\chapter{التوتر ومقياس التوتر}\label{ch:g8:voltage-voltmeter}

منذ أضاءت أول \omterm{def:g3:first-electric-circuit:battery}{بطارية} أول \omterm{def:g3:first-electric-circuit:bulb}{مصباح}، حملت كلمة واحدة في هذا الكتاب
عبئًا ثقيلًا غامضًا: ``الدفعة''. فالبطاريات تدفع دفعًا ضعيفًا،
والشبكة تدفع دفعًا قاتلًا، وبطاريتان رأسًا بعقب تدفعان معًا.
والدفعة تصير الآن مقدارًا مقيسًا \omterm{def:g6:measurement-in-science:measuring}{بوحدة} مشهورة --- وآلتها،
بخلاف آلة الفصل الماضي، تأبى أن تقف في الطريق.

\section{التوتر}

\begin{definition}[التوتر]\label{def:g8:voltage-voltmeter:voltage}
\emph{التوتر}\index{توتر} $U$ بين نقطتين من دارة يقيس الدفعة
الكهربائية المتاحة بينهما --- أي كم يُساق التيار بقوة من نقطة
إلى الأخرى، كما يقيس ارتفاع الشلّال قوّة سَوقه لدولاب الطاحونة.
والتوتر \emph{بين} دائمًا: بين مأخذَي \omterm{def:g3:first-electric-circuit:battery}{بطارية}، وبين جانبَي
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} --- ولا يكون عند نقطة واحدة وحدها أبدًا. ووحدته
\emph{الفولت}\index{فولت} (\unit{V})، تكريمًا لمخترع أول
\omterm{def:g3:first-electric-circuit:battery}{بطارية}.
\end{definition}

\begin{example}[ألبوم عائلة الفولت]\label{ex:g8:voltage-voltmeter:values}
خلية مستديرة واحدة: \qty{1.5}{V}. \omterm{def:g3:first-electric-circuit:battery}{والبطارية} المسطّحة الجيبية:
\qty{4.5}{V} --- ثلاث خلايا رأسًا بعقب داخل غلاف واحد.
\omterm{def:g3:first-electric-circuit:battery}{والبطارية} المستطيلة ذات المأخذين في أعلاها: \qty{9}{V}، وفيها
ست خلايا مخفيّة. \omterm{def:g3:first-electric-circuit:battery}{وبطارية} سيارة: \qty{12}{V}. ومأخذ الجدار:
\qty{230}{V} --- أي دفعة أكثر من مئة وخمسين خلية صغيرة، وهو
المضمون الكمّي كله لعبارة ``المأخذ، أبدًا''. وأمّا البرق بين
السحابة والأرض: فمئات الملايين من الفولتات، لحظة قصيرة.
\end{example}

\begin{proposition}[توتر بلا تيار]\label{prop:g8:voltage-voltmeter:nocurrent}
\omterm{def:g8:voltage-voltmeter:voltage}{التوتر} والتيار مقداران مختلفان، \omterm{def:g8:voltage-voltmeter:voltage}{والتوتر} أسبق. \omterm{def:g3:first-electric-circuit:battery}{فبطارية} وحدها في
درج، غير موصولة بشيء، لا تسوق تيارًا البتّة --- ومع ذلك يقف بين
مأخذيها \qty{1.5}{V} كاملة مستعدّة، قابلة \omterm{def:g6:measurement-in-science:measuring}{للقياس}، كارتفاع
شلّال في يوم تعطّلت فيه الطاحونة. فالدفعة قد توجد بلا مسير؛
وأمّا المسير فلا يوجد بلا دفعة أبدًا.
\end{proposition}

\begin{definition}[مقياس التوتر]\label{def:g8:voltage-voltmeter:voltmeter}
\emph{مقياس التوتر}\index{مقياس التوتر} (الفولطمتر) يقيس
\omterm{def:g8:voltage-voltmeter:voltage}{التوتر} بين نقطتين --- فيوصل \emph{على طرفيهما}: أي \emph{\omterm{def:g5:series-parallel-circuits:parallel}{على
التفرع}} مع العنصر المقصود، سلك على كل جانب، كمسّاح يمسك ميزان
تسوية بين ارتفاعين. وهو مبنيّ على نقيض \omterm{def:g8:intensity-ammeter:ammeter}{مقياس التيار} في كل شيء،
فيقاوم التيار مقاومة \emph{شبه تامّة}: فإذا وُصل على طرفَي أيّ
شيء لم يسحب منه ما يُذكر --- فتعليقه لا يزعج شيئًا، ولا يحتاج
الأمر إلى قطع أيّ حلقة أبدًا.
\end{definition}

\begin{method}[قياس توتر]\label{met:g8:voltage-voltmeter:measure}
\begin{enumerate}
  \item دع الدارة سليمة --- فلا يُقطع شيء من أجل مقياس \omterm{def:g8:voltage-voltmeter:voltage}{توتر}؛
  \item علّق سلكًا على كل جانب من العنصر: سلك \unit{V} حيث يدخل
        التيار الاصطلاحي، والسلك المشترك حيث يخرج؛
  \item المجال الأكبر أولًا، ثم انزل حتى الراحة؛
  \item اقرأ، واكتب \omterm{def:g6:measurement-in-science:measuring}{الوحدة}.
\end{enumerate}
وآداب المقياسين جنبًا إلى جنب: \omterm{def:g8:intensity-ammeter:ammeter}{مقياس التيار} --- اقطع الطريق،
وقف فيه، ولا تقاوم شيئًا؛ \omterm{def:g8:voltage-voltmeter:voltmeter}{ومقياس التوتر} --- لا تقطع شيئًا، وقف
جانبًا، وقاوم كل شيء. والخلط بين الأدبين أغلى عادة في الورشة.
\end{method}

\section{التوترات الاسمية}

\begin{definition}[التوتر الاسمي]\label{def:g8:voltage-voltmeter:nominal}
كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح} ومحرّك وجهاز يحمل \emph{توترًا اسميًّا}\index{توتر
اسمي} --- أي الدفعة التي بُني لها، مطبوعة على قاعدته أو
لوحته: \omterm{def:g3:first-electric-circuit:bulb}{مصباح} كشّاف يدوي ``3.5 V''، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} مصدح سيارة
``12 V''، وغلّاية ``230 V''. فإذا غُذّي الجهاز بتوتره الاسمي
أدّى كما صُمّم؛ وإذا غُذّي بأقلّ منه بكثير تجهّم --- \omterm{def:g3:first-electric-circuit:bulb}{مصباح}
خافت، ومحرّك كسول؛ وإذا غُذّي بأكثر منه بكثير احترق --- ببهاء،
مرة واحدة.
\end{definition}

\begin{example}[مطابقة المصباح للبطارية]\label{ex:g8:voltage-voltmeter:matching}
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} ``3.5 V'' على \omterm{def:g3:first-electric-circuit:battery}{البطارية} المسطّحة \qty{4.5}{V}: مُجهَد
قليلًا --- ساطع، وقصير العمر. وعلى خلية \qty{1.5}{V} واحدة:
توهّج برتقالي نعسان. وعلى \omterm{def:g3:first-electric-circuit:battery}{بطارية} \qty{9}{V}: لحظة خاطفة واحدة،
ثم ينفصم الخيط الرفيع --- نعي الخيط. فاللافتة عقد: أوفِ به
يعِش \omterm{def:g3:first-electric-circuit:bulb}{المصباح} عمره المعلن.
\end{example}

\begin{remark}[لماذا تقتل 230 V]\label{rem:g8:voltage-voltmeter:mains}
الآن تنال القاعدة المطلقة القديمة عددها. فطريق الجسم الرطب
\omterm{def:g4:conductors-insulators:conductor}{الناقل}، إذا عُرضت عليه \qty{4.5}{V}، مرّر تيارًا أصغر من أن
يُحسّ. والطريق نفسه عند \qty{230}{V} --- أي خمسين ضعف الدفعة
--- يمرّر خمسين ضعف التيار، أي أبعد بكثير من أجزاء المئة من
\omterm{def:g8:intensity-ammeter:intensity}{الأمبير} التي توقف قلبًا. فالمأخذ ليس أقوى ``بعض الشيء'' من
منضدة بطارياتك: بل هو أبعد منها برتبتَي قدر، وصرامة كتاب
القواعد حساب صريح.
\end{remark}

\section{أول قراءات التوتر}

\begin{example}[حول حلقة بسيطة]\label{ex:g8:voltage-voltmeter:readings}
\omterm{def:g3:first-electric-circuit:battery}{بطارية} واحدة، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} واحد، والمقاييس في اليد. على طرفَي
\omterm{def:g3:first-electric-circuit:battery}{البطارية}: \qty{4.5}{V}. وعلى طرفَي \omterm{def:g3:first-electric-circuit:bulb}{المصباح}: \qty{4.5}{V} ---
\omterm{def:g3:first-electric-circuit:bulb}{فالمصباح} ينال ما تعرضه \omterm{def:g3:first-electric-circuit:battery}{البطارية}. وعلى طرفَي قطعة من
\emph{السلك} الواصل: \qty{0}{V}، أو ما يقارب ذلك بقدر ما يقول
المقياس --- فالسلك الجيّد طريق مستوٍ لا ينفق شيئًا من الدفعة.
فالدفعة تسكن على طرفَي \emph{العمّال}، لا على امتداد الطرق:
قراءة تستحقّ التذكّر حين تجعلها قوانين الفصل التالي رسمية.
\end{example}

\begin{example}[اعتراف القاطعة المفتوحة]\label{ex:g8:voltage-voltmeter:openswitch}
\omterm{def:g3:first-electric-circuit:bulb}{المصباح} مظلم، \omterm{ex:g3:first-electric-circuit:switch}{والقاطعة} مفتوحة --- فأين ذهبت الدفعة؟ قِس: على
طرفَي \omterm{def:g3:first-electric-circuit:bulb}{المصباح} المظلم \qty{0}{V}؛ وعلى طرفَي \emph{\omterm{ex:g3:first-electric-circuit:switch}{القاطعة}
المفتوحة} \qty{4.5}{V} كاملة. فدفعة \omterm{def:g3:first-electric-circuit:battery}{البطارية} تقف بتمامها على
طرفَي الفجوة، منتظرة --- وهي الموضع الوحيد في الحلقة الذي
تكون فيه نقطتان ``متباعدتين'' حقًّا من الوجهة الكهربائية.
أغلق \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} تتبادل القراءتان مكانيهما. \omterm{ex:g3:first-electric-circuit:switch}{فالقاطعة} المفتوحة هي
حيث ينتظر \omterm{def:g8:voltage-voltmeter:voltage}{التوتر}.
\end{example}

\section{مسألة: تدقيق درج البطاريات}

\begin{problem}[{مسألة نهاية الأسبوع --- تدقيق درج البطاريات
الكبير؛ مقياس توتر، وعلبة أحذية من الخلايا، ورفّ أجهزة
العائلة}]\label{pb:g8:voltage-voltmeter:1}
في كل بيت ذلك الدرج: خلايا سائبة، وبطاريات مجهولة، وأجهزة
``تحتاج إلى جديدة''. وقد حلّ يوم التدقيق --- \omterm{def:g8:voltage-voltmeter:voltmeter}{ومقياس التوتر} في
اليد.

\medskip\noindent\textbf{القسم الأول --- فرز الخلايا.}
\begin{enumerate}
  \item صِف الوصل الصحيح \omterm{def:g8:voltage-voltmeter:voltmeter}{لمقياس التوتر} لفحص خلية سائبة
        \qty{1.5}{V} --- ولماذا لا يلزم بناء دارة أولًا.
  \item قراءات التدقيق، توًّا من الدرج: \qty{1.58}{V}،
        و\qty{1.32}{V}، و\qty{0.9}{V}، و\qty{0.1}{V}. افرزها:
        نضِرة، وصالحة، ومتعبة، وميّتة.
  \item \omterm{def:g3:first-electric-circuit:battery}{بطارية} تسعة \omterm{def:g8:voltage-voltmeter:voltage}{فولت} تقرأ \qty{8.9}{V}. فكم تسهم كل خلية
        من خلاياها الستّ الداخلية تقريبًا؟
  \item ولماذا تُظهر خلية شبه ميّتة \emph{بعض} \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} مع أنها
        لم تعد تضيء شيئًا؟ (فدفعتها تنجو واهنة؛ فما الذي فقدت
        \omterm{def:g2:pushes-and-pulls:force}{القوة} على سَوقه؟)
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- رفّ الأجهزة.}
\begin{enumerate}[resume]
  \item آلة التصوير تريد خليتين رأسًا بعقب. أيّ \omterm{def:g8:voltage-voltmeter:voltage}{توتر} تتوقّع،
        وماذا سيسلّم زوج الناجيتين \qty{1.32}{V} من السؤال
        الثاني؟
  \item المذياع القديم يأخذ أربع خلايا رأسًا بعقب ويتجهّم دون
        \qty{5}{V}. أيّ خلايا الدرج، إن وُجدت، تصلح طاقمًا
        له؟
  \item يحشر الأخ الصغير ثلاث خلايا في كشّاف الخليتين ``لمزيد
        من السطوع''، مقحمًا الثالثة على النابض. \omterm{def:g3:first-electric-circuit:bulb}{والمصباح} مقنَّن
        على \qty{3}{V}. تنبّأ بالعرض، بلغة \omterm{def:g8:voltage-voltmeter:nominal}{التوتر الاسمي}.
  \item لوحة محوّل جرس الباب تقرأ ``\qty{8}{V}''. ويقترح أحدهم
        وصل الجرس مباشرة \omterm{def:g3:first-electric-circuit:battery}{ببطارية} تسعة \omterm{def:g8:voltage-voltmeter:voltage}{فولت} ``ما دامت الثمانية
        تكاد تكون تسعة''. قيّم: أمطابقة عادلة أم قتل بطيء؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- قوانين التدقيق.}
\begin{enumerate}[resume]
  \item في الكشّاف (\omterm{def:g3:first-electric-circuit:bulb}{المصباح} والخلايا \omterm{def:g4:series-circuits:series}{على التسلسل} \omterm{ex:g3:first-electric-circuit:switch}{والقاطعة}
        مفتوحة)، يقيس المدقّق: على طرفَي \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} المفتوحة
        \qty{3.1}{V}؛ وعلى طرفَي \omterm{def:g3:first-electric-circuit:bulb}{المصباح} \qty{0}{V}. اشرح
        القراءتين، وتنبّأ بهما بعد النقرة.
  \item وعلى طرفَي كل قطعة من أسلاك الكشّاف يقول المقياس
        \qty{0}{V}، مضيئًا كان أم مظلمًا. فعلامَ يدلّ هذا في
        شأن السلك الجيّد، بلغة الدفعة؟
  \item ولا يُدقَّق مأخذ الشبكة، بقانون عائلي دائم. أعِد صوغ
        القانون بعدد هذا الفصل، وبجملة الحساب التي وراءه.
  \item أودِع تقرير التدقيق: ثلاث جمل --- ما \omterm{def:g8:voltage-voltmeter:voltmeter}{مقياس التوتر} وما
        لا يكون أبدًا؛ وبمَ يعد \omterm{def:g8:voltage-voltmeter:nominal}{التوتر الاسمي} على اللافتة
        وبمَ يهدّد؛ وأين ينتظر \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} في أيّ دارة عاطلة.
\end{enumerate}
\end{problem}
