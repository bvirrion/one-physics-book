\chapter{الصوت: السرعة والحدّة والعلوّ}\label{ch:g8:sound-pitch-loudness}

الأوتار القصيرة تغنّي حادّة، والنقرات القوية تغنّي عالية ---
وقد وجدت قيثارتك ذات الشريط المطّاطي القواعد منذ سنوات. وفي
هذه السنة تنال القواعد أعدادًا: عدّة في الثانية للحدّة، وحجمًا
للتأرجح للعلوّ، ونافذة --- بحوافّ --- على ما تستطيع الآذان
البشرية التقاطه أصلًا. وفيما وراء إحدى الحافّتين، تضحك
الخفافيش علينا.

\section{التردّد: الحدّة معدودةً}

\begin{definition}[التردّد]\label{def:g8:sound-pitch-loudness:frequency}
\emph{تردّد}\index{تردّد} \omterm{def:g3:sound-around-us:vibration}{اهتزاز} $f$ هو عدد التأرجحات الكاملة
ذهابًا وإيابًا التي يؤدّيها في كل ثانية. ووحدته
\emph{الهرتز}\index{هرتز} (\unit{Hz}): تأرجحة واحدة في
الثانية --- تكريمًا للفيزيائي الذي صنع الأمواج غير المرئية
والتقطها أول مرة. فوتر قيثارة يرتعش $110$ مرة في الثانية يهتزّ
بتردّد \qty{110}{Hz}؛ وجناحا بعوضة، قرب \qty{600}{Hz}؛
وخفقان علم كبير البطيء، بضعة هرتزات.
\end{definition}

\begin{proposition}[الحدّة هي التردّد]\label{prop:g8:sound-pitch-loudness:pitch}
ما تسمّيه الأذن \emph{حدّة} هو \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد} ولا شيء غيره: فكلما
ارتعش المنبع أسرع، ارتفعت النغمة. وقواعد الشريط المطّاطي
تُترجَم في الحال --- فالأشياء القصيرة المشدودة الرفيعة ترتعش
\omterm{def:g8:sound-pitch-loudness:frequency}{بتردّد} $f$ أعلى. ومرساة الموسيقى: آلات الحفلات في العالم كله
تُدوزَن على النغمة ``لا'' عند \qty{440}{Hz}. ومضاعفة \omterm{def:g8:sound-pitch-loudness:frequency}{تردّد} أيّ
نغمة ترفعها بأشدّ الأبعاد الموسيقية توافقًا --- الأوكتاف:
فنغمة ``لا'' ثانية، أوكتافًا أعلى، عند \qty{880}{Hz}.
\end{proposition}

\begin{method}[رؤية الصوت]\label{met:g8:sound-pitch-loudness:seeing}
طبل حبّات الأرزّ أظهر ارتعاش \omterm{def:g3:sound-around-us:sound}{الصوت}؛ والمِذياع يرسمه.
\begin{enumerate}
  \item شغّل تطبيق تحليل \omterm{def:g3:sound-around-us:sound}{صوت} على هاتف أو حاسوب --- فشاشته
        تمثّل تأرجح المِذياع في مقابل الزمن، وهي راسم ذبذبات
        حديث؛
  \item دندن منخفضًا ثابتًا: أثر كسول متباعد؛ ثم غنِّ عاليًا:
        تتزاحم القمم؛
  \item اقرأ قراءة \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد} وأنت تصفر: فمعظم الصفير يقع بين
        $500$ و\qty{2000}{Hz}؛
  \item اضرب الشوكة الرنّانة المعلّمة ``لا 440'' وتحقّق أن
        القراءة تفي بنقشها.
\end{enumerate}
شاشة واحدة تحلّ محلّ فصل من التخمين: فالحدّة معدودة أمام
عينيك.
\end{method}

\section{نافذة الأذن}

\begin{proposition}[مدى السمع البشري]\label{prop:g8:sound-pitch-loudness:range}
الآذان الفتيّة السليمة تلتقط الاهتزازات بين نحو \qty{20}{Hz}
و\qty{20000}{Hz}. ودون النافذة تقع \emph{الأصوات دون
السمعية}\index{صوت دون سمعي} --- تُحسّ أحيانًا هديرًا في
الصدر، ولا تُسمع أبدًا؛ وفوقها \emph{الأصوات فوق
السمعية}\index{صوت فوق سمعي} --- صمت لنا، مهما عنف الارتعاش.
والنافذة تضيق بالعمر وبسوء المعاملة: فالأوكتاف الأعلى عادةً
أول ضحايا العادات الصاخبة.
\end{proposition}

\begin{example}[فيما وراء الحافّتين]\label{ex:g8:sound-pitch-loudness:beyond}
الطبيعة والصناعة كلتاهما تعملان فيما وراء نافذتنا. فالخفافيش
تصطاد بنقرات فوق سمعية قرب \qty{50000}{Hz}، فتقرأ أصداءً أدقّ
مما يستطيع أيّ \omterm{def:g3:sound-around-us:sound}{صوت} مسموع أن يرسمه --- والدلافين كذلك، على
روح فصل السونار. والكلاب تجيب صفّارات ``صامتة'' حدّتها فوق
حافّتنا بقليل. والماسحات الطبّية تصوّر الأجنّة بأصداء فوق
سمعية غير ضارّة. وعند الحافّة الأخرى، تتحادث الفيلة بهدير دون
سمعي عبر الكيلومترات، وتعلن العواصف والزلازل عن نفسها بنغمات
لا تصغي إليها أذن.
\end{example}

\section{العلوّ: حجم التأرجحة}

\begin{proposition}[العلوّ هو السعة]\label{prop:g8:sound-pitch-loudness:loudness}
ما تسمّيه الأذن \emph{علوًّا} يقابل \emph{سعة}\index{سعة}
\omterm{def:g3:sound-around-us:vibration}{الاهتزاز} --- حجم التأرجحة، \omterm{def:g2:pushes-and-pulls:force}{وقوة} التضاغط المسافر. انقر برفق أو
بضراوة: تقف قمم الأثر منخفضة أو عالية بينما يبقى
\emph{تباعدها} --- أي الحدّة --- بلا مساس. \omterm{prop:g8:sound-pitch-loudness:loudness}{فالسعة} \omterm{def:g8:sound-pitch-loudness:frequency}{والتردّد}
قرصا ضبط مستقلّان: فأيّ حدّة قد تُهمَس أو تُزأر.
\end{proposition}

\begin{example}[سلّم الديسيبل]\label{ex:g8:sound-pitch-loudness:decibels}
يُدرَّج العلوّ \emph{بالديسيبل}\index{ديسيبل} (\unit{dB}) ---
وهو سلّم مبنيّ لمدى الأذن المذهل، يضيف عليه كل تقوية \omterm{def:g3:sound-around-us:sound}{للصوت}
بمقدار عشرة أضعاف العددَ نفسه من الدرجات. ومعالمه: حفيف
الأوراق، \qty{10}{dB}؛ ومحادثة هادئة، \qty{50}{dB}؛ وحركة
مرور مزدحمة، \qty{80}{dB}؛ والصفوف الأمامية في حفلة صخب،
\qty{110}{dB}؛ والألم والأذى الفوري، قرب \qty{120}{dB}. وحساب
السلّم الغريب --- خطوات متساوية لقوى عشرة أضعاف --- لوغاريتم،
وهو من عظيم الاختراعات التي تنتظرك في رياضيات الثانوية.
\end{example}

\begin{remark}[ميزانية الأذن]\label{rem:g8:sound-pitch-loudness:earsafety}
الطبل الصغير في أذنك، الذي لقيته في فصل سنتك الثالثة، يمسك
حسابات العمر كله. ففوق نحو \qty{85}{dB} يتراكم الضرر مع \omterm{def:g2:measuring-time:duration}{مدة}
التعرّض --- حدّ يوم عمل في مصنع صاخب، ودقائق لا غير عند
مستويات مقدّمة الحفلات؛ وسمّاعات الرأس على أقصى \omterm{def:g3:sound-around-us:sound}{الصوت} تجلس في
نطاق الخطر تمامًا. والسمع الضائع في التأرجحات العالية لا يعود
أبدًا: فالنغمات الأعلى تغادر أولًا، بهدوء، وإلى الأبد. فليس
للآذان أجفان --- ولا قطع غيار.
\end{remark}

\begin{example}[السرعة، لا يمسّها أيّ من القرصين]\label{ex:g8:sound-pitch-loudness:speed}
لا يمسّ أيّ من القرصين التسليم. فالحادّ والغليظ، والعالي
والخافت، كلها تركب تناقل \omterm{def:g7:sound-production:medium}{الوسط} الواحد بوتيرة \omterm{def:g7:sound-production:medium}{الوسط} الواحدة ---
\qty{340}{m/s} في \omterm{def:g2:air-around-us:air}{الهواء}. والبرهان يُؤدّى كل ليلة: فأوركسترا
تُسمع عبر حديقة تصل \emph{منسجمة وفي وقتها} --- ولو كانت
النغمات الحادّة أسرع من الغليظة لتلطّخ كل وتر بعيد إلى تعاقب
نغمات. (والعلوّ يخبو مع البعد --- قشور فصل الصف السابع
المنتشرة --- لكن الخبوّ إضعاف، لا إبطاء أبدًا.)
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:1}
عرّف \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد} ووحدته. وما حدّة نغمة ``لا'' في الحفلات، وما
حدّة ``لا'' التي فوقها بأوكتاف؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:2}
أثران يمتدّان على العشرين ملّي ثانية نفسها: أحدهما يُظهر $4$
تأرجحات كاملة، والآخر $16$. احسب التردّدين؛ وأيّهما أحدّ
صوتًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:3}
أعطِ نافذة السمع البشري، واسمَي البلدين اللذين وراء حدودها ---
مع ساكن واحد من كل بلد.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:4}
على شاشة المحلّل، ما الذي يتغيّر حين تغنّي النغمة نفسها
بعلوّ أكبر؟ وحين تغنّي نغمة أحدّ بالعلوّ نفسه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:5}
ترجِم قواعد الشريط المطّاطي القديمة إلى لغة \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد}: ماذا يفعل
كلٌّ من القِصَر والشدّ والرقّة \omterm{def:g8:sound-pitch-loudness:frequency}{بالتردّد} $f$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:6}
ضع على سلّم الديسيبل: حفيف الأوراق؛ وحركة المرور؛ والصفّ
الأمامي في الحفلة؛ والألم. وأين يبدأ نطاق الخطر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:7}
لماذا تبدو صفّارات الكلاب معطّلة لأصحابها؟ ولماذا يخالفهم
الكلب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:8}
أوركسترا بعيدة تصل منسجمة وفي وقتها. فعلامَ يبرهن هذا في شأن
\omterm{def:g7:motion-average-speed:formula}{سرعة} \omterm{def:g3:sound-around-us:sound}{الصوت} عبر الحدّة والعلوّ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:9}
خفّاش ينقر عند \qty{50000}{Hz}. كم تأرجحة كاملة تحوي نقرة
واحدة مدّتها \qty{0.01}{s}؟ ولماذا يجب أن يستعمل التصوير
الدقيق \omterm{ex:g4:how-sound-travels:echo}{بالصدى} ارتعاشًا بهذه \omterm{def:g7:motion-average-speed:formula}{السرعة}؟ (تذكّر ما تشترك فيه أدقّ
الفراشي.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:10}
أخفض مفاتيح البيانو الكبير يهتزّ قرب \qty{27}{Hz}؛ وأعلاها
قرب \qty{4200}{Hz}. حدّد موضع المفتاحين داخل نافذة السمع (أو
في مواجهتها) --- واشرح لماذا قد يحتاج مدوزن بيانو مسنّ إلى
عون المحلّل عند أحد طرفَي لوحة المفاتيح أولًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:11}
قواعد الحفلات تسمح بثماني ساعات عند \qty{85}{dB} لكن بدقائق
لا غير قرب \qty{110}{dB}، فتنصّف الميزانية كل بضع درجات. أيّ
نوع من النموّ يختبئ في خطوات السلّم المتساوية --- وماذا يعلّم
انهيار الميزانية في شأن ``أعلى قليلًا لا غير''؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:sound-pitch-loudness:12}
صمّم فحص \omterm{def:g3:sound-around-us:sound}{الصوت} الكامل لقاعة مدرسة بهاتف محلّل واحد: كيف
تتحقّق من نغمة ``لا'' في البيانو في مقابل $440$؛ وكيف تجد
علوّ القاعة عند الصفّ الخلفي أثناء اختبار الطبول؛ وكيف تمسك
ما إذا كانت التهوية تصدر هديرًا دون سمعي تخطئه الأذن؛ وأيّ
النتائج الثلاث \emph{أقلّها} تجهيزًا لدى مِذياع الهاتف
للتصديق عليه (راجع نافذة سمع الهاتف نفسه --- المبنيّة
للأصوات البشرية).
\end{exercise}

\section{مسألة: فحص الصوت}

\begin{problem}[{مسألة نهاية الأسبوع --- فحص الصوت في مهرجان
المدرسة؛ محلّل واحد، وثلاث فرق، وممرّضة قلِقة؛ أمسية منضدة
المزج}]\label{pb:g8:sound-pitch-loudness:1}
عشيّة المهرجان: تدير المحلّل عند منضدة المزج، وممرّضة المدرسة
تراقب مقياس الديسيبل من فوق كتفك.

\medskip\noindent\textbf{القسم الأول --- ساعة الدوزنة.}
\begin{enumerate}
  \item شوكة الجوقة الرنّانة تغنّي عند \qty{440}{Hz}؛ ويُظهر
        المحلّل نغمة ``لا'' في البيانو عند \qty{432}{Hz}.
        شخّص البيانو، وسمِّ القرص --- الحدّة أم العلوّ ---
        الذي يحتاج إلى المدوزن.
  \item أخفض أوتار غيتار الجهير يقرأ \qty{41}{Hz}. حدّد
        موضعه في نافذة السمع --- واشرح لماذا
        \emph{سيحسّه} الجمهور في الصدر بقدر ما يسمعه.
  \item نغمة المغنّي الرئيسي الممدودة تُظهر قممًا تباعدها
        كتباعد قمم الشوكة تمامًا، لكنها أعلى ثلاث مرات.
        قارن الصوتين في القرصين معًا.
  \item ويزعم فنّي شابّ أن مكبّرات القاعة الجديدة ``تعيد
        الإنتاج حتى \qty{22000}{Hz}''. فمن في الجمهور يصدّق
        على قمّة ذلك المدى --- ومن لا يستطيع؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- دفتر الممرّضة.}
\begin{enumerate}[resume]
  \item مستويات البروفة: الوصلة الصوتية \qty{75}{dB}؛ ووصلة
        الصخب \qty{100}{dB}؛ والختام المخطَّط ``أعلى بلمسة''
        عند \qty{110}{dB}. أيّ الوصلات تؤشّر عليها
        الممرّضة، وفي مواجهة أيّ بند من الميزانية؟
  \item وتوزّع الممرّضة سدادات أذن إسفنجية مقنَّنة لتخفيف
        الأصوات \qty{20}{dB}. فماذا تصير وصلة الصخب خلفها
        --- ولماذا يلبس الفنّيون الحكماء السدادات في كل
        عرض؟
  \item وبالوقوف على ضعف البعد من الخشبة يهبط المقياس هبوطًا
        ملحوظًا. أيّ قانون قديم من قوانين القشور المنتشرة
        يعمل هنا --- وأيّ قرص (الحدّة أم العلوّ) يديره
        البعد؟
  \item ويحتجّ عازف الطبول: ``الأخفت أبلد --- فالنغمات
        الحادّة تموت أولًا حين نعزف بهدوء!'' فكّ له القرصين:
        ماذا يغيّر العزف الخافت فعلًا؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- فيزياء المهرجان.}
\begin{enumerate}[resume]
  \item تختم الألعاب النارية العرض على بعد \qty{680}{m}.
        احسب التأخّر بين الوميض والدويّ الذي سيلحظه الحشد
        --- \omterm{def:g8:sound-pitch-loudness:frequency}{وتردّد} الهدير إذا دقّت القذائف الكبيرة مرتين في
        الثانية (تحقّق من النافذة: أيُسمع أم يُحسّ؟).
  \item ولغز آخر الليل: مقطع فيديو من الحفلة على هاتف يبدو،
        عند تشغيله، نحيلًا --- وقد ذهب هدير الصدر. راجع
        تحفّظ التمرين الثاني عشر: ما الذي لم يُبنَ المِذياع
        قطّ لالتقاطه؟
  \item ويطلب تقرير الممرّضة ``جملة واحدة عن كل'' من قرصَي
        الأمسية والشيء الثالث الذي لا يمسّه أيّ قرص. صُغها.
  \item وعند التفكيك: طارد الآفات فوق السمعي عند حارس
        المبنى، وقد ظلّ يعمل طوال الأمسية بجانب باب الخشبة،
        لم يزعج مستمعًا واحدًا، وأزعج --- كما يُزعم ---
        فئرانًا كثيرة. اشرح الجمهورين معًا بقضية نافذة
        واحدة.
\end{enumerate}
