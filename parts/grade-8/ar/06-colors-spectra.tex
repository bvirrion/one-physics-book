\chapter{ألوان الضوء وأطيافه}\label{ch:g8:colors-spectra}

أين تسكن الألوان؟ في التفاحة، يقول الحسّ المشترك --- فالحمرة
هناك على القشرة تمامًا. لكن احمل التفاحة إلى صالة رياضية مضاءة
بأزرق عميق تصر التفاحة ``الحمراء'' سوداء كالفحم. فالألوان،
كما يتبيّن، محادثة ثلاثية بين \omterm{def:g1:light-and-shadows:source}{الضوء} والجسم والعين --- وتفتتح
بأغرب حقيقة في البصريات: \omterm{def:g1:light-and-shadows:source}{الضوء} الأبيض ليس أبيض.

\section{الضوء الأبيض مفكوكًا}

\begin{proposition}[الضوء الأبيض مزيج]\label{prop:g8:colors-spectra:mixture}
\omterm{def:g1:light-and-shadows:source}{الضوء} ``الأبيض'' العادي --- \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس \omterm{def:g1:light-and-shadows:source}{وضوء} \omterm{def:g3:first-electric-circuit:bulb}{المصباح} --- مزيج من
الألوان كلها دفعة واحدة. فإذا أُرسل عبر موشور زجاجي انفكّ
المزيج: إذ ينكسر كل لون بمقدار مختلف قليلًا وهو يعبر الزجاج،
فتتباعد الألوان مروحة في شريط متّصل من الأحمر (أقلّها انكسارًا)
إلى البنفسجي (أكثرها انكسارًا). ويسمّى الشريط
\emph{طيفًا}\index{طيف}، ويسمّى الفكّ \emph{تبديدًا}\index{تبديد}.
\end{proposition}

\begin{example}[أطياف بلا مختبر]\label{ex:g8:colors-spectra:everyday}
الطبيعة ودرج الخردوات مملوءان مواشير. فقوس قزح: كل قطرة مطر
موشور \omterm{def:g5:mirrors-reflection:reflection}{ومرآة} صغيران، والقوس بسعة السماء \omterm{def:g1:light-and-shadows:source}{ضوء} شمس بدّدته ملايين
القطرات --- وهو دائمًا في نصف السماء \emph{المقابل} للشمس،
ولهذا يقف صائدو أقواس قزح والشمس خلف ظهورهم. ورذاذ خرطوم
الحديقة الناعم يستحضر واحدًا عند الطلب. ووجه القرص المدمج
المحزّز يقذف أطيافًا صغيرة على السقف؛ وحافة \omterm{def:g5:mirrors-reflection:reflection}{المرآة} المشطوفة
تخطّط الجدار في أصائل مشمسة. \omterm{def:g1:light-and-shadows:source}{فالضوء} الأبيض يفضح مزيجه في كل
موضع يلقى فيه بنية دقيقة أو زجاجًا مائلًا.
\end{example}

\begin{example}[إعادة تعبئة المزيج]\label{ex:g8:colors-spectra:newtondisc}
الفكّ يجري بالمقلوب أيضًا. لوّن قرصًا بقطاعات قوس قزح وأدره
سريعًا: فالعين، وهي أبطأ من أن تتابع، تستقبل الألوان كلها من كل
بقعة في وقت واحد عمليًّا --- فتقرأ المزيج: بياضًا رماديًّا شاحبًا
مغسولًا. وقد أدار الفيزيائي العظيم الذي فكّ \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس بموشور
أولَ مرة قرصًا كهذا بعينه ليحتجّ للتعبئة --- وكانت أجرأ ضرباته
موشورًا ثانيًا يلتقط الطيف ويطويه ثانية في \omterm{def:g6:light-propagation:ray}{حزمة} بيضاء واحدة:
فالمزيج مبرهَن من الجهتين.
\end{example}

\section{لماذا التفاحة حمراء}

\begin{proposition}[لون الجسم]\label{prop:g8:colors-spectra:objectcolor}
لون الجسم \omterm{def:g7:light-sources-propagation:coats}{المعتم} مصنوع بالطرح. فالجسم، إذ يستقبل \omterm{def:g1:light-and-shadows:source}{الضوء}،
\emph{يمتصّ} جزءًا من المزيج ويبعثر الباقي --- والباقي المبعثَر
هو ما يبلغ عينك ويسمّي اللون. فالتفاحة الحمراء تمتصّ جميع ألوان
\omterm{def:g1:light-and-shadows:source}{الضوء} الأبيض تقريبًا وتعيد الأحمر أساسًا. والصفحة البيضاء تعيد
كل شيء تقريبًا؛ والمعطف الأسود لا يعيد شيئًا تقريبًا --- بل
يمتصّ الجميع (ويسخن تبعًا لذلك، كما تشهد السيارات الداكنة في
الصيف). فاللون ليس \emph{في} القشرة: بل هو جواب القشرة على
\omterm{def:g1:light-and-shadows:source}{الضوء} الذي عُرض عليها.
\end{proposition}

\begin{example}[اختبار الصالة الرياضية]\label{ex:g8:colors-spectra:gym}
والآن إلى الصالة الرياضية المضاءة بالأزرق. قشرة التفاحة
الحمراء تفعل ما تفعله دائمًا: تمتصّ كل شيء إلا الأحمر، وتقف
مستعدّة لإعادة الأحمر --- لكن الكشّاف الأزرق \emph{لا يحوي}
أحمر لتعيده. فلا شيء يُبعثر ولا شيء يُستقبل: فتبدو التفاحة
سوداء. أما الصفحة البيضاء تحت \omterm{def:g1:light-and-shadows:source}{الضوء} نفسه فتعيد ما أُعطيت:
فتبدو زرقاء. وقاعدة المسرح: لا يستطيع جسم أن يعيد إلا ألوانًا
جاء بها \omterm{def:g1:light-and-shadows:source}{الضوء} --- فغيّر \omterm{def:g1:light-and-shadows:source}{الضوء} تغيّر كل زيّ في القاعة.
\end{example}

\begin{method}[التنبّؤ بلون تحت أيّ ضوء]\label{met:g8:colors-spectra:predict}
\begin{enumerate}
  \item اسرد ما يجيء به \omterm{def:g1:light-and-shadows:source}{الضوء} (الأبيض: كل شيء؛ \omterm{def:g3:first-electric-circuit:bulb}{والمصباح}
        الملوّن: لونه هو أساسًا)؛
  \item اسرد ما يعيده الجسم في \omterm{def:g1:light-and-shadows:source}{الضوء} الأبيض --- فلونه في \omterm{def:g1:light-and-shadows:source}{ضوء}
        النهار يسمّيه؛
  \item قاطِع بين السردين: فالجسم يبعثر ما يظهر في كليهما؛
  \item وإذا كان التقاطع خاليًا: بدا الجسم أسود.
\end{enumerate}
قبّعة زرقاء تحت \omterm{def:g1:light-and-shadows:source}{ضوء} أحمر: تعيد الأزرق لا غير، وعُرض عليها
الأحمر لا غير --- فالتقاطع خالٍ: قبّعة سوداء. وتحت \omterm{def:g1:light-and-shadows:source}{ضوء} أبيض:
يلقى العرض الكامل الجوابَ الأزرق --- فقبّعة زرقاء.
\end{method}

\section{بناء الألوان من الضوء}

\begin{example}[ثلاثة مصابيح تصنعها كلها]\label{ex:g8:colors-spectra:additive}
مزج \emph{الأضواء} جمع. راكِب ثلاثة كشّافات --- أحمر وأخضر
وأزرق: فالأحمر والأخضر معًا يصنعان الأصفر؛ والأخضر والأزرق
يصنعان سماويًّا فيروزيًّا؛ والأزرق والأحمر يصنعان الأرجواني؛
والثلاثة جميعًا تصنع الأبيض. فمن ثلاثة ألوان أساسية مختارة
جيّدًا لا غير، يبني مزج \omterm{def:g1:light-and-shadows:source}{الضوء} تقريبًا كل لون تستطيع العين أن
تسمّيه --- ولهذا كانت الثلاثة هي العدد السحري لكل شاشة
متوهّجة.
\end{example}

\begin{example}[الشاشة التي في جيبك]\label{ex:g8:colors-spectra:screen}
افحص منطقة بيضاء من شاشة هاتف أو تلفاز بمكبّرة قوية: فلا بياض
في أيّ موضع --- بل شرائط دقيقة حمراء وخضراء وزرقاء لا غير،
كلها مشتعلة. وكل لون تُظهره الشاشة يومًا هو هذه المصابيح
الثلاثة بنسب مجرّعة؛ وكل ``بياض'' هو الثلاثي في أوج غنائه،
ممزوجًا بمسافة عينك. أما علبة الألوان فتعمل في الاتجاه الآخر
--- فالأصباغ تمتصّ، ومزجها يطرح \omterm{def:g1:light-and-shadows:source}{الضوء} بدل أن يجمعه --- ولهذا
يعطي مزج كل الأصباغ ظلمة كدرة بينما يعطي مزج كل الأضواء
بياضًا؛ وأمّا حكاية الرسّام كاملة فمن نصيب حصة الفنّ
والكيمياء.
\end{example}

\begin{remark}[الغروب، مدفوعًا بالتمام]\label{rem:g8:colors-spectra:sunset}
وعد \omterm{def:g1:day-and-night:daynight}{الغروب} في السنة الماضية يحلّ أجله. \omterm{def:g2:air-around-us:air}{الهواء} يبعثر \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس
--- لكن لا بحياد: فالطرف الأزرق من المزيج يتبعثر أسهل بكثير من
الأحمر. فانظر بعيدًا عن الشمس نهارًا ترَ الحصة المسروقة،
مرسَلةً إليك من السماء كلها: زرقاء. وعند \omterm{def:g1:day-and-night:daynight}{الغروب} يكون طريق
\omterm{def:g6:light-propagation:ray}{الحزمة} المباشرة الملامس عبر \omterm{def:g2:air-around-us:air}{الهواء} أطول ما يكون، وتكاد سرقة
الأزرق تتمّ --- والناجي الذي يبلغ عينك هو الباقي الأحمر
البرتقالي، والغيوم الوردية تلتقط التبعثرات المتأخّرة. فالسماء
الزرقاء \omterm{def:g1:day-and-night:daynight}{والغروب} الأحمر سرقة واحدة، منظورًا إليها من جانبَي
الدفتر.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:7}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:8}

\begin{exercise}[$\star\star\star$]\label{exo:g8:colors-spectra:12}
حِيَل المسرح: يحتاج مصمّم إلى لافتة بحروف ساطعة مقروءة على

أن تختفي، سوداء بانتظام، حروفًا وأرضيةً معًا، تحت الغمر الأحمر

الاختيار المغري --- ستخون الختام.
\end{exercise}

\section{مسألة: مسرح الضوء}

\begin{problem}[{مسألة نهاية الأسبوع --- بروفة الأزياء تحت
كشّافات ملوّنة؛ أزياء ومرشّحات ومشهد واحد مخرَّب؛ امتحان مصمّم
الإضاءة}]\label{pb:g8:colors-spectra:1}
بروفة الأزياء: ثلاثة كشّافات (أحمر وأخضر وأزرق) بمخفّتات، ورفّ
أزياء، ونصّ فيه حِيَل. وأنت تمسك خطة الإضاءة.

\medskip\noindent\textbf{القسم الأول --- تمارين الإحماء.}
\begin{enumerate}
  \item الكشّافات الثلاثة بكامل قوّتها: أيّ \omterm{def:g1:light-and-shadows:source}{ضوء} يغمر الخشبة،
        ولماذا؟
  \item الأحمر والأخضر بكامل قوّتهما والأزرق مطفأ: سمِّ \omterm{def:g1:light-and-shadows:source}{ضوء}
        الخشبة.
  \item تحت ذلك المزيج الأحمر الأخضر، تنبّأ بمظهر: قميص
        أبيض؛ ووشاح أزرق؛ وثوب أصفر.
  \item والخلفية البيضاء تحت الأزرق وحده --- وسبب أن تكون
        الخلفية قماشًا أبيض دائمًا.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- عيادة الأزياء.}
\begin{enumerate}[resume]
  \item يجب أن يبدو معطف الشرّير أسود في المشهد الأول (غمر
        أحمر) وأخضر غنيًّا في المشهد الثاني (\omterm{def:g1:light-and-shadows:source}{ضوء} أبيض). أيّ
        لون نهاري للمعطف يفي بالغرض، ولماذا؟
  \item ويجب أن يبقى رداء الشبح مظهره أبيض في \emph{كل}
        إضاءة. أيّ لون للرداء، وماذا سيُظهر فعلًا تحت كل
        كشّاف على حدة؟
  \item ووشاح ممثّل أحمر يجب أن ``يذوب'' --- فلا يُميَّز عن
        الستارة السوداء --- في لحظة بعينها. أيّ الكشّافات
        يجوز إشعالها للحيلة، ولماذا تفشل الحيلة لحظة ينضمّ
        إليها الكشّاف الأحمر؟
  \item وحروف البرنامج الذهبية بدت رائعة عند الظهيرة لكنها
        بدت كدرة تحت ضبط المساء الغنيّ بالأزرق. اشرح، معاملًا
        الذهبي أصفرَ دافئًا.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- التخريب.}
المشهد الافتتاحي: قوس قزح الديكور (خطوط مطلية: أحمر، وبرتقالي،
وأصفر، وأخضر، وأزرق، وبنفسجي) تحت \omterm{def:g1:light-and-shadows:source}{ضوء} أبيض كامل.
\begin{enumerate}[resume]
  \item يبدّل مشاكس الضبط الأبيض بأحمر خالص. خطًّا خطًّا، ماذا
        يرى الجمهور الآن؟
  \item ويرتجل المخرج فيطلب الكشّافين الأحمر \emph{والأزرق}
        معًا. أيّ الخطوط يحيا، وماذا يفعل الغمر المائل إلى
        الأرجواني بالخطّ الأخضر؟
  \item ويسأل مدير الخشبة لماذا لا يمكن ببساطة ``إعادة إضاءة
        القوس إلى طبيعته'' بكشّاف أخضر وحيد شديد السطوع. أجب
        \omterm{def:g1:light-and-shadows:source}{بالضوء} الوحيد الذي يُظهر الخطوط الستّة كلها.
  \item اكتب جواب امتحان المصمّم: قانونا مسرح \omterm{def:g1:light-and-shadows:source}{الضوء} ---
        واحد عمّا تفعله الأضواء إذا مُزجت، وواحد عمّا تفعله
        الأجسام بما أُعطيت --- كلٌّ في جملة واحدة.
\end{enumerate}
\end{problem}
