\chapter{التيار الكهربائي وآثاره}\label{ch:g8:electric-current-effects}

سار التيار نافعًا بلا اسم خمس سنوات من الدارات: يضيء الخيوط،
ويدير المحرّكات، ويصوّت الطنّانات. وفي هذه السنة تصير الكهرباء
كمّية --- مقاييس ووحدات وقوانين --- ونفتتحها بجرد الحساب: ماذا
يستطيع تيار أن \emph{يفعل} بالضبط؟ ثلاثة آثار، كما يتبيّن،
والعالم الكهربائي كله مبنيّ منها.

\section{الآثار الثلاثة}

\begin{proposition}[آثار التيار الكهربائي]\label{prop:g8:electric-current-effects:three}
التيار السائر في المادة يظهر نفسه بثلاثة آثار:
\begin{enumerate}
  \item \emph{الأثر الحراري}\index{أثر حراري}: كل تيار يُدفئ
        \omterm{def:g4:conductors-insulators:conductor}{الناقل} الذي يعبره --- برفق في الأسلاك الجيّدة، وبضراوة
        في الرفيعة أو المثقلة، وأبيض ساخنًا عن قصد في الخيوط
        والمحمصات؛
  \item \emph{الأثر المغناطيسي}\index{أثر مغناطيسي}: التيار
        يجعل ما حوله مغناطيسيًّا --- فالسلك يزيح إبرة \omterm{def:g4:magnets-and-compass:compass}{بوصلة}
        قريبة، والسلك الملفوف يصير مغناطيسًا حقيقيًّا ذا قطبين،
        قابلًا للتشغيل متى شئت؛
  \item \emph{الأثر الكيميائي}\index{أثر كيميائي}: التيار، إذ
        يعبر سوائل بعينها، يسوق تغيّرًا كيميائيًّا --- فيشطر
        الماء إلى غازين، ويطلي الملاعق بالفضّة، ويشحن كل
        \omterm{def:g3:first-electric-circuit:battery}{بطارية} ويفرّغها.
\end{enumerate}
\end{proposition}

\begin{example}[الأثر الحراري في العمل]\label{ex:g8:electric-current-effects:heating}
عن قصد: قضبان المحمصة المتوهّجة، ووشيعة الغلّاية الخفية، وخيط
\omterm{def:g3:first-electric-circuit:bulb}{المصباح} القديم الأبيض الساخن، \omterm{def:g7:short-circuits-safety:fuse}{والمصهر} المبنيّ \omterm{def:g2:ice-water-steam:meltfreeze}{لينصهر} أولًا.
وبغير رغبة: شواحن الهواتف الدافئة، وبطون الحواسيب المحمولة
الحارّة --- وتهديد \omterm{def:g6:circuits-and-safety:short}{الدارة القصيرة}، وقد درسته توقيعًا للشرّير.
أثر واحد، خادمًا وخطرًا، جرعته بمقدار ما يشتدّ سير التيار ---
وهي جرعة نتعلّم \emph{قياسها} في الفصل التالي.
\end{example}

\section{التيار يصنع المغانط}

\begin{example}[البوصلة تفضح السلك]\label{ex:g8:electric-current-effects:oersted}
قبل قرنين، وأثناء محاضرة، لاحظ فيزيائي إبرة \omterm{def:g4:magnets-and-compass:compass}{بوصلة} ترتجف كلما
أغلق دارة على المنضدة بجانبها. وتلك الرجفة --- وما هي إلا
ارتعاشة إبرة --- كانت أول جسر يُرى بين الكهرباء والمغناطيسية،
وهما علمان كانا إلى ذلك الحين منفصلين انفصال الطقس عن الموسيقى.
ضع \omterm{def:g4:magnets-and-compass:compass}{بوصلة} بجانب سلك وشغّل: تنحرف الإبرة جانبًا؛ واعكس التيار
تنحرف في الجهة الأخرى. فالتيار يخلق مغناطيسية حول نفسه، ولها
اتجاه.
\end{example}

\begin{definition}[المغناطيس الكهربائي]\label{def:g8:electric-current-effects:electromagnet}
\emph{المغناطيس الكهربائي}\index{مغناطيس كهربائي} وشيعة من سلك
معزول --- ملفوفة غالبًا على لبّ حديدي --- تصير مغناطيسًا ما دام
التيار يسري فيها: \omterm{def:g4:magnets-and-compass:poles}{بقطب شمالي} \omterm{def:g4:magnets-and-compass:poles}{وقطب جنوبي} وقدرة على خطف الحديد
وكل شيء. وهو، بخلاف أيّ \omterm{def:g1:magnets:magnet}{مغناطيس} حجري، يطيع قاطعة: فالتيار
مشغَّل، \omterm{def:g1:magnets:magnet}{فالمغناطيس} مشغَّل؛ والتيار مقطوع، \omterm{def:g1:magnets:magnet}{فالمغناطيس} زائل؛
والتيار معكوس، فالقطبان متبادلان.
\end{definition}

\begin{example}[المغناطيسية القابلة للإطفاء تدير العالم]\label{ex:g8:electric-current-effects:uses}
\omterm{def:g3:levers-and-scales:lever}{رافعة} ساحة الخردة تمسك سيارة \omterm{def:g8:electric-current-effects:electromagnet}{بمغناطيس كهربائي} وتُسقطها بفتح
قاطعة --- فجرّب ذلك \omterm{def:g1:magnets:magnet}{بمغناطيس} حجري. وأقفال الأبواب الكهربائية،
وطقطقة المُرحِّل، ومطرقة جرس الباب (\omterm{def:g8:electric-current-effects:electromagnet}{مغناطيس كهربائي} يخطفها إلى
الجرس، فيقطع تيار نفسه، فترتدّ بالنابض: خشخشة صارت قانونًا)،
ورأس القراءة في الهواتف القديمة، وقبل كل شيء كل محرّك كهربائي:
مغناطيسية صنعها تيار تدفع في وجه المغانط، آلاف المرات في
الثانية. \omterm{prop:g8:electric-current-effects:three}{فالأثر المغناطيسي} عضلة العصر الكهربائي.
\end{example}

\begin{method}[لفّ مغناطيس كهربائي]\label{met:g8:electric-current-effects:wind}
\begin{enumerate}
  \item لُفّ مترًا أو مترين من سلك رفيع معزول لفّات محكمة
        مرتّبة حول مسمار حديدي كبير أو برغي، مبقيًا طرفين
        حرّين؛
  \item صِل الطرفين \omterm{def:g3:first-electric-circuit:battery}{ببطارية} عبر قاطعة (لا مباشرة زمنًا طويلًا
        --- فالوشيعة تكاد تكون \omterm{def:g7:short-circuits-safety:device}{دارة قصيرة} وتسخن: \omterm{prop:g8:electric-current-effects:three}{فالأثر
        الحراري} يرافق التجربة)؛
  \item أغلق \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} لحظة: يخطف رأس المسمار مشابك الورق؛
        وافتحها: فتسقط؛
  \item اختبر \omterm{def:g4:magnets-and-compass:compass}{ببوصلة}: فطرفا الوشيعة يسلكان مسلك \omterm{def:g4:magnets-and-compass:poles}{قطب شمالي} \omterm{def:g4:magnets-and-compass:poles}{وقطب
        جنوبي} --- ويتبادلان حين تعكس \omterm{def:g3:first-electric-circuit:battery}{البطارية}.
\end{enumerate}
لفّات أكثر، فقبضة أقوى: جرِّع المغناطيسية باللفّ.
\end{method}

\section{الأثر الكيميائي}

\begin{example}[تيار في الماء]\label{ex:g8:electric-current-effects:electrolysis}
اغمس مسريَين من رصاص أقلام موصولين \omterm{def:g3:first-electric-circuit:battery}{ببطارية} في ماء جُعل ناقلًا
قليلًا (وملعقة من كربونات الغسيل تعين)، وراقب: فقاقيع دقيقة
تنساب من الرصاصتين معًا --- غازان مختلفان، مولودان من الماء
نفسه، \omterm{def:g6:mass-and-volume:volume}{وحجم} أحدهما عند إحدى الرصاصتين ضِعف الآخر عند الأخرى.
فالتيار يفكّك الماء إلى مكوّنيه غير المرئيين. وهذا الباب بين
الكهرباء والكيمياء --- \emph{التحليل الكهربائي} --- يطلي
الملاعق الرخيصة بالفضّة، وينتزع الألمنيوم من خامه، وهو في
معظمه من نصيب منهج الكيمياء؛ والفيزياء تحيّيه وتمضي.
\end{example}

\begin{example}[كل بطارية هي الأثر معكوسًا]\label{ex:g8:electric-current-effects:battery}
داخل كل \omterm{def:g3:first-electric-circuit:battery}{بطارية} تدفع الكيمياء التيار: \omterm{def:g5:everyday-energy:energy}{فالطاقة} الكيميائية
المخزونة تسوق المسير، وتنفق نفسها وهي تخدم --- وهو الإفراغ
الذي عرفته منذ خفت \omterm{def:g3:first-electric-circuit:bulb}{المصباح} اليدوي. والخلايا القابلة لإعادة
الشحن تغلق الدائرة: فالشاحن يسوق التيار عبرها
\emph{بالمقلوب}، فيعيد \omterm{prop:g8:electric-current-effects:three}{الأثر الكيميائي} بناء المخزون. من
كيميائية إلى كهربائية، ومن كهربائية إلى كيميائية: أثر واحد،
واتجاهان، وصناعة البطاريات كلها.
\end{example}

\begin{remark}[الآثار كواشف]\label{rem:g8:electric-current-effects:detect}
الآثار الثلاثة تجيب عن سؤال حيّرك يوم كنت صبيّ الفاحص: كيف
\emph{ترى} تيارًا --- مسيرًا غير مرئي في سلك \omterm{def:g7:light-sources-propagation:coats}{معتم}؟ بأفعاله.
سلك يدفأ، \omterm{def:g4:magnets-and-compass:compass}{وبوصلة} ترتجف، وفقاعة تنساب: كل منها يفضح المسير. وكل
آلة قيست بها التيارات يومًا تقرأ واحدًا من الآثار الثلاثة ---
والآلة التي تنتظر في الفصل التالي تقرأ الرجفة المغناطيسية،
مصقولةً في قرص مرقّم.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g8:electric-current-effects:9}

وهكذا. تتبّع دورة كاملة واحدة واشرح لماذا \emph{يجب} أن يخشخش

\begin{exercise}[$\star\star$]\label{exo:g8:electric-current-effects:10}

وأيّ دارتين لا تتلامسان أبدًا.
\end{exercise}

إبرة بوصلتك ترتجف كلما حمل سلك مخفيّ في الجدار تيارًا. صمّم من

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:electric-current-effects:12}

\end{exercise}

\section{مسألة: ورشة المخترع}

\begin{problem}[{مسألة نهاية الأسبوع --- بطارية واحدة، وثلاثة
آثار، وخمسة اختراعات؛ أمسية في ورشة
المخترع}]\label{pb:g8:electric-current-effects:1}
تحمل منضدة الورشة بطاريات وسلكًا ومسامير حديدية \omterm{def:g4:magnets-and-compass:compass}{وبوصلة}
ورصاصتَي قلمين وجرّة ماء بالصودا --- ودفترًا من الاختراعات
نصف المبنية لتقييمها.

\medskip\noindent\textbf{القسم الأول --- جرد الحساب.}
\begin{enumerate}
  \item ارسم منضدة اختبار الآثار الثلاثة في الورشة: \omterm{def:g3:first-electric-circuit:battery}{لبطارية}
        واحدة وقاطعة، صِف تركيبًا يُظهر الآثار الثلاثة دفعة
        واحدة (سلك رفيع يدفأ، \omterm{def:g4:magnets-and-compass:compass}{وبوصلة} بجانب سلك، والجرّة في
        الحلقة). وبأيّ ترتيب يجب أن تجلس العروض الثلاثة ---
        \omterm{def:g4:series-circuits:series}{على التسلسل} أم \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} --- كي تديرها كلها قاطعة
        واحدة؟
  \item سلك العرض الرفيع يدفأ بينما تبقى أسلاك التغذية
        الغليظة باردة، في الحلقة نفسها. أيّ حقيقة من حقائق
        الفصل الأول عن حلقة التسلسل تجعل هذا \emph{أطرف} مما
        يبدو --- ما المتساوي في السلكين، وما المختلف؟
  \item ترتجف \omterm{def:g4:magnets-and-compass:compass}{البوصلة} شرقًا حين تُغلق \omterm{ex:g3:first-electric-circuit:switch}{القاطعة}. أيّ تغييرين،
        إذا أُجريا معًا، يتركان الرجفة كما كانت بالضبط؟
  \item تنساب الفقاقيع من إحدى رصاصتَي القلمين \omterm{def:g7:motion-average-speed:formula}{بسرعة} ضِعف
        الأخرى. علامَ يلمّح هذا في شأن مكوّنَي الماء؟
        (والكيمياء ستؤكّد النسبة.)
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- تقييم الاختراعات.}
\begin{enumerate}[resume]
  \item اختراع: ``جرس الباب الصامت'' --- \omterm{def:g8:electric-current-effects:electromagnet}{مغناطيس كهربائي} يجذب
        مطرقة من اللبّاد الليّن مرة واحدة ويمسكها حتى يُترك
        الزرّ. أيّ قطعة من آليّة جرس الباب الكلاسيكي يجب على
        المخترع \emph{إزالتها}، وماذا سيسمع الزائر؟
  \item اختراع: ``الفارز المغناطيسي'' --- خردة على \omterm{def:g4:conductors-insulators:conductor}{ناقل} تمرّ
        تحت \omterm{def:g8:electric-current-effects:electromagnet}{مغناطيس كهربائي} ينتشل الحديد والفولاذ. ماذا يحدث
        عند صندوق الإسقاط، ولماذا يترك الفارز علب الألمنيوم
        على السير (تجيب حقيقتان عن المغانط من سنوات سابقة)؟
  \item اختراع: ``قالب الأقطاب'' --- \omterm{def:g4:magnets-and-compass:compass}{بوصلة} مركّبة فوق وشيعة
        تنقلب كلما عكست قاطعة مخفيّة \omterm{def:g3:first-electric-circuit:battery}{البطارية}. أيّ خاصّية من
        خواصّ المغناطيسية التي يصنعها التيار تُستثمر هنا؟
  \item اختراع: ``الطالي الكهربائي الأبدي'' --- طلي بالفضّة
        تغذّيه \omterm{def:g3:first-electric-circuit:battery}{بطارية} ``تعيد شحن نفسها من الطلي''. ارفضه،
        مستشهدًا باتّجاهَي \omterm{def:g3:first-electric-circuit:battery}{البطارية} في هذا الفصل وبقانون أقدم
        عن \omterm{def:g5:everyday-energy:energy}{الطاقة}.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- تحفة الليلة.}
يختم المخترع بلعبة \omterm{def:g3:levers-and-scales:lever}{رافعة} للمهرجان: عصا توجيه، \omterm{def:g8:electric-current-effects:electromagnet}{ومغناطيس كهربائي}
صغير على خيوط، وصندوق من الحلي الحديدية.
\begin{enumerate}[resume]
  \item اكتب سيناريو الحركة الفائزة بلغة الآثار: ماذا يحدث
        عند ``الإمساك''، وأثناء الأرجحة، وعند ``الإطلاق''؟
  \item يشكو روّاد المهرجان أن \omterm{def:g3:levers-and-scales:lever}{الرافعة} تُسقط الجوائز في منتصف
        الأرجحة كلما هزلت \omterm{def:g3:first-electric-circuit:battery}{البطارية} القديمة. أيّ خاصّية من
        خواصّ \omterm{def:g8:electric-current-effects:electromagnet}{المغناطيس الكهربائي} --- وهي فضيلة وضعف في آن ---
        تُعرَض هنا؟
  \item يضاعف المخترع لفّات الوشيعة ليساعد. تنبّأ بالأثر في
        القبضة، وسمِّ الجرعة الأخرى (من جهة \omterm{def:g3:first-electric-circuit:battery}{البطارية}) التي
        سيتيح لنا الفصل التالي أن نعطيها عددًا.
  \item آخر صفحة في الدفتر: اكتب خلاصة الورشة --- جملة لكل
        أثر، تسمّي كل واحدة أفضل اختراع خادم لها في الأمسية.
\end{enumerate}
\end{problem}
