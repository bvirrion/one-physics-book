\chapter{العدسات والصور}\label{ch:g8:lenses-images}

آلة التصوير ذات الثقب اشترت صورتها الصغيرة الحادّة بثمن قاسٍ:
\omterm{def:g1:light-and-shadows:source}{ضوء} لا يكاد يُذكر. ومع ذلك ترسم عينك أنت عالمًا ساطعًا واضحًا
على جدارها الخلفي طوال النهار، ولا ثقب في الأفق. وسرّ العين
قرص من مادة زجاجية منحنية يجمع \omterm{def:g6:light-propagation:ray}{حزمة} كاملة ويكسرها كما يشاء ---
\omterm{def:g8:lenses-images:families}{العدسة}، نجمة آلات التصوير والنظّارات والمكبّرات وهذا الفصل.

\section{عائلتا العدسات}

\begin{definition}[العدسات المجمِّعة والمفرِّقة]\label{def:g8:lenses-images:families}
\emph{العدسة}\index{عدسة} قطعة من مادة \omterm{def:g7:light-sources-propagation:coats}{شفّافة} --- زجاج أو
بلاستيك صافٍ --- بوجهين منحنيين، مبنيّة لتكسر \omterm{def:g1:light-and-shadows:source}{الضوء} كسرًا
مضبوطًا (\omterm{def:g1:light-and-shadows:source}{فالضوء} ينكسر حين يعبر إلى الزجاج؛ وأمّا قانون ذلك
الكسر بدقّة فكنز الثانوية). وهما عائلتان:
\begin{enumerate}
  \item \emph{العدسات المجمِّعة}\index{عدسة مجمِّعة} ---
        وهي أغلظ في الوسط منها عند الحافة --- تكسر أشعّة
        \omterm{def:g6:light-propagation:ray}{الحزمة} \emph{بعضها نحو بعض}؛
  \item \emph{العدسات المفرِّقة}\index{عدسة مفرِّقة} --- وهي
        أرقّ في الوسط --- تباعد بينها.
\end{enumerate}
مرّر طرف إصبعك على عدسة نظّارة: فنتوء في الوسط أو تجويف في
الوسط يفرز كل عدسة على وجه الأرض إلى عائلتها.
\end{definition}

\begin{definition}[البؤرة والبعد البؤري]\label{def:g8:lenses-images:focus}
أرسل \omterm{def:g6:light-propagation:ray}{حزمة} من أشعّة متوازية --- \omterm{def:g1:light-and-shadows:source}{وضوء} الشمس يفي بالغرض --- عبر
\omterm{def:g8:lenses-images:families}{عدسة مجمِّعة}: فالأشعّة المنكسرة تتقاطع كلها عند نقطة واحدة
خلفها، هي \emph{البؤرة}\index{بؤرة}. والمسافة من \omterm{def:g8:lenses-images:families}{العدسة} إلى
البؤرة هي \emph{البعد البؤري}\index{بعد بؤري} $f$: عدد \omterm{def:g8:lenses-images:families}{العدسة}
المميّز الأوحد. فالعدسات السمينة الشديدة الانحناء تكسر بقوة ---
فيقصر $f$؛ والعدسات الرقيقة الانحناء تكسر قليلًا --- فيطول
$f$.
\end{definition}

\begin{method}[قياس بعد بؤري]\label{met:g8:lenses-images:measure}
لأيّ \omterm{def:g8:lenses-images:families}{عدسة مجمِّعة}، دقيقة واحدة ونافذة بعيدة:
\begin{enumerate}
  \item واجه جدارًا مقابلًا لمشهد \emph{بعيد} ساطع --- النافذة
        والعالم وراءها؛
  \item أمسك \omterm{def:g8:lenses-images:families}{العدسة} قرب الجدار وأبعدها ببطء حتى يقفز المشهد
        صورة صغيرة حادّة على الجدار (مقلوبة رأسًا على عقب ---
        كما هو متوقّع الآن)؛
  \item قِس المسافة من \omterm{def:g8:lenses-images:families}{العدسة} إلى الجدار: فلمشهد بعيد، تلك
        المسافة \emph{هي} \omterm{def:g8:lenses-images:focus}{البعد البؤري} --- إذ تصل الأشعّة
        البعيدة متوازية عمليًّا، والأشعّة المتوازية تجتمع عند
        \omterm{def:g8:lenses-images:focus}{البؤرة}.
\end{enumerate}
المكبّرة \omterm{def:g6:measurement-in-science:measuring}{تقيس} نحو \qty{10}{cm}؛ وعدسات النظّارات، عشرات
السنتيمترات؛ \omterm{def:g8:lenses-images:families}{وعدسة} آلة التصوير القصيرة، بضعة سنتيمترات.
\end{method}

\begin{remark}[الزجاجة الحارقة]\label{rem:g8:lenses-images:burning}
عند \omterm{def:g8:lenses-images:focus}{بؤرة} \omterm{def:g8:lenses-images:families}{عدسة مجمِّعة}، يتزاحم عرض \omterm{def:g6:light-propagation:ray}{حزمة} كامل من \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس في
نقطة واحدة متوهّجة --- مركَّزًا بما يكفي لتفحيم الورق وإشعال
الحرائق في ثوانٍ. فعامِله بما هو: مبدأ الطبّاخ الشمسي في \omterm{def:g6:mass-and-volume:volume}{حجم}
الجيب. ولا تبئّر \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس قرب شيء قابل للاشتعال أبدًا، ولا
نحو جلد أو ثياب --- ولا تنظر إلى الشمس، والقانون القديم مضاعف
هنا، \emph{عبر} أيّ \omterm{def:g8:lenses-images:families}{عدسة}: فتبئير العين نفسها مضافًا إلى تبئير
\omterm{def:g8:lenses-images:families}{العدسة} يصنع حرقًا لحظيًّا بلا ألم ودائمًا.
\end{remark}

\section{العدسة صانعةَ صور}

\begin{proposition}[الصور الحقيقية]\label{prop:g8:lenses-images:realimage}
\omterm{def:g8:lenses-images:families}{العدسة المجمِّعة} تجمع الأشعّة التي تبعثرها كل نقطة من الجسم
وتسلّمها إلى نقطة مقابلة على الجانب الآخر: فتركّب، نقطة نقطة،
\omterm{prop:g8:lenses-images:realimage}{صورة حقيقية} --- \emph{صورة حقيقية}\index{صورة حقيقية} ---
يمكن التقاطها على حاجز. والصورة مقلوبة، كصورة الثقب، لكنها
ساطعة: \omterm{def:g8:lenses-images:families}{فالعدسة} تحصد ملء \omterm{def:g8:lenses-images:families}{عدسة} من \omterm{def:g1:light-and-shadows:source}{الضوء} لكل نقطة حيث كان الثقب
يمرّر خيطًا واحدًا. وقواعد الهندسة التقريبية: الجسم البعيد
يُصوَّر صغيرًا وقرب \omterm{def:g8:lenses-images:focus}{البؤرة}؛ وكلما اقترب الجسم من \omterm{def:g8:lenses-images:families}{العدسة} تراجعت
صورته وكبرت.
\end{proposition}

\begin{example}[آلة التصوير والعين، هندسة واحدة]\label{ex:g8:lenses-images:cameraeye}
آلة التصوير هي صانعة الصور معلَّبةً: \omterm{def:g8:lenses-images:families}{عدسة مجمِّعة} في المقدّمة،
وحاجز حسّاس \omterm{def:g1:light-and-shadows:source}{للضوء} في الخلف، وصورة تتكوّن عليه مقلوبة (والبرمجية
تدير الملفّ بأدب). وعينك هي الآلة نفسها في نسيج حيّ: \omterm{def:g8:lenses-images:families}{عدسة
مجمِّعة} خلف البؤبؤ، و\emph{الشبكية} حاجزًا في الخلف، وصورة
مقلوبة --- والدماغ يقلبها، كما لوحظ يوم علّمنا الثقب أول مرة.
ويختلف التبئير اختلافًا جميلًا: فآلة التصوير تحرّك عدستها؛ \omterm{def:g8:lenses-images:families}{وعدسة}
العين الليّنة \emph{تغيّر شكلها}، تُشدّ أسمن للقريب، وتُرخى
أرقّ للبعيد --- تحسّ الجهد بعد ساعة من القراءة القريبة.
\end{example}

\begin{example}[النظّارات]\label{ex:g8:lenses-images:spectacles}
حين يقصر تبئير العين نفسها، تكمّله \omterm{def:g8:lenses-images:families}{عدسة} أمامها. فالعين القصيرة
البصر تبئّر بقوة زائدة --- فتتركّب المشاهد البعيدة \emph{قبل}
الشبكية وتصل مشوّشة: \omterm{def:g8:lenses-images:families}{وعدسة} نظّارة \emph{مفرِّقة} تفكّ كسر \omterm{def:g1:light-and-shadows:source}{الضوء}
بالقدر الكافي. والعيون الطويلة البصر (ومعظم العيون المسنّة)
تبئّر بضعف لا يكفي للعمل القريب: \omterm{def:g8:lenses-images:families}{وعدسة} \emph{مجمِّعة} تقرضها
\omterm{def:g2:pushes-and-pulls:force}{القوة} الناقصة --- فنظّارات القراءة أبناء عمومة صغار أمناء
للمكبّرة. عائلتان من العدسات، وعائلتان من التشويش.
\end{example}

\section{المكبّرة}

\begin{example}[داخل البعد البؤري]\label{ex:g8:lenses-images:magnifier}
أمسك \omterm{def:g8:lenses-images:families}{عدسة مجمِّعة} \emph{أقرب} إلى طابع بريد من بعدها البؤري
تتغيّر طبيعة الصورة: فلا صورة تحطّ على أيّ حاجز --- بل ترى
الطابع، وأنت تنظر عبر \omterm{def:g8:lenses-images:families}{العدسة}، قائمًا في مكانه ومكبَّرًا.
\omterm{def:g8:lenses-images:families}{فالعدسة} تكسر أشعّة الطابع حتى تبلغ العين \emph{كأنها} مبعثَرة
عن طابع أفخم خلف الزجاج --- وهْم مصنوع على المقاس، ابن عمّ
عالم \omterm{def:g5:mirrors-reflection:reflection}{المرآة} الشبحي. وقد استعمله الساعاتيّون وجامعو الطوابع
ونفسك الماضية بمكبّرة الطفولة؛ وأمّا هندسة الأشعّة الكاملة
لهذه الحيلة فجوهرة من جواهر مجلّد الثانوية.
\end{example}

\begin{remark}[آلات فوق آلات]\label{rem:g8:lenses-images:instruments}
كدّس العدسات ينفتح العالم. فعدستان مجمِّعتان تصنعان مجهرًا:
إحداهما تقذف \omterm{prop:g8:lenses-images:realimage}{صورة حقيقية} كبيرة لقطيرة، والثانية تكبّر تلك
الصورة --- وهكذا نال المدافعون عن \omterm{def:g7:states-of-matter:molecule}{الجزيئات} أول حلفائهم
المباشرين. \omterm{def:g8:lenses-images:families}{وعدسة} كبيرة (أو \omterm{def:g5:mirrors-reflection:reflection}{مرآة} منحنية) مع مكبّرة تصنعان
مقرابًا: فأشعّة المجرّة البعيدة المتوازية الخافتة تُجمع في \omterm{prop:g8:lenses-images:realimage}{صورة
حقيقية}، ثم تُكبَّر في عين مدهوشة. وكل آلة بصرية في الحضارة هي
فكرتا هذا الفصل --- اجمع، ثم كبّر --- في ترتيبات شتّى.
\end{remark}

\section{مسألة: منضدة النظّاراتي}

\begin{problem}[{مسألة نهاية الأسبوع --- متدرّبًا عند نظّاراتي
البلدة؛ أبعاد بؤرية، وتركيبات، ومقراب واحد
مشروخ}]\label{pb:g8:lenses-images:1}
يوم عند النظّاراتي: عدسات تُفرَز، ونظّارات تُطابَق، ومقراب
زبون قطعًا.

\medskip\noindent\textbf{القسم الأول --- فرز الصينية.}
انسكاب درج خلط المخزون.
\begin{enumerate}
  \item اقترح اختبارين سريعين --- واحدًا باللمس وواحدًا
        بالنافذة والجدار --- لفرز \omterm{def:g8:lenses-images:families}{العدسات المجمِّعة} عن
        المفرِّقة.
  \item \omterm{def:g8:lenses-images:families}{العدسة} أ تقذف صورة حادّة للشارع على الجدار عند
        \qty{25}{cm}. ما بعدها البؤري وما عائلتها؟
  \item \omterm{def:g8:lenses-images:families}{والعدسة} ب تأبى أن تقذف أيّ صورة على الجدار عند أيّ
        مسافة، والأشياء المرئية عبرها تبدو أصغر. ما عائلتها؟
  \item والعدستان ج ود مجمِّعتان كلتاهما؛ \omterm{def:g8:lenses-images:families}{فالعدسة} ج تصنع صورة
        النافذة عند \qty{10}{cm}، ود عند \qty{40}{cm}.
        فأيّهما أشدّ انحناءً وأقوى كسرًا؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- التركيبات.}
\begin{enumerate}[resume]
  \item زبون يرى لافتات الطريق البعيدة مشوّشة لكنه يقرأ قوائم
        الطعام بلا جهد. شخّص العين وصِف العائلة.
  \item وآخر يمسك الجريدة على مدّ ذراعه ويشكو أن ذراعيه
        ``قصيرتان''. شخّص وصِف.
  \item ويحذّر النظّاراتي الزبون الثاني: ``نظّارة القراءة
        الجديدة ستشوّش الجبال عليك.'' فلماذا تكون نظّارات
        القراءة للصفحات القريبة لا غير؟
  \item ويسأل طفل لماذا تجعل عدسات النظّاراتي التجريبية لوحة
        الفحص تسبح بينما عيناه هو ``سليمتان''. فماذا تفعل
        \omterm{def:g8:lenses-images:families}{عدسة} تصحيحية بعين مبئَّرة جيّدًا؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- المقراب المشروخ.}
يجيء زبون بمقراب لرصد النجوم وعدسته الأمامية مشروخة، يسأل إن
كانت المكبّرة الاحتياطية في العلبة تصلح بديلًا لها.
\begin{enumerate}[resume]
  \item سمِّ وظيفتَي العدستين في الآلة السليمة: ماذا تستقبل
        \omterm{def:g8:lenses-images:families}{العدسة} الأمامية الكبيرة من \omterm{def:g4:solar-system:star}{نجم} وماذا تصنع منه، وماذا
        تفعل \omterm{def:g8:lenses-images:families}{عدسة} العين الصغيرة بذلك؟
  \item \omterm{def:g8:lenses-images:families}{العدسة} الأمامية بعرض صحن؛ \omterm{def:g8:lenses-images:families}{وعدسة} العين \omterm{def:g6:mass-and-volume:volume}{بحجم} قطعة
        نقدية. فلماذا يهمّ \emph{عرض} \omterm{def:g8:lenses-images:families}{العدسة} الأمامية كل هذا
        القدر للنجوم الخافتة؟
  \item والمكبّرة الاحتياطية شديدة الانحناء لكنها \omterm{def:g6:mass-and-volume:volume}{بحجم} قطعة
        نقدية. احكم في الاستبدال: ماذا يُكسب وماذا يُفقد في
        مقدّمة الآلة؟
  \item اختم اليوم بعقيدة النظّاراتي في سطر واحد عن كل آلة
        على الرفّ --- النظّارات، والمكبّرات، وآلات التصوير،
        والمقاريب --- وبخلاصة الفعلين لما تفعله العدسات
        للحضارة.
\end{enumerate}
\end{problem}
