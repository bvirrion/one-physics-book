\chapter{المقاومة وقانون أوم}\label{ch:g8:ohms-law}

سؤال واحد ظلّ يتبع الحكاية الكهربائية كلها: فبعض العناصر تدع
المسير يندفع، وبعضها يخنقه حتى يصير قطرات --- والمصابيح تقتسم
التوترات على غير سواء، والأسلاك الرفيعة تدفأ، والغليظة تبقى
باردة. \omterm{def:g8:ohms-law:resistance}{والمقاومة} نفسها تنال الآن اسمًا \omterm{def:g6:measurement-in-science:measuring}{ووحدة}، وتنال ---
تتويجًا لأربع سنوات من الدارات --- أنفعَ قانون في هذا الكتاب:
ثلاثة حروف تربط الدفعة والمسير \omterm{def:g8:ohms-law:resistance}{والمقاومة} في سطر جبر واحد.

\section{المقاومة}

\begin{definition}[المقاومة]\label{def:g8:ohms-law:resistance}
\emph{مقاومة}\index{مقاومة} عنصر $R$ \omterm{def:g6:measurement-in-science:measuring}{تقيس} مدى شدّة مقاومته
\omterm{def:g7:circuits-series-parallel:current}{للتيار الكهربائي}: فعند \omterm{def:g8:voltage-voltmeter:voltage}{توتر} معطى، كلما كبرت المقاومة ضعف
المسير الذي تسمح بمروره. ووحدتها \emph{الأوم}\index{أوم}
(ورمزه $\Omega$، وهو حرف أوميغا اليوناني الكبير)، تكريمًا
للفيزيائي المعلّم الذي وجد قانون هذا الفصل بأسلاك منزلية الصنع
وصبر بطولي.
\end{definition}

\begin{definition}[المقاوم]\label{def:g8:ohms-law:resistor}
\emph{المقاوم}\index{مقاوم} عنصر مصنوع ليكون له \omterm{def:g8:ohms-law:resistance}{مقاومة} مختارة
ثابتة --- عائق معاير، يُباع من أجزاء \omterm{def:g8:ohms-law:resistance}{الأوم} إلى ملايينه.
والدارات تستعمل المقاومات لضبط التيارات عن قصد: ترويض المسير
الذي كان سيحرق قطعة رقيقة، والتخفيت، والتقسيم، والتعديل.
وأشرطتها الملوّنة تهجّي قيمتها؛ ورمزها مستطيل بسيط.
\end{definition}

\begin{example}[مقاومات من حولك]\label{ex:g8:ohms-law:values}
\omterm{def:g2:measuring-length:units}{متر} من سلك نحاسي غليظ: أجزاء من مئة من \omterm{def:g8:ohms-law:resistance}{الأوم} --- الطريق
المستوي. \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} كشّاف يدوي متوهّج: بضعة أومات. ومحرّك كهربائي
صغير: أومات قليلة. ووشيعة تسخين الغلّاية: نحو $25\,\Omega$.
وجلد إنسان جافّ، من يد إلى يد: عشرات الآلاف من الأومات ---
وبالبلل، أقلّ بصورة كارثية: فقاعدة اليدين الرطبتين القديمة
عبارة عن \omterm{def:g8:ohms-law:resistance}{مقاومة}. \omterm{ex:g3:first-electric-circuit:switch}{والقاطعة} المفتوحة: \omterm{def:g8:ohms-law:resistance}{مقاومة} فوق كل \omterm{def:g6:measurement-in-science:measuring}{قياس} ---
العائق اللانهائي.
\end{example}

\section{التجربة}

\begin{method}[قياس صورة $U$--$I$ لعنصر]\label{met:g8:ohms-law:portrait}
\omterm{def:g8:ohms-law:resistor}{مقاوم} واحد، ومغذٍّ قابل للضبط (أو طابور خلايا متنامٍ)،
والمقياسان معًا:
\begin{enumerate}
  \item صِل المغذّي \omterm{def:g8:intensity-ammeter:ammeter}{ومقياس التيار} \omterm{def:g8:ohms-law:resistor}{والمقاوم} \omterm{def:g4:series-circuits:series}{على التسلسل}؛ وعلّق
        \omterm{def:g8:voltage-voltmeter:voltmeter}{مقياس التوتر} على طرفَي \omterm{def:g8:ohms-law:resistor}{المقاوم}؛
  \item اضبط أصغر \omterm{def:g8:voltage-voltmeter:voltage}{توتر}؛ وسجّل الثنائية ($U$، $I$)؛
  \item ارفع \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} خطوة خطوة --- خلية، فخليتان، فثلاث ---
        مسجّلًا ثنائية في كل مرة؛
  \item مثّل النقاط، $I$ أفقيًّا و$U$ عموديًّا، وانظر.
\end{enumerate}
\end{method}

\begin{example}[اعتراف مقاوم]\label{ex:g8:ohms-law:confession}
جدول جلسة واحدة، \omterm{def:g8:ohms-law:resistor}{لمقاوم} واحد:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
ضاعف الدفعة يتضاعف المسير؛ وثلّثها يتثلّث: فالمقداران $U$
و$I$ \emph{متناسبان}، وكل نسبة $U/I$ تعيد العدد نفسه: $15$. وليست
تلك النسبة الثابتة مصادفة --- بل هي \omterm{def:g8:ohms-law:resistance}{مقاومة} \omterm{def:g8:ohms-law:resistor}{المقاوم}،
$R = 15\,\Omega$، تُظهر نفسها في كل سطر.
\end{example}

\begin{proposition}[قانون أوم]\label{prop:g8:ohms-law:law}
\omterm{def:g8:ohms-law:resistor}{لمقاوم} عند \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} ثابتة، يتناسب \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} على طرفيه مع التيار
المارّ فيه، وثابت التناسب هو مقاومته:
\[
U = R \times I ,
\]
مع $U$ \omterm{def:g8:voltage-voltmeter:voltage}{بالفولت}، و$I$ \omterm{def:g8:intensity-ammeter:intensity}{بالأمبير}، و$R$ \omterm{def:g8:ohms-law:resistance}{بالأوم}. ويُعاد ترتيبه على
الهوى: $I = U/R$ (التيار الذي تسوقه دفعة عبر عائق)،
و$R = U/I$ (العائق مكشوفًا بثنائية قراءات أمينة واحدة).
\end{proposition}

\begin{example}[القانون يحسب كل شيء]\label{ex:g8:ohms-law:compute}
متى عُلم $R$، أجاب القانون في كل اتجاه. وشيعة الغلّاية،
$25\,\Omega$ على شبكة \qty{230}{V}: $I = U/R = 230 \div 25 =
\qty{9.2}{A}$ --- وهكذا يُفسَّر جهاز العشرة \omterm{def:g8:intensity-ammeter:intensity}{أمبير}. \omterm{def:g8:ohms-law:resistor}{ومقاوم} يجب
أن يحدّ تيار \omterm{def:g3:first-electric-circuit:bulb}{مصباح} رقيق عند \qty{0.02}{A} من \qty{4.5}{V}:
$R = U/I = 4.5 \div 0.02 = 225\,\Omega$ --- فاختر القيمة
المعيارية التالية. ويسري \qty{0.05}{A} عبر $60\,\Omega$:
\omterm{def:g8:voltage-voltmeter:voltage}{فالتوتر} على طرفيه $U = 60 \times 0.05 = \qty{3}{V}$. قانون
واحد، وثلاث وصفات، والورشة كلها.
\end{example}

\begin{remark}[عكّاز المثلّث]\label{rem:g8:ohms-law:triangle}
يرسم بعض المتعلّمين $U$ فوق $R$ و$I$ في مثلّث صغير، ثم يغطّون
الحرف المطلوب ليقرأوا الوصفة. وهو عكّاز لا ضرر فيه --- لكن
جبرك يعيد الآن ترتيب $U = R I$ بأمانة في سطر واحد، والطريقة
الأمينة تسافر معك إلى كل صيغة ستلقاها يومًا، بمثلّث أو بغير
مثلّث. فاتّكئ على العكّاز هذه السنة إن اضطررت؛ ولكن انوِ
المشي.
\end{remark}

\section{ما تفسّره المقاومة}

\begin{example}[ألغاز قديمة على فاتورة واحدة]\label{ex:g8:ohms-law:mysteries}
اقتسام \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} غير المتساوي في فصل القوانين: فالمصابيح \omterm{def:g4:series-circuits:series}{على
التسلسل} تتقاسم $I$ واحدًا، فبمقتضى $U = R I$ يأخذ $R$ الأكبر
$U$ الأكبر --- هبوط العامل الأشقّ الأكبر، وقد صار قابلًا
للحساب. \omterm{def:g5:series-parallel-circuits:parallel}{والفرع} المخنوق في فصل التفرع: فالسلك الطويل الرفيع
أضاف \omterm{def:g8:ohms-law:resistance}{مقاومة}، فقلّصت $I = U/R$ حصة ذلك \omterm{def:g5:series-parallel-circuits:parallel}{الفرع}. \omterm{def:g6:circuits-and-safety:short}{والدارة القصيرة}:
التفاف مقاومته تكاد تكون صفرًا، فتنفجر $I = U/R$ --- فالاندفاع
كان دائمًا قسمة على ما يقارب الصفر. أربع سنوات من معرفة
الدارات الوصفية، على فاتورة قانون واحد.
\end{example}

\begin{example}[الأسلاك، بالمتر]\label{ex:g8:ohms-law:wires}
\omterm{def:g8:ohms-law:resistance}{مقاومة} السلك تكبر بطوله وتصغر بغلظه --- فالسلك الطويل الرفيع
طريق جبلي ضيّق، والقصير الغليظ طريق سريع. ومن هنا كبلات
الكهربائي السمينة للأجهزة النهمة، وسبيكة وشيعة التسخين
\omterm{def:g8:ohms-law:resistance}{المقاومة} الطويلة الرفيعة عن قصد --- ومن هنا أيضًا سبب أن
القانون نفسه الذي يضيء مصباحًا يسخّن محمصة: \omterm{def:g8:ohms-law:resistance}{فالمقاومة} هي حيث
ينفق المسير دفعته، والدفعة المنفقة، كما سيؤكّد فصل \omterm{def:g5:everyday-energy:energy}{الطاقة} في
السنة القادمة، تصير دفئًا.
\end{example}

\begin{remark}[أين تنتهي ولاية القانون]\label{rem:g8:ohms-law:limits}
قانون \omterm{def:g8:ohms-law:resistance}{أوم} قانون \omterm{def:g8:ohms-law:resistor}{مقاوم}، لا حقّ كوني. فقِس صورة \emph{\omterm{def:g3:first-electric-circuit:bulb}{مصباح} ذي
خيط} تجد النقاط تنحني بعيدًا عن الخطّ المستقيم: فالخيط يسخن
كلما نما التيار، والمعدن الساخن يقاوم أكثر --- فيأبى $R$ أن
يستقرّ. والثنائيات والمحرّكات والجلد الحيّ تحني صورها هي أيضًا،
كلٌّ على طريقته. فالخطّ المستقيم المارّ بالمبدأ توقيع \omterm{def:g8:ohms-law:resistor}{المقاوم}،
لا توقيع المادة --- فاسأل كل عنصر عن صورته قبل أن تأتمنه على
القانون.
\end{remark}

\section{مسألة: منضدة ضبط الجودة}

\begin{problem}[{مسألة نهاية الأسبوع --- يوم الفحص في مصنع
المقاومات؛ صور وتفاوتات، وعنصر واحد ليس مقاومًا
البتّة}]\label{pb:g8:ohms-law:1}
تختبر منضدة الجودة في المصنع إنتاج اليوم --- ومحتالًا واحدًا.
وأنت تدير المقاييس.

\medskip\noindent\textbf{القسم الأول --- معايرة المنضدة.}
\begin{enumerate}
  \item ارسم (أو صِف) دارة الاختبار: المغذّي، \omterm{def:g8:intensity-ammeter:ammeter}{ومقياس التيار}،
        والعنصر تحت الاختبار، \omterm{def:g8:voltage-voltmeter:voltmeter}{ومقياس التوتر} --- فمن \omterm{def:g4:series-circuits:series}{على
        التسلسل}، ومن على الطرفين؟
  \item العيّنة الأولى، عند \qty{4.5}{V}، تمرّر \qty{0.15}{A}.
        فما مقاومتها؟
  \item وتزعم اللافتة $33\,\Omega$ بتفاوت خمسة في المئة. احسب
        النطاق المقبول، واحكم في العيّنة الأولى. (خمسة في
        المئة من $33$ أولًا.)
  \item العيّنة الثانية، ولافتتها $150\,\Omega$: تنبّأ
        بتيارها عند \qty{4.5}{V}، لتعرف المنضدة ما تتوقّع
        (برقم عشري واحد، \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- صور كاملة.}
\begin{enumerate}[resume]
  \item جدول العيّنة الثالثة: $(1.5, 0.05)$، و$(3.0, 0.10)$،
        و$(4.5, 0.15)$، و$(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{بالفولت} \omterm{def:g8:intensity-ammeter:intensity}{والأمبير}.
        لائم $R$، مستشهدًا بخاصّية الأعداد التي تجيز قيمة
        واحدة.
  \item جدول العيّنة الرابعة: $(1.5, 0.30)$، و$(3.0, 0.45)$،
        و$(4.5, 0.54)$، و$(6.0, 0.60)$. اكشف المحتال: فلا $R$
        واحدة، ثم بيّن اتجاه الانجراف.
  \item والعيّنة الرابعة هي في الحقيقة \omterm{def:g3:first-electric-circuit:bulb}{مصباح} صغير ذو خيط من
        خطّ الإنتاج المجاور. احكِ قصة انحنائها --- ولماذا
        يكون اختبار الخطّ المستقيم في المنضدة كاشفًا
        للمقاومات بالضبط.
  \item ويقول كتاب قواعد المنضدة: ``كل صورة يجب أن تُؤخذ عند
        \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} ثابتة.'' فأيّ بند من بنود قانون \omterm{def:g8:ohms-law:resistance}{أوم} تحمي
        هذه القاعدة؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- قسم التطبيقات.}
\begin{enumerate}[resume]
  \item يطلب طلب مقاومًا يحدّ التيار عند \qty{30}{mA} على مغذٍّ
        \qty{12}{V} (وحصة الحمل نفسه مهمَلة). احسب القيمة
        المطلوب تسعيرها.
  \item ويشكو زبون أن وشيعة مسخّنه $25\,\Omega$ ``لا'' تسحب
        إلا \qty{9.2}{A} بينما يسمح \omterm{def:g7:short-circuits-safety:fuse}{المصهر} بعشرة. احسب أيّ
        \omterm{def:g8:voltage-voltmeter:voltage}{توتر} تغذية تلمّح إليه شكواه، وطمئنه بالحساب.
  \item ويطلب مختبر النماذج أغلظ وصلة نحاسية وأقصرها في
        المصنع ``بأقرب ما يمكن إلى صفر \omterm{def:g8:ohms-law:resistance}{أوم}''. فلأيّ دور في
        الدارة وُلدت تلك الوصلة --- وقرب أيّ شرّير يومي تضعها
        مقاومتها شبه المعدومة إن أُسيء استعمالها؟
  \item أغلِق سجلّ الفحص: ثلاث جمل --- القانون، وصورته،
        والشرط الوحيد الذي يجوز لعنصر أن يوقّعه بموجبه.
\end{enumerate}
\end{problem}
