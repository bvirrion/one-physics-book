\chapter{قانونا التيار والتوتر في الدارات}\label{ch:g8:circuit-laws}

مقياسان، ومقداران --- والآن، أخيرًا، الدستور. فكل دارة ستبنيها
يومًا، من كشّاف يدوي إلى شبكة مدينة، تطيع أربعة قوانين قصيرة:
اثنان في محاسبة التيار، واثنان في محاسبة \omterm{def:g8:voltage-voltmeter:voltage}{التوتر}. وبها وبقليل
من الجبر تستطيع أن تحسب قراءات لم تقسها بعد --- وأن تمسك أيّ
\omterm{def:g6:measurement-in-science:measuring}{قياس} يكذب.

\section{قانونا التيار}

\begin{proposition}[قانون التيار في حلقة على التسلسل]\label{prop:g8:circuit-laws:seriesI}
على امتداد حلقة واحدة، \omterm{def:g8:intensity-ammeter:intensity}{الشدّة} هي نفسها في كل مكان:
\[
I_{\text{البطارية}} = I_{\text{المصباح 1}} = I_{\text{المصباح 2}} =
\dots
\]
طريق واحد، ومسير واحد --- نقاط مكس الفصل الماضي، وقد صارت
قانونًا من الدستور.
\end{proposition}

\begin{proposition}[قانون العقد]\label{prop:g8:circuit-laws:junction}
عند أيّ \omterm{def:g5:series-parallel-circuits:parallel}{عقدة}، تساوي الشدّات الداخلة مجتمعةً الشدّاتِ الخارجة
مجتمعةً: فلطريق رئيسي ينشطر إلى فرعين،
\[
I = I_1 + I_2 .
\]
فالشحنة لا تُخلق ولا تفنى ولا تُخزَّن عند مفترق: فما يصل
يغادر. (وهذا صحيح لأيّ عدد من الفروع --- اجمع الوافدين، واجمع
المغادرين، تتوازن الدفاتر.)
\end{proposition}

\section{قانونا التوتر}

\begin{proposition}[قانون التوتر في حلقة على التسلسل]\label{prop:g8:circuit-laws:seriesU}
حول حلقة \omterm{def:g4:series-circuits:series}{على التسلسل}، يُقتسم \omterm{def:g8:voltage-voltmeter:voltage}{توتر} \omterm{def:g3:first-electric-circuit:battery}{البطارية} بين الأجهزة، وتجتمع
الحصص بالضبط:
\[
U_{\text{البطارية}} = U_1 + U_2 + \dots
\]
وصورة الشلّال تمسك الدفاتر: \omterm{def:g3:first-electric-circuit:battery}{فالبطارية} ترفع المسير عبر ارتفاعها
كاملًا، والأجهزة تنفق ذلك الارتفاع فيما بينها، قطرة قطرة ---
والأسلاك الجيّدة لا تنفق شيئًا.
\end{proposition}

\begin{proposition}[قانون التوتر للفروع على التفرع]\label{prop:g8:circuit-laws:parallelU}
الأجهزة \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} تمتدّ كلها بين العقدتين نفسيهما --- فيخدمها
\omterm{def:g8:voltage-voltmeter:voltage}{توتر} واحد:
\[
U_1 = U_2 = U_{\text{بين العقدتين}} .
\]
فكل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} يتمتّع بالدفعة كاملة بين العقدتين --- ولهذا تسطع
المصابيح \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} بسطوعها الاسمي كاملًا، ولهذا يعرض كل مأخذ
في بيتك \qty{230}{V} نفسها.
\end{proposition}

\begin{example}[لماذا الحصص غير متساوية؟]\label{ex:g8:circuit-laws:unequal}
مصباحان مختلفان \omterm{def:g4:series-circuits:series}{على التسلسل} يحملان التيار نفسه (القانون
الأول) ومع ذلك يقتسمان \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} اقتسامًا غير متساوٍ ---
\qty{2.1}{V} في مقابل \qty{2.4}{V}. والحصة الأكبر تذهب إلى
\omterm{def:g3:first-electric-circuit:bulb}{المصباح} الذي يقاوم المسير أكثر: فالعامل الأشقّ يأخذ الهبوط
الأكبر. وأمّا ما تعنيه ``يقاوم أكثر'' بالضبط، وكيف تقيسه،
فموضوع الفصل التالي كله --- وقوانين هذا الفصل مسرحه.
\end{example}

\section{الحساب قبل القياس}

\begin{method}[حلّ دارة بالقوانين الأربعة]\label{met:g8:circuit-laws:solve}
لدارة معلوم بعض قراءاتها:
\begin{enumerate}
  \item ارسم المخطّط وعلّم كل $I$ وكل $U$ معلوم؛
  \item اكتب القوانين المنطبقة معادلاتٍ --- مجاميع العقد
        للتيارات، ومجاميع الحلقة للتوترات، وتساوي $I$ \omterm{def:g4:series-circuits:series}{على التسلسل}
        وتساوي $U$ \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}؛
  \item حُلّ للمجاهيل (فمعادلات منهج الجبر عندك تكسب قوتها
        هنا)؛
  \item تحقّق من المعقولية: فلا \omterm{def:g3:first-electric-circuit:bulb}{مصباح} ينال أكثر مما تعرضه
        \omterm{def:g3:first-electric-circuit:battery}{البطارية}، ولا تيار \omterm{def:g5:series-parallel-circuits:parallel}{فرع} يتجاوز تيار الطريق الرئيسي.
\end{enumerate}
\end{method}

\begin{example}[مثال محلول: الدارة المختلطة]\label{ex:g8:circuit-laws:worked}
\omterm{def:g3:first-electric-circuit:battery}{بطارية} \qty{6}{V} تغذّي \omterm{def:g3:first-electric-circuit:bulb}{المصباح} $A$ \omterm{def:g4:series-circuits:series}{على التسلسل} مع زوج \omterm{def:g5:series-parallel-circuits:parallel}{على
التفرع} ($B$ و$C$). والمقيس: $I_{\text{البطارية}} = \qty{0.5}{A}$،
و$I_B = \qty{0.3}{A}$، و$U_A = \qty{2.5}{V}$. احسب الباقي.
قانون العقد: $I_C = 0.5 - 0.3 = \qty{0.2}{A}$. وقانون التيار
\omterm{def:g4:series-circuits:series}{على التسلسل}: \omterm{def:g3:first-electric-circuit:bulb}{المصباح} $A$ يحمل \qty{0.5}{A} كاملة. وقانون \omterm{def:g8:voltage-voltmeter:voltage}{توتر}
الحلقة: ينال الزوج $U_{\text{الزوج}} = 6 - 2.5 = \qty{3.5}{V}$؛
وقانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}: $U_B = U_C = \qty{3.5}{V}$. فكل
قراءة مقياس في الدارة صارت معلومة --- أربعة قوانين، وسطر جبر
لكل واحد.
\end{example}

\begin{example}[القوانين كاشفةَ كذب]\label{ex:g8:circuit-laws:liedetector}
يزعم تقرير مختبر: \omterm{def:g3:first-electric-circuit:battery}{البطارية} \qty{4.5}{V}؛ ومصباحان \omterm{def:g4:series-circuits:series}{على التسلسل}
عند \qty{2.1}{V} و\qty{2.9}{V}. فقانون الحلقة يجمع الحصص:
\qty{5.0}{V} منفقة من كيس \qty{4.5}{V} --- محال؛ فقد أساء أحدهم
قراءة قرص. وتقرير آخر: \omterm{def:g5:series-parallel-circuits:parallel}{عقدة} يدخلها \qty{0.40}{A}، ويخرج من
فرعيها \qty{0.25}{A} و\qty{0.20}{A} --- فالمفترق ``يخلق''
\qty{0.05}{A}: مرفوض. وقبل القوانين كانت القراءات الخاطئة تبدو
حسنة كالصائبة؛ أمّا الآن فالدستور يدقّق كل صفحة.
\end{example}

\begin{remark}[البطاريات تحت القوانين نفسها]\label{rem:g8:circuit-laws:batteries}
قاعدة ``رأسًا بعقب'' هي قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g4:series-circuits:series}{على التسلسل} بأقدم ثيابه:
فالبطاريات \omterm{def:g4:series-circuits:series}{على التسلسل} تجمع توتراتها ($1.5 + 1.5 + 1.5 =
\qty{4.5}{V}$ --- سرّ \omterm{def:g3:first-electric-circuit:battery}{البطارية} المسطّحة)، والخلية المعكوسة
\emph{تطرح}، وقد أُنفقت دفعتها في مقاتلة الأخريات. فالقوانين
تحكم المعطين والمنفقين سواءً؛ ولا يختلف إلا إشارة إسهامهم.
\end{remark}

\section{مسألة: الصناديق المختومة الغامضة}

\begin{problem}[{مسألة نهاية الأسبوع --- تحدّي الصناديق
المختومة في نادي الإلكترونيات؛ أربعة قوانين في مواجهة أربعة
ألغاز؛ التدقيق الكبير}]\label{pb:g8:circuit-laws:1}
يختم النادي دارات بسيطة داخل صناديق، ولا يترك إلا مآخذ
للمقاييس. وعلى الفرق أن تستنتج الداخل --- والدستور في اليد.
\omterm{def:g3:first-electric-circuit:battery}{وبطارية} المنضدة \qty{6}{V} في كل مكان.

\medskip\noindent\textbf{القسم الأول --- الصندوق أ: مأخذان،
ومصباحان.}
تعترف اللافتة: مصباحان متماثلان ولا شيء غيرهما، إمّا \omterm{def:g4:series-circuits:series}{على
التسلسل} وإمّا \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}.
\begin{enumerate}
  \item موصولًا \omterm{def:g3:first-electric-circuit:battery}{بالبطارية}، يسحب الصندوق أ \qty{0.2}{A}؛
        ومعلوم أن مصباحًا واحدًا كهذا وحده على \qty{6}{V} يسحب
        نحو \qty{0.4}{A}. أعلى التسلسل هو أم \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}؟
        استدلّ بقانون تيار.
  \item تنبّأ بما كان الصندوق سيسحبه في الترتيب \emph{الآخر}.
  \item وأيّ \omterm{def:g8:voltage-voltmeter:voltage}{توتر} ينال كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح} داخلي في الترتيب الفعلي،
        وبأيّ قانون \omterm{def:g8:voltage-voltmeter:voltage}{توتر}؟
  \item وفي أيّ ترتيب كانت المصابيح المخفيّة ستسطع بسطوعها
        الاسمي كاملًا --- ولماذا لا تفعل ذلك، بشهادة العين من
        كوّة الصندوق؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الصندوق ب: لغز
المصابيح الثلاثة.}
يعترف الصندوق ب: \omterm{def:g3:first-electric-circuit:bulb}{مصباح} واحد ($X$) \omterm{def:g4:series-circuits:series}{على التسلسل} مع زوج \omterm{def:g5:series-parallel-circuits:parallel}{على
التفرع} ($Y$ و$Z$ --- وليسا متماثلين بالضرورة).
\begin{enumerate}[resume]
  \item يقيس الفريق $I_{\text{البطارية}} = \qty{0.5}{A}$،
        وعند مأخذ اختبار $Y$ نفسه $I_Y = \qty{0.35}{A}$. جِد
        $I_Z$ و$I_X$، مستشهدًا بالقوانين.
  \item مقياس \omterm{def:g8:voltage-voltmeter:voltage}{توتر} على طرفَي $X$: \qty{2.8}{V}. جِد \omterm{def:g8:voltage-voltmeter:voltage}{توتر} الزوج
        \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} --- وكلًّا من $U_Y$ و$U_Z$.
  \item $Y$ و$Z$ يحملان تيارين مختلفين عند \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} نفسه. فعلامَ
        يدلّ هذا في شأن المصباحين، بلغة
        \cref{ex:g8:circuit-laws:unequal}؟
  \item يفكّ الفريق $Y$ من فتحة. تنبّأ باتجاه تغيّر
        $I_{\text{البطارية}}$ الجديد (ارتفاع أم هبوط أم
        ثبات)، مستدلًّا بالطرق ومقاومتها.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الصندوق ج: الغشّ.}
\begin{enumerate}[resume]
  \item يبلّغ باني الصندوق ج: ``\omterm{def:g3:first-electric-circuit:battery}{البطارية} \qty{6}{V}؛
        ومصباحان \omterm{def:g4:series-circuits:series}{على التسلسل} في الداخل عند \qty{2.5}{V}
        و\qty{4.1}{V}؛ \omterm{def:g5:series-parallel-circuits:parallel}{وعقدة} في الداخل تستقبل \qty{0.3}{A}
        وترسل \qty{0.2}{A} و\qty{0.15}{A}.'' أدِن التقرير
        مرتين، بقانون لكل إدانة.
  \item ويتبيّن لاحقًا أن إحدى قراءتَي \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} المبلَّغ عنهما
        أمينة. فإذا كانت هي \qty{2.5}{V}، فماذا يجب أن تكون
        شريكتها حقًّا؟
  \item ويعترف الباني بأن أعداد \omterm{def:g5:series-parallel-circuits:parallel}{العقدة} ``قُرّبت بسخاء''. فإذا
        كانت قراءتا الفرعين \qty{0.2}{A} و\qty{0.15}{A}
        أمينتين، فماذا يجب أن يحمل الطريق الرئيسي حقًّا؟
  \item ولماذا تجعل القوانين الأربعة الغشّ في الصناديق
        المختومة لعبة خاسرة؟ جملة واحدة.
\end{enumerate}

\medskip\noindent\textbf{القسم الرابع --- الدستور.}
\begin{enumerate}[resume]
  \item اكتب ميثاق النادي: القوانين الأربعة في أربعة أسطر،
        ولكل واحد صورته اليومية التي تحمله (الطرق، ونقاط
        المكس، والشلّالات، والمفترقات).
\end{enumerate}
\end{problem}
