\chapter{سرعة الضوء؛ المسافات في الكون}\label{ch:g8:speed-of-light}

سبع سنوات وهذا الكتاب يهمس بأن \omterm{def:g1:light-and-shadows:source}{الضوء} ``كأنه لحظي'' --- الوميض
قبل الرعد، والضربة المرئية تشغّل ساعة الإيقاف. وهذا الفصل يفي
بأقدم وعود العائلة: \omterm{def:g7:motion-average-speed:formula}{فسرعة} \omterm{def:g1:light-and-shadows:source}{الضوء} ليست لانهائية، والبشر قاسوها
رغم كل شيء، والعدد من الضخامة بحيث يصير، في النفَس نفسه، مسطرة
علم الفلك وآلته الزمنية.

\section{القياس المستحيل}

\begin{example}[فانوسان على تلّين]\label{ex:g8:speed-of-light:galileo}
قبل أربعة قرون، أوقف رائد التجارب العظيم راصدين على تلّين
متباعدين ليلًا، مع كلٍّ فانوس بمصراع: اكشف فانوسك حين ترى
فانوسي، وينبغي أن يفضح تأخّر الذهاب والإياب \omterm{def:g7:motion-average-speed:formula}{سرعة} \omterm{def:g1:light-and-shadows:source}{الضوء}. لكن
التأخّر الذي وجداه لم يكن إلا ردود فعل بشرية؛ فعبور \omterm{def:g1:light-and-shadows:source}{الضوء}
الحقيقي بين التلّين استغرق أجزاءً من \emph{مليون} من الثانية.
فشلت التجربة فشلًا مجيدًا --- وقد أثبتت، على الأقلّ، أن \omterm{def:g1:light-and-shadows:source}{الضوء}
أسرع من أن توقّته قمّة تلّ، وتحدّت علم الفلك أن يجد مضامير
أطول.
\end{example}

\begin{example}[القمر الذي تأخّر]\label{ex:g8:speed-of-light:romer}
وجاء المضمار هديةً من المشتري. فقمره الكبير الأقرب إليه ينخسف
في \omterm{def:g1:light-and-shadows:shadow}{ظلّ} المشتري بجدول بالغ الانتظام --- ساعة في السماء. لكن
فلكيًّا صبورًا وجد الساعة \emph{تتأخّر} دقائق كثيرة في الشهور
التي كان \omterm{def:g6:sun-earth-moon:axisorbit}{مدار} الأرض قد حملنا فيها إلى الجانب البعيد من الشمس
عن المشتري، وتتقدّم ثانية كلما تأرجحنا قريبًا. وكانت قفزته:
\omterm{def:g2:moon-in-the-sky:moon}{القمر} منضبط --- بل \emph{\omterm{def:g1:light-and-shadows:source}{ضوءه}} هو المتأخّر، إذ يحتاج إلى زمن
إضافي ليعبر العرض الإضافي \omterm{def:g6:sun-earth-moon:axisorbit}{لمدار} الأرض. ومن التأخّر ومن \omterm{def:g6:mass-and-volume:volume}{حجم}
\omterm{def:g6:sun-earth-moon:axisorbit}{المدار} جاء أول تقدير \omterm{def:g7:motion-average-speed:formula}{لسرعة} \omterm{def:g1:light-and-shadows:source}{الضوء} --- قريبًا من الحقيقة قربًا
يحبس الأنفاس، بمقاييس القرن السابع عشر.
\end{example}

\begin{proposition}[سرعة الضوء]\label{prop:g8:speed-of-light:value}
في الخلاء --- وفي \omterm{def:g2:air-around-us:air}{الهواء}، بتقريب ممتاز --- يسافر \omterm{def:g1:light-and-shadows:source}{الضوء} \omterm{def:g7:motion-average-speed:formula}{بسرعة}
\[
c = \qty{300000}{km/s} = 3 \times 10^{8}\ \unit{m/s},
\]
أي ثلاثمئة ألف كيلومتر في كل ثانية: سبع لفّات ونصف حول الأرض
بين نبضتَي قلب. وكل لون من المزيج يسافر بهذه $c$ نفسها في
الخلاء؛ ولم يُوقَّت قطّ شيء يحمل مادة أو خبرًا أسرع منها. ثم
أكّد دهاء أرضي كلامَ الفلكيين لاحقًا --- \omterm{def:g6:light-propagation:ray}{حزمة} تقطّعها عجلة
مسنّنة دوّارة، تركض ثمانية كيلومترات إلى \omterm{def:g5:mirrors-reflection:reflection}{مرآة} وترجع عبر الفجوة
التالية بين السنّين --- وصقلت الآلات الحديثة $c$ حتى يقين تامّ
بحيث صار \omterm{def:g2:measuring-length:units}{المتر} نفسه يُعرَّف منها الآن.
\end{proposition}

\begin{example}[أزمنة السفر في الجوار]\label{ex:g8:speed-of-light:times}
بالعلاقة $t = d/c$ ينال النظام الشمسي جدول مواقيت. \omterm{def:g2:moon-in-the-sky:moon}{القمر}،
\qty{384000}{km}: $384000 \div 300000 \approx \qty{1.3}{s}$
--- ومحادثات الراديو بين الأرض \omterm{def:g2:moon-in-the-sky:moon}{والقمر} تحمل تلك الوقفة
المحرجة. والشمس، $150$ مليون كيلومتر: $150000000 \div 300000 =
\qty{500}{s}$ --- أي ثماني دقائق وعشرين ثانية: \omterm{def:g1:light-and-shadows:source}{فضوء} الشمس خبر
عمره ثماني دقائق دائمًا. والمرّيخ، عند مسافة معتادة قدرها
$2 \times 10^{8}$ كيلومتر: نحو إحدى عشرة دقيقة --- فسائقو
المركبات الجوّالة يرسلون خطة اليوم وينتظرون؛ ولا عصا توجيه
تقود عبر رحلة ذهاب وإياب مدّتها عشرون دقيقة.
\end{example}

\section{السنة الضوئية}

\begin{definition}[السنة الضوئية]\label{def:g8:speed-of-light:lightyear}
\emph{السنة الضوئية}\index{سنة ضوئية} \emph{مسافة}: هي
المسافة التي يقطعها \omterm{def:g1:light-and-shadows:source}{الضوء} في سنة واحدة --- نحو
$9.5 \times 10^{12}$ كيلومتر، أي قرابة عشرة ملايين مليون
كيلومتر. والاسم يضلّل المتعجّل (فهي ليست زمنًا، كما أن
``الخطوة'' ليست قدمًا)؛ والفلكيون يستعملونها لأن الكيلومترات
تنهار تحت مقياس السماء الحقيقي.
\end{definition}

\begin{example}[الجوار بالسنوات الضوئية]\label{ex:g8:speed-of-light:ladder}
أقرب \omterm{def:g4:solar-system:star}{نجم} بعد الشمس: نحو $4$ سنوات ضوئية --- ففوتونات الليلة
غادرته وأنت في فصولك الفيزيائية الأولى. ونجوم سماء الشتاء
الساطعة: عشرات إلى مئات السنوات الضوئية. \omterm{ex:g5:stars-night-sky:milkyway}{ودرب التبّانة}، مدينة
نجومنا: نحو $100000$ \omterm{def:g8:speed-of-light:lightyear}{سنة ضوئية} عرضًا. وأقرب مجرّة كبيرة
مجاورة: نحو $2.5$ مليون \omterm{def:g8:speed-of-light:lightyear}{سنة ضوئية} --- ولطخة ضوئها الخافتة،
المرئية لعين مجرّدة تكيّفت مع الظلام، هي أقدم \omterm{def:g1:light-and-shadows:source}{ضوء} يستطيع إنسان
أن يراه بلا عون.
\end{example}

\begin{proposition}[المقاريب آلات زمنية]\label{prop:g8:speed-of-light:timemachine}
لأن أخبار \omterm{def:g1:light-and-shadows:source}{الضوء} تسافر \omterm{def:g7:motion-average-speed:formula}{بسرعة} $c$ ولأن الكون شاسع، فإن
\emph{النظر بعيدًا نظرٌ إلى الوراء}. \omterm{def:g4:solar-system:star}{فالنجم} على $4$ سنوات
ضوئية يُري نفسه كما كان قبل $4$ سنوات؛ والمجرّة المجاورة، كما
كانت يوم نحت أسلافنا أول أدواتهم الحجرية. وشكّ راصد النجوم في
الصف الخامس صار الآن مبرهنة حسابية: فكل مقراب مصوَّب إلى
الخارج مصوَّب إلى الماضي، وعلم الفلك تاريخ يُقرأ \omterm{def:g1:light-and-shadows:source}{بضوء} النجوم.
\end{proposition}

\begin{method}[تحويل المسافة وعمق النظر]\label{met:g8:speed-of-light:convert}
\begin{enumerate}
  \item المسافة بالسنوات الضوئية $\to$ عمق النظر بالسنين:
        العدد نفسه --- وتلك عبقرية \omterm{def:g6:measurement-in-science:measuring}{الوحدة} كلها؛
  \item المسافة بالكيلومترات $\to$ زمن \omterm{def:g1:light-and-shadows:source}{الضوء}: اقسم على
        \qty{300000}{km/s}، ثم روّض الثواني إلى دقائق أو
        ساعات أو سنوات؛
  \item السنوات الضوئية $\to$ الكيلومترات (ونادرًا ما يستحقّ
        الأمر): اضرب في $9.5 \times 10^{12}$ --- وتذكّر لماذا
        قلّما يفعل الفلكيون ذلك.
\end{enumerate}
\end{method}

\begin{remark}[ماذا تعني ``الآن'' هناك]\label{rem:g8:speed-of-light:now}
أما زال \omterm{def:g4:solar-system:star}{النجم} البعيد أربع سنوات ضوئية يسطع \emph{الآن}؟ على
الأرجح --- فالنجوم تعمّر طويلًا --- لكن لا جواب أسرع من ضوئه
هو يستطيع أن يبلغنا يومًا: فأنضر خبر ممكن قديم بأربع سنوات،
دائمًا. وأمّا في عمق السماء فيلين السؤال إلى غرابة: فبعض أخفت
اللطخات في المقاريب الكبيرة مجرّات غادر ضوؤها قبل أن توجد
الأرض. وستشحذ الفيزياء ما يمكن أن تعنيه ``الآن عن بُعد'' أصلًا
في سنوات الجامعة؛ وأمّا هذه السنة فاحمل الصيغة الأمينة:
السماء ليست مشهدًا --- بل أرشيف.
\end{remark}

\section{مسألة: مركز التحكّم}

\begin{problem}[{مسألة نهاية الأسبوع --- نوبة في مركز التحكّم
بالفضاء السحيق؛ محادثات بتأخير الضوء من القمر إلى النجوم؛
دليل العمليات}]\label{pb:g8:speed-of-light:1}
الليلة تدير مكتب الاتّصالات: كل رسالة تسافر \omterm{def:g7:motion-average-speed:formula}{بسرعة}
$c = \qty{300000}{km/s}$، وتأخير \omterm{def:g1:light-and-shadows:source}{الضوء} يحكم كل قرار.

\medskip\noindent\textbf{القسم الأول --- العمليات القريبة.}
\begin{enumerate}
  \item القاعدة القمرية (\qty{384000}{km}) تطلب تأكيد إنزال
        إمدادات. بعد كم من قولك ``مؤكَّد'' تسمعه القاعدة، وما
        أقرب وقت تستطيع فيه سماع إقرارها؟
  \item \omterm{def:g2:moon-in-the-sky:moon}{قمر} صناعي في \omterm{def:g6:sun-earth-moon:axisorbit}{مدار} عالٍ على \qty{36000}{km} يرحّل بثًّا
        تلفازيًّا. احسب تأخيره في اتجاه واحد --- وقل لماذا
        تحمل المكالمات الهاتفية العابرة للقارّات عبر أقمار
        كهذه ترددًا يكاد يُلحظ (ذهابًا وإيابًا صعودًا
        وهبوطًا).
  \item يبلّغ المرصد الشمسي عن توهّج عظيم ``بينما هو يحدث''.
        صحّح عبارة المشغّل: متى اندلع التوهّج حقًّا؟
  \item ويقتضي البروتوكول أن تحسب الساعات كلها حساب التأخير.
        فأيّ وصلة قريبة من الأرض --- \omterm{def:g2:moon-in-the-sky:moon}{القمر} أم \omterm{def:g2:moon-in-the-sky:moon}{القمر} الصناعي
        العالي --- تحتاج إلى الهامش الأكبر، وبأيّ عامل
        تقريبًا؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- نافذة المرّيخ.}
يقف المرّيخ الليلة على $1.8 \times 10^{8}$ كيلومتر.
\begin{enumerate}[resume]
  \item احسب تأخير الاتجاه الواحد بالثواني، ثم بالدقائق.
  \item تلقى المركبة الجوّالة صدعًا غير متوقّع، وآلات تصويرها
        تُظهره مقتربًا \omterm{def:g7:motion-average-speed:formula}{بسرعة} المشي. اشرح، برقم الذهاب
        والإياب، لماذا يكون الإنقاذ بعصا توجيه ميؤوسًا منه ---
        وما الذي يجب أن تحمله المركبة على متنها بناءً على
        ذلك.
  \item يستغرق رفع خطة المساء \qty{3}{min} من الإرسال زائد
        التأخير. فمتى يجب أن يبدأ الرفع ليصل كاملًا إلى
        المركبة عند الساعة 21:00 بتوقيت التحكّم؟
  \item وبعد ستّة أشهر سيقف المرّيخ على $3.8 \times 10^{8}$
        كيلومتر. فماذا يصير التأخير، وما الذي يعلّمه هذا في
        شأن ``المسافة'' إلى \omterm{def:g4:solar-system:planet}{كوكب}؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- المكتب البعيد.}
\begin{enumerate}[resume]
  \item بلغ مسبار بين النجوم مسافة \emph{يوم} ضوئي كامل من
        الأرض. عبّر عن تلك المسافة بالكيلومترات (بقوى العشرة؛
        واليوم يحمل $86400$ ثانية)، واذكر زمن المحادثة ذهابًا
        وإيابًا مع المسبار.
  \item ويصوغ الحالمون في المكتب الخلفي ``رسالة إلى أقرب
        \omterm{def:g4:solar-system:star}{نجم}''. صُغ الجدول الزمني بدل ذلك: أُرسلت الليلة، فمتى
        أقرب وقت يصل فيه أيّ ردّ؟
  \item ويُظهر جدار الأرشيف صورة الليلة العميقة للمجرّة
        المجاورة، على بعد $2.5$ مليون \omterm{def:g8:speed-of-light:lightyear}{سنة ضوئية}. أضف الزمن
        الأمين إلى التعليق --- أيّ عصر من ماضي الأرض تبثّه
        هذه اللقطة راجعةً، لو كان هناك من ينظر؟
  \item سجلّ نهاية النوبة: اكتب قواعد المكتب الثلاث ---
        واحدة عن اللحظية (وأنه لا شيء لحظي)، وواحدة عن عصيّ
        التوجيه والتأخير، وواحدة عن المقاريب والأرشيف.
\end{enumerate}
\end{problem}
