\chapter{شدّة التيار ومقياس التيار}\label{ch:g8:intensity-ammeter}

``تيار قوي، وتيار ضعيف'' --- هكذا تكلّمنا سنوات، نجرّع السطوع
والقبضة بالحدس. واليوم يتقاعد الحدس. فتنال \omterm{def:g2:pushes-and-pulls:force}{قوة} التيار اسمًا
\omterm{def:g6:measurement-in-science:measuring}{ووحدة} وآلة؛ وأول ما تثبته الآلة قانون طالما ارتبته ولم تقسه
مرة واحدة.

\section{الشدّة ووحدتها}

\begin{definition}[الشدّة]\label{def:g8:intensity-ammeter:intensity}
\emph{شدّة}\index{شدّة} \omterm{def:g7:circuits-series-parallel:current}{التيار الكهربائي} $I$ \omterm{def:g6:measurement-in-science:measuring}{تقيس} كم يبلغ
المسير من \omterm{def:g2:pushes-and-pulls:force}{القوة}: أي كم من الشحنة الكهربائية يستعرض أمام نقطة من
الدارة في كل ثانية --- معدّل جريان التيار، كما أن معدّل النهر
ماء في الثانية. ووحدتها \emph{الأمبير}\index{أمبير} (\unit{A})،
تكريمًا لرائد عظيم من روّاد قوانين الكهرباء؛ ودارات الحياة
اليومية تنطق كثيرًا \emph{بالملّي أمبير}: $\qty{1}{mA} =
\qty{0.001}{A}$، أي جزء من ألف.
\end{definition}

\begin{example}[أمبيرات تُعرف بالنظر]\label{ex:g8:intensity-ammeter:values}
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} مؤشّر صغير يرتشف نحو \qty{0.02}{A} --- أي \qty{20}{mA}؛
\omterm{def:g3:first-electric-circuit:bulb}{ومصباح} كشّاف يدوي نحو \qty{0.3}{A}؛ \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} مصدح سيارة الكبير
\qty{5}{A}؛ وغلّاية بكامل هديرها \qty{10}{A}؛ ومحرّك سيارة عند
الإقلاع يجرع \qty{100}{A} من بطاريته؛ وصاعقة برق، في ملّي
ثانيتها، تدكّ عشرات الآلاف من الأمبيرات في قناتها. وسلّم الخطر
يبعث على الرصانة: فتيارات مقدارها أجزاء من مئة من \omterm{def:g8:intensity-ammeter:intensity}{الأمبير} عبر
جسم إنسان قد تكون قاتلة --- فالاحترام يبدأ دون \omterm{def:g8:intensity-ammeter:intensity}{الأمبير} الواحد
بكثير.
\end{example}

\begin{remark}[الشخص والوحدة]\label{rem:g8:intensity-ammeter:person}
تُكتب \omterm{def:g6:measurement-in-science:measuring}{الوحدة} اسمًا عامًّا --- ``تيار شدّته أمبيران'' --- بينما
تسمّي الكلمة نفسها بحرفها الكبير في لغات لاتينية ذلك الشخص، كما
في كل \omterm{def:g6:measurement-in-science:measuring}{وحدة} تكرّم عالِمًا. أمّا الرمز \unit{A} فيبقى بحرف كبير.
وستلقى هذا العرف ثانية عمّا قريب، مرتين.
\end{remark}

\section{مقياس التيار}

\begin{definition}[مقياس التيار]\label{def:g8:intensity-ammeter:ammeter}
\emph{مقياس التيار}\index{مقياس التيار} (الأمبيرمتر) يقيس
\omterm{def:g8:intensity-ammeter:intensity}{الشدّة}. وهو يعدّ المسير المارّ \emph{عبر نفسه} --- فيجب أن
يُدخَل \emph{\omterm{def:g4:series-circuits:series}{على التسلسل}}، في الحلقة التي يقيسها بعينها، كنقطة
مكس على الطريق. وهو مبنيّ ليزعج المسير أقلّ ما يمكن، فلا يكاد
يقاوم التيار البتّة --- ولهذا أيضًا يكون مقياس تيار موصول خطأً
مباشرة على طرفَي \omterm{def:g3:first-electric-circuit:battery}{بطارية} دارةً قصيرة ذات قرص: وهي الطريقة
الكلاسيكية لإتلاف واحد.
\end{definition}

\begin{method}[قياس شدّة]\label{met:g8:intensity-ammeter:measure}
\begin{enumerate}
  \item أطفئ الدارة؛ ثم \emph{اقطع} الحلقة حيث يُراد عدّ
        التيار؛
  \item أدخل \omterm{def:g8:intensity-ammeter:ammeter}{مقياس التيار} في الفجوة: فيدخل التيار من مأخذه
        \unit{A}، ويخرج من مأخذه \emph{المشترك}، حتى يدخل
        الاتجاه الاصطلاحي من باب المقياس الأمامي؛
  \item وعلى مقياس متعدّد المجالات، ابدأ بالمجال \emph{الأكبر}،
        ثم انزل درجة درجة حتى تستقرّ القراءة مريحة على القرص
        --- حمايةً للمقياس من المفاجآت؛
  \item أغلق \omterm{ex:g3:first-electric-circuit:switch}{القاطعة}، واقرأ --- واكتب \omterm{def:g6:measurement-in-science:measuring}{الوحدة}.
\end{enumerate}
والوصل بالمقلوب على مقياس ذي إبرة يصفع الإبرة على حاجزها؛ أما
المقاييس الرقمية فلا تُظهر إلا إشارة ناقص، فتغفر للطلبة كل
يوم.
\end{method}

\section{أول قانون مقيس}

\begin{proposition}[حلقة واحدة، وشدّة واحدة]\label{prop:g8:intensity-ammeter:sameI}
في \omterm{def:g4:series-circuits:series}{الدارة على التسلسل}، \omterm{def:g8:intensity-ammeter:intensity}{الشدّة} واحدة عند كل نقطة من الحلقة.
انقل \omterm{def:g8:intensity-ammeter:ammeter}{مقياس التيار} إلى أيّ مكان --- قبل \omterm{def:g3:first-electric-circuit:bulb}{المصباح}، أو بعده، أو
بجانب \omterm{def:g3:first-electric-circuit:battery}{البطارية}: لا تتغيّر القراءة. فما لمّحت إليه خمس سنوات من
مصابيح متساوية السطوع تقوله الآلة الآن بالأعداد: لا يُستهلك شيء
في الطريق؛ طريق واحد، ومسير واحد، و$I$ واحد.
\end{proposition}

\begin{example}[قراءة الحروف الدقيقة في القانون]\label{ex:g8:intensity-ammeter:fineprint}
القانون يقول \emph{على امتداد حلقة واحدة}، لا \emph{مهما فعلت
بالحلقة}. أضف مصباحًا ثانيًا \omterm{def:g4:series-circuits:series}{على التسلسل} يضعف المسير في كل مكان
دفعة واحدة --- فلعلّ \qty{0.30}{A} تهبط إلى \qty{0.17}{A} ---
لكنه يضعف \emph{بالتساوي} في كل مكان: فنقاط المكس الثلاث تتّفق
على العدد الجديد أيضًا. حلقة واحدة، \omterm{def:g8:intensity-ammeter:intensity}{وشدّة} واحدة --- مهما تكن
تلك \omterm{def:g8:intensity-ammeter:intensity}{الشدّة} الآن.
\end{example}

\begin{example}[نظرة إلى العقدة]\label{ex:g8:intensity-ammeter:junction}
انصب نقاط المكس على زوج \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} بدل ذلك: الطريق الرئيسي
\qty{0.50}{A}؛ \omterm{def:g5:series-parallel-circuits:parallel}{والفرع} الأول \qty{0.30}{A}؛ \omterm{def:g5:series-parallel-circuits:parallel}{والفرع} الثاني
\qty{0.20}{A}. وليس المجموع الموحي بمصادفة --- فالمسايير
الداخلة إلى \omterm{def:g5:series-parallel-circuits:parallel}{عقدة} والخارجة منها يجب أن تتوازن، إذ لا تُخلق
الشحنة ولا تفنى عند مفترق الطريق. أمّا القانون كاملًا، ومعه
شريكه الخاصّ بالتوتر، فيعقد مجلسه في الفصل بعد التالي؛ وقد
لقيته مقاييسك سلفًا.
\end{example}

\begin{remark}[المقياس المتعدّد]\label{rem:g8:intensity-ammeter:multimeter}
طوبة الورشة الصفراء --- \emph{المقياس المتعدّد} --- تطوي \omterm{def:g8:intensity-ammeter:ammeter}{مقياس
التيار} وعدّة آلات أخرى في علبة واحدة ذات قرص اختيار. أدر القرص
إلى \unit{A} أو \unit{mA}، وضع السلكين في المأخذين المعلّمين،
فإذا هو مقياس تيارك، بقواعد التسلسل وكل شيء. وفصلان من هذا
الكتاب يسكنان فيه سلفًا؛ وثالث ينتقل إليه في الفصل التالي ---
بقواعد وصل \emph{معاكسة}، فانتبه.
\end{remark}

\section{مسألة: منضدة القياس}

\begin{problem}[{مسألة نهاية الأسبوع --- يوم التأهيل في ورشة
الإلكترونيات؛ أربع محطّات، ومقياس تيار لكل واحدة؛ رخصة
المتدرّب}]\label{pb:g8:intensity-ammeter:1}
لينال المتدرّب امتيازات الورشة، يجول على أربع محطّات \omterm{def:g6:measurement-in-science:measuring}{قياس}.
واصحب الفصل معك.

\medskip\noindent\textbf{القسم الأول --- المحطة الأولى: الحلقة
البسيطة.}
\omterm{def:g3:first-electric-circuit:battery}{بطارية}، وقاطعة، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} واحد.
\begin{enumerate}
  \item صِف، خطوة خطوة، إدخال \omterm{def:g8:intensity-ammeter:ammeter}{مقياس التيار} \omterm{def:g6:measurement-in-science:measuring}{لقياس} تيار
        \omterm{def:g3:first-electric-circuit:bulb}{المصباح}.
  \item القراءة: \qty{280}{mA}. عبّر عنها \omterm{def:g8:intensity-ammeter:intensity}{بالأمبير}.
  \item ينقل الممتحِن مقياسك إلى الجهة الأخرى من \omterm{def:g3:first-electric-circuit:battery}{البطارية}.
        تنبّأ بالقراءة، مستشهدًا بالقانون.
  \item ``وماذا لو كان في هذه الحلقة مقياسا تيار معًا؟'' أين
        يجلسان، وماذا سيُظهران؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- المحطة الثانية: زوج
\omterm{def:g4:series-circuits:series}{على التسلسل}.}
مصباحان مختلفان، \omterm{def:g3:first-electric-circuit:bulb}{والمصباح} أ أسطع من ب، \omterm{def:g4:series-circuits:series}{على التسلسل}.
\begin{enumerate}[resume]
  \item يتنبّأ متدرّب: ``\omterm{def:g3:first-electric-circuit:bulb}{المصباح} أ الأسطع يحمل تيارًا أكبر من ب
        الخافت.'' اختبر التنبّؤ بالقانون --- ووفّق بين تساوي
        التيارين واختلاف السطوع (ما الذي يجب أن يختلف بين
        المصباحين؟).
  \item يقرأ المقياس \qty{150}{mA} بين أ وب. اكتب كل تيار آخر
        في الحلقة.
  \item يُقصَر \omterm{def:g3:first-electric-circuit:bulb}{المصباح} ب بسلك ذي ملقطين. تنبّأ بالقراءة
        الجديدة وبسلوك المصباحين.
  \item يطلب الممتحِن جملة واحدة عن سبب منع الورشة أسلاك
        الملاقط على طرفَي \emph{\omterm{def:g3:first-electric-circuit:battery}{البطارية}} مع سماحها بالعرض على
        طرفَي ب.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- المحطة الثالثة:
\omterm{def:g5:series-parallel-circuits:parallel}{العقدة}.}
\omterm{def:g5:series-parallel-circuits:parallel}{فرع} \omterm{def:g3:first-electric-circuit:bulb}{مصباح} \omterm{def:g5:series-parallel-circuits:parallel}{وفرع} محرّك \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}.
\begin{enumerate}[resume]
  \item الطريق الرئيسي: \qty{0.52}{A}؛ \omterm{def:g5:series-parallel-circuits:parallel}{وفرع} \omterm{def:g3:first-electric-circuit:bulb}{المصباح}:
        \qty{0.30}{A}. فما تيار \omterm{def:g5:series-parallel-circuits:parallel}{فرع} \omterm{ex:g6:circuits-and-safety:motor}{المحرّك}، بالتوازن؟
  \item يعلق \omterm{ex:g6:circuits-and-safety:motor}{المحرّك} (وهو موصول لكنه لا يدور) فيرتقي فرعه إلى
        \qty{0.40}{A}. وبفرض بقاء \omterm{def:g5:series-parallel-circuits:parallel}{فرع} \omterm{def:g3:first-electric-circuit:bulb}{المصباح} على
        \qty{0.30}{A}، ماذا يحمل الطريق الرئيسي الآن --- وأيّ

  \item يُفكّ \omterm{def:g3:first-electric-circuit:bulb}{المصباح}. تنبّأ بقراءات نقاط المكس الثلاث.
  \item ولماذا تنفد \omterm{def:g3:first-electric-circuit:battery}{البطارية} أسرع ما تكون والفرعان سليمان ---
