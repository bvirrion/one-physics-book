\chapter{Spanning en de voltmeter}\label{ch:g8:voltage-voltmeter}

Sinds de eerste \omterm{def:g3:first-electric-circuit:battery}{batterij} het eerste \omterm{def:g3:first-electric-circuit:bulb}{lampje} liet
branden, doet één woord in dit boek zware, vage dienst: de ``duw''.
\omterm{def:g3:first-electric-circuit:battery}{Batterijen} duwen zwak, het net duwt moorddadig, twee \omterm{def:g3:first-electric-circuit:battery}{batterijen} achter
elkaar duwen samen. De duw wordt nu een gemeten grootheid met een
beroemde \omterm{def:g6:measurement-in-science:measuring}{eenheid} --- en haar instrument weigert, anders dan dat
van het vorige hoofdstuk, op de weg te gaan staan.

\section{Spanning}

\begin{definition}[Spanning]\label{def:g8:voltage-voltmeter:voltage}
De \emph{spanning}\index{spanning} $U$ tussen twee punten van een
schakeling meet de elektrische duw die daartussen beschikbaar is
--- hoe hard de stroom van het ene punt naar het andere wordt gedreven,
zoals de hoogte van een waterval meet hoe hard die het molenrad
aandrijft. Spanning is altijd een \emph{tussen}: tussen de twee
klemmen van een \omterm{def:g3:first-electric-circuit:battery}{batterij}, tussen de twee kanten van een
lamp --- nooit in één punt alleen. Haar \omterm{def:g6:measurement-in-science:measuring}{eenheid} is de
\emph{volt}\index{volt} (\unit{V}), ter ere van de uitvinder van de
eerste \omterm{def:g3:first-electric-circuit:battery}{batterij}.
\end{definition}

\begin{example}[Het familiealbum van de volt]\label{ex:g8:voltage-voltmeter:values}
Eén ronde cel: \qty{1.5}{V}. De platte zakbatterij: \qty{4.5}{V} ---
drie cellen achter elkaar in één omhulsel. De rechthoekige
negenvolt met beide klemmen bovenop: \qty{9}{V}, zes
verborgen cellen. Een autoaccu: \qty{12}{V}. Het stopcontact:
\qty{230}{V} --- meer dan honderdvijftig kleine cellen aan duw, en dat
is de hele kwantitatieve inhoud van ``het stopcontact, nooit''.
Bliksem tussen wolk en grond: honderden miljoenen
\omterm{def:g8:voltage-voltmeter:voltage}{volt}, heel even.
\end{example}

\begin{proposition}[Spanning zonder stroom]\label{prop:g8:voltage-voltmeter:nocurrent}
\omterm{def:g8:voltage-voltmeter:voltage}{Spanning} en stroom zijn verschillende grootheden, en de
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} komt eerst. Een \omterm{def:g3:first-electric-circuit:battery}{batterij} die alleen in een la
ligt, aan niets aangesloten, drijft helemaal geen stroom aan --- en
toch staat tussen haar klemmen de volle \qty{1.5}{V} klaar,
meetbaar, als de hoogte van een waterval op een dag dat de molen
gesloten is. Duw kan bestaan zonder mars; de mars bestaat nooit zonder
duw.
\end{proposition}

\begin{definition}[Voltmeter]\label{def:g8:voltage-voltmeter:voltmeter}
Een \emph{voltmeter}\index{voltmeter} meet de
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} tussen twee punten --- dus wordt hij \emph{over} die
punten heen aangesloten: \emph{\omterm{def:g5:series-parallel-circuits:parallel}{parallel}} aan het onderdeel dat je
interesseert, één snoer aan elke kant, als een landmeter die een
waterpas tussen twee hoogtes houdt. Hij is in elk opzicht tegengesteld
aan de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} gebouwd en verzet zich vrijwel
\emph{volledig} tegen de stroom: waar je hem ook overheen zet, hij
trekt er bijna geen --- dus verstoort het aanklemmen niets, en hoeft er
nooit een lus te worden verbroken.
\end{definition}

\end{omfigure}

\begin{method}[Een spanning meten]\label{met:g8:voltage-voltmeter:measure}
\begin{enumerate}
  \item laat de schakeling intact --- voor een \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} wordt
        niets verbroken;
  \item klem één snoer aan elke kant van het onderdeel: het
        \unit{V}-snoer waar de conventionele stroom binnenkomt, het
        gemeenschappelijke snoer waar hij vertrekt;
  \item eerst het grootste bereik, dan omlaag stappen tot het prettig
        afleest;
  \item lees af, en schrijf de \omterm{def:g6:measurement-in-science:measuring}{eenheid} erbij.
\end{enumerate}
De manieren van de twee \omterm{def:g2:measuring-length:units}{meters}, naast elkaar:
\omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} --- verbreek de weg, ga erin staan, verzet je
nergens tegen; \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} --- verbreek niets, blijf ernaast staan,
verzet je overal tegen. De twee manieren verwarren is de duurste
gewoonte van de werkplaats.
\end{method}

\section{Nominale spanningen}

\begin{definition}[Nominale spanning]\label{def:g8:voltage-voltmeter:nominal}
Elke lamp, \omterm{ex:g6:circuits-and-safety:motor}{motor} en elk apparaat draagt een \emph{nominale
spanning}\index{nominale spanning} --- de duw waarvoor het gebouwd is,
in de voet of op het plaatje gestempeld: een zaklamplampje
``3,5 V'', een autokoplamp ``12 V'', een waterkoker ``230 V''. Op zijn
nominale \omterm{def:g8:voltage-voltmeter:voltage}{spanning} presteert het apparaat zoals ontworpen; op veel
minder mokt het --- doffe lamp, luie \omterm{ex:g6:circuits-and-safety:motor}{motor}; op veel meer brandt
het door --- schitterend, eenmalig.
\end{definition}

\begin{example}[Lamp bij batterij zoeken]\label{ex:g8:voltage-voltmeter:matching}
Een ``3,5 V''-lampje op de platte \qty{4.5}{V}-batterij: licht
overbelast --- helder, en kortlevend. Op één \qty{1.5}{V}-cel: een
slaperige oranje gloed. Op de \qty{9}{V}-batterij: één
verblindend ogenblik, dan breekt de dunne draad --- het overlijdensbericht
van de gloeidraad. Het etiket is een contract: eer het, en de lamp
leeft haar geadverteerde leven.
\end{example}

\begin{remark}[Waarom 230 V doodt]\label{rem:g8:voltage-voltmeter:mains}
Nu krijgt de oude absolute regel haar getal. De vochtige geleidende weg
van het lichaam laat bij \qty{4.5}{V} een stroom door die te klein is
om te voelen. Dezelfde weg bij \qty{230}{V} --- vijftig keer de duw ---
laat vijftig keer de stroom door, ver voorbij de enkele honderdsten van
een \omterm{def:g8:intensity-ammeter:intensity}{ampère} die een hart stilzetten. Het stopcontact is niet
``een beetje'' sterker dan je batterijtafel: het ligt er twee ordes van
grootte boven, en de strengheid van het reglement is doodgewone
rekenkunde.
\end{remark}

\section{Eerste spanningsaflezingen}

\begin{example}[Rond een eenvoudige lus]\label{ex:g8:voltage-voltmeter:readings}
Eén \omterm{def:g3:first-electric-circuit:battery}{batterij}, één lamp, \omterm{def:g2:measuring-length:units}{meters} in de hand. Over de
\omterm{def:g3:first-electric-circuit:battery}{batterij}: \qty{4.5}{V}. Over de lamp: \qty{4.5}{V} --- de lamp
ontvangt wat de \omterm{def:g3:first-electric-circuit:battery}{batterij} aanbiedt. Over een stuk
verbindings\emph{draad}: \qty{0}{V}, of zo dicht daarbij als de \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter}
kan zeggen --- een goede draad is een vlakke weg, die geen duw uitgeeft.
De duw woont over de \emph{werkers}, niet langs de wegen: een aflezing
die het onthouden waard is voor wanneer de wetten van het volgende
hoofdstuk haar officieel maken.
\end{example}

\begin{example}[De bekentenis van de open schakelaar]\label{ex:g8:voltage-voltmeter:openswitch}
Lamp donker, \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} open --- waar is de duw gebleven? Meet:
over de donkere lamp \qty{0}{V}; over de \emph{open
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}} de volle \qty{4.5}{V}. De duw van de
\omterm{def:g3:first-electric-circuit:battery}{batterij} staat in zijn geheel over de opening te wachten --- de
enige plek in de lus waar twee punten elektrisch echt ``ver uit elkaar''
liggen. Sluit de \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} en de aflezingen wisselen van plaats.
Een open \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} is waar de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} wacht.
\end{example}

\section{Probleem: de inventarisatie van de batterijla}

\begin{problem}[{Weekendprobleem --- de grote inventarisatie van de
batterijla; een voltmeter, een schoenendoos vol cellen en de
apparatenplank van het gezin}]\label{pb:g8:voltage-voltmeter:1}
Elk huis heeft die la: losse cellen, mysterieuze \omterm{def:g3:first-electric-circuit:battery}{batterijen}, apparaten
die ``nieuwe nodig hebben''. Inventarisatiedag ---
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} in de hand.

\medskip\noindent\textbf{Deel I --- De cellen sorteren.}
\begin{enumerate}
  \item Beschrijf de juiste aansluiting van de
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} om een losse \qty{1.5}{V}-cel te
        controleren --- en waarom er eerst geen schakeling hoeft te
        worden gebouwd.
  \item De aflezingen van de inventarisatie, vers uit de la:
        \qty{1.58}{V}, \qty{1.32}{V}, \qty{0.9}{V}, \qty{0.1}{V}.
        Sorteer ze: vers, bruikbaar, moe, dood.
  \item Een negenvoltbatterij leest \qty{8.9}{V}. Hoeveel
        draagt elk van haar zes binnencellen ruwweg bij?
  \item Waarom toont een bijna dode cel nog \emph{enige}
        \omterm{def:g8:voltage-voltmeter:voltage}{spanning} hoewel ze niets meer kan laten
        branden? (Haar duw overleeft zwakjes; waartoe mist ze
        de \omterm{def:g2:pushes-and-pulls:force}{kracht}?)
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De apparatenplank.}
\begin{enumerate}[resume]
  \item De camera wil twee cellen achter elkaar. Welke
        \omterm{def:g8:voltage-voltmeter:voltage}{spanning} verwacht hij, en wat zal het paar overlevers van
        \qty{1.32}{V} uit vraag 2 leveren?
  \item De oude radio neemt vier cellen achter elkaar en mokt onder
        \qty{5}{V}. Welke cellen uit de la kunnen hem bemannen, als
        er al zijn?
  \item Kleine broer propt drie cellen in de tweecelszaklamp ``voor
        extra helderheid'' en wringt de derde over de veer. Het
        \omterm{def:g3:first-electric-circuit:bulb}{lampje} is voor \qty{3}{V} bedoeld. Voorspel de
        voorstelling, in de taal van de
        \omterm{def:g8:voltage-voltmeter:nominal}{nominale spanning}.
  \item Op het plaatje van de deurbeltransformator staat
        ``\qty{8}{V}''. Iemand stelt voor de deurbel rechtstreeks aan
        een negenvoltbatterij te hangen ``want acht is
        bijna negen''. Beoordeel: eerlijke match of langzame moord?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De wetten van de inventarisatie.}
\begin{enumerate}[resume]
  \item In de zaklamp (\omterm{def:g3:first-electric-circuit:bulb}{lampje} en cellen \omterm{def:g4:series-circuits:series}{in serie},
        \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} open) meet de inventariseerder: over
        de open \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} \qty{3.1}{V}; over het
        \omterm{def:g3:first-electric-circuit:bulb}{lampje} \qty{0}{V}. Verklaar beide, en voorspel beide na
        de klik.
  \item Over elk stuk bedrading van de zaklamp zegt de \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter}
        \qty{0}{V}, of hij nu brandt of donker is. Wat verklaart dit
        over goede draad, in duwtaal?
  \item Het stopcontact wordt \emph{niet} geïnventariseerd, volgens
        staande huisregel. Herformuleer die regel met het getal van
        dit hoofdstuk, en met de rekenzin die erachter zit.
  \item Dien het inventarisatierapport in: drie zinnen --- wat een
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} is en nooit is; wat de
        \omterm{def:g8:voltage-voltmeter:nominal}{nominale spanning} op een etiket belooft en dreigt; en
        waar in een stilstaande schakeling de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} wacht.
\end{enumerate}
\end{problem}
