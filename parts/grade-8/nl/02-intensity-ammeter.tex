\chapter{Stroomsterkte en de ampèremeter}\label{ch:g8:intensity-ammeter}

``Sterke stroom, zwakke stroom'' --- zo praten we al jaren, terwijl we
helderheid en greep op gevoel doseren. Vandaag gaat het gevoel met
pensioen. De sterkte van de stroom krijgt een naam, een
\omterm{def:g6:measurement-in-science:measuring}{eenheid} en een instrument; en het eerste wat het instrument
bewijst, is een wet die je allang vermoedde maar nooit één keer hebt
gemeten.

\section{Stroomsterkte en haar eenheid}

\begin{definition}[Stroomsterkte]\label{def:g8:intensity-ammeter:intensity}
De \emph{stroomsterkte}\index{stroomsterkte} $I$ van een
\omterm{def:g7:circuits-series-parallel:current}{elektrische stroom} meet hoe sterk de mars is: hoeveel
elektrische lading er elke \omterm{def:g2:measuring-time:second}{seconde} langs een punt van de kring
paradeert --- het debiet van de stroom, zoals dat van een rivier
water-per-seconde is. Haar \omterm{def:g6:measurement-in-science:measuring}{eenheid} is de
\emph{ampère}\index{ampère} (\unit{A}), ter ere van een groot pionier
van de wetten van de elektriciteit; alledaagse schakelingen spreken
vaak in \emph{milliampère}: $\qty{1}{mA} = \qty{0.001}{A}$, een
duizendste.
\end{definition}

\begin{example}[Ampères om op het zicht te kennen]\label{ex:g8:intensity-ammeter:values}
Een klein controlelampje nipt ongeveer \qty{0.02}{A} --- \qty{20}{mA};
een zaklamplampje rond de \qty{0.3}{A}; de grote lamp van een
autokoplamp \qty{5}{A}; een waterkoker op volle \omterm{def:g2:pushes-and-pulls:force}{kracht} \qty{10}{A}; een
startende automotor slokt \qty{100}{A} uit zijn
accu; een bliksemschicht ramt gedurende zijn milliseconde
tienduizenden \omterm{def:g8:intensity-ammeter:intensity}{ampère} door zijn kanaal. En de gevarenschaal is
ontnuchterend: stromen van enkele honderdsten van een
\omterm{def:g8:intensity-ammeter:intensity}{ampère} door een menselijk lichaam kunnen al dodelijk zijn ---
respect begint ver onder één \omterm{def:g8:intensity-ammeter:intensity}{ampère}.
\end{example}

\begin{remark}[De persoon en de eenheid]\label{rem:g8:intensity-ammeter:person}
Voluit geschreven is de \omterm{def:g6:measurement-in-science:measuring}{eenheid} kleingeschreven --- ``een stroom
van twee \omterm{def:g8:intensity-ammeter:intensity}{ampère}'' --- terwijl het woord met hoofdletter de
persoon benoemt, zoals bij elke \omterm{def:g6:measurement-in-science:measuring}{eenheid} die een wetenschapper
eert. Het symbool \unit{A} blijft een hoofdletter. Je komt deze
afspraak binnenkort nog twee keer tegen.
\end{remark}

\section{De ampèremeter}

\begin{definition}[Ampèremeter]\label{def:g8:intensity-ammeter:ammeter}
Een \emph{ampèremeter}\index{ampèremeter} meet de
\omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte}. Hij telt de mars die \emph{door hemzelf} gaat ---
dus moet hij \emph{\omterm{def:g4:series-circuits:series}{in serie}} worden opgenomen, in precies de kring
die hij meet, als een tolhuisje langs de weg. Hij is gebouwd om
de mars zo weinig mogelijk te storen en verzet zich vrijwel niet tegen
de stroom --- en daarom is een ampèremeter die per ongeluk rechtstreeks
over een \omterm{def:g3:first-electric-circuit:battery}{batterij} wordt gezet een \omterm{def:g7:short-circuits-safety:device}{kortsluiting} met een
wijzerplaat: de klassieke manier om er een te slopen.
\end{definition}

\begin{method}[Een stroomsterkte meten]\label{met:g8:intensity-ammeter:measure}
\begin{enumerate}
  \item zet de kring uit; \emph{verbreek} de lus daar waar de stroom
        geteld moet worden;
  \item plaats de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} in de opening: stroom naar binnen
        via zijn \unit{A}-klem, naar buiten via zijn
        \emph{gemeenschappelijke} klem, zodat de conventionele
        richting door de voordeur van de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} binnenkomt;
  \item begin bij een \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} met meerdere bereiken op het
        \emph{grootste} bereik en stap dan omlaag tot de aflezing
        comfortabel op de schaal valt --- dat beschermt de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} tegen
        verrassingen;
  \item sluit de \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}, lees af --- en schrijf de
        \omterm{def:g6:measurement-in-science:measuring}{eenheid} erbij.
\end{enumerate}
Een omgekeerde aansluiting slaat bij een naaldmeter de naald tegen zijn
aanslag; digitale \omterm{def:g2:measuring-length:units}{meters} tonen slechts een minteken en vergeven
dagelijks leerlingen.
\end{method}

\section{De eerste gemeten wet}

\begin{proposition}[Eén lus, één stroomsterkte]\label{prop:g8:intensity-ammeter:sameI}
In een \omterm{def:g4:series-circuits:series}{serieschakeling} is de \omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte} in elk punt van
de lus gelijk. Verplaats de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} waarheen je wilt ---
voor de lamp, erna, naast de \omterm{def:g3:first-electric-circuit:battery}{batterij}: de aflezing verandert niet.
Wat vijf jaar even helder brandende \omterm{def:g3:first-electric-circuit:bulb}{lampjes} suggereerden, stelt de
\omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} nu in getallen vast: onderweg wordt niets verbruikt; één weg, één
mars, één $I$.
\end{proposition}

\begin{tikzpicture}[scale=0.85]
  \draw (0,0) to[battery1] (0,3.0)
    to[ammeter, l=\qty{0.30}{A}] (2.6,3.0)

    to[short] (0,0);
\end{tikzpicture}

{\small Drie tolhuisjes op één lus: voor, tussen en na de lampen meldt
elke \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} dezelfde \qty{0.30}{A}.}
\end{omfigure}

\begin{example}[De kleine lettertjes van de wet]\label{ex:g8:intensity-ammeter:fineprint}
De wet zegt \emph{langs één lus}, niet \emph{wat je ook met de lus
doet}. Zet er een tweede lamp \omterm{def:g4:series-circuits:series}{in serie} bij en de mars verzwakt
overal tegelijk --- misschien zakt \qty{0.30}{A} naar \qty{0.17}{A} ---
maar hij verzwakt \emph{overal even veel}: de drie tolhuisjes zijn het
ook over het nieuwe getal eens. Eén lus, één
\omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte} --- welke \omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte} dat op dat
moment ook is.
\end{example}

\begin{example}[Een blik op het knooppunt]\label{ex:g8:intensity-ammeter:junction}
Zet in plaats daarvan tolhuisjes op een \omterm{def:g5:series-parallel-circuits:parallel}{parallel} paar: hoofdweg
\qty{0.50}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{tak} één \qty{0.30}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{tak} twee
\qty{0.20}{A}. De veelzeggende som is geen toeval --- de marsen een
\omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} in en eruit moeten in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht} zijn, want lading wordt
bij een splitsing van de weg niet geschapen en niet vernietigd. De
volledige wet, met haar partner voor de spanning, houdt over twee
hoofdstukken zitting; je \omterm{def:g2:measuring-length:units}{meters} hebben haar al ontmoet.
\end{example}

\begin{remark}[De multimeter]\label{rem:g8:intensity-ammeter:multimeter}
De gele baksteen van het werkplaats --- de \emph{multimeter} --- vouwt
de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} en verscheidene andere instrumenten in één doos
met een keuzeknop. Knop op \unit{A} of \unit{mA}, snoeren in de
gemerkte bussen, en hij is je \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter}, serieregels en al.
Twee hoofdstukken van dit boek wonen er al in; een derde trekt volgend
hoofdstuk in --- met, pas op, \emph{tegengestelde} aansluitregels.
\end{remark}

\section{Oefeningen}

Wat meet de \omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte}, in riviertaal? Geef haar
\omterm{def:g6:measurement-in-science:measuring}{eenheid} en symbool, en reken \qty{250}{mA} om naar
\omterm{def:g8:intensity-ammeter:intensity}{ampère}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:2}
Koppel de \omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte} aan haar tafereel: \qty{0.02}{A}; \
\qty{0.3}{A}; \ \qty{10}{A}; \ \qty{100}{A} --- waterkoker,

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:3}

\end{exercise}

Formuleer de wet van één lus, één \omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte}. Welke oude

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:6}

\qty{0.45}{A}. Wat zal hij ná de \omterm{ex:g6:circuits-and-safety:motor}{motor} lezen? En naast de

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:7}
Een tweede lamp voegt zich bij een serielus en de \qty{0.30}{A} van de

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:8}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:9}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:10}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:intensity-ammeter:12}

ampèremeterposities er nodig zijn om \emph{elke} stroom in de
schakeling te kennen; het minimum aantal metingen dat volstaat (één
mag worden afgeleid --- door welk \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}?); en de volgorde van
handelingen die nooit een \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} gevaarlijk aangesloten laat staan.
\end{exercise}

\section{Probleem: de meetbank}

\begin{problem}[{Weekendprobleem --- examendag in de
elektronicawerkplaats; vier posten, per post één ampèremeter; het
diploma van de leerling}]\label{pb:g8:intensity-ammeter:1}
Om werkplaatsrechten te verdienen doet elke leerling vier meetposten
aan. Neem het hoofdstuk mee.

\medskip\noindent\textbf{Deel I --- Post één: de eenvoudige lus.}
\omterm{def:g3:first-electric-circuit:battery}{Batterij}, \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}, één lamp.
\begin{enumerate}
  \item Beschrijf stap voor stap hoe je de \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} opneemt
        om de stroom door de lamp te \omterm{def:g6:measurement-in-science:measuring}{meten}.
  \item De aflezing: \qty{280}{mA}. Druk die uit in
        \omterm{def:g8:intensity-ammeter:intensity}{ampère}.
  \item De examinator verplaatst je \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} naar de andere kant van de
        \omterm{def:g3:first-electric-circuit:battery}{batterij}. Voorspel de aflezing, met een beroep op de
        wet.
  \item ``En als ik twee \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeters} tegelijk in deze lus had?''
        Waar zouden ze kunnen staan, en wat zouden ze tonen?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Post twee: het seriepaar.}
Twee verschillende lampen, A helderder dan B, \omterm{def:g4:series-circuits:series}{in serie}.
\begin{enumerate}[resume]
  \item Een leerling voorspelt: ``De helderdere A voert meer stroom
        dan de doffe B.'' Toets de voorspelling aan de wet --- en
        breng de gelijke stromen in overeenstemming met de ongelijke
        helderheid (wat moet er tussen de lampen verschillen?).
  \item De \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} leest \qty{150}{mA} tussen A en B. Schrijf elke
        andere stroom in de lus op.
  \item Lamp B wordt door een krokodillensnoer
        \omterm{def:g7:short-circuits-safety:device}{kortgesloten}. Voorspel de nieuwe aflezing en het gedrag
        van beide lampen.
  \item De examinator vraagt één zin over waarom de werkplaats
        krokodillensnoeren over de \emph{\omterm{def:g3:first-electric-circuit:battery}{batterij}} verbiedt maar de
        demonstratie over B toestaat.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Post drie: het
\omterm{def:g5:series-parallel-circuits:parallel}{knooppunt}.}
Een lamptak en een motortak
\omterm{def:g5:series-parallel-circuits:parallel}{parallel}.
\begin{enumerate}[resume]
  \item Hoofdweg: \qty{0.52}{A}; lamptak: \qty{0.30}{A}. De
        stroom in de motortak, volgens het \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}?
  \item De \omterm{ex:g6:circuits-and-safety:motor}{motor} loopt vast (nog aangesloten, niet draaiend) en
        zijn \omterm{def:g5:series-parallel-circuits:parallel}{tak} klimt naar \qty{0.40}{A}. Wat voert de
        hoofdweg nu, aangenomen dat de lamptak zijn
        \qty{0.30}{A} houdt --- en welke bewaker van de werkplaats
        luistert mee?
  \item De lamp wordt losgedraaid. Voorspel alle drie de aflezingen
        van de tolhuisjes.
  \item Waarom loopt de \omterm{def:g3:first-electric-circuit:battery}{batterij} het snelst leeg met beide
        \omterm{def:g5:series-parallel-circuits:parallel}{takken} gezond --- waar wordt de vrijgevigheid betaald?
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Het diploma.}
\begin{enumerate}[resume]
  \item Schrijf de belofte van de leerling: drie regels --- waar een
        \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} thuishoort en nooit thuishoort; wat één lus
        garandeert; wat een \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht} brengt.
\end{enumerate}
\end{problem}
