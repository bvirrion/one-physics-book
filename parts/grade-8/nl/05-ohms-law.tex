\chapter{Weerstand en de wet van Ohm}\label{ch:g8:ohms-law}

Eén vraag heeft het hele elektrische verhaal achtervolgd: sommige
onderdelen laten de mars razen, andere knijpen hem af tot een straaltje
--- lampen verdelen \omterm{def:g8:voltage-voltmeter:voltage}{spanningen} ongelijk, dunne draden worden warm,
dikke blijven koel. Het verzet zelf krijgt nu een naam, een
\omterm{def:g6:measurement-in-science:measuring}{eenheid} en --- als kroon op vier jaar schakelingen --- de
allernuttigste wet uit dit boek: drie letters die de duw, de mars en
het verzet in één regel algebra aan elkaar knopen.

\section{Weerstand}

\begin{definition}[Weerstand]\label{def:g8:ohms-law:resistance}
De \emph{weerstand}\index{weerstand} $R$ van een onderdeel
meet hoe sterk het zich tegen de
\omterm{def:g7:circuits-series-parallel:current}{elektrische stroom} verzet: bij een gegeven
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} geldt: hoe groter de weerstand, hoe zwakker de mars
die hij doorlaat. Haar \omterm{def:g6:measurement-in-science:measuring}{eenheid} is de \emph{ohm}\index{ohm}
(symbool $\Omega$, de Griekse hoofdletter omega), ter ere van de
schoolmeester-natuurkundige die de wet van dit hoofdstuk vond met
zelfgemaakte draden en heldhaftig geduld.
\end{definition}

\begin{definition}[Weerstand als onderdeel]\label{def:g8:ohms-law:resistor}
Een \emph{weerstand}\index{weerstand} --- als onderdeel, ook wel
resistor genoemd --- wordt gefabriceerd met een gekozen, vaste
weerstandswaarde: een geijkte hindernis, verkrijgbaar van
fracties van een \omterm{def:g8:ohms-law:resistance}{ohm} tot miljoenen. Schakelingen gebruiken
\omterm{def:g8:ohms-law:resistance}{weerstanden} om stromen met opzet in te stellen: de mars temmen die een
teer onderdeel zou verbranden, dimmen, verdelen, bijstellen. Hun
gekleurde ringen spellen hun waarde uit; hun symbool is een kale
rechthoek.
\end{definition}

\begin{example}[Weerstanden om je heen]\label{ex:g8:ohms-law:values}
Een \omterm{def:g2:measuring-length:units}{meter} dikke koperdraad: honderdsten van een
\omterm{def:g8:ohms-law:resistance}{ohm} --- de vlakke weg. Een gloeiend
zaklamplampje: enkele \omterm{def:g8:ohms-law:resistance}{ohm}. Een kleine
elektrische \omterm{ex:g6:circuits-and-safety:motor}{motor}: een paar \omterm{def:g8:ohms-law:resistance}{ohm}. De
verwarmingsspiraal van de waterkoker: ongeveer $25\,\Omega$. Droge
mensenhuid, hand tot hand: tienduizenden \omterm{def:g8:ohms-law:resistance}{ohm} --- nat,
catastrofaal minder: de oude regel over vochtige handen is een uitspraak
over \omterm{def:g8:ohms-law:resistance}{weerstand}. Een open \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}:
\omterm{def:g8:ohms-law:resistance}{weerstand} voorbij alle \omterm{def:g6:measurement-in-science:measuring}{meten} --- de oneindige hindernis.
\end{example}

\section{Het experiment}

\begin{method}[Het $U$--$I$-portret van een onderdeel meten]\label{met:g8:ohms-law:portrait}
Eén \omterm{def:g8:ohms-law:resistance}{weerstand}, één regelbare voeding (of een groeiende rij
cellen), beide \omterm{def:g2:measuring-length:units}{meters}:
\begin{enumerate}
  \item schakel voeding, \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} en
        \omterm{def:g8:ohms-law:resistance}{weerstand} \omterm{def:g4:series-circuits:series}{in serie}; klem de
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} over de \omterm{def:g8:ohms-law:resistance}{weerstand};
  \item stel de kleinste \omterm{def:g8:voltage-voltmeter:voltage}{spanning} in; noteer het paar ($U$,
        $I$);
  \item verhoog de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} stap voor stap --- één cel, twee,
        drie --- en noteer telkens een paar;
  \item zet de punten uit, $I$ horizontaal, $U$ verticaal, en kijk.
\end{enumerate}
\end{method}

\begin{example}[De bekentenis van een weerstand]\label{ex:g8:ohms-law:confession}
De tabel van één sessie, voor één \omterm{def:g8:ohms-law:resistance}{weerstand}:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
Dubbele duw, dubbele mars; drievoudig, drievoudig: $U$ en $I$ zijn
\emph{evenredig}, en elke verhouding $U/I$ geeft hetzelfde getal terug:
$15$. Die constante verhouding is geen toeval --- het is de
\omterm{def:g8:ohms-law:resistance}{weerstand} van de \omterm{def:g8:ohms-law:resistance}{weerstand}, $R = 15\,\Omega$, die
zich in elke rij vertoont.
\end{example}

\begin{proposition}[De wet van Ohm]\label{prop:g8:ohms-law:law}
Voor een \omterm{def:g8:ohms-law:resistance}{weerstand} bij constante
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur} zijn de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} erover en de
stroom erdoor evenredig, en de evenredigheidsconstante is haar
\omterm{def:g8:ohms-law:resistance}{weerstand}:
\[
U = R \times I ,
\]
met $U$ in \omterm{def:g8:voltage-voltmeter:voltage}{volt}, $I$ in \omterm{def:g8:intensity-ammeter:intensity}{ampère}, $R$ in
\omterm{def:g8:ohms-law:resistance}{ohm}. Naar smaak herschikt: $I = U/R$ (de stroom die een duw
door een hindernis stuurt) en $R = U/I$ (de hindernis, ontmaskerd door
één eerlijk paar aflezingen).
\end{proposition}

\begin{example}[De wet rekent alles uit]\label{ex:g8:ohms-law:compute}
Zodra $R$ bekend is, antwoordt de wet in elke richting. De
waterkokerspiraal, $25\,\Omega$ op het \qty{230}{V}-net: $I = U/R = 230
\div 25 = \qty{9.2}{A}$ --- het tienampèreapparaat verklaard.
Een \omterm{def:g8:ohms-law:resistance}{weerstand} moet de stroom door een teer \omterm{def:g3:first-electric-circuit:bulb}{lampje} bij
\qty{4.5}{V} tot \qty{0.02}{A} beperken: $R = U/I = 4.5 \div 0.02 =
225\,\Omega$ --- kies de eerstvolgende standaardwaarde. Door
$60\,\Omega$ loopt \qty{0.05}{A}: de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} erover is $U = 60
\times 0.05 = \qty{3}{V}$. Eén wet, drie recepten, de hele werkplaats.
\end{example}

\begin{remark}[De driehoekkruk]\label{rem:g8:ohms-law:triangle}
Sommige leerlingen tekenen $U$ boven $R$ en $I$ in een driehoekje en
dekken de gezochte letter af om het recept af te lezen. Als kruk
onschuldig --- maar jouw algebra herschikt $U = R I$ nu eerlijk in één
regel, en de eerlijke manier reist mee naar elke formule die je ooit
tegenkomt, driehoek of niet. Leun dit jaar op de kruk als het moet;
maar plan om te lopen.
\end{remark}

\section{Wat weerstand verklaart}

\begin{example}[Oude mysteries op één rekening]\label{ex:g8:ohms-law:mysteries}
De ongelijke spanningsverdeling uit het wettenhoofdstuk: lampen
\omterm{def:g4:series-circuits:series}{in serie} delen één $I$, dus neemt volgens $U = R I$ de grotere $R$ de
grotere $U$ --- het grotere verval van de zwaarste werker, nu
berekenbaar. De afgeknepen \omterm{def:g5:series-parallel-circuits:parallel}{tak} uit het parallelhoofdstuk: de
lange dunne draad voegde \omterm{def:g8:ohms-law:resistance}{weerstand} toe, en $I = U/R$ liet het
deel van die \omterm{def:g5:series-parallel-circuits:parallel}{tak} krimpen. De \omterm{def:g7:short-circuits-safety:device}{kortsluiting}: een
sluiproute met bijna nul $R$, zodat $I = U/R$ ontploft --- de stormloop
was altijd al een deling door bijna nul. Vier jaar kwalitatieve
schakelwijsheid, op de rekening van één wet.
\end{example}

\begin{example}[Draden, per meter]\label{ex:g8:ohms-law:wires}
De \omterm{def:g8:ohms-law:resistance}{weerstand} van een draad groeit met zijn lengte en krimpt
met zijn dikte --- een lange dunne draad is een smalle bergweg, een
korte dikke een snelweg. Vandaar de vette kabels van de elektricien
voor gulzige apparaten, de met opzet lange dunne weerstandslegering van
de verwarmingsspiraal --- en de reden dat dezelfde wet die een lamp
laat branden een broodrooster verwarmt:
\omterm{def:g8:ohms-law:resistance}{weerstand} is waar de mars zijn duw uitgeeft, en uitgegeven duw
wordt, zoals het energiehoofdstuk volgend jaar zal bevestigen,
\omterm{def:g5:heat-and-insulation:heat}{warmte}.
\end{example}

\begin{remark}[Waar het gezag van de wet ophoudt]\label{rem:g8:ohms-law:limits}
De wet van Ohm is de wet van een \omterm{def:g8:ohms-law:resistance}{weerstand}, geen universeel
recht. Meet het portret van een \emph{gloeilamp} en de punten buigen
van de rechte lijn af: de gloeidraad verhit naarmate de stroom
groeit, en heet metaal biedt meer verzet --- $R$ weigert te blijven
zitten. Dioden, \omterm{ex:g6:circuits-and-safety:motor}{motoren} en levende huid buigen hun portretten
ook, elk op hun eigen manier. De rechte lijn door de oorsprong is de
handtekening van de \omterm{def:g8:ohms-law:resistance}{weerstand}, niet die van de materie ---
vraag een onderdeel altijd om zijn portret voor je het de wet
toevertrouwt.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
Wat meet de \omterm{def:g8:ohms-law:resistance}{weerstand}? Geef haar
\omterm{def:g6:measurement-in-science:measuring}{eenheid} en symbool, en schrijf de wet van Ohm met haar drie
gedaantes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
Vul elk paar aan voor een \omterm{def:g8:ohms-law:resistance}{weerstand} van $30\,\Omega$: $U =
\qty{6}{V}$, $I = {?}$; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
Een onderdeel toont $U = \qty{4.5}{V}$ bij $I = \qty{0.09}{A}$. Zijn
\omterm{def:g8:ohms-law:resistance}{weerstand}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
Welke vorm tekenen de punten van een echte \omterm{def:g8:ohms-law:resistance}{weerstand} in het
portretexperiment, en wat betekent een steilere lijn?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
Verifieer $R$ uit de tweede en de vierde rij van de tabel in dit
hoofdstuk. Waarom is het beide keren hetzelfde getal krijgen precies
waar het om gaat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
De spiraal van de waterkoker is $25\,\Omega$; die van een tweede
waterkoker $35\,\Omega$. Welke trekt op hetzelfde net de sterkste
stroom --- en hoeveel trekt elk (één decimaal)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
Verklaar met $U = R I$ waarom van twee lampen \omterm{def:g4:series-circuits:series}{in serie} die één stroom
delen, de lamp met de grootste \omterm{def:g8:ohms-law:resistance}{weerstand} de grootste
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} neemt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
Vertaal naar weerstandstaal: de stormloop van de
\omterm{def:g7:short-circuits-safety:device}{kortsluiting}; \ de volledige blokkade van de open
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
Een teer controlelampje verdraagt \qty{20}{mA}. Welke
\omterm{def:g8:ohms-law:resistance}{weerstand} \omterm{def:g4:series-circuits:series}{in serie} beschermt het op een
\qty{9}{V}-batterij, als het \omterm{def:g3:first-electric-circuit:bulb}{lampje} zelf ongeveer \qty{2}{V}
van het deel neemt? (Eerst het deel van de
\omterm{def:g8:ohms-law:resistance}{weerstand}, dan $R$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
Het gemeten portret van een gloeilamp: (\qty{1}{V}, \qty{0.25}{A}),
(\qty{4}{V}, \qty{0.5}{A}). Toon aan dat geen enkele $R$ bij beide
punten past, en vertel het natuurkundige verhaal achter de buiging.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
Twee \omterm{def:g8:ohms-law:resistor}{weerstanden} \omterm{def:g4:series-circuits:series}{in serie}, $10\,\Omega$ en
$20\,\Omega$, op \qty{6}{V}. Vind met de seriewetten plus de wet van
Ohm op elk: de stroom in de lus en de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} over elke
\omterm{def:g8:ohms-law:resistance}{weerstand}. (Tip: één stroom $I$ voldoet aan $6 = 10 I + 20
I$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
Dezelfde twee \omterm{def:g8:ohms-law:resistor}{weerstanden}, nu \omterm{def:g5:series-parallel-circuits:parallel}{parallel} op
\qty{6}{V}. Vind de stroom in elke \omterm{def:g5:series-parallel-circuits:parallel}{tak}, het totaal van de
\omterm{def:g3:first-electric-circuit:battery}{batterij}, en dan --- de elegante afsluiting --- de enkele
\omterm{def:g8:ohms-law:resistance}{weerstand} $R$ die datzelfde totaal uit \qty{6}{V} zou
trekken. Vergelijk hem met beide \omterm{def:g8:ohms-law:resistor}{weerstanden} en verklaar in
wegentaal waarom het team van twee zich \emph{minder} verzet dan elk
van beide leden alleen.
\end{exercise}

\section{Probleem: de kwaliteitscontrolebank}

\begin{problem}[{Weekendprobleem --- keuringsdag in de
weerstandenfabriek; portretten, toleranties en één onderdeel dat
helemaal geen weerstand is}]\label{pb:g8:ohms-law:1}
De kwaliteitsbank van de fabriek test de productie van de dag --- en
één bedrieger. Jij bedient de \omterm{def:g2:measuring-length:units}{meters}.

\medskip\noindent\textbf{Deel I --- De bank ijken.}
\begin{enumerate}
  \item Schets (of beschrijf) de testschakeling: voeding,
        \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter}, het te testen onderdeel,
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} --- wie staat \omterm{def:g4:series-circuits:series}{in serie}, wie
        eroverheen?
  \item Monster één laat bij \qty{4.5}{V} \qty{0.15}{A} door. Zijn
        \omterm{def:g8:ohms-law:resistance}{weerstand}?
  \item Het etiket claimt $33\,\Omega$, tolerantie vijf procent.
        Bereken de aanvaardbare band, en vel je oordeel over monster
        één. (Eerst vijf procent van $33$.)
  \item Monster twee, met etiket $150\,\Omega$: voorspel zijn stroom
        bij \qty{4.5}{V}, zodat de bank weet wat te verwachten (één
        decimaal, in \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Volledige portretten.}
\begin{enumerate}[resume]
  \item De tabel van monster drie: $(1.5, 0.05)$, $(3.0, 0.10)$,
        $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{volt},
        \omterm{def:g8:intensity-ammeter:intensity}{ampère}. Bepaal $R$, met daarbij het kenmerk van de
        getallen dat één enkele waarde toestaat.
  \item De tabel van monster vier: $(1.5, 0.30)$, $(3.0, 0.45)$,
        $(4.5, 0.54)$, $(6.0, 0.60)$. Ontmasker de bedrieger: geen
        enkele $R$, en de richting van de drift.
  \item Monster vier is in werkelijkheid een klein gloeilampje van de
        volgende productielijn. Vertel zijn buigverhaal --- en waarom
        de rechtelijntest van de bank precies een weerstanddetector
        is.
  \item Het reglement van de bank zegt: ``elk portret moet bij
        constante \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} worden opgenomen.'' Welke
        bepaling van de wet van Ohm beschermt die regel?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Afdeling toepassingen.}
\begin{enumerate}[resume]
  \item Een bestelling vraagt om een \omterm{def:g8:ohms-law:resistance}{weerstand} die de stroom
        op een \qty{12}{V}-voeding tot \qty{30}{mA} beperkt (het
        eigen deel van de belasting verwaarloosbaar). Bereken de
        waarde die je moet offreren.
  \item Een klant klaagt dat zijn verwarmingsspiraal van
        $25\,\Omega$ ``maar'' \qty{9.2}{A} trekt terwijl de
        \omterm{def:g7:short-circuits-safety:fuse}{zekering} er $10$ toestaat. Bereken welke
        voedingsspanning zijn klacht impliceert, en stel
        hem met de rekensom gerust.
  \item Het prototypelab vraagt om de dikste, kortste koperen
        doorverbinder van de fabriek, ``met zo dicht mogelijk bij nul
        \omterm{def:g8:ohms-law:resistance}{ohm}''. Voor welke rol in een schakeling is die
        doorverbinder geboren --- en in de buurt van welke alledaagse
        schurk plaatst zijn bijna-nul-weerstand hem bij
        verkeerd gebruik?
  \item Sluit het keuringslogboek af: drie zinnen --- de wet, haar
        portret, en de ene voorwaarde waaronder een onderdeel haar
        mag ondertekenen.
\end{enumerate}
\end{problem}
