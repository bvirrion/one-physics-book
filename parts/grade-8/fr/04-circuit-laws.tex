\chapter{Les lois des courants et des tensions dans les circuits}\label{ch:g8:circuit-laws}

Deux appareils, deux grandeurs --- et voici enfin la constitution.
Tout circuit que tu construiras jamais, de la lampe de poche au réseau
d'une ville, obéit à quatre lois brèves~: deux sur la comptabilité du
courant, deux sur celle de la \omterm{def:g8:voltage-voltmeter:voltage}{tension}. Muni d'elles et d'un peu
d'algèbre, tu peux calculer des lectures que tu n'as pas encore
mesurées --- et confondre toute mesure qui ment.

\section{Les lois des courants}

\begin{proposition}[Loi des courants dans une boucle en série]\label{prop:g8:circuit-laws:seriesI}
Le long d'une seule boucle, l'\omterm{def:g8:intensity-ammeter:intensity}{intensité} est la même partout~:
\[
I_{\text{pile}} = I_{\text{lampe 1}} = I_{\text{lampe 2}} =
\dots
\]
Une route, une marche --- les péages du chapitre précédent, devenus
une loi de la constitution.
\end{proposition}

\begin{proposition}[Loi des nœuds]\label{prop:g8:circuit-laws:junction}
En tout \omterm{def:g5:series-parallel-circuits:parallel}{nœud}, les \omterm{def:g8:intensity-ammeter:intensity}{intensités} qui entrent, prises ensemble, égalent
les \omterm{def:g8:intensity-ammeter:intensity}{intensités} qui sortent, prises ensemble~: pour une route
principale qui se divise en deux \omterm{def:g5:series-parallel-circuits:parallel}{branches},
\[
I = I_1 + I_2 .
\]
La charge n'est ni créée, ni détruite, ni stockée à un
embranchement~: tout ce qui arrive repart. (Vrai pour un nombre
quelconque de \omterm{def:g5:series-parallel-circuits:parallel}{branches} --- somme des arrivées, somme des départs, et
les comptes s'équilibrent.)
\end{proposition}

\section{Les lois des tensions}

\begin{proposition}[Loi des tensions dans une boucle en série]\label{prop:g8:circuit-laws:seriesU}
Le long d'une boucle \omterm{def:g4:series-circuits:series}{en série}, la \omterm{def:g8:voltage-voltmeter:voltage}{tension} de la \omterm{def:g3:first-electric-circuit:battery}{pile} se partage entre
les appareils, et les parts s'additionnent exactement~:
\[
U_{\text{pile}} = U_1 + U_2 + \dots
\]
L'\omterm{prop:g5:mirrors-reflection:image}{image} de la chute d'eau tient les comptes~: la \omterm{def:g3:first-electric-circuit:battery}{pile} hisse la marche
sur toute sa hauteur, et les appareils dépensent cette hauteur entre
eux, goutte à goutte --- les bons fils n'en dépensant rien.
\end{proposition}

\begin{proposition}[Loi des tensions pour des branches en dérivation]\label{prop:g8:circuit-laws:parallelU}
Des appareils \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} sont tous placés entre les deux mêmes
\omterm{def:g5:series-parallel-circuits:parallel}{nœuds} --- une seule \omterm{def:g8:voltage-voltmeter:voltage}{tension} les sert donc tous~:
\[
U_1 = U_2 = U_{\text{entre les deux nœuds}} .
\]
Chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} jouit de toute la poussée entre les \omterm{def:g5:series-parallel-circuits:parallel}{nœuds} --- et c'est
pourquoi des lampes \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} brillent à leur pleine valeur
nominale, et pourquoi chaque prise de ta maison offre les mêmes
\qty{230}{V}.
\end{proposition}

\begin{example}[Pourquoi des parts inégales~?]\label{ex:g8:circuit-laws:unequal}
Deux lampes différentes \omterm{def:g4:series-circuits:series}{en série} portent le même courant (première
loi) et se partagent pourtant la \omterm{def:g8:voltage-voltmeter:voltage}{tension} inégalement --- \qty{2.1}{V}
contre \qty{2.4}{V}. La plus grosse part revient à la lampe qui
s'oppose le plus à la marche~: le travailleur le plus rude prend la
plus grande chute. Ce que «~s'opposer davantage~» veut dire au juste,
et comment le \omterm{def:g6:measurement-in-science:measuring}{mesurer}, fait tout le sujet du chapitre suivant --- les
lois de ce chapitre-ci en sont la scène.
\end{example}

\section{Calculer avant de mesurer}

\begin{method}[Résoudre un circuit avec les quatre lois]\label{met:g8:circuit-laws:solve}
Étant donné un circuit dont on connaît quelques lectures~:
\begin{enumerate}
  \item dessiner le schéma et marquer tous les $I$ et $U$ connus~;
  \item écrire les lois applicables sous forme d'équations ---
        sommes aux \omterm{def:g5:series-parallel-circuits:parallel}{nœuds} pour les courants, sommes sur la boucle
        pour les \omterm{def:g8:voltage-voltmeter:voltage}{tensions}, égalité pour le $I$ \omterm{def:g4:series-circuits:series}{en série} et le $U$
        \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}~;
  \item résoudre en cherchant les inconnues (les équations du cours
        d'algèbre, enfin utiles)~;
  \item vérifier le bon sens~: aucune lampe ne reçoit plus que ce
        que la \omterm{def:g3:first-electric-circuit:battery}{pile} offre, aucun courant de \omterm{def:g5:series-parallel-circuits:parallel}{branche} ne dépasse
        celui de la route principale.
\end{enumerate}
\end{method}

\begin{example}[Exemple traité~: le circuit mixte]\label{ex:g8:circuit-laws:worked}
Une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{6}{V} alimente la lampe A \omterm{def:g4:series-circuits:series}{en série} avec une paire \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation} (B et C). Mesures~: $I_{\text{pile}} = \qty{0.5}{A}$,
$I_B = \qty{0.3}{A}$, $U_A = \qty{2.5}{V}$. Calculons le reste. Loi
des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}~: $I_C = 0.5 - 0.3 = \qty{0.2}{A}$. Loi des courants \omterm{def:g4:series-circuits:series}{en
série}~: la lampe A porte les \qty{0.5}{A} en entier. Loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions}
sur la boucle~: la paire reçoit
$U_{\text{paire}} = 6 - 2.5 = \qty{3.5}{V}$~; loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}~: $U_B = U_C = \qty{3.5}{V}$. Toutes les lectures
d'appareil du circuit, désormais connues --- quatre lois, une ligne
d'algèbre chacune.
\end{example}

\begin{example}[Les lois comme détecteurs de mensonge]\label{ex:g8:circuit-laws:liedetector}
Un compte rendu de travaux pratiques affirme~: \omterm{def:g3:first-electric-circuit:battery}{pile} \qty{4.5}{V}~;
deux lampes \omterm{def:g4:series-circuits:series}{en série} à \qty{2.1}{V} et \qty{2.9}{V}. La loi de la
boucle additionne les parts~: \qty{5.0}{V} dépensés depuis une bourse
de \qty{4.5}{V} --- impossible~; quelqu'un a mal lu un cadran. Un
autre compte rendu~: \qty{0.40}{A} entrant au \omterm{def:g5:series-parallel-circuits:parallel}{nœud}, \qty{0.25}{A} et
\qty{0.20}{A} sortant par les \omterm{def:g5:series-parallel-circuits:parallel}{branches} --- l'embranchement «~crée~»
\qty{0.05}{A}~: rejeté. Avant les lois, des lectures fausses
paraissaient aussi bonnes que des justes~; désormais, la constitution
audite chaque page.
\end{example}

\begin{remark}[Les piles sous les mêmes lois]\label{rem:g8:circuit-laws:batteries}
La règle du bout à bout n'est que la loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série} dans
ses plus vieux habits~: des \omterm{def:g3:first-electric-circuit:battery}{piles} \omterm{def:g4:series-circuits:series}{en série} additionnent leurs
\omterm{def:g8:voltage-voltmeter:voltage}{tensions} ($1.5 + 1.5 + 1.5 = \qty{4.5}{V}$ --- le secret de la \omterm{def:g3:first-electric-circuit:battery}{pile}
plate), et un élément retourné se \emph{retranche}, sa poussée passée
à combattre les autres. Les lois gouvernent les donneurs comme les
dépensiers~; seul le signe de leur contribution diffère.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:1}
Énonce les quatre lois --- deux pour le courant, deux pour la
\omterm{def:g8:voltage-voltmeter:voltage}{tension} --- chacune en une ligne.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:2}
Un \omterm{def:g5:series-parallel-circuits:parallel}{nœud} reçoit \qty{0.8}{A} et envoie \qty{0.55}{A} dans une \omterm{def:g5:series-parallel-circuits:parallel}{branche}.
Que porte l'autre \omterm{def:g5:series-parallel-circuits:parallel}{branche}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:3}
Une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{9}{V} alimente trois lampes \omterm{def:g4:series-circuits:series}{en série}~; deux d'entre
elles prennent \qty{2.5}{V} et \qty{3.8}{V}. La part de la
troisième~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:4}
Deux lampes \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} aux \omterm{def:g3:first-electric-circuit:battery}{bornes} d'une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{12}{V}~: la
\omterm{def:g8:voltage-voltmeter:voltage}{tension} de chaque lampe~? Quel fait quotidien à propos des prises
d'une maison relève de la même loi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:5}
Dans le circuit mixte traité, quelle mesure supplémentaire aurait été
redondante, et pourquoi~? (Choisis-en une, justifie par une loi.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:6}
Deux lampes différentes \omterm{def:g4:series-circuits:series}{en série} portent le même courant mais se
partagent la \omterm{def:g8:voltage-voltmeter:voltage}{tension} $1.8$ contre \qty{2.7}{V}. Quelle lampe
s'oppose le plus à la marche~? Quel chapitre à venir mesure cette
opposition~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:7}
Audit~: \omterm{def:g3:first-electric-circuit:battery}{pile} \qty{6}{V}~; lampes \omterm{def:g4:series-circuits:series}{en série} mesurées à \qty{2.2}{V},
\qty{1.9}{V}, \qty{1.4}{V}. Les comptes s'équilibrent-ils à
l'honnêteté d'un appareil près~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:8}
Audit~: route principale \qty{0.9}{A}~; trois \omterm{def:g5:series-parallel-circuits:parallel}{branches} \qty{0.4}{A},
\qty{0.3}{A}, \qty{0.3}{A}. Verdict~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:9}
Une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{4.5}{V}, la lampe D \omterm{def:g4:series-circuits:series}{en série} avec un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}
--- puis, \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} avec la lampe D, un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} indique
\qty{4.5}{V} aux \omterm{def:g3:first-electric-circuit:battery}{bornes} de l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} \emph{ouvert} et \qty{0}{V}
aux \omterm{def:g3:first-electric-circuit:battery}{bornes} de D. Montre que les deux lectures obéissent à la loi des
\omterm{def:g8:voltage-voltmeter:voltage}{tensions} sur la boucle.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:10}
Trois éléments identiques bout à bout allument une lampe~; un
quatrième, retourné, rejoint la file. \omterm{def:g8:voltage-voltmeter:voltage}{Tension} totale avant et après,
par la loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série} avec les signes~? Que fait la
lampe~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:11}
Un circuit à deux \omterm{def:g5:series-parallel-circuits:parallel}{branches}~: la \omterm{def:g5:series-parallel-circuits:parallel}{branche} une porte les lampes E et F
\omterm{def:g4:series-circuits:series}{en série}~; la \omterm{def:g5:series-parallel-circuits:parallel}{branche} deux porte la lampe G seule. \omterm{def:g3:first-electric-circuit:battery}{Pile} \qty{6}{V}~;
$U_E = \qty{2.2}{V}$. Calcule $U_F$ et $U_G$, en citant une loi à
chaque étape.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:circuit-laws:12}
L'audit complet~: une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{6}{V}~; la lampe P \omterm{def:g4:series-circuits:series}{en série} avec
un \omterm{def:g5:series-parallel-circuits:parallel}{nœud}~; les \omterm{def:g5:series-parallel-circuits:parallel}{branches} Q et R se rejoignent et reviennent. Mesures~:
$I_P = \qty{0.6}{A}$, $I_Q = \qty{0.45}{A}$, $U_Q = \qty{2.3}{V}$.
Calcule tous les courants et toutes les \omterm{def:g8:voltage-voltmeter:voltage}{tensions} restants du circuit
($I_R$, $U_P$, $U_R$), en énonçant la loi derrière chaque ligne ---
puis invente une «~mesure~» supplémentaire qu'un élève négligent
pourrait rapporter et que ta solution achevée confondrait
immédiatement.
\end{exercise}

\section{Problème~: les boîtes mystères scellées}

\begin{problem}[{Problème du week-end --- le défi des boîtes
scellées du club d'électronique~; quatre lois contre quatre
mystères~; le grand audit}]\label{pb:g8:circuit-laws:1}
Le club scelle des circuits simples dans des boîtes, ne laissant que
des prises pour les appareils. Les équipes doivent en déduire
l'intérieur --- constitution en main. La \omterm{def:g3:first-electric-circuit:battery}{pile} de l'établi est de
\qty{6}{V} partout.

\medskip\noindent\textbf{Partie I --- boîte A~: deux \omterm{def:g3:first-electric-circuit:battery}{bornes}, deux
lampes.}
L'étiquette avoue~: deux lampes identiques et rien d'autre, soit \omterm{def:g4:series-circuits:series}{en
série}, soit \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}.
\begin{enumerate}
  \item Branchée sur la \omterm{def:g3:first-electric-circuit:battery}{pile}, la boîte A appelle \qty{0.2}{A}~; une
        seule lampe de ce genre, seule sous \qty{6}{V}, appelle
        environ \qty{0.4}{A}. Série ou dérivation~? Argumente par
        une loi des courants.
  \item Prédis ce que la boîte appellerait dans l'\emph{autre}
        disposition.
  \item Quelle \omterm{def:g8:voltage-voltmeter:voltage}{tension} chaque lampe intérieure reçoit-elle dans la
        disposition réelle, et par quelle loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions}~?
  \item Dans quelle disposition les lampes cachées brilleraient-elles
        à leur pleine valeur nominale --- et pourquoi, visiblement,
        ne le font-elles pas, vu par le judas de la boîte~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- boîte B~: l'énigme des trois
lampes.}
La boîte B avoue~: une lampe (X) \omterm{def:g4:series-circuits:series}{en série} avec une paire \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation} (Y, Z --- pas nécessairement identiques).
\begin{enumerate}[resume]
  \item L'équipe mesure $I_{\text{pile}} = \qty{0.5}{A}$ et, à la
        prise d'essai propre à Y, $I_Y = \qty{0.35}{A}$. Trouve
        $I_Z$ et $I_X$, en citant les lois.
  \item Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} aux \omterm{def:g3:first-electric-circuit:battery}{bornes} de X~: \qty{2.8}{V}. Trouve la
        \omterm{def:g8:voltage-voltmeter:voltage}{tension} de la paire \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} --- puis $U_Y$ et $U_Z$.
  \item Y et Z portent des courants différents sous la même \omterm{def:g8:voltage-voltmeter:voltage}{tension}.
        Que révèle cela sur les deux lampes, dans le langage de
        \cref{ex:g8:circuit-laws:unequal}~?
  \item L'équipe dévisse Y par une trappe. Prédis le sens de
        variation du nouveau $I_{\text{pile}}$ (hausse, baisse ou
        maintien), en raisonnant sur les routes et leur opposition.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- boîte C~: la triche.}
\begin{enumerate}[resume]
  \item Le constructeur de la boîte C rapporte~: «~\omterm{def:g3:first-electric-circuit:battery}{pile}
        \qty{6}{V}~; deux lampes \omterm{def:g4:series-circuits:series}{en série} à l'intérieur, à
        \qty{2.5}{V} et \qty{4.1}{V}~; \omterm{def:g5:series-parallel-circuits:parallel}{nœud} intérieur recevant
        \qty{0.3}{A} et envoyant \qty{0.2}{A} plus \qty{0.15}{A}.~»
        Confonds ce rapport deux fois, une loi par condamnation.
  \item L'une des deux \omterm{def:g8:voltage-voltmeter:voltage}{tensions} rapportées se révèle par la suite
        honnête. Si c'est celle de \qty{2.5}{V}, que doit valoir
        réellement sa partenaire~?
  \item Le constructeur avoue que les nombres du \omterm{def:g5:series-parallel-circuits:parallel}{nœud} ont été
        «~arrondis généreusement~». Si les deux lectures de \omterm{def:g5:series-parallel-circuits:parallel}{branche}
        \qty{0.2}{A} et \qty{0.15}{A} sont honnêtes, que doit porter
        réellement la route principale~?
  \item Pourquoi les quatre lois font-elles de la triche en boîte
        scellée un jeu perdu d'avance~? Une phrase.
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- la constitution.}
\begin{enumerate}[resume]
  \item Rédige la charte du club~: les quatre lois en quatre lignes,
        chacune avec l'\omterm{prop:g5:mirrors-reflection:image}{image} quotidienne qui la porte (routes,
        péages, chutes d'eau, embranchements).
\end{enumerate}
\end{problem}
