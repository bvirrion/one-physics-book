\chapter{Le son~: vitesse, hauteur, intensité sonore}\label{ch:g8:sound-pitch-loudness}

Les cordes courtes chantent aigu, les pincements vigoureux chantent
fort --- ta guitare en élastiques avait trouvé les règles il y a des
années. Cette année, les règles reçoivent des nombres~: un compte par
seconde pour la hauteur, une taille d'oscillation pour l'intensité
sonore, et une fenêtre --- avec des bords --- sur ce que les oreilles
humaines peuvent seulement attraper. Au-delà d'un des bords, les
chauves-souris se moquent de nous.

\section{La fréquence~: la hauteur comptée}

\begin{definition}[Fréquence]\label{def:g8:sound-pitch-loudness:frequency}
La \emph{fréquence}\index{fréquence} $f$ d'une \omterm{def:g3:sound-around-us:vibration}{vibration} est le nombre
d'allers-retours complets qu'elle accomplit chaque seconde. Son \omterm{def:g6:measurement-in-science:measuring}{unité}
est le \emph{hertz}\index{hertz} (\unit{Hz})~: une oscillation par
seconde --- en l'honneur du physicien qui produisit et attrapa le
premier des ondes invisibles. Une corde de guitare qui tremble $110$
fois par seconde vibre à \qty{110}{Hz}~; les ailes d'un moustique,
près de \qty{600}{Hz}~; le claquement lent d'un grand drapeau,
quelques hertz.
\end{definition}

\begin{proposition}[La hauteur, c'est la fréquence]\label{prop:g8:sound-pitch-loudness:pitch}
Ce que l'oreille appelle la \emph{hauteur} est la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} et rien
d'autre~: plus la source tremble vite, plus la note est aiguë. Les
règles de l'élastique se traduisent aussitôt --- ce qui est court,
tendu et fin tremble à un $f$ plus élevé. L'ancre de la musique~: les
instruments de concert du monde entier s'accordent sur le la à
\qty{440}{Hz}. Doubler la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} d'une note l'élève du plus
consonant de tous les intervalles --- l'octave~: le la de nouveau, une
octave au-dessus, à \qty{880}{Hz}.
\end{proposition}

\begin{method}[Voir le son]\label{met:g8:sound-pitch-loudness:seeing}
Le tambour aux grains de riz montrait le tremblement du \omterm{def:g3:sound-around-us:sound}{son}~; un
microphone le dessine.
\begin{enumerate}
  \item lancer une application d'analyse du \omterm{def:g3:sound-around-us:sound}{son} sur un téléphone ou
        un ordinateur --- son écran trace l'oscillation du
        microphone en fonction du temps, un oscilloscope moderne~;
  \item fredonner grave et régulier~: un tracé paresseux, aux
        crêtes bien espacées~; chanter aigu~: les crêtes se
        serrent~;
  \item lire l'affichage de la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} en sifflant~: la plupart
        des sifflements tombent entre $500$ et \qty{2000}{Hz}~;
  \item frapper le diapason marqué «~la 440~» et vérifier que
        l'affichage honore sa gravure.
\end{enumerate}
Un écran remplace tout un chapitre de conjectures~: la hauteur
comptée sous tes yeux.
\end{method}

\section{La fenêtre de l'oreille}

\begin{proposition}[L'étendue de l'audition humaine]\label{prop:g8:sound-pitch-loudness:range}
Des oreilles jeunes et saines attrapent les \omterm{def:g3:sound-around-us:vibration}{vibrations} comprises entre
environ \qty{20}{Hz} et \qty{20000}{Hz}. En dessous de la fenêtre
s'étendent les \emph{infrasons}\index{infrason} --- parfois sentis
comme un grondement dans la poitrine, jamais entendus~; au-dessus, les
\emph{ultrasons}\index{ultrason} --- silence pour nous, si violent que
soit le tremblement. La fenêtre se rétrécit avec l'âge et avec les
abus~: l'octave la plus aiguë est d'ordinaire la première victime des
habitudes bruyantes.
\end{proposition}

\begin{example}[Au-delà des bords]\label{ex:g8:sound-pitch-loudness:beyond}
La nature et l'industrie travaillent toutes deux au-delà de notre
fenêtre. Les chauves-souris chassent par des clics ultrasonores voisins
de \qty{50000}{Hz} et lisent des \omterm{ex:g4:how-sound-travels:echo}{échos} plus fins que ne saurait en
peindre aucun son audible --- les dauphins de même, dans l'esprit du
chapitre sur le sonar. Les chiens répondent à des sifflets
«~silencieux~» accordés juste au-dessus de notre bord. Les appareils
médicaux forment l'\omterm{prop:g5:mirrors-reflection:image}{image} des enfants à naître avec d'inoffensifs \omterm{ex:g4:how-sound-travels:echo}{échos}
d'\omterm{prop:g8:sound-pitch-loudness:range}{ultrasons}. À l'autre bord, les éléphants conversent en grondements
infrasonores sur des kilomètres, et les tempêtes comme les séismes
s'annoncent sur des tons qu'aucune oreille n'écoute.
\end{example}

\section{L'intensité sonore~: la taille de l'oscillation}

\begin{proposition}[L'intensité sonore, c'est l'amplitude]\label{prop:g8:sound-pitch-loudness:loudness}
Ce que l'oreille appelle l'\emph{intensité sonore} correspond à
l'\emph{amplitude}\index{amplitude} de la \omterm{def:g3:sound-around-us:vibration}{vibration} --- la taille de
l'oscillation, la \omterm{def:g2:pushes-and-pulls:force}{force} du resserrement qui voyage. Pince doucement ou
sauvagement~: les crêtes du tracé se tiennent basses ou hautes tandis
que leur \emph{espacement} --- la hauteur --- reste intact.
L'\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude} et la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} sont deux réglages indépendants~: toute
hauteur peut être murmurée ou hurlée.
\end{proposition}

\end{omfigure}

\begin{example}[L'échelle des décibels]\label{ex:g8:sound-pitch-loudness:decibels}
L'intensité sonore se gradue en \emph{décibels}\index{décibel}
(\unit{dB}) --- une échelle bâtie pour l'étendue stupéfiante de
l'oreille, sur laquelle chaque renforcement du \omterm{def:g3:sound-around-us:sound}{son} d'un facteur dix
ajoute le même nombre de barreaux. Repères~: le bruissement des
feuilles, \qty{10}{dB}~; une conversation calme, \qty{50}{dB}~; une
circulation dense, \qty{80}{dB}~; les premiers rangs d'un concert de
rock, \qty{110}{dB}~; la douleur et le dommage immédiat, près de
\qty{120}{dB}. L'étrange arithmétique de cette échelle --- des pas
égaux pour des \omterm{def:g2:pushes-and-pulls:force}{forces} multipliées par dix --- est un logarithme, l'une
des grandes inventions qui t'attendent en mathématiques au lycée.
\end{example}

\begin{remark}[Le budget de l'oreille]\label{rem:g8:sound-pitch-loudness:earsafety}
Le petit tambour de ton oreille, rencontré dans ton chapitre de
troisième année, tient des comptes à vie. Au-dessus d'environ
\qty{85}{dB}, le dommage s'accumule avec la \omterm{def:g2:measuring-time:duration}{durée} d'exposition ---
une journée de travail à la limite dans une usine bruyante, quelques
minutes seulement aux niveaux du premier rang d'un concert~; un casque
à plein volume se tient franchement dans la bande dangereuse.
L'audition perdue à cause des grandes oscillations ne revient jamais~:
les notes les plus aiguës partent les premières, discrètement,
définitivement. Les oreilles n'ont pas de paupières --- et pas de
pièces de rechange.
\end{remark}

\begin{example}[La vitesse, que ni l'un ni l'autre réglage ne touche]\label{ex:g8:sound-pitch-loudness:speed}
Aucun des deux réglages ne touche à la livraison. Aigus et graves,
forts et doux, tous les \omterm{def:g3:sound-around-us:sound}{sons} montent l'unique relais du milieu à
l'unique allure du milieu --- \qty{340}{m/s} dans l'\omterm{def:g2:air-around-us:air}{air}. La preuve se
joue tous les soirs~: un orchestre entendu de l'autre bout d'un parc
arrive \emph{juste et à l'heure} --- si les notes aiguës allaient plus
vite que les graves, tout accord lointain s'étalerait en arpèges.
(L'intensité sonore, elle, faiblit avec la distance --- les coquilles
qui s'étendent du chapitre du grade 7 --- mais faiblir, c'est
s'affaiblir, jamais ralentir.)
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:1}
Définis la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} et son \omterm{def:g6:measurement-in-science:measuring}{unité}. Quelle est la hauteur du la de
concert, et celle du la situé une octave au-dessus~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:2}
Deux tracés couvrent les mêmes $20$ millisecondes~: l'un montre $4$
oscillations complètes, l'autre $16$. Calcule les deux \omterm{def:g8:sound-pitch-loudness:frequency}{fréquences}~;
lequel sonne le plus aigu~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:3}
Donne la fenêtre de l'audition humaine, et les noms des deux pays qui
s'étendent au-delà de ses frontières --- avec un habitant de chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:4}
Sur l'écran de l'analyseur, qu'est-ce qui change quand tu chantes la
même note plus fort~? Et quand tu chantes une note plus aiguë avec la
même \omterm{def:g2:pushes-and-pulls:force}{force}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:5}
Traduis en langage de \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} les vieilles règles de l'élastique~:
que font au $f$ le fait d'être court, tendu et fin~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:6}
Place sur l'échelle des \omterm{ex:g8:sound-pitch-loudness:decibels}{décibels}~: le murmure des feuilles~; la
circulation~; le premier rang du concert~; la douleur. Où commence la
bande dangereuse~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:7}
Pourquoi les sifflets pour chiens paraissent-ils cassés à leurs
maîtres~? Et pourquoi le chien n'est-il pas d'accord~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:8}
Un orchestre lointain arrive juste et à l'heure. Que prouve cela sur
la \omterm{def:g7:motion-average-speed:formula}{vitesse} du \omterm{def:g3:sound-around-us:sound}{son} selon la hauteur et l'intensité sonore~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:9}
Une chauve-souris émet des clics à \qty{50000}{Hz}. Combien
d'oscillations complètes contient un clic de \qty{0.01}{s}~? Pourquoi
une imagerie fine par \omterm{ex:g4:how-sound-travels:echo}{échos} doit-elle employer un tremblement aussi
rapide~? (Rappelle-toi ce qu'ont en commun les pinceaux les plus
fins.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:10}
La touche la plus grave d'un piano à queue vibre près de
\qty{27}{Hz}~; la plus aiguë, près de \qty{4200}{Hz}. Situe les deux
dans la fenêtre de l'audition (ou contre elle) --- et explique
pourquoi un accordeur de piano vieillissant peut avoir besoin de
l'aide de l'analyseur à l'une des extrémités du clavier d'abord.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:11}
Les règles des concerts autorisent $8$ \omterm{ex:g2:measuring-time:clock}{heures} à \qty{85}{dB} mais
quelques minutes seulement près de \qty{110}{dB}, le budget étant
divisé par deux tous les quelques barreaux. Quel genre de croissance
se cache dans les pas égaux de cette échelle --- et que nous apprend
l'effondrement du budget sur le «~juste un peu plus fort~»~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:sound-pitch-loudness:12}
Conçois la \omterm{def:g3:levers-and-scales:scale}{balance} complète du \omterm{def:g3:sound-around-us:sound}{son} d'une salle scolaire avec un seul
téléphone analyseur~: comment vérifier le la du piano contre $440$~;
comment trouver l'intensité sonore de la salle au dernier rang
pendant un essai de batterie~; comment repérer si la ventilation émet
un grondement infrasonore que l'oreille manque~; et lequel des trois
constats le microphone du téléphone est le \emph{moins} équipé pour
certifier (consulte la propre fenêtre d'audition du téléphone ---
bâtie pour les voix).
\end{exercise}

\section{Problème~: la balance du son}

\begin{problem}[{Problème du week-end --- balance du son à la fête de
l'école~; un analyseur, trois groupes et une infirmière nerveuse~; la
soirée de la table de mixage}]\label{pb:g8:sound-pitch-loudness:1}
Veille de fête~: tu tiens l'analyseur à la table de mixage, avec
l'infirmière scolaire qui surveille le sonomètre par-dessus ton
épaule.

\medskip\noindent\textbf{Partie I --- l'heure de l'accord.}
\begin{enumerate}
  \item Le diapason de la chorale chante à \qty{440}{Hz}~;
        l'analyseur montre le la du piano à \qty{432}{Hz}.
        Diagnostique le piano, et nomme le réglage --- hauteur ou
        intensité sonore --- qui réclame l'accordeur.
  \item La corde la plus grave de la basse indique \qty{41}{Hz}.
        Situe-la dans la fenêtre de l'audition --- et explique
        pourquoi le public la \emph{sentira} dans la poitrine
        autant qu'il l'entendra.
  \item La note tenue du chanteur principal montre des crêtes
        espacées exactement comme celles du diapason, mais trois
        fois plus hautes. Compare les deux \omterm{def:g3:sound-around-us:sound}{sons} sur les deux
        réglages.
  \item Un jeune technicien affirme que les nouvelles enceintes de
        la salle «~reproduisent jusqu'à \qty{22000}{Hz}~». Qui,
        dans le public, pourra certifier le haut de cette étendue
        --- et qui ne le pourra pas~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- le registre de l'infirmière.}
\begin{enumerate}[resume]
  \item Niveaux des répétitions~: set acoustique \qty{75}{dB}~; set
        rock \qty{100}{dB}~; final prévu «~un cheveu au-dessus~» à
        \qty{110}{dB}. Quels sets l'infirmière signale-t-elle, et
        contre quelle ligne de budget~?
  \item L'infirmière distribue des bouchons en mousse annoncés pour
        adoucir les \omterm{def:g3:sound-around-us:sound}{sons} de \qty{20}{dB}. Que devient le set rock
        derrière eux --- et pourquoi les techniciens avisés
        portent-ils des bouchons à tous les concerts~?
  \item En se tenant deux fois plus loin de la scène, le sonomètre
        baisse nettement. Quelle vieille loi des coquilles qui
        s'étendent est à l'œuvre --- et lequel des deux réglages
        (hauteur ou intensité sonore) la distance tourne-t-elle~?
  \item Le batteur proteste~: «~Plus doux veut dire plus terne ---
        les aigus meurent les premiers quand on joue doucement~!~»
        Démêle les deux réglages pour lui~: que change réellement
        le fait de jouer doucement~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- physique de la fête.}
\begin{enumerate}[resume]
  \item Un feu d'artifice clôt le spectacle à \qty{680}{m}. Calcule
        le décalage entre l'éclat et la détonation que remarquera
        la foule --- et la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} du grondement si les grosses
        bombes cognent deux fois par seconde (contrôle de la
        fenêtre~: entendu ou senti~?).
  \item Dernier mystère de la nuit~: une vidéo du concert prise au
        téléphone, une fois rejouée, sonne maigre --- le grondement
        dans la poitrine a disparu. Consulte la mise en garde de
        l'exercice 12~: qu'est-ce que le microphone n'a jamais été
        construit pour attraper~?
  \item Le rapport de l'infirmière demande «~une phrase chacun~»
        sur les deux réglages de la soirée et sur la troisième
        chose qu'aucun des deux réglages ne peut toucher.
        Rédige-les.
  \item Rangement~: le répulsif à nuisibles à \omterm{prop:g8:sound-pitch-loudness:range}{ultrasons} du
        gardien, en marche toute la soirée près de l'entrée des
        artistes, n'a gêné aucun auditeur et, dit-on, beaucoup de
        souris. Explique les deux publics avec une seule
        proposition sur la fenêtre.
\end{enumerate}
\end{problem}
