\chapter{La résistance et la loi d'Ohm}\label{ch:g8:ohms-law}

Une question a suivi toute l'histoire électrique~: certains composants
laissent la marche filer, d'autres l'étranglent jusqu'au filet d'eau
--- les lampes se partagent les \omterm{def:g8:voltage-voltmeter:voltage}{tensions} inégalement, les fils fins
chauffent, les gros restent froids. L'opposition elle-même reçoit
aujourd'hui un nom, une \omterm{def:g6:measurement-in-science:measuring}{unité} et --- couronnement de quatre années de
circuits --- la loi la plus utile de ce livre~: trois lettres qui
nouent la poussée, la marche et l'opposition en une seule ligne
d'algèbre.

\section{La résistance}

\begin{definition}[Résistance]\label{def:g8:ohms-law:resistance}
La \emph{résistance}\index{résistance} $R$ d'un composant mesure avec
quelle \omterm{def:g2:pushes-and-pulls:force}{force} il s'oppose au \omterm{def:g7:circuits-series-parallel:current}{courant électrique}~: sous une \omterm{def:g8:voltage-voltmeter:voltage}{tension}
donnée, plus la résistance est grande, plus faible est la marche
qu'il laisse passer. Son \omterm{def:g6:measurement-in-science:measuring}{unité} est l'\emph{ohm}\index{ohm} (symbole
$\Omega$, la capitale grecque oméga), en l'honneur du
maître d'école devenu physicien qui trouva la loi de ce chapitre avec
des fils faits maison et une patience héroïque.
\end{definition}

\begin{definition}[Résistor]\label{def:g8:ohms-law:resistor}
Un \emph{résistor}\index{résistor} --- on dit aussi un \omterm{def:g4:conductors-insulators:conductor}{conducteur}
ohmique --- est un composant fabriqué pour avoir une \omterm{def:g8:ohms-law:resistance}{résistance}
choisie et stable~: un obstacle étalonné, vendu de la fraction d'\omterm{def:g8:ohms-law:resistance}{ohm}
au million d'\omterm{def:g8:ohms-law:resistance}{ohms}. Les circuits emploient les résistors pour régler
les courants à dessein~: dompter la marche qui brûlerait une pièce
délicate, atténuer, diviser, ajuster. Leurs anneaux de couleur
épellent leur valeur~; leur symbole est un simple rectangle.
\end{definition}

\begin{example}[Des résistances autour de toi]\label{ex:g8:ohms-law:values}
Un \omterm{def:g2:measuring-length:units}{mètre} de gros fil de cuivre~: des centièmes d'\omterm{def:g8:ohms-law:resistance}{ohm} --- la route
plate. Une ampoule de lampe de poche allumée~: quelques \omterm{def:g8:ohms-law:resistance}{ohms}. Un petit
\omterm{ex:g6:circuits-and-safety:motor}{moteur} électrique~: quelques \omterm{def:g8:ohms-law:resistance}{ohms}. La \omterm{def:g8:ohms-law:resistance}{résistance} chauffante d'une
bouilloire~: environ $25\,\Omega$. La peau humaine sèche, d'une main à
l'autre~: des dizaines de milliers d'\omterm{def:g8:ohms-law:resistance}{ohms} --- mouillée,
catastrophiquement moins~: la vieille règle des mains humides est un
énoncé de \omterm{def:g8:ohms-law:resistance}{résistance}. Un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} ouvert~: une \omterm{def:g8:ohms-law:resistance}{résistance}
au-delà de toute mesure --- l'obstacle infini.
\end{example}

\section{L'expérience}

\begin{method}[Relever le portrait $U$--$I$ d'un composant]\label{met:g8:ohms-law:portrait}
Un \omterm{def:g8:ohms-law:resistor}{résistor}, une alimentation réglable (ou une file croissante de
\omterm{def:g3:first-electric-circuit:battery}{piles}), les deux appareils~:
\begin{enumerate}
  \item brancher l'alimentation, l'\omterm{def:g8:intensity-ammeter:ammeter}{ampèremètre} et le \omterm{def:g8:ohms-law:resistor}{résistor} \omterm{def:g4:series-circuits:series}{en
        série}~; poser le \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} aux \omterm{def:g3:first-electric-circuit:battery}{bornes} du \omterm{def:g8:ohms-law:resistor}{résistor}~;
  \item régler la plus petite \omterm{def:g8:voltage-voltmeter:voltage}{tension}~; noter le couple ($U$, $I$)~;
  \item augmenter la \omterm{def:g8:voltage-voltmeter:voltage}{tension} pas à pas --- une \omterm{def:g3:first-electric-circuit:battery}{pile}, deux, trois ---
        en notant un couple à chaque fois~;
  \item placer les points, $I$ en abscisse, $U$ en ordonnée, et
        regarder.
\end{enumerate}
\end{method}

\begin{example}[L'aveu d'un résistor]\label{ex:g8:ohms-law:confession}
Le tableau d'une séance, pour un \omterm{def:g8:ohms-law:resistor}{résistor}~:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
Double la poussée, double la marche~; triple-la, elle triple~: $U$ et
$I$ sont \emph{proportionnels}, et chaque rapport $U/I$ redonne le
même nombre~: $15$. Ce rapport constant n'est pas un hasard --- c'est
la \omterm{def:g8:ohms-law:resistance}{résistance} du \omterm{def:g8:ohms-law:resistor}{résistor}, $R = 15\,\Omega$, qui se montre à chaque
ligne.
\end{example}

\begin{proposition}[Loi d'Ohm]\label{prop:g8:ohms-law:law}
Pour un \omterm{def:g8:ohms-law:resistor}{résistor} à \omterm{def:g3:temperature-thermometers:temperature}{température} constante, la \omterm{def:g8:voltage-voltmeter:voltage}{tension} à ses \omterm{def:g3:first-electric-circuit:battery}{bornes} et
le courant qui le traverse sont proportionnels, et la constante de
proportionnalité est sa \omterm{def:g8:ohms-law:resistance}{résistance}~:
\[
U = R \times I ,
\]
avec $U$ en \omterm{def:g8:voltage-voltmeter:voltage}{volts}, $I$ en \omterm{def:g8:intensity-ammeter:intensity}{ampères}, $R$ en \omterm{def:g8:ohms-law:resistance}{ohms}. Réarrangée à
volonté~: $I = U/R$ (le courant qu'une poussée entraîne à travers un
obstacle) et $R = U/I$ (l'obstacle démasqué par un seul couple de
lectures honnêtes).
\end{proposition}

\begin{example}[La loi calcule tout]\label{ex:g8:ohms-law:compute}
Une fois $R$ connu, la loi répond dans tous les sens. La \omterm{def:g8:ohms-law:resistance}{résistance}
chauffante de la bouilloire, $25\,\Omega$ sur le secteur à
\qty{230}{V}~: $I = U/R = 230 \div 25 = \qty{9.2}{A}$ --- l'appareil
à dix \omterm{def:g8:intensity-ammeter:intensity}{ampères} s'explique. Un \omterm{def:g8:ohms-law:resistor}{résistor} doit limiter à \qty{0.02}{A} le
courant d'une lampe délicate depuis \qty{4.5}{V}~:
$R = U/I = 4.5 \div 0.02 = 225\,\Omega$ --- on prendra la valeur
normalisée immédiatement supérieure. Dans $60\,\Omega$ passent
\qty{0.05}{A}~: la \omterm{def:g8:voltage-voltmeter:voltage}{tension} à ses \omterm{def:g3:first-electric-circuit:battery}{bornes} vaut
$U = 60 \times 0.05 = \qty{3}{V}$. Une loi, trois recettes, tout
l'atelier.
\end{example}

\begin{remark}[La béquille du triangle]\label{rem:g8:ohms-law:triangle}
Certains élèves crayonnent $U$ au-dessus de $R$ et $I$ dans un petit
triangle et masquent la lettre cherchée pour y lire la recette.
Inoffensif comme béquille --- mais ton algèbre réarrange désormais
$U = R I$ honnêtement en une ligne, et la manière honnête voyage
jusqu'à toutes les formules que tu rencontreras jamais, triangle ou
non. Appuie-toi sur la béquille cette année s'il le faut~; prévois de
marcher.
\end{remark}

\section{Ce que la résistance explique}

\begin{example}[De vieux mystères sur une seule facture]\label{ex:g8:ohms-law:mysteries}
Le partage inégal des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} du chapitre des lois~: des lampes \omterm{def:g4:series-circuits:series}{en
série} partagent un même $I$, donc par $U = R I$ le plus grand $R$
prend le plus grand $U$ --- la plus grande chute du travailleur le
plus rude, désormais calculable. La \omterm{def:g5:series-parallel-circuits:parallel}{branche} étranglée du chapitre de
la dérivation~: le long fil fin ajoutait de la \omterm{def:g8:ohms-law:resistance}{résistance}, et
$I = U/R$ rapetissait la part de cette \omterm{def:g5:series-parallel-circuits:parallel}{branche}. Le \omterm{def:g7:short-circuits-safety:device}{court-circuit}~:
une déviation de $R$ presque nul, donc $I = U/R$ explose --- la
bousculade a toujours été une division par presque zéro. Quatre
années de savoir qualitatif sur les circuits, la facture d'une seule
loi.
\end{example}

\begin{example}[Les fils, au mètre]\label{ex:g8:ohms-law:wires}
La \omterm{def:g8:ohms-law:resistance}{résistance} d'un fil croît avec sa longueur et décroît avec son
épaisseur --- un fil long et fin est une route de montagne étroite,
un fil court et gros une autoroute. D'où les gros câbles de
l'électricien pour les appareils voraces, l'alliage résistant
délibérément long et fin de la \omterm{def:g8:ohms-law:resistance}{résistance} chauffante --- et la raison
pour laquelle la même loi qui allume une lampe échauffe un
grille-pain~: la \omterm{def:g8:ohms-law:resistance}{résistance} est l'endroit où la marche dépense sa
poussée, et la poussée dépensée, comme le confirmera le chapitre sur
l'\omterm{def:g5:everyday-energy:energy}{énergie} l'an prochain, devient de la \omterm{def:g5:heat-and-insulation:heat}{chaleur}.
\end{example}

\begin{remark}[Où s'arrête l'autorité de la loi]\label{rem:g8:ohms-law:limits}
La loi d'Ohm est la loi d'un \omterm{def:g8:ohms-law:resistor}{résistor}, non un droit universel. Relève
le portrait d'une \emph{lampe à filament} et les points s'écartent de
la droite~: le filament s'échauffe à mesure que le courant croît, et
le métal chaud résiste davantage --- $R$ refuse de rester en place.
Les diodes, les \omterm{ex:g6:circuits-and-safety:motor}{moteurs} et la peau vivante courbent aussi leurs
portraits, chacun à sa façon. La droite passant par l'origine est la
signature du \omterm{def:g8:ohms-law:resistor}{résistor}, non celle de la matière --- demande toujours à
un composant son portrait avant de lui confier la loi.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
Que mesure la \omterm{def:g8:ohms-law:resistance}{résistance}~? Donne son \omterm{def:g6:measurement-in-science:measuring}{unité} et son symbole, et écris
la loi d'Ohm avec ses trois écritures.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
Complète chaque couple pour un \omterm{def:g8:ohms-law:resistor}{résistor} de $30\,\Omega$~:
$U = \qty{6}{V}$, $I = {?}$~; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
Un composant présente $U = \qty{4.5}{V}$ sous $I = \qty{0.09}{A}$.
Sa \omterm{def:g8:ohms-law:resistance}{résistance}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
Dans l'expérience du portrait, quelle forme dessinent les points d'un
vrai \omterm{def:g8:ohms-law:resistor}{résistor}, et que signifie une droite plus raide~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
À partir du tableau du chapitre, vérifie $R$ sur la deuxième et la
quatrième colonne. Pourquoi obtenir deux fois le même nombre est-il
tout l'enjeu~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
La \omterm{def:g8:ohms-law:resistance}{résistance} chauffante d'une bouilloire vaut $25\,\Omega$~; celle
d'une seconde bouilloire, $35\,\Omega$. Sur le même secteur, laquelle
appelle le courant le plus fort --- et combien appelle chacune (une
décimale)~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
Explique avec $U = R I$ pourquoi, de deux lampes \omterm{def:g4:series-circuits:series}{en série} partageant
un même courant, celle qui résiste le plus prend la plus grande
\omterm{def:g8:voltage-voltmeter:voltage}{tension}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
Traduis en langage de \omterm{def:g8:ohms-law:resistance}{résistance}~: la bousculade du \omterm{def:g7:short-circuits-safety:device}{court-circuit}~;
\ le blocage total de l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} ouvert.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
Un voyant lumineux délicat tolère \qty{20}{mA}. Quel \omterm{def:g8:ohms-law:resistor}{résistor} \omterm{def:g4:series-circuits:series}{en
série} le protège sur une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{9}{V}, si le voyant prend
lui-même environ \qty{2}{V} de la part~? (La part du \omterm{def:g8:ohms-law:resistor}{résistor}
d'abord, puis $R$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
Portrait mesuré d'une lampe à filament~: (\qty{1}{V},
\qty{0.25}{A}), (\qty{4}{V}, \qty{0.5}{A}). Montre qu'aucun $R$
unique ne convient aux deux points, et raconte l'histoire physique
derrière cette courbure.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
Deux \omterm{def:g8:ohms-law:resistor}{résistors} \omterm{def:g4:series-circuits:series}{en série}, $10\,\Omega$ et $20\,\Omega$, sous
\qty{6}{V}. En utilisant les lois de la série puis la loi d'Ohm sur
chacun~: trouve le courant de la boucle et la \omterm{def:g8:voltage-voltmeter:voltage}{tension} de chaque
\omterm{def:g8:ohms-law:resistor}{résistor}. (Indice~: un seul courant $I$ vérifie
$6 = 10 I + 20 I$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
Les deux mêmes \omterm{def:g8:ohms-law:resistor}{résistors}, maintenant \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} sous \qty{6}{V}.
Trouve le courant de chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche}, le total fourni par la \omterm{def:g3:first-electric-circuit:battery}{pile},
puis --- la fin élégante --- la \omterm{def:g8:ohms-law:resistance}{résistance} unique $R$ qui appellerait
ce même total sous \qty{6}{V}. Compare-la aux deux \omterm{def:g8:ohms-law:resistor}{résistors} et
explique, en langage de routes, pourquoi l'équipe de deux résiste
\emph{moins} que chacun de ses membres seul.
\end{exercise}

\section{Problème~: le banc de contrôle qualité}

\begin{problem}[{Problème du week-end --- jour d'inspection à
l'usine de résistors~; des portraits, des tolérances et un composant
qui n'est pas du tout un résistor}]\label{pb:g8:ohms-law:1}
Le banc qualité de l'usine éprouve la production du jour --- et un
imposteur. C'est toi qui tiens les appareils.

\medskip\noindent\textbf{Partie I --- étalonner le banc.}
\begin{enumerate}
  \item Dessine (ou décris) le circuit d'essai~: alimentation,
        \omterm{def:g8:intensity-ammeter:ammeter}{ampèremètre}, composant testé, \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} --- qui est \omterm{def:g4:series-circuits:series}{en
        série}, qui est aux \omterm{def:g3:first-electric-circuit:battery}{bornes}~?
  \item L'échantillon un, sous \qty{4.5}{V}, laisse passer
        \qty{0.15}{A}. Sa \omterm{def:g8:ohms-law:resistance}{résistance}~?
  \item L'étiquette annonce $33\,\Omega$, tolérance cinq pour cent.
        Calcule la fourchette acceptable, et statue sur
        l'échantillon un. (Cinq pour cent de $33$ d'abord.)
  \item L'échantillon deux, étiqueté $150\,\Omega$~: prédis son
        courant sous \qty{4.5}{V}, pour que le banc sache à quoi
        s'attendre (une décimale, en \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{Partie II --- des portraits complets.}
\begin{enumerate}[resume]
  \item Tableau de l'échantillon trois~: $(1.5, 0.05)$,
        $(3.0, 0.10)$, $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{volts},
        \omterm{def:g8:intensity-ammeter:intensity}{ampères}. Ajuste $R$, en citant le trait des nombres qui
        autorise une valeur unique.
  \item Tableau de l'échantillon quatre~: $(1.5, 0.30)$,
        $(3.0, 0.45)$, $(4.5, 0.54)$, $(6.0, 0.60)$. Démasque
        l'imposteur~: aucun $R$ unique, et le sens de la dérive.
  \item L'échantillon quatre est en fait une petite lampe à
        filament venue de la chaîne voisine. Raconte l'histoire de
        sa courbure --- et pourquoi l'essai de la droite, sur ce
        banc, est précisément un détecteur de \omterm{def:g8:ohms-law:resistor}{résistors}.
  \item Le règlement du banc dit~: «~tout portrait doit être relevé
        à \omterm{def:g3:temperature-thermometers:temperature}{température} constante.~» Quelle clause de la loi d'Ohm
        cette règle protège-t-elle~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- le service des
applications.}
\begin{enumerate}[resume]
  \item Une commande demande un \omterm{def:g8:ohms-law:resistor}{résistor} limitant le courant à
        \qty{30}{mA} sur une alimentation de \qty{12}{V} (la part
        de la charge elle-même étant négligeable). Calcule la
        valeur à proposer.
  \item Un client se plaint que sa \omterm{def:g8:ohms-law:resistance}{résistance} chauffante de
        $25\,\Omega$ n'appelle «~que~» \qty{9.2}{A} alors que le
        \omterm{def:g7:short-circuits-safety:fuse}{fusible} en autorise $10$. Calcule la \omterm{def:g8:voltage-voltmeter:voltage}{tension} d'alimentation
        que sa plainte suppose, et rassure-le par l'arithmétique.
  \item Le laboratoire des prototypes réclame le cavalier de cuivre
        le plus gros et le plus court de l'usine, «~avec une
        \omterm{def:g8:ohms-law:resistance}{résistance} aussi proche de zéro \omterm{def:g8:ohms-law:resistance}{ohm} que possible~». Pour
        quel rôle de circuit ce cavalier est-il né --- et de quel
        coupable familier sa \omterm{def:g8:ohms-law:resistance}{résistance} presque nulle le rapproche
        s'il est mal employé~?
  \item Clos le registre d'inspection~: trois phrases --- la loi,
        son portrait, et l'unique condition sous laquelle un
        composant a le droit de la signer.
\end{enumerate}
\end{problem}
