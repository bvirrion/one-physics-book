\chapter{Las lentes y las imágenes}\label{ch:g8:lenses-images}

La cámara estenopeica compraba su \omterm{prop:g5:mirrors-reflection:image}{imagen} pequeña y nítida a un
precio cruel: casi nada de \omterm{def:g1:light-and-shadows:source}{luz}. Y sin embargo tu propio ojo pinta
todo el día un mundo brillante y nítido en su pared del fondo, y no se
ve ningún agujerito por ninguna parte. El secreto del ojo es un disco
de materia vidriosa y curvada que recoge un \omterm{def:g6:light-propagation:ray}{haz} entero y lo dobla a
la orden --- la \omterm{def:g8:lenses-images:families}{lente}, \omterm{def:g4:solar-system:star}{estrella} de las cámaras, de las gafas,
de las lupas y de este capítulo.

\section{Dos familias de lentes}

\begin{definition}[Lentes convergentes y divergentes]\label{def:g8:lenses-images:families}
Una \emph{lente}\index{lente} es un trozo de material
\omterm{def:g7:light-sources-propagation:coats}{transparente} --- vidrio o plástico claro --- con las caras
curvadas, construido para doblar la \omterm{def:g1:light-and-shadows:source}{luz} de manera controlada (la
\omterm{def:g1:light-and-shadows:source}{luz} se dobla al pasar al vidrio; la ley exacta de ese doblez es un
tesoro del bachillerato). Dos familias:
\begin{enumerate}
  \item las \emph{lentes convergentes}\index{lente convergente} ---
        más gruesas por el centro que por el borde --- doblan los
        rayos de un \omterm{def:g6:light-propagation:ray}{haz} \emph{unos hacia otros};
  \item las \emph{lentes divergentes}\index{lente divergente} ---
        más delgadas por el centro --- los separan.
\end{enumerate}
Pasa la yema de un dedo por el cristal de unas gafas: un abultamiento
central o un hueco central clasifica en su familia a todas las lentes
de la Tierra.
\end{definition}

\begin{definition}[Foco y distancia focal]\label{def:g8:lenses-images:focus}
Manda un manojo de rayos paralelos --- la \omterm{def:g1:light-and-shadows:source}{luz} del Sol sirve --- a
través de una \omterm{def:g8:lenses-images:families}{lente convergente}: los rayos doblados se cruzan
todos en un punto por detrás de ella, el \emph{foco}\index{foco}. La
distancia de la \omterm{def:g8:lenses-images:families}{lente} al foco es la \emph{distancia
focal}\index{distancia focal} $f$: el único número que define a la
\omterm{def:g8:lenses-images:families}{lente}. Las \omterm{def:g8:lenses-images:families}{lentes} muy curvadas y gordas doblan con fuerza --- $f$
corta; las poco curvadas doblan poco --- $f$ larga.
\end{definition}

\begin{method}[Medir una distancia focal]\label{met:g8:lenses-images:measure}
Para cualquier \omterm{def:g8:lenses-images:families}{lente convergente}, un minuto y una ventana
lejana:
\begin{enumerate}
  \item ponte de cara a una pared enfrente de una escena
        \emph{lejana} y brillante --- la ventana y el mundo de más
        allá;
  \item sujeta la \omterm{def:g8:lenses-images:families}{lente} cerca de la pared y sepárala despacio
        hasta que la escena salte a un cuadrito nítido sobre la
        pared (del revés --- como ya era de esperar);
  \item mide la distancia de la \omterm{def:g8:lenses-images:families}{lente} a la pared: para una
        escena lejana, esa distancia \emph{es} la \omterm{def:g8:lenses-images:focus}{distancia
        focal} --- los rayos de lejos llegan prácticamente paralelos,
        y los rayos paralelos se juntan en el \omterm{def:g8:lenses-images:focus}{foco}.
\end{enumerate}
Una lupa mide unos \qty{10}{cm}; los cristales de unas gafas,
decenas de \omterm{def:g2:measuring-length:units}{centímetros}; la \omterm{def:g8:lenses-images:families}{lente} rechoncha de una cámara,
unos pocos.
\end{method}

\begin{remark}[La lupa que quema]\label{rem:g8:lenses-images:burning}
En el \omterm{def:g8:lenses-images:focus}{foco} de una \omterm{def:g8:lenses-images:families}{lente convergente}, todo un \omterm{def:g6:light-propagation:ray}{haz} de \omterm{def:g1:light-and-shadows:source}{luz}
del Sol se apretuja en un punto deslumbrante --- lo bastante
concentrado como para chamuscar papel y empezar un incendio en unos
segundos. Trátalo por lo que es: el principio de la cocina solar en
formato de bolsillo. No concentres nunca la \omterm{def:g1:light-and-shadows:source}{luz} del Sol cerca de nada
inflamable, ni hacia la piel o la ropa --- y, por dos veces la vieja
ley, no mires nunca al Sol \emph{a través} de ninguna
\omterm{def:g8:lenses-images:families}{lente}: el enfoque propio del ojo más el de la \omterm{def:g8:lenses-images:families}{lente}
producen una quemadura instantánea, indolora y permanente.
\end{remark}

\section{La lente como fabricante de imágenes}

\begin{proposition}[Imágenes reales]\label{prop:g8:lenses-images:realimage}
Una \omterm{def:g8:lenses-images:families}{lente convergente} recoge los rayos difundidos por cada punto
de un objeto y los entrega a un punto correspondiente del otro lado:
punto a punto, monta un cuadro genuino --- una \emph{imagen
real}\index{imagen real} --- que se puede recoger en una pantalla. La
\omterm{prop:g5:mirrors-reflection:image}{imagen} está invertida, como la del agujerito, pero brillante: la
\omterm{def:g8:lenses-images:families}{lente} cosecha una \omterm{def:g8:lenses-images:families}{lente} entera de \omterm{def:g1:light-and-shadows:source}{luz} por punto allí
donde el agujerito dejaba pasar un solo hilo. Reglas prácticas de la
geometría: un objeto lejano da una \omterm{prop:g5:mirrors-reflection:image}{imagen} pequeña y cerca del
\omterm{def:g8:lenses-images:focus}{foco}; a medida que el objeto se acerca a la \omterm{def:g8:lenses-images:families}{lente}, su
\omterm{prop:g5:mirrors-reflection:image}{imagen} retrocede y crece.
\end{proposition}

\begin{example}[La cámara y el ojo, una sola arquitectura]\label{ex:g8:lenses-images:cameraeye}
Una cámara es el fabricante de imágenes metido en una caja:
\omterm{def:g8:lenses-images:families}{lente convergente} delante, pantalla sensible a la \omterm{def:g1:light-and-shadows:source}{luz} detrás e
\omterm{prop:g5:mirrors-reflection:image}{imagen} formada invertida sobre ella (el programa gira el archivo
con toda educación). Tu ojo es la misma máquina en tejido vivo: una
\omterm{def:g8:lenses-images:families}{lente convergente} detrás de la pupila, la \emph{retina} como
pantalla al fondo e \omterm{prop:g5:mirrors-reflection:image}{imagen} invertida --- el cerebro le da la
vuelta, como ya se dijo cuando nos lo enseñó por primera vez el
agujerito. El enfoque difiere de manera preciosa: la cámara mueve su
\omterm{def:g8:lenses-images:families}{lente}; la \omterm{def:g8:lenses-images:families}{lente} blanda del ojo \emph{cambia de forma},
estirada hasta engordar para las cosas cercanas y relajada hasta
adelgazar para las lejanas --- nota el esfuerzo después de una hora de
lectura de cerca.
\end{example}

\begin{example}[Las gafas]\label{ex:g8:lenses-images:spectacles}
Cuando el enfoque propio del ojo se queda corto, una \omterm{def:g8:lenses-images:families}{lente}
delante lo completa. El ojo miope enfoca con demasiada \omterm{def:g2:pushes-and-pulls:force}{fuerza} --- las
escenas lejanas se montan \emph{antes} de la retina y llegan borrosas:
una \omterm{def:g8:lenses-images:families}{lente} \emph{divergente} en las gafas desdobla la
\omterm{def:g1:light-and-shadows:source}{luz} justo lo necesario. Los ojos hipermétropes (y casi todos los
ojos mayores) enfocan con demasiada poca \omterm{def:g2:pushes-and-pulls:force}{fuerza} para el trabajo de
cerca: una \omterm{def:g8:lenses-images:families}{lente} \emph{convergente} presta la \omterm{def:g2:pushes-and-pulls:force}{fuerza} que falta
--- las gafas de leer son primas honradas y pequeñas de la lupa. Dos
familias de \omterm{def:g8:lenses-images:families}{lentes}, dos familias de borrosidad.
\end{example}

\section{La lupa}

{\small La lupa en funcionamiento: sujeta más cerca que su \omterm{def:g8:lenses-images:focus}{distancia focal}, la \omterm{def:g8:lenses-images:families}{lente} enseña la letra derecha y agrandada.}
\end{omfigure}

\begin{example}[Por dentro de la distancia focal]\label{ex:g8:lenses-images:magnifier}
Sujeta una \omterm{def:g8:lenses-images:families}{lente convergente} \emph{más cerca} de un sello que su
\omterm{def:g8:lenses-images:focus}{distancia focal} y el cuadro cambia de naturaleza: no aterriza
ninguna \omterm{prop:g5:mirrors-reflection:image}{imagen} en ninguna pantalla --- en cambio, mirando a
través de la \omterm{def:g8:lenses-images:families}{lente}, ves el sello derecho, en su sitio y
agrandado. La \omterm{def:g8:lenses-images:families}{lente} dobla los rayos del sello de modo que lleguen
al ojo \emph{como si} los hubiera difundido un sello más grandioso
situado detrás del vidrio --- una ilusión hecha a medida, prima del
mundo fantasma del \omterm{def:g5:mirrors-reflection:reflection}{espejo}. Los relojeros, los coleccionistas de
sellos y tú mismo de pequeño con aquella lupa la habéis usado toda la
vida; la geometría completa de los rayos de este truco es una joya del
volumen de bachillerato.
\end{example}

\begin{remark}[Instrumentos sobre instrumentos]\label{rem:g8:lenses-images:instruments}
Apila \omterm{def:g8:lenses-images:families}{lentes} y el mundo se abre. Dos \omterm{def:g8:lenses-images:families}{lentes convergentes} hacen un
microscopio: una lanza una \omterm{prop:g8:lenses-images:realimage}{imagen real} grande de una gotita y la
segunda agranda esa \omterm{prop:g5:mirrors-reflection:image}{imagen} --- así consiguieron sus primeros
aliados directos los defensores de las \omterm{def:g7:states-of-matter:molecule}{moléculas}. Una
\omterm{def:g8:lenses-images:families}{lente} grande (o un \omterm{def:g5:mirrors-reflection:reflection}{espejo} curvo) más una lupa hacen un
telescopio: los débiles rayos paralelos de una galaxia lejana,
recogidos en una \omterm{prop:g8:lenses-images:realimage}{imagen real} y agrandados después hacia un ojo
asombrado. Todos los instrumentos ópticos de la civilización son las
dos ideas de este capítulo --- recoger y después agrandar --- en
disposiciones diversas.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:1}
Clasifica al tacto: ¿cómo distingue la yema de un dedo una \omterm{def:g8:lenses-images:families}{lente
convergente} de una divergente? ¿Qué le hace cada una a un
\omterm{def:g6:light-propagation:ray}{haz} paralelo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:2}
Define \omterm{def:g8:lenses-images:focus}{foco} y \omterm{def:g8:lenses-images:focus}{distancia focal}. ¿Qué dice una
\omterm{def:g8:lenses-images:focus}{distancia focal} corta sobre la curvatura de una
\omterm{def:g8:lenses-images:families}{lente} y sobre su poder de doblar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:3}
Describe la medida de $f$ con la ventana y la pared. ¿Por qué tiene que
estar lejos la escena?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:4}
Enuncia dos reglas de seguridad de la lupa que quema, y la razón de
existir de cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:5}
¿Qué es una \omterm{prop:g8:lenses-images:realimage}{imagen real}? Da sus dos propiedades características y
la ventaja de la \omterm{def:g8:lenses-images:families}{lente} sobre el agujerito.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:6}
Traza el mapa de la cámara sobre el ojo: ¿qué hace de \omterm{def:g8:lenses-images:families}{lente}, de
pantalla y de mecanismo de enfoque en cada uno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:7}
¿Qué familia de cristales sirve al ojo miope, y por qué? ¿Y al ojo
hipermétrope que envejece?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:8}
¿Qué tiene que cumplir la posición del sello para el truco de la lupa
--- y de qué clase de ``\omterm{prop:g5:mirrors-reflection:image}{imagen}'' disfruta entonces el ojo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:lenses-images:9}
El modelo de un fotógrafo camina hacia la cámara. ¿Hacia dónde tiene
que moverse la \omterm{def:g8:lenses-images:families}{lente} para mantener nítida la \omterm{prop:g5:mirrors-reflection:image}{imagen} sobre
el sensor, según las reglas prácticas? ¿Qué hace en cambio el ojo en la
misma situación?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:lenses-images:10}
Un día soleado, una \omterm{def:g8:lenses-images:families}{lente} dejada de cualquier manera sobre unas
hojas secas es un incendio a punto de ocurrir --- pero solo a una
distancia concreta entre la \omterm{def:g8:lenses-images:families}{lente} y las hojas. ¿Qué distancia, y por
qué exactamente ahí?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:lenses-images:11}
Las gafas de leer del abuelo hacen también de lupa de emergencia y, en
vacaciones, de encendedor solar de puros --- pero las gafas de su nieto
miope no hacen ni lo uno ni lo otro. Explica los dos hechos de familia.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:lenses-images:12}
El microscopio y el telescopio apilan los dos \omterm{def:g8:lenses-images:families}{lentes
convergentes} --- y sin embargo uno agranda gotitas y el otro galaxias.
Contrasta sus dos etapas: qué recibe y qué produce la \omterm{def:g8:lenses-images:families}{lente} uno
en cada instrumento (objeto cercano frente a \omterm{def:g1:light-and-shadows:source}{luz} de \omterm{def:g4:solar-system:star}{estrella}
prácticamente paralela), y por qué el telescopio quiere ante todo una
\omterm{def:g8:lenses-images:families}{lente} uno \emph{grande}, mientras que el microscopio la quiere
\emph{muy curvada}.
\end{exercise}

\section{Problema: el banco del óptico}

\begin{problem}[{Problema de fin de semana --- de aprendiz con el
óptico del pueblo; distancias focales, adaptaciones y un telescopio
roto}]\label{pb:g8:lenses-images:1}
Un día en la óptica: \omterm{def:g8:lenses-images:families}{lentes} que clasificar, gafas que emparejar y el
telescopio de un cliente hecho piezas.

\medskip\noindent\textbf{Parte I --- Clasificar la bandeja.}
Un cajón volcado ha mezclado las existencias.
\begin{enumerate}
  \item Propón dos pruebas rápidas --- una al tacto y otra con la
        ventana y la pared --- para separar las convergentes de las
        \omterm{def:g8:lenses-images:families}{lentes divergentes}.
  \item La \omterm{def:g8:lenses-images:families}{lente} A lanza sobre la pared una \omterm{prop:g5:mirrors-reflection:image}{imagen}
        nítida de la calle a \qty{25}{cm}. ¿Su \omterm{def:g8:lenses-images:focus}{distancia focal}
        y su familia?
  \item La \omterm{def:g8:lenses-images:families}{lente} B se niega a lanzar ninguna \omterm{prop:g5:mirrors-reflection:image}{imagen}
        sobre la pared a ninguna distancia, y las cosas vistas a
        través de ella se ven más pequeñas. ¿Su familia?
  \item Las \omterm{def:g8:lenses-images:families}{lentes} C y D son las dos convergentes; C hace su
        \omterm{prop:g5:mirrors-reflection:image}{imagen} de la ventana a \qty{10}{cm} y D a \qty{40}{cm}.
        ¿Cuál es la \omterm{def:g8:lenses-images:families}{lente} más curvada y que dobla con más
        \omterm{def:g2:pushes-and-pulls:force}{fuerza}?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Adaptaciones.}
\begin{enumerate}[resume]
  \item Un cliente ve borrosas las señales de tráfico lejanas pero
        lee las cartas de los restaurantes sin esfuerzo. Diagnostica
        el ojo y receta la familia.
  \item Otro sostiene el periódico con el brazo estirado y se queja
        de que tiene los brazos ``demasiado cortos''. Diagnostica y
        receta.
  \item El óptico avisa al segundo cliente: ``sus gafas nuevas de
        leer le van a emborronar las montañas''. ¿Por qué las gafas
        de leer son solo para las páginas cercanas?
  \item Una niña pregunta por qué las \omterm{def:g8:lenses-images:families}{lentes} de prueba del óptico
        hacen que el panel de letras se le vaya, si sus ojos están
        ``bien''. ¿Qué le hace una \omterm{def:g8:lenses-images:families}{lente} correctora a un ojo
        bien enfocado?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El telescopio roto.}
Un cliente trae un telescopio de mirar \omterm{def:g4:solar-system:star}{estrellas} con la \omterm{def:g8:lenses-images:families}{lente}
delantera partida y pregunta si la lupa de repuesto de la caja puede
sustituirla.
\begin{enumerate}[resume]
  \item Nombra los oficios de las dos \omterm{def:g8:lenses-images:families}{lentes} en el instrumento sano:
        ¿qué recibe de una \omterm{def:g4:solar-system:star}{estrella} la gran \omterm{def:g8:lenses-images:families}{lente} delantera
        y qué hace con ello, y qué hace con eso la pequeña
        \omterm{def:g8:lenses-images:families}{lente} ocular?
  \item La \omterm{def:g8:lenses-images:families}{lente} delantera es ancha como un platillo; la
        \omterm{def:g8:lenses-images:families}{lente} ocular, del tamaño de una moneda. ¿Por qué importa
        tanto la \emph{anchura} de la \omterm{def:g8:lenses-images:families}{lente} delantera para las
        \omterm{def:g4:solar-system:star}{estrellas} débiles?
  \item La lupa de repuesto está muy curvada, pero es del tamaño de
        una moneda. Juzga el recambio: ¿qué se ganaría y qué se
        perdería en la parte delantera del instrumento?
  \item Cierra el día con el credo de una línea del óptico sobre
        todos los instrumentos del estante --- gafas, lupas, cámaras
        y telescopios --- y el resumen de dos verbos de lo que hacen
        las \omterm{def:g8:lenses-images:families}{lentes} por la civilización.
\end{enumerate}
\end{problem}
