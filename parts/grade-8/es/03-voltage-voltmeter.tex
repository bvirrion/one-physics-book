\chapter{La tensión y el voltímetro}\label{ch:g8:voltage-voltmeter}

Desde que la primera \omterm{def:g3:first-electric-circuit:battery}{pila} encendió la primera
\omterm{def:g3:first-electric-circuit:bulb}{bombilla}, una palabra ha hecho en este libro un servicio pesado y
vago: el ``empujón''. Las \omterm{def:g3:first-electric-circuit:battery}{pilas} empujan flojo, la red empuja de manera
asesina, dos \omterm{def:g3:first-electric-circuit:battery}{pilas} puestas en fila empujan juntas. El empujón se
convierte ahora en una magnitud medida con una \omterm{def:g6:measurement-in-science:measuring}{unidad} famosa --- y
su instrumento, al contrario que el del capítulo anterior, se niega a
plantarse en la carretera.

\section{La tensión}

\begin{definition}[Tensión]\label{def:g8:voltage-voltmeter:voltage}
La \emph{tensión}\index{tensión} $U$ entre dos puntos de un circuito
mide el empujón eléctrico disponible entre ellos --- con cuánta
\omterm{def:g2:pushes-and-pulls:force}{fuerza} se impulsa la corriente de un punto al otro, igual que la altura
de una cascada mide con cuánta \omterm{def:g2:pushes-and-pulls:force}{fuerza} mueve esta la rueda del
molino. La tensión es siempre un \emph{entre}: entre los dos
\omterm{def:g3:first-electric-circuit:battery}{bornes} de una \omterm{def:g3:first-electric-circuit:battery}{pila}, entre los dos lados de una lámpara
--- nunca en un solo punto aislado. Su \omterm{def:g6:measurement-in-science:measuring}{unidad} es el
\emph{voltio}\index{voltio} (\unit{V}), en honor del inventor de la
primera \omterm{def:g3:first-electric-circuit:battery}{pila}.
\end{definition}

\begin{example}[El álbum de familia del voltio]\label{ex:g8:voltage-voltmeter:values}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} redonda suelta: \qty{1.5}{V}. La \omterm{def:g3:first-electric-circuit:battery}{pila} plana de
bolsillo: \qty{4.5}{V} --- tres \omterm{def:g3:first-electric-circuit:battery}{pilas} en fila dentro de una sola
envoltura. La rectangular de nueve \omterm{def:g8:voltage-voltmeter:voltage}{voltios} con los dos
\omterm{def:g3:first-electric-circuit:battery}{bornes} arriba: \qty{9}{V}, seis \omterm{def:g3:first-electric-circuit:battery}{pilas} escondidas. Una
batería de coche: \qty{12}{V}. El enchufe de la pared:
\qty{230}{V} --- más empujón que el de ciento cincuenta pilitas, que es
todo el contenido cuantitativo de ``al enchufe, jamás''. Un rayo entre
la nube y el suelo: cientos de millones de \omterm{def:g8:voltage-voltmeter:voltage}{voltios}, un instante.
\end{example}

\begin{proposition}[Tensión sin corriente]\label{prop:g8:voltage-voltmeter:nocurrent}
La \omterm{def:g8:voltage-voltmeter:voltage}{tensión} y la corriente son magnitudes distintas, y la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} va primero. Una \omterm{def:g3:first-electric-circuit:battery}{pila} sola en un cajón, sin
conectar a nada, no impulsa ninguna corriente --- y sin embargo entre
sus \omterm{def:g3:first-electric-circuit:battery}{bornes} están listos y medibles sus \qty{1.5}{V} enteros, como
la altura de una cascada un día en que el molino está cerrado. El
empujón puede existir sin desfile; el desfile no existe nunca sin
empujón.
\end{proposition}

\begin{definition}[Voltímetro]\label{def:g8:voltage-voltmeter:voltmeter}
Un \emph{voltímetro}\index{voltímetro} mide la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} entre dos puntos --- así que se conecta \emph{entre}
ellos: \emph{\omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}} con el componente que interesa, una
punta a cada lado, como un topógrafo que sostiene un nivel entre dos
alturas. Construido al revés que el \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} en todos los
sentidos, se opone a la corriente casi \emph{del todo}: conectado entre
dos puntos cualesquiera, no absorbe casi nada --- así que pinzarlo no
molesta a nadie y no hace falta romper ningún lazo.
\end{definition}

{\small El \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} adopta la postura del topógrafo: cruzado
sobre la lámpara, una punta a cada lado, midiendo el empujón que hay
entre ellas --- y sin plantarse nunca en la carretera.}
\end{omfigure}

\begin{method}[Medir una tensión]\label{met:g8:voltage-voltmeter:measure}
\begin{enumerate}
  \item deja el circuito intacto --- para un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} no se
        rompe nada;
  \item pinza una punta a cada lado del componente: la punta
        \unit{V} por donde entra la corriente convencional y la
        punta común por donde sale;
  \item alcance mayor primero, y ve bajando hasta la comodidad;
  \item lee y escribe la \omterm{def:g6:measurement-in-science:measuring}{unidad}.
\end{enumerate}
Los modales de los dos aparatos, uno al lado del otro:
\omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} --- romper la carretera, plantarse en ella, no
oponerse a nada; \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} --- no romper nada, quedarse a un
lado, oponerse a todo. Confundir los dos modales es la costumbre más
cara del taller.
\end{method}

\section{Tensiones nominales}

\begin{definition}[Tensión nominal]\label{def:g8:voltage-voltmeter:nominal}
Toda lámpara, todo \omterm{ex:g6:circuits-and-safety:motor}{motor} y todo aparato lleva una \emph{tensión
nominal}\index{tensión nominal} --- el empujón para el que está
construido, estampado en su casquillo o en su placa: una
\omterm{def:g3:first-electric-circuit:bulb}{bombilla} de linterna, ``3.5 V''; un faro de coche, ``12 V''; un
hervidor, ``230 V''. Alimentado con su \omterm{def:g8:voltage-voltmeter:voltage}{tensión} nominal, el
aparato rinde como se diseñó; alimentado con mucho menos, se enfurruña
--- lámpara mortecina, \omterm{ex:g6:circuits-and-safety:motor}{motor} perezoso; alimentado con mucho más, se
quema --- brillantemente, una sola vez.
\end{definition}

\begin{example}[Emparejar lámpara y pila]\label{ex:g8:voltage-voltmeter:matching}
Una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} de ``3.5 V'' en la \omterm{def:g3:first-electric-circuit:battery}{pila} plana de
\qty{4.5}{V}: algo pasada de rosca --- brillante y de vida corta. En
una sola \omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{1.5}{V}: un resplandor naranja soñoliento. En la
\omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{9}{V}: un instante deslumbrante y el hilo fino se
parte --- la esquela del filamento. La etiqueta es un contrato:
respétalo y la lámpara vivirá la vida que anuncia.
\end{example}

\begin{remark}[Por qué matan los 230 V]\label{rem:g8:voltage-voltmeter:mains}
Ahora la vieja regla absoluta recibe su número. La carretera conductora
y húmeda del cuerpo, ofreciéndole \qty{4.5}{V}, deja pasar una
corriente demasiado pequeña para notarla. Esa misma carretera a
\qty{230}{V} --- cincuenta veces el empujón --- deja pasar cincuenta
veces la corriente, muy por encima de las pocas centésimas de
\omterm{def:g8:intensity-ammeter:intensity}{amperio} que paran un corazón. El enchufe no es ``un poco'' más
fuerte que tu banco de \omterm{def:g3:first-electric-circuit:battery}{pilas}: está dos órdenes de magnitud por encima,
y la severidad del reglamento es aritmética pura.
\end{remark}

\section{Primeras lecturas de tensión}

\begin{example}[Alrededor de un lazo sencillo]\label{ex:g8:voltage-voltmeter:readings}
Una \omterm{def:g3:first-electric-circuit:battery}{pila}, una lámpara y los aparatos en la mano. Entre los
\omterm{def:g3:first-electric-circuit:battery}{bornes} de la \omterm{def:g3:first-electric-circuit:battery}{pila}: \qty{4.5}{V}. En la lámpara: \qty{4.5}{V} ---
la lámpara recibe lo que la \omterm{def:g3:first-electric-circuit:battery}{pila} ofrece. En un tramo de
\emph{cable} de conexión: \qty{0}{V}, o lo más cerca de cero que el
aparato sepa decir --- un buen cable es una carretera llana que no
gasta nada del empujón. El empujón vive en los \emph{trabajadores}, no
a lo largo de las carreteras: una lectura que conviene recordar para
cuando las leyes del capítulo siguiente la hagan oficial.
\end{example}

\begin{example}[La confesión del interruptor abierto]\label{ex:g8:voltage-voltmeter:openswitch}
Lámpara apagada, \omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto --- ¿adónde ha ido el
empujón? Mide: en la lámpara apagada, \qty{0}{V}; en el
\emph{\omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto}, los \qty{4.5}{V} enteros. El empujón de
la \omterm{def:g3:first-electric-circuit:battery}{pila} está entero en el hueco, esperando --- el único sitio del
lazo donde dos puntos están de verdad ``lejos'' eléctricamente. Cierra
el \omterm{ex:g3:first-electric-circuit:switch}{interruptor} y las lecturas se intercambian. Un
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto es donde espera la \omterm{def:g8:voltage-voltmeter:voltage}{tensión}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:1}
¿Qué mide la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} y por qué es siempre un ``entre''?
Da su \omterm{def:g6:measurement-in-science:measuring}{unidad} y la \omterm{prop:g5:mirrors-reflection:image}{imagen} de la cascada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:2}
De memoria: las tensiones de una \omterm{def:g3:first-electric-circuit:battery}{pila} redonda, de la
\omterm{def:g3:first-electric-circuit:battery}{pila} plana, de la \omterm{def:g3:first-electric-circuit:battery}{pila} rectangular, de una
batería de coche y de la red.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:3}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} está tirada sin conectar a nada. ¿Qué corriente
circula? ¿Qué lee un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} puesto entre sus \omterm{def:g3:first-electric-circuit:battery}{bornes}? ¿Qué
proposición habla?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:4}
Contrasta los modales de los dos aparatos: conexión, rotura de la
carretera y oposición a la corriente. ¿Qué error construye un
\omterm{def:g7:short-circuits-safety:device}{cortocircuito} con esfera?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:5}
¿Qué es una \omterm{def:g8:voltage-voltmeter:nominal}{tensión nominal}? Predice el comportamiento de una
lámpara de ``6 V'' con \qty{1.5}{V}; con \qty{6}{V}; con
\qty{12}{V}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:6}
En el lazo sencillo, ¿qué lee el \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} en la
\omterm{def:g3:first-electric-circuit:battery}{pila}, en la lámpara y en un tramo de cable? ¿Dónde vive el
empujón?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:7}
\omterm{ex:g3:first-electric-circuit:switch}{Interruptor} abierto: ¿qué hay en la lámpara apagada y qué en el
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto? ¿Qué les pasa a las dos lecturas con el
clic?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:8}
¿Por qué pinzar un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} sobre una lámpara encendida no
molesta casi nada al circuito? ¿Qué es lo que el aparato se niega a
hacer?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:9}
La \omterm{def:g3:first-electric-circuit:battery}{pila} plana de \qty{4.5}{V} esconde tres \omterm{def:g3:first-electric-circuit:battery}{pilas}. Dibuja (o
describe) su disposición y calcula la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} si una de ellas
estuviera puesta del revés por accidente --- el capítulo de la linterna
de hace años, ahora con números. (Los empujones opuestos se restan.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:10}
La lámpara de una acampada es de \qty{12}{V}; la batería de la
furgoneta marca \qty{12.6}{V} recién cargada y \qty{11.2}{V} después de
una larga velada. Describe la velada de la lámpara y por qué no se
quemó nada.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:11}
Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} conectado por error \emph{\omterm{def:g4:series-circuits:series}{en serie}} con la
lámpara (lazo roto y aparato en el hueco): predice el comportamiento de
la lámpara y la lectura del aparato --- acuérdate de a qué se opone un
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro}. (El silencio de la lámpara y el número del aparato
son los dos instructivos.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:voltage-voltmeter:12}
Dos lámparas idénticas \omterm{def:g4:series-circuits:series}{en serie} con \qty{4.5}{V}: un
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} encuentra unos \qty{2.25}{V} en cada una. Las mismas
lámparas \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} con la misma \omterm{def:g3:first-electric-circuit:battery}{pila}: predice la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} de cada lámpara y concilia los dos resultados con las
reglas de brillo que conoces desde hace años (las parejas \omterm{def:g4:series-circuits:series}{en serie}
lucen poco, las parejas \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} lucen a pleno rendimiento). ¿Qué
capítulo por venir convertirá tus predicciones en ley?
\end{exercise}

\section{Problema: la auditoría del cajón de las pilas}

\begin{problem}[{Problema de fin de semana --- la gran auditoría del
cajón de las pilas; un voltímetro, una caja de zapatos llena de pilas y
el estante de los cacharros de la familia}]\label{pb:g8:voltage-voltmeter:1}
Todas las casas tienen ese cajón: \omterm{def:g3:first-electric-circuit:battery}{pilas} sueltas, baterías misteriosas,
cacharros que ``necesitan \omterm{def:g3:first-electric-circuit:battery}{pilas} nuevas''. Día de auditoría ---
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} en mano.

\medskip\noindent\textbf{Parte I --- Clasificar las \omterm{def:g3:first-electric-circuit:battery}{pilas}.}
\begin{enumerate}
  \item Describe la conexión correcta del \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} para
        comprobar una \omterm{def:g3:first-electric-circuit:battery}{pila} suelta de \qty{1.5}{V} --- y por qué no
        hay que montar antes ningún circuito.
  \item Las lecturas de la auditoría, recién sacadas del cajón:
        \qty{1.58}{V}, \qty{1.32}{V}, \qty{0.9}{V}, \qty{0.1}{V}.
        Clasifícalas: nueva, utilizable, cansada, muerta.
  \item Una \omterm{def:g3:first-electric-circuit:battery}{pila} de nueve \omterm{def:g8:voltage-voltmeter:voltage}{voltios} marca \qty{8.9}{V}.
        ¿Con cuánto contribuye aproximadamente cada una de sus seis
        \omterm{def:g3:first-electric-circuit:battery}{pilas} interiores?
  \item ¿Por qué una \omterm{def:g3:first-electric-circuit:battery}{pila} casi muerta sigue enseñando \emph{algo} de
        \omterm{def:g8:voltage-voltmeter:voltage}{tensión} aunque ya no pueda encender nada? (Su
        empujón sobrevive debilucho; ¿qué es lo que ya no tiene
        \omterm{def:g2:pushes-and-pulls:force}{fuerzas} para mover?)
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El estante de los cacharros.}
\begin{enumerate}[resume]
  \item La cámara quiere dos \omterm{def:g3:first-electric-circuit:battery}{pilas} en fila. ¿Qué \omterm{def:g8:voltage-voltmeter:voltage}{tensión}
        espera y qué entregará la pareja de supervivientes de
        \qty{1.32}{V} de la pregunta 2?
  \item La radio vieja lleva cuatro \omterm{def:g3:first-electric-circuit:battery}{pilas} en fila y se enfurruña por
        debajo de \qty{5}{V}. ¿Qué \omterm{def:g3:first-electric-circuit:battery}{pilas} del cajón, si es que hay
        alguna, pueden tripularla?
  \item El hermano pequeño embute tres \omterm{def:g3:first-electric-circuit:battery}{pilas} en la linterna de dos
        ``para que alumbre más'', encajando la tercera contra el
        muelle. La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} es de \qty{3}{V}. Predice el
        espectáculo, en lenguaje de \omterm{def:g8:voltage-voltmeter:nominal}{tensión nominal}.
  \item La placa del transformador del timbre pone ``\qty{8}{V}''.
        Alguien propone conectar el timbre directamente a una
        \omterm{def:g3:first-electric-circuit:battery}{pila} de nueve \omterm{def:g8:voltage-voltmeter:voltage}{voltios} ``ya que ocho es casi
        nueve''. Evalúalo: ¿emparejamiento razonable o asesinato
        lento?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Las leyes de la auditoría.}
\begin{enumerate}[resume]
  \item En la linterna (\omterm{def:g3:first-electric-circuit:bulb}{bombilla} y \omterm{def:g3:first-electric-circuit:battery}{pilas} \omterm{def:g4:series-circuits:series}{en serie},
        \omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto), el auditor mide: en el
        \omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto, \qty{3.1}{V}; en la
        \omterm{def:g3:first-electric-circuit:bulb}{bombilla}, \qty{0}{V}. Explica las dos y predice las dos
        después del clic.
  \item En todos los tramos del cableado de la linterna el aparato
        dice \qty{0}{V}, encendida o apagada. ¿Qué declara esto sobre
        el buen cable, en lenguaje de empujones?
  \item El enchufe de la red \emph{no} se audita, por ley familiar
        permanente. Reformula la ley con el número de este capítulo y
        con la frase aritmética que hay detrás.
  \item Archiva el informe de la auditoría: tres frases --- qué es y
        qué no es nunca un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro}; qué promete y qué amenaza
        la \omterm{def:g8:voltage-voltmeter:nominal}{tensión nominal} de una etiqueta; y dónde espera la
        \omterm{def:g8:voltage-voltmeter:voltage}{tensión} en cualquier circuito parado.
\end{enumerate}
\end{problem}
