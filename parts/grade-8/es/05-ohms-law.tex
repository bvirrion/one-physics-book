\chapter{La resistencia y la ley de Ohm}\label{ch:g8:ohms-law}

Una pregunta ha ido siguiendo a toda la historia eléctrica: algunos
componentes dejan que el desfile corra y otros lo estrangulan hasta
dejarlo en un hilillo --- las lámparas se reparten las
tensiones de forma desigual, los cables finos se calientan y los
gruesos se quedan fríos. La oposición en sí recibe ahora un nombre, una
\omterm{def:g6:measurement-in-science:measuring}{unidad} y --- coronando cuatro años de circuitos --- la ley más
útil de este libro: tres letras que atan el empujón, el desfile y la
oposición en una sola línea de álgebra.

\section{La resistencia}

\begin{definition}[Resistencia]\label{def:g8:ohms-law:resistance}
La \emph{resistencia}\index{resistencia} $R$ de un componente
mide con cuánta \omterm{def:g2:pushes-and-pulls:force}{fuerza} se opone a la \omterm{def:g7:circuits-series-parallel:current}{corriente
eléctrica}: a una \omterm{def:g8:voltage-voltmeter:voltage}{tensión} dada, cuanto mayor es la resistencia,
más débil es el desfile que deja pasar. Su \omterm{def:g6:measurement-in-science:measuring}{unidad} es el
\emph{ohmio}\index{ohmio} (símbolo $\Omega$, la omega mayúscula
griega), en honor del físico y maestro de escuela que encontró la ley
de este capítulo con cables caseros y una paciencia heroica.
\end{definition}

\begin{definition}[Resistor]\label{def:g8:ohms-law:resistor}
Un \emph{resistor}\index{resistor} es un componente fabricado para
tener una \omterm{def:g8:ohms-law:resistance}{resistencia} elegida y estable --- un obstáculo
calibrado, que se vende desde fracciones de \omterm{def:g8:ohms-law:resistance}{ohmio} hasta
millones. Los circuitos usan resistores para fijar corrientes a
propósito: para domar el desfile que quemaría una pieza delicada, para
atenuar, para repartir, para ajustar. Sus bandas de colores deletrean
su valor; su símbolo es un rectángulo liso.
\end{definition}

\begin{example}[Resistencias que te rodean]\label{ex:g8:ohms-law:values}
Un \omterm{def:g2:measuring-length:units}{metro} de cable grueso de cobre: centésimas de
\omterm{def:g8:ohms-law:resistance}{ohmio} --- la carretera llana. Una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} de linterna
encendida: unos cuantos \omterm{def:g8:ohms-law:resistance}{ohmios}. Un \omterm{ex:g6:circuits-and-safety:motor}{motor} eléctrico
pequeño: unos pocos \omterm{def:g8:ohms-law:resistance}{ohmios}. La \omterm{def:g8:ohms-law:resistance}{resistencia} calefactora de un
hervidor: unos $25\,\Omega$. La piel humana seca, de mano a mano:
decenas de miles de \omterm{def:g8:ohms-law:resistance}{ohmios} --- mojada, catastróficamente menos: la
vieja regla de las manos húmedas es una afirmación sobre la
\omterm{def:g8:ohms-law:resistance}{resistencia}. Un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto:
\omterm{def:g8:ohms-law:resistance}{resistencia} más allá de toda medida --- el obstáculo infinito.
\end{example}

\section{El experimento}

\begin{method}[Medir el retrato $U$--$I$ de un componente]\label{met:g8:ohms-law:portrait}
Un \omterm{def:g8:ohms-law:resistor}{resistor}, una fuente regulable (o una cola cada vez más larga de \omterm{def:g3:first-electric-circuit:battery}{pilas})
y los dos aparatos:
\begin{enumerate}
  \item conecta la fuente, el \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} y el
        \omterm{def:g8:ohms-law:resistor}{resistor} \omterm{def:g4:series-circuits:series}{en serie}; pinza el
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} sobre el \omterm{def:g8:ohms-law:resistor}{resistor};
  \item pon la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} más pequeña; anota la pareja ($U$,
        $I$);
  \item sube la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} paso a paso --- una \omterm{def:g3:first-electric-circuit:battery}{pila}, dos, tres
        --- anotando una pareja cada vez;
  \item dibuja los puntos, $I$ en horizontal y $U$ en vertical, y
        mira.
\end{enumerate}
\end{method}

\begin{example}[La confesión de un resistor]\label{ex:g8:ohms-law:confession}
La tabla de una sesión, para un \omterm{def:g8:ohms-law:resistor}{resistor}:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
El doble de empujón, el doble de desfile; el triple, el triple: $U$ e
$I$ son \emph{proporcionales}, y todos los cocientes $U/I$ devuelven el
mismo número: $15$. Ese cociente constante no es casualidad --- es la
\omterm{def:g8:ohms-law:resistance}{resistencia} del \omterm{def:g8:ohms-law:resistor}{resistor}, $R = 15\,\Omega$,
enseñándose en todas las filas.
\end{example}

\begin{proposition}[Ley de Ohm]\label{prop:g8:ohms-law:law}
Para un \omterm{def:g8:ohms-law:resistor}{resistor} a \omterm{def:g3:temperature-thermometers:temperature}{temperatura} estable, la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} entre sus extremos y la corriente que lo recorre son
proporcionales, y la constante de proporcionalidad es su
\omterm{def:g8:ohms-law:resistance}{resistencia}:
\[
U = R \times I ,
\]
con $U$ en \omterm{def:g8:voltage-voltmeter:voltage}{voltios}, $I$ en \omterm{def:g8:intensity-ammeter:intensity}{amperios} y $R$ en
\omterm{def:g8:ohms-law:resistance}{ohmios}. Reordenada al gusto: $I = U/R$ (la corriente que un
empujón mueve a través de un obstáculo) y $R = U/I$ (el obstáculo,
desenmascarado por una pareja honrada de lecturas).
\end{proposition}

\begin{example}[La ley lo calcula todo]\label{ex:g8:ohms-law:compute}
Una vez conocida $R$, la ley responde en todas las direcciones. La
\omterm{def:g8:ohms-law:resistance}{resistencia} del hervidor, $25\,\Omega$ en la red de \qty{230}{V}: $I =
U/R = 230 \div 25 = \qty{9.2}{A}$ --- explicado el aparato de
diez \omterm{def:g8:intensity-ammeter:intensity}{amperios}. Un \omterm{def:g8:ohms-law:resistor}{resistor} tiene que limitar a
\qty{0.02}{A} la corriente de una lámpara delicada a partir de
\qty{4.5}{V}: $R = U/I = 4.5 \div 0.02 = 225\,\Omega$ --- elige el
siguiente valor normalizado. Por $60\,\Omega$ circulan \qty{0.05}{A}:
la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} entre sus extremos es $U = 60 \times 0.05 =
\qty{3}{V}$. Una ley, tres recetas, el taller entero.
\end{example}

\begin{remark}[La muleta del triángulo]\label{rem:g8:ohms-law:triangle}
Algunos estudiantes dibujan a lápiz $U$ encima de $R$ e $I$ en un
triangulito y tapan la letra buscada para leer la receta. Inofensivo
como muleta --- pero tu álgebra ya reordena $U = R I$ con honradez en
una línea, y la manera honrada viaja a todas las fórmulas que te
encuentres en la vida, haya triángulo o no. Apóyate en la muleta este
año si te hace falta; hazte el propósito de andar.
\end{remark}

\section{Lo que explica la resistencia}

\begin{example}[Viejos misterios en una sola factura]\label{ex:g8:ohms-law:mysteries}
El reparto desigual de \omterm{def:g8:voltage-voltmeter:voltage}{tensión} del capítulo de las leyes: las
lámparas \omterm{def:g4:series-circuits:series}{en serie} comparten una sola $I$, así que por $U = R I$ la $R$
mayor se lleva la $U$ mayor --- el salto más grande del trabajador más
duro, ahora calculable. La \omterm{def:g5:series-parallel-circuits:parallel}{rama} estrangulada del capítulo del
paralelo: el cable largo y fino añadió \omterm{def:g8:ohms-law:resistance}{resistencia}, e $I = U/R$
encogió la parte de esa \omterm{def:g5:series-parallel-circuits:parallel}{rama}. El \omterm{def:g7:short-circuits-safety:device}{cortocircuito}: un
desvío de $R$ casi nula, así que $I = U/R$ explota --- la estampida
siempre fue una división por casi cero. Cuatro años de saber
cualitativo sobre circuitos, en la factura de una sola ley.
\end{example}

\begin{example}[Los cables, al metro]\label{ex:g8:ohms-law:wires}
La \omterm{def:g8:ohms-law:resistance}{resistencia} de un cable crece con su longitud y encoge con su
grosor --- un cable largo y fino es una carretera de montaña estrecha;
uno corto y grueso, una autopista. De ahí los cables gordos del
electricista para los aparatos hambrientos, la aleación resistiva
deliberadamente larga y fina de una \omterm{def:g8:ohms-law:resistance}{resistencia} calefactora --- y la
razón de que la misma ley que enciende una lámpara caliente un
tostador: la \omterm{def:g8:ohms-law:resistance}{resistencia} es donde el desfile gasta su empujón, y
el empujón gastado, como confirmará el año que viene el capítulo de la
\omterm{def:g5:everyday-energy:energy}{energía}, se convierte en \omterm{def:g5:heat-and-insulation:heat}{calor}.
\end{example}

\begin{remark}[Dónde acaba la jurisdicción de la ley]\label{rem:g8:ohms-law:limits}
La ley de Ohm es una ley de los \omterm{def:g8:ohms-law:resistor}{resistores}, no un derecho
universal. Mide el retrato de una \emph{lámpara de filamento} y los
puntos se apartan de la recta: el filamento se calienta a medida
que crece la corriente, y el metal caliente resiste más --- $R$ se
niega a quedarse quieta. Los diodos, los \omterm{ex:g6:circuits-and-safety:motor}{motores} y la piel viva
también doblan sus retratos, cada uno a su manera. La recta que pasa
por el origen es la firma del \omterm{def:g8:ohms-law:resistor}{resistor}, no la de la materia ---
pídele siempre su retrato a un componente antes de confiarle la ley.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
¿Qué mide la \omterm{def:g8:ohms-law:resistance}{resistencia}? Da su \omterm{def:g6:measurement-in-science:measuring}{unidad} y su
símbolo, y escribe la ley de Ohm con sus tres reordenaciones.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
Completa cada pareja para un \omterm{def:g8:ohms-law:resistor}{resistor} de $30\,\Omega$: $U =
\qty{6}{V}$, $I = {?}$; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
Un componente enseña $U = \qty{4.5}{V}$ con $I = \qty{0.09}{A}$. ¿Su
\omterm{def:g8:ohms-law:resistance}{resistencia}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
En el experimento del retrato, ¿qué forma dibujan los puntos de un
\omterm{def:g8:ohms-law:resistor}{resistor} de verdad, y qué significa una recta más empinada?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
Con la tabla del capítulo, comprueba $R$ a partir de la segunda fila y
de la cuarta. ¿Por qué obtener el mismo número las dos veces es
justamente lo importante?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
La \omterm{def:g8:ohms-law:resistance}{resistencia} de un hervidor es de $25\,\Omega$; la de un segundo
hervidor, de $35\,\Omega$. Con la misma red, ¿cuál absorbe la corriente
más fuerte --- y cuánto absorbe cada uno (un decimal)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
Explica con $U = R I$ por qué, de dos lámparas \omterm{def:g4:series-circuits:series}{en serie} que comparten
una corriente, la de mayor \omterm{def:g8:ohms-law:resistance}{resistencia} se lleva la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} más grande.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
Traduce al lenguaje de la \omterm{def:g8:ohms-law:resistance}{resistencia}: la estampida del
\omterm{def:g7:short-circuits-safety:device}{cortocircuito}; \ el bloqueo total del \omterm{ex:g3:first-electric-circuit:switch}{interruptor}
abierto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
Una lamparita indicadora delicada aguanta \qty{20}{mA}. ¿Qué
\omterm{def:g8:ohms-law:resistor}{resistor} \omterm{def:g4:series-circuits:series}{en serie} la protege con una \omterm{def:g3:first-electric-circuit:battery}{pila} de
\qty{9}{V}, si la propia lámpara se lleva unos \qty{2}{V} de la parte?
(Primero la parte del \omterm{def:g8:ohms-law:resistor}{resistor}; después, $R$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
El retrato medido de una lámpara de filamento: (\qty{1}{V},
\qty{0.25}{A}), (\qty{4}{V}, \qty{0.5}{A}). Demuestra que ninguna $R$
única encaja con los dos puntos y cuenta la historia física que hay
detrás de la curvatura.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
Dos \omterm{def:g8:ohms-law:resistor}{resistores} \omterm{def:g4:series-circuits:series}{en serie}, de $10\,\Omega$ y
$20\,\Omega$, con \qty{6}{V}. Usando las leyes de la serie más la ley
de Ohm en cada uno: halla la corriente del lazo y la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} de cada \omterm{def:g8:ohms-law:resistor}{resistor}. (Pista: hay una única
corriente $I$ que cumple $6 = 10 I + 20 I$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
Los mismos dos \omterm{def:g8:ohms-law:resistor}{resistores}, ahora \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} con
\qty{6}{V}. Halla la corriente de cada \omterm{def:g5:series-parallel-circuits:parallel}{rama}, el total de la
\omterm{def:g3:first-electric-circuit:battery}{pila} y después --- el final elegante --- la
\omterm{def:g8:ohms-law:resistance}{resistencia} $R$ única que absorbería ese mismo total con
\qty{6}{V}. Compárala con los dos \omterm{def:g8:ohms-law:resistor}{resistores} y explica, en
lenguaje de carreteras, por qué el equipo de dos resiste \emph{menos}
que cualquiera de sus miembros por separado.
\end{exercise}

\section{Problema: el banco de control de calidad}

\begin{problem}[{Problema de fin de semana --- día de inspección en la
fábrica de resistores; retratos, tolerancias y un componente que no es
resistor en absoluto}]\label{pb:g8:ohms-law:1}
El banco de calidad de la fábrica prueba la producción del día --- y a
un impostor. Tú manejas los aparatos.

\medskip\noindent\textbf{Parte I --- Calibrar el banco.}
\begin{enumerate}
  \item Dibuja (o describe) el circuito de prueba: fuente,
        \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro}, el componente que se prueba,
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} --- ¿quién va \omterm{def:g4:series-circuits:series}{en serie} y quién
        cruzado?
  \item La muestra uno, a \qty{4.5}{V}, deja pasar \qty{0.15}{A}.
        ¿Su \omterm{def:g8:ohms-law:resistance}{resistencia}?
  \item La etiqueta declara $33\,\Omega$, con una tolerancia del
        cinco por ciento. Calcula la banda aceptable y dictamina
        sobre la muestra uno. (Primero, el cinco por ciento de
        $33$.)
  \item La muestra dos, etiquetada con $150\,\Omega$: predice su
        corriente a \qty{4.5}{V}, para que el banco sepa qué esperar
        (un decimal, en \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Retratos completos.}
\begin{enumerate}[resume]
  \item Tabla de la muestra tres: $(1.5, 0.05)$, $(3.0, 0.10)$,
        $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{voltios},
        \omterm{def:g8:intensity-ammeter:intensity}{amperios}. Ajusta $R$, citando el rasgo de los números
        que permite un valor único.
  \item Tabla de la muestra cuatro: $(1.5, 0.30)$, $(3.0, 0.45)$,
        $(4.5, 0.54)$, $(6.0, 0.60)$. Destapa al impostor: ninguna
        $R$ única, y el sentido de la deriva.
  \item La muestra cuatro es, de hecho, una lamparita de filamento
        de la línea de producción de al lado. Cuenta la historia de
        su curvatura --- y por qué la prueba de la recta del banco es
        exactamente un detector de \omterm{def:g8:ohms-law:resistor}{resistores}.
  \item El reglamento del banco dice: ``todos los retratos deben
        tomarse a \omterm{def:g3:temperature-thermometers:temperature}{temperatura} estable''. ¿Qué cláusula de la
        ley de Ohm está protegiendo la regla?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Departamento de
aplicaciones.}
\begin{enumerate}[resume]
  \item Un pedido solicita un \omterm{def:g8:ohms-law:resistor}{resistor} que limite la corriente a
        \qty{30}{mA} con una fuente de \qty{12}{V} (la parte de la
        propia carga, despreciable). Calcula el valor que hay que
        presupuestar.
  \item Un cliente se queja de que su \omterm{def:g8:ohms-law:resistance}{resistencia} calefactora de
        $25\,\Omega$ ``solo'' absorbe \qty{9.2}{A} cuando el
        \omterm{def:g7:short-circuits-safety:fuse}{fusible} permite $10$. Calcula qué \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de
        alimentación implica su queja y tranquilízalo con la
        aritmética.
  \item El laboratorio de prototipos pide el puente de cobre más
        grueso y más corto de la fábrica, ``con tan cerca de cero
        \omterm{def:g8:ohms-law:resistance}{ohmios} como sea posible''. ¿Para qué papel en el
        circuito ha nacido ese puente --- y cerca de qué villano
        cotidiano lo coloca su \omterm{def:g8:ohms-law:resistance}{resistencia} casi nula si se usa
        mal?
  \item Cierra el diario de inspección: tres frases --- la ley, su
        retrato y la única condición bajo la cual puede firmarla un
        componente.
\end{enumerate}
\end{problem}
