\chapter{La corriente eléctrica y sus efectos}\label{ch:g8:electric-current-effects}

La corriente lleva cinco años de circuitos desfilando útil y sin
nombre: hilos que se encienden, \omterm{ex:g6:circuits-and-safety:motor}{motores} que giran,
\omterm{ex:g6:circuits-and-safety:motor}{zumbadores} que suenan. Este año la electricidad se vuelve
cuantitativa --- aparatos de medida, \omterm{def:g6:measurement-in-science:measuring}{unidades} y leyes --- y abrimos
haciendo inventario: ¿qué puede \emph{hacer} exactamente una corriente?
Tres efectos, resulta, y con ellos está construido el mundo eléctrico
entero.

\section{Los tres efectos}

\begin{proposition}[Los efectos de la corriente eléctrica]\label{prop:g8:electric-current-effects:three}
Una corriente que atraviesa la materia se delata por tres efectos:
\begin{enumerate}
  \item el \emph{efecto calorífico}\index{efecto calorífico}: toda
        corriente calienta el \omterm{def:g4:conductors-insulators:conductor}{conductor} que atraviesa --- con
        suavidad en los buenos cables, con furia en los delgados o
        sobrecargados, al rojo blanco y a propósito en los
        filamentos y en los tostadores;
  \item el \emph{efecto magnético}\index{efecto magnético}: una
        corriente vuelve magnético lo que la rodea --- un cable
        desvía la aguja de una \omterm{def:g4:magnets-and-compass:compass}{brújula} cercana, y un cable
        enrollado se convierte en un verdadero \omterm{def:g1:magnets:magnet}{imán} con polos,
        conectable a voluntad;
  \item el \emph{efecto químico}\index{efecto químico}: al
        atravesar ciertos \omterm{def:g3:solids-liquids-gases:liquid}{líquidos}, una corriente provoca cambios
        químicos --- parte el agua en dos gases, platea cucharas con
        plata y carga y descarga todas las \omterm{def:g3:first-electric-circuit:battery}{pilas}.
\end{enumerate}
\end{proposition}

\begin{example}[El efecto calorífico en funcionamiento]\label{ex:g8:electric-current-effects:heating}
Deliberado: las barras al rojo del tostador, la resistencia escondida
del hervidor, el filamento al rojo blanco de la lámpara antigua, el
\omterm{def:g7:short-circuits-safety:fuse}{fusible} fabricado para fundirse el primero. No deseado: los
cargadores de móvil tibios, los bajos calientes de los portátiles ---
y la amenaza del \omterm{def:g7:short-circuits-safety:device}{cortocircuito}, que estudiaste como la firma del
villano. Un solo efecto, criado y peligro, dosificado por lo fuerte
que corra la corriente --- una dosis que aprenderemos a
\emph{\omterm{def:g6:measurement-in-science:measuring}{medir}} en el capítulo siguiente.
\end{example}

\section{La corriente fabrica imanes}

\begin{example}[La brújula delata al cable]\label{ex:g8:electric-current-effects:oersted}
Hace dos siglos, durante una clase, un físico notó que la aguja de una
\omterm{def:g4:magnets-and-compass:compass}{brújula} daba un respingo cada vez que él cerraba un circuito en la
mesa de al lado. Aquel respingo --- apenas el temblor de una aguja ---
fue el primer puente que se vio jamás entre la electricidad y el
magnetismo, dos ciencias hasta entonces tan separadas como el tiempo
atmosférico y la música. Pon una \omterm{def:g4:magnets-and-compass:compass}{brújula} junto a un cable y conecta:
la aguja se desvía hacia un lado; invierte la corriente y se desvía
hacia el otro. La corriente crea magnetismo a su alrededor, y con una
dirección.
\end{example}

\begin{definition}[Electroimán]\label{def:g8:electric-current-effects:electromagnet}
Un \emph{electroimán}\index{electroimán} es una bobina de cable
aislado --- a menudo enrollada sobre un núcleo de hierro --- que se
convierte en \omterm{def:g1:magnets:magnet}{imán} mientras circula corriente por ella:
\omterm{def:g4:magnets-and-compass:poles}{polo norte}, \omterm{def:g4:magnets-and-compass:poles}{polo sur}, poder de agarrar hierro y todo lo demás.
A diferencia de cualquier \omterm{def:g1:magnets:magnet}{imán} de piedra, obedece a un
\omterm{ex:g3:first-electric-circuit:switch}{interruptor}: corriente conectada, \omterm{def:g1:magnets:magnet}{imán} en marcha; corriente
cortada, \omterm{def:g1:magnets:magnet}{imán} desaparecido; corriente invertida, polos
intercambiados.
\end{definition}

\begin{example}[El magnetismo con interruptor mueve el mundo]\label{ex:g8:electric-current-effects:uses}
La grúa del desguace agarra un coche con un \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} y lo suelta
abriendo un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} --- inténtalo con un \omterm{def:g1:magnets:magnet}{imán} de piedra.
Las cerraduras eléctricas, el clac de un relé, el martillo del timbre
(un \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} que lo arranca contra la campana, corta su propia
corriente y lo devuelve de golpe: el traqueteo hecho ley), el cabezal
lector de los teléfonos antiguos y --- por encima de todo --- cada
\omterm{ex:g6:circuits-and-safety:motor}{motor} eléctrico: magnetismo fabricado por la corriente empujando
contra imanes, miles de veces por segundo. El \omterm{prop:g8:electric-current-effects:three}{efecto magnético}
es el músculo de la era eléctrica.
\end{example}

\begin{method}[Devanar un electroimán]\label{met:g8:electric-current-effects:wind}
\begin{enumerate}
  \item enrolla uno o dos \omterm{def:g2:measuring-length:units}{metros} de cable fino aislado en vueltas
        apretadas y ordenadas alrededor de un clavo o un perno
        grande de hierro, dejando dos extremos libres;
  \item conecta los extremos a una \omterm{def:g3:first-electric-circuit:battery}{pila} a través de un
        \omterm{ex:g3:first-electric-circuit:switch}{interruptor} (nunca directamente durante mucho rato ---
        la bobina es casi un \omterm{def:g7:short-circuits-safety:device}{cortocircuito} y se calienta: el
        \omterm{prop:g8:electric-current-effects:three}{efecto calorífico} hace de carabina del experimento);
  \item cierra el \omterm{ex:g3:first-electric-circuit:switch}{interruptor} un instante: la cabeza del clavo
        agarra clips; ábrelo: se caen;
  \item comprueba con una \omterm{def:g4:magnets-and-compass:compass}{brújula}: los dos extremos de la bobina se
        comportan como polos N y S --- y se intercambian cuando das
        la vuelta a la \omterm{def:g3:first-electric-circuit:battery}{pila}.
\end{enumerate}
Más vueltas, agarre más fuerte: el magnetismo se dosifica devanando.
\end{method}

\section{El efecto químico}

\begin{example}[Corriente a través del agua]\label{ex:g8:electric-current-effects:electrolysis}
Mete en agua ligeramente conductora (una cucharada de sosa ayuda) dos
minas de lápiz conectadas con cable a una \omterm{def:g3:first-electric-circuit:battery}{pila}, y mira: de las
dos minas sale un chorro de burbujitas --- dos gases distintos,
nacidos del agua misma, con el doble de \omterm{def:g6:mass-and-volume:volume}{volumen} en una mina que en
la otra. La corriente está desmontando el agua en sus dos componentes
invisibles. Esta puerta entre la electricidad y la química --- la
\emph{electrólisis} --- platea cucharas baratas, arranca el aluminio
de su mineral y pertenece sobre todo a tu curso de química; la física
la saluda y sigue su camino.
\end{example}

\begin{example}[Toda pila es el efecto al revés]\label{ex:g8:electric-current-effects:battery}
Dentro de cada \omterm{def:g3:first-electric-circuit:battery}{pila}, la química empuja la corriente:
\omterm{def:g5:everyday-energy:energy}{energía} química almacenada mueve el desfile y se gasta a sí misma
mientras sirve --- el vaciarse que conoces desde que se apagó aquella
linterna. Las \omterm{def:g3:first-electric-circuit:battery}{pilas} recargables cierran el círculo: el cargador
\omterm{def:g2:pushes-and-pulls:force}{fuerza} la corriente \emph{al revés} y el \omterm{prop:g8:electric-current-effects:three}{efecto químico}
reconstruye la reserva. De química a eléctrica y de eléctrica a
química: un solo efecto, las dos direcciones y toda la industria de
las \omterm{def:g3:first-electric-circuit:battery}{pilas}.
\end{example}

\begin{remark}[Los efectos como detectores]\label{rem:g8:electric-current-effects:detect}
Los tres efectos responden a una pregunta que te dejó perplejo cuando
eras aprendiz de comprobador: ¿cómo se \emph{ve} una corriente --- un
desfile invisible dentro de un cable \omterm{def:g7:light-sources-propagation:coats}{opaco}? Por sus obras. Un
cable que se calienta, una \omterm{def:g4:magnets-and-compass:compass}{brújula} que da un respingo, un chorro de
burbujas: cada uno delata el desfile. Todos los instrumentos de \omterm{def:g6:measurement-in-science:measuring}{medir}
corriente que se han construido leen uno de los tres efectos --- y el
que espera en el capítulo siguiente lee el respingo magnético,
refinado hasta convertirlo en una esfera con números.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:1}
Nombra los tres efectos de la \omterm{def:g7:circuits-series-parallel:current}{corriente eléctrica}, cada uno con un
uso a nuestro servicio.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:2}
¿Qué efecto hay en cada escena: un \omterm{def:g7:short-circuits-safety:fuse}{fusible} que se
funde; \ un timbre que martillea; \ burbujas en las minas de
lápiz; \ el rojo de un tostador; \ el agarre de la grúa de un
desguace?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:3}
¿Qué reveló el respingo de la \omterm{def:g4:magnets-and-compass:compass}{brújula} junto a la mesa de la clase?
¿Qué le pasa al respingo cuando se invierte la corriente?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:4}
Da tres poderes que tenga un \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} y que ningún \omterm{def:g1:magnets:magnet}{imán}
de piedra pueda igualar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:5}
En el método del devanado, ¿por qué tiene que estar aislado el cable?
¿Y por qué no debe quedarse la bobina conectada mucho rato sin una
carga que trabaje?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:6}
¿Cómo suelta el coche el operario de la grúa? Contrástalo con lo que
exigiría un \omterm{def:g1:magnets:magnet}{imán} de piedra de la misma \omterm{def:g2:pushes-and-pulls:force}{fuerza}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:7}
¿Qué conversión de \omterm{def:g5:everyday-energy:energy}{energía} ocurre dentro de una \omterm{def:g3:first-electric-circuit:battery}{pila} en uso?
¿Y dentro de una \omterm{def:g3:first-electric-circuit:battery}{pila} recargable puesta a cargar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:electric-current-effects:8}
¿Por qué son preciosos los tres efectos como \emph{detectores}? ¿Qué
efecto refinará hasta convertirlo en un número el instrumento del
capítulo siguiente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:electric-current-effects:9}
El timbre: un \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} arranca el martillo hacia la campana,
pero el martillo al moverse rompe su propio circuito; un muelle lo
devuelve y rehace el contacto --- y así sucesivamente. Recorre un
ciclo completo y explica por qué el diseño \emph{tiene} que traquetear
en lugar de quedarse quieto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:electric-current-effects:10}
Un relé es un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} que cierra un \omterm{def:g8:electric-current-effects:electromagnet}{electroimán}: una
corriente pequeña en la bobina cierra unos contactos que llevan una
corriente grande en otro circuito. Explica por qué esto permite que un
termostato delicado mande con seguridad sobre un calefactor eléctrico
hambriento --- y qué dos circuitos no se tocan jamás.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:electric-current-effects:11}
La aguja de tu \omterm{def:g4:magnets-and-compass:compass}{brújula} da un respingo cada vez que un cable
escondido en la pared lleva corriente. Diseña con este hecho un
buscador de cables de pared: protocolo, qué significa un respingo
fuerte y un límite honrado del instrumento (piensa en qué otras cosas
mueven las agujas de las \omterm{def:g4:magnets-and-compass:compass}{brújulas}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:electric-current-effects:12}
Un \omterm{ex:g6:circuits-and-safety:motor}{motor} eléctrico, a grandes rasgos: una bobina montada sobre un
\omterm{def:g6:sun-earth-moon:axisorbit}{eje} se sitúa entre los polos de un \omterm{def:g1:magnets:magnet}{imán}; la corriente convierte la
bobina en \omterm{def:g1:magnets:magnet}{imán}; el tira y afloja de los polos hace girar el \omterm{def:g6:sun-earth-moon:axisorbit}{eje} ---
y, justo cuando la bobina iba a quedarse alineada, se invierte la
corriente que la recorre, así que tiene que perseguir la alineación
para siempre. Explica, polo a polo, por qué la inversión es
indispensable --- ¿qué haría el \omterm{ex:g6:circuits-and-safety:motor}{motor} sin ella? (Acabas de explicar
por qué los \omterm{ex:g6:circuits-and-safety:motor}{motores} llevan dentro un contacto inversor --- y la
mitad de cómo la era eléctrica hace girar sus ruedas.)
\end{exercise}

\section{Problema: el taller del inventor}

\begin{problem}[{Problema de fin de semana --- una pila, tres efectos,
cinco inventos; una tarde en el taller del
inventor}]\label{pb:g8:electric-current-effects:1}
En la mesa del taller hay \omterm{def:g3:first-electric-circuit:battery}{pilas}, cable, clavos de hierro, una
\omterm{def:g4:magnets-and-compass:compass}{brújula}, dos minas de lápiz, un tarro de agua con sosa --- y un
cuaderno de inventos a medio construir que hay que evaluar.

\medskip\noindent\textbf{Parte I --- El inventario.}
\begin{enumerate}
  \item Dibuja el banco de pruebas de los tres efectos del taller:
        con una \omterm{def:g3:first-electric-circuit:battery}{pila} y un \omterm{ex:g3:first-electric-circuit:switch}{interruptor}, describe un montaje que
        enseñe los tres efectos a la vez (un cable fino que se
        caliente, una \omterm{def:g4:magnets-and-compass:compass}{brújula} junto a un \omterm{def:g4:conductors-insulators:conductor}{conductor}, el tarro dentro
        del lazo). ¿En qué disposición tienen que ir las tres
        demostraciones --- \omterm{def:g4:series-circuits:series}{en serie} o \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} --- para que un
        solo \omterm{ex:g3:first-electric-circuit:switch}{interruptor} las gobierne todas?
  \item El cable fino de la demostración se calienta, pero los
        cables gruesos de alimentación siguen fríos, en el mismo
        lazo. ¿Qué hecho del primer curso sobre el lazo \omterm{def:g4:series-circuits:series}{en serie}
        vuelve esto \emph{más} interesante de lo que parece --- qué
        es igual en los dos cables y qué es distinto?
  \item La \omterm{def:g4:magnets-and-compass:compass}{brújula} da un respingo hacia el este cuando se cierra
        el \omterm{ex:g3:first-electric-circuit:switch}{interruptor}. ¿Qué dos cambios, hechos a la vez,
        dejarían el respingo exactamente como estaba?
  \item De una mina de lápiz salen burbujas al doble de \omterm{def:g7:motion-average-speed:formula}{velocidad}
        que de la otra. ¿Qué sugiere esto sobre los dos componentes
        del agua? (La química confirmará la proporción.)
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Evaluar los inventos.}
\begin{enumerate}[resume]
  \item Invento: ``el timbre silencioso'' --- un
        \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} que tira una sola vez de un martillo de
        fieltro blando y lo mantiene sujeto hasta que se suelta el
        pulsador. ¿Qué pieza del mecanismo del timbre clásico tiene
        que \emph{quitar} el inventor, y qué oirá la visita?
  \item Invento: ``el clasificador magnético'' --- la chatarra pasa
        en una cinta por debajo de un \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} que saca el
        hierro y el acero. ¿Qué pasa en el contenedor de descarga y
        por qué el clasificador deja las latas de aluminio en la
        cinta (responden dos hechos sobre los imanes de años
        anteriores)?
  \item Invento: ``el invertidor de polos'' --- una \omterm{def:g4:magnets-and-compass:compass}{brújula}
        montada sobre una bobina, que se da la vuelta cada vez que
        un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} escondido invierte la \omterm{def:g3:first-electric-circuit:battery}{pila}. ¿Qué
        propiedad del magnetismo fabricado por la corriente se está
        aprovechando?
  \item Invento: ``el electroplateador eterno'' --- un plateado con
        plata alimentado por una \omterm{def:g3:first-electric-circuit:battery}{pila} que ``se recarga sola con el
        propio plateado''. Vétalo citando las dos direcciones de la
        \omterm{def:g3:first-electric-circuit:battery}{pila} del capítulo y una ley más antigua sobre la
        \omterm{def:g5:everyday-energy:energy}{energía}.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La obra maestra de la noche.}
El inventor cierra con una máquina de garra para la feria: una
\omterm{def:g3:levers-and-scales:lever}{palanca}, un pequeño \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} colgado de cuerdas y un cajón de
baratijas de hierro.
\begin{enumerate}[resume]
  \item Escribe el guion de la jugada ganadora en lenguaje de
        efectos: ¿qué pasa al ``agarrar'', durante el balanceo y al
        ``soltar''?
  \item Los feriantes se quejan de que la grúa deja caer los premios
        a media pasada siempre que flaquea la \omterm{def:g3:first-electric-circuit:battery}{pila} vieja. ¿Qué
        propiedad del \omterm{def:g8:electric-current-effects:electromagnet}{electroimán} --- virtud y debilidad en una
        --- queda a la vista?
  \item El inventor duplica las vueltas de la bobina para ayudar.
        Predice el efecto sobre el agarre y nombra la otra dosis (del
        lado de la \omterm{def:g3:first-electric-circuit:battery}{pila}) que el capítulo siguiente nos dejará
        poner en números.
  \item Última página del cuaderno: escribe el resumen del taller
        --- una frase por efecto, cada una nombrando su mejor
        invento servicial de la tarde.
\end{enumerate}
\end{problem}
