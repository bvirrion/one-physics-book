\chapter{La intensidad de la corriente y el amperímetro}\label{ch:g8:intensity-ammeter}

``Corriente fuerte, corriente floja'' --- llevamos años hablando así,
dosificando el brillo y el agarre a ojo. El ojo se jubila hoy. La
\omterm{def:g2:pushes-and-pulls:force}{fuerza} de la corriente recibe un nombre, una \omterm{def:g6:measurement-in-science:measuring}{unidad} y un
instrumento; y lo primero que demuestra el instrumento es una ley que
llevabas mucho tiempo sospechando y que no habías medido ni una vez.

\section{La intensidad y su unidad}

\begin{definition}[Intensidad]\label{def:g8:intensity-ammeter:intensity}
La \emph{intensidad}\index{intensidad} $I$ de una \omterm{def:g7:circuits-series-parallel:current}{corriente
eléctrica} mide lo fuerte que es el desfile: cuánta carga eléctrica
desfila por un punto del circuito cada segundo --- el caudal de la
corriente, igual que el de un río es agua por segundo. Su
\omterm{def:g6:measurement-in-science:measuring}{unidad} es el \emph{amperio}\index{amperio} (\unit{A}), en honor de
un gran pionero de las leyes de la electricidad; los circuitos de todos
los días hablan a menudo en \emph{miliamperios}: $\qty{1}{mA} =
\qty{0.001}{A}$, una milésima.
\end{definition}

\begin{example}[Amperios que conviene conocer de vista]\label{ex:g8:intensity-ammeter:values}
Una lamparita indicadora sorbe unos \qty{0.02}{A} --- \qty{20}{mA};
una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} de linterna, unos \qty{0.3}{A}; la gran lámpara del faro
de un coche, \qty{5}{A}; un hervidor a pleno rugido, \qty{10}{A}; el
\omterm{ex:g6:circuits-and-safety:motor}{motor} de un coche al arrancar engulle \qty{100}{A} de su
batería; un rayo, durante su milisegundo, mete decenas de miles de
\omterm{def:g8:intensity-ammeter:intensity}{amperios} por su canal. Y la escala del peligro da que pensar:
corrientes de unas pocas centésimas de \omterm{def:g8:intensity-ammeter:intensity}{amperio} a través de un
cuerpo humano ya pueden ser mortales --- el respeto empieza muy por
debajo de un \omterm{def:g8:intensity-ammeter:intensity}{amperio}.
\end{example}

\begin{remark}[La persona y la unidad]\label{rem:g8:intensity-ammeter:person}
Escrita con todas sus letras, la \omterm{def:g6:measurement-in-science:measuring}{unidad} va en minúscula --- ``una
corriente de dos \omterm{def:g8:intensity-ammeter:intensity}{amperios}'' ---, mientras que la palabra con
mayúscula nombra a la persona, como con toda \omterm{def:g6:measurement-in-science:measuring}{unidad} que honra a un
científico. El símbolo \unit{A} se queda en mayúscula. Volverás a
encontrarte muy pronto con esta convención, dos veces.
\end{remark}

\section{El amperímetro}

\begin{definition}[Amperímetro]\label{def:g8:intensity-ammeter:ammeter}
Un \emph{amperímetro}\index{amperímetro} mide la
\omterm{def:g8:intensity-ammeter:intensity}{intensidad}. Cuenta el desfile que pasa \emph{por dentro de él}
--- así que hay que intercalarlo \emph{\omterm{def:g4:series-circuits:series}{en serie}}, dentro del propio
lazo que mide, como un peaje en la carretera. Construido para
molestar al desfile lo menos posible, casi no se opone a la corriente
--- que es también la razón de que un amperímetro mal conectado
directamente entre los \omterm{def:g3:first-electric-circuit:battery}{bornes} de una \omterm{def:g3:first-electric-circuit:battery}{pila} sea un
\omterm{def:g7:short-circuits-safety:device}{cortocircuito} con esfera: la manera clásica de estropear uno.
\end{definition}

\begin{method}[Medir una intensidad]\label{met:g8:intensity-ammeter:measure}
\begin{enumerate}
  \item apaga el circuito; \emph{rompe} el lazo por donde haya que
        contar la corriente;
  \item intercala el \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} en el hueco: la corriente entra
        por su \omterm{def:g3:first-electric-circuit:battery}{borne} \unit{A} y sale por su \omterm{def:g3:first-electric-circuit:battery}{borne}
        \emph{común}, de modo que el sentido convencional entre por
        la puerta principal del aparato;
  \item en un aparato de varios alcances, empieza por el
        \emph{mayor} y ve bajando hasta que la lectura quede holgada
        en la esfera --- así proteges el aparato de las sorpresas;
  \item cierra el \omterm{ex:g3:first-electric-circuit:switch}{interruptor}, lee --- y escribe la
        \omterm{def:g6:measurement-in-science:measuring}{unidad}.
\end{enumerate}
La conexión al revés en un aparato de aguja estampa la aguja contra su
tope; los digitales se limitan a enseñar un signo menos, perdonando a
los estudiantes a diario.
\end{method}

\section{La primera ley medida}

\begin{proposition}[Un lazo, una intensidad]\label{prop:g8:intensity-ammeter:sameI}
En un \omterm{def:g4:series-circuits:series}{circuito en serie}, la \omterm{def:g8:intensity-ammeter:intensity}{intensidad} es la misma en todos
los puntos del lazo. Mueve el \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} a donde quieras ---
antes de la lámpara, después de ella, junto a la \omterm{def:g3:first-electric-circuit:battery}{pila}: la lectura
no cambia. Lo que cinco años de \omterm{def:g3:first-electric-circuit:bulb}{bombillas} igual de brillantes
sugerían, el aparato lo enuncia ahora en números: nada se consume por
el camino; una carretera, un desfile, una $I$.
\end{proposition}

\end{omfigure}

\begin{example}[Leer la letra pequeña de la ley]\label{ex:g8:intensity-ammeter:fineprint}
La ley dice \emph{a lo largo de un lazo}, no \emph{hagas lo que hagas
con el lazo}. Añade una segunda lámpara \omterm{def:g4:series-circuits:series}{en serie} y el desfile se
debilita en todas partes a la vez --- quizá los \qty{0.30}{A} bajen a
\qty{0.17}{A} ---, pero se debilita \emph{por igual} en todas partes:
los tres peajes coinciden también en el número nuevo. Un lazo, una
\omterm{def:g8:intensity-ammeter:intensity}{intensidad} --- sea cual sea esa \omterm{def:g8:intensity-ammeter:intensity}{intensidad} en cada
momento.
\end{example}

\begin{example}[Un vistazo al nudo]\label{ex:g8:intensity-ammeter:junction}
Pon los peajes en un par \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}: carretera principal
\qty{0.50}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{rama} uno \qty{0.30}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{rama} dos
\qty{0.20}{A}. La suma sugerente no es casualidad --- los desfiles que
entran en un \omterm{def:g5:series-parallel-circuits:parallel}{nudo} y los que salen de él tienen que cuadrar, ya que
la carga ni se crea ni se destruye en una bifurcación. La ley completa,
con su compañera para la tensión, preside el capítulo de dentro de dos;
tus aparatos ya se la han encontrado.
\end{example}

\begin{remark}[El multímetro]\label{rem:g8:intensity-ammeter:multimeter}
El ladrillo amarillo del taller --- el \emph{multímetro} --- pliega el
\omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} y varios instrumentos más dentro de una sola caja con
un selector giratorio. Selector en \unit{A} o \unit{mA}, puntas en las
tomas marcadas, y es tu \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro}, con reglas de serie y todo.
Dos capítulos de este libro viven ya dentro de él; un tercero se muda
en el capítulo siguiente --- con reglas de conexión, cuidado,
\emph{opuestas}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:1}
¿Qué mide la \omterm{def:g8:intensity-ammeter:intensity}{intensidad}, en lenguaje de ríos? Da su
\omterm{def:g6:measurement-in-science:measuring}{unidad} y su símbolo, y convierte \qty{250}{mA} en
\omterm{def:g8:intensity-ammeter:intensity}{amperios}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:2}
Empareja cada \omterm{def:g8:intensity-ammeter:intensity}{intensidad} con su escena: \qty{0.02}{A}; \
\qty{0.3}{A}; \ \qty{10}{A}; \ \qty{100}{A} --- hervidor, lamparita
indicadora, \omterm{ex:g6:circuits-and-safety:motor}{motor} arrancando, \omterm{def:g3:first-electric-circuit:bulb}{bombilla} de linterna.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:3}
¿Por qué hay que intercalar un \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} \omterm{def:g4:series-circuits:series}{en serie}? ¿A qué
está construido para no oponerse casi nada, y qué accidente hace eso
posible?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:4}
Enumera los cuatro pasos de una medida correcta de
\omterm{def:g8:intensity-ammeter:intensity}{intensidad}. ¿Por qué empezar por el alcance mayor?

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:5}
Enuncia la ley de un lazo, una \omterm{def:g8:intensity-ammeter:intensity}{intensidad}. ¿Qué vieja \omterm{def:g1:five-senses:observation}{observación}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:6}
Un \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} colocado antes de un \omterm{ex:g6:circuits-and-safety:motor}{motor} \omterm{def:g4:series-circuits:series}{en serie} marca
\qty{0.45}{A}. ¿Qué marcará después del \omterm{ex:g6:circuits-and-safety:motor}{motor}? ¿Y junto a la

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:7}
Una segunda lámpara se suma a un lazo \omterm{def:g4:series-circuits:series}{en serie} y los \qty{0.30}{A} del
\omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} pasan a ser \qty{0.17}{A}. ¿Ha fallado la ley?

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:8}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:9}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:intensity-ammeter:12}
Diseña el plan de medida completo de un \omterm{def:g5:series-parallel-circuits:parallel}{circuito en paralelo} de
dos \omterm{def:g5:series-parallel-circuits:parallel}{ramas} (lámpara y \omterm{ex:g6:circuits-and-safety:motor}{motor}): cuántas posiciones de
\omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} hacen falta para conocer \emph{todas} las corrientes
del circuito; el número mínimo de medidas que basta (una puede
deducirse --- ¿con qué balance?); y el orden de operaciones que no deja
nunca un aparato conectado peligrosamente.
\end{exercise}

\section{Problema: el banco de metrología}

\begin{problem}[{Problema de fin de semana --- día de acreditación en
el taller de electrónica; cuatro puestos, un amperímetro en cada uno;
el carné del aprendiz}]\label{pb:g8:intensity-ammeter:1}
Para ganarse los privilegios del taller, cada aprendiz recorre cuatro
puestos de medida. Trae el capítulo puesto.

\medskip\noindent\textbf{Parte I --- Puesto uno: el lazo sencillo.}
\omterm{def:g3:first-electric-circuit:battery}{Pila}, \omterm{ex:g3:first-electric-circuit:switch}{interruptor} y una lámpara.
\begin{enumerate}
  \item Describe, paso a paso, cómo intercalar el
        \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} para \omterm{def:g6:measurement-in-science:measuring}{medir} la corriente de la
        lámpara.
  \item La lectura: \qty{280}{mA}. Exprésala en \omterm{def:g8:intensity-ammeter:intensity}{amperios}.
  \item El examinador te lleva el aparato al otro lado de la
        \omterm{def:g3:first-electric-circuit:battery}{pila}. Predice la lectura, citando la ley.
  \item ``¿Y si tuviera dos \omterm{def:g8:intensity-ammeter:ammeter}{amperímetros} a la vez en este
        lazo?''. ¿Dónde podrían ir y qué enseñarían?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Puesto dos: pareja \omterm{def:g4:series-circuits:series}{en serie}.}
Dos lámparas distintas, A más brillante que B, \omterm{def:g4:series-circuits:series}{en serie}.
\begin{enumerate}[resume]
  \item Un aprendiz predice: ``A, más brillante, lleva más corriente
        que B, que está apagadilla''. Pon a prueba la predicción con
        la ley --- y concilia las corrientes iguales con los brillos
        desiguales (¿qué tiene que ser distinto entre las lámparas?).
  \item El aparato marca \qty{150}{mA} entre A y B. Escribe todas
        las demás corrientes del lazo.
  \item La lámpara B queda cortocircuitada por un cable con
        pinzas. Predice la nueva lectura y el comportamiento de las
        dos lámparas.
  \item El examinador pide una frase sobre por qué el taller prohíbe
        los cables con pinzas entre los \omterm{def:g3:first-electric-circuit:battery}{bornes} de la
        \emph{\omterm{def:g3:first-electric-circuit:battery}{pila}} y en cambio permite la demostración sobre
        B.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Puesto tres: el
\omterm{def:g5:series-parallel-circuits:parallel}{nudo}.}
Una \omterm{def:g5:series-parallel-circuits:parallel}{rama} de lámpara y una \omterm{def:g5:series-parallel-circuits:parallel}{rama} de \omterm{ex:g6:circuits-and-safety:motor}{motor}
\omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}.
\begin{enumerate}[resume]
  \item Carretera principal: \qty{0.52}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{rama} de la lámpara:
        \qty{0.30}{A}. ¿La corriente de la \omterm{def:g5:series-parallel-circuits:parallel}{rama} del
        \omterm{ex:g6:circuits-and-safety:motor}{motor}, por balance?
  \item El \omterm{ex:g6:circuits-and-safety:motor}{motor} se atasca (sigue conectado, pero no gira) y su
        \omterm{def:g5:series-parallel-circuits:parallel}{rama} sube a \qty{0.40}{A}. Suponiendo que la
        \omterm{def:g5:series-parallel-circuits:parallel}{rama} de la lámpara conserva sus \qty{0.30}{A}, ¿qué
        lleva ahora la carretera principal --- y qué guardián del
        taller está escuchando?
  \item Se desenrosca la lámpara. Predice las tres lecturas de los
        peajes.
  \item ¿Por qué se gasta más deprisa la \omterm{def:g3:first-electric-circuit:battery}{pila} con las dos
        \omterm{def:g5:series-parallel-circuits:parallel}{ramas} sanas --- dónde se paga la generosidad?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- El carné.}
\begin{enumerate}[resume]
  \item Escribe el juramento del aprendiz: tres líneas --- dónde va y
        dónde no va nunca un \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro}; qué garantiza un solo
        lazo; qué cuadra un \omterm{def:g5:series-parallel-circuits:parallel}{nudo}.
\end{enumerate}
\end{problem}
