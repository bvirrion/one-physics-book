\chapter{Leyes de la corriente y de la tensión en los circuitos}\label{ch:g8:circuit-laws}

Dos aparatos, dos magnitudes --- y ahora, por fin, la constitución.
Todos los circuitos que construyas en tu vida, de una linterna a la red
de una ciudad, obedecen a cuatro leyes breves: dos sobre la
contabilidad de la corriente y dos sobre la de la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión}. Armado con ellas y con un poco de álgebra, puedes
calcular lecturas que todavía no has medido --- y pillar cualquier
medida que mienta.

\section{Las leyes de la corriente}

\begin{proposition}[Ley de la corriente en un lazo en serie]\label{prop:g8:circuit-laws:seriesI}
A lo largo de un solo lazo, la \omterm{def:g8:intensity-ammeter:intensity}{intensidad} es la misma en todas
partes:
\[
I_{\text{pila}} = I_{\text{lámpara 1}} = I_{\text{lámpara 2}} =
\dots
\]
Una carretera, un desfile --- los peajes del capítulo anterior,
convertidos en ley constitucional.
\end{proposition}

\begin{proposition}[La ley del nudo]\label{prop:g8:circuit-laws:junction}
En cualquier \omterm{def:g5:series-parallel-circuits:parallel}{nudo}, las \omterm{def:g8:intensity-ammeter:intensity}{intensidades} que entran juntas igualan a
las \omterm{def:g8:intensity-ammeter:intensity}{intensidades} que salen juntas: para una carretera principal que se
parte en dos \omterm{def:g5:series-parallel-circuits:parallel}{ramas},
\[
I = I_1 + I_2 .
\]
La carga ni se crea, ni se destruye, ni se almacena en una
bifurcación: todo lo que llega, sale. (Vale para cualquier número de
\omterm{def:g5:series-parallel-circuits:parallel}{ramas} --- suma las llegadas, suma las salidas, y las cuentas
cuadran.)
\end{proposition}

\section{Las leyes de la tensión}

\begin{proposition}[Ley de la tensión en un lazo en serie]\label{prop:g8:circuit-laws:seriesU}
A lo largo de un lazo \omterm{def:g4:series-circuits:series}{en serie}, la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de la
\omterm{def:g3:first-electric-circuit:battery}{pila} se reparte entre los aparatos, y las partes suman
exactamente:
\[
U_{\text{pila}} = U_1 + U_2 + \dots
\]
La \omterm{prop:g5:mirrors-reflection:image}{imagen} de la cascada lleva las cuentas: la \omterm{def:g3:first-electric-circuit:battery}{pila} sube
el desfile por toda su altura, y los aparatos se gastan esa altura
entre ellos, salto a salto --- sin que los buenos cables gasten nada.
\end{proposition}

\begin{proposition}[Ley de la tensión en ramas en paralelo]\label{prop:g8:circuit-laws:parallelU}
Los aparatos \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} abarcan todos los mismos dos
\omterm{def:g5:series-parallel-circuits:parallel}{nudos} --- así que una sola \omterm{def:g8:voltage-voltmeter:voltage}{tensión} los sirve a todos:
\[
U_1 = U_2 = U_{\text{entre los dos nudos}} .
\]
Cada \omterm{def:g5:series-parallel-circuits:parallel}{rama} disfruta del empujón entero que hay entre los
\omterm{def:g5:series-parallel-circuits:parallel}{nudos} --- por eso las lámparas \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} lucen con todo su
brillo nominal, y por eso todos los enchufes de tu casa ofrecen los
mismos \qty{230}{V}.
\end{proposition}

\begin{example}[¿Por qué partes desiguales?]\label{ex:g8:circuit-laws:unequal}
Dos lámparas distintas \omterm{def:g4:series-circuits:series}{en serie} llevan la misma corriente (ley
primera) y en cambio se reparten la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de forma desigual
--- \qty{2.1}{V} frente a \qty{2.4}{V}. La parte mayor va a la lámpara
que más se opone al desfile: el trabajador más duro se lleva el salto
más grande. Qué significa exactamente ``oponerse más'', y cómo
medirlo, es el tema entero del capítulo siguiente --- y las leyes
de este capítulo son su escenario.
\end{example}

\section{Calcular antes de medir}

\begin{method}[Resolver un circuito con las cuatro leyes]\label{met:g8:circuit-laws:solve}
Dado un circuito con algunas lecturas conocidas:
\begin{enumerate}
  \item dibuja el esquema y marca todas las $I$ y $U$ conocidas;
  \item escribe como ecuaciones las leyes aplicables --- sumas de
        \omterm{def:g5:series-parallel-circuits:parallel}{nudo} para las corrientes, sumas de lazo para las
        tensiones, igualdad para la $I$ \omterm{def:g4:series-circuits:series}{en serie} y para la $U$
        \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo};
  \item despeja las incógnitas (las ecuaciones de tu curso de
        álgebra, ganándose el sueldo);
  \item comprueba que todo tiene sentido: ninguna lámpara recibiendo
        más de lo que la \omterm{def:g3:first-electric-circuit:battery}{pila} ofrece, ninguna corriente de
        \omterm{def:g5:series-parallel-circuits:parallel}{rama} mayor que la de la carretera principal.
\end{enumerate}
\end{method}

\begin{example}[Resuelto: el circuito mixto]\label{ex:g8:circuit-laws:worked}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{6}{V} alimenta la lámpara A \omterm{def:g4:series-circuits:series}{en serie}
con una pareja \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} (B y C). Medido: $I_{\text{pila}} =
\qty{0.5}{A}$, $I_B = \qty{0.3}{A}$, $U_A = \qty{2.5}{V}$. Calcula lo
demás. Ley del \omterm{def:g5:series-parallel-circuits:parallel}{nudo}: $I_C = 0.5 - 0.3 = \qty{0.2}{A}$. Ley de la
corriente \omterm{def:g4:series-circuits:series}{en serie}: la lámpara A lleva los \qty{0.5}{A} enteros. Ley de
la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} en el lazo: la pareja recibe $U_{\text{pareja}} = 6 -
2.5 = \qty{3.5}{V}$; ley de la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}: $U_B = U_C =
\qty{3.5}{V}$. Todas las lecturas del circuito, ya conocidas --- cuatro
leyes, una línea de álgebra cada una.
\end{example}

\begin{example}[Las leyes como detectores de mentiras]\label{ex:g8:circuit-laws:liedetector}
Un informe de laboratorio afirma: \omterm{def:g3:first-electric-circuit:battery}{pila} \qty{4.5}{V}; dos
lámparas \omterm{def:g4:series-circuits:series}{en serie} a \qty{2.1}{V} y \qty{2.9}{V}. La ley del lazo suma
las partes: \qty{5.0}{V} gastados de un monedero de \qty{4.5}{V} ---
imposible; alguien leyó mal una esfera. Otro informe: al \omterm{def:g5:series-parallel-circuits:parallel}{nudo}
entran \qty{0.40}{A} y por las \omterm{def:g5:series-parallel-circuits:parallel}{ramas} salen \qty{0.25}{A} y
\qty{0.20}{A} --- la bifurcación ``crea'' \qty{0.05}{A}: rechazado.
Antes de las leyes, las lecturas equivocadas tenían tan buena pinta
como las correctas; ahora la constitución audita todas las páginas.
\end{example}

\begin{remark}[Las pilas, bajo las mismas leyes]\label{rem:g8:circuit-laws:batteries}
La regla de ponerlas en fila es la ley de la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} \omterm{def:g4:series-circuits:series}{en serie}
con su ropa más vieja: las \omterm{def:g3:first-electric-circuit:battery}{pilas} \omterm{def:g4:series-circuits:series}{en serie} suman sus
tensiones ($1.5 + 1.5 + 1.5 = \qty{4.5}{V}$ --- el secreto de la
\omterm{def:g3:first-electric-circuit:battery}{pila} plana), y una \omterm{def:g3:first-electric-circuit:battery}{pila} del revés \emph{resta}, con su empujón
gastado en pelearse con las otras. Las leyes gobiernan por igual a los
que dan y a los que gastan; lo único que cambia es el signo de su
contribución.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:1}
Enuncia las cuatro leyes --- dos de corriente y dos de
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} ---, cada una en una línea.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:2}
A un \omterm{def:g5:series-parallel-circuits:parallel}{nudo} le llegan \qty{0.8}{A} y manda \qty{0.55}{A} por una
\omterm{def:g5:series-parallel-circuits:parallel}{rama}. ¿Qué lleva la otra \omterm{def:g5:series-parallel-circuits:parallel}{rama}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:3}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{9}{V} alimenta tres lámparas \omterm{def:g4:series-circuits:series}{en serie}; dos de
ellas se llevan \qty{2.5}{V} y \qty{3.8}{V}. ¿La parte de la tercera?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:4}
Dos lámparas \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} sobre una \omterm{def:g3:first-electric-circuit:battery}{pila} de
\qty{12}{V}: ¿la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de cada lámpara? ¿Qué hecho cotidiano
sobre los enchufes de una casa es esa misma ley?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:5}
En el circuito mixto resuelto, ¿qué medida más habría sido redundante,
y por qué? (Elige una cualquiera y justifícalo con una ley.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:6}
Dos lámparas distintas \omterm{def:g4:series-circuits:series}{en serie} llevan la misma corriente pero se
reparten la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} $1.8$ frente a \qty{2.7}{V}. ¿Qué lámpara se
opone más al desfile? ¿Qué capítulo por venir mide esa
oposición?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:7}
Auditoría: \omterm{def:g3:first-electric-circuit:battery}{pila} \qty{6}{V}; lámparas \omterm{def:g4:series-circuits:series}{en serie} medidas a
\qty{2.2}{V}, \qty{1.9}{V} y \qty{1.4}{V}. ¿Cuadran las cuentas dentro
de la honradez de un aparato?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:8}
Auditoría: carretera principal \qty{0.9}{A}; tres \omterm{def:g5:series-parallel-circuits:parallel}{ramas} con
\qty{0.4}{A}, \qty{0.3}{A} y \qty{0.3}{A}. ¿Veredicto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:9}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{4.5}{V}, la lámpara D \omterm{def:g4:series-circuits:series}{en serie} con un
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} --- y entonces un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} enseña
\qty{4.5}{V} en el \emph{\omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto} y \qty{0}{V} en D.
Demuestra que las dos lecturas obedecen a la ley de la
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} en el lazo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:10}
Tres \omterm{def:g3:first-electric-circuit:battery}{pilas} idénticas en fila encienden una lámpara; una cuarta,
del revés, se suma a la cola. ¿\omterm{def:g8:voltage-voltmeter:voltage}{Tensión} total antes y después,
por la ley de la serie con signos? ¿Qué hace la lámpara?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:11}
Un circuito de dos \omterm{def:g5:series-parallel-circuits:parallel}{ramas}: la \omterm{def:g5:series-parallel-circuits:parallel}{rama} uno lleva las
lámparas E y F \omterm{def:g4:series-circuits:series}{en serie}; la \omterm{def:g5:series-parallel-circuits:parallel}{rama} dos lleva la lámpara G
sola. \omterm{def:g3:first-electric-circuit:battery}{Pila} de \qty{6}{V}; $U_E = \qty{2.2}{V}$. Calcula $U_F$ y
$U_G$, citando una ley en cada paso.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:circuit-laws:12}
La auditoría completa: una \omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{6}{V}; la lámpara P
\omterm{def:g4:series-circuits:series}{en serie} con un \omterm{def:g5:series-parallel-circuits:parallel}{nudo}; las \omterm{def:g5:series-parallel-circuits:parallel}{ramas} Q y R se
vuelven a juntar y regresan. Medido: $I_P = \qty{0.6}{A}$, $I_Q =
\qty{0.45}{A}$, $U_Q = \qty{2.3}{V}$. Calcula todas las corrientes y
tensiones que quedan en el circuito ($I_R$, $U_P$, $U_R$),
enunciando la ley que hay detrás de cada línea --- y después inventa
una ``medida'' extra que podría declarar un estudiante descuidado y que
tu solución terminada condenaría al instante.
\end{exercise}

\section{Problema: las cajas misteriosas selladas}

\begin{problem}[{Problema de fin de semana --- el desafío de las cajas
selladas del club de electrónica; cuatro leyes contra cuatro misterios;
la gran auditoría}]\label{pb:g8:circuit-laws:1}
El club sella circuitos sencillos dentro de cajas y deja fuera solo las
tomas para los aparatos. Los equipos tienen que deducir el interior ---
con la constitución en la mano. La \omterm{def:g3:first-electric-circuit:battery}{pila} del banco es de
\qty{6}{V} en todo el problema.

\medskip\noindent\textbf{Parte I --- Caja A: dos \omterm{def:g3:first-electric-circuit:battery}{bornes}, dos
lámparas.}
La etiqueta admite: dos lámparas idénticas y nada más, o
\omterm{def:g4:series-circuits:series}{en serie} o \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}.
\begin{enumerate}
  \item Conectada a la \omterm{def:g3:first-electric-circuit:battery}{pila}, la caja A absorbe \qty{0.2}{A}; se
        sabe que una sola de esas lámparas a \qty{6}{V} absorbe unos
        \qty{0.4}{A}. ¿\omterm{def:g4:series-circuits:series}{En serie} o \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}? Razónalo con una ley
        de la corriente.
  \item Predice cuánto absorbería la caja en la \emph{otra}
        disposición.
  \item ¿Qué \omterm{def:g8:voltage-voltmeter:voltage}{tensión} recibe cada lámpara interior en la
        disposición real, y por qué ley de la \omterm{def:g8:voltage-voltmeter:voltage}{tensión}?
  \item ¿En qué disposición lucirían las lámparas escondidas con
        todo su brillo nominal --- y por qué se ve por la mirilla de
        la caja que no lo hacen?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Caja B: el rompecabezas de las
tres lámparas.}
La caja B admite: una lámpara (X) \omterm{def:g4:series-circuits:series}{en serie} con una pareja \omterm{def:g5:series-parallel-circuits:parallel}{en
paralelo} (Y, Z --- no necesariamente idénticas).
\begin{enumerate}[resume]
  \item El equipo mide $I_{\text{pila}} = \qty{0.5}{A}$ y, en
        la toma de prueba propia de Y, $I_Y = \qty{0.35}{A}$. Halla
        $I_Z$ e $I_X$, citando las leyes.
  \item Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} en X: \qty{2.8}{V}. Halla la
        \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de la pareja \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} --- y cada una de $U_Y$
        y $U_Z$.
  \item Y y Z llevan corrientes distintas con la misma
        \omterm{def:g8:voltage-voltmeter:voltage}{tensión}. ¿Qué revela esto sobre las dos lámparas, en el
        lenguaje del \cref{ex:g8:circuit-laws:unequal}?
  \item El equipo desenrosca Y por una trampilla. Predice en qué
        sentido cambia la nueva $I_{\text{pila}}$ (sube, baja o se
        mantiene), razonando con las carreteras y su oposición.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Caja C: el tramposo.}
\begin{enumerate}[resume]
  \item El constructor de la caja C declara: ``\omterm{def:g3:first-electric-circuit:battery}{pila}
        \qty{6}{V}; dos lámparas \omterm{def:g4:series-circuits:series}{en serie} dentro, a \qty{2.5}{V} y
        \qty{4.1}{V}; un \omterm{def:g5:series-parallel-circuits:parallel}{nudo} interior que recibe \qty{0.3}{A}
        y manda \qty{0.2}{A} más \qty{0.15}{A}''. Condena el informe
        dos veces, una ley por condena.
  \item Después se descubre que una de las dos lecturas de
        \omterm{def:g8:voltage-voltmeter:voltage}{tensión} declaradas es honrada. Si es la de
        \qty{2.5}{V}, ¿cuánto tiene que valer de verdad su
        compañera?
  \item El constructor confiesa que los números del \omterm{def:g5:series-parallel-circuits:parallel}{nudo}
        estaban ``redondeados con generosidad''. Si las dos lecturas
        de \omterm{def:g5:series-parallel-circuits:parallel}{rama} \qty{0.2}{A} y \qty{0.15}{A} son honradas,
        ¿cuánto tiene que llevar de verdad la carretera principal?
  \item ¿Por qué las cuatro leyes convierten hacer trampas con las
        cajas selladas en un juego perdido? Una frase.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- La constitución.}
\begin{enumerate}[resume]
  \item Escribe la carta fundacional del club: las cuatro leyes en
        cuatro líneas, cada una con la \omterm{prop:g5:mirrors-reflection:image}{imagen} cotidiana que la
        transporta (carreteras, peajes, cascadas, bifurcaciones).
\end{enumerate}
\end{problem}
