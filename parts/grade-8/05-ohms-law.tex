\chapter{Resistance and Ohm's Law}\label{ch:g8:ohms-law}

One question has trailed the whole electrical story: some
components let the march rush, others throttle it to a trickle
--- lamps split \omterm{def:g8:voltage-voltmeter:voltage}{voltages} unequally, thin wires warm, thick ones
stay cool. The opposition itself now gets a name, a \omterm{def:g6:measurement-in-science:measuring}{unit}, and
--- crowning four years of circuits --- the single most useful
law in this book: three letters that tie the push, the march
and the opposition into one line of algebra.

\section{Resistance}

\begin{definition}[Resistance]\label{def:g8:ohms-law:resistance}
The \emph{resistance}\index{resistance} $R$ of a component
\omterm{def:g6:measurement-in-science:measuring}{measures} how strongly it opposes the \omterm{def:g7:circuits-series-parallel:current}{electric current}: at a
given \omterm{def:g8:voltage-voltmeter:voltage}{voltage}, the greater the resistance, the weaker the
march it lets through. Its \omterm{def:g6:measurement-in-science:measuring}{unit} is the \emph{ohm}\index{ohm}
(symbol $\Omega$, the Greek capital omega), honoring the
schoolmaster-physicist who found this chapter's law with
homemade wires and heroic patience.
\end{definition}

\begin{definition}[Resistor]\label{def:g8:ohms-law:resistor}
A \emph{resistor}\index{resistor} is a component manufactured
to have a chosen, steady \omterm{def:g8:ohms-law:resistance}{resistance} --- a calibrated obstacle,
sold from fractions of an \omterm{def:g8:ohms-law:resistance}{ohm} to millions. Circuits use
resistors to set currents on purpose: taming the march that
would burn a delicate part, dimming, dividing, adjusting.
Their striped color bands spell out their value; their symbol
is a plain rectangle.
\end{definition}

\begin{example}[Resistances around you]\label{ex:g8:ohms-law:values}
A \omterm{def:g2:measuring-length:units}{metre} of thick copper wire: hundredths of an \omterm{def:g8:ohms-law:resistance}{ohm} --- the
level road. A glowing flashlight \omterm{def:g3:first-electric-circuit:bulb}{bulb}: some \omterm{def:g8:ohms-law:resistance}{ohms}. A small
electric \omterm{ex:g6:circuits-and-safety:motor}{motor}: a few \omterm{def:g8:ohms-law:resistance}{ohms}. The kettle's heating coil: about
$25\,\Omega$. Dry human skin, hand to hand: tens of thousands
of \omterm{def:g8:ohms-law:resistance}{ohms} --- wet, catastrophically less: the old damp-hands rule
is a \omterm{def:g8:ohms-law:resistance}{resistance} statement. An open \omterm{ex:g3:first-electric-circuit:switch}{switch}: \omterm{def:g8:ohms-law:resistance}{resistance} beyond
all measuring --- the infinite obstacle.
\end{example}

\section{The experiment}

\begin{method}[Measuring a component's $U$--$I$ portrait]\label{met:g8:ohms-law:portrait}
One \omterm{def:g8:ohms-law:resistor}{resistor}, one adjustable supply (or a growing queue of
cells), both meters:
\begin{enumerate}
  \item wire supply, \omterm{def:g8:intensity-ammeter:ammeter}{ammeter} and \omterm{def:g8:ohms-law:resistor}{resistor} \omterm{def:g4:series-circuits:series}{in series}; clip the
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} across the \omterm{def:g8:ohms-law:resistor}{resistor};
  \item set the smallest \omterm{def:g8:voltage-voltmeter:voltage}{voltage}; record the pair ($U$, $I$);
  \item raise the \omterm{def:g8:voltage-voltmeter:voltage}{voltage} step by step --- one cell, two,
        three --- recording a pair each time;
  \item plot the points, $I$ across, $U$ up, and look.
\end{enumerate}
\end{method}

\begin{example}[A resistor's confession]\label{ex:g8:ohms-law:confession}
A session's table, for one \omterm{def:g8:ohms-law:resistor}{resistor}:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
Double the push, double the march; triple, triple: $U$ and $I$
are \emph{proportional}, and every ratio $U/I$ returns the same
number: $15$. That constant ratio is no accident --- it is the
\omterm{def:g8:ohms-law:resistor}{resistor}'s \omterm{def:g8:ohms-law:resistance}{resistance}, $R = 15\,\Omega$, showing itself in
every row.
\end{example}

\begin{proposition}[Ohm's law]\label{prop:g8:ohms-law:law}
For a \omterm{def:g8:ohms-law:resistor}{resistor} at steady \omterm{def:g3:temperature-thermometers:temperature}{temperature}, the \omterm{def:g8:voltage-voltmeter:voltage}{voltage} across it
and the current through it are proportional, and the constant
of proportionality is its \omterm{def:g8:ohms-law:resistance}{resistance}:
\[
U = R \times I ,
\]
with $U$ in \omterm{def:g8:voltage-voltmeter:voltage}{volts}, $I$ in \omterm{def:g8:intensity-ammeter:intensity}{amperes}, $R$ in \omterm{def:g8:ohms-law:resistance}{ohms}. Rearranged to
taste: $I = U/R$ (the current a push drives through an
obstacle) and $R = U/I$ (the obstacle, unmasked by one honest
pair of readings).
\end{proposition}

\begin{example}[The law computes everything]\label{ex:g8:ohms-law:compute}
Once $R$ is known, the law answers in every direction.
The kettle coil, $25\,\Omega$ on the \qty{230}{V} mains:
$I = U/R = 230 \div 25 = \qty{9.2}{A}$ --- the \omterm{def:g8:intensity-ammeter:intensity}{ten-ampere}
appliance explained. A \omterm{def:g8:ohms-law:resistor}{resistor} must limit a delicate lamp's
current to \qty{0.02}{A} from \qty{4.5}{V}: $R = U/I = 4.5
\div 0.02 = 225\,\Omega$ --- pick the next standard value.
Through $60\,\Omega$ flows \qty{0.05}{A}: the \omterm{def:g8:voltage-voltmeter:voltage}{voltage} across
it is $U = 60 \times 0.05 = \qty{3}{V}$. One law, three
recipes, the whole workshop.
\end{example}

\begin{remark}[The triangle crutch]\label{rem:g8:ohms-law:triangle}
Some learners pencil $U$ over $R$ and $I$ in a little
triangle, covering the wanted letter to read off the recipe.
Harmless as a crutch --- but your algebra now rearranges
$U = R I$ honestly in one line, and the honest way travels to
every formula you will ever meet, triangle or none. Lean on
the crutch if you must this year; plan to walk.
\end{remark}

\section{What resistance explains}

\begin{example}[Old mysteries on one invoice]\label{ex:g8:ohms-law:mysteries}
The unequal \omterm{def:g8:voltage-voltmeter:voltage}{voltage} split of the laws chapter: series lamps
share one $I$, so by $U = R I$ the bigger $R$ takes the bigger
$U$ --- the harder worker's larger drop, now computable. The
throttled \omterm{def:g5:series-parallel-circuits:parallel}{branch} of the parallel chapter: the long thin wire
added \omterm{def:g8:ohms-law:resistance}{resistance}, and $I = U/R$ shrank that \omterm{def:g5:series-parallel-circuits:parallel}{branch}'s share.
The \omterm{def:g7:short-circuits-safety:device}{short circuit}: a bypass of nearly zero $R$, so $I = U/R$
explodes --- the stampede was always division by almost-zero.
Four years of qualitative circuit lore, one law's invoice.
\end{example}

\begin{example}[Wires, by the metre]\label{ex:g8:ohms-law:wires}
A wire's \omterm{def:g8:ohms-law:resistance}{resistance} grows with its length and shrinks with its
thickness --- a long thin wire is a narrow mountain road, a
short thick one a motorway. Hence the electrician's fat cables
for hungry appliances, the heating coil's deliberately long
thin resistive alloy --- and the reason the same law that \omterm{def:g1:light-and-shadows:source}{lights}
a lamp warms a toaster: \omterm{def:g8:ohms-law:resistance}{resistance} is where the march spends
its push, and spent push, as the \omterm{def:g5:everyday-energy:energy}{energy} chapter will confirm
next year, becomes warmth.
\end{example}

\begin{remark}[Where the law's writ ends]\label{rem:g8:ohms-law:limits}
Ohm's law is a \omterm{def:g8:ohms-law:resistor}{resistor}'s law, not a universal right. Measure
a \emph{filament lamp's} portrait and the points bend away
from the straight line: the filament \omterm{def:g5:heat-and-insulation:heat}{heats} as the current
grows, and hot metal resists more --- $R$ refuses to stay
put. Diodes, \omterm{ex:g6:circuits-and-safety:motor}{motors} and living skin bend their portraits too,
each in its own way. The straight line through the origin is
the \omterm{def:g8:ohms-law:resistor}{resistor}'s signature, not matter's --- always ask a
component for its portrait before trusting it with the law.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
What does \omterm{def:g8:ohms-law:resistance}{resistance} \omterm{def:g6:measurement-in-science:measuring}{measure}? Give its \omterm{def:g6:measurement-in-science:measuring}{unit} and symbol, and
write Ohm's law with its three rearrangements.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
Complete each pair for a $30\,\Omega$ \omterm{def:g8:ohms-law:resistor}{resistor}: $U =
\qty{6}{V}$, $I = {?}$; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
A component shows $U = \qty{4.5}{V}$ at $I = \qty{0.09}{A}$.
Its \omterm{def:g8:ohms-law:resistance}{resistance}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
In the portrait experiment, what shape do a true \omterm{def:g8:ohms-law:resistor}{resistor}'s
points draw, and what does a steeper line mean?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
From the chapter's table, verify $R$ from the second and
fourth rows. Why is getting the same number both times the
whole point?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
The kettle coil is $25\,\Omega$; a second kettle's is
$35\,\Omega$. On the same mains, which draws the stronger
current --- and how much does each draw (one decimal)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
Explain with $U = R I$ why, of two series lamps sharing one
current, the \omterm{def:g8:ohms-law:resistance}{higher-resistance} lamp takes the larger \omterm{def:g8:voltage-voltmeter:voltage}{voltage}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
Translate into resistance-language: the \omterm{def:g7:short-circuits-safety:device}{short circuit}'s
stampede; \ the open \omterm{ex:g3:first-electric-circuit:switch}{switch}'s total blockade.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
A delicate indicator lamp tolerates \qty{20}{mA}. What series
\omterm{def:g8:ohms-law:resistor}{resistor} protects it on a \qty{9}{V} \omterm{def:g3:first-electric-circuit:battery}{battery}, if the lamp
itself takes about \qty{2}{V} of the share? (\omterm{def:g8:ohms-law:resistor}{Resistor}'s share
first, then $R$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
A filament lamp's portrait, measured: (\qty{1}{V},
\qty{0.25}{A}), (\qty{4}{V}, \qty{0.5}{A}). Show that no
single $R$ fits both points, and tell the physical story
behind the bending.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
Two \omterm{def:g8:ohms-law:resistor}{resistors} \omterm{def:g4:series-circuits:series}{in series}, $10\,\Omega$ and $20\,\Omega$, on
\qty{6}{V}. Using the series laws plus Ohm's law on each:
find the loop's current and each \omterm{def:g8:ohms-law:resistor}{resistor}'s \omterm{def:g8:voltage-voltmeter:voltage}{voltage}. (Hint:
one current $I$ satisfies $6 = 10 I + 20 I$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
Same two \omterm{def:g8:ohms-law:resistor}{resistors}, now \omterm{def:g5:series-parallel-circuits:parallel}{in parallel} on \qty{6}{V}. Find each
\omterm{def:g5:series-parallel-circuits:parallel}{branch}'s current, the \omterm{def:g7:short-circuits-safety:battery}{battery's} total, and then --- the
elegant finish --- the single \omterm{def:g8:ohms-law:resistance}{resistance} $R$ that would draw
that same total from \qty{6}{V}. Compare it with both
\omterm{def:g8:ohms-law:resistor}{resistors} and explain, in road-language, why the team of two
resists \emph{less} than either member alone.
\end{exercise}

\section{Problem: The Quality Control Bench}

\begin{problem}[{Weekend problem --- inspection day at the
resistor factory; portraits, tolerances and one component
that is no resistor at all}]\label{pb:g8:ohms-law:1}
The factory's quality bench tests the day's production ---
and one impostor. You run the meters.

\medskip\noindent\textbf{Part I --- Calibrating the bench.}
\begin{enumerate}
  \item Sketch (or describe) the test circuit: supply,
        \omterm{def:g8:intensity-ammeter:ammeter}{ammeter}, the component under test, \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} --- who
        is \omterm{def:g4:series-circuits:series}{in series}, who across?
  \item Sample one, at \qty{4.5}{V}, passes \qty{0.15}{A}.
        Its \omterm{def:g8:ohms-law:resistance}{resistance}?
  \item The label claims $33\,\Omega$, tolerance five
        percent. Compute the acceptable band, and rule on
        sample one. (Five percent of $33$ first.)
  \item Sample two, labeled $150\,\Omega$: predict its current
        at \qty{4.5}{V}, so the bench knows what to expect
        (one decimal, in \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{Part II --- Full portraits.}
\begin{enumerate}[resume]
  \item Sample three's table: $(1.5, 0.05)$, $(3.0, 0.10)$,
        $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{volts}, \omterm{def:g8:intensity-ammeter:intensity}{amperes}. Fit
        $R$, citing the feature of the numbers that permits a
        single value.
  \item Sample four's table: $(1.5, 0.30)$, $(3.0, 0.45)$,
        $(4.5, 0.54)$, $(6.0, 0.60)$. Show the impostor: no
        single $R$, and the drift's direction.
  \item Sample four is, in fact, a small filament lamp from
        the next production line. Tell its bending story ---
        and why the bench's straight-line test is precisely a
        resistor-detector.
  \item The bench's rule book says: ``every portrait must be
        taken at steady \omterm{def:g3:temperature-thermometers:temperature}{temperature}.'' Which clause of Ohm's
        law is the rule protecting?
\end{enumerate}

\medskip\noindent\textbf{Part III --- Applications
department.}
\begin{enumerate}[resume]
  \item An order asks for a \omterm{def:g8:ohms-law:resistor}{resistor} limiting current to
        \qty{30}{mA} on a \qty{12}{V} supply (the load's own
        share negligible). Compute the value to quote.
  \item A customer complains their $25\,\Omega$ heater coil
        ``only'' draws \qty{9.2}{A} while the \omterm{def:g7:short-circuits-safety:fuse}{fuse} allows
        $10$. Compute what supply \omterm{def:g8:voltage-voltmeter:voltage}{voltage} their complaint
        implies, and reassure them with the arithmetic.
  \item The prototype lab requests the factory's thickest,
        shortest copper jumper ``with as close to zero \omterm{def:g8:ohms-law:resistance}{ohms}
        as possible''. What circuit role is that jumper born
        for --- and near what everyday villain does its
        near-zero \omterm{def:g8:ohms-law:resistance}{resistance} place it if misused?
  \item Close the inspection log: three sentences --- the
        law, its portrait, and the one condition under which
        a component may sign it.
\end{enumerate}
\end{problem}
