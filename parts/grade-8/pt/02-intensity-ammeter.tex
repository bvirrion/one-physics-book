\chapter{A intensidade da corrente e o amperímetro}\label{ch:g8:intensity-ammeter}

``Corrente forte, corrente fraca'' --- falamos assim há anos, dosando
brilho e agarrada no olhômetro. O olhômetro se aposenta hoje. A \omterm{def:g2:pushes-and-pulls:force}{força}
da corrente ganha um nome, uma \omterm{def:g6:measurement-in-science:measuring}{unidade} e um instrumento; e a primeira
coisa que o instrumento prova é uma lei que você suspeita há muito
tempo e nunca mediu.

\section{A intensidade e a unidade dela}

\begin{definition}[Intensidade]\label{def:g8:intensity-ammeter:intensity}
A \emph{intensidade}\index{intensidade} $I$ de uma \omterm{def:g7:circuits-series-parallel:current}{corrente elétrica}
mede quão forte é a marcha: quanta carga elétrica desfila por um
ponto do circuito a cada segundo --- a vazão da corrente, como a de
um rio é água por segundo. A \omterm{def:g6:measurement-in-science:measuring}{unidade} dela é o
\emph{ampère}\index{ampère} (\unit{A}), em homenagem a um grande
pioneiro das leis da eletricidade; os circuitos do dia a dia falam
muitas vezes em \emph{miliampères}: $\qty{1}{mA} = \qty{0.001}{A}$,
um milésimo.
\end{definition}

\begin{example}[Ampères para conhecer de vista]\label{ex:g8:intensity-ammeter:values}
Uma lampadinha indicadora bebe uns \qty{0.02}{A} --- \qty{20}{mA};
uma lâmpada de lanterna, cerca de \qty{0.3}{A}; a grande lâmpada do
farol de um carro, \qty{5}{A}; uma chaleira a todo vapor,
\qty{10}{A}; um \omterm{ex:g6:circuits-and-safety:motor}{motor} de carro dando a partida engole \qty{100}{A} da
bateria; um raio, no milésimo de segundo dele, enfia dezenas de
milhares de \omterm{def:g8:intensity-ammeter:intensity}{ampères} pelo canal. E a escala do perigo é séria:
correntes de alguns centésimos de \omterm{def:g8:intensity-ammeter:intensity}{ampère} atravessando um corpo humano
já podem ser fatais --- o respeito começa bem abaixo de um \omterm{def:g8:intensity-ammeter:intensity}{ampère}.
\end{example}

\begin{remark}[A pessoa e a unidade]\label{rem:g8:intensity-ammeter:person}
Escrita por extenso, a \omterm{def:g6:measurement-in-science:measuring}{unidade} vai em minúscula --- ``uma corrente de
dois \omterm{def:g8:intensity-ammeter:intensity}{ampères}'' --- enquanto a palavra com maiúscula nomeia a pessoa,
como em toda \omterm{def:g6:measurement-in-science:measuring}{unidade} que homenageia um cientista. O símbolo \unit{A}
continua maiúsculo. Você vai reencontrar essa convenção em breve,
duas vezes.
\end{remark}

\section{O amperímetro}

\begin{definition}[Amperímetro]\label{def:g8:intensity-ammeter:ammeter}
Um \emph{amperímetro}\index{amperímetro} mede a \omterm{def:g8:intensity-ammeter:intensity}{intensidade}. Ele
conta a marcha que passa \emph{por dentro dele} --- então precisa ser
inserido \emph{\omterm{def:g4:series-circuits:series}{em série}}, dentro da própria volta que ele mede, como
um pedágio na estrada. Construído para atrapalhar a marcha o mínimo
possível, ele quase não se opõe à corrente --- e é também por isso
que um amperímetro ligado por engano direto nos \omterm{def:g3:first-electric-circuit:battery}{terminais} de uma
\omterm{def:g3:first-electric-circuit:battery}{pilha} é um \omterm{def:g7:short-circuits-safety:device}{curto-circuito} com mostrador: a maneira clássica de
quebrar um.
\end{definition}

\begin{method}[Medir uma intensidade]\label{met:g8:intensity-ammeter:measure}
\begin{enumerate}
  \item desligue o circuito; \emph{abra} a volta no lugar onde a
        corrente deve ser contada;
  \item encaixe o \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} na abertura: corrente entrando pelo
        \omterm{def:g3:first-electric-circuit:battery}{terminal} \unit{A} e saindo pelo \omterm{def:g3:first-electric-circuit:battery}{terminal} \emph{comum}, de modo
        que o sentido convencional entre pela porta da frente do
        aparelho;
  \item num aparelho de várias escalas, comece pela \emph{maior}
        escala e vá descendo até a leitura ficar confortável no
        mostrador --- protegendo o aparelho de surpresas;
  \item feche o \omterm{ex:g3:first-electric-circuit:switch}{interruptor}, leia --- e escreva a \omterm{def:g6:measurement-in-science:measuring}{unidade}.
\end{enumerate}
A ligação ao contrário num aparelho de ponteiro joga o ponteiro
contra o batente; os digitais apenas mostram um sinal de menos,
perdoando estudantes diariamente.
\end{method}

\section{A primeira lei medida}

\begin{proposition}[Uma volta, uma intensidade]\label{prop:g8:intensity-ammeter:sameI}
Num \omterm{def:g4:series-circuits:series}{circuito em série}, a \omterm{def:g8:intensity-ammeter:intensity}{intensidade} é a mesma em todos os pontos da
volta. Mude o \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} de lugar --- antes da lâmpada, depois dela,
ao lado da \omterm{def:g3:first-electric-circuit:battery}{pilha}: a leitura não muda. O que cinco anos de lâmpadas
igualmente brilhantes sugeriam, o aparelho agora enuncia em números:
nada é consumido pelo caminho; uma estrada, uma marcha, um $I$.
\end{proposition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.85]
  \draw (0,0) to[battery1] (0,3.0)
    to[ammeter, l=\qty{0.30}{A}] (2.6,3.0)
    to[lamp] (4.8,3.0)
    to[ammeter, l=\qty{0.30}{A}] (7.4,3.0)
    to[lamp] (9.6,3.0)
    to[short] (9.6,0)
    to[ammeter, l=\qty{0.30}{A}] (4.8,0)
    to[short] (0,0);
\end{tikzpicture}

{\small Três pedágios numa volta só: antes, entre e depois das
lâmpadas, todo \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} informa os mesmos \qty{0.30}{A}.}
\end{omfigure}

\begin{example}[Lendo as letras miúdas da lei]\label{ex:g8:intensity-ammeter:fineprint}
A lei diz \emph{ao longo de uma volta}, e não \emph{aconteça o que
acontecer com a volta}. Acrescente uma segunda lâmpada \omterm{def:g4:series-circuits:series}{em série} e a
marcha enfraquece em todo lugar de uma vez --- talvez os
\qty{0.30}{A} caiam para \qty{0.17}{A} ---, mas enfraquece
\emph{igualmente} em todo lugar: os três pedágios concordam também
com o número novo. Uma volta, uma \omterm{def:g8:intensity-ammeter:intensity}{intensidade} --- seja qual for essa
\omterm{def:g8:intensity-ammeter:intensity}{intensidade} no momento.
\end{example}

\begin{example}[Uma espiada no nó]\label{ex:g8:intensity-ammeter:junction}
Ponha pedágios num par \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo}: estrada principal, \qty{0.50}{A};
\omterm{def:g5:series-parallel-circuits:parallel}{ramo} um, \qty{0.30}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{ramo} dois, \qty{0.20}{A}. A soma sugestiva
não é acaso --- as marchas que entram num \omterm{def:g5:series-parallel-circuits:parallel}{nó} e as que saem dele
precisam se equilibrar, já que carga nenhuma é criada ou destruída
numa bifurcação da estrada. A lei completa, com a companheira dela
para a tensão, dá as cartas no capítulo depois do próximo; os seus
aparelhos já a encontraram.
\end{example}

\begin{remark}[O multímetro]\label{rem:g8:intensity-ammeter:multimeter}
O tijolo amarelo da oficina --- o \emph{multímetro} --- dobra o
\omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} e vários outros instrumentos dentro de uma caixa só, com
um seletor. Seletor em \unit{A} ou \unit{mA}, pontas nos bornes
certos, e ele é o seu \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro}, com regras de série e tudo mais.
Dois capítulos deste livro já moram dentro dele; um terceiro se muda
para lá no próximo capítulo --- com regras de ligação, atenção,
\emph{opostas}.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:1}
O que a \omterm{def:g8:intensity-ammeter:intensity}{intensidade} mede, na linguagem dos rios? Dê a \omterm{def:g6:measurement-in-science:measuring}{unidade} e o
símbolo dela, e converta \qty{250}{mA} em \omterm{def:g8:intensity-ammeter:intensity}{ampères}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:2}
Ligue a \omterm{def:g8:intensity-ammeter:intensity}{intensidade} à cena: \qty{0.02}{A}; \ \qty{0.3}{A}; \
\qty{10}{A}; \ \qty{100}{A} --- chaleira, lampadinha indicadora,
\omterm{ex:g6:circuits-and-safety:motor}{motor} dando a partida, lâmpada de lanterna.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:3}
Por que um \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} precisa ser inserido \omterm{def:g4:series-circuits:series}{em série}? A que ele foi
construído para quase não se opor, e que acidente isso torna
possível?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:4}
Liste os quatro passos de uma medida de \omterm{def:g8:intensity-ammeter:intensity}{intensidade} bem feita. Por
que começar pela maior escala?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:5}
Enuncie a lei ``uma volta, uma \omterm{def:g8:intensity-ammeter:intensity}{intensidade}''. Que velha observação
sobre brilhos iguais ela transformou em números?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:6}
Um \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} antes de um \omterm{ex:g6:circuits-and-safety:motor}{motor} \omterm{def:g4:series-circuits:series}{em série} marca \qty{0.45}{A}. O que
ele vai marcar depois do \omterm{ex:g6:circuits-and-safety:motor}{motor}? E ao lado da \omterm{def:g3:first-electric-circuit:battery}{pilha}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:7}
Uma segunda lâmpada entra numa volta \omterm{def:g4:series-circuits:series}{em série} e os \qty{0.30}{A} do

miúdas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:8}
Um \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} digital marca $-0.25$ \unit{A}. O que o estudante fez,

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:9}
Estrada principal, \qty{0.60}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{ramo} um, \qty{0.35}{A}. Quanto

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:11}

Quanto carrega a estrada principal da \omterm{def:g3:first-electric-circuit:battery}{pilha}? Uma terceira lâmpada
igual entra \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo}: preveja a nova carga da estrada principal e

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:intensity-ammeter:12}
Projete o plano de medida completo de um \omterm{def:g5:series-parallel-circuits:parallel}{circuito em paralelo} de dois
\omterm{def:g5:series-parallel-circuits:parallel}{ramos} (lâmpada e \omterm{ex:g6:circuits-and-safety:motor}{motor}): quantas posições de \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} são
necessárias para conhecer \emph{todas} as correntes do circuito; o
número mínimo de medidas que basta (uma pode ser deduzida --- por qual
\omterm{def:g2:balance-equilibrium:equilibrium}{equilíbrio}?); e a ordem das operações que nunca deixa um aparelho
ligado perigosamente.
\end{exercise}

\section{Problema: A bancada de metrologia}

\begin{problem}[{Problema de fim de semana --- dia de qualificação na
oficina de eletrônica; quatro estações, um amperímetro em cada uma; a
licença do aprendiz}]\label{pb:g8:intensity-ammeter:1}
Para ganhar direito de bancada, cada aprendiz percorre quatro
estações de medida. Leve o capítulo junto.

\medskip\noindent\textbf{Parte I --- \omterm{def:g3:sun-days-seasons:year}{Estação} um: a volta simples.}
\omterm{def:g3:first-electric-circuit:battery}{Pilha}, \omterm{ex:g3:first-electric-circuit:switch}{interruptor}, uma lâmpada.
\begin{enumerate}
  \item Descreva, passo a passo, a inserção do \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} para \omterm{def:g6:measurement-in-science:measuring}{medir}
        a corrente da lâmpada.
  \item A leitura: \qty{280}{mA}. Expresse-a em \omterm{def:g8:intensity-ammeter:intensity}{ampères}.
  \item O examinador muda o seu aparelho para o outro lado da \omterm{def:g3:first-electric-circuit:battery}{pilha}.
        Preveja a leitura, citando a lei.
  \item ``E se eu tivesse dois \omterm{def:g8:intensity-ammeter:ammeter}{amperímetros} nesta volta ao mesmo
        tempo?'' Onde eles poderiam ficar, e o que mostrariam?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- \omterm{def:g3:sun-days-seasons:year}{Estação} dois: o par \omterm{def:g4:series-circuits:series}{em série}.}
Duas lâmpadas diferentes, A mais brilhante que B, \omterm{def:g4:series-circuits:series}{em série}.
\begin{enumerate}[resume]
  \item Um aprendiz prevê: ``A brilhante A carrega mais corrente que
        a fraca B.'' Teste a previsão com a lei --- e concilie as
        correntes iguais com os brilhos desiguais (o que precisa ser
        diferente entre as lâmpadas?).
  \item O aparelho marca \qty{150}{mA} entre A e B. Escreva todas as
        outras correntes da volta.
  \item A lâmpada B é curto-circuitada por um fio com garras. Preveja
        a nova leitura e o comportamento das duas lâmpadas.
  \item O examinador pede uma frase sobre por que a oficina proíbe
        garras atravessadas na \emph{\omterm{def:g3:first-electric-circuit:battery}{pilha}} e permite a demonstração
        atravessada em B.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- \omterm{def:g3:sun-days-seasons:year}{Estação} três: o \omterm{def:g5:series-parallel-circuits:parallel}{nó}.}
Um \omterm{def:g5:series-parallel-circuits:parallel}{ramo} com lâmpada e um \omterm{def:g5:series-parallel-circuits:parallel}{ramo} com \omterm{ex:g6:circuits-and-safety:motor}{motor}, \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo}.
\begin{enumerate}[resume]
  \item Estrada principal: \qty{0.52}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{ramo} da lâmpada:
        \qty{0.30}{A}. A corrente do \omterm{def:g5:series-parallel-circuits:parallel}{ramo} do \omterm{ex:g6:circuits-and-safety:motor}{motor}, pelo \omterm{def:g2:balance-equilibrium:equilibrium}{equilíbrio}?
  \item O \omterm{ex:g6:circuits-and-safety:motor}{motor} trava (ainda ligado, mas sem girar) e o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} dele
        sobe para \qty{0.40}{A}. Supondo que o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} da lâmpada
        mantenha os \qty{0.30}{A} dele, quanto carrega agora a
        estrada principal --- e que guarda da oficina está de ouvido
        atento?
  \item A lâmpada é desenroscada. Preveja as três leituras dos
        pedágios.
  \item Por que a \omterm{def:g3:first-electric-circuit:battery}{pilha} se esgota mais depressa com os dois \omterm{def:g5:series-parallel-circuits:parallel}{ramos}
        saudáveis --- onde a generosidade é paga?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- A licença.}
\begin{enumerate}[resume]
  \item Escreva o juramento do aprendiz: três linhas --- onde um
        \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} fica e onde ele nunca fica; o que uma volta
        garante; o que um \omterm{def:g5:series-parallel-circuits:parallel}{nó} equilibra.
\end{enumerate}
\end{problem}
