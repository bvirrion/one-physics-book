\chapter{A resistência e a lei de Ohm}\label{ch:g8:ohms-law}

Uma pergunta persegue a história elétrica inteira: alguns
componentes deixam a marcha disparar, outros a estrangulam até virar
um fiapo --- as lâmpadas dividem tensões de forma desigual, os fios
finos esquentam, os grossos ficam frios. A própria oposição agora
ganha um nome, uma \omterm{def:g6:measurement-in-science:measuring}{unidade} e --- coroando quatro anos de circuitos ---
a lei mais útil deste livro: três letras que amarram o empurrão, a
marcha e a oposição numa linha só de álgebra.

\section{A resistência}

\begin{definition}[Resistência]\label{def:g8:ohms-law:resistance}
A \emph{resistência}\index{resistência} $R$ de um componente mede com
que \omterm{def:g2:pushes-and-pulls:force}{força} ele se opõe à \omterm{def:g7:circuits-series-parallel:current}{corrente elétrica}: a uma dada \omterm{def:g8:voltage-voltmeter:voltage}{tensão}, quanto
maior a resistência, mais fraca a marcha que ele deixa passar. A
\omterm{def:g6:measurement-in-science:measuring}{unidade} dela é o \emph{ohm}\index{ohm} (símbolo $\Omega$, o ômega
grego maiúsculo), em homenagem ao físico e professor de escola que
achou a lei deste capítulo com fios caseiros e uma paciência heroica.
\end{definition}

\begin{definition}[Resistor]\label{def:g8:ohms-law:resistor}
Um \emph{resistor}\index{resistor} é um componente fabricado para ter
uma \omterm{def:g8:ohms-law:resistance}{resistência} escolhida e estável --- um obstáculo calibrado,
vendido desde frações de \omterm{def:g8:ohms-law:resistance}{ohm} até milhões deles. Os circuitos usam
resistores para fixar correntes de propósito: domando a marcha que
queimaria uma peça delicada, diminuindo o brilho, dividindo,
ajustando. As faixas coloridas deles soletram o valor de cada um; o
símbolo deles é um retângulo simples.
\end{definition}

\begin{example}[Resistências em volta de você]\label{ex:g8:ohms-law:values}
Um \omterm{def:g2:measuring-length:units}{metro} de fio grosso de cobre: centésimos de \omterm{def:g8:ohms-law:resistance}{ohm} --- a estrada
plana. Uma lâmpada de lanterna acesa: alguns \omterm{def:g8:ohms-law:resistance}{ohms}. Um motorzinho
elétrico: uns poucos \omterm{def:g8:ohms-law:resistance}{ohms}. A resistência da chaleira: cerca de
$25\,\Omega$. Pele humana seca, de mão a mão: dezenas de milhares de
\omterm{def:g8:ohms-law:resistance}{ohms} --- molhada, catastroficamente menos: a velha regra das mãos
úmidas é uma afirmação sobre \omterm{def:g8:ohms-law:resistance}{resistência}. Um \omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto:
\omterm{def:g8:ohms-law:resistance}{resistência} além de qualquer medida --- o obstáculo infinito.
\end{example}

\section{A experiência}

\begin{method}[Medir o retrato $U$--$I$ de um componente]\label{met:g8:ohms-law:portrait}
Um \omterm{def:g8:ohms-law:resistor}{resistor}, uma fonte ajustável (ou uma fila \omterm{def:g7:moon-phases:phases}{crescente} de \omterm{def:g3:first-electric-circuit:battery}{pilhas}), os
dois aparelhos:
\begin{enumerate}
  \item ligue a fonte, o \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} e o \omterm{def:g8:ohms-law:resistor}{resistor} \omterm{def:g4:series-circuits:series}{em série}; prenda o
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} atravessado no \omterm{def:g8:ohms-law:resistor}{resistor};
  \item ajuste a menor \omterm{def:g8:voltage-voltmeter:voltage}{tensão}; anote o par ($U$, $I$);
  \item aumente a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} passo a passo --- uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, duas, três ---
        anotando um par a cada vez;
  \item marque os pontos, $I$ na horizontal, $U$ na vertical, e
        olhe.
\end{enumerate}
\end{method}

\begin{example}[A confissão de um resistor]\label{ex:g8:ohms-law:confession}
A tabela de uma sessão, para um \omterm{def:g8:ohms-law:resistor}{resistor}:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
Dobre o empurrão, dobra a marcha; triplique, triplica: $U$ e $I$ são
\emph{proporcionais}, e toda razão $U/I$ devolve o mesmo número:
$15$. Essa razão constante não é acaso --- é a \omterm{def:g8:ohms-law:resistance}{resistência} do
\omterm{def:g8:ohms-law:resistor}{resistor}, $R = 15\,\Omega$, se mostrando em cada linha.
\end{example}

\begin{proposition}[A lei de Ohm]\label{prop:g8:ohms-law:law}
Para um \omterm{def:g8:ohms-law:resistor}{resistor} a \omterm{def:g3:temperature-thermometers:temperature}{temperatura} estável, a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} atravessada nele e a
corrente que passa por ele são proporcionais, e a constante de
proporcionalidade é a \omterm{def:g8:ohms-law:resistance}{resistência} dele:
\[
U = R \times I ,
\]
com $U$ em \omterm{def:g8:voltage-voltmeter:voltage}{volts}, $I$ em \omterm{def:g8:intensity-ammeter:intensity}{ampères} e $R$ em \omterm{def:g8:ohms-law:resistance}{ohms}. Rearranjada a gosto:
$I = U/R$ (a corrente que um empurrão impele através de um obstáculo)
e $R = U/I$ (o obstáculo, desmascarado por um par honesto de
leituras).
\end{proposition}

\begin{example}[A lei calcula tudo]\label{ex:g8:ohms-law:compute}
Uma vez conhecido $R$, a lei responde em todas as direções. A
resistência da chaleira, $25\,\Omega$ nos \qty{230}{V} da tomada:
$I = U/R = 230 \div 25 = \qty{9.2}{A}$ --- o aparelho de dez \omterm{def:g8:intensity-ammeter:intensity}{ampères}
explicado. Um \omterm{def:g8:ohms-law:resistor}{resistor} precisa limitar a corrente de uma lâmpada
delicada a \qty{0.02}{A} a partir de \qty{4.5}{V}:
$R = U/I = 4.5 \div 0.02 = 225\,\Omega$ --- escolha o próximo valor
padrão. Por $60\,\Omega$ passam \qty{0.05}{A}: a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} atravessada
nele é $U = 60 \times 0.05 = \qty{3}{V}$. Uma lei, três receitas, a
oficina inteira.
\end{example}

\begin{remark}[A muleta do triângulo]\label{rem:g8:ohms-law:triangle}
Alguns estudantes rabiscam $U$ em cima de $R$ e $I$ num
triangulozinho, tapando a letra procurada para ler a receita.
Inofensivo como muleta --- mas a sua álgebra agora rearranja
$U = R I$ honestamente numa linha, e o jeito honesto viaja para toda
fórmula que você vier a encontrar, com triângulo ou sem. Apoie-se na
muleta se precisar neste ano; planeje andar.
\end{remark}

\section{O que a resistência explica}

\begin{example}[Velhos mistérios numa nota fiscal só]\label{ex:g8:ohms-law:mysteries}
A divisão desigual da \omterm{def:g8:voltage-voltmeter:voltage}{tensão} do capítulo das leis: lâmpadas \omterm{def:g4:series-circuits:series}{em série}
repartem um mesmo $I$, então, por $U = R I$, o maior $R$ fica com o
maior $U$ --- a queda maior de quem trabalha mais duro, agora
calculável. O \omterm{def:g5:series-parallel-circuits:parallel}{ramo} estrangulado do capítulo do paralelo: o fio longo
e fino acrescentou \omterm{def:g8:ohms-law:resistance}{resistência}, e $I = U/R$ encolheu a parcela desse
\omterm{def:g5:series-parallel-circuits:parallel}{ramo}. O \omterm{def:g7:short-circuits-safety:device}{curto-circuito}: um desvio de $R$ quase zero, e então $I = U/R$
explode --- o estouro da boiada sempre foi uma divisão por quase
zero. Quatro anos de sabedoria qualitativa sobre circuitos, na nota
fiscal de uma lei só.
\end{example}

\begin{example}[Fios, por metro]\label{ex:g8:ohms-law:wires}
A \omterm{def:g8:ohms-law:resistance}{resistência} de um fio cresce com o comprimento dele e encolhe com a
espessura --- um fio longo e fino é uma estradinha estreita de
montanha, um curto e grosso é uma autoestrada. Daí os cabos gordos do
eletricista para os aparelhos famintos, a liga resistiva de propósito
longa e fina da resistência de aquecimento --- e o motivo de a mesma
lei que acende uma lâmpada aquecer uma torradeira: a \omterm{def:g8:ohms-law:resistance}{resistência} é
onde a marcha gasta o empurrão dela, e empurrão gasto, como o
capítulo da \omterm{def:g5:everyday-energy:energy}{energia} vai confirmar no ano que vem, vira \omterm{def:g5:heat-and-insulation:heat}{calor}.
\end{example}

\begin{remark}[Onde termina a jurisdição da lei]\label{rem:g8:ohms-law:limits}
A lei de Ohm é uma lei dos \omterm{def:g8:ohms-law:resistor}{resistores}, não um direito universal. Meça
o retrato de uma \emph{lâmpada de filamento} e os pontos se afastam
da reta: o filamento esquenta conforme a corrente cresce, e metal
quente resiste mais --- $R$ se recusa a ficar parado. Diodos, \omterm{ex:g6:circuits-and-safety:motor}{motores}
e pele viva também entortam os retratos deles, cada um do seu jeito. A
reta pela origem é a assinatura do \omterm{def:g8:ohms-law:resistor}{resistor}, não a da matéria ---
sempre peça o retrato de um componente antes de confiar a lei a ele.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
O que a \omterm{def:g8:ohms-law:resistance}{resistência} mede? Dê a \omterm{def:g6:measurement-in-science:measuring}{unidade} e o símbolo dela, e escreva a
lei de Ohm com os três rearranjos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
Complete cada par para um \omterm{def:g8:ohms-law:resistor}{resistor} de $30\,\Omega$:
$U = \qty{6}{V}$, $I = {?}$; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
Um componente mostra $U = \qty{4.5}{V}$ com $I = \qty{0.09}{A}$. A
\omterm{def:g8:ohms-law:resistance}{resistência} dele?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
Na experiência do retrato, que forma os pontos de um \omterm{def:g8:ohms-law:resistor}{resistor} de
verdade desenham, e o que uma reta mais inclinada quer dizer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
A partir da tabela do capítulo, confira $R$ pela segunda e pela
quarta coluna. Por que achar o mesmo número nas duas vezes é
justamente a questão toda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
A \omterm{def:g8:ohms-law:resistance}{resistência} de uma chaleira é de $25\,\Omega$; a de uma segunda
chaleira, $35\,\Omega$. Na mesma tomada, qual delas puxa a corrente
mais forte --- e quanto cada uma puxa (uma casa decimal)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
Explique com $U = R I$ por que, de duas lâmpadas \omterm{def:g4:series-circuits:series}{em série} que
repartem uma corrente, a de maior \omterm{def:g8:ohms-law:resistance}{resistência} fica com a maior
\omterm{def:g8:voltage-voltmeter:voltage}{tensão}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
Traduza para a linguagem da \omterm{def:g8:ohms-law:resistance}{resistência}: o estouro da boiada do
\omterm{def:g7:short-circuits-safety:device}{curto-circuito}; \ o bloqueio total do \omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
Uma lampadinha indicadora delicada tolera \qty{20}{mA}. Que \omterm{def:g8:ohms-law:resistor}{resistor}
\omterm{def:g4:series-circuits:series}{em série} a protege numa bateria de \qty{9}{V}, se a própria lâmpada
fica com uns \qty{2}{V} da parcela? (A parcela do \omterm{def:g8:ohms-law:resistor}{resistor} primeiro,
depois $R$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
O retrato de uma lâmpada de filamento, medido: (\qty{1}{V},
\qty{0.25}{A}), (\qty{4}{V}, \qty{0.5}{A}). Mostre que nenhum $R$
único serve para os dois pontos, e conte a história física por trás
da entortada.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
Dois \omterm{def:g8:ohms-law:resistor}{resistores} \omterm{def:g4:series-circuits:series}{em série}, $10\,\Omega$ e $20\,\Omega$, em
\qty{6}{V}. Usando as leis da série mais a lei de Ohm em cada um:
ache a corrente da volta e a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} de cada \omterm{def:g8:ohms-law:resistor}{resistor}. (Dica: uma
corrente $I$ satisfaz $6 = 10 I + 20 I$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
Os mesmos dois \omterm{def:g8:ohms-law:resistor}{resistores}, agora \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo} em \qty{6}{V}. Ache a
corrente de cada \omterm{def:g5:series-parallel-circuits:parallel}{ramo}, o total da bateria e então --- o fecho
elegante --- a \omterm{def:g8:ohms-law:resistance}{resistência} única $R$ que puxaria esse mesmo total de
\qty{6}{V}. Compare-a com os dois \omterm{def:g8:ohms-law:resistor}{resistores} e explique, na linguagem
das estradas, por que a dupla resiste \emph{menos} do que qualquer um
dos membros sozinho.
\end{exercise}

\section{Problema: A bancada do controle de qualidade}

\begin{problem}[{Problema de fim de semana --- dia de inspeção na
fábrica de resistores; retratos, tolerâncias e um componente que não
é resistor nenhum}]\label{pb:g8:ohms-law:1}
A bancada de qualidade da fábrica testa a produção do dia --- e um
impostor. Os aparelhos ficam com você.

\medskip\noindent\textbf{Parte I --- Calibrando a bancada.}
\begin{enumerate}
  \item Esboce (ou descreva) o circuito de teste: fonte, \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro},
        o componente sob teste, \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} --- quem fica \omterm{def:g4:series-circuits:series}{em série},
        quem fica atravessado?
  \item A amostra um, em \qty{4.5}{V}, deixa passar \qty{0.15}{A}. A
        \omterm{def:g8:ohms-law:resistance}{resistência} dela?
  \item A etiqueta afirma $33\,\Omega$, com tolerância de cinco por
        cento. Calcule a faixa aceitável e dê o veredito sobre a
        amostra um. (Cinco por cento de $33$ primeiro.)
  \item Amostra dois, etiquetada $150\,\Omega$: preveja a corrente
        dela em \qty{4.5}{V}, para a bancada saber o que esperar (uma
        casa decimal, em \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Retratos completos.}
\begin{enumerate}[resume]
  \item A tabela da amostra três: $(1.5, 0.05)$, $(3.0, 0.10)$,
        $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{volts}, \omterm{def:g8:intensity-ammeter:intensity}{ampères}. Ajuste $R$,
        citando a característica dos números que permite um valor
        único.
  \item A tabela da amostra quatro: $(1.5, 0.30)$, $(3.0, 0.45)$,
        $(4.5, 0.54)$, $(6.0, 0.60)$. Mostre o impostor: nenhum $R$
        único, e o sentido da deriva.
  \item A amostra quatro é, na verdade, uma lampadinha de filamento
        da linha de produção vizinha. Conte a história da entortada
        dela --- e por que o teste da reta da bancada é justamente um
        detector de \omterm{def:g8:ohms-law:resistor}{resistores}.
  \item O regulamento da bancada diz: ``todo retrato precisa ser
        tirado a \omterm{def:g3:temperature-thermometers:temperature}{temperatura} estável''. Que cláusula da lei de Ohm a
        regra está protegendo?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Departamento de aplicações.}
\begin{enumerate}[resume]
  \item Um pedido pede um \omterm{def:g8:ohms-law:resistor}{resistor} que limite a corrente a
        \qty{30}{mA} numa fonte de \qty{12}{V} (a parcela da própria
        carga é desprezível). Calcule o valor a cotar.
  \item Um cliente reclama que a resistência de aquecimento de
        $25\,\Omega$ dele ``só'' puxa \qty{9.2}{A} enquanto o \omterm{def:g7:short-circuits-safety:fuse}{fusível}
        permite $10$. Calcule que \omterm{def:g8:voltage-voltmeter:voltage}{tensão} de alimentação a reclamação
        dele implica, e tranquilize-o com a aritmética.
  \item O laboratório de protótipos pede o fio de ligação de cobre
        mais grosso e mais curto da fábrica, ``com o mais perto de
        zero \omterm{def:g8:ohms-law:resistance}{ohm} possível''. Para que papel de circuito esse fio
        nasceu --- e perto de que vilão do dia a dia a \omterm{def:g8:ohms-law:resistance}{resistência}
        quase nula dele o coloca, se for mal usado?
  \item Encerre o registro da inspeção: três frases --- a lei, o
        retrato dela e a única condição sob a qual um componente pode
        assiná-la.
\end{enumerate}
\end{problem}
