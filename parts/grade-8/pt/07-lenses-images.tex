\chapter{As lentes e as imagens}\label{ch:g8:lenses-images}

A câmara de furinho comprou a imagenzinha nítida dela por um preço
cruel: quase nada de \omterm{def:g1:light-and-shadows:source}{luz}. E no entanto o seu próprio olho pinta um
mundo claro e definido na parede do fundo dele o dia inteiro, e não há
furinho nenhum à vista. O segredo do olho é um disco de matéria
vítrea e curva que recolhe um \omterm{def:g6:light-propagation:ray}{feixe} inteiro e o entorta sob encomenda
--- a \omterm{def:g8:lenses-images:families}{lente}, \omterm{def:g4:solar-system:star}{estrela} das câmeras, dos óculos, das lupas e deste
capítulo.

\section{Duas famílias de lente}

\begin{definition}[Lentes convergentes e divergentes]\label{def:g8:lenses-images:families}
Uma \emph{lente}\index{lente} é um pedaço de material \omterm{def:g7:light-sources-propagation:coats}{transparente}
--- vidro ou plástico claro --- de faces curvas, construído para
entortar a \omterm{def:g1:light-and-shadows:source}{luz} de um jeito controlado (a \omterm{def:g1:light-and-shadows:source}{luz} se entorta ao passar
para dentro do vidro; a lei exata dessa entortada é um tesouro do
ensino médio). Duas famílias:
\begin{enumerate}
  \item as \emph{lentes convergentes}\index{lente convergente} ---
        mais grossas no meio do que na borda --- entortam os raios de
        um \omterm{def:g6:light-propagation:ray}{feixe} \emph{um na direção do outro};
  \item as \emph{lentes divergentes}\index{lente divergente} ---
        mais finas no meio --- afastam os raios uns dos outros.
\end{enumerate}
Passe a ponta do dedo por uma lente de óculos: uma barriga no meio ou
uma cova no meio classifica toda lente da Terra na família dela.
\end{definition}

\begin{definition}[Foco e distância focal]\label{def:g8:lenses-images:focus}
Mande um punhado de raios paralelos --- a \omterm{def:g1:light-and-shadows:source}{luz} do Sol serve --- por uma
\omterm{def:g8:lenses-images:families}{lente convergente}: os raios entortados se cruzam todos num ponto
atrás dela, o \emph{foco}\index{foco}. A distância da \omterm{def:g8:lenses-images:families}{lente} ao foco é
a \emph{distância focal}\index{distância focal} $f$: o único número
que define a \omterm{def:g8:lenses-images:families}{lente}. \omterm{def:g8:lenses-images:families}{Lentes} gordas e muito curvas entortam com força
--- $f$ curto; as de curvatura suave entortam pouco --- $f$ longo.
\end{definition}

\begin{method}[Medir uma distância focal]\label{met:g8:lenses-images:measure}
Para qualquer \omterm{def:g8:lenses-images:families}{lente convergente}, um minuto e uma janela distante:
\begin{enumerate}
  \item fique de frente para uma parede oposta a uma cena clara e
        \emph{distante} --- a janela e o mundo lá fora;
  \item segure a \omterm{def:g8:lenses-images:families}{lente} perto da parede e a afaste devagar até a cena
        estalar numa figurinha nítida na parede (de cabeça para baixo
        --- como você já esperava);
  \item meça a distância da \omterm{def:g8:lenses-images:families}{lente} à parede: para uma cena distante,
        essa distância \emph{é} a \omterm{def:g8:lenses-images:focus}{distância focal} --- os raios de
        longe chegam praticamente paralelos, e raios paralelos se
        recolhem no \omterm{def:g8:lenses-images:focus}{foco}.
\end{enumerate}
Uma lupa mede uns \qty{10}{cm}; as \omterm{def:g8:lenses-images:families}{lentes} de óculos, dezenas de
\omterm{def:g2:measuring-length:units}{centímetros}; a \omterm{def:g8:lenses-images:families}{lente} atarracada de uma câmera, uns poucos.
\end{method}

\begin{remark}[O vidro que queima]\label{rem:g8:lenses-images:burning}
No \omterm{def:g8:lenses-images:focus}{foco} de uma \omterm{def:g8:lenses-images:families}{lente convergente}, uma largura inteira de \omterm{def:g6:light-propagation:ray}{feixe} de \omterm{def:g1:light-and-shadows:source}{luz}
solar se aglomera num pontinho em brasa --- concentrado o bastante
para carbonizar papel e começar incêndios em segundos. Trate-o pelo
que ele é: o princípio do fogão solar em formato de bolso. Nunca
concentre \omterm{def:g1:light-and-shadows:source}{luz} do Sol perto de nada inflamável, nunca na direção da
pele ou da roupa --- e, com dobro de razão sobre a velha lei, nunca
olhe para o Sol \emph{através} de \omterm{def:g8:lenses-images:families}{lente} nenhuma: a focalização do
próprio olho mais a da \omterm{def:g8:lenses-images:families}{lente} fazem uma queimadura instantânea,
indolor e permanente.
\end{remark}

\section{A lente como fabricante de imagens}

\begin{proposition}[Imagens reais]\label{prop:g8:lenses-images:realimage}
Uma \omterm{def:g8:lenses-images:families}{lente convergente} recolhe os raios difundidos por cada ponto de
um objeto e os entrega a um ponto correspondente do outro lado: ponto
a ponto, ela monta uma figura de verdade --- uma \emph{imagem
real}\index{imagem real} --- que pode ser apanhada numa tela. A
\omterm{prop:g5:mirrors-reflection:image}{imagem} é invertida, como a do furinho, mas clara: a \omterm{def:g8:lenses-images:families}{lente} colhe uma
\omterm{def:g8:lenses-images:families}{lente} inteira de \omterm{def:g1:light-and-shadows:source}{luz} por ponto, onde o furinho deixava passar um
único fiozinho. Regras de bolso da geometria: um objeto distante se
imagina pequeno e perto do \omterm{def:g8:lenses-images:focus}{foco}; conforme o objeto se aproxima da
\omterm{def:g8:lenses-images:families}{lente}, a \omterm{prop:g5:mirrors-reflection:image}{imagem} dele recua e cresce.
\end{proposition}

\begin{example}[Câmera e olho, uma arquitetura só]\label{ex:g8:lenses-images:cameraeye}
Uma câmera é a fabricante de imagens dentro de uma caixa: \omterm{def:g8:lenses-images:families}{lente
convergente} na frente, tela sensível à \omterm{def:g1:light-and-shadows:source}{luz} atrás, \omterm{prop:g5:mirrors-reflection:image}{imagem} formada
invertida sobre ela (o programa gira o arquivo com educação). O seu
olho é a mesma máquina em tecido vivo: uma \omterm{def:g8:lenses-images:families}{lente convergente} atrás da
pupila, a \emph{retina} como tela lá no fundo, \omterm{prop:g5:mirrors-reflection:image}{imagem} invertida --- e
o cérebro a desvira, como ficou dito quando o furinho nos ensinou
isso pela primeira vez. A focalização difere de um jeito bonito: a
câmera desloca a \omterm{def:g8:lenses-images:families}{lente} dela; a \omterm{def:g8:lenses-images:families}{lente} mole do olho \emph{muda de
forma}, puxada mais gorda para as coisas perto, relaxada mais fina
para as de longe --- sinta o esforço depois de uma hora de leitura de
perto.
\end{example}

\begin{example}[Óculos]\label{ex:g8:lenses-images:spectacles}
Quando a focalização do próprio olho fica devendo, uma \omterm{def:g8:lenses-images:families}{lente} na
frente completa o serviço. O olho míope focaliza forte demais --- as
cenas distantes se montam \emph{antes} da retina e chegam borradas:
uma \omterm{def:g8:lenses-images:families}{lente} de óculos \emph{divergente} desentorta a \omterm{def:g1:light-and-shadows:source}{luz} na medida
certa. Os olhos hipermetropes (e a maioria dos mais velhos) focalizam
fraco demais para o trabalho de perto: uma \omterm{def:g8:lenses-images:families}{lente} \emph{convergente}
empresta a \omterm{def:g2:pushes-and-pulls:force}{força} que falta --- os óculos de leitura são honestos
priminhos da lupa. Duas famílias de \omterm{def:g8:lenses-images:families}{lente}, duas famílias de borrão.
\end{example}

\section{A lupa}

\end{omfigure}

\begin{example}[Dentro da distância focal]\label{ex:g8:lenses-images:magnifier}
Segure uma \omterm{def:g8:lenses-images:families}{lente convergente} \emph{mais perto} de um selo do que a
\omterm{def:g8:lenses-images:focus}{distância focal} dela, e a figura muda de caráter: \omterm{prop:g5:mirrors-reflection:image}{imagem} nenhuma cai
em tela nenhuma --- em vez disso, olhando pela \omterm{def:g8:lenses-images:families}{lente}, você vê o selo
direito, no lugar dele, e aumentado. A \omterm{def:g8:lenses-images:families}{lente} entorta os raios do selo
de modo que eles chegam ao olho \emph{como se} tivessem sido
difundidos por um selo mais grandioso atrás do vidro --- uma ilusão
sob encomenda, prima do mundo fantasma do \omterm{def:g5:mirrors-reflection:reflection}{espelho}. Relojoeiros,
colecionadores de selos e você mesmo, no passado, com a lupa de
criança já usaram isso; a geometria completa dos raios desse truque é
uma joia do volume do ensino médio.
\end{example}

\begin{remark}[Instrumentos sobre instrumentos]\label{rem:g8:lenses-images:instruments}
Empilhe \omterm{def:g8:lenses-images:families}{lentes} e o mundo se abre. Duas \omterm{def:g8:lenses-images:families}{lentes convergentes} fazem um
microscópio: uma joga uma \omterm{prop:g8:lenses-images:realimage}{imagem real} grande de uma gotinha, a
segunda amplia essa \omterm{prop:g5:mirrors-reflection:image}{imagem} --- foi assim que os defensores das
\omterm{def:g7:states-of-matter:molecule}{moléculas} ganharam os primeiros aliados diretos deles. Uma \omterm{def:g8:lenses-images:families}{lente}
grande (ou um \omterm{def:g5:mirrors-reflection:reflection}{espelho} curvo) mais uma lupa fazem um telescópio: os
raios paralelos e fracos de uma galáxia distante recolhidos numa
\omterm{prop:g8:lenses-images:realimage}{imagem real}, depois ampliados dentro de um olho espantado. Todo
instrumento óptico da civilização são as duas ideias deste capítulo
--- recolher, depois ampliar --- em arranjos variados.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:1}
Classifique pelo tato: como a ponta do dedo distingue uma \omterm{def:g8:lenses-images:families}{lente
convergente} de uma divergente? O que cada uma faz com um \omterm{def:g6:light-propagation:ray}{feixe}
paralelo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:2}
Defina \omterm{def:g8:lenses-images:focus}{foco} e \omterm{def:g8:lenses-images:focus}{distância focal}. O que uma \omterm{def:g8:lenses-images:focus}{distância focal} curta diz
sobre a curvatura e sobre o poder de entortar de uma \omterm{def:g8:lenses-images:families}{lente}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:3}
Descreva a medida de $f$ com a janela e a parede. Por que a cena
precisa ser distante?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:4}
Enuncie duas regras de segurança do vidro que queima, e o motivo de

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:5}
O que é uma \omterm{prop:g8:lenses-images:realimage}{imagem real}? Dê as duas propriedades que a assinam, e a

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:6}
Mapeie a câmera sobre o olho: quem faz o papel de \omterm{def:g8:lenses-images:families}{lente}, de tela e de

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:7}
Que família de \omterm{def:g8:lenses-images:families}{lente} de óculos serve ao olho míope, e por quê? E ao

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:lenses-images:8}
O que precisa ser verdade sobre a posição do selo para o truque da

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:lenses-images:9}

a \omterm{def:g8:lenses-images:families}{lente} precisa se mover para manter a \omterm{prop:g5:mirrors-reflection:image}{imagem} nítida no sensor, pelas

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:lenses-images:10}
Num dia de sol, uma \omterm{def:g8:lenses-images:families}{lente} largada com descuido sobre folhas secas é

\begin{exercise}[$\star\star$]\label{exo:g8:lenses-images:11}
Os óculos de leitura do avô servem também de lupa de emergência e,
nos feriados, de acendedor solar de charuto --- mas os óculos do neto

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:lenses-images:12}
O microscópio e o telescópio empilham os dois duas \omterm{def:g8:lenses-images:families}{lentes

galáxias. Contraste os dois estágios deles: o que a \omterm{def:g8:lenses-images:families}{lente} um recebe e
produz em cada instrumento (objeto perto contra \omterm{def:g1:light-and-shadows:source}{luz} de \omterm{def:g4:solar-system:star}{estrela}
praticamente paralela), e por que o telescópio quer a \omterm{def:g8:lenses-images:families}{lente} um
\emph{grande} acima de tudo, enquanto o microscópio a quer
\emph{muito curva}?
\end{exercise}

\section{Problema: A bancada da óptica}

\begin{problem}[{Problema de fim de semana --- de aprendiz na óptica
da cidade; distâncias focais, adaptações e um telescópio
rachado}]\label{pb:g8:lenses-images:1}
Um dia na óptica: \omterm{def:g8:lenses-images:families}{lentes} para classificar, óculos para casar e o
telescópio de um cliente em pedaços.

\medskip\noindent\textbf{Parte I --- Classificando a bandeja.}
Uma gaveta derrubada misturou o estoque.
\begin{enumerate}
  \item Proponha dois testes rápidos --- um pelo tato, outro com a
        janela e a parede --- para separar as \omterm{def:g8:lenses-images:families}{lentes convergentes} das
        divergentes.
  \item A \omterm{def:g8:lenses-images:families}{lente} A joga uma \omterm{prop:g5:mirrors-reflection:image}{imagem} nítida da rua na parede a
        \qty{25}{cm}. A \omterm{def:g8:lenses-images:focus}{distância focal} e a família dela?
  \item A \omterm{def:g8:lenses-images:families}{lente} B se recusa a jogar \omterm{prop:g5:mirrors-reflection:image}{imagem} nenhuma na parede a
        distância nenhuma, e as coisas vistas por ela parecem
        menores. A família dela?
  \item As \omterm{def:g8:lenses-images:families}{lentes} C e D são as duas convergentes; C faz a \omterm{prop:g5:mirrors-reflection:image}{imagem} da
        janela a \qty{10}{cm}, D a \qty{40}{cm}. Qual é a mais curva
        e a que entorta com mais \omterm{def:g2:pushes-and-pulls:force}{força}?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Adaptações.}
\begin{enumerate}[resume]
  \item Um cliente vê borradas as placas distantes da estrada mas lê
        cardápios sem esforço. Diagnostique o olho e receite a
        família.
  \item Outro segura o jornal com o braço esticado e reclama que os
        braços dele são ``curtos demais''. Diagnostique e receite.
  \item O óptico avisa o segundo cliente: ``os seus óculos de leitura
        novos vão borrar as montanhas.'' Por que óculos de leitura
        servem só para as páginas de perto?
  \item Uma criança pergunta por que as \omterm{def:g8:lenses-images:families}{lentes} de teste do óptico
        fazem a tabela de letras nadar enquanto os olhos dela estão
        ``ótimos''. O que uma \omterm{def:g8:lenses-images:families}{lente} corretiva está fazendo com um
        olho bem focalizado?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O telescópio rachado.}
Um cliente traz um telescópio de observar \omterm{def:g4:solar-system:star}{estrelas} com a \omterm{def:g8:lenses-images:families}{lente} da
frente rachada, perguntando se a lupa reserva que está na caixa pode
substituí-la.

\begin{enumerate}[resume]
  \item Diga o serviço das duas \omterm{def:g8:lenses-images:families}{lentes} no instrumento saudável: o que
        a grande \omterm{def:g8:lenses-images:families}{lente} da frente recebe de uma \omterm{def:g4:solar-system:star}{estrela} e o que faz
        disso, e o que a \omterm{def:g8:lenses-images:families}{lente} pequena do olho faz com aquilo?
  \item A \omterm{def:g8:lenses-images:families}{lente} da frente é larga como um pires; a do olho, do
        tamanho de uma moeda. Por que a \emph{largura} da \omterm{def:g8:lenses-images:families}{lente} da
        frente importa tanto para as \omterm{def:g4:solar-system:star}{estrelas} fracas?
  \item A lupa reserva é muito curva mas do tamanho de uma moeda.
        Julgue a substituição: o que se ganharia e o que se perderia
        na frente do instrumento?
  \item Encerre o dia com o credo de uma linha do óptico sobre todo
        instrumento da prateleira --- óculos, lupas, câmeras,
        telescópios --- e o resumo em dois verbos do que as \omterm{def:g8:lenses-images:families}{lentes}
        fazem pela civilização.
\end{enumerate}
\end{problem}
