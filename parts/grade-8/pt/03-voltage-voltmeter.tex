\chapter{A tensão e o voltímetro}\label{ch:g8:voltage-voltmeter}

Desde que a primeira \omterm{def:g3:first-electric-circuit:battery}{pilha} acendeu a primeira lâmpada, uma palavra
faz um serviço pesado e vago neste livro: o ``empurrão''. As \omterm{def:g3:first-electric-circuit:battery}{pilhas}
empurram fraquinho, a tomada empurra assassinamente, duas \omterm{def:g3:first-electric-circuit:battery}{pilhas} uma
atrás da outra empurram juntas. O empurrão agora vira uma grandeza
medida, com uma \omterm{def:g6:measurement-in-science:measuring}{unidade} famosa --- e o instrumento dele, ao contrário
do do capítulo passado, se recusa a ficar dentro da estrada.

\section{A tensão}

\begin{definition}[Tensão]\label{def:g8:voltage-voltmeter:voltage}
A \emph{tensão}\index{tensão} $U$ entre dois pontos de um circuito
mede o empurrão elétrico disponível entre eles --- com que \omterm{def:g2:pushes-and-pulls:force}{força} a
corrente é impelida de um ponto ao outro, assim como a altura de uma
cachoeira mede com que \omterm{def:g2:pushes-and-pulls:force}{força} ela toca a roda do moinho. A tensão é
sempre um \emph{entre}: entre os dois \omterm{def:g3:first-electric-circuit:battery}{terminais} de uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, entre os
dois lados de uma lâmpada --- nunca num único ponto sozinho. A
\omterm{def:g6:measurement-in-science:measuring}{unidade} dela é o \emph{volt}\index{volt} (\unit{V}), em homenagem ao
inventor da primeira \omterm{def:g3:first-electric-circuit:battery}{pilha}.
\end{definition}

\begin{example}[O álbum de família dos volts]\label{ex:g8:voltage-voltmeter:values}
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha} redonda sozinha: \qty{1.5}{V}. A bateria chata de bolso:
\qty{4.5}{V} --- três \omterm{def:g3:first-electric-circuit:battery}{pilhas} uma atrás da outra dentro de uma
embalagem só. A retangular de nove \omterm{def:g8:voltage-voltmeter:voltage}{volts}, com os dois \omterm{def:g3:first-electric-circuit:battery}{terminais} em
cima: \qty{9}{V}, seis \omterm{def:g3:first-electric-circuit:battery}{pilhas} escondidas. Uma bateria de carro:
\qty{12}{V}. A tomada da parede: \qty{230}{V} --- mais empurrão do que
cento e cinquenta pilhinhas, que é todo o conteúdo quantitativo do
``na tomada, nunca''. Um raio entre a nuvem e o chão: centenas de
milhões de \omterm{def:g8:voltage-voltmeter:voltage}{volts}, por um instante.
\end{example}

\begin{proposition}[Tensão sem corrente]\label{prop:g8:voltage-voltmeter:nocurrent}
\omterm{def:g8:voltage-voltmeter:voltage}{Tensão} e corrente são grandezas diferentes, e a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} vem primeiro.
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha} sozinha numa gaveta, ligada a nada, não impele corrente
nenhuma --- e mesmo assim, entre os \omterm{def:g3:first-electric-circuit:battery}{terminais} dela, os \qty{1.5}{V}
inteiros estão de prontidão, mensuráveis, como a altura de uma
cachoeira num dia em que o moinho está fechado. O empurrão pode
existir sem marcha; a marcha nunca existe sem empurrão.
\end{proposition}

\begin{definition}[Voltímetro]\label{def:g8:voltage-voltmeter:voltmeter}
Um \emph{voltímetro}\index{voltímetro} mede a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} entre dois
pontos --- então é ligado \emph{atravessado} neles: \emph{\omterm{def:g5:series-parallel-circuits:parallel}{em
paralelo}} com o componente de interesse, uma ponta de cada lado, como
um topógrafo segurando um nível entre duas alturas. Construído ao
contrário do \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} em tudo, ele se opõe à corrente quase
\emph{completamente}: ligado atravessado em qualquer coisa, ele puxa
quase nada --- então prendê-lo não atrapalha nada, e volta nenhuma
precisa jamais ser aberta.
\end{definition}

{\small O \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} assume a pose do topógrafo: atravessado na

nunca ficando dentro da estrada.}
\end{omfigure}

\begin{method}[Medir uma tensão]\label{met:g8:voltage-voltmeter:measure}
\begin{enumerate}
  \item deixe o circuito intacto --- nada se abre para um
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro};
  \item prenda uma ponta de cada lado do componente: a ponta \unit{V}
        onde a corrente convencional entra, a ponta comum onde ela
        sai;
  \item maior escala primeiro, descendo até ficar confortável;
  \item leia, e escreva a \omterm{def:g6:measurement-in-science:measuring}{unidade}.
\end{enumerate}
Os modos dos dois aparelhos, lado a lado: \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} --- abra a
estrada, fique dentro dela, não se oponha a nada; \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} --- não
abra nada, fique de lado, oponha-se a tudo. Confundir os dois modos é
o hábito mais caro da oficina.
\end{method}

\section{Tensões de projeto}

\begin{definition}[Tensão nominal]\label{def:g8:voltage-voltmeter:nominal}
Toda lâmpada, \omterm{ex:g6:circuits-and-safety:motor}{motor} e aparelho carrega uma \emph{tensão
nominal}\index{tensão nominal} --- o empurrão para o qual foi
construído, gravado na base ou na placa: uma lâmpada de lanterna
``3,5 V'', um farol de carro ``12 V'', uma chaleira ``230 V''.
Alimentado com a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} nominal, o aparelho funciona como foi
projetado; alimentado com muito menos, fica emburrado --- lâmpada
fraca, \omterm{ex:g6:circuits-and-safety:motor}{motor} preguiçoso; alimentado com muito mais, queima ---
brilhantemente, uma vez só.
\end{definition}

\begin{example}[Casando lâmpada e pilha]\label{ex:g8:voltage-voltmeter:matching}
Uma lâmpada de ``3,5 V'' na bateria chata de \qty{4.5}{V}: um pouco
forçada --- brilhante e de vida curta. Numa única \omterm{def:g3:first-electric-circuit:battery}{pilha} de
\qty{1.5}{V}: um brilho alaranjado e sonolento. Na bateria de
\qty{9}{V}: um instante ofuscante, e então o fio fininho se parte ---
o obituário do filamento. A etiqueta é um contrato: cumpra-o, e a
lâmpada vive a vida anunciada dela.
\end{example}

\begin{remark}[Por que 230 V matam]\label{rem:g8:voltage-voltmeter:mains}
Agora a velha regra absoluta ganha o número dela. A estrada
condutora úmida do corpo, quando lhe oferecem \qty{4.5}{V}, deixa
passar uma corrente pequena demais para ser sentida. A mesma estrada
a \qty{230}{V} --- cinquenta vezes o empurrão --- deixa passar
cinquenta vezes a corrente, bem além dos poucos centésimos de \omterm{def:g8:intensity-ammeter:intensity}{ampère}
que param um coração. A tomada não é ``um pouquinho'' mais forte que
a sua bancada de \omterm{def:g3:first-electric-circuit:battery}{pilhas}: ela está duas ordens de grandeza além, e a
severidade do regulamento é aritmética pura.
\end{remark}

\section{As primeiras leituras de tensão}

\begin{example}[Em volta de uma volta simples]\label{ex:g8:voltage-voltmeter:readings}
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, uma lâmpada, aparelhos na mão. Atravessado na \omterm{def:g3:first-electric-circuit:battery}{pilha}:
\qty{4.5}{V}. Atravessado na lâmpada: \qty{4.5}{V} --- a lâmpada
recebe o que a \omterm{def:g3:first-electric-circuit:battery}{pilha} oferece. Atravessado num trecho de \emph{fio} de
ligação: \qty{0}{V}, ou o mais perto disso que o aparelho consegue
dizer --- um bom fio é uma estrada plana, que não gasta nada do
empurrão. O empurrão mora atravessado nos \emph{trabalhadores}, e não
ao longo das estradas: uma leitura que vale a pena guardar para
quando as leis do próximo capítulo a tornarem oficial.
\end{example}

\begin{example}[A confissão do interruptor aberto]\label{ex:g8:voltage-voltmeter:openswitch}
Lâmpada apagada, \omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto --- para onde foi o empurrão?
Meça: atravessado na lâmpada apagada, \qty{0}{V}; atravessado no
\emph{\omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto}, os \qty{4.5}{V} inteiros. O empurrão da
\omterm{def:g3:first-electric-circuit:battery}{pilha} fica inteiro sobre a abertura, esperando --- o único lugar da
volta onde dois pontos estão de verdade ``longe um do outro''
eletricamente. Feche o \omterm{ex:g3:first-electric-circuit:switch}{interruptor} e as leituras trocam de lugar. Um
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto é o lugar onde a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} espera.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:1}
O que a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} mede, e por que ela é sempre um ``entre''? Dê a
\omterm{def:g6:measurement-in-science:measuring}{unidade} dela e a \omterm{prop:g5:mirrors-reflection:image}{imagem} da cachoeira.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:2}
De cabeça: as tensões de uma \omterm{def:g3:first-electric-circuit:battery}{pilha} redonda, da bateria chata, da
bateria retangular, de uma bateria de carro, da tomada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:3}
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha} está largada, ligada a nada. Que corrente circula? O que um
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} atravessado nela marca? Qual proposição fala?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:4}
Contraste os modos dos dois aparelhos: ligação, abertura da estrada,
oposição à corrente. Qual erro constrói um \omterm{def:g7:short-circuits-safety:device}{curto-circuito} com
mostrador?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:5}
O que é uma \omterm{def:g8:voltage-voltmeter:nominal}{tensão nominal}? Preveja o comportamento de uma lâmpada de
``6 V'' em \qty{1.5}{V}; em \qty{6}{V}; em \qty{12}{V}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:6}
Na volta simples, o que o \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} marca atravessado na \omterm{def:g3:first-electric-circuit:battery}{pilha}, na
lâmpada e num trecho de fio? Onde mora o empurrão?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:7}
\omterm{ex:g3:first-electric-circuit:switch}{Interruptor} aberto: o que fica atravessado na lâmpada apagada, e no
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto? O que acontece com as duas leituras no clique?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:8}
Por que prender um \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} atravessado numa lâmpada acesa quase
não atrapalha o circuito? O que o aparelho se recusa a fazer?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:9}
A bateria chata de \qty{4.5}{V} esconde três \omterm{def:g3:first-electric-circuit:battery}{pilhas}. Desenhe (ou
descreva) o arranjo delas e calcule a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} se uma \omterm{def:g3:first-electric-circuit:battery}{pilha} fosse
invertida sem querer --- o capítulo da lanterna, de anos atrás, agora
com números. (Empurrões opostos se subtraem.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:10}
A lâmpada de um campista é de \qty{12}{V}; a bateria da van marca
\qty{12.6}{V} carregada e \qty{11.2}{V} depois de uma longa noite.
Descreva a noite da lâmpada, e por que nada queimou.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:11}
Um \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} ligado por engano \emph{\omterm{def:g4:series-circuits:series}{em série}} com a lâmpada (volta
aberta, aparelho na abertura): preveja o comportamento da lâmpada e a
leitura do aparelho --- lembre do que um \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} se opõe. (O
silêncio da lâmpada e o número do aparelho são os dois instrutivos.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:voltage-voltmeter:12}
Duas lâmpadas iguais \omterm{def:g4:series-circuits:series}{em série} em \qty{4.5}{V}: um \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} acha uns
\qty{2.25}{V} atravessados em cada uma. As mesmas lâmpadas \omterm{def:g5:series-parallel-circuits:parallel}{em
paralelo} na mesma bateria: preveja a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} de cada lâmpada e
concilie as duas descobertas com as regras de brilho que você conhece
há anos (pares \omterm{def:g4:series-circuits:series}{em série} apagados, pares \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo} brilhando
inteiro). Qual capítulo por vir vai transformar as suas previsões em
lei?
\end{exercise}

\section{Problema: A auditoria da gaveta de pilhas}

\begin{problem}[{Problema de fim de semana --- a grande auditoria da
gaveta de pilhas; um voltímetro, uma caixa de sapato cheia de pilhas
e a estante de bugigangas da família}]\label{pb:g8:voltage-voltmeter:1}
Toda casa tem a gaveta: \omterm{def:g3:first-electric-circuit:battery}{pilhas} soltas, baterias misteriosas,
aparelhos que ``precisam de \omterm{def:g3:first-electric-circuit:battery}{pilhas} novas''. Dia de auditoria ---
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} na mão.

\medskip\noindent\textbf{Parte I --- Separando as \omterm{def:g3:first-electric-circuit:battery}{pilhas}.}
\begin{enumerate}
  \item Descreva a ligação correta do \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} para conferir uma
        \omterm{def:g3:first-electric-circuit:battery}{pilha} solta de \qty{1.5}{V} --- e por que circuito nenhum
        precisa ser montado antes.
  \item As leituras da auditoria, direto da gaveta: \qty{1.58}{V},
        \qty{1.32}{V}, \qty{0.9}{V}, \qty{0.1}{V}. Separe-as: nova,
        usável, cansada, morta.
  \item Uma bateria de nove \omterm{def:g8:voltage-voltmeter:voltage}{volts} marca \qty{8.9}{V}. Quanto
        contribui, mais ou menos, cada uma das seis \omterm{def:g3:first-electric-circuit:battery}{pilhas} internas
        dela?
  \item Por que uma \omterm{def:g3:first-electric-circuit:battery}{pilha} quase morta ainda mostra \emph{alguma}
        \omterm{def:g8:voltage-voltmeter:voltage}{tensão} embora já não consiga acender nada? (O empurrão dela
        sobrevive fraquinho; o que ela perdeu a \omterm{def:g2:pushes-and-pulls:force}{força} de impelir?)
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A estante de bugigangas.}
\begin{enumerate}[resume]
  \item A câmera quer duas \omterm{def:g3:first-electric-circuit:battery}{pilhas} uma atrás da outra. Que \omterm{def:g8:voltage-voltmeter:voltage}{tensão} ela
        espera, e o que o par de sobreviventes de \qty{1.32}{V} da
        questão 2 vai entregar?
  \item O rádio velho leva quatro \omterm{def:g3:first-electric-circuit:battery}{pilhas} uma atrás da outra e fica
        emburrado abaixo de \qty{5}{V}. Que \omterm{def:g3:first-electric-circuit:battery}{pilhas} da gaveta, se
        alguma, podem formar a tripulação dele?
  \item O irmãozinho enfia três \omterm{def:g3:first-electric-circuit:battery}{pilhas} na lanterna de duas ``para dar
        mais brilho'', entalando a terceira contra a mola. A lâmpada
        é de \qty{3}{V}. Preveja o espetáculo, na linguagem da \omterm{def:g8:voltage-voltmeter:nominal}{tensão
        nominal}.
  \item A placa do transformador da \omterm{ex:g6:circuits-and-safety:motor}{campainha} diz ``\qty{8}{V}''.
        Alguém propõe ligar a \omterm{ex:g6:circuits-and-safety:motor}{campainha} direto numa bateria de nove
        \omterm{def:g8:voltage-voltmeter:voltage}{volts} ``já que oito é quase nove''. Avalie: casamento justo
        ou assassinato lento?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- As leis da auditoria.}
\begin{enumerate}[resume]
  \item Na lanterna (lâmpada e \omterm{def:g3:first-electric-circuit:battery}{pilhas} \omterm{def:g4:series-circuits:series}{em série}, \omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto),
        o auditor mede: atravessado no \omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto,
        \qty{3.1}{V}; atravessado na lâmpada, \qty{0}{V}. Explique as
        duas leituras e preveja as duas depois do clique.
  \item Atravessado em qualquer trecho da fiação da lanterna o
        aparelho diz \qty{0}{V}, acesa ou apagada. O que isso declara
        sobre um bom fio, na linguagem do empurrão?
  \item A tomada \emph{não} é auditada, por lei permanente da
        família. Reformule a lei com o número deste capítulo e a
        frase aritmética por trás dela.
  \item Protocole o relatório da auditoria: três frases --- o que um
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} é e o que ele nunca é; o que a \omterm{def:g8:voltage-voltmeter:nominal}{tensão nominal} de
        uma etiqueta promete e ameaça; e onde, em qualquer circuito
        parado, a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} espera.
\end{enumerate}
\end{problem}
