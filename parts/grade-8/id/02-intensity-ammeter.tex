\chapter{Kuat Arus dan Amperemeter}\label{ch:g8:intensity-ammeter}

``Arus kuat, arus lemah'' --- kita sudah bertahun-tahun berbicara
seperti ini, sambil menakar terang dan cengkeraman menurut perasaan.
Perasaan dipensiunkan hari ini. Kekuatan arusnya mendapat sebuah nama,
sebuah \omterm{def:g6:measurement-in-science:measuring}{satuan}, dan sebuah alat; dan hal pertama yang dibuktikan
alatnya adalah sebuah hukum yang sudah lama kamu curigai tetapi tidak
pernah sekali pun kamu ukur.

\section{Kuat arus dan satuannya}

\begin{definition}[Kuat arus]\label{def:g8:intensity-ammeter:intensity}
\emph{Kuat arus}\index{kuat arus} $I$ sebuah \omterm{def:g7:circuits-series-parallel:current}{arus listrik} \omterm{def:g6:measurement-in-science:measuring}{mengukur}
seberapa kuat arak-arakannya: berapa banyak muatan listrik yang
berarak melewati sebuah titik rangkaiannya tiap sekon --- laju aliran
arusnya, sebagaimana laju sebuah sungai berupa air-tiap-sekon.
\omterm{def:g6:measurement-in-science:measuring}{Satuannya} adalah \emph{ampere}\index{ampere} (\unit{A}), yang
menghormati seorang perintis besar hukum kelistrikan; rangkaian
sehari-hari sering berbicara dalam \emph{miliampere}:
$\qty{1}{mA} = \qty{0.001}{A}$, seperseribu.
\end{definition}

\begin{example}[Ampere yang patut dikenali sekilas]\label{ex:g8:intensity-ammeter:values}
Sebuah lampu penunjuk kecil menyeruput kira-kira \qty{0.02}{A} ---
\qty{20}{mA}; sebuah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} senter, sekitar \qty{0.3}{A}; lampu besar
pada lampu depan mobil, \qty{5}{A}; sebuah cerek yang meraung penuh,
\qty{10}{A}; mesin mobil yang sedang dinyalakan meneguk \qty{100}{A}
dari akinya; sebuah sambaran petir, selama semilisekonnya, menghantam
puluhan ribu \omterm{def:g8:intensity-ammeter:intensity}{ampere} menuruni salurannya. Dan skala bahayanya membuat
kita sadar: arus beberapa perseratus \omterm{def:g8:intensity-ammeter:intensity}{ampere} yang menyeberangi tubuh
manusia sudah dapat mematikan --- rasa hormat mulai jauh di bawah satu
\omterm{def:g8:intensity-ammeter:intensity}{ampere}.
\end{example}

\begin{remark}[Orangnya dan satuannya]\label{rem:g8:intensity-ammeter:person}
Kalau ditulis lengkap, \omterm{def:g6:measurement-in-science:measuring}{satuannya} berhuruf kecil --- ``arus sebesar dua
\omterm{def:g8:intensity-ammeter:intensity}{ampere}'' --- sedangkan kata yang berhuruf besar menamai orangnya,
seperti pada setiap \omterm{def:g6:measurement-in-science:measuring}{satuan} yang menghormati seorang ilmuwan. Lambang
\unit{A} tetap huruf besar. Kamu akan bertemu perjanjian ini lagi
sebentar lagi, dua kali.
\end{remark}

\section{Amperemeternya}

\begin{definition}[Amperemeter]\label{def:g8:intensity-ammeter:ammeter}
Sebuah \emph{amperemeter}\index{amperemeter} \omterm{def:g6:measurement-in-science:measuring}{mengukur} \omterm{def:g8:intensity-ammeter:intensity}{kuat arus}. Ia
menghitung arak-arakan yang lewat \emph{menembus dirinya sendiri} ---
sehingga ia harus disisipkan \emph{secara seri}, ke dalam lingkar yang
diukurnya itu sendiri, seperti gardu tol di sebuah jalan. Karena
dibangun untuk seminimal mungkin mengganggu arak-arakannya, ia hampir
sama sekali tidak melawan arusnya --- dan itu pula sebabnya
amperemeter yang salah dihubungkan langsung melintang pada sebuah
\omterm{def:g3:first-electric-circuit:battery}{baterai} adalah \omterm{def:g7:short-circuits-safety:device}{korsleting} berpapan ukur: cara klasik untuk merusak
sebuah amperemeter.
\end{definition}

\begin{method}[Mengukur sebuah kuat arus]\label{met:g8:intensity-ammeter:measure}
\begin{enumerate}
  \item matikan rangkaiannya; \emph{putuslah} lingkarnya di tempat
        arusnya hendak dihitung;
  \item sisipkan \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya} ke dalam celahnya: arus masuk lewat
        kutub \unit{A}-nya, keluar lewat kutub \emph{bersama}-nya,
        supaya arah perjanjiannya memasuki pintu depan alatnya;
  \item pada alat berjangkauan banyak, mulailah pada jangkauan yang
        \emph{terbesar}, lalu turunlah setingkat demi setingkat
        sampai bacaannya duduk dengan nyaman pada papannya ---
        sehingga alatnya terlindung dari kejutan;
  \item tutuplah \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}, bacalah --- lalu tulislah \omterm{def:g6:measurement-in-science:measuring}{satuannya}.
\end{enumerate}
Sambungan yang terbalik pada alat berjarum menghantamkan jarumnya ke
penahannya; alat digital sekadar menunjukkan tanda minus, sehingga
memaafkan para murid tiap hari.
\end{method}

\section{Hukum terukur yang pertama}

\begin{proposition}[Satu lingkar, satu kuat arus]\label{prop:g8:intensity-ammeter:sameI}
Pada \omterm{def:g4:series-circuits:series}{rangkaian seri}, kuat arusnya sama pada setiap titik lingkarnya.
Pindahkan \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya} ke mana pun --- sebelum lampunya, sesudahnya,
di samping \omterm{def:g3:first-electric-circuit:battery}{baterainya}: bacaannya tidak berubah. Apa yang disarankan
lima tahun \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang sama terangnya, kini dinyatakan alatnya dalam
bilangan: tidak ada yang terpakai habis di tengah jalan; satu jalan,
satu arak-arakan, satu $I$.
\end{proposition}

\begin{example}[Membaca cetakan halus hukumnya]\label{ex:g8:intensity-ammeter:fineprint}
Hukumnya berkata \emph{di sepanjang satu lingkar}, bukan \emph{apa pun
yang kamu lakukan terhadap lingkarnya}. Tambahkan lampu kedua secara
seri dan arak-arakannya melemah di mana-mana sekaligus --- mungkin
\qty{0.30}{A} turun menjadi \qty{0.17}{A} --- tetapi ia melemah
\emph{dengan sama} di mana-mana: ketiga gardu tolnya sepakat pula
tentang bilangan yang baru. Satu lingkar, satu \omterm{def:g8:intensity-ammeter:intensity}{kuat arus} --- berapa pun
\omterm{def:g8:intensity-ammeter:intensity}{kuat arus} itu saat ini.
\end{example}

\begin{example}[Sekilas pandang ke percabangannya]\label{ex:g8:intensity-ammeter:junction}
Pasanglah gardu tol pada sepasang \omterm{def:g5:series-parallel-circuits:parallel}{cabang} paralel: jalan utamanya
\qty{0.50}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{cabang} satu \qty{0.30}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{cabang} dua
\qty{0.20}{A}. Jumlahnya yang mengisyaratkan sesuatu itu bukan
kebetulan --- arak-arakan yang masuk ke sebuah percabangan dan yang
keluar darinya harus seimbang, karena muatan tidak diciptakan dan
tidak dimusnahkan pada sebuah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} jalan. Hukum penuhnya, beserta
pasangannya untuk tegangan, bersidang pada bab sesudah yang berikutnya;
alat ukurmu sudah bertemu dengannya.
\end{example}

\begin{remark}[Multimeternya]\label{rem:g8:intensity-ammeter:multimeter}
Batu bata kuning milik bengkelnya --- \emph{multimeter} --- melipat
\omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} dan beberapa alat lain menjadi satu kotak dengan sebuah
papan pemilih. Papannya pada \unit{A} atau \unit{mA}, kabelnya pada
lubang yang bertanda, dan ia pun menjadi amperemetermu, lengkap dengan
aturan serinya. Dua bab buku ini sudah berdiam di dalamnya; yang
ketiga pindah masuk pada bab berikutnya --- dengan, awas, aturan
pemasangan kabel yang \emph{berlawanan}.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:1}
Apa yang diukur \omterm{def:g8:intensity-ammeter:intensity}{kuat arus}, dalam bahasa sungai? Berikan \omterm{def:g6:measurement-in-science:measuring}{satuan} dan
lambangnya, lalu ubahlah \qty{250}{mA} menjadi \omterm{def:g8:intensity-ammeter:intensity}{ampere}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:2}
Pasangkan kuat arusnya dengan adegannya: \qty{0.02}{A}; \
\qty{0.3}{A}; \ \qty{10}{A}; \ \qty{100}{A} --- cerek, lampu penunjuk,
mesin yang sedang dinyalakan, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} senter.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:3}
Mengapa sebuah \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} harus disisipkan secara seri? Apa yang
dibangun padanya supaya hampir sama sekali tidak dilawannya, dan
kecelakaan apa yang dimungkinkan hal itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:4}
Sebutkan keempat langkah pengukuran \omterm{def:g8:intensity-ammeter:intensity}{kuat arus} yang sepatutnya. Mengapa
mulai pada jangkauan yang terbesar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:5}
Nyatakan hukum satu-lingkar-satu-kuat-arusnya. Pengamatan lama tentang
terang yang sama yang mana yang diubahnya menjadi bilangan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:6}
Sebuah \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} sebelum sebuah \omterm{ex:g6:circuits-and-safety:motor}{motor} seri membaca \qty{0.45}{A}.
Berapa yang akan dibacanya sesudah \omterm{ex:g6:circuits-and-safety:motor}{motornya}? Dan di samping \omterm{def:g3:first-electric-circuit:battery}{baterainya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:7}
Sebuah lampu kedua bergabung ke sebuah lingkar seri dan \qty{0.30}{A}
milik \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya} menjadi \qty{0.17}{A}. Apakah hukumnya gagal?
Jelaskan cetakan halusnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:intensity-ammeter:8}
Sebuah \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} digital membaca $-0.25$ \unit{A}. Apa yang telah
dilakukan muridnya, dan seberapa \omterm{def:g4:weight-and-mass:weight}{berat} itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:9}
Jalan utamanya \qty{0.60}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{cabang} satu \qty{0.35}{A}. Berapa yang
dibawa \omterm{def:g5:series-parallel-circuits:parallel}{cabang} dua, dan menurut penalaran yang mana? Berapa yang dibaca
gardu tol pada percabangan yang kedua?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:10}
Seorang murid memasang \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya} \emph{secara paralel} dengan
lampunya ``untuk membandingkan keduanya''. Ramalkan kedua akibatnya
--- bagi lampunya, dan bagi alatnya --- lalu sebutkan penjahat yang
tanpa sengaja dibangun muridnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:intensity-ammeter:11}
Dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} serupa yang terpasang paralel masing-masing membawa
\qty{0.30}{A}. Berapa yang dibawa jalan utama \omterm{def:g3:first-electric-circuit:battery}{baterainya}? Sebuah
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} ketiga yang serupa bergabung secara paralel: ramalkan beban
baru jalan utamanya, lalu sebutkan apa yang terkuras lebih cepat.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:intensity-ammeter:12}
Rancanglah rencana pengukuran yang lengkap bagi sebuah \omterm{def:g5:series-parallel-circuits:parallel}{rangkaian
paralel} bercabang dua (lampu dan \omterm{ex:g6:circuits-and-safety:motor}{motor}): berapa banyak kedudukan
\omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} yang diperlukan untuk mengetahui \emph{setiap} arus di
dalam rangkaiannya; berapa jumlah pengukuran terkecil yang mencukupi
(satu di antaranya dapat disimpulkan --- menurut keseimbangan yang
mana?); dan urutan pengerjaan yang tidak pernah meninggalkan sebuah
alat terpasang secara berbahaya.
\end{exercise}

\section{Soal: Meja Metrologi}

\begin{problem}[{Soal akhir pekan --- hari kelulusan di bengkel
elektronika; empat pos, satu amperemeter tiap pos; surat izin sang
murid}]\label{pb:g8:intensity-ammeter:1}
Untuk memperoleh hak memakai bengkelnya, tiap murid berkeliling empat
pos pengukuran. Bawalah babnya.

\medskip\noindent\textbf{Bagian I --- Pos satu: lingkar yang
sederhana.}
\omterm{def:g3:first-electric-circuit:battery}{Baterai}, \omterm{ex:g3:first-electric-circuit:switch}{sakelar}, satu lampu.
\begin{enumerate}
  \item Gambarkan, langkah demi langkah, penyisipan \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya}
        untuk \omterm{def:g6:measurement-in-science:measuring}{mengukur} arus lampunya.
  \item Bacaannya: \qty{280}{mA}. Nyatakan dalam \omterm{def:g8:intensity-ammeter:intensity}{ampere}.
  \item Pengujinya memindahkan alatmu ke sisi \omterm{def:g3:first-electric-circuit:battery}{baterainya} yang lain.
        Ramalkan bacaannya, sambil menyebut hukumnya.
  \item ``Dan kalau aku punya dua \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} di dalam lingkar ini
        sekaligus?'' Di mana keduanya boleh duduk, dan apa yang akan
        ditunjukkannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pos dua: pasangan
seri.}
Dua lampu yang berbeda, A lebih terang daripada B, \omterm{def:g4:series-circuits:series}{terangkai seri}.
\begin{enumerate}[resume]
  \item Seorang murid meramalkan: ``A yang lebih terang membawa arus
        lebih banyak daripada B yang redup.'' Ujilah ramalannya dengan
        hukumnya --- lalu rukunkan arus yang sama itu dengan terang
        yang tidak sama (apa yang harus berbeda di antara lampunya?).
  \item Alatnya membaca \qty{150}{mA} di antara A dan B. Tuliskan
        setiap arus lain di dalam lingkarnya.
  \item Lampu B dikorsletingkan oleh sepotong kabel berjepit.
        Ramalkan bacaan yang baru dan perilaku kedua lampunya.
  \item Pengujinya meminta satu kalimat tentang mengapa bengkelnya
        melarang kabel berjepit melintang pada \emph{\omterm{def:g3:first-electric-circuit:battery}{baterainya}}
        sementara mengizinkan peragaan yang melintang pada B.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pos tiga:
percabangannya.}
Sebuah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} lampu dan sebuah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} \omterm{ex:g6:circuits-and-safety:motor}{motor} terpasang paralel.
\begin{enumerate}[resume]
  \item Jalan utamanya: \qty{0.52}{A}; \omterm{def:g5:series-parallel-circuits:parallel}{cabang} lampunya: \qty{0.30}{A}.
        Berapa arus \omterm{def:g5:series-parallel-circuits:parallel}{cabang} \omterm{ex:g6:circuits-and-safety:motor}{motornya}, menurut keseimbangannya?
  \item \omterm{ex:g6:circuits-and-safety:motor}{Motornya} macet (masih terhubung, tidak berputar) dan
        \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} memanjat ke \qty{0.40}{A}. Dengan menganggap \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
        lampunya menjaga \qty{0.30}{A}-nya, berapa yang dibawa jalan
        utamanya sekarang --- dan pengawal bengkel yang mana yang
        sedang mendengarkan?
  \item Lampunya dilepas dari fittingnya. Ramalkan ketiga bacaan
        gardu tolnya.
  \item Mengapa \omterm{def:g3:first-electric-circuit:battery}{baterainya} terkuras paling cepat ketika kedua
        \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sehat --- di mana kemurahan hatinya dibayar?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Surat izinnya.}
\begin{enumerate}[resume]
  \item Tulislah janji sang murid: tiga baris --- di mana sebuah
        \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} berada dan tidak pernah boleh berada; apa yang
        dijamin satu lingkar; apa yang diseimbangkan sebuah
        percabangan.
\end{enumerate}
\end{problem}
