\chapter{Lensa dan Bayangan}\label{ch:g8:lenses-images}

Kamera lubang jarum membeli \omterm{prop:g5:mirrors-reflection:image}{bayangan} kecilnya yang tajam dengan harga
yang kejam: hampir tidak ada \omterm{def:g1:light-and-shadows:source}{cahaya}. Namun matamu sendiri melukiskan
dunia yang terang dan tajam pada dinding belakangnya sepanjang hari,
dan tidak ada lubang jarum yang terlihat. Rahasia matanya adalah
sekeping bahan \omterm{def:g7:light-sources-propagation:coats}{bening} yang melengkung yang mengumpulkan seluruh \omterm{def:g6:light-propagation:ray}{berkas}

kacamata, kaca pembesar, dan bab ini.

\section{Dua keluarga lensa}

\begin{definition}[Lensa cembung dan lensa cekung]\label{def:g8:lenses-images:families}
Sebuah \emph{lensa}\index{lensa} adalah sekeping bahan \omterm{def:g7:light-sources-propagation:coats}{bening} ---
kaca atau plastik jernih --- dengan wajah yang melengkung, yang
dibangun untuk membelokkan \omterm{def:g1:light-and-shadows:source}{cahaya} dengan terkendali (\omterm{def:g1:light-and-shadows:source}{cahaya} membelok
ketika ia menyeberang masuk ke dalam kaca; hukum pembelokan itu yang
persis adalah harta jilid Sekolah Menengah). Dua keluarganya:
\begin{enumerate}
  \item \emph{lensa cembung}\index{lensa cembung} --- lebih tebal di
        tengah daripada di tepinya --- membelokkan sinar sebuah \omterm{def:g6:light-propagation:ray}{berkas}
        \emph{ke arah} satu sama lain;
  \item \emph{lensa cekung}\index{lensa cekung} --- lebih tipis di
        tengah --- merenggangkannya terpisah.
\end{enumerate}
Jalankan ujung jari menyeberangi lensa kacamata: tonjolan di tengah
atau cekungan di tengah memilah setiap lensa di Bumi ke dalam
keluarganya.
\end{definition}

\begin{definition}[Titik fokus dan jarak fokus]\label{def:g8:lenses-images:focus}
Kirimkan seberkas sinar sejajar --- \omterm{def:g1:light-and-shadows:source}{cahaya} matahari sudah cukup ---
menembus sebuah \omterm{def:g8:lenses-images:families}{lensa cembung}: sinar yang membelok itu semuanya
berpotongan pada satu titik di belakangnya, yaitu \emph{titik
fokus}\index{titik fokus}. Jarak dari \omterm{def:g8:lenses-images:families}{lensanya} ke titik fokusnya
adalah \emph{jarak fokus}\index{jarak fokus} $f$: satu bilangan yang
menjadi ciri \omterm{def:g8:lenses-images:families}{lensanya}. \omterm{def:g8:lenses-images:families}{Lensa} gemuk yang melengkung kuat membelokkan
dengan keras --- $f$ yang pendek; \omterm{def:g8:lenses-images:families}{lensa} yang melengkung lembut
membelokkan sedikit --- $f$ yang panjang.
\end{definition}

\begin{method}[Mengukur sebuah jarak fokus]\label{met:g8:lenses-images:measure}
Untuk \omterm{def:g8:lenses-images:families}{lensa cembung} mana pun, semenit dan sebuah jendela yang jauh:
\begin{enumerate}
  \item hadapilah sebuah dinding yang berseberangan dengan pemandangan
        yang terang dan \emph{jauh} --- jendelanya dan dunia di
        baliknya;
  \item peganglah \omterm{def:g8:lenses-images:families}{lensanya} dekat dindingnya lalu tariklah ia menjauh
        perlahan sampai pemandangannya menjeplak menjadi gambar kecil
        yang tajam pada dindingnya (terbalik --- sebagaimana sudah
        diharapkan sekarang);
  \item ukurlah jarak \omterm{def:g8:lenses-images:families}{lensa} ke dindingnya: untuk pemandangan yang
        jauh, jarak itu \emph{adalah} jarak fokusnya --- sinar dari kejauhan tiba
        praktis sejajar, dan sinar yang sejajar berkumpul di fokusnya.
\end{enumerate}
Sebuah kaca pembesar terukur sekitar \qty{10}{cm}; \omterm{def:g8:lenses-images:families}{lensa} kacamata,
puluhan \omterm{def:g2:measuring-length:units}{sentimeter}; \omterm{def:g8:lenses-images:families}{lensa} kamera yang pendek, beberapa \omterm{def:g2:measuring-length:units}{sentimeter}.
\end{method}

\begin{remark}[Kaca pembakar]\label{rem:g8:lenses-images:burning}
Pada \omterm{def:g8:lenses-images:focus}{titik fokus} sebuah \omterm{def:g8:lenses-images:families}{lensa cembung}, \omterm{def:g1:light-and-shadows:source}{cahaya} matahari selebar seluruh
berkasnya berdesak ke dalam satu titik yang membara --- cukup terpusat
untuk menghanguskan kertas lalu menyalakan api dalam hitungan sekon.
Perlakukanlah ia sebagaimana adanya: asas kompor tenaga surya dalam
bentuk sebesar saku. Jangan pernah memfokuskan \omterm{def:g1:light-and-shadows:source}{cahaya} matahari dekat
benda apa pun yang mudah terbakar, jangan pernah ke arah kulit atau
pakaian --- dan, dua kali lipat hukum yang lama, jangan pernah
memandang Matahari \emph{menembus} \omterm{def:g8:lenses-images:families}{lensa} apa pun: pemfokusan mata
sendiri ditambah pemfokusan \omterm{def:g8:lenses-images:families}{lensanya} membuat luka bakar yang seketika,
tanpa rasa sakit, dan tetap selamanya.
\end{remark}

\section{Lensa sebagai pembuat bayangan}

\begin{proposition}[Bayangan nyata]\label{prop:g8:lenses-images:realimage}
Sebuah \omterm{def:g8:lenses-images:families}{lensa cembung} mengumpulkan sinar yang dibaurkan tiap titik
sebuah benda lalu mengantarkannya ke titik yang berpadanan di sisi
yang lain: titik demi titik, ia merakit gambar yang sungguhan ---
sebuah \emph{bayangan nyata}\index{bayangan nyata} --- yang dapat
ditangkap pada sebuah layar. \omterm{prop:g5:mirrors-reflection:image}{Bayangannya} terbalik, seperti \omterm{prop:g5:mirrors-reflection:image}{bayangan}
lubang jarumnya, tetapi terang: \omterm{def:g8:lenses-images:families}{lensanya} memanen \omterm{def:g1:light-and-shadows:source}{cahaya} selebar
seluruh \omterm{def:g8:lenses-images:families}{lensa} tiap titiknya, sedangkan lubang jarumnya melewatkan satu
helai saja. Aturan praktis geometrinya: benda yang jauh membayang
kecil dan dekat fokusnya; selagi bendanya mendekati \omterm{def:g8:lenses-images:families}{lensanya},
\omterm{prop:g5:mirrors-reflection:image}{bayangannya} mundur lalu \omterm{def:g7:moon-phases:phases}{membesar}.
\end{proposition}

\begin{example}[Kamera dan mata, satu tata bangun]\label{ex:g8:lenses-images:cameraeye}
Sebuah kamera adalah pembuat \omterm{prop:g5:mirrors-reflection:image}{bayangan} yang dikotakkan: \omterm{def:g8:lenses-images:families}{lensa cembung}
di depan, layar peka \omterm{def:g1:light-and-shadows:source}{cahaya} di belakang, \omterm{prop:g5:mirrors-reflection:image}{bayangan} terbentuk terbalik
padanya (perangkat lunaknya dengan sopan memutar berkasnya). Matamu
adalah mesin yang sama dalam jaringan yang hidup: \omterm{def:g8:lenses-images:families}{lensa cembung} di
belakang manik matanya, \emph{retina} sebagai layar di bagian
belakangnya, \omterm{prop:g5:mirrors-reflection:image}{bayangannya} terbalik --- otaknya yang membalikkannya,
sebagaimana dicatat ketika lubang jarumnya pertama kali mengajari
kita. Pemfokusannya berbeda dengan indah: kameranya menggerakkan
\omterm{def:g8:lenses-images:families}{lensanya}; \omterm{def:g8:lenses-images:families}{lensa} matamu yang lunak \emph{berubah bentuk}, ditarik lebih
gemuk untuk benda yang dekat, dikendurkan lebih tipis untuk yang jauh
--- rasakanlah upayanya sesudah sejam membaca dari dekat.
\end{example}

\begin{example}[Kacamata]\label{ex:g8:lenses-images:spectacles}
Ketika pemfokusan mata sendiri kurang panjang, sebuah \omterm{def:g8:lenses-images:families}{lensa} di
depannya menambalnya. Mata yang rabun jauh memfokuskan terlalu kuat
--- pemandangan yang jauh terakit \emph{sebelum} retinanya lalu tiba
kabur: sebuah \omterm{def:g8:lenses-images:families}{lensa} kacamata yang \emph{cekung} melepaskan
pembelokannya tepat secukupnya. Mata yang rabun dekat (dan kebanyakan
mata yang lebih tua) memfokuskan terlalu lemah untuk pekerjaan dekat:
sebuah \omterm{def:g8:lenses-images:families}{lensa} \emph{cembung} meminjamkan kekuatan yang kurang itu ---
kacamata baca adalah sepupu kecil kaca pembesar yang jujur. Dua
keluarga \omterm{def:g8:lenses-images:families}{lensa}, dua keluarga kekaburan.
\end{example}

\section{Kaca pembesarnya}

\begin{omfigure}

{\small Kaca pembesarnya sedang bekerja: kalau dipegang lebih dekat daripada jarak fokusnya, \omterm{def:g8:lenses-images:families}{lensanya} menunjukkan cetakannya tegak dan diperbesar.}
\end{omfigure}

\begin{example}[Di dalam jarak fokusnya]\label{ex:g8:lenses-images:magnifier}
Peganglah sebuah \omterm{def:g8:lenses-images:families}{lensa cembung} \emph{lebih dekat} ke sebuah perangko
daripada jarak fokusnya, dan gambarnya berganti watak: tidak ada
\omterm{prop:g5:mirrors-reflection:image}{bayangan} yang mendarat pada layar mana pun --- sebaliknya, sambil
memandang menembus \omterm{def:g8:lenses-images:families}{lensanya}, kamu melihat perangkonya tegak, di
tempatnya, dan diperbesar. \omterm{def:g8:lenses-images:families}{Lensanya} membelokkan sinar perangkonya
sehingga mereka mencapai mata \emph{seakan-akan} dibaurkan oleh
perangko yang lebih megah di balik kacanya --- sebuah ilusi sesuai
pesanan, sepupu dunia hantu milik \omterm{def:g5:mirrors-reflection:reflection}{cerminnya}. Tukang arloji, pengumpul
perangko, dan dirimu sendiri pada masa lalu dengan kaca pembesar masa
kecilmu semuanya sudah memakainya; geometri sinar muslihat ini
selengkapnya adalah permata jilid Sekolah Menengah.
\end{example}

\begin{remark}[Alat di atas alat]\label{rem:g8:lenses-images:instruments}
Tumpuklah \omterm{def:g8:lenses-images:families}{lensanya} dan dunianya pun terbuka. Dua \omterm{def:g8:lenses-images:families}{lensa cembung} membuat
sebuah mikroskop: yang satu melemparkan \omterm{prop:g8:lenses-images:realimage}{bayangan nyata} yang besar dari
setetes cairan, yang kedua memperbesar \omterm{prop:g5:mirrors-reflection:image}{bayangan} itu --- para pembela
\omterm{def:g7:states-of-matter:molecule}{molekul} memperoleh sekutu langsungnya yang pertama dengan cara ini.
Sebuah \omterm{def:g8:lenses-images:families}{lensa} yang besar (atau \omterm{def:g5:mirrors-reflection:reflection}{cermin} yang melengkung) ditambah sebuah
kaca pembesar membuat sebuah teleskop: sinar sejajar yang samar milik
galaksi yang jauh dikumpulkan menjadi \omterm{prop:g8:lenses-images:realimage}{bayangan nyata}, lalu diperbesar
ke dalam mata yang terkesima. Setiap alat optik peradaban adalah kedua
gagasan bab ini --- kumpulkan, lalu perbesar --- dalam berbagai susunan.
\end{remark}

\section{Soal: Meja Sang Ahli Kacamata}

\begin{problem}[{Soal akhir pekan --- menjadi murid ahli kacamata
kotanya; jarak fokus, pemasangan, dan satu teleskop yang
retak}]\label{pb:g8:lenses-images:1}
Sehari di tempat ahli kacamatanya: \omterm{def:g8:lenses-images:families}{lensa} yang harus dipilah, kacamata
yang harus dipadankan, dan sebuah teleskop pelanggan yang berkeping.

\medskip\noindent\textbf{Bagian I --- Memilah nampannya.}
Sebuah laci yang tumpah telah mengacaukan persediaannya.
\begin{enumerate}
  \item Usulkan dua uji yang cepat --- satu dengan sentuhan, satu
        dengan jendela dan dinding --- untuk memilah \omterm{def:g8:lenses-images:families}{lensa cembung}
        dari \omterm{def:g8:lenses-images:families}{lensa cekung}.
  \item \omterm{def:g8:lenses-images:families}{Lensa} A melemparkan \omterm{prop:g5:mirrors-reflection:image}{bayangan} jalanan yang tajam ke dindingnya
        pada \qty{25}{cm}. Berapa jarak fokusnya dan apa keluarganya?
  \item \omterm{def:g8:lenses-images:families}{Lensa} B menolak melemparkan \omterm{prop:g5:mirrors-reflection:image}{bayangan} apa pun ke dindingnya
        pada jarak berapa pun, dan benda yang dipandang menembusnya
        tampak lebih kecil. Apa keluarganya?
  \item \omterm{def:g8:lenses-images:families}{Lensa} C dan D keduanya cembung; C membuat \omterm{prop:g5:mirrors-reflection:image}{bayangan} jendelanya
        pada \qty{10}{cm}, D pada \qty{40}{cm}. Yang mana yang lebih
        kuat melengkung dan lebih keras membelokkan?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pemasangannya.}
\begin{enumerate}[resume]
  \item Seorang pelanggan melihat rambu jalan yang jauh kabur tetapi
        membaca daftar menu tanpa susah payah. Buatlah diagnosis
        matanya lalu resepkan keluarganya.
  \item Yang lain memegang korannya sepanjang lengan lalu mengeluh
        bahwa lengannya ``terlalu pendek''. Buatlah diagnosisnya lalu
        resepkan.
  \item Ahli kacamatanya memperingatkan pelanggan yang kedua:
        ``kacamata baca barumu akan mengaburkan pegunungan.'' Mengapa
        kacamata baca hanya untuk halaman yang dekat?
  \item Seorang anak bertanya mengapa \omterm{def:g8:lenses-images:families}{lensa} coba milik ahli
        kacamatanya membuat papan uji matanya berenang padahal
        matanya sendiri ``baik-baik saja''. Apa yang sedang dilakukan
        \omterm{def:g8:lenses-images:families}{lensa} pembetul terhadap mata yang sudah terfokus baik?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Teleskop yang
retak.}
Seorang pelanggan membawa teleskop pengamat \omterm{def:g4:solar-system:star}{bintang} dengan \omterm{def:g8:lenses-images:families}{lensa}
depannya yang retak, lalu bertanya apakah kaca pembesar cadangan di
dalam kotaknya dapat menggantikannya.
\begin{enumerate}[resume]
  \item Sebutkan tugas kedua \omterm{def:g8:lenses-images:families}{lensanya} pada alat yang sehat: apa yang
        diterima \omterm{def:g8:lenses-images:families}{lensa} depan yang besar itu dari sebuah \omterm{def:g4:solar-system:star}{bintang} dan
        apa yang dibuatnya, lalu apa yang dilakukan \omterm{def:g8:lenses-images:families}{lensa} mata yang
        kecil terhadapnya?
  \item \omterm{def:g8:lenses-images:families}{Lensa} depannya selebar sebuah lepek; \omterm{def:g8:lenses-images:families}{lensa} matanya sekecil
        sekeping koin. Mengapa \emph{lebar} \omterm{def:g8:lenses-images:families}{lensa} depannya begitu
        berarti bagi \omterm{def:g4:solar-system:star}{bintang} yang samar?
  \item Kaca pembesar cadangannya melengkung kuat tetapi sekecil
        koin. Nilailah penggantiannya: apa yang akan diperoleh dan
        hilang di bagian depan alatnya?
  \item Tutuplah harinya dengan pengakuan iman satu baris milik ahli
        kacamatanya tentang setiap alat di raknya --- kacamata, kaca
        pembesar, kamera, teleskop --- dan ringkasan dua kata kerja
        tentang apa yang dilakukan \omterm{def:g8:lenses-images:families}{lensa} bagi peradaban.
\end{enumerate}
\end{problem}
