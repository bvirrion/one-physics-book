\chapter{Hambatan dan Hukum Ohm}\label{ch:g8:ohms-law}

Satu pertanyaan telah mengekori seluruh kisah kelistrikannya: sebagian
komponen membiarkan arak-arakannya menyerbu, yang lain mencekiknya
menjadi rembesan --- lampu membelah \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} secara tidak sama, kawat
tipis menghangat, yang tebal tetap sejuk. Perlawanannya sendiri kini
mendapat sebuah nama, sebuah \omterm{def:g6:measurement-in-science:measuring}{satuan}, dan --- sebagai mahkota empat
tahun rangkaian --- hukum yang paling berguna di dalam buku ini: tiga
huruf yang mengikat dorongan, arak-arakan, dan perlawanan menjadi satu
baris aljabar.

\section{Hambatan}

\begin{definition}[Hambatan]\label{def:g8:ohms-law:resistance}
\emph{Hambatan}\index{hambatan} $R$ sebuah komponen \omterm{def:g6:measurement-in-science:measuring}{mengukur} seberapa
kuat ia melawan \omterm{def:g7:circuits-series-parallel:current}{arus listrik}: pada \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} tertentu, makin besar
hambatannya, makin lemah arak-arakan yang diloloskannya. \omterm{def:g6:measurement-in-science:measuring}{Satuannya}
adalah \emph{ohm}\index{ohm} (lambangnya $\Omega$, huruf besar Yunani
omega), yang menghormati guru sekolah sekaligus fisikawan yang
menemukan hukum bab ini dengan kawat buatan sendiri dan kesabaran yang
kepahlawanan.
\end{definition}

\begin{definition}[Resistor]\label{def:g8:ohms-law:resistor}
Sebuah \emph{resistor}\index{resistor} adalah komponen yang dibuat
supaya mempunyai \omterm{def:g8:ohms-law:resistance}{hambatan} yang terpilih dan mantap --- sebuah rintangan
bertera, yang dijual mulai dari pecahan \omterm{def:g8:ohms-law:resistance}{ohm} sampai jutaan \omterm{def:g8:ohms-law:resistance}{ohm}.
Rangkaian memakai resistor untuk menetapkan arus dengan sengaja:
menjinakkan arak-arakan yang akan membakar bagian yang halus,
meredupkan, membagi, menyetel. Pita warnanya yang berjalur mengeja
nilainya; lambangnya berupa persegi panjang polos.
\end{definition}

\begin{example}[Hambatan di sekelilingmu]\label{ex:g8:ohms-law:values}
Semeter kawat tembaga yang tebal: perseratusan \omterm{def:g8:ohms-law:resistance}{ohm} --- jalan yang
datar. \omterm{def:g3:first-electric-circuit:bulb}{Bohlam} senter yang berpijar: beberapa \omterm{def:g8:ohms-law:resistance}{ohm}. \omterm{ex:g6:circuits-and-safety:motor}{Motor} listrik kecil:
beberapa \omterm{def:g8:ohms-law:resistance}{ohm}. Kumparan pemanas cerek: kira-kira $25\,\Omega$. Kulit
manusia yang kering, dari tangan ke tangan: puluhan ribu \omterm{def:g8:ohms-law:resistance}{ohm} --- kalau
basah, jauh lebih kecil sampai membawa malapetaka: aturan tangan
lembap yang lama itu adalah pernyataan tentang \omterm{def:g8:ohms-law:resistance}{hambatan}. \omterm{ex:g3:first-electric-circuit:switch}{Sakelar} yang
terbuka: \omterm{def:g8:ohms-law:resistance}{hambatan} di luar segala pengukuran --- rintangan yang tak
terhingga.
\end{example}

\section{Percobaannya}

\begin{method}[Mengukur potret $U$--$I$ sebuah komponen]\label{met:g8:ohms-law:portrait}
Satu \omterm{def:g8:ohms-law:resistor}{resistor}, satu sumber yang dapat disetel (atau barisan sel yang
bertambah panjang), dan kedua alat ukurnya:
\begin{enumerate}
  \item pasanglah sumbernya, \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya}, dan \omterm{def:g8:ohms-law:resistor}{resistornya} secara
        seri; jepitkan \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeternya} melintang pada \omterm{def:g8:ohms-law:resistor}{resistornya};
  \item aturlah \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang terkecil; catatlah pasangannya ($U$, $I$);
  \item naikkan \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} setingkat demi setingkat --- satu sel, dua,
        tiga --- sambil mencatat sepasang tiap kalinya;
  \item gambarkan titiknya, $I$ mendatar, $U$ ke atas, lalu lihatlah.
\end{enumerate}
\end{method}

\begin{example}[Pengakuan sebuah resistor]\label{ex:g8:ohms-law:confession}
Tabel sebuah sesi, untuk satu \omterm{def:g8:ohms-law:resistor}{resistor}:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
Gandakan dorongannya, gandalah arak-arakannya; tigakalikan, tiga kali
pula: $U$ dan $I$ \emph{sebanding}, dan setiap nisbah $U/I$
mengembalikan bilangan yang sama: $15$. Nisbah tetap itu bukan
kebetulan --- ia adalah \omterm{def:g8:ohms-law:resistance}{hambatan} \omterm{def:g8:ohms-law:resistor}{resistornya}, $R = 15\,\Omega$, yang
memperlihatkan dirinya pada setiap barisnya.
\end{example}

\begin{proposition}[Hukum Ohm]\label{prop:g8:ohms-law:law}
Untuk sebuah \omterm{def:g8:ohms-law:resistor}{resistor} pada \omterm{def:g3:temperature-thermometers:temperature}{suhu} yang mantap, \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang melintang
padanya dan arus yang menembusnya sebanding, dan tetapan
kesebandingannya adalah \omterm{def:g8:ohms-law:resistance}{hambatannya}:
\[
U = R \times I ,
\]
dengan $U$ dalam \omterm{def:g8:voltage-voltmeter:voltage}{volt}, $I$ dalam \omterm{def:g8:intensity-ammeter:intensity}{ampere}, $R$ dalam \omterm{def:g8:ohms-law:resistance}{ohm}. Disusun ulang
sesuai selera: $I = U/R$ (arus yang digerakkan sebuah dorongan
menembus sebuah rintangan) dan $R = U/I$ (rintangannya, yang dibongkar
kedoknya oleh sepasang bacaan yang jujur).
\end{proposition}

\end{omfigure}

\begin{example}[Hukumnya menghitung segalanya]\label{ex:g8:ohms-law:compute}
Begitu $R$ diketahui, hukumnya menjawab ke segala arah. Kumparan
cereknya, $25\,\Omega$ pada listrik rumah \qty{230}{V}:
$I = U/R = 230 \div 25 = \qty{9.2}{A}$ --- perkakas sepuluh \omterm{def:g8:intensity-ammeter:intensity}{ampere}
yang terjelaskan. Sebuah \omterm{def:g8:ohms-law:resistor}{resistor} harus membatasi arus sebuah lampu
yang halus sampai \qty{0.02}{A} dari \qty{4.5}{V}: $R = U/I = 4.5
\div 0.02 = 225\,\Omega$ --- pilihlah nilai baku berikutnya. Menembus
$60\,\Omega$ mengalir \qty{0.05}{A}: \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang melintang padanya
adalah $U = 60 \times 0.05 = \qty{3}{V}$. Satu hukum, tiga resep, dan
seluruh bengkelnya.
\end{example}

\begin{remark}[Tongkat penopang segitiganya]\label{rem:g8:ohms-law:triangle}
Sebagian pembelajar menuliskan $U$ di atas $R$ dan $I$ di dalam sebuah
segitiga kecil, lalu menutup huruf yang dicari untuk membaca resepnya.
Tidak berbahaya sebagai tongkat penopang --- tetapi aljabarmu kini
menyusun ulang $U = R I$ dengan jujur dalam satu baris, dan cara yang
jujur itu berlaku bagi setiap rumus yang akan pernah kamu temui,
bersegitiga atau tidak. Bersandarlah pada tongkatnya kalau memang
perlu tahun ini; berencanalah untuk berjalan.
\end{remark}

\section{Apa yang dijelaskan hambatan}

\begin{example}[Misteri lama pada satu tagihan]\label{ex:g8:ohms-law:mysteries}
Pembelahan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang tidak sama pada bab hukumnya: lampu seri
berbagi satu $I$, sehingga menurut $U = R I$ yang $R$-nya lebih besar
mengambil $U$ yang lebih besar --- turunan yang lebih besar milik
pekerja yang lebih berat, kini dapat dihitung. \omterm{def:g5:series-parallel-circuits:parallel}{Cabang} yang tercekik
pada bab paralelnya: kawat panjang yang tipis itu menambahkan
\omterm{def:g8:ohms-law:resistance}{hambatan}, dan $I = U/R$ menyusutkan jatah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} itu. \omterm{def:g7:short-circuits-safety:device}{Korsletingnya}:
sebuah jalan pintas yang $R$-nya hampir nol, sehingga $I = U/R$
meledak --- lari serempaknya sejak dahulu adalah pembagian dengan
hampir-nol. Empat tahun pengetahuan rangkaian yang kualitatif, satu
tagihan sebuah hukum.
\end{example}

\begin{example}[Kawat, dihitung tiap meter]\label{ex:g8:ohms-law:wires}
\omterm{def:g8:ohms-law:resistance}{Hambatan} sepotong kawat bertumbuh menurut panjangnya dan \omterm{def:g7:moon-phases:phases}{menyusut}
menurut tebalnya --- kawat yang panjang dan tipis adalah jalan gunung
yang sempit, yang pendek dan tebal adalah jalan bebas \omterm{def:g8:ohms-law:resistance}{hambatan}. Karena
itulah tukang listrik memakai kabel gemuk untuk perkakas yang rakus,
kumparan pemanas memakai paduan berhambatan yang sengaja dibuat panjang
dan tipis --- dan itulah sebabnya hukum yang sama yang menyalakan
lampu menghangatkan pemanggang roti: \omterm{def:g8:ohms-law:resistance}{hambatan} adalah tempat
arak-arakannya membelanjakan dorongannya, dan dorongan yang
dibelanjakan, sebagaimana akan ditegaskan bab \omterm{def:g5:everyday-energy:energy}{energi} tahun depan,
menjadi kehangatan.
\end{example}

\begin{remark}[Di mana kekuasaan hukumnya berakhir]\label{rem:g8:ohms-law:limits}
Hukum Ohm adalah hukum sebuah \omterm{def:g8:ohms-law:resistor}{resistor}, bukan hak yang berlaku
semesta. Ukurlah potret sebuah \emph{lampu berfilamen} dan titiknya
membelok menjauhi garis lurusnya: filamennya memanas selagi arusnya
bertumbuh, dan logam yang panas lebih banyak melawan --- $R$ menolak
tinggal diam. Dioda, \omterm{ex:g6:circuits-and-safety:motor}{motor}, dan kulit yang hidup juga membelokkan
potretnya, masing-masing dengan caranya sendiri. Garis lurus yang
melewati titik asal adalah tanda tangan sebuah \omterm{def:g8:ohms-law:resistor}{resistor}, bukan tanda
tangan materi --- selalu mintalah potret sebuah komponen sebelum
memercayakan hukumnya kepadanya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
Apa yang diukur \omterm{def:g8:ohms-law:resistance}{hambatan}? Berikan \omterm{def:g6:measurement-in-science:measuring}{satuan} dan lambangnya, lalu tulislah
hukum Ohm beserta ketiga susunan ulangnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
Lengkapilah tiap pasangannya untuk \omterm{def:g8:ohms-law:resistor}{resistor} $30\,\Omega$: $U =
\qty{6}{V}$, $I = {?}$; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
Sebuah komponen menunjukkan $U = \qty{4.5}{V}$ pada $I = \qty{0.09}{A}$.
Berapa \omterm{def:g8:ohms-law:resistance}{hambatannya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
Pada percobaan potretnya, bentuk apa yang digambar titik sebuah
\omterm{def:g8:ohms-law:resistor}{resistor} sejati, dan apa arti garis yang lebih curam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
Dari tabel babnya, periksalah $R$ dari baris kedua dan keempat.
Mengapa memperoleh bilangan yang sama pada kedua kalinya justru
menjadi pokoknya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
Kumparan cereknya $25\,\Omega$; kumparan cerek kedua $35\,\Omega$.
Pada listrik rumah yang sama, yang mana yang menarik arus lebih kuat
--- dan berapa yang ditarik masing-masing (satu angka di belakang
koma)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
Jelaskan dengan $U = R I$ mengapa, di antara dua lampu seri yang
berbagi satu arus, lampu yang \omterm{def:g8:ohms-law:resistance}{hambatannya} lebih besar mengambil
\omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang lebih besar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
Terjemahkan ke dalam bahasa \omterm{def:g8:ohms-law:resistance}{hambatan}: lari serempak \omterm{def:g7:short-circuits-safety:device}{korsletingnya}; \
penghadangan total \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang terbuka.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
Sebuah lampu penunjuk yang halus tahan \qty{20}{mA}. \omterm{def:g8:ohms-law:resistor}{Resistor} seri
yang mana yang melindunginya pada \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{9}{V}, kalau lampunya
sendiri mengambil kira-kira \qty{2}{V} dari bagiannya? (Bagian
\omterm{def:g8:ohms-law:resistor}{resistornya} lebih dahulu, lalu $R$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
Potret sebuah lampu berfilamen, terukur: (\qty{1}{V},
\qty{0.25}{A}), (\qty{4}{V}, \qty{0.5}{A}). Tunjukkan bahwa tidak ada
$R$ tunggal yang cocok bagi kedua titiknya, lalu ceritakan kisah
fisika di balik pembelokannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
Dua \omterm{def:g8:ohms-law:resistor}{resistor} \omterm{def:g4:series-circuits:series}{terangkai seri}, $10\,\Omega$ dan $20\,\Omega$, pada
\qty{6}{V}. Dengan memakai hukum serinya ditambah hukum Ohm pada
masing-masing: carilah arus lingkarnya dan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} tiap \omterm{def:g8:ohms-law:resistor}{resistornya}.
(Petunjuk: satu arus $I$ memenuhi $6 = 10 I + 20 I$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
Kedua \omterm{def:g8:ohms-law:resistor}{resistor} yang sama, kini terpasang paralel pada \qty{6}{V}.
Carilah arus tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}, arus total \omterm{def:g3:first-electric-circuit:battery}{baterainya}, lalu --- sebagai
penutup yang anggun --- \omterm{def:g8:ohms-law:resistance}{hambatan} tunggal $R$ yang akan menarik total
yang sama itu dari \qty{6}{V}. Bandingkan dengan kedua \omterm{def:g8:ohms-law:resistor}{resistornya}
lalu jelaskan, dalam bahasa jalan, mengapa regu berdua itu melawan
\emph{lebih sedikit} daripada anggotanya masing-masing sendirian.
\end{exercise}

\section{Soal: Meja Pengawasan Mutu}

\begin{problem}[{Soal akhir pekan --- hari pemeriksaan di pabrik
resistor; potret, toleransi, dan satu komponen yang sama sekali bukan
resistor}]\label{pb:g8:ohms-law:1}
Meja mutu pabriknya menguji hasil produksi hari itu --- dan satu
pengelabu. Kamu menjalankan alat ukurnya.

\medskip\noindent\textbf{Bagian I --- Menera mejanya.}
\begin{enumerate}
  \item Sketsakan (atau gambarkan) rangkaian ujinya: sumber,
        \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter}, komponen yang diuji, \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} --- siapa yang
        \omterm{def:g4:series-circuits:series}{terangkai seri}, siapa yang melintang?
  \item Contoh satu, pada \qty{4.5}{V}, melewatkan \qty{0.15}{A}.
        Berapa \omterm{def:g8:ohms-law:resistance}{hambatannya}?
  \item Labelnya mengaku $33\,\Omega$, dengan toleransi lima persen.
        Hitunglah pita yang dapat diterima, lalu berilah putusan atas
        contoh satu. (Lima persen dari $33$ lebih dahulu.)
  \item Contoh dua, berlabel $150\,\Omega$: ramalkan arusnya pada
        \qty{4.5}{V}, supaya mejanya tahu apa yang harus diharapkan
        (satu angka di belakang koma, dalam \unit{mA}).
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Potret yang lengkap.}
\begin{enumerate}[resume]
  \item Tabel contoh tiga: $(1.5, 0.05)$, $(3.0, 0.10)$,
        $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{volt}, \omterm{def:g8:intensity-ammeter:intensity}{ampere}. Cocokkanlah
        $R$-nya, sambil menyebut ciri bilangannya yang mengizinkan
        satu nilai tunggal.
  \item Tabel contoh empat: $(1.5, 0.30)$, $(3.0, 0.45)$,
        $(4.5, 0.54)$, $(6.0, 0.60)$. Tunjukkan pengelabunya: tidak
        ada $R$ tunggal, dan ke mana arah pergeserannya.
  \item Contoh empat sebenarnya adalah sebuah lampu berfilamen kecil
        dari jalur produksi berikutnya. Ceritakan kisah
        pembelokannya --- lalu mengapa uji garis lurus mejanya justru
        merupakan pengesan \omterm{def:g8:ohms-law:resistor}{resistor}.
  \item Kitab aturan mejanya berkata: ``setiap potret harus diambil
        pada \omterm{def:g3:temperature-thermometers:temperature}{suhu} yang mantap.'' Pasal hukum Ohm yang mana yang
        sedang dilindungi aturan itu?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Bagian
penerapannya.}
\begin{enumerate}[resume]
  \item Sebuah pesanan meminta \omterm{def:g8:ohms-law:resistor}{resistor} yang membatasi arus sampai
        \qty{30}{mA} pada sumber \qty{12}{V} (bagian bebannya sendiri
        dapat diabaikan). Hitunglah nilai yang harus ditawarkan.
  \item Seorang pelanggan mengeluh bahwa kumparan pemanas
        $25\,\Omega$ mereka ``hanya'' menarik \qty{9.2}{A} sementara
        \omterm{def:g7:short-circuits-safety:fuse}{sekringnya} mengizinkan $10$. Hitunglah \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} sumber yang
        disiratkan keluhannya, lalu tenangkan mereka dengan
        hitungannya.
  \item Laboratorium purwarupanya meminta kabel penghubung tembaga
        pabriknya yang paling tebal dan paling pendek ``dengan \omterm{def:g8:ohms-law:resistance}{ohm}
        sedekat mungkin ke nol''. Peran rangkaian apa yang menjadi
        kelahiran kabel itu --- dan dekat penjahat sehari-hari yang
        mana \omterm{def:g8:ohms-law:resistance}{hambatannya} yang hampir-nol itu menempatkannya kalau
        disalahgunakan?
  \item Tutuplah buku pemeriksaannya: tiga kalimat --- hukumnya,
        potretnya, dan satu syarat yang membuat sebuah komponen boleh
        menandatanganinya.
\end{enumerate}
\end{problem}
