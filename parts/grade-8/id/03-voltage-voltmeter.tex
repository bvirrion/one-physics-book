\chapter{Tegangan dan Voltmeter}\label{ch:g8:voltage-voltmeter}

Sejak \omterm{def:g3:first-electric-circuit:battery}{baterai} yang pertama menyalakan \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang pertama, satu kata
telah menjalankan tugas yang \omterm{def:g4:weight-and-mass:weight}{berat} dan kabur di dalam buku ini:
``dorongan''. \omterm{def:g3:first-electric-circuit:battery}{Baterai} mendorong dengan lemah, listrik rumah mendorong
sampai mematikan, dua \omterm{def:g3:first-electric-circuit:battery}{baterai} kepala-ke-ekor mendorong bersama.
Dorongannya kini menjadi besaran terukur dengan \omterm{def:g6:measurement-in-science:measuring}{satuan} yang termasyhur
--- dan alatnya, tidak seperti alat bab yang lalu, menolak berdiri di
tengah jalan.

\section{Tegangan}

\begin{definition}[Tegangan]\label{def:g8:voltage-voltmeter:voltage}
\emph{Tegangan}\index{tegangan} $U$ di antara dua titik sebuah
rangkaian \omterm{def:g6:measurement-in-science:measuring}{mengukur} dorongan listrik yang tersedia di antara keduanya
--- seberapa keras arusnya digerakkan dari titik yang satu ke titik
yang lain, sebagaimana tinggi sebuah air terjun \omterm{def:g6:measurement-in-science:measuring}{mengukur} seberapa
keras ia menggerakkan roda kincirnya. Tegangan selalu berupa
\emph{di antara}: di antara kedua kutub sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai}, di antara
kedua sisi sebuah lampu --- tidak pernah pada satu titik sendirian.
\omterm{def:g6:measurement-in-science:measuring}{Satuannya} adalah \emph{volt}\index{volt} (\unit{V}), yang menghormati
penemu \omterm{def:g3:first-electric-circuit:battery}{baterai} yang pertama.
\end{definition}

\begin{example}[Album keluarga volt]\label{ex:g8:voltage-voltmeter:values}
Sebuah sel bulat tunggal: \qty{1.5}{V}. \omterm{def:g3:first-electric-circuit:battery}{Baterai} saku yang pipih:
\qty{4.5}{V} --- tiga sel kepala-ke-ekor di dalam satu bungkus.
\omterm{def:g3:first-electric-circuit:battery}{Baterai} sembilan \omterm{def:g8:voltage-voltmeter:voltage}{volt} yang berbentuk balok dengan kedua kutubnya di
atas: \qty{9}{V}, enam sel yang tersembunyi. Aki mobil: \qty{12}{V}.
Stopkontak dinding: \qty{230}{V} --- dorongan yang lebih besar
daripada seratus lima puluh sel kecil, dan itulah seluruh isi
kuantitatif ``stopkontak, jangan pernah''. Petir di antara awan dan
tanah: ratusan juta \omterm{def:g8:voltage-voltmeter:voltage}{volt}, sejenak.
\end{example}

\begin{proposition}[Tegangan tanpa arus]\label{prop:g8:voltage-voltmeter:nocurrent}
\omterm{def:g8:voltage-voltmeter:voltage}{Tegangan} dan arus adalah besaran yang berbeda, dan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} datang
lebih dahulu. Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} yang sendirian di dalam laci, yang
terhubung kabel ke apa pun, sama sekali tidak menggerakkan arus ---
namun di antara kutubnya \qty{1.5}{V} yang penuh berdiri siap, dapat
diukur, seperti tinggi sebuah air terjun pada hari kincirnya tutup.
Dorongan dapat ada tanpa arak-arakan; arak-arakannya tidak pernah ada
tanpa dorongan.
\end{proposition}

\begin{definition}[Voltmeter]\label{def:g8:voltage-voltmeter:voltmeter}
Sebuah \emph{voltmeter}\index{voltmeter} \omterm{def:g6:measurement-in-science:measuring}{mengukur} \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} di antara
dua titik --- sehingga ia dihubungkan \emph{melintang} padanya:
\emph{secara paralel} dengan komponen yang diminati, satu kabel pada
tiap sisinya, seperti juru ukur yang memegang alat datar di antara dua
ketinggian. Karena dibangun berlawanan dengan \omterm{def:g8:intensity-ammeter:ammeter}{amperemeternya} dalam
segala hal, ia melawan arus hampir \emph{sepenuhnya}: kalau
dihubungkan melintang pada apa pun, ia menarik hampir tidak ada ---
sehingga menjepitkannya tidak mengganggu apa pun, dan tidak ada
lingkar yang perlu diputus.
\end{definition}

\begin{method}[Mengukur sebuah tegangan]\label{met:g8:voltage-voltmeter:measure}
\begin{enumerate}
  \item biarkan rangkaiannya utuh --- tidak ada yang diputus untuk
        sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter};
  \item jepitkan satu kabel pada tiap sisi komponennya: kabel
        \unit{V} di tempat arus perjanjiannya masuk, kabel bersamanya
        di tempat arusnya keluar;
  \item jangkauan terbesar lebih dahulu, lalu turunlah sampai nyaman;
  \item bacalah, lalu tulislah \omterm{def:g6:measurement-in-science:measuring}{satuannya}.
\end{enumerate}
Tata cara kedua alatnya, bersebelahan: \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} --- putuskan
jalannya, berdirilah di dalamnya, jangan lawan apa pun; \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} ---
jangan putuskan apa pun, berdirilah di samping, lawanlah segalanya.
Mengacaukan kedua tata caranya adalah kebiasaan yang paling mahal di
bengkelnya.
\end{method}

\section{Tegangan bertakaran}

\begin{definition}[Tegangan nominal]\label{def:g8:voltage-voltmeter:nominal}
Setiap lampu, \omterm{ex:g6:circuits-and-safety:motor}{motor}, dan perkakas membawa sebuah \emph{tegangan
nominal}\index{tegangan nominal} --- dorongan yang menjadi rancangan
pembuatannya, yang dicap pada alas atau pelatnya: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} senter ``3,5
V'', lampu depan mobil ``12 V'', cerek ``230 V''. Kalau diberi
\omterm{def:g8:voltage-voltmeter:voltage}{tegangan} nominalnya, alatnya bekerja sebagaimana dirancang; kalau
diberi jauh lebih sedikit, ia merajuk --- lampu yang redup, \omterm{ex:g6:circuits-and-safety:motor}{motor} yang
malas; kalau diberi jauh lebih banyak, ia terbakar putus --- dengan
gemilang, sekali saja.
\end{definition}

\begin{example}[Memadankan lampu dengan baterai]\label{ex:g8:voltage-voltmeter:matching}
\omterm{def:g3:first-electric-circuit:bulb}{Bohlam} ``3.5 V'' pada \omterm{def:g3:first-electric-circuit:battery}{baterai} pipih \qty{4.5}{V}: sedikit
kelebihan dorongan --- terang, dan berumur pendek. Pada satu sel
\qty{1.5}{V}: nyala jingga yang mengantuk. Pada \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{9}{V}:
satu saat yang menyilaukan, lalu benang tipisnya pun putus --- itulah
berita duka filamennya. Labelnya adalah sebuah perjanjian: hormatilah,
dan lampunya menjalani umur yang diiklankannya.
\end{example}

\begin{remark}[Mengapa 230 V membunuh]\label{rem:g8:voltage-voltmeter:mains}
Kini aturan mutlak yang lama mendapat bilangannya. Jalan penghantar
tubuh yang lembap, kalau ditawari \qty{4.5}{V}, melewatkan arus yang
terlalu kecil untuk dirasakan. Jalan yang sama pada \qty{230}{V} ---
lima puluh kali dorongannya --- melewatkan lima puluh kali arusnya,
jauh melampaui beberapa perseratus \omterm{def:g8:intensity-ammeter:intensity}{ampere} yang menghentikan jantung.
Stopkontaknya bukan ``agak'' lebih kuat daripada meja \omterm{def:g3:first-electric-circuit:battery}{baterai} kalian:
ia dua orde besaran di luarnya, dan kerasnya kitab aturannya adalah
hitungan biasa.
\end{remark}

\section{Bacaan tegangan yang pertama}

\begin{example}[Berkeliling sebuah lingkar yang sederhana]\label{ex:g8:voltage-voltmeter:readings}
Satu \omterm{def:g3:first-electric-circuit:battery}{baterai}, satu lampu, alat ukur di tangan. Melintang pada
\omterm{def:g3:first-electric-circuit:battery}{baterainya}: \qty{4.5}{V}. Melintang pada lampunya: \qty{4.5}{V} ---
lampunya menerima apa yang ditawarkan \omterm{def:g3:first-electric-circuit:battery}{baterainya}. Melintang pada
sebentang \emph{kawat} penghubung: \qty{0}{V}, atau sedekat yang dapat
dikatakan alatnya --- kawat yang baik adalah jalan yang datar, yang
sama sekali tidak membelanjakan dorongannya. Dorongannya berdiam
melintang pada \emph{pekerjanya}, bukan di sepanjang jalannya: sebuah
bacaan yang patut diingat ketika hukum bab berikutnya meresmikannya.
\end{example}

\begin{example}[Pengakuan sakelar yang terbuka]\label{ex:g8:voltage-voltmeter:openswitch}
Lampu gelap, \omterm{ex:g3:first-electric-circuit:switch}{sakelar} terbuka --- ke mana dorongannya pergi? Ukurlah:
melintang pada lampu yang gelap, \qty{0}{V}; melintang pada
\emph{\omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang terbuka}, \qty{4.5}{V} yang penuh. Dorongan
\omterm{def:g3:first-electric-circuit:battery}{baterainya} berdiri seluruhnya melintang pada celahnya, menanti ---
satu-satunya tempat di dalam lingkarnya tempat dua titik sungguh
``berjauhan'' secara listrik. Tutuplah \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} dan bacaannya pun
bertukar tempat. \omterm{ex:g3:first-electric-circuit:switch}{Sakelar} yang terbuka adalah tempat \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya}
menanti.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:1}
Apa yang diukur \omterm{def:g8:voltage-voltmeter:voltage}{tegangan}, dan mengapa ia selalu berupa ``di antara''?
Berikan \omterm{def:g6:measurement-in-science:measuring}{satuannya} dan gambaran air terjunnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:2}
Dari ingatan: \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} sebuah sel bulat, \omterm{def:g3:first-electric-circuit:battery}{baterai} pipihnya, \omterm{def:g3:first-electric-circuit:battery}{baterai}
baloknya, sebuah aki mobil, dan listrik rumahnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:3}
Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} tergeletak tanpa terhubung kabel ke apa pun. Arus apa
yang mengalir? Berapa yang dibaca \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} yang melintang padanya?
Proposisi yang mana yang berbicara?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:4}
Bandingkan tata cara kedua alatnya: sambungannya, pemutusan jalannya,
perlawanannya terhadap arus. Kekeliruan yang mana yang membangun
\omterm{def:g7:short-circuits-safety:device}{korsleting} berpapan ukur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:5}
Apa itu \omterm{def:g8:voltage-voltmeter:nominal}{tegangan nominal}? Ramalkan perilaku lampu ``6 V'' pada
\qty{1.5}{V}; pada \qty{6}{V}; pada \qty{12}{V}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:6}
Pada lingkar yang sederhana itu, berapa yang dibaca \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeternya}
melintang pada \omterm{def:g3:first-electric-circuit:battery}{baterainya}, pada lampunya, dan pada sebentang kawat? Di
mana dorongannya berdiam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:7}
\omterm{ex:g3:first-electric-circuit:switch}{Sakelar} terbuka: apa yang berdiri melintang pada lampu yang gelap, dan
melintang pada \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang terbuka? Apa yang terjadi pada kedua
bacaannya ketika terdengar \omterm{def:g3:sound-around-us:sound}{bunyi} klik?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:voltage-voltmeter:8}
Mengapa menjepitkan \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} melintang pada lampu yang bekerja hampir
sama sekali tidak mengganggu rangkaiannya? Apa yang ditolak alatnya
untuk dilakukan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:9}
\omterm{def:g3:first-electric-circuit:battery}{Baterai} pipih \qty{4.5}{V} menyembunyikan tiga sel. Gambarlah (atau
gambarkan) susunannya, lalu hitunglah \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} seandainya satu
selnya tanpa sengaja terbalik --- bab senter bertahun-tahun yang lalu,
kini dengan bilangan. (Dorongan yang berlawanan saling mengurangi.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:10}
Lampu seorang pekemah bertakaran \qty{12}{V}; aki kendaraannya membaca
\qty{12.6}{V} ketika baru dan \qty{11.2}{V} sesudah sesore yang
panjang. Gambarkan sore lampunya, dan mengapa tidak ada yang terbakar
putus.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:voltage-voltmeter:11}
Sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} yang keliru dipasang \emph{secara seri} dengan
lampunya (lingkarnya terputus, alatnya di dalam celahnya): ramalkan
perilaku lampunya dan bacaan alatnya --- ingatlah apa yang dilawan
sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter}. (Kesunyian lampunya dan bilangan alatnya
sama-sama mengajarkan sesuatu.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:voltage-voltmeter:12}
Dua lampu serupa yang \omterm{def:g4:series-circuits:series}{terangkai seri} pada \qty{4.5}{V}: sebuah
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} menemukan kira-kira \qty{2.25}{V} melintang pada
masing-masing. Lampu yang sama secara paralel pada \omterm{def:g3:first-electric-circuit:battery}{baterai} yang sama:
ramalkan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} tiap lampunya, lalu rukunkan kedua temuannya dengan
aturan terang yang sudah kamu kenal bertahun-tahun (pasangan seri
redup, pasangan paralel bersinar penuh). Bab yang akan datang yang
mana yang akan menjadikan ramalanmu hukum?
\end{exercise}

\section{Soal: Pemeriksaan Laci Baterai}

\begin{problem}[{Soal akhir pekan --- pemeriksaan besar laci
baterainya; sebuah voltmeter, sekardus sepatu berisi sel, dan rak
perkakas keluarganya}]\label{pb:g8:voltage-voltmeter:1}
Setiap rumah mempunyai lacinya: sel yang berserakan, \omterm{def:g3:first-electric-circuit:battery}{baterai}
misterius, perkakas yang ``perlu yang baru''. Hari pemeriksaan ---
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} di tangan.

\medskip\noindent\textbf{Bagian I --- Memilah selnya.}
\begin{enumerate}
  \item Gambarkan sambungan \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} yang benar untuk memeriksa
        sebuah sel \qty{1.5}{V} yang berserakan --- dan mengapa tidak
        ada rangkaian yang perlu dibangun lebih dahulu.
  \item Bacaan pemeriksaannya, langsung dari lacinya:
        \qty{1.58}{V}, \qty{1.32}{V}, \qty{0.9}{V},
        \qty{0.1}{V}. Pilahlah: baru, masih terpakai, lelah, mati.
  \item Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} sembilan \omterm{def:g8:voltage-voltmeter:voltage}{volt} membaca \qty{8.9}{V}. Kasarnya
        berapa yang disumbangkan masing-masing keenam sel di dalamnya?
  \item Mengapa sel yang hampir mati masih menunjukkan \emph{sedikit}
        \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} padahal ia tidak dapat lagi menyalakan apa pun?
        (Dorongannya selamat dengan lemah; ia kehilangan kekuatan
        untuk menggerakkan apa?)
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Rak perkakasnya.}
\begin{enumerate}[resume]
  \item Kameranya menghendaki dua sel kepala-ke-ekor. \omterm{def:g8:voltage-voltmeter:voltage}{Tegangan} berapa
        yang diharapkannya, dan berapa yang akan diantarkan pasangan
        \qty{1.32}{V} yang selamat dari pertanyaan 2?
  \item Radio tuanya memakai empat sel kepala-ke-ekor lalu merajuk di
        bawah \qty{5}{V}. Sel laci yang mana, kalau ada, yang dapat
        mengawakinya?
  \item Adik lelakinya menjejalkan tiga sel ke dalam senter berisi dua
        sel ``supaya lebih terang'', dengan mengganjal sel ketiganya
        melintang pada pegasnya. \omterm{def:g3:first-electric-circuit:bulb}{Bohlamnya} bertakaran \qty{3}{V}.
        Ramalkan pertunjukannya, dalam bahasa \omterm{def:g8:voltage-voltmeter:nominal}{tegangan nominal}.
  \item Pelat trafo bel pintunya membaca ``\qty{8}{V}''. Seseorang
        mengusulkan memasang bel pintunya langsung ke sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai}
        sembilan \omterm{def:g8:voltage-voltmeter:voltage}{volt} ``karena delapan itu hampir sembilan''.
        Nilailah: padanan yang adil atau pembunuhan yang lambat?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Hukum pemeriksaannya.}
\begin{enumerate}[resume]
  \item Pada senternya (\omterm{def:g3:first-electric-circuit:bulb}{bohlam} dan selnya \omterm{def:g4:series-circuits:series}{terangkai seri}, \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}
        terbuka), pemeriksanya \omterm{def:g6:measurement-in-science:measuring}{mengukur}: melintang pada \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang
        terbuka, \qty{3.1}{V}; melintang pada \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}, \qty{0}{V}.
        Jelaskan keduanya, lalu ramalkan keduanya sesudah \omterm{def:g3:sound-around-us:sound}{bunyi}
        kliknya.
  \item Melintang pada tiap bentang kabel senternya alatnya berkata
        \qty{0}{V}, entah menyala entah gelap. Apa yang dinyatakan
        hal ini tentang kawat yang baik, dalam bahasa dorongan?
  \item Stopkontak listrik rumahnya \emph{tidak} diperiksa, menurut
        hukum keluarga yang tetap. Nyatakan ulang hukumnya dengan
        bilangan bab ini, dan kalimat hitungan di baliknya.
  \item Arsipkan laporan pemeriksaannya: tiga kalimat --- sebuah
        \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} itu apa dan tidak pernah apa; apa yang dijanjikan
        dan diancamkan \omterm{def:g8:voltage-voltmeter:nominal}{tegangan nominal} pada sebuah label; dan di
        mana, di dalam rangkaian yang menganggur mana pun, \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya}
        menanti.
\end{enumerate}
\end{problem}
