\chapter{Hukum Arus dan Tegangan di dalam Rangkaian}\label{ch:g8:circuit-laws}

Dua alat ukur, dua besaran --- dan kini, akhirnya, undang-undang
dasarnya. Setiap rangkaian yang akan pernah kamu bangun, dari sebuah
senter sampai jaringan sebuah kota, menaati empat hukum yang pendek:
dua tentang pembukuan arusnya, dua tentang pembukuan \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya}.
Berbekal keempatnya dan sedikit aljabar, kamu dapat menghitung bacaan
yang belum kamu ukur --- lalu menangkap pengukuran mana pun yang
berdusta.

\section{Hukum arusnya}

\begin{proposition}[Hukum arus bagi lingkar seri]\label{prop:g8:circuit-laws:seriesI}
Di sepanjang satu lingkar tunggal, kuat arusnya sama di mana-mana:
\[
I_{\text{baterai}} = I_{\text{lampu 1}} = I_{\text{lampu 2}} =
\dots
\]
Satu jalan, satu arak-arakan --- gardu tol bab yang lalu, kini menjadi
sebuah hukum undang-undang dasarnya.
\end{proposition}

\begin{proposition}[Hukum percabangan]\label{prop:g8:circuit-laws:junction}
Pada percabangan mana pun, \omterm{def:g8:intensity-ammeter:intensity}{kuat arus} yang mengalir masuk bersama-sama
sama dengan \omterm{def:g8:intensity-ammeter:intensity}{kuat arus} yang mengalir keluar bersama-sama: untuk sebuah
jalan utama yang membelah menjadi dua \omterm{def:g5:series-parallel-circuits:parallel}{cabang},
\[
I = I_1 + I_2 .
\]
Muatan tidak diciptakan, tidak dimusnahkan, dan tidak disimpan pada
sebuah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} jalan: apa pun yang tiba, pergi. (Benar untuk berapa pun
banyak \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} --- jumlahkan yang tiba, jumlahkan yang berangkat, dan
pembukuannya seimbang.)
\end{proposition}

\section{Hukum tegangannya}

\begin{proposition}[Hukum tegangan bagi lingkar seri]\label{prop:g8:circuit-laws:seriesU}
Mengelilingi sebuah lingkar seri, \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} \omterm{def:g3:first-electric-circuit:battery}{baterainya} dibagi di antara
alatnya, dan bagiannya terjumlah dengan tepat:
\[
U_{\text{baterai}} = U_1 + U_2 + \dots
\]
Gambaran air terjunnya menjaga pembukuannya: \omterm{def:g3:first-electric-circuit:battery}{baterainya} mengangkat
arak-arakannya menembus seluruh tingginya, dan alatnya membelanjakan
tinggi itu di antara mereka, tetes demi tetes --- dengan kawat yang
baik sama sekali tidak membelanjakan apa-apa.
\end{proposition}

\begin{proposition}[Hukum tegangan bagi cabang paralel]\label{prop:g8:circuit-laws:parallelU}
Alat yang terpasang paralel semuanya membentang pada dua percabangan
yang sama --- sehingga satu \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} melayani mereka semua:
\[
U_1 = U_2 = U_{\text{melintang pada pasangan percabangannya}} .
\]
Tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} menikmati dorongan penuh di antara percabangannya ---
itulah sebabnya lampu paralel bersinar pada terang nominal penuh, dan
sebabnya setiap stopkontak di rumahmu menawarkan \qty{230}{V} yang
sama.
\end{proposition}

\begin{example}[Mengapa bagiannya tidak sama?]\label{ex:g8:circuit-laws:unequal}
Dua lampu yang berbeda yang \omterm{def:g4:series-circuits:series}{terangkai seri} membawa arus yang sama
(hukum satu) namun membelah \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} secara tidak sama ---
\qty{2.1}{V} lawan \qty{2.4}{V}. Bagian yang lebih besar jatuh kepada
lampu yang lebih banyak melawan arak-arakannya: pekerja yang lebih
berat mengambil turunan yang lebih besar. Apa tepatnya arti ``lebih
banyak melawan'', dan bagaimana \omterm{def:g6:measurement-in-science:measuring}{mengukurnya}, adalah seluruh pokok bab
berikutnya --- hukum bab ini adalah panggungnya.
\end{example}

\section{Menghitung sebelum mengukur}

\begin{method}[Memecahkan sebuah rangkaian dengan keempat hukumnya]\label{met:g8:circuit-laws:solve}
Diberi sebuah rangkaian dengan beberapa bacaan yang diketahui:
\begin{enumerate}
  \item gambarlah diagramnya lalu tandai setiap $I$ dan $U$ yang
        diketahui;
  \item tulislah hukum yang berlaku sebagai persamaan --- jumlah
        percabangan untuk arusnya, jumlah lingkar untuk \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya},
        kesamaan untuk $I$ seri dan $U$ paralel;
  \item pecahkan untuk yang belum diketahui (persamaan pelajaran
        aljabarmu, yang kini memperoleh nafkahnya);
  \item periksalah kewajarannya: tidak ada lampu yang menerima lebih
        daripada yang ditawarkan \omterm{def:g3:first-electric-circuit:battery}{baterainya}, tidak ada arus \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
        yang melampaui arus jalan utamanya.
\end{enumerate}
\end{method}

\begin{example}[Dikerjakan: rangkaian campurannya]\label{ex:g8:circuit-laws:worked}
Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{6}{V} memberi makan lampu A yang \omterm{def:g4:series-circuits:series}{terangkai seri}
dengan sepasang \omterm{def:g5:series-parallel-circuits:parallel}{cabang} paralel (B dan C). Terukur:
$I_{\text{baterai}} = \qty{0.5}{A}$,
$I_B = \qty{0.3}{A}$, $U_A = \qty{2.5}{V}$. Hitunglah sisanya.
Hukum percabangan: $I_C = 0.5 - 0.3 = \qty{0.2}{A}$. Hukum arus
serinya: lampu A membawa \qty{0.5}{A} yang penuh. Hukum \omterm{def:g8:voltage-voltmeter:voltage}{tegangan}
lingkarnya: pasangannya menerima $U_{\text{pasangan}} = 6 - 2.5 = \qty{3.5}{V}$;
hukum \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} paralelnya: $U_B = U_C = \qty{3.5}{V}$. Setiap bacaan
alat di dalam rangkaiannya, kini diketahui --- empat hukum, satu baris
aljabar masing-masing.
\end{example}

\begin{example}[Hukumnya sebagai pengesan dusta]\label{ex:g8:circuit-laws:liedetector}
Sebuah laporan praktikum mengaku: \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{4.5}{V}; dua lampu seri
pada \qty{2.1}{V} dan \qty{2.9}{V}. Hukum lingkarnya menjumlahkan
bagiannya: \qty{5.0}{V} dibelanjakan dari dompet \qty{4.5}{V} ---
mustahil; seseorang salah membaca sebuah papan ukur. Laporan yang
lain: percabangan masuk \qty{0.40}{A}, \omterm{def:g5:series-parallel-circuits:parallel}{cabang} keluar \qty{0.25}{A} dan
\qty{0.20}{A} --- \omterm{def:g5:series-parallel-circuits:parallel}{cabang} jalannya ``menciptakan'' \qty{0.05}{A}:
ditolak. Sebelum ada hukumnya, bacaan yang keliru tampak sebaik yang
benar; kini undang-undang dasarnya memeriksa setiap halaman.
\end{example}

\begin{remark}[Baterai di bawah hukum yang sama]\label{rem:g8:circuit-laws:batteries}
Aturan kepala-ke-ekor adalah hukum \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} seri yang mengenakan
pakaiannya yang tertua: \omterm{def:g3:first-electric-circuit:battery}{baterai} yang \omterm{def:g4:series-circuits:series}{terangkai seri} menjumlahkan
\omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} ($1.5 +
1.5 + 1.5 = \qty{4.5}{V}$ --- rahasia \omterm{def:g3:first-electric-circuit:battery}{baterai} pipihnya), dan satu sel
yang terbalik \emph{mengurangi}, dengan dorongannya dibelanjakan untuk
melawan yang lain. Hukumnya mengatur pemberi dan pembelanja sama saja;
hanya tanda sumbangan mereka yang berbeda.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:1}
Nyatakan keempat hukumnya --- dua untuk arus, dua untuk \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} ---
masing-masing dalam satu baris.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:2}
Sebuah percabangan menerima \qty{0.8}{A} lalu mengirim \qty{0.55}{A}
menuruni satu \omterm{def:g5:series-parallel-circuits:parallel}{cabang}. \omterm{def:g5:series-parallel-circuits:parallel}{Cabang} yang lain membawa berapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:3}
Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{9}{V} memberi makan tiga lampu seri; dua di
antaranya mengambil \qty{2.5}{V} dan \qty{3.8}{V}. Berapa bagian yang
ketiga?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:4}
Dua lampu terpasang paralel melintang pada \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{12}{V}: berapa
\omterm{def:g8:voltage-voltmeter:voltage}{tegangan} tiap lampunya? Kenyataan sehari-hari apa tentang stopkontak
rumah yang merupakan hukum yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:5}
Pada rangkaian campuran yang dikerjakan itu, satu pengukuran tambahan
yang mana yang akan mubazir, dan mengapa? (Pilihlah salah satu, lalu
belalah dengan sebuah hukum.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:6}
Dua lampu seri yang berbeda membawa arus yang sama tetapi membelah
\omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} $1.8$ lawan \qty{2.7}{V}. Lampu yang mana yang lebih
banyak melawan arak-arakannya? Bab yang akan datang yang mana yang
\omterm{def:g6:measurement-in-science:measuring}{mengukur} perlawanan itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:7}
Periksalah: \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{6}{V}; lampu seri terukur \qty{2.2}{V},
\qty{1.9}{V}, \qty{1.4}{V}. Apakah pembukuannya seimbang dalam batas
kejujuran sebuah alat ukur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:8}
Periksalah: jalan utama \qty{0.9}{A}; tiga \omterm{def:g5:series-parallel-circuits:parallel}{cabang} \qty{0.4}{A},
\qty{0.3}{A}, \qty{0.3}{A}. Apa putusannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:9}
Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{4.5}{V}, lampu D \omterm{def:g4:series-circuits:series}{terangkai seri} dengan sebuah
\omterm{ex:g3:first-electric-circuit:switch}{sakelar} --- lalu, secara paralel dengan lampu D, sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter}
menunjukkan \qty{4.5}{V} melintang pada \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang \emph{terbuka}
dan \qty{0}{V} melintang pada D. Tunjukkan bahwa kedua bacaannya
menaati hukum \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} lingkarnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:10}
Tiga sel serupa kepala-ke-ekor menyalakan sebuah lampu; sel keempat,
yang terbalik, bergabung ke barisannya. Berapa \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} totalnya
sebelum dan sesudahnya, menurut hukum serinya dengan tanda? Apa yang
dilakukan lampunya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:11}
Sebuah rangkaian bercabang dua: \omterm{def:g5:series-parallel-circuits:parallel}{cabang} satu membawa lampu E dan F
secara seri; \omterm{def:g5:series-parallel-circuits:parallel}{cabang} dua membawa lampu G sendirian. \omterm{def:g3:first-electric-circuit:battery}{Baterainya}
\qty{6}{V}; $U_E = \qty{2.2}{V}$. Hitunglah $U_F$ dan $U_G$, sambil
menyebut sebuah hukum untuk tiap langkahnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:circuit-laws:12}
Pemeriksaan yang lengkap: sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{6}{V}; lampu P \omterm{def:g4:series-circuits:series}{terangkai
seri} dengan sebuah percabangan; \omterm{def:g5:series-parallel-circuits:parallel}{cabang} Q dan R bergabung kembali lalu
pulang. Terukur: $I_P = \qty{0.6}{A}$, $I_Q = \qty{0.45}{A}$,
$U_Q = \qty{2.3}{V}$. Hitunglah setiap arus dan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang tersisa
di dalam rangkaiannya ($I_R$, $U_P$, $U_R$), sambil menyatakan hukum
di balik tiap barisnya --- lalu karanglah satu ``pengukuran'' tambahan
yang mungkin dilaporkan murid yang ceroboh yang akan langsung
dihukum oleh penyelesaianmu yang sudah jadi.
\end{exercise}

\section{Soal: Kotak Misteri yang Tersegel}

\begin{problem}[{Soal akhir pekan --- tantangan kotak tersegel milik
klub elektronika; empat hukum melawan empat misteri; pemeriksaan
besarnya}]\label{pb:g8:circuit-laws:1}
Klubnya menyegel rangkaian sederhana ke dalam kotak, dengan hanya
menyisakan lubang untuk alat ukur. Regunya harus menyimpulkan isinya
--- dengan undang-undang dasarnya di tangan. \omterm{def:g3:first-electric-circuit:battery}{Baterai} mejanya
\qty{6}{V} sepanjang acaranya.

\medskip\noindent\textbf{Bagian I --- Kotak A: dua kutub, dua
lampu.}
Labelnya mengakui: dua lampu yang serupa dan tidak ada yang lain,
entah \omterm{def:g4:series-circuits:series}{terangkai seri} entah paralel.
\begin{enumerate}
  \item Ketika terhubung ke \omterm{def:g3:first-electric-circuit:battery}{baterainya}, kotak A menarik \qty{0.2}{A};
        satu lampu semacam itu sendirian pada \qty{6}{V} diketahui
        menarik kira-kira \qty{0.4}{A}. Seri atau paralel?
        Berhujahlah dengan sebuah hukum arus.
  \item Ramalkan berapa yang akan ditarik kotaknya dalam susunan yang
        \emph{lain}.
  \item \omterm{def:g8:voltage-voltmeter:voltage}{Tegangan} berapa yang diterima tiap lampu di dalamnya pada
        susunan yang sesungguhnya, menurut hukum \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang mana?
  \item Pada susunan yang mana lampu tersembunyinya akan bersinar
        pada terang nominal penuh --- dan mengapa mereka jelas-jelas
        tidak begitu, kalau dilihat lewat lubang intip kotaknya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Kotak B: teka-teki
tiga lampunya.}
Kotak B mengakui: satu lampu (X) \omterm{def:g4:series-circuits:series}{terangkai seri} dengan sepasang \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
paralel (Y, Z --- tidak harus serupa).
\begin{enumerate}[resume]
  \item Regunya \omterm{def:g6:measurement-in-science:measuring}{mengukur} $I_{\text{baterai}} = \qty{0.5}{A}$
        dan, pada lubang uji milik Y sendiri, $I_Y = \qty{0.35}{A}$.
        Carilah $I_Z$ dan $I_X$, sambil menyebut hukumnya.
  \item Sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} melintang pada X: \qty{2.8}{V}. Carilah
        \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} pasangan paralelnya --- dan masing-masing $U_Y$, $U_Z$.
  \item Y dan Z membawa arus yang berbeda pada \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang sama. Apa
        yang diungkapkan hal ini tentang kedua lampunya, dalam bahasa
        \cref{ex:g8:circuit-laws:unequal}?
  \item Regunya melepas Y dari fittingnya lewat sebuah pintu kecil.
        Ramalkan arah perubahan $I_{\text{baterai}}$ yang baru (naik,
        turun, atau tetap), dengan bernalar dari jalannya dan
        perlawanannya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Kotak C: kecurangannya.}
\begin{enumerate}[resume]
  \item Pembangun kotak C melaporkan: ``\omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{6}{V}; dua lampu
        seri di dalamnya pada \qty{2.5}{V} dan \qty{4.1}{V};
        percabangan di dalamnya menerima \qty{0.3}{A} lalu mengirim
        \qty{0.2}{A} ditambah \qty{0.15}{A}.'' Hukumlah laporannya
        dua kali, satu hukum tiap hukuman.
  \item Salah satu di antara kedua bacaan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang dilaporkan
        kemudian ternyata jujur. Kalau itu \qty{2.5}{V}-nya, berapa
        sebenarnya pasangannya?
  \item Pembangunnya mengaku bahwa bilangan percabangannya
        ``dibulatkan dengan murah hati''. Kalau kedua bacaan \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
        \qty{0.2}{A} dan \qty{0.15}{A} itu jujur, berapa sebenarnya
        yang harus dibawa jalan utamanya?
  \item Mengapa keempat hukumnya menjadikan kecurangan kotak tersegel
        permainan yang pasti kalah? Satu kalimat.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Undang-undang dasarnya.}
\begin{enumerate}[resume]
  \item Tulislah piagam klubnya: keempat hukumnya dalam empat baris,
        masing-masing dengan gambaran sehari-hari yang membawanya
        (jalan, gardu tol, air terjun, \omterm{def:g5:series-parallel-circuits:parallel}{cabang} jalan).
\end{enumerate}
\end{problem}
