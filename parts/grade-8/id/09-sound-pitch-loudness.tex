\chapter{Bunyi: Kelajuan, Nada, Kenyaringan}\label{ch:g8:sound-pitch-loudness}

Dawai yang pendek bernyanyi tinggi, petikan yang keras bernyanyi
nyaring --- gitar karet gelangmu menemukan aturannya bertahun-tahun
yang lalu. Tahun ini aturannya mendapat bilangan: sebuah hitungan tiap
sekon untuk nadanya, sebuah ukuran ayunan untuk kenyaringannya, dan
sebuah jendela --- dengan tepinya --- tentang apa yang sama sekali dapat
ditangkap telinga manusia. Di luar salah satu tepinya, kelelawar
menertawakan kita.

\section{Frekuensi: nada yang dihitung}

\begin{definition}[Frekuensi]\label{def:g8:sound-pitch-loudness:frequency}
\emph{Frekuensi}\index{frekuensi} $f$ sebuah \omterm{def:g3:sound-around-us:vibration}{getaran} adalah banyaknya
ayunan bolak-balik lengkap yang dibuatnya tiap sekon. \omterm{def:g6:measurement-in-science:measuring}{Satuannya} adalah
\emph{hertz}\index{hertz} (\unit{Hz}): satu ayunan tiap sekon ---
yang menghormati fisikawan yang pertama kali membuat lalu menangkap
gelombang yang tak terlihat. Dawai gitar yang bergetar $110$ kali tiap
sekon bergetar pada \qty{110}{Hz}; sayap seekor nyamuk, dekat
\qty{600}{Hz}; kibasan lambat sehelai bendera besar, beberapa hertz.
\end{definition}

\begin{proposition}[Nada adalah frekuensi]\label{prop:g8:sound-pitch-loudness:pitch}
Apa yang disebut telinga sebagai \emph{nada} adalah \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} dan
tidak lain: makin cepat sumbernya bergetar, makin tinggi nadanya.
Aturan karet gelangnya langsung diterjemahkan --- benda yang pendek,
tegang, dan tipis bergetar pada $f$ yang lebih tinggi. Jangkar
musiknya: alat musik konser di seluruh dunia menyetel diri pada nada A
pada \qty{440}{Hz}. Menggandakan \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} nada mana pun menaikkannya
sebesar selang yang paling selaras di antara semuanya --- yaitu oktaf:
A lagi, seoktaf di atasnya, pada \qty{880}{Hz}.
\end{proposition}

\begin{method}[Melihat bunyi]\label{met:g8:sound-pitch-loudness:seeing}
Gendang berbutir beras menunjukkan \omterm{def:g3:sound-around-us:vibration}{getaran} \omterm{def:g3:sound-around-us:sound}{bunyinya}; sebuah mikrofon
menggambarnya.
\begin{enumerate}
  \item jalankan sebuah aplikasi penganalisis \omterm{def:g3:sound-around-us:sound}{bunyi} pada telepon atau
        komputer --- layarnya menggambarkan ayunan mikrofonnya
        terhadap waktu, sebuah osiloskop zaman sekarang;
  \item berdengunglah rendah dan mantap: jejak yang malas dan
        berjarak lebar; bernyanyilah tinggi: puncaknya berdesak
        merapat;
  \item bacalah petunjuk \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensinya} sambil bersiul: kebanyakan
        siulan mendarat di antara $500$ dan \qty{2000}{Hz};
  \item pukullah garpu tala bertanda ``A 440'' lalu periksalah apakah
        petunjuknya menghormati ukirannya.
\end{enumerate}
Satu layar menggantikan sebuah bab penuh terkaan: nada yang dihitung
di depan matamu.
\end{method}

\section{Jendela telinganya}

\begin{proposition}[Jangkauan pendengaran manusia]\label{prop:g8:sound-pitch-loudness:range}
Telinga muda yang sehat menangkap \omterm{def:g3:sound-around-us:vibration}{getaran} di antara kira-kira
\qty{20}{Hz} dan \qty{20000}{Hz}. Di bawah jendelanya terletak
\emph{infrasonik}\index{infrasonik} --- kadang-kadang terasa sebagai
gemuruh di dada, tetapi tidak pernah terdengar; di atasnya,
\emph{ultrasonik}\index{ultrasonik} --- kesunyian bagi kita, sedahsyat
apa pun \omterm{def:g3:sound-around-us:vibration}{getarannya}. Jendelanya menyempit menurut usia dan menurut
penyalahgunaan: oktaf yang tertinggi biasanya menjadi korban pertama
kebiasaan yang nyaring.
\end{proposition}

\begin{example}[Di luar tepinya]\label{ex:g8:sound-pitch-loudness:beyond}
Alam dan industri sama-sama bekerja di luar jendela kita. Kelelawar
berburu dengan detak \omterm{prop:g8:sound-pitch-loudness:range}{ultrasonik} dekat \qty{50000}{Hz}, sambil membaca
\omterm{ex:g4:how-sound-travels:echo}{gema} yang lebih halus daripada yang dapat dilukis \omterm{def:g3:sound-around-us:sound}{bunyi} terdengar mana
pun --- lumba-lumba demikian pula, dalam semangat bab sonarnya. Anjing
menjawab siulan yang ``sunyi'' yang nadanya tepat di atas tepi kita.
Pemindai kedokteran mencitrakan anak yang belum lahir dengan \omterm{ex:g4:how-sound-travels:echo}{gema}
\omterm{prop:g8:sound-pitch-loudness:range}{ultrasonik} yang tidak berbahaya. Di tepi yang lain, gajah bercakap
dengan gemuruh \omterm{prop:g8:sound-pitch-loudness:range}{infrasonik} melintasi berkilo-kilometer, dan badai serta
gempa bumi mengumumkan dirinya dengan nada yang tidak diperhatikan
telinga mana pun.
\end{example}

\section{Kenyaringan: ukuran ayunannya}

\begin{proposition}[Kenyaringan adalah amplitudo]\label{prop:g8:sound-pitch-loudness:loudness}
Apa yang disebut telinga sebagai \emph{kenyaringan} berpadanan dengan
\emph{amplitudo}\index{amplitudo} \omterm{def:g3:sound-around-us:vibration}{getarannya} --- ukuran ayunannya,
kekuatan perasan yang berjalan. Petiklah dengan lembut atau dengan
buas: puncak jejaknya berdiri rendah atau tinggi sementara
\emph{jaraknya} --- yaitu nadanya --- tidak tersentuh. \omterm{prop:g8:sound-pitch-loudness:loudness}{Amplitudo} dan
\omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} adalah dua tombol yang mandiri: nada mana pun boleh
dibisikkan atau diteriakkan.
\end{proposition}

\end{omfigure}

\begin{example}[Tangga desibel]\label{ex:g8:sound-pitch-loudness:decibels}
Kenyaringan ditingkatkan dalam \emph{desibel}\index{desibel}
(\unit{dB}) --- sebuah tangga yang dibangun bagi jangkauan telinga yang
menakjubkan, yang padanya tiap penguatan \omterm{def:g3:sound-around-us:sound}{bunyi} sepuluh kali lipat
menambahkan jumlah anak tangga yang sama. Tonggaknya: gemerisik daun,
\qty{10}{dB}; percakapan yang tenang, \qty{50}{dB}; lalu lintas yang
sibuk, \qty{80}{dB}; barisan depan konser rok, \qty{110}{dB}; rasa
sakit dan luka seketika, dekat \qty{120}{dB}. Hitungan tangganya yang
aneh --- langkah yang sama untuk kekuatan sepuluh kali lipat ---
adalah sebuah logaritma, salah satu penemuan besar yang menantimu pada
matematika sekolah menengah.
\end{example}

\begin{remark}[Anggaran telinganya]\label{rem:g8:sound-pitch-loudness:earsafety}
Gendang kecil di dalam telingamu, yang ditemui pada babmu di tahun
ketiga, menyimpan pembukuan seumur hidup. Di atas kira-kira
\qty{85}{dB}, kerusakannya bertumpuk seiring lama paparannya --- batas
sehari kerja di pabrik yang nyaring, hanya beberapa menit pada tingkat
barisan depan konser; headphone pada suara penuh duduk tepat di dalam
pita bahayanya. \omterm{def:g1:five-senses:senses}{Pendengaran} yang hilang karena ayunan yang tinggi
tidak pernah kembali: nada yang tertinggi pergi lebih dahulu, dengan
diam-diam, untuk selamanya. Telinga tidak mempunyai kelopak --- dan
tidak mempunyai suku cadang.
\end{remark}

\begin{example}[Kelajuan, yang tak tergoyahkan oleh kedua tombolnya]\label{ex:g8:sound-pitch-loudness:speed}
Tidak satu pun tombolnya menyentuh pengantarannya. Tinggi dan rendah,
nyaring dan lirih, semua \omterm{def:g3:sound-around-us:sound}{bunyi} menunggangi satu estafet \omterm{def:g7:sound-production:medium}{mediumnya} pada
satu langkah \omterm{def:g7:sound-production:medium}{mediumnya} --- \qty{340}{m/s} di dalam \omterm{def:g2:air-around-us:air}{udara}. Buktinya
dipentaskan tiap malam: sebuah orkes yang terdengar menyeberangi
sebuah taman tiba \emph{selaras dan tepat waktu} --- seandainya nada
yang tinggi lebih cepat daripada yang rendah, setiap kunci nada yang
jauh akan mengabur menjadi arpegio. (Kenyaringan memudar menurut jarak
--- kulit yang menyebar pada bab kelas tujuh --- tetapi memudar itu
melemah, tidak pernah melambat.)
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:1}
Batasilah \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} dan \omterm{def:g6:measurement-in-science:measuring}{satuannya}. Berapa nada A konser, dan nada A
seoktaf di atasnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:2}
Dua jejak membentang pada $20$ milisekon yang sama: yang satu
menunjukkan $4$ ayunan lengkap, yang lain $16$. Hitunglah kedua
\omterm{def:g8:sound-pitch-loudness:frequency}{frekuensinya}; yang mana yang terdengar lebih tinggi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:3}
Berikan jendela \omterm{def:g1:five-senses:senses}{pendengaran} manusianya, dan nama kedua negeri di luar
perbatasannya --- dengan satu penghuni masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:4}
Pada layar penganalisisnya, apa yang berubah ketika kamu menyanyikan
nada yang sama lebih nyaring? Dan ketika kamu menyanyikan nada yang
lebih tinggi dengan kenyaringan yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:5}
Terjemahkan aturan karet gelang yang lama ke dalam bahasa \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi}:
apa yang dilakukan pendek, tegang, dan tipis terhadap $f$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:6}
Tempatkan pada tangga \omterm{ex:g8:sound-pitch-loudness:decibels}{desibelnya}: daun yang berbisik; lalu lintas;
barisan depan konsernya; rasa sakit. Di mana pita bahayanya mulai?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:7}
Mengapa siulan anjing tampak rusak bagi pemiliknya? Dan mengapa
anjingnya tidak sependapat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:sound-pitch-loudness:8}
Sebuah orkes yang jauh tiba selaras dan tepat waktu. Apa yang
dibuktikan hal ini tentang \omterm{def:g7:motion-average-speed:formula}{kelajuan} \omterm{def:g3:sound-around-us:sound}{bunyi} menurut nada dan
kenyaringannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:9}
Seekor kelelawar berdetak pada \qty{50000}{Hz}. Berapa banyak ayunan
lengkap yang dimuat satu detak \qty{0.01}{s}? Mengapa pencitraan \omterm{ex:g4:how-sound-travels:echo}{gema}
yang halus harus memakai \omterm{def:g3:sound-around-us:vibration}{getaran} yang secepat itu? (Ingatlah apa yang
sama-sama dimiliki kuas yang paling halus.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:10}
Tuts terendah sebuah piano besar bergetar dekat \qty{27}{Hz}; yang
tertinggi, dekat \qty{4200}{Hz}. Tempatkan keduanya di dalam (atau di
luar) jendela \omterm{def:g1:five-senses:senses}{pendengarannya} --- lalu jelaskan mengapa penyetel piano
yang menua mungkin memerlukan bantuan penganalisisnya lebih dahulu di
salah satu ujung papan tutsnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:sound-pitch-loudness:11}
Aturan konser mengizinkan $8$ \omterm{ex:g2:measuring-time:clock}{jam} pada \qty{85}{dB} tetapi hanya
beberapa menit dekat \qty{110}{dB}, dengan anggarannya memaruh tiap
beberapa anak tangga. Pertumbuhan macam apa yang bersembunyi di dalam
langkah tangganya yang sama --- dan apa yang diajarkan runtuhnya
anggaran itu tentang ``sedikit lebih nyaring saja''?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:sound-pitch-loudness:12}
Rancanglah pemeriksaan \omterm{def:g3:sound-around-us:sound}{bunyi} yang lengkap bagi sebuah aula sekolah
dengan satu telepon penganalisis: bagaimana memeriksa nada A pianonya
terhadap $440$; bagaimana menemukan kenyaringan aulanya di barisan
belakang selama uji gendangnya; bagaimana menangkap apakah
peranginannya memancarkan gemuruh \omterm{prop:g8:sound-pitch-loudness:range}{infrasonik} yang tidak ditangkap
telinga; dan temuan yang mana di antara ketiganya yang \emph{paling
sedikit} dapat disahkan mikrofon teleponnya (lihatlah jendela
\omterm{def:g1:five-senses:senses}{pendengaran} teleponnya sendiri --- yang dibangun untuk suara orang).
\end{exercise}

\section{Soal: Pemeriksaan Bunyi}

\begin{problem}[{Soal akhir pekan --- pemeriksaan bunyi pada festival
sekolah; satu penganalisis, tiga kelompok musik, dan seorang perawat
yang gugup; sesore di meja pencampurannya}]\label{pb:g8:sound-pitch-loudness:1}
Malam menjelang festival: kamu menjalankan penganalisisnya di meja
pencampurannya, dengan perawat sekolahnya mengawasi alat ukur
\omterm{ex:g8:sound-pitch-loudness:decibels}{desibelnya} dari balik bahumu.

\medskip\noindent\textbf{Bagian I --- \omterm{ex:g2:measuring-time:clock}{Jam} penyetelan.}
\begin{enumerate}
  \item Garpu tala paduan suaranya bernyanyi pada \qty{440}{Hz};
        penganalisisnya menunjukkan nada A pianonya pada
        \qty{432}{Hz}. Buatlah diagnosis pianonya, lalu sebutkan
        tombol yang mana --- nada atau kenyaringan --- yang memerlukan
        penyetelnya.
  \item Dawai terendah gitar basnya terbaca \qty{41}{Hz}. Tempatkan
        ia di dalam jendela \omterm{def:g1:five-senses:senses}{pendengarannya} --- lalu jelaskan mengapa
        penontonnya akan \emph{merasakannya} di dada sebanyak
        mendengarnya.
  \item Nada yang ditahan penyanyi utamanya menunjukkan puncak yang
        berjarak persis seperti puncak garpu talanya, tetapi tiga kali
        lebih tinggi. Bandingkan kedua \omterm{def:g3:sound-around-us:sound}{bunyinya} pada kedua tombolnya.
  \item Seorang kru muda mengaku bahwa pengeras suara baru aulanya
        ``menghasilkan ulang sampai \qty{22000}{Hz}''. Siapa di antara
        penontonnya yang akan mengesahkan puncak jangkauan itu ---
        dan siapa yang tidak dapat?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Buku sang perawat.}
\begin{enumerate}[resume]
  \item Tingkat latihannya: bagian akustik \qty{75}{dB}; bagian rok
        \qty{100}{dB}; babak penutup yang direncanakan ``sedikit di
        atasnya'' pada \qty{110}{dB}. Bagian yang mana yang ditandai
        perawatnya, terhadap baris anggaran yang mana?
  \item Perawatnya membagikan penyumbat telinga busa yang bertakaran
        melembutkan \omterm{def:g3:sound-around-us:sound}{bunyi} sebesar \qty{20}{dB}. Bagian roknya menjadi
        berapa di baliknya --- dan mengapa kru yang bijak memakai
        penyumbat pada setiap pertunjukan?
  \item Ketika berdiri dua kali lebih jauh dari panggungnya, alat
        ukurnya turun dengan jelas. Hukum lama tentang kulit yang
        menyebar yang mana yang sedang bekerja --- dan tombol yang mana
        (nada atau kenyaringan) yang diputar jarak?
  \item Penabuh gendangnya membantah: ``Lebih lirih berarti lebih
        tumpul --- nada yang tinggi mati lebih dahulu ketika kami
        bermain lembut!'' Uraikan kedua tombolnya untuknya: apa
        sebenarnya yang diubah bermain dengan lembut?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Fisika festivalnya.}
\begin{enumerate}[resume]
  \item Kembang api menutup pertunjukannya \qty{680}{m} jauhnya.
        Hitunglah kelambatan kilat-ke-dentum yang akan diperhatikan
        khalayaknya --- dan \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} gemuruhnya kalau peluru
        besarnya menggedebuk dua kali tiap sekon (periksalah
        jendelanya: terdengar atau terasa?).
  \item Misteri terakhir malam itu: sebuah video telepon tentang
        konsernya, ketika diputar ulang, terdengar tipis --- gemuruh
        dadanya lenyap. Lihatlah peringatan latihan 12: apa yang
        memang tidak pernah dibangun untuk ditangkap mikrofonnya?
  \item Laporan perawatnya meminta ``satu kalimat masing-masing''
        tentang kedua tombol malam itu dan hal ketiga yang tidak dapat
        disentuh kedua tombolnya. Rancanglah ketiganya.
  \item Pembongkaran: pengusir hama \omterm{prop:g8:sound-pitch-loudness:range}{ultrasonik} milik penjaganya, yang
        berjalan sepanjang malam di dekat pintu panggungnya, tidak
        mengganggu satu pendengar pun dan, konon, mengganggu banyak
        tikus. Jelaskan kedua khalayaknya dengan satu proposisi
        jendelanya.
\end{enumerate}
\end{problem}
