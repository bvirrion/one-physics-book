\section*{Chapitre \ref{ch:g8:circuit-laws} --- Les lois des courants et des tensions dans les circuits}

\begin{solution}{exo:g8:circuit-laws:1}
Loi des courants \omterm{def:g4:series-circuits:series}{en série}~: une boucle, une \omterm{def:g8:intensity-ammeter:intensity}{intensité} partout. Loi
des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}~: autant de courants entrants que de courants sortants. Loi
des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série}~: la \omterm{def:g8:voltage-voltmeter:voltage}{tension} de la \omterm{def:g3:first-electric-circuit:battery}{pile} égale la somme des
parts des appareils le long de la boucle. Loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}~: des \omterm{def:g5:series-parallel-circuits:parallel}{branches} tendues entre les mêmes \omterm{def:g5:series-parallel-circuits:parallel}{nœuds} se partagent
une seule et même \omterm{def:g8:voltage-voltmeter:voltage}{tension}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:2}
$0.8 - 0.55 = \qty{0.25}{A}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:3}
$9 - 2.5 - 3.8 = \qty{2.7}{V}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:4}
\qty{12}{V} chacune --- la loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}. La même
loi pose les mêmes \qty{230}{V} sur chaque prise du câblage \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation} de la maison.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:5}
Par exemple $I_C$~: une fois $I_{\text{pile}}$ et $I_B$ connus, la
loi des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds} fixe $I_C = \qty{0.2}{A}$ --- tout appareil envoyé là
ne pourrait que confirmer (de même $U_C$, fixé par la loi des
\omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} dès que $U_B$ est connu).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:6}
La lampe à \qty{2.7}{V}~: à courant égal, la plus grande part de
\omterm{def:g8:voltage-voltmeter:voltage}{tension} désigne l'opposant le plus fort. Le chapitre suivant mesure
cette opposition --- la résistance.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:7}
$2.2 + 1.9 + 1.4 = \qty{5.5}{V}$ contre une bourse de \qty{6}{V}~:
un \omterm{def:g8:voltage-voltmeter:voltage}{demi-volt} égaré --- au-delà de la dispersion honnête d'un appareil
pour de telles lectures~; un cadran a été mal lu (ou une part est
restée non mesurée).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:8}
$0.4 + 0.3 + 0.3 = \qty{1.0}{A}$ qui sortent contre \qty{0.9}{A} qui
entrent~: l'embranchement «~crée~» \qty{0.1}{A} --- condamné par la
loi des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:9}
La dépense de la boucle doit totaliser les \qty{4.5}{V} de la \omterm{def:g3:first-electric-circuit:battery}{pile}~:
\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} ouvert \qty{4.5}{V} $+$ lampe au repos \qty{0}{V}
$= 4.5$ --- équilibré. (Après le déclic, les parts s'échangent~:
\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} près de zéro, lampe près de \qty{4.5}{V} --- équilibré
de nouveau.)
\end{solution}

\begin{solution}{exo:g8:circuit-laws:10}
Avant~: $3 \times 1.5 = \qty{4.5}{V}$. Après~:
$4.5 - 1.5 = \qty{3.0}{V}$ --- l'élément retourné se retranche. La
lampe passe de sa lueur de \qty{4.5}{V} à celle de \qty{3}{V}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:11}
La \omterm{def:g5:series-parallel-circuits:parallel}{branche} une est tendue aux \omterm{def:g3:first-electric-circuit:battery}{bornes} de la \omterm{def:g3:first-electric-circuit:battery}{pile}~:
$U_E + U_F = \qty{6}{V}$ (loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série} le long de la
\omterm{def:g5:series-parallel-circuits:parallel}{branche}), donc $U_F = 6 - 2.2 = \qty{3.8}{V}$. La \omterm{def:g5:series-parallel-circuits:parallel}{branche} deux est
tendue entre les mêmes \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}~: $U_G = \qty{6}{V}$ (loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions}
\omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:12}
$I_R = 0.6 - 0.45 = \qty{0.15}{A}$ (loi des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}).
$U_R = U_Q = \qty{2.3}{V}$ (loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}).
$U_P = 6 - 2.3 = \qty{3.7}{V}$ (loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série}). Une
addition condamnable~: «~$I_R = \qty{0.2}{A}$~» --- les comptes du
\omterm{def:g5:series-parallel-circuits:parallel}{nœud} donneraient $0.45 + 0.2 = 0.65 \neq 0.6$~; ou
«~$U_R = \qty{2.6}{V}$~» --- des \omterm{def:g5:series-parallel-circuits:parallel}{branches} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} n'ont pas le
droit d'être en désaccord.
\end{solution}

\begin{solution}{pb:g8:circuit-laws:1}
\textbf{1.} \omterm{def:g4:series-circuits:series}{En série}. \omterm{def:g4:series-circuits:series}{En série}, les deux lampes s'opposent à la
marche l'une après l'autre~: le courant de la boucle tombe bien en
dessous des \qty{0.4}{A} d'une lampe seule --- les \qty{0.2}{A}
mesurés collent. (La dérivation \emph{ajouterait} deux \omterm{def:g5:series-parallel-circuits:parallel}{branches}
entières.)
\textbf{2.} \omterm{def:g5:series-parallel-circuits:parallel}{En dérivation}~: deux \omterm{def:g5:series-parallel-circuits:parallel}{branches} d'environ \qty{0.4}{A}
chacune --- à peu près \qty{0.8}{A} tirés de la \omterm{def:g3:first-electric-circuit:battery}{pile}.
\textbf{3.} La loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série} partage les \qty{6}{V}
entre deux lampes identiques~: environ \qty{3}{V} chacune.
\textbf{4.} \omterm{def:g5:series-parallel-circuits:parallel}{En dérivation} --- chaque lampe serait aux \omterm{def:g3:first-electric-circuit:battery}{bornes} des
\qty{6}{V} entiers. Par le judas, la paire \omterm{def:g4:series-circuits:series}{en série} luit faiblement~:
chacune est sous-alimentée, à la moitié de sa poussée habituelle.
\textbf{5.} $I_Z = 0.5 - 0.35 = \qty{0.15}{A}$ (loi des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds})~;
$I_X = \qty{0.5}{A}$ --- X se tient sur la route non divisée (loi des
courants \omterm{def:g4:series-circuits:series}{en série}).
\textbf{6.} $U_{\text{paire}} = 6 - 2.8 = \qty{3.2}{V}$ (loi de la
boucle)~; $U_Y = U_Z = \qty{3.2}{V}$ (loi de la dérivation).
\textbf{7.} Même poussée, marches différentes~: les lampes s'opposent
inégalement au courant --- Z, qui en porte moins sous la même
\omterm{def:g8:voltage-voltmeter:voltage}{tension}, est l'opposant le plus fort (la route la plus dure).
\textbf{8.} Baisse~: perdre la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de Y supprime une route~; la
route de Z restante s'oppose plus que la paire ne le faisait
ensemble, si bien que la \omterm{def:g3:first-electric-circuit:battery}{pile} entraîne moins de courant total.
\textbf{9.} Première condamnation, loi de la boucle~:
$2.5 + 4.1 = \qty{6.6}{V}$ dépensés depuis une bourse de \qty{6}{V}.
Seconde condamnation, loi des \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}~: \qty{0.3}{A} entrants,
$0.2 + 0.15 = \qty{0.35}{A}$ sortants.
\textbf{10.} $6 - 2.5 = \qty{3.5}{V}$.
\textbf{11.} $0.2 + 0.15 = \qty{0.35}{A}$.
\textbf{12.} Parce que les quatre lois surdéterminent tout circuit
honnête~: chaque lecture doit s'accorder aux sommes des autres, si
bien qu'un nombre inventé doit mentir à au moins une loi --- et se
trouve condamné par la seule arithmétique.
\textbf{13.} Par exemple~: «~Une route, une marche --- la loi des
courants \omterm{def:g4:series-circuits:series}{en série}. À chaque embranchement, les marches
s'équilibrent, autant qui entrent que qui sortent --- la loi des
\omterm{def:g5:series-parallel-circuits:parallel}{nœuds}. Le long de toute boucle, la chute d'eau de la \omterm{def:g3:first-electric-circuit:battery}{pile} est
dépensée jusqu'à la dernière goutte par les travailleurs --- la loi
des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g4:series-circuits:series}{en série}. Et des \omterm{def:g5:series-parallel-circuits:parallel}{branches} suspendues entre les deux
mêmes \omterm{def:g5:series-parallel-circuits:parallel}{nœuds} boivent à la même hauteur --- la loi des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}.~»
\end{solution}
