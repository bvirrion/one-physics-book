\section*{Chapitre \ref{ch:g8:sound-pitch-loudness} --- Le son~: vitesse, hauteur, intensité sonore}

\begin{solution}{exo:g8:sound-pitch-loudness:1}
La \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence}~: le nombre d'oscillations complètes par seconde, en
\omterm{def:g8:sound-pitch-loudness:frequency}{hertz} (\unit{Hz}). Le la de concert~: \qty{440}{Hz}~; l'octave
au-dessus~: \qty{880}{Hz} --- le double.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:2}
$4$ oscillations en \qty{0.02}{s}~: \qty{200}{Hz}~; $16$
oscillations~: \qty{800}{Hz} --- le second sonne plus aigu (de deux
octaves, en fait).
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:3}
D'environ \qty{20}{Hz} à \qty{20000}{Hz}. En dessous~: les \omterm{prop:g8:sound-pitch-loudness:range}{infrasons}
--- les grondements à longue portée des éléphants. Au-dessus~: les
\omterm{prop:g8:sound-pitch-loudness:range}{ultrasons} --- les clics de chasse de la chauve-souris.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:4}
Plus fort, même note~: les crêtes grandissent, l'espacement ne change
pas. Plus aigu, même intensité sonore~: les crêtes se serrent, la
hauteur des crêtes ne change pas. Deux réglages indépendants.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:5}
Chacun élève la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence}~: plus courte, plus tendue et plus fine, une
corde tremble davantage de fois par seconde --- $f$ plus grand, note
plus aiguë.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:6}
Les feuilles \qty{10}{dB}~; la circulation \qty{80}{dB}~; le premier
rang \qty{110}{dB}~; la douleur près de \qty{120}{dB}. La bande des
dommages s'ouvre vers \qty{85}{dB}.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:7}
La hauteur du sifflet se tient juste au-dessus de \qty{20000}{Hz} ---
hors de la fenêtre du maître, dans celle du chien. Cassé pour une
paire d'oreilles, perçant pour l'autre~: les fenêtres diffèrent selon
l'espèce.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:8}
Que la \omterm{def:g7:motion-average-speed:formula}{vitesse} est une et la même pour toute hauteur et toute
intensité sonore~: des \omterm{def:g7:motion-average-speed:formula}{vitesses} différentes étaleraient les accords
lointains en arpèges et feraient traîner le doux derrière le fort ---
le concert dans le parc réfute les deux tous les soirs.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:9}
$50000 \times 0.01 = 500$ oscillations par clic. Une peinture fine
exige des pinceaux fins~: seules des \omterm{def:g3:sound-around-us:vibration}{vibrations} aux oscillations très
courtes --- $f$ très élevé --- peuvent palper de petits détails dans
leurs \omterm{ex:g4:how-sound-travels:echo}{échos}~; les grondements grossiers de basse \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} passent sur
un papillon de nuit sans le remarquer.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:10}
Les deux touches se tiennent dans la fenêtre --- \qty{27}{Hz} juste
au-dessus du plancher de \qty{20}{Hz}, \qty{4200}{Hz} confortablement
sous le plafond. Mais l'âge ronge la fenêtre par le \emph{haut}~:
l'oreille de l'accordeur s'émousse d'abord dans l'aigu, si bien que
l'analyseur est appelé pour les octaves les plus hautes avant les plus
basses.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:11}
Une croissance par multiplication par dix cachée dans des pas égaux
--- chaque poignée de barreaux multiplie la \omterm{def:g2:pushes-and-pulls:force}{force} du \omterm{def:g3:sound-around-us:sound}{son} au lieu de
l'augmenter d'une quantité fixe. Ainsi «~un peu plus fort~» sur
l'échelle est \emph{beaucoup} plus fort pour l'oreille~: les budgets
s'effondrent d'\omterm{ex:g2:measuring-time:clock}{heures} en minutes sur ce qui se lit comme un petit
tour de bouton.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:12}
Contrôle du piano~: l'analyseur à côté de la corde, comparer son
affichage à $440$. \omterm{def:g8:intensity-ammeter:intensity}{Intensité} sonore au dernier rang~: s'y tenir avec
l'affichage en \omterm{ex:g8:sound-pitch-loudness:decibels}{décibels} pendant l'essai de batterie. Ventilation~:
braquer l'affichage du \omterm{prop:g8:colors-spectra:mixture}{spectre} de l'analyseur sur le bas de l'étendue
et guetter une pointe sous \qty{20}{Hz}. Le moins fiable~: le verdict
sur les \omterm{prop:g8:sound-pitch-loudness:range}{infrasons} --- les microphones de téléphone sont bâtis pour les
voix et deviennent sourds près du plancher de la fenêtre~; un écran
silencieux ne prouve donc pas grand-chose là en bas.
\end{solution}

\begin{solution}{pb:g8:sound-pitch-loudness:1}
\textbf{1.} Le piano est trop bas --- chaque la tremble huit
oscillations par seconde de trop peu. Le réglage de la hauteur~: la
clé d'un accordeur, non un amplificateur.
\textbf{2.} Les \qty{41}{Hz} se tiennent juste au-dessus du plancher
de \qty{20}{Hz} de la fenêtre~: entendus, profondément --- et avec des
oscillations aussi lentes, le tambour de la poitrine répond de
concert~: senti autant qu'entendu.
\textbf{3.} Même hauteur (espacement égal, \qty{440}{Hz}), trois fois
l'\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude}~: le chanteur chante la note du diapason, bien plus fort.
\textbf{4.} Seules les oreilles les plus jeunes --- le plafond de la
fenêtre, près de \qty{20000}{Hz}, appartient aux enfants~; la fenêtre
de la plupart des adultes s'est déjà abaissée, et aucun auditeur ne
certifie \qty{22000}{Hz}.
\textbf{5.} Le set rock (\qty{100}{dB}) et le final (\qty{110}{dB})
se tiennent tous deux au-dessus de la ligne des dommages de
\qty{85}{dB}~: territoire de quelques minutes seulement~; le set
acoustique passe.
\textbf{6.} Environ \qty{80}{dB} derrière les bouchons --- de nouveau
sous la ligne des dommages~: une soirée d'exposition rendue
survivable, et c'est pourquoi les avisés en portent toujours.
\textbf{7.} Les coquilles qui s'étendent~: chaque resserrement se
partage sur une coquille plus vaste à mesure qu'on s'éloigne ---
l'intensité sonore (l'\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude}) faiblit. La hauteur voyage
inchangée~: la distance ne tourne qu'un seul réglage.
\textbf{8.} Jouer doucement ne tourne que le réglage de l'\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude}
--- la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} de chaque note, aiguë comme grave, reste exactement
en place. (Son «~plus terne~» est un tour que joue l'oreille
elle-même à faible \omterm{def:g8:intensity-ammeter:intensity}{intensité}, non la physique des cordes.)
\textbf{9.} $680 \div 340 = \qty{2}{s}$ entre l'éclat et la
détonation. Deux coups par seconde~: \qty{2}{Hz} --- bien en dessous
du plancher~: senti dans la poitrine, jamais entendu comme un \omterm{def:g3:sound-around-us:sound}{son}
tenu.
\textbf{10.} Les oscillations les plus graves de la basse~: près du
plancher de la fenêtre et en dessous, précisément là où le microphone
bâti pour la voix devient sourd --- le grondement était réel~; le
téléphone ne l'a jamais attrapé.
\textbf{11.} Par exemple~: «~La hauteur est la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} --- le
compte des oscillations par seconde, et toute l'affaire de
l'accordeur. L'intensité sonore est l'\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude} --- la taille des
oscillations, et tout le registre de l'infirmière. Et aucun des deux
réglages ne touche à la \omterm{def:g7:motion-average-speed:formula}{vitesse}~: toutes les notes de ce soir ont
traversé la salle aux mêmes \qty{340}{m/s}.~»
\textbf{12.} La proposition sur la fenêtre~: le répulsif est accordé
au-dessus de \qty{20000}{Hz} --- hors de toute fenêtre humaine, dans
celle de la souris. Un seul tremblement, deux publics, aucune
contradiction.
\end{solution}
