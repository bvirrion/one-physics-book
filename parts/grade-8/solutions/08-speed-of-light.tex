\section*{Chapter \ref{ch:g8:speed-of-light} --- The Speed of Light; Distances in the Universe}

\begin{solution}{exo:g8:speed-of-light:1}
$c = \qty{300000}{km/s} = 3 \times 10^{8}\ \unit{m/s}$. The
hilltop lanterns measured only human reflexes: \omterm{def:g1:light-and-shadows:source}{light} crossed
the hills in millionths of a second, far beneath any hand's
timing.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:2}
Jupiter's moon eclipsed on a punctual schedule --- a sky
clock. The \emph{news} of each eclipse arrived late whenever
the Earth stood far from Jupiter, early when near. The
lateness measured \omterm{def:g1:light-and-shadows:source}{light}'s travel time across the width of the
Earth's \omterm{def:g6:sun-earth-moon:axisorbit}{orbit} --- and so \omterm{def:g1:light-and-shadows:source}{light}'s \omterm{def:g7:motion-average-speed:formula}{speed}.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:3}
\omterm{def:g2:moon-in-the-sky:moon}{Moon}: $384000 \div 300000 \approx \qty{1.3}{s}$. Sun:
$150000000 \div 300000 = \qty{500}{s}$ --- eight minutes
twenty. Sunshine always shows the Sun as it was eight minutes
ago: old news by construction.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:4}
A distance: the stretch \omterm{def:g1:light-and-shadows:source}{light} covers in one year --- about
$9.5 \times 10^{12}$ km.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:5}
Four years. Answers vary --- four years ago you were several
grades back down this very book.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:6}
At eleven-ish minutes one way, a steering correction answers a
picture at least twenty-two minutes stale --- the rover would
be in the crevasse before the joystick's twitch arrived. Hence
plans, not joysticks.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:7}
\omterm{def:g2:moon-in-the-sky:moon}{Moon}: about \qty{1.3}{s}. Sun: eight minutes. Nearest \omterm{def:g4:solar-system:star}{star}:
four years. Neighbor galaxy: two and a half million years.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:8}
\omterm{def:g1:light-and-shadows:source}{Light}. The \omterm{def:g7:motion-average-speed:formula}{speed} $c$ proved steadier than any metal bar or
national ruler --- so the ruler is now derived from the \omterm{def:g1:light-and-shadows:source}{light},
not the \omterm{def:g1:light-and-shadows:source}{light} measured by the ruler.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:9}
Round trip: $300000 \times 2.6 = 780000$ km; the \omterm{def:g2:moon-in-the-sky:moon}{Moon} stands
at half --- \qty{390000}{km}. The sonar rule carries over
whole: every \omterm{ex:g4:how-sound-travels:echo}{echo} pays for the round trip, halve before
believing.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:10}
Lamp: $3 \div (3 \times 10^{8}) = 10^{-8}$ s --- a hundredth
of a millionth of a second: even the room is an archive, too
shallow to notice. Then the Sun (eight minutes), the \omterm{def:g4:solar-system:star}{star} (a
century), the galaxy (two and a half million years) --- the
deeper you look, the older the page.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:11}
``Last Tuesday, the \omterm{def:g1:light-and-shadows:source}{light} of a \omterm{def:g4:solar-system:star}{star}'s explosion \emph{reached
us} --- the explosion itself happened some $8000$ years ago,
and its news has been traveling ever since.''
\end{solution}

\begin{solution}{exo:g8:speed-of-light:12}
$3 \times 10^{8}\ \text{km} \div 1000\ \text{s} = 3 \times
10^{5}$ \unit{km/s} --- the modern value on the nose (the
historical estimate, with rougher \omterm{def:g6:sun-earth-moon:axisorbit}{orbit} figures, landed
within about a quarter of it: for the sixteen-hundreds, an
astonishing first answer).
\end{solution}

\begin{solution}{pb:g8:speed-of-light:1}
\textbf{1.} About \qty{1.3}{s} after you speak; the soonest
acknowledgment returns after the round trip, about
\qty{2.6}{s}.
\textbf{2.} $36000 \div 300000 = \qty{0.12}{s}$ one way; a
question-and-answer rides up and down twice --- near half a
second of hesitation, just at the ear's threshold.
\textbf{3.} ``The flare erupted eight minutes twenty before
our screens lit --- we watch it as it \emph{was}.''
\textbf{4.} The \omterm{def:g2:moon-in-the-sky:moon}{Moon} link: \qty{1.3}{s} against \qty{0.12}{s}
--- roughly ten times the allowance.
\textbf{5.} $1.8 \times 10^{8} \div (3 \times 10^{5}) = 600$
s: ten minutes one way.
\textbf{6.} The round trip is twenty minutes: the rescue
picture is ten minutes old and the rescue command ten minutes
late --- the crevasse wins by nineteen minutes. The rover
must carry its own reflexes: on-board hazard-stopping.
\textbf{7.} Last bit must arrive by 21:00; it leaves Earth at
$21{:}00 - 10$ min $= 20{:}50$; transmission began three
minutes earlier: start at 20:47.
\textbf{8.} The lag doubles to about twenty-one minutes one
way: \omterm{def:g4:solar-system:planet}{planets} have no ``the'' distance --- both worlds \omterm{def:g6:sun-earth-moon:axisorbit}{orbit},
and every conversation is scheduled against a moving target.
\textbf{9.} $300000 \times 86400 \approx 2.6 \times 10^{10}$
km --- twenty-six thousand million kilometres; a
question-and-answer takes two full days.
\textbf{10.} \omterm{def:g1:light-and-shadows:source}{Light} reaches the \omterm{def:g4:solar-system:star}{star} in about four years; the
soonest reply lands about eight years after tonight's
transmission.
\textbf{11.} ``This \omterm{def:g1:light-and-shadows:source}{light} left its galaxy two and a half
million years ago: the snapshot broadcasts the Earth of the
earliest stone tools --- our deepest naked-eye page of the
archive.''
\textbf{12.} For example: ``Nothing is instantaneous ---
every signal, ours included, travels at best at $c$.
Joysticks lose to lag beyond the \omterm{def:g2:moon-in-the-sky:moon}{Moon}: distant machines must
think for themselves. And every telescope is an archive
reader --- the farther the object, the older the news, with
no fresher edition printable.''
\end{solution}
