\section*{Hoofdstuk \ref{ch:g8:ohms-law} --- Weerstand en de wet van Ohm}

\begin{solution}{exo:g8:ohms-law:1}
Hoe sterk een onderdeel zich tegen de stroom verzet;
\omterm{def:g6:measurement-in-science:measuring}{eenheid} de \omterm{def:g8:ohms-law:resistance}{ohm}, symbool $\Omega$. $U = R I$; \
$I = U/R$; \ $R = U/I$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:2}
$I = 6 \div 30 = \qty{0.2}{A}$; \ $U = 30 \times 0.5 = \qty{15}{V}$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:3}
$R = 4.5 \div 0.09 = 50\,\Omega$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:4}
Een rechte lijn door de oorsprong --- evenredigheid. Een steilere lijn
betekent meer \omterm{def:g8:voltage-voltmeter:voltage}{volt} per \omterm{def:g8:intensity-ammeter:intensity}{ampère}: een grotere
\omterm{def:g8:ohms-law:resistance}{weerstand}.
\end{solution}

\begin{solution}{exo:g8:ohms-law:5}
$3.0 \div 0.20 = 15$; $6.0 \div 0.40 = 15$. Dat de verhouding over de
rijen heen gelijk blijft, \emph{is} de wet: één constante $R$ die elke
aflezing bedient, is wat van ``de \omterm{def:g8:ohms-law:resistance}{weerstand}'' een eigenschap
van het onderdeel maakt.
\end{solution}

\begin{solution}{exo:g8:ohms-law:6}
De kleinste \omterm{def:g8:ohms-law:resistance}{weerstand} drinkt het meest: $230 \div 25 =
\qty{9.2}{A}$ tegen $230 \div 35 \approx \qty{6.6}{A}$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:7}
Lampen \omterm{def:g4:series-circuits:series}{in serie} delen één $I$; de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} over elke lamp is
$U = R I$ met dezelfde $I$ --- dus vermenigvuldigt de grotere $R$ zich
tot de grotere $U$: de sterkste tegenstander neemt het grootste deel.
\end{solution}

\begin{solution}{exo:g8:ohms-law:8}
\omterm{def:g7:short-circuits-safety:device}{Kortsluiting}: $R$ bijna nul, dus ontploft $I = U/R$ --- de
stormloop is een deling door bijna niets. Open
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}: $R$ voorbij elk \omterm{def:g6:measurement-in-science:measuring}{meten}, dus zakt $I = U/R$
in tot nul --- de volledige blokkade.
\end{solution}

\begin{solution}{exo:g8:ohms-law:9}
De \omterm{def:g8:ohms-law:resistance}{weerstand} moet $9 - 2 = \qty{7}{V}$ opnemen bij
\qty{0.02}{A}: $R = 7 \div 0.02 = 350\,\Omega$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:10}
Eerste punt: $R = 1 \div 0.25 = 4\,\Omega$; tweede: $4 \div 0.5 =
8\,\Omega$ --- geen enkele waarde past bij beide. Het verhaal: de
groeiende stroom verhit de gloeidraad, en heet metaal biedt
meer verzet; het portret buigt omdat het onderdeel verandert terwijl
het werkt.
\end{solution}

\begin{solution}{exo:g8:ohms-law:11}
Eén lus, één $I$: de twee vervallen tellen op tot de duw van de
\omterm{def:g3:first-electric-circuit:battery}{batterij}, $6 = 10 I + 20 I = 30 I$, dus $I = \qty{0.2}{A}$. Dan
$U_{10} = 10 \times 0.2 = \qty{2}{V}$ en $U_{20} = 20 \times 0.2 =
\qty{4}{V}$ --- samen \qty{6}{V}, zoals de luswet eist.
\end{solution}

\begin{solution}{exo:g8:ohms-law:12}
Elke \omterm{def:g5:series-parallel-circuits:parallel}{tak} overspant \qty{6}{V}: $I_{10} = 6 \div 10 =
\qty{0.6}{A}$, $I_{20} = 6 \div 20 = \qty{0.3}{A}$; samen
$\qty{0.9}{A}$. De vervanger: $R = 6 \div 0.9 \approx 6.7\,\Omega$ ---
\emph{minder} dan elk van beide leden. In wegentaal: twee wegen tussen
dezelfde steden dragen meer verkeer dan elke weg alleen; elke
toegevoegde parallelle weg vergemakkelijkt de doorgang, dus zakt het
verzet van het team onder dat van zijn zwakste lid.
\end{solution}

\begin{solution}{pb:g8:ohms-law:1}
\textbf{1.} Voeding, \omterm{def:g8:intensity-ammeter:ammeter}{ampèremeter} en onderdeel in één serielus
--- het tolhuisje op de weg; de \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} over het onderdeel
--- de landmeter ernaast.
\textbf{2.} $R = 4.5 \div 0.15 = 30\,\Omega$.
\textbf{3.} Vijf procent van $33$ is ongeveer $1.65$: aanvaardbaar van
$31.35$ tot $34.65\,\Omega$. De $30\,\Omega$ van monster één valt
erbuiten: afgekeurd.
\textbf{4.} $I = 4.5 \div 150 = \qty{0.03}{A} = \qty{30.0}{mA}$.
\textbf{5.} Elke rij geeft $R = U/I = 30\,\Omega$ ($1.5 \div 0.05$,
$3.0 \div 0.10$, \dots): constante verhouding, portret als rechte lijn
--- één waarde past bij alles: $30\,\Omega$.
\textbf{6.} De verhoudingen lopen $5$, $6.7$, $8.3$, $10\,\Omega$: bij
elke stap hoger --- de \omterm{def:g8:ohms-law:resistance}{weerstand} groeit met de stroom mee;
er bestaat geen enkele $R$.
\textbf{7.} Terwijl de stroom groeit, verhit de gloeidraad tot
hij gloeit, en heet metaal biedt meer verzet --- het portret buigt
omhoog. De rechtelijntest van de bank laat dus precies de onderdelen
toe waarvan $R$ blijft zitten: hij is, van bouw, een weerstanddetector.
\textbf{8.} De bepaling over de constante \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}: de wet
van Ohm belooft evenredigheid \emph{bij constante
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur}} --- verwarm een onderdeel midden in zijn portret
en zelfs een eerlijke \omterm{def:g8:ohms-law:resistance}{weerstand} gaat driften.
\textbf{9.} $R = 12 \div 0.03 = 400\,\Omega$.
\textbf{10.} $U = R I = 25 \times 9.2 = \qty{230}{V}$ --- precies het
lichtnet: de spiraal trekt exact wat de wet van Ohm haar toebedeelt, en
de marge van de \omterm{def:g7:short-circuits-safety:fuse}{zekering} werkt zoals ontworpen. Er is niets
``maar''; alles is rekenkunde.
\textbf{11.} De doorverbinder is geboren om verbindingsdraad te zijn
--- de vlakke weg, die geen duw uitgeeft. Verkeerd gebruikt, over een
voeding gelegd, maken zijn bijna nul \omterm{def:g8:ohms-law:resistance}{ohm} hem tot de perfecte
\omterm{def:g7:short-circuits-safety:device}{kortsluiting}: de schurk was altijd al gewoon een heel goede
draad op de verkeerde plek.
\textbf{12.} Bijvoorbeeld: ``De wet: $U = R I$, duw is verzet maal
mars. Haar portret: een rechte lijn door de oorsprong, steiler voor
sterkere tegenstanders. En alleen een onderdeel bij constante
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur} mag haar ondertekenen --- \omterm{def:g5:heat-and-insulation:heat}{warmte}
herschrijft de \omterm{def:g8:ohms-law:resistance}{weerstand}, en koortsig opgenomen portretten
bewijzen niets.''
\end{solution}
