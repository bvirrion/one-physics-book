\section*{Hoofdstuk \ref{ch:g8:sound-pitch-loudness} --- Geluid: snelheid, toonhoogte, luidheid}

\begin{solution}{exo:g8:sound-pitch-loudness:1}
\omterm{def:g8:sound-pitch-loudness:frequency}{Frequentie}: het aantal volledige zwaaien per \omterm{def:g2:measuring-time:second}{seconde}, in
\omterm{def:g8:sound-pitch-loudness:frequency}{hertz} (\unit{Hz}). Concert-a: \qty{440}{Hz}; het octaaf
erboven: \qty{880}{Hz} --- verdubbeld.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:2}
$4$ zwaaien in \qty{0.02}{s}: \qty{200}{Hz}; $16$ zwaaien:
\qty{800}{Hz} --- de tweede klinkt hoger (twee octaven,
zelfs).
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:3}
Ongeveer \qty{20}{Hz} tot \qty{20000}{Hz}. Eronder:
\omterm{prop:g8:sound-pitch-loudness:range}{infrageluid} --- het gerommel van olifanten over grote afstand.
Erboven: \omterm{prop:g8:sound-pitch-loudness:range}{ultrageluid} --- de jachtklikjes van de vleermuis.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:4}
Luider, dezelfde noot: de pieken worden hoger, de onderlinge afstand
verandert niet. Hoger, even luid: de pieken dringen dichter op elkaar,
de hoogte verandert niet. Twee onafhankelijke knoppen.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:5}
Elk verhoogt de \omterm{def:g8:sound-pitch-loudness:frequency}{frequentie}: kortere, strakkere en dunnere
snaren trillen allemaal vaker per \omterm{def:g2:measuring-time:second}{seconde} --- hogere $f$, hogere
toonhoogte.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:6}
Bladeren \qty{10}{dB}; verkeer \qty{80}{dB}; eerste rij \qty{110}{dB};
pijn rond \qty{120}{dB}. De schadeband opent rond \qty{85}{dB}.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:7}
De toonhoogte van het fluitje ligt net boven \qty{20000}{Hz} --- buiten
het venster van het baasje, binnen dat van de hond. Kapot voor het ene
paar oren, snerpend voor het andere: vensters verschillen per soort.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:8}
Dat de \omterm{def:g7:motion-average-speed:formula}{snelheid} voor elke toonhoogte en elke luidheid één en
dezelfde is: verschillende snelheden zouden verre akkoorden tot
arpeggio's uitsmeren en het zachte achter het luide aan laten komen ---
het parkconcert weerlegt allebei elke avond.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:9}
$50000 \times 0.01 = 500$ zwaaien per klik. Fijn schilderen vraagt
fijne penselen: alleen \omterm{def:g3:sound-around-us:vibration}{trillingen} met heel korte zwaaien --- heel
hoge $f$ --- kunnen kleine details in hun \omterm{ex:g4:how-sound-travels:echo}{echo}'s betasten; grove
laagfrequente dreunen spoelen ongemerkt over een mot heen.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:10}
Beide toetsen liggen binnen het venster --- \qty{27}{Hz} net boven de
vloer van \qty{20}{Hz}, \qty{4200}{Hz} comfortabel onder het plafond.
Maar de leeftijd knabbelt het venster van \emph{boven} af: het oor van
de stemmer wordt het eerst dof in de hoogte, dus wordt de analyser
eerder voor de hoogste octaven geroepen dan voor de laagste.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:11}
Tienvoudige groei verstopt in gelijke stappen --- elke paar sporten
vermenigvuldigen de sterkte van het \omterm{def:g3:sound-around-us:sound}{geluid}, ze tellen er niet
bij op. Dus ``een beetje luider'' op de ladder is \emph{veel} sterker
voor het oor: budgetten storten in van \omterm{ex:g2:measuring-time:clock}{uren} naar minuten over wat
als een klein knopje draaien leest.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:12}
Pianocontrole: analyser naast de snaar, vergelijk de aflezing met
$440$. Luidheid op de achterste rij: ga daar tijdens de drumtest staan
met de decibelaflezing. Ventilatie: richt de
spectrumweergave van de analyser op het lage einde en zoek naar
een piek onder \qty{20}{Hz}. Het minst betrouwbaar: het oordeel over
\omterm{prop:g8:sound-pitch-loudness:range}{infrageluid} --- telefoonmicrofoons zijn voor stemmen gebouwd en
worden doof bij de vloer van het venster, dus een stil scherm bewijst
daar beneden weinig.
\end{solution}

\begin{solution}{pb:g8:sound-pitch-loudness:1}
\textbf{1.} De piano staat te laag --- elke a trilt acht zwaaien per
\omterm{def:g2:measuring-time:second}{seconde} te weinig. De toonhoogteknop: de sleutel van een stemmer, geen
versterker.
\textbf{2.} \qty{41}{Hz} ligt net boven de vloer van \qty{20}{Hz} van
het venster: gehoord, diep --- en bij zo trage zwaaien antwoordt de
eigen trommel van de borst mee: net zo goed gevoeld als gehoord.
\textbf{3.} Dezelfde toonhoogte (gelijke onderlinge afstand,
\qty{440}{Hz}), drie keer de \omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude}: de zanger zingt de noot
van de stemvork, veel luider.
\textbf{4.} Alleen de jongste oren --- het plafond van het venster rond
\qty{20000}{Hz} hoort bij kinderen; bij de meeste volwassenen is het
venster al gezakt, en niemand in het publiek bevestigt \qty{22000}{Hz}.
\textbf{5.} De rockset (\qty{100}{dB}) en de finale (\qty{110}{dB})
staan allebei boven de schadegrens van \qty{85}{dB}: gebied van slechts
minuten; de akoestische set komt erdoor.
\textbf{6.} Ongeveer \qty{80}{dB} achter de doppen --- terug onder de
schadegrens: een avond blootstelling overleefbaar gemaakt, en daarom
dragen de verstandigen ze altijd.
\textbf{7.} De uitdijende schillen: elke samenpersing wordt met de
afstand over een grotere schil gedeeld --- de luidheid
(\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude}) vervaagt. De toonhoogte rijdt onveranderd mee: de
afstand draait maar één knop.
\textbf{8.} Zacht spelen draait alleen aan de knop van de
\omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude} --- de \omterm{def:g8:sound-pitch-loudness:frequency}{frequentie} van elke noot, hoog en
laag, blijft precies zitten. (Zijn ``doffer'' is een truc van het oor
zelf bij lage luidheid, niet de natuurkunde van de snaren.)
\textbf{9.} $680 \div 340 = \qty{2}{s}$ tussen flits en knal. Twee
dreunen per \omterm{def:g2:measuring-time:second}{seconde}: \qty{2}{Hz} --- ver onder de vloer: in de borst
gevoeld, nooit als toon gehoord.
\textbf{10.} De laagste zwaaien van de bas: bij de vloer van het venster
en eronder, precies waar de voor stemmen gebouwde microfoon doof wordt
--- het gerommel was echt; de telefoon ving het nooit op.
\textbf{11.} Bijvoorbeeld: ``Toonhoogte is \omterm{def:g8:sound-pitch-loudness:frequency}{frequentie} --- de
telling van de zwaaien per \omterm{def:g2:measuring-time:second}{seconde}, en het hele vak van de stemmer.
Luidheid is \omterm{prop:g8:sound-pitch-loudness:loudness}{amplitude} --- de grootte van de zwaaien, het hele
grootboek van de verpleegkundige. En geen van beide knoppen raakt de
\omterm{def:g7:motion-average-speed:formula}{snelheid} aan: elke noot van vanavond stak de zaal over met
dezelfde \qty{340}{m/s}.''
\textbf{12.} Het venstervoorstel: de verjager zit boven \qty{20000}{Hz}
--- buiten elk menselijk venster, binnen dat van de muis. Eén \omterm{def:g3:sound-around-us:vibration}{trilling},
twee publieken, geen tegenspraak.
\end{solution}
