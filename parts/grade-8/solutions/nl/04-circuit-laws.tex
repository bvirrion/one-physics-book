\section*{Hoofdstuk \ref{ch:g8:circuit-laws} --- Stroom- en spanningswetten in schakelingen}

\begin{solution}{exo:g8:circuit-laws:1}
Stroomwet voor serie: één lus, overal één
\omterm{def:g8:intensity-ammeter:intensity}{stroomsterkte}. Knooppuntwet: stromen erin is
stromen eruit. Spanningswet voor serie: de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} van de
\omterm{def:g3:first-electric-circuit:battery}{batterij} is gelijk aan de som van de delen van de apparaten rond
de lus. Spanningswet voor \omterm{def:g5:series-parallel-circuits:parallel}{parallel}: \omterm{def:g5:series-parallel-circuits:parallel}{takken} die dezelfde
\omterm{def:g5:series-parallel-circuits:parallel}{knooppunten} overspannen, delen één en dezelfde
\omterm{def:g8:voltage-voltmeter:voltage}{spanning}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:2}
$0.8 - 0.55 = \qty{0.25}{A}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:3}
$9 - 2.5 - 3.8 = \qty{2.7}{V}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:4}
Elk \qty{12}{V} --- de spanningswet voor \omterm{def:g5:series-parallel-circuits:parallel}{parallel}. Dezelfde wet zet op
elk stopcontact van de parallelle huisbedrading dezelfde \qty{230}{V}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:5}
Bijvoorbeeld $I_C$: zodra $I_{\text{batterij}}$ en $I_B$ bekend zijn,
legt de knooppuntwet $I_C = \qty{0.2}{A}$ vast --- een meetinstrument
die daarheen wordt gestuurd, kan alleen nog bevestigen (net zo $U_C$,
vastgelegd door de spanningswet voor \omterm{def:g5:series-parallel-circuits:parallel}{parallel} zodra $U_B$ bekend is).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:6}
De \qty{2.7}{V}-lamp: bij gelijke stroom markeert het grotere
spanningsdeel de sterkste tegenstander. Het volgende hoofdstuk
meet dat verzet --- de weerstand.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:7}
$2.2 + 1.9 + 1.4 = \qty{5.5}{V}$ tegen een beurs van \qty{6}{V}: een
halve \omterm{def:g8:voltage-voltmeter:voltage}{volt} zoek --- meer dan de eerlijke spreiding van een
meetinstrument bij zulke aflezingen; één schaal is verkeerd afgelezen (of een
deel is niet gemeten).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:8}
$0.4 + 0.3 + 0.3 = \qty{1.0}{A}$ eruit tegen \qty{0.9}{A} erin: de
splitsing ``schept'' \qty{0.1}{A} --- veroordeeld door de
knooppuntwet.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:9}
De uitgaven van de lus moeten optellen tot de \qty{4.5}{V} van de
\omterm{def:g3:first-electric-circuit:battery}{batterij}: open \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} \qty{4.5}{V} $+$
stilstaande lamp \qty{0}{V} $= 4.5$ --- in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}. (Na de klik
ruilen de delen: \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} bijna nul, lamp bijna \qty{4.5}{V}
--- opnieuw in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}.)
\end{solution}

\begin{solution}{exo:g8:circuit-laws:10}
Ervoor: $3 \times 1.5 = \qty{4.5}{V}$. Erna: $4.5 - 1.5 =
\qty{3.0}{V}$ --- de omgekeerde cel trekt af. De lamp dooft van haar
\qty{4.5}{V}-gloed naar een \qty{3}{V}-gloed.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:11}
\omterm{def:g5:series-parallel-circuits:parallel}{Tak} één overspant de \omterm{def:g3:first-electric-circuit:battery}{batterij}: $U_E + U_F =
\qty{6}{V}$ (spanningswet voor serie langs de \omterm{def:g5:series-parallel-circuits:parallel}{tak}), dus
$U_F = 6 - 2.2 = \qty{3.8}{V}$. \omterm{def:g5:series-parallel-circuits:parallel}{Tak} twee overspant dezelfde
\omterm{def:g5:series-parallel-circuits:parallel}{knooppunten}: $U_G = \qty{6}{V}$ (spanningswet voor \omterm{def:g5:series-parallel-circuits:parallel}{parallel}).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:12}
$I_R = 0.6 - 0.45 = \qty{0.15}{A}$ (knooppuntwet). $U_R = U_Q
= \qty{2.3}{V}$ (spanningswet voor \omterm{def:g5:series-parallel-circuits:parallel}{parallel}). $U_P = 6 - 2.3 =
\qty{3.7}{V}$ (spanningswet voor serie). Een veroordeelbare extra:
``$I_R = \qty{0.2}{A}$'' --- de boeken van het \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} zouden
$0.45 + 0.2 = 0.65 \neq 0.6$ tonen; of ``$U_R = \qty{2.6}{V}$'' ---
parallelle \omterm{def:g5:series-parallel-circuits:parallel}{takken} mogen het niet oneens zijn.
\end{solution}

\begin{solution}{pb:g8:circuit-laws:1}
\textbf{1.} Serie. \omterm{def:g4:series-circuits:series}{In serie} verzetten de twee lampen zich na
elkaar tegen de mars: de stroom in de lus valt ver onder de
\qty{0.4}{A} van één lamp --- de gemeten \qty{0.2}{A} past. (\omterm{def:g5:series-parallel-circuits:parallel}{Parallel}
zou twee volle \omterm{def:g5:series-parallel-circuits:parallel}{takken} \emph{optellen}.)
\textbf{2.} \omterm{def:g5:series-parallel-circuits:parallel}{Parallel}: twee \omterm{def:g5:series-parallel-circuits:parallel}{takken} van elk ongeveer
\qty{0.4}{A} --- ruwweg \qty{0.8}{A} uit de \omterm{def:g3:first-electric-circuit:battery}{batterij}.
\textbf{3.} De spanningswet voor serie verdeelt de \qty{6}{V} over twee
identieke lampen: elk ongeveer \qty{3}{V}.
\textbf{4.} \omterm{def:g5:series-parallel-circuits:parallel}{Parallel} --- elke lamp zou de volle \qty{6}{V}
overspannen. Door het kijkgaatje gloeit het seriepaar dof: elke lamp is
ondervoed op de helft van haar gebruikelijke duw.
\textbf{5.} $I_Z = 0.5 - 0.35 = \qty{0.15}{A}$
(knooppuntwet); $I_X = \qty{0.5}{A}$ --- X zit op de
onverdeelde weg (stroomwet voor serie).
\textbf{6.} $U_{\text{paar}} = 6 - 2.8 = \qty{3.2}{V}$ (luswet);
$U_Y = U_Z = \qty{3.2}{V}$ (parallelwet).
\textbf{7.} Dezelfde duw, verschillende marsen: de lampen verzetten
zich ongelijk tegen de stroom --- Z, die bij dezelfde
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} minder voert, is de sterkste tegenstander (de
zwaardere weg).
\textbf{8.} Dalen: het wegvallen van de \omterm{def:g5:series-parallel-circuits:parallel}{tak} van Y neemt een weg
weg; de overblijvende Z-weg verzet zich meer dan het paar samen deed,
dus drijft de batterij minder totale stroom aan.
\textbf{9.} Veroordeling één, luswet: $2.5 + 4.1 = \qty{6.6}{V}$
uitgegeven uit een beurs van \qty{6}{V}. Veroordeling twee,
knooppuntwet: erin \qty{0.3}{A}, eruit $0.2 + 0.15 =
\qty{0.35}{A}$.
\textbf{10.} $6 - 2.5 = \qty{3.5}{V}$.
\textbf{11.} $0.2 + 0.15 = \qty{0.35}{A}$.
\textbf{12.} Omdat de vier wetten elke eerlijke schakeling
overbepalen: elke aflezing moet kloppen met de sommen van de andere,
dus moet één verzonnen getal aan minstens één wet liegen --- en het
wordt door de rekenkunde alleen al veroordeeld.
\textbf{13.} Bijvoorbeeld: ``Eén weg, één mars --- de stroomwet voor
serie. Bij elke splitsing zijn de marsen in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}, erin is eruit
--- de knooppuntwet. Rond elke lus wordt de waterval van de
\omterm{def:g3:first-electric-circuit:battery}{batterij} tot de laatste druppel door de werkers uitgegeven ---
de spanningswet voor serie. En \omterm{def:g5:series-parallel-circuits:parallel}{takken} die tussen dezelfde twee
\omterm{def:g5:series-parallel-circuits:parallel}{knooppunten} hangen, drinken van dezelfde hoogte --- de
spanningswet voor \omterm{def:g5:series-parallel-circuits:parallel}{parallel}.''
\end{solution}
