\section*{Bab \ref{ch:g8:ohms-law} --- Hambatan dan Hukum Ohm}

\begin{solution}{exo:g8:ohms-law:1}
Seberapa kuat sebuah komponen melawan arusnya; \omterm{def:g6:measurement-in-science:measuring}{satuannya} \omterm{def:g8:ohms-law:resistance}{ohm}, dengan
lambang $\Omega$. $U = R I$; \ $I = U/R$; \ $R = U/I$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:2}
$I = 6 \div 30 = \qty{0.2}{A}$; \ $U = 30 \times 0.5 =
\qty{15}{V}$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:3}
$R = 4.5 \div 0.09 = 50\,\Omega$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:4}
Sebuah garis lurus yang melewati titik asal --- kesebandingan. Garis
yang lebih curam berarti lebih banyak \omterm{def:g8:voltage-voltmeter:voltage}{volt} yang diperlukan tiap
\omterm{def:g8:intensity-ammeter:intensity}{ampere}: \omterm{def:g8:ohms-law:resistance}{hambatan} yang lebih besar.
\end{solution}

\begin{solution}{exo:g8:ohms-law:5}
$3.0 \div 0.20 = 15$; $6.0 \div 0.40 = 15$. Kesamaan nisbahnya di
seluruh barisnya \emph{adalah} hukumnya: satu $R$ tetap yang melayani
setiap bacaan itulah yang menjadikan ``\omterm{def:g8:ohms-law:resistance}{hambatannya}'' sebuah sifat
komponennya.
\end{solution}

\begin{solution}{exo:g8:ohms-law:6}
\omterm{def:g8:ohms-law:resistance}{Hambatan} yang lebih kecil minum lebih banyak: $230 \div 25 =
\qty{9.2}{A}$ lawan $230 \div 35 \approx \qty{6.6}{A}$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:7}
Lampu seri berbagi satu $I$; \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} tiap lampunya adalah $U = R I$
dengan $I$ yang sama --- sehingga $R$ yang lebih besar terkalikan
menjadi $U$ yang lebih besar: penentang yang lebih kuat mengambil
bagian yang lebih besar.
\end{solution}

\begin{solution}{exo:g8:ohms-law:8}
\omterm{def:g7:short-circuits-safety:device}{Korsleting}: $R$ dekat nol, sehingga $I = U/R$ meledak --- lari
serempaknya adalah pembagian dengan hampir-tiada. \omterm{ex:g3:first-electric-circuit:switch}{Sakelar} terbuka: $R$
di luar segala ukuran, sehingga $I = U/R$ runtuh menjadi nol ---
penghadangan yang total.
\end{solution}

\begin{solution}{exo:g8:ohms-law:9}
\omterm{def:g8:ohms-law:resistor}{Resistornya} harus mengambil $9 - 2 = \qty{7}{V}$ pada \qty{0.02}{A}:
$R = 7 \div 0.02 = 350\,\Omega$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:10}
Titik pertama: $R = 1 \div 0.25 = 4\,\Omega$; yang kedua: $4 \div
0.5 = 8\,\Omega$ --- tidak ada nilai tunggal yang cocok. Kisahnya:
arus yang bertumbuh memanaskan filamennya, dan logam yang panas lebih
banyak melawan; potretnya membelok karena komponennya berubah selagi
ia bekerja.
\end{solution}

\begin{solution}{exo:g8:ohms-law:11}
Satu lingkar, satu $I$: kedua turunannya berjumlah sama dengan
dorongan \omterm{def:g3:first-electric-circuit:battery}{baterainya}, $6 = 10 I + 20 I = 30 I$, sehingga $I = \qty{0.2}{A}$.
Lalu $U_{10}
= 10 \times 0.2 = \qty{2}{V}$ dan $U_{20} = 20 \times 0.2 =
\qty{4}{V}$ --- berjumlah \qty{6}{V}, sebagaimana dituntut hukum
lingkarnya.
\end{solution}

\begin{solution}{exo:g8:ohms-law:12}
Tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} membentang pada \qty{6}{V}: $I_{10} = 6 \div 10 =
\qty{0.6}{A}$, $I_{20} = 6 \div 20 = \qty{0.3}{A}$; totalnya
$\qty{0.9}{A}$. Penggantinya: $R = 6 \div 0.9 \approx
6.7\,\Omega$ --- \emph{lebih kecil} daripada anggotanya masing-masing.
Dalam bahasa jalan: dua jalan di antara kota yang sama membawa lebih
banyak lalu lintas daripada masing-masing sendirian; setiap jalan
paralel yang ditambahkan memperingan \omterm{def:g6:describing-motion:trajectory}{lintasannya}, sehingga perlawanan
regunya jatuh di bawah perlawanan anggotanya yang terlemah.
\end{solution}

\begin{solution}{pb:g8:ohms-law:1}
\textbf{1.} Sumber, \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter}, dan komponennya dalam satu lingkar
seri --- gardu tolnya di tengah jalan; \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeternya} melintang pada
komponennya --- juru ukurnya di samping.
\textbf{2.} $R = 4.5 \div 0.15 = 30\,\Omega$.
\textbf{3.} Lima persen dari $33$ adalah kira-kira $1.65$: dapat
diterima dari $31.35$ sampai $34.65\,\Omega$. Angka $30\,\Omega$
milik contoh satu jatuh di luarnya: ditolak.
\textbf{4.} $I = 4.5 \div 150 = \qty{0.03}{A} =
\qty{30.0}{mA}$.
\textbf{5.} Setiap barisnya mengembalikan $R = U/I = 30\,\Omega$ ($1.5
\div 0.05$, $3.0 \div 0.10$, \dots): nisbah yang tetap, potret berupa
garis lurus --- satu nilai cocok bagi semuanya: $30\,\Omega$.
\textbf{6.} Nisbahnya berjalan $5$, $6.7$, $8.3$, $10\,\Omega$:
memanjat pada tiap langkahnya --- \omterm{def:g8:ohms-law:resistance}{hambatannya} bertumbuh selagi arusnya
bertumbuh; tidak ada $R$ tunggal yang ada.
\textbf{7.} Selagi arusnya bertumbuh filamennya memanas menuju
berpijar, dan logam yang panas lebih banyak melawan --- potretnya
membelok ke atas. Uji garis lurus mejanya karena itu tepat menerima
komponen yang $R$-nya tinggal diam: menurut bangunannya, ia sebuah
pengesan \omterm{def:g8:ohms-law:resistor}{resistor}.
\textbf{8.} Pasal \omterm{def:g3:temperature-thermometers:temperature}{suhu} yang mantapnya: hukum Ohm menjanjikan
kesebandingan \emph{pada \omterm{def:g3:temperature-thermometers:temperature}{suhu} yang mantap} --- hangatkan sebuah
komponen di tengah potretnya dan bahkan \omterm{def:g8:ohms-law:resistor}{resistor} yang jujur pun
bergeser.
\textbf{9.} $R = 12 \div 0.03 = 400\,\Omega$.
\textbf{10.} $U = R I = 25 \times 9.2 = \qty{230}{V}$ ---
tepat listrik rumahnya: kumparannya menarik persis apa yang
dijatahkan hukum Ohm kepadanya, dan selisih \omterm{def:g7:short-circuits-safety:fuse}{sekringnya} bekerja
sebagaimana dirancang. Tidak ada yang ``hanya''; segalanya hitungan.
\textbf{11.} Kabel penghubungnya lahir untuk menjadi kawat penghubung
--- jalan yang datar, yang tidak membelanjakan dorongan.
Disalahgunakan melintang pada sebuah sumber, \omterm{def:g8:ohms-law:resistance}{ohmnya} yang hampir-nol
menjadikannya \omterm{def:g7:short-circuits-safety:device}{korsleting} yang sempurna: sejak dahulu penjahatnya
hanyalah kawat yang sangat baik di tempat yang keliru.
\textbf{12.} Misalnya: ``Hukumnya: $U = R I$, dorongan sama dengan
perlawanan kali arak-arakan. Potretnya: sebuah garis lurus yang
melewati titik asal, lebih curam bagi penentang yang lebih kuat. Dan
hanya komponen pada \omterm{def:g3:temperature-thermometers:temperature}{suhu} yang mantap yang boleh menandatanganinya ---
panas menulis ulang \omterm{def:g8:ohms-law:resistance}{hambatan}, dan potret yang diambil ketika demam
tidak membuktikan apa pun.''
\end{solution}
