\section*{Bab \ref{ch:g8:speed-of-light} --- Kelajuan Cahaya; Jarak di Alam Semesta}

\begin{solution}{exo:g8:speed-of-light:1}
$c = \qty{300000}{km/s} = 3 \times 10^{8}\ \unit{m/s}$. Lentera di
puncak bukitnya hanya \omterm{def:g6:measurement-in-science:measuring}{mengukur} waktu tanggap manusia: \omterm{def:g1:light-and-shadows:source}{cahaya}
menyeberangi bukitnya dalam perjutaan sekon, jauh di bawah pencatatan
waktu tangan mana pun.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:2}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} Jupiter tergerhana menurut jadwal yang tepat waktu --- sebuah
\omterm{ex:g2:measuring-time:clock}{jam} langit. \emph{Berita} tiap gerhananya tiba terlambat setiap kali
Bumi berdiri jauh dari Jupiter, lebih awal ketika dekat.
Keterlambatannya \omterm{def:g6:measurement-in-science:measuring}{mengukur} waktu tempuh \omterm{def:g1:light-and-shadows:source}{cahaya} menyeberangi lebar \omterm{def:g6:sun-earth-moon:axisorbit}{orbit}
Bumi --- dan dengan demikian \omterm{def:g6:measurement-in-science:measuring}{mengukur} \omterm{def:g7:motion-average-speed:formula}{kelajuan} \omterm{def:g1:light-and-shadows:source}{cahaya}.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:3}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan}: $384000 \div 300000 \approx \qty{1.3}{s}$. Matahari:
$150000000 \div 300000 = \qty{500}{s}$ --- delapan menit dua puluh.
Sinar matahari selalu menunjukkan Matahari sebagaimana ia delapan
menit yang lalu: berita basi menurut bangunannya.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:4}
Sebuah jarak: bentangan yang ditempuh \omterm{def:g1:light-and-shadows:source}{cahaya} dalam satu tahun ---
kira-kira $9.5 \times 10^{12}$ km.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:5}
Empat tahun. Jawabannya berbeda-beda --- empat tahun yang lalu kamu
masih beberapa tingkat di bawah pada buku ini juga.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:6}
Pada kira-kira sebelas menit sekali jalan, sebuah pembetulan kemudi
menjawab sebuah gambar yang sedikitnya sudah basi dua puluh dua menit
--- penjelajahnya akan sudah masuk ke jurangnya sebelum kedutan \omterm{def:g3:levers-and-scales:lever}{tuas}
kendalinya tiba. Karena itulah rencana, bukan \omterm{def:g3:levers-and-scales:lever}{tuas} kendali.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:7}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan}: kira-kira \qty{1.3}{s}. Matahari: delapan menit. \omterm{def:g4:solar-system:star}{Bintang}
terdekat: empat tahun. Galaksi tetangganya: dua setengah juta tahun.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:8}
\omterm{def:g1:light-and-shadows:source}{Cahaya}. \omterm{def:g7:motion-average-speed:formula}{Kelajuan} $c$ terbukti lebih mantap daripada batang logam atau
penggaris kebangsaan mana pun --- sehingga penggarisnya kini
diturunkan dari \omterm{def:g1:light-and-shadows:source}{cahayanya}, bukan \omterm{def:g1:light-and-shadows:source}{cahayanya} yang diukur penggarisnya.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:9}
Pulang perginya: $300000 \times 2.6 = 780000$ km; \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berdiri pada
separuhnya --- \qty{390000}{km}. Aturan sonarnya berlaku pula di sini
seutuhnya: setiap \omterm{ex:g4:how-sound-travels:echo}{gema} membayar perjalanan pulang perginya, paruhlah
sebelum memercayainya.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:10}
Lampu: $3 \div (3 \times 10^{8}) = 10^{-8}$ s --- seperseratus
perjutaan sekon: bahkan kamarnya pun sebuah arsip, hanya terlalu
dangkal untuk diperhatikan. Lalu Matahari (delapan menit), \omterm{def:g4:solar-system:star}{bintangnya}
(seabad), galaksinya (dua setengah juta tahun) --- makin dalam kamu
memandang, makin tua halamannya.
\end{solution}

\begin{solution}{exo:g8:speed-of-light:11}
``Selasa lalu, \omterm{def:g1:light-and-shadows:source}{cahaya} sebuah ledakan \omterm{def:g4:solar-system:star}{bintang} \emph{mencapai kita} ---
ledakannya sendiri terjadi kira-kira $8000$ tahun yang lalu, dan
beritanya sudah bepergian sejak saat itu.''
\end{solution}

\begin{solution}{exo:g8:speed-of-light:12}
$3 \times 10^{8}\ \text{km} \div 1000\ \text{s} = 3 \times
10^{5}$ \unit{km/s} --- persis nilai zaman sekarang (taksiran
sejarahnya, dengan angka \omterm{def:g6:sun-earth-moon:axisorbit}{orbit} yang lebih kasar, mendarat dalam
selisih kira-kira seperempatnya: untuk abad ketujuh belas, jawaban
pertama yang menakjubkan).
\end{solution}

\begin{solution}{pb:g8:speed-of-light:1}
\textbf{1.} Kira-kira \qty{1.3}{s} sesudah kamu berbicara; pengakuan
penerimaan yang paling cepat kembali sesudah perjalanan pulang
perginya, kira-kira \qty{2.6}{s}.
\textbf{2.} $36000 \div 300000 = \qty{0.12}{s}$ sekali jalan; sebuah
tanya-jawab menunggangi naik dan turun dua kali --- hampir setengah
sekon keraguan, tepat pada ambang telinganya.
\textbf{3.} ``Suarnya meletus delapan menit dua puluh sebelum layar
kita menyala --- kita menontonnya sebagaimana ia \emph{dahulu}.''
\textbf{4.} Sambungan \omterm{def:g2:moon-in-the-sky:moon}{Bulannya}: \qty{1.3}{s} lawan \qty{0.12}{s}
--- kasarnya sepuluh kali kelonggarannya.
\textbf{5.} $1.8 \times 10^{8} \div (3 \times 10^{5}) = 600$
s: sepuluh menit sekali jalan.
\textbf{6.} Perjalanan pulang perginya dua puluh menit: gambar
penyelamatannya sudah berumur sepuluh menit dan perintah
penyelamatannya terlambat sepuluh menit --- jurangnya menang dengan
selisih sembilan belas menit. Penjelajahnya harus membawa waktu
tanggapnya sendiri: penghentian bahaya di dalam dirinya.
\textbf{7.} Kepingan terakhirnya harus tiba pada pukul 21:00; ia
meninggalkan Bumi pada $21{:}00 - 10$ menit $= 20{:}50$; pemancarannya
mulai tiga menit lebih awal: mulailah pada pukul 20:47.
\textbf{8.} Kelambatannya menggandakan diri menjadi kira-kira dua
puluh satu menit sekali jalan: \omterm{def:g4:solar-system:planet}{planet} tidak mempunyai ``sang'' jarak
--- kedua dunianya mengorbit, dan setiap percakapan dijadwalkan
terhadap sasaran yang bergerak.
\textbf{9.} $300000 \times 86400 \approx 2.6 \times 10^{10}$
km --- dua puluh enam ribu juta kilometer; sebuah tanya-jawab memakan
dua hari penuh.
\textbf{10.} \omterm{def:g1:light-and-shadows:source}{Cahaya} mencapai \omterm{def:g4:solar-system:star}{bintangnya} dalam kira-kira empat tahun;
jawaban yang paling cepat mendarat kira-kira delapan tahun sesudah
pemancaran malam ini.
\textbf{11.} ``\omterm{def:g1:light-and-shadows:source}{Cahaya} ini meninggalkan galaksinya dua setengah juta
tahun yang lalu: potretnya menyiarkan Bumi pada zaman perkakas batu
yang paling awal --- halaman arsip mata telanjang kita yang
terdalam.''
\textbf{12.} Misalnya: ``Tidak ada yang seketika --- setiap isyarat,
termasuk isyarat kita, paling cepat berjalan pada $c$. \omterm{def:g3:levers-and-scales:lever}{Tuas} kendali
kalah oleh kelambatan di luar \omterm{def:g2:moon-in-the-sky:moon}{Bulan}: mesin yang jauh harus berpikir
sendiri. Dan setiap teleskop adalah pembaca arsip --- makin jauh
bendanya, makin tua beritanya, tanpa terbitan lebih segar yang dapat
dicetak.''
\end{solution}
