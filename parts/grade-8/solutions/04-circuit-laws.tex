\section*{Chapter \ref{ch:g8:circuit-laws} --- Current and Voltage Laws in Circuits}

\begin{solution}{exo:g8:circuit-laws:1}
Series current law: one loop, one \omterm{def:g8:intensity-ammeter:intensity}{intensity} everywhere.
\omterm{def:g5:series-parallel-circuits:parallel}{Junction} law: currents in equal currents out. Series \omterm{def:g8:voltage-voltmeter:voltage}{voltage}
law: the \omterm{def:g7:short-circuits-safety:battery}{battery's} \omterm{def:g8:voltage-voltmeter:voltage}{voltage} equals the sum of the devices'
shares around the loop. Parallel \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law: \omterm{def:g5:series-parallel-circuits:parallel}{branches} spanning
the same \omterm{def:g5:series-parallel-circuits:parallel}{junctions} share one and the same \omterm{def:g8:voltage-voltmeter:voltage}{voltage}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:2}
$0.8 - 0.55 = \qty{0.25}{A}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:3}
$9 - 2.5 - 3.8 = \qty{2.7}{V}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:4}
\qty{12}{V} each --- the parallel \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law. The same law
puts the identical \qty{230}{V} on every socket of the house's
parallel wiring.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:5}
For instance $I_C$: once $I_{\text{battery}}$ and $I_B$ are
known, the \omterm{def:g5:series-parallel-circuits:parallel}{junction} law fixes $I_C = \qty{0.2}{A}$ --- any
meter sent there could only confirm (likewise $U_C$, fixed by
the parallel \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law once $U_B$ is known).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:6}
The \qty{2.7}{V} lamp: at equal current, the larger \omterm{def:g8:voltage-voltmeter:voltage}{voltage}
share marks the stronger opposer. The next chapter \omterm{def:g6:measurement-in-science:measuring}{measures}
that opposition --- resistance.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:7}
$2.2 + 1.9 + 1.4 = \qty{5.5}{V}$ against a \qty{6}{V} purse:
half a \omterm{def:g8:voltage-voltmeter:voltage}{volt} astray --- beyond honest meter scatter for such
readings; one dial was misread (or a share went unmeasured).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:8}
$0.4 + 0.3 + 0.3 = \qty{1.0}{A}$ leaving against \qty{0.9}{A}
arriving: the fork ``creates'' \qty{0.1}{A} --- convicted by
the \omterm{def:g5:series-parallel-circuits:parallel}{junction} law.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:9}
The loop's spending must sum to the \omterm{def:g7:short-circuits-safety:battery}{battery's} \qty{4.5}{V}:
open \omterm{ex:g3:first-electric-circuit:switch}{switch} \qty{4.5}{V} $+$ idle lamp \qty{0}{V} $= 4.5$ ---
balanced. (After the click the shares trade: \omterm{ex:g3:first-electric-circuit:switch}{switch} near zero,
lamp near \qty{4.5}{V} --- balanced again.)
\end{solution}

\begin{solution}{exo:g8:circuit-laws:10}
Before: $3 \times 1.5 = \qty{4.5}{V}$. After: $4.5 - 1.5 =
\qty{3.0}{V}$ --- the reversed cell subtracts. The lamp dims
from its \qty{4.5}{V} glow to a \qty{3}{V} one.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:11}
\omterm{def:g5:series-parallel-circuits:parallel}{Branch} one spans the \omterm{def:g3:first-electric-circuit:battery}{battery}: $U_E + U_F = \qty{6}{V}$ (series
\omterm{def:g8:voltage-voltmeter:voltage}{voltage} law along the \omterm{def:g5:series-parallel-circuits:parallel}{branch}), so $U_F = 6 - 2.2 =
\qty{3.8}{V}$. \omterm{def:g5:series-parallel-circuits:parallel}{Branch} two spans the same \omterm{def:g5:series-parallel-circuits:parallel}{junctions}: $U_G =
\qty{6}{V}$ (parallel \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:12}
$I_R = 0.6 - 0.45 = \qty{0.15}{A}$ (\omterm{def:g5:series-parallel-circuits:parallel}{junction} law). $U_R = U_Q
= \qty{2.3}{V}$ (parallel \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law). $U_P = 6 - 2.3 =
\qty{3.7}{V}$ (series \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law). A convictable extra:
``$I_R = \qty{0.2}{A}$'' --- the \omterm{def:g5:series-parallel-circuits:parallel}{junction} books would show
$0.45 + 0.2 = 0.65 \neq 0.6$; or ``$U_R = \qty{2.6}{V}$'' ---
parallel \omterm{def:g5:series-parallel-circuits:parallel}{branches} may not disagree.
\end{solution}

\begin{solution}{pb:g8:circuit-laws:1}
\textbf{1.} Series. \omterm{def:g4:series-circuits:series}{In series} the two lamps oppose the march
one after the other: the loop's current falls well below a
single lamp's \qty{0.4}{A} --- the measured \qty{0.2}{A} fits.
(Parallel would \emph{add} two full \omterm{def:g5:series-parallel-circuits:parallel}{branches}.)
\textbf{2.} Parallel: two \omterm{def:g5:series-parallel-circuits:parallel}{branches} of about \qty{0.4}{A} each
--- roughly \qty{0.8}{A} from the \omterm{def:g3:first-electric-circuit:battery}{battery}.
\textbf{3.} The series \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law splits the \qty{6}{V}
between two identical lamps: about \qty{3}{V} each.
\textbf{4.} Parallel --- each lamp would span the full
\qty{6}{V}. Through the peephole the series pair glows dim:
each is underfed at half its usual push.
\textbf{5.} $I_Z = 0.5 - 0.35 = \qty{0.15}{A}$ (\omterm{def:g5:series-parallel-circuits:parallel}{junction} law);
$I_X = \qty{0.5}{A}$ --- X sits on the undivided road (series
current law).
\textbf{6.} $U_{\text{pair}} = 6 - 2.8 = \qty{3.2}{V}$ (loop
law); $U_Y = U_Z = \qty{3.2}{V}$ (parallel law).
\textbf{7.} Same push, different marches: the lamps oppose the
current unequally --- Z, carrying less at the same \omterm{def:g8:voltage-voltmeter:voltage}{voltage}, is
the stronger opposer (the harder road).
\textbf{8.} Fall: losing Y's \omterm{def:g5:series-parallel-circuits:parallel}{branch} removes a road; the
remaining Z-road opposes more than the pair did together, so
the \omterm{def:g3:first-electric-circuit:battery}{battery} drives less total current.
\textbf{9.} Conviction one, loop law: $2.5 + 4.1 =
\qty{6.6}{V}$ spent from a \qty{6}{V} purse. Conviction two,
\omterm{def:g5:series-parallel-circuits:parallel}{junction} law: in \qty{0.3}{A}, out $0.2 + 0.15 =
\qty{0.35}{A}$.
\textbf{10.} $6 - 2.5 = \qty{3.5}{V}$.
\textbf{11.} $0.2 + 0.15 = \qty{0.35}{A}$.
\textbf{12.} Because the four laws over-determine every honest
circuit: each reading must agree with the others' sums, so one
invented number must lie to at least one law --- and is
convicted by arithmetic alone.
\textbf{13.} For example: ``One road, one march --- the series
current law. At every fork the marches balance, in equals out
--- the \omterm{def:g5:series-parallel-circuits:parallel}{junction} law. Around any loop, the \omterm{def:g7:short-circuits-safety:battery}{battery's} waterfall
is spent to the last drop by the workers --- the series \omterm{def:g8:voltage-voltmeter:voltage}{voltage}
law. And \omterm{def:g5:series-parallel-circuits:parallel}{branches} hanging between the same two \omterm{def:g5:series-parallel-circuits:parallel}{junctions} drink
from the same height --- the parallel \omterm{def:g8:voltage-voltmeter:voltage}{voltage} law.''
\end{solution}
