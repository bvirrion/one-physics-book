\section*{الفصل \ref{ch:g8:circuit-laws} --- قانونا التيار والتوتر في الدارات}

\begin{solution}{exo:g8:circuit-laws:1}
قانون التيار \omterm{def:g4:series-circuits:series}{على التسلسل}: حلقة واحدة، \omterm{def:g8:intensity-ammeter:intensity}{وشدّة} واحدة في كل مكان.
وقانون العقد: التيارات الداخلة تساوي التيارات الخارجة. وقانون
\omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g4:series-circuits:series}{على التسلسل}: \omterm{def:g8:voltage-voltmeter:voltage}{توتر} \omterm{def:g3:first-electric-circuit:battery}{البطارية} يساوي مجموع حصص الأجهزة حول
الحلقة. وقانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}: الفروع الممتدّة بين العقدتين
نفسيهما تتقاسم توترًا واحدًا بعينه.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:2}
$0.8 - 0.55 = \qty{0.25}{A}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:3}
$9 - 2.5 - 3.8 = \qty{2.7}{V}$.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:4}
\qty{12}{V} لكل واحد --- قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}. والقانون
نفسه يضع \qty{230}{V} عينها على كل مأخذ في تمديدات البيت
المتفرّعة.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:5}
مثلًا $I_C$: فمتى عُلم $I_{\text{البطارية}}$ و$I_B$، ثبّت قانون
العقد $I_C = \qty{0.2}{A}$ --- وأيّ مقياس يُرسَل إلى هناك لا
يملك إلا التأكيد (وكذلك $U_C$، يثبّته قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}
متى عُلم $U_B$).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:6}
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} \qty{2.7}{V}: فعند تساوي التيار، تدلّ حصة \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} الأكبر
على المقاوم الأقوى. والفصل التالي يقيس تلك المقاومة ---
المقاومة الكهربائية.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:7}
$2.2 + 1.9 + 1.4 = \qty{5.5}{V}$ في مقابل كيس \qty{6}{V}: نصف
\omterm{def:g8:voltage-voltmeter:voltage}{فولت} ضائع --- وهو أبعد من \omterm{def:g7:light-sources-propagation:diffusion}{تبعثر} المقاييس الأمين عند قراءات
كهذه؛ فقد أُسيئت قراءة قرص (أو بقيت حصة بلا \omterm{def:g6:measurement-in-science:measuring}{قياس}).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:8}
$0.4 + 0.3 + 0.3 = \qty{1.0}{A}$ خارجة في مقابل \qty{0.9}{A}
واصلة: فالمفترق ``يخلق'' \qty{0.1}{A} --- مُدان بقانون العقد.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:9}
يجب أن يبلغ إنفاق الحلقة \omterm{def:g8:voltage-voltmeter:voltage}{توتر} \omterm{def:g3:first-electric-circuit:battery}{البطارية} \qty{4.5}{V}: \omterm{ex:g3:first-electric-circuit:switch}{القاطعة}
المفتوحة \qty{4.5}{V} $+$ \omterm{def:g3:first-electric-circuit:bulb}{المصباح} العاطل \qty{0}{V} $= 4.5$ ---
متوازن. (وبعد النقرة تتبادل الحصص: \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} قرب الصفر،
\omterm{def:g3:first-electric-circuit:bulb}{والمصباح} قرب \qty{4.5}{V} --- متوازن ثانية.)
\end{solution}

\begin{solution}{exo:g8:circuit-laws:10}
قبل: $3 \times 1.5 = \qty{4.5}{V}$. وبعد: $4.5 - 1.5 =
\qty{3.0}{V}$ --- فالخلية المعكوسة تطرح. ويخفت \omterm{def:g3:first-electric-circuit:bulb}{المصباح} من
توهّج \qty{4.5}{V} إلى توهّج \qty{3}{V}.
\end{solution}

\begin{solution}{exo:g8:circuit-laws:11}
\omterm{def:g5:series-parallel-circuits:parallel}{الفرع} الأول يمتدّ على \omterm{def:g3:first-electric-circuit:battery}{البطارية}: $U_E + U_F = \qty{6}{V}$
(قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g4:series-circuits:series}{على التسلسل} على امتداد \omterm{def:g5:series-parallel-circuits:parallel}{الفرع})، ومنه $U_F = 6 -
2.2 = \qty{3.8}{V}$. \omterm{def:g5:series-parallel-circuits:parallel}{والفرع} الثاني يمتدّ بين العقدتين
نفسيهما: $U_G = \qty{6}{V}$ (قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}).
\end{solution}

\begin{solution}{exo:g8:circuit-laws:12}
$I_R = 0.6 - 0.45 = \qty{0.15}{A}$ (قانون العقد). و$U_R = U_Q
= \qty{2.3}{V}$ (قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}). و$U_P = 6 - 2.3 =
\qty{3.7}{V}$ (قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g4:series-circuits:series}{على التسلسل}). ومن الإضافات
القابلة للإدانة: ``$I_R = \qty{0.2}{A}$'' --- فدفاتر \omterm{def:g5:series-parallel-circuits:parallel}{العقدة}
ستُظهر $0.45 + 0.2 = 0.65 \neq 0.6$؛ أو ``$U_R = \qty{2.6}{V}$''
--- فالفروع \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} لا يجوز أن تختلف.
\end{solution}

\begin{solution}{pb:g8:circuit-laws:1}
\textbf{1.} \omterm{def:g4:series-circuits:series}{على التسلسل}. \omterm{def:g4:series-circuits:series}{فعلى التسلسل} يقاوم المصباحان المسير
واحدًا بعد الآخر: فيهبط تيار الحلقة دون \qty{0.4}{A} الخاصّة
\omterm{def:g3:first-electric-circuit:bulb}{بمصباح} واحد بكثير --- و\qty{0.2}{A} المقيسة تلائم ذلك. (أمّا
التفرع فكان \emph{سيجمع} فرعين كاملين.)
\textbf{2.} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}: فرعان بنحو \qty{0.4}{A} لكل واحد ---
أي نحو \qty{0.8}{A} من \omterm{def:g3:first-electric-circuit:battery}{البطارية}.
\textbf{3.} قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g4:series-circuits:series}{على التسلسل} يقسم \qty{6}{V} بين
مصباحين متماثلين: نحو \qty{3}{V} لكل واحد.
\textbf{4.} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} --- فكان كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح} سيمتدّ على \qty{6}{V}
كاملة. ومن الكوّة يتوهّج الزوج المتسلسل خافتًا: فكل واحد ناقص
التغذية عند نصف دفعته المعتادة.
\textbf{5.} $I_Z = 0.5 - 0.35 = \qty{0.15}{A}$ (قانون العقد)؛
و$I_X = \qty{0.5}{A}$ --- \omterm{def:g3:first-electric-circuit:bulb}{فالمصباح} $X$ يجلس على الطريق غير المنقسم
(قانون التيار \omterm{def:g4:series-circuits:series}{على التسلسل}).
\textbf{6.} $U_{\text{الزوج}} = 6 - 2.8 = \qty{3.2}{V}$ (قانون
الحلقة)؛ و$U_Y = U_Z = \qty{3.2}{V}$ (قانون التفرع).
\textbf{7.} الدفعة نفسها، والمسيران مختلفان: فالمصباحان
يقاومان التيار على غير سواء --- و$Z$، وهو يحمل أقلّ عند \omterm{def:g8:voltage-voltmeter:voltage}{التوتر}
نفسه، هو المقاوم الأقوى (الطريق الأشقّ).
\textbf{8.} هبوط: ففقد \omterm{def:g5:series-parallel-circuits:parallel}{فرع} $Y$ يزيل طريقًا؛ وطريق $Z$ الباقي
يقاوم أكثر مما كان الفرعان يقاومانه معًا، فتسوق \omterm{def:g3:first-electric-circuit:battery}{البطارية}
تيارًا كلّيًّا أقلّ.
\textbf{9.} الإدانة الأولى، بقانون الحلقة: $2.5 + 4.1 =
\qty{6.6}{V}$ منفقة من كيس \qty{6}{V}. والإدانة الثانية،
بقانون العقد: داخل \qty{0.3}{A}، وخارج $0.2 + 0.15 =
\qty{0.35}{A}$.
\textbf{10.} $6 - 2.5 = \qty{3.5}{V}$.
\textbf{11.} $0.2 + 0.15 = \qty{0.35}{A}$.
\textbf{12.} لأن القوانين الأربعة تُفرِط في تحديد كل دارة
أمينة: فكل قراءة يجب أن توافق مجاميع أخواتها، فلا بدّ لعدد
مخترَع أن يكذب على قانون واحد على الأقلّ --- فيُدان بالحساب
وحده.
\textbf{13.} مثلًا: ``طريق واحد، ومسير واحد --- قانون التيار
\omterm{def:g4:series-circuits:series}{على التسلسل}. وعند كل مفترق تتوازن المسايير، فالداخل يساوي
الخارج --- قانون العقد. وحول أيّ حلقة يُنفق شلّال \omterm{def:g3:first-electric-circuit:battery}{البطارية} إلى
آخر قطرة على العمّال --- قانون \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g4:series-circuits:series}{على التسلسل}. والفروع
المعلّقة بين العقدتين نفسيهما تشرب من الارتفاع نفسه --- قانون
\omterm{def:g8:voltage-voltmeter:voltage}{التوتر} \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}.''
\end{solution}
