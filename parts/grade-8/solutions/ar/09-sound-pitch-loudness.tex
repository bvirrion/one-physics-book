\section*{الفصل \ref{ch:g8:sound-pitch-loudness} --- الصوت: السرعة والحدّة والعلوّ}

\begin{solution}{exo:g8:sound-pitch-loudness:1}
\omterm{def:g8:sound-pitch-loudness:frequency}{التردّد}: عدد التأرجحات الكاملة في الثانية، \omterm{def:g8:sound-pitch-loudness:frequency}{بالهرتز}
(\unit{Hz}). ونغمة ``لا'' في الحفلات: \qty{440}{Hz}؛ والتي
فوقها بأوكتاف: \qty{880}{Hz} --- مضاعفة.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:2}
$4$ تأرجحات في \qty{0.02}{s}: \qty{200}{Hz}؛ و$16$ تأرجحة:
\qty{800}{Hz} --- فالثاني أحدّ صوتًا (بأوكتافين في الحقيقة).
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:3}
من نحو \qty{20}{Hz} إلى \qty{20000}{Hz}. ودونها: \omterm{prop:g8:sound-pitch-loudness:range}{الأصوات دون
السمعية} --- هدير الفيلة البعيد المدى. وفوقها: \omterm{prop:g8:sound-pitch-loudness:range}{الأصوات فوق
السمعية} --- نقرات صيد الخفّاش.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:4}
أعلى بالنغمة نفسها: تعلو القمم ويبقى التباعد كما هو. وأحدّ
بالعلوّ نفسه: تتزاحم القمم ويبقى الارتفاع كما هو. قرصان
مستقلّان.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:5}
كلٌّ منها يرفع \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد}: فالأوتار الأقصر والأشدّ شدًّا والأرقّ
كلها ترتعش مرات أكثر في الثانية --- \omterm{def:g8:sound-pitch-loudness:frequency}{تردّد} أعلى، فحدّة أعلى.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:6}
الأوراق \qty{10}{dB}؛ وحركة المرور \qty{80}{dB}؛ والصفّ
الأمامي \qty{110}{dB}؛ والألم قرب \qty{120}{dB}. ونطاق الضرر
يفتح قرب \qty{85}{dB}.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:7}
حدّة الصفّارة تقع فوق \qty{20000}{Hz} بقليل --- خارج نافذة
صاحبها، وداخل نافذة الكلب. معطّلة لزوج من الآذان، ثاقبة
للآخر: فالنوافذ تختلف باختلاف الأنواع.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:8}
أن \omterm{def:g7:motion-average-speed:formula}{السرعة} واحدة بعينها لكل حدّة ولكل علوّ: فسرعات مختلفة كانت
ستلطّخ الأوتار البعيدة إلى تعاقب نغمات وتؤخّر الخافت خلف
العالي --- وحفلة الحديقة تدحض الاثنين كل ليلة.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:9}
$50000 \times 0.01 = 500$ تأرجحة في النقرة. والرسم الدقيق
يحتاج إلى فراشٍ دقيقة: فلا يستطيع تحسّس التفاصيل الصغيرة في
أصدائها إلا اهتزازات ذات تأرجحات قصيرة جدًّا --- أي \omterm{def:g8:sound-pitch-loudness:frequency}{تردّد} $f$
عالٍ جدًّا؛ أما الهدير الخشن المنخفض \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد} فيغسل فراشة
عابرة من غير أن يلحظها.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:10}
كلا المفتاحين داخل النافذة --- \qty{27}{Hz} فوق أرضية
\qty{20}{Hz} بقليل، و\qty{4200}{Hz} دون السقف براحة. لكن
العمر يقضم النافذة من \emph{أعلى}: فأذن المدوزن نفسها تبلد
أولًا في الطبقات الحادّة، فيُستدعى المحلّل للأوكتافات العليا
قبل الدنيا.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:11}
نموّ بعشرة أضعاف مختبئًا في خطوات متساوية --- فكل بضع درجات
تضرب \omterm{def:g2:pushes-and-pulls:force}{قوة} \omterm{def:g3:sound-around-us:sound}{الصوت} في عدد، لا تضيف إليها. فعبارة ``أعلى قليلًا'' على
السلّم أقوى \emph{بكثير} على الأذن: فتنهار الميزانيات من
ساعات إلى دقائق عبر ما يُقرأ لفّة صغيرة على القرص.
\end{solution}

\begin{solution}{exo:g8:sound-pitch-loudness:12}
فحص البيانو: المحلّل بجانب الوتر، وتُقارن قراءته بالعدد $440$.
وعلوّ الصفّ الخلفي: قِف هناك مع قراءة الديسيبل أثناء اختبار
الطبول. والتهوية: صوّب شاشة الطيف في المحلّل إلى الطرف
المنخفض وابحث عن شوكة دون \qty{20}{Hz}. وأقلّها جدارة بالثقة:
حكم \omterm{prop:g8:sound-pitch-loudness:range}{الأصوات دون السمعية} --- فمذياع الهاتف مبنيّ للأصوات
البشرية ويصمّ قرب أرضية النافذة، فشاشة صامتة هناك لا تثبت
كثيرًا.
\end{solution}

\begin{solution}{pb:g8:sound-pitch-loudness:1}
\textbf{1.} البيانو منخفض الدوزنة --- فكل ``لا'' فيه ترتعش
ثماني تأرجحات في الثانية أقلّ مما ينبغي. والقرص: الحدّة ---
مفتاح مدوزن، لا مضخّم.
\textbf{2.} \qty{41}{Hz} تجلس فوق أرضية النافذة \qty{20}{Hz}
بقليل: فتُسمع، عميقة --- ومع تأرجحات بهذا البطء يجيب طبل
الصدر نفسه: فتُحسّ بقدر ما تُسمع.
\textbf{3.} الحدّة نفسها (تباعد متساوٍ، \qty{440}{Hz})، \omterm{prop:g8:sound-pitch-loudness:loudness}{والسعة}
ثلاثة أضعاف: فالمغنّي يغنّي نغمة الشوكة، لكن أعلى بكثير.
\textbf{4.} أصغر الآذان لا غير --- فسقف النافذة قرب
\qty{20000}{Hz} ملك الأطفال؛ ونوافذ معظم البالغين قد هبطت
سلفًا، ولا مستمع يصدّق على \qty{22000}{Hz}.
\textbf{5.} وصلة الصخب (\qty{100}{dB}) والختام
(\qty{110}{dB}) كلتاهما فوق خطّ الضرر \qty{85}{dB}: أرض
الدقائق لا غير؛ أما الوصلة الصوتية فتمرّ.
\textbf{6.} نحو \qty{80}{dB} خلف السدادات --- أي رجوع دون خطّ
الضرر: فيصير تعرّض أمسية قابلًا للنجاة، ولهذا يلبسها الحكماء
دائمًا.
\textbf{7.} القشور المنتشرة: فكل ضغطة تُقتسم على قشرة أوسع مع
البعد --- فيخبو العلوّ (\omterm{prop:g8:sound-pitch-loudness:loudness}{السعة}). أما الحدّة فتركب بلا تغيّر:
فالبعد يدير قرصًا واحدًا لا غير.
\textbf{8.} العزف الخافت يدير قرص \omterm{prop:g8:sound-pitch-loudness:loudness}{السعة} وحده --- \omterm{def:g8:sound-pitch-loudness:frequency}{وتردّد} كل
نغمة، حادّة كانت أم غليظة، يبقى في مكانه تمامًا. (و``بلادته''
حيلة الأذن نفسها عند العلوّ المنخفض، لا فيزياء الأوتار.)
\textbf{9.} $680 \div 340 = \qty{2}{s}$ بين الوميض والدويّ.
ودقّتان في الثانية: \qty{2}{Hz} --- أي دون الأرضية بكثير:
فتُحسّ في الصدر، ولا تُسمع نغمةً أبدًا.
\textbf{10.} أخفض تأرجحات الجهير: قرب أرضية النافذة ودونها،
وهو بالضبط حيث يصمّ المِذياع المبنيّ للأصوات البشرية ---
فالهدير كان حقيقيًّا؛ والهاتف لم يلتقطه قطّ.
\textbf{11.} مثلًا: ``الحدّة هي \omterm{def:g8:sound-pitch-loudness:frequency}{التردّد} --- عدّ التأرجحات في
الثانية، وشغل المدوزن كله. والعلوّ هو \omterm{prop:g8:sound-pitch-loudness:loudness}{السعة} --- حجم
التأرجحات، ودفتر الممرّضة كله. ولا يمسّ أيّ قرص \omterm{def:g7:motion-average-speed:formula}{السرعة}: فكل
نغمة الليلة عبرت القاعة عند \qty{340}{m/s} نفسها.''
\textbf{12.} قضية النافذة: الطارد حدّته فوق \qty{20000}{Hz}
--- خارج كل نافذة بشرية، وداخل نافذة الفأر. ارتعاش واحد،
وجمهوران، ولا تناقض.
\end{solution}
