\section*{अध्याय \ref{ch:g8:ohms-law} --- प्रतिरोध और ओम का नियम}

\begin{solution}{exo:g8:ohms-law:1}
यह कि कोई पुरज़ा धारा का कितने ज़ोर से विरोध करता है; \omterm{def:g6:measurement-in-science:measuring}{मात्रक} \omterm{def:g8:ohms-law:resistance}{ओम},
प्रतीक $\Omega$. $U = R I$; \ $I = U/R$; \ $R = U/I$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:2}
$I = 6 \div 30 = \qty{0.2}{A}$; \ $U = 30 \times 0.5 =
\qty{15}{V}$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:3}
$R = 4.5 \div 0.09 = 50\,\Omega$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:4}
मूल बिंदु से गुज़रती एक सीधी रेखा --- यानी समानुपात। और ज़्यादा खड़ी
रेखा का मतलब है हर \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} के लिए ज़्यादा \omterm{def:g8:voltage-voltmeter:voltage}{वोल्ट}: यानी बड़ा \omterm{def:g8:ohms-law:resistance}{प्रतिरोध}।
\end{solution}

\begin{solution}{exo:g8:ohms-law:5}
$3.0 \div 0.20 = 15$; $6.0 \div 0.40 = 15$. पंक्तियों भर इस अनुपात
का एक-सा रहना ही वह नियम \emph{है}: हर पाठ की सेवा करता एक ही अटल $R$
ही ``\omterm{def:g8:ohms-law:resistance}{प्रतिरोध}'' को उस पुरज़े का गुण बनाता है।
\end{solution}

\begin{solution}{exo:g8:ohms-law:6}
छोटा \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} ज़्यादा पीता है: $230 \div 25 = \qty{9.2}{A}$, जबकि
$230 \div 35 \approx \qty{6.6}{A}$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:7}
\omterm{def:g4:series-circuits:series}{श्रेणी में} लगे लैंप एक ही $I$ बाँटते हैं; और हर लैंप का \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} उसी
$I$ के साथ $U = R I$ होता है --- इसलिए बड़ा $R$ गुणा होकर बड़ा $U$ बना
देता है: ज़्यादा विरोध करने वाला बड़ा हिस्सा ले जाता है।
\end{solution}

\begin{solution}{exo:g8:ohms-law:8}
\omterm{def:g7:short-circuits-safety:device}{लघुपथन}: $R$ लगभग शून्य, इसलिए $I = U/R$ फट पड़ती है --- वह भगदड़
लगभग-कुछ-नहीं से भाग देना है। खुला \omterm{ex:g3:first-electric-circuit:switch}{स्विच}: $R$ मापन से परे, इसलिए
$I = U/R$ ढहकर शून्य हो जाती है --- पूरी नाकाबंदी।
\end{solution}

\begin{solution}{exo:g8:ohms-law:9}
\omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} को \qty{0.02}{A} पर $9 - 2 = \qty{7}{V}$ लेने हैं:
$R = 7 \div 0.02 = 350\,\Omega$.
\end{solution}

\begin{solution}{exo:g8:ohms-law:10}
पहला बिंदु: $R = 1 \div 0.25 = 4\,\Omega$; दूसरा: $4 \div 0.5 =
8\,\Omega$ --- यानी कोई एक मान दोनों पर नहीं बैठता। कहानी: बढ़ती धारा
तंतु को गरम कर देती है, और गरम धातु ज़्यादा विरोध करती है; चित्र
इसलिए मुड़ता है कि पुरज़ा काम करते-करते ख़ुद बदल जाता है।
\end{solution}

\begin{solution}{exo:g8:ohms-law:11}
एक फंदा, एक $I$: दोनों गिरावटें जुड़कर \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का धक्का बनती हैं,
$6 = 10 I + 20 I = 30 I$, इसलिए $I = \qty{0.2}{A}$. फिर $U_{10} =
10 \times 0.2 = \qty{2}{V}$ और $U_{20} = 20 \times 0.2 =
\qty{4}{V}$ --- जिनका जोड़ \qty{6}{V}, जैसा फंदे का नियम माँगता है।
\end{solution}

\begin{solution}{exo:g8:ohms-law:12}
हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} \qty{6}{V} पर फैली है: $I_{10} = 6 \div 10 = \qty{0.6}{A}$,
$I_{20} = 6 \div 20 = \qty{0.3}{A}$; कुल \qty{0.9}{A}। और वह अकेला
बदली: $R = 6 \div 0.9 \approx 6.7\,\Omega$ --- दोनों सदस्यों से
\emph{कम}। सड़क की ज़बान में: एक ही दो क़स्बों के बीच दो सड़कें किसी
भी अकेली सड़क से ज़्यादा आवाजाही ढोती हैं; \omterm{def:g5:series-parallel-circuits:parallel}{समांतर में} जुड़ती हर नई
सड़क आना-जाना आसान कर देती है, इसलिए टोली का विरोध उसके सबसे कमज़ोर
सदस्य से भी नीचे गिर जाता है।
\end{solution}

\begin{solution}{pb:g8:ohms-law:1}
\textbf{1.} आपूर्ति, \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} और वह पुरज़ा एक ही श्रेणी फंदे में ---
यानी सड़क में खड़ा नाका; और \omterm{def:g8:voltage-voltmeter:voltmeter}{वोल्टमीटर} उस पुरज़े पर --- यानी एक ओर खड़ा
सर्वेक्षक।
\textbf{2.} $R = 4.5 \div 0.15 = 30\,\Omega$.
\textbf{3.} $33$ का पाँच प्रतिशत क़रीब $1.65$ है: स्वीकार्य पट्टी
$31.35$ से $34.65\,\Omega$ तक। पहले नमूने का $30\,\Omega$ उससे बाहर
गिरता है: अस्वीकार।
\textbf{4.} $I = 4.5 \div 150 = \qty{0.03}{A} = \qty{30.0}{mA}$.
\textbf{5.} हर पंक्ति $R = U/I = 30\,\Omega$ लौटाती है ($1.5 \div
0.05$, $3.0 \div 0.10$, \dots): अटल अनुपात, और सीधी रेखा वाला चित्र
--- एक ही मान सब पर बैठता है: $30\,\Omega$.
\textbf{6.} अनुपात चलते हैं $5$, $6.7$, $8.3$, $10\,\Omega$: हर
क़दम के साथ चढ़ते हुए --- यानी धारा बढ़ने के साथ बढ़ता \omterm{def:g8:ohms-law:resistance}{प्रतिरोध}; कोई
एक $R$ है ही नहीं।
\textbf{7.} धारा बढ़ने के साथ तंतु दहकने की ओर गरम होता जाता है, और
गरम धातु ज़्यादा विरोध करती है --- इसलिए चित्र ऊपर की ओर मुड़ जाता है।
इस तरह मेज़ की सीधी-रेखा वाली परख ठीक उन्हीं पुरज़ों को पास करती है
जिनका $R$ अपनी जगह टिका रहता है: यानी वह बनावट से ही \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} पहचानने
वाला यंत्र है।
\textbf{8.} टिके हुए \omterm{def:g3:temperature-thermometers:temperature}{ताप} वाली शर्त: \omterm{def:g8:ohms-law:resistance}{ओम} का नियम समानुपात का वादा
\emph{टिके हुए \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर} करता है --- चित्र लेते-लेते किसी पुरज़े को गरम
कर दो और ईमानदार \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} भी बहक जाता है।
\textbf{9.} $R = 12 \div 0.03 = 400\,\Omega$.
\textbf{10.} $U = R I = 25 \times 9.2 = \qty{230}{V}$ --- ठीक
मेन: कुंडली उतनी ही खींचती है जितनी \omterm{def:g8:ohms-law:resistance}{ओम} का नियम उसे बाँटता है, और
\omterm{def:g7:short-circuits-safety:fuse}{फ़्यूज़} का हाशिया वैसे ही काम कर रहा है जैसा डिज़ाइन किया गया था।
यहाँ ``सिर्फ़'' कुछ भी नहीं है; सब अंकगणित है।
\textbf{11.} वह जंपर जोड़ने वाले तार बनने के लिए जन्मा है --- यानी
समतल सड़क, जो धक्का ज़रा भी ख़र्च नहीं करती। पर आपूर्ति के दोनों
सिरों पर ग़लत जगह लगा दो तो उसका लगभग-शून्य \omterm{def:g8:ohms-law:resistance}{ओम} उसे बेहतरीन \omterm{def:g7:short-circuits-safety:device}{लघुपथन} बना
देता है: वह खलनायक हमेशा से ग़लत जगह लगा एक बहुत अच्छा तार ही था।
\textbf{12.} जैसे: ``नियम: $U = R I$, यानी धक्का बराबर विरोध गुणा
क़तार। उसका चित्र: मूल बिंदु से गुज़रती एक सीधी रेखा, जो ज़्यादा
विरोध करने वालों के लिए ज़्यादा खड़ी होती है। और उस पर दस्तख़त सिर्फ़
वही पुरज़ा कर सकता है जो टिके हुए \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर हो --- गरमी \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} दोबारा
लिख देती है, और बुख़ार में लिए गए चित्र कुछ भी साबित नहीं करते।''
\end{solution}
