\chapter{प्रतिरोध और ओम का नियम}\label{ch:g8:ohms-law}

पूरी बिजली की कहानी के पीछे-पीछे एक सवाल चलता आया है: कुछ पुरज़े उस
क़तार को दौड़ लगाने देते हैं, और कुछ उसे गला-गलाकर बूँद भर कर देते हैं
--- लैंप \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} असमान बाँटते हैं, पतले तार गरम होते हैं, मोटे ठंडे
रहते हैं। अब उस विरोध को ख़ुद अपना नाम, अपना \omterm{def:g6:measurement-in-science:measuring}{मात्रक} और --- चार साल के
परिपथों के सिरमौर के तौर पर --- इस पुस्तक का सबसे काम का नियम मिल जाता
है: तीन अक्षर, जो धक्के, क़तार और विरोध को बीजगणित की एक ही पंक्ति में
बाँध देते हैं।

\section{प्रतिरोध}

\begin{definition}[प्रतिरोध]\label{def:g8:ohms-law:resistance}
किसी पुरज़े का \emph{प्रतिरोध}\index{प्रतिरोध} $R$ यह मापता है कि वह
\omterm{def:g7:circuits-series-parallel:current}{विद्युत धारा} का कितने ज़ोर से विरोध करता है: किसी दिए हुए \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} पर
प्रतिरोध जितना बड़ा, उसमें से गुज़रने वाली क़तार उतनी ही कमज़ोर। इसका
\omterm{def:g6:measurement-in-science:measuring}{मात्रक} \emph{ओम}\index{ओम} (प्रतीक $\Omega$, यानी यूनानी बड़ा ओमेगा)
है, जो उस अध्यापक-भौतिकशास्त्री के सम्मान में रखा गया जिसने घर पर बने
तारों और वीरतापूर्ण धैर्य से इस अध्याय का नियम खोज निकाला।
\end{definition}

\begin{definition}[प्रतिरोधक]\label{def:g8:ohms-law:resistor}
\emph{प्रतिरोधक}\index{प्रतिरोधक} वह पुरज़ा है जिसे किसी चुने हुए,
टिके हुए \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} के साथ बनाया जाता है --- यानी एक अंशांकित अड़चन, जो
\omterm{def:g8:ohms-law:resistance}{ओम} के अंशों से लेकर लाखों \omterm{def:g8:ohms-law:resistance}{ओम} तक बिकती है। परिपथ प्रतिरोधकों से धाराएँ
जान-बूझकर तय करते हैं: उस क़तार को साधना जो किसी नाज़ुक पुरज़े को जला
देती, मंद करना, बाँटना, समायोजित करना। उन पर बनी रंगीन धारियाँ उनका
मान बोल देती हैं; और उनका प्रतीक एक सादा आयत है।
\end{definition}

\begin{example}[तुम्हारे आस-पास के प्रतिरोध]\label{ex:g8:ohms-law:values}
मोटे ताँबे के तार का एक \omterm{def:g2:measuring-length:units}{मीटर}: \omterm{def:g8:ohms-law:resistance}{ओम} के सौवें हिस्से --- समतल सड़क।
दहकता टॉर्च का \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}: कुछ \omterm{def:g8:ohms-law:resistance}{ओम}। छोटी बिजली की \omterm{ex:g6:circuits-and-safety:motor}{मोटर}: कुछेक \omterm{def:g8:ohms-law:resistance}{ओम}। केतली की
गरम करने वाली कुंडली: क़रीब $25\,\Omega$. इंसान की सूखी त्वचा, एक हाथ
से दूसरे तक: दसियों हज़ार \omterm{def:g8:ohms-law:resistance}{ओम} --- और गीली हो तो तबाही की हद तक कम: नम
हाथों वाला पुराना नियम दरअसल \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} का बयान है। खुला \omterm{ex:g3:first-electric-circuit:switch}{स्विच}: हर मापन
से परे \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} --- यानी अनंत अड़चन।
\end{example}

\section{प्रयोग}

\begin{method}[किसी पुरज़े का $U$--$I$ चित्र लेना]\label{met:g8:ohms-law:portrait}
एक \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक}, एक समायोज्य आपूर्ति (या सेलों की बढ़ती क़तार), और दोनों
उपकरण:
\begin{enumerate}
  \item आपूर्ति, \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} और \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} को \omterm{def:g4:series-circuits:series}{श्रेणी में} जोड़ो; और
        \omterm{def:g8:voltage-voltmeter:voltmeter}{वोल्टमीटर} को \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} पर टाँक दो;
  \item सबसे छोटा \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} लगाओ; और जोड़ी ($U$, $I$) दर्ज करो;
  \item \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} क़दम दर क़दम बढ़ाओ --- एक सेल, दो, तीन --- और हर बार
        एक जोड़ी दर्ज करो;
  \item बिंदुओं को आलेखित करो, $I$ आड़े और $U$ खड़े, और देखो।
\end{enumerate}
\end{method}

\begin{example}[प्रतिरोधक का इक़बाल]\label{ex:g8:ohms-law:confession}
एक बैठक की सारणी, एक ही \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} के लिए:
\begin{center}
\begin{tabular}{l|cccc}
$U$ (\unit{V}) & $1.5$ & $3.0$ & $4.5$ & $6.0$ \\
\hline
$I$ (\unit{A}) & $0.10$ & $0.20$ & $0.30$ & $0.40$
\end{tabular}
\end{center}
धक्का दोगुना, क़तार दोगुनी; तिगुना, तो तिगुनी: $U$ और $I$
\emph{समानुपाती} हैं, और हर अनुपात $U/I$ वही अंक लौटाता है: $15$. वह
अटल अनुपात कोई इत्तेफ़ाक़ नहीं है --- वह उस \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} का \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} है,
$R = 15\,\Omega$, जो हर पंक्ति में ख़ुद को दिखा देता है।
\end{example}

\begin{proposition}[ओम का नियम]\label{prop:g8:ohms-law:law}
टिके हुए \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर किसी \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} के लिए, उस पर लगा \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} और उसमें से
गुज़रती धारा समानुपाती होते हैं, और समानुपात का स्थिरांक उसका \omterm{def:g8:ohms-law:resistance}{प्रतिरोध}
होता है:
\[
U = R \times I ,
\]
जहाँ $U$ \omterm{def:g8:voltage-voltmeter:voltage}{वोल्ट} में, $I$ \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} में और $R$ \omterm{def:g8:ohms-law:resistance}{ओम} में है। मरज़ी के मुताबिक़
पलटा हुआ: $I = U/R$ (यानी वह धारा जो कोई धक्का किसी अड़चन में से
चलाता है) और $R = U/I$ (यानी वह अड़चन, जिसे पाठों की एक ईमानदार जोड़ी
बेनक़ाब कर देती है)।
\end{proposition}

गुज़रती एक सीधी रेखा पर --- खींचा हुआ समानुपात। रेखा जितनी खड़ी,

\begin{example}[नियम सब कुछ हिसाब लगा देता है]\label{ex:g8:ohms-law:compute}
$R$ पता हो जाने के बाद यह नियम हर दिशा में जवाब देता है। केतली की
कुंडली, \qty{230}{V} मेन पर $25\,\Omega$: $I = U/R = 230 \div 25 =
\qty{9.2}{A}$ --- यानी दस \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} वाला उपकरण समझ में आ गया। किसी
\omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} को \qty{4.5}{V} से किसी नाज़ुक लैंप की धारा \qty{0.02}{A}
तक सीमित करनी है: $R = U/I = 4.5 \div 0.02 = 225\,\Omega$ --- अब
इसके ऊपर वाला मानक मान उठा लो। $60\,\Omega$ में से \qty{0.05}{A}
बहती है: उस पर \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} $U = 60 \times 0.05 = \qty{3}{V}$ होगा। एक
नियम, तीन नुस्ख़े, और पूरी कार्यशाला।
\end{example}

\begin{remark}[त्रिभुज वाली बैसाखी]\label{rem:g8:ohms-law:triangle}
कुछ सीखने वाले $U$ को $R$ तथा $I$ के ऊपर एक नन्हे त्रिभुज में लिख
लेते हैं और मनचाहे अक्षर को ढककर नुस्ख़ा पढ़ लेते हैं। बैसाखी के तौर
पर बेज़रर --- पर तुम्हारा बीजगणित अब $U = R I$ को एक ही पंक्ति में
ईमानदारी से पलट देता है, और ईमानदार तरीक़ा हर उस सूत्र तक साथ चलता है
जो तुम्हें कभी मिलेगा, चाहे त्रिभुज हो या न हो। इस साल ज़रूरत हो तो
बैसाखी टेक लो; पर चलने की योजना बनाओ।
\end{remark}

\section{प्रतिरोध क्या-क्या समझा देता है}

\begin{example}[पुरानी पहेलियाँ, एक ही बीजक पर]\label{ex:g8:ohms-law:mysteries}
नियमों वाले अध्याय का असमान \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} बँटवारा: \omterm{def:g4:series-circuits:series}{श्रेणी में} लगे लैंप एक
ही $I$ बाँटते हैं, इसलिए $U = R I$ के मुताबिक़ बड़ा $R$ बड़ा $U$ ले
जाता है --- ज़्यादा मेहनत करने वाले की बड़ी गिरावट, अब हिसाब लगाने
लायक। समांतर वाले अध्याय की वह गली हुई \omterm{def:g5:series-parallel-circuits:parallel}{शाखा}: लंबे पतले तार ने \omterm{def:g8:ohms-law:resistance}{प्रतिरोध}
जोड़ दिया, और $I = U/R$ ने उस \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} का हिस्सा सिकोड़ दिया। और \omterm{def:g7:short-circuits-safety:device}{लघुपथन}:
लगभग शून्य $R$ वाला एक बाईपास, इसलिए $I = U/R$ फट पड़ता है --- वह
भगदड़ हमेशा से लगभग-शून्य से भाग देना ही थी। चार साल की गुणात्मक
परिपथ-विद्या, और एक ही नियम का बीजक।
\end{example}

\begin{example}[तार, मीटर के हिसाब से]\label{ex:g8:ohms-law:wires}
किसी तार का \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} उसकी लंबाई के साथ बढ़ता है और उसकी मोटाई के साथ
घटता है --- लंबा पतला तार पहाड़ की तंग सड़क है, और छोटा मोटा तार
राजमार्ग। इसीलिए बिजली मिस्त्री भूखे उपकरणों के लिए मोटी केबल लगाता
है, और इसीलिए गरम करने वाली कुंडली जान-बूझकर लंबे, पतले प्रतिरोधी
मिश्रातु की बनी होती है --- और इसीलिए वही नियम, जो लैंप जलाता है,
टोस्टर गरम भी करता है: \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} ही वह जगह है जहाँ क़तार अपना धक्का
ख़र्च करती है, और ख़र्च हुआ धक्का, जैसा अगले साल \omterm{def:g5:everyday-energy:energy}{ऊर्जा} वाला अध्याय
पुष्टि करेगा, गरमाहट बन जाता है।
\end{example}

\begin{remark}[नियम का राज कहाँ ख़त्म होता है]\label{rem:g8:ohms-law:limits}
\omterm{def:g8:ohms-law:resistance}{ओम} का नियम \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} का नियम है, कोई सर्वव्यापी अधिकार नहीं। किसी
\emph{तंतु वाले लैंप} का चित्र मापो और बिंदु उस सीधी रेखा से हटकर मुड़
जाते हैं: धारा बढ़ने के साथ तंतु गरम होता है, और गरम धातु ज़्यादा
विरोध करती है --- यानी $R$ अपनी जगह टिकने से इनकार कर देता है।
डायोड, मोटरें और जीती-जागती त्वचा भी अपने-अपने ढंग से अपने चित्र मोड़
देती हैं। मूल बिंदु से गुज़रती सीधी रेखा \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} के दस्तख़त हैं,
पदार्थ के नहीं --- इसलिए किसी भी पुरज़े को इस नियम के हवाले करने से
पहले उससे उसका चित्र हमेशा माँग लो।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:1}
\omterm{def:g8:ohms-law:resistance}{प्रतिरोध} क्या मापता है? उसका \omterm{def:g6:measurement-in-science:measuring}{मात्रक} और प्रतीक बताओ, और \omterm{def:g8:ohms-law:resistance}{ओम} का नियम
उसके तीनों रूपों समेत लिखो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:2}
$30\,\Omega$ के \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} के लिए हर जोड़ी पूरी करो: $U =
\qty{6}{V}$, $I = {?}$; \ $I = \qty{0.5}{A}$, $U = {?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:3}
कोई पुरज़ा $I = \qty{0.09}{A}$ पर $U = \qty{4.5}{V}$ दिखाता है।
उसका \omterm{def:g8:ohms-law:resistance}{प्रतिरोध}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:4}
चित्र वाले प्रयोग में किसी सच्चे \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} के बिंदु कौन-सी आकृति
खींचते हैं, और ज़्यादा खड़ी रेखा का क्या मतलब है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:5}
अध्याय की सारणी से दूसरी और चौथी पंक्ति के $R$ की जाँच करो। दोनों बार
वही अंक मिलना ही पूरी बात क्यों है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:6}
केतली की कुंडली $25\,\Omega$ की है; दूसरी केतली की $35\,\Omega$ की।
उसी मेन पर कौन-सी ज़्यादा तगड़ी धारा खींचती है --- और हर एक कितनी
खींचती है (एक दशमलव अंक)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:7}
$U = R I$ से समझाओ कि एक ही धारा बाँटते श्रेणी वाले दो लैंपों में से
ज़्यादा \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} वाला लैंप बड़ा \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} क्यों लेता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:ohms-law:8}
\omterm{def:g8:ohms-law:resistance}{प्रतिरोध} की ज़बान में अनुवाद करो: \omterm{def:g7:short-circuits-safety:device}{लघुपथन} की भगदड़; \ खुले \omterm{ex:g3:first-electric-circuit:switch}{स्विच} की
पूरी नाकाबंदी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:9}
संकेत देने वाला एक नाज़ुक लैंप \qty{20}{mA} झेल सकता है। \qty{9}{V}
की \omterm{def:g3:first-electric-circuit:battery}{बैटरी} पर उसकी हिफ़ाज़त कौन-सा श्रेणी वाला \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} करेगा, अगर
ख़ुद लैंप हिस्से में से क़रीब \qty{2}{V} ले लेता है? (पहले \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक}
का हिस्सा, फिर $R$।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:10}
किसी तंतु वाले लैंप का मापा हुआ चित्र: (\qty{1}{V}, \qty{0.25}{A}),
(\qty{4}{V}, \qty{0.5}{A}). दिखाओ कि कोई एक $R$ दोनों बिंदुओं पर
नहीं बैठता, और उस मुड़ाव के पीछे की भौतिक कहानी भी बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:ohms-law:11}
\omterm{def:g4:series-circuits:series}{श्रेणी में} लगे दो \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक}, $10\,\Omega$ और $20\,\Omega$,
\qty{6}{V} पर। श्रेणी वाले नियमों और हर एक पर \omterm{def:g8:ohms-law:resistance}{ओम} के नियम से: फंदे की
धारा और हर \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} का \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} निकालो। (संकेत: एक ही धारा $I$
$6 = 10 I + 20 I$ को संतुष्ट करती है।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:ohms-law:12}
वही दोनों \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक}, अब \qty{6}{V} पर \omterm{def:g5:series-parallel-circuits:parallel}{समांतर में}। हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} की धारा
निकालो, \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का कुल भी --- और फिर, शानदार समापन के तौर पर, वह अकेला
\omterm{def:g8:ohms-law:resistance}{प्रतिरोध} $R$ भी जो \qty{6}{V} से वही कुल धारा खींचता। उसकी तुलना
दोनों प्रतिरोधकों से करो और सड़क की ज़बान में समझाओ कि दोनों की टोली
अपने किसी भी सदस्य से \emph{कम} विरोध क्यों करती है।
\end{exercise}

\section{समस्या: गुणवत्ता जाँच की मेज़}

\begin{problem}[{सप्ताहांत समस्या --- प्रतिरोधक कारख़ाने में जाँच का
दिन; चित्र, सहनशीलताएँ, और एक ऐसा पुरज़ा जो प्रतिरोधक है ही
नहीं}]\label{pb:g8:ohms-law:1}
कारख़ाने की गुणवत्ता वाली मेज़ दिन भर के उत्पादन की जाँच करती है ---
और साथ में एक भेस बदले नक़लची की भी। उपकरण तुम चलाते हो।

\medskip\noindent\textbf{भाग I --- मेज़ का अंशांकन।}
\begin{enumerate}
  \item परख वाला परिपथ खींचो (या बयान करो): आपूर्ति, \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर}, जाँच के
        नीचे वाला पुरज़ा, \omterm{def:g8:voltage-voltmeter:voltmeter}{वोल्टमीटर} --- कौन \omterm{def:g4:series-circuits:series}{श्रेणी में} है, और कौन
        किस पर?
  \item पहला नमूना \qty{4.5}{V} पर \qty{0.15}{A} गुज़ारता है। उसका
        \omterm{def:g8:ohms-law:resistance}{प्रतिरोध}?
  \item लेबल $33\,\Omega$ का दावा करता है, सहनशीलता पाँच प्रतिशत।
        स्वीकार्य पट्टी निकालो, और पहले नमूने पर फ़ैसला दो। (पहले
        $33$ का पाँच प्रतिशत।)
  \item दूसरा नमूना, जिस पर $150\,\Omega$ लिखा है: \qty{4.5}{V} पर
        उसकी धारा की भविष्यवाणी करो, ताकि मेज़ को पता रहे कि उम्मीद
        क्या रखनी है (एक दशमलव अंक, \unit{mA} में)।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- पूरे चित्र।}
\begin{enumerate}[resume]
  \item तीसरे नमूने की सारणी: $(1.5, 0.05)$, $(3.0, 0.10)$,
        $(4.5, 0.15)$, $(6.0, 0.20)$ --- \omterm{def:g8:voltage-voltmeter:voltage}{वोल्ट}, \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर}। $R$
        बिठाओ, और अंकों की उस ख़ासियत का हवाला भी दो जो एक ही मान की
        इजाज़त देती है।
  \item चौथे नमूने की सारणी: $(1.5, 0.30)$, $(3.0, 0.45)$,
        $(4.5, 0.54)$, $(6.0, 0.60)$. नक़लची को पकड़ो: कोई एक $R$
        नहीं, और बहकाव की दिशा भी बताओ।
  \item चौथा नमूना दरअसल अगली उत्पादन-पंक्ति का एक छोटा तंतु वाला
        लैंप है। उसके मुड़ाव की कहानी सुनाओ --- और यह भी कि मेज़ की
        सीधी-रेखा वाली परख ठीक-ठीक \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} पहचानने वाला यंत्र
        क्यों है।
  \item मेज़ की नियम-पुस्तिका कहती है: ``हर चित्र टिके हुए \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर ही
        लिया जाए।'' यह नियम \omterm{def:g8:ohms-law:resistance}{ओम} के नियम की किस शर्त की हिफ़ाज़त कर रहा
        है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- उपयोग विभाग।}
\begin{enumerate}[resume]
  \item एक ऑर्डर ऐसा \omterm{def:g8:ohms-law:resistor}{प्रतिरोधक} माँगता है जो \qty{12}{V} की आपूर्ति
        पर धारा \qty{30}{mA} तक सीमित कर दे (\omterm{def:g4:weight-and-mass:weight}{भार} का अपना हिस्सा
        नगण्य है)। बताने लायक मान निकालो।
  \item एक ग्राहक शिकायत करता है कि उनकी $25\,\Omega$ वाली हीटर
        कुंडली ``सिर्फ़'' \qty{9.2}{A} खींचती है, जबकि \omterm{def:g7:short-circuits-safety:fuse}{फ़्यूज़} $10$
        की इजाज़त देता है। निकालो कि उनकी शिकायत किस आपूर्ति \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर}
        की ओर इशारा करती है, और अंकगणित से उन्हें तसल्ली दो।
  \item प्रतिरूप वाली प्रयोगशाला कारख़ाने का सबसे मोटा, सबसे छोटा
        ताँबे का जंपर माँगती है, ``जिसका \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} जितना हो सके शून्य
        के क़रीब हो''। वह जंपर परिपथ की किस भूमिका के लिए जन्मा है
        --- और ग़लत जगह लगा दिया जाए तो उसका लगभग-शून्य \omterm{def:g8:ohms-law:resistance}{प्रतिरोध} उसे
        किस रोज़मर्रा खलनायक के पास ले जाता है?
  \item जाँच की बही बंद करो: तीन वाक्य --- नियम, उसका चित्र, और वह
        अकेली शर्त जिसके तहत कोई पुरज़ा उस पर दस्तख़त कर सकता है।
\end{enumerate}
\end{problem}
