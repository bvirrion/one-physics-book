\section{सफ़ेद प्रकाश, खोलकर देखा हुआ}

\begin{proposition}[सफ़ेद प्रकाश एक मिश्रण है]\label{prop:g8:colors-spectra:mixture}
आम ``सफ़ेद'' \omterm{def:g1:light-and-shadows:source}{प्रकाश} --- धूप, लैंप का उजाला --- एक साथ सारे रंगों का
मेल है। काँच के प्रिज़्म में से भेजो तो यह मेल खुल जाता है: काँच पार
करते हुए हर रंग ज़रा-ज़रा अलग मात्रा में मुड़ता है, इसलिए रंग पंखे की
तरह फैलकर लाल (सबसे कम मुड़ा हुआ) से बैंगनी (सबसे ज़्यादा मुड़ा हुआ)
तक एक लगातार पट्टी बना देते हैं। उस पट्टी को
\emph{वर्णक्रम}\index{वर्णक्रम} कहते हैं, और इस खुलने को
\emph{विक्षेपण}\index{विक्षेपण}।
\end{proposition}

\begin{example}[बिना प्रयोगशाला के वर्णक्रम]\label{ex:g8:colors-spectra:everyday}
क़ुदरत और घर की रद्दी दराज़, दोनों प्रिज़्मों से भरी हैं। इंद्रधनुष:
बारिश की हर बूँद एक नन्हा प्रिज़्म-सह-दर्पण है, और आकाश भर फैला वह
मेहराब लाखों बूँदों से विक्षेपित हुई धूप है --- और हमेशा आकाश के उसी
आधे हिस्से में जो सूर्य के \emph{सामने} हो, इसीलिए इंद्रधनुष खोजने वाले
सूर्य को पीठ देकर खड़े होते हैं। बाग़ के पाइप की महीन फुहार माँगते ही
एक इंद्रधनुष हाज़िर कर देती है। किसी सीडी का खाँचेदार चेहरा छत पर
नन्हे \omterm{prop:g8:colors-spectra:mixture}{वर्णक्रम} फेंकता है; और धूप वाली दोपहरों में \omterm{def:g5:mirrors-reflection:reflection}{दर्पण} का ढालू किनारा
दीवार पर धारियाँ खींच देता है। सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश} जहाँ भी महीन बनावट या
तिरछे काँच से मिलता है, वहीं अपने मिश्रण होने का भेद खोल देता है।
\end{example}

\begin{example}[मिश्रण को दोबारा बाँध देना]\label{ex:g8:colors-spectra:newtondisc}
यह खुलना उलटा भी चलता है। किसी चकती पर इंद्रधनुष के खंड रँगो और उसे
तेज़ी से घुमाओ: आँख, जो इतनी तेज़ी का पीछा नहीं कर पाती, हर जगह से
सारे रंग लगभग एक साथ पा लेती है --- और उस मेल को पढ़ लेती है: एक फीका,
धुला-सा सफ़ेद-सलेटी। जिस महान भौतिकशास्त्री ने सबसे पहले प्रिज़्म से
धूप खोली थी, उसने दोबारा बँधने का तर्क करने के लिए ठीक ऐसी ही चकती
घुमाई --- और उसका सबसे साहसी दाँव एक दूसरा प्रिज़्म था, जो उस \omterm{prop:g8:colors-spectra:mixture}{वर्णक्रम}
को पकड़कर वापस एक ही सफ़ेद \omterm{def:g6:light-propagation:ray}{किरणपुंज} में मोड़ देता था: मिश्रण दोनों ओर
से साबित।
\end{example}

\section{सेब लाल क्यों है}

\begin{proposition}[किसी वस्तु का रंग]\label{prop:g8:colors-spectra:objectcolor}
किसी \omterm{def:g7:light-sources-propagation:coats}{अपारदर्शी} वस्तु का रंग घटाव से बनता है। \omterm{def:g1:light-and-shadows:source}{प्रकाश} पाकर वह वस्तु उस
मेल का एक हिस्सा \emph{सोख} लेती है और बाक़ी विसरित कर देती है --- और
विसरित हुआ वही बचा हिस्सा तुम्हारी आँख तक पहुँचकर रंग का नाम रखता है।
लाल सेब सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश} के लगभग सारे रंग सोख लेता है और मुख्य रूप से लाल
लौटाता है। सफ़ेद पन्ना लगभग सब कुछ लौटा देता है; और काला कोट लगभग कुछ
भी नहीं --- पूरा का पूरा सोखते हुए (और उसी हिसाब से गरम होते हुए, जैसा
गरमियों में खड़ी गहरे रंग की कारें गवाही देती हैं)। रंग छिलके
\emph{में} नहीं है: वह उस \omterm{def:g1:light-and-shadows:source}{प्रकाश} को छिलके का जवाब है जो उसे पेश किया
गया था।
\end{proposition}

\begin{example}[खेल-कक्ष वाली परख]\label{ex:g8:colors-spectra:gym}
अब वह नीले उजाले वाला खेल-कक्ष। लाल सेब का छिलका वही करता है जो वह
हमेशा करता है: लाल के सिवा सब कुछ सोख लेता है और लाल लौटाने को तैयार
खड़ा रहता है --- पर नीली प्रकाश-बौछार में लौटाने लायक लाल है ही नहीं।
कुछ विसरित नहीं, कुछ मिला नहीं: सेब काला दिखता है। उसी \omterm{def:g1:light-and-shadows:source}{प्रकाश} में
सफ़ेद पन्ना वही लौटाता है जो उसे मिलता है: वह नीला दिखता है। रंगमंच का
नियम: कोई वस्तु सिर्फ़ वही रंग लौटा सकती है जो \omterm{def:g1:light-and-shadows:source}{प्रकाश} लेकर आया ---
\omterm{def:g1:light-and-shadows:source}{प्रकाश} बदलो, और हॉल की हर पोशाक बदल जाती है।
\end{example}

\begin{method}[किसी भी प्रकाश में रंग की भविष्यवाणी]\label{met:g8:colors-spectra:predict}
\begin{enumerate}
  \item गिनाओ कि \omterm{def:g1:light-and-shadows:source}{प्रकाश} क्या लेकर आया है (सफ़ेद: सब कुछ; रंगीन लैंप:
        मुख्य रूप से अपना ही रंग);
  \item गिनाओ कि वह वस्तु सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश} में क्या लौटाती है --- उसका
        दिन के उजाले वाला रंग ही उसका नाम रख देता है;
  \item दोनों सूचियों का सर्वनिष्ठ भाग लो: वस्तु वही विसरित करती है जो
        दोनों में आता हो;
  \item सर्वनिष्ठ भाग ख़ाली हो: वस्तु काली दिखती है।
\end{enumerate}
लाल \omterm{def:g1:light-and-shadows:source}{प्रकाश} में नीली टोपी: लौटाती सिर्फ़ नीला, और पेश किया गया सिर्फ़
लाल --- सर्वनिष्ठ भाग ख़ाली: काली टोपी। सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश} में: पूरी पेशकश
नीले जवाब से मिलती है --- नीली टोपी।
\end{method}

\section{प्रकाश से रंग बनाना}

\begin{example}[तीन लैंप सारे रंग बना देते हैं]\label{ex:g8:colors-spectra:additive}
\emph{प्रकाशों} को मिलाना जोड़ता है। तीन प्रकाश-बौछारें एक-दूसरे पर
डालो --- लाल, हरी, नीली: लाल और हरी मिलकर पीला बनाती हैं; हरी और
नीली, फ़िरोज़ी सियान; नीली और लाल, मैजेंटा; और तीनों मिलकर, सफ़ेद।
सिर्फ़ तीन ठीक से चुने प्राथमिक रंगों से प्रकाश-मिश्रण लगभग हर वह रंग
बना देता है जिसका नाम आँख रख सकती है --- और इसीलिए तीन हर दमकते पर्दे
का जादुई अंक है।
\end{example}

\begin{example}[तुम्हारी जेब में बैठा पर्दा]\label{ex:g8:colors-spectra:screen}
किसी फ़ोन या टेलीविज़न के पर्दे के सफ़ेद हिस्से को तगड़े आवर्धक से
देखो: कहीं कोई सफ़ेद नहीं --- सिर्फ़ लाल, हरी और नीली नन्ही पट्टियाँ,
सब की सब दहकती हुई। पर्दा जो भी रंग कभी दिखाता है, वह इन्हीं तीन
लैंपों की नपी-तुली ख़ुराकें हैं; और हर ``सफ़ेद'' उस तिकड़ी का पूरे
सुर में गाना है, जिसे तुम्हारी आँख फ़ासले की मदद से मिला देती है। रंगों
की डिबिया उलटे ढंग से काम करती है --- रंजक सोखते हैं, इसलिए उन्हें
मिलाना \omterm{def:g1:light-and-shadows:source}{प्रकाश} जोड़ता नहीं, घटाता है --- और इसीलिए सारे रंग मिलाओ तो
गँदला अँधेरा मिलता है, जबकि सारे \omterm{def:g1:light-and-shadows:source}{प्रकाश} मिलाओ तो सफ़ेद; चित्रकार वाली
पूरी कहानी कला की \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} और रसायन की है।
\end{example}

\begin{remark}[सूर्यास्त का पूरा दाम चुका दिया गया]\label{rem:g8:colors-spectra:sunset}
पिछले साल का \omterm{def:g1:day-and-night:daynight}{सूर्यास्त} वाला वादा आज चुकता होता है। \omterm{def:g2:air-around-us:air}{हवा} धूप को विसरित
करती है --- पर बराबरी से नहीं: मेल का नीला सिरा लाल के मुक़ाबले कहीं
ज़्यादा आसानी से बिखेरा जाता है। दिन में सूर्य से हटकर देखो और
तुम्हें वही चुराया हुआ हिस्सा दिखता है, जिसे पूरा आकाश आगे बढ़ाकर
भेजता है: नीला। और \omterm{def:g1:day-and-night:daynight}{सूर्यास्त} पर सीधे \omterm{def:g6:light-propagation:ray}{किरणपुंज} का \omterm{def:g2:air-around-us:air}{हवा} में से छूता हुआ
रास्ता सबसे लंबा होता है, इसलिए नीले की चोरी लगभग पूरी हो जाती है ---
और तुम्हारी आँख तक पहुँचता बचा हुआ हिस्सा वही लाल-नारंगी है, जबकि
गुलाबी बादल देर से हुए बिखराव पकड़ लेते हैं। नीला आकाश और लाल \omterm{def:g1:day-and-night:daynight}{सूर्यास्त}
एक ही चोरी हैं, बहीखाते के दो पहलुओं से देखी हुई।
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:4}
घटाव की ज़बान में: घास हरी क्यों है? सफ़ेद पन्ना सफ़ेद और काला
पट्ट काला क्यों है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:5}
विधि से भविष्यवाणी करो: सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश} में पीला दुपट्टा; और नीले \omterm{def:g1:light-and-shadows:source}{प्रकाश}
में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:6}
नीले खेल-कक्ष में लाल सेब काला क्यों पड़ जाता है --- जबकि उसके बग़ल
रखा सफ़ेद रूमाल नीला क्यों हो जाता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:7}
प्रकाश-मिश्रण के तीनों प्राथमिक रंग बताओ, और हर जोड़ी के मिलने का
नतीजा भी --- और तीनों के एक साथ मिलने का भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:colors-spectra:8}
आवर्धक किसी पर्दे के ``सफ़ेद'' में क्या ज़ाहिर कर देता है? मिलावट होती
कहाँ है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:colors-spectra:9}
सारे \emph{\omterm{def:g1:light-and-shadows:source}{प्रकाश}} मिलाने पर सफ़ेद क्यों मिलता है, जबकि सारे
\emph{रंग} मिलाने पर गँदला अँधेरा? बहीखाते की हर दिशा पर एक-एक वाक्य।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:colors-spectra:10}
\omterm{def:g1:day-and-night:daynight}{सूर्यास्त} का बहीखाता पूरा करो: \omterm{def:g2:air-around-us:air}{हवा} ख़ास तौर पर क्या चुराती है, चुराया
हुआ माल किसे और किस रूप में मिलता है, और सबसे लंबे छूते हुए रास्ते से
बचकर क्या निकलता है --- और यह भी कि पिछले साल वाला
\emph{ग्रहण-ग्रस्त \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}} ठीक उसी बचे हुए हिस्से से क्यों दमकता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:colors-spectra:11}
किसी दुकान का कपड़े बदलने का कमरा ``त्वचा के रंग पर फबने वाले'' गरम
लाली भरे लैंपों से रोशन है। एक ग्राहक ऐसी जैकेट ख़रीदता है जो गहरी,
गरम भूरी दिख रही थी --- और दिन के उजाले में वह फीकी हरी निकलती है।
भविष्यवाणी वाली विधि से इस धोखे को दोबारा खड़ा करो।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:colors-spectra:12}
रंगमंच का हुनर: किसी डिज़ाइनर को ऐसा बैनर चाहिए जिस पर सफ़ेद हॉल-बत्ती
में काली ज़मीन पर चमकीले, पढ़े जा सकने वाले अक्षर हों --- पर समापन की
गहरी लाल बौछार में पूरा बैनर, अक्षर और ज़मीन दोनों, एक-सा काला होकर
ग़ायब हो जाए। अक्षरों का रंग और ज़मीन का रंग चुनो (इनमें से: सफ़ेद,
लाल, हरा, नीला, काला), हर चुनाव को घटाव के नियमों से सही ठहराओ, और यह
भी समझाओ कि \emph{सफ़ेद} अक्षर --- जो लुभाने वाला चुनाव हैं --- समापन
के साथ ग़द्दारी क्यों कर देते।
\end{exercise}

\section{समस्या: प्रकाश का रंगमंच}

\begin{problem}[{सप्ताहांत समस्या --- रंगीन बौछारों में पोशाक वाला
पूर्वाभ्यास; पोशाकें, छन्ने और एक तोड़-फोड़ किया गया दृश्य;
प्रकाश-डिज़ाइनर की परीक्षा}]\label{pb:g8:colors-spectra:1}
पोशाक वाला पूर्वाभ्यास: मंद करने वालों समेत तीन प्रकाश-बौछारें (लाल,
हरी, नीली), पोशाकों का एक ताक, और तरकीबों से भरी एक पटकथा।
प्रकाश-योजना तुम्हारे हाथ में है।

\medskip\noindent\textbf{भाग I --- गरमाने वाली कवायद।}
\begin{enumerate}
  \item तीनों बौछारें पूरी: मंच किस \omterm{def:g1:light-and-shadows:source}{प्रकाश} में नहाता है, और क्यों?
  \item लाल और हरी पूरी, नीली बुझी: मंच के \omterm{def:g1:light-and-shadows:source}{प्रकाश} का नाम बताओ।
  \item उसी लाल-हरे मेल में इनके दिखने की भविष्यवाणी करो: सफ़ेद
        क़मीज़; नीला पटका; पीला गाउन।
  \item अकेली नीली बौछार में सफ़ेद पृष्ठभूमि वाला परदा --- और यह भी
        कि पृष्ठभूमि का परदा हमेशा सफ़ेद कपड़े का ही क्यों होता है।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- पोशाकों की दवाख़ाना-जाँच।}
\begin{enumerate}[resume]
  \item खलनायक का कोट पहले दृश्य (लाल बौछार) में काला दिखना चाहिए,
        और दूसरे दृश्य (सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश}) में गाढ़ा हरा। दिन के उजाले
        वाला कौन-सा रंग का कोट काम देगा, और क्यों?
  \item भूत का चोग़ा \emph{हर} रोशनी में सफ़ेद जैसा ही दिखना चाहिए।
        कौन-से रंग का चोग़ा, और हर अकेली बौछार में वह असल में क्या
        दिखाएगा?
  \item किसी अभिनेता का लाल दुपट्टा एक पल के लिए ``घुल जाना'' चाहिए
        --- यानी काले परदे से अलग न पहचाना जाए। इस तरकीब के लिए
        कौन-सी बौछारें जल सकती हैं, और लाल बौछार के शामिल होते ही वह
        क्यों नाकाम हो जाती है?
  \item कार्यक्रम-पत्र के सुनहरे अक्षर दोपहर में शानदार लगे थे पर
        नीले-भारी शाम वाले पूर्वनियोजन में गँदले पड़ गए। सोने को गरम
        पीला मानकर समझाओ।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- तोड़-फोड़।}
पहला दृश्य: पूरे सफ़ेद \omterm{def:g1:light-and-shadows:source}{प्रकाश} में सजावट का इंद्रधनुषी मेहराब (रँगी हुई
धारियाँ: लाल, नारंगी, पीली, हरी, नीली, बैंगनी)।
\begin{enumerate}[resume]
  \item कोई शरारती सफ़ेद पूर्वनियोजन बदलकर शुद्ध लाल कर देता है। अब
        दर्शकों को धारी दर धारी क्या दिखता है?
  \item निर्देशक मौक़े पर ही लाल \emph{और} नीली बौछारें एक साथ माँग
        लेता है। कौन-सी धारियाँ जी उठती हैं, और मैजेंटा-सी बौछार हरी
        धारी के साथ क्या करती है?
  \item मंच-प्रबंधक पूछता है कि मेहराब को किसी बहुत तेज़ अकेली हरी
        बौछार से ``वापस सामान्य'' क्यों नहीं किया जा सकता। उस अकेले
        \omterm{def:g1:light-and-shadows:source}{प्रकाश} का नाम लेकर उत्तर दो जो छहों धारियाँ दिखा देता है।
  \item डिज़ाइनर की परीक्षा का उत्तर लिखो: \omterm{def:g1:light-and-shadows:source}{प्रकाश} के रंगमंच के दोनों
        नियम --- एक इस बारे में कि मिलाए जाने पर \omterm{def:g1:light-and-shadows:source}{प्रकाश} क्या करते
        हैं, और एक इस बारे में कि वस्तुएँ अपने मिले हुए के साथ क्या
        करती हैं --- हर एक एक वाक्य में।
\end{enumerate}
\end{problem}
