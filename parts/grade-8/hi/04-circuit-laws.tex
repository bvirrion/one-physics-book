\chapter{परिपथों में धारा और विभवांतर के नियम}\label{ch:g8:circuit-laws}

दो उपकरण, दो राशियाँ --- और अब, आख़िरकार, संविधान। तुम आगे जो भी परिपथ
बनाओगे, टॉर्च से लेकर किसी शहर के जाल तक, वह चार छोटे नियम मानता है:
दो धारा के बहीखाते के बारे में, और दो \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} के बारे में। इन्हें और
थोड़े-से बीजगणित को हथियार बनाकर तुम वे पाठ भी निकाल सकते हो जिन्हें
तुमने अभी मापा ही नहीं --- और झूठ बोलता कोई भी मापन पकड़ भी सकते हो।

\section{धारा के नियम}

\begin{proposition}[श्रेणी फंदे के लिए धारा का नियम]\label{prop:g8:circuit-laws:seriesI}
अकेले फंदे के साथ-साथ \omterm{def:g8:intensity-ammeter:intensity}{तीव्रता} हर जगह एक ही रहती है:
\[
I_{\text{बैटरी}} = I_{\text{लैंप 1}} = I_{\text{लैंप 2}} =
\dots
\]
एक सड़क, एक क़तार --- यानी पिछले अध्याय के नाके, अब संविधान का नियम
बनकर।
\end{proposition}

\begin{proposition}[संधि का नियम]\label{prop:g8:circuit-laws:junction}
किसी भी \omterm{def:g5:series-parallel-circuits:parallel}{संधि} पर भीतर बहती तीव्रताओं का जोड़ बाहर बहती तीव्रताओं के
जोड़ के बराबर होता है: दो \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} में बँटती किसी मुख्य सड़क के लिए,
\[
I = I_1 + I_2 .
\]
किसी दोराहे पर आवेश न बनता है, न मिटता है, न जमा होता है: जो पहुँचता
है, वही निकल जाता है। (यह चाहे जितनी \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} के लिए सच है --- आने
वालों का जोड़ लो, जाने वालों का जोड़ लो, और बहीखाता बराबर बैठ जाता
है।)
\end{proposition}

\section{विभवांतर के नियम}

\begin{proposition}[श्रेणी फंदे के लिए विभवांतर का नियम]\label{prop:g8:circuit-laws:seriesU}
किसी श्रेणी फंदे के गिर्द \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} उपकरणों में बँट जाता है,
और उनके हिस्से ठीक-ठीक जुड़कर पूरे हो जाते हैं:
\[
U_{\text{बैटरी}} = U_1 + U_2 + \dots
\]
जलप्रपात वाला बिंब बहीखाता रखता है: \omterm{def:g3:first-electric-circuit:battery}{बैटरी} उस क़तार को अपनी पूरी ऊँचाई
तक उठाती है, और उपकरण उस ऊँचाई को आपस में बाँटकर बूँद-बूँद ख़र्च कर
देते हैं --- जबकि अच्छे तार कुछ भी ख़र्च नहीं करते।
\end{proposition}

\begin{proposition}[समांतर शाखाओं के लिए विभवांतर का नियम]\label{prop:g8:circuit-laws:parallelU}
\omterm{def:g5:series-parallel-circuits:parallel}{समांतर में} लगे उपकरण उन्हीं दो संधियों पर फैले होते हैं --- इसलिए एक
ही \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} उन सबकी सेवा करता है:
\[
U_1 = U_2 = U_{\text{दोनों संधियों के बीच}} .
\]
हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} को दोनों संधियों के बीच का पूरा धक्का मिलता है --- और इसीलिए
\omterm{def:g5:series-parallel-circuits:parallel}{समांतर में} लगे लैंप अपनी पूरी नामिक चमक से जलते हैं, और इसीलिए
तुम्हारे घर का हर सॉकेट वही \qty{230}{V} पेश करता है।
\end{proposition}

\begin{example}[हिस्से असमान क्यों?]\label{ex:g8:circuit-laws:unequal}
\omterm{def:g4:series-circuits:series}{श्रेणी में} लगे दो अलग-अलग लैंप एक ही धारा ढोते हैं (पहला नियम) फिर भी
\omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} असमान बाँटते हैं --- \qty{2.1}{V} के मुक़ाबले \qty{2.4}{V}।
बड़ा हिस्सा उसी लैंप को जाता है जो उस क़तार का ज़्यादा विरोध करता है:
जो ज़्यादा मेहनत करता है, उसी की गिरावट बड़ी होती है। ``ज़्यादा विरोध
करना'' का ठीक-ठीक मतलब क्या है, और उसे मापा कैसे जाए, यही अगले अध्याय
का पूरा विषय है --- और इस अध्याय के नियम उसी का मंच हैं।
\end{example}

\section{मापने से पहले हिसाब लगाना}

\begin{method}[चारों नियमों से परिपथ हल करना]\label{met:g8:circuit-laws:solve}
कुछ ज्ञात पाठों वाला कोई परिपथ दिया हो:
\begin{enumerate}
  \item चित्र खींचो और हर ज्ञात $I$ तथा $U$ पर निशान लगाओ;
  \item लागू होने वाले नियम समीकरणों के तौर पर लिखो --- धाराओं के लिए
        \omterm{def:g5:series-parallel-circuits:parallel}{संधि} वाले जोड़, विभवांतरों के लिए फंदे वाले जोड़, श्रेणी की
        $I$ तथा समांतर की $U$ के लिए बराबरी;
  \item अज्ञातों के लिए हल करो (तुम्हारे बीजगणित पाठ्यक्रम के समीकरण
        अपनी रोटी कमाते हुए);
  \item समझदारी की जाँच: कोई लैंप \omterm{def:g3:first-electric-circuit:battery}{बैटरी} की पेशकश से ज़्यादा न पा रहा
        हो, और किसी \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} की धारा मुख्य सड़क वाली से बड़ी न हो।
\end{enumerate}
\end{method}

\begin{example}[हल करके दिखाया: मिला-जुला परिपथ]\label{ex:g8:circuit-laws:worked}
\qty{6}{V} की एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी} लैंप A को एक समांतर जोड़ी (B और C) के साथ
\omterm{def:g4:series-circuits:series}{श्रेणी में} खिलाती है। मापा हुआ: $I_{\text{बैटरी}} = \qty{0.5}{A}$,
$I_B = \qty{0.3}{A}$, $U_A = \qty{2.5}{V}$. बाक़ी सब निकालो। \omterm{def:g5:series-parallel-circuits:parallel}{संधि} का
नियम: $I_C = 0.5 - 0.3 = \qty{0.2}{A}$. श्रेणी वाला धारा नियम: लैंप A
पूरी \qty{0.5}{A} ढोता है। फंदे वाला \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} नियम: उस जोड़ी को
$U_{\text{जोड़ी}} = 6 - 2.5 = \qty{3.5}{V}$ मिलते हैं; और समांतर वाला
\omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} नियम: $U_B = U_C = \qty{3.5}{V}$. परिपथ का हर उपकरण-पाठ अब
ज्ञात है --- चार नियम, और हर एक पर बीजगणित की एक पंक्ति।
\end{example}

\begin{example}[नियम, झूठ पकड़ने वाले यंत्र की तरह]\label{ex:g8:circuit-laws:liedetector}
कोई प्रयोगशाला-रिपोर्ट दावा करती है: \omterm{def:g3:first-electric-circuit:battery}{बैटरी} \qty{4.5}{V}; और \omterm{def:g4:series-circuits:series}{श्रेणी
में} लगे दो लैंप \qty{2.1}{V} तथा \qty{2.9}{V} पर। फंदे वाला नियम
हिस्सों को जोड़ देता है: \qty{4.5}{V} की थैली से \qty{5.0}{V} का
ख़र्च --- नामुमकिन; किसी ने कोई सुई-पट्टी ग़लत पढ़ी। एक और रिपोर्ट:
\omterm{def:g5:series-parallel-circuits:parallel}{संधि} में \qty{0.40}{A} भीतर, और \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} से \qty{0.25}{A} तथा
\qty{0.20}{A} बाहर --- दोराहा \qty{0.05}{A} ``बना'' रहा है:
अस्वीकार। इन नियमों से पहले ग़लत पाठ सही पाठों जितने ही अच्छे लगते थे;
अब संविधान हर पन्ने का लेखा करता है।
\end{example}

\begin{remark}[बैटरियाँ भी इन्हीं नियमों के अधीन]\label{rem:g8:circuit-laws:batteries}
नाक-से-पूँछ वाला नियम दरअसल श्रेणी का \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} नियम ही है, अपने सबसे
पुराने कपड़ों में: \omterm{def:g4:series-circuits:series}{श्रेणी में} लगी बैटरियाँ अपने \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} जोड़ लेती हैं
($1.5 + 1.5 + 1.5 = \qty{4.5}{V}$ --- चपटी \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का भेद), और उलटा बैठा
एक सेल \emph{घट} जाता है, क्योंकि उसका धक्का बाक़ियों से लड़ने में
ख़र्च होता है। ये नियम देने वालों और ख़र्च करने वालों, दोनों पर हुक्म
चलाते हैं; फ़र्क़ सिर्फ़ उनके योगदान के चिह्न का है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:1}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:2}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:3}
\qty{9}{V} की \omterm{def:g3:first-electric-circuit:battery}{बैटरी} \omterm{def:g4:series-circuits:series}{श्रेणी में} लगे तीन लैंप खिलाती है; उनमें से दो

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:4}

घर के सॉकेटों के बारे में कौन-सा रोज़मर्रा तथ्य यही नियम है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:5}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:circuit-laws:6}
\omterm{def:g4:series-circuits:series}{श्रेणी में} लगे दो अलग-अलग लैंप एक ही धारा ढोते हैं पर \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} $1.8$

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:circuit-laws:11}
दो \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} वाला परिपथ: पहली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} में लैंप E और F \omterm{def:g4:series-circuits:series}{श्रेणी में} हैं; दूसरी
\omterm{def:g5:series-parallel-circuits:parallel}{शाखा} में अकेला लैंप G। \omterm{def:g3:first-electric-circuit:battery}{बैटरी} \qty{6}{V}; $U_E = \qty{2.2}{V}$. $U_F$
और $U_G$ निकालो, और हर क़दम के लिए एक नियम का हवाला भी दो।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:circuit-laws:12}
पूरा लेखा: \qty{6}{V} की \omterm{def:g3:first-electric-circuit:battery}{बैटरी}; लैंप P एक \omterm{def:g5:series-parallel-circuits:parallel}{संधि} के साथ \omterm{def:g4:series-circuits:series}{श्रेणी में};
शाखाएँ Q और R फिर मिलकर लौट आती हैं। मापा हुआ: $I_P = \qty{0.6}{A}$,
$I_Q = \qty{0.45}{A}$, $U_Q = \qty{2.3}{V}$. परिपथ की बाक़ी हर धारा
और हर \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} निकालो ($I_R$, $U_P$, $U_R$), और हर पंक्ति के पीछे का
नियम भी बताओ --- फिर एक ऐसा फ़ालतू ``मापन'' गढ़ो जिसे कोई लापरवाह
विद्यार्थी दर्ज कर सकता था और जिसे तुम्हारा तैयार हल फ़ौरन दोषी ठहरा
देता।
\end{exercise}

\section{समस्या: सीलबंद पहेली-डिब्बे}

\begin{problem}[{सप्ताहांत समस्या --- इलेक्ट्रॉनिकी क्लब की सीलबंद
डिब्बों वाली चुनौती; चार नियम बनाम चार पहेलियाँ; महान
लेखा}]\label{pb:g8:circuit-laws:1}
क्लब सादे परिपथों को डिब्बों में सीलकर सिर्फ़ उपकरण जोड़ने के ख़ाने
खुले छोड़ देता है। टोलियों को भीतर का हाल निकालना है --- हाथ में
संविधान लेकर। मेज़ की \omterm{def:g3:first-electric-circuit:battery}{बैटरी} हर जगह \qty{6}{V} की है।

\medskip\noindent\textbf{भाग I --- डिब्बा A: दो सिरे, दो लैंप।}
लेबल इतना क़बूल करता है: दो एक जैसे लैंप और कुछ नहीं, या तो \omterm{def:g4:series-circuits:series}{श्रेणी में}
या \omterm{def:g5:series-parallel-circuits:parallel}{समांतर में}।
\begin{enumerate}
  \item \omterm{def:g3:first-electric-circuit:battery}{बैटरी} से जोड़ने पर डिब्बा A \qty{0.2}{A} खींचता है; जबकि पता
        है कि ऐसा अकेला लैंप \qty{6}{V} पर क़रीब \qty{0.4}{A} खींचता
        है। श्रेणी या समांतर? धारा वाले किसी नियम से तर्क करो।
  \item भविष्यवाणी करो कि \emph{दूसरे} बंदोबस्त में डिब्बा कितनी
        खींचता।
  \item असली बंदोबस्त में भीतर के हर लैंप को कितना \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} मिलता है,
        और किस \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} नियम से?
  \item किस बंदोबस्त में छिपे लैंप अपनी पूरी नामिक चमक से जलते ---
        और डिब्बे के झरोखे से देखने पर वे साफ़-साफ़ वैसे क्यों नहीं
        जल रहे?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- डिब्बा B: तीन लैंपों की पहेली।}
डिब्बा B इतना क़बूल करता है: एक लैंप (X) एक समांतर जोड़ी (Y, Z ---
ज़रूरी नहीं कि एक जैसे हों) के साथ \omterm{def:g4:series-circuits:series}{श्रेणी में} है।
\begin{enumerate}[resume]
  \item टोली $I_{\text{बैटरी}} = \qty{0.5}{A}$ मापती है, और Y के अपने
        परख-ख़ाने पर $I_Y = \qty{0.35}{A}$. नियमों का हवाला देते हुए
        $I_Z$ और $I_X$ निकालो।
  \item X पर लगा \omterm{def:g8:voltage-voltmeter:voltmeter}{वोल्टमीटर}: \qty{2.8}{V}. समांतर जोड़ी का \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर}
        निकालो --- और $U_Y$ तथा $U_Z$ भी।
  \item Y और Z एक ही \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} पर अलग-अलग धाराएँ ढोते हैं।
        \cref{ex:g8:circuit-laws:unequal} की ज़बान में यह दोनों लैंपों
        के बारे में क्या ज़ाहिर करता है?
  \item टोली एक झरोखे से Y को खोलकर निकाल लेती है। सड़कों और उनके
        विरोध से तर्क करते हुए भविष्यवाणी करो कि नई
        $I_{\text{बैटरी}}$ किस ओर बदलेगी (चढ़ेगी, गिरेगी, या टिकी
        रहेगी)।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- डिब्बा C: बेईमानी।}
\begin{enumerate}[resume]
  \item डिब्बा C बनाने वाला रिपोर्ट करता है: ``\omterm{def:g3:first-electric-circuit:battery}{बैटरी} \qty{6}{V};
        भीतर \omterm{def:g4:series-circuits:series}{श्रेणी में} लगे दो लैंप \qty{2.5}{V} तथा \qty{4.1}{V} पर;
        और भीतर की \omterm{def:g5:series-parallel-circuits:parallel}{संधि} को \qty{0.3}{A} मिलती है जबकि वह \qty{0.2}{A}
        तथा \qty{0.15}{A} भेजती है।'' इस रिपोर्ट को दो बार दोषी
        ठहराओ, हर सज़ा पर एक नियम।
  \item दोनों में से एक \omterm{def:g8:voltage-voltmeter:voltage}{विभवांतर} पाठ बाद में ईमानदार निकलता है। अगर
        वह \qty{2.5}{V} वाला है, तो उसका साथी सचमुच कितना होना
        चाहिए?
  \item बनाने वाला क़बूल करता है कि \omterm{def:g5:series-parallel-circuits:parallel}{संधि} वाले अंक ``उदारता से गोल
        किए गए थे''। अगर \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} के दोनों पाठ \qty{0.2}{A} और
        \qty{0.15}{A} ईमानदार हैं, तो मुख्य सड़क सचमुच कितनी ढो रही
        होगी?
  \item चारों नियम सीलबंद डिब्बों में बेईमानी को हारा हुआ खेल क्यों
        बना देते हैं? एक वाक्य।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- संविधान।}
\begin{enumerate}[resume]
  \item क्लब का घोषणापत्र लिखो: चार पंक्तियों में चारों नियम, और हर
        एक के साथ वह रोज़मर्रा बिंब भी जो उसे ढोता है (सड़कें, नाके,
        जलप्रपात, दोराहे)।
\end{enumerate}
\end{problem}
