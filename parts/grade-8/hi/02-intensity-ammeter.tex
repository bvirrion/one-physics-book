\chapter{धारा की तीव्रता और ऐमीटर}\label{ch:g8:intensity-ammeter}

``तेज़ धारा, कमज़ोर धारा'' --- हम बरसों इसी तरह बात करते आए हैं, और
चमक तथा पकड़ की ख़ुराक अंदाज़े से तय करते आए हैं। आज अंदाज़ा सेवामुक्त
हो जाता है। धारा की ताक़त को एक नाम, एक \omterm{def:g6:measurement-in-science:measuring}{मात्रक} और एक उपकरण मिल जाता
है; और वह उपकरण सबसे पहले जो साबित करता है, वह ऐसा नियम है जिस पर
तुम्हें बहुत पहले से शक़ था, पर जिसे तुमने एक बार भी मापा नहीं था।

\section{तीव्रता और उसका मात्रक}

\begin{definition}[तीव्रता]\label{def:g8:intensity-ammeter:intensity}
\omterm{def:g7:circuits-series-parallel:current}{विद्युत धारा} की \emph{तीव्रता}\index{तीव्रता} $I$ यह मापती है कि वह
क़तार कितनी तगड़ी है: यानी परिपथ के किसी बिंदु से हर सेकंड कितना
विद्युत आवेश परेड करके गुज़रता है --- यानी धारा के बहाव की दर, ठीक
वैसे ही जैसे किसी नदी की दर पानी-प्रति-सेकंड होती है। इसका \omterm{def:g6:measurement-in-science:measuring}{मात्रक}
\emph{ऐम्पियर}\index{ऐम्पियर} (\unit{A}) है, जो बिजली के नियमों के एक
महान अग्रदूत के सम्मान में रखा गया; और रोज़मर्रा के परिपथ अकसर
\emph{मिलीऐम्पियर} में बोलते हैं: $\qty{1}{mA} = \qty{0.001}{A}$,
यानी एक हज़ारवाँ हिस्सा।
\end{definition}

\begin{example}[देखते ही पहचान लेने लायक ऐम्पियर]\label{ex:g8:intensity-ammeter:values}
संकेत देने वाला छोटा-सा लैंप क़रीब \qty{0.02}{A} घूँटता है ---
\qty{20}{mA}; टॉर्च का \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} क़रीब \qty{0.3}{A}; कार की अगली बत्ती का
बड़ा लैंप \qty{5}{A}; पूरी गरजती केतली \qty{10}{A}; चालू होता कार का
इंजन अपनी \omterm{def:g3:first-electric-circuit:battery}{बैटरी} से \qty{100}{A} गटक जाता है; और बिजली की कौंध अपने उस
मिलीसेकंड भर में अपनी नाली से दसियों हज़ार \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} ठूँस देती है। और
ख़तरे का पैमाना होश उड़ा देने वाला है: इंसानी शरीर में से गुज़रती
\omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} के कुछ सौवें हिस्सों जितनी धाराएँ भी जानलेवा हो सकती हैं ---
आदर एक \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} से काफ़ी नीचे ही शुरू हो जाता है।
\end{example}

\begin{remark}[व्यक्ति और मात्रक]\label{rem:g8:intensity-ammeter:person}
\omterm{def:g6:measurement-in-science:measuring}{मात्रक} का नाम एक व्यक्ति से आया है, पर उसे लिखते हुए हम उसे आम \omterm{def:g6:measurement-in-science:measuring}{मात्रक}
की तरह बरतते हैं --- ``दो \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} की धारा'' --- जबकि प्रतीक \unit{A}
हमेशा बड़े अक्षर में लिखा जाता है, जैसा किसी वैज्ञानिक के सम्मान में
रखे हर \omterm{def:g6:measurement-in-science:measuring}{मात्रक} के साथ होता है। यह रिवाज तुम्हें जल्द ही दोबारा मिलेगा,
वह भी दो बार।
\end{remark}

\section{ऐमीटर}

\begin{definition}[ऐमीटर]\label{def:g8:intensity-ammeter:ammeter}
\emph{ऐमीटर}\index{ऐमीटर} \omterm{def:g8:intensity-ammeter:intensity}{तीव्रता} मापता है। वह उसी क़तार को गिनता है
जो \emph{उसके अपने भीतर} से गुज़रती है --- इसलिए उसे उसी फंदे में
\emph{श्रेणी} में डालना पड़ता है जिसे वह माप रहा है, ठीक सड़क पर बने
किसी नाके की तरह। वह क़तार को कम से कम छेड़ने के लिए बना है, इसलिए
धारा का विरोध लगभग करता ही नहीं --- और यही वजह है कि ग़लती से सीधे
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} के दोनों सिरों पर जोड़ दिया गया ऐमीटर सुई-पट्टी लगा हुआ \omterm{def:g7:short-circuits-safety:device}{लघुपथन}
है: किसी ऐमीटर को बरबाद करने का चिरपरिचित तरीक़ा।
\end{definition}

\begin{method}[तीव्रता मापना]\label{met:g8:intensity-ammeter:measure}
\begin{enumerate}
  \item परिपथ बंद करो; और जहाँ धारा गिननी है वहाँ फंदा \emph{तोड़ो};
  \item उस दरार में \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} डालो: धारा उसके \unit{A} वाले सिरे से भीतर
        और \emph{साझा} सिरे से बाहर, ताकि पारंपरिक दिशा उपकरण के
        अगले दरवाज़े से घुसे;
  \item कई परासों वाले उपकरण पर \emph{सबसे बड़े} परास से शुरू करो, और
        फिर तब तक नीचे उतरो जब तक पाठ सुई-पट्टी पर आराम से न बैठ जाए
        --- इससे उपकरण अचरजों से बचा रहता है;
  \item \omterm{ex:g3:first-electric-circuit:switch}{स्विच} बंद करो, पाठ पढ़ो --- और \omterm{def:g6:measurement-in-science:measuring}{मात्रक} भी लिखो।
\end{enumerate}
सुई वाले उपकरण पर उलटा जोड़ सुई को उसके ठिये से भिड़ा देता है; जबकि
अंकीय उपकरण बस एक ऋण चिह्न दिखा देते हैं, और रोज़ विद्यार्थियों को माफ़
कर देते हैं।
\end{method}

\section{पहला मापा हुआ नियम}

\begin{proposition}[एक फंदा, एक तीव्रता]\label{prop:g8:intensity-ammeter:sameI}
\omterm{def:g4:series-circuits:series}{श्रेणी परिपथ} में \omterm{def:g8:intensity-ammeter:intensity}{तीव्रता} फंदे के हर बिंदु पर एक ही रहती है। \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} को
कहीं भी हटाकर लगा दो --- लैंप से पहले, उसके बाद, या \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के बग़ल:
पाठ नहीं बदलता। जिसकी ओर पाँच साल तक बराबर चमकते \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} इशारा करते रहे,
उसी को अब यह उपकरण अंकों में कह देता है: रास्ते में कुछ भी ख़र्च नहीं
होता; एक सड़क, एक क़तार, एक $I$।
\end{proposition}

\begin{example}[नियम की बारीक इबारत पढ़ना]\label{ex:g8:intensity-ammeter:fineprint}
नियम कहता है \emph{एक ही फंदे के साथ-साथ}, यह नहीं कि \emph{तुम फंदे
के साथ चाहे जो करो}। \omterm{def:g4:series-circuits:series}{श्रेणी में} दूसरा लैंप जोड़ो और क़तार एक साथ हर
जगह कमज़ोर पड़ जाती है --- शायद \qty{0.30}{A} गिरकर \qty{0.17}{A} रह
जाए --- पर वह हर जगह \emph{बराबर} कमज़ोर पड़ती है: तीनों नाके नए अंक
पर भी एकमत रहते हैं। एक फंदा, एक \omterm{def:g8:intensity-ammeter:intensity}{तीव्रता} --- वह \omterm{def:g8:intensity-ammeter:intensity}{तीव्रता} इस वक़्त जो भी
हो।
\end{example}

\begin{example}[संधि पर एक नज़र]\label{ex:g8:intensity-ammeter:junction}
इसके बजाय समांतर जोड़ी पर नाके बिठाओ: मुख्य सड़क \qty{0.50}{A}; पहली
\omterm{def:g5:series-parallel-circuits:parallel}{शाखा} \qty{0.30}{A}; दूसरी \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} \qty{0.20}{A}। यह इशारा करता जोड़ कोई
इत्तेफ़ाक़ नहीं है --- किसी \omterm{def:g5:series-parallel-circuits:parallel}{संधि} में घुसती और उससे निकलती क़तारों का
हिसाब बराबर बैठना ही चाहिए, क्योंकि सड़क के किसी दोराहे पर आवेश न
बनता है, न मिटता है। पूरा नियम, विभवांतर वाले अपने साथी समेत, अगले से
अगले अध्याय में दरबार लगाता है; तुम्हारे उपकरण उससे अभी ही मिल चुके
हैं।
\end{example}

\begin{remark}[मल्टीमीटर]\label{rem:g8:intensity-ammeter:multimeter}
कार्यशाला की वह पीली ईंट --- \emph{मल्टीमीटर} --- \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} और कई दूसरे
उपकरणों को एक ही डिब्बे में मोड़कर रख देती है, जिस पर चुनने वाली एक
सुई-पट्टी लगी होती है। सुई \unit{A} या \unit{mA} पर, तार निशान लगे
ख़ानों में, और वह तुम्हारा \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} है --- श्रेणी वाले नियमों समेत। इस
पुस्तक के दो अध्याय अभी से उसके भीतर बसते हैं; तीसरा अगले अध्याय में
आकर बस जाएगा --- पर सावधान, \emph{उलटे} जोड़ने के नियमों के साथ।
\end{remark}

\omterm{def:g5:series-parallel-circuits:parallel}{समांतर में} लगे दो एक जैसे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} \qty{0.30}{A} ढोते हैं। \omterm{def:g3:first-electric-circuit:battery}{बैटरी} की मुख्य
सड़क कितनी ढोती है? \omterm{def:g5:series-parallel-circuits:parallel}{समांतर में} तीसरा एक जैसा \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} जुड़ता है: मुख्य
सड़क के नए बोझ की भविष्यवाणी करो, और यह भी बताओ कि तेज़ी से ख़ाली
क्या होता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:intensity-ammeter:12}
दो \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} वाले \omterm{def:g5:series-parallel-circuits:parallel}{समांतर परिपथ} (लैंप और \omterm{ex:g6:circuits-and-safety:motor}{मोटर}) के लिए मापन की पूरी योजना
बनाओ: परिपथ की \emph{हर} धारा जानने के लिए \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} की कितनी जगहें चाहिए;
कम से कम कितने मापन काफ़ी हैं (एक तो निकाला जा सकता है --- किस हिसाब
के बराबर बैठने से?); और कामों का वह क्रम भी जो किसी उपकरण को कभी
ख़तरनाक ढंग से जुड़ा नहीं छोड़ता।
\end{exercise}

\section{समस्या: मापन की मेज़}

\begin{problem}[{सप्ताहांत समस्या --- इलेक्ट्रॉनिकी कार्यशाला में
योग्यता का दिन; चार अड्डे, हर एक पर एक ऐमीटर; शागिर्द का
लाइसेंस}]\label{pb:g8:intensity-ammeter:1}
कार्यशाला के अधिकार कमाने के लिए हर शागिर्द को मापन के चार अड्डों का
चक्कर लगाना है। अध्याय साथ लेकर चलो।

\medskip\noindent\textbf{भाग I --- पहला अड्डा: सादा फंदा।}
\omterm{def:g3:first-electric-circuit:battery}{बैटरी}, \omterm{ex:g3:first-electric-circuit:switch}{स्विच}, एक लैंप।
\begin{enumerate}
  \item लैंप की धारा मापने के लिए \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} डालना क़दम दर क़दम बताओ।
  \item पाठ: \qty{280}{mA}. इसे \omterm{def:g8:intensity-ammeter:intensity}{ऐम्पियर} में लिखो।
  \item परीक्षक तुम्हारा उपकरण \omterm{def:g3:first-electric-circuit:battery}{बैटरी} की दूसरी ओर हटा देता है। नियम का
        हवाला देते हुए पाठ की भविष्यवाणी करो।
  \item ``और अगर इसी फंदे में एक साथ दो \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} होते?'' वे कहाँ-कहाँ
        बैठ सकते थे, और क्या दिखाते?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दूसरा अड्डा: श्रेणी वाली जोड़ी।}
दो अलग-अलग लैंप, जिनमें क ख से ज़्यादा चमकीला है, \omterm{def:g4:series-circuits:series}{श्रेणी में}।
\begin{enumerate}[resume]
  \item एक शागिर्द भविष्यवाणी करता है: ``ज़्यादा चमकीला क मंद ख से
        ज़्यादा धारा ढोता है।'' इस भविष्यवाणी को नियम के सामने जाँचो
        --- और बराबर धाराओं को असमान चमक से मिलाओ भी (दोनों लैंपों में
        फ़र्क़ किस चीज़ का होना चाहिए?)।
  \item उपकरण क और ख के बीच \qty{150}{mA} बताता है। फंदे की बाक़ी हर
        धारा लिख डालो।
  \item ख वाले लैंप को एक क्लिप वाले तार से \omterm{def:g7:short-circuits-safety:device}{लघुपथित} कर दिया जाता है।
        नए पाठ और दोनों लैंपों के बरताव की भविष्यवाणी करो।
  \item परीक्षक एक वाक्य माँगता है: कार्यशाला ख पर यह प्रदर्शन करने
        की इजाज़त देती है, पर \emph{\omterm{def:g3:first-electric-circuit:battery}{बैटरी}} के दोनों सिरों पर क्लिप
        वाले तार मना क्यों करती है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- तीसरा अड्डा: \omterm{def:g5:series-parallel-circuits:parallel}{संधि}।}
एक लैंप वाली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} और एक \omterm{ex:g6:circuits-and-safety:motor}{मोटर} वाली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा}, \omterm{def:g5:series-parallel-circuits:parallel}{समांतर में}।
\begin{enumerate}[resume]
  \item मुख्य सड़क: \qty{0.52}{A}; लैंप वाली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा}: \qty{0.30}{A}.
        हिसाब बराबर बैठाकर \omterm{ex:g6:circuits-and-safety:motor}{मोटर} वाली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} की धारा?
  \item \omterm{ex:g6:circuits-and-safety:motor}{मोटर} जाम हो जाती है (जुड़ी हुई, पर घूमती नहीं) और उसकी \omterm{def:g5:series-parallel-circuits:parallel}{शाखा}
        चढ़कर \qty{0.40}{A} हो जाती है। मान लो लैंप वाली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} अपनी
        \qty{0.30}{A} बनाए रखती है, तो अब मुख्य सड़क कितनी ढो रही है
        --- और कार्यशाला का कौन-सा पहरेदार कान लगाए बैठा है?
  \item लैंप खोलकर निकाल लिया जाता है। तीनों नाकों के पाठों की
        भविष्यवाणी करो।
  \item दोनों शाखाएँ भली-चंगी हों तो \omterm{def:g3:first-electric-circuit:battery}{बैटरी} सबसे तेज़ी से क्यों ख़ाली
        होती है --- उदारता का दाम कहाँ चुकाया जाता है?
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- लाइसेंस।}
\begin{enumerate}[resume]
  \item शागिर्द की शपथ लिखो: तीन पंक्तियाँ --- \omterm{def:g8:intensity-ammeter:ammeter}{ऐमीटर} की जगह कहाँ है
        और कहाँ कभी नहीं; एक फंदा किसकी ज़मानत देता है; और \omterm{def:g5:series-parallel-circuits:parallel}{संधि} किस
        चीज़ का हिसाब बराबर बैठाती है।
\end{enumerate}
\end{problem}
