\chapter{A potência e a energia elétricas}\label{ch:g9:electric-power-energy}

A conta de luz não cobra por \omterm{def:g8:voltage-voltmeter:voltage}{volts}, nem por \omterm{def:g8:intensity-ammeter:intensity}{ampères} --- ela cobra por
algo que os aparelhos da sua família vêm multiplicando discretamente
o mês inteiro. Este capítulo casa as duas grandezas elétricas numa
terceira, dá ao gasto rápido e ao gasto lento os nomes certos deles e
termina no medidor ao lado da porta de entrada: física que você pode
auditar contra dinheiro de verdade.

\section{A potência: o ritmo da entrega}

\begin{definition}[A potência e o watt]\label{def:g9:electric-power-energy:power}
A \emph{potência}\index{potência} $P$ de um aparelho é o ritmo com
que ele converte \omterm{def:g5:everyday-energy:energy}{energia} --- \omterm{def:g9:kinetic-energy-safety:joule}{joules} por segundo. A \omterm{def:g6:measurement-in-science:measuring}{unidade} dela é o
\emph{watt}\index{watt} (\unit{W}): um \omterm{def:g9:kinetic-energy-safety:joule}{joule} a cada segundo, em
homenagem ao engenheiro da era do vapor; o \emph{quilowatt}
(\qty{1}{kW} $=$ \qty{1000}{W}) serve os pesos-pesados da cozinha. A
potência não é uma quantidade de \omterm{def:g5:everyday-energy:energy}{energia}, mas um \emph{ritmo} de
gasto --- a vazão de uma torneira, não o conteúdo de um balde.
\end{definition}

\begin{proposition}[A potência elétrica]\label{prop:g9:electric-power-energy:UI}
Um aparelho alimentado com a \omterm{def:g8:voltage-voltmeter:voltage}{tensão} $U$ e que puxa a corrente $I$
recebe a \omterm{def:g9:electric-power-energy:power}{potência} elétrica
\[
P = U \times I ,
\]
\omterm{def:g9:electric-power-energy:power}{watts} a partir de \omterm{def:g8:voltage-voltmeter:voltage}{volts} vezes \omterm{def:g8:intensity-ammeter:intensity}{ampères}. A fórmula calcula nas três
direções, e a segunda receita dela comanda a caixa de fusíveis:
$I = P/U$ diz que corrente a \omterm{def:g9:electric-power-energy:power}{potência} nominal de um aparelho vai
puxar da rede.
\end{proposition}

\begin{example}[A escada doméstica]\label{ex:g9:electric-power-energy:ladder}
Leia as placas das máquinas da família: uma \omterm{def:g1:light-and-shadows:source}{luz} indicadora,
\qty{1}{W}; uma lâmpada moderna, \qty{5}{W} (a antepassada
incandescente dela gastava sessenta pela mesma \omterm{def:g1:light-and-shadows:source}{luz} --- a diferença
era \omterm{def:g5:heat-and-insulation:heat}{calor}); um notebook, \qty{50}{W}; uma televisão, \qty{100}{W};
uma máquina de lavar aquecendo a água dela, \qty{2000}{W}; a
chaleira, \qty{2300}{W}; um aquecedor de água instantâneo, até
\qty{9000}{W}. O padrão da escada: tudo o que tem por serviço
\emph{aquecer} sobe aos quilowatts --- o \omterm{def:g5:heat-and-insulation:heat}{calor} é a forma de \omterm{def:g5:everyday-energy:energy}{energia}
mais cara de fornecer.
\end{example}

\begin{example}[A aritmética da caixa de fusíveis]\label{ex:g9:electric-power-energy:fuses}
A chaleira na rede de \qty{230}{V}: $I = 2300 \div 230 = \qty{10}{A}$
--- o aparelho de dez \omterm{def:g8:intensity-ammeter:intensity}{ampères} dos capítulos antigos, agora deduzido. O
aquecedor de água de \qty{9000}{W}: quase \qty{40}{A}, exigindo um
\omterm{def:g5:series-parallel-circuits:parallel}{ramo} robusto e um disjuntor só dele. E o filtro de tomadas
sobrecarregado da investigação do incêndio: chaleira, aquecedor e
fritadeira somando as correntes $P/U$ delas para além da
especificação do \omterm{def:g5:series-parallel-circuits:parallel}{ramo} --- três inocentes, um crime aritmético,
exatamente como na denúncia.
\end{example}

\section{A energia: a quantidade entregue}

\begin{proposition}[A energia a partir da potência e do tempo]\label{prop:g9:electric-power-energy:Pt}
Um aparelho de \omterm{def:g9:electric-power-energy:power}{potência} $P$ que funciona por um tempo $t$ converte a
\omterm{def:g5:everyday-energy:energy}{energia}
\[
E = P \times t ,
\]
\omterm{def:g9:kinetic-energy-safety:joule}{joules} a partir de \omterm{def:g9:electric-power-energy:power}{watts} vezes segundos. Torneira aberta ou fio de
água demorado, o balde se enche pelo produto: os três minutos de uma
chaleira de \qty{2300}{W} ($2300 \times 180 \approx \qty{4.1e5}{J}$)
gastam mais do que um dia inteiro de uma lâmpada de \qty{5}{W}
($5 \times 86400 = \qty{4.3e5}{J}$) --- quase exatamente um empate,
na verdade: uma fervura equivale a um dia de boa \omterm{def:g1:light-and-shadows:source}{luz}.
\end{proposition}

\begin{definition}[O quilowatt-hora]\label{def:g9:electric-power-energy:kwh}
As \omterm{def:g5:everyday-energy:energy}{energias} domésticas dão contagens de \omterm{def:g9:kinetic-energy-safety:joule}{joules} desajeitadas, e por
isso o medidor fala em \emph{quilowatt-horas}\index{quilowatt-hora}
(\unit{kWh}): a \omterm{def:g5:everyday-energy:energy}{energia} de um quilowatt sustentado por uma hora,
\[
\qty{1}{kWh} = 1000 \times 3600 = \qty{3.6e6}{J}
\]
--- três milhões e seiscentos mil \omterm{def:g9:kinetic-energy-safety:joule}{joules} por \omterm{def:g6:measurement-in-science:measuring}{unidade} da conta. Com $P$
em quilowatts e $t$ em \omterm{ex:g2:measuring-time:clock}{horas}, $E = P \times t$ entrega
\omterm{ex:g2:measuring-time:clock}{quilowatt-horas} diretamente: a fórmula com que o medidor vive.
\end{definition}

\begin{method}[Auditar um aparelho]\label{met:g9:electric-power-energy:audit}
Para qualquer máquina e qualquer mês:
\begin{enumerate}
  \item leia a \omterm{def:g9:electric-power-energy:power}{potência} $P$ dele na placa (ou calcule $U \times I$);
  \item estime com honestidade o tempo de funcionamento $t$ --- por
        dia, depois por mês;
  \item $E = P \times t$ em \omterm{def:g9:electric-power-energy:kwh}{quilowatt-horas} ($P$ em \unit{kW}, $t$ em
        \omterm{ex:g2:measuring-time:clock}{horas});
  \item multiplique pela tarifa (tome cerca de $0.25$ \omterm{def:g6:measurement-in-science:measuring}{unidade}
        monetária por \unit{kWh}) para o custo do mês.
\end{enumerate}
A chaleira, seis minutos por dia:
$2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$ por mês. A televisão,
três \omterm{ex:g2:measuring-time:clock}{horas} por dia: $0.1 \times 3 \times 30 = \qty{9}{kWh}$.
\omterm{def:g9:electric-power-energy:power}{Potências} pequenas com muitas \omterm{ex:g2:measuring-time:clock}{horas} rivalizam com \omterm{def:g9:electric-power-energy:power}{potências} grandes
de poucas \omterm{ex:g2:measuring-time:clock}{horas} --- a lição recorrente da auditoria.
\end{method}

\section{O medidor e a conta}

\begin{example}[Ler a conta]\label{ex:g9:electric-power-energy:bill}
O medidor ao lado da porta de entrada conta cada \omterm{def:g9:electric-power-energy:kwh}{quilowatt-hora} que
entra na casa; a conta é a diferença do mês dele vezes a tarifa. As
$250$ \omterm{def:g6:measurement-in-science:measuring}{unidades} mensais de uma família típica se decompõem pela
auditoria: primeiro o aquecimento de água e os aquecedores de
ambiente (o clube do quilowatt), depois os aparelhos frios ---
poucos \omterm{def:g9:electric-power-energy:power}{watts}, mas funcionando sempre ---, depois o cozimento, a
lavagem, a \omterm{def:g1:light-and-shadows:source}{luz} e a eletrônica. A \omterm{def:g2:pushes-and-pulls:force}{força} da auditoria: ela acha as
\omterm{def:g3:levers-and-scales:lever}{alavancas} de verdade. Trocar dez lâmpadas velhas de \qty{60}{W} por
outras de \qty{5}{W} economiza mais do que desligar todo carregador
da casa mil vezes seguidas.
\end{example}

\begin{example}[Os vampiros do modo de espera]\label{ex:g9:electric-power-energy:standby}
Letra miúda, muitas \omterm{ex:g2:measuring-time:clock}{horas}: a luzinha de espera de uma televisão bebe
\qty{0.5}{W} --- mas durante as $720$ \omterm{ex:g2:measuring-time:clock}{horas} do mês:
$0.0005 \times 720 = \qty{0.36}{kWh}$. Um vampiro é inofensivo; uma
casa com vinte bebedores de um \omterm{def:g9:electric-power-energy:power}{watt} cada paga
$0.02 \times 720 \approx \qty{14}{kWh}$ por mês por nada nenhum. A
\omterm{def:g9:electric-power-energy:power}{potência} vezes o tempo não perdoa fator nenhum: o segundo morde
quando o primeiro se esconde.
\end{example}

\begin{remark}[A potência no mundo mais amplo]\label{rem:g9:electric-power-energy:wider}
O \omterm{def:g9:electric-power-energy:power}{watt}, nascido elétrico aqui, mede todo ritmo de \omterm{def:g5:everyday-energy:energy}{energia}. O seu
corpo em repouso funciona perto de \qty{100}{W} --- uma lâmpada
antepassada incandescente de \omterm{def:g5:heat-and-insulation:heat}{calor} (lembre do abafamento da sala de
aula cheia: trinta estudantes, três quilowatts de biologia). Um
ciclista sustenta \qty{200}{W}; o \omterm{ex:g6:circuits-and-safety:motor}{motor} de um carro solta dezenas de
quilowatts; a usina da cidade, centenas de milhões de \omterm{def:g9:electric-power-energy:power}{watts}; a \omterm{def:g1:light-and-shadows:source}{luz} do
Sol que cai sobre a cidade, confortavelmente mais --- o teto sob o
qual todos os outros ritmos trabalham. O próximo capítulo segue os
\omterm{def:g9:electric-power-energy:power}{watts} da rede até a fonte girante deles.
\end{remark}

\section{Problema: A auditoria energética da casa}

\begin{problem}[{Problema de fim de semana --- a auditoria energética
da família; placas, medidores e a conta do mês; achar as alavancas de
verdade}]\label{pb:g9:electric-power-energy:1}
Armada com as duas fórmulas e a tarifa ($0.25$ por \unit{kWh}), a
família audita um mês (30 dias). O inventário: chaleira
\qty{2300}{W}, $6$ minutos por dia; televisão \qty{100}{W}, $4$ \omterm{ex:g2:measuring-time:clock}{horas}
por dia; geladeira --- caso especial abaixo; dez lâmpadas de
\qty{5}{W} cada, $5$ \omterm{ex:g2:measuring-time:clock}{horas} por dia; máquina de lavar \qty{2}{kW}, um
ciclo quente de $1$ hora em dias alternados; aquecedor de água
\qty{2.5}{kW}, $2$ \omterm{ex:g2:measuring-time:clock}{horas} por dia; bebedores de espera somando
\qty{10}{W}, sempre ligados.

\medskip\noindent\textbf{Parte I --- Máquina por máquina.}
\begin{enumerate}
  \item O mês da chaleira, em \unit{kWh}.
  \item Os meses da televisão e das dez lâmpadas.
  \item Os meses da máquina de lavar ($15$ ciclos) e do aquecedor de
        água.
  \item O mês dos bebedores de espera --- o segundo fator em ação.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A geladeira e a soma.}
O compressor da geladeira funciona a \qty{120}{W}, mas só um terço do
tempo, ligando e desligando dia e noite.
\begin{enumerate}[resume]
  \item A \omterm{def:g9:electric-power-energy:power}{potência} média efetiva dele, e o mês dele em \unit{kWh}.
  \item Some as sete linhas da auditoria. Que três linhas dominam?
  \item A conta do mês pela tarifa --- e a parcela do aquecedor de
        água nela, em dinheiro.
  \item O medidor marcava $2150$ \omterm{def:g6:measurement-in-science:measuring}{unidades} no início do mês. Preveja a
        leitura dele no fim do mês.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- As \omterm{def:g3:levers-and-scales:lever}{alavancas}.}
\begin{enumerate}[resume]
  \item Proposta um: banhos mais curtos --- o aquecedor cai para
        \qty{1.5}{h} por dia. Economia mensal, em \omterm{def:g5:everyday-energy:energy}{energia} e em
        dinheiro?
  \item Proposta dois: uma campanha pública insiste em desplugar os
        carregadores de celular (\qty{0.2}{W} cada um quando
        ociosos). Audite o mês ocioso de um carregador e compare com
        a proposta um; o que o contraste ensina sobre auditorias
        contra slogans?
  \item Proposta três: a avó sugere o velho hábito de encher a
        chaleira só com o necessário --- metade da água, metade da
        \omterm{def:g5:everyday-energy:energy}{energia}. Nova linha da chaleira, e a economia do mês?
  \item Escreva as conclusões da auditoria: três frases --- onde os
        \omterm{def:g9:electric-power-energy:kwh}{quilowatt-horas} de fato moram, que fator da fórmula o caso do
        modo de espera ensina e qual mudança doméstica isolada rende
        mais.
\end{enumerate}
\end{problem}
