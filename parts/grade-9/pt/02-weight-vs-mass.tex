\chapter{Peso e massa: medir forças}\label{ch:g9:weight-vs-mass}

Cinco anos atrás, numa \omterm{def:g2:moon-in-the-sky:moon}{Lua} de papel, você partiu ``pesado'' em dois:
a \omterm{def:g3:measuring-mass:mass}{massa} para a matéria, o \omterm{def:g4:weight-and-mass:weight}{peso} para o puxão. A separação era
sabedoria; agora ela vira aritmética. As \omterm{def:g2:pushes-and-pulls:force}{forças} ganham a \omterm{def:g6:measurement-in-science:measuring}{unidade}
delas --- batizada com o nome do homem da maçã que cai ---, o \omterm{def:g4:weight-and-mass:weight}{peso}
ganha a fórmula dele, e a balança de elástico da sua infância volta em
traje profissional.

\section{O newton}

\begin{definition}[O newton]\label{def:g9:weight-vs-mass:newton}
As \omterm{def:g2:pushes-and-pulls:force}{forças} --- empurrões, puxões, \omterm{def:g4:weight-and-mass:weight}{pesos}, a garra da \omterm{def:g9:gravitation:gravitation}{gravitação} --- são
medidas em \emph{newtons}\index{newton} (\unit{N}), em homenagem ao
autor da lei da \omterm{def:g9:gravitation:gravitation}{gravitação}. Âncoras práticas: um newton é mais ou
menos o \omterm{def:g4:weight-and-mass:weight}{peso} de uma maçã pequena (a história sorri); a sua mochila
pesa uns \qty{50}{N}; um aperto de mão firme espreme com uns
\qty{100}{N}; o contrato de aluguel da \omterm{def:g2:moon-in-the-sky:moon}{Lua} do capítulo passado
chegava a $10^{20}$.
\end{definition}

\begin{definition}[Dinamômetro]\label{def:g9:weight-vs-mass:dynamometer}
Um \emph{dinamômetro}\index{dinamômetro} é o medidor de \omterm{def:g2:pushes-and-pulls:force}{força}: uma
mola calibrada dentro de uma caixa, que se estica em proporção à
\omterm{def:g2:pushes-and-pulls:force}{força} aplicada, com o ponteiro lendo \omterm{def:g9:weight-vs-mass:newton}{newtons} diretamente. É a sua
balança de elástico já crescida --- e, como o \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} e o
\omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} antes dele, ele aposenta mais um pedaço do ``parece
forte'' e o troca por números honestos.
\end{definition}

\begin{method}[Medir uma força]\label{met:g9:weight-vs-mass:measure}
\begin{enumerate}
  \item escolha um \omterm{def:g9:weight-vs-mass:dynamometer}{dinamômetro} cuja faixa sirva ao serviço --- um
        instrumento de \qty{2}{N} para estojos, um de \qty{100}{N}
        para mochilas;
  \item zere-o na posição de uso (pendurado, se a carga vai ficar
        pendurada --- o próprio \omterm{def:g4:weight-and-mass:weight}{peso} da mola desloca o zero);
  \item aplique a \omterm{def:g2:pushes-and-pulls:force}{força} de modo firme, ao longo do \omterm{def:g6:sun-earth-moon:axisorbit}{eixo} do
        instrumento, e leia na altura dos olhos;
  \item registre com a \omterm{def:g6:measurement-in-science:measuring}{unidade} --- e, como as \omterm{def:g2:pushes-and-pulls:force}{forças} têm direções,
        anote também a direção quando ela importar.
\end{enumerate}
Nos esquemas, uma \omterm{def:g2:pushes-and-pulls:force}{força} medida vira uma seta desenhada numa escala
declarada --- ``um \omterm{def:g2:measuring-length:units}{centímetro} por dez \omterm{def:g9:weight-vs-mass:newton}{newtons}'' --- apontando para
onde a \omterm{def:g2:pushes-and-pulls:force}{força} aponta: as setas de \omterm{def:g2:pushes-and-pulls:force}{força} da infância, agora
quantitativas.
\end{method}

\section{A fórmula do peso}

\begin{proposition}[O peso é proporcional à massa]\label{prop:g9:weight-vs-mass:formula}
Num dado lugar, o \omterm{def:g4:weight-and-mass:weight}{peso} $P$ de um objeto (em \unit{N}) é proporcional à
\omterm{def:g3:measuring-mass:mass}{massa} $m$ dele (em \unit{kg}):
\[
P = m \times g ,
\]
em que $g$, a \emph{intensidade gravitacional}\index{intensidade
gravitacional} do lugar, dá os \omterm{def:g9:weight-vs-mass:newton}{newtons} de puxão por \omterm{def:g3:measuring-mass:units}{quilograma} de
matéria. Perto da superfície da Terra,
\[
g \approx \qty{9.8}{N/kg}:
\]
cada \omterm{def:g3:measuring-mass:units}{quilograma} é puxado com pouco menos de dez \omterm{def:g9:weight-vs-mass:newton}{newtons}. A fórmula é
a lei do capítulo passado em traje local: $g$ empacota a \omterm{def:g3:measuring-mass:mass}{massa} e o
raio da Terra num número só para ``aqui''.
\end{proposition}

\begin{example}[A fórmula em ação]\label{ex:g9:weight-vs-mass:atwork}
Um estudante de \qty{60}{kg}: $P = 60 \times 9.8 = \qty{588}{N}$. Uma
bola de futebol de \qty{450}{g}:
$P = 0.45 \times 9.8 \approx \qty{4.4}{N}$ --- atenção aos
\omterm{def:g3:measuring-mass:units}{quilogramas}. De trás para a frente: um engradado que pesa \qty{343}{N}
tem \omterm{def:g3:measuring-mass:mass}{massa} $m = P/g = 343 \div 9.8 = \qty{35}{kg}$. E de cabeça para
baixo: numa montanha em que uma mochila de \qty{10}{kg} pesa
\qty{97.7}{N}, o $g$ local vale
$97.7 \div 10 = \qty{9.77}{N/kg}$ --- $g$ afina um pouquinho com a
altitude, exatamente como o inverso do quadrado prometia.
\end{example}

\begin{example}[Uma mala, muitos pesos]\label{ex:g9:weight-vs-mass:planets}
Cada mundo fixa o $g$ dele, e a velha história dos astronautas vira
uma tabela. A sua fiel mala de \qty{10}{kg}:
\begin{center}
\begin{tabular}{l|c|c}
lugar & $g$ (\unit{N/kg}) & \omterm{def:g4:weight-and-mass:weight}{peso} da mala \\
\hline
Terra & $9.8$ & \qty{98}{N} \\
\omterm{def:g2:moon-in-the-sky:moon}{Lua} & $1.6$ & \qty{16}{N} \\
Marte & $3.7$ & \qty{37}{N} \\
Topo das nuvens de Júpiter & $24.8$ & \qty{248}{N}
\end{tabular}
\end{center}
Dez \omterm{def:g3:measuring-mass:units}{quilogramas} de matéria em todo lugar --- e um \omterm{def:g4:weight-and-mass:weight}{peso} diferente em
cada mundo: a \omterm{def:g3:measuring-mass:mass}{massa} fica inalterada, enquanto o \omterm{def:g4:weight-and-mass:weight}{peso} segue o $g$
local.
\end{example}

\begin{remark}[As duas balanças, julgadas de vez]\label{rem:g9:weight-vs-mass:scales}
Agora os vereditos da infância se fecham com fórmulas. A \omterm{def:g3:levers-and-scales:scale}{balança de
pratos} compara $m_1 g$ com $m_2 g$: o $g$ local multiplica os dois
pratos e se cancela --- as \omterm{def:g3:levers-and-scales:scale}{balanças de pratos} medem \emph{\omterm{def:g3:measuring-mass:mass}{massa}}, e
são verdadeiras em qualquer mundo. A balança de mola lê $P = m g$
diretamente: ela mede \emph{\omterm{def:g4:weight-and-mass:weight}{peso}}, leal ao lugar dela. A balança de
banheiro é uma balança de mola disfarçada, marcada de fábrica em
\omterm{def:g3:measuring-mass:units}{quilogramas} pela divisão do seu \omterm{def:g4:weight-and-mass:weight}{peso} pelo $g$ \emph{da Terra} ---
honesta em casa, bajuladora na \omterm{def:g2:moon-in-the-sky:moon}{Lua} (\qty{10}{kg} marcados para cada
\qty{60}{kg} seus) e caluniadora em Júpiter.
\end{remark}

\section{As forças no papel}

\begin{example}[O livro sobre a mesa, em newtons]\label{ex:g9:weight-vs-mass:book}
O \omterm{def:g2:balance-equilibrium:equilibrium}{equilíbrio} mais antigo deste livro, enfim com números: um
dicionário de \qty{2}{kg} descansa sobre uma escrivaninha. O \omterm{def:g4:weight-and-mass:weight}{peso}
dele: $P = 2 \times 9.8 = \qty{19.6}{N}$, desenhado como uma seta de
\qty{2}{cm} numa escala de dez \omterm{def:g9:weight-vs-mass:newton}{newtons} por \omterm{def:g2:measuring-length:units}{centímetro}, apontando
direto para baixo a partir do livro. O empurrão de sustentação da
escrivaninha: \qty{19.6}{N} direto para cima --- uma seta igual, no
sentido oposto. O empate é exato, e a imobilidade é a prova dele: a
figura da infância, agora um orçamento medido.
\end{example}

\begin{remark}[O que as forças fazem --- a lei que falta]\label{rem:g9:weight-vs-mass:dynamics}
Este capítulo mede \omterm{def:g2:pushes-and-pulls:force}{forças}; ele ainda não diz o que um \emph{excedente}
de \omterm{def:g2:pushes-and-pulls:force}{força} faz com o movimento. Os quatro serviços da infância ---
começar, parar, virar, deformar --- esperam a lei exata deles, e ela é
uma das maiores de toda a física: o volume do ensino médio abre a
mecânica dele com ela. Este ano termina o alicerce: \omterm{def:g2:pushes-and-pulls:force}{forças} em
\omterm{def:g9:weight-vs-mass:newton}{newtons}, \omterm{def:g4:weight-and-mass:weight}{pesos} por fórmula, setas em escala --- a gramática da frase
que vem.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:1}
Diga o nome da \omterm{def:g6:measurement-in-science:measuring}{unidade} de \omterm{def:g2:pushes-and-pulls:force}{força} e do instrumento dela. Dê duas
âncoras do dia a dia para um \omterm{def:g9:weight-vs-mass:newton}{newton} e para cem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:2}
Escreva a fórmula do \omterm{def:g4:weight-and-mass:weight}{peso} com as \omterm{def:g6:measurement-in-science:measuring}{unidades}, e o significado de $g$ em
palavras.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:3}
Calcule \omterm{def:g4:weight-and-mass:weight}{pesos} terrestres: um cachorro de \qty{25}{kg}; um pão de
\qty{700}{g}; você mesmo (escolha a sua \omterm{def:g3:measuring-mass:mass}{massa}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:4}
Um melão pendurado estica o \omterm{def:g9:weight-vs-mass:dynamometer}{dinamômetro} até \qty{14.7}{N}. A \omterm{def:g3:measuring-mass:mass}{massa}
dele?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:5}
Leia o gráfico do capítulo: o que a inclinação dele quer dizer, e o
que a reta faria na \omterm{def:g2:moon-in-the-sky:moon}{Lua}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:6}
A mala de \qty{10}{kg} faz o tour Terra, \omterm{def:g2:moon-in-the-sky:moon}{Lua}, Marte, Júpiter. Que
números da tabela nunca mudam, e quais mudam --- e por quê?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:7}
Por que uma \omterm{def:g3:levers-and-scales:scale}{balança de pratos} diz a verdade em todos os mundos
enquanto a balança de banheiro bajula na \omterm{def:g2:moon-in-the-sky:moon}{Lua}? Fórmulas, não
sentimentos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:8}
Desenhe (ou descreva) as duas setas de \omterm{def:g2:pushes-and-pulls:force}{força} do dicionário de
\qty{2}{kg} a um \omterm{def:g2:measuring-length:units}{centímetro} por cinco \omterm{def:g9:weight-vs-mass:newton}{newtons}: comprimentos,
sentidos e o veredito do orçamento.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:9}
Uma companhia aérea permite \qty{23}{kg} por mala. Um passageiro
argumenta: ``Na \omterm{def:g2:moon-in-the-sky:moon}{Lua} isso pesaria só uns poucos \omterm{def:g9:weight-vs-mass:newton}{newtons}!'' Por que o
limite da companhia --- combustível e \omterm{def:g8:ohms-law:resistance}{resistência} do piso à parte ---
diz respeito com razão à \omterm{def:g3:measuring-mass:mass}{massa}, e que instrumento deveria conferi-lo
num aeroporto lunar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:10}
Em Marte, o braço de um jipe robô pesa \qty{111}{N}. A \omterm{def:g3:measuring-mass:mass}{massa} dele? E
quanto o mesmo braço pesaria de volta no galpão de montagem, na
Terra?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:11}
Um \omterm{def:g9:weight-vs-mass:dynamometer}{dinamômetro} zerado deitado na horizontal e depois usado pendurado
mente um pouquinho --- para que lado, e por quê? (Que \omterm{def:g4:weight-and-mass:weight}{peso} a mais se
junta ao da carga?)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:weight-vs-mass:12}
Projete a experiência que desenha o gráfico do capítulo e extrai $g$:
equipamento, tabela de medidas, o gráfico e como a inclinação é lida.
Depois explique o que o mesmo protocolo, rodado num laboratório da
\omterm{def:g2:moon-in-the-sky:moon}{Lua}, entregaria --- que números mudam, que método sobrevive intocado.
\end{exercise}

\section{Problema: Os equipadores interplanetários}

\begin{problem}[{Problema de fim de semana --- o provador dos
equipadores interplanetários; um corpo, quatro mundos; balanças,
dinamômetros e o balcão de bagagens}]\label{pb:g9:weight-vs-mass:1}
Os equipadores preparam viajantes para os portos do \omterm{def:g4:solar-system:family}{Sistema Solar}
($g$, em \unit{N/kg}: Terra $9.8$; \omterm{def:g2:moon-in-the-sky:moon}{Lua} $1.6$; Marte $3.7$; convés de
nuvens de Júpiter $24.8$). O provador fica com você.

\medskip\noindent\textbf{Parte I --- A prova de roupa.}
Uma viajante de \qty{70}{kg} de \omterm{def:g3:measuring-mass:mass}{massa} se apresenta.
\begin{enumerate}
  \item O \omterm{def:g4:weight-and-mass:weight}{peso} dela em cada um dos quatro portos?
  \item Em qual porto a clássica balança de banheiro (marcada em
        \omterm{def:g3:measuring-mass:units}{quilogramas}, calibrada para a Terra) lê corretamente a \omterm{def:g3:measuring-mass:mass}{massa}
        dela --- e o que ela lê nos outros três? (Divida cada \omterm{def:g4:weight-and-mass:weight}{peso}
        pelo $g$ da Terra.)
  \item A balança honesta dos equipadores é uma \omterm{def:g3:levers-and-scales:scale}{balança de pratos} com
        \omterm{def:g3:measuring-mass:mass}{massas} marcadas. O que ela informa em cada porto, e por quê?
  \item As botas da viajante nunca podem fazer pressão no chão com
        mais de \qty{2000}{N} (os pisos das \omterm{def:g3:sun-days-seasons:year}{estações} são finos). Há
        algum porto em que botas mais corpo (\qty{70}{kg} no total)
        quebrem a regra?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O balcão de bagagens.}
\begin{enumerate}[resume]
  \item O baú com destino a Marte tem \qty{40}{kg} de \omterm{def:g3:measuring-mass:mass}{massa}. Calcule
        o \omterm{def:g4:weight-and-mass:weight}{peso} dele na partida (Terra) e na chegada (Marte).
  \item O sindicato dos carregadores limita o \emph{\omterm{def:g4:weight-and-mass:weight}{peso}} carregado à
        mão em \qty{200}{N} em todos os portos. Que \emph{\omterm{def:g3:measuring-mass:mass}{massa}}
        máxima um carregador pode levar em cada um?
  \item Um engradado etiquetado ``\qty{150}{N} na \omterm{def:g2:moon-in-the-sky:moon}{Lua}'' precisa
        passar pela alfândega da Terra pela \omterm{def:g3:measuring-mass:mass}{massa}. Faça a conversão.
  \item O \omterm{def:g9:weight-vs-mass:dynamometer}{dinamômetro} do balcão lê uma bolsa pendurada em
        \qty{294}{N} na Terra. A \omterm{def:g3:measuring-mass:mass}{massa} e o \omterm{def:g4:weight-and-mass:weight}{peso} lunar dela para o
        manifesto?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Reclamações e curiosidades.}
\begin{enumerate}[resume]
  \item Um fisiculturista volta da \omterm{def:g2:moon-in-the-sky:moon}{Lua} furioso: ``Ganhos de \omterm{def:g2:pushes-and-pulls:force}{força} de
        seis vezes, perdidos de uma noite para a outra!'' Console-o
        com a fórmula: o que de fato mudou em cada porto, e o que
        nunca mudou?
  \item Um chef do convés de Júpiter reclama que bater claras parece
        levantar bigornas, embora o batedor ``tenha meros \qty{200}{g}
        de \omterm{def:g3:measuring-mass:mass}{massa}''. Calcule o \omterm{def:g4:weight-and-mass:weight}{peso} do batedor no convés e julgue.
  \item A loja de presentes vende ``um \omterm{def:g9:weight-vs-mass:newton}{newton} de chocolate, em
        qualquer lugar''. Em qual porto o comprador leva mais
        \emph{matéria} de chocolate, e quanta \omterm{def:g3:measuring-mass:mass}{massa} em cada um de
        Terra, \omterm{def:g2:moon-in-the-sky:moon}{Lua} e Júpiter?
  \item Escreva o lema dos equipadores: uma frase separando o que
        viaja com você do que cada mundo cobra.
\end{enumerate}
\end{problem}
