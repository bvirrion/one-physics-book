\chapter{As formas de energia e as conversões}\label{ch:g9:energy-conversions}

A \omterm{def:g5:everyday-energy:energy}{energia} atravessou este livro como um rio por baixo de uma cidade
--- entrevista em cada esquina, nunca mapeada por inteiro. Chegou a
hora. Cada forma ganha o nome dela, uma delas ganha a segunda grande
fórmula deste ano, e acima de todas preside a mais poderosa lei de
contabilidade da ciência: o total nunca muda. Nunca.

\section{As formas, reunidas}

\begin{definition}[As formas de energia]\label{def:g9:energy-conversions:forms}
As formas de \omterm{def:g5:everyday-energy:energy}{energia}, apresentadas formalmente: a \emph{\omterm{prop:g9:kinetic-energy-safety:formula}{energia
cinética}} --- a parcela do movimento, $\frac12 m v^2$; a \emph{energia
potencial}\index{energia potencial} --- guardada pela posição ou pelo
arranjo (o que foi levantado, esticado, \omterm{prop:g7:states-of-matter:squeeze}{comprimido}); a \emph{\omterm{def:g5:everyday-energy:energy}{energia}
térmica} --- o sacolejo molecular desordenado que a \omterm{def:g3:temperature-thermometers:temperature}{temperatura} mede;
a \emph{\omterm{def:g5:everyday-energy:energy}{energia} química} --- guardada nos arranjos da matéria:
alimentos, combustíveis, \omterm{def:g3:first-electric-circuit:battery}{pilhas}; a \emph{\omterm{def:g5:everyday-energy:energy}{energia} elétrica} --- o que
a corrente em marcha consegue entregar; a \emph{\omterm{def:g5:everyday-energy:energy}{energia} radiante} ---
carregada pela própria \omterm{def:g1:light-and-shadows:source}{luz}, a forma de transporte da \omterm{def:g1:light-and-shadows:source}{luz} do Sol; e a
\emph{\omterm{def:g5:everyday-energy:energy}{energia} nuclear} --- trancada no coração do átomo, a reserva
mais profunda que os seres humanos já abriram.
\end{definition}

\begin{proposition}[A energia potencial gravitacional]\label{prop:g9:energy-conversions:ep}
Levantar uma \omterm{def:g3:measuring-mass:mass}{massa} $m$ (em \unit{kg}) por uma altura $h$ (em
\unit{m}) contra o $g$ da gravidade guarda nela a \omterm{def:g9:energy-conversions:forms}{energia potencial}
\[
E_p = m \times g \times h
\]
em \omterm{def:g9:kinetic-energy-safety:joule}{joules} --- recuperável por inteiro na volta para baixo. O machado
levantado, o balanço puxado para trás, o lago de montanha: todos
mantêm $m g h$ em conta corrente. (É a fórmula emprestada dos
cartazes de segurança no trânsito --- agora oficialmente sua.)
\end{proposition}

\begin{example}[A grande troca: $E_p$ contra $E_k$]\label{ex:g9:energy-conversions:trade}
Solte uma bola de uma altura $h$: enquanto ela cai, a conta dela
transfere, \omterm{def:g9:kinetic-energy-safety:joule}{joule} por \omterm{def:g9:kinetic-energy-safety:joule}{joule}, da potencial para a cinética. No chão,
todo o $m g h$ virou $\frac12 m v^2$ --- iguale os dois e a \omterm{def:g3:measuring-mass:mass}{massa} se
cancela:
\[
v = \sqrt{2 g h},
\]
com a raiz quadrada deste ano ganhando o salário dela. De
\qty{20}{m}: $v = \sqrt{2 \times 9.8 \times 20} \approx \qty{20}{m/s}$
--- setenta quilômetros por hora, seja qual for a \omterm{def:g3:measuring-mass:mass}{massa}. O pêndulo
joga a troca nos dois sentidos para sempre (menos um sussurro para o
\omterm{def:g2:air-around-us:air}{ar}); o skatista na meia-cana também: altura vira \omterm{def:g7:motion-average-speed:formula}{velocidade} que vira
altura, com as duas fórmulas passando uma soma só de uma para a
outra.
\end{example}

\section{A grande lei}

\begin{proposition}[A conservação da energia]\label{prop:g9:energy-conversions:conservation}
Em todo processo já examinado --- colisões e química, máquinas e
tempestades, \omterm{def:g4:solar-system:star}{estrelas} e células --- a \omterm{def:g5:everyday-energy:energy}{energia} muda de forma e de
dono, mas o \emph{total} é exatamente conservado: nenhuma criada,
nenhuma destruída, cada \omterm{def:g9:kinetic-energy-safety:joule}{joule} com endereço. Esta é a lei de
contabilidade mais profunda da física: ela aposentou as máquinas
eternas dos capítulos da infância, ela audita cada corrente deste
livro, e experiência nenhuma em três séculos a pegou em erro. (Por
que a natureza cumpre essa lei com tanta fidelidade é uma pergunta
profunda com uma resposta bonita --- uma das joias dos anos de
universidade.)
\end{proposition}

\begin{method}[Auditar qualquer processo]\label{met:g9:energy-conversions:audit}
A contabilidade universal do físico:
\begin{enumerate}
  \item cerque o sistema --- o que está dentro da auditoria;
  \item liste as contas de \omterm{def:g5:everyday-energy:energy}{energia} de antes: cada forma, cada
        reserva;
  \item liste as de depois --- incluindo, sempre, o trocado térmico:
        o \omterm{def:g5:heat-and-insulation:heat}{calor} do atrito, o sussurro do \omterm{def:g3:sound-around-us:sound}{som};
  \item feche as contas: os totais precisam bater. Um rombo quer
        dizer que alguma forma ficou sem ser contada --- procure de
        novo o vazamento; ele costuma estar morno.
\end{enumerate}
\end{method}

\begin{example}[Auditorias, resolvidas]\label{ex:g9:energy-conversions:audits}
O carro que freia: conta cinética esvaziada, conta térmica dos discos
creditada por inteiro --- aposentadoria, não destruição. A bola que
quica e vai morrendo: cada quique raspa um pouco da cinética para o
\omterm{def:g5:heat-and-insulation:heat}{calor} da bola e do chão; a resposta da infância, agora auditável. O
seu café da manhã pedalando ladeira acima: reserva química para
baixo, conta potencial para cima ($m g h$ seu e da bicicleta),
parcela térmica irradiada --- os músculos funcionam quentes. Todo
mistério dos primeiros capítulos de \omterm{def:g5:everyday-energy:energy}{energia} se encerra sob as mesmas
três colunas.
\end{example}

\section{O rendimento}

\begin{definition}[Rendimento]\label{def:g9:energy-conversions:efficiency}
O \emph{rendimento}\index{rendimento} de um conversor é a fração da
\omterm{def:g5:everyday-energy:energy}{energia} de entrada dele que é entregue na forma desejada:
\[
\text{rendimento} =
\frac{\text{energia útil que sai}}{\text{energia total que entra}},
\]
um número entre $0$ e $1$ (ou a porcentagem correspondente). O
restante não é destruído --- a conservação proíbe ---, mas escapa em
formas indesejadas, quase sempre \omterm{def:g5:heat-and-insulation:heat}{calor}.
\end{definition}

\begin{example}[Boletins]\label{ex:g9:energy-conversions:reportcards}
A lâmpada antepassada incandescente: cinco por cento de \omterm{def:g1:light-and-shadows:source}{luz}, noventa
e cinco de \omterm{def:g5:heat-and-insulation:heat}{calor} --- um aquecedor com um hobby. A sucessora moderna
dela: trinta e tantos por cento, e o resto ainda \omterm{def:g5:heat-and-insulation:heat}{calor}. Um \omterm{ex:g6:circuits-and-safety:motor}{motor} de
carro: cerca de um terço em movimento útil, dois terços saindo pelo
radiador e pelo escapamento. Um \omterm{ex:g6:circuits-and-safety:motor}{motor} elétrico: acima de noventa. Os
grandes \omterm{def:g9:alternator:alternator}{alternadores}: perto de noventa e nove --- os melhores
conversores da civilização. Conversor honesto nenhum chega a cem:
algum vazamento para o \omterm{def:g5:heat-and-insulation:heat}{calor} é a comissão universal da natureza.
\end{example}

\begin{remark}[O que ``economizar energia'' quer dizer de verdade]\label{rem:g9:energy-conversions:saving}
Se a conservação guarda cada \omterm{def:g9:kinetic-energy-safety:joule}{joule}, por que é preciso ``economizar
\omterm{def:g5:everyday-energy:energy}{energia}''? Porque a lei preserva a \emph{quantidade}, não a
\emph{utilidade}. As reservas concentradas --- combustível, carga,
altura --- podem ser gastas em \omterm{def:g5:heat-and-insulation:heat}{calor} espalhado finamente pelo mundo,
e o \omterm{def:g5:heat-and-insulation:heat}{calor} espalhado, embora cada \omterm{def:g9:kinetic-energy-safety:joule}{joule} dele sobreviva, já não
consegue mover chaleiras ou trens: o rio desceu a ladeira. O problema
energético da civilização não é a falta de \omterm{def:g9:kinetic-energy-safety:joule}{joules} --- o Sol nos
banha com milhares de vezes as nossas necessidades ---, e sim a boa
administração dos \omterm{def:g9:kinetic-energy-safety:joule}{joules} \emph{úteis}. Essa ideia, tornada precisa, é
uma das histórias mais célebres da física; ela espera nos anos de
universidade sob o nome famoso dela.
\end{remark}

\section{Problema: A comissão do parque de diversões}

\begin{problem}[{Problema de fim de semana --- o parque de diversões
contrata uma consultoria de física; montanhas-russas, torres e o
livro-caixa que não sabe mentir}]\label{pb:g9:energy-conversions:1}
As novas atrações do parque precisam passar pela sua auditoria de
\omterm{def:g5:everyday-energy:energy}{energia} antes do dia da inauguração. Tome $g = \qty{9.8}{N/kg}$;
ignore o atrito até ele se recusar a ser ignorado.

\medskip\noindent\textbf{Parte I --- A Grande Queda.}
Um trem de montanha-russa de \qty{2000}{kg} é guinchado até um cume
de \qty{45}{m} e solto do repouso.
\begin{enumerate}
  \item O $E_p$ do trem no cume?
  \item A \omterm{def:g7:motion-average-speed:formula}{velocidade} ideal dele no fim da queda, pela fórmula em que
        a \omterm{def:g3:measuring-mass:mass}{massa} se cancela --- em \unit{m/s} e em \unit{km/h}?
  \item Os projetistas acrescentam uma segunda colina de \qty{50}{m}
        logo depois da queda. Vete a ideia com o livro-caixa --- e
        diga qual é a segunda colina mais alta que o trem ideal
        conseguiria vencer.
  \item A \omterm{def:g7:motion-average-speed:formula}{velocidade} medida lá embaixo é \qty{28}{m/s}, e não a
        ideal. Calcule as duas \omterm{prop:g9:kinetic-energy-safety:formula}{energias cinéticas} e lance a diferença
        na conta certa.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O guincho e a conta.}
\begin{enumerate}[resume]
  \item O guincho leva o trem até o cume em \qty{60}{s}. Com
        \omterm{def:g9:energy-conversions:efficiency}{rendimento} perfeito, a \omterm{def:g9:electric-power-energy:power}{potência} dele?
  \item O guincho de verdade puxa \qty{20}{kW}. O \omterm{def:g9:energy-conversions:efficiency}{rendimento} dele?
  \item Cada volta sai a cada quatro minutos, dez \omterm{ex:g2:measuring-time:clock}{horas} por dia. A
        \omterm{def:g5:everyday-energy:energy}{energia} diária do guincho em \omterm{def:g9:electric-power-energy:kwh}{quilowatt-horas} (guincho real,
        conte as subidas) e o custo dela a $0.25$ por \omterm{def:g6:measurement-in-science:measuring}{unidade}?
  \item Um projetista propõe recuperar a \omterm{def:g5:everyday-energy:energy}{energia} de chegada do trem
        com um freio-gerador que devolve à rede --- ``o brinquedo se
        paga sozinho!''. Audite o sonho: que fração poderia
        realisticamente voltar, e que lei proíbe o círculo completo?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A torre de queda livre e o
relatório.}
A torre de \qty{60}{m} solta uma gôndola de \qty{500}{kg} sobre
freios magnéticos a \qty{10}{m} do chão.
\begin{enumerate}[resume]
  \item A \omterm{def:g7:motion-average-speed:formula}{velocidade} ao entrar na zona de frenagem (uma queda ideal
        de \qty{50}{m})?
  \item A \omterm{def:g5:everyday-energy:energy}{energia} que os freios precisam aposentar, e para onde a
        indução a manda. (Os freios são primos do capítulo do
        \omterm{def:g9:alternator:alternator}{alternador} --- redemoinhos de corrente induzida aquecendo
        metal.)
  \item Os passageiros gritam que ficaram ``sem \omterm{def:g4:weight-and-mass:weight}{peso}'' durante a
        queda. A partir da ideia dos companheiros de queda do
        capítulo da \omterm{def:g9:gravitation:gravitation}{gravitação}, julgue a alegação em uma frase.
  \item Assine o relatório da comissão: três frases --- a troca de
        que a montanha-russa vive, a lei que toda atração obedeceu e
        a conta que discretamente cobrou imposto de todas elas.
\end{enumerate}
\end{problem}
