\section*{Chapitre \ref{ch:g9:atoms-to-galaxies} --- Des atomes aux galaxies~: les échelles de l'Univers}

\begin{solution}{exo:g9:atoms-to-galaxies:1}
\omterm{def:g9:atoms-to-galaxies:atom}{Noyau} ($10^{-15}$ \unit{m}), \omterm{def:g9:atoms-to-galaxies:atom}{atome} ($10^{-10}$), \omterm{def:g7:states-of-matter:molecule}{molécule} (autour de
$10^{-9}$), fourmi ($10^{-2}$, ou $10^{-3}$ pour une modeste).
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:2}
Un \omterm{def:g9:atoms-to-galaxies:atom}{noyau} minuscule, lourd et chargé positivement au centre~; un nuage
d'\omterm{def:g9:atoms-to-galaxies:atom}{électrons} négatifs et presque sans \omterm{def:g3:measuring-mass:mass}{masse} très loin autour. Les
tailles~: l'\omterm{def:g9:atoms-to-galaxies:atom}{atome} près de $10^{-10}$ \unit{m}, le \omterm{def:g9:atoms-to-galaxies:atom}{noyau} près de
$10^{-15}$ --- l'écart de cent mille fois qui fait le petit pois dans
un stade.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:3}
Les \omterm{def:g9:atoms-to-galaxies:atom}{électrons} --- négatifs, marchant du $-$ vers le $+$~: à l'opposé
de la convention. Les lois survivent parce qu'elles n'ont jamais
dépendu de qui marche~: les courants, les sommes aux \omterm{def:g5:series-parallel-circuits:parallel}{nœuds}, les
portraits d'Ohm et les factures de \omterm{def:g9:electric-power-energy:power}{puissance} se lisent à l'identique
avec la flèche conservée par accord mondial.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:4}
La puissance de dix la plus proche. Taille~: $10^{0}$ \unit{m}~;
salle de classe~: $10^{1}$~; diamètre de la Terre~: $10^{7}$.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:5}
De l'\omterm{def:g9:atoms-to-galaxies:atom}{atome} au \omterm{def:g9:atoms-to-galaxies:atom}{noyau}~: cinq barreaux. De toi à la Terre~: sept. De la
distance du Soleil ($10^{11}$) à l'\omterm{def:g4:solar-system:star}{étoile} la plus proche
($4 \times 10^{16}$)~: cinq à six barreaux --- quelques centaines de
milliers de fois.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:6}
L'\omterm{def:g9:atoms-to-galaxies:atom}{atome} est $10^{-10}$ \unit{m} de presque rien autour d'un \omterm{def:g9:atoms-to-galaxies:atom}{noyau} de
$10^{-15}$~: entasse des \omterm{def:g9:atoms-to-galaxies:atom}{atomes} dans une table et le vide vient avec
eux. La solidité est le refus électrique des nuages d'\omterm{def:g9:atoms-to-galaxies:atom}{électrons} de se
recouvrir --- du vide, fermement défendu.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:7}
Millimètre, $10^{-3}$~; \omterm{def:g9:atoms-to-galaxies:atom}{atome}, $10^{-10}$~: sept barreaux --- environ
dix millions d'\omterm{def:g9:atoms-to-galaxies:atom}{atomes} dans la rangée.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:8}
\omterm{def:g8:speed-of-light:lightyear}{Année-lumière}~: $10^{16}$ \unit{m} (neuf mille cinq cents millions de
millions). \omterm{ex:g5:stars-night-sky:milkyway}{Voie lactée}~: $10^{21}$. Ciel le plus profond exploré~:
près de $10^{26}$.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:9}
$10^{5}$ petits pois~: $0.01 \times 10^{5} = \qty{1000}{m}$ --- un
«~stade~» large d'un kilomètre~: plus vaste qu'aucun stade réel, mais
l'\omterm{prop:g5:mirrors-reflection:image}{image} tient --- un petit pois au point central, et les petits pois
suivants à un kilomètre de là.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:10}
Gouttes dans les mers~: environ $10^{21} \times 10^{5} = 10^{26}$ ---
la vieille affirmation, prise au pied de la lettre, allait donc trop
loin~: les mers contiennent plus de gouttes qu'une goutte ne contient
de \omterm{def:g7:states-of-matter:molecule}{molécules}, et de plusieurs barreaux. La version honnête --- les
\omterm{def:g7:states-of-matter:molecule}{molécules} d'une goutte dépassent en nombre tout ce qu'on pourrait
compter dans une vie --- survit~; les échelles gardent honnêtes même
les affirmations qu'on aime.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:11}
Environ dix barreaux vers le bas jusqu'à l'\omterm{def:g9:atoms-to-galaxies:atom}{atome} ($10^{0}$ à
$10^{-10}$) et vingt-six vers le haut jusqu'au ciel profond ---
proche du milieu, légèrement penché du côté du petit. Neuf années
d'école mettent un lecteur à portée honnête des deux extrémités de
tout ce qui a été mesuré.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:12}
Les réponses sont personnelles --- l'exercice est la lettre
elle-même. (Un spécimen, pour la forme~: au lecteur de première année
du chapitre des \omterm{def:g1:light-and-shadows:shadow}{ombres} --- «~le jumeau sombre sur le trottoir
expliquera un jour les \omterm{def:g7:shadows-eclipses:solar}{éclipses de Soleil}~»~; au lecteur de cinquième
année du chapitre sur l'\omterm{def:g5:everyday-energy:energy}{énergie} --- «~les chaînes que tu traces
deviendront des formules avec des \omterm{def:g9:kinetic-energy-safety:joule}{joules} dedans~»~; au lecteur de
huitième année de la loi d'Ohm --- «~les portraits que tu traces sont
l'habitude d'éprouver les lois, et c'est là tout le secret~».)
\end{solution}

\begin{solution}{pb:g9:atoms-to-galaxies:1}
\textbf{1.} Cour de l'école~: seconde $2$~; la Terre~: seconde $7$~;
l'\omterm{def:g6:sun-earth-moon:axisorbit}{orbite} de la \omterm{def:g2:moon-in-the-sky:moon}{Lune}~: seconde $9$.
\textbf{2.} Seconde $11$ (à $1.5 \times 10^{11}$ \unit{m})~; le
narrateur~: «~la \omterm{def:g1:light-and-shadows:source}{lumière} du Soleil sur la nappe a huit minutes~».
\textbf{3.} Grossièrement des secondes $13$ à $16$~: le cadre tient
tout le \omterm{def:g4:solar-system:family}{Système solaire} en un point et aucune \omterm{def:g4:solar-system:star}{étoile} encore ---
honnêtement, un vide presque noir, et des secondes entières~: les
scènes les plus vraies du film.
\textbf{4.} Le ciel le plus profond exploré, $10^{26}$ \unit{m} de
galaxies --- sa \omterm{def:g1:light-and-shadows:source}{lumière} la plus ancienne étant plus vieille que la
Terre elle-même.
\textbf{5.} Fourmi~: seconde $2$ vers l'intérieur~; cellule~: seconde
$5$~; \omterm{def:g7:states-of-matter:molecule}{molécule}~: seconde $9$~; \omterm{def:g9:atoms-to-galaxies:atom}{atome}~: seconde $10$.
\textbf{6.} À travers le vide propre de l'\omterm{def:g9:atoms-to-galaxies:atom}{atome} --- cinq secondes de
rien entre le nuage et le centre~: «~si cet \omterm{def:g9:atoms-to-galaxies:atom}{atome} était un stade, nous
avons quitté les gradins les plus hauts et nous volons encore vers un
petit pois~».
\textbf{7.} Le \omterm{def:g9:atoms-to-galaxies:atom}{noyau} --- et honnêtement~: «~ici s'arrête notre
échelle, non parce que la nature s'arrête, mais parce que ce sont les
plus petits barreaux que les humains aient mesurés~; la construction
continue~».
\textbf{8.} Vers l'extérieur, $26$ secondes~; vers l'intérieur,
$15$~: un univers de quarante et une secondes.
\textbf{9.} Chaque seconde multiplie la vue par dix --- des
\emph{rapports} égaux, non des additions égales~: la logique de
l'échelle est qu'un barreau est un facteur, et qu'en facteurs le pas
de la rue et le pas de l'amas ont la même taille.
\textbf{10.} Par exemple~: «~Si plein que cela paraisse, presque tout
ici est vide --- la matière comme le ciel ne sont que de rares îles
dans un vaste rien.~»
\textbf{11.} Le microscope (chapitre des \omterm{def:g8:lenses-images:families}{lentilles}~: deux \omterm{def:g8:lenses-images:families}{lentilles
convergentes} au-dessus du petit) et la lunette astronomique (même
chapitre, rassemblant le faible et le lointain) --- les clés des deux
directions, toutes deux taillées dans l'optique de ce livre.
\textbf{12.} Par exemple~: «~Neuf années, une seule échelle~: des
pattes comptables d'une coccinelle aux galaxies plus anciennes que le
sol sous nos pieds, chaque barreau a été posé par une mesure que
n'importe qui peut vérifier. L'an prochain, les questions que nous
avons laissées ouvertes --- la \omterm{def:g2:pushes-and-pulls:force}{force} et le mouvement, la nature de la
\omterm{def:g1:light-and-shadows:source}{lumière}, le cœur de l'\omterm{def:g9:atoms-to-galaxies:atom}{atome} --- commencent à recevoir leur réponse~;
apporte l'échelle.~»
\end{solution}
