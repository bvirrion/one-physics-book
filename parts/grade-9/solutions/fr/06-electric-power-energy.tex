\section*{Chapitre \ref{ch:g9:electric-power-energy} --- La puissance et l'énergie électriques}

\begin{solution}{exo:g9:electric-power-energy:1}
La \omterm{def:g9:electric-power-energy:power}{puissance} est l'allure de conversion de l'\omterm{def:g5:everyday-energy:energy}{énergie} --- des \omterm{def:g9:kinetic-energy-safety:joule}{joules}
par seconde, en \omterm{def:g9:electric-power-energy:power}{watts}~: le débit du robinet, non le seau.
$P = U \times I$ et $E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
Lampe~: $230 \times 0.022 \approx \qty{5}{W}$. Démarreur~:
$12 \times 100 = \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
Bouilloire~: $2300 \div 230 = \qty{10}{A}$~; micro-ondes~:
$700 \div 230 \approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
Parce que la \omterm{def:g5:heat-and-insulation:heat}{chaleur} est la forme d'\omterm{def:g5:everyday-energy:energy}{énergie} la plus coûteuse à
fournir~: élever la \omterm{def:g3:temperature-thermometers:temperature}{température} de l'eau ou d'une pièce dévore des
\omterm{def:g9:kinetic-energy-safety:joule}{joules} à une allure dont les lampes et l'électronique n'approchent
jamais --- si bien que la plaque de tout appareil de chauffage
s'exprime en kilowatts.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. Un mois de \omterm{def:g9:kinetic-energy-safety:joule}{joules} compte dix chiffres~; les
fabricants de compteurs ont choisi une \omterm{def:g6:measurement-in-science:measuring}{unité} qui garde les mois
domestiques dans de confortables centaines.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
Radiateur~: $2 \times 0.5 = \qty{1}{kWh}$. Télévision~:
$0.1 \times 5 = \qty{0.5}{kWh}$. La demi-heure de radiateur dépense
le double de toute la soirée de télévision.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$~; coût
$6 \times 0.25 = 1.5$ \omterm{def:g6:measurement-in-science:measuring}{unité} monétaire --- un mois de devoirs pour le
prix d'une viennoiserie.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}
Chaque \omterm{def:g9:electric-power-energy:kwh}{kilowattheure} qui entre dans la maison. La facture retranche la
lecture du mois dernier de celle de ce mois-ci et multiplie la
différence par le tarif.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
Ancêtres~: $0.6 \times 4 \times 30 = \qty{72}{kWh}$. Modernes~:
$0.05 \times 4 \times 30 = \qty{6}{kWh}$. Économie~: \qty{66}{kWh}
--- $16.5$ \omterm{def:g6:measurement-in-science:measuring}{unités} monétaires par mois, pour un après-midi sur un
escabeau.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}
Sèche-cheveux~: $2000 \div 230 \approx \qty{8.7}{A}$. Avec les
\qty{10}{A} de la bouilloire~: $18.7$ --- au-delà du calibre de
\qty{16}{A} de la \omterm{def:g5:series-parallel-circuits:parallel}{branche}~: le disjoncteur mettra (à juste titre) fin
à l'expérience.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
Bouilloire~: $60 \div 2.3 \approx 26$ \omterm{ex:g2:measuring-time:clock}{heures}. Lampe~:
$60 \div 0.005 = 12000$ \omterm{ex:g2:measuring-time:clock}{heures} --- un an et quatre mois. Et pourtant
la même réserve ne fait rouler la voiture que quelques centaines de
kilomètres~: déplacer une tonne de machine dévore de l'\omterm{def:g5:everyday-energy:energy}{énergie} à une
échelle à côté de laquelle l'éclairage domestique est une erreur
d'arrondi.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
Prédiction~: $t = E/P = \num{5.0e5} \div 2300 \approx \qty{217}{s}$
--- environ trois minutes et demie. La vraie bouilloire met plus
longtemps~: une part de ses \omterm{def:g9:kinetic-energy-safety:joule}{joules} fuit à côté de l'eau --- en
échauffant le corps même de la bouilloire, l'\omterm{def:g2:air-around-us:air}{air}, en s'échappant en
vapeur précoce. L'\omterm{def:g5:everyday-energy:energy}{énergie} n'est jamais détruite, mais elle n'est pas
non plus livrée tout entière là où on le voudrait~: la fuite de la
chaîne, enfin chronométrée dans ta propre cuisine.
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} Télévision~: $0.1 \times 4 \times 30 = \qty{12}{kWh}$~;
lampes~: $0.05 \times 5 \times 30 = \qty{7.5}{kWh}$.
\textbf{3.} Lave-linge~: $2 \times 1 \times 15 = \qty{30}{kWh}$~;
chauffe-eau~: $2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- dix \omterm{def:g9:electric-power-energy:power}{watts}
cachés, tout un mois de bouilloire.
\textbf{5.} Moyenne $120 \div 3 = \qty{40}{W}$~; mois~:
$0.04 \times 720 = \qty{28.8}{kWh}$.
\textbf{6.} Total $\approx \qty{242}{kWh}$. Lignes dominantes~: le
chauffe-eau (\qty{150}{}), le lave-linge (\qty{30}{}), le
réfrigérateur (\qty{28.8}{kWh}) --- le chauffage et ce qui ne
s'arrête jamais.
\textbf{7.} $242 \times 0.25 \approx 60.6$ \omterm{def:g6:measurement-in-science:measuring}{unités}~; part du
chauffe-eau~: $150 \times 0.25 = 37.5$ --- bien plus de la moitié de
la facture.
\textbf{8.} $2150 + 242 = 2392$ \omterm{def:g6:measurement-in-science:measuring}{unités}.
\textbf{9.} Nouvelle ligne du chauffe-eau~:
$2.5 \times 1.5 \times 30 = \qty{112.5}{kWh}$~: économie
\qty{37.5}{kWh}, environ $9.4$ \omterm{def:g6:measurement-in-science:measuring}{unités} monétaires par mois.
\textbf{10.} Un chargeur au repos~:
$0.0002 \times 720 = \qty{0.14}{kWh}$ --- environ quatre centièmes
d'\omterm{def:g6:measurement-in-science:measuring}{unité} monétaire~: la proposition une économise deux cents fois plus
par chargeur. Les audits trouvent des \omterm{def:g3:levers-and-scales:lever}{leviers}~; les slogans trouvent
des chargeurs.
\textbf{11.} La ligne de la bouilloire tombe de moitié, à
\qty{3.45}{kWh}~: économie \qty{3.45}{kWh}, moins d'une \omterm{def:g6:measurement-in-science:measuring}{unité}
monétaire --- l'habitude de la grand-mère a raison, et reste modeste.
\textbf{12.} Par exemple~: «~Les \omterm{def:g9:electric-power-energy:kwh}{kilowattheures} vivent là où l'eau
chauffe et là où les machines ne dorment jamais. La ligne de la veille
enseigne le facteur temps~: de minuscules \omterm{def:g9:electric-power-energy:power}{puissances} font grossir les
factures quand $t$ vaut toujours. Et le changement le plus rentable
est la douche plus courte --- le chauffe-eau possède la facture, donc
ses \omterm{ex:g2:measuring-time:clock}{heures} possèdent les économies.~»
\end{solution}
