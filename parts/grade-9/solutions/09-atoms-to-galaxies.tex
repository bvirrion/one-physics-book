\section*{Chapter \ref{ch:g9:atoms-to-galaxies} --- From Atoms to Galaxies: Scales of the Universe}

\begin{solution}{exo:g9:atoms-to-galaxies:1}
\omterm{def:g9:atoms-to-galaxies:atom}{Nucleus} ($10^{-15}$ \unit{m}), \omterm{def:g9:atoms-to-galaxies:atom}{atom} ($10^{-10}$), \omterm{def:g7:states-of-matter:molecule}{molecule}
(around $10^{-9}$), ant ($10^{-2}$, or $10^{-3}$ for a

\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:2}
A minuscule, heavy, positively charged \omterm{def:g9:atoms-to-galaxies:atom}{nucleus} at the
center; a cloud of nearly weightless negative \omterm{def:g9:atoms-to-galaxies:atom}{electrons} far
outside it. The sizes: \omterm{def:g9:atoms-to-galaxies:atom}{atom} near $10^{-10}$ \unit{m},
\omterm{def:g9:atoms-to-galaxies:atom}{nucleus} near $10^{-15}$ --- the hundred-thousandfold gap
that makes the pea-in-a-stadium.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:3}
\omterm{def:g9:atoms-to-galaxies:atom}{Electrons} --- negative, marching from $-$ to $+$: opposite
to the convention. The laws survive because they never
depended on who marches: currents, \omterm{def:g5:series-parallel-circuits:parallel}{junction} sums, Ohm's
portraits and \omterm{def:g9:electric-power-energy:power}{power} bills read identically with the arrow
kept by worldwide agreement.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:4}
The nearest power of ten. Height: $10^{0}$ \unit{m};
classroom: $10^{1}$; Earth's diameter: $10^{7}$.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:5}
\omterm{def:g9:atoms-to-galaxies:atom}{Atom} to \omterm{def:g9:atoms-to-galaxies:atom}{nucleus}: five rungs. You to the Earth: seven. Sun's
distance ($10^{11}$) to nearest \omterm{def:g4:solar-system:star}{star} ($4 \times 10^{16}$):
five to six rungs --- a few hundred thousandfold.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:6}
The \omterm{def:g9:atoms-to-galaxies:atom}{atom} is $10^{-10}$ \unit{m} of mostly nothing around a
$10^{-15}$ \omterm{def:g9:atoms-to-galaxies:atom}{nucleus}: pack \omterm{def:g9:atoms-to-galaxies:atom}{atoms} into a table and the void
comes with them. Solidity is the \omterm{def:g9:atoms-to-galaxies:atom}{electron} clouds' electric
refusal to overlap --- emptiness, firmly defended.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:7}
Millimetre, $10^{-3}$; \omterm{def:g9:atoms-to-galaxies:atom}{atom}, $10^{-10}$: seven rungs ---
about ten million \omterm{def:g9:atoms-to-galaxies:atom}{atoms} in the row.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:8}
\omterm{def:g8:speed-of-light:lightyear}{Light-year}: $10^{16}$ \unit{m} (nine and a half thousand
million million). \omterm{ex:g5:stars-night-sky:milkyway}{Milky Way}: $10^{21}$. Deepest surveyed
sky: near $10^{26}$.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:9}
$10^{5}$ peas: $0.01 \times 10^{5} = \qty{1000}{m}$ --- a
kilometre-wide ``stadium'': grander than any real one, but
the picture holds --- pea at the center spot, and the next
peas a kilometre away.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:10}
Drops in the seas: about $10^{21} \times 10^{5} = 10^{26}$
--- so the old claim, taken literally, overreached: the seas
hold more drops than a drop holds \omterm{def:g7:states-of-matter:molecule}{molecules} by several
rungs. The honest version --- a drop's \omterm{def:g7:states-of-matter:molecule}{molecules} outnumber
anything countable in a lifetime --- survives; ladders keep
even beloved claims honest.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:11}
About ten rungs down to the \omterm{def:g9:atoms-to-galaxies:atom}{atom} ($10^{0}$ to $10^{-10}$)
and twenty-six up to the deep sky --- close to the middle,
tilted a little toward the small. Nine years of schooling
put a reader within honest reach of both ends of everything
measured.
\end{solution}

\begin{solution}{pb:g9:atoms-to-galaxies:1}
\textbf{1.} Schoolyard: second $2$; the Earth: second $7$;
the \omterm{def:g2:moon-in-the-sky:moon}{Moon}'s \omterm{def:g6:sun-earth-moon:axisorbit}{orbit}: second $9$.
\textbf{2.} Second $11$ (at $1.5 \times 10^{11}$ \unit{m});
the narrator: ``the sunshine on the blanket is eight
minutes old.''
\textbf{3.} Roughly seconds $13$ to $16$: the frame holds
the whole \omterm{def:g4:solar-system:family}{Solar System} as a dot and no \omterm{def:g4:solar-system:star}{star} yet ---
honestly, near-black emptiness, seconds of it: the film's
truest scenes.
\textbf{4.} The deepest surveyed sky, $10^{26}$ \unit{m} of
galaxies --- its oldest \omterm{def:g1:light-and-shadows:source}{light} older than the Earth itself.
\textbf{5.} Ant: second $2$ inward; cell: second $5$;
\omterm{def:g7:states-of-matter:molecule}{molecule}: second $9$; \omterm{def:g9:atoms-to-galaxies:atom}{atom}: second $10$.
\textbf{6.} Through the \omterm{def:g9:atoms-to-galaxies:atom}{atom}'s own emptiness --- five
seconds of nothing between cloud and center: ``if this \omterm{def:g9:atoms-to-galaxies:atom}{atom}
were a stadium, we have left the highest stands and are
still flying toward a pea.''
\textbf{7.} The \omterm{def:g9:atoms-to-galaxies:atom}{nucleus} --- and honestly: ``here our ladder
ends, not because nature stops, but because these are the
smallest rungs humans have measured; the building
continues.''
\textbf{8.} Outward $26$ seconds; inward $15$: a
forty-one-second universe.
\textbf{9.} Each second multiplies the view tenfold ---
equal \emph{ratios}, not equal additions: the ladder's
logic is that a rung is a factor, and by factors the
street-step and the cluster-step are the same size.
\textbf{10.} For example: ``However full it looks, almost
everything here is empty --- matter and sky alike are rare
islands in wide nothing.''
\textbf{11.} The microscope (lenses chapter: two \omterm{def:g8:lenses-images:families}{converging
lenses} over the small) and the telescope (same chapter,
gathering the faint far) --- the two directions' keys, both
ground from this book's optics.
\textbf{12.} For example: ``Nine years, one ladder: from a
ladybug's countable legs to galaxies older than the ground
underfoot, every rung was set by measurement anyone may
check. Next year the questions we left open --- \omterm{def:g2:pushes-and-pulls:force}{force} and
motion, the nature of \omterm{def:g1:light-and-shadows:source}{light}, the \omterm{def:g9:atoms-to-galaxies:atom}{atom}'s heart --- begin to
answer; bring the ladder.''
\end{solution}
