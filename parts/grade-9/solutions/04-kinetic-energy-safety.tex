\section*{Chapter \ref{ch:g9:kinetic-energy-safety} --- Kinetic Energy and Road Safety}

\begin{solution}{exo:g9:kinetic-energy-safety:1}
The \omterm{def:g9:kinetic-energy-safety:joule}{joule} (\unit{J}): a few \omterm{def:g9:kinetic-energy-safety:joule}{joules} lift this book to its
shelf; the heart spends about one per beat. $E_k = \frac12 m
v^2$ --- $m$ in \omterm{def:g3:measuring-mass:units}{kilograms}, $v$ in \omterm{def:g2:measuring-length:units}{metres} per second, $E_k$ in
\omterm{def:g9:kinetic-energy-safety:joule}{joules}.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:2}
Hare: $0.5 \times 2 \times 100 = \qty{100}{J}$. Player: $0.5
\times 80 \times 25 = \qty{1000}{J}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:3}
Half more \omterm{def:g3:measuring-mass:mass}{mass}: factor $1.5$. Half more \omterm{def:g7:motion-average-speed:formula}{speed}: factor $1.5^2
= 2.25$. The \omterm{def:g7:motion-average-speed:formula}{speed} dial wins --- it always does, wearing the
square.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:4}
\omterm{prop:g9:kinetic-energy-safety:stopping}{Reaction distance} --- one second's travel, growing in
proportion to $v$; \omterm{prop:g9:kinetic-energy-safety:stopping}{braking distance} --- the retirement of
$E_k$, growing with $v^2$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:5}
At \qty{50}{km/h}: $13$ of \qty{27}{m} --- about half. At
\qty{130}{km/h}: $85$ of \qty{121}{m} --- some seventy
percent. The squared component outgrows the proportional one:
at \omterm{def:g7:motion-average-speed:formula}{speed}, braking devours the total.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:6}
Into \omterm{def:g5:heat-and-insulation:heat}{heat} at the brake discs --- the amber glow after
mountain descents, the hot-brake smell. \omterm{def:g5:everyday-energy:energy}{Energy} is passed on,
never erased: the chain proposition of the childhood \omterm{def:g5:everyday-energy:energy}{energy}
chapter, still in command.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:7}
$36 \div 3.6 = \qty{10}{m}$; then \qty{20}{m}; then
\qty{30}{m} --- marching in proportion to the \omterm{def:g7:motion-average-speed:formula}{speed}, as a
one-second stretch must.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:8}
Your $E_k$ is fixed by your \omterm{def:g3:measuring-mass:mass}{mass} and the car's \omterm{def:g7:motion-average-speed:formula}{speed} --- no
strap changes it. The belt civilizes the \emph{retirement}:
stopping the body over a longer distance and time, gently,
instead of against the dashboard, all at once.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:9}
Braking scales with the square of \omterm{def:g7:motion-average-speed:formula}{speed}: at \qty{60}{km/h}
(double), $8 \times 4 = \qty{32}{m}$; at \qty{90}{km/h}
(triple), $8 \times 9 = \qty{72}{m}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:10}
At \qty{30}{km/h}: reaction $\approx \qty{8.3}{m}$, braking
$13 \times (30/50)^2 \approx \qty{4.7}{m}$ --- stopping in
about \qty{13}{m}, against \qty{27}{m} at fifty: less than
half the road. Energies: $(30/50)^2 = 0.36$ --- barely a
third of the crash \omterm{def:g5:everyday-energy:energy}{energy}. Both persuade; the poster-ready
line is usually the road one --- ``at 30 you stop before the
child; at 50 you reach them still moving''.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:11}
Double only the braking share --- rain lengthens the
retirement, not the brain: at \qty{50}{km/h}, $14 + 26 =
\qty{40}{m}$; at \qty{90}{km/h}, $25 + 80 = \qty{105}{m}$.
The reaction second is dry and wet alike.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:12}
Truck: $v = 25$ \unit{m/s}; $E_k = 0.5 \times 20000 \times
625 = \qty{6.25e6}{J}$. Matching car: $0.5 \times 1000 \times
v^2 = \num{6.25e6}$ gives $v^2 = 12500$, $v \approx
\qty{112}{m/s} \approx \qty{400}{km/h}$ --- no road car
approaches it. A truck at highway \omterm{def:g7:motion-average-speed:formula}{speed} carries racing-car
energies with lorry brakes: when they overheat and fail on a
long descent, only a gravel ramp can retire megajoules
safely. Cars never need one; their energies die in ordinary
brakes.
\end{solution}

\begin{solution}{pb:g9:kinetic-energy-safety:1}
\textbf{1.} $50 \div 3.6 = \qty{13.9}{m/s}$; $60 \div 3.6 =
\qty{16.7}{m/s}$.
\textbf{2.} \qty{13.9}{m} and \qty{16.7}{m}.
\textbf{3.} \qty{13}{m}; at sixty, $13 \times (60/50)^2 =
13 \times 1.44 \approx \qty{18.7}{m}$.
\textbf{4.} Stopping: $26.9$ against \qty{35.4}{m} --- the
poster's blank: about \emph{eight and a half \omterm{def:g2:measuring-length:units}{metres}} more,
two car-lengths past where the fifty-driver has stopped.
\textbf{5.} At fifty: about \qty{97}{kJ}. At ninety ($v =
\qty{25}{m/s}$): $0.5 \times 1000 \times 625 =
\qty{312}{kJ}$.
\textbf{6.} $97 \div 9.8 \approx \qty{10}{m}$; $312 \div 9.8
\approx \qty{32}{m}$.
\textbf{7.} Ten \omterm{def:g2:measuring-length:units}{metres}: a three-storey house; thirty-two: a
ten-storey block. Draft: ``A crash at 50 is a fall from the
third floor. At 90, from the tenth. Choose your window.''
\textbf{8.} The heights honestly \omterm{def:g6:measurement-in-science:measuring}{measure} the \omterm{def:g5:everyday-energy:energy}{energy} that
must be retired --- that number no engineering changes.
Crumple zones and belts change the retirement's
\emph{manner}: \omterm{def:g2:measuring-length:units}{metres} of gentle stopping instead of
\omterm{def:g2:measuring-length:units}{centimetres} of cruel one --- which is why the fall analogy
overstates injury in a modern car, and why the \omterm{def:g5:everyday-energy:energy}{energy} it
depicts is real all the same.
\textbf{9.} \qty{25}{m} blind per second; a two-second
glance, \qty{50}{m} --- half a pitch before the reaction
second even starts.
\textbf{10.} $65 + 25 = \qty{90}{m}$ --- the tired driver
adds a bus-length and a half to the alert driver's total.
\textbf{11.} Most wrong at \emph{low} \omterm{def:g7:motion-average-speed:formula}{speeds}: there the
squared braking share is small and the reaction stretch is
most of the stopping distance --- in town, the second
\emph{is} the danger. (At high \omterm{def:g7:motion-average-speed:formula}{speed} braking dominates ---
but \qty{25}{m} per second is no trifle there either.)
\textbf{12.} For example: ``Ten more of \omterm{def:g7:motion-average-speed:formula}{speed}, ten more
\omterm{def:g2:measuring-length:units}{metres} of stopping --- \omterm{def:g7:motion-average-speed:formula}{speed} enters twice, once squared.
A crash is a fall: \omterm{def:g5:everyday-energy:energy}{energy} grows with the square of \omterm{def:g7:motion-average-speed:formula}{speed}.
One second of glance is twenty-five \omterm{def:g2:measuring-length:units}{metres} of blindness ---
attention is the shortest \omterm{prop:g9:kinetic-energy-safety:stopping}{braking distance} of all.''
\end{solution}
