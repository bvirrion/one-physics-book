\section*{Chapter \ref{ch:g9:weight-vs-mass} --- Weight vs Mass: Measuring Forces}

\begin{solution}{exo:g9:weight-vs-mass:1}
The \omterm{def:g9:weight-vs-mass:newton}{newton} (\unit{N}), measured with the \omterm{def:g9:weight-vs-mass:dynamometer}{dynamometer}. Anchors:
one \omterm{def:g9:weight-vs-mass:newton}{newton} --- a small apple's \omterm{def:g4:weight-and-mass:weight}{weight}; a hundred --- a solid
handshake (or a \qty{10}{kg} suitcase's \omterm{def:g4:weight-and-mass:weight}{weight}).
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:2}
$P = m \times g$: \omterm{def:g4:weight-and-mass:weight}{weight} in \omterm{def:g9:weight-vs-mass:newton}{newtons}, \omterm{def:g3:measuring-mass:mass}{mass} in \omterm{def:g3:measuring-mass:units}{kilograms}, and
$g$ --- the place's \omterm{prop:g9:weight-vs-mass:formula}{gravitational strength} --- the \omterm{def:g9:weight-vs-mass:newton}{newtons} of
pull each \omterm{def:g3:measuring-mass:units}{kilogram} receives there (\qty{9.8}{N/kg} on Earth).
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:3}
Dog: $25 \times 9.8 = \qty{245}{N}$. Loaf: $0.7 \times 9.8
\approx \qty{6.9}{N}$. A \qty{55}{kg} reader: about
\qty{539}{N}.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:4}
$m = 14.7 \div 9.8 = \qty{1.5}{kg}$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:5}
The slope is the local $g$ --- \omterm{def:g9:weight-vs-mass:newton}{newtons} per \omterm{def:g3:measuring-mass:units}{kilogram}. On the
\omterm{def:g2:moon-in-the-sky:moon}{Moon} the line stays straight through the origin but slumps to
a gentle $1.6$ slope: same proportionality, weaker world.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:6}
The \omterm{def:g3:measuring-mass:mass}{mass} column never changes --- \qty{10}{kg} of stuff
travels intact. The $g$ and \omterm{def:g4:weight-and-mass:weight}{weight} columns change with the
world: \omterm{def:g4:weight-and-mass:weight}{weight} is the local toll, $m \times g$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:7}
The balance compares $m_1 g$ with $m_2 g$: $g$ multiplies both
pans and cancels --- \omterm{def:g3:measuring-mass:mass}{mass}, truthfully, anywhere. The bathroom
scale \omterm{def:g6:measurement-in-science:measuring}{measures} $P = m g_{\text{here}}$ but divides by
$g_{\text{Earth}}$ to print \omterm{def:g3:measuring-mass:units}{kilograms}: on the \omterm{def:g2:moon-in-the-sky:moon}{Moon} it prints
$m \times 1.6/9.8$ --- one sixth of the truth.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:8}
\omterm{def:g4:weight-and-mass:weight}{Weight}: \qty{19.6}{N} down --- an arrow about \qty{3.9}{cm}
long at five \omterm{def:g9:weight-vs-mass:newton}{newtons} per \omterm{def:g2:measuring-length:units}{centimetre}, from the book downward.
Support: an equal \qty{3.9}{cm} arrow upward. Equal lengths,
opposite ways: budget balanced, book still.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:9}
The plane must carry, lift and brake the bag's \emph{stuff} ---
its \omterm{def:g3:measuring-mass:mass}{mass} --- and \qty{23}{kg} of stuff is \qty{23}{kg} on any
world. The lunar desk needs a balance with marked masses: the
only scale that survives the change of $g$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:10}
$m = 111 \div 3.7 = \qty{30}{kg}$; on Earth it weighs $30
\times 9.8 = \qty{294}{N}$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:11}
It reads too high: hanging, the spring also carries its own
(and its hook's) \omterm{def:g4:weight-and-mass:weight}{weight}, which the flat zero never included
--- the instrument adds a constant surplus. Zero it hanging,
and the surplus is subtracted before the load arrives.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:12}
Equipment: \omterm{def:g9:weight-vs-mass:dynamometer}{dynamometer}, set of marked masses ($1$ to
\qty{5}{kg}), hook. Hang each \omterm{def:g3:measuring-mass:mass}{mass}, record ($m$, $P$) pairs,
plot $P$ against $m$: the points align through the origin, and
the slope --- rise in \omterm{def:g9:weight-vs-mass:newton}{newtons} over run in \omterm{def:g3:measuring-mass:units}{kilograms} --- is
$g \approx \qty{9.8}{N/kg}$. On the \omterm{def:g2:moon-in-the-sky:moon}{Moon}: identical protocol,
identical straightness, new slope $1.6$ --- the method
\omterm{def:g6:measurement-in-science:measuring}{measures} whatever world it stands on; only the number is
local.
\end{solution}

\begin{solution}{pb:g9:weight-vs-mass:1}
\textbf{1.} Earth $70 \times 9.8 = \qty{686}{N}$; \omterm{def:g2:moon-in-the-sky:moon}{Moon}
\qty{112}{N}; Mars \qty{259}{N}; Jupiter deck
\qty{1736}{N}.
\textbf{2.} Correct on Earth: \qty{70}{kg}. \omterm{def:g2:moon-in-the-sky:moon}{Moon}: $112 \div
9.8 \approx \qty{11.4}{kg}$; Mars: $259 \div 9.8 \approx
\qty{26.4}{kg}$; Jupiter: $1736 \div 9.8 \approx
\qty{177}{kg}$ --- flattery and slander by formula.
\textbf{3.} \qty{70}{kg} at every port: the balance's
comparison cancels the local $g$.
\textbf{4.} Only Jupiter's deck: \qty{1736}{N} approaches the
limit but stays under --- no port breaks \qty{2000}{N} for
this traveler (a heavier colleague of \qty{90}{kg} would
break it there: $90 \times 24.8 = \qty{2232}{N}$).
\textbf{5.} Departure: $40 \times 9.8 = \qty{392}{N}$;
arrival: $40 \times 3.7 = \qty{148}{N}$.
\textbf{6.} $m = 200/g$: Earth \qty{20.4}{kg}; \omterm{def:g2:moon-in-the-sky:moon}{Moon}
\qty{125}{kg}; Mars \qty{54}{kg}; Jupiter \qty{8.1}{kg} ---
the union's fixed \omterm{def:g9:weight-vs-mass:newton}{newtons} buy very different masses.
\textbf{7.} $m = 150 \div 1.6 = \qty{93.75}{kg}$ --- about
\qty{94}{kg} of crate for customs.
\textbf{8.} $m = 294 \div 9.8 = \qty{30}{kg}$; \omterm{def:g2:moon-in-the-sky:moon}{Moon} \omterm{def:g4:weight-and-mass:weight}{weight}
$30 \times 1.6 = \qty{48}{N}$.
\textbf{9.} Muscles and \omterm{def:g3:measuring-mass:mass}{mass} came home unchanged; only the
opponent did. On the \omterm{def:g2:moon-in-the-sky:moon}{Moon} each \omterm{def:g3:measuring-mass:units}{kilogram} cost \qty{1.6}{N} to
lift; home charges \qty{9.8}{N} again. The gains were the
\omterm{def:g2:moon-in-the-sky:moon}{Moon}'s discount, not the body's growth --- $P = m g$, with
$m$ loyal and $g$ fickle.
\textbf{10.} $0.2 \times 24.8 \approx \qty{5}{N}$ --- an
anvil it is not: on Earth the whisk weighed \qty{2}{N}, so
deck whisking is two-and-a-half-fold heavier, wearying for a
wrist but hardly foundry work. Complaint: sustained,
partially.
\textbf{11.} A \omterm{def:g9:weight-vs-mass:newton}{newton} of chocolate is $m = 1/g$: Earth about
\qty{102}{g}; \omterm{def:g2:moon-in-the-sky:moon}{Moon} \qty{625}{g}; Jupiter \qty{40}{g}. Buy
your \omterm{def:g9:weight-vs-mass:newton}{newtons} on the \omterm{def:g2:moon-in-the-sky:moon}{Moon}.
\textbf{12.} For example: ``\omterm{def:g3:measuring-mass:mass}{Mass} travels with you; \omterm{def:g4:weight-and-mass:weight}{weight} is
each world's toll on it --- pack \omterm{def:g3:measuring-mass:units}{kilograms}, and let the
ports charge their \omterm{def:g9:weight-vs-mass:newton}{newtons}.''
\end{solution}
