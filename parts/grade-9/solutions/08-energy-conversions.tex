\section*{Chapter \ref{ch:g9:energy-conversions} --- Forms of Energy and Conversions}

\begin{solution}{exo:g9:energy-conversions:1}
Kinetic (the rolling ball), potential (the raised bucket),
thermal (the warm bath), chemical (breakfast, the \omterm{def:g3:first-electric-circuit:battery}{battery}),
electrical (the wall's supply), radiant (sunshine through the
window), nuclear (the power station's core).
\end{solution}

\begin{solution}{exo:g9:energy-conversions:2}
Bucket: $2 \times 9.8 \times 10 = \qty{196}{J}$. Hiker: $60
\times 9.8 \times 500 = \qty{2.9e5}{J}$ --- nearly three
hundred kilojoules of altitude.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:3}
In every process, \omterm{def:g5:everyday-energy:energy}{energy} changes form and owner but the total
is exactly conserved. The forever-machines died of it: a
machine that outputs more than it stores or receives would
create \omterm{def:g5:everyday-energy:energy}{energy} --- forbidden without appeal.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:4}
Top of swing: all potential, momentarily still. Bottom: all
kinetic, fastest. The drift: each pass skims a little into
warmth of \omterm{def:g2:air-around-us:air}{air} and \omterm{def:g3:levers-and-scales:lever}{pivot}, and the swing dies down --- accounts
transferred, total intact.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:5}
$v = \sqrt{2 \times 9.8 \times 5} \approx \qty{9.9}{m/s}$ ---
about \qty{36}{km/h}. The \omterm{def:g3:measuring-mass:mass}{mass} canceled when $m g h$ was set
against $\frac12 m v^2$: heavy and \omterm{def:g1:light-and-shadows:source}{light} divers splash at the
same \omterm{def:g7:motion-average-speed:formula}{speed}.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:6}
Useful over total: $170 \div 200 = 0.85$ --- eighty-five
percent. The missing \qty{30}{J}: warmth in windings and
bearings.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:7}
Five percent of its input left as \omterm{def:g1:light-and-shadows:source}{light}; ninety-five left as
warmth. A device sorted by its outputs was a heater --- that
happened to glow.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:8}
Chemical (breakfast) $\to$ muscles' work $\to$ \omterm{prop:g9:kinetic-energy-safety:formula}{kinetic energy}
of rider and machine $\to$ stored as \omterm{def:g9:energy-conversions:forms}{potential energy}, $m g
h$, at the summit --- with thermal leaks throughout: warm
muscles, warm \omterm{def:g2:air-around-us:air}{air} stirred, warm tires. Total: conserved.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:9}
Per tonne: $1000 \times 9.8 \times 50 = \qty{4.9e5}{J}$;
at ninety percent, about \qty{4.4e5}{J} electrical. For
\qty{4.4e6}{W}: $\num{4.4e6} \div \num{4.4e5} = 10$ tonnes
per second through the turbines.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:10}
Before: $0.5 \times 9.8 \times 2 = \qty{9.8}{J}$. After the
bounce: $0.5 \times 9.8 \times 1.2 \approx \qty{5.9}{J}$. The
skim: about \qty{3.9}{J} into warmth of ball and floor (and a
click of \omterm{def:g3:sound-around-us:sound}{sound} --- itself soon warmth). The audit ends when
rebounds reach zero height: every \omterm{def:g9:kinetic-energy-safety:joule}{joule} of the original $9.8$
then rests in the thermal account.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:11}
$E_p = 500 \times 9.8 \times 30 = \qty{1.47e5}{J}$; at ninety
percent the meter must supply $\num{1.47e5} \div 0.9 \approx
\qty{1.6e5}{J}$. The extra tenth is not lost to the universe
--- it warms \omterm{ex:g6:circuits-and-safety:motor}{motor}, gears and shaft: conservation holds; the
lift's ledger simply includes a thermal line.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:12}
Bottom \omterm{def:g7:motion-average-speed:formula}{speed}: $v = \sqrt{2 \times 9.8 \times 4} \approx
\qty{8.9}{m/s}$. Ideally the right rim is regained at
exactly \qty{4}{m}; really a little lower, each crossing
lower still, until the skater rocks to rest at the bottom
--- the whole $m g h$ retired into warmth of wheels, ramp
and \omterm{def:g2:air-around-us:air}{air}. Nothing was destroyed, but everything useful was
spent: the world's ledger gained spread-out warmth, and the
skater lost the concentrated, reusable height --- the
amount survives; the usefulness does not return.
\end{solution}

\begin{solution}{pb:g9:energy-conversions:1}
\textbf{1.} $E_p = 2000 \times 9.8 \times 45 =
\qty{8.8e5}{J}$.
\textbf{2.} $v = \sqrt{2 \times 9.8 \times 45} \approx
\qty{29.7}{m/s} \approx \qty{107}{km/h}$.
\textbf{3.} A \qty{50}{m} crest demands more \omterm{def:g5:everyday-energy:energy}{energy} than the
summit granted: the ledger forbids it. From a \qty{45}{m}
start at rest, no later hill may exceed \qty{45}{m} --- and
really, friction demands visibly less.
\textbf{4.} Measured: $0.5 \times 2000 \times 28^2 =
\qty{7.8e5}{J}$ against the ideal \qty{8.8e5}{J}: about
\qty{1e5}{J} charged to the thermal account --- rails,
wheels and \omterm{def:g2:air-around-us:air}{air}, warmed.
\textbf{5.} $P = \num{8.8e5} \div 60 \approx
\qty{1.5e4}{W}$ --- some fifteen kilowatts.
\textbf{6.} $14.7 \div 20 \approx 0.74$: seventy-four
percent.
\textbf{7.} Ten \omterm{ex:g2:measuring-time:clock}{hours} at one lift per four minutes: $150$
lifts; each costs $20 \times \tfrac{1}{60} =
\qty{0.33}{kWh}$; daily $150 \times 0.33 = \qty{50}{kWh}$
--- $12.5$ currency \omterm{def:g6:measurement-in-science:measuring}{units}.
\textbf{8.} A generous brake-generator might recover a fair
share of the arrival \omterm{prop:g9:kinetic-energy-safety:formula}{kinetic energy} --- but every stage
(rails, brake, generator, transformer) pays its thermal
commission, and the recovered \omterm{def:g9:kinetic-energy-safety:joule}{joules} can never equal the
winch's spending: conservation allows recycling;
efficiency-less-than-one forbids the full circle. The ride
can defray itself, never pay for itself.
\textbf{9.} $v = \sqrt{2 \times 9.8 \times 50} \approx
\qty{31.3}{m/s}$.
\textbf{10.} Kinetic at entry $0.5 \times 500 \times 980 =
\qty{2.45e5}{J}$, plus the last ten \omterm{def:g2:measuring-length:units}{metres}' $500 \times 9.8
\times 10 = \qty{4.9e4}{J}$: about \qty{2.9e5}{J} for the
brakes --- sent, by induced eddy currents, into warmth of
the brake fins: induction as retirement plan.
\textbf{11.} The claim stands, felt if not worded exactly:
gondola and riders fell together, and falling companions
press on nothing --- the scream is shared free fall,
gravity present in full and unfelt.
\textbf{12.} For example: ``Our coasters live by one trade:
height for \omterm{def:g7:motion-average-speed:formula}{speed} and back, $m g h$ against $\frac12 m v^2$.
Every attraction obeyed one law: the \omterm{def:g5:everyday-energy:energy}{energy} total, audited
at every point, never moved. And one account taxed them
all: warmth --- friction's quiet commission, payable on
every ride.''
\end{solution}
