\section*{Chapter \ref{ch:g9:electric-power-energy} --- Electric Power and Energy}

\begin{solution}{exo:g9:electric-power-energy:1}
\omterm{def:g9:electric-power-energy:power}{Power} is the pace of \omterm{def:g5:everyday-energy:energy}{energy} conversion --- \omterm{def:g9:kinetic-energy-safety:joule}{joules} per second,
in \omterm{def:g9:electric-power-energy:power}{watts}: the tap's flow, not the bucket. $P = U \times I$
and $E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
Lamp: $230 \times 0.022 \approx \qty{5}{W}$. Starter: $12
\times 100 = \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
Kettle: $2300 \div 230 = \qty{10}{A}$; microwave: $700 \div
230 \approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
Because warmth is the costliest form of \omterm{def:g5:everyday-energy:energy}{energy} to supply: raising
water's or a room's \omterm{def:g3:temperature-thermometers:temperature}{temperature} devours \omterm{def:g9:kinetic-energy-safety:joule}{joules} at a pace
lamps and electronics never approach --- so every heater's
plate reads in kilowatts.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. A month of \omterm{def:g9:kinetic-energy-safety:joule}{joules} runs to ten digits; the
meter's makers chose a \omterm{def:g6:measurement-in-science:measuring}{unit} that keeps household months in
comfortable hundreds.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
Heater: $2 \times 0.5 = \qty{1}{kWh}$. Television: $0.1
\times 5 = \qty{0.5}{kWh}$. The half-hour heater outspends
the whole television evening, twofold.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$; cost $6 \times
0.25 = 1.5$ currency \omterm{def:g6:measurement-in-science:measuring}{units} --- a month of homework for the
price of a pastry.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}

\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
Ancestors: $0.6 \times 4 \times 30 = \qty{72}{kWh}$.
Moderns: $0.05 \times 4 \times 30 = \qty{6}{kWh}$. Saving:
\qty{66}{kWh} --- $16.5$ currency \omterm{def:g6:measurement-in-science:measuring}{units} monthly, from one

\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}

kettle's \qty{10}{A}: $18.7$ --- past the \qty{16}{A}
\omterm{def:g5:series-parallel-circuits:parallel}{branch}'s rating: the breaker will (rightly) end the

\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
Kettle: $60 \div 2.3 \approx 26$ \omterm{ex:g2:measuring-time:clock}{hours}. Lamp: $60 \div
0.005 = 12000$ \omterm{ex:g2:measuring-time:clock}{hours} --- a year and four months. Yet the
same store drives the car only a few hundred kilometres:
moving a tonne of machine devours \omterm{def:g5:everyday-energy:energy}{energy} on a scale beside
which household lighting is a rounding error.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
Prediction: $t = E/P = \num{5.0e5} \div 2300 \approx
\qty{217}{s}$ --- about three and a half minutes. The real
kettle runs longer: part of its \omterm{def:g9:kinetic-energy-safety:joule}{joules} leak past the water
--- warming the kettle's own body, the \omterm{def:g2:air-around-us:air}{air}, escaping as
early steam. \omterm{def:g5:everyday-energy:energy}{Energy} is never destroyed, but neither is it
all delivered where intended: the chain's leak, timed at
last in your own kitchen.
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} Television: $0.1 \times 4 \times 30 =
\qty{12}{kWh}$; lamps: $0.05 \times 5 \times 30 =
\qty{7.5}{kWh}$.
\textbf{3.} Washer: $2 \times 1 \times 15 = \qty{30}{kWh}$;
heater: $2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- ten
hidden \omterm{def:g9:electric-power-energy:power}{watts}, a kettle's whole month.
\textbf{5.} Average $120 \div 3 = \qty{40}{W}$; month:
$0.04 \times 720 = \qty{28.8}{kWh}$.
\textbf{6.} Total $\approx \qty{242}{kWh}$. Dominant: the
water heater (\qty{150}{}), the washer (\qty{30}{}), the
refrigerator (\qty{28.8}{kWh}) --- heating and always-on.
\textbf{7.} $242 \times 0.25 \approx 60.6$ \omterm{def:g6:measurement-in-science:measuring}{units}; the
heater's share: $150 \times 0.25 = 37.5$ --- well over
half the bill.
\textbf{8.} $2150 + 242 = 2392$ \omterm{def:g6:measurement-in-science:measuring}{units}.
\textbf{9.} New heater line: $2.5 \times 1.5 \times 30 =
\qty{112.5}{kWh}$: saving \qty{37.5}{kWh}, about $9.4$
\omterm{def:g6:measurement-in-science:measuring}{units} of currency monthly.
\textbf{10.} One idle charger: $0.0002 \times 720 =
\qty{0.14}{kWh}$ --- about four hundredths of a currency
\omterm{def:g6:measurement-in-science:measuring}{unit}: proposal one outsaves it two-hundredfold per
charger. Audits find \omterm{def:g3:levers-and-scales:lever}{levers}; slogans find chargers.
\textbf{11.} Kettle line halves to \qty{3.45}{kWh}: saving
\qty{3.45}{kWh}, under one currency \omterm{def:g6:measurement-in-science:measuring}{unit} --- grandmother's
habit is right, and modest.
\textbf{12.} For example: ``The \omterm{def:g9:electric-power-energy:kwh}{kilowatt-hours} live where
water gets hot and where machines never sleep. The standby
line teaches the time factor: tiny \omterm{def:g9:electric-power-energy:power}{powers} grow bills when
$t$ is always. And the best-paying change is the shorter
shower --- the heater owns the bill, so its \omterm{ex:g2:measuring-time:clock}{hours} own the
savings.''
\end{solution}
