\section*{Hoofdstuk \ref{ch:g9:electric-power-energy} --- Elektrisch vermogen en energie}

\begin{solution}{exo:g9:electric-power-energy:1}
\omterm{def:g9:electric-power-energy:power}{Vermogen} is het tempo van energieomzetting ---
\omterm{def:g9:kinetic-energy-safety:joule}{joule} per \omterm{def:g2:measuring-time:second}{seconde}, in \omterm{def:g9:electric-power-energy:power}{watt}: de stroom uit de kraan,
niet de emmer. $P = U \times I$ en $E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
Lamp: $230 \times 0.022 \approx \qty{5}{W}$. Startmotor: $12 \times 100
= \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
Waterkoker: $2300 \div 230 = \qty{10}{A}$; magnetron: $700 \div 230
\approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
Omdat \omterm{def:g5:heat-and-insulation:heat}{warmte} de duurste vorm van \omterm{def:g5:everyday-energy:energy}{energie} is om te leveren: de
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur} van water of van een kamer verhogen verslindt
\omterm{def:g9:kinetic-energy-safety:joule}{joule} in een tempo dat lampen en elektronica nooit benaderen
--- dus staat op het plaatje van elke kachel een getal in kilowatt.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. Een maand in \omterm{def:g9:kinetic-energy-safety:joule}{joule} loopt op tot tien cijfers;
de makers van de elektriciteitsmeter kozen een \omterm{def:g6:measurement-in-science:measuring}{eenheid} die
huishoudmaanden in comfortabele honderdtallen houdt.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
Kachel: $2 \times 0.5 = \qty{1}{kWh}$. Televisie: $0.1 \times 5 =
\qty{0.5}{kWh}$. Het halve uur kachel geeft twee keer zoveel uit als de
hele televisieavond.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$; kosten $6 \times 0.25 =
1.5$ munteenheden --- een maand huiswerk voor de prijs van een
gebakje.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}
Elk \omterm{def:g9:electric-power-energy:kwh}{kilowattuur} dat het huis binnenkomt. De rekening trekt de
stand van vorige maand af van die van deze maand en vermenigvuldigt het
verschil met het tarief.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
Voorouders: $0.6 \times 4 \times 30 = \qty{72}{kWh}$. Modernen: $0.05
\times 4 \times 30 = \qty{6}{kWh}$. Besparing: \qty{66}{kWh} --- $16.5$
munteenheden per maand, na één middag op een trapladder.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}
Droger: $2000 \div 230 \approx \qty{8.7}{A}$. Met de \qty{10}{A} van de
waterkoker: $18.7$ --- voorbij de grenswaarde van de
\qty{16}{A}-groep: de automaat maakt (terecht) een eind aan het
experiment.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
Waterkoker: $60 \div 2.3 \approx 26$ uur. Lamp: $60 \div 0.005 =
12000$ uur --- een jaar en vier maanden. En toch rijdt diezelfde
voorraad de auto maar een paar honderd kilometer: een ton machine
verplaatsen verslindt \omterm{def:g5:everyday-energy:energy}{energie} op een schaal waarnaast
huisverlichting een afrondingsfout is.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
Voorspelling: $t = E/P = \num{5.0e5} \div 2300 \approx \qty{217}{s}$
--- ongeveer drieënhalve minuut. De echte waterkoker doet er langer
over: een deel van zijn \omterm{def:g9:kinetic-energy-safety:joule}{joule} lekt langs het water weg --- het
verwarmt het lichaam van de waterkoker zelf en de \omterm{def:g2:air-around-us:air}{lucht}, of
ontsnapt als vroege stoom. \omterm{def:g5:everyday-energy:energy}{Energie} wordt nooit vernietigd, maar
ze wordt evenmin volledig afgeleverd waar het de bedoeling was: het lek
in de keten, eindelijk geklokt in je eigen keuken.
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} Televisie: $0.1 \times 4 \times 30 = \qty{12}{kWh}$;
lampen: $0.05 \times 5 \times 30 = \qty{7.5}{kWh}$.
\textbf{3.} Wasmachine: $2 \times 1 \times 15 = \qty{30}{kWh}$;
boiler: $2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- tien verborgen
\omterm{def:g9:electric-power-energy:power}{watt}, een hele maand waterkoker.
\textbf{5.} Gemiddeld $120 \div 3 = \qty{40}{W}$; maand: $0.04 \times
720 = \qty{28.8}{kWh}$.
\textbf{6.} Totaal $\approx \qty{242}{kWh}$. Overheersend: de boiler
(\qty{150}{}), de wasmachine (\qty{30}{}), de koelkast
(\qty{28.8}{kWh}) --- verwarmen en altijd-aan.
\textbf{7.} $242 \times 0.25 \approx 60.6$ eenheden; het aandeel
van de boiler: $150 \times 0.25 = 37.5$ --- ruim meer dan de helft van
de rekening.
\textbf{8.} $2150 + 242 = 2392$ eenheden.
\textbf{9.} Nieuwe boilerregel: $2.5 \times 1.5 \times 30 =
\qty{112.5}{kWh}$: een besparing van \qty{37.5}{kWh}, ongeveer $9.4$
munteenheden per maand.
\textbf{10.} Eén nietsdoende lader: $0.0002 \times 720 =
\qty{0.14}{kWh}$ --- ongeveer vier honderdsten van een
munteenheid: voorstel één bespaart er per lader tweehonderd keer
zoveel als dit. Doorlichtingen vinden hefbomen; leuzen vinden
laders.
\textbf{11.} De regel van de waterkoker halveert tot \qty{3.45}{kWh}:
een besparing van \qty{3.45}{kWh}, minder dan één
munteenheid --- de gewoonte van oma is juist, en bescheiden.
\textbf{12.} Bijvoorbeeld: ``De kilowatturen wonen waar water
heet wordt en waar machines nooit slapen. De stand-byregel onderwijst
de tijdfactor: piepkleine \omterm{def:g9:electric-power-energy:power}{vermogens} laten rekeningen groeien
wanneer $t$ gelijk is aan altijd. En de best renderende verandering is
korter douchen --- de boiler bezit de rekening, dus bezitten zijn
\omterm{ex:g2:measuring-time:clock}{uren} de besparing.''
\end{solution}
