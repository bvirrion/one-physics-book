\section*{Hoofdstuk \ref{ch:g9:weight-vs-mass} --- Gewicht tegenover massa: krachten meten}

\begin{solution}{exo:g9:weight-vs-mass:1}
De \omterm{def:g9:weight-vs-mass:newton}{newton} (\unit{N}), gemeten met de
\omterm{def:g9:weight-vs-mass:dynamometer}{krachtmeter}. Ankers: één \omterm{def:g9:weight-vs-mass:newton}{newton} --- het
\omterm{def:g4:weight-and-mass:weight}{gewicht} van een kleine appel; honderd --- een stevige handdruk
(of het \omterm{def:g4:weight-and-mass:weight}{gewicht} van een koffer van \qty{10}{kg}).
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:2}
$P = m \times g$: \omterm{def:g4:weight-and-mass:weight}{gewicht} in \omterm{def:g9:weight-vs-mass:newton}{newton},
\omterm{def:g3:measuring-mass:mass}{massa} in \omterm{def:g3:measuring-mass:units}{kilogram}, en $g$ --- de
\omterm{prop:g9:weight-vs-mass:formula}{zwaartekrachtsterkte} van de plaats --- het aantal
\omterm{def:g9:weight-vs-mass:newton}{newton} trek dat elke \omterm{def:g3:measuring-mass:units}{kilogram} daar ontvangt
(\qty{9.8}{N/kg} op Aarde).
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:3}
Hond: $25 \times 9.8 = \qty{245}{N}$. Brood: $0.7 \times 9.8 \approx
\qty{6.9}{N}$. Een lezer van \qty{55}{kg}: ongeveer \qty{539}{N}.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:4}
$m = 14.7 \div 9.8 = \qty{1.5}{kg}$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:5}
De helling is de plaatselijke $g$ --- \omterm{def:g9:weight-vs-mass:newton}{newton} per
\omterm{def:g3:measuring-mass:units}{kilogram}. Op de \omterm{def:g2:moon-in-the-sky:moon}{Maan} blijft de lijn recht door de
oorsprong lopen maar zakt ze naar een flauwe helling van $1.6$:
dezelfde evenredigheid, een zwakkere wereld.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:6}
De kolom met de \omterm{def:g3:measuring-mass:mass}{massa} verandert nooit --- \qty{10}{kg} stof reist
ongeschonden mee. De kolommen met $g$ en \omterm{def:g4:weight-and-mass:weight}{gewicht} veranderen met
de wereld: \omterm{def:g4:weight-and-mass:weight}{gewicht} is de plaatselijke tol, $m \times g$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:7}
De balans vergelijkt $m_1 g$ met $m_2 g$: $g$ vermenigvuldigt beide
schalen en valt weg --- \omterm{def:g3:measuring-mass:mass}{massa}, waarheidsgetrouw, overal. De
badkamerweegschaal meet $P = m g_{\text{hier}}$ maar deelt door
$g_{\text{Aarde}}$ om \omterm{def:g3:measuring-mass:units}{kilogram} af te drukken: op de
\omterm{def:g2:moon-in-the-sky:moon}{Maan} drukt ze $m \times 1.6/9.8$ af --- een zesde van de
waarheid.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:8}
\omterm{def:g4:weight-and-mass:weight}{Gewicht}: \qty{19.6}{N} omlaag --- een pijl van ongeveer
\qty{3.9}{cm} bij vijf \omterm{def:g9:weight-vs-mass:newton}{newton} per
\omterm{def:g2:measuring-length:units}{centimeter}, vanaf het boek naar beneden. Steun: een even lange
pijl van \qty{3.9}{cm} omhoog. Gelijke lengtes, tegengestelde
richtingen: begroting in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}, boek stil.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:9}
Het vliegtuig moet de \emph{stof} van de koffer dragen, optillen en
afremmen --- haar \omterm{def:g3:measuring-mass:mass}{massa} --- en \qty{23}{kg} stof is
\qty{23}{kg} op elke wereld. De maanbalie heeft een balans met gemerkte
\omterm{def:g3:measuring-mass:mass}{massa}'s nodig: de enige weegschaal die een verandering van $g$
overleeft.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:10}
$m = 111 \div 3.7 = \qty{30}{kg}$; op Aarde weegt hij $30 \times 9.8 =
\qty{294}{N}$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:11}
Hij leest te hoog: hangend draagt de veer ook haar eigen
\omterm{def:g4:weight-and-mass:weight}{gewicht} (en dat van haar haak), dat het platte nulpunt nooit
meerekende --- het instrument telt er een vast overschot bij. Zet hem
hangend op nul, en het overschot wordt afgetrokken voordat de last
arriveert.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:12}
Uitrusting: \omterm{def:g9:weight-vs-mass:dynamometer}{krachtmeter}, set gemerkte \omterm{def:g3:measuring-mass:mass}{massa}'s ($1$ tot
\qty{5}{kg}), haak. Hang elke \omterm{def:g3:measuring-mass:mass}{massa} op, noteer de paren ($m$,
$P$), en zet $P$ uit tegen $m$: de punten liggen op één lijn door de
oorsprong, en de helling --- stijging in \omterm{def:g9:weight-vs-mass:newton}{newton} gedeeld door
de horizontale afstand in \omterm{def:g3:measuring-mass:units}{kilogram} --- is $g \approx
\qty{9.8}{N/kg}$. Op de \omterm{def:g2:moon-in-the-sky:moon}{Maan}: hetzelfde protocol, dezelfde
rechtheid, een nieuwe helling van $1.6$ --- de methode
meet elke wereld waarop ze staat; alleen het getal is
plaatselijk.
\end{solution}

\begin{solution}{pb:g9:weight-vs-mass:1}
\textbf{1.} Aarde $70 \times 9.8 = \qty{686}{N}$; \omterm{def:g2:moon-in-the-sky:moon}{Maan}
\qty{112}{N}; Mars \qty{259}{N}; dek van Jupiter \qty{1736}{N}.
\textbf{2.} Correct op Aarde: \qty{70}{kg}. \omterm{def:g2:moon-in-the-sky:moon}{Maan}: $112 \div
9.8 \approx \qty{11.4}{kg}$; Mars: $259 \div 9.8 \approx
\qty{26.4}{kg}$; Jupiter: $1736 \div 9.8 \approx \qty{177}{kg}$ ---
vleierij en laster volgens formule.
\textbf{3.} \qty{70}{kg} in elke haven: de vergelijking van de balans
heft de plaatselijke $g$ op.
\textbf{4.} Alleen het dek van Jupiter: \qty{1736}{N} nadert de grens
maar blijft eronder --- geen enkele haven breekt de \qty{2000}{N} voor
deze reiziger (een zwaardere collega van \qty{90}{kg} zou hem daar wel
breken: $90 \times 24.8 = \qty{2232}{N}$).
\textbf{5.} Vertrek: $40 \times 9.8 = \qty{392}{N}$; aankomst: $40
\times 3.7 = \qty{148}{N}$.
\textbf{6.} $m = 200/g$: Aarde \qty{20.4}{kg}; \omterm{def:g2:moon-in-the-sky:moon}{Maan}
\qty{125}{kg}; Mars \qty{54}{kg}; Jupiter \qty{8.1}{kg} --- de vaste
\omterm{def:g9:weight-vs-mass:newton}{newton} van de bond kopen zeer verschillende \omterm{def:g3:measuring-mass:mass}{massa}'s.
\textbf{7.} $m = 150 \div 1.6 = \qty{93.75}{kg}$ --- ongeveer
\qty{94}{kg} krat voor de douane.
\textbf{8.} $m = 294 \div 9.8 = \qty{30}{kg}$; \omterm{def:g4:weight-and-mass:weight}{gewicht} op de
\omterm{def:g2:moon-in-the-sky:moon}{Maan} $30 \times 1.6 = \qty{48}{N}$.
\textbf{9.} Spieren en \omterm{def:g3:measuring-mass:mass}{massa} kwamen onveranderd thuis; alleen de
tegenstander was anders. Op de \omterm{def:g2:moon-in-the-sky:moon}{Maan} kostte elke
\omterm{def:g3:measuring-mass:units}{kilogram} \qty{1.6}{N} om op te tillen; thuis wordt weer
\qty{9.8}{N} in rekening gebracht. De winst was de korting van de
\omterm{def:g2:moon-in-the-sky:moon}{Maan}, niet de groei van het lichaam --- $P = m g$, met $m$
trouw en $g$ wispelturig.
\textbf{10.} $0.2 \times 24.8 \approx \qty{5}{N}$ --- een aambeeld is
het niet: op Aarde woog de garde \qty{2}{N}, dus kloppen op het dek is
tweeënhalf keer zo zwaar, vermoeiend voor een pols maar nauwelijks
gieterijwerk. Klacht: gedeeltelijk gegrond.
\textbf{11.} Een \omterm{def:g9:weight-vs-mass:newton}{newton} chocolade is $m = 1/g$: Aarde ongeveer
\qty{102}{g}; \omterm{def:g2:moon-in-the-sky:moon}{Maan} \qty{625}{g}; Jupiter \qty{40}{g}. Koop je
\omterm{def:g9:weight-vs-mass:newton}{newton} op de \omterm{def:g2:moon-in-the-sky:moon}{Maan}.
\textbf{12.} Bijvoorbeeld: ``\omterm{def:g3:measuring-mass:mass}{Massa} reist met je mee;
\omterm{def:g4:weight-and-mass:weight}{gewicht} is de tol die elke wereld erop heft --- pak
kilogrammen in, en laat de havens hun
\omterm{def:g9:weight-vs-mass:newton}{newton} in rekening brengen.''
\end{solution}
