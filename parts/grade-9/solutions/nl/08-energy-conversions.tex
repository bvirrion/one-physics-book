\section*{Hoofdstuk \ref{ch:g9:energy-conversions} --- Energievormen en omzettingen}

\begin{solution}{exo:g9:energy-conversions:1}
Kinetisch (de rollende bal), potentieel (de opgehesen emmer),
thermisch (het warme bad), chemisch (het ontbijt, de
\omterm{def:g3:first-electric-circuit:battery}{batterij}), elektrisch (de voeding uit de muur), straling
(zonneschijn door het raam), kernenergie (de kern van de centrale).
\end{solution}

\begin{solution}{exo:g9:energy-conversions:2}
Emmer: $2 \times 9.8 \times 10 = \qty{196}{J}$. Wandelaar: $60 \times
9.8 \times 500 = \qty{2.9e5}{J}$ --- bijna driehonderd kilojoule aan
hoogte.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:3}
In elk proces verandert \omterm{def:g5:everyday-energy:energy}{energie} van vorm en van eigenaar, maar
blijft het totaal exact behouden. De eeuwigdurende machines stierven
eraan: een machine die meer levert dan ze opslaat of ontvangt, zou
\omterm{def:g5:everyday-energy:energy}{energie} scheppen --- verboden zonder beroep.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:4}
Boven in de zwaai: alles potentieel, even stil. Onderaan: alles
kinetisch, het snelst. De drift: elke passage roomt een beetje af naar
\omterm{def:g5:heat-and-insulation:heat}{warmte} van \omterm{def:g2:air-around-us:air}{lucht} en ophangpunt, en de schommel sterft
uit --- rekeningen overgeboekt, totaal intact.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:5}
$v = \sqrt{2 \times 9.8 \times 5} \approx \qty{9.9}{m/s}$ --- ongeveer
\qty{36}{km/h}. De \omterm{def:g3:measuring-mass:mass}{massa} viel weg toen $m g h$ tegenover
$\frac12 m v^2$ werd gezet: zware en lichte duikers plonzen met
dezelfde \omterm{def:g7:motion-average-speed:formula}{snelheid}.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:6}
Nuttig gedeeld door totaal: $170 \div 200 = 0.85$ --- vijfentachtig
procent. De ontbrekende \qty{30}{J}: \omterm{def:g5:heat-and-insulation:heat}{warmte} in wikkelingen en lagers.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:7}
Vijf procent van haar toevoer vertrok als \omterm{def:g1:light-and-shadows:source}{licht};
vijfennegentig als \omterm{def:g5:heat-and-insulation:heat}{warmte}. Een apparaat dat op zijn uitgangen wordt
gesorteerd, was een kachel --- die toevallig gloeide.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:8}
Chemisch (het ontbijt) $\to$ spierarbeid $\to$
\omterm{prop:g9:kinetic-energy-safety:formula}{kinetische energie} van fietser en fiets $\to$ opgeslagen als
\omterm{def:g9:energy-conversions:forms}{potentiële energie}, $m g h$, op de top --- met overal thermische
lekken: warme spieren, warme \omterm{def:g2:air-around-us:air}{lucht} die in beroering is gebracht,
warme banden. Totaal: behouden.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:9}
Per ton: $1000 \times 9.8 \times 50 = \qty{4.9e5}{J}$; bij negentig
procent ongeveer \qty{4.4e5}{J} elektrisch. Voor \qty{4.4e6}{W}:
$\num{4.4e6} \div \num{4.4e5} = 10$ ton per \omterm{def:g2:measuring-time:second}{seconde} door de turbines.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:10}
Ervoor: $0.5 \times 9.8 \times 2 = \qty{9.8}{J}$. Na de stuit: $0.5
\times 9.8 \times 1.2 \approx \qty{5.9}{J}$. Het afroomsel: ongeveer
\qty{3.9}{J} naar \omterm{def:g5:heat-and-insulation:heat}{warmte} van bal en vloer (en een tik
\omterm{def:g3:sound-around-us:sound}{geluid} --- zelf ook al snel \omterm{def:g5:heat-and-insulation:heat}{warmte}). De doorlichting eindigt
wanneer de stuiten hoogte nul bereiken: elke \omterm{def:g9:kinetic-energy-safety:joule}{joule} van de
oorspronkelijke $9.8$ rust dan op de thermische rekening.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:11}
$E_p = 500 \times 9.8 \times 30 = \qty{1.47e5}{J}$; bij negentig
procent moet de elektriciteitsmeter $\num{1.47e5} \div 0.9 \approx \qty{1.6e5}{J}$
leveren. Het extra tiende deel is niet aan het heelal verloren --- het
verwarmt \omterm{ex:g6:circuits-and-safety:motor}{motor}, tandwielen en aandrijfas: het behoud houdt stand; het
grootboek van de lift bevat eenvoudigweg een thermische regel.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:12}
\omterm{def:g7:motion-average-speed:formula}{Snelheid} onderaan: $v = \sqrt{2 \times 9.8 \times 4} \approx
\qty{8.9}{m/s}$. Ideaal wordt de rechterrand op precies \qty{4}{m}
teruggewonnen; werkelijk een beetje lager, en bij elke passage nog
lager, tot de skater onderaan tot stilstand wiegt --- de hele $m g h$
met pensioen gestuurd naar \omterm{def:g5:heat-and-insulation:heat}{warmte} van wielen, \omterm{def:g6:describing-motion:trajectory}{baan} en
\omterm{def:g2:air-around-us:air}{lucht}. Er is niets vernietigd, maar alles wat bruikbaar was is
uitgegeven: het grootboek van de wereld won uitgesmeerde \omterm{def:g5:heat-and-insulation:heat}{warmte}, en de
skater verloor de geconcentreerde, herbruikbare hoogte --- de
hoeveelheid overleeft; de bruikbaarheid komt niet terug.
\end{solution}

\begin{solution}{pb:g9:energy-conversions:1}
\textbf{1.} $E_p = 2000 \times 9.8 \times 45 = \qty{8.8e5}{J}$.
\textbf{2.} $v = \sqrt{2 \times 9.8 \times 45} \approx \qty{29.7}{m/s}
\approx \qty{107}{km/h}$.
\textbf{3.} Een top van \qty{50}{m} eist meer \omterm{def:g5:everyday-energy:energy}{energie} dan de
heuvel schonk: het grootboek verbiedt het. Vanuit rust op \qty{45}{m}
mag geen latere heuvel de \qty{45}{m} overschrijden --- en in
werkelijkheid eist de wrijving zichtbaar minder.
\textbf{4.} Gemeten: $0.5 \times 2000 \times 28^2 = \qty{7.8e5}{J}$
tegenover de ideale \qty{8.8e5}{J}: ongeveer \qty{1e5}{J} geboekt op de
thermische rekening --- rails, wielen en \omterm{def:g2:air-around-us:air}{lucht}, opgewarmd.
\textbf{5.} $P = \num{8.8e5} \div 60 \approx \qty{1.5e4}{W}$ --- zo'n
vijftien kilowatt.
\textbf{6.} $14.7 \div 20 \approx 0.74$: vierenzeventig procent.
\textbf{7.} Tien uur bij één liftbeweging per vier minuten: $150$
liftbewegingen; elk kost $20 \times \tfrac{1}{60} = \qty{0.33}{kWh}$;
per dag $150 \times 0.33 = \qty{50}{kWh}$ --- $12.5$
munteenheden.
\textbf{8.} Een royale remgenerator zou een behoorlijk deel van de
\omterm{prop:g9:kinetic-energy-safety:formula}{kinetische energie} bij aankomst kunnen terugwinnen --- maar elke
trap (rails, rem, generator, transformator) betaalt haar thermische
commissie, en de teruggewonnen \omterm{def:g9:kinetic-energy-safety:joule}{joule} kunnen nooit gelijk zijn
aan de uitgaven van de lier: het behoud staat hergebruik toe; een
\omterm{def:g9:energy-conversions:efficiency}{rendement} kleiner dan één verbiedt de volledige cirkel. De rit kan
zichzelf verlichten, nooit zichzelf betalen.
\textbf{9.} $v = \sqrt{2 \times 9.8 \times 50} \approx \qty{31.3}{m/s}$.
\textbf{10.} Kinetisch bij binnenkomst $0.5 \times 500 \times 980 =
\qty{2.45e5}{J}$, plus de laatste tien elektriciteitsmeter $500 \times 9.8
\times 10 = \qty{4.9e4}{J}$: ongeveer \qty{2.9e5}{J} voor de remmen ---
door geïnduceerde wervelstromen gestuurd naar \omterm{def:g5:heat-and-insulation:heat}{warmte} van de
remvinnen: inductie als pensioenregeling.
\textbf{11.} De bewering houdt stand, gevoeld al is ze niet precies
geformuleerd: gondel en passagiers vielen samen, en samen vallende
metgezellen drukken nergens op --- de schreeuw is gedeelde vrije val,
met de zwaartekracht volledig aanwezig en toch onvoelbaar.
\textbf{12.} Bijvoorbeeld: ``Onze achtbanen leven van één ruil: hoogte
voor \omterm{def:g7:motion-average-speed:formula}{snelheid} en terug, $m g h$ tegen $\frac12 m v^2$. Elke
attractie gehoorzaamde één wet: het totaal aan \omterm{def:g5:everyday-energy:energy}{energie}, op elk
punt doorgelicht, bewoog nooit. En één rekening belastte ze allemaal:
\omterm{def:g5:heat-and-insulation:heat}{warmte} --- de stille commissie van de wrijving, verschuldigd bij elke
rit.''
\end{solution}
