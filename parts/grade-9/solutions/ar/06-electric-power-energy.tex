\section*{الفصل \ref{ch:g9:electric-power-energy} --- القدرة والطاقة الكهربائيتان}

\begin{solution}{exo:g9:electric-power-energy:1}
\omterm{def:g9:electric-power-energy:power}{القدرة} وتيرة تحويل \omterm{def:g5:everyday-energy:energy}{الطاقة} --- جولات في الثانية، \omterm{def:g9:electric-power-energy:power}{بالواط}: جريان
الصنبور لا الدلو. و$P = U \times I$ و$E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
\omterm{def:g3:first-electric-circuit:bulb}{المصباح}: $230 \times 0.022 \approx \qty{5}{W}$. ومحرّك
الإقلاع: $12 \times 100 = \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
الغلّاية: $2300 \div 230 = \qty{10}{A}$؛ والميكروويف:
$700 \div 230 \approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
لأن الدفء أغلى أشكال \omterm{def:g5:everyday-energy:energy}{الطاقة} تسليمًا: فرفع \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} ماء أو
غرفة يلتهم الجولات بوتيرة لا تقاربها المصابيح ولا
الإلكترونيات --- فتقرأ لوحة كل مسخّن بالكيلوواطات.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. وشهر من الجولات يبلغ عشرة أرقام؛ فاختار
صانعو العدّادات \omterm{def:g6:measurement-in-science:measuring}{وحدة} تُبقي شهور البيوت في مئات مريحة.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
المسخّن: $2 \times 0.5 = \qty{1}{kWh}$. والتلفاز:
$0.1 \times 5 = \qty{0.5}{kWh}$. فمسخّن نصف الساعة ينفق ضعف
ما تنفقه أمسية التلفاز كلها.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$؛ والكلفة
$6 \times 0.25 = 1.5$ وحدة نقدية --- شهر من الواجبات بثمن
قطعة حلوى.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}
كل \omterm{def:g9:electric-power-energy:kwh}{كيلوواط ساعة} يدخل البيت. والفاتورة تطرح قراءة الشهر
الماضي من قراءة هذا الشهر وتضرب الفرق في التعرفة.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
الأسلاف: $0.6 \times 4 \times 30 = \qty{72}{kWh}$. والمحدثون:
$0.05 \times 4 \times 30 = \qty{6}{kWh}$. والتوفير:
\qty{66}{kWh} --- أي $16.5$ وحدة نقدية شهريًّا، من أصيل واحد
على سلّم.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}
المجفّف: $2000 \div 230 \approx \qty{8.7}{A}$. ومع
\qty{10}{A} للغلّاية: $18.7$ --- أي فوق تقنين \omterm{def:g5:series-parallel-circuits:parallel}{الفرع}
\qty{16}{A}: فسينهي القاطع التجربة (وهو محقّ).
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
الغلّاية: $60 \div 2.3 \approx 26$ ساعة. \omterm{def:g3:first-electric-circuit:bulb}{والمصباح}:
$60 \div 0.005 = 12000$ ساعة --- أي سنة وأربعة أشهر. ومع ذلك
لا يقود المخزون نفسه السيارة إلا بضع مئات من الكيلومترات:
فتحريك طنّ من الآلة يلتهم \omterm{def:g5:everyday-energy:energy}{طاقة} تصير بجانبها إضاءة البيت خطأ
تقريب.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
التنبّؤ: $t = E/P = \num{5.0e5} \div 2300 \approx
\qty{217}{s}$ --- أي نحو ثلاث دقائق ونصف. والغلّاية الحقيقية
تعمل أطول: فجزء من جولاتها يتسرّب متجاوزًا الماء --- يدفئ جسم
الغلّاية نفسه \omterm{def:g2:air-around-us:air}{والهواء}، ويهرب بخارًا مبكّرًا. \omterm{def:g5:everyday-energy:energy}{فالطاقة} لا تُفنى
أبدًا، لكنها لا تُسلَّم كلها حيث يُراد أيضًا: تسرّب السلسلة،
موقَّتًا أخيرًا في مطبخك أنت.
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} التلفاز: $0.1 \times 4 \times 30 =
\qty{12}{kWh}$؛ والمصابيح: $0.05 \times 5 \times 30 =
\qty{7.5}{kWh}$.
\textbf{3.} الغسّالة: $2 \times 1 \times 15 = \qty{30}{kWh}$؛
والسخّان: $2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- عشرة
واطات مخفيّة، أي شهر غلّاية كامل.
\textbf{5.} المتوسّط $120 \div 3 = \qty{40}{W}$؛ والشهر:
$0.04 \times 720 = \qty{28.8}{kWh}$.
\textbf{6.} المجموع $\approx \qty{242}{kWh}$. والمهيمنون:
سخّان الماء (\qty{150}{})، والغسّالة (\qty{30}{})، والثلّاجة
(\qty{28.8}{kWh}) --- التسخين وما يعمل دائمًا.
\textbf{7.} $242 \times 0.25 \approx 60.6$ \omterm{def:g6:measurement-in-science:measuring}{وحدة}؛ وحصة
السخّان: $150 \times 0.25 = 37.5$ --- أي أكثر من نصف
الفاتورة بكثير.
\textbf{8.} $2150 + 242 = 2392$ \omterm{def:g6:measurement-in-science:measuring}{وحدة}.
\textbf{9.} سطر السخّان الجديد: $2.5 \times 1.5 \times 30 =
\qty{112.5}{kWh}$: فالتوفير \qty{37.5}{kWh}، أي نحو $9.4$
وحدة نقدية شهريًّا.
\textbf{10.} شاحن عاطل واحد: $0.0002 \times 720 =
\qty{0.14}{kWh}$ --- أي نحو أربعة أجزاء من مئة من \omterm{def:g6:measurement-in-science:measuring}{الوحدة}
النقدية: فالاقتراح الأول يوفّر مئتَي ضعف ذلك لكل شاحن.
فالتدقيقات تجد الروافع؛ والشعارات تجد الشواحن.
\textbf{11.} سطر الغلّاية ينتصف إلى \qty{3.45}{kWh}:
فالتوفير \qty{3.45}{kWh}، أي دون وحدة نقدية واحدة ---
فعادة الجدّة صائبة، ومتواضعة.
\textbf{12.} مثلًا: ``\omterm{def:g9:electric-power-energy:kwh}{الكيلوواط ساعة} تسكن حيث يسخن الماء
وحيث لا تنام الآلات. وسطر الاستعداد يعلّم عامل الزمن: فالقدرات
الضئيلة تنفخ الفواتير حين يكون $t$ هو الدوام. وأفضل تغيير
عائدًا هو الاستحمام الأقصر --- فالسخّان يملك الفاتورة، ومن ثمّ
تملك ساعاته التوفير.''
\end{solution}
