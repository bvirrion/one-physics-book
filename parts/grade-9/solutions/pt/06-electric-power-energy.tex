\section*{Capítulo \ref{ch:g9:electric-power-energy} --- A potência e a energia elétricas}

\begin{solution}{exo:g9:electric-power-energy:1}
A \omterm{def:g9:electric-power-energy:power}{potência} é o ritmo da conversão de \omterm{def:g5:everyday-energy:energy}{energia} --- \omterm{def:g9:kinetic-energy-safety:joule}{joules} por segundo,
em \omterm{def:g9:electric-power-energy:power}{watts}: a vazão da torneira, não o balde. $P = U \times I$ e
$E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
Lâmpada: $230 \times 0.022 \approx \qty{5}{W}$. \omterm{ex:g6:circuits-and-safety:motor}{Motor} de partida:
$12 \times 100 = \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
Chaleira: $2300 \div 230 = \qty{10}{A}$; micro-ondas:
$700 \div 230 \approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
Porque o \omterm{def:g5:heat-and-insulation:heat}{calor} é a forma de \omterm{def:g5:everyday-energy:energy}{energia} mais cara de fornecer: elevar a
\omterm{def:g3:temperature-thermometers:temperature}{temperatura} da água ou de um cômodo devora \omterm{def:g9:kinetic-energy-safety:joule}{joules} num ritmo que as
lâmpadas e a eletrônica nunca chegam perto --- e por isso a placa de
todo aquecedor vem em quilowatts.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. Um mês de \omterm{def:g9:kinetic-energy-safety:joule}{joules} chega a dez algarismos; os
fabricantes do medidor escolheram uma \omterm{def:g6:measurement-in-science:measuring}{unidade} que mantém os meses
domésticos em centenas confortáveis.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
Aquecedor: $2 \times 0.5 = \qty{1}{kWh}$. Televisão:
$0.1 \times 5 = \qty{0.5}{kWh}$. O aquecedor de meia hora gasta mais
que a noite inteira de televisão, o dobro.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$; custo
$6 \times 0.25 = 1.5$ \omterm{def:g6:measurement-in-science:measuring}{unidade} monetária --- um mês de lição de casa
pelo preço de um salgado.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}
Cada \omterm{def:g9:electric-power-energy:kwh}{quilowatt-hora} que entra na casa. A conta subtrai a leitura do
mês passado da deste mês e multiplica a diferença pela tarifa.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
Antepassadas: $0.6 \times 4 \times 30 = \qty{72}{kWh}$. Modernas:
$0.05 \times 4 \times 30 = \qty{6}{kWh}$. Economia: \qty{66}{kWh} ---
$16.5$ \omterm{def:g6:measurement-in-science:measuring}{unidades} monetárias por mês, a partir de uma tarde em cima de
uma escadinha.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}
Secador: $2000 \div 230 \approx \qty{8.7}{A}$. Com os \qty{10}{A} da
chaleira: $18.7$ --- passando da especificação do \omterm{def:g5:series-parallel-circuits:parallel}{ramo} de
\qty{16}{A}: o disjuntor vai (com razão) encerrar a experiência.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
Chaleira: $60 \div 2.3 \approx 26$ \omterm{ex:g2:measuring-time:clock}{horas}. Lâmpada:
$60 \div 0.005 = 12000$ \omterm{ex:g2:measuring-time:clock}{horas} --- um ano e quatro meses. E mesmo
assim a mesma reserva leva o carro só umas poucas centenas de
quilômetros: mover uma tonelada de máquina devora \omterm{def:g5:everyday-energy:energy}{energia} numa escala
diante da qual a iluminação doméstica é um erro de arredondamento.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
Previsão: $t = E/P = \num{5.0e5} \div 2300 \approx \qty{217}{s}$ ---
cerca de três minutos e meio. A chaleira de verdade demora mais:
parte dos \omterm{def:g9:kinetic-energy-safety:joule}{joules} dela vaza para fora da água --- aquecendo o próprio
corpo da chaleira e o \omterm{def:g2:air-around-us:air}{ar}, escapando como vapor precoce. A \omterm{def:g5:everyday-energy:energy}{energia}
nunca é destruída, mas também não é toda entregue onde se pretendia:
o vazamento da corrente, cronometrado enfim na sua própria cozinha.
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} Televisão: $0.1 \times 4 \times 30 = \qty{12}{kWh}$;
lâmpadas: $0.05 \times 5 \times 30 = \qty{7.5}{kWh}$.
\textbf{3.} Máquina de lavar: $2 \times 1 \times 15 = \qty{30}{kWh}$;
aquecedor: $2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- dez \omterm{def:g9:electric-power-energy:power}{watts}
escondidos, um mês inteiro de chaleira.
\textbf{5.} Média $120 \div 3 = \qty{40}{W}$; mês:
$0.04 \times 720 = \qty{28.8}{kWh}$.
\textbf{6.} Total $\approx \qty{242}{kWh}$. Dominantes: o aquecedor
de água (\qty{150}{}), a máquina de lavar (\qty{30}{}) e a geladeira
(\qty{28.8}{kWh}) --- aquecimento e sempre ligado.
\textbf{7.} $242 \times 0.25 \approx 60.6$ \omterm{def:g6:measurement-in-science:measuring}{unidades}; a parcela do
aquecedor: $150 \times 0.25 = 37.5$ --- bem mais da metade da conta.
\textbf{8.} $2150 + 242 = 2392$ \omterm{def:g6:measurement-in-science:measuring}{unidades}.
\textbf{9.} Nova linha do aquecedor:
$2.5 \times 1.5 \times 30 = \qty{112.5}{kWh}$: economia de
\qty{37.5}{kWh}, cerca de $9.4$ \omterm{def:g6:measurement-in-science:measuring}{unidades} monetárias por mês.
\textbf{10.} Um carregador ocioso: $0.0002 \times 720 =
\qty{0.14}{kWh}$ --- cerca de quatro centésimos de \omterm{def:g6:measurement-in-science:measuring}{unidade} monetária:
a proposta um economiza duzentas vezes mais por carregador. As
auditorias acham \omterm{def:g3:levers-and-scales:lever}{alavancas}; os slogans acham carregadores.
\textbf{11.} A linha da chaleira cai à metade, \qty{3.45}{kWh}:
economia de \qty{3.45}{kWh}, menos de uma \omterm{def:g6:measurement-in-science:measuring}{unidade} monetária --- o
hábito da avó está certo, e é modesto.
\textbf{12.} Por exemplo: ``Os \omterm{def:g9:electric-power-energy:kwh}{quilowatt-horas} moram onde a água
esquenta e onde as máquinas nunca dormem. A linha do modo de espera
ensina o fator tempo: \omterm{def:g9:electric-power-energy:power}{potências} minúsculas fazem contas crescerem
quando $t$ é sempre. E a mudança que mais rende é o banho mais curto
--- o aquecedor é dono da conta, então as \omterm{ex:g2:measuring-time:clock}{horas} dele são donas da
economia.''
\end{solution}
