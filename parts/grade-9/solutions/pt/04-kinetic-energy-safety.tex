\section*{Capítulo \ref{ch:g9:kinetic-energy-safety} --- A energia cinética e a segurança no trânsito}

\begin{solution}{exo:g9:kinetic-energy-safety:1}
O \omterm{def:g9:kinetic-energy-safety:joule}{joule} (\unit{J}): uns poucos \omterm{def:g9:kinetic-energy-safety:joule}{joules} levantam este livro até a
prateleira dele; o coração gasta cerca de um por batida.
$E_k = \frac12 m v^2$ --- $m$ em \omterm{def:g3:measuring-mass:units}{quilogramas}, $v$ em \omterm{def:g2:measuring-length:units}{metros} por
segundo, $E_k$ em \omterm{def:g9:kinetic-energy-safety:joule}{joules}.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:2}
Lebre: $0.5 \times 2 \times 100 = \qty{100}{J}$. Jogador:
$0.5 \times 80 \times 25 = \qty{1000}{J}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:3}
Metade a mais de \omterm{def:g3:measuring-mass:mass}{massa}: fator $1.5$. Metade a mais de \omterm{def:g7:motion-average-speed:formula}{velocidade}:
fator $1.5^2 = 2.25$. O botão da \omterm{def:g7:motion-average-speed:formula}{velocidade} vence --- ele sempre
vence, vestindo o quadrado.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:4}
A \omterm{prop:g9:kinetic-energy-safety:stopping}{distância de reação} --- a viagem de um segundo, crescendo em
proporção a $v$; a \omterm{prop:g9:kinetic-energy-safety:stopping}{distância de frenagem} --- a aposentadoria de
$E_k$, crescendo com $v^2$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:5}
A \qty{50}{km/h}: $13$ de \qty{27}{m} --- cerca de metade. A
\qty{130}{km/h}: $85$ de \qty{121}{m} --- uns setenta por cento. O
componente ao quadrado cresce mais que o proporcional: em \omterm{def:g7:motion-average-speed:formula}{velocidade},
a frenagem devora o total.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:6}
Vira \omterm{def:g5:heat-and-insulation:heat}{calor} nos discos de freio --- o brilho âmbar depois das descidas
de montanha, o cheiro de freio quente. A \omterm{def:g5:everyday-energy:energy}{energia} é repassada, nunca
apagada: a proposição da corrente do capítulo de \omterm{def:g5:everyday-energy:energy}{energia} da infância,
ainda no comando.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:7}
$36 \div 3.6 = \qty{10}{m}$; depois \qty{20}{m}; depois \qty{30}{m}
--- marchando em proporção à \omterm{def:g7:motion-average-speed:formula}{velocidade}, como um trecho de um segundo
tem de fazer.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:8}
A sua $E_k$ é fixada pela sua \omterm{def:g3:measuring-mass:mass}{massa} e pela \omterm{def:g7:motion-average-speed:formula}{velocidade} do carro ---
cinta nenhuma a muda. O cinto civiliza a \emph{aposentadoria} dela:
para o corpo ao longo de uma distância e de um tempo maiores, com
delicadeza, em vez de contra o painel, tudo de uma vez.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:9}
A frenagem escala com o quadrado da \omterm{def:g7:motion-average-speed:formula}{velocidade}: a \qty{60}{km/h} (o
dobro), $8 \times 4 = \qty{32}{m}$; a \qty{90}{km/h} (o triplo),
$8 \times 9 = \qty{72}{m}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:10}
A \qty{30}{km/h}: reação $\approx \qty{8.3}{m}$, frenagem
$13 \times (30/50)^2 \approx \qty{4.7}{m}$ --- parada em cerca de
\qty{13}{m}, contra \qty{27}{m} a cinquenta: menos da metade da
estrada. \omterm{def:g5:everyday-energy:energy}{Energias}: $(30/50)^2 = 0.36$ --- mal um terço da \omterm{def:g5:everyday-energy:energy}{energia} de
colisão. As duas persuadem; a frase que costuma ir para o cartaz é a
da estrada --- ``a 30 você para antes da criança; a 50 você chega até
ela ainda em movimento''.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:11}
Dobre só a parcela da frenagem --- a chuva alonga a aposentadoria,
não o cérebro: a \qty{50}{km/h}, $14 + 26 = \qty{40}{m}$; a
\qty{90}{km/h}, $25 + 80 = \qty{105}{m}$. O segundo de reação é igual
no seco e no molhado.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:12}
Caminhão: $v = 25$ \unit{m/s};
$E_k = 0.5 \times 20000 \times 625 = \qty{6.25e6}{J}$. Carro
equivalente: $0.5 \times 1000 \times v^2 = \num{6.25e6}$ dá
$v^2 = 12500$, $v \approx \qty{112}{m/s} \approx \qty{400}{km/h}$ ---
carro de rua nenhum chega perto. Um caminhão na \omterm{def:g7:motion-average-speed:formula}{velocidade} da estrada
carrega \omterm{def:g5:everyday-energy:energy}{energias} de carro de corrida com freios de caminhão: quando
eles superaquecem e falham numa descida longa, só uma rampa de brita
consegue aposentar megajoules com segurança. Os carros nunca precisam
de uma; as \omterm{def:g5:everyday-energy:energy}{energias} deles morrem em freios comuns.
\end{solution}

\begin{solution}{pb:g9:kinetic-energy-safety:1}
\textbf{1.} $50 \div 3.6 = \qty{13.9}{m/s}$;
$60 \div 3.6 = \qty{16.7}{m/s}$.
\textbf{2.} \qty{13.9}{m} e \qty{16.7}{m}.
\textbf{3.} \qty{13}{m}; a sessenta,
$13 \times (60/50)^2 = 13 \times 1.44 \approx \qty{18.7}{m}$.
\textbf{4.} Parada: $26.9$ contra \qty{35.4}{m} --- a lacuna do
cartaz: cerca de \emph{oito \omterm{def:g2:measuring-length:units}{metros} e meio} a mais, dois comprimentos
de carro além de onde o motorista dos cinquenta já parou.
\textbf{5.} A cinquenta: cerca de \qty{97}{kJ}. A noventa
($v = \qty{25}{m/s}$): $0.5 \times 1000 \times 625 = \qty{312}{kJ}$.
\textbf{6.} $97 \div 9.8 \approx \qty{10}{m}$;
$312 \div 9.8 \approx \qty{32}{m}$.
\textbf{7.} Dez \omterm{def:g2:measuring-length:units}{metros}: um prédio de três andares; trinta e dois: um
bloco de dez andares. Rascunho: ``Uma batida a 50 é uma queda do
terceiro andar. A 90, do décimo. Escolha a sua janela.''
\textbf{8.} As alturas medem honestamente a \omterm{def:g5:everyday-energy:energy}{energia} que precisa ser
aposentada --- esse número engenharia nenhuma muda. As zonas de
deformação e os cintos mudam a \emph{maneira} da aposentadoria:
\omterm{def:g2:measuring-length:units}{metros} de parada suave em vez de \omterm{def:g2:measuring-length:units}{centímetros} de parada cruel --- e é
por isso que a analogia da queda exagera a lesão num carro moderno, e
por que a \omterm{def:g5:everyday-energy:energy}{energia} que ela retrata é real do mesmo jeito.
\textbf{9.} \qty{25}{m} às cegas por segundo; uma olhadinha de dois
segundos, \qty{50}{m} --- meio campo antes mesmo de o segundo de
reação começar.
\textbf{10.} $65 + 25 = \qty{90}{m}$ --- o motorista cansado
acrescenta um ônibus e meio ao total do motorista atento.
\textbf{11.} Mais errado nas \omterm{def:g7:motion-average-speed:formula}{velocidades} \emph{baixas}: ali a parcela
da frenagem ao quadrado é pequena e o trecho de reação é a maior
parte da distância de parada --- na cidade, o segundo \emph{é} o
perigo. (Em alta \omterm{def:g7:motion-average-speed:formula}{velocidade} a frenagem domina --- mas \qty{25}{m} por
segundo também não são ninharia lá.)
\textbf{12.} Por exemplo: ``Dez a mais de \omterm{def:g7:motion-average-speed:formula}{velocidade}, dez \omterm{def:g2:measuring-length:units}{metros} a
mais de parada --- a \omterm{def:g7:motion-average-speed:formula}{velocidade} entra duas vezes, uma delas ao
quadrado. Uma batida é uma queda: a \omterm{def:g5:everyday-energy:energy}{energia} cresce com o quadrado da
\omterm{def:g7:motion-average-speed:formula}{velocidade}. Um segundo de olhadinha são vinte e cinco \omterm{def:g2:measuring-length:units}{metros} de
cegueira --- a atenção é a mais curta das \omterm{prop:g9:kinetic-energy-safety:stopping}{distâncias de frenagem}.''
\end{solution}
