\section*{अध्याय \ref{ch:g9:weight-vs-mass} --- भार बनाम द्रव्यमान: बलों का मापन}

\begin{solution}{exo:g9:weight-vs-mass:1}
\omterm{def:g9:weight-vs-mass:newton}{न्यूटन} (\unit{N}), जिसे \omterm{def:g9:weight-vs-mass:dynamometer}{बलमापी} से मापा जाता है। लंगर: एक \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} ---
किसी छोटे सेब का \omterm{def:g4:weight-and-mass:weight}{भार}; और सौ --- एक मज़बूत हाथ मिलाना (या \qty{10}{kg}
के सूटकेस का \omterm{def:g4:weight-and-mass:weight}{भार})।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:2}
$P = m \times g$: \omterm{def:g4:weight-and-mass:weight}{भार} \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} में, \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \omterm{def:g3:measuring-mass:units}{किलोग्राम} में, और $g$ ---
यानी उस जगह की \omterm{prop:g9:weight-vs-mass:formula}{गुरुत्वीय प्रबलता} --- वहाँ हर \omterm{def:g3:measuring-mass:units}{किलोग्राम} को मिलने वाले
खिंचाव के \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} (पृथ्वी पर \qty{9.8}{N/kg})।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:3}
कुत्ता: $25 \times 9.8 = \qty{245}{N}$. रोटी: $0.7 \times 9.8
\approx \qty{6.9}{N}$. \qty{55}{kg} का पाठक: क़रीब \qty{539}{N}।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:4}
$m = 14.7 \div 9.8 = \qty{1.5}{kg}$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:5}
ढाल ही स्थानीय $g$ है --- यानी \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} प्रति \omterm{def:g3:measuring-mass:units}{किलोग्राम}। \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर वह
रेखा मूल बिंदु से गुज़रती सीधी ही रहती, पर ढलकर $1.6$ की नरम ढाल पर आ
जाती: वही समानुपात, कमज़ोर दुनिया।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:6}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} वाला स्तंभ कभी नहीं बदलता --- \qty{10}{kg} सामान सही-सलामत
सफ़र करता है। $g$ और \omterm{def:g4:weight-and-mass:weight}{भार} वाले स्तंभ दुनिया के साथ बदलते हैं: \omterm{def:g4:weight-and-mass:weight}{भार} तो
स्थानीय चुंगी है, $m \times g$.
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:7}
\omterm{def:g3:levers-and-scales:scale}{तुला} $m_1 g$ की तुलना $m_2 g$ से करती है: $g$ दोनों पलड़ों को गुणा
करके कट जाती है --- \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, सच्चाई से, कहीं भी। जबकि स्नानघर
वाली तराज़ू मापती तो $P = m g_{\text{यहाँ}}$ है, पर \omterm{def:g3:measuring-mass:units}{किलोग्राम} छापने के
लिए $g_{\text{पृथ्वी}}$ से भाग देती है: इसलिए \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर वह
$m \times 1.6/9.8$ छाप देती है --- यानी सच का छठा हिस्सा।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:8}
\omterm{def:g4:weight-and-mass:weight}{भार}: नीचे की ओर \qty{19.6}{N} --- पाँच \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} प्रति \omterm{def:g2:measuring-length:units}{सेंटीमीटर} पर
क़रीब \qty{3.9}{cm} लंबा तीर, किताब से नीचे की ओर। सहारा: उतना ही
\qty{3.9}{cm} का तीर, ऊपर की ओर। बराबर लंबाइयाँ, उलटी दिशाएँ:
बहीखाता बराबर, और किताब ठहरी हुई।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:9}
हवाई जहाज़ को थैले का \emph{सामान} --- यानी उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} --- ढोना,
उठाना और रोकना पड़ता है; और \qty{23}{kg} सामान हर दुनिया पर
\qty{23}{kg} ही रहता है। \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} की मेज़ को अंकित द्रव्यमानों वाली \omterm{def:g3:levers-and-scales:scale}{तुला}
चाहिए: वही अकेली तराज़ू है जो $g$ के बदलने से बच निकलती है।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:10}
$m = 111 \div 3.7 = \qty{30}{kg}$; और पृथ्वी पर उसका \omterm{def:g4:weight-and-mass:weight}{भार}
$30 \times 9.8 = \qty{294}{N}$ होगा।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:11}
वह ज़्यादा पढ़ता है: लटकी हालत में कमानी अपना (और अपने हुक का) \omterm{def:g4:weight-and-mass:weight}{भार} भी
ढोती है, जिसे सपाट वाले शून्य ने कभी गिना ही नहीं --- यंत्र एक
अटल अधिशेष जोड़ देता है। लटकाकर शून्य करो, और वह अधिशेष बोझ आने से
पहले ही घटा दिया जाता है।
\end{solution}

\begin{solution}{exo:g9:weight-vs-mass:12}
सामान: \omterm{def:g9:weight-vs-mass:dynamometer}{बलमापी}, अंकित द्रव्यमानों का सेट ($1$ से \qty{5}{kg} तक), और
एक हुक। हर \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} लटकाओ, ($m$, $P$) जोड़ियाँ दर्ज करो, और $P$ को
$m$ के सामने आलेखित करो: बिंदु मूल बिंदु से गुज़रती एक सीध में बैठ जाते
हैं, और ढाल --- यानी न्यूटनों में चढ़ाई बटा किलोग्रामों में लंबाई ---
$g \approx \qty{9.8}{N/kg}$ है। \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर: वही तरीक़ा, वही सीधापन, और
नई ढाल $1.6$ --- यह विधि जिस भी दुनिया पर खड़ी हो उसी को माप लेती है;
सिर्फ़ अंक स्थानीय होता है।
\end{solution}

\begin{solution}{pb:g9:weight-vs-mass:1}
\textbf{1.} पृथ्वी $70 \times 9.8 = \qty{686}{N}$; \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}
\qty{112}{N}; मंगल \qty{259}{N}; बृहस्पति का पटल \qty{1736}{N}।
\textbf{2.} पृथ्वी पर सही: \qty{70}{kg}. \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}: $112 \div 9.8
\approx \qty{11.4}{kg}$; मंगल: $259 \div 9.8 \approx \qty{26.4}{kg}$;
बृहस्पति: $1736 \div 9.8 \approx \qty{177}{kg}$ --- सूत्र से की
गई चापलूसी और बदनामी।
\textbf{3.} हर बंदरगाह पर \qty{70}{kg}: \omterm{def:g3:levers-and-scales:scale}{तुला} की तुलना स्थानीय $g$ को
काट देती है।
\textbf{4.} सिर्फ़ बृहस्पति का पटल: \qty{1736}{N} सीमा के पास पहुँचता
है पर उससे नीचे ही रहता है --- यानी इस यात्री के लिए कोई भी बंदरगाह
\qty{2000}{N} नहीं तोड़ता (जबकि \qty{90}{kg} का कोई भारी साथी वहाँ
तोड़ देता: $90 \times 24.8 = \qty{2232}{N}$).
\textbf{5.} रवानगी: $40 \times 9.8 = \qty{392}{N}$; पहुँचने पर:
$40 \times 3.7 = \qty{148}{N}$.
\textbf{6.} $m = 200/g$: पृथ्वी \qty{20.4}{kg}; \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}
\qty{125}{kg}; मंगल \qty{54}{kg}; बृहस्पति \qty{8.1}{kg} --- संघ के
तय \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} बहुत अलग-अलग \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} ख़रीदते हैं।
\textbf{7.} $m = 150 \div 1.6 = \qty{93.75}{kg}$ ---
सीमा-शुल्क के लिए क़रीब \qty{94}{kg} का क्रेट।
\textbf{8.} $m = 294 \div 9.8 = \qty{30}{kg}$; और \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर \omterm{def:g4:weight-and-mass:weight}{भार}
$30 \times 1.6 = \qty{48}{N}$.
\textbf{9.} मांसपेशियाँ और \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} दोनों जस के तस लौटे; बदला सिर्फ़
मुक़ाबिल। \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर हर \omterm{def:g3:measuring-mass:units}{किलोग्राम} उठाने का दाम \qty{1.6}{N} था; और घर
फिर \qty{9.8}{N} वसूल रहा है। वह कमाई \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} की छूट थी, शरीर की बढ़त
नहीं --- $P = m g$, जिसमें $m$ वफ़ादार है और $g$ चंचल।
\textbf{10.} $0.2 \times 24.8 \approx \qty{5}{N}$ --- निहाई तो नहीं
है: पृथ्वी पर उसी फेंटनी का \omterm{def:g4:weight-and-mass:weight}{भार} \qty{2}{N} था, इसलिए पटल पर फेंटना
ढाई गुना भारी है, कलाई थकाने लायक, पर लोहारख़ाने का काम शायद ही।
शिकायत: आंशिक रूप से सही।
\textbf{11.} एक \omterm{def:g9:weight-vs-mass:newton}{न्यूटन} चॉकलेट यानी $m = 1/g$: पृथ्वी पर क़रीब
\qty{102}{g}; \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर \qty{625}{g}; बृहस्पति पर \qty{40}{g}। अपने
\omterm{def:g9:weight-vs-mass:newton}{न्यूटन} \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर ख़रीदो।
\textbf{12.} जैसे: ``\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} तुम्हारे साथ सफ़र करता है; और \omterm{def:g4:weight-and-mass:weight}{भार} उस पर
हर दुनिया की चुंगी है --- इसलिए \omterm{def:g3:measuring-mass:units}{किलोग्राम} बाँधो, और बंदरगाहों को अपने
\omterm{def:g9:weight-vs-mass:newton}{न्यूटन} वसूलने दो।''
\end{solution}
