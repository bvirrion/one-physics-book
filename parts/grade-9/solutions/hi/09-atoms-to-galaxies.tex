\section*{अध्याय \ref{ch:g9:atoms-to-galaxies} --- परमाणुओं से मंदाकिनियों तक: ब्रह्मांड के पैमाने}

\begin{solution}{exo:g9:atoms-to-galaxies:1}
\omterm{def:g9:atoms-to-galaxies:atom}{नाभिक} ($10^{-15}$ \unit{m}), \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} ($10^{-10}$), \omterm{def:g7:states-of-matter:molecule}{अणु} (क़रीब
$10^{-9}$), चींटी ($10^{-2}$, या किसी मामूली चींटी के लिए $10^{-3}$)।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:2}
बीच में एक नन्हा, भारी, धन आवेश वाला \omterm{def:g9:atoms-to-galaxies:atom}{नाभिक}; और उससे बहुत बाहर लगभग
भारहीन ऋण \omterm{def:g9:atoms-to-galaxies:atom}{इलेक्ट्रॉनों} का एक बादल। नाप: \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} क़रीब $10^{-10}$
\unit{m}, और \omterm{def:g9:atoms-to-galaxies:atom}{नाभिक} क़रीब $10^{-15}$ --- यानी वही एक लाख गुना का
फ़ासला, जो स्टेडियम में रखे मटर वाला चित्र बना देता है।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:3}
\omterm{def:g9:atoms-to-galaxies:atom}{इलेक्ट्रॉन} --- जो ऋण हैं और $-$ से $+$ की ओर चलते हैं: यानी परंपरा के
उलट। और नियम इसलिए बचे रहते हैं कि वे कभी इस पर टिके ही नहीं थे कि
चलता कौन है: धाराएँ, \omterm{def:g5:series-parallel-circuits:parallel}{संधि} के जोड़, \omterm{def:g8:ohms-law:resistance}{ओम} के चित्र और \omterm{def:g9:electric-power-energy:power}{शक्ति} के बिल, सब उसी
तीर के साथ एक-सा पढ़े जाते हैं जिसे दुनिया भर की सहमति ने बनाए रखा है।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:4}
सबसे पास वाली दस की घात। ऊँचाई: $10^{0}$ \unit{m}; \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा}: $10^{1}$;
पृथ्वी का व्यास: $10^{7}$.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:5}
\omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} से \omterm{def:g9:atoms-to-galaxies:atom}{नाभिक}: पाँच डंडे। तुम से पृथ्वी: सात। और सूर्य की दूरी
($10^{11}$) से सबसे पास के तारे ($4 \times 10^{16}$) तक: पाँच से छह
डंडे --- कुछ लाख गुना।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:6}
\omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} $10^{-15}$ के \omterm{def:g9:atoms-to-galaxies:atom}{नाभिक} के गिर्द $10^{-10}$ \unit{m} का ज़्यादातर
कुछ-नहीं है: \omterm{def:g9:atoms-to-galaxies:atom}{परमाणुओं} को मेज़ में ठूँसो तो वह शून्य भी उनके साथ चला
आता है। और ठोसपन इलेक्ट्रॉन-बादलों का एक-दूसरे में घुसने से विद्युत
इनकार है --- यानी ख़ालीपन, मज़बूती से बचाया हुआ।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:7}
मिलीमीटर, $10^{-3}$; \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु}, $10^{-10}$: यानी सात डंडे --- उस
क़तार में क़रीब एक करोड़ \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु}।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:8}
\omterm{def:g8:speed-of-light:lightyear}{प्रकाश वर्ष}: $10^{16}$ \unit{m} (यानी साढ़े नौ हज़ार अरब)। \omterm{ex:g5:stars-night-sky:milkyway}{आकाशगंगा}:
$10^{21}$. और सबसे गहरा सर्वेक्षित आकाश: क़रीब $10^{26}$.
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:9}
$10^{5}$ मटर: $0.01 \times 10^{5} = \qty{1000}{m}$ --- यानी एक
किलोमीटर चौड़ा ``स्टेडियम'': किसी भी असली स्टेडियम से कहीं भव्य, पर
चित्र टिका रहता है --- बीच वाले निशान पर एक मटर, और अगले मटर एक
किलोमीटर दूर।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:10}
समुद्रों में बूँदें: क़रीब $10^{21} \times 10^{5} = 10^{26}$ ---
इसलिए वह पुराना दावा, अक्षरशः लिया जाए तो, हद से आगे निकल गया था:
समुद्रों में जितनी बूँदें हैं, वे एक बूँद के \omterm{def:g7:states-of-matter:molecule}{अणुओं} से कई डंडे ज़्यादा
हैं। ईमानदार रूप बच जाता है --- कि एक बूँद के \omterm{def:g7:states-of-matter:molecule}{अणु} उस हर चीज़ से
ज़्यादा हैं जिसे कोई ज़िंदगी भर में गिन सके; सीढ़ियाँ प्यारे से प्यारे
दावों को भी ईमानदार रखती हैं।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:11}
\omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} तक क़रीब दस डंडे नीचे ($10^{0}$ से $10^{-10}$) और गहरे आकाश तक
छब्बीस ऊपर --- बीच के क़रीब, और ज़रा-सा छोटे की ओर झुका हुआ। नौ
साल की पढ़ाई पाठक को हर मापी गई चीज़ के दोनों सिरों की ईमानदार पहुँच
में ला देती है।
\end{solution}

\begin{solution}{exo:g9:atoms-to-galaxies:12}
जवाब निजी हैं --- यह अभ्यास ख़ुद वही चिट्ठी है। (रूप भर के लिए एक
नमूना: \omterm{def:g1:light-and-shadows:shadow}{छाया} वाले अध्याय के पहली \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} वाले पाठक से --- ``फ़र्श पर पड़ा
वह अँधेरा जुड़वाँ एक दिन सूर्य के ग्रहण समझा देगा''; \omterm{def:g5:everyday-energy:energy}{ऊर्जा} वाले अध्याय
के पाँचवीं \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} वाले पाठक से --- ``जिन शृंखलाओं को तुम खींच रहे हो,
वे \omterm{def:g9:kinetic-energy-safety:joule}{जूल} वाले सूत्र बन जाएँगी''; और \omterm{def:g8:ohms-law:resistance}{ओम} के नियम वाले आठवीं \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} के पाठक
से --- ``जो चित्र तुम खींच रहे हो, वे नियमों को जाँचने की आदत हैं, और
पूरा भेद वही है''।)
\end{solution}

\begin{solution}{pb:g9:atoms-to-galaxies:1}
\textbf{1.} स्कूल-आँगन: सेकंड $2$; पृथ्वी: सेकंड $7$; और \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} की
\omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा}: सेकंड $9$.
\textbf{2.} सेकंड $11$ ($1.5 \times 10^{11}$ \unit{m} पर); और वाचक:
``चादर पर पड़ी यह धूप आठ मिनट पुरानी है।''
\textbf{3.} मोटे तौर पर सेकंड $13$ से $16$ तक: फ़्रेम पूरे \omterm{def:g4:solar-system:family}{सौर मंडल} को
एक बिंदु की तरह थामे है और कोई \omterm{def:g4:solar-system:star}{तारा} अभी घुसा नहीं --- इसलिए ईमानदारी
से, लगभग काला ख़ालीपन, और वह भी कई सेकंड: फ़िल्म के सबसे सच्चे दृश्य।
\textbf{4.} सबसे गहरा सर्वेक्षित आकाश, यानी मंदाकिनियों के $10^{26}$
\unit{m} --- और उसमें मौजूद सबसे पुराना \omterm{def:g1:light-and-shadows:source}{प्रकाश} ख़ुद पृथ्वी से भी पुराना।
\textbf{5.} चींटी: भीतर की ओर सेकंड $2$; कोशिका: सेकंड $5$; \omterm{def:g7:states-of-matter:molecule}{अणु}:
सेकंड $9$; \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु}: सेकंड $10$.
\textbf{6.} ख़ुद \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} के ख़ालीपन में से --- यानी बादल और केंद्र के
बीच पाँच सेकंड का कुछ-नहीं: ``अगर यह \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} कोई स्टेडियम होता, तो हम
सबसे ऊँची दीर्घाएँ छोड़ चुके हैं और अब भी एक मटर की ओर उड़े जा रहे
हैं।''
\textbf{7.} \omterm{def:g9:atoms-to-galaxies:atom}{नाभिक} --- और ईमानदारी से: ``यहाँ हमारी सीढ़ी ख़त्म होती
है, इसलिए नहीं कि क़ुदरत रुक जाती है, बल्कि इसलिए कि इंसानों ने अब तक
इन्हीं सबसे छोटे डंडों को नापा है; इमारत बनती जा रही है।''
\textbf{8.} बाहर की ओर $26$ सेकंड; और भीतर की ओर $15$: यानी इकतालीस
सेकंड का ब्रह्मांड।
\textbf{9.} हर सेकंड नज़ारे को दस गुना कर देता है --- बराबर
\emph{अनुपात}, बराबर जोड़ नहीं: सीढ़ी का तर्क यही है कि हर डंडा एक
गुणक है, और गुणकों के हिसाब से सड़क वाला क़दम तथा गुच्छ वाला क़दम एक
ही नाप के हैं।
\textbf{10.} जैसे: ``यह चाहे कितना ही भरा दिखे, यहाँ लगभग सब कुछ
ख़ाली है --- पदार्थ और आकाश, दोनों चौड़े कुछ-नहीं में बिखरे विरले
टापू हैं।''
\textbf{11.} सूक्ष्मदर्शी (\omterm{def:g8:lenses-images:families}{लेंस} वाला अध्याय: छोटी चीज़ों के ऊपर दो
\omterm{def:g8:lenses-images:families}{अभिसारी लेंस}) और दूरबीन (उसी अध्याय में, दूर की मद्धम चीज़ें बटोरती
हुई) --- यानी दोनों दिशाओं की चाबियाँ, और दोनों इसी पुस्तक की
प्रकाशिकी में घिसी हुईं।
\textbf{12.} जैसे: ``नौ साल, एक सीढ़ी: भृंगी की गिनी जा सकने वाली
टाँगों से लेकर पैरों तले की ज़मीन से भी पुरानी मंदाकिनियों तक, हर डंडा
ऐसे मापन से बिठाया गया जिसे कोई भी जाँच सकता है। और अगले साल वे सवाल
जवाब देने लगते हैं जिन्हें हमने खुला छोड़ा था --- \omterm{def:g2:pushes-and-pulls:force}{बल} और गति, \omterm{def:g1:light-and-shadows:source}{प्रकाश} की
प्रकृति, \omterm{def:g9:atoms-to-galaxies:atom}{परमाणु} का दिल; सीढ़ी साथ लाना।''
\end{solution}
