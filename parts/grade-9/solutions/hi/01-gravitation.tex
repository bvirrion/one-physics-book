\section*{अध्याय \ref{ch:g9:gravitation} --- गुरुत्वाकर्षण}

\begin{solution}{exo:g9:gravitation:1}
$F = G \, m_A m_B / d^2$, जहाँ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \omterm{def:g3:measuring-mass:units}{किलोग्राम} में, $d$ केंद्रों
के बीच \omterm{def:g2:measuring-length:units}{मीटर} में, $F$ न्यूटन में और
$G = \qty{6.67e-11}{N.m^2/kg^2}$. शब्दों में: यह खिंचाव हर \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}
के साथ बढ़ता है और दूरी के वर्ग के साथ मरता है --- और यह दोनों साथियों
पर बराबर, उलटी दिशाओं में लगता है।
\end{solution}

\begin{solution}{exo:g9:gravitation:2}
ठीक \qty{1}{N} से --- इस नियम के खिंचाव हमेशा बराबर, उलटी जोड़ियों में
ही आते हैं। सेब का त्वरण इसलिए दिखता है कि उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} नन्हा है;
जबकि वही \qty{1}{N} \qty{5.97e24}{kg} के \omterm{def:g4:solar-system:planet}{ग्रह} पर लगकर उसे नापने लायक
हद से भी कम हिलाता है।
\end{solution}

\begin{solution}{exo:g9:gravitation:3}
$2d$ पर: $d$ वाले \omterm{def:g2:pushes-and-pulls:force}{बल} का चौथाई --- यानी दो का वर्ग। और $10d$ पर:
सौवाँ हिस्सा।
\end{solution}

\begin{solution}{exo:g9:gravitation:4}
$F = \num{6.67e-11} \times 80 \times 80 \div 0.25 =
\qty{1.7e-6}{N}$ --- न्यूटन के बीस लाखवें हिस्से से भी कम: नाचने
वाले साथियों को जो भी पास खींचता हो, वह \omterm{def:g9:gravitation:gravitation}{गुरुत्वाकर्षण} तो नहीं है।
\end{solution}

\begin{solution}{exo:g9:gravitation:5}
वह दर जिससे गोला गिरता है, और वह दर जिससे गोल पृथ्वी की सतह उसके नीचे
मुड़ती चली जाती है। \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा}: गिरना और चूकना।
\end{solution}

\begin{solution}{exo:g9:gravitation:6}
अंतरिक्ष केंद्र की ऊँचाई पर गुरुत्व अपनी लगभग पूरी सतह वाली ताक़त बनाए
रखता है --- वही तो केंद्र को उसके वृत्त पर थामे है। केंद्र और चालक दल,
दोनों एक साथ पृथ्वी के गिर्द \emph{गिर} रहे हैं: साथ-साथ गिरने वाले
किसी पर दबाव नहीं डालते, और \omterm{def:g1:floating-and-sinking:words}{तैरना} उसी साझा मुक्त पतन का एहसास है।
\end{solution}

\begin{solution}{exo:g9:gravitation:7}
क्योंकि आकाश राक्षसी द्रव्यमानों में सौदा करता है --- \omterm{def:g4:solar-system:planet}{ग्रह} और तारे उस
कमज़ोर $G$ को गुणा करके ताक़तवर \omterm{def:g2:pushes-and-pulls:force}{बल} बना देते हैं --- और क्योंकि
\omterm{def:g9:gravitation:gravitation}{गुरुत्वाकर्षण} कभी कटता नहीं: वह सिर्फ़ खींचता है, इसलिए जुड़ता हर
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} खिंचाव और गहरा कर देता है।
\end{solution}

\begin{solution}{exo:g9:gravitation:8}
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का खिंचाव समुद्रों को उसकी ओर भी ढेर करता है (सबसे पास का
पानी, सबसे ज़ोर से खिंचा हुआ) और उससे दूर की ओर भी (सबसे दूर का पानी,
सबसे कम खिंचा हुआ); और घूमती पृथ्वी हर किनारे को रोज़ दोनों ढेरों के
नीचे से ले जाती है --- यानी दो ज्वार। सबसे बड़े अमावस और पूर्णिमा पर,
जब सूर्य का हाथ \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} के साथ एक क़तार में आ जाता है: बड़े ज्वार।
\end{solution}

\begin{solution}{exo:g9:gravitation:9}
$F = \num{6.67e-11} \times (\num{7.3e22} \times \num{5.97e24})
\div (\num{3.8e8})^2 \approx \qty{2.0e20}{N}$ --- एक $2$, और
उसके पीछे न्यूटनों के बीस शून्य, जो \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} को उसके पट्टे पर थामे हैं।
\end{solution}

\begin{solution}{exo:g9:gravitation:10}
दोनों घुंडियाँ एक-दूसरे के ख़िलाफ़ घूमती हैं: \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान},
क़रीब अस्सी गुना छोटा, खिंचाव को अस्सी गुना काट देता है; पर उसकी
त्रिज्या, लगभग चार गुना छोटी, सतह वाले खिंचाव को चार के वर्ग जितना
\emph{चढ़ा} देती है --- यानी क़रीब चौदह गुना। अस्सी नीचे, चौदह ऊपर: और
लगभग छठा हिस्सा बचा रह जाता है।
\end{solution}

\begin{solution}{exo:g9:gravitation:11}
प्रतिलोम वर्ग दूरी के साथ सूर्य की पकड़ पतली कर देता है, और कमज़ोर
पकड़ सिर्फ़ धीमे गिरते-चक्कर को ही थाम सकती है: बाहरी \omterm{def:g4:solar-system:planet}{ग्रह} ढीले पट्टे
पर बँधे हैं और टहलते हैं, जबकि भीतरी ज़ोर से जकड़े हैं और दौड़ते हैं
--- बुध की दौड़ और वरुण की $164$ साल लंबी सैर एक ही नियम हैं, दो
दूरियों पर पढ़े हुए।
\end{solution}

\begin{solution}{exo:g9:gravitation:12}
उसका आवर्तकाल ठीक एक दिन होना चाहिए, और चक्कर पूरब की ओर ---
ख़ुद पृथ्वी के घूमने की दिशा में। तब उपग्रह और एंटेना ठीक क़दम मिलाकर
घूमते हैं: वह ``टँगी'' तश्तरी दरअसल ऐसे तोप-गोले को देख रही है जो \omterm{def:g4:solar-system:planet}{ग्रह}
के गिर्द ठीक उसी दर से गिरता है जिस दर से \omterm{def:g4:solar-system:planet}{ग्रह} उसके नीचे घूम रहा है
--- यानी वह सिर्फ़ घूमते हुए प्रेक्षक की आँखों में ही ठहरा हुआ है।
\end{solution}

\begin{solution}{pb:g9:gravitation:1}
\textbf{1.} $F = \num{6.67e-11} \times (0.1 \times \num{5.97e24})
\div (\num{6.4e6})^2 \approx \qty{0.97}{N}$ --- कहो एक न्यूटन:
बग़ीचे वाला खिंचाव, ब्रह्मांड के स्थिरांक से निकाला हुआ।
\textbf{2.} उतने ही \qty{0.97}{N} से --- यानी आपसीपन: इस नियम के
खिंचाव सिर्फ़ बराबर, उलटी जोड़ियों में आते हैं।
\textbf{3.} केंद्र से दूरी दोगुनी होते ही खिंचाव चौथाई रह जाता है:
क़रीब \qty{0.24}{N}।
\textbf{4.} \omterm{def:g4:solar-system:planet}{ग्रह} का हर कण सेब को खींचता है --- पास के पहाड़, गहरी
क्रोड, दूसरी ओर का सिरा --- और उन सारे खिंचावों का जोड़ ठीक ऐसे बनता
है मानो पूरा \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} केंद्र पर बैठा हो: यानी एक प्रमेय, कोई मान्यता
नहीं, और उसके लेखक ने इसे साबित कर पाने तक बरसों छपाई रोके रखी।
\textbf{5.} क़रीब $2000 \times 9.8 \approx \qty{2.0e4}{N}$ ---
बीस हज़ार न्यूटन ($d$ को एक पृथ्वी-त्रिज्या लेकर नियम भी वही देता
है)।
\textbf{6.} दो त्रिज्याओं पर प्रतिलोम वर्ग चौथाई छोड़ता है: उसके
\qty{19600}{N} के सतह वाले \omterm{def:g4:weight-and-mass:weight}{भार} के मुक़ाबले क़रीब \qty{4900}{N}।
\textbf{7.} हाँ --- पूरे गुरुत्व का चौथाई कोई भारहीन शून्य नहीं है।
उपग्रह उस खिंचाव को अपना रास्ता मोड़ने में ख़र्च करता है: वह पृथ्वी के
गिर्द गिरता है और हमेशा चूकता रहता है, ठीक जैसा पहाड़ी तोप ने सिखाया
था।
\textbf{8.} ``काफ़ी तेज़ दाग़ा गया तोप का गोला गिरते हुए ठीक उतना ही
उतरता है जितना पृथ्वी मुड़ जाती है, और वह हमेशा के लिए चक्कर लगाने
लगता है --- गिरता हुआ, पर उतरता कभी नहीं। हर उपग्रह वही तोप का गोला
है, जिसे बग़ल की ओर इतना तेज़ छोड़ा गया कि वह ज़मीन को चूकता ही रहे।''
\textbf{9.} $\num{3.8e8} \div \num{6.4e6} \approx 59.4$
पृथ्वी-त्रिज्याएँ।
\textbf{10.} क़रीब $1 \div 59.4^2 \approx 1/3500$, यानी सतह वाली
ताक़त का लगभग साढ़े तीन हज़ारवाँ हिस्सा।
\textbf{11.} नरम इसलिए कि साठ त्रिज्याओं पर गुरुत्व साढ़े तीन हज़ार
गुना तनु हो चुका है --- पहले सेकंड की पाँच \omterm{def:g2:measuring-length:units}{मीटर} वाली गिरावट सिमटकर
क़रीब एक मिलीमीटर रह जाती है। और अनंत इसलिए कि गिरने का वही एक
मिलीमीटर ठीक उतना ही है जितना पृथ्वी की सतह \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} की बग़ल वाली
सरकन के नीचे मुड़ जाती है: वह गिरावट कभी बढ़त नहीं बनाती।
\textbf{12.} जैसे: ``सेब, उपग्रह और \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} एक ही वाक्य मानते हैं:
हर \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} हर दूसरे \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} की ओर गिरता है, और वह भी
$G m_A m_B / d^2$ के हिसाब से --- सेब उतर जाता है, जबकि उपग्रह और
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} चूकते रहते हैं; और फ़र्क़ सिर्फ़ बग़ल वाली \omterm{def:g7:motion-average-speed:formula}{चाल} का है।''
\end{solution}
