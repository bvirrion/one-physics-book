\section*{अध्याय \ref{ch:g9:kinetic-energy-safety} --- गतिज ऊर्जा और सड़क सुरक्षा}

\begin{solution}{exo:g9:kinetic-energy-safety:1}
\omterm{def:g9:kinetic-energy-safety:joule}{जूल} (\unit{J}): इस किताब को उसके ताक तक उठाने में कुछ \omterm{def:g9:kinetic-energy-safety:joule}{जूल} लगते हैं;
और दिल हर धड़कन पर क़रीब एक \omterm{def:g9:kinetic-energy-safety:joule}{जूल} ख़र्च करता है। $E_k = \frac12 m v^2$
--- $m$ \omterm{def:g3:measuring-mass:units}{किलोग्राम} में, $v$ \omterm{def:g2:measuring-length:units}{मीटर} प्रति सेकंड में, और $E_k$ \omterm{def:g9:kinetic-energy-safety:joule}{जूल} में।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:2}
ख़रगोश: $0.5 \times 2 \times 100 = \qty{100}{J}$. खिलाड़ी: $0.5
\times 80 \times 25 = \qty{1000}{J}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:3}
आधा \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और: गुणक $1.5$. आधी \omterm{def:g7:motion-average-speed:formula}{चाल} और: गुणक $1.5^2 = 2.25$. \omterm{def:g7:motion-average-speed:formula}{चाल}
वाली घुंडी जीत जाती है --- वह हमेशा जीतती है, वर्ग पहने हुए।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:4}
\omterm{prop:g9:kinetic-energy-safety:stopping}{प्रतिक्रिया दूरी} --- एक सेकंड का सफ़र, जो $v$ के समानुपात में बढ़ती
है; और \omterm{prop:g9:kinetic-energy-safety:stopping}{ब्रेक-दूरी} --- $E_k$ की सेवामुक्ति, जो $v^2$ के साथ बढ़ती है।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:5}
\qty{50}{km/h} पर: \qty{27}{m} में से $13$ --- यानी क़रीब आधा। और
\qty{130}{km/h} पर: \qty{121}{m} में से $85$ --- कोई सत्तर
प्रतिशत। वर्ग वाला हिस्सा समानुपाती हिस्से से आगे निकल जाता है: तेज़
\omterm{def:g7:motion-average-speed:formula}{चाल} पर ब्रेक ही पूरा जोड़ हड़प लेते हैं।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:6}
ब्रेक की चक्रिकाओं पर \omterm{def:g5:heat-and-insulation:heat}{ऊष्मा} में --- पहाड़ी ढलानों के बाद वही नारंगी
दमक, और गरम ब्रेकों की वही गंध। \omterm{def:g5:everyday-energy:energy}{ऊर्जा} आगे बढ़ाई जाती है, मिटाई कभी
नहीं: बचपन वाले \omterm{def:g5:everyday-energy:energy}{ऊर्जा} अध्याय की शृंखला वाली प्रतिज्ञप्ति आज भी हुक्म
चला रही है।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:7}
$36 \div 3.6 = \qty{10}{m}$; फिर \qty{20}{m}; फिर \qty{30}{m} ---
यानी \omterm{def:g7:motion-average-speed:formula}{चाल} के समानुपात में क़दम मिलाते हुए, जैसा एक सेकंड भर के हिस्से को
करना ही चाहिए।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:8}
तुम्हारी $E_k$ तुम्हारे \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और कार की \omterm{def:g7:motion-average-speed:formula}{चाल} से तय है --- कोई पट्टा
उसे बदलता नहीं। पेटी \emph{सेवामुक्ति} को सभ्य बनाती है: शरीर को
ज़्यादा लंबी दूरी और समय में, नरमी से रोककर, डैशबोर्ड पर एक ही झटके में
नहीं।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:9}
ब्रेक \omterm{def:g7:motion-average-speed:formula}{चाल} के वर्ग के साथ बढ़ते हैं: \qty{60}{km/h} पर (दोगुनी),
$8 \times 4 = \qty{32}{m}$; और \qty{90}{km/h} पर (तिगुनी),
$8 \times 9 = \qty{72}{m}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:10}
\qty{30}{km/h} पर: प्रतिक्रिया $\approx \qty{8.3}{m}$, ब्रेक
$13 \times (30/50)^2 \approx \qty{4.7}{m}$ --- यानी क़रीब \qty{13}{m}
में रुकना, जबकि पचास पर \qty{27}{m}: आधे से भी कम सड़क। ऊर्जाएँ:
$(30/50)^2 = 0.36$ --- टक्कर की \omterm{def:g5:everyday-energy:energy}{ऊर्जा} का बमुश्किल एक तिहाई। दोनों
मनवा लेती हैं; पर पोस्टर के लिए तैयार पंक्ति आम तौर पर सड़क वाली ही
होती है --- ``तीस पर तुम बच्चे से पहले रुक जाते हो; पचास पर तुम उस तक
चलते हुए पहुँचते हो''।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:11}
सिर्फ़ ब्रेक वाला हिस्सा दोगुना करो --- बारिश सेवामुक्ति लंबी करती है,
दिमाग़ नहीं: \qty{50}{km/h} पर $14 + 26 = \qty{40}{m}$; और
\qty{90}{km/h} पर $25 + 80 = \qty{105}{m}$. प्रतिक्रिया वाला सेकंड
सूखे और गीले, दोनों में एक-सा रहता है।
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:12}
ट्रक: $v = 25$ \unit{m/s}; $E_k = 0.5 \times 20000 \times 625 =
\qty{6.25e6}{J}$. बराबरी करती कार: $0.5 \times 1000 \times v^2 =
\num{6.25e6}$ से $v^2 = 12500$, यानी $v \approx \qty{112}{m/s}
\approx \qty{400}{km/h}$ --- सड़क की कोई कार इसके आस-पास भी नहीं
पहुँचती। राजमार्ग की \omterm{def:g7:motion-average-speed:formula}{चाल} पर चलता ट्रक लॉरी के ब्रेकों के साथ रेसिंग कार
जितनी ऊर्जाएँ ढोता है: और लंबी ढलान पर वे ब्रेक तपकर नाकाम हो जाएँ, तो
मेगाजूलों को सुरक्षित सेवामुक्त सिर्फ़ बजरी की ढलान ही कर सकती है।
कारों को ऐसी ढलान कभी नहीं चाहिए; उनकी ऊर्जाएँ आम ब्रेकों में ही मर
जाती हैं।
\end{solution}

\begin{solution}{pb:g9:kinetic-energy-safety:1}
\textbf{1.} $50 \div 3.6 = \qty{13.9}{m/s}$; $60 \div 3.6 =
\qty{16.7}{m/s}$.
\textbf{2.} \qty{13.9}{m} और \qty{16.7}{m}।
\textbf{3.} \qty{13}{m}; और साठ पर $13 \times (60/50)^2 =
13 \times 1.44 \approx \qty{18.7}{m}$.
\textbf{4.} रुकना: \qty{35.4}{m} के मुक़ाबले $26.9$ --- यानी पोस्टर
की ख़ाली जगह: क़रीब \emph{साढ़े आठ \omterm{def:g2:measuring-length:units}{मीटर}} और, यानी वहाँ से दो कार भर
आगे जहाँ पचास वाला चालक रुक चुका होता है।
\textbf{5.} पचास पर: क़रीब \qty{97}{kJ}। और नब्बे पर ($v =
\qty{25}{m/s}$): $0.5 \times 1000 \times 625 = \qty{312}{kJ}$.
\textbf{6.} $97 \div 9.8 \approx \qty{10}{m}$; और $312 \div 9.8
\approx \qty{32}{m}$.
\textbf{7.} दस \omterm{def:g2:measuring-length:units}{मीटर}: तीन मंज़िला मकान; और बत्तीस: दस मंज़िला इमारत।
मसौदा: ``पचास की टक्कर तीसरी मंज़िल से गिरना है। नब्बे पर, दसवीं से।
अपनी खिड़की चुन लो।''
\textbf{8.} वे ऊँचाइयाँ ईमानदारी से उस \omterm{def:g5:everyday-energy:energy}{ऊर्जा} को मापती हैं जिसे
सेवामुक्त करना ही है --- और उस अंक को कोई अभियांत्रिकी नहीं बदलती।
मुड़ने वाले हिस्से और पेटियाँ सेवामुक्ति का \emph{ढंग} बदलते हैं:
सेंटीमीटरों में क्रूर रुकावट के बजाय मीटरों में नरम रुकावट --- और
इसीलिए गिरने वाली उपमा किसी आधुनिक कार में चोट को बढ़ा-चढ़ाकर बताती है,
और इसीलिए वह जिस \omterm{def:g5:everyday-energy:energy}{ऊर्जा} को दिखाती है वह फिर भी उतनी ही असली है।
\textbf{9.} हर सेकंड \qty{25}{m} आँख मूँदे हुए; और दो सेकंड की नज़र
पर \qty{50}{m} --- प्रतिक्रिया वाला सेकंड शुरू होने से पहले ही
आधा मैदान।
\textbf{10.} $65 + 25 = \qty{90}{m}$ --- यानी थका चालक चौकन्ने चालक
के जोड़ में डेढ़ बस भर और जोड़ देता है।
\textbf{11.} सबसे ज़्यादा ग़लत \emph{कम} चालों पर: वहाँ वर्ग वाला
ब्रेक हिस्सा छोटा होता है और रुकने की दूरी का ज़्यादातर हिस्सा
प्रतिक्रिया ही होती है --- शहर में वह सेकंड ही ख़तरा \emph{है}। (तेज़
\omterm{def:g7:motion-average-speed:formula}{चाल} पर ब्रेक हावी हो जाते हैं --- पर वहाँ भी हर सेकंड के \qty{25}{m}
कोई मामूली बात नहीं।)
\textbf{12.} जैसे: ``\omterm{def:g7:motion-average-speed:formula}{चाल} के दस और, तो रुकने के दस \omterm{def:g2:measuring-length:units}{मीटर} और --- \omterm{def:g7:motion-average-speed:formula}{चाल} दो
बार भीतर घुसती है, और एक बार वर्ग होकर। टक्कर एक गिरना है: \omterm{def:g5:everyday-energy:energy}{ऊर्जा} \omterm{def:g7:motion-average-speed:formula}{चाल}
के वर्ग के साथ बढ़ती है। और नज़र का एक सेकंड पच्चीस \omterm{def:g2:measuring-length:units}{मीटर} का अंधापन है
--- ध्यान ही सबसे छोटी \omterm{prop:g9:kinetic-energy-safety:stopping}{ब्रेक-दूरी} है।''
\end{solution}
