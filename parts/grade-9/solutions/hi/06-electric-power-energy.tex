\section*{अध्याय \ref{ch:g9:electric-power-energy} --- विद्युत शक्ति और ऊर्जा}

\begin{solution}{exo:g9:electric-power-energy:1}
\omterm{def:g9:electric-power-energy:power}{शक्ति} \omterm{def:g5:everyday-energy:energy}{ऊर्जा} बदलने की रफ़्तार है --- यानी \omterm{def:g9:kinetic-energy-safety:joule}{जूल} प्रति सेकंड, \omterm{def:g9:electric-power-energy:power}{वाट} में: नल
का बहाव, बाल्टी नहीं। $P = U \times I$ और $E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
लैंप: $230 \times 0.022 \approx \qty{5}{W}$. स्टार्टर: $12 \times 100
= \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
केतली: $2300 \div 230 = \qty{10}{A}$; माइक्रोवेव: $700 \div 230
\approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
क्योंकि गरमाहट \omterm{def:g5:everyday-energy:energy}{ऊर्जा} का सबसे महँगा रूप है: पानी का या किसी कमरे का \omterm{def:g3:temperature-thermometers:temperature}{ताप}
चढ़ाना जूलों को उस रफ़्तार से हड़पता है जिसके आस-पास लैंप और
इलेक्ट्रॉनिकी कभी नहीं पहुँचते --- इसलिए हर हीटर की पट्टी किलोवाटों में
पढ़ी जाती है।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. जूलों में गिना महीना दस अंकों तक चला जाता है; इसलिए
मीटर बनाने वालों ने ऐसा \omterm{def:g6:measurement-in-science:measuring}{मात्रक} चुना जो घरों के महीनों को आरामदेह
सैकड़ों में रखता है।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
हीटर: $2 \times 0.5 = \qty{1}{kWh}$. टेलीविज़न: $0.1 \times 5 =
\qty{0.5}{kWh}$. आधे घंटे का हीटर पूरी टेलीविज़न वाली शाम से दोगुना
ख़र्च कर जाता है।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$; लागत $6 \times 0.25 =
1.5$ मुद्रा-मात्रक --- एक मिठाई के दाम में पूरे महीने का गृहकार्य।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}
घर में घुसता हर \omterm{def:g9:electric-power-energy:kwh}{किलोवाट-घंटा}। और बिल इस महीने के पाठ में से पिछले महीने
का पाठ घटाकर उस अंतर को दर से गुणा कर देता है।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
पुरखे: $0.6 \times 4 \times 30 = \qty{72}{kWh}$. आधुनिक: $0.05 \times
4 \times 30 = \qty{6}{kWh}$. बचत: \qty{66}{kWh} --- यानी हर महीने
$16.5$ मुद्रा-मात्रक, और वह भी सीढ़ी पर बिताई एक दोपहर से।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}
ड्रायर: $2000 \div 230 \approx \qty{8.7}{A}$. केतली की \qty{10}{A}
के साथ: $18.7$ --- \qty{16}{A} वाली \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} की सीमा के पार: विच्छेदक
(और ठीक ही) इस प्रयोग को ख़त्म कर देगा।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
केतली: $60 \div 2.3 \approx 26$ घंटे। लैंप: $60 \div 0.005 = 12000$
घंटे --- यानी एक साल चार महीने। फिर भी वही भंडार कार को सिर्फ़ कुछ सौ
किलोमीटर चलाता है: एक टन मशीन को हिलाना \omterm{def:g5:everyday-energy:energy}{ऊर्जा} को उस पैमाने पर हड़पता
है जिसके बग़ल घर का उजाला गोल कर दी जाने वाली ग़लती भर है।
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
भविष्यवाणी: $t = E/P = \num{5.0e5} \div 2300 \approx \qty{217}{s}$
--- क़रीब साढ़े तीन मिनट। असली केतली इससे ज़्यादा देर चलती है:
उसके कुछ \omterm{def:g9:kinetic-energy-safety:joule}{जूल} पानी के पास से रिसकर निकल जाते हैं --- ख़ुद केतली का बदन
गरम करते हुए, \omterm{def:g2:air-around-us:air}{हवा} गरम करते हुए, और जल्दी उठती भाप बनकर भागते हुए।
\omterm{def:g5:everyday-energy:energy}{ऊर्जा} कभी नष्ट नहीं होती, पर वह पूरी की पूरी वहीं पहुँच भी नहीं जाती
जहाँ इरादा था: यानी शृंखला का वही रिसाव, जिसका समय आख़िरकार तुमने अपनी
ही रसोई में नाप लिया।
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} टेलीविज़न: $0.1 \times 4 \times 30 = \qty{12}{kWh}$;
लैंप: $0.05 \times 5 \times 30 = \qty{7.5}{kWh}$.
\textbf{3.} मशीन: $2 \times 1 \times 15 = \qty{30}{kWh}$; यंत्र:
$2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- यानी दस छिपे हुए
\omterm{def:g9:electric-power-energy:power}{वाट}, और केतली का पूरा महीना।
\textbf{5.} औसत $120 \div 3 = \qty{40}{W}$; महीना: $0.04 \times 720
= \qty{28.8}{kWh}$.
\textbf{6.} कुल $\approx \qty{242}{kWh}$. हावी: पानी गरम करने वाला
यंत्र (\qty{150}{}), कपड़े धोने की मशीन (\qty{30}{}), और रेफ़्रिजरेटर
(\qty{28.8}{kWh}) --- गरम करना और हमेशा चलते रहना।
\textbf{7.} $242 \times 0.25 \approx 60.6$ \omterm{def:g6:measurement-in-science:measuring}{मात्रक}; और यंत्र का
हिस्सा: $150 \times 0.25 = 37.5$ --- यानी बिल का आधे से भी ज़्यादा।
\textbf{8.} $2150 + 242 = 2392$ \omterm{def:g6:measurement-in-science:measuring}{मात्रक}।
\textbf{9.} यंत्र की नई पंक्ति: $2.5 \times 1.5 \times 30 =
\qty{112.5}{kWh}$: यानी \qty{37.5}{kWh} की बचत, और हर महीने क़रीब
$9.4$ मुद्रा-मात्रक।
\textbf{10.} ख़ाली बैठा एक चार्जर: $0.0002 \times 720 =
\qty{0.14}{kWh}$ --- मुद्रा के क़रीब चार सौवें हिस्से: पहला सुझाव
हर चार्जर के मुक़ाबले दो सौ गुना ज़्यादा बचाता है। लेखे लीवर ढूँढ़ते
हैं; नारे चार्जर।
\textbf{11.} केतली की पंक्ति आधी होकर \qty{3.45}{kWh} रह जाती है:
यानी \qty{3.45}{kWh} की बचत, और एक मुद्रा-मात्रक से भी कम --- दादी जी
की आदत सही है, और मामूली भी।
\textbf{12.} जैसे: ``किलोवाट-घंटे वहीं रहते हैं जहाँ पानी गरम होता है
और जहाँ मशीनें कभी सोती ही नहीं। स्टैंडबाय वाली पंक्ति समय वाले गुणक
की सीख देती है: जब $t$ हमेशा हो, तब नन्ही शक्तियाँ भी बिल उगा देती
हैं। और सबसे ज़्यादा चुकाने वाला बदलाव छोटा स्नान है --- बिल का मालिक
वही यंत्र है, इसलिए बचत का मालिक भी उसके घंटे ही हैं।''
\end{solution}
