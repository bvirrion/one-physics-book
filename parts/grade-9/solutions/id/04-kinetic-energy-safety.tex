\section*{Bab \ref{ch:g9:kinetic-energy-safety} --- Energi Kinetik dan Keselamatan Jalan}

\begin{solution}{exo:g9:kinetic-energy-safety:1}
Joule (\unit{J}): beberapa \omterm{def:g9:kinetic-energy-safety:joule}{joule} mengangkat buku ini ke raknya;
jantungnya membelanjakan kira-kira satu tiap detak. $E_k = \frac12 m
v^2$ --- $m$ dalam \omterm{def:g3:measuring-mass:units}{kilogram}, $v$ dalam \omterm{def:g2:measuring-length:units}{meter} tiap sekon, $E_k$ dalam
\omterm{def:g9:kinetic-energy-safety:joule}{joule}.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:2}
Kelinci: $0.5 \times 2 \times 100 = \qty{100}{J}$. Pemainnya: $0.5
\times 80 \times 25 = \qty{1000}{J}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:3}
\omterm{def:g3:measuring-mass:mass}{Massa} separuh lebih banyak: faktor $1.5$. \omterm{def:g7:motion-average-speed:formula}{Kelajuan} separuh
lebih banyak: faktor $1.5^2
= 2.25$. Tombol \omterm{def:g7:motion-average-speed:formula}{kelajuannya} menang --- ia selalu menang, karena
mengenakan kuadratnya.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:4}
\omterm{prop:g9:kinetic-energy-safety:stopping}{Jarak tanggapnya} --- perjalanan satu sekon, yang bertumbuh sebanding
dengan $v$; \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak pengeremannya} --- pensiunnya $E_k$, yang bertumbuh
menurut $v^2$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:5}
Pada \qty{50}{km/h}: $13$ dari \qty{27}{m} --- kira-kira separuhnya.
Pada \qty{130}{km/h}: $85$ dari \qty{121}{m} --- kira-kira tujuh puluh
persen. Komponen yang berkuadrat mengalahkan pertumbuhan komponen yang
sebanding: pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} tinggi, pengeremanlah yang melahap totalnya.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:6}
Menjadi \omterm{def:g5:heat-and-insulation:heat}{kalor} pada cakram remnya --- pijar jingga sesudah menuruni
gunung, bau rem yang panas. \omterm{def:g5:everyday-energy:energy}{Energi} diteruskan, tidak pernah dihapus:
proposisi rantai pada bab \omterm{def:g5:everyday-energy:energy}{energi} masa kecilnya, yang masih memerintah.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:7}
$36 \div 3.6 = \qty{10}{m}$; lalu \qty{20}{m}; lalu \qty{30}{m} ---
berbaris sebanding dengan \omterm{def:g7:motion-average-speed:formula}{kelajuannya}, sebagaimana seharusnya sebuah
bentangan satu sekon.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:8}
Nilai $E_k$-mu dipancangkan massamu dan \omterm{def:g7:motion-average-speed:formula}{kelajuan} mobilnya --- tidak ada
tali yang mengubahnya. Sabuknya memberadabkan \emph{pensiunnya}:
menghentikan tubuhnya atas jarak dan waktu yang lebih panjang, dengan
lembut, alih-alih menabrak papan kemudinya sekaligus.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:9}
Pengereman berskala menurut kuadrat \omterm{def:g7:motion-average-speed:formula}{kelajuannya}: pada \qty{60}{km/h}
(dua kali lipat), $8 \times 4 = \qty{32}{m}$; pada \qty{90}{km/h}
(tiga kali lipat), $8 \times 9 = \qty{72}{m}$.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:10}
Pada \qty{30}{km/h}: tanggapnya $\approx \qty{8.3}{m}$, pengeremannya
$13 \times (30/50)^2 \approx \qty{4.7}{m}$ --- berhenti dalam
kira-kira \qty{13}{m}, lawan \qty{27}{m} pada lima puluh: kurang dari
separuh jalannya. \omterm{def:g5:everyday-energy:energy}{Energinya}: $(30/50)^2 = 0.36$ --- nyaris sepertiga
\omterm{def:g5:everyday-energy:energy}{energi} tabrakannya. Keduanya meyakinkan; baris yang siap poster
biasanya baris jalannya --- ``pada 30 kamu berhenti sebelum anaknya;
pada 50 kamu mencapainya sambil masih bergerak''.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:11}
Gandakan hanya bagian pengeremannya --- hujan memanjangkan pensiunnya,
bukan otaknya: pada \qty{50}{km/h}, $14 + 26 =
\qty{40}{m}$; pada \qty{90}{km/h}, $25 + 80 = \qty{105}{m}$.
Sekon tanggapnya sama saja kering maupun basah.
\end{solution}

\begin{solution}{exo:g9:kinetic-energy-safety:12}
Truknya: $v = 25$ \unit{m/s}; $E_k = 0.5 \times 20000 \times
625 = \qty{6.25e6}{J}$. Mobil yang menyamainya: $0.5 \times 1000 \times
v^2 = \num{6.25e6}$ memberi $v^2 = 12500$, $v \approx
\qty{112}{m/s} \approx \qty{400}{km/h}$ --- tidak ada mobil jalan raya
yang mendekatinya. Sebuah truk pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} jalan bebas \omterm{def:g8:ohms-law:resistance}{hambatan}
membawa \omterm{def:g5:everyday-energy:energy}{energi} mobil balap dengan rem truk: ketika remnya terlalu
panas lalu gagal pada turunan yang panjang, hanya sebuah landaian
kerikil yang dapat memensiunkan megajoule dengan aman. Mobil tidak
pernah memerlukannya; \omterm{def:g5:everyday-energy:energy}{energinya} mati di dalam rem yang biasa.
\end{solution}

\begin{solution}{pb:g9:kinetic-energy-safety:1}
\textbf{1.} $50 \div 3.6 = \qty{13.9}{m/s}$; $60 \div 3.6 =
\qty{16.7}{m/s}$.
\textbf{2.} \qty{13.9}{m} dan \qty{16.7}{m}.
\textbf{3.} \qty{13}{m}; pada enam puluh, $13 \times (60/50)^2 =
13 \times 1.44 \approx \qty{18.7}{m}$.
\textbf{4.} Berhentinya: $26.9$ lawan \qty{35.4}{m} --- isian
posternya: kira-kira \emph{delapan setengah \omterm{def:g2:measuring-length:units}{meter}} lebih banyak, dua
panjang mobil melewati tempat pengemudi lima puluh sudah berhenti.
\textbf{5.} Pada lima puluh: kira-kira \qty{97}{kJ}. Pada sembilan
puluh ($v =
\qty{25}{m/s}$): $0.5 \times 1000 \times 625 =
\qty{312}{kJ}$.
\textbf{6.} $97 \div 9.8 \approx \qty{10}{m}$; $312 \div 9.8
\approx \qty{32}{m}$.
\textbf{7.} Sepuluh \omterm{def:g2:measuring-length:units}{meter}: sebuah rumah bertingkat tiga; tiga puluh
dua: sebuah blok bertingkat sepuluh. Rancangannya: ``Tabrakan pada 50
adalah jatuh dari lantai tiga. Pada 90, dari lantai sepuluh. Pilihlah
jendelamu.''
\textbf{8.} Tingginya dengan jujur \omterm{def:g6:measurement-in-science:measuring}{mengukur} \omterm{def:g5:everyday-energy:energy}{energi} yang harus
dipensiunkan --- bilangan itu tidak diubah rekayasa apa pun. Zona
remuk dan sabuknya mengubah \emph{tata cara} pensiunnya: \omterm{def:g2:measuring-length:units}{bermeter-meter}
penghentian yang lembut alih-alih beberapa \omterm{def:g2:measuring-length:units}{sentimeter} penghentian yang
kejam --- dan itulah sebabnya perumpamaan jatuhnya melebih-lebihkan
cederanya di dalam mobil zaman sekarang, dan sebabnya \omterm{def:g5:everyday-energy:energy}{energi} yang
digambarkannya tetap saja nyata.
\textbf{9.} \qty{25}{m} dengan buta tiap sekon; sebuah lirikan dua
sekon, \qty{50}{m} --- separuh lapangan sebelum sekon tanggapnya
bahkan mulai.
\textbf{10.} $65 + 25 = \qty{90}{m}$ --- pengemudi yang lelah
menambahkan satu setengah panjang bus pada total pengemudi yang
waspada.
\textbf{11.} Paling keliru pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} yang \emph{rendah}: di sana
bagian pengereman yang berkuadrat itu kecil dan bentangan tanggapnya
menjadi sebagian besar jarak berhentinya --- di dalam kota, sekonnya
\emph{adalah} bahayanya. (Pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} tinggi pengeremannya menguasai
--- tetapi \qty{25}{m} tiap sekon juga bukan hal sepele di sana.)
\textbf{12.} Misalnya: ``Sepuluh \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g7:motion-average-speed:formula}{kelajuan} lebih banyak, sepuluh
\omterm{def:g2:measuring-length:units}{meter} jarak berhenti lebih banyak --- \omterm{def:g7:motion-average-speed:formula}{kelajuan} masuk dua kali, sekali
berkuadrat. Sebuah tabrakan adalah sebuah jatuh: \omterm{def:g5:everyday-energy:energy}{energi} bertumbuh
menurut kuadrat \omterm{def:g7:motion-average-speed:formula}{kelajuannya}. Satu sekon lirikan adalah dua puluh lima
\omterm{def:g2:measuring-length:units}{meter} kebutaan --- perhatian adalah \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak pengereman} yang terpendek di
antara semuanya.''
\end{solution}
