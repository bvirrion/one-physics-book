\section*{Bab \ref{ch:g9:energy-conversions} --- Bentuk Energi dan Perubahannya}

\begin{solution}{exo:g9:energy-conversions:1}
Kinetik (bola yang menggelinding), potensial (ember yang dinaikkan),
termal (air mandi yang hangat), kimia (sarapan, \omterm{def:g3:first-electric-circuit:battery}{baterainya}), listrik
(pasokan dindingnya), radiasi (sinar matahari lewat jendelanya),
nuklir (inti pembangkit listriknya).
\end{solution}

\begin{solution}{exo:g9:energy-conversions:2}
Embernya: $2 \times 9.8 \times 10 = \qty{196}{J}$. Pendakinya: $60
\times 9.8 \times 500 = \qty{2.9e5}{J}$ --- hampir tiga ratus
kilojoule ketinggian.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:3}
Pada setiap proses, \omterm{def:g5:everyday-energy:energy}{energi} berganti bentuk dan berganti pemilik tetapi
totalnya kekal dengan tepat. Mesin abadinya mati karenanya: sebuah
mesin yang mengeluarkan lebih banyak daripada yang disimpan atau
diterimanya akan menciptakan \omterm{def:g5:everyday-energy:energy}{energi} --- terlarang tanpa banding.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:4}
Puncak ayunannya: seluruhnya potensial, diam sejenak. Dasarnya:
seluruhnya kinetik, tercepat. Pergeserannya: tiap \omterm{def:g6:describing-motion:trajectory}{lintasan} mengikis
sedikit menjadi kehangatan \omterm{def:g2:air-around-us:air}{udara} dan porosnya, dan ayunannya pun mati
perlahan --- rekeningnya berpindah, totalnya utuh.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:5}
$v = \sqrt{2 \times 9.8 \times 5} \approx \qty{9.9}{m/s}$ ---
kira-kira \qty{36}{km/h}. \omterm{def:g3:measuring-mass:mass}{Massanya} meniadakan diri ketika $m g h$
disamakan dengan $\frac12 m v^2$: peloncat yang \omterm{def:g4:weight-and-mass:weight}{berat} dan yang ringan
mencebur pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} yang sama.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:6}
Yang berguna dibagi totalnya: $170 \div 200 = 0.85$ --- delapan puluh
lima persen. Angka \qty{30}{J} yang hilang: kehangatan di dalam
lilitan dan bantalannya.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:7}
Lima persen masukannya pergi sebagai \omterm{def:g1:light-and-shadows:source}{cahaya}; sembilan puluh lima
pergi sebagai kehangatan. Sebuah alat yang dipilah menurut keluarannya
ternyata sebuah pemanas --- yang kebetulan berpijar.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:8}
Kimia (sarapan) $\to$ kerja ototnya $\to$ \omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetik} pengendara
dan mesinnya $\to$ tersimpan sebagai \omterm{def:g9:energy-conversions:forms}{energi potensial}, $m g
h$, di puncaknya --- dengan kebocoran termal sepanjang jalannya: otot yang
hangat, \omterm{def:g2:air-around-us:air}{udara} yang teraduk hangat, ban yang hangat. Totalnya: kekal.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:9}
Tiap tonnya: $1000 \times 9.8 \times 50 = \qty{4.9e5}{J}$;
pada sembilan puluh persen, kira-kira \qty{4.4e5}{J} listrik. Untuk
\qty{4.4e6}{W}: $\num{4.4e6} \div \num{4.4e5} = 10$ ton tiap sekon
menembus turbinnya.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:10}
Sebelumnya: $0.5 \times 9.8 \times 2 = \qty{9.8}{J}$. Sesudah
pantulannya: $0.5 \times 9.8 \times 1.2 \approx \qty{5.9}{J}$.
Kikisannya: kira-kira \qty{3.9}{J} menjadi kehangatan bola dan
lantainya (dan sebuah \omterm{def:g3:sound-around-us:sound}{bunyi} klik --- yang sendiri segera menjadi
kehangatan). Pemeriksaannya berakhir ketika lentingannya mencapai
tinggi nol: setiap \omterm{def:g9:kinetic-energy-safety:joule}{joule} dari $9.8$ yang asli lalu beristirahat di
rekening termalnya.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:11}
$E_p = 500 \times 9.8 \times 30 = \qty{1.47e5}{J}$; pada sembilan
puluh persen meterannya harus memasok $\num{1.47e5} \div 0.9 \approx
\qty{1.6e5}{J}$. Sepersepuluh tambahannya tidak hilang dari alam
semesta --- ia menghangatkan \omterm{ex:g6:circuits-and-safety:motor}{motor}, gigi, dan porosnya: kekekalannya
bertahan; pembukuan liftnya sekadar memuat sebuah baris termal.
\end{solution}

\begin{solution}{exo:g9:energy-conversions:12}
\omterm{def:g7:motion-average-speed:formula}{Kelajuan} dasarnya: $v = \sqrt{2 \times 9.8 \times 4} \approx
\qty{8.9}{m/s}$. Secara ideal bibir kanannya diraih kembali tepat pada
\qty{4}{m}; sesungguhnya sedikit lebih rendah, tiap penyeberangannya
lebih rendah lagi, sampai pemain papan luncurnya berayun sampai diam
di dasarnya --- seluruh $m g h$-nya dipensiunkan menjadi kehangatan
roda, landaian, dan \omterm{def:g2:air-around-us:air}{udaranya}. Tidak ada yang dimusnahkan, tetapi
segala yang berguna dibelanjakan: pembukuan dunianya memperoleh
kehangatan yang tersebar, dan pemain papan luncurnya kehilangan tinggi
yang terpusat dan dapat dipakai ulang --- jumlahnya selamat;
kegunaannya tidak kembali.
\end{solution}

\begin{solution}{pb:g9:energy-conversions:1}
\textbf{1.} $E_p = 2000 \times 9.8 \times 45 =
\qty{8.8e5}{J}$.
\textbf{2.} $v = \sqrt{2 \times 9.8 \times 45} \approx
\qty{29.7}{m/s} \approx \qty{107}{km/h}$.
\textbf{3.} Puncak \qty{50}{m} menuntut \omterm{def:g5:everyday-energy:energy}{energi} yang lebih banyak
daripada yang dianugerahkan puncaknya: pembukuannya melarangnya. Dari
awal \qty{45}{m} pada keadaan diam, tidak ada bukit berikutnya yang
boleh melampaui \qty{45}{m} --- dan sesungguhnya, gesekannya menuntut
jelas lebih rendah lagi.
\textbf{4.} Yang terukur: $0.5 \times 2000 \times 28^2 =
\qty{7.8e5}{J}$ terhadap \qty{8.8e5}{J} yang ideal: kira-kira
\qty{1e5}{J} dibebankan pada rekening termalnya --- rel, roda, dan
\omterm{def:g2:air-around-us:air}{udaranya}, yang menghangat.
\textbf{5.} $P = \num{8.8e5} \div 60 \approx
\qty{1.5e4}{W}$ --- kira-kira lima belas kilowatt.
\textbf{6.} $14.7 \div 20 \approx 0.74$: tujuh puluh empat persen.
\textbf{7.} Sepuluh \omterm{ex:g2:measuring-time:clock}{jam} dengan satu angkatan tiap empat menit: $150$
angkatan; masing-masing memakan $20 \times \tfrac{1}{60} =
\qty{0.33}{kWh}$; hariannya $150 \times 0.33 = \qty{50}{kWh}$
--- $12.5$ \omterm{def:g6:measurement-in-science:measuring}{satuan} mata uang.
\textbf{8.} Sebuah rem-pembangkit yang murah hati mungkin memulihkan
sebagian besar \omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetik} kedatangannya --- tetapi setiap tahapnya
(rel, rem, pembangkit, trafo) membayar komisi termalnya, dan \omterm{def:g9:kinetic-energy-safety:joule}{joule}
yang dipulihkan tidak akan pernah dapat menyamai belanja dereknya:
kekekalannya mengizinkan daur ulang; efisiensi-kurang-dari-satu
melarang lingkaran penuhnya. Wahananya dapat meringankan biayanya
sendiri, tidak pernah membayarnya sendiri.
\textbf{9.} $v = \sqrt{2 \times 9.8 \times 50} \approx
\qty{31.3}{m/s}$.
\textbf{10.} Kinetiknya ketika masuk $0.5 \times 500 \times 980 =
\qty{2.45e5}{J}$, ditambah $500 \times 9.8
\times 10 = \qty{4.9e4}{J}$ milik sepuluh \omterm{def:g2:measuring-length:units}{meter} terakhirnya:
kira-kira \qty{2.9e5}{J} bagi remnya --- yang dikirim, lewat arus
pusar yang diinduksikan, menjadi kehangatan sirip remnya: induksi
sebagai rencana pensiun.
\textbf{11.} Pengakuannya bertahan, terasa walaupun tidak terkata
dengan tepat: gondola dan penumpangnya jatuh bersama-sama, dan teman
yang sama-sama jatuh tidak menekan apa pun --- jeritannya adalah jatuh
bebas yang dijalani bersama, dengan \omterm{def:g9:gravitation:gravitation}{gravitasi} hadir sepenuhnya namun
tidak terasa.
\textbf{12.} Misalnya: ``Koaster kami hidup oleh satu pertukaran:
tinggi ditukar \omterm{def:g7:motion-average-speed:formula}{kelajuan} lalu kembali, $m g h$ lawan $\frac12 m v^2$.
Setiap wahananya menaati satu hukum: total \omterm{def:g5:everyday-energy:energy}{energinya}, yang diperiksa
di setiap titik, tidak pernah bergerak. Dan satu rekening menagih
pajak kepada semuanya: kehangatan --- komisi gesekan yang diam-diam,
yang harus dibayar pada setiap wahananya.''
\end{solution}
