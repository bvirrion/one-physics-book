\section*{Bab \ref{ch:g9:electric-power-energy} --- Daya dan Energi Listrik}

\begin{solution}{exo:g9:electric-power-energy:1}
\omterm{def:g9:electric-power-energy:power}{Daya} adalah langkah pengubahan \omterm{def:g5:everyday-energy:energy}{energi} --- \omterm{def:g9:kinetic-energy-safety:joule}{joule} tiap sekon, dalam
\omterm{def:g9:electric-power-energy:power}{watt}: aliran kerannya, bukan embernya. $P = U \times I$
dan $E = P \times t$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:2}
Lampunya: $230 \times 0.022 \approx \qty{5}{W}$. \omterm{ex:g6:circuits-and-safety:motor}{Motor} penyalanya: $12
\times 100 = \qty{1200}{W}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:3}
Cereknya: $2300 \div 230 = \qty{10}{A}$; oven gelombang mikronya: $700 \div
230 \approx \qty{3}{A}$.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:4}
Karena kehangatan adalah bentuk \omterm{def:g5:everyday-energy:energy}{energi} yang paling mahal untuk
dipasok: menaikkan \omterm{def:g3:temperature-thermometers:temperature}{suhu} air atau \omterm{def:g3:temperature-thermometers:temperature}{suhu} sebuah ruangan melahap \omterm{def:g9:kinetic-energy-safety:joule}{joule}
pada langkah yang tidak pernah didekati lampu dan elektronika ---
sehingga pelat setiap pemanas terbaca dalam kilowatt.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:5}
$\qty{1}{kWh} = 1000 \, \text{W} \times 3600 \, \text{s} =
\qty{3.6e6}{J}$. Sebulan \omterm{def:g9:kinetic-energy-safety:joule}{joule} berjalan sampai sepuluh angka;
pembuat meterannya memilih \omterm{def:g6:measurement-in-science:measuring}{satuan} yang menjaga bulan rumah tangganya
pada ratusan yang nyaman.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:6}
Pemanasnya: $2 \times 0.5 = \qty{1}{kWh}$. Televisinya: $0.1
\times 5 = \qty{0.5}{kWh}$. Pemanas setengah \omterm{ex:g2:measuring-time:clock}{jamnya} melampaui
belanja sesore penuh televisinya, dua kali lipat.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:7}
$E = 0.05 \times 4 \times 30 = \qty{6}{kWh}$; biayanya $6 \times
0.25 = 1.5$ \omterm{def:g6:measurement-in-science:measuring}{satuan} mata uang --- sebulan pekerjaan rumah seharga
sepotong kue.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:8}
Setiap \omterm{def:g9:electric-power-energy:kwh}{kilowatt-jam} yang masuk ke rumahnya. Tagihannya mengurangkan
bacaan bulan lalu dari bacaan bulan ini lalu mengalikan selisihnya
dengan tarifnya.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:9}
Leluhurnya: $0.6 \times 4 \times 30 = \qty{72}{kWh}$. Yang sezaman:
$0.05 \times 4 \times 30 = \qty{6}{kWh}$. Penghematannya:
\qty{66}{kWh} --- $16.5$ \omterm{def:g6:measurement-in-science:measuring}{satuan} mata uang tiap bulan, dari sesore di
atas sebuah tangga lipat.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:10}
Pengeringnya: $2000 \div 230 \approx \qty{8.7}{A}$. Bersama
\qty{10}{A} milik cereknya: $18.7$ --- melewati takaran \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
\qty{16}{A}-nya: pemutus arusnya akan (dengan benar) mengakhiri
percobaannya.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:11}
Cereknya: $60 \div 2.3 \approx 26$ \omterm{ex:g2:measuring-time:clock}{jam}. Lampunya: $60 \div
0.005 = 12000$ \omterm{ex:g2:measuring-time:clock}{jam} --- setahun empat bulan. Namun simpanan yang sama
menggerakkan mobilnya hanya beberapa ratus kilometer: menggerakkan
seton mesin melahap \omterm{def:g5:everyday-energy:energy}{energi} pada skala yang di sampingnya penerangan
rumah tangga hanyalah kesalahan pembulatan.
\end{solution}

\begin{solution}{exo:g9:electric-power-energy:12}
Ramalannya: $t = E/P = \num{5.0e5} \div 2300 \approx
\qty{217}{s}$ --- kira-kira tiga setengah menit. Cerek yang
sesungguhnya berjalan lebih lama: sebagian \omterm{def:g9:kinetic-energy-safety:joule}{joulenya} bocor melewati
airnya --- menghangatkan badan cereknya sendiri, \omterm{def:g2:air-around-us:air}{udaranya}, lalu lolos
sebagai uap yang awal. \omterm{def:g5:everyday-energy:energy}{Energi} tidak pernah dimusnahkan, tetapi tidak
pula seluruhnya diantarkan ke tempat yang dituju: kebocoran rantainya,
yang akhirnya dicatat waktunya di dapurmu sendiri.
\end{solution}

\begin{solution}{pb:g9:electric-power-energy:1}
\textbf{1.} $2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$.
\textbf{2.} Televisinya: $0.1 \times 4 \times 30 =
\qty{12}{kWh}$; lampunya: $0.05 \times 5 \times 30 =
\qty{7.5}{kWh}$.
\textbf{3.} Mesin cucinya: $2 \times 1 \times 15 = \qty{30}{kWh}$;
pemanasnya: $2.5 \times 2 \times 30 = \qty{150}{kWh}$.
\textbf{4.} $0.010 \times 720 = \qty{7.2}{kWh}$ --- sepuluh \omterm{def:g9:electric-power-energy:power}{watt} yang
tersembunyi, sebulan penuh milik sebuah cerek.
\textbf{5.} Rata-ratanya $120 \div 3 = \qty{40}{W}$; sebulannya:
$0.04 \times 720 = \qty{28.8}{kWh}$.
\textbf{6.} Totalnya $\approx \qty{242}{kWh}$. Yang menguasai: pemanas
airnya (\qty{150}{}), mesin cucinya (\qty{30}{}), dan lemari esnya
(\qty{28.8}{kWh}) --- yang memanaskan dan yang selalu menyala.
\textbf{7.} $242 \times 0.25 \approx 60.6$ \omterm{def:g6:measurement-in-science:measuring}{satuan}; bagian pemanasnya:
$150 \times 0.25 = 37.5$ --- jauh lebih dari separuh tagihannya.
\textbf{8.} $2150 + 242 = 2392$ \omterm{def:g6:measurement-in-science:measuring}{satuan}.
\textbf{9.} Baris pemanas yang baru: $2.5 \times 1.5 \times 30 =
\qty{112.5}{kWh}$: penghematannya \qty{37.5}{kWh}, kira-kira $9.4$
\omterm{def:g6:measurement-in-science:measuring}{satuan} mata uang tiap bulan.
\textbf{10.} Satu pengisi \omterm{def:g9:electric-power-energy:power}{daya} yang menganggur: $0.0002 \times 720 =
\qty{0.14}{kWh}$ --- kira-kira empat perseratus \omterm{def:g6:measurement-in-science:measuring}{satuan} mata uang:
usul satu mengalahkan penghematannya dua ratus kali lipat tiap
pengisi \omterm{def:g9:electric-power-energy:power}{dayanya}. Pemeriksaan menemukan \omterm{def:g3:levers-and-scales:lever}{tuas}; semboyan menemukan
pengisi \omterm{def:g9:electric-power-energy:power}{daya}.
\textbf{11.} Baris cereknya memaruh menjadi \qty{3.45}{kWh}:
penghematannya \qty{3.45}{kWh}, kurang dari satu \omterm{def:g6:measurement-in-science:measuring}{satuan} mata uang ---
kebiasaan neneknya benar, dan sederhana.
\textbf{12.} Misalnya: ``\omterm{def:g9:electric-power-energy:kwh}{Kilowatt-jamnya} berdiam di tempat airnya
menjadi panas dan di tempat mesinnya tidak pernah tidur. Baris
siaganya mengajarkan faktor waktunya: \omterm{def:g9:electric-power-energy:power}{daya} yang mungil menumbuhkan
tagihan ketika $t$-nya selalu. Dan perubahan yang paling menguntungkan
adalah mandi yang lebih singkat --- pemanasnya memiliki tagihannya,
sehingga \omterm{ex:g2:measuring-time:clock}{jamnya} memiliki penghematannya.''
\end{solution}
