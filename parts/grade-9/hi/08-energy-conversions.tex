\chapter{ऊर्जा के रूप और रूपांतरण}\label{ch:g9:energy-conversions}

\omterm{def:g5:everyday-energy:energy}{ऊर्जा} इस पुस्तक में यूँ बहती रही है जैसे किसी शहर के नीचे बहती नदी ---
हर मोड़ पर झलकती हुई, पर पूरी कभी नक़्शे पर नहीं उतरी। अब वह वक़्त आ
गया है। हर रूप को उसका नाम मिलता है, उनमें से एक को इस साल का दूसरा
महान सूत्र, और उन सब पर विज्ञान का सबसे ताक़तवर बहीखाता नियम राज करता
है: कुल कभी नहीं बदलता। कभी नहीं।

\section{सारे रूप, एक जगह}

\begin{definition}[ऊर्जा के रूप]\label{def:g9:energy-conversions:forms}
\omterm{def:g5:everyday-energy:energy}{ऊर्जा} के रूप, औपचारिक ढंग से पेश किए हुए:
\emph{\omterm{prop:g9:kinetic-energy-safety:formula}{गतिज ऊर्जा}} --- यानी गति का हिस्सा, $\frac12 m v^2$;
\emph{स्थितिज ऊर्जा}\index{स्थितिज ऊर्जा} --- जो जगह या बंदोबस्त से
जमा होती है (उठाई हुई, खिंची हुई, दबाई हुई चीज़ें); \emph{तापीय
\omterm{def:g5:everyday-energy:energy}{ऊर्जा}} --- वह बेतरतीब अणु-हिलावट जिसे \omterm{def:g3:temperature-thermometers:temperature}{ताप} मापता है;
\emph{रासायनिक \omterm{def:g5:everyday-energy:energy}{ऊर्जा}} --- जो पदार्थ के बंदोबस्तों में जमा है: भोजन,
ईंधन, बैटरियाँ; \emph{विद्युत \omterm{def:g5:everyday-energy:energy}{ऊर्जा}} --- यानी चलती क़तार की पहुँचाई
जाने वाली देन; \emph{विकिरण \omterm{def:g5:everyday-energy:energy}{ऊर्जा}} --- जिसे ख़ुद \omterm{def:g1:light-and-shadows:source}{प्रकाश} ढोता है, और जो
धूप का ढुलाई वाला रूप है; और \emph{नाभिकीय \omterm{def:g5:everyday-energy:energy}{ऊर्जा}} --- जो परमाणु की
क्रोड में बंद है, यानी इंसानों का खोला हुआ सबसे गहरा भंडार।
\end{definition}

\begin{proposition}[गुरुत्वीय स्थितिज ऊर्जा]\label{prop:g9:energy-conversions:ep}
$m$ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} (\unit{kg} में) को गुरुत्व की $g$ के ख़िलाफ़ $h$ ऊँचाई
(\unit{m} में) तक उठाने से उसमें यह \omterm{def:g9:energy-conversions:forms}{स्थितिज ऊर्जा} जमा हो जाती है:
\[
E_p = m \times g \times h
\]
जो \omterm{def:g9:kinetic-energy-safety:joule}{जूल} में है --- और नीचे लौटते हुए पूरी की पूरी वापस मिल जाती है।
उठाई हुई कुल्हाड़ी, पीछे खींचा गया झूला, पहाड़ी झील: सब अपने खाते में
$m g h$ रखे हैं। (यह वही सूत्र है जो सड़क-सुरक्षा वाले पोस्टरों के लिए
उधार लिया गया था --- अब आधिकारिक रूप से तुम्हारा।)
\end{proposition}

\begin{example}[महान सौदा: $E_p$ बनाम $E_k$]\label{ex:g9:energy-conversions:trade}
किसी गेंद को $h$ ऊँचाई से गिराओ: गिरते हुए उसका खाता \omterm{def:g9:kinetic-energy-safety:joule}{जूल} दर \omterm{def:g9:kinetic-energy-safety:joule}{जूल}
स्थितिज से गतिज में चला जाता है। ज़मीन पर पहुँचते-पहुँचते पूरी $m g h$
$\frac12 m v^2$ बन चुकी होती है --- दोनों को बराबर रखो और \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} कट
जाता है:
\[
v = \sqrt{2 g h},
\]
यानी इस साल का वर्गमूल अपनी रोटी कमाता हुआ। \qty{20}{m} से:
$v = \sqrt{2 \times 9.8 \times 20} \approx \qty{20}{m/s}$ ---
सत्तर किलोमीटर प्रति घंटा, \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} चाहे जो हो। लोलक यही सौदा दोनों
ओर हमेशा खेलता रहता है (\omterm{def:g2:air-around-us:air}{हवा} को दी जाने वाली एक फुसफुसाहट घटाकर); और
अर्ध-नली में स्केट चलाने वाला भी वैसे ही: ऊँचाई से \omterm{def:g7:motion-average-speed:formula}{चाल}, और \omterm{def:g7:motion-average-speed:formula}{चाल} से
ऊँचाई --- दोनों सूत्र एक ही जोड़ आपस में आगे-पीछे थमाते हुए।
\end{example}

\section{वह महान नियम}

\begin{proposition}[ऊर्जा संरक्षण]\label{prop:g9:energy-conversions:conservation}
अब तक जाँची गई हर प्रक्रिया में --- टक्करों और रसायन में, मशीनों और
तूफ़ानों में, तारों और कोशिकाओं में --- \omterm{def:g5:everyday-energy:energy}{ऊर्जा} अपना रूप और मालिक बदलती
है, पर \emph{कुल} ठीक-ठीक संरक्षित रहता है: न कुछ बनता है, न कुछ मिटता
है, और हर \omterm{def:g9:kinetic-energy-safety:joule}{जूल} का हिसाब रहता है। यह भौतिकी का सबसे गहरा बहीखाता नियम है:
इसी ने बचपन के अध्यायों वाली हमेशा चलने वाली मशीनें सेवामुक्त कीं, यही
इस पुस्तक की हर शृंखला का लेखा करता है, और तीन सदियों में किसी भी
प्रयोग ने इसे ग़लती करते नहीं पकड़ा। (क़ुदरत इस नियम को इतनी वफ़ादारी
से क्यों निभाती है, यह एक गहरा सवाल है जिसका जवाब बेहद ख़ूबसूरत है ---
और विश्वविद्यालय वाले सालों के नगीनों में से एक।)
\end{proposition}

\begin{method}[किसी भी प्रक्रिया का लेखा]\label{met:g9:energy-conversions:audit}
भौतिकशास्त्री का सार्वत्रिक बहीखाता:
\begin{enumerate}
  \item तंत्र की बाड़ लगाओ --- यानी लेखे के भीतर क्या-क्या है;
  \item पहले वाले ऊर्जा-खाते गिनाओ: हर रूप, हर भंडार;
  \item बाद वाले गिनाओ --- और उनमें तापीय रेज़गारी हमेशा शामिल रखो:
        घर्षण की गरमाहट, \omterm{def:g3:sound-around-us:sound}{ध्वनि} की फुसफुसाहट;
  \item बहीखाता बराबर करो: दोनों जोड़ मिलने ही चाहिए। कमी का मतलब है
        कोई रूप गिना नहीं गया --- रिसाव दोबारा ढूँढ़ो; वह आम तौर पर
        गरम होता है।
\end{enumerate}
\end{method}

\begin{example}[लेखे, करके दिखाए हुए]\label{ex:g9:energy-conversions:audits}
ब्रेक लगाती कार: गतिज खाता ख़ाली, और चक्रिकाओं पर तापीय खाता पूरा जमा
--- सेवामुक्ति, विनाश नहीं। धीरे-धीरे मरती उछलती गेंद: हर उछाल
गतिज में से थोड़ा गेंद और फ़र्श की गरमाहट में छील लेता है; बचपन वाला
जवाब, अब लेखा करने लायक। और चढ़ाई पर साइकिल चलाता तुम्हारा नाश्ता:
रासायनिक भंडार नीचे, स्थितिज खाता ऊपर (तुम्हारे और साइकिल के $m g h$
जितना), और तापीय हिस्सा विकिरित --- मांसपेशियाँ गरम चलती हैं। शुरुआती
\omterm{def:g5:everyday-energy:energy}{ऊर्जा} अध्यायों की हर पहेली इन्हीं तीन स्तंभों के नीचे बंद हो जाती है।
\end{example}

\section{दक्षता}

\begin{definition}[दक्षता]\label{def:g9:energy-conversions:efficiency}
किसी रूपांतरक की \emph{दक्षता}\index{दक्षता} उसकी भीतर आई \omterm{def:g5:everyday-energy:energy}{ऊर्जा} का वह
अंश है जो मनचाहे रूप में पहुँचाया जाता है:
\[
\text{दक्षता} =
\frac{\text{काम की निकली ऊर्जा}}{\text{कुल भीतर आई ऊर्जा}},
\]
यानी $0$ और $1$ के बीच का एक अंक (या उसका प्रतिशत)। बचा हुआ हिस्सा
नष्ट नहीं होता --- संरक्षण मना करता है --- बल्कि बिन माँगे रूपों में
भाग निकलता है, और लगभग हमेशा गरमाहट में।
\end{definition}

\begin{example}[रिपोर्ट कार्ड]\label{ex:g9:energy-conversions:reportcards}
दहकता पुरखा \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}: पाँच प्रतिशत \omterm{def:g1:light-and-shadows:source}{प्रकाश}, पंचानवे गरमाहट --- यानी शौक़
पाले हुए एक हीटर। उसका आधुनिक उत्तराधिकारी: तीस प्रतिशत से ऊपर, और
बाक़ी अब भी गरमाहट। कार का इंजन: क़रीब एक तिहाई काम की गति, और दो
तिहाई रेडिएटर तथा निकास से बाहर। बिजली की \omterm{ex:g6:circuits-and-safety:motor}{मोटर}: नब्बे से ऊपर। और वे
महान \omterm{def:g9:alternator:alternator}{प्रत्यावर्तित्र}: क़रीब निन्यानवे --- सभ्यता के सबसे बढ़िया
रूपांतरक। कोई भी ईमानदार रूपांतरक सौ तक नहीं पहुँचता: गरमाहट में
थोड़ा-सा रिसाव क़ुदरत का सार्वत्रिक कमीशन है।
\end{example}

\begin{remark}[``ऊर्जा बचाने'' का असल मतलब]\label{rem:g9:energy-conversions:saving}
अगर संरक्षण हर \omterm{def:g9:kinetic-energy-safety:joule}{जूल} की पहरेदारी करता है, तो किसी को ``\omterm{def:g5:everyday-energy:energy}{ऊर्जा} बचाने'' की
ज़रूरत ही क्यों पड़े? क्योंकि यह नियम \emph{मात्रा} बचाता है,
\emph{उपयोगिता} नहीं। सांद्र भंडार --- ईंधन, आवेश, ऊँचाई --- ख़र्च
होकर दुनिया भर में पतली-पतली फैली गरमाहट बन सकते हैं; और फैली हुई
गरमाहट, चाहे उसका हर \omterm{def:g9:kinetic-energy-safety:joule}{जूल} बचा हो, फिर केतलियाँ या रेलगाड़ियाँ नहीं चला
सकती: नदी ढलान उतर चुकी। सभ्यता की ऊर्जा-समस्या जूलों की कमी नहीं है
--- सूर्य हम पर अपनी ज़रूरत से हज़ारों गुना बरसाता है --- बल्कि
\emph{काम के} जूलों की सँभाल है। इसी विचार को सटीक बनाना भौतिकी की सबसे
मशहूर कहानियों में से एक है; वह विश्वविद्यालय वाले सालों में अपने
प्रसिद्ध नाम के साथ इंतज़ार कर रही है।
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:energy-conversions:6}
\omterm{def:g9:energy-conversions:efficiency}{दक्षता} की परिभाषा दो। कोई \omterm{ex:g6:circuits-and-safety:motor}{मोटर} \qty{200}{J} लेती है और \qty{170}{J}
गति पहुँचाती है: \omterm{def:g9:energy-conversions:efficiency}{दक्षता}, और बाक़ी का अंजाम?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:energy-conversions:7}
पुरखे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} को ``शौक़ पाले हुए हीटर'' कहना वाजिब क्यों था? दोनों
प्रतिशत बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:energy-conversions:8}
साइकिल सवार की चढ़ाई की पूरी रूपांतरण-शृंखला लिखो --- नाश्ते से चोटी
तक --- और हर रूप तथा हर रिसाव का नाम भी बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:energy-conversions:9}
बाँध का बहीखाता: \qty{1000}{kg} पानी टरबाइनों में से \qty{50}{m}
गिरता है। पहुँचाई गई $E_p$ निकालो; नब्बे प्रतिशत रूपांतरण पर प्रति टन
विद्युत \omterm{def:g5:everyday-energy:energy}{ऊर्जा} --- और \qty{4.4e6}{W} के निर्गम के लिए ज़रूरी टन प्रति
सेकंड भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:energy-conversions:10}
\qty{2}{m} से गिराई गई \qty{0.5}{kg} की गेंद उछलकर \qty{1.2}{m} तक
जाती है। उस उछाल का लेखा करो: पहले और बाद की ऊर्जाएँ, कतरन का नाप और
उसका ठिकाना --- और वह उछाल-ऊँचाई भी जिसके बाद किसी लेखे को न गतिज
मिलेगी, न \omterm{def:g9:energy-conversions:forms}{स्थितिज ऊर्जा}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:energy-conversions:11}
``लिफ़्ट विद्युत \omterm{def:g5:everyday-energy:energy}{ऊर्जा} को नब्बे प्रतिशत \omterm{def:g9:energy-conversions:efficiency}{दक्षता} पर \omterm{def:g9:energy-conversions:forms}{स्थितिज ऊर्जा} में
बदल देती है।'' \qty{500}{kg} के डिब्बे को \qty{30}{m} उठाने के लिए
ज़रूरी विद्युत \omterm{def:g5:everyday-energy:energy}{ऊर्जा} निकालो --- और उन फ़ालतू दस प्रतिशत को संरक्षण के
नियम से मिलाओ।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:energy-conversions:12}
अर्ध-नली वाला स्केट चलाने वाला बाईं कोर पर \qty{4}{m} की ऊँचाई से
ठहरे हुए शुरू करता है। नीचे पार करते वक़्त की \omterm{def:g7:motion-average-speed:formula}{चाल}, और दाईं ओर पहुँची
ऊँचाई की भविष्यवाणी करो --- आदर्श हालत में और असल में --- और गति तथा
उसकी \omterm{def:g5:everyday-energy:energy}{ऊर्जा} के लंबे वक़्त वाले अंजाम का ख़ाका भी खींचो। फिर
\cref{rem:g9:energy-conversions:saving} की मदद से समझाओ कि कोर पर दिया
गया कोई भी धक्का पूरा का पूरा वापस क्यों नहीं मिल सकता: दुनिया के
बहीखाते ने क्या कमाया, और स्केट वाले ने हमेशा के लिए क्या गँवाया?
\end{exercise}

\section{समस्या: मनोरंजन पार्क का आयोग}

\begin{problem}[{सप्ताहांत समस्या --- मनोरंजन पार्क किसी भौतिकी
सलाहकार को रखता है; झूला-गाड़ियाँ, मीनारें और वह बहीखाता जो झूठ बोल
ही नहीं सकता}]\label{pb:g9:energy-conversions:1}
पार्क के नए झूलों को खुलने के दिन से पहले तुम्हारे ऊर्जा-लेखे से पास
होना है। $g = \qty{9.8}{N/kg}$ लो; और घर्षण को तब तक अनदेखा करो जब तक
वह अनदेखा होने से इनकार न कर दे।

\medskip\noindent\textbf{भाग I --- महान गिरावट।}
\qty{2000}{kg} की झूला-गाड़ी को चरखी से \qty{45}{m} की चोटी तक खींचकर
ठहरे हुए छोड़ा जाता है।
\begin{enumerate}
  \item चोटी पर गाड़ी की $E_p$?
  \item \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} काटने वाले सूत्र से गिरावट के पेंदे पर उसकी आदर्श
        \omterm{def:g7:motion-average-speed:formula}{चाल} --- \unit{m/s} में और \unit{km/h} में?
  \item डिज़ाइनर उस गिरावट के ठीक बाद \qty{50}{m} की दूसरी पहाड़ी जोड़
        देते हैं। बहीखाते से इस पर वीटो लगाओ --- और वह सबसे ऊँची
        दूसरी पहाड़ी भी बताओ जिसे आदर्श गाड़ी पार कर सकती।
  \item पेंदे पर मापी गई \omterm{def:g7:motion-average-speed:formula}{चाल} आदर्श नहीं, \qty{28}{m/s} निकलती है।
        दोनों गतिज ऊर्जाएँ निकालो और उस अंतर को उसके ठीक खाते में
        डालो।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- चरखी और बिल।}
\begin{enumerate}[resume]
  \item चरखी गाड़ी को \qty{60}{s} में उसकी चोटी तक उठाती है। पूरी
        \omterm{def:g9:energy-conversions:efficiency}{दक्षता} पर उसकी \omterm{def:g9:electric-power-energy:power}{शक्ति}?
  \item असली चरखी \qty{20}{kW} खींचती है। उसकी \omterm{def:g9:energy-conversions:efficiency}{दक्षता}?
  \item हर सवारी हर चार मिनट पर, रोज़ दस घंटे चलती है। असली चरखी की
        रोज़ की \omterm{def:g5:everyday-energy:energy}{ऊर्जा} किलोवाट-घंटों में (उठान गिनकर), और $0.25$ प्रति
        \omterm{def:g6:measurement-in-science:measuring}{मात्रक} की दर पर उसकी लागत?
  \item एक डिज़ाइनर सुझाता है कि गाड़ी की पहुँचने वाली \omterm{def:g5:everyday-energy:energy}{ऊर्जा} किसी
        जनित्र-ब्रेक से वापस लेकर जाल को खिला दी जाए --- ``सवारी अपना
        ख़र्च ख़ुद निकाल लेगी!'' इस सपने का लेखा करो: हक़ीक़त में
        कितना हिस्सा लौट सकता है, और पूरा घेरा कौन-सा नियम मना कर देता
        है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- मुक्त पतन वाली मीनार और
रिपोर्ट।}
\qty{60}{m} की मीनार \qty{500}{kg} के डिब्बे को \qty{10}{m} पर लगे
चुंबकीय ब्रेकों में गिरा देती है।
\begin{enumerate}[resume]
  \item ब्रेक वाले इलाक़े में घुसते वक़्त की \omterm{def:g7:motion-average-speed:formula}{चाल} (यानी \qty{50}{m} का
        आदर्श पतन)?
  \item वह \omterm{def:g5:everyday-energy:energy}{ऊर्जा} जो ब्रेकों को सेवामुक्त करनी है, और प्रेरण उसे भेजता
        कहाँ है। (ये ब्रेक \omterm{def:g9:alternator:alternator}{प्रत्यावर्तित्र} वाले अध्याय के चचेरे भाई
        हैं --- यानी धातु को गरम करती प्रेरित धाराओं की भँवरें।)
  \item सवार चीख़ते हैं कि गिरने के दौरान वे ``भारहीन'' थे।
        \omterm{def:g9:gravitation:gravitation}{गुरुत्वाकर्षण} वाले अध्याय के साथ-साथ गिरने वाले साथियों के
        विचार से इस दावे पर एक वाक्य में फ़ैसला दो।
  \item आयोग की रिपोर्ट पर दस्तख़त करो: तीन वाक्य --- वह सौदा जिस पर
        झूला-गाड़ी जीती है, वह नियम जिसे हर झूले ने माना, और वह खाता
        जिसने चुपचाप उन सब पर चुंगी लगाई।
\end{enumerate}
\end{problem}
