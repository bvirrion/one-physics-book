\chapter{गतिज ऊर्जा और सड़क सुरक्षा}\label{ch:g9:kinetic-energy-safety}

बचपन के चाबी वाले खिलौनों से लेकर अब तक \omterm{def:g5:everyday-energy:energy}{ऊर्जा} भंडारों और रूपों की
कहानी रही है --- जीवंत, काम की, और बिना अंकों के। वह इंतज़ार अब ख़त्म
होता है। गति की \omterm{def:g5:everyday-energy:energy}{ऊर्जा} को उसका सूत्र मिलता है, \omterm{def:g5:everyday-energy:energy}{ऊर्जा} को उसका \omterm{def:g6:measurement-in-science:measuring}{मात्रक},
और यह अंकगणित ज़िंदगी और मौत के एक मामले पर हुक्म चलाता निकलता है: कि
दोगुनी \omterm{def:g7:motion-average-speed:formula}{चाल} वाली कार को रोकना चार गुना मुश्किल क्यों है, और बेध्यानी का
एक अकेला सेकंड कितने \omterm{def:g2:measuring-length:units}{मीटर} वसूल कर लेता है।

\section{गति की ऊर्जा}

\begin{definition}[जूल]\label{def:g9:kinetic-energy-safety:joule}
\omterm{def:g5:everyday-energy:energy}{ऊर्जा} --- उसका हर रूप --- \emph{जूल}\index{जूल} (\unit{J}) में मापी
जाती है, जो उस प्रयोगकर्ता के सम्मान में है जिसने साबित किया कि \omterm{def:g5:heat-and-insulation:heat}{ऊष्मा}
भी \omterm{def:g5:everyday-energy:energy}{ऊर्जा} का ही एक रूप है। लंगर: इस किताब को फ़र्श से ताक तक उठाने में
कुछ जूल ख़र्च होते हैं; धड़कते दिल की हर धड़कन क़रीब एक जूल; और
चॉकलेट की एक पट्टी दस लाख जूल भोजन-ऊर्जा जमा किए रहती है।
\end{definition}

\begin{proposition}[गतिज ऊर्जा]\label{prop:g9:kinetic-energy-safety:formula}
$m$ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} (\unit{kg} में) वाला कोई पिंड $v$ \omterm{def:g7:motion-average-speed:formula}{चाल} (\unit{m/s} में)
से चलते हुए, सिर्फ़ अपनी गति के नाते, यह \emph{गतिज ऊर्जा}\index{गतिज
ऊर्जा} ढोता है:
\[
E_k = \frac{1}{2} \times m \times v^2 ,
\]
जो \omterm{def:g9:kinetic-energy-safety:joule}{जूल} में है। दो घुंडियाँ, और असमान ताक़तें: \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} दोगुना करने से
$E_k$ दोगुनी होती है --- पर \emph{\omterm{def:g7:motion-average-speed:formula}{चाल}} दोगुनी करने से वह
\emph{चौगुनी} हो जाती है, क्योंकि \omterm{def:g7:motion-average-speed:formula}{चाल} वर्ग होकर भीतर घुसती है। ख़तरनाक
घुंडी \omterm{def:g7:motion-average-speed:formula}{चाल} ही है।
\end{proposition}

\begin{example}[पहले हिसाब]\label{ex:g9:kinetic-energy-safety:first}
\qty{20}{m/s} पर \qty{0.45}{kg} की फ़ुटबॉल: $E_k = 0.5 \times 0.45
\times 20^2 = \qty{90}{J}$. \qty{10}{m/s} पर \qty{60}{kg} का धावक:
$0.5 \times 60 \times 100 = \qty{3000}{J}$. \qty{50}{km/h} पर
\qty{1000}{kg} की कार --- पहले बदलो: $50 \div 3.6 \approx
\qty{13.9}{m/s}$ --- $E_k = 0.5 \times 1000 \times 13.9^2 \approx
\qty{9.7e4}{J}$: यानी लगभग एक लाख \omterm{def:g9:kinetic-energy-safety:joule}{जूल}। वही कार \qty{100}{km/h} पर:
चार गुना --- \qty{3.9e5}{J}. सूत्र की चेतावनी, अंकों में।
\end{example}

\begin{example}[ब्रेक उसे भेजते कहाँ हैं]\label{ex:g9:kinetic-energy-safety:brakes}
रुकने के लिए कार को अपनी पूरी $E_k$ किसी न किसी को थमानी ही पड़ती है
--- \omterm{def:g5:everyday-energy:energy}{ऊर्जा} आगे बढ़ाई जाती है, मिटाई कभी नहीं, जैसा बचपन की शृंखलाएँ
ज़ोर देकर कहती थीं। ब्रेकों की पकड़ उसे \omterm{def:g5:heat-and-insulation:heat}{ऊष्मा} में बदल देती है: पहाड़ी
ढलानों के बाद चक्रिकाएँ नारंगी दमकती हैं, और गरम ब्रेकों की गंध दरअसल
\omterm{prop:g9:kinetic-energy-safety:formula}{गतिज ऊर्जा} का सेवामुक्त होना \emph{है}। और किसी टक्कर में वह सौंपना
हिंसक तथा तात्कालिक होता है --- सीधे मुड़-तुड़ गए इस्पात में। नीचे आने
वाला सड़क-सुरक्षा का हर अंक यही बहीखाता है: $E_k$ जितनी बड़ी, उसकी
सेवामुक्ति उतनी ही लंबी, या उतनी ही भद्दी।
\end{example}

\section{रुकने की दूरी}

\begin{proposition}[रुकना $=$ सोचना $+$ ब्रेक लगाना]\label{prop:g9:kinetic-energy-safety:stopping}
ख़तरा भाँप लेने वाला चालक भी सड़क के दो हिस्से पार करने के बाद ही रुक
पाता है:
\begin{enumerate}
  \item \emph{प्रतिक्रिया दूरी}\index{प्रतिक्रिया दूरी}: वह सड़क, जो
        दिमाग़ के ताड़ने और पैर के हिलने तक तय हो जाती है ---
        बिना बदली \omterm{def:g7:motion-average-speed:formula}{चाल} पर क़रीब एक पूरा सेकंड, इसलिए यह हिस्सा $v$ के
        \emph{समानुपात} में बढ़ता है;
  \item \emph{ब्रेक-दूरी}\index{ब्रेक-दूरी}: वह सड़क, जो ब्रेकों को
        \omterm{prop:g9:kinetic-energy-safety:formula}{गतिज ऊर्जा} सेवामुक्त करने के लिए चाहिए --- और चूँकि $E_k$
        $v^2$ ढोती है, यह हिस्सा \omterm{def:g7:motion-average-speed:formula}{चाल} के \emph{वर्ग} के साथ बढ़ता है:
        \omterm{def:g7:motion-average-speed:formula}{चाल} दोगुनी, ब्रेक की सड़क चौगुनी।
\end{enumerate}
इन दोनों का जोड़, यानी \emph{रुकने की दूरी}, वही अंक है जिसका सामना
गेंद के पीछे भागते बच्चे से होता है।
\end{proposition}

\begin{example}[वह सारणी जो हर चालक के पास होनी चाहिए]\label{ex:g9:kinetic-energy-safety:table}
सूखी सड़क, चौकन्ना चालक (एक सेकंड की प्रतिक्रिया; अच्छे ब्रेक):
\begin{center}
\begin{tabular}{l|c|c|c}
\omterm{def:g7:motion-average-speed:formula}{चाल} & प्रतिक्रिया & ब्रेक & रुकना \\
\hline
\qty{50}{km/h} & \qty{14}{m} & \qty{13}{m} & \qty{27}{m} \\
\qty{90}{km/h} & \qty{25}{m} & \qty{40}{m} & \qty{65}{m} \\
\qty{130}{km/h} & \qty{36}{m} & \qty{85}{m} & \qty{121}{m}
\end{tabular}
\end{center}
इसे दो बार पढ़ो। शहर की \omterm{def:g7:motion-average-speed:formula}{चाल} पर डेढ़ बस भर; और राजमार्ग की \omterm{def:g7:motion-average-speed:formula}{चाल} पर एक
पूरा दौड़-पथ और उसका पाँचवाँ हिस्सा। और अनुपात दोनों नियमों की पुष्टि
कर देते हैं: प्रतिक्रिया $v$ की तरह बढ़ती है, और ब्रेक $v^2$ की तरह
--- इसलिए तेज़ \omterm{def:g7:motion-average-speed:formula}{चाल} पर ब्रेक ही पूरा जोड़ हड़प लेते हैं।
\end{example}

\begin{method}[रुकने की दूरी का अंदाज़ा लगाना]\label{met:g9:kinetic-energy-safety:estimate}
\begin{enumerate}
  \item \omterm{def:g7:motion-average-speed:formula}{चाल} को \unit{m/s} में बदलो (\unit{km/h} को $3.6$ से भाग दो);
  \item \omterm{prop:g9:kinetic-energy-safety:stopping}{प्रतिक्रिया दूरी}: एक सेकंड का सफ़र --- यानी वही \unit{m/s}
        वाला अंक, मीटरों में;
  \item \omterm{prop:g9:kinetic-energy-safety:stopping}{ब्रेक-दूरी}: किसी जाने-पहचाने लंगर को वर्ग के हिसाब से बढ़ाओ
        --- \qty{50}{km/h} पर \qty{13}{m} से शुरू करके $(v/50)^2$ से
        गुणा करो;
  \item जोड़ो, और आगे की सड़क सचमुच जितनी पेश कर रही हो उससे मिलाकर
        देखो।
\end{enumerate}
गीली सड़कें ब्रेक वाला हिस्सा दोगुना कर देती हैं; और थके या बँटे हुए
चालक प्रतिक्रिया वाले सेकंड को दो की ओर खींच देते हैं --- ईमानदार
आँकड़ों के साथ जोड़ दोबारा चलाओ।
\end{method}

\begin{example}[एक नज़र की क़ीमत]\label{ex:g9:kinetic-energy-safety:glance}
\qty{90}{km/h} पर फ़ोन की ओर दो सेकंड की एक नज़र: कार $2 \times 25 =
\qty{50}{m}$ तय कर जाती है --- आधा फ़ुटबॉल मैदान --- और वह भी
आँख मूँदकर, इससे पहले कि प्रतिक्रिया वाला सेकंड शुरू भी हो। शहर की \omterm{def:g7:motion-average-speed:formula}{चाल}
पर वही नज़र आँख मूँदकर \qty{28}{m} ख़र्च कर देती है: चौराहे के पार,
स्कूल के फाटक के पार। कार का सबसे ख़तरनाक पुरज़ा यहाँ के किसी सूत्र से
नहीं मापा जाता: वह है चालक का ध्यान।
\end{example}

\begin{remark}[पेटियाँ, हवा-थैले और हेलमेट]\label{rem:g9:kinetic-energy-safety:belts}
किसी टक्कर में कार एक \omterm{def:g2:measuring-length:units}{मीटर} भर मुड़-तुड़कर रुक जाती है --- पर बिना पेटी
वाला सवार पूरी \omterm{def:g7:motion-average-speed:formula}{चाल} पर चलता रहता है (पिछले अध्याय का वह शांत वाक्य, यहाँ
भयावह ढंग से लागू) जब तक उसे जो भी सामने आए वही न रोक दे: डैशबोर्ड,
सामने का शीशा, सड़क। सीट-पेटियाँ और हवा-थैले शरीर को ज़्यादा लंबी दूरी
और ज़्यादा लंबे समय में रोकते हैं, और उसकी \omterm{prop:g9:kinetic-energy-safety:formula}{गतिज ऊर्जा} एक ही झटके के
बजाय नरमी से सेवामुक्त करते हैं; और साइकिल सवार का हेलमेट खोपड़ी की
अनमोल सामग्री के लिए यही करता है, जिसका कुचला जा सकने वाला फ़ोम नरमी से
रुकने के कुछ \omterm{def:g2:measuring-length:units}{सेंटीमीटर} ख़रीद लेता है। इनमें से कोई तुम्हारी $E_k$ एक
\omterm{def:g9:kinetic-energy-safety:joule}{जूल} भी कम नहीं करता --- ये सब बस उसकी सेवामुक्ति को सभ्य बना देते हैं।
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:2}

के तौर पर जोड़ना, या उसकी \omterm{def:g7:motion-average-speed:formula}{चाल} आधी और बढ़ा देना? दोनों गुणक दिखाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:4}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:5}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:6}

और बचपन वाले उस नियम का भी जो उसे यूँ ही ग़ायब होने से मना करता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:7}

निकालो (चौकन्ना एक सेकंड)। जवाब किस सीधे-सादे नमूने पर क़दम मिलाते

\end{exercise}

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:kinetic-energy-safety:10}
नगर परिषद स्कूल वाले इलाक़ों की सीमा \qty{50}{} से घटाकर
\qty{30}{km/h} कर देती है। दोनों रुकने की दूरियों की तुलना करो (लंगर:
$50$ पर \qty{13}{m} की \omterm{prop:g9:kinetic-energy-safety:stopping}{ब्रेक-दूरी}; एक सेकंड की प्रतिक्रिया) --- और
दोनों गतिज ऊर्जाओं की भी। पोस्टर पर तुम्हें कौन-सी तुलना ज़्यादा असरदार
लगती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:kinetic-energy-safety:11}
बारिश ब्रेक-दूरियाँ दोगुनी कर देती है। चालक की सारणी का रुकने वाला
स्तंभ गीली सड़कों के लिए $50$ और \qty{90}{km/h} पर दोबारा बनाओ ---
तुमने कौन-सा हिस्सा दोगुना किया, कौन-सा नहीं, और क्यों?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:kinetic-energy-safety:12}
\qty{20000}{kg} का एक ट्रक \qty{90}{km/h} पर लुढ़क रहा है। उसकी $E_k$
वैज्ञानिक संकेतन में निकालो; वह \omterm{def:g7:motion-average-speed:formula}{चाल} ढूँढ़ो जिस पर \qty{1000}{kg} की कार
उसी \omterm{def:g5:everyday-energy:energy}{ऊर्जा} की बराबरी कर लेती (दोनों सूत्र बराबर रखो --- इस साल बीजगणित
तुम्हारा अपना है); और यह नतीजा निकालो कि पहाड़ी सड़कों पर बेक़ाबू ट्रकों
के लिए बचाव-ढलानें क्यों होती हैं जबकि कारों के लिए ऐसी कोई ढलान नहीं
होती।
\end{exercise}

\section{समस्या: सुरक्षा अभियान}

\begin{problem}[{सप्ताहांत समस्या --- कक्षा क़स्बे का सड़क-सुरक्षा
अभियान बनाती है; सूत्र से जाँचे गए पोस्टर; महापौर के मुश्किल
सवाल}]\label{pb:g9:kinetic-energy-safety:1}
क़स्बा एक अभियान का काम सौंपता है, एक ही शर्त पर: हर पोस्टर का हर अंक
भौतिकी की जाँच से बचकर निकलना चाहिए। \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} हिसाब लगाती है। (लंगर:
प्रतिक्रिया एक सेकंड; \qty{50}{km/h} पर \qty{13}{m} की \omterm{prop:g9:kinetic-energy-safety:stopping}{ब्रेक-दूरी}, जो
वर्ग के हिसाब से बढ़ती है; कार का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \qty{1000}{kg}.)

\medskip\noindent\textbf{भाग I --- पहला पोस्टर: ``50, 60 नहीं''।}
\begin{enumerate}
  \item दोनों चालें \unit{m/s} में निकालो (एक दशमलव अंक)।
  \item दोनों चालों पर प्रतिक्रिया दूरियाँ?
  \item दोनों पर ब्रेक-दूरियाँ (लंगर को बढ़ाकर)?
  \item पोस्टर की मुख्य पंक्ति: ``\omterm{def:g7:motion-average-speed:formula}{चाल} के दस \omterm{def:g6:measurement-in-science:measuring}{मात्रक} और --- तो रुकने के
        कितने \omterm{def:g2:measuring-length:units}{मीटर} और?'' वह अंक भरो।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दूसरा पोस्टर: ``\omterm{def:g7:motion-average-speed:formula}{चाल} की दीवार''।}
डिज़ाइन टक्कर की ऊर्जाओं को गिरने की ऊँचाइयों के तौर पर दिखाता है:
यानी $v$ \omterm{def:g7:motion-average-speed:formula}{चाल} पर हुई टक्कर उस ऊँचाई से गिरने के ``बराबर'' है जिससे उतनी
ही \omterm{def:g5:everyday-energy:energy}{ऊर्जा} कमाई जाती।
\begin{enumerate}[resume]
  \item \qty{50}{km/h} पर (भाग I से) और \qty{90}{km/h} पर कार की
        $E_k$ निकालो।
  \item कलाकारों को ऊँचाइयाँ चाहिए: इस कार के लिए \qty{1}{m} से गिरने
        पर क़रीब \qty{9.8}{kJ} कमाई जाती है ($m g h$ --- यानी अगले
        अध्याय का दूसरा सूत्र, पहले ही उधार लिया हुआ)। टक्कर की ये
        दोनों ऊर्जाएँ गिरने की कौन-सी ऊँचाइयों से मेल खाती हैं? (भाग दो;
        \omterm{def:g2:measuring-length:units}{मीटर} तक गोल करो।)
  \item रोज़मर्रा की कौन-सी इमारतें उन ऊँचाइयों से मेल खाती हैं? पोस्टर
        की पंक्ति का मसौदा बनाओ।
  \item महापौर एतराज़ करते हैं: ``कोई दीवारों में गाड़ी नहीं ठोकता;
        कारें मुड़-तुड़ जाती हैं, पेटियाँ थाम लेती हैं।'' पोस्टर की
        भौतिकी का बचाव करो और साथ ही उनकी बात भी मानो --- मुड़ने वाले
        हिस्से और पेटियाँ क्या बदलती हैं, और कौन-सा अंक वे नहीं
        बदलतीं?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- तीसरा पोस्टर: ``एक सेकंड''।}
\begin{enumerate}[resume]
  \item \qty{90}{km/h} पर बेध्यानी का एक सेकंड ब्रेक लगने से पहले
        कितने \omterm{def:g2:measuring-length:units}{मीटर} वसूल लेता है? और फ़ोन की दो सेकंड की एक नज़र?
  \item थके चालक वाला रूप: प्रतिक्रिया खिंचकर दो सेकंड। \qty{90}{km/h}
        पर रुकने की पूरी दूरी दोबारा निकालो और चौकन्ने चालक की
        \qty{65}{m} से तुलना करो।
  \item एक शक़ी पार्षद: ``ब्रेक वाले उन बड़े अंकों के बग़ल एक सेकंड
        से भला क्या फ़र्क़ पड़ता होगा।'' वह किन चालों पर सबसे ज़्यादा
        ग़लत है --- कम या ज़्यादा? (दोनों हिस्सों के बढ़ने के नियमों की
        तुलना करो।)
  \item अभियान पर दस्तख़त करो: पोस्टरों के नीचे लिखी तीन पंक्तियाँ, हर
        पोस्टर पर एक, और हर एक में उसका सूत्र सादे शब्दों में।
\end{enumerate}
\end{problem}
