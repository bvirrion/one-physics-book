\chapter{गुरुत्वाकर्षण}\label{ch:g9:gravitation}

एक सेब अपनी डाल छोड़ता है और गिर जाता है। उसी बग़ीचे के ऊपर बैठा
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} नहीं गिरता। हज़ारों साल तक ये दो अलग-अलग तथ्य लगते रहे --- जब
तक प्रतिभा की एक कौंध ने उन्हें जोड़ नहीं दिया: \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} गिर \emph{रहा}
है, लगातार, पृथ्वी के गिर्द, और उसे थामे हुए वही खिंचाव है जो सेब पर
अपना दावा ठोकता है। बग़ीचे और आकाश के लिए एक ही नियम --- उस दिन तक लिखा
गया सबसे साहसी समीकरण, और इस पुस्तक का तुम्हारा आख़िरी साल उसी से खुलता
है।

\section{सार्वत्रिक आकर्षण}

\begin{definition}[गुरुत्वाकर्षण]\label{def:g9:gravitation:gravitation}
\emph{गुरुत्वाकर्षण}\index{गुरुत्वाकर्षण} सारे द्रव्यमानों के बीच का
आपसी \omterm{def:g1:magnets:magnet}{आकर्षण} है: ब्रह्मांड में पदार्थ का हर हिस्सा हर दूसरे हिस्से को
खींचता है। यह वही बिना-छुए वाला \omterm{def:g2:pushes-and-pulls:force}{बल} है जिससे बरसों पहले ``पृथ्वी का
खिंचाव'' कहकर तुम्हारी मुलाक़ात हुई थी --- और जो अब सार्वत्रिक होकर
सामने आता है: पृथ्वी सेब को खींचती है, सेब पृथ्वी को, \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} समुद्रों
को, सूर्य ग्रहों को, और ताक पर रखी दो किताबें भी एक-दूसरे को खींचती हैं
--- कमज़ोरी से, पर हमेशा।
\end{definition}

\begin{proposition}[सार्वत्रिक गुरुत्वाकर्षण का नियम]\label{prop:g9:gravitation:law}
$m_A$ और $m_B$ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} वाले दो पिंड (\unit{kg} में), जिनके केंद्र
$d$ दूरी पर हों (\unit{m} में), एक-दूसरे को उन्हें जोड़ने वाली रेखा के
साथ-साथ बराबर और उलटे बलों से खींचते हैं, जिनकी ताक़त है
\[
F = G \times \frac{m_A \times m_B}{d^2},
\]
जहाँ $F$ न्यूटन (\unit{N}) में है --- यानी \omterm{def:g2:pushes-and-pulls:force}{बल} का वह \omterm{def:g6:measurement-in-science:measuring}{मात्रक} जिसकी पूरी
कहानी अगले अध्याय से शुरू होती है --- और
$G = \qty{6.67e-11}{N.m^2/kg^2}$ \emph{गुरुत्वीय स्थिरांक} है, जो
ब्रह्मांड में हर जगह वही अंक रहता है। यह \omterm{def:g2:pushes-and-pulls:force}{बल} हर \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} के साथ बढ़ता
है, और दूरी के \emph{वर्ग} के साथ मरता है: दोगुनी दूर, चार गुना
कमज़ोर; दस गुना दूर, सौ गुना कमज़ोर।
\end{proposition}

\begin{example}[नियम को पढ़ना]\label{ex:g9:gravitation:reading}
इस सूत्र की हर ख़ासियत कुछ कहती है। \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} गुणा होते हैं: दोनों में
से किसी एक साथी को दोगुना कर दो, \omterm{def:g2:pushes-and-pulls:force}{बल} दोगुना। दूरी हर में वर्ग होकर
बैठती है: गुरुत्व एक तेज़ी से फीकी पड़ती फुसफुसाहट है --- वही
प्रतिलोम-वर्ग वाली तनुता, जो तुम्हें \omterm{def:g1:light-and-shadows:source}{प्रकाश} और \omterm{def:g3:sound-around-us:sound}{ध्वनि} के साथ भी दोबारा
मिलेगी, क्योंकि अंतरिक्ष में फैलती हर चीज़ की ज्यामिति यही है। और $G$
का नन्हा-सा नाप --- दशमलव के बाद ग्यारह शून्य --- ऐलान करता है कि
\omterm{def:g9:gravitation:gravitation}{गुरुत्वाकर्षण} क़ुदरत के बलों का कमज़ोर सदस्य है: वह ब्रह्मांड पर सिर्फ़
इसलिए राज करता है कि वहाँ ऊपर \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} राक्षसी हैं और उसे काटने वाला
कुछ भी नहीं।
\end{example}

\begin{example}[ताक की किताबें आपस में क्यों नहीं भिड़ जातीं]\label{ex:g9:gravitation:books}
\qty{1}{kg} की दो किताबें, जिनके केंद्र \qty{0.1}{m} दूर हैं:
\[
F = \num{6.67e-11} \times \frac{1 \times 1}{0.1^2}
= \qty{6.67e-9}{N}
\]
--- यानी न्यूटन के कुछ अरबवें हिस्से, जो धूल के किसी कण के कण के \omterm{def:g4:weight-and-mass:weight}{भार}
जितना है। अब पृथ्वी और सेब: एक किताब की जगह \qty{5.97e24}{kg} का \omterm{def:g4:solar-system:planet}{ग्रह}
रख दो और दूरी की जगह उसकी \qty{6.4e6}{m} की त्रिज्या, और वही नियम
बग़ीचे की दोपहर वाला जाना-पहचाना खिंचाव थमा देता है। सार्वत्रिक का
मतलब सार्वत्रिक है --- बस \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} तय करते हैं कि किसे पता चलेगा।
\end{example}

\section{गिरता हुआ चंद्रमा}

\begin{example}[पहाड़ पर रखी तोप]\label{ex:g9:gravitation:cannon}
उसी कौंध का अपना विचार-प्रयोग। किसी ऊँची चोटी से तोप का गोला आड़ा
दाग़ो: वह चाप बनाकर गिर पड़ता है। तेज़ दाग़ो: वह और दूर गिरता है ---
और इस पूरे वक़्त पृथ्वी का खिंचाव उसका रास्ता मोड़ता रहता है। अब कल्पना
करो कि उसे इतना तेज़ दाग़ा जाए कि गिरते-गिरते गोल पृथ्वी की सतह
\emph{उसके नीचे उतनी ही तेज़ी से मुड़ती चली जाए}: तब गोला बिना रुके
गिरता रहेगा और उतरेगा कभी नहीं --- वह \omterm{def:g4:solar-system:planet}{ग्रह} का चक्कर लगाने लगेगा। इस
लगातार गिरते रहने का एक जाना-पहचाना नाम है: \emph{\omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा}}। \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} में
घूमते किसी पिंड को कोई ``ऊपर'' नहीं थामे रखता; वह बस शान से गिर रहा
होता है।
\end{example}

\begin{proposition}[गुरुत्वाकर्षण क्या-क्या चलाता है]\label{prop:g9:gravitation:runs}
एक नियम, पूरा आकाश: \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पृथ्वी का सनातन तोप-गोला है, जो महीने में
एक बार हमारे गिर्द गिरता है; \omterm{def:g4:solar-system:planet}{ग्रह} वैसे ही सूर्य के गिर्द गिरते हैं ---
और पास वाले तेज़, जैसा प्रतिलोम वर्ग माँगता है; धूमकेतु अँधेरे से लंबी
गिरती हुई तोरणों पर भीतर की ओर झपटते हैं; और हर बनावटी उपग्रह ---
मौसम की आँखें, दिशा बताने वाले प्रकाश-स्तंभ, हर नब्बे मिनट में चक्कर
लगाता चालक दल वाला अंतरिक्ष केंद्र --- पहाड़ी तोप के उसी गणित पर सवार
है। आकाश में कुछ भी लटका हुआ नहीं है। हर चीज़ गिरती है, और चूक जाती है।
\end{proposition}

\begin{example}[ज्वार-भाटा: चंद्रमा भी खींचता है]\label{ex:g9:gravitation:tides}
आपसीपन के दस्तख़त पानी पर लिखे हैं। \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का खिंचाव पृथ्वी पर सबसे
तगड़ा उस महासागर पर पड़ता है जो उसके सामने हो, और सबसे कमज़ोर दूसरी ओर
--- इसलिए समुद्र एक साथ \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} की ओर और उससे दूर की ओर, दोनों ओर
हौले-हौले ढेर हो जाते हैं; और उन ढेरों के नीचे पृथ्वी के घूमते जाने से
किनारों को पानी चढ़ता-उतरता दिखता है: यानी \emph{ज्वार-भाटा}, ज़्यादातर
दिनों में दो बार। सूर्य भी अपना हाथ जोड़ता है --- अमावस और पूर्णिमा पर
एक क़तार में आकर, जिससे बड़े ज्वार आते हैं। मछुआरे हज़ारों साल से
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} को पढ़ते आए हैं; और यह नियम उसे उन्हें पढ़कर सुना देता है।
\end{example}

\begin{remark}[नियम जो नहीं कहता]\label{rem:g9:gravitation:notsay}
वह महान समीकरण \omterm{def:g2:pushes-and-pulls:force}{बल} का हिसाब बेहद बारीकी से लगा देता है और इस बात की एक
रत्ती भी नहीं समझाता कि दो पिंड ख़ाली शून्य के पार एक-दूसरे को थामते
\emph{कैसे} हैं --- ख़ुद उसके लेखक ने इस सवाल को खुला कहा और शानदार ढंग
से बिना जवाब छोड़ दिया। गहरे जवाब मौजूद हैं --- इस पाठ्यक्रम के
विश्वविद्यालय वाले साल आधुनिक जवाब तक पहुँचते हैं, जहाँ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} उसी
ज्यामिति को मोड़ देता है जिसमें वह बसता है --- पर तब से हर ईमानदार
भौतिकशास्त्री ने वही सबक़ दोहराया है जो तुम इस पन्ने से ले सकते हो:
कोई नियम पूरी तरह काम का, पूरी तरह जाँचा हुआ हो सकता है, और फिर भी
अपना आख़िरी भेद अपने पास रख सकता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:1}
सार्वत्रिक \omterm{def:g9:gravitation:gravitation}{गुरुत्वाकर्षण} का नियम बताओ --- सूत्र, \omterm{def:g6:measurement-in-science:measuring}{मात्रक}, और दोनों
ख़ासियतें (\omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, दूरी) शब्दों में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:2}
पृथ्वी सेब को \qty{1}{N} से खींचती है। सेब पृथ्वी को कितने \omterm{def:g2:pushes-and-pulls:force}{बल} से
खींचता है? दोनों में से सिर्फ़ एक का ही त्वरण दिखाई क्यों देता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:3}
दो उपग्रह पृथ्वी के केंद्र से $d$ और $2d$ दूरी पर \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} में हैं। उन
पर लगते गुरुत्वीय बलों की तुलना करो (\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बराबर)। और $10 d$ पर?
\end{exercise}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:5}

ठीक-ठीक मेल खाती हैं? \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} क्या है, तीन शब्दों में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:6}
``अंतरिक्ष-यात्री इसलिए तैरते हैं कि वहाँ ऊपर गुरुत्व है ही नहीं।''
इसे तोप के गोले से ढहाओ: केंद्र और उसका चालक दल, दोनों क्या कर रहे
हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:7}
क़ुदरत का कमज़ोर सदस्य होने के बावजूद \omterm{def:g9:gravitation:gravitation}{गुरुत्वाकर्षण} ब्रह्मांड पर हुक्म
क्यों चलाता है? अध्याय से दो वजहें।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:8}
ज्वार-भाटा आता कहाँ से है, और ज़्यादातर दिनों में दो बार क्यों? सबसे
बड़े कब आते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:gravitation:9}
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} (\qty{7.3e22}{kg}) पृथ्वी (\qty{5.97e24}{kg}) से
\qty{3.8e8}{m} दूर \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} में है। उनके बीच का \omterm{def:g1:magnets:magnet}{आकर्षण} निकालो, शुरू से
आख़िर तक वैज्ञानिक संकेतन में --- और जवाब की दस की घात कितने अंक ढोती
है, इसे निहारो भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:gravitation:10}
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} की सतह पर तुम्हारा \omterm{def:g4:weight-and-mass:weight}{भार} सिमटकर छठा हिस्सा रह गया था (उछलते
अंतरिक्ष-यात्री याद करो)। नियम की दोनों घुंडियों से --- \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का
छोटा \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, और उसकी छोटी त्रिज्या --- समझाओ कि पृथ्वी से अस्सी
गुना हलके पिंड से छठा हिस्सा कैसे निकल आता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:gravitation:11}
बुध सूर्य का चक्कर $88$ दिन में पूरा करता है, और वरुण $165$ साल में।
इस नमूने को प्रतिलोम वर्ग से पढ़ो: दूरी के साथ गुरुत्व का फीका पड़ना
सूर्य की पकड़ के साथ क्या करता है, और इस तरह बाहरी कुनबे की रफ़्तार के
साथ?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:gravitation:12}
भूस्थिर उपग्रह भूमध्य रेखा के किसी एक बिंदु के ऊपर देखने में बिना हिले
टँगा रहता है --- तश्तरी वाले एंटेना कभी हिलते ही नहीं। इस दिखावे को
\cref{prop:g9:gravitation:runs} से मिलाओ: उसका कक्षीय आवर्तकाल क्या
होना चाहिए, उसे किस ओर चक्कर लगाना चाहिए, और यहाँ ``टँगे रहने'' का
मतलब ``घूमती पृथ्वी के साथ ठीक क़दम मिलाकर गिरते रहना'' क्यों है?
\end{exercise}

\section{समस्या: तोप के गोले की विरासत}

\begin{problem}[{सप्ताहांत समस्या --- बग़ीचे से कक्षा तक; एक ही नियम से
सेब, तोप और चंद्रमा का लेखा; उपग्रह प्रक्षेपण का
पटल}]\label{pb:g9:gravitation:1}
अंतरिक्ष एजेंसी की जनसंपर्क मेज़ पर एक शाम: एक नियम,
$G = \qty{6.67e-11}{N.m^2/kg^2}$, पृथ्वी का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}
\qty{5.97e24}{kg}, और पृथ्वी की त्रिज्या \qty{6.4e6}{m}।

\medskip\noindent\textbf{भाग I --- सेब का लेखा।}
\begin{enumerate}
  \item सतह पर रखे \qty{0.1}{kg} के सेब पर पृथ्वी का खिंचाव निकालो
        (केंद्रों के बीच एक पृथ्वी-त्रिज्या का फ़ासला)।
  \item सेब पृथ्वी को कितने \omterm{def:g2:pushes-and-pulls:force}{बल} से वापस खींचता है? इस दो-तरफ़ा
        बहीखाते के लिए नियम का शब्द बताओ।
  \item सेब को उठाकर केंद्र से पृथ्वी की दोगुनी त्रिज्या पर ले जाओ।
        खिंचाव का क्या होता है?
  \item जनसंपर्क में आया एक दर्शक पूछता है कि सूत्र ज़मीन के बजाय
        पृथ्वी के \emph{केंद्र} तक की दूरी क्यों लेता है। ईमानदार
        छोटा जवाब दो (पूरा \omterm{def:g4:solar-system:planet}{ग्रह} खींचता है, और उसके बिखरे हुए खिंचाव
        जुड़कर ठीक ऐसे बनते हैं मानो सब कुछ केंद्र से आ रहा हो ---
        यानी एक प्रमेय, जिसे साबित करने में ख़ुद उस महान लेखक को
        बरसों लगे थे)।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- प्रक्षेपण का पटल।}
\begin{enumerate}[resume]
  \item आज के उपग्रह का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \qty{2000}{kg} है। नियम से सतह पर
        उसका \omterm{def:g4:weight-and-mass:weight}{भार}? (या अगले अध्याय में नाम पाने वाले उस शॉर्टकट से:
        क़रीब \qty{9.8}{N} प्रति \omterm{def:g3:measuring-mass:units}{किलोग्राम}।)
  \item केंद्र से \qty{1.28e7}{m} की अपनी कामकाजी \omterm{def:g6:sun-earth-moon:axisorbit}{कक्षा} में ---
        दो पृथ्वी-त्रिज्याओं पर --- गुरुत्व उस पर उसके सतह वाले \omterm{def:g4:weight-and-mass:weight}{भार}
        का कौन-सा अंश अब भी लगाता है?
  \item ``तो वहाँ ऊपर सचमुच गुरुत्व है?'' अंक से पुष्टि करो --- और
        समझाओ कि उपग्रह उस खिंचाव को गिरकर टकराने के बजाय किस काम में
        लगा रहा है।
  \item प्रक्षेपण की टिप्पणी कहती है ``कक्षीय \omterm{def:g7:motion-average-speed:formula}{चाल} को गिरने से मेल
        खाना चाहिए''। दर्शकों की पुस्तिका के लिए पहाड़ी तोप वाली कहानी
        दो वाक्यों में दोबारा सुनाओ।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का बहीखाता।}
\begin{enumerate}[resume]
  \item \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} \qty{3.8e8}{m} पर है, तो वह कितनी पृथ्वी-त्रिज्याएँ
        दूर सवार है? (एक दशमलव अंक।)
  \item प्रतिलोम वर्ग के हिसाब से वहाँ गुरुत्व फीका पड़कर सतह वाली
        ताक़त का मोटे तौर पर कौन-सा अंश रह जाता है? (अपने पिछले जवाब
        का वर्ग करो।)
  \item वह महान कौंध, दोबारा: बग़ीचे का सेब अपने पहले सेकंड में
        क़रीब पाँच \omterm{def:g2:measuring-length:units}{मीटर} गिरता है; जबकि \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} हर सेकंड पृथ्वी की ओर
        बमुश्किल एक मिलीमीटर ``गिरता'' है --- और फिर भी कभी पहुँचता
        नहीं। अपने जवाबों $9$ और $10$ से इन दोनों तथ्यों को जोड़ो:
        \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का गिरना हज़ारों गुना नरम क्यों है, और अनंत क्यों?
  \item जनसंपर्क की शाम बंद करो: एक नियम, तीन कलाकार --- सेब, उपग्रह,
        \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}। पटल का वह समापन वाक्य लिखो जो तीनों को जोड़ देता है।
\end{enumerate}
\end{problem}
