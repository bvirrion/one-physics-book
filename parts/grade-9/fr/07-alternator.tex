\chapter{De la centrale à la maison~: l'alternateur}\label{ch:g9:alternator}

Une année entière de \omterm{def:g9:electric-power-energy:kwh}{kilowattheures} a traversé tes audits --- mais
d'où venaient-ils~? Suis le câble hors de la maison, sous la rue, le
long des pylônes, et il s'achève sur une machine qui tourne dans une
grande halle~: l'\omterm{def:g9:alternator:alternator}{alternateur}. Son secret est la plus jolie surprise de
l'électricité de ce livre~: la découverte de l'an dernier, le courant
qui fabrique du magnétisme, prise à l'envers.

\section{L'induction~: l'effet retourné}

\begin{proposition}[Induction électromagnétique]\label{prop:g9:alternator:induction}
Déplace un \omterm{def:g1:magnets:magnet}{aimant} près d'un \omterm{def:g3:first-electric-circuit:circuit}{circuit fermé} --- ou déplace le circuit
près de l'\omterm{def:g1:magnets:magnet}{aimant} --- et, tant que dure le mouvement, une \omterm{def:g8:voltage-voltmeter:voltage}{tension}
apparaît et entraîne un courant~: de l'électricité \emph{induite} par
un magnétisme qui change. Les règles, tirées de l'établi~:
\begin{enumerate}
  \item seul le \emph{changement} induit~: \omterm{def:g1:magnets:magnet}{aimant} au repos à côté
        de la bobine, rien --- si puissant que soit l'\omterm{def:g1:magnets:magnet}{aimant}~;
  \item mouvement plus rapide, \omterm{def:g1:magnets:magnet}{aimant} plus fort, plus de spires à
        la bobine~: \omterm{def:g8:voltage-voltmeter:voltage}{tension} induite plus grande~;
  \item inverse le mouvement, tu inverses le courant induit ---
        l'approche et le retrait entraînent des marches opposées.
\end{enumerate}
L'an dernier, un courant fabriquait du magnétisme~; cette année, un
magnétisme en mouvement fabrique un courant. Les deux effets sont une
seule pièce, lue sur ses deux faces --- et la seconde face alimente la
civilisation.
\end{proposition}

\begin{method}[L'induction sur l'établi]\label{met:g9:alternator:bench}
Une bobine à nombreuses spires, un \omterm{def:g1:magnets:magnet}{aimant} droit, un appareil sensible
à zéro central~:
\begin{enumerate}
  \item brancher l'appareil aux \omterm{def:g3:first-electric-circuit:battery}{bornes} de la bobine~; tenir
        l'\omterm{def:g1:magnets:magnet}{aimant} immobile à l'intérieur~: l'aiguille dort à zéro ---
        la \omterm{def:g2:pushes-and-pulls:force}{force} seule n'induit rien~;
  \item enfoncer l'\omterm{def:g1:magnets:magnet}{aimant}~: l'aiguille bondit d'un côté --- et
        revient à zéro à l'instant où le mouvement cesse~;
  \item le retirer~: un bond égal, de l'autre côté~;
  \item l'enfoncer plus vite~: un bond plus grand. Fais aller et
        venir l'\omterm{def:g1:magnets:magnet}{aimant} en cadence, et l'aiguille oscille à gauche, à
        droite, à gauche --- tu fabriques à la main un \omterm{def:g9:alternating-current:dcac}{courant
        alternatif}.
\end{enumerate}
\end{method}

\section{L'alternateur}

\begin{definition}[Alternateur]\label{def:g9:alternator:alternator}
Un \emph{alternateur}\index{alternateur} est l'induction rendue
infatigable~: un \omterm{def:g1:magnets:magnet}{aimant} tourné régulièrement à côté d'une bobine fixe
(ou une bobine tournée dans un champ magnétique). Chaque demi-tour
amène un pôle vers la bobine, puis l'emporte --- du changement, sans
cesse renouvelé --- si bien que la \omterm{def:g8:voltage-voltmeter:voltage}{tension} induite oscille positive,
négative, positive~: \emph{alternative}, par construction, traçant
exactement la douce sinusoïde du chapitre de l'\omterm{met:g9:alternating-current:scope}{oscilloscope}. Tourne à
cinquante tours par seconde et la sortie alterne à \qty{50}{Hz}~: la
\omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} du réseau \emph{est} une \omterm{def:g7:motion-average-speed:formula}{vitesse} de rotation, tenue au pas
par chaque centrale du réseau.
\end{definition}

\begin{example}[La dynamo de ta roue]\label{ex:g9:alternator:dynamo}
La dynamo de bicyclette est un \omterm{def:g9:alternator:alternator}{alternateur} de poche~: la roue fait
tourner un petit \omterm{def:g1:magnets:magnet}{aimant} dans une bobine fixe, et la lampe s'allume ---
plus brillante et à plus haute \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} à mesure que tu pédales plus
fort, exactement comme le promettent les règles de l'établi (et
papillotante au pas, comme l'observait le chapitre de l'alternatif).
Son prix est honnête aussi~: la dynamo freine --- allumer la lampe
coûte à tes jambes un effort bien réel. L'induction ne donne aucune
\omterm{def:g5:everyday-energy:energy}{énergie}~; elle \emph{convertit} du mouvement en électricité, \omterm{def:g9:kinetic-energy-safety:joule}{joule}
pour \omterm{def:g9:kinetic-energy-safety:joule}{joule}, moins l'impôt du frottement.
\end{example}

\section{Toutes les centrales sont une seule machine}

\begin{proposition}[Le cœur commun]\label{prop:g9:alternator:core}
La quasi-totalité de l'électricité du monde, quelle que soit sa
réclame, se fabrique de la même façon~: quelque chose fait tourner une
\emph{turbine} --- un robuste moulin à pales --- et la turbine fait
tourner un \omterm{def:g9:alternator:alternator}{alternateur}. Les centrales ne diffèrent que par \emph{ce
qui pousse les pales}~:
\begin{enumerate}
  \item les centrales thermiques font bouillir de l'eau en brûlant
        du charbon ou du gaz --- des jets de vapeur entraînent la
        turbine~;
  \item les centrales nucléaires font bouillir la même eau avec la
        \omterm{def:g5:heat-and-insulation:heat}{chaleur} d'atomes que l'on brise --- de la vapeur encore~;
  \item les barrages hydroélectriques laissent un lac de montagne
        tomber au travers des pales --- la chaîne du barrage de
        l'enfance, achevée~;
  \item les éoliennes se passent d'intermédiaire~: le \omterm{def:g2:air-around-us:wind}{vent} est la
        poussée des pales, avec un \omterm{def:g9:alternator:alternator}{alternateur} dans chaque nacelle.
\end{enumerate}
Une exception confirme la règle~: les panneaux solaires fabriquent de
l'électricité sans la moindre rotation --- par un effet de la \omterm{def:g1:light-and-shadows:source}{lumière}
qui appartient aux années d'université.
\end{proposition}

\begin{example}[Des chaînes, auditées]\label{ex:g9:alternator:chains}
Écris chaque centrale comme une chaîne d'\omterm{def:g5:everyday-energy:energy}{énergie} de l'enfance,
désormais dotée d'un vocabulaire professionnel. Le barrage~: réserve
de hauteur $\to$ mouvement de l'eau qui tombe $\to$ rotation $\to$
électricité --- et c'est le Soleil, rappelle-toi, qui a élevé ce lac
par évaporation~: l'hydroélectricité est du soleil différé en pluie.
La centrale à charbon~: réserve chimique $\to$ \omterm{def:g5:heat-and-insulation:heat}{chaleur} $\to$ mouvement
de la vapeur $\to$ rotation $\to$ électricité --- avec de la \omterm{def:g5:heat-and-insulation:heat}{chaleur}
qui fuit à chaque flèche, comme l'avouent les panaches des tours de
refroidissement. Le \omterm{def:g2:air-around-us:wind}{vent}~: de l'\omterm{def:g2:air-around-us:air}{air} en mouvement (brassé par le
Soleil) directement en rotation. Les sources que la nature remplit à
nouveau --- l'eau, le \omterm{def:g2:air-around-us:wind}{vent}, la \omterm{def:g1:light-and-shadows:source}{lumière} du Soleil --- sont dites
\emph{renouvelables}\index{renouvelable}~; les réserves brûlées et
brisées sont dépensées pour toujours à l'échelle humaine.
\end{example}

\begin{remark}[Pourquoi les pylônes ronronnent à 400000 volts]\label{rem:g9:alternator:transport}
La dernière promesse du chapitre de l'alternatif arrive à échéance.
Une ville a besoin, mettons, de cent millions de \omterm{def:g9:electric-power-energy:power}{watts}. D'après
$P = U I$, cette \omterm{def:g9:electric-power-energy:power}{puissance} peut voyager comme un courant énorme sous
une \omterm{def:g8:voltage-voltmeter:voltage}{tension} modeste --- ou comme un courant modeste sous une \omterm{def:g8:voltage-voltmeter:voltage}{tension}
énorme. Or les fils s'échauffent avec le \emph{courant} (le vieil
\omterm{prop:g8:electric-current-effects:three}{effet thermique}), et chaque degré de pylône échauffé est de l'\omterm{def:g5:everyday-energy:energy}{énergie}
payée puis perdue pour les corbeaux. Le réseau transporte donc sous
des \omterm{def:g8:voltage-voltmeter:voltage}{tensions} à couper le souffle --- des centaines de milliers de
\omterm{def:g8:voltage-voltmeter:voltage}{volts}, des courants minuscules, des pertes minimales --- puis abaisse
la \omterm{def:g8:voltage-voltmeter:voltage}{tension}, quartier après quartier, jusqu'aux \qty{230}{V}
apprivoisés de ton mur. La machine qui abaisse, le
\emph{transformateur}, n'est que deux bobines et de l'induction une
fois de plus --- et elle ne fonctionne que sur un courant qui
alterne~: tout le réseau parle alternatif parce que le transformateur
le parle.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g9:alternator:1}
Énonce les trois règles d'établi de l'induction. Qu'induit un \omterm{def:g1:magnets:magnet}{aimant}
puissant \emph{au repos}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:2}
Pourquoi la sortie d'un \omterm{def:g9:alternator:alternator}{alternateur} est-elle alternative~? Relie les
changements de signe à la rotation.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:3}
À quelle \omterm{def:g7:motion-average-speed:formula}{vitesse} de rotation un \omterm{def:g9:alternator:alternator}{alternateur} alimente-t-il un réseau à
\qty{50}{Hz}~? Sur quoi toutes les centrales du réseau doivent-elles
donc s'accorder~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:4}
Nomme les deux cases universelles de toute centrale, et la case qui
change entre le charbon, l'atome, le barrage et le \omterm{def:g2:air-around-us:wind}{vent}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:5}
Écris la chaîne hydroélectrique du lac de montagne jusqu'à ta lampe
--- et nomme la machine céleste qui a rempli le lac.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:6}
Pourquoi pédaler devient-il plus dur à l'instant où la lampe de la
dynamo s'allume~? Quel principe interdit qu'il en soit autrement~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:7}
Trie en renouvelable et non renouvelable~: le \omterm{def:g2:air-around-us:wind}{vent}~; le charbon~; le
lac du barrage~; l'uranium~; la \omterm{def:g1:light-and-shadows:source}{lumière} du Soleil. Qu'est-ce qui
définit ce tri~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternator:8}
Pourquoi le réseau expédie-t-il sa \omterm{def:g9:electric-power-energy:power}{puissance} sous des centaines de
milliers de \omterm{def:g8:voltage-voltmeter:voltage}{volts}~? Nomme la formule qui offre le choix et l'effet qui
le tranche.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternator:9}
Un \omterm{def:g9:alternator:alternator}{alternateur} de démonstration actionné lentement à la manivelle
allume sa lampe faiblement, avec un papillotement visible~; actionné
vite, il l'allume vivement et régulièrement. Rends compte des deux
changements avec les règles de l'établi et le chapitre des
\omterm{def:g8:sound-pitch-loudness:frequency}{fréquences}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternator:10}
Une centrale délivre \qty{1e8}{W}. Compare les courants nécessaires
sous \qty{230}{V} et sous \qty{400000}{V} (en notation
scientifique). Lequel de vrais câbles pourraient-ils porter~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternator:11}
Les centrales thermiques et nucléaires sont, au fond, des bouilloires
élaborées. Défends ce slogan par leurs chaînes --- et situe l'unique
maillon par lequel les deux diffèrent.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:alternator:12}
Le grand tour, parcouru à l'envers~: ce soir, ta lampe brille sous
\qty{230}{V}. Remonte son \omterm{def:g5:everyday-energy:energy}{énergie} à travers chaque case ---
transformateur, pylônes, \omterm{def:g9:alternator:alternator}{alternateur}, turbine, et une réserve primaire
de ton choix --- en nommant la forme d'\omterm{def:g5:everyday-energy:energy}{énergie} à chaque étape et les
fuites en chemin. Puis réponds à la question de l'enfance par une
phrase de dernière année~: d'où vient, en dernier ressort, la \omterm{def:g1:light-and-shadows:source}{lumière}
de ta chambre~?
\end{exercise}

\section{Problème~: garde de nuit à la centrale}

\begin{problem}[{Problème du week-end --- garde de nuit à la centrale
hydroélectrique~; induction, marche au pas et tempête~; le journal de
l'ingénieure}]\label{pb:g9:alternator:1}
Tu accompagnes l'ingénieure de nuit au barrage de la vallée --- le lac
au-dessus, la ville en dessous, les \omterm{def:g9:alternator:alternator}{alternateurs} ronronnant entre les
deux.

\medskip\noindent\textbf{Partie I --- la halle des machines.}
\begin{enumerate}
  \item L'ingénieure ouvre une vanne~: l'eau tonne à travers la
        turbine trois, et les appareils de son \omterm{def:g9:alternator:alternator}{alternateur} se
        réveillent. Écris la chaîne d'\omterm{def:g5:everyday-energy:energy}{énergie} du lac au jeu de
        barres.
  \item À l'intérieur de l'\omterm{def:g9:alternator:alternator}{alternateur}, qu'est-ce qui bouge et
        qu'est-ce qui reste immobile --- et pourquoi faut-il que
        quelque chose bouge~?
  \item Le réseau tourne à \qty{50}{Hz}. L'ingénieure surveille un
        cadran qui maintient la turbine trois exactement à la
        rotation correspondante. Qu'est-ce qui se gâte à la
        mauvaise \omterm{def:g7:motion-average-speed:formula}{vitesse}~? (Que ferait la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} de sa
        \omterm{def:g8:voltage-voltmeter:voltage}{tension}~?)
  \item Un apprenti demande si un \omterm{def:g1:magnets:magnet}{aimant} plus puissant dans le
        rotor donnerait de la \omterm{def:g9:electric-power-energy:power}{puissance} «~gratuite~» en plus.
        Corrige ce rêve avec le prix honnête de la dynamo.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- la demande de la ville.}
\begin{enumerate}[resume]
  \item À 03:00, la ville appelle \qty{2e7}{W}. Sous les
        \qty{200000}{V} de la ligne de transport, quel courant
        quitte la centrale~?
  \item La même \omterm{def:g9:electric-power-energy:power}{puissance} sous \qty{230}{V} demanderait quel
        courant --- et que dit cette comparaison sur le métier du
        transformateur~?
  \item À 07:00, les bouilloires et les radiateurs se réveillent~:
        la demande double. Que l'ingénieure doit-elle donner en
        plus aux turbines, et qu'arriverait-il à la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} du
        réseau si elle en donnait trop peu~? (Les machines
        ralentissent sous la charge, comme un cycliste dans une
        côte.)
  \item La centrale sœur --- un parc éolien sur la crête --- se
        déconnecte quand le \omterm{def:g2:air-around-us:wind}{vent} tombe. Quelle case de sa chaîne a
        failli, et quelles cases se portaient parfaitement bien~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- la tempête.}
\begin{enumerate}[resume]
  \item La foudre coupe une ligne de transport~; les disjoncteurs
        s'ouvrent. L'ingénieure réachemine la \omterm{def:g9:electric-power-energy:power}{puissance} par les
        lignes voisines. Quelle propriété du réseau --- un seul
        grand maillage \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} --- rend ce réacheminement
        possible~?
  \item Pendant la manœuvre, l'éclairage d'un entrepôt vire au brun
        pendant une minute. Interprète~: qu'est-ce qui a fléchi, et
        de quoi les \omterm{ex:g6:circuits-and-safety:motor}{moteurs} et les lampes de l'entrepôt ont-ils été
        brièvement sous-alimentés~?
  \item L'aube~: le journal doit expliquer aux apprentis pourquoi
        toute l'électricité de la nuit était, à l'origine, de la
        \omterm{def:g1:light-and-shadows:source}{lumière} du Soleil. Rédige l'explication en deux lignes
        (lac, évaporation, pluie).
  \item Dernière ligne du journal, par tradition~: la physique de
        la nuit en une phrase --- avec le mouvement, le magnétisme
        et la bouilloire matinale de la ville dedans.
\end{enumerate}
\end{problem}
