\chapter{La puissance et l'énergie électriques}\label{ch:g9:electric-power-energy}

La facture d'électricité ne facture ni les \omterm{def:g8:voltage-voltmeter:voltage}{volts}, ni les \omterm{def:g8:intensity-ammeter:intensity}{ampères} ---
elle facture quelque chose que les appareils de ta famille multiplient
discrètement l'un par l'autre tout le mois. Ce chapitre marie les deux
grandeurs électriques en une troisième, donne à la dépense rapide et à
la dépense lente leurs noms propres, et s'achève au compteur près de
la porte d'entrée~: de la physique que tu peux auditer contre de
l'argent réel.

\section{La puissance~: le débit de la livraison}

\begin{definition}[La puissance et le watt]\label{def:g9:electric-power-energy:power}
La \emph{puissance}\index{puissance} $P$ d'un appareil est le débit
auquel il convertit de l'\omterm{def:g5:everyday-energy:energy}{énergie} --- des \omterm{def:g9:kinetic-energy-safety:joule}{joules} par seconde. Son \omterm{def:g6:measurement-in-science:measuring}{unité}
est le \emph{watt}\index{watt} (\unit{W})~: un \omterm{def:g9:kinetic-energy-safety:joule}{joule} chaque seconde,
en l'honneur de l'ingénieur de l'âge de la vapeur~; le
\emph{kilowatt} (\qty{1}{kW} $=$ \qty{1000}{W}) sert les \omterm{def:g4:weight-and-mass:weight}{poids} lourds
de la cuisine. La puissance n'est pas une quantité d'\omterm{def:g5:everyday-energy:energy}{énergie} mais une
\emph{allure} de dépense --- le débit d'un robinet, non le contenu
d'un seau.
\end{definition}

\begin{proposition}[La puissance électrique]\label{prop:g9:electric-power-energy:UI}
Un appareil alimenté sous la \omterm{def:g8:voltage-voltmeter:voltage}{tension} $U$ et appelant le courant $I$
reçoit la \omterm{def:g9:electric-power-energy:power}{puissance} électrique
\[
P = U \times I ,
\]
des \omterm{def:g9:electric-power-energy:power}{watts} à partir de \omterm{def:g8:voltage-voltmeter:voltage}{volts} multipliés par des \omterm{def:g8:intensity-ammeter:intensity}{ampères}. La formule
calcule dans les trois sens, et sa seconde recette fait tourner le
tableau des \omterm{def:g7:short-circuits-safety:fuse}{fusibles}~: $I = P/U$ dit quel courant la \omterm{def:g9:electric-power-energy:power}{puissance}
nominale d'un appareil tirera du secteur.
\end{proposition}

\begin{example}[L'échelle domestique]\label{ex:g9:electric-power-energy:ladder}
Lis les plaques des machines de la famille~: un voyant lumineux,
\qty{1}{W}~; une lampe moderne, \qty{5}{W} (son ancêtre incandescente
en dépensait soixante pour la même \omterm{def:g1:light-and-shadows:source}{lumière} --- la différence était de
la \omterm{def:g5:heat-and-insulation:heat}{chaleur})~; un ordinateur portable, \qty{50}{W}~; une télévision,
\qty{100}{W}~; un lave-linge qui chauffe son eau, \qty{2000}{W}~; la
bouilloire, \qty{2300}{W}~; un chauffe-eau instantané, jusqu'à
\qty{9000}{W}. Le motif de l'échelle~: tout ce dont le métier est de
\emph{chauffer} grimpe jusqu'aux kilowatts --- la \omterm{def:g5:heat-and-insulation:heat}{chaleur} est la forme
d'\omterm{def:g5:everyday-energy:energy}{énergie} la plus coûteuse à fournir.
\end{example}

\begin{example}[L'arithmétique du tableau des fusibles]\label{ex:g9:electric-power-energy:fuses}
La bouilloire sur le secteur à \qty{230}{V}~:
$I = 2300 \div 230 = \qty{10}{A}$ --- l'appareil à dix \omterm{def:g8:intensity-ammeter:intensity}{ampères} des
vieux chapitres, désormais démontré. Le chauffe-eau de \qty{9000}{W}~:
près de \qty{40}{A}, exigeant sa propre \omterm{def:g5:series-parallel-circuits:parallel}{branche} robuste et son propre
disjoncteur. Et la multiprise surchargée de l'enquête sur l'incendie~:
la bouilloire, le radiateur et la friteuse additionnant leurs courants
$P/U$ au-delà du calibre de la \omterm{def:g5:series-parallel-circuits:parallel}{branche} --- trois innocents, un seul
crime arithmétique, exactement comme il était reproché.
\end{example}

\section{L'énergie~: la quantité livrée}

\begin{proposition}[L'énergie à partir de la puissance et du temps]\label{prop:g9:electric-power-energy:Pt}
Un appareil de \omterm{def:g9:electric-power-energy:power}{puissance} $P$ qui fonctionne pendant une \omterm{def:g2:measuring-time:duration}{durée} $t$
convertit l'\omterm{def:g5:everyday-energy:energy}{énergie}
\[
E = P \times t ,
\]
des \omterm{def:g9:kinetic-energy-safety:joule}{joules} à partir de \omterm{def:g9:electric-power-energy:power}{watts} multipliés par des secondes. Robinet
rapide ou long filet d'eau, le seau se remplit selon le produit~: les
trois minutes d'une bouilloire de \qty{2300}{W}
($2300 \times 180 \approx \qty{4.1e5}{J}$) dépensent plus que la
journée entière d'une lampe de \qty{5}{W}
($5 \times 86400 = \qty{4.3e5}{J}$) --- presque exactement à égalité,
en fait~: une ébullition vaut une journée de bonne \omterm{def:g1:light-and-shadows:source}{lumière}.
\end{proposition}

\begin{definition}[Le kilowattheure]\label{def:g9:electric-power-energy:kwh}
Les \omterm{def:g5:everyday-energy:energy}{énergies} domestiques font des comptes de \omterm{def:g9:kinetic-energy-safety:joule}{joules} disgracieux~; le
compteur parle donc en
\emph{kilowattheures}\index{kilowattheure} (\unit{kWh})~: l'\omterm{def:g5:everyday-energy:energy}{énergie}
d'un kilowatt maintenu pendant une heure,
\[
\qty{1}{kWh} = 1000 \times 3600 = \qty{3.6e6}{J}
\]
--- trois millions six cent mille \omterm{def:g9:kinetic-energy-safety:joule}{joules} par \omterm{def:g6:measurement-in-science:measuring}{unité} sur la facture.
Avec $P$ en kilowatts et $t$ en \omterm{ex:g2:measuring-time:clock}{heures}, $E = P \times t$ livre
directement des kilowattheures~: la formule dont vit le compteur.
\end{definition}

\begin{method}[Auditer un appareil]\label{met:g9:electric-power-energy:audit}
Pour toute machine et tout mois~:
\begin{enumerate}
  \item lire sa \omterm{def:g9:electric-power-energy:power}{puissance} $P$ sur la plaque (ou calculer
        $U \times I$)~;
  \item estimer honnêtement sa \omterm{def:g2:measuring-time:duration}{durée} de fonctionnement $t$ --- par
        jour, puis par mois~;
  \item $E = P \times t$ en \omterm{def:g9:electric-power-energy:kwh}{kilowattheures} ($P$ en \unit{kW}, $t$
        en \omterm{ex:g2:measuring-time:clock}{heures})~;
  \item multiplier par le tarif (prends environ $0.25$ \omterm{def:g6:measurement-in-science:measuring}{unité}
        monétaire par \unit{kWh}) pour le coût du mois.
\end{enumerate}
La bouilloire, six minutes par jour~:
$2.3 \times 0.1 \times 30 = \qty{6.9}{kWh}$ par mois. La télévision,
trois \omterm{ex:g2:measuring-time:clock}{heures} par jour~: $0.1 \times 3 \times 30 = \qty{9}{kWh}$. De
petites \omterm{def:g9:electric-power-energy:power}{puissances} sur de longues \omterm{ex:g2:measuring-time:clock}{heures} rivalisent avec de grandes
\omterm{def:g9:electric-power-energy:power}{puissances} sur de courtes --- la leçon récurrente de l'audit.
\end{method}

\section{Le compteur et la facture}

\begin{example}[Lire la facture]\label{ex:g9:electric-power-energy:bill}
Le compteur près de la porte d'entrée compte chaque \omterm{def:g9:electric-power-energy:kwh}{kilowattheure} qui
entre dans la maison~; la facture est sa différence du mois multipliée
par le tarif. Les $250$ \omterm{def:g6:measurement-in-science:measuring}{unités} mensuelles d'une famille typique se
décomposent à l'audit~: le chauffage de l'eau et les radiateurs
d'abord (le club des kilowatts), puis les appareils du froid --- de
modestes \omterm{def:g9:electric-power-energy:power}{watts}, mais qui tournent toujours --- puis la cuisson, le
lavage, l'éclairage et l'électronique. La \omterm{def:g9:electric-power-energy:power}{puissance} de l'audit~: il
trouve les vrais \omterm{def:g3:levers-and-scales:lever}{leviers}. Remplacer dix vieilles ampoules de
\qty{60}{W} par des \qty{5}{W} économise mille fois plus que
débrancher tous les chargeurs de la maison.
\end{example}

\begin{example}[Les vampires de la veille]\label{ex:g9:electric-power-energy:standby}
Petits caractères, longues \omterm{ex:g2:measuring-time:clock}{heures}~: le voyant de veille d'une
télévision sirote \qty{0.5}{W} --- mais pendant les $720$ \omterm{ex:g2:measuring-time:clock}{heures} du
mois~: $0.0005 \times 720 = \qty{0.36}{kWh}$. Un vampire est
inoffensif~; une maison avec vingt siroteurs d'un \omterm{def:g9:electric-power-energy:power}{watt} chacun paie
$0.02 \times 720 \approx \qty{14}{kWh}$ par mois pour rien. La
\omterm{def:g9:electric-power-energy:power}{puissance} multipliée par le temps ne pardonne aucun facteur~: le
second mord quand le premier se cache.
\end{example}

\begin{remark}[La puissance dans le monde plus large]\label{rem:g9:electric-power-energy:wider}
Le \omterm{def:g9:electric-power-energy:power}{watt}, né électrique ici, mesure toute allure d'\omterm{def:g5:everyday-energy:energy}{énergie}. Ton corps au
repos tourne au ralenti près de \qty{100}{W} --- une ampoule ancestrale
de \omterm{def:g5:heat-and-insulation:heat}{chaleur} (rappelle-toi l'\omterm{def:g2:air-around-us:air}{air} lourd de la salle de classe bondée~:
trente élèves, trois kilowatts de biologie). Un cycliste soutient
\qty{200}{W}~; le moteur d'une voiture déchaîne des dizaines de
kilowatts~; la centrale électrique de la ville, des centaines de
millions de \omterm{def:g9:electric-power-energy:power}{watts}~; la \omterm{def:g1:light-and-shadows:source}{lumière} du Soleil qui tombe sur la ville,
confortablement davantage --- le plafond sous lequel travaillent
toutes les autres allures. Le chapitre suivant suit les \omterm{def:g9:electric-power-energy:power}{watts} du
réseau jusqu'à leur source tournante.
\end{remark}

\section{Problème~: l'audit énergétique de la maison}

\begin{problem}[{Problème du week-end --- l'audit énergétique de la
famille~; des plaques, des compteurs et la facture du mois~; trouver
les vrais leviers}]\label{pb:g9:electric-power-energy:1}
Armée des deux formules et du tarif ($0.25$ le \unit{kWh}), la
famille audite un mois (30 jours). L'inventaire~: bouilloire
\qty{2300}{W}, $6$ minutes par jour~; télévision \qty{100}{W}, $4$
\omterm{ex:g2:measuring-time:clock}{heures} par jour~; réfrigérateur --- cas particulier ci-dessous~; dix
lampes de \qty{5}{W} chacune, $5$ \omterm{ex:g2:measuring-time:clock}{heures} par jour~; lave-linge
\qty{2}{kW}, un cycle chaud d'une heure un jour sur deux~;
chauffe-eau \qty{2.5}{kW}, $2$ \omterm{ex:g2:measuring-time:clock}{heures} par jour~; siroteurs de veille
totalisant \qty{10}{W}, toujours allumés.

\medskip\noindent\textbf{Partie I --- machine par machine.}
\begin{enumerate}
  \item Le mois de la bouilloire, en \unit{kWh}.
  \item Les mois de la télévision et des dix lampes.
  \item Les mois du lave-linge ($15$ cycles) et du chauffe-eau.
  \item Le mois des siroteurs de veille --- le second facteur à
        l'œuvre.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- le réfrigérateur et la
somme.}
Le compresseur du réfrigérateur tourne à \qty{120}{W}, mais un tiers
du temps seulement, en cyclant jour et nuit.
\begin{enumerate}[resume]
  \item Sa \omterm{def:g9:electric-power-energy:power}{puissance} moyenne effective, et son mois en \unit{kWh}.
  \item Totalise les sept lignes de l'audit. Quelles trois lignes
        dominent~?
  \item La facture du mois au tarif --- et la part du chauffe-eau
        dedans, en \omterm{def:g6:measurement-in-science:measuring}{unités} monétaires.
  \item Le compteur indiquait $2150$ \omterm{def:g6:measurement-in-science:measuring}{unités} au début du mois.
        Prédis sa lecture de fin de mois.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- les \omterm{def:g3:levers-and-scales:lever}{leviers}.}
\begin{enumerate}[resume]
  \item Proposition une~: des douches plus courtes --- le
        chauffe-eau tombe à \qty{1.5}{h} par jour. Économie
        mensuelle, en \omterm{def:g5:everyday-energy:energy}{énergie} et en argent~?
  \item Proposition deux~: une campagne publique presse de
        débrancher les chargeurs de téléphone (\qty{0.2}{W} chacun
        au repos). Audite le mois au repos d'un chargeur et compare
        à la proposition une~; que nous apprend ce contraste sur
        les audits face aux slogans~?
  \item Proposition trois~: la grand-mère suggère la vieille
        habitude de ne remplir la bouilloire que du nécessaire ---
        moitié moins d'eau, moitié moins d'\omterm{def:g5:everyday-energy:energy}{énergie}. Nouvelle ligne
        de la bouilloire, et économie du mois~?
  \item Écris les conclusions de l'audit~: trois phrases --- où
        vivent vraiment les \omterm{def:g9:electric-power-energy:kwh}{kilowattheures}, quel facteur de la
        formule enseigne le cas de la veille, et quel changement
        domestique unique rapporte le plus.
\end{enumerate}
\end{problem}
