\chapter{Le courant alternatif}\label{ch:g9:alternating-current}

Tous les circuits que tu as construits jusqu'ici marchaient sur une
\omterm{def:g3:first-electric-circuit:battery}{pile}~: un courant défilant régulièrement du $+$ vers le $-$,
consciencieux comme l'aiguille d'une horloge. Mais les prises de tes
murs délivrent quelque chose de plus sauvage --- un courant qui
s'inverse cent fois par seconde, sa \omterm{def:g8:voltage-voltmeter:voltage}{tension} oscillant sans cesse entre
le plus et le moins. Voici le \omterm{def:g9:alternating-current:dcac}{courant alternatif}~: la forme sous
laquelle naît, voyage et se livre la quasi-totalité de l'électricité
du monde.

\section{Deux sortes de courant}

\begin{definition}[Courant continu et courant alternatif]\label{def:g9:alternating-current:dcac}
Un \emph{courant continu}\index{courant continu} circule toujours dans
le même sens, entraîné par une \omterm{def:g8:voltage-voltmeter:voltage}{tension} constante --- le cadeau de la
\omterm{def:g3:first-electric-circuit:battery}{pile}, constant jusqu'à l'épuisement de la réserve. Un \emph{courant
alternatif}\index{courant alternatif} change de sens périodiquement,
entraîné par une \omterm{def:g8:voltage-voltmeter:voltage}{tension} qui oscille en douceur du positif au négatif
et retour, encore et encore --- le régime de la prise, et celui de
l'alternateur (le chapitre suivant dira de qui il s'agit).
\end{definition}

\begin{definition}[Période et fréquence]\label{def:g9:alternating-current:period}
La \emph{période}\index{période} $T$ d'une \omterm{def:g8:voltage-voltmeter:voltage}{tension} alternative est la
\omterm{def:g2:measuring-time:duration}{durée} d'un cycle complet --- une oscillation entière, montée,
descente et retour --- en secondes. Sa \emph{\omterm{def:g8:sound-pitch-loudness:frequency}{fréquence}} $f$ est le
nombre de cycles par seconde, en \omterm{def:g8:sound-pitch-loudness:frequency}{hertz} --- l'\omterm{def:g6:measurement-in-science:measuring}{unité} du \omterm{def:g3:sound-around-us:sound}{son}, qui
resservira --- et les deux sont inverses l'une de l'autre~:
\[
f = \frac{1}{T}, \qquad T = \frac{1}{f}.
\]
\end{definition}

\begin{proposition}[L'état civil du secteur]\label{prop:g9:alternating-current:mains}
Dans la plus grande partie du monde, le secteur alterne à
$f = \qty{50}{Hz}$ --- cinquante cycles complets par seconde, une
\omterm{def:g9:alternating-current:period}{période} de $T = 1/50 = \qty{0.02}{s}$~: vingt millisecondes par
oscillation (certaines régions tournent à soixante). Le courant dans
le cordon de ta lampe de chevet change de sens cent fois par seconde
--- deux fois par cycle --- et n'a jamais défilé régulièrement d'une
\omterm{def:g3:first-electric-circuit:battery}{borne} vers l'autre.
\end{proposition}

\begin{example}[Arithmétique des fréquences]\label{ex:g9:alternating-current:arithmetic}
Le couple d'inverses se calcule dans les deux sens. Un générateur qui
cycle à \qty{60}{Hz}~: $T = 1/60 \approx \qty{0.017}{s}$ --- dix-sept
millisecondes. Un \omterm{met:g9:alternating-current:scope}{oscilloscope} montre un cycle qui dure
\qty{0.04}{s}~: $f = 1/0.04 = \qty{25}{Hz}$. Une dynamo de bicyclette
à pédalage lent fait \qty{10}{Hz}~: une \omterm{def:g9:alternating-current:period}{période} d'un dixième de
seconde --- et sa lampe papillote visiblement, dix pulsations par
seconde, jusqu'à ce que la \omterm{def:g7:motion-average-speed:formula}{vitesse} les lisse au-delà de ce que l'œil
remarque.
\end{example}

\section{Lire l'oscillation}

\begin{method}[Lire un tracé d'oscilloscope]\label{met:g9:alternating-current:scope}
L'\emph{oscilloscope}\index{oscilloscope} --- ou n'importe quelle
application d'analyse --- trace la \omterm{def:g8:voltage-voltmeter:voltage}{tension} en fonction du temps sur un
écran quadrillé~; ce que vaut chaque carreau, en \omterm{def:g8:voltage-voltmeter:voltage}{volts} et en
millisecondes, est fixé par deux boutons. Pour récolter un tracé~:
\begin{enumerate}
  \item compter les carreaux d'un cycle complet et multiplier par
        le temps par carreau~: c'est $T$~; puis $f = 1/T$~;
  \item compter les carreaux de la ligne médiane jusqu'à une crête
        et multiplier par les \omterm{def:g8:voltage-voltmeter:voltage}{volts} par carreau~: c'est la
        \emph{\omterm{def:g8:voltage-voltmeter:voltage}{tension} maximale} $U_{\text{max}}$~;
  \item contrôler la forme~: le secteur et les alternateurs
        dessinent l'onde douce et sans cesse répétée de la figure
        --- la \emph{sinusoïde}, nom que les mathématiques donnent à
        l'oscillation parfaite.
\end{enumerate}
\end{method}

\begin{proposition}[Tension efficace]\label{prop:g9:alternating-current:effective}
Quelle est «~la~» \omterm{def:g8:voltage-voltmeter:voltage}{tension} d'une oscillation qui visite toutes les
valeurs de $+U_{\text{max}}$ à $-U_{\text{max}}$ cent fois par
seconde~? Le nombre unique honnête est la \emph{tension
efficace}\index{tension efficace} $U$~: la \omterm{def:g8:voltage-voltmeter:voltage}{tension} continue constante
qui échaufferait un \omterm{def:g8:ohms-law:resistor}{résistor} tout aussi bien. Pour la sinusoïde, elle
se tient bien en dessous de la crête --- $U_{\text{max}}$ vaut
environ $1.4$ fois $U$ --- et c'est elle que tout \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} pour
\omterm{def:g9:alternating-current:dcac}{courant alternatif} est construit pour lire. Les fameux
«~\qty{230}{V}~» de la prise sont la valeur efficace~; les crêtes
atteignent en fait environ \qty{325}{V}, cinquante fois par seconde,
dans chaque sens.
\end{proposition}

\begin{example}[Qui s'en soucie]\label{ex:g9:alternating-current:whocares}
L'\omterm{prop:g8:electric-current-effects:three}{effet thermique} fonctionne très bien dans les deux cas --- un
filament ou une \omterm{def:g8:ohms-law:resistance}{résistance} de bouilloire s'échauffe quel que soit le
sens de la marche, et c'est pourquoi lampes et radiateurs boivent
l'alternatif sans se plaindre. Mais l'électronique --- téléphones,
ordinateurs portables, tout ce qui contient des puces --- réclame du
continu constant~: leurs blocs chargeurs et leurs alimentations
cachées convertissent l'oscillation de la prise en une constance de
\omterm{def:g3:first-electric-circuit:battery}{pile} (sens la tiédeur du bloc~: la conversion paie son impôt). Et le
monde des \omterm{def:g3:first-electric-circuit:battery}{piles} ne peut pas non plus se recharger sur une
oscillation~: charger, c'est de la chimie à l'envers, et la chimie
exige un sens constant.
\end{example}

\begin{remark}[Pourquoi le monde a choisi l'oscillation]\label{rem:g9:alternating-current:why}
Si les \omterm{def:g3:first-electric-circuit:battery}{piles} sont si agréablement constantes, pourquoi la civilisation
est-elle câblée en alternatif~? Deux raisons, qui attendent toutes
deux deux chapitres plus loin~: le \omterm{def:g9:alternating-current:dcac}{courant alternatif} est ce que les
machines tournantes \emph{fabriquent naturellement} --- et, de façon
décisive, seul l'alternatif nourrit le transformateur, l'humble
boîtier ronronnant qui permet à l'électricité de traverser un pays
sous une \omterm{def:g8:voltage-voltmeter:voltage}{tension} colossale et d'entrer chez toi sous une \omterm{def:g8:voltage-voltmeter:voltage}{tension}
apprivoisée. Le réseau est alternatif parce que le voyage l'exige~;
les deux chapitres suivants suivent le courant de la bobine tournante
au mur de la cuisine.
\end{remark}

\begin{remark}[L'oscillation et le corps]\label{rem:g9:alternating-current:safety}
La vieille arithmétique de la sécurité gagne un détail sinistre~: les
cinquante oscillations par seconde du secteur sont voisines du rythme
qui dérègle le plus facilement la propre cadence électrique du
cœur --- les chocs alternatifs sont, à \omterm{def:g8:intensity-ammeter:intensity}{ampère} égal, plus dangereux que
les chocs constants. Toutes les règles de l'enfance tiennent,
redoublées~: les \qty{230}{V} efficaces de la prise --- \qty{325}{V} à
la crête --- ne sont pas simplement une plus grosse \omterm{def:g3:first-electric-circuit:battery}{pile}. C'est un
animal différent, et moins indulgent.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:1}
Distingue le continu de l'alternatif en une phrase chacun, avec la
source domestique de chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:2}
Définis la \omterm{def:g9:alternating-current:period}{période} et la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} et écris leur relation
d'inverses. Calcule $T$ pour \qty{50}{Hz} et $f$ pour
$T = \qty{0.01}{s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:3}
Combien de fois par seconde le courant du secteur change-t-il de sens
à \qty{50}{Hz}~? (Attention~: les inversions par cycle.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:4}
Un \omterm{met:g9:alternating-current:scope}{oscilloscope} montre un cycle qui couvre $4$ carreaux à
\qty{5}{ms} par carreau. Trouve $T$ et $f$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:5}
La prise annonce \qty{230}{V}~; la crête atteint \qty{325}{V}. Nomme
les deux valeurs et le sens de la plus petite.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:6}
Pourquoi les bouilloires acceptent-elles l'alternatif de bon cœur
alors que les chargeurs de téléphone doivent le convertir~? Un effet,
une chimie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:7}
Esquisse (ou décris) les deux tracés~: une \omterm{def:g3:first-electric-circuit:battery}{pile} de \qty{9}{V}, et une
\omterm{def:g8:voltage-voltmeter:voltage}{tension} alternative de \omterm{def:g9:alternating-current:period}{période} \qty{20}{ms} et de crête \qty{9}{V}.
Quel unique trait dessiné par l'instrument les distingue d'un coup
d'œil~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:8}
Donne les deux raisons pour lesquelles la civilisation s'est câblée en
alternatif, telles que le chapitre les annonce.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:9}
Une dynamo de bicyclette lente allume sa lampe avec un papillotement
visible~; pédaler plus fort stabilise la lueur. Explique les deux
régimes par la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} --- et estime la \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence} en dessous de
laquelle le papillotement agace, sachant que le cinéma trompe l'œil à
environ $24$ images par seconde.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:10}
Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} pour alternatif branché aux \omterm{def:g3:first-electric-circuit:battery}{bornes} d'une prise indique
\qty{230}{V}. Un \omterm{met:g9:alternating-current:scope}{oscilloscope} (théorique et intrépide) branché sur la
même prise montre des crêtes à quelle valeur --- et combien de crêtes
par seconde, en comptant les deux signes~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:11}
Le réseau d'une région tourne à \qty{60}{Hz} sous \qty{120}{V}
efficaces. Calcule sa \omterm{def:g9:alternating-current:period}{période} et sa \omterm{def:g8:voltage-voltmeter:voltage}{tension} de crête approchée --- et
nomme lequel de tes appareils de voyage protesterait (consulte
\cref{ex:g9:alternating-current:whocares}~; les chargeurs modernes
lisent les petites lignes de leur étiquette).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:alternating-current:12}
De vieux films d'écrans et de tubes fluorescents montrent parfois des
barres sombres qui défilent --- les images de la caméra attrapant
l'éclat de la lampe en pleine oscillation. Une lampe du secteur à
\qty{50}{Hz} s'illumine à chaque crête, de \emph{n'importe quel}
signe~: combien de pulsations d'éclat par seconde~? Une caméra à $25$
images par seconde échantillonne ce battement~: explique par le
raisonnement pourquoi le battement entre les deux rythmes peint des
barres à déplacement lent, et pourquoi filmer dans une région à
\qty{60}{Hz} change le motif.
\end{exercise}

\section{Problème~: l'établi de l'oscilloscope}

\begin{problem}[{Problème du week-end --- soirée de qualification à
l'établi de l'oscilloscope~; quatre alimentations mystères, un
écran~; le registre du technicien}]\label{pb:g9:alternating-current:1}
Quatre alimentations sans étiquette doivent être identifiées d'après
leurs tracés. Les réglages de l'\omterm{met:g9:alternating-current:scope}{oscilloscope}~: \qty{5}{ms} par
carreau horizontal, \qty{2}{V} par carreau vertical (sauf indication
contraire). C'est toi qui tiens le registre.

\medskip\noindent\textbf{Partie I --- les quatre tracés.}
\begin{enumerate}
  \item Alimentation A~: une ligne plate, $2.25$ carreaux au-dessus
        du milieu. Identifie sa nature et sa \omterm{def:g8:voltage-voltmeter:voltage}{tension} --- et son
        identité domestique probable.
  \item Alimentation B~: une onde douce, un cycle par $4$
        carreaux, crêtes à $3$ carreaux. Nature, \omterm{def:g9:alternating-current:period}{période},
        \omterm{def:g8:sound-pitch-loudness:frequency}{fréquence}, \omterm{def:g8:voltage-voltmeter:voltage}{tension} de crête~?
  \item Alimentation C~: une ligne plate \emph{en dessous} du
        milieu de $2.25$ carreaux. Qu'est-ce que c'est, et quelle
        minuscule erreur l'explique~?
  \item Alimentation D~: une onde douce, un cycle par $2$
        carreaux, crêtes à $1.5$ carreau. Toutes ses
        caractéristiques --- et pourquoi la lampe de D
        papilloterait moins que celle de B si chacune en
        alimentait une.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- les questions du
technicien.}
\begin{enumerate}[resume]
  \item Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmètre} pour alternatif aux \omterm{def:g3:first-electric-circuit:battery}{bornes} de l'alimentation B
        indique environ \qty{4.2}{V}, alors que ses crêtes touchent
        \qty{6}{V}. Concilie les deux nombres.
  \item Laquelle des quatre alimentations pourrait charger un
        téléphone directement, et que doit-on placer entre les
        autres et sa batterie~?
  \item La plaque d'avertissement de l'établi~: «~L'alternatif mord
        plus fort, \omterm{def:g8:voltage-voltmeter:voltage}{volt} pour \omterm{def:g8:voltage-voltmeter:voltage}{volt}.~» Redis-en la raison.
  \item Un apprenti demande pourquoi l'\omterm{met:g9:alternating-current:scope}{oscilloscope} dessine l'onde
        de B si joliment arrondie plutôt qu'en dents de scie. Nomme
        cette forme et son origine (quel genre de machine, au
        chapitre suivant, la dessine naturellement~?).
\end{enumerate}

\medskip\noindent\textbf{Partie III --- le tracé du réseau
lui-même.}
Un démonstrateur bien isolé montre le secteur, à \qty{100}{V} et
\qty{5}{ms} par carreau.
\begin{enumerate}[resume]
  \item Prédis le tracé~: carreaux par cycle, et hauteur des crêtes
        en carreaux.
  \item À partir du tracé, retrouve $f$ et la \omterm{prop:g9:alternating-current:effective}{tension efficace} ---
        et vérifie les deux contre la fameuse étiquette de la
        prise.
  \item On bascule le démonstrateur sur un banc de \omterm{def:g3:first-electric-circuit:battery}{piles} de même
        puissance de chauffe efficace. Décris le nouveau tracé en
        une ligne.
  \item Clos le registre par le résumé en trois lignes du
        technicien~: ce que signifie la ligne plate, ce que
        signifie l'onde, et quel nombre unique permet de comparer
        honnêtement les deux.
\end{enumerate}
\end{problem}
