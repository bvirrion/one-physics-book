\chapter{Berat lawan Massa: Mengukur Gaya}\label{ch:g9:weight-vs-mass}

Lima tahun yang lalu, di atas sebuah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} kertas, kamu membelah
``\omterm{def:g4:weight-and-mass:weight}{berat}'' menjadi dua: \omterm{def:g3:measuring-mass:mass}{massa} untuk zatnya, \omterm{def:g4:weight-and-mass:weight}{berat} untuk tarikannya.
Pembelahan itu kebijaksanaan; kini ia menjadi hitungan. \omterm{def:g2:pushes-and-pulls:force}{Gaya} mendapat
\omterm{def:g6:measurement-in-science:measuring}{satuannya} --- yang dinamai menurut orang dengan apel yang jatuh ---
\omterm{def:g4:weight-and-mass:weight}{beratnya} mendapat rumusnya, dan timbangan karet gelang masa kecilmu
kembali dengan pakaian yang profesional.

\section{Newton}

\begin{definition}[Newton]\label{def:g9:weight-vs-mass:newton}
\omterm{def:g2:pushes-and-pulls:force}{Gaya} --- dorongan, tarikan, \omterm{def:g4:weight-and-mass:weight}{berat}, dan cengkeraman \omterm{def:g9:gravitation:gravitation}{gravitasi} ---
diukur dalam \emph{newton}\index{newton} (\unit{N}), yang menghormati
pengarang hukum \omterm{def:g9:gravitation:gravitation}{gravitasi}. Jangkar yang mudah dipakai: satu newton
kira-kira seberat sebutir apel kecil (sejarahnya tersenyum); tas
sekolahmu berbobot sekitar \qty{50}{N}; jabat tangan yang mantap
mengepal dengan sekitar \qty{100}{N}; sewa \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pada bab yang lalu
mencapai $10^{20}$.
\end{definition}

\begin{definition}[Dinamometer]\label{def:g9:weight-vs-mass:dynamometer}
Sebuah \emph{dinamometer}\index{dinamometer} adalah alat ukur \omterm{def:g2:pushes-and-pulls:force}{gaya}:
sebuah pegas bertera di dalam sarungnya, yang meregang sebanding
dengan gaya yang dikenakan, dengan jarumnya membaca \omterm{def:g9:weight-vs-mass:newton}{newton} secara
langsung. Ia adalah timbangan karet gelangmu yang sudah dewasa ---
dan seperti \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} dan \omterm{def:g8:intensity-ammeter:ammeter}{amperemeter} sebelumnya, ia memensiunkan satu
petak lagi ``terasa kuat'' menjadi bilangan yang jujur.
\end{definition}

\begin{method}[Mengukur sebuah gaya]\label{met:g9:weight-vs-mass:measure}
\begin{enumerate}
  \item pilihlah \omterm{def:g9:weight-vs-mass:dynamometer}{dinamometer} yang jangkauannya cocok dengan
        pekerjaannya --- alat \qty{2}{N} untuk kotak pensil, alat
        \qty{100}{N} untuk tas sekolah;
  \item nolkan ia pada kedudukan pemakaiannya (menggantung, kalau
        bebannya akan menggantung --- \omterm{def:g4:weight-and-mass:weight}{berat} pegasnya sendiri
        menggeser nolnya);
  \item kenakan \omterm{def:g2:pushes-and-pulls:force}{gayanya} dengan mantap, di sepanjang \omterm{def:g6:sun-earth-moon:axisorbit}{sumbu} alatnya,
        lalu bacalah setinggi mata;
  \item catatlah beserta \omterm{def:g6:measurement-in-science:measuring}{satuannya} --- dan, karena \omterm{def:g2:pushes-and-pulls:force}{gaya} mempunyai
        arah, catatlah arahnya juga ketika itu berarti.
\end{enumerate}
Pada diagram, \omterm{def:g2:pushes-and-pulls:force}{gaya} yang terukur menjadi sebuah anak panah yang digambar
menurut skala yang dinyatakan --- ``satu \omterm{def:g2:measuring-length:units}{sentimeter} tiap sepuluh
\omterm{def:g9:weight-vs-mass:newton}{newton}'' --- yang menunjuk sebagaimana \omterm{def:g2:pushes-and-pulls:force}{gayanya} menunjuk: anak panah
\omterm{def:g2:pushes-and-pulls:force}{gaya} masa kecilnya, kini kuantitatif.
\end{method}

\section{Rumus beratnya}

\begin{proposition}[Berat sebanding dengan massa]\label{prop:g9:weight-vs-mass:formula}
Pada tempat mana pun yang diberikan, \omterm{def:g4:weight-and-mass:weight}{berat} sebuah benda $P$ (dalam
\unit{N}) sebanding dengan \omterm{def:g3:measuring-mass:mass}{massanya} $m$ (dalam \unit{kg}):
\[
P = m \times g ,
\]
dengan $g$, yaitu \emph{kuat gravitasi}\index{kuat gravitasi} tempat
itu, memberikan berapa \omterm{def:g9:weight-vs-mass:newton}{newton} tarikannya tiap \omterm{def:g3:measuring-mass:units}{kilogram} zatnya. Dekat
permukaan Bumi,
\[
g \approx \qty{9.8}{N/kg}:
\]
tiap \omterm{def:g3:measuring-mass:units}{kilogram} ditarik dengan sedikit di bawah sepuluh \omterm{def:g9:weight-vs-mass:newton}{newton}. Rumusnya
adalah hukum bab yang lalu dalam pakaian setempat: $g$ membungkus
\omterm{def:g3:measuring-mass:mass}{massa} dan jari-jari Bumi menjadi satu bilangan untuk ``di sini''.
\end{proposition}

dan seperangkat \omterm{def:g3:measuring-mass:mass}{massa} bertanda: sebuah garis lurus yang melewati titik

\begin{example}[Rumusnya sedang bekerja]\label{ex:g9:weight-vs-mass:atwork}
Seorang murid \qty{60}{kg}: $P = 60 \times 9.8 = \qty{588}{N}$. Sebuah
bola sepak \qty{450}{g}: $P = 0.45 \times 9.8 \approx
\qty{4.4}{N}$ --- awas \omterm{def:g3:measuring-mass:units}{kilogramnya}. Terbalik: sebuah peti yang \omterm{def:g4:weight-and-mass:weight}{beratnya}
\qty{343}{N} bermassa $m = P/g = 343 \div 9.8 =
\qty{35}{kg}$. Dan terbalik ke atas: di sebuah gunung tempat sebuah
ransel \qty{10}{kg} berbobot \qty{97.7}{N}, $g$ setempatnya adalah
$97.7 \div 10 = \qty{9.77}{N/kg}$ --- $g$ menipis sedikit menurut
ketinggian, tepat sebagaimana dijanjikan kuadrat terbaliknya.
\end{example}

\begin{example}[Satu koper, banyak berat]\label{ex:g9:weight-vs-mass:planets}
Tiap dunia menetapkan $g$-nya sendiri, dan kisah antariksawan yang
lama pun menjadi sebuah tabel. Koper \qty{10}{kg} setiamu:
\begin{center}
\begin{tabular}{l|c|c}
tempat & $g$ (\unit{N/kg}) & \omterm{def:g4:weight-and-mass:weight}{berat} kopernya \\
\hline
Bumi & $9.8$ & \qty{98}{N} \\
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} & $1.6$ & \qty{16}{N} \\
Mars & $3.7$ & \qty{37}{N} \\
Puncak awan Jupiter & $24.8$ & \qty{248}{N}
\end{tabular}
\end{center}
Sepuluh \omterm{def:g3:measuring-mass:units}{kilogram} zat di mana-mana --- dan \omterm{def:g4:weight-and-mass:weight}{berat} yang berbeda di setiap
dunianya: \omterm{def:g3:measuring-mass:mass}{massanya} tidak berubah, sedangkan \omterm{def:g4:weight-and-mass:weight}{beratnya} mengikuti $g$
setempatnya.
\end{example}

\begin{remark}[Kedua timbangannya, diadili untuk selamanya]\label{rem:g9:weight-vs-mass:scales}
Kini putusan masa kecilnya ditutup dengan rumus. \omterm{def:g3:levers-and-scales:scale}{Neracanya}
membandingkan $m_1 g$ terhadap $m_2 g$: $g$ setempatnya mengalikan
kedua piringnya lalu meniadakan diri --- \omterm{def:g3:levers-and-scales:scale}{neraca} \omterm{def:g6:measurement-in-science:measuring}{mengukur} \emph{\omterm{def:g3:measuring-mass:mass}{massa}},
dan benar di dunia mana pun. Timbangan pegas membaca $P = m g$ secara
langsung: ia \omterm{def:g6:measurement-in-science:measuring}{mengukur} \emph{\omterm{def:g4:weight-and-mass:weight}{berat}}, dan setia kepada tempatnya.
Timbangan kamar mandi adalah timbangan pegas yang menyamar, yang
ditandai pabriknya dalam \omterm{def:g3:measuring-mass:units}{kilogram} dengan membagi beratmu dengan $g$
\emph{milik Bumi} --- jujur di rumah, penyanjung di \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
(\qty{10}{kg} terbaca tiap \qty{60}{kg} dirimu), dan pemfitnah di
Jupiter.
\end{remark}

\section{Gaya di atas kertas}

\begin{example}[Buku di atas meja, dalam newton]\label{ex:g9:weight-vs-mass:book}
\omterm{def:g2:balance-equilibrium:equilibrium}{Kesetimbangan} yang tertua di dalam buku ini, akhirnya dengan bilangan:
sebuah kamus \qty{2}{kg} berbaring di atas sebuah meja. \omterm{def:g4:weight-and-mass:weight}{Beratnya}: $P = 2
\times 9.8 = \qty{19.6}{N}$, yang digambar sebagai anak panah sepanjang
\qty{2}{cm} pada skala sepuluh \omterm{def:g9:weight-vs-mass:newton}{newton} tiap \omterm{def:g2:measuring-length:units}{sentimeter}, yang menunjuk
lurus ke bawah dari bukunya. Dorongan penyangga mejanya:
\qty{19.6}{N} lurus ke atas --- anak panah yang sama, arah yang
berlawanan. Ikatannya tepat, dan diamnya adalah buktinya: gambaran masa
kecilnya, kini sebuah anggaran yang terukur.
\end{example}

\begin{remark}[Apa yang dilakukan gaya --- hukum yang belum ada]\label{rem:g9:weight-vs-mass:dynamics}
Bab ini \omterm{def:g6:measurement-in-science:measuring}{mengukur} \omterm{def:g2:pushes-and-pulls:force}{gaya}; ia belum mengatakan apa yang dilakukan
\emph{kelebihan} \omterm{def:g2:pushes-and-pulls:force}{gaya} terhadap gerak. Keempat pekerjaan masa kecilnya
--- memulai, menghentikan, membelokkan, mengubah bentuk --- menanti
hukumnya yang tepat, dan hukum itu termasuk yang terbesar di seluruh
fisika: jilid Sekolah Menengah membuka mekanikanya dengan itu. Tahun
ini menyelesaikan pekerjaan dasarnya: \omterm{def:g2:pushes-and-pulls:force}{gaya} dalam \omterm{def:g9:weight-vs-mass:newton}{newton}, \omterm{def:g4:weight-and-mass:weight}{berat} menurut
rumus, anak panah menurut skala --- yaitu tata bahasa bagi kalimat yang
akan datang.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:1}
Sebutkan \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g2:pushes-and-pulls:force}{gaya} dan alatnya. Berikan dua jangkar sehari-hari
untuk satu \omterm{def:g9:weight-vs-mass:newton}{newton} dan untuk seratus \omterm{def:g9:weight-vs-mass:newton}{newton}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:2}
Tuliskan rumus \omterm{def:g4:weight-and-mass:weight}{beratnya} beserta \omterm{def:g6:measurement-in-science:measuring}{satuannya}, dan arti $g$ dengan
kata-kata.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:3}
Hitunglah \omterm{def:g4:weight-and-mass:weight}{berat} di Bumi: seekor anjing \qty{25}{kg}; seketul roti
\qty{700}{g}; dirimu sendiri (pilihlah massamu).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:4}
Sebutir melon yang tergantung meregangkan \omterm{def:g9:weight-vs-mass:dynamometer}{dinamometernya} sampai
\qty{14.7}{N}. Berapa \omterm{def:g3:measuring-mass:mass}{massanya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:5}
Bacalah grafik babnya: apa arti kemiringannya, dan apa yang akan
dilakukan garisnya di \omterm{def:g2:moon-in-the-sky:moon}{Bulan}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:6}
Koper \qty{10}{kg} itu berkeliling Bumi, \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, Mars, Jupiter.
Bilangan yang mana di dalam tabelnya yang tidak pernah berubah, dan
yang mana yang berubah --- lalu mengapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:7}
Mengapa sebuah \omterm{def:g3:levers-and-scales:scale}{neraca} berkata benar di setiap dunia sementara
timbangan kamar mandi menyanjung di \omterm{def:g2:moon-in-the-sky:moon}{Bulan}? Dengan rumus, bukan dengan
perasaan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:8}
Gambarlah (atau gambarkan) kedua anak panah \omterm{def:g2:pushes-and-pulls:force}{gaya} milik kamus
\qty{2}{kg} itu pada satu \omterm{def:g2:measuring-length:units}{sentimeter} tiap lima \omterm{def:g9:weight-vs-mass:newton}{newton}: panjangnya,
arahnya, dan putusan anggarannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:9}
Sebuah maskapai penerbangan mengizinkan \qty{23}{kg} tiap tas. Seorang
penumpang berhujah: ``Di \omterm{def:g2:moon-in-the-sky:moon}{Bulan} ia hanya akan berbobot beberapa
\omterm{def:g9:weight-vs-mass:newton}{newton}!'' Mengapa batas maskapainya --- bahan bakar dan kekuatan
lantainya dikesampingkan --- sepatutnya memedulikan \omterm{def:g3:measuring-mass:mass}{massa}, dan alat
apa yang harus memeriksanya di sebuah bandara bulan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:10}
Di Mars, lengan sebuah penjelajah berbobot \qty{111}{N}. Berapa
\omterm{def:g3:measuring-mass:mass}{massanya}? Berapa \omterm{def:g4:weight-and-mass:weight}{berat} lengan yang sama itu kembali di aula
perakitannya di Bumi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:11}
Sebuah \omterm{def:g9:weight-vs-mass:dynamometer}{dinamometer} yang dinolkan ketika terbaring rata, lalu dipakai
menggantung, berdusta sedikit --- ke arah mana, dan mengapa? (\omterm{def:g4:weight-and-mass:weight}{Berat}
tambahan apa yang bergabung dengan \omterm{def:g4:weight-and-mass:weight}{berat} bebannya?)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:weight-vs-mass:12}
Rancanglah percobaan yang menggambar grafik babnya lalu menyarikan
$g$: perlengkapannya, tabel pengukurannya, gambarnya, dan bagaimana
kemiringannya dibaca. Lalu jelaskan apa yang akan diserahkan tata cara
yang sama itu kalau dijalankan di sebuah laboratorium \omterm{def:g2:moon-in-the-sky:moon}{Bulan} ---
bilangan yang mana yang berubah, cara yang mana yang selamat tanpa
tersentuh.
\end{exercise}

\section{Soal: Toko Perlengkapan Antarplanet}

\begin{problem}[{Soal akhir pekan --- ruang pas milik toko
perlengkapan antarplanet; satu tubuh, empat dunia; timbangan,
dinamometer, dan meja bagasinya}]\label{pb:g9:weight-vs-mass:1}
Tokonya melengkapi para pelancong untuk pelabuhan \omterm{def:g4:solar-system:family}{Tata Suryanya}
($g$, dalam \unit{N/kg}: Bumi $9.8$; \omterm{def:g2:moon-in-the-sky:moon}{Bulan} $1.6$; Mars $3.7$;
geladak awan Jupiter $24.8$). Kamu bertugas di ruang pasnya.

\medskip\noindent\textbf{Bagian I --- Pengepasannya.}
Seorang pelancong bermassa \qty{70}{kg} maju.
\begin{enumerate}
  \item Berapa \omterm{def:g4:weight-and-mass:weight}{beratnya} di masing-masing keempat pelabuhannya?
  \item Di pelabuhan yang mana timbangan kamar mandi yang klasik
        (yang bertanda dalam \omterm{def:g3:measuring-mass:units}{kilogram}, tertera untuk Bumi) membaca
        \omterm{def:g3:measuring-mass:mass}{massanya} dengan benar --- dan berapa yang dibacanya di ketiga
        yang lain? (Bagilah tiap \omterm{def:g4:weight-and-mass:weight}{beratnya} dengan $g$ Bumi.)
  \item Timbangan jujur tokonya berupa sebuah \omterm{def:g3:levers-and-scales:scale}{neraca} dengan \omterm{def:g3:measuring-mass:mass}{massa}
        bertanda. Berapa yang dilaporkannya di tiap pelabuhannya, dan
        mengapa?
  \item Sepatu bot pelancongnya tidak boleh menekan tanahnya dengan
        lebih daripada \qty{2000}{N} (lantai stasiunnya tipis). Adakah
        pelabuhan tempat sepatu-bot-ditambah-tubuhnya (totalnya
        \qty{70}{kg}) melanggar aturannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Meja bagasinya.}
\begin{enumerate}[resume]
  \item Peti yang menuju Mars bermassa \qty{40}{kg}. Hitunglah
        \omterm{def:g4:weight-and-mass:weight}{beratnya} pada keberangkatannya (Bumi) dan pada
        kedatangannya (Mars).
  \item Serikat kuli angkut membatasi \emph{\omterm{def:g4:weight-and-mass:weight}{berat}} yang dibawa
        tangan pada \qty{200}{N} di setiap pelabuhan. \emph{\omterm{def:g3:measuring-mass:mass}{Massa}}
        terbesar berapa yang boleh dibawa seorang kuli di
        masing-masingnya?
  \item Sebuah peti berlabel ``\qty{150}{N} di \omterm{def:g2:moon-in-the-sky:moon}{Bulan}'' harus lulus
        bea cukai Bumi menurut \omterm{def:g3:measuring-mass:mass}{massa}. Lakukanlah pengubahannya.
  \item \omterm{def:g9:weight-vs-mass:dynamometer}{Dinamometer} mejanya membaca sebuah tas jinjing yang
        menggantung sebesar \qty{294}{N} di Bumi. Berapa \omterm{def:g3:measuring-mass:mass}{massanya},
        dan berapa \omterm{def:g4:weight-and-mass:weight}{beratnya} di \omterm{def:g2:moon-in-the-sky:moon}{Bulan} untuk daftar muatannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Keluhan dan hal yang
menarik.}
\begin{enumerate}[resume]
  \item Seorang binaragawan pulang dari \omterm{def:g2:moon-in-the-sky:moon}{Bulan} dengan marah:
        ``Tambahan kekuatan enam kali lipat, lenyap dalam
        semalam!'' Hiburlah dia dengan rumusnya: apa sebenarnya yang
        berubah di tiap pelabuhannya, dan apa yang tidak pernah?
  \item Seorang juru masak geladak Jupiter mengeluh bahwa mengocok
        terasa seperti mengangkat landasan tempa, padahal
        pengocoknya ``bermassa hanya \qty{200}{g}''. Hitunglah \omterm{def:g4:weight-and-mass:weight}{berat}
        pengocoknya di geladak itu lalu berilah putusan.
  \item Toko cendera matanya menjual ``senewton cokelat, di mana
        pun''. Di pelabuhan yang mana pembelinya memperoleh
        \emph{zat} cokelat yang terbanyak, dan berapa \omterm{def:g3:measuring-mass:mass}{massanya} di
        Bumi, \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, dan Jupiter masing-masing?
  \item Tulislah semboyan tokonya: satu kalimat yang memisahkan apa
        yang bepergian bersamamu dari apa yang ditagih tiap dunianya.
\end{enumerate}
\end{problem}
