\chapter{Daya dan Energi Listrik}\label{ch:g9:electric-power-energy}

Tagihan listriknya tidak menagih \omterm{def:g8:voltage-voltmeter:voltage}{volt}, dan tidak pula \omterm{def:g8:intensity-ammeter:intensity}{ampere} --- ia
menagih sesuatu yang telah diam-diam dikalikan bersama oleh perkakas
keluargamu sepanjang bulan. Bab ini menikahkan kedua besaran
listriknya menjadi yang ketiga, memberi nama yang sepatutnya kepada
pembelanjaan yang cepat dan yang lambat, lalu berakhir di meteran di
dekat pintu depan: fisika yang dapat kamu periksa terhadap uang yang
sungguhan.

\section{Daya: laju pengantaran}

\begin{definition}[Daya dan watt]\label{def:g9:electric-power-energy:power}
\emph{Daya}\index{daya} $P$ sebuah alat adalah laju ia mengubah \omterm{def:g5:everyday-energy:energy}{energi}
--- \omterm{def:g9:kinetic-energy-safety:joule}{joule} tiap sekon. \omterm{def:g6:measurement-in-science:measuring}{Satuannya} adalah \emph{watt}\index{watt}
(\unit{W}): satu \omterm{def:g9:kinetic-energy-safety:joule}{joule} tiap sekon, yang menghormati insinyur zaman uap;
\emph{kilowatt} (\qty{1}{kW} $=$ \qty{1000}{W}) melayani para pemberat
dapurnya. Daya bukanlah jumlah \omterm{def:g5:everyday-energy:energy}{energi} melainkan sebuah \emph{langkah}
pembelanjaan --- aliran sebuah keran, bukan isi sebuah ember.
\end{definition}

\begin{proposition}[Daya listrik]\label{prop:g9:electric-power-energy:UI}
Sebuah alat yang diberi \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} $U$ dan menarik arus $I$ menerima \omterm{def:g9:electric-power-energy:power}{daya}
listrik
\[
P = U \times I ,
\]
\omterm{def:g9:electric-power-energy:power}{watt} dari \omterm{def:g8:voltage-voltmeter:voltage}{volt} kali \omterm{def:g8:intensity-ammeter:intensity}{ampere}. Rumusnya menghitung ke ketiga arahnya, dan
resep keduanya menjalankan kotak \omterm{def:g7:short-circuits-safety:fuse}{sekringnya}: $I = P/U$ memberitahu arus
apa yang akan ditarik \omterm{def:g9:electric-power-energy:power}{daya} tertakar sebuah perkakas dari listrik
rumahnya.
\end{proposition}

\begin{example}[Tangga rumah tangganya]\label{ex:g9:electric-power-energy:ladder}
Bacalah pelat pada mesin keluarganya: sebuah lampu penunjuk,
\qty{1}{W}; sebuah lampu zaman sekarang, \qty{5}{W} (leluhurnya yang
berpijar membelanjakan enam puluh untuk \omterm{def:g1:light-and-shadows:source}{cahaya} yang sama ---
selisihnya berupa panas); sebuah komputer jinjing, \qty{50}{W}; sebuah
televisi, \qty{100}{W}; sebuah mesin cuci yang memanaskan airnya,
\qty{2000}{W}; cereknya, \qty{2300}{W}; sebuah pemanas air seketika,
sampai \qty{9000}{W}. Pola tangganya: apa pun yang tugasnya
\emph{memanaskan} memanjat ke kilowatt --- kehangatan adalah bentuk
\omterm{def:g5:everyday-energy:energy}{energi} yang paling mahal untuk dipasok.
\end{example}

\begin{example}[Hitungan kotak sekringnya]\label{ex:g9:electric-power-energy:fuses}
Cereknya pada listrik rumah \qty{230}{V}: $I = 2300 \div 230 =
\qty{10}{A}$ --- perkakas sepuluh \omterm{def:g8:intensity-ammeter:intensity}{ampere} pada bab yang lama, kini
diturunkan. Pemanas air \qty{9000}{W}: hampir \qty{40}{A}, yang
menuntut \omterm{def:g5:series-parallel-circuits:parallel}{cabang} dan pemutus arusnya sendiri yang kekar. Dan stopkontak
jamak yang kelebihan beban pada penyelidikan kebakarannya: cerek,
pemanas, dan penggorengnya menjumlahkan arus $P/U$ mereka melewati
takaran \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} --- tiga yang tidak bersalah, satu kejahatan
hitungan, tepat sebagaimana didakwakan.
\end{example}

\section{Energi: jumlah yang diantarkan}

\begin{proposition}[Energi dari daya dan waktu]\label{prop:g9:electric-power-energy:Pt}
Sebuah alat berdaya $P$ yang berjalan selama waktu $t$ mengubah \omterm{def:g5:everyday-energy:energy}{energi}
sebesar
\[
E = P \times t ,
\]
\omterm{def:g9:kinetic-energy-safety:joule}{joule} dari \omterm{def:g9:electric-power-energy:power}{watt} kali sekon. Keran yang cepat atau rembesan yang
panjang, embernya terisi menurut hasil kalinya: tiga menit sebuah
cerek \qty{2300}{W} ($2300 \times 180 \approx \qty{4.1e5}{J}$)
membelanjakan lebih banyak daripada sehari penuh sebuah lampu
\qty{5}{W} ($5 \times
86400 = \qty{4.3e5}{J}$) --- sebenarnya hampir tepat seri: satu
pendidihan setara dengan sehari \omterm{def:g1:light-and-shadows:source}{cahaya} yang baik.
\end{proposition}

\begin{definition}[Kilowatt-jam]\label{def:g9:electric-power-energy:kwh}
\omterm{def:g5:everyday-energy:energy}{Energi} rumah tangga menghasilkan hitungan \omterm{def:g9:kinetic-energy-safety:joule}{joule} yang janggal, sehingga
meterannya berbicara \emph{kilowatt-jam}\index{kilowatt-jam}
(\unit{kWh}): \omterm{def:g5:everyday-energy:energy}{energi} satu kilowatt yang dipertahankan selama satu \omterm{ex:g2:measuring-time:clock}{jam},
\[
\qty{1}{kWh} = 1000 \times 3600 = \qty{3.6e6}{J}
\]
--- tiga koma enam juta \omterm{def:g9:kinetic-energy-safety:joule}{joule} tiap \omterm{def:g6:measurement-in-science:measuring}{satuan} pada tagihannya. Dengan $P$
dalam kilowatt dan $t$ dalam \omterm{ex:g2:measuring-time:clock}{jam}, $E = P \times t$ menyerahkan
kilowatt-jam secara langsung: itulah rumus yang menjadi hidup
meterannya.
\end{definition}

\begin{method}[Memeriksa sebuah perkakas]\label{met:g9:electric-power-energy:audit}
Untuk mesin dan bulan mana pun:
\begin{enumerate}
  \item bacalah \omterm{def:g9:electric-power-energy:power}{dayanya} $P$ pada pelatnya (atau hitunglah
        $U \times I$);
  \item taksirlah dengan jujur waktu jalannya $t$ --- tiap hari, lalu
        tiap bulan;
  \item $E = P \times t$ dalam \omterm{def:g9:electric-power-energy:kwh}{kilowatt-jam} ($P$ dalam \unit{kW}, $t$
        dalam \omterm{ex:g2:measuring-time:clock}{jam});
  \item kalikan dengan tarifnya (ambillah kira-kira $0.25$ \omterm{def:g6:measurement-in-science:measuring}{satuan} mata
        uang tiap \unit{kWh}) untuk memperoleh biaya bulannya.
\end{enumerate}
Cereknya, enam menit tiap hari: $2.3 \times 0.1 \times 30 =
\qty{6.9}{kWh}$ tiap bulan. Televisinya, tiga \omterm{ex:g2:measuring-time:clock}{jam} tiap hari:
$0.1 \times 3 \times 30 = \qty{9}{kWh}$. \omterm{def:g9:electric-power-energy:power}{Daya} yang kecil dengan \omterm{ex:g2:measuring-time:clock}{jam}
yang panjang menyaingi \omterm{def:g9:electric-power-energy:power}{daya} yang besar dengan \omterm{ex:g2:measuring-time:clock}{jam} yang pendek ---
itulah pelajaran pemeriksaannya yang berulang.
\end{method}

\section{Meteran dan tagihannya}

\begin{example}[Membaca tagihannya]\label{ex:g9:electric-power-energy:bill}
Meteran di dekat pintu depan menghitung setiap \omterm{def:g9:electric-power-energy:kwh}{kilowatt-jam} yang masuk
ke rumahnya; tagihannya adalah selisih sebulannya dikali tarifnya.
Angka $250$ \omterm{def:g6:measurement-in-science:measuring}{satuan} bulanan sebuah keluarga yang khas terurai menurut
pemeriksaannya: pemanasan air dan pemanas ruangannya lebih dahulu (klub
kilowattnya), lalu perkakas dinginnya --- \omterm{def:g9:electric-power-energy:power}{wattnya} sederhana, tetapi
berjalan selalu --- lalu memasak, mencuci, \omterm{def:g1:light-and-shadows:source}{cahaya}, dan elektronikanya.
Kekuatan pemeriksaannya: ia menemukan \omterm{def:g3:levers-and-scales:lever}{tuas} yang sesungguhnya.
Mengganti sepuluh \omterm{def:g3:first-electric-circuit:bulb}{bohlam} tua \qty{60}{W} dengan yang \qty{5}{W}
menghemat lebih banyak daripada mencabut setiap pengisi \omterm{def:g9:electric-power-energy:power}{daya} di
rumahnya seribu kali lipat.
\end{example}

\begin{example}[Vampir siaganya]\label{ex:g9:electric-power-energy:standby}
Cetakan kecil, \omterm{ex:g2:measuring-time:clock}{jam} yang panjang: lampu siaga sebuah televisi
menyeruput \qty{0.5}{W} --- tetapi selama seluruh $720$ \omterm{ex:g2:measuring-time:clock}{jam} bulannya:
$0.0005 \times 720 = \qty{0.36}{kWh}$. Satu vampir tidak berbahaya;
sebuah rumah dengan dua puluh penyeruput sewatt masing-masing membayar
$0.02 \times 720 \approx \qty{14}{kWh}$ tiap bulan untuk tidak apa-apa.
\omterm{def:g9:electric-power-energy:power}{Daya} kali waktu tidak memaafkan suku mana pun: faktor keduanya
menggigit ketika yang pertama bersembunyi.
\end{example}

\begin{remark}[Daya di dunia yang lebih luas]\label{rem:g9:electric-power-energy:wider}
Watt, yang lahir sebagai \omterm{def:g6:measurement-in-science:measuring}{satuan} listrik di sini, \omterm{def:g6:measurement-in-science:measuring}{mengukur} setiap
langkah \omterm{def:g5:everyday-energy:energy}{energi}. Tubuhmu yang beristirahat menganggur dekat
\qty{100}{W} --- sebuah bohlam-leluhur kehangatan yang berpijar
(ingatlah pengapnya ruang kelas yang penuh sesak: tiga puluh murid,
tiga kilowatt biologi). Seorang pengendara sepeda bertahan pada
\qty{200}{W}; mesin sebuah mobil melepaskan puluhan kilowatt;
pembangkit listrik kotanya, ratusan juta \omterm{def:g9:electric-power-energy:power}{watt}; \omterm{def:g1:light-and-shadows:source}{cahaya} matahari yang
jatuh pada kotanya, dengan nyaman lebih banyak lagi --- itulah
langit-langit yang di bawahnya semua langkah yang lain bekerja. Bab
berikutnya mengikuti \omterm{def:g9:electric-power-energy:power}{watt} jaringannya sampai ke sumbernya yang
berputar.
\end{remark}

\section{Soal: Pemeriksaan Energi Rumah}

\begin{problem}[{Soal akhir pekan --- pemeriksaan energi keluarganya;
pelat, meteran, dan tagihan bulannya; menemukan tuas yang
sesungguhnya}]\label{pb:g9:electric-power-energy:1}
Berbekal kedua rumusnya dan tarifnya ($0.25$ tiap \unit{kWh}),
keluarganya memeriksa sebulan (30 hari). Daftarnya: cerek
\qty{2300}{W}, $6$ menit tiap hari; televisi \qty{100}{W}, $4$ \omterm{ex:g2:measuring-time:clock}{jam}
tiap hari; lemari es --- perkara khusus di bawah; sepuluh lampu
\qty{5}{W} masing-masing, $5$ \omterm{ex:g2:measuring-time:clock}{jam} tiap hari; mesin cuci \qty{2}{kW},
satu daur panas $1$ \omterm{ex:g2:measuring-time:clock}{jam} tiap dua hari; pemanas air \qty{2.5}{kW}, $2$
\omterm{ex:g2:measuring-time:clock}{jam} tiap hari; penyeruput siaga yang berjumlah \qty{10}{W}, selalu
menyala.

\medskip\noindent\textbf{Bagian I --- Mesin demi mesin.}
\begin{enumerate}
  \item Berapa sebulan cereknya, dalam \unit{kWh}.
  \item Berapa sebulan televisinya dan sepuluh lampunya.
  \item Berapa sebulan mesin cucinya ($15$ daur) dan pemanas airnya.
  \item Berapa sebulan penyeruput siaganya --- faktor kedua yang
        sedang bekerja.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Lemari es dan
jumlahnya.}
Kompresor lemari esnya berjalan pada \qty{120}{W}, tetapi hanya
sepertiga waktunya, berdaur siang dan malam.
\begin{enumerate}[resume]
  \item Berapa \omterm{def:g9:electric-power-energy:power}{daya} rata-rata efektifnya, dan berapa sebulannya dalam
        \unit{kWh}.
  \item Jumlahkan ketujuh baris pemeriksaannya. Ketiga baris yang mana
        yang menguasainya?
  \item Berapa tagihan bulannya pada tarifnya --- dan berapa bagian
        pemanas airnya di dalamnya dalam mata uang.
  \item Meterannya membaca $2150$ \omterm{def:g6:measurement-in-science:measuring}{satuan} pada awal bulannya. Ramalkan
        bacaannya pada akhir bulannya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Tuasnya.}
\begin{enumerate}[resume]
  \item Usul satu: mandi yang lebih singkat --- pemanasnya turun
        menjadi \qty{1.5}{h} tiap hari. Berapa penghematan bulanannya,
        dalam \omterm{def:g5:everyday-energy:energy}{energi} dan dalam uang?
  \item Usul dua: sebuah kampanye umum mendesak orang mencabut pengisi
        \omterm{def:g9:electric-power-energy:power}{daya} telepon (\qty{0.2}{W} masing-masing ketika menganggur).
        Periksalah sebulan menganggurnya satu pengisi \omterm{def:g9:electric-power-energy:power}{daya} lalu
        bandingkan dengan usul satu; apa yang diajarkan
        perbandingannya tentang pemeriksaan lawan semboyan?
  \item Usul tiga: neneknya menyarankan kebiasaan lama mengisi
        cereknya hanya seperlunya --- separuh airnya, separuh
        \omterm{def:g5:everyday-energy:energy}{energinya}. Berapa baris cereknya yang baru, dan berapa
        penghematan bulannya?
  \item Tulislah kesimpulan pemeriksaannya: tiga kalimat --- di mana
        \omterm{def:g9:electric-power-energy:kwh}{kilowatt-jamnya} sungguh berdiam, faktor rumus yang mana yang
        diajarkan perkara siaganya, dan satu perubahan rumah tangga
        yang mana yang paling menguntungkan.
\end{enumerate}
\end{problem}
