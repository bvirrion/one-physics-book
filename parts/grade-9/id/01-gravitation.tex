\chapter{Gravitasi}\label{ch:g9:gravitation}

Sebuah apel melepaskan diri dari dahannya lalu jatuh. \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, di atas
kebun yang sama, tidak. Selama ribuan tahun keduanya tampak sebagai
dua kenyataan yang tidak berhubungan --- sampai satu wawasan yang
jenius menyatukannya: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} \emph{sedang} jatuh, tanpa henti,
mengelilingi Bumi, ditahan oleh tarikan yang sama yang menuntut
apelnya. Satu hukum bagi kebun dan langit --- persamaan paling berani
yang pernah ditulis sampai hari itu, dan tahun terakhirmu pada buku
ini dibuka dengannya.

\section{Tarikan yang semesta}

\begin{definition}[Gravitasi]\label{def:g9:gravitation:gravitation}
\emph{Gravitasi}\index{gravitasi} adalah tarikan timbal balik di
antara semua \omterm{def:g3:measuring-mass:mass}{massa}: setiap bagian materi di alam semesta menarik
setiap bagian yang lain. Ia adalah \omterm{def:g2:pushes-and-pulls:force}{gaya} tanpa sentuhan yang kamu temui
sebagai ``tarikan Bumi'' bertahun-tahun yang lalu --- kini terungkap
sebagai semesta: Bumi menarik apelnya, apelnya menarik Bumi, \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
menarik lautan, Matahari menarik \omterm{def:g4:solar-system:planet}{planet}, dan dua buku di sebuah rak
saling menarik, dengan lemah, untuk selamanya.
\end{definition}

\begin{proposition}[Hukum gravitasi semesta]\label{prop:g9:gravitation:law}
Dua benda bermassa $m_A$ dan $m_B$ (dalam \unit{kg}), dengan pusatnya
terpisah sejauh $d$ (dalam \unit{m}), saling menarik dengan gaya yang
sama besar dan berlawanan arah di sepanjang garis yang
menghubungkannya, dengan kekuatan
\[
F = G \times \frac{m_A \times m_B}{d^2},
\]
$F$ dalam newton (\unit{N}) --- \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g2:pushes-and-pulls:force}{gaya} yang kisah penuhnya mulai
pada bab berikutnya --- dengan
$G = \qty{6.67e-11}{N.m^2/kg^2}$, yaitu \emph{tetapan \omterm{def:g9:gravitation:gravitation}{gravitasi}},
bilangan yang sama di mana pun di alam semesta. \omterm{def:g2:pushes-and-pulls:force}{Gayanya} bertumbuh
menurut tiap \omterm{def:g3:measuring-mass:mass}{massanya}, dan mati menurut \emph{kuadrat} jaraknya: dua
kali lebih jauh, empat kali lebih lemah; sepuluh kali lebih jauh,
seratus kali lebih lemah.
\end{proposition}

\begin{example}[Membaca hukumnya]\label{ex:g9:gravitation:reading}
Setiap ciri rumusnya berbicara. \omterm{def:g3:measuring-mass:mass}{Massanya} mengalikan: gandakan salah
satu pasangannya, gandalah \omterm{def:g2:pushes-and-pulls:force}{gayanya}. Jaraknya dikuadratkan di
penyebutnya: \omterm{def:g9:gravitation:gravitation}{gravitasi} adalah bisikan yang memudar dengan cepat ---
pengenceran kuadrat terbalik yang akan kamu temui lagi bersama \omterm{def:g1:light-and-shadows:source}{cahaya}
dan \omterm{def:g3:sound-around-us:sound}{bunyi}, karena ia adalah geometri apa pun yang menyebar ke dalam
ruang. Dan ukuran $G$ yang sangat kecil --- sebelas nol sesudah
titiknya --- memaklumkan \omterm{def:g9:gravitation:gravitation}{gravitasi} sebagai yang terlemah di antara
\omterm{def:g2:pushes-and-pulls:force}{gaya} alam: ia menguasai kosmos hanya karena \omterm{def:g3:measuring-mass:mass}{massa} di sana raksasa dan
tidak ada yang meniadakannya.
\end{example}

\begin{example}[Mengapa rak tidak saling membanting]\label{ex:g9:gravitation:books}
Dua buku \qty{1}{kg}, dengan pusatnya terpisah \qty{0.1}{m}:
\[
F = \num{6.67e-11} \times \frac{1 \times 1}{0.1^2}
= \qty{6.67e-9}{N}
\]
--- beberapa permiliaran newton, seberat serpih dari serpih sebutir
debu. Kini Bumi dan apelnya: gantilah satu bukunya dengan \omterm{def:g4:solar-system:planet}{planet}
bermassa \qty{5.97e24}{kg} dan jaraknya dengan jari-jarinya
\qty{6.4e6}{m}, dan hukum yang sama menyerahkan tarikan yang akrab
pada sore di kebun. Semesta memang berarti semesta --- hanya \omterm{def:g3:measuring-mass:mass}{massanya}
yang memutuskan siapa yang memperhatikan.
\end{example}

\section{Bulan yang jatuh}

\begin{example}[Meriam di atas gunung]\label{ex:g9:gravitation:cannon}
Percobaan pikiran milik wawasannya sendiri. Tembakkan sebutir peluru
meriam mendatar dari sebuah puncak yang tinggi: ia melengkung lalu
mendarat. Tembakkan lebih cepat: ia mendarat lebih jauh --- dengan
tarikan Bumi membengkokkan jalurnya sepanjang waktu itu. Kini
bayangkan menembakkannya begitu cepat sehingga, selagi ia jatuh,
permukaan Bumi yang bulat \emph{melengkung menjauh di bawahnya dengan
laju yang sama}: bolanya jatuh tanpa henti lalu tidak pernah mendarat
--- ia mengelilingi \omterm{def:g4:solar-system:planet}{planetnya}. Kejatuhan yang abadi itu mempunyai nama
yang akrab: sebuah \emph{\omterm{def:g6:sun-earth-moon:axisorbit}{orbit}}. Tidak ada yang menahan benda yang
mengorbit ``tetap di atas''; ia sedang jatuh dengan gaya.
\end{example}

\begin{proposition}[Apa yang dijalankan gravitasi]\label{prop:g9:gravitation:runs}
Satu hukum, seluruh langitnya: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} adalah peluru meriam abadi milik
Bumi, yang jatuh mengelilingi kita sekali tiap bulan; \omterm{def:g4:solar-system:planet}{planetnya} jatuh
demikian pula mengelilingi Matahari --- yang lebih dekat lebih cepat,
sebagaimana dituntut kuadrat terbaliknya; komet menerjun masuk dari
kegelapan pada lingkar jatuh yang panjang; dan setiap satelit buatan
--- mata cuaca, mercusuar pandu arah, stasiun berawak yang berkeliling
tiap sembilan puluh menit --- menunggangi matematika meriam gunungnya.
Tidak ada di langit yang tergantung. Segalanya jatuh, lalu luput.
\end{proposition}

\begin{example}[Pasang surut: Bulan menarik balik]\label{ex:g9:gravitation:tides}
Sifat timbal baliknya mempunyai tanda tangan yang tertulis di air.
Tarikan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pada Bumi paling kuat pada lautan yang menghadapnya dan
paling lemah di sisi yang jauh --- sehingga lautannya menumpuk lembut
ke arah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} dan menjauhinya sekaligus, dan selagi Bumi berputar di
bawah tumpukannya, pantainya melihat airnya naik lalu turun: itulah
\emph{pasang surut}, dua kali pada kebanyakan hari. Matahari
menambahkan tangannya --- sejajar pada bulan baru dan purnama untuk
pasang purnama yang besar. Nelayan telah membaca \omterm{def:g2:moon-in-the-sky:moon}{Bulan} selama
ribuan tahun; hukumnya membacakannya kembali.
\end{example}

\begin{remark}[Apa yang tidak dikatakan hukumnya]\label{rem:g9:gravitation:notsay}
Persamaan besar itu menghitung \omterm{def:g2:pushes-and-pulls:force}{gayanya} dengan ketelitian yang indah
lalu sama sekali tidak menjelaskan \emph{bagaimana} dua benda
mencengkeram menyeberangi kehampaan --- pengarangnya sendiri menyebut
pertanyaannya terbuka lalu meninggalkannya, dengan megah, tanpa
jawaban. Jawaban yang lebih dalam memang ada --- tahun-tahun perguruan
tinggi pada pelajaran ini mencapai jawaban zaman sekarang, yang
padanya \omterm{def:g3:measuring-mass:mass}{massa} melengkungkan geometri yang didiaminya itu sendiri ---
tetapi setiap fisikawan yang jujur sejak itu mengulangi pelajaran yang
boleh kamu ambil dari halaman ini: sebuah hukum dapat sempurna
berguna, sempurna teruji, dan tetap menjaga rahasia terakhirnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:1}
Nyatakan hukum \omterm{def:g9:gravitation:gravitation}{gravitasi} semesta --- rumusnya, \omterm{def:g6:measurement-in-science:measuring}{satuannya}, dan kedua
cirinya (\omterm{def:g3:measuring-mass:mass}{massa}, jarak) dengan kata-kata.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:2}
Bumi menarik apelnya dengan \qty{1}{N}. Dengan \omterm{def:g2:pushes-and-pulls:force}{gaya} berapa apelnya
menarik Bumi? Mengapa hanya satu di antara keduanya yang jelas
terlihat dipercepat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:3}
Dua satelit mengorbit pada jarak $d$ dan $2d$ dari pusat Bumi.
Bandingkan \omterm{def:g2:pushes-and-pulls:force}{gaya} \omterm{def:g9:gravitation:gravitation}{gravitasi} padanya (\omterm{def:g3:measuring-mass:mass}{massanya} sama). Dan pada $10 d$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:4}
Hitunglah tarikan di antara dua pasangan penari bermassa \qty{80}{kg}
yang terpisah \qty{0.5}{m} --- lalu berilah komentar tentang asmara
lawan \omterm{def:g9:gravitation:gravitation}{gravitasi}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:5}
Pada kisah meriam gunungnya, dua laju apa yang berpadanan dengan tepat
bagi tembakan yang mengorbit? Apa itu \omterm{def:g6:sun-earth-moon:axisorbit}{orbit}, dalam tiga kata?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:6}
``Para antariksawan mengapung karena tidak ada \omterm{def:g9:gravitation:gravitation}{gravitasi} di atas
sana.'' Runtuhkan ini dengan peluru meriamnya: stasiun dan awaknya
sama-sama sedang melakukan apa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:7}
Mengapa \omterm{def:g9:gravitation:gravitation}{gravitasi}, yang terlemah di alam, tetap saja mengatur
kosmosnya? Dua alasan dari babnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:gravitation:8}
Dari mana datangnya pasang surut, dan mengapa dua kali pada
kebanyakan hari? Kapan yang terbesar menyerang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:gravitation:9}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} (\qty{7.3e22}{kg}) mengorbit pada \qty{3.8e8}{m} dari Bumi
(\qty{5.97e24}{kg}). Hitunglah tarikannya, dengan notasi ilmiah dari
awal sampai akhir --- lalu kagumilah berapa banyak angka yang dibawa
pangkat sepuluh pada jawabannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:gravitation:10}
Di permukaan \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, beratmu \omterm{def:g7:moon-phases:phases}{menyusut} menjadi seperenam (ingatlah
antariksawan yang melompat-lompat). Dengan memakai kedua tombol
hukumnya --- \omterm{def:g3:measuring-mass:mass}{massa} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang lebih kecil, jari-jarinya yang lebih
kecil --- jelaskan bagaimana seperenam dapat muncul dari benda yang
delapan puluh kali lebih ringan daripada Bumi.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:gravitation:11}
Merkurius menempuh satu putaran mengelilingi Matahari dalam $88$ hari,
Neptunus dalam $165$ tahun. Bacalah pola ini dari kuadrat terbaliknya:
apa yang dilakukan \omterm{def:g9:gravitation:gravitation}{gravitasi} yang memudar menurut jarak terhadap
cengkeraman Matahari, dan dengan demikian terhadap langkah keluarga
yang di luar?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:gravitation:12}
Sebuah satelit geostasioner tergantung seakan-akan tak bergerak di
atas satu titik khatulistiwa --- antena parabolanya tidak pernah
bergerak. Rukunkan penampakannya dengan
\cref{prop:g9:gravitation:runs}: berapa kalanya mengorbit, ke arah
mana ia harus berkeliling, dan mengapa ``tergantung'' di sini berarti
``jatuh dalam langkah yang sempurna sejalan dengan Bumi yang
berputar''?
\end{exercise}

\section{Soal: Warisan Sang Peluru Meriam}

\begin{problem}[{Soal akhir pekan --- dari kebun ke orbit; apelnya,
meriamnya, dan Bulan yang diperiksa dengan satu hukum; papan
peluncuran satelitnya}]\label{pb:g9:gravitation:1}
Sesore di meja penyuluhan badan antariksanya: satu hukum,
$G = \qty{6.67e-11}{N.m^2/kg^2}$, \omterm{def:g3:measuring-mass:mass}{massa} Bumi
\qty{5.97e24}{kg}, jari-jari Bumi \qty{6.4e6}{m}.

\medskip\noindent\textbf{Bagian I --- Pemeriksaan apelnya.}
\begin{enumerate}
  \item Hitunglah tarikan Bumi pada sebutir apel \qty{0.1}{kg} di
        permukaannya (pusatnya terpisah satu jari-jari Bumi).
  \item Apelnya menarik Bumi balik dengan gaya berapa? Nyatakan kata
        hukumnya untuk pembukuan dua arah ini.
  \item Angkatlah apelnya sampai dua kali jari-jari Bumi dari
        pusatnya. Apa jadinya tarikannya?
  \item Seorang pengunjung penyuluhannya bertanya mengapa rumusnya
        memakai jarak ke \emph{pusat} Bumi alih-alih ke tanahnya.
        Tawarkan jawaban pendek yang jujur (seluruh \omterm{def:g4:solar-system:planet}{planetnya}
        menarik, dan tarikannya yang berserakan terjumlah seakan-akan
        datang dari pusatnya --- sebuah teorema yang bahkan pengarang
        besarnya sendiri perlu bertahun-tahun untuk membuktikannya).
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Papan peluncurannya.}
\begin{enumerate}[resume]
  \item Satelit hari ini bermassa \qty{2000}{kg}. Berapa \omterm{def:g4:weight-and-mass:weight}{beratnya} di
        permukaan, menurut hukumnya? (Atau menurut jalan pintas yang
        dinamai bab berikutnya: kira-kira \qty{9.8}{N} tiap
        \omterm{def:g3:measuring-mass:units}{kilogram}.)
  \item Pada \omterm{def:g6:sun-earth-moon:axisorbit}{orbit} kerjanya pada \qty{1.28e7}{m} dari pusatnya ---
        dua jari-jari Bumi --- berapa bagian dari \omterm{def:g4:weight-and-mass:weight}{beratnya} di
        permukaan yang masih dikerahkan \omterm{def:g9:gravitation:gravitation}{gravitasi} padanya?
  \item ``Jadi masih ada \omterm{def:g9:gravitation:gravitation}{gravitasi} yang sungguhan di atas sana?''
        Tegaskan dengan bilangannya --- lalu jelaskan apa yang sedang
        dilakukan satelitnya dengan tarikan itu alih-alih menabrak.
  \item Ulasan peluncurannya berkata ``\omterm{def:g7:motion-average-speed:formula}{kelajuan} \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} harus
        berpadanan dengan kejatuhannya''. Ceritakan ulang kisah
        meriam gunungnya dalam dua kalimat untuk brosur para
        pengunjungnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pembukuan \omterm{def:g2:moon-in-the-sky:moon}{Bulannya}.}
\begin{enumerate}[resume]
  \item Dengan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pada \qty{3.8e8}{m}, berapa jari-jari Bumi
        jauhnya ia menunggangi \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya}? (Satu angka di belakang
        koma.)
  \item Menurut kuadrat terbaliknya, \omterm{def:g9:gravitation:gravitation}{gravitasi} di sana telah memudar
        menjadi kira-kira berapa bagian dari kekuatannya di
        permukaan? (Kuadratkanlah jawabanmu yang sebelumnya.)
  \item Wawasan besarnya, diceritakan ulang: apel kebunnya jatuh
        kira-kira lima \omterm{def:g2:measuring-length:units}{meter} pada sekon pertamanya; \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, tiap
        sekonnya, ``jatuh'' ke arah Bumi hanya semilimeter --- namun
        tidak pernah tiba. Hubungkan kedua kenyataannya lewat
        jawabanmu 9 dan 10: mengapa kejatuhan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} ribuan kali lebih
        lembut, dan mengapa tanpa akhir?
  \item Tutuplah sore penyuluhannya: satu hukum, tiga pemeran ---
        apel, satelit, \omterm{def:g2:moon-in-the-sky:moon}{Bulan}. Tulislah kalimat penutup papannya yang
        menyatukan ketiganya.
\end{enumerate}
\end{problem}
