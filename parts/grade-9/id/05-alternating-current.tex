\chapter{Arus Bolak-Balik}\label{ch:g9:alternating-current}

Setiap rangkaian yang pernah kamu bangun berjalan dengan sebuah
\omterm{def:g3:first-electric-circuit:battery}{baterai}: arus yang berarak mantap dari $+$ ke $-$, setia seperti jarum
\omterm{ex:g2:measuring-time:clock}{jam}. Tetapi stopkontak di dindingmu mengantarkan sesuatu yang lebih
liar --- arus yang membalik arah seratus kali tiap sekon, dengan
\omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} berayun tanpa henti di antara plus dan minus. Temuilah \omterm{def:g9:alternating-current:dcac}{arus
bolak-balik}: bentuk yang di dalamnya boleh dikatakan seluruh listrik
dunia lahir, dikirim, dan diantarkan.

\section{Dua jenis arus}

\begin{definition}[Arus searah dan arus bolak-balik]\label{def:g9:alternating-current:dcac}
Sebuah \emph{arus searah}\index{arus searah} (DC) selalu mengalir ke
arah yang sama, digerakkan oleh \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang mantap --- pemberian
\omterm{def:g3:first-electric-circuit:battery}{baterainya}, tetap sampai simpanannya habis. Sebuah \emph{arus
bolak-balik}\index{arus bolak-balik} (AC) membalik arah secara
berkala, digerakkan oleh \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang berayun mulus dari positif ke
negatif lalu kembali, berulang-ulang --- itulah rezim stopkontaknya,
dan rezim alternatornya (bab berikutnya menceritakan milik siapa).
\end{definition}

\begin{definition}[Perioda dan frekuensi]\label{def:g9:alternating-current:period}
\emph{Perioda}\index{perioda} $T$ sebuah \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} bolak-balik adalah
lamanya satu daur lengkap --- satu ayunan penuh naik, turun, dan
kembali, dalam sekon. \emph{\omterm{def:g8:sound-pitch-loudness:frequency}{Frekuensinya}} $f$ adalah banyaknya daur
tiap sekon, dalam \omterm{def:g8:sound-pitch-loudness:frequency}{hertz} --- \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g3:sound-around-us:sound}{bunyinya}, yang mengabdi lagi --- dan
keduanya saling merupakan kebalikan:
\[
f = \frac{1}{T}, \qquad T = \frac{1}{f}.
\]
\end{definition}

\begin{proposition}[Angka penting listrik rumahnya]\label{prop:g9:alternating-current:mains}
Di sebagian besar dunia listrik rumahnya berbolak-balik pada
$f = \qty{50}{Hz}$ --- lima puluh daur penuh tiap sekon, dengan
\omterm{def:g9:alternating-current:period}{perioda} $T = 1/50 = \qty{0.02}{s}$: dua puluh milisekon tiap ayunan
(sebagian wilayah berjalan pada enam puluh). Arus di dalam kabel lampu
bacamu membalik arah seratus kali tiap sekon --- dua kali tiap daurnya
--- dan tidak pernah sekali pun berarak mantap dari kutub yang satu ke
kutub yang lain.
\end{proposition}

\begin{example}[Hitungan frekuensinya]\label{ex:g9:alternating-current:arithmetic}
Pasangan kebalikannya menghitung ke dua arah. Sebuah pembangkit yang
berdaur pada \qty{60}{Hz}: $T = 1/60 \approx \qty{0.017}{s}$ --- tujuh
belas milisekon. Sebuah \omterm{met:g9:alternating-current:scope}{osiloskop} menunjukkan sebuah daur yang
berlangsung \qty{0.04}{s}: $f = 1/0.04 = \qty{25}{Hz}$. Sebuah dinamo
sepeda pada kayuhan yang lambat membuat \qty{10}{Hz}: \omterm{def:g9:alternating-current:period}{periodanya}
sepersepuluh sekon --- dan lampunya jelas berkedip-kedip, sepuluh
denyut tiap sekon, sampai \omterm{def:g7:motion-average-speed:formula}{kelajuannya} melicinkannya di luar
pemperhatian mata.
\end{example}

\section{Membaca ayunannya}

\begin{method}[Membaca jejak sebuah osiloskop]\label{met:g9:alternating-current:scope}
Sebuah \emph{osiloskop}\index{osiloskop} --- atau aplikasi penganalisis mana
pun --- menggambar \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} terhadap waktu pada sebuah layar berpetak;
nilai tiap petaknya dalam \omterm{def:g8:voltage-voltmeter:voltage}{volt} dan dalam milisekon diatur oleh dua
tombol. Untuk memanen sebuah jejak:
\begin{enumerate}
  \item hitunglah petak untuk satu daur penuh lalu kalikan dengan
        waktu tiap petaknya: itulah $T$; lalu $f = 1/T$;
  \item hitunglah petak dari garis tengahnya sampai sebuah puncak lalu
        kalikan dengan \omterm{def:g8:voltage-voltmeter:voltage}{volt} tiap petaknya: itulah \emph{\omterm{def:g8:voltage-voltmeter:voltage}{tegangan}
        puncak} $U_{\text{maks}}$;
  \item periksalah bentuknya: listrik rumah dan alternator menggambar
        gelombang yang mulus dan terulang tanpa henti pada gambarnya
        --- yaitu \emph{sinusoid}, nama matematika bagi ayunan yang
        sempurna.
\end{enumerate}
\end{method}

\begin{proposition}[Tegangan efektif]\label{prop:g9:alternating-current:effective}
Berapa ``sang'' \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} sebuah ayunan yang mengunjungi setiap nilai
dari $+U_{\text{maks}}$ sampai $-U_{\text{maks}}$ seratus kali tiap
sekon? Bilangan tunggal yang jujur adalah \emph{tegangan
efektif}\index{tegangan efektif} $U$: \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} DC yang mantap yang
akan memanaskan sebuah \omterm{def:g8:ohms-law:resistor}{resistor} sama baiknya. Untuk sinusoidnya ia
duduk jauh di bawah puncaknya --- $U_{\text{maks}}$ kira-kira $1.4$
kali $U$ --- dan itulah yang dibangun untuk dibaca setiap \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter}
AC. Angka ``\qty{230}{V}'' yang termasyhur milik stopkontaknya adalah
nilai efektifnya; puncaknya sebenarnya mencapai kira-kira
\qty{325}{V}, lima puluh kali tiap sekon, ke tiap arahnya.
\end{proposition}

\begin{example}[Alat mana yang peduli]\label{ex:g9:alternating-current:whocares}
Efek panasnya bekerja baik-baik saja ke arah mana pun --- sehelai
filamen atau kumparan cerek menghangat tanpa peduli arah
arak-arakannya, dan itulah sebabnya lampu dan pemanas meminum AC tanpa
mengeluh. Tetapi elektronika --- telepon, komputer jinjing, apa pun
yang berkeping --- memerlukan DC yang mantap: bata pengisi dayanya dan
catu daya tersembunyinya mengubah ayunan stopkontaknya menjadi
ketetapan serupa \omterm{def:g3:first-electric-circuit:battery}{baterai} (rasakan hangat batanya: pengubahannya
membayar pajaknya). Dan dunia \omterm{def:g3:first-electric-circuit:battery}{baterainya} juga tidak dapat mengisi
dengan sebuah ayunan: pengisian adalah kimia yang dijalankan terbalik,
dan kimia bersikeras pada arah yang tetap.
\end{example}

\begin{remark}[Mengapa dunianya memilih ayunannya]\label{rem:g9:alternating-current:why}
Kalau \omterm{def:g3:first-electric-circuit:battery}{baterai} begitu menyenangkan mantapnya, mengapa peradaban dipasangi
kabel untuk AC? Dua alasan, keduanya menanti dua bab di depan: \omterm{def:g9:alternating-current:dcac}{arus
bolak-balik} adalah apa yang \emph{secara alami dibuat} mesin yang
berputar --- dan, yang menentukan, hanya AC yang memberi makan trafo,
kotak berdengung yang bersahaja yang membiarkan listrik menempuh
perjalanan sepanjang sebuah negeri pada \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang raksasa lalu
masuk ke rumahmu pada \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} yang jinak. Jaringannya bolak-balik
karena perjalanannya menuntutnya; kedua bab berikutnya mengikuti
arusnya dari gulungan yang berputar sampai ke dinding dapurnya.
\end{remark}

\begin{remark}[Ayunannya dan tubuhnya]\label{rem:g9:alternating-current:safety}
Hitungan lama keselamatannya memperoleh satu perincian yang suram: lima
puluh ayunan tiap sekon milik listrik rumahnya terletak dekat irama
yang paling mudah mengacaukan pewaktuan listrik jantung itu sendiri ---
sengatan bolak-balik, \omterm{def:g8:intensity-ammeter:intensity}{ampere} demi \omterm{def:g8:intensity-ammeter:intensity}{ampere}, lebih berbahaya daripada
sengatan yang mantap. Setiap aturan masa kecilnya tetap berlaku,
berlipat dua: \qty{230}{V} efektif milik stopkontaknya ---
\qty{325}{V} pada puncaknya --- bukan sekadar \omterm{def:g3:first-electric-circuit:battery}{baterai} yang lebih besar.
Ia binatang yang lain, yang kurang memaafkan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:1}
Bedakan DC dari AC dalam satu kalimat masing-masing, beserta sumber
rumah tangga masing-masingnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:2}
Batasilah \omterm{def:g9:alternating-current:period}{perioda} dan \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} lalu tuliskan hubungan kebalikannya.
Hitunglah $T$ untuk \qty{50}{Hz} dan $f$ untuk
$T = \qty{0.01}{s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:3}
Berapa kali tiap sekon \omterm{def:g7:circuits-series-parallel:current}{arus listrik} rumahnya membalik arah pada
\qty{50}{Hz}? (Hati-hati: pembalikan tiap daurnya.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:4}
Sebuah \omterm{met:g9:alternating-current:scope}{osiloskop} menunjukkan satu daur membentang $4$ petak pada
\qty{5}{ms} tiap petak. Carilah $T$ dan $f$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:5}
Stopkontaknya berkata \qty{230}{V}; puncaknya mencapai \qty{325}{V}.
Sebutkan kedua nilainya dan arti yang lebih kecil.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:6}
Mengapa cerek menerima AC dengan senang hati sementara pengisi daya
telepon harus mengubahnya? Satu efek, satu kimia.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:7}
Sketsakan (atau gambarkan) kedua jejaknya: sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} \qty{9}{V},
dan sebuah \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} bolak-balik berperioda \qty{20}{ms} dan berpuncak
\qty{9}{V}. Satu ciri yang digambar alatnya yang mana yang membedakan
keduanya dalam sekilas pandang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:8}
Berikan kedua alasan peradaban memasangi dirinya kabel untuk AC,
sebagaimana dicuplik babnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:9}
Sebuah dinamo sepeda yang lambat menyalakan lampunya dengan kedipan
yang terlihat; mengayuh lebih keras memantapkan nyalanya. Jelaskan
kedua rezimnya dengan \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} --- lalu taksirlah \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensi} yang di
bawahnya kedipannya mengganggu, mengingat bahwa bioskop menipu mata
pada kira-kira $24$ gambar tiap sekon.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:10}
Sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} AC yang melintang pada sebuah stopkontak membaca
\qty{230}{V}. Sebuah \omterm{met:g9:alternating-current:scope}{osiloskop} (khayalan) yang tak kenal takut yang
melintang pada stopkontak yang sama menunjukkan puncak pada nilai
berapa --- dan berapa puncak tiap sekon, dengan menghitung kedua
tandanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:11}
Jaringan sebuah wilayah berjalan pada \qty{60}{Hz} dengan efektif
\qty{120}{V}. Hitunglah \omterm{def:g9:alternating-current:period}{periodanya} dan hampiran \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} puncaknya ---
lalu sebutkan alat perjalananmu yang mana yang akan keberatan (lihatlah
\cref{ex:g9:alternating-current:whocares}; pengisi daya zaman sekarang
membaca cetakan halus pada labelnya).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:alternating-current:12}
Film lama tentang layar dan lampu neon kadang-kadang menunjukkan palang
gelap yang bergulir --- bingkai kameranya yang menangkap terang lampunya
di tengah ayunannya. Sebuah lampu listrik rumah pada \qty{50}{Hz}
menjadi terang pada setiap puncak dari tanda yang \emph{mana pun}:
berapa denyut terang tiap sekon? Sebuah kamera pada $25$ bingkai tiap
sekon mencuplik denyut itu: nalarkanlah mengapa layangan di antara
kedua iramanya melukiskan palang yang bergerak lambat, dan mengapa
memfilmkan di wilayah \qty{60}{Hz} mengubah polanya.
\end{exercise}

\section{Soal: Meja Osiloskop}

\begin{problem}[{Soal akhir pekan --- malam kelulusan di meja
osiloskopnya; empat sumber misterius, satu layar; buku sang
teknisi}]\label{pb:g9:alternating-current:1}
Empat sumber tanpa label harus dikenali dari jejaknya. Tombol
\omterm{met:g9:alternating-current:scope}{osiloskopnya}: \qty{5}{ms} tiap petak mendatar, \qty{2}{V} tiap petak
tegak (kecuali kalau disebutkan lain). Kamu menjaga bukunya.

\medskip\noindent\textbf{Bagian I --- Keempat jejaknya.}
\begin{enumerate}
  \item Sumber A: sebuah garis datar, $2.25$ petak di atas garis
        tengahnya. Kenalilah jenis dan \omterm{def:g8:voltage-voltmeter:voltage}{tegangannya} --- dan jati diri
        rumah tangganya yang mungkin.
  \item Sumber B: sebuah gelombang yang mulus, satu daur tiap $4$
        petak, dengan puncak setinggi $3$ petak. Apa jenisnya,
        \omterm{def:g9:alternating-current:period}{periodanya}, \omterm{def:g8:sound-pitch-loudness:frequency}{frekuensinya}, dan \omterm{def:g8:voltage-voltmeter:voltage}{tegangan} puncaknya?
  \item Sumber C: sebuah garis datar \emph{di bawah} garis tengahnya
        sejauh $2.25$ petak. Ia itu apa, dan kekeliruan kecil apa yang
        menjelaskannya?
  \item Sumber D: sebuah gelombang yang mulus, satu daur tiap $2$
        petak, dengan puncak $1.5$ petak. Berikan angka lengkapnya ---
        lalu mengapa lampu D akan lebih sedikit berkedip daripada
        lampu B seandainya keduanya memberi makan satu lampu.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pertanyaan sang
teknisi.}
\begin{enumerate}[resume]
  \item Sebuah \omterm{def:g8:voltage-voltmeter:voltmeter}{voltmeter} AC yang melintang pada sumber B membaca
        kira-kira \qty{4.2}{V}, padahal puncaknya mencapai \qty{6}{V}.
        Rukunkan kedua bilangannya.
  \item Yang mana di antara keempat sumbernya yang dapat mengisi
        sebuah telepon secara langsung, dan apa yang harus berdiri di
        antara yang lain dan \omterm{def:g3:first-electric-circuit:battery}{baterainya}?
  \item Pelat peringatan mejanya: ``AC menggigit lebih buruk, \omterm{def:g8:voltage-voltmeter:voltage}{volt}
        demi \omterm{def:g8:voltage-voltmeter:voltage}{volt}.'' Nyatakan ulang alasannya.
  \item Seorang murid magang bertanya mengapa \omterm{met:g9:alternating-current:scope}{osiloskopnya} menggambar
        gelombang B begitu mulus membulat alih-alih zigzag. Sebutkan
        bentuknya dan asalnya (mesin macam apa, pada bab berikutnya,
        yang menggambarnya secara alami?).
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Jejak milik
jaringannya sendiri.}
Sebuah alat peraga yang terpencil dengan aman menunjukkan listrik
rumahnya, pada \qty{100}{V} dan \qty{5}{ms} tiap petak.
\begin{enumerate}[resume]
  \item Ramalkan jejaknya: berapa petak tiap daur, dan berapa tinggi
        puncaknya dalam petak.
  \item Dari jejaknya, pulihkan $f$ dan \omterm{prop:g9:alternating-current:effective}{tegangan efektifnya} --- lalu
        periksalah keduanya terhadap label stopkontaknya yang
        termasyhur.
  \item Alat peraganya dialihkan ke sebuah bank \omterm{def:g3:first-electric-circuit:battery}{baterai} yang daya
        pemanasan efektifnya sama. Gambarkan jejak yang baru dalam
        satu baris.
  \item Tutuplah bukunya dengan ringkasan tiga baris sang teknisi: apa
        arti garis yang datar, apa arti gelombangnya, dan satu bilangan
        yang mana yang membiarkan keduanya dibandingkan dengan adil.
\end{enumerate}
\end{problem}
