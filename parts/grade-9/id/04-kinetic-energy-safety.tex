\chapter{Energi Kinetik dan Keselamatan Jalan}\label{ch:g9:kinetic-energy-safety}

Sejak mainan berpegas masa kecilmu, \omterm{def:g5:everyday-energy:energy}{energi} telah menjadi kisah tentang
simpanan dan bentuk --- hidup, berguna, dan tanpa bilangan.
Penantiannya berakhir sekarang. \omterm{def:g5:everyday-energy:energy}{Energi} geraknya mendapat rumusnya,
\omterm{def:g5:everyday-energy:energy}{energinya} mendapat \omterm{def:g6:measurement-in-science:measuring}{satuannya}, dan hitungannya ternyata mengatur perkara
hidup dan mati: mengapa sebuah mobil pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} dua kali lipat empat
kali lebih sukar dihentikan, dan berapa \omterm{def:g2:measuring-length:units}{meter} harga satu sekon kelalaian.

\section{Energi gerak}

\begin{definition}[Joule]\label{def:g9:kinetic-energy-safety:joule}
\omterm{def:g5:everyday-energy:energy}{Energi} --- setiap bentuknya --- diukur dalam
\emph{joule}\index{joule} (\unit{J}), yang menghormati pelaku
percobaan yang membuktikan \omterm{def:g5:heat-and-insulation:heat}{kalor} itu sendiri sebuah bentuk \omterm{def:g5:everyday-energy:energy}{energi}.
Jangkarnya: mengangkat buku ini dari lantai ke rak membelanjakan
beberapa joule; jantung yang berdetak, kira-kira satu joule tiap
detak; sebatang cokelat menyimpan sejuta joule \omterm{def:g5:everyday-energy:energy}{energi} makanan.
\end{definition}

\begin{proposition}[Energi kinetik]\label{prop:g9:kinetic-energy-safety:formula}
Sebuah benda bermassa $m$ (dalam \unit{kg}) yang bergerak pada
\omterm{def:g7:motion-average-speed:formula}{kelajuan} $v$ (dalam \unit{m/s}) membawa, berkat geraknya,
\emph{energi kinetik}\index{energi kinetik}
\[
E_k = \frac{1}{2} \times m \times v^2 ,
\]
dalam \omterm{def:g9:kinetic-energy-safety:joule}{joule}. Dua tombol, dengan pangkat yang tidak sama: menggandakan
\omterm{def:g3:measuring-mass:mass}{massanya} menggandakan $E_k$ --- tetapi menggandakan \emph{\omterm{def:g7:motion-average-speed:formula}{kelajuannya}}
\emph{mengempatkalikannya}, karena \omterm{def:g7:motion-average-speed:formula}{kelajuannya} masuk dalam bentuk
kuadrat. \omterm{def:g7:motion-average-speed:formula}{Kelajuan} adalah tombol yang berbahaya.
\end{proposition}

\begin{example}[Perhitungan yang pertama]\label{ex:g9:kinetic-energy-safety:first}
Sebuah bola sepak \qty{0.45}{kg} pada \qty{20}{m/s}: $E_k = 0.5 \times
0.45 \times 20^2 = \qty{90}{J}$. Seorang pelari cepat \qty{60}{kg} pada
\qty{10}{m/s}: $0.5 \times 60 \times 100 = \qty{3000}{J}$. Sebuah mobil
\qty{1000}{kg} pada \qty{50}{km/h} --- ubahlah lebih dahulu:
$50 \div 3.6 \approx \qty{13.9}{m/s}$ --- $E_k = 0.5 \times
1000 \times 13.9^2 \approx \qty{9.7e4}{J}$: hampir seratus ribu \omterm{def:g9:kinetic-energy-safety:joule}{joule}.
Mobil yang sama pada \qty{100}{km/h}: empat kali lipatnya ---
\qty{3.9e5}{J}. Peringatan rumusnya, dalam bilangan.
\end{example}

\begin{example}[Ke mana pengereman mengirimnya]\label{ex:g9:kinetic-energy-safety:brakes}
Untuk berhenti, sebuah mobil harus menyerahkan seluruh $E_k$-nya
kepada sesuatu --- \omterm{def:g5:everyday-energy:energy}{energi} diteruskan, tidak pernah dihapus, sebagaimana
dituntut rantai masa kecilmu. Cengkeraman remnya mengubahnya menjadi
\omterm{def:g5:heat-and-insulation:heat}{kalor}: cakramnya berpijar jingga sesudah menuruni gunung, dan bau rem
yang panas \emph{adalah} \omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetik} yang sedang dipensiunkan. Pada
sebuah tabrakan penyerahannya ganas dan seketika --- ke dalam baja yang
teremuk. Setiap bilangan keselamatan jalan di bawah ini adalah
pembukuan ini: makin besar $E_k$-nya, makin panjang, atau makin buruk,
pensiunnya.
\end{example}

\section{Jarak berhenti}

\begin{proposition}[Berhenti = berpikir + mengerem]\label{prop:g9:kinetic-energy-safety:stopping}
Seorang pengemudi yang melihat bahaya berhenti hanya sesudah dua
bentangan jalan:
\begin{enumerate}
  \item \emph{jarak tanggap}\index{jarak tanggap}: jalan yang ditempuh
        selagi otaknya memperhatikan dan kakinya bergerak --- kira-kira
        satu sekon penuh pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} yang tak berubah, sehingga
        bentangan ini bertumbuh \emph{sebanding} dengan $v$;
  \item \emph{jarak pengereman}\index{jarak pengereman}: jalan yang
        diperlukan remnya untuk memensiunkan \omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetiknya} ---
        dan karena $E_k$ membawa $v^2$, bentangan ini bertumbuh
        menurut \emph{kuadrat} \omterm{def:g7:motion-average-speed:formula}{kelajuannya}: \omterm{def:g7:motion-average-speed:formula}{kelajuan} dua kali lipat,
        jalan pengereman empat kali lipat.
\end{enumerate}
Jumlahnya, yaitu \emph{jarak berhenti}, adalah bilangan yang bertemu
anak yang mengejar bolanya.
\end{proposition}

\begin{example}[Tabel yang seharusnya dimiliki setiap pengemudi]\label{ex:g9:kinetic-energy-safety:table}
Jalan kering, pengemudi yang waspada (waktu tanggap satu sekon; rem yang baik):
\begin{center}
\begin{tabular}{l|c|c|c}
\omterm{def:g7:motion-average-speed:formula}{kelajuan} & tanggap & pengereman & berhenti \\
\hline
\qty{50}{km/h} & \qty{14}{m} & \qty{13}{m} & \qty{27}{m} \\
\qty{90}{km/h} & \qty{25}{m} & \qty{40}{m} & \qty{65}{m} \\
\qty{130}{km/h} & \qty{36}{m} & \qty{85}{m} & \qty{121}{m}
\end{tabular}
\end{center}
Bacalah dua kali. Pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} kota, sepanjang satu setengah bus; pada
\omterm{def:g7:motion-average-speed:formula}{kelajuan} jalan bebas \omterm{def:g8:ohms-law:resistance}{hambatan}, sepanjang satu \omterm{def:g6:describing-motion:trajectory}{lintasan} lari penuh
ditambah seperlima. Dan perbandingannya menegaskan kedua hukumnya:
tanggapnya bertumbuh seperti $v$, pengeremannya seperti $v^2$ --- pada
\omterm{def:g7:motion-average-speed:formula}{kelajuan} tinggi, pengeremanlah yang melahap totalnya.
\end{example}

\begin{method}[Menaksir sebuah jarak berhenti]\label{met:g9:kinetic-energy-safety:estimate}
\begin{enumerate}
  \item ubahlah \omterm{def:g7:motion-average-speed:formula}{kelajuannya} menjadi \unit{m/s} (bagilah \unit{km/h}
        dengan $3.6$);
  \item \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak tanggapnya}: perjalanan satu sekon --- bilangan
        \unit{m/s}-nya sendiri, dalam \omterm{def:g2:measuring-length:units}{meter};
  \item \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak pengeremannya}: skalakan sebuah jangkar yang diketahui
        menurut kuadratnya --- dari \qty{13}{m} pada \qty{50}{km/h},
        kalikan dengan $(v/50)^2$;
  \item jumlahkan, lalu bandingkan dengan apa yang sesungguhnya
        ditawarkan jalan di depannya.
\end{enumerate}
Jalan yang basah menggandakan bagian pengeremannya; pengemudi yang
lelah atau terganggu perhatiannya meregangkan sekon tanggapnya menuju
dua --- jalankan ulang jumlahnya dengan masukan yang jujur.
\end{method}

\begin{example}[Harga sebuah lirikan]\label{ex:g9:kinetic-energy-safety:glance}
Sebuah lirikan dua sekon ke sebuah telepon pada \qty{90}{km/h}:
mobilnya menempuh $2 \times 25 = \qty{50}{m}$ --- separuh lapangan
sepak bola --- yang dikemudikan dengan buta, sebelum sekon tanggapnya
bahkan mulai. Pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} kota lirikan yang sama membelanjakan
\qty{28}{m} dengan buta: melewati persimpangannya, melewati gerbang
sekolahnya. Bagian mobil yang paling berbahaya tidak terukur oleh
rumus mana pun di sini: perhatian pengemudinya.
\end{example}

\begin{remark}[Sabuk, kantong, dan helm]\label{rem:g9:kinetic-energy-safety:belts}
Pada sebuah tabrakan mobilnya berhenti dalam semeter keremukan ---
tetapi penumpang yang tidak bersabuk berjalan terus pada \omterm{def:g7:motion-average-speed:formula}{kelajuan}
penuh (kalimat yang tenang pada bab yang lalu, diterapkan dengan
suram) sampai dihentikan oleh apa pun yang datang: papan kemudinya,
kaca depannya, jalannya. Sabuk pengaman dan kantong \omterm{def:g2:air-around-us:air}{udara} menghentikan
tubuhnya atas jarak dan waktu yang lebih panjang, sehingga
memensiunkan \omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetiknya} dengan lembut alih-alih sekaligus; helm
pengendara sepeda melakukan hal yang sama bagi isi tengkorak yang tidak
tergantikan, dengan busanya yang dapat remuk membeli beberapa
\omterm{def:g2:measuring-length:units}{sentimeter} penghentian yang lembut. Tidak satu pun di antaranya
mengurangi $E_k$-mu walau sejoule --- mereka memberadabkan pensiunnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:1}
Berikan \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g5:everyday-energy:energy}{energinya} dan dua jangkarnya, lalu tulislah rumus \omterm{prop:g9:kinetic-energy-safety:formula}{energi
kinetiknya} beserta \omterm{def:g6:measurement-in-science:measuring}{satuannya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:2}
Hitunglah $E_k$: seekor kelinci \qty{2}{kg} pada \qty{10}{m/s}; seorang
pemain rugbi \qty{80}{kg} pada \qty{5}{m/s}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:3}
Yang mana yang lebih menaikkan $E_k$ sebuah mobil: menambahkan muatan
seberat separuh \omterm{def:g3:measuring-mass:mass}{massanya}, atau menaikkan \omterm{def:g7:motion-average-speed:formula}{kelajuannya} sebesar
separuhnya? Tunjukkan kedua faktornya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:4}
Sebutkan kedua bentangan jarak berhentinya dan hukum pertumbuhannya
--- satu sebanding, satu kuadrat.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:5}
Dari tabel pengemudinya: berapa bagian jarak berhentinya yang berupa
pengereman pada \qty{50}{km/h} --- dan pada \qty{130}{km/h}? Apa yang
berubah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:6}
Ke mana \omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetik} sebuah mobil yang mengerem pergi? Sebutkan
baunya dan pijarnya --- serta hukum masa kecil yang melarangnya
sekadar lenyap.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:7}
Hitunglah \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak tanggapnya} pada \qty{36}{km/h}, \qty{72}{km/h}, dan
\qty{108}{km/h} (satu sekon yang waspada). Pola sederhana apa yang
diikuti jawabannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:kinetic-energy-safety:8}
Mengapa sabuk pengaman tidak mengurangi \omterm{def:g5:everyday-energy:energy}{energi} kinetikmu --- dan apa
yang justru diberadabkannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:kinetic-energy-safety:9}
\omterm{prop:g9:kinetic-energy-safety:stopping}{Jarak pengereman} sebuah skuter \qty{8}{m} pada \qty{30}{km/h}.
Taksirlah pada \qty{60}{km/h} dan pada \qty{90}{km/h} --- lalu
nyatakan hukum yang kamu pakai untuk menskalakannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:kinetic-energy-safety:10}
Dewan kota menurunkan batas \omterm{def:g7:motion-average-speed:formula}{kelajuan} zona sekolah dari \qty{50}{}
menjadi \qty{30}{km/h}. Bandingkan kedua jarak berhentinya (jangkar:
\qty{13}{m} pengereman pada $50$; tanggap satu sekon) --- lalu kedua
\omterm{prop:g9:kinetic-energy-safety:formula}{energi kinetiknya}. Perbandingan yang mana yang menurutmu lebih

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:kinetic-energy-safety:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:kinetic-energy-safety:12}
Sebuah truk \qty{20000}{kg} melaju pada \qty{90}{km/h}. Hitunglah

mobil \qty{1000}{kg} menyamai \omterm{def:g5:everyday-energy:energy}{energi} itu (samakan kedua rumusnya ---
aljabarnya menjadi milikmu tahun ini); lalu simpulkan mengapa landaian
penyelamat truk yang lepas kendali ada di jalan pegunungan sementara
tidak ada landaian semacam itu yang melayani mobil.
\end{exercise}

\section{Soal: Kampanye Keselamatan}

\begin{problem}[{Soal akhir pekan --- kelas merancang kampanye
keselamatan jalan kotanya; poster yang diperiksa dengan rumus;
pertanyaan sulit sang wali kota}]\label{pb:g9:kinetic-energy-safety:1}
Kotanya memesan sebuah kampanye, dengan satu syarat: setiap bilangan
pada setiap posternya harus selamat melewati pemeriksaan fisika.
Kelasnya menghitung. (Jangkarnya: tanggap satu sekon; pengereman
\qty{13}{m} pada \qty{50}{km/h}, yang diskalakan menurut kuadratnya;
\omterm{def:g3:measuring-mass:mass}{massa} mobil \qty{1000}{kg}.)

\medskip\noindent\textbf{Bagian I --- Poster satu: ``50 bukan
60''.}
\begin{enumerate}
  \item Hitunglah kedua \omterm{def:g7:motion-average-speed:formula}{kelajuannya} dalam \unit{m/s} (satu angka di
        belakang koma).
  \item Berapa \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak tanggapnya} pada kedua \omterm{def:g7:motion-average-speed:formula}{kelajuannya}?
  \item Berapa \omterm{prop:g9:kinetic-energy-safety:stopping}{jarak pengeremannya} pada keduanya (skalakan
        jangkarnya)?
  \item Judul posternya: ``Sepuluh \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g7:motion-average-speed:formula}{kelajuan} lebih banyak, ---
        berapa \omterm{def:g2:measuring-length:units}{meter} jarak berhenti lebih banyak?'' Isilah
        bilangannya.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Poster dua: ``Dinding
\omterm{def:g7:motion-average-speed:formula}{kelajuan}''.}
Rancangannya menunjukkan \omterm{def:g5:everyday-energy:energy}{energi} tabrakan sebagai tinggi jatuh: sebuah
tabrakan pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} $v$ ``sama dengan'' jatuh dari tinggi yang
padanya \omterm{def:g5:everyday-energy:energy}{energi} yang sama akan diperoleh.
\begin{enumerate}[resume]
  \item Hitunglah $E_k$ mobilnya pada \qty{50}{km/h} (dari
        Bagian I) dan pada \qty{90}{km/h}.
  \item Para senimannya memerlukan tingginya: jatuh dari \qty{1}{m}
        memperoleh kira-kira \qty{9.8}{kJ} bagi mobil ini ($m g h$ ---
        rumus lain bab berikutnya, yang dipinjam lebih awal). Kedua
        \omterm{def:g5:everyday-energy:energy}{energi} tabrakannya berpadanan dengan tinggi jatuh berapa?
        (Bagilah; bulatkan ke \omterm{def:g2:measuring-length:units}{meter} terdekat.)
  \item Bangunan sehari-hari yang mana yang berpadanan dengan tinggi
        itu? Rancanglah baris posternya.
  \item Wali kotanya membantah: ``Tidak ada yang mengemudi menabrak
        dinding; mobil meremuk, sabuknya menangkap.'' Belalah fisika
        posternya sambil mengakui pokoknya --- apa yang diubah zona
        remuk dan sabuknya, dan bilangan apa yang tidak diubahnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Poster tiga: ``Satu
sekon''.}
\begin{enumerate}[resume]
  \item Pada \qty{90}{km/h}, berapa \omterm{def:g2:measuring-length:units}{meter} harga satu sekon yang
        terganggu perhatiannya sebelum pengereman apa pun? Dan lirikan
        telepon dua sekon?
  \item Ragam pengemudi yang lelah: waktu tanggapnya diregangkan
        menjadi dua sekon. Hitunglah ulang jarak berhenti penuhnya
        pada \qty{90}{km/h} lalu bandingkan dengan \qty{65}{m} milik
        pengemudi yang waspada.
  \item Seorang anggota dewan yang meragukan: ``Pastilah sesekon
        tidak banyak berarti di samping bilangan pengereman yang
        besar itu.'' Pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} yang mana ia paling keliru ---
        rendah atau tinggi? (Bandingkan hukum pertumbuhan kedua
        komponennya.)
  \item Sahkan kampanyenya: tiga baris di bagian bawah posternya,
        satu tiap poster, masing-masing membawa rumusnya dalam kata
        yang sederhana.
\end{enumerate}
\end{problem}
