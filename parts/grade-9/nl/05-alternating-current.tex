\chapter{Wisselstroom}\label{ch:g9:alternating-current}

Elke schakeling die je ooit hebt gebouwd, liep op een
\omterm{def:g3:first-electric-circuit:battery}{batterij}: een stroom die gestaag van $+$ naar $-$ marcheerde,
plichtsgetrouw als de wijzer van een klok. Maar de stopcontacten in je
muren leveren iets wilders --- een stroom die honderd keer per \omterm{def:g2:measuring-time:second}{seconde}
van richting keert, met een \omterm{def:g8:voltage-voltmeter:voltage}{spanning} die onophoudelijk tussen plus
en min heen en weer zwaait. Maak kennis met de
\omterm{def:g9:alternating-current:dcac}{wisselstroom}: de vorm waarin vrijwel alle elektriciteit ter
wereld wordt geboren, vervoerd en geleverd.

\section{Twee soorten stroom}

\begin{definition}[Gelijkstroom en wisselstroom]\label{def:g9:alternating-current:dcac}
Een \emph{gelijkstroom}\index{gelijkstroom} (DC) loopt altijd in
dezelfde richting, aangedreven door een gestage
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} --- het geschenk van de \omterm{def:g3:first-electric-circuit:battery}{batterij},
constant tot de voorraad opraakt. Een
\emph{wisselstroom}\index{wisselstroom} (AC) keert periodiek van
richting, aangedreven door een \omterm{def:g8:voltage-voltmeter:voltage}{spanning} die soepel van positief
naar negatief en terug zwaait, keer op keer --- het regime van het
stopcontact, en dat van de wisselstroomgenerator (het volgende
hoofdstuk vertelt van wie).
\end{definition}

\begin{definition}[Periode en frequentie]\label{def:g9:alternating-current:period}
De \emph{periode}\index{periode} $T$ van een wisselspanning is de
duur van één volledige kringloop --- één volle zwaai omhoog,
omlaag en terug, in \omterm{def:g2:measuring-time:second}{seconden}. Haar \emph{\omterm{def:g8:sound-pitch-loudness:frequency}{frequentie}} $f$ is
het aantal kringlopen per \omterm{def:g2:measuring-time:second}{seconde}, in \omterm{def:g8:sound-pitch-loudness:frequency}{hertz} --- de
\omterm{def:g6:measurement-in-science:measuring}{eenheid} van het \omterm{def:g3:sound-around-us:sound}{geluid}, opnieuw in dienst --- en de
twee zijn elkaars omgekeerde:
\[
f = \frac{1}{T}, \qquad T = \frac{1}{f}.
\]
\end{definition}

\begin{proposition}[De kerngegevens van het lichtnet]\label{prop:g9:alternating-current:mains}
In het grootste deel van de wereld wisselt het lichtnet met
$f = \qty{50}{Hz}$ --- vijftig volledige kringlopen per \omterm{def:g2:measuring-time:second}{seconde}, een
\omterm{def:g9:alternating-current:period}{periode} van $T = 1/50 = \qty{0.02}{s}$: twintig milliseconden
per zwaai (sommige gebieden draaien op zestig). De stroom in het snoer
van je leeslamp keert honderd keer per \omterm{def:g2:measuring-time:second}{seconde} van richting --- twee
keer per kringloop --- en heeft nog nooit gestaag van de ene
klem naar de andere gemarcheerd.
\end{proposition}

\begin{example}[Rekenen met frequentie]\label{ex:g9:alternating-current:arithmetic}
Het omgekeerdenpaar rekent beide kanten op. Een generator die op
\qty{60}{Hz} draait: $T = 1/60 \approx \qty{0.017}{s}$ --- zeventien
milliseconden. Een \omterm{met:g9:alternating-current:scope}{oscilloscoop} toont een kringloop die
\qty{0.04}{s} duurt: $f = 1/0.04 = \qty{25}{Hz}$. Een fietsdynamo maakt
bij langzaam trappen \qty{10}{Hz}: een \omterm{def:g9:alternating-current:period}{periode} van een tiende
\omterm{def:g2:measuring-time:second}{seconde} --- en zijn \omterm{def:g3:first-electric-circuit:bulb}{lampje} flikkert zichtbaar, tien pulsen per \omterm{def:g2:measuring-time:second}{seconde},
tot de \omterm{def:g7:motion-average-speed:formula}{snelheid} ze gladstrijkt tot voorbij wat het oog opmerkt.
\end{example}

\section{De zwaai lezen}

\begin{method}[Een oscilloscoopbeeld lezen]\label{met:g9:alternating-current:scope}
De \emph{oscilloscoop}\index{oscilloscoop} --- of elke analyse-app ---
tekent de \omterm{def:g8:voltage-voltmeter:voltage}{spanning} tegen de tijd op een geruit scherm; wat elk
hokje waard is in \omterm{def:g8:voltage-voltmeter:voltage}{volt} en in milliseconden wordt door twee
knoppen ingesteld. Om een spoor te oogsten:
\begin{enumerate}
  \item tel de hokjes van één volledige kringloop en vermenigvuldig
        met de tijd per hokje: dat is $T$; daarna $f = 1/T$;
  \item tel de hokjes van de middellijn tot een top en vermenigvuldig
        met de \omterm{def:g8:voltage-voltmeter:voltage}{volt} per hokje: de \emph{piekspanning}
        $U_{\text{max}}$;
  \item controleer de vorm: het lichtnet en de generatoren tekenen de
        soepele, eindeloos herhaalde golf uit de figuur --- de
        \emph{sinusoïde}, de naam die de wiskunde aan de volmaakte
        zwaai geeft.
\end{enumerate}
\end{method}

\begin{proposition}[Effectieve spanning]\label{prop:g9:alternating-current:effective}
Wat is ``de'' \omterm{def:g8:voltage-voltmeter:voltage}{spanning} van een zwaai die honderd keer per
\omterm{def:g2:measuring-time:second}{seconde} elke waarde van $+U_{\text{max}}$ tot $-U_{\text{max}}$
bezoekt? Het eerlijke enkele getal is de \emph{effectieve
spanning}\index{effectieve spanning} $U$: de gestage
gelijkspanning die een \omterm{def:g8:ohms-law:resistance}{weerstand} even goed zou
verwarmen. Voor de sinusoïde ligt ze ruim onder de top ---
$U_{\text{max}}$ is ongeveer $1.4$ keer $U$ --- en het is wat elke
wisselstroomvoltmeter gebouwd is om af te lezen. De beroemde
``\qty{230}{V}'' van het stopcontact is de effectieve waarde; de toppen
halen in werkelijkheid ongeveer \qty{325}{V}, vijftig keer per \omterm{def:g2:measuring-time:second}{seconde},
beide kanten op.
\end{proposition}

\begin{example}[Welke apparaten er iets om geven]\label{ex:g9:alternating-current:whocares}
De \omterm{prop:g8:electric-current-effects:three}{warmtewerking} doet het beide kanten op prima --- een
gloeidraad of een spiraal van een waterkoker wordt warm, ongeacht de
richting van de mars, en daarom drinken lampen en kachels \omterm{def:g9:alternating-current:dcac}{wisselstroom}
zonder klagen. Maar elektronica --- telefoons, laptops, alles met chips
--- heeft gestage \omterm{def:g9:alternating-current:dcac}{gelijkstroom} nodig: hun laderblokken en verborgen
voedingen zetten de zwaai van het stopcontact om in een
batterijachtige standvastigheid (voel de \omterm{def:g5:heat-and-insulation:heat}{warmte} van het blok: de
omzetting betaalt haar belasting). En de wereld van de
\omterm{def:g3:first-electric-circuit:battery}{batterij} laat zich evenmin op een zwaai opladen: laden is
scheikunde achterstevoren, en scheikunde staat op een vaste richting.
\end{example}

\begin{remark}[Waarom de wereld voor de zwaai koos]\label{rem:g9:alternating-current:why}
Als \omterm{def:g3:first-electric-circuit:battery}{batterijen} zo aangenaam gestaag zijn, waarom is de beschaving dan
voor \omterm{def:g9:alternating-current:dcac}{wisselstroom} bedraad? Twee redenen, die allebei twee hoofdstukken
verderop wachten: \omterm{def:g9:alternating-current:dcac}{wisselstroom} is wat draaiende machines
\emph{van nature maken} --- en, doorslaggevend, alleen \omterm{def:g9:alternating-current:dcac}{wisselstroom}
voedt de transformator, het nederige zoemende kastje waardoor
elektriciteit een land kan doorkruisen op kolossale
\omterm{def:g8:voltage-voltmeter:voltage}{spanning} en je huis kan binnenkomen op een tamme. Het net is
wisselend omdat de reis het eist; de volgende twee hoofdstukken volgen
de stroom van draaiende spoel tot keukenmuur.
\end{remark}

\begin{remark}[De zwaai en het lichaam]\label{rem:g9:alternating-current:safety}
De oude rekenkunde van de veiligheid krijgt er één grimmig detail bij:
de vijftig zwaaien per \omterm{def:g2:measuring-time:second}{seconde} van het lichtnet liggen dicht bij het
ritme dat de eigen elektrische timing van het hart het gemakkelijkst
ontregelt --- wisselstroomschokken zijn, \omterm{def:g8:intensity-ammeter:intensity}{ampère} voor
\omterm{def:g8:intensity-ammeter:intensity}{ampère}, gevaarlijker dan gestage. Elke regel uit je
kindertijd staat, dubbel onderstreept: de effectieve \qty{230}{V} van
het stopcontact --- \qty{325}{V} in de top --- is niet zomaar een
grotere \omterm{def:g3:first-electric-circuit:battery}{batterij}. Het is een ander, minder vergevingsgezind
beest.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:9}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:11}

\begin{exercise}[$\star\star\star$]\label{exo:g9:alternating-current:12}

balken --- de beeldjes van de camera vangen de helderheid van de lamp
midden in haar zwaai. Een netlamp op \qty{50}{Hz} \omterm{def:g1:light-and-shadows:source}{licht} op bij elke top
van \emph{beide} tekens: hoeveel helderheidspulsen per \omterm{def:g2:measuring-time:second}{seconde}? Een
camera van $25$ beeldjes per \omterm{def:g2:measuring-time:second}{seconde} bemonstert dat pulseren: beredeneer
waarom de zweving tussen de twee ritmes langzaam bewegende balken
schildert, en waarom filmen in een gebied met \qty{60}{Hz} het patroon
verandert.
\end{exercise}

\section{Probleem: de oscilloscoopbank}

\begin{problem}[{Weekendprobleem --- examenavond aan de
oscilloscoopbank; vier mysterieuze voedingen, één scherm; het grootboek
van de technicus}]\label{pb:g9:alternating-current:1}
Vier voedingen zonder etiket moeten aan hun spoor worden herkend. De
knoppen van de \omterm{met:g9:alternating-current:scope}{oscilloscoop}: \qty{5}{ms} per hokje horizontaal,
\qty{2}{V} per hokje verticaal (behalve waar anders vermeld). Jij houdt
het grootboek bij.

\medskip\noindent\textbf{Deel I --- De vier sporen.}
\begin{enumerate}
  \item Voeding A: een vlakke lijn, $2.25$ hokjes boven het midden.
        Benoem haar soort en \omterm{def:g8:voltage-voltmeter:voltage}{spanning} --- en haar
        waarschijnlijke huishoudelijke identiteit.
  \item Voeding B: een soepele golf, één kringloop per $4$ hokjes,
        toppen $3$ hokjes hoog. Soort, \omterm{def:g9:alternating-current:period}{periode},
        \omterm{def:g8:sound-pitch-loudness:frequency}{frequentie}, piekspanning?
  \item Voeding C: een vlakke lijn $2.25$ hokjes \emph{onder} het
        midden. Wat is het, en welke kleine fout verklaart het?
  \item Voeding D: een soepele golf, één kringloop per $2$ hokjes,
        toppen van $1.5$ hokje. Volledige kerngegevens --- en waarom
        het \omterm{def:g3:first-electric-circuit:bulb}{lampje} van D minder zou flikkeren dan dat van B als ze
        allebei er een voedden.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De vragen van de technicus.}
\begin{enumerate}[resume]
  \item Een wisselstroomvoltmeter over voeding B leest ongeveer
        \qty{4.2}{V}, hoewel haar toppen \qty{6}{V} halen. Breng de
        twee getallen met elkaar in overeenstemming.
  \item Welke van de vier voedingen zou een telefoon rechtstreeks
        kunnen opladen, en wat moet er tussen de andere en zijn
        \omterm{def:g3:first-electric-circuit:battery}{batterij} staan?
  \item Het waarschuwingsplaatje van de bank: ``\omterm{def:g9:alternating-current:dcac}{wisselstroom} bijt
        erger, \omterm{def:g8:voltage-voltmeter:voltage}{volt} voor \omterm{def:g8:voltage-voltmeter:voltage}{volt}.'' Geef de reden
        opnieuw.
  \item Een leerling vraagt waarom de \omterm{met:g9:alternating-current:scope}{oscilloscoop} de golf van B zo
        soepel afgerond tekent in plaats van als een zigzag. Noem de
        vorm en haar oorsprong (welk soort machine, in het volgende
        hoofdstuk, tekent haar van nature?).
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het spoor van het net zelf.}
Een veilig gescheiden demonstratieopstelling toont het lichtnet, bij
\qty{100}{V} en \qty{5}{ms} per hokje.
\begin{enumerate}[resume]
  \item Voorspel het spoor: hokjes per kringloop en tophoogte in
        hokjes.
  \item Haal uit het spoor $f$ en de \omterm{prop:g9:alternating-current:effective}{effectieve spanning} --- en
        toets allebei aan het beroemde etiket van het stopcontact.
  \item De opstelling wordt omgeschakeld naar een batterijbank met
        hetzelfde effectieve verwarmingsvermogen. Beschrijf het nieuwe
        spoor in één regel.
  \item Sluit het grootboek af met de samenvatting van de technicus in
        drie regels: wat de vlakke lijn betekent, wat de golf betekent,
        en welk enkel getal het mogelijk maakt de twee eerlijk te
        vergelijken.
\end{enumerate}
\end{problem}
