\chapter{Elektrisch vermogen en energie}\label{ch:g9:electric-power-energy}

De energierekening brengt geen \omterm{def:g8:voltage-voltmeter:voltage}{volt} in rekening, en ook geen
\omterm{def:g8:intensity-ammeter:intensity}{ampère} --- ze brengt iets in rekening dat de apparaten van je
gezin de hele maand stilletjes met elkaar hebben vermenigvuldigd. Dit
hoofdstuk huwt de twee elektrische grootheden tot een derde, geeft snel
en traag uitgeven hun juiste namen, en eindigt bij de elektriciteitsmeter naast de
voordeur: natuurkunde die je aan echt geld kunt toetsen.

\section{Vermogen: het tempo van levering}

\begin{definition}[Vermogen en de watt]\label{def:g9:electric-power-energy:power}
Het \emph{vermogen}\index{vermogen} $P$ van een apparaat is het tempo
waarmee het \omterm{def:g5:everyday-energy:energy}{energie} omzet --- \omterm{def:g9:kinetic-energy-safety:joule}{joule} per \omterm{def:g2:measuring-time:second}{seconde}. De
\omterm{def:g6:measurement-in-science:measuring}{eenheid} is de \emph{watt}\index{watt} (\unit{W}): één
\omterm{def:g9:kinetic-energy-safety:joule}{joule} per \omterm{def:g2:measuring-time:second}{seconde}, ter ere van de ingenieur van het
stoomtijdperk; de \emph{kilowatt} (\qty{1}{kW} $=$ \qty{1000}{W})
bedient de zwaargewichten van de keuken. Vermogen is geen hoeveelheid
\omterm{def:g5:everyday-energy:energy}{energie} maar een \emph{tempo} van uitgeven --- de stroom uit een
kraan, niet de inhoud van een emmer.
\end{definition}

\begin{proposition}[Elektrisch vermogen]\label{prop:g9:electric-power-energy:UI}
Een apparaat dat \omterm{def:g8:voltage-voltmeter:voltage}{spanning} $U$ krijgt en stroom $I$ trekt,
ontvangt het elektrische \omterm{def:g9:electric-power-energy:power}{vermogen}
\[
P = U \times I ,
\]
\omterm{def:g9:electric-power-energy:power}{watt} uit \omterm{def:g8:voltage-voltmeter:voltage}{volt} maal \omterm{def:g8:intensity-ammeter:intensity}{ampère}. De formule
rekent in alle drie de richtingen, en haar tweede recept bestuurt de
meterkast: $I = P/U$ vertelt welke stroom het opgegeven
\omterm{def:g9:electric-power-energy:power}{vermogen} van een apparaat uit het lichtnet zal trekken.
\end{proposition}

\begin{example}[De huishoudladder]\label{ex:g9:electric-power-energy:ladder}
Lees de plaatjes op de machines van het gezin: een controlelampje
\qty{1}{W}; een moderne lamp \qty{5}{W} (haar gloeiende voorouder gaf
er zestig uit voor hetzelfde \omterm{def:g1:light-and-shadows:source}{licht} --- het verschil was
\omterm{def:g5:heat-and-insulation:heat}{warmte}); een laptop \qty{50}{W}; een televisie \qty{100}{W};
een wasmachine die haar water verwarmt \qty{2000}{W}; de waterkoker
\qty{2300}{W}; een doorstroomboiler tot \qty{9000}{W}. Het patroon van
de ladder: alles wat als taak \emph{verwarmen} heeft, klimt naar de
kilowatts --- \omterm{def:g5:heat-and-insulation:heat}{warmte} is de duurste vorm van \omterm{def:g5:everyday-energy:energy}{energie} om te
leveren.
\end{example}

\begin{example}[De rekenkunde van de meterkast]\label{ex:g9:electric-power-energy:fuses}
De waterkoker op het \qty{230}{V}-net: $I = 2300 \div 230 =
\qty{10}{A}$ --- het tienampèreapparaat uit de oude
hoofdstukken, nu afgeleid. De doorstroomboiler van \qty{9000}{W}: bijna
\qty{40}{A}, die zijn eigen stevige groep en automaat opeist. En de
overbelaste stekkerdoos uit het brandonderzoek: waterkoker, kachel en
friteuse die hun $P/U$-stromen bij elkaar optellen tot voorbij de
grenswaarde van de groep --- drie onschuldigen, één
rekenkundige misdaad, precies zoals ten laste gelegd.
\end{example}

\section{Energie: de geleverde hoeveelheid}

\begin{proposition}[Energie uit vermogen en tijd]\label{prop:g9:electric-power-energy:Pt}
Een apparaat met \omterm{def:g9:electric-power-energy:power}{vermogen} $P$ dat gedurende een tijd $t$ draait,
zet de \omterm{def:g5:everyday-energy:energy}{energie}
\[
E = P \times t ,
\]
om --- \omterm{def:g9:kinetic-energy-safety:joule}{joule} uit \omterm{def:g9:electric-power-energy:power}{watt} maal \omterm{def:g2:measuring-time:second}{seconden}. Snelle kraan
of lang straaltje, de emmer vult zich volgens het product: de drie
minuten van een waterkoker van \qty{2300}{W} ($2300 \times 180 \approx
\qty{4.1e5}{J}$) geven meer uit dan een hele dag van een lamp van
\qty{5}{W} ($5 \times 86400 = \qty{4.3e5}{J}$) --- vrijwel precies
gelijkspel eigenlijk: één keer koken evenaart één dag goed
\omterm{def:g1:light-and-shadows:source}{licht}.
\end{proposition}

\begin{definition}[Het kilowattuur]\label{def:g9:electric-power-energy:kwh}
Huishoudelijke energieën leveren onhandelbare joulegetallen op, dus
spreekt de elektriciteitsmeter in \emph{kilowattuur}\index{kilowattuur} (\unit{kWh}):
de \omterm{def:g5:everyday-energy:energy}{energie} van één kilowatt die één uur wordt volgehouden,
\[
\qty{1}{kWh} = 1000 \times 3600 = \qty{3.6e6}{J}
\]
--- drie komma zes miljoen \omterm{def:g9:kinetic-energy-safety:joule}{joule} per \omterm{def:g6:measurement-in-science:measuring}{eenheid} op de
rekening. Met $P$ in kilowatt en $t$ in \omterm{ex:g2:measuring-time:clock}{uren} levert $E = P
\times t$ rechtstreeks kilowattuur op: de formule waar de elektriciteitsmeter
van leeft.
\end{definition}

\begin{method}[Een apparaat doorlichten]\label{met:g9:electric-power-energy:audit}
Voor elke machine en elke maand:
\begin{enumerate}
  \item lees haar \omterm{def:g9:electric-power-energy:power}{vermogen} $P$ op het plaatje (of bereken
        $U \times I$);
  \item schat haar draaitijd $t$ eerlijk --- per dag, dan per maand;
  \item $E = P \times t$ in \omterm{def:g9:electric-power-energy:kwh}{kilowattuur} ($P$ in \unit{kW}, $t$
        in \omterm{ex:g2:measuring-time:clock}{uren});
  \item vermenigvuldig met het tarief (neem ongeveer $0.25$
        munteenheden per \unit{kWh}) voor de kosten van de
        maand.
\end{enumerate}
De waterkoker, zes minuten per dag: $2.3 \times 0.1 \times 30 =
\qty{6.9}{kWh} $ per maand. De televisie, drie uur per dag: $0.1
\times 3 \times 30 = \qty{9}{kWh}$. Kleine \omterm{def:g9:electric-power-energy:power}{vermogens} met lange
\omterm{ex:g2:measuring-time:clock}{uren} evenaren grote \omterm{def:g9:electric-power-energy:power}{vermogens} met korte --- de
terugkerende les van de doorlichting.
\end{method}

\section{De meter en de rekening}

\begin{example}[De rekening lezen]\label{ex:g9:electric-power-energy:bill}
De elektriciteitsmeter naast de voordeur telt elk \omterm{def:g9:electric-power-energy:kwh}{kilowattuur} dat het huis
binnenkomt; de rekening is het maandverschil maal het tarief. De $250$
maandeenheden van een doorsneegezin vallen bij doorlichting
uiteen: eerst waterverwarming en radiatoren (de kilowattclub), dan de
koude apparaten --- bescheiden \omterm{def:g9:electric-power-energy:power}{watts}, maar altijd draaiend ---
dan koken, wassen, \omterm{def:g1:light-and-shadows:source}{licht} en elektronica. De \omterm{def:g2:pushes-and-pulls:force}{kracht} van de
doorlichting: ze vindt de echte hefbomen. Tien oude
lampen van \qty{60}{W} vervangen door van \qty{5}{W} bespaart
duizend keer meer dan elke lader in huis uit het stopcontact halen.
\end{example}

\begin{example}[De sluipvampiers]\label{ex:g9:electric-power-energy:standby}
Kleine lettertjes, lange \omterm{ex:g2:measuring-time:clock}{uren}: het stand-bylampje van
een televisie nipt \qty{0.5}{W} --- maar wel alle $720$
uur van de maand: $0.0005 \times 720 = \qty{0.36}{kWh}$. Eén
vampier is onschuldig; een huis met twintig nippers van elk een
\omterm{def:g9:electric-power-energy:power}{watt} betaalt maandelijks $0.02 \times 720 \approx
\qty{14}{kWh}$ voor niets. \omterm{def:g9:electric-power-energy:power}{Vermogen} maal tijd vergeeft geen enkele
factor: de tweede bijt wanneer de eerste zich verstopt.
\end{example}

\begin{remark}[Vermogen in de wijdere wereld]\label{rem:g9:electric-power-energy:wider}
De \omterm{def:g9:electric-power-energy:power}{watt}, hier elektrisch geboren, meet elk
energietempo. Je rustende lichaam draait stationair op zo'n
\qty{100}{W} --- een gloeiende voorouderlamp van \omterm{def:g5:heat-and-insulation:heat}{warmte} (denk aan
de bedomptheid van het volle klaslokaal: dertig leerlingen, drie
kilowatt biologie). Een fietser houdt \qty{200}{W} vol; de \omterm{ex:g6:circuits-and-safety:motor}{motor} van
een auto ontketent tientallen kilowatts; de elektriciteitscentrale van
de stad honderden miljoenen \omterm{def:g9:electric-power-energy:power}{watt}; het zonlicht dat op de stad
valt, comfortabel meer --- het plafond waaronder alle andere tempo's
werken. Het volgende hoofdstuk volgt de \omterm{def:g9:electric-power-energy:power}{watts} van het net naar
hun draaiende bron.
\end{remark}

\section{Probleem: de energiedoorlichting van het huis}

\begin{problem}[{Weekendprobleem --- de energiedoorlichting van het
gezin; plaatjes, meters en de rekening van de maand; de echte hefbomen
vinden}]\label{pb:g9:electric-power-energy:1}
Gewapend met de twee formules en het tarief ($0.25$ per \unit{kWh})
\omterm{def:g1:light-and-shadows:source}{licht} het gezin een maand door (30 dagen). De inventaris: waterkoker
\qty{2300}{W}, $6$ minuten per dag; televisie \qty{100}{W}, $4$
uur per dag; koelkast --- speciaal geval hieronder; tien lampen
van elk \qty{5}{W}, $5$ uur per dag; wasmachine \qty{2}{kW}, om
de dag één warm programma van $1$ uur; boiler \qty{2.5}{kW}, $2$
uur per dag; stand-bynippers samen \qty{10}{W}, altijd aan.

\medskip\noindent\textbf{Deel I --- Machine voor machine.}
\begin{enumerate}
  \item De maand van de waterkoker, in \unit{kWh}.
  \item De maanden van de televisie en van de tien lampen.
  \item De maanden van de wasmachine ($15$ programma's) en van de
        boiler.
  \item De maand van de stand-bynippers --- de tweede factor aan het
        werk.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De koelkast en de som.}
De compressor van de koelkast draait op \qty{120}{W}, maar slechts een
derde van de tijd, dag en nacht in en uit schakelend.
\begin{enumerate}[resume]
  \item Haar effectieve gemiddelde \omterm{def:g9:electric-power-energy:power}{vermogen}, en haar maand in
        \unit{kWh}.
  \item Tel de zeven regels van de doorlichting op. Welke drie regels
        overheersen?
  \item De rekening van de maand tegen het tarief --- en het aandeel
        van de boiler daarin in geld.
  \item De elektriciteitsmeter stond aan het begin van de maand op $2150$
        eenheden. Voorspel zijn stand aan het eind van de maand.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De hefbomen.}
\begin{enumerate}[resume]
  \item Voorstel één: korter douchen --- de boiler zakt naar
        \qty{1.5}{h} per dag. Maandelijkse besparing, in
        \omterm{def:g5:everyday-energy:energy}{energie} en in geld?
  \item Voorstel twee: een publiekscampagne roept op telefoonladers
        uit het stopcontact te halen (elk \qty{0.2}{W} als ze niets
        doen). \omterm{def:g1:light-and-shadows:source}{Licht} de nietsdoende maand van één lader door en
        vergelijk met voorstel één; wat leert het contrast over
        doorlichtingen tegenover leuzen?
  \item Voorstel drie: oma stelt de oude gewoonte voor om de
        waterkoker alleen te vullen met wat nodig is --- half zoveel
        water, half zoveel \omterm{def:g5:everyday-energy:energy}{energie}. De nieuwe regel voor de
        waterkoker, en de besparing van de maand?
  \item Schrijf de conclusies van de doorlichting: drie zinnen ---
        waar de kilowatturen werkelijk wonen, welke factor van de
        formule het stand-bygeval onderwijst, en welke enkele
        verandering in huis het best rendeert.
\end{enumerate}
\end{problem}
