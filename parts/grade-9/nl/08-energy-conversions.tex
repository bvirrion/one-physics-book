\section{De vormen, bijeengebracht}

\begin{definition}[De vormen van energie]\label{def:g9:energy-conversions:forms}
De vormen van \omterm{def:g5:everyday-energy:energy}{energie}, formeel voorgesteld:
\emph{\omterm{prop:g9:kinetic-energy-safety:formula}{kinetische energie}} --- het aandeel van de beweging,
$\frac12 m v^2$; \emph{potentiële energie}\index{potentiële energie}
--- opgeslagen door plaats of ordening (wat is opgetild, uitgerekt of
\omterm{prop:g7:states-of-matter:squeeze}{samengedrukt}); \emph{thermische
\omterm{def:g5:everyday-energy:energy}{energie}} --- het wanordelijke moleculaire gewiebel dat de
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur} meet; \emph{chemische
\omterm{def:g5:everyday-energy:energy}{energie}} --- opgeslagen in de ordening van de materie: voedsel,
brandstoffen, \omterm{def:g3:first-electric-circuit:battery}{batterijen}; \emph{elektrische \omterm{def:g5:everyday-energy:energy}{energie}} --- de
leverbare waar van de marcherende stroom; \emph{stralingsenergie} ---
gedragen door het \omterm{def:g1:light-and-shadows:source}{licht} zelf, de vervoersvorm van de zonneschijn;
en \emph{kernenergie} --- opgesloten in de kern van het atoom, de
diepste voorraad die mensen hebben geopend.
\end{definition}

\begin{proposition}[Zwaarte-energie]\label{prop:g9:energy-conversions:ep}
Een \omterm{def:g3:measuring-mass:mass}{massa} $m$ (in \unit{kg}) over een hoogte $h$ (in \unit{m})
tegen de $g$ van de zwaartekracht in optillen slaat er de
\omterm{def:g9:energy-conversions:forms}{potentiële energie}
\[
E_p = m \times g \times h
\]
in \omterm{def:g9:kinetic-energy-safety:joule}{joule} in op --- op de weg omlaag volledig terug te winnen. De
opgeheven bijl, de achteruitgetrokken schommel, het bergmeer: allemaal
houden ze $m g h$ in rekening. (Dit is de geleende formule van de
verkeersveiligheidsposters --- nu officieel de jouwe.)
\end{proposition}

\begin{example}[De grote ruil: $E_p$ tegen $E_k$]\label{ex:g9:energy-conversions:trade}
Laat een bal van hoogte $h$ vallen: terwijl hij valt, verhuist zijn
rekening, \omterm{def:g9:kinetic-energy-safety:joule}{joule} voor \omterm{def:g9:kinetic-energy-safety:joule}{joule}, van potentieel naar
kinetisch. Bij de grond is de hele $m g h$ $\frac12 m v^2$ geworden ---
stel ze gelijk en de \omterm{def:g3:measuring-mass:mass}{massa} valt weg:
\[
v = \sqrt{2 g h},
\]
de vierkantswortel van dit jaar die zijn brood verdient. Vanaf
\qty{20}{m}: $v = \sqrt{2 \times 9.8 \times 20} \approx \qty{20}{m/s}$
--- zeventig kilometer per uur, wat de \omterm{def:g3:measuring-mass:mass}{massa} ook zij. De
slinger speelt de ruil eeuwig beide kanten op (op een fluistering aan de
\omterm{def:g2:air-around-us:air}{lucht} na); de skater in de halfpipe evenzo: hoogte naar
\omterm{def:g7:motion-average-speed:formula}{snelheid} naar hoogte, terwijl de twee formules één som heen en
weer doorgeven.
\end{example}

\section{De grote wet}

\begin{proposition}[Behoud van energie]\label{prop:g9:energy-conversions:conservation}
In elk proces dat ooit is onderzocht --- botsingen en scheikunde,
machines en stormen, sterren en cellen --- verandert
\omterm{def:g5:everyday-energy:energy}{energie} van vorm en van eigenaar, maar blijft het \emph{totaal}
exact behouden: niets geschapen, niets vernietigd, elke
\omterm{def:g9:kinetic-energy-safety:joule}{joule} verantwoord. Dit is de diepste boekhoudwet van de
natuurkunde: ze stuurde de eeuwigdurende machines uit je
kinderhoofdstukken met pensioen, ze controleert elke keten in dit boek,
en in drie eeuwen heeft geen enkel experiment haar op een fout betrapt.
(Waarom de natuur deze wet zo trouw naleeft, is een diepe vraag met een
prachtig antwoord --- een van de juwelen van de universitaire jaren.)
\end{proposition}

\begin{method}[Elk proces doorlichten]\label{met:g9:energy-conversions:audit}
De universele boekhouding van de natuurkundige:
\begin{enumerate}
  \item zet een hek om het systeem --- wat hoort er binnen de
        doorlichting;
  \item som de energierekeningen vooraf op: elke vorm, elke
        voorraad;
  \item som ze achteraf op --- altijd inclusief het thermische
        kleingeld: de \omterm{def:g5:heat-and-insulation:heat}{warmte} van de wrijving, de fluistering van het
        \omterm{def:g3:sound-around-us:sound}{geluid};
  \item breng de boeken in \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}: de totalen moeten kloppen. Een
        tekort betekent dat een vorm niet is meegeteld --- zoek het lek
        opnieuw; het is meestal warm.
\end{enumerate}
\end{method}

\begin{example}[Doorlichtingen, uitgewerkt]\label{ex:g9:energy-conversions:audits}
De remmende auto: de kinetische rekening leeggehaald, de thermische
rekening bij de schijven volledig gecrediteerd --- pensionering, geen
vernietiging. De stuiterende bal die uitsterft: elke stuit roomt
\omterm{prop:g9:kinetic-energy-safety:formula}{kinetische energie} af tot \omterm{def:g5:heat-and-insulation:heat}{warmte} van bal en vloer; het antwoord uit je
kindertijd, nu controleerbaar. Jouw ontbijt terwijl je bergop fietst:
chemische voorraad omlaag, potentiële rekening omhoog ($m g h$ van jou
en de fiets), thermisch aandeel uitgestraald --- spieren draaien warm.
Elk mysterie uit de vroege energiehoofdstukken sluit onder
dezelfde drie kolommen.
\end{example}

\section{Rendement}

\begin{definition}[Rendement]\label{def:g9:energy-conversions:efficiency}
Het \emph{rendement}\index{rendement} van een omzetter is het deel van
de toegevoerde \omterm{def:g5:everyday-energy:energy}{energie} dat in de gewenste vorm wordt geleverd:
\[
\text{rendement} =
\frac{\text{nuttige energie eruit}}{\text{totale energie erin}},
\]
een getal tussen $0$ en $1$ (of het bijbehorende percentage). De rest
wordt niet vernietigd --- het behoud verbiedt dat --- maar ontsnapt in
ongewenste vormen, vrijwel altijd \omterm{def:g5:heat-and-insulation:heat}{warmte}.
\end{definition}

\begin{example}[Rapportcijfers]\label{ex:g9:energy-conversions:reportcards}
De gloeiende voorouderlamp: vijf procent \omterm{def:g1:light-and-shadows:source}{licht},
vijfennegentig \omterm{def:g5:heat-and-insulation:heat}{warmte} --- een kachel met een hobby. Haar moderne
opvolger: dertig procent en meer, de rest nog steeds \omterm{def:g5:heat-and-insulation:heat}{warmte}. Een
automotor: ongeveer een derde nuttige beweging, twee derde de radiateur
en de uitlaat uit. Een elektrische \omterm{ex:g6:circuits-and-safety:motor}{motor}: boven de negentig. De
grote generatoren: bijna negenennegentig --- de fijnste omzetters
van de beschaving. Geen enkele eerlijke omzetter haalt honderd: een
zeker lek naar \omterm{def:g5:heat-and-insulation:heat}{warmte} is de universele commissie van de natuur.
\end{example}

\begin{remark}[Wat ``energie besparen'' werkelijk betekent]\label{rem:g9:energy-conversions:saving}
Als het behoud elke \omterm{def:g9:kinetic-energy-safety:joule}{joule} bewaakt, waarom moet iemand dan
``\omterm{def:g5:everyday-energy:energy}{energie} besparen''? Omdat de wet de \emph{hoeveelheid}
bewaart, niet de \emph{bruikbaarheid}. Geconcentreerde voorraden ---
brandstof, lading, hoogte --- kunnen worden uitgegeven aan \omterm{def:g5:heat-and-insulation:heat}{warmte} die
dun over de wereld wordt uitgesmeerd, en uitgesmeerde \omterm{def:g5:heat-and-insulation:heat}{warmte} kan, hoewel
elke \omterm{def:g9:kinetic-energy-safety:joule}{joule} ervan overleeft, geen waterkokers of treinen meer
aandrijven: de rivier liep bergafwaarts. Het
energieprobleem van de beschaving is geen tekort aan
\omterm{def:g9:kinetic-energy-safety:joule}{joule} --- de Zon overgiet ons met duizenden keren onze behoefte
--- maar het zuinig omgaan met \emph{bruikbare}. Dat idee, precies
gemaakt, is een van de beroemdste verhalen uit de natuurkunde; het
wacht in de universitaire jaren onder zijn befaamde naam.
\end{remark}

\section{Probleem: de opdracht van het pretpark}

\begin{problem}[{Weekendprobleem --- het pretpark huurt een
natuurkundig adviseur in; achtbanen, torens en het grootboek dat niet
kan liegen}]\label{pb:g9:energy-conversions:1}
De nieuwe attracties van het park moeten voor de openingsdag jouw
energiedoorlichting doorstaan. Neem $g = \qty{9.8}{N/kg}$;
verwaarloos de wrijving tot ze weigert verwaarloosd te worden.

\medskip\noindent\textbf{Deel I --- De Grote Duik.}
Een achtbaantrein van \qty{2000}{kg} wordt naar een top van \qty{45}{m}
gelierd en vanuit rust losgelaten.
\begin{enumerate}
  \item De $E_p$ van de trein op de top?
  \item Zijn ideale \omterm{def:g7:motion-average-speed:formula}{snelheid} onder aan de duik, volgens de formule
        waarin de \omterm{def:g3:measuring-mass:mass}{massa} wegvalt --- in \unit{m/s} en \unit{km/h}?
  \item De ontwerpers voegen vlak na de duik een tweede heuvel van
        \qty{50}{m} toe. Spreek je veto uit met het grootboek --- en
        noem de hoogste tweede heuvel die de ideale trein zou kunnen
        nemen.
  \item De gemeten \omterm{def:g7:motion-average-speed:formula}{snelheid} onderaan is \qty{28}{m/s}, niet de
        ideale. Bereken beide kinetische energieën en boek het verschil
        op de juiste rekening.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De lier en de rekening.}
\begin{enumerate}[resume]
  \item De lier tilt de trein in \qty{60}{s} naar zijn top. Bij
        volmaakt \omterm{def:g9:energy-conversions:efficiency}{rendement}, haar \omterm{def:g9:electric-power-energy:power}{vermogen}?
  \item De echte lier trekt \qty{20}{kW}. Haar
        \omterm{def:g9:energy-conversions:efficiency}{rendement}?
  \item Elke rit vertrekt om de vier minuten, tien uur per dag.
        De dagelijkse \omterm{def:g5:everyday-energy:energy}{energie} van de lier in
        \omterm{def:g9:electric-power-energy:kwh}{kilowattuur} (de echte lier, tel de liftbewegingen), en de
        kosten daarvan bij $0.25$ per \omterm{def:g6:measurement-in-science:measuring}{eenheid}?
  \item Een ontwerper stelt voor de aankomstenergie van de trein
        terug te winnen met een generatorrem die het net voedt --- ``de
        rit betaalt zichzelf!'' \omterm{def:g1:light-and-shadows:source}{Licht} de droom door: welk deel zou
        realistisch kunnen terugkomen, en welke wet verbiedt de
        volledige cirkel?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De valtoren en het rapport.}
De toren van \qty{60}{m} laat een gondel van \qty{500}{kg} vallen tot
in de magnetische remmen op \qty{10}{m}.
\begin{enumerate}[resume]
  \item De \omterm{def:g7:motion-average-speed:formula}{snelheid} bij het binnengaan van de remzone (een ideale
        val van \qty{50}{m})?
  \item De \omterm{def:g5:everyday-energy:energy}{energie} die de remmen met pensioen moeten sturen, en
        waarheen de inductie haar stuurt. (De remmen zijn neefjes van
        het generatorhoofdstuk --- wervels van geïnduceerde stroom
        die metaal verwarmen.)
  \item Passagiers schreeuwen dat ze tijdens de val ``gewichtloos''
        waren. Doe met het idee van de samen vallende metgezellen uit
        het gravitatiehoofdstuk in één zin uitspraak over die
        bewering.
  \item Onderteken het rapport van de opdracht: drie zinnen --- de ruil
        waarvan de achtbaan leeft, de wet die elke attractie
        gehoorzaamde, en de rekening die ze allemaal stilletjes
        belastte.
\end{enumerate}
\end{problem}
