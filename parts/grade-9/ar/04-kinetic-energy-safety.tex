\chapter{الطاقة الحركية وسلامة الطرق}\label{ch:g9:kinetic-energy-safety}

منذ ألعاب الزنبرك في طفولتك، \omterm{def:g5:everyday-energy:energy}{والطاقة} حكاية مخازن وأشكال ---
حيّة نافعة بلا أعداد. وينتهي الانتظار الآن. فتنال \omterm{def:g5:everyday-energy:energy}{طاقة} الحركة
صيغتها، وتنال \omterm{def:g5:everyday-energy:energy}{الطاقة} وحدتها، ويتبيّن أن الحساب يحكم أمر حياة
أو موت: لماذا تكون سيارة على ضعف \omterm{def:g7:motion-average-speed:formula}{السرعة} أصعب في الإيقاف أربع
مرات، وكم تكلّف ثانية واحدة من الغفلة بالأمتار.

\section{طاقة الحركة}

\begin{definition}[الجول]\label{def:g9:kinetic-energy-safety:joule}
تُقاس \omterm{def:g5:everyday-energy:energy}{الطاقة} --- بكل أشكالها --- \emph{بالجول}\index{جول}
(\unit{J})، تكريمًا للمجرّب الذي برهن أن \omterm{def:g5:heat-and-insulation:heat}{الحرارة} نفسها شكل من
أشكال \omterm{def:g5:everyday-energy:energy}{الطاقة}. ومراسٍ: رفع هذا الكتاب من الأرض إلى الرفّ ينفق
بضعة جولات؛ والقلب النابض نحو جول لكل نبضة؛ ولوح شوكولاتة يخزن
مليون جول من \omterm{def:g5:everyday-energy:energy}{طاقة} الغذاء.
\end{definition}

\begin{proposition}[الطاقة الحركية]\label{prop:g9:kinetic-energy-safety:formula}
جسم كتلته $m$ (\omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{kg}) يتحرّك \omterm{def:g7:motion-average-speed:formula}{بسرعة} $v$ (\omterm{def:g6:measurement-in-science:measuring}{بوحدة}
\unit{m/s}) يحمل، بحكم حركته، \emph{الطاقة
الحركية}\index{طاقة حركية}
\[
E_k = \frac{1}{2} \times m \times v^2 ,
\]
\omterm{def:g9:kinetic-energy-safety:joule}{بالجول}. قرصان، وأُسّان غير متساويين: فمضاعفة \omterm{def:g3:measuring-mass:mass}{الكتلة} تضاعف
$E_k$ --- أما مضاعفة \emph{\omterm{def:g7:motion-average-speed:formula}{السرعة}} فتربّعها أربع مرات، لأن
\omterm{def:g7:motion-average-speed:formula}{السرعة} تدخل مربّعة. \omterm{def:g7:motion-average-speed:formula}{فالسرعة} هي القرص الخطر.
\end{proposition}

\begin{example}[أول الحسابات]\label{ex:g9:kinetic-energy-safety:first}
كرة قدم كتلتها \qty{0.45}{kg} \omterm{def:g7:motion-average-speed:formula}{بسرعة} \qty{20}{m/s}: $E_k = 0.5
\times 0.45 \times 20^2 = \qty{90}{J}$. وعدّاء كتلته
\qty{60}{kg} \omterm{def:g7:motion-average-speed:formula}{بسرعة} \qty{10}{m/s}: $0.5 \times 60 \times 100 =
\qty{3000}{J}$. وسيارة كتلتها \qty{1000}{kg} \omterm{def:g7:motion-average-speed:formula}{بسرعة}
\qty{50}{km/h} --- وحوّل أولًا: $50 \div 3.6 \approx
\qty{13.9}{m/s}$ --- فتكون $E_k = 0.5 \times 1000 \times 13.9^2
\approx \qty{9.7e4}{J}$: أي قرابة مئة ألف \omterm{def:g9:kinetic-energy-safety:joule}{جول}. والسيارة نفسها
عند \qty{100}{km/h}: أربعة أضعاف --- \qty{3.9e5}{J}. تحذير
الصيغة، بالأعداد.
\end{example}

\begin{example}[إلى أين يرسلها الكبح]\label{ex:g9:kinetic-energy-safety:brakes}
لكي تتوقّف السيارة، عليها أن تسلّم $E_k$ كلها إلى شيء ما ---
\omterm{def:g5:everyday-energy:energy}{فالطاقة} تُمرَّر ولا تُمحى أبدًا، كما أصرّت سلاسل طفولتك. وقبضة
المكابح تحوّلها إلى \omterm{def:g5:heat-and-insulation:heat}{حرارة}: فالأقراص تتوهّج كهرمانية بعد
الهبوط من الجبال، ورائحة المكابح الساخنة \emph{هي} \omterm{prop:g9:kinetic-energy-safety:formula}{الطاقة
الحركية} تتقاعد. وفي الاصطدام يكون التسليم عنيفًا لحظيًّا ---
إلى فولاذ منسحق. وكل عدد من أعداد سلامة الطرق أدناه هو هذه
المحاسبة: فكلما كبرت $E_k$ طال تقاعدها أو قبح.
\end{example}

\section{مسافة التوقّف}

\begin{proposition}[التوقّف = تفكير + كبح]\label{prop:g9:kinetic-energy-safety:stopping}
السائق الذي يلمح خطرًا لا يتوقّف إلا بعد قطعتين من الطريق:
\begin{enumerate}
  \item \emph{مسافة ردّ الفعل}\index{مسافة ردّ الفعل}: الطريق
        المقطوع بينما يلحظ الدماغ وتتحرّك القدم --- نحو ثانية
        كاملة \omterm{def:g7:motion-average-speed:formula}{بالسرعة} نفسها، فتنمو هذه القطعة \emph{بالتناسب}
        مع $v$؛
  \item \emph{مسافة الكبح}\index{مسافة الكبح}: الطريق الذي
        تحتاج إليه المكابح لتقاعد \omterm{prop:g9:kinetic-energy-safety:formula}{الطاقة الحركية} --- وإذ تحمل
        $E_k$ المقدار $v^2$، تنمو هذه القطعة مع \emph{مربّع}
        \omterm{def:g7:motion-average-speed:formula}{السرعة}: فبضعف \omterm{def:g7:motion-average-speed:formula}{السرعة} يصير طريق الكبح أربعة أضعاف.
\end{enumerate}
ومجموعهما، \emph{مسافة التوقّف}، هو العدد الذي يلقى الطفل
الجاري خلف الكرة.
\end{proposition}

\begin{example}[الجدول الذي ينبغي أن يملكه كل سائق]\label{ex:g9:kinetic-energy-safety:table}
طريق جافّ، وسائق منتبه (ردّ فعل ثانية واحدة؛ ومكابح جيّدة):
\begin{center}
\begin{tabular}{l|c|c|c}
\omterm{def:g7:motion-average-speed:formula}{السرعة} & ردّ الفعل & الكبح & التوقّف \\
\hline
\qty{50}{km/h} & \qty{14}{m} & \qty{13}{m} & \qty{27}{m} \\
\qty{90}{km/h} & \qty{25}{m} & \qty{40}{m} & \qty{65}{m} \\
\qty{130}{km/h} & \qty{36}{m} & \qty{85}{m} & \qty{121}{m}
\end{tabular}
\end{center}
اقرأه مرتين. فعند \omterm{def:g7:motion-average-speed:formula}{سرعة} المدينة، طول حافلة ونصف؛ وعند \omterm{def:g7:motion-average-speed:formula}{سرعة}
الطريق السريع، مضمار جري كامل وخُمس. والنسب تؤكّد القانونين:
فردّ الفعل ينمو بمقدار يتناسب مع $v$، والكبح مع $v^2$ --- وعند \omterm{def:g7:motion-average-speed:formula}{السرعة}
العالية يلتهم الكبح المجموع.
\end{example}

\begin{method}[تقدير مسافة توقّف]\label{met:g9:kinetic-energy-safety:estimate}
\begin{enumerate}
  \item حوّل \omterm{def:g7:motion-average-speed:formula}{السرعة} إلى \unit{m/s} (اقسم \unit{km/h} على
        $3.6$)؛
  \item \omterm{prop:g9:kinetic-energy-safety:stopping}{مسافة ردّ الفعل}: سفر ثانية واحدة --- أي العدد \omterm{def:g6:measurement-in-science:measuring}{بوحدة}
        \unit{m/s} نفسه، بالأمتار؛
  \item \omterm{prop:g9:kinetic-energy-safety:stopping}{مسافة الكبح}: قِس مرساةً معلومة بالمربّع --- فمن
        \qty{13}{m} عند \qty{50}{km/h}، اضرب في $(v/50)^2$؛
  \item اجمع، وقارن بما يعرضه الطريق أمامك فعلًا.
\end{enumerate}
والطرق المبلّلة تضاعف حصة الكبح؛ والسائقون المتعبون أو
المشتّتون يمدّون ثانية ردّ الفعل نحو ثانيتين --- فأعِد الجمع
بالمدخلات الأمينة.
\end{method}

\begin{example}[ثمن نظرة]\label{ex:g9:kinetic-energy-safety:glance}
نظرة إلى هاتف مدّتها ثانيتان عند \qty{90}{km/h}: تقطع السيارة
$2 \times 25 = \qty{50}{m}$ --- أي نصف ملعب كرة قدم ---
مقودةً بلا بصر، قبل أن تبدأ أيّ ثانية من ردّ الفعل أصلًا. وعند
\omterm{def:g7:motion-average-speed:formula}{سرعة} المدينة تنفق النظرة نفسها \qty{28}{m} عمياء: أي بعد
التقاطع، وبعد بوّابة المدرسة. وأخطر مكوّنات السيارة لا تقيسه
أيّ صيغة هنا: انتباه السائق.
\end{example}

\begin{remark}[الأحزمة والوسائد والخوذ]\label{rem:g9:kinetic-energy-safety:belts}
في الاصطدام تتوقّف السيارة في \omterm{def:g2:measuring-length:units}{متر} من الانسحاق --- أما الراكب
غير المربوط فيمضي \omterm{def:g7:motion-average-speed:formula}{بالسرعة} كاملة (جملة الفصل الماضي الهادئة،
مطبَّقةً بقسوة) حتى يوقفه ما يصادفه: لوحة القيادة، أو الزجاج
الأمامي، أو الطريق. وأحزمة الأمان والوسائد الهوائية توقف الجسم
على مسافة أطول وزمن أطول، فتقاعد طاقته الحركية برفق بدل أن
تفعل ذلك دفعة واحدة؛ وخوذة الدرّاج تفعل الشيء نفسه لمحتوى
الجمجمة الذي لا يُعوَّض، فرغوتها القابلة للسحق تشتري سنتيمترات
من التوقّف الرفيق. ولا يخفض شيء منها $E_k$ عندك \omterm{def:g9:kinetic-energy-safety:joule}{بجول} واحد ---
بل تمدّن تقاعدها.
\end{remark}

\section{مسألة: حملة السلامة}

\begin{problem}[{مسألة نهاية الأسبوع --- الصفّ يصمّم حملة
سلامة الطرق في البلدة؛ ملصقات تُدقَّق بالصيغة؛ أسئلة العمدة
الصعبة}]\label{pb:g9:kinetic-energy-safety:1}
تكلّف البلدة الصفّ بحملة، بشرط واحد: أن يجتاز كل عدد على كل
ملصق تدقيقًا فيزيائيًّا. فيحسب الصفّ. (المراسي: ردّ فعل ثانية
واحدة؛ وكبح \qty{13}{m} عند \qty{50}{km/h}، \omterm{def:g6:measurement-in-science:measuring}{بالقياس} على
المربّع؛ \omterm{def:g3:measuring-mass:mass}{وكتلة} السيارة \qty{1000}{kg}.)

\medskip\noindent\textbf{القسم الأول --- الملصق الأول: ``50 لا
60''.}
\begin{enumerate}
  \item احسب السرعتين \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{m/s} (برقم عشري واحد).
  \item ما مسافتا ردّ الفعل عند السرعتين؟
  \item وما مسافتا الكبح عندهما (\omterm{def:g6:measurement-in-science:measuring}{بقياس} المرساة)؟
  \item وعنوان الملصق: ``عشر وحدات \omterm{def:g7:motion-average-speed:formula}{سرعة} زيادة --- فكم مترًا
        زيادة في التوقّف؟'' املأ العدد.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الملصق الثاني:
``جدار \omterm{def:g7:motion-average-speed:formula}{السرعة}''.}
يُظهر التصميم طاقات الاصطدام ارتفاعاتِ سقوط: فالاصطدام عند
\omterm{def:g7:motion-average-speed:formula}{سرعة} $v$ ``يساوي'' السقوط من الارتفاع الذي تُكتسب عنده \omterm{def:g5:everyday-energy:energy}{الطاقة}
نفسها.
\begin{enumerate}[resume]
  \item احسب $E_k$ للسيارة عند \qty{50}{km/h} (من القسم
        الأول) وعند \qty{90}{km/h}.
  \item ويحتاج الفنّانون إلى ارتفاعات: فالسقوط من \qty{1}{m}
        يكسب نحو \qty{9.8}{kJ} لهذه السيارة ($m g h$ ---
        صيغة الفصل التالي الأخرى، مستعارةً باكرًا). فإلى أيّ
        ارتفاعَي سقوط تقابل طاقتا الاصطدام؟ (اقسم؛ وقرّب إلى
        \omterm{def:g2:measuring-length:units}{المتر}.)
  \item وأيّ مبانٍ يومية تطابق تينك الارتفاعين؟ صُغ سطر
        الملصق.
  \item ويعترض العمدة: ``لا أحد يقود نحو الجدران؛ فالسيارات
        تنسحق، والأحزمة تمسك.'' دافع عن فيزياء الملصق مع
        التسليم بوجهة نظره --- ماذا تغيّر مناطق الانسحاق
        والأحزمة، وأيّ عدد لا تغيّره؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الملصق الثالث:
``ثانية واحدة''.}
\begin{enumerate}[resume]
  \item عند \qty{90}{km/h}، كم مترًا تكلّف ثانية واحدة

  \item ونسخة السائق المتعب: ردّ فعل ممدود إلى ثانيتين. أعِد
        حساب مسافة التوقّف الكاملة عند \qty{90}{km/h}
        وقارنها بالمسافة \qty{65}{m} للسائق المنتبه.
  \item ويقول عضو مجلس متشكّك: ``لا شكّ أن ثانية أمر يسير
        بجانب أعداد الكبح الكبيرة تلك.'' فعند أيّ السرعات
        يكون أشدّ خطأً --- المنخفضة أم العالية؟ (قارن
        قانونَي نموّ المكوّنين.)
  \item وقّع على الحملة: ثلاثة أسطر لأسفل الملصقات، سطر لكل
        ملصق، يحمل كل واحد صيغته بكلام صريح.
