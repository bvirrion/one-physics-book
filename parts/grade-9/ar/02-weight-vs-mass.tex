\chapter{الوزن في مقابل الكتلة: قياس القوى}\label{ch:g9:weight-vs-mass}

قبل خمس سنوات، وعلى \omterm{def:g2:moon-in-the-sky:moon}{قمر} من ورق، شطرت ``الثقيل'' شطرين: \omterm{def:g3:measuring-mass:mass}{الكتلة}
للمادة، \omterm{def:g4:weight-and-mass:weight}{والوزن} للجذب. وكان الشطر حكمة؛ وهو يصير الآن حسابًا.
فتنال القوى وحدتها --- المسمّاة باسم صاحب التفاحة الساقطة ---
وينال \omterm{def:g4:weight-and-mass:weight}{الوزن} صيغته، ويعود ميزان الشريط المطّاطي في طفولتك بثوب
احترافي.

\section{النيوتن}

\begin{definition}[النيوتن]\label{def:g9:weight-vs-mass:newton}
تُقاس القوى --- الدفعات والجذبات والأوزان وقبضة \omterm{def:g9:gravitation:gravitation}{الجاذبية} ---
\emph{بالنيوتن}\index{نيوتن} (\unit{N})، تكريمًا لصاحب قانون
الجذب. ومراسٍ مفيدة: النيوتن الواحد نحو \omterm{def:g4:weight-and-mass:weight}{وزن} تفاحة صغيرة
(والتاريخ يبتسم)؛ وحقيبتك المدرسية تزن نحو \qty{50}{N}؛ ومصافحة
متينة تعصر بنحو \qty{100}{N}؛ وعقد \omterm{def:g2:moon-in-the-sky:moon}{القمر} في الفصل الماضي بلغ
$10^{20}$.
\end{definition}

\begin{definition}[مقياس القوة]\label{def:g9:weight-vs-mass:dynamometer}
\emph{مقياس القوة}\index{مقياس القوة} (الدينامومتر) هو مقياس
القوى: نابض معاير في علبة، يستطيل بالتناسب مع \omterm{def:g2:pushes-and-pulls:force}{القوة} المطبَّقة،
ومؤشّره يقرأ النيوتنات مباشرة. وهو ميزان شريطك المطّاطي وقد
كبر --- وهو، \omterm{def:g8:voltage-voltmeter:voltmeter}{كمقياس التوتر} \omterm{def:g8:intensity-ammeter:ammeter}{ومقياس التيار} قبله، يقاعد رقعة أخرى
من ``يبدو قويًّا'' إلى أعداد أمينة.
\end{definition}

\begin{method}[قياس قوة]\label{met:g9:weight-vs-mass:measure}
\begin{enumerate}
  \item اختر مقياس \omterm{def:g2:pushes-and-pulls:force}{قوة} يلائم مداه المهمّة --- آلة \qty{2}{N}
        لمقالم الأقلام، وآلة \qty{100}{N} للحقائب المدرسية؛
  \item صفّره في وضع الاستعمال (معلَّقًا، إن كان الحمل
        سيُعلَّق --- \omterm{def:g4:weight-and-mass:weight}{فوزن} النابض نفسه يزيح الصفر)؛
  \item طبّق \omterm{def:g2:pushes-and-pulls:force}{القوة} بثبات، على امتداد محور الآلة، واقرأ على
        مستوى العين؛
  \item سجّل مع \omterm{def:g6:measurement-in-science:measuring}{الوحدة} --- وإذ للقوى اتجاهات، دوّن الاتجاه
        أيضًا حين يهمّ.
\end{enumerate}
وعلى المخطّطات، تصير \omterm{def:g2:pushes-and-pulls:force}{القوة} المقيسة سهمًا يُرسم بمقياس معلَن ---
``\omterm{def:g2:measuring-length:units}{سنتيمتر} واحد لكل عشرة نيوتنات'' --- ويشير حيث تشير \omterm{def:g2:pushes-and-pulls:force}{القوة}:
فأسهم قوى الطفولة صارت كمّية.
\end{method}

\section{صيغة الوزن}

\begin{proposition}[الوزن يتناسب مع الكتلة]\label{prop:g9:weight-vs-mass:formula}
في أيّ موضع معطى، يتناسب \omterm{def:g4:weight-and-mass:weight}{وزن} الجسم $P$ (\omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{N}) مع
كتلته $m$ (\omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{kg}):
\[
P = m \times g ,
\]
حيث $g$، وهي \emph{شدّة الجاذبية}\index{شدّة الجاذبية} في
الموضع، تعطي نيوتنات الجذب لكل \omterm{def:g3:measuring-mass:units}{كيلوغرام} من المادة. وقرب سطح
الأرض،
\[
g \approx \qty{9.8}{N/kg}:
\]
فكل \omterm{def:g3:measuring-mass:units}{كيلوغرام} يُجذب بما دون عشرة نيوتنات بقليل. والصيغة هي
قانون الفصل الماضي بثوب محلّي: فالمقدار $g$ يحزم \omterm{def:g3:measuring-mass:mass}{كتلة} الأرض
ونصف قطرها في عدد واحد يخصّ ``هنا''.
\end{proposition}

\begin{example}[الصيغة في العمل]\label{ex:g9:weight-vs-mass:atwork}
تلميذ كتلته \qty{60}{kg}: $P = 60 \times 9.8 = \qty{588}{N}$.
وكرة قدم كتلتها \qty{450}{g}: $P = 0.45 \times 9.8 \approx
\qty{4.4}{N}$ --- وانتبه للكيلوغرامات. وبالعكس: صندوق وزنه
\qty{343}{N} كتلته $m = P/g = 343 \div 9.8 = \qty{35}{kg}$.
وبالمقلوب: على جبل تزن فيه حقيبة كتلتها \qty{10}{kg}
\qty{97.7}{N}، تكون $g$ المحلّية $97.7 \div 10 =
\qty{9.77}{N/kg}$ --- فالمقدار $g$ يرقّ قليلًا مع الارتفاع، تمامًا
كما وعد مربّع المسافة.
\end{example}

\begin{example}[حقيبة واحدة، وأوزان كثيرة]\label{ex:g9:weight-vs-mass:planets}
كل عالَم يضع $g$ الخاصّة به، فتصير حكاية روّاد الفضاء القديمة
جدولًا. حقيبتك الوفيّة كتلتها \qty{10}{kg}:
\begin{center}
\begin{tabular}{l|c|c}
الموضع & $g$ (\unit{N/kg}) & \omterm{def:g4:weight-and-mass:weight}{وزن} الحقيبة \\
\hline
الأرض & $9.8$ & \qty{98}{N} \\
\omterm{def:g2:moon-in-the-sky:moon}{القمر} & $1.6$ & \qty{16}{N} \\
المرّيخ & $3.7$ & \qty{37}{N} \\
قمم غيوم المشتري & $24.8$ & \qty{248}{N}
\end{tabular}
\end{center}
عشرة كيلوغرامات من المادة في كل مكان --- \omterm{def:g4:weight-and-mass:weight}{ووزن} مختلف على كل
عالَم: \omterm{def:g3:measuring-mass:mass}{فالكتلة} لا تتغيّر، \omterm{def:g4:weight-and-mass:weight}{والوزن} يتبع $g$ المحلّية.
\end{example}

\begin{remark}[الميزانان، محكومًا لهما نهائيًّا]\label{rem:g9:weight-vs-mass:scales}
والآن تنغلق أحكام الطفولة بالصيغ. \omterm{def:g3:levers-and-scales:scale}{فميزان الكفّتين} يقارن
$m_1 g$ في مقابل $m_2 g$: فتضرب $g$ المحلّية في الكفّتين معًا
فتُختصر --- فموازين الكفّتين \omterm{def:g6:measurement-in-science:measuring}{تقيس} \emph{\omterm{def:g3:measuring-mass:mass}{الكتلة}}، صادقةً على
أيّ عالَم. \omterm{def:g4:weight-and-mass:springscale}{والميزان النابضي} يقرأ $P = m g$ مباشرة: فهو يقيس
\emph{\omterm{def:g4:weight-and-mass:weight}{الوزن}}، وفيًّا لموضعه. وميزان الحمّام \omterm{def:g4:weight-and-mass:springscale}{ميزان نابضي} متنكّر،
معلَّم في المصنع بالكيلوغرامات بقسمة وزنك على $g$
\emph{الأرض} --- أمينًا في البيت، ومجامِلًا على \omterm{def:g2:moon-in-the-sky:moon}{القمر}
(\qty{10}{kg} تُقرأ لكل \qty{60}{kg} منك)، ومفتريًا على
المشتري.
\end{remark}

\section{القوى على الورق}

\begin{example}[الكتاب على الطاولة، بالنيوتن]\label{ex:g9:weight-vs-mass:book}
أقدم توازن في هذا الكتاب، وقد صار بالأعداد أخيرًا: معجم كتلته
\qty{2}{kg} يستقرّ على مكتب. وزنه: $P = 2 \times 9.8 =
\qty{19.6}{N}$، مرسومًا سهمًا طوله \qty{2}{cm} بمقياس عشرة
نيوتنات لكل \omterm{def:g2:measuring-length:units}{سنتيمتر}، مصوَّبًا مستقيمًا إلى الأسفل من الكتاب.
ودفعة المكتب السانِدة: \qty{19.6}{N} مستقيمة إلى الأعلى ---
سهم مساوٍ في الجهة المعاكسة. والتعادل تامّ، والسكون برهانه:
صورة الطفولة، وقد صارت ميزانية مقيسة.
\end{example}

\begin{remark}[ماذا تفعل القوى --- القانون الغائب]\label{rem:g9:weight-vs-mass:dynamics}
هذا الفصل يقيس القوى؛ ولا يقول بعدُ ماذا يفعل \emph{فائض}
\omterm{def:g2:pushes-and-pulls:force}{القوة} بالحركة. فالأعمال الأربعة في الطفولة --- التحريك،
والإيقاف، والتدوير، والتشويه --- تنتظر قانونها المضبوط، وهو من
أعظم قوانين الفيزياء كلها: ومجلّد الثانوية يفتتح ميكانيكه به.
وهذه السنة تُتمّ التأسيس: قوى \omterm{def:g9:weight-vs-mass:newton}{بالنيوتن}، وأوزان بالصيغة، وأسهم
بمقياس --- أي قواعد الجملة الآتية.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:1}
سمِّ \omterm{def:g6:measurement-in-science:measuring}{وحدة} \omterm{def:g2:pushes-and-pulls:force}{القوة} وآلتها. وأعطِ مرساتين يوميّتين \omterm{def:g9:weight-vs-mass:newton}{للنيوتن} الواحد
وللمئة \omterm{def:g9:weight-vs-mass:newton}{نيوتن}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:2}
اكتب صيغة \omterm{def:g4:weight-and-mass:weight}{الوزن} بوحداتها، ومعنى $g$ بالكلام.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:3}
احسب الأوزان على الأرض: كلب كتلته \qty{25}{kg}؛ ورغيف كتلته

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:4}
شمّامة معلَّقة تمدّ \omterm{def:g9:weight-vs-mass:dynamometer}{مقياس القوة} إلى \qty{14.7}{N}. فما
كتلتها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:5}
اقرأ مخطّط الفصل: ماذا يعني ميله، وماذا كان الخطّ سيفعل على
\omterm{def:g2:moon-in-the-sky:moon}{القمر}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:6}
الحقيبة \qty{10}{kg} تجول على الأرض \omterm{def:g2:moon-in-the-sky:moon}{والقمر} والمرّيخ والمشتري.
أيّ أعداد الجدول لا تتغيّر أبدًا، وأيّها يتغيّر --- ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:7}
لماذا يقول \omterm{def:g3:levers-and-scales:scale}{ميزان الكفّتين} الصدق على كل عالَم بينما يجامل ميزان
الحمّام على \omterm{def:g2:moon-in-the-sky:moon}{القمر}؟ بالصيغ لا بالمشاعر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:8}
ارسم (أو صِف) سهمَي \omterm{def:g2:pushes-and-pulls:force}{القوة} للمعجم \qty{2}{kg} بمقياس \omterm{def:g2:measuring-length:units}{سنتيمتر}
واحد لكل خمسة نيوتنات: الأطوال والاتجاهات وحكم الميزانية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:9}
تسمح شركة طيران بحدّ قدره \qty{23}{kg} لكل حقيبة. فيحتجّ مسافر: ``على
\omterm{def:g2:moon-in-the-sky:moon}{القمر} لن تزن إلا بضعة نيوتنات!'' لماذا يخصّ حدّ الشركة ---
بمعزل عن الوقود ومتانة الأرضيات --- الكتلةَ بحقّ، وأيّ آلة
ينبغي أن تفحصه في مطار قمري؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:10}
على المرّيخ، تزن ذراع مركبة جوّالة \qty{111}{N}. فما كتلتها؟

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:11}
مقياس \omterm{def:g2:pushes-and-pulls:force}{قوة} صُفّر وهو مسطّح ثم استُعمل معلَّقًا يكذب قليلًا ---
في أيّ اتجاه، ولماذا؟ (أيّ \omterm{def:g4:weight-and-mass:weight}{وزن} إضافي ينضمّ إلى \omterm{def:g4:weight-and-mass:weight}{وزن} الحمل؟)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:weight-vs-mass:12}
صمّم التجربة التي ترسم مخطّط الفصل وتستخرج $g$: العُدّة، وجدول

البروتوكول نفسه سيسلّم لو أُجري في مختبر على \omterm{def:g2:moon-in-the-sky:moon}{القمر} --- أيّ
الأعداد يتغيّر، وأيّ الطريقة تنجو بلا مساس.
\end{exercise}

\section{مسألة: مجهّزو ما بين الكواكب}

\begin{problem}[{مسألة نهاية الأسبوع --- غرفة القياس عند
مجهّزي ما بين الكواكب؛ جسم واحد وأربعة عوالم؛ موازين ومقاييس
قوة ومكتب الأمتعة}]\label{pb:g9:weight-vs-mass:1}
يجهّز المتجر المسافرين إلى مرافئ النظام الشمسي ($g$ \omterm{def:g6:measurement-in-science:measuring}{بوحدة}
\unit{N/kg}: الأرض $9.8$؛ \omterm{def:g2:moon-in-the-sky:moon}{والقمر} $1.6$؛ والمرّيخ $3.7$؛ وسطح
غيوم المشتري $24.8$). وأنت تعمل في غرفة \omterm{def:g6:measurement-in-science:measuring}{القياس}.

\medskip\noindent\textbf{القسم الأول --- \omterm{def:g6:measurement-in-science:measuring}{القياس}.}
يتقدّم مسافر كتلته \qty{70}{kg}.
\begin{enumerate}
  \item ما وزنه عند كل من المرافئ الأربعة؟
  \item وعند أيّ مرفأ يقرأ ميزان الحمّام الكلاسيكي (المعلَّم
        بالكيلوغرامات، والمعاير للأرض) كتلته قراءة صحيحة ---
        وماذا يقرأ عند الثلاثة الأخرى؟ (اقسم كل \omterm{def:g4:weight-and-mass:weight}{وزن} على $g$
        الأرض.)
  \item وميزان المتجر الأمين \omterm{def:g3:levers-and-scales:scale}{ميزان كفّتين} بكتل معلَّمة. فبماذا
        يبلّغ عند كل مرفأ، ولماذا؟
  \item ويجب ألّا يضغط حذاء المسافر الأرض بأكثر من
        \qty{2000}{N} (فأرضيات المحطّات رقيقة). فهل من مرفأ
        يخرق فيه الحذاء مع الجسم (\qty{70}{kg} في المجموع)
        القاعدة؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- مكتب الأمتعة.}
\begin{enumerate}[resume]
  \item \omterm{def:g3:measuring-mass:mass}{كتلة} الصندوق المتّجه إلى المرّيخ \qty{40}{kg}. احسب
        وزنه عند المغادرة (الأرض) وعند الوصول (المرّيخ).
  \item ونقابة العتّالين تحدّ \emph{\omterm{def:g4:weight-and-mass:weight}{الوزن}} المحمول باليد
        عند \qty{200}{N} في كل مرفأ. فما أقصى \emph{\omterm{def:g3:measuring-mass:mass}{كتلة}}
        يجوز لعتّال أن يحملها عند كل واحد؟
  \item وصندوق ملصقته ``\qty{150}{N} على \omterm{def:g2:moon-in-the-sky:moon}{القمر}'' يجب أن يجتاز
        جمارك الأرض \omterm{def:g3:measuring-mass:mass}{بالكتلة}. أجرِ التحويل.
  \item ويقرأ مقياس \omterm{def:g2:pushes-and-pulls:force}{قوة} المكتب حقيبة معلَّقة عند \qty{294}{N}
        على الأرض. فما \omterm{def:g3:measuring-mass:mass}{الكتلة}، وما وزنها على \omterm{def:g2:moon-in-the-sky:moon}{القمر} لبيان
        الشحن؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- شكاوى وطرائف.}
\begin{enumerate}[resume]
  \item يعود لاعب كمال أجسام من \omterm{def:g2:moon-in-the-sky:moon}{القمر} غاضبًا: ``مكاسب قوّة
        بستّة أضعاف، ذهبت في ليلة!'' عزِّه بالصيغة: ما الذي
        تغيّر فعلًا عند كل مرفأ، وما الذي لم يتغيّر قطّ؟
  \item ويشكو طاهٍ على سطح المشتري أن الخفق يشبه رفع
        السنادين، مع أن المضرب ``كتلته \qty{200}{g} لا
        غير''. احسب \omterm{def:g4:weight-and-mass:weight}{وزن} المضرب على السطح واحكم.
  \item ويبيع متجر الهدايا ``نيوتنًا من الشوكولاتة، في أيّ
        مكان''. فعند أيّ مرفأ ينال المشتري أكثر
        \emph{مادة} من الشوكولاتة، وكم \omterm{def:g3:measuring-mass:mass}{كتلة} عند كلٍّ من
        الأرض \omterm{def:g2:moon-in-the-sky:moon}{والقمر} والمشتري؟
  \item اكتب شعار المتجر: جملة واحدة تفصل ما يسافر معك عمّا
        يفرضه كل عالَم عليه.
\end{enumerate}
