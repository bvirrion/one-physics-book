\chapter{التيار المتناوب}\label{ch:g9:alternating-current}

كل دارة بنيتها يومًا كانت تعمل \omterm{def:g3:first-electric-circuit:battery}{ببطارية}: تيار يسير بثبات من $+$
إلى $-$، منضبطًا انضباط عقرب ساعة. لكن المآخذ في جدرانك تسلّم
شيئًا أكثر جموحًا --- تيارًا يعكس اتجاهه مئة مرة في كل ثانية،
وتوتره يتأرجح بلا كلل بين الموجب والسالب. فتعرّف إلى \omterm{def:g9:alternating-current:dcac}{التيار
المتناوب}: الشكل الذي تولد به كهرباء العالم كلها تقريبًا،
وتُشحن، وتُسلَّم.

\section{نوعا التيار}

\begin{definition}[التيار المستمرّ والتيار المتناوب]\label{def:g9:alternating-current:dcac}
\emph{التيار المستمرّ}\index{تيار مستمرّ} (DC) يسري دائمًا في
الاتجاه نفسه، مساقًا \omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} ثابت --- هبة \omterm{def:g3:first-electric-circuit:battery}{البطارية}، ثابتة حتى
ينفد المخزون. و\emph{التيار المتناوب}\index{تيار متناوب} (AC)
يعكس اتجاهه دوريًّا، مساقًا \omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} يتأرجح بسلاسة من الموجب إلى
السالب ثم يعود، مرة بعد مرة --- نظام المأخذ، ونظام المولّد
المتناوب (والفصل التالي يقول لمن).
\end{definition}

\begin{definition}[الدور والتردّد]\label{def:g9:alternating-current:period}
\emph{دور}\index{دور} \omterm{def:g8:voltage-voltmeter:voltage}{توتر} متناوب $T$ هو \omterm{def:g2:measuring-time:duration}{مدة} دورة كاملة واحدة
--- تأرجحة تامّة صعودًا وهبوطًا وعودة، بالثواني. و\emph{تردّده}
$f$ هو عدد الدورات في الثانية، \omterm{def:g8:sound-pitch-loudness:frequency}{بالهرتز} --- \omterm{def:g6:measurement-in-science:measuring}{وحدة} \omterm{def:g3:sound-around-us:sound}{الصوت}، تخدم
ثانية --- وكلٌّ منهما مقلوب الآخر:
\[
f = \frac{1}{T}, \qquad T = \frac{1}{f}.
\]
\end{definition}

\begin{proposition}[إحصاءات الشبكة الحيوية]\label{prop:g9:alternating-current:mains}
في معظم أنحاء العالم تتناوب الشبكة عند $f = \qty{50}{Hz}$ ---
أي خمسين دورة كاملة في الثانية، \omterm{def:g9:alternating-current:period}{بدور} $T = 1/50 =
\qty{0.02}{s}$: عشرون ملّي ثانية لكل تأرجحة (وبعض المناطق
تعمل عند ستّين). فالتيار في سلك \omterm{def:g3:first-electric-circuit:bulb}{مصباح} قراءتك يعكس اتجاهه مئة
مرة في كل ثانية --- مرتين في الدورة --- ولم يسر قطّ سيرًا
ثابتًا من مأخذ إلى آخر.
\end{proposition}

\begin{example}[حساب التردّد]\label{ex:g9:alternating-current:arithmetic}
الزوج المتقالب يحسب في الاتجاهين. مولّد يدور عند
\qty{60}{Hz}: $T = 1/60 \approx \qty{0.017}{s}$ --- سبع عشرة
ملّي ثانية. \omterm{met:g9:alternating-current:scope}{وراسم ذبذبات} يُظهر دورة تدوم \qty{0.04}{s}:
$f = 1/0.04 = \qty{25}{Hz}$. ودينامو درّاجة عند دوس بطيء
يصنع \qty{10}{Hz}: \omterm{def:g9:alternating-current:period}{فالدور} عُشر ثانية --- ومصباحه يخفق خفقانًا
مرئيًّا، عشر نبضات في الثانية، حتى تنعّمها \omterm{def:g7:motion-average-speed:formula}{السرعة} إلى ما وراء
انتباه العين.
\end{example}

\section{قراءة التأرجح}

\begin{method}[قراءة أثر على راسم الذبذبات]\label{met:g9:alternating-current:scope}
\emph{راسم الذبذبات}\index{راسم ذبذبات} --- أو أيّ تطبيق
محلّل --- يرسم \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} في مقابل الزمن على شاشة مربّعات؛ ويضبط
قرصان قيمة كل مربّع \omterm{def:g8:voltage-voltmeter:voltage}{بالفولت} وبالملّي ثانية. ولحصاد أثر:
\begin{enumerate}
  \item عُدّ المربّعات لدورة كاملة واضربها في الزمن لكل مربّع:
        فذلك $T$؛ ثم $f = 1/T$؛
  \item عُدّ المربّعات من الخطّ الأوسط إلى قمّة واضربها في
        الفولتات لكل مربّع: فذلك \emph{\omterm{def:g8:voltage-voltmeter:voltage}{التوتر} الأقصى}
        $U_{\text{الأقصى}}$؛
  \item تحقّق من الشكل: فالشبكة والمولّدات المتناوبة ترسم
        الموجة الناعمة المكرّرة بلا انتهاء التي في الشكل ---
        \emph{الجيبية}، وهو اسم الرياضيات للتأرجحة الكاملة.
\end{enumerate}
\end{method}

\begin{proposition}[التوتر الفعّال]\label{prop:g9:alternating-current:effective}
ما ``\omterm{def:g8:voltage-voltmeter:voltage}{التوتر}'' الخاصّ بتأرجحة تزور كل قيمة من
$+U_{\text{الأقصى}}$ إلى $-U_{\text{الأقصى}}$ مئة مرة في
الثانية؟ العدد الواحد الأمين هو \emph{التوتر
الفعّال}\index{توتر فعّال} $U$: أي \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} المستمرّ الثابت الذي
كان سيسخّن مقاومًا بالقدر نفسه. وهو للجيبية أدنى من القمّة
بكثير --- فالقيمة $U_{\text{الأقصى}}$ نحو $1.4$ ضعف $U$ --- وهو ما
بُني كل مقياس \omterm{def:g8:voltage-voltmeter:voltage}{توتر} متناوب ليقرأه. و\qty{230}{V} المشهورة
للمأخذ هي القيمة الفعّالة؛ أمّا القمم فتبلغ نحو \qty{325}{V}،
خمسين مرة في الثانية، في كل جهة.
\end{proposition}

\begin{example}[أيّ الأجهزة يبالي]\label{ex:g9:alternating-current:whocares}
\omterm{prop:g8:electric-current-effects:three}{الأثر الحراري} يعمل جيّدًا في الحالين --- فخيط \omterm{def:g3:first-electric-circuit:bulb}{المصباح} أو وشيعة
الغلّاية يدفأ بغضّ النظر عن اتجاه المسير، ولهذا تشرب المصابيح
والمسخّنات \omterm{def:g9:alternating-current:dcac}{التيار المتناوب} بلا شكوى. أما الإلكترونيات ---
الهواتف والحواسيب المحمولة وكل ما فيه رقاقات --- فتحتاج إلى
\omterm{def:g9:alternating-current:dcac}{تيار مستمرّ} ثابت: فطوبات شواحنها ومغذّياتها الخفية تحوّل تأرجح
المأخذ إلى ثبات شبيه بثبات \omterm{def:g3:first-electric-circuit:battery}{البطارية} (وتحسّس دفء الطوبة:
فالتحويل يدفع ضريبته). وعالم البطاريات لا يستطيع الشحن على
تأرجحة أيضًا: فالشحن كيمياء معكوسة، والكيمياء تصرّ على اتجاه
واحد ثابت.
\end{example}

\begin{remark}[لماذا اختار العالم التأرجحة]\label{rem:g9:alternating-current:why}
إن كانت البطاريات ثابتة إلى هذا الحدّ المريح، فلماذا وُصلت
الحضارة \omterm{def:g9:alternating-current:dcac}{بالتيار المتناوب}؟ لسببين، كلاهما ينتظر بعد فصلين:
\omterm{def:g9:alternating-current:dcac}{فالتيار المتناوب} هو ما \emph{تصنعه بطبيعتها} الآلات الدوّارة
--- وهو، وهذا هو الحاسم، وحده يغذّي المحوّل، تلك العلبة
الطنّانة المتواضعة التي تتيح للكهرباء أن تسافر بلدًا بأكمله
\omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} هائل وتدخل بيتك \omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} مروَّض. فالشبكة متناوبة لأن الرحلة
تقتضي ذلك؛ والفصلان التاليان يتبعان التيار من الوشيعة الدائرة
إلى جدار المطبخ.
\end{remark}

\begin{remark}[التأرجحة والجسم]\label{rem:g9:alternating-current:safety}
يكسب حساب السلامة القديم تفصيلًا كئيبًا واحدًا: فتأرجحات
الشبكة الخمسون في الثانية تقع قرب الإيقاع الذي يعطّل توقيت
القلب الكهربائي أسهل ما يكون --- فالصدمات المتناوبة، أمبيرًا
\omterm{def:g8:intensity-ammeter:intensity}{بأمبير}، أخطر من الثابتة. وكل قاعدة من قواعد الطفولة قائمة،
مضاعفة: \omterm{def:g8:voltage-voltmeter:voltage}{فتوتر} \qty{230}{V} الفعّال في المأخذ --- و\qty{325}{V}
عند القمّة --- ليست مجرّد \omterm{def:g3:first-electric-circuit:battery}{بطارية} أكبر. بل هي حيوان آخر، أقلّ
تسامحًا.
\end{remark}

\section{مسألة: منضدة راسم الذبذبات}

\begin{problem}[{مسألة نهاية الأسبوع --- ليلة التأهيل على
منضدة راسم الذبذبات؛ أربعة مغذّيات غامضة، وشاشة واحدة؛ دفتر
الفنّي}]\label{pb:g9:alternating-current:1}
أربعة مغذّيات بلا لافتات يجب التعرّف إليها من آثارها. وقرصا
الراسم: \qty{5}{ms} لكل مربّع أفقي، و\qty{2}{V} لكل مربّع
عمودي (إلا حيث يُذكر خلاف ذلك). وأنت تمسك الدفتر.

\medskip\noindent\textbf{القسم الأول --- الآثار الأربعة.}
\begin{enumerate}
  \item المغذّي أ: خطّ مستوٍ، على ارتفاع $2.25$ مربّع فوق
        الوسط. حدّد نوعه وتوتره --- وهويّته المنزلية
        المرجّحة.
  \item المغذّي ب: موجة ناعمة، دورة لكل $4$ مربّعات، وقمم
        بارتفاع $3$ مربّعات. النوع، \omterm{def:g9:alternating-current:period}{والدور}، \omterm{def:g8:sound-pitch-loudness:frequency}{والتردّد}،
        \omterm{def:g8:voltage-voltmeter:voltage}{والتوتر} الأقصى؟
  \item المغذّي ج: خطّ مستوٍ \emph{دون} \omterm{def:g7:sound-production:medium}{الوسط} بمقدار $2.25$
        مربّع. ما هو، وأيّ خطأ صغير يفسّره؟
  \item المغذّي د: موجة ناعمة، دورة لكل مربّعين، وقمم بارتفاع
        $1.5$ مربّع. الإحصاءات كاملة --- ولماذا كان \omterm{def:g3:first-electric-circuit:bulb}{مصباح} د
        سيخفق أقلّ من \omterm{def:g3:first-electric-circuit:bulb}{مصباح} ب لو غذّى كلٌّ منهما واحدًا.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- أسئلة الفنّي.}
\begin{enumerate}[resume]
  \item مقياس \omterm{def:g8:voltage-voltmeter:voltage}{توتر} متناوب على المغذّي ب يقرأ نحو
        \qty{4.2}{V}، مع أن قممه تبلغ \qty{6}{V}. وفّق بين
        العددين.
  \item وأيّ المغذّيات الأربعة يستطيع شحن هاتف مباشرة، وما
        الذي يجب أن يقف بين البقية وبطاريته؟
  \item ولوحة تحذير المنضدة: ``المتناوب يعضّ أشدّ، فولتًا
        \omterm{def:g8:voltage-voltmeter:voltage}{بفولت}.'' أعِد ذكر السبب.
  \item ويسأل متدرّب لماذا يرسم الراسم موجة ب مستديرة ناعمة
        بدل أن تكون متعرّجة. سمِّ الشكل ومنشأه (أيّ آلة، في
        الفصل التالي، ترسمه بطبيعتها؟).
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- أثر الشبكة نفسها.}
عارض معزول بأمان يُظهر الشبكة، عند \qty{100}{V} و\qty{5}{ms}
لكل مربّع.
\begin{enumerate}[resume]
  \item تنبّأ بالأثر: عدد المربّعات لكل دورة، وارتفاع القمّة
        بالمربّعات.
  \item ومن الأثر، استخرج $f$ \omterm{prop:g9:alternating-current:effective}{والتوتر الفعّال} --- وتحقّق
        منهما في مقابل لافتة المأخذ المشهورة.
  \item ويُبدَّل العارض ببنك بطاريات ذي قدرة تسخين فعّالة
        مساوية. صِف الأثر الجديد في سطر واحد.
  \item أغلِق الدفتر بخلاصة الفنّي في ثلاثة أسطر: ماذا يعني
        الخطّ المستوي، وماذا تعني الموجة، وأيّ عدد واحد يتيح
        المقارنة العادلة بينهما.
\end{enumerate}
\end{problem}
