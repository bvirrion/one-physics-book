\chapter{من محطّة التوليد إلى البيت: المولّد المتناوب}\label{ch:g9:alternator}

سنة كاملة من \omterm{def:g9:electric-power-energy:kwh}{الكيلوواط ساعة} جرت عبر تدقيقاتك --- لكن من أين؟
اتبع الكبل خارجًا من البيت، وتحت الشارع، وعبر الأبراج، ينتهِ
بك عند آلة تدور في قاعة عظيمة: \omterm{def:g9:alternator:alternator}{المولّد المتناوب}. وسرّه أجمل
مفاجأة في كهرباء هذا الكتاب: اكتشاف السنة الماضية أن التيار
يصنع مغناطيسية، معكوسًا.

\section{التحريض: الأثر معكوسًا}

\begin{proposition}[التحريض الكهرمغناطيسي]\label{prop:g9:alternator:induction}
حرّك مغناطيسًا قرب \omterm{def:g3:first-electric-circuit:circuit}{دارة مغلقة} --- أو حرّك الدارة قرب \omterm{def:g1:magnets:magnet}{المغناطيس}
--- فما دامت الحركة يظهر \omterm{def:g8:voltage-voltmeter:voltage}{توتر} ويسوق تيارًا: كهرباء
\emph{محرَّضة} بمغناطيسية متغيّرة. والقواعد، من المنضدة:
\begin{enumerate}
  \item لا يحرّض إلا \emph{التغيّر}: \omterm{def:g1:magnets:magnet}{فالمغناطيس} الساكن بجانب
        الوشيعة لا يعطي شيئًا --- مهما قوي \omterm{def:g1:magnets:magnet}{المغناطيس}؛
  \item وحركة أسرع، \omterm{def:g1:magnets:magnet}{ومغناطيس} أقوى، ولفّات أكثر في الوشيعة:
        \omterm{def:g8:voltage-voltmeter:voltage}{توتر} محرَّض أكبر؛
  \item واعكس الحركة ينعكس التيار المحرَّض --- فالإقبال
        والإدبار يسوقان مسيرين متعاكسين.
\end{enumerate}
في السنة الماضية صنع تيارٌ مغناطيسية؛ وفي هذه السنة تصنع
مغناطيسيةٌ متحرّكة تيارًا. والأثران وجها عملة واحدة --- والوجه
الثاني يشغّل الحضارة.
\end{proposition}

\begin{method}[التحريض على المنضدة]\label{met:g9:alternator:bench}
وشيعة كثيرة اللفّات، \omterm{def:g1:magnets:magnet}{ومغناطيس} قضيبي، ومقياس حسّاس صفره في
الوسط:
\begin{enumerate}
  \item صِل المقياس على طرفَي الوشيعة؛ وأمسك \omterm{def:g1:magnets:magnet}{المغناطيس} ساكنًا
        داخلها: تنام الإبرة عند الصفر --- \omterm{def:g2:pushes-and-pulls:force}{فالقوة} وحدها لا
        تحرّض شيئًا؛
  \item ادفع \omterm{def:g1:magnets:magnet}{المغناطيس} إلى الداخل: ترفس الإبرة في جهة ---
        وتعود إلى الصفر لحظة تتوقّف الحركة؛
  \item واسحبه: رفسة مساوية، في الجهة الأخرى؛
  \item وادفعه أسرع: رفسة أكبر. وهزّ \omterm{def:g1:magnets:magnet}{المغناطيس} دخولًا وخروجًا
        بإيقاع، تتأرجح الإبرة يسارًا ويمينًا ويسارًا --- فأنت
        تدير تيارًا متناوبًا بيدك.
\end{enumerate}
\end{method}

\section{المولّد المتناوب}

\begin{definition}[المولّد المتناوب]\label{def:g9:alternator:alternator}
\emph{المولّد المتناوب}\index{مولّد متناوب} تحريض لا يكلّ:
\omterm{def:g1:magnets:magnet}{مغناطيس} يُدار بثبات بجانب وشيعة ثابتة (أو وشيعة تُدار في حقل
مغناطيسي). فكل نصف دورة تجلب قطبًا نحو الوشيعة ثم تحمله
بعيدًا --- تغيّر يتجدّد بلا انتهاء --- فيتأرجح \omterm{def:g8:voltage-voltmeter:voltage}{التوتر} المحرَّض
موجبًا فسالبًا فموجبًا: \emph{متناوبًا}، بحكم البناء، راسمًا
تمامًا الجيبية الناعمة في فصل \omterm{met:g9:alternating-current:scope}{راسم الذبذبات}. وأدِره خمسين دورة
في الثانية يتناوب الخرج عند \qty{50}{Hz}: \omterm{def:g8:sound-pitch-loudness:frequency}{فتردّد} الشبكة
\emph{هو} \omterm{def:g7:motion-average-speed:formula}{سرعة} دوران، تحفظها كل محطّة على الشبكة على خطوة
واحدة.
\end{definition}

\begin{example}[الدينامو على عجلتك]\label{ex:g9:alternator:dynamo}
دينامو الدرّاجة \omterm{def:g9:alternator:alternator}{مولّد متناوب} في الجيب: فالعجلة تدير مغناطيسًا
صغيرًا داخل وشيعة ثابتة، فيضيء \omterm{def:g3:first-electric-circuit:bulb}{المصباح} --- أسطع وبتردّد أعلى
كلما دُست أقوى، تمامًا كما تعد قواعد المنضدة (ويخفق عند الزحف،
كما لاحظ فصل \omterm{def:g9:alternating-current:dcac}{التيار المتناوب}). وثمنه أمين أيضًا: فالدينامو
يجرّ --- وإضاءة \omterm{def:g3:first-electric-circuit:bulb}{المصباح} تكلّف ساقيك جهدًا حقيقيًّا. فالتحريض لا
يعطي \omterm{def:g5:everyday-energy:energy}{طاقة}؛ بل \emph{يحوّل} الحركة إلى كهرباء، جولًا \omterm{def:g9:kinetic-energy-safety:joule}{بجول}، ناقص
ضريبة الاحتكاك.
\end{example}

\section{كل محطّة توليد آلة واحدة}

\begin{proposition}[اللبّ المشترك]\label{prop:g9:alternator:core}
كهرباء العالم كلها تقريبًا، مهما كان إعلانها، تُصنع بالطريقة
نفسها: شيء يدير \emph{عنفة} --- طاحونة \omterm{def:g2:air-around-us:air}{هواء} متينة من الريش ---
والعنفة تدير مولّدًا متناوبًا. ولا تختلف محطّات التوليد إلا في
\emph{ما الذي يدفع الريش}:
\begin{enumerate}
  \item المحطّات الحرارية تغلي الماء بحرق الفحم أو \omterm{def:g3:solids-liquids-gases:gas}{الغاز} ---
        فنفّاثات البخار تدير العنفة؛
  \item والمحطّات النووية تغلي الماء \omterm{def:g5:heat-and-insulation:heat}{بحرارة} شطر الذرّات بدل
        ذلك --- فبخار ثانية؛
  \item والسدود الكهرمائية تدع بحيرة جبلية تسقط عبر الريش ---
        سلسلة السدّ في الطفولة، مكتملةً؛
  \item وعنفات الرياح تتخطّى الوسيط: \omterm{def:g2:air-around-us:wind}{فالريح} هي دفعة الريش،
        \omterm{def:g9:alternator:alternator}{ومولّد متناوب} في كل حجرة.
\end{enumerate}
واستثناء واحد يؤكّد القاعدة: فالألواح الشمسية تصنع الكهرباء
بلا دوران البتّة --- بأثر ضوئي من نصيب سنوات الجامعة.
\end{proposition}

\begin{example}[سلاسل، مدقَّقة]\label{ex:g9:alternator:chains}
اكتب كل محطّة سلسلةَ \omterm{def:g5:everyday-energy:energy}{طاقة} كما في الطفولة، بمفردات احترافية
الآن. السدّ: ارتفاع مخزون $\to$ حركة ماء ساقط $\to$ دوران
$\to$ كهرباء --- والشمس، تذكّر، هي التي رفعت تلك البحيرة
\omterm{def:g6:water-states:changes}{بالتبخّر}: \omterm{def:g9:electric-power-energy:power}{فالقدرة} الكهرمائية \omterm{def:g1:light-and-shadows:source}{ضوء} شمس مؤجَّل مطرًا. ومحطّة
الفحم: مخزون كيميائي $\to$ \omterm{def:g5:heat-and-insulation:heat}{حرارة} $\to$ حركة بخار $\to$ دوران
$\to$ كهرباء --- مع تسرّب الدفء عند كل سهم، كما تعترف أعمدة
دخان أبراج التبريد. \omterm{def:g2:air-around-us:wind}{والريح}: \omterm{def:g2:air-around-us:air}{هواء} متحرّك (حرّكته الشمس) إلى
دوران مباشرة. وتُسمّى المنابع التي تعيد الطبيعة ملأها --- الماء
\omterm{def:g2:air-around-us:wind}{والريح} \omterm{def:g1:light-and-shadows:source}{وضوء} الشمس --- \emph{متجدّدة}\index{متجدّدة}؛ أما
المخزونات المحروقة والمشطورة فتُنفق إلى الأبد بمقاييس الزمن
البشري.
\end{example}

\begin{remark}[لماذا تطنّ الأبراج عند 400000 فولت]\label{rem:g9:alternator:transport}
يحلّ أجل آخر وعود فصل \omterm{def:g9:alternating-current:dcac}{التيار المتناوب}. تحتاج بلدة، ولنقل، إلى
مئة مليون \omterm{def:g9:electric-power-energy:power}{واط}. وبمقتضى $P = U I$، تستطيع تلك \omterm{def:g9:electric-power-energy:power}{القدرة} أن تسافر
تيارًا هائلًا \omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} متواضع --- أو تيارًا متواضعًا \omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} هائل.
لكن الأسلاك تسخن بفعل \emph{التيار} (\omterm{prop:g8:electric-current-effects:three}{الأثر الحراري} القديم)، وكل
درجة من تدفئة الأبراج \omterm{def:g5:everyday-energy:energy}{طاقة} مدفوعة الثمن ضائعة على الغربان.
فالشبكة تنقل عند توترات تحبس الأنفاس --- مئات الآلاف من
الفولتات، وتيارات ضئيلة، وضياع أدنى --- ثم تخفض \omterm{def:g8:voltage-voltmeter:voltage}{التوتر}، حيًّا
حيًّا، إلى \qty{230}{V} المروَّضة في جدارك. وآلة الخفض،
\emph{المحوّل}، وشيعتان وتحريض مرة أخرى --- ولا تعمل إلا على
تيار يتناوب: فالشبكة كلها تنطق \omterm{def:g9:alternating-current:dcac}{بالتيار المتناوب} لأن المحوّل
كذلك.
\end{remark}

\section{مسألة: نوبة ليل في المحطّة}

\begin{problem}[{مسألة نهاية الأسبوع --- نوبة ليل في المحطّة
الكهرمائية؛ تحريض وخطوة واحدة وعاصفة؛ سجلّ
المهندسة}]\label{pb:g9:alternator:1}
ترافق مهندسة الليل عند سدّ الوادي --- بحيرة في الأعلى، وبلدة
في الأسفل، ومولّدات متناوبة تطنّ بينهما.

\medskip\noindent\textbf{القسم الأول --- قاعة الآلات.}
\begin{enumerate}
  \item تفتح المهندسة بوّابة: فيرعد الماء عبر العنفة الثالثة،
        وتستيقظ مقاييس مولّدها المتناوب. اكتب سلسلة \omterm{def:g5:everyday-energy:energy}{الطاقة} من
        البحيرة إلى القضيب العمومي.
  \item وداخل \omterm{def:g9:alternator:alternator}{المولّد المتناوب}، ما الذي يتحرّك وما الذي يقف
        ساكنًا --- ولماذا يجب أن يتحرّك شيء أصلًا؟
  \item والشبكة تعمل عند \qty{50}{Hz}. وتراقب المهندسة قرصًا
        يبقي العنفة الثالثة عند الدوران المطابق تمامًا. فماذا
        يفسد عند \omterm{def:g7:motion-average-speed:formula}{السرعة} الخطأ؟ (ماذا كان سيفعل \omterm{def:g8:sound-pitch-loudness:frequency}{تردّد}
        توترها؟)
  \item ويسأل متدرّب إن كان \omterm{def:g1:magnets:magnet}{مغناطيس} أقوى في الدوّار سيعطي
        \omterm{def:g9:electric-power-energy:power}{قدرة} إضافية ``مجّانية''. صحّح الحلم بثمن الدينامو
        الأمين.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- طلب البلدة.}
\begin{enumerate}[resume]
  \item عند الثالثة فجرًا تسحب البلدة \qty{2e7}{W}. فعند
        \qty{200000}{V} على خطّ النقل، أيّ تيار يغادر
        المحطّة؟
  \item \omterm{def:g9:electric-power-energy:power}{والقدرة} نفسها عند \qty{230}{V} كانت ستحتاج إلى أيّ
        تيار --- وماذا تقول المقارنة عن عمل المحوّل؟
  \item وعند السابعة تستيقظ الغلّايات والمسخّنات: فيتضاعف
        الطلب. فما الذي يجب أن تطعمه المهندسة العنفات أكثر،
        وماذا كان سيحدث \omterm{def:g8:sound-pitch-loudness:frequency}{لتردّد} الشبكة لو أطعمتها أقلّ مما
        ينبغي؟ (فالآلات تبطئ تحت الحمل، كدرّاج على تلّة.)
  \item وتخرج المحطّة الشقيقة --- مزرعة رياح على الحافة ---
        من الخدمة إذ تموت \omterm{def:g2:air-around-us:wind}{الريح}. فأيّ صندوق من سلسلتها خذل،
        وأيّ الصناديق وقفت سليمة تمامًا؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- العاصفة.}
\begin{enumerate}[resume]
  \item يقطع البرق خطّ نقل؛ فتنطلق القواطع. وتعيد المهندسة
        توجيه القدرة على الخطوط المجاورة. فأيّ خاصّية من
        خواصّ الشبكة --- وهي شبكة تفرّع واحدة عظيمة --- تجعل
        إعادة التوجيه ممكنة؟
  \item وأثناء الهرج تخفت أضواء مستودع خفوتًا بنّيًّا دقيقة
        واحدة. فسّر: ما الذي هبط، وبمَ كانت محرّكات المستودع
        ومصابيحه ناقصة التغذية لحظتها؟
  \item والفجر: على السجلّ أن يشرح للمتدرّبين لماذا كانت
        كهرباء الليل كلها، في أصلها، \omterm{def:g1:light-and-shadows:source}{ضوء} شمس. صُغ الشرح في
        سطرين (البحيرة، \omterm{def:g6:water-states:changes}{والتبخّر}، والمطر).
  \item وسطر السجلّ الأخير، كما جرت العادة: فيزياء الليلة في
        جملة واحدة --- الحركة والمغناطيسية وغلّاية البلدة
        الصباحية فيها.
\end{enumerate}
\end{problem}
