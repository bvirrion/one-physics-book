\chapter{الحركة: المنتظمة والمتغيّرة}\label{ch:g9:uniform-varied-motion}

قبل سنتين شخّصت الحركة بالعين: نقاط متساوية التباعد، \omterm{def:g6:describing-motion:uniform}{فحركة
منتظمة}؛ ونقاط متباعدة، فتسارع. وفي هذه السنة تنال النقاط
تدريجًا بالملّيمتر وينال التشخيص أعدادًا --- فترة ففترة، \omterm{def:g7:motion-average-speed:formula}{وسرعة}
\omterm{def:g7:motion-average-speed:formula}{فسرعة}. وفي آخر الفصل تقف جملة هادئة عن \omterm{def:g2:pushes-and-pulls:force}{القوة} والحركة ستحمل
يومًا نصف الفيزياء على ظهرها.

\section{سجلّ الحركة}

\begin{definition}[سجلّات المواضع المؤقَّتة]\label{def:g9:uniform-varied-motion:record}
\emph{سجلّ المواضع المؤقَّت}\index{سجلّ مواضع مؤقَّت} لحركة
يعلّم مواضع الجسم المتحرّك عند نقرات زمنية متساوية --- لقطات
فيديو تُتصفّح واحدة واحدة، أو صورة بومضات متقطّعة، أو مؤقّت
النقاط في المختبر يطبع شريطًا ورقيًّا يجرّه الجسم، بخمسين نقطة
في الثانية. وبين كل علامتين متجاورتين يقع سفر نقرة واحدة:
فالسجلّ يحوّل الحركة إلى قطع قابلة \omterm{def:g6:measurement-in-science:measuring}{للقياس}.
\end{definition}

\begin{method}[قراءة سجلّ بالأعداد]\label{met:g9:uniform-varied-motion:read}
لسجلّ طول نقرته $\tau$ (وليكن \qty{0.02}{s} لخمسين في الثانية،
أو \qty{0.1}{s} مريحة من الفيديو):
\begin{enumerate}
  \item قِس كل فجوة بين علامتين متجاورتين بمسطرة الملّيمتر؛
  \item اقسم كل فجوة على النقرة: فالنسبة $v = d/\tau$ تعطي \omterm{def:g7:motion-average-speed:formula}{السرعة}
        (المتوسّطة) على تلك الفترة الصغيرة --- \omterm{def:g6:measurement-in-science:measuring}{بوحدة}
        \unit{m/s} إن كانت الفجوات بالأمتار؛
  \item ضع السرعات جنبًا إلى جنب واقرأ القصة: أعداد ثابتة،
        \omterm{def:g6:describing-motion:uniform}{فحركة منتظمة}؛ ومتسلّقة، فمتسارعة؛ وهابطة، فمتباطئة؛
  \item اذكر \omterm{def:g7:motion-average-speed:formula}{سرعة} أيّ فترة بوصفها \omterm{def:g7:motion-average-speed:formula}{السرعة} ``عند'' تلك اللحظة:
        فعلى نقرة بهذا القِصَر، تتصافح المتوسّطة واللحظية.
\end{enumerate}
\end{method}

\begin{example}[شريط، محلولًا]\label{ex:g9:uniform-varied-motion:worked}
شريط عربة، ونقراته \qty{0.1}{s}، وفجواته \omterm{def:g2:measuring-length:units}{بالسنتيمتر}: $2.0$
و$3.0$ و$4.0$ و$5.0$ و$6.0$. والسرعات، فجوةً ففجوة:
$0.02 \div 0.1 = \qty{0.2}{m/s}$، ثم $0.3$ و$0.4$ و$0.5$
و\qty{0.6}{m/s}. الحكم: متسارعة --- وبانتظام جميل: \omterm{def:g7:motion-average-speed:formula}{فالسرعة}
تنمو بمقدار \qty{0.1}{m/s} بالضبط في كل نقرة. وللحركات التي
تتسلّق سرعتها بخطوات متساوية في أزمنة متساوية اسم خاصّ ومستقبل
عظيم --- \emph{متسارعة بانتظام}، وهي توقيع الأسباب الثابتة.
\end{example}

\begin{example}[أشهر حركة متسارعة]\label{ex:g9:uniform-varied-motion:freefall}
أسقط حجرًا وصوّره: تتمدّد الفجوات نقرةً فنقرة --- فالسقوط الحرّ
متسارع. ومقيسًا، تنمو سرعته بنحو \qty{9.8}{m/s} في كل ثانية من
السقوط: فبعد ثانية، \qty{9.8}{m/s}؛ وبعد ثانيتين، قرابة
\qty{20}{m/s}. وليست مساواة معدّل النموّ للمقدار $g$ في الموضع
مصادفة --- بل هي أعمق قافية في فيزياء هذه السنة، ومجلّد
الثانوية يبني ميكانيكه عليها. فدوّنها، وسطّر تحتها، ودعها
تنتظر.
\end{example}

\section{السرعة اللحظية، بأمانة}

\begin{definition}[السرعة اللحظية]\label{def:g9:uniform-varied-motion:instant}
\emph{السرعة اللحظية}\index{سرعة لحظية} لحركة عند لحظة ما هي
\omterm{def:g7:motion-average-speed:average}{السرعة المتوسّطة} على فترة قصيرة جدًّا حول تلك اللحظة --- قصيرة
حتى لا يجد فيها للحركة متّسع لتغيير وتيرتها. وهي عدد عدّاد
\omterm{def:g7:motion-average-speed:formula}{السرعة} وعدد الرادار: فكلاهما، في الحقيقة، يقيس تمامًا كما
تفعل \cref{met:g9:uniform-varied-motion:read}، على نقرات
كطرفة عين. (وأمّا ما تعنيه ``قصيرة جدًّا'' بصرامة تامّة فهو
السؤال العظيم الذي تجيب عنه رياضيات السنة الأخيرة من الثانوية
--- وذلك الجواب أنشأ العلم الحديث.)
\end{definition}

\begin{example}[المتوسّطة واللحظية جنبًا إلى جنب]\label{ex:g9:uniform-varied-motion:sidebyside}
رحلة مترو بين محطّتين: \omterm{def:g7:motion-average-speed:average}{السرعة المتوسّطة} --- المسافة الكلّية
على الزمن الكلّي --- \qty{35}{km/h}. \omterm{def:g9:uniform-varied-motion:instant}{والسرعة اللحظية}: $0$ عند
الرصيفين، و\qty{70}{km/h} في منتصف النفق. وفي \omterm{def:g6:describing-motion:uniform}{الحركة المنتظمة}،
وفيها وحدها، يندمج المفهومان في عدد ثابت واحد: \omterm{def:g6:describing-motion:uniform}{فالحركة
المنتظمة} هي بالضبط الحركة التي تجعل \omterm{def:g7:motion-average-speed:formula}{السرعة} مقدارًا واحدًا
أمينًا لا يتعلّق بفترة.
\end{example}

\section{الجملة الهادئة}

\begin{proposition}[القوى المتوازنة تترك الحركة بلا تغيير]\label{prop:g9:uniform-varied-motion:inertia}
حين تلغي القوى المؤثّرة في جسم بعضها بعضًا --- أو حين لا يؤثّر
فيه شيء منها البتّة --- لا تتغيّر حركته: فالساكن يبقى ساكنًا،
والمتحرّك يمضي في خطّ مستقيم \omterm{def:g7:motion-average-speed:formula}{بسرعة} ثابتة. ولا يقع تغيّر
الحركة --- تسارعًا أو تباطؤًا أو انعطافًا --- إلا ما دامت
تؤثّر \omterm{def:g2:pushes-and-pulls:force}{قوة} \emph{غير متوازنة}. وهذه الجملة، مذكورةً هنا
ومؤجَّلةً بأمانة، هي قانون الميكانيك الافتتاحي؛ ومجلّد
الثانوية يبدأ بها ولا يفلتها أبدًا.
\end{proposition}

\begin{example}[قراءة العالم بالجملة]\label{ex:g9:uniform-varied-motion:reading}
القرص على الجليد الأملس ينزلق مستقيمًا ثابتًا --- فالقوى
متوازنة (\omterm{def:g4:weight-and-mass:weight}{الوزن} هابطًا والجليد صاعدًا)، والحركة بلا تغيير: فلا
حاجة إلى دفعة \emph{لمواصلة} السير. والسيارة الكابحة تبطئ:
\omterm{def:g2:pushes-and-pulls:force}{قوة} واحدة غير متوازنة، هي قبضة الطريق على الإطارات، مؤثّرة
إلى الخلف. والدرّاج المنعطف: \omterm{def:g2:pushes-and-pulls:force}{قوة} غير متوازنة جانبًا، تجهّزها
الإطارات المائلة. والطائرة المسافرة \omterm{def:g7:motion-average-speed:formula}{بسرعة} ثابتة قدرها
\qty{900}{km/h}: الدفع يوازن السحب، والرفع يوازن \omterm{def:g4:weight-and-mass:weight}{الوزن} ---
أربع قوى، وتعادل تامّ، \omterm{def:g6:describing-motion:uniform}{وحركة منتظمة}. فالسكون والسير المستقرّ،
عند الفيزياء، حالة مسالمة واحدة.
\end{example}

\begin{remark}[لماذا تفاجئ الجملة]\label{rem:g9:uniform-varied-motion:surprise}
تبدو الحياة اليومية محتجّة: كفّ عن الدوس تبطئ الدرّاجة ---
أفلا تحتاج الحركة \emph{إلى} \omterm{def:g2:pushes-and-pulls:force}{قوة} إذن؟ لكن الدرّاجة المبطئة
ليست بلا \omterm{def:g2:pushes-and-pulls:force}{قوة}: فالاحتكاك \omterm{def:g2:air-around-us:air}{والهواء} يدفعان إلى الخلف، بلا توازن،
ولهذا بالضبط تتحلّل الوتيرة. أزِلهما --- حلبة الجليد، وخلاء
الفضاء --- تنطلق الحركة إلى الأبد: فالمسابر الفضائية التي
أُطلقت قبل أن يلتقي جدّاك ما زالت منطلقة الآن، ومحرّكاتها
باردة. وقد احتاجت البشرية ألفَي سنة لترى ما وراء قناع
الاحتكاك؛ ولك أنت ميزة حلبات الجليد والمسابر الفضائية.
\end{remark}

\section{مسألة: مكتب الأشرطة}

\begin{problem}[{مسألة نهاية الأسبوع --- أصيل في مكتب تحليل
الحركة؛ أربعة أشرطة، وأربعة أحكام؛ قافية المفتّش}]\label{pb:g9:uniform-varied-motion:1}
يحلّل المكتب السجلّات المؤقَّتة للمدارس والأندية الرياضية
والمحاكم. والنقرات \qty{0.1}{s} في كل مكان؛ والفجوات تجيء
\omterm{def:g2:measuring-length:units}{بالسنتيمتر}. وأنت تمسك المسطرة.

\medskip\noindent\textbf{القسم الأول --- أحكام روتينية.}
\begin{enumerate}
  \item الشريط أ (ماشٍ في ممرّ): $8.0$ و$8.0$ و$8.0$ و$8.0$
        و$8.0$. الحكم \omterm{def:g7:motion-average-speed:formula}{والسرعة} --- بوحدتَي \unit{m/s}
        و\unit{km/h}.
  \item الشريط ب (انطلاقة عدّاء): $2.0$ و$4.0$ و$6.0$ و$8.0$
        و$10.0$. الحكم، والسرعتان الأولى والأخيرة، ونموّ
        \omterm{def:g7:motion-average-speed:formula}{السرعة} لكل ثانية.
  \item الشريط ج (كرة متدحرجة تلقى رملًا): $10.0$ و$7.0$
        و$4.5$ و$2.5$ و$1.0$. الحكم --- وسرد جملة \omterm{def:g2:pushes-and-pulls:force}{القوة}
        للرمل.
  \item الشريط د (لغز): $5.0$ و$5.0$ و$5.0$ و$7.5$ و$10.0$.
        احكِ قصة د من فصلين، وعلّم النقرة التي وقع عندها
        شيء.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- مكتب الفيزياء في
المكتب.}
\begin{enumerate}[resume]
  \item يصرّ زبون أن درّاجته النارية للتوصيل ``كانت تسير
        \qty{30}{km/h}، يا سيّدي الضابط، في المتوسّط''. وقد
        سجّل الرادار \qty{55}{km/h} عند التقاطع. اشرح للزبون
        لماذا يمكن أن يصدق العددان كلاهما، وأيّهما يقرأ
        القانون.
  \item عدّاء الشريط ب يكسب \omterm{def:g7:motion-average-speed:formula}{السرعة} بخطوات متساوية. أيّ كلمة
        واحدة يختم بها المكتب حركةً كهذه، وعلامَ يدلّ ثبات
        النموّ في شأن \omterm{def:g2:pushes-and-pulls:force}{القوة} الدافعة؟
  \item وماشي الشريط أ، بمقتضى جملة \omterm{def:g2:pushes-and-pulls:force}{القوة}، يمشي تحت قوى
        متوازنة. سمِّ التوازن لماشٍ ثابت (ما الذي يدفع، وما
        الذي يقاوم).
  \item وشريط سقوط حجر يُظهر سرعات فترات $0.98$ و$1.96$
        و\qty{2.94}{m/s}. تحقّق من النموّ لكل ثانية وسمِّ
        العدد الذي يستنسخه.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- القضية.}
لوح تزلّج تدحرج من منحدر عبر ساحة المدرسة إلى حوض زهور؛
والحارس يتّهم ``\omterm{def:g7:motion-average-speed:formula}{السرعة} الطائشة''، والمتزلّج يزعم ``كنت أبطئ
طول الطريق''. وتعطي كاميرا الساحة الفجوات: عند مخرج المنحدر
$12.0$، ثم $11.5$ و$11.0$ و$10.5$ و$10.0$ إلى حوض الزهور.
\begin{enumerate}[resume]
  \item الحكم في زعم المتزلّج --- بسرعات الفترات.
  \item \omterm{def:g7:motion-average-speed:formula}{سرعة} المخرج \omterm{def:g7:motion-average-speed:formula}{وسرعة} دخول حوض الزهور \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{km/h}:
        أكانت إحداهما ``طائشة'' بجانب \qty{15}{km/h} لدرّاج
