\chapter{القدرة والطاقة الكهربائيتان}\label{ch:g9:electric-power-energy}

فاتورة الكهرباء لا تحاسبك على الفولتات، ولا على الأمبيرات ---
بل تحاسبك على شيء ظلّت أجهزة عائلتك تضربها ببعضها في صمت طوال
الشهر. وهذا الفصل يزوّج المقدارين الكهربائيين في ثالث، ويعطي
الإنفاق السريع والبطيء اسميهما الصحيحين، وينتهي عند العدّاد
قرب الباب: فيزياء تستطيع تدقيقها في مقابل مال حقيقي.

\section{القدرة: معدّل التسليم}

\begin{definition}[القدرة والواط]\label{def:g9:electric-power-energy:power}
\emph{قدرة}\index{قدرة} جهاز $P$ هي المعدّل الذي يحوّل به
\omterm{def:g5:everyday-energy:energy}{الطاقة} --- جولات في الثانية. ووحدتها \emph{الواط}\index{واط}
(\unit{W}): \omterm{def:g9:kinetic-energy-safety:joule}{جول} واحد في كل ثانية، تكريمًا لمهندس عصر البخار؛
و\emph{الكيلوواط} (\qty{1}{kW} $=$ \qty{1000}{W}) يخدم أثقال
المطبخ. والقدرة ليست مقدارًا من \omterm{def:g5:everyday-energy:energy}{الطاقة} بل \emph{وتيرة} إنفاق
--- جريان الصنبور، لا محتوى الدلو.
\end{definition}

\begin{proposition}[القدرة الكهربائية]\label{prop:g9:electric-power-energy:UI}
جهاز يُغذّى \omterm{def:g8:voltage-voltmeter:voltage}{بتوتر} $U$ ويسحب تيارًا $I$ ينال \omterm{def:g9:electric-power-energy:power}{قدرة} كهربائية
\[
P = U \times I ,
\]
واطات من فولتات مضروبة في أمبيرات. والصيغة تحسب في الاتّجاهات
الثلاثة كلها، ووصفتها الثانية تدير علبة المصاهر: فالعلاقة $I = P/U$
تقول أيّ تيار ستسحبه \omterm{def:g9:electric-power-energy:power}{قدرة} الجهاز الاسمية من الشبكة.
\end{proposition}

\begin{example}[سلّم البيت]\label{ex:g9:electric-power-energy:ladder}
اقرأ اللوحات على آلات العائلة: \omterm{def:g1:light-and-shadows:source}{ضوء} مؤشّر، \qty{1}{W}؛ \omterm{def:g3:first-electric-circuit:bulb}{ومصباح}
حديث، \qty{5}{W} (وسلفه المتوهّج كان ينفق ستّين \omterm{def:g1:light-and-shadows:source}{للضوء} نفسه ---
والفرق كان \omterm{def:g5:heat-and-insulation:heat}{حرارة})؛ وحاسوب محمول، \qty{50}{W}؛ وتلفاز،
\qty{100}{W}؛ وغسّالة تسخّن ماءها، \qty{2000}{W}؛ وغلّاية،
\qty{2300}{W}؛ وسخّان ماء فوري، حتى \qty{9000}{W}. ونمط
السلّم: كل ما وظيفته \emph{التسخين} يتسلّق إلى الكيلوواطات ---
فالدفء أغلى أشكال \omterm{def:g5:everyday-energy:energy}{الطاقة} تسليمًا.
\end{example}

\begin{example}[حساب علبة المصاهر]\label{ex:g9:electric-power-energy:fuses}
الغلّاية على شبكة \qty{230}{V}: $I = 2300 \div 230 =
\qty{10}{A}$ --- جهاز العشرة \omterm{def:g8:intensity-ammeter:intensity}{أمبير} في الفصول القديمة، مستنتَجًا
الآن. وسخّان الماء \qty{9000}{W}: قرابة \qty{40}{A}، يطالب
بفرعه المتين وقاطعه الخاصّ. وأمّا مشترك المآخذ المثقل في تحقيق
الحريق: فغلّاية ومسخّن ومقلاة تجمع تيارات $P/U$ لها فوق تقنين
\omterm{def:g5:series-parallel-circuits:parallel}{الفرع} --- ثلاثة أبرياء، وجريمة حسابية واحدة، كما جاء في
الاتّهام تمامًا.
\end{example}

\section{الطاقة: المقدار المسلَّم}

\begin{proposition}[الطاقة من القدرة والزمن]\label{prop:g9:electric-power-energy:Pt}
جهاز قدرته $P$ يعمل زمنًا $t$ يحوّل \omterm{def:g5:everyday-energy:energy}{الطاقة}
\[
E = P \times t ,
\]
جولات من واطات مضروبة في ثوانٍ. صنبور سريع أو تقطير طويل،
يمتلئ الدلو بالجداء: فثلاث دقائق لغلّاية \qty{2300}{W}
($2300 \times 180 \approx \qty{4.1e5}{J}$) تنفق أكثر من يوم
كامل \omterm{def:g3:first-electric-circuit:bulb}{لمصباح} \qty{5}{W} ($5 \times 86400 = \qty{4.3e5}{J}$)
--- بل هما متعادلان تقريبًا في الحقيقة: فغليان واحد يعادل يوم
\omterm{def:g1:light-and-shadows:source}{ضوء} جيّد.
\end{proposition}

\begin{definition}[الكيلوواط ساعة]\label{def:g9:electric-power-energy:kwh}
طاقات البيوت تصنع أعداد جولات ثقيلة، فينطق العدّاد
\emph{بالكيلوواط ساعة}\index{كيلوواط ساعة} (\unit{kWh}): وهي
\omterm{def:g5:everyday-energy:energy}{طاقة} كيلوواط واحد مستدامًا ساعة واحدة،
\[
\qty{1}{kWh} = 1000 \times 3600 = \qty{3.6e6}{J}
\]
--- أي ثلاثة ملايين وستّمئة ألف \omterm{def:g9:kinetic-energy-safety:joule}{جول} لكل \omterm{def:g6:measurement-in-science:measuring}{وحدة} على الفاتورة.
ومع $P$ بالكيلوواط و$t$ \omterm{ex:g2:measuring-time:clock}{بالساعات}، تسلّم $E = P \times t$
الكيلوواط ساعة مباشرة: وهي الصيغة التي يحيا بها العدّاد.
\end{definition}

\begin{method}[تدقيق جهاز]\label{met:g9:electric-power-energy:audit}
لأيّ آلة وأيّ شهر:
\begin{enumerate}
  \item اقرأ قدرتها $P$ على اللوحة (أو احسب $U \times I$)؛
  \item قدّر زمن تشغيلها $t$ بأمانة --- في اليوم، ثم في
        الشهر؛
  \item $E = P \times t$ \omterm{def:g9:electric-power-energy:kwh}{بالكيلوواط ساعة} ($P$ \omterm{def:g6:measurement-in-science:measuring}{بوحدة}
        \unit{kW}، و$t$ \omterm{ex:g2:measuring-time:clock}{بالساعات})؛
  \item اضرب في التعرفة (وخذ نحو $0.25$ وحدة نقدية لكل
        \unit{kWh}) لتحصل على كلفة الشهر.
\end{enumerate}
الغلّاية، ستّ دقائق يوميًّا: $2.3 \times 0.1 \times 30 =
\qty{6.9}{kWh}$ في الشهر. والتلفاز، ثلاث ساعات يوميًّا:
$0.1 \times 3 \times 30 = \qty{9}{kWh}$. فالقدرات الصغيرة ذات
الساعات الطويلة تنافس القدرات الكبيرة ذات الساعات القصيرة ---
وهو درس التدقيق المتكرّر.
\end{method}

\section{العدّاد والفاتورة}

\begin{example}[قراءة الفاتورة]\label{ex:g9:electric-power-energy:bill}
العدّاد قرب الباب يعدّ كل \omterm{def:g9:electric-power-energy:kwh}{كيلوواط ساعة} يدخل البيت؛ والفاتورة
هي فرق شهره مضروبًا في التعرفة. و$250$ \omterm{def:g6:measurement-in-science:measuring}{وحدة} شهرية لعائلة
معتادة تتفكّك بالتدقيق: تسخين الماء والمشعّات أولًا (نادي
الكيلوواط)، ثم الأجهزة الباردة --- واطات متواضعة، لكنها تعمل
دائمًا --- ثم الطبخ والغسيل \omterm{def:g1:light-and-shadows:source}{والضوء} والإلكترونيات. وقدرة
التدقيق أنه يجد الروافع الحقيقية: فاستبدال عشرة مصابيح قديمة
\qty{60}{W} بمصابيح \qty{5}{W} يوفّر أكثر ألف مرة من فصل كل
شاحن في البيت.
\end{example}

\begin{example}[مصّاصو الاستعداد]\label{ex:g9:electric-power-energy:standby}
حروف صغيرة وساعات طويلة: \omterm{def:g1:light-and-shadows:source}{ضوء} استعداد التلفاز يرتشف
\qty{0.5}{W} --- لكن على مدى ساعات الشهر البالغة $720$:
$0.0005 \times 720 = \qty{0.36}{kWh}$. ومصّاص واحد لا ضرر
منه؛ لكن بيتًا فيه عشرون مرتشفًا \omterm{def:g9:electric-power-energy:power}{بواط} لكل واحد يدفع
$0.02 \times 720 \approx \qty{14}{kWh}$ شهريًّا في لا شيء.
\omterm{def:g9:electric-power-energy:power}{فالقدرة} في الزمن لا تغفر لأيّ حدّ: والعامل الثاني يعضّ حين
يختبئ الأول.
\end{example}

\begin{remark}[القدرة في العالم الأوسع]\label{rem:g9:electric-power-energy:wider}
\omterm{def:g9:electric-power-energy:power}{الواط}، وقد وُلد كهربائيًّا هنا، يقيس كل وتيرة \omterm{def:g5:everyday-energy:energy}{طاقة}. فجسمك في
الراحة يعمل قرب \qty{100}{W} --- مصباحًا متوهّجًا سلفيًّا من
الدفء (وتذكّر خنقة الصفّ المزدحم: ثلاثون تلميذًا، وثلاثة
كيلوواطات من علم الأحياء). والدرّاج يستديم \qty{200}{W}؛ ومحرّك
سيارة يطلق عشرات الكيلوواطات؛ ومحطّة توليد البلدة، مئات
الملايين من الواطات؛ \omterm{def:g1:light-and-shadows:source}{وضوء} الشمس الساقط على البلدة، أكثر من ذلك
براحة --- وهو السقف الذي تعمل تحته كل الوتائر الأخرى. والفصل
التالي يتبع واطات الشبكة إلى منبعها الدوّار.
\end{remark}

\section{مسألة: تدقيق طاقة البيت}

\begin{problem}[{مسألة نهاية الأسبوع --- تدقيق طاقة العائلة؛
لوحات وعدّادات وفاتورة الشهر؛ العثور على الروافع
الحقيقية}]\label{pb:g9:electric-power-energy:1}
مسلَّحةً بالصيغتين وبالتعرفة ($0.25$ لكل \unit{kWh})، تدقّق
العائلة شهرًا (30 يومًا). والجرد: غلّاية \qty{2300}{W}، $6$
دقائق يوميًّا؛ وتلفاز \qty{100}{W}، $4$ ساعات يوميًّا؛ وثلّاجة
--- حالة خاصّة أدناه؛ وعشرة مصابيح \qty{5}{W} لكل واحد، $5$
ساعات يوميًّا؛ وغسّالة \qty{2}{kW}، بدورة ساخنة مدّتها ساعة
واحدة كل يومين؛ وسخّان ماء \qty{2.5}{kW}، ساعتان يوميًّا؛
ومرتشفو الاستعداد بمجموع \qty{10}{W}، يعملون دائمًا.

\medskip\noindent\textbf{القسم الأول --- آلةً آلة.}
\begin{enumerate}
  \item شهر الغلّاية، \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{kWh}.
  \item شهر التلفاز وشهر المصابيح العشرة.
  \item شهر الغسّالة ($15$ دورة) وشهر سخّان الماء.
  \item شهر مرتشفي الاستعداد --- العامل الثاني في العمل.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الثلّاجة
والمجموع.}
ضاغط الثلّاجة يعمل عند \qty{120}{W}، لكن ثلث الوقت لا غير،
متناوبًا ليلًا ونهارًا.
\begin{enumerate}[resume]
  \item قدرتها المتوسّطة الفعّالة، وشهرها \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{kWh}.
  \item اجمع سطور التدقيق السبعة. فأيّ ثلاثة سطور تهيمن؟
  \item فاتورة الشهر بالتعرفة --- وحصة سخّان الماء منها
        بالنقد.
  \item وقد قرأ العدّاد $2150$ \omterm{def:g6:measurement-in-science:measuring}{وحدة} في أول الشهر. تنبّأ
        بقراءته في آخره.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الروافع.}
\begin{enumerate}[resume]
  \item الاقتراح الأول: استحمام أقصر --- فيهبط السخّان إلى
        \qty{1.5}{h} يوميًّا. ما التوفير الشهري، طاقةً
        ومالًا؟
  \item الاقتراح الثاني: حملة عامّة تحضّ على فصل شواحن
        الهواتف (\qty{0.2}{W} لكل واحد وهو عاطل). دقّق شهر
        شاحن واحد عاطل وقارنه بالاقتراح الأول؛ وعلامَ يدلّ
        التباين في شأن التدقيقات في مقابل الشعارات؟
  \item الاقتراح الثالث: تقترح الجدّة العادة القديمة في ملء
        الغلّاية بقدر الحاجة لا غير --- نصف الماء، فنصف
        \omterm{def:g5:everyday-energy:energy}{الطاقة}. ما سطر الغلّاية الجديد، وما توفير الشهر؟
  \item اكتب خلاصات التدقيق: ثلاث جمل --- أين تسكن \omterm{def:g9:electric-power-energy:kwh}{الكيلوواط
        ساعة} حقًّا، وأيّ عامل من عاملَي الصيغة تعلّمه حالة
        الاستعداد، وأيّ تغيير منزلي واحد يدفع أفضل عائد.
\end{enumerate}
