\chapter{La corriente alterna}\label{ch:g9:alternating-current}

Todos los circuitos que has construido en tu vida funcionaban con una
\omterm{def:g3:first-electric-circuit:battery}{pila}: la corriente desfilando sin pausa de $+$ a $-$, obediente
como la aguja de un reloj. Pero los enchufes de tus paredes entregan
algo más salvaje --- una corriente que invierte el sentido cien veces
por segundo, con su \omterm{def:g8:voltage-voltmeter:voltage}{tensión} oscilando sin descanso entre el más y
el menos. Te presento a la \omterm{def:g9:alternating-current:dcac}{corriente alterna}: la forma en que nace,
se transporta y se entrega prácticamente toda la electricidad del
mundo.

\section{Dos clases de corriente}

\begin{definition}[Corriente continua y corriente alterna]\label{def:g9:alternating-current:dcac}
Una \emph{corriente continua}\index{corriente continua} (CC) circula
siempre en el mismo sentido, movida por una \omterm{def:g8:voltage-voltmeter:voltage}{tensión} constante ---
el regalo de la \omterm{def:g3:first-electric-circuit:battery}{pila}, constante hasta que se agota la reserva.
Una \emph{corriente alterna}\index{corriente alterna} (CA) invierte
periódicamente el sentido, movida por una \omterm{def:g8:voltage-voltmeter:voltage}{tensión} que oscila con
suavidad de positiva a negativa y vuelta, una y otra vez --- el régimen
del enchufe, y el del alternador (el capítulo siguiente cuenta de
quién).
\end{definition}

\begin{definition}[Periodo y frecuencia]\label{def:g9:alternating-current:period}
El \emph{periodo}\index{periodo} $T$ de una \omterm{def:g8:voltage-voltmeter:voltage}{tensión} alterna es la
\omterm{def:g2:measuring-time:duration}{duración} de un ciclo completo --- una oscilación entera hacia
arriba, hacia abajo y de vuelta ---, en segundos. Su
\emph{\omterm{def:g8:sound-pitch-loudness:frequency}{frecuencia}} $f$ es el número de ciclos por segundo, en
\omterm{def:g8:sound-pitch-loudness:frequency}{hercios} --- la \omterm{def:g6:measurement-in-science:measuring}{unidad} del \omterm{def:g3:sound-around-us:sound}{sonido}, otra vez de
servicio --- y las dos son inversas la una de la otra:
\[
f = \frac{1}{T}, \qquad T = \frac{1}{f}.
\]
\end{definition}

\begin{proposition}[Los datos vitales de la red]\label{prop:g9:alternating-current:mains}
En casi todo el mundo, la red alterna con $f = \qty{50}{Hz}$ ---
cincuenta ciclos completos por segundo, un \omterm{def:g9:alternating-current:period}{periodo} de $T = 1/50 =
\qty{0.02}{s}$: veinte milisegundos por oscilación (algunas regiones
van a sesenta). La corriente del cable de tu lámpara de leer invierte
el sentido cien veces por segundo --- dos por ciclo --- y no ha
desfilado sin pausa de un \omterm{def:g3:first-electric-circuit:battery}{borne} al otro ni una sola vez.
\end{proposition}

\begin{example}[Aritmética de frecuencias]\label{ex:g9:alternating-current:arithmetic}
La pareja de inversos calcula en los dos sentidos. Un generador que
cicla a \qty{60}{Hz}: $T = 1/60 \approx \qty{0.017}{s}$ --- diecisiete
milisegundos. Un \omterm{met:g9:alternating-current:scope}{osciloscopio} enseña un ciclo que dura
\qty{0.04}{s}: $f = 1/0.04 = \qty{25}{Hz}$. Una dinamo de bicicleta con
pedaleo lento hace \qty{10}{Hz}: \omterm{def:g9:alternating-current:period}{periodo} de una décima de segundo
--- y su lámpara parpadea de manera visible, diez pulsos por segundo,
hasta que la \omterm{def:g7:motion-average-speed:formula}{velocidad} los alisa más allá de lo que nota el ojo.
\end{example}

\section{Leer la oscilación}

\begin{method}[Leer un trazo de osciloscopio]\label{met:g9:alternating-current:scope}
El \emph{osciloscopio}\index{osciloscopio} --- o cualquier aplicación
de análisis --- dibuja la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} frente al tiempo sobre una
pantalla cuadriculada; lo que vale cada cuadro en \omterm{def:g8:voltage-voltmeter:voltage}{voltios} y en
milisegundos lo fijan dos mandos. Para cosechar un trazo:
\begin{enumerate}
  \item cuenta los cuadros de un ciclo completo y multiplica por el
        tiempo por cuadro: eso es $T$; después, $f = 1/T$;
  \item cuenta los cuadros de la línea central a una cresta y
        multiplica por los \omterm{def:g8:voltage-voltmeter:voltage}{voltios} por cuadro: la \emph{\omterm{def:g8:voltage-voltmeter:voltage}{tensión} de
        pico} $U_{\text{max}}$;
  \item comprueba la forma: la red y los alternadores dibujan la
        onda suave y repetida sin fin de la figura --- la
        \emph{sinusoide}, el nombre que le dan las matemáticas a la
        oscilación perfecta.
\end{enumerate}
\end{method}

\begin{proposition}[Tensión eficaz]\label{prop:g9:alternating-current:effective}
¿Cuál es ``la'' \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de una oscilación que visita todos los
valores de $+U_{\text{max}}$ a $-U_{\text{max}}$ cien veces por
segundo? El número único honrado es la \emph{tensión
eficaz}\index{tensión eficaz} $U$: la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} continua constante
que calentaría igual de bien un \omterm{def:g8:ohms-law:resistor}{resistor}. Para la
sinusoide queda bastante por debajo de la cresta --- $U_{\text{max}}$
es unas $1.4$ veces $U$ --- y es lo que están construidos para leer
todos los \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetros} de CA. Los famosos ``\qty{230}{V}'' del
enchufe son el valor eficaz; las crestas llegan en realidad a unos
\qty{325}{V}, cincuenta veces por segundo en cada sentido.
\end{proposition}

\begin{example}[A qué aparatos les importa]\label{ex:g9:alternating-current:whocares}
El \omterm{prop:g8:electric-current-effects:three}{efecto calorífico} funciona igual de bien de las dos maneras ---
un filamento o la \omterm{def:g8:ohms-law:resistance}{resistencia} de un hervidor se calientan sin importar
el sentido del desfile, que es la razón de que las lámparas y las
estufas beban CA sin quejarse. Pero la electrónica --- móviles,
portátiles, todo lo que lleva chips --- necesita CC constante: sus
ladrillos de cargador y sus fuentes de alimentación escondidas
convierten la oscilación del enchufe en una constancia de \omterm{def:g3:first-electric-circuit:battery}{pila} (nota el
calorcito del ladrillo: la conversión paga su impuesto). Y el mundo de
las \omterm{def:g3:first-electric-circuit:battery}{pilas} tampoco puede cargarse con una oscilación: cargar es
química al revés, y la química insiste en un sentido coherente.
\end{example}

\begin{remark}[Por qué el mundo eligió la oscilación]\label{rem:g9:alternating-current:why}
Si las \omterm{def:g3:first-electric-circuit:battery}{pilas} son tan agradablemente constantes, ¿por qué está la
civilización cableada para CA? Dos razones, las dos esperando dos
capítulos más adelante: la \omterm{def:g9:alternating-current:dcac}{corriente alterna} es lo que
\emph{fabrican de forma natural} las máquinas que giran --- y, de manera
decisiva, solo la CA alimenta al transformador, esa humilde caja que
zumba y que permite que la electricidad recorra un país a una
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} colosal y entre en tu casa con una mansa. La red es
alterna porque el viaje lo exige; los dos capítulos siguientes siguen a
la corriente desde la bobina que gira hasta la pared de la cocina.
\end{remark}

\begin{remark}[La oscilación y el cuerpo]\label{rem:g9:alternating-current:safety}
La vieja aritmética de la seguridad gana un detalle sombrío: las
cincuenta oscilaciones por segundo de la red están cerca del ritmo que
más fácilmente altera el propio compás eléctrico del corazón --- las
descargas alternas son, \omterm{def:g8:intensity-ammeter:intensity}{amperio} a \omterm{def:g8:intensity-ammeter:intensity}{amperio}, más
peligrosas que las constantes. Todas las reglas de la infancia siguen
en pie, y por partida doble: los \qty{230}{V} eficaces del enchufe ---
\qty{325}{V} en la cresta --- no son simplemente una \omterm{def:g3:first-electric-circuit:battery}{pila} más
grande. Son un animal distinto y menos indulgente.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:1}
Distingue la CC de la CA en una frase cada una, con la fuente doméstica
de cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:2}
Define \omterm{def:g9:alternating-current:period}{periodo} y \omterm{def:g8:sound-pitch-loudness:frequency}{frecuencia} y escribe su relación de
inversos. Calcula $T$ para \qty{50}{Hz} y $f$ para $T =
\qty{0.01}{s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:3}
¿Cuántas veces por segundo invierte el sentido la corriente de la red a
\qty{50}{Hz}? (Ojo: inversiones por ciclo.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:4}
Un \omterm{met:g9:alternating-current:scope}{osciloscopio} enseña un ciclo que abarca $4$ cuadros a \qty{5}{ms}
por cuadro. Halla $T$ y $f$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:5}
El enchufe dice \qty{230}{V}; la cresta llega a \qty{325}{V}. Nombra
los dos valores y el significado del más pequeño.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:6}
¿Por qué aceptan los hervidores la CA con alegría mientras que los
cargadores de móvil tienen que convertirla? Un efecto, una química.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:7}
Dibuja (o describe) los dos trazos: una \omterm{def:g3:first-electric-circuit:battery}{pila} de \qty{9}{V} y una
\omterm{def:g8:voltage-voltmeter:voltage}{tensión} alterna de \omterm{def:g9:alternating-current:period}{periodo} \qty{20}{ms} y pico
\qty{9}{V}. ¿Qué rasgo único dibujado por el instrumento los distingue
de un vistazo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:alternating-current:8}
Da las dos razones por las que la civilización se cableó para CA, según
las adelanta el capítulo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:9}
Una dinamo de bicicleta lenta enciende su lámpara con un parpadeo
visible; pedaleando más fuerte, el resplandor se vuelve constante.
Explica los dos regímenes con la \omterm{def:g8:sound-pitch-loudness:frequency}{frecuencia} --- y estima la
\omterm{def:g8:sound-pitch-loudness:frequency}{frecuencia} por debajo de la cual el parpadeo molesta, sabiendo que
el cine engaña al ojo con unas $24$ imágenes por segundo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:10}
Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} de CA puesto en un enchufe marca \qty{230}{V}. Un
\omterm{met:g9:alternating-current:scope}{osciloscopio} (teórico) sin miedo puesto en ese mismo enchufe enseña
crestas ¿de qué valor --- y cuántas crestas por segundo, contando los
dos signos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:alternating-current:11}
La red de una región funciona a \qty{60}{Hz} con \qty{120}{V}
eficaces. Calcula su \omterm{def:g9:alternating-current:period}{periodo} y su \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de pico
aproximada --- y di cuáles de tus aparatos de viaje protestarían
(consulta el \cref{ex:g9:alternating-current:whocares}; los cargadores
modernos leen la letra pequeña de sus etiquetas).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:alternating-current:12}
Las películas antiguas de pantallas y de tubos fluorescentes enseñan a
veces barras oscuras que ruedan --- los fotogramas de la cámara
pillando el brillo de la lámpara a media oscilación. Una lámpara de red
a \qty{50}{Hz} se aviva en cada cresta de \emph{cualquiera} de los dos
signos: ¿cuántos pulsos de brillo por segundo? Una cámara a $25$
fotogramas por segundo muestrea ese pulsar: razona por qué el batido
entre los dos ritmos pinta barras que se mueven despacio, y por qué
grabar en una región de \qty{60}{Hz} cambia el dibujo.
\end{exercise}

\section{Problema: el banco del osciloscopio}

\begin{problem}[{Problema de fin de semana --- noche de acreditación en
el banco del osciloscopio; cuatro fuentes misteriosas, una pantalla; el
libro de registro del técnico}]\label{pb:g9:alternating-current:1}
Hay que identificar cuatro fuentes sin etiqueta a partir de sus trazos.
Los mandos del \omterm{met:g9:alternating-current:scope}{osciloscopio}: \qty{5}{ms} por cuadro horizontal y
\qty{2}{V} por cuadro vertical (salvo donde se indique). Tú llevas el
registro.

\medskip\noindent\textbf{Parte I --- Los cuatro trazos.}
\begin{enumerate}
  \item Fuente A: una línea plana, $2.25$ cuadros por encima de la
        línea central. Identifica su clase y su \omterm{def:g8:voltage-voltmeter:voltage}{tensión} --- y su
        identidad doméstica más probable.
  \item Fuente B: una onda suave, un ciclo cada $4$ cuadros y
        crestas de $3$ cuadros de altura. ¿Clase,
        \omterm{def:g9:alternating-current:period}{periodo}, \omterm{def:g8:sound-pitch-loudness:frequency}{frecuencia} y \omterm{def:g8:voltage-voltmeter:voltage}{tensión} de
        pico?
  \item Fuente C: una línea plana \emph{por debajo} de la central
        $2.25$ cuadros. ¿Qué es, y qué error minúsculo lo explica?
  \item Fuente D: una onda suave, un ciclo cada $2$ cuadros y
        crestas de $1.5$ cuadros. Estadísticas completas --- y por
        qué la lámpara de D parpadearía menos que la de B si cada una
        alimentara una.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Las preguntas del técnico.}
\begin{enumerate}[resume]
  \item Un \omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} de CA puesto en la fuente B marca unos
        \qty{4.2}{V}, aunque sus crestas llegan a \qty{6}{V}.
        Concilia los dos números.
  \item ¿Cuál de las cuatro fuentes podría cargar un móvil
        directamente, y qué tiene que haber entre las otras y su
        batería?
  \item La placa de aviso del banco: ``La CA muerde peor,
        \omterm{def:g8:voltage-voltmeter:voltage}{voltio} a \omterm{def:g8:voltage-voltmeter:voltage}{voltio}''. Vuelve a enunciar la razón.
  \item Un aprendiz pregunta por qué el \omterm{met:g9:alternating-current:scope}{osciloscopio} dibuja la onda
        de B tan suavemente redondeada y no en zigzag. Nombra la
        forma y su origen (¿qué clase de máquina, en el capítulo
        siguiente, la dibuja de forma natural?).
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El trazo de la propia red.}
Un demostrador convenientemente aislado enseña la red, a \qty{100}{V} y
\qty{5}{ms} por cuadro.
\begin{enumerate}[resume]
  \item Predice el trazo: cuadros por ciclo y altura de la cresta en
        cuadros.
  \item A partir del trazo, recupera $f$ y la \omterm{prop:g9:alternating-current:effective}{tensión eficaz} ---
        y comprueba las dos contra la famosa etiqueta del enchufe.
  \item El demostrador se cambia por un banco de \omterm{def:g3:first-electric-circuit:battery}{pilas} de la
        misma potencia calorífica eficaz. Describe el nuevo trazo en
        una línea.
  \item Cierra el registro con el resumen de tres líneas del
        técnico: qué significa la línea plana, qué significa la onda
        y qué único número permite compararlas con justicia.
\end{enumerate}
\end{problem}
