\chapter{Peso frente a masa: medir fuerzas}\label{ch:g9:weight-vs-mass}

Hace cinco años, en una \omterm{def:g2:moon-in-the-sky:moon}{Luna} de papel, partiste ``pesado'' en dos: la
\omterm{def:g3:measuring-mass:mass}{masa} para la materia, el \omterm{def:g4:weight-and-mass:weight}{peso} para el tirón. La partición
era sabiduría; ahora se convierte en aritmética. Las
\omterm{def:g2:pushes-and-pulls:force}{fuerzas} reciben su \omterm{def:g6:measurement-in-science:measuring}{unidad} --- con el nombre del hombre de
la manzana que cae ---, el \omterm{def:g4:weight-and-mass:weight}{peso} recibe su fórmula, y la \omterm{def:g3:levers-and-scales:scale}{balanza}
de goma elástica de tu infancia vuelve con traje de profesional.

\section{El newton}

\begin{definition}[El newton]\label{def:g9:weight-vs-mass:newton}
Las \omterm{def:g2:pushes-and-pulls:force}{fuerzas} --- empujones, tirones, \omterm{def:g4:weight-and-mass:weight}{pesos}, el agarre de
la \omterm{def:g9:gravitation:gravitation}{gravitación} --- se miden en \emph{newtons}\index{newton}
(\unit{N}), en honor del autor de la ley de la
\omterm{def:g9:gravitation:gravitation}{gravitación}. Anclas prácticas: un newton es más o menos el
\omterm{def:g4:weight-and-mass:weight}{peso} de una manzana pequeña (la historia sonríe); tu mochila
pesa unos \qty{50}{N}; un apretón de manos firme aprieta con unos
\qty{100}{N}; la correa de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} del capítulo anterior llegaba a
$10^{20}$.
\end{definition}

\begin{definition}[Dinamómetro]\label{def:g9:weight-vs-mass:dynamometer}
Un \emph{dinamómetro}\index{dinamómetro} es el medidor de \omterm{def:g2:pushes-and-pulls:force}{fuerzas}: un
muelle calibrado dentro de una carcasa, que se estira en proporción a
la fuerza aplicada y cuya aguja lee \omterm{def:g9:weight-vs-mass:newton}{newtons}
directamente. Es tu \omterm{def:g3:levers-and-scales:scale}{balanza} de goma elástica hecha mayor --- y, como el
\omterm{def:g8:voltage-voltmeter:voltmeter}{voltímetro} y el \omterm{def:g8:intensity-ammeter:ammeter}{amperímetro} antes que él, jubila otro
trozo de ``parece fuerte'' y lo convierte en números honrados.
\end{definition}

\begin{method}[Medir una fuerza]\label{met:g9:weight-vs-mass:measure}
\begin{enumerate}
  \item elige un \omterm{def:g9:weight-vs-mass:dynamometer}{dinamómetro} cuyo alcance encaje con el trabajo
        --- uno de \qty{2}{N} para estuches, uno de \qty{100}{N}
        para mochilas;
  \item ponlo a cero en la posición de uso (colgando, si la carga va
        a colgar --- el propio \omterm{def:g4:weight-and-mass:weight}{peso} del muelle desplaza el cero);
  \item aplica la \omterm{def:g2:pushes-and-pulls:force}{fuerza} de manera sostenida, a lo largo del \omterm{def:g6:sun-earth-moon:axisorbit}{eje}
        del instrumento, y lee a la altura de los ojos;
  \item anota con la \omterm{def:g6:measurement-in-science:measuring}{unidad} --- y, como las
        \omterm{def:g2:pushes-and-pulls:force}{fuerzas} tienen dirección, anota también la dirección
        cuando importe.
\end{enumerate}
En los esquemas, una \omterm{def:g2:pushes-and-pulls:force}{fuerza} medida se convierte en una flecha
dibujada a una escala declarada --- ``un \omterm{def:g2:measuring-length:units}{centímetro} por cada diez
\omterm{def:g9:weight-vs-mass:newton}{newtons}'' --- que apunta hacia donde apunta la
\omterm{def:g2:pushes-and-pulls:force}{fuerza}: las flechas de \omterm{def:g2:pushes-and-pulls:force}{fuerza} de la infancia, ahora
cuantitativas.
\end{method}

\section{La fórmula del peso}

\begin{proposition}[El peso es proporcional a la masa]\label{prop:g9:weight-vs-mass:formula}
En un lugar dado, el \omterm{def:g4:weight-and-mass:weight}{peso} $P$ (en \unit{N}) de un objeto es
proporcional a su \omterm{def:g3:measuring-mass:mass}{masa} $m$ (en \unit{kg}):
\[
P = m \times g ,
\]
donde $g$, la \emph{intensidad gravitatoria}\index{intensidad
gravitatoria} del lugar, da los \omterm{def:g9:weight-vs-mass:newton}{newtons} de tirón por
\omterm{def:g3:measuring-mass:units}{kilogramo} de materia. Cerca de la superficie de la Tierra,
\[
g \approx \qty{9.8}{N/kg}:
\]
de cada \omterm{def:g3:measuring-mass:units}{kilogramo} se tira con algo menos de diez
\omterm{def:g9:weight-vs-mass:newton}{newtons}. La fórmula es la ley del capítulo anterior con ropa
local: $g$ empaqueta la \omterm{def:g3:measuring-mass:mass}{masa} y el radio de la Tierra en un solo
número para el ``aquí''.
\end{proposition}

\begin{example}[La fórmula en funcionamiento]\label{ex:g9:weight-vs-mass:atwork}
Un alumno de \qty{60}{kg}: $P = 60 \times 9.8 = \qty{588}{N}$. Un balón
de \qty{450}{g}: $P = 0.45 \times 9.8 \approx \qty{4.4}{N}$ --- ojo con
los \omterm{def:g3:measuring-mass:units}{kilogramos}. Hacia atrás: una caja que pesa \qty{343}{N} tiene
una \omterm{def:g3:measuring-mass:mass}{masa} $m = P/g = 343 \div 9.8 = \qty{35}{kg}$. Y del revés: en
una montaña donde una mochila de \qty{10}{kg} pesa \qty{97.7}{N}, el
$g$ local es $97.7 \div 10 = \qty{9.77}{N/kg}$ --- $g$ adelgaza
ligeramente con la altitud, exactamente como prometía el inverso del
cuadrado.
\end{example}

\begin{example}[Una maleta, muchos pesos]\label{ex:g9:weight-vs-mass:planets}
Cada mundo fija su propio $g$, y la vieja historia de los astronautas
se convierte en una tabla. Tu fiel maleta de \qty{10}{kg}:
\begin{center}
\begin{tabular}{l|c|c}
lugar & $g$ (\unit{N/kg}) & \omterm{def:g4:weight-and-mass:weight}{peso} de la maleta \\
\hline
Tierra & $9.8$ & \qty{98}{N} \\
\omterm{def:g2:moon-in-the-sky:moon}{Luna} & $1.6$ & \qty{16}{N} \\
Marte & $3.7$ & \qty{37}{N} \\
cimas de nubes de Júpiter & $24.8$ & \qty{248}{N}
\end{tabular}
\end{center}
Diez \omterm{def:g3:measuring-mass:units}{kilogramos} de materia en todas partes --- y un
\omterm{def:g4:weight-and-mass:weight}{peso} distinto en cada mundo: la \omterm{def:g3:measuring-mass:mass}{masa} no cambia,
mientras que el \omterm{def:g4:weight-and-mass:weight}{peso} sigue al $g$ local.
\end{example}

\begin{remark}[Las dos balanzas, juzgadas para siempre]\label{rem:g9:weight-vs-mass:scales}
Ahora los veredictos de la infancia se cierran con fórmulas. La \omterm{def:g3:levers-and-scales:scale}{balanza}
de platillos compara $m_1 g$ con $m_2 g$: el $g$ local multiplica los
dos platillos y se cancela --- las \omterm{def:g3:levers-and-scales:scale}{balanzas} miden
\emph{\omterm{def:g3:measuring-mass:mass}{masa}}, verídicas en cualquier mundo. El
\omterm{def:g9:weight-vs-mass:dynamometer}{dinamómetro} lee $P = m g$ directamente: mide
\emph{\omterm{def:g4:weight-and-mass:weight}{peso}}, leal a su ubicación. La báscula de baño es un
\omterm{def:g9:weight-vs-mass:dynamometer}{dinamómetro} disfrazado, marcado en fábrica en
\omterm{def:g3:measuring-mass:units}{kilogramos} dividiendo tu \omterm{def:g4:weight-and-mass:weight}{peso} entre el $g$ de la
\emph{Tierra} --- honrada en casa, aduladora en la \omterm{def:g2:moon-in-the-sky:moon}{Luna}
(\qty{10}{kg} marcados por cada \qty{60}{kg} tuyos) y difamadora en
Júpiter.
\end{remark}

\section{Las fuerzas sobre el papel}

\begin{example}[El libro sobre la mesa, en newtons]\label{ex:g9:weight-vs-mass:book}
El \omterm{def:g2:balance-equilibrium:equilibrium}{equilibrio} más antiguo de este libro, por fin con números: un
diccionario de \qty{2}{kg} descansa sobre un escritorio. Su
\omterm{def:g4:weight-and-mass:weight}{peso}: $P = 2 \times 9.8 = \qty{19.6}{N}$, dibujado como una flecha
de \qty{2}{cm} a una escala de diez \omterm{def:g9:weight-vs-mass:newton}{newtons} por
\omterm{def:g2:measuring-length:units}{centímetro}, apuntando derecha hacia abajo desde el libro. El
empujón de sostén del escritorio: \qty{19.6}{N} derechos hacia arriba
--- una flecha igual y de sentido contrario. El empate es exacto, y la
quietud es su demostración: el dibujo de la infancia, ahora un
presupuesto medido.
\end{example}

\begin{remark}[Lo que hacen las fuerzas --- la ley que falta]\label{rem:g9:weight-vs-mass:dynamics}
Este capítulo mide \omterm{def:g2:pushes-and-pulls:force}{fuerzas}; todavía no dice qué le
hace al movimiento un \emph{excedente} de \omterm{def:g2:pushes-and-pulls:force}{fuerza}. Los cuatro
oficios de la infancia --- arrancar, parar, girar, deformar --- esperan
su ley exacta, y es una de las más grandes de toda la física: el
volumen de bachillerato abre con ella su mecánica. Este año se termina
la preparación del terreno: \omterm{def:g2:pushes-and-pulls:force}{fuerzas} en \omterm{def:g9:weight-vs-mass:newton}{newtons},
\omterm{def:g4:weight-and-mass:weight}{pesos} por fórmula, flechas a escala --- la gramática de la frase
que viene.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:1}
Nombra la \omterm{def:g6:measurement-in-science:measuring}{unidad} de \omterm{def:g2:pushes-and-pulls:force}{fuerza} y su instrumento. Da dos
anclas cotidianas para un \omterm{def:g9:weight-vs-mass:newton}{newton} y para cien.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:2}
Escribe la fórmula del \omterm{def:g4:weight-and-mass:weight}{peso} con \omterm{def:g6:measurement-in-science:measuring}{unidades}, y el
significado de $g$ con palabras.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:3}
Calcula \omterm{def:g4:weight-and-mass:weight}{pesos} en la Tierra: un perro de \qty{25}{kg}; una hogaza
de \qty{700}{g}; tú mismo (elige tu \omterm{def:g3:measuring-mass:mass}{masa}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:4}
Un melón colgado estira el \omterm{def:g9:weight-vs-mass:dynamometer}{dinamómetro} hasta \qty{14.7}{N}. ¿Su
\omterm{def:g3:measuring-mass:mass}{masa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:5}
Lee la gráfica del capítulo: ¿qué significa su pendiente y qué haría la
recta en la \omterm{def:g2:moon-in-the-sky:moon}{Luna}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:6}
La maleta de \qty{10}{kg} recorre la Tierra, la \omterm{def:g2:moon-in-the-sky:moon}{Luna}, Marte y
Júpiter. ¿Qué números de la tabla no cambian nunca y cuáles cambian ---
y por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:7}
¿Por qué dice la verdad una \omterm{def:g3:levers-and-scales:scale}{balanza} de platillos en todos los mundos
mientras que la báscula de baño adula en la \omterm{def:g2:moon-in-the-sky:moon}{Luna}? Con fórmulas,
no con sensaciones.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:weight-vs-mass:8}
Dibuja (o describe) las dos flechas de \omterm{def:g2:pushes-and-pulls:force}{fuerza} del diccionario de
\qty{2}{kg} a un \omterm{def:g2:measuring-length:units}{centímetro} por cada cinco \omterm{def:g9:weight-vs-mass:newton}{newtons}:
longitudes, direcciones y el veredicto del presupuesto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:9}
Una compañía aérea permite \qty{23}{kg} por bulto. Un pasajero
argumenta: ``¡En la \omterm{def:g2:moon-in-the-sky:moon}{Luna} solo pesaría unos pocos
\omterm{def:g9:weight-vs-mass:newton}{newtons}!''. ¿Por qué el límite de la compañía --- dejando de
lado el combustible y la \omterm{def:g8:ohms-law:resistance}{resistencia} del suelo --- se refiere con razón
a la \omterm{def:g3:measuring-mass:mass}{masa}, y qué instrumento debería comprobarlo en un
aeropuerto lunar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:10}
En Marte, el brazo de un vehículo explorador pesa \qty{111}{N}. ¿Su
\omterm{def:g3:measuring-mass:mass}{masa}? ¿Cuánto pesaría ese mismo brazo de vuelta en la nave de
montaje, en la Tierra?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:weight-vs-mass:11}
Un \omterm{def:g9:weight-vs-mass:dynamometer}{dinamómetro} puesto a cero tumbado y usado después colgando
miente un poquito --- ¿en qué sentido, y por qué? (¿Qué
\omterm{def:g4:weight-and-mass:weight}{peso} extra se suma al de la carga?)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:weight-vs-mass:12}
Diseña el experimento que dibuja la gráfica del capítulo y extrae $g$:
material, tabla de medidas, la representación gráfica y cómo se lee la
pendiente. Explica después qué entregaría ese mismo protocolo llevado a
cabo en un laboratorio de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} --- qué números cambian y qué
método sobrevive intacto.
\end{exercise}

\section{Problema: los equipadores interplanetarios}

\begin{problem}[{Problema de fin de semana --- el probador de los
equipadores interplanetarios; un cuerpo, cuatro mundos; básculas,
dinamómetros y el mostrador de equipajes}]\label{pb:g9:weight-vs-mass:1}
Los equipadores preparan a los viajeros para los puertos del
\omterm{def:g4:solar-system:family}{Sistema Solar} ($g$, en \unit{N/kg}: Tierra $9.8$; \omterm{def:g2:moon-in-the-sky:moon}{Luna} $1.6$;
Marte $3.7$; cubierta de nubes de Júpiter $24.8$). Tú atiendes el
probador.

\medskip\noindent\textbf{Parte I --- La prueba.}
Se presenta un viajero de \omterm{def:g3:measuring-mass:mass}{masa} \qty{70}{kg}.
\begin{enumerate}
  \item ¿Su \omterm{def:g4:weight-and-mass:weight}{peso} en cada uno de los cuatro puertos?
  \item ¿En qué puerto lee correctamente su \omterm{def:g3:measuring-mass:mass}{masa} la clásica
        báscula de baño (marcada en \omterm{def:g3:measuring-mass:units}{kilogramos} y calibrada para
        la Tierra) --- y qué marca en los otros tres? (Divide cada
        \omterm{def:g4:weight-and-mass:weight}{peso} entre el $g$ de la Tierra.)
  \item La \omterm{def:g3:levers-and-scales:scale}{balanza} honrada de los equipadores es una \omterm{def:g3:levers-and-scales:scale}{balanza} de
        platillos con \omterm{def:g3:measuring-mass:mass}{masas} marcadas. ¿Qué declara en cada puerto, y
        por qué?
  \item Las botas del viajero no deben apretar nunca el suelo con
        más de \qty{2000}{N} (los suelos de las estaciones son
        finos). ¿Hay algún puerto donde botas más cuerpo
        (\qty{70}{kg} en total) rompan la regla?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El mostrador de equipajes.}
\begin{enumerate}[resume]
  \item El baúl con destino a Marte tiene una \omterm{def:g3:measuring-mass:mass}{masa} de \qty{40}{kg}.
        Calcula su \omterm{def:g4:weight-and-mass:weight}{peso} en la salida (Tierra) y en la llegada
        (Marte).
  \item El sindicato de mozos limita a \qty{200}{N} el
        \emph{\omterm{def:g4:weight-and-mass:weight}{peso}} que se lleva a mano en todos los puertos.
        ¿Qué \emph{\omterm{def:g3:measuring-mass:mass}{masa}} máxima puede llevar un mozo en cada
        uno?
  \item Una caja etiquetada como ``\qty{150}{N} en la
        \omterm{def:g2:moon-in-the-sky:moon}{Luna}'' tiene que pasar la aduana terrestre por
        \omterm{def:g3:measuring-mass:mass}{masa}. \omterm{def:g6:light-propagation:ray}{Haz} la conversión.
  \item El \omterm{def:g9:weight-vs-mass:dynamometer}{dinamómetro} del mostrador marca \qty{294}{N} para
        una bolsa colgada en la Tierra. ¿\omterm{def:g3:measuring-mass:mass}{Masa}, y su
        \omterm{def:g4:weight-and-mass:weight}{peso} en la \omterm{def:g2:moon-in-the-sky:moon}{Luna} para el manifiesto?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Quejas y curiosidades.}
\begin{enumerate}[resume]
  \item Un culturista vuelve furioso de la \omterm{def:g2:moon-in-the-sky:moon}{Luna}: ``¡Ganancias
        de \omterm{def:g2:pushes-and-pulls:force}{fuerza} por seis, perdidas de un día para otro!''.
        Consuélalo con la fórmula: ¿qué cambió en realidad en cada
        puerto y qué no cambió nunca?
  \item Un cocinero de la cubierta de Júpiter se queja de que batir
        se parece a levantar yunques, aunque las varillas ``tienen
        una \omterm{def:g3:measuring-mass:mass}{masa} de solo \qty{200}{g}''. Calcula el
        \omterm{def:g4:weight-and-mass:weight}{peso} de las varillas en la cubierta y dictamina.
  \item La tienda de regalos vende ``un \omterm{def:g9:weight-vs-mass:newton}{newton} de chocolate,
        en cualquier sitio''. ¿En qué puerto se lleva el comprador
        más \emph{materia} de chocolate, y cuánta \omterm{def:g3:measuring-mass:mass}{masa} en
        cada uno de la Tierra, la \omterm{def:g2:moon-in-the-sky:moon}{Luna} y Júpiter?
  \item Escribe el lema de los equipadores: una frase que separe lo
        que viaja contigo de lo que cobra cada mundo.
\end{enumerate}
\end{problem}
