\chapter{Potencia y energía eléctricas}\label{ch:g9:electric-power-energy}

La factura de la \omterm{def:g1:light-and-shadows:source}{luz} no cobra por \omterm{def:g8:voltage-voltmeter:voltage}{voltios}, ni por
\omterm{def:g8:intensity-ammeter:intensity}{amperios} --- cobra por algo que los aparatos de tu familia llevan
todo el mes multiplicando en silencio. Este capítulo casa las dos
magnitudes eléctricas en una tercera, les pone su nombre propio al
gasto rápido y al lento, y termina en el contador de la entrada: física
que puedes auditar contra dinero de verdad.

\section{La potencia: el ritmo de la entrega}

\begin{definition}[La potencia y el vatio]\label{def:g9:electric-power-energy:power}
La \emph{potencia}\index{potencia} $P$ de un aparato es el ritmo al que
convierte \omterm{def:g5:everyday-energy:energy}{energía} --- \omterm{def:g9:kinetic-energy-safety:joule}{julios} por segundo. Su
\omterm{def:g6:measurement-in-science:measuring}{unidad} es el \emph{vatio}\index{vatio} (\unit{W}): un
\omterm{def:g9:kinetic-energy-safety:joule}{julio} cada segundo, en honor del ingeniero de la era del vapor;
el \emph{kilovatio} (\qty{1}{kW} $=$ \qty{1000}{W}) sirve a los \omterm{def:g4:weight-and-mass:weight}{pesos}
pesados de la cocina. La potencia no es una cantidad de
\omterm{def:g5:everyday-energy:energy}{energía}, sino un \emph{ritmo} de gasto --- el caudal de un grifo,
no el contenido de un cubo.
\end{definition}

\begin{proposition}[La potencia eléctrica]\label{prop:g9:electric-power-energy:UI}
Un aparato alimentado con la \omterm{def:g8:voltage-voltmeter:voltage}{tensión} $U$ y que absorbe la
corriente $I$ recibe la \omterm{def:g9:electric-power-energy:power}{potencia} eléctrica
\[
P = U \times I ,
\]
\omterm{def:g9:electric-power-energy:power}{vatios} a partir de \omterm{def:g8:voltage-voltmeter:voltage}{voltios} por \omterm{def:g8:intensity-ammeter:intensity}{amperios}. La
fórmula calcula en las tres direcciones, y su segunda receta gobierna
el cuadro de \omterm{def:g7:short-circuits-safety:fuse}{fusibles}: $I = P/U$ dice qué corriente le va a
sacar a la red la \omterm{def:g9:electric-power-energy:power}{potencia} nominal de un aparato.
\end{proposition}

\begin{example}[La escalera doméstica]\label{ex:g9:electric-power-energy:ladder}
Lee las placas de las máquinas de la familia: un piloto indicador,
\qty{1}{W}; una lámpara moderna, \qty{5}{W} (su antepasada
incandescente gastaba sesenta para la misma \omterm{def:g1:light-and-shadows:source}{luz} --- la diferencia
era \omterm{def:g5:heat-and-insulation:heat}{calor}); un portátil, \qty{50}{W}; un televisor,
\qty{100}{W}; una lavadora calentando su agua, \qty{2000}{W}; el
hervidor, \qty{2300}{W}; un calentador instantáneo de agua, hasta
\qty{9000}{W}. La pauta de la escalera: todo lo que tiene por oficio
\emph{calentar} sube hasta los kilovatios --- el calorcito es la forma
más cara de \omterm{def:g5:everyday-energy:energy}{energía} de suministrar.
\end{example}

\begin{example}[La aritmética del cuadro de fusibles]\label{ex:g9:electric-power-energy:fuses}
El hervidor en la red de \qty{230}{V}: $I = 2300 \div 230 =
\qty{10}{A}$ --- el aparato de diez \omterm{def:g8:intensity-ammeter:intensity}{amperios} de los capítulos
antiguos, ahora deducido. El calentador de agua de \qty{9000}{W}: casi
\qty{40}{A}, que exigen su propia \omterm{def:g5:series-parallel-circuits:parallel}{rama} robusta y su propio
automático. Y la regleta sobrecargada de la investigación del incendio:
hervidor, estufa y freidora sumando sus corrientes $P/U$ por encima de
lo que aguanta la \omterm{def:g5:series-parallel-circuits:parallel}{rama} --- tres inocentes, un crimen aritmético,
exactamente como decía la acusación.
\end{example}

\section{La energía: la cantidad entregada}

\begin{proposition}[La energía a partir de la potencia y el tiempo]\label{prop:g9:electric-power-energy:Pt}
Un aparato de \omterm{def:g9:electric-power-energy:power}{potencia} $P$ funcionando durante un tiempo $t$
convierte la \omterm{def:g5:everyday-energy:energy}{energía}
\[
E = P \times t ,
\]
\omterm{def:g9:kinetic-energy-safety:joule}{julios} a partir de \omterm{def:g9:electric-power-energy:power}{vatios} por segundos. Grifo rápido o
hilillo largo, el cubo se llena según el producto: los tres minutos de
un hervidor de \qty{2300}{W} ($2300 \times 180 \approx
\qty{4.1e5}{J}$) gastan más que el día entero de una lámpara de
\qty{5}{W} ($5 \times 86400 = \qty{4.3e5}{J}$) --- casi exactamente un
empate, de hecho: un hervido equivale a un día de buena
\omterm{def:g1:light-and-shadows:source}{luz}.
\end{proposition}

\begin{definition}[El kilovatio-hora]\label{def:g9:electric-power-energy:kwh}
Las \omterm{def:g5:everyday-energy:energy}{energías} domésticas dan cuentas de \omterm{def:g9:kinetic-energy-safety:joule}{julios} desmañadas, así que el
contador habla en \emph{kilovatios-hora}\index{kilovatio-hora}
(\unit{kWh}): la \omterm{def:g5:everyday-energy:energy}{energía} de un kilovatio mantenido durante una
hora,
\[
\qty{1}{kWh} = 1000 \times 3600 = \qty{3.6e6}{J}
\]
--- tres millones y seiscientos mil \omterm{def:g9:kinetic-energy-safety:joule}{julios} por
\omterm{def:g6:measurement-in-science:measuring}{unidad} de la factura. Con $P$ en kilovatios y $t$ en
\omterm{ex:g2:measuring-time:clock}{horas}, $E = P \times t$ entrega kilovatios-hora directamente: la
fórmula de la que vive el contador.
\end{definition}

\begin{method}[Auditar un aparato]\label{met:g9:electric-power-energy:audit}
Para cualquier máquina y cualquier mes:
\begin{enumerate}
  \item lee su \omterm{def:g9:electric-power-energy:power}{potencia} $P$ en la placa (o calcula $U \times
        I$);
  \item estima con honradez su tiempo de funcionamiento $t$ --- por
        día y después por mes;
  \item $E = P \times t$ en \omterm{def:g9:electric-power-energy:kwh}{kilovatios-hora} ($P$ en \unit{kW},
        $t$ en \omterm{ex:g2:measuring-time:clock}{horas});
  \item multiplica por la tarifa (toma unas $0.25$
        \omterm{def:g6:measurement-in-science:measuring}{unidades} monetarias por \unit{kWh}) para el coste del
        mes.
\end{enumerate}
El hervidor, seis minutos diarios: $2.3 \times 0.1 \times 30 =
\qty{6.9}{kWh}$ al mes. El televisor, tres \omterm{ex:g2:measuring-time:clock}{horas} diarias: $0.1
\times 3 \times 30 = \qty{9}{kWh}$. Las \omterm{def:g9:electric-power-energy:power}{potencias} pequeñas con
muchas \omterm{ex:g2:measuring-time:clock}{horas} compiten con las grandes de pocas --- la lección
recurrente de la auditoría.
\end{method}

\section{El contador y la factura}

\begin{example}[Leer la factura]\label{ex:g9:electric-power-energy:bill}
El contador de la entrada cuenta todos los \omterm{def:g9:electric-power-energy:kwh}{kilovatios-hora} que
entran en la casa; la factura es la diferencia de su mes multiplicada
por la tarifa. Las $250$ \omterm{def:g6:measurement-in-science:measuring}{unidades} mensuales de una familia típica
se descomponen por auditoría: primero el agua caliente y los radiadores
(el club del kilovatio), después los aparatos de frío ---
\omterm{def:g9:electric-power-energy:power}{vatios} modestos, pero funcionando siempre --- y luego la cocina,
el lavado, la \omterm{def:g1:light-and-shadows:source}{luz} y la electrónica. El poder de la auditoría:
encuentra las \omterm{def:g3:levers-and-scales:lever}{palancas} de verdad. Cambiar diez
\omterm{def:g3:first-electric-circuit:bulb}{bombillas} viejas de \qty{60}{W} por unas de \qty{5}{W} ahorra mil
veces más que desenchufar todos los cargadores de la casa.
\end{example}

\begin{example}[Los vampiros del modo de espera]\label{ex:g9:electric-power-energy:standby}
Letra pequeña, muchas \omterm{ex:g2:measuring-time:clock}{horas}: el pilotito de espera de un televisor
sorbe \qty{0.5}{W} --- pero durante las $720$ \omterm{ex:g2:measuring-time:clock}{horas} del mes:
$0.0005 \times 720 = \qty{0.36}{kWh}$. Un vampiro es inofensivo; una
casa con veinte sorbedores de un \omterm{def:g9:electric-power-energy:power}{vatio} cada uno paga $0.02 \times
720 \approx \qty{14}{kWh}$ al mes por nada. La \omterm{def:g9:electric-power-energy:power}{potencia} por el
tiempo no perdona ningún término: el segundo factor muerde cuando el
primero se esconde.
\end{example}

\begin{remark}[La potencia en el mundo ancho]\label{rem:g9:electric-power-energy:wider}
El \omterm{def:g9:electric-power-energy:power}{vatio}, nacido eléctrico aquí, mide todos los ritmos de
\omterm{def:g5:everyday-energy:energy}{energía}. Tu cuerpo en reposo ronronea cerca de \qty{100}{W} --- una
\omterm{def:g3:first-electric-circuit:bulb}{bombilla} antepasada de calorcito (acuérdate del bochorno de un aula
llena: treinta alumnos, tres kilovatios de biología). Un ciclista
sostiene \qty{200}{W}; el \omterm{ex:g6:circuits-and-safety:motor}{motor} de un coche desata decenas de
kilovatios; la central eléctrica del pueblo, cientos de millones de
\omterm{def:g9:electric-power-energy:power}{vatios}; la \omterm{def:g1:light-and-shadows:source}{luz} del Sol que cae sobre el pueblo, bastante más --- el
techo bajo el que trabajan todos los demás ritmos. El capítulo
siguiente sigue los \omterm{def:g9:electric-power-energy:power}{vatios} de la red hasta su fuente giratoria.
\end{remark}

\section{Problema: la auditoría energética de la casa}

\begin{problem}[{Problema de fin de semana --- la auditoría energética
familiar; placas, contadores y la factura del mes; encontrar las
palancas de verdad}]\label{pb:g9:electric-power-energy:1}
Armada con las dos fórmulas y la tarifa ($0.25$ por \unit{kWh}), la
familia audita un mes (30 días). El inventario: hervidor
\qty{2300}{W}, $6$ minutos diarios; televisor \qty{100}{W}, $4$
\omterm{ex:g2:measuring-time:clock}{horas} diarias; frigorífico --- caso especial más abajo; diez
lámparas de \qty{5}{W} cada una, $5$ \omterm{ex:g2:measuring-time:clock}{horas} diarias; lavadora
\qty{2}{kW}, un ciclo caliente de $1$ hora un día sí y otro no;
calentador de agua \qty{2.5}{kW}, $2$ \omterm{ex:g2:measuring-time:clock}{horas} diarias; sorbedores
en espera que suman \qty{10}{W}, siempre encendidos.

\medskip\noindent\textbf{Parte I --- Máquina por máquina.}
\begin{enumerate}
  \item El mes del hervidor, en \unit{kWh}.
  \item Los meses del televisor y de las diez lámparas.
  \item Los meses de la lavadora ($15$ ciclos) y del calentador de
        agua.
  \item El mes de los sorbedores en espera --- el segundo factor en
        acción.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El frigorífico y la suma.}
El compresor del frigorífico funciona a \qty{120}{W}, pero solo un
tercio del tiempo, arrancando y parando día y noche.
\begin{enumerate}[resume]
  \item Su \omterm{def:g9:electric-power-energy:power}{potencia} media efectiva y su mes en \unit{kWh}.
  \item Suma las siete líneas de la auditoría. ¿Qué tres líneas
        dominan?
  \item La factura del mes con la tarifa --- y la parte que le
        corresponde en dinero al calentador de agua.
  \item El contador marcaba $2150$ \omterm{def:g6:measurement-in-science:measuring}{unidades} al empezar el mes.
        Predice su lectura de fin de mes.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Las \omterm{def:g3:levers-and-scales:lever}{palancas}.}
\begin{enumerate}[resume]
  \item Propuesta uno: duchas más cortas --- el calentador baja a
        \qty{1.5}{h} diarias. ¿Ahorro mensual, en
        \omterm{def:g5:everyday-energy:energy}{energía} y en dinero?
  \item Propuesta dos: una campaña pública insta a desenchufar los
        cargadores de móvil (\qty{0.2}{W} cada uno cuando están
        parados). Audita el mes parado de un cargador y compáralo con
        la propuesta uno; ¿qué enseña el contraste sobre las
        auditorías frente a los eslóganes?
  \item Propuesta tres: la abuela sugiere la vieja costumbre de
        llenar el hervidor solo con lo que hace falta --- la mitad de
        agua, la mitad de \omterm{def:g5:everyday-energy:energy}{energía}. ¿Nueva línea del hervidor y
        ahorro del mes?
  \item Escribe las conclusiones de la auditoría: tres frases ---
        dónde viven de verdad los \omterm{def:g9:electric-power-energy:kwh}{kilovatios-hora}, qué factor de la
        fórmula enseña el caso del modo de espera y qué único cambio
        doméstico sale más a cuenta.
\end{enumerate}
\end{problem}
