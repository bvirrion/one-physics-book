\chapter{Formas de la energía y conversiones}\label{ch:g9:energy-conversions}

La \omterm{def:g5:everyday-energy:energy}{energía} ha recorrido este libro como un río por debajo de una
ciudad --- entrevista en todas las esquinas, sin haberse cartografiado
nunca entera. Ha llegado la hora. Todas las formas reciben su nombre,
una de ellas recibe la segunda gran fórmula de este año, y sobre todas
ellas preside la ley de contabilidad más poderosa de la ciencia: el
total no cambia nunca. Jamás.

\section{Las formas, reunidas}

\begin{definition}[Las formas de la energía]\label{def:g9:energy-conversions:forms}
Las formas de la \omterm{def:g5:everyday-energy:energy}{energía}, presentadas formalmente: la
\emph{\omterm{prop:g9:kinetic-energy-safety:formula}{energía cinética}} --- la parte del movimiento, $\frac12 m
v^2$; la \emph{energía potencial}\index{energía potencial} ---
almacenada por la posición o la disposición (lo levantado, lo estirado,
lo comprimido); la \emph{\omterm{def:g5:everyday-energy:energy}{energía} térmica} --- el meneo
molecular desordenado que mide la \omterm{def:g3:temperature-thermometers:temperature}{temperatura}; la
\emph{\omterm{def:g5:everyday-energy:energy}{energía} química} --- almacenada en las disposiciones de la
materia: comida, combustibles, \omterm{def:g3:first-electric-circuit:battery}{pilas}; la \emph{\omterm{def:g5:everyday-energy:energy}{energía}
eléctrica} --- lo que entrega la corriente que desfila; la
\emph{\omterm{def:g5:everyday-energy:energy}{energía} radiante} --- transportada por la propia
\omterm{def:g1:light-and-shadows:source}{luz}, la forma de envío de la \omterm{def:g1:light-and-shadows:source}{luz} del Sol; y la
\emph{\omterm{def:g5:everyday-energy:energy}{energía} nuclear} --- encerrada en el corazón del átomo, la
reserva más profunda que han abierto los humanos.
\end{definition}

\begin{proposition}[Energía potencial gravitatoria]\label{prop:g9:energy-conversions:ep}
Levantar una \omterm{def:g3:measuring-mass:mass}{masa} $m$ (en \unit{kg}) una altura $h$ (en \unit{m})
contra el $g$ de la gravedad almacena en ella la \omterm{def:g9:energy-conversions:forms}{energía
potencial}
\[
E_p = m \times g \times h
\]
\omterm{def:g9:kinetic-energy-safety:joule}{julios} --- recuperables por completo a la vuelta, cuando baje. El
hacha levantada, el columpio echado hacia atrás, el lago de montaña:
todos tienen $m g h$ en su cuenta. (Es la fórmula que tomaron prestada
los carteles de seguridad vial --- ahora oficialmente tuya.)
\end{proposition}

\begin{example}[El gran trueque: $E_p$ por $E_k$]\label{ex:g9:energy-conversions:trade}
Suelta una pelota desde una altura $h$: mientras cae, su cuenta se
transfiere, \omterm{def:g9:kinetic-energy-safety:joule}{julio} a \omterm{def:g9:kinetic-energy-safety:joule}{julio}, de potencial a cinética. En
el suelo, todo el $m g h$ se ha convertido en $\frac12 m v^2$ ---
iguálalos y la \omterm{def:g3:measuring-mass:mass}{masa} se cancela:
\[
v = \sqrt{2 g h},
\]
con la raíz cuadrada de este año ganándose el sueldo. Desde
\qty{20}{m}: $v = \sqrt{2 \times 9.8 \times 20} \approx \qty{20}{m/s}$
--- setenta kilómetros por hora, sea cual sea la \omterm{def:g3:measuring-mass:mass}{masa}. El péndulo
juega al trueque en los dos sentidos para siempre (menos un susurro que
se lleva el \omterm{def:g2:air-around-us:air}{aire}); el patinador del medio tubo, igual: altura a
\omterm{def:g7:motion-average-speed:formula}{velocidad} y a altura, con las dos fórmulas pasándose una misma suma
de un lado a otro.
\end{example}

\section{La gran ley}

\begin{proposition}[Conservación de la energía]\label{prop:g9:energy-conversions:conservation}
En todos los procesos examinados hasta hoy --- choques y química,
máquinas y tormentas, \omterm{def:g4:solar-system:star}{estrellas} y células --- la
\omterm{def:g5:everyday-energy:energy}{energía} cambia de forma y de dueño, pero el \emph{total} se
conserva exactamente: no se crea ninguna, no se destruye ninguna, todos
los \omterm{def:g9:kinetic-energy-safety:joule}{julios} rinden cuentas. Es la ley de contabilidad más profunda
de la física: jubiló a las máquinas eternas de los capítulos de la
infancia, audita todas las cadenas de este libro y en tres siglos
ningún experimento la ha pillado en un error. (Por qué cumple la
naturaleza esta ley con tanta fidelidad es una pregunta profunda con
una respuesta preciosa --- una de las joyas de los años de
universidad.)
\end{proposition}

\begin{method}[Auditar cualquier proceso]\label{met:g9:energy-conversions:audit}
La contabilidad universal del físico:
\begin{enumerate}
  \item pon una valla alrededor del sistema --- lo que queda dentro
        de la auditoría;
  \item enumera las cuentas de \omterm{def:g5:everyday-energy:energy}{energía} de antes: cada forma,
        cada reserva;
  \item enumera las de después --- incluyendo siempre la calderilla
        térmica: el calorcito del rozamiento, el susurro del
        \omterm{def:g3:sound-around-us:sound}{sonido};
  \item cuadra los libros: los totales tienen que coincidir. Un
        déficit significa que se quedó una forma sin contar --- busca
        otra vez la fuga; suele estar templada.
\end{enumerate}
\end{method}

\begin{example}[Auditorías resueltas]\label{ex:g9:energy-conversions:audits}
El coche que frena: cuenta cinética vaciada, cuenta térmica de los
discos abonada por completo --- jubilación, no destrucción. La pelota
que bota y se va apagando: cada bote desnata algo de cinética y lo pasa
a calorcito de la pelota y del suelo; la respuesta de la infancia, ahora
auditable. Tu desayuno subiendo una cuesta en bici: reserva química
abajo, cuenta potencial arriba ($m g h$ tuyo y de la bicicleta), parte
térmica radiada --- los músculos van calientes. Todos los misterios de
los primeros capítulos de la \omterm{def:g5:everyday-energy:energy}{energía} se cierran bajo las mismas tres
columnas.
\end{example}

\section{El rendimiento}

\begin{definition}[Rendimiento]\label{def:g9:energy-conversions:efficiency}
El \emph{rendimiento}\index{rendimiento} de un convertidor es la
fracción de la \omterm{def:g5:everyday-energy:energy}{energía} de entrada que se entrega en la forma
deseada:
\[
\text{rendimiento} =
\frac{\text{energía útil de salida}}{\text{energía total de entrada}},
\]
un número entre $0$ y $1$ (o su porcentaje). El resto no se destruye
--- la conservación lo prohíbe ---, sino que se escapa en formas no
deseadas, casi siempre calorcito.
\end{definition}

\begin{example}[Boletines de notas]\label{ex:g9:energy-conversions:reportcards}
La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} antepasada incandescente: un cinco por ciento de
\omterm{def:g1:light-and-shadows:source}{luz} y un noventa y cinco de calorcito --- una estufa con una
afición. Su sucesora moderna: más de un treinta por ciento, y el resto
todavía calorcito. El \omterm{ex:g6:circuits-and-safety:motor}{motor} de un coche: alrededor de un tercio de
movimiento útil y dos tercios por el radiador y el escape. Un
\omterm{ex:g6:circuits-and-safety:motor}{motor} eléctrico: por encima del noventa. Los grandes
\omterm{def:g9:alternator:alternator}{alternadores}: casi el noventa y nueve --- los mejores convertidores
de la civilización. Ningún convertidor honrado llega a cien: algo de
fuga a calorcito es la comisión universal de la naturaleza.
\end{example}

\begin{remark}[Qué significa de verdad ``ahorrar energía'']\label{rem:g9:energy-conversions:saving}
Si la conservación custodia todos los \omterm{def:g9:kinetic-energy-safety:joule}{julios}, ¿por qué tiene nadie
que ``ahorrar \omterm{def:g5:everyday-energy:energy}{energía}''? Porque la ley conserva la \emph{cantidad},
no la \emph{utilidad}. Las reservas concentradas --- combustible, carga,
altura --- se pueden gastar convirtiéndolas en calorcito repartido con
fineza por el mundo, y el calorcito repartido, aunque sobreviva cada uno
de sus \omterm{def:g9:kinetic-energy-safety:joule}{julios}, ya no puede mover hervidores ni trenes: el río bajó
la cuesta. El problema energético de la civilización no es una escasez
de \omterm{def:g9:kinetic-energy-safety:joule}{julios} --- el Sol nos ducha con miles de veces lo que
necesitamos ---, sino la administración de los \emph{útiles}. Esa idea,
hecha precisa, es una de las historias más célebres de la física;
espera en los años de universidad bajo su famoso nombre.
\end{remark}

\section{Problema: el encargo del parque de atracciones}

\begin{problem}[{Problema de fin de semana --- el parque de atracciones
contrata a un consultor de física; montañas rusas, torres y el libro de
cuentas que no puede mentir}]\label{pb:g9:energy-conversions:1}
Las atracciones nuevas del parque tienen que pasar tu auditoría de
\omterm{def:g5:everyday-energy:energy}{energía} antes del día de la inauguración. Toma $g =
\qty{9.8}{N/kg}$; desprecia el rozamiento hasta que se niegue a que lo
desprecien.

\medskip\noindent\textbf{Parte I --- La Gran Caída.}
Un tren de montaña rusa de \qty{2000}{kg} se sube con cabrestante a una
cima de \qty{45}{m} y se suelta desde el reposo.
\begin{enumerate}
  \item ¿El $E_p$ del tren en la cima?
  \item ¿Su \omterm{def:g7:motion-average-speed:formula}{velocidad} ideal en el fondo de la caída, según la
        fórmula en la que se cancela la \omterm{def:g3:measuring-mass:mass}{masa} --- en \unit{m/s} y en
        \unit{km/h}?
  \item Los diseñadores añaden una segunda colina de \qty{50}{m}
        justo después de la caída. Vétala con el libro de cuentas ---
        y di cuál es la segunda colina más alta que podría coronar el
        tren ideal.
  \item La \omterm{def:g7:motion-average-speed:formula}{velocidad} medida en el fondo es de \qty{28}{m/s}, no
        la ideal. Calcula las dos \omterm{prop:g9:kinetic-energy-safety:formula}{energías cinéticas} y cárgale la
        diferencia a la cuenta que corresponde.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El cabrestante y la factura.}
\begin{enumerate}[resume]
  \item El cabrestante sube el tren a su cima en \qty{60}{s}. Con un
        \omterm{def:g9:energy-conversions:efficiency}{rendimiento} perfecto, ¿su potencia?
  \item El cabrestante real absorbe \qty{20}{kW}. ¿Su
        \omterm{def:g9:energy-conversions:efficiency}{rendimiento}?
  \item Cada vuelta sale cada cuatro minutos, diez \omterm{ex:g2:measuring-time:clock}{horas} al
        día. ¿La \omterm{def:g5:everyday-energy:energy}{energía} diaria del cabrestante en
        \omterm{def:g9:electric-power-energy:kwh}{kilovatios-hora} (con el cabrestante real; cuenta las
        subidas) y su coste a $0.25$ por \omterm{def:g6:measurement-in-science:measuring}{unidad}?
  \item Un diseñador propone recuperar la \omterm{def:g5:everyday-energy:energy}{energía} de llegada del
        tren con un freno-generador que alimente a la red --- ``¡la
        atracción se paga sola!''. Audita el sueño: ¿qué fracción
        podría volver de forma realista y qué ley prohíbe el círculo
        completo?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La torre de caída libre y el
informe.}
La torre de \qty{60}{m} deja caer una góndola de \qty{500}{kg} sobre
unos frenos magnéticos situados a \qty{10}{m}.
\begin{enumerate}[resume]
  \item ¿\omterm{def:g7:motion-average-speed:formula}{Velocidad} de entrada en la zona de frenado (una caída
        ideal de \qty{50}{m})?
  \item ¿\omterm{def:g5:everyday-energy:energy}{Energía} que tienen que jubilar los frenos, y adónde la
        manda la inducción? (Los frenos son primos del capítulo del
        \omterm{def:g9:alternator:alternator}{alternador} --- remolinos de corriente inducida que
        calientan el metal.)
  \item Los pasajeros gritan que iban ``sin \omterm{def:g4:weight-and-mass:weight}{peso}'' durante la caída.
        A partir de la idea de los compañeros de caída del capítulo
        de la \omterm{def:g9:gravitation:gravitation}{gravitación}, dictamina sobre la afirmación en una
        frase.
  \item Firma el informe del encargo: tres frases --- el trueque del
        que vive la montaña rusa, la ley que obedecieron todas las
        atracciones y la cuenta que en silencio les cobró impuestos a
        todas.
\end{enumerate}
\end{problem}
