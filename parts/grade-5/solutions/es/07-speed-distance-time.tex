\section*{Capítulo \ref{ch:g5:speed-distance-time} --- La velocidad: distancia y tiempo}

\begin{solution}{exo:g5:speed-distance-time:1}
Misma distancia --- gana el que emplea menos tiempo; mismo tiempo ---
gana el que recorre más distancia. ``He corrido más lejos'' no fija ni
una cosa ni la otra: con tiempos distintos, más distancia no demuestra
nada sobre ser rápido.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:2}
Bo --- misma distancia (una piscina), menos tiempo ($25 < 30$
segundos): la regla uno.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:3}
La moto --- mismo tiempo (una hora), más distancia ($45 > 20$
kilómetros): la regla dos.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:4}
Que en cada hora recorre \qty{200}{km}. En $2$ \omterm{ex:g2:measuring-time:clock}{horas}:
$200 \times 2 = \qty{400}{km}$. En media hora: la mitad de $200$, o
sea \qty{100}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:5}
El caracol ($5$ \omterm{def:g2:measuring-length:units}{metros} en cada hora), el caminante ($5$
kilómetros), el autobús urbano ($30$) y la paloma ($80$ kilómetros en
cada hora).
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:6}
En una hora caben $30$ raciones de $2$ minutos:
$30 \times 100 = \qty{3000}{m}$ --- una \omterm{def:g5:speed-distance-time:speed}{velocidad} de $3$ kilómetros
en cada hora.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:7}
El coche: $80 \times 3 = \qty{240}{km}$. El ciclista:
$15 \times 3 = \qty{45}{km}$.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:8}
La recta del viajero más rápido sube con más pendiente. Al cabo de $2$
\omterm{ex:g2:measuring-time:clock}{horas}, la recta del ciclista está en \qty{30}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:9}
Hora a hora: $30$, $60$, $90$ --- al cabo de $3$ \omterm{ex:g2:measuring-time:clock}{horas} el
transbordador ha recorrido exactamente \qty{90}{km}: la travesía dura
$3$ \omterm{ex:g2:measuring-time:clock}{horas}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:10}
Cada hora Mara recorta \qty{20}{km} y Jon \qty{10}{km}: el hueco
encoge \qty{30}{km} en cada hora. Partiendo de \qty{60}{km}, llega a
cero al cabo de $2$ \omterm{ex:g2:measuring-time:clock}{horas} --- se encuentran a \qty{40}{km} del pueblo
de Mara.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:11}
Después de su único minuto corriendo, la liebre está en el
\qty{60}{m} --- la mitad --- y se queda dormida. La tortuga necesita
$120 \div 2 = 60$ minutos y llega en el instante $60$. La liebre se
despierta en el instante $1 + 70 = 71$, corre sus últimos \qty{60}{m}
en un minuto y llega en el instante $72$. La tortuga gana por $12$
minutos --- la constancia vence a la \omterm{def:g5:speed-distance-time:speed}{velocidad}, ahora con aritmética.
\end{solution}
