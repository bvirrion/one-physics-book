\section*{Chapitre \ref{ch:g5:speed-distance-time} --- La vitesse~: distance et durée}

\begin{solution}{exo:g5:speed-distance-time:1}
Même distance --- le moins de temps gagne~; même \omterm{def:g2:measuring-time:duration}{durée} --- la plus
grande distance gagne. «~J'ai couru plus loin~» ne fixe ni l'une ni
l'autre~: avec des \omterm{def:g2:measuring-time:duration}{durées} inégales, une plus grande distance ne
prouve rien sur la rapidité.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:2}
Bo --- même distance (un bassin), moins de temps ($25 < 30$
secondes)~: règle première.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:3}
Le scooter --- même \omterm{def:g2:measuring-time:duration}{durée} (une heure), plus de distance ($45 > 20$
kilomètres)~: règle deuxième.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:4}
À chaque heure, il parcourt \qty{200}{km}. En $2$ \omterm{ex:g2:measuring-time:clock}{heures}~:
$200 \times 2 = \qty{400}{km}$. En une demi-heure~: la moitié de
$200$, soit \qty{100}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:5}
L'escargot ($5$ \omterm{def:g2:measuring-length:units}{mètres} par heure), le marcheur ($5$ kilomètres), le
bus de ville ($30$), le pigeon ($80$ kilomètres par heure).
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:6}
Une heure contient $30$ parts de $2$ minutes~:
$30 \times 100 = \qty{3000}{m}$ --- une \omterm{def:g5:speed-distance-time:speed}{vitesse} de $3$ kilomètres par
heure.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:7}
La voiture~: $80 \times 3 = \qty{240}{km}$. Le cycliste~:
$15 \times 3 = \qty{45}{km}$.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:8}
La droite du voyageur le plus rapide monte plus fort. Après $2$
\omterm{ex:g2:measuring-time:clock}{heures}, la droite du cycliste se trouve à \qty{30}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:9}
Heure après heure~: $30$, $60$, $90$ --- au bout de $3$ \omterm{ex:g2:measuring-time:clock}{heures}, le
ferry a parcouru exactement \qty{90}{km}~: la traversée dure $3$
\omterm{ex:g2:measuring-time:clock}{heures}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:10}
Chaque heure, Mara comble \qty{20}{km} et Jon \qty{10}{km}~: l'écart
diminue de \qty{30}{km} par heure. Parti de \qty{60}{km}, il atteint
zéro au bout de $2$ \omterm{ex:g2:measuring-time:clock}{heures} --- ils se rencontrent à \qty{40}{km} de
la ville de Mara.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:11}
Après sa minute de course, le lièvre se trouve à \qty{60}{m} --- à
mi-parcours --- et s'endort. La tortue met $120 \div 2 = 60$ minutes
et arrive à l'instant $60$. Le lièvre se réveille à l'instant
$1 + 70 = 71$, court ses derniers \qty{60}{m} en une minute, et
arrive à l'instant $72$. La tortue gagne de $12$ minutes ---
la régularité bat la \omterm{def:g5:speed-distance-time:speed}{vitesse}, désormais avec de l'arithmétique.
\end{solution}
