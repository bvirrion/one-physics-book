\section*{Hoofdstuk \ref{ch:g5:speed-distance-time} --- Snelheid: afstand en tijd}

\begin{solution}{exo:g5:speed-distance-time:1}
Dezelfde afstand --- de minste tijd wint; dezelfde tijd --- de meeste
afstand wint. ``Ik heb verder gelopen'' zet geen van beide vast: bij
ongelijke tijden bewijst meer afstand niets over snel zijn.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:2}
Bo --- dezelfde afstand (één bad), minder tijd ($25 < 30$ \omterm{def:g2:measuring-time:second}{seconden}):
regel één.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:3}
De scooter --- dezelfde tijd (één uur), meer afstand ($45 > 20$
kilometer): regel twee.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:4}
In elk uur legt hij \qty{200}{km} af. In $2$ uur:
$200 \times 2 = \qty{400}{km}$. In een half uur: de helft van $200$,
dus \qty{100}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:5}
Slak ($5$ \omterm{def:g2:measuring-length:units}{meter} per uur), wandelaar ($5$ kilometer), stadsbus
($30$), duif ($80$ kilometer per uur).
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:6}
In een uur passen $30$ porties van $2$ minuten: $30 \times 100 =
\qty{3000}{m}$ --- een \omterm{def:g5:speed-distance-time:speed}{snelheid} van $3$ kilometer per uur.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:7}
De auto: $80 \times 3 = \qty{240}{km}$. De fietser:
$15 \times 3 = \qty{45}{km}$.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:8}
De lijn van de snelste reiziger klimt steiler. Na $2$ uur staat de
lijn van de fietser op \qty{30}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:9}
Uur na uur: $30$, $60$, $90$ --- na $3$ uur heeft de veerboot
precies \qty{90}{km} afgelegd: de overtocht duurt $3$ uur.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:10}
Elk uur haalt Mara \qty{20}{km} weg en Jon \qty{10}{km}: de opening
krimpt met \qty{30}{km} per uur. Vanaf \qty{60}{km} bereikt ze na $2$
uur nul --- ze ontmoeten elkaar op \qty{40}{km} van het dorp van
Mara.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:11}
Na zijn ene minuut rennen staat de haas op \qty{60}{m} --- halverwege
--- en valt hij in slaap. De schildpad heeft $120 \div 2 = 60$ minuten
nodig en finisht op \omterm{def:g2:measuring-time:duration}{tijdstip} $60$. De haas wordt wakker op \omterm{def:g2:measuring-time:duration}{tijdstip}
$1 + 70 = 71$, rent zijn laatste \qty{60}{m} in één minuut en finisht
op \omterm{def:g2:measuring-time:duration}{tijdstip} $72$. De schildpad wint met $12$ minuten --- gestaag
verslaat snel, nu met rekenwerk erbij.
\end{solution}
