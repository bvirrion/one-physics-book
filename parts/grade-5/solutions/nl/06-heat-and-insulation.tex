\section*{Hoofdstuk \ref{ch:g5:heat-and-insulation} --- Warmte en isolatie}

\begin{solution}{exo:g5:heat-and-insulation:1}
\omterm{def:g5:heat-and-insulation:heat}{Warmte} is \omterm{def:g5:everyday-energy:energy}{energie} die onderweg is van een heter lichaam naar een
kouder; vanzelf stroomt ze alleen bergafwaarts, heet naar koud, en ze
stopt zodra de twee temperaturen elkaar hebben bereikt.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:2}
``Doe de deur dicht, je laat onze \omterm{def:g5:heat-and-insulation:heat}{warmte} naar buiten!'' --- kou stroomt
niet naar binnen; de \omterm{def:g5:heat-and-insulation:heat}{warmte} van het huis stroomt naar de koudere straat.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:3}
Snelle rijstroken: de koperen pan, de stalen lepel. Trage rijstroken:
de wollen sjaal, stilstaande \omterm{def:g2:air-around-us:air}{lucht}, de houten lepel, het dekbed.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:4}
De bodem moet een snelle rijstrook zijn en de \omterm{def:g5:heat-and-insulation:heat}{warmte} van het fornuis
het eten in jagen; het handvat moet een trage rijstrook zijn en
diezelfde \omterm{def:g5:heat-and-insulation:heat}{warmte} uit je hand houden. Eén werktuig, twee talenten, elk
op zijn plaats.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:5}
Ze staan allebei op dezelfde \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}, maar metaal is een snelle
rijstrook: het rukt \omterm{def:g5:heat-and-insulation:heat}{warmte} in één ruk uit je warmere hand, en die ruk
is het ``ijskoude'' gevoel. De huid meet de snelheid van haar eigen
warmteverlies, niet de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:6}
Een eerlijke race verandert maar één ding --- de omwikkeling.
Identieke mokken, evenveel \omterm{def:g5:heat-and-insulation:heat}{warmte} bij de start en dezelfde kamer zorgen
ervoor dat de sjaal in zijn eentje elk verschil tussen de twee krommen
verklaart.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:7}
Na $15$ minuten: de ingepakte mok ongeveer \qty{41}{\celsius}, de
blote mok ongeveer \qty{30}{\celsius}. Beide krommen lopen naar de
\qty{20}{\celsius} van de kamer: de stroom loopt tot de
temperaturen elkaar ontmoeten, en isolatie vertraagt de reis alleen.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:8}
Opgeblazen veren sluiten een dikke deken stilstaande \omterm{def:g2:air-around-us:air}{lucht} op --- de
kampioen trage rijstrook --- rond het warme lijf van de vogel.
Dezelfde kaart: wollen kleren, donzen dekbedden, dubbele ramen (of
fleece, geïsoleerde muren).
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:9}
De thermoskan maakt geen \omterm{def:g5:heat-and-insulation:heat}{warmte} en geen kou: ze sluit de rijstroken
van de \omterm{def:g5:heat-and-insulation:heat}{warmte} --- leegte tussen haar twee wanden, glans om de gloed
terug te kaatsen. In januari vertraagt ze het ontsnappen van de
\omterm{def:g5:heat-and-insulation:heat}{warmte} van de chocolademelk; in juli vertraagt ze het binnensluipen
van de \omterm{def:g5:heat-and-insulation:heat}{warmte} van de zomer. Eén apparaat, beide kanten op, omdat ze
alleen de stroom beheert.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:10}
Op een hete dag probeert \omterm{def:g5:heat-and-insulation:heat}{warmte} van de warme \omterm{def:g2:air-around-us:air}{lucht} \emph{de} koude
bakken \emph{in} te stromen. De wol maakt van die stroom naar binnen
een straaltje --- dezelfde vertraging die ze rond een warm lichaam
uitvoert. Isolatie heeft geen lievelingen; de verkoopster, en niet de
klant, had het begrepen.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:11}
Het dubbele raam sluit tussen zijn ruiten een laag stilstaande
\omterm{def:g2:air-around-us:air}{lucht} op --- een trage rijstrook --- zodat de \omterm{def:g5:heat-and-insulation:heat}{warmte} van het huis
naar buiten sijpelt waar het enkele raam haar laat stromen. Minder
ontsnappende \omterm{def:g5:heat-and-insulation:heat}{warmte}, minder brandstof om haar te vervangen. Maar
isolatie vertraagt de bergafwaartse stroom alleen en stopt hem nooit:
er lekt altijd wat \omterm{def:g5:heat-and-insulation:heat}{warmte} de winter in, en de kachel moet die blijven
bijvullen.
\end{solution}
