\section*{Hoofdstuk \ref{ch:g5:series-parallel-circuits} --- Serie- en parallelschakelingen}

\begin{solution}{exo:g5:series-parallel-circuits:1}
Een \omterm{def:g5:series-parallel-circuits:parallel}{tak} is een van de afzonderlijke wegen waarin de kring zich
splitst; een \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} is een punt waar wegen splitsen of weer
samenkomen. Twee \omterm{def:g3:first-electric-circuit:bulb}{lampjes} \omterm{def:g5:series-parallel-circuits:parallel}{parallel} rijden op twee
lampjesdragende \omterm{def:g5:series-parallel-circuits:parallel}{takken}.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:2}
Onafhankelijkheid (een onderbreking legt alleen haar eigen
\omterm{def:g5:series-parallel-circuits:parallel}{tak} stil); volle felheid (elk \omterm{def:g3:first-electric-circuit:bulb}{lampje} schijnt alsof het alleen was,
waarvoor de \omterm{def:g3:first-electric-circuit:battery}{batterij} in ruil sneller leegraakt); commandoposten (een
takschakelaar heerst over zijn \omterm{def:g5:series-parallel-circuits:parallel}{tak}, een \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} vóór de
splitsing over alles).
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:3}
\omterm{def:g3:first-electric-circuit:bulb}{Lampje} 2 brandt ongestoord door --- zijn eigen weg is compleet.
\omterm{def:g4:series-circuits:series}{In serie} ging het uit, omdat de twee \omterm{def:g3:first-electric-circuit:bulb}{lampjes} daar één enkele weg
deelden, en elke onderbreking die voor allebei verbrak.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:4}
Paar B (\omterm{def:g5:series-parallel-circuits:parallel}{parallel}) schijnt per \omterm{def:g3:first-electric-circuit:bulb}{lampje} feller --- elk op volle felheid.
Paar B maakt zijn \omterm{def:g3:first-electric-circuit:battery}{batterij} ook sneller leeg: het voedt twee volle
\omterm{def:g5:series-parallel-circuits:parallel}{takken} tegelijk.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:5}
Allebei dicht: beide \omterm{def:g3:first-electric-circuit:bulb}{lampjes} schijnen. Hoofdschakelaar open: allebei
donker --- die heerst over de gemeenschappelijke weg. Hoofdschakelaar
dicht, takschakelaar open: \omterm{def:g3:first-electric-circuit:bulb}{lampje} 1 schijnt, \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2 donker.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:6}
\omterm{def:g4:series-circuits:series}{In serie} zou één kapot \omterm{def:g3:first-electric-circuit:bulb}{lampje} (of één uitgeschakelde lamp!) het
hele huis verduisteren, en zou alles zwakker worden naarmate er meer
lampen bijkwamen. Gemakken van parallelle bedrading: de
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar} van elke kamer heerst alleen over die kamer; een
doorgebrand keukenlampje laat de rest van het huis verlicht; elke lamp
brandt op volle felheid.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:7}
Vóór de splitsing --- op de gemeenschappelijke weg waar de stroom het
huis binnenkomt: de hoofdschakelaar in de meterkast. Hij sluit het hele
huis in één keer af, voor reparaties en noodgevallen.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:8}
\omterm{def:g5:series-parallel-circuits:parallel}{Parallel}: de plafondlamp uitzetten doodt het bedlampje niet, en het
\omterm{def:g3:first-electric-circuit:bulb}{lampje} uit het stopcontact halen verduistert het plafond niet --- elk
rijdt op zijn eigen \omterm{def:g5:series-parallel-circuits:parallel}{tak}.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:9}
Met onze gewone \omterm{ex:g3:first-electric-circuit:switch}{schakelaars} blijven de ontwerpen mislukken: een
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar} \omterm{def:g4:series-circuits:series}{in serie} met het \omterm{def:g3:first-electric-circuit:bulb}{lampje} kun je maar vanaf één kant
openen; twee gewone \omterm{ex:g3:first-electric-circuit:switch}{schakelaars} \omterm{def:g4:series-circuits:series}{in serie} eisen dat ze
\emph{allebei} dicht zijn (elk uiteinde kan het \omterm{def:g1:light-and-shadows:source}{licht} wel uitdoen
maar niet altijd aan); twee \omterm{def:g5:series-parallel-circuits:parallel}{parallel} eisen dat ze allebei open zijn
voor duisternis. Het gemak in de gang heeft een \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} nodig die
\emph{tussen twee wegen kiest} in plaats van er één te verbreken ---
de wisselschakelaar, die echte gangen in gelijke paren gebruiken.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:10}
Het gemak: de \omterm{def:g3:first-electric-circuit:bulb}{lampjes} rijden op parallelle \omterm{def:g5:series-parallel-circuits:parallel}{takken}, dus verduistert
een kapot \omterm{def:g3:first-electric-circuit:bulb}{lampje} alleen zichzelf. De waarschuwing: elk ontbrekend
\omterm{def:g3:first-electric-circuit:bulb}{lampje} haalt een \omterm{def:g5:series-parallel-circuits:parallel}{tak} weg, en de \omterm{def:g3:first-electric-circuit:battery}{batterij} (of de kleine voeding
van de sliert) duwt haar volle inspanning door de \omterm{def:g5:series-parallel-circuits:parallel}{takken} die
overblijven --- met te veel weg krijgen de overlevenden elk meer dan
waarvoor ze gebouwd zijn, worden ze heter en gaan ze jong kapot.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:11}
Op het eigen stuk weg van de takschakelaar: tussen de
takschakelaar en \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2 (voordat hun \omterm{def:g5:series-parallel-circuits:parallel}{tak} zich weer bij de rest
voegt). Daar brandt het precies wanneer die \omterm{def:g5:series-parallel-circuits:parallel}{tak} loopt --- en het
staat \omterm{def:g4:series-circuits:series}{in serie} met \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2, want ze rijden op dezelfde
\omterm{def:g5:series-parallel-circuits:parallel}{tak}.
\end{solution}
