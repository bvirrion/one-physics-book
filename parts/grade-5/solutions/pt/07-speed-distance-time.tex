\section*{Capítulo \ref{ch:g5:speed-distance-time} --- Velocidade: distância e tempo}

\begin{solution}{exo:g5:speed-distance-time:1}
Mesma distância --- ganha quem gasta menos tempo; mesmo tempo ---
ganha quem percorre mais distância. ``Eu corri mais longe'' não fixa
nem uma coisa nem outra: com tempos diferentes, mais distância não
prova nada sobre ser rápido.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:2}
Bo --- mesma distância (uma piscina), menos tempo ($25 < 30$
segundos): regra um.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:3}
A motoneta --- mesmo tempo (uma hora), mais distância ($45 > 20$
quilômetros): regra dois.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:4}
Em cada hora ele percorre \qty{200}{km}. Em $2$ \omterm{ex:g2:measuring-time:clock}{horas}:
$200 \times 2 = \qty{400}{km}$. Em meia hora: metade de $200$, ou
seja, \qty{100}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:5}
Caracol ($5$ \omterm{def:g2:measuring-length:units}{metros} em cada hora), pessoa caminhando ($5$
quilômetros), ônibus urbano ($30$), pombo ($80$ quilômetros em cada
hora).
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:6}
Uma hora guarda $30$ porções de $2$ minutos: $30 \times 100 =
\qty{3000}{m}$ --- uma \omterm{def:g5:speed-distance-time:speed}{velocidade} de $3$ quilômetros em cada hora.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:7}
O carro: $80 \times 3 = \qty{240}{km}$. O ciclista:
$15 \times 3 = \qty{45}{km}$.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:8}
A reta do viajante mais rápido sobe com mais inclinação. Depois de
$2$ \omterm{ex:g2:measuring-time:clock}{horas} a reta do ciclista está em \qty{30}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:9}
Hora a hora: $30$, $60$, $90$ --- depois de $3$ \omterm{ex:g2:measuring-time:clock}{horas} a balsa
percorreu exatamente \qty{90}{km}: a travessia leva $3$ \omterm{ex:g2:measuring-time:clock}{horas}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:10}
A cada hora Mara fecha \qty{20}{km} e Jon fecha \qty{10}{km}: o vão
encolhe \qty{30}{km} em cada hora. Partindo de \qty{60}{km}, ele
chega a zero depois de $2$ \omterm{ex:g2:measuring-time:clock}{horas} --- eles se encontram a
\qty{40}{km} da cidade de Mara.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:11}
Depois do seu único minuto de corrida, a lebre está em \qty{60}{m}
--- na metade do caminho --- e adormece. A tartaruga precisa de
$120 \div 2 = 60$ minutos e chega no instante $60$. A lebre acorda no
instante $1 + 70 = 71$, corre os últimos \qty{60}{m} em um minuto e
chega no instante $72$. A tartaruga ganha por $12$ minutos ---
constância vence \omterm{def:g5:speed-distance-time:speed}{velocidade}, agora com aritmética.
\end{solution}
