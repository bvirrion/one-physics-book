\section*{अध्याय \ref{ch:g5:air-pressure-weather} --- वायुदाब और मौसम}

\begin{solution}{exo:g5:air-pressure-weather:1}
वही गेंद पहले कसी हुई और फिर पिचकी हुई तौली जाती है: उसमें से सिर्फ़
\omterm{def:g2:air-around-us:air}{हवा} गई, और \omterm{def:g3:levers-and-scales:scale}{तुला} कम बताती है --- यह फ़र्क़ भागी हुई \omterm{def:g2:air-around-us:air}{हवा} का \omterm{def:g4:weight-and-mass:weight}{भार} है। \omterm{def:g2:air-around-us:air}{हवा}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} वाला पदार्थ है।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:2}
\omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} की तली की दबी हुई \omterm{def:g2:air-around-us:air}{हवा} का वह धक्का, जो वह हर छूती हुई सतह पर
लगाती है। वह हमारे ऊपर फैले \omterm{def:g2:air-around-us:air}{हवा} के गहरे महासागर के \omterm{def:g4:weight-and-mass:weight}{भार} से आता है, और
हर दिशा में धकेलता है --- नीचे, बग़ल में, और ऊपर भी।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:3}
क्योंकि धक्के बराबरी में आते हैं: तुम्हारी हथेली की दूसरी ओर की ---
और तुम्हारे शरीर के भीतर की --- \omterm{def:g2:air-around-us:air}{हवा} ठीक उतने ही ज़ोर से वापस धकेलती
है। अपनी ताक़त सिर्फ़ \emph{बेमेल} धक्का दिखाता है, जैसे कार्ड पर या
पिचकती बोतल पर।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:4}
पानी का \omterm{def:g4:weight-and-mass:weight}{भार} कार्ड को नीचे धकेलता है; \omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} कार्ड को ऊपर धकेलता है।
\omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} जीत जाता है --- कार्ड टिका रहता है और पानी भीतर रुका रहता है।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:5}
तुम नली के भीतर से \emph{\omterm{def:g2:air-around-us:air}{हवा}} हटाते हो। फिर गिलास के रस पर नीचे दबाता
\omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} रस को ख़ाली हुई नली में ऊपर धकेल देता है: उठाने का काम आकाश
करता है।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:6}
कप को सपाट दबाने से उसके पीछे की \omterm{def:g2:air-around-us:air}{हवा} निकल जाती है। नीचे से वापस
धकेलने वाला कुछ न बचने पर \omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} का धक्का कप को टाइलों से चिपकाए
रखता है।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:7}
वह कमज़ोर पड़ जाता है --- ऊपर दबाने के लिए \omterm{def:g2:air-around-us:air}{हवा} का महासागर कम बचता है।
सुराग़: तेज़ चढ़ाई पर कानों का चटकना, पतली \omterm{def:g2:air-around-us:air}{हवा} में हाँफना (और बहुत
ठंडी उबलती चाय)।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:8}
\omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} का धक्का --- यानी \omterm{def:g5:air-pressure-weather:pressure}{वायुदाब}। तेज़ी से गिरना रास्ते में तूफ़ान
की चेतावनी है: सैर-सपाटा भीतर ले आओ।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:9}
चोटी पर बंद की गई बोतल के भीतर पतली पहाड़ी \omterm{def:g2:air-around-us:air}{हवा} क़ैद रह गई; घाटी में
बाहर का धक्का भीतर क़ैद धक्के से ताक़तवर हो गया, और बोतल मुड़ गई।
उलटी दिशा में, घाटी में बंद बोतल के भीतर \emph{तगड़ी} \omterm{def:g2:air-around-us:air}{हवा} रहती है
जबकि ऊँचाई के साथ बाहरी धक्का कमज़ोर पड़ता जाता है: बोतल तनकर फूल जाती
है --- और बंद चिप्स का पैकेट भी हर पहाड़ी सफ़र में यही करता है।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:10}
\omterm{def:g2:air-around-us:air}{हवा} लगातार भारी ऊँचे दाब वाले इलाक़ों से हल्के नीचे दाब वाले इलाक़ों की
ओर उँडेलती है, और वही उँडेलना \omterm{def:g2:air-around-us:wind}{पवन} है। यानी पुरानी परिभाषा अपने आप पूरी
हो जाती है: \omterm{def:g2:air-around-us:wind}{पवन} यानी चलती हुई \omterm{def:g2:air-around-us:air}{हवा} --- और उसे चलाते हैं \omterm{def:g5:air-pressure-weather:pressure}{वायुदाब} के
अंतर।
\end{solution}

\begin{solution}{exo:g5:air-pressure-weather:11}
ऊँचाई के साथ पानी पर \omterm{def:g5:air-pressure-weather:pressure}{वायुमंडल} का धक्का कमज़ोर पड़ता है, इसलिए उबलना
आसान हो जाता है: \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} \qty{100}{\celsius} से काफ़ी नीचे सरक जाता
है। और पठार के नियम से उबलता पानी अपने \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} से आगे कभी चढ़ ही
नहीं सकता --- ज़्यादा आँच सिर्फ़ ज़्यादा भाप बनाती है। इसलिए चोटी का
पानी ठंडा उबलता है \emph{और वहीं अटका रह जाता है}: चाय फीकी, सिवइयाँ
कच्ची, और भौतिकी संतुष्ट।
\end{solution}
