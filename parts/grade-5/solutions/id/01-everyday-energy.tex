\section*{Bab \ref{ch:g5:everyday-energy} --- Energi dalam Kehidupan Sehari-hari}

\begin{solution}{exo:g5:everyday-energy:1}
\omterm{def:g5:everyday-energy:energy}{Energi} adalah apa yang diperlukan untuk membuat segalanya terjadi.
Tanpanya, tidak ada yang dapat bergerak, terangkat, bersinar,
menghangat, atau berbunyi --- pilih empat saja.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:2}
Pegas yang terputar; makanan si pesepeda; kayu yang terbakar; \omterm{def:g2:air-around-us:air}{udara}
yang bergerak sebagai \omterm{def:g2:air-around-us:wind}{angin}; \omterm{def:g3:first-electric-circuit:battery}{baterainya}.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:3}
Gerak (bola yang menggelinding), \omterm{def:g1:light-and-shadows:source}{cahaya} (lampu), panas (air mandi yang
hangat), listrik (kawat menuju pemanggang roti), tersimpan (sarapan,
sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai}, pegas yang terputar).
\end{solution}

\begin{solution}{exo:g5:everyday-energy:4}
\omterm{def:g5:everyday-energy:energy}{Energi} listrik dari dinding (yang lebih jauh ke belakang diberi makan
oleh penyimpan sebuah pembangkit listrik) $\to$ kawat berpijar di
dalam pemanggang roti (pengubah) $\to$ panas menuju rotinya --- dengan
sedikit \omterm{def:g1:light-and-shadows:source}{cahaya}, pijar jingganya, di sampingnya.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:5}
Air yang tinggi (penyimpan) $\to$ air yang jatuh memutar turbin
(pengubah) $\to$ \omterm{def:g5:everyday-energy:energy}{energi} listrik di dalam kawat $\to$ lampunya
(pengubah kedua) $\to$ \omterm{def:g1:light-and-shadows:source}{cahaya}, ditambah sedikit kehangatan.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:6}
Penyimpan \omterm{def:g3:first-electric-circuit:battery}{baterainya} sudah habis. \omterm{def:g5:everyday-energy:energy}{Energinya} tidak dimusnahkan: \omterm{def:g5:everyday-energy:energy}{energi}
itu dibelanjakan sepanjang rantai harian --- menyalakan layar, membuat
\omterm{def:g3:sound-around-us:sound}{bunyi} --- dan berakhir tersebar sebagai kehangatan lembut di dalam
teleponnya dan di dalam ruangan.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:7}
Itulah kebocorannya: setiap pengubah meneruskan sebagian \omterm{def:g5:everyday-energy:energy}{energinya}
menjadi kehangatan di samping kerja bergunanya. Cip laptop mengubah
\omterm{def:g5:everyday-energy:energy}{energi} listrik, dan kehangatan di bawahnya adalah saluran samping
rantai itu yang tidak terhindarkan.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:8}
Layarnya didorong \omterm{def:g2:air-around-us:wind}{angin}; \omterm{def:g2:air-around-us:wind}{angin} adalah \omterm{def:g2:air-around-us:air}{udara} yang teraduk menjadi gerak
karena Matahari menghangatkan daratan dan laut secara tidak merata;
jadi \omterm{def:g5:everyday-energy:energy}{energi} gerak perahunya tiba, dua mata rantai ke belakang, dari
sinar matahari.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:9}
Bahwa \omterm{def:g5:everyday-energy:energy}{energi} tidak pernah disulap dari \omterm{def:g2:air-around-us:air}{udara} kosong --- apa pun yang
berjalan sedang diberi makan. Pertanyaan pertama: ``Apa penyimpannya,
dan dari mana lampunya menariknya?'' Kalau penemunya tidak punya
jawaban, lampu itu masuk golongan roda abadi: ke dalam laci kegagalan
yang termasyhur.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:10}
\omterm{def:g5:everyday-energy:energy}{Energinya} diteruskan, bukan dimusnahkan: setiap pantulan membelanjakan
sebagian \omterm{def:g5:everyday-energy:energy}{energi} gerak bolanya untuk memipihkan bola serta menggesek
\omterm{def:g2:air-around-us:air}{udara} dan lantai, dan semua pembayaran itu berakhir sebagai bisikan
kehangatan di dalam bola, lantai, dan \omterm{def:g2:air-around-us:air}{udara}. Terlalu tersebar untuk
dikumpulkan, tetapi tidak lenyap.
\end{solution}

\begin{solution}{exo:g5:everyday-energy:11}
Di antara Matahari dan api kayu berdiri para perantara: pohon, yang
bertahun-tahun menangkap sinar matahari lalu menyimpannya sebagai
kayu, dan api yang akhirnya membongkar simpanan itu. Panel di atap
mengambil \omterm{def:g5:everyday-energy:energy}{energi} sinar mataharinya langsung --- tanpa pohon, tanpa
tahunan, tanpa api.
\end{solution}
