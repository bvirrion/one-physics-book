\section*{Bab \ref{ch:g5:heat-and-insulation} --- Kalor dan Isolasi}

\begin{solution}{exo:g5:heat-and-insulation:1}
\omterm{def:g5:heat-and-insulation:heat}{Kalor} adalah \omterm{def:g5:everyday-energy:energy}{energi} yang sedang berpindah dari benda yang lebih panas
ke benda yang lebih dingin; dengan sendirinya ia hanya mengalir
menurun, dari panas ke dingin, dan berhenti ketika kedua \omterm{def:g3:temperature-thermometers:temperature}{suhunya} sudah
bertemu.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:2}
``Tutup pintunya, kamu membiarkan \omterm{def:g5:heat-and-insulation:heat}{kalor} kita keluar!'' --- dingin
tidak mengalir masuk; \omterm{def:g5:heat-and-insulation:heat}{kalor} rumahnyalah yang mengalir keluar ke jalan
yang lebih dingin.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:3}
Jalur cepat: panci tembaga dan sendok baja. Jalur lambat: syal wol,
\omterm{def:g2:air-around-us:air}{udara} yang diam, sendok kayu, dan selimut bulu.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:4}
Dasarnya harus jalur cepat, yang menyerbukan \omterm{def:g5:heat-and-insulation:heat}{kalor} kompor ke dalam
makanannya; gagangnya harus jalur lambat, yang menjauhkan \omterm{def:g5:heat-and-insulation:heat}{kalor} yang
sama dari tanganmu. Satu alat, dua bakat, masing-masing pada
tempatnya.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:5}
Keduanya berada pada \omterm{def:g3:temperature-thermometers:temperature}{suhu} yang sama, tetapi logam adalah jalur cepat:
ia menyapu \omterm{def:g5:heat-and-insulation:heat}{kalor} keluar dari tanganmu yang lebih hangat dengan
terburu, dan keterburuan itulah rasa ``sedingin es''. Kulit \omterm{def:g2:measuring-length:measure}{mengukur}
laju kehilangan \omterm{def:g5:heat-and-insulation:heat}{kalornya} sendiri, bukan \omterm{def:g3:temperature-thermometers:temperature}{suhu}.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:6}
Lomba yang adil hanya mengubah satu hal --- pembungkusnya. Cangkir
yang serupa, panas awal yang sama, dan ruangan yang sama memastikan
hanya syalnya yang menjelaskan perbedaan apa pun di antara kedua
kurvanya.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:7}
Sesudah $15$ menit: cangkir yang terbungkus kira-kira
\qty{41}{\celsius}, cangkir yang telanjang kira-kira
\qty{30}{\celsius}. Kedua kurvanya menuju \qty{20}{\celsius} milik
ruangan itu: alirannya berjalan sampai \omterm{def:g3:temperature-thermometers:temperature}{suhunya} bertemu, dan isolasi
hanya memperlambat perjalanannya.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:8}
Bulu yang menggembung memerangkap selimut tebal berisi \omterm{def:g2:air-around-us:air}{udara} yang diam
--- jalur lambat sang juara --- di sekeliling tubuh burung yang
hangat. Kartu yang sama: pakaian wol, selimut bulu, jendela ganda
(atau kain bulu domba, dinding berisolasi).
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:9}
Termos tidak membuat \omterm{def:g5:heat-and-insulation:heat}{kalor} maupun dingin: ia menutup jalur \omterm{def:g5:heat-and-insulation:heat}{kalor} ---
kekosongan di antara kedua dindingnya, kilap untuk memantulkan
pijarnya kembali. Pada bulan Januari ia memperlambat \omterm{def:g5:heat-and-insulation:heat}{kalor} cokelatnya
lolos keluar; pada bulan Juli ia memperlambat \omterm{def:g5:heat-and-insulation:heat}{kalor} \omterm{def:g3:sun-days-seasons:year}{musim} panas
merayap masuk. Satu alat, dua arah, karena hanya alirannya yang ia
urus.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:10}
Pada hari yang panas, \omterm{def:g5:heat-and-insulation:heat}{kalor} mencoba mengalir dari \omterm{def:g2:air-around-us:air}{udara} yang hangat
\emph{ke dalam} wadah yang dingin. Wolnya mengubah aliran masuk itu
menjadi rembesan --- perlambatan yang sama seperti yang dilakukannya
di sekeliling tubuh yang hangat. Isolasi tidak pilih kasih;
penjualnya, bukan pembelinya, yang memahaminya.
\end{solution}

\begin{solution}{exo:g5:heat-and-insulation:11}
Jendela ganda memenjarakan selapis \omterm{def:g2:air-around-us:air}{udara} yang diam --- jalur lambat
--- di antara kacanya, sehingga \omterm{def:g5:heat-and-insulation:heat}{kalor} rumahnya merembes keluar di
tempat jendela tunggal membiarkannya mengalir deras. \omterm{def:g5:heat-and-insulation:heat}{Kalor} yang lolos
lebih sedikit, bahan bakar penggantinya pun lebih sedikit. Tetapi
isolasi hanya memperlambat aliran menurunnya, tidak pernah
menghentikannya: sebagian \omterm{def:g5:heat-and-insulation:heat}{kalor} selalu bocor ke arah \omterm{def:g3:sun-days-seasons:year}{musim} dingin, dan
tungkunya harus terus menambahnya.
\end{solution}
