\section*{Bab \ref{ch:g5:series-parallel-circuits} --- Rangkaian Seri dan Paralel}

\begin{solution}{exo:g5:series-parallel-circuits:1}
\omterm{def:g5:series-parallel-circuits:parallel}{Cabang} adalah salah satu jalan terpisah tempat rangkaiannya memisah;
\omterm{def:g5:series-parallel-circuits:parallel}{titik cabang} adalah titik tempat jalan memisah atau bergabung kembali.
Dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} paralel menunggangi dua \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang membawa \omterm{def:g3:first-electric-circuit:bulb}{bohlam}.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:2}
Kemandirian (celah hanya menghentikan \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sendiri); terang penuh
(tiap \omterm{def:g3:first-electric-circuit:bulb}{bohlam} bersinar seolah sendirian, dengan \omterm{def:g3:first-electric-circuit:battery}{baterai} yang lebih
cepat habis sebagai gantinya); pos komando (\omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabang} menguasai
\omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}, \omterm{ex:g3:first-electric-circuit:switch}{sakelar} sebelum percabangan menguasai semuanya).
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:3}
\omterm{def:g3:first-electric-circuit:bulb}{Bohlam} 2 tetap menyala, tanpa terganggu --- jalannya sendiri lengkap.
Pada \omterm{def:g4:series-circuits:series}{rangkaian seri} ia menjadi gelap, karena di sana kedua \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}
berbagi satu jalan tunggal, dan celah apa pun memutuskannya bagi
keduanya.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:4}
Pasangan B (paralel) bersinar lebih terang untuk tiap \omterm{def:g3:first-electric-circuit:bulb}{bohlam} ---
masing-masing pada terang penuh. Pasangan B juga lebih cepat
menghabiskan \omterm{def:g3:first-electric-circuit:battery}{baterainya}: ia memberi makan dua \omterm{def:g5:series-parallel-circuits:parallel}{cabang} penuh sekaligus.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:5}
Keduanya tertutup: kedua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} menyala. \omterm{ex:g3:first-electric-circuit:switch}{Sakelar} utama terbuka:
keduanya gelap --- ia menguasai jalan bersamanya. \omterm{ex:g3:first-electric-circuit:switch}{Sakelar} utama
tertutup dan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabang} terbuka: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 menyala, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 gelap.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:6}
Pada \omterm{def:g4:series-circuits:series}{rangkaian seri}, satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang mati (atau satu lampu yang
dimatikan!) akan menggelapkan seluruh rumah, dan segalanya meredup
selagi lampu bertambah. Kenyamanan dari pemasangan paralel: \omterm{ex:g3:first-electric-circuit:switch}{sakelar}
tiap ruangan hanya menguasai ruangannya; \omterm{def:g3:first-electric-circuit:bulb}{bohlam} dapur yang putus
membiarkan sisa rumah tetap terang; setiap lampu menyala pada terang
penuh.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:7}
Sebelum percabangannya --- pada jalan bersama tempat arusnya masuk ke
rumah: \omterm{ex:g3:first-electric-circuit:switch}{sakelar} utama di kotak sekring. Ia memutus seluruh rumah
sekaligus, dipakai untuk perbaikan dan keadaan darurat.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:8}
Paralel: mematikan lampu langit-langit tidak membunuh lampu meja, dan
mencabut lampu mejanya tidak menggelapkan langit-langit ---
masing-masing menunggangi \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sendiri.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:9}
Dengan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} biasa milik kita, rancangannya terus gagal: \omterm{ex:g3:first-electric-circuit:switch}{sakelar}
yang seri dengan \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} hanya dapat dibuka dari satu ujung; dua
\omterm{ex:g3:first-electric-circuit:switch}{sakelar} biasa yang seri menuntut \emph{keduanya} tertutup (tiap ujung
dapat mematikan lampunya tetapi tidak selalu menyalakannya); dua yang
paralel menuntut keduanya terbuka supaya gelap. Kenyamanan lorongnya
memerlukan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang \emph{memilih di antara dua jalan} alih-alih
sekadar memutus satu jalan --- itulah \omterm{ex:g3:first-electric-circuit:switch}{sakelar} dua arah, yang dipakai
lorong sungguhan secara berpasangan.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:10}
Kenyamanannya: \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} menunggangi \omterm{def:g5:series-parallel-circuits:parallel}{cabang} paralel, sehingga \omterm{def:g3:first-electric-circuit:bulb}{bohlam}
yang mati hanya menggelapkan dirinya sendiri. Peringatannya: setiap
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang hilang menghapus satu \omterm{def:g5:series-parallel-circuits:parallel}{cabang}, dan \omterm{def:g3:first-electric-circuit:battery}{baterainya} (atau catu
daya kecil untaiannya) mendorong seluruh usahanya lewat \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang
tersisa --- kalau terlalu banyak yang hilang, tiap penyintasnya
menerima lebih daripada rancangannya, menyala lebih panas, dan mati
muda.
\end{solution}

\begin{solution}{exo:g5:series-parallel-circuits:11}
Pada ruas jalan milik \omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabang} itu sendiri: di antara \omterm{ex:g3:first-electric-circuit:switch}{sakelar}
\omterm{def:g5:series-parallel-circuits:parallel}{cabang} dan \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 (sebelum \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} bergabung kembali dengan yang
lain). Di situ ia menyala tepat ketika \omterm{def:g5:series-parallel-circuits:parallel}{cabang} itu berjalan --- dan ia
\omterm{def:g4:series-circuits:series}{terangkai seri} dengan \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2, keduanya menunggangi \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang sama.
\end{solution}
