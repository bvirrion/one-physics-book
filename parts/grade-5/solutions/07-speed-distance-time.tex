\section*{Chapter \ref{ch:g5:speed-distance-time} --- Speed: Distance and Time}

\begin{solution}{exo:g5:speed-distance-time:1}
Same distance --- less time wins; same time --- more distance wins.
``I ran farther'' fixes neither: with unequal times, more distance
proves nothing about being fast.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:2}
Bo --- same distance (one pool), less time ($25 < 30$ seconds):
rule one.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:3}
The scooter --- same time (one hour), more distance ($45 > 20$
kilometres): rule two.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:4}
In every hour it covers \qty{200}{km}. In $2$ \omterm{ex:g2:measuring-time:clock}{hours}: $200 \times 2
= \qty{400}{km}$. In half an hour: half of $200$, so \qty{100}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:5}
Snail ($5$ \omterm{def:g2:measuring-length:units}{metres} each hour), walker ($5$ kilometres), city bus
($30$), pigeon ($80$ kilometres each hour).
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:6}
An hour holds $30$ helpings of $2$ minutes: $30 \times 100 =
\qty{3000}{m}$ --- a \omterm{def:g5:speed-distance-time:speed}{speed} of $3$ kilometres in each hour.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:7}
The car: $80 \times 3 = \qty{240}{km}$. The cyclist: $15 \times 3 =
\qty{45}{km}$.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:8}
The faster traveler's line climbs more steeply. After $2$ \omterm{ex:g2:measuring-time:clock}{hours} the
cyclist's line stands at \qty{30}{km}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:9}
Hour by hour: $30$, $60$, $90$ --- after $3$ \omterm{ex:g2:measuring-time:clock}{hours} the ferry has
covered exactly \qty{90}{km}: the crossing takes $3$ \omterm{ex:g2:measuring-time:clock}{hours}.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:10}
Each hour Mara closes \qty{20}{km} and Jon \qty{10}{km}: the gap
shrinks by \qty{30}{km} each hour. From \qty{60}{km}, it reaches
zero after $2$ \omterm{ex:g2:measuring-time:clock}{hours} --- they meet, \qty{40}{km} from Mara's town.
\end{solution}

\begin{solution}{exo:g5:speed-distance-time:11}
After its one running minute the hare stands at \qty{60}{m} ---
halfway --- and falls asleep. The tortoise needs $120 \div 2 = 60$
minutes and finishes at time $60$. The hare wakes at time $1 + 70 =
71$, runs its last \qty{60}{m} in one minute, and finishes at time
$72$. The tortoise wins by $12$ minutes --- steady beats speedy,
now with arithmetic.
\end{solution}
