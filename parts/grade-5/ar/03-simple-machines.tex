\chapter{الآلات البسيطة: الروافع والبكرات والمستويات المائلة}\label{ch:g5:simple-machines}

لا محرّك ولا كهرباء --- ومع ذلك رفع البنّاؤون القدماء ببضعة
عوارض وحبال ومنحدرات حجارةً عجزت فرق كاملة من الثيران عن جرّها.
وكانت في عُدّتهم \emph{الآلات البسيطة}: اختراعات بلا محرّك البتة،
تحوّل سحبة ضعيفة مريحة إلى سحبة جبّارة. وأنت تملك أولاها أصلًا؛
فتعرّف إلى الأسرة.

\section{أسرة موفّرات الجهد}

\begin{definition}[الآلة البسيطة]\label{def:g5:simple-machines:machine}
\emph{الآلة البسيطة}\index{آلة بسيطة} أداة بلا محرّك تغيّر قوةً
لتلائمنا: فتجعلها أشدّ، أو تغيّر اتجاهها، أو الأمرين معًا.
والكلاسيكيات: \emph{\omterm{def:g3:levers-and-scales:lever}{الرافعة}} التي تعرفها من الأرجوحة والعتلة،
و\emph{\omterm{def:g5:simple-machines:pulley}{البكرة}}، و\emph{\omterm{def:g5:simple-machines:ramp}{المستوي المائل}} --- أي المنحدر
المتواضع.
\end{definition}

\begin{example}[الرافعة، من جديد]\label{ex:g5:simple-machines:lever}
قاعدة \omterm{def:g3:levers-and-scales:lever}{الرافعة} عندك أصلًا: القطع في العلامات --- فقوّتك الصغيرة
بعيدًا عن \omterm{def:g3:levers-and-scales:lever}{محور الارتكاز} تغلب حِملًا كبيرًا قريبًا منه. لكن
انتبه الآن إلى شيء آخر: طرفك من العتلة يكنس مسافة طويلة نازلًا
بينما لا ترتفع الصخرة إلا قليلًا. احتفظ بهذه الملاحظة في جيبك؛
فهي توشك أن تصير قانونًا.
\end{example}

\section{البكرات}

\begin{definition}[البكرة]\label{def:g5:simple-machines:pulley}
\emph{البكرة}\index{بكرة} عجلة فيها أخدود للحبل. والبكرة
\emph{الثابتة} تتعلّق بعارضة ولا تغيّر إلا اتجاه السحب: اسحب إلى
الأسفل يصعد الحِمل. أما البكرة \emph{المتحرّكة} فتركب الحبل
والحِمل معلّق بها: فيتعلّق الحِمل عندئذ \emph{بجديلتين} من
الحبل، ولا تحمل كل جديلة إلا نصف \omterm{def:g4:weight-and-mass:weight}{الوزن}.
\end{definition}

\begin{example}[البكرات في عملها]\label{ex:g5:simple-machines:pulleywork}
يصعد العلم في ساريته بينما يداك تسحبان \emph{إلى الأسفل} في
راحة: فهناك \omterm{def:g5:simple-machines:pulley}{بكرة} ثابتة في الأعلى --- تغيّر الاتجاه ولا تغيّر
الجهد، لكن السحب إلى الأسفل يجعل وزنك أنت يساعدك. أما مجموعة
بكرات البنّاء فتصل عدّة بكرات حتى يتعلّق الحِمل بأربع جدائل أو
ستّ: فتحمل كل جديلة رُبع \omterm{def:g4:weight-and-mass:weight}{الوزن} أو سُدسه، فيرفع عامل واحد بيانو.
لكن راقب العامل: تمرّ في يديه أمتار وأمتار من الحبل بينما يزحف
البيانو صاعدًا على مهل.
\end{example}

\section{المستوي المائل}

\begin{definition}[المستوي المائل]\label{def:g5:simple-machines:ramp}
\emph{المستوي المائل}\index{مستوٍ مائل} --- أي المنحدر --- منحدر
لرفع الأحمال من غير رفعها مستقيمة إلى الأعلى. وكلما لطف الميل
صغرت الدفعة اللازمة لتحريك الحِمل عليه: فدحرجة برميل على منحدر
طويل لطيف سهلة؛ وعلى منحدر قصير حادّ قاسية؛ ورفعه مستقيمًا إلى
الأعلى مستحيل.
\end{definition}

\begin{method}[تحسّسه بميزان الشريط المطّاطي]\label{met:g5:simple-machines:measure}
يستطيع ميزان الشريط المطّاطي من السنة الماضية أن يتذوّق الفرق:
\begin{enumerate}
  \item ضع سيارة لعبة على لوح واعلق الشريط المطّاطي بها؛
  \item وأسنِد اللوح منحدرًا لطيفًا إلى كومة كتب منخفضة؛ ثم
        اجرر السيارة صاعدة على مهل بالشريط ولاحظ المدّ؛
  \item ثم أسنِد اللوح نفسه أشدّ حدّة بكثير، إلى مقعد كرسيّ،
        واجرر من جديد.
\end{enumerate}
فالجرّ على المنحدر الحادّ يمدّ الشريط أكثر بكثير: فكلما حدّ
الميل كبر الجهد. والمنحدرات اللطيفة لطيفة حقًّا --- وجهازك يقول
ذلك.
\end{method}

\begin{example}[المنحدرات في المشهد]\label{ex:g5:simple-machines:ramps}
منحدرات الكراسي المتحرّكة تمتدّ طويلة ضحلة إلى جانب سلالم قصيرة
--- لطفٌ مقصود. وطرق الجبال تتعرّج: فبدل أن تهاجم القمّة
مستقيمة تلفّ مستويًا مائلًا طويلًا لطيفًا ذهابًا وإيابًا عبر
المنحدر. واللولب منحدر متنكّر --- منحدر ملفوف حول قضيب، فتسرق
كل دورة سهلة من المفكّ للولب خطوة دقيقة أعمق.
\end{example}

\section{القاعدة الذهبية}

\begin{proposition}[القاعدة الذهبية للآلات]\label{prop:g5:simple-machines:golden}
لا تعطي \omterm{def:g5:simple-machines:machine}{آلة بسيطة} شيئًا بلا مقابل: فما توفّره من \omterm{def:g2:pushes-and-pulls:force}{قوة} تستوفيه
مسافةً. نصف \omterm{def:g2:pushes-and-pulls:force}{القوة} --- ضِعف الحبل المسحوب، أو ضِعف الطريق
المقطوع؛ وعُشر \omterm{def:g2:pushes-and-pulls:force}{القوة} --- عشرة أضعاف الطريق. وفي كل آلة تكون
المقايضة دقيقة: فالراحة تُشترى بالأمتار.
\end{proposition}

\begin{example}[تحقّق من القاعدة]\label{ex:g5:simple-machines:check}
\omterm{def:g5:simple-machines:pulley}{البكرة} المتحرّكة: تسحب يدك بنصف \omterm{def:g2:pushes-and-pulls:force}{القوة}، ولرفع الحِمل \qty{1}{m}
تسحب \qty{2}{m} من الحبل --- فلا بدّ أن تقصر الجديلتان كلتاهما
مترًا. والعتلة: طرفك يكنس قوسًا طويلًا، والصخرة تكاد لا تتحرّك.
والمنحدر: يبلغ الصندوق الارتفاع نفسه، لكنك دفعته على المنحدر
كله. فكل توفير مدفوع ثمنه، بدقّة.
\end{example}

\begin{remark}[لماذا لا يمكن الغشّ في القاعدة]\label{rem:g5:simple-machines:whygolden}
تحت القاعدة الذهبية يختبئ صديقنا القديم من فصل \omterm{def:g5:everyday-energy:energy}{الطاقة}: فرفع
حِمل إلى ارتفاع يكلّف حصة ثابتة من \omterm{def:g5:everyday-energy:energy}{الطاقة}، ولا يغيّر أي ترتيب
للحبال والألواح ذلك الثمن. والآلات لا تفعل إلا أن تدعك تدفع
الثمن بعملات أصغر --- وأكثر عددًا. \omterm{def:g2:pushes-and-pulls:force}{فقوة} صغيرة على طريق طويل
تسلّم \omterm{def:g5:everyday-energy:energy}{الطاقة} نفسها التي تسلّمها \omterm{def:g2:pushes-and-pulls:force}{قوة} كبيرة على طريق قصير. ولهذا
السبب بعينه فشلت آلات الأبد: فالفاتورة فاتورة دائمًا.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:1}
سمِّ الآلات البسيطة الكلاسيكية الثلاث، وموضعًا يوميًّا لكل
واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:2}
ما الذي تغيّره \omterm{def:g5:simple-machines:pulley}{البكرة} الثابتة في سحبك؟ وما الذي تغيّره \omterm{def:g5:simple-machines:pulley}{البكرة}
المتحرّكة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:3}
لماذا يكون سحب علم \emph{إلى الأسفل} لرفعه أريح من رفع \omterm{def:g4:weight-and-mass:weight}{الوزن}
نفسه مستقيمًا إلى الأعلى، مع أن الجهد بالحجم نفسه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:4}
\omterm{def:g5:simple-machines:pulley}{ببكرة} متحرّكة واحدة يتعلّق الحِمل بجديلتين. فأيّ حصة من \omterm{def:g4:weight-and-mass:weight}{الوزن}
تحملها يدك؟ ولرفع الحِمل \qty{3}{m}، كم حبلًا يجب أن تسحب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:5}
في \cref{met:g5:simple-machines:measure}، أيّ منحدر يمدّ الشريط
المطّاطي أكثر؟ وما الذي يدفع ثمن المدّ الأصغر في المنحدر
اللطيف؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:6}
لماذا تتعرّج طرق الجبال بدل أن تصعد المنحدر مستقيمة؟ وأيّ \omterm{def:g5:simple-machines:machine}{آلة
بسيطة} هي الطريق كله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:7}
اذكر القاعدة الذهبية للآلات. آلة تتيح لك أن تسحب بعُشر \omterm{def:g2:pushes-and-pulls:force}{القوة}
--- فبمَ تُطالبك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:8}
كيف يكون اللولب مستويًا مائلًا متنكّرًا؟ وما الذي يقوم بدور
الطريق الطويل اللطيف؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:simple-machines:9}
مجموعة بكرات تعلّق بيانو بأربع جدائل. فأيّ كسر من \omterm{def:g4:weight-and-mass:weight}{وزن} البيانو
تسحبه يد العامل؟ ولرفع البيانو \qty{2}{m}، كم مترًا من الحبل
يمرّ في يديه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:simple-machines:10}
منحدر تحميل طوله \qty{6}{m} ويرتفع \qty{1}{m}؛ ودفع صندوق صاعدًا
عليه يحتاج إلى دفعة معتدلة. فيقطع النقّالون المتعجّلون منحدرًا
جديدًا طوله \qty{3}{m} فقط إلى الرصيف نفسه. استعمل القاعدة
الذهبية لمقارنة الدفعة الجديدة بالقديمة --- واشرح لماذا لم يكن
``نصف الطريق'' صفقة رابحة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:simple-machines:11}
يعرض إعلان مجموعة بكرات ``من الذكاء بحيث تسحب مترًا واحدًا من
الحبل بنصف \omterm{def:g2:pushes-and-pulls:force}{القوة} --- ويرتفع الحِمل مترًا كاملًا''. حاكِم الدعوى
أمام القاعدة الذهبية وفاتورة \omterm{def:g5:everyday-energy:energy}{الطاقة} في
\cref{rem:g5:simple-machines:whygolden}: أيستطيع أيّ ترتيب من
العجلات والحبال أن يفي بها؟
\end{exercise}
