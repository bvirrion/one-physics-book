\chapter{الحرارة والعزل}\label{ch:g5:heat-and-insulation}

لماذا يبدو المِزلج المعدني جليديًّا بينما لا يبدو كذلك المقعد
الخشبي إلى جانبه؟ ذلك اللغز يتبعنا منذ السنة الأولى من هذا
الكتاب. واليوم يسقط --- ومعه أسرار الترمس والنافذة المزدوجة
والعصفور المنفوش في الشتاء. والمفتاح أن نكفّ عن السؤال كيف
\emph{تبدو} الأشياء ونبدأ بتتبّع أين \emph{تجري} \omterm{def:g5:heat-and-insulation:heat}{الحرارة}.

\section{الحرارة في حركة}

\begin{definition}[الحرارة]\label{def:g5:heat-and-insulation:heat}
\emph{الحرارة}\index{حرارة} \omterm{def:g5:everyday-energy:energy}{طاقة} في حركتها من جسم أسخن إلى جسم
أبرد. فالحساء يمرّر حرارة إلى الملعقة، والمدفأة إلى الغرفة، ويدك
إلى كرة ثلج. والحرارة ليست مادة مختبئة داخل الأشياء الساخنة ---
إنما هي \emph{جريان}، ولها اتجاه دائمًا.
\end{definition}

\begin{proposition}[الحرارة تجري نازلة]\label{prop:g5:heat-and-insulation:downhill}
تجري \omterm{def:g5:heat-and-insulation:heat}{الحرارة} من تلقاء نفسها من الجسم الأسخن إلى الأبرد دائمًا
--- ولا تجري في الاتجاه الآخر أبدًا. ويستمرّ الجريان ما دامت
درجتا \omterm{def:g5:heat-and-insulation:heat}{الحرارة} مختلفتين، ويخبو حين تلتقيان: فإذا تُرك جاران
ساخن وبارد معًا \omterm{def:g2:measuring-time:duration}{مدة} كافية انتهيا إلى \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} واحدة مشتركة.
\end{proposition}

\begin{example}[قراءة الاتجاه]\label{ex:g5:heat-and-insulation:direction}
ملعقة الشاي في الشاي تسخن: فقد جرت \omterm{def:g5:heat-and-insulation:heat}{الحرارة} من الشاي $\to$ إلى
الملعقة. والليموناضة مع الثلج تبرد --- وانتبه! لم يدخلها شيء
اسمه ``البرودة'': بل جرت \omterm{def:g5:heat-and-insulation:heat}{الحرارة} \emph{خارجة من الليموناضة إلى
الثلج}، فصهرته. فالبرودة ليست جريانًا؛ إنما هي الاسم الذي نعطيه
لجانب الخاسر. وكتب الفيزياء كلها تقول ``يفقد المشروب \omterm{def:g5:heat-and-insulation:heat}{حرارة}''،
ولا تقول ``يكسب المشروب برودة'' أبدًا.
\end{example}

\section{مسارب سريعة ومسارب بطيئة}

\begin{definition}[نواقل الحرارة وعوازلها]\label{def:g5:heat-and-insulation:conductors}
تختلف الموادّ اختلافًا شديدًا في سرعة تمرير \omterm{def:g5:heat-and-insulation:heat}{الحرارة} فيها.
\emph{فناقل الحرارة}\index{ناقل للحرارة} مسرب سريع --- والمعادن
قبل كل شيء. \emph{وعازل الحرارة}\index{عازل للحرارة} مسرب بطيء:
الخشب، والصوف، والبلاستيك، والريش --- والبطل: \emph{\omterm{def:g2:air-around-us:air}{الهواء}
الساكن}، محبوسًا حتى لا يدور. (والكلمتان مستعارتان عن قصد من
نواقل الكهرباء وعوازلها؛ والخاصّيتان تجتمعان كثيرًا لا دائمًا.)
\end{definition}

\begin{example}[لغز المِزلج، وقد حُلّ]\label{ex:g5:heat-and-insulation:slide}
قضى المِزلج المعدني والمقعد الخشبي الليل عند \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة}
الباردة نفسها --- وقد أقسم \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} على ذلك من زمن بعيد.
المِسهما: يدك أدفأ من كليهما، فتجري \omterm{def:g5:heat-and-insulation:heat}{الحرارة} منك إلى كل واحد
منهما. لكن المعدن مسرب سريع --- يخطف حرارتك خطفًا، وذلك
\emph{الخطف} هو ما يبدو ``جليديًّا''. أما الخشب فلا يأخذ حرارتك
إلا قطرةً قطرة: فبالكاد يبدو باردًا. ولم يكن جلدك يقيس \omterm{def:g3:temperature-thermometers:temperature}{درجة
الحرارة} قطّ --- إنما يقيس سرعة فقدانه هو \omterm{def:g5:heat-and-insulation:heat}{للحرارة}. أُغلقت
القضية، بعد أربع سنوات.
\end{example}

\begin{method}[سباق التبريد]\label{met:g5:heat-and-insulation:race}
شاهِد العزل يفوز، بالأعداد:
\begin{enumerate}
  \item املأ كوبين متماثلين بماء صنبور ساخن بالتساوي
        (\emph{واطلب من شخص كبير ماءً ساخنًا يُحتمل باليد، لا
        ماءً غاليًا})؛
  \item ولفّ أحد الكوبين بوشاح صوفي لفًّا سميكًا؛ واترك الآخر
        عاريًا؛
  \item واغرز \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} في كل كوب واكتب درجتَي \omterm{def:g5:heat-and-insulation:heat}{الحرارة} في
        جدول كل خمس دقائق، ستّ مرات؛
  \item وارسم السجلّين خطّين على بيان واحد، \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} في
        مقابل الزمن.
\end{enumerate}
فيغوص خطّ الكوب العاري؛ وينحدر خطّ الكوب الملفوف انحدارًا
لطيفًا. البداية نفسها والغرفة نفسها --- والصوف جعل مخرج \omterm{def:g5:heat-and-insulation:heat}{الحرارة}
مسربًا بطيئًا.
\end{method}

\begin{remark}[العزل يُبطئ لا غير]\label{rem:g5:heat-and-insulation:onlyslows}
انظر إلى أين يتّجه المنحنيان: إلى \qty{20}{\celsius} في الغرفة.
فلا وشاح ولا ترمس ولا جدار كوخ جليدي يوقف جريان \omterm{def:g5:heat-and-insulation:heat}{الحرارة} النازل
--- إنما العزل يجعل المسرب البطيء أبطأ لا غير. ولهذا أيضًا أخّر
المعطفُ على رجل الثلج (أتذكره؟) انصهاره ولم يستطع إنقاذه: \omterm{def:g5:heat-and-insulation:heat}{فحرارة}
الصيف ظلّت تتسرّب إليه قطرةً قطرة.
\end{remark}

\section{فنّ حبس الهواء}

\begin{example}[هندسة الشتاء]\label{ex:g5:heat-and-insulation:air}
كل شيء دافئ تملكه تقريبًا آلة لحبس \omterm{def:g2:air-around-us:air}{الهواء}. فالصوف يُدفئك لا
بألياف نفسها بل بجيوب \omterm{def:g2:air-around-us:air}{الهواء} التي لا تُحصى بينها؛ ولحف الريش،
والفليس، والعصفور المنفوش كرةً على سلك متجمّد، كلها تلعب الورقة
نفسها. والبيوت تنضمّ إلى اللعبة: فالنوافذ المزدوجة تسجن طبقة
\omterm{def:g2:air-around-us:air}{هواء} بين لوحين، والجدران المعزولة محشوّة بحشوة منفوشة تكنز
\omterm{def:g2:air-around-us:air}{الهواء}. \omterm{def:g2:air-around-us:air}{فالهواء} الساكن بطل المسارب البطيئة --- والحيلة أن يُبقى
\emph{ساكنًا}.
\end{example}

\begin{example}[الترمس]\label{ex:g5:heat-and-insulation:thermos}
قارورة الترمس تحفة العزل: قنّينة داخل قنّينة، \omterm{def:g2:air-around-us:air}{والهواء} بينهما
مسحوب حتى يكاد لا يبقى منه شيء --- وحيث لا تكاد توجد مادة لا
يكاد جريان \omterm{def:g5:heat-and-insulation:heat}{الحرارة} يجد مسربًا البتة (وهي حيلة تستحقّ التذكّر:
فالخلاء نفسه هو الذي أسكت الجرس في الوعاء). ثم أضِف جدرانًا
لامعة تردّ توهّج الدفء إلى الداخل، فتلقى الشوكولاتة الساخنة
العصرَ وهي ساخنة --- أو الليموناضة وهي باردة: فالترمس لا يعرف
شيئًا عن الساخن والبارد، إنما يُغلق المسارب لا غير، في
الاتجاهين.
\end{example}

\begin{remark}[التدفئة، مقولةً على أصولها]\label{rem:g5:heat-and-insulation:coat}
والآن تُغلق مسألة المعطف القديمة إلى الأبد. فالمعطف \omterm{def:g2:air-around-us:air}{هواء} محبوس
حول صانع \omterm{def:g5:heat-and-insulation:heat}{حرارة} --- هو أنت. فجسمك يبثّ \omterm{def:g5:heat-and-insulation:heat}{الحرارة} إلى الخارج طوال
النهار؛ والمعطف يحوّل ذلك البثّ إلى قطرات، فتبقى دافئًا من
إنتاجك أنت. أما على رجل الثلج --- ولا صانع \omterm{def:g5:heat-and-insulation:heat}{حرارة} في داخله ---
فالمعطف نفسه لا يفعل إلا أن يُبطئ تسرّب الصيف إلى الداخل.
والعزل لا يحابي أحدًا: فهو يُبطئ الجريان في أي اتجاه جرى.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:1}
ما \omterm{def:g5:heat-and-insulation:heat}{الحرارة}، وفي أيّ اتجاه تجري من تلقاء نفسها؟ ومتى يتوقّف
الجريان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:2}
``أغلق الباب، فأنت تُدخل البرد!'' أعد كتابة جملة الجدّة كما
يكتبها كتاب فيزياء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:3}
صنّف إلى مسارب سريعة ومسارب بطيئة \omterm{def:g5:heat-and-insulation:heat}{للحرارة}: مقلاة نحاسية؛ \ وشاح
صوفي؛ \ ملعقة فولاذية؛ \ \omterm{def:g2:air-around-us:air}{هواء} ساكن؛ \ ملعقة خشبية؛ \ لحاف
ريش.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:4}
لماذا تكون قيعان المقالي معدنية ومقابضها بلاستيكية أو خشبية؟
أجب بالمسارب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:5}
أخيرًا، بكلماتك أنت: لماذا يبدو المعدن أبرد من الخشب عند \omterm{def:g3:temperature-thermometers:temperature}{درجة
الحرارة} نفسها؟ وما الذي يقيسه الجلد حقًّا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:6}
في سباق التبريد، لماذا يجب أن يكون الكوبان متماثلين وأن يبدآ
ساخنين بالتساوي؟ وأيّ فرق واحد تختبره التجربة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:7}
اقرأ بيان التبريد: عند أيّ \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} تقريبًا يقف كل كوب بعد
$15$ دقيقة؟ وإلى أيّ \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} يتّجه المنحنيان، ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:8}
لماذا ينفش العصفور نفسه كرةً على سلك متجمّد؟ وسمِّ ثلاثة
اختراعات بشرية تلعب الورقة نفسها.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:9}
الترمس نفسه يُبقي الشوكولاتة ساخنة في كانون الثاني والليموناضة
باردة في تموز. اشرح كيف يقوم جهاز واحد بالعملين --- ماذا يفعل
الترمس حقًّا، وبأيّ شيء؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:10}
تلفّ بائعة مثلّجات أوعيتها ببطّانيات صوفية سميكة في يوم حارّ،
فيضحك زبون: ``الصوف لإبقاء الأشياء دافئة!'' دافع عن البائعة بما
في \cref{rem:g5:heat-and-insulation:coat} --- في أيّ اتجاه
تحاول \omterm{def:g5:heat-and-insulation:heat}{الحرارة} أن تجري، وماذا يفعل بها الصوف؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:11}
بيتان في الشتاء نفسه: أحدهما بنوافذ مفردة والآخر بنوافذ مزدوجة.
اشرح بالمسارب \omterm{def:g2:air-around-us:air}{والهواء} المحبوس لماذا يحرق البيت ذو النوافذ
المزدوجة وقودًا أقلّ --- ولماذا يظلّ محتاجًا، وفق
\cref{rem:g5:heat-and-insulation:onlyslows}، إلى \emph{بعض}
التدفئة.
\end{exercise}
