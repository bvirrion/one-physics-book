\chapter{الدارات على التسلسل وعلى التفرع}\label{ch:g5:series-parallel-circuits}

حين مات \omterm{def:g3:first-electric-circuit:bulb}{مصباح} واحد أظلمت سلسلة الزينة كلها --- تلك لعنة الحلقة
الواحدة. ومع ذلك حين يحترق \omterm{def:g3:first-electric-circuit:bulb}{مصباح} في مطبخك يظلّ سائر البيت مضيئًا.
فقد وصّل أحدهم بيتك على نحو أذكى. واليوم نتعلّمه: طريقة الفروع.

\section{طريقتان للتوصيل}

أنت تعرف الطريقة الأولى: \emph{\omterm{def:g4:series-circuits:series}{على التسلسل}}، وفيها ينتظم اللاعبون
كلهم في حلقة واحدة، خرزات في قلادة واحدة. وإليك الثانية.

\begin{definition}[على التفرع]\label{def:g5:series-parallel-circuits:parallel}
يكون اللاعبون \emph{على التفرع}\index{دارة على التفرع} حين تنقسم
الدارة \emph{فروعًا}\index{فرع} منفصلة --- لاعب واحد في كل فرع
--- تتشعّب ثم تلتقي. وتُسمّى النقاط التي تتشعّب عندها الطرق أو
تلتقي \emph{عقدًا}\index{عقدة}. وكل فرع طريق كامل من أحد قطبَي
\omterm{def:g3:first-electric-circuit:battery}{البطارية} إلى الآخر.
\end{definition}

\omterm{def:g3:first-electric-circuit:bulb}{مصباح} فرعه الخاص، ثم يلتقي الفرعان قبل العودة إلى \omterm{def:g3:first-electric-circuit:battery}{البطارية}.}

\begin{method}[كيف تبنيها]\label{met:g5:series-parallel-circuits:build}
\omterm{def:g3:first-electric-circuit:battery}{ببطارية} واحدة ومصباحين وأسلاك:
\begin{enumerate}
  \item وصّل \omterm{def:g3:first-electric-circuit:bulb}{المصباح} الأول \omterm{def:g3:first-electric-circuit:battery}{بالبطارية} وحده، حلقةً بسيطة ---
        وهذا \omterm{def:g5:series-parallel-circuits:parallel}{الفرع} الأول؛
  \item ثم وصّل \omterm{def:g3:first-electric-circuit:bulb}{المصباح} الثاني \emph{عبر} الأول مباشرة: سلكًا من
        كل جانب من \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 1 إلى الجانب الموافق في \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 2
        --- وهذا \omterm{def:g5:series-parallel-circuits:parallel}{الفرع} الثاني؛
  \item وتتبّع كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} بإصبعك: فمن $+$، يجب أن يمرّ كل طريق
        \omterm{def:g3:first-electric-circuit:bulb}{بمصباح} واحد بالضبط قبل أن يبلغ $-$.
\end{enumerate}
فيضيء المصباحان --- ويسطع كل واحد منهما سطوعَ \omterm{def:g3:first-electric-circuit:bulb}{مصباح} مفرد.
\end{method}

\section{قواعد الفروع}

\begin{proposition}[قواعد الدارة على التفرع]\label{prop:g5:series-parallel-circuits:rules}
مع وجود كل لاعب في فرعه الخاص:
\begin{enumerate}
  \item \emph{الاستقلال}: الفراغ في \omterm{def:g5:series-parallel-circuits:parallel}{فرع} لا يوقف إلا ذلك \omterm{def:g5:series-parallel-circuits:parallel}{الفرع}
        --- فُكّ مصباحًا يظلّ الآخر مشتعلًا؛
  \item \emph{السطوع الكامل}: يسطع كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح} كأنه وحده ---
        فلا تقاسم لدفع \omterm{def:g3:first-electric-circuit:battery}{البطارية} بين الفروع؛
  \item \emph{مواقع الأمر}: \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} داخل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} لا تأمر إلا ذلك
        \omterm{def:g5:series-parallel-circuits:parallel}{الفرع}؛ \omterm{ex:g3:first-electric-circuit:switch}{والقاطعة} على الطريق المشترك، قبل التشعّب، تأمر
        الجميع. (والثمن: أن \omterm{def:g3:first-electric-circuit:battery}{البطارية} تغذّي كل الفروع دفعة
        واحدة، فتنفد أسرع.)
\end{enumerate}
\end{proposition}

\begin{example}[اختبار الفكّ، معادًا]\label{ex:g5:series-parallel-circuits:test}
افعل بالزوج المتفرّع ما قتل زوج التسلسل يومًا: فُكّ \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 2.
فلا يرتعش \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 1 حتى --- إذ إن فرعه الخاص، وطريقه الكامل، لم
يُمسّ. ثم أعده وفُكّ \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 1 بدلًا منه: يظلّ \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 2 مشتعلًا
هادئًا. فقد رُفعت لعنة الحلقة الواحدة: \omterm{def:g5:series-parallel-circuits:parallel}{فعلى التفرع} يعتني كل
لاعب بفرعه هو.
\end{example}

\begin{example}[بيتك دارة على التفرع]\label{ex:g5:series-parallel-circuits:house}
كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح} ومأخذ وجهاز في بيتك يركب فرعه الخاص من دارة كبيرة واحدة
\omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}. ولهذا يستطيع \omterm{def:g3:first-electric-circuit:bulb}{مصباح} المطبخ أن يحترق بينما تواصل غرفة
الجلوس قراءتها غير مضطربة؛ ولهذا لا تأمر قاطعة كل غرفة إلا
ضوءها؛ ولهذا تستطيع قاطعة رئيسية واحدة --- في علبة المصاهر عند
العدّاد --- أن تقطع البيت كله دفعة واحدة للإصلاح. وقد انضمّت
أضواء الزينة الحديثة إلى النادي: فمصابيحها موصّلة بحيث لا يُظلم
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} ميت السلسلةَ بعد اليوم.
\end{example}

\begin{remark}[التسلسل والتفرع جنبًا إلى جنب]\label{rem:g5:series-parallel-circuits:compare}
فراغ واحد: \omterm{def:g4:series-circuits:series}{على التسلسل} --- الكلّ معتم؛ \omterm{def:g5:series-parallel-circuits:parallel}{وعلى التفرع} --- \omterm{def:g5:series-parallel-circuits:parallel}{فرع} واحد
معتم. ومصابيح أكثر: \omterm{def:g4:series-circuits:series}{على التسلسل} --- الكلّ أخفت؛ \omterm{def:g5:series-parallel-circuits:parallel}{وعلى التفرع} ---
الكلّ بسطوع كامل، لكن \omterm{def:g3:first-electric-circuit:battery}{البطارية} تنفد أسرع. وقاطعة واحدة لكل شيء:
\omterm{def:g4:series-circuits:series}{على التسلسل} --- في أي موضع؛ \omterm{def:g5:series-parallel-circuits:parallel}{وعلى التفرع} --- قبل التشعّب فقط.
وليست إحدى الطريقتين ``أفضل'': \omterm{def:g3:first-electric-circuit:bulb}{فالمصباح} اليدوي سعيد \omterm{def:g4:series-circuits:series}{على التسلسل}
(\omterm{def:g3:first-electric-circuit:bulb}{مصباح} واحد وقاطعة واحدة)، والبيت لا بدّ أن يكون \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}.
وأول سؤال يسأله الكهربائي عن أي دارة هو ببساطة: \emph{أين
يتشعّب الطريق؟}
\end{remark}

\section{قراءة الدارات}

\begin{method}[تتبّع الطرق]\label{met:g5:series-parallel-circuits:follow}
لتتوقّع مصير أي \omterm{def:g3:first-electric-circuit:bulb}{مصباح} في رسم دارة:
\begin{enumerate}
  \item ضع إصبعك على قطب \omterm{def:g3:first-electric-circuit:battery}{البطارية} $+$ وامشِ كل طريق ممكن إلى
        $-$، مُحصيًا اللاعبين الذين يمرّ بهم كل طريق؛
  \item يضيء \omterm{def:g3:first-electric-circuit:bulb}{المصباح} إن مرّ به طريق واحد غير مقطوع على الأقلّ؛
  \item واللاعبون الذين يشتركون في كل الطرق هم \omterm{def:g4:series-circuits:series}{على التسلسل} بعضهم
        مع بعض؛ واللاعبون على تشعّبات مختلفة هم \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}.
\end{enumerate}
وامشِ الطرق قبل أن تتوقّع --- فالحدس من شكل الرسم وحده هو ما
تخدع به الداراتُ الناسَ.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:1}
ما \omterm{def:g5:series-parallel-circuits:parallel}{الفرع}؟ وما \omterm{def:g5:series-parallel-circuits:parallel}{العقدة}؟ وفي \omterm{def:g5:series-parallel-circuits:parallel}{دارة على التفرع} فيها مصباحان، كم
فرعًا يحمل مصباحًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:2}
اذكر قواعد \omterm{def:g5:series-parallel-circuits:parallel}{الدارة على التفرع} الثلاث.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:3}
مصباحان \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}: فُكّ \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 1. فماذا يفعل \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 2؟ وماذا
فعل في زوج التسلسل، ولماذا الفرق؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:4}
مصباحان متماثلان: الزوج A \omterm{def:g4:series-circuits:series}{على التسلسل}، والزوج B \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}،
بالبطاريات نفسها. فأيّ الزوجين أسطع في كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح}؟ وأيّهما ينفد
بطاريته أسرع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:5}
في الشكل ذي القاطعتين، ماذا يفعل كل \omterm{def:g3:first-electric-circuit:bulb}{مصباح} حين: تكون القاطعتان
مغلقتين؟ \ وحين تكون الرئيسية مفتوحة؟ \ وحين تكون الرئيسية
مغلقة وقاطعة \omterm{def:g5:series-parallel-circuits:parallel}{الفرع} مفتوحة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:6}
لماذا يُوصَّل البيت \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} لا \omterm{def:g4:series-circuits:series}{على التسلسل}؟ اذكر راحتين
يوميّتين تعتمدان على ذلك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:7}
أين في دارة البيت تأمر القاطعةُ كلَّ شيء دفعة واحدة؟ وما اسم
تلك \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} اليومي واستعمالها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:8}
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} السرير ومأخذه: أهما \omterm{def:g4:series-circuits:series}{على التسلسل} أم \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} مع \omterm{def:g1:light-and-shadows:source}{ضوء}
السقف؟ وكيف تعرف ذلك من الحياة اليومية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:9}
ارسم (بالرموز) دارةً لممرّ: \omterm{def:g3:first-electric-circuit:battery}{بطارية} واحدة، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} واحد، وهذه
الراحة --- أن يُشعَل \omterm{def:g3:first-electric-circuit:bulb}{المصباح} ويُطفَأ من قاطعة عند \emph{أيّ} من
طرفَي الممرّ\dots\ حاول بصدق؛ فإن أصرّت تصاميمك على الفشل فاشرح
ما الصعب في الأمر. (فالتبديل من طرفين حقًّا يحتاج إلى قاطعة أذكى
من قاطعتنا --- والمحاولة تعلّم أكثر من الجواب.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:10}
سلسلة أضواء زينة حديثة تظلّ مضيئة حين يموت \omterm{def:g3:first-electric-circuit:bulb}{مصباح} واحد --- لكن
العلبة تحذّر: ``لا تشغّل السلسلة وقد نُزع منها أكثر من ثلاثة
مصابيح.'' استعمل قواعد الفروع لتشرح الراحة، ثم --- وهو الأصعب
--- سببًا معقولًا للتحذير. (فكّر في ما يناله كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} باقٍ من
\omterm{def:g3:first-electric-circuit:battery}{البطارية}، وفي جهد \omterm{def:g3:first-electric-circuit:battery}{البطارية} نفسها.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:11}
في شكل القاطعتين يريد كهربائي أن يضيف مصباحًا آخر يضيء
\emph{فقط} حين تكون قاطعة \omterm{def:g5:series-parallel-circuits:parallel}{الفرع} مغلقة، ويبقى معتمًا فيما عدا
ذلك --- ولو كانت \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} الرئيسية مغلقة. فعلى أيّ امتداد من
الطريق يجب أن يُوصَّل \omterm{def:g3:first-electric-circuit:bulb}{المصباح} الجديد، وهل يكون عندئذ \omterm{def:g4:series-circuits:series}{على
التسلسل} أم \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع} مع \omterm{def:g3:first-electric-circuit:bulb}{المصباح} 2؟
\end{exercise}
