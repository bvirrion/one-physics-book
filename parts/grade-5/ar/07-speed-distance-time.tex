\chapter{السرعة: المسافة والزمن}\label{ch:g5:speed-distance-time}

``أنا أسرع!'' --- ``لا، بل أنا!'' وكل ساحة لعب تحسم هذا بالطريقة
الأمينة: اصطفّا، استعدّا، انطلقا. لكن كيف تقارن راكب درّاجة
بعدّاء تسابق أمس، على طريق أخرى؟ \omterm{def:g5:speed-distance-time:speed}{السرعة} هي كيف تحسم الفيزياء
سباقات لا يمكن أن تُجرى جنبًا إلى جنب أبدًا.

\section{طريقتان للفوز في سباق}

\begin{proposition}[قواعد السباق العادل]\label{prop:g5:speed-distance-time:rules}
طريقتان أمينتان لمقارنة السريع بالبطيء:
\begin{enumerate}
  \item على \emph{المسافة نفسها}، يحتاج الأسرع إلى \emph{زمن
        أقلّ} --- فأول من يعبر الخطّ يفوز بسباق العَدو؛
  \item في \emph{الزمن نفسه}، يقطع الأسرع \emph{مسافة أكبر} ---
        فبعد دقيقة واحدة يكون السبّاح الأسرع أبعد ببساطة.
\end{enumerate}
المسافة نفسها أو الزمن نفسه: ثبّت أحدهما وقارن الآخر. أما
المقارنة و\emph{كلاهما} مختلف --- ``قطعتُ مسافة أبعد!'' ``لكنك
جريت بعد الظهر كله!'' --- فلا تحسم شيئًا.
\end{proposition}

\begin{example}[غير عادل وعادل]\label{ex:g5:speed-distance-time:fair}
قطعت لينا \qty{10}{km} بالدرّاجة؛ وقطع ماركو \qty{15}{km}. أفهو
أسرع؟ لا سبيل إلى معرفة ذلك --- فلعلّه ركب ثلاثة أضعاف \omterm{def:g2:measuring-time:duration}{المدة}.
لكن إذا ركب كلاهما ساعة واحدة بالضبط صارت المقارنة عادلة: الزمن
نفسه ومسافة أكبر، فماركو أسرع. وعبارة ``في ساعة واحدة'' الصغيرة
هذه توشك أن تصير عبارتنا المفضّلة.
\end{example}

\section{السرعة: المسافة في ساعة واحدة}

\begin{definition}[السرعة]\label{def:g5:speed-distance-time:speed}
\emph{سرعة}\index{سرعة} مسافر منتظم هي المسافة التي يقطعها في
حصة ثابتة من الزمن --- ساعة واحدة عادةً. فقولنا ``تقطع هذه
السيارة $60$ كيلومترًا في كل ساعة'' يسمّي سرعتها؛ والماشي النشيط
يقطع نحو $5$ كيلومترات في كل ساعة، وراكب درّاجة السباق $40$.
وللزاحفات نستعمل حصصًا ألطف: فالحلزون يقطع نحو $5$ أمتار في كل
ساعة.
\end{definition}

\begin{method}[كيف تقيس سرعتك]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item قِس أو جِد امتدادًا طوله \qty{100}{m} بالضبط (وكثير من
        مضامير الرياضة يعلّم واحدًا)؛
  \item وامشِه بخطوتك الطبيعية بينما يقيس صديق زمنك --- ولنفرض
        أنه استغرق دقيقة واحدة؛
  \item والساعة تسع $60$ دقيقة كهذه، فتقطع في ساعة
        $60 \times 100 = \qty{6000}{m}$؛
  \item وبالكيلومترات: سرعتك في المشي $6$ كيلومترات في كل ساعة.
\end{enumerate}
وأعِد ذلك عَدوًا وقارن --- فأكثر الأطفال يضاعفون سرعتهم
تقريبًا.
\end{method}

\section{توقّع المسافات}

\begin{proposition}[المسافة من السرعة]\label{prop:g5:speed-distance-time:distance}
المسافر المنتظم يقطع مسافة ساعته الواحدة من جديد في كل ساعة
تالية. ومن ثمّ:
\[
\text{المسافة المقطوعة} = \text{المسافة في ساعة واحدة} \times
\text{عدد الساعات}.
\]
فسيارة تقطع $60$ كيلومترًا في كل ساعة تقطع في $3$ ساعات
$60 \times 3 = \qty{180}{km}$.
\end{proposition}

\begin{example}[حسابات الرحلات]\label{ex:g5:speed-distance-time:journeys}
الماشي عند $4$ كيلومترات في كل ساعة، خارجًا $2$ ساعة:
$4 \times 2 = \qty{8}{km}$. وراكب الدرّاجة عند $15$ كيلومترًا في
كل ساعة، راكبًا $4$ ساعات: $15 \times 4 = \qty{60}{km}$. والطائرة
عند $900$ كيلومتر في كل ساعة، طائرةً $5$ ساعات: $900 \times 5 =
\qty{4500}{km}$ --- قارّة تُعبَر بين وجبتين.
\end{example}

\begin{remark}[الانتظام تبسيط]\label{rem:g5:speed-distance-time:steady}
المسافرون الحقيقيون نادرًا ما ينتظمون: فالسيارة تبطئ في القرى،
وراكب الدرّاجة يطير نازلًا ويكدح صاعدًا، وأنت تتمهّل عند
المخبز. وحساباتنا تتظاهر بأن الرحلة كلها تجري بوتيرة واحدة
أمينة --- وهو تظاهر نافع، يكفي للتخطيط. أما كيف نكون أمناء في
رحلة ذات وتائر كثيرة --- وما معنى ``المتوسط'' حقًّا --- فقصة
لسنة لاحقة، حين تملك الرياضيات المناسبة.
\end{remark}

\begin{example}[قراءة السرعات من حولك]\label{ex:g5:speed-distance-time:signs}
لافتات الطرق ولوحات السيارات تتكلّم بالكيلومترات في الساعة
الواحدة --- فاللافتة المستديرة التي عليها $50$ تقول للسائقين:
اقطعوا $50$ كيلومترًا في كل ساعة على الأكثر، وهي وتيرة اختيرت
حتى يبقى في الوسع تفادي طفل يجري وراء كرة. وقد تتباهى شاشة قطار
\omterm{def:g5:speed-distance-time:speed}{السرعة} العالية بالعدد $300$؛ أما لافتات المشاة في الجبال فتفضّل
الساعات على الكيلومترات --- ``البحيرة: ساعتان'' --- واثقةً بأن
الماشي يعرف سرعته هو.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
اذكر الطريقتين العادلتين لمقارنة السريع بالبطيء. ولماذا لا تثبت
عبارة ``جريتُ أبعد منك'' وحدها شيئًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
سبّاحان يعبران المسبح نفسه؛ فتحتاج آنا إلى $30$ ثانية ويحتاج بو
إلى $25$. فأيّهما أسرع، وبأيّ قاعدة من قواعد السباق العادل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
بعد ساعة واحدة قطع جرّار \qty{20}{km} وقطعت درّاجة نارية
\qty{45}{km}. فأيّهما أسرع، وبأيّ قاعدة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
ماذا يعني قولنا ``يقطع هذا القطار $200$ كيلومتر في كل ساعة''؟
وكم يبلغ في $2$ ساعة؟ وفي نصف ساعة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
رتّب بحسب \omterm{def:g5:speed-distance-time:speed}{السرعة}: ماشٍ ($5$ كيلومترات في كل ساعة)؛ \ حمامة
($80$ كيلومترًا في كل ساعة)؛ \ حلزون ($5$ \emph{أمتار} في كل
ساعة)؛ \ حافلة مدينة ($30$ كيلومترًا في كل ساعة).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
تتمشّى \qty{100}{m} معلَّمة في $2$ دقيقة. فكم مترًا تقطع في ساعة
بتلك الوتيرة؟ وقل سرعتك بالكيلومترات في كل ساعة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
تحفظ سيارة $80$ كيلومترًا في كل ساعة \omterm{def:g2:measuring-time:duration}{مدة} $3$ ساعات. فكم تسافر؟
وراكب درّاجة عند $15$ كيلومترًا في كل ساعة في الساعات الثلاث
نفسها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
في بيان الرحلات في الفصل، كيف تكتشف المسافر الأسرع بنظرة واحدة؟
واقرأ منه: كم قطع راكب الدرّاجة بعد $2$ ساعة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
عبّارة تقطع $30$ كيلومترًا في كل ساعة. والعبور \qty{90}{km}.
باستعمال \cref{prop:g5:speed-distance-time:distance} بالمقلوب
--- وجرّب الساعات واحدة واحدة --- كم يستغرق العبور؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
بين مدينتين \qty{60}{km}. تنطلق مارا من إحداهما بدرّاجتها عند
$20$ كيلومترًا في كل ساعة؛ وفي اللحظة نفسها ينطلق جون من الأخرى
نحوها عند $10$ كيلومترات في كل ساعة. فبكم كيلومترًا تتقلّص
المسافة بينهما في كل ساعة؟ وبعد كم ساعة يلتقيان؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
الحكاية القديمة، بالأعداد: طول السباق \qty{120}{m}. يجري الأرنب
$60$ مترًا في كل دقيقة --- لكنه بعد دقيقة واحدة من الجري يقف
لقيلولة مدتها $70$ دقيقة، ثم يواصل الجري. وتدبّ السلحفاة $2$
\omterm{def:g2:measuring-length:units}{متر} في كل دقيقة ولا تقف أبدًا. فأين يكون الأرنب حين يبدأ
قيلولته؟ وفي أيّ لحظة يعبر كل واحد منهما خطّ النهاية؟ ومَن
يفوز؟
\end{exercise}
