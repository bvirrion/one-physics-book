\chapter{الضغط الجوي والطقس}\label{ch:g5:air-pressure-weather}

أنت تعيش في قاع محيط --- محيط من \omterm{def:g2:air-around-us:air}{الهواء}، عمقه مئات الكيلومترات،
وثقله كله مستقرّ على كتفيك. فلماذا لا يسحقك؟ ولماذا لا تحسّ به
حتى؟ وما شأن هذا كله بطقس الغد؟ \omterm{def:g2:air-around-us:air}{للهواء}، صديقنا الخفيّ القديم،
سرّ كبير أخير يسلّمه.

\section{للهواء وزن}

\begin{example}[وزن ما لا يُرى]\label{ex:g5:air-pressure-weather:weighing}
تُوضع كرة قدم منفوخة على \omterm{def:g3:levers-and-scales:scale}{ميزان كفّتين} دقيق وتُوازَن تمامًا. ثم
يُفتح صمّامها فيهسّ \omterm{def:g2:air-around-us:air}{الهواء} الزائد خارجًا --- الكرة نفسها،
والصمّام نفسه، وكل شيء نفسه، ناقصًا بعض \omterm{def:g2:air-around-us:air}{الهواء}. ثم تعود إلى
الميزان: فإذا الكرة المفرَّغة \emph{أخفّ}. والفرق هو \omterm{def:g4:weight-and-mass:weight}{وزن} \omterm{def:g2:air-around-us:air}{الهواء}
الهارب. \omterm{def:g2:air-around-us:air}{فالهواء} مادة، وللمادة \omterm{def:g3:measuring-mass:mass}{كتلة}، \omterm{def:g3:measuring-mass:mass}{والكتلة} تجذبها الأرض إلى
الأسفل: فالمحيط الذي فوقنا ثقيل.
\end{example}

\begin{definition}[الغلاف الجوي وضغطه]\label{def:g5:air-pressure-weather:pressure}
\emph{الغلاف الجوي}\index{غلاف جوي} طبقة \omterm{def:g2:air-around-us:air}{الهواء} العميقة الملفوفة
حول الأرض كلها. ووزنه يعصر \omterm{def:g2:air-around-us:air}{الهواء} في القاع --- حيث نعيش ---
\omterm{def:g2:air-around-us:air}{والهواء} المعصور يدفع بشدّة كل سطح يلمسه. ويُسمّى ذلك الدفع
\emph{الضغط الجوي}\index{ضغط جوي}.
\end{definition}

\begin{proposition}[الدفع من كل الجهات]\label{prop:g5:air-pressure-weather:allsides}
يدفع \omterm{def:g5:air-pressure-weather:pressure}{الضغط الجوي} كل سطح من \emph{كل} جهة --- إلى الأسفل على
الطاولات، وجانبًا على الجدران، بل \emph{إلى الأعلى} على السقوف
والراحات. وهو جبّار: فعلى مساحة بقدر راحة يدك يدفع \omterm{def:g2:air-around-us:air}{الهواء} بقدر
\omterm{def:g4:weight-and-mass:weight}{وزن} حقيبة سفر كبيرة ممتلئة تقريبًا. ونحن لا ننتبه أبدًا، لأن
الدفعات تأتي متعادلة: \omterm{def:g2:air-around-us:air}{فالهواء} الذي في داخلنا وخلف كل سطح يدفع
بالشدّة نفسها.
\end{proposition}

\begin{method}[البطاقة التي تحبس الماء]\label{met:g5:air-pressure-weather:card}
دفع \omterm{def:g2:air-around-us:air}{الهواء} إلى الأعلى، ممسوكًا بالجرم المشهود (افعله فوق
الحوض):
\begin{enumerate}
  \item املأ كأسًا بالماء حتى الحافة تمامًا؛
  \item وضع \omterm{def:g5:everyday-energy:energy}{بطاقة} قاسية ملساء مستوية فوقها، ملامسةً للماء من كل
        جانب؛
  \item وأمسك البطاقة في موضعها، واقلب الكأس رأسًا على عقب
        بالضبط، ثم أفلت البطاقة برفق.
\end{enumerate}
فتبقى البطاقة --- حابسةً ملء الكأس كله. \omterm{def:g4:weight-and-mass:weight}{فوزن} الماء يضغط البطاقة
إلى الأسفل، لكن \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} يضغطها \emph{إلى الأعلى} بشدّة
أكبر. لقد رأيت للتوّ دفعة الحقيبة من تحت.
\end{method}

\section{آلات يشغّلها الغلاف الجوي}

\begin{example}[الماصّة]\label{ex:g5:air-pressure-weather:straw}
تظنّ أنك تسحب العصير صاعدًا في الماصّة. والحقيقة أنك لا تستطيع
سحب \omterm{def:g3:solids-liquids-gases:liquid}{سائل} البتة --- جرّب أن تسحب الماء بأصابعك! وإنما تمصّ
\emph{\omterm{def:g2:air-around-us:air}{الهواء}} من الماصّة. فإذا خلا داخلها من \omterm{def:g2:air-around-us:air}{الهواء} دفع الغلافُ
الجوي الضاغط على العصير في الكأس العصيرَ صاعدًا في الماصّة
الخالية إلى فمك. فكل رشفة رفعةٌ يؤدّيها \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي}؛ وأنت لا
تفعل إلا أن تُخلي الطريق.
\end{example}

\begin{example}[كؤوس الشفط والقنّينات المهروسة]\label{ex:g5:air-pressure-weather:suction}
كأس الشفط المطّاطية في خطّاف الحمّام تُضغط مستوية على البلاط،
فيُعصر \omterm{def:g2:air-around-us:air}{الهواء} من تحتها: فيدفع \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} الآن الكأسَ إلى
الجدار ولا شيء خلفها يدفع في المقابل --- مثبَّتة بالسماء نفسها.
واربط غطاء قنّينة بلاستيكية فارغة عاليًا في الجبل، ثم انزل إلى
الوادي: فتتكرمش القنّينة كأن يدًا شبحية عصرتها. \omterm{def:g2:air-around-us:air}{فالهواء} الذي
يضغط من الخارج أكثر من \omterm{def:g2:air-around-us:air}{هواء} الجبل الرقيق المحبوس في الداخل ---
فيفوز \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} وتنطوي القنّينة.
\end{example}

\begin{example}[أرقّ مع الارتفاع]\label{ex:g5:air-pressure-weather:height}
اصعد تترك جزءًا من المحيط تحتك: \omterm{def:g2:air-around-us:air}{فهواء} أقلّ فوقك يعني دفعًا
أضعف. تطقطق أذناك في تلفريك سريع بينما يتغيّر الضغط في الخارج؛
ويلهث متسلّقو الجبال في \omterm{def:g2:air-around-us:air}{هواء} صار رقيقًا. وهنا يستسلم لغز شاي
الجبل من السنة الماضية: فبضغط \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} الأقلّ على الماء يأتي
الغليان أسهل --- فالماء على القمم العظيمة يغلي دون
\qty{100}{\celsius} بكثير، أبرد من أن يُنقَع فيه شاي على أصوله.
\end{example}

\section{البارومتر والطقس}

\begin{definition}[البارومتر]\label{def:g5:air-pressure-weather:barometer}
\emph{البارومتر}\index{بارومتر} هو الجهاز الذي يقيس \omterm{def:g5:air-pressure-weather:pressure}{الضغط الجوي}
--- وهو عضو آخر في أسرة قضاتنا الأمناء، إلى جانب \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة}
والميزان. ويرتفع مؤشّره حين يضغط \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} فوقه بشدّة أكبر،
وينخفض حين يخفّ الضغط.
\end{definition}

\begin{proposition}[الضغط يتكلّم عن الغد]\label{prop:g5:air-pressure-weather:weather}
ضغط \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} لا يهدأ: \omterm{def:g2:air-around-us:air}{فالهواء} الثقيل الهابط يجلب سماء صافية
هادئة؛ \omterm{def:g2:air-around-us:air}{والهواء} الخفيف الصاعد يستدعي الغيوم والمطر. \omterm{def:g5:air-pressure-weather:barometer}{فالبارومتر}
عرّاف من نوع متواضع أمين: فالضغط \emph{الثابت أو الصاعد} يعد
بطقس صحو؛ والضغط \emph{الهابط} ينذر بغيوم قادمة؛ والهبوط
\emph{السريع} يعني أن عاصفة قد تكون في الطريق.
\end{proposition}

\begin{remark}[الحرفان م وخ على خريطة الطقس]\label{rem:g5:air-pressure-weather:map}
ستُظهر نشرة الطقس الليلة دوّامات كبيرة معلَّمة بحرف م وحرف خ:
مناطق ضغط مرتفع ومنخفض تنجرف عبر الخريطة مثل حيتان بطيئة.
\omterm{def:g2:air-around-us:air}{والهواء} ينسكب أبدًا من مناطق الضغط المرتفع الثقيلة نحو مناطق
الضغط المنخفض الخفيفة --- وذلك الانسكاب \emph{هو \omterm{def:g2:air-around-us:wind}{الريح}}. فصل
واحد وثلاثة أصدقاء قدامى مفسَّرون: تنبّؤ \omterm{def:g5:air-pressure-weather:barometer}{البارومتر}، وأصل \omterm{def:g2:air-around-us:wind}{الريح}،
وسبب معرفة أذنيك أن الطقس على وشك أن ينكسر.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:1}
كيف تبيّن تجربة كرة القدم أن \omterm{def:g2:air-around-us:air}{للهواء} وزنًا؟ وما الذي يُقارَن
بالضبط؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:2}
ما \omterm{def:g5:air-pressure-weather:pressure}{الضغط الجوي}، ومن أين يأتي؟ وفي أيّ الجهات يدفع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:3}
يضغط \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} على راحة يدك بقدر \omterm{def:g4:weight-and-mass:weight}{وزن} حقيبة سفر ممتلئة تقريبًا
--- ومع ذلك لا تحسّ بشيء. فلماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:4}
في حيلة البطاقة تتصارع دفعتان على البطاقة. سمِّهما، وقل أيّهما
يفوز.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:5}
``أنا أسحب العصير صاعدًا في الماصّة.'' صحّح هذه الجملة: ما الذي
تنزعه حقًّا، ومَن الذي يرفع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:6}
لماذا تتشبّث كأس الشفط بالبلاط؟ وما الذي عُصر خارجًا، وما الذي
يمسك الكأس؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:7}
ماذا يحدث \omterm{def:g5:air-pressure-weather:pressure}{للضغط الجوي} وأنت تصعد جبلًا، ولماذا؟ واذكر علامتين
عليه ينتبه إليهما مسافر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:air-pressure-weather:8}
ماذا يقيس \omterm{def:g5:air-pressure-weather:barometer}{البارومتر}؟ وبارومترك يهبط \omterm{def:g5:speed-distance-time:speed}{بسرعة} منذ الفطور --- فماذا
ينبغي أن تتوقّع خطط النزهة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:air-pressure-weather:9}
تكرمشت القنّينة المغلقة في النزول من الجبل. اشرح التكرمش
بالدفعات من الداخل ومن الخارج --- وتوقّع: ماذا يحدث بدل ذلك
لقنّينة أُغلقت في الوادي وحُملت \emph{صاعدة}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:air-pressure-weather:10}
من أين تأتي \omterm{def:g2:air-around-us:wind}{الريح}، وفق دوّامات م وخ على خريطة الطقس؟ اربط ذلك
بأقدم تعريف في هذا الكتاب: \omterm{def:g2:air-around-us:wind}{الريح} \omterm{def:g2:air-around-us:air}{هواء} متحرّك --- حرّكه ماذا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:air-pressure-weather:11}
يشكو متسلّقو الجبال أن الشاي على القمم العظيمة لا يُنقَع قويًّا
أبدًا وأن المعكرونة تبقى قاسية مهما طال غليانها. اجمع التفسير
كله عبر سنتين: ماذا يفعل \omterm{def:g5:air-pressure-weather:pressure}{الغلاف الجوي} الرقيق \omterm{def:g4:changes-of-state:boiling-point}{بدرجة الغليان}،
ولماذا لا يفيد إطالة الغليان؟ (قضيّتان، إحداهما من السنة
الماضية، تُغلقان هذه القضية.)
\end{exercise}
