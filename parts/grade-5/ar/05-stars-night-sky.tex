\chapter{النجوم وسماء الليل}\label{ch:g5:stars-night-sky}

بعيدًا عن مصابيح المدينة، في ليلة صافية بلا \omterm{def:g2:moon-in-the-sky:moon}{قمر}، تكون السماء
مزدحمة: آلاف الشرارات، خافتة وعنيفة، بيضاء وبرتقالية. بها اهتدى
البحّارة، وعلى مواقيتها زرع الفلّاحون، وبالعجب منها بدأ كل علم.
فما هي --- ولماذا تدور فوق رؤوسنا في نظام بديع ونحن نيام؟

\section{ما النجم}

\begin{example}[شموس لا تُحصى]\label{ex:g5:stars-night-sky:suns}
كل \omterm{def:g4:solar-system:star}{نجم} شمس: كرة هائلة من مادة متوهّجة الحرارة، تصبّ ضوءها الخاص
--- وكثير منها أجبر من شمسنا. وهي لا تبدو شرارات متواضعة إلا
لسبب واحد: البعد. فالشمس نجمنا المجاور؛ وأما البقية فمن البعد
بحيث ينفق \omterm{def:g1:light-and-shadows:source}{الضوء} نفسه، ذلك العدّاء البطل الذي يعبر الغرفة في لا
شيء من الزمن، \emph{سنوات} في الطريق من أقربها. فسماء الليل حشد
من الشموس، صغّرها البعد حتى صارت بريقًا.
\end{example}

\begin{example}[ألوان النجوم]\label{ex:g5:stars-night-sky:colors}
انظر بعناية تجد الشرارات ليست كلها بيضاء: فبعضها يشتعل برتقاليًّا
محمرًّا، وبعضها أصفر مثل شمسنا، وبعضها أزرق مبيضّ. واللون رسالة
عن الحرارة --- وهو يجري عكس صنابير الماء وعُلَب الطلاء: فالنجوم
البرتقالية \emph{أبرد} الأسرة، والزرقاء المبيضّة أسخنها، أسخن
من الشمس بكثير. وجمرة متوهّجة ونصل فولاذ محمًّى حتى البياض
يحكيان القصة نفسها هنا على الأرض: فالأسخن يتوهّج أكثر بياضًا.
\end{example}

\begin{remark}[أين تذهب النجوم في النهار]\label{rem:g5:stars-night-sky:daytime}
النجوم لا تغادر أبدًا. فهي تشرق فوقك عند الظهر كما تشرق في منتصف
الليل --- لكن سماء النهار مغمورة \omterm{def:g1:light-and-shadows:source}{بضوء} شمس يبعثره \omterm{def:g2:air-around-us:air}{الهواء}، وهجٍ
أزرق أسطع بكثير من أي شرارة. والشمعة غير مرئية إلى جانب نار
كبيرة. فإذا غربت الشمس ونزح الوهج عاد حشد السماء الدائم ظاهرًا
من جديد.
\end{remark}

\section{خرائط من نجوم}

\begin{definition}[الكوكبة]\label{def:g5:stars-night-sky:constellation}
\emph{الكوكبة}\index{كوكبة} نمط مسمّى من النجوم الساطعة ---
صيّاد، أو بجعة، أو دبّ كبير --- ابتكره مراقبو السماء قديمًا
خريطةً ومعينًا على الحفظ. ونجوم الكوكبة ليست جارات حقًّا؛ إنما
تصطفّ كما تُرى من الأرض لا غير. لكن الأنماط ثابتة: فقد تعلّم
جدّاك الأشكال نفسها بعينها.
\end{definition}

\begin{method}[كيف تجد النجم القطبي]\label{met:g5:stars-night-sky:northstar}
تقدّم السماء \omterm{def:g4:magnets-and-compass:compass}{بوصلة} مجّانية:
\begin{enumerate}
  \item جِد الدبّ الأكبر --- سبعة نجوم ساطعة، على شكل مغرفة
        بمقبض طويل، يسهل التقاطها طوال السنة؛
  \item وخذ النجمين عند الحافة الأمامية من المغرفة --- وهما
        \emph{الدليلان}؛
  \item ومُدّ الخطّ من أحدهما إلى الآخر، ماضيًا عبر السماء، نحو
        خمسة أمثال المسافة بينهما؛
  \item \omterm{def:g4:solar-system:star}{فالنجم} المتواضع الذي تحطّ عنده هو \emph{النجم
        القطبي}\index{نجم قطبي} --- أنزِل منه خيط شاقول إلى
        الأفق، فهناك الشمال.
\end{enumerate}
وتحقّق منه ببوصلتك: تجد الإبرة \omterm{def:g4:solar-system:star}{والنجم} متّفقين. وقد راهن
البحّارة بأرواحهم على ذلك الاتّفاق قرونًا.
\end{method}

\section{السماء الدائرة}

\begin{proposition}[السماء تدور حول النجم القطبي]\label{prop:g5:stars-night-sky:wheel}
خلال الليل يدور حشد النجوم كله على مهل عبر السماء في دوائر
عظيمة --- كلها حول نقطة واحدة تكاد لا تتحرّك: \omterm{met:g5:stars-night-sky:northstar}{النجم القطبي}.
ولا شيء هناك في الأعلى يتحرّك بهذا النظام حقًّا؛ إنما هي أرضنا
تدور، كما تدير الشمس عبر سماء النهار. \omterm{met:g5:stars-night-sky:northstar}{والنجم القطبي} بالكاد
يتزحزح لأنه يصادف أن يقف فوق محور دوران الأرض تمامًا تقريبًا ---
وهو الاتجاه الوحيد الذي يتركه الدوران في سلام.
\end{proposition}

\begin{example}[مدينتنا من النجوم]\label{ex:g5:stars-night-sky:milkyway}
في أشدّ الليالي ظلمة يتقوّس شريط لبني خافت عبر السماء كلها:
\emph{درب التبّانة}\index{درب التبّانة}. ويحلّ المنظار المقرّب
اللبنَ إلى حقيقته: نجوم لا تُحصى، أخفت وأشدّ ازدحامًا من أن
تفصلها العين. إنها مدينتنا من النجوم مرئيةً من الداخل ---
جزيرة مسطّحة دوّارة من الشموس أكثر من حبّات الرمل على شاطئ،
وشمسنا تسكن في ضاحية هادئة منها. وكل \omterm{def:g4:solar-system:star}{نجم} تراه بعينك جار من هذه
المدينة الواحدة.
\end{example}

\begin{remark}[واصِل النظر إلى الأعلى]\label{rem:g5:stars-night-sky:lookup}
علم الفلك أقدم العلوم، وهو الوحيد الذي يفتح مختبره مجّانًا كل
ليلة صافية. بطّانية، وموضع مظلم بعيد عن مصابيح الشوارع، وصبر
حتى تستيقظ العينان على الظلمة --- تلك قائمة العُدّة كلها. والدبّ
الأكبر \omterm{met:g5:stars-night-sky:northstar}{والنجم القطبي} \omterm{def:g4:solar-system:planet}{وكوكب} أو كوكبان \omterm{ex:g5:stars-night-sky:milkyway}{ودرب التبّانة} رحلة أولى
جميلة؛ أما قوانين \omterm{def:g1:light-and-shadows:source}{ضوء} النجوم الأعمق فستجدك مستعدًّا جيدًا، بعد
سنوات، قرب نهاية هذا المنهاج.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:1}
ما \omterm{def:g4:solar-system:star}{النجم}؟ ولماذا تبدو النجوم شرارات دقيقة مع أن كثيرًا منها
يفوق شمسنا سطوعًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:2}
أالنجوم ما زالت في السماء عند الظهر؟ ولماذا لا نستطيع رؤيتها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:3}
\omterm{def:g4:solar-system:star}{نجم} برتقالي \omterm{def:g4:solar-system:star}{ونجم} أزرق مبيضّ يشرقان جنبًا إلى جنب. فأيّهما أسخن؟
وأيّ توهّج يومي يحكي قصة الألوان نفسها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:4}
ما \omterm{def:g5:stars-night-sky:constellation}{الكوكبة}؟ وهل نجومها قريبة بعضها من بعض حقًّا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:5}
صِف خطوة خطوة كيف تجد \omterm{met:g5:stars-night-sky:northstar}{النجم القطبي} من الدبّ الأكبر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:6}
ما الذي يميّز موضع \omterm{met:g5:stars-night-sky:northstar}{النجم القطبي} في السماء الدائرة، وأيّ جهاز
يومي يحلّ محلّه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:7}
تدور النجوم عبر الليل في دوائر مرتّبة. فما الذي يدور حقًّا؟ وما
الذي يستعرضه الدوران نفسه عبر سماء النهار؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:8}
مِمّ يتبيّن أن الشريط اللبني في أشدّ السماوات ظلمة مؤلَّف، حين
يُنظر إليه بمنظار مقرّب؟ وما هو حقًّا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:9}
آلة تصوير تُركت طوال الليل مصوَّبة نحو \omterm{met:g5:stars-night-sky:northstar}{النجم القطبي} تسجّل
دوائر؛ وإذا صُوّبت نحو الأفق الشرقي بدل ذلك سجّلت خطوطًا مائلة
صاعدة. اشرح الصورتين بأرض واحدة دائرة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:10}
بحّارة غريقة في ليلة صافية ليس معها \omterm{def:g4:magnets-and-compass:compass}{بوصلة} ولا هاتف --- ولا
عجلة. اشرح كيف تسلّمها السماء اتجاهًا و--- بشيء من الصبر ---
ساعةً. (فكرتان من الفصل، وسماء دائرة واحدة.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:11}
\omterm{def:g1:light-and-shadows:source}{ضوء} \omterm{def:g4:solar-system:star}{نجم} بعيد ينفق سنوات في الطريق إلى عينك. فحين تنظر إلى ذلك
\omterm{def:g4:solar-system:star}{النجم} الليلة، أتراه كما هو \emph{الآن}؟ وإلى ماذا تنظر حقًّا ---
ولماذا يجعل هذا كل مقراب ضربًا من آلات الزمن؟
\end{exercise}
