\chapter{Warmte en isolatie}\label{ch:g5:heat-and-insulation}

Waarom voelt de metalen glijbaan ijskoud terwijl de houten bank ernaast
dat niet doet? Dat raadsel sleept al sinds het eerste jaar van dit boek
achter ons aan. Vandaag valt het --- samen met de geheimen van de
thermoskan, het dubbele raam en de opgebolde wintermus. De sleutel is
om niet langer te vragen hoe dingen \emph{aanvoelen} maar te volgen
waarheen \omterm{def:g5:heat-and-insulation:heat}{warmte} \emph{stroomt}.

\section{Warmte is onderweg}

\begin{definition}[Warmte]\label{def:g5:heat-and-insulation:heat}
\emph{Warmte}\index{warmte} is \omterm{def:g5:everyday-energy:energy}{energie} die onderweg is van een heter
ding naar een kouder. De soep geeft warmte door aan de lepel, de
radiator aan de kamer, je hand aan een sneeuwbal. Warmte is geen stof
die zich in hete dingen verstopt --- het is een \emph{stroom}, en ze
heeft altijd een richting.
\end{definition}

\begin{proposition}[Warmte stroomt bergafwaarts]\label{prop:g5:heat-and-insulation:downhill}
Vanzelf stroomt \omterm{def:g5:heat-and-insulation:heat}{warmte} altijd van het hetere lichaam naar het koudere
--- nooit andersom. De stroom loopt zolang de twee
temperaturen verschillen, en sterft weg naarmate ze elkaar naderen:
lang genoeg bij elkaar gelaten eindigen hete en koude buren op één
gedeelde \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}.
\end{proposition}

\begin{example}[De richting aflezen]\label{ex:g5:heat-and-insulation:direction}
Het theelepeltje in de thee wordt heet: er stroomde \omterm{def:g5:heat-and-insulation:heat}{warmte} van thee
$\to$ lepel. De limonade met ijs wordt koud --- pas op! Er stroomde
niets naar binnen dat ``kou'' heet: er stroomde \omterm{def:g5:heat-and-insulation:heat}{warmte} \emph{uit de
limonade het ijs in}, waardoor dat smolt. Kou is geen stroom; het is
alleen de naam die we de verliezende kant geven. Natuurkundeboeken
zeggen allemaal ``het drankje verliest \omterm{def:g5:heat-and-insulation:heat}{warmte}'', nooit ``het drankje
wint kou''.
\end{example}

\section{Snelle en trage rijstroken}

\begin{definition}[Warmtegeleiders en warmte-isolatoren]\label{def:g5:heat-and-insulation:conductors}
Stoffen verschillen enorm in hoe snel ze \omterm{def:g5:heat-and-insulation:heat}{warmte} erdoorheen laten
stromen. Een \emph{warmtegeleider}\index{warmtegeleider} is een snelle
rijstrook --- metalen bovenal. Een
\emph{warmte-isolator}\index{warmte-isolator} is een trage rijstrook:
hout, wol, plastic, veren --- en de kampioen, \emph{stilstaande
\omterm{def:g2:air-around-us:air}{lucht}}, zo opgesloten dat ze niet kan wervelen. (De woorden zijn met
opzet geleend van de \omterm{def:g4:conductors-insulators:conductor}{geleiders} en \omterm{def:g4:conductors-insulators:insulator}{isolatoren} van de elektriciteit;
de twee eigenschappen gaan vaak, maar niet altijd, samen.)
\end{definition}

\begin{example}[Het glijbaanraadsel, opgelost]\label{ex:g5:heat-and-insulation:slide}
De metalen glijbaan en de houten bank brachten de nacht door op
dezelfde kille \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} --- de \omterm{def:g3:temperature-thermometers:temperature}{thermometer} zwoer het lang geleden
al. Raak ze aan: jouw hand is warmer dan allebei, dus stroomt er
\omterm{def:g5:heat-and-insulation:heat}{warmte} van jou in elk van beide. Maar het metaal is een snelle
rijstrook --- het rukt jouw \omterm{def:g5:heat-and-insulation:heat}{warmte} in één ruk weg, en die \emph{ruk} is
wat ``ijskoud'' aanvoelt. Het hout neemt jouw \omterm{def:g5:heat-and-insulation:heat}{warmte} alleen in een
straaltje aan: nauwelijks koel. Je huid mat nooit \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} --- ze
meet de snelheid van haar eigen warmteverlies. Zaak gesloten, na vier
jaar.
\end{example}

\begin{method}[De afkoelrace]\label{met:g5:heat-and-insulation:race}
Zie isolatie winnen, met getallen:
\begin{enumerate}
  \item vul twee identieke mokken met even heet kraanwater
        (\emph{vraag een volwassene om handwarm, niet kokend});
  \item wikkel één mok dik in een wollen sjaal; laat de andere bloot;
  \item plant in elke mok een \omterm{def:g3:temperature-thermometers:temperature}{thermometer} en schrijf beide
        temperaturen elke vijf minuten in een tabel, zes keer;
  \item teken beide reeksen als twee lijnen in één grafiek,
        \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} tegen tijd.
\end{enumerate}
De lijn van de blote mok duikt omlaag; die van de ingepakte mok
helt zachtjes. Dezelfde start, dezelfde kamer --- de wol maakte de
uitgang van de \omterm{def:g5:heat-and-insulation:heat}{warmte} tot een trage rijstrook.
\end{method}

\begin{remark}[Isolatie vertraagt alleen]\label{rem:g5:heat-and-insulation:onlyslows}
Kijk waar beide krommen naartoe gaan: naar de \qty{20}{\celsius} van
de kamer. Geen sjaal, geen thermoskan, geen iglomuur houdt de
bergafwaartse stroom van \omterm{def:g5:heat-and-insulation:heat}{warmte} ooit tegen --- isolatie maakt de trage
rijstrook alleen nog trager. Daarom stelde de jas om de sneeuwpop (weet
je nog?) zijn \omterm{def:g2:ice-water-steam:meltfreeze}{smelten} wel uit maar kon hij hem niet redden: de
\omterm{def:g5:heat-and-insulation:heat}{warmte} van de zomer sijpelde toch naar binnen.
\end{remark}

\section{De kunst van lucht opsluiten}

\begin{example}[Wintertechniek]\label{ex:g5:heat-and-insulation:air}
Bijna elk warm ding dat je bezit is een machine om \omterm{def:g2:air-around-us:air}{lucht} op te
sluiten. Wol houdt je niet warm door de vezels zelf maar door de
ontelbare luchtzakjes ertussen; donzen dekbedden, fleece en de mus die
zich op een bevroren draad tot een bolletje opblaast spelen allemaal
dezelfde kaart. Huizen doen mee: dubbele ramen sluiten een laag
\omterm{def:g2:air-around-us:air}{lucht} tussen twee ruiten op, en geïsoleerde muren zijn gevuld met
pluizige, luchthamsterende vulling. Stilstaande \omterm{def:g2:air-around-us:air}{lucht} is de kampioen
trage rijstrook --- de kunst is haar \emph{stil} te houden.
\end{example}

\begin{example}[De thermoskan]\label{ex:g5:heat-and-insulation:thermos}
De thermoskan is het meesterwerk van de isolatie: een fles in een
fles, met de \omterm{def:g2:air-around-us:air}{lucht} ertussen bijna helemaal weggepompt --- en waar
bijna geen stof is, vindt de stroom van \omterm{def:g5:heat-and-insulation:heat}{warmte} bijna geen rijstrook
(een truc om te onthouden: dezelfde leegte die de bel onder de stolp
het zwijgen oplegde). Voeg glanzende wanden toe om de gloed van de
\omterm{def:g5:heat-and-insulation:heat}{warmte} terug naar binnen te kaatsen, en de warme chocolademelk haalt
nog heet de middag --- of de limonade nog koud: de thermoskan weet
niets van heet of koud, ze sluit alleen de rijstroken, in beide
richtingen.
\end{example}

\begin{remark}[Warm blijven, netjes gezegd]\label{rem:g5:heat-and-insulation:coat}
Nu sluit de oude jassenvraag voorgoed. Een jas is opgesloten
\omterm{def:g2:air-around-us:air}{lucht} rond een warmtemaker --- jou. Je lichaam stroomt de hele dag
\omterm{def:g5:heat-and-insulation:heat}{warmte} naar buiten; de jas maakt van die stroom een straaltje, en je
blijft warm op je eigen productie. Bij de sneeuwpop --- geen
warmtemaker binnenin --- vertraagt dezelfde jas alleen het straaltje
zomer dat naar binnen sijpelt. Isolatie heeft geen lievelingen: ze
vertraagt de stroom, welke kant hij ook op loopt.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:1}
Wat is \omterm{def:g5:heat-and-insulation:heat}{warmte}, en welke kant stroomt ze vanzelf op? Wanneer stopt de
stroom?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:2}
``Doe de deur dicht, je laat de kou binnen!'' Herschrijf de zin van
oma zoals een natuurkundeboek dat zou doen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:3}
Sorteer in snelle en trage rijstroken voor \omterm{def:g5:heat-and-insulation:heat}{warmte}: een koperen pan; \
een wollen sjaal; \ een stalen lepel; \ stilstaande \omterm{def:g2:air-around-us:air}{lucht}; \ een
houten lepel; \ een dekbed.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:4}
Waarom hebben pannen metalen bodems maar plastic of houten handvatten?
Antwoord met rijstroken.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:5}
Eindelijk, in je eigen woorden: waarom voelt metaal kouder dan hout
bij dezelfde \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}? Wat meet de huid werkelijk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:6}
Waarom moeten de twee mokken in de afkoelrace identiek zijn en even
heet beginnen? Welk enkele verschil onderzoekt de proef?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:7}
Lees de afkoelgrafiek af: welke \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} laat elke mok na $15$
minuten ongeveer zien? Naar welke \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} lopen beide krommen
toe, en waarom?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:8}
Waarom blaast een mus zich op een vrieskoude draad tot een bolletje
op? Noem drie menselijke uitvindingen die dezelfde kaart spelen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:9}
Dezelfde thermoskan houdt chocolademelk heet in januari en limonade
koud in juli. Leg uit hoe één apparaat allebei die klussen aankan ---
wat doet de thermoskan eigenlijk, en waarmee?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:10}
Een ijsverkoopster wikkelt haar bakken op een hete dag in dikke wollen
dekens, en een klant lacht: ``Wol is om dingen warm te houden!''
Verdedig de verkoopster met \cref{rem:g5:heat-and-insulation:coat} ---
welke kant probeert de \omterm{def:g5:heat-and-insulation:heat}{warmte} op te stromen, en wat doet de wol ermee?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:11}
Twee huizen staan in dezelfde winter: het ene met enkele ramen, het
andere met dubbele. Leg met rijstroken en opgesloten \omterm{def:g2:air-around-us:air}{lucht} uit waarom
het huis met dubbele ramen minder brandstof verstookt --- en waarom
het volgens \cref{rem:g5:heat-and-insulation:onlyslows} tóch
\emph{enige} verwarming nodig heeft.
\end{exercise}
