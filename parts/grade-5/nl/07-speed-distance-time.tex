\chapter{Snelheid: afstand en tijd}\label{ch:g5:speed-distance-time}

``Ik ben sneller!'' --- ``Nee, ik!'' Elk schoolplein beslecht dit op de
eerlijke manier: op een rij, klaar voor de start, af. Maar hoe
vergelijk je een fietser met een hardloper die gisteren op een andere
weg racete? \omterm{def:g5:speed-distance-time:speed}{Snelheid} is hoe de natuurkunde races beslecht die nooit
naast elkaar gelopen kunnen worden.

\section{Twee manieren om een race te winnen}

\begin{proposition}[De regels van de eerlijke race]\label{prop:g5:speed-distance-time:rules}
Twee eerlijke manieren om snel en langzaam te vergelijken:
\begin{enumerate}
  \item over \emph{dezelfde afstand} heeft de snelste \emph{minder
        tijd} nodig --- wie het eerst over de streep is wint de
        sprint;
  \item in \emph{dezelfde tijd} legt de snelste \emph{meer afstand}
        af --- na één minuut is de snelste zwemmer gewoon verder.
\end{enumerate}
Dezelfde afstand of dezelfde tijd: zet er één vast en vergelijk de
andere. Vergelijken met \emph{allebei} verschillend --- ``Ik heb
verder gelopen!'' ``Maar jij liep de hele middag!'' --- beslist niets.
\end{proposition}

\begin{example}[Oneerlijk en eerlijk]\label{ex:g5:speed-distance-time:fair}
Lena fietste \qty{10}{km}; Marco fietste \qty{15}{km}. Is Marco
sneller? Niet te zeggen --- misschien reed hij wel drie keer zo lang.
Maar als ze allebei precies één uur reden, wordt de vergelijking
eerlijk: dezelfde tijd, meer afstand, Marco is sneller. Dat kleine ``in
één uur'' wordt zo onze lievelingsuitdrukking.
\end{example}

\section{Snelheid: de afstand in één uur}

\begin{definition}[Snelheid]\label{def:g5:speed-distance-time:speed}
De \emph{snelheid}\index{snelheid} van een gelijkmatige reiziger is de
afstand die hij in één vaste portie tijd aflegt --- meestal één uur.
``Deze auto legt $60$ kilometer per uur af'' noemt zijn snelheid; een
stevige wandelaar haalt ongeveer $5$ kilometer per uur, een
wielrenner $40$. Voor kruipers gebruiken we vriendelijkere porties:
een slak legt ongeveer $5$ \omterm{def:g2:measuring-length:units}{meter} per uur af.
\end{definition}

\begin{method}[Je eigen snelheid meten]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item Meet of zoek een stuk van precies \qty{100}{m} (veel
        sportbanen hebben er een uitgezet);
  \item loop het in je natuurlijke tempo terwijl een vriend je klokt
        --- stel dat je er één minuut over doet;
  \item in één uur passen $60$ van zulke minuten, dus in een uur zou
        je $60 \times 100 = \qty{6000}{m}$ afleggen;
  \item in kilometers gezegd: je wandelsnelheid is $6$ kilometer per
        uur.
\end{enumerate}
Herhaal het rennend en vergelijk --- bij de meeste kinderen verdubbelt
het ongeveer.
\end{method}

\section{Afstanden voorspellen}

\begin{proposition}[Afstand uit snelheid]\label{prop:g5:speed-distance-time:distance}
Een gelijkmatige reiziger legt zijn uurafstand in elk volgend uur
opnieuw af. Dus:
\[
\text{afgelegde afstand} = \text{afstand in één uur} \times
\text{aantal uren}.
\]
Een auto die $60$ kilometer per uur rijdt legt in $3$ uur
$60 \times 3 = \qty{180}{km}$ af.
\end{proposition}

\begin{example}[Reissommen]\label{ex:g5:speed-distance-time:journeys}
De wandelaar met $4$ kilometer per uur, $2$ uur onderweg:
$4 \times 2 = \qty{8}{km}$. De fietser met $15$ kilometer per uur, $4$
uur in het zadel: $15 \times 4 = \qty{60}{km}$. Het
verkeersvliegtuig met $900$ kilometer per uur, $5$ uur in de \omterm{def:g2:air-around-us:air}{lucht}:
$900 \times 5 = \qty{4500}{km}$ --- een continent overgestoken tussen
twee maaltijden.
\end{example}

\begin{remark}[Gelijkmatig is een vereenvoudiging]\label{rem:g5:speed-distance-time:steady}
Echte reizigers zijn zelden gelijkmatig: de auto vertraagt in dorpen,
de fietser vliegt bergaf en zwoegt bergop, jij treuzelt langs de
bakker. Onze sommen doen alsof de hele reis in één trouw tempo
verloopt --- een nuttige schijn, goed genoeg om mee te plannen. Hoe je
eerlijk bent over een reis met vele tempo's --- en wat ``gemiddeld''
werkelijk betekent --- is een verhaal voor een later jaar, zodra je de
juiste wiskunde bezit.
\end{remark}

\begin{example}[Snelheden om je heen lezen]\label{ex:g5:speed-distance-time:signs}
Verkeersborden en dashboards spreken in kilometers per uur --- het
ronde bord met $50$ zegt bestuurders: leg per uur hooguit $50$
kilometer af, een tempo dat zo gekozen is dat een kind achter een bal
aan nog gespaard kan worden. Het scherm van de hogesnelheidstrein mag
met $300$ pronken; wegwijzers voor wandelaars in de bergen geven de
voorkeur aan \omterm{ex:g2:measuring-time:clock}{uren} boven kilometers --- ``meer: 2 u'' --- in het
vertrouwen dat de wandelaar zijn eigen \omterm{def:g5:speed-distance-time:speed}{snelheid} kent.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
Noem de twee eerlijke manieren om snel en langzaam te vergelijken.
Waarom bewijst ``ik heb verder gelopen dan jij'' op zichzelf niets?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
Twee zwemmers steken hetzelfde bad over; Ana heeft $30$ \omterm{def:g2:measuring-time:second}{seconden}
nodig, Bo $25$. Wie is sneller, en volgens welke regel van de
eerlijke race?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
Na één uur heeft een tractor \qty{20}{km} afgelegd en een scooter
\qty{45}{km}. Wie is sneller, en volgens welke regel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
Wat betekent ``deze trein legt $200$ kilometer per uur af''? Hoe ver
komt hij in $2$ uur? En in een half uur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
Zet op volgorde van \omterm{def:g5:speed-distance-time:speed}{snelheid}: een wandelaar ($5$ kilometer per uur); \
een duif ($80$ kilometer per uur); \ een slak ($5$ \emph{\omterm{def:g2:measuring-length:units}{meter}} per
uur); \ een stadsbus ($30$ kilometer per uur).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
Je slentert een uitgezette \qty{100}{m} in $2$ minuten. Hoeveel
\omterm{def:g2:measuring-length:units}{meter} zou je in dat tempo in een uur afleggen? Zeg je \omterm{def:g5:speed-distance-time:speed}{snelheid} in
kilometers per uur.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
Een auto houdt $3$ uur lang gelijkmatig $80$ kilometer per uur aan.
Hoe ver rijdt hij? En een fietser met $15$ kilometer per uur gedurende
diezelfde $3$ uur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
Hoe herken je in de reisgrafiek van dit hoofdstuk in één oogopslag de
snelste reiziger? Lees eruit af: hoe ver is de fietser na $2$
uur gekomen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
Een veerboot legt $30$ kilometer per uur af. De overtocht is
\qty{90}{km}. Gebruik \cref{prop:g5:speed-distance-time:distance}
achterstevoren --- probeer de \omterm{ex:g2:measuring-time:clock}{uren} één voor één --- hoe lang duurt de
overtocht?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
Twee dorpen liggen \qty{60}{km} uit elkaar. Mara fietst vanuit het ene
met $20$ kilometer per uur; op hetzelfde moment fietst Jon vanuit het
andere haar tegemoet met $10$ kilometer per uur. Met hoeveel kilometer
krimpt de opening tussen hen per uur? Na hoeveel uur ontmoeten ze
elkaar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
De oude fabel, met getallen: de race is \qty{120}{m} lang. De haas
rent $60$ \omterm{def:g2:measuring-length:units}{meter} per minuut --- maar na één minuut rennen stopt hij
voor een dutje van $70$ minuten, en rent daarna door. De schildpad
sjokt $2$ \omterm{def:g2:measuring-length:units}{meter} per minuut en stopt nooit. Waar is de haas als hij
aan zijn dutje begint? Op welk \omterm{def:g2:measuring-time:duration}{tijdstip} komt elk van beiden over de
finish? Wie wint?
\end{exercise}
