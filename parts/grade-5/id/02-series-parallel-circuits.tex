\chapter{Rangkaian Seri dan Paralel}\label{ch:g5:series-parallel-circuits}

Ketika satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} mati, seluruh untaian lampu pesta menjadi gelap ---
kutukan lingkar tunggal. Namun ketika sebuah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} putus di dapurmu,
sisa rumahmu tetap bercahaya. Ada orang yang memasang kabel rumahmu
dengan cara yang lebih cerdik. Hari ini kita mempelajarinya: cara
percabangan.

\section{Dua cara memasang kabel}

Kamu sudah mengenal cara yang pertama: \emph{\omterm{def:g4:series-circuits:series}{terangkai seri}}, semua
pemain dirangkai pada satu lingkar, manik pada satu kalung. Inilah
cara yang kedua.

\begin{definition}[Terangkai paralel]\label{def:g5:series-parallel-circuits:parallel}
Para pemain \emph{terangkai paralel}\index{rangkaian paralel} ketika
rangkaiannya terbelah menjadi beberapa \emph{cabang}\index{cabang}
yang terpisah --- satu pemain untuk tiap cabang --- yang memisah lalu
bergabung kembali. Titik tempat jalan memisah atau bergabung kembali
disebut \emph{titik cabang}\index{titik cabang}. Setiap cabang adalah
jalan yang lengkap dari satu kutub \omterm{def:g3:first-electric-circuit:battery}{baterai} ke kutub yang lain.
\end{definition}

\begin{method}[Membangunnya]\label{met:g5:series-parallel-circuits:build}
Dengan satu \omterm{def:g3:first-electric-circuit:battery}{baterai}, dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan kawat:
\begin{enumerate}
  \item pasanglah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} pertama ke \omterm{def:g3:first-electric-circuit:battery}{baterai}, sendirian, sebagai
        lingkar biasa --- inilah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} pertama;
  \item sekarang pasanglah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} kedua \emph{tepat melintang} pada
        \omterm{def:g3:first-electric-circuit:bulb}{bohlam} pertama: satu kawat dari tiap sisi \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 ke sisi
        \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 yang bersesuaian --- inilah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} kedua;
  \item telusurilah tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} dengan jarimu: dari $+$, setiap jalan
        harus melewati tepat satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} sebelum mencapai $-$.
\end{enumerate}
Kedua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} menyala --- dan keduanya bersinar seterang satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}
yang menyala sendirian.
\end{method}

\section{Aturan percabangan}

\begin{proposition}[Aturan rangkaian paralel]\label{prop:g5:series-parallel-circuits:rules}
Dengan tiap pemain pada \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sendiri:
\begin{enumerate}
  \item \emph{kemandirian}: celah pada satu \omterm{def:g5:series-parallel-circuits:parallel}{cabang} hanya
        menghentikan \omterm{def:g5:series-parallel-circuits:parallel}{cabang} itu --- lepaskan satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan yang
        lain tetap menyala;
  \item \emph{terang penuh}: tiap \omterm{def:g3:first-electric-circuit:bulb}{bohlam} bersinar seolah-olah ia
        sendirian --- tidak ada pembagian dorongan \omterm{def:g3:first-electric-circuit:battery}{baterai} di antara
        \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya};
  \item \emph{pos komando}: \omterm{ex:g3:first-electric-circuit:switch}{sakelar} di dalam sebuah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} hanya
        memerintah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} itu; \omterm{ex:g3:first-electric-circuit:switch}{sakelar} pada jalan bersama, sebelum
        percabangannya, memerintah semuanya. (Harganya: \omterm{def:g3:first-electric-circuit:battery}{baterainya}
        memberi makan setiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} sekaligus, sehingga ia lebih
        cepat habis.)
\end{enumerate}
\end{proposition}

\begin{example}[Uji melepas bohlam, diulang]\label{ex:g5:series-parallel-circuits:test}
Lakukan pada pasangan paralel ini persis apa yang dahulu membunuh
pasangan seri: lepaskan \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2. \omterm{def:g3:first-electric-circuit:bulb}{Bohlam} 1 bahkan tidak berkedip ---
\omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sendiri, jalannya sendiri yang lengkap, tidak tersentuh.
Pasang kembali, lalu lepaskan \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 menyala tenang terus.
Kutukan lingkar tunggal terangkat: pada \omterm{def:g5:series-parallel-circuits:parallel}{rangkaian paralel}, setiap
pemain mengurus \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sendiri.
\end{example}

\begin{example}[Rumahmu adalah rangkaian paralel]\label{ex:g5:series-parallel-circuits:house}
Setiap lampu, stopkontak, dan peralatan di rumahmu menunggangi
\omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sendiri pada satu \omterm{def:g5:series-parallel-circuits:parallel}{rangkaian paralel} yang besar. Itulah
sebabnya \omterm{def:g3:first-electric-circuit:bulb}{bohlam} dapur dapat putus sementara ruang keluarga tetap
membaca tanpa terganggu; mengapa \omterm{ex:g3:first-electric-circuit:switch}{sakelar} tiap ruangan hanya memerintah
lampunya sendiri; dan mengapa satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} utama --- di kotak sekring
dekat meteran --- dapat memutus seluruh rumah sekaligus untuk
perbaikan. Lampu pesta zaman sekarang ikut bergabung: \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}
dipasang sedemikian rupa sehingga satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} mati tidak lagi
menggelapkan untaiannya.
\end{example}

\begin{remark}[Seri dan paralel, berdampingan]\label{rem:g5:series-parallel-circuits:compare}
Satu celah: seri --- semuanya gelap; paralel --- satu \omterm{def:g5:series-parallel-circuits:parallel}{cabang} gelap.
\omterm{def:g3:first-electric-circuit:bulb}{Bohlam} lebih banyak: seri --- semuanya lebih redup; paralel ---
semuanya terang penuh, tetapi \omterm{def:g3:first-electric-circuit:battery}{baterainya} lebih cepat habis. Satu
\omterm{ex:g3:first-electric-circuit:switch}{sakelar} untuk semuanya: seri --- di mana pun; paralel --- hanya
sebelum percabangannya. Tidak ada cara yang ``lebih baik'': senter
dengan senang hati memakai seri (satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar}), rumah mau
tidak mau memakai paralel. Pertanyaan pertama seorang tukang listrik
tentang rangkaian apa pun sederhana saja: \emph{di mana jalannya
memisah?}
\end{remark}

\section{Membaca rangkaian}

\begin{method}[Menelusuri jalannya]\label{met:g5:series-parallel-circuits:follow}
Untuk meramalkan nasib \omterm{def:g3:first-electric-circuit:bulb}{bohlam} mana pun pada gambar rangkaian:
\begin{enumerate}
  \item letakkan jarimu pada kutub $+$ \omterm{def:g3:first-electric-circuit:battery}{baterai} lalu jalanilah setiap
        jalan yang mungkin sampai ke $-$, sambil mencatat pemain yang
        dilewati tiap jalan;
  \item sebuah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} menyala kalau ada setidaknya satu jalan tanpa
        putus yang melewatinya;
  \item pemain yang berbagi setiap jalan \omterm{def:g4:series-circuits:series}{terangkai seri} satu sama
        lain; pemain pada percabangan yang berbeda \omterm{def:g5:series-parallel-circuits:parallel}{terangkai paralel}.
\end{enumerate}
Jalanilah jalannya sebelum kamu meramal --- menebak dari bentuk
gambarnya saja adalah cara rangkaian menipu orang.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:1}
Apa itu \omterm{def:g5:series-parallel-circuits:parallel}{cabang}? Apa itu \omterm{def:g5:series-parallel-circuits:parallel}{titik cabang}? Pada \omterm{def:g5:series-parallel-circuits:parallel}{rangkaian paralel} dengan
dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, ada berapa \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang membawa \omterm{def:g3:first-electric-circuit:bulb}{bohlam}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:2}
Sebutkan ketiga aturan \omterm{def:g5:series-parallel-circuits:parallel}{rangkaian paralel}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:3}
Dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} \omterm{def:g5:series-parallel-circuits:parallel}{terangkai paralel}: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 dilepaskan. Apa yang dilakukan
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2? Dan pada pasangan seri --- apa yang dilakukannya di sana,
dan mengapa berbeda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:4}
Dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang serupa: pasangan A \omterm{def:g4:series-circuits:series}{terangkai seri}, pasangan B
\omterm{def:g5:series-parallel-circuits:parallel}{terangkai paralel}, dengan \omterm{def:g3:first-electric-circuit:battery}{baterai} yang sama. Pasangan mana yang tiap
\omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} bersinar lebih terang? Yang mana yang lebih cepat
menghabiskan \omterm{def:g3:first-electric-circuit:battery}{baterainya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:5}
Pada gambar dengan dua \omterm{ex:g3:first-electric-circuit:switch}{sakelar}, apa yang dilakukan tiap \omterm{def:g3:first-electric-circuit:bulb}{bohlam} ketika:
kedua \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} tertutup? \ \omterm{ex:g3:first-electric-circuit:switch}{sakelar} utamanya terbuka? \ \omterm{ex:g3:first-electric-circuit:switch}{sakelar}
utamanya tertutup dan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} terbuka?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:6}
Mengapa rumah dipasangi kabel secara paralel dan bukan secara seri?
Sebutkan dua kenyamanan sehari-hari yang bergantung padanya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:7}
Di bagian mana pada rangkaian rumah sebuah \omterm{ex:g3:first-electric-circuit:switch}{sakelar} memerintah semuanya
sekaligus? Apa nama sehari-hari dan kegunaan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:8}
Lampu meja dan stopkontak dindingnya: \omterm{def:g4:series-circuits:series}{terangkai seri} atau paralel
terhadap lampu langit-langit? Dari mana kamu tahu dari kehidupan
sehari-hari?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:9}
Gambarlah (dengan lambang) sebuah rangkaian untuk lorong: satu
\omterm{def:g3:first-electric-circuit:battery}{baterai}, satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan kenyamanan ini --- \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} harus dapat
dinyalakan dan dimatikan dari \omterm{ex:g3:first-electric-circuit:switch}{sakelar} di \emph{salah satu} ujung
lorong\dots\ Cobalah dengan jujur; kalau rancanganmu keras kepala
gagal, jelaskan apa yang sulit. (\omterm{ex:g3:first-electric-circuit:switch}{Sakelar} dua arah yang sebenarnya
memerlukan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang lebih cerdik daripada milik kita --- percobaan
itu mengajarkan lebih banyak daripada jawabannya.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:10}
Seuntai lampu pesta zaman sekarang tetap menyala ketika satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}
mati --- tetapi kotaknya memperingatkan: ``Jangan menyalakan
untaiannya kalau lebih dari tiga \omterm{def:g3:first-electric-circuit:bulb}{bohlam} dilepas.'' Dengan aturan
percabangan, jelaskan kenyamanannya dan, yang lebih sulit, satu alasan
yang masuk akal untuk peringatan itu. (Pikirkan apa yang diterima tiap
\omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang tersisa dari \omterm{def:g3:first-electric-circuit:battery}{baterainya}, dan usaha \omterm{def:g3:first-electric-circuit:battery}{baterainya} sendiri.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:11}
Pada gambar dua \omterm{ex:g3:first-electric-circuit:switch}{sakelar}, seorang tukang listrik ingin menambahkan satu
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} lagi yang menyala \emph{hanya} ketika \omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}
tertutup, dan gelap dalam keadaan lain --- bahkan ketika \omterm{ex:g3:first-electric-circuit:switch}{sakelar}
utamanya tertutup. Pada ruas jalan yang mana \omterm{def:g3:first-electric-circuit:bulb}{bohlam} baru itu harus
dipasang, dan apakah ia lalu \omterm{def:g4:series-circuits:series}{terangkai seri} atau paralel terhadap
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2?
\end{exercise}
