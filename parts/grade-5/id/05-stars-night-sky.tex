\chapter{Bintang dan Langit Malam}\label{ch:g5:stars-night-sky}

Jauh dari lampu kota, pada malam cerah tanpa bulan, langit menjadi
sesak: ribuan percikan, samar dan garang, putih dan jingga. Pelaut
berkemudi dengannya, petani menanam dengannya, dan setiap ilmu bermula
dari rasa takjub kepadanya. Apa sebenarnya mereka --- dan mengapa
mereka berputar begitu rapi di atas kepala kita selagi kita tidur?

\section{Apa itu bintang}

\begin{example}[Matahari yang tak terhitung]\label{ex:g5:stars-night-sky:suns}
Setiap \omterm{def:g4:solar-system:star}{bintang} adalah sebuah matahari: bola raksasa dari zat yang
berpijar panas, mencurahkan \omterm{def:g1:light-and-shadows:source}{cahayanya} sendiri --- banyak di antaranya
lebih perkasa daripada Matahari kita. Mereka tampak sebagai percikan
yang sederhana hanya karena satu hal: jarak. Matahari adalah \omterm{def:g4:solar-system:star}{bintang}
tetangga sebelah kita; yang lain begitu jauh sehingga \omterm{def:g1:light-and-shadows:source}{cahaya} pun, sang
juara lari yang menyeberangi ruangan tanpa memakan waktu sama sekali,
menghabiskan \emph{bertahun-tahun} di jalan dari yang terdekat di
antaranya. Langit malam adalah kerumunan matahari, yang disusutkan
jarak menjadi kerlip.
\end{example}

\begin{example}[Warna bintang]\label{ex:g5:stars-night-sky:colors}
Amati baik-baik dan percikan itu tidak semuanya putih: ada yang
menyala jingga kemerahan, ada yang kuning seperti Matahari kita, ada
yang putih kebiruan. Warnanya adalah pesan tentang panas --- dan ia
berjalan berlawanan dengan keran dan kaleng cat: \omterm{def:g4:solar-system:star}{bintang} yang jingga
adalah yang \emph{paling sejuk} di keluarga itu, dan yang putih
kebiruan yang paling panas, jauh lebih panas daripada Matahari. Bara
yang berpijar dan bilah baja yang membara putih menceritakan kisah
yang sama di Bumi ini: yang lebih panas berpijar lebih putih.
\end{example}

\begin{remark}[Ke mana bintang pergi pada siang hari]\label{rem:g5:stars-night-sky:daytime}
\omterm{def:g4:solar-system:star}{Bintang} tidak pernah pergi. Mereka bersinar di atasmu pada tengah hari
persis seperti pada tengah malam --- tetapi langit siang dibanjiri
\omterm{def:g1:light-and-shadows:source}{cahaya} matahari yang dihamburkan \omterm{def:g2:air-around-us:air}{udara}, silau biru yang jauh lebih
terang daripada percikan mana pun. Sebatang lilin tidak terlihat di
sebelah api unggun. Ketika \omterm{def:g1:day-and-night:daynight}{Matahari terbenam} dan silaunya surut,
kerumunan tetap milik langit itu tinggal menjadi terlihat lagi.
\end{remark}

\section{Peta yang terbuat dari bintang}

\begin{definition}[Rasi bintang]\label{def:g5:stars-night-sky:constellation}
Sebuah \emph{rasi bintang}\index{rasi bintang} adalah pola \omterm{def:g4:solar-system:star}{bintang}
terang yang diberi nama --- seorang pemburu, seekor angsa, seekor
beruang besar --- yang diciptakan para pengamat langit zaman dahulu
sebagai peta sekaligus alat bantu ingatan. \omterm{def:g4:solar-system:star}{Bintang} sebuah rasi
sebenarnya bukan tetangga; mereka hanya berbaris jika dilihat dari
Bumi. Tetapi polanya mantap: kakek-nenekmu mempelajari bentuk yang
persis sama.
\end{definition}

\begin{method}[Menemukan Bintang Utara]\label{met:g5:stars-night-sky:northstar}
Langit menawarkan \omterm{def:g4:magnets-and-compass:compass}{kompas} cuma-cuma:
\begin{enumerate}
  \item temukan rasi Biduk --- tujuh \omterm{def:g4:solar-system:star}{bintang} terang berbentuk panci
        bergagang panjang, mudah dikenali sepanjang tahun;
  \item ambillah dua \omterm{def:g4:solar-system:star}{bintang} di tepi depan pancinya ---
        \emph{sang penunjuk};
  \item panjangkan garis dari yang satu ke yang lain, terus
        menyeberangi langit, kira-kira lima kali jarak keduanya;
  \item \omterm{def:g4:solar-system:star}{bintang} sederhana yang kamu temui itulah
        \emph{Bintang Utara}\index{Bintang Utara} --- turunkan garis
        tegak lurus darinya sampai ke cakrawala, dan di situlah utara.
\end{enumerate}
Periksalah dengan kompasmu: jarum dan \omterm{def:g4:solar-system:star}{bintang} sependapat. Pelaut
mempertaruhkan nyawanya pada kesepakatan itu selama berabad-abad.
\end{method}

\section{Langit yang berputar}

\begin{proposition}[Langit berputar mengelilingi Bintang Utara]\label{prop:g5:stars-night-sky:wheel}
Sepanjang malam, seluruh kerumunan \omterm{def:g4:solar-system:star}{bintang} berputar pelan menyeberangi
langit dalam lingkaran besar --- semuanya mengelilingi satu titik yang
nyaris tidak bergerak: \omterm{met:g5:stars-night-sky:northstar}{Bintang Utara}. Tidak ada apa pun di sana yang
sungguh-sungguh bergerak serapi itu; Bumi kita sendirilah yang
berputar, sebagaimana ia memutar Matahari menyeberangi langit siang.
\omterm{met:g5:stars-night-sky:northstar}{Bintang Utara} nyaris tidak bergeming karena kebetulan ia berdiri
hampir tepat di atas sumbu putar Bumi --- satu-satunya arah yang
dibiarkan tenang oleh putaran itu.
\end{proposition}

\begin{example}[Kota bintang kita]\label{ex:g5:stars-night-sky:milkyway}
Pada malam yang paling gelap, pita susu yang samar melengkung
menyeberangi seluruh langit: itulah
\emph{Bima Sakti}\index{Bima Sakti}. Teropong melarutkan susunya
menjadi kebenaran: \omterm{def:g4:solar-system:star}{bintang} yang tak terhitung banyaknya, terlalu samar
dan terlalu berjejal untuk dipisahkan mata. Itulah kota asal \omterm{def:g4:solar-system:star}{bintang}
kita, dilihat dari dalam --- pulau matahari yang pipih dan berpusar,
berisi lebih banyak matahari daripada butir pasir di sebuah pantai,
dengan Matahari kita tinggal di pinggiran yang sepi. Setiap \omterm{def:g4:solar-system:star}{bintang}
yang dapat kamu lihat dengan mata adalah tetangga dari satu kota ini.
\end{example}

\begin{remark}[Teruslah menengadah]\label{rem:g5:stars-night-sky:lookup}
Astronomi adalah ilmu yang tertua, dan satu-satunya ilmu yang
laboratoriumnya buka tanpa bayaran setiap malam yang cerah. Sehelai
selimut, satu tempat gelap jauh dari lampu jalan, kesabaran selagi
mata terbiasa dengan gelap --- itulah seluruh daftar perlengkapannya.
Biduk, \omterm{met:g5:stars-night-sky:northstar}{Bintang Utara}, satu dua \omterm{def:g4:solar-system:planet}{planet}, dan \omterm{ex:g5:stars-night-sky:milkyway}{Bima Sakti} sudah menjadi
ekspedisi pertama yang baik; hukum yang lebih dalam tentang \omterm{def:g1:light-and-shadows:source}{cahaya}
\omterm{def:g4:solar-system:star}{bintang} akan menemukanmu dalam keadaan siap, bertahun-tahun dari
sekarang, menjelang akhir pelajaran ini.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:1}
Apa itu \omterm{def:g4:solar-system:star}{bintang}? Mengapa \omterm{def:g4:solar-system:star}{bintang} tampak sebagai percikan mungil
padahal banyak di antaranya mengalahkan \omterm{def:g1:light-and-shadows:source}{cahaya} Matahari kita?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:2}
Apakah \omterm{def:g4:solar-system:star}{bintang} masih ada di langit pada tengah hari? Mengapa kita
tidak dapat melihatnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:3}
Sebuah \omterm{def:g4:solar-system:star}{bintang} jingga dan sebuah \omterm{def:g4:solar-system:star}{bintang} putih kebiruan bersinar
berdampingan. Mana yang lebih panas? Pijar sehari-hari apa yang
menceritakan kisah warna yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:4}
Apa itu \omterm{def:g5:stars-night-sky:constellation}{rasi bintang}? Apakah \omterm{def:g4:solar-system:star}{bintangnya} sungguh-sungguh berdekatan
satu sama lain?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:5}
Ceritakan, langkah demi langkah, cara menemukan \omterm{met:g5:stars-night-sky:northstar}{Bintang Utara} dari
rasi Biduk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:6}
Apa yang istimewa dari tempat \omterm{met:g5:stars-night-sky:northstar}{Bintang Utara} di langit yang berputar,
dan alat sehari-hari apa yang digantikannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:7}
\omterm{def:g4:solar-system:star}{Bintang} berputar menyeberangi malam dalam lingkaran yang rapi. Apa
yang sebenarnya sedang berputar? Apa lagi yang diarak putaran yang
sama menyeberangi langit siang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:8}
Ternyata apa pita susu yang melintasi langit paling gelap itu, dilihat
lewat teropong? Sebenarnya ia apa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:9}
Kalau dibiarkan semalaman, kamera yang diarahkan ke \omterm{met:g5:stars-night-sky:northstar}{Bintang Utara}
merekam lingkaran; kalau diarahkan ke cakrawala timur, ia merekam
goresan menyerong yang menaik. Jelaskan kedua gambar itu dengan satu
Bumi yang berputar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:10}
Seorang pelaut yang kapalnya karam pada malam yang cerah tidak punya
\omterm{def:g4:magnets-and-compass:compass}{kompas}, tidak punya telepon --- dan tidak terburu-buru. Jelaskan
bagaimana langit menyerahkan kepadanya sebuah arah sekaligus, dengan
kesabaran, sebuah \omterm{ex:g2:measuring-time:clock}{jam}. (Dua gagasan bab, satu langit yang berputar.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:11}
\omterm{def:g1:light-and-shadows:source}{Cahaya} sebuah \omterm{def:g4:solar-system:star}{bintang} yang jauh menghabiskan bertahun-tahun di jalan
menuju matamu. Ketika kamu memandang \omterm{def:g4:solar-system:star}{bintang} itu malam ini, apakah
kamu melihatnya sebagaimana ia \emph{sekarang}? Apa yang sebenarnya
kamu pandang --- dan mengapa hal ini membuat setiap teleskop menjadi
semacam mesin waktu?
\end{exercise}
