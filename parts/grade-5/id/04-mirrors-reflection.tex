\chapter{Cermin dan Pemantulan}\label{ch:g5:mirrors-reflection}

\omterm{def:g1:light-and-shadows:source}{Cahaya}, seperti kita buktikan, tidak pernah membelok mengitari
tikungan --- namun setiap pagi ada salinan dirimu yang memandang
keluar dari dinding kamar mandi, dan tahun lalu kita menjanjikan
tipuan untuk melihat di balik tikungan. Tipuannya bukan
membengkokkan jalannya melainkan \emph{melipatnya}. Segalanya melipat
pada satu permukaan yang luar biasa: \omterm{def:g5:mirrors-reflection:reflection}{cermin}.

\section{Pantulan}

\begin{definition}[Pemantulan]\label{def:g5:mirrors-reflection:reflection}
Ketika \omterm{def:g1:light-and-shadows:source}{cahaya} menumbuk sebuah permukaan lalu melenting kembali
alih-alih menembusnya atau mati di sana, kita menyebutnya
\emph{pemantulan}\index{pemantulan}: \omterm{def:g1:light-and-shadows:source}{cahayanya} dipantulkan. Sebuah
\emph{cermin}\index{cermin} adalah permukaan yang begitu halus
sehingga memantulkan \omterm{def:g1:light-and-shadows:source}{cahaya} dengan cara yang tertib sempurna --- logam
yang dipoles di belakang kaca, air yang tenang, sendok yang berkilau.
\end{definition}

\begin{proposition}[Aturan pantulan]\label{prop:g5:mirrors-reflection:bounce}
\omterm{def:g4:light-straight-lines:ray}{Sinar cahaya} memantul dari \omterm{def:g5:mirrors-reflection:reflection}{cermin} seperti bola sempurna memantul dari
dinding: ia pergi persis securam ia datang, di sisi yang lain. Datang
lurus --- kembali lurus. Datang menyerong --- pergi menyerong dengan
kecondongan yang sepadan. Kedua sudutnya, ketika datang dan ketika
pergi, selalu sama besar.
\end{proposition}

\begin{example}[Membidikkan pantulan]\label{ex:g5:mirrors-reflection:aiming}
Karena pantulannya begitu tertib, ia dapat \emph{dibidikkan}: dengan
\omterm{def:g5:mirrors-reflection:reflection}{cermin} kecil di bawah sinar matahari kamu dapat melemparkan bintik
\omterm{def:g1:light-and-shadows:source}{cahaya} ke dinding mana pun yang kamu pilih --- miringkan \omterm{def:g5:mirrors-reflection:reflection}{cerminnya},
dan bintiknya menurut. Pelaut dan pendaki yang dalam kesulitan memakai
persis cara ini: kilatan \omterm{def:g5:mirrors-reflection:reflection}{cermin} yang dibidikkan ke kapal atau
helikopter yang jauh adalah isyarat yang terlihat sampai
berkilo-kilometer, seluruhnya bertenaga Matahari.
\end{example}

\begin{example}[Mengapa dinding bukan cermin]\label{ex:g5:mirrors-reflection:walls}
Dinding putih memantulkan banyak \omterm{def:g1:light-and-shadows:source}{cahaya} --- kalau tidak, ruangannya
akan suram --- jadi mengapa ia tidak menampilkan \omterm{prop:g5:mirrors-reflection:image}{bayangan}? Dari dekat,
permukaannya adalah bentang alam berbukit-bukit mungil: tiap sinar
yang datang menuruti aturan pantulan pada lerengnya sendiri yang
kecil, dan pasukan sinar yang tertib itu terhambur ke segala arah.
Kilap \omterm{def:g5:mirrors-reflection:reflection}{cermin} menjaga lerengnya tetap datar, pasukannya tetap seirama,
dan gambarnya selamat melewati pantulan. Ketertiban, bukan kilap saja,
itulah yang membuat sebuah \omterm{prop:g5:mirrors-reflection:image}{bayangan}.
\end{example}

\section{Sosok di balik kaca}

\begin{proposition}[Aturan bayangan]\label{prop:g5:mirrors-reflection:image}
\omterm{def:g5:mirrors-reflection:reflection}{Cermin} datar menampilkan setiap benda sebagai
\emph{bayangan}\index{bayangan} yang berdiri \emph{di belakang}
kacanya, tepat sejauh benda itu berdiri di depannya, sama besar
dengannya, titik demi titik. Mundurlah, dan bayanganmu ikut mundur;
dunia \omterm{def:g5:mirrors-reflection:reflection}{cermin} adalah salinan dunia kita yang setia dan terlipat.
\end{proposition}

\begin{example}[Mengapa bayangannya tinggal di belakang]\label{ex:g5:mirrors-reflection:why}
Matamu berjiwa sederhana: ia menganggap setiap sinar berjalan lurus.
Sinar yang memantul itu memang datang dari lilinnya lewat \omterm{def:g5:mirrors-reflection:reflection}{cermin} ---
tetapi mata memanjangkannya \emph{mundur menembus kacanya}, sampai ke
tempat sinar itu seolah-olah bermula. Semua sinar yang memantul
sepakat tentang titik awal hantu itu: sinar sebanyak satu lilin, satu
lilin hantu yang tajam. Tentu saja tidak ada apa pun yang berdiri di
balik dinding --- \omterm{prop:g5:mirrors-reflection:image}{bayangan} adalah tipuan \omterm{def:g1:light-and-shadows:source}{cahaya} yang paling
meyakinkan.
\end{example}

\begin{example}[Dunia yang tertukar]\label{ex:g5:mirrors-reflection:swap}
Hadapkan selembar tulisan ke \omterm{def:g5:mirrors-reflection:reflection}{cermin}: hurufnya kembali terbalik, kiri
dan kanan bertukar. Itulah sebabnya kata AMBULANS dicat
\omterm{def:g5:mirrors-reflection:reflection}{terbalik-cermin} di bagian depan mobil ambulans: pengemudi di depannya,
yang melirik ke kaca spionnya, melihatnya berbalik menjadi tulisan
biasa --- lalu menepi. Dua \omterm{def:g5:mirrors-reflection:reflection}{cermin} dapat membatalkan pertukaran itu:
pantulan ganda membalik pembalikannya.
\end{example}

\section{Melipat jalannya dengan sengaja}

\begin{method}[Membuat periskop]\label{met:g5:mirrors-reflection:periscope}
Peneropong tikungan yang dijanjikan, kini diserahkan:
\begin{enumerate}
  \item ambillah kotak yang tinggi dan sempit; guntinglah jendela di
        dekat puncak salah satu sisinya dan satu lagi di dekat dasar
        sisi yang berseberangan;
  \item pasanglah \omterm{def:g5:mirrors-reflection:reflection}{cermin} kecil di dalam, di belakang tiap jendela,
        masing-masing dimiringkan setengah sudut siku-siku, dan kedua
        kemiringannya sejajar satu sama lain;
  \item arahkan jendela atasnya melewati dinding (atau mengitari
        tikungan), lalu tengoklah ke jendela bawahnya.
\end{enumerate}
Jalan \omterm{def:g1:light-and-shadows:source}{cahayanya} melipat dua kali: turun di dalam kotak di antara kedua
\omterm{def:g5:mirrors-reflection:reflection}{cerminnya}, lalu keluar ke matamu. Awak kapal selam mengamati ombak
lewat persis alat ini --- yang dibuat panjang dan kedap air.
\end{method}

\begin{remark}[Sendok rumah hantu]\label{rem:g5:mirrors-reflection:curved}
Tengoklah ke dalam cekungan sendok yang dipoles: kamu tergantung
terbalik. Baliklah sendoknya: kamu tegak lagi, tetapi terentang.
\omterm{def:g5:mirrors-reflection:reflection}{Cermin} melengkung tetap menuruti aturan pantulan di setiap titik ---
tetapi lerengnya ikut berputar bersama lengkungnya, sehingga pantulan
yang tertib itu berkumpul atau memencar, dan \omterm{prop:g5:mirrors-reflection:image}{bayangannya} menyusut,
membesar, atau berjungkir balik. \omterm{def:g5:mirrors-reflection:reflection}{Cermin} melengkung juga memusatkan
\omterm{def:g1:light-and-shadows:source}{cahaya}: sebuah piring melengkung berisi sinar matahari dapat memasak
sepanci --- ingat kompor surya pada bab \omterm{def:g5:everyday-energy:energy}{energi}. Hukumnya yang persis
milik tahun yang lebih tinggi; sendok itu pratinjaunya yang jujur.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:1}
Nyatakan aturan pantulan. Apa yang dilakukan sinar yang datang tegak
lurus?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:2}
Dinding putih memantulkan banyak \omterm{def:g1:light-and-shadows:source}{cahaya}. Mengapa ia tidak menampilkan
\omterm{prop:g5:mirrors-reflection:image}{bayangan}, sementara \omterm{def:g5:mirrors-reflection:reflection}{cermin} yang dipoles menampilkannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:3}
Kamu berdiri \qty{2}{m} di depan \omterm{def:g5:mirrors-reflection:reflection}{cermin} datar. Di mana bayanganmu, dan
seberapa jauh jaraknya darimu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:4}
Kamu berjalan mendekati bayanganmu di \omterm{def:g5:mirrors-reflection:reflection}{cermin} dengan langkah pelan. Apa
yang dilakukan \omterm{prop:g5:mirrors-reflection:image}{bayangannya}, dan secepat apa jarak di antara kalian
menutup?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:5}
Mengapa kata AMBULANS dicat terbalik di bagian depan mobil ambulans?
Bagi siapa tulisan itu terbaca benar, dan lewat apa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:6}
Pada periskop, berapa kali jalan \omterm{def:g1:light-and-shadows:source}{cahayanya} melipat, dan apa yang
dilakukan tiap lipatan? Pada kemiringan berapa \omterm{def:g5:mirrors-reflection:reflection}{cerminnya} berdiri?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:7}
Bagaimana seorang pendaki berbekal \omterm{def:g5:mirrors-reflection:reflection}{cermin} saku dapat memberi isyarat
kepada helikopter yang jauh? Dua gagasan bab yang mana yang sedang
bekerja --- sebutkan aturan yang membidikkan kilatannya dan penyimpan
yang menenagainya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:mirrors-reflection:8}
Air yang tenang mencerminkan pegunungan; \omterm{def:g2:air-around-us:wind}{angin} mengeriputkan danaunya
dan gambarnya pecah menjadi kerlip. Jelaskan kedua keadaan itu dengan
\cref{ex:g5:mirrors-reflection:walls}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:mirrors-reflection:9}
Mata ``memanjangkan sinar ke belakang''. Pakailah hal itu untuk
menjelaskan mengapa bayanganmu berdiri \emph{di belakang} dinding
tempat \omterm{def:g5:mirrors-reflection:reflection}{cermin} itu tergantung --- di tempat yang pasti tidak ditempati
siapa pun. Semua sinar yang memantul dari hidungmu sepakat tentang apa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:mirrors-reflection:10}
Sebuah toko ingin agar pembeli di pintu dapat melihat mengitari
tikungan buta ke dalam lorong raknya. Satu \omterm{def:g5:mirrors-reflection:reflection}{cermin} datar harus dipasang
di tempat kedua lorong bertemu. Bagaimana \omterm{def:g5:mirrors-reflection:reflection}{cermin} itu harus
dimiringkan, dan mengapa pemasangan yang sama membuat lorong raknya
dapat melihat pintunya? (Jalan \omterm{def:g1:light-and-shadows:source}{cahaya} bekerja dua arah.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:mirrors-reflection:11}
Tulisan \omterm{def:g5:mirrors-reflection:reflection}{cermin} kembali terbalik; pantulan ganda ``membalik
pembalikannya''. Jelaskan mengapa periskop pada
\cref{met:g5:mirrors-reflection:periscope}, dengan kedua \omterm{def:g5:mirrors-reflection:reflection}{cerminnya},
menampilkan pemandangannya dengan arah yang benar dan bukan
\omterm{def:g5:mirrors-reflection:reflection}{terbalik-cermin}.
\end{exercise}
