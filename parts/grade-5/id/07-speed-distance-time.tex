\chapter{Kelajuan: Jarak dan Waktu}\label{ch:g5:speed-distance-time}

``Aku lebih cepat!'' --- ``Bukan, aku!'' Setiap halaman bermain
menyelesaikannya dengan cara yang jujur: berbaris, siap, ya, mulai.
Tetapi bagaimana kamu membandingkan seorang pesepeda dengan seorang
pelari yang berlomba kemarin, di jalan yang lain? \omterm{def:g5:speed-distance-time:speed}{Kelajuan} adalah cara
fisika menyelesaikan lomba yang tidak pernah dapat dijalankan
berdampingan.

\section{Dua cara memenangkan lomba}

\begin{proposition}[Aturan lomba yang adil]\label{prop:g5:speed-distance-time:rules}
Dua cara yang jujur untuk membandingkan cepat dan lambat:
\begin{enumerate}
  \item pada \emph{jarak yang sama}, yang lebih cepat memerlukan
        \emph{waktu yang lebih sedikit} --- yang pertama melewati
        garis memenangkan lomba lari;
  \item dalam \emph{waktu yang sama}, yang lebih cepat menempuh
        \emph{jarak yang lebih jauh} --- sesudah satu menit, perenang
        yang lebih cepat memang sudah lebih jauh.
\end{enumerate}
Jarak yang sama atau waktu yang sama: tetapkan salah satunya,
bandingkan yang lain. Membandingkan dengan \emph{keduanya} berbeda ---
``Aku berlari lebih jauh!'' ``Tetapi kamu berlari sepanjang sore!''
--- tidak memutuskan apa pun.
\end{proposition}

\begin{example}[Tidak adil dan adil]\label{ex:g5:speed-distance-time:fair}
Lena bersepeda \qty{10}{km}; Marco bersepeda \qty{15}{km}. Apakah
Marco lebih cepat? Tidak ada cara untuk mengetahuinya --- barangkali
ia bersepeda tiga kali lebih lama. Tetapi kalau keduanya bersepeda
tepat selama satu \omterm{ex:g2:measuring-time:clock}{jam}, perbandingannya menjadi adil: waktu yang sama,
jarak yang lebih jauh, Marco lebih cepat. Sepenggal kata ``dalam satu
\omterm{ex:g2:measuring-time:clock}{jam}'' itu sebentar lagi menjadi ungkapan kesayangan kita.
\end{example}

\section{Kelajuan: jarak dalam satu jam}

\begin{definition}[Kelajuan]\label{def:g5:speed-distance-time:speed}
Yang disebut \emph{kelajuan}\index{kelajuan} sebuah pengembara yang
ajek adalah jarak yang ditempuhnya dalam satu jatah waktu yang tetap
--- biasanya satu \omterm{ex:g2:measuring-time:clock}{jam}. ``Mobil ini menempuh $60$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}''
menyebutkan kelajuannya; pejalan kaki yang cekatan sanggup kira-kira
$5$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}, pembalap sepeda $40$. Untuk yang merayap kita
memakai jatah yang lebih bersahabat: siput menempuh kira-kira $5$
\omterm{def:g2:measuring-length:units}{meter} tiap \omterm{ex:g2:measuring-time:clock}{jam}.
\end{definition}

\begin{method}[Mengukur kelajuanmu sendiri]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item Ukurlah atau carilah bentangan sepanjang tepat \qty{100}{m}
        (banyak lintasan olahraga menandainya);
  \item berjalanlah menyusurinya dengan langkah wajarmu sementara
        seorang teman menghitung waktumu --- misalkan perlu satu
        menit;
  \item satu \omterm{ex:g2:measuring-time:clock}{jam} berisi $60$ menit seperti itu, jadi dalam satu \omterm{ex:g2:measuring-time:clock}{jam}
        kamu akan menempuh $60 \times 100 = \qty{6000}{m}$;
  \item dikatakan dalam kilometer: \omterm{def:g5:speed-distance-time:speed}{kelajuan} jalanmu adalah $6$
        kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}.
\end{enumerate}
Ulangi sambil berlari lalu bandingkan --- kebanyakan anak kira-kira
menjadi dua kali lipat.
\end{method}

\section{Meramalkan jarak}

\begin{proposition}[Jarak dari kelajuan]\label{prop:g5:speed-distance-time:distance}
Pengembara yang ajek menempuh jarak satu \omterm{ex:g2:measuring-time:clock}{jamnya} lagi pada setiap \omterm{ex:g2:measuring-time:clock}{jam}
berikutnya. Jadi:
\[
\text{jarak yang ditempuh} = \text{jarak dalam satu jam} \times
\text{banyaknya jam}.
\]
Mobil yang menempuh $60$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam} menempuh, dalam $3$ \omterm{ex:g2:measuring-time:clock}{jam},
$60 \times 3 = \qty{180}{km}$.
\end{proposition}

\begin{example}[Hitungan perjalanan]\label{ex:g5:speed-distance-time:journeys}
Pejalan kaki dengan $4$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}, berjalan selama $2$ \omterm{ex:g2:measuring-time:clock}{jam}:
$4 \times 2 = \qty{8}{km}$. Pesepeda dengan $15$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam},
bersepeda $4$ \omterm{ex:g2:measuring-time:clock}{jam}: $15 \times 4 = \qty{60}{km}$. Pesawat penumpang
dengan $900$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}, terbang $5$ \omterm{ex:g2:measuring-time:clock}{jam}: $900 \times 5 =
\qty{4500}{km}$ --- satu benua diseberangi di antara dua kali makan.
\end{example}

\begin{remark}[Ajek adalah penyederhanaan]\label{rem:g5:speed-distance-time:steady}
Pengembara yang sesungguhnya jarang ajek: mobil melambat di desa,
pesepeda melesat menuruni bukit lalu bersusah payah mendaki, kamu
berlambat-lambat melewati toko roti. Hitungan kita berpura-pura bahwa
seluruh perjalanannya berlangsung pada satu langkah yang setia ---
pura-pura yang berguna, cukup baik untuk membuat rencana. Cara
bersikap jujur tentang perjalanan yang berlangkah-langkah banyak ---
dan apa arti ``rata-rata'' yang sebenarnya --- adalah cerita untuk
tahun yang lebih tinggi, ketika kamu sudah memiliki matematikanya.
\end{remark}

\begin{example}[Membaca kelajuan di sekitarmu]\label{ex:g5:speed-distance-time:signs}
Rambu jalan dan papan penunjuk di mobil berbicara dalam kilometer tiap
satu \omterm{ex:g2:measuring-time:clock}{jam} --- rambu bundar bertuliskan $50$ memberitahu pengemudi:
tempuhlah paling banyak $50$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}, sebuah langkah yang
dipilih supaya anak yang mengejar bola masih dapat diselamatkan. Layar
kereta cepat mungkin membanggakan $300$; papan penunjuk pejalan di
pegunungan lebih suka \omterm{ex:g2:measuring-time:clock}{jam} daripada kilometer --- ``danau: 2 j'' ---
dengan memercayai pejalannya mengenal \omterm{def:g5:speed-distance-time:speed}{kelajuannya} sendiri.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
Nyatakan kedua cara yang adil untuk membandingkan cepat dan lambat.
Mengapa ``aku berlari lebih jauh daripada kamu'' saja tidak
membuktikan apa pun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
Dua perenang menyeberangi kolam yang sama; Ana memerlukan $30$ sekon,
Bo memerlukan $25$. Siapa yang lebih cepat, dan menurut aturan lomba
adil yang mana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
Sesudah satu \omterm{ex:g2:measuring-time:clock}{jam}, sebuah traktor sudah menempuh \qty{20}{km} dan
sebuah skuter \qty{45}{km}. Siapa yang lebih cepat, dan menurut aturan
yang mana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
Apa arti ``kereta ini menempuh $200$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}''? Sejauh apa
ia sampai dalam $2$ \omterm{ex:g2:measuring-time:clock}{jam}? Dalam setengah \omterm{ex:g2:measuring-time:clock}{jam}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
Urutkan menurut \omterm{def:g5:speed-distance-time:speed}{kelajuan}: pejalan kaki ($5$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam});
\ seekor merpati ($80$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}); \ seekor siput ($5$
\emph{\omterm{def:g2:measuring-length:units}{meter}} tiap \omterm{ex:g2:measuring-time:clock}{jam}); \ sebuah bus kota ($30$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
Kamu berjalan santai menyusuri \qty{100}{m} yang ditandai dalam $2$
menit. Berapa \omterm{def:g2:measuring-length:units}{meter} yang akan kamu tempuh dalam satu \omterm{ex:g2:measuring-time:clock}{jam} pada langkah
itu? Sebutkan kelajuanmu dalam kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
Sebuah mobil menjaga \omterm{def:g5:speed-distance-time:speed}{kelajuan} ajek $80$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam} selama $3$
\omterm{ex:g2:measuring-time:clock}{jam}. Sejauh apa ia berjalan? Dan seorang pesepeda dengan $15$
kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam} selama $3$ \omterm{ex:g2:measuring-time:clock}{jam} yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
Pada grafik perjalanan dalam bab ini, bagaimana kamu mengenali
pengembara yang lebih cepat dengan sekali lihat? Bacalah darinya:
sejauh apa pesepedanya sudah berjalan sesudah $2$ \omterm{ex:g2:measuring-time:clock}{jam}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
Sebuah kapal feri menempuh $30$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}. Penyeberangannya
sepanjang \qty{90}{km}. Dengan memakai
\cref{prop:g5:speed-distance-time:distance} secara terbalik ---
cobalah \omterm{ex:g2:measuring-time:clock}{jam} demi \omterm{ex:g2:measuring-time:clock}{jam} --- berapa lama penyeberangannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
Dua kota berjarak \qty{60}{km}. Mara bersepeda dari kota yang satu
dengan $20$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}; pada saat yang sama Jon bersepeda dari
kota yang lain ke arahnya dengan $10$ kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}. Berapa
kilometer jarak di antara mereka menyusut tiap \omterm{ex:g2:measuring-time:clock}{jam}? Sesudah berapa \omterm{ex:g2:measuring-time:clock}{jam}
mereka bertemu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
Fabel lama, dengan bilangan: lombanya sepanjang \qty{120}{m}. Kelinci
berlari $60$ \omterm{def:g2:measuring-length:units}{meter} tiap menit --- tetapi sesudah satu menit berlari ia
berhenti untuk tidur selama $70$ menit, lalu berlari lagi. Kura-kura
melangkah $2$ \omterm{def:g2:measuring-length:units}{meter} tiap menit dan tidak pernah berhenti. Di mana
kelincinya ketika ia mulai tidur? Pada menit ke berapa masing-masing
melewati garis akhir? Siapa yang menang?
\end{exercise}
