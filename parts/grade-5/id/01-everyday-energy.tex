\chapter{Energi dalam Kehidupan Sehari-hari}\label{ch:g5:everyday-energy}

Mainan berpegas berlari cepat menyeberangi lantai, melambat, lalu
berhenti --- sampai kamu memutar pegasnya lagi. Senter meredup ketika
\omterm{def:g3:first-electric-circuit:battery}{baterainya} letih. Kamu sendiri kehabisan tenaga sebelum makan siang.
Segala yang bergerak, bersinar, berbunyi, atau menghangatkan tampak
sedang membelanjakan sesuatu. Fisika punya nama untuk sesuatu itu, dan
mungkin itulah kata yang paling penting dalam seluruh buku ini:
\omterm{def:g5:everyday-energy:energy}{energi}.

\section{Sesuatu yang membuat segalanya terjadi}

\begin{definition}[Energi]\label{def:g5:everyday-energy:energy}
Yang disebut \emph{energi}\index{energi} adalah apa yang diperlukan
untuk membuat segalanya terjadi: menggerakkan benda, mengangkatnya,
menyalakannya, menghangatkannya, membuatnya berbunyi. Tidak ada yang
berjalan, berpijar, atau tumbuh tanpa energi yang dibelanjakan
untuknya --- dan apa pun yang membelanjakannya harus lebih dahulu
punya energi untuk dibelanjakan.
\end{definition}

\begin{definition}[Penyimpan energi]\label{def:g5:everyday-energy:store}
Sebuah \emph{penyimpan energi}\index{penyimpan energi} adalah apa pun
yang menyimpan \omterm{def:g5:everyday-energy:energy}{energi} siap pakai. Penyimpan sehari-hari yang besar:
\emph{makanan} (bahan bakar tubuhmu), \emph{bahan bakar} seperti kayu
dan bensin, \emph{\omterm{def:g3:first-electric-circuit:battery}{baterai}} yang terisi, \emph{pegas} yang terputar,
air yang ditahan \emph{tinggi} di belakang bendungan --- dan
\emph{Matahari} yang panas, yang mencurahkan \omterm{def:g5:everyday-energy:energy}{energi} ke segala sesuatu,
sepanjang hari, dengan cuma-cuma.
\end{definition}

\begin{example}[Temukan penyimpannya]\label{ex:g5:everyday-energy:spot}
Setiap benda yang berjalan punya penyimpan di belakangnya. Mainan yang
berlari: pegasnya yang terputar. Senter: \omterm{def:g3:first-electric-circuit:battery}{baterainya}. Mobil: bahan
bakar di tangkinya. Kamu: sarapanmu. Perahu layar: \omterm{def:g2:air-around-us:air}{udara} yang bergerak
sebagai \omterm{def:g2:air-around-us:wind}{angin}. Ketika penyimpannya habis --- pegas terurai, \omterm{def:g3:first-electric-circuit:battery}{baterai}
kosong, tangki kering, waktu makan siang mendekat --- berjalannya pun
berhenti.
\end{example}

\begin{remark}[Tanpa diberi makan, tidak berjalan]\label{rem:g5:everyday-energy:nofree}
Selama berabad-abad, para penemu mengejar mesin yang akan berjalan
selamanya tanpa apa-apa --- roda yang berputar sendiri, \omterm{ex:g2:measuring-time:clock}{jam} yang tidak
pernah perlu diputar. Setiap satunya gagal. Kegagalan mereka adalah
salah satu pelajaran terdalam fisika: \omterm{def:g5:everyday-energy:energy}{energi} tidak pernah disulap dari
\omterm{def:g2:air-around-us:air}{udara} kosong. Apa pun yang berjalan sedang diberi makan --- temukan
penyimpannya, dan kamu memahami mesinnya.
\end{remark}

\section{Energi berubah bentuk}

\begin{example}[Bentuk-bentuk energi]\label{ex:g5:everyday-energy:forms}
\omterm{def:g5:everyday-energy:energy}{Energi} menampakkan diri dalam bentuk yang berbeda-beda: \omterm{def:g5:everyday-energy:energy}{energi}
\emph{gerak} pada bola yang menggelinding dan \omterm{def:g2:air-around-us:wind}{angin} yang bertiup;
\omterm{def:g5:everyday-energy:energy}{energi} \emph{\omterm{def:g1:light-and-shadows:source}{cahaya}} yang mencurah dari lampu dan Matahari; \omterm{def:g5:everyday-energy:energy}{energi}
\emph{panas} pada segala yang hangat; \omterm{def:g5:everyday-energy:energy}{energi} \emph{listrik} yang
merambat lewat kawat; \omterm{def:g5:everyday-energy:energy}{energi} \emph{tersimpan} yang menunggu diam di
dalam makanan, bahan bakar, pegas, dan benda yang terangkat. Sesuatu
yang sama, banyak bentuknya.
\end{example}

\begin{proposition}[Rantai energi]\label{prop:g5:everyday-energy:chain}
Setiap kali sesuatu terjadi, \omterm{def:g5:everyday-energy:energy}{energi} ditarik dari sebuah penyimpan lalu
diteruskan, biasanya berubah bentuk di sepanjang jalan. \omterm{def:g5:everyday-energy:energy}{Energi} tidak
pernah diciptakan dari ketiadaan --- dan tidak pernah benar-benar
dimusnahkan juga: apa yang ``dihabiskan'' sebuah mesin hanya
diteruskan, paling sering berakhir, cepat atau lambat, sebagai
kehangatan lembut yang menyebar ke sekelilingnya.
\end{proposition}

\begin{method}[Membaca rantai energi]\label{met:g5:everyday-energy:read}
Untuk memahami alat apa pun, telusurilah rantainya:
\begin{enumerate}
  \item temukan \emph{penyimpan} asal \omterm{def:g5:everyday-energy:energy}{energinya};
  \item temukan \emph{pengubahnya} --- bagian tempat \omterm{def:g5:everyday-energy:energy}{energinya}
        berubah bentuk;
  \item sebutkan bentuk yang keluar --- yang berguna, dan kehangatan
        yang bocor di sampingnya.
\end{enumerate}
Senter, ditelusuri: \omterm{def:g3:first-electric-circuit:battery}{baterai} (penyimpan) $\to$ \omterm{def:g3:first-electric-circuit:bulb}{bohlam} (pengubah) $\to$
\omterm{def:g1:light-and-shadows:source}{cahaya} (yang berguna) $+$ sedikit kehangatan (kebocorannya --- rabalah
\omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}).
\end{method}

\begin{example}[Rantai di mana-mana]\label{ex:g5:everyday-energy:chains}
Perjalanan pagimu, ditelusuri: sarapan (penyimpan) $\to$ ototmu
(pengubah) $\to$ gerak sepeda $+$ kehangatan (kamu merasakannya ---
tubuhmu memanas). Ladang \omterm{def:g2:air-around-us:wind}{angin}: \omterm{def:g2:air-around-us:air}{udara} yang bergerak $\to$ bilah yang
berputar $\to$ \omterm{def:g5:everyday-energy:energy}{energi} listrik di dalam kawat $\to$ \omterm{def:g1:light-and-shadows:source}{cahaya} di lampumu
malam ini. Bendungan: air yang tinggi $\to$ air yang jatuh memutar
turbin $\to$ \omterm{def:g5:everyday-energy:energy}{energi} listrik. Rantai yang panjang hanyalah rantai
pendek yang disambung: bentuk demi bentuk demi bentuk.
\end{example}

\section{Matahari di kepala meja}

\begin{example}[Hampir setiap rantai bermula dari Matahari]\label{ex:g5:everyday-energy:sun}
Telusurilah kebanyakan rantai ke belakang dan kamu tiba di Matahari.
Sarapan? Tumbuhan menumbuhkannya dengan menangkap \omterm{def:g1:light-and-shadows:source}{cahaya} matahari;
serealmu adalah sinar matahari dalam kaleng. \omterm{def:g2:air-around-us:wind}{Angin}? Matahari
menghangatkan daratan secara tidak merata, dan \omterm{def:g2:air-around-us:air}{udara} yang teraduk
itulah \omterm{def:g2:air-around-us:wind}{angin} --- jadi perahu layar pun berjalan dengan sinar matahari.
Kayu? Sinar matahari yang disimpan pohon. Air tinggi di bendungan?
Matahari yang mengangkatnya: ia menguapkan laut, lalu uapnya turun
sebagai hujan di pegunungan. Panel surya hanya melewati para
perantara.
\end{example}

\begin{remark}[Kata yang akan kita tajamkan]\label{rem:g5:everyday-energy:promise}
Untuk sekarang, \omterm{def:g5:everyday-energy:energy}{energi} adalah cerita yang diceritakan dengan baik:
penyimpan, bentuk, rantai. Dapatkah ia \emph{diukur}, seperti panjang
dalam \omterm{def:g2:measuring-length:units}{meter} dan \omterm{def:g3:measuring-mass:mass}{massa} dalam \omterm{def:g3:measuring-mass:units}{gram}? Dapat --- \omterm{def:g5:everyday-energy:energy}{energi} punya \omterm{def:g2:measuring-length:measure}{satuannya}
sendiri dan pembukuannya yang persis, dan keduanya datang kemudian
dalam buku ini, setelah kamu berkenalan dengan kelajuan dan \omterm{def:g2:pushes-and-pulls:force}{gaya}
sebagaimana mestinya. Rantai yang kamu telusuri tahun ini adalah
petanya; bilangannya akan datang untuk tinggal di atasnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:1}
Apa itu \omterm{def:g5:everyday-energy:energy}{energi}, menurut kata-kata bab ini? Sebutkan empat hal yang
tidak dapat terjadi tanpanya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:2}
Temukan penyimpannya: tikus mainan berpegas yang berlari; \ seorang
pengendara sepeda; \ api unggun; \ perahu layar; \ sebuah senter.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:3}
Sebutkan lima bentuk \omterm{def:g5:everyday-energy:energy}{energi}, masing-masing dengan satu contoh dari
harimu sendiri.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:4}
Telusurilah rantai \omterm{def:g5:everyday-energy:energy}{energi} sebuah pemanggang roti, seperti pada
\cref{met:g5:everyday-energy:read}: penyimpan, pengubah, bentuk keluar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:5}
Telusurilah rantai dari danau sebuah bendungan sampai ke lampu bacamu
malam ini.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:6}
Sebuah telepon ``mati''. Apa yang sebenarnya habis? Ke mana perginya
\omterm{def:g5:everyday-energy:energy}{energi} itu, menurut proposisi rantai?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:7}
Mengapa laptop menjadi hangat di bagian bawahnya selagi kamu
memakainya? Mata rantai yang mana yang sedang menampakkan diri?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:8}
Tunjukkan bahwa perahu layar berjalan dengan sinar matahari, dengan
menelusuri rantainya ke belakang selangkah demi selangkah.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:everyday-energy:9}
Seorang penemu mengumumkan lampu yang ``tidak memerlukan \omterm{def:g3:first-electric-circuit:battery}{baterai},
tidak memerlukan colokan, tidak memerlukan bahan bakar --- ia sekadar
bersinar selamanya''. Pelajaran mana pada
\cref{rem:g5:everyday-energy:nofree} yang dilawan pengumuman itu?
Pertanyaan apa yang harus kamu ajukan lebih dahulu kepada penemunya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:everyday-energy:10}
Sebuah bola karet yang dijatuhkan dari setinggi bahu memantul makin
rendah dan akhirnya terbaring diam. ``\omterm{def:g5:everyday-energy:energy}{Energinya} dimusnahkan,'' kata
Leo. Pakailah \cref{prop:g5:everyday-energy:chain} untuk membela akhir
cerita yang lebih baik --- ke dalam bentuk akhir yang mana \omterm{def:g5:everyday-energy:energy}{energi}
pantulan itu menyelinap?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:everyday-energy:11}
Ada rumah yang memanaskan air di panel hitam pada atapnya; ada yang
memasak dengan sinar matahari dan cermin melengkung; sel surya membuat
listrik dari \omterm{def:g1:light-and-shadows:source}{cahaya} siang. Jelaskan ungkapan ``panel surya hanya
melewati para perantara'' pada \cref{ex:g5:everyday-energy:sun}:
perantara mana saja yang berdiri di antara Matahari dan kehangatan api
kayu?
\end{exercise}
