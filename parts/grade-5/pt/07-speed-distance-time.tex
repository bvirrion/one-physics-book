\chapter{Velocidade: distância e tempo}\label{ch:g5:speed-distance-time}

``Eu sou mais rápido!'' --- ``Não, eu é que sou!'' Todo pátio resolve
isso do jeito honesto: fila, preparar, apontar, já. Mas como comparar
um ciclista com um corredor que competiu ontem, em outra estrada? A
\omterm{def:g5:speed-distance-time:speed}{velocidade} é o jeito que a física tem de decidir corridas que nunca
podem ser disputadas lado a lado.

\section{Dois jeitos de ganhar uma corrida}

\begin{proposition}[As regras da corrida justa]\label{prop:g5:speed-distance-time:rules}
Dois jeitos honestos de comparar rápido e devagar:
\begin{enumerate}
  \item na \emph{mesma distância}, o mais rápido precisa de
        \emph{menos tempo} --- quem cruza a linha primeiro ganha a
        prova;
  \item no \emph{mesmo tempo}, o mais rápido percorre \emph{mais
        distância} --- depois de um minuto, o nadador mais rápido
        está simplesmente mais longe.
\end{enumerate}
Mesma distância ou mesmo tempo: fixe um, compare o outro. Comparar
com os \emph{dois} diferentes --- ``eu corri mais longe!'' ``mas você
correu a tarde inteira!'' --- não decide nada.
\end{proposition}

\begin{example}[Injusto e justo]\label{ex:g5:speed-distance-time:fair}
Lena pedalou \qty{10}{km}; Marco pedalou \qty{15}{km}. Marco é mais
rápido? Não dá para dizer --- talvez ele tenha pedalado três vezes
mais tempo. Mas, se os dois pedalaram exatamente uma hora, a
comparação fica justa: mesmo tempo, mais distância, Marco é mais
rápido. Esse ``em uma hora'' está prestes a virar a nossa frase
predileta.
\end{example}

\section{Velocidade: a distância em uma hora}

\begin{definition}[Velocidade]\label{def:g5:speed-distance-time:speed}
A \emph{velocidade}\index{velocidade} de um viajante constante é a
distância que ele percorre numa porção fixa de tempo --- em geral uma
hora. ``Este carro percorre $60$ quilômetros em cada hora'' diz a
velocidade dele; uma pessoa andando depressa faz cerca de $5$
quilômetros em cada hora, e um ciclista de corrida, $40$. Para os
lerdos usamos porções mais amigáveis: um caracol percorre cerca de
$5$ \omterm{def:g2:measuring-length:units}{metros} em cada hora.
\end{definition}

\begin{method}[Medir a sua própria velocidade]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item Meça ou encontre um trecho de exatamente \qty{100}{m} (muitas
        pistas de atletismo têm um marcado);
  \item percorra-o no seu passo natural enquanto um colega cronometra
        --- suponha que leve um minuto;
  \item uma hora guarda $60$ desses minutos, então em uma hora você
        percorreria $60 \times 100 = \qty{6000}{m}$;
  \item dito em quilômetros: a sua \omterm{def:g5:speed-distance-time:speed}{velocidade} caminhando é de $6$
        quilômetros em cada hora.
\end{enumerate}
Repita correndo e compare --- a maioria das crianças mais ou menos
dobra o valor.
\end{method}

\section{Prever distâncias}

\begin{proposition}[Distância a partir da velocidade]\label{prop:g5:speed-distance-time:distance}
Um viajante constante percorre de novo a distância de uma hora em
cada hora seguinte. Então:
\[
\text{distância percorrida} = \text{distância em uma hora} \times
\text{número de horas}.
\]
Um carro que faz $60$ quilômetros em cada hora percorre, em $3$
\omterm{ex:g2:measuring-time:clock}{horas}, $60 \times 3 = \qty{180}{km}$.
\end{proposition}

\begin{example}[Contas de viagem]\label{ex:g5:speed-distance-time:journeys}
A pessoa a $4$ quilômetros em cada hora, caminhando por $2$ \omterm{ex:g2:measuring-time:clock}{horas}:
$4 \times 2 = \qty{8}{km}$. O ciclista a $15$ quilômetros em cada
hora, pedalando $4$ \omterm{ex:g2:measuring-time:clock}{horas}: $15 \times 4 = \qty{60}{km}$. O avião de
passageiros a $900$ quilômetros em cada hora, voando $5$ \omterm{ex:g2:measuring-time:clock}{horas}:
$900 \times 5 = \qty{4500}{km}$ --- um continente atravessado entre
duas refeições.
\end{example}

\begin{remark}[Constante é uma simplificação]\label{rem:g5:speed-distance-time:steady}
Os viajantes de verdade quase nunca são constantes: o carro diminui a
\omterm{def:g5:speed-distance-time:speed}{velocidade} nos vilarejos, o ciclista voa na descida e sofre na
subida, você enrola na frente da padaria. As nossas contas fingem que
a viagem inteira corre num único ritmo fiel --- um fingimento útil,
bom o bastante para planejar. Como ser honesto sobre uma viagem de
muitos ritmos --- e o que ``média'' quer dizer de verdade --- é uma
história para um ano mais adiante, quando você tiver a matemática
certa.
\end{remark}

\begin{example}[Lendo velocidades ao seu redor]\label{ex:g5:speed-distance-time:signs}
As placas de trânsito e os painéis dos carros falam em
quilômetros-por-uma-hora --- a placa redonda com $50$ diz aos
motoristas: percorra no máximo $50$ quilômetros em cada hora, um
ritmo escolhido para que uma criança correndo atrás de uma bola ainda
possa ser poupada. O painel de um trem de alta \omterm{def:g5:speed-distance-time:speed}{velocidade} pode exibir
$300$; as placas de trilha nas montanhas preferem \omterm{ex:g2:measuring-time:clock}{horas} a
quilômetros --- ``lago: 2 h'' --- confiando que o caminhante conheça
a própria \omterm{def:g5:speed-distance-time:speed}{velocidade}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
Enuncie os dois jeitos justos de comparar rápido e devagar. Por que
``eu corri mais longe que você'' sozinho não prova nada?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
Dois nadadores atravessam a mesma piscina; Ana precisa de $30$
segundos, Bo precisa de $25$. Quem é mais rápido, e por qual regra da
corrida justa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
Depois de uma hora, um trator percorreu \qty{20}{km} e uma
motoneta, \qty{45}{km}. Quem é mais rápido, e por qual regra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
O que quer dizer ``este trem percorre $200$ quilômetros em cada
hora''? Que distância ele alcança em $2$ \omterm{ex:g2:measuring-time:clock}{horas}? E em meia hora?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
Ordene por \omterm{def:g5:speed-distance-time:speed}{velocidade}: uma pessoa caminhando ($5$
quilômetros em cada hora); \ um pombo ($80$ quilômetros em cada
hora); \ um caracol ($5$ \emph{\omterm{def:g2:measuring-length:units}{metros}} em cada hora); \ um ônibus
urbano ($30$ quilômetros em cada hora).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
Você percorre tranquilamente um trecho marcado de \qty{100}{m} em $2$
minutos. Quantos \omterm{def:g2:measuring-length:units}{metros} você percorreria em uma hora nesse ritmo?
Diga a sua \omterm{def:g5:speed-distance-time:speed}{velocidade} em quilômetros em cada hora.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
Um carro mantém $80$ quilômetros em cada hora durante $3$ \omterm{ex:g2:measuring-time:clock}{horas}. Que
distância ele percorre? E um ciclista a $15$ quilômetros em cada hora
nas mesmas $3$ \omterm{ex:g2:measuring-time:clock}{horas}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
No gráfico das viagens do capítulo, como identificar de relance o
viajante mais rápido? Leia no gráfico: que distância o ciclista já
percorreu depois de $2$ \omterm{ex:g2:measuring-time:clock}{horas}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
Uma balsa percorre $30$ quilômetros em cada hora. A travessia tem
\qty{90}{km}. Usando a
\cref{prop:g5:speed-distance-time:distance} ao contrário --- teste
hora por hora --- quanto tempo leva a travessia?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
Duas cidades ficam a \qty{60}{km} uma da outra. Mara pedala de uma
delas a $20$ quilômetros em cada hora; no mesmo instante, Jon pedala
da outra na direção dela a $10$ quilômetros em cada hora. Quantos
quilômetros o vão entre os dois encolhe a cada hora? Depois de
quantas \omterm{ex:g2:measuring-time:clock}{horas} eles se encontram?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
A velha fábula, com números: a corrida tem \qty{120}{m}. A lebre
corre $60$ \omterm{def:g2:measuring-length:units}{metros} em cada minuto --- mas, depois de um minuto de
corrida, ela para para um cochilo de $70$ minutos e depois volta a
correr. A tartaruga se arrasta a $2$ \omterm{def:g2:measuring-length:units}{metros} em cada minuto e nunca
para. Onde está a lebre quando começa o cochilo? Em que instante cada
uma cruza a linha de chegada? Quem ganha?
\end{exercise}
