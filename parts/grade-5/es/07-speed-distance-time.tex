\chapter{La velocidad: distancia y tiempo}\label{ch:g5:speed-distance-time}

``¡Yo soy más rápido!'' --- ``¡Qué va, yo!''. En todos los patios esto
se zanja de la manera honrada: en fila, preparados, listos, ya. Pero
¿cómo comparas a un ciclista con un corredor que compitió ayer, en
otra carretera? La \omterm{def:g5:speed-distance-time:speed}{velocidad} es la manera que tiene la física de
zanjar las carreras que no se pueden correr juntas.

\section{Dos maneras de ganar una carrera}

\begin{proposition}[Las reglas de la carrera justa]\label{prop:g5:speed-distance-time:rules}
Dos maneras honradas de comparar lo rápido y lo lento:
\begin{enumerate}
  \item con la \emph{misma distancia}, el más rápido necesita
        \emph{menos tiempo} --- gana el esprint quien cruza primero la
        línea;
  \item en el \emph{mismo tiempo}, el más rápido recorre \emph{más
        distancia} --- al cabo de un minuto, el nadador más rápido
        está sencillamente más lejos.
\end{enumerate}
Misma distancia o mismo tiempo: fija una cosa y compara la otra.
Comparar con las \emph{dos} distintas --- ``¡yo he corrido más lejos!''
``¡pero tú has corrido toda la tarde!'' --- no decide nada.
\end{proposition}

\begin{example}[Injusto y justo]\label{ex:g5:speed-distance-time:fair}
Lena recorrió \qty{10}{km} en bici; Marco, \qty{15}{km}. ¿Es Marco más
rápido? No hay manera de saberlo --- puede que él pedaleara el triple
de tiempo. Pero si los dos pedalearon exactamente una hora, la
comparación se vuelve justa: mismo tiempo, más distancia, Marco es más
rápido. Ese pequeño ``en una hora'' está a punto de convertirse en
nuestra frase favorita.
\end{example}

\section{La velocidad: la distancia en una hora}

\begin{definition}[La velocidad]\label{def:g5:speed-distance-time:speed}
La \emph{velocidad}\index{velocidad} de un viajero constante es la
distancia que recorre en una ración fija de tiempo --- normalmente una
hora. ``Este coche recorre $60$ kilómetros en cada hora'' nombra su
velocidad; una persona que camina a buen paso hace unos $5$ kilómetros
en cada hora y un ciclista de carreras, $40$. Para los que se
arrastran usamos raciones más amables: un caracol recorre unos $5$
\omterm{def:g2:measuring-length:units}{metros} en cada hora.
\end{definition}

\begin{method}[Medir tu propia velocidad]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item mide o busca un tramo de exactamente \qty{100}{m} (muchas
        pistas de atletismo tienen uno marcado);
  \item recórrelo a tu paso natural mientras un amigo te cronometra
        --- supongamos que tardas un minuto;
  \item en una hora caben $60$ de esos minutos, así que en una hora
        recorrerías $60 \times 100 = \qty{6000}{m}$;
  \item dicho en kilómetros: tu \omterm{def:g5:speed-distance-time:speed}{velocidad} andando es de $6$
        kilómetros en cada hora.
\end{enumerate}
Repítelo corriendo y compara --- casi todos los niños más o menos la
duplican.
\end{method}

\section{Predecir distancias}

\begin{proposition}[La distancia a partir de la velocidad]\label{prop:g5:speed-distance-time:distance}
Un viajero constante vuelve a recorrer su distancia de una hora en
cada hora siguiente. Así que:
\[
\text{distancia recorrida} = \text{distancia en una hora} \times
\text{número de horas}.
\]
Un coche que hace $60$ kilómetros en cada hora recorre, en $3$
\omterm{ex:g2:measuring-time:clock}{horas}, $60 \times 3 = \qty{180}{km}$.
\end{proposition}

\begin{example}[Cuentas de viaje]\label{ex:g5:speed-distance-time:journeys}
El caminante a $4$ kilómetros en cada hora, de paseo durante $2$
\omterm{ex:g2:measuring-time:clock}{horas}: $4 \times 2 = \qty{8}{km}$. El ciclista a $15$ kilómetros en
cada hora, pedaleando $4$ \omterm{ex:g2:measuring-time:clock}{horas}: $15 \times 4 = \qty{60}{km}$. El
avión de línea a $900$ kilómetros en cada hora, volando $5$
\omterm{ex:g2:measuring-time:clock}{horas}: $900 \times 5 = \qty{4500}{km}$ --- un continente cruzado
entre dos comidas.
\end{example}

\begin{remark}[Lo de constante es una simplificación]\label{rem:g5:speed-distance-time:steady}
Los viajeros de verdad casi nunca son constantes: el coche frena en
los pueblos, el ciclista vuela cuesta abajo y sufre cuesta arriba, y
tú te entretienes delante de la panadería. Nuestras cuentas fingen que
el viaje entero se recorre a un único ritmo fiel --- un fingimiento
útil y suficiente para hacer planes. Cómo ser honrado con un viaje de
muchos ritmos --- y qué significa de verdad la palabra ``media'' ---
es historia para un año posterior, cuando tengas las matemáticas
adecuadas.
\end{remark}

\begin{example}[Leer velocidades a tu alrededor]\label{ex:g5:speed-distance-time:signs}
Las señales de tráfico y los cuadros de los coches hablan en
kilómetros por cada hora --- la señal redonda que dice $50$ les dice a
los \omterm{def:g4:conductors-insulators:conductor}{conductores}: recorred como mucho $50$ kilómetros en cada hora, un
ritmo elegido para que un niño que persigue una pelota todavía pueda
salvarse. La pantalla del tren de alta \omterm{def:g5:speed-distance-time:speed}{velocidad} puede presumir de
$300$; los carteles de los senderos de montaña prefieren las
\omterm{ex:g2:measuring-time:clock}{horas} a los kilómetros --- ``lago: 2 h'' --- y confían en que el
caminante conozca su propia \omterm{def:g5:speed-distance-time:speed}{velocidad}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
Enuncia las dos maneras justas de comparar lo rápido y lo lento. ¿Por
qué no demuestra nada, por sí solo, ``he corrido más lejos que tú''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
Dos nadadores cruzan la misma piscina; Ana necesita $30$ segundos y Bo
necesita $25$. ¿Quién es más rápido y por cuál de las reglas de la
carrera justa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
Al cabo de una hora, un tractor ha recorrido \qty{20}{km} y una moto
\qty{45}{km}. ¿Quién es más rápido y por qué regla?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
¿Qué significa ``este tren recorre $200$ kilómetros en cada hora''?
¿Hasta dónde llega en $2$ \omterm{ex:g2:measuring-time:clock}{horas}? ¿Y en media hora?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
Ordena por \omterm{def:g5:speed-distance-time:speed}{velocidad}: un caminante ($5$ kilómetros en cada hora); \
una paloma ($80$ kilómetros en cada hora); \ un caracol ($5$
\emph{\omterm{def:g2:measuring-length:units}{metros}} en cada hora); \ un autobús urbano ($30$ kilómetros en
cada hora).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
Recorres paseando un tramo marcado de \qty{100}{m} en $2$ minutos.
¿Cuántos \omterm{def:g2:measuring-length:units}{metros} recorrerías en una hora a ese ritmo? Di tu
\omterm{def:g5:speed-distance-time:speed}{velocidad} en kilómetros en cada hora.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
Un coche mantiene $80$ kilómetros en cada hora durante $3$
\omterm{ex:g2:measuring-time:clock}{horas}. ¿Cuánto recorre? ¿Y un ciclista a $15$ kilómetros en cada
hora durante esas mismas $3$ \omterm{ex:g2:measuring-time:clock}{horas}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
En la gráfica de los viajes del capítulo, ¿cómo localizas de un
vistazo al viajero más rápido? Lee en ella: ¿cuánto ha recorrido el
ciclista al cabo de $2$ \omterm{ex:g2:measuring-time:clock}{horas}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
Un transbordador recorre $30$ kilómetros en cada hora. La travesía es
de \qty{90}{km}. Usando la \cref{prop:g5:speed-distance-time:distance}
al revés --- prueba con las \omterm{ex:g2:measuring-time:clock}{horas} una a una ---, ¿cuánto dura la
travesía?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
Dos pueblos están a \qty{60}{km} el uno del otro. Mara sale en bici de
uno a $20$ kilómetros en cada hora; en ese mismo momento Jon sale del
otro hacia ella a $10$ kilómetros en cada hora. ¿Cuántos kilómetros
encoge el hueco entre los dos en cada hora? ¿Al cabo de cuántas
\omterm{ex:g2:measuring-time:clock}{horas} se encuentran?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
La vieja fábula, con números: la carrera mide \qty{120}{m}. La liebre
corre $60$ \omterm{def:g2:measuring-length:units}{metros} en cada minuto --- pero, después de un minuto
corriendo, se para a echar una siesta de $70$ minutos y luego sigue
corriendo. La tortuga avanza $2$ \omterm{def:g2:measuring-length:units}{metros} en cada minuto y no se para
nunca. ¿Dónde está la liebre cuando empieza su siesta? ¿En qué momento
cruza la meta cada una? ¿Quién gana?
\end{exercise}
