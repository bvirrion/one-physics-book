\chapter{El calor y el aislamiento}\label{ch:g5:heat-and-insulation}

¿Por qué se nota helado el tobogán de metal y el banco de madera que
está a su lado no? Ese misterio nos persigue desde el primer año de
este libro. Hoy cae --- junto con los secretos del termo, de la
ventana doble y del gorrión hinchado de invierno. La clave es dejar de
preguntar cómo se \emph{notan} las cosas y empezar a seguir por dónde
\emph{fluye} el \omterm{def:g5:heat-and-insulation:heat}{calor}.

\section{El calor está de viaje}

\begin{definition}[El calor]\label{def:g5:heat-and-insulation:heat}
El \emph{calor}\index{calor} es \omterm{def:g5:everyday-energy:energy}{energía} de viaje desde una cosa más
caliente hacia otra más fría. La sopa le pasa calor a la cuchara, el
radiador a la habitación, tu mano a una bola de nieve. El calor no es
una sustancia escondida dentro de las cosas calientes --- es un
\emph{flujo}, y siempre tiene una dirección.
\end{definition}

\begin{proposition}[El calor fluye cuesta abajo]\label{prop:g5:heat-and-insulation:downhill}
Por sí solo, el \omterm{def:g5:heat-and-insulation:heat}{calor} fluye siempre del cuerpo más caliente al más
frío --- nunca al revés. El flujo dura mientras las dos
\omterm{def:g3:temperature-thermometers:temperature}{temperaturas} sean distintas y se apaga cuando se igualan: si se
los deja juntos el tiempo suficiente, un vecino caliente y otro frío
acaban a una misma \omterm{def:g3:temperature-thermometers:temperature}{temperatura} compartida.
\end{proposition}

\begin{example}[Leer la dirección]\label{ex:g5:heat-and-insulation:direction}
La cucharilla dentro del té se pone caliente: el \omterm{def:g5:heat-and-insulation:heat}{calor} fluyó del té
$\to$ a la cuchara. La limonada con hielo se enfría --- ¡cuidado! No
entró nada llamado ``frío'': el \omterm{def:g5:heat-and-insulation:heat}{calor} salió \emph{de la limonada
hacia el hielo} y lo derritió. El frío no es un flujo; es solo el
nombre que le damos al lado del perdedor. Todos los libros de física
dicen ``la bebida pierde \omterm{def:g5:heat-and-insulation:heat}{calor}'', nunca ``la bebida gana frío''.
\end{example}

\section{Carriles rápidos y carriles lentos}

\begin{definition}[Conductores y aislantes del calor]\label{def:g5:heat-and-insulation:conductors}
Las sustancias se diferencian muchísimo en la velocidad con la que
dejan fluir el \omterm{def:g5:heat-and-insulation:heat}{calor} por su interior. Un \emph{conductor del
calor}\index{conductor del calor} es un carril rápido --- los metales
por encima de todos. Un \emph{aislante del calor}\index{aislante del
calor} es un carril lento: la madera, la lana, el plástico, las plumas
--- y el campeón, el \emph{\omterm{def:g2:air-around-us:air}{aire} quieto}, atrapado para que no pueda
arremolinarse. (Las palabras están tomadas a propósito de los
\omterm{def:g4:conductors-insulators:conductor}{conductores} y \omterm{def:g4:conductors-insulators:insulator}{aislantes} de la electricidad; las dos
propiedades van a menudo juntas, aunque no siempre.)
\end{definition}

\begin{example}[El misterio del tobogán, resuelto]\label{ex:g5:heat-and-insulation:slide}
El tobogán de metal y el banco de madera pasaron la noche a la misma
\omterm{def:g3:temperature-thermometers:temperature}{temperatura} helada --- el \omterm{def:g3:temperature-thermometers:temperature}{termómetro} lo juró hace ya años.
Tócalos: tu mano está más caliente que los dos, así que el \omterm{def:g5:heat-and-insulation:heat}{calor}
fluye de ti hacia cada uno. Pero el metal es un carril rápido --- se
lleva tu \omterm{def:g5:heat-and-insulation:heat}{calor} a toda prisa, y esa \emph{prisa} es lo que se nota
como ``helado''. La madera te toma el \omterm{def:g5:heat-and-insulation:heat}{calor} solo a gotas: apenas
fresca. Tu piel no estuvo nunca midiendo la \omterm{def:g3:temperature-thermometers:temperature}{temperatura} --- mide
la velocidad de su propia pérdida de \omterm{def:g5:heat-and-insulation:heat}{calor}. Caso cerrado, después de
cuatro años.
\end{example}

\begin{method}[La carrera del enfriamiento]\label{met:g5:heat-and-insulation:race}
Mira ganar al aislamiento, con números:
\begin{enumerate}
  \item llena dos tazas iguales con agua del grifo igual de caliente
        (\emph{pide a una persona adulta agua caliente de la mano, no
        hirviendo});
  \item envuelve una taza con una bufanda de lana bien gruesa y deja
        la otra desnuda;
  \item planta un \omterm{def:g3:temperature-thermometers:temperature}{termómetro} en cada una y anota las dos
        \omterm{def:g3:temperature-thermometers:temperature}{temperaturas} en una tabla cada cinco minutos, seis veces;
  \item dibuja los dos registros como dos líneas en una misma
        gráfica, la \omterm{def:g3:temperature-thermometers:temperature}{temperatura} frente al tiempo.
\end{enumerate}
La línea de la taza desnuda se despeña; la de la taza envuelta baja
con suavidad. Mismo comienzo, misma habitación --- la lana convirtió
en carril lento la salida del \omterm{def:g5:heat-and-insulation:heat}{calor}.
\end{method}

\begin{remark}[El aislamiento solo frena]\label{rem:g5:heat-and-insulation:onlyslows}
Fíjate hacia dónde se dirigen las dos curvas: hacia los
\qty{20}{\celsius} de la habitación. Ninguna bufanda, ningún termo y
ninguna pared de iglú detienen jamás el flujo cuesta abajo del
\omterm{def:g5:heat-and-insulation:heat}{calor} --- el aislamiento solo hace el carril lento todavía más
lento. Por eso también el abrigo del muñeco de nieve (¿te acuerdas de
él?) retrasó su derretimiento pero no pudo salvarlo: el \omterm{def:g5:heat-and-insulation:heat}{calor} del
verano seguía colándose a gotas.
\end{remark}

\section{El arte de atrapar aire}

\begin{example}[Ingeniería de invierno]\label{ex:g5:heat-and-insulation:air}
Casi todas las cosas calentitas que tienes son máquinas de atrapar
\omterm{def:g2:air-around-us:air}{aire}. La lana no te da \omterm{def:g5:heat-and-insulation:heat}{calor} por las fibras en sí, sino por las
incontables bolsas de \omterm{def:g2:air-around-us:air}{aire} que quedan entre ellas; los edredones de
plumas, el forro polar y el gorrión hinchado como una bola sobre un
cable helado juegan todos la misma carta. Las casas se apuntan:
las ventanas dobles encierran una capa de \omterm{def:g2:air-around-us:air}{aire} entre dos cristales, y
las paredes aisladas van rellenas de un acolchado esponjoso y
acaparador de \omterm{def:g2:air-around-us:air}{aire}. El \omterm{def:g2:air-around-us:air}{aire} quieto es el campeón de los carriles
lentos --- el truco está en mantenerlo \emph{quieto}.
\end{example}

\begin{example}[El termo]\label{ex:g5:heat-and-insulation:thermos}
El termo es la obra maestra del aislamiento: una botella dentro de
otra botella, con el \omterm{def:g2:air-around-us:air}{aire} de entre las dos sacado hasta dejarlo casi
en nada --- y donde casi no hay materia, el flujo del \omterm{def:g5:heat-and-insulation:heat}{calor} casi no
encuentra carril (un truco que vale la pena recordar: el mismo vacío
que silenció la campana del recipiente). Añade paredes brillantes que
devuelvan hacia dentro el resplandor del \omterm{def:g5:heat-and-insulation:heat}{calor} y el chocolate caliente
llegará a la tarde todavía caliente --- o la limonada todavía fría: el
termo no sabe nada de caliente ni de frío, solo cierra los carriles,
en los dos sentidos.
\end{example}

\begin{remark}[Abrigarse, dicho como es debido]\label{rem:g5:heat-and-insulation:coat}
Ahora la vieja cuestión del abrigo se cierra para siempre. Un abrigo
es \omterm{def:g2:air-around-us:air}{aire} atrapado alrededor de una fábrica de \omterm{def:g5:heat-and-insulation:heat}{calor} --- tú. Tu
cuerpo echa \omterm{def:g5:heat-and-insulation:heat}{calor} hacia fuera a chorro todo el día; el abrigo
convierte ese chorro en un hilillo, y tú te quedas calentito con tu
propia producción. Sobre el muñeco de nieve --- que no tiene ninguna
fábrica de \omterm{def:g5:heat-and-insulation:heat}{calor} dentro --- ese mismo abrigo se limita a frenar el
hilillo del verano que entra. El aislamiento no tiene favoritos: frena
el flujo, corra en el sentido que corra.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:1}
¿Qué es el \omterm{def:g5:heat-and-insulation:heat}{calor} y en qué dirección fluye por sí solo? ¿Cuándo se
para el flujo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:2}
``¡Cierra la puerta, que estás dejando entrar el frío!''. Reescribe la
frase de la abuela como la escribiría un libro de física.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:3}
Reparte en carriles rápidos y carriles lentos para el \omterm{def:g5:heat-and-insulation:heat}{calor}: una
sartén de cobre; \ una bufanda de lana; \ una cuchara de acero; \ el
\omterm{def:g2:air-around-us:air}{aire} quieto; \ una cuchara de madera; \ un edredón.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:4}
¿Por qué tienen las sartenes el fondo de metal y el mango de plástico
o de madera? Responde con carriles.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:5}
Por fin, con tus palabras: ¿por qué se nota más frío el metal que la
madera a la misma \omterm{def:g3:temperature-thermometers:temperature}{temperatura}? ¿Qué mide de verdad la piel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:6}
En la carrera del enfriamiento, ¿por qué tienen que ser iguales las
dos tazas y empezar igual de calientes? ¿Qué única diferencia está
poniendo a prueba el experimento?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:7}
Lee la gráfica del enfriamiento: ¿a qué \omterm{def:g3:temperature-thermometers:temperature}{temperatura} está más o
menos cada taza al cabo de $15$ minutos? ¿Hacia qué \omterm{def:g3:temperature-thermometers:temperature}{temperatura} se
dirigen las dos curvas y por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:heat-and-insulation:8}
¿Por qué se hincha un gorrión hasta quedar hecho una bola sobre un
cable helado? Nombra tres inventos humanos que jueguen la misma carta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:9}
El mismo termo mantiene caliente el chocolate en enero y fría la
limonada en julio. Explica cómo se apaña un solo aparato con los dos
trabajos --- ¿qué hace en realidad el termo y sobre qué?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:10}
Una heladera envuelve sus cubetas en mantas gruesas de lana un día de
mucho \omterm{def:g5:heat-and-insulation:heat}{calor}, y un cliente se ríe: ``¡La lana es para mantener las
cosas calientes!''. Defiende a la heladera con la
\cref{rem:g5:heat-and-insulation:coat} --- ¿en qué sentido intenta
fluir el \omterm{def:g5:heat-and-insulation:heat}{calor} y qué le hace la lana?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:heat-and-insulation:11}
Dos casas pasan el mismo invierno: una con ventanas sencillas y otra
con ventanas dobles. Explica con carriles y \omterm{def:g2:air-around-us:air}{aire} atrapado por qué la
casa de ventanas dobles quema menos combustible --- y por qué, según
la \cref{rem:g5:heat-and-insulation:onlyslows}, sigue necesitando
\emph{algo} de calefacción.
\end{exercise}
