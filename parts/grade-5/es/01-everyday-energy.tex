\chapter{La energía en la vida diaria}\label{ch:g5:everyday-energy}

Un juguete de cuerda corretea por el suelo, va frenando y se para ---
hasta que le vuelves a dar cuerda. Una linterna pierde brillo mientras
sus \omterm{def:g3:first-electric-circuit:battery}{pilas} se cansan. Y tú mismo te quedas sin fuelle antes de comer.
Todo lo que se mueve, brilla, suena o calienta parece estar gastando
algo. La física tiene un nombre para ese algo, y quizá sea la palabra
más importante de todo este libro: \omterm{def:g5:everyday-energy:energy}{energía}.

\section{El algo que consigue que las cosas pasen}

\begin{definition}[La energía]\label{def:g5:everyday-energy:energy}
La \emph{energía}\index{energía} es lo que hace falta para que las
cosas ocurran: para poner cosas en movimiento, para levantarlas, para
iluminarlas, para calentarlas, para hacerlas sonar. Nada anda,
brilla ni crece sin que se gaste energía en ello --- y quien la gasta
tiene que tener antes energía que gastar.
\end{definition}

\begin{definition}[Almacenes de energía]\label{def:g5:everyday-energy:store}
Un \emph{almacén de energía}\index{almacén de energía} es cualquier
cosa que guarde \omterm{def:g5:everyday-energy:energy}{energía} lista para usarse. Los grandes almacenes de
cada día: la \emph{comida} (el combustible de tu cuerpo), los
\emph{combustibles} como la leña y la gasolina, las \emph{\omterm{def:g3:first-electric-circuit:battery}{pilas}}
cargadas, un \emph{muelle} al que se le ha dado cuerda, el agua
retenida \emph{en alto} detrás de una presa --- y el \emph{Sol}
ardiente, que derrama \omterm{def:g5:everyday-energy:energy}{energía} sobre todo, todo el día y gratis.
\end{definition}

\begin{example}[Encuentra el almacén]\label{ex:g5:everyday-energy:spot}
Detrás de todo lo que anda hay un almacén. El juguete que corretea: su
muelle con cuerda. La linterna: sus \omterm{def:g3:first-electric-circuit:battery}{pilas}. El coche: el combustible
del depósito. Tú: tu desayuno. El velero: el \omterm{def:g2:air-around-us:air}{aire} en movimiento del
\omterm{def:g2:air-around-us:wind}{viento}. Cuando el almacén se vacía --- muelle sin cuerda, \omterm{def:g3:first-electric-circuit:battery}{pilas}
gastadas, depósito seco, hora de comer ---, el andar se para.
\end{example}

\begin{remark}[Sin alimentar no hay marcha]\label{rem:g5:everyday-energy:nofree}
Durante siglos, los inventores persiguieron máquinas que funcionaran
para siempre sin nada --- ruedas que se giran solas, relojes a los que
nunca hay que dar cuerda. Todas y cada una fracasaron. Su fracaso es
una de las lecciones más profundas de la física: la \omterm{def:g5:everyday-energy:energy}{energía} no se
saca nunca de la nada. Todo lo que funciona está siendo alimentado ---
encuentra el almacén y entenderás la máquina.
\end{remark}

\section{La energía cambia de forma}

\begin{example}[Formas de la energía]\label{ex:g5:everyday-energy:forms}
La \omterm{def:g5:everyday-energy:energy}{energía} se deja ver en formas distintas: \omterm{def:g5:everyday-energy:energy}{energía} de
\emph{movimiento} en la pelota que rueda y en el \omterm{def:g2:air-around-us:wind}{viento} que sopla;
\omterm{def:g5:everyday-energy:energy}{energía} \emph{luminosa} que sale a chorros de la lámpara y del
Sol; \omterm{def:g5:everyday-energy:energy}{energía} \emph{térmica} en todo lo que está caliente;
\omterm{def:g5:everyday-energy:energy}{energía} \emph{eléctrica} viajando por los cables; \omterm{def:g5:everyday-energy:energy}{energía}
\emph{almacenada} esperando tranquilamente en la comida, el
combustible, los muelles y las cosas levantadas. El mismo algo, muchas
formas.
\end{example}

\begin{proposition}[La cadena de la energía]\label{prop:g5:everyday-energy:chain}
Siempre que ocurre algo, se saca \omterm{def:g5:everyday-energy:energy}{energía} de un almacén y se la pasa
adelante, casi siempre cambiando de forma por el camino. Nunca se crea
de la nada --- y tampoco se destruye de verdad: lo que una máquina
``gasta'' solo se ha pasado adelante, y acaba casi siempre, tarde o
temprano, convertido en un calor suave repartido por los alrededores.
\end{proposition}

\begin{method}[Leer una cadena de energía]\label{met:g5:everyday-energy:read}
Para entender cualquier aparato, sigue su cadena:
\begin{enumerate}
  \item busca el \emph{almacén} del que viene la \omterm{def:g5:everyday-energy:energy}{energía};
  \item busca el \emph{convertidor} --- la pieza donde la \omterm{def:g5:everyday-energy:energy}{energía}
        cambia de forma;
  \item nombra las formas que salen --- la útil y el calor que se
        escapa a su lado.
\end{enumerate}
La linterna, seguida: \omterm{def:g3:first-electric-circuit:battery}{pila} (almacén) $\to$ \omterm{def:g3:first-electric-circuit:bulb}{bombilla}
(convertidor) $\to$ \omterm{def:g1:light-and-shadows:source}{luz} (útil) $+$ un poco de calor (la fuga ---
tócala).
\end{method}

\begin{example}[Cadenas por todas partes]\label{ex:g5:everyday-energy:chains}
Tu paseo en bicicleta de la mañana, seguido: el desayuno (almacén)
$\to$ tus músculos (convertidor) $\to$ movimiento de la bicicleta $+$
calor (lo notas --- estás sudando). El parque eólico: \omterm{def:g2:air-around-us:air}{aire} en
movimiento $\to$ palas que giran $\to$ \omterm{def:g5:everyday-energy:energy}{energía} eléctrica en los cables
$\to$ \omterm{def:g1:light-and-shadows:source}{luz} en tu lámpara esta noche. La presa: agua en alto $\to$ el
agua que cae hace girar las turbinas $\to$ \omterm{def:g5:everyday-energy:energy}{energía} eléctrica. Las
cadenas largas no son más que cadenas cortas enlazadas: forma tras
forma tras forma.
\end{example}

\section{El Sol a la cabecera de la mesa}

\begin{example}[Casi todas las cadenas empiezan en el Sol]\label{ex:g5:everyday-energy:sun}
Sigue casi cualquier cadena hacia atrás y llegarás al Sol. ¿El
desayuno? Lo hicieron crecer unas plantas atrapando la \omterm{def:g1:light-and-shadows:source}{luz} del Sol; tu
cereal es sol enlatado. ¿El \omterm{def:g2:air-around-us:wind}{viento}? El Sol calienta las tierras de
manera desigual, y el \omterm{def:g2:air-around-us:air}{aire} removido es \omterm{def:g2:air-around-us:wind}{viento} --- así que hasta los
veleros funcionan con \omterm{def:g1:light-and-shadows:source}{luz} del Sol. ¿La leña? \omterm{def:g1:light-and-shadows:source}{Luz} del Sol almacenada
por los árboles. ¿El agua en alto de la presa? La levantó el Sol:
evaporó el mar y el vapor llovió sobre las montañas. Los paneles
solares se saltan a los intermediarios y ya está.
\end{example}

\begin{remark}[Una palabra que afilaremos]\label{rem:g5:everyday-energy:promise}
Por ahora, la \omterm{def:g5:everyday-energy:energy}{energía} es un relato bien contado: almacenes, formas y
cadenas. ¿Se puede \emph{\omterm{def:g2:measuring-length:measure}{medir}}, como la longitud en \omterm{def:g2:measuring-length:units}{metros} y la
\omterm{def:g3:measuring-mass:mass}{masa} en \omterm{def:g3:measuring-mass:units}{gramos}? Se puede --- la \omterm{def:g5:everyday-energy:energy}{energía} tiene su propia
\omterm{def:g2:measuring-length:measure}{unidad} y su propia contabilidad exacta, y las dos llegan más
adelante en este libro, cuando hayas conocido como es debido la
velocidad y la \omterm{def:g2:pushes-and-pulls:force}{fuerza}. Las cadenas que sigues este año son el mapa;
los números vendrán a vivir en él.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:1}
¿Qué es la \omterm{def:g5:everyday-energy:energy}{energía}, con las palabras de este capítulo? Nombra cuatro
cosas que no puedan ocurrir sin ella.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:2}
Encuentra el almacén: un ratón de cuerda que corretea; \ un ciclista;
\ una hoguera; \ un velero; \ una linterna.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:3}
Nombra cinco formas de la \omterm{def:g5:everyday-energy:energy}{energía}, con un ejemplo de cada una sacado
de tu propio día.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:4}
Sigue la cadena de \omterm{def:g5:everyday-energy:energy}{energía} de una tostadora, como en el
\cref{met:g5:everyday-energy:read}: almacén, convertidor y formas que
salen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:5}
Sigue la cadena desde el embalse de una presa hasta tu lámpara de leer
de esta noche.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:6}
Un teléfono ``se muere''. ¿Qué se ha agotado en realidad? ¿Adónde se
fue esa \omterm{def:g5:everyday-energy:energy}{energía}, según la proposición de la cadena?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:7}
¿Por qué se calienta un portátil por debajo mientras lo usas? ¿Qué
eslabón de la cadena se está dejando ver?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:everyday-energy:8}
Demuestra que un velero funciona con \omterm{def:g1:light-and-shadows:source}{luz} del Sol, siguiendo su cadena
hacia atrás paso a paso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:everyday-energy:9}
Un inventor anuncia una lámpara que ``no necesita \omterm{def:g3:first-electric-circuit:battery}{pila}, ni enchufe,
ni combustible --- sencillamente brilla para siempre''. ¿Qué lección
de la \cref{rem:g5:everyday-energy:nofree} desafía el anuncio? ¿Qué
pregunta deberías hacerle al inventor lo primero?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:everyday-energy:10}
Una pelota de goma soltada a la altura del hombro rebota cada vez más
bajo y acaba quieta. ``Su \omterm{def:g5:everyday-energy:energy}{energía} se ha destruido'', dice Leo. Usa la
\cref{prop:g5:everyday-energy:chain} para defender un final mejor para
la historia --- ¿en qué forma final se coló la \omterm{def:g5:everyday-energy:energy}{energía} de los
rebotes?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:everyday-energy:11}
Algunas casas calientan agua en paneles negros del tejado; otras
cocinan con \omterm{def:g1:light-and-shadows:source}{luz} del Sol y espejos curvos; las células solares fabrican
electricidad a partir de la \omterm{def:g1:light-and-shadows:source}{luz} del día. Explica la frase ``los
paneles solares se saltan a los intermediarios'' del
\cref{ex:g5:everyday-energy:sun}: ¿qué intermediarios hay entre el Sol
y el calor de una fogata de leña?
\end{exercise}
