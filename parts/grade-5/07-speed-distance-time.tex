\chapter{Speed: Distance and Time}\label{ch:g5:speed-distance-time}

``I'm faster!'' --- ``No, I am!'' Every playground settles this the
honest way: line up, ready, set, go. But how do you compare a
cyclist with a runner who raced yesterday, on another road? \omterm{def:g5:speed-distance-time:speed}{Speed}
is how physics settles races that can never be run side by side.

\section{Two ways to win a race}

\begin{proposition}[The fair race rules]\label{prop:g5:speed-distance-time:rules}
Two honest ways to compare fast and slow:
\begin{enumerate}
  \item over the \emph{same distance}, the faster one needs
        \emph{less time} --- first across the line wins the sprint;
  \item in the \emph{same time}, the faster one covers \emph{more
        distance} --- after one minute, the faster swimmer is
        simply farther.
\end{enumerate}
Same distance or same time: fix one, compare the other. Comparing
with \emph{both} different --- ``I ran farther!'' ``But you ran all
afternoon!'' --- decides nothing.
\end{proposition}

\begin{example}[Unfair and fair]\label{ex:g5:speed-distance-time:fair}
Lena cycled \qty{10}{km}; Marco cycled \qty{15}{km}. Is Marco
faster? No way to tell --- perhaps he rode three times as long.
But if both rode for exactly one hour, the comparison turns fair:
same time, more distance, Marco is faster. That little ``in one
hour'' is about to become our favorite phrase.
\end{example}

\section{Speed: the distance in one hour}

\begin{definition}[Speed]\label{def:g5:speed-distance-time:speed}
The \emph{speed}\index{speed} of a steady traveler is the distance
it covers in one fixed helping of time --- usually one hour.
``This car travels $60$ kilometres in each hour'' names its speed;
a brisk walker manages about $5$ kilometres in each hour, a
racing cyclist $40$. For crawlers we use friendlier helpings: a
snail covers about $5$ \omterm{def:g2:measuring-length:units}{metres} in each hour.
\end{definition}

\begin{method}[Measuring your own speed]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item Measure or find a stretch of exactly \qty{100}{m} (many
        sports tracks mark one);
  \item walk it at your natural pace while a friend times you ---
        suppose it takes one minute;
  \item one hour holds $60$ of those minutes, so in an hour you
        would cover $60 \times 100 = \qty{6000}{m}$;
  \item said in kilometres: your walking \omterm{def:g5:speed-distance-time:speed}{speed} is $6$ kilometres in
        each hour.
\end{enumerate}
Repeat at a run and compare --- most children roughly double.
\end{method}

\section{Predicting distances}

\begin{proposition}[Distance from speed]\label{prop:g5:speed-distance-time:distance}
A steady traveler covers its one-hour distance again in every
further hour. So:
\[
\text{distance covered} = \text{distance in one hour} \times
\text{number of hours}.
\]
A car doing $60$ kilometres each hour covers, in $3$ \omterm{ex:g2:measuring-time:clock}{hours},
$60 \times 3 = \qty{180}{km}$.
\end{proposition}

\begin{example}[Journey sums]\label{ex:g5:speed-distance-time:journeys}
The walker at $4$ kilometres each hour, out for $2$ \omterm{ex:g2:measuring-time:clock}{hours}:
$4 \times 2 = \qty{8}{km}$. The cyclist at $15$ kilometres each
hour, riding $4$ \omterm{ex:g2:measuring-time:clock}{hours}: $15 \times 4 = \qty{60}{km}$. The airliner
at $900$ kilometres each hour, flying $5$ \omterm{ex:g2:measuring-time:clock}{hours}: $900 \times 5 =
\qty{4500}{km}$ --- a continent crossed between two meals.
\end{example}

\begin{remark}[Steady is a simplification]\label{rem:g5:speed-distance-time:steady}
Real travelers are rarely steady: the car slows in villages, the
cyclist flies downhill and toils up, you dawdle past the bakery.
Our sums pretend the whole journey runs at one faithful pace ---
a useful pretense, good enough for planning. How to be honest
about a journey of many paces --- and what ``average'' really
means --- is a story for a later year, once you own the right
mathematics.
\end{remark}

\begin{example}[Reading speeds around you]\label{ex:g5:speed-distance-time:signs}
Road signs and car dashboards speak in kilometres-per-one-hour ---
the round sign saying $50$ tells drivers: cover at most $50$
kilometres in each hour, a pace chosen so that a child chasing a
ball can still be spared. The \omterm{def:g5:speed-distance-time:speed}{high-speed} train's screen may boast
$300$; walkers' signposts in the mountains prefer \omterm{ex:g2:measuring-time:clock}{hours} over
kilometres --- ``lake: 2 h'' --- trusting the walker to know their
own \omterm{def:g5:speed-distance-time:speed}{speed}.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
State the two fair ways to compare fast and slow. Why does ``I ran
farther than you'' alone prove nothing?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
Two swimmers cross the same pool; Ana needs $30$ seconds, Bo needs
$25$. Who is faster, and by which fair-race rule?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
After one hour, a tractor has covered \qty{20}{km} and a scooter
\qty{45}{km}. Who is faster, and by which rule?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
What does ``this train travels $200$ kilometres in each hour''
mean? How far does it get in $2$ \omterm{ex:g2:measuring-time:clock}{hours}? In half an hour?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
Rank by \omterm{def:g5:speed-distance-time:speed}{speed}: a walker ($5$ kilometres each hour); \ a pigeon
($80$ kilometres each hour); \ a snail ($5$ \emph{\omterm{def:g2:measuring-length:units}{metres}} each
hour); \ a city bus ($30$ kilometres each hour).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
You stroll a marked \qty{100}{m} in $2$ minutes. How many \omterm{def:g2:measuring-length:units}{metres}
would you cover in an hour at that pace? Say your \omterm{def:g5:speed-distance-time:speed}{speed} in
kilometres in each hour.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
A car keeps a steady $80$ kilometres each hour for $3$ \omterm{ex:g2:measuring-time:clock}{hours}. How
far does it travel? And a cyclist at $15$ kilometres each hour for
the same $3$ \omterm{ex:g2:measuring-time:clock}{hours}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
On the journey graph of the chapter, how do you spot the faster
traveler at a glance? Read from it: how far has the cyclist gone
after $2$ \omterm{ex:g2:measuring-time:clock}{hours}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
A ferry covers $30$ kilometres in each hour. The crossing is
\qty{90}{km}. Using
\cref{prop:g5:speed-distance-time:distance} backwards --- try
\omterm{ex:g2:measuring-time:clock}{hours} one by one --- how long does the crossing take?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
Two towns are \qty{60}{km} apart. Mara cycles from one at $20$
kilometres each hour; at the same moment Jon cycles from the other
toward her at $10$ kilometres each hour. How many kilometres does
the gap between them shrink each hour? After how many \omterm{ex:g2:measuring-time:clock}{hours} do they
meet?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
The old fable, with numbers: the race is \qty{120}{m} long. The
hare runs $60$ \omterm{def:g2:measuring-length:units}{metres} in each minute --- but after one minute of
running it stops for a $70$-minute nap, then runs on. The tortoise
plods $2$ \omterm{def:g2:measuring-length:units}{metres} in each minute and never stops. Where is the hare
when it starts its nap? At what time does each cross the finish
line? Who wins?
\end{exercise}
