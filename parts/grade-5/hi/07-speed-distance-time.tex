\chapter{चाल: दूरी और समय}\label{ch:g5:speed-distance-time}

``मैं ज़्यादा तेज़ हूँ!'' --- ``नहीं, मैं हूँ!'' हर मैदान इस झगड़े को
ईमानदार तरीक़े से निपटाता है: क़तार में लगो, तैयार, ध्यान, चलो। पर उस
साइकिल सवार की तुलना उस दौड़ने वाले से कैसे करोगे जो कल, किसी और सड़क
पर दौड़ा था? \omterm{def:g5:speed-distance-time:speed}{चाल} वही तरीक़ा है जिससे भौतिकी उन दौड़ों का फ़ैसला करती है
जो कभी अगल-बग़ल हो ही नहीं सकतीं।

\section{दौड़ जीतने के दो तरीक़े}

\begin{proposition}[निष्पक्ष दौड़ के नियम]\label{prop:g5:speed-distance-time:rules}
तेज़ और धीमे की तुलना करने के दो ईमानदार तरीक़े:
\begin{enumerate}
  \item \emph{एक ही दूरी} पर, ज़्यादा तेज़ वाले को \emph{कम समय}
        लगता है --- फ़ीते तक पहले पहुँचने वाला दौड़ जीत जाता है;
  \item \emph{एक ही समय} में, ज़्यादा तेज़ वाला \emph{ज़्यादा दूरी}
        तय करता है --- एक मिनट बाद तेज़ तैराक बस आगे होता है।
\end{enumerate}
या तो दूरी एक जैसी रखो या समय: एक को पक्का करो, और दूसरे की तुलना
करो। \emph{दोनों} अलग-अलग रखकर तुलना करना --- ``मैं ज़्यादा दूर
दौड़ा!'' ``पर तुम तो पूरी दोपहर दौड़ते रहे!'' --- कुछ भी तय नहीं
करता।
\end{proposition}

\begin{example}[नाइंसाफ़ी और इंसाफ़]\label{ex:g5:speed-distance-time:fair}
लीना ने \qty{10}{km} साइकिल चलाई; मार्को ने \qty{15}{km}। तो क्या
मार्को ज़्यादा तेज़ है? बताया ही नहीं जा सकता --- हो सकता है उसने तीन
गुना समय लगाया हो। पर अगर दोनों ने ठीक एक-एक घंटा चलाई हो, तो तुलना
निष्पक्ष हो जाती है: समय वही, दूरी ज़्यादा, यानी मार्को ज़्यादा तेज़
है। वह छोटा-सा ``एक घंटे में'' अब हमारा सबसे पसंदीदा वाक्यांश बनने
वाला है।
\end{example}

\section{चाल: एक घंटे की दूरी}

\begin{definition}[चाल]\label{def:g5:speed-distance-time:speed}
किसी एक-सी रफ़्तार से चलने वाले यात्री की \emph{चाल}\index{चाल} वह
दूरी है जो वह समय के एक तय हिस्से में तय करता है --- आम तौर पर एक
घंटे में। ``यह गाड़ी हर घंटे में $60$ किलोमीटर चलती है'' उसकी चाल
बताता है; तेज़ चलने वाला पैदल यात्री हर घंटे में लगभग $5$ किलोमीटर तय
कर लेता है, और दौड़ का साइकिल सवार $40$। रेंगने वालों के लिए हम
ज़्यादा दोस्ताना हिस्से चुन लेते हैं: घोंघा हर घंटे में लगभग $5$ \omterm{def:g2:measuring-length:units}{मीटर}
तय करता है।
\end{definition}

\begin{method}[अपनी चाल मापना]\label{met:g5:speed-distance-time:measure}
\begin{enumerate}
  \item ठीक \qty{100}{m} का कोई हिस्सा नापो या ढूँढ़ो (बहुत-से खेल के
        मैदानों पर निशान लगा होता है);
  \item अपनी सहज \omterm{def:g5:speed-distance-time:speed}{चाल} से उसे पार करो जबकि कोई साथी तुम्हारा समय ले
        --- मान लो एक मिनट लगा;
  \item एक घंटे में ऐसे $60$ मिनट होते हैं, इसलिए एक घंटे में तुम
        $60 \times 100 = \qty{6000}{m}$ तय करते;
  \item किलोमीटर में कहो: तुम्हारे चलने की \omterm{def:g5:speed-distance-time:speed}{चाल} हर घंटे में $6$
        किलोमीटर है।
\end{enumerate}
अब दौड़कर यही दोहराओ और तुलना करो --- ज़्यादातर बच्चों की \omterm{def:g5:speed-distance-time:speed}{चाल} मोटे
तौर पर दोगुनी हो जाती है।
\end{method}

\section{दूरियों का अंदाज़ा लगाना}

\begin{proposition}[चाल से दूरी]\label{prop:g5:speed-distance-time:distance}
एक-सी रफ़्तार वाला यात्री हर अगले घंटे में फिर से उतनी ही दूरी तय करता
है जितनी उसने एक घंटे में की थी। इसलिए:
\[
\text{तय की गई दूरी} = \text{एक घंटे की दूरी} \times
\text{घंटों की संख्या}.
\]
हर घंटे $60$ किलोमीटर चलने वाली गाड़ी $3$ \omterm{ex:g2:measuring-time:clock}{घंटों} में
$60 \times 3 = \qty{180}{km}$ तय करती है।
\end{proposition}

\begin{example}[सफ़र के सवाल]\label{ex:g5:speed-distance-time:journeys}
हर घंटे $4$ किलोमीटर चलने वाला पैदल यात्री, $2$ घंटे बाहर रहने पर:
$4 \times 2 = \qty{8}{km}$। हर घंटे $15$ किलोमीटर चलने वाला साइकिल
सवार, $4$ घंटे चलाने पर: $15 \times 4 = \qty{60}{km}$। हर घंटे $900$
किलोमीटर उड़ने वाला यात्री विमान, $5$ घंटे उड़ने पर: $900 \times 5 =
\qty{4500}{km}$ --- यानी दो भोजनों के बीच एक पूरा महाद्वीप पार।
\end{example}

\begin{remark}[एक-सी रफ़्तार एक सरलीकरण है]\label{rem:g5:speed-distance-time:steady}
असली यात्री शायद ही कभी एक-सी रफ़्तार से चलते हैं: गाड़ी गाँवों में
धीमी हो जाती है, साइकिल सवार ढलान पर उड़ता है और चढ़ाई पर हाँफता है,
और तुम हलवाई की दुकान के पास सुस्ता लेते हो। हमारे सवाल यह मान लेते
हैं कि पूरा सफ़र एक ही वफ़ादार रफ़्तार से चला --- यह मान लेना काम का
है और योजना बनाने के लिए काफ़ी अच्छा। कई-कई रफ़्तारों वाले सफ़र के
बारे में ईमानदारी से कैसे बोला जाए --- और ``औसत'' का असली मतलब क्या है
--- यह कहानी किसी आगे वाले \omterm{def:g3:sun-days-seasons:year}{वर्ष} की है, जब तुम्हारे पास ज़रूरी गणित आ
चुका होगा।
\end{remark}

\begin{example}[आस-पास की चालें पढ़ना]\label{ex:g5:speed-distance-time:signs}
सड़क के चिह्न और गाड़ियों के पटल ``एक घंटे में इतने किलोमीटर'' की
भाषा बोलते हैं --- $50$ लिखा गोल चिह्न चालकों से कहता है: हर घंटे में
ज़्यादा से ज़्यादा $50$ किलोमीटर चलो, यानी वह रफ़्तार जो इसलिए चुनी गई
है कि गेंद के पीछे भागते किसी बच्चे को बचाया जा सके। तेज़ रेलगाड़ी का
परदा शायद $300$ की डींग मारे; और पहाड़ों में पैदल यात्रियों के दिशा-चिह्न
किलोमीटर से ज़्यादा घंटे पसंद करते हैं --- ``झील: 2 घंटे'' --- यह भरोसा
करते हुए कि यात्री अपनी \omterm{def:g5:speed-distance-time:speed}{चाल} ख़ुद जानता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:1}
तेज़ और धीमे की तुलना के दोनों निष्पक्ष तरीक़े बताओ। अकेला ``मैं
तुमसे ज़्यादा दूर दौड़ा'' कुछ भी साबित क्यों नहीं करता?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:2}
दो तैराक एक ही तरण-ताल पार करते हैं; अना को $30$ सेकंड लगते हैं, बो
को $25$। कौन ज़्यादा तेज़ है, और निष्पक्ष दौड़ के किस नियम से?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:3}
एक घंटे बाद ट्रैक्टर ने \qty{20}{km} और स्कूटर ने \qty{45}{km} तय
कर लिए हैं। कौन ज़्यादा तेज़ है, और किस नियम से?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:4}
``यह रेलगाड़ी हर घंटे में $200$ किलोमीटर चलती है'' का क्या मतलब है?
वह $2$ \omterm{ex:g2:measuring-time:clock}{घंटों} में कितनी दूर पहुँचती है? और आधे घंटे में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:5}
\omterm{def:g5:speed-distance-time:speed}{चाल} के हिसाब से क्रम में लगाओ: पैदल यात्री (हर घंटे $5$ किलोमीटर);
\ कबूतर (हर घंटे $80$ किलोमीटर); \ घोंघा (हर घंटे $5$
\emph{\omterm{def:g2:measuring-length:units}{मीटर}}); \ शहर की बस (हर घंटे $30$ किलोमीटर)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:6}
तुम निशान लगे \qty{100}{m} को $2$ मिनट में टहलकर पार करते हो। उसी
रफ़्तार से एक घंटे में तुम कितने \omterm{def:g2:measuring-length:units}{मीटर} तय करते? अपनी \omterm{def:g5:speed-distance-time:speed}{चाल} हर घंटे के
किलोमीटरों में बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:7}
एक गाड़ी $3$ \omterm{ex:g2:measuring-time:clock}{घंटों} तक हर घंटे $80$ किलोमीटर की एक-सी रफ़्तार रखती है।
वह कितनी दूर जाती है? और उन्हीं $3$ \omterm{ex:g2:measuring-time:clock}{घंटों} में हर घंटे $15$ किलोमीटर
चलने वाला साइकिल सवार?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:speed-distance-time:8}
अध्याय के सफ़र वाले आलेख में ज़्यादा तेज़ यात्री को एक नज़र में कैसे
पहचानोगे? उसी से पढ़ो: $2$ घंटे बाद साइकिल सवार कितनी दूर पहुँच चुका
है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:9}
एक नाव हर घंटे $30$ किलोमीटर तय करती है। पार करने की दूरी
\qty{90}{km} है। \cref{prop:g5:speed-distance-time:distance} को
उलटी दिशा में काम में लेकर --- यानी एक-एक घंटा जोड़कर देखो ---
बताओ कि पार करने में कितना समय लगेगा।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:10}
दो क़स्बे \qty{60}{km} दूर हैं। मारा एक से हर घंटे $20$ किलोमीटर की
\omterm{def:g5:speed-distance-time:speed}{चाल} से साइकिल चलाती है; ठीक उसी पल जॉन दूसरे से उसकी ओर हर घंटे $10$
किलोमीटर की \omterm{def:g5:speed-distance-time:speed}{चाल} से चलता है। उनके बीच का फ़ासला हर घंटे कितने किलोमीटर
घटता है? कितने \omterm{ex:g2:measuring-time:clock}{घंटों} बाद वे मिलेंगे?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:speed-distance-time:11}
पुरानी कहानी, अब संख्याओं के साथ: दौड़ \qty{120}{m} लंबी है। ख़रगोश
हर मिनट $60$ \omterm{def:g2:measuring-length:units}{मीटर} दौड़ता है --- पर एक मिनट दौड़ने के बाद वह $70$ मिनट
की नींद लेने रुक जाता है, फिर आगे दौड़ता है। कछुआ हर मिनट $2$ \omterm{def:g2:measuring-length:units}{मीटर}
चलता है और कभी नहीं रुकता। नींद शुरू करते समय ख़रगोश कहाँ है? दोनों
में से हर एक फ़ीते को किस समय पार करता है? जीतता कौन है?
\end{exercise}
