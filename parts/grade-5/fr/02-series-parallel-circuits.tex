\chapter{Circuits en série et en dérivation}\label{ch:g5:series-parallel-circuits}

Quand une lampe grillait, toute la guirlande s'éteignait --- la
malédiction de la boucle unique. Pourtant, quand une lampe grille
dans ta cuisine, le reste de la maison continue de briller.
Quelqu'un a câblé ta maison d'une façon plus astucieuse. C'est ce que
nous apprenons aujourd'hui~: la voie des \omterm{def:g5:series-parallel-circuits:parallel}{branches}.

\section{Deux façons de câbler}

Tu connais la première~: \emph{\omterm{def:g4:series-circuits:series}{en série}}, tous les joueurs enfilés
sur une seule boucle, comme des perles sur un collier. Voici la
seconde.

\begin{definition}[En dérivation]\label{def:g5:series-parallel-circuits:parallel}
Des joueurs sont \emph{en dérivation}\index{circuit en dérivation}
--- on dit aussi en parallèle --- quand le circuit se sépare en
\emph{branches}\index{branche} distinctes --- un joueur par
branche --- qui s'écartent puis se rejoignent. Les points où les
routes se séparent ou se rejoignent s'appellent des
\emph{nœuds}\index{nœud}. Chaque branche est une route complète d'une
\omterm{def:g3:first-electric-circuit:battery}{borne} de la \omterm{def:g3:first-electric-circuit:battery}{pile} à l'autre.
\end{definition}

\begin{method}[Le construire]\label{met:g5:series-parallel-circuits:build}
Avec une \omterm{def:g3:first-electric-circuit:battery}{pile}, deux lampes et des fils~:
\begin{enumerate}
  \item câbler la première lampe à la \omterm{def:g3:first-electric-circuit:battery}{pile}, seule, en boucle
        simple --- \omterm{def:g5:series-parallel-circuits:parallel}{branche} numéro un~;
  \item câbler maintenant la seconde lampe \emph{directement en
        travers} de la première~: un fil de chaque côté de la lampe 1
        vers le côté correspondant de la lampe 2 --- \omterm{def:g5:series-parallel-circuits:parallel}{branche} numéro
        deux~;
  \item suivre chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} du doigt~: en partant du $+$, chaque
        route doit traverser exactement une lampe avant d'atteindre
        le $-$.
\end{enumerate}
Les deux lampes s'allument --- et chacune brille aussi fort qu'une
lampe seule.
\end{method}

\section{Les règles des branches}

\begin{proposition}[Les règles du circuit en
dérivation]\label{prop:g5:series-parallel-circuits:rules}
Chaque joueur étant sur sa propre \omterm{def:g5:series-parallel-circuits:parallel}{branche}~:
\begin{enumerate}
  \item \emph{indépendance}~: une coupure dans une \omterm{def:g5:series-parallel-circuits:parallel}{branche} n'arrête
        que cette \omterm{def:g5:series-parallel-circuits:parallel}{branche} --- dévisse une lampe, et l'autre continue
        de briller~;
  \item \emph{plein éclat}~: chaque lampe brille comme si elle était
        seule --- pas de partage de la poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile} entre les
        \omterm{def:g5:series-parallel-circuits:parallel}{branches}~;
  \item \emph{postes de commande}~: un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} placé dans une
        \omterm{def:g5:series-parallel-circuits:parallel}{branche} ne commande que cette \omterm{def:g5:series-parallel-circuits:parallel}{branche}~; un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} placé
        sur la route commune, avant la séparation, commande tout le
        monde. (Le prix~: la \omterm{def:g3:first-electric-circuit:battery}{pile} alimente toutes les \omterm{def:g5:series-parallel-circuits:parallel}{branches} à la
        fois, et se vide donc plus vite.)
\end{enumerate}
\end{proposition}

\begin{example}[L'essai du dévissage, rejoué]\label{ex:g5:series-parallel-circuits:test}
Fais à la paire \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} exactement ce qui avait tué la paire \omterm{def:g4:series-circuits:series}{en
série}~: dévisse la lampe 2. La lampe 1 ne vacille même pas --- sa
propre \omterm{def:g5:series-parallel-circuits:parallel}{branche}, sa route complète à elle, est intacte. Revisse-la, et
dévisse la lampe 1 à la place~: la lampe 2 continue tranquillement de
briller. La malédiction de la boucle unique est levée~: \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}, chaque joueur s'occupe de sa branche.
\end{example}

\begin{example}[Ta maison est un circuit en
dérivation]\label{ex:g5:series-parallel-circuits:house}
Chaque lampe, chaque prise et chaque appareil de ta maison monte sur
sa propre \omterm{def:g5:series-parallel-circuits:parallel}{branche} d'un grand \omterm{def:g5:series-parallel-circuits:parallel}{circuit en dérivation}. Voilà pourquoi la
lampe de la cuisine peut griller pendant que le salon continue de
lire sans être dérangé~; pourquoi l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} de chaque pièce ne
commande que sa propre \omterm{def:g1:light-and-shadows:source}{lumière}~; et pourquoi un unique \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}
général --- dans le tableau électrique, près du compteur --- peut
couper toute la maison d'un coup pour les réparations. Les guirlandes
modernes ont rejoint le club~: leurs lampes sont câblées de façon
qu'une lampe grillée n'éteigne plus la guirlande.
\end{example}

\begin{remark}[Série et dérivation, côte à côte]\label{rem:g5:series-parallel-circuits:compare}
Une coupure~: \omterm{def:g4:series-circuits:series}{en série}, tout s'éteint~; \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}, une seule
\omterm{def:g5:series-parallel-circuits:parallel}{branche} s'éteint. Plus de lampes~: \omterm{def:g4:series-circuits:series}{en série}, toutes plus faibles~; \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}, toutes à plein éclat, mais la \omterm{def:g3:first-electric-circuit:battery}{pile} se vide plus vite. Un
seul \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} pour tout~: \omterm{def:g4:series-circuits:series}{en série}, n'importe où~; \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation},
seulement avant la séparation. Aucune des deux façons n'est
«~meilleure~»~: la lampe de poche est très heureuse \omterm{def:g4:series-circuits:series}{en série} (une
lampe, un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}), la maison est nécessairement \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}.
La première question d'un électricien devant un circuit est
simplement~: \emph{où la route se sépare-t-elle~?}
\end{remark}

\section{Lire les circuits}

\begin{method}[Suivre les routes]\label{met:g5:series-parallel-circuits:follow}
Pour prédire le sort de n'importe quelle lampe sur un schéma~:
\begin{enumerate}
  \item poser le doigt sur le $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} et parcourir toutes les
        routes possibles jusqu'au $-$, en énumérant les joueurs que
        chaque route traverse~;
  \item une lampe s'allume s'il existe au moins une route
        ininterrompue qui la traverse~;
  \item les joueurs traversés par toutes les routes sont \omterm{def:g4:series-circuits:series}{en série}
        les uns avec les autres~; les joueurs situés sur des
        embranchements différents sont \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}.
\end{enumerate}
Parcours les routes avant de prédire --- deviner d'après la seule
allure du schéma, c'est ainsi que les circuits trompent les gens.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:1}
Qu'est-ce qu'une \omterm{def:g5:series-parallel-circuits:parallel}{branche}~? Un \omterm{def:g5:series-parallel-circuits:parallel}{nœud}~? Dans un \omterm{def:g5:series-parallel-circuits:parallel}{circuit en dérivation} à
deux lampes, combien de \omterm{def:g5:series-parallel-circuits:parallel}{branches} portent une lampe~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:2}
Énonce les trois règles du \omterm{def:g5:series-parallel-circuits:parallel}{circuit en dérivation}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:3}
Deux lampes \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}~: la lampe 1 est dévissée. Que fait la
lampe 2~? Et dans la paire \omterm{def:g4:series-circuits:series}{en série} --- que faisait-elle là-bas, et
d'où vient la différence~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:4}
Deux lampes identiques~: la paire A est \omterm{def:g4:series-circuits:series}{en série}, la paire B \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}, avec les mêmes \omterm{def:g3:first-electric-circuit:battery}{piles}. Quelle paire brille le plus fort
par lampe~? Laquelle vide sa \omterm{def:g3:first-electric-circuit:battery}{pile} le plus vite~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:5}
Sur la figure aux deux \omterm{ex:g3:first-electric-circuit:switch}{interrupteurs}, que fait chaque lampe quand~:
les deux \omterm{ex:g3:first-electric-circuit:switch}{interrupteurs} sont fermés~? \ le général est ouvert~? \ le
général est fermé et l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} de \omterm{def:g5:series-parallel-circuits:parallel}{branche} ouvert~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:6}
Pourquoi une maison est-elle câblée \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} et non \omterm{def:g4:series-circuits:series}{en série}~?
Donne deux conforts de tous les jours qui en dépendent.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:7}
À quel endroit du circuit d'une maison un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} commande-t-il
tout à la fois~? Quel est le nom courant de cet \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}, et à
quoi sert-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:series-parallel-circuits:8}
Une lampe de chevet et sa prise murale~: sont-elles \omterm{def:g4:series-circuits:series}{en série} ou \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation} avec le plafonnier~? Comment le sais-tu, par la vie de
tous les jours~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:9}
Dessine (avec des symboles) le circuit d'un couloir~: une \omterm{def:g3:first-electric-circuit:battery}{pile}, une
lampe, et ce confort --- la lampe doit pouvoir s'allumer et
s'éteindre depuis un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} placé à \emph{l'une ou l'autre}
extrémité du couloir\dots\ Essaie honnêtement~; si tes montages
échouent obstinément, explique ce qui est difficile. (Un vrai
va-et-vient demande un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} plus astucieux que le nôtre ---
la tentative apprend plus que la réponse.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:10}
Une guirlande moderne continue de briller quand une lampe grille ---
mais la boîte prévient~: «~Ne pas faire fonctionner la guirlande avec
plus de trois lampes retirées.~» À l'aide des règles des \omterm{def:g5:series-parallel-circuits:parallel}{branches},
explique le confort et, plus difficile, une raison sensée de cet
avertissement. (Pense à ce que reçoit chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} restante de la
part de la \omterm{def:g3:first-electric-circuit:battery}{pile}, et à l'effort de la \omterm{def:g3:first-electric-circuit:battery}{pile} elle-même.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:series-parallel-circuits:11}
Sur la figure aux deux \omterm{ex:g3:first-electric-circuit:switch}{interrupteurs}, un électricien veut ajouter une
lampe de plus qui s'allume \emph{seulement} quand l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} de
\omterm{def:g5:series-parallel-circuits:parallel}{branche} est fermé, et reste éteinte sinon --- même quand
l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} général est fermé. Sur quel tronçon de route la
nouvelle lampe doit-elle être câblée, et se retrouve-t-elle alors \omterm{def:g4:series-circuits:series}{en
série} ou \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} avec la lampe 2~?
\end{exercise}
