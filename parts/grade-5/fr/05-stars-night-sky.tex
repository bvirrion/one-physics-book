\chapter{Les étoiles et le ciel nocturne}\label{ch:g5:stars-night-sky}

Loin des lampadaires de la ville, par une nuit claire et sans lune,
le ciel est bondé~: des milliers d'étincelles, faibles ou vives,
blanches ou orangées. Les marins s'y sont guidés, les paysans y ont
réglé leurs semailles, et toutes les sciences ont commencé par s'en
émerveiller. Que sont-elles --- et pourquoi tournent-elles si
sagement au-dessus de nos têtes pendant que nous dormons~?

\section{Ce qu'est une étoile}

\begin{example}[Des soleils innombrables]\label{ex:g5:stars-night-sky:suns}
Chaque \omterm{def:g4:solar-system:star}{étoile} est un soleil~: une gigantesque boule de matière
incandescente qui déverse sa propre \omterm{def:g1:light-and-shadows:source}{lumière} --- beaucoup sont plus
puissantes que notre Soleil. Elles n'ont l'\omterm{def:g2:air-around-us:air}{air} que de modestes
étincelles pour une seule raison~: la distance. Le Soleil est notre
\omterm{def:g4:solar-system:star}{étoile} d'à côté~; les autres sont si loin que même la \omterm{def:g1:light-and-shadows:source}{lumière}, la
championne de la course qui traverse une pièce en un rien de temps,
passe des \emph{années} sur la route depuis la plus proche d'entre
elles. Le ciel nocturne est une foule de soleils, rapetissés par la
distance jusqu'à n'être que des paillettes.
\end{example}

\begin{example}[Les couleurs des étoiles]\label{ex:g5:stars-night-sky:colors}
Regarde attentivement~: les étincelles ne sont pas toutes blanches.
Certaines brûlent d'un orange rougeoyant, d'autres sont jaunes comme
notre Soleil, d'autres encore blanc-bleu. La couleur est un message
sur la chaleur --- et il va à l'inverse des robinets et des pots de
peinture~: les \omterm{def:g4:solar-system:star}{étoiles} orangées sont les \emph{plus froides} de la
famille, et les blanc-bleu les plus chaudes, bien plus chaudes que le
Soleil. Un charbon incandescent et une lame d'acier chauffée à blanc
racontent la même histoire ici, sur Terre~: plus c'est chaud, plus
c'est blanc.
\end{example}

\begin{remark}[Où vont les étoiles le jour]\label{rem:g5:stars-night-sky:daytime}
Les \omterm{def:g4:solar-system:star}{étoiles} ne s'en vont jamais. Elles brillent au-dessus de toi à
midi exactement comme à minuit --- mais le ciel du jour est inondé de
\omterm{def:g1:light-and-shadows:source}{lumière} solaire diffusée par l'\omterm{def:g2:air-around-us:air}{air}, un éclat bleu bien plus vif que
n'importe quelle étincelle. Une bougie est invisible à côté d'un feu
de joie. Quand le Soleil se couche et que l'éclat s'écoule, la foule
permanente du ciel redevient tout simplement visible.
\end{remark}

\section{Des cartes faites d'étoiles}

\begin{definition}[Constellation]\label{def:g5:stars-night-sky:constellation}
Une \emph{constellation}\index{constellation} est une figure d'\omterm{def:g4:solar-system:star}{étoiles}
brillantes portant un nom --- un chasseur, un cygne, une grande
ourse --- inventée il y a très longtemps par les observateurs du ciel
comme carte et aide-mémoire. Les \omterm{def:g4:solar-system:star}{étoiles} d'une constellation ne sont
pas de vraies voisines~; elles ne font que s'aligner vues depuis la
Terre. Mais les figures sont stables~: tes grands-parents ont appris
exactement les mêmes.
\end{definition}

\begin{method}[Trouver l'étoile Polaire]\label{met:g5:stars-night-sky:northstar}
Le ciel offre une \omterm{def:g4:magnets-and-compass:compass}{boussole} gratuite~:
\begin{enumerate}
  \item repérer la Grande Ourse --- sept \omterm{def:g4:solar-system:star}{étoiles} brillantes, une
        casserole au long manche, facile à trouver toute l'année~;
  \item prendre les deux \omterm{def:g4:solar-system:star}{étoiles} du bord avant de la casserole ---
        les \emph{pointeuses}~;
  \item prolonger la droite qui va de l'une à l'autre, en travers du
        ciel, sur environ cinq fois leur écartement~;
  \item la modeste \omterm{def:g4:solar-system:star}{étoile} où l'on arrive est l'\emph{étoile
        Polaire}\index{étoile Polaire} --- laisse tomber un fil à
        plomb depuis elle jusqu'à l'horizon, et voilà le nord.
\end{enumerate}
Vérifie avec ta \omterm{def:g4:magnets-and-compass:compass}{boussole}~: l'aiguille et l'\omterm{def:g4:solar-system:star}{étoile} sont d'accord. Les
marins ont joué leur vie sur cet accord pendant des siècles.
\end{method}

\section{Le ciel qui tourne}

\begin{proposition}[Le ciel tourne autour de l'étoile
Polaire]\label{prop:g5:stars-night-sky:wheel}
Au fil de la nuit, toute la foule des \omterm{def:g4:solar-system:star}{étoiles} tourne lentement à
travers le ciel en grands cercles --- tous autour d'un point presque
immobile~: l'\omterm{met:g5:stars-night-sky:northstar}{étoile Polaire}. Rien, là-haut, ne bouge vraiment avec
une telle discipline~; c'est notre Terre qui tourne, comme elle fait
défiler le Soleil dans le ciel du jour. L'\omterm{met:g5:stars-night-sky:northstar}{étoile Polaire} bouge à
peine parce qu'elle se trouve presque exactement au-dessus de l'axe
de rotation de la Terre --- la seule direction que la rotation laisse
en paix.
\end{proposition}

\begin{example}[Notre ville d'étoiles]\label{ex:g5:stars-night-sky:milkyway}
Les nuits les plus noires, une faible bande laiteuse s'arque en
travers du ciel~: la \emph{Voie lactée}\index{Voie lactée}. Des
jumelles dissolvent le lait en vérité~: d'innombrables \omterm{def:g4:solar-system:star}{étoiles}, trop
faibles et trop serrées pour être séparées à l'œil. C'est notre ville
natale d'\omterm{def:g4:solar-system:star}{étoiles} vue de l'intérieur --- une île plate et tourbillonnante
comptant plus de soleils qu'il n'y a de grains de sable sur une
plage, notre Soleil habitant dans une banlieue tranquille. Chaque
\omterm{def:g4:solar-system:star}{étoile} que tu vois à l'œil nu est une voisine de cette même ville.
\end{example}

\begin{remark}[Continue de lever les yeux]\label{rem:g5:stars-night-sky:lookup}
L'astronomie est la plus vieille des sciences, et la seule dont le
laboratoire ouvre gratuitement chaque nuit claire. Une couverture, un
coin sombre loin des lampadaires, de la patience pendant que les yeux
s'éveillent à l'obscurité --- voilà toute la liste du matériel. La
Grande Ourse, l'\omterm{met:g5:stars-night-sky:northstar}{étoile Polaire}, une \omterm{def:g4:solar-system:planet}{planète} ou deux et la \omterm{ex:g5:stars-night-sky:milkyway}{Voie lactée}
font une belle première expédition~; les lois plus profondes de la
\omterm{def:g1:light-and-shadows:source}{lumière} des \omterm{def:g4:solar-system:star}{étoiles} te trouveront bien préparé, dans quelques années,
vers la fin de ce cours.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:1}
Qu'est-ce qu'une \omterm{def:g4:solar-system:star}{étoile}~? Pourquoi les \omterm{def:g4:solar-system:star}{étoiles} ressemblent-elles à de
minuscules étincelles alors que beaucoup surpassent notre Soleil~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:2}
Les \omterm{def:g4:solar-system:star}{étoiles} sont-elles encore dans le ciel à midi~? Pourquoi ne
pouvons-nous pas les voir~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:3}
Une \omterm{def:g4:solar-system:star}{étoile} orangée et une \omterm{def:g4:solar-system:star}{étoile} blanc-bleu brillent côte à côte.
Laquelle est la plus chaude~? Quelle lueur de tous les jours raconte
la même histoire de couleurs~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:4}
Qu'est-ce qu'une \omterm{def:g5:stars-night-sky:constellation}{constellation}~? Ses \omterm{def:g4:solar-system:star}{étoiles} sont-elles vraiment
proches les unes des autres~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:5}
Décris, étape par étape, comment trouver l'\omterm{met:g5:stars-night-sky:northstar}{étoile Polaire} à partir de
la Grande Ourse.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:6}
Qu'a de particulier la place de l'\omterm{met:g5:stars-night-sky:northstar}{étoile Polaire} dans le ciel qui
tourne, et quel instrument de tous les jours remplace-t-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:7}
Les \omterm{def:g4:solar-system:star}{étoiles} tournent dans la nuit en cercles bien rangés. Qu'est-ce
qui tourne réellement~? Que fait encore défiler cette même rotation
dans le ciel du jour~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:stars-night-sky:8}
Que se révèle être la bande laiteuse des ciels les plus noirs, vue
aux jumelles~? Qu'est-ce vraiment~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:9}
Laissé seul toute la nuit, un appareil photo pointé sur l'\omterm{met:g5:stars-night-sky:northstar}{étoile
Polaire} enregistre des cercles~; pointé sur l'horizon est, il
enregistre des traînées obliques qui montent. Explique les deux
\omterm{prop:g5:mirrors-reflection:image}{images} avec une seule Terre qui tourne.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:10}
Une naufragée, par une nuit claire, n'a ni \omterm{def:g4:magnets-and-compass:compass}{boussole}, ni téléphone ---
et aucune raison de se presser. Explique comment le ciel lui donne à
la fois une direction et, avec de la patience, une horloge. (Deux
idées du chapitre, un seul ciel qui tourne.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:stars-night-sky:11}
La \omterm{def:g1:light-and-shadows:source}{lumière} d'une \omterm{def:g4:solar-system:star}{étoile} lointaine passe des années sur la route
jusqu'à ton œil. Quand tu regardes cette \omterm{def:g4:solar-system:star}{étoile} ce soir, la vois-tu
telle qu'elle est \emph{maintenant}~? Que regardes-tu en réalité ---
et pourquoi cela fait-il de tout télescope une sorte de machine à
remonter le temps~?
\end{exercise}
