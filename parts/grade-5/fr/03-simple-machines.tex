\chapter{Les machines simples~: leviers, poulies, plans inclinés}\label{ch:g5:simple-machines}

Pas de moteur, pas d'électricité --- et pourtant, avec quelques
poutres, quelques cordes et quelques rampes, les bâtisseurs de
l'Antiquité ont soulevé des pierres que des attelages entiers de
bœufs ne pouvaient pas traîner. Leur boîte à outils contenait les
\emph{\omterm{def:g5:simple-machines:machine}{machines simples}}~: des inventions sans le moindre moteur, qui
transforment une traction faible et confortable en une traction
puissante. Tu possèdes déjà la première~; fais connaissance avec la
famille.

\section{La famille des économiseurs d'effort}

\begin{definition}[Machine simple]\label{def:g5:simple-machines:machine}
Une \emph{machine simple}\index{machine simple} est un dispositif
sans moteur qui modifie une \omterm{def:g2:pushes-and-pulls:force}{force} à notre convenance~: en la rendant
plus intense, ou en changeant sa direction, ou les deux. Les
classiques~: le \emph{\omterm{def:g3:levers-and-scales:lever}{levier}} que tu connais par la bascule et la
barre à mine, la \emph{\omterm{def:g5:simple-machines:pulley}{poulie}}, et le \emph{\omterm{def:g5:simple-machines:ramp}{plan incliné}} --- la
modeste rampe.
\end{definition}

\begin{example}[Le levier, revisité]\label{ex:g5:simple-machines:lever}
Tu possèdes déjà la règle du \omterm{def:g3:levers-and-scales:lever}{levier}~: les pièces multipliées par les
graduations --- ta petite \omterm{def:g2:pushes-and-pulls:force}{force} loin du \omterm{def:g3:levers-and-scales:lever}{pivot} l'emporte sur une
grosse charge placée tout près. Remarque maintenant autre chose~: ton
extrémité de la barre à mine descend d'un long trajet pendant que le
rocher ne monte que d'un peu. Garde cette observation en poche~: elle
va devenir une loi.
\end{example}

\section{Les poulies}

\begin{definition}[Poulie]\label{def:g5:simple-machines:pulley}
Une \emph{poulie}\index{poulie} est une roue creusée d'une gorge où
passe une corde. Une poulie \emph{fixe} pend à une poutre et ne
change que la direction de la traction~: on tire vers le bas, la
charge monte. Une poulie \emph{mobile} chemine sur la corde avec la
charge accrochée à elle~: la charge pend alors à \emph{deux} brins de
corde, et chaque brin ne porte que la moitié du \omterm{def:g4:weight-and-mass:weight}{poids}.
\end{definition}

\begin{example}[Les poulies au travail]\label{ex:g5:simple-machines:pulleywork}
Le drapeau monte le long de son mât pendant que tes mains tirent
confortablement \emph{vers le bas}~: une \omterm{def:g5:simple-machines:pulley}{poulie} fixe est en haut ---
direction changée, effort inchangé, mais tirer vers le bas laisse ton
propre \omterm{def:g4:weight-and-mass:weight}{poids} t'aider. Le palan du bâtisseur enchaîne plusieurs
\omterm{def:g5:simple-machines:pulley}{poulies} pour que la charge pende à quatre ou six brins~: chaque brin
porte le quart ou le sixième du \omterm{def:g4:weight-and-mass:weight}{poids}, et un seul ouvrier soulève un
piano. Regarde bien l'ouvrier, pourtant~: des \omterm{def:g2:measuring-length:units}{mètres} et des \omterm{def:g2:measuring-length:units}{mètres} de
corde filent entre ses mains pendant que le piano monte lentement.
\end{example}

\section{Le plan incliné}

\begin{definition}[Plan incliné]\label{def:g5:simple-machines:ramp}
Un \emph{plan incliné}\index{plan incliné} --- une rampe --- est une
pente qui permet d'élever des charges sans les soulever à la
verticale. Plus la pente est douce, plus la poussée nécessaire pour y
faire avancer la charge est faible~: rouler un tonneau sur une longue
rampe douce est facile~; sur une rampe courte et raide, brutal~; tout
droit vers le haut, impossible.
\end{definition}

\begin{method}[Le sentir avec le dynamomètre en
élastique]\label{met:g5:simple-machines:measure}
Ton \omterm{def:g4:weight-and-mass:springscale}{dynamomètre} en élastique de l'an dernier peut goûter la
différence~:
\begin{enumerate}
  \item poser une petite voiture sur une planche et y accrocher
        l'élastique~;
  \item caler la planche en rampe douce sur une \omterm{def:g3:first-electric-circuit:battery}{pile} de livres
        basse~; tirer lentement la voiture vers le haut par
        l'élastique et noter l'allongement~;
  \item caler la même planche bien plus raide, sur l'assise d'une
        chaise, et tirer de nouveau.
\end{enumerate}
La traction sur la pente raide étire beaucoup plus l'élastique~:
pente plus raide, effort plus grand. Les pentes douces sont vraiment
douces --- ton instrument le dit.
\end{method}

\begin{example}[Des rampes dans le paysage]\label{ex:g5:simple-machines:ramps}
Les rampes d'accès pour fauteuils roulants s'étirent, longues et peu
pentues, à côté de courts escaliers --- de la douceur voulue. Les
routes de montagne zigzaguent~: plutôt que d'attaquer le sommet tout
droit, elles enroulent un long \omterm{def:g5:simple-machines:ramp}{plan incliné} bien doux en
allers-retours sur le versant. Et la vis est une rampe déguisée ---
une pente enroulée autour d'une tige, si bien que chaque tour facile
du tournevis fait descendre la vis d'un tout petit pas.
\end{example}

\section{La règle d'or}

\begin{proposition}[La règle d'or des machines]\label{prop:g5:simple-machines:golden}
Aucune \omterm{def:g5:simple-machines:machine}{machine simple} ne donne quelque chose pour rien~: ce qu'elle
économise en \omterm{def:g2:pushes-and-pulls:force}{force}, elle le facture en distance. Moitié moins de
\omterm{def:g2:pushes-and-pulls:force}{force} --- deux fois plus de corde tirée, ou deux fois plus de route
parcourue~; un dixième de la \omterm{def:g2:pushes-and-pulls:force}{force} --- dix fois plus de route.
Machine après machine, l'échange est exact~: le confort s'achète en
\omterm{def:g2:measuring-length:units}{mètres}.
\end{proposition}

\begin{example}[Vérifier la règle]\label{ex:g5:simple-machines:check}
La \omterm{def:g5:simple-machines:pulley}{poulie} mobile~: ta main tire avec la moitié de la \omterm{def:g2:pushes-and-pulls:force}{force}, et pour
élever la charge de \qty{1}{m} tu ramènes \qty{2}{m} de corde --- les
deux brins doivent raccourcir d'un \omterm{def:g2:measuring-length:units}{mètre} chacun. La barre à mine~:
ton extrémité décrit un long arc, le rocher bouge à peine. La rampe~:
la caisse atteint la même hauteur, mais tu l'as poussée sur toute la
pente. Chaque économie payée, exactement.
\end{example}

\begin{remark}[Pourquoi la règle ne se triche
pas]\label{rem:g5:simple-machines:whygolden}
Sous la règle d'or se cache notre vieil ami du chapitre sur
l'\omterm{def:g5:everyday-energy:energy}{énergie}~: élever une charge jusqu'à une hauteur coûte une part
fixe d'\omterm{def:g5:everyday-energy:energy}{énergie}, et aucun arrangement de cordes et de planches ne
change ce prix. Les machines te permettent seulement de le payer en
plus petites pièces --- en plus grand nombre. Une petite \omterm{def:g2:pushes-and-pulls:force}{force} sur
une longue route livre la même \omterm{def:g5:everyday-energy:energy}{énergie} qu'une grande \omterm{def:g2:pushes-and-pulls:force}{force} sur une
route courte. Les machines à mouvement perpétuel ont échoué pour
cette raison même~: l'addition est toujours l'addition.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:1}
Nomme les trois \omterm{def:g5:simple-machines:machine}{machines simples} classiques, et un endroit de la vie
courante où l'on rencontre chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:2}
Que change une \omterm{def:g5:simple-machines:pulley}{poulie} fixe à ta traction~? Que change une \omterm{def:g5:simple-machines:pulley}{poulie}
mobile~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:3}
Pourquoi tirer un drapeau vers le \emph{bas} pour le hisser
est-il plus confortable que soulever le même \omterm{def:g4:weight-and-mass:weight}{poids} à la verticale,
alors que l'effort a la même intensité~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:4}
Avec une \omterm{def:g5:simple-machines:pulley}{poulie} mobile, la charge pend à deux brins. Quelle part du
\omterm{def:g4:weight-and-mass:weight}{poids} ta main tient-elle~? Pour élever la charge de \qty{3}{m},
combien de corde faut-il tirer~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:5}
Dans \cref{met:g5:simple-machines:measure}, quelle rampe étire le
plus l'élastique~? Qu'est-ce qui paie le plus faible allongement de
la rampe douce~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:6}
Pourquoi les routes de montagne zigzaguent-elles au lieu de monter
tout droit dans la pente~? Quelle \omterm{def:g5:simple-machines:machine}{machine simple} est la route tout
entière~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:7}
Énonce la règle d'or des machines. Une machine te permet de tirer
avec un dixième de la \omterm{def:g2:pushes-and-pulls:force}{force} --- que te facture-t-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:simple-machines:8}
En quoi une vis est-elle un \omterm{def:g5:simple-machines:ramp}{plan incliné} déguisé~? Qu'est-ce qui joue
le rôle de la longue route douce~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:simple-machines:9}
Un palan suspend un piano à quatre brins. Quelle fraction du \omterm{def:g4:weight-and-mass:weight}{poids} du
piano la main de l'ouvrier tire-t-elle~? Pour élever le piano de
\qty{2}{m}, combien de \omterm{def:g2:measuring-length:units}{mètres} de corde passent entre ses mains~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:simple-machines:10}
Une rampe de chargement mesure \qty{6}{m} de long et s'élève de
\qty{1}{m}~; pousser une caisse dessus demande une poussée modeste.
Les déménageurs impatients taillent une nouvelle rampe de \qty{3}{m}
seulement jusqu'à la même plateforme. À l'aide de la règle d'or,
compare la nouvelle poussée à l'ancienne --- et explique pourquoi
«~la moitié de la route~» n'était pas une bonne affaire.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:simple-machines:11}
Une publicité propose un jeu de \omterm{def:g5:simple-machines:pulley}{poulies} «~si astucieux qu'on tire un
\omterm{def:g2:measuring-length:units}{mètre} de corde avec la moitié de la \omterm{def:g2:pushes-and-pulls:force}{force} --- et que la charge monte
d'un \omterm{def:g2:measuring-length:units}{mètre} entier~». Fais passer l'affirmation en jugement devant la
règle d'or et devant l'addition d'\omterm{def:g5:everyday-energy:energy}{énergie} de
\cref{rem:g5:simple-machines:whygolden}~: un arrangement de roues et
de cordes pourrait-il tenir cette promesse~?
\end{exercise}
