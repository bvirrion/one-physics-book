\chapter{Matahari di Langit: Hari dan Musim}\label{ch:g3:sun-days-seasons}

Pada bulan Juni kamu bermain di luar sesudah makan malam dalam \omterm{def:g1:light-and-shadows:source}{cahaya}
keemasan; pada bulan Desember lampu jalan sudah menyala sebelum kamu
pulang dari sekolah. Matahari tetap dengan kebiasaan lamanya --- naik
di satu sisi, melewati puncak, turun di sisi yang lain --- tetapi
jalannya melintasi langit tidak sama sepanjang tahun. Ikutilah ia
selama satu tahun penuh, dan kamu menemukan \omterm{def:g3:sun-days-seasons:year}{musim}.

\section{Jalan Matahari yang berubah}

\begin{example}[Jalan musim panas, jalan musim dingin]\label{ex:g3:sun-days-seasons:roads}
Amati busur Matahari pada pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} panas: ia terbit pagi
sekali, memanjat sangat tinggi --- hampir tepat di atas kepala pada
tengah hari --- lalu terbenam larut: jalan yang panjang dan tinggi,
dan hari yang panjang. Pada pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} dingin, busurnya rendah
dan pendek: \omterm{def:g1:day-and-night:daynight}{Matahari terbit} terlambat, tidak pernah memanjat jauh di
atas atap rumah, dan terbenam pada pertengahan sore. Di antara
keduanya, pada \omterm{def:g3:sun-days-seasons:year}{musim} semi dan \omterm{def:g3:sun-days-seasons:year}{musim} gugur, jalannya sedang-sedang saja
--- dan siang serta malam berbagi \omterm{ex:g2:measuring-time:clock}{jam} hampir sama rata.
\end{example}

\begin{proposition}[Aturan tengah hari]\label{prop:g3:sun-days-seasons:midday}
Pada setiap \omterm{def:g3:sun-days-seasons:year}{musim}, Matahari berdiri paling tinggi pada pertengahan
hari --- dan pada saat itu setiap \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} paling pendek untuk
hari itu dan menunjuk ke arah yang sama dengan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah
hari kemarin, dan besok. \omterm{def:g1:light-and-shadows:shadow}{Bayang-bayang} tengah hari adalah anak panah
paling tepercaya di alam.
\end{proposition}

\begin{example}[Membaca musim dari bayang-bayang]\label{ex:g3:sun-days-seasons:shadowlength}
\emph{Panjang} \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari berubah menurut \omterm{def:g3:sun-days-seasons:year}{musim}: pada
\omterm{def:g3:sun-days-seasons:year}{musim} panas, dengan Matahari hampir tepat di atas kepala,
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari sebuah tiang hanyalah tunggul yang pendek;
pada \omterm{def:g3:sun-days-seasons:year}{musim} dingin, dengan Matahari tergantung rendah, tiang yang sama
melemparkan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari yang panjang. Tandailah ujung
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari di tanah sebulan sekali, dan tandanya akan
merayap maju-mundur sepanjang tahun seperti bandul yang lambat.
\end{example}

\section{Jam bayang-bayang}

\begin{definition}[Jam matahari]\label{def:g3:sun-days-seasons:sundial}
Sebuah \emph{jam matahari}\index{jam matahari} adalah \omterm{ex:g2:measuring-time:clock}{jam} yang dibuat
dari \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}: sebatang tongkat atau bilah yang berdiri di bawah
sinar matahari, dengan tanda di sekelilingnya yang menunjukkan tempat
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} jatuh pada tiap \omterm{ex:g2:measuring-time:clock}{jam}. Selagi Matahari menempuh
busurnya, \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} menyapu tanda itu --- pelan, tanpa suara,
dan tanpa satu pun bagian yang bergerak.
\end{definition}

\begin{method}[Membuat jam bayang-bayang]\label{met:g3:sun-days-seasons:build}
Pada pagi yang cerah, di tempat yang terkena matahari sepanjang hari:
\begin{enumerate}
  \item tancapkan tongkat lurus tegak di tanah yang rata (atau
        tegakkan di dalam ember berisi pasir);
  \item setiap \omterm{ex:g2:measuring-time:clock}{jam}, tepat pada \omterm{ex:g2:measuring-time:clock}{jamnya}, tandai ujung \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya}
        dengan sebutir kerikil lalu tulis \omterm{ex:g2:measuring-time:clock}{jamnya} pada kerikil itu
        dengan kapur;
  \item teruskan sampai petang.
\end{enumerate}
Besok, \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} akan menyinggahi kerikilmu tepat waktu:
tongkatmu kini penunjuk waktu. (Setidaknya untuk sementara --- selagi
\omterm{def:g3:sun-days-seasons:year}{musim} bergeser, \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} ikut bergeser, dan kerikilmu
perlahan-lahan akan meleset.)
\end{method}

\section{Tahun dan musimnya}

\begin{definition}[Tahun dan musim]\label{def:g3:sun-days-seasons:year}
Sementara Bumi berputar sekali sehari, ia juga menempuh perjalanan
raksasa \emph{mengelilingi Matahari} --- satu putaran penuh memakan
waktu satu \emph{tahun}\index{tahun}, kira-kira $365$ hari. Sepanjang
putaran itu, jalan Matahari melintasi langit kita perlahan naik dan
turun, dan kita melewati empat \emph{musim}\index{musim}: semi, panas,
gugur, dingin --- lalu semi lagi, putaran demi putaran.
\end{definition}

\begin{example}[Empat perempat]\label{ex:g3:sun-days-seasons:quarters}
\omterm{def:g3:sun-days-seasons:year}{Musim} panas: Matahari tinggi, hari panjang, \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari
pendek. \omterm{def:g3:sun-days-seasons:year}{Musim} dingin: Matahari rendah, hari pendek, \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}
tengah hari panjang. \omterm{def:g3:sun-days-seasons:year}{Musim} semi dan \omterm{def:g3:sun-days-seasons:year}{musim} gugur berada di antaranya
--- dan pada masing-masingnya datang satu hari istimewa ketika siang
dan malam berlangsung sama lama. Petani, pelaut, dan pembangun zaman
dahulu belajar membaca kalender ini langsung dari langit.
\end{example}

\begin{remark}[Musim dingin bukan karena ``lebih jauh'']\label{rem:g3:sun-days-seasons:notfarther}
Menggoda sekali untuk menebak bahwa \omterm{def:g3:sun-days-seasons:year}{musim} dingin terasa dingin karena
Bumi menjauh dari Matahari. Ternyata tidak --- jaraknya hampir tidak
berubah, dan ketika di sini \omterm{def:g3:sun-days-seasons:year}{musim} dingin, pada saat yang persis sama
sedang \omterm{def:g3:sun-days-seasons:year}{musim} panas bagi anak-anak di belahan Bumi yang jauh, dan itu
tidak mungkin dijelaskan oleh ``lebih jauh''! Kuncinya yang sebenarnya
adalah \emph{jalan rendah} Matahari: matahari rendah, \omterm{def:g1:light-and-shadows:source}{cahaya}
menyerong, pemanasan lemah. Penjelasan lengkapnya dijanjikan untuk bab
kemudian.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:1}
Pada \omterm{def:g3:sun-days-seasons:year}{musim} apa busur Matahari paling tinggi dan paling panjang? Paling
rendah dan paling pendek? Apa akibatnya terhadap panjang harinya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:2}
Pada saat apa dalam sehari \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} sebuah tiang paling pendek?
Apa lagi yang istimewa pada \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari, hari demi hari?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:3}
Tiang yang sama melemparkan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari sepanjang $2$
sepatu bot pada bulan Juni dan $7$ sepatu bot pada bulan Desember.
Jelaskan perbedaannya dengan kedua jalan Matahari.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:4}
Bagaimana \omterm{def:g3:sun-days-seasons:sundial}{jam matahari} menunjukkan waktu? Jarum bergeraknya terbuat
dari apa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:5}
Pada \cref{met:g3:sun-days-seasons:build}, mengapa tongkatnya harus
berada di tempat yang terkena matahari sepanjang hari? Apa yang
terjadi pada \omterm{ex:g2:measuring-time:clock}{jam} itu ketika cuaca berawan --- dan pada malam hari?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:6}
Berapa lama Bumi menempuh satu putaran mengelilingi Matahari? Berapa
\omterm{def:g3:sun-days-seasons:year}{musim} yang termuat dalam satu putaran? Urutkan, mulai dari \omterm{def:g3:sun-days-seasons:year}{musim} semi.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:7}
Kira-kira ada berapa hari dalam satu tahun? Dan --- dari bab
sebelumnya --- ada berapa \omterm{ex:g2:measuring-time:clock}{jam} dalam satu hari? Perjalanan Bumi yang
mana yang membuat masing-masingnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:8}
Sebutkan dua petunjuk, yang terlihat dari jendelamu dalam satu minggu
saja, yang memberitahumu apakah kamu sedang berada di pertengahan
\omterm{def:g3:sun-days-seasons:year}{musim} panas atau pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} dingin tanpa melihat kalender.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:9}
Ketika di tempat tinggalmu \omterm{def:g3:sun-days-seasons:year}{musim} dingin, \omterm{def:g3:sun-days-seasons:year}{musim} apa yang sedang dialami
anak-anak di belahan Bumi yang jauh? Apa yang dibuktikan hal ini
tentang tebakan ``\omterm{def:g3:sun-days-seasons:year}{musim} dingin terjadi karena seluruh Bumi lebih jauh
dari Matahari''?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sun-days-seasons:10}
Seorang tukang kebun menancapkan kerikil \omterm{ex:g2:measuring-time:clock}{jam} \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} pada
bulan Maret. Pada bulan Juni \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} tidak lagi menyentuh
kerikil pada \omterm{ex:g2:measuring-time:clock}{jam} yang ditandai. Apakah tongkatnya bergeser? Jelaskan
apa yang bergeser, dengan \cref{ex:g3:sun-days-seasons:shadowlength}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sun-days-seasons:11}
Seorang penjelajah terbangun tanpa arloji dan tanpa telepon, melihat
Matahari tepat di titik tertingginya, dan satu \omterm{ex:g2:measuring-time:clock}{jam} kemudian mendapati
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tendanya sudah lebih panjang daripada tinggi tendanya.
Kira-kira pukul berapa ketika ia terbangun, dan apakah ia lebih dekat
ke pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} panas atau pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} dingin? Beralasanlah
dari aturan tengah hari dan dari panjang \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya}.
\end{exercise}
