\chapter{Tuas dan Neraca}\label{ch:g3:levers-and-scales}

Tahun lalu, di jungkat-jungkit, kita menemukan bahwa berat dan jarak
sama-sama menentukan --- dan kita berjanji memberikan aturan yang
persis dengan bilangan sungguhan di dalamnya. Saatnya menepati janji
itu. Di sepanjang jalan kita akan mengangkat batu yang jauh terlalu
berat bagi lengan kita, dan berkenalan dengan alat yang menimbang
barang dagangan dunia selama ribuan tahun.

\section{Tuas}

\begin{definition}[Tuas dan titik tumpu]\label{def:g3:levers-and-scales:lever}
Sebuah \emph{tuas}\index{tuas} adalah batang kaku yang bertumpu pada
sebuah titik penyangga yang membuatnya dapat berjungkit. Titik
penyangga itu disebut \emph{titik tumpu}\index{titik tumpu}.
Jungkat-jungkit adalah tuas; begitu pula linggis di bawah batu, dan
cakar palu yang mencabut paku.
\end{definition}

\begin{example}[Pelipat ganda kekuatan]\label{ex:g3:levers-and-scales:crowbar}
Seorang tukang kebun menyelipkan batang besi yang panjang ke bawah
sebuah bongkah batu, menumpukannya pada sepotong kayu dekat bongkah
itu, lalu menekan ujung yang jauh --- dan bongkah batu yang tidak
terangkat oleh sepuluh tangan itu pun terguling pelan. Rahasia \omterm{def:g3:levers-and-scales:lever}{tuas}
adalah jarak: menekanlah \emph{jauh} dari \omterm{def:g3:levers-and-scales:lever}{titik tumpu}, dan dorongan
kecilmu melakukan pekerjaan raksasa yang berada dekat dengannya.
Sebuah kisah tua menceritakan seorang ilmuwan besar zaman kuno
berbangga: ``Berilah aku tempat berpijak dan \omterm{def:g3:levers-and-scales:lever}{tuas} yang cukup panjang,
dan akan kugerakkan Bumi.''
\end{example}

\section{Aturan tuas}

\begin{method}[Jungkat-jungkit koin]\label{met:g3:levers-and-scales:coins}
Bangunlah jungkat-jungkit mini untuk memburu aturannya:
\begin{enumerate}
  \item seimbangkan sebuah penggaris melintang di atas pensil pada
        tanda tengahnya;
  \item pakailah koin yang serupa sebagai penunggang, diletakkan pada
        tanda bilangan bulat: ``satu koin di $4$'' berarti empat tanda
        dari pensilnya;
  \item bebani kedua sisinya dan catatlah kapan penggarisnya seimbang.
\end{enumerate}
Cobalah: satu koin di $6$ seimbang dengan satu koin di $6$. Dua koin
bertumpuk di $3$ seimbang dengan satu koin di $6$. Tiga koin di $2$
--- sekali lagi seimbang dengan satu koin di $6$!
\end{method}

\begin{proposition}[Aturan tuas]\label{prop:g3:levers-and-scales:rule}
Untuk masing-masing sisi \omterm{def:g3:levers-and-scales:lever}{tuas}, kalikan banyaknya koin dengan jaraknya
dari \omterm{def:g3:levers-and-scales:lever}{titik tumpu}. Tuasnya seimbang tepat ketika kedua sisinya punya
\emph{hasil kali yang sama}:
\[
2 \text{ koin} \times 3 \text{ tanda} = 6
\qquad\text{seimbang dengan}\qquad
1 \text{ koin} \times 6 \text{ tanda} = 6 .
\]
Kalau hasil kali salah satu sisi lebih besar, sisi itulah yang turun.
\end{proposition}

\begin{example}[Memakai aturannya]\label{ex:g3:levers-and-scales:using}
Empat koin duduk di tanda $2$: hasil kalinya $4 \times 2 = 8$. Untuk
menyeimbangkannya dengan koin di tanda $4$, kita memerlukan hasil kali
$8$ juga di sana: $8 = 2 \times 4$, jadi dua koin di $4$ sudah cukup.
Dan dapatkah satu koin saja menyeimbangkan keempatnya? Koin itu
harus duduk di tanda $8$: $1 \times 8
= 8$. Makin ringan penunggangnya, makin jauh ia harus duduk ---
persis yang diajarkan jungkat-jungkit Ayah kepada kita, sekarang
dengan bilangan.
\end{example}

\section{Neraca}

\begin{definition}[Neraca]\label{def:g3:levers-and-scales:scale}
Sebuah \emph{neraca}\index{neraca} adalah \omterm{def:g3:levers-and-scales:lever}{tuas} yang dibangun untuk
keadilan: dua piring tergantung pada jarak yang \emph{persis sama}
dari \omterm{def:g3:levers-and-scales:lever}{titik tumpu}. Dengan jarak yang sama, hasil kalinya cocok tepat
ketika beratnya cocok --- sehingga neraca berjungkit ke arah piring
yang lebih berat, dan tergantung mendatar ketika kedua piringnya
memuat beban yang sama beratnya.
\end{definition}

\begin{example}[Menimbang dengan membandingkan]\label{ex:g3:levers-and-scales:weighing}
Selama ribuan tahun, para pedagang menjual tepung, emas, dan merica
dengan \omterm{def:g3:levers-and-scales:scale}{neraca}: barangnya di satu piring, balok logam acuan yang kecil
di piring yang lain, ditambahkan sampai kedua piring tergantung
mendatar. \omterm{def:g3:levers-and-scales:scale}{Neraca} tidak mengenal bilangan apa pun --- ia hanya menjawab
``lebih berat'', ``lebih ringan'', atau ``sama''. Tetapi membandingkan
dengan balok yang dipilih baik-baik sudah cukup bagi pasar yang adil
--- dan, seperti akan ditunjukkan bab terakhir bagian ini, itulah
jalan menuju pengukuran massa itu sendiri.
\end{example}

\begin{remark}[Periksalah neraca yang kosong]\label{rem:g3:levers-and-scales:fair}
\omterm{def:g3:levers-and-scales:scale}{Neraca} yang adil harus tergantung mendatar ketika \emph{kosong}. Kalau
satu lengannya lebih panjang atau satu piringnya lebih berat, hasil
kalinya sudah berbeda bahkan sebelum barangnya datang, dan setiap
penimbangan sesudah itu curang. Pedagang yang jujur membuktikan \omterm{def:g3:levers-and-scales:scale}{neraca}
kosongnya mendatar di depan pembeli --- cara lama tanpa kata untuk
mengatakan ``main jujur''.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:1}
Tunjukkan \omterm{def:g3:levers-and-scales:lever}{tuas} dan \omterm{def:g3:levers-and-scales:lever}{titik tumpunya} pada: jungkat-jungkit; \ linggis di
atas sepotong kayu; \ gunting. (Gunting menyembunyikan dua \omterm{def:g3:levers-and-scales:lever}{tuas} --- di
mana keduanya berjungkit?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:2}
Untuk mengangkat batu yang berat dengan linggis, sebaiknya kamu
menekan dekat \omterm{def:g3:levers-and-scales:lever}{titik tumpu} atau jauh darinya? Mengapa, dalam satu
kalimat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:3}
Pada jungkat-jungkit koin, hitunglah hasil kali untuk: $3$ koin di
tanda $4$; \ $2$ koin di tanda $5$; \ $1$ koin di tanda $6$. Pasangan
mana di antaranya yang saling seimbang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:4}
Satu koin duduk di tanda $8$ di sebelah kiri. Di mana kamu harus
meletakkan $2$ koin di sebelah kanan supaya seimbang? Dan $4$ koin?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:5}
Sisi kiri: $3$ koin di tanda $3$. Sisi kanan: $2$ koin di tanda $4$.
Hitunglah kedua hasil kalinya. Sisi mana yang turun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:6}
Mengapa kedua piring sebuah \omterm{def:g3:levers-and-scales:scale}{neraca} digantung pada jarak yang persis
sama dari \omterm{def:g3:levers-and-scales:lever}{titik tumpu}? Apa yang akan dilakukan lengan kiri yang lebih
panjang terhadap setiap penimbangan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:7}
Sebuah \omterm{def:g3:levers-and-scales:scale}{neraca} tergantung mendatar dengan sekantong kelereng di satu
piring dan $6$ balok yang serupa di piring yang lain. Apa yang kita
pelajari tentang kantong itu? Apa jawaban \omterm{def:g3:levers-and-scales:scale}{neraca} kalau kita mengambil
satu balok?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:8}
Sebuah jungkat-jungkit: Tom seberat $2$ karung dan duduk di tanda $3$;
adik perempuannya seberat $1$ karung. Di tanda berapa adiknya harus
duduk supaya seimbang dengannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:9}
Cakar palu mencabut sebuah paku: tanganmu mendorong ujung gagangnya,
jauh dari tempat kepala palu bertumpu pada kayu. Mana \omterm{def:g3:levers-and-scales:lever}{titik tumpunya},
dan mengapa paku itu --- yang tidak pernah tercabut oleh jarimu ---
akhirnya keluar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:levers-and-scales:10}
Kakek (seberat $4$ karung) ingin bermain jungkat-jungkit dengan Zoe
kecil (seberat $1$ karung). Jungkat-jungkitnya bertanda $1$ sampai $8$
di tiap sisi. Carilah \emph{dua} cara berbeda untuk mendudukkan mereka
supaya jungkat-jungkitnya seimbang.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:levers-and-scales:11}
Seorang pedagang curang diam-diam membuat satu lengan \omterm{def:g3:levers-and-scales:scale}{neracanya}
sedikit lebih panjang, dan selalu meletakkan barangnya di sisi lengan
yang panjang. Apakah pembeli mendapat barang lebih banyak daripada
yang dibayarnya, atau lebih sedikit? Jelaskan dengan aturan \omterm{def:g3:levers-and-scales:lever}{tuas} ---
dan dengan \cref{rem:g3:levers-and-scales:fair}.
\end{exercise}
