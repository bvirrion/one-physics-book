\section{Sebuah bohlam, sebuah baterai, dua kawat}

\begin{definition}[Baterai]\label{def:g3:first-electric-circuit:battery}
Sebuah \emph{baterai}\index{baterai} adalah tempat penyimpan listrik
yang kecil. Ia punya dua ujung yang berbeda, yang disebut
\emph{kutub}\index{kutub}: satu bertanda $+$ (plus), yang lain
bertanda $-$ (minus). Kedua ujungnya penting --- listrik hanya bermain
kalau keduanya ikut dalam permainan.
\end{definition}

\begin{definition}[Bohlam]\label{def:g3:first-electric-circuit:bulb}
Sebuah \emph{bohlam}\index{bohlam} adalah lampu kecil dengan benang
tipis di dalam kacanya yang berpijar putih membara ketika listrik
mengalir melewatinya. Amati baik-baik kaki logamnya: bohlam juga punya
\emph{dua} tempat sentuh --- ujung di bagian paling bawah, dan sisi
logam kakinya. Dua \omterm{def:g3:first-electric-circuit:battery}{kutub} pada \omterm{def:g3:first-electric-circuit:battery}{baterai}, dua tempat sentuh pada bohlam:
itu bukan kebetulan.
\end{definition}

\begin{method}[Menyalakan bohlam]\label{met:g3:first-electric-circuit:light}
Dengan satu \omterm{def:g3:first-electric-circuit:battery}{baterai}, satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan dua kawat:
\begin{enumerate}
  \item pasang satu kawat dari \omterm{def:g3:first-electric-circuit:battery}{kutub} $+$ \omterm{def:g3:first-electric-circuit:battery}{baterai} ke ujung kaki
        \omterm{def:g3:first-electric-circuit:bulb}{bohlam};
  \item pasang kawat yang lain dari \omterm{def:g3:first-electric-circuit:battery}{kutub} $-$ \omterm{def:g3:first-electric-circuit:battery}{baterai} ke sisi logam
        kakinya;
  \item \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} menyala --- kamu sudah membangun sebuah rangkaian!
\end{enumerate}
Kalau tetap gelap, periksalah tersangka yang biasa: kawat yang
menyentuh kacanya alih-alih tempat sentuh logamnya, sambungan yang
longgar, atau \omterm{def:g3:first-electric-circuit:battery}{baterai} yang sudah habis.
\end{method}

\section{Rangkaiannya harus tertutup}

\begin{definition}[Rangkaian listrik]\label{def:g3:first-electric-circuit:circuit}
Sebuah \emph{rangkaian listrik}\index{rangkaian listrik} adalah
lingkar lengkap bagi listrik: keluar dari satu \omterm{def:g3:first-electric-circuit:battery}{kutub} \omterm{def:g3:first-electric-circuit:battery}{baterai}, melewati
kawat dan \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, lalu kembali masuk ke \omterm{def:g3:first-electric-circuit:battery}{kutub} yang lain. Ketika
lingkarnya lengkap kita katakan rangkaiannya
\emph{tertutup}\index{rangkaian tertutup}; kalau ada celah di mana
pun, rangkaiannya \emph{terbuka}\index{rangkaian terbuka}.
\end{definition}

\begin{proposition}[Hukum lingkar]\label{prop:g3:first-electric-circuit:loop}
\omterm{def:g3:first-electric-circuit:bulb}{Bohlam} menyala hanya ketika rangkaiannya tertutup --- satu lingkar
tanpa putus dari satu \omterm{def:g3:first-electric-circuit:battery}{kutub}, melewati \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, kembali ke \omterm{def:g3:first-electric-circuit:battery}{kutub} yang
lain. Satu celah saja, di mana pun pada lingkar itu, dan \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}
padam: listrik tidak melompati celah, dan ia tidak mengalir ke jalan
buntu.
\end{proposition}

\begin{example}[Mengapa kedua ujungnya?]\label{ex:g3:first-electric-circuit:bothends}
Hubungkan \emph{dua} kawat dari \omterm{def:g3:first-electric-circuit:battery}{kutub} $+$ \omterm{def:g3:first-electric-circuit:battery}{baterai} ke kedua tempat
sentuh \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan biarkan \omterm{def:g3:first-electric-circuit:battery}{kutub} $-$ sendirian: tidak terjadi apa-apa.
Satu kawat dari $+$ dan satu dari $-$, tetapi keduanya ke ujung
\omterm{def:g3:first-electric-circuit:bulb}{bohlam}: sekali lagi tidak terjadi apa-apa. \omterm{def:g3:first-electric-circuit:bulb}{Bohlam} menyala hanya ketika
lingkarnya menyinggahi kedua \omterm{def:g3:first-electric-circuit:battery}{kutub} \omterm{def:g3:first-electric-circuit:battery}{baterai} yang berbeda \emph{dan}
kedua tempat sentuh \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang berbeda --- satu perjalanan
bolak-balik yang lengkap.
\end{example}

\begin{example}[Sakelar]\label{ex:g3:first-electric-circuit:switch}
Sebuah \emph{sakelar}\index{sakelar} adalah pintu di dalam rangkaian:
jembatan logam kecil yang direbahkan oleh bunyi klik (tertutup ---
lingkarnya lengkap, terang!) atau diangkat sampai terbuka (ada celah
--- gelap). Tombol senter, \omterm{ex:g3:first-electric-circuit:switch}{sakelar} lampu di dinding, tombol bel pintu:
semuanya pintu di dalam sebuah lingkar. Kamu dapat membuatnya sendiri
dari jepitan jemuran dan sepotong aluminium dapur yang dilipatkan pada
rahangnya.
\end{example}

\begin{remark}[Gambar yang rapi]\label{rem:g3:first-electric-circuit:symbols}
Alih-alih menggambar setiap tonjolan \omterm{def:g3:first-electric-circuit:battery}{baterai} dan ulir \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, para
ahli listrik memakai \emph{lambang}: dua garis yang tidak sama panjang
untuk \omterm{def:g3:first-electric-circuit:battery}{baterai} (yang panjang adalah $+$), lingkaran bersilang untuk
lampu, jembatan angkat kecil untuk \omterm{ex:g3:first-electric-circuit:switch}{sakelar}, dan garis lurus untuk
kawat. Gambar lambang sebuah rangkaian lebih cepat dibuat, dan siapa
pun di dunia dapat membacanya --- kita akan memakai gambar seperti ini
bertahun-tahun lamanya.
\end{remark}

\begin{remark}[Baterai boleh, stopkontak jangan pernah]\label{rem:g3:first-electric-circuit:safety}
Semua yang ada di bab ini dimainkan dengan \omterm{def:g3:first-electric-circuit:battery}{baterai} bulat atau pipih
yang kecil: dorongannya mungil dan aman untuk percobaan.
\emph{Stopkontak} di dinding adalah dunia yang lain --- dorongannya
ratusan kali lebih perkasa, cukup untuk melukai atau membunuh. Jangan
pernah memasukkan kawat, jari, atau apa pun ke dalam stopkontak, dan
jangan pernah membongkar apa pun yang dicolokkan ke sana. Permainan
\omterm{def:g3:first-electric-circuit:battery}{baterai}: boleh. Stopkontak: jangan pernah.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:1}
Sebutkan kedua \omterm{def:g3:first-electric-circuit:battery}{kutub} sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} dan kedua tempat sentuh sebuah
\omterm{def:g3:first-electric-circuit:bulb}{bohlam}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:2}
Apa itu \omterm{def:g3:first-electric-circuit:circuit}{rangkaian tertutup}? Apa itu \omterm{def:g3:first-electric-circuit:circuit}{rangkaian terbuka}? Pada yang mana
\omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} menyala?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:3}
Lea menghubungkan \omterm{def:g3:first-electric-circuit:battery}{kutub} $+$ \omterm{def:g3:first-electric-circuit:battery}{baterai} ke ujung \omterm{def:g3:first-electric-circuit:bulb}{bohlam} dengan satu kawat,
lalu berhenti di situ. \omterm{def:g3:first-electric-circuit:bulb}{Bohlamnya} tetap gelap. Apa yang kurang, menurut
kata-kata hukum lingkar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:4}
Apa pekerjaan sebuah \omterm{ex:g3:first-electric-circuit:switch}{sakelar} di dalam lingkar? Apa yang terdapat di
dalam \omterm{ex:g3:first-electric-circuit:switch}{sakelar} ketika ia tertutup?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:5}
Gambarlah, dengan lambang yang rapi pada
\cref{rem:g3:first-electric-circuit:symbols}, sebuah rangkaian dengan
satu \omterm{def:g3:first-electric-circuit:battery}{baterai}, satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} tertutup, dan satu lampu. Lambang mana
untuk apa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:6}
Senter padam di tengah perjalanan berkemah. Sebutkan tiga celah atau
kerusakan yang mungkin untuk diperiksa, dari
\cref{met:g3:first-electric-circuit:light}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:7}
Pada rangkaian bel pintu, di mana ``\omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}''? Mengapa belnya
berbunyi hanya selama tombolnya ditekan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:8}
Mengapa kamu tidak boleh menyodokkan apa pun ke dalam stopkontak,
padahal bermain dengan \omterm{def:g3:first-electric-circuit:battery}{baterai} tidak apa-apa? (Lihat
\cref{rem:g3:first-electric-circuit:safety}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:9}
Sebuah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} dipegang dengan ujungnya menempel pada \omterm{def:g3:first-electric-circuit:battery}{kutub} $+$
\omterm{def:g3:first-electric-circuit:battery}{baterai}, dan satu kawat dari \omterm{def:g3:first-electric-circuit:battery}{kutub} $-$ \omterm{def:g3:first-electric-circuit:battery}{baterai} ditekankan pada sisi
logam kaki \omterm{def:g3:first-electric-circuit:bulb}{bohlam}: apakah \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} akan menyala? Telusurilah
lingkarnya dengan jarimu untuk memutuskan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:10}
Sam membangun sebuah lingkar: \omterm{def:g3:first-electric-circuit:battery}{baterai}, kawat, \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, kawat, kembali
ke \omterm{def:g3:first-electric-circuit:battery}{baterai} --- tetapi ia memasang kedua kawatnya pada \omterm{def:g3:first-electric-circuit:battery}{kutub} yang sama,
$-$, ``karena letaknya lebih dekat''. \omterm{def:g3:first-electric-circuit:bulb}{Bohlamnya} tetap gelap walaupun
lingkarnya tampak lengkap. Jelaskan apa yang kurang pada lingkarnya,
dengan \cref{ex:g3:first-electric-circuit:bothends}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:11}
Seuntai lampu pesta kehilangan satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan \emph{seluruh}
untaiannya gelap, padahal semua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang lain baik-baik saja. Apa
yang diberitahukan hal ini tentang cara \omterm{def:g3:first-electric-circuit:bulb}{bohlam} itu disusun di dalam
lingkar? (Bab rangkaian tahun depan akan memberi susunan itu sebuah
nama.)
\end{exercise}
