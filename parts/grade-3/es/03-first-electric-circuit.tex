\chapter{Un primer circuito eléctrico}\label{ch:g3:first-electric-circuit}

Clic --- la linterna lanza su haz por la habitación a oscuras. Dentro
de ese tubo de plástico viven una \omterm{def:g3:first-electric-circuit:battery}{pila}, una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} pequeña y dos
tiras de metal, jugando a un juego de reglas estrictas. Hoy vamos a
desmontar el juego: con una \omterm{def:g3:first-electric-circuit:battery}{pila}, una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} y dos cables
fabricaremos \omterm{def:g1:light-and-shadows:source}{luz} con nuestras propias manos.

\section{Una bombilla, una pila, dos cables}

\begin{definition}[La pila]\label{def:g3:first-electric-circuit:battery}
Una \emph{pila}\index{pila} es un pequeño almacén de electricidad.
Tiene dos extremos distintos, llamados sus
\emph{bornes}\index{borne}: uno marcado con $+$ (más) y el otro con
$-$ (menos). Los dos extremos importan --- la electricidad solo juega
cuando los dos entran en el juego.
\end{definition}

\begin{definition}[La bombilla]\label{def:g3:first-electric-circuit:bulb}
Una \emph{bombilla}\index{bombilla} es una lamparita con un hilo fino
dentro del vidrio que se pone al rojo blanco cuando la electricidad
pasa por él. Fíjate bien en su casquillo metálico: la bombilla tiene también
\emph{dos} sitios de contacto --- la punta del fondo y el lado
metálico del casquillo. Dos \omterm{def:g3:first-electric-circuit:battery}{bornes} en la \omterm{def:g3:first-electric-circuit:battery}{pila}, dos contactos en la
bombilla: eso no es casualidad.
\end{definition}

\begin{method}[Encender la bombilla]\label{met:g3:first-electric-circuit:light}
Con una \omterm{def:g3:first-electric-circuit:battery}{pila}, una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} y dos cables:
\begin{enumerate}
  \item une con un cable el \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pila} con la punta del
        casquillo de la \omterm{def:g3:first-electric-circuit:bulb}{bombilla};
  \item une con el otro cable el \omterm{def:g3:first-electric-circuit:battery}{borne} $-$ de la \omterm{def:g3:first-electric-circuit:battery}{pila} con el lado
        metálico del casquillo;
  \item la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} se enciende --- ¡has construido un circuito!
\end{enumerate}
Si se queda apagada, revisa a los sospechosos de siempre: un cable que
toca el vidrio en vez del contacto metálico, una unión floja o una
\omterm{def:g3:first-electric-circuit:battery}{pila} gastada.
\end{method}

\section{El circuito tiene que cerrarse}

\begin{definition}[Circuito eléctrico]\label{def:g3:first-electric-circuit:circuit}
Un \emph{circuito eléctrico}\index{circuito eléctrico} es un bucle
completo para la electricidad: sale por un \omterm{def:g3:first-electric-circuit:battery}{borne} de la \omterm{def:g3:first-electric-circuit:battery}{pila},
recorre los cables y la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} y vuelve a entrar por el otro
\omterm{def:g3:first-electric-circuit:battery}{borne}. Cuando el bucle está completo decimos que el circuito está
\emph{cerrado}\index{circuito cerrado}; si tiene un hueco en algún
sitio, está \emph{abierto}\index{circuito abierto}.
\end{definition}

\begin{proposition}[La ley del bucle]\label{prop:g3:first-electric-circuit:loop}
La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} se enciende solo cuando el circuito está cerrado ---
un bucle sin cortes que sale de un \omterm{def:g3:first-electric-circuit:battery}{borne}, atraviesa la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} y
vuelve al otro \omterm{def:g3:first-electric-circuit:battery}{borne}. Un solo hueco, en cualquier punto del bucle,
y la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} se apaga: la electricidad no salta huecos y no corre
por callejones sin salida.
\end{proposition}

\begin{example}[¿Por qué los dos extremos?]\label{ex:g3:first-electric-circuit:bothends}
Une \emph{dos} cables desde el \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pila} hasta los dos
contactos de la \omterm{def:g3:first-electric-circuit:bulb}{bombilla}, dejando el $-$ en paz: nada. Un cable
desde el $+$ y otro desde el $-$, pero los dos a la punta de la
\omterm{def:g3:first-electric-circuit:bulb}{bombilla}: nada otra vez. La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} se enciende solo cuando el
bucle visita los dos \omterm{def:g3:first-electric-circuit:battery}{bornes} distintos de la \omterm{def:g3:first-electric-circuit:battery}{pila} \emph{y} los dos
contactos distintos de la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} --- un viaje de ida y vuelta
completo.
\end{example}

\begin{example}[El interruptor]\label{ex:g3:first-electric-circuit:switch}
Un \emph{interruptor}\index{interruptor} es una puerta en el circuito:
un puentecito de metal que el clic deja caer (cerrado --- el bucle
está completo, ¡\omterm{def:g1:light-and-shadows:source}{luz}!) o levanta (un hueco --- oscuridad). El botón
de la linterna, el \omterm{ex:g3:first-electric-circuit:switch}{interruptor} de la \omterm{def:g1:light-and-shadows:source}{luz} de la pared, el pulsador de
un timbre: todos ellos son puertas en un bucle. Puedes construir uno
con una pinza de la ropa y una tira de papel de aluminio doblada sobre
sus mordazas.
\end{example}

\begin{remark}[Dibujos ordenados]\label{rem:g3:first-electric-circuit:symbols}
En vez de dibujar cada bulto de la \omterm{def:g3:first-electric-circuit:battery}{pila} y cada rosca de la
\omterm{def:g3:first-electric-circuit:bulb}{bombilla}, los electricistas usan \emph{símbolos}: dos barras
desiguales para la \omterm{def:g3:first-electric-circuit:battery}{pila} (la larga es el $+$), un círculo con una
cruz para la lámpara, un puentecito levadizo para el \omterm{ex:g3:first-electric-circuit:switch}{interruptor} y
líneas rectas para los cables. Un dibujo con símbolos se hace más
deprisa y lo puede leer cualquier persona del mundo --- usaremos estos
dibujos durante muchos años.
\end{remark}

\begin{remark}[Pilas sí, enchufe de la pared jamás]\label{rem:g3:first-electric-circuit:safety}
Todo lo de este capítulo se juega con \omterm{def:g3:first-electric-circuit:battery}{pilas} pequeñas, redondas o
planas: su empuje es diminuto y no tiene peligro para los
experimentos. El \emph{enchufe de la pared} es otro mundo --- su
empuje es cientos de veces más tremendo, suficiente para hacer daño o
matar. No metas nunca cables, dedos ni ninguna otra cosa en un enchufe
de la pared, y no desmontes nunca nada que se enchufe en uno. Juegos
con \omterm{def:g3:first-electric-circuit:battery}{pilas}: sí. Enchufe: jamás.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:1}
Nombra los dos \omterm{def:g3:first-electric-circuit:battery}{bornes} de una \omterm{def:g3:first-electric-circuit:battery}{pila} y los dos sitios de contacto de
una \omterm{def:g3:first-electric-circuit:bulb}{bombilla}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:2}
¿Qué es un \omterm{def:g3:first-electric-circuit:circuit}{circuito cerrado}? ¿Y un \omterm{def:g3:first-electric-circuit:circuit}{circuito abierto}? ¿En cuál de
los dos se enciende la \omterm{def:g3:first-electric-circuit:bulb}{bombilla}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:3}
Lea une con un cable el \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pila} con la punta de la
\omterm{def:g3:first-electric-circuit:bulb}{bombilla} y se para ahí. La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} se queda apagada. ¿Qué falta,
con las palabras de la ley del bucle?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:4}
¿Qué trabajo hace un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} en el bucle? ¿Qué hay dentro de un
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} cuando está cerrado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:5}
Dibuja, con los símbolos ordenados de la
\cref{rem:g3:first-electric-circuit:symbols}, un circuito con una
\omterm{def:g3:first-electric-circuit:battery}{pila}, un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} cerrado y una lámpara. ¿Qué símbolo es
cada cosa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:6}
La linterna se apaga en mitad de una acampada. Da tres huecos o
averías posibles que haya que revisar, a partir del
\cref{met:g3:first-electric-circuit:light}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:7}
En el circuito del timbre, ¿dónde está el ``\omterm{ex:g3:first-electric-circuit:switch}{interruptor}''? ¿Por qué
suena el timbre solo mientras se aprieta el pulsador?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:8}
¿Por qué no hay que meter nunca nada en un enchufe de la pared, si
jugar con una \omterm{def:g3:first-electric-circuit:battery}{pila} no tiene peligro? (Mira la
\cref{rem:g3:first-electric-circuit:safety}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:9}
Una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} sujeta con la punta sobre el \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la
\omterm{def:g3:first-electric-circuit:battery}{pila}, y un cable que va del \omterm{def:g3:first-electric-circuit:battery}{borne} $-$ de la \omterm{def:g3:first-electric-circuit:battery}{pila} apretado
contra el lado metálico del casquillo de la \omterm{def:g3:first-electric-circuit:bulb}{bombilla}: ¿se encenderá?
Recorre el bucle con el dedo para decidirlo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:10}
Sam construye un bucle: \omterm{def:g3:first-electric-circuit:battery}{pila}, cable, \omterm{def:g3:first-electric-circuit:bulb}{bombilla}, cable y vuelta a
la \omterm{def:g3:first-electric-circuit:battery}{pila} --- pero engancha los dos cables al mismo \omterm{def:g3:first-electric-circuit:battery}{borne}, el $-$,
``porque estaba más cerca''. La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} se queda apagada aunque el
bucle parezca completo. Explica qué le falta a su bucle, usando el
\cref{ex:g3:first-electric-circuit:bothends}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:11}
A una guirnalda de luces de fiesta le falta una \omterm{def:g3:first-electric-circuit:bulb}{bombilla}, y la
guirnalda \emph{entera} está apagada, aunque todas las demás
\omterm{def:g3:first-electric-circuit:bulb}{bombillas} están bien. ¿Qué te dice esto sobre cómo están colocadas
las \omterm{def:g3:first-electric-circuit:bulb}{bombillas} en el bucle? (El capítulo de circuitos del año que
viene le pondrá nombre a esta colocación.)
\end{exercise}
