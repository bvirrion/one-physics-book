\chapter{Un premier circuit électrique}\label{ch:g3:first-electric-circuit}

Clic --- la \omterm{def:g3:first-electric-circuit:bulb}{lampe} de poche jette son faisceau à travers la pièce
sombre. Dans ce tube de plastique vivent une \omterm{def:g3:first-electric-circuit:battery}{pile}, une petite \omterm{def:g3:first-electric-circuit:bulb}{lampe}
et deux bandes de métal, qui jouent à un jeu aux règles strictes.
Aujourd'hui, nous démontons le jeu~: avec une \omterm{def:g3:first-electric-circuit:battery}{pile}, une \omterm{def:g3:first-electric-circuit:bulb}{lampe} et deux
fils, nous allons faire de la \omterm{def:g1:light-and-shadows:source}{lumière} de nos propres mains.

\section{Une lampe, une pile, deux fils}

\begin{definition}[La pile]\label{def:g3:first-electric-circuit:battery}
Une \emph{pile}\index{pile} est une petite réserve d'électricité.
Elle a deux extrémités différentes, appelées ses
\emph{bornes}\index{borne}~: l'une marquée $+$ (plus), l'autre
marquée $-$ (moins). Les deux comptent --- l'électricité ne joue que
lorsque toutes les deux entrent dans la partie.
\end{definition}

\begin{definition}[La lampe]\label{def:g3:first-electric-circuit:bulb}
Une \emph{lampe}\index{lampe} est une petite ampoule dont le mince
filament, à l'intérieur du verre, devient blanc de chaleur quand
l'électricité le traverse. Regarde de près son culot de métal~: la
lampe a elle aussi \emph{deux} endroits de contact --- le plot tout
en bas, et le côté métallique du culot. Deux \omterm{def:g3:first-electric-circuit:battery}{bornes} sur la \omterm{def:g3:first-electric-circuit:battery}{pile}, deux
contacts sur la lampe~: ce n'est pas un hasard.
\end{definition}

\begin{method}[Allumer la lampe]\label{met:g3:first-electric-circuit:light}
Avec une \omterm{def:g3:first-electric-circuit:battery}{pile}, une \omterm{def:g3:first-electric-circuit:bulb}{lampe} et deux fils~:
\begin{enumerate}
  \item relier un fil de la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} au plot du culot de
        la \omterm{def:g3:first-electric-circuit:bulb}{lampe}~;
  \item relier l'autre fil de la \omterm{def:g3:first-electric-circuit:battery}{borne} $-$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} au côté
        métallique du culot~;
  \item la \omterm{def:g3:first-electric-circuit:bulb}{lampe} s'allume --- tu viens de construire un circuit~!
\end{enumerate}
Si elle reste éteinte, cherche les suspects habituels~: un fil qui
touche le verre au lieu du contact métallique, une connexion mal
serrée, ou une \omterm{def:g3:first-electric-circuit:battery}{pile} usée.
\end{method}

\section{Le circuit doit être fermé}

\begin{definition}[Circuit électrique]\label{def:g3:first-electric-circuit:circuit}
Un \emph{circuit électrique}\index{circuit électrique} est une boucle
complète pour l'électricité~: on sort d'une \omterm{def:g3:first-electric-circuit:battery}{borne} de la \omterm{def:g3:first-electric-circuit:battery}{pile}, on
traverse les fils et la \omterm{def:g3:first-electric-circuit:bulb}{lampe}, et on revient par l'autre \omterm{def:g3:first-electric-circuit:battery}{borne}. Quand
la boucle est complète, on dit que le circuit est
\emph{fermé}\index{circuit fermé}~; s'il présente une coupure quelque
part, il est \emph{ouvert}\index{circuit ouvert}.
\end{definition}

\begin{proposition}[La loi de la boucle]\label{prop:g3:first-electric-circuit:loop}
La \omterm{def:g3:first-electric-circuit:bulb}{lampe} ne s'allume que si le circuit est fermé --- une boucle
ininterrompue partant d'une \omterm{def:g3:first-electric-circuit:battery}{borne}, traversant la \omterm{def:g3:first-electric-circuit:bulb}{lampe}, et revenant à
l'autre \omterm{def:g3:first-electric-circuit:battery}{borne}. Une seule coupure, n'importe où dans la boucle, et la
\omterm{def:g3:first-electric-circuit:bulb}{lampe} s'éteint~: l'électricité ne saute pas les coupures, et elle ne
s'engage pas dans les impasses.
\end{proposition}

\begin{example}[Pourquoi les deux
extrémités~?]\label{ex:g3:first-electric-circuit:bothends}
Relie \emph{deux} fils de la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} aux deux contacts
de la \omterm{def:g3:first-electric-circuit:bulb}{lampe}, en laissant le $-$ de côté~: rien. Un fil depuis le $+$
et un depuis le $-$, mais tous les deux au plot de la \omterm{def:g3:first-electric-circuit:bulb}{lampe}~: rien
encore. La \omterm{def:g3:first-electric-circuit:bulb}{lampe} ne s'allume que si la boucle passe par les deux
\omterm{def:g3:first-electric-circuit:battery}{bornes} différentes de la \omterm{def:g3:first-electric-circuit:battery}{pile} \emph{et} par les deux contacts
différents de la \omterm{def:g3:first-electric-circuit:bulb}{lampe} --- un aller-retour complet.
\end{example}

\begin{example}[L'interrupteur]\label{ex:g3:first-electric-circuit:switch}
Un \emph{interrupteur}\index{interrupteur} est une porte dans le
circuit~: un petit pont de métal que le clic pose (fermé --- la
boucle est complète, de la \omterm{def:g1:light-and-shadows:source}{lumière}~!) ou soulève (une coupure ---
l'obscurité). Le bouton de la \omterm{def:g3:first-electric-circuit:bulb}{lampe} de poche, l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} du mur,
le bouton d'une sonnette~: tous sont des portes dans une boucle. Tu
peux en fabriquer un avec une pince à linge et une bande de papier
d'aluminium repliée sur ses mâchoires.
\end{example}

\begin{remark}[Des dessins bien propres]\label{rem:g3:first-electric-circuit:symbols}
Plutôt que de dessiner chaque bosse de la \omterm{def:g3:first-electric-circuit:battery}{pile} et chaque pas de vis
de la \omterm{def:g3:first-electric-circuit:bulb}{lampe}, les électriciens utilisent des \emph{symboles}~: deux
barres inégales pour la \omterm{def:g3:first-electric-circuit:battery}{pile} (la longue est le $+$), un cercle barré
d'une croix pour la \omterm{def:g3:first-electric-circuit:bulb}{lampe}, un petit pont-levis pour l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur},
et des traits droits pour les fils. Un schéma en symboles se dessine
plus vite, et n'importe qui dans le monde peut le lire --- nous
utiliserons ces schémas pendant de nombreuses années.
\end{remark}

\begin{remark}[La pile oui, la prise jamais]\label{rem:g3:first-electric-circuit:safety}
Tout ce chapitre se joue avec de petites \omterm{def:g3:first-electric-circuit:battery}{piles} rondes ou plates~:
leur poussée est minuscule et sans danger pour les expériences. La
\emph{prise de courant} est un autre monde --- sa poussée est des
centaines de fois plus puissante, assez pour blesser ou tuer.
N'introduis jamais de fils, de doigts ni quoi que ce soit dans une
prise, et ne démonte jamais un appareil qui s'y branche. Les jeux
avec une \omterm{def:g3:first-electric-circuit:battery}{pile}~: oui. La prise~: jamais.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:1}
Nomme les deux \omterm{def:g3:first-electric-circuit:battery}{bornes} d'une \omterm{def:g3:first-electric-circuit:battery}{pile} et les deux endroits de contact
d'une \omterm{def:g3:first-electric-circuit:bulb}{lampe}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:2}
Qu'est-ce qu'un \omterm{def:g3:first-electric-circuit:circuit}{circuit fermé}~? Qu'est-ce qu'un \omterm{def:g3:first-electric-circuit:circuit}{circuit ouvert}~? Dans
lequel la \omterm{def:g3:first-electric-circuit:bulb}{lampe} s'allume-t-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:3}
Léa relie la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} au plot de la \omterm{def:g3:first-electric-circuit:bulb}{lampe} avec un fil, et
s'arrête là. La \omterm{def:g3:first-electric-circuit:bulb}{lampe} reste éteinte. Que manque-t-il, dans les mots
de la loi de la boucle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:4}
Quel métier fait un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} dans la boucle~? Qu'y a-t-il à
l'intérieur d'un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} quand il est fermé~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:5}
Dessine, avec les symboles bien propres de
\cref{rem:g3:first-electric-circuit:symbols}, un circuit comportant
une \omterm{def:g3:first-electric-circuit:battery}{pile}, un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} fermé et une \omterm{def:g3:first-electric-circuit:bulb}{lampe}. Quel symbole
correspond à quoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:6}
La \omterm{def:g3:first-electric-circuit:bulb}{lampe} de poche s'éteint au milieu d'un camping. Donne trois
coupures ou pannes possibles à vérifier, d'après
\cref{met:g3:first-electric-circuit:light}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:7}
Dans le circuit d'une sonnette, où se trouve l'«~\omterm{ex:g3:first-electric-circuit:switch}{interrupteur}~»~?
Pourquoi la sonnerie ne retentit-elle que tant qu'on appuie sur le
bouton~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:8}
Pourquoi ne faut-il jamais enfoncer quoi que ce soit dans une prise
de courant, alors que jouer avec une \omterm{def:g3:first-electric-circuit:battery}{pile} ne pose pas de problème~?
(Voir \cref{rem:g3:first-electric-circuit:safety}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:9}
Une \omterm{def:g3:first-electric-circuit:bulb}{lampe} tenue par son plot sur la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile}, et un fil
partant de la \omterm{def:g3:first-electric-circuit:battery}{borne} $-$ pressé contre le côté métallique du culot~:
s'allumera-t-elle~? Suis la boucle du doigt pour décider.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:10}
Sam construit une boucle~: \omterm{def:g3:first-electric-circuit:battery}{pile}, fil, \omterm{def:g3:first-electric-circuit:bulb}{lampe}, fil, retour à la \omterm{def:g3:first-electric-circuit:battery}{pile}
--- mais il branche les deux fils sur la même \omterm{def:g3:first-electric-circuit:battery}{borne}, le $-$, «~parce
qu'elle était plus près~». La \omterm{def:g3:first-electric-circuit:bulb}{lampe} reste éteinte alors que la boucle
paraît complète. Explique ce qui manque à sa boucle, à l'aide de
\cref{ex:g3:first-electric-circuit:bothends}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:11}
Dans une guirlande de fête, il manque une \omterm{def:g3:first-electric-circuit:bulb}{lampe}, et la guirlande
\emph{entière} est éteinte, alors que toutes les autres \omterm{def:g3:first-electric-circuit:bulb}{lampes} sont
en bon état. Que t'apprend cela sur la façon dont les \omterm{def:g3:first-electric-circuit:bulb}{lampes} sont
disposées dans la boucle~? (Le chapitre sur les circuits de l'an
prochain donnera un nom à cette disposition.)
\end{exercise}
