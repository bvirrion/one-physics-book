\chapter{Mesurer la masse}\label{ch:g3:measuring-mass}

Au marché~: «~Un \omterm{def:g3:measuring-mass:units}{kilogramme} de pommes, s'il vous plaît~!~» Le vendeur
verse des pommes dans le plateau jusqu'à ce que la \omterm{def:g3:levers-and-scales:scale}{balance} soit
contente. Qu'a donc commandé exactement la cliente --- et comment la
\omterm{def:g3:levers-and-scales:scale}{balance} sait-elle quand s'arrêter~? Avec les \omterm{def:g3:levers-and-scales:lever}{leviers} et les \omterm{def:g3:levers-and-scales:scale}{balances}
dans notre boîte à outils, nous voilà prêts à \omterm{def:g2:measuring-length:measure}{mesurer} une grandeur
nouvelle~: la \omterm{def:g3:measuring-mass:mass}{masse}.

\section{La masse~: la quantité de matière}

\begin{definition}[Masse]\label{def:g3:measuring-mass:mass}
La \emph{masse}\index{masse} d'un objet dit quelle quantité de
matière le compose. Plus il y a de matière, plus la masse est
grande --- et plus la Terre le tire fort vers le bas, ce qui explique
que les objets de grande masse paraissent plus lourds dans la main et
font descendre leur plateau de \omterm{def:g3:levers-and-scales:scale}{balance}.
\end{definition}

\begin{example}[Gros ne veut pas dire lourd]\label{ex:g3:measuring-mass:bignotcheavy}
Un oreiller remplit tes deux bras, et pourtant tu le portes sans
peine. Une pierre plus petite que ton poing fatigue ton poignet.
L'oreiller est gros mais contient peu de matière --- surtout des
plumes et de l'\omterm{def:g2:air-around-us:air}{air}~; la pierre est petite mais bourrée de matière. La
\omterm{def:g3:measuring-mass:mass}{masse} parle de la \emph{quantité de matière}, pas de la place
occupée. (Ces deux idées, et la danse délicate entre elles, se
retrouveront un jour dans l'histoire de ce qui \omterm{def:g1:floating-and-sinking:words}{flotte} et de ce qui
\omterm{def:g1:floating-and-sinking:words}{coule}.)
\end{example}

\begin{definition}[Gramme et kilogramme]\label{def:g3:measuring-mass:units}
La \omterm{def:g3:measuring-mass:mass}{masse} se mesure en \emph{grammes}\index{gramme} (notés \unit{g})
et en \emph{kilogrammes}\index{kilogramme} (notés \unit{kg})~:
\[
\qty{1}{kg} = \qty{1000}{g}.
\]
Un trombone a une \omterm{def:g3:measuring-mass:mass}{masse} d'environ \qty{1}{g}~; une pomme, environ
\qty{150}{g}~; une brique de lait, environ \qty{1}{kg}~; toi,
quelques dizaines de kilogrammes. Comme le \omterm{def:g2:measuring-length:units}{mètre}, ces \omterm{def:g2:measuring-length:measure}{unités} sont les
mêmes pour tout le monde sur Terre.
\end{definition}

\section{Peser avec une balance}

\begin{method}[Peser avec des masses marquées]\label{met:g3:measuring-mass:weigh}
Pour \omterm{def:g2:measuring-length:measure}{mesurer} la \omterm{def:g3:measuring-mass:mass}{masse} d'un melon avec une \omterm{def:g3:levers-and-scales:scale}{balance} et une boîte de
\omterm{def:g3:measuring-mass:mass}{masses} marquées (\qty{500}{g}, \qty{200}{g}, \qty{100}{g},
\qty{50}{g}, \dots)~:
\begin{enumerate}
  \item vérifier que la \omterm{def:g3:levers-and-scales:scale}{balance} à vide reste horizontale~;
  \item poser le melon dans un plateau~;
  \item ajouter des \omterm{def:g3:measuring-mass:mass}{masses} dans l'autre plateau, en commençant par
        les plus grandes qui conviennent, jusqu'à ce que les plateaux
        soient au même niveau~;
  \item additionner les \omterm{def:g3:measuring-mass:mass}{masses}~: leur total est la \omterm{def:g3:measuring-mass:mass}{masse} du melon.
\end{enumerate}
Si les plateaux s'équilibrent avec une \omterm{def:g3:measuring-mass:mass}{masse} de \qty{500}{g}, une de
\qty{200}{g} et une de \qty{100}{g}, la \omterm{def:g3:measuring-mass:mass}{masse} du melon est
$500 + 200 + 100 = \qty{800}{g}$.
\end{method}

\begin{example}[D'autres balances]\label{ex:g3:measuring-mass:otherscales}
La \omterm{def:g3:levers-and-scales:scale}{balance} de cuisine et le pèse-personne n'ont pas de second
plateau~: on pose la charge, et une aiguille tourne (ou un nombre
s'affiche) pour indiquer directement la \omterm{def:g3:measuring-mass:mass}{masse}. À l'intérieur, c'est
un ressort comprimé qui fait la comparaison à ta place. Pratique et
rapide --- mais la vieille \omterm{def:g3:levers-and-scales:scale}{balance} à deux plateaux, elle, ne cache
aucun mécanisme~: rien que la règle du \omterm{def:g3:levers-and-scales:lever}{levier} que tu comprends déjà.
\end{example}

\begin{example}[Lire le marché]\label{ex:g3:measuring-mass:market}
«~Un \omterm{def:g3:measuring-mass:units}{kilogramme} de pommes~» veut dire~: des pommes jusqu'à ce que la
\omterm{def:g3:levers-and-scales:scale}{balance} équilibre une \omterm{def:g3:measuring-mass:mass}{masse} de \qty{1}{kg} --- environ $6$ ou $7$
pommes. La farine se vend en sacs de \qty{1}{kg}, les tablettes de
chocolat font souvent \qty{100}{g}, une lettre demande environ
\qty{20}{g} de papier, et la règle du facteur «~moins de
\qty{20}{g}, un seul timbre~» est une règle de \omterm{def:g3:measuring-mass:mass}{masse}.
\end{example}

\section{La masse suit la matière}

\begin{proposition}[Même matière, même masse]\label{prop:g3:measuring-mass:conserved}
Changer la \emph{forme} d'un objet ne change pas sa \omterm{def:g3:measuring-mass:mass}{masse}. Roule une
boule de pâte en serpent, aplatis-la en galette, ou coupe-la en
quatre et pèse tous les morceaux ensemble~: la \omterm{def:g3:levers-and-scales:scale}{balance} annonce la
même \omterm{def:g3:measuring-mass:mass}{masse} à chaque fois. La \omterm{def:g3:measuring-mass:mass}{masse} compte la matière --- et écraser,
rouler et couper n'ajoutent ni n'enlèvent de matière.
\end{proposition}

\begin{remark}[La balance ignore les apparences]\label{rem:g3:measuring-mass:appearances}
Voilà la \omterm{def:g2:pushes-and-pulls:force}{force} tranquille de la mesure~: les yeux jugent une galette
aplatie «~plus grosse~» que la boule dont elle vient, et un oreiller
gonflé «~plus lourd~» qu'une pierre --- mais la \omterm{def:g3:levers-and-scales:scale}{balance}, comme le
\omterm{def:g3:temperature-thermometers:temperature}{thermomètre} et la règle graduée avant elle, annonce la matière et
rien d'autre. L'un après l'autre, nos instruments remplacent «~on
dirait~» par «~c'est~».
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:1}
Que dit la \omterm{def:g3:measuring-mass:mass}{masse} d'un objet~? Quelles deux \omterm{def:g2:measuring-length:measure}{unités} la mesurent, et
combien de \omterm{def:g3:measuring-mass:units}{grammes} font un \omterm{def:g3:measuring-mass:units}{kilogramme}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:2}
Choisis la \omterm{def:g3:measuring-mass:mass}{masse} la plus probable~: un trombone --- \qty{1}{g} ou
\qty{1}{kg}~? Une pomme --- \qty{150}{g} ou \qty{15}{kg}~? Une brique
de lait --- \qty{1}{g} ou \qty{1}{kg}~? Un chat --- \qty{4}{kg} ou
\qty{400}{kg}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:3}
Les plateaux s'équilibrent quand un potiron fait face à une \omterm{def:g3:measuring-mass:mass}{masse} de
\qty{500}{g} et à deux \omterm{def:g3:measuring-mass:mass}{masses} de \qty{200}{g}. Quelle est la \omterm{def:g3:measuring-mass:mass}{masse} du
potiron~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:4}
Pourquoi la \omterm{def:g3:levers-and-scales:scale}{balance} à vide doit-elle rester horizontale avant toute
pesée~? (Souviens-toi du marchand tricheur.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:5}
Un ballon de plage est bien plus gros qu'une boîte de conserve, et
pourtant c'est la boîte qui fait descendre son plateau. Explique avec
\cref{ex:g3:measuring-mass:bignotcheavy}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:6}
Combien de \omterm{def:g3:measuring-mass:units}{grammes} y a-t-il dans \qty{2}{kg}~? Dans un \omterm{def:g3:measuring-mass:units}{demi-kilogramme}~?
Qu'est-ce qui est le plus~: \qty{1200}{g} ou \qty{1}{kg}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:7}
Une tablette de chocolat de \qty{100}{g} est cassée en tous ses
petits carrés, et tous les carrés vont ensemble sur la \omterm{def:g3:levers-and-scales:scale}{balance}.
Quelle \omterm{def:g3:measuring-mass:mass}{masse} la \omterm{def:g3:levers-and-scales:scale}{balance} indique-t-elle, et quelle loi le dit~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:8}
La \omterm{def:g3:levers-and-scales:scale}{balance} de cuisine et la \omterm{def:g3:levers-and-scales:scale}{balance} à deux plateaux pèsent toutes
deux un sac de farine. Qu'utilise chacune pour faire la
comparaison --- et laquelle fonctionne grâce à une règle apprise il y
a trois chapitres~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:9}
Le boulanger a besoin de \qty{800}{g} de farine et dispose de \omterm{def:g3:measuring-mass:mass}{masses}
de \qty{500}{g}, \qty{200}{g}, \qty{200}{g}, \qty{100}{g} et
\qty{50}{g}. Trouve deux façons différentes de peser exactement
\qty{800}{g} --- sachant que les \omterm{def:g3:measuring-mass:mass}{masses} ont le droit de se placer
dans \emph{l'un ou l'autre} plateau.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:10}
Nina pèse une orange entière~: \qty{200}{g}. Elle l'épluche et pèse
l'orange épluchée seule~: \qty{160}{g}.
\cref{prop:g3:measuring-mass:conserved} est-elle en défaut~? Où sont
passés les \omterm{def:g3:measuring-mass:units}{grammes} manquants, et comment Nina pourrait-elle le
prouver avec une pesée de plus~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:11}
Une devinette, désormais facile pour toi~: qu'est-ce qui a la plus
grande \omterm{def:g3:measuring-mass:mass}{masse}, un \omterm{def:g3:measuring-mass:units}{kilogramme} de plumes ou un \omterm{def:g3:measuring-mass:units}{kilogramme} de fer~? Et
qu'est-ce qui prend le plus de place~? Explique pourquoi tant de gens
répondent mal à la première question.
\end{exercise}
