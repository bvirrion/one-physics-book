\chapter{Leviers et balances}\label{ch:g3:levers-and-scales}

L'an dernier, sur la bascule, nous avons découvert que le poids et la
distance comptent tous les deux --- et nous avons promis une règle
précise, avec de vrais nombres dedans. Il est temps de tenir la
promesse. Au passage, nous soulèverons des pierres bien trop lourdes
pour nos bras, et nous rencontrerons l'instrument qui a pesé les
marchandises du monde pendant des milliers d'années.

\section{Le levier}

\begin{definition}[Levier et pivot]\label{def:g3:levers-and-scales:lever}
Un \emph{levier}\index{levier} est une barre rigide posée sur un
point d'appui autour duquel elle peut basculer. Ce point d'appui
s'appelle le \emph{pivot}\index{pivot}. Une bascule est un levier~;
une barre à mine glissée sous un rocher aussi, tout comme la panne
d'un marteau qui arrache un clou.
\end{definition}

\begin{example}[Le multiplicateur de force]\label{ex:g3:levers-and-scales:crowbar}
Un jardinier glisse une longue barre de fer sous un gros rocher, la
fait reposer sur une bûche placée tout près du rocher, et appuie sur
l'extrémité opposée --- et le rocher, que dix mains n'arrivaient pas
à soulever, bascule sans effort. Le secret du \omterm{def:g3:levers-and-scales:lever}{levier} est la
distance~: appuie \emph{loin} du \omterm{def:g3:levers-and-scales:lever}{pivot}, et ta petite poussée fait le
travail d'un géant placé tout près. Une vieille histoire raconte
qu'un grand savant de l'Antiquité se vantait ainsi~: «~Donnez-moi un
point d'appui et un \omterm{def:g3:levers-and-scales:lever}{levier} assez long, et je soulèverai la Terre.~»
\end{example}

\section{La règle du levier}

\begin{method}[La bascule à pièces]\label{met:g3:levers-and-scales:coins}
Construis une bascule miniature pour partir à la chasse à la règle~:
\begin{enumerate}
  \item mettre une règle en équilibre en travers d'un crayon, sur sa
        graduation du milieu~;
  \item utiliser des pièces de monnaie identiques comme passagers,
        posées sur des graduations entières~: «~une pièce au $4$~»
        veut dire quatre graduations à partir du crayon~;
  \item charger les deux côtés et noter quand la règle est en
        équilibre.
\end{enumerate}
Essaie~: une pièce au $6$ équilibre une pièce au $6$. Deux pièces
empilées au $3$ équilibrent une pièce au $6$. Trois pièces au $2$ ---
équilibre de nouveau avec une pièce au $6$~!
\end{method}

\begin{proposition}[La règle du levier]\label{prop:g3:levers-and-scales:rule}
Pour chaque côté du \omterm{def:g3:levers-and-scales:lever}{levier}, multiplie le nombre de pièces par leur
distance au \omterm{def:g3:levers-and-scales:lever}{pivot}. Le \omterm{def:g3:levers-and-scales:lever}{levier} est en équilibre exactement quand les
deux côtés donnent le \emph{même produit}~:
\[
2 \text{ pièces} \times 3 \text{ graduations} = 6
\qquad\text{équilibre}\qquad
1 \text{ pièce} \times 6 \text{ graduations} = 6 .
\]
Si le produit d'un côté est plus grand, ce côté descend.
\end{proposition}

\begin{example}[Se servir de la règle]\label{ex:g3:levers-and-scales:using}
Quatre pièces sont posées à la graduation $2$~: leur produit vaut
$4 \times 2 = 8$. Pour les équilibrer avec des pièces à la graduation
$4$, il faut aussi un produit de $8$~: $8 = 2 \times 4$, donc deux
pièces au $4$ font l'affaire. Et une seule pièce pourrait-elle
équilibrer les quatre~? Il faudrait qu'elle soit à la graduation
$8$~: $1 \times 8 = 8$. Plus le passager est léger, plus il doit
s'asseoir loin --- exactement ce que la bascule de Papa nous a
appris, mais cette fois avec des nombres.
\end{example}

\section{La balance}

\begin{definition}[Balance]\label{def:g3:levers-and-scales:scale}
Une \emph{balance}\index{balance} est un \omterm{def:g3:levers-and-scales:lever}{levier} construit pour être
juste~: deux plateaux suspendus à des distances \emph{exactement
égales} du \omterm{def:g3:levers-and-scales:lever}{pivot}. Avec des distances égales, les produits coïncident
exactement quand les poids coïncident --- la balance penche donc du
côté du plateau le plus lourd, et reste horizontale quand les deux
plateaux portent des charges de même poids.
\end{definition}

\begin{example}[Peser en comparant]\label{ex:g3:levers-and-scales:weighing}
Pendant des milliers d'années, les marchands ont vendu la farine,
l'or et le poivre avec des \omterm{def:g3:levers-and-scales:scale}{balances}~: la marchandise dans un plateau,
de petits blocs de métal de référence dans l'autre, en ajoutant des
blocs jusqu'à ce que les plateaux soient au même niveau. La \omterm{def:g3:levers-and-scales:scale}{balance}
ne connaît aucun nombre --- elle répond seulement «~plus lourd~»,
«~plus léger~» ou «~égal~». Mais comparer avec des blocs bien choisis
est tout ce dont un marché honnête a besoin --- et, comme le montrera
le dernier chapitre de cette partie, c'est le chemin qui mène à la
mesure de la masse elle-même.
\end{example}

\begin{remark}[Vérifier la balance à vide]\label{rem:g3:levers-and-scales:fair}
Une \omterm{def:g3:levers-and-scales:scale}{balance} juste doit rester horizontale quand elle est \emph{à
vide}. Si un bras est plus long ou un plateau plus lourd, les
produits diffèrent avant même l'arrivée de la marchandise, et chaque
pesée qui suit triche. Les marchands honnêtes montraient au client
leur \omterm{def:g3:levers-and-scales:scale}{balance} à vide bien horizontale --- une vieille façon, sans
paroles, de dire «~jeu franc~».
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:1}
Montre le \omterm{def:g3:levers-and-scales:lever}{levier} et le \omterm{def:g3:levers-and-scales:lever}{pivot} dans~: une bascule~; \ une barre à mine
posée sur une bûche~; \ une paire de ciseaux. (Les ciseaux cachent
deux \omterm{def:g3:levers-and-scales:lever}{leviers} --- où pivotent-ils~?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:2}
Pour soulever un lourd rocher avec une barre à mine, faut-il appuyer
près du \omterm{def:g3:levers-and-scales:lever}{pivot} ou loin de lui~? Pourquoi, en une phrase~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:3}
Sur la bascule à pièces, calcule le produit pour~: $3$ pièces à la
graduation $4$~; \ $2$ pièces à la graduation $5$~; \ $1$ pièce à la
graduation $6$. Quelles paires parmi celles-ci s'équilibrent~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:4}
Une pièce est posée à la graduation $8$, à gauche. Où faut-il poser
$2$ pièces à droite pour équilibrer~? Et $4$ pièces~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:5}
Côté gauche~: $3$ pièces à la graduation $3$. Côté droit~: $2$ pièces
à la graduation $4$. Calcule les deux produits. Quel côté descend~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:6}
Pourquoi les deux plateaux d'une \omterm{def:g3:levers-and-scales:scale}{balance} sont-ils suspendus à des
distances exactement égales du \omterm{def:g3:levers-and-scales:lever}{pivot}~? Que ferait un bras gauche plus
long à chaque pesée~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:7}
Une \omterm{def:g3:levers-and-scales:scale}{balance} reste horizontale avec un sac de billes dans un plateau
et $6$ blocs identiques dans l'autre. Qu'apprend-on sur le sac~? Que
répond la \omterm{def:g3:levers-and-scales:scale}{balance} si on retire un bloc~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:8}
Une bascule~: Tom pèse autant que $2$ sacs et s'assoit à la
graduation $3$~; sa petite sœur pèse autant qu'$1$ sac. À quelle
graduation doit-elle s'asseoir pour l'équilibrer~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:9}
La panne du marteau arrache un clou~: ta main pousse le bout du
manche, loin de l'endroit où la tête du marteau repose sur le bois.
Où est le \omterm{def:g3:levers-and-scales:lever}{pivot}, et pourquoi le clou --- que tes doigts ne
pourraient jamais arracher --- sort-il~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:levers-and-scales:10}
Papi (lourd comme $4$ sacs) veut faire de la bascule avec la petite
Zoé (lourde comme $1$ sac). La bascule porte les graduations $1$ à
$8$ de chaque côté. Trouve \emph{deux} façons différentes de les
asseoir pour que la bascule soit en équilibre.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:levers-and-scales:11}
Un marchand tricheur allonge en cachette un des bras de sa \omterm{def:g3:levers-and-scales:scale}{balance},
et pose toujours la marchandise du côté du bras long. Les clients
reçoivent-ils plus de marchandise qu'ils n'en paient, ou moins~?
Explique avec la règle du \omterm{def:g3:levers-and-scales:lever}{levier} --- et avec
\cref{rem:g3:levers-and-scales:fair}.
\end{exercise}
