\chapter{Température et thermomètres}\label{ch:g3:temperature-thermometers}

Il y a deux hivers, trois bols d'eau ont trompé tes deux mains~: la
même eau paraissait chaude à l'une et fraîche à l'autre. Nous
promettions alors qu'un instrument trancherait un jour ce genre de
dispute. Le voici --- le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre}, le petit juge de verre qui
transforme le chaud et le froid en un nombre.

\section{Pourquoi le toucher ne suffit pas}

\begin{example}[Des juges partiaux]\label{ex:g3:temperature-thermometers:unfair}
Le toucher juge mal en toute saison~: le toboggan en métal paraît
plus froid que le banc en bois placé à la même \omterm{def:g1:light-and-shadows:shadow}{ombre}~; l'eau tiède
paraît chaude à une main froide et fraîche à une main chaude~; et au
retour de la luge, le couloir glacial paraît douillet. Pour comparer
honnêtement le chaud et le froid --- d'une pièce à l'autre, d'un jour
à l'autre, d'une personne à l'autre --- il nous faut un juge qui ne
change jamais d'avis~: un instrument.
\end{example}

\section{Le thermomètre et ses degrés}

\begin{definition}[Température]\label{def:g3:temperature-thermometers:temperature}
La \emph{température}\index{température} d'un objet est le nombre qui
dit exactement à quel point il est chaud ou froid. Nombre plus grand,
objet plus chaud~; nombre plus petit, objet plus froid. La
température se mesure en \emph{degrés Celsius}\index{degré Celsius},
notés \unit{\celsius}, avec un
\emph{thermomètre}\index{thermomètre}.
\end{definition}

\begin{proposition}[Les deux repères de l'eau]\label{prop:g3:temperature-thermometers:milestones}
L'échelle Celsius est accrochée aux deux grands changements d'état de
l'eau~:
\begin{enumerate}
  \item à \qty{0}{\celsius}, l'eau \omterm{def:g2:ice-water-steam:meltfreeze}{gèle} et devient de la glace (et la
        glace fond)~;
  \item à \qty{100}{\celsius}, l'eau bout et devient de la vapeur.
\end{enumerate}
Entre les deux repères, l'échelle est découpée en $100$ degrés
égaux --- un escalier de petites marches, de la glace à la vapeur.
\end{proposition}

\begin{example}[Comment travaille le juge de
verre]\label{ex:g3:temperature-thermometers:works}
Dans le tube fin du \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} est enfermé un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} coloré. La
chaleur fait gonfler ce \omterm{def:g3:solids-liquids-gases:liquid}{liquide} d'un tout petit peu --- et, dans un
tube aussi fin, un petit gonflement pousse visiblement la colonne
vers le haut. Le froid la fait redescendre. Le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} ne se laisse
pas tromper comme une main~: il gonfle de la même façon pour la même
chaleur, aujourd'hui, demain, pour tout le monde.
\end{example}

\section{Lire un thermomètre}

\begin{method}[Lire un thermomètre]\label{met:g3:temperature-thermometers:read}
\begin{enumerate}
  \item Placer le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} là où l'on veut la \omterm{def:g3:temperature-thermometers:temperature}{température} --- dans
        l'eau, à l'\omterm{def:g1:light-and-shadows:shadow}{ombre} dehors, sous la langue~;
  \item \emph{attendre} --- le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} a besoin de temps pour finir
        de monter ou de descendre~; le regarder jusqu'à ce qu'il
        s'arrête~;
  \item mettre l'œil au niveau du sommet de la colonne~;
  \item lire le nombre au trait atteint par la colonne, et l'écrire
        avec son \omterm{def:g2:measuring-length:measure}{unité}~: \qty{18}{\celsius}.
\end{enumerate}
Lire trop tôt, ou de biais, donne un nombre faux --- le juge est
honnête, mais il faut l'interroger correctement.
\end{method}

\begin{example}[Des températures à retenir]\label{ex:g3:temperature-thermometers:livedby}
Quelques nombres qui méritent d'être sus par cœur~: une pièce
agréable est à environ \qty{20}{\celsius}~; ton corps se maintient à
environ \qty{37}{\celsius}, été comme hiver (bien au-dessus, c'est de
la fièvre --- le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} du médecin le sait)~; le réfrigérateur
tient environ \qty{4}{\celsius}~; une journée d'été chaude atteint
\qty{30}{\celsius}~; l'eau du bain est agréable vers
\qty{35}{\celsius}.
\end{example}

\section{Au-dessous de zéro}

\begin{example}[Les nombres de l'hiver]\label{ex:g3:temperature-thermometers:belowzero}
Un matin de gel, la colonne descend \emph{au-dessous} du trait $0$.
Les traits continuent là-dessous, et on les lit avec le mot
«~moins~»~: un trait sous zéro, c'est \emph{moins un} degré, cinq
traits sous zéro, c'est \emph{moins cinq}, que l'on écrit
\qty{-5}{\celsius}. Les nombres négatifs sont plus froids que zéro,
et plus ils descendent, plus le gel est rude~: un matin à $-10$ mord
plus fort qu'un matin à $-2$. Le bulletin météo les annonce tous les
soirs d'hiver.
\end{example}

\begin{remark}[Zéro ne veut pas dire «~rien~»]\label{rem:g3:temperature-thermometers:zero}
Sur cette échelle, $0$ ne signifie pas «~aucune \omterm{def:g3:temperature-thermometers:temperature}{température}~» --- ce
n'est que le trait où l'eau \omterm{def:g2:ice-water-steam:meltfreeze}{gèle}, un repère comme un panneau au bord
de la route. Le temps peut être plus chaud que ce repère ou plus
froid~; le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} dit simplement de quel côté, et de combien de
traits. (Les nombres au-dessous de zéro ont toute une arithmétique à
eux --- ton cours de mathématiques s'en occupera pour de bon dans une
année plus avancée.)
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:1}
Pourquoi un \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} juge-t-il le chaud et le froid plus
honnêtement que ta main~? Cite une façon dont les mains ont été
trompées dans des chapitres précédents.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:2}
À quelle \omterm{def:g3:temperature-thermometers:temperature}{température} l'eau gèle-t-elle~? À laquelle bout-elle~? Dans
quelle \omterm{def:g2:measuring-length:measure}{unité} sont ces deux nombres~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:3}
Qu'arrive-t-il au \omterm{def:g3:solids-liquids-gases:liquid}{liquide} du tube fin quand on réchauffe le
\omterm{def:g3:temperature-thermometers:temperature}{thermomètre}~? Quand on le refroidit~? Pourquoi un si petit changement
se voit-il si nettement~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:4}
Cite les deux erreurs contre lesquelles
\cref{met:g3:temperature-thermometers:read} met en garde au moment de
lire un \omterm{def:g3:temperature-thermometers:temperature}{thermomètre}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:5}
Associe chaque objet à sa \omterm{def:g3:temperature-thermometers:temperature}{température} probable ---
\qty{4}{\celsius}, \qty{20}{\celsius}, \qty{37}{\celsius},
\qty{100}{\celsius}~: de l'eau qui bout~; \ un salon douillet~; \
l'intérieur du réfrigérateur~; \ ton propre corps.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:6}
Le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} de bain indique \qty{35}{\celsius}~; la casserole sur
la cuisinière indique \qty{100}{\celsius}. Laquelle est la plus
chaude, et peut-on toucher la casserole sans danger~? De combien de
degrés la casserole est-elle plus chaude que le bain~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:7}
Un \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} d'hiver montre la colonne $3$ traits sous zéro.
Comment dit-on et comment écrit-on cette \omterm{def:g3:temperature-thermometers:temperature}{température}~? Fait-il plus
froid ou plus chaud qu'à \qty{-8}{\celsius}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:8}
Hier, le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} du jardin indiquait \qty{6}{\celsius} au
petit-déjeuner et \qty{14}{\celsius} au déjeuner. De combien de
degrés la matinée s'est-elle réchauffée~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:9}
Les flaques du trottoir se sont changées en glace pendant la nuit.
Que peux-tu dire de la \omterm{def:g3:temperature-thermometers:temperature}{température} de la nuit, sans aucun
\omterm{def:g3:temperature-thermometers:temperature}{thermomètre}~? Quel repère utilises-tu~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:temperature-thermometers:10}
En rentrant tout droit de la luge, Emma se précipite dans le couloir
et s'écrie~: «~On étouffe ici~!~» Le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} du couloir indique
\qty{16}{\celsius}. Fait-il chaud dans le couloir~? Pourquoi Emma
le ressent-elle ainsi --- et quel juge la famille doit-elle croire~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:temperature-thermometers:11}
Une cuisinière veut savoir si une soupe à gros bouillons devient plus
chaude à mesure qu'elle bout. Elle mesure~: après une minute
d'ébullition, \qty{100}{\celsius}~; après dix minutes, toujours
\qty{100}{\celsius}. Que lui apprend le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} sur l'eau qui
bout --- et sur les repères de l'eau~?
\end{exercise}
