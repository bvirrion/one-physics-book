\chapter{درجة الحرارة وموازينها}\label{ch:g3:temperature-thermometers}

قبل شتاءين خدعت ثلاثة أوعية ماء يديك: فبدا الماء نفسه دافئًا
لإحداهما وباردًا للأخرى. ووعدنا حينها بأن جهازًا سيفصل يومًا في
مثل هذه الخصومات. وها هو ذا --- \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة}، ذلك القاضي الزجاجي
الصغير الذي يحوّل الساخن والبارد إلى عدد.

\section{لماذا لا يكفي اللمس}

\begin{example}[القضاة غير العادلين]\label{ex:g3:temperature-thermometers:unfair}
اللمس يقضي بغير عدل في كل فصل من فصول السنة: فالمِزلج المعدني يبدو
أبرد من المقعد الخشبي الواقف في \omterm{def:g1:light-and-shadows:shadow}{الظلّ} نفسه؛ والماء الفاتر يبدو
دافئًا ليد باردة وباردًا ليد دافئة؛ وبعد التزحلق يبدو الممرّ
القارس دافئًا. فلكي نقارن الساخن والبارد مقارنة أمينة --- بين
الغرف، وبين الأيام، وبين الناس --- نحتاج إلى قاضٍ لا يغيّر رأيه
أبدًا: إلى جهاز.
\end{example}

\section{ميزان الحرارة ودرجاته}

\begin{definition}[درجة الحرارة]\label{def:g3:temperature-thermometers:temperature}
\emph{درجة الحرارة}\index{درجة حرارة} في شيء هي العدد الذي يقول
بالضبط كم هو ساخن أو بارد. فالعدد الأعلى للشيء الأسخن؛ والأدنى
للأبرد. وتُقاس درجة الحرارة \emph{بالدرجة
السلسيوسية}\index{درجة سلسيوسية}، وتُكتب \unit{\celsius}،
بواسطة \emph{ميزان الحرارة}\index{ميزان حرارة}.
\end{definition}

\begin{proposition}[معلَما الماء]\label{prop:g3:temperature-thermometers:milestones}
يُثبَّت السلّم السلسيوسي على تغيّرَي حالة الماء العظيمين:
\begin{enumerate}
  \item عند \qty{0}{\celsius} \omterm{def:g2:ice-water-steam:meltfreeze}{يتجمّد} الماء ثلجًا (\omterm{def:g2:ice-water-steam:meltfreeze}{وينصهر} الثلج)؛
  \item وعند \qty{100}{\celsius} يغلي الماء بخارًا.
\end{enumerate}
وبين المعلمين يُقسَّم السلّم $100$ درجة متساوية --- سلّم من
خطوات أصغر من الثلج إلى البخار.
\end{proposition}

\begin{example}[كيف يعمل القاضي الزجاجي]\label{ex:g3:temperature-thermometers:works}
داخل أنبوب \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} الرفيع \omterm{def:g3:solids-liquids-gases:liquid}{سائل} ملوّن محبوس. والدفء يجعل ذلك
\omterm{def:g3:solids-liquids-gases:liquid}{السائل} يتمدّد قليلًا --- وفي أنبوب بهذه الرفاعة يدفع التمدّد
القليل العمودَ صاعدًا دفعًا ظاهرًا للعين. والبرد يقلّصه نازلًا.
\omterm{def:g3:solids-liquids-gases:liquid}{والسائل} لا يُخدع كما تُخدع اليد: فهو يتمدّد التمدّد نفسه للدفء
نفسه، اليوم وغدًا، وعند كل الناس.
\end{example}

\section{قراءة ميزان الحرارة}

\begin{method}[كيف تقرأ ميزان الحرارة]\label{met:g3:temperature-thermometers:read}
\begin{enumerate}
  \item ضع \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} حيث تريد معرفة \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} --- في
        الماء، أو في \omterm{def:g1:light-and-shadows:shadow}{الظلّ} في الخارج، أو تحت اللسان؛
  \item \emph{وانتظر} --- \omterm{def:g3:solids-liquids-gases:liquid}{فالسائل} يحتاج إلى وقت ليتمّ صعوده أو
        نزوله؛ وراقبه حتى يكفّ عن الحركة؛
  \item واجعل عينك في مستوى قمة العمود؛
  \item واقرأ العدد عند العلامة التي يبلغها العمود، واكتبه مع
        وحدته: \qty{18}{\celsius}.
\end{enumerate}
والقراءة قبل الأوان، أو من زاوية مائلة، تعطي عددًا خاطئًا ---
فالقاضي أمين، لكن لا بدّ من مساءلته على أصولها.
\end{method}

\begin{example}[درجات حرارة تُعاش بها]\label{ex:g3:temperature-thermometers:livedby}
بعض الأعداد جدير بأن يُحفظ عن ظهر قلب: فالغرفة اللطيفة نحو
\qty{20}{\celsius}؛ وجسمك يحفظ نفسه عند نحو \qty{37}{\celsius}،
في الصيف والشتاء سواء (وما زاد على ذلك كثيرًا فحمّى --- \omterm{def:g3:temperature-thermometers:temperature}{وميزان
حرارة} الطبيب يعرف)؛ والثلاجة عند نحو \qty{4}{\celsius}؛ ويوم
الصيف الحارّ يبلغ \qty{30}{\celsius}؛ وماء الحمّام لطيف قرب
\qty{35}{\celsius}.
\end{example}

\section{تحت الصفر}

\begin{example}[أعداد الشتاء]\label{ex:g3:temperature-thermometers:belowzero}
في صباح متجمّد ينزل العمود \emph{تحت} العلامة $0$. والعلامات
تتابع نزولها هناك، ونقرؤها بكلمة ``ناقص'': فعلامة واحدة تحت
الصفر هي \emph{ناقص واحد} درجة، وخمس علامات تحته \emph{ناقص
خمسة}، وتُكتب \qty{-5}{\celsius}. والأعداد الناقصة أبرد من الصفر،
وكلما نزلت اشتدّ \omterm{def:g2:ice-water-steam:meltfreeze}{التجمّد}: فصباح عند $-10$ يقرص أشدّ من صباح عند
$-2$. وتذكرها نشرة الطقس كل مساء شتوي.
\end{example}

\begin{remark}[الصفر ليس ``لا شيء'']\label{rem:g3:temperature-thermometers:zero}
في هذا السلّم لا يعني $0$ ``لا \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة}'' --- إنما هو العلامة
التي \omterm{def:g2:ice-water-steam:meltfreeze}{يتجمّد} عندها الماء، معلمٌ مثل لافتة على الطريق. والطقس قد
يكون أدفأ من المعلم أو أبرد منه؛ \omterm{def:g3:temperature-thermometers:temperature}{وميزان الحرارة} لا يخبرك إلا في
أيّ الجهتين هو، وبكم علامة. (وللأعداد تحت الصفر حساب كامل خاصّ
بها --- وسيتناوله منهاج الرياضيات على أصوله في سنة لاحقة.)
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:1}
لماذا يكون \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} قاضيًا أعدل من يدك في الساخن والبارد؟
واذكر وجهًا واحدًا خُدعت فيه الأيدي في فصول سابقة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:2}
عند أيّ \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} \omterm{def:g2:ice-water-steam:meltfreeze}{يتجمّد} الماء؟ وعند أيّها يغلي؟ وبأيّ \omterm{def:g2:measuring-length:measure}{وحدة}
العددان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:3}
ماذا يحدث \omterm{def:g3:solids-liquids-gases:liquid}{للسائل} في الأنبوب الرفيع حين يسخن \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة}؟ وحين
يبرد؟ ولماذا يظهر تغيّر بهذا الصغر ظهورًا بيّنًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:4}
اذكر الخطأين اللذين تحذّر منهما
\cref{met:g3:temperature-thermometers:read} عند قراءة \omterm{def:g3:temperature-thermometers:temperature}{ميزان
الحرارة}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:5}
وافِق بين كل شيء ودرجة حرارته المرجّحة ---
\qty{4}{\celsius}، \qty{20}{\celsius}، \qty{37}{\celsius}،
\qty{100}{\celsius}: ماء يغلي؛ \ غرفة جلوس دافئة؛ \ داخل
الثلاجة؛ \ جسمك أنت.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:6}
يقرأ \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} الحمّام \qty{35}{\celsius}؛ ويقرأ القِدر على
الموقد \qty{100}{\celsius}. أيّهما أسخن، وهل يؤمَن لمس القِدر؟
وبكم درجة يكون القِدر أسخن من الحمّام؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:7}
يُظهر \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} شتوي العمودَ عند $3$ علامات تحت الصفر. كيف
نقول تلك الدرجة ونكتبها؟ وأهي أبرد من \qty{-8}{\celsius} أم
أدفأ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:8}
قرأ \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} الحديقة أمس \qty{6}{\celsius} عند الفطور
و\qty{14}{\celsius} عند الغداء. بكم درجة دفئ الصباح؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:temperature-thermometers:9}
تحوّلت البِرَك على الرصيف ثلجًا خلال الليل. فماذا تستطيع أن تقول
عن \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} الليل من غير أي \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة}؟ وأيّ معلم تستعمل؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:temperature-thermometers:10}
بعد التزحلق مباشرة تجري إيما إلى الممرّ وتصيح: ``الجوّ هنا يغلي!''
ويقرأ \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} الممرّ \qty{16}{\celsius}. أالممرّ ساخن؟ ولماذا
تحسّه إيما كذلك --- وأيّ قاضٍ ينبغي أن تثق به الأسرة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:temperature-thermometers:11}
تريد طاهية أن تعرف أيسخن الحساء الغالي أكثر كلما طال غليانه.
\omterm{def:g2:measuring-length:measure}{فتقيس}: بعد دقيقة من الغليان \qty{100}{\celsius}؛ وبعد عشر دقائق
\qty{100}{\celsius} أيضًا. فماذا يعلّمها \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} عن الماء
الغالي --- وعن معلمَي الماء؟
\end{exercise}
