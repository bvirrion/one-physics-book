\chapter{قياس الكتلة}\label{ch:g3:measuring-mass}

في السوق: ``كيلوغرامًا من التفّاح من فضلك!'' فيصبّ البائع التفّاح
في الكفّة حتى يرضى الميزان. فما الذي طلبه الزبون بالضبط --- وكيف
يعرف الميزان متى يتوقّف؟ ومعنا في عُدّتنا الروافع وموازين
الكفّتين، فقد صرنا مستعدّين \omterm{def:g2:measuring-length:measure}{لقياس} شيء جديد: \omterm{def:g3:measuring-mass:mass}{الكتلة}.

\section{الكتلة: كم فيه من مادة}

\begin{definition}[الكتلة]\label{def:g3:measuring-mass:mass}
\emph{كتلة}\index{كتلة} الشيء تقول كم فيه من مادة صُنع منها.
فكلما زادت المادة كبرت الكتلة --- واشتدّ جذب الأرض له إلى الأسفل،
ولهذا تبدو الأشياء الأكبر كتلةً أثقل في يدك وتضغط كفّتها في
\omterm{def:g3:levers-and-scales:scale}{ميزان الكفّتين} إلى الأسفل.
\end{definition}

\begin{example}[الكبير ليس ثقيلًا]\label{ex:g3:measuring-mass:bignotcheavy}
وسادة تملأ ذراعيك كلتيهما، ومع ذلك تحملها بسهولة. وحجر أصغر من
قبضتك يُجهد معصمك. فالوسادة كبيرة لكنها تحمل قليلًا من المادة ---
أكثرها ريش \omterm{def:g2:air-around-us:air}{وهواء}؛ والحجر صغير لكنه محشوّ بالمادة. \omterm{def:g3:measuring-mass:mass}{فالكتلة} عن
\emph{كمّ المادة}، لا عن الحيّز الذي يشغله الشيء. (وهاتان
الفكرتان، والرقصة الملتبسة بينهما، ستلتقيان من جديد يومًا في قصة
\omterm{def:g1:floating-and-sinking:words}{الطفو} \omterm{def:g1:floating-and-sinking:words}{والغرق}.)
\end{example}

\begin{definition}[الغرام والكيلوغرام]\label{def:g3:measuring-mass:units}
تُقاس \omterm{def:g3:measuring-mass:mass}{الكتلة} \emph{بالغرام}\index{غرام} (ويُكتب \unit{g})
و\emph{بالكيلوغرام}\index{كيلوغرام} (ويُكتب \unit{kg}):
\[
\qty{1}{kg} = \qty{1000}{g}.
\]
\omterm{def:g3:measuring-mass:mass}{فكتلة} مشبك ورق نحو \qty{1}{g}؛ \omterm{def:g3:measuring-mass:mass}{وكتلة} تفّاحة نحو \qty{150}{g}؛
\omterm{def:g3:measuring-mass:mass}{وكتلة} علبة حليب نحو \qty{1}{kg}؛ وأنت بضع عشرات من
الكيلوغرامات. وهاتان الوحدتان، مثل \omterm{def:g2:measuring-length:units}{المتر}، واحدة عند كل من على
الأرض.
\end{definition}

\section{الوزن بميزان الكفّتين}

\begin{method}[الوزن بالقطع المرجعية]\label{met:g3:measuring-mass:weigh}
\omterm{def:g2:measuring-length:measure}{لقياس} \omterm{def:g3:measuring-mass:mass}{كتلة} بطّيخة \omterm{def:g3:levers-and-scales:scale}{بميزان كفّتين} وعلبة من القطع المعدنية المعلَّمة
(\qty{500}{g}، \qty{200}{g}، \qty{100}{g}،
\qty{50}{g}، \dots):
\begin{enumerate}
  \item تحقّق من أن الميزان الفارغ مستوٍ؛
  \item ضع البطّيخة في إحدى الكفّتين؛
  \item أضِف القطع في الكفّة الأخرى، وابدأ بأكبر ما ينفع منها،
        حتى تستوي الكفّتان؛
  \item اجمع القطع: فمجموعها \omterm{def:g3:measuring-mass:mass}{كتلة} البطّيخة.
\end{enumerate}
فإذا استوت الكفّتان مع قطعة \qty{500}{g} وقطعة \qty{200}{g}
وقطعة \qty{100}{g}، كانت \omterm{def:g3:measuring-mass:mass}{كتلة} البطّيخة $500 + 200 + 100 =
\qty{800}{g}$.
\end{method}

\begin{example}[موازين أخرى]\label{ex:g3:measuring-mass:otherscales}
ميزان المطبخ وميزان الحمّام ليس لهما كفّة ثانية: تضع الحِمل
عليهما، فيدور مؤشّر (أو يضيء عدد) ليبيّن \omterm{def:g3:measuring-mass:mass}{الكتلة} مباشرة. وفي
الداخل يقوم نابض مضغوط بالمقارنة نيابة عنك. مريح وسريع --- لكن
\omterm{def:g3:levers-and-scales:scale}{ميزان الكفّتين} القديم لا يخبّئ آلة البتة: إنما هو قاعدة \omterm{def:g3:levers-and-scales:lever}{الرافعة}
التي فهمتها أصلًا.
\end{example}

\begin{example}[قراءة السوق]\label{ex:g3:measuring-mass:market}
``\omterm{def:g3:measuring-mass:units}{كيلوغرام} من التفّاح'' يعني: تفّاحًا حتى يوازن الميزانُ قطعةَ
\qty{1}{kg} --- أي نحو $6$ أو $7$ تفّاحات. والطحين يأتي في أكياس
\qty{1}{kg}، وألواح الشوكولاتة كثيرًا ما تكون \qty{100}{g}،
والرسالة تحتاج إلى نحو \qty{20}{g} من الورق، وقاعدة ساعي البريد
``ما دون \qty{20}{g} بطابع واحد'' قاعدة \omterm{def:g3:measuring-mass:mass}{كتلة}.
\end{example}

\section{الكتلة تتبع المادة}

\begin{proposition}[المادة نفسها والكتلة نفسها]\label{prop:g3:measuring-mass:conserved}
تغيير \emph{شكل} الشيء لا يغيّر كتلته. دحرج كرة عجين حتى تصير
أفعى، أو افرشها فطيرة، أو قطّعها أربعًا وزِن القطع كلها معًا:
يُبلّغ الميزان \omterm{def:g3:measuring-mass:mass}{الكتلة} نفسها في كل مرة. \omterm{def:g3:measuring-mass:mass}{فالكتلة} تعدّ المادة ---
والعصر والدحرجة والتقطيع لا تضيف مادة ولا تنقص منها.
\end{proposition}

\begin{remark}[الميزان لا يعبأ بالمظاهر]\label{rem:g3:measuring-mass:appearances}
هذه هي \omterm{def:g2:pushes-and-pulls:force}{القوة} الهادئة في \omterm{def:g2:measuring-length:measure}{القياس}: فالعيون تحكم بأن الفطيرة
المفروشة ``أكبر'' من الكرة التي جاءت منها، وبأن الوسادة المنفوشة
``أثقل'' من حجر --- أما \omterm{def:g3:levers-and-scales:scale}{ميزان الكفّتين}، مثل \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة}
والمسطرة قبله، فيُبلّغ المادة لا غير. وهكذا تحلّ أجهزتنا، واحدًا
بعد واحد، ``هو كذلك'' محلّ ``يبدو كذلك''.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:1}
ماذا تقول \omterm{def:g3:measuring-mass:mass}{كتلة} الشيء؟ وبأيّ وحدتين تُقاس، وكم غرامًا يصنع
كيلوغرامًا واحدًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:2}
اختر \omterm{def:g3:measuring-mass:mass}{الكتلة} الأرجح: مشبك ورق --- \qty{1}{g} أم \qty{1}{kg}؟
وتفّاحة --- \qty{150}{g} أم \qty{15}{kg}؟ وعلبة حليب ---
\qty{1}{g} أم \qty{1}{kg}؟ وقطّ --- \qty{4}{kg} أم \qty{400}{kg}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:3}
استوت الكفّتان حين واجهت يقطينةٌ قطعةَ \qty{500}{g} وقطعتين من
\qty{200}{g}. فما \omterm{def:g3:measuring-mass:mass}{كتلة} اليقطينة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:4}
لماذا يجب أن يستوي الميزان الفارغ قبل أي وزن؟ (تذكّر التاجر
الغشّاش.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:5}
كرة الشاطئ أكبر بكثير من علبة حساء، ومع ذلك تضغط العلبة كفّتها
إلى الأسفل. اشرح ذلك بما في
\cref{ex:g3:measuring-mass:bignotcheavy}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:6}
كم غرامًا في \qty{2}{kg}؟ وكم في نصف \omterm{def:g3:measuring-mass:units}{كيلوغرام}؟ وأيّهما أكثر:
\qty{1200}{g} أم \qty{1}{kg}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:7}
لوح شوكولاتة كتلته \qty{100}{g} كُسر إلى مربّعاته الصغيرة كلها،
ووُضعت المربّعات كلها معًا على الميزان. فأيّ \omterm{def:g3:measuring-mass:mass}{كتلة} يُظهر الميزان،
وأيّ قانون يقول ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:8}
يزن ميزان المطبخ \omterm{def:g3:levers-and-scales:scale}{وميزان الكفّتين} كلاهما كيس طحين. فبمَ يقارن كل
واحد منهما --- وأيّهما يعمل بقاعدة تعلّمتها قبل ثلاثة فصول؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:9}
يحتاج الخبّاز إلى \qty{800}{g} من الطحين، وعنده قطع من
\qty{500}{g} و\qty{200}{g} و\qty{200}{g} و\qty{100}{g}
و\qty{50}{g}. جِد طريقتين مختلفتين لوزن \qty{800}{g} بالضبط ---
علمًا بأنه يجوز أن توضع القطع في \emph{أيّ} من الكفّتين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:10}
تزن نينا برتقالة كاملة: \qty{200}{g}. ثم تقشّرها وتزن البرتقالة
المقشّرة وحدها: \qty{160}{g}. أفشلت
\cref{prop:g3:measuring-mass:conserved}؟ وأين الغرامات
الناقصة، وكيف تستطيع نينا أن تثبت ذلك بعملية وزن واحدة أخرى؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:11}
أحجية صارت سهلة عليك أخيرًا: أيّهما أكبر كتلةً، \omterm{def:g3:measuring-mass:units}{كيلوغرام} من
الريش أم \omterm{def:g3:measuring-mass:units}{كيلوغرام} من الحديد؟ وأيّهما يشغل حيّزًا أكبر؟ واشرح
لماذا يجيب كثير من الناس عن السؤال الأول إجابة خاطئة.
\end{exercise}
