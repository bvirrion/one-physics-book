\chapter{الصلب والسائل والغاز}\label{ch:g3:solids-liquids-gases}

حجر، وكأس حليب، \omterm{def:g2:air-around-us:air}{والهواء} في بالون: ثلاثة أصناف من المادة تتصرّف
بثلاث طرائق مختلفة اختلافًا تامًّا. في السنة الماضية أرانا الماء
حالاته الثلاث؛ وهذه السنة نكتشف أن \emph{كل شيء} في واحدة من هذه
الحالات الثلاث --- ولكل حالة قواعد صارمة.

\section{عائلات المادة الثلاث}

يمكن تصنيف كل مادة في العالم في ثلاث عائلات تُسمّى \emph{الحالات}
الثلاث: صلبة كالحجر، وسائلة كالحليب، وغازية \omterm{def:g2:air-around-us:air}{كالهواء}. وكان الثلج
والماء والبخار مثالنا الأول: مادة واحدة تزور العائلات الثلاث. فهيا
نتعلّم قواعد كل عائلة.

\section{الأجسام الصلبة تحفظ شكلها}

\begin{definition}[الجسم الصلب]\label{def:g3:solids-liquids-gases:solid}
\emph{الجسم الصلب}\index{جسم صلب} مادة تحفظ شكلها الخاص. فالمفتاح
يحفظ تعرّجه في جيبك، وفي درج، وتحت الماء. وتستطيع أن تمسك جسمًا
صلبًا بين إصبعين، ولا يغيّر تحريكه هيئته. وبعض الأجسام الصلبة
قاسية (الحجر)، وبعضها طرية (الخبز)، وبعضها لدنة (الشريط
المطّاطي) --- لكن كلًّا منها، إذا تُرك وشأنه، يحفظ شكله.
\end{definition}

\begin{example}[جسم صلب من حبيبات]\label{ex:g3:solids-liquids-gases:sand}
يبدو الرمل خارقًا للقاعدة: فهو حين يُسكب من دلو يأخذ شكل كومته،
كأنه \omterm{def:g3:solids-liquids-gases:liquid}{سائل} تقريبًا. لكن انظر عن قرب: الرمل حشد من حبيبات صلبة
دقيقة، و\emph{كل حبيبة} تحفظ شكلها حفظًا تامًّا. والطحين والسكّر
والأرزّ تصنع الحيلة نفسها --- حشود من أجسام صلبة صغيرة تجري معًا
وكل فرد منها يبقى صلبًا.
\end{example}

\section{السوائل تجري --- لكنها تحفظ كمّيتها}

\begin{definition}[السائل]\label{def:g3:solids-liquids-gases:liquid}
\emph{السائل}\index{سائل} مادة تجري: لا شكل خاصّ به، ويأخذ شكل
الإناء الذي يحمله كائنًا ما كان. لكن السائل يحفظ \emph{كمّيته}:
اسكب كأسًا ممتلئة من الحليب في وعاء فتبقى كمّية الحليب هي هي ---
لا قطرة أكثر ولا أقلّ، وإنما شكل جديد.
\end{definition}

\begin{proposition}[سطح السائل الساكن مستوٍ]\label{prop:g3:solids-liquids-gases:level}
سطح \omterm{def:g3:solids-liquids-gases:liquid}{السائل} الساكن مستوٍ أفقيّ دائمًا --- موازٍ للأرض. أمِل
القنّينة: لا يميل الماء الذي فيها معها؛ بل يبقى سطحه أفقيًّا
والقنّينة مائلة.
\end{proposition}

\begin{example}[الاستواء في العمل]\label{ex:g3:solids-liquids-gases:level}
يستعمل البنّاؤون هذا القانون كل يوم: فميزان الماء عندهم نافذة
صغيرة من \omterm{def:g3:solids-liquids-gases:liquid}{سائل} فيها فقاعة واحدة؛ فإذا استقرّت الفقاعة بين
علامتيها كان الرفّ مستويًا استواء \omterm{def:g3:solids-liquids-gases:liquid}{السائل} الذي في داخله. وعند
الفطور، راقب العصير في علبة مائلة: مهما أملتها بقي السطح في
الداخل مستويًا أفقيًّا --- حتى يجد الفوهة.
\end{example}

\section{الغازات تملأ كل شيء}

\begin{definition}[الغاز]\label{def:g3:solids-liquids-gases:gas}
\emph{الغاز}\index{غاز} مادة تنتشر حتى تملأ \emph{كل} الحيّز الذي
يُعطى لها. \omterm{def:g2:air-around-us:air}{والهواء} غاز: ضعه في قنّينة يملأ القنّينة إلى كل ركن
منها؛ وافتح القنّينة في غرفة أكبر فينتشر فيها أيضًا. فليس للغاز
شكل خاصّ به ولا موضع خاصّ يستقرّ فيه --- اعرض عليه حيّزًا أكبر
يأخذه كله.
\end{definition}

\begin{example}[العطر المسافر]\label{ex:g3:solids-liquids-gases:perfume}
يفتح أحدهم قارورة عطر في طرف من الغرفة. وبعد لحظة تنتبه إليه
الأنوف في الطرف \emph{الآخر}. لم يبقَ \omterm{def:g3:solids-liquids-gases:gas}{غاز} العطر مؤدَّبًا قرب
قارورته: بل انتشر حين أُطلق في \omterm{def:g2:air-around-us:air}{هواء} الغرفة كله، من ركن إلى ركن
--- وهذا بالضبط ما تفعله عائلة الغازات دائمًا.
\end{example}

\begin{remark}[تصنيف الحالات الملتبسة]\label{rem:g3:solids-liquids-gases:tricky}
بعض المواد تحتاج إلى عين متأنّية. فالعسل يجري --- ببطء، لكنه
يجري، ويأخذ شكل إنائه، ويحفظ كمّيته: فهو \omterm{def:g3:solids-liquids-gases:liquid}{سائل}، ولكنه كسول. ومعجون
الأسنان والكتشب يجريان حين يُعصران: سائلان ثخينان. والإسفنجة \omterm{def:g3:solids-liquids-gases:solid}{جسم
صلب} مليء بثقوب فيها \omterm{def:g2:air-around-us:air}{هواء}. وأما الدخان؟ فذرّات صلبة دقيقة تركب
غازًا ساخنًا. وعند الشكّ، اطرح أسئلة العائلة: أيحفظ شكله؟ أيجري؟
أيملأ كل الحيّز الذي يناله؟
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:1}
صنّف في العائلات الثلاث: قطعة نقود؛ \ عصير برتقال؛ \ \omterm{def:g2:air-around-us:air}{الهواء} في
إطار سيارة؛ \ خرزة خشبية؛ \ زيت؛ \ بخار الماء الغازي فوق الفجوة
الصافية في قِدر غالٍ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:2}
ما الذي يحفظه \omterm{def:g3:solids-liquids-gases:solid}{الجسم الصلب} ولا يحفظه \omterm{def:g3:solids-liquids-gases:liquid}{السائل}؟ وما الذي يحفظه
\omterm{def:g3:solids-liquids-gases:liquid}{السائل} حين يُسكب من كأس إلى وعاء؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:3}
سُكبت قنّينة ليموناضة ممتلئة في وعاء سلطة عريض. أصار عندنا
ليموناضة أكثر أم أقلّ أم الكمّية نفسها؟ وما الذي تغيّر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:4}
ارسم حوض سمك نصفه ماء: أولًا واقفًا مستويًا، ثم مائلًا على
حافة. ارسم سطح الماء في الحالتين --- في أيّ اتجاه يستوي؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:5}
كيف يخبر ميزان الماء عند البنّاء بأن الرفّ مستوٍ؟ وأيّ قانون من
قوانين هذا الفصل يستعمله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:6}
لماذا تبلغ رائحة الفطائر غرفة نومك في الأعلى؟ أيّ عائلة تعمل
هنا، وما قاعدتها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:7}
الرمل يُسكب \omterm{def:g3:solids-liquids-gases:liquid}{كالسائل}، ومع ذلك نسمّيه صلبًا. اشرح ذلك مستعينًا
بما في \cref{ex:g3:solids-liquids-gases:sand}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:8}
يحتاج العسل إلى دقيقة كاملة لينساب عن الملعقة. أصلب هو أم \omterm{def:g3:solids-liquids-gases:liquid}{سائل}؟
اطرح أسئلة العائلة في
\cref{rem:g3:solids-liquids-gases:tricky} وأجب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:solids-liquids-gases:9}
أيّ عائلة: تحفظ شكلها الخاص؟ \ تأخذ شكل إنائها وتحفظ كمّيتها؟
\ تأخذ كل الحيّز الذي تناله؟ أعطِ فردًا من كل عائلة تجده في
مطبخك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:solids-liquids-gases:10}
مكعّب ثلج (صلب) \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} ماءً (سائلًا)، فتجفّفه شمس الصيف ماءً
طائرًا لا يُرى (غازًا). أيّ قاعدة عائلية تنكسر في كل خطوة: ما
الذي تكفّ المادة عن حفظه حين \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} الثلج؟ وحين \omterm{def:g2:ice-water-steam:evapcond}{يتبخّر} الماء؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:solids-liquids-gases:11}
نُفخ بالون وعُقد. تقول ميرا إن \omterm{def:g2:air-around-us:air}{الهواء} في داخله ``على شكل
البالون، إذن \omterm{def:g3:solids-liquids-gases:gas}{للغاز} شكل''. اشرح ما الذي يعطي الشكل حقًّا، وماذا
يفعل \omterm{def:g2:air-around-us:air}{الهواء} في اللحظة التي ينفجر فيها البالون.
\end{exercise}
