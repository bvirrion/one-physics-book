\chapter{الصوت من حولنا}\label{ch:g3:sound-around-us}

دندن أغنيتك المفضّلة --- وبينما تدندن ضع إصبعين برفق على مقدّمة
حنجرتك. أتحسّ بذلك الارتعاش؟ لقد لمست توًّا صوتًا في مهده. فكل
\omterm{def:g3:sound-around-us:sound}{صوت}، وكل طبل، وكل باب صرّار، يصنع صوته بالطريقة نفسها تمامًا، وهذا
الفصل يخرج باحثًا عن الحيلة.

\section{أين تُولد الأصوات}

\begin{definition}[الاهتزاز]\label{def:g3:sound-around-us:vibration}
\emph{الاهتزاز}\index{اهتزاز} ارتعاش سريع جدًّا --- حركة ذهاب
وإياب تتكرّر مرات كثيرة في كل ثانية، وكثيرًا ما تكون أسرع من أن
تتابعها العين. فوتر القيثارة المنقور يهتزّ: لا ترى إلا لطخة
مضطربة، لكن طرف إصبعك يحسّ بالطنين.
\end{definition}

\begin{definition}[الصوت]\label{def:g3:sound-around-us:sound}
\emph{الصوت}\index{صوت} ما نسمعه حين يهتزّ شيء وينتقل ارتعاشه
عبر \omterm{def:g2:air-around-us:air}{الهواء} إلى آذاننا. والشيء المهتزّ هو \emph{منبع} الصوت ---
وتر، أو جلد طبل، أو جرس، أو حنجرتك أنت.
\end{definition}

\begin{method}[كيف تجعل الأرزّ يرقص]\label{met:g3:sound-around-us:rice}
لكي \emph{ترى} صوتًا وهو يُولد:
\begin{enumerate}
  \item انثر بضع حبّات أرزّ على جلد طبل (أو على غطاء علبة
        بلاستيكي، أو على جلد بالون ممدود)؛
  \item واطرق الجلد قرب الحبّات --- دمدمة: فتقفز الحبّات!
  \item والآن \emph{اصرخ} على الجلد من قرب شديد فحسب: ترتعش
        الحبّات من جديد، من غير أن يلمسها شيء.
\end{enumerate}
فالجلد يهتزّ كلما أصدر صوتًا --- بل حتى \omterm{def:g3:sound-around-us:sound}{الصوت} القادم من فمك
يجعله يرتعش.
\end{method}

\begin{proposition}[لا صوت بلا اهتزاز]\label{prop:g3:sound-around-us:novib}
كل \omterm{def:g3:sound-around-us:sound}{صوت} يأتي من \omterm{def:g3:sound-around-us:vibration}{اهتزاز} --- وحين يتوقّف \omterm{def:g3:sound-around-us:vibration}{الاهتزاز} يتوقّف \omterm{def:g3:sound-around-us:sound}{الصوت} معه
في الحال. اضرب صنجًا ودعه يرنّ؛ ثم أمسكه بيديك إمساكًا محكمًا:
صمت، في اللحظة التي توقف فيها قبضتُك الارتعاش.
\end{proposition}

\begin{example}[منابع الصوت في كل مكان]\label{ex:g3:sound-around-us:sources}
اصطد \omterm{def:g3:sound-around-us:vibration}{الاهتزاز} وراء كل \omterm{def:g3:sound-around-us:sound}{صوت}: القيثارة --- أوتار منقورة ترتعش؛
والطبل --- جلد مضروب يرتعش؛ وصوتك --- طيّتان صغيرتان في حنجرتك
ترتعشان بينما يندفع \omterm{def:g2:air-around-us:air}{هواء} رئتيك بجوارهما (وذاك هو الطنين الذي
أحسّت به أصابعك)؛ والذبابة الطنّانة --- جناحاها يرتعشان مئات
المرات في الثانية؛ والباب المصفوق --- الباب كله وإطاره يرتجفان
لحظة.
\end{example}

\section{عالٍ وخافت، حادّ وغليظ}

\begin{example}[العلوّ ارتعاش واسع]\label{ex:g3:sound-around-us:loud}
انقر شريطًا مطّاطيًّا ممدودًا برفق: لطخة صغيرة، ورنّة خافتة.
وانقره بشدّة: لطخة عريضة، ورنّة عالية. فكلما اشتدّ ارتعاش المنبع
--- أي اتّسع ذهابه وإيابه --- كان \omterm{def:g3:sound-around-us:sound}{الصوت} \emph{أعلى}. وراقب
حبّات الأرزّ أيضًا: فالطرقة الرفيقة ترعشها، والدمدمة الجبّارة
تقذفها في \omterm{def:g2:air-around-us:air}{الهواء}.
\end{example}

\begin{example}[حادّ وغليظ]\label{ex:g3:sound-around-us:pitch}
مُدّ شريطًا مطّاطيًّا على علبة مفتوحة وانقره؛ ثم اقرصه من الوسط
وانقر نصفه القصير: تصير الرنّة \emph{أحدّ}، مثل زقزقة عصفور.
فالأشياء القصيرة المشدودة الرفيعة ترتعش أسرع فتغنّي حادًّا؛
والطويلة المرتخية الغليظة ترتعش أبطأ فتغنّي غليظًا. ولهذا تزقزق
الناي الصغيرة وتزمجر التوبا الكبيرة --- ولهذا يغنّي أرفع أوتار
القيثارة أحدّها.
\end{example}

\begin{remark}[احمِ الطبل الصغير في أذنك]\label{rem:g3:sound-around-us:ears}
في عمق أذنك جلد دقيق --- أرقّ من الورق --- يرتعش مع كل \omterm{def:g3:sound-around-us:sound}{صوت}
يصله، تمامًا مثل طبل حبّات الأرزّ. والأصوات الشديدة العلوّ تهزّه
هزًّا عنيفًا: فالموسيقى العالية جدًّا والآلات والألعاب النارية
تستطيع أن تتلفه إلى الأبد، وهو لا ينبت من جديد. فغطِّ أذنيك قرب
الضجيج العالي، ولا تصرخ في أذن أحد أبدًا --- فليس للآذان
جفون.
\end{remark}

\section{من المنبع إلى الأذن}

\begin{example}[رحلة تصفيقة]\label{ex:g3:sound-around-us:journey}
يصفّق صديق في الجهة الأخرى من الساحة. فتجعل التصفيقة \omterm{def:g2:air-around-us:air}{الهواء}
يرتعش؛ وينتشر الارتعاش في \omterm{def:g2:air-around-us:air}{الهواء} في كل اتجاه، مثل الحلقات حول
حصاة أُلقيت في بركة؛ ويصل بعضه إلى أذنك فيجعل الجلد الصغير في
أذنك يرتعش --- فتسمع التصفيقة. منبع، فرحلة، فأذن: وكل \omterm{def:g3:sound-around-us:sound}{صوت} سمعته
في حياتك قطع هذه الرحلة. فكم سرعة الرحلة، وفي أيّ شيء تستطيع أن
تنتقل؟ ذانك بالضبط سؤالا فصل \omterm{def:g3:sound-around-us:sound}{الصوت} في السنة القادمة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:1}
ما \omterm{def:g3:sound-around-us:vibration}{الاهتزاز}؟ وسمِّ الجزء المهتزّ في: قيثارة؛ \ طبل؛
\ صوتك؛ \ ذبابة طنّانة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:2}
في تجربة الأرزّ، ماذا تثبت الحبّات القافزة عن جلد طبل مصوّت؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:3}
لماذا يُسكِت الإمساكُ صنجًا رنّانًا في الحال؟ وأيّ قانون من
قوانين الفصل يعمل هنا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:4}
لكي تصير رنّة الشريط المطّاطي \emph{أعلى}، ما الذي تغيّره في
نقرتك؟ وكيف تبدو لطخة الشريط حينئذ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:5}
أيّهما يغنّي أحدّ: الشريط المقروص القصير أم الشريط كله؟ \ الناي
الصغيرة أم التوبا الكبيرة؟ \ أرفع أوتار القيثارة أم أغلظها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:6}
دندن وأصابعك على حنجرتك، ثم توقّف عن الدندنة. بمَ تحسّ أصابعك،
ومتى يتوقّف الإحساس بالضبط؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:7}
احكِ رحلة تصفيقة من يدَي صديقك إلى أذنك، في ثلاث خطوات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:8}
لماذا يجب إبعاد الأصوات الشديدة العلوّ عن الآذان؟ وأيّ شيء دقيق
داخل الأذن في خطر، وبمَ يشبه ما في هذا الفصل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:9}
مشط صندوق الموسيقى فيه أسنان مختلفة الأطوال. فأيّ الأسنان تعزف
النغمات الحادّة وأيّها يعزف الغليظة؟ أجب بقاعدة
\cref{ex:g3:sound-around-us:pitch}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sound-around-us:10}
صُبّ ماءً في قنّينة زجاجية وانفخ فوق فوهتها: صفير! أضِف ماءً
أكثر وانفخ من جديد: يصير الصفير أحدّ. فما الذي يرتعش ليصنع
الصفير --- الزجاج أم الماء أم \omterm{def:g2:air-around-us:air}{الهواء} الذي في الداخل؟ استعمل
قاعدة ``القصير يغنّي حادًّا'' للدفاع عن جوابك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sound-around-us:11}
تقول مايا: ``حين يكون التلفاز صامتًا تظلّ الشاشة تتحرّك ---
فالحركة وحدها إذن تصنع صوتًا.'' استعمل
\cref{def:g3:sound-around-us:sound} وتجربة الأرزّ لتشرح أيّ نوع
من الحركة يصنع صوتًا، ولماذا لا تصنع الصور على الشاشة صوتًا.
\end{exercise}
