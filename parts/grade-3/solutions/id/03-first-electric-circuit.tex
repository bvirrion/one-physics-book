\section*{Bab \ref{ch:g3:first-electric-circuit} --- Rangkaian Listrik Pertama}

\begin{solution}{exo:g3:first-electric-circuit:1}
\omterm{def:g3:first-electric-circuit:battery}{Kutub} \omterm{def:g3:first-electric-circuit:battery}{baterai}: $+$ (plus) dan $-$ (minus). Tempat sentuh \omterm{def:g3:first-electric-circuit:bulb}{bohlam}: ujung
di bagian bawah kakinya, dan sisi logam kakinya.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:2}
Tertutup: satu lingkar tanpa putus dari \omterm{def:g3:first-electric-circuit:battery}{kutub} ke \omterm{def:g3:first-electric-circuit:battery}{kutub}. Terbuka:
lingkar yang punya celah di suatu tempat. \omterm{def:g3:first-electric-circuit:bulb}{Bohlam} menyala hanya pada
yang tertutup.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:3}
Perjalanan pulangnya. Satu kawat hanyalah separuh lingkar: listriknya
harus keluar dari satu \omterm{def:g3:first-electric-circuit:battery}{kutub}, melewati \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, dan \emph{kembali ke
\omterm{def:g3:first-electric-circuit:battery}{kutub} yang lain} --- Lea memerlukan kawat kedua dari tempat sentuh
samping \omterm{def:g3:first-electric-circuit:bulb}{bohlam} ke \omterm{def:g3:first-electric-circuit:battery}{kutub} $-$.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:4}
\omterm{ex:g3:first-electric-circuit:switch}{Sakelar} adalah pintu di dalam lingkar: kalau tertutup ia melengkapi
rangkaiannya; kalau terbuka ia membuat celah. Di dalam \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang
tertutup, sebuah jembatan logam kecil menyatukan kedua kawatnya.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:5}
Sebuah lingkar dari garis lurus yang menghubungkan: dua garis yang
tidak sama panjang (\omterm{def:g3:first-electric-circuit:battery}{baterainya}, garis panjangnya adalah $+$), lambang
jembatan angkat yang terbaring tertutup (\omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}), dan lingkaran
bersilang (lampunya).
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:6}
\omterm{def:g3:first-electric-circuit:battery}{Baterai} yang sudah habis; sambungan yang longgar (kawat atau \omterm{def:g3:first-electric-circuit:bulb}{bohlam}
tidak terpasang erat, pegasnya bengkok); \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang rusak --- benang
berpijarnya putus, sehingga ada celah di dalam \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} sendiri.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:7}
Tombolnya \emph{adalah} \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}. Menekannya merebahkan jembatan
logam dan menutup lingkarnya, sehingga belnya berbunyi; melepaskannya
mengangkat jembatan itu, celahnya terbuka, dan bunyinya berhenti.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:8}
Dorongan \omterm{def:g3:first-electric-circuit:battery}{baterai} mungil dan aman; dorongan stopkontak ratusan kali
lebih perkasa dan dapat melukai atau membunuh. Stopkontak bukan tempat
percobaan --- tidak pernah.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:9}
Ya, \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} menyala. Telusuri lingkarnya: \omterm{def:g3:first-electric-circuit:battery}{kutub} $+$, masuk ke
ujungnya, melewati benang berpijarnya, lalu keluar lewat tempat
sentuh sampingnya, menyusuri kawatnya, dan kembali ke $-$. Satu perjalanan penuh
yang menyinggahi kedua \omterm{def:g3:first-electric-circuit:battery}{kutub} dan kedua tempat sentuhnya.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:10}
Lingkarnya tidak pernah menyinggahi \omterm{def:g3:first-electric-circuit:battery}{kutub} yang kedua. Sebuah rangkaian
harus berjalan dari satu \omterm{def:g3:first-electric-circuit:battery}{kutub} \emph{melewati \omterm{def:g3:first-electric-circuit:bulb}{bohlam} ke \omterm{def:g3:first-electric-circuit:battery}{kutub} yang
lain}: keluar dari $-$ lalu kembali ke $-$ adalah perjalanan yang
tidak ke mana-mana --- dorongan \omterm{def:g3:first-electric-circuit:battery}{baterai} memerlukan kedua ujungnya yang
berbeda ikut bermain.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:11}
Semua \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} pasti duduk pada satu lingkar tunggal, satu demi satu:
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang hilang meninggalkan celah, dan satu celah menggelapkan
seluruh untaian --- itulah hukum lingkar. (Tahun depan susunan
satu-demi-satu ini mendapat namanya: rangkaian seri.)
\end{solution}
