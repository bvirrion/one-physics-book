\section*{Bab \ref{ch:g3:solids-liquids-gases} --- Zat Padat, Zat Cair, Gas}

\begin{solution}{exo:g3:solids-liquids-gases:1}
Padat: koinnya, manik-manik kayunya. Cair: jus jeruk, minyak. \omterm{def:g3:solids-liquids-gases:gas}{Gas}:
\omterm{def:g2:air-around-us:air}{udara} di dalam ban, uap tak terlihat di celah bening.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:2}
\omterm{def:g3:solids-liquids-gases:solid}{Zat padat} mempertahankan bentuknya sendiri; \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} tidak. Ketika
dituang dari gelas ke mangkuk, \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} mempertahankan
\emph{jumlahnya}.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:3}
Jumlahnya sama saja --- hanya bentuknya yang berubah, dari bentuk
botol menjadi bentuk mangkuk yang lebar.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:4}
Pada kedua gambar, permukaan airnya terbaring rata dan mendatar,
sejajar dengan lantai --- pada akuarium yang dimiringkan, permukaannya
tidak ikut miring bersama kacanya; airnya berkumpul di sisi yang
rendah.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:5}
\omterm{def:g3:solids-liquids-gases:liquid}{Zat cair} di dalam jendela kecilnya selalu terbaring mendatar, sehingga
gelembungnya menunjukkan di mana ``atas'' berada: gelembung di antara
kedua tanda berarti raknya sejajar dengan \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} yang mendatar itu.
Waterpas memakai hukum bahwa \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} yang diam terbaring mendatar.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:6}
Keluarga \omterm{def:g3:solids-liquids-gases:gas}{gas}. Bau panekuk adalah \omterm{def:g3:solids-liquids-gases:gas}{gas}, dan \omterm{def:g3:solids-liquids-gases:gas}{gas} menyebar untuk mengisi
seluruh ruang yang diberikan kepadanya --- dapur, tangga, kamar tidur.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:7}
Pasir adalah kerumunan butiran padat yang mungil. Kerumunannya
mengalir dan mengambil bentuk timbunannya, tetapi setiap butirnya
mempertahankan bentuknya sendiri dengan sempurna --- jadi setiap
butirnya \omterm{def:g3:solids-liquids-gases:solid}{zat padat} sejati.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:8}
Madu mengalir (pelan), mengambil bentuk wadahnya, dan mempertahankan
jumlahnya: madu menjawab pertanyaannya seperti \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} --- \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}
yang malas.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:9}
Mempertahankan bentuknya: padat (sebuah sendok). Mengambil bentuk
wadahnya tetapi mempertahankan jumlahnya: cair (susu). Mengambil
seluruh ruangnya: \omterm{def:g3:solids-liquids-gases:gas}{gas} (\omterm{def:g2:air-around-us:air}{udara} di dapur).
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:10}
Ketika es \omterm{def:g2:ice-water-steam:meltfreeze}{mencair}, zatnya berhenti mempertahankan \emph{bentuknya}
sendiri (tetapi masih mempertahankan jumlahnya). Ketika air \omterm{def:g2:ice-water-steam:evapcond}{menguap},
ia bahkan berhenti mempertahankan \emph{tempat diamnya}: sebagai \omterm{def:g3:solids-liquids-gases:gas}{gas}
ia menyebar ke seluruh ruang yang ada.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:11}
Kulit karetnyalah yang memberikan bentuk: kulit itu sangkar padat yang
teregang, dan \omterm{def:g2:air-around-us:air}{udara} yang tertekan mengisi setiap sudut bentuk apa pun
yang diambil sangkarnya. Letuskan balonnya, dan \omterm{def:g2:air-around-us:air}{udaranya} seketika
menyebar ke seluruh ruangan --- menunjukkan bahwa ia memang tidak
pernah punya bentuk sendiri.
\end{solution}
