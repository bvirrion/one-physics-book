\section*{Chapter \ref{ch:g3:solids-liquids-gases} --- Solids, Liquids, Gases}

\begin{solution}{exo:g3:solids-liquids-gases:1}
\omterm{def:g3:solids-liquids-gases:solid}{Solids}: the coin, the wooden bead. \omterm{def:g3:solids-liquids-gases:liquid}{Liquids}: orange juice, oil.
Gases: the \omterm{def:g2:air-around-us:air}{air} in the tire, the invisible steam in the clear gap.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:2}
A \omterm{def:g3:solids-liquids-gases:solid}{solid} keeps its own shape; a \omterm{def:g3:solids-liquids-gases:liquid}{liquid} does not. Poured from glass to
bowl, a \omterm{def:g3:solids-liquids-gases:liquid}{liquid} keeps its \emph{amount}.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:3}
The same amount --- only the shape changed, from bottle-shape to
wide-bowl-shape.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:4}
In both drawings the water surface lies flat and level, parallel to
the floor --- in the tilted tank the surface does not tilt with the
glass; the water gathers at the low side.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:5}
The \omterm{def:g3:solids-liquids-gases:liquid}{liquid} in its little window always lies level, so the bubble
shows where ``up'' is: bubble between the marks means the shelf
matches the level \omterm{def:g3:solids-liquids-gases:liquid}{liquid}. It uses the law that a resting \omterm{def:g3:solids-liquids-gases:liquid}{liquid} lies
level.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:6}
The \omterm{def:g3:solids-liquids-gases:gas}{gas} family. The pancake smell is a \omterm{def:g3:solids-liquids-gases:gas}{gas}, and a \omterm{def:g3:solids-liquids-gases:gas}{gas} spreads to
fill all the room it is given --- kitchen, staircase, bedroom.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:7}
Sand is a crowd of tiny \omterm{def:g3:solids-liquids-gases:solid}{solid} grains. The crowd flows and takes the
shape of its pile, but each grain keeps its own shape perfectly ---
so each grain is a true \omterm{def:g3:solids-liquids-gases:solid}{solid}.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:8}
It flows (slowly), takes its container's shape, and keeps its
amount: honey answers the questions like a \omterm{def:g3:solids-liquids-gases:liquid}{liquid} --- a lazy one.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:9}
Keeps its shape: \omterm{def:g3:solids-liquids-gases:solid}{solid} (a spoon). Takes the container's shape but
keeps its amount: \omterm{def:g3:solids-liquids-gases:liquid}{liquid} (milk). Takes all the room: \omterm{def:g3:solids-liquids-gases:gas}{gas} (the \omterm{def:g2:air-around-us:air}{air} in
the kitchen).
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:10}
When ice \omterm{def:g2:ice-water-steam:meltfreeze}{melts}, the stuff stops keeping its own \emph{shape} (but
still keeps its amount). When water \omterm{def:g2:ice-water-steam:evapcond}{evaporates}, it stops keeping
even its \emph{resting place}: as a \omterm{def:g3:solids-liquids-gases:gas}{gas} it spreads into all the room
there is.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:11}
The rubber skin gives the shape: it is a stretched \omterm{def:g3:solids-liquids-gases:solid}{solid} cage, and
the squeezed \omterm{def:g2:air-around-us:air}{air} fills every corner of whatever shape the cage
takes. Pop the balloon, and the \omterm{def:g2:air-around-us:air}{air} instantly spreads into the whole
room --- showing it never had a shape of its own at all.
\end{solution}
