\section*{Chapitre \ref{ch:g3:first-electric-circuit} --- Un premier circuit électrique}

\begin{solution}{exo:g3:first-electric-circuit:1}
Les \omterm{def:g3:first-electric-circuit:battery}{bornes} de la \omterm{def:g3:first-electric-circuit:battery}{pile}~: $+$ (plus) et $-$ (moins). Les contacts de la
\omterm{def:g3:first-electric-circuit:bulb}{lampe}~: le plot, au bas du culot, et le côté métallique du culot.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:2}
Fermé~: une boucle ininterrompue d'une \omterm{def:g3:first-electric-circuit:battery}{borne} à l'autre. Ouvert~: une
boucle avec une coupure quelque part. La \omterm{def:g3:first-electric-circuit:bulb}{lampe} ne s'allume que dans
le \omterm{def:g3:first-electric-circuit:circuit}{circuit fermé}.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:3}
Le trajet de retour. Un seul fil ne fait que la moitié d'une boucle~:
l'électricité doit sortir d'une \omterm{def:g3:first-electric-circuit:battery}{borne}, traverser la \omterm{def:g3:first-electric-circuit:bulb}{lampe} et
\emph{revenir à l'autre \omterm{def:g3:first-electric-circuit:battery}{borne}} --- il manque à Léa un second fil
allant du contact latéral de la \omterm{def:g3:first-electric-circuit:bulb}{lampe} à la \omterm{def:g3:first-electric-circuit:battery}{borne} $-$.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:4}
L'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} est une porte dans la boucle~: fermé, il complète le
circuit~; ouvert, il crée une coupure. À l'intérieur d'un
\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} fermé, un petit pont de métal relie les deux fils.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:5}
Une boucle de traits droits reliant~: deux barres inégales (la \omterm{def:g3:first-electric-circuit:battery}{pile},
la barre longue étant le $+$), le symbole du pont-levis posé à plat
(l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur}), et un cercle barré d'une croix (la \omterm{def:g3:first-electric-circuit:bulb}{lampe}).
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:6}
Une \omterm{def:g3:first-electric-circuit:battery}{pile} usée~; une connexion mal serrée (fil ou \omterm{def:g3:first-electric-circuit:bulb}{lampe} mal vissés,
ressort tordu)~; une \omterm{def:g3:first-electric-circuit:bulb}{lampe} grillée --- son filament brisé fait une
coupure à l'intérieur même de la \omterm{def:g3:first-electric-circuit:bulb}{lampe}.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:7}
Le bouton \emph{est} l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur}. En appuyant, on pose le pont de
métal et on ferme la boucle~: la sonnerie retentit~; en relâchant, on
soulève le pont, la coupure s'ouvre et la sonnerie s'arrête.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:8}
La poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile} est minuscule et sans danger~; celle de la
prise est des centaines de fois plus puissante et peut blesser ou
tuer. Les prises ne sont pas faites pour les expériences --- jamais.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:9}
Oui, elle s'allume. Suis la boucle~: \omterm{def:g3:first-electric-circuit:battery}{borne} $+$, entrée par le plot,
traversée du filament, sortie par le contact latéral, parcours du
fil, retour au $-$. Un aller-retour complet qui passe par les deux
\omterm{def:g3:first-electric-circuit:battery}{bornes} et les deux contacts.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:10}
Sa boucle ne passe jamais par la seconde \omterm{def:g3:first-electric-circuit:battery}{borne}. Un circuit doit aller
d'une \omterm{def:g3:first-electric-circuit:battery}{borne} \emph{à l'autre en traversant la \omterm{def:g3:first-electric-circuit:bulb}{lampe}}~: sortir du $-$
pour revenir au $-$ est un aller-retour vers nulle part --- la
poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile} a besoin de ses deux extrémités différentes dans
la partie.
\end{solution}

\begin{solution}{exo:g3:first-electric-circuit:11}
Les \omterm{def:g3:first-electric-circuit:bulb}{lampes} sont forcément toutes sur une seule et même boucle, les
unes après les autres~: la \omterm{def:g3:first-electric-circuit:bulb}{lampe} manquante laisse une coupure, et une
seule coupure éteint toute la guirlande --- c'est la loi de la
boucle. (L'an prochain, cette disposition les unes après les autres
recevra son nom~: un circuit en série.)
\end{solution}
