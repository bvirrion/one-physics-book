\section*{Chapitre \ref{ch:g3:solids-liquids-gases} --- Solides, liquides, gaz}

\begin{solution}{exo:g3:solids-liquids-gases:1}
\omterm{def:g3:solids-liquids-gases:solid}{Solides}~: la pièce de monnaie, la perle de bois. \omterm{def:g3:solids-liquids-gases:liquid}{Liquides}~: le jus
d'orange, l'huile. \omterm{def:g3:solids-liquids-gases:gas}{Gaz}~: l'\omterm{def:g2:air-around-us:air}{air} du pneu, la vapeur d'eau invisible de
l'espace transparent.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:2}
Un \omterm{def:g3:solids-liquids-gases:solid}{solide} garde sa propre forme~; un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} non. Versé d'un verre
dans un bol, un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} garde sa \emph{quantité}.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:3}
La même quantité --- seule la forme a changé, de la forme de la
bouteille à celle du large saladier.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:4}
Dans les deux dessins, la surface de l'eau est plane et horizontale,
parallèle au sol --- dans l'aquarium basculé, la surface ne penche
pas avec le verre~; l'eau se rassemble du côté bas.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:5}
Le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} de sa petite fenêtre reste toujours horizontal, et la
bulle montre donc où est le haut~: la bulle entre les repères
signifie que l'étagère suit le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} horizontal. Il utilise la loi
selon laquelle un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} au repos a une surface horizontale.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:6}
La famille des \omterm{def:g3:solids-liquids-gases:gas}{gaz}. L'odeur des crêpes est un \omterm{def:g3:solids-liquids-gases:gas}{gaz}, et un \omterm{def:g3:solids-liquids-gases:gas}{gaz} s'étale
pour occuper toute la place qu'on lui donne --- cuisine, escalier,
chambre.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:7}
Le sable est une foule de minuscules grains \omterm{def:g3:solids-liquids-gases:solid}{solides}. La foule
s'écoule et prend la forme de son tas, mais chaque grain garde
parfaitement sa propre forme --- chaque grain est donc un vrai
\omterm{def:g3:solids-liquids-gases:solid}{solide}.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:8}
Il \omterm{def:g1:floating-and-sinking:words}{coule} (lentement), prend la forme de son récipient et garde sa
quantité~: le miel répond aux questions comme un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} --- un
\omterm{def:g3:solids-liquids-gases:liquid}{liquide} paresseux.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:9}
Garde sa forme~: un \omterm{def:g3:solids-liquids-gases:solid}{solide} (une cuillère). Prend la forme du
récipient mais garde sa quantité~: un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} (le lait). Occupe toute
la place~: un \omterm{def:g3:solids-liquids-gases:gas}{gaz} (l'\omterm{def:g2:air-around-us:air}{air} de la cuisine).
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:10}
Quand la glace fond, la matière cesse de garder sa \emph{forme}
propre (mais garde encore sa quantité). Quand l'eau \omterm{def:g2:ice-water-steam:evapcond}{s'évapore}, elle
cesse même de garder sa \emph{place}~: à l'état de \omterm{def:g3:solids-liquids-gases:gas}{gaz}, elle se
répand dans tout l'espace disponible.
\end{solution}

\begin{solution}{exo:g3:solids-liquids-gases:11}
C'est la peau de caoutchouc qui donne la forme~: c'est une cage
\omterm{def:g3:solids-liquids-gases:solid}{solide} tendue, et l'\omterm{def:g2:air-around-us:air}{air} comprimé en remplit le moindre recoin, quelle
que soit la forme de la cage. Fais éclater le ballon, et l'\omterm{def:g2:air-around-us:air}{air} se
répand aussitôt dans toute la pièce --- montrant qu'il n'a jamais eu
de forme propre.
\end{solution}
