\section*{Chapitre \ref{ch:g3:measuring-mass} --- Mesurer la masse}

\begin{solution}{exo:g3:measuring-mass:1}
Quelle quantité de matière compose l'objet. \omterm{def:g2:measuring-length:measure}{Unités}~: le \omterm{def:g3:measuring-mass:units}{gramme} et le
\omterm{def:g3:measuring-mass:units}{kilogramme}, avec $\qty{1}{kg} = \qty{1000}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:2}
Trombone~: \qty{1}{g}. Pomme~: \qty{150}{g}. Brique de lait~:
\qty{1}{kg}. Chat~: \qty{4}{kg}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:3}
$500 + 200 + 200 = 900$~: la \omterm{def:g3:measuring-mass:mass}{masse} du potiron est \qty{900}{g}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:4}
Si la \omterm{def:g3:levers-and-scales:scale}{balance} à vide penche déjà, chaque pesée ment en faveur d'un
côté --- comme le bras long du marchand tricheur. Rester horizontale
à vide est la preuve d'un départ honnête.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:5}
Le ballon de plage est gros mais surtout rempli d'\omterm{def:g2:air-around-us:air}{air} --- peu de
matière, petite \omterm{def:g3:measuring-mass:mass}{masse}. La boîte est petite mais bourrée de
matière --- plus de \omterm{def:g3:measuring-mass:mass}{masse}, et elle fait donc descendre son plateau.
La \omterm{def:g3:measuring-mass:mass}{masse} compte la matière, pas la taille.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:6}
$\qty{2}{kg} = \qty{2000}{g}$~; un \omterm{def:g3:measuring-mass:units}{demi-kilogramme} fait
\qty{500}{g}~; $\qty{1200}{g}$ est plus que
$\qty{1}{kg} = \qty{1000}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:7}
Toujours \qty{100}{g}~: casser change la forme, pas la matière ---
c'est la loi «~même matière, même \omterm{def:g3:measuring-mass:mass}{masse}~», tous les morceaux étant
ensemble sur le plateau.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:8}
La \omterm{def:g3:levers-and-scales:scale}{balance} de cuisine compare grâce à un ressort comprimé placé à
l'intérieur~; la \omterm{def:g3:levers-and-scales:scale}{balance} à deux plateaux compare avec des \omterm{def:g3:measuring-mass:mass}{masses}
marquées, sur des bras égaux --- la règle du \omterm{def:g3:levers-and-scales:lever}{levier} du chapitre sur
les \omterm{def:g3:levers-and-scales:lever}{leviers}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:9}
Première façon~: $500 + 200 + 100 = \qty{800}{g}$ de \omterm{def:g3:measuring-mass:mass}{masses} en face
de la farine. Deuxième façon~: poser la \omterm{def:g3:measuring-mass:mass}{masse} de \qty{100}{g}
\emph{avec la farine} et $500 + 200 + 200 = \qty{900}{g}$ en face~;
des plateaux au même niveau signifient alors que farine
$+ \qty{100}{g}$ vaut \qty{900}{g}, donc la farine pèse
$900 - 100 = \qty{800}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:10}
La loi n'est pas en défaut~: aucune matière n'a disparu --- elle a
seulement été séparée. Les \qty{40}{g} manquants sont la peau. Une
pesée de plus le prouve~: la peau seule doit indiquer \qty{40}{g}, et
$160 + 40 = \qty{200}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:11}
Ils ont la \emph{même} \omterm{def:g3:measuring-mass:mass}{masse} --- un \omterm{def:g3:measuring-mass:units}{kilogramme} chacun. Les plumes
prennent bien plus de place. Les gens répondent mal parce que les
mains et les yeux jugent le fer «~matière lourde~»~; mais la question
a fixé la \omterm{def:g3:measuring-mass:mass}{masse} à un \omterm{def:g3:measuring-mass:units}{kilogramme} de part et d'autre --- seule la place
occupée diffère.
\end{solution}
