\section*{Chapter \ref{ch:g3:measuring-mass} --- Measuring Mass}

\begin{solution}{exo:g3:measuring-mass:1}
How much stuff a thing is made of. \omterm{def:g2:measuring-length:measure}{Units}: the \omterm{def:g3:measuring-mass:units}{gram} and the
\omterm{def:g3:measuring-mass:units}{kilogram}, with $\qty{1}{kg} = \qty{1000}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:2}
Paperclip: \qty{1}{g}. Apple: \qty{150}{g}. Milk carton:
\qty{1}{kg}. Cat: \qty{4}{kg}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:3}
$500 + 200 + 200 = 900$: the pumpkin's \omterm{def:g3:measuring-mass:mass}{mass} is \qty{900}{g}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:4}
If the empty scale already tips, every weighing lies in favor of
one side --- like the cheating merchant's long arm. Level when
empty is the proof of a fair start.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:5}
The beach ball is big but mostly \omterm{def:g2:air-around-us:air}{air} --- little stuff, small \omterm{def:g3:measuring-mass:mass}{mass}.
The can is small but packed with stuff --- more \omterm{def:g3:measuring-mass:mass}{mass}, so it presses
its pan down. \omterm{def:g3:measuring-mass:mass}{Mass} counts stuff, not size.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:6}
$\qty{2}{kg} = \qty{2000}{g}$; half a \omterm{def:g3:measuring-mass:units}{kilogram} is \qty{500}{g};
$\qty{1200}{g}$ is more than $\qty{1}{kg} = \qty{1000}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:7}
Still \qty{100}{g}: breaking changes the shape, not the stuff ---
the law ``same stuff, same \omterm{def:g3:measuring-mass:mass}{mass}'', with all the pieces on the pan
together.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:8}
The kitchen scale compares with a squeezed spring inside; the
two-pan balance compares with blocks across equal arms --- the
\omterm{def:g3:levers-and-scales:lever}{lever} rule from the \omterm{def:g3:levers-and-scales:lever}{levers} chapter.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:9}
First way: $500 + 200 + 100 = \qty{800}{g}$ of blocks opposite the
flour. Second way: put the \qty{100}{g} block \emph{with the flour}
and $500 + 200 + 200 = \qty{900}{g}$ opposite; level pans then mean
flour $+ \qty{100}{g}$ equals \qty{900}{g}, so the flour is
$900 - 100 = \qty{800}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:10}
The law has not failed: no stuff vanished --- it was only
separated. The missing \qty{40}{g} are the peel. One more weighing
proves it: the peel alone should read \qty{40}{g}, and $160 + 40 =
\qty{200}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:11}
They have the \emph{same} \omterm{def:g3:measuring-mass:mass}{mass} --- one \omterm{def:g3:measuring-mass:units}{kilogram} each. The feathers
take far more room. People answer wrongly because hands and eyes
judge iron ``heavy stuff''; but the question fixed the \omterm{def:g3:measuring-mass:mass}{mass} at one
\omterm{def:g3:measuring-mass:units}{kilogram} each --- only the room differs.
\end{solution}
