\section*{Hoofdstuk \ref{ch:g3:temperature-thermometers} --- Temperatuur en thermometers}

\begin{solution}{exo:g3:temperature-thermometers:1}
De \omterm{def:g3:temperature-thermometers:temperature}{thermometer} geeft hetzelfde getal bij dezelfde warmte, wie hem ook
afleest en wanneer ook. Handen werden voor de gek gehouden door de
drie kommen (hetzelfde lauwe water voelde tegelijk warm en koel) en
doordat metaal in dezelfde \omterm{def:g1:light-and-shadows:shadow}{schaduw} kouder aanvoelt dan hout.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:2}
Water \omterm{def:g2:ice-water-steam:meltfreeze}{bevriest} bij \qty{0}{\celsius} en kookt bij
\qty{100}{\celsius} --- allebei in \omterm{def:g3:temperature-thermometers:temperature}{graden Celsius}.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:3}
Verwarmd zet de opgesloten \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} uit en klimt de kolom; afgekoeld
krimpt ze en zakt de kolom. Het buisje is zo dun dat zelfs een
piepkleine uitzetting de kolom zichtbaar omhoog duwt.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:4}
Te vroeg aflezen (voordat de kolom is gestopt met bewegen), en schuin
van opzij aflezen in plaats van met je oog op gelijke hoogte met de
bovenkant van de kolom.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:5}
Kokend water: \qty{100}{\celsius}. Behaaglijke woonkamer:
\qty{20}{\celsius}. Koelkast: \qty{4}{\celsius}. Je lichaam:
\qty{37}{\celsius}.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:6}
De pan is warmer, met $100 - 35 = 65$ graden --- veel te heet om aan
te raken: hij staat op kooktemperatuur.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:7}
Min drie \omterm{def:g3:temperature-thermometers:temperature}{graden Celsius}, geschreven \qty{-3}{\celsius}. Dat is
\emph{warmer} dan \qty{-8}{\celsius}: acht streepjes onder nul zit
dieper in de vorst dan drie.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:8}
$14 - 6 = 8$: die ochtend werd het $8$ graden warmer.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:9}
De nacht is onder \qty{0}{\celsius} gezakt --- de vriesmijlpaal van
water. Bevroren plassen zijn een \omterm{def:g3:temperature-thermometers:temperature}{thermometer} gemaakt van weer.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:10}
Nee --- \qty{16}{\celsius} zit aan de koele kant van aangenaam. De
huid van Emma vergelijkt de gang met de vrieskoude tuin die ze net
verlaten heeft, dus voelt hij voor haar behaaglijk. Het gezin moet de
\omterm{def:g3:temperature-thermometers:temperature}{thermometer} geloven: die vergelijkt met de vaste streepjes van zijn
schaal, niet met waar hij zojuist vandaan kwam.
\end{solution}

\begin{solution}{exo:g3:temperature-thermometers:11}
Kokend water blijft op \qty{100}{\celsius}, hoe lang het ook kookt ---
de mijlpaal is een plafond: extra verwarmen maakt meer damp, niet
heter water. Juist daarom kon \qty{100}{\celsius} überhaupt als
betrouwbare mijlpaal worden gekozen.
\end{solution}
