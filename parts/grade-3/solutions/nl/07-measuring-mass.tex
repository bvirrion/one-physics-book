\section*{Hoofdstuk \ref{ch:g3:measuring-mass} --- Massa meten}

\begin{solution}{exo:g3:measuring-mass:1}
Uit hoeveel stof een ding gemaakt is. Eenheden: de \omterm{def:g3:measuring-mass:units}{gram} en de
\omterm{def:g3:measuring-mass:units}{kilogram}, met $\qty{1}{kg} = \qty{1000}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:2}
Paperclip: \qty{1}{g}. Appel: \qty{150}{g}. Pak melk: \qty{1}{kg}.
Kat: \qty{4}{kg}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:3}
$500 + 200 + 200 = 900$: de \omterm{def:g3:measuring-mass:mass}{massa} van de pompoen is \qty{900}{g}.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:4}
Als de lege weegschaal al kantelt, liegt elke weging in het voordeel
van één kant --- net als de lange arm van de bedrieglijke koopman.
Waterpas als ze leeg is, is het bewijs van een eerlijke start.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:5}
De strandbal is groot maar vooral \omterm{def:g2:air-around-us:air}{lucht} --- weinig stof, kleine
\omterm{def:g3:measuring-mass:mass}{massa}. Het blik is klein maar volgepakt met stof --- meer
\omterm{def:g3:measuring-mass:mass}{massa}, dus drukt het zijn schaal omlaag. \omterm{def:g3:measuring-mass:mass}{Massa} telt stof, geen
grootte.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:6}
$\qty{2}{kg} = \qty{2000}{g}$; een halve \omterm{def:g3:measuring-mass:units}{kilogram} is \qty{500}{g};
$\qty{1200}{g}$ is meer dan $\qty{1}{kg} = \qty{1000}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:7}
Nog steeds \qty{100}{g}: breken verandert de vorm, niet de stof --- de
wet ``dezelfde stof, dezelfde \omterm{def:g3:measuring-mass:mass}{massa}'', met alle stukjes samen op de
schaal.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:8}
De keukenweegschaal vergelijkt met een ingedrukte veer binnenin; de
balans met twee schalen vergelijkt met blokjes over gelijke armen ---
de hefboomregel uit het hoofdstuk over hefbomen.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:9}
Eerste manier: $500 + 200 + 100 = \qty{800}{g}$ aan blokjes tegenover
het meel. Tweede manier: leg het blokje van \qty{100}{g} \emph{bij het
meel} en $500 + 200 + 200 = \qty{900}{g}$ ertegenover; waterpasse
schalen betekenen dan dat meel $+ \qty{100}{g}$ gelijk is aan
\qty{900}{g}, dus is het meel $900 - 100 = \qty{800}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:10}
De wet heeft niet gefaald: er is geen stof verdwenen --- ze is alleen
gescheiden. De ontbrekende \qty{40}{g} zijn de schil. Eén extra weging
bewijst het: de schil alleen moet \qty{40}{g} aanwijzen, en
$160 + 40 = \qty{200}{g}$.
\end{solution}

\begin{solution}{exo:g3:measuring-mass:11}
Ze hebben \emph{dezelfde} \omterm{def:g3:measuring-mass:mass}{massa} --- elk één \omterm{def:g3:measuring-mass:units}{kilogram}. De veren nemen
veel meer ruimte in. Mensen antwoorden verkeerd omdat handen en ogen
ijzer als ``zware stof'' beoordelen; maar de vraag legde de
\omterm{def:g3:measuring-mass:mass}{massa} van allebei op één \omterm{def:g3:measuring-mass:units}{kilogram} vast --- alleen de ruimte
verschilt.
\end{solution}
