\section*{Hoofdstuk \ref{ch:g3:levers-and-scales} --- Hefbomen en balansweegschalen}

\begin{solution}{exo:g3:levers-and-scales:1}
Wip: de plank is de \omterm{def:g3:levers-and-scales:lever}{hefboom}, de steun in het midden het
\omterm{def:g3:levers-and-scales:lever}{draaipunt}. Koevoet: de staaf is de \omterm{def:g3:levers-and-scales:lever}{hefboom}, het blok het
\omterm{def:g3:levers-and-scales:lever}{draaipunt}. Schaar: elk blad met zijn handvat is een \omterm{def:g3:levers-and-scales:lever}{hefboom}, en ze
draaien allebei om de schroef in het midden.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:2}
Ver van het \omterm{def:g3:levers-and-scales:lever}{draaipunt}: hoe verder naar buiten de duw, hoe groter zijn
product --- een kleine duw ver weg verslaat een zware rots vlakbij.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:3}
$3 \times 4 = 12$; \ $2 \times 5 = 10$; \ $1 \times 6 = 6$. Geen van
deze producten is gelijk, dus geen twee ervan houden elkaar in
\omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht}.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:4}
Het linkerproduct is $1 \times 8 = 8$. Twee muntjes hebben
$8 = 2 \times 4$ nodig: streepje $4$. Vier muntjes hebben
$8 = 4 \times 2$ nodig: streepje $2$.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:5}
Links: $3 \times 3 = 9$. Rechts: $2 \times 4 = 8$. Het product links
is groter, dus gaat de linkerkant omlaag.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:6}
Bij gelijke armen betekenen waterpasse schalen gelijke gewichten ---
de vergelijking is eerlijk. Een langere linkerarm zou de linkerlading
met een grotere afstand vermenigvuldigen, dus zou de weegschaal bij
elke weging in het voordeel van die kant liegen.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:7}
Het zakje weegt precies evenveel als de $6$ blokjes. Met één blokje
minder is de schaal met het zakje zwaarder, en kantelt de weegschaal
naar het zakje toe.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:8}
Het product van Tom: $2 \times 3 = 6$. Zijn zusje heeft
$1 \times 6 = 6$ nodig: ze moet op streepje $6$ gaan zitten.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:9}
Het \omterm{def:g3:levers-and-scales:lever}{draaipunt} is de plek waar de hamerkop op het hout rust. Jouw hand
duwt aan het verre uiteinde van de steel --- een grote afstand, dus
een groot product --- terwijl de spijker vlak bij het \omterm{def:g3:levers-and-scales:lever}{draaipunt}
tegenstribbelt: de \omterm{def:g3:levers-and-scales:lever}{hefboom} maakt van jouw comfortabele duw de trek van
een reus.
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:10}
Voor \omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht} moet gelden: $4 \times (\text{streepje van opa}) =
1 \times (\text{streepje van Zoe})$. Twee manieren: opa op $1$ met Zoe
op $4$ ($4 = 4$), of opa op $2$ met Zoe op $8$ ($8 = 8$).
\end{solution}

\begin{solution}{exo:g3:levers-and-scales:11}
Minder. Op de lange arm maakt zelfs een lichte lading een groot
product, dus hangen de schalen waterpas terwijl de waar \emph{minder}
weegt dan de eerlijke blokjes op de korte arm. De controle met de lege
weegschaal betrapt hem: met niets in de schalen kantelt zijn
weegschaal al naar de lange arm.
\end{solution}
