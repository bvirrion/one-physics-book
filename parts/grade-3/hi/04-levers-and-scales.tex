\chapter{उत्तोलक और तुला}\label{ch:g3:levers-and-scales}

पिछले साल सी-सॉ पर हमने पाया था कि भार और दूरी, दोनों गिनती में आते
हैं --- और हमने असली संख्याओं वाले एक सटीक नियम का वादा किया था। वह
वादा निभाने का समय आ गया। रास्ते में हम ऐसे पत्थर उठाएँगे जो हमारी
बाँहों के लिए कहीं ज़्यादा भारी हैं, और उस उपकरण से भी मिलेंगे जिसने
हज़ारों वर्षों तक दुनिया का सामान तौला।

\section{उत्तोलक}

\begin{definition}[उत्तोलक और धुरी]\label{def:g3:levers-and-scales:lever}
\emph{उत्तोलक}\index{उत्तोलक} एक कड़ी छड़ है जो किसी टेक-बिंदु पर
टिकी होती है और उसी के चारों ओर झुक सकती है। उस टेक-बिंदु को
\emph{धुरी}\index{धुरी} कहते हैं। सी-सॉ उत्तोलक है; चट्टान के नीचे
लगी सब्बल भी, और कील खींचता हथौड़े का पंजा भी।
\end{definition}

\begin{example}[ताक़त बढ़ाने वाला]\label{ex:g3:levers-and-scales:crowbar}
एक माली लोहे की लंबी छड़ किसी बड़ी चट्टान के नीचे सरकाता है, उसे
चट्टान के पास रखे एक कुंदे पर टिकाता है, और दूर वाले सिरे को नीचे
दबाता है --- और जो चट्टान दस हाथ मिलकर नहीं उठा सके, वह आराम से लुढ़क
जाती है। \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} का राज़ दूरी है: \omterm{def:g3:levers-and-scales:lever}{धुरी} से \emph{दूर} दबाओ, और
तुम्हारा छोटा-सा धक्का \omterm{def:g3:levers-and-scales:lever}{धुरी} के पास खड़े किसी दानव का काम कर देता है।
एक पुरानी कहानी कहती है कि प्राचीन काल के एक महान वैज्ञानिक ने डींग
मारी थी: ``मुझे खड़े होने की जगह और काफ़ी लंबा \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} दे दो, मैं
पृथ्वी हिला दूँगा।''
\end{example}

\section{उत्तोलक का नियम}

\begin{method}[सिक्कों का सी-सॉ]\label{met:g3:levers-and-scales:coins}
नियम की तलाश में एक नन्हा सी-सॉ बनाओ:
\begin{enumerate}
  \item एक रूलर को उसके बीच वाले निशान पर पेंसिल के ऊपर \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} करो;
  \item सवार के तौर पर एक जैसे सिक्के लो और उन्हें पूरी संख्या वाले
        निशानों पर रखो: ``$4$ पर एक सिक्का'' यानी पेंसिल से चार
        निशान दूर;
  \item दोनों ओर सिक्के रखो और देखो कि रूलर कब \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} होता है।
\end{enumerate}
करके देखो: $6$ पर रखा एक सिक्का $6$ पर रखे एक सिक्के से \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} होता
है। $3$ पर एक के ऊपर एक रखे दो सिक्के $6$ पर रखे एक सिक्के से \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित}
होते हैं। $2$ पर तीन सिक्के --- फिर से $6$ पर रखे एक से संतुलन!
\end{method}

\begin{proposition}[उत्तोलक का नियम]\label{prop:g3:levers-and-scales:rule}
\omterm{def:g3:levers-and-scales:lever}{उत्तोलक} की हर ओर के लिए सिक्कों की संख्या को \omterm{def:g3:levers-and-scales:lever}{धुरी} से उनकी दूरी से
गुणा करो। \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} ठीक तभी \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} होता है जब दोनों ओर का
\emph{गुणनफल एक जैसा} हो:
\[
2 \text{ सिक्के} \times 3 \text{ निशान} = 6
\qquad\text{संतुलित होते हैं}\qquad
1 \text{ सिक्का} \times 6 \text{ निशान} = 6 .
\]
अगर एक ओर का गुणनफल बड़ा हो, तो वही ओर नीचे जाती है।
\end{proposition}

\begin{example}[नियम को काम में लाना]\label{ex:g3:levers-and-scales:using}
निशान $2$ पर चार सिक्के रखे हैं: उनका गुणनफल $4 \times 2 = 8$ है।
निशान $4$ पर रखे सिक्कों से उन्हें \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} करना हो तो वहाँ भी गुणनफल
$8$ चाहिए: $8 = 2 \times 4$, यानी $4$ पर दो सिक्के काम कर देंगे।
और क्या अकेला एक सिक्का उन चारों से संतुलन बना सकता है? उसे निशान $8$
पर बैठना पड़ेगा: $1 \times 8 = 8$। सवार जितना हल्का, उसे उतना ही दूर
बैठना पड़ता है --- ठीक वही बात जो पापा वाले सी-सॉ ने सिखाई थी, अब
संख्याओं के साथ।
\end{example}

\section{तुला}

\begin{definition}[तुला]\label{def:g3:levers-and-scales:scale}
\emph{तुला}\index{तुला} निष्पक्षता के लिए बना एक \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} है: \omterm{def:g3:levers-and-scales:lever}{धुरी} से
\emph{बिलकुल बराबर} दूरियों पर लटके दो पलड़े। दूरियाँ बराबर हों तो
गुणनफल ठीक तभी बराबर होते हैं जब भार बराबर हों --- इसलिए तुला भारी
पलड़े की ओर झुक जाती है, और जब दोनों पलड़ों का बोझ बराबर भारी हो तब
सीधी टिकी रहती है।
\end{definition}

\begin{example}[तुलना करके तौलना]\label{ex:g3:levers-and-scales:weighing}
हज़ारों वर्षों तक व्यापारी आटा, सोना और काली मिर्च \omterm{def:g3:levers-and-scales:scale}{तुला} से बेचते
रहे: एक पलड़े में सामान, दूसरे में धातु के छोटे संदर्भ बाट, और बाट
तब तक जोड़ते जाओ जब तक दोनों पलड़े सीधे न हो जाएँ। \omterm{def:g3:levers-and-scales:scale}{तुला} कोई संख्या
नहीं जानती --- वह बस ``भारी'', ``हल्का'' या ``बराबर'' का उत्तर देती
है। पर सोच-समझकर चुने गए बाटों से तुलना ही किसी निष्पक्ष बाज़ार के
लिए काफ़ी है --- और, जैसा इस भाग का आख़िरी अध्याय दिखाएगा, यही
द्रव्यमान मापने की राह भी है।
\end{example}

\begin{remark}[ख़ाली तुला जाँच लो]\label{rem:g3:levers-and-scales:fair}
निष्पक्ष \omterm{def:g3:levers-and-scales:scale}{तुला} को \emph{ख़ाली} हालत में सीधा टिकना चाहिए। अगर एक भुजा
लंबी हो या एक पलड़ा भारी, तो सामान आने से पहले ही गुणनफल अलग-अलग हो
जाते हैं, और उसके बाद हर तौल बेईमानी है। ईमानदार व्यापारी अपनी ख़ाली
\omterm{def:g3:levers-and-scales:scale}{तुला} ग्राहक के सामने सीधी करके दिखाते थे --- ``ईमानदार सौदा'' कहने का
एक पुराना, बिना बोले वाला तरीक़ा।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:1}
इनमें \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} और \omterm{def:g3:levers-and-scales:lever}{धुरी} पहचानो: सी-सॉ; \ कुंदे पर रखी सब्बल; \ कैंची।
(कैंची में दो \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} छिपे हैं --- वे किस पर घूमते हैं?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:2}
सब्बल से भारी चट्टान उठानी हो, तो \omterm{def:g3:levers-and-scales:lever}{धुरी} के पास दबाना चाहिए या उससे
दूर? एक वाक्य में क्यों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:3}
सिक्कों वाले सी-सॉ पर गुणनफल निकालो: निशान $4$ पर $3$ सिक्के; \
निशान $5$ पर $2$ सिक्के; \ निशान $6$ पर $1$ सिक्का। इनमें से कौन-सी
जोड़ियाँ आपस में \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} होती हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:4}
बाईं ओर निशान $8$ पर एक सिक्का रखा है। संतुलन के लिए दाईं ओर $2$
सिक्के कहाँ रखने होंगे? और $4$ सिक्के?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:5}
बाईं ओर: निशान $3$ पर $3$ सिक्के। दाईं ओर: निशान $4$ पर $2$ सिक्के।
दोनों गुणनफल निकालो। कौन-सी ओर नीचे जाएगी?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:6}
\omterm{def:g3:levers-and-scales:scale}{तुला} के दोनों पलड़े \omterm{def:g3:levers-and-scales:lever}{धुरी} से बिलकुल बराबर दूरियों पर क्यों लटकाए जाते
हैं? बाईं भुजा लंबी हो जाए तो हर तौल का क्या होगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:7}
एक पलड़े में काँच की गोलियों की थैली और दूसरे में $6$ एक जैसे बाट
रखने पर \omterm{def:g3:levers-and-scales:scale}{तुला} सीधी टिकी है। थैली के बारे में हमें क्या पता चला? एक
बाट हटा लें तो \omterm{def:g3:levers-and-scales:scale}{तुला} क्या उत्तर देगी?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:8}
एक सी-सॉ: टॉम का भार $2$ बोरियों जितना है और वह निशान $3$ पर बैठा
है; उसकी छोटी बहन का भार $1$ बोरी जितना है। उससे संतुलन बनाने के लिए
उसे किस निशान पर बैठना होगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:levers-and-scales:9}
हथौड़े का पंजा एक कील खींचता है: तुम्हारा हाथ मूठ के उस सिरे को
धकेलता है जो हथौड़े के सिर की लकड़ी पर टिकी जगह से दूर है। \omterm{def:g3:levers-and-scales:lever}{धुरी}
कौन-सी है, और वह कील --- जिसे तुम्हारी उँगलियाँ कभी नहीं खींच सकती
थीं --- बाहर आ क्यों जाती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:levers-and-scales:10}
दादाजी ($4$ बोरियों जितने भारी) नन्ही ज़ो ($1$ बोरी जितनी भारी) के
साथ सी-सॉ खेलना चाहते हैं। सी-सॉ की हर ओर $1$ से $8$ तक निशान हैं।
उन्हें बैठाने के \emph{दो} अलग-अलग तरीक़े ढूँढ़ो जिनसे सी-सॉ \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित}
रहे।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:levers-and-scales:11}
