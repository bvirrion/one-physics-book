\chapter{आकाश में सूर्य: दिन और ऋतुएँ}\label{ch:g3:sun-days-seasons}

जून में तुम रात के खाने के बाद भी सुनहरे उजाले में बाहर खेलते हो;
दिसंबर में स्कूल से घर लौटने से पहले ही सड़क की बत्तियाँ जल जाती हैं।
सूर्य अपनी पुरानी आदत निभाता रहता है --- एक ओर से ऊपर, चोटी के पार,
दूसरी ओर से नीचे --- पर आकाश में उसकी राह पूरे साल एक-सी नहीं रहती।
पूरे एक साल उसका पीछा करो, और तुम्हें ऋतुएँ मिल जाएँगी।

\section{सूर्य की बदलती राह}

\begin{example}[गरमी की राह, जाड़े की राह]\label{ex:g3:sun-days-seasons:roads}
भरी गरमी में सूर्य की चाप देखो: वह जल्दी उगता है, बहुत ऊँचा चढ़ता है
--- दोपहर में लगभग सिर के ऊपर --- और देर से डूबता है: लंबी और ऊँची
राह, यानी लंबा दिन। भरे जाड़े में चाप नीची और छोटी होती है: सूर्य देर
से उगता है, छतों से ज़्यादा ऊपर कभी नहीं चढ़ता, और तीसरे पहर के बीच
में ही डूब जाता है। इन दोनों के बीच, वसंत और शरद में, राह मँझोली रहती
है --- और दिन तथा रात \omterm{ex:g2:measuring-time:clock}{घंटों} को लगभग बराबर बाँट लेते हैं।
\end{example}

\begin{proposition}[दोपहर का नियम]\label{prop:g3:sun-days-seasons:midday}
हर \omterm{def:g3:sun-days-seasons:year}{ऋतु} में सूर्य दिन के बीचोंबीच सबसे ऊँचा होता है --- और उसी पल हर
\omterm{def:g1:light-and-shadows:shadow}{छाया} उस दिन की सबसे छोटी \omterm{def:g1:light-and-shadows:shadow}{छाया} होती है और उसी दिशा में इशारा करती है
जिस दिशा में कल की दोपहर वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} करती थी, और कल की भी करेगी। दोपहर
की छायाएँ प्रकृति के सबसे भरोसेमंद तीर हैं।
\end{proposition}

\begin{example}[छाया से ऋतु पढ़ना]\label{ex:g3:sun-days-seasons:shadowlength}
दोपहर वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} की \emph{लंबाई} \omterm{def:g3:sun-days-seasons:year}{ऋतु} के साथ बदलती है: गरमी में, जब
सूर्य लगभग सिर के ऊपर होता है, किसी खंभे की दोपहर वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} एक छोटा
ठूँठ भर होती है; जाड़े में, जब सूर्य नीचा टँगा रहता है, वही खंभा
दोपहर में लंबी \omterm{def:g1:light-and-shadows:shadow}{छाया} डालता है। महीने में एक बार दोपहर वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} की नोक
ज़मीन पर निशान लगाओ, और साल भर में वे निशान किसी सुस्त लोलक की तरह
आगे-पीछे रेंगते दिखेंगे।
\end{example}

\section{छाया की घड़ी}

\begin{definition}[धूपघड़ी]\label{def:g3:sun-days-seasons:sundial}
\emph{धूपघड़ी}\index{धूपघड़ी} \omterm{def:g1:light-and-shadows:shadow}{छाया} से बनी एक घड़ी है: धूप में खड़ी
कोई छड़ी या पत्ती, और उसके चारों ओर बने निशान, जो बताते हैं कि हर
घंटे \omterm{def:g1:light-and-shadows:shadow}{छाया} कहाँ पड़ती है। सूर्य जैसे-जैसे अपनी चाप पर चलता है,
\omterm{def:g1:light-and-shadows:shadow}{छाया} निशानों के चारों ओर घूमती जाती है --- धीरे-धीरे, चुपचाप, और बिना
किसी हिलने वाले पुरज़े के।
\end{definition}

\begin{method}[छाया की घड़ी बनाना]\label{met:g3:sun-days-seasons:build}
किसी धूप वाली सुबह, ऐसी जगह पर जहाँ दिन भर धूप रहती हो:
\begin{enumerate}
  \item समतल ज़मीन में एक सीधी छड़ी खड़ी गाड़ो (या उसे रेत भरी बालटी
        में खड़ा कर दो);
  \item हर घंटे, ठीक घंटे पर, उसकी \omterm{def:g1:light-and-shadows:shadow}{छाया} की नोक पर एक कंकड़ रखो और
        खड़िया से उस पर घंटा लिख दो;
  \item शाम तक यही करते जाओ।
\end{enumerate}
कल \omterm{def:g1:light-and-shadows:shadow}{छाया} तुम्हारे कंकड़ों से ठीक समय पर मिलने आएगी: तुम्हारी छड़ी अब
समय बताती है। (कुछ समय तक, कम से कम --- \omterm{def:g3:sun-days-seasons:year}{ऋतु} सरकती है तो छायाएँ भी
सरकती हैं, और तुम्हारे कंकड़ धीरे-धीरे ग़लत पड़ने लगेंगे।)
\end{method}

\section{वर्ष और उसकी ऋतुएँ}

\begin{definition}[वर्ष और ऋतुएँ]\label{def:g3:sun-days-seasons:year}
पृथ्वी दिन में एक बार अपनी जगह घूमती है, और साथ ही एक विशाल यात्रा
\emph{सूर्य के चारों ओर} भी करती है --- एक पूरा चक्कर एक
\emph{वर्ष}\index{वर्ष} लेता है, यानी लगभग $365$ दिन। इस चक्कर के
दौरान हमारे आकाश में सूर्य की राह धीरे-धीरे ऊपर उठती और नीचे उतरती
है, और हम चार \emph{ऋतुओं}\index{ऋतु} से गुज़रते हैं: वसंत, ग्रीष्म,
शरद, शिशिर --- और फिर वसंत, चक्कर-दर-चक्कर।
\end{definition}

\begin{example}[चार पहर]\label{ex:g3:sun-days-seasons:quarters}
ग्रीष्म: ऊँचा सूर्य, लंबे दिन, दोपहर की छोटी छायाएँ। शिशिर: नीचा
सूर्य, छोटे दिन, दोपहर की लंबी छायाएँ। वसंत और शरद इनके बीच बैठते हैं
--- और उनमें से हर एक में एक ख़ास दिन आता है जब दिन और रात बराबर लंबे
होते हैं। पुराने ज़माने के किसान, नाविक और राजगीर यह पंचांग सीधे आकाश
से पढ़ना सीख गए थे।
\end{example}

\begin{remark}[शिशिर का मतलब ``ज़्यादा दूर'' नहीं है]\label{rem:g3:sun-days-seasons:notfarther}
यह अंदाज़ा लगाने का मन करता है कि सर्दी इसलिए पड़ती है कि पृथ्वी
सूर्य से दूर खिसक जाती है। ऐसा नहीं है --- दूरी लगभग बदलती ही नहीं,
और जब यहाँ शिशिर होता है तब ठीक उसी पल पृथ्वी के दूसरे आधे हिस्से के
बच्चों के यहाँ ग्रीष्म होता है, जिसे कोई ``ज़्यादा दूर'' समझा ही नहीं
सकता! असली कुंजी सूर्य की \emph{नीची राह} है: नीचा सूर्य, तिरछा
\omterm{def:g1:light-and-shadows:source}{प्रकाश}, कमज़ोर गरमी। पूरी व्याख्या का वादा किसी आगे वाले अध्याय के लिए
है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:1}
किस \omterm{def:g3:sun-days-seasons:year}{ऋतु} में सूर्य की चाप सबसे ऊँची और सबसे लंबी होती है? और सबसे
नीची तथा सबसे छोटी? इससे दिन की लंबाई पर क्या असर पड़ता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:2}
दिन के किस पल खंभे की \omterm{def:g1:light-and-shadows:shadow}{छाया} सबसे छोटी होती है? दिन-ब-दिन दोपहर वाली
\omterm{def:g1:light-and-shadows:shadow}{छाया} में और क्या ख़ास बात होती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:3}
वही खंभा जून में $2$ जूतों जितनी और दिसंबर में $7$ जूतों जितनी लंबी
दोपहर वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} डालता है। सूर्य की दोनों राहों से यह अंतर समझाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:4}
\omterm{def:g3:sun-days-seasons:sundial}{धूपघड़ी} समय कैसे बताती है? उसकी चलती हुई सुई किस चीज़ से बनी है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:5}
\cref{met:g3:sun-days-seasons:build} में छड़ी ऐसी जगह क्यों रहनी
चाहिए जहाँ दिन भर धूप रहती हो? बादलों वाले मौसम में --- और रात में
--- इस घड़ी का क्या होता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:6}
पृथ्वी को सूर्य के चारों ओर एक चक्कर लगाने में कितना समय लगता है? एक
चक्कर में कितनी ऋतुएँ समाती हैं? उन्हें वसंत से शुरू करके क्रम में
लगाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:7}
एक \omterm{def:g3:sun-days-seasons:year}{वर्ष} में लगभग कितने दिन होते हैं? और --- पिछले किसी अध्याय से ---
एक दिन में कितने घंटे? पृथ्वी की कौन-सी यात्रा इनमें से कौन-सा बनाती
है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:8}
अपनी खिड़की से एक ही हफ़्ते में दिखने वाले ऐसे दो सुराग़ बताओ जो
पंचांग देखे बिना बता दें कि भरी गरमी चल रही है या भरा जाड़ा।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sun-days-seasons:9}
जब तुम्हारे यहाँ शिशिर होता है, तब पृथ्वी के दूसरे आधे हिस्से के
बच्चों के यहाँ कौन-सी \omterm{def:g3:sun-days-seasons:year}{ऋतु} होती है? इससे ``सर्दी इसलिए पड़ती है कि
पूरी पृथ्वी सूर्य से दूर चली जाती है'' वाले अंदाज़े के बारे में क्या
साबित होता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sun-days-seasons:10}
एक माली मार्च में अपनी छाया-घड़ी के कंकड़ जमाती है। जून में \omterm{def:g1:light-and-shadows:shadow}{छाया}
निशान लगे \omterm{ex:g2:measuring-time:clock}{घंटों} पर कंकड़ों तक पहुँचती ही नहीं। क्या छड़ी हिल गई?
\cref{ex:g3:sun-days-seasons:shadowlength} की मदद से बताओ कि खिसका
क्या।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sun-days-seasons:11}
एक खोजी यात्री बिना घड़ी और बिना फ़ोन के जागती है, सूर्य को ठीक अपनी
सबसे ऊँची जगह पर देखती है, और एक घंटे बाद पाती है कि उसके तंबू की \omterm{def:g1:light-and-shadows:shadow}{छाया}
तंबू की ऊँचाई से भी लंबी हो चुकी है। जागते समय लगभग कितने बजे थे, और
वह भरी गरमी के पास है या भरे जाड़े के? दोपहर के नियम और \omterm{def:g1:light-and-shadows:shadow}{छाया} की लंबाई
से तर्क दो।
\end{exercise}
