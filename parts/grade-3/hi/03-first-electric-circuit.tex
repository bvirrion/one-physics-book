\section{एक बल्ब, एक बैटरी, दो तार}

\begin{definition}[बैटरी]\label{def:g3:first-electric-circuit:battery}
\emph{बैटरी}\index{बैटरी} बिजली का एक छोटा भंडार है। उसके दो अलग सिरे
होते हैं, जिन्हें उसके \emph{सिरे}\index{सिरा} कहते हैं: एक पर $+$
(धन) का निशान होता है, दूसरे पर $-$ (ऋण) का। दोनों सिरे ज़रूरी हैं
--- बिजली तभी खेलती है जब दोनों खेल में शामिल हों।
\end{definition}

\begin{definition}[बल्ब]\label{def:g3:first-electric-circuit:bulb}
\emph{बल्ब}\index{बल्ब} एक नन्हा लैंप है, जिसके काँच के भीतर एक पतला
तंतु होता है; बिजली उसमें से बहे तो वह सफ़ेद-गरम होकर दमकने लगता है।
उसके धातु के पेंदे को ध्यान से देखो: बल्ब के भी \emph{दो} संपर्क
स्थान होते हैं --- बिलकुल नीचे की नोक, और पेंदे की धातु वाली बग़ल।
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} पर दो सिरे, बल्ब पर दो संपर्क: यह संयोग नहीं है।
\end{definition}

\begin{method}[बल्ब जलाना]\label{met:g3:first-electric-circuit:light}
एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} और दो तारों से:
\begin{enumerate}
  \item एक तार \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $+$ सिरे से \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} के पेंदे की नोक तक जोड़ो;
  \item दूसरा तार \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $-$ सिरे से पेंदे की धातु वाली बग़ल तक
        जोड़ो;
  \item \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} जल उठता है --- तुमने परिपथ बना लिया!
\end{enumerate}
अगर वह अँधेरा ही रहे, तो हमेशा वाले संदिग्धों को जाँचो: कोई तार धातु
के संपर्क की जगह काँच को छू रहा हो, कोई जोड़ ढीला हो, या \omterm{def:g3:first-electric-circuit:battery}{बैटरी} चुक
गई हो।
\end{method}

\section{परिपथ का बंद होना ज़रूरी है}

\begin{definition}[विद्युत परिपथ]\label{def:g3:first-electric-circuit:circuit}
\emph{विद्युत परिपथ}\index{विद्युत परिपथ} बिजली के लिए बना एक पूरा
फंदा है: \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के एक सिरे से निकलकर, तारों और \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} में से होकर,
वापस दूसरे सिरे में। जब यह फंदा पूरा हो तब हम कहते हैं कि परिपथ
\emph{बंद}\index{बंद परिपथ} है; अगर उसमें कहीं भी दरार हो तो वह
\emph{खुला}\index{खुला परिपथ} है।
\end{definition}

\begin{proposition}[फंदे का नियम]\label{prop:g3:first-electric-circuit:loop}
\omterm{def:g3:first-electric-circuit:bulb}{बल्ब} तभी जलता है जब परिपथ बंद हो --- एक सिरे से \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} में से होकर
दूसरे सिरे तक जाता हुआ एक अटूट फंदा। फंदे में कहीं भी एक अकेली दरार
हो, और \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} बुझ जाता है: बिजली दरारें फाँदती नहीं, और बंद गलियों में
बहती नहीं।
\end{proposition}

\begin{example}[दोनों सिरे क्यों?]\label{ex:g3:first-electric-circuit:bothends}
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $+$ सिरे से \emph{दो} तार \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} के दोनों संपर्कों तक जोड़ो
और $-$ को छोड़ दो: कुछ नहीं। एक तार $+$ से और एक $-$ से, पर दोनों
\omterm{def:g3:first-electric-circuit:bulb}{बल्ब} की नोक तक: फिर कुछ नहीं। \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} तभी जलता है जब फंदा \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के दोनों
अलग-अलग सिरों \emph{और} \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} के दोनों अलग-अलग संपर्कों से होकर गुज़रे
--- यानी पूरा चक्कर लगाए।
\end{example}

\begin{example}[स्विच]\label{ex:g3:first-electric-circuit:switch}
\emph{स्विच}\index{स्विच} परिपथ में लगा एक दरवाज़ा है: धातु का एक
नन्हा पुल, जिसे क्लिक करते ही या तो बिछा दिया जाता है (बंद --- फंदा
पूरा, \omterm{def:g1:light-and-shadows:source}{प्रकाश}!) या उठा दिया जाता है (दरार --- अँधेरा)। टॉर्च का बटन,
दीवार पर लगा बत्ती का \omterm{ex:g3:first-electric-circuit:switch}{स्विच}, दरवाज़े की घंटी का बटन: ये सब फंदे में
लगे दरवाज़े हैं। ऐसा एक \omterm{ex:g3:first-electric-circuit:switch}{स्विच} तुम कपड़े की चिमटी और उसके दोनों जबड़ों
पर मोड़ी गई रसोई की पन्नी की पट्टी से ख़ुद बना सकते हो।
\end{example}

\begin{remark}[सलीक़े के चित्र]\label{rem:g3:first-electric-circuit:symbols}
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} का हर उभार और \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} का हर चूड़ीदार पेंच बनाने के बजाय बिजली वाले
\emph{संकेत} काम में लेते हैं: \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के लिए अलग-अलग लंबाई की दो
लकीरें (लंबी वाली $+$ है), लैंप के लिए क्रॉस वाला एक वृत्त, \omterm{ex:g3:first-electric-circuit:switch}{स्विच} के
लिए एक नन्हा उठाऊ पुल, और तारों के लिए सीधी रेखाएँ। संकेतों वाला
परिपथ-चित्र बनाने में तेज़ होता है, और उसे दुनिया में कोई भी पढ़ सकता
है --- ऐसे चित्र हम कई वर्षों तक काम में लेंगे।
\end{remark}

\begin{remark}[बैटरी हाँ, दीवार का सॉकेट कभी नहीं]\label{rem:g3:first-electric-circuit:safety}
इस अध्याय का हर खेल छोटी गोल या चपटी बैटरियों से खेला जाता है: उनका
धक्का बहुत हल्का होता है और प्रयोगों के लिए सुरक्षित। \emph{दीवार का
सॉकेट} दूसरी ही दुनिया है --- उसका धक्का सैकड़ों गुना ज़्यादा
ताक़तवर होता है, इतना कि चोट पहुँचा दे या जान ले ले। सॉकेट में कभी
तार, उँगलियाँ या कुछ और मत डालो, और सॉकेट में लगने वाली किसी भी चीज़
को कभी मत खोलो। \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के खेल: हाँ। सॉकेट: कभी नहीं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:1}
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} के दोनों सिरों और \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} के दोनों संपर्क स्थानों के नाम बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:2}
\omterm{def:g3:first-electric-circuit:circuit}{बंद परिपथ} क्या है? \omterm{def:g3:first-electric-circuit:circuit}{खुला परिपथ} क्या है? \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} इनमें से किसमें जलता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:3}
लिया \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $+$ सिरे को एक तार से \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} की नोक तक जोड़ती है और वहीं
रुक जाती है। \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} अँधेरा ही रहता है। फंदे के नियम के शब्दों में कहो
कि क्या कमी है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:4}
फंदे में \omterm{ex:g3:first-electric-circuit:switch}{स्विच} का काम क्या है? बंद \omterm{ex:g3:first-electric-circuit:switch}{स्विच} के भीतर क्या होता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:5}
\cref{rem:g3:first-electric-circuit:symbols} वाले सलीक़ेदार संकेतों
से एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, एक बंद \omterm{ex:g3:first-electric-circuit:switch}{स्विच} और एक लैंप वाला परिपथ बनाओ। कौन-सा संकेत
किसका है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:6}
कहीं घूमने गए हो और बीच में टॉर्च बुझ जाती है।
\cref{met:g3:first-electric-circuit:light} की मदद से जाँचने लायक तीन
संभव दरारें या ख़राबियाँ बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:7}
दरवाज़े की घंटी वाले परिपथ में ``\omterm{ex:g3:first-electric-circuit:switch}{स्विच}'' कहाँ है? घंटी सिर्फ़ बटन
दबाए रखने तक ही क्यों बजती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:8}
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} से खेलना ठीक है, फिर भी दीवार के सॉकेट में कभी कुछ क्यों नहीं
घुसाना चाहिए? (\cref{rem:g3:first-electric-circuit:safety} देखो।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:9}
एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} की नोक \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $+$ सिरे पर टिकी है, और \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $-$ सिरे से
आता एक तार \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} के पेंदे की धातु वाली बग़ल से दबाया गया है: क्या वह
जलेगा? उँगली से फंदे का पीछा करके तय करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:10}
सैम एक फंदा बनाता है: \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, तार, \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}, तार, और वापस \omterm{def:g3:first-electric-circuit:battery}{बैटरी} --- पर वह
दोनों तार एक ही सिरे, यानी $-$ से जोड़ देता है, ``क्योंकि वह पास
था''। फंदा पूरा दिखने पर भी \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} अँधेरा रहता है।
\cref{ex:g3:first-electric-circuit:bothends} की मदद से समझाओ कि उसके
फंदे में क्या कमी है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:11}
सजावटी बत्तियों की एक लड़ी में से एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} ग़ायब है, और \emph{पूरी}
लड़ी अँधेरी है, जबकि बाक़ी सारे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} ठीक हैं। इससे तुम्हें क्या पता
चलता है कि \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} फंदे में किस तरह लगे हैं? (अगले साल के परिपथ वाले
अध्याय में इस तरतीब को एक नाम मिलेगा।)
\end{exercise}
