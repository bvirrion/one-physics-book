\chapter{द्रव्यमान मापना}\label{ch:g3:measuring-mass}

बाज़ार में: ``एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} सेब देना!'' दुकानदार पलड़े में तब तक सेब
डालता जाता है जब तक \omterm{def:g3:levers-and-scales:scale}{तुला} संतुष्ट न हो जाए। ग्राहक ने माँगा ठीक-ठीक
क्या --- और \omterm{def:g3:levers-and-scales:scale}{तुला} को कैसे पता चलता है कि कब रुकना है? \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} और \omterm{def:g3:levers-and-scales:scale}{तुला}
हमारे औज़ार-डिब्बे में आ चुके हैं, इसलिए अब हम एक नई चीज़ मापने के
लिए तैयार हैं: \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}।

\section{द्रव्यमान: कितना पदार्थ}

\begin{definition}[द्रव्यमान]\label{def:g3:measuring-mass:mass}
किसी चीज़ का \emph{द्रव्यमान}\index{द्रव्यमान} बताता है कि वह कितने
पदार्थ से बनी है। पदार्थ जितना ज़्यादा, द्रव्यमान उतना ही बड़ा ---
और पृथ्वी उसे उतने ही ज़ोर से नीचे खींचती है, इसीलिए ज़्यादा
द्रव्यमान वाली चीज़ें हाथ में भारी लगती हैं और \omterm{def:g3:levers-and-scales:scale}{तुला} के अपने पलड़े को
नीचे दबा देती हैं।
\end{definition}

\begin{example}[बड़ा होना भारी होना नहीं है]\label{ex:g3:measuring-mass:bignotcheavy}
तकिया तुम्हारी पूरी बाँहें भर देता है, फिर भी तुम उसे आसानी से उठा
लेते हो। तुम्हारी मुट्ठी से भी छोटा एक पत्थर कलाई पर ज़ोर डाल देता
है। तकिया बड़ा है पर उसमें पदार्थ कम है --- ज़्यादातर पंख और \omterm{def:g2:air-around-us:air}{हवा};
पत्थर छोटा है पर पदार्थ से ठसाठस भरा। \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} का सवाल यह है कि
\emph{पदार्थ कितना है}, यह नहीं कि जगह कितनी घिरती है। (ये दोनों
विचार और उनका मुश्किल नाच किसी दिन तैरने-डूबने की कहानी में फिर
मिलेंगे।)
\end{example}

\begin{definition}[ग्राम और किलोग्राम]\label{def:g3:measuring-mass:units}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \emph{ग्राम}\index{ग्राम} (लिखते हैं \unit{g}) और
\emph{किलोग्राम}\index{किलोग्राम} (लिखते हैं \unit{kg}) में मापा
जाता है:
\[
\qty{1}{kg} = \qty{1000}{g}.
\]
एक क्लिप का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} लगभग \qty{1}{g} होता है; एक सेब का लगभग
\qty{150}{g}; दूध के एक डिब्बे का लगभग \qty{1}{kg}; और तुम्हारा,
कुछ दसियों किलोग्राम। \omterm{def:g2:measuring-length:units}{मीटर} की तरह ये \omterm{def:g2:measuring-length:measure}{मात्रक} भी पृथ्वी पर सबके लिए एक
जैसे हैं।
\end{definition}

\section{तुला से तौलना}

\begin{method}[बाटों से तौलना]\label{met:g3:measuring-mass:weigh}
\omterm{def:g3:levers-and-scales:scale}{तुला} और निशान लगे धातु के बाटों के डिब्बे (\qty{500}{g},
\qty{200}{g}, \qty{100}{g}, \qty{50}{g}, \dots) से किसी ख़रबूज़े का
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} मापने के लिए:
\begin{enumerate}
  \item जाँच लो कि ख़ाली \omterm{def:g3:levers-and-scales:scale}{तुला} सीधी टिकी है;
  \item ख़रबूज़ा एक पलड़े में रखो;
  \item दूसरे पलड़े में बाट डालो, पहले सबसे बड़े काम के बाट, जब तक
        दोनों पलड़े सीधे न हो जाएँ;
  \item बाटों को जोड़ लो: उनका जोड़ ही ख़रबूज़े का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} है।
\end{enumerate}
अगर पलड़े \qty{500}{g}, \qty{200}{g} और \qty{100}{g} के बाटों से
सीधे हो जाएँ, तो ख़रबूज़े का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} $500 + 200 + 100 =
\qty{800}{g}$ है।
\end{method}

\begin{example}[दूसरी तुलाएँ]\label{ex:g3:measuring-mass:otherscales}
रसोई की तराज़ू और वज़न तौलने वाली तराज़ू में दूसरा पलड़ा होता ही नहीं:
तुम बोझ रखते हो, और एक सुई घूमकर (या कोई संख्या जलकर) सीधे \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}
दिखा देती है। भीतर एक दबी हुई स्प्रिंग तुम्हारे लिए तुलना कर देती है।
यह सुविधाजनक और तेज़ है --- पर पुरानी दो-पलड़े वाली \omterm{def:g3:levers-and-scales:scale}{तुला} में कोई
मशीनरी छिपी ही नहीं होती: बस वही \omterm{def:g3:levers-and-scales:lever}{उत्तोलक} का नियम, जिसे तुम पहले से
समझते हो।
\end{example}

\begin{example}[बाज़ार पढ़ना]\label{ex:g3:measuring-mass:market}
``एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} सेब'' का मतलब है: उतने सेब जब तक \omterm{def:g3:levers-and-scales:scale}{तुला} \qty{1}{kg} के
बाट से \omterm{def:g2:balance-equilibrium:equilibrium}{संतुलित} न हो जाए --- यानी लगभग $6$ या $7$ सेब। आटा
\qty{1}{kg} की थैलियों में आता है, चॉकलेट की पट्टियाँ अकसर
\qty{100}{g} की होती हैं, एक चिट्ठी में लगभग \qty{20}{g} काग़ज़ लगता
है, और डाकिये का नियम ``\qty{20}{g} से कम पर एक टिकट'' असल में
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} का नियम है।
\end{example}

\section{द्रव्यमान पदार्थ के पीछे चलता है}

\begin{proposition}[वही पदार्थ, वही द्रव्यमान]\label{prop:g3:measuring-mass:conserved}
किसी चीज़ का \emph{आकार} बदलने से उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} नहीं बदलता। आटे की
लोई को बेलकर साँप बना लो, चपटा करके रोटी बना लो, या चार टुकड़ों में
काटकर सारे टुकड़े एक साथ तौल लो: \omterm{def:g3:levers-and-scales:scale}{तुला} हर बार वही \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बताएगी।
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} पदार्थ गिनता है --- और दबाने, बेलने या काटने से न पदार्थ
जुड़ता है, न घटता है।
\end{proposition}

\begin{remark}[तुला दिखावे को अनदेखा कर देती है]\label{rem:g3:measuring-mass:appearances}
मापने की चुपचाप ताक़त यही है: आँखें चपटी रोटी को उसी लोई से ``बड़ा''
मानती हैं जिससे वह बनी, और फूले हुए तकिये को पत्थर से ``भारी'' ---
पर \omterm{def:g3:levers-and-scales:scale}{तुला}, अपने से पहले आए \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} और रूलर की तरह, सिर्फ़ पदार्थ की
ख़बर देती है, और कुछ नहीं। एक-एक करके हमारे उपकरण ``ऐसा लगता है'' की
जगह ``ऐसा है'' रख रहे हैं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:1}
किसी चीज़ का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} क्या बताता है? उसे कौन-से दो \omterm{def:g2:measuring-length:measure}{मात्रक} मापते
हैं, और एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} में कितने \omterm{def:g3:measuring-mass:units}{ग्राम} होते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:2}
ज़्यादा संभव \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} चुनो: एक क्लिप --- \qty{1}{g} या
\qty{1}{kg}? एक सेब --- \qty{150}{g} या \qty{15}{kg}? दूध का एक
डिब्बा --- \qty{1}{g} या \qty{1}{kg}? एक बिल्ली --- \qty{4}{kg} या
\qty{400}{kg}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:3}
एक कद्दू के सामने \qty{500}{g} का एक बाट और \qty{200}{g} के दो बाट
रखने पर पलड़े सीधे हो जाते हैं। कद्दू का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} कितना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:4}
किसी भी तौल से पहले ख़ाली \omterm{def:g3:levers-and-scales:scale}{तुला} का सीधा टिकना क्यों ज़रूरी है?
(बेईमान व्यापारी याद करो।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:5}
समुद्र-तट वाली गेंद सूप के डिब्बे से कहीं बड़ी है, फिर भी डिब्बा
अपना पलड़ा नीचे दबा देता है।
\cref{ex:g3:measuring-mass:bignotcheavy} की मदद से समझाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:6}
\qty{2}{kg} में कितने \omterm{def:g3:measuring-mass:units}{ग्राम} होते हैं? आधे \omterm{def:g3:measuring-mass:units}{किलोग्राम} में? ज़्यादा
कौन है: \qty{1200}{g} या \qty{1}{kg}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:7}
\qty{100}{g} की एक चॉकलेट पट्टी को उसके सारे छोटे टुकड़ों में तोड़
दिया जाता है, और सारे टुकड़े एक साथ \omterm{def:g3:levers-and-scales:scale}{तुला} पर रखे जाते हैं। \omterm{def:g3:levers-and-scales:scale}{तुला}
कितना \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} दिखाएगी, और कौन-सा नियम ऐसा कहता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:8}
रसोई की तराज़ू और दो-पलड़े वाली \omterm{def:g3:levers-and-scales:scale}{तुला}, दोनों आटे की थैली तौलती हैं।
तुलना करने के लिए हर एक क्या काम में लेती है --- और इनमें से कौन उस
नियम से चलती है जो तुमने तीन अध्याय पहले सीखा था?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:9}
हलवाई को \qty{800}{g} आटा चाहिए और उसके पास \qty{500}{g},
\qty{200}{g}, \qty{200}{g}, \qty{100}{g} और \qty{50}{g} के बाट
हैं। ठीक \qty{800}{g} तौलने के दो अलग-अलग तरीक़े ढूँढ़ो --- यह जानते
हुए कि बाट \emph{किसी भी} पलड़े में रखे जा सकते हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:10}
नीना पूरा संतरा तौलती है: \qty{200}{g}। फिर वह उसे छीलकर छिला हुआ
संतरा अकेला तौलती है: \qty{160}{g}। क्या
\cref{prop:g3:measuring-mass:conserved} झूठा पड़ गया? ग़ायब \omterm{def:g3:measuring-mass:units}{ग्राम}
कहाँ हैं, और नीना एक और तौल से इसे कैसे साबित कर सकती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:11}
आख़िरकार तुम्हारे लिए आसान हो चुकी एक पहेली: \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} किसका ज़्यादा
है, एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} पंखों का या एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} लोहे का? और जगह कौन
ज़्यादा घेरता है? बताओ कि इतने सारे लोग पहले सवाल का ग़लत उत्तर
क्यों देते हैं।
\end{exercise}
