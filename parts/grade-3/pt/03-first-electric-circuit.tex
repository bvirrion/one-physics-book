\chapter{Um primeiro circuito elétrico}\label{ch:g3:first-electric-circuit}

Clique --- a lanterna joga o feixe dela pelo cômodo escuro. Dentro
daquele tubo de plástico moram uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, uma lampadinha e duas tiras
de metal, jogando um jogo de regras rigorosas. Hoje vamos desmontar o
jogo: com uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, uma \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} e dois fios, vamos fabricar \omterm{def:g1:light-and-shadows:source}{luz} com
as nossas próprias mãos.

\section{Uma lâmpada, uma pilha, dois fios}

\begin{definition}[A pilha]\label{def:g3:first-electric-circuit:battery}
Uma \emph{pilha}\index{pilha} é um pequeno estoque de eletricidade.
Ela tem duas pontas diferentes, chamadas de
\emph{terminais}\index{terminal}: uma marcada com $+$ (mais), a outra
com $-$ (menos). As duas pontas importam --- a eletricidade só brinca
quando as duas entram no jogo.
\end{definition}

\begin{definition}[A lâmpada]\label{def:g3:first-electric-circuit:bulb}
Uma \emph{lâmpada}\index{lâmpada} é uma luminária pequena com um fio
fininho dentro do vidro que fica incandescente quando a eletricidade
passa por ele. Olhe com atenção a base de metal dela: a lâmpada
também tem \emph{dois} pontos de contato --- a pontinha lá embaixo e
a lateral de metal da base. Dois \omterm{def:g3:first-electric-circuit:battery}{terminais} na \omterm{def:g3:first-electric-circuit:battery}{pilha}, dois contatos na
lâmpada: isso não é por acaso.
\end{definition}

\begin{method}[Acender a lâmpada]\label{met:g3:first-electric-circuit:light}
Com uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, uma \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} e dois fios:
\begin{enumerate}
  \item ligue um fio do \omterm{def:g3:first-electric-circuit:battery}{terminal} $+$ da \omterm{def:g3:first-electric-circuit:battery}{pilha} à pontinha da base da
        \omterm{def:g3:first-electric-circuit:bulb}{lâmpada};
  \item ligue o outro fio do \omterm{def:g3:first-electric-circuit:battery}{terminal} $-$ da \omterm{def:g3:first-electric-circuit:battery}{pilha} à lateral de metal
        da base;
  \item a \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} acende --- você montou um circuito!
\end{enumerate}
Se ela continuar apagada, confira os suspeitos de sempre: um fio
encostando no vidro em vez do contato de metal, uma ligação frouxa ou
uma \omterm{def:g3:first-electric-circuit:battery}{pilha} gasta.
\end{method}

\section{O circuito precisa fechar}

\begin{definition}[Circuito elétrico]\label{def:g3:first-electric-circuit:circuit}
Um \emph{circuito elétrico}\index{circuito elétrico} é uma volta
completa para a eletricidade: sai de um \omterm{def:g3:first-electric-circuit:battery}{terminal} da \omterm{def:g3:first-electric-circuit:battery}{pilha}, passa
pelos fios e pela \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} e volta para o outro \omterm{def:g3:first-electric-circuit:battery}{terminal}. Quando a
volta está completa, dizemos que o circuito está
\emph{fechado}\index{circuito fechado}; se houver uma falha em
qualquer ponto, ele está \emph{aberto}\index{circuito aberto}.
\end{definition}

\begin{proposition}[A lei da volta]\label{prop:g3:first-electric-circuit:loop}
A \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} acende só quando o circuito está fechado --- uma volta sem
interrupção de um \omterm{def:g3:first-electric-circuit:battery}{terminal}, passando pela \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}, de volta ao outro
\omterm{def:g3:first-electric-circuit:battery}{terminal}. Uma única falha, em qualquer ponto da volta, e a \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}
apaga: a eletricidade não pula buracos e não escorre para becos sem
saída.
\end{proposition}

\begin{example}[Por que as duas pontas?]\label{ex:g3:first-electric-circuit:bothends}
Ligue \emph{dois} fios do \omterm{def:g3:first-electric-circuit:battery}{terminal} $+$ da \omterm{def:g3:first-electric-circuit:battery}{pilha} aos dois contatos da
\omterm{def:g3:first-electric-circuit:bulb}{lâmpada}, deixando o $-$ de fora: nada. Um fio do $+$ e um do $-$, mas
os dois na pontinha da \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}: nada de novo. A \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} acende só
quando a volta visita os dois \omterm{def:g3:first-electric-circuit:battery}{terminais} diferentes da \omterm{def:g3:first-electric-circuit:battery}{pilha} \emph{e}
os dois contatos diferentes da \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} --- uma viagem de ida e volta
completa.
\end{example}

\begin{example}[O interruptor]\label{ex:g3:first-electric-circuit:switch}
Um \emph{interruptor}\index{interruptor} é uma porta no circuito: uma
pontezinha de metal que o clique abaixa (fechado --- a volta está
completa, \omterm{def:g1:light-and-shadows:source}{luz}!) ou levanta (uma falha --- escuro). O botão da
lanterna, o \omterm{ex:g3:first-electric-circuit:switch}{interruptor} de \omterm{def:g1:light-and-shadows:source}{luz} da parede, o botão de uma campainha:
todos são portas numa volta. Você pode montar um com um prendedor de
roupa e uma tira de papel-alumínio dobrada sobre as pontas dele.
\end{example}

\begin{remark}[Desenhos organizados]\label{rem:g3:first-electric-circuit:symbols}
Em vez de desenhar cada saliência da \omterm{def:g3:first-electric-circuit:battery}{pilha} e cada rosca da \omterm{def:g3:first-electric-circuit:bulb}{lâmpada},
os eletricistas usam \emph{símbolos}: duas barras de tamanhos
diferentes para a \omterm{def:g3:first-electric-circuit:battery}{pilha} (a comprida é o $+$), um círculo com uma cruz
para a \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}, uma pontezinha levadiça para o \omterm{ex:g3:first-electric-circuit:switch}{interruptor} e linhas
retas para os fios. Um desenho de circuito com símbolos é mais rápido
de fazer, e qualquer pessoa do mundo consegue ler --- vamos usar
esses desenhos por muitos anos.
\end{remark}

\begin{remark}[Pilha sim, tomada nunca]\label{rem:g3:first-electric-circuit:safety}
Tudo neste capítulo é brincado com \omterm{def:g3:first-electric-circuit:battery}{pilhas} pequenas, redondas ou
chatas: o empurrão delas é pequenininho e seguro para experiências. A
\emph{tomada} da parede é outro mundo --- o empurrão dela é centenas
de vezes mais possante, o bastante para machucar ou matar. Nunca
enfie fios, dedos nem qualquer outra coisa numa tomada, e nunca
desmonte nada que se ligue numa. Brincadeiras com \omterm{def:g3:first-electric-circuit:battery}{pilha}: sim. Tomada:
nunca.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:1}
Diga os dois \omterm{def:g3:first-electric-circuit:battery}{terminais} de uma \omterm{def:g3:first-electric-circuit:battery}{pilha} e os dois pontos de contato de
uma \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:2}
O que é um \omterm{def:g3:first-electric-circuit:circuit}{circuito fechado}? O que é um \omterm{def:g3:first-electric-circuit:circuit}{circuito aberto}? Em qual dos
dois a \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} acende?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:3}
Lea liga o \omterm{def:g3:first-electric-circuit:battery}{terminal} $+$ da \omterm{def:g3:first-electric-circuit:battery}{pilha} à pontinha da \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} com um fio e
para por aí. A \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} continua apagada. O que está faltando, nas
palavras da lei da volta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:4}
Que serviço um \omterm{ex:g3:first-electric-circuit:switch}{interruptor} faz na volta? O que há dentro de um
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} quando ele está fechado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:5}
Desenhe, com os símbolos organizados da
\cref{rem:g3:first-electric-circuit:symbols}, um circuito com uma
\omterm{def:g3:first-electric-circuit:battery}{pilha}, um \omterm{ex:g3:first-electric-circuit:switch}{interruptor} fechado e uma \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}. Qual símbolo é qual?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:6}
A lanterna apaga no meio de um acampamento. Dê três falhas ou
problemas possíveis para conferir, a partir do
\cref{met:g3:first-electric-circuit:light}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:7}
No circuito da campainha, onde está o ``\omterm{ex:g3:first-electric-circuit:switch}{interruptor}''? Por que a
campainha só toca enquanto o botão está apertado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:8}
Por que nunca se deve enfiar nada numa tomada, se brincar com uma
\omterm{def:g3:first-electric-circuit:battery}{pilha} não tem problema? (Veja a
\cref{rem:g3:first-electric-circuit:safety}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:9}
Uma \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} com a pontinha encostada no \omterm{def:g3:first-electric-circuit:battery}{terminal} $+$ da \omterm{def:g3:first-electric-circuit:battery}{pilha}, e um
fio do \omterm{def:g3:first-electric-circuit:battery}{terminal} $-$ da \omterm{def:g3:first-electric-circuit:battery}{pilha} apertado contra a lateral de metal da
base da \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}: ela vai acender? Siga a volta com o dedo para
decidir.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:10}
Sam monta uma volta: \omterm{def:g3:first-electric-circuit:battery}{pilha}, fio, \omterm{def:g3:first-electric-circuit:bulb}{lâmpada}, fio, de volta à \omterm{def:g3:first-electric-circuit:battery}{pilha} ---
mas ele liga os dois fios no mesmo \omterm{def:g3:first-electric-circuit:battery}{terminal}, o $-$, ``porque estava
mais perto''. A \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} continua apagada, embora a volta pareça
completa. Explique o que falta na volta dele, usando o
\cref{ex:g3:first-electric-circuit:bothends}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:11}
Um cordão de luzinhas de festa está com uma \omterm{def:g3:first-electric-circuit:bulb}{lâmpada} faltando, e o
cordão \emph{inteiro} está apagado, embora todas as outras \omterm{def:g3:first-electric-circuit:bulb}{lâmpadas}
estejam boas. O que isso diz a você sobre o jeito como as \omterm{def:g3:first-electric-circuit:bulb}{lâmpadas}
estão arrumadas na volta? (O capítulo de circuitos do ano que vem vai
dar um nome a essa arrumação.)
\end{exercise}
