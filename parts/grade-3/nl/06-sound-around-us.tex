\chapter{Geluid om ons heen}\label{ch:g3:sound-around-us}

Neurie je lievelingsliedje --- en leg terwijl je neuriet twee vingers
zachtjes op de voorkant van je keel. Voel je dat trillen? Je hebt zojuist
een \omterm{def:g3:sound-around-us:sound}{geluid} op zijn geboorteplek aangeraakt. Elke stem, elke trommel,
elke piepende deur maakt haar \omterm{def:g3:sound-around-us:sound}{geluid} op precies dezelfde manier, en dit
hoofdstuk gaat op jacht naar die truc.

\section{Waar geluiden geboren worden}

\begin{definition}[Trilling]\label{def:g3:sound-around-us:vibration}
Een \emph{trilling}\index{trilling} is een heel snel beven --- een
heen-en-weerbeweging die zich vele keren per \omterm{def:g2:measuring-time:second}{seconde} herhaalt, vaak te
snel voor het oog. Een aangeslagen gitaarsnaar trilt: je ziet alleen
een waas, maar je vingertop voelt het zoemen.
\end{definition}

\begin{definition}[Geluid]\label{def:g3:sound-around-us:sound}
\emph{Geluid}\index{geluid} is wat we horen wanneer iets trilt en dat
beven door de \omterm{def:g2:air-around-us:air}{lucht} naar onze oren reist. Het trillende ding is de
\emph{bron} van het geluid --- een snaar, een trommelvel, een bel, je
eigen keel.
\end{definition}

\begin{method}[Rijst laten dansen]\label{met:g3:sound-around-us:rice}
Om een \omterm{def:g3:sound-around-us:sound}{geluid} geboren te \emph{zien} worden:
\begin{enumerate}
  \item strooi een paar rijstkorrels op het vel van een trommel (of op
        het plastic deksel van een blik, of op een strak gespannen
        ballonvel);
  \item tik op het vel vlak bij de korrels --- boem: de korrels
        springen op!
  \item \emph{schreeuw} nu alleen maar van heel dichtbij tegen het
        vel: de korrels trillen weer, zonder aangeraakt te worden.
\end{enumerate}
Het vel trilt telkens als het klinkt --- en zelfs een \omterm{def:g3:sound-around-us:sound}{geluid} dat uit
jouw mond aankomt, zet het aan het beven.
\end{method}

\begin{proposition}[Geen trilling, geen geluid]\label{prop:g3:sound-around-us:novib}
Elk \omterm{def:g3:sound-around-us:sound}{geluid} komt van een \omterm{def:g3:sound-around-us:vibration}{trilling} --- en zodra de \omterm{def:g3:sound-around-us:vibration}{trilling} stopt,
stopt het \omterm{def:g3:sound-around-us:sound}{geluid} er ogenblikkelijk mee. Sla op een bekken en laat het
naklinken; pak het dan stevig met beide handen vast: stilte, op het
moment dat je greep het beven stillegt.
\end{proposition}

\begin{example}[Overal geluidsbronnen]\label{ex:g3:sound-around-us:sources}
Jaag op de \omterm{def:g3:sound-around-us:vibration}{trilling} achter elk \omterm{def:g3:sound-around-us:sound}{geluid}: de gitaar --- aangeslagen
snaren beven; de trommel --- het aangeslagen vel beeft; je stem ---
twee plooitjes in je keel beven terwijl de \omterm{def:g2:air-around-us:air}{lucht} uit je longen
erlangs stroomt (dat is het zoemen dat je vingers voelden); de zoemende
vlieg --- haar vleugels beven honderden keren per \omterm{def:g2:measuring-time:second}{seconde}; de
dichtgeslagen deur --- de hele deur en haar kozijn schudden even na.
\end{example}

\section{Hard en zacht, hoog en laag}

\begin{example}[Hard is een grote trilling]\label{ex:g3:sound-around-us:loud}
Tokkel zachtjes aan een gespannen elastiekje: een smalle waas, een
zachte tsjing. Tokkel hard: een brede waas, een harde tsjing. Hoe
heftiger de bron beeft --- hoe wijder haar heen en weer --- hoe
\emph{harder} het \omterm{def:g3:sound-around-us:sound}{geluid}. Kijk ook naar de rijstkorrels: een zacht
tikje laat ze rillen, een machtige dreun gooit ze de \omterm{def:g2:air-around-us:air}{lucht} in.
\end{example}

\begin{example}[Hoog en laag]\label{ex:g3:sound-around-us:pitch}
Span een elastiekje over een open doos en tokkel; knijp het dan in het
midden vast en tokkel de korte helft: de tsjing wordt \emph{hoger},
als het tsjilpen van een vogel. Korte, strakke, dunne dingen beven
sneller en zingen hoog; lange, slappe, dikke dingen beven langzamer en
zingen laag. Daarom tsjilpt de kleine fluit en gromt de grote tuba ---
en daarom zingt de dunste snaar van de gitaar het hoogst.
\end{example}

\begin{remark}[Bescherm het trommeltje in je oor]\label{rem:g3:sound-around-us:ears}
Diep in je oor beeft een piepklein velletje --- dunner dan papier ---
bij elk aankomend \omterm{def:g3:sound-around-us:sound}{geluid} mee, net als de trommel met de rijstkorrels.
\omterm{def:g3:sound-around-us:sound}{Geluiden} die te hard zijn schudden het te heftig: heel harde
muziek, machines en vuurwerk kunnen het voorgoed beschadigen, en het
groeit niet terug. Houd je oren dicht in de buurt van hard lawaai, en
schreeuw nooit in iemands oor --- oren hebben geen oogleden.
\end{remark}

\section{Van de bron naar het oor}

\begin{example}[De reis van een klap]\label{ex:g3:sound-around-us:journey}
Een vriend klapt aan de overkant van het schoolplein in zijn handen.
De klap zet de \omterm{def:g2:air-around-us:air}{lucht} aan het beven; het beven spreidt zich in alle
richtingen door de \omterm{def:g2:air-around-us:air}{lucht} uit, als de kringen om een kiezel die in een
vijver valt; een deel ervan bereikt je oor en zet het velletje in je
oor aan het beven --- en je hoort de klap. Bron, reis, oor: elk
\omterm{def:g3:sound-around-us:sound}{geluid} dat je ooit hebt gehoord heeft die tocht gemaakt. Hoe snel is
die reis, en waar kan ze doorheen? Het geluidshoofdstuk van volgend
jaar pakt precies die vragen op.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:1}
Wat is een \omterm{def:g3:sound-around-us:vibration}{trilling}? Noem het trillende onderdeel bij: een gitaar; \
een trommel; \ je stem; \ een zoemende vlieg.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:2}
Wat bewijzen de springende korrels in de rijstproef over een klinkend
trommelvel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:3}
Waarom valt een naklinkend bekken meteen stil als je het vastpakt?
Welke wet uit het hoofdstuk is hier aan het werk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:4}
Wat verander je aan je tokkel om het elastiekje \emph{harder} te laten
klinken? Hoe ziet de waas van het elastiekje er dan uit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:5}
Wat zingt hoger: het korte geknepen elastiekje of het hele elastiekje?
\ De kleine fluit of de grote tuba? \ De dunste of de dikste
gitaarsnaar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:6}
Neurie met je vingers op je keel, en stop dan met neuriën. Wat voelen
je vingers, en precies wanneer houdt dat gevoel op?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:7}
Vertel in drie stappen de reis van een klap, van de handen van je
vriend naar jouw oor.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:8}
Waarom moeten heel harde \omterm{def:g3:sound-around-us:sound}{geluiden} bij oren vandaan blijven? Welk
piepklein ding in het oor loopt gevaar, en waar lijkt het uit dit
hoofdstuk op?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sound-around-us:9}
De kam van een speeldoos heeft tanden van verschillende lengtes. Welke
tanden spelen de hoge noten en welke de lage? Antwoord met de regel
van \cref{ex:g3:sound-around-us:pitch}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sound-around-us:10}
Giet water in een glazen fles en blaas over de opening: een toeter!
Doe er meer water bij en blaas opnieuw: de toon wordt hoger. Wat trilt
er om die toon te maken --- het glas, het water, of de \omterm{def:g2:air-around-us:air}{lucht} erin?
Verdedig je antwoord met de regel dat kort hoog zingt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sound-around-us:11}
Maya zegt: ``Als de televisie op stil staat, beweegt het scherm nog
steeds --- dus beweging alleen moet \omterm{def:g3:sound-around-us:sound}{geluid} maken.'' Leg met
\cref{def:g3:sound-around-us:sound} en de rijstproef uit wat voor
beweging \omterm{def:g3:sound-around-us:sound}{geluid} maakt, en waarom de beelden op een scherm er geen
maken.
\end{exercise}
