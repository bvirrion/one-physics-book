\chapter{Een eerste stroomkring}\label{ch:g3:first-electric-circuit}

Klik --- de zaklamp werpt haar bundel door de donkere kamer. In dat
plastic buisje wonen een \omterm{def:g3:first-electric-circuit:battery}{batterij}, een \omterm{def:g3:first-electric-circuit:bulb}{lampje} en twee strookjes metaal,
die een spel met strenge regels spelen. Vandaag halen we dat spel uit
elkaar: met een \omterm{def:g3:first-electric-circuit:battery}{batterij}, een \omterm{def:g3:first-electric-circuit:bulb}{lampje} en twee draadjes maken we met onze
eigen handen \omterm{def:g1:light-and-shadows:source}{licht}.

\section{Een lampje, een batterij, twee draadjes}

\begin{definition}[De batterij]\label{def:g3:first-electric-circuit:battery}
Een \emph{batterij}\index{batterij} is een kleine voorraad
elektriciteit. Ze heeft twee verschillende uiteinden, haar
\emph{polen}\index{pool} genoemd: de ene gemerkt met $+$ (plus), de
andere met $-$ (min). Beide uiteinden doen ertoe --- elektriciteit
speelt pas mee als die twee allebei meedoen.
\end{definition}

\begin{definition}[Het lampje]\label{def:g3:first-electric-circuit:bulb}
Een \emph{lampje}\index{lampje} is een klein lampje met een dun
draadje in zijn glas dat witheet gloeit als er elektriciteit
doorheen stroomt. Kijk goed naar zijn metalen voet: het lampje heeft
ook \emph{twee} contactplekken --- het puntje helemaal onderaan, en de
metalen zijkant van de voet. Twee \omterm{def:g3:first-electric-circuit:battery}{polen} op de \omterm{def:g3:first-electric-circuit:battery}{batterij}, twee
contacten op het lampje: dat is geen toeval.
\end{definition}

\begin{method}[Het lampje laten branden]\label{met:g3:first-electric-circuit:light}
Met één \omterm{def:g3:first-electric-circuit:battery}{batterij}, één \omterm{def:g3:first-electric-circuit:bulb}{lampje} en twee draadjes:
\begin{enumerate}
  \item maak één draadje vast van de $+$\omterm{def:g3:first-electric-circuit:battery}{pool} van de \omterm{def:g3:first-electric-circuit:battery}{batterij} aan het
        puntje van de voet van het \omterm{def:g3:first-electric-circuit:bulb}{lampje};
  \item maak het andere draadje vast van de $-$\omterm{def:g3:first-electric-circuit:battery}{pool} aan de metalen
        zijkant van de voet;
  \item het \omterm{def:g3:first-electric-circuit:bulb}{lampje} gaat branden --- je hebt een \omterm{def:g3:first-electric-circuit:circuit}{stroomkring} gebouwd!
\end{enumerate}
Blijft het donker, kijk dan naar de gebruikelijke verdachten: een
draadje dat het glas raakt in plaats van het metalen contact, een losse
verbinding, of een lege \omterm{def:g3:first-electric-circuit:battery}{batterij}.
\end{method}

\section{De stroomkring moet gesloten zijn}

\begin{definition}[Stroomkring]\label{def:g3:first-electric-circuit:circuit}
Een \emph{stroomkring}\index{stroomkring} is een volledige lus voor
elektriciteit: uit de ene \omterm{def:g3:first-electric-circuit:battery}{pool} van de \omterm{def:g3:first-electric-circuit:battery}{batterij}, door de draadjes en
het \omterm{def:g3:first-electric-circuit:bulb}{lampje}, en terug in de andere \omterm{def:g3:first-electric-circuit:battery}{pool}. Is de lus compleet, dan
zeggen we dat de kring \emph{gesloten}\index{gesloten stroomkring} is;
zit er ergens een onderbreking, dan is hij
\emph{open}\index{open stroomkring}.
\end{definition}

\begin{proposition}[De luswet]\label{prop:g3:first-electric-circuit:loop}
Het \omterm{def:g3:first-electric-circuit:bulb}{lampje} brandt alleen als de kring gesloten is --- één
ononderbroken lus van de ene \omterm{def:g3:first-electric-circuit:battery}{pool}, door het \omterm{def:g3:first-electric-circuit:bulb}{lampje}, terug naar de
andere \omterm{def:g3:first-electric-circuit:battery}{pool}. Eén enkele onderbreking, waar dan ook in de lus, en het
\omterm{def:g3:first-electric-circuit:bulb}{lampje} gaat uit: elektriciteit springt niet over onderbrekingen, en
ze stroomt niet in doodlopende einden.
\end{proposition}

\begin{example}[Waarom allebei de polen?]\label{ex:g3:first-electric-circuit:bothends}
Verbind \emph{twee} draadjes van de $+$\omterm{def:g3:first-electric-circuit:battery}{pool} van de \omterm{def:g3:first-electric-circuit:battery}{batterij} met de
twee contacten van het \omterm{def:g3:first-electric-circuit:bulb}{lampje} en laat $-$ met rust: niets. Eén
draadje van $+$ en één van $-$, maar allebei naar het puntje van het
\omterm{def:g3:first-electric-circuit:bulb}{lampje}: opnieuw niets. Het \omterm{def:g3:first-electric-circuit:bulb}{lampje} brandt pas als de lus de twee
verschillende \omterm{def:g3:first-electric-circuit:battery}{polen} van de \omterm{def:g3:first-electric-circuit:battery}{batterij} \emph{én} de twee verschillende
contacten van het \omterm{def:g3:first-electric-circuit:bulb}{lampje} aandoet --- een volledige rondreis.
\end{example}

\begin{example}[De schakelaar]\label{ex:g3:first-electric-circuit:switch}
Een \emph{schakelaar}\index{schakelaar} is een deur in de kring: een
klein metalen bruggetje dat de klik neerlegt (gesloten --- de lus is
compleet, \omterm{def:g1:light-and-shadows:source}{licht}!) of optilt (een onderbreking --- donker). De knop van
de zaklamp, de lichtschakelaar aan de muur, de knop van een deurbel:
allemaal zijn het deuren in een lus. Je kunt er zelf een bouwen van
een wasknijper en een strookje aluminiumfolie dat je om zijn bekken
vouwt.
\end{example}

\begin{remark}[Nette tekeningen]\label{rem:g3:first-electric-circuit:symbols}
In plaats van elke bobbel van de \omterm{def:g3:first-electric-circuit:battery}{batterij} en elke schroefdraad van het
\omterm{def:g3:first-electric-circuit:bulb}{lampje} te tekenen, gebruiken elektriciens \emph{symbolen}: twee
ongelijke streepjes voor de \omterm{def:g3:first-electric-circuit:battery}{batterij} (het lange is $+$), een cirkel met
een kruis voor de lamp, een klein ophaalbruggetje voor de
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}, en rechte lijnen voor de draadjes. Een symbolentekening
van een kring is sneller getekend, en iedereen ter wereld kan hem
lezen --- die tekeningen zullen we nog jarenlang gebruiken.
\end{remark}

\begin{remark}[Batterij ja, stopcontact nooit]\label{rem:g3:first-electric-circuit:safety}
Alles in dit hoofdstuk wordt gespeeld met kleine ronde of platte
\omterm{def:g3:first-electric-circuit:battery}{batterijen}: hun duw is minuscuul en veilig om mee te proeven. Het
\emph{stopcontact} is een andere wereld --- zijn duw is honderden
keren machtiger, genoeg om te verwonden of te doden. Steek nooit
draden, vingers of wat dan ook in een stopcontact, en haal nooit iets
uit elkaar dat op een stopcontact wordt aangesloten.
Batterijspelletjes: ja. Stopcontact: nooit.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:1}
Noem de twee \omterm{def:g3:first-electric-circuit:battery}{polen} van een \omterm{def:g3:first-electric-circuit:battery}{batterij} en de twee contactplekken van een
\omterm{def:g3:first-electric-circuit:bulb}{lampje}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:2}
Wat is een \omterm{def:g3:first-electric-circuit:circuit}{gesloten stroomkring}? Wat is een \omterm{def:g3:first-electric-circuit:circuit}{open stroomkring}? In
welke van de twee brandt het \omterm{def:g3:first-electric-circuit:bulb}{lampje}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:3}
Lea verbindt de $+$\omterm{def:g3:first-electric-circuit:battery}{pool} van de \omterm{def:g3:first-electric-circuit:battery}{batterij} met één draadje met het
puntje van het \omterm{def:g3:first-electric-circuit:bulb}{lampje} en houdt het daarbij. Het \omterm{def:g3:first-electric-circuit:bulb}{lampje} blijft donker.
Wat ontbreekt er, in de woorden van de luswet?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:4}
Welke taak heeft een \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} in de lus? Wat ligt er in een
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar} als hij dicht is?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:5}
Teken met de nette symbolen van
\cref{rem:g3:first-electric-circuit:symbols} een kring met één
\omterm{def:g3:first-electric-circuit:battery}{batterij}, één gesloten \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} en één lamp. Welk symbool is wat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:6}
De zaklamp gaat midden op een kampeertocht uit. Noem drie mogelijke
onderbrekingen of storingen om na te kijken, uit
\cref{met:g3:first-electric-circuit:light}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:7}
Waar zit in de deurbelkring de ``\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}''? Waarom rinkelt de bel
alleen zolang je op de knop drukt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:8}
Waarom mag je nooit iets in een stopcontact steken, terwijl spelen met
een \omterm{def:g3:first-electric-circuit:battery}{batterij} prima is? (Zie
\cref{rem:g3:first-electric-circuit:safety}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:first-electric-circuit:9}
Een \omterm{def:g3:first-electric-circuit:bulb}{lampje} met zijn puntje op de $+$\omterm{def:g3:first-electric-circuit:battery}{pool} van de \omterm{def:g3:first-electric-circuit:battery}{batterij}
gehouden, en één draadje van de $-$\omterm{def:g3:first-electric-circuit:battery}{pool} tegen de metalen zijkant van
de voet van het \omterm{def:g3:first-electric-circuit:bulb}{lampje} gedrukt: gaat het branden? Volg de lus met je
vinger om te beslissen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:10}
Sam bouwt een lus: \omterm{def:g3:first-electric-circuit:battery}{batterij}, draadje, \omterm{def:g3:first-electric-circuit:bulb}{lampje}, draadje, terug naar de
\omterm{def:g3:first-electric-circuit:battery}{batterij} --- maar hij maakt beide draadjes vast aan dezelfde
\omterm{def:g3:first-electric-circuit:battery}{pool}, $-$, ``omdat die dichterbij was''. Het \omterm{def:g3:first-electric-circuit:bulb}{lampje} blijft donker
hoewel de lus compleet lijkt. Leg met
\cref{ex:g3:first-electric-circuit:bothends} uit wat zijn lus mist.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:first-electric-circuit:11}
In een sliert feestlichtjes ontbreekt één \omterm{def:g3:first-electric-circuit:bulb}{lampje}, en de \emph{hele}
sliert is donker, hoewel alle andere \omterm{def:g3:first-electric-circuit:bulb}{lampjes} in orde zijn. Wat
vertelt dat je over hoe de \omterm{def:g3:first-electric-circuit:bulb}{lampjes} in de lus zijn opgesteld? (Het
hoofdstuk over \omterm{def:g3:first-electric-circuit:circuit}{stroomkringen} van volgend jaar geeft die opstelling een
naam.)
\end{exercise}
