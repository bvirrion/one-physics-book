\chapter{Massa meten}\label{ch:g3:measuring-mass}

Op de markt: ``Een \omterm{def:g3:measuring-mass:units}{kilogram} appels, alstublieft!'' De verkoper laat
appels in de schaal rollen tot de weegschaal tevreden is. Wat heeft de
klant precies besteld --- en hoe weet de weegschaal wanneer ze moet
stoppen? Met hefbomen en balansweegschalen in onze gereedschapskist
zijn we klaar om iets nieuws te \omterm{def:g2:measuring-length:measure}{meten}: \omterm{def:g3:measuring-mass:mass}{massa}.

\section{Massa: hoeveel stof}

\begin{definition}[Massa]\label{def:g3:measuring-mass:mass}
De \emph{massa}\index{massa} van een ding vertelt uit hoeveel stof het
gemaakt is. Hoe meer stof, hoe groter de massa --- en hoe harder de
Aarde eraan omlaag trekt, waardoor dingen met meer massa zwaarder
aanvoelen in je hand en hun schaal van een \omterm{def:g3:levers-and-scales:scale}{balansweegschaal} omlaag
drukken.
\end{definition}

\begin{example}[Groot is niet zwaar]\label{ex:g3:measuring-mass:bignotcheavy}
Een kussen vult je hele armen, en toch draag je het moeiteloos. Een
steen die kleiner is dan je vuist belast je pols. Het kussen is groot
maar bevat weinig stof --- vooral veren en \omterm{def:g2:air-around-us:air}{lucht}; de steen is klein
maar volgepakt met stof. \omterm{def:g3:measuring-mass:mass}{Massa} gaat over \emph{hoeveel stof}, niet over
hoeveel ruimte iets inneemt. (Die twee ideeën, en de lastige dans
ertussen, komen ooit terug in het verhaal van \omterm{def:g1:floating-and-sinking:words}{drijven} en \omterm{def:g1:floating-and-sinking:words}{zinken}.)
\end{example}

\begin{definition}[Gram en kilogram]\label{def:g3:measuring-mass:units}
\omterm{def:g3:measuring-mass:mass}{Massa} wordt gemeten in \emph{gram}\index{gram} (geschreven
\unit{g}) en \emph{kilogram}\index{kilogram} (geschreven \unit{kg}):
\[
\qty{1}{kg} = \qty{1000}{g}.
\]
Een paperclip heeft een \omterm{def:g3:measuring-mass:mass}{massa} van ongeveer \qty{1}{g}; een appel
ongeveer \qty{150}{g}; een pak melk ongeveer \qty{1}{kg}; jij een
paar tientallen kilogram. Net als de \omterm{def:g2:measuring-length:units}{meter} zijn deze eenheden voor
iedereen op Aarde hetzelfde.
\end{definition}

\section{Wegen met een balansweegschaal}

\begin{method}[Wegen met blokjes]\label{met:g3:measuring-mass:weigh}
Om de \omterm{def:g3:measuring-mass:mass}{massa} van een meloen te \omterm{def:g2:measuring-length:measure}{meten} met een \omterm{def:g3:levers-and-scales:scale}{balansweegschaal} en een
doos gemerkte metalen blokjes (\qty{500}{g}, \qty{200}{g},
\qty{100}{g}, \qty{50}{g}, \dots):
\begin{enumerate}
  \item controleer of de lege weegschaal waterpas hangt;
  \item leg de meloen in de ene schaal;
  \item leg blokjes in de andere schaal, de grootste bruikbare eerst,
        tot de schalen waterpas hangen;
  \item tel de blokjes op: hun totaal is de \omterm{def:g3:measuring-mass:mass}{massa} van de meloen.
\end{enumerate}
Hangen de schalen waterpas bij een blokje van \qty{500}{g}, een van
\qty{200}{g} en een van \qty{100}{g}, dan is de \omterm{def:g3:measuring-mass:mass}{massa} van de meloen
$500 + 200 + 100 = \qty{800}{g}$.
\end{method}

\begin{example}[Andere weegschalen]\label{ex:g3:measuring-mass:otherscales}
De keukenweegschaal en de personenweegschaal hebben geen tweede
schaal: je legt de lading erop, en een naald draait (of een getal
\omterm{def:g1:light-and-shadows:source}{licht} op) om de \omterm{def:g3:measuring-mass:mass}{massa} rechtstreeks te tonen. Binnenin doet een
ingedrukte veer het vergelijken voor je. Handig en snel --- maar de
oude balans met twee schalen verbergt helemaal geen machinerie: alleen
de hefboomregel die je al begrijpt.
\end{example}

\begin{example}[De markt lezen]\label{ex:g3:measuring-mass:market}
``Een \omterm{def:g3:measuring-mass:units}{kilogram} appels'' betekent: appels tot de weegschaal in
\omterm{def:g2:balance-equilibrium:equilibrium}{evenwicht} is met een blok van \qty{1}{kg} --- ongeveer $6$ of $7$
appels. Meel komt in zakken van \qty{1}{kg}, repen chocola zijn vaak
\qty{100}{g}, een brief heeft ongeveer \qty{20}{g} papier nodig, en de
regel van de postbode ``onder de \qty{20}{g}, één postzegel'' is een
massaregel.
\end{example}

\section{Massa volgt de stof}

\begin{proposition}[Dezelfde stof, dezelfde massa]\label{prop:g3:measuring-mass:conserved}
De \emph{vorm} van een ding veranderen verandert zijn \omterm{def:g3:measuring-mass:mass}{massa} niet. Rol
een bal deeg tot een slang, plet hem tot een pannenkoek, of snijd hem
in vieren en weeg alle stukken samen: de weegschaal meldt elke keer
dezelfde \omterm{def:g3:measuring-mass:mass}{massa}. \omterm{def:g3:measuring-mass:mass}{Massa} telt de stof --- en platdrukken, rollen en
snijden voegen geen stof toe en halen er niets af.
\end{proposition}

\begin{remark}[De weegschaal negeert de schijn]\label{rem:g3:measuring-mass:appearances}
Dit is de stille \omterm{def:g2:pushes-and-pulls:force}{kracht} van het \omterm{def:g2:measuring-length:measure}{meten}: ogen vinden een platgedrukte
pannenkoek ``groter'' dan de bal waar hij van komt, en een opgeschud
kussen ``zwaarder'' dan een steen --- maar de \omterm{def:g3:levers-and-scales:scale}{balansweegschaal}
rapporteert, net als de \omterm{def:g3:temperature-thermometers:temperature}{thermometer} en de liniaal daarvoor, de stof en
niets anders. Eén voor één vervangen onze instrumenten ``het lijkt''
door ``het is''.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:1}
Wat vertelt de \omterm{def:g3:measuring-mass:mass}{massa} van een ding? Met welke twee eenheden meet je
haar, en hoeveel \omterm{def:g3:measuring-mass:units}{gram} gaan er in één \omterm{def:g3:measuring-mass:units}{kilogram}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:2}
Kies de waarschijnlijkste \omterm{def:g3:measuring-mass:mass}{massa}: een paperclip --- \qty{1}{g} of
\qty{1}{kg}? Een appel --- \qty{150}{g} of \qty{15}{kg}? Een pak melk
--- \qty{1}{g} of \qty{1}{kg}? Een kat --- \qty{4}{kg} of
\qty{400}{kg}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:3}
De schalen hangen waterpas als een pompoen tegenover een blokje van
\qty{500}{g} en twee blokjes van \qty{200}{g} staat. Wat is de
\omterm{def:g3:measuring-mass:mass}{massa} van de pompoen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:4}
Waarom moet de lege weegschaal vóór elke weging waterpas hangen?
(Denk aan de bedrieglijke koopman.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:5}
Een strandbal is veel groter dan een blik soep, en toch drukt het blik
zijn schaal omlaag. Leg dat uit met
\cref{ex:g3:measuring-mass:bignotcheavy}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:6}
Hoeveel \omterm{def:g3:measuring-mass:units}{gram} zit er in \qty{2}{kg}? In een halve \omterm{def:g3:measuring-mass:units}{kilogram}? Wat is
meer: \qty{1200}{g} of \qty{1}{kg}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:7}
Een reep chocola van \qty{100}{g} wordt in al zijn blokjes gebroken,
en alle blokjes gaan samen op de weegschaal. Welke \omterm{def:g3:measuring-mass:mass}{massa} laat de
weegschaal zien, en welke wet zegt dat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:8}
De keukenweegschaal en de balans met twee schalen wegen allebei een
zak meel. Wat gebruikt elk van beide om te vergelijken --- en welke
werkt volgens een regel die je drie hoofdstukken geleden hebt geleerd?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measuring-mass:9}
De bakker heeft \qty{800}{g} meel nodig en heeft blokjes van
\qty{500}{g}, \qty{200}{g}, \qty{200}{g}, \qty{100}{g} en
\qty{50}{g}. Zoek twee verschillende manieren om precies \qty{800}{g}
af te wegen --- wetend dat blokjes in \emph{allebei} de schalen mogen
liggen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:10}
Nina weegt een hele sinaasappel: \qty{200}{g}. Ze pelt hem en weegt de
gepelde sinaasappel alleen: \qty{160}{g}. Heeft
\cref{prop:g3:measuring-mass:conserved} gefaald? Waar zijn de
ontbrekende grammen, en hoe zou Nina dat met één extra weging kunnen
bewijzen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measuring-mass:11}
Een raadsel, dat nu eindelijk makkelijk voor je is: wat heeft meer
\omterm{def:g3:measuring-mass:mass}{massa}, een \omterm{def:g3:measuring-mass:units}{kilogram} veren of een \omterm{def:g3:measuring-mass:units}{kilogram} ijzer? En wat neemt meer
ruimte in? Leg uit waarom zoveel mensen de eerste vraag verkeerd
beantwoorden.
\end{exercise}
