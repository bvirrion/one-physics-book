\chapter{مخطّطات الحركة والسرعة المتوسّطة}\label{ch:g7:motion-average-speed}

قبل سنتين كانت \omterm{def:g7:motion-average-speed:formula}{السرعة} تعني ``الكيلومترات المقطوعة في ساعة
واحدة''، معدودة على الأصابع. أما الآن، وأنت مسلّح بالحروف
والمخطّطات، فتنبت للفكرة أنياب: صيغة تحسب في ثلاثة اتّجاهات،
وصورة تُظهر رحلة كاملة في نظرة واحدة --- وفخّ مشهور في المتوسّطات
يقع فيه الكبار كل يوم.

\section{الصيغة}

\begin{definition}[السرعة، بالصيغة]\label{def:g7:motion-average-speed:formula}
لرحلة (أو لجزء منها) تُقطع بوتيرة ثابتة، \emph{السرعة}\index{سرعة}
$v$ هي المسافة $d$ مقسومة على زمن السير $t$:
\[
v = \frac{d}{t}.
\]
فإن كانت $d$ بالكيلومترات و$t$ \omterm{ex:g2:measuring-time:clock}{بالساعات}، نطقت $v$
\emph{بالكيلومتر في الساعة} (\unit{km/h})؛ وبالأمتار والثواني،
\emph{\omterm{def:g2:measuring-length:units}{بالمتر} في الثانية} (\unit{m/s}) --- وهي المفضّلة عند
العالِم. وهذه الصيغة، كصيغة \omterm{def:g7:volume-mass-density:formula}{الكثافة}، تحسب في الاتّجاهات الثلاثة
كلها: $v = d/t$، \ و$d = v \times t$، \ و$t = d/v$.
\end{definition}

\begin{example}[الوصفات في العمل]\label{ex:g7:motion-average-speed:recipes}
قطار يقطع \qty{240}{km} في $2$ ساعة: $v = 240 \div 2 =
\qty{120}{km/h}$. وكم يقطع بهذه الوتيرة في $5$ ساعات؟ $d = 120
\times 5 = \qty{600}{km}$. وكم يستغرق \qty{300}{km}؟ $t = 300
\div 120 = 2.5$ ساعة --- ساعتان وثلاثون دقيقة (وانتبه للمتمرّد:
$0.5$ ساعة هي $30$ دقيقة، لا ``$50$'' أبدًا).
\end{example}

\begin{example}[وحدتان، وسرعة واحدة]\label{ex:g7:motion-average-speed:units}
عدّاء يجري \qty{100}{m} في \qty{10}{s}: $v = \qty{10}{m/s}$.
فكم تساوي بوحدات الطريق؟ في ساعة واحدة --- $3600$ ثانية ---
يقطع العدّاء $10 \times 3600 = \qty{36000}{m} = \qty{36}{km}$:
أي $\qty{10}{m/s} = \qty{36}{km/h}$. وسعر الصرف العامّ: كل
\unit{m/s} تساوي \qty{3.6}{km/h} --- لأن الساعة تحمل $3600$
ثانية والكيلومتر لا يحمل إلا $1000$ \omterm{def:g2:measuring-length:units}{متر}. وأمّا \qty{340}{m/s}
\omterm{def:g3:sound-around-us:sound}{للصوت}، بوحدات الطريق: فأكثر من \qty{1200}{km/h}.
\end{example}

\section{الرحلة صورةً}

\begin{proposition}[قوانين المخطّط]\label{prop:g7:motion-average-speed:graphs}
مثّل المسافة المقطوعة في مقابل الزمن، تكتب الحركةُ سيرتها
الذاتية:
\begin{enumerate}
  \item \omterm{def:g6:describing-motion:uniform}{الحركة المنتظمة} ترسم خطًّا \emph{مستقيمًا} --- مسافات
        متساوية في أزمنة متساوية، خطوة بعد خطوة؛
  \item وكلما كان الخطّ \emph{أشدّ ميلًا} كانت الحركة أسرع:
        فالميل هو \omterm{def:g7:motion-average-speed:formula}{السرعة} مرئيةً؛
  \item والجزء \emph{المستوي} توقّف: يمضي الزمن وتقف المسافة.
\end{enumerate}
والانحناءات تحدّث عن التغيّر: فالانحناء صعودًا تسارع، والاستواء
التدريجي تباطؤ.
\end{proposition}

\begin{method}[قراءة مخطّط المسافة والزمن]\label{met:g7:motion-average-speed:read}
\begin{enumerate}
  \item تحقّق من المحورين ووحدتيهما أولًا --- دقائق أم ساعات،
        أمتار أم كيلومترات؛
  \item قسّم الخطّ عند انحناءاته إلى أجزاء، وسمِّ كل جزء:
        مستقيم شديد الميل، مستقيم ليّن الميل، مستوٍ؛
  \item ولأيّ جزء مستقيم، اقرأ ارتفاعه وامتداده --- المسافة
        المكتسبة والزمن المستغرق --- ثم اقسم: فتلك \omterm{def:g7:motion-average-speed:formula}{سرعة} الجزء؛
  \item وللقصة كلها، اقرأ المسافة الكلّية عند الزمن الأخير ---
        وتذكّر أن الأجزاء المستوية جزء من الزمن الكلّي.
\end{enumerate}
\end{method}

\begin{example}[الساعي، مفكوك الشيفرة]\label{ex:g7:motion-average-speed:decoded}
طبّق الطريقة على الشكل. الجزء الأول: \qty{10}{km} في $20$
دقيقة --- ثلث ساعة --- أي $v = 10 \div \tfrac{1}{3} =
\qty{30}{km/h}$. والثاني: مستوٍ من الدقيقة $20$ إلى $35$ ---
توقّف تسليم. والثالث: \qty{10}{km} في $25$ دقيقة، أي دون
\qty{24}{km/h} بقليل. والمجموع: \qty{20}{km} في ساعة واحدة ---
وهو ما يقدّم لنا فكرة الفصل التالية على طبق.
\end{example}

\section{السرعة المتوسّطة --- وفخّها المشهور}

\begin{definition}[السرعة المتوسّطة]\label{def:g7:motion-average-speed:average}
\emph{السرعة المتوسّطة}\index{سرعة متوسّطة} لرحلة كاملة هي
المسافة الكلّية مقسومة على الزمن \emph{الكلّي} --- بما فيه
التوقّفات:
\[
v_{\text{المتوسّط}} = \frac{d_{\text{الكلّية}}}{t_{\text{الكلّي}}}.
\]
وهي الوتيرة الثابتة الوحيدة التي كانت ستقطع الطريق نفسه في
الزمن الإجمالي نفسه --- \qty{20}{km} للساعي في ساعة واحدة:
متوسّط \qty{20}{km/h}، وإن لم تدر العجلات مرة واحدة بتلك
\omterm{def:g7:motion-average-speed:formula}{السرعة}.
\end{definition}

\begin{proposition}[الفخّ]\label{prop:g7:motion-average-speed:trap}
\omterm{def:g7:motion-average-speed:average}{السرعة المتوسّطة} لرحلة \emph{ليست}، على العموم، منتصف سرعاتها.
فالأجزاء البطيئة تلتهم من الزمن أكثر مما تلتهم السريعة، فتثقل
في المتوسّط أكثر. ولا يُوثق إلا بالوصفة الكاملة --- المسافة
الكلّية على الزمن الكلّي؛ وأمّا حساب متوسّط أعداد \omterm{def:g7:motion-average-speed:formula}{السرعة} نفسها
فأشدّ الخطوات الخاطئة إغراءً في هذا الفصل.
\end{proposition}

\begin{example}[الفخّ، منقضًّا]\label{ex:g7:motion-average-speed:sprung}
درّاج يقطع \qty{60}{km} ذهابًا \omterm{def:g7:motion-average-speed:formula}{بسرعة} \qty{30}{km/h}، ثم
\qty{60}{km} نفسها عودةً \omterm{def:g7:motion-average-speed:formula}{بسرعة} \qty{20}{km/h}. ``المتوسّط:
\qty{25}{km/h}''؟ تحقّق بأمانة. الذهاب: $t = 60 \div 30 = 2$
ساعة. والعودة: $t = 60 \div 20 = 3$ ساعات. والرحلة كلها:
\qty{120}{km} في $5$ ساعات ---
$v_{\text{المتوسّط}} = \qty{24}{km/h}$. فالنصف البطيء طالب بثلاث
ساعات من الخمس، وجرّ المتوسّط دون المنتصف. وكلما أسرعت في نصف،
قلّ الزمن الذي يوجد فيه ذلك النصف أصلًا.
\end{example}

\begin{remark}[ما يعرفه عدّاد السرعة]\label{rem:g7:motion-average-speed:speedometer}
\omterm{def:g7:motion-average-speed:average}{السرعة المتوسّطة} تصف رحلة كاملة؛ أما إبرة عدّاد \omterm{def:g7:motion-average-speed:formula}{السرعة} فتجيب عن
سؤال آخر --- \emph{كم \omterm{def:g7:motion-average-speed:formula}{السرعة} الآن}. وعلى المخطّط، ``الآن'' يسكن
في ميل الخطّ \emph{عند نقطة واحدة} --- سهل الرؤية على جزء
مستقيم، دقيق حيث ينحني الخطّ. وجعل ``الميل عند نقطة'' دقيقًا هو
من أعظم اختراعات الرياضيات، وهو ينتظرك في آخر الثانوية. وحتى
ذلك الحين: الأجزاء المستقيمة تنال أعدادًا، والانحناءات تنال
حكايات.
\end{remark}

\section{مسألة: جمعة الساعي}

\begin{problem}[{مسألة نهاية الأسبوع --- ساعي درّاجة واحد، وجمعة
مدينة واحدة؛ مخطّط المرسِل، والفخّ في ورقة
المكافأة}]\label{pb:g7:motion-average-speed:1}
يسجّل المرسِل جمعة الساعية لينا على مخطّط مسافة وزمن، ويحسم
أجرها منه. وأنت تدقّق الورقة.

\medskip\noindent\textbf{القسم الأول --- الصباح، من المخطّط.}
يُظهر المخطّط: صعودًا مستقيمًا من ($0$ دقيقة، \qty{0}{km}) إلى
($30$ دقيقة، \qty{9}{km})؛ ثم استواءً حتى الدقيقة $45$؛ ثم
مستقيمًا من هناك إلى ($75$ دقيقة، \qty{15}{km})؛ ثم استواءً حتى
الدقيقة $90$.
\begin{enumerate}
  \item احكِ قصة الصباح بالكلام: الأجزاء، والتوقّفات، وما كانت
        التوقّفات على الأرجح.
  \item ما \omterm{def:g7:motion-average-speed:formula}{السرعة} على الجزء الأول، \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{km/h}؟
  \item وما \omterm{def:g7:motion-average-speed:formula}{السرعة} على جزء السير الثاني؟
  \item وما \emph{متوسّط} \omterm{def:g7:motion-average-speed:formula}{سرعة} لينا على صباح الدقائق التسعين
        كله، بما فيه التوقّفات؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- بعد الظهر، من
الصيغة.}
\begin{enumerate}[resume]
  \item مرحلة ما بعد الظهر الأولى: \qty{12}{km} بوتيرة ثابتة
        مقدارها \qty{24}{km/h}. كم استغرقت، بالدقائق؟
  \item المرحلة الثانية: تسليم في أعالي المدينة،
        \qty{20}{minutes} \omterm{def:g7:motion-average-speed:formula}{بسرعة} \qty{18}{km/h}. كم كيلومترًا؟
  \item المرحلة الثالثة: السير الطويل إلى المستودع،
        \qty{10}{km}، أُنجز في $25$ دقيقة. ما \omterm{def:g7:motion-average-speed:formula}{سرعة} الوتيرة
        الثابتة \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{km/h}؟
  \item مجموع ما بعد الظهر: اجمع المسافات والأزمنة (المراحل
        وحدها، بلا استراحات): أيّ \omterm{def:g7:motion-average-speed:average}{سرعة متوسّطة} حفظتها العجلات
        وهي دائرة؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- فخّ ورقة المكافأة.}
قاعدة المكافأة: ``\omterm{def:g7:motion-average-speed:average}{السرعة المتوسّطة} على جولة كاملة فوق
\qty{20}{km/h} تكسب مكافأة الرّاكب السريع.''
\begin{enumerate}[resume]
  \item جمعة لينا كاملة: \qty{37}{km} في $3$ ساعات (بكل
        التوقّفات). أتكسب المكافأة؟
  \item ويطالب بها زميلها ماكس: ``ركبت نصف جولتي \omterm{def:g7:motion-average-speed:formula}{بسرعة}
        \qty{30}{km/h} ونصفها \omterm{def:g7:motion-average-speed:formula}{بسرعة} \qty{15}{km/h} ---
        فالمتوسّط $22.5$!'' فيتحقّق المرسِل: النصفان كلاهما
        \qty{12}{km}. احسب متوسّط ماكس الحقيقي واحكم في
        مكافأته.
  \item اشرح لماكس، بلغة الزمن، أين أخطأ عدده $22.5$.
  \item تقترح لينا قاعدة مكافأة أعدل للسعاة --- قاعدة لا
        تعاقب على توقّفات التسليم. اقترح واحدة (فالمخطّط يعرف
        كيف)، وقل أيّ \omterm{def:g7:motion-average-speed:formula}{سرعة} تقيسها.
\end{enumerate}

\medskip\noindent\textbf{القسم الرابع --- درس المرسِل.}
