\chapter{الظلال والكسوف والخسوف}\label{ch:g7:shadows-eclipses}

مرة أو مرتين في العمر، ولدقائق معدودة، يصير النهار ليلًا: يسودّ
قرص الشمس خلف \omterm{def:g2:moon-in-the-sky:moon}{القمر} وتظهر النجوم عند الظهر. وقد قرأ القدماء في
ذلك نذير شؤم؛ أما أنت فستقرأ هندسة. \omterm{def:g7:shadows-eclipses:solar}{فالكسوف} ليس إلا ظلًّا ---
أكبر \omterm{def:g1:light-and-shadows:shadow}{ظلّ} ستقف فيه يومًا --- وعُدّة المسطرة والأشعّة عندك هي
بالضبط ما يطيعه.

\section{ظلال من مصابيح أمينة}

\begin{proposition}[الظلّ التامّ وشبه الظلّ]\label{prop:g7:shadows-eclipses:umbra}
المنبع \emph{شبه النقطي} يلقي ظلًّا واحدًا حادّ الحافة. أما
المنبع \emph{العريض} --- كرة مصنفرة، أو قرص الشمس كله ---
فيلقي ظلًّا من جزأين:
\begin{enumerate}
  \item \emph{الظلّ التامّ}\index{ظلّ تامّ}: المنطقة التي لا
        يبلغها \omterm{def:g1:light-and-shadows:source}{ضوء} من \emph{أيّ} جزء من المنبع --- ظلمة
        كاملة؛
  \item \emph{شبه الظلّ}\index{شبه الظلّ}: الحاشية التي يبلغها
        \omterm{def:g1:light-and-shadows:source}{ضوء} من \emph{جزء} من المنبع لا غير --- نصف \omterm{def:g1:light-and-shadows:shadow}{ظلّ}، يخفت
        كلما ابتعد.
\end{enumerate}
فمن داخل \omterm{prop:g7:shadows-eclipses:umbra}{الظلّ التامّ} يكون المنبع مختفيًا كله؛ ومن \omterm{prop:g7:shadows-eclipses:umbra}{شبه الظلّ}
مختفيًا جزئيًّا --- يطلّ من فوق حافة الحاجب.
\end{proposition}

\begin{method}[إنشاء الظلّين]\label{met:g7:shadows-eclipses:construct}
مع منبع عريض تحكم الخريطةَ الأشعّةُ الخارجة من \emph{حافّتيه}:
\begin{enumerate}
  \item ارسم أشعّة من حافة المنبع العليا تمسّ جانبَي الحاجب،
        ومثلها من الحافة السفلى؛
  \item فالمنطقة خلف الحاجب التي تخطئها أشعّة \emph{الحافتين}
        معًا هي \omterm{prop:g7:shadows-eclipses:umbra}{الظلّ التامّ}؛
  \item والمنطقتان على الجانبين اللتان تخطئهما أشعّة حافة
        وتبلغهما أشعّة الأخرى هما \omterm{prop:g7:shadows-eclipses:umbra}{شبه الظلّ}؛
  \item وحيث تصل الأشعّة كلها يكون \omterm{def:g1:light-and-shadows:source}{الضوء} كاملًا.
\end{enumerate}
\omterm{prop:g7:shadows-eclipses:umbra}{والظلّ التامّ} لحاجب مستدير مخروط --- وبلوغ ذلك المخروط حاجزًا
بعينه أو عدمُه هو ما يحسم كل شيء في \omterm{def:g7:shadows-eclipses:solar}{الكسوف} \omterm{def:g7:shadows-eclipses:lunar}{والخسوف}.
\end{method}

\section{القمر يدخل الحزمة}

\begin{definition}[كسوف الشمس]\label{def:g7:shadows-eclipses:solar}
يحدث \emph{كسوف الشمس}\index{كسوف} حين يمرّ \omterm{def:g2:moon-in-the-sky:moon}{القمر} بين الشمس
والأرض تمامًا، فيقع ظلّه علينا. وحيث يلمس \emph{\omterm{prop:g7:shadows-eclipses:umbra}{الظلّ التامّ}}
\omterm{def:g2:moon-in-the-sky:moon}{للقمر} الأرضَ --- بقعة راكضة نادرًا ما يتجاوز عرضها مئتَي
كيلومتر --- يختفي قرص الشمس كله: كسوف \emph{كلّي}، وليل عند
الظهر، ونجوم بازغة. وفي \emph{\omterm{prop:g7:shadows-eclipses:umbra}{شبه الظلّ}} الواسع المحيط لا
تُقضَم الشمس إلا جزئيًّا: كسوف \emph{جزئي}.
\end{definition}

\begin{definition}[خسوف القمر]\label{def:g7:shadows-eclipses:lunar}
ويحدث \emph{خسوف القمر}\index{خسوف} حين يبحر \emph{\omterm{def:g2:moon-in-the-sky:moon}{القمر}} في
مخروط \omterm{def:g1:light-and-shadows:shadow}{ظلّ} \emph{الأرض}: فيعتم \omterm{ex:g2:moon-in-the-sky:shapes}{البدر} --- إلى توهّج نحاسي عميق
في الغالب --- ساعةً أو أكثر، مرئيًّا للنصف الليلي من العالم كله
دفعة واحدة. ونحن هذه المرة الحاجب، \omterm{def:g2:moon-in-the-sky:moon}{والقمر} واقف في ظلّنا
التامّ.
\end{definition}

\begin{example}[لماذا هو نادر؟]\label{ex:g7:shadows-eclipses:rare}
المحاق يأتي كل شهر --- فلماذا لا يحدث \omterm{def:g7:shadows-eclipses:solar}{كسوف} كل شهر؟ لأن \omterm{def:g6:sun-earth-moon:axisorbit}{مدار}
\omterm{def:g2:moon-in-the-sky:moon}{القمر} \emph{مائل} بنحو خمس درجات على \omterm{def:g6:describing-motion:trajectory}{مسار} الأرض حول الشمس.
ففي أكثر الشهور يمرّ المحاق \emph{فوق} خطّ الشمس والأرض أو
\emph{تحته}، فينطلق ظلّه في الفضاء بلا ضرر. ولا تصطفّ العوالم
الثلاثة إلا حين يصادف المحاقُ عبورَ \omterm{def:g2:moon-in-the-sky:moon}{القمر} مستوى الاصطفاف ---
وذلك بضع مرات في السنة في مكان ما من الأرض، وأما لبلدة
\emph{بعينها} فالكلّية زائرة مرة في عدة أعمار.
\end{example}

\begin{example}[المصادفة التي تصنع الكلّية]\label{ex:g7:shadows-eclipses:coincidence}
وجدها نموذج ملعبك: الشمس أعرض من \omterm{def:g2:moon-in-the-sky:moon}{القمر} بنحو $400$ مرة ---
وتقف أبعد منه بنحو $400$ مرة. فيتطابق القرصان في السماء تطابقًا
يكاد يكون تامًّا، فيستطيع \omterm{def:g2:moon-in-the-sky:moon}{القمر} أن يغطّي الشمس \emph{بالكاد}:
فالكلّية قصيرة وبقعة ظلّها التامّ صغيرة. وحين يكون \omterm{def:g2:moon-in-the-sky:moon}{القمر} في
الطرف البعيد من مداره البيضوي قليلًا يقصّر قرصه، فتحفّ حلقة
مشتعلة من الشمس \omterm{def:g2:moon-in-the-sky:moon}{بالقمر} الأسود --- وهو \omterm{def:g7:shadows-eclipses:solar}{كسوف} \emph{حلقي}، بديع
ولا يجوز مشاهدته بلا حماية أبدًا.
\end{example}

\begin{remark}[مشاهدة بلا بكاء]\label{rem:g7:shadows-eclipses:safety}
القانون القديم لا استثناء فيه \omterm{def:g7:shadows-eclipses:solar}{للكسوف}: \emph{لا} تنظر إلى الشمس
أبدًا --- مكسوفةً أو محلَّقة أو كاملة --- بعين مجرّدة، ولا
بنظّارة شمس، ولا بأي عدسة. فالشمس المختفية جزئيًّا تحرق
الشبكيّتين كما تحرقهما الكاملة، وبلا ألم. والأجهزة الآمنة
أصدقاء قدامى: ثقب الدبّوس --- اسقط صورة الشمس عبر ثقب صغير على
ورقة وراقب القضمة تكبر في راحة --- أو شجرة وارفة، تنثر آلاف
ثقوبها بين الأوراق أهلّة صغيرة على الأرض. ولا يجوز أن تواجه
الشمس مباشرة إلا نظّارات \omterm{def:g7:shadows-eclipses:solar}{كسوف} معتمَدة، مفحوصة غير متضرّرة؛ وفي
دقائق \emph{الكلّية} وحدها يجوز التأمّل في الإكليل عاريًا ---
وأول شظيّة عائدة تنهي الهدنة.
\end{remark}

\section{مسألة: بعثة الكسوف}

\begin{problem}[{مسألة نهاية الأسبوع --- الصفّ يخطّط لبعثة
الكسوف؛ الهندسة تختار المخيّم، والفيزياء تحزم الحقائب؛
الكلّية، دقيقة دقيقة}]\label{pb:g7:shadows-eclipses:1}
في الربيع القادم يعبر \omterm{def:g7:shadows-eclipses:solar}{كسوف} كلّي للشمس القارّة. ويخطّط الصفّ
لبعثة --- برسوم الأشعّة لا بنشرات السفر.

\medskip\noindent\textbf{الجزء الأول --- اختيار موضع الوقوف.}
يُظهر الأطلس الطريق المتوقَّع: شريط عرضه \qty{160}{km} يعبر
الخريطة، وتحفّ به منطقة \omterm{def:g7:shadows-eclipses:solar}{كسوف} جزئي عرضها آلاف الكيلومترات.
\begin{enumerate}
  \item أيّ منطقة \omterm{def:g1:light-and-shadows:shadow}{ظلّ} يخطّها الشريط، وأيّ منطقة تحفّ به؟
  \item وتقع بلدة الصفّ على \qty{300}{km} من خطّ منتصف الشريط.
