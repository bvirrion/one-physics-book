\chapter{الحجم والكتلة والكثافة}\label{ch:g7:volume-mass-density}

في السنة الماضية فضحت صائغًا غشّاشًا بقسمة الغرامات على
السنتيمترات المكعّبة. وهذه السنة سلّمك منهاج الرياضيات \omterm{def:g2:pushes-and-pulls:force}{قوة}
جديدة: حروفًا تنوب عن الأعداد. \omterm{def:g7:volume-mass-density:formula}{والكثافة} أفضل موضع تُنفق فيه
--- صيغة قصيرة واحدة ترصّ ثلاث وصفات وعُدّة محقّق وقاعدة \omterm{def:g1:floating-and-sinking:words}{الطفو}
كلها في خمسة رموز.

\section{الصيغة}

\begin{definition}[الكثافة، بالصيغة]\label{def:g7:volume-mass-density:formula}
\emph{كثافة}\index{كثافة} مادة، ويُرمز إليها بالحرف $\rho$ (وهو
الحرف اليوناني \emph{رو})، هي كتلتها لكل \omterm{def:g6:measurement-in-science:measuring}{وحدة} \omterm{def:g6:mass-and-volume:volume}{حجم}: فلعيّنة
كتلتها $m$ وحجمها $V$،
\[
\rho = \frac{m}{V}.
\]
فإذا كان $m$ بالغرامات و$V$ بالسنتيمترات المكعّبة خرج $\rho$
\emph{\omterm{def:g3:measuring-mass:units}{بالغرام} لكل \omterm{def:g6:mass-and-volume:volume}{سنتيمتر مكعّب}} (\unit{g/cm^3}) --- وهي
``\omterm{def:g3:measuring-mass:mass}{كتلة} \omterm{def:g6:mass-and-volume:volume}{سنتيمتر مكعّب} واحد'' في السنة الماضية، بزيّها الرسمي
الآن. وكثافة الماء: $\rho = \qty{1.0}{g/cm^3}$. ولأن \omterm{def:g6:mass-and-volume:volume}{اللتر}
$1000$ \omterm{def:g6:mass-and-volume:volume}{سنتيمتر مكعّب}، فالعدد نفسه يصلح \omterm{def:g3:measuring-mass:units}{بالكيلوغرام} لكل \omterm{def:g6:mass-and-volume:volume}{لتر}:
فالماء \qty{1.0}{kg} لكل \omterm{def:g6:mass-and-volume:volume}{لتر}.
\end{definition}

\begin{example}[قراءة الحروف]\label{ex:g7:volume-mass-density:reading}
الصيغة جملة بالاختزال. والعلاقة $\rho = m/V$ تُقرأ: ``لإيجاد \omterm{def:g7:volume-mass-density:formula}{الكثافة}
اقسم \omterm{def:g3:measuring-mass:mass}{كتلة} العيّنة على حجمها.'' والحروف مواضع شاغرة --- فلهذه
الحصاة يصير $m$ هو \qty{75}{g} ويصير $V$ هو \qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
القسمة نفسها التي في السنة الماضية --- لكن الصيغة تحفظ الوصفة
لكل حصاة تأتي.
\end{example}

\begin{proposition}[ثلاث وصفات في صيغة واحدة]\label{prop:g7:volume-mass-density:recipes}
لأن \omterm{def:g7:volume-mass-density:formula}{الكثافة} تربط \omterm{def:g3:measuring-mass:mass}{الكتلة} \omterm{def:g6:mass-and-volume:volume}{والحجم} بتناسب، تخدم صيغة واحدة ثلاث
حاجات:
\begin{enumerate}
  \item \emph{إيجاد \omterm{def:g7:volume-mass-density:formula}{كثافة}}: $\rho = m/V$ --- اقسم \omterm{def:g3:measuring-mass:mass}{الكتلة} على
        \omterm{def:g6:mass-and-volume:volume}{الحجم}؛
  \item \emph{إيجاد \omterm{def:g3:measuring-mass:mass}{كتلة}}: $m = \rho \times V$ --- فكل \omterm{def:g6:mass-and-volume:volume}{سنتيمتر
        مكعّب} يحمل $\rho$ من الغرامات، وعددها $V$؛
  \item \emph{إيجاد \omterm{def:g6:mass-and-volume:volume}{حجم}}: $V = m/\rho$ --- وزّع \omterm{def:g3:measuring-mass:mass}{الكتلة} حصصًا
        قدر كل حصة $\rho$ من الغرامات؛ فعدد الحصص هو عدد
        السنتيمترات المكعّبة.
\end{enumerate}
\end{proposition}

\begin{example}[الوصفات في عملها]\label{ex:g7:volume-mass-density:atwork}
\emph{\omterm{def:g3:measuring-mass:mass}{الكتلة} من \omterm{def:g6:mass-and-volume:volume}{الحجم}}: كم يزن \qty{250}{cm^3} من الحديد
($\rho = 7.9$)؟ $m = 7.9 \times 250 = \qty{1975}{g}$ ---
أي قرب كيلوغرامين في \omterm{def:g6:mass-and-volume:volume}{حجم} فنجان قهوة.
\emph{\omterm{def:g6:mass-and-volume:volume}{والحجم} من \omterm{def:g3:measuring-mass:mass}{الكتلة}}: أيّ حيّز يشغله \qty{540}{g} من
البلّوط ($\rho = 0.6$)؟ $V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{ورفيق السلامة}: أجوبة الحديد يجب أن تكون ثقيلة صغيرة،
وأجوبة الخشب خفيفة كبيرة --- فجدول الكثافات خير صديق لخطوة
الحكم.
\end{example}

\section{قياس الكثافات}

\begin{method}[كثافة سائل]\label{met:g7:volume-mass-density:liquid}
السوائل لا تستقرّ على كفّة ميزان وحدها --- فاستعِر حيلة
التصفير:
\begin{enumerate}
  \item ضع مخبارًا مدرَّجًا فارغًا على الميزان وصفّره؛
  \item واسكب فيه حجمًا مريحًا --- ولنقل $V = \qty{80}{cm^3}$
        من \omterm{def:g3:solids-liquids-gases:liquid}{السائل} الغامض، قارئًا \omterm{met:g6:mass-and-volume:cylinder}{السطح الهلالي} على أصوله؛
  \item فيُظهر الميزان الآن \omterm{def:g3:measuring-mass:mass}{كتلة} \omterm{def:g3:solids-liquids-gases:liquid}{السائل} وحدها: ولنقل
        $m = \qty{68}{g}$؛
  \item واقسم: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ ---
        أخفّ من الماء: فهو زيت على الأرجح.
\end{enumerate}
\end{method}

\begin{example}[كثافة جسم صلب من البداية إلى النهاية]\label{ex:g7:volume-mass-density:solid}
ميدالية تدّعي أنها من فضة ($\rho = 10.5$). الميزان:
$m = \qty{84}{g}$. والمخبار: يصعد الماء من \qty{60.0}{cm^3} إلى
\qty{68.0}{cm^3}، ومنه $V = \qty{8.0}{cm^3}$. ثم
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- فالتوقيع مطابق:
وقد تكون فضة حقًّا. (ومعدن أرخص مكسوّ بطبقة فضة كان سيفضح نفسه
هنا --- إلا أن يُختار بدهاء، كما ستبيّن مسألة نهاية الأسبوع.)
\end{example}

\begin{remark}[الوحدات في الصيغ]\label{rem:g7:volume-mass-density:units}
لا تكون الصيغة أمينة إلا إذا اتّفقت وحداتها. فمع $m$ بالغرامات
و$V$ \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{cm^3} يتكلّم $\rho$ \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{g/cm^3}؛ وأطعِمها
كيلوغرامات ولترات تجب \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{kg/L} --- ولحسن الحظّ بالعدد
نفسه لأي مادة. لكن اخلط الغرامات باللترات فتسلّم الصيغةُ، بلا
شكوى، هراءً. وقاعدة المحترفين: استعرض الوحدات قبل الحساب،
واكتبها في الجواب بعده.
\end{remark}

\section{التفكير بالكثافة}

\begin{example}[السبائك وما بينهما]\label{ex:g7:volume-mass-density:alloys}
امزج مادّتين فتحطّ \omterm{def:g7:volume-mass-density:formula}{كثافة} المزيج \emph{بين} توقيعيهما --- أقرب
إلى المكوّن الأسخى. فالبرونز (نحاس $9.0$ مع قليل من القصدير)
يوقّع قرب $8.8$؛ وخليط التاج من ذهب وفضة وقّع بين $10.5$
و$19.3$؛ وماء البحر، وهو ملح مذاب في ماء، يرتقي إلى نحو $1.03$.
والبينيّة نفسها قرينة: فقراءة $12$ من سبيكة ``ذهب خالص''
اعتراف بصحبة.
\end{example}

\begin{example}[الكثافة تحسم عالَم الطفو]\label{ex:g7:volume-mass-density:floating}
قاعدة \omterm{def:g1:floating-and-sinking:words}{الطفو}، بزيّها الرسمي الآن: \omterm{def:g1:floating-and-sinking:words}{يطفو} الجسم في \omterm{def:g3:solids-liquids-gases:liquid}{سائل} حين يكون
$\rho_{\text{الجسم}} < \rho_{\text{السائل}}$. فالبلّوط ($0.6$)
على الماء ($1.0$): \omterm{def:g1:floating-and-sinking:words}{يطفو}. والثلج ($0.9$) في زيت ($0.85$): \omterm{def:g1:floating-and-sinking:words}{يغرق}
--- فراجع الجدول لا حدسك. وسبّاح ($\approx 1.0$) في البحيرات
الشديدة الملوحة المشهورة ($\approx 1.2$): \omterm{def:g1:floating-and-sinking:words}{يطفو} كالفلّينة،
وجريدته في يده. متراجحة واحدة، والمرفأ كله.
\end{example}

\begin{remark}[انتبه للحشد لا للطرد]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ تصف \emph{الموادّ}؛ أما السفن وسترات النجاة
المنتفخة فهي \emph{طرود} --- مادة مع \omterm{def:g2:air-around-us:air}{هواء} محبوس --- والذي يواجه
قاعدة \omterm{def:g1:floating-and-sinking:words}{الطفو} هو $m/V$ الكلّي للطرد. والصيغة تعالج الاثنين إن
أطعمتها $m$ و$V$ الصحيحين: \omterm{def:g7:volume-mass-density:formula}{كثافة} الفولاذ نفسه للمادة، \omterm{def:g3:measuring-mass:mass}{وكتلة}
البدن كله على \omterm{def:g6:mass-and-volume:volume}{حجم} البدن كله للطرد. وأكثر ``مفارقات'' \omterm{def:g1:floating-and-sinking:words}{الطفو}
ليست إلا خلطًا بين الحسابين.
\end{remark}

\section{مسألة: محقّق ساحة الخردة}

\begin{problem}[{مسألة نهاية الأسبوع --- عصريّة مع محقّق معادن
ساحة الخردة؛ أربع دفعات غامضة، وسبيكة منتحلة، وحدود اختبار
الكثافة}]\label{pb:g7:volume-mass-density:1}
تشتري ساحة الخردة المعدن بما \emph{هو}، لا بما يبدو، ويعمل
محقّق الساحة بميزان وإناء مدرَّج كبير وجدول الكثافات. وأنت
المتدرّب. (مقتطف من الجدول: الألمنيوم $2.7$؛ والحديد $7.9$؛
والنحاس $9.0$؛ والرصاص $11.3$؛ والذهب $19.3$؛ والتنغستن
$19.3$.)

\medskip\noindent\textbf{الجزء الأول --- أربع دفعات.}
\begin{enumerate}
  \item الدفعة A، سبيكة رمادية: $m = \qty{5400}{g}$،
        والإزاحة $V = \qty{2000}{cm^3}$. فما \omterm{def:g7:volume-mass-density:formula}{الكثافة} والهوية؟
  \item والدفعة B، لفّة سلك: $m = \qty{1800}{g}$،
        $V = \qty{200}{cm^3}$. فما \omterm{def:g7:volume-mass-density:formula}{الكثافة} والهوية؟
  \item والدفعة C، لوح ثقيل باهت: $m = \qty{4520}{g}$،
        $V = \qty{400}{cm^3}$. فما \omterm{def:g7:volume-mass-density:formula}{الكثافة} والهوية --- ولماذا
        يجب أن تُعالَج الدفعة C بقفّازين وباحترام؟
  \item والدفعة D، كيس خردة شاحبة مختلطة: $m = \qty{8100}{g}$،
        $V = \qty{1500}{cm^3}$. بيّن أن \omterm{def:g7:volume-mass-density:formula}{كثافة} الدفعة D تحطّ بين
        معدنين من الجدول، وقل ما الذي يحويه الكيس على
        الأرجح.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- الأسعار
والتوقّعات.}
تدفع الساحة \omterm{def:g3:measuring-mass:mass}{بالكتلة} وتخطّط للنقل \omterm{def:g6:mass-and-volume:volume}{بالحجم}.
\begin{enumerate}[resume]
  \item يريد مشترٍ \qty{540}{g} من الألمنيوم تُقطع من الدفعة
        A. فأيّ \omterm{def:g6:mass-and-volume:volume}{حجم} من السبيكة هو ذلك؟
  \item ولفّة النحاس في الدفعة B ستُصهر مكعّبات \omterm{def:g6:mass-and-volume:volume}{حجم} كل واحد
        \qty{25}{cm^3}. فما \omterm{def:g3:measuring-mass:mass}{كتلة} المكعّب الواحد، وكم مكعّبًا
        كاملًا تعطي اللفّة؟
  \item وصندوق يحمل \qty{20}{kg} على الأكثر. فكم مكعّبًا من
        مكعّبات النحاس تلك يجوز أن يحمل؟
  \item وحوض الشاحنة يسع \qty{3000}{cm^3} أخرى من خردة الحديد.
        فأيّ \omterm{def:g3:measuring-mass:mass}{كتلة} إضافية، بالكيلوغرامات، يُسمح للمحقّق
        بتحميلها؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- المنتحل.}
يصل بائع بسبيكة ``ذهب'' لامعة: $m = \qty{3860}{g}$،
والإزاحة $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item احسب \omterm{def:g7:volume-mass-density:formula}{كثافة} السبيكة. أتطابق توقيع الذهب؟
  \item ولا يتأثّر المحقّق، بل يراجع آخر سطر في الجدول ويتنهّد.
        فأيّ معدن أرخص يلبس \omterm{def:g7:volume-mass-density:formula}{كثافة} الذهب بالضبط --- وماذا
        يعلّمنا هذا عن حدود اختبار \omterm{def:g7:volume-mass-density:formula}{الكثافة}؟
  \item وأيّ جهازَي الساحة هُزم هنا: الميزان أم الإناء أم
        كلاهما أم لا هذا ولا ذاك؟ قل بدقّة ما الذي أثبته
        اختبار \omterm{def:g7:volume-mass-density:formula}{الكثافة} \emph{بأمانة} عن السبيكة.
  \item واقترح اختبارًا فيزيائيًّا آخر من سنوات سابقة في هذا
        المنهاج يميّز الذهب من قرينه من غير إتلاف السبيكة.
        (فدرجتا انصهارهما مختلفتان اختلافًا هائلًا --- الذهب
        قرب \qty{1064}{\celsius}، وقرينه فوق كل فرن هنا بكثير
        --- لكن لا ساحة تصهر كنزًا محتملًا: فجِد دليلًا أرفق،
        ولعلّه حكم \omterm{def:g1:magnets:magnet}{المغناطيس} على النوى الحديدية، أو رنين قضيب
        مضروب، ودافع عن اختيارك بأمانة.)
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- صنعة المحقّق.}
\begin{enumerate}[resume]
  \item اكتب عقيدة المحقّق في ثلاثة أسطر: ما الذي تستطيع
        \omterm{def:g7:volume-mass-density:formula}{الكثافة} أن تثبته، وما الذي لا تستطيع إلا أن توحي به،
        وما الذي لا تستطيعه وحدها أبدًا.
\end{enumerate}
\end{problem}
