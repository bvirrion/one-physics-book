\chapter{الدارات: الحلقات والتسلسل والتفرع}\label{ch:g7:circuits-series-parallel}

أربع سنوات من الدارات ملأت عُدّتك: حلقات، ومخططات، والتوصيلتان
العظيمتان. لكن شخصية واحدة عملت طوال ذلك بلا اسم --- ذلك الشيء
الذي يجري. وهذه السنة ينال اسمه واتجاهه وأغرب خصائصه: أنه لا
يُستهلك أبدًا. ومع صعود التيار على الخشبة يكفّ التسلسل والتفرع
عن أن يكونا وصفتين ويصيران قانونين.

\section{التيار يصعد على الخشبة}

\begin{definition}[التيار الكهربائي]\label{def:g7:circuits-series-parallel:current}
\emph{التيار الكهربائي}\index{تيار كهربائي} هو الجريان الذي
يسير في \omterm{def:g3:first-electric-circuit:circuit}{دارة مغلقة}: حشد من جسيمات مشحونة --- كثيرة كثرة لا
تُتخيّل، دقيقة دقّة لا تُتخيّل، من أبناء عمومة العالم الجزيئي
--- يسيّرها دفع \omterm{def:g3:first-electric-circuit:battery}{البطارية} عبر معدن الأسلاك. فلا حلقة، فلا مسير:
والتيار لا يوجد إلا ما دام الطريق مغلقًا. (أما ما هم السائرون
بالضبط، وكيف يدعهم المعدن يعبرون، فقصة جميلة محفوظة للسنوات
الأخيرة من هذا الكتاب.)
\end{definition}

\begin{proposition}[الاتجاه الاصطلاحي]\label{prop:g7:circuits-series-parallel:direction}
باصطلاح عالمي، تعلّم المخططات الكهربائية التيارَ جاريًا
\emph{خارج \omterm{def:g3:first-electric-circuit:battery}{البطارية} من القطب $+$، حول الدارة، إلى القطب $-$}.
وكل سهم على كل مخطط في العالم يطيع هذا الاصطلاح --- وقد اختير
قبل أن يستطيع أحد أن يسأل السائرين أنفسهم بزمن طويل. (وحين
استطاعت الفيزياء أن تسأل أخيرًا، حمل الجواب نكتة صغيرة مشهورة،
مدّخرة لسنة لاحقة.)
\end{proposition}

\begin{proposition}[التيار لا يُستهلك]\label{prop:g7:circuits-series-parallel:notconsumed}
العناصر لا تأكل التيار. فأيّ تيار يدخل مصباحًا أو محرّكًا أو
طنّانًا يغادره كاملًا مواصلًا مسيره: وإنما الذي يأخذه العنصر
\emph{\omterm{def:g5:everyday-energy:energy}{طاقة}} من مخزن \omterm{def:g3:first-electric-circuit:battery}{البطارية}، يسلّمها المسير العابر. ولذلك يجري
في أي حلقة تسلسل تيار واحد بعينه عبر كل عنصر --- ويعود إلى
\omterm{def:g3:first-electric-circuit:battery}{البطارية} بقدر ما غادرها.
\end{proposition}

\begin{example}[الوهم في قفص الاتهام]\label{ex:g7:circuits-series-parallel:trial}
الخطأ المغري: ``\omterm{def:g3:first-electric-circuit:bulb}{المصباح} الأول يستهلك بعض التيار، فلا بدّ أن
يتوهّج الثاني أضعف.'' والدارة نفسها تدحضه. وصّل مصباحين
\emph{متماثلين} \omterm{def:g4:series-circuits:series}{على التسلسل} وانظر: يتوهّجان بسطوع متساوٍ تمامًا
--- وتبديلهما لا يغيّر شيئًا. ولو كان التيار يُستهلك في
الطريق لوجب أن يكون \omterm{def:g3:first-electric-circuit:bulb}{المصباح} الثاني هو الخافت دائمًا، أيّ \omterm{def:g3:first-electric-circuit:bulb}{مصباح}
جلس هناك. ولا يكون كذلك أبدًا. فالتيار يصل كاملًا؛ والذي
يتقاسمه الزوج هو \emph{دفع} \omterm{def:g3:first-electric-circuit:battery}{البطارية}، ولهذا يكونان أخفت من
\omterm{def:g3:first-electric-circuit:bulb}{مصباح} وحيد --- أخفت بالتساوي.
\end{example}

\section{العائلتان قانونين}

\begin{proposition}[قانون التسلسل]\label{prop:g7:circuits-series-parallel:series}
في \omterm{def:g4:series-circuits:series}{دارة على التسلسل} --- حلقة واحدة --- يجري التيار نفسه عبر كل
عنصر: طريق واحد، ومسير واحد، بلا استثناء. والنتائج، مفسَّرةً
الآن: فراغ واحد يوقف المسير في كل مكان دفعة واحدة؛ والعناصر
المتماثلة تتقاسم العمل بالتساوي أينما جلست؛ وترتيب العناصر على
الحلقة لا يمكن أن يهمّ مسيرًا يمرّ بها كلها على السواء.
\end{proposition}

\begin{proposition}[قانون التفرع]\label{prop:g7:circuits-series-parallel:parallel}
عند \omterm{def:g5:series-parallel-circuits:parallel}{عقدة} \emph{ينقسم} التيار: فيسلك جزء من المسير كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع}، ثم
تلتقي الأجزاء كاملةً عند \omterm{def:g5:series-parallel-circuits:parallel}{العقدة} البعيدة --- لا شيء يُفقد ولا
شيء يُكسب. وينال كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} دفع \omterm{def:g3:first-electric-circuit:battery}{البطارية} كاملًا، فيعمل كل عنصر كأنه
وحده؛ \omterm{def:g5:series-parallel-circuits:parallel}{والفرع} المسدود لا يفعل إلا أن يرسل لا أحد، وسائر الفروع
تواصل مسيرها.
\end{proposition}

\begin{example}[قراءة السطوع قراءة كهربائي]\label{ex:g7:circuits-series-parallel:brightness}
\omterm{def:g3:first-electric-circuit:battery}{بطارية} واحدة ومصابيح متماثلة. وحيدًا: سطوع كامل --- وهو
المرجع. واثنان \omterm{def:g4:series-circuits:series}{على التسلسل}: كلاهما خافت بالتساوي --- مسير واحد
ودفع متقاسَم. واثنان \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}: كلاهما بسطوع كامل --- فكل \omterm{def:g5:series-parallel-circuits:parallel}{فرع}
ينعم بالدفع كله (وتفرغ \omterm{def:g3:first-electric-circuit:battery}{البطارية} أسرع بمرتين تقريبًا: فلا شيء
مجّاني). وثلاثة \omterm{def:g4:series-circuits:series}{على التسلسل}: أخفت بعد، وكلها متساوية. ونظرة
الكهربائي: عُدّ رفاق الحلقة للخفوت، وعُدّ الفروع لفواتير
\omterm{def:g3:first-electric-circuit:battery}{البطارية}.
\end{example}

\begin{method}[كيف تتوقّع مصير أي مصباح]\label{met:g7:circuits-series-parallel:predict}
أمام مخطط، لكل \omterm{def:g3:first-electric-circuit:bulb}{مصباح}:
\begin{enumerate}
  \item \emph{الوجود}: أثمّة حلقة واحدة غير مقطوعة على الأقلّ
        من $+$ عبر هذا \omterm{def:g3:first-electric-circuit:bulb}{المصباح} إلى $-$؟ فلا حلقة، فلا \omterm{def:g1:light-and-shadows:source}{ضوء}.
  \item \emph{الرفقة على الطريق}: أيّ العناصر يتقاسم \emph{كل}
        حلقة تمرّ بهذا \omterm{def:g3:first-electric-circuit:bulb}{المصباح}؟ فتلك رفاقه \omterm{def:g4:series-circuits:series}{على التسلسل} ---
        وكلما كثر الرفاق خفت التوهّج.
  \item \emph{جيران الفروع}: أيّ العناصر يجلس على فروع أخرى؟
        فهي لا تُخفت هذا \omterm{def:g3:first-electric-circuit:bulb}{المصباح} ولا تعتم معه.
  \item \emph{القواطع}: \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} المفتوحة تقتل بالضبط الحلقات
        التي تجلس عليها --- فأعِد الخطوة الأولى وقد نُزعت.
\end{enumerate}
\end{method}

\begin{remark}[لماذا يعسر موت الوهم]\label{rem:g7:circuits-series-parallel:whyhard}
``لا بدّ أن شيئًا يُستهلك --- \omterm{def:g3:first-electric-circuit:battery}{فالبطارية} تفرغ!'' وهذا صحيح:
\omterm{def:g5:everyday-energy:energy}{فطاقة} \omterm{def:g3:first-electric-circuit:battery}{البطارية} المخزونة تُنفَق، وتُسلَّم إلى المصابيح ضوءًا
ودفئًا عبر المسير العابر. أما السائرون أنفسهم فيدورون سالمين،
مثل كراسي مصعد التزلّج تسلّم المتزلّجين إلى القمّة طوال النهار
من غير أن يُستهلك كرسيّ واحد. فأبقِ الحسابين متمايزين ---
التيار يدور \omterm{def:g5:everyday-energy:energy}{والطاقة} تُسلَّم --- تبقَ كل دارة في هذا الكتاب
أمينة. أما الجهاز الذي يعدّ المسير نفسه فينضمّ إلينا في السنة
القادمة.
\end{remark}

\section{مسألة: لوحة حارس المنارة}

\begin{problem}[{مسألة نهاية الأسبوع --- إعادة توصيل المنارة
القديمة؛ مجموعة بطاريات واحدة وواجبات كثيرة؛ ليلة الأعطال عند
الحارس}]\label{pb:g7:circuits-series-parallel:1}
يجب أن تغذّي مجموعة بطاريات المنارة: \omterm{def:g3:first-electric-circuit:bulb}{المصباح} العظيم، وبوق
الضباب (طنّان جبّار)، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} قراءة الحارس، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} الدرج ---
وأن تنجو من ليلة تفتيش الأعطال عند الحارس.

\medskip\noindent\textbf{الجزء الأول --- التصميم.}
\begin{enumerate}
  \item يجب ألّا يخفت \omterm{def:g3:first-electric-circuit:bulb}{المصباح} العظيم أبدًا لأن شيئًا آخر
        اشتغل. فأيّ توصيل يقتضيه هذا للمنشأة كلها، ولماذا يخون
        أيُّ اقتران \omterm{def:g4:series-circuits:series}{على التسلسل} السفنَ؟
  \item ويحتاج كل واجب إلى قاطعته الخاصة، ويريد الحارس قاطع
        فصل رئيسيًّا واحدًا للعواصف. ضع القواطع الخمسة كلها.
  \item ويجب ألّا يمنع بوق الضباب \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} الدرج أحدهما الآخر من
        العمل بأعطاله. أهذا مضمون أصلًا؟ وبأيّ قانون؟
  \item ارسم أو صِف المخطط كاملًا: المجموعة، \omterm{ex:g3:first-electric-circuit:switch}{والقاطعة}
        الرئيسية، وأربعة فروع، وأربع قواطع فروع.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- أسئلة الحارس.}
\begin{enumerate}[resume]
  \item يقلق الحارس: ``أربعة واجبات تشرب دفعة واحدة --- أينال
        كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} دفع المجموعة كاملًا مع ذلك؟'' أجب بقانون
        التفرع.
  \item ``وهل يُستهلك التيار وهو يعمل عبر \omterm{def:g3:first-electric-circuit:bulb}{المصباح} العظيم؟''
        صحّح للحارس، بمصعد التزلّج إن أفاد.
  \item وفي ليلة ضبابية يعمل كل شيء دفعة واحدة. فما ثمن سخاء
        التفرع، وأين يُدفَع في المنارة؟
  \item \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} القراءة وحده مضاء؛ ويتتبّع الحارس، بروح
        \omterm{def:g6:measurement-in-science:measuring}{القياس}، المسير: صِف رحلته الكاملة ذهابًا وإيابًا،
        مسمّيًا كل عنصر يمرّ به وكل عنصر يتجنّبه.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- ليلة الأعطال.}
\begin{enumerate}[resume]
  \item منتصف الليل: يحترق \omterm{def:g3:first-electric-circuit:bulb}{مصباح} الدرج. فماذا يعتم معه؟ وأيّ
        قانون يجيب؟
  \item الواحدة: يترك إصلاح أخرق امتدادًا من سلك عارٍ يصل
        قطبَي المجموعة مباشرة، قبل كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع}. سمِّ الحدث، واشرح
        لماذا يفشل الآن كل واجب في المنارة دفعة واحدة ---
        فإلى أين يذهب المسير؟
  \item الثانية، وقد أُزيل العطل: يريد الحارس أن يحمي \omterm{def:g3:first-electric-circuit:bulb}{المصباح}
        العظيم حتى لا يُسكته اختصار كهذا مرة أخرى أبدًا.
        وعنده مجموعة بطاريات ثانية مستقلّة. اقترح التوصيل.
  \item الفجر: اكتب سجلّ الحارس --- ثلاث جمل: أيّ قانون حفظ
        السفن الليلة، وأيّ وهم دحضته الليلة، وأيّ سلك واحد
        سبّب أسوأ ساعة.
\end{enumerate}
\end{problem}
