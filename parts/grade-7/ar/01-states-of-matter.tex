\chapter{حالات المادة وتغيّراتها}\label{ch:g7:states-of-matter}

ستّ سنوات من أجسام صلبة تحفظ شكلها، وسوائل تستوي أفقيًّا، وغازات
تملأ كل ركن --- ولم نسأل مرة واحدة \emph{لماذا}. وهذه السنة ترفع
الفيزياء الغطاء. فتحت الحالات الثلاث تختبئ واحدة من أعظم الأفكار
التي خطرت للبشر: أن كل شيء، كل شيء على الإطلاق، مصنوع من جسيمات
أصغر من أن تُرى. سلّم بتلك الفكرة الواحدة تفسّر القواعد التي
جمعتها منذ الطفولة نفسها بضربة واحدة.

\section{الفكرة العظيمة}

\begin{definition}[الجزيئات]\label{def:g7:states-of-matter:molecule}
كل مادة مصنوعة من \emph{جزيئات}\index{جزيء}: جسيمات دقيقة إلى
حدّ لا يُتخيّل، أصغر بكثير ممّا تراه عين أو مجهر مدرسي. فقطرة
ماء واحدة تحوي من الجزيئات أكثر ممّا في البحار كلها من قطرات.
ولكل مادة نقية نوع جزيئاتها الخاص --- فالماء من جزيئات ماء،
والحديد من جسيماته الخاصة --- وجزيئات المادة الواحدة كلها
متماثلة.
\end{definition}

\begin{remark}[التصديق بلا رؤية]\label{rem:g7:states-of-matter:believing}
لم يشاهد أحد قطّ جزيئًا واحدًا بالعين المجرّدة --- فلماذا نصدّق
\omterm{def:g7:states-of-matter:molecule}{بالجزيئات}؟ لأن الفكرة \emph{تستحقّ} التصديق: فهي تتوقّع، توقّعًا
صحيحًا، كل ما في هذا الفصل، وجبالًا فوقه. وهكذا تقبل الفيزياء
ضيفًا خفيًّا: لا على الإيمان، بل على النفع الذي لا ينقطع. (وقد
وصلت أخيرًا صور مباشرة من أجهزة حديثة عجيبة قبل قرن --- بعد أن
كان الفيزيائيون على يقين بزمن طويل.)
\end{remark}

\begin{proposition}[النموذج الجسيمي للحالات الثلاث]\label{prop:g7:states-of-matter:model}
الحالات الثلاث ثلاثة ترتيبات \omterm{def:g7:states-of-matter:molecule}{للجزيئات} نفسها:
\begin{enumerate}
  \item \emph{الصلبة}: \omterm{def:g7:states-of-matter:molecule}{جزيئات} محشوّة متراصّة في رصّة منظَّمة،
        يرتعش كل واحد منها في موضعه محافظًا عليه --- ومن هنا
        الشكل الثابت؛
  \item \emph{السائلة}: \omterm{def:g7:states-of-matter:molecule}{جزيئات} ما زالت متلامسة، لكنها غير
        منظَّمة وحرّة في الانزلاق بعضها بجوار بعض --- ومن هنا
        الجريان والسطح المستوي؛
  \item \emph{الغازية}: \omterm{def:g7:states-of-matter:molecule}{جزيئات} متباعدة، تطير سريعة في كل
        الاتجاهات، تتصادم وترتدّ --- ومن هنا ملء كل ركن
        يُعرَض عليها.
\end{enumerate}
\end{proposition}

\begin{example}[ستّ سنوات من القواعد، وتفسير واحد]\label{ex:g7:states-of-matter:rules}
أجرِ القواعد القديمة في النموذج. الأجسام الصلبة تحفظ شكلها:
فكل \omterm{def:g7:states-of-matter:molecule}{جزيء} يحفظ موقعه. والسوائل تأخذ شكل الإناء وتستوي أفقيًّا:
فالحشد المنزلق ينهار حتى لا يستطيع \omterm{def:g7:states-of-matter:molecule}{جزيء} أن ينزلق أخفض. والغازات
تملأ كل حيّز يُعرَض عليها: فلا شيء يمسك الطائرين. والعطر يعبر
الغرفة: \omterm{def:g7:states-of-matter:molecule}{جزيئات} \omterm{def:g3:solids-liquids-gases:gas}{غاز} تطير وتتصادم وتنتشر. والرمل يُسكب \omterm{def:g3:solids-liquids-gases:liquid}{كالسائل}:
لا \omterm{def:g7:states-of-matter:molecule}{جزيئات} تنزلق، بل حبيبات --- وكل حبيبة نفسها \omterm{def:g3:solids-liquids-gases:solid}{جسم صلب} دقيق
منظَّم من تريليونات \omterm{def:g7:states-of-matter:molecule}{الجزيئات}.
\end{example}

\section{اختبار الحيّز الخالي}

\begin{method}[محقنتان]\label{met:g7:states-of-matter:syringe}
يدّعي النموذج أن \omterm{def:g7:states-of-matter:molecule}{جزيئات} \omterm{def:g3:solids-liquids-gases:gas}{الغاز} تطير متباعدة، وأن \omterm{def:g7:states-of-matter:molecule}{جزيئات} \omterm{def:g3:solids-liquids-gases:liquid}{السائل}
متلامسة. فينبغي إذن أن ينضغط \omterm{def:g3:solids-liquids-gases:gas}{الغاز} وألّا ينضغط \omterm{def:g3:solids-liquids-gases:liquid}{السائل}. اختبر
ذلك:
\begin{enumerate}
  \item املأ محقنة بلاستيكية هواءً، وسُدّ فوهتها بطرف إصبعك،
        وادفع المكبس: فيستجيب --- تستطيع أن تنصّف حيّز \omterm{def:g2:air-around-us:air}{الهواء}،
        وتحسّ بمقاومة مرنة؛
  \item واملأ المحقنة نفسها ماءً، وسُدّها، وادفع: فيقف المكبس
        كالجدار --- يأبى الماء أن ينكمش ولو شعرة؛
  \item وأفلت كل مكبس: فيرتدّ \omterm{def:g2:air-around-us:air}{الهواء}؛ أما الماء فلم يتحرّك
        قطّ.
\end{enumerate}
\end{method}

\begin{proposition}[الغازات تنضغط والسوائل والصلبة لا]\label{prop:g7:states-of-matter:squeeze}
\omterm{def:g3:solids-liquids-gases:gas}{الغاز} \emph{قابل للانضغاط}\index{قابل للانضغاط} --- يمكن أن
يُدفع إلى حيّز أقلّ --- لأن أكثر \omterm{def:g3:solids-liquids-gases:gas}{الغاز} حيّز خالٍ بين \omterm{def:g7:states-of-matter:molecule}{جزيئات}
طائرة. أما السوائل والأجسام الصلبة فغير قابلة للانضغاط عمليًّا:
فجزيئاتها متلامسة أصلًا، والمادة المتلامسة لا يمكن أن تُرصّ
أكثر بأي دفع تستطيعه يد --- أو مكبس هيدروليكي.
\end{proposition}

\begin{example}[الهواء المضغوط في عمله]\label{ex:g7:states-of-matter:squeezed}
منفاخ الدرّاجة يضغط ملء غرف من \omterm{def:g2:air-around-us:air}{الهواء} في الإطار --- ولا يمكن
ذلك إلا لأن أكثر \omterm{def:g2:air-around-us:air}{الهواء} خواء. والحشد المضغوط، وجزيئاته تطرق
جدران الإطار أسرع وأكثف، يحمل الدرّاجة: فدفع \omterm{def:g3:solids-liquids-gases:gas}{الغاز} \emph{هو}
عاصفة برَده الجزيئية. وقنّينات الغوّاصين ترصّ تنفّس عصريّة
كاملة في قارورة فولاذية؛ أما \omterm{def:g3:solids-liquids-gases:liquid}{سائل} المكابح غير \omterm{prop:g7:states-of-matter:squeeze}{القابل للانضغاط}
فينقل دفعة قدمك إلى العجلات بلا نقصان --- فكل حالة مستأجَرة
لموهبتها.
\end{example}

\section{تغيّرات الحالة، مقروءةً من جديد}

\begin{example}[الأبواب، جزيئًا جزيئًا]\label{ex:g7:states-of-matter:doors}
سخّن جسمًا صلبًا فترتعش جزيئاته أشدّ؛ وعند \omterm{def:g4:changes-of-state:melting-point}{درجة الانصهار} يفجّر
الارتعاش الرصّة المنظَّمة --- فيبدأ الحشد بالانزلاق: وهو
\omterm{def:g6:water-states:changes}{الانصهار}. وسخّن \omterm{def:g3:solids-liquids-gases:liquid}{السائل} أكثر فتنتزع أسرع \omterm{def:g7:states-of-matter:molecule}{الجزيئات} نفسها من قبضة
الحشد كليًّا وتطير: وهو \omterm{def:g6:water-states:changes}{التبخّر} السطحي --- وعند \omterm{def:g4:changes-of-state:boiling-point}{درجة الغليان}
يصير الهرب اندفاعًا جماعيًّا من داخل \omterm{def:g3:solids-liquids-gases:liquid}{السائل} نفسه. والتبريد يجري
الشريط بالمقلوب: فيُمسَك الطائرون (\omterm{def:g6:water-states:changes}{تكاثف})، وينغلق المنزلقون في
صفوف (\omterm{def:g6:water-states:changes}{تصلّب}). فأبواب السنة الماضية الستّة قصة واحدة: قبضة
جزيئية في مواجهة ارتعاش جزيئي.
\end{example}

\begin{example}[قوانين قديمة بالمجّان]\label{ex:g7:states-of-matter:free}
قانونان كُسبا بعناء يأتيان الآن بالمجّان. \emph{\omterm{def:g3:measuring-mass:mass}{الكتلة} تنجو من
تغيّرات الحالة}: \omterm{def:g6:water-states:changes}{فالانصهار} يعيد ترتيب \omterm{def:g7:states-of-matter:molecule}{الجزيئات} ولا يخلق واحدًا
ولا يُفني واحدًا --- الجسيمات نفسها \omterm{def:g3:measuring-mass:mass}{والكتلة} نفسها.
\emph{ومسطبات \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة}}: فعند \omterm{def:g4:changes-of-state:melting-point}{درجة الانصهار} يُنفَق الدفء
الوارد على كسر قبضة الرصّة، جزيئًا جزيئًا، لا على ارتعاش أسرع
--- فينتظر \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} حتى ينكسر آخر صفّ. وما اكتشفه \omterm{def:g6:measurement-in-science:measuring}{القياس}
يفسّره النموذج.
\end{example}

\begin{remark}[انتفاخ الماء العجيب]\label{rem:g7:states-of-matter:iceswell}
حتى انتفاخ الماء الغريب عند \omterm{def:g2:ice-water-steam:meltfreeze}{التجمّد} ينضمّ إلى القافلة: \omterm{def:g7:states-of-matter:molecule}{فجزيئات}
الماء يصادف أنها تنغلق في رصّة \emph{هوائية} غير معتادة ---
شبكة مفتوحة فيها حيّز خالٍ أكثر ممّا كان في الحشد المنزلق.
ورصّة أهوى تعني حيّزًا أكبر \omterm{def:g6:density-floating:density}{وكثافة} أدنى: \omterm{def:g1:floating-and-sinking:words}{فيطفو} الثلج على سائله.
وأكثر الموادّ ترصّ أضيق ممّا تنزلق؛ وعمارة الماء هي الاستثناء
البديع.
\end{remark}

\section{الحرارة، مرئيةً من الداخل}

\begin{proposition}[درجة الحرارة والحركة الجزيئية]\label{prop:g7:states-of-matter:temperature}
\omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} \omterm{def:g6:measurement-in-science:measuring}{تقيس} نشاط الحركة الجزيئية: فكلما سخنت المادة
ارتعشت جزيئاتها أو انزلقت أو طارت أسرع. وجريان \omterm{def:g5:heat-and-insulation:heat}{الحرارة} من
الساخن إلى البارد هو الحشد النشيط يهدأ وهو يحرّك الحشد الخامل
--- تصادمًا تصادمًا --- حتى يتقاسما وتيرة واحدة: أي \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة}
واحدة. فقانون \omterm{def:g5:heat-and-insulation:heat}{الحرارة} النازل القديم كرة طاولة جزيئية.
\end{proposition}

\begin{example}[البرودة ليست مادة]\label{ex:g7:states-of-matter:cold}
يُقاعد النموذج إغراءً قديمًا إلى الأبد: فلا شيء اسمه ``البرودة''
يجري أبدًا. إنما هناك حركة جزيئية أبطأ وأسرع لا غير؛ والنشيط
يُقرض النشاط، ولا عكس. وعند أبرد ما يُتخيّل --- \omterm{def:g7:states-of-matter:molecule}{والجزيئات} ساكنة
بقدر ما تسمح الطبيعة --- تقع أرضية حقيقية \omterm{def:g3:temperature-thermometers:temperature}{لدرجة الحرارة}، أدنى
بكثير من أي مجمّدة؛ وعددها الغريب المشهور يخصّ مجلّد المرحلة
الثانوية.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:1}
اذكر صورة النموذج الجسيمي لكل حالة من الحالات الثلاث، في سطر
واحد لكل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:2}
لماذا يحفظ \omterm{def:g3:solids-liquids-gases:solid}{الجسم الصلب} شكله بينما يأخذ \omterm{def:g3:solids-liquids-gases:liquid}{السائل} شكل إنائه؟ أجب
\omterm{def:g7:states-of-matter:molecule}{بالجزيئات} لا بالقواعد.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:3}
في اختبار المحقنة، لماذا يستجيب \omterm{def:g2:air-around-us:air}{الهواء} ويرتدّ، بينما يقف الماء
كالجدار؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:4}
منفاخ الدرّاجة يرصّ ملء غرف كثيرة من \omterm{def:g2:air-around-us:air}{الهواء} في إطار واحد. فماذا
يثبت هذا عن تباعد \omterm{def:g7:states-of-matter:molecule}{جزيئات} \omterm{def:g3:solids-liquids-gases:gas}{الغاز}؟ وهل يستطيع منفاخ أن يفعل ذلك
بالماء؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:5}
احكِ قصة \omterm{def:g6:water-states:changes}{الانصهار} باللغة الجزيئية: ماذا يفعل الدفء، وما الذي
ينكسر عند \omterm{def:g4:changes-of-state:melting-point}{درجة الانصهار}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:6}
لماذا تنجو \omterm{def:g3:measuring-mass:mass}{الكتلة} من كل تغيّر حالة، وفق النموذج؟ وما الذي كان
يجب أن يحدث \omterm{def:g7:states-of-matter:molecule}{للجزيئات} حتى تُفقد \omterm{def:g3:measuring-mass:mass}{كتلة}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:7}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:8}

\begin{exercise}[$\star\star$]\label{exo:g7:states-of-matter:9}

\begin{exercise}[$\star\star$]\label{exo:g7:states-of-matter:11}
لماذا \omterm{def:g1:floating-and-sinking:words}{يطفو} الثلج، باللغة الجزيئية؟ وما غير المعتاد في الرصّة
التي تبنيها \omterm{def:g7:states-of-matter:molecule}{جزيئات} الماء، وماذا كانت البِرَك ستفعل في الشتاء لو
رصّ الماء كما ترصّ الموادّ العادية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:states-of-matter:12}
قطرة حبر أُطلقت برفق في قاع كأس ماء \emph{ساكن} تنتشر على مهل،
عبر ساعات، حتى يصير الماء كله أزرق خافتًا --- بلا تحريك ولا
تيّارات. اشرح ما يكشفه هذا الانتشار الصبور عن \omterm{def:g7:states-of-matter:molecule}{جزيئات} \omterm{def:g3:solids-liquids-gases:liquid}{السائل} ---
\omterm{def:g7:states-of-matter:molecule}{جزيئات} الماء \omterm{def:g7:states-of-matter:molecule}{وجزيئات} الحبر معًا --- ولماذا تنتشر القطرة نفسها
أسرع في ماء دافئ. (وقد كانت هذه الملاحظة المتواضعة من أوائل
الأدلّة العظيمة على الفكرة الجزيئية كلها.)
\end{exercise}

\section{مسألة: المسرح الجزيئي}

\begin{problem}[{مسألة نهاية الأسبوع --- الصفّ يمثّل النموذج
الجسيمي في الصالة الرياضية؛ التلاميذ جزيئات، والحالات
تصميم رقص؛ وأسئلة النقّاد}]\label{pb:g7:states-of-matter:1}
يمثّل صفّ الفيزياء المادةَ نفسها: ثلاثون تلميذًا يؤدّون دور
\omterm{def:g7:states-of-matter:molecule}{جزيئات} مادة واحدة، والصالة الرياضية هي الإناء.

\medskip\noindent\textbf{الجزء الأول --- ثلاثة مشاهد.}
\begin{enumerate}
  \item المشهد الأول، الحالة الصلبة: كيف ينبغي أن يقف
        الثلاثون، وأيّ حركة صغيرة يجب أن يواصلها كل واحد؟ وما
        الذي لا يجوز لأحد منهم فعله؟
  \item المشهد الثاني، الحالة السائلة: ما الذي يتغيّر في
        الترتيب، وأيّ حرّية جديدة تظهر؟ وما الذي يجب أن يواصله
        التلاميذ ممّا سيلغيه مشهد \omterm{def:g3:solids-liquids-gases:gas}{الغاز}؟
  \item المشهد الثالث، الحالة الغازية: صِف تصميم الرقص ---
        وقل لماذا يحتاج هذا المشهد إلى \emph{كامل} الصالة بينما
        يتكوّم المشهدان الأولان في ركن.
  \item ويسأل الجمهور: ``أهم التلاميذ الثلاثون أنفسهم في
        المشاهد الثلاثة؟'' فأيّ قانون من قوانين السنة الماضية
        يمثّله الجواب؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- النقّاد يختبرون
المسرحية.}
\begin{enumerate}[resume]
  \item يدفع ناقد برفق حشد مشهد \omterm{def:g3:solids-liquids-gases:gas}{الغاز}، فيسوقه إلى نصف الصالة.
        أيسمح المشهد بذلك؟ وماذا عن الدفعة نفسها على كومة مشهد
        \omterm{def:g3:solids-liquids-gases:liquid}{السائل}؟ سمِّ القضية الممثَّلة.
  \item ولتمثيل \emph{\omterm{def:g6:water-states:changes}{الانصهار}} يرفع المخرج قرص \omterm{def:g5:heat-and-insulation:heat}{حرارة}
        متخيَّلًا. فما الذي يجب أن يزيد منه تلاميذ المشهد
        الصلب، وفي أيّ لحظة يصير المشهد قانونًا مشهدَ
        \omterm{def:g3:solids-liquids-gases:liquid}{السائل}؟
  \item ويجب أن تُظهر المسرحية مسطبة \omterm{def:g6:water-states:changes}{الانصهار}: فبينما يحدث
        التغيّر، على ماذا تُنفَق \omterm{def:g5:everyday-energy:energy}{طاقة} قرص \omterm{def:g5:heat-and-insulation:heat}{الحرارة} --- وبماذا
        يجب أن يبلّغ ``الناقد \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة}'' عن نشاط التلاميذ
        الوسطيّ خلالها؟
  \item ومثّل \omterm{def:g6:water-states:changes}{التبخّر} من مشهد \omterm{def:g3:solids-liquids-gases:liquid}{السائل}: أيّ التلاميذ يغادرون
        الكومة --- أيّ واحد على غير قاعدة أم نوع بعينه؟ وماذا
        يحدث لنشاط الكومة الوسطيّ وهم يغادرون، وأيّ قشعريرة
        يومية يفسّرها هذا؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- المؤثّرات الخاصة.}
\begin{enumerate}[resume]
  \item مؤثّر البالون: يُسيَّج مشهد \omterm{def:g3:solids-liquids-gases:gas}{الغاز} بتلاميذ يمسكون حلقة
        حبل (وهي غلاف البالون). فماذا يفعل الممثّلون الجزيئيون
        بالحبل، وماذا يحدث للحلقة حين يتحرّك الممثّلون أسرع؟
  \item ويجب تمثيل \omterm{def:g2:ice-water-steam:meltfreeze}{تجمّد} الماء العجيب. فأيّ تعليمة غير معتادة
        عن \emph{التباعد} يعطيها المخرج لرصّة المشهد الصلب،
        وأيّ واقعة في \omterm{def:g1:floating-and-sinking:words}{الطفو} تفسّرها؟
  \item وختام قطرة الحبر: يبدأ تلميذ بقميص أزرق في ركن من مشهد
        \emph{\omterm{def:g3:solids-liquids-gases:liquid}{السائل}}. فمن غير أي دفعة من المخرج، كيف ينتهي
        القميص الأزرق بعيدًا عن ركنه --- ولماذا أسرع إن رُفع
        قرص \omterm{def:g5:heat-and-insulation:heat}{الحرارة}؟
  \item واكتب ملاحظة البرنامج: قل في ثلاث جمل أيّ فكرة واحدة
        تمثّلها المسرحية كلها، وسمِّ قانونين قديمين تفسّرهما
        بالمجّان، واعترف بأمانة بما لا يستطيع أي مشهد أن يُظهره
        (كم هم \emph{صغار} الممثّلون الحقيقيون وكم هم
        \emph{كثيرون}).
\end{enumerate}
\end{problem}
