\chapter{منابع الضوء وانتشاره}\label{ch:g7:light-sources-propagation}

لماذا تستطيع أن تقرأ هذه الصفحة؟ \omterm{def:g3:first-electric-circuit:bulb}{مصباح} يضيء --- والصفحة ليست
مصباحًا، ومع ذلك تستقبل عينك ضوءًا من كل سطر مطبوع. وبين ما
يصنع \omterm{def:g1:light-and-shadows:source}{الضوء} وما يكتفي بتمريره تمييزٌ لا يستطيع علم البصريات أن
يعيش من دونه؛ ومعه يأتي الجواب الكامل عن أقدم سؤال في هذا
الكتاب: ما الذي يلزم لرؤية شيء؟

\section{الصانعون والمُمرِّرون}

\begin{definition}[المنابع الأوّلية]\label{def:g7:light-sources-propagation:primary}
\emph{المنبع الأوّلي}\index{منبع أوّلي} يصنع \omterm{def:g1:light-and-shadows:source}{ضوءه} بنفسه: الشمس
والنجوم، واللهب، والخيوط المتوهّجة، وفولاذ المسبك الأبيض
الساخن، واليراعات، والبرق. وضوؤه موجود سواء أضاء غيره أم لم
يُضئ --- أطفئ العالم يظلّ المنبع الأوّلي متوهّجًا.
\end{definition}

\begin{definition}[الأجسام المبعثِرة]\label{def:g7:light-sources-propagation:diffusion}
وكل شيء مرئي آخر \emph{جسم مبعثِر}\index{تبعثر}: يستقبل \omterm{def:g1:light-and-shadows:source}{الضوء}
ويعيد إرسال --- \emph{يبعثر} --- جزءًا منه في كل الاتجاهات.
الصفحة، والتفاحة، \omterm{def:g2:moon-in-the-sky:moon}{والقمر}، ووجهك أنت: لا يصنع أيّ منها ومضة
خاصة به؛ وكل واحد ينثر \omterm{def:g1:light-and-shadows:source}{الضوء} المستقبَل بسخاء في كل جهة. والجسم
المبعثِر في الظلمة غير مرئي ببساطة.
\end{definition}

\begin{example}[التبعثر ليس انعكاسًا في مرآة]\label{ex:g7:light-sources-propagation:notmirror}
المرايا والصفحات كلتاهما تردّ \omterm{def:g1:light-and-shadows:source}{الضوء} المستقبَل --- لكن بأسلوبين
متعاكسين. \omterm{def:g5:mirrors-reflection:reflection}{فالمرآة} ترتدّ \omterm{def:g1:light-and-shadows:source}{بالضوء} في اتجاه \emph{واحد} منضبط
(زاويتان متساويتان --- وهي قاعدة الارتداد القديمة عندك)، فتحفظ
الصورة؛ أما خشونة الصفحة المجهرية فترشّ \omterm{def:g1:light-and-shadows:source}{الضوء} في \emph{كل
مكان}، فتُتلف الصورة لكنها تقدّم نفسها لكل عين دفعة واحدة.
ولهذا تُظهر \omterm{def:g5:mirrors-reflection:reflection}{المرآة} \omterm{def:g3:first-electric-circuit:bulb}{المصباح}، بينما لا تُظهر الصفحة إلا نفسها،
من أيّ موضع وقفت.
\end{example}

\begin{example}[ملفّ القمر، مغلقًا]\label{ex:g7:light-sources-propagation:moon}
حكم الطفولة ينال الآن كلماته الرسمية: \omterm{def:g2:moon-in-the-sky:moon}{القمر} ليس منبعًا أوّليًّا؛
إنما هو \omterm{def:g7:light-sources-propagation:diffusion}{جسم مبعثِر} \omterm{def:g6:mass-and-volume:volume}{بحجم} \omterm{def:g4:solar-system:planet}{كوكب} --- صخر رمادي ينثر \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس في
كل الاتجاهات، ومنها جهتنا. \omterm{def:g1:light-and-shadows:source}{فضوء} \omterm{def:g2:moon-in-the-sky:moon}{القمر} شمسٌ مستعملة، و``الجانب
المظلم'' هو ببساطة النصف الذي لا يحمل ملفّه، في تلك اللحظة،
ضوءًا يُمرَّر.
\end{example}

\section{الرؤية، محسومةً}

\begin{proposition}[شرطا الرؤية]\label{prop:g7:light-sources-propagation:vision}
ترى العينُ جسمًا بالضبط حين
\begin{enumerate}
  \item \emph{يغادر} الضوءُ الجسمَ --- مصنوعًا إن كان الجسم
        منبعًا أوّليًّا؛ ومبعثَرًا إن كان مُمرِّرًا --- و
  \item ينتقل ذلك \omterm{def:g1:light-and-shadows:source}{الضوء} في خطّ مستقيم غير محجوب إلى العين.
\end{enumerate}
فاكسر أيّ الشرطين --- جسمًا مبعثِرًا بلا إضاءة، أو جدارًا عبر
خطّ النظر --- يصر الجسم غير مرئي. وكل أحجية رؤية في هذا الكتاب
تنفكّ إلى هذين الفحصين.
\end{proposition}

\begin{example}[الحزمة، مرئيةً]\label{ex:g7:light-sources-propagation:beam}
في \omterm{def:g2:air-around-us:air}{هواء} نظيف لا يمكن أن تُرى \omterm{def:g6:light-propagation:ray}{حزمة} \omterm{def:g3:first-electric-circuit:bulb}{مصباح} يدوي تعبر الغرفة من
الجانب --- \omterm{def:g2:air-around-us:air}{فالهواء} لا يمرّر شيئًا يُذكر، ولا ينحرف من \omterm{def:g1:light-and-shadows:source}{ضوء}
\omterm{def:g6:light-propagation:ray}{الحزمة} شيء نحو عينك. انفض قطعة قماش مغبرّة: تقفز \omterm{def:g6:light-propagation:ray}{الحزمة} إلى
الرؤية. فكل ذرّة غبار \omterm{def:g7:light-sources-propagation:diffusion}{جسم مبعثِر} دقيق، يلتقط \omterm{def:g1:light-and-shadows:source}{ضوء} \omterm{def:g6:light-propagation:ray}{الحزمة} وينثر
بعضه في جهتك --- وآلاف الذرّات ترسم لك خطّ \omterm{def:g6:light-propagation:ray}{الحزمة} المستقيم.
والضباب والغبش والدخان تؤدّي الخدمة نفسها لحزم المنارات وأجهزة
عرض السينما.
\end{example}

\section{ثلاثة أثواب للضوء}

\begin{definition}[شفّاف ونصف شفّاف ومعتم]\label{def:g7:light-sources-propagation:coats}
تلقى الموادّ الضوءَ بثلاثة أساليب:
\begin{enumerate}
  \item \emph{شفّافة}\index{شفّاف}: يعبرها \omterm{def:g1:light-and-shadows:source}{الضوء} من غير اضطراب
        يُذكر، حافظًا نظامه --- الزجاج الصافي، والماء الساكن،
        \omterm{def:g2:air-around-us:air}{والهواء}؛ وتعبرها الصور سليمة؛
  \item \emph{نصف شفّافة}\index{نصف شفّاف}: يعبرها \omterm{def:g1:light-and-shadows:source}{الضوء} لكن
        مشوَّشًا --- الزجاج المصنفر، وورق الشفّ، والستارة
        الرقيقة؛ فيعبر السطوع ولا تعبر الصور؛
  \item \emph{معتمة}\index{معتم}: يُوقَف \omterm{def:g1:light-and-shadows:source}{الضوء} --- يُمتصّ أو
        يُبعثَر راجعًا --- الخشب والحجر والمعدن ويدك؛ وخلف
        الجسم المعتم يقع \omterm{def:g1:light-and-shadows:shadow}{الظلّ}.
\end{enumerate}
\end{definition}

\begin{method}[تصنيف الموادّ بضوء مصباح يدوي]\label{met:g7:light-sources-propagation:sort}
غرفة مظلمة، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} يدوي، وصفحة مطبوعة، والعيّنة بينهما:
\begin{enumerate}
  \item أمسك العيّنة فوق المطبوع \omterm{def:g3:first-electric-circuit:bulb}{والمصباح} خلفها؛
  \item أتقرأ المطبوع عبرها؟ --- \omterm{def:g7:light-sources-propagation:coats}{شفّافة}؛
  \item أترى توهّجًا بلا حروف؟ --- \omterm{def:g7:light-sources-propagation:coats}{نصف شفّافة}؛
  \item أترى ظلمة، وظلًّا وراءها؟ --- \omterm{def:g7:light-sources-propagation:coats}{معتمة}.
\end{enumerate}
واختبر بأمانة: فكثير من الموادّ يبدّل فريقه بتبدّل السماكة ---
فصفيحة الماء \omterm{def:g7:light-sources-propagation:coats}{شفّافة}، والبحر العميق مظلم كالليل؛ وورقة واحدة
\omterm{def:g7:light-sources-propagation:coats}{نصف شفّافة}، والكتاب المغلق \omterm{def:g7:light-sources-propagation:coats}{معتم}.
\end{method}

\begin{example}[فرق العمارة الثلاثة]\label{ex:g7:light-sources-propagation:architecture}
ينشر البنّاؤون الفرق الثلاثة عن قصد. فزجاج النافذة \omterm{def:g7:light-sources-propagation:coats}{الشفّاف}:
منظر \omterm{def:g1:light-and-shadows:source}{وضوء} معًا. وزجاج الحمّام \omterm{def:g7:light-sources-propagation:coats}{نصف الشفّاف} وحواجز المكاتب
المصنفرة: \omterm{def:g1:light-and-shadows:source}{ضوء} نهار داخل وخصوصية محفوظة --- فالسطوع بلا صور هو
وصف عملها بالضبط. والجدران والمصاريع \omterm{def:g7:light-sources-propagation:coats}{المعتمة}: ظلمة عند الطلب.
وأباجورات المصابيح نصف شفافية في خدمة الراحة: فوهج الخيط
المشتعل يُشوَّش حتى يصير توهّجًا ليّنًا بلا منبع.
\end{example}

\begin{remark}[لا شيء يُمرَّر، فلا شيء يُرى]\label{rem:g7:light-sources-propagation:space}
أحجية: رائدة فضاء بين النجوم تنظر \emph{جانبًا} عبر خواء مملوء
بأشعّة الشمس. فماذا ترى؟ سوادًا --- مع أن \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس يغمر ما
حول عينيها. فالفضاء لا يحمل غبارًا يُذكر: فلا شيء يبعثر \omterm{def:g1:light-and-shadows:source}{الضوء}
العابر نحوها، \omterm{def:g1:light-and-shadows:source}{والضوء} الذي لا يدخل العين لا يرسم شيئًا. وسماء
النهار زرقاء بالضبط لأن هواءنا \emph{يمرّر} \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس نازلًا
إلينا --- \omterm{def:g4:solar-system:planet}{كوكب} كامل من \omterm{def:g7:light-sources-propagation:diffusion}{التبعثر} الرقيق فوقنا. فلا \omterm{def:g2:air-around-us:air}{هواء}، فسماء
سوداء، حتى عند الظهر: وقد وقف روّاد \omterm{def:g2:moon-in-the-sky:moon}{القمر} في \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس تحت
قبّة سوداء بلا نجوم.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g7:light-sources-propagation:2}
ماذا يفعل \omterm{def:g7:light-sources-propagation:diffusion}{الجسم المبعثِر} \omterm{def:g1:light-and-shadows:source}{بالضوء} الذي يستقبله --- وماذا يصير في

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:light-sources-propagation:3}
اذكر شرطَي الرؤية، واستعملهما: لماذا لا تستطيع أن ترى سطح
\omterm{def:g5:mirrors-reflection:reflection}{مرآة} نظيفة تمامًا \emph{نفسه}، بل الأشياء التي تعكسها لا غير؟

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:light-sources-propagation:4}

\begin{exercise}[$\star$]\label{exo:g7:light-sources-propagation:6}

\begin{exercise}[$\star\star\star$]\label{exo:g7:light-sources-propagation:12}

الشمس زرقاء، وقرص الشمس أبيض يخطف البصر --- وعند \omterm{def:g1:day-and-night:daynight}{الغروب} تصير
الشمس المنخفضة برتقالية بينما تلتقط الغيوم فوقها لونًا
ورديًّا. مستعملًا عبارة ``\omterm{def:g2:air-around-us:air}{الهواء} يبعثر قليلًا من \omterm{def:g1:light-and-shadows:source}{الضوء} العابر
فيه، وأكثر كلما طال الطريق'' لا غير، اشرح لماذا تُسلَب الشمس
المنخفضة جزءًا من ضوئها --- ومَن يتلقّى الحصة المسلوبة. (وقصة
الألوان كاملة في فصل الأطياف في السنة القادمة؛ فاستدلّ هنا
بطريق أطول \omterm{def:g7:light-sources-propagation:diffusion}{وتبعثر} أكثر لا غير.)
\end{exercise}

\section{مسألة: إنارة المتحف}

\begin{problem}[{مسألة نهاية الأسبوع --- قاعة النحت الجديدة؛
وردية ليلية لمصمّم إضاءة؛ منابع وأجسام مبعثِرة وثلاثة أنواع من
الزجاج}]\label{pb:g7:light-sources-propagation:1}
تُفتَح قاعة المتحف الجديدة يوم الاثنين: تمثال رخامي واحد،
ولوحات على الجدران، وواجهة عرض زجاجية، وأمين متحف قلق. وأنت
تساعد مصمّم الإضاءة.

\medskip\noindent\textbf{الجزء الأول --- المبادئ الأولى في
الموقع.}
\begin{enumerate}
  \item يسأل أمين المتحف لماذا يحتاج التمثال، وهو ``ناصع
        البياض''، إلى مصابيح أصلًا. أجب بشرطَي الرؤية.
  \item أحصِ المنابع الأوّلية والأجسام المبعثِرة في القاعة بعد
        إشعال المصابيح: المصابيح، والتمثال، واللوحات،
        والزوّار، وزجاج الواجهة.
  \item يبدو الرخام متوهّجًا من كل مكان بلا بقعة تخطف البصر.
        فأيّ أسلوب في ردّ \omterm{def:g1:light-and-shadows:source}{الضوء} يعمل، ولماذا يناله كل موضع
        يقف فيه زائر؟
  \item وفي المقابل تلقي لوحة نحاسية مصقولة \omterm{def:g5:mirrors-reflection:reflection}{انعكاسًا} واحدًا
        خاطفًا \omterm{def:g3:first-electric-circuit:bulb}{لمصباح}. سمِّ الأسلوب الآخر، واذكر القاعدة التي
        تحكم أين يذهب ذلك الخطف.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- ترويض الوهج.}
\begin{enumerate}[resume]
  \item المصابيح العارية توهج في أعين الزوّار. فيغلّف المصمّم
        كل واحد \omterm{def:g5:simple-machines:pulley}{بكرة} مصنفرة. فإلى أيّ فريق من الموادّ تنتمي
        الكرة، وماذا تفعل \omterm{def:g1:light-and-shadows:source}{بضوء} الخيط؟
  \item ويجب أن تُضاء اللوحات، وألّا يظهر أيّ \omterm{def:g3:first-electric-circuit:bulb}{مصباح}
        \emph{منعكسًا} في ورنيشها من موضع المشاهدة الرئيسي.
        فباستعمال قاعدة \omterm{def:g5:mirrors-reflection:reflection}{المرآة}، أين يضع المصمّم المصابيح
        \omterm{def:g6:measurement-in-science:measuring}{بالقياس} إلى اللوحة والمشاهد؟
  \item وزجاج واجهة العرض الصافي يكاد يكون غير مرئي --- حتى
        يظهر فيه كمّ زائر أبيض كالشبح. فلماذا ينفذ الزجاج
        الصافي ويعكس عكسًا خافتًا في آن؟ وما الذي يحقّقه
        تجويف المصمّم الأسود الجدران خلف الواجهة؟
  \item وأحيانًا تنجرف ذرّات غبار عبر حزم الكشّافات، فترسم
        خيوطًا ساطعة في \omterm{def:g2:air-around-us:air}{الهواء}. فيقول أمين المتحف: ``بديع!
        أبقِه.'' ويقول المصمّم: ``إنه يعني هواءً متّسخًا.''
        فمن المصيب في الفيزياء، وما تلك الخيوط؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- اختبار الليل.}
\begin{enumerate}[resume]
  \item الأضواء مطفأة والمصاريع محكمة: يبلّغ الحارس أن القاعة
        ``سوداء تمامًا --- ولا حتى التمثال الأبيض''. اشرح
        بشرطَي الرؤية لماذا يعجز البياض في الظلمة.
  \item ويُقترح منور سقفي: زجاج مصنفر في الأعلى. فماذا سيسلّم
        نهارًا، وما الذي سيأبى تسليمه (مناظر الغيوم، وخطف
        الشمس المباشر)؟
  \item ويخشى أمين المتحف أن يبهت \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس اللوحات فيطلب
        ``ضوءًا بلا شمس البتة''. فيبتسم المصمّم: فحتى \omterm{def:g1:light-and-shadows:source}{ضوء}
        النوافذ الشمالية اللين نهارًا \emph{هو} \omterm{def:g1:light-and-shadows:source}{ضوء} شمس. اشرح
        --- ومَن الذي مرّره؟
  \item واكتب تقرير المصمّم: ثلاث جمل --- واحدة في لماذا كان
        \omterm{def:g7:light-sources-propagation:diffusion}{التبعثر} خير صديق للمتحف، وواحدة في أين يجب أن تُراقب
        الانعكاسات على أسلوب \omterm{def:g5:mirrors-reflection:reflection}{المرآة}، وواحدة في فريق الزجاج
        المختار لكل واجب.
\end{enumerate}
\end{problem}
