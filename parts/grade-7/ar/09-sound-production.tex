\chapter{الصوت: إنتاجه وانتشاره}\label{ch:g7:sound-production}

تعرف منذ الطفولة أن \omterm{def:g3:sound-around-us:sound}{الصوت} يولد في ارتعاش ويموت في الخواء ---
فقد رقصت حبّات الأرزّ، وسكت الجرس في الجرّة. وكان ينقصك اسم
الرسول. أمّا الآن فتملك العالم الجزيئي، ورحلة \omterm{def:g3:sound-around-us:sound}{الصوت} كلها ---
من وتر القيثارة إلى طبلة أذنك، عبر \omterm{def:g2:air-around-us:air}{الهواء} أو الماء أو الفولاذ
--- صارت تُشاهَد من الداخل أخيرًا.

\section{الرُّسل}

\begin{definition}[الوسط]\label{def:g7:sound-production:medium}
\emph{الوسط}\index{وسط} (وجمعه \emph{أوساط}) هو المادة التي
يسافر فيها \omterm{def:g3:sound-around-us:sound}{الصوت}: \omterm{def:g2:air-around-us:air}{الهواء} والماء والخشب والفولاذ --- وكل حالة
تفي بالغرض، فكلها حشود من \omterm{def:g7:states-of-matter:molecule}{الجزيئات}. وأمّا ما لا يستطيع \omterm{def:g3:sound-around-us:sound}{صوت} أن
يعبره فهو غياب الحشد: الخلاء، المسرح الفارغ.
\end{definition}

\begin{proposition}[كيف يسافر الصوت]\label{prop:g7:sound-production:travel}
المنبع المهتزّ يدفع \omterm{def:g7:states-of-matter:molecule}{الجزيئات} التي بجانبه؛ فتتزاحم على جيرانها
وترتدّ؛ فيدفع الجيران الصفّ التالي بدورهم. والدفعة --- تضاغط
مسافر في الحشد --- تنطلق من المنبع إلى الخارج في كل اتجاه،
\omterm{def:g7:states-of-matter:molecule}{جزيء} يناقل جزيئًا، مع أن كل \omterm{def:g7:states-of-matter:molecule}{جزيء} نفسه لا يفعل إلا أن يهتزّ حول
موضعه. \omterm{def:g3:sound-around-us:sound}{فالصوت} هو \emph{التناقل}، لا العدّاؤون: الرسالة تسافر،
والرُّسل يمكثون في بيوتهم.
\end{proposition}

\begin{example}[الجرّة، مشروحة أخيرًا]\label{ex:g7:sound-production:jar}
الجرس المُسكَت المشهور يعترف الآن بآليّته. فشفط \omterm{def:g2:air-around-us:air}{الهواء} يزيل
الحشد: فجدران الجرس المرتعشة تدفع في لا شيء، ولا يتكوّن تناقل،
ولا تعبر رسالة الزجاج. والجرس لم يتوقّف قطّ --- بل فقد رُسله.
ويتبع ذلك فورًا صمت الفضاء العظيم: فليس بين النجوم حشد يحمل
صرخة.
\end{example}

\begin{example}[عالٍ وخافت ومتهالك]\label{ex:g7:sound-production:loud}
الدفعة الأقوى تصنع تضاغطًا أشدّ: صوتًا أعلى. وإذ ينتشر التناقل
--- حِلَق بركة في ثلاثة أبعاد --- تُقتسم كل ضغطة على قشرة من
الحشد أوسع فأوسع، فتضعف مع البعد: ولهذا تخبو الصيحات، ولهذا
تكون موسيقى الجار همسًا عبر جدارين. ولم يحتج شيء من هذا إلى
قوانين جديدة: ففيزياء الحشد تدفع الثمن كله.
\end{example}

\section{لكل وسط وتيرته}

\begin{proposition}[سرعة الصوت تتوقّف على الوسط]\label{prop:g7:sound-production:speeds}
\omterm{def:g7:motion-average-speed:formula}{سرعة} التناقل خاصّية من خواصّ الحشد:
\begin{enumerate}
  \item في \omterm{def:g2:air-around-us:air}{الهواء}: نحو \qty{340}{m/s}؛
  \item في الماء: نحو \qty{1500}{m/s} --- أسرع أربع مرات؛
  \item في الفولاذ: نحو \qty{5000}{m/s} --- أسرع من \omterm{def:g2:air-around-us:air}{الهواء}
        خمس عشرة مرة.
\end{enumerate}
والنمط: كلما تلامست \omterm{def:g7:states-of-matter:molecule}{جزيئات} \omterm{def:g7:sound-production:medium}{الوسط} تلامسًا أشدّ، أسرعت في تسليم
الدفعة. \omterm{def:g7:states-of-matter:molecule}{فجزيئات} \omterm{def:g2:air-around-us:air}{الهواء} الطائرة لا بدّ أن تعبر الفجوات أولًا كي
تسلّم؛ وحشد الماء المتلامس يناقل بنشاط؛ وصفوف الفولاذ المقفلة
تمرّر الرسالة يدًا بيد تقريبًا.
\end{proposition}

\begin{example}[مشاهد قديمة وأعداد جديدة]\label{ex:g7:sound-production:scenes}
أذن الكشّاف على القضيب: تناقل الفولاذ يسبق تناقل \omterm{def:g2:air-around-us:air}{الهواء} بعامل
خمسة عشر --- فينذر القضيب قبل الهدير بزمن طويل. وأغاني الحيتان
تعبر البحار: تناقل الماء السريع السخيّ. وأمّا صوتك أنت فيبدو
غريبًا في التسجيلات، وذلك جزئيًّا لأنك حين تتكلّم تسمع نفسك عبر
تناقل عظم الجمجمة إلى جانب \omterm{def:g2:air-around-us:air}{الهواء} --- وسطان، ونسختان، ولا يسمع
سواك إلا نسخة \omterm{def:g2:air-around-us:air}{الهواء} وحدها.
\end{example}

\section{الصوت بساعة إيقاف}

\begin{method}[قياس سرعة الصوت]\label{met:g7:sound-production:measure}
أقدم طريقة، وما زالت تعمل في أيّ ملعب رياضي:
\begin{enumerate}
  \item يقف شريك على مسافة مقيسة قدرها \qty{680}{m} (طولا
        ملعب وزيادة --- قِسها بالخطوات أو استعمل علامات
        الملعب) ومعه غطاءا قِدرين؛
  \item فيصفق بالغطاءين فوق رأسه: فأنت \emph{ترى} الضربة في
        الحال (فرحلة \omterm{def:g1:light-and-shadows:source}{الضوء} كأنها لحظية)، وتشغّل ساعة الإيقاف
        عند ومضة الحركة؛
  \item وتوقفها عند وصول الرنين: وتوقّع ما يقارب \qty{2}{s}
        تمامًا؛
  \item ثم احسب: $v = d/t = 680 \div 2 = \qty{340}{m/s}$.
\end{enumerate}
كرّر وخذ المتوسّط --- فأزمنة ردّ الفعل تتبعثر، كما علّم فصل
\omterm{def:g6:measurement-in-science:measuring}{القياس} الأمين. وعدّ الرعد هو هذه الطريقة معكوسة: فإذ تعرف $v$،
تعطيك الثواني الصامتة المسافة.
\end{method}

\begin{example}[الصدى، محسوبًا]\label{ex:g7:sound-production:echo}
تصفيقة نحو جُرف تعود في \qty{3}{s}. فقد سافر \omterm{def:g3:sound-around-us:sound}{الصوت} إلى الجرف
وعاد: $d_{\text{الذهاب والإياب}} = 340 \times 3 = \qty{1020}{m}$،
فالجرف يقف على النصف: \qty{510}{m}. وقاعدة كل حساب صدى: الزمن
المقيس يدفع ثمن الرحلة ذهابًا وإيابًا --- ونسيان ``والإياب'' هو
الزلّة الكلاسيكية، وهي تضاعف كل جواب خطأً.
\end{example}

\begin{example}[السونار، بجدّ]\label{ex:g7:sound-production:sonar}
سونار سفينة المسح ينقر إلى الأسفل؛ فيعود صدى القاع في
\qty{0.6}{s}. وفي \emph{الماء}: $d = 1500 \times 0.6 =
\qty{900}{m}$ ذهابًا وإيابًا: فالعمق \qty{450}{m}. والدلافين
والخفافيش تجري الحساب نفسه بالغريزة؛ والسفن تطبعه على الخرائط.
ولاحظ الانضباط: \omterm{def:g7:sound-production:medium}{الوسط} هو الذي يختار \omterm{def:g7:motion-average-speed:formula}{السرعة} --- وإطعام صدى تحت
الماء عددَ \omterm{def:g2:air-around-us:air}{الهواء} $340$ هو زلّة الفصل الكلاسيكية الثانية.
\end{example}

\begin{remark}[ما لا تزال ``السرعة'' ليست إياه]\label{rem:g7:sound-production:light}
\qty{5000}{m/s} للفولاذ تبدو بطولية --- حتى يُستذكر \omterm{def:g1:light-and-shadows:source}{الضوء}. ففي
العاصفة تُعدّ رحلة الوميض لحظية بينما يهرول الرعد كيلومتره في
ثلاث ثوانٍ؛ ولا \omterm{def:g7:sound-production:medium}{وسط}، لا الفولاذ ولا الماس، يقرّب \omterm{def:g3:sound-around-us:sound}{الصوت} من وتيرة
\omterm{def:g1:light-and-shadows:source}{الضوء} ولو على مرمى صيحة. وأمّا كم يجري \omterm{def:g1:light-and-shadows:source}{الضوء} بالضبط --- وكيف
وقّت البشر أخيرًا شيئًا يعبر غرفة في جزء من مئة مليون من
الثانية --- فقصة السنة القادمة، ولها فصل خاصّ بها.
\end{remark}

\section{تمارين}

ما \omterm{def:g7:sound-production:medium}{الوسط}؟ سمِّ ثلاثة، وسمِّ ``اللاوسط'' الوحيد الذي لا يستطيع

\begin{exercise}[$\star$]\label{exo:g7:sound-production:2}
في صورة التناقل، ما الذي يسافر من المنبع إلى الأذن --- وماذا

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:sound-production:3}
اشرح جرس الجرّة المُسكَت \omterm{def:g7:states-of-matter:molecule}{بالجزيئات} --- ولماذا تكون معارك الفضاء

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:sound-production:4}
أعطِ السرعات الثلاث في جدول الفصل، والنمط الذي يربط ترتيب

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:sound-production:5}
لماذا تسبق رسالة قضيب الكشّاف رسالة \omterm{def:g2:air-around-us:air}{الهواء}؟ وبأيّ عامل تقريبًا؟

\begin{exercise}[$\star$]\label{exo:g7:sound-production:6}
في \omterm{def:g6:measurement-in-science:measuring}{قياس} الملعب، لماذا تكون مشاهدة ضربة الغطاءين كمسدّس انطلاق

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:sound-production:7}
صدى تصفيقة عن جدار مستودع يعود في \qty{1}{s}. كم يبعد الجدار؟

\begin{exercise}[$\star$]\label{exo:g7:sound-production:8}

\begin{exercise}[$\star\star$]\label{exo:g7:sound-production:9}
ألعاب نارية فوق الخليج: ترى الانفجار، وتسمعه بعده

\begin{exercise}[$\star\star$]\label{exo:g7:sound-production:10}

\begin{exercise}[$\star\star$]\label{exo:g7:sound-production:11}

\begin{exercise}[$\star\star\star$]\label{exo:g7:sound-production:12}
صمّم قياسًا \omterm{def:g7:motion-average-speed:formula}{لسرعة} \omterm{def:g3:sound-around-us:sound}{الصوت} \emph{في الماء} على بحيرة هادئة،

\section{مسألة: مسح الأخدود}

\begin{problem}[{مسألة نهاية الأسبوع --- رسم خريطة الأخدود
الصامت بالأذن؛ أصداء وقضبان وأعماق نهر؛ عُدّة
المسّاح}]\label{pb:g7:sound-production:1}
يرسم فريق مسح خريطة أخدود ناءٍ بساعات إيقاف، \omterm{def:g6:measurement-in-science:measuring}{ووحدة} سونار
واحدة، وفيزياء هذا الفصل. والسرعات: \omterm{def:g2:air-around-us:air}{الهواء} \qty{340}{m/s}،
والماء \qty{1500}{m/s}، والفولاذ \qty{5000}{m/s}.

\medskip\noindent\textbf{القسم الأول --- العروض \omterm{ex:g4:how-sound-travels:echo}{بالصدى}.}
\begin{enumerate}
  \item من الحافة الشرقية، يعود صدى تصفيقة عن الجدار الغربي في
        \qty{4}{s}. كم يبلغ عرض الأخدود هنا؟
  \item وعند موضع أضيق يعود \omterm{ex:g4:how-sound-travels:echo}{الصدى} في \qty{1.5}{s}. ما العرض؟
  \item ومسّاح يقف \emph{بين} جدارين متوازيين يصفّق مرة واحدة
        فيسمع صدَيين متمايزين: بعد \qty{1}{s} وبعد \qty{2}{s}.
        كم يبعد كل جدار، وما عرض الأخدود على هذا الخطّ؟
  \item ولماذا يأتي أكبر خطأ في قياسات القسم الأول من يد ساعة
        الإيقاف لا من الصيغة؟ (استشهد بحكمة قاعدة تكرار
        \omterm{def:g6:measurement-in-science:measuring}{القياس}.)
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- النهر في الأسفل.}
\begin{enumerate}[resume]
  \item \omterm{def:g6:measurement-in-science:measuring}{وحدة} السونار، طافية على النهر، تنقر القاع: فيعود \omterm{ex:g4:how-sound-travels:echo}{الصدى}
        في \qty{0.2}{s}. ما عمق النهر؟
  \item \omterm{def:g6:measurement-in-science:measuring}{والوحدة} نفسها منقورة جانبًا نحو صخرة مغمورة تعيد صدى
        في \qty{0.4}{s}. ما المسافة إلى الصخرة؟
  \item ويقترح زميل توفير البطاريات: ``اصرخ من القارب وقِس زمن
        صدى القاع بالأذن.'' اذكر سببين فيزيائيّين لفشل هذا حيث
        ينجح السونار. (فكّر في الحدّ بين الوسطين، وفيما يجب أن
        يفعله تناقل الصرخة الهوائي عند السطح.)
  \item وقاع النهر يهبط هبوطًا حادًّا في \omterm{def:g7:sound-production:medium}{وسط} المجرى: تنبّأ بما
        يفعله زمن صدى السونار بينما ينجرف القارب عابرًا
        الهبوط.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- قضيب المنجم القديم.}
قضيب مستقيم مهجور يمتدّ \qty{2}{km} على أرض الأخدود حتى بوابة
المنجم.
\begin{enumerate}[resume]
  \item ضربة مطرقة على القضيب عند البوابة: احسب زمنَي الوصول
        عند طرف المسّاحين --- عبر الفولاذ، وعبر \omterm{def:g2:air-around-us:air}{الهواء}.
  \item ويسمع الفريق رنين القضيب ودويّ \omterm{def:g2:air-around-us:air}{الهواء} يفصل بينهما ذلك
        الفارق. اشرح كيف يستطيع \emph{الفارق وحده} أن يقيس
        المسافة إلى البوابة لو كانت \qty{2}{km} مجهولة (صِف
        الاستدلال؛ والحساب هدية القسم الثالث للمتحمّسين).
  \item ويملأ الضباب الأخدود ليلًا --- فلا نفع للمصابيح ولا
        للمرايا. أيّ قنوات \omterm{def:g6:measurement-in-science:measuring}{القياس} الثلاث عند الفريق (\omterm{def:g3:sound-around-us:sound}{صوت}
        \omterm{def:g2:air-around-us:air}{الهواء}، \omterm{def:g3:sound-around-us:sound}{صوت} الماء، \omterm{def:g3:sound-around-us:sound}{صوت} القضيب) يزعجها الضباب أقلّ ما
        يكون، ولماذا؟
  \item اكتب سجلّ المسّاح: ثلاث جمل --- قاعدة الذهاب
        والإياب، وقاعدة ``\omterm{def:g7:sound-production:medium}{الوسط} يختار \omterm{def:g7:motion-average-speed:formula}{السرعة}''، وأيّ أداة
        واحدة (العين أم الأذن) بدأت كل توقيت في الأخدود،
        ولماذا يجوز الوثوق بها بوصفها ``لحظية''.
\end{enumerate}
\end{problem}
