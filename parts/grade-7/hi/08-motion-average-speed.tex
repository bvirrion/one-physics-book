\chapter{गति के आलेख और औसत चाल}\label{ch:g7:motion-average-speed}

दो साल पहले \omterm{def:g7:motion-average-speed:formula}{चाल} का मतलब था ``एक घंटे में तय किए गए किलोमीटर'', जो
उँगलियों पर गिने जाते थे। अब, अक्षरों और आलेखों से लैस होकर, वह विचार
दाँत निकाल लेता है: एक सूत्र जो तीन दिशाओं में हिसाब लगाता है, एक चित्र
जो पूरी यात्रा एक नज़र में दिखा देता है --- और औसतों के बारे में वह
मशहूर फंदा, जिसमें बड़े-बड़े रोज़ फँसते हैं।

\section{सूत्र}

\begin{definition}[चाल, सूत्र से]\label{def:g7:motion-average-speed:formula}
किसी यात्रा (या उसके किसी हिस्से) के लिए, जो एक-सी रफ़्तार से तय की
गई हो, \emph{चाल}\index{चाल} $v$ दूरी $d$ को यात्रा के समय $t$ से भाग
देने पर मिलती है:
\[
v = \frac{d}{t}.
\]
$d$ किलोमीटर में और $t$ \omterm{ex:g2:measuring-time:clock}{घंटों} में हो, तो $v$ \emph{किलोमीटर प्रति
घंटा} (\unit{km/h}) में बोलती है; और \omterm{def:g2:measuring-length:units}{मीटर} तथा सेकंड में हो, तो
\emph{\omterm{def:g2:measuring-length:units}{मीटर} प्रति सेकंड} (\unit{m/s}) में --- वैज्ञानिकों का प्यारा
\omterm{def:g6:measurement-in-science:measuring}{मात्रक}। \omterm{def:g7:volume-mass-density:formula}{घनत्व} के सूत्र की तरह यह भी तीनों दिशाओं में हिसाब लगाता है:
$v = d/t$, \ $d = v \times t$, \ $t = d/v$।
\end{definition}

\begin{example}[नुस्ख़े काम पर]\label{ex:g7:motion-average-speed:recipes}
एक रेलगाड़ी $2$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{240}{km} तय करती है: $v = 240 \div 2 =
\qty{120}{km/h}$। उसी रफ़्तार से $5$ \omterm{ex:g2:measuring-time:clock}{घंटों} में कितनी दूर? $d = 120
\times 5 = \qty{600}{km}$। और \qty{300}{km} के लिए कितना समय? $t = 300
\div 120 = 2.5$ घंटे --- यानी दो घंटे तीस मिनट (उस बाग़ी का ध्यान
रखो: $0.5$ घंटा $30$ मिनट होता है, कभी ``$50$'' नहीं)।
\end{example}

\begin{example}[दो मात्रक, एक ही चाल]\label{ex:g7:motion-average-speed:units}
एक धावक \qty{10}{s} में \qty{100}{m} दौड़ता है: $v = \qty{10}{m/s}$।
सड़क वाले मात्रकों में यह कितनी तेज़ हुई? एक घंटे में --- यानी $3600$
सेकंड में --- धावक $10 \times 3600 = \qty{36000}{m} = \qty{36}{km}$
तय कर लेता; यानी $\qty{10}{m/s} = \qty{36}{km/h}$। आम विनिमय दर: हर
\unit{m/s} \qty{3.6}{km/h} के बराबर है --- क्योंकि एक घंटे में $3600$
सेकंड होते हैं और एक किलोमीटर में सिर्फ़ $1000$ \omterm{def:g2:measuring-length:units}{मीटर}। \omterm{def:g3:sound-around-us:sound}{ध्वनि} की
\qty{340}{m/s}, सड़क वाले मात्रकों में: \qty{1200}{km/h} से भी
ज़्यादा।
\end{example}

\section{यात्रा, एक चित्र के तौर पर}

\begin{proposition}[आलेख के नियम]\label{prop:g7:motion-average-speed:graphs}
तय की गई दूरी को समय के सामने आलेखित करो, और गति अपनी आत्मकथा लिख
देती है:
\begin{enumerate}
  \item एक-सी गति एक \emph{सीधी} रेखा खींचती है --- बराबर समयों में
        बराबर दूरियाँ, क़दम दर क़दम;
  \item रेखा जितनी \emph{खड़ी}, गति उतनी तेज़: खड़ापन ही आँखों के
        सामने रखी हुई \omterm{def:g7:motion-average-speed:formula}{चाल} है;
  \item और \emph{सपाट} हिस्सा एक ठहराव है: समय गुज़रता है, दूरी खड़ी
        रह जाती है।
\end{enumerate}
मोड़ बदलाव की ख़बर देते हैं: ऊपर की ओर मुड़ती रेखा यानी तेज़ होना;
सपाट पड़ती रेखा यानी धीमा होना।
\end{proposition}

\begin{method}[दूरी--समय आलेख पढ़ना]\label{met:g7:motion-average-speed:read}
\begin{enumerate}
  \item सबसे पहले अक्षों और उनके मात्रकों की जाँच करो --- मिनट या
        घंटे, \omterm{def:g2:measuring-length:units}{मीटर} या किलोमीटर;
  \item रेखा को उसके मोड़ों पर हिस्सों में बाँटो, और हर एक पर नाम
        चढ़ाओ: सीधी-और-खड़ी, सीधी-और-नरम, सपाट;
  \item किसी भी सीधे हिस्से के लिए उसकी चढ़ाई और उसकी लंबाई पढ़ो ---
        यानी कमाई हुई दूरी और लगा हुआ समय --- और भाग दे दो: यही उस
        हिस्से की \omterm{def:g7:motion-average-speed:formula}{चाल} है;
  \item और पूरी कहानी के लिए आख़िरी समय पर कुल दूरी पढ़ो --- और यह
        याद रखो कि सपाट हिस्से भी कुल समय का हिस्सा हैं।
\end{enumerate}
\end{method}

\begin{example}[हरकारा, खोला हुआ]\label{ex:g7:motion-average-speed:decoded}
इस विधि को चित्र पर लगाओ। पहला हिस्सा: $20$ मिनट में \qty{10}{km} ---
यानी घंटे का एक तिहाई --- इसलिए $v = 10 \div \tfrac{1}{3} =
\qty{30}{km/h}$। दूसरा: मिनट $20$ से $35$ तक सपाट --- यानी सामान
पहुँचाने का ठहराव। तीसरा: $25$ मिनट में \qty{10}{km}, यानी
\qty{24}{km/h} से ज़रा-सा कम। कुल: एक घंटे में \qty{20}{km} --- और यही
अध्याय का अगला विचार थाली में सजाकर हमें थमा देता है।
\end{example}

\section{औसत चाल --- और उसका मशहूर फंदा}

\begin{definition}[औसत चाल]\label{def:g7:motion-average-speed:average}
पूरी यात्रा की \emph{औसत चाल}\index{औसत चाल} कुल दूरी को \emph{कुल}
समय से भाग देने पर मिलती है --- ठहरावों समेत:
\[
v_{\text{औसत}} = \frac{d_{\text{कुल}}}{t_{\text{कुल}}}.
\]
यह वह अकेली एक-सी रफ़्तार है जो उतनी ही सड़क उतने ही कुल समय में तय कर
देती --- हरकारे की एक घंटे में \qty{20}{km}: औसत \qty{20}{km/h}, जबकि
पहिए एक बार भी ठीक उसी \omterm{def:g7:motion-average-speed:formula}{चाल} पर नहीं घूमे।
\end{definition}

\begin{proposition}[फंदा]\label{prop:g7:motion-average-speed:trap}
किसी यात्रा की \omterm{def:g7:motion-average-speed:average}{औसत चाल} आम तौर पर उसकी चालों का बीच वाला बिंदु
\emph{नहीं} होती। धीमे हिस्से तेज़ हिस्सों से ज़्यादा समय खा जाते हैं,
इसलिए औसत में उनका वज़न ज़्यादा पड़ता है। भरोसे लायक सिर्फ़ पूरा
नुस्ख़ा है --- कुल दूरी बटा कुल समय; जबकि ख़ुद \omterm{def:g7:motion-average-speed:formula}{चाल} के अंकों का औसत
निकालना इस अध्याय की सबसे लुभावनी ग़लत \omterm{def:g7:motion-average-speed:formula}{चाल} है।
\end{proposition}

\begin{example}[फंदा, चलता हुआ]\label{ex:g7:motion-average-speed:sprung}
एक साइकिल सवार \qty{30}{km/h} से \qty{60}{km} जाता है, और उतने ही
\qty{60}{km} \qty{20}{km/h} से वापस आता है। ``औसत: \qty{25}{km/h}''?
ईमानदारी से जाँचो। जाते हुए: $t = 60 \div 30 = 2$ घंटे। लौटते हुए:
$t = 60 \div 20 = 3$ घंटे। पूरी यात्रा: $5$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{120}{km}
--- $v_{\text{औसत}} = \qty{24}{km/h}$। धीमे आधे हिस्से ने पाँच में से
तीन घंटे हड़प लिए, और औसत को बीच वाले बिंदु से नीचे घसीट लिया। किसी
एक आधे हिस्से पर तुम जितना तेज़ चलोगे, वह आधा हिस्सा उतनी ही कम देर
तक मौजूद रहेगा।
\end{example}

\begin{remark}[गतिमापी क्या जानता है]\label{rem:g7:motion-average-speed:speedometer}
\omterm{def:g7:motion-average-speed:average}{औसत चाल} पूरी यात्रा का बयान देती है; जबकि गतिमापी की सुई एक अलग सवाल
का जवाब देती है --- \emph{इसी वक़्त कितनी तेज़}। आलेख पर ``इसी वक़्त''
रेखा के \emph{किसी एक बिंदु पर} खड़ेपन में बसता है --- सीधे हिस्से पर
आसानी से दिख जाने वाला, और जहाँ रेखा मुड़ती है वहाँ बारीक। ``किसी
बिंदु पर खड़ापन'' को सटीक बनाना गणित के सबसे बड़े आविष्कारों में से एक
है, और वह हाई स्कूल के आख़िर में तुम्हारा इंतज़ार कर रहा है। तब तक:
सीधे हिस्सों को अंक मिलते हैं, और मुड़ती रेखाओं को कहानियाँ।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:1}
\omterm{def:g7:motion-average-speed:formula}{चाल} का सूत्र और उसके दो रोज़मर्रा \omterm{def:g6:measurement-in-science:measuring}{मात्रक} लिखो। दूरी कौन-सा नुस्ख़ा
निकालता है? और समय?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:2}
एक नाव $3$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{45}{km} तय करती है; एक ख़रगोश \qty{8}{s} में
\qty{100}{m} दौड़ता है। दोनों की \omterm{def:g7:motion-average-speed:formula}{चाल} निकालो, हर एक अपने सहज \omterm{def:g6:measurement-in-science:measuring}{मात्रक}
में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:3}
विनिमय दर से बदलो: \qty{5}{m/s} को \unit{km/h} में; \
\qty{72}{km/h} को \unit{m/s} में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:4}
आलेख के तीनों नियम बताओ। धीरे-धीरे सपाट पड़ता मोड़ क्या ख़बर देता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:5}
हरकारे के आलेख पर: किन मिनटों के बीच रफ़्तार सबसे नरम है (पर शून्य
नहीं)? बिना हिसाब लगाए यह कैसे बताओगे?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:6}
किसी पैदल चलने वाले का आलेख ($30$ मिनट, \qty{2}{km}) और ($60$ मिनट,
\qty{4}{km}) बिंदुओं से गुज़रती एक सीधी रेखा दिखाता है। एक-सी गति या
बदलती हुई? \unit{km/h} में \omterm{def:g7:motion-average-speed:formula}{चाल}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:7}
\omterm{def:g7:motion-average-speed:average}{औसत चाल} की परिभाषा दो। एक पैदल यात्रा: $4$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{12}{km},
जिसमें एक घंटे का जलपान भी शामिल है। \omterm{def:g7:motion-average-speed:average}{औसत चाल} --- और \emph{चलते हुए}
की \omterm{def:g7:motion-average-speed:average}{औसत चाल}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:8}
बराबर लंबाई के तेज़ हिस्से के मुक़ाबले धीमा हिस्सा औसत में ज़्यादा
``भारी'' क्यों पड़ता है? उत्तर समय से दो, सूत्रों से नहीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:9}
फंदा दोबारा चलाओ: \qty{15}{km/h} से \qty{30}{km} जाना, और
\qty{30}{km/h} से \qty{30}{km} लौटना। अंदाज़े वाला बीच का बिंदु,
ईमानदार औसत --- और यात्रा के किस आधे हिस्से ने घड़ी का ज़्यादातर
हिस्सा हड़पा?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:10}
इस सफ़र का आलेख खींचो (या ठीक-ठीक बयान करो): आधे घंटे तक एक-सी
\qty{40}{km/h}; चौथाई घंटे तक ठहराव; चौथाई घंटे तक एक-सी
\qty{60}{km/h}। फिर सफ़र की \omterm{def:g7:motion-average-speed:average}{औसत चाल} निकालो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:11}
तूफ़ान की गिनती, उन्नत रूप में: कौंध के \qty{6}{s} बाद गरज पहुँचती
है। \omterm{def:g3:sound-around-us:sound}{ध्वनि} की \qty{340}{m/s} लेकर $d = v \times t$ से तूफ़ान की दूरी
निकालो --- और ``किलोमीटर के लिए तीन से भाग दो'' वाले पुराने नियम को
अपने सूत्र के सामने जाँचो भी।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:motion-average-speed:12}
एक पुरानी पहेली, जो अपने हर मिनट के लायक़ है: कोई \omterm{def:g4:conductors-insulators:conductor}{चालक} किसी सफ़र की
\emph{दूरी} का पहला आधा हिस्सा \qty{30}{km/h} से तय करता है और कुल
औसत \qty{60}{km/h} चाहता है। \qty{60}{km} के सफ़र पर कुल दूरी बटा कुल
समय वाले नुस्ख़े से दिखाओ कि दूसरा आधा हिस्सा शून्य समय में तय करना
पड़ेगा: यानी यह ख़्वाहिश मुश्किल भर नहीं, नामुमकिन है।
\end{exercise}

\section{समस्या: हरकारे का शुक्रवार}

\begin{problem}[{सप्ताहांत समस्या --- एक साइकिल हरकारा, शहर का एक
शुक्रवार; प्रेषक का आलेख, और बोनस वाली परची में छिपा
फंदा}]\label{pb:g7:motion-average-speed:1}
प्रेषक हरकारा लेना का शुक्रवार एक दूरी--समय आलेख पर दर्ज करता है और
उसी से उसकी मज़दूरी तय करता है। उस परची की जाँच तुम्हें करनी है।

\medskip\noindent\textbf{भाग I --- सुबह, आलेख से।}
आलेख दिखाता है: ($0$ मिनट, \qty{0}{km}) से ($30$ मिनट, \qty{9}{km})
तक सीधी चढ़ाई; $45$ मिनट तक सपाट; वहाँ से ($75$ मिनट, \qty{15}{km})
तक सीधी रेखा; और $90$ मिनट तक सपाट।
\begin{enumerate}
  \item सुबह की कहानी शब्दों में कहो: हिस्से, ठहराव, और यह भी कि वे
        ठहराव संभवतः क्या रहे होंगे।
  \item पहले हिस्से पर \unit{km/h} में \omterm{def:g7:motion-average-speed:formula}{चाल}?
  \item चलने वाले दूसरे हिस्से पर \omterm{def:g7:motion-average-speed:formula}{चाल}?
  \item ठहरावों समेत, पूरे $90$ मिनट की सुबह पर लेना की \emph{औसत}
        \omterm{def:g7:motion-average-speed:formula}{चाल}?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दोपहर बाद, सूत्र से।}
\begin{enumerate}[resume]
  \item दोपहर बाद का पहला चरण: एक-सी \qty{24}{km/h} से \qty{12}{km}।
        इसमें कितने मिनट लगे?
  \item दूसरा चरण: ऊपरी बस्ती में एक सामान पहुँचाना, \qty{18}{km/h}
        से \qty{20}{minutes}। कितने किलोमीटर?
  \item तीसरा चरण: गोदाम तक की लंबी सवारी, \qty{10}{km}, जो $25$ मिनट
        में पूरी हुई। एक-सी रफ़्तार वाली \omterm{def:g7:motion-average-speed:formula}{चाल} \unit{km/h} में?
  \item पूरी दोपहर: दूरियाँ और समय जोड़ो (सिर्फ़ चरण, बीच के
        विश्राम नहीं): चलते रहने के दौरान पहियों ने कौन-सी \omterm{def:g7:motion-average-speed:average}{औसत चाल}
        बनाए रखी?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- बोनस वाली परची का फंदा।}
बोनस का नियम: ``पूरे दौरे पर \qty{20}{km/h} से ऊपर की \omterm{def:g7:motion-average-speed:average}{औसत चाल} तेज़
सवार वाला बोनस दिलाती है।''
\begin{enumerate}[resume]
  \item लेना का पूरा शुक्रवार: $3$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{37}{km} (सारे
        ठहराव शामिल)। क्या उसे बोनस मिलेगा?
  \item उसका साथी मैक्स दावा ठोकता है: ``मैंने अपना आधा दौरा
        \qty{30}{km/h} से चलाया और आधा \qty{15}{km/h} से --- यानी
        औसत $22.5$!'' प्रेषक जाँचता है: दोनों आधे हिस्से \qty{12}{km}
        के थे। मैक्स का सच्चा औसत निकालो और उसके बोनस पर फ़ैसला दो।
  \item मैक्स को समय की ज़बान में समझाओ कि उसका $22.5$ कहाँ ग़लत हो
        गया।
  \item लेना हरकारों के लिए एक ज़्यादा इंसाफ़ वाला बोनस नियम सुझाती
        है --- ऐसा, जो सामान पहुँचाने के ठहरावों की सज़ा न दे। एक
        नियम सुझाओ (आलेख जानता है कैसे), और बताओ कि वह कौन-सी \omterm{def:g7:motion-average-speed:formula}{चाल}
        मापता है।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- प्रेषक का सबक़।}
\begin{enumerate}[resume]
  \item शुक्रवार की समीक्षा: प्रेषक के तीन वाक्य लिखो --- आलेख पर
        सीधे हिस्से, सपाट हिस्से और ज़्यादा खड़े हिस्से का क्या-क्या
        मतलब है --- और साथ में चालों का औसत निकालने के बारे में वह
        अकेली चेतावनी भी।
\end{enumerate}
\end{problem}
