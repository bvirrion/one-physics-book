\chapter{परिपथ: फंदे, श्रेणी, समांतर}\label{ch:g7:circuits-series-parallel}

चार साल के परिपथों ने तुम्हारा औज़ार-डिब्बा भर दिया है: फंदे, आरेख,
और जोड़ने के दो बड़े तरीक़े। पर एक पात्र इस पूरे समय बिना नाम के काम
करता रहा --- वह ``कुछ'' जो बहता है। इस साल उसे उसका नाम मिलता है,
उसकी दिशा मिलती है, और उसका सबसे अनोखा गुण भी: वह कभी ख़र्च नहीं होता।
धारा के मंच पर आते ही श्रेणी और समांतर नुस्ख़े नहीं रह जाते, नियम बन
जाते हैं।

\section{धारा मंच पर आती है}

\begin{definition}[विद्युत धारा]\label{def:g7:circuits-series-parallel:current}
\emph{विद्युत धारा}\index{विद्युत धारा} वह बहाव है जो किसी \omterm{def:g3:first-electric-circuit:circuit}{बंद परिपथ}
में चक्कर लगाता है: आवेशित कणों की एक भीड़ --- कल्पना से परे इतने कि
गिने न जा सकें, और कल्पना से परे इतने छोटे, और \omterm{def:g7:states-of-matter:molecule}{अणुओं} की दुनिया के
रिश्तेदार --- जिन्हें \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का धक्का तारों की धातु में से क़तार बाँधकर
चला देता है। फंदा नहीं, तो क़तार नहीं: धारा तभी तक रहती है जब तक सड़क
बंद हो। (ये चलने वाले हैं ठीक-ठीक क्या, और धातु उन्हें अपने में से
कैसे गुज़रने देती है, यह बढ़िया कहानी इस पुस्तक के आख़िरी वर्षों के
लिए बचाई गई है।)
\end{definition}

\begin{proposition}[परंपरागत दिशा]\label{prop:g7:circuits-series-parallel:direction}
दुनिया भर की परंपरा के अनुसार परिपथ आरेखों में धारा को \emph{\omterm{def:g3:first-electric-circuit:battery}{बैटरी} के
बाहर, $+$ सिरे से निकलकर, परिपथ का चक्कर लगाकर, $-$ सिरे तक} बहती
दिखाया जाता है। दुनिया के हर आरेख का हर तीर यही परंपरा मानता है ---
और यह परंपरा उस समय से बहुत पहले चुनी गई थी जब कोई ख़ुद उन चलने वालों
से पूछ भी सकता। (और जब भौतिकी आख़िर पूछ सकी, तब उत्तर में एक मशहूर
छोटा-सा मज़ाक़ निकला, जो किसी आगे वाले \omterm{def:g3:sun-days-seasons:year}{वर्ष} के लिए रखा है।)
\end{proposition}

\begin{proposition}[धारा ख़र्च नहीं होती]\label{prop:g7:circuits-series-parallel:notconsumed}
उपकरण धारा को खाते नहीं। जितनी धारा किसी लैंप, \omterm{ex:g6:circuits-and-safety:motor}{मोटर} या \omterm{ex:g6:circuits-and-safety:motor}{बज़र} में घुसती
है उतनी ही पूरी बाहर निकलकर आगे बढ़ जाती है: उपकरण जो लेता है वह
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} के भंडार की \emph{\omterm{def:g5:everyday-energy:energy}{ऊर्जा}} है, जिसे गुज़रती क़तार पहुँचा देती है।
इसलिए किसी भी श्रेणी फंदे में हर उपकरण में से एक ही धारा बहती है ---
और \omterm{def:g3:first-electric-circuit:battery}{बैटरी} तक उतनी ही लौटती है जितनी कभी उससे निकली थी।
\end{proposition}

\begin{example}[कठघरे में एक ग़लतफ़हमी]\label{ex:g7:circuits-series-parallel:trial}
लुभावनी ग़लती: ``पहला \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} कुछ धारा ख़र्च कर देता है, इसलिए दूसरे को
मंद चमकना चाहिए।'' ख़ुद परिपथ इसका खंडन कर देता है। \emph{एक जैसे} दो
\omterm{def:g3:first-electric-circuit:bulb}{बल्ब} \omterm{def:g4:series-circuits:series}{श्रेणी में} जोड़ो और देखो: दोनों ठीक बराबर चमकते हैं --- और उन्हें
आपस में बदल देने से कुछ नहीं बदलता। अगर रास्ते में धारा ख़र्च होती,
तो \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} दो हमेशा मंद रहता, चाहे वहाँ कोई भी \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} बैठा हो। ऐसा कभी
नहीं होता। धारा पूरी पहुँचती है; जो जोड़ी बाँटती है वह \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का
\emph{धक्का} है, और इसीलिए दोनों अकेले \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} से मंद हैं --- और बराबर
मंद।
\end{example}

\section{दोनों कुनबे, नियमों के रूप में}

\begin{proposition}[श्रेणी का नियम]\label{prop:g7:circuits-series-parallel:series}
\omterm{def:g4:series-circuits:series}{श्रेणी परिपथ} में --- यानी एक ही अकेले फंदे में --- हर उपकरण में से
वही एक धारा बहती है: एक सड़क, एक क़तार, कोई अपवाद नहीं। नतीजे, जिनकी
अब व्याख्या भी है: एक दरार पूरी क़तार को एक साथ हर जगह रोक देती है;
एक जैसे उपकरण काम बराबर बाँट लेते हैं, चाहे वे कहीं भी बैठे हों; और
जो क़तार सब में से एक जैसी गुज़रती हो, उसके लिए फंदे पर उपकरणों का क्रम
मायने रख ही नहीं सकता।
\end{proposition}

\begin{proposition}[समांतर का नियम]\label{prop:g7:circuits-series-parallel:parallel}
किसी \omterm{def:g5:series-parallel-circuits:parallel}{संधि} पर धारा \emph{बँट} जाती है: क़तार का एक हिस्सा हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा}
पकड़ता है, और दूर वाली \omterm{def:g5:series-parallel-circuits:parallel}{संधि} पर सारे हिस्से मिलकर फिर पूरे हो जाते हैं
--- न कुछ खोया, न कुछ जुड़ा। हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} को \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का पूरा धक्का मिलता है,
इसलिए हर उपकरण ऐसे काम करता है मानो वह अकेला हो; और रुकी हुई \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} बस
किसी को नहीं भेजती, जबकि बाक़ी \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} की क़तार चलती रहती है।
\end{proposition}

\begin{example}[बिजली वाले की तरह चमक पढ़ना]\label{ex:g7:circuits-series-parallel:brightness}
एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, और एक जैसे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}। अकेला: पूरी चमक --- यही हमारा संदर्भ है।
\omterm{def:g4:series-circuits:series}{श्रेणी में} दो: दोनों बराबर मंद --- एक क़तार, बँटा हुआ धक्का। \omterm{def:g5:series-parallel-circuits:parallel}{समांतर में}
दो: दोनों पूरी चमक पर --- हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} को पूरा धक्का (और \omterm{def:g3:first-electric-circuit:battery}{बैटरी} लगभग दोगुनी
तेज़ी से ख़ाली होती है: मुफ़्त कुछ भी नहीं)। \omterm{def:g4:series-circuits:series}{श्रेणी में} तीन: और भी मंद,
और सब बराबर। बिजली वाले की नज़र: मंदी के लिए फंदे के साथी गिनो, और
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} के बिल के लिए शाखाएँ।
\end{example}

\begin{method}[किसी भी लैंप का भविष्य बताना]\label{met:g7:circuits-series-parallel:predict}
कोई आरेख मिले, तो हर लैंप के लिए:
\begin{enumerate}
  \item \emph{अस्तित्व}: क्या $+$ से इस लैंप में से होकर $-$ तक कम
        से कम एक अटूट फंदा है? फंदा नहीं, तो \omterm{def:g1:light-and-shadows:source}{प्रकाश} नहीं।
  \item \emph{सड़क के साथी}: इस लैंप से गुज़रने वाले \emph{हर} फंदे
        में कौन-से उपकरण साझा हैं? वही उसके श्रेणी वाले साथी हैं ---
        साथी जितने ज़्यादा, चमक उतनी मंद।
  \item \emph{\omterm{def:g5:series-parallel-circuits:parallel}{शाखा} के पड़ोसी}: दूसरी \omterm{def:g5:series-parallel-circuits:parallel}{शाखाओं} पर कौन-से उपकरण बैठे हैं?
        वे न इस लैंप को मंद करते हैं, न इसके साथ बुझते हैं।
  \item \emph{\omterm{ex:g3:first-electric-circuit:switch}{स्विच}}: खुला \omterm{ex:g3:first-electric-circuit:switch}{स्विच} ठीक उन्हीं फंदों को मारता है जिन पर
        वह बैठा है --- उसे हटाकर पहला क़दम दोबारा चलो।
\end{enumerate}
\end{method}

\begin{remark}[यह ग़लतफ़हमी आसानी से मरती क्यों नहीं]\label{rem:g7:circuits-series-parallel:whyhard}
``कुछ तो ख़र्च होता ही होगा --- \omterm{def:g3:first-electric-circuit:battery}{बैटरी} जो ख़ाली हो जाती है!'' सच: \omterm{def:g3:first-electric-circuit:battery}{बैटरी}
की संचित \emph{\omterm{def:g5:everyday-energy:energy}{ऊर्जा}} ख़र्च होती है, जिसे गुज़रती क़तार लैंपों को
\omterm{def:g1:light-and-shadows:source}{प्रकाश} और गरमी के रूप में सौंप देती है। पर चलने वाले ख़ुद बिना किसी
नुक़सान के चक्कर लगाते रहते हैं, ठीक स्की-लिफ़्ट की कुर्सियों की तरह,
जो दिन भर स्की करने वालों को ऊपर पहुँचाती हैं और उनमें से एक भी कुर्सी
ख़र्च नहीं होती। दोनों हिसाब अलग-अलग रखो --- धारा चक्कर लगाती है,
\omterm{def:g5:everyday-energy:energy}{ऊर्जा} पहुँचाई जाती है --- और इस पुस्तक का हर परिपथ ईमानदार बना रहता
है। ख़ुद क़तार को गिनने वाला उपकरण अगले साल हमसे मिलेगा।
\end{remark}

से उसका हिस्सा कैसे गला सकती है? (अगला साल इस गला देने को उसका नाम और

\section{समस्या: लाइटहाउस के रखवाले का पटल}

\begin{problem}[{सप्ताहांत समस्या --- पुराने लाइटहाउस की तारबंदी फिर
से; एक बैटरी-भंडार, कई ज़िम्मेदारियाँ; और रखवाले की ख़राबियों भरी
रात}]\label{pb:g7:circuits-series-parallel:1}
लाइटहाउस के बैटरी-भंडार को खिलाना है: विशाल लैंप, कोहरे का भोंपू (एक
ज़ोरदार \omterm{ex:g6:circuits-and-safety:motor}{बज़र}), रखवाले की पढ़ने वाली बत्ती, और सीढ़ियों की बत्ती ---
और साथ ही रखवाले की गड़बड़ियाँ ढूँढ़ने वाली रात भी झेलनी है।

\medskip\noindent\textbf{भाग I --- डिज़ाइन।}
\begin{enumerate}
  \item विशाल लैंप कभी इसलिए मंद नहीं पड़ना चाहिए कि कोई और चीज़ चालू
        हो गई। पूरे ढाँचे के लिए यह कौन-सी तारबंदी माँगता है, और
        \omterm{def:g4:series-circuits:series}{श्रेणी में} कोई भी जोड़ी जहाज़ों के साथ धोखा क्यों करती?
  \item हर ज़िम्मेदारी को अपना \omterm{ex:g3:first-electric-circuit:switch}{स्विच} चाहिए, और रखवाला तूफ़ानों के लिए
        एक मुख्य कटाव-स्विच भी चाहता है। पाँचों \omterm{ex:g3:first-electric-circuit:switch}{स्विच} लगाओ।
  \item कोहरे के भोंपू और सीढ़ियों की बत्ती को एक-दूसरे की ख़राबियों
        से कभी नहीं रुकना चाहिए। क्या यह पहले से पक्का है? किस नियम
        से?
  \item पूरा आरेख बनाओ या बताओ: भंडार, मुख्य \omterm{ex:g3:first-electric-circuit:switch}{स्विच}, चार शाखाएँ, और
        चार शाखा-स्विच।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- रखवाले के सवाल।}
\begin{enumerate}[resume]
  \item रखवाले को चिंता है: ``चारों ज़िम्मेदारियाँ एक साथ पी रही हैं
        --- क्या हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} को अब भी भंडार का पूरा धक्का मिलता है?''
        समांतर के नियम से उत्तर दो।
  \item ``और विशाल लैंप में काम करते हुए क्या धारा ख़र्च हो जाती
        है?'' रखवाले की ग़लतफ़हमी दूर करो, और मदद लगे तो स्की-लिफ़्ट
        का सहारा लो।
  \item कोहरे वाली रात में सब कुछ एक साथ चलता है। समांतर की उदारता की
        क़ीमत क्या है, और लाइटहाउस में वह चुकाई कहाँ जाती है?
  \item सिर्फ़ पढ़ने वाली बत्ती जल रही है; मापने के शौक़ीन रखवाले
        क़तार का पीछा करते हैं: उसका पूरा चक्कर बताओ, और वे सारे
        पुरज़े गिनाओ जिनसे वह गुज़रती है तथा जिनसे बचकर निकल जाती है।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- ख़राबियों की रात।}
\begin{enumerate}[resume]
  \item आधी रात: सीढ़ियों का \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} फुँक जाता है। और क्या-क्या अँधेरा
        हो जाता है? कौन-सा नियम उत्तर देता है?
  \item एक बजे: एक भोंडी मरम्मत नंगे तार का एक टुकड़ा भंडार के दोनों
        सिरों को सीधे जोड़ता हुआ छोड़ जाती है, और वह भी हर \omterm{def:g5:series-parallel-circuits:parallel}{शाखा} से
        पहले। इस घटना का नाम बताओ, और समझाओ कि अब लाइटहाउस की हर
        ज़िम्मेदारी एक साथ क्यों नाकाम हो जाती है --- क़तार जाती
        कहाँ है?
  \item दो बजे, ख़राबी दूर: रखवाला चाहता है कि विशाल लैंप ऐसा सुरक्षित
        हो कि कोई शॉर्टकट उसे फिर कभी चुप न कर सके। एक दूसरा,
        स्वतंत्र बैटरी-भंडार उपलब्ध है। तारबंदी सुझाओ।
  \item भोर: रखवाले की डायरी लिखो --- तीन वाक्य: आज रात जहाज़ों को
        किस नियम ने बचाया, रात ने किस ग़लतफ़हमी का खंडन किया, और
        कौन-से अकेले तार ने सबसे बुरा घंटा दिया।
\end{enumerate}
\end{problem}
