\chapter{आयतन, द्रव्यमान, घनत्व}\label{ch:g7:volume-mass-density}

पिछले साल तुमने ग्रामों को घन सेंटीमीटरों से भाग देकर एक बेईमान सुनार
का पर्दाफ़ाश किया था। इस साल तुम्हारे गणित के पाठ्यक्रम ने तुम्हें एक
नई ताक़त थमाई है: वे अक्षर जो संख्याओं की जगह खड़े होते हैं। \omterm{def:g7:volume-mass-density:formula}{घनत्व}
उसे ख़र्च करने की सबसे बढ़िया जगह है --- एक छोटा-सा सूत्र, जो तीन
नुस्ख़े, एक जासूस का औज़ार-डिब्बा और तैरने का पूरा नियम पाँच संकेतों में
भर देता है।

\section{सूत्र}

\begin{definition}[घनत्व, सूत्र से]\label{def:g7:volume-mass-density:formula}
किसी पदार्थ का \emph{घनत्व}\index{घनत्व} $\rho$ (यूनानी अक्षर
\emph{रो}) उसका प्रति \omterm{def:g6:measurement-in-science:measuring}{मात्रक} \omterm{def:g6:mass-and-volume:volume}{आयतन} \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} है: $m$ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और $V$
\omterm{def:g6:mass-and-volume:volume}{आयतन} वाले नमूने के लिए,
\[
\rho = \frac{m}{V}.
\]
$m$ \omterm{def:g3:measuring-mass:units}{ग्राम} में और $V$ \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} में लो, तो $\rho$
\emph{\omterm{def:g3:measuring-mass:units}{ग्राम} प्रति \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर}} (\unit{g/cm^3}) में निकलता है ---
यानी पिछले साल वाला ``एक \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}'', अब वर्दी पहने
हुए। पानी का घनत्व: $\rho = \qty{1.0}{g/cm^3}$। और चूँकि एक \omterm{def:g6:mass-and-volume:volume}{लीटर}
$1000$ \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} होता है, वही संख्या \omterm{def:g3:measuring-mass:units}{किलोग्राम} प्रति \omterm{def:g6:mass-and-volume:volume}{लीटर} का काम भी
कर देती है: पानी \qty{1.0}{kg} प्रति \omterm{def:g6:mass-and-volume:volume}{लीटर}।
\end{definition}

\begin{example}[अक्षरों को पढ़ना]\label{ex:g7:volume-mass-density:reading}
सूत्र संक्षेप में लिखा गया एक वाक्य है। $\rho = m/V$ पढ़ा जाता है:
``\omterm{def:g7:volume-mass-density:formula}{घनत्व} निकालने के लिए नमूने के \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} को उसके \omterm{def:g6:mass-and-volume:volume}{आयतन} से भाग दो।''
अक्षर जगह रोकने वाले हैं --- \emph{इस} कंकड़ के लिए $m$ बन जाता है
\qty{75}{g} और $V$ बन जाता है \qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
भाग वही है जो पिछले साल था --- पर सूत्र आगे आने वाले हर कंकड़ के लिए
नुस्ख़ा याद रखता है।
\end{example}

\begin{proposition}[एक सूत्र में तीन नुस्ख़े]\label{prop:g7:volume-mass-density:recipes}
चूँकि \omterm{def:g7:volume-mass-density:formula}{घनत्व} \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और \omterm{def:g6:mass-and-volume:volume}{आयतन} को समानुपात में बाँध देता है, इसलिए एक
ही सूत्र तीन ज़रूरतें पूरी कर देता है:
\begin{enumerate}
  \item \emph{\omterm{def:g7:volume-mass-density:formula}{घनत्व} निकालना}: $\rho = m/V$ --- \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} को \omterm{def:g6:mass-and-volume:volume}{आयतन} से
        भाग दो;
  \item \emph{\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} निकालना}: $m = \rho \times V$ --- हर घन
        सेंटीमीटर $\rho$ \omterm{def:g3:measuring-mass:units}{ग्राम} लिए चलता है, और ऐसे $V$ घन हैं;
  \item \emph{\omterm{def:g6:mass-and-volume:volume}{आयतन} निकालना}: $V = m/\rho$ --- \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} को $\rho$
        \omterm{def:g3:measuring-mass:units}{ग्राम} वाले हिस्सों में बाँट दो; हिस्सों की संख्या ही घन
        सेंटीमीटरों की संख्या है।
\end{enumerate}
\end{proposition}

\begin{example}[नुस्ख़े काम पर]\label{ex:g7:volume-mass-density:atwork}
\emph{\omterm{def:g6:mass-and-volume:volume}{आयतन} से \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}}: लोहे ($\rho = 7.9$) के \qty{250}{cm^3}
का \omterm{def:g4:weight-and-mass:weight}{भार} कितना होगा? $m = 7.9 \times 250 = \qty{1975}{g}$ --- यानी एक
कॉफ़ी-मग जितनी जगह में लगभग दो \omterm{def:g3:measuring-mass:units}{किलोग्राम}।
\emph{\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} से \omterm{def:g6:mass-and-volume:volume}{आयतन}}: बलूत ($\rho = 0.6$) के \qty{540}{g}
कितनी जगह घेरेंगे? $V = 540 \div 0.6 = \qty{900}{cm^3}$।
\emph{समझदारी का साथी}: लोहे के उत्तर भारी-और-छोटे होने चाहिए, और
लकड़ी के हल्के-और-बड़े --- \omterm{def:g7:volume-mass-density:formula}{घनत्व} की तालिका परखने वाले क़दम की सबसे
अच्छी दोस्त है।
\end{example}

\section{घनत्व मापना}

\begin{method}[किसी द्रव का घनत्व]\label{met:g7:volume-mass-density:liquid}
\omterm{def:g3:solids-liquids-gases:liquid}{द्रव} अकेले \omterm{def:g3:levers-and-scales:scale}{तुला} के पलड़े पर नहीं बैठते --- इसलिए टेयर वाली तरकीब उधार
लो:
\begin{enumerate}
  \item ख़ाली मापक बेलन \omterm{def:g3:levers-and-scales:scale}{तुला} पर रखो और उसे शून्य पर टेयर कर दो;
  \item उसमें कोई सुविधाजनक \omterm{def:g6:mass-and-volume:volume}{आयतन} डालो --- मान लो रहस्यमय \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} का
        $V = \qty{80}{cm^3}$, और \omterm{met:g6:mass-and-volume:cylinder}{नवचंद्रक} ठीक से पढ़ो;
  \item अब \omterm{def:g3:levers-and-scales:scale}{तुला} अकेले \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} दिखाती है: मान लो
        $m = \qty{68}{g}$;
  \item भाग दो: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ --- पानी
        से हल्का: शायद कोई तेल।
\end{enumerate}
\end{method}

\begin{example}[किसी ठोस का घनत्व, शुरू से अंत तक]\label{ex:g7:volume-mass-density:solid}
एक पदक चाँदी ($\rho = 10.5$) का होने का दावा करता है। \omterm{def:g3:levers-and-scales:scale}{तुला}:
$m = \qty{84}{g}$। बेलन: पानी \qty{60.0}{cm^3} से \qty{68.0}{cm^3}
तक चढ़ता है, यानी $V = \qty{8.0}{cm^3}$। तब
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- दस्तख़त मेल खाता है:
हो सकता है वह सचमुच चाँदी हो। (चाँदी की परत चढ़ाई हुई कोई सस्ती धातु
यहीं पकड़ी जाती --- जब तक कि उसे बड़ी चालाकी से न चुना गया हो, जैसा
सप्ताहांत की समस्या दिखाएगी।)
\end{example}

\begin{remark}[सूत्रों में मात्रक]\label{rem:g7:volume-mass-density:units}
कोई सूत्र तभी ईमानदार है जब उसके \omterm{def:g6:measurement-in-science:measuring}{मात्रक} आपस में मेल खाएँ। $m$ \omterm{def:g3:measuring-mass:units}{ग्राम}
में और $V$ \unit{cm^3} में हो, तो $\rho$ \unit{g/cm^3} बोलता है;
\omterm{def:g3:measuring-mass:units}{किलोग्राम} और \omterm{def:g6:mass-and-volume:volume}{लीटर} खिलाओ तो वह \unit{kg/L} में उत्तर देता है --- और
ख़ुशी की बात यह कि किसी भी पदार्थ के लिए दोनों संख्याएँ एक ही होती हैं।
पर \omterm{def:g3:measuring-mass:units}{ग्राम} को \omterm{def:g6:mass-and-volume:volume}{लीटर} के साथ मिला दो, और सूत्र बिना शिकायत किए बकवास थमा
देता है। पेशेवरों का नियम: गणना से पहले मात्रकों की परेड कराओ; और गणना
के बाद उन्हें उत्तर में लिख दो।
\end{remark}

\section{घनत्व वाली सोच}

\begin{example}[मिश्रधातुएँ और बीच के मान]\label{ex:g7:volume-mass-density:alloys}
दो पदार्थ मिलाओ और मिश्रण का \omterm{def:g7:volume-mass-density:formula}{घनत्व} उनके दस्तख़तों के \emph{बीच} आ
जाता है --- और जिस घटक की मात्रा ज़्यादा हो, उसके ज़्यादा पास। काँसा
(ताँबा $9.0$, और थोड़ा-सा राँगा) लगभग $8.8$ के दस्तख़त करता है; मुकुट
की सोना-चाँदी मिलावट $10.5$ और $19.3$ के बीच दस्तख़त कर रही थी; और
समुद्र का पानी, यानी पानी में घुला नमक, चढ़कर लगभग $1.03$ पर आ जाता
है। बीच में पड़ना ख़ुद एक सुराग़ है: ``शुद्ध सोने'' की सिल्ली का $12$
वाला पाठ किसी साथी की मौजूदगी का इक़बाल है।
\end{example}

\begin{example}[तैरने की दुनिया घनत्व तय करता है]\label{ex:g7:volume-mass-density:floating}
तैरने का नियम, अब वर्दी में: कोई वस्तु किसी \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} में तब \omterm{def:g1:floating-and-sinking:words}{तैरती} है जब
$\rho_{\text{वस्तु}} < \rho_{\text{द्रव}}$ हो। बलूत ($0.6$) पानी
($1.0$) पर: तैरता है। बर्फ़ ($0.9$) तेल ($0.85$) में: \omterm{def:g1:floating-and-sinking:words}{डूबती} है ---
यहाँ अपनी अंतर्दृष्टि नहीं, तालिका देखो। कोई तैराक ($\approx 1.0$)
उन मशहूर बेहद खारी झीलों ($\approx 1.2$) में: कॉर्क की तरह तैरता है,
हाथ में अख़बार लिए। एक असमिका, और पूरा बंदरगाह।
\end{example}

\begin{remark}[भीड़ देखो, पुड़िया नहीं]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ \emph{पदार्थों} का वर्णन करता है; और जहाज़ तथा फूली हुई
जीवन-रक्षक जैकेट \emph{पुड़ियाँ} हैं --- यानी पदार्थ और उसमें क़ैद \omterm{def:g2:air-around-us:air}{हवा}
--- और तैरने के नियम का सामना पुड़िया का कुल $m/V$ करता है। सूत्र दोनों
सँभाल लेता है, बशर्ते तुम उसे सही $m$ और $V$ खिलाओ: पदार्थ के लिए
इस्पात का अपना \omterm{def:g7:volume-mass-density:formula}{घनत्व}, और पुड़िया के लिए पूरे ढाँचे का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बटा पूरे
ढाँचे का \omterm{def:g6:mass-and-volume:volume}{आयतन}। तैरने के ज़्यादातर ``विरोधाभास'' असल में इन्हीं दो
हिसाबों की गड़बड़ी हैं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:1}
\omterm{def:g7:volume-mass-density:formula}{घनत्व} का सूत्र लिखो, हर अक्षर का नाम बताओ, और वह \omterm{def:g6:measurement-in-science:measuring}{मात्रक} भी जो \omterm{def:g3:measuring-mass:units}{ग्राम}
तथा \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} के साथ आता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:2}
\omterm{def:g7:volume-mass-density:formula}{घनत्व} निकालो: $m = \qty{270}{g}$, $V = \qty{100}{cm^3}$।
क्षितिज-रेखा वाले आलेख की यह कौन-सी धातु है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:3}
सही नुस्ख़ा काम में लो: ताँबे ($\rho = 9.0$) के \qty{40}{cm^3} का
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान}; \ सोने ($\rho = 19.3$) के \qty{1930}{g} का \omterm{def:g6:mass-and-volume:volume}{आयतन}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:4}
\cref{met:g7:volume-mass-density:liquid} में बेलन को पहले टेयर क्यों
किया जाता है? फिर कौन-से दो पाठ सूत्र को खिलाए जाते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:5}
एक \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}: $V = \qty{50}{cm^3}$, $m = \qty{70}{g}$। उसका \omterm{def:g7:volume-mass-density:formula}{घनत्व}? क्या
बर्फ़ का टुकड़ा ($0.9$) उसमें तैरेगा या डूबेगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:6}
पानी का \omterm{def:g7:volume-mass-density:formula}{घनत्व} \unit{g/cm^3} में और \omterm{def:g3:measuring-mass:units}{किलोग्राम} प्रति \omterm{def:g6:mass-and-volume:volume}{लीटर} में एक ही
संख्या क्यों है? \cref{rem:g7:volume-mass-density:units} \omterm{def:g3:measuring-mass:units}{ग्राम} को
\omterm{def:g6:mass-and-volume:volume}{लीटर} के साथ मिलाने पर कौन-सी चेतावनी जोड़ती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:7}
``शुद्ध सोना'' लिखे एक कड़े का $m = \qty{58}{g}$ और
$V = \qty{5.0}{cm^3}$ है। उसका \omterm{def:g7:volume-mass-density:formula}{घनत्व} निकालो और तालिका को गवाह बनाकर
अपना फ़ैसला सुनाओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:8}
एक घन मीटर $100 \times 100 \times 100$ \omterm{def:g2:measuring-length:units}{सेंटीमीटर} वाला घन है। यह
कितने \unit{cm^3} हुए --- और एक घन \omterm{def:g2:measuring-length:units}{मीटर} पानी का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \omterm{def:g3:measuring-mass:units}{किलोग्राम}
में कितना? (उत्तर बता देता है कि मकान मालिक पानी वाले बिस्तरों से
क्यों घबराते हैं।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:9}
एक ``चाँदी'' की ट्रॉफ़ी: $m = \qty{420}{g}$, और पानी
\qty{200}{cm^3} से \qty{260}{cm^3} तक चढ़ता है। \omterm{def:g7:volume-mass-density:formula}{घनत्व}? वह तालिका के
किन दो पदार्थों के बीच पड़ता है --- और बीच में पड़ना क्या फुसफुसाता
है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:10}
ख़ाली बोतल का \omterm{def:g4:weight-and-mass:weight}{भार} \qty{380}{g} है; किसी रहस्यमय \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} से उसके
\qty{75}{cL} के निशान तक भरने पर उसका \omterm{def:g4:weight-and-mass:weight}{भार} \qty{1010}{g} हो जाता है।
\omterm{def:g3:solids-liquids-gases:liquid}{द्रव} का \omterm{def:g7:volume-mass-density:formula}{घनत्व} \unit{g/cm^3} में निकालो और उसकी पहचान सुझाओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:11}
एल्यूमिनियम के एक खोखले बोये का कुल \omterm{def:g6:mass-and-volume:volume}{आयतन} \qty{4000}{cm^3} और कुल
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \qty{1200}{g} है। \emph{पुड़िया} का \omterm{def:g7:volume-mass-density:formula}{घनत्व} निकालो और तैरने या
डूबने का अंदाज़ा लगाओ --- फिर
\cref{rem:g7:volume-mass-density:parcel} की मदद से बताओ कि
एल्यूमिनियम का अपना $2.7$ यहाँ ग़लत संख्या क्यों थी।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:volume-mass-density:12}
कोई ज़ंजीर शुद्ध ताँबे की है या नहीं, यह तय करने के लिए पूरी कार्यविधि
बनाओ: हर मापन गिनाओ, काम में लिया गया नुस्ख़ा बताओ, शुद्धता के लिए
अपेक्षित संख्या लिखो, और \emph{दो} ईमानदार वजहें भी बताओ जिनसे
तुम्हारा फ़ैसला फिर भी ग़लत हो सकता है (खोखली कड़ियाँ सोचो, और वे
चालाक मिश्रण भी जिनका \omterm{def:g7:volume-mass-density:formula}{घनत्व} ताँबे के पास ही आ बैठता है)। कौन-सी
अतिरिक्त जाँच मुक़दमे को कसेगी?
\end{exercise}

\section{समस्या: कबाड़ख़ाने का जासूस}

\begin{problem}[{सप्ताहांत समस्या --- कबाड़ख़ाने की धातु-जासूस के साथ
एक दोपहर; चार रहस्यमय ढेर, एक नक़ली सिल्ली; और घनत्व वाली जाँच की
सीमाएँ}]\label{pb:g7:volume-mass-density:1}
कबाड़ख़ाना धातु उस हिसाब से ख़रीदता है कि वह \emph{है} क्या, उस हिसाब से
नहीं कि दिखती कैसी है, और वहाँ की जासूस एक \omterm{def:g3:levers-and-scales:scale}{तुला}, एक बड़े अंशांकित बरतन
और उसी क्षितिज-रेखा वाली तालिका से काम करती है। तुम उसके नौसिखिए हो।
(तालिका का अंश: एल्यूमिनियम $2.7$; लोहा $7.9$; ताँबा $9.0$; सीसा
$11.3$; सोना $19.3$; टंगस्टन $19.3$।)

\medskip\noindent\textbf{भाग I --- चार ढेर।}
\begin{enumerate}
  \item ढेर क, एक सलेटी सिल्ली: $m = \qty{5400}{g}$, विस्थापन
        $V = \qty{2000}{cm^3}$। \omterm{def:g7:volume-mass-density:formula}{घनत्व} और पहचान?
  \item ढेर ख, तार का एक लच्छा: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$। \omterm{def:g7:volume-mass-density:formula}{घनत्व} और पहचान?
  \item ढेर ग, एक फीकी भारी पट्टी: $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$। \omterm{def:g7:volume-mass-density:formula}{घनत्व} और पहचान --- और ढेर ग को
        दस्ताने पहनकर तथा आदर से क्यों सँभालना चाहिए?
  \item ढेर घ, मिले-जुले फीके कबाड़ की एक बोरी: $m = \qty{8100}{g}$,
        $V = \qty{1500}{cm^3}$। दिखाओ कि ढेर घ का \omterm{def:g7:volume-mass-density:formula}{घनत्व} तालिका की दो
        धातुओं के बीच पड़ता है, और बताओ कि बोरी में सबसे संभव क्या
        है।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दाम और अंदाज़े।}
कबाड़ख़ाना दाम \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} से देता है, पर ढुलाई की योजना \omterm{def:g6:mass-and-volume:volume}{आयतन} से बनाता है।
\begin{enumerate}[resume]
  \item एक ख़रीदार ढेर क से \qty{540}{g} एल्यूमिनियम कटवाना चाहता
        है। यह सिल्ली का कितना \omterm{def:g6:mass-and-volume:volume}{आयतन} हुआ?
  \item ढेर ख के ताँबे के लच्छे को गलाकर \qty{25}{cm^3} वाले घन
        बनाने हैं। एक घन का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} कितना होगा, और लच्छे से कितने
        पूरे घन बनेंगे?
  \item एक पेटी ज़्यादा से ज़्यादा \qty{20}{kg} ढो सकती है। उसमें
        ताँबे के कितने घन वैध रूप से रखे जा सकते हैं?
  \item ट्रक की टंकी वाली जगह में लोहे के कबाड़ के \qty{3000}{cm^3}
        और समा सकते हैं। जासूस कितना अतिरिक्त \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, \omterm{def:g3:measuring-mass:units}{किलोग्राम}
        में, लाद सकती है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- नक़ली।}
एक विक्रेता चमकती ``सोने'' की सिल्ली लेकर आता है:
$m = \qty{3860}{g}$, विस्थापन $V = \qty{200}{cm^3}$।
\begin{enumerate}[resume]
  \item सिल्ली का \omterm{def:g7:volume-mass-density:formula}{घनत्व} निकालो। क्या वह सोने के दस्तख़त से मेल खाता
        है?
  \item जासूस बिना डिगे तालिका की आख़िरी पंक्ति देखती है और आह भरती
        है। कौन-सी सस्ती धातु ठीक सोने वाला \omterm{def:g7:volume-mass-density:formula}{घनत्व} पहने बैठी है ---
        और इससे \omterm{def:g7:volume-mass-density:formula}{घनत्व} वाली जाँच की सीमाओं के बारे में क्या पता चलता
        है?
  \item यहाँ कबाड़ख़ाने का कौन-सा उपकरण हारा: \omterm{def:g3:levers-and-scales:scale}{तुला}, बरतन, दोनों, या
        कोई नहीं? ठीक-ठीक बताओ कि \omterm{def:g7:volume-mass-density:formula}{घनत्व} वाली जाँच ने सिल्ली के बारे
        में ईमानदारी से क्या \emph{साबित} कर दिया।
  \item इस पाठ्यक्रम के पिछले वर्षों में से कोई और भौतिक जाँच सुझाओ
        जो सिल्ली को नुक़सान पहुँचाए बिना सोने को उसके हमशक्ल से अलग
        कर दे। (उनके \omterm{def:g4:changes-of-state:melting-point}{गलनांक} बहुत अलग हैं --- सोना लगभग
        \qty{1064}{\celsius} पर, और हमशक्ल यहाँ की हर भट्ठी से कहीं
        ऊपर --- पर कोई कबाड़ख़ाना शायद-ख़ज़ाने को गलाता नहीं: कोई
        नरम सबूत ढूँढ़ो, शायद लोहे के क्रोड पर \omterm{def:g1:magnets:magnet}{चुंबक} का फ़ैसला, या
        ठोंकी गई छड़ की खनक, और अपने चुनाव के पक्ष में ईमानदारी से
        तर्क दो।)
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- जासूस का हुनर।}
\begin{enumerate}[resume]
  \item जासूस का तीन पंक्तियों वाला घोषणापत्र लिखो: \omterm{def:g7:volume-mass-density:formula}{घनत्व} क्या साबित
        कर सकता है, क्या सिर्फ़ सुझा सकता है, और अकेला कभी क्या नहीं
        कर सकता।
\end{enumerate}
\end{problem}
