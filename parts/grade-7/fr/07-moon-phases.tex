\chapter{Les phases de la Lune}\label{ch:g7:moon-phases}

Il y a des années, tu tenais un journal de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} et tu regardais la
forme argentée passer du croissant à la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} et revenir, et
nous t'avions promis le «~pourquoi~» pour plus tard. Plus tard, c'est
maintenant. Tu possèdes tout ce dont l'explication a besoin~: une
\omterm{def:g2:moon-in-the-sky:moon}{Lune} qui parcourt son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite} en un mois, de la \omterm{def:g1:light-and-shadows:source}{lumière} solaire en
lignes droites, et une sphère qui est toujours --- toujours ---
éclairée à moitié.

\section{Le fait unique qui est dessous}

\begin{proposition}[Éclairée à moitié, toujours]\label{prop:g7:moon-phases:halflit}
Le Soleil éclaire une moitié de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} à chaque instant --- celle
qui lui fait face --- exactement comme il éclaire la moitié de la
Terre. La \omterm{def:g2:moon-in-the-sky:moon}{Lune} n'a pas de forme changeante, elle ne grandit ni ne
rétrécit~: c'est une boule de pierre qui porte un côté jour permanent
et un côté nuit permanent, la frontière glissant à mesure qu'elle
parcourt son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite}. Tout ce que nous appelons une «~\omterm{def:g7:moon-phases:phases}{phase}~» n'est
que la \emph{vue}~: la part du côté jour qui se trouve tournée vers
la Terre.
\end{proposition}

\begin{definition}[Phases]\label{def:g7:moon-phases:phases}
Les \emph{phases}\index{phase} de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} sont ses formes apparentes
au fil du mois~: la \emph{nouvelle lune} (côté jour entièrement
détourné --- nous ne voyons rien), le \emph{croissant}, le
\emph{premier quartier} (la moitié du disque éclairée), la
\emph{lune gibbeuse} (plus de la moitié), la \emph{\omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune}} (côté
jour entièrement tourné vers nous), puis de nouveau gibbeuse, le
\emph{dernier quartier} et le croissant, refermant le cycle en
environ $29.5$ jours. De la nouvelle lune à la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune}, la partie
éclairée grandit de nuit en nuit~: la \omterm{def:g2:moon-in-the-sky:moon}{Lune} est
\emph{croissante}\index{croissante}~; de la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} à la
nouvelle, elle diminue~: la \omterm{def:g2:moon-in-the-sky:moon}{Lune} est
\emph{décroissante}\index{décroissante}.
\end{definition}

\begin{method}[Sois la Terre, tiens la Lune]\label{met:g7:moon-phases:walk}
L'explication, jouée en une minute~:
\begin{enumerate}
  \item dans une pièce sombre, une unique lampe nue est le Soleil~;
        toi, tu es la Terre~; une balle blanche au bout de ton bras
        tendu est la \omterm{def:g2:moon-in-the-sky:moon}{Lune}~;
  \item fais face à la lampe, la balle vers elle~: la moitié
        éclairée de la balle est détournée de toi --- tu vois son
        côté sombre~: c'est la nouvelle lune~;
  \item tourne lentement sur place (la balle décrit son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite}
        autour de toi)~: un croissant éclairé apparaît au bord de la
        balle --- croissant de lune \omterm{def:g7:moon-phases:phases}{croissante}~; après un quart de
        tour, la moitié de la balle apparaît éclairée~: c'est le
        premier quartier~;
  \item continue de tourner~: gibbeuse --- puis, la lampe dans le
        dos, la balle entièrement éclairée devant toi~: c'est la
        \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune}~; achève le tour par la gibbeuse \omterm{def:g7:moon-phases:phases}{décroissante},
        le quartier et le croissant, jusqu'à la nouvelle lune.
\end{enumerate}
Un tour sur toi-même rejoue un mois de ciel.
\end{method}

\section{L'idée fausse, mise à la retraite}

\begin{proposition}[Les phases ne sont pas l'ombre de la
Terre]\label{prop:g7:moon-phases:notshadow}
La réponse erronée la plus répandue de toute l'astronomie~: «~les
\omterm{def:g7:moon-phases:phases}{phases} sont l'\omterm{def:g1:light-and-shadows:shadow}{ombre} de la Terre sur la \omterm{def:g2:moon-in-the-sky:moon}{Lune}~». La géométrie la réfute
de trois façons. L'\omterm{def:g1:light-and-shadows:shadow}{ombre} de la Terre pointe à l'\emph{opposé} du
Soleil --- elle ne peut toucher qu'une \omterm{def:g2:moon-in-the-sky:moon}{Lune} située à l'opposé du
Soleil, ce qui n'arrive qu'à la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} (et encore, seulement
lors de rares éclipses). Une \omterm{def:g2:moon-in-the-sky:moon}{Lune} en croissant se tient
\emph{du côté du Soleil}, très loin de notre \omterm{def:g1:light-and-shadows:shadow}{ombre}. Et le bord de
notre \omterm{def:g1:light-and-shadows:shadow}{ombre} sur la \omterm{def:g2:moon-in-the-sky:moon}{Lune} est toujours un arc de grand cercle --- il ne
pourrait jamais découper la frontière à double courbure de la forme
gibbeuse. Les \omterm{def:g7:moon-phases:phases}{phases} n'ont besoin d'aucune \omterm{def:g1:light-and-shadows:shadow}{ombre}~: une boule éclairée
à moitié, vue sous différents angles, fait tout le travail.
\end{proposition}

\begin{example}[Éclipse et phase]\label{ex:g7:moon-phases:eclipsevs}
Garde les deux assombrissements de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} dans des tiroirs séparés.
\emph{\omterm{def:g7:moon-phases:phases}{Phase}}~: le lent cycle mensuel, le côté nuit propre à la \omterm{def:g2:moon-in-the-sky:moon}{Lune}
qui se tourne vers nous --- aucune \omterm{def:g1:light-and-shadows:shadow}{ombre} de la nôtre n'y est mêlée.
\emph{\omterm{def:g7:shadows-eclipses:lunar}{Éclipse de Lune}}~: une heure rare où la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} traverse
notre \omterm{prop:g7:shadows-eclipses:umbra}{zone d'ombre} --- l'\omterm{def:g1:light-and-shadows:shadow}{ombre} de la Terre tombant véritablement sur
elle, cuivrée et rapide. Le rythme du journal est une vue~; l'éclipse
est une visite.
\end{example}

\section{Lire le ciel comme une horloge}

\begin{example}[Où les phases tiennent leurs
horaires]\label{ex:g7:moon-phases:hours}
Le schéma attache chaque \omterm{def:g7:moon-phases:phases}{phase} à une position --- et chaque position
à un horaire. La \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} se tient à l'opposé du Soleil~: elle
doit donc se lever au \omterm{def:g1:day-and-night:daynight}{coucher du Soleil}, régner toute la nuit, et se
coucher au \omterm{def:g1:day-and-night:daynight}{lever du Soleil}. La nouvelle lune voyage \emph{avec} le
Soleil~: présente tout le jour, perdue dans l'éblouissement --- et
voilà pourquoi personne ne «~voit~» une nouvelle lune. Le premier
quartier suit le Soleil d'un quart de tour~: il se lève vers midi et
se couche vers minuit --- c'est la \omterm{def:g2:moon-in-the-sky:moon}{Lune} d'après le dîner. La surprise
de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} de plein jour de ton enfance est désormais un horaire~:
toute \omterm{def:g2:moon-in-the-sky:moon}{Lune} qui n'est ni nouvelle ni pleine partage une partie de son
service avec le Soleil.
\end{example}

\begin{example}[De quel côté pointe le croissant~?]\label{ex:g7:moon-phases:points}
Le côté éclairé de toute \omterm{def:g7:moon-phases:phases}{phase} fait face au Soleil --- le croissant
est un arc pointé vers son archer. Croissant du soir à l'ouest~: les
cornes tournées vers le haut, à l'opposé de la lueur du coucher
au-dessous de l'horizon. Le ciel ne le dessine jamais de travers~;
les peintres, souvent. Un croissant dont les cornes pointent vers le
Soleil est l'erreur préférée des chasseurs d'erreurs de
l'astronomie.
\end{example}

\begin{remark}[Le visage qui ne se détourne jamais]\label{rem:g7:moon-phases:sameface}
Encore une observation du journal, enfin honorée~: les taches sombres
de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} forment le même visage chaque nuit, \omterm{def:g7:moon-phases:phases}{phase} après \omterm{def:g7:moon-phases:phases}{phase},
année après année. La \omterm{def:g2:moon-in-the-sky:moon}{Lune} tourne exactement une fois sur son \omterm{def:g6:sun-earth-moon:axisorbit}{axe} par
\omterm{def:g6:sun-earth-moon:axisorbit}{orbite} --- la même moitié fait donc face à la Terre pour toujours.
Personne n'a vu la face cachée avant qu'une sonde spatiale n'en fasse
le tour avec un appareil photo, du vivant de tes grands-parents.
Pourquoi la rotation et la révolution s'accordent si parfaitement est
une profonde histoire de marées, gardée pour les années de licence de
ce cours.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g7:moon-phases:4}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:moon-phases:9}

\begin{exercise}[$\star\star$]\label{exo:g7:moon-phases:10}

\begin{exercise}[$\star\star$]\label{exo:g7:moon-phases:11}

\begin{exercise}[$\star\star\star$]\label{exo:g7:moon-phases:12}

\end{exercise}

\section{Problème~: un mois avec la Lune}

\begin{problem}[{Problème du week-end --- le mois portes ouvertes de
l'observatoire~; trente nuits de Lune, un tableau mural~; le quiz du
médiateur}]\label{pb:g7:moon-phases:1}
L'observatoire de la ville organise un «~mois avec la \omterm{def:g2:moon-in-the-sky:moon}{Lune}~» ouvert
au public. Comme jeune médiateur, tu tiens le tableau mural et tu
réponds aux visiteurs.

\medskip\noindent\textbf{Partie I --- monter le tableau.}
\begin{enumerate}
  \item La nuit 0 est celle de la nouvelle lune. Remplis les
        étapes du tableau~: vers quelles nuits tombent le premier
        quartier, la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} et le dernier quartier~?
  \item À côté de chaque étape, le tableau réclame les \omterm{ex:g2:measuring-time:clock}{heures} de
        lever. Associe~: se lève au \omterm{def:g1:day-and-night:daynight}{coucher du Soleil}~; \ se lève
        vers midi~; \ se lève vers minuit~; \ se lève et se couche
        avec le Soleil.
  \item Une visiteuse demande pourquoi le tableau montre le
        croissant bas dans le ciel du soir à l'\emph{ouest}, alors
        que la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} s'élève de l'\emph{est} à la tombée du
        jour. Explique avec les positions orbitales.
  \item Un autre visiteur jure avoir vu la \omterm{def:g2:moon-in-the-sky:moon}{Lune} à dix \omterm{ex:g2:measuring-time:clock}{heures} du
        matin. Quelles \omterm{def:g7:moon-phases:phases}{phases} du tableau rendent possible une \omterm{def:g2:moon-in-the-sky:moon}{Lune}
        matinale~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- les démonstrations du
médiateur.}
\begin{enumerate}[resume]
  \item Tu joues la promenade de la lampe et de la balle devant une
        classe. Un élève t'immobilise en plein tour~: la balle
        apparaît éclairée aux trois quarts. Nomme la \omterm{def:g7:moon-phases:phases}{phase}, et
        situe la lampe, la Terre (toi) et la balle pour le groupe.
  \item «~Alors c'est bien l'\omterm{def:g1:light-and-shadows:shadow}{ombre} de la Terre qui fait les
        \omterm{def:g7:moon-phases:phases}{phases}~?~» demande l'élève en montrant le côté sombre de ta
        balle. Donne la réfutation que contient la démonstration
        elle-même~: le côté nuit de qui voient-ils, et où se
        trouverait réellement l'\omterm{def:g1:light-and-shadows:shadow}{ombre} de la Terre~?
  \item Le groupe demande pourquoi on voit toujours «~le lapin~»
        (ou «~le visage~») dans la \omterm{def:g2:moon-in-the-sky:moon}{Lune}, jamais son dos. Réponds
        avec le fait du même visage.
  \item Un parent sceptique~: «~Si la \omterm{def:g2:moon-in-the-sky:moon}{Lune} tourne sur elle-même,
        comme vous le dites, ne devrait-on pas la voir tourner~?~»
        Résous la contradiction apparente~: comment la \omterm{def:g2:moon-in-the-sky:moon}{Lune} peut-elle
        tourner exactement une fois par \omterm{def:g6:sun-earth-moon:axisorbit}{orbite} et, par là même,
        nous \emph{cacher} sa rotation~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- la soirée quiz.}
\begin{enumerate}[resume]
  \item Carte A~: «~La \omterm{def:g2:moon-in-the-sky:moon}{Lune} de ce soir s'est levée à minuit et
        montre sa moitié \emph{gauche} éclairée (vue en allant vers
        le matin). \omterm{def:g7:moon-phases:phases}{Croissante} ou \omterm{def:g7:moon-phases:phases}{décroissante} --- et quel
        quartier~?~»
  \item Carte B~: «~L'affiche d'un artiste montre une \omterm{def:g2:moon-in-the-sky:moon}{Lune} en
        croissant à minuit, haute au-dessus de nos têtes, les
        cornes pointées droit vers les couleurs du couchant à
        l'horizon.~» Énumère toutes les erreurs astronomiques de
        l'affiche.
  \item Carte C~: «~Pendant la \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} de ce soir, des
        visiteurs pourraient-ils assister quelque part sur Terre à
        une \omterm{def:g7:shadows-eclipses:solar}{éclipse de Soleil}~?~» Tranche avec la géométrie des
        \omterm{def:g7:moon-phases:phases}{phases}.
  \item Discours de clôture~: en trois phrases, donne le fait
        unique qui sous-tend tout le mois, l'idée fausse que le
        mois a enterrée, et l'astuce d'horaire que les visiteurs
        doivent emporter chez eux (ce que toute \omterm{def:g7:moon-phases:phases}{phase} dit de son
        heure de lever).
\end{enumerate}
\end{problem}
