\chapter{États de la matière et changements d'état}\label{ch:g7:states-of-matter}

Six années de \omterm{def:g3:solids-liquids-gases:solid}{solides} qui gardent leur forme, de \omterm{def:g3:solids-liquids-gases:liquid}{liquides} dont la
surface reste horizontale, de \omterm{def:g3:solids-liquids-gases:gas}{gaz} qui remplissent chaque recoin ---
et pas une seule fois nous n'avons demandé \emph{pourquoi}. Cette
année, la physique soulève le couvercle. Sous les trois états se
cache l'une des plus grandes idées que les humains aient jamais
eues~: tout, absolument tout, est fait de particules trop petites
pour être vues. Accorde cette seule idée, et les règles que tu
collectionnes depuis l'enfance s'expliquent d'un seul coup.

\section{La grande idée}

\begin{definition}[Molécules]\label{def:g7:states-of-matter:molecule}
Toute la matière est faite de \emph{molécules}\index{molécule}~: des
particules d'une petitesse inimaginable, bien au-dessous de ce que
peuvent voir un œil ou un microscope d'école. Une seule goutte d'eau
contient plus de molécules qu'il n'y a de gouttes d'eau dans toutes
les mers. Chaque substance pure est bâtie avec son propre type de
molécule --- l'eau avec des molécules d'eau, le fer avec les
particules qui lui sont propres --- et les molécules d'une substance
sont toutes identiques.
\end{definition}

\begin{remark}[Croire sans voir]\label{rem:g7:states-of-matter:believing}
Personne n'a jamais observé une seule \omterm{def:g7:states-of-matter:molecule}{molécule} à l'œil nu --- alors
pourquoi y croire~? Parce que l'idée \emph{mérite} la croyance~: elle
prédit correctement tout ce que contient ce chapitre, et des
montagnes de choses en plus. C'est ainsi que la physique accepte un
invité invisible~: non par foi, mais par utilité acharnée. (Des
\omterm{prop:g5:mirrors-reflection:image}{images} directes, obtenues par de merveilleux instruments modernes,
sont enfin arrivées il y a un siècle --- longtemps après que les
physiciens étaient déjà convaincus.)
\end{remark}

\begin{proposition}[Le modèle particulaire des trois
états]\label{prop:g7:states-of-matter:model}
Les trois états sont trois arrangements des mêmes \omterm{def:g7:states-of-matter:molecule}{molécules}~:
\begin{enumerate}
  \item \emph{\omterm{def:g3:solids-liquids-gases:solid}{solide}}~: des \omterm{def:g7:states-of-matter:molecule}{molécules} serrées les unes contre les
        autres en un empilement ordonné, chacune tremblant sur
        place mais gardant sa position --- d'où la forme fixe~;
  \item \emph{\omterm{def:g3:solids-liquids-gases:liquid}{liquide}}~: des \omterm{def:g7:states-of-matter:molecule}{molécules} qui se touchent encore, mais
        en désordre et libres de glisser les unes contre les
        autres --- d'où l'écoulement et la surface horizontale~;
  \item \emph{\omterm{def:g3:solids-liquids-gases:gas}{gaz}}~: des \omterm{def:g7:states-of-matter:molecule}{molécules} très éloignées les unes des
        autres, volant vite dans toutes les directions, se heurtant
        et rebondissant --- d'où le remplissage de tout recoin
        offert.
\end{enumerate}
\end{proposition}

\begin{example}[Six ans de règles, une seule
explication]\label{ex:g7:states-of-matter:rules}
Passe les vieilles règles dans le modèle. Les \omterm{def:g3:solids-liquids-gases:solid}{solides} gardent leur
forme~: chaque \omterm{def:g7:states-of-matter:molecule}{molécule} tient son poste. Les \omterm{def:g3:solids-liquids-gases:liquid}{liquides} prennent la
forme du récipient et restent horizontaux~: la foule glissante
s'affaisse jusqu'à ce qu'aucune \omterm{def:g7:states-of-matter:molecule}{molécule} ne puisse glisser plus bas.
Les \omterm{def:g3:solids-liquids-gases:gas}{gaz} remplissent toute la place offerte~: rien ne retient les
voltigeuses. Le parfum qui traverse la pièce~: des \omterm{def:g7:states-of-matter:molecule}{molécules} de \omterm{def:g3:solids-liquids-gases:gas}{gaz}
qui volent, se heurtent, se répandent. Le sable qui se verse comme un
\omterm{def:g3:solids-liquids-gases:liquid}{liquide}~: ce ne sont pas des \omterm{def:g7:states-of-matter:molecule}{molécules} qui glissent, mais des grains
--- chaque grain étant lui-même un minuscule \omterm{def:g3:solids-liquids-gases:solid}{solide} ordonné de
milliers de milliards de \omterm{def:g7:states-of-matter:molecule}{molécules}.
\end{example}

\section{Le test de l'espace vide}

\begin{method}[Deux seringues]\label{met:g7:states-of-matter:syringe}
Le modèle affirme que les \omterm{def:g7:states-of-matter:molecule}{molécules} d'un \omterm{def:g3:solids-liquids-gases:gas}{gaz} volent loin les unes des
autres et que celles d'un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} se touchent. Alors un \omterm{def:g3:solids-liquids-gases:gas}{gaz} devrait
se laisser comprimer, et un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} non. Mettons cela à l'épreuve~:
\begin{enumerate}
  \item remplir d'\omterm{def:g2:air-around-us:air}{air} une seringue en plastique, boucher l'embout du
        bout du doigt, et pousser le piston~: il cède --- on peut
        réduire de moitié la place de l'\omterm{def:g2:air-around-us:air}{air}, en sentant une
        résistance élastique~;
  \item remplir la même seringue d'eau, boucher, et pousser~: le
        piston tient comme un mur --- l'eau refuse de rétrécir
        d'un cheveu~;
  \item relâcher chaque piston~: l'\omterm{def:g2:air-around-us:air}{air} revient d'un bond~; l'eau
        n'a jamais bougé.
\end{enumerate}
\end{method}

\begin{proposition}[Les gaz se compriment~; les liquides et les
solides, non]\label{prop:g7:states-of-matter:squeeze}
Un \omterm{def:g3:solids-liquids-gases:gas}{gaz} peut être \emph{comprimé}\index{compressible} --- forcé dans
moins de place --- parce qu'un \omterm{def:g3:solids-liquids-gases:gas}{gaz} est surtout de l'espace vide entre
des \omterm{def:g7:states-of-matter:molecule}{molécules} qui volent. Les \omterm{def:g3:solids-liquids-gases:liquid}{liquides} et les \omterm{def:g3:solids-liquids-gases:solid}{solides} sont
pratiquement \emph{incompressibles}\index{incompressible}~: leurs
\omterm{def:g7:states-of-matter:molecule}{molécules} se touchent déjà, et une matière dont les \omterm{def:g7:states-of-matter:molecule}{molécules} se
touchent ne peut pas être tassée davantage par la moindre poussée
qu'une main --- ou même une presse hydraulique --- puisse fournir.
\end{proposition}

\begin{example}[L'air comprimé au travail]\label{ex:g7:states-of-matter:squeezed}
La pompe à vélo entasse des pièces entières d'\omterm{def:g2:air-around-us:air}{air} dans le pneu ---
ce qui n'est possible que parce que l'\omterm{def:g2:air-around-us:air}{air} est surtout du vide. La
foule comprimée, dont les \omterm{def:g7:states-of-matter:molecule}{molécules} martèlent les parois du pneu plus
vite et plus dru, soutient le vélo~: la poussée d'un \omterm{def:g3:solids-liquids-gases:gas}{gaz} \emph{est}
sa grêle moléculaire. Les bouteilles des plongeurs tassent un
après-midi de respiration dans une bouteille d'acier~; le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} de
frein, incompressible, transmet quant à lui la poussée de ton pied
jusqu'aux roues sans rien perdre --- chaque état engagé pour son
propre talent.
\end{example}

\section{Les changements d'état, relus}

\begin{example}[Les portes, molécule par
molécule]\label{ex:g7:states-of-matter:doors}
Chauffe un \omterm{def:g3:solids-liquids-gases:solid}{solide} et ses \omterm{def:g7:states-of-matter:molecule}{molécules} tremblent plus fort~; à la
\omterm{def:g4:changes-of-state:melting-point}{température de fusion}, ce tremblement fait éclater l'empilement
ordonné --- la foule se met à glisser~: c'est la \omterm{def:g6:water-states:changes}{fusion}. Chauffe
encore le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} et les \omterm{def:g7:states-of-matter:molecule}{molécules} les plus rapides s'arrachent
entièrement à l'emprise de la foule et s'envolent~: c'est
l'évaporation --- et, à la \omterm{def:g4:changes-of-state:boiling-point}{température d'ébullition}, l'évasion
devient une ruée depuis l'intérieur même du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}. Le
refroidissement passe le film à l'envers~: des voltigeuses attrapées
(\omterm{def:g6:water-states:changes}{condensation}), des glissantes qui se rangent en rangs
(\omterm{def:g6:water-states:changes}{solidification}). Les six portes de l'an dernier ne sont qu'une seule
histoire~: l'emprise moléculaire contre le tremblement moléculaire.
\end{example}

\begin{example}[De vieilles lois, gratuitement]\label{ex:g7:states-of-matter:free}
Deux lois durement acquises viennent maintenant gratuitement. \emph{La
\omterm{def:g3:measuring-mass:mass}{masse} survit aux changements d'état}~: la \omterm{def:g6:water-states:changes}{fusion} réarrange les
\omterm{def:g7:states-of-matter:molecule}{molécules} mais n'en crée ni n'en détruit une seule --- mêmes
particules, même \omterm{def:g3:measuring-mass:mass}{masse}. \emph{Les paliers de \omterm{def:g3:temperature-thermometers:temperature}{température}}~: à la
\omterm{def:g4:changes-of-state:melting-point}{température de fusion}, la \omterm{def:g5:heat-and-insulation:heat}{chaleur} qui arrive est dépensée à briser
l'emprise de l'empilement, \omterm{def:g7:states-of-matter:molecule}{molécule} après \omterm{def:g7:states-of-matter:molecule}{molécule}, et non à
accélérer le tremblement --- le \omterm{def:g3:temperature-thermometers:temperature}{thermomètre} attend que le dernier
rang soit rompu. Ce que la mesure a découvert, le modèle l'explique.
\end{example}

\begin{remark}[L'étrange gonflement de l'eau]\label{rem:g7:states-of-matter:iceswell}
Même l'étrange gonflement de l'eau qui \omterm{def:g2:ice-water-steam:meltfreeze}{gèle} rejoint le tableau~: il
se trouve que les \omterm{def:g7:states-of-matter:molecule}{molécules} d'eau se verrouillent en un empilement
inhabituellement \emph{aéré} --- un réseau ouvert, avec plus d'espace
vide que n'en avait la foule glissante. Empilement plus aéré, plus de
place, \omterm{def:g6:density-floating:density}{masse volumique} plus faible~: la glace \omterm{def:g1:floating-and-sinking:words}{flotte} sur son propre
\omterm{def:g3:solids-liquids-gases:liquid}{liquide}. La plupart des substances s'empilent plus serré qu'elles ne
glissent~; l'architecture de l'eau est la magnifique exception.
\end{remark}

\section{La chaleur, vue de l'intérieur}

\begin{proposition}[Température et agitation
moléculaire]\label{prop:g7:states-of-matter:temperature}
La \omterm{def:g3:temperature-thermometers:temperature}{température} mesure la vivacité de l'agitation moléculaire~: plus
une substance est chaude, plus ses \omterm{def:g7:states-of-matter:molecule}{molécules} tremblent, glissent ou
volent vite. La \omterm{def:g5:heat-and-insulation:heat}{chaleur} qui \omterm{def:g1:floating-and-sinking:words}{coule} du chaud vers le froid, c'est la
foule vive qui se calme en secouant la foule engourdie --- choc après
choc --- jusqu'à ce que les deux partagent une même allure~: une
seule \omterm{def:g3:temperature-thermometers:temperature}{température}. La vieille loi de la \omterm{def:g5:heat-and-insulation:heat}{chaleur} qui descend la pente
est du ping-pong moléculaire.
\end{proposition}

\begin{example}[Le froid n'est pas une matière]\label{ex:g7:states-of-matter:cold}
Le modèle met définitivement à la retraite une vieille tentation~:
rien qui s'appellerait «~froid~» ne \omterm{def:g1:floating-and-sinking:words}{coule} jamais. Il n'y a qu'une
agitation moléculaire plus lente ou plus rapide~; la vive prête sa
vivacité, jamais l'inverse. Au plus froid imaginable --- des
\omterm{def:g7:states-of-matter:molecule}{molécules} aussi immobiles que la nature le permet --- se trouve un
véritable plancher de \omterm{def:g3:temperature-thermometers:temperature}{température}, bien au-dessous de tout
congélateur~; son nombre étrange et célèbre appartient au \omterm{def:g6:mass-and-volume:volume}{volume} du
lycée.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:1}
Énonce l'\omterm{prop:g5:mirrors-reflection:image}{image} que le modèle particulaire donne de chacun des trois
états, en une ligne chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:2}
Pourquoi un \omterm{def:g3:solids-liquids-gases:solid}{solide} garde-t-il sa forme alors qu'un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} prend
celle de son récipient~? Réponds avec des \omterm{def:g7:states-of-matter:molecule}{molécules}, pas avec des
règles.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:3}
Dans le test des seringues, pourquoi l'\omterm{def:g2:air-around-us:air}{air} cède-t-il et revient-il
d'un bond, alors que l'eau tient comme un mur~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:4}
Une pompe à vélo entasse plusieurs pièces d'\omterm{def:g2:air-around-us:air}{air} dans un seul pneu.
Que cela prouve-t-il sur l'espacement des \omterm{def:g7:states-of-matter:molecule}{molécules} d'un \omterm{def:g3:solids-liquids-gases:gas}{gaz}~? Une
pompe pourrait-elle en faire autant avec de l'eau~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:5}
Raconte l'histoire de la \omterm{def:g6:water-states:changes}{fusion} en langage moléculaire~: que fait la
\omterm{def:g5:heat-and-insulation:heat}{chaleur}, et qu'est-ce qui se brise à la \omterm{def:g4:changes-of-state:melting-point}{température de fusion}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:6}
Pourquoi la \omterm{def:g3:measuring-mass:mass}{masse} survit-elle à tout changement d'état, d'après le
modèle~? Que faudrait-il qu'il arrive aux \omterm{def:g7:states-of-matter:molecule}{molécules} pour que de la
\omterm{def:g3:measuring-mass:mass}{masse} soit perdue~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:7}
Que mesure la \omterm{def:g3:temperature-thermometers:temperature}{température}, dans le modèle~? Décris l'écoulement de la
\omterm{def:g5:heat-and-insulation:heat}{chaleur} vers le bas de la pente comme du ping-pong moléculaire.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:states-of-matter:8}
Un parfum ouvert à un bout d'une pièce immobile est senti à l'autre
bout une minute plus tard. Explique le voyage --- et pourquoi un
courant d'\omterm{def:g2:air-around-us:air}{air} l'accélère.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:states-of-matter:9}
L'évaporation refroidit ce qu'elle laisse derrière elle~: la sueur
rafraîchit la peau, et un doigt mouillé levé en l'\omterm{def:g2:air-around-us:air}{air} sent le côté
froid du \omterm{def:g2:air-around-us:wind}{vent}. Explique avec l'évasion des \omterm{def:g7:states-of-matter:molecule}{molécules} les \emph{plus
rapides} --- quelle sorte de foule reste en arrière~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:states-of-matter:10}
Un ballon fermé laissé au soleil d'été gonfle~; au réfrigérateur, il
s'affaisse. Le nombre de \omterm{def:g7:states-of-matter:molecule}{molécules} à l'intérieur n'a jamais changé.
Explique les deux transformations avec la \omterm{def:g5:speed-distance-time:speed}{vitesse} moléculaire et la
grêle sur les parois.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:states-of-matter:11}
Pourquoi la glace flotte-t-elle, en langage moléculaire~? Qu'y a-t-il
d'inhabituel dans l'empilement que bâtissent les \omterm{def:g7:states-of-matter:molecule}{molécules} d'eau, et
que feraient les étangs en hiver si l'eau s'empilait comme les
substances ordinaires~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:states-of-matter:12}
Une goutte d'encre déposée doucement au fond d'un verre d'eau
\emph{parfaitement immobile} se répand lentement, en quelques \omterm{ex:g2:measuring-time:clock}{heures},
jusqu'à ce que toute l'eau soit d'un bleu léger --- sans agitation,
sans courant. Explique ce que cette diffusion patiente révèle sur les
\omterm{def:g7:states-of-matter:molecule}{molécules} d'un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} --- celles de l'eau comme celles de
l'encre --- et pourquoi la même goutte se répand plus vite dans une
eau chaude. (Cette humble observation compte parmi les grandes
premières preuves de toute l'idée moléculaire.)
\end{exercise}

\section{Problème~: le théâtre moléculaire}

\begin{problem}[{Problème du week-end --- la classe met en scène le
modèle particulaire dans le gymnase~; des élèves en guise de
molécules, les états en guise de chorégraphie~; les questions des
critiques}]\label{pb:g7:states-of-matter:1}
Le cours de physique met en scène la matière elle-même~: trente
élèves jouent les \omterm{def:g7:states-of-matter:molecule}{molécules} d'une substance, et le gymnase est le
récipient.

\medskip\noindent\textbf{Partie I --- trois tableaux.}
\begin{enumerate}
  \item Premier tableau, le \omterm{def:g3:solids-liquids-gases:solid}{solide}~: comment les trente
        doivent-ils se tenir, et quel petit mouvement chacun
        doit-il continuer de faire~? Que n'ont-ils pas le droit de
        faire~?
  \item Deuxième tableau, le \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~: qu'est-ce qui change dans
        l'arrangement, et quelle liberté nouvelle apparaît~? Que
        doivent-ils continuer de faire, et que le tableau du \omterm{def:g3:solids-liquids-gases:gas}{gaz}
        abolira~?
  \item Troisième tableau, le \omterm{def:g3:solids-liquids-gases:gas}{gaz}~: décris la chorégraphie --- et
        dis pourquoi ce tableau a besoin du gymnase \emph{tout
        entier} alors que les deux premiers se blottissent dans un
        coin.
  \item Le public demande~: «~Sont-ce les mêmes trente élèves dans
        les trois tableaux~?~» Quelle loi de l'an dernier la
        réponse met-elle en scène~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- les critiques éprouvent la
pièce.}
\begin{enumerate}[resume]
  \item Un critique pousse doucement la foule du tableau du \omterm{def:g3:solids-liquids-gases:gas}{gaz} et
        la refoule dans la moitié du gymnase. Le tableau le
        permet-il~? Et la même poussée sur le groupe du tableau du
        \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~? Nomme la proposition mise en scène.
  \item Pour mettre en scène la \emph{\omterm{def:g6:water-states:changes}{fusion}}, le metteur en scène
        monte un bouton de \omterm{def:g5:heat-and-insulation:heat}{chaleur} imaginaire. Que les élèves du
        \omterm{def:g3:solids-liquids-gases:solid}{solide} doivent-ils faire de plus en plus, et à quel moment
        le tableau devient-il légitimement celui du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~?
  \item La pièce doit montrer le palier de \omterm{def:g6:water-states:changes}{fusion}~: pendant que le
        changement se produit, à quoi l'\omterm{def:g5:everyday-energy:energy}{énergie} du bouton de
        \omterm{def:g5:heat-and-insulation:heat}{chaleur} est-elle dépensée --- et que doit rapporter le
        «~\omterm{def:g3:temperature-thermometers:temperature}{critique-thermomètre}~» sur la vivacité moyenne des
        élèves pendant ce temps~?
  \item Mets en scène l'évaporation à partir du tableau du
        \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~: quels élèves quittent le groupe --- n'importe
        lesquels au hasard, ou d'un genre particulier~? Qu'advient-il
        de la vivacité moyenne du groupe à mesure qu'ils partent, et
        quelle fraîcheur de tous les jours cela explique-t-il~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- effets spéciaux.}
\begin{enumerate}[resume]
  \item L'effet ballon~: le tableau du \omterm{def:g3:solids-liquids-gases:gas}{gaz} est clôturé par des
        élèves tenant un cercle de corde (la peau du ballon). Que
        font les acteurs moléculaires à la corde, et qu'arrive-t-il
        au cercle quand les acteurs vont plus vite~?
  \item L'étrange congélation de l'eau doit être mise en scène.
        Quelle consigne inhabituelle sur l'\emph{espacement} le
        metteur en scène donne-t-il à l'empilement du tableau du
        \omterm{def:g3:solids-liquids-gases:solid}{solide}, et quel fait de flottaison cela explique-t-il~?
  \item Le final de la goutte d'encre~: un élève en chemise bleue
        commence dans un coin du tableau du \emph{\omterm{def:g3:solids-liquids-gases:liquid}{liquide}}. Sans
        aucune poussée du metteur en scène, comment la chemise
        bleue finit-elle loin de son coin --- et pourquoi plus vite
        si le bouton de \omterm{def:g5:heat-and-insulation:heat}{chaleur} est monté~?
  \item Rédige la note du programme~: en trois phrases, dis quelle
        idée unique toute la pièce met en scène, nomme deux
        vieilles lois qu'elle explique gratuitement, et reconnais
        honnêtement ce qu'aucun tableau ne peut montrer (à quel
        point les vrais acteurs sont \emph{petits} et \emph{nombreux}).
\end{enumerate}
\end{problem}
