\chapter{Circuits~: boucles, série, dérivation}\label{ch:g7:circuits-series-parallel}

Quatre années de circuits ont rempli ta boîte à outils~: les boucles,
les schémas, les deux grands câblages. Mais un personnage a travaillé
sans nom depuis le début --- ce quelque chose qui circule. Cette
année, il reçoit son nom, son sens, et sa propriété la plus
étrange~: il n'est jamais consommé. Le courant entrant en scène, la
série et la dérivation cessent d'être des recettes et deviennent des
lois.

\section{Le courant entre en scène}

\begin{definition}[Courant électrique]\label{def:g7:circuits-series-parallel:current}
Le \emph{courant électrique}\index{courant électrique} est
l'écoulement qui parcourt un \omterm{def:g3:first-electric-circuit:circuit}{circuit fermé}~: une foule de particules
chargées --- d'un nombre inimaginable, d'une petitesse inimaginable,
cousines du monde moléculaire --- mises en marche à travers le métal
des fils par la poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile}. Pas de boucle, pas de marche~: le
courant n'existe que tant que la route est fermée. (Ce que sont
exactement ces marcheurs, et comment un métal les laisse passer, est
une belle histoire réservée aux dernières années de ce livre.)
\end{definition}

\begin{proposition}[Le sens conventionnel]\label{prop:g7:circuits-series-parallel:direction}
Par convention mondiale, les \omterm{def:g6:circuits-and-safety:diagram}{schémas électriques} indiquent le courant
comme circulant, \emph{à l'extérieur de la \omterm{def:g3:first-electric-circuit:battery}{pile}, de la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$,
autour du circuit, jusqu'à la \omterm{def:g3:first-electric-circuit:battery}{borne} $-$}. Toutes les flèches de tous
les schémas du monde obéissent à cette convention --- choisie
longtemps avant qu'on ait pu interroger les marcheurs eux-mêmes.
(Quand la physique a enfin pu les interroger, la réponse recelait une
petite plaisanterie célèbre, gardée pour une année plus avancée.)
\end{proposition}

\begin{proposition}[Le courant n'est pas
consommé]\label{prop:g7:circuits-series-parallel:notconsumed}
Les appareils ne mangent pas le courant. Le courant qui entre dans
une lampe, un \omterm{ex:g6:circuits-and-safety:motor}{moteur} ou un \omterm{ex:g6:circuits-and-safety:motor}{vibreur} en ressort entier et poursuit sa
marche~: ce que l'appareil prend, c'est de l'\emph{\omterm{def:g5:everyday-energy:energy}{énergie}} au
réservoir de la \omterm{def:g3:first-electric-circuit:battery}{pile}, livrée par la marche qui passe. Dans toute
boucle \omterm{def:g4:series-circuits:series}{en série}, un seul et même courant traverse donc chaque
appareil --- il en revient à la \omterm{def:g3:first-electric-circuit:battery}{pile} exactement autant qu'il en est
parti.
\end{proposition}

\begin{example}[L'idée fausse en jugement]\label{ex:g7:circuits-series-parallel:trial}
L'erreur tentante~: «~la première lampe consomme un peu de courant,
la seconde doit donc briller plus faiblement~». Le circuit lui-même
la réfute. Câble deux lampes \emph{identiques} \omterm{def:g4:series-circuits:series}{en série} et
regarde~: toutes deux brillent exactement du même éclat --- et les
échanger ne change rien. Si le courant était consommé en chemin, la
lampe numéro deux devrait toujours être la plus faible, quelle que
soit la lampe qui s'y trouve. Elle ne l'est jamais. Le courant arrive
entier~; ce que la paire se partage, c'est la \emph{poussée} de la
\omterm{def:g3:first-electric-circuit:battery}{pile}, et c'est pourquoi toutes deux sont plus faibles qu'une lampe
seule --- également plus faibles.
\end{example}

\section{Les deux familles, en lois}

\begin{proposition}[La loi de la série]\label{prop:g7:circuits-series-parallel:series}
Dans un \omterm{def:g4:series-circuits:series}{circuit en série} --- une boucle unique --- le même courant
traverse chaque appareil~: une route, une marche, sans exception.
Conséquences, désormais expliquées~: une seule coupure arrête la
marche partout à la fois~; des appareils identiques se partagent le
travail à égalité, où qu'ils se trouvent~; et l'ordre des appareils
sur la boucle ne peut pas compter pour une marche qui les traverse
tous de la même façon.
\end{proposition}

\begin{proposition}[La loi de la dérivation]\label{prop:g7:circuits-series-parallel:parallel}
À un \omterm{def:g5:series-parallel-circuits:parallel}{nœud}, le courant \emph{se divise}~: une part de la marche prend
chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche}, et les parts se rejoignent, au complet, au \omterm{def:g5:series-parallel-circuits:parallel}{nœud}
d'arrivée --- rien de perdu, rien de gagné. Chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} reçoit la
pleine poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile}, si bien que chaque appareil travaille
comme s'il était seul~; et une \omterm{def:g5:series-parallel-circuits:parallel}{branche} bloquée se contente de
n'envoyer personne, tandis que les autres \omterm{def:g5:series-parallel-circuits:parallel}{branches} poursuivent leur
marche.
\end{proposition}

\begin{example}[Lire l'éclat comme un
électricien]\label{ex:g7:circuits-series-parallel:brightness}
Une \omterm{def:g3:first-electric-circuit:battery}{pile}, des lampes identiques. Seule~: plein éclat --- la
référence. Deux \omterm{def:g4:series-circuits:series}{en série}~: toutes deux également faibles --- une
marche, une poussée partagée. Deux \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}~: toutes deux à
plein éclat --- chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} jouit de toute la poussée (et la \omterm{def:g3:first-electric-circuit:battery}{pile}
se vide à peu près deux fois plus vite~: rien n'est gratuit). Trois
\omterm{def:g4:series-circuits:series}{en série}~: plus faibles encore, toutes égales. Le coup d'œil de
l'électricien~: compte les compagnons de boucle pour la faiblesse,
compte les \omterm{def:g5:series-parallel-circuits:parallel}{branches} pour la facture de la \omterm{def:g3:first-electric-circuit:battery}{pile}.
\end{example}

\begin{method}[Prédire le sort de n'importe quelle
lampe]\label{met:g7:circuits-series-parallel:predict}
Devant un schéma, pour chaque lampe~:
\begin{enumerate}
  \item \emph{Existence}~: y a-t-il au moins une boucle
        ininterrompue du $+$ jusqu'au $-$ en passant par cette
        lampe~? Pas de boucle, pas de \omterm{def:g1:light-and-shadows:source}{lumière}.
  \item \emph{Compagnons de route}~: quels appareils partagent
        \emph{toutes} les boucles passant par cette lampe~? Ce sont
        ses compagnons \omterm{def:g4:series-circuits:series}{en série} --- plus de compagnons, éclat plus
        faible.
  \item \emph{Voisins de \omterm{def:g5:series-parallel-circuits:parallel}{branche}}~: quels appareils se trouvent sur
        d'autres \omterm{def:g5:series-parallel-circuits:parallel}{branches}~? Ils n'affaiblissent pas cette lampe et
        ne s'éteignent pas avec elle.
  \item \emph{\omterm{ex:g3:first-electric-circuit:switch}{Interrupteurs}}~: un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} ouvert tue exactement
        les boucles sur lesquelles il se trouve --- refais l'étape 1
        en le supprimant.
\end{enumerate}
\end{method}

\begin{remark}[Pourquoi l'idée fausse a la vie
dure]\label{rem:g7:circuits-series-parallel:whyhard}
«~Il faut bien que quelque chose se consomme --- la \omterm{def:g3:first-electric-circuit:battery}{pile} se vide~!~»
C'est vrai~: c'est l'\emph{\omterm{def:g5:everyday-energy:energy}{énergie}} en réserve dans la \omterm{def:g3:first-electric-circuit:battery}{pile} qui est
dépensée, remise aux lampes sous forme de \omterm{def:g1:light-and-shadows:source}{lumière} et de \omterm{def:g5:heat-and-insulation:heat}{chaleur} par
la marche qui passe. Mais les marcheurs, eux, circulent sans dommage,
comme les sièges d'un télésiège qui montent des skieurs toute la
journée sans qu'un seul siège soit jamais consommé. Garde les deux
comptabilités séparées --- le courant circule, l'\omterm{def:g5:everyday-energy:energy}{énergie} est livrée
--- et tous les circuits de ce livre resteront honnêtes.
L'instrument qui compte la marche elle-même nous rejoint l'an
prochain.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:1}
Qu'est-ce que le \omterm{def:g7:circuits-series-parallel:current}{courant électrique}~? Quand existe-t-il dans un
circuit, et quand n'existe-t-il pas~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:2}
Énonce le sens conventionnel du courant. Sur le schéma d'une boucle à
une lampe, où pointent les flèches~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:3}
Deux lampes identiques \omterm{def:g4:series-circuits:series}{en série} brillent exactement du même éclat.
Quelle erreur répandue cette égalité réfute-t-elle, et comment~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:4}
Dans une boucle \omterm{def:g4:series-circuits:series}{en série} comportant un \omterm{ex:g6:circuits-and-safety:motor}{moteur} et une lampe, on
compare le courant qui traverse le \omterm{ex:g6:circuits-and-safety:motor}{moteur} à celui qui traverse la
lampe. Que dit la loi de la série~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:5}
Qu'arrive-t-il au courant au premier \omterm{def:g5:series-parallel-circuits:parallel}{nœud} d'un \omterm{def:g5:series-parallel-circuits:parallel}{circuit en
dérivation}~? Et au second~? Qu'est-il vrai de la part de poussée de
la \omterm{def:g3:first-electric-circuit:battery}{pile} que reçoit chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:6}
Une \omterm{def:g3:first-electric-circuit:battery}{pile}, des lampes identiques~: classe l'éclat d'une lampe ---
seule~; \ avec un compagnon \omterm{def:g4:series-circuits:series}{en série}~; \ avec un voisin \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation}~; \ avec deux compagnons \omterm{def:g4:series-circuits:series}{en série}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:7}
Quelle disposition vide la \omterm{def:g3:first-electric-circuit:battery}{pile} le plus vite~: deux lampes \omterm{def:g4:series-circuits:series}{en série},
ou les deux mêmes \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}~? Pourquoi, en langage de marche~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:8}
Explique l'\omterm{prop:g5:mirrors-reflection:image}{image} du télésiège de
\cref{rem:g7:circuits-series-parallel:whyhard}~: qui joue les
sièges, qui joue les skieurs, et qu'est-ce qui est réellement
consommé~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:9}
Un schéma~: \omterm{def:g3:first-electric-circuit:battery}{pile}, \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}, puis la lampe A~; après A, un \omterm{def:g5:series-parallel-circuits:parallel}{nœud}
envoie deux \omterm{def:g5:series-parallel-circuits:parallel}{branches} --- la lampe B sur l'une, la lampe C sur
l'autre --- qui se rejoignent avant de revenir à la \omterm{def:g3:first-electric-circuit:battery}{pile}. À l'aide de
\cref{met:g7:circuits-series-parallel:predict}~: qui sont les
compagnons \omterm{def:g4:series-circuits:series}{en série} de A~? Que fait l'éclat de A quand la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de
C est coupée~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:10}
Une guirlande décorative demande une conception~: dix lampes qui
brillent toutes à plein éclat, indépendamment, sur une seule
alimentation --- mais l'ensemble doit obéir à un unique \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}
général. Dessine ou décris le câblage, en citant les deux lois.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:11}
Dans le \omterm{def:g5:series-parallel-circuits:parallel}{circuit en dérivation} à deux lampes, la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de la lampe 1
est en gros fil de cuivre tandis que celle de la lampe 2 comporte un
long fil fin. La lampe 2 brille un peu plus faiblement. Que cela
suggère-t-il sur la façon dont la route d'une \omterm{def:g5:series-parallel-circuits:parallel}{branche} peut, à elle
seule, étrangler sa part de la marche~? (L'an prochain donnera un nom
et une loi à cet étranglement.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:circuits-series-parallel:12}
Une \omterm{def:g3:first-electric-circuit:battery}{pile} alimente deux \omterm{def:g5:series-parallel-circuits:parallel}{branches} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}~: la \omterm{def:g5:series-parallel-circuits:parallel}{branche} numéro un
porte la lampe X seule~; la \omterm{def:g5:series-parallel-circuits:parallel}{branche} numéro deux porte les lampes Y et
Z \omterm{def:g4:series-circuits:series}{en série}. Compare les éclats des trois lampes, dans l'ordre, avec
des raisons tirées des deux lois. Puis prédis le sort de chaque lampe
si Z grille --- et si, à la place, c'est X qui grille.
\end{exercise}

\section{Problème~: le tableau du gardien de phare}

\begin{problem}[{Problème du week-end --- recâbler le vieux phare~;
une seule batterie, bien des services~; la nuit de pannes du
gardien}]\label{pb:g7:circuits-series-parallel:1}
La batterie du phare doit alimenter~: la grande lampe, une corne de
brume (un puissant \omterm{ex:g6:circuits-and-safety:motor}{vibreur}), la lampe de lecture du gardien et
l'éclairage de l'escalier --- et survivre à la nuit de dépannage du
gardien.

\medskip\noindent\textbf{Partie I --- la conception.}
\begin{enumerate}
  \item La grande lampe ne doit jamais faiblir parce qu'autre chose
        s'allume. Quel câblage cela impose-t-il à toute
        l'installation, et pourquoi tout appariement \omterm{def:g4:series-circuits:series}{en série}
        trahirait-il les navires~?
  \item Chaque service a besoin de son propre \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}, et le
        gardien veut une coupure générale pour les tempêtes. Place
        les cinq \omterm{ex:g3:first-electric-circuit:switch}{interrupteurs}.
  \item La corne de brume et l'éclairage de l'escalier ne doivent
        jamais être empêchés de fonctionner par les pannes l'un de
        l'autre. Est-ce déjà garanti~? Par quelle loi~?
  \item Esquisse ou décris le schéma complet~: la batterie,
        l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} général, quatre \omterm{def:g5:series-parallel-circuits:parallel}{branches}, quatre
        \omterm{ex:g3:first-electric-circuit:switch}{interrupteurs} de \omterm{def:g5:series-parallel-circuits:parallel}{branche}.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- les questions du gardien.}
\begin{enumerate}[resume]
  \item Le gardien s'inquiète~: «~Quatre services qui boivent en même
        temps --- chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} reçoit-elle encore toute la poussée
        de la batterie~?~» Réponds avec la loi de la dérivation.
  \item «~Et le courant est-il consommé à mesure qu'il travaille
        dans la grande lampe~?~» Remets le gardien sur la voie, avec
        le télésiège si cela aide.
  \item Par une nuit de brouillard, tout fonctionne à la fois. Quel
        est le prix de la générosité de la dérivation, et où
        est-il payé dans le phare~?
  \item Seule la lampe de lecture est allumée~; en bon mesureur, le
        gardien suit la marche~: décris son aller-retour complet, en
        nommant chaque composant qu'elle traverse et chacun qu'elle
        évite.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- la nuit des pannes.}
\begin{enumerate}[resume]
  \item Minuit~: la lampe de l'escalier grille. Qu'est-ce qui
        s'éteint en plus~? Quelle loi répond~?
  \item Une heure~: une réparation maladroite laisse un tronçon de
        fil nu relier directement les deux \omterm{def:g3:first-electric-circuit:battery}{bornes} de la batterie, en
        amont de toutes les \omterm{def:g5:series-parallel-circuits:parallel}{branches}. Nomme l'événement, et explique
        pourquoi tous les services du phare tombent d'un coup ---
        où va la marche~?
  \item Deux \omterm{ex:g2:measuring-time:clock}{heures}, panne écartée~: le gardien veut protéger la
        grande lampe pour qu'un tel raccourci ne puisse plus jamais
        la faire taire. Une seconde batterie indépendante est
        disponible. Propose le câblage.
  \item À l'aube, rédige le journal du gardien --- trois phrases~:
        quelle loi a mis les navires en sécurité cette nuit, quelle
        idée fausse la nuit a réfutée, et quel unique fil a causé la
        pire heure.
\end{enumerate}
\end{problem}
