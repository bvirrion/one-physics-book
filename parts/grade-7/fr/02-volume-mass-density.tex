\chapter{Volume, masse, masse volumique}\label{ch:g7:volume-mass-density}

L'an dernier, tu as démasqué un orfèvre fraudeur en divisant des
\omterm{def:g3:measuring-mass:units}{grammes} par des \omterm{def:g6:mass-and-volume:volume}{centimètres cubes}. Cette année, ton cours de
mathématiques t'a remis un pouvoir nouveau~: des lettres qui
représentent des nombres. La \omterm{def:g7:volume-mass-density:formula}{masse volumique} est l'endroit parfait
pour le dépenser --- une courte formule qui tasse trois recettes, la
trousse d'un détective et toute la règle de la flottaison en cinq
symboles.

\section{La formule}

\begin{definition}[La masse volumique, par la
formule]\label{def:g7:volume-mass-density:formula}
La \emph{masse volumique}\index{masse volumique} $\rho$ (la lettre
grecque \emph{rhô}) d'une substance est sa \omterm{def:g3:measuring-mass:mass}{masse} par \omterm{def:g6:measurement-in-science:measuring}{unité} de
\omterm{def:g6:mass-and-volume:volume}{volume}~: pour un échantillon de \omterm{def:g3:measuring-mass:mass}{masse} $m$ et de \omterm{def:g6:mass-and-volume:volume}{volume} $V$,
\[
\rho = \frac{m}{V}.
\]
Avec $m$ en \omterm{def:g3:measuring-mass:units}{grammes} et $V$ en \omterm{def:g6:mass-and-volume:volume}{centimètres cubes}, $\rho$ sort en
\emph{\omterm{def:g3:measuring-mass:units}{grammes} par \omterm{def:g6:mass-and-volume:volume}{centimètre cube}} (\unit{g/cm^3}) --- la «~\omterm{def:g3:measuring-mass:mass}{masse}
d'un \omterm{def:g6:mass-and-volume:volume}{centimètre cube}~» de l'an dernier, désormais en uniforme. \omterm{def:g3:measuring-mass:mass}{Masse}
volumique de l'eau~: $\rho = \qty{1.0}{g/cm^3}$. Et comme un \omterm{def:g6:mass-and-volume:volume}{litre}
vaut $1000$ \omterm{def:g6:mass-and-volume:volume}{centimètres cubes}, le même nombre se lit aussi en
\omterm{def:g3:measuring-mass:units}{kilogrammes} par \omterm{def:g6:mass-and-volume:volume}{litre}~: l'eau fait \qty{1.0}{kg} par \omterm{def:g6:mass-and-volume:volume}{litre}.
\end{definition}

\begin{example}[Lire les lettres]\label{ex:g7:volume-mass-density:reading}
Une formule est une phrase en sténographie. $\rho = m/V$ se lit~:
«~pour trouver la \omterm{def:g7:volume-mass-density:formula}{masse volumique}, diviser la \omterm{def:g3:measuring-mass:mass}{masse} de l'échantillon
par son \omterm{def:g6:mass-and-volume:volume}{volume}~». Les lettres sont des cases à remplir --- pour
\emph{ce} caillou-ci, $m$ devient \qty{75}{g} et $V$ devient
\qty{30}{cm^3}~:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
La même division que l'an dernier --- mais la formule, elle, se
souvient de la recette pour tous les cailloux à venir.
\end{example}

\begin{proposition}[Trois recettes dans une seule
formule]\label{prop:g7:volume-mass-density:recipes}
Comme la \omterm{def:g7:volume-mass-density:formula}{masse volumique} lie la \omterm{def:g3:measuring-mass:mass}{masse} et le \omterm{def:g6:mass-and-volume:volume}{volume} par une
proportion, une seule formule sert trois besoins~:
\begin{enumerate}
  \item \emph{trouver une \omterm{def:g7:volume-mass-density:formula}{masse volumique}}~: $\rho = m/V$ --- on
        divise la \omterm{def:g3:measuring-mass:mass}{masse} par le \omterm{def:g6:mass-and-volume:volume}{volume}~;
  \item \emph{trouver une \omterm{def:g3:measuring-mass:mass}{masse}}~: $m = \rho \times V$ --- chaque
        \omterm{def:g6:mass-and-volume:volume}{centimètre cube} porte $\rho$ \omterm{def:g3:measuring-mass:units}{grammes}, et il y en a $V$~;
  \item \emph{trouver un \omterm{def:g6:mass-and-volume:volume}{volume}}~: $V = m/\rho$ --- on partage la
        \omterm{def:g3:measuring-mass:mass}{masse} en parts de $\rho$ \omterm{def:g3:measuring-mass:units}{grammes}~; le nombre de parts est le
        nombre de \omterm{def:g6:mass-and-volume:volume}{centimètres cubes}.
\end{enumerate}
\end{proposition}

\begin{example}[Les recettes au travail]\label{ex:g7:volume-mass-density:atwork}
\emph{La \omterm{def:g3:measuring-mass:mass}{masse} à partir du \omterm{def:g6:mass-and-volume:volume}{volume}}~: que pèsent \qty{250}{cm^3} de
fer ($\rho = 7.9$)~? $m = 7.9 \times 250 = \qty{1975}{g}$ ---
presque deux \omterm{def:g3:measuring-mass:units}{kilogrammes} dans le \omterm{def:g6:mass-and-volume:volume}{volume} d'un mug.
\emph{Le \omterm{def:g6:mass-and-volume:volume}{volume} à partir de la \omterm{def:g3:measuring-mass:mass}{masse}}~: quelle place occupent
\qty{540}{g} de chêne ($\rho = 0.6$)~?
$V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{Le compagnon de bon sens}~: les réponses en fer doivent être
lourdes-et-petites, celles en bois légères-et-grandes --- le tableau
des \omterm{def:g7:volume-mass-density:formula}{masses volumiques} est le meilleur ami de l'étape du jugement.
\end{example}

\section{Mesurer des masses volumiques}

\begin{method}[Masse volumique d'un liquide]\label{met:g7:volume-mass-density:liquid}
Les \omterm{def:g3:solids-liquids-gases:liquid}{liquides} ne tiennent pas seuls sur un plateau de \omterm{def:g3:levers-and-scales:scale}{balance} ---
empruntons donc l'astuce de la tare~:
\begin{enumerate}
  \item poser une éprouvette graduée vide sur la \omterm{def:g3:levers-and-scales:scale}{balance} et la tarer
        à zéro~;
  \item y verser un \omterm{def:g6:mass-and-volume:volume}{volume} commode --- disons
        $V = \qty{80}{cm^3}$ du \omterm{def:g3:solids-liquids-gases:liquid}{liquide} mystérieux, en lisant
        correctement le \omterm{met:g6:mass-and-volume:cylinder}{ménisque}~;
  \item la \omterm{def:g3:levers-and-scales:scale}{balance} affiche alors la \omterm{def:g3:measuring-mass:mass}{masse} du \omterm{def:g3:solids-liquids-gases:liquid}{liquide} seul~: disons
        $m = \qty{68}{g}$~;
  \item diviser~: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ ---
        plus léger que l'eau~: sans doute une huile.
\end{enumerate}
\end{method}

\begin{example}[Masse volumique d'un solide, du début à la
fin]\label{ex:g7:volume-mass-density:solid}
Une médaille se prétend en argent ($\rho = 10.5$). \omterm{def:g3:levers-and-scales:scale}{Balance}~:
$m = \qty{84}{g}$. Éprouvette~: l'eau monte de \qty{60.0}{cm^3} à
\qty{68.0}{cm^3}, donc $V = \qty{8.0}{cm^3}$. Alors
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- la signature
correspond~: ce peut bien être de l'argent. (Un métal bon marché
habillé d'un placage d'argent se serait trahi ici --- à moins d'être
choisi avec ruse, comme le montrera le problème du week-end.)
\end{example}

\begin{remark}[Les unités dans les formules]\label{rem:g7:volume-mass-density:units}
Une formule n'est honnête que si ses \omterm{def:g6:measurement-in-science:measuring}{unités} s'accordent. Avec $m$ en
\omterm{def:g3:measuring-mass:units}{grammes} et $V$ en \unit{cm^3}, $\rho$ parle en \unit{g/cm^3}~;
nourris-la de \omterm{def:g3:measuring-mass:units}{kilogrammes} et de \omterm{def:g6:mass-and-volume:volume}{litres}, elle répond en \unit{kg/L}
--- heureusement, le même nombre pour une substance donnée. Mais
mélange des \omterm{def:g3:measuring-mass:units}{grammes} et des \omterm{def:g6:mass-and-volume:volume}{litres}, et la formule, sans se plaindre,
livre un non-sens. Règle des professionnels~: avant de calculer,
passer les \omterm{def:g6:measurement-in-science:measuring}{unités} en revue~; après avoir calculé, les écrire dans la
réponse.
\end{remark}

\section{Penser en masses volumiques}

\begin{example}[Alliages et intermédiaires]\label{ex:g7:volume-mass-density:alloys}
Mélange deux substances et la \omterm{def:g7:volume-mass-density:formula}{masse volumique} du mélange tombe
\emph{entre} leurs signatures --- plus près de celle de l'ingrédient
le plus généreux. Le bronze (du cuivre à $9.0$ avec un peu d'étain)
signe près de $8.8$~; le mélange or-argent de la couronne signait
entre $10.5$ et $19.3$~; l'eau de mer, du sel dissous dans de l'eau,
grimpe à environ $1.03$. L'intermédiaire est en soi un indice~: une
lecture de $12$ pour un lingot «~d'or pur~» est un aveu de mauvaise
compagnie.
\end{example}

\begin{example}[La masse volumique décide du monde
flottant]\label{ex:g7:volume-mass-density:floating}
La règle de la flottaison, désormais en uniforme~: un objet \omterm{def:g1:floating-and-sinking:words}{flotte}
dans un \omterm{def:g3:solids-liquids-gases:liquid}{liquide} quand
$\rho_{\text{objet}} < \rho_{\text{liquide}}$. Le chêne ($0.6$) sur
l'eau ($1.0$)~: il \omterm{def:g1:floating-and-sinking:words}{flotte}. La glace ($0.9$) dans l'huile ($0.85$)~:
elle \omterm{def:g1:floating-and-sinking:words}{coule} --- consulte le tableau, pas ton intuition. Un nageur
($\approx 1.0$) dans les fameux lacs ultra-salés ($\approx 1.2$)~: il
\omterm{def:g1:floating-and-sinking:words}{flotte} comme un bouchon, le journal à la main. Une seule inégalité,
et tout le port.
\end{example}

\begin{remark}[Attention à la foule, pas au colis]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ décrit des \emph{substances}~; les navires et les gilets
de sauvetage gonflés sont des \emph{colis} --- de la substance plus
de l'\omterm{def:g2:air-around-us:air}{air} emprisonné --- et c'est le $m/V$ d'ensemble du colis qui
affronte la règle de la flottaison. La formule traite les deux, à
condition de la nourrir des bons $m$ et $V$~: la \omterm{def:g7:volume-mass-density:formula}{masse volumique}
propre de l'acier pour la substance, la \omterm{def:g3:measuring-mass:mass}{masse} de toute la coque sur
le \omterm{def:g6:mass-and-volume:volume}{volume} de toute la coque pour le colis. La plupart des
«~paradoxes~» de la flottaison ne sont que les deux comptabilités
confondues.
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:4}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:5}

volumique}~? Un glaçon ($0.9$) y flotte-t-il ou y coule-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:6}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:7}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:8}

\omterm{def:g3:measuring-mass:mass}{masse} d'un \omterm{def:g2:measuring-length:units}{mètre} cube d'eau, en \omterm{def:g3:measuring-mass:units}{kilogrammes}~? (La réponse explique

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:10}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:11}

\end{exercise}

\section{Problème~: le détective de la casse}

\begin{problem}[{Problème du week-end --- un après-midi avec le
détective des métaux de la casse~; quatre lots mystérieux, un lingot
imposteur~; les limites du test de masse
volumique}]\label{pb:g7:volume-mass-density:1}
La casse achète le métal pour ce qu'il \emph{est}, non pour ce dont
il a l'\omterm{def:g2:air-around-us:air}{air}, et la détective du chantier travaille avec une \omterm{def:g3:levers-and-scales:scale}{balance},
un grand récipient gradué et le tableau des \omterm{def:g7:volume-mass-density:formula}{masses volumiques}. Tu es
l'apprenti. (Extrait du tableau~: aluminium $2.7$~; fer $7.9$~;
cuivre $9.0$~; plomb $11.3$~; or $19.3$~; tungstène $19.3$.)

\medskip\noindent\textbf{Partie I --- quatre lots.}
\begin{enumerate}
  \item Lot A, un lingot gris~: $m = \qty{5400}{g}$, déplacement
        $V = \qty{2000}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Masse volumique} et identité~?
  \item Lot B, une bobine de fil~: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Masse volumique} et identité~?
  \item Lot C, une plaque terne et lourde~: $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Masse volumique} et identité --- et
        pourquoi le lot C doit-il être manipulé avec des gants et
        du respect~?
  \item Lot D, un sac de ferraille claire et mêlée~:
        $m = \qty{8100}{g}$, $V = \qty{1500}{cm^3}$. Montre que la
        \omterm{def:g7:volume-mass-density:formula}{masse volumique} du lot D tombe entre deux métaux du
        tableau, et dis ce que le sac contient le plus
        vraisemblablement.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- prix et prévisions.}
La casse paie à la \omterm{def:g3:measuring-mass:mass}{masse} mais organise le transport au \omterm{def:g6:mass-and-volume:volume}{volume}.
\begin{enumerate}[resume]
  \item Un acheteur veut \qty{540}{g} d'aluminium découpés dans le
        lot A. Quel \omterm{def:g6:mass-and-volume:volume}{volume} de lingot cela représente-t-il~?
  \item La bobine de cuivre du lot B doit être fondue en cubes de
        \qty{25}{cm^3} chacun. Quelle est la \omterm{def:g3:measuring-mass:mass}{masse} d'un cube, et
        combien de cubes pleins la bobine donne-t-elle~?
  \item Une caisse peut porter au plus \qty{20}{kg}. Combien de ces
        cubes de cuivre a-t-elle le droit de contenir~?
  \item Le puits de chargement du camion peut encore recevoir
        \qty{3000}{cm^3} de ferraille de fer. Quelle \omterm{def:g3:measuring-mass:mass}{masse}
        supplémentaire, en \omterm{def:g3:measuring-mass:units}{kilogrammes}, la détective a-t-elle le
        droit de charger~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- l'imposteur.}
Un vendeur se présente avec un lingot «~d'or~» étincelant~:
$m = \qty{3860}{g}$, déplacement $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item Calcule la \omterm{def:g7:volume-mass-density:formula}{masse volumique} du lingot. Correspond-elle à la
        signature de l'or~?
  \item La détective, imperturbable, consulte la dernière ligne du
        tableau et soupire. Quel métal bon marché porte exactement
        la \omterm{def:g7:volume-mass-density:formula}{masse volumique} de l'or --- et que cela apprend-il sur
        les limites du test de \omterm{def:g7:volume-mass-density:formula}{masse volumique}~?
  \item Lequel des deux instruments de la casse a été mis en échec
        ici~: la \omterm{def:g3:levers-and-scales:scale}{balance}, le récipient, les deux, ou aucun~? Dis
        précisément ce que le test de \omterm{def:g7:volume-mass-density:formula}{masse volumique} a
        honnêtement établi au sujet du lingot.
  \item Propose un autre test physique, tiré des années
        précédentes de ce cours, qui distingue l'or de son sosie
        sans abîmer le lingot. (Leurs \omterm{def:g4:changes-of-state:melting-point}{températures de fusion}
        diffèrent énormément --- l'or vers \qty{1064}{\celsius}, le
        sosie bien au-delà de tous les fours d'ici --- mais aucune
        casse ne fond un trésor éventuel~: trouve une preuve plus
        douce, peut-être le verdict de l'\omterm{def:g1:magnets:magnet}{aimant} sur les âmes de
        fer, ou le \omterm{def:g3:sound-around-us:sound}{son} d'un lingot frappé, et défends honnêtement
        ton choix.)
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- l'art de la détective.}
\begin{enumerate}[resume]
  \item Rédige le credo en trois lignes de la détective~: ce que la
        \omterm{def:g7:volume-mass-density:formula}{masse volumique} peut prouver, ce qu'elle ne peut que
        suggérer, et ce qu'elle ne peut jamais faire toute seule.
\end{enumerate}
\end{problem}
