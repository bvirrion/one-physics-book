\chapter{Courts-circuits et sécurité électrique}\label{ch:g7:short-circuits-safety}

Toutes les règles de sécurité que tu as apprises remontent à un seul
coupable, brièvement rencontré l'an dernier~: le \omterm{def:g7:short-circuits-safety:device}{court-circuit}. Ce
chapitre l'étudie comme il faut --- comment il naît, pourquoi il
brûle, comment un humble fil sacrificiel en vient à bout --- et
transforme le tableau électrique de la famille, de placard mystérieux
en allié que tu comprends.

\section{La méthode du coupable}

\begin{proposition}[Le courant préfère la route sans
effort]\label{prop:g7:short-circuits-safety:effortless}
Entre deux routes reliant les deux mêmes points, le courant en marche
se presse sur celle qui s'oppose le moins à lui. Les appareils ---
lampes, \omterm{ex:g6:circuits-and-safety:motor}{moteurs}, \omterm{ex:g6:circuits-and-safety:motor}{vibreurs} --- s'opposent tous à la marche pendant
qu'ils travaillent~; un simple gros fil ne s'y oppose presque pas.
Offre à la marche une dérivation en fil nu, et elle déserte la route
qui travaille, presque entièrement, pour celle qui ne coûte aucun
effort.
\end{proposition}

\begin{definition}[Court-circuit d'un appareil]\label{def:g7:short-circuits-safety:device}
Un appareil est \emph{court-circuité}\index{court-circuit} quand une
route conductrice relie directement ses deux \omterm{def:g3:first-electric-circuit:battery}{bornes} en le
contournant. La marche déserte l'appareil --- une lampe
court-circuitée s'éteint, un \omterm{ex:g6:circuits-and-safety:motor}{moteur} s'arrête --- pendant que le reste
de la boucle continue, avec désormais un travailleur de moins à
s'opposer à la marche.
\end{definition}

\begin{definition}[Court-circuit de la pile]\label{def:g7:short-circuits-safety:battery}
Le cas le plus grave~: une route conductrice relie les deux \omterm{def:g3:first-electric-circuit:battery}{bornes}
\emph{de la \omterm{def:g3:first-electric-circuit:battery}{pile}} sans le moindre appareil sur son trajet. Plus rien
ne s'oppose à la marche, qui devient une ruée --- et cette ruée
folle échauffe le fil et la \omterm{def:g3:first-electric-circuit:battery}{pile} elle-même, détruisant les
éléments, faisant \omterm{def:g2:ice-water-steam:meltfreeze}{fondre} les \omterm{def:g4:conductors-insulators:insulator}{isolants} et, sur le secteur, mettant le
feu aux maisons.
\end{definition}

\begin{method}[Le coupable, démontré sans
danger]\label{met:g7:short-circuits-safety:demo}
Avec une \omterm{def:g3:first-electric-circuit:battery}{pile} et deux lampes \omterm{def:g4:series-circuits:series}{en série} (sans danger~: la poussée de la
\omterm{def:g3:first-electric-circuit:battery}{pile} est minuscule)~:
\begin{enumerate}
  \item allumer les deux lampes --- également faibles, comme le
        promet la loi de la série~;
  \item poser maintenant un simple fil en travers des deux \omterm{def:g3:first-electric-circuit:battery}{bornes} de
        la lampe 2, pour la contourner~;
  \item observer~: la lampe 2 meurt instantanément --- désertée ---
        pendant que la lampe 1 devient \emph{plus brillante}~: un
        opposant contourné, la marche de toute la boucle se
        renforce~;
  \item retirer le fil~: la marche revient dans la lampe 2, et
        l'égale faiblesse revient elle aussi.
\end{enumerate}
Une seule expérience, trois leçons~: la règle du contournement,
l'appareil déserté et la ruée renforcée.
\end{method}

\begin{example}[Pourquoi les fils chauffent]\label{ex:g7:short-circuits-safety:heating}
Un courant en marche réchauffe toujours un peu sa route --- la lueur
de la lampe \emph{est} ce réchauffement, mis au travail exprès dans
un filament conçu pour rougir à blanc. Dans une ruée, le
réchauffement devient une menace~: la ruée d'un \omterm{def:g7:short-circuits-safety:device}{court-circuit} peut
porter un fil de cuivre au-delà de la \omterm{def:g4:changes-of-state:melting-point}{température de fusion} de sa
gaine plastique en quelques secondes. L'\omterm{def:g1:five-senses:senses}{avant-goût} de poche~: une
\omterm{def:g3:first-electric-circuit:battery}{pile} court-circuitée par un fil devient chaude dans ta main --- la
même physique qui, à l'échelle du secteur, noircit les murs. (Comment
le réchauffement croît avec la ruée sera honnêtement mesuré dans deux
ans, avec le chapitre sur la puissance électrique.)
\end{example}

\section{Les gardes du corps}

\begin{definition}[Fusible]\label{def:g7:short-circuits-safety:fuse}
Un \emph{fusible}\index{fusible} est un garde du corps qui meurt pour
toi~: un fil volontairement fin, placé \omterm{def:g4:series-circuits:series}{en série} avec le circuit qu'il
protège, choisi pour \omterm{def:g2:ice-water-steam:meltfreeze}{fondre} --- et donc couper la boucle --- dès que
le courant dépasse son calibre. La ruée qui carboniserait des \omterm{def:g2:measuring-length:units}{mètres}
de câblage de maison détruit à la place trois \omterm{def:g2:measuring-length:units}{centimètres} de fil
sacrificiel dans sa cartouche ininflammable. Un fusible grillé se
remplace --- \emph{après} avoir trouvé le défaut --- par un fusible
neuf de même calibre.
\end{definition}

\begin{example}[Disjoncteur et fusible, deux gardes un seul
devoir]\label{ex:g7:short-circuits-safety:breaker}
Le disjoncteur moderne fait le devoir du \omterm{def:g7:short-circuits-safety:fuse}{fusible} sans mourir~: un
\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} à ressort qui s'ouvre d'un coup sous la ruée et se
réarme d'une pichenette --- après que le défaut a été soigné.
\omterm{def:g7:short-circuits-safety:fuse}{Fusible} ou disjoncteur, le poste du garde est le même~: \emph{\omterm{def:g4:series-circuits:series}{en
série}}, sur la \omterm{def:g5:series-parallel-circuits:parallel}{branche} qu'il protège, pour que la ruée soit obligée
de passer par son poste de contrôle. Un garde placé \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}
regarderait la ruée passer en tonnant sur l'autre route --- pure
décoration.
\end{example}

\begin{example}[La surcharge~: la ruée polie]\label{ex:g7:short-circuits-safety:overload}
Toute ruée dangereuse n'est pas un \omterm{def:g7:short-circuits-safety:device}{court-circuit}. Une multiprise qui
alimente à la fois un radiateur, une bouilloire et un grille-pain
demande à son unique \omterm{def:g5:series-parallel-circuits:parallel}{branche} de porter trois marches complètes en
même temps --- chaque appareil est innocent, mais leur somme est une
surcharge qui réchauffe le câblage du mur comme un grille-pain lent.
Le disjoncteur saute sur la somme, et il a bien raison. Règle de la
maison~: un seul gros appareil de chauffe par prise~; la multiprise
est faite pour les lampes et les chargeurs, non pour une cuisine
privée.
\end{example}

\begin{remark}[La réparation interdite]\label{rem:g7:short-circuits-safety:nail}
La vieille histoire est vraie et tue encore~: le \omterm{def:g7:short-circuits-safety:fuse}{fusible} d'une maison
grille sans arrêt, et quelqu'un le «~répare~» avec un clou ou une
pièce de monnaie --- un \omterm{def:g3:solids-liquids-gases:solid}{solide} \omterm{def:g4:conductors-insulators:conductor}{conducteur} qui ne fondra jamais. Le
symptôme est réduit au silence~; la maladie continue de faire rage.
La ruée suivante ne rencontre aucun poste de contrôle, et les fils de
la maison deviennent eux-mêmes le \omterm{def:g7:short-circuits-safety:fuse}{fusible} --- à l'intérieur des murs.
Ne contourne jamais un garde~: un \omterm{def:g7:short-circuits-safety:fuse}{fusible} qui grille sans arrêt est
un garde qui, sans arrêt, dit la vérité.
\end{remark}

\begin{remark}[Les règles, forme définitive]\label{rem:g7:short-circuits-safety:rules}
Le code électrique de la famille, avec ses raisons désormais
complètes~: ne jamais ponter les \omterm{def:g3:first-electric-circuit:battery}{bornes} d'une \omterm{def:g3:first-electric-circuit:battery}{pile}, et tenir les clés
plates loin des batteries de voiture (une batterie court-circuitée
par du métal est un arc de soudure)~; un seul appareil gourmand en
\omterm{def:g5:heat-and-insulation:heat}{chaleur} par prise (la surcharge)~; les cordons abîmés mis à la
retraite (une gaine effilochée est un \omterm{def:g7:short-circuits-safety:device}{court-circuit} qui répète son
rôle)~; l'eau et le secteur séparés, les mains sèches toujours (ton
corps ne doit jamais servir de contournement)~; les gardes respectés
--- des \omterm{def:g7:short-circuits-safety:fuse}{fusibles} au bon calibre uniquement, et le défaut trouvé avant
tout réarmement~; et la prise elle-même, à jamais, ni pour les doigts
ni pour les expériences.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:1}
Énonce la proposition de la route sans effort. Pourquoi un fil nu
enlève-t-il la marche à une lampe~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:2}
Distingue le \omterm{def:g7:short-circuits-safety:device}{court-circuit} d'un appareil du \omterm{def:g7:short-circuits-safety:device}{court-circuit} de la \omterm{def:g3:first-electric-circuit:battery}{pile}.
Lequel est le plus grave, et pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:3}
Dans la démonstration sans danger, pourquoi la lampe 2 meurt-elle ---
et pourquoi la lampe 1 devient-elle plus brillante au même instant~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:4}
Pourquoi un \omterm{def:g7:short-circuits-safety:device}{court-circuit} échauffe-t-il son fil~? Quel appareil de
tous les jours utilise le même réchauffement à dessein~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:5}
Qu'est-ce qu'un \omterm{def:g7:short-circuits-safety:fuse}{fusible}, et que sacrifie-t-il~? Pourquoi doit-il se
placer \omterm{def:g4:series-circuits:series}{en série} avec ce qu'il protège~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:6}
En quoi un disjoncteur améliore-t-il le \omterm{def:g7:short-circuits-safety:fuse}{fusible}~? Que doit-il se
passer avant de réarmer ou de remplacer l'un ou l'autre garde~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:7}
Un radiateur, une bouilloire et un grille-pain se partagent une seule
multiprise. Aucun fil n'est nu, aucun appareil n'est défectueux ---
et pourtant le disjoncteur saute. Nomme l'événement et explique la
plainte de la \omterm{def:g5:series-parallel-circuits:parallel}{branche}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:8}
Pourquoi «~réparer~» un \omterm{def:g7:short-circuits-safety:fuse}{fusible} avec un clou compte-t-il parmi les
gestes les plus dangereux d'une maison~? Qu'est-ce qui devient le
\omterm{def:g7:short-circuits-safety:fuse}{fusible} ensuite~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:9}
Les deux fils intérieurs d'un cordon, aux gaines effilochées, se
touchent à l'intérieur du câble --- en amont de la lampe qu'ils
alimentent. Diagnostique~: quel genre de \omterm{def:g7:short-circuits-safety:device}{court-circuit},
qu'arrive-t-il à la lampe, et que fait le garde de la \omterm{def:g5:series-parallel-circuits:parallel}{branche}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:10}
Dans la démonstration à deux lampes, prédis ce qui se passe si le fil
de contournement est posé en travers des \emph{deux} lampes à la fois
--- d'avant la lampe 1 à après la lampe 2. Quelle définition
s'applique alors, et pourquoi l'expérimentateur ne doit-il pas
l'essayer~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:11}
Mystère du matin de Noël~: la guirlande \omterm{def:g4:series-circuits:series}{en série} de la grand-mère a
une lampe dont les fils de soutien intérieurs ont fusionné en
mourant --- elle s'est court-circuitée elle-même. Contrairement à la
tragédie habituelle, \emph{le reste de la guirlande continue de
briller}, un peu plus fort. Explique --- et dis ce qui se prépare
maintenant, lentement, pour les lampes restantes.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:short-circuits-safety:12}
Conçois le plan de protection d'un petit atelier~: trois \omterm{def:g5:series-parallel-circuits:parallel}{branches} \omterm{def:g5:series-parallel-circuits:parallel}{en
dérivation} (l'éclairage~; les outils électriques~; un chargeur de
batterie), une seule alimentation. Place les gardes et les
\omterm{ex:g3:first-electric-circuit:switch}{interrupteurs}~; choisis quelles \omterm{def:g5:series-parallel-circuits:parallel}{branches} méritent leur propre garde
et pourquoi~; et rédige l'avis en deux lignes à afficher au-dessus du
tiroir à \omterm{def:g7:short-circuits-safety:fuse}{fusibles}, qui résume
\cref{rem:g7:short-circuits-safety:nail}.
\end{exercise}

\section{Problème~: le rapport de l'enquêtrice incendie}

\begin{problem}[{Problème du week-end --- après l'incendie de
l'appartement~; l'enquêtrice lit les aveux du câblage~; trois
défauts, une leçon}]\label{pb:g7:short-circuits-safety:1}
Personne n'a été blessé --- la famille était absente --- mais le mur
de la cuisine est carbonisé, et l'enquêtrice doit reconstituer la
nuit à partir des preuves. Tu portes son carnet.

\medskip\noindent\textbf{Partie I --- lire les ruines.}
\begin{enumerate}
  \item Pièce à conviction numéro un~: une multiprise qui alimentait
        un radiateur, une bouilloire et une friteuse, son plastique
        à moitié fondu. Aucun fil nu à l'intérieur. Nomme le défaut
        que cette pièce suggère, et dis pourquoi aucun «~fil
        coupable~» n'est nécessaire pour l'expliquer.
  \item Pièce numéro deux~: un cordon derrière le placard, rongé
        par un animal, ses deux \omterm{def:g4:conductors-insulators:conductor}{conducteurs} se touchant à l'endroit
        de la morsure. Nomme ce défaut, et dis où est allée la
        marche à partir de l'instant du contact.
  \item Pièce numéro trois, la plus accablante~: dans le tableau
        électrique, la cartouche de \omterm{def:g7:short-circuits-safety:fuse}{fusible} de la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de la
        cuisine contient --- une bande de papier d'aluminium
        roulée. Dis ce que cette «~réparation~» a fait à la
        protection de la \omterm{def:g5:series-parallel-circuits:parallel}{branche}.
  \item Assemble les trois pièces en une histoire vraisemblable~:
        quel défaut est venu en premier, qu'a fait le vrai \omterm{def:g7:short-circuits-safety:fuse}{fusible}
        à ce sujet (avant la pièce numéro trois), et pourquoi la
        nuit finale s'est-elle terminée par un incendie plutôt que
        par une cuisine plongée dans le noir~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- la déposition de physique.}
\begin{enumerate}[resume]
  \item Le tribunal demande~: pourquoi un fil \omterm{def:g7:short-circuits-safety:device}{court-circuité} ou
        surchargé \emph{chauffe-t-il}~? Cite l'appareil de tous les
        jours qui prouve que le réchauffement par le courant peut
        être apprivoisé --- et dis ce qui manquait ici.
  \item Pourquoi un \omterm{def:g7:short-circuits-safety:fuse}{fusible} est-il fin exprès~? Explique
        l'expression «~un garde du corps qui meurt pour toi~».
  \item La bande d'aluminium conduisait magnifiquement.
        Précisément \emph{à cause de cela}, elle a échoué comme
        garde. Résous ce petit paradoxe pour le jury.
  \item Le radiateur seul était sans danger, la bouilloire seule
        sans danger, la friteuse seule sans danger. Écris une phrase
        sur la façon dont trois sécurités ont fait la somme d'un
        danger.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- le verdict et les leçons.}
\begin{enumerate}[resume]
  \item L'enquêtrice doit classer les trois défauts par gravité
        morale~: la surcharge, le cordon rongé, l'aluminium. Lequel
        a transformé l'incident en catastrophe, et pourquoi~?
  \item La famille demande ce qui se serait passé, la nuit de
        l'incendie, si un vrai \omterm{def:g7:short-circuits-safety:fuse}{fusible} avait été en place. Fais-lui
        vivre la scène, minute par minute.
  \item Recommande le nouveau câblage~: comment le radiateur, la
        bouilloire et la friteuse doivent-ils être alimentés
        désormais, et quelle règle permanente gouverne l'avenir de
        la multiprise~?
  \item Referme le rapport par le résumé en trois phrases de
        l'enquêtrice~: la méthode du coupable, le devoir du garde,
        et l'unique réparation qui ne doit jamais être improvisée.
\end{enumerate}
\end{problem}
