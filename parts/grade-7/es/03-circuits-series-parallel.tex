\chapter{Circuitos: bucles, serie y paralelo}\label{ch:g7:circuits-series-parallel}

Cuatro años de circuitos te han llenado la caja de herramientas:
bucles, esquemas y los dos grandes montajes. Pero ha habido un
personaje trabajando sin nombre todo este tiempo --- ese algo que
fluye. Este año recibe su nombre, su sentido y su propiedad más
extraña: no se gasta nunca. Con la corriente en escena, la serie y el
paralelo dejan de ser recetas y se convierten en leyes.

\section{La corriente sale a escena}

\begin{definition}[La corriente eléctrica]\label{def:g7:circuits-series-parallel:current}
La \emph{corriente eléctrica}\index{corriente eléctrica} es el flujo
que recorre un \omterm{def:g3:first-electric-circuit:circuit}{circuito cerrado}: una multitud de partículas
cargadas --- inimaginablemente numerosas, inimaginablemente pequeñas,
primas del mundo molecular --- puestas a desfilar por el metal de los
cables gracias al empuje de la \omterm{def:g3:first-electric-circuit:battery}{pila}. Sin bucle no hay desfile: la
corriente existe solo mientras la carretera está cerrada. (Qué son
exactamente los que desfilan, y cómo los deja pasar un metal, es una
historia estupenda reservada para los últimos años de este libro.)
\end{definition}

\begin{proposition}[El sentido convencional]\label{prop:g7:circuits-series-parallel:direction}
Por convenio mundial, los \omterm{def:g6:circuits-and-safety:diagram}{esquemas de circuito} marcan la corriente
como si fluyera \emph{por fuera de la \omterm{def:g3:first-electric-circuit:battery}{pila} desde el \omterm{def:g3:first-electric-circuit:battery}{borne} $+$,
alrededor del circuito, hasta el \omterm{def:g3:first-electric-circuit:battery}{borne} $-$}. Todas las flechas de
todos los esquemas del mundo obedecen este convenio --- elegido mucho
antes de que nadie pudiera preguntarles a los propios desfilantes.
(Cuando la física por fin pudo preguntar, la respuesta guardaba una
pequeña broma famosa, reservada para un año posterior.)
\end{proposition}

\begin{proposition}[La corriente no se consume]\label{prop:g7:circuits-series-parallel:notconsumed}
Los aparatos no se comen la corriente. Toda la corriente que entra en
una lámpara, un \omterm{ex:g6:circuits-and-safety:motor}{motor} o un \omterm{ex:g6:circuits-and-safety:motor}{zumbador} sale entera y sigue
desfilando: lo que el aparato toma es \emph{\omterm{def:g5:everyday-energy:energy}{energía}} del almacén de
la \omterm{def:g3:first-electric-circuit:battery}{pila}, entregada por el desfile que pasa. En cualquier bucle \omterm{def:g4:series-circuits:series}{en
serie} fluye, por lo tanto, una única y misma corriente a través de
todos los aparatos --- a la \omterm{def:g3:first-electric-circuit:battery}{pila} vuelve tanta como salió de ella.
\end{proposition}

\begin{example}[El error, en el banquillo]\label{ex:g7:circuits-series-parallel:trial}
El error tentador: ``la primera \omterm{def:g3:first-electric-circuit:bulb}{bombilla} gasta algo de corriente,
así que la segunda tiene que brillar menos''. El propio circuito lo
refuta. Monta dos \omterm{def:g3:first-electric-circuit:bulb}{bombillas} \emph{iguales} \omterm{def:g4:series-circuits:series}{en serie} y mira: las
dos brillan exactamente lo mismo --- e intercambiarlas no cambia nada.
Si la corriente se consumiera por el camino, la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} dos
debería ser siempre la floja, fuera cual fuera la \omterm{def:g3:first-electric-circuit:bulb}{bombilla}
colocada ahí. Nunca lo es. La corriente llega entera; lo que la pareja
se reparte es el \emph{empuje} de la \omterm{def:g3:first-electric-circuit:battery}{pila}, y por eso las dos brillan
menos que una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} sola --- y menos por igual.
\end{example}

\section{Las dos familias, como leyes}

\begin{proposition}[La ley de la serie]\label{prop:g7:circuits-series-parallel:series}
En un \omterm{def:g4:series-circuits:series}{circuito en serie} --- un único bucle --- fluye la misma
corriente por todos los aparatos: una carretera, un desfile, sin
excepciones. Consecuencias, ahora explicadas: un hueco para el desfile
en todas partes a la vez; los aparatos iguales se reparten el trabajo
por igual, estén donde estén; y el orden de los aparatos en el bucle
no puede importarle a un desfile que pasa por todos ellos igualmente.
\end{proposition}

\begin{proposition}[La ley del paralelo]\label{prop:g7:circuits-series-parallel:parallel}
En un \omterm{def:g5:series-parallel-circuits:parallel}{nudo}, la corriente \emph{se divide}: una parte del desfile
toma cada \omterm{def:g5:series-parallel-circuits:parallel}{rama}, y las partes se vuelven a juntar, completas, en el
\omterm{def:g5:series-parallel-circuits:parallel}{nudo} del otro extremo --- sin perder ni ganar nada. Cada
\omterm{def:g5:series-parallel-circuits:parallel}{rama} recibe el empuje completo de la \omterm{def:g3:first-electric-circuit:battery}{pila}, así que cada
aparato trabaja como si estuviera solo; y una \omterm{def:g5:series-parallel-circuits:parallel}{rama} bloqueada se
limita a no mandar a nadie, mientras las demás \omterm{def:g5:series-parallel-circuits:parallel}{ramas} siguen
desfilando.
\end{proposition}

\begin{example}[Leer el brillo como un electricista]\label{ex:g7:circuits-series-parallel:brightness}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} y \omterm{def:g3:first-electric-circuit:bulb}{bombillas} iguales. Sola: brillo completo --- la
referencia. Dos \omterm{def:g4:series-circuits:series}{en serie}: las dos igual de flojas --- un desfile,
empuje repartido. Dos \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo}: las dos a pleno brillo --- cada
\omterm{def:g5:series-parallel-circuits:parallel}{rama} disfruta del empuje entero (y la \omterm{def:g3:first-electric-circuit:battery}{pila} se vacía más o
menos el doble de rápido: nada es gratis). Tres \omterm{def:g4:series-circuits:series}{en serie}: más
flojas todavía, todas iguales. El vistazo del electricista: cuenta
compañeros de bucle para saber la flojera y cuenta \omterm{def:g5:series-parallel-circuits:parallel}{ramas} para
saber la factura de la \omterm{def:g3:first-electric-circuit:battery}{pila}.
\end{example}

\begin{method}[Predecir el destino de cualquier lámpara]\label{met:g7:circuits-series-parallel:predict}
Dado un esquema, para cada lámpara:
\begin{enumerate}
  \item \emph{Existencia}: ¿hay por lo menos un bucle sin cortes
        desde el $+$ que pase por esta lámpara hasta el $-$? Sin
        bucle no hay \omterm{def:g1:light-and-shadows:source}{luz}.
  \item \emph{Compañía en la carretera}: ¿qué aparatos comparten
        \emph{todos} los bucles que pasan por esta lámpara? Esos son
        sus compañeros de serie --- más compañeros, brillo más flojo.
  \item \emph{Vecinos de \omterm{def:g5:series-parallel-circuits:parallel}{rama}}: ¿qué aparatos están en otras
        \omterm{def:g5:series-parallel-circuits:parallel}{ramas}? Ni le quitan brillo a esta lámpara ni se apagan con
        ella.
  \item \emph{\omterm{ex:g3:first-electric-circuit:switch}{Interruptores}}: un \omterm{ex:g3:first-electric-circuit:switch}{interruptor} abierto mata
        exactamente los bucles en los que está --- repite el paso 1
        quitándolo.
\end{enumerate}
\end{method}

\begin{remark}[Por qué el error se resiste a morir]\label{rem:g7:circuits-series-parallel:whyhard}
``¡Algo tiene que gastarse --- la \omterm{def:g3:first-electric-circuit:battery}{pila} se vacía!''. Cierto: lo que
se gasta es la \emph{\omterm{def:g5:everyday-energy:energy}{energía}} almacenada de la \omterm{def:g3:first-electric-circuit:battery}{pila}, entregada
a las lámparas como \omterm{def:g1:light-and-shadows:source}{luz} y \omterm{def:g5:heat-and-insulation:heat}{calor} por el desfile que pasa. Pero los
desfilantes circulan intactos, como las sillas de un telesilla que
suben esquiadores todo el día sin que se consuma ni una silla. Mantén
separadas las dos contabilidades --- la corriente circula, la
\omterm{def:g5:everyday-energy:energy}{energía} se entrega --- y todos los circuitos de este libro seguirán
siendo honrados. El instrumento que cuenta el propio desfile se nos
une el año que viene.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:1}
¿Qué es la \omterm{def:g7:circuits-series-parallel:current}{corriente eléctrica}? ¿Cuándo existe en un circuito y
cuándo no?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:3}
Dos \omterm{def:g3:first-electric-circuit:bulb}{bombillas} iguales \omterm{def:g4:series-circuits:series}{en serie} brillan exactamente igual. ¿Qué

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:4}
En un bucle \omterm{def:g4:series-circuits:series}{en serie} con un \omterm{ex:g6:circuits-and-safety:motor}{motor} y una lámpara, se compara la
corriente que pasa por el \omterm{ex:g6:circuits-and-safety:motor}{motor} con la que pasa por la lámpara. ¿Qué

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:5}
¿Qué le pasa a la corriente en el primer \omterm{def:g5:series-parallel-circuits:parallel}{nudo} de un
\omterm{def:g5:series-parallel-circuits:parallel}{circuito en paralelo}? ¿Y en el segundo? ¿Qué se cumple con la
porción de empuje de la \omterm{def:g3:first-electric-circuit:battery}{pila} que recibe cada \omterm{def:g5:series-parallel-circuits:parallel}{rama}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:6}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:9}
Un esquema: \omterm{def:g3:first-electric-circuit:battery}{pila}, \omterm{ex:g3:first-electric-circuit:switch}{interruptor} y después la lámpara A;
después de A, un \omterm{def:g5:series-parallel-circuits:parallel}{nudo} lanza dos \omterm{def:g5:series-parallel-circuits:parallel}{ramas} --- la lámpara B en
una y la lámpara C en la otra ---, que se vuelven a juntar antes de
regresar a la \omterm{def:g3:first-electric-circuit:battery}{pila}. Usando el
\cref{met:g7:circuits-series-parallel:predict}: ¿quiénes son los
compañeros de serie de A? ¿Qué hace el brillo de A cuando se corta la
\omterm{def:g5:series-parallel-circuits:parallel}{rama} de C?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:10}
Unas luces decorativas piden un diseño: diez \omterm{def:g3:first-electric-circuit:bulb}{bombillas} que
brillen todas a pleno brillo y de manera independiente desde una sola
alimentación --- pero el conjunto entero tiene que obedecer a un único
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} general. Dibuja o describe el montaje, citando las
dos leyes.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:11}
En el \omterm{def:g5:series-parallel-circuits:parallel}{circuito en paralelo} de dos \omterm{def:g3:first-electric-circuit:bulb}{bombillas}, la \omterm{def:g5:series-parallel-circuits:parallel}{rama} de
la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 1 es cable grueso de cobre mientras que la \omterm{def:g5:series-parallel-circuits:parallel}{rama} de
la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2 incluye un cable largo y fino. La \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2
brilla un poco menos. ¿Qué sugiere esto sobre cómo la propia carretera
de una \omterm{def:g5:series-parallel-circuits:parallel}{rama} puede estrangular su porción del desfile? (El año que
viene le pone nombre y ley a ese estrangulamiento.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:circuits-series-parallel:12}
Una \omterm{def:g3:first-electric-circuit:battery}{pila} alimenta dos \omterm{def:g5:series-parallel-circuits:parallel}{ramas} paralelas: la \omterm{def:g5:series-parallel-circuits:parallel}{rama} uno
lleva la lámpara X sola; la \omterm{def:g5:series-parallel-circuits:parallel}{rama} dos lleva las lámparas Y y Z
\omterm{def:g4:series-circuits:series}{en serie}. Compara por orden los brillos de las tres lámparas, con
razones sacadas de las dos leyes. Predice después el destino de cada
lámpara si se funde Z --- y si, en cambio, se funde X.
\end{exercise}

\section{Problema: el cuadro del farero}

\begin{problem}[{Problema de fin de semana --- reinstalar el viejo
faro; un banco de pilas y muchas obligaciones; la noche de averías del
farero}]\label{pb:g7:circuits-series-parallel:1}
El banco de \omterm{def:g3:first-electric-circuit:battery}{pilas} del faro tiene que alimentar: la gran lámpara,
una sirena de niebla (un \omterm{ex:g6:circuits-and-safety:motor}{zumbador} tremendo), la lámpara de lectura del
farero y la \omterm{def:g1:light-and-shadows:source}{luz} de la escalera --- y sobrevivir a la noche de
averías del farero.

\medskip\noindent\textbf{Parte I --- El diseño.}
\begin{enumerate}
  \item La gran lámpara no puede perder brillo nunca porque se
        encienda otra cosa. ¿Qué montaje exige esto para la
        instalación entera y por qué traicionaría a los barcos
        cualquier emparejamiento \omterm{def:g4:series-circuits:series}{en serie}?
  \item Cada obligación necesita su propio \omterm{ex:g3:first-electric-circuit:switch}{interruptor}, y el farero
        quiere un corte general para las tormentas. Coloca los cinco
        \omterm{ex:g3:first-electric-circuit:switch}{interruptores}.
  \item La sirena de niebla y la \omterm{def:g1:light-and-shadows:source}{luz} de la escalera no deben
        impedirse nunca funcionar la una a la otra por sus averías.
        ¿Está eso ya garantizado? ¿Por qué ley?
  \item Dibuja o describe el esquema completo: banco,
        \omterm{ex:g3:first-electric-circuit:switch}{interruptor} general, cuatro \omterm{def:g5:series-parallel-circuits:parallel}{ramas} y cuatro
        \omterm{ex:g3:first-electric-circuit:switch}{interruptores} de \omterm{def:g5:series-parallel-circuits:parallel}{rama}.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Las preguntas del farero.}
\begin{enumerate}[resume]
  \item El farero se preocupa: ``Cuatro obligaciones bebiendo a la
        vez --- ¿sigue recibiendo cada \omterm{def:g5:series-parallel-circuits:parallel}{rama} el empuje completo del
        banco?''. Responde con la ley del paralelo.
  \item ``¿Y se está gastando la corriente mientras trabaja a través
        de la gran lámpara?''. Saca de su error al farero, con el
        telesilla si te ayuda.
  \item Una noche de niebla funciona todo a la vez. ¿Cuál es el
        precio de la generosidad del paralelo y en qué parte del faro
        se paga?
  \item Está encendida solo la lámpara de lectura; con espíritu de
        medidor, el farero sigue el desfile: describe su viaje
        completo de ida y vuelta, nombrando todos los componentes por
        los que pasa y todos los que evita.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La noche de las averías.}
\begin{enumerate}[resume]
  \item Medianoche: se funde la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} de la escalera. ¿Qué más se
        apaga? ¿Qué ley responde?
  \item La una: una reparación chapucera deja un tramo de cable
        desnudo uniendo directamente los dos \omterm{def:g3:first-electric-circuit:battery}{bornes} del banco,
        aguas arriba de todas las \omterm{def:g5:series-parallel-circuits:parallel}{ramas}. Nombra el suceso y
        explica por qué fallan ahora de golpe todas las obligaciones
        del faro --- ¿adónde se va el desfile?
  \item Las dos, con la avería resuelta: el farero quiere proteger la
        gran lámpara para que un atajo así no pueda volver a
        apagarla. Hay disponible un segundo banco de \omterm{def:g3:first-electric-circuit:battery}{pilas}
        independiente. Propón el montaje.
  \item Amanece: escribe el cuaderno de bitácora del farero --- tres
        frases: qué ley mantuvo a salvo a los barcos esta noche, qué
        error refutó la noche y qué único cable causó la peor hora.
\end{enumerate}
\end{problem}
