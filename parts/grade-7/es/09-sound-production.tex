\chapter{El sonido: producción y propagación}\label{ch:g7:sound-production}

Sabes desde la infancia que el \omterm{def:g3:sound-around-us:sound}{sonido} nace en un temblor y muere en el
vacío --- los granos de arroz bailaron, la campana del frasco se
calló. Lo que te faltaba era el nombre del mensajero. Ahora el mundo
molecular es tuyo, y el viaje entero del \omterm{def:g3:sound-around-us:sound}{sonido} --- de una cuerda de
guitarra a tu tímpano, por el \omterm{def:g2:air-around-us:air}{aire}, por el agua o por el acero ---
puede por fin mirarse desde dentro.

\section{Los mensajeros}

\begin{definition}[Medio]\label{def:g7:sound-production:medium}
Un \emph{medio}\index{medio} es la sustancia por la que viaja un
\omterm{def:g3:sound-around-us:sound}{sonido}: \omterm{def:g2:air-around-us:air}{aire}, agua, madera, acero --- cualquier estado sirve,
porque todos son multitudes de \omterm{def:g7:states-of-matter:molecule}{moléculas}. Lo que no puede cruzar
ningún \omterm{def:g3:sound-around-us:sound}{sonido} es la ausencia de multitud: el vacío, el escenario
desierto.
\end{definition}

\begin{proposition}[Cómo viaja el sonido]\label{prop:g7:sound-production:travel}
Una fuente que vibra empuja a las \omterm{def:g7:states-of-matter:molecule}{moléculas} que tiene al lado; estas
se apretujan contra sus vecinas y rebotan; las vecinas empujan a su
vez a la fila siguiente. El empujón --- un apretujamiento viajero de
la multitud --- sale disparado de la fuente en todas las direcciones,
con una \omterm{def:g7:states-of-matter:molecule}{molécula} relevando a otra, aunque cada \omterm{def:g7:states-of-matter:molecule}{molécula} en sí
solo se agite en torno a su sitio. El \omterm{def:g3:sound-around-us:sound}{sonido} es el \emph{relevo}, no
los corredores: el mensaje viaja; los mensajeros se quedan en casa.
\end{proposition}

\begin{example}[El frasco, explicado por fin]\label{ex:g7:sound-production:jar}
La famosa campana silenciada confiesa por fin su mecanismo. Sacar el
\omterm{def:g2:air-around-us:air}{aire} retira la multitud: las paredes temblorosas de la campana
empujan a la nada, no se forma ningún relevo y no cruza el vidrio
ningún mensaje. La campana nunca dejó de sonar --- perdió a sus
mensajeros. Y de ahí sale al instante el gran silencio del espacio:
entre las \omterm{def:g4:solar-system:star}{estrellas} no hay multitud que lleve un grito.
\end{example}

\begin{example}[Fuerte, flojo y gastado]\label{ex:g7:sound-production:loud}
Un empujón más fuerte produce un apretujamiento más intenso:
\omterm{def:g3:sound-around-us:sound}{sonido} más fuerte. Y, a medida que el relevo se extiende ---
círculos de estanque en tres dimensiones ---, cada apretujamiento se
reparte por una cáscara de multitud cada vez más vasta y se debilita
con la distancia: por eso se apagan los gritos y por eso la música del
vecino es un murmullo a través de dos paredes. Nada de esto ha
necesitado leyes nuevas: la física de multitudes lo paga todo.
\end{example}

\section{Cada medio tiene su paso}

\begin{proposition}[La velocidad del sonido depende del medio]\label{prop:g7:sound-production:speeds}
La \omterm{def:g7:motion-average-speed:formula}{velocidad} del relevo es una propiedad de la multitud:
\begin{enumerate}
  \item en el \omterm{def:g2:air-around-us:air}{aire}: unos \qty{340}{m/s};
  \item en el agua: unos \qty{1500}{m/s} --- cuatro veces más
        rápido;
  \item en el acero: unos \qty{5000}{m/s} --- quince veces más
        rápido que en el \omterm{def:g2:air-around-us:air}{aire}.
\end{enumerate}
El patrón: cuanto más apretadas se tocan las \omterm{def:g7:states-of-matter:molecule}{moléculas} de un
\omterm{def:g7:sound-production:medium}{medio}, más rápido se pasan el empujón. Las \omterm{def:g7:states-of-matter:molecule}{moléculas} voladoras del
\omterm{def:g2:air-around-us:air}{aire} tienen que cruzar antes unos huecos para entregarlo; la
multitud del agua, que se toca, releva con viveza; las filas trabadas
del acero se pasan el mensaje casi de mano en mano.
\end{proposition}

\begin{example}[Escenas viejas, números nuevos]\label{ex:g7:sound-production:scenes}
La oreja del explorador sobre el raíl: el relevo del acero le saca al
del \omterm{def:g2:air-around-us:air}{aire} un factor de quince --- el aviso del raíl llega mucho antes
que el estruendo. Los cantos de las ballenas cruzando mares: el relevo
veloz y generoso del agua. Y tu propia voz te suena rara en las
grabaciones en parte porque, al hablar, te oyes también por el relevo
óseo del cráneo además de por el \omterm{def:g2:air-around-us:air}{aire} --- dos \omterm{def:g7:sound-production:medium}{medios}, dos versiones,
y los demás solo oyen la de \omterm{def:g2:air-around-us:air}{aire}.
\end{example}

\section{El sonido con cronómetro}

\begin{method}[Medir la velocidad del sonido]\label{met:g7:sound-production:measure}
El método más antiguo sigue funcionando en cualquier campo de
deportes:
\begin{enumerate}
  \item un compañero se coloca a \qty{680}{m} medidos (dos largos de
        campo y pico --- mídelo a pasos o usa las marcas del campo)
        con dos tapas de cazuela;
  \item las choca por encima de la cabeza: tú \emph{ves} el golpe al
        instante (el viaje de la \omterm{def:g1:light-and-shadows:source}{luz} es como si fuera
        instantáneo) y arrancas el cronómetro con el destello del
        movimiento;
  \item páralo con el clanc que llega: espera casi exactamente
        \qty{2}{s};
  \item calcula: $v = d/t = 680 \div 2 = \qty{340}{m/s}$.
\end{enumerate}
Repite y promedia --- los tiempos de reacción se dispersan, como
enseñaba el capítulo de la medida honrada. Contar la tormenta es este
método al revés: sabiendo $v$, los segundos de silencio te entregan la
distancia.
\end{method}

\begin{example}[El eco, calculado]\label{ex:g7:sound-production:echo}
Una palmada hacia un acantilado vuelve en \qty{3}{s}. El
\omterm{def:g3:sound-around-us:sound}{sonido} viajó hasta el acantilado y de vuelta: $d_{\text{ida y
vuelta}} = 340 \times 3 = \qty{1020}{m}$, así que el acantilado está a
la mitad: \qty{510}{m}. La regla de toda cuenta con \omterm{ex:g4:how-sound-travels:echo}{eco}: el tiempo
medido paga el viaje de ida y vuelta --- olvidar la ``vuelta'' es el
desliz clásico, y duplica mal todas las respuestas.
\end{example}

\begin{example}[El sonar, en serio]\label{ex:g7:sound-production:sonar}
El sonar del barco de prospección hace clic hacia abajo; el
\omterm{ex:g4:how-sound-travels:echo}{eco} del fondo vuelve en \qty{0.6}{s}. En el \emph{agua}, $d = 1500
\times 0.6 = \qty{900}{m}$ de ida y vuelta: profundidad \qty{450}{m}.
Los delfines y los murciélagos hacen la misma aritmética por instinto;
los barcos la imprimen en las cartas náuticas. Fíjate en la
disciplina: el \omterm{def:g7:sound-production:medium}{medio} elige la \omterm{def:g7:motion-average-speed:formula}{velocidad} --- darle a un
\omterm{ex:g4:how-sound-travels:echo}{eco} submarino los $340$ del \omterm{def:g2:air-around-us:air}{aire} es el segundo desliz clásico
del capítulo.
\end{example}

\begin{remark}[Lo que sigue sin ser ``rápido'']\label{rem:g7:sound-production:light}
Los \qty{5000}{m/s} del acero suenan heroicos --- hasta que uno se
acuerda de la \omterm{def:g1:light-and-shadows:source}{luz}. En la tormenta, el viaje del relámpago cuenta como
instantáneo mientras el trueno trota su kilómetro en tres segundos; y
ningún \omterm{def:g7:sound-production:medium}{medio}, ni el acero ni el diamante, acerca el \omterm{def:g3:sound-around-us:sound}{sonido} ni
de lejos al paso de la \omterm{def:g1:light-and-shadows:source}{luz}. A qué \omterm{def:g7:motion-average-speed:formula}{velocidad} corre exactamente la
\omterm{def:g1:light-and-shadows:source}{luz} --- y cómo los humanos consiguieron por fin cronometrar algo que
cruza una habitación en una cienmillonésima de segundo --- es la
historia del año que viene, con capítulo propio.
\end{remark}

\section{Problema: la prospección del cañón}

\begin{problem}[{Problema de fin de semana --- cartografiar de oído el
cañón silencioso; ecos, raíles y profundidades de río; el maletín del
topógrafo}]\label{pb:g7:sound-production:1}
Un equipo de topografía cartografía un cañón remoto con cronómetros,
un equipo de sonar y la física de este capítulo.
\omterm{def:g7:motion-average-speed:formula}{Velocidades}: \omterm{def:g2:air-around-us:air}{aire} \qty{340}{m/s}, agua \qty{1500}{m/s}, acero
\qty{5000}{m/s}.

\medskip\noindent\textbf{Parte I --- Anchuras por \omterm{ex:g4:how-sound-travels:echo}{eco}.}
\begin{enumerate}
  \item Desde el borde este, el \omterm{ex:g4:how-sound-travels:echo}{eco} de una palmada en la pared
        oeste vuelve en \qty{4}{s}. ¿Cuánto mide de ancho el cañón
        aquí?
  \item En un punto más estrecho, el \omterm{ex:g4:how-sound-travels:echo}{eco} vuelve en \qty{1.5}{s}.
        ¿La anchura?
  \item Un topógrafo colocado \emph{entre} dos paredes paralelas da
        una palmada y oye dos \omterm{ex:g4:how-sound-travels:echo}{ecos} distintos: a \qty{1}{s} y a
        \qty{2}{s}. ¿A qué distancia está cada pared y cuánto mide
        de ancho el cañón en esta línea?
  \item ¿Por qué el mayor error de medida de la Parte I viene de la
        mano del cronómetro y no de la fórmula? (Cita la sabiduría
        de la regla de las medidas repetidas.)
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El río de abajo.}
\begin{enumerate}[resume]
  \item El equipo de sonar, a flote en el río, lanza un pulso al
        fondo: \omterm{ex:g4:how-sound-travels:echo}{eco} en \qty{0.2}{s}. ¿La profundidad del río?
  \item El mismo equipo, apuntado de lado hacia un peñasco
        sumergido, devuelve un \omterm{ex:g4:how-sound-travels:echo}{eco} en \qty{0.4}{s}. ¿Distancia al
        peñasco?
  \item Un compañero propone ahorrar \omterm{def:g3:first-electric-circuit:battery}{pilas}: ``gritamos desde la
        barca y cronometramos de oído el \omterm{ex:g4:how-sound-travels:echo}{eco} del fondo''. Dos
        razones físicas por las que esto falla donde el sonar
        triunfa. (Piensa en la frontera entre \omterm{def:g7:sound-production:medium}{medios} y en lo que
        tiene que hacer en la superficie el relevo aéreo de un
        grito.)
  \item El lecho del río cae en picado a media corriente: predice
        qué hace el tiempo de \omterm{ex:g4:how-sound-travels:echo}{eco} del sonar mientras la barca
        deriva por encima del escalón.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El viejo raíl minero.}
Un raíl abandonado y recto recorre \qty{2}{km} por el fondo del cañón
hasta la boca de la mina.
\begin{enumerate}[resume]
  \item Un martillazo en el raíl junto a la boca: calcula los dos
        tiempos de llegada al extremo de los topógrafos --- por el
        acero y por el \omterm{def:g2:air-around-us:air}{aire}.
  \item El equipo oye el clanc del raíl y el bum del \omterm{def:g2:air-around-us:air}{aire} separados
        por ese intervalo. Explica cómo \emph{el intervalo solo}
        podría \omterm{def:g6:measurement-in-science:measuring}{medir} la distancia a la boca si los \qty{2}{km}
        fueran desconocidos (describe el razonamiento; el cálculo es
        el regalo de la Parte III a los aplicados).
  \item La niebla llena el cañón durante la noche --- inútil para
        lámparas y \omterm{def:g5:mirrors-reflection:reflection}{espejos}. ¿Cuál de los tres canales de medida del
        equipo (\omterm{def:g3:sound-around-us:sound}{sonido} por el \omterm{def:g2:air-around-us:air}{aire}, \omterm{def:g3:sound-around-us:sound}{sonido} por el agua,
        \omterm{def:g3:sound-around-us:sound}{sonido} por el raíl) perturba menos la niebla, y por qué?
  \item Escribe el diario del topógrafo: tres frases --- la regla
        del viaje de ida y vuelta, la regla de que el \omterm{def:g7:sound-production:medium}{medio} elige la
        \omterm{def:g7:motion-average-speed:formula}{velocidad}, y qué único instrumento (ojo u oído) arrancó
        todos los cronometrajes del cañón y por qué puede fiarse uno
        de él como ``instantáneo''.
\end{enumerate}
\end{problem}
