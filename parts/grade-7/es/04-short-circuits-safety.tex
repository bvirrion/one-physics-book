\chapter{Cortocircuitos y seguridad eléctrica}\label{ch:g7:short-circuits-safety}

Todas las reglas de seguridad que has aprendido se remontan a un
villano que conociste brevemente el año pasado: el
\omterm{def:g7:short-circuits-safety:device}{cortocircuito}. Este capítulo lo estudia como es debido --- cómo
empieza, por qué quema y cómo lo derrota un humilde cable de
sacrificio --- y convierte el cuadro de \omterm{def:g7:short-circuits-safety:fuse}{fusibles} de la familia de
armario misterioso en un aliado al que entiendes.

\section{El método del villano}

\begin{proposition}[La corriente prefiere la carretera sin esfuerzo]\label{prop:g7:short-circuits-safety:effortless}
Entre dos carreteras que unen los mismos dos puntos, la corriente que
desfila se apiña en la que menos se le opone. Los aparatos ---
lámparas, \omterm{ex:g6:circuits-and-safety:motor}{motores}, \omterm{ex:g6:circuits-and-safety:motor}{zumbadores} --- se oponen todos al desfile
mientras trabajan; un simple cable grueso no se le opone casi nada.
Ofrécele al desfile un desvío de cable desnudo y abandonará casi por
completo la carretera que trabaja por la que no cuesta esfuerzo.
\end{proposition}

\begin{definition}[Cortocircuito de un aparato]\label{def:g7:short-circuits-safety:device}
Un aparato está \emph{cortocircuitado}\index{cortocircuito} cuando una
carretera conductora une directamente sus dos \omterm{def:g3:first-electric-circuit:battery}{bornes} y lo esquiva.
El desfile abandona el aparato --- una lámpara cortocircuitada se
apaga, un \omterm{ex:g6:circuits-and-safety:motor}{motor} se para --- mientras el resto del bucle sigue
adelante, ahora con un trabajador menos oponiéndose al desfile.
\end{definition}

\begin{definition}[Cortocircuito de la pila]\label{def:g7:short-circuits-safety:battery}
El caso más grave: una carretera conductora une los dos \omterm{def:g3:first-electric-circuit:battery}{bornes}
\emph{de la \omterm{def:g3:first-electric-circuit:battery}{pila}} sin ningún aparato por el camino. Nada se opone al
desfile y este se convierte en una estampida --- la avalancha
desbocada calienta el cable y la propia \omterm{def:g3:first-electric-circuit:battery}{pila}, arruina las
\omterm{def:g3:first-electric-circuit:battery}{pilas}, funde el aislamiento y, en la red de la casa, incendia
viviendas.
\end{definition}

\begin{method}[El villano, demostrado sin peligro]\label{met:g7:short-circuits-safety:demo}
Con una \omterm{def:g3:first-electric-circuit:battery}{pila} y dos \omterm{def:g3:first-electric-circuit:bulb}{bombillas} \omterm{def:g4:series-circuits:series}{en serie} (sin peligro: el
empuje de la \omterm{def:g3:first-electric-circuit:battery}{pila} es diminuto):
\begin{enumerate}
  \item enciende las dos \omterm{def:g3:first-electric-circuit:bulb}{bombillas} --- igual de flojas, como
        promete la ley de la serie;
  \item toca ahora con un cable normal los dos \omterm{def:g3:first-electric-circuit:battery}{bornes} de la
        \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2, esquivándola;
  \item mira: la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2 muere al instante --- abandonada ---
        mientras la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 1 salta a \emph{más brillo}: con un
        oponente esquivado, el desfile del bucle entero se refuerza;
  \item quita el cable: el desfile vuelve a entrar en la
        \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2 y regresa la flojera igualada.
\end{enumerate}
Un solo experimento, tres lecciones: la regla del desvío, el aparato
abandonado y la avalancha reforzada.
\end{method}

\begin{example}[Por qué se calientan los cables]\label{ex:g7:short-circuits-safety:heating}
Una corriente que desfila calienta siempre un poco su carretera --- el
resplandor de la lámpara es ese calentamiento, puesto a trabajar a
propósito en un hilo construido para ponerse al rojo blanco. En una
estampida, el calentamiento se vuelve amenaza: la avalancha de un
\omterm{def:g7:short-circuits-safety:device}{cortocircuito} puede calentar un cable de cobre por encima del
\omterm{def:g4:changes-of-state:melting-point}{punto de fusión} de su funda de plástico en cuestión de segundos. El
avance a escala de bolsillo: una \omterm{def:g3:first-electric-circuit:battery}{pila} cortocircuitada por un cable
se calienta en la mano --- la misma física que, a la escala de la red,
tizna paredes. (Cómo crece el calentamiento con la avalancha se mide
con honradez dentro de dos años, con el capítulo de la potencia
eléctrica.)
\end{example}

\section{Los guardaespaldas}

\begin{definition}[Fusible]\label{def:g7:short-circuits-safety:fuse}
Un \emph{fusible}\index{fusible} es un guardaespaldas que muere por
ti: un cable deliberadamente fino, \omterm{def:g4:series-circuits:series}{en serie} con el circuito que
protege, elegido para fundirse --- y cortar así el bucle --- cuando la
corriente supera su valor nominal. La estampida que chamuscaría
\omterm{def:g2:measuring-length:units}{metros} de instalación doméstica destruye en su lugar tres
\omterm{def:g2:measuring-length:units}{centímetros} de cable de sacrificio dentro de su cartucho
ignífugo. Un fusible fundido se sustituye --- \emph{después} de
encontrar la avería --- por otro nuevo del mismo valor.
\end{definition}

\begin{example}[Magnetotérmico y fusible, dos guardias con un mismo deber]\label{ex:g7:short-circuits-safety:breaker}
El magnetotérmico moderno cumple el deber del \omterm{def:g7:short-circuits-safety:fuse}{fusible} sin morir: un
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} con muelle que salta abierto ante la avalancha y se
rearma con un gesto --- después de curar la avería. \omterm{def:g7:short-circuits-safety:fuse}{Fusible} o
magnetotérmico, el puesto del guardia es el mismo: \emph{\omterm{def:g4:series-circuits:series}{en
serie}}, en la \omterm{def:g5:series-parallel-circuits:parallel}{rama} que protege, para que la estampida tenga que
pasar por su control. Un guardia \omterm{def:g5:series-parallel-circuits:parallel}{en paralelo} vería tronar la
avalancha por la otra carretera --- pura decoración.
\end{example}

\begin{example}[La sobrecarga: la estampida educada]\label{ex:g7:short-circuits-safety:overload}
No todas las avalanchas peligrosas son \omterm{def:g7:short-circuits-safety:device}{cortocircuitos}. Una regleta
que alimenta a la vez un radiador, un hervidor y una tostadora le pide
a su única \omterm{def:g5:series-parallel-circuits:parallel}{rama} que cargue con tres desfiles completos juntos ---
cada aparato es inocente, pero su suma es una sobrecarga que calienta
la instalación de la pared como una tostadora lenta. El
magnetotérmico salta por la suma, y con razón. Regla de la casa: un
solo aparato hambriento que dé \omterm{def:g5:heat-and-insulation:heat}{calor} por enchufe; la regleta es para
lámparas y cargadores, no para una cocina particular.
\end{example}

\begin{remark}[La reparación prohibida]\label{rem:g7:short-circuits-safety:nail}
La vieja historia es verdadera y sigue matando: en una casa salta una
y otra vez el \omterm{def:g7:short-circuits-safety:fuse}{fusible} y alguien lo ``repara'' con un clavo o una
moneda --- un \omterm{def:g4:conductors-insulators:conductor}{conductor} robusto que no se fundirá jamás. Se calla
el síntoma; la enfermedad sigue haciendo estragos. La siguiente
estampida no encuentra ningún control, y los propios cables de la casa
se convierten en el \omterm{def:g7:short-circuits-safety:fuse}{fusible} --- dentro de las paredes. No puentees
nunca a un guardia: un \omterm{def:g7:short-circuits-safety:fuse}{fusible} que se funde una y otra vez es un
guardia que dice la verdad una y otra vez.
\end{remark}

\begin{remark}[Las reglas, en su forma definitiva]\label{rem:g7:short-circuits-safety:rules}
El código eléctrico de la familia, ya con todas sus razones: no unir
nunca los \omterm{def:g3:first-electric-circuit:battery}{bornes} de una \omterm{def:g3:first-electric-circuit:battery}{pila} y mantener las llaves lejos de las
baterías de coche (una batería cortocircuitada con metal es un arco de
soldadura); un solo aparato hambriento de \omterm{def:g5:heat-and-insulation:heat}{calor} por enchufe
(sobrecarga); los cables dañados, jubilados (una funda deshilachada es
un \omterm{def:g7:short-circuits-safety:device}{cortocircuito} ensayando); el agua y la red, separadas, y las
manos siempre secas (tu cuerpo no debe ser nunca el desvío); los
guardias, respetados --- solo \omterm{def:g7:short-circuits-safety:fuse}{fusibles} equivalentes y averías
encontradas antes de rearmar ---; y el enchufe, para siempre, ni para
dedos ni para experimentos.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:1}
Enuncia la proposición de la carretera sin esfuerzo. ¿Por qué le gana
un cable desnudo el desfile a una lámpara?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:2}
Distingue el \omterm{def:g7:short-circuits-safety:device}{cortocircuito} de un aparato del
\omterm{def:g7:short-circuits-safety:device}{cortocircuito} de la \omterm{def:g3:first-electric-circuit:battery}{pila}. ¿Cuál es más grave y por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:3}
En la demostración sin peligro, ¿por qué muere la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2 --- y
por qué gana brillo la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 1 en ese mismo instante?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:4}
¿Por qué calienta su cable un \omterm{def:g7:short-circuits-safety:device}{cortocircuito}? ¿Qué aparato de
cada día usa ese mismo calentamiento a propósito?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:5}
¿Qué es un \omterm{def:g7:short-circuits-safety:fuse}{fusible} y qué sacrifica? ¿Por qué tiene que estar
\omterm{def:g4:series-circuits:series}{en serie} con lo que protege?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:6}
¿En qué mejora un magnetotérmico al \omterm{def:g7:short-circuits-safety:fuse}{fusible}? ¿Qué tiene que pasar
antes de rearmar o de sustituir a cualquiera de los dos guardias?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:7}
Un radiador, un hervidor y una tostadora comparten una regleta. No hay
ningún cable desnudo ni ningún aparato averiado --- y, aun así, salta
el magnetotérmico. Nombra el suceso y explica la queja de la
\omterm{def:g5:series-parallel-circuits:parallel}{rama}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:8}
¿Por qué es ``reparar'' un \omterm{def:g7:short-circuits-safety:fuse}{fusible} con un clavo uno de los actos más
peligrosos de una casa? ¿Qué se convierte después en el
\omterm{def:g7:short-circuits-safety:fuse}{fusible}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:9}
Los dos hilos interiores de un cable, con las fundas deshilachadas, se
tocan dentro de la manguera --- aguas arriba de la lámpara que
alimentan. Diagnostica: ¿qué clase de \omterm{def:g7:short-circuits-safety:device}{cortocircuito} es, qué le pasa
a la lámpara y qué hace el guardia de la \omterm{def:g5:series-parallel-circuits:parallel}{rama}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:10}
En la demostración de dos \omterm{def:g3:first-electric-circuit:bulb}{bombillas}, predice qué pasa si el cable
del desvío se coloca en cambio sobre \emph{las dos} \omterm{def:g3:first-electric-circuit:bulb}{bombillas}
juntas --- de antes de la \omterm{def:g3:first-electric-circuit:bulb}{bombilla} 1 a después de la
\omterm{def:g3:first-electric-circuit:bulb}{bombilla} 2. ¿Qué definición se aplica ahora y por qué no debe
probarlo quien experimenta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:11}
Misterio de la mañana de Navidad: la guirnalda \omterm{def:g4:series-circuits:series}{en serie} de la abuela
tiene una \omterm{def:g3:first-electric-circuit:bulb}{bombilla} cuyos hilos de soporte interiores se han soldado
entre sí al morir --- cortocircuitándose a sí misma. Al contrario de
lo que suele ser una tragedia, \emph{el resto de la guirnalda sigue
brillando}, y un poco más. Explícalo --- y di qué se está preparando
ahora despacio para las \omterm{def:g3:first-electric-circuit:bulb}{bombillas} que quedan.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:short-circuits-safety:12}
Diseña el plan de protección de un pequeño taller: tres \omterm{def:g5:series-parallel-circuits:parallel}{ramas}
paralelas (luces; herramientas eléctricas; un cargador de
baterías) y una sola alimentación. Coloca guardias e
\omterm{ex:g3:first-electric-circuit:switch}{interruptores}; elige qué \omterm{def:g5:series-parallel-circuits:parallel}{ramas} merecen su propio guardia y por
qué; y escribe el aviso de dos líneas que irá encima del cajón de los
\omterm{def:g7:short-circuits-safety:fuse}{fusibles} y que resuma la
\cref{rem:g7:short-circuits-safety:nail}.
\end{exercise}

\section{Problema: el informe de la investigadora de incendios}

\begin{problem}[{Problema de fin de semana --- después del incendio
del piso; la investigadora lee la confesión de la instalación; tres
averías, una lección}]\label{pb:g7:short-circuits-safety:1}
No hubo heridos --- la familia estaba fuera ---, pero la pared de la
cocina está chamuscada, y la investigadora tiene que reconstruir la
noche a partir de las pruebas. Tú llevas su cuaderno.

\medskip\noindent\textbf{Parte I --- Leer las ruinas.}
\begin{enumerate}
  \item Prueba número uno: una regleta que alimentaba un radiador, un
        hervidor y una freidora, con el plástico medio derretido. Sin
        ningún cable desnudo por ninguna parte. Nombra la avería que
        sugiere esta prueba y di por qué no hace falta ningún ``cable
        villano'' para ella.
  \item Prueba número dos: un cable detrás del armario, mordido por
        una mascota, con sus dos \omterm{def:g4:conductors-insulators:conductor}{conductores} tocándose en el
        mordisco. Nombra esta avería y di adónde se fue el desfile a
        partir del momento del contacto.
  \item Prueba número tres, la incriminatoria: en el cuadro de
        \omterm{def:g7:short-circuits-safety:fuse}{fusibles}, el cartucho del \omterm{def:g7:short-circuits-safety:fuse}{fusible} de la \omterm{def:g5:series-parallel-circuits:parallel}{rama} de la
        cocina contiene --- una tira enrollada de papel de aluminio.
        Di qué le hizo esa ``reparación'' a la protección de la
        \omterm{def:g5:series-parallel-circuits:parallel}{rama}.
  \item Ordena las tres pruebas en una historia verosímil: ¿qué
        avería llegó primero, qué hizo el \omterm{def:g7:short-circuits-safety:fuse}{fusible} de verdad con
        ella (antes de la prueba número tres) y por qué acabó la
        última noche en incendio en vez de en una cocina a oscuras?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- La declaración de la física.}
\begin{enumerate}[resume]
  \item El tribunal pregunta: ¿por qué \emph{se calienta} un cable
        \omterm{def:g7:short-circuits-safety:device}{cortocircuitado} o sobrecargado? Di qué aparato de cada día
        demuestra que el calentamiento por corriente se puede
        domesticar --- y qué faltaba aquí.
  \item ¿Por qué es fino a propósito un \omterm{def:g7:short-circuits-safety:fuse}{fusible}? Explica la frase
        ``un guardaespaldas que muere por ti''.
  \item La tira de aluminio conducía de maravilla. Y precisamente
        \emph{por eso} fracasó como guardia. Resuelve la pequeña
        paradoja para el jurado.
  \item El radiador solo era seguro, el hervidor solo era seguro y la
        freidora sola era segura. Escribe una frase sobre cómo tres
        seguridades sumaron un solo peligro.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El veredicto y las lecciones.}
\begin{enumerate}[resume]
  \item La investigadora tiene que ordenar las tres averías por su
        \omterm{def:g4:weight-and-mass:weight}{peso} moral: la sobrecarga, el cable mordido y el aluminio.
        ¿Cuál convirtió el percance en catástrofe y por qué?
  \item La familia pregunta qué habría pasado la noche del incendio
        si hubiera habido un \omterm{def:g7:short-circuits-safety:fuse}{fusible} en condiciones. Guíalos por
        ello, minuto a minuto.
  \item Recomienda la nueva instalación: ¿cómo hay que alimentar
        ahora el radiador, el hervidor y la freidora, y qué regla
        permanente gobierna el futuro de la regleta?
  \item Cierra el informe con el resumen de tres frases de la
        investigadora: el método del villano, el deber del guardia y
        la única reparación que no se puede improvisar jamás.
\end{enumerate}
\end{problem}
