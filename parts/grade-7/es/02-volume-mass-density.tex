\chapter{Volumen, masa y densidad}\label{ch:g7:volume-mass-density}

El año pasado desenmascaraste a un orfebre fraudulento dividiendo
\omterm{def:g3:measuring-mass:units}{gramos} entre \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}. Este año, tu curso de
matemáticas te ha entregado un poder nuevo: letras que representan
números. La \omterm{def:g7:volume-mass-density:formula}{densidad} es el sitio perfecto para gastarlo --- una
fórmula corta que empaqueta tres recetas, un maletín de detective y la
regla entera de la flotación en cinco símbolos.

\section{La fórmula}

\begin{definition}[La densidad, en fórmula]\label{def:g7:volume-mass-density:formula}
La \emph{densidad}\index{densidad} $\rho$ (la letra griega
\emph{ro}) de una sustancia es su \omterm{def:g3:measuring-mass:mass}{masa} por \omterm{def:g6:measurement-in-science:measuring}{unidad} de
\omterm{def:g6:mass-and-volume:volume}{volumen}: para una muestra de \omterm{def:g3:measuring-mass:mass}{masa} $m$ y \omterm{def:g6:mass-and-volume:volume}{volumen} $V$,
\[
\rho = \frac{m}{V}.
\]
Con $m$ en \omterm{def:g3:measuring-mass:units}{gramos} y $V$ en \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}, $\rho$ sale
en \emph{\omterm{def:g3:measuring-mass:units}{gramos} por \omterm{def:g6:mass-and-volume:volume}{centímetro cúbico}} (\unit{g/cm^3}) ---
la ``\omterm{def:g3:measuring-mass:mass}{masa} de un \omterm{def:g6:mass-and-volume:volume}{centímetro cúbico}'' del año pasado, ahora de
uniforme. La densidad del agua: $\rho = \qty{1.0}{g/cm^3}$. Y, como un
\omterm{def:g6:mass-and-volume:volume}{litro} son $1000$ \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}, el mismo número sirve
también como \omterm{def:g3:measuring-mass:units}{kilogramos} por \omterm{def:g6:mass-and-volume:volume}{litro}: el agua es \qty{1.0}{kg}
por \omterm{def:g6:mass-and-volume:volume}{litro}.
\end{definition}

\begin{example}[Leer las letras]\label{ex:g7:volume-mass-density:reading}
Una fórmula es una frase en taquigrafía. $\rho = m/V$ se lee: ``para
hallar la \omterm{def:g7:volume-mass-density:formula}{densidad}, divide la \omterm{def:g3:measuring-mass:mass}{masa} de la muestra entre su
\omterm{def:g6:mass-and-volume:volume}{volumen}''. Las letras son huecos que se rellenan --- para
\emph{este} guijarro, $m$ se convierte en \qty{75}{g} y $V$ en
\qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
La misma división que el año pasado --- pero la fórmula se acuerda de
la receta para todos los guijarros que estén por venir.
\end{example}

\begin{proposition}[Tres recetas en una fórmula]\label{prop:g7:volume-mass-density:recipes}
Como la \omterm{def:g7:volume-mass-density:formula}{densidad} ata la \omterm{def:g3:measuring-mass:mass}{masa} y el volumen en proporción,
una sola fórmula sirve para tres necesidades:
\begin{enumerate}
  \item \emph{hallar una \omterm{def:g7:volume-mass-density:formula}{densidad}}: $\rho = m/V$ --- divide la
        \omterm{def:g3:measuring-mass:mass}{masa} entre el volumen;
  \item \emph{hallar una \omterm{def:g3:measuring-mass:mass}{masa}}: $m = \rho \times V$ --- cada
        \omterm{def:g6:mass-and-volume:volume}{centímetro cúbico} lleva $\rho$ \omterm{def:g3:measuring-mass:units}{gramos}, y hay $V$ de
        ellos;
  \item \emph{hallar un \omterm{def:g6:mass-and-volume:volume}{volumen}}: $V = m/\rho$ --- reparte la
        \omterm{def:g3:measuring-mass:mass}{masa} en raciones de $\rho$ \omterm{def:g3:measuring-mass:units}{gramos}; el número de
        raciones es el número de \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}.
\end{enumerate}
\end{proposition}

\begin{example}[Las recetas trabajando]\label{ex:g7:volume-mass-density:atwork}
\emph{La \omterm{def:g3:measuring-mass:mass}{masa} a partir del \omterm{def:g6:mass-and-volume:volume}{volumen}}: ¿cuánto pesan
\qty{250}{cm^3} de hierro ($\rho = 7.9$)?
$m = 7.9 \times 250 = \qty{1975}{g}$ --- casi dos \omterm{def:g3:measuring-mass:units}{kilogramos} en el
volumen de una taza de café.
\emph{El \omterm{def:g6:mass-and-volume:volume}{volumen} a partir de la \omterm{def:g3:measuring-mass:mass}{masa}}: ¿cuánto sitio ocupan
\qty{540}{g} de roble ($\rho = 0.6$)?
$V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{Compañera de sensatez}: las respuestas del hierro deben salir
pesadas y pequeñas, y las de la madera, ligeras y grandes --- la tabla
de \omterm{def:g7:volume-mass-density:formula}{densidades} es la mejor amiga del paso de juzgar.
\end{example}

\section{Medir densidades}

\begin{method}[Densidad de un líquido]\label{met:g7:volume-mass-density:liquid}
Los \omterm{def:g3:solids-liquids-gases:liquid}{líquidos} no se sostienen solos en el platillo de una \omterm{def:g3:levers-and-scales:scale}{balanza}
--- así que toma prestado el truco de la tara:
\begin{enumerate}
  \item coloca una probeta vacía en la \omterm{def:g3:levers-and-scales:scale}{balanza} y tárala a cero;
  \item echa dentro un \omterm{def:g6:mass-and-volume:volume}{volumen} cómodo --- pongamos
        $V = \qty{80}{cm^3}$ del \omterm{def:g3:solids-liquids-gases:liquid}{líquido} misterioso ---, leyendo
        bien el \omterm{met:g6:mass-and-volume:cylinder}{menisco};
  \item la \omterm{def:g3:levers-and-scales:scale}{balanza} enseña ahora solo la \omterm{def:g3:measuring-mass:mass}{masa} del \omterm{def:g3:solids-liquids-gases:liquid}{líquido}:
        pongamos $m = \qty{68}{g}$;
  \item divide: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ --- más
        ligero que el agua: probablemente un aceite.
\end{enumerate}
\end{method}

\begin{example}[Densidad de un sólido, de principio a fin]\label{ex:g7:volume-mass-density:solid}
Una medalla dice ser de plata ($\rho = 10.5$). \omterm{def:g3:levers-and-scales:scale}{Balanza}:
$m = \qty{84}{g}$. Probeta: el agua sube de \qty{60.0}{cm^3} a
\qty{68.0}{cm^3}, así que $V = \qty{8.0}{cm^3}$. Entonces
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- la firma coincide: bien
podría ser plata. (Un metal más barato vestido con un baño de plata se
habría delatado aquí --- salvo que estuviera elegido con astucia, como
enseñará el problema de fin de semana.)
\end{example}

\begin{remark}[Las unidades en las fórmulas]\label{rem:g7:volume-mass-density:units}
Una fórmula solo es honrada si sus \omterm{def:g6:measurement-in-science:measuring}{unidades} coinciden. Con $m$ en
\omterm{def:g3:measuring-mass:units}{gramos} y $V$ en \unit{cm^3}, $\rho$ habla en \unit{g/cm^3};
aliméntala con \omterm{def:g3:measuring-mass:units}{kilogramos} y \omterm{def:g6:mass-and-volume:volume}{litros} y responderá en
\unit{kg/L} --- y, por fortuna, con el mismo número para cualquier
sustancia. Pero mezcla \omterm{def:g3:measuring-mass:units}{gramos} con \omterm{def:g6:mass-and-volume:volume}{litros} y la fórmula, sin
quejarse, entregará un disparate. Regla de los profesionales: antes de
calcular, pasa revista a las \omterm{def:g6:measurement-in-science:measuring}{unidades}; después de calcular,
escríbelas en la respuesta.
\end{remark}

\section{Pensar en densidades}

\begin{example}[Aleaciones e intermedios]\label{ex:g7:volume-mass-density:alloys}
Mezcla dos sustancias y la \omterm{def:g7:volume-mass-density:formula}{densidad} de la mezcla aterriza
\emph{entre} sus firmas --- más cerca de la del ingrediente más
generoso. El bronce (cobre $9.0$ con un poco de estaño) firma cerca de
$8.8$; la mezcla de oro y plata de la corona firmaba entre $10.5$ y
$19.3$; el agua del mar, sal disuelta en agua, se sube hasta alrededor
de $1.03$. Estar en medio es en sí mismo una pista: una lectura de $12$
en un lingote de ``oro puro'' es una confesión de compañía.
\end{example}

\begin{example}[La densidad decide el mundo de la flotación]\label{ex:g7:volume-mass-density:floating}
La regla de la flotación, ya de uniforme: un objeto \omterm{def:g1:floating-and-sinking:words}{flota} en un
\omterm{def:g3:solids-liquids-gases:liquid}{líquido} cuando $\rho_{\text{objeto}} < \rho_{\text{líquido}}$. El
roble ($0.6$) sobre el agua ($1.0$): \omterm{def:g1:floating-and-sinking:words}{flota}. El hielo ($0.9$) en
aceite ($0.85$): \omterm{def:g1:floating-and-sinking:words}{se hunde} --- consulta la tabla, no tus instintos.
Un nadador ($\approx 1.0$) en los famosos lagos ultrasalados
($\approx 1.2$): \omterm{def:g1:floating-and-sinking:words}{flota} como un corcho, con el periódico en la mano.
Una sola desigualdad, el puerto entero.
\end{example}

\begin{remark}[Fíjate en la multitud, no en el paquete]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ describe \emph{sustancias}; los barcos y los chalecos
salvavidas hinchados son \emph{paquetes} --- sustancia más \omterm{def:g2:air-around-us:air}{aire}
atrapado ---, y es el $m/V$ global del paquete el que se enfrenta a la
regla de la flotación. La fórmula sirve para los dos casos si la
alimentas con la $m$ y la $V$ correctas: la \omterm{def:g7:volume-mass-density:formula}{densidad} propia del
acero para la sustancia, y la \omterm{def:g3:measuring-mass:mass}{masa} del casco entero entre el
volumen del casco entero para el paquete. Casi todas las
``paradojas'' de la flotación no son más que las dos contabilidades
confundidas.
\end{remark}

\end{exercise}

\section{Problema: el detective de la chatarrería}

\begin{problem}[{Problema de fin de semana --- una tarde con la
detective de metales de la chatarrería; cuatro lotes misteriosos, un
lingote impostor; los límites de la prueba de la
densidad}]\label{pb:g7:volume-mass-density:1}
La chatarrería compra el metal por lo que \emph{es}, no por lo que
parece, y la detective del patio trabaja con una \omterm{def:g3:levers-and-scales:scale}{balanza}, un recipiente
graduado grande y la tabla de rascacielos. Tú eres el aprendiz.
(Extracto de la tabla: aluminio $2.7$; hierro $7.9$; cobre $9.0$;
plomo $11.3$; oro $19.3$; wolframio $19.3$.)

\medskip\noindent\textbf{Parte I --- Cuatro lotes.}
\begin{enumerate}
  \item Lote A, un lingote gris: $m = \qty{5400}{g}$, desplazamiento
        $V = \qty{2000}{cm^3}$. ¿\omterm{def:g7:volume-mass-density:formula}{Densidad} e identidad?
  \item Lote B, una bobina de hilo: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$. ¿\omterm{def:g7:volume-mass-density:formula}{Densidad} e identidad?
  \item Lote C, una placa mate y pesada: $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$. ¿\omterm{def:g7:volume-mass-density:formula}{Densidad} e identidad --- y por qué
        hay que manejar el lote C con guantes y respeto?
  \item Lote D, un saco de chatarra pálida mezclada:
        $m = \qty{8100}{g}$, $V = \qty{1500}{cm^3}$. Demuestra que la
        \omterm{def:g7:volume-mass-density:formula}{densidad} del lote D cae entre dos metales de la tabla y di
        qué contiene el saco con mayor probabilidad.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Precios y predicciones.}
El patio paga por \omterm{def:g3:measuring-mass:mass}{masa} pero planifica el transporte por
\omterm{def:g6:mass-and-volume:volume}{volumen}.
\begin{enumerate}[resume]
  \item Un comprador quiere \qty{540}{g} de aluminio cortados del
        lote A. ¿Qué \omterm{def:g6:mass-and-volume:volume}{volumen} de lingote es eso?
  \item La bobina de cobre del lote B se va a fundir en cubos de
        \qty{25}{cm^3} cada uno. ¿Cuál es la \omterm{def:g3:measuring-mass:mass}{masa} de un cubo y
        cuántos cubos enteros da la bobina?
  \item Una caja puede cargar como mucho \qty{20}{kg}. ¿Cuántos de
        esos cubos de cobre puede llevar legalmente?
  \item El hueco del depósito del camión admite \qty{3000}{cm^3} más
        de chatarra de hierro. ¿Qué \omterm{def:g3:measuring-mass:mass}{masa} adicional, en
        \omterm{def:g3:measuring-mass:units}{kilogramos}, puede cargar la detective?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El impostor.}
Llega un vendedor con un reluciente lingote de ``oro'':
$m = \qty{3860}{g}$, desplazamiento $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item Calcula la \omterm{def:g7:volume-mass-density:formula}{densidad} del lingote. ¿Coincide con la firma del
        oro?
  \item La detective, impasible, consulta la última línea de la tabla
        y suspira. ¿Qué metal más barato lleva exactamente la
        \omterm{def:g7:volume-mass-density:formula}{densidad} del oro --- y qué enseña esto sobre los límites
        de la prueba de la \omterm{def:g7:volume-mass-density:formula}{densidad}?
  \item ¿Cuál de los dos instrumentos del patio ha sido derrotado
        aquí: la \omterm{def:g3:levers-and-scales:scale}{balanza}, el recipiente, los dos o ninguno? Di con
        precisión qué estableció \emph{sí} con honradez la prueba de
        la \omterm{def:g7:volume-mass-density:formula}{densidad} sobre el lingote.
  \item Sugiere otra prueba física de años anteriores de este curso
        que distinga el oro de su doble sin dañar el lingote. (Sus
        \omterm{def:g4:changes-of-state:melting-point}{puntos de fusión} son enormemente distintos --- el del oro
        cerca de \qty{1064}{\celsius} y el del doble muy por encima de
        cualquier horno de aquí ---, pero ninguna chatarrería
        funde un posible tesoro: busca pruebas más suaves, quizá
        el veredicto del \omterm{def:g1:magnets:magnet}{imán} sobre los núcleos de hierro o el
        tañido de una barra golpeada, y defiende tu elección con
        honradez.)
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- El oficio de la detective.}
\begin{enumerate}[resume]
  \item Escribe el credo de tres líneas de la detective: qué puede
        demostrar la \omterm{def:g7:volume-mass-density:formula}{densidad}, qué solo puede sugerir y qué no
        puede hacer nunca ella sola.
\end{enumerate}
\end{problem}
