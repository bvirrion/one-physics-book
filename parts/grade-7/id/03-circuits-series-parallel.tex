\chapter{Rangkaian: Lingkar, Seri, Paralel}\label{ch:g7:circuits-series-parallel}

Empat tahun bergaul dengan rangkaian telah mengisi kotak
perkakasmu: lingkar, diagram, dan kedua pemasangan kabel yang besar.
Tetapi satu tokoh sudah bekerja tanpa nama sejak semula --- sesuatu
yang mengalir. Tahun ini ia menerima namanya, arahnya, dan sifatnya
yang paling aneh: ia tidak pernah terpakai habis. Dengan arus di atas
panggung, seri dan paralel berhenti menjadi resep lalu menjadi hukum.

\section{Arus naik panggung}

\begin{definition}[Arus listrik]\label{def:g7:circuits-series-parallel:current}
\emph{Arus listrik}\index{arus listrik} adalah aliran yang berjalan
mengelilingi \omterm{def:g3:first-electric-circuit:circuit}{rangkaian tertutup}: kerumunan partikel bermuatan ---
yang banyaknya tak terbayangkan dan kecilnya tak terbayangkan, sepupu
dunia \omterm{def:g7:states-of-matter:molecule}{molekul} --- yang disuruh berarak menembus logam kawatnya oleh dorongan
\omterm{def:g3:first-electric-circuit:battery}{baterainya}. Tanpa lingkar, tanpa arak-arakan: arusnya ada hanya selagi
jalannya tertutup. (Apa sebenarnya para pengaraknya, dan bagaimana
sebuah logam meloloskan mereka, adalah kisah bagus yang disimpan untuk
tahun-tahun terakhir buku ini.)
\end{definition}

\begin{proposition}[Arah perjanjiannya]\label{prop:g7:circuits-series-parallel:direction}
Menurut perjanjian sedunia, \omterm{def:g6:circuits-and-safety:diagram}{diagram rangkaian} menandai arusnya sebagai
mengalir \emph{di luar \omterm{def:g3:first-electric-circuit:battery}{baterai} dari kutub $+$, mengelilingi
rangkaiannya, menuju kutub $-$}. Setiap anak panah pada setiap diagram
di seluruh dunia menaati perjanjian ini --- yang dipilih jauh sebelum
siapa pun dapat bertanya kepada para pengaraknya sendiri. (Ketika
fisika akhirnya dapat bertanya, jawabannya menyimpan lelucon kecil
yang termasyhur, yang disimpan untuk tahun berikutnya.)
\end{proposition}

\begin{proposition}[Arus tidak terpakai habis]\label{prop:g7:circuits-series-parallel:notconsumed}
Alat tidak memakan arusnya. Arus berapa pun yang masuk ke sebuah
lampu, \omterm{ex:g6:circuits-and-safety:motor}{motor}, atau bel meninggalkannya utuh, terus berarak: yang
diambil alatnya adalah \emph{\omterm{def:g5:everyday-energy:energy}{energi}} dari simpanan \omterm{def:g3:first-electric-circuit:battery}{baterainya}, yang
diantarkan oleh arak-arakan yang lewat. Pada lingkar seri mana pun,
karena itu, satu arus yang sama mengalir menembus setiap alat ---
sebanyak yang tiba kembali di \omterm{def:g3:first-electric-circuit:battery}{baterainya} seperti yang dahulu
meninggalkannya.
\end{proposition}

\begin{example}[Salah paham yang diadili]\label{ex:g7:circuits-series-parallel:trial}
Kekeliruan yang menggoda: ``\omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang pertama memakai habis sebagian
arusnya, jadi yang kedua pasti menyala lebih redup.'' Rangkaiannya
sendiri membantahnya. Pasanglah dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang \emph{serupa} secara
seri lalu lihatlah: keduanya menyala dengan terang yang persis sama
--- dan menukar tempatnya tidak mengubah apa pun. Kalau arusnya
terpakai habis di tengah jalan, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} kedua seharusnya selalu menjadi
yang redup, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} mana pun yang duduk di sana. Ia tidak pernah
begitu. Arusnya tiba utuh; yang dibagi pasangannya adalah
\emph{dorongan} \omterm{def:g3:first-electric-circuit:battery}{baterainya}, dan itulah sebabnya keduanya lebih redup
daripada \omterm{def:g3:first-electric-circuit:bulb}{bohlam} tunggal --- redupnya sama.
\end{example}

\section{Kedua keluarganya, sebagai hukum}

\begin{proposition}[Hukum seri]\label{prop:g7:circuits-series-parallel:series}
Pada \omterm{def:g4:series-circuits:series}{rangkaian seri} --- satu lingkar tunggal --- arus yang sama
mengalir menembus setiap alat: satu jalan, satu arak-arakan, tanpa
kekecualian. Akibatnya, kini terjelaskan: satu celah menghentikan
arak-arakannya di mana-mana sekaligus; alat yang serupa berbagi
pekerjaannya sama rata, di mana pun mereka duduk; dan urutan alat pada
lingkarnya tidak mungkin berarti bagi arak-arakan yang menembus
semuanya dengan sama.
\end{proposition}

\begin{proposition}[Hukum paralel]\label{prop:g7:circuits-series-parallel:parallel}
Pada sebuah percabangan, arusnya \emph{terbagi}: sebagian
arak-arakannya menempuh tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang}, lalu bagian-bagiannya bergabung
kembali, lengkap, pada percabangan yang jauh --- tidak ada yang
hilang, tidak ada yang bertambah. Tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} menerima dorongan penuh
\omterm{def:g3:first-electric-circuit:battery}{baterainya}, sehingga tiap alat bekerja seakan-akan sendirian; dan
\omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang tersumbat hanya tidak mengirim siapa pun, sementara \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
yang lain terus berarak.
\end{proposition}

\begin{example}[Membaca terang seperti tukang listrik]\label{ex:g7:circuits-series-parallel:brightness}
Satu \omterm{def:g3:first-electric-circuit:battery}{baterai}, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang serupa. Sendirian: terang penuh --- itulah
acuannya. Dua secara seri: keduanya sama-sama redup --- satu
arak-arakan, dorongan yang dibagi. Dua secara paralel: keduanya terang
penuh --- tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} menikmati dorongan seluruhnya (dan \omterm{def:g3:first-electric-circuit:battery}{baterainya}
kosong kira-kira dua kali lebih cepat: tidak ada yang cuma-cuma). Tiga
secara seri: lebih redup lagi, semuanya sama. Sekilas pandang tukang
listriknya: hitunglah kawan selingkarnya untuk keredupan, hitunglah
\omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} untuk tagihan \omterm{def:g3:first-electric-circuit:battery}{baterainya}.
\end{example}

\begin{method}[Meramalkan nasib lampu mana pun]\label{met:g7:circuits-series-parallel:predict}
Diberi sebuah diagram, untuk tiap lampu:
\begin{enumerate}
  \item \emph{Keberadaan}: adakah sedikitnya satu lingkar tak terputus
        dari $+$ menembus lampu ini menuju $-$? Tanpa lingkar, tanpa
        \omterm{def:g1:light-and-shadows:source}{cahaya}.
  \item \emph{Teman di jalan}: alat mana yang berbagi \emph{setiap}
        lingkar yang menembus lampu ini? Itulah kawan serinya ---
        makin banyak kawan, makin redup nyalanya.
  \item \emph{Tetangga \omterm{def:g5:series-parallel-circuits:parallel}{cabang}}: alat mana yang duduk pada \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang
        lain? Mereka tidak meredupkan lampu ini dan tidak pula ikut
        gelap bersamanya.
  \item \emph{\omterm{ex:g3:first-electric-circuit:switch}{Sakelar}}: \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang terbuka membunuh tepat lingkar
        yang didudukinya --- jalankan ulang langkah 1 dengan lingkar
        itu dibuang.
\end{enumerate}
\end{method}

\begin{remark}[Mengapa salah pahamnya sukar mati]\label{rem:g7:circuits-series-parallel:whyhard}
``Pasti ada yang terpakai habis --- \omterm{def:g3:first-electric-circuit:battery}{baterainya} kan kosong!'' Benar:
\emph{\omterm{def:g5:everyday-energy:energy}{energi}} yang tersimpan di \omterm{def:g3:first-electric-circuit:battery}{baterainya} memang dibelanjakan,
diserahkan kepada lampu sebagai \omterm{def:g1:light-and-shadows:source}{cahaya} dan kehangatan oleh arak-arakan
yang lewat. Tetapi para pengaraknya sendiri beredar tanpa cedera,
seperti kursi kereta gantung ski yang mengantar pemain ski ke atas
sepanjang hari tanpa satu kursi pun terpakai habis. Pisahkan kedua
pembukuannya --- arus beredar, \omterm{def:g5:everyday-energy:energy}{energi} diantarkan --- dan setiap
rangkaian di dalam buku ini tetap jujur. Alat yang menghitung
arak-arakannya sendiri bergabung dengan kita tahun depan.
\end{remark}

Apa itu \omterm{def:g7:circuits-series-parallel:current}{arus listrik}? Kapan ia ada di dalam sebuah rangkaian, dan

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:2}
Nyatakan arah perjanjian arusnya. Pada diagram sebuah lingkar
berbohlam satu, ke mana anak panahnya menunjuk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:3}
Dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} serupa yang \omterm{def:g4:series-circuits:series}{terangkai seri} menyala persis sama terang.
Kekeliruan populer apa yang dibantah kesamaan ini, dan bagaimana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:4}
Pada sebuah lingkar seri berisi sebuah \omterm{ex:g6:circuits-and-safety:motor}{motor} dan sebuah lampu, arus
yang menembus \omterm{ex:g6:circuits-and-safety:motor}{motornya} dibandingkan dengan arus yang menembus lampunya.
Apa kata hukum serinya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:5}
Apa yang terjadi pada arusnya di percabangan pertama sebuah \omterm{def:g5:series-parallel-circuits:parallel}{rangkaian
paralel}? Dan di percabangan yang kedua? Apa yang benar tentang jatah
tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} atas dorongan \omterm{def:g3:first-electric-circuit:battery}{baterainya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:6}

sendirian; \ dengan satu kawan seri; \ dengan satu tetangga paralel;
\ dengan dua kawan seri.

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:7}
Susunan mana yang mengosongkan \omterm{def:g3:first-electric-circuit:battery}{baterainya} lebih cepat: dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:9}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:10}
Lampu hias meminta sebuah rancangan: sepuluh \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang semuanya

seluruh perangkatnya harus menaati satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} induk. Gambarlah atau

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:11}

tembaga yang tebal sedangkan \omterm{def:g5:series-parallel-circuits:parallel}{cabang} \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 memuat kawat yang panjang

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:circuits-series-parallel:12}
Sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai} memberi makan dua \omterm{def:g5:series-parallel-circuits:parallel}{cabang} paralel: \omterm{def:g5:series-parallel-circuits:parallel}{cabang} pertama

dan kalau X yang putus.
\end{exercise}

\section{Soal: Panel Penjaga Mercusuar}

\begin{problem}[{Soal akhir pekan --- memasang ulang kabel mercusuar
tua; satu bank baterai, banyak tugas; malam penuh kerusakan bagi
penjaganya}]\label{pb:g7:circuits-series-parallel:1}
Bank \omterm{def:g3:first-electric-circuit:battery}{baterai} mercusuarnya harus memberi makan: lampu besarnya,
serunai kabut (bel yang perkasa), lampu baca penjaganya, dan lampu
tangga --- lalu selamat melewati malam penelusuran kerusakan sang
penjaga.

\medskip\noindent\textbf{Bagian I --- Rancangan.}
\begin{enumerate}
  \item Lampu besarnya tidak boleh meredup karena sesuatu yang lain
        dinyalakan. Pemasangan kabel apa yang dituntut hal ini bagi
        seluruh pemasangannya, dan mengapa pemasangan seri yang mana
        pun akan mengkhianati kapalnya?
  \item Tiap tugas memerlukan \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} sendiri, dan penjaganya
        menginginkan satu pemutus induk untuk \omterm{def:g3:sun-days-seasons:year}{musim} badai. Tempatkan
        kelima \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}.
  \item Serunai kabut dan lampu tangganya tidak boleh sampai
        terhalang bekerja oleh kerusakan yang satu lagi. Apakah hal
        ini sudah terjamin? Oleh hukum yang mana?
  \item Sketsakan atau gambarkan diagram lengkapnya: bank, \omterm{ex:g3:first-electric-circuit:switch}{sakelar}
        induk, empat \omterm{def:g5:series-parallel-circuits:parallel}{cabang}, empat \omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabang}.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pertanyaan penjaganya.}
\begin{enumerate}[resume]
  \item Penjaganya khawatir: ``Empat tugas minum sekaligus ---
        apakah tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} masih mendapat dorongan penuh banknya?''
        Jawablah dengan hukum paralelnya.
  \item ``Dan apakah arusnya terpakai habis selagi ia bekerja
        menembus lampu besarnya?'' Luruskan penjaganya, dengan kereta
        gantung ski kalau itu membantu.
  \item Pada malam berkabut segalanya menyala sekaligus. Apa harga
        kemurahan hati paralelnya, dan di bagian mercusuar mana harga
        itu dibayar?
  \item Hanya lampu bacanya yang menyala; dengan pikiran pengukur,
        penjaganya mengikuti arak-arakannya: gambarkan perjalanan
        pulang perginya yang penuh, sambil menyebut setiap komponen
        yang dilewatinya dan setiap yang dihindarinya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Malam penuh kerusakan.}
\begin{enumerate}[resume]
  \item Tengah malam: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} tangganya putus. Apa lagi yang menjadi
        gelap? Hukum yang mana yang menjawabnya?
  \item Pukul satu: sebuah perbaikan yang ceroboh meninggalkan
        sebentang kawat telanjang yang menghubungkan kedua kutub
        banknya secara langsung, di hulu setiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang}. Sebutkan
        peristiwanya, lalu jelaskan mengapa setiap tugas di
        mercusuarnya kini gagal sekaligus --- ke mana arak-arakannya
        pergi?
  \item Pukul dua, kerusakannya beres: penjaganya ingin lampu
        besarnya terlindung supaya jalan pintas seperti itu tidak
        pernah lagi dapat membungkamnya. Tersedia bank \omterm{def:g3:first-electric-circuit:battery}{baterai} kedua
        yang mandiri. Usulkan pemasangan kabelnya.
  \item Fajar: tulislah buku harian penjaganya --- tiga kalimat:
        hukum mana yang menjaga kapalnya selamat malam ini, salah
        paham mana yang dibantah malam itu, dan satu kawat mana yang
        menyebabkan \omterm{ex:g2:measuring-time:clock}{jam} yang terburuk.
\end{enumerate}
\end{problem}
