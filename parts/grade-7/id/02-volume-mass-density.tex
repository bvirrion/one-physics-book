\chapter{Volume, Massa, Massa Jenis}\label{ch:g7:volume-mass-density}

Tahun lalu kamu membongkar kedok seorang pandai emas yang curang
dengan membagi \omterm{def:g3:measuring-mass:units}{gram} dengan \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}. Tahun ini, pelajaran
matematikamu menyerahkan kekuatan baru: huruf yang mewakili bilangan.
\omterm{def:g7:volume-mass-density:formula}{Massa jenis} adalah tempat yang sempurna untuk membelanjakannya --- satu
rumus pendek yang mengemas tiga resep, perlengkapan seorang detektif,
dan seluruh aturan \omterm{def:g1:floating-and-sinking:words}{terapung} ke dalam lima lambang.

\section{Rumusnya}

\begin{definition}[Massa jenis, menurut rumus]\label{def:g7:volume-mass-density:formula}
Yang disebut \emph{massa jenis}\index{massa jenis} $\rho$ (huruf Yunani
\emph{rho}) sebuah zat adalah \omterm{def:g3:measuring-mass:mass}{massanya} tiap \omterm{def:g6:measurement-in-science:measuring}{satuan} \omterm{def:g6:mass-and-volume:volume}{volume}: untuk
sebuah cuplikan bermassa $m$ dan bervolume $V$,
\[
\rho = \frac{m}{V}.
\]
Dengan $m$ dalam \omterm{def:g3:measuring-mass:units}{gram} dan $V$ dalam \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}, $\rho$ keluar
dalam \emph{\omterm{def:g3:measuring-mass:units}{gram} tiap \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}} (\unit{g/cm^3}) --- ``\omterm{def:g3:measuring-mass:mass}{massa}
satu \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}'' tahun lalu, kini mengenakan seragamnya. \omterm{def:g3:measuring-mass:mass}{Massa}
jenis air: $\rho = \qty{1.0}{g/cm^3}$. Dan karena satu \omterm{def:g6:mass-and-volume:volume}{liter} adalah
$1000$ \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}, bilangan yang sama berlaku pula sebagai
\omterm{def:g3:measuring-mass:units}{kilogram} tiap \omterm{def:g6:mass-and-volume:volume}{liter}: air adalah \qty{1.0}{kg} tiap \omterm{def:g6:mass-and-volume:volume}{liter}.
\end{definition}

\begin{example}[Membaca hurufnya]\label{ex:g7:volume-mass-density:reading}
Sebuah rumus adalah kalimat yang disingkat. $\rho = m/V$ terbaca:
``untuk menemukan \omterm{def:g3:measuring-mass:mass}{massa} jenisnya, bagilah \omterm{def:g3:measuring-mass:mass}{massa} cuplikannya dengan
\omterm{def:g6:mass-and-volume:volume}{volumenya}.'' Hurufnya adalah tempat penampung --- untuk kerikil
\emph{ini}, $m$ menjadi
\qty{75}{g} dan $V$ menjadi \qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
Pembagian yang sama seperti tahun lalu --- tetapi rumusnya mengingat
resepnya bagi setiap kerikil yang akan datang.
\end{example}

\begin{proposition}[Tiga resep dalam satu rumus]\label{prop:g7:volume-mass-density:recipes}
Karena \omterm{def:g7:volume-mass-density:formula}{massa jenis} mengikat \omterm{def:g3:measuring-mass:mass}{massa} dan \omterm{def:g6:mass-and-volume:volume}{volume} secara sebanding, satu
rumus melayani tiga kebutuhan:
\begin{enumerate}
  \item \emph{menemukan \omterm{def:g7:volume-mass-density:formula}{massa jenis}}: $\rho = m/V$ --- bagilah \omterm{def:g3:measuring-mass:mass}{massa}
        dengan \omterm{def:g6:mass-and-volume:volume}{volume};
  \item \emph{menemukan \omterm{def:g3:measuring-mass:mass}{massa}}: $m = \rho \times V$ --- tiap
        \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik} membawa $\rho$ \omterm{def:g3:measuring-mass:units}{gram}, dan ada $V$ buah di
        antaranya;
  \item \emph{menemukan \omterm{def:g6:mass-and-volume:volume}{volume}}: $V = m/\rho$ --- bagikan \omterm{def:g3:measuring-mass:mass}{massanya}
        dalam jatah $\rho$ \omterm{def:g3:measuring-mass:units}{gram}; banyaknya jatah adalah banyaknya
        \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}.
\end{enumerate}
\end{proposition}

\begin{example}[Resepnya sedang bekerja]\label{ex:g7:volume-mass-density:atwork}
\emph{\omterm{def:g3:measuring-mass:mass}{Massa} dari \omterm{def:g6:mass-and-volume:volume}{volume}}: berapa \omterm{def:g4:weight-and-mass:weight}{berat} \qty{250}{cm^3} besi
($\rho = 7.9$)? $m = 7.9 \times 250 = \qty{1975}{g}$ ---
hampir dua \omterm{def:g3:measuring-mass:units}{kilogram} dalam ruah segelas kopi.
\emph{\omterm{def:g6:mass-and-volume:volume}{Volume} dari \omterm{def:g3:measuring-mass:mass}{massa}}: berapa ruang yang dipakai \qty{540}{g} kayu
ek ($\rho = 0.6$)? $V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{Kawan pemeriksa}: jawaban besi seharusnya berat-dan-kecil,
jawaban kayu ringan-dan-besar --- tabel \omterm{def:g7:volume-mass-density:formula}{massa jenis} adalah sahabat
terbaik langkah penilaiannya.
\end{example}

\section{Mengukur massa jenis}

\begin{method}[Massa jenis sebuah zat cair]\label{met:g7:volume-mass-density:liquid}
\omterm{def:g3:solids-liquids-gases:liquid}{Zat cair} tidak mau duduk sendirian di piring \omterm{def:g3:levers-and-scales:scale}{neraca} --- jadi
pinjamlah muslihat tara:
\begin{enumerate}
  \item letakkan gelas ukur yang kosong di atas \omterm{def:g3:levers-and-scales:scale}{neracanya} lalu
        taranya jadikan nol;
  \item tuangkan \omterm{def:g6:mass-and-volume:volume}{volume} yang enak --- misalnya $V = \qty{80}{cm^3}$
        \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} misteriusnya, sambil membaca meniskusnya dengan
        benar;
  \item \omterm{def:g3:levers-and-scales:scale}{neracanya} kini menunjukkan \omterm{def:g3:measuring-mass:mass}{massa} \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} saja: misalnya
        $m = \qty{68}{g}$;
  \item bagilah: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ ---
        lebih ringan daripada air: kemungkinan sebuah minyak.
\end{enumerate}
\end{method}

\begin{example}[Massa jenis zat padat, dari awal sampai akhir]\label{ex:g7:volume-mass-density:solid}
Sebuah medali mengaku terbuat dari perak ($\rho = 10.5$). \omterm{def:g3:levers-and-scales:scale}{Neraca}:
$m = \qty{84}{g}$. Gelas ukur: airnya memanjat dari \qty{60.0}{cm^3} ke
\qty{68.0}{cm^3}, sehingga $V = \qty{8.0}{cm^3}$. Maka
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- tanda tangannya
cocok: boleh jadi ia memang perak. (Logam yang lebih murah yang
berpakaian sepuhan perak akan mengkhianati dirinya di sini ---
kecuali kalau dipilih dengan licik, sebagaimana ditunjukkan soal akhir
pekannya.)
\end{example}

\begin{remark}[Satuan di dalam rumus]\label{rem:g7:volume-mass-density:units}
Sebuah rumus jujur hanya kalau \omterm{def:g6:measurement-in-science:measuring}{satuannya} sepakat. Dengan $m$ dalam
\omterm{def:g3:measuring-mass:units}{gram} dan $V$ dalam \unit{cm^3}, $\rho$ berbicara \unit{g/cm^3};
berilah ia \omterm{def:g3:measuring-mass:units}{kilogram} dan \omterm{def:g6:mass-and-volume:volume}{liter} dan ia menjawab \unit{kg/L} ---
untungnya, bilangan yang sama bagi zat mana pun. Tetapi campurkan \omterm{def:g3:measuring-mass:units}{gram}
dengan \omterm{def:g6:mass-and-volume:volume}{liter} dan rumusnya, tanpa mengeluh, menyerahkan omong kosong.
Aturan para ahli: sebelum menghitung, arak \omterm{def:g6:measurement-in-science:measuring}{satuannya}; sesudah
menghitung, tuliskan \omterm{def:g6:measurement-in-science:measuring}{satuannya} ke dalam jawabannya.
\end{remark}

\section{Berpikir dengan massa jenis}

\begin{example}[Paduan dan yang di antara]\label{ex:g7:volume-mass-density:alloys}
Campurkan dua zat dan \omterm{def:g7:volume-mass-density:formula}{massa jenis} campurannya mendarat \emph{di
antara} kedua tanda tangannya --- lebih dekat ke bahan yang lebih
murah hati. Perunggu (tembaga $9.0$ dengan sedikit timah)
menandatangani dekat $8.8$; campuran emas-dan-perak pada mahkotanya
menandatangani di antara $10.5$ dan $19.3$; air laut, garam yang
larut di dalam air, merangkak naik sampai kira-kira $1.03$. Keadaan di
antara itu sendiri sebuah petunjuk: bacaan $12$ dari sebatang ``emas
murni'' adalah pengakuan bahwa ia berteman.
\end{example}

\begin{example}[Massa jenis memutuskan dunia terapung]\label{ex:g7:volume-mass-density:floating}
Aturan \omterm{def:g1:floating-and-sinking:words}{terapungnya}, kini berseragam: sebuah benda \omterm{def:g1:floating-and-sinking:words}{terapung} di dalam
\omterm{def:g3:solids-liquids-gases:liquid}{zat cair} bila $\rho_{\text{benda}} < \rho_{\text{zat cair}}$. Kayu ek
($0.6$) di atas air ($1.0$): \omterm{def:g1:floating-and-sinking:words}{terapung}. Es ($0.9$) di dalam minyak
($0.85$): \omterm{def:g1:floating-and-sinking:words}{tenggelam} --- periksalah tabelnya, bukan nalurimu. Seorang
perenang ($\approx 1.0$) di danau yang termasyhur sangat asin
($\approx 1.2$): \omterm{def:g1:floating-and-sinking:words}{terapung} seperti gabus, sambil memegang koran. Satu
ketaksamaan, seluruh pelabuhannya.
\end{example}

\begin{remark}[Perhatikan kerumunannya, bukan bungkusannya]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ menggambarkan \emph{zat}; kapal dan jaket pelampung yang
menggembung adalah \emph{bungkusan} --- zat ditambah \omterm{def:g2:air-around-us:air}{udara} yang
terperangkap --- dan $m/V$ bungkusan itu secara keseluruhanlah yang
menghadapi aturan \omterm{def:g1:floating-and-sinking:words}{terapungnya}. Rumusnya menangani keduanya, kalau kamu
memberinya $m$ dan $V$ yang benar: \omterm{def:g7:volume-mass-density:formula}{massa jenis} bajanya sendiri untuk
zatnya, dan \omterm{def:g3:measuring-mass:mass}{massa} seluruh lambungnya dibagi \omterm{def:g6:mass-and-volume:volume}{volume} seluruh lambungnya
untuk bungkusannya. Kebanyakan ``paradoks'' \omterm{def:g1:floating-and-sinking:words}{terapung} hanyalah kedua
pembukuan itu yang tertukar.
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:3}

Pada \cref{met:g7:volume-mass-density:liquid}, mengapa gelas ukurnya

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:5}
Sebuah \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}: $V = \qty{50}{cm^3}$, $m = \qty{70}{g}$. Berapa \omterm{def:g3:measuring-mass:mass}{massa}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:6}

dan dalam \omterm{def:g3:measuring-mass:units}{kilogram} tiap \omterm{def:g6:mass-and-volume:volume}{liter}? Peringatan apa yang dilekatkan
\cref{rem:g7:volume-mass-density:units} pada pencampuran \omterm{def:g3:measuring-mass:units}{gram} dengan

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:7}
Sebuah gelang bertanda ``emas murni'' mempunyai $m = \qty{58}{g}$ dan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:8}
Satu \omterm{def:g2:measuring-length:units}{meter} kubik adalah kubus berukuran $100 \times 100 \times 100$
\omterm{def:g2:measuring-length:units}{sentimeter}. Berapa \unit{cm^3} itu --- dan berapa \omterm{def:g3:measuring-mass:mass}{massa} satu \omterm{def:g2:measuring-length:units}{meter}
kubik air, dalam \omterm{def:g3:measuring-mass:units}{kilogram}? (Jawabannya menjelaskan mengapa kasur air

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:9}
Sebuah piala ``perak'': $m = \qty{420}{g}$, airnya naik dari

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:10}
Sebuah botol kosong bermassa \qty{380}{g}; ketika diisi sampai tanda

Carilah \omterm{def:g7:volume-mass-density:formula}{massa jenis} \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} dalam \unit{g/cm^3} lalu usulkan
jati dirinya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:11}
Sebuah pelampung aluminium yang berongga mempunyai \omterm{def:g6:mass-and-volume:volume}{volume} total

\emph{bungkusannya} lalu ramalkan \omterm{def:g1:floating-and-sinking:words}{terapung} atau \omterm{def:g1:floating-and-sinking:words}{tenggelam} --- lalu

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:volume-mass-density:12}
Rancanglah sebuah tata cara yang lengkap untuk memutuskan apakah
sebuah rantai terbuat dari tembaga murni: sebutkan setiap
pengukurannya, resep yang dipakai, bilangan yang diharapkan bagi
kemurniannya, dan \emph{dua} alasan yang jujur mengapa putusanmu masih
mungkin keliru (pikirkan mata rantai yang berongga, dan campuran licik
yang \omterm{def:g3:measuring-mass:mass}{massa} jenisnya mendarat dekat milik tembaga). Uji tambahan apa
yang akan memperkuat perkaranya?
\end{exercise}

\section{Soal: Detektif Tempat Rongsokan}

\begin{problem}[{Soal akhir pekan --- sesore bersama detektif logam di
tempat rongsokan; empat kelompok misterius, satu batangan pengelabu;
batas uji massa jenis}]\label{pb:g7:volume-mass-density:1}
Tempat rongsokan membeli logam menurut \emph{apa adanya}, bukan
menurut rupanya, dan detektif tempat itu bekerja dengan sebuah \omterm{def:g3:levers-and-scales:scale}{neraca},
sebuah bejana berskala yang besar, dan tabel garis langitnya. Kamu
muridnya. (Kutipan tabel: aluminium $2.7$; besi $7.9$; tembaga $9.0$;
timbal $11.3$; emas $19.3$; wolfram $19.3$.)

\medskip\noindent\textbf{Bagian I --- Empat kelompok.}
\begin{enumerate}
  \item Kelompok A, sebuah batangan kelabu: $m = \qty{5400}{g}$,
        pendesakan $V = \qty{2000}{cm^3}$. Berapa \omterm{def:g3:measuring-mass:mass}{massa} jenisnya dan
        apa jati dirinya?
  \item Kelompok B, segulung kawat: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$. Berapa \omterm{def:g3:measuring-mass:mass}{massa} jenisnya dan apa jati
        dirinya?
  \item Kelompok C, sebuah lempeng suram yang \omterm{def:g4:weight-and-mass:weight}{berat}:
        $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$. Berapa \omterm{def:g3:measuring-mass:mass}{massa} jenisnya dan apa jati
        dirinya --- lalu mengapa kelompok C harus ditangani dengan
        sarung tangan dan rasa hormat?
  \item Kelompok D, sekarung rongsokan pucat yang bercampur:
        $m = \qty{8100}{g}$,
        $V = \qty{1500}{cm^3}$. Tunjukkan bahwa \omterm{def:g7:volume-mass-density:formula}{massa jenis} kelompok D
        mendarat di antara dua logam tabelnya, lalu katakan apa yang
        paling mungkin dimuat karungnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Harga dan ramalan.}
Tempat itu membayar menurut \omterm{def:g3:measuring-mass:mass}{massa} tetapi merencanakan pengangkutan
menurut \omterm{def:g6:mass-and-volume:volume}{volume}.
\begin{enumerate}[resume]
  \item Seorang pembeli menginginkan \qty{540}{g} aluminium yang
        dipotong dari kelompok A. Berapa \omterm{def:g6:mass-and-volume:volume}{volume} batangan itu?
  \item Gulungan tembaga kelompok B akan dilebur menjadi kubus yang
        masing-masing \qty{25}{cm^3}. Berapa \omterm{def:g3:measuring-mass:mass}{massa} satu kubusnya, dan
        berapa banyak kubus penuh yang dihasilkan gulungannya?
  \item Sebuah peti dapat membawa paling banyak \qty{20}{kg}. Berapa
        banyak kubus tembaga itu yang boleh dimuatnya dengan sah?
  \item Sumur tangki truknya memuat \qty{3000}{cm^3} rongsokan besi
        lagi. \omterm{def:g3:measuring-mass:mass}{Massa} tambahan berapa, dalam \omterm{def:g3:measuring-mass:units}{kilogram}, yang boleh dimuat
        detektifnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Si pengelabu.}
Seorang penjual tiba dengan batangan ``emas'' yang berkilau:
$m = \qty{3860}{g}$, pendesakan $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g7:volume-mass-density:formula}{massa jenis} batangannya. Apakah ia cocok dengan
        tanda tangan emas?
  \item Detektifnya, tanpa bergeming, melihat baris terakhir tabelnya
        lalu mengeluh. Logam yang lebih murah mana yang mengenakan
        \omterm{def:g7:volume-mass-density:formula}{massa jenis} emas dengan tepat --- dan apa yang diajarkannya
        tentang batas uji \omterm{def:g3:measuring-mass:mass}{massa} jenisnya?
  \item Alat mana di antara kedua alat tempat itu yang dikalahkan di
        sini: \omterm{def:g3:levers-and-scales:scale}{neracanya}, bejananya, keduanya, atau tidak satu pun?
        Katakan dengan tepat apa yang \emph{sungguh} ditegakkan uji
        \omterm{def:g3:measuring-mass:mass}{massa} jenisnya tentang batangan itu.
  \item Usulkan satu uji fisika lagi dari tahun-tahun sebelumnya pada
        pelajaran ini yang membedakan emas dari kembarannya tanpa
        merusak batangannya. (\omterm{def:g4:changes-of-state:melting-point}{Titik leburnya} berbeda jauh sekali ---
        emas dekat \qty{1064}{\celsius}, kembarannya jauh di atas
        setiap tungku di sini --- tetapi tidak ada tempat rongsokan
        yang melebur harta yang mungkin sungguhan: carilah bukti yang
        lebih lembut, mungkin putusan \omterm{def:g1:magnets:magnet}{magnet} atas inti besi, atau
        dentang sebatang logam yang dipukul, lalu belalah pilihanmu
        dengan jujur.)
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Keahlian detektifnya.}
\begin{enumerate}[resume]
  \item Tuliskan pengakuan iman tiga baris milik detektifnya: apa yang
        dapat dibuktikan \omterm{def:g7:volume-mass-density:formula}{massa jenis}, apa yang hanya dapat
        disarankannya, dan apa yang tidak pernah dapat dilakukannya
        sendirian.
\end{enumerate}
\end{problem}
