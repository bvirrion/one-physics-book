\chapter{Wujud Zat dan Perubahan Wujud}\label{ch:g7:states-of-matter}

Enam tahun bergaul dengan \omterm{def:g3:solids-liquids-gases:solid}{zat padat} yang mempertahankan bentuknya, \omterm{def:g3:solids-liquids-gases:liquid}{zat
cair} yang terbaring mendatar, dan \omterm{def:g3:solids-liquids-gases:gas}{gas} yang mengisi setiap sudut ---
dan tidak sekali pun kita bertanya \emph{mengapa}. Tahun ini, fisika
membuka tutupnya. Di bawah ketiga wujud itu bersembunyi salah satu
gagasan terbesar yang pernah dimiliki manusia: segalanya, sungguh
segalanya, terbuat dari partikel yang terlalu kecil untuk dilihat.
Terimalah satu gagasan itu, dan aturan yang kamu kumpulkan sejak masa
kecil menjelaskan dirinya sendiri dalam satu sapuan.

\section{Gagasan yang besar}

\begin{definition}[Molekul]\label{def:g7:states-of-matter:molecule}
Semua materi terbuat dari \emph{molekul}\index{molekul}: partikel yang
sangat kecil sampai tak terbayangkan, jauh di bawah apa yang dapat
dilihat mata atau mikroskop sekolah mana pun. Setetes air saja memuat
lebih banyak molekul daripada banyaknya tetes air di seluruh lautan.
Setiap zat murni dibangun dari jenis molekulnya sendiri --- air dari
molekul air, besi dari partikel milik besi --- dan molekul sebuah zat
semuanya serupa.
\end{definition}

\begin{remark}[Memercayai tanpa melihat]\label{rem:g7:states-of-matter:believing}
Tidak seorang pun pernah mengamati satu \omterm{def:g7:states-of-matter:molecule}{molekul} dengan mata telanjang
--- jadi mengapa memercayainya? Karena gagasan itu \emph{memperoleh}
kepercayaan: ia meramalkan, dengan benar, segala yang dimuat bab ini,
dan bergunung-gunung lagi. Begitulah cara fisika menerima tamu yang
tak terlihat: bukan atas dasar keyakinan, melainkan atas dasar
kegunaan yang tak henti-hentinya. (Gambar langsung dari alat zaman
sekarang yang menakjubkan akhirnya tiba seabad yang lalu --- lama
sesudah para fisikawan sudah yakin.)
\end{remark}

\begin{proposition}[Model partikel bagi ketiga wujud]\label{prop:g7:states-of-matter:model}
Ketiga wujud itu adalah tiga susunan \omterm{def:g7:states-of-matter:molecule}{molekul} yang sama:
\begin{enumerate}
  \item \emph{padat}: \omterm{def:g7:states-of-matter:molecule}{molekulnya} terkemas rapat dalam tumpukan yang
        tertib, masing-masing bergetar di tempatnya tetapi
        mempertahankan kedudukannya --- karena itu bentuknya tetap;
  \item \emph{cair}: \omterm{def:g7:states-of-matter:molecule}{molekulnya} masih bersentuhan, tetapi tidak
        tertib dan bebas meluncur melewati satu sama lain --- karena
        itu ia mengalir, dan permukaannya mendatar;
  \item \emph{\omterm{def:g3:solids-liquids-gases:gas}{gas}}: \omterm{def:g7:states-of-matter:molecule}{molekulnya} berjauhan, terbang cepat ke segala
        arah, bertumbukan dan memantul --- karena itu ia mengisi
        setiap sudut yang ditawarkan.
\end{enumerate}
\end{proposition}

\begin{example}[Enam tahun aturan, satu penjelasan]\label{ex:g7:states-of-matter:rules}
Jalankan aturan lama itu lewat modelnya. \omterm{def:g3:solids-liquids-gases:solid}{Zat padat} mempertahankan
bentuknya: setiap \omterm{def:g7:states-of-matter:molecule}{molekul} menjaga posnya. \omterm{def:g3:solids-liquids-gases:liquid}{Zat cair} mengambil bentuk
wadahnya dan terbaring mendatar: kerumunan yang meluncur itu merosot
sampai tidak ada \omterm{def:g7:states-of-matter:molecule}{molekul} yang dapat meluncur lebih rendah. \omterm{def:g3:solids-liquids-gases:gas}{Gas} mengisi
seluruh ruang yang ditawarkan: tidak ada yang menahan para penerbang.
Parfum yang menyeberangi ruangan: \omterm{def:g7:states-of-matter:molecule}{molekul} \omterm{def:g3:solids-liquids-gases:gas}{gas} yang terbang,
bertumbukan, menyebar. Pasir yang mengucur seperti \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}: bukan
\omterm{def:g7:states-of-matter:molecule}{molekul} yang meluncur, melainkan butiran --- tiap butirnya sendiri \omterm{def:g3:solids-liquids-gases:solid}{zat
padat} mungil yang tertib berisi triliunan \omterm{def:g7:states-of-matter:molecule}{molekul}.
\end{example}

\section{Uji ruang kosong}

\begin{method}[Dua alat suntik]\label{met:g7:states-of-matter:syringe}
Modelnya mengaku bahwa \omterm{def:g7:states-of-matter:molecule}{molekul} \omterm{def:g3:solids-liquids-gases:gas}{gas} terbang berjauhan sedangkan \omterm{def:g7:states-of-matter:molecule}{molekul}
\omterm{def:g3:solids-liquids-gases:liquid}{zat cair} bersentuhan. Kalau begitu \omterm{def:g3:solids-liquids-gases:gas}{gas} seharusnya dapat diperas, dan
\omterm{def:g3:solids-liquids-gases:liquid}{zat cair} tidak. Ujilah:
\begin{enumerate}
  \item isilah alat suntik plastik dengan \omterm{def:g2:air-around-us:air}{udara}, tutup mulutnya
        dengan ujung jari, lalu doronglah penghisapnya: ia mengalah
        --- kamu dapat memaruh ruang \omterm{def:g2:air-around-us:air}{udaranya}, sambil merasakan
        perlawanan yang kenyal;
  \item isilah alat suntik yang sama dengan air, tutup, lalu dorong:
        penghisapnya berdiri seperti dinding --- air menolak menyusut
        walau sehelai rambut;
  \item lepaskan tiap penghisapnya: \omterm{def:g2:air-around-us:air}{udaranya} melenting kembali;
        airnya tidak pernah bergerak.
\end{enumerate}
\end{method}

\begin{proposition}[Gas dapat dimampatkan; zat cair dan padat tidak]\label{prop:g7:states-of-matter:squeeze}
Sebuah \omterm{def:g3:solids-liquids-gases:gas}{gas} dapat \emph{dimampatkan}\index{dimampatkan} --- dipaksa
masuk ke ruang yang lebih sedikit --- karena sebagian besar isi \omterm{def:g3:solids-liquids-gases:gas}{gas}
adalah ruang kosong di antara \omterm{def:g7:states-of-matter:molecule}{molekul} yang beterbangan. \omterm{def:g3:solids-liquids-gases:liquid}{Zat cair} dan
\omterm{def:g3:solids-liquids-gases:solid}{zat padat} praktis \emph{tidak dapat \omterm{prop:g7:states-of-matter:squeeze}{dimampatkan}}: \omterm{def:g7:states-of-matter:molecule}{molekulnya} sudah
bersentuhan, dan materi yang sudah bersentuhan tidak dapat dikemas
lebih rapat oleh dorongan apa pun yang dapat diberikan tangan --- atau
sebuah kempa hidraulik.
\end{proposition}

\begin{example}[Udara mampat dan pekerjaannya]\label{ex:g7:states-of-matter:squeezed}
Pompa sepeda memeras beberapa ruangan penuh \omterm{def:g2:air-around-us:air}{udara} ke dalam sebuah ban
--- mungkin hanya karena \omterm{def:g2:air-around-us:air}{udara} sebagian besar kekosongan. Kerumunan
yang termampatkan itu, dengan \omterm{def:g7:states-of-matter:molecule}{molekul} yang menghantam dinding bannya
lebih cepat dan lebih rapat, menahan sepedanya tetap tegak: dorongan
sebuah \omterm{def:g3:solids-liquids-gases:gas}{gas} \emph{adalah} badai \omterm{def:g7:states-of-matter:molecule}{molekulnya}. Tabung penyelam memampatkan
persediaan napas sesore penuh ke dalam sebuah botol baja; sementara
itu minyak rem yang tidak dapat \omterm{prop:g7:states-of-matter:squeeze}{dimampatkan} membawa dorongan kakimu ke
rodanya tanpa berkurang --- tiap wujud dipekerjakan menurut bakatnya
sendiri.
\end{example}

\section{Perubahan wujud, dibaca ulang}

\begin{example}[Pintunya, molekul demi molekul]\label{ex:g7:states-of-matter:doors}
Hangatkan sebuah \omterm{def:g3:solids-liquids-gases:solid}{zat padat} dan \omterm{def:g7:states-of-matter:molecule}{molekulnya} yang bergetar berguncang
lebih keras; pada \omterm{def:g4:changes-of-state:melting-point}{titik leburnya} guncangannya membongkar tumpukan yang
tertib --- kerumunannya mulai meluncur: itulah \omterm{def:g6:water-states:changes}{peleburan}. Hangatkan
\omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} lebih lanjut dan \omterm{def:g7:states-of-matter:molecule}{molekul} yang tercepat merobek diri
sepenuhnya dari cengkeraman kerumunannya lalu terbang: itulah
\omterm{def:g6:water-states:changes}{penguapan} --- dan pada \omterm{def:g4:changes-of-state:boiling-point}{titik didihnya} pelariannya menjadi lari serempak
dari dalam \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} sendiri. Pendinginan memutar filmnya mundur:
penerbang yang tertangkap (\omterm{def:g6:water-states:changes}{pengembunan}), peluncur yang mengunci diri
ke dalam barisannya (\omterm{def:g6:water-states:changes}{pembekuan}). Enam pintu tahun lalu adalah satu
kisah tentang cengkeraman \omterm{def:g7:states-of-matter:molecule}{molekul} melawan guncangan \omterm{def:g7:states-of-matter:molecule}{molekul}.
\end{example}

\begin{example}[Hukum lama dengan cuma-cuma]\label{ex:g7:states-of-matter:free}
Dua hukum yang dahulu diraih dengan susah payah kini datang dengan
cuma-cuma. \emph{\omterm{def:g3:measuring-mass:mass}{Massa} selamat melewati perubahan wujud}: \omterm{def:g6:water-states:changes}{peleburan}
menyusun ulang \omterm{def:g7:states-of-matter:molecule}{molekulnya} tetapi tidak menciptakan maupun memusnahkan
satu pun --- partikel yang sama, \omterm{def:g3:measuring-mass:mass}{massa} yang sama. \emph{Dataran \omterm{def:g3:temperature-thermometers:temperature}{suhu}}:
pada \omterm{def:g4:changes-of-state:melting-point}{titik leburnya}, kehangatan yang masuk habis dipakai memecahkan
cengkeraman tumpukannya, \omterm{def:g7:states-of-matter:molecule}{molekul} demi \omterm{def:g7:states-of-matter:molecule}{molekul}, bukan untuk guncangan
yang lebih cepat --- \omterm{def:g3:temperature-thermometers:temperature}{termometernya} menunggu sampai barisan yang
terakhir terpecah. Apa yang ditemukan pengukuran, dijelaskan modelnya.
\end{example}

\begin{remark}[Penggembungan air yang aneh]\label{rem:g7:states-of-matter:iceswell}
Bahkan penggembungan air yang aneh ketika \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} pun ikut bergabung:
\omterm{def:g7:states-of-matter:molecule}{molekul} air kebetulan mengunci diri ke dalam tumpukan yang luar biasa
\emph{lapang} --- kisi terbuka dengan ruang kosong lebih banyak
daripada yang tadi dimiliki kerumunan yang meluncur. Tumpukan yang
lebih lapang, ruang lebih banyak, \omterm{def:g6:density-floating:density}{massa jenis} lebih kecil: es \omterm{def:g1:floating-and-sinking:words}{terapung}
di atas \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} sendiri. Kebanyakan zat menumpuk lebih rapat
daripada ketika meluncur; tata bangunan air adalah kekecualian yang
indah.
\end{remark}

\section{Kalor, dilihat dari dalam}

\begin{proposition}[Suhu dan gerak molekul]\label{prop:g7:states-of-matter:temperature}
\omterm{def:g3:temperature-thermometers:temperature}{Suhu} \omterm{def:g6:measurement-in-science:measuring}{mengukur} kelincahan gerak \omterm{def:g7:states-of-matter:molecule}{molekul}: makin panas sebuah zat, makin
cepat \omterm{def:g7:states-of-matter:molecule}{molekulnya} bergetar, meluncur, atau terbang. \omterm{def:g5:heat-and-insulation:heat}{Kalor} yang mengalir
dari panas ke dingin adalah kerumunan yang lincah menenangkan diri
selagi ia membangkitkan kerumunan yang lamban --- tumbukan demi
tumbukan --- sampai keduanya berbagi satu langkah: satu \omterm{def:g3:temperature-thermometers:temperature}{suhu}. Hukum
menurun tentang \omterm{def:g5:heat-and-insulation:heat}{kalor} yang lama itu adalah tenis meja \omterm{def:g7:states-of-matter:molecule}{molekul}.
\end{proposition}

\begin{example}[Dingin bukan sebuah zat]\label{ex:g7:states-of-matter:cold}
Modelnya memensiunkan godaan lama untuk selamanya: tidak ada yang
bernama ``dingin'' yang pernah mengalir. Yang ada hanyalah gerak
\omterm{def:g7:states-of-matter:molecule}{molekul} yang lebih lambat dan lebih cepat; yang lincah meminjamkan
kelincahan, tidak pernah sebaliknya. Pada keadaan terdingin yang dapat
dibayangkan --- \omterm{def:g7:states-of-matter:molecule}{molekul} sediam yang diizinkan alam --- terletak lantai
\omterm{def:g3:temperature-thermometers:temperature}{suhu} yang sejati, jauh di bawah lemari beku mana pun; bilangannya yang
aneh dan termasyhur menjadi milik jilid Sekolah Menengah.
\end{example}

tetes yang sama menyebar lebih cepat di air yang hangat. (Pengamatan

\omterm{def:g7:states-of-matter:molecule}{molekul}.)
\end{exercise}

\section{Soal: Teater Molekul}

\begin{problem}[{Soal akhir pekan --- kelas mementaskan model partikel
di gedung olahraga; murid sebagai molekul, wujud sebagai koreografi;
pertanyaan para pengkritik}]\label{pb:g7:states-of-matter:1}
Kelas fisika mementaskan materi itu sendiri: tiga puluh murid
memerankan \omterm{def:g7:states-of-matter:molecule}{molekul} sebuah zat, dan gedung olahraganya menjadi
wadahnya.

\medskip\noindent\textbf{Bagian I --- Tiga babak.}
\begin{enumerate}
  \item Babak pertama, \omterm{def:g3:solids-liquids-gases:solid}{zat padat}: bagaimana ketiga puluhnya harus
        berdiri, dan gerak kecil apa yang harus terus dilakukan
        masing-masing? Apa yang tidak boleh dilakukan satu pun di
        antaranya?
  \item Babak kedua, \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}: apa yang berubah pada susunannya, dan
        kebebasan baru apa yang muncul? Apa yang masih harus terus
        dilakukan para muridnya, yang nanti dihapus oleh babak
        gasnya?
  \item Babak ketiga, \omterm{def:g3:solids-liquids-gases:gas}{gas}: gambarkan koreografinya --- lalu katakan
        mengapa babak ini memerlukan \emph{seluruh} gedung olahraga
        sementara kedua babak pertama berdesak di satu sudut.
  \item Penontonnya bertanya: ``Apakah ketiga puluh muridnya sama
        pada ketiga babaknya?'' Hukum mana dari tahun lalu yang
        dipentaskan oleh jawabannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Para pengkritik menguji
sandiwaranya.}
\begin{enumerate}[resume]
  \item Seorang pengkritik mendorong pelan kerumunan pada babak
        gasnya, dan menggiringnya ke separuh gedung. Apakah babaknya
        mengizinkannya? Dan dorongan yang sama pada desakan babak \omterm{def:g3:solids-liquids-gases:liquid}{zat
        cairnya}? Sebutkan proposisi yang sedang dipentaskan.
  \item Untuk mementaskan \emph{\omterm{def:g6:water-states:changes}{peleburan}}, sutradaranya memutar
        tombol \omterm{def:g5:heat-and-insulation:heat}{kalor} khayalan. Apa yang harus makin banyak dilakukan
        murid pada babak padatnya, dan pada saat apa babaknya sah
        menjadi babak \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}?
  \item Sandiwaranya harus menunjukkan dataran \omterm{def:g6:water-states:changes}{peleburannya}: selama
        perubahannya berlangsung, \omterm{def:g5:everyday-energy:energy}{energi} tombol \omterm{def:g5:heat-and-insulation:heat}{kalornya} dibelanjakan
        untuk apa --- dan apa yang harus dilaporkan ``pengkritik
        \omterm{def:g3:temperature-thermometers:temperature}{termometer}'' tentang kelincahan rata-rata muridnya selama
        itu?
  \item Pentaskan \omterm{def:g6:water-states:changes}{penguapan} dari babak \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya}: murid yang mana
        yang meninggalkan desakannya --- sembarang saja, atau jenis
        tertentu? Apa yang terjadi pada kelincahan rata-rata
        desakannya selagi mereka pergi, dan rasa sejuk sehari-hari
        apa yang dijelaskannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Efek khusus.}
\begin{enumerate}[resume]
  \item Efek balon: babak gasnya dipagari murid yang memegang
        lingkaran tali (kulit balonnya). Apa yang dilakukan para
        pemeran \omterm{def:g7:states-of-matter:molecule}{molekul} terhadap talinya, dan apa yang terjadi pada
        lingkarannya ketika para pemerannya bergerak lebih cepat?
  \item \omterm{def:g6:water-states:changes}{Pembekuan} air yang aneh harus dipentaskan. Petunjuk yang
        tidak biasa tentang \emph{jarak} apa yang diberikan
        sutradaranya kepada tumpukan pada babak padatnya, dan
        kenyataan \omterm{def:g1:floating-and-sinking:words}{terapung} apa yang dijelaskannya?
  \item Babak penutup tetes tinta: seorang murid berbaju biru mulai
        di sebuah sudut pada babak \emph{\omterm{def:g3:solids-liquids-gases:liquid}{zat cair}}. Tanpa dorongan
        sutradara apa pun, bagaimana baju biru itu berakhir jauh dari
        sudutnya --- dan mengapa lebih cepat kalau tombol \omterm{def:g5:heat-and-insulation:heat}{kalornya}
        dinaikkan?
  \item Tulislah catatan acaranya: dalam tiga kalimat, katakan satu
        gagasan apa yang dipentaskan seluruh sandiwaranya, sebutkan
        dua hukum lama yang dijelaskannya dengan cuma-cuma, dan
        akuilah dengan jujur apa yang tidak dapat ditunjukkan babak
        mana pun (betapa \emph{kecil} dan betapa \emph{banyak} para
        pemeran yang sesungguhnya).
\end{enumerate}
\end{problem}
