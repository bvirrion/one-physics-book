\chapter{Grafik Gerak dan Kelajuan Rata-Rata}\label{ch:g7:motion-average-speed}

Dua tahun lalu, \omterm{def:g7:motion-average-speed:formula}{kelajuan} berarti ``kilometer yang ditempuh dalam satu
\omterm{ex:g2:measuring-time:clock}{jam},'' yang dihitung dengan jari. Kini, berbekal huruf dan grafik,
gagasannya bertumbuh gigi: sebuah rumus yang menghitung ke tiga arah,
sebuah gambar yang memperlihatkan seluruh perjalanan dalam sekilas
pandang --- dan sebuah perangkap termasyhur tentang rata-rata yang
menjerat orang dewasa tiap hari.

\section{Rumusnya}

\begin{definition}[Kelajuan, menurut rumus]\label{def:g7:motion-average-speed:formula}
Untuk sebuah perjalanan (atau sebentang darinya) yang ditempuh pada
langkah yang mantap, \emph{kelajuan}\index{kelajuan} $v$ adalah jarak
$d$ dibagi waktu tempuh $t$:
\[
v = \frac{d}{t}.
\]
Dengan $d$ dalam kilometer dan $t$ dalam \omterm{ex:g2:measuring-time:clock}{jam}, $v$ berbicara
\emph{kilometer tiap \omterm{ex:g2:measuring-time:clock}{jam}} (\unit{km/h}); dengan \omterm{def:g2:measuring-length:units}{meter} dan sekon,
\emph{\omterm{def:g2:measuring-length:units}{meter} tiap sekon} (\unit{m/s}) --- kesayangan para ilmuwan.
Seperti rumus \omterm{def:g7:volume-mass-density:formula}{massa jenis}, rumus ini menghitung ke ketiga arahnya:
$v = d/t$, \ $d = v \times t$, \
$t = d/v$.
\end{definition}

\begin{example}[Resepnya sedang bekerja]\label{ex:g7:motion-average-speed:recipes}
Sebuah kereta menempuh \qty{240}{km} dalam $2$ \omterm{ex:g2:measuring-time:clock}{jam}: $v = 240 \div 2 =
\qty{120}{km/h}$. Berapa jauh pada langkah itu dalam $5$ \omterm{ex:g2:measuring-time:clock}{jam}? $d = 120
\times 5 = \qty{600}{km}$. Berapa lama untuk \qty{300}{km}? $t = 300
\div 120 = 2.5$ \omterm{ex:g2:measuring-time:clock}{jam} --- dua \omterm{ex:g2:measuring-time:clock}{jam} tiga puluh menit (awas si pemberontak:
$0.5$ \omterm{ex:g2:measuring-time:clock}{jam} adalah $30$ menit, bukan ``$50$'').
\end{example}

\begin{example}[Dua satuan, satu kelajuan]\label{ex:g7:motion-average-speed:units}
Seorang pelari cepat menempuh \qty{100}{m} dalam \qty{10}{s}: $v = \qty{10}{m/s}$.
Secepat apa itu dalam \omterm{def:g6:measurement-in-science:measuring}{satuan} jalan raya? Dalam satu \omterm{ex:g2:measuring-time:clock}{jam} --- $3600$
sekon --- pelarinya akan menempuh $10 \times 3600 = \qty{36000}{m} =
\qty{36}{km}$: jadi $\qty{10}{m/s} = \qty{36}{km/h}$. Kurs tukarnya
secara umum: tiap \unit{m/s} bernilai \qty{3.6}{km/h} ---
karena satu \omterm{ex:g2:measuring-time:clock}{jam} memuat $3600$ sekon sedangkan satu kilometer hanya
$1000$ \omterm{def:g2:measuring-length:units}{meter}. Angka \qty{340}{m/s} milik \omterm{def:g3:sound-around-us:sound}{bunyi}, dalam \omterm{def:g6:measurement-in-science:measuring}{satuan} jalan
raya: lebih daripada \qty{1200}{km/h}.
\end{example}

\section{Perjalanan sebagai sebuah gambar}

\begin{proposition}[Hukum grafiknya]\label{prop:g7:motion-average-speed:graphs}
Gambarkan jarak yang ditempuh terhadap waktu, dan geraknya menuliskan
riwayat hidupnya sendiri:
\begin{enumerate}
  \item \omterm{def:g6:describing-motion:uniform}{gerak beraturan} menggambar garis \emph{lurus} --- jarak yang
        sama dalam waktu yang sama, langkah demi langkah;
  \item makin \emph{curam} garisnya, makin cepat geraknya:
        kecuraman adalah \omterm{def:g7:motion-average-speed:formula}{kelajuan} yang dibuat terlihat;
  \item bentangan yang \emph{mendatar} adalah sebuah perhentian:
        waktu berlalu, jaraknya diam.
\end{enumerate}
Belokannya bercerita tentang perubahan: melengkung ke atas, makin
cepat; mendatar, makin lambat.
\end{proposition}

\begin{method}[Membaca grafik jarak--waktu]\label{met:g7:motion-average-speed:read}
\begin{enumerate}
  \item periksalah \omterm{def:g6:sun-earth-moon:axisorbit}{sumbu} dan \omterm{def:g6:measurement-in-science:measuring}{satuannya} lebih dahulu --- menit atau
        \omterm{ex:g2:measuring-time:clock}{jam}, \omterm{def:g2:measuring-length:units}{meter} atau kilometer;
  \item belahlah garisnya pada belokannya menjadi bentangan; berilah
        nama pada masing-masingnya: lurus-dan-curam,
        lurus-dan-landai, mendatar;
  \item untuk bentangan lurus mana pun, bacalah naiknya dan
        mendatarnya --- jarak yang diperoleh, waktu yang dipakai ---
        lalu bagilah: itulah \omterm{def:g7:motion-average-speed:formula}{kelajuan} bentangannya;
  \item untuk cerita seluruhnya, bacalah jarak total pada waktu yang
        terakhir --- lalu ingatlah bahwa bentangan mendatarnya
        termasuk bagian waktu totalnya.
\end{enumerate}
\end{method}

\begin{example}[Kurirnya, dipecahkan sandinya]\label{ex:g7:motion-average-speed:decoded}
Terapkan caranya pada gambarnya. Bentangan pertama: \qty{10}{km} dalam
$20$ menit --- sepertiga \omterm{ex:g2:measuring-time:clock}{jam} --- sehingga $v = 10 \div
\tfrac{1}{3} = \qty{30}{km/h}$. Kedua: mendatar dari menit $20$ sampai
$35$ --- sebuah perhentian pengantaran. Ketiga: \qty{10}{km} dalam $25$
menit, sedikit di bawah \qty{24}{km/h}. Totalnya: \qty{20}{km} dalam
satu \omterm{ex:g2:measuring-time:clock}{jam} --- yang menyerahkan gagasan bab ini berikutnya kepada kita
di atas piring.
\end{example}

\section{Kelajuan rata-rata --- dan perangkapnya yang termasyhur}

\begin{definition}[Kelajuan rata-rata]\label{def:g7:motion-average-speed:average}
\emph{Kelajuan rata-rata}\index{kelajuan rata-rata} sebuah perjalanan
utuh adalah jarak totalnya dibagi waktu \emph{total} --- termasuk
perhentiannya:
\[
v_{\text{rata-rata}} = \frac{d_{\text{total}}}{t_{\text{total}}}.
\]
Ia adalah satu langkah mantap yang akan menempuh jalan yang sama dalam
waktu keseluruhan yang sama --- \qty{20}{km} milik kurirnya dalam satu
\omterm{ex:g2:measuring-time:clock}{jam}: rata-rata \qty{20}{km/h}, padahal rodanya tidak pernah sekali pun
berputar pada \omterm{def:g7:motion-average-speed:formula}{kelajuan} itu.
\end{definition}

\begin{proposition}[Perangkapnya]\label{prop:g7:motion-average-speed:trap}
\omterm{def:g7:motion-average-speed:average}{Kelajuan rata-rata} sebuah perjalanan pada umumnya \emph{bukan} titik
tengah \omterm{def:g7:motion-average-speed:formula}{kelajuannya}. Bentangan yang lambat memakan waktu lebih banyak
daripada yang cepat, sehingga bobotnya lebih berat dalam
rata-ratanya. Hanya resep yang penuh --- jarak total dibagi waktu
total --- yang dapat dipercaya; merata-ratakan bilangan \omterm{def:g7:motion-average-speed:formula}{kelajuannya}
sendiri adalah langkah keliru yang paling menggoda di dalam bab ini.
\end{proposition}

\begin{example}[Perangkapnya, terjeplak]\label{ex:g7:motion-average-speed:sprung}
Seorang pengendara sepeda mengayuh \qty{60}{km} pergi pada
\qty{30}{km/h}, dan \qty{60}{km} yang sama pulang pada \qty{20}{km/h}.
``Rata-ratanya \qty{25}{km/h}''? Periksalah dengan jujur. Perginya:
$t = 60 \div 30 = 2$ \omterm{ex:g2:measuring-time:clock}{jam}. Pulangnya: $t = 60 \div 20 = 3$ \omterm{ex:g2:measuring-time:clock}{jam}.
Perjalanan seluruhnya: \qty{120}{km} dalam $5$ \omterm{ex:g2:measuring-time:clock}{jam} ---
$v_{\text{rata-rata}} = \qty{24}{km/h}$. Separuh yang lambat menuntut
tiga \omterm{ex:g2:measuring-time:clock}{jam} dari kelimanya, dan menyeret rata-ratanya ke bawah titik
tengahnya. Makin cepat kamu melaju pada satu separuhnya, makin sedikit
waktu separuh itu bahkan ada.
\end{example}

\begin{remark}[Apa yang diketahui jarum kelajuan]\label{rem:g7:motion-average-speed:speedometer}
\omterm{def:g7:motion-average-speed:average}{Kelajuan rata-rata} menggambarkan sebuah perjalanan utuh; jarum
\omterm{def:g7:motion-average-speed:formula}{kelajuan} menjawab pertanyaan yang lain --- \emph{secepat apa saat
ini}. Pada grafiknya, ``saat ini'' berdiam di dalam kecuraman garisnya
\emph{pada satu titik} --- mudah dilihat pada bentangan yang lurus,
halus di tempat garisnya melengkung. Membuat ``kecuraman pada sebuah
titik'' menjadi cermat adalah salah satu penemuan terbesar matematika,
dan ia menantimu di akhir sekolah menengah. Sampai saat itu: bentangan
yang lurus mendapat bilangan, lengkungannya mendapat cerita.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:1}
Tuliskan rumus \omterm{def:g7:motion-average-speed:formula}{kelajuannya} dan kedua \omterm{def:g6:measurement-in-science:measuring}{satuan} sehari-harinya. Resep yang
mana yang menemukan jarak? Yang mana yang menemukan waktu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:2}
Sebuah kapal feri menempuh \qty{45}{km} dalam $3$ \omterm{ex:g2:measuring-time:clock}{jam}; seekor kelinci
berlari \qty{100}{m} dalam \qty{8}{s}. Hitunglah kedua \omterm{def:g7:motion-average-speed:formula}{kelajuannya},
masing-masing dalam \omterm{def:g6:measurement-in-science:measuring}{satuan} alaminya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:3}
Ubahlah, dengan kurs tukarnya: \qty{5}{m/s} menjadi \unit{km/h}; \
\qty{72}{km/h} menjadi \unit{m/s}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:4}
Nyatakan ketiga hukum grafiknya. Apa yang diceritakan belokan yang
mendatar sedikit demi sedikit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:5}
Pada grafik kurirnya: di antara menit yang mana langkahnya paling
landai (tetapi bukan nol)? Bagaimana kamu dapat mengetahuinya tanpa
menghitung?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:6}
Grafik seorang pejalan kaki menunjukkan garis lurus yang melewati
titik ($30$ menit, \qty{2}{km}) dan ($60$ menit, \qty{4}{km}).
Beraturan atau berubah? Berapa \omterm{def:g7:motion-average-speed:formula}{kelajuannya} dalam \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:7}
Batasilah \omterm{def:g7:motion-average-speed:average}{kelajuan rata-rata}. Sebuah pendakian: \qty{12}{km} dalam $4$
\omterm{ex:g2:measuring-time:clock}{jam} termasuk piknik sejam. Berapa \omterm{def:g7:motion-average-speed:average}{kelajuan rata-ratanya} --- dan berapa
\omterm{def:g7:motion-average-speed:average}{kelajuan rata-rata} \emph{selagi berjalan}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:8}
Mengapa bentangan yang lambat ``lebih berat'' dalam sebuah rata-rata
daripada bentangan cepat yang sama panjangnya? Jawablah dengan waktu,
bukan dengan rumus.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:9}
Putar ulang perangkapnya: \qty{30}{km} pergi pada \qty{15}{km/h},
\qty{30}{km} kembali pada \qty{30}{km/h}. Berapa titik tengah yang
diramalkan, berapa rata-rata yang jujur --- dan separuh perjalanan
yang mana yang memiliki sebagian besar \omterm{ex:g2:measuring-time:clock}{jamnya}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:10}
Sketsakan (atau gambarkan dengan cermat) grafik perjalanan ini:
beraturan \qty{40}{km/h} selama setengah \omterm{ex:g2:measuring-time:clock}{jam}; berhenti selama
seperempat \omterm{ex:g2:measuring-time:clock}{jam}; beraturan \qty{60}{km/h} selama seperempat \omterm{ex:g2:measuring-time:clock}{jam}. Lalu
hitunglah \omterm{def:g7:motion-average-speed:average}{kelajuan rata-rata} perjalanannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:11}
Menghitung badai, ditingkatkan: guruhnya tiba \qty{6}{s} sesudah
kilatnya. Dengan \omterm{def:g3:sound-around-us:sound}{bunyi} pada \qty{340}{m/s}, pakailah $d = v \times t$
untuk jarak badainya --- lalu periksalah aturan lama ``bagilah dengan
tiga untuk mendapat kilometer'' terhadap rumusmu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:motion-average-speed:12}
Buah kenari yang lama, yang berharga tiap menitnya: seorang pengemudi
menempuh separuh pertama \emph{jaraknya} pada \qty{30}{km/h} lalu
menginginkan rata-rata keseluruhan \qty{60}{km/h}. Tunjukkan --- dengan
resep jarak-total-dibagi-waktu-total pada perjalanan \qty{60}{km} ---
bahwa separuh yang kedua harus ditempuh dalam waktu nol: keinginannya
mustahil, bukan sekadar sukar.
\end{exercise}

\section{Soal: Hari Jumat Sang Kurir}

\begin{problem}[{Soal akhir pekan --- satu kurir sepeda, satu hari
Jumat di kota; grafik sang penyalur tugas, perangkap pada lembar
bonusnya}]\label{pb:g7:motion-average-speed:1}
Penyalur tugasnya mencatat hari Jumat kurir Lena pada sebuah grafik
jarak--waktu lalu menetapkan upahnya dari grafik itu. Kamu memeriksa
lembarnya.

\medskip\noindent\textbf{Bagian I --- Paginya, dari
grafiknya.}
Grafiknya menunjukkan: panjatan lurus dari ($0$ menit, \qty{0}{km})
sampai ($30$ menit, \qty{9}{km}); mendatar sampai menit $45$; lurus
dari sana sampai ($75$ menit, \qty{15}{km}); mendatar sampai menit $90$.
\begin{enumerate}
  \item Ceritakan kisah paginya dengan kata-kata: bentangan,
        perhentian, dan kira-kira apa perhentiannya itu.
  \item Berapa \omterm{def:g7:motion-average-speed:formula}{kelajuan} pada bentangan pertama, dalam \unit{km/h}?
  \item Berapa \omterm{def:g7:motion-average-speed:formula}{kelajuan} pada bentangan mengayuh yang kedua?
  \item Berapa \omterm{def:g7:motion-average-speed:formula}{kelajuan} \emph{rata-rata} Lena atas seluruh pagi $90$
        menitnya, termasuk perhentiannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Sorenya, dari
rumusnya.}
\begin{enumerate}[resume]
  \item Tahap sore satu: \qty{12}{km} pada \qty{24}{km/h} yang
        mantap. Berapa lama waktunya, dalam menit?
  \item Tahap dua: pengantaran ke kota atas, \qty{20}{minutes} pada
        \qty{18}{km/h}. Berapa kilometer?
  \item Tahap tiga: perjalanan panjang ke gudangnya, \qty{10}{km},
        yang diselesaikan dalam $25$ menit. Berapa \omterm{def:g7:motion-average-speed:formula}{kelajuan} langkah
        mantapnya dalam \unit{km/h}?
  \item Sore seluruhnya: jumlahkan jaraknya dan waktunya (tahapnya
        saja, tanpa istirahat): \omterm{def:g7:motion-average-speed:average}{kelajuan rata-rata} berapa yang dijaga
        rodanya selagi berputar?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Perangkap pada lembar bonusnya.}
Aturan bonusnya: ``\omterm{def:g7:motion-average-speed:average}{kelajuan rata-rata} atas satu ronde penuh di atas
\qty{20}{km/h} memperoleh bonus pengayuh cepat.''
\begin{enumerate}[resume]
  \item Hari Jumat penuh milik Lena: \qty{37}{km} dalam $3$ \omterm{ex:g2:measuring-time:clock}{jam}
        (termasuk semua perhentiannya). Apakah ia memperoleh bonusnya?
  \item Rekannya Max menuntutnya: ``Aku mengayuh separuh rondeku pada
        \qty{30}{km/h} dan separuhnya pada \qty{15}{km/h} ---
        rata-ratanya $22.5$!'' Penyalur tugasnya memeriksa: kedua
        separuhnya sama-sama \qty{12}{km}. Hitunglah rata-rata Max
        yang sebenarnya lalu berilah putusan atas bonusnya.
  \item Jelaskan kepada Max, dalam bahasa waktu, di mana letak
        kekeliruan $22.5$-nya.
  \item Lena mengusulkan aturan bonus yang lebih adil bagi kurir ---
        yang tidak menghukum perhentian pengantaran. Usulkan satu
        (grafiknya tahu caranya), lalu katakan \omterm{def:g7:motion-average-speed:formula}{kelajuan} yang mana yang
        diukurnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Pelajaran sang penyalur tugas.}
\begin{enumerate}[resume]
  \item Tinjauan hari Jumat: tulislah tiga kalimat milik penyalur
        tugasnya --- apa arti bentangan yang lurus, bentangan yang
        mendatar, dan bentangan yang lebih curam pada grafiknya
        masing-masing --- lalu satu peringatan tentang merata-ratakan
        \omterm{def:g7:motion-average-speed:formula}{kelajuan}.
\end{enumerate}
\end{problem}
