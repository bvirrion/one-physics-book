\chapter{Korsleting dan Keselamatan Listrik}\label{ch:g7:short-circuits-safety}

Setiap aturan keselamatan yang kamu pelajari berpangkal pada satu
penjahat yang ditemui sekilas tahun lalu: \omterm{def:g7:short-circuits-safety:device}{korsleting}. Bab ini
mempelajarinya dengan sungguh-sungguh --- bagaimana ia bermula,
mengapa ia membakar, bagaimana sepotong kawat kurban yang bersahaja
mengalahkannya --- lalu mengubah kotak \omterm{def:g7:short-circuits-safety:fuse}{sekring} keluarga dari lemari
misterius menjadi sekutu yang kamu pahami.

\section{Cara kerja sang penjahat}

\begin{proposition}[Arus lebih menyukai jalan tanpa upaya]\label{prop:g7:short-circuits-safety:effortless}
Di antara dua jalan yang menghubungkan dua titik yang sama, arus yang
berarak berdesak ke jalan yang lebih sedikit melawannya. Alat ---
lampu, \omterm{ex:g6:circuits-and-safety:motor}{motor}, dan bel --- semuanya melawan arak-arakannya selagi
mereka bekerja; kawat tebal yang polos hampir sama sekali tidak
melawannya.
Tawarkan jalan pintas berupa kawat telanjang kepada arak-arakannya,
dan ia meninggalkan jalan yang bekerja itu hampir seluruhnya demi
jalan yang tanpa upaya.
\end{proposition}

\begin{definition}[Korsleting sebuah alat]\label{def:g7:short-circuits-safety:device}
Sebuah alat mengalami \emph{korsleting}\index{korsleting} bila sebuah
jalan penghantar menghubungkan kedua kutubnya secara langsung, dengan
melewatinya. Arak-arakannya meninggalkan alatnya --- lampu yang
terkorsleting menjadi gelap, \omterm{ex:g6:circuits-and-safety:motor}{motor} berhenti --- sementara sisa
lingkarnya berjalan terus, kini dengan satu pekerja lebih sedikit yang
melawan arak-arakannya.
\end{definition}

\begin{definition}[Korsleting baterainya]\label{def:g7:short-circuits-safety:battery}
Perkara yang paling berat: sebuah jalan penghantar menghubungkan kedua
kutub \emph{\omterm{def:g3:first-electric-circuit:battery}{baterainya}} tanpa alat apa pun padanya. Sama sekali tidak
ada yang melawan arak-arakannya, dan ia menjadi lari serempak ---
serbuan liarnya memanaskan kawatnya dan \omterm{def:g3:first-electric-circuit:battery}{baterainya} sendiri, merusak
selnya, melelehkan isolasinya, dan, pada listrik rumah, membakar rumah.
\end{definition}

\begin{method}[Sang penjahat, diperagakan dengan aman]\label{met:g7:short-circuits-safety:demo}
Dengan satu \omterm{def:g3:first-electric-circuit:battery}{baterai} dan dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} secara seri (aman: dorongan
\omterm{def:g3:first-electric-circuit:battery}{baterainya} mungil):
\begin{enumerate}
  \item nyalakan kedua \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} --- sama-sama redup, sebagaimana
        dijanjikan hukum serinya;
  \item kini sentuhkan sepotong kawat polos melintang pada kedua kutub
        \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2, sehingga melewatinya;
  \item lihatlah: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 mati seketika --- ditinggalkan ---
        sementara \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 melonjak \emph{lebih terang}: dengan satu
        penentang yang terlewati, arak-arakan seluruh lingkarnya
        menguat;
  \item singkirkan kawatnya: arak-arakannya masuk kembali ke \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2,
        dan keredupan yang sama pun kembali.
\end{enumerate}
Satu percobaan, tiga pelajaran: aturan jalan pintasnya, alat yang
ditinggalkan, dan serbuan yang menguat.
\end{method}

\begin{example}[Mengapa kawatnya memanas]\label{ex:g7:short-circuits-safety:heating}
Arus yang berarak selalu menghangatkan jalannya sedikit --- nyala
lampunya adalah penghangatan itu, yang dipekerjakan dengan sengaja di
dalam sehelai benang yang dibangun untuk berpijar putih. Pada lari
serempak, penghangatannya berubah menjadi ancaman: serbuan
\omterm{def:g7:short-circuits-safety:device}{korsletingnya} dapat memanaskan kawat tembaga melewati \omterm{def:g4:changes-of-state:melting-point}{titik lebur}
selubung plastiknya dalam hitungan sekon. Cuplikan sebesar saku:
\omterm{def:g3:first-electric-circuit:battery}{baterai} yang terkorsleting oleh sepotong kawat menjadi hangat di
tanganmu --- fisika yang sama yang, pada ukuran listrik rumah,
menghanguskan dinding. (Bagaimana penghangatannya tumbuh bersama
serbuannya diukur dengan jujur dua tahun lagi, pada bab daya listrik.)
\end{example}

\section{Para pengawal}

\begin{definition}[Sekring]\label{def:g7:short-circuits-safety:fuse}
Sebuah \emph{sekring}\index{sekring} adalah pengawal yang mati
untukmu: sepotong kawat yang sengaja dibuat tipis, \omterm{def:g4:series-circuits:series}{terangkai seri}
dengan rangkaian yang dijaganya, dan dipilih supaya melebur --- lalu
memutus lingkarnya --- ketika arusnya melampaui takarannya. Lari
serempak yang akan menghanguskan \omterm{def:g2:measuring-length:units}{bermeter-meter} pemasangan kabel rumah
justru memusnahkan tiga \omterm{def:g2:measuring-length:units}{sentimeter} kawat kurban di dalam selongsong
tahan apinya. Sekring yang putus diganti --- \emph{sesudah}
kerusakannya ditemukan --- dengan yang baru bertakaran sama.
\end{definition}

\begin{example}[Pemutus arus dan sekring, dua pengawal satu tugas]\label{ex:g7:short-circuits-safety:breaker}
Pemutus arus zaman sekarang menjalankan tugas \omterm{def:g7:short-circuits-safety:fuse}{sekringnya} tanpa mati:
sebuah \omterm{ex:g3:first-electric-circuit:switch}{sakelar} berpegas yang menjeplak terbuka pada serbuannya lalu
dipasang ulang dengan sentilan --- sesudah kerusakannya disembuhkan.
\omterm{def:g7:short-circuits-safety:fuse}{Sekring} atau pemutus arus, pos pengawalnya sama: \emph{secara seri},
pada \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang dilindunginya, supaya lari serempaknya harus melewati
pos pemeriksaannya. Pengawal yang terpasang paralel hanya akan
menonton serbuannya menggelegar lewat di jalan yang lain --- hiasan
belaka.
\end{example}

\begin{example}[Kelebihan beban: lari serempak yang sopan]\label{ex:g7:short-circuits-safety:overload}
Tidak setiap serbuan yang berbahaya berupa \omterm{def:g7:short-circuits-safety:device}{korsleting}. Sebuah
stopkontak jamak yang memberi makan pemanas, cerek, dan pemanggang roti
sekaligus meminta satu \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} membawa tiga arak-arakan penuh
bersama --- tiap perkakasnya tidak bersalah, tetapi jumlahnya adalah
kelebihan beban yang menghangatkan kabel dindingnya seperti pemanggang
roti yang lambat. Pemutus arusnya turun karena jumlahnya, dan memang
sepatutnya. Aturan rumah: satu perkakas pembuat panas yang rakus tiap
stopkontak; stopkontak jamaknya untuk lampu dan pengisi daya, bukan
untuk dapur pribadi.
\end{example}

\begin{remark}[Perbaikan yang terlarang]\label{rem:g7:short-circuits-safety:nail}
Cerita lamanya benar dan masih membunuh: \omterm{def:g7:short-circuits-safety:fuse}{sekring} sebuah rumah tangga
putus terus, lalu seseorang ``memperbaikinya'' dengan sebatang paku
atau sekeping koin --- penghantar kekar yang tidak akan pernah
melebur. Gejalanya dibungkam; penyakitnya mengamuk terus. Lari serempak
yang berikutnya tidak bertemu pos pemeriksaan mana pun, dan kawat
rumahnya sendiri menjadi \omterm{def:g7:short-circuits-safety:fuse}{sekringnya} --- di dalam dindingnya. Jangan
pernah melewati seorang pengawal: \omterm{def:g7:short-circuits-safety:fuse}{sekring} yang putus terus adalah
pengawal yang berkata benar terus.
\end{remark}

\begin{remark}[Aturannya, dalam bentuk akhir]\label{rem:g7:short-circuits-safety:rules}
Kitab hukum listrik keluarganya, dengan alasannya kini lengkap: jangan
pernah menjembatani kutub sebuah \omterm{def:g3:first-electric-circuit:battery}{baterai}, dan jauhkan kunci pas dari
aki mobil (\omterm{def:g7:short-circuits-safety:device}{korsleting} \omterm{def:g3:first-electric-circuit:battery}{baterai} di dalam logam adalah busur las); satu
perkakas yang rakus panas tiap stopkontak (kelebihan beban); kabel
yang rusak dipensiunkan (selubung yang berjumbai adalah \omterm{def:g7:short-circuits-safety:device}{korsleting} yang
sedang berlatih); air dan listrik rumah dipisahkan, tangan selalu
kering (tubuhmu tidak boleh menjadi jalan pintasnya); pengawalnya
dihormati --- hanya \omterm{def:g7:short-circuits-safety:fuse}{sekring} yang cocok, kerusakannya ditemukan sebelum
dipasang ulang; dan stopkontaknya sendiri, untuk selamanya, bukan
untuk jari atau percobaan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:1}
Nyatakan proposisi jalan tanpa upayanya. Mengapa kawat telanjang
memenangkan arak-arakannya dari sebuah lampu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:2}
Bedakan \omterm{def:g7:short-circuits-safety:device}{korsleting} sebuah alat dari \omterm{def:g7:short-circuits-safety:device}{korsleting} \omterm{def:g3:first-electric-circuit:battery}{baterainya}. Yang mana
yang lebih berat, dan mengapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:3}
Pada peragaan yang aman itu, mengapa \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 mati --- dan mengapa
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 menjadi lebih terang pada saat yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:4}
Mengapa \omterm{def:g7:short-circuits-safety:device}{korsleting} memanaskan kawatnya? Alat sehari-hari apa yang
memakai penghangatan yang sama dengan sengaja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:5}
Apa itu \omterm{def:g7:short-circuits-safety:fuse}{sekring}, dan apa yang dikurbankannya? Mengapa ia harus duduk
secara seri dengan apa yang dijaganya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:6}
Bagaimana pemutus arus menyempurnakan \omterm{def:g7:short-circuits-safety:fuse}{sekringnya}? Apa yang harus
terjadi sebelum salah satu pengawalnya dipasang ulang atau diganti?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:7}
Sebuah pemanas, cerek, dan pemanggang roti berbagi satu stopkontak
jamak. Tidak ada kawat yang telanjang, tidak ada alat yang rusak ---
tetapi pemutus arusnya turun. Sebutkan peristiwanya lalu jelaskan
keluhan \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:8}
Mengapa ``memperbaiki'' \omterm{def:g7:short-circuits-safety:fuse}{sekring} dengan sebatang paku termasuk perbuatan
yang paling berbahaya di dalam sebuah rumah? Apa yang menjadi
\omterm{def:g7:short-circuits-safety:fuse}{sekringnya} sesudah itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:9}
Kedua kawat dalam sebuah kabel, dengan selubungnya yang berjumbai,
saling bersentuhan di dalam kabelnya --- di hulu lampu yang diberinya
makan. Buatlah diagnosisnya: \omterm{def:g7:short-circuits-safety:device}{korsleting} jenis yang mana, apa yang
terjadi pada lampunya, dan apa yang dilakukan pengawal \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:10}
Pada peragaan berbohlam dua itu, ramalkan apa yang terjadi kalau kawat
pintasnya justru dibaringkan melintang pada \emph{kedua} \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}
sekaligus --- dari sebelum \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 sampai sesudah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2. Batasan
yang mana yang kini berlaku, dan mengapa pelaku percobaannya tidak
boleh mencobanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:11}
Misteri pagi hari raya: untaian lampu peri seri milik nenek mempunyai
satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang kawat penyangga dalamnya melebur menyatu ketika mati
--- sehingga mengorsletingkan dirinya sendiri. Berlawanan dengan
tragedi yang biasa, \emph{sisa untaiannya tetap bersinar}, sedikit
lebih terang. Jelaskan --- lalu katakan apa yang kini sedang
perlahan-lahan disiapkan bagi \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang tersisa.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:short-circuits-safety:12}
Rancanglah rencana perlindungan sebuah bengkel kecil: tiga \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
paralel (lampu; perkakas bertenaga listrik; sebuah pengisi aki), satu
sumber. Tempatkan pengawal dan \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}; pilihlah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} mana yang
layak mendapat pengawalnya sendiri dan mengapa; lalu tulislah
pengumuman dua baris di atas laci \omterm{def:g7:short-circuits-safety:fuse}{sekringnya} yang meringkas
\cref{rem:g7:short-circuits-safety:nail}.
\end{exercise}

\section{Soal: Laporan Penyelidik Kebakaran}

\begin{problem}[{Soal akhir pekan --- sesudah kebakaran apartemennya;
penyelidiknya membaca pengakuan pemasangan kabelnya; tiga kerusakan,
satu pelajaran}]\label{pb:g7:short-circuits-safety:1}
Tidak ada yang terluka --- keluarganya sedang pergi --- tetapi dinding
dapurnya hangus, dan penyelidiknya harus menyusun ulang malam itu dari
buktinya. Kamu membawa buku catatannya.

\medskip\noindent\textbf{Bagian I --- Membaca puingnya.}
\begin{enumerate}
  \item Barang bukti satu: sebuah stopkontak jamak yang memberi makan
        sebuah pemanas, sebuah cerek, dan sebuah penggoreng dalam,
        dengan plastiknya separuh meleleh. Tidak ada kawat telanjang
        di mana pun padanya. Sebutkan kerusakan yang disarankan barang
        bukti ini, dan mengapa tidak diperlukan ``kawat penjahat'' apa
        pun untuknya.
  \item Barang bukti dua: sebuah kabel di balik lemari, yang digigiti
        seekor binatang peliharaan, dengan kedua penghantarnya
        bersentuhan pada bekas gigitannya. Sebutkan kerusakan ini,
        lalu nyatakan ke mana arak-arakannya pergi sejak saat
        persentuhannya.
  \item Barang bukti tiga, yang memberatkan: di dalam kotak
        \omterm{def:g7:short-circuits-safety:fuse}{sekringnya}, selongsong \omterm{def:g7:short-circuits-safety:fuse}{sekring} \omterm{def:g5:series-parallel-circuits:parallel}{cabang} dapurnya memuat ---
        segulung kerat kertas timah dapur. Nyatakan apa yang dilakukan
        ``perbaikan'' ini terhadap perlindungan \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}.
  \item Susunlah ketiga barang buktinya menjadi cerita yang masuk
        akal: kerusakan mana yang datang lebih dahulu, apa yang
        dilakukan \omterm{def:g7:short-circuits-safety:fuse}{sekring} yang sejati terhadapnya (sebelum barang
        bukti tiga), dan mengapa malam yang terakhir berakhir dengan
        kebakaran alih-alih dengan dapur yang gelap?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Kesaksian fisikanya.}
\begin{enumerate}[resume]
  \item Pengadilannya bertanya: mengapa kawat yang terkorsleting atau
        kelebihan beban \emph{memanas}? Sebutkan alat sehari-hari yang
        membuktikan bahwa penghangatan oleh arus dapat dijinakkan ---
        lalu apa yang tidak ada di sini.
  \item Mengapa \omterm{def:g7:short-circuits-safety:fuse}{sekring} sengaja dibuat tipis? Jelaskan ungkapan
        ``pengawal yang mati untukmu''.
  \item Kerat kertas timahnya menghantar dengan hebat. Justru
        \emph{karena} itulah ia gagal sebagai pengawal. Pecahkan
        paradoks kecil itu untuk juri.
  \item Pemanasnya sendirian bertakaran aman, cereknya sendirian
        aman, penggorengnya sendirian aman. Tulislah satu kalimat
        tentang bagaimana tiga keamanan terjumlah menjadi satu bahaya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Putusan dan
pelajarannya.}
\begin{enumerate}[resume]
  \item Penyelidiknya harus mengurutkan ketiga kerusakannya menurut
        bobot moralnya: kelebihan bebannya, kabel yang digigitnya,
        kertas timahnya. Yang mana yang mengubah kemalangan menjadi
        malapetaka, dan mengapa?
  \item Keluarganya bertanya apa yang akan terjadi pada malam
        kebakarannya seandainya \omterm{def:g7:short-circuits-safety:fuse}{sekring} yang sepatutnya terpasang.
        Tuntunlah mereka melewatinya, menit demi menit.
  \item Sarankan pemasangan kabel ulangnya: bagaimana seharusnya
        pemanas, cerek, dan penggorengnya diberi makan sekarang, dan
        aturan tetap apa yang mengatur masa depan stopkontak jamaknya?
  \item Tutuplah laporannya dengan ringkasan tiga kalimat milik
        penyelidiknya: cara kerja sang penjahat, tugas sang pengawal,
        dan satu perbaikan yang tidak boleh pernah dikarang mendadak.
\end{enumerate}
\end{problem}
