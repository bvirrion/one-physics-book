\chapter{Fase Bulan}\label{ch:g7:moon-phases}

Bertahun-tahun yang lalu kamu menyimpan buku harian \omterm{def:g2:moon-in-the-sky:moon}{Bulan} lalu
menonton bentuk peraknya berbaris dari \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} menjadi purnama lalu
kembali, dengan ``mengapa''-nya dijanjikan untuk nanti. Nanti itu
sekarang. Kamu memiliki segala yang diperlukan penjelasannya: \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
yang mengorbit tiap bulan, \omterm{def:g1:light-and-shadows:source}{cahaya} matahari menurut garis lurus, dan
sebuah bola yang selalu --- selalu --- separuh tersinari.

\section{Satu kenyataan yang ada di bawahnya}

\begin{proposition}[Separuh tersinari, selalu]\label{prop:g7:moon-phases:halflit}
Matahari menyinari separuh \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pada setiap saat --- separuh yang
menghadapnya --- persis seperti ia menyinari separuh Bumi. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} tidak
mempunyai bentuk yang berubah-ubah, tidak tumbuh dan tidak \omterm{def:g7:moon-phases:phases}{menyusut}:
ia bola batu yang mengenakan sisi siang yang tetap dan sisi malam yang
tetap, dengan batasnya meluncur selagi ia mengorbit. Segala yang kita
sebut ``\omterm{def:g7:moon-phases:phases}{fase}'' sekadar \emph{pemandangannya}: berapa banyak sisi
siangnya yang kebetulan menghadap Bumi.
\end{proposition}

\begin{definition}[Fase]\label{def:g7:moon-phases:phases}
\emph{Fase}\index{fase} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} adalah bentuknya yang tampak sepanjang
bulan: \emph{bulan baru} (sisi siangnya berpaling sepenuhnya --- kita
tidak melihat apa-apa), \emph{\omterm{ex:g2:moon-in-the-sky:shapes}{sabit}}, \emph{perempat pertama}
(separuh cakramnya tersinari), \emph{cembung} (lebih daripada
separuh), \emph{purnama} (sisi siangnya berpaling sepenuhnya ke arah
kita), lalu cembung, \emph{perempat terakhir} dan \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} lagi, yang
menutup daurnya dalam kira-kira $29.5$ hari. Dari bulan baru sampai
purnama bagian yang tersinari tumbuh malam demi malam: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} sedang
\emph{membesar}\index{membesar}; dari purnama kembali ke bulan baru ia
menyusut: sedang \emph{menyusut}\index{menyusut}.
\end{definition}

\begin{method}[Jadilah Bumi, peganglah Bulan]\label{met:g7:moon-phases:walk}
Penjelasannya, dipentaskan dalam semenit:
\begin{enumerate}
  \item di dalam ruangan gelap, sebuah lampu telanjang tunggal
        menjadi Matahari; kamu menjadi Bumi; sebuah bola putih di
        atas tanganmu yang terjulur menjadi \omterm{def:g2:moon-in-the-sky:moon}{Bulan};
  \item menghadaplah ke lampunya, dengan bolanya ke arah lampu:
        separuh bola yang tersinari menghadap menjauhimu --- kamu
        melihat sisi gelapnya: bulan baru;
  \item berputarlah perlahan di tempat (bolanya mengorbitmu):
        serpihan yang tersinari muncul di tepi bolanya --- \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} yang
        \omterm{def:g7:moon-phases:phases}{membesar}; sesudah seperempat putaran, separuh bolanya tampak
        tersinari: perempat pertama;
  \item teruslah berputar: cembung --- lalu, dengan lampu di
        punggungmu, bolanya tersinari sepenuhnya di depanmu: purnama;
        selesaikan putarannya menembus separuh dan \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} yang
        \omterm{def:g7:moon-phases:phases}{menyusut} sampai ke bulan baru.
\end{enumerate}
Satu putaranmu sendiri memutar ulang satu bulan langit.
\end{method}

\section{Salah pahamnya, dipensiunkan}

\begin{proposition}[Fase bukan bayang-bayang Bumi]\label{prop:g7:moon-phases:notshadow}
Jawaban keliru yang paling populer di dalam ilmu falak: ``\omterm{def:g7:moon-phases:phases}{fasenya}
adalah \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} Bumi pada \omterm{def:g2:moon-in-the-sky:moon}{Bulan}.'' Geometri membantahnya dengan
tiga cara. \omterm{def:g1:light-and-shadows:shadow}{Bayang-bayang} Bumi menunjuk \emph{menjauhi} Matahari ---
ia hanya dapat menyentuh \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang berdiri berseberangan dengan
Matahari, dan itu terjadi semata-mata pada \omterm{ex:g2:moon-in-the-sky:shapes}{bulan purnama} (dan pun
hanya pada gerhana yang jarang). \omterm{def:g2:moon-in-the-sky:moon}{Bulan} \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} berdiri di sisi
\emph{matahari} dari kita, sama sekali tidak dekat \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} kita.
Dan tepi \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} itu pada \omterm{def:g2:moon-in-the-sky:moon}{Bulan} selalu berupa bagian sebuah
lingkaran besar --- ia tidak akan pernah dapat memahat batas bentuk
cembungnya yang melengkung dua kali. \omterm{def:g7:moon-phases:phases}{Fasenya} tidak memerlukan
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}: sebuah bola yang separuh tersinari, dilihat dari sisi
yang berbeda-beda, melakukan segalanya.
\end{proposition}

\begin{example}[Gerhana lawan fase]\label{ex:g7:moon-phases:eclipsevs}
Simpanlah kedua penggelapan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} itu di laci yang terpisah.
\emph{\omterm{def:g7:moon-phases:phases}{Fase}}: daur bulanan yang lambat, sisi malam \omterm{def:g2:moon-in-the-sky:moon}{Bulan} sendiri yang
berpaling ke arah kita --- tanpa melibatkan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} kita.
\emph{\omterm{def:g7:shadows-eclipses:lunar}{Gerhana bulan}}: sejam yang jarang ketika \omterm{ex:g2:moon-in-the-sky:shapes}{Bulan purnama}
menyeberangi \omterm{prop:g7:shadows-eclipses:umbra}{umbra} kita --- \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} Bumi yang sungguh jatuh
padanya, kemerahan tembaga dan cepat. Irama buku hariannya adalah
sebuah pemandangan; gerhananya adalah sebuah kunjungan.
\end{example}

\section{Membaca langit seperti membaca jam}

\begin{example}[Di mana fasenya menepati jamnya]\label{ex:g7:moon-phases:hours}
Diagramnya memakukan tiap \omterm{def:g7:moon-phases:phases}{fase} pada sebuah kedudukan --- dan tiap
kedudukan pada sebuah jadwal. \omterm{ex:g2:moon-in-the-sky:shapes}{Bulan purnama} berdiri berseberangan
dengan Matahari: ia pasti terbit ketika \omterm{def:g1:day-and-night:daynight}{matahari terbenam}, bertakhta
sepanjang malam, lalu terbenam ketika \omterm{def:g1:day-and-night:daynight}{matahari terbit}. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru
bepergian \emph{bersama} Matahari: naik sepanjang siang, hilang di
dalam silaunya --- itulah sebabnya tidak seorang pun ``melihat''
bulan baru. Perempat pertama tertinggal seperempat putaran di belakang
Matahari: ia terbit menjelang tengah hari lalu terbenam menjelang
tengah malam --- \omterm{def:g2:moon-in-the-sky:moon}{Bulan} sehabis makan malam. Kejutan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} siang hari
masa kecilmu kini menjadi sebuah jadwal: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} mana pun yang bukan
bulan baru dan bukan purnama berbagi sebagian gilirannya dengan
Matahari.
\end{example}

\begin{example}[Ke mana sabitnya menunjuk?]\label{ex:g7:moon-phases:points}
Sisi yang tersinari pada \omterm{def:g7:moon-phases:phases}{fase} mana pun menghadap Matahari --- \omterm{ex:g2:moon-in-the-sky:shapes}{sabitnya}
adalah busur panah yang dibidikkan pada pemanahnya. \omterm{ex:g2:moon-in-the-sky:shapes}{Sabit} senja di
barat: tanduknya menunjuk ke atas dan menjauhi nyala \omterm{def:g1:day-and-night:daynight}{matahari terbenam}
di bawah kaki langit. Langit tidak pernah menggambarnya keliru; pelukis
sering keliru. \omterm{ex:g2:moon-in-the-sky:shapes}{Sabit} dengan tanduknya menghadap Matahari adalah
permainan cari-kesalahan kesayangan ilmu falak.
\end{example}

\begin{remark}[Wajah yang tidak pernah berpaling]\label{rem:g7:moon-phases:sameface}
Satu pengamatan buku harian lagi, akhirnya dihormati: bercak gelap
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} membentuk wajah yang sama tiap malam, \omterm{def:g7:moon-phases:phases}{fase} demi \omterm{def:g7:moon-phases:phases}{fase}, tahun demi
tahun. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berputar tepat sekali pada \omterm{def:g6:sun-earth-moon:axisorbit}{sumbunya} tiap \omterm{def:g6:sun-earth-moon:axisorbit}{orbit} ---
sehingga separuh yang sama menghadap Bumi selamanya. Tidak seorang pun
melihat sisi jauhnya sampai sebuah wahana antariksa terbang
mengelilinginya dengan sebuah kamera, masih dalam ingatan hidup
kakek-nenekmu. Mengapa putaran dan \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} sepakat begitu sempurna
adalah kisah pasang surut yang mendalam, yang disimpan untuk
tahun-tahun perguruan tinggi pada pelajaran ini.
\end{remark}

\section{Soal: Sebulan Bersama Bulan}

\begin{problem}[{Soal akhir pekan --- bulan terbuka di
observatoriumnya; tiga puluh malam bersama Bulan, satu bagan dinding;
kuis sang pemandu}]\label{pb:g7:moon-phases:1}
Observatorium kotanya menjalankan acara umum ``Sebulan Bersama
\omterm{def:g2:moon-in-the-sky:moon}{Bulan}''. Sebagai pemandu muda, kamu menjaga bagan dindingnya lalu
menjawab para pengunjungnya.

\medskip\noindent\textbf{Bagian I --- Menyiapkan bagannya.}
\begin{enumerate}
  \item Malam 0 adalah bulan baru. Isilah tonggak bagannya: kira-kira
        pada malam yang mana jatuh perempat pertama, purnama, dan
        perempat terakhir?
  \item Di samping tiap tonggaknya, bagannya menghendaki waktu
        terbit. Pasangkanlah: terbit ketika \omterm{def:g1:day-and-night:daynight}{matahari terbenam}; \
        terbit menjelang tengah hari; \ terbit menjelang tengah
        malam; \ terbit dan terbenam bersama Matahari.
  \item Seorang pengunjung bertanya mengapa bagannya menunjukkan
        \omterm{ex:g2:moon-in-the-sky:shapes}{sabitnya} rendah di langit senja sebelah \emph{barat} tetapi
        purnamanya memanjat dari sebelah \emph{timur} ketika senja.
        Jelaskan dengan kedudukan \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya}.
  \item Pengunjung yang lain bersumpah bahwa mereka melihat \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
        pada pukul sepuluh pagi. \omterm{def:g7:moon-phases:phases}{Fase} bagan yang mana yang
        memungkinkan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pagi hari?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Peragaan sang
pemandu.}
\begin{enumerate}[resume]
  \item Kamu mementaskan jalan lampu-dan-bola untuk sekelompok murid
        sekolah. Seorang murid membekukanmu di tengah putaran:
        bolanya tampak tersinari tiga perempat. Sebutkan \omterm{def:g7:moon-phases:phases}{fasenya},
        lalu tentukan letak lampunya, kamu-sebagai-Bumi, dan bolanya
        untuk kelompoknya.
  \item ``Jadi \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} Bumi yang membuat \omterm{def:g7:moon-phases:phases}{fasenya}?'' tanya
        muridnya, sambil menunjuk bolamu yang bersisi gelap. Berikan
        bantahan milik peragaannya sendiri: sisi malam siapa yang
        sedang mereka lihat, dan di mana sebenarnya \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}
        Bumi berada?
  \item Kelompoknya bertanya mengapa mereka selalu melihat
        ``kelincinya'' (atau ``wajahnya'') di \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, tidak pernah
        punggungnya. Jawablah dengan kenyataan wajah yang sama.
  \item Seorang orang tua yang meragukan: ``Kalau \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berputar,
        seperti yang kamu katakan, bukankah kita seharusnya melihatnya
        berputar?'' Pecahkanlah pertentangan yang tampak itu:
        bagaimana \omterm{def:g2:moon-in-the-sky:moon}{Bulan} dapat berputar tepat sekali tiap \omterm{def:g6:sun-earth-moon:axisorbit}{orbit}
        sehingga justru \emph{menyembunyikan} putarannya dari kita?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Malam kuis.}
\begin{enumerate}[resume]
  \item Kartu kuis A: ``\omterm{def:g2:moon-in-the-sky:moon}{Bulan} malam ini terbit pada tengah malam lalu
        memperlihatkan separuh \emph{kirinya} tersinari (sebagaimana
        terlihat menjelang pagi). \omterm{def:g7:moon-phases:phases}{Membesar} atau \omterm{def:g7:moon-phases:phases}{menyusut} --- dan
        perempat yang mana?''
  \item Kartu kuis B: ``Sebuah poster seniman menunjukkan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} \omterm{ex:g2:moon-in-the-sky:shapes}{sabit}
        pada tengah malam, tinggi di atas kepala, dengan tanduknya
        menunjuk lurus ke bawah ke arah warna \omterm{def:g1:day-and-night:daynight}{matahari terbenam} di
        kaki langit.'' Sebutkan setiap kesalahan ilmu falak pada
        posternya.
  \item Kartu kuis C: ``Selama \omterm{ex:g2:moon-in-the-sky:shapes}{bulan purnama} malam ini, dapatkah para
