\chapter{Bayang-Bayang dan Gerhana}\label{ch:g7:shadows-eclipses}

Sekali atau dua kali seumur hidup, selama beberapa menit, siang
menjadi malam: cakram Matahari menghitam di balik \omterm{def:g2:moon-in-the-sky:moon}{Bulan} dan \omterm{def:g4:solar-system:star}{bintang}
muncul pada tengah hari. Bangsa purba membaca kiamat di dalamnya; kamu
akan membaca geometri. Gerhana tidak lain daripada \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} ---
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} terbesar yang akan pernah kamu diami --- dan perlengkapan
penggaris-dan-sinarmu tepat menjadi apa yang ditaatinya.

\section{Bayang-bayang dari lampu yang jujur}

\begin{proposition}[Umbra dan penumbra]\label{prop:g7:shadows-eclipses:umbra}
Sumber yang \emph{serupa titik} melemparkan satu \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}
bertepi tajam. Sumber yang \emph{lebar} --- sebuah bola kaca beku,
seluruh cakram Matahari --- melemparkan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} berbagian dua:
\begin{enumerate}
  \item \emph{umbra}\index{umbra}: daerah yang dicapai \omterm{def:g1:light-and-shadows:source}{cahaya} dari
        \emph{tidak satu} bagian pun sumbernya --- kegelapan penuh;
  \item \emph{penumbra}\index{penumbra}: pinggiran yang dicapai
        \omterm{def:g1:light-and-shadows:source}{cahaya} dari \emph{sebagian} sumbernya saja --- setengah
        \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}, yang memudar ke arah luar.
\end{enumerate}
Dari dalam \omterm{prop:g7:shadows-eclipses:umbra}{umbranya}, sumbernya tersembunyi seluruhnya; dari
\omterm{prop:g7:shadows-eclipses:umbra}{penumbranya}, tersembunyi sebagian --- mengintip di atas tepi
penghalangnya.
\end{proposition}

\begin{method}[Membangun kedua bayang-bayangnya]\label{met:g7:shadows-eclipses:construct}
Untuk sumber yang lebar, sinar dari \emph{tepinya} yang memerintah
petanya:
\begin{enumerate}
  \item gambarlah sinar dari tepi atas sumbernya yang menyerempet
        kedua sisi penghalangnya, dan sinar yang sama dari tepi
        bawahnya;
  \item daerah di belakang penghalangnya yang diluputkan sinar
        \emph{kedua} tepinya adalah \omterm{prop:g7:shadows-eclipses:umbra}{umbranya};
  \item daerah di sampingnya yang diluputkan satu tepi tetapi dicapai
        tepi yang lain adalah \omterm{prop:g7:shadows-eclipses:umbra}{penumbranya};
  \item di tempat semua sinarnya tiba, \omterm{def:g1:light-and-shadows:source}{cahaya} penuh.
\end{enumerate}
\omterm{prop:g7:shadows-eclipses:umbra}{Umbra} sebuah penghalang bundar berupa kerucut --- dan apakah kerucut
itu mencapai sebuah layar tertentu memutuskan segalanya tentang
gerhana.
\end{method}

\begin{example}[Tepi yang lembut di mana-mana]\label{ex:g7:shadows-eclipses:soft}
Kini \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} yang mengabur dari lampu beku dan langit berawan
pun terjelaskan: sumber yang lebar membungkus setiap \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang}
dengan \omterm{prop:g7:shadows-eclipses:umbra}{penumbra}, dan sumber yang cukup besar (seluruh langit yang
mendung) seluruhnya \omterm{prop:g7:shadows-eclipses:umbra}{penumbra} --- \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} terluntur sampai
tiada. Tepi \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah harimu sendiri, kalau dilihat dari
dekat, berupa keliman \omterm{prop:g7:shadows-eclipses:umbra}{penumbra} yang lembut selebar sesentimeter:
Matahari berupa cakram, bukan titik, dan keliman itu pengakuannya.
\end{example}

\section{Bulan melangkah ke dalam berkasnya}

\begin{definition}[Gerhana matahari]\label{def:g7:shadows-eclipses:solar}
\emph{Gerhana matahari}\index{gerhana matahari} terjadi ketika \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
lewat tepat di antara Matahari dan Bumi, lalu \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} jatuh
pada kita. Di tempat \emph{\omterm{prop:g7:shadows-eclipses:umbra}{umbra}} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menyentuh Bumi --- sebuah
bintik yang melesat yang jarang selebar dua ratus kilometer --- cakram
Matahari tersembunyi seluruhnya: gerhana \emph{total}, malam pada
tengah hari, \omterm{def:g4:solar-system:star}{bintang} bermunculan. Di dalam \emph{\omterm{prop:g7:shadows-eclipses:umbra}{penumbra}} luas yang
mengelilinginya, Matahari hanya tergigit sebagian: gerhana
\emph{sebagian}.
\end{definition}

\begin{definition}[Gerhana bulan]\label{def:g7:shadows-eclipses:lunar}
\emph{Gerhana bulan}\index{gerhana bulan} terjadi ketika \emph{\omterm{def:g2:moon-in-the-sky:moon}{Bulan}}
berlayar masuk ke kerucut \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} \emph{Bumi}: \omterm{ex:g2:moon-in-the-sky:shapes}{bulan purnamanya}
menggelap --- sering sampai menjadi nyala tembaga yang pekat ---
selama sejam atau lebih, dan terlihat oleh seluruh belahan malam
dunianya sekaligus. Kali ini kitalah penghalangnya, dan \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berdiri
di dalam \omterm{prop:g7:shadows-eclipses:umbra}{umbra} kita.
\end{definition}

\begin{example}[Mengapa begitu jarang?]\label{ex:g7:shadows-eclipses:rare}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru datang tiap bulan --- mengapa bukan \omterm{def:g7:shadows-eclipses:solar}{gerhana matahari} tiap
bulan? Karena \omterm{def:g6:sun-earth-moon:axisorbit}{orbit} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} \emph{miring}, kira-kira lima derajat,
terhadap jalur Bumi mengelilingi Matahari. Pada kebanyakan bulan,
bulan barunya lewat \emph{di atas} atau \emph{di bawah} garis
Matahari--Bumi, dan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} menombak tanpa bahaya ke
antariksa. Hanya ketika bulan baru terjadi tepat pada saat \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
menyeberangi bidang kesejajarannya, ketiga dunianya berbaris ---
beberapa kali setahun di suatu tempat di Bumi, dan bagi sebuah kota
\emph{tertentu}, totalitas adalah tamu yang datang sekali dalam
beberapa umur.
\end{example}

\begin{example}[Kebetulan yang membuat totalitas]\label{ex:g7:shadows-eclipses:coincidence}
Model lapangan olahragamu sudah menemukannya: Matahari kira-kira $400$
kali lebih lebar daripada \omterm{def:g2:moon-in-the-sky:moon}{Bulan} --- dan berdiri kira-kira $400$ kali
lebih jauh. Karena itu kedua cakramnya berpadanan di langit dengan
hampir tepat, sehingga \omterm{def:g2:moon-in-the-sky:moon}{Bulan} dapat menutupi Matahari \emph{nyaris
saja}: totalitasnya singkat dan bintik \omterm{prop:g7:shadows-eclipses:umbra}{umbranya} kecil. Ketika \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
berada di ujung jauh \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} yang sedikit lonjong, cakramnya kurang
panjang, dan sebentuk cincin Matahari yang membara membingkai \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
yang hitam --- gerhana \emph{cincin}, indah dan tidak pernah aman
ditonton tanpa pelindung.
\end{example}

\begin{remark}[Menonton tanpa menangis]\label{rem:g7:shadows-eclipses:safety}
Hukum lamanya tidak mempunyai kekecualian gerhana: \emph{jangan
pernah} memandang Matahari --- tergerhana, bercincin, atau utuh ---
dengan mata telanjang, kacamata hitam, atau lensa apa pun. Matahari
yang tersembunyi sebagian membakar selaput jala mata sepasti Matahari
yang utuh, dan tanpa rasa sakit. Alat yang aman adalah kawan lama:
lubang jarum --- proyeksikan \omterm{prop:g5:mirrors-reflection:image}{bayangan} Matahari menembus lubang kecil
ke atas kertas lalu tontonlah gigitannya tumbuh dengan nyaman --- atau
sebatang pohon yang rimbun, yang seribu lubang jarum celah daunnya
menaburi tanah dengan \omterm{ex:g2:moon-in-the-sky:shapes}{sabit-sabit} kecil. Hanya kacamata gerhana
bersertifikat, yang telah diperiksa dan tidak rusak, yang boleh
menghadapi Matahari secara langsung; hanya selama beberapa menit
\emph{totalitasnya}, koronanya yang telanjang boleh dikagumi --- dan
serpihan pertama yang kembali mengakhiri gencatan senjatanya.
\end{remark}

\section{Soal: Ekspedisi Gerhana}

\begin{problem}[{Soal akhir pekan --- kelas merencanakan ekspedisi
gerhananya; geometri memilih tempat berkemahnya, fisika mengemasi
tasnya; totalitas, menit demi menit}]\label{pb:g7:shadows-eclipses:1}
\omterm{def:g3:sun-days-seasons:year}{Musim} semi depan, \omterm{def:g7:shadows-eclipses:solar}{gerhana matahari} total menyeberangi benuanya. Kelas
itu merencanakan sebuah ekspedisi --- dengan diagram sinar, bukan
dengan brosur.

\medskip\noindent\textbf{Bagian I --- Memilih tempat berdiri.}
Atlasnya menunjukkan jalur yang diramalkan: sebuah jalur selebar
\qty{160}{km} yang membentang menyeberangi petanya, diapit daerah
gerhana sebagian yang lebarnya ribuan kilometer.
\begin{enumerate}
  \item Daerah \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} yang mana yang digariskan jalurnya, dan
        daerah mana yang mengapitnya?
  \item Kota kelasnya terletak \qty{300}{km} dari garis tengah
        jalurnya. Apa yang akan dilihat kotanya pada hari gerhananya?
  \item Mengapa ekspedisinya harus bepergian ke jalurnya --- mengapa
        ``sebagian sembilan puluh lima persen'' di rumah tidak akan
        memperlihatkan malam pada tengah hari, \omterm{def:g4:solar-system:star}{bintang}, atau koronanya?
  \item Ramalan jalurnya menandai totalitas pada pukul 10:41, yang
        berlangsung $3$ menit $12$ sekon pada garis tengahnya dan
        kurang dari itu ke arah tepinya. Mengapa totalitasnya paling
        lama di tengah jalurnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Mengemasi fisikanya.}
\begin{enumerate}[resume]
  \item Gurunya mengemasi kacamata gerhana bersertifikat lalu menyita
        dua kacamata hitam. Belalah kedua perbuatannya.
  \item Tiap murid harus membangun proyektor lubang jarum dari sebuah
        kardus sepatu. Ingatlah kembali tata bangunnya, lalu nyatakan
        harus menghadap ke mana ketika berdiri terhadap Matahari.
  \item Seorang murid mengusulkan ``menontonnya pada pantulan seember
        air''. Berilah putusan atas usulnya.
  \item Kelompoknya juga akan membawa sebuah saringan untuk dipegang
        di atas seprai putih. Ramalkan, lalu jelaskan, apa yang akan
        dilukiskan berpuluh-puluh lubangnya selama fase sebagiannya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Harinya, menit demi
menit.}
\begin{enumerate}[resume]
  \item Pukul 09:30, sentuhan pertama: lewat kacamatanya, sebuah
        gigitan kecil muncul di tepi Matahari. Benda apa yang sedang
        mengambil gigitan itu, dan daerah \omterm{def:g1:light-and-shadows:shadow}{bayang-bayangnya} yang mana
        yang sudah mencapai perkemahannya?
  \item Pukul 10:39: \omterm{def:g1:light-and-shadows:source}{cahayanya} berubah menjadi ganjil dan tipis;
        \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} di tanah menjadi bertepi tajam secara aneh.
        Jelaskan penajamannya dengan pemikiran umbra-dan-penumbra ---
        Matahari sedang menjadi apa selagi serpihannya yang terlihat
        menyempit?
  \item Pukul 10:41, totalitas: kacamatanya dilepas, menurut
        aturannya. Selama $3$ menit kelas itu berdiri di dalam
        sesuatu. Sebutkan namanya dengan tepat, lalu nyatakan apa yang
        telah menjadi terlihat mengelilingi cakram hitamnya --- dan
        mengapa gencatan senjatanya berakhir pada serpihan pertama
        yang kembali.
  \item Pukul 12:00, di dalam bus pulang, gurunya meminta ringkasan
        satu kalimat: ``Gerhana adalah\dots'' Lengkapilah dengan
        layak, dengan geometri babnya di dalamnya.
\end{enumerate}
\end{problem}
