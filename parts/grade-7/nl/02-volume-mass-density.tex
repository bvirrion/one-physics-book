\chapter{Volume, massa, dichtheid}\label{ch:g7:volume-mass-density}

Vorig jaar ontmaskerde je een frauduleuze goudsmid door grammen door
\omterm{def:g6:mass-and-volume:volume}{kubieke centimeters} te delen. Dit jaar heeft je wiskundecursus je een
nieuwe macht gegeven: letters die voor getallen staan.
\omterm{def:g7:volume-mass-density:formula}{Dichtheid} is de perfecte plek om die te besteden --- één korte
formule die drie recepten, de gereedschapskist van een detective en de
hele drijfregel in vijf tekens propt.

\section{De formule}

\begin{definition}[Dichtheid, in formulevorm]\label{def:g7:volume-mass-density:formula}
De \emph{dichtheid}\index{dichtheid} $\rho$ (de Griekse letter
\emph{rho}) van een stof is haar \omterm{def:g3:measuring-mass:mass}{massa} per \omterm{def:g6:measurement-in-science:measuring}{eenheid}
\omterm{def:g6:mass-and-volume:volume}{volume}: voor een monster met \omterm{def:g3:measuring-mass:mass}{massa} $m$ en \omterm{def:g6:mass-and-volume:volume}{volume} $V$ geldt
\[
\rho = \frac{m}{V}.
\]
Met $m$ in \omterm{def:g3:measuring-mass:units}{gram} en $V$ in \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} komt $\rho$ eruit in
\emph{\omterm{def:g3:measuring-mass:units}{gram} per \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter}} (\unit{g/cm^3}) --- de
``\omterm{def:g3:measuring-mass:mass}{massa} van één \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter}'' van vorig jaar, nu in
uniform. De dichtheid van water: $\rho = \qty{1.0}{g/cm^3}$. En
aangezien een \omterm{def:g6:mass-and-volume:volume}{liter} $1000$ \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} is, dient hetzelfde
getal ook als \omterm{def:g3:measuring-mass:units}{kilogram} per \omterm{def:g6:mass-and-volume:volume}{liter}: water is \qty{1.0}{kg} per
\omterm{def:g6:mass-and-volume:volume}{liter}.
\end{definition}

\begin{example}[De letters lezen]\label{ex:g7:volume-mass-density:reading}
Een formule is een zin in steno. $\rho = m/V$ leest: ``om de
\omterm{def:g7:volume-mass-density:formula}{dichtheid} te vinden, deel je de \omterm{def:g3:measuring-mass:mass}{massa} van het monster door zijn
\omterm{def:g6:mass-and-volume:volume}{volume}.'' De letters zijn plaatshouders --- voor \emph{deze}
kiezelsteen wordt $m$ gelijk aan \qty{75}{g} en $V$ gelijk aan
\qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
Dezelfde deling als vorig jaar --- maar de formule onthoudt het recept
voor elke kiezel die nog komt.
\end{example}

\begin{proposition}[Drie recepten in één formule]\label{prop:g7:volume-mass-density:recipes}
Doordat \omterm{def:g7:volume-mass-density:formula}{dichtheid} \omterm{def:g3:measuring-mass:mass}{massa} en \omterm{def:g6:mass-and-volume:volume}{volume} evenredig aan elkaar knoopt,
dient één formule drie behoeften:
\begin{enumerate}
  \item \emph{een \omterm{def:g7:volume-mass-density:formula}{dichtheid} vinden}: $\rho = m/V$ --- deel
        \omterm{def:g3:measuring-mass:mass}{massa} door \omterm{def:g6:mass-and-volume:volume}{volume};
  \item \emph{een \omterm{def:g3:measuring-mass:mass}{massa} vinden}: $m = \rho \times V$ --- elke
        \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} draagt $\rho$ \omterm{def:g3:measuring-mass:units}{gram}, en er zijn er $V$;
  \item \emph{een \omterm{def:g6:mass-and-volume:volume}{volume} vinden}: $V = m/\rho$ --- verdeel de
        \omterm{def:g3:measuring-mass:mass}{massa} in porties van $\rho$ \omterm{def:g3:measuring-mass:units}{gram}; het aantal porties is
        het aantal \omterm{def:g6:mass-and-volume:volume}{kubieke centimeters}.
\end{enumerate}
\end{proposition}

\begin{example}[De recepten aan het werk]\label{ex:g7:volume-mass-density:atwork}
\emph{\omterm{def:g3:measuring-mass:mass}{Massa} uit \omterm{def:g6:mass-and-volume:volume}{volume}}: wat weegt \qty{250}{cm^3} ijzer
($\rho = 7.9$)? $m = 7.9 \times 250 = \qty{1975}{g}$ --- bijna twee
\omterm{def:g3:measuring-mass:units}{kilogram} in het bestek van een koffiemok.
\emph{\omterm{def:g6:mass-and-volume:volume}{Volume} uit \omterm{def:g3:measuring-mass:mass}{massa}}: hoeveel ruimte neemt \qty{540}{g}
eikenhout ($\rho = 0.6$) in? $V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{Gezondverstandpartner}: antwoorden voor ijzer horen
zwaar-en-klein te zijn, die voor hout licht-en-groot --- de
dichthedentabel is de beste vriend van de beoordelingsstap.
\end{example}

\section{Dichtheden meten}

\begin{method}[Dichtheid van een vloeistof]\label{met:g7:volume-mass-density:liquid}
Vloeistoffen gaan niet los op een weegschaal zitten --- dus leen de
tarreertruc:
\begin{enumerate}
  \item zet een lege maatcilinder op de balans en tarreer hem op nul;
  \item giet er een handig \omterm{def:g6:mass-and-volume:volume}{volume} in --- zeg $V = \qty{80}{cm^3}$
        van de raadselachtige \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}, met de \omterm{met:g6:mass-and-volume:cylinder}{meniscus} netjes
        afgelezen;
  \item de balans toont nu de \omterm{def:g3:measuring-mass:mass}{massa} van de \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} alleen: zeg
        $m = \qty{68}{g}$;
  \item deel: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ --- lichter
        dan water: waarschijnlijk een olie.
\end{enumerate}
\end{method}

\begin{example}[Dichtheid van een vaste stof, van begin tot eind]\label{ex:g7:volume-mass-density:solid}
Een medaille beweert van zilver te zijn ($\rho = 10.5$). Balans:
$m = \qty{84}{g}$. Cilinder: het water klimt van \qty{60.0}{cm^3} naar
\qty{68.0}{cm^3}, dus $V = \qty{8.0}{cm^3}$. Dan is
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- de handtekening klopt:
zilver zou het best kunnen zijn. (Een goedkoper metaal in een zilveren
jasje zou zich hier hebben verraden --- tenzij het sluw gekozen was,
zoals het weekendprobleem zal laten zien.)
\end{example}

\begin{remark}[Eenheden in formules]\label{rem:g7:volume-mass-density:units}
Een formule is alleen eerlijk als haar eenheden kloppen. Met $m$ in
\omterm{def:g3:measuring-mass:units}{gram} en $V$ in \unit{cm^3} spreekt $\rho$ in \unit{g/cm^3}; voer haar
\omterm{def:g3:measuring-mass:units}{kilogram} en \omterm{def:g6:mass-and-volume:volume}{liter} en ze antwoordt in \unit{kg/L} --- gelukkig voor
elke stof hetzelfde getal. Maar meng \omterm{def:g3:measuring-mass:units}{gram} met \omterm{def:g6:mass-and-volume:volume}{liter} en de formule
levert, zonder klagen, onzin. Regel van de vakmensen: laat vóór het
rekenen de eenheden paraderen; schrijf ze na het rekenen in het
antwoord.
\end{remark}

\section{Denken in dichtheden}

\begin{example}[Legeringen en tussenwaarden]\label{ex:g7:volume-mass-density:alloys}
Meng twee stoffen en de \omterm{def:g7:volume-mass-density:formula}{dichtheid} van het mengsel landt
\emph{tussen} hun handtekeningen --- dichter bij het royaalste
ingrediënt. Brons (koper $9.0$ met een beetje tin) tekent rond $8.8$;
het goud-zilvermengsel van de kroon tekende tussen $10.5$ en $19.3$;
zeewater, zout opgelost in water, kruipt naar ongeveer $1.03$.
Tussenin liggen is zelf een aanwijzing: een aflezing van $12$ bij een
staaf ``zuiver goud'' is een bekentenis van gezelschap.
\end{example}

\begin{example}[Dichtheid beslist de drijvende wereld]\label{ex:g7:volume-mass-density:floating}
De drijfregel, nu in uniform: een voorwerp \omterm{def:g1:floating-and-sinking:words}{drijft} in een
\omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} wanneer
$\rho_{\text{voorwerp}} < \rho_{\text{vloeistof}}$. Eikenhout ($0.6$)
op water ($1.0$): \omterm{def:g1:floating-and-sinking:words}{drijft}. IJs ($0.9$) in olie ($0.85$): \omterm{def:g1:floating-and-sinking:words}{zinkt} ---
raadpleeg de tabel, niet je onderbuik. Een zwemmer ($\approx 1.0$) in
de beroemde ultrazoute meren ($\approx 1.2$): \omterm{def:g1:floating-and-sinking:words}{drijft} als een kurk,
krant in de hand. Eén ongelijkheid, de hele haven.
\end{example}

\begin{remark}[Let op de menigte, niet op het pakket]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ beschrijft \emph{stoffen}; schepen en opgeblazen
zwemvesten zijn \emph{pakketten} --- stof plus opgesloten
\omterm{def:g2:air-around-us:air}{lucht} --- en het is de totale $m/V$ van het pakket die tegenover de
drijfregel staat. De formule kan allebei aan, als je haar de juiste
$m$ en $V$ voert: de eigen \omterm{def:g7:volume-mass-density:formula}{dichtheid} van het staal voor de stof, de
\omterm{def:g3:measuring-mass:mass}{massa} van de hele romp over het \omterm{def:g6:mass-and-volume:volume}{volume} van de hele romp voor het
pakket. De meeste ``drijfparadoxen'' zijn gewoon de twee
boekhoudingen door elkaar gehaald.
\end{remark}

\section{Probleem: de detective van het schroothandelterrein}

\begin{problem}[{Weekendprobleem --- een middag met de metaaldetective
van de schroothandel; vier raadselachtige partijen, één bedrieglijke
staaf; de grenzen van de dichtheidstoets}]\label{pb:g7:volume-mass-density:1}
De schroothandel koopt metaal om wat het \emph{is}, niet om hoe het
eruitziet, en de detective van het terrein werkt met een balans, een
grote maatbak en de skylinetabel. Jij bent de leerling. (Uittreksel
uit de tabel: aluminium $2.7$; ijzer $7.9$; koper $9.0$; lood $11.3$;
goud $19.3$; wolfraam $19.3$.)

\medskip\noindent\textbf{Deel I --- Vier partijen.}
\begin{enumerate}
  \item Partij A, een grijze staaf: $m = \qty{5400}{g}$, verdringing
        $V = \qty{2000}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Dichtheid} en identiteit?
  \item Partij B, een rol draad: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Dichtheid} en identiteit?
  \item Partij C, een doffe zware plaat: $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Dichtheid} en identiteit --- en waarom
        moet partij C met handschoenen en respect behandeld worden?
  \item Partij D, een zak gemengd bleek schroot: $m = \qty{8100}{g}$,
        $V = \qty{1500}{cm^3}$. Laat zien dat de \omterm{def:g7:volume-mass-density:formula}{dichtheid} van partij
        D tussen twee metalen uit de tabel valt, en zeg wat de zak
        hoogstwaarschijnlijk bevat.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Prijzen en voorspellingen.}
Het terrein betaalt per \omterm{def:g3:measuring-mass:mass}{massa} maar plant transport per \omterm{def:g6:mass-and-volume:volume}{volume}.
\begin{enumerate}[resume]
  \item Een koper wil \qty{540}{g} aluminium uit partij A gezaagd.
        Welk \omterm{def:g6:mass-and-volume:volume}{volume} staaf is dat?
  \item De koperen rol van partij B wordt omgesmolten tot kubussen
        van elk \qty{25}{cm^3}. Wat is de \omterm{def:g3:measuring-mass:mass}{massa} van één kubus, en
        hoeveel volle kubussen levert de rol op?
  \item Een krat mag hoogstens \qty{20}{kg} dragen. Hoeveel van die
        koperen kubussen mag hij bevatten?
  \item In de laadbak van de vrachtwagen past nog \qty{3000}{cm^3}
        ijzerschroot. Welke extra \omterm{def:g3:measuring-mass:mass}{massa}, in \omterm{def:g3:measuring-mass:units}{kilogram}, mag de
        detective laden?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De bedrieger.}
Er komt een verkoper met een glanzende ``gouden'' staaf:
$m = \qty{3860}{g}$, verdringing $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:g7:volume-mass-density:formula}{dichtheid} van de staaf. Klopt die met de
        handtekening van goud?
  \item De detective, onbewogen, raadpleegt de laatste regel van de
        tabel en zucht. Welk goedkoper metaal draagt precies de
        \omterm{def:g7:volume-mass-density:formula}{dichtheid} van goud --- en wat leert dat over de grenzen van
        de dichtheidstoets?
  \item Welk van de twee instrumenten van het terrein is hier
        verslagen: de balans, de bak, allebei, of geen van beide? Zeg
        precies wat de dichtheidstoets \emph{wel} eerlijk over de
        staaf heeft vastgesteld.
  \item Stel een verdere natuurkundige toets uit eerdere jaren van
        deze cursus voor die goud van zijn dubbelganger onderscheidt
        zonder de staaf te beschadigen. (Hun \omterm{def:g4:changes-of-state:melting-point}{smeltpunten}
        verschillen enorm --- goud rond \qty{1064}{\celsius}, de
        dubbelganger ver boven elke oven hier --- maar geen
        schroothandel \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} een misschien-schat: zoek zachtere
        aanwijzingen, misschien het oordeel van de \omterm{def:g1:magnets:magnet}{magneet} over
        ijzeren kernen, of de klank van een aangeslagen staaf, en
        verdedig je keuze eerlijk.)
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Het vak van de detective.}
\begin{enumerate}[resume]
  \item Schrijf de geloofsbelijdenis van de detective in drie regels:
        wat \omterm{def:g7:volume-mass-density:formula}{dichtheid} kan bewijzen, wat ze alleen kan suggereren, en
        wat ze in haar eentje nooit kan.
\end{enumerate}
\end{problem}
