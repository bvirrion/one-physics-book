\chapter{Kortsluiting en elektrische veiligheid}\label{ch:g7:short-circuits-safety}

Elke veiligheidsregel die je hebt geleerd voert terug op één schurk die
je vorig jaar kort ontmoette: de \omterm{def:g7:short-circuits-safety:device}{kortsluiting}. Dit hoofdstuk bestudeert
haar grondig --- hoe ze begint, waarom ze brandt, hoe een nederig
opofferend draadje haar verslaat --- en maakt van de zekeringkast van
het gezin een raadselachtige kast tot een bondgenoot die je begrijpt.

\section{De werkwijze van de schurk}

\begin{proposition}[De stroom verkiest de moeiteloze weg]\label{prop:g7:short-circuits-safety:effortless}
Tussen twee wegen die dezelfde twee punten verbinden, dringt de
marcherende stroom zich op de weg die hem het minst tegenwerkt.
Apparaten --- lampen, \omterm{ex:g6:circuits-and-safety:motor}{motoren}, \omterm{ex:g6:circuits-and-safety:motor}{zoemers} --- werken de mars alle tegen
terwijl ze hun werk doen; een gewone dikke draad werkt hem zo goed als
niet tegen. Bied de mars een omleiding van blote draad, en hij laat de
werkende weg vrijwel volledig in de steek voor de moeiteloze.
\end{proposition}

\begin{definition}[Kortsluiting van een apparaat]\label{def:g7:short-circuits-safety:device}
Een apparaat is \emph{kortgesloten}\index{kortsluiting} wanneer een
geleidende weg zijn twee aansluitingen rechtstreeks verbindt en het
zo overslaat. De mars laat het apparaat in de steek --- een
kortgesloten lamp gaat uit, een \omterm{ex:g6:circuits-and-safety:motor}{motor} stopt --- terwijl de rest van
de lus doorgaat, nu met één werker minder die de mars tegenwerkt.
\end{definition}

\begin{definition}[Kortsluiting van de batterij]\label{def:g7:short-circuits-safety:battery}
Het ernstigste geval: een geleidende weg verbindt de twee
polen van de \emph{\omterm{def:g3:first-electric-circuit:battery}{batterij}} zonder ergens een apparaat erop.
Niets werkt de mars nog tegen, en hij wordt een stormloop --- de wilde
stroom verhit de draad en de \omterm{def:g3:first-electric-circuit:battery}{batterij} zelf, verwoest cellen,
\omterm{def:g2:ice-water-steam:meltfreeze}{smelt} isolatie en zet op het net huizen in brand.
\end{definition}

\begin{method}[De schurk, veilig gedemonstreerd]\label{met:g7:short-circuits-safety:demo}
Met één \omterm{def:g3:first-electric-circuit:battery}{batterij} en twee \omterm{def:g3:first-electric-circuit:bulb}{lampjes} \omterm{def:g4:series-circuits:series}{in serie} (veilig: de duw van
de \omterm{def:g3:first-electric-circuit:battery}{batterij} is minuscuul):
\begin{enumerate}
  \item laat beide \omterm{def:g3:first-electric-circuit:bulb}{lampjes} branden --- even zwak, zoals de seriewet
        belooft;
  \item leg nu een gewone draad over de twee aansluitingen van
        \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2 en sla het zo over;
  \item kijk: \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2 sterft onmiddellijk --- in de steek gelaten
        --- terwijl \omterm{def:g3:first-electric-circuit:bulb}{lampje} 1 \emph{feller} opspringt: met één
        tegenwerker overgeslagen versterkt de mars van de hele lus;
  \item haal de draad weg: de mars gaat \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2 weer binnen, en de
        gelijke zwakte keert terug.
\end{enumerate}
Eén proef, drie lessen: de omleidingsregel, het verlaten apparaat en
de versterkte stormloop.
\end{method}

\begin{example}[Waarom de draden verhitten]\label{ex:g7:short-circuits-safety:heating}
Een marcherende stroom verwarmt zijn weg altijd een beetje --- de
gloed van de lamp \emph{is} die verwarming, met opzet aan het werk
gezet in een draadje dat gebouwd is om witheet te lopen. In een
stormloop wordt de verwarming een bedreiging: de stroom van een
\omterm{def:g7:short-circuits-safety:device}{kortsluiting} kan een koperdraad in \omterm{def:g2:measuring-time:second}{seconden} voorbij het
\omterm{def:g4:changes-of-state:melting-point}{smeltpunt} van zijn plastic mantel verhitten. De voorproef op
zakformaat: een \omterm{def:g3:first-electric-circuit:battery}{batterij} die met een draad wordt \omterm{def:g7:short-circuits-safety:device}{kortgesloten} wordt warm
in je hand --- dezelfde natuurkunde die op netschaal muren verkoolt.
(Hoe de verwarming met de stormloop groeit, wordt over twee jaar
eerlijk gemeten, in het hoofdstuk over elektrisch vermogen.)
\end{example}

\section{De lijfwachten}

\begin{definition}[Zekering]\label{def:g7:short-circuits-safety:fuse}
Een \emph{zekering}\index{zekering} is een lijfwacht die voor je
sterft: een met opzet dun draadje, \omterm{def:g4:series-circuits:series}{in serie} met de kring die het
bewaakt, zo gekozen dat het \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} --- en zo de lus doorsnijdt ---
zodra de stroom zijn waarde overschrijdt. De stormloop die
\omterm{def:g2:measuring-length:units}{meters} huisbedrading zou verkolen, verwoest in plaats daarvan drie
\omterm{def:g2:measuring-length:units}{centimeter} opofferingsdraad in zijn brandwerende patroon. Een
gesprongen zekering wordt --- \emph{nadat} de storing gevonden is ---
vervangen door een nieuwe van dezelfde waarde.
\end{definition}

\begin{example}[Onderbreker en zekering, twee wachters één taak]\label{ex:g7:short-circuits-safety:breaker}
De moderne stroomonderbreker doet het werk van de \omterm{def:g7:short-circuits-safety:fuse}{zekering} zonder te
sterven: een veerbelaste \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} die bij de stormloop openklapt en
met een tik teruggezet wordt --- nadat de storing verholpen is.
\omterm{def:g7:short-circuits-safety:fuse}{Zekering} of onderbreker, de post van de wachter is dezelfde:
\emph{\omterm{def:g4:series-circuits:series}{in serie}}, op de \omterm{def:g5:series-parallel-circuits:parallel}{tak} die hij beschermt, zodat de
stormloop door zijn controlepost moet. Een wachter \omterm{def:g5:series-parallel-circuits:parallel}{parallel} zou de
stormloop op de andere weg voorbij zien donderen --- pure versiering.
\end{example}

\begin{example}[Overbelasting: de beleefde stormloop]\label{ex:g7:short-circuits-safety:overload}
Niet elke gevaarlijke stormloop is een \omterm{def:g7:short-circuits-safety:device}{kortsluiting}. Een
stekkerdoos die tegelijk een kachel, een waterkoker en een
broodrooster voedt, vraagt haar ene \omterm{def:g5:series-parallel-circuits:parallel}{tak} om drie volle marsen samen
te dragen --- elk apparaat onschuldig, hun som een overbelasting die
de wandbedrading als een trage broodrooster verwarmt. De onderbreker
klapt op de som eruit, en terecht. Huisregel: één hongerig
warmteapparaat per stopcontact; de stekkerdoos is voor lampen en
opladers, niet voor een privékeuken.
\end{example}

\begin{remark}[De verboden reparatie]\label{rem:g7:short-circuits-safety:nail}
Het oude verhaal is waar en doodt nog steeds: de \omterm{def:g7:short-circuits-safety:fuse}{zekering} van een
huishouden springt telkens, en iemand ``repareert'' hem met een
spijker of een muntje --- een stevige \omterm{def:g4:conductors-insulators:conductor}{geleider} die nooit zal
\omterm{def:g6:water-states:changes}{smelten}. Het symptoom is het zwijgen opgelegd; de ziekte woedt door.
De volgende stormloop treft geen controlepost aan, en de eigen draden
van het huis worden de \omterm{def:g7:short-circuits-safety:fuse}{zekering} --- binnen in de muren. Sla nooit een
wachter over: een \omterm{def:g7:short-circuits-safety:fuse}{zekering} die telkens springt is een wachter die
telkens de waarheid vertelt.
\end{remark}

\begin{remark}[De regels, in hun definitieve vorm]\label{rem:g7:short-circuits-safety:rules}
Het elektrische wetboek van het gezin, met zijn redenen nu compleet:
overbrug nooit de polen van een \omterm{def:g3:first-electric-circuit:battery}{batterij}, en houd sleutels weg
van accu's (een \omterm{def:g7:short-circuits-safety:device}{kortgesloten} accu in metaal is een lasboog); één
warmtehongerig apparaat per stopcontact (overbelasting); beschadigde
snoeren met pensioen (een rafelende mantel is een \omterm{def:g7:short-circuits-safety:device}{kortsluiting} die
repeteert); water en net gescheiden, altijd droge handen (je lichaam
mag nooit de omleiding zijn); wachters gerespecteerd --- alleen
passende \omterm{def:g7:short-circuits-safety:fuse}{zekeringen}, storingen gevonden vóór het terugzetten; en het
stopcontact zelf, voor altijd, niet voor vingers of proeven.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:1}
Formuleer de propositie van de moeiteloze weg. Waarom wint een blote
draad de mars weg bij een lamp?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:2}
Onderscheid de \omterm{def:g7:short-circuits-safety:device}{kortsluiting} van een apparaat van de
\omterm{def:g7:short-circuits-safety:device}{kortsluiting} van de \omterm{def:g3:first-electric-circuit:battery}{batterij}. Welke is ernstiger, en waarom?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:3}
Waarom sterft \omterm{def:g3:first-electric-circuit:bulb}{lampje} 2 in de veilige demonstratie --- en waarom wordt
\omterm{def:g3:first-electric-circuit:bulb}{lampje} 1 op datzelfde moment feller?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:4}
Waarom verhit een \omterm{def:g7:short-circuits-safety:device}{kortsluiting} haar draad? Welk alledaags apparaat
gebruikt diezelfde verwarming met opzet?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:5}
Wat is een \omterm{def:g7:short-circuits-safety:fuse}{zekering}, en wat offert ze op? Waarom moet ze
\omterm{def:g4:series-circuits:series}{in serie} staan met wat ze bewaakt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:6}
Hoe verbetert een stroomonderbreker de \omterm{def:g7:short-circuits-safety:fuse}{zekering}? Wat moet er
gebeuren voordat een van beide wachters wordt teruggezet of vervangen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:7}
Een kachel, een waterkoker en een broodrooster delen één stekkerdoos.
Geen draad ligt bloot, geen apparaat is kapot --- en toch klapt de
onderbreker eruit. Noem het verschijnsel en leg de klacht van de
\omterm{def:g5:series-parallel-circuits:parallel}{tak} uit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:short-circuits-safety:8}
Waarom hoort een \omterm{def:g7:short-circuits-safety:fuse}{zekering} ``repareren'' met een spijker bij de
gevaarlijkste daden in een huis? Wat wordt daarna de \omterm{def:g7:short-circuits-safety:fuse}{zekering}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:9}
De twee binnendraden van een snoer raken elkaar met gerafelde mantels
binnen in de kabel --- stroomopwaarts van de lamp die ze voeden.
Diagnosticeer: welke soort \omterm{def:g7:short-circuits-safety:device}{kortsluiting}, wat gebeurt er met de lamp,
en wat doet de wachter van de \omterm{def:g5:series-parallel-circuits:parallel}{tak}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:10}
Voorspel wat er in de demonstratie met twee \omterm{def:g3:first-electric-circuit:bulb}{lampjes} gebeurt als de
omleidingsdraad in plaats daarvan over \emph{allebei} de
\omterm{def:g3:first-electric-circuit:bulb}{lampjes} samen wordt gelegd --- van vóór \omterm{def:g3:first-electric-circuit:bulb}{lampje} 1 tot na
\omterm{def:g3:first-electric-circuit:bulb}{lampje} 2. Welke definitie geldt nu, en waarom mag de proefnemer het
niet proberen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:short-circuits-safety:11}
Kerstochtendraadsel: in de sliert kerstlampjes van oma \omterm{def:g4:series-circuits:series}{in serie} zit
één \omterm{def:g3:first-electric-circuit:bulb}{lampje} waarvan de inwendige steundraadjes bij het sterven aan
elkaar zijn gesmolten --- het heeft zichzelf \omterm{def:g7:short-circuits-safety:device}{kortgesloten}. Tegen de
gebruikelijke tragedie in \emph{blijft de rest van de sliert
schijnen}, een beetje feller. Verklaar dat --- en zeg wat er nu
langzaam voor de overgebleven \omterm{def:g3:first-electric-circuit:bulb}{lampjes} in de maak is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:short-circuits-safety:12}
Ontwerp het beschermingsplan van een kleine werkplaats: drie
parallelle \omterm{def:g5:series-parallel-circuits:parallel}{takken} (verlichting; elektrisch gereedschap; een
batterijlader), één voeding. Plaats wachters en
\omterm{ex:g3:first-electric-circuit:switch}{schakelaars}; kies welke \omterm{def:g5:series-parallel-circuits:parallel}{takken} een eigen wachter verdienen en
waarom; en schrijf het bordje van twee regels boven de
zekeringlade dat \cref{rem:g7:short-circuits-safety:nail} samenvat.
\end{exercise}

\section{Probleem: het rapport van de brandonderzoekster}

\begin{problem}[{Weekendprobleem --- na de flatbrand; de onderzoekster
leest de bekentenis van de bedrading; drie storingen, één
les}]\label{pb:g7:short-circuits-safety:1}
Er is niemand gewond geraakt --- het gezin was weg --- maar de
keukenmuur is verkoold, en de onderzoekster moet de nacht uit het
bewijsmateriaal reconstrueren. Jij draagt haar notitieboekje.

\medskip\noindent\textbf{Deel I --- De ruïne lezen.}
\begin{enumerate}
  \item Stuk één: een stekkerdoos die een kachel, een waterkoker en
        een frituurpan voedde, met half gesmolten plastic. Nergens
        blote draad. Noem de storing die dit stuk suggereert, en
        waarom er geen ``schurkendraad'' voor nodig is.
  \item Stuk twee: een snoer achter de kast, door een huisdier
        aangevreten, met de twee \omterm{def:g4:conductors-insulators:conductor}{geleiders} die elkaar bij de beet
        raken. Noem deze storing, en zeg waar de mars vanaf het
        moment van contact heen ging.
  \item Stuk drie, het vernietigende: in de zekeringkast bevat het
        patroon van de keukentak --- een opgerold strookje
        aluminiumfolie. Zeg wat die ``reparatie'' met de bescherming
        van de \omterm{def:g5:series-parallel-circuits:parallel}{tak} deed.
  \item Zet de drie stukken in een waarschijnlijk verhaal: welke
        storing kwam eerst, wat deed de echte \omterm{def:g7:short-circuits-safety:fuse}{zekering} eraan (vóór
        stuk drie), en waarom eindigde de laatste nacht in brand in
        plaats van in een donkere keuken?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De natuurkundige verklaring.}
\begin{enumerate}[resume]
  \item Het gerecht vraagt: waarom verhit een \omterm{def:g7:short-circuits-safety:device}{kortgesloten} of
        overbelaste draad? Noem het alledaagse apparaat dat bewijst
        dat stroomverwarming te temmen is --- en wat hier ontbrak.
  \item Waarom is een \omterm{def:g7:short-circuits-safety:fuse}{zekering} met opzet dun? Leg de uitdrukking
        ``een lijfwacht die voor je sterft'' uit.
  \item Het foliestrookje geleidde voortreffelijk. Juist
        \emph{daarom} faalde het als wachter. Los de kleine paradox
        voor de jury op.
  \item De kachel alleen was veilig verklaard, de waterkoker alleen
        veilig, de frituurpan alleen veilig. Schrijf één zin over hoe
        drie veiligheden optelden tot één gevaar.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het vonnis en de lessen.}
\begin{enumerate}[resume]
  \item De onderzoekster moet de drie storingen op moreel
        \omterm{def:g4:weight-and-mass:weight}{gewicht} ordenen: de overbelasting, het aangevreten snoer, de
        folie. Welke maakte van een ongeluk een ramp, en waarom?
  \item Het gezin vraagt wat er in de brandnacht zou zijn gebeurd
        als er een deugdelijke \omterm{def:g7:short-circuits-safety:fuse}{zekering} in had gezeten. Loop het
        minuut voor minuut met hen door.
  \item Adviseer over de nieuwe bedrading: hoe moeten kachel,
        waterkoker en frituurpan nu gevoed worden, en welke vaste
        regel geldt voortaan voor de stekkerdoos?
  \item Sluit het rapport af met de samenvatting van de
        onderzoekster in drie zinnen: de werkwijze van de schurk, de
        taak van de wachter, en de ene reparatie die nooit
        geïmproviseerd mag worden.
\end{enumerate}
\end{problem}
