\chapter{Bewegingsgrafieken en gemiddelde snelheid}\label{ch:g7:motion-average-speed}

Twee jaar geleden betekende \omterm{def:g7:motion-average-speed:formula}{snelheid} ``kilometers afgelegd in één uur'',
op je vingers geteld. Nu, gewapend met letters en grafieken, krijgt het
idee tanden: een formule die in drie richtingen rekent, een plaatje dat
een hele reis in één oogopslag toont --- en een beroemde val over
gemiddelden waar volwassenen dagelijks in trappen.

\section{De formule}

\begin{definition}[Snelheid, per formule]\label{def:g7:motion-average-speed:formula}
Voor een reis (of een stuk ervan) die in een gelijkmatig tempo wordt
afgelegd, is de \emph{snelheid}\index{snelheid} $v$ de afstand $d$
gedeeld door de reistijd $t$:
\[
v = \frac{d}{t}.
\]
Met $d$ in kilometers en $t$ in \omterm{ex:g2:measuring-time:clock}{uren} spreekt $v$
\emph{kilometer per uur} (\unit{km/h}); met \omterm{def:g2:measuring-length:units}{meters} en \omterm{def:g2:measuring-time:second}{seconden},
\emph{\omterm{def:g2:measuring-length:units}{meter} per \omterm{def:g2:measuring-time:second}{seconde}} (\unit{m/s}) --- de lieveling van de
natuurkundige. Net als die van de \omterm{def:g7:volume-mass-density:formula}{dichtheid} rekent deze formule in
alle drie de richtingen: $v = d/t$, \ $d = v \times t$, \
$t = d/v$.
\end{definition}

\begin{example}[De recepten aan het werk]\label{ex:g7:motion-average-speed:recipes}
Een trein legt \qty{240}{km} af in $2$ uur: $v = 240 \div 2 =
\qty{120}{km/h}$. Hoe ver in dat tempo in $5$ uur? $d = 120
\times 5 = \qty{600}{km}$. Hoe lang over \qty{300}{km}? $t = 300
\div 120 = 2.5$ uur --- twee uur en dertig minuten (let op de
weerspannige: $0.5$ uur is $30$ minuten, nooit ``$50$'').
\end{example}

\begin{example}[Twee eenheden, één snelheid]\label{ex:g7:motion-average-speed:units}
Een sprinter loopt \qty{100}{m} in \qty{10}{s}: $v = \qty{10}{m/s}$.
Hoe snel is dat in wegeenheden? In één uur --- $3600$ \omterm{def:g2:measuring-time:second}{seconden} ---
zou de sprinter $10 \times 3600 = \qty{36000}{m} = \qty{36}{km}$
afleggen: dus $\qty{10}{m/s} = \qty{36}{km/h}$. De algemene
wisselkoers: elke \unit{m/s} is \qty{3.6}{km/h} waard --- omdat een uur
$3600$ \omterm{def:g2:measuring-time:second}{seconden} bevat en een kilometer slechts $1000$
\omterm{def:g2:measuring-length:units}{meter}. De \qty{340}{m/s} van \omterm{def:g3:sound-around-us:sound}{geluid}, in
wegeenheden: ruim \qty{1200}{km/h}.
\end{example}

\section{De reis als plaatje}

\begin{proposition}[De grafiekwetten]\label{prop:g7:motion-average-speed:graphs}
Zet de afgelegde afstand uit tegen de tijd, en de beweging schrijft haar
autobiografie:
\begin{enumerate}
  \item \omterm{def:g6:describing-motion:uniform}{eenparige beweging} tekent een \emph{rechte} lijn ---
        gelijke afstanden in gelijke tijden, stap na stap;
  \item hoe \emph{steiler} de lijn, hoe sneller de beweging:
        steilheid is zichtbaar gemaakte \omterm{def:g7:motion-average-speed:formula}{snelheid};
  \item een \emph{vlak} stuk is een stop: de tijd verstrijkt, de
        afstand staat stil.
\end{enumerate}
Bochten vertellen van verandering: omhoog krommend, versnellen;
afvlakkend, vertragen.
\end{proposition}

\begin{method}[Een afstand--tijdgrafiek lezen]\label{met:g7:motion-average-speed:read}
\begin{enumerate}
  \item controleer eerst de assen en hun eenheden --- minuten of
        \omterm{ex:g2:measuring-time:clock}{uren}, \omterm{def:g2:measuring-length:units}{meters} of kilometers;
  \item knip de lijn bij haar knikken in stukken; benoem elk stuk:
        recht-en-steil, recht-en-rustig, vlak;
  \item lees voor elk recht stuk de stijging en de horizontale
        afstand af --- gewonnen afstand, verstreken tijd --- en deel:
        de \omterm{def:g7:motion-average-speed:formula}{snelheid} van dat stuk;
  \item lees voor het hele verhaal de totale afstand af bij het
        laatste \omterm{def:g2:measuring-time:duration}{tijdstip} --- en onthoud dat de vlakke stukken deel
        uitmaken van de totale tijd.
\end{enumerate}
\end{method}

\begin{example}[De koerier, ontcijferd]\label{ex:g7:motion-average-speed:decoded}
Pas de methode toe op de figuur. Eerste stuk: \qty{10}{km} in $20$
minuten --- een derde van een uur --- dus $v = 10 \div \tfrac{1}{3} =
\qty{30}{km/h}$. Tweede: vlak van minuut $20$ tot $35$ --- een
bezorgstop. Derde: \qty{10}{km} in $25$ minuten, een tikje onder
\qty{24}{km/h}. Totaal: \qty{20}{km} in één uur --- wat ons het
volgende idee van dit hoofdstuk op een presenteerblaadje aanreikt.
\end{example}

\section{Gemiddelde snelheid --- en haar beroemde val}

\begin{definition}[Gemiddelde snelheid]\label{def:g7:motion-average-speed:average}
De \emph{gemiddelde snelheid}\index{gemiddelde snelheid} van een hele
reis is de totale afstand gedeeld door de \emph{totale} tijd --- stops
inbegrepen:
\[
v_{\text{gemiddeld}} = \frac{d_{\text{totaal}}}{t_{\text{totaal}}}.
\]
Het is het ene gelijkmatige tempo waarmee dezelfde weg in dezelfde
totale tijd zou zijn afgelegd --- de \qty{20}{km} van de koerier in één
uur: gemiddeld \qty{20}{km/h}, ook al draaiden de wielen geen enkel
moment op die \omterm{def:g7:motion-average-speed:formula}{snelheid}.
\end{definition}

\begin{proposition}[De val]\label{prop:g7:motion-average-speed:trap}
De \omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid} van een reis is in het algemeen
\emph{niet} het midden van haar snelheden. Trage stukken vreten
meer tijd dan snelle, dus wegen ze zwaarder in het gemiddelde. Alleen
het volledige recept --- totale afstand gedeeld door totale tijd --- is
betrouwbaar; het middelen van de snelheidsgetallen zelf is de meest
verleidelijke misstap van dit hoofdstuk.
\end{proposition}

\begin{example}[De val, dichtgeklapt]\label{ex:g7:motion-average-speed:sprung}
Een fietser rijdt \qty{60}{km} heen met \qty{30}{km/h}, en dezelfde
\qty{60}{km} terug met \qty{20}{km/h}. ``Gemiddeld: \qty{25}{km/h}''?
Controleer het eerlijk. Heen: $t = 60 \div 30 = 2$ uur. Terug: $t =
60 \div 20 = 3$ uur. Hele reis: \qty{120}{km} in $5$
uur --- $v_{\text{gemiddeld}} = \qty{24}{km/h}$. De trage helft
eiste drie van de vijf uur op en trok het gemiddelde onder het
midden. Hoe sneller je op de ene helft gaat, hoe korter die helft
überhaupt bestaat.
\end{example}

\begin{remark}[Wat de snelheidsmeter weet]\label{rem:g7:motion-average-speed:speedometer}
\omterm{def:g7:motion-average-speed:average}{Gemiddelde snelheid} beschrijft een hele reis; de naald van de
snelheidsmeter beantwoordt een andere vraag --- \emph{hoe snel op dit
moment}. Op de grafiek woont ``op dit moment'' in de steilheid van de
lijn \emph{in één punt} --- makkelijk te zien op een recht stuk,
subtiel waar de lijn kromt. ``Steilheid in een punt'' precies maken is
een van de grootste uitvindingen van de wiskunde, en die wacht aan het
eind van de middelbare school op je. Tot dan: rechte stukken krijgen
getallen, krommen krijgen verhalen.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:1}
Schrijf de snelheidsformule op en haar twee alledaagse
eenheden. Welk recept vindt een afstand? En een tijd?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:2}
Een veerboot legt \qty{45}{km} af in $3$ uur; een haas loopt
\qty{100}{m} in \qty{8}{s}. Bereken beide snelheden, elk in haar
natuurlijke \omterm{def:g6:measurement-in-science:measuring}{eenheid}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:3}
Reken om, met de wisselkoers: \qty{5}{m/s} naar \unit{km/h}; \
\qty{72}{km/h} naar \unit{m/s}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:4}
Noem de drie grafiekwetten. Wat vertelt een bocht die geleidelijk
afvlakt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:5}
Tussen welke minuten is het tempo op de grafiek van de koerier het
rustigst (maar niet nul)? Hoe zie je dat zonder te rekenen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:6}
De grafiek van een wandelaar toont een rechte lijn door de punten
($30$ min, \qty{2}{km}) en ($60$ min, \qty{4}{km}). Eenparig of
veranderlijk? \omterm{def:g7:motion-average-speed:formula}{Snelheid} in \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:7}
Definieer \omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid}. Een tocht: \qty{12}{km} in $4$
uur, inclusief een picknick van een uur. \omterm{def:g7:motion-average-speed:average}{Gemiddelde snelheid} ---
en \omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid} \emph{tijdens het lopen}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:8}
Waarom weegt een traag stuk ``zwaarder'' in een gemiddelde dan een snel
stuk van dezelfde lengte? Antwoord met tijd, niet met formules.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:9}
Speel de val opnieuw af: \qty{30}{km} heen met \qty{15}{km/h},
\qty{30}{km} terug met \qty{30}{km/h}. Voorspeld midden, eerlijk
gemiddelde --- en welke helft van de reis bezat het grootste deel van
de klok?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:10}
Schets (of beschrijf nauwkeurig) de grafiek van deze rit: eenparig
\qty{40}{km/h} gedurende een half uur; stilstaand gedurende een
kwartier; eenparig \qty{60}{km/h} gedurende een kwartier. Bereken
daarna de \omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid} van de rit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:11}
Onweer tellen, opgewaardeerd: de donder komt \qty{6}{s} na de flits
aan. Gebruik met \omterm{def:g3:sound-around-us:sound}{geluid} op \qty{340}{m/s} de formule $d = v \times
t$ voor de afstand tot het onweer --- en toets de oude regel ``deel
door drie voor kilometers'' aan je formule.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:motion-average-speed:12}
Een oude klassieker, elke minuut waard: een bestuurder legt de eerste
helft \emph{van de afstand} van een rit af met \qty{30}{km/h} en wil
een totaalgemiddelde van \qty{60}{km/h}. Toon --- met het recept
totale afstand gedeeld door totale tijd op een rit van \qty{60}{km}
--- aan dat de tweede helft in nul tijd zou moeten worden afgelegd: de
wens is onmogelijk, niet slechts moeilijk.
\end{exercise}

\section{Probleem: de vrijdag van de koerier}

\begin{problem}[{Weekendprobleem --- één fietskoerier, één vrijdag in
de stad; de grafiek van de coördinator, de val in het
bonusblad}]\label{pb:g7:motion-average-speed:1}
De coördinator legt de vrijdag van koerier Lena vast op een
afstand--tijdgrafiek en berekent daaruit haar loon. Jij controleert het
blad.

\medskip\noindent\textbf{Deel I --- De ochtend, uit de grafiek.}
De grafiek toont: een rechte klim van ($0$ min, \qty{0}{km}) naar
($30$ min, \qty{9}{km}); vlak tot $45$ min; recht vandaar naar
($75$ min, \qty{15}{km}); vlak tot $90$ min.
\begin{enumerate}
  \item Vertel het verhaal van de ochtend in woorden: stukken, stops,
        en wat de stops vermoedelijk waren.
  \item \omterm{def:g7:motion-average-speed:formula}{Snelheid} op het eerste stuk, in \unit{km/h}?
  \item \omterm{def:g7:motion-average-speed:formula}{Snelheid} op het tweede rijstuk?
  \item Lena's \emph{gemiddelde} \omterm{def:g7:motion-average-speed:formula}{snelheid} over de hele ochtend van
        $90$ minuten, stops inbegrepen?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De middag, uit de formule.}
\begin{enumerate}[resume]
  \item Eerste rit van de middag: \qty{12}{km} met een gelijkmatige
        \qty{24}{km/h}. Hoe lang duurde die, in minuten?
  \item Tweede rit: een bezorging in het noorden van de stad,
        \qty{20}{minutes} met \qty{18}{km/h}. Hoeveel kilometer?
  \item Derde rit: de lange tocht naar het depot, \qty{10}{km},
        gedaan in $25$ minuten. \omterm{def:g7:motion-average-speed:formula}{Snelheid} bij gelijkmatig tempo in
        \unit{km/h}?
  \item Totale middag: tel de afstanden en de tijden op (alleen de
        ritten, geen pauzes): welke \omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid} hielden de
        wielen aan terwijl ze rolden?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De val van het bonusblad.}
De bonusregel: ``een \omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid} over een volledige ronde
boven \qty{20}{km/h} levert de snelrijdersbonus op.''
\begin{enumerate}[resume]
  \item Lena's volledige vrijdag: \qty{37}{km} in $3$ uur (alle
        stops inbegrepen). Verdient ze de bonus?
  \item Haar collega Max claimt hem: ``Ik reed de helft van mijn
        ronde met \qty{30}{km/h} en de helft met \qty{15}{km/h} ---
        dat is gemiddeld $22.5$!'' De coördinator controleert: beide
        helften waren \qty{12}{km}. Bereken het echte gemiddelde van
        Max en beslis over zijn bonus.
  \item Leg Max in tijdtaal uit waar zijn $22.5$ misging.
  \item Lena stelt een eerlijker bonusregel voor koeriers voor ---
        een die bezorgstops niet bestraft. Doe een voorstel (de
        grafiek weet hoe), en zeg welke \omterm{def:g7:motion-average-speed:formula}{snelheid} die
        meet.
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- De les van de coördinator.}
\begin{enumerate}[resume]
  \item Vrijdagevaluatie: schrijf de drie zinnen van de coördinator
        --- wat een recht stuk, een vlak stuk en een steiler stuk
        elk op de grafiek betekenen --- en de ene waarschuwing over
        het middelen van snelheden.
\end{enumerate}
\end{problem}
