\chapter{Lichtbronnen en voortplanting}\label{ch:g7:light-sources-propagation}

Waarom kun je deze bladzijde lezen? Er schijnt een lamp --- maar de
bladzijde is geen lamp, en toch ontvangt je oog \omterm{def:g1:light-and-shadows:source}{licht} van elke
gedrukte regel. Tussen de dingen die \omterm{def:g1:light-and-shadows:source}{licht} maken en de dingen die het
alleen maar doorgeven ligt een onderscheid waar de optica niet zonder
kan; en daarmee komt ook het volledige antwoord op de oudste vraag van
dit boek: wat is er nodig om iets te zien?

\section{Makers en doorgevers}

\begin{definition}[Primaire lichtbronnen]\label{def:g7:light-sources-propagation:primary}
Een \emph{primaire lichtbron}\index{primaire lichtbron} maakt haar
eigen \omterm{def:g1:light-and-shadows:source}{licht}: de Zon en de sterren, vlammen, gloeiende
gloeidraden, het witgloeiende staal van de smidse, glimwormen, de
bliksem. Haar \omterm{def:g1:light-and-shadows:source}{licht} bestaat of er nu iets anders schijnt of niet ---
zet de wereld uit, en een primaire bron gloeit nog steeds.
\end{definition}

\begin{definition}[Verstrooiende voorwerpen]\label{def:g7:light-sources-propagation:diffusion}
Elk ander zichtbaar ding is een \emph{verstrooiend
voorwerp}\index{verstrooiing}: het ontvangt \omterm{def:g1:light-and-shadows:source}{licht} en stuurt een deel
ervan in alle richtingen weer weg --- het \emph{verstrooit} het. De
bladzijde, de appel, de \omterm{def:g2:moon-in-the-sky:moon}{Maan}, je eigen gezicht: geen van alle maakt
ook maar een glimp eigen \omterm{def:g1:light-and-shadows:source}{licht}; elk strooit ontvangen \omterm{def:g1:light-and-shadows:source}{licht}
royaal naar alle kanten. Een verstrooier in het donker is eenvoudigweg
onzichtbaar.
\end{definition}

\begin{example}[Verstrooien is niet spiegelen]\label{ex:g7:light-sources-propagation:notmirror}
Zowel \omterm{def:g5:mirrors-reflection:reflection}{spiegels} als bladzijden sturen ontvangen \omterm{def:g1:light-and-shadows:source}{licht} terug ---
maar op tegengestelde manieren. De \omterm{def:g5:mirrors-reflection:reflection}{spiegel} kaatst in \emph{één}
gedisciplineerde richting (gelijke hoeken --- je oude kaatsregel) en
behoudt het plaatje; de microscopische ruwheid van de bladzijde sproeit
\omterm{def:g1:light-and-shadows:source}{licht} \emph{overal} heen, verwoest het plaatje maar biedt zichzelf
aan elk oog tegelijk aan. Daarom toont een \omterm{def:g5:mirrors-reflection:reflection}{spiegel} de lamp, terwijl
de bladzijde alleen zichzelf toont, vanaf welke plek je ook staat.
\end{example}

\begin{example}[Het dossier van de Maan, gesloten]\label{ex:g7:light-sources-propagation:moon}
Het oordeel uit je kindertijd heeft nu zijn officiële woorden: de
\omterm{def:g2:moon-in-the-sky:moon}{Maan} is geen \omterm{def:g7:light-sources-propagation:primary}{primaire lichtbron}; ze is een
\omterm{def:g7:light-sources-propagation:diffusion}{verstrooiend voorwerp} ter grootte van een \omterm{def:g4:solar-system:planet}{planeet} --- grijs
gesteente dat zonlicht in alle richtingen strooit, de onze inbegrepen.
Maanlicht is dus tweedehands zonneschijn, en de ``donkere kant'' is
gewoon de helft waarvan het dossier op dat moment geen \omterm{def:g1:light-and-shadows:source}{licht} bevat
om door te geven.
\end{example}

\section{Zien, beslecht}

\begin{proposition}[De twee voorwaarden voor zien]\label{prop:g7:light-sources-propagation:vision}
Een oog ziet een voorwerp precies wanneer
\begin{enumerate}
  \item er \omterm{def:g1:light-and-shadows:source}{licht} van het voorwerp \emph{vertrekt} --- gemaakt, als
        het voorwerp een primaire bron is; verstrooid, als het een
        doorgever is --- en
  \item dat \omterm{def:g1:light-and-shadows:source}{licht} langs een rechte, onbelemmerde lijn het oog
        binnenkomt.
\end{enumerate}
Breek een van beide voorwaarden --- een onverlichte verstrooier, of
een muur dwars over de zichtlijn --- en het voorwerp blijft ongezien.
Elke zichtbaarheidspuzzel in dit boek valt uiteen in deze twee
controles.
\end{proposition}

\begin{example}[De bundel zichtbaar gemaakt]\label{ex:g7:light-sources-propagation:beam}
In schone \omterm{def:g2:air-around-us:air}{lucht} is een zaklampbundel die de kamer oversteekt van
opzij niet te zien --- \omterm{def:g2:air-around-us:air}{lucht} geeft vrijwel niets door, en er draait
geen \omterm{def:g1:light-and-shadows:source}{licht} uit de \omterm{def:g6:light-propagation:ray}{bundel} jouw oog in. Schud een stoffige doek uit:
de \omterm{def:g6:light-propagation:ray}{bundel} springt in \omterm{prop:g5:mirrors-reflection:image}{beeld}. Elk stofje is een piepkleine verstrooier
die bundellicht opvangt en een deel jouw kant op strooit --- duizenden
stipjes die de rechte lijn van de \omterm{def:g6:light-propagation:ray}{bundel} voor je tekenen. Mist, nevel
en rook bewijzen dezelfde dienst aan vuurtorenbundels en
bioscoopprojectoren.
\end{example}

\section{Drie jassen voor het licht}

\begin{definition}[Doorzichtig, doorschijnend, ondoorzichtig]\label{def:g7:light-sources-propagation:coats}
Materialen ontmoeten \omterm{def:g1:light-and-shadows:source}{licht} op drie manieren:
\begin{enumerate}
  \item \emph{doorzichtig}\index{doorzichtig}: \omterm{def:g1:light-and-shadows:source}{licht} steekt vrijwel
        ongestoord over en houdt zijn orde --- helder glas, stil
        water, \omterm{def:g2:air-around-us:air}{lucht}; \omterm{prop:g5:mirrors-reflection:image}{beelden} komen intact door;
  \item \emph{doorschijnend}\index{doorschijnend}: \omterm{def:g1:light-and-shadows:source}{licht} steekt
        over, maar door elkaar gehusseld --- matglas, calqueerpapier,
        een dun gordijn; helderheid komt door, \omterm{prop:g5:mirrors-reflection:image}{beelden} niet;
  \item \emph{ondoorzichtig}\index{ondoorzichtig}: \omterm{def:g1:light-and-shadows:source}{licht} wordt
        tegengehouden --- opgeslokt of teruggestrooid --- hout,
        steen, metaal, je hand; achter een ondoorzichtig ding ligt
        \omterm{def:g1:light-and-shadows:shadow}{schaduw}.
\end{enumerate}
\end{definition}

\begin{method}[Materialen sorteren bij zaklamplicht]\label{met:g7:light-sources-propagation:sort}
Een donkere kamer, een zaklamp, een bedrukte bladzijde, en het monster
ertussen:
\begin{enumerate}
  \item houd het monster boven de druk met de zaklamp erachter;
  \item lees je de druk erdoorheen? --- \omterm{def:g7:light-sources-propagation:coats}{doorzichtig};
  \item zie je gloed maar geen letters? --- \omterm{def:g7:light-sources-propagation:coats}{doorschijnend};
  \item zie je duisternis, en een \omterm{def:g1:light-and-shadows:shadow}{schaduw} erachter? ---
        \omterm{def:g7:light-sources-propagation:coats}{ondoorzichtig}.
\end{enumerate}
Toets eerlijk: veel materialen wisselen van ploeg met de dikte --- een
laagje water is \omterm{def:g7:light-sources-propagation:coats}{doorzichtig}, de diepzee is nachtdonker; één vel
papier is \omterm{def:g7:light-sources-propagation:coats}{doorschijnend}, het gesloten boek \omterm{def:g7:light-sources-propagation:coats}{ondoorzichtig}.
\end{method}

\begin{example}[De drie ploegen van de architectuur]\label{ex:g7:light-sources-propagation:architecture}
Bouwers zetten alle drie met opzet in. \omterm{def:g7:light-sources-propagation:coats}{Doorzichtig} vensterglas:
uitzicht en \omterm{def:g1:light-and-shadows:source}{licht} tegelijk. Doorschijnende badkamerruiten en
matglazen kantoorwanden: daglicht binnen, privacy bewaard ---
helderheid zonder \omterm{prop:g5:mirrors-reflection:image}{beelden} is precies hun functieomschrijving.
Ondoorzichtige muren en luiken: duisternis op afroep. Lampenkappen
zijn doorschijnendheid in dienst van het comfort: de felle glans van de
gloeidraad wordt door elkaar gehusseld tot een zachte, bronloze gloed.
\end{example}

\begin{remark}[Niets om door te geven, niets te zien]\label{rem:g7:light-sources-propagation:space}
Raadsel: een astronaute tussen de sterren kijkt \emph{zijwaarts}
over een met zonnestralen gevulde leegte. Wat ziet ze? Zwartheid ---
terwijl er zonlicht langs haar ogen stroomt. De ruimte draagt geen
noemenswaardig stof: niets verstrooit het passerende \omterm{def:g1:light-and-shadows:source}{licht} naar
haar toe, en \omterm{def:g1:light-and-shadows:source}{licht} dat het oog niet binnenkomt schildert niets. De
daghemel is juist blauw doordat onze \omterm{def:g2:air-around-us:air}{lucht} zonlicht \emph{wél} naar
ons omlaag doorgeeft --- een \omterm{def:g4:solar-system:planet}{planeet} vol zachte \omterm{def:g7:light-sources-propagation:diffusion}{verstrooiing} boven
ons hoofd. Geen \omterm{def:g2:air-around-us:air}{lucht}, zwarte hemel, zelfs om twaalf uur 's middags: de
astronauten van de \omterm{def:g2:moon-in-the-sky:moon}{Maan} stonden in de zon onder een sterloze zwarte
koepel.
\end{remark}

\section{Probleem: het museum verlichten}

\begin{problem}[{Weekendprobleem --- de nieuwe beeldhouwzaal; de
nachtdienst van een lichtontwerper; bronnen, verstrooiers en drie
soorten glas}]\label{pb:g7:light-sources-propagation:1}
De nieuwe zaal van het museum gaat maandag open: één marmeren \omterm{prop:g5:mirrors-reflection:image}{beeld},
schilderijen aan de muren, een glazen vitrine, en een nerveuze
conservator. Jij assisteert de lichtontwerper.

\medskip\noindent\textbf{Deel I --- Grondbeginselen ter plaatse.}
\begin{enumerate}
  \item De conservator vraagt waarom het \omterm{prop:g5:mirrors-reflection:image}{beeld}, ``zo schitterend
        wit'', überhaupt nog lampen nodig heeft. Antwoord met de twee
        voorwaarden voor zien.
  \item Noem de primaire bronnen en de verstrooiers in de zaal zodra
        de lampen aan zijn: lampen, \omterm{prop:g5:mirrors-reflection:image}{beeld}, schilderijen, bezoekers,
        het glas van de vitrine.
  \item Het marmer lijkt van overal te gloeien, zonder verblindende
        vlek. Welke manier van \omterm{def:g1:light-and-shadows:source}{licht} terugsturen is hier aan het werk,
        en waarom ontvangt elke bezoekerspositie het?
  \item Eén gepolijste koperen plaquette werpt daarentegen één
        verblindende \omterm{def:g5:mirrors-reflection:reflection}{weerkaatsing} van een lamp. Noem de andere manier,
        en geef de regel die bepaalt waar die verblinding heen gaat.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De verblinding temmen.}
\begin{enumerate}[resume]
  \item Blote lampen schitteren in de ogen van de bezoekers. De
        ontwerper hult elke lamp in een matglazen bol. Bij welke
        materiaalploeg hoort die bol, en wat doet hij met het
        \omterm{def:g1:light-and-shadows:source}{licht} van de gloeidraad?
  \item De schilderijen moeten verlicht worden, maar er mag vanaf de
        belangrijkste kijkplek geen lamp in hun vernis
        \emph{weerspiegeld} verschijnen. Waar zet de ontwerper met de
        spiegelregel de lampen ten opzichte van schilderij en
        kijker?
  \item Het heldere glas van de vitrine is bijna onzichtbaar --- tot
        de witte mouw van een bezoeker er als een spook in
        verschijnt. Waarom laat helder glas tegelijk door en
        spiegelt het zwakjes? Wat bereikt de zwartgewande nis die de
        ontwerper achter de vitrine plaatst?
  \item Soms zweven er stofjes door de \omterm{def:g6:light-propagation:ray}{bundels} van de spots, die
        heldere draden in de \omterm{def:g2:air-around-us:air}{lucht} tekenen. Conservator: ``Prachtig!
        Laat maar zo.'' Ontwerper: ``Het betekent vuile
        \omterm{def:g2:air-around-us:air}{lucht}.'' Wie heeft gelijk over de natuurkunde, en wat zijn de
        draden?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De nachttest.}
\begin{enumerate}[resume]
  \item \omterm{def:g1:light-and-shadows:source}{Licht} uit, luiken dicht: de bewaker meldt dat de zaal
        ``volslagen zwart is --- zelfs het witte \omterm{prop:g5:mirrors-reflection:image}{beeld} niet''. Leg met
        de twee voorwaarden uit waarom witheid in het donker
        machteloos is.
  \item Er wordt een dakraam voorgesteld: matglas boven het hoofd.
        Wat zal het overdag leveren, en wat zal het weigeren te
        leveren (uitzicht op wolken, rechtstreekse zonneschittering)?
  \item De conservator vreest dat zonlicht de schilderijen doet
        verbleken en vraagt om ``\omterm{def:g1:light-and-shadows:source}{licht} zonder enige zon''. De
        ontwerper glimlacht: overdag \emph{is} zelfs het zachte
        \omterm{def:g1:light-and-shadows:source}{licht} van de noordramen zonneschijn. Leg dat uit --- wat
        heeft het doorgegeven?
  \item Schrijf het rapport van de ontwerper: drie zinnen --- één
        over waarom \omterm{def:g7:light-sources-propagation:diffusion}{verstrooiing} de beste vriend van het museum is, één
        over waar \omterm{def:g5:mirrors-reflection:reflection}{weerkaatsingen} van spiegelsoort in de gaten moeten
        worden gehouden, en één over de glasploeg die voor elke taak
        gekozen is.
\end{enumerate}
\end{problem}
