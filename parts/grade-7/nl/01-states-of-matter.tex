\chapter{Toestanden van materie en toestandsovergangen}\label{ch:g7:states-of-matter}

Zes jaar vaste stoffen die hun vorm houden, vloeistoffen die waterpas
liggen, gassen die elke hoek vullen --- en nooit vroegen we
\emph{waarom}. Dit jaar \omterm{def:g1:light-and-shadows:source}{licht} de natuurkunde het deksel op. Onder de
drie toestanden schuilt een van de grootste ideeën die mensen ooit
hadden: alles, werkelijk alles, is opgebouwd uit deeltjes die te klein
zijn om te zien. Gun ons dat ene idee, en de regels die je sinds je
kindertijd verzamelde verklaren zichzelf in één beweging.

\section{Het grote idee}

\begin{definition}[Moleculen]\label{def:g7:states-of-matter:molecule}
Alle materie is opgebouwd uit \emph{moleculen}\index{molecuul}:
onvoorstelbaar kleine deeltjes, ver onder wat welk oog of welke
schoolmicroscoop ook kan zien. Eén druppel water bevat meer moleculen
dan er waterdruppels in alle zeeën zijn. Elke zuivere stof is
opgebouwd uit haar eigen soort molecuul --- water uit
watermoleculen, ijzer uit de eigen deeltjes van ijzer --- en de
moleculen van een stof zijn allemaal identiek.
\end{definition}

\begin{remark}[Geloven zonder zien]\label{rem:g7:states-of-matter:believing}
Niemand heeft ooit één \omterm{def:g7:states-of-matter:molecule}{molecuul} met het blote oog gezien --- waarom
erin geloven? Omdat het idee zijn geloofwaardigheid \emph{verdient}:
het voorspelt, correct, alles wat in dit hoofdstuk staat, en bergen
meer. Zo aanvaardt de natuurkunde een onzichtbare gast: niet op geloof,
maar op meedogenloze bruikbaarheid. (Rechtstreekse \omterm{prop:g5:mirrors-reflection:image}{beelden} van
schitterende moderne instrumenten kwamen pas een eeuw geleden --- lang
nadat natuurkundigen al zeker waren.)
\end{remark}

\begin{proposition}[Het deeltjesmodel van de drie toestanden]\label{prop:g7:states-of-matter:model}
De drie toestanden zijn drie ordeningen van dezelfde
\omterm{def:g7:states-of-matter:molecule}{moleculen}:
\begin{enumerate}
  \item \emph{vast}: \omterm{def:g7:states-of-matter:molecule}{moleculen} dicht opeengepakt in een geordende
        stapeling, elk trillend op zijn plaats maar zijn positie
        vasthoudend --- vandaar de vaste vorm;
  \item \emph{vloeibaar}: \omterm{def:g7:states-of-matter:molecule}{moleculen} raken elkaar nog, maar
        wanordelijk en vrij om langs elkaar te glijden --- vandaar
        het stromen, en het waterpasse oppervlak;
  \item \emph{\omterm{def:g3:solids-liquids-gases:gas}{gas}}: \omterm{def:g7:states-of-matter:molecule}{moleculen} ver uit elkaar, snel in alle
        richtingen vliegend, botsend en terugkaatsend --- vandaar het
        vullen van elke aangeboden hoek.
\end{enumerate}
\end{proposition}

\begin{example}[Zes jaar regels, één verklaring]\label{ex:g7:states-of-matter:rules}
Laat de oude regels door het model lopen. Vaste stoffen houden hun
vorm: elk \omterm{def:g7:states-of-matter:molecule}{molecuul} blijft op zijn post. Vloeistoffen nemen de vorm
van de houder aan en liggen waterpas: de glijdende menigte zakt in
tot geen enkel \omterm{def:g7:states-of-matter:molecule}{molecuul} lager kan glijden. Gassen vullen alle
aangeboden ruimte: niets houdt de vliegers tegen. De parfum die de
kamer oversteekt: gasmoleculen die vliegen, botsen, zich
verspreiden. Zand dat stroomt als een \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}: geen glijdende
\omterm{def:g7:states-of-matter:molecule}{moleculen}, maar korrels --- en elke korrel is zelf een piepklein
geordend vast bouwsel van biljoenen \omterm{def:g7:states-of-matter:molecule}{moleculen}.
\end{example}

\section{De proef op de lege ruimte}

\begin{method}[Twee injectiespuiten]\label{met:g7:states-of-matter:syringe}
Het model beweert dat gasmoleculen ver uiteen vliegen en dat
vloeistofmoleculen elkaar raken. Dan zou \omterm{def:g3:solids-liquids-gases:gas}{gas} zich moeten laten
samendrukken en \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} niet. Toets het:
\begin{enumerate}
  \item vul een plastic injectiespuit met \omterm{def:g2:air-around-us:air}{lucht}, sluit de tuit af met
        een vingertop en duw de zuiger in: hij geeft mee --- je kunt
        de ruimte van de \omterm{def:g2:air-around-us:air}{lucht} halveren en voelt een veerkrachtig
        terugvechten;
  \item vul dezelfde spuit met water, sluit af en duw: de zuiger
        staat als een muur --- water weigert ook maar een haar te
        krimpen;
  \item laat elke zuiger los: de \omterm{def:g2:air-around-us:air}{lucht} veert terug; het water is
        nooit bewogen.
\end{enumerate}
\end{method}

\begin{proposition}[Gassen laten zich samendrukken; vloeistoffen en vaste stoffen niet]\label{prop:g7:states-of-matter:squeeze}
Een \omterm{def:g3:solids-liquids-gases:gas}{gas} kan \emph{samengedrukt}\index{samendrukbaar} worden --- in
minder ruimte gedwongen --- doordat een \omterm{def:g3:solids-liquids-gases:gas}{gas} grotendeels lege ruimte
tussen vliegende \omterm{def:g7:states-of-matter:molecule}{moleculen} is. Vloeistoffen en vaste stoffen
zijn praktisch \emph{onsamendrukbaar}: hun \omterm{def:g7:states-of-matter:molecule}{moleculen} raken elkaar al,
en materie die elkaar raakt kun je niet dichter opeen pakken met welke
duw een hand --- of een hydraulische pers --- ook kan geven.
\end{proposition}

\begin{example}[Samengeperste lucht aan het werk]\label{ex:g7:states-of-matter:squeezed}
De fietspomp perst kamers vol \omterm{def:g2:air-around-us:air}{lucht} in de band --- alleen mogelijk
doordat \omterm{def:g2:air-around-us:air}{lucht} vooral leegte is. De samengeperste menigte, met
\omterm{def:g7:states-of-matter:molecule}{moleculen} die sneller en dichter op de bandwanden hameren, houdt de
fiets omhoog: de duw van een \omterm{def:g3:solids-liquids-gases:gas}{gas} \emph{is} zijn moleculaire hagelbui.
Duikflessen persen een middag ademen in een stalen fles; de
onsamendrukbare remvloeistof draagt intussen de duw van je voet
onverminderd naar de wielen --- elke toestand ingehuurd voor haar
eigen talent.
\end{example}

\section{Toestandsovergangen, herlezen}

\begin{example}[De deuren, molecuul voor molecuul]\label{ex:g7:states-of-matter:doors}
Verwarm een \omterm{def:g3:solids-liquids-gases:solid}{vaste stof} en haar trillende \omterm{def:g7:states-of-matter:molecule}{moleculen} schudden
heftiger; bij het \omterm{def:g4:changes-of-state:melting-point}{smeltpunt} laat het schudden de geordende stapeling
barsten --- de menigte begint te glijden: \omterm{def:g6:water-states:changes}{smelten}. Verwarm de
\omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} verder en de snelste \omterm{def:g7:states-of-matter:molecule}{moleculen} rukken zich helemaal los uit
de greep van de menigte en vliegen weg: \omterm{def:g6:water-states:changes}{verdamping} --- en bij het
\omterm{def:g4:changes-of-state:boiling-point}{kookpunt} wordt de ontsnapping een stormloop vanuit de
\omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} zelf. Afkoelen draait de film terug: vliegers gevangen
(\omterm{def:g6:water-states:changes}{condensatie}), glijders die in het gelid klikken
(\omterm{def:g6:water-states:changes}{stollen}). De zes deuren van vorig jaar zijn één verhaal van
moleculaire greep tegen moleculair schudden.
\end{example}

\begin{example}[Oude wetten gratis]\label{ex:g7:states-of-matter:free}
Twee zwaarbevochten wetten komen nu gratis. \emph{\omterm{def:g3:measuring-mass:mass}{Massa} overleeft
toestandsovergangen}: \omterm{def:g6:water-states:changes}{smelten} herschikt \omterm{def:g7:states-of-matter:molecule}{moleculen} maar maakt er
geen enkele bij en vernietigt er geen --- dezelfde deeltjes, dezelfde
\omterm{def:g3:measuring-mass:mass}{massa}. \emph{Temperatuurplateaus}: bij het \omterm{def:g4:changes-of-state:melting-point}{smeltpunt} gaat de
binnenkomende \omterm{def:g5:heat-and-insulation:heat}{warmte} op aan het breken van de greep van de stapeling,
\omterm{def:g7:states-of-matter:molecule}{molecuul} voor \omterm{def:g7:states-of-matter:molecule}{molecuul}, en niet aan sneller schudden --- de
\omterm{def:g3:temperature-thermometers:temperature}{thermometer} wacht tot het laatste gelid gebroken is. Wat het \omterm{def:g6:measurement-in-science:measuring}{meten}
ontdekte, verklaart het model.
\end{example}

\begin{remark}[Het rare zwellen van water]\label{rem:g7:states-of-matter:iceswell}
Zelfs het vreemde vriesgezwel van water doet mee:
watermoleculen klikken toevallig in een ongewoon \emph{luchtige}
stapeling --- een open rooster met meer lege ruimte dan de glijdende
menigte had. Luchtiger stapeling, meer ruimte, lagere
\omterm{def:g6:density-floating:density}{dichtheid}: ijs \omterm{def:g1:floating-and-sinking:words}{drijft} op zijn eigen \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}. De meeste stoffen
stapelen dichter dan ze glijden; de architectuur van water is de
prachtige uitzondering.
\end{remark}

\section{Warmte, van binnenuit gezien}

\begin{proposition}[Temperatuur en moleculaire beweging]\label{prop:g7:states-of-matter:temperature}
\omterm{def:g3:temperature-thermometers:temperature}{Temperatuur} meet de levendigheid van de moleculaire beweging: hoe
heter een stof, hoe sneller haar \omterm{def:g7:states-of-matter:molecule}{moleculen} trillen, glijden of
vliegen. \omterm{def:g5:heat-and-insulation:heat}{Warmte} die van heet naar koud stroomt, is de levendige
menigte die tot bedaren komt terwijl ze de trage opjaagt --- botsing
na botsing --- tot beide één tempo delen: één \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}. De oude
bergafwaartse wet van de \omterm{def:g5:heat-and-insulation:heat}{warmte} is moleculair tafeltennis.
\end{proposition}

\begin{example}[Kou is geen stof]\label{ex:g7:states-of-matter:cold}
Het model bergt een oude verleiding voorgoed op: er stroomt nooit iets
dat ``kou'' heet. Er is alleen tragere en snellere moleculaire
beweging; de levendige leent levendigheid uit, nooit andersom. Bij het
koudste dat denkbaar is --- \omterm{def:g7:states-of-matter:molecule}{moleculen} zo stil als de natuur toestaat
--- ligt een echte bodem van \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}, ver onder elke vriezer; haar
vreemde en beroemde getal hoort bij het hogeschooldeel.
\end{example}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:states-of-matter:12}

\emph{stilstaand} water wordt losgelaten verspreidt zich langzaam, over

de inkt --- en waarom dezelfde druppel zich in warm water sneller
verspreidt. (Deze nederige waarneming behoorde tot de grote vroege
bewijzen voor het hele moleculaire idee.)
\end{exercise}

\section{Probleem: het moleculaire theater}

\begin{problem}[{Weekendprobleem --- de klas voert het deeltjesmodel op
in de gymzaal; leerlingen als moleculen, toestanden als choreografie;
de vragen van de critici}]\label{pb:g7:states-of-matter:1}
De natuurkundeklas voert de materie zelf op: dertig leerlingen spelen
de \omterm{def:g7:states-of-matter:molecule}{moleculen} van één stof, en de gymzaal is de houder.

\medskip\noindent\textbf{Deel I --- Drie scènes.}
\begin{enumerate}
  \item Scène één, de \omterm{def:g3:solids-liquids-gases:solid}{vaste stof}: hoe moeten de dertig staan, en
        welke kleine beweging moet elk blijven maken? Wat mag geen
        van hen doen?
  \item Scène twee, de \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}: wat verandert er aan de ordening,
        en welke nieuwe vrijheid verschijnt er? Wat moeten de
        leerlingen nog steeds blijven doen wat de gasscène zal
        afschaffen?
  \item Scène drie, het \omterm{def:g3:solids-liquids-gases:gas}{gas}: beschrijf de choreografie --- en zeg
        waarom deze scène de \emph{hele} gymzaal nodig heeft terwijl
        de eerste twee in een hoek samendringen.
  \item Het publiek vraagt: ``Zijn het in alle drie de scènes
        dezelfde dertig leerlingen?'' Welke wet van vorig jaar voert
        het antwoord op?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De critici toetsen het stuk.}
\begin{enumerate}[resume]
  \item Een criticus duwt zachtjes tegen de menigte van de
        gasscène en \omterm{def:g1:floating-and-sinking:words}{drijft} haar in de halve zaal. Laat de scène dat
        toe? En dezelfde duw tegen de kluit van de
        vloeistofscène? Noem de propositie die hier wordt
        opgevoerd.
  \item Om het \emph{\omterm{def:g6:water-states:changes}{smelten}} op te voeren draait de regisseur aan
        een denkbeeldige warmteknop. Wat moeten de leerlingen van de
        \omterm{def:g3:solids-liquids-gases:solid}{vaste stof} steeds meer doen, en op welk moment wordt de scène
        rechtsgeldig de vloeistofscène?
  \item Het stuk moet het smeltplateau tonen: waaraan wordt de
        \omterm{def:g5:everyday-energy:energy}{energie} van de warmteknop tijdens de overgang uitgegeven ---
        en wat moet de ``thermometercriticus'' ondertussen melden over
        de gemiddelde levendigheid van de leerlingen?
  \item Voer de \omterm{def:g6:water-states:changes}{verdamping} vanuit de vloeistofscène op: welke
        leerlingen verlaten de kluit --- willekeurige, of een
        bepaalde soort? Wat gebeurt er met de gemiddelde levendigheid
        van de kluit terwijl ze gaan, en welke alledaagse verkoeling
        verklaart dat?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Speciale effecten.}
\begin{enumerate}[resume]
  \item Het balloneffect: de gasscène wordt omheind door leerlingen
        die een touwring vasthouden (het vel van de ballon). Wat doen
        de moleculaire acteurs met het touw, en wat gebeurt er met de
        ring als de acteurs sneller bewegen?
  \item Het rare \omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen} van water moet opgevoerd worden. Welke
        ongewone instructie over \emph{tussenruimte} geeft de
        regisseur aan de stapeling van de vaste stofscène, en welk
        drijffeit verklaart die?
  \item De inktdruppelfinale: één leerling in een blauw shirt begint
        in een hoek van de \emph{\omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}}scène. Hoe komt het blauwe
        shirt zonder enige duw van de regisseur ver van zijn hoek
        terecht --- en waarom sneller als de warmteknop hoger staat?
  \item Schrijf de programmatoelichting: zeg in drie zinnen welk ene
        idee het hele stuk opvoert, noem twee oude wetten die het
        gratis verklaart, en geef eerlijk toe wat geen enkele scène
        kan tonen (hoe \emph{klein} en hoe \emph{talrijk} de echte
        acteurs zijn).
\end{enumerate}
\end{problem}
