\chapter{Gráficos do movimento e velocidade média}\label{ch:g7:motion-average-speed}

Dois anos atrás, \omterm{def:g7:motion-average-speed:formula}{velocidade} queria dizer ``quilômetros percorridos em
uma hora'', contados nos dedos. Agora, armada de letras e gráficos, a
ideia ganha dentes: uma fórmula que calcula em três direções, um
retrato que mostra a viagem inteira num relance --- e uma armadilha
famosa sobre médias, que pega adultos todos os dias.

\section{A fórmula}

\begin{definition}[Velocidade, por fórmula]\label{def:g7:motion-average-speed:formula}
Para uma viagem (ou um trecho dela) percorrida a um ritmo constante,
a \emph{velocidade}\index{velocidade} $v$ é a distância $d$ dividida
pelo tempo de percurso $t$:
\[
v = \frac{d}{t}.
\]
Com $d$ em quilômetros e $t$ em \omterm{ex:g2:measuring-time:clock}{horas}, $v$ fala em
\emph{quilômetros por hora} (\unit{km/h}); com \omterm{def:g2:measuring-length:units}{metros} e segundos, em
\emph{\omterm{def:g2:measuring-length:units}{metros} por segundo} (\unit{m/s}) --- a predileta dos
cientistas. Como a da \omterm{def:g7:volume-mass-density:formula}{densidade}, esta fórmula calcula nas três
direções: $v = d/t$, \ $d = v \times t$, \ $t = d/v$.
\end{definition}

\begin{example}[As receitas em ação]\label{ex:g7:motion-average-speed:recipes}
Um trem percorre \qty{240}{km} em $2$ \omterm{ex:g2:measuring-time:clock}{horas}:
$v = 240 \div 2 = \qty{120}{km/h}$. Que distância nesse ritmo em $5$
\omterm{ex:g2:measuring-time:clock}{horas}? $d = 120 \times 5 = \qty{600}{km}$. Quanto tempo para
\qty{300}{km}? $t = 300 \div 120 = 2.5$ \omterm{ex:g2:measuring-time:clock}{horas} --- duas \omterm{ex:g2:measuring-time:clock}{horas} e trinta
minutos (cuidado com o rebelde: $0.5$ hora são $30$ minutos, nunca
``$50$'').
\end{example}

\begin{example}[Duas unidades, uma velocidade]\label{ex:g7:motion-average-speed:units}
Um velocista corre \qty{100}{m} em \qty{10}{s}:
$v = \qty{10}{m/s}$. Quanto é isso nas \omterm{def:g6:measurement-in-science:measuring}{unidades} de estrada? Em uma
hora --- $3600$ segundos --- o velocista percorreria
$10 \times 3600 = \qty{36000}{m} = \qty{36}{km}$: então
$\qty{10}{m/s} = \qty{36}{km/h}$. A taxa geral de câmbio: cada
\unit{m/s} vale \qty{3.6}{km/h} --- porque uma hora guarda $3600$
segundos e um quilômetro guarda apenas $1000$ \omterm{def:g2:measuring-length:units}{metros}. Os
\qty{340}{m/s} do \omterm{def:g3:sound-around-us:sound}{som}, em \omterm{def:g6:measurement-in-science:measuring}{unidades} de estrada: mais de
\qty{1200}{km/h}.
\end{example}

\section{A viagem como retrato}

\begin{proposition}[As leis do gráfico]\label{prop:g7:motion-average-speed:graphs}
Coloque num gráfico a distância percorrida contra o tempo, e o
movimento escreve a autobiografia dele:
\begin{enumerate}
  \item o \omterm{def:g6:describing-motion:uniform}{movimento uniforme} desenha uma \emph{reta} --- distâncias
        iguais em tempos iguais, passo após passo;
  \item quanto mais \emph{inclinada} a reta, mais rápido o
        movimento: a inclinação é a \omterm{def:g7:motion-average-speed:formula}{velocidade} tornada visível;
  \item um trecho \emph{horizontal} é uma parada: o tempo passa, a
        distância fica onde está.
\end{enumerate}
As curvas contam mudanças: curvando para cima, acelerando; ficando
menos inclinada, freando.
\end{proposition}

{\small A manhã de um entregador contada por uma linha: pedalada
firme, uma parada de quinze minutos e depois um ritmo mais suave. O
gráfico é a autobiografia da viagem.}
\end{omfigure}

\begin{method}[Ler um gráfico distância--tempo]\label{met:g7:motion-average-speed:read}
\begin{enumerate}
  \item confira primeiro os \omterm{def:g6:sun-earth-moon:axisorbit}{eixos} e as \omterm{def:g6:measurement-in-science:measuring}{unidades} deles --- minutos ou
        \omterm{ex:g2:measuring-time:clock}{horas}, \omterm{def:g2:measuring-length:units}{metros} ou quilômetros;
  \item divida a linha nos pontos em que ela dobra, em trechos;
        classifique cada um: reto e inclinado, reto e suave,
        horizontal;
  \item para qualquer trecho reto, leia a subida e o avanço dele ---
        distância ganha, tempo gasto --- e divida: é a \omterm{def:g7:motion-average-speed:formula}{velocidade} do
        trecho;
  \item para a história inteira, leia a distância total no instante
        final --- e lembre que os trechos horizontais fazem parte do
        tempo total.
\end{enumerate}
\end{method}

\begin{example}[O entregador, decifrado]\label{ex:g7:motion-average-speed:decoded}
Aplique o método à figura. Primeiro trecho: \qty{10}{km} em $20$
minutos --- um terço de hora --- então
$v = 10 \div \tfrac{1}{3} = \qty{30}{km/h}$. Segundo: horizontal do
minuto $20$ ao $35$ --- uma parada de entrega. Terceiro:
\qty{10}{km} em $25$ minutos, um pouquinho abaixo de \qty{24}{km/h}.
Total: \qty{20}{km} em uma hora --- o que nos entrega de bandeja a
próxima ideia do capítulo.
\end{example}

\section{Velocidade média --- e a armadilha famosa dela}

\begin{definition}[Velocidade média]\label{def:g7:motion-average-speed:average}
A \emph{velocidade média}\index{velocidade média} de uma viagem
inteira é a distância total dividida pelo tempo \emph{total} ---
paradas incluídas:
\[
v_{\text{média}} = \frac{d_{\text{total}}}{t_{\text{total}}}.
\]
Ela é o único ritmo constante que teria percorrido a mesma estrada no
mesmo tempo total --- os \qty{20}{km} do entregador em uma hora:
média de \qty{20}{km/h}, embora as rodas nunca tenham girado nem uma
vez nessa \omterm{def:g7:motion-average-speed:formula}{velocidade}.
\end{definition}

\begin{proposition}[A armadilha]\label{prop:g7:motion-average-speed:trap}
A \omterm{def:g7:motion-average-speed:average}{velocidade média} de uma viagem \emph{não} é, em geral, o ponto
médio entre as \omterm{def:g7:motion-average-speed:formula}{velocidades} dela. Os trechos lentos comem mais tempo
que os rápidos, e por isso pesam mais na média. Só a receita completa
--- distância total sobre tempo total --- é confiável; tirar a média
dos próprios números de \omterm{def:g7:motion-average-speed:formula}{velocidade} é o passo errado mais sedutor do
capítulo.
\end{proposition}

\begin{example}[A armadilha, acionada]\label{ex:g7:motion-average-speed:sprung}
Um ciclista pedala \qty{60}{km} na ida a \qty{30}{km/h} e os mesmos
\qty{60}{km} na volta a \qty{20}{km/h}. ``Média: \qty{25}{km/h}''?
Confira com honestidade. Ida: $t = 60 \div 30 = 2$ \omterm{ex:g2:measuring-time:clock}{horas}. Volta:
$t = 60 \div 20 = 3$ \omterm{ex:g2:measuring-time:clock}{horas}. Viagem inteira: \qty{120}{km} em $5$
\omterm{ex:g2:measuring-time:clock}{horas} --- $v_{\text{média}} = \qty{24}{km/h}$. A metade lenta
reivindicou três das cinco \omterm{ex:g2:measuring-time:clock}{horas} e puxou a média para baixo do ponto
médio. Quanto mais rápido você vai numa metade, menos tempo essa
metade sequer existe.
\end{example}

\begin{remark}[O que o velocímetro sabe]\label{rem:g7:motion-average-speed:speedometer}
A \omterm{def:g7:motion-average-speed:average}{velocidade média} descreve uma viagem inteira; o ponteiro do
velocímetro responde a outra pergunta --- \emph{quão rápido agora}.
No gráfico, o ``agora'' mora na inclinação da linha \emph{num único
ponto} --- fácil de ver num trecho reto, sutil onde a linha faz
curva. Tornar precisa a ``inclinação num ponto'' é uma das maiores
invenções da matemática, e ela espera por você no fim do ensino
médio. Até lá: os trechos retos ganham números, as curvas ganham
histórias.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:1}
Escreva a fórmula da \omterm{def:g7:motion-average-speed:formula}{velocidade} e as duas \omterm{def:g6:measurement-in-science:measuring}{unidades} do dia a dia dela.
Qual receita encontra uma distância? E um tempo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:2}
Uma balsa percorre \qty{45}{km} em $3$ \omterm{ex:g2:measuring-time:clock}{horas}; uma lebre corre
\qty{100}{m} em \qty{8}{s}. Calcule as duas \omterm{def:g7:motion-average-speed:formula}{velocidades}, cada uma na
\omterm{def:g6:measurement-in-science:measuring}{unidade} natural dela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:3}
Converta, com a taxa de câmbio: \qty{5}{m/s} para \unit{km/h}; \
\qty{72}{km/h} para \unit{m/s}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:4}
Enuncie as três leis do gráfico. O que uma curva que vai ficando
menos inclinada aos poucos conta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:5}
No gráfico do entregador: entre quais minutos o ritmo é o mais suave
(mas não nulo)? Como dá para saber sem calcular?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:6}
O gráfico de um caminhante mostra uma reta passando pelos pontos
($30$ min, \qty{2}{km}) e ($60$ min, \qty{4}{km}). Uniforme ou
variado? Qual a \omterm{def:g7:motion-average-speed:formula}{velocidade} em \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:motion-average-speed:7}
Defina \omterm{def:g7:motion-average-speed:average}{velocidade média}. Uma trilha: \qty{12}{km} em $4$ \omterm{ex:g2:measuring-time:clock}{horas},
incluindo um piquenique de uma hora. Qual é a \omterm{def:g7:motion-average-speed:average}{velocidade média} --- e
a \omterm{def:g7:motion-average-speed:average}{velocidade média} \emph{enquanto se caminhava}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:8}
Por que um trecho lento é ``mais pesado'' numa média do que um trecho
rápido de mesmo comprimento? Responda com tempo, e não com fórmulas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:9}
Reencene a armadilha: \qty{30}{km} de ida a \qty{15}{km/h} e
\qty{30}{km} de volta a \qty{30}{km/h}. Ponto médio previsto, média
honesta --- e qual metade da viagem foi dona da maior parte do
relógio?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:10}
Esboce (ou descreva com precisão) o gráfico desta viagem:
\qty{40}{km/h} uniformes por meia hora; parado por um quarto de hora;
\qty{60}{km/h} uniformes por um quarto de hora. Depois calcule a
\omterm{def:g7:motion-average-speed:average}{velocidade média} da viagem.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:motion-average-speed:11}
A contagem da tempestade, aprimorada: o trovão chega \qty{6}{s}
depois do clarão. Com o \omterm{def:g3:sound-around-us:sound}{som} a \qty{340}{m/s}, use $d = v \times t$
para a distância da tempestade --- e confira a velha regra do
``dividir por três para dar quilômetros'' contra a sua fórmula.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:motion-average-speed:12}
Um clássico que vale cada minuto: um motorista percorre a primeira
metade \emph{da distância} de uma viagem a \qty{30}{km/h} e quer uma
média geral de \qty{60}{km/h}. Mostre --- com a receita da distância
total sobre o tempo total, numa viagem de \qty{60}{km} --- que a
segunda metade teria de ser percorrida em tempo zero: o desejo é
impossível, e não apenas difícil.
\end{exercise}

\section{Problema: A sexta-feira do entregador}

\begin{problem}[{Problema de fim de semana --- uma entregadora de
bicicleta, uma sexta-feira na cidade; o gráfico do despachante, a
armadilha na planilha de bônus}]\label{pb:g7:motion-average-speed:1}
O despachante registra a sexta-feira da entregadora Lena num gráfico
distância--tempo e calcula o pagamento dela a partir dele. Você
audita a planilha.

\medskip\noindent\textbf{Parte I --- A manhã, pelo gráfico.}
O gráfico mostra: uma subida reta de ($0$ min, \qty{0}{km}) até
($30$ min, \qty{9}{km}); horizontal até $45$ min; reta dali até
($75$ min, \qty{15}{km}); horizontal até $90$ min.
\begin{enumerate}
  \item Conte a história da manhã em palavras: trechos, paradas e o
        que as paradas presumivelmente foram.
  \item Qual a \omterm{def:g7:motion-average-speed:formula}{velocidade} no primeiro trecho, em \unit{km/h}?
  \item Qual a \omterm{def:g7:motion-average-speed:formula}{velocidade} no segundo trecho de pedalada?
  \item Qual a \omterm{def:g7:motion-average-speed:formula}{velocidade} \emph{média} de Lena ao longo dos $90$
        minutos inteiros da manhã, paradas incluídas?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A tarde, pela fórmula.}
\begin{enumerate}[resume]
  \item Perna um da tarde: \qty{12}{km} a \qty{24}{km/h} constantes.
        Quanto tempo levou, em minutos?
  \item Perna dois: uma entrega na parte alta da cidade,
        \qty{20}{minutos} a \qty{18}{km/h}. Quantos quilômetros?
  \item Perna três: a pedalada longa até o depósito, \qty{10}{km},
        feita em $25$ minutos. Qual a \omterm{def:g7:motion-average-speed:formula}{velocidade} constante em
        \unit{km/h}?
  \item Total da tarde: some as distâncias e os tempos (só as pernas,
        sem pausas): que \omterm{def:g7:motion-average-speed:average}{velocidade média} as rodas mantiveram
        enquanto rodavam?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A armadilha da planilha de
bônus.}
A regra do bônus: ``\omterm{def:g7:motion-average-speed:average}{velocidade média} acima de \qty{20}{km/h} numa
volta completa ganha o bônus de pedalada rápida''.
\begin{enumerate}[resume]
  \item A sexta-feira inteira de Lena: \qty{37}{km} em $3$ \omterm{ex:g2:measuring-time:clock}{horas}
        (todas as paradas incluídas). Ela ganha o bônus?
  \item O colega dela, Max, reivindica o bônus: ``Pedalei metade da
        minha volta a \qty{30}{km/h} e metade a \qty{15}{km/h} ---
        isso dá média de $22.5$!'' O despachante confere: as duas
        metades tinham \qty{12}{km}. Calcule a média verdadeira de
        Max e decida sobre o bônus dele.
  \item Explique a Max, na linguagem do tempo, onde os $22.5$ dele
        deram errado.
  \item Lena propõe uma regra de bônus mais justa para entregadores
        --- uma que não castigue as paradas de entrega. Sugira uma (o
        gráfico sabe como) e diga qual \omterm{def:g7:motion-average-speed:formula}{velocidade} ela mede.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- A lição do despachante.}
\begin{enumerate}[resume]
  \item Revisão da sexta-feira: escreva as três frases do
        despachante --- o que um trecho reto, um trecho horizontal e
        um trecho mais inclinado significam no gráfico --- e o único
        aviso sobre tirar média de \omterm{def:g7:motion-average-speed:formula}{velocidades}.
\end{enumerate}
\end{problem}
