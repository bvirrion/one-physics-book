\chapter{Circuitos: voltas, série, paralelo}\label{ch:g7:circuits-series-parallel}

Quatro anos de circuitos encheram a sua caixa de ferramentas: voltas,
esquemas, as duas grandes ligações. Mas um personagem trabalhou sem
nome esse tempo todo --- aquilo que flui. Neste ano ele recebe o
nome, o sentido e a propriedade mais estranha dele: ele nunca é
gasto. Com a corrente em cena, série e paralelo deixam de ser receitas
e viram leis.

\section{A corrente entra em cena}

\begin{definition}[Corrente elétrica]\label{def:g7:circuits-series-parallel:current}
A \emph{corrente elétrica}\index{corrente elétrica} é o fluxo que
corre por um \omterm{def:g3:first-electric-circuit:circuit}{circuito fechado}: uma multidão de partículas carregadas
--- inimaginavelmente numerosas, inimaginavelmente pequenas, primas
do mundo molecular --- postas a marchar pelo metal dos fios pelo
empurrão da \omterm{def:g3:first-electric-circuit:battery}{pilha}. Sem volta, sem marcha: a corrente só existe
enquanto a estrada está fechada. (O que exatamente são os
marchadores, e como um metal os deixa passar, é uma bela história
reservada para os últimos anos deste livro.)
\end{definition}

\begin{proposition}[O sentido convencional]\label{prop:g7:circuits-series-parallel:direction}
Por convenção mundial, os \omterm{def:g6:circuits-and-safety:diagram}{esquemas elétricos} marcam a corrente como
saindo \emph{do \omterm{def:g3:first-electric-circuit:battery}{terminal} $+$ da \omterm{def:g3:first-electric-circuit:battery}{pilha}, por fora dela, dando a volta
pelo circuito, até o \omterm{def:g3:first-electric-circuit:battery}{terminal} $-$}. Toda seta de todo esquema do
mundo obedece a essa convenção --- escolhida muito antes de alguém
poder perguntar aos próprios marchadores. (Quando a física enfim pôde
perguntar, a resposta guardava uma pequena piada famosa, guardada
para um ano mais adiante.)
\end{proposition}

\begin{proposition}[A corrente não é consumida]\label{prop:g7:circuits-series-parallel:notconsumed}
Os dispositivos não comem a corrente. Toda corrente que entra numa
lâmpada, num \omterm{ex:g6:circuits-and-safety:motor}{motor} ou numa \omterm{ex:g6:circuits-and-safety:motor}{campainha} sai inteira, marchando adiante:
o que o dispositivo pega é \emph{\omterm{def:g5:everyday-energy:energy}{energia}} da reserva da \omterm{def:g3:first-electric-circuit:battery}{pilha},
entregue pela marcha que passa. Numa volta \omterm{def:g4:series-circuits:series}{em série}, portanto, uma
única e mesma corrente flui por todos os dispositivos --- volta para
a \omterm{def:g3:first-electric-circuit:battery}{pilha} tanto quanto saiu dela.
\end{proposition}

\begin{example}[A ideia errada em julgamento]\label{ex:g7:circuits-series-parallel:trial}
O erro tentador: ``a primeira lâmpada gasta um pouco de corrente,
então a segunda tem de brilhar mais fraco''. O próprio circuito o
refuta. Ligue duas lâmpadas \emph{iguais} \omterm{def:g4:series-circuits:series}{em série} e observe: as duas
brilham exatamente com o mesmo brilho --- e trocá-las de lugar não
muda nada. Se a corrente fosse consumida no caminho, a lâmpada dois
deveria ser sempre a mais fraca, seja qual for a lâmpada colocada
ali. Nunca é. A corrente chega inteira; o que a dupla reparte é o
\emph{empurrão} da \omterm{def:g3:first-electric-circuit:battery}{pilha}, e é por isso que as duas são mais fracas
que uma lâmpada sozinha --- igualmente mais fracas.
\end{example}

\section{As duas famílias, como leis}

\begin{proposition}[A lei da série]\label{prop:g7:circuits-series-parallel:series}
Num \omterm{def:g4:series-circuits:series}{circuito em série} --- uma única volta --- a mesma corrente flui
por todos os dispositivos: uma estrada, uma marcha, sem exceção.
Consequências, agora explicadas: uma falha para a marcha em toda
parte ao mesmo tempo; dispositivos iguais repartem o trabalho por
igual, esteja cada um onde estiver; e a ordem dos dispositivos na
volta não pode importar para uma marcha que atravessa todos eles do
mesmo jeito.
\end{proposition}

\begin{proposition}[A lei do paralelo]\label{prop:g7:circuits-series-parallel:parallel}
Num \omterm{def:g5:series-parallel-circuits:parallel}{nó}, a corrente \emph{se divide}: uma parte da marcha entra em
cada \omterm{def:g5:series-parallel-circuits:parallel}{ramo}, e as partes se juntam de novo, completas, no \omterm{def:g5:series-parallel-circuits:parallel}{nó} distante
--- nada perdido, nada ganho. Cada \omterm{def:g5:series-parallel-circuits:parallel}{ramo} recebe o empurrão inteiro da
\omterm{def:g3:first-electric-circuit:battery}{pilha}, então cada dispositivo funciona como se estivesse sozinho; e
um \omterm{def:g5:series-parallel-circuits:parallel}{ramo} bloqueado apenas não manda ninguém, enquanto os outros \omterm{def:g5:series-parallel-circuits:parallel}{ramos}
continuam marchando.
\end{proposition}

\begin{example}[Ler o brilho como um eletricista]\label{ex:g7:circuits-series-parallel:brightness}
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, lâmpadas iguais. Sozinha: brilho total --- a referência.
Duas \omterm{def:g4:series-circuits:series}{em série}: as duas igualmente fracas --- uma marcha, empurrão
repartido. Duas \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo}: as duas no brilho total --- cada \omterm{def:g5:series-parallel-circuits:parallel}{ramo}
aproveita o empurrão inteiro (e a \omterm{def:g3:first-electric-circuit:battery}{pilha} se esvazia mais ou menos duas
vezes mais depressa: nada é de graça). Três \omterm{def:g4:series-circuits:series}{em série}: mais fracas
ainda, todas iguais. O olhar do eletricista: conte os companheiros de
volta para saber a fraqueza, conte os \omterm{def:g5:series-parallel-circuits:parallel}{ramos} para saber a conta da
\omterm{def:g3:first-electric-circuit:battery}{pilha}.
\end{example}

\begin{method}[Prever o destino de qualquer lâmpada]\label{met:g7:circuits-series-parallel:predict}
Diante de um esquema, para cada lâmpada:
\begin{enumerate}
  \item \emph{Existência}: há pelo menos uma volta sem interrupção do
        $+$ até o $-$ passando por esta lâmpada? Sem volta, sem \omterm{def:g1:light-and-shadows:source}{luz}.
  \item \emph{Companhia na estrada}: quais dispositivos compartilham
        \emph{todas} as voltas que passam por esta lâmpada? Esses são
        os companheiros de série dela --- mais companheiros, brilho
        mais fraco.
  \item \emph{Vizinhos de \omterm{def:g5:series-parallel-circuits:parallel}{ramo}}: quais dispositivos ficam em outros
        \omterm{def:g5:series-parallel-circuits:parallel}{ramos}? Eles nem enfraquecem esta lâmpada nem apagam junto com
        ela.
  \item \emph{\omterm{ex:g3:first-electric-circuit:switch}{Interruptores}}: um \omterm{ex:g3:first-electric-circuit:switch}{interruptor} aberto mata exatamente
        as voltas em que ele está --- refaça o passo 1 sem elas.
\end{enumerate}
\end{method}

\begin{remark}[Por que a ideia errada custa a morrer]\label{rem:g7:circuits-series-parallel:whyhard}
``Alguma coisa tem de ser gasta --- a \omterm{def:g3:first-electric-circuit:battery}{pilha} se esvazia!'' Verdade: a
\emph{\omterm{def:g5:everyday-energy:energy}{energia}} armazenada da \omterm{def:g3:first-electric-circuit:battery}{pilha} é gasta, entregue às lâmpadas como
\omterm{def:g1:light-and-shadows:source}{luz} e \omterm{def:g5:heat-and-insulation:heat}{calor} pela marcha que passa. Mas os próprios marchadores
circulam ilesos, como as cadeiras de um teleférico levando
esquiadores morro acima o dia inteiro sem que uma cadeira sequer seja
consumida. Mantenha as duas contabilidades separadas --- a corrente
circula, a \omterm{def:g5:everyday-energy:energy}{energia} é entregue --- e todo circuito deste livro
continua honesto. O instrumento que conta a própria marcha se junta a
\omterm{def:g5:series-parallel-circuits:parallel}{nós} no ano que vem.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:1}
O que é a \omterm{def:g7:circuits-series-parallel:current}{corrente elétrica}? Quando ela existe num circuito, e quando
não existe?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:2}
Enuncie o sentido convencional da corrente. Num esquema de uma volta
com uma lâmpada, para onde as setas apontam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:3}
Duas lâmpadas iguais \omterm{def:g4:series-circuits:series}{em série} brilham exatamente igual. Qual erro
popular essa igualdade refuta, e como?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:4}
Numa volta \omterm{def:g4:series-circuits:series}{em série} com um \omterm{ex:g6:circuits-and-safety:motor}{motor} e uma lâmpada, compara-se a corrente
que passa pelo \omterm{ex:g6:circuits-and-safety:motor}{motor} com a que passa pela lâmpada. O que a lei da
série diz?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:5}
O que acontece com a corrente no primeiro \omterm{def:g5:series-parallel-circuits:parallel}{nó} de um \omterm{def:g5:series-parallel-circuits:parallel}{circuito em
paralelo}? E no segundo? O que é verdade sobre a parte do empurrão da
\omterm{def:g3:first-electric-circuit:battery}{pilha} que cabe a cada \omterm{def:g5:series-parallel-circuits:parallel}{ramo}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:6}
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha}, lâmpadas iguais: coloque em ordem o brilho de uma lâmpada
--- sozinha; \ com um companheiro \omterm{def:g4:series-circuits:series}{em série}; \ com um vizinho \omterm{def:g5:series-parallel-circuits:parallel}{em
paralelo}; \ com dois companheiros \omterm{def:g4:series-circuits:series}{em série}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:7}
Qual arranjo esvazia a \omterm{def:g3:first-electric-circuit:battery}{pilha} mais depressa: duas lâmpadas \omterm{def:g4:series-circuits:series}{em série} ou
as mesmas duas \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo}? Por quê, na linguagem da marcha?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:circuits-series-parallel:8}
Explique a \omterm{prop:g5:mirrors-reflection:image}{imagem} do teleférico da
\cref{rem:g7:circuits-series-parallel:whyhard}: o que faz o papel das
cadeiras, o que faz o papel dos esquiadores e o que realmente acaba?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:9}
Um esquema: \omterm{def:g3:first-electric-circuit:battery}{pilha}, \omterm{ex:g3:first-electric-circuit:switch}{interruptor} e depois a lâmpada A; depois de A um
\omterm{def:g5:series-parallel-circuits:parallel}{nó} manda dois \omterm{def:g5:series-parallel-circuits:parallel}{ramos} --- a lâmpada B num deles, a lâmpada C no outro
--- que se juntam antes de voltar à \omterm{def:g3:first-electric-circuit:battery}{pilha}. Usando o
\cref{met:g7:circuits-series-parallel:predict}: quem são os
companheiros de série de A? O que o brilho de A faz quando o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} de
C é cortado?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:10}
Umas luzinhas decorativas pedem um projeto: dez lâmpadas que brilhem
todas no brilho total, de forma independente, a partir de uma única
fonte --- mas o conjunto inteiro precisa obedecer a um único
\omterm{ex:g3:first-electric-circuit:switch}{interruptor} geral. Desenhe ou descreva a ligação, citando as duas
leis.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:circuits-series-parallel:11}
No \omterm{def:g5:series-parallel-circuits:parallel}{circuito em paralelo} de duas lâmpadas, o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} da lâmpada 1 é de
fio grosso de cobre, enquanto o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} da lâmpada 2 inclui um fio
comprido e fino. A lâmpada 2 brilha um pouco mais fraco. O que isso
sugere sobre como a própria estrada de um \omterm{def:g5:series-parallel-circuits:parallel}{ramo} pode estrangular a
parte da marcha que cabe a ele? (O ano que vem dá a esse
estrangulamento um nome e uma lei.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:circuits-series-parallel:12}
Uma \omterm{def:g3:first-electric-circuit:battery}{pilha} alimenta dois \omterm{def:g5:series-parallel-circuits:parallel}{ramos} \omterm{def:g5:series-parallel-circuits:parallel}{em paralelo}: o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} um contém a lâmpada
X sozinha; o \omterm{def:g5:series-parallel-circuits:parallel}{ramo} dois contém as lâmpadas Y e Z \omterm{def:g4:series-circuits:series}{em série}. Compare os
brilhos das três lâmpadas, em ordem, com justificativas das duas
leis. Depois preveja o destino de cada lâmpada se Z queimar --- e se,
em vez disso, X queimar.
\end{exercise}

\section{Problema: O painel do faroleiro}

\begin{problem}[{Problema de fim de semana --- refazendo a
instalação do velho farol; um banco de pilhas, muitas tarefas; a
noite de defeitos do faroleiro}]\label{pb:g7:circuits-series-parallel:1}
O banco de \omterm{def:g3:first-electric-circuit:battery}{pilhas} do farol precisa alimentar: a grande lâmpada, uma
sirene de neblina (uma \omterm{ex:g6:circuits-and-safety:motor}{campainha} possante), a \omterm{def:g1:light-and-shadows:source}{luz} de leitura do
faroleiro e a \omterm{def:g1:light-and-shadows:source}{luz} da escada --- e sobreviver à noite de investigação
dele.

\medskip\noindent\textbf{Parte I --- Projeto.}
\begin{enumerate}
  \item A grande lâmpada nunca pode enfraquecer porque outra coisa
        foi ligada. Que tipo de ligação isso exige para a instalação
        inteira, e por que qualquer emparelhamento \omterm{def:g4:series-circuits:series}{em série} trairia
        os navios?
  \item Cada tarefa precisa do próprio \omterm{ex:g3:first-electric-circuit:switch}{interruptor}, e o faroleiro
        quer um corte geral para as tempestades. Posicione os cinco
        \omterm{ex:g3:first-electric-circuit:switch}{interruptores}.
  \item A sirene de neblina e a \omterm{def:g1:light-and-shadows:source}{luz} da escada nunca podem ser
        impedidas de funcionar pelos defeitos uma da outra. Isso já
        está garantido? Por qual lei?
  \item Esboce ou descreva o esquema completo: banco, \omterm{ex:g3:first-electric-circuit:switch}{interruptor}
        geral, quatro \omterm{def:g5:series-parallel-circuits:parallel}{ramos}, quatro \omterm{ex:g3:first-electric-circuit:switch}{interruptores} de \omterm{def:g5:series-parallel-circuits:parallel}{ramo}.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- As perguntas do faroleiro.}
\begin{enumerate}[resume]
  \item O faroleiro se preocupa: ``Quatro tarefas bebendo ao mesmo
        tempo --- cada \omterm{def:g5:series-parallel-circuits:parallel}{ramo} ainda recebe o empurrão inteiro do
        banco?'' Responda com a lei do paralelo.
  \item ``E a corrente está sendo gasta enquanto trabalha na grande
        lâmpada?'' Corrija o faroleiro, com o teleférico se ajudar.
  \item Numa noite de neblina tudo funciona ao mesmo tempo. Qual é o
        preço da generosidade do paralelo, e em que parte do farol
        ele é pago?
  \item Só a \omterm{def:g1:light-and-shadows:source}{luz} de leitura está acesa; com espírito de medidor, o
        faroleiro acompanha a marcha: descreva a viagem completa de
        ida e volta dela, dizendo cada componente por que ela passa e
        cada um que ela evita.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A noite dos defeitos.}
\begin{enumerate}[resume]
  \item Meia-noite: a lâmpada da escada queima. O que mais apaga?
        Qual lei responde?
  \item Uma hora: um conserto desastrado deixa um trecho de fio nu
        ligando diretamente os dois \omterm{def:g3:first-electric-circuit:battery}{terminais} do banco, antes de
        todos os \omterm{def:g5:series-parallel-circuits:parallel}{ramos}. Diga o nome do acontecimento e explique por
        que agora todas as tarefas do farol falham ao mesmo tempo ---
        para onde vai a marcha?
  \item Duas \omterm{ex:g2:measuring-time:clock}{horas}, defeito resolvido: o faroleiro quer a grande
        lâmpada protegida, para que um atalho desses nunca mais possa
        silenciá-la. Há um segundo banco de \omterm{def:g3:first-electric-circuit:battery}{pilhas} independente
        disponível. Proponha a ligação.
  \item Amanhecer: escreva o diário do faroleiro --- três frases:
        qual lei manteve os navios seguros esta noite, qual ideia
        errada a noite refutou e qual único fio causou a pior hora.
\end{enumerate}
\end{problem}
