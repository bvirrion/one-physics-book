\chapter{Volume, massa, densidade}\label{ch:g7:volume-mass-density}

No ano passado você desmascarou um ourives fraudulento dividindo
\omterm{def:g3:measuring-mass:units}{gramas} por \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}. Neste ano, o seu curso de matemática
entregou a você um poder novo: letras que representam números. A
\omterm{def:g7:volume-mass-density:formula}{densidade} é o lugar perfeito para gastá-lo --- uma fórmula curtinha
que empacota três receitas, um kit de detetive e a regra inteira da
flutuação em cinco símbolos.

\section{A fórmula}

\begin{definition}[Densidade, por fórmula]\label{def:g7:volume-mass-density:formula}
A \emph{densidade}\index{densidade} $\rho$ (a letra grega
\emph{rô}) de uma substância é a \omterm{def:g3:measuring-mass:mass}{massa} dela por \omterm{def:g6:measurement-in-science:measuring}{unidade} de \omterm{def:g6:mass-and-volume:volume}{volume}:
para uma amostra de \omterm{def:g3:measuring-mass:mass}{massa} $m$ e \omterm{def:g6:mass-and-volume:volume}{volume} $V$,
\[
\rho = \frac{m}{V}.
\]
Com $m$ em \omterm{def:g3:measuring-mass:units}{gramas} e $V$ em \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}, $\rho$ sai em
\emph{\omterm{def:g3:measuring-mass:units}{gramas} por \omterm{def:g6:mass-and-volume:volume}{centímetro cúbico}} (\unit{g/cm^3}) --- a ``\omterm{def:g3:measuring-mass:mass}{massa} de
um \omterm{def:g6:mass-and-volume:volume}{centímetro cúbico}'' do ano passado, agora de uniforme. A densidade
da água: $\rho = \qty{1.0}{g/cm^3}$. E, como um \omterm{def:g6:mass-and-volume:volume}{litro} são $1000$
\omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}, o mesmo número serve também como \omterm{def:g3:measuring-mass:units}{quilogramas} por
\omterm{def:g6:mass-and-volume:volume}{litro}: a água tem \qty{1.0}{kg} por \omterm{def:g6:mass-and-volume:volume}{litro}.
\end{definition}

\begin{example}[Lendo as letras]\label{ex:g7:volume-mass-density:reading}
Uma fórmula é uma frase em taquigrafia. $\rho = m/V$ se lê: ``para
encontrar a \omterm{def:g7:volume-mass-density:formula}{densidade}, divida a \omterm{def:g3:measuring-mass:mass}{massa} da amostra pelo \omterm{def:g6:mass-and-volume:volume}{volume} dela''.
As letras são lugares vazios --- para \emph{este} seixo, $m$ vira
\qty{75}{g} e $V$ vira \qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
A mesma divisão do ano passado --- mas a fórmula guarda a receita
para todos os seixos que ainda vierem.
\end{example}

\begin{proposition}[Três receitas numa fórmula]\label{prop:g7:volume-mass-density:recipes}
Como a \omterm{def:g7:volume-mass-density:formula}{densidade} amarra a \omterm{def:g3:measuring-mass:mass}{massa} e o \omterm{def:g6:mass-and-volume:volume}{volume} em proporção, uma única
fórmula atende a três necessidades:
\begin{enumerate}
  \item \emph{encontrar uma \omterm{def:g7:volume-mass-density:formula}{densidade}}: $\rho = m/V$ --- divida a
        \omterm{def:g3:measuring-mass:mass}{massa} pelo \omterm{def:g6:mass-and-volume:volume}{volume};
  \item \emph{encontrar uma \omterm{def:g3:measuring-mass:mass}{massa}}: $m = \rho \times V$ --- cada
        \omterm{def:g6:mass-and-volume:volume}{centímetro cúbico} carrega $\rho$ \omterm{def:g3:measuring-mass:units}{gramas}, e há $V$ deles;
  \item \emph{encontrar um \omterm{def:g6:mass-and-volume:volume}{volume}}: $V = m/\rho$ --- reparta a \omterm{def:g3:measuring-mass:mass}{massa}
        em porções de $\rho$ \omterm{def:g3:measuring-mass:units}{gramas}; o número de porções é o número
        de \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}.
\end{enumerate}
\end{proposition}

\begin{example}[As receitas em ação]\label{ex:g7:volume-mass-density:atwork}
\emph{\omterm{def:g3:measuring-mass:mass}{Massa} a partir do \omterm{def:g6:mass-and-volume:volume}{volume}}: quanto pesam \qty{250}{cm^3} de
ferro ($\rho = 7.9$)? $m = 7.9 \times 250 = \qty{1975}{g}$ --- quase
dois \omterm{def:g3:measuring-mass:units}{quilogramas} no \omterm{def:g6:mass-and-volume:volume}{volume} de uma caneca de café.
\emph{\omterm{def:g6:mass-and-volume:volume}{Volume} a partir da \omterm{def:g3:measuring-mass:mass}{massa}}: que espaço ocupam \qty{540}{g} de
carvalho ($\rho = 0.6$)? $V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{Parceiro do bom senso}: as respostas do ferro devem ser
pesadas-e-pequenas, as da madeira leves-e-grandes --- a tabela de
\omterm{def:g7:volume-mass-density:formula}{densidades} é a melhor amiga do passo do julgamento.
\end{example}

\section{Medir densidades}

\begin{method}[Densidade de um líquido]\label{met:g7:volume-mass-density:liquid}
Os \omterm{def:g3:solids-liquids-gases:liquid}{líquidos} não ficam sozinhos no prato de uma balança --- então
empreste o truque da tara:
\begin{enumerate}
  \item ponha uma proveta vazia na balança e tare para zero;
  \item despeje um \omterm{def:g6:mass-and-volume:volume}{volume} conveniente --- digamos
        $V = \qty{80}{cm^3}$ do \omterm{def:g3:solids-liquids-gases:liquid}{líquido} misterioso, lendo o \omterm{met:g6:mass-and-volume:cylinder}{menisco}
        direito;
  \item a balança agora mostra só a \omterm{def:g3:measuring-mass:mass}{massa} do \omterm{def:g3:solids-liquids-gases:liquid}{líquido}: digamos
        $m = \qty{68}{g}$;
  \item divida: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ --- mais
        leve que a água: provavelmente um óleo.
\end{enumerate}
\end{method}

\begin{example}[Densidade de um sólido, do começo ao fim]\label{ex:g7:volume-mass-density:solid}
Uma medalha afirma ser de prata ($\rho = 10.5$). Balança:
$m = \qty{84}{g}$. Proveta: a água sobe de \qty{60.0}{cm^3} para
\qty{68.0}{cm^3}, então $V = \qty{8.0}{cm^3}$. Aí
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- a assinatura bate: bem
pode ser prata. (Um metal mais barato vestido de banho de prata teria
se traído aqui --- a não ser que fosse escolhido com astúcia, como o
problema de fim de semana vai mostrar.)
\end{example}

\begin{remark}[Unidades nas fórmulas]\label{rem:g7:volume-mass-density:units}
Uma fórmula só é honesta se as \omterm{def:g6:measurement-in-science:measuring}{unidades} dela combinarem. Com $m$ em
\omterm{def:g3:measuring-mass:units}{gramas} e $V$ em \unit{cm^3}, $\rho$ fala em \unit{g/cm^3}; alimente-a
com \omterm{def:g3:measuring-mass:units}{quilogramas} e \omterm{def:g6:mass-and-volume:volume}{litros} e ela responde em \unit{kg/L} --- felizmente,
o mesmo número para qualquer substância. Mas misture \omterm{def:g3:measuring-mass:units}{gramas} com
\omterm{def:g6:mass-and-volume:volume}{litros} e a fórmula, sem reclamar, entrega uma bobagem. Regra dos
profissionais: antes de calcular, passe as \omterm{def:g6:measurement-in-science:measuring}{unidades} em revista;
depois de calcular, escreva-as na resposta.
\end{remark}

\section{Pensar por densidade}

\begin{example}[Ligas e intermediários]\label{ex:g7:volume-mass-density:alloys}
Misture duas substâncias e a \omterm{def:g7:volume-mass-density:formula}{densidade} da mistura cai \emph{entre} as
assinaturas delas --- mais perto do ingrediente mais generoso. O
bronze (cobre $9.0$ com um pouco de estanho) assina perto de $8.8$; a
mistura de ouro com prata da coroa assinou entre $10.5$ e $19.3$; a
água do mar, sal dissolvido em água, sobe para cerca de $1.03$. Estar
no meio já é uma pista: uma leitura de $12$ numa barra de ``ouro
puro'' é uma confissão de companhia.
\end{example}

\begin{example}[A densidade decide o mundo da flutuação]\label{ex:g7:volume-mass-density:floating}
A regra da flutuação, agora de uniforme: um objeto \omterm{def:g1:floating-and-sinking:words}{flutua} num \omterm{def:g3:solids-liquids-gases:liquid}{líquido}
quando $\rho_{\text{objeto}} < \rho_{\text{líquido}}$. Carvalho
($0.6$) na água ($1.0$): \omterm{def:g1:floating-and-sinking:words}{flutua}. Gelo ($0.9$) no óleo ($0.85$):
\omterm{def:g1:floating-and-sinking:words}{afunda} --- consulte a tabela, e não os seus instintos. Um nadador
($\approx 1.0$) nos famosos lagos ultrassalgados ($\approx 1.2$):
\omterm{def:g1:floating-and-sinking:words}{flutua} como uma rolha, com o jornal na mão. Uma desigualdade, o porto
inteiro.
\end{example}

\begin{remark}[Olhe a multidão, não o pacote]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ descreve \emph{substâncias}; os navios e os coletes
salva-vidas inflados são \emph{pacotes} --- substância mais \omterm{def:g2:air-around-us:air}{ar} preso
--- e é o $m/V$ geral do pacote que enfrenta a regra da flutuação. A
fórmula dá conta dos dois, desde que você a alimente com o $m$ e o
$V$ certos: a \omterm{def:g7:volume-mass-density:formula}{densidade} do próprio aço para a substância, a \omterm{def:g3:measuring-mass:mass}{massa} do
casco inteiro sobre o \omterm{def:g6:mass-and-volume:volume}{volume} do casco inteiro para o pacote. Quase
todos os ``paradoxos'' da flutuação são só as duas contabilidades
confundidas.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:1}
Escreva a fórmula da \omterm{def:g7:volume-mass-density:formula}{densidade}, diga o nome de cada letra e dê a
\omterm{def:g6:measurement-in-science:measuring}{unidade} que acompanha \omterm{def:g3:measuring-mass:units}{gramas} e \omterm{def:g6:mass-and-volume:volume}{centímetros cúbicos}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:2}
Calcule a \omterm{def:g7:volume-mass-density:formula}{densidade}: $m = \qty{270}{g}$, $V = \qty{100}{cm^3}$. Qual
metal do gráfico é esse?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:3}
Use a receita certa: a \omterm{def:g3:measuring-mass:mass}{massa} de \qty{40}{cm^3} de cobre
($\rho = 9.0$); \ o \omterm{def:g6:mass-and-volume:volume}{volume} de \qty{1930}{g} de ouro
($\rho = 19.3$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:4}
No \cref{met:g7:volume-mass-density:liquid}, por que a proveta é
tarada primeiro? Quais duas leituras alimentam então a fórmula?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:5}
Um \omterm{def:g3:solids-liquids-gases:liquid}{líquido}: $V = \qty{50}{cm^3}$, $m = \qty{70}{g}$. Qual é a
\omterm{def:g7:volume-mass-density:formula}{densidade} dele? Um cubo de gelo ($0.9$) \omterm{def:g1:floating-and-sinking:words}{flutua} ou \omterm{def:g1:floating-and-sinking:words}{afunda} nele?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:6}
Por que a \omterm{def:g7:volume-mass-density:formula}{densidade} da água é o mesmo número em \unit{g/cm^3} e em
\omterm{def:g3:measuring-mass:units}{quilogramas} por \omterm{def:g6:mass-and-volume:volume}{litro}? Qual aviso a
\cref{rem:g7:volume-mass-density:units} prega sobre misturar \omterm{def:g3:measuring-mass:units}{gramas}
com \omterm{def:g6:mass-and-volume:volume}{litros}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:7}
Uma pulseira marcada como ``ouro puro'' tem $m = \qty{58}{g}$ e
$V = \qty{5.0}{cm^3}$. Calcule a \omterm{def:g7:volume-mass-density:formula}{densidade} dela e dê o seu veredito,
com a tabela como testemunha.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:8}
Um \omterm{def:g2:measuring-length:units}{metro} cúbico é um cubo de $100 \times 100 \times 100$ \omterm{def:g2:measuring-length:units}{centímetros}.
Quantos \unit{cm^3} são --- e qual é a \omterm{def:g3:measuring-mass:mass}{massa} de um \omterm{def:g2:measuring-length:units}{metro} cúbico de
água, em \omterm{def:g3:measuring-mass:units}{quilogramas}? (A resposta explica por que camas d'água
preocupam os síndicos.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:9}
Um troféu ``de prata'': $m = \qty{420}{g}$, subida da água de
\qty{200}{cm^3} para \qty{260}{cm^3}. Qual é a \omterm{def:g7:volume-mass-density:formula}{densidade}? Entre quais
duas substâncias da tabela ela cai --- e o que esse estar-no-meio
sussurra?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:10}
Uma garrafa vazia pesa \qty{380}{g}; cheia até a marca de
\qty{75}{cL} com um \omterm{def:g3:solids-liquids-gases:liquid}{líquido} misterioso, ela pesa \qty{1010}{g}.
Encontre a \omterm{def:g7:volume-mass-density:formula}{densidade} do \omterm{def:g3:solids-liquids-gases:liquid}{líquido} em \unit{g/cm^3} e proponha a
identidade dele.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:11}
Uma boia oca de alumínio tem \omterm{def:g6:mass-and-volume:volume}{volume} total de \qty{4000}{cm^3} e \omterm{def:g3:measuring-mass:mass}{massa}
total de \qty{1200}{g}. Calcule a \omterm{def:g7:volume-mass-density:formula}{densidade} do \emph{pacote} e
preveja se ela \omterm{def:g1:floating-and-sinking:words}{flutua} ou \omterm{def:g1:floating-and-sinking:words}{afunda} --- e depois explique, com a
\cref{rem:g7:volume-mass-density:parcel}, por que o $2.7$ do próprio
alumínio era o número errado a consultar.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:volume-mass-density:12}
Projete um protocolo completo para decidir se uma corrente é de cobre
puro: liste cada medida, a receita aplicada, o número esperado para a
pureza e \emph{dois} motivos honestos pelos quais o seu veredito
ainda poderia estar errado (pense em elos ocos e em misturas astutas
cuja \omterm{def:g7:volume-mass-density:formula}{densidade} caia perto da do cobre). Qual teste extra apertaria o
caso?
\end{exercise}

\section{Problema: O detetive do ferro-velho}

\begin{problem}[{Problema de fim de semana --- uma tarde com a
detetive de metais do ferro-velho; quatro lotes misteriosos, um
lingote impostor; os limites do teste de
densidade}]\label{pb:g7:volume-mass-density:1}
O ferro-velho compra metal pelo que ele \emph{é}, e não pelo que
parece, e a detetive do pátio trabalha com uma balança, um recipiente
graduado grande e a tabela de \omterm{def:g7:volume-mass-density:formula}{densidades}. Você é o aprendiz. (Trecho
da tabela: alumínio $2.7$; ferro $7.9$; cobre $9.0$; chumbo $11.3$;
ouro $19.3$; tungstênio $19.3$.)

\medskip\noindent\textbf{Parte I --- Quatro lotes.}
\begin{enumerate}
  \item Lote A, um lingote cinzento: $m = \qty{5400}{g}$,
        deslocamento $V = \qty{2000}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Densidade} e identidade?
  \item Lote B, um rolo de fio: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Densidade} e identidade?
  \item Lote C, uma chapa opaca e pesada: $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Densidade} e identidade --- e por que
        o lote C precisa ser manuseado com luvas e respeito?
  \item Lote D, um saco de sucata clara e misturada:
        $m = \qty{8100}{g}$, $V = \qty{1500}{cm^3}$. Mostre que a
        \omterm{def:g7:volume-mass-density:formula}{densidade} do lote D cai entre dois metais da tabela e diga o
        que o saco provavelmente contém.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Preços e previsões.}
O pátio paga por \omterm{def:g3:measuring-mass:mass}{massa}, mas planeja o transporte por \omterm{def:g6:mass-and-volume:volume}{volume}.
\begin{enumerate}[resume]
  \item Um comprador quer \qty{540}{g} de alumínio cortados do lote
        A. Que \omterm{def:g6:mass-and-volume:volume}{volume} de lingote é isso?
  \item O rolo de cobre do lote B vai ser derretido em cubos de
        \qty{25}{cm^3} cada um. Qual é a \omterm{def:g3:measuring-mass:mass}{massa} de um cubo, e quantos
        cubos cheios o rolo rende?
  \item Um engradado pode carregar no máximo \qty{20}{kg}. Quantos
        desses cubos de cobre ele pode legalmente levar?
  \item O poço do caminhão comporta mais \qty{3000}{cm^3} de sucata
        de ferro. Que \omterm{def:g3:measuring-mass:mass}{massa} extra, em \omterm{def:g3:measuring-mass:units}{quilogramas}, a detetive está
        autorizada a carregar?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O impostor.}
Um vendedor chega com um lingote reluzente de ``ouro'':
$m = \qty{3860}{g}$, deslocamento $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item Calcule a \omterm{def:g7:volume-mass-density:formula}{densidade} do lingote. Ela bate com a assinatura do
        ouro?
  \item A detetive, impassível, consulta a última linha da tabela e
        suspira. Qual metal mais barato tem exatamente a \omterm{def:g7:volume-mass-density:formula}{densidade} do
        ouro --- e o que isso ensina sobre os limites do teste de
        \omterm{def:g7:volume-mass-density:formula}{densidade}?
  \item Qual dos dois instrumentos do pátio foi derrotado aqui: a
        balança, o recipiente, os dois ou nenhum? Diga com precisão o
        que o teste de \omterm{def:g7:volume-mass-density:formula}{densidade} \emph{de fato} estabeleceu com
        honestidade sobre o lingote.
  \item Sugira mais um teste físico, de anos anteriores deste curso,
        capaz de distinguir o ouro do sósia dele sem danificar o
        lingote. (Os \omterm{def:g4:changes-of-state:melting-point}{pontos de fusão} dos dois diferem enormemente ---
        o ouro perto de \qty{1064}{\celsius}, o sósia bem acima de
        qualquer forno daqui --- mas ferro-velho nenhum \omterm{def:g2:ice-water-steam:meltfreeze}{derrete} um
        possível tesouro: encontre uma evidência mais gentil, talvez
        o veredito do \omterm{def:g1:magnets:magnet}{ímã} sobre núcleos de ferro, ou o \omterm{def:g3:sound-around-us:sound}{som} de uma
        barra golpeada, e defenda a sua escolha com honestidade.)
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- O ofício da detetive.}
\begin{enumerate}[resume]
  \item Escreva o credo de três linhas da detetive: o que a \omterm{def:g7:volume-mass-density:formula}{densidade}
        consegue provar, o que ela só consegue sugerir e o que ela
        nunca consegue fazer sozinha.
\end{enumerate}
\end{problem}
