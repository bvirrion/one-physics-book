\section*{Capítulo \ref{ch:g7:motion-average-speed} --- Gráficos do movimento e velocidade média}

\begin{solution}{exo:g7:motion-average-speed:1}
$v = d/t$; \omterm{def:g6:measurement-in-science:measuring}{unidades} \unit{km/h} e \unit{m/s}. Distância:
$d = v \times t$; tempo: $t = d/v$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:2}
Balsa: $45 \div 3 = \qty{15}{km/h}$. Lebre:
$100 \div 8 = \qty{12.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:3}
$5 \times 3.6 = \qty{18}{km/h}$; \ $72 \div 3.6 = \qty{20}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:4}
Reta: \omterm{def:g6:describing-motion:uniform}{movimento uniforme}; mais inclinada: mais rápido; horizontal:
parado. Uma curva que vai ficando menos inclinada aos poucos conta que
o movimento está freando.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:5}
Do minuto $35$ ao $60$ --- o terceiro trecho: é o menos inclinado
entre os trechos que sobem, o que se lê direto da inclinação mais
suave dele.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:6}
Uniforme --- uma única reta. Ela ganha \qty{2}{km} a cada meia hora:
$v = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:7}
Distância total sobre tempo total, paradas incluídas. Trilha
inteira: $12 \div 4 = \qty{3}{km/h}$. Só caminhando:
$12 \div 3 = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:8}
Em distâncias iguais, o trecho lento simplesmente dura mais --- uma
parte maior do relógio da viagem é vivida no ritmo lento, e por isso
o ritmo lento fala com uma parcela maior da voz da viagem.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:9}
Palpite do ponto médio: \qty{22.5}{km/h}. Honestamente: ida
$30 \div 15 = 2$ \omterm{ex:g2:measuring-time:clock}{horas}, volta $30 \div 30 = 1$ hora; total
\qty{60}{km} em $3$ \omterm{ex:g2:measuring-time:clock}{horas}: \qty{20}{km/h}. A metade lenta foi dona de
duas das três \omterm{ex:g2:measuring-time:clock}{horas}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:10}
Gráfico: reta até (\qty{30}{min}, \qty{20}{km}); horizontal até $45$
min; reta e mais inclinada até (\qty{60}{min}, \qty{35}{km}). Média:
\qty{35}{km} em uma hora --- \qty{35}{km/h}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:11}
$d = 340 \times 6 = \qty{2040}{m}$ --- cerca de \qty{2}{km}. A velha
regra: $6 \div 3 = 2$ quilômetros --- concordando, porque
\qty{340}{m/s} querem dizer quase exatamente um quilômetro a cada
três segundos.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:12}
Primeira metade: \qty{30}{km} a \qty{30}{km/h} custam exatamente $1$
hora. Uma média geral de \qty{60}{km/h} em \qty{60}{km} permite um
total de exatamente $1$ hora --- já gasta até o último segundo. Os
\qty{30}{km} restantes teriam de levar $0$ hora: \omterm{def:g7:motion-average-speed:formula}{velocidade} nenhuma,
por mais heroica, alcança a média; o orçamento acabou.
\end{solution}

\begin{solution}{pb:g7:motion-average-speed:1}
\textbf{1.} Uma pedalada firme ($30$ min), uma parada de $15$ minutos
(entrega), uma pedalada mais suave ($30$ min) e uma segunda parada de
$15$ minutos --- duas corridas, duas entregas.
\textbf{2.} \qty{9}{km} em meia hora: \qty{18}{km/h}.
\textbf{3.} $15 - 9 = \qty{6}{km}$ em meia hora: \qty{12}{km/h}.
\textbf{4.} \qty{15}{km} em $1.5$ hora: \qty{10}{km/h}.
\textbf{5.} $t = 12 \div 24 = 0.5$ hora: $30$ minutos.
\textbf{6.} $20$ minutos são um terço de hora:
$d = 18 \times \tfrac{1}{3} = \qty{6}{km}$.
\textbf{7.} $25$ minutos são $\tfrac{25}{60}$ de hora;
$v = 10 \div \tfrac{25}{60} = \qty{24}{km/h}$.
\textbf{8.} Distâncias: $12 + 6 + 10 = \qty{28}{km}$; tempos:
$30 + 20 + 25 = 75$ minutos $= 1.25$ hora; média rodando:
$28 \div 1.25 = \qty{22.4}{km/h}$.
\textbf{9.} $37 \div 3 \approx \qty{12.3}{km/h}$ --- bem abaixo de
$20$: sem bônus (as paradas dela são trabalho honesto, mas a regra as
conta).
\textbf{10.} Max: metade rápida $12 \div 30 = 0.4$ h; metade lenta
$12 \div 15 = 0.8$ h; total \qty{24}{km} em $1.2$ h: exatamente
\qty{20}{km/h} --- e não \emph{acima} de $20$: sem bônus.
\textbf{11.} O $22.5$ dele tirou a média dos dois \emph{números},
como se cada ritmo fosse dono de metade do relógio. Na verdade a
metade lenta foi dona de $0.8$ das $1.2$ \omterm{ex:g2:measuring-time:clock}{horas} dele --- dois terços
da voz da viagem --- e puxou a média verdadeira para $20$.
\textbf{12.} Por exemplo: pagar o bônus pela \omterm{def:g7:motion-average-speed:average}{velocidade média}
\emph{rodando} --- a distância total dividida só pelo tempo de
pedalada, com os trechos horizontais do gráfico excluídos. Ela mede o
ritmo que as rodas realmente mantêm, premiando pedalar rápido em vez
de pular entregas.
\textbf{13.} ``Um trecho reto é um ritmo constante; um trecho
horizontal é uma parada com o relógio correndo; um trecho mais
inclinado é um ritmo mais rápido --- a inclinação é a \omterm{def:g7:motion-average-speed:formula}{velocidade}. E
nunca tire a média de números de \omterm{def:g7:motion-average-speed:formula}{velocidade}: tire a média da viagem
--- distância total sobre tempo total --- ou as \omterm{ex:g2:measuring-time:clock}{horas} lentas farão da
planilha uma piada.''
\end{solution}
