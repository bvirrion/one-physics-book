\section*{Hoofdstuk \ref{ch:g7:motion-average-speed} --- Bewegingsgrafieken en gemiddelde snelheid}

\begin{solution}{exo:g7:motion-average-speed:1}
$v = d/t$; eenheden \unit{km/h} en \unit{m/s}. Afstand:
$d = v \times t$; tijd: $t = d/v$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:2}
Veerboot: $45 \div 3 = \qty{15}{km/h}$. Haas: $100 \div 8 =
\qty{12.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:3}
$5 \times 3.6 = \qty{18}{km/h}$; \ $72 \div 3.6 = \qty{20}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:4}
Rechte lijn: \omterm{def:g6:describing-motion:uniform}{eenparige beweging}; steiler: sneller; vlak:
stilstaand. Een geleidelijk afvlakkende bocht vertelt van vertragen.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:5}
Van minuut $35$ tot $60$ --- het derde stuk: het is het minst steile
van de stijgende stukken, rechtstreeks af te lezen aan zijn flauwere
helling.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:6}
Eenparig --- één rechte lijn. Er komt elk half uur \qty{2}{km} bij:
$v = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:7}
Totale afstand gedeeld door totale tijd, stops inbegrepen. Hele tocht:
$12 \div 4 = \qty{3}{km/h}$. Alleen lopend: $12 \div 3 =
\qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:8}
Over gelijke afstanden duurt het trage stuk eenvoudigweg langer --- een
groter deel van de klok van de reis wordt in het trage tempo geleefd,
dus spreekt dat trage tempo met een groter deel van de stem van de
reis.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:9}
Gok op het midden: \qty{22.5}{km/h}. Eerlijk: heen $30 \div 15 = 2$
uur, terug $30 \div 30 = 1$ uur; totaal \qty{60}{km} in $3$
uur: \qty{20}{km/h}. De trage helft bezat twee van de drie
\omterm{ex:g2:measuring-time:clock}{uren}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:10}
Grafiek: recht tot (\qty{30}{min}, \qty{20}{km}); vlak tot $45$ min;
recht en steiler tot (\qty{60}{min}, \qty{35}{km}). Gemiddelde:
\qty{35}{km} in één uur --- \qty{35}{km/h}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:11}
$d = 340 \times 6 = \qty{2040}{m}$ --- ongeveer \qty{2}{km}. De oude
regel: $6 \div 3 = 2$ kilometer --- in overeenstemming, omdat
\qty{340}{m/s} vrijwel een kilometer per drie \omterm{def:g2:measuring-time:second}{seconden} betekent.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:12}
Eerste helft: \qty{30}{km} met \qty{30}{km/h} kost precies $1$ uur. Een
totaalgemiddelde van \qty{60}{km/h} over \qty{60}{km} staat in totaal
precies $1$ uur toe --- al tot op de laatste \omterm{def:g2:measuring-time:second}{seconde} uitgegeven. De
resterende \qty{30}{km} zou $0$ uur mogen kosten: geen enkele
\omterm{def:g7:motion-average-speed:formula}{snelheid}, hoe heldhaftig ook, haalt het gemiddelde nog; het budget
is op.
\end{solution}

\begin{solution}{pb:g7:motion-average-speed:1}
\textbf{1.} Een stevige rit ($30$ min), een stop van $15$ minuten
(bezorging), een rustiger rit ($30$ min), en een tweede stop van $15$
minuten --- twee ritten, twee adressen.
\textbf{2.} \qty{9}{km} in een half uur: \qty{18}{km/h}.
\textbf{3.} $15 - 9 = \qty{6}{km}$ in een half uur: \qty{12}{km/h}.
\textbf{4.} \qty{15}{km} in $1.5$ uur: \qty{10}{km/h}.
\textbf{5.} $t = 12 \div 24 = 0.5$ uur: $30$ minuten.
\textbf{6.} $20$ minuten is een derde van een uur: $d = 18 \times
\tfrac{1}{3} = \qty{6}{km}$.
\textbf{7.} $25$ minuten is $\tfrac{25}{60}$ uur; $v = 10
\div \tfrac{25}{60} = \qty{24}{km/h}$.
\textbf{8.} Afstanden: $12 + 6 + 10 = \qty{28}{km}$; tijden:
$30 + 20 + 25 = 75$ minuten $= 1.25$ uur; rolgemiddelde:
$28 \div 1.25 = \qty{22.4}{km/h}$.
\textbf{9.} $37 \div 3 \approx \qty{12.3}{km/h}$ --- ver onder $20$:
geen bonus (haar stops zijn eerlijk werk, maar de regel telt ze mee).
\textbf{10.} Max: snelle helft $12 \div 30 = 0.4$ h; trage helft
$12 \div 15 = 0.8$ h; totaal \qty{24}{km} in $1.2$ h: precies
\qty{20}{km/h} --- niet \emph{boven} $20$: geen bonus.
\textbf{11.} Zijn $22.5$ middelde de twee \emph{getallen}, alsof elk
tempo de helft van de klok bezat. In werkelijkheid bezat de trage helft
$0.8$ van zijn $1.2$ uur --- twee derde van de stem van de reis ---
en trok het echte gemiddelde omlaag tot $20$.
\textbf{12.} Bijvoorbeeld: keer de bonus uit op de \emph{rollende}
\omterm{def:g7:motion-average-speed:average}{gemiddelde snelheid} --- totale afstand gedeeld door alleen de
rijtijd, met de vlakke stukken van de grafiek eruit. Die meet het
tempo dat de wielen werkelijk aanhouden, en beloont snel rijden in
plaats van overgeslagen bezorgingen.
\textbf{13.} ``Een recht stuk is een gelijkmatig tempo; een vlak stuk
is een stop terwijl de klok doorloopt; een steiler stuk is een sneller
tempo --- steilheid is \omterm{def:g7:motion-average-speed:formula}{snelheid}. En middel nooit
snelheidsgetallen: middel de reis --- totale afstand gedeeld door
totale tijd --- anders maken de trage \omterm{ex:g2:measuring-time:clock}{uren} het blad belachelijk.''
\end{solution}
