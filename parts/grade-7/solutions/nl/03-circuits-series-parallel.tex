\section*{Hoofdstuk \ref{ch:g7:circuits-series-parallel} --- Stroomkringen: lussen, serie, parallel}

\begin{solution}{exo:g7:circuits-series-parallel:1}
De stroming van geladen deeltjes die onder de duw van de
\omterm{def:g3:first-electric-circuit:battery}{batterij} door de kring marcheren. Hij bestaat alleen zolang minstens
één gesloten lus de twee polen verbindt; open de lus en de mars
stopt overal tegelijk.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:2}
Buiten de \omterm{def:g3:first-electric-circuit:battery}{batterij}, van $+$ rond de kring naar $-$. Op de lus met
één \omterm{def:g3:first-electric-circuit:bulb}{lampje} vertrekken de pijlen bij $+$, gaan ze door het
\omterm{def:g3:first-electric-circuit:bulb}{lampje}, en keren ze terug naar $-$.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:3}
De fout dat apparaten stroom verbruiken, zodat \omterm{def:g3:first-electric-circuit:bulb}{lampjes}
``stroomafwaarts'' zwakker zouden moeten gloeien. Gelijke felheid ---
die elke verwisseling overleeft --- laat zien dat door allebei
dezelfde hele stroom gaat: er wordt onderweg niets opgegeten.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:4}
Het is één en dezelfde stroom: één enkele lus draagt één enkele mars
door elk apparaat erop.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:5}
Hij splitst zich --- een deel van de mars neemt elke \omterm{def:g5:series-parallel-circuits:parallel}{tak} --- en bij
het tweede \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} voegen de delen zich compleet weer samen. Elke
\omterm{def:g5:series-parallel-circuits:parallel}{tak} geniet van de volle duw van de \omterm{def:g3:first-electric-circuit:battery}{batterij}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:6}
Alleen $=$ met één parallelbuur (volle felheid), dan met één
seriegenoot (zwak), dan met twee seriegenoten (nog zwakker).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:7}
\omterm{def:g5:series-parallel-circuits:parallel}{Parallel}: twee \omterm{def:g5:series-parallel-circuits:parallel}{takken} die elk een volle mars trekken verdubbelen de
totale levering van de \omterm{def:g3:first-electric-circuit:battery}{batterij}, terwijl het seriepaar één afgeknepen
mars door allebei stuurt. Volle felheid wordt bij de \omterm{def:g3:first-electric-circuit:battery}{batterij}
betaald.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:8}
De stoeltjes zijn de geladen marcheerders, die eindeloos rondgaan en
nooit verbruikt worden; de skiërs zijn de \omterm{def:g5:everyday-energy:energy}{energie}, ingeladen bij de
\omterm{def:g3:first-electric-circuit:battery}{batterij} en afgezet bij de lampen als \omterm{def:g1:light-and-shadows:source}{licht} en \omterm{def:g5:heat-and-insulation:heat}{warmte}. Wat opraakt
is de voorraad skiërs van het wintersportoord --- de opgeslagen
\omterm{def:g5:everyday-energy:energy}{energie} van de \omterm{def:g3:first-electric-circuit:battery}{batterij}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:9}
Elke lus door A passeert de \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} en dan B óf C --- dus de
seriegenoten van A zijn de \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} en het \emph{paar}
\omterm{def:g5:series-parallel-circuits:parallel}{takken} als geheel, maar noch B noch C afzonderlijk. De \omterm{def:g5:series-parallel-circuits:parallel}{tak} van
C doorknippen stuurt de hele mars door B: A blijft gloeien (een beetje
anders --- één weg in plaats van twee --- zoals de instrumenten van
volgend jaar zullen \omterm{def:g6:measurement-in-science:measuring}{meten}).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:10}
Eén hoofdschakelaar op de gemeenschappelijke weg vlak na de voeding,
en dan tien parallelle \omterm{def:g5:series-parallel-circuits:parallel}{takken}, elk met één \omterm{def:g3:first-electric-circuit:bulb}{lampje}. De
parallelwet gunt elk \omterm{def:g3:first-electric-circuit:bulb}{lampje} volle, onafhankelijke felheid; de
serieplaatsing van de hoofdschakelaar --- op de weg die elke lus deelt
--- laat één \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} alle tien commanderen.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:11}
De splitsing bij het \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} is geen gelijke verdeling: de eigen
weg van een \omterm{def:g5:series-parallel-circuits:parallel}{tak} kan de mars tegenwerken, en de lange dunne draad
knijpt het aandeel van zijn \omterm{def:g5:series-parallel-circuits:parallel}{tak} af --- minder marcheerders per
\omterm{def:g2:measuring-time:second}{seconde}, zwakkere gloed. Wegen zelf vechten tegen de stroom: het idee
achter de weerstand van volgend jaar.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:12}
X, alleen op zijn \omterm{def:g5:series-parallel-circuits:parallel}{tak}, krijgt de volle duw: het felst. Y en Z delen
de duw van hun \omterm{def:g5:series-parallel-circuits:parallel}{tak} \omterm{def:g4:series-circuits:series}{in serie}: even zwak, allebei onder X. Brandt
Z door, dan is hun lus verbroken: Y gaat mee uit --- X schijnt
onverschillig door. Brandt X door, dan sterft alleen de \omterm{def:g5:series-parallel-circuits:parallel}{tak} van X:
Y en Z gaan door, zwak als voorheen.
\end{solution}

\begin{solution}{pb:g7:circuits-series-parallel:1}
\textbf{1.} Vier parallelle \omterm{def:g5:series-parallel-circuits:parallel}{takken}, elk met één taak:
\omterm{def:g5:series-parallel-circuits:parallel}{parallel} werkt elke \omterm{def:g5:series-parallel-circuits:parallel}{tak} op volle \omterm{def:g2:pushes-and-pulls:force}{kracht}, wat de andere ook doen.
Elke koppeling \omterm{def:g4:series-circuits:series}{in serie} zou de \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} van een leeslamp of een
doorgebrand traplampje de grote lamp laten verzwakken of doven ---
onvergeeflijk.
\textbf{2.} De hoofdschakelaar op de gemeenschappelijke weg vlak na de
bank; elk van de vier takschakelaars \omterm{def:g4:series-circuits:series}{in serie} met zijn eigen
taak, binnen zijn \omterm{def:g5:series-parallel-circuits:parallel}{tak}.
\textbf{3.} Ja --- de parallelwet: een onderbreking in de ene
\omterm{def:g5:series-parallel-circuits:parallel}{tak} legt alleen het aandeel van die \omterm{def:g5:series-parallel-circuits:parallel}{tak} stil; de andere
marcheren door.
\textbf{4.} Bank, hoofdschakelaar, dan een \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} naar vier
\omterm{def:g5:series-parallel-circuits:parallel}{takken} --- (grote lamp $+$ \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}), (misthoorn $+$
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}), (leeslicht $+$ \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}), (traplicht $+$
\omterm{ex:g3:first-electric-circuit:switch}{schakelaar}) --- die weer samenkomen en terugkeren naar de bank.
\textbf{5.} Ja: elke \omterm{def:g5:series-parallel-circuits:parallel}{tak} overspant de twee kanten van de bank, dus
krijgt elk de volle duw, hoeveel buren er ook meedrinken.
\textbf{6.} Nee: de stroom door de lamp verlaat haar heel --- de
stoeltjes gaan onopgegeten rond. Wat de lamp neemt is \omterm{def:g5:everyday-energy:energy}{energie} uit de
voorraad van de bank, bezorgd door de langstrekkende mars.
\textbf{7.} Vier volle marsen tegelijk: de \omterm{def:g5:everyday-energy:energy}{energie} van de bank loopt
door vier \omterm{def:g5:series-parallel-circuits:parallel}{taken} snel weg. De rekening wordt in de batterijkamer
betaald --- in opgeslagen \omterm{def:g5:everyday-energy:energy}{energie}, nooit in verloren stroom.
\textbf{8.} Uit de $+$ van de bank, door de hoofdschakelaar, naar
het \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt}; de \omterm{def:g5:series-parallel-circuits:parallel}{tak} van het leeslicht in, door zijn
gesloten \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} en de lamp; langs het \omterm{def:g5:series-parallel-circuits:parallel}{knooppunt} waar de andere
drie (donkere) \omterm{def:g5:series-parallel-circuits:parallel}{takken} weer samenkomen; terug naar de $-$ van de
bank. Hij mijdt de \omterm{def:g5:series-parallel-circuits:parallel}{takken} van de grote lamp, de misthoorn en de trap
volledig.
\textbf{9.} Niets anders --- alleen de \omterm{def:g5:series-parallel-circuits:parallel}{tak} van de trap sterft met
haar \omterm{def:g3:first-electric-circuit:bulb}{lampje}: de onafhankelijkheid uit de parallelwet.
\textbf{10.} Een \omterm{def:g6:circuits-and-safety:short}{kortsluiting} stroomopwaarts van elke \omterm{def:g5:series-parallel-circuits:parallel}{tak}. De
mars laat elke werkende weg in de steek voor de moeiteloze blote
sluiproute: de lampen en de hoorn verhongeren terwijl de sluipdraad
een wilde, verhittende stormloop draagt --- elke taak valt tegelijk
uit tot de draad wordt weggehaald.
\textbf{11.} Geef de grote lamp een eigen onafhankelijke lus op de
tweede bank: bank twee, een eigen \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}, de grote lamp --- zonder
ook maar één draad met het algemene paneel te delen. Een sluiproute
over bank één legt dan alles het zwijgen op \emph{behalve} het
\omterm{def:g1:light-and-shadows:source}{licht} voor de schepen.
\textbf{12.} Bijvoorbeeld: ``De parallelwet hield vannacht elke taak
soeverein. De nacht weerlegde opnieuw het geloof dat lampen stroom
verslinden --- de mars kwam heel terug terwijl de banken leegliepen.
En één blote draad, die de bank kortsloot, kostte ons het slechtste
uur: elke weg leeg terwijl de sluiproute gloeide.''
\end{solution}
