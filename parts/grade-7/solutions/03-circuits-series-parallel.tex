\section*{Chapter \ref{ch:g7:circuits-series-parallel} --- Circuits: Loops, Series, Parallel}

\begin{solution}{exo:g7:circuits-series-parallel:1}
The flow of charged particles marching through the circuit under
the \omterm{def:g3:first-electric-circuit:battery}{battery}'s push. It exists only while at least one closed loop
joins the two \omterm{def:g3:first-electric-circuit:battery}{terminals}; open the loop and the march stops
everywhere at once.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:2}
Outside the \omterm{def:g3:first-electric-circuit:battery}{battery}, from $+$ around the circuit to $-$. On the
\omterm{def:g3:first-electric-circuit:bulb}{one-bulb} loop the arrows leave $+$, pass through the \omterm{def:g3:first-electric-circuit:bulb}{bulb}, and
return to $-$.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:3}
The error that devices consume current, so ``downstream'' \omterm{def:g3:first-electric-circuit:bulb}{bulbs}
should glow weaker. Equal brightness --- surviving every swap ---
shows the same whole current passes through both: nothing is
eaten en route.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:4}
They are one and the same current: a single loop carries a single
march through every device on it.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:5}
It divides --- part of the march takes each \omterm{def:g5:series-parallel-circuits:parallel}{branch} --- and at the
second \omterm{def:g5:series-parallel-circuits:parallel}{junction} the parts rejoin, complete. Every \omterm{def:g5:series-parallel-circuits:parallel}{branch} enjoys
the \omterm{def:g3:first-electric-circuit:battery}{battery}'s full push.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:6}
Alone $=$ with one parallel neighbor (full brightness), then one
series companion (dim), then two series companions (dimmer
still).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:7}
Parallel: two \omterm{def:g5:series-parallel-circuits:parallel}{branches} each drawing a full march doubles the
\omterm{def:g3:first-electric-circuit:battery}{battery}'s total delivery, while the series pair sends one
throttled march through both. Full brightness is paid for at the
\omterm{def:g3:first-electric-circuit:battery}{battery}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:8}
The chairs are the charged marchers, circulating endlessly and
never consumed; the skiers are the \omterm{def:g5:everyday-energy:energy}{energy}, loaded at the \omterm{def:g3:first-electric-circuit:battery}{battery}
and dropped off at the lamps as \omterm{def:g1:light-and-shadows:source}{light} and warmth. What runs out
is the resort's stock of skiers --- the \omterm{def:g3:first-electric-circuit:battery}{battery}'s stored \omterm{def:g5:everyday-energy:energy}{energy}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:9}
Every loop through A passes the \omterm{ex:g3:first-electric-circuit:switch}{switch} and then either B or C ---
so A's series companions are the \omterm{ex:g3:first-electric-circuit:switch}{switch} and the \emph{pair} of
\omterm{def:g5:series-parallel-circuits:parallel}{branches} as a whole, but neither B nor C alone. Cutting C's
\omterm{def:g5:series-parallel-circuits:parallel}{branch} sends the whole march through B: A keeps glowing (a little
differently --- one road instead of two --- as next year's
instruments will \omterm{def:g6:measurement-in-science:measuring}{measure}).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:10}
One master switch on the common road just after the supply, then
ten parallel \omterm{def:g5:series-parallel-circuits:parallel}{branches}, one \omterm{def:g3:first-electric-circuit:bulb}{bulb} each. The parallel law grants
full, independent brightness to every \omterm{def:g3:first-electric-circuit:bulb}{bulb}; the series placement
of the master --- on the road every loop shares --- lets one
\omterm{ex:g3:first-electric-circuit:switch}{switch} command all ten.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:11}
The \omterm{def:g5:series-parallel-circuits:parallel}{junction}'s division is not an equal split: a \omterm{def:g5:series-parallel-circuits:parallel}{branch}'s own
road can resist the march, and the long thin wire throttles its
\omterm{def:g5:series-parallel-circuits:parallel}{branch}'s share --- fewer marchers per second, weaker glow. Roads
themselves fight the current: the idea behind next year's
resistance.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:12}
X, alone on its \omterm{def:g5:series-parallel-circuits:parallel}{branch}, gets the full push: brightest. Y and Z
share their \omterm{def:g5:series-parallel-circuits:parallel}{branch}'s push \omterm{def:g4:series-circuits:series}{in series}: equally dim, both below X.
If Z burns out, its loop is broken: Y darkens with it --- X
shines on, indifferent. If X burns out, only X's \omterm{def:g5:series-parallel-circuits:parallel}{branch} dies: Y
and Z continue, dim as before.
\end{solution}

\begin{solution}{pb:g7:circuits-series-parallel:1}
\textbf{1.} Four parallel \omterm{def:g5:series-parallel-circuits:parallel}{branches}, one duty each: \omterm{def:g5:series-parallel-circuits:parallel}{in parallel},
each \omterm{def:g5:series-parallel-circuits:parallel}{branch} works at full strength whatever the others do. Any
series pairing would let a reading lamp's \omterm{ex:g3:first-electric-circuit:switch}{switch} or a burnt
staircase \omterm{def:g3:first-electric-circuit:bulb}{bulb} dim or darken the great lamp --- unforgivable.
\textbf{2.} The master on the common road just after the bank;
each of the four \omterm{def:g5:series-parallel-circuits:parallel}{branch} \omterm{ex:g3:first-electric-circuit:switch}{switches} \omterm{def:g4:series-circuits:series}{in series} with its own duty,
inside its \omterm{def:g5:series-parallel-circuits:parallel}{branch}.
\textbf{3.} Yes --- the parallel law: a gap in one \omterm{def:g5:series-parallel-circuits:parallel}{branch} stops
only that \omterm{def:g5:series-parallel-circuits:parallel}{branch}'s share; the others march on.
\textbf{4.} Bank, master \omterm{ex:g3:first-electric-circuit:switch}{switch}, then a \omterm{def:g5:series-parallel-circuits:parallel}{junction} into four
\omterm{def:g5:series-parallel-circuits:parallel}{branches} --- (great lamp $+$ \omterm{ex:g3:first-electric-circuit:switch}{switch}), (foghorn $+$ \omterm{ex:g3:first-electric-circuit:switch}{switch}),
(reading \omterm{def:g1:light-and-shadows:source}{light} $+$ \omterm{ex:g3:first-electric-circuit:switch}{switch}), (staircase \omterm{def:g1:light-and-shadows:source}{light} $+$ \omterm{ex:g3:first-electric-circuit:switch}{switch}) ---
rejoining and returning to the bank.
\textbf{5.} Yes: every \omterm{def:g5:series-parallel-circuits:parallel}{branch} spans the bank's two sides, so
each receives the full push regardless of how many neighbors
drink alongside.
\textbf{6.} No: the current through the lamp leaves it whole ---
the chairs circulate uneaten. What the lamp takes is \omterm{def:g5:everyday-energy:energy}{energy} from
the bank's store, delivered by the passing march.
\textbf{7.} Four full marches at once: the bank's \omterm{def:g5:everyday-energy:energy}{energy} drains
four duties fast. The bill is paid in the \omterm{def:g3:first-electric-circuit:battery}{battery} room --- in
stored \omterm{def:g5:everyday-energy:energy}{energy}, never in lost current.
\textbf{8.} Out of the bank's $+$, through the master \omterm{ex:g3:first-electric-circuit:switch}{switch},
to the \omterm{def:g5:series-parallel-circuits:parallel}{junction}; into the reading \omterm{def:g1:light-and-shadows:source}{light}'s \omterm{def:g5:series-parallel-circuits:parallel}{branch}, through its
closed \omterm{ex:g3:first-electric-circuit:switch}{switch} and the lamp; past the \omterm{def:g5:series-parallel-circuits:parallel}{junction} where the other
three (dark) \omterm{def:g5:series-parallel-circuits:parallel}{branches} rejoin; back to the bank's $-$. It avoids
the great lamp, foghorn and staircase \omterm{def:g5:series-parallel-circuits:parallel}{branches} entirely.
\textbf{9.} Nothing else --- only the staircase \omterm{def:g5:series-parallel-circuits:parallel}{branch} dies with
its \omterm{def:g3:first-electric-circuit:bulb}{bulb}: the parallel law's independence.
\textbf{10.} A \omterm{def:g6:circuits-and-safety:short}{short circuit} upstream of every \omterm{def:g5:series-parallel-circuits:parallel}{branch}. The march
abandons every working road for the effortless bare shortcut:
the lamps and horn starve while the shortcut wire carries a
wild, heating rush --- every duty fails at once until the wire
is removed.
\textbf{11.} Give the great lamp its own independent loop on the
second bank: bank two, its own \omterm{ex:g3:first-electric-circuit:switch}{switch}, the great lamp --- sharing
no wire with the general panel. A shortcut across bank one then
silences everything \emph{except} the ships' \omterm{def:g1:light-and-shadows:source}{light}.
\textbf{12.} For example: ``The parallel law kept each duty
sovereign tonight. The night refuted, once more, the belief that
lamps devour current --- the march returned whole even as the
banks drained. And one bare wire, shorting the bank, cost us our
worst hour: every road empty while the shortcut burned.''
\end{solution}
