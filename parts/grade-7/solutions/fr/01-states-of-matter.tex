\section*{Chapitre \ref{ch:g7:states-of-matter} --- États de la matière et changements d'état}

\begin{solution}{exo:g7:states-of-matter:1}
\omterm{def:g3:solids-liquids-gases:solid}{Solide}~: des \omterm{def:g7:states-of-matter:molecule}{molécules} serrées en un empilement ordonné, tremblant
sur place. \omterm{def:g3:solids-liquids-gases:liquid}{Liquide}~: des \omterm{def:g7:states-of-matter:molecule}{molécules} en contact mais en désordre, qui
glissent les unes contre les autres. \omterm{def:g3:solids-liquids-gases:gas}{Gaz}~: des \omterm{def:g7:states-of-matter:molecule}{molécules} très
éloignées, volant vite dans toutes les directions.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:2}
Dans le \omterm{def:g3:solids-liquids-gases:solid}{solide}, chaque \omterm{def:g7:states-of-matter:molecule}{molécule} tient son poste dans l'empilement~:
l'ensemble garde donc sa forme. Dans le \omterm{def:g3:solids-liquids-gases:liquid}{liquide}, les \omterm{def:g7:states-of-matter:molecule}{molécules}
glissent librement les unes contre les autres tout en restant en
contact~: la foule s'affaisse pour épouser le récipient, jusqu'à ce
que rien ne puisse glisser plus bas --- d'où la surface horizontale.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:3}
L'\omterm{def:g2:air-around-us:air}{air} est surtout de l'espace vide entre des \omterm{def:g7:states-of-matter:molecule}{molécules} qui volent~:
pousser rapproche les voltigeuses (et leur martèlement résiste ---
d'où l'élasticité). Les \omterm{def:g7:states-of-matter:molecule}{molécules} de l'eau se touchent déjà~; il n'y
a aucun vide à chasser, et le piston rencontre donc un mur.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:4}
Que les \omterm{def:g7:states-of-matter:molecule}{molécules} d'un \omterm{def:g3:solids-liquids-gases:gas}{gaz} volent très éloignées les unes des autres,
avec un grand vide entre elles --- des pièces entières peuvent être
tassées ensemble. L'eau, dont les \omterm{def:g7:states-of-matter:molecule}{molécules} se touchent déjà, est
incompressible~: aucune pompe ne la tasse davantage.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:5}
La \omterm{def:g5:heat-and-insulation:heat}{chaleur} fait trembler de plus en plus fort les \omterm{def:g7:states-of-matter:molecule}{molécules}
empilées~; à la \omterm{def:g4:changes-of-state:melting-point}{température de fusion}, le tremblement l'emporte sur
l'emprise de l'empilement, les rangs se rompent, et les \omterm{def:g7:states-of-matter:molecule}{molécules} se
mettent à glisser~: le \omterm{def:g3:solids-liquids-gases:solid}{solide} est devenu \omterm{def:g3:solids-liquids-gases:liquid}{liquide}.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:6}
Un changement d'état ne fait que réarranger les mêmes \omterm{def:g7:states-of-matter:molecule}{molécules} ---
aucune n'est créée ni détruite, et la \omterm{def:g3:measuring-mass:mass}{masse} ne peut donc pas changer.
Perdre de la \omterm{def:g3:measuring-mass:mass}{masse} exigerait que des \omterm{def:g7:states-of-matter:molecule}{molécules} disparaissent (ou
s'échappent du récipient --- la fausse alerte de la bouteille non
bouchée).
\end{solution}

\begin{solution}{exo:g7:states-of-matter:7}
La vivacité de l'agitation moléculaire. Quand le chaud touche le
froid, les \omterm{def:g7:states-of-matter:molecule}{molécules} vives bousculent les engourdies et leur
transmettent de la vivacité, choc après choc --- les vives se
calment, les engourdies s'animent --- jusqu'à ce que les deux foules
partagent une même allure~: des \omterm{def:g3:temperature-thermometers:temperature}{températures} égales.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:8}
Les \omterm{def:g7:states-of-matter:molecule}{molécules} de parfum volent, heurtent des \omterm{def:g7:states-of-matter:molecule}{molécules} d'\omterm{def:g2:air-around-us:air}{air},
rebondissent et vagabondent --- zigzag après zigzag à travers la
pièce. Un courant d'\omterm{def:g2:air-around-us:air}{air} en transporte des foules entières d'un
bloc~: chevaucher le \omterm{def:g2:air-around-us:wind}{vent} vaut mieux que tituber dans la foule.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:9}
Seules les \omterm{def:g7:states-of-matter:molecule}{molécules} les \emph{plus rapides} peuvent s'arracher à
l'emprise du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~: l'évaporation est donc un impôt sur la
vivacité~; les évadées emportent plus que leur part, et la foule
laissée derrière est, en moyenne, plus lente --- c'est-à-dire plus
froide. La sueur et le doigt mouillé refroidissent en exportant leurs
plus vives.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:10}
Au soleil, les \omterm{def:g7:states-of-matter:molecule}{molécules} enfermées volent plus vite et martèlent la
peau du ballon plus fort et plus souvent~: la grêle gonfle le ballon
jusqu'à ce que la peau tendue lui fasse contrepoids. Au froid, les
mêmes \omterm{def:g7:states-of-matter:molecule}{molécules} ralentissent, le martèlement faiblit, et la poussée
de l'\omterm{def:g2:air-around-us:air}{air} extérieur écrase le ballon.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:11}
En gelant, l'eau se verrouille en un empilement inhabituellement
aéré et ouvert --- avec plus d'espace vide que n'en avait la foule
glissante --- si bien que la glace occupe plus de place pour les
mêmes \omterm{def:g7:states-of-matter:molecule}{molécules}~: \omterm{def:g6:density-floating:density}{masse volumique} inférieure à celle du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}, et
elle \omterm{def:g1:floating-and-sinking:words}{flotte}. Si l'eau était ordinaire, la glace s'empilerait serré,
coulerait à mesure de sa formation, et les étangs gèleraient du fond
vers le haut --- sans laisser d'eau abritée pour les poissons.
\end{solution}

\begin{solution}{exo:g7:states-of-matter:12}
Rien n'a agité l'eau, et pourtant l'encre a voyagé~: les \omterm{def:g7:states-of-matter:molecule}{molécules} du
\omterm{def:g3:solids-liquids-gases:liquid}{liquide} lui-même sont en mouvement incessant et bousculent les
\omterm{def:g7:states-of-matter:molecule}{molécules} d'encre, pas au hasard après pas au hasard, à travers la
foule --- les deux \omterm{def:g3:solids-liquids-gases:liquid}{liquides} sont pleins de mouvement à chaque
instant, même quand le verre paraît parfaitement immobile. La \omterm{def:g5:heat-and-insulation:heat}{chaleur}
accélère chaque bousculade, et la diffusion se presse donc. L'encre
patiente a ainsi rendu visible la danse invisible --- une preuve qui
a convaincu les sceptiques des \omterm{def:g7:states-of-matter:molecule}{molécules} elles-mêmes.
\end{solution}

\begin{solution}{pb:g7:states-of-matter:1}
\textbf{1.} En rangs bien alignés, épaule contre épaule, chacun
gigotant sur place --- et aucun n'a le droit d'échanger sa place ni
de quitter son poste.
\textbf{2.} Les rangs se dissolvent~: les élèves restent épaule
contre épaule (en contact~!) mais se faufilent et glissent désormais
les uns contre les autres. Ils doivent toujours garder le contact
avec la foule --- personne ne peut courir librement.
\textbf{3.} Les élèves se dispersent, courant vite et tout droit
jusqu'à se heurter ou heurter un mur, rebondissant sans fin --- et
comme rien ne les retient ensemble, seuls les murs les arrêtent~: la
foule s'empare de tous les recoins offerts par le gymnase.
\textbf{4.} Oui --- les mêmes trente dans chaque tableau~: la \omterm{def:g3:measuring-mass:mass}{masse}
survit aux changements d'état, mise en scène.
\textbf{5.} Le tableau du \omterm{def:g3:solids-liquids-gases:gas}{gaz} le permet --- les coureurs se
resserrent dans la moitié du gymnase (et martèlent plus fort les
rabatteurs). Le groupe du \omterm{def:g3:solids-liquids-gases:liquid}{liquide} ne peut pas rétrécir~: les élèves
se touchent déjà. Mise en scène~: les \omterm{def:g3:solids-liquids-gases:gas}{gaz} se compriment, les \omterm{def:g3:solids-liquids-gases:liquid}{liquides}
non.
\textbf{6.} Gigoter de plus en plus fort à leur poste~; le tableau
devient celui du \omterm{def:g3:solids-liquids-gases:liquid}{liquide} à l'instant où le gigotement fait éclater
les rangs et où les élèves se mettent à glisser les uns contre les
autres.
\textbf{7.} L'\omterm{def:g5:everyday-energy:energy}{énergie} du bouton est dépensée à briser l'emprise de
l'empilement --- rang après rang --- et non à accélérer le
mouvement~: le \omterm{def:g3:temperature-thermometers:temperature}{critique-thermomètre} doit rapporter une vivacité
moyenne \emph{inchangée} jusqu'à ce que le dernier rang se rompe. Le
palier, mis en scène.
\textbf{8.} Les élèves les \emph{plus rapides} --- eux seuls peuvent
s'arracher. La vivacité moyenne du groupe restant baisse~: le \omterm{def:g3:solids-liquids-gases:liquid}{liquide}
se refroidit lui-même en s'évaporant, exactement la fraîcheur de la
sueur et des doigts mouillés.
\textbf{9.} Les acteurs heurtent sans cesse la corde et la poussent
vers l'extérieur~; des acteurs plus rapides poussent plus fort et
plus souvent, et le cercle gonfle --- le ballon au soleil.
\textbf{10.} «~Empilez-vous en ordre \emph{ouvert} --- à bout de
bras, bien aéré~»~: le \omterm{def:g3:solids-liquids-gases:solid}{solide} de l'eau prend plus de place que son
\omterm{def:g3:solids-liquids-gases:liquid}{liquide}, et le radeau du tableau de la glace flotterait donc sur la
foule du tableau du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}.
\textbf{11.} La chemise bleue est bousculée --- poussée pas au
hasard après pas au hasard à travers la foule glissante, jusqu'à
avoir vagabondé partout~: c'est la diffusion de l'encre. Monte le
bouton et chaque bousculade s'accélère, si bien que le vagabondage se
presse.
\textbf{12.} Par exemple~: «~La pièce de ce soir met en scène une
seule idée~: toute la matière est faite de \omterm{def:g7:states-of-matter:molecule}{molécules} en mouvement.
Cette idée accordée, deux vieilles lois viennent gratuitement --- la
\omterm{def:g3:measuring-mass:mass}{masse} qui survit à tout changement d'état, et le palier de \omterm{def:g6:water-states:changes}{fusion}. Ce
que nous ne pouvons pas mettre en scène, c'est l'échelle~: les vrais
acteurs sont plus petits que tout regard et plus nombreux que tous
les publics du monde.~»
\end{solution}
