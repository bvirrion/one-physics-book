\section*{Chapitre \ref{ch:g7:volume-mass-density} --- Volume, masse, masse volumique}

\begin{solution}{exo:g7:volume-mass-density:1}
$\rho = m/V$~: $\rho$ la \omterm{def:g7:volume-mass-density:formula}{masse volumique}, $m$ la \omterm{def:g3:measuring-mass:mass}{masse} de
l'échantillon, $V$ son \omterm{def:g6:mass-and-volume:volume}{volume}. Avec des \omterm{def:g3:measuring-mass:units}{grammes} et des \omterm{def:g6:mass-and-volume:volume}{centimètres
cubes}, $\rho$ sort en \unit{g/cm^3}.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:2}
$\rho = 270 \div 100 = \qty{2.7}{g/cm^3}$~: de l'aluminium.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:3}
$m = 9.0 \times 40 = \qty{360}{g}$ de cuivre.
$V = 1930 \div 19.3 = \qty{100}{cm^3}$ d'or --- le nombre même de
l'affaire de la couronne.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:4}
La tare retire la \omterm{def:g3:measuring-mass:mass}{masse} propre de l'éprouvette pour que la \omterm{def:g3:levers-and-scales:scale}{balance}
annonce le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} seul --- la règle du zéro honnête. La formule est
ensuite nourrie de la \omterm{def:g3:measuring-mass:mass}{masse} $m$ donnée par la \omterm{def:g3:levers-and-scales:scale}{balance} et du \omterm{def:g6:mass-and-volume:volume}{volume}
$V$ lu au \omterm{met:g6:mass-and-volume:cylinder}{ménisque}.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:5}
$\rho = 70 \div 50 = \qty{1.4}{g/cm^3}$ --- du genre miel. La glace,
à $0.9$, est bien moins dense~: elle y \omterm{def:g1:floating-and-sinking:words}{flotte} haut.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:6}
Les deux \omterm{def:g6:measurement-in-science:measuring}{unités} changent d'échelle ensemble~: un \omterm{def:g6:mass-and-volume:volume}{litre} vaut $1000$
\unit{cm^3} et un \omterm{def:g3:measuring-mass:units}{kilogramme} vaut $1000$ \omterm{def:g3:measuring-mass:units}{grammes}, si bien que les
deux milliers se simplifient --- un seul nombre sert dans les deux
cas. Mélanger des \omterm{def:g3:measuring-mass:units}{grammes} et des \omterm{def:g6:mass-and-volume:volume}{litres} saute cette simplification et
livre un nombre mille fois faux.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:7}
$\rho = 58 \div 5.0 = \qty{11.6}{g/cm^3}$ --- proche du $11.3$ du
plomb, très loin du $19.3$ de l'or. Verdict~: ce n'est pas de l'or
pur~; sans doute un imposteur à cœur de plomb en habit doré.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:8}
$100 \times 100 \times 100 = 1000000$ \unit{cm^3} --- un million. À
\qty{1}{g} chacun, cela fait \qty{1000000}{g} $=$ \qty{1000}{kg}~:
une tonne entière par \omterm{def:g2:measuring-length:units}{mètre} cube d'eau --- et voilà pourquoi un grand
lit à eau est un meuble de rez-de-chaussée.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:9}
$V = 260 - 200 = \qty{60}{cm^3}$~;
$\rho = 420 \div 60 = \qty{7.0}{g/cm^3}$ --- entre l'aluminium
($2.7$) et le fer ($7.9$), tout près sous le fer. L'entre-deux
murmure~: un mélange ou un imposteur plaqué, certainement pas le
$10.5$ de l'argent.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:10}
\omterm{def:g3:measuring-mass:mass}{Masse} du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~: $1010 - 380 = \qty{630}{g}$~; \omterm{def:g6:mass-and-volume:volume}{volume}~:
$\qty{75}{cL} = \qty{750}{cm^3}$~; donc
$\rho = 630 \div 750 = \qty{0.84}{g/cm^3}$ --- une huile.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:11}
\omterm{def:g7:volume-mass-density:formula}{Masse volumique} du colis~: $1200 \div 4000 = \qty{0.3}{g/cm^3}$ ---
bien au-dessous du $1.0$ de l'eau~: la bouée \omterm{def:g1:floating-and-sinking:words}{flotte} haut. Le $2.7$ de
l'aluminium ne décrit que la peau métallique~; le colis est surtout
de l'\omterm{def:g2:air-around-us:air}{air} enfermé, et c'est le colis --- \omterm{def:g3:measuring-mass:mass}{masse} totale sur \omterm{def:g6:mass-and-volume:volume}{volume}
total --- qui affronte la règle de la flottaison.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:12}
Protocole~: peser la chaîne ($m$)~; \omterm{def:g6:measurement-in-science:measuring}{mesurer} son \omterm{def:g6:mass-and-volume:volume}{volume} par la méthode
de la montée d'eau, entièrement immergée, sans bulles ($V$)~;
calculer $\rho = m/V$~; la pureté attend environ \qty{9.0}{g/cm^3}.
Doutes honnêtes~: des maillons creux emprisonnent de l'\omterm{def:g2:air-around-us:air}{air} et
gonflent $V$, en simulant une \omterm{def:g7:volume-mass-density:formula}{masse volumique} faible~; et un mélange
rusé (ou une âme plaquée) peut tomber près de $9.0$ sans contenir le
moindre cuivre pur. Test qui resserre le dossier~: recommencer la
mesure de \omterm{def:g6:mass-and-volume:volume}{volume} après avoir noyé les maillons (en chassant les
bulles), et ajouter une signature indépendante --- par exemple
l'\omterm{def:g1:magnets:magnet}{aimant} (une âme d'acier se trahit instantanément) ou, en atelier,
un comportement de \omterm{def:g6:water-states:changes}{fusion} mesuré.
\end{solution}

\begin{solution}{pb:g7:volume-mass-density:1}
\textbf{1.} $5400 \div 2000 = \qty{2.7}{g/cm^3}$~: de l'aluminium.
\textbf{2.} $1800 \div 200 = \qty{9.0}{g/cm^3}$~: du cuivre.
\textbf{3.} $4520 \div 400 = \qty{11.3}{g/cm^3}$~: du plomb ---
dense, mou et toxique si on le manipule sans précaution~: des gants.
\textbf{4.} $8100 \div 1500 = \qty{5.4}{g/cm^3}$ --- entre
l'aluminium ($2.7$) et le fer ($7.9$)~: un sac mêlé des deux, à peu
près moitié-moitié en \omterm{def:g6:mass-and-volume:volume}{volume}.
\textbf{5.} $V = 540 \div 2.7 = \qty{200}{cm^3}$.
\textbf{6.} Un cube~: $m = 9.0 \times 25 = \qty{225}{g}$. Les
\qty{1800}{g} de la bobine donnent $1800 \div 225 = 8$ cubes pleins.
\textbf{7.} $\qty{20}{kg} = \qty{20000}{g}$~;
$20000 \div 225 = 88.9$~: $88$ cubes.
\textbf{8.} $m = 7.9 \times 3000 = \qty{23700}{g} \approx
\qty{23.7}{kg}$.
\textbf{9.} $3860 \div 200 = \qty{19.3}{g/cm^3}$ --- une
correspondance parfaite avec l'or.
\textbf{10.} Le tungstène --- \omterm{def:g7:volume-mass-density:formula}{masse volumique} $19.3$, la même
signature à la décimale près. Le test de \omterm{def:g7:volume-mass-density:formula}{masse volumique} identifie
des candidats~; il ne peut pas distinguer des substances qui, par
hasard, partagent une signature.
\textbf{11.} Aucun instrument n'a échoué~: la \omterm{def:g3:measuring-mass:mass}{masse} et le \omterm{def:g6:mass-and-volume:volume}{volume} sont
justes, et la division aussi. Le test a honnêtement établi que le
lingot est \emph{soit} de l'or, \emph{soit} quelque chose ayant
exactement la \omterm{def:g7:volume-mass-density:formula}{masse volumique} de l'or --- il a réduit les suspects à
deux.
\textbf{12.} Choix défendables~: le \omterm{def:g3:sound-around-us:sound}{son} et le toucher d'un lingot
frappé (le tungstène est bien plus dur --- une lime ou un test de
dureté sur un coin caché les sépare vite), ou une mesure reconnue de
la façon dont le lingot conduit la \omterm{def:g5:heat-and-insulation:heat}{chaleur} ou le courant (l'or est
parmi les meilleurs \omterm{def:g4:conductors-insulators:conductor}{conducteurs}, le tungstène bien loin derrière) ---
plus doux que n'importe quel four, et décisif conjointement avec la
\omterm{def:g7:volume-mass-density:formula}{masse volumique}.
\textbf{13.} Par exemple~: «~La \omterm{def:g7:volume-mass-density:formula}{masse volumique} peut prouver qu'une
substance n'est \emph{pas} ce qu'elle prétend. Elle ne peut que
suggérer ce qu'elle est --- les signatures réduisent les suspects. Et
seule, elle ne peut jamais condamner~: une identité réclame deux
témoins indépendants.~»
\end{solution}
