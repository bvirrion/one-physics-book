\section*{Chapitre \ref{ch:g7:circuits-series-parallel} --- Circuits~: boucles, série, dérivation}

\begin{solution}{exo:g7:circuits-series-parallel:1}
L'écoulement de particules chargées qui marchent à travers le circuit
sous la poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile}. Il n'existe que tant qu'au moins une
boucle fermée relie les deux \omterm{def:g3:first-electric-circuit:battery}{bornes}~; ouvre la boucle et la marche
s'arrête partout à la fois.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:2}
À l'extérieur de la \omterm{def:g3:first-electric-circuit:battery}{pile}, du $+$ autour du circuit jusqu'au $-$. Sur
la boucle à une lampe, les flèches quittent le $+$, traversent la
lampe, et reviennent au $-$.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:3}
L'erreur selon laquelle les appareils consomment le courant, de sorte
que les lampes «~en aval~» devraient briller plus faiblement. Un
éclat égal --- qui survit à tous les échanges --- montre que le même
courant entier traverse les deux~: rien n'est mangé en route.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:4}
C'est un seul et même courant~: une boucle unique porte une marche
unique à travers chacun de ses appareils.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:5}
Il se divise --- une part de la marche prend chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} --- et,
au second \omterm{def:g5:series-parallel-circuits:parallel}{nœud}, les parts se rejoignent, au complet. Chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche}
jouit de la pleine poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:6}
Seule $=$ avec un voisin \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation} (plein éclat), puis avec un
compagnon \omterm{def:g4:series-circuits:series}{en série} (faible), puis avec deux compagnons \omterm{def:g4:series-circuits:series}{en série} (plus
faible encore).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:7}
La dérivation~: deux \omterm{def:g5:series-parallel-circuits:parallel}{branches} tirant chacune une marche complète
doublent la livraison totale de la \omterm{def:g3:first-electric-circuit:battery}{pile}, alors que la paire \omterm{def:g4:series-circuits:series}{en série}
n'envoie qu'une seule marche étranglée à travers les deux. Le plein
éclat se paie à la \omterm{def:g3:first-electric-circuit:battery}{pile}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:8}
Les sièges sont les marcheurs chargés, qui circulent sans fin et ne
sont jamais consommés~; les skieurs sont l'\omterm{def:g5:everyday-energy:energy}{énergie}, chargée à la \omterm{def:g3:first-electric-circuit:battery}{pile}
et déposée aux lampes sous forme de \omterm{def:g1:light-and-shadows:source}{lumière} et de \omterm{def:g5:heat-and-insulation:heat}{chaleur}. Ce qui
s'épuise, c'est le stock de skieurs de la station --- l'\omterm{def:g5:everyday-energy:energy}{énergie} en
réserve dans la \omterm{def:g3:first-electric-circuit:battery}{pile}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:9}
Toute boucle passant par A traverse l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur}, puis soit B, soit
C --- les compagnons \omterm{def:g4:series-circuits:series}{en série} de A sont donc l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} et la
\emph{paire} de \omterm{def:g5:series-parallel-circuits:parallel}{branches} prise dans son ensemble, mais ni B ni C
séparément. Couper la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de C envoie toute la marche dans B~: A
continue de briller (un peu différemment --- une route au lieu de
deux --- comme le mesureront les instruments de l'an prochain).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:10}
Un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} général sur la route commune, juste après
l'alimentation, puis dix \omterm{def:g5:series-parallel-circuits:parallel}{branches} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}, une lampe chacune.
La loi de la dérivation accorde à chaque lampe un plein éclat
indépendant~; le placement \omterm{def:g4:series-circuits:series}{en série} du général --- sur la route que
toutes les boucles partagent --- permet à un seul \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} de
commander les dix.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:11}
La division au \omterm{def:g5:series-parallel-circuits:parallel}{nœud} n'est pas un partage à parts égales~: la route
propre d'une \omterm{def:g5:series-parallel-circuits:parallel}{branche} peut résister à la marche, et le long fil fin
étrangle la part de sa branche --- moins de marcheurs par seconde, un
éclat plus faible. Les routes elles-mêmes combattent le courant~:
c'est l'idée derrière la résistance de l'an prochain.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:12}
X, seule sur sa branche, reçoit la pleine poussée~: c'est la plus
brillante. Y et Z se partagent \omterm{def:g4:series-circuits:series}{en série} la poussée de leur \omterm{def:g5:series-parallel-circuits:parallel}{branche}~:
également faibles, toutes deux en dessous de X. Si Z grille, sa
boucle est rompue~: Y s'éteint avec elle --- X continue de briller,
indifférente. Si c'est X qui grille, seule la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de X meurt~: Y
et Z continuent, aussi faibles qu'avant.
\end{solution}

\begin{solution}{pb:g7:circuits-series-parallel:1}
\textbf{1.} Quatre \omterm{def:g5:series-parallel-circuits:parallel}{branches} \omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}, un service par \omterm{def:g5:series-parallel-circuits:parallel}{branche}~:
\omterm{def:g5:series-parallel-circuits:parallel}{en dérivation}, chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} travaille à pleine puissance quoi que
fassent les autres. Tout appariement \omterm{def:g4:series-circuits:series}{en série} laisserait
l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} d'une lampe de lecture ou une lampe d'escalier grillée
affaiblir ou éteindre la grande lampe --- impardonnable.
\textbf{2.} Le général sur la route commune, juste après la
batterie~; chacun des quatre \omterm{ex:g3:first-electric-circuit:switch}{interrupteurs} de \omterm{def:g5:series-parallel-circuits:parallel}{branche} \omterm{def:g4:series-circuits:series}{en série} avec
son propre service, à l'intérieur de sa branche.
\textbf{3.} Oui --- par la loi de la dérivation~: une coupure dans
une \omterm{def:g5:series-parallel-circuits:parallel}{branche} n'arrête que la part de cette \omterm{def:g5:series-parallel-circuits:parallel}{branche}~; les autres
poursuivent leur marche.
\textbf{4.} Batterie, \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} général, puis un \omterm{def:g5:series-parallel-circuits:parallel}{nœud} d'où partent
quatre \omterm{def:g5:series-parallel-circuits:parallel}{branches} --- (grande lampe $+$ \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}), (corne de brume
$+$ \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}), (lampe de lecture $+$ \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}), (éclairage
d'escalier $+$ \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}) --- qui se rejoignent et reviennent à la
batterie.
\textbf{5.} Oui~: chaque \omterm{def:g5:series-parallel-circuits:parallel}{branche} relie les deux côtés de la batterie,
et chacune reçoit donc la pleine poussée, quel que soit le nombre de
voisines qui boivent à côté.
\textbf{6.} Non~: le courant qui traverse la lampe en ressort entier
--- les sièges circulent sans être mangés. Ce que la lampe prend,
c'est de l'\omterm{def:g5:everyday-energy:energy}{énergie} au réservoir de la batterie, livrée par la marche
qui passe.
\textbf{7.} Quatre marches complètes à la fois~: l'\omterm{def:g5:everyday-energy:energy}{énergie} de la
batterie s'épuise vite avec quatre services. La facture se paie dans
la salle des batteries --- en \omterm{def:g5:everyday-energy:energy}{énergie} stockée, jamais en courant
perdu.
\textbf{8.} Sortie par le $+$ de la batterie, passage par
l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} général jusqu'au \omterm{def:g5:series-parallel-circuits:parallel}{nœud}~; entrée dans la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de la
lampe de lecture, traversée de son \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} fermé et de la
lampe~; passage par le \omterm{def:g5:series-parallel-circuits:parallel}{nœud} où les trois autres \omterm{def:g5:series-parallel-circuits:parallel}{branches} (éteintes)
se rejoignent~; retour au $-$ de la batterie. Elle évite entièrement
les \omterm{def:g5:series-parallel-circuits:parallel}{branches} de la grande lampe, de la corne de brume et de
l'escalier.
\textbf{9.} Rien d'autre --- seule la \omterm{def:g5:series-parallel-circuits:parallel}{branche} de l'escalier meurt
avec sa lampe~: c'est l'indépendance de la loi de la dérivation.
\textbf{10.} Un \omterm{def:g6:circuits-and-safety:short}{court-circuit} en amont de toutes les \omterm{def:g5:series-parallel-circuits:parallel}{branches}. La
marche abandonne toutes les routes utiles pour le raccourci nu et
sans effort~: les lampes et la corne sont affamées pendant que le fil
de raccourci encaisse une ruée folle et brûlante --- tous les
services tombent d'un coup jusqu'à ce que le fil soit retiré.
\textbf{11.} Donner à la grande lampe sa propre boucle indépendante
sur la seconde batterie~: batterie deux, son propre \omterm{ex:g3:first-electric-circuit:switch}{interrupteur}, la
grande lampe --- sans partager le moindre fil avec le tableau
général. Un raccourci en travers de la première batterie fait alors
taire tout \emph{sauf} la \omterm{def:g1:light-and-shadows:source}{lumière} des navires.
\textbf{12.} Par exemple~: «~La loi de la dérivation a gardé chaque
service souverain cette nuit. La nuit a réfuté une fois de plus
l'idée que les lampes dévorent le courant --- la marche revenait
entière alors même que les batteries se vidaient. Et un seul fil nu,
court-circuitant la batterie, nous a coûté notre pire heure~: toutes
les routes vides pendant que le raccourci brûlait.~»
\end{solution}
