\section*{Chapitre \ref{ch:g7:motion-average-speed} --- Graphiques du mouvement et vitesse moyenne}

\begin{solution}{exo:g7:motion-average-speed:1}
$v = d/t$~; \omterm{def:g6:measurement-in-science:measuring}{unités} \unit{km/h} et \unit{m/s}. La distance~:
$d = v \times t$~; la \omterm{def:g2:measuring-time:duration}{durée}~: $t = d/v$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:2}
Le ferry~: $45 \div 3 = \qty{15}{km/h}$. Le lièvre~:
$100 \div 8 = \qty{12.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:3}
$5 \times 3.6 = \qty{18}{km/h}$~; \ $72 \div 3.6 = \qty{20}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:4}
Droite~: \omterm{def:g6:describing-motion:uniform}{mouvement uniforme}~; plus raide~: plus rapide~;
horizontal~: à l'arrêt. Une courbe qui s'aplatit progressivement
raconte un ralentissement.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:5}
De la minute $35$ à la minute $60$ --- le troisième tronçon~: c'est
le moins raide des tronçons montants, cela se lit directement à sa
pente plus douce.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:6}
Uniforme --- une seule droite. Elle gagne \qty{2}{km} par demi-heure~:
$v = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:7}
La distance totale sur la \omterm{def:g2:measuring-time:duration}{durée} totale, arrêts compris. Randonnée
entière~: $12 \div 4 = \qty{3}{km/h}$. En marchant seulement~:
$12 \div 3 = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:8}
Sur des distances égales, le tronçon lent dure tout simplement plus
longtemps --- une plus grande part de l'horloge du trajet est vécue à
l'allure lente, et cette allure lente parle donc avec une plus grande
part de la voix du trajet.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:9}
Milieu supposé~: \qty{22.5}{km/h}. Honnêtement~: aller
$30 \div 15 = 2$ \omterm{ex:g2:measuring-time:clock}{heures}, retour $30 \div 30 = 1$ heure~; total
\qty{60}{km} en $3$ \omterm{ex:g2:measuring-time:clock}{heures}~: \qty{20}{km/h}. La moitié lente a
possédé deux des trois \omterm{ex:g2:measuring-time:clock}{heures}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:10}
Graphique~: droite jusqu'à (\qty{30}{min}, \qty{20}{km})~; palier
jusqu'à $45$ min~; droite plus raide jusqu'à (\qty{60}{min},
\qty{35}{km}). Moyenne~: \qty{35}{km} en une heure ---
\qty{35}{km/h}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:11}
$d = 340 \times 6 = \qty{2040}{m}$ --- environ \qty{2}{km}. La
vieille règle~: $6 \div 3 = 2$ kilomètres --- d'accord avec la
formule, parce que \qty{340}{m/s} signifie très exactement à peu près
un kilomètre toutes les trois secondes.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:12}
Première moitié~: \qty{30}{km} à \qty{30}{km/h} coûte exactement $1$
heure. Une moyenne d'ensemble de \qty{60}{km/h} sur \qty{60}{km}
autorise un total d'exactement $1$ heure --- déjà dépensée jusqu'à la
dernière seconde. Les \qty{30}{km} restants devraient être parcourus
en $0$ heure~: aucune \omterm{def:g7:motion-average-speed:formula}{vitesse}, si héroïque soit-elle, ne rattrape la
moyenne~; le budget est épuisé.
\end{solution}

\begin{solution}{pb:g7:motion-average-speed:1}
\textbf{1.} Une course vive ($30$ min), un arrêt de $15$ minutes (une
livraison), une course plus douce ($30$ min), et un second arrêt de
$15$ minutes --- deux courses, deux visites.
\textbf{2.} \qty{9}{km} en une demi-heure~: \qty{18}{km/h}.
\textbf{3.} $15 - 9 = \qty{6}{km}$ en une demi-heure~:
\qty{12}{km/h}.
\textbf{4.} \qty{15}{km} en $1.5$ heure~: \qty{10}{km/h}.
\textbf{5.} $t = 12 \div 24 = 0.5$ heure~: $30$ minutes.
\textbf{6.} $20$ minutes font un tiers d'heure~:
$d = 18 \times \tfrac{1}{3} = \qty{6}{km}$.
\textbf{7.} $25$ minutes font $\tfrac{25}{60}$ heure~;
$v = 10 \div \tfrac{25}{60} = \qty{24}{km/h}$.
\textbf{8.} Distances~: $12 + 6 + 10 = \qty{28}{km}$~; \omterm{def:g2:measuring-time:duration}{durées}~:
$30 + 20 + 25 = 75$ minutes $= 1.25$ heure~; moyenne en roulant~:
$28 \div 1.25 = \qty{22.4}{km/h}$.
\textbf{9.} $37 \div 3 \approx \qty{12.3}{km/h}$ --- bien au-dessous
de $20$~: pas de prime (ses arrêts sont un travail honnête, mais la
règle les compte).
\textbf{10.} Max~: moitié rapide $12 \div 30 = 0.4$ h~; moitié lente
$12 \div 15 = 0.8$ h~; total \qty{24}{km} en $1.2$ h~: exactement
\qty{20}{km/h} --- pas \emph{au-dessus} de $20$~: pas de prime.
\textbf{11.} Son $22.5$ faisait la moyenne des deux \emph{nombres},
comme si chaque allure possédait la moitié de l'horloge. En vérité,
la moitié lente possédait $0.8$ de ses $1.2$ heure --- les deux tiers
de la voix du trajet --- et a tiré la vraie moyenne jusqu'à $20$.
\textbf{12.} Par exemple~: verser la prime sur la \omterm{def:g7:motion-average-speed:average}{vitesse moyenne}
\emph{en roulant} --- la distance totale divisée par le seul temps de
roulage, en excluant les paliers du graphique. Elle mesure l'allure
que les roues tiennent réellement, et récompense la \omterm{def:g7:motion-average-speed:formula}{vitesse} plutôt
que les livraisons sautées.
\textbf{13.} «~Un tronçon droit est une allure régulière~; un tronçon
horizontal est un arrêt avec l'horloge qui continue de tourner~; un
tronçon plus raide est une allure plus rapide --- la pente est la
\omterm{def:g7:motion-average-speed:formula}{vitesse}. Et jamais de moyenne de \omterm{def:g7:motion-average-speed:formula}{vitesses}~: faites la moyenne du
trajet --- distance totale sur \omterm{def:g2:measuring-time:duration}{durée} totale --- ou les \omterm{ex:g2:measuring-time:clock}{heures} lentes
se moqueront de la fiche.~»
\end{solution}
