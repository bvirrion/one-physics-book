\section*{अध्याय \ref{ch:g7:volume-mass-density} --- आयतन, द्रव्यमान, घनत्व}

\begin{solution}{exo:g7:volume-mass-density:1}
$\rho = m/V$: $\rho$ \omterm{def:g7:volume-mass-density:formula}{घनत्व}, $m$ नमूने का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, $V$ उसका \omterm{def:g6:mass-and-volume:volume}{आयतन}।
\omterm{def:g3:measuring-mass:units}{ग्राम} और \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} लेने पर $\rho$ \unit{g/cm^3} में निकलता है।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:2}
$\rho = 270 \div 100 = \qty{2.7}{g/cm^3}$: एल्यूमिनियम।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:3}
$m = 9.0 \times 40 = \qty{360}{g}$ ताँबा।
$V = 1930 \div 19.3 = \qty{100}{cm^3}$ सोना --- मुकुट वाले मुक़दमे
की अपनी संख्या।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:4}
टेयर करने से बेलन का अपना \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} हट जाता है और \omterm{def:g3:levers-and-scales:scale}{तुला} अकेले \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} की
ख़बर देती है --- यानी वही ईमानदार-शून्य वाला नियम। फिर सूत्र को \omterm{def:g3:levers-and-scales:scale}{तुला}
का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} $m$ और \omterm{met:g6:mass-and-volume:cylinder}{नवचंद्रक} का पाठ $V$ खिलाया जाता है।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:5}
$\rho = 70 \div 50 = \qty{1.4}{g/cm^3}$ --- शहद जैसा। $0.9$ वाली
बर्फ़ उससे कहीं कम घनी है: वह उसमें ऊपर-ऊपर तैरेगी।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:6}
दोनों \omterm{def:g6:measurement-in-science:measuring}{मात्रक} साथ-साथ बढ़ते हैं: एक \omterm{def:g6:mass-and-volume:volume}{लीटर} $1000$ \unit{cm^3} है और एक
\omterm{def:g3:measuring-mass:units}{किलोग्राम} $1000$ \omterm{def:g3:measuring-mass:units}{ग्राम}, इसलिए दोनों हज़ार आपस में कट जाते हैं --- और
एक ही संख्या दोनों का काम कर देती है। \omterm{def:g3:measuring-mass:units}{ग्राम} को \omterm{def:g6:mass-and-volume:volume}{लीटर} के साथ मिलाने पर
यह कटाव छूट जाता है और उत्तर हज़ार गुना ग़लत निकलता है।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:7}
$\rho = 58 \div 5.0 = \qty{11.6}{g/cm^3}$ --- सीसे के $11.3$ के
पास, और सोने के $19.3$ के आस-पास भी नहीं। फ़ैसला: शुद्ध सोना नहीं;
शायद सोने के कपड़े पहने सीसे के दिल वाला कोई नक़ली।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:8}
$100 \times 100 \times 100 = 1000000$ \unit{cm^3} --- यानी दस लाख।
हर एक \qty{1}{g} का, तो कुल \qty{1000000}{g} $=$ \qty{1000}{kg}:
एक घन \omterm{def:g2:measuring-length:units}{मीटर} पानी का पूरा एक टन --- और इसीलिए पानी वाला बड़ा बिस्तर
भूतल का फ़र्नीचर है।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:9}
$V = 260 - 200 = \qty{60}{cm^3}$;
$\rho = 420 \div 60 = \qty{7.0}{g/cm^3}$ --- एल्यूमिनियम ($2.7$)
और लोहे ($7.9$) के बीच, और लोहे से ज़रा नीचे। बीच में पड़ना
फुसफुसाता है: कोई मिश्रण या परत चढ़ा नक़ली, चाँदी का $10.5$ तो पक्के
तौर पर नहीं।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:10}
\omterm{def:g3:solids-liquids-gases:liquid}{द्रव} का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}: $1010 - 380 = \qty{630}{g}$; \omterm{def:g6:mass-and-volume:volume}{आयतन}:
$\qty{75}{cL} = \qty{750}{cm^3}$; इसलिए $\rho = 630 \div 750 =
\qty{0.84}{g/cm^3}$ --- कोई तेल।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:11}
पुड़िया का \omterm{def:g7:volume-mass-density:formula}{घनत्व}: $1200 \div 4000 = \qty{0.3}{g/cm^3}$ --- पानी के
$1.0$ से बहुत नीचे: बोया ऊपर-ऊपर तैरता है। एल्यूमिनियम का $2.7$
सिर्फ़ धातु की खाल का वर्णन करता है; पुड़िया तो ज़्यादातर बंद \omterm{def:g2:air-around-us:air}{हवा} है,
और तैरने के नियम का सामना पुड़िया करती है --- कुल \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बटा कुल
\omterm{def:g6:mass-and-volume:volume}{आयतन}।
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:12}
कार्यविधि: ज़ंजीर तौलो ($m$); उठान वाली विधि से उसका \omterm{def:g6:mass-and-volume:volume}{आयतन} मापो, पूरी
डूबी हुई, बिना बुलबुलों के ($V$); $\rho = m/V$ निकालो; शुद्धता के
लिए लगभग \qty{9.0}{g/cm^3} अपेक्षित है। ईमानदार संदेह: खोखली कड़ियाँ
\omterm{def:g2:air-around-us:air}{हवा} क़ैद करके $V$ फुला देती हैं और \omterm{def:g7:volume-mass-density:formula}{घनत्व} झूठा कम दिखा देती हैं; और
कोई चालाक मिश्रण (या परत चढ़ा क्रोड) $9.0$ के पास आ सकता है जबकि उसमें
शुद्ध ताँबा हो ही नहीं। कसने वाली जाँच: कड़ियों में पानी भरकर (बुलबुले
झाड़कर) \omterm{def:g6:mass-and-volume:volume}{आयतन} दोबारा मापो, और कोई स्वतंत्र दस्तख़त भी जोड़ो --- जैसे
\omterm{def:g1:magnets:magnet}{चुंबक} (इस्पात का क्रोड ख़ुद को उसी पल पकड़ा देता है), या कार्यशाला में
मापा गया गलन-व्यवहार।
\end{solution}

\begin{solution}{pb:g7:volume-mass-density:1}
\textbf{1.} $5400 \div 2000 = \qty{2.7}{g/cm^3}$: एल्यूमिनियम।
\textbf{2.} $1800 \div 200 = \qty{9.0}{g/cm^3}$: ताँबा।
\textbf{3.} $4520 \div 400 = \qty{11.3}{g/cm^3}$: सीसा --- घना,
नरम, और लापरवाही से छूने पर ज़हरीला: दस्ताने पहनो।
\textbf{4.} $8100 \div 1500 = \qty{5.4}{g/cm^3}$ --- एल्यूमिनियम
($2.7$) और लोहे ($7.9$) के बीच: यानी दोनों का मिला-जुला कबाड़,
\omterm{def:g6:mass-and-volume:volume}{आयतन} के हिसाब से मोटे तौर पर आधा-आधा।
\textbf{5.} $V = 540 \div 2.7 = \qty{200}{cm^3}$।
\textbf{6.} एक घन: $m = 9.0 \times 25 = \qty{225}{g}$। लच्छे के
\qty{1800}{g} से $1800 \div 225 = 8$ पूरे घन बनते हैं।
\textbf{7.} $\qty{20}{kg} = \qty{20000}{g}$; $20000 \div 225 =
88.9$: यानी $88$ घन।
\textbf{8.} $m = 7.9 \times 3000 = \qty{23700}{g} \approx
\qty{23.7}{kg}$।
\textbf{9.} $3860 \div 200 = \qty{19.3}{g/cm^3}$ --- सोने के
दस्तख़त से पूरा-पूरा मेल।
\textbf{10.} टंगस्टन --- \omterm{def:g7:volume-mass-density:formula}{घनत्व} $19.3$, दशमलव तक वही दस्तख़त। \omterm{def:g7:volume-mass-density:formula}{घनत्व}
वाली जाँच संदिग्ध छाँट देती है; पर जो पदार्थ संयोग से एक ही दस्तख़त
रखते हों, उन्हें वह अलग नहीं कर सकती।
\textbf{11.} कोई भी उपकरण नहीं हारा: \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और \omterm{def:g6:mass-and-volume:volume}{आयतन} दोनों सही हैं,
और भाग भी। जाँच ने ईमानदारी से इतना साबित किया कि सिल्ली \emph{या तो}
सोना है \emph{या} ठीक सोने वाले \omterm{def:g7:volume-mass-density:formula}{घनत्व} की कोई और चीज़ --- यानी उसने
संदिग्ध दो तक सीमित कर दिए।
\textbf{12.} बचाव लायक चुनाव: ठोंकी गई छड़ की खनक और उसका एहसास
(टंगस्टन कहीं ज़्यादा कठोर है --- किसी छिपे कोने पर रेती या कठोरता की
जाँच उन्हें झट से अलग कर देती है), या यह \omterm{def:g6:measurement-in-science:measuring}{मापना} कि छड़ \omterm{def:g5:heat-and-insulation:heat}{ऊष्मा} या धारा
कितनी अच्छी तरह ले जाती है (सोना सबसे अच्छे चालकों में है, टंगस्टन
काफ़ी पीछे) --- ये किसी भट्ठी से कहीं नरम हैं, और \omterm{def:g7:volume-mass-density:formula}{घनत्व} के साथ मिलकर
फ़ैसलाकुन भी।
\textbf{13.} जैसे: ``\omterm{def:g7:volume-mass-density:formula}{घनत्व} यह साबित कर सकता है कि कोई पदार्थ वह
\emph{नहीं} है जो होने का दावा करता है। वह क्या है, यह वह सिर्फ़ सुझा
सकता है --- दस्तख़त संदिग्ध छाँटते भर हैं। और अकेला वह कभी सज़ा नहीं
सुना सकता: पहचान को दो स्वतंत्र गवाह चाहिए।''
\end{solution}
