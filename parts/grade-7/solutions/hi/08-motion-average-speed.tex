\section*{अध्याय \ref{ch:g7:motion-average-speed} --- गति के आलेख और औसत चाल}

\begin{solution}{exo:g7:motion-average-speed:1}
$v = d/t$; \omterm{def:g6:measurement-in-science:measuring}{मात्रक} \unit{km/h} और \unit{m/s}। दूरी: $d = v \times t$;
समय: $t = d/v$।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:2}
नाव: $45 \div 3 = \qty{15}{km/h}$। ख़रगोश: $100 \div 8 =
\qty{12.5}{m/s}$।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:3}
$5 \times 3.6 = \qty{18}{km/h}$; \ $72 \div 3.6 = \qty{20}{m/s}$।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:4}
सीधी रेखा: एक-सी गति; ज़्यादा खड़ी: ज़्यादा तेज़; सपाट: ठहराव। और
धीरे-धीरे सपाट पड़ता मोड़ धीमा होने की ख़बर देता है।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:5}
मिनट $35$ से $60$ तक --- यानी तीसरा हिस्सा: चढ़ते हिस्सों में वही सबसे
कम खड़ा है, जो उसके ज़्यादा नरम झुकाव से सीधे पढ़ा जा सकता है।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:6}
एक-सी गति --- एक ही सीधी रेखा। वह हर आधे घंटे में \qty{2}{km} कमाती
है: $v = \qty{4}{km/h}$।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:7}
कुल दूरी बटा कुल समय, ठहरावों समेत। पूरी यात्रा: $12 \div 4 =
\qty{3}{km/h}$। सिर्फ़ चलते हुए: $12 \div 3 = \qty{4}{km/h}$।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:8}
बराबर दूरियों पर धीमा हिस्सा बस ज़्यादा देर तक चलता है --- यानी
यात्रा की घड़ी का ज़्यादा हिस्सा उसी धीमी रफ़्तार में जीते हुए बीतता
है, इसलिए यात्रा की आवाज़ में धीमी रफ़्तार का हिस्सा ज़्यादा हो जाता
है।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:9}
बीच के बिंदु वाला अंदाज़ा: \qty{22.5}{km/h}। ईमानदारी से: जाते हुए
$30 \div 15 = 2$ घंटे, लौटते हुए $30 \div 30 = 1$ घंटा; कुल $3$ \omterm{ex:g2:measuring-time:clock}{घंटों}
में \qty{60}{km}: \qty{20}{km/h}। तीनों में से दो घंटे धीमे आधे हिस्से
के हड़पे हुए थे।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:10}
आलेख: (\qty{30}{min}, \qty{20}{km}) तक सीधी रेखा; $45$ मिनट तक सपाट;
और फिर (\qty{60}{min}, \qty{35}{km}) तक सीधी तथा ज़्यादा खड़ी रेखा।
औसत: एक घंटे में \qty{35}{km} --- \qty{35}{km/h}।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:11}
$d = 340 \times 6 = \qty{2040}{m}$ --- यानी क़रीब \qty{2}{km}। पुराना
नियम: $6 \div 3 = 2$ किलोमीटर --- यानी सहमति, क्योंकि \qty{340}{m/s}
का मतलब है हर तीन सेकंड में लगभग ठीक एक किलोमीटर।
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:12}
पहला आधा हिस्सा: \qty{30}{km/h} से \qty{30}{km} में ठीक $1$ घंटा लगता
है। और \qty{60}{km} पर कुल \qty{60}{km/h} का औसत कुल ठीक $1$ घंटे की
ही इजाज़त देता है --- यानी वह आख़िरी सेकंड तक पहले ही ख़र्च हो चुका
है। बचे हुए \qty{30}{km} $0$ घंटे में तय करने पड़ेंगे: कोई भी \omterm{def:g7:motion-average-speed:formula}{चाल},
चाहे कितनी ही वीर हो, वह औसत नहीं बना सकती; बजट ख़त्म है।
\end{solution}

\begin{solution}{pb:g7:motion-average-speed:1}
\textbf{1.} तेज़ सवारी ($30$ मिनट), $15$ मिनट का ठहराव (सामान
पहुँचाना), फिर नरम सवारी ($30$ मिनट), और दूसरा $15$ मिनट का ठहराव ---
यानी दो दौड़, दो पड़ाव।
\textbf{2.} आधे घंटे में \qty{9}{km}: \qty{18}{km/h}।
\textbf{3.} आधे घंटे में $15 - 9 = \qty{6}{km}$: \qty{12}{km/h}।
\textbf{4.} $1.5$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{15}{km}: \qty{10}{km/h}।
\textbf{5.} $t = 12 \div 24 = 0.5$ घंटा: $30$ मिनट।
\textbf{6.} $20$ मिनट घंटे का एक तिहाई है: $d = 18 \times
\tfrac{1}{3} = \qty{6}{km}$।
\textbf{7.} $25$ मिनट $\tfrac{25}{60}$ घंटा है; $v = 10 \div
\tfrac{25}{60} = \qty{24}{km/h}$।
\textbf{8.} दूरियाँ: $12 + 6 + 10 = \qty{28}{km}$; समय:
$30 + 20 + 25 = 75$ मिनट $= 1.25$ घंटे; चलते रहने का औसत:
$28 \div 1.25 = \qty{22.4}{km/h}$।
\textbf{9.} $37 \div 3 \approx \qty{12.3}{km/h}$ --- यानी $20$ से
कहीं नीचे: बोनस नहीं (उसके ठहराव ईमानदार काम हैं, पर नियम उन्हें भी
गिनता है)।
\textbf{10.} मैक्स: तेज़ आधा $12 \div 30 = 0.4$ घंटा; धीमा आधा
$12 \div 15 = 0.8$ घंटा; कुल $1.2$ \omterm{ex:g2:measuring-time:clock}{घंटों} में \qty{24}{km}: ठीक-ठीक
\qty{20}{km/h} --- यानी $20$ से \emph{ऊपर} नहीं: बोनस नहीं।
\textbf{11.} उसने दोनों \emph{अंकों} का औसत निकाल लिया, मानो हर
रफ़्तार के पास घड़ी का आधा-आधा हिस्सा हो। सच में तो उसके $1.2$ \omterm{ex:g2:measuring-time:clock}{घंटों}
में से $0.8$ धीमे आधे हिस्से के थे --- यानी यात्रा की आवाज़ का
दो-तिहाई --- और उसी ने सच्चे औसत को $20$ तक घसीट लिया।
\textbf{12.} जैसे: बोनस \emph{चलते रहने} की \omterm{def:g7:motion-average-speed:average}{औसत चाल} पर दो --- यानी
कुल दूरी बटा सिर्फ़ सवारी का समय, और आलेख के सपाट हिस्से बाहर। यह उस
रफ़्तार को मापता है जो पहिए सचमुच बनाए रखते हैं, और तेज़ सवारी को
इनाम देता है, न कि छोड़ी हुई सुपुर्दगियों को।
\textbf{13.} ``सीधा हिस्सा एक-सी रफ़्तार है; सपाट हिस्सा वह ठहराव है
जिसमें घड़ी चलती रहती है; और ज़्यादा खड़ा हिस्सा ज़्यादा तेज़ रफ़्तार
--- खड़ापन ही \omterm{def:g7:motion-average-speed:formula}{चाल} है। और \omterm{def:g7:motion-average-speed:formula}{चाल} के अंकों का औसत कभी मत निकालो: औसत
यात्रा का निकालो --- कुल दूरी बटा कुल समय --- वरना धीमे घंटे परची का
मज़ाक़ बना देंगे।''
\end{solution}
