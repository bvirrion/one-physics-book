\section*{Bab \ref{ch:g7:volume-mass-density} --- Volume, Massa, Massa Jenis}

\begin{solution}{exo:g7:volume-mass-density:1}
$\rho = m/V$: $\rho$ \omterm{def:g3:measuring-mass:mass}{massa} jenisnya, $m$ \omterm{def:g3:measuring-mass:mass}{massa} cuplikannya, $V$
\omterm{def:g6:mass-and-volume:volume}{volumenya}. Dengan \omterm{def:g3:measuring-mass:units}{gram} dan \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}, $\rho$ keluar dalam
\unit{g/cm^3}.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:2}
$\rho = 270 \div 100 = \qty{2.7}{g/cm^3}$: aluminium.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:3}
$m = 9.0 \times 40 = \qty{360}{g}$ tembaga.
$V = 1930 \div 19.3 = \qty{100}{cm^3}$ emas --- bilangan milik
perkara mahkotanya sendiri.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:4}
Menara menyingkirkan \omterm{def:g3:measuring-mass:mass}{massa} gelas ukurnya sendiri sehingga \omterm{def:g3:levers-and-scales:scale}{neracanya}
melaporkan \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} saja --- itulah aturan nol yang jujur. Rumusnya
lalu diberi \omterm{def:g3:measuring-mass:mass}{massa} $m$ dari \omterm{def:g3:levers-and-scales:scale}{neracanya} dan bacaan \omterm{met:g6:mass-and-volume:cylinder}{meniskus} $V$.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:5}
$\rho = 70 \div 50 = \qty{1.4}{g/cm^3}$ --- serupa madu. Es pada
$0.9$ jauh kurang padat: ia \omterm{def:g1:floating-and-sinking:words}{terapung} tinggi di dalamnya.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:6}
Kedua \omterm{def:g6:measurement-in-science:measuring}{satuannya} berskala bersama: satu \omterm{def:g6:mass-and-volume:volume}{liter} adalah $1000$
\unit{cm^3} dan satu \omterm{def:g3:measuring-mass:units}{kilogram} adalah $1000$ \omterm{def:g3:measuring-mass:units}{gram}, sehingga kedua
ribuannya saling meniadakan --- satu bilangan melayani keduanya.
Mencampur \omterm{def:g3:measuring-mass:units}{gram} dengan \omterm{def:g6:mass-and-volume:volume}{liter} melewatkan peniadaan itu lalu menyerahkan
bilangan yang keliru seribu kali lipat.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:7}
$\rho = 58 \div 5.0 = \qty{11.6}{g/cm^3}$ --- dekat $11.3$ milik
timbal, sama sekali jauh dari $19.3$ milik emas. Putusannya: bukan
emas murni; kemungkinan pengelabu berhati timbal berbaju emas.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:8}
$100 \times 100 \times 100 = 1000000$ \unit{cm^3} --- sejuta.
Pada \qty{1}{g} masing-masing, itu \qty{1000000}{g} $=$ \qty{1000}{kg}:
seton penuh tiap \omterm{def:g2:measuring-length:units}{meter} kubik air --- dan itulah sebabnya kasur air
yang besar adalah perabot bagi lantai dasar.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:9}
$V = 260 - 200 = \qty{60}{cm^3}$;
$\rho = 420 \div 60 = \qty{7.0}{g/cm^3}$ --- di antara aluminium
($2.7$) dan besi ($7.9$), paling dekat di bawah besi. Keadaan di
antara itu berbisik: sebuah campuran atau pengelabu bersepuh, dan
pasti bukan $10.5$ milik perak.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:10}
\omterm{def:g3:measuring-mass:mass}{Massa} \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya}: $1010 - 380 = \qty{630}{g}$; \omterm{def:g6:mass-and-volume:volume}{volumenya}:
$\qty{75}{cL} = \qty{750}{cm^3}$; sehingga $\rho = 630 \div 750 =
\qty{0.84}{g/cm^3}$ --- sebuah minyak.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:11}
\omterm{def:g7:volume-mass-density:formula}{Massa jenis} bungkusannya: $1200 \div 4000 = \qty{0.3}{g/cm^3}$ ---
jauh di bawah $1.0$ milik air: pelampungnya \omterm{def:g1:floating-and-sinking:words}{terapung} tinggi. Angka
$2.7$ milik aluminium hanya menggambarkan kulit logamnya; bungkusannya
sebagian besar berupa \omterm{def:g2:air-around-us:air}{udara} yang terkurung, dan bungkusan itulah ---
\omterm{def:g3:measuring-mass:mass}{massa} total dibagi \omterm{def:g6:mass-and-volume:volume}{volume} total --- yang menghadapi aturan
\omterm{def:g1:floating-and-sinking:words}{terapungnya}.
\end{solution}

\begin{solution}{exo:g7:volume-mass-density:12}
Tata caranya: timbanglah rantainya ($m$); ukurlah \omterm{def:g6:mass-and-volume:volume}{volumenya} dengan
cara kenaikan air, terendam sepenuhnya, tanpa gelembung ($V$);
hitunglah $\rho = m/V$; kemurniannya mengharapkan kira-kira
\qty{9.0}{g/cm^3}. Keraguan yang jujur: mata rantai yang berongga
memerangkap \omterm{def:g2:air-around-us:air}{udara} lalu menggembungkan $V$, sehingga memalsukan \omterm{def:g7:volume-mass-density:formula}{massa
jenis} yang rendah; dan campuran yang licik (atau inti bersepuh) dapat
mendarat dekat $9.0$ padahal sama sekali tidak memuat tembaga murni.
Uji yang memperkuatnya: ulangi pengukuran \omterm{def:g6:mass-and-volume:volume}{volumenya} sesudah membanjiri
mata rantainya (kibaskan gelembungnya keluar), lalu tambahkan satu
tanda tangan yang mandiri --- misalnya \omterm{def:g1:magnets:magnet}{magnet} (inti baja mengkhianati
dirinya seketika) atau, di sebuah bengkel, perilaku \omterm{def:g6:water-states:changes}{peleburan} yang
terukur.
\end{solution}

\begin{solution}{pb:g7:volume-mass-density:1}
\textbf{1.} $5400 \div 2000 = \qty{2.7}{g/cm^3}$: aluminium.
\textbf{2.} $1800 \div 200 = \qty{9.0}{g/cm^3}$: tembaga.
\textbf{3.} $4520 \div 400 = \qty{11.3}{g/cm^3}$: timbal --- padat,
lunak, dan beracun kalau ditangani sembarangan: pakailah sarung
tangan.
\textbf{4.} $8100 \div 1500 = \qty{5.4}{g/cm^3}$ --- di antara
aluminium ($2.7$) dan besi ($7.9$): sekarung campuran keduanya,
kasarnya separuh-separuh menurut \omterm{def:g6:mass-and-volume:volume}{volume}.
\textbf{5.} $V = 540 \div 2.7 = \qty{200}{cm^3}$.
\textbf{6.} Satu kubus: $m = 9.0 \times 25 = \qty{225}{g}$. Angka
\qty{1800}{g} milik gulungannya menghasilkan $1800 \div 225 = 8$ kubus
penuh.
\textbf{7.} $\qty{20}{kg} = \qty{20000}{g}$; $20000 \div 225 =
88.9$: $88$ kubus.
\textbf{8.} $m = 7.9 \times 3000 = \qty{23700}{g} \approx
\qty{23.7}{kg}$.
\textbf{9.} $3860 \div 200 = \qty{19.3}{g/cm^3}$ --- padanan yang
sempurna bagi emas.
\textbf{10.} Wolfram --- \omterm{def:g3:measuring-mass:mass}{massa} jenisnya $19.3$, tanda tangan yang sama
sampai ke angka di belakang koma. Uji \omterm{def:g3:measuring-mass:mass}{massa} jenisnya mengenali calon;
ia tidak dapat membedakan zat yang kebetulan berbagi satu tanda
tangan.
\textbf{11.} Tidak satu alat pun gagal: \omterm{def:g3:measuring-mass:mass}{massa} dan \omterm{def:g6:mass-and-volume:volume}{volumenya} benar,
dan pembagiannya juga benar. Ujinya dengan jujur menegakkan bahwa
batangannya \emph{entah} emas entah sesuatu yang \omterm{def:g3:measuring-mass:mass}{massa} jenisnya persis
\omterm{def:g7:volume-mass-density:formula}{massa jenis} emas --- ia mempersempit tersangkanya menjadi dua.
\textbf{12.} Pilihan yang dapat dibela: dentang dan rasa batangan yang
dipukul (wolfram jauh lebih keras --- sebuah kikir atau uji kekerasan
pada sudut yang tersembunyi membedakan keduanya dengan cepat), atau
ukuran ahli yang diterima tentang bagaimana batangannya menghantar
\omterm{def:g5:heat-and-insulation:heat}{kalor} atau arus (emas termasuk penghantar terbaik, wolfram jauh
tertinggal) --- lebih lembut daripada tungku mana pun, dan menentukan
kalau digabungkan dengan \omterm{def:g3:measuring-mass:mass}{massa} jenisnya.
\textbf{13.} Misalnya: ``\omterm{def:g7:volume-mass-density:formula}{Massa jenis} dapat membuktikan bahwa sebuah
zat \emph{bukan} apa yang diakuinya. Ia hanya dapat menyarankan ia itu
apa --- tanda tangannya mempersempit tersangkanya. Dan sendirian ia
tidak akan pernah dapat menjatuhkan hukuman: jati diri menghendaki dua
saksi yang mandiri.''
\end{solution}
