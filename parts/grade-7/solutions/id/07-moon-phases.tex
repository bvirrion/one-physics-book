\section*{Bab \ref{ch:g7:moon-phases} --- Fase Bulan}

\begin{solution}{exo:g7:moon-phases:1}
Tepat separuh, selalu --- separuh yang menghadap matahari. Sebuah \omterm{def:g7:moon-phases:phases}{fase}
hanyalah pemandangannya: bagian separuh yang tersinari itu yang
kebetulan menghadap Bumi.
\end{solution}

\begin{solution}{exo:g7:moon-phases:2}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru, \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} yang \omterm{def:g7:moon-phases:phases}{membesar}, perempat pertama, cembung yang
\omterm{def:g7:moon-phases:phases}{membesar}, purnama, cembung yang \omterm{def:g7:moon-phases:phases}{menyusut}, perempat terakhir, \omterm{ex:g2:moon-in-the-sky:shapes}{sabit}
yang \omterm{def:g7:moon-phases:phases}{menyusut} --- kira-kira $29.5$ hari sekelilingnya. Yang \omterm{def:g7:moon-phases:phases}{membesar}
adalah separuh yang bertumbuh, dari bulan baru sampai purnama.
\end{solution}

\begin{solution}{exo:g7:moon-phases:3}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru: kamu menghadap lampunya dengan bolanya ke arah lampu ---
separuhnya yang tersinari berpaling darimu. Purnama: lampu di
punggungmu, bolanya di depanmu --- separuhnya yang tersinari berpaling
sepenuhnya ke arahmu.
\end{solution}

\begin{solution}{exo:g7:moon-phases:4}
Dua saja di antaranya: \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} Bumi menunjuk menjauhi Matahari
sehingga hanya dapat mencapai \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pada kedudukan purnama; \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
\omterm{ex:g2:moon-in-the-sky:shapes}{sabit} berdiri di sisi matahari dari kita, sama sekali tidak dekat
\omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} itu; dan tepi \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} yang bundar tidak akan
pernah dapat memahat lengkung ganda cembungnya.
\end{solution}

\begin{solution}{exo:g7:moon-phases:5}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru gelap bagi kita karena sisi malam \emph{miliknya sendiri}
yang menghadap kita --- tanpa melibatkan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} kita, dan sisi
jauhnya bermandi sinar matahari penuh. \omterm{def:g7:shadows-eclipses:lunar}{Gerhana bulan} adalah \omterm{ex:g2:moon-in-the-sky:shapes}{Bulan
purnama} yang menyeberangi \omterm{prop:g7:shadows-eclipses:umbra}{umbra} sejati Bumi: \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} kita sungguh
jatuh padanya, sejenak.
\end{solution}

\begin{solution}{exo:g7:moon-phases:6}
\omterm{ex:g2:moon-in-the-sky:shapes}{Bulan purnama} berdiri berseberangan dengan Matahari. Selagi Bumi yang
berputar membawa Matahari turun di bawah kaki langit sebelah barat,
titik langit yang berseberangan --- tempat \omterm{ex:g2:moon-in-the-sky:shapes}{Bulan purnama} menanti ---
pasti sedang terbit di timur: \omterm{def:g1:day-and-night:daynight}{matahari terbenam} dan terbitnya purnama
adalah satu peristiwa yang dilihat dari dua arah.
\end{solution}

\begin{solution}{exo:g7:moon-phases:7}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru bepergian bersama Matahari: pada tengah malam keduanya
berdiri di bawah kakimu, di sisi jauh Bumi. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru pada tengah
malam sama mustahilnya dengan Matahari pada tengah malam.
\end{solution}

\begin{solution}{exo:g7:moon-phases:8}
Menjauhi titik \omterm{def:g1:day-and-night:daynight}{matahari terbenamnya} --- busur \omterm{ex:g2:moon-in-the-sky:shapes}{sabitnya} melengkung ke
arah Matahari, tanduknya menjauh. Sisi yang tersinari harus menghadap
Matahari, dan tanduknya adalah tepi sisi tersinari yang sedang mundur:
geometri melarang mereka membidik ke arah matahari.
\end{solution}

\begin{solution}{exo:g7:moon-phases:9}
Di antara perempat pertama dan purnama --- kira-kira tiga perdelapan
jalan mengelilinginya, di sisi senjanya. Ia terbit pada pertengahan
sore. Lima malam kemudian: purnama --- terbit ketika \omterm{def:g1:day-and-night:daynight}{matahari terbenam},
naik sepanjang malam.
\end{solution}

\begin{solution}{exo:g7:moon-phases:10}
\omterm{def:g7:shadows-eclipses:solar}{Gerhana matahari}: hanya pada bulan baru --- Matahari--Bulan--Bumi
sebaris. \omterm{def:g7:shadows-eclipses:lunar}{Gerhana bulan}: hanya pada purnama ---
Matahari--Bumi--Bulan sebaris. Tetap jarang bahkan ketika itu, karena
\omterm{def:g6:sun-earth-moon:axisorbit}{orbit} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang miring biasanya membawanya di atas atau di bawah
garisnya yang persis: \omterm{def:g7:moon-phases:phases}{fasenya} perlu, kesejajarannya tidak terjamin.
\end{solution}

\begin{solution}{exo:g7:moon-phases:11}
Bumi memperlihatkan \omterm{def:g7:moon-phases:phases}{fase} yang berlawanan dengan \omterm{def:g7:moon-phases:phases}{fase} kita, pada \omterm{ex:g2:measuring-time:clock}{jam}
yang sama: ketika kita melihat bulan baru, para antariksawan bulan
melihat \emph{Bumi purnama} --- seluruh sisi siang kita menghadap
mereka; ketika bulan kita purnama, mereka melihat Bumi baru. Satu
geometri, yang dibaca dari kursi yang lain.
\end{solution}

\begin{solution}{exo:g7:moon-phases:12}
Jalannya: \omterm{def:g1:light-and-shadows:source}{cahaya} matahari menumbuk Bumi, Bumi yang terang
membaurkannya ke arah bulan, lalu sisi malam \omterm{def:g2:moon-in-the-sky:moon}{Bulan} membaurkan sedikit
kembali ke mata kita --- \omterm{def:g1:light-and-shadows:source}{cahaya} matahari yang dua kali diteruskan. Pada
\omterm{def:g7:moon-phases:phases}{fase} \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} kita, \omterm{def:g2:moon-in-the-sky:moon}{Bulan} melihat \emph{Bumi yang hampir purnama}
(aturan \omterm{def:g7:moon-phases:phases}{fase} yang berlawanan): sisi malamnya dibanjiri \omterm{def:g1:light-and-shadows:source}{cahaya} oleh
Bumi paling terang yang mungkin --- \omterm{def:g1:light-and-shadows:source}{cahaya} bumi yang terkuat. Menjelang
purnama kita, \omterm{def:g2:moon-in-the-sky:moon}{Bulan} melihat Bumi \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} yang tipis: lampu banjirnya
mengecil, dan silau \omterm{def:g7:moon-phases:phases}{fase} purnamanya menenggelamkan sisanya.
\end{solution}

\begin{solution}{pb:g7:moon-phases:1}
\textbf{1.} Perempat pertama dekat malam $7$; purnama dekat malam
$15$; lalu perempat terakhir dekat malam $22$ (dan daurnya menutup
dekat malam $29$--$30$).
\textbf{2.} Purnama: terbit ketika \omterm{def:g1:day-and-night:daynight}{matahari terbenam}. Perempat
pertama: terbit menjelang tengah hari. Perempat terakhir: terbit
menjelang tengah malam. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} baru: terbit dan terbenam bersama
Matahari.
\textbf{3.} \omterm{ex:g2:moon-in-the-sky:shapes}{Sabit} yang muda berdiri hanya sedikit di belakang Matahari
pada \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya}: ia tergantung di barat sesudah \omterm{def:g1:day-and-night:daynight}{matahari terbenam} lalu
segera menyusul Matahari turun. Purnama berdiri berseberangan: selagi
\omterm{def:g1:day-and-night:daynight}{Matahari terbenam} di barat, berseberangan berarti terbit di timur.
\textbf{4.} \omterm{def:g7:moon-phases:phases}{Fase} mana pun yang berbagi sebagian giliran Matahari:
cembung yang \omterm{def:g7:moon-phases:phases}{membesar} sepanjang sore, dan terutama \omterm{def:g7:moon-phases:phases}{fase} yang \omterm{def:g7:moon-phases:phases}{menyusut}
--- perempat terakhir dan \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} yang \omterm{def:g7:moon-phases:phases}{menyusut} menunggangi langit pagi
dengan tinggi.
\textbf{5.} Cembung yang \omterm{def:g7:moon-phases:phases}{membesar}: bolanya tersinari tiga perempat
berarti kamu berdiri di antara perempat pertama dan purnama --- lampu
di balik bahumu, bolanya di depan dan ke samping.
\textbf{6.} Mereka sedang melihat sisi malam \emph{milik} bolanya
sendiri --- separuh yang berpaling dari lampunya. Kamu-sebagai-Bumi
melemparkan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} juga, tetapi ia membentang di belakangmu,
menjauhi lampunya; bolanya tidak berada di dalamnya kecuali pada
kedudukan purnama --- dan pun hanya kalau sejajar dengan persis:
sebuah gerhana, bukan sebuah \omterm{def:g7:moon-phases:phases}{fase}.
\textbf{7.} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berputar tepat sekali tiap \omterm{def:g6:sun-earth-moon:axisorbit}{orbit}, sehingga separuh
yang sama menghadap Bumi selamanya --- kelinci, wajah, dan segalanya;
sisi jauhnya menunggu sebuah kamera terbang mengelilinginya.
\textbf{8.} Putaran dan \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} menjaga langkah yang sempurna: selagi
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} maju di sepanjang jalurnya, ia berputar tepat secukupnya untuk
menjaga wajah yang sama tetap menghadap kita --- seperti penari yang
mengelilingi pasangannya sambil selalu menghadapnya: sungguh berputar
(ruangannya melihat setiap sisinya), namun memperlihatkan satu wajah
saja kepada pasangannya.
\textbf{9.} Terbit pada tengah malam adalah jadwal perempat terakhir
--- sisi yang \omterm{def:g7:moon-phases:phases}{menyusut}: perempat terakhir (separuh yang tersinari
menghadap ke arah Matahari pagi).
\textbf{10.} Kesalahannya: \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} berdiri dekat Matahari, sehingga ia
tidak pernah dapat menunggangi langit dengan tinggi pada tengah malam;
tanduknya menunjuk \emph{menjauhi} Matahari, tidak pernah ke arahnya;
dan warna \omterm{def:g1:day-and-night:daynight}{matahari terbenam} pada tengah malam adalah kemustahilan
tersendiri. Tiga kekeliruan.
\textbf{11.} Tidak: \omterm{def:g7:shadows-eclipses:solar}{gerhana matahari} menuntut \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berada di antara
Matahari dan Bumi --- kedudukan bulan baru. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} malam ini berdiri
pada purnama, di sisi yang berseberangan: kedua peristiwanya tidak
mungkin berbagi satu malam.
\textbf{12.} Misalnya: ``Satu kenyataan yang menjalankan bulan itu:
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} selalu separuh tersinari, dan kita menonton separuh itu berkeliling
dari sisi yang berubah-ubah. Yang dikubur bulan ini: \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} Bumi
tidak ada hubungannya dengan \omterm{def:g7:moon-phases:phases}{fasenya} --- ia berkunjung hanya pada
gerhana yang jarang. Dan bawalah \omterm{ex:g2:measuring-time:clock}{jamnya} pulang: tiap \omterm{def:g7:moon-phases:phases}{fase} menepati \omterm{ex:g2:measuring-time:clock}{jam}
terbitnya sendiri --- purnama ketika \omterm{def:g1:day-and-night:daynight}{matahari terbenam}, perempatnya
pada tengah hari dan tengah malam --- sehingga bentuk \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
memberitahumu jadwalnya.''
\end{solution}
