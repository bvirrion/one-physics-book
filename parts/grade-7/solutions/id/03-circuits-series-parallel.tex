\section*{Bab \ref{ch:g7:circuits-series-parallel} --- Rangkaian: Lingkar, Seri, Paralel}

\begin{solution}{exo:g7:circuits-series-parallel:1}
Aliran partikel bermuatan yang berarak menembus rangkaiannya di bawah
dorongan \omterm{def:g3:first-electric-circuit:battery}{baterainya}. Ia ada hanya selagi sedikitnya satu lingkar
tertutup menghubungkan kedua kutubnya; bukalah lingkarnya dan
arak-arakannya berhenti di mana-mana sekaligus.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:2}
Di luar \omterm{def:g3:first-electric-circuit:battery}{baterainya}, dari $+$ mengelilingi rangkaiannya menuju $-$.
Pada lingkar berbohlam satu itu anak panahnya meninggalkan $+$,
menembus \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}, lalu pulang ke $-$.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:3}
Kekeliruan bahwa alat memakan habis arusnya, sehingga \omterm{def:g3:first-electric-circuit:bulb}{bohlam} ``di
hilir'' seharusnya menyala lebih lemah. Terang yang sama --- yang
selamat melewati setiap pertukaran --- menunjukkan bahwa arus utuh
yang sama menembus keduanya: tidak ada yang dimakan di tengah jalan.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:4}
Keduanya satu arus yang sama juga: satu lingkar tunggal membawa satu
arak-arakan tunggal menembus setiap alat padanya.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:5}
Ia terbagi --- sebagian arak-arakannya menempuh tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} --- dan
pada percabangan yang kedua bagian-bagiannya bergabung kembali,
lengkap. Tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} menikmati dorongan penuh \omterm{def:g3:first-electric-circuit:battery}{baterainya}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:6}
Sendirian $=$ dengan satu tetangga paralel (terang penuh), lalu dengan
satu kawan seri (redup), lalu dengan dua kawan seri (lebih redup
lagi).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:7}
Paralel: dua \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang masing-masing menarik arak-arakan penuh
menggandakan penyaluran total \omterm{def:g3:first-electric-circuit:battery}{baterainya}, sedangkan pasangan serinya
mengirim satu arak-arakan yang tercekik menembus keduanya. Terang
penuh dibayar di \omterm{def:g3:first-electric-circuit:battery}{baterainya}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:8}
Kursinya adalah para pengarak yang bermuatan, yang beredar tanpa henti
dan tidak pernah terpakai habis; para pemain skinya adalah \omterm{def:g5:everyday-energy:energy}{energinya},
yang dimuat di \omterm{def:g3:first-electric-circuit:battery}{baterainya} lalu diturunkan di lampunya sebagai \omterm{def:g1:light-and-shadows:source}{cahaya}
dan kehangatan. Yang habis adalah persediaan pemain ski milik
tempatnya --- yaitu \omterm{def:g5:everyday-energy:energy}{energi} yang tersimpan di \omterm{def:g3:first-electric-circuit:battery}{baterainya}.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:9}
Setiap lingkar yang menembus A melewati \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} lalu melewati B
atau C --- sehingga kawan seri A adalah \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} dan \emph{pasangan}
\omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} sebagai keseluruhan, tetapi bukan B sendirian dan bukan pula
C sendirian. Memutus \omterm{def:g5:series-parallel-circuits:parallel}{cabang} C mengirim seluruh arak-arakannya menembus
B: A tetap menyala (sedikit berbeda --- satu jalan alih-alih dua ---
sebagaimana akan diukur alat tahun depan).
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:10}
Satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} induk pada jalan bersamanya tepat sesudah sumbernya, lalu
sepuluh \omterm{def:g5:series-parallel-circuits:parallel}{cabang} paralel, satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} pada masing-masingnya. Hukum
paralelnya menganugerahkan terang penuh yang mandiri kepada setiap
\omterm{def:g3:first-electric-circuit:bulb}{bohlam}; penempatan seri induknya --- pada jalan yang dibagi setiap
lingkar --- membuat satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} memerintah kesepuluhnya.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:11}
Pembagian pada percabangannya bukan pembelahan yang sama rata: jalan
sebuah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} sendiri dapat melawan arak-arakannya, dan kawat yang
panjang dan tipis itu mencekik jatah \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} --- pengarak lebih
sedikit tiap sekon, nyala lebih lemah. Jalan itu sendiri melawan
arusnya: itulah gagasan di balik hambatan tahun depan.
\end{solution}

\begin{solution}{exo:g7:circuits-series-parallel:12}
X, sendirian pada \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}, mendapat dorongan penuh: paling terang. Y
dan Z berbagi dorongan \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} secara seri: sama-sama redup,
keduanya di bawah X. Kalau Z putus, lingkarnya patah: Y ikut gelap
bersamanya --- X terus bersinar, tanpa peduli. Kalau X yang putus,
hanya \omterm{def:g5:series-parallel-circuits:parallel}{cabang} X yang mati: Y dan Z berjalan terus, seredup sebelumnya.
\end{solution}

\begin{solution}{pb:g7:circuits-series-parallel:1}
\textbf{1.} Empat \omterm{def:g5:series-parallel-circuits:parallel}{cabang} paralel, satu tugas pada masing-masingnya:
secara paralel, tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} bekerja pada kekuatan penuh apa pun yang
dilakukan \omterm{def:g5:series-parallel-circuits:parallel}{cabang} yang lain. Pemasangan seri yang mana pun akan
membiarkan \omterm{ex:g3:first-electric-circuit:switch}{sakelar} lampu baca atau \omterm{def:g3:first-electric-circuit:bulb}{bohlam} tangga yang putus
meredupkan atau menggelapkan lampu besarnya --- tidak termaafkan.
\textbf{2.} Induknya pada jalan bersamanya tepat sesudah banknya;
masing-masing keempat \omterm{ex:g3:first-electric-circuit:switch}{sakelar} \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya} \omterm{def:g4:series-circuits:series}{terangkai seri} dengan tugasnya
sendiri, di dalam \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}.
\textbf{3.} Ya --- menurut hukum paralelnya: sebuah celah pada satu
\omterm{def:g5:series-parallel-circuits:parallel}{cabang} hanya menghentikan jatah \omterm{def:g5:series-parallel-circuits:parallel}{cabang} itu; yang lain terus berarak.
\textbf{4.} Bank, \omterm{ex:g3:first-electric-circuit:switch}{sakelar} induk, lalu sebuah percabangan menjadi empat
\omterm{def:g5:series-parallel-circuits:parallel}{cabang} --- (lampu besar $+$ \omterm{ex:g3:first-electric-circuit:switch}{sakelar}), (serunai kabut $+$ \omterm{ex:g3:first-electric-circuit:switch}{sakelar}),
(lampu baca $+$ \omterm{ex:g3:first-electric-circuit:switch}{sakelar}), (lampu tangga $+$ \omterm{ex:g3:first-electric-circuit:switch}{sakelar}) --- yang bergabung
kembali lalu pulang ke banknya.
\textbf{5.} Ya: tiap \omterm{def:g5:series-parallel-circuits:parallel}{cabang} membentang pada kedua sisi banknya,
sehingga masing-masing menerima dorongan penuh tanpa peduli berapa
banyak tetangga yang minum di sampingnya.
\textbf{6.} Tidak: arus yang menembus lampunya meninggalkannya utuh
--- kursinya beredar tanpa dimakan. Yang diambil lampunya adalah
\omterm{def:g5:everyday-energy:energy}{energi} dari simpanan banknya, yang diantarkan oleh arak-arakan yang
lewat.
\textbf{7.} Empat arak-arakan penuh sekaligus: \omterm{def:g5:everyday-energy:energy}{energi} banknya terkuras
oleh empat tugas dengan cepat. Tagihannya dibayar di ruang \omterm{def:g3:first-electric-circuit:battery}{baterainya}
--- dalam \omterm{def:g5:everyday-energy:energy}{energi} yang tersimpan, tidak pernah dalam arus yang hilang.
\textbf{8.} Keluar dari $+$ banknya, menembus \omterm{ex:g3:first-electric-circuit:switch}{sakelar} induknya, menuju
percabangannya; masuk ke \omterm{def:g5:series-parallel-circuits:parallel}{cabang} lampu bacanya, menembus \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya}
yang tertutup dan lampunya; lewat percabangan tempat ketiga \omterm{def:g5:series-parallel-circuits:parallel}{cabang}
yang lain (yang gelap) bergabung kembali; lalu kembali ke $-$ banknya.
Ia sama sekali menghindari \omterm{def:g5:series-parallel-circuits:parallel}{cabang} lampu besar, serunai kabut, dan
tangganya.
\textbf{9.} Tidak ada yang lain --- hanya \omterm{def:g5:series-parallel-circuits:parallel}{cabang} tangganya yang mati
bersama \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}: itulah kemandirian hukum paralelnya.
\textbf{10.} Sebuah \omterm{def:g6:circuits-and-safety:short}{korsleting} di hulu setiap \omterm{def:g5:series-parallel-circuits:parallel}{cabangnya}.
Arak-arakannya meninggalkan setiap jalan yang bekerja demi jalan
pintas telanjang yang tanpa upaya: lampu dan serunainya kelaparan
sementara kawat pintasnya membawa serbuan yang liar dan memanas ---
setiap tugasnya gagal sekaligus sampai kawatnya disingkirkan.
\textbf{11.} Berilah lampu besarnya lingkarnya sendiri yang mandiri
pada bank kedua: bank dua, \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} sendiri, lampu besarnya ---
tanpa berbagi kawat apa pun dengan panel umumnya. Jalan pintas yang
melintang pada bank satu lalu membungkam segalanya \emph{kecuali}
lampu bagi kapalnya.
\textbf{12.} Misalnya: ``Hukum paralelnya menjaga tiap tugas tetap
berdaulat malam ini. Malam itu membantah, sekali lagi, kepercayaan
bahwa lampu melahap arus --- arak-arakannya pulang utuh bahkan selagi
banknya terkuras. Dan satu kawat telanjang, yang mengorsletingkan
banknya, merenggut \omterm{ex:g2:measuring-time:clock}{jam} terburuk kami: setiap jalan kosong sementara
jalan pintasnya terbakar.''
\end{solution}
