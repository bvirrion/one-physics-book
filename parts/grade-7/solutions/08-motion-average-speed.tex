\section*{Chapter \ref{ch:g7:motion-average-speed} --- Motion Graphs and Average Speed}

\begin{solution}{exo:g7:motion-average-speed:1}
$v = d/t$; \omterm{def:g6:measurement-in-science:measuring}{units} \unit{km/h} and \unit{m/s}. Distance:
$d = v \times t$; time: $t = d/v$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:2}
Ferry: $45 \div 3 = \qty{15}{km/h}$. Hare: $100 \div 8 =
\qty{12.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:3}
$5 \times 3.6 = \qty{18}{km/h}$; \ $72 \div 3.6 = \qty{20}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:4}
Straight line: \omterm{def:g6:describing-motion:uniform}{uniform motion}; steeper: faster; flat: stopped. A
gradually flattening bend tells of slowing down.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:5}
From minute $35$ to $60$ --- the third stretch: it is the least
steep of the rising stretches, read directly from its gentler
slant.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:6}
Uniform --- one straight line. It gains \qty{2}{km} each half
hour: $v = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:7}
Total distance over total time, stops included. Whole hike:
$12 \div 4 = \qty{3}{km/h}$. Walking only: $12 \div 3 =
\qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:8}
Over equal distances, the slow stretch simply lasts longer ---
more of the journey's clock is spent living at the slow pace, so
the slow pace speaks with more of the journey's voice.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:9}
Midpoint guess: \qty{22.5}{km/h}. Honestly: out $30 \div 15 = 2$
\omterm{ex:g2:measuring-time:clock}{hours}, back $30 \div 30 = 1$ hour; total \qty{60}{km} in $3$
\omterm{ex:g2:measuring-time:clock}{hours}: \qty{20}{km/h}. The slow half owned two of the three
\omterm{ex:g2:measuring-time:clock}{hours}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:10}
Graph: straight to (\qty{30}{min}, \qty{20}{km}); flat to
$45$ min; straight and steeper to (\qty{60}{min}, \qty{35}{km}).
Average: \qty{35}{km} in one hour --- \qty{35}{km/h}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:11}
$d = 340 \times 6 = \qty{2040}{m}$ --- about \qty{2}{km}. The
old rule: $6 \div 3 = 2$ kilometres --- agreeing, because
\qty{340}{m/s} means very nearly a kilometre every three
seconds.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:12}
First half: \qty{30}{km} at \qty{30}{km/h} costs exactly $1$
hour. An overall \qty{60}{km/h} over \qty{60}{km} allows a total
of exactly $1$ hour --- already spent to the last second. The
remaining \qty{30}{km} would need to take $0$ \omterm{ex:g2:measuring-time:clock}{hours}: no \omterm{def:g7:motion-average-speed:formula}{speed},
however heroic, makes the average; the budget is gone.
\end{solution}

\begin{solution}{pb:g7:motion-average-speed:1}
\textbf{1.} A brisk ride ($30$ min), a $15$-minute stop
(delivery), a gentler ride ($30$ min), and a second $15$-minute
stop --- two runs, two calls.
\textbf{2.} \qty{9}{km} in half an hour: \qty{18}{km/h}.
\textbf{3.} $15 - 9 = \qty{6}{km}$ in half an hour:
\qty{12}{km/h}.
\textbf{4.} \qty{15}{km} in $1.5$ \omterm{ex:g2:measuring-time:clock}{hours}: \qty{10}{km/h}.
\textbf{5.} $t = 12 \div 24 = 0.5$ \omterm{ex:g2:measuring-time:clock}{hours}: $30$ minutes.
\textbf{6.} $20$ minutes is a third of an hour: $d = 18 \times
\tfrac{1}{3} = \qty{6}{km}$.
\textbf{7.} $25$ minutes is $\tfrac{25}{60}$ \omterm{ex:g2:measuring-time:clock}{hours}; $v = 10
\div \tfrac{25}{60} = \qty{24}{km/h}$.
\textbf{8.} Distances: $12 + 6 + 10 = \qty{28}{km}$; times:
$30 + 20 + 25 = 75$ minutes $= 1.25$ \omterm{ex:g2:measuring-time:clock}{hours}; rolling average:
$28 \div 1.25 = \qty{22.4}{km/h}$.
\textbf{9.} $37 \div 3 \approx \qty{12.3}{km/h}$ --- far below
$20$: no bonus (her stops are honest work, but the rule counts
them).
\textbf{10.} Max: fast half $12 \div 30 = 0.4$ h; slow half
$12 \div 15 = 0.8$ h; total \qty{24}{km} in $1.2$ h: exactly
\qty{20}{km/h} --- not \emph{above} $20$: no bonus.
\textbf{11.} His $22.5$ averaged the two \emph{numbers}, as if
each pace owned half the clock. In truth the slow half owned
$0.8$ of his $1.2$ \omterm{ex:g2:measuring-time:clock}{hours} --- two thirds of the journey's voice
--- and dragged the true average down to $20$.
\textbf{12.} For example: pay the bonus on \emph{rolling}
\omterm{def:g7:motion-average-speed:average}{average speed} --- total distance divided by riding time only,
the flat stretches of the graph excluded. It \omterm{def:g6:measurement-in-science:measuring}{measures} the pace
the wheels actually keep, rewarding fast riding rather than
skipped deliveries.
\textbf{13.} ``A straight stretch is a steady pace; a flat
stretch is a stop with the clock still running; a steeper
stretch is a faster pace --- steepness is \omterm{def:g7:motion-average-speed:formula}{speed}. And never
\omterm{def:g7:motion-average-speed:average}{average speed} numbers: average the journey --- total distance
over total time --- or the slow \omterm{ex:g2:measuring-time:clock}{hours} will make a fool of the
sheet.''
\end{solution}
