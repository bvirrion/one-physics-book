\section*{Capítulo \ref{ch:g7:motion-average-speed} --- Gráficas del movimiento y velocidad media}

\begin{solution}{exo:g7:motion-average-speed:1}
$v = d/t$; \omterm{def:g6:measurement-in-science:measuring}{unidades} \unit{km/h} y \unit{m/s}. Distancia: $d = v
\times t$; tiempo: $t = d/v$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:2}
Transbordador: $45 \div 3 = \qty{15}{km/h}$. Liebre: $100 \div 8 =
\qty{12.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:3}
$5 \times 3.6 = \qty{18}{km/h}$; \ $72 \div 3.6 = \qty{20}{m/s}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:4}
Línea recta: \omterm{def:g6:describing-motion:uniform}{movimiento uniforme}; más empinada: más rápido;
horizontal: parado. Un quiebro que se va aplanando poco a poco cuenta
que el movimiento frena.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:5}
Del minuto $35$ al $60$ --- el tercer tramo: es el menos empinado de
los tramos que suben, y se lee directamente en su inclinación más
suave.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:6}
Uniforme --- una sola línea recta. Gana \qty{2}{km} cada media hora:
$v = \qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:7}
Distancia total entre tiempo total, paradas incluidas. Excursión
entera: $12 \div 4 = \qty{3}{km/h}$. Solo caminando: $12 \div 3 =
\qty{4}{km/h}$.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:8}
A distancias iguales, el tramo lento simplemente dura más --- se
consume más reloj del viaje viviendo al ritmo lento, así que el ritmo
lento habla con más voz del viaje.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:9}
Punto medio previsto: \qty{22.5}{km/h}. Con honradez: ida $30 \div 15
= 2$ \omterm{ex:g2:measuring-time:clock}{horas}, vuelta $30 \div 30 = 1$ hora; total \qty{60}{km} en $3$
\omterm{ex:g2:measuring-time:clock}{horas}: \qty{20}{km/h}. La mitad lenta se quedó con dos de las tres
\omterm{ex:g2:measuring-time:clock}{horas}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:10}
Gráfica: recta hasta (\qty{30}{min}, \qty{20}{km}); horizontal hasta
el minuto $45$; recta y más empinada hasta (\qty{60}{min},
\qty{35}{km}). Media: \qty{35}{km} en una hora --- \qty{35}{km/h}.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:11}
$d = 340 \times 6 = \qty{2040}{m}$ --- unos \qty{2}{km}. La vieja
regla: $6 \div 3 = 2$ kilómetros --- coincide, porque \qty{340}{m/s}
significa casi exactamente un kilómetro cada tres segundos.
\end{solution}

\begin{solution}{exo:g7:motion-average-speed:12}
Primera mitad: \qty{30}{km} a \qty{30}{km/h} cuesta exactamente $1$
hora. Una media global de \qty{60}{km/h} en \qty{60}{km} permite un
total de exactamente $1$ hora --- ya gastada hasta el último segundo.
Los \qty{30}{km} que quedan tendrían que llevar $0$ \omterm{ex:g2:measuring-time:clock}{horas}: ninguna
\omterm{def:g7:motion-average-speed:formula}{velocidad}, por heroica que sea, alcanza la media; el presupuesto se
acabó.
\end{solution}

\begin{solution}{pb:g7:motion-average-speed:1}
\textbf{1.} Un pedaleo vivo ($30$ min), una parada de $15$ minutos
(entrega), un pedaleo más suave ($30$ min) y una segunda parada de
$15$ minutos --- dos carreras, dos visitas.
\textbf{2.} \qty{9}{km} en media hora: \qty{18}{km/h}.
\textbf{3.} $15 - 9 = \qty{6}{km}$ en media hora: \qty{12}{km/h}.
\textbf{4.} \qty{15}{km} en $1.5$ \omterm{ex:g2:measuring-time:clock}{horas}: \qty{10}{km/h}.
\textbf{5.} $t = 12 \div 24 = 0.5$ \omterm{ex:g2:measuring-time:clock}{horas}: $30$ minutos.
\textbf{6.} $20$ minutos son un tercio de hora: $d = 18 \times
\tfrac{1}{3} = \qty{6}{km}$.
\textbf{7.} $25$ minutos son $\tfrac{25}{60}$ \omterm{ex:g2:measuring-time:clock}{horas}; $v = 10 \div
\tfrac{25}{60} = \qty{24}{km/h}$.
\textbf{8.} Distancias: $12 + 6 + 10 = \qty{28}{km}$; tiempos: $30 +
20 + 25 = 75$ minutos $= 1.25$ \omterm{ex:g2:measuring-time:clock}{horas}; media rodando: $28 \div 1.25
= \qty{22.4}{km/h}$.
\textbf{9.} $37 \div 3 \approx \qty{12.3}{km/h}$ --- muy por debajo de
$20$: sin prima (sus paradas son trabajo honrado, pero la regla las
cuenta).
\textbf{10.} Max: mitad rápida $12 \div 30 = 0.4$ h; mitad lenta $12
\div 15 = 0.8$ h; total \qty{24}{km} en $1.2$ h: exactamente
\qty{20}{km/h} --- no \emph{por encima} de $20$: sin prima.
\textbf{11.} Su $22.5$ promedió los dos \emph{números}, como si cada
ritmo se quedara con la mitad del reloj. En realidad la mitad lenta se
quedó con $0.8$ de sus $1.2$ \omterm{ex:g2:measuring-time:clock}{horas} --- dos tercios de la voz del
viaje --- y arrastró la media verdadera hasta $20$.
\textbf{12.} Por ejemplo: pagar la prima según la \omterm{def:g7:motion-average-speed:average}{velocidad media} de
\emph{rodadura} --- distancia total dividida entre el tiempo de
pedaleo solamente, excluidos los tramos horizontales de la gráfica.
Mide el ritmo que llevan de verdad las ruedas, y premia el pedalear
rápido en lugar de las entregas saltadas.
\textbf{13.} ``Un tramo recto es un ritmo constante; un tramo
horizontal es una parada con el reloj en marcha; un tramo más empinado
es un ritmo más rápido --- la pendiente es la \omterm{def:g7:motion-average-speed:formula}{velocidad}. Y nunca
promedies números de \omterm{def:g7:motion-average-speed:formula}{velocidad}: promedia el viaje --- distancia total
entre tiempo total --- o las \omterm{ex:g2:measuring-time:clock}{horas} lentas dejarán la hoja en
ridículo''.
\end{solution}
