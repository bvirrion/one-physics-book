\chapter{Volume, Mass, Density}\label{ch:g7:volume-mass-density}

Last year you unmasked a fraudulent goldsmith by dividing \omterm{def:g3:measuring-mass:units}{grams} by
\omterm{def:g6:mass-and-volume:volume}{cubic centimetres}. This year, your mathematics course has handed
you a new power: letters that stand for numbers. \omterm{def:g7:volume-mass-density:formula}{Density} is the
perfect place to spend it --- one short formula that packs three
recipes, a detective's toolkit, and the whole floating rule into
five symbols.

\section{The formula}

\begin{definition}[Density, by formula]\label{def:g7:volume-mass-density:formula}
The \emph{density}\index{density} $\rho$ (the Greek letter
\emph{rho}) of a substance is its \omterm{def:g3:measuring-mass:mass}{mass} per \omterm{def:g6:measurement-in-science:measuring}{unit} of \omterm{def:g6:mass-and-volume:volume}{volume}: for a
sample of \omterm{def:g3:measuring-mass:mass}{mass} $m$ and \omterm{def:g6:mass-and-volume:volume}{volume} $V$,
\[
\rho = \frac{m}{V}.
\]
With $m$ in \omterm{def:g3:measuring-mass:units}{grams} and $V$ in \omterm{def:g6:mass-and-volume:volume}{cubic centimetres}, $\rho$ comes out
in \emph{\omterm{def:g3:measuring-mass:units}{grams} per \omterm{def:g6:mass-and-volume:volume}{cubic centimetre}} (\unit{g/cm^3}) --- last
year's ``\omterm{def:g3:measuring-mass:mass}{mass} of one \omterm{def:g6:mass-and-volume:volume}{cubic centimetre}'', now wearing its uniform.
Water's density: $\rho = \qty{1.0}{g/cm^3}$. And since a \omterm{def:g6:mass-and-volume:volume}{litre} is
$1000$ \omterm{def:g6:mass-and-volume:volume}{cubic centimetres}, the same number doubles as \omterm{def:g3:measuring-mass:units}{kilograms}
per \omterm{def:g6:mass-and-volume:volume}{litre}: water is \qty{1.0}{kg} per \omterm{def:g6:mass-and-volume:volume}{litre}.
\end{definition}

\begin{example}[Reading the letters]\label{ex:g7:volume-mass-density:reading}
A formula is a sentence in shorthand. $\rho = m/V$ reads: ``to
find the \omterm{def:g7:volume-mass-density:formula}{density}, divide the sample's \omterm{def:g3:measuring-mass:mass}{mass} by its \omterm{def:g6:mass-and-volume:volume}{volume}.'' The
letters are placeholders --- for \emph{this} pebble, $m$ becomes
\qty{75}{g} and $V$ becomes \qty{30}{cm^3}:
\[
\rho = \frac{m}{V} = \frac{75}{30} = \qty{2.5}{g/cm^3}.
\]
Same division as last year --- but the formula remembers the
recipe for every pebble to come.
\end{example}

\begin{proposition}[Three recipes in one formula]\label{prop:g7:volume-mass-density:recipes}
Because \omterm{def:g7:volume-mass-density:formula}{density} ties \omterm{def:g3:measuring-mass:mass}{mass} and \omterm{def:g6:mass-and-volume:volume}{volume} in proportion, one formula
serves three needs:
\begin{enumerate}
  \item \emph{find a \omterm{def:g7:volume-mass-density:formula}{density}}: $\rho = m/V$ --- divide \omterm{def:g3:measuring-mass:mass}{mass} by
        \omterm{def:g6:mass-and-volume:volume}{volume};
  \item \emph{find a \omterm{def:g3:measuring-mass:mass}{mass}}: $m = \rho \times V$ --- each \omterm{def:g6:mass-and-volume:volume}{cubic
        centimetre} carries $\rho$ \omterm{def:g3:measuring-mass:units}{grams}, and there are $V$ of
        them;
  \item \emph{find a \omterm{def:g6:mass-and-volume:volume}{volume}}: $V = m/\rho$ --- share the \omterm{def:g3:measuring-mass:mass}{mass} out
        in helpings of $\rho$ \omterm{def:g3:measuring-mass:units}{grams}; the number of helpings is
        the number of \omterm{def:g6:mass-and-volume:volume}{cubic centimetres}.
\end{enumerate}
\end{proposition}

\begin{example}[The recipes at work]\label{ex:g7:volume-mass-density:atwork}
\emph{\omterm{def:g3:measuring-mass:mass}{Mass} from \omterm{def:g6:mass-and-volume:volume}{volume}}: what does \qty{250}{cm^3} of iron
($\rho = 7.9$) weigh? $m = 7.9 \times 250 = \qty{1975}{g}$ ---
nearly two \omterm{def:g3:measuring-mass:units}{kilograms} in a coffee-mug's bulk.
\emph{\omterm{def:g6:mass-and-volume:volume}{Volume} from \omterm{def:g3:measuring-mass:mass}{mass}}: what room does \qty{540}{g} of oak
($\rho = 0.6$) take? $V = 540 \div 0.6 = \qty{900}{cm^3}$.
\emph{Sanity partner}: iron answers should be heavy-and-small,
wood answers light-and-large --- the \omterm{def:g7:volume-mass-density:formula}{density} table is the judging
step's best friend.
\end{example}

\section{Measuring densities}

\begin{method}[Density of a liquid]\label{met:g7:volume-mass-density:liquid}
\omterm{def:g3:solids-liquids-gases:liquid}{Liquids} will not sit on a balance pan alone --- so borrow the
tare trick:
\begin{enumerate}
  \item place an empty measuring cylinder on the balance and tare
        it to zero;
  \item pour in a convenient \omterm{def:g6:mass-and-volume:volume}{volume} --- say $V = \qty{80}{cm^3}$
        of the mystery \omterm{def:g3:solids-liquids-gases:liquid}{liquid}, reading the \omterm{met:g6:mass-and-volume:cylinder}{meniscus} properly;
  \item the balance now shows the \omterm{def:g3:solids-liquids-gases:liquid}{liquid}'s \omterm{def:g3:measuring-mass:mass}{mass} alone: say
        $m = \qty{68}{g}$;
  \item divide: $\rho = 68 \div 80 = \qty{0.85}{g/cm^3}$ ---
        lighter than water: likely an oil.
\end{enumerate}
\end{method}

\begin{example}[Density of a solid, start to finish]\label{ex:g7:volume-mass-density:solid}
A medal claims to be silver ($\rho = 10.5$). Balance:
$m = \qty{84}{g}$. Cylinder: the water climbs from \qty{60.0}{cm^3} to
\qty{68.0}{cm^3}, so $V = \qty{8.0}{cm^3}$. Then
$\rho = 84 \div 8.0 = \qty{10.5}{g/cm^3}$ --- the signature
matches: silver it may well be. (A cheaper metal dressed in
silver plating would have betrayed itself here --- unless chosen
with cunning, as the weekend problem will show.)
\end{example}

\begin{remark}[Units in formulas]\label{rem:g7:volume-mass-density:units}
A formula is honest only if its \omterm{def:g6:measurement-in-science:measuring}{units} agree. With $m$ in \omterm{def:g3:measuring-mass:units}{grams}
and $V$ in \unit{cm^3}, $\rho$ speaks \unit{g/cm^3}; feed it
\omterm{def:g3:measuring-mass:units}{kilograms} and \omterm{def:g6:mass-and-volume:volume}{litres} and it answers \unit{kg/L} --- happily, the
same number for any substance. But mix \omterm{def:g3:measuring-mass:units}{grams} with \omterm{def:g6:mass-and-volume:volume}{litres} and the
formula, uncomplaining, delivers nonsense. Rule of the
professionals: before computing, parade the \omterm{def:g6:measurement-in-science:measuring}{units}; after
computing, write them into the answer.
\end{remark}

\section{Density thinking}

\begin{example}[Alloys and in-betweens]\label{ex:g7:volume-mass-density:alloys}
Blend two substances and the blend's \omterm{def:g7:volume-mass-density:formula}{density} lands \emph{between}
their signatures --- nearer the more generous ingredient. Bronze
(copper $9.0$ with a little tin) signs near $8.8$; the crown's
gold-and-silver blend signed between $10.5$ and $19.3$; sea water,
salt dissolved in water, edges up to about $1.03$. Between-ness
is itself a clue: a reading of $12$ from a ``pure gold'' bar is a
confession of company.
\end{example}

\begin{example}[Density decides the floating world]\label{ex:g7:volume-mass-density:floating}
The floating rule, now in uniform: an object \omterm{def:g1:floating-and-sinking:words}{floats} in a \omterm{def:g3:solids-liquids-gases:liquid}{liquid}
when $\rho_{\text{object}} < \rho_{\text{liquid}}$. Oak ($0.6$)
on water ($1.0$): \omterm{def:g1:floating-and-sinking:words}{floats}. Ice ($0.9$) in oil ($0.85$): \omterm{def:g1:floating-and-sinking:words}{sinks} ---
check the table, not your instincts. A swimmer ($\approx 1.0$)
in the famous ultra-salty lakes ($\approx 1.2$): \omterm{def:g1:floating-and-sinking:words}{floats} like a
cork, newspaper in hand. One inequality, the whole harbor.
\end{example}

\begin{remark}[Mind the crowd, not the parcel]\label{rem:g7:volume-mass-density:parcel}
$\rho = m/V$ describes \emph{substances}; ships and swollen
life-jackets are \emph{parcels} --- substance plus trapped \omterm{def:g2:air-around-us:air}{air}
--- and it is the parcel's overall $m/V$ that faces the floating
rule. The formula handles both, if you feed it the right $m$ and
$V$: the steel's own \omterm{def:g7:volume-mass-density:formula}{density} for the substance, the whole hull's
\omterm{def:g3:measuring-mass:mass}{mass} over the whole hull's \omterm{def:g6:mass-and-volume:volume}{volume} for the parcel. Most floating
``paradoxes'' are just the two bookkeepings confused.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:1}
Write the \omterm{def:g7:volume-mass-density:formula}{density} formula, name each letter, and give the \omterm{def:g6:measurement-in-science:measuring}{unit}
that goes with \omterm{def:g3:measuring-mass:units}{grams} and \omterm{def:g6:mass-and-volume:volume}{cubic centimetres}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:2}
Compute the \omterm{def:g7:volume-mass-density:formula}{density}: $m = \qty{270}{g}$, $V = \qty{100}{cm^3}$.
Which metal of the skyline chart is this?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:3}
Use the right recipe: the \omterm{def:g3:measuring-mass:mass}{mass} of \qty{40}{cm^3} of copper
($\rho = 9.0$); \ the \omterm{def:g6:mass-and-volume:volume}{volume} of \qty{1930}{g} of gold
($\rho = 19.3$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:4}
In \cref{met:g7:volume-mass-density:liquid}, why is the cylinder
tared first? What two readings then feed the formula?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:5}
A \omterm{def:g3:solids-liquids-gases:liquid}{liquid}: $V = \qty{50}{cm^3}$, $m = \qty{70}{g}$. Its \omterm{def:g7:volume-mass-density:formula}{density}?
Does an ice cube ($0.9$) \omterm{def:g1:floating-and-sinking:words}{float} or \omterm{def:g1:floating-and-sinking:words}{sink} in it?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:6}
Why is water's \omterm{def:g7:volume-mass-density:formula}{density} the same number in \unit{g/cm^3} and in
\omterm{def:g3:measuring-mass:units}{kilograms} per \omterm{def:g6:mass-and-volume:volume}{litre}? What warning does
\cref{rem:g7:volume-mass-density:units} attach to mixing \omterm{def:g3:measuring-mass:units}{grams}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:volume-mass-density:7}
A bracelet marked ``pure gold'' has $m = \qty{58}{g}$ and
$V = \qty{5.0}{cm^3}$. Compute its \omterm{def:g7:volume-mass-density:formula}{density} and give your

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:8}
One cubic \omterm{def:g2:measuring-length:units}{metre} is a cube of $100 \times 100 \times 100$
\omterm{def:g2:measuring-length:units}{centimetres}. How many \unit{cm^3} is that --- and what is the
\omterm{def:g3:measuring-mass:mass}{mass} of a cubic \omterm{def:g2:measuring-length:units}{metre} of water, in \omterm{def:g3:measuring-mass:units}{kilograms}? (The answer
explains why waterbeds worry landlords.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:9}
A ``silver'' trophy: $m = \qty{420}{g}$, water rise from
\qty{200}{cm^3} to \qty{260}{cm^3}. \omterm{def:g7:volume-mass-density:formula}{Density}? Between which two

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:10}

\omterm{def:g3:solids-liquids-gases:liquid}{liquid}'s \omterm{def:g7:volume-mass-density:formula}{density} in \unit{g/cm^3} and propose its identity.

\begin{exercise}[$\star\star$]\label{exo:g7:volume-mass-density:11}
A hollow aluminium buoy has a total \omterm{def:g6:mass-and-volume:volume}{volume} of \qty{4000}{cm^3}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:volume-mass-density:12}

\end{exercise}

\section{Problem: The Scrapyard Detective}

\begin{problem}[{Weekend problem --- an afternoon with the
scrapyard's metal detective; four mystery lots, one impostor
ingot; the limits of the density test}]\label{pb:g7:volume-mass-density:1}
The scrapyard buys metal by what it \emph{is}, not what it looks
like, and the yard's detective works with a balance, a big
graduated vessel, and the skyline table. You are the apprentice.
(Table extract: aluminium $2.7$; iron $7.9$; copper $9.0$; lead
$11.3$; gold $19.3$; tungsten $19.3$.)

\medskip\noindent\textbf{Part I --- Four lots.}
\begin{enumerate}
  \item Lot A, a gray ingot: $m = \qty{5400}{g}$, displacement
        $V = \qty{2000}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Density} and identity?
  \item Lot B, a coil of wire: $m = \qty{1800}{g}$,
        $V = \qty{200}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Density} and identity?
  \item Lot C, a dull heavy plate: $m = \qty{4520}{g}$,
        $V = \qty{400}{cm^3}$. \omterm{def:g7:volume-mass-density:formula}{Density} and identity --- and why
        must lot C be handled with gloves and respect?
  \item Lot D, a sack of mixed pale scrap: $m = \qty{8100}{g}$,
        $V = \qty{1500}{cm^3}$. Show that lot D's \omterm{def:g7:volume-mass-density:formula}{density} lands
        between two table metals, and say what the sack most
        likely contains.
\end{enumerate}

\medskip\noindent\textbf{Part II --- Prices and predictions.}
The yard pays by \omterm{def:g3:measuring-mass:mass}{mass} but plans transport by \omterm{def:g6:mass-and-volume:volume}{volume}.
\begin{enumerate}[resume]
  \item A buyer wants \qty{540}{g} of aluminium cut from lot A.
        What \omterm{def:g6:mass-and-volume:volume}{volume} of ingot is that?
  \item The copper coil of lot B is to be melted into cubes of
        \qty{25}{cm^3} each. What is the \omterm{def:g3:measuring-mass:mass}{mass} of one cube, and
        how many full cubes does the coil yield?
  \item A crate can carry at most \qty{20}{kg}. How many of
        those copper cubes may it legally hold?
  \item The truck's tank-well holds \qty{3000}{cm^3} more of
        iron scrap. What extra \omterm{def:g3:measuring-mass:mass}{mass}, in \omterm{def:g3:measuring-mass:units}{kilograms}, is the
        detective allowed to load?
\end{enumerate}

\medskip\noindent\textbf{Part III --- The impostor.}
A seller arrives with a gleaming ``gold'' ingot:
$m = \qty{3860}{g}$, displacement $V = \qty{200}{cm^3}$.
\begin{enumerate}[resume]
  \item Compute the ingot's \omterm{def:g7:volume-mass-density:formula}{density}. Does it match gold's
        signature?
  \item The detective, unmoved, consults the table's last line
        and sighs. Which cheaper metal wears exactly gold's
        \omterm{def:g7:volume-mass-density:formula}{density} --- and what does this teach about the \omterm{def:g7:volume-mass-density:formula}{density}
        test's limits?
  \item Which of the yard's two instruments has been defeated
        here: the balance, the vessel, both, or neither? Say
        precisely what the \omterm{def:g7:volume-mass-density:formula}{density} test \emph{did} honestly
        establish about the ingot.
  \item Suggest a further physical test from earlier years of
        this course that tells gold from its double without
        harming the ingot. (Their \omterm{def:g4:changes-of-state:melting-point}{melting points} differ
        enormously --- gold near \qty{1064}{\celsius}, the
        double far above every furnace here --- but no yard
        \omterm{def:g2:ice-water-steam:meltfreeze}{melts} a maybe-treasure: find gentler evidence, perhaps
        the \omterm{def:g1:magnets:magnet}{magnet}'s verdict on iron cores, or the ring of a
        struck bar, and defend your choice honestly.)
\end{enumerate}

\medskip\noindent\textbf{Part IV --- The detective's craft.}
\begin{enumerate}[resume]
  \item Write the detective's three-line creed: what \omterm{def:g7:volume-mass-density:formula}{density} can
        prove, what it can only suggest, and what it can never
        do alone.
\end{enumerate}
\end{problem}
