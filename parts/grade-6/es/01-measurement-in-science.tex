\chapter{Medir en ciencia: unidades e instrumentos}\label{ch:g6:measurement-in-science}

Un carpintero, un médico y una astrónoma empiezan su jornada: uno
mide una estantería, otro una fiebre y la tercera los vagabundeos
de Marte. Los tres se fían de la misma maquinaria silenciosa ---
\omterm{def:g6:measurement-in-science:measuring}{unidades} acordadas, instrumentos honrados y lecturas cuidadosas.
Este año la física se vuelve cuantitativa de verdad, así que empezamos
puliendo las herramientas en las que se apoyarán todos los capítulos
siguientes.

\section{Del parecer al número}

Cinco años de experimentos nos han enseñado un ritmo: los ojos y las
manos suponen, los instrumentos deciden. La regla contra el ``parece
más largo''; la \omterm{def:g3:levers-and-scales:scale}{balanza} contra el ``parece más pesado''; el
\omterm{def:g3:temperature-thermometers:temperature}{termómetro} contra el ``se nota frío''. Cada vez, el veredicto del
instrumento es un \emph{número con una \omterm{def:g6:measurement-in-science:measuring}{unidad}} --- y esa pareja
viaja, se compara y la puede comprobar cualquiera, cosa que ninguna
sensación pudo hacer jamás.

\begin{definition}[Medir una magnitud]\label{def:g6:measurement-in-science:measuring}
\emph{Medir}\index{medir} una magnitud --- una longitud, una
\omterm{def:g3:measuring-mass:mass}{masa}, una \omterm{def:g2:measuring-time:duration}{duración}, una \omterm{def:g3:temperature-thermometers:temperature}{temperatura} --- es contar cuántas
veces cabe en ella su \emph{unidad}\index{unidad} acordada, usando un
instrumento construido para el trabajo. El resultado se escribe
siempre como un número \emph{con} su unidad: \qty{28.5}{cm},
\qty{450}{g}, \qty{37.2}{\celsius}. Un número solo es un rumor;
número más unidad es una medida.
\end{definition}

\begin{example}[El precio de una unidad que falta]\label{ex:g6:measurement-in-science:missing}
``Añade 3 de harina''. ¿\omterm{def:g3:measuring-mass:units}{Gramos}? ¿Cucharadas? ¿Tazas? El destino del
bizcocho depende de la palabra que falta. Ha habido desastres de
verdad por \omterm{def:g6:measurement-in-science:measuring}{unidades} olvidadas o confundidas --- ha habido pilotos que
se quedaron sin combustible en pleno vuelo porque se confundieron
\omterm{def:g3:measuring-mass:units}{kilogramos} con libras. Los profesionales escriben la \omterm{def:g6:measurement-in-science:measuring}{unidad}
todas las veces, sin excepción; a partir de este capítulo, nosotros
también.
\end{example}

\section{Las unidades del mundo}

\begin{definition}[El equipo básico del científico]\label{def:g6:measurement-in-science:units}
El mundo se ha puesto de acuerdo en unas \omterm{def:g6:measurement-in-science:measuring}{unidades} comunes, una
familia por magnitud:
\begin{enumerate}
  \item longitud: el \emph{\omterm{def:g2:measuring-length:units}{metro}} (\unit{m}), con sus crías el
        kilómetro ($\qty{1}{km} = \qty{1000}{m}$), el
        \omterm{def:g2:measuring-length:units}{centímetro} ($\qty{100}{cm} = \qty{1}{m}$) y el milímetro
        ($\qty{1000}{mm} = \qty{1}{m}$);
  \item \omterm{def:g3:measuring-mass:mass}{masa}: el \emph{\omterm{def:g3:measuring-mass:units}{kilogramo}} (\unit{kg}), con el \omterm{def:g3:measuring-mass:units}{gramo}
        ($\qty{1}{kg} = \qty{1000}{g}$) y el miligramo
        ($\qty{1000}{mg} = \qty{1}{g}$);
  \item tiempo: el \emph{segundo} (\unit{s}), con el minuto
        ($\qty{1}{min} = \qty{60}{s}$) y la hora
        ($\qty{1}{h} = \qty{60}{min} = \qty{3600}{s}$);
  \item \omterm{def:g3:temperature-thermometers:temperature}{temperatura}: el \omterm{def:g3:temperature-thermometers:temperature}{grado Celsius} (\unit{\celsius});
  \item volumen de \omterm{def:g3:solids-liquids-gases:liquid}{líquidos}: el litro (\unit{L}), que
        mediremos en el capítulo siguiente.
\end{enumerate}
\end{definition}

\begin{example}[Convertir como un científico]\label{ex:g6:measurement-in-science:convert}
Las conversiones son paseos por la escalera decimal.
$\qty{2.35}{m}$ en \omterm{def:g2:measuring-length:units}{centímetros}: se multiplica por $100$ --- la coma
se desliza dos lugares a la derecha: \qty{235}{cm}. $\qty{4500}{g}$ en
\omterm{def:g3:measuring-mass:units}{kilogramos}: se divide entre $1000$: \qty{4.5}{kg}. El tiempo es el
rebelde de la familia --- sus peldaños van de $60$ en $60$ y no de
$10$ en $10$: $\qty{2.5}{min}$ no son $250$ segundos, sino
$2 \times 60 + 30 = \qty{150}{s}$. Respeta al rebelde.
\end{example}

\begin{remark}[¿Quién custodia el metro?]\label{rem:g6:measurement-in-science:guard}
Durante dos siglos, el \omterm{def:g2:measuring-length:units}{metro} del mundo fue una barra de metal
precioso guardada en una cámara acorazada cerca de una gran capital
europea, y el \omterm{def:g3:measuring-mass:units}{kilogramo}, un cilindro reluciente a su lado: las
reglas y las pesas de todos los países eran copias de copias de
aquellos. Hoy los guardias se han jubilado: el \omterm{def:g2:measuring-length:units}{metro} y sus amigos
se definen ahora a partir de hechos inmutables de la propia naturaleza
--- la \omterm{def:g1:light-and-shadows:source}{luz} y los relojes internos de los átomos ---, de modo que
cualquier laboratorio bien equipado de la Tierra (¡o de fuera de ella!)
puede reconstruirlos con exactitud. Las \omterm{def:g6:measurement-in-science:measuring}{unidades} son el acuerdo
pacífico más duradero de la humanidad.
\end{remark}

\section{Usar bien los instrumentos}

\begin{method}[Leer cualquier escala graduada]\label{met:g6:measurement-in-science:read}
Reglas, \omterm{def:g3:temperature-thermometers:temperature}{termómetros}, jarras medidoras, esferas: una sola destreza
las lee todas.
\begin{enumerate}
  \item busca el \emph{valor de la división}: toma dos marcas
        numeradas vecinas, réstalas y divide entre el número de
        pasitos que hay entre ellas --- en una regla numerada cada
        \omterm{def:g2:measuring-length:units}{centímetro} con diez pasitos, cada paso vale
        $1 \div 10 = \qty{0.1}{cm}$, un milímetro;
  \item pon el ojo justo enfrente de la marca (una vista de lado
        desplaza la lectura);
  \item cuenta a partir de la última marca numerada: $7$ y $3$
        pasitos son $7.3$;
  \item escribe el número \emph{con su \omterm{def:g6:measurement-in-science:measuring}{unidad}}.
\end{enumerate}
\end{method}

\begin{method}[Estimar, medir, juzgar]\label{met:g6:measurement-in-science:estimate}
Los profesionales no miden nunca en frío:
\begin{enumerate}
  \item \emph{estima} primero --- ``esta mesa mide unos dos
        \omterm{def:g2:measuring-length:units}{metros}'';
  \item \emph{elige} el instrumento cuyo alcance y cuyos pasos
        encajen con el trabajo (una regla para un sello, una cinta
        métrica para un pasillo, la báscula de cocina para la harina,
        la del baño para las personas);
  \item \emph{mide}, siguiendo las reglas de lectura;
  \item \emph{juzga}: ¿el resultado coincide más o menos con la
        estimación? Una mesa medida en \qty{21.3}{m} no ha crecido
        --- te has equivocado por un factor de diez, y la estimación
        acaba de pillarte.
\end{enumerate}
La estimación es tu red de seguridad: atrapa las comas decimales que
se han deslizado antes de que ellas te atrapen a ti.
\end{method}

\begin{example}[Elegir con cabeza]\label{ex:g6:measurement-in-science:choose}
Para pesar una carta antes de enviarla, la báscula del baño no sirve
de nada --- los \qty{18}{g} de la carta desaparecen entre sus pasos;
la báscula de cocina, que avanza de \omterm{def:g3:measuring-mass:units}{gramo} en \omterm{def:g3:measuring-mass:units}{gramo}, responde a la
perfección. Para cronometrar un esprint de $100$ \omterm{def:g2:measuring-length:units}{metros}, los pasos de
minuto del reloj de cocina son un desastre; un cronómetro cuenta en
centésimas de segundo. Ningún instrumento es ``mejor'': cada uno tiene
su territorio, marcado por su alcance y por su paso más pequeño.
\end{example}

\section{Números honrados}

\begin{remark}[Entre las marcas]\label{rem:g6:measurement-in-science:between}
A las agujas de verdad les encanta caer entre marcas. La honradez
tiene entonces dos niveles: como mínimo, nombrar a las dos vecinas ---
``entre $7.3$ y $7.4$, más cerca de $7.3$''; mejor todavía, estimar
una cifra más allá de las marcas --- ``unos \qty{7.32}{cm}''. Lo que
la honradez prohíbe es inventarse precisión: leer \qty{7.3214}{cm} en
unas marcas de milímetro es ficción con bata de laboratorio. Las
marcas de un instrumento fijan el límite de lo que ese instrumento
puede prometer.
\end{remark}

\begin{example}[Mide dos veces y fíate más]\label{ex:g6:measurement-in-science:repeat}
Tres amigos cronometran la misma bajada de un coche de juguete con el
mismo cronómetro: \qty{4.6}{s}, \qty{4.8}{s}, \qty{4.7}{s}. Nadie ha
metido la pata --- los dedos, sencillamente, arrancan y paran con un
pelo de diferencia. Una dispersión pequeña es la respiración normal de
cualquier medida de verdad. Las costumbres sabias: repetir la medida,
mirar cómo se agrupan las lecturas y desconfiar de todo número
solitario --- sobre todo del propio.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:1}
Completa cada medida con una \omterm{def:g6:measurement-in-science:measuring}{unidad} sensata: una puerta mide $2$
\dots\ de alto; \ una moneda tiene una \omterm{def:g3:measuring-mass:mass}{masa} de $8$ \dots; \ una clase
dura $55$ \dots; \ una fiebre es de $39$ \dots
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:2}
Convierte: \qty{3.2}{m} a \unit{cm}; \ \qty{750}{g} a \unit{kg}; \
\qty{4}{min} a segundos; \ \qty{5600}{m} a \unit{km}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:3}
¿Por qué es el tiempo el rebelde de la familia de las
\omterm{def:g6:measurement-in-science:measuring}{unidades}? Convierte \qty{3.5}{min} a segundos --- y di qué
respuesta equivocada habría dado la escalera decimal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:4}
Una jarra está numerada cada \qty{100}{mL} con cinco pasitos entre
números. ¿Cuánto vale un paso? El zumo está dos pasos por encima de la
marca del $300$ --- ¿cuánto zumo hay?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:5}
¿Qué instrumento para cada trabajo y por qué: pesar una pizca de
levadura; \ cronometrar un recuento de latidos; \ el ancho de esta
página; \ el largo del patio del colegio?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:6}
Nadia mide su lápiz: \qty{17.4}{m}. ¿Qué red de seguridad del
\cref{met:g6:measurement-in-science:estimate} debería haber atrapado
esto y cuál es el valor verdadero más probable?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:7}
La columna de un \omterm{def:g3:temperature-thermometers:temperature}{termómetro} se para entre las marcas del $18$ y del
$19$, más cerca del $19$. Da una manera honrada de comunicarlo --- y
una deshonesta que hay que evitar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:8}
Tres lecturas de cronómetro de una misma bajada: \qty{6.1}{s},
\qty{6.3}{s}, \qty{6.2}{s}. ¿Tiene alguien la culpa? ¿Cuál es la
manera sabia de comunicarlo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:9}
Una receta extranjera pide ``un cuarto de libra de mantequilla''. Tu
báscula de cocina habla en \omterm{def:g3:measuring-mass:units}{gramos}, y una libra entera son unos
\qty{450}{g}. ¿Cuántos \omterm{def:g3:measuring-mass:units}{gramos} pesas? ¿Qué enseña este recado sobre
las \omterm{def:g6:measurement-in-science:measuring}{unidades} compartidas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:10}
El primer \omterm{def:g2:measuring-length:units}{centímetro} de una regla está gastado, de modo que su borde
empieza en la marca del $1$. Un sello colocado desde el borde marca
$3.4$ en su otro extremo. ¿Cuánto mide de ancho el sello en realidad?
¿Qué costumbre derrota a las reglas gastadas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:11}
La clase mide el pasillo dos veces con una cinta de \qty{20}{m}:
\qty{18.25}{m} y \qty{18.35}{m}. La maestra escribe
``\qty{18.3}{m}''. Explica su elección usando la
\cref{rem:g6:measurement-in-science:between} y el
\cref{ex:g6:measurement-in-science:repeat}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:measurement-in-science:12}
Un cuento antiguo: un rey decretó que su propio pie fuera la
\omterm{def:g6:measurement-in-science:measuring}{unidad} de longitud del reino. Enumera dos desastres prácticos que
se siguen de ahí (piensa en los carpinteros de pueblos lejanos y en el
hijo del rey, que crece y hereda) y explica cómo resolvió los dos la
jubilada barra de metal de la
\cref{rem:g6:measurement-in-science:guard} --- y qué problema dejó sin
resolver incluso la barra y que las constantes de la propia naturaleza
arreglaron por fin.
\end{exercise}

\section{Problema: el inspector de instrumentos}

\begin{problem}[{Problema de fin de semana --- un día con la
inspectora de instrumentos del pueblo; básculas, jarras y relojes en
el banquillo; el cuaderno del aprendiz}]\label{pb:g6:measurement-in-science:1}
Todos los años la inspectora del pueblo pone a prueba los instrumentos
de las tiendas y de los colegios. Hoy tú eres el aprendiz, cuaderno en
mano.

\medskip\noindent\textbf{Parte I --- La báscula de la panadería.}
La inspectora lleva pesas certificadas: \qty{500}{g}, \qty{200}{g},
\qty{200}{g}, \qty{100}{g} y \qty{50}{g}.
\begin{enumerate}
  \item Comprueba que la aguja de la báscula vacía está exactamente
        en el cero. ¿Por qué tiene que ser esta la primera prueba de
        todas?
  \item Carga la pesa de \qty{500}{g} y una de \qty{200}{g}. ¿Qué
        debería marcar una báscula honrada?
  \item La aguja marca en cambio \qty{680}{g}. ¿Favorece esta báscula
        al panadero o a los clientes? Explica quién pierde dinero en
        cada venta.
  \item ¿Qué combinaciones de sus pesas le permiten comprobar la
        lectura de \qty{750}{g}? Da una.
  \item La báscula de la panadería avanza de \qty{5}{g} en
        \qty{5}{g}. ¿Podría usarla la inspectora para comprobar un
        sello de \qty{2}{g}? ¿Por qué no?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Las jarras medidoras del
colegio.}
\begin{enumerate}[resume]
  \item Una jarra está numerada cada \qty{200}{mL}, con cuatro
        pasitos entre números. ¿Cuánto vale cada paso?
  \item El agua está un paso por encima de la marca de
        \qty{600}{mL}. El cuaderno de la maestra dice ``unos
        \qty{700}{mL}''. Califica ese informe: ¿honrado, demasiado
        vago o ficción?
  \item Un alumno lee la jarra desde una silla baja, con el ojo muy
        por debajo de la línea del agua, y obtiene un número distinto
        del de la inspectora. ¿Qué regla de lectura se ha incumplido?
  \item La inspectora vierte los \qty{650}{mL} declarados de la jarra
        en una jarra certificada, que marca \qty{640}{mL}. ¿En cuánto
        se equivoca la jarra del colegio y es eso peor o mejor que
        uno de sus propios pasos?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Los relojes y el veredicto del
aprendiz.}
\begin{enumerate}[resume]
  \item El reloj de la estación de tren se compara con la señal horaria de la
        radio: marca las $12$ h $2$ min en el mediodía de la señal.
        ¿Adelanta o atrasa, y cuánto?
  \item Si nadie lo corrige, ¿cuánto adelantará al cabo de una
        semana? (Gana lo mismo cada día.)
  \item Los relojes de tres tiendas marcan las $12$ h $2$ min, las
        $11$ h $58$ min y las $12$ h en punto en el mediodía de la
        señal. La inspectora dice: ``Aquí promediar sería una tontería
        --- hay que arreglar cada reloj''. ¿Qué arreglo necesita cada
        uno?
  \item Anochece. Escribe en tu cuaderno las tres reglas de oro de la
        inspectora --- una sobre el cero, una sobre la lectura y una
        sobre las \omterm{def:g6:measurement-in-science:measuring}{unidades} ---, cada una en una sola frase tuya.
\end{enumerate}
\end{problem}
