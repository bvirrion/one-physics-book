\chapter{Describir el movimiento: trayectoria y velocidad}\label{ch:g6:describing-motion}

Una hoja que cae, un tren de alta \omterm{def:g5:speed-distance-time:speed}{velocidad}, un caballito de tiovivo,
un penalti: el mundo no se está nunca quieto. Para estudiar el
movimiento, la física le hace dos preguntas ordenadas a todo lo que se
mueve: \emph{¿por qué camino?} y \emph{¿a qué ritmo por él?} Camino y
ritmo --- \omterm{def:g6:describing-motion:trajectory}{trayectoria} y \omterm{def:g5:speed-distance-time:speed}{velocidad} --- son las gafas de este
capítulo.

\section{La trayectoria: el camino recorrido}

\begin{definition}[La trayectoria]\label{def:g6:describing-motion:trajectory}
La \emph{trayectoria}\index{trayectoria} de un objeto en movimiento es
la línea que traza su viaje --- el camino que dibuja por el espacio.
Tres formas cubren casi toda la vida diaria:
\emph{rectilínea}\index{trayectoria rectilínea} (el ascensor, la
manzana que cae), \emph{circular}\index{trayectoria circular} (el
caballito del tiovivo, la punta de la aguja de un reloj) y
\emph{curva} de maneras más libres (el arco de una pelota lanzada, el
picado de una golondrina).
\end{definition}

\begin{example}[¿La trayectoria de quién?]\label{ex:g6:describing-motion:whose}
Ten cuidado de nombrar \emph{qué punto} estás siguiendo. El morro del
caballito del tiovivo dibuja un círculo; el centro del tiovivo entero
no dibuja nada --- se queda quieto. El eje de una rueda que rueda se
desliza en línea recta mientras un punto de su llanta traza saltitos
elegantes. Una sola máquina, muchas \omterm{def:g6:describing-motion:trajectory}{trayectorias}: la física sigue un
punto cada vez.
\end{example}

\begin{remark}[¿En movimiento respecto a qué?]\label{rem:g6:describing-motion:relative}
Un pasajero que dormita en el tren no dibuja ninguna
\omterm{def:g6:describing-motion:trajectory}{trayectoria} para el pasajero de enfrente --- y una línea de cien
kilómetros sobre el mapa para una vaca que lo mira desde el prado. Las
dos descripciones son honradas: el movimiento es siempre
\emph{movimiento respecto a algo}, y quien lo describe tiene que decir
respecto a qué. Por ahora comparamos todo, calladamente, con el suelo;
toda la potencia de elegir otros puntos de vista es una historia
célebre que se guarda para el volumen de secundaria.
\end{remark}

\section{¿Uniforme o variado?}

\begin{definition}[Movimiento uniforme y variado]\label{def:g6:describing-motion:uniform}
Un movimiento es \emph{uniforme}\index{movimiento uniforme} cuando
recorre distancias iguales en tiempos iguales --- el paso constante de
quien camina al son de un metrónomo, la marcha de un tren entre
paradas. Es \emph{variado}\index{movimiento variado} cuando el
ritmo cambia: acelerando (el velocista al salir de los tacos),
frenando (el autobús al llegar) o las dos cosas por turnos (el
interminable arrancar y parar del tráfico urbano).
\end{definition}

\begin{proposition}[Leer la película de puntos]\label{prop:g6:describing-motion:dots}
Fotografía un objeto en movimiento a intervalos iguales de tiempo ---
muchas instantáneas rápidas sobre una misma \omterm{prop:g5:mirrors-reflection:image}{imagen} --- y su
movimiento firmará con su propio nombre, en puntos:
\begin{enumerate}
  \item puntos separados por igual: \omterm{def:g6:describing-motion:uniform}{movimiento uniforme};
  \item puntos que se van separando por el camino: acelerando;
  \item puntos que se van apretando: frenando.
\end{enumerate}
Entre cada par de puntos hay tiempos iguales, así que los huecos entre
puntos comparan las distancias --- el diagnóstico entero de un
vistazo.
\end{proposition}

\begin{method}[Hacer una película de puntos]\label{met:g6:describing-motion:film}
Cualquier cámara de teléfono puede producir estos retratos:
\begin{enumerate}
  \item graba el objeto en movimiento --- un amigo pasando en bici,
        una pelota bajando una rampa --- sujetando la cámara quieta;
  \item avanza por el vídeo parándote a intervalos iguales (cada
        medio segundo de reproducción, por ejemplo);
  \item en cada parada, marca la posición del objeto sobre un papel
        de calco puesto en la pantalla, o anótala respecto a los
        postes de una valla del fondo;
  \item pon tus marcas en fila y léelas con la
        \cref{prop:g6:describing-motion:dots}.
\end{enumerate}
Un aviso de los profesionales: sujeta la cámara quieta --- una cámara
que se mueve añade su propio movimiento a la película y enreda quién
tiene qué \omterm{def:g6:describing-motion:trajectory}{trayectoria} (exactamente el aviso de la
\cref{rem:g6:describing-motion:relative}).
\end{method}

\section{Poner números al ritmo}

\begin{example}[La velocidad por proporcionalidad]\label{ex:g6:describing-motion:proportion}
Un tranvía en \omterm{def:g6:describing-motion:uniform}{movimiento uniforme} recorre \qty{9}{km} en $20$
minutos. ¿Qué \omterm{def:g5:speed-distance-time:speed}{velocidad} es esa, en los conocidos kilómetros por
hora? La proporcionalidad responde sin ninguna maquinaria nueva: una
hora son tres raciones de $20$ minutos, así que el tranvía recorre
$3 \times 9 = \qty{27}{km}$ en una hora --- su \omterm{def:g5:speed-distance-time:speed}{velocidad} es de $27$
kilómetros por hora. En una tabla:
\begin{center}
\begin{tabular}{l|cc}
minutos & $20$ & $60$ \\
\hline
kilómetros & $9$ & $27$
\end{tabular}
\end{center}
El \omterm{def:g6:describing-motion:uniform}{movimiento uniforme} \emph{es} proporcionalidad entre la distancia
y el tiempo --- tu curso de matemáticas y tu curso de física acaban de
darse la mano.
\end{example}

\begin{example}[El tablero de ritmos]\label{ex:g6:describing-motion:paces}
Ritmos de cada día, para juzgar cualquier respuesta: pasear, unos $4$
kilómetros por hora; caminar a buen paso, $6$; ir en bici sin
esfuerzo, $15$; un autobús urbano con paradas, $20$; un coche por
carretera abierta, $90$; un tren de alta \omterm{def:g5:speed-distance-time:speed}{velocidad}, hasta $300$.
Cuando un cálculo te entregue un paseante a $40$ kilómetros por hora,
el tablero de ritmos lo devuelve.
\end{example}

\begin{remark}[La velocidad en un instante]\label{rem:g6:describing-motion:instant}
Un \omterm{def:g6:describing-motion:uniform}{movimiento variado} no tiene una única \omterm{def:g5:speed-distance-time:speed}{velocidad} --- el
autobús que frena va rápido, después despacio y después parado. La
aguja del velocímetro de un coche responde a una pregunta más sutil:
\emph{¿a qué \omterm{def:g5:speed-distance-time:speed}{velocidad} ahora mismo?} Definir con precisión esa
\omterm{def:g5:speed-distance-time:speed}{velocidad} instantánea es genuinamente profundo --- necesita unas
matemáticas que conocerás en el último año de secundaria, y abrió uno
de los capítulos más grandiosos de toda la ciencia. Por este año: los
\omterm{def:g6:describing-motion:uniform}{movimientos uniformes} reciben un número honrado; los
\omterm{def:g6:describing-motion:uniform}{movimientos variados} reciben un relato.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:1}
Di la forma de la \omterm{def:g6:describing-motion:trajectory}{trayectoria}: una gota de lluvia bajando por un
cristal quieto; \ la punta del minutero; \ el tiro de un balón de
baloncesto hacia la canasta; \ el eslalon de un esquiador.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:2}
En una bicicleta que rueda, ¿qué \omterm{def:g6:describing-motion:trajectory}{trayectoria} dibuja el eje de la
rueda? ¿Y la válvula de la llanta? (Una palabra y un dibujito para
cada una.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:3}
¿Uniforme o variado: un tren en marcha entre paradas; \ ese mismo
tren arrancando del andén; \ un paracaidista bajando a ritmo
constante; \ un disco que se desliza y frena sobre hielo áspero?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:4}
Una película de puntos enseña huecos de \qty{2}{cm}, \qty{2}{cm},
\qty{2}{cm} y \qty{2}{cm} entre instantáneas. Diagnostica el
movimiento. Otra enseña \qty{1}{cm}, \qty{2}{cm}, \qty{4}{cm},
\qty{7}{cm}: diagnostica esa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:5}
¿Por qué tiene que quedarse quieta la cámara al hacer una película de
puntos? ¿Qué \omterm{def:g1:five-senses:observation}{observación} del capítulo está en juego?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:6}
Un transbordador en \omterm{def:g6:describing-motion:uniform}{movimiento uniforme} recorre \qty{12}{km} en $30$
minutos. Construye la tabla de proporcionalidad y da su
\omterm{def:g5:speed-distance-time:speed}{velocidad} en kilómetros por hora.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:7}
Un ciclista dice mantener $14$ kilómetros por hora. ¿Cuánto recorre en
$30$ minutos? ¿Y en $15$ minutos? (Tablas, no fórmulas.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:8}
Con el tablero de ritmos del \cref{ex:g6:describing-motion:paces},
juzga estas afirmaciones: un paseante que recorre \qty{12}{km} en una
hora; \ un autobús urbano que recorre \qty{10}{km} en $30$ minutos; \
un tren de alta \omterm{def:g5:speed-distance-time:speed}{velocidad} que recorre \qty{75}{km} en $15$ minutos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:9}
El pasajero dormido de la \cref{rem:g6:describing-motion:relative}:
describe su movimiento --- \omterm{def:g6:describing-motion:trajectory}{trayectoria} y ritmo --- primero para el
pasajero de enfrente y después para la vaca del prado. ¿Por qué son
honradas las dos descripciones?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:10}
Una línea de \omterm{def:g2:measuring-length:units}{metro} recorre \qty{9}{km} de punta a punta. El horario
concede $18$ minutos, incluidas seis paradas de un minuto en las
paradas. Durante el tiempo real de marcha, ¿el ritmo medio del
\omterm{def:g2:measuring-length:units}{metro} está por encima o por debajo de $45$ kilómetros por hora? (Busca
primero el tiempo de marcha y usa después una tabla.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:11}
Diseña una película de puntos para zanjar una discusión familiar:
¿corre el perro a ritmo constante cuando va a buscar la pelota o hace
esprint y luego se deja llevar? Enumera tus pasos, qué vas a
\omterm{def:g6:measurement-in-science:measuring}{medir} y qué veredicto daría cada patrón de puntos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:describing-motion:12}
Una pelota rueda hasta el borde de una mesa y cae. Dibuja su
\omterm{def:g6:describing-motion:trajectory}{trayectoria} desde el borde de la mesa hasta el suelo, tal como la
ve alguien de pie en la habitación. Razona después --- ¡otra vez los
puntos de vista! --- qué \omterm{def:g6:describing-motion:trajectory}{trayectoria} vería un observador diminuto
montado en un carrito que rueda por debajo de la pelota que cae,
justo al mismo ritmo de avance que ella. (Uno de los dos ve una
curva; el otro, algo mucho más sencillo. Este enigma vuelve, resuelto,
en el volumen de secundaria.)
\end{exercise}

\section{Problema: la película de puntos de la línea 7}

\begin{problem}[{Problema de fin de semana --- una ingeniera de
transportes lee las películas de puntos de una línea de tranvía;
diagnosticar el trayecto, cronometrar la línea y pillar la mentira del
horario}]\label{pb:g6:describing-motion:1}
Una ingeniera de transportes ha grabado trayectos de tranvía con una
cámara de carretera que marca la posición del tranvía en cada
intervalo (un intervalo $=$ $10$ segundos) y te entrega las películas.

\medskip\noindent\textbf{Parte I --- Leer las películas.}
La película A, entre dos paradas, enseña huecos (en
\omterm{def:g2:measuring-length:units}{metros}): $50, 50, 50, 50, 50$.
\begin{enumerate}
  \item Diagnostica el movimiento de la película A.
  \item ¿Cuánto recorrió el tranvía durante los $50$ segundos de la
        película A?
  \item La película B, saliendo de una parada, enseña huecos
        $10, 25, 40, 50, 50$. Cuenta con palabras la historia del
        trayecto.
  \item En la película B, ¿durante qué intervalos es (casi) uniforme
        el movimiento?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El ritmo verdadero de la línea.}
\begin{enumerate}[resume]
  \item En su tramo uniforme, el tranvía recorre \qty{50}{m} cada
        $10$ segundos. Construye la tabla hasta un minuto: ¿cuánto
        recorre por minuto?
  \item Sigue hasta la hora: ¿cuál es la \omterm{def:g5:speed-distance-time:speed}{velocidad} de marcha del
        tranvía en kilómetros por hora?
  \item La línea entera mide \qty{12}{km}. A \omterm{def:g5:speed-distance-time:speed}{velocidad} de marcha y
        sin paradas, ¿cuántos minutos de punta a punta?
  \item El horario concede $50$ minutos de punta a punta. ¿Cuántos
        minutos pierde la línea de verdad en paradas, semáforos y
        zonas lentas, todo junto?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El anuncio, en el banquillo.}
El cartel de la empresa presume: ``Línea 7 --- de punta a punta de la
ciudad a $18$ kilómetros por hora, ¡el doble del mejor paso de un
caminante!''.
\begin{enumerate}[resume]
  \item Comprueba la aritmética del cartel con el horario:
        \qty{12}{km} en $50$ minutos --- construye la tabla hasta una
        hora. ¿Es honrado ``$18$ kilómetros por hora'' para el viaje
        \emph{entero}?
  \item Un pedante objeta: ``Ningún número de esos aparece jamás en
        el velocímetro --- ¡el tranvía o va a $18$ o está parado a
        $0$!''. Explica qué describe en realidad una
        \omterm{def:g5:speed-distance-time:speed}{velocidad} de viaje entero y por qué sigue siendo el número
        justo para un viaje de muchos ritmos.
  \item La ingeniera propone saltarse dos paradas para ahorrar
        cuatro minutos. ¿Cuál sería el nuevo tiempo de punta a punta
        y --- otra vez con tabla --- la nueva \omterm{def:g5:speed-distance-time:speed}{velocidad} de viaje
        entero en kilómetros por hora? (Redondea con sensatez.)
  \item Escribe el informe de una sola frase de la ingeniera: ¿de qué
        número deberían fiarse los viajeros para hacer planes, de la
        \omterm{def:g5:speed-distance-time:speed}{velocidad} de marcha o de la \omterm{def:g5:speed-distance-time:speed}{velocidad} de viaje entero, y
        por qué?
\end{enumerate}
\end{problem}
