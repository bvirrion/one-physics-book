\chapter{Air dalam Segala Wujudnya}\label{ch:g6:water-states}

Kamu sudah memburu air melewati wujud-wujudnya sejak tahun kedua
sekolahmu: es, air, uap; manusia salju yang \omterm{def:g2:ice-water-steam:meltfreeze}{mencair} dan jendela yang
berkabut. Tahun ini perburuannya berubah menjadi pembukuan. Ketika es
\omterm{def:g2:ice-water-steam:meltfreeze}{mencair}, adakah zat yang hilang? Ketika air \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}, mengapa pipa
pecah? Dengan \omterm{def:g3:levers-and-scales:scale}{neraca} dan gelas ukur di tangan, kita mengaudit ketiga
wujud air.

\section{Ketiga wujud, dengan label yang benar}

\begin{definition}[Enam perubahan wujud]\label{def:g6:water-states:changes}
Di antara ketiga wujudnya, air punya enam pintu, masing-masing dengan
nama ilmiahnya:
\begin{enumerate}
  \item padat $\to$ cair: \emph{peleburan}\index{peleburan}; cair
        $\to$ padat: \emph{pembekuan}\index{pembekuan} (atau
        pemadatan);
  \item cair $\to$ \omterm{def:g3:solids-liquids-gases:gas}{gas}: \emph{penguapan}\index{penguapan} ---
        menderu cepat ketika mendidih, tenang dan sejuk ketika
        berlangsung dari permukaan; \omterm{def:g3:solids-liquids-gases:gas}{gas} $\to$ cair:
        \emph{pengembunan}\index{pengembunan};
  \item padat $\to$ \omterm{def:g3:solids-liquids-gases:gas}{gas} langsung: \emph{sublimasi}\index{sublimasi},
        dan \omterm{def:g3:solids-liquids-gases:gas}{gas} $\to$ padat langsung --- pintu yang lebih langka,
        tetapi nyata: bulu embun beku pada jendela yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}
        tumbuh langsung dari uap yang tak terlihat, dengan
        melewatkan wujud cairnya sama sekali.
\end{enumerate}
\end{definition}

\begin{example}[Pintu di sekelilingmu]\label{ex:g6:water-states:doors}
Embun beku pagi hari yang \omterm{def:g2:ice-water-steam:meltfreeze}{mencair} dari rumput; cucian yang \omterm{def:g2:ice-water-steam:evapcond}{menguap} di
jemuran; \omterm{def:g5:mirrors-reflection:reflection}{cermin} dingin yang mengembunkan uap mandimu; lemari beku yang
membekukan bongkahan es untuk besok; bulu embun beku yang tumbuh
semalaman di kaca jendela. Satu zat, enam pintu, semuanya dalam sehari
kehidupan biasa --- namailah, dan kamu memilikinya.
\end{example}

\section{Audit: adakah zat yang hilang?}

\begin{method}[Menimbang sebuah perubahan wujud]\label{met:g6:water-states:weigh}
Ajukan pertanyaannya kepada \omterm{def:g3:levers-and-scales:scale}{neraca}:
\begin{enumerate}
  \item tuangkan sedikit air ke dalam botol plastik kecil, putar
        tutupnya sampai rapat, keringkan bagian luarnya;
  \item timbanglah: misalkan \qty{368}{g} --- catatlah;
  \item baringkan di lemari beku semalaman: \omterm{def:g6:water-states:changes}{pembekuan};
  \item timbanglah botol yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} itu segera (lapkan dulu embun
        bekunya).
\end{enumerate}
\omterm{def:g3:levers-and-scales:scale}{Neracanya} menjawab: \qty{368}{g}, sampai ke \omterm{def:g3:measuring-mass:units}{gramnya}. Cairkan kembali
di ambang jendela lalu timbang lagi: \qty{368}{g}. Wujudnya berubah
dua kali; zatnya tidak pernah berkedip.
\end{method}

\begin{proposition}[Massa selamat melewati perubahan wujud]\label{prop:g6:water-states:mass}
Perubahan wujud mengubah penampilan sebuah zat, bukan banyaknya: \omterm{def:g3:measuring-mass:mass}{massa}
sebelumnya sama persis dengan \omterm{def:g3:measuring-mass:mass}{massa} sesudahnya. Melebur, \omterm{def:g2:ice-water-steam:meltfreeze}{membeku},
mendidih di dalam panci tertutup, \omterm{def:g2:ice-water-steam:evapcond}{mengembun} --- lewat setiap pintu,
\omterm{def:g3:levers-and-scales:scale}{neracanya} membaca hal yang sama. (Tetapi biarkan tutupnya terbuka, dan
uapnya melayang pergi bersama \omterm{def:g3:measuring-mass:units}{gramnya}: zatnya tetap ada, tetapi
auditmu tidak lagi memuatnya.)
\end{proposition}

\begin{example}[Ke mana gram genangannya pergi]\label{ex:g6:water-states:puddle}
Sebuah genangan \qty{300}{g} mengering. Lenyap? Tidak berbobot? Bukan:
\qty{300}{g} air kini menunggangi \omterm{def:g2:air-around-us:air}{udara} sebagai uap yang tak terlihat,
setiap \omterm{def:g3:measuring-mass:units}{gramnya} utuh --- sebagian akan jatuh sebagai hujan minggu
depan. \omterm{def:g6:water-states:changes}{Penguapan} menipu mata, tidak pernah menipu pembukuan. Garis
kapur pada percobaan genangan masa kecilmu adalah sebuah audit;
sekarang ia punya bilangan.
\end{example}

\section{Audit: apakah ruangnya tetap?}

\begin{proposition}[Air mengembang ketika membeku]\label{prop:g6:water-states:volume}
\omterm{def:g3:measuring-mass:mass}{Massa} selamat melewati pintunya --- \omterm{def:g6:mass-and-volume:volume}{volume} tidak. Air adalah zat yang
aneh: ketika menjadi padat, ia \emph{mengembang}. Satu \omterm{def:g6:mass-and-volume:volume}{liter} air
menjadi kira-kira $1.1$ \omterm{def:g6:mass-and-volume:volume}{liter} es: \omterm{def:g3:measuring-mass:mass}{massa} yang sama, ruang yang
kira-kira sepersepuluh lebih banyak. Kebanyakan zat menyusut ketika
memadat; air justru menggembung --- dan keanehan itu menggambar ulang
bentang alam.
\end{proposition}

\begin{example}[Pipa yang pecah dan gunung yang retak]\label{ex:g6:water-states:burst}
Kalau terlupakan di lemari beku, botol kaca berisi penuh air akan
retak: es yang menggembung memerlukan tambahan sepersepuluh ruangnya
dan mengambilnya, ada kaca atau tidak. \omterm{def:g3:sun-days-seasons:year}{Musim} dingin melakukan hal yang
sama pada perpipaan --- air yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} di dalam pipa memecahkannya
sepasti palu --- dan pada pegunungan: hujan merembes ke celah batu,
\omterm{def:g2:ice-water-steam:meltfreeze}{membeku} pada malam hari, menggembung, lalu memaksa celahnya melebar.
Tahun demi tahun, beku demi beku, penggembungan kecil yang aneh pada
\cref{prop:g6:water-states:volume} membelah bongkah batu dan
meruntuhkan puncak.
\end{example}

\begin{remark}[Mengapa es terapung]\label{rem:g6:water-states:floats}
Keanehan yang sama menjawab pertanyaan yang kamu ajukan bertahun-tahun
lalu waktu mandi. Es menyimpan zat seliter di dalam ruang $1.1$ \omterm{def:g6:mass-and-volume:volume}{liter}:
untuk ruang yang sama, es \emph{lebih ringan} daripada air --- jadi es
\omterm{def:g1:floating-and-sinking:words}{terapung} di atas \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} sendiri. Gunung es menunggangi laut,
kolam \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} dari atas ke bawah (menyelamatkan ikan di bawahnya), dan
bongkahan es di limunmu \omterm{def:g1:floating-and-sinking:words}{terapung} alih-alih \omterm{def:g1:floating-and-sinking:words}{tenggelam}. Sedikit zat yang
\omterm{def:g1:floating-and-sinking:words}{terapung} di atas dirinya sendiri; kehidupan di kolam yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}
bersyukur bahwa air melakukannya.
\end{remark}

\section{Daur air, diaudit}

\begin{example}[Perjalanan besar bolak-balik, dengan kosakatanya]\label{ex:g6:water-states:cycle}
Kisah masa kecil tentang laut, awan, dan hujan kini terbaca seperti
buku besar. Kehangatan Matahari menggerakkan \emph{\omterm{def:g6:water-states:changes}{penguapan}} dari
laut --- berton-ton naik tanpa terlihat. Tinggi dan dingin, uapnya
\emph{\omterm{def:g2:ice-water-steam:evapcond}{mengembun}} menjadi tetes awan; makin dingin lagi, tetesnya
\omterm{def:g2:ice-water-steam:meltfreeze}{membeku} menjadi salju. Salju menumpuk menjadi gletser; gletser
\emph{melebur} menjadi sungai; sungai mengembalikan setiap \omterm{def:g3:measuring-mass:units}{gramnya} ke
laut. Di sepanjang seluruh lingkarnya,
\cref{prop:g6:water-states:mass} tetap berlaku: \omterm{def:g4:solar-system:planet}{planet} ini tidak
bertambah maupun kehilangan setetes pun --- air di gelasmu sudah
mengarungi lingkar ini sejak sebelum zaman dinosaurus.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g6:water-states:1}
Sebutkan keenam pintu pada \cref{def:g6:water-states:changes},
masing-masing dengan satu penampakan sehari-hari.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:2}
Pintu yang mana: embun yang terbentuk pada rumput dingin saat fajar?
\ manusia salju yang menyusut pada hari \qty{-5}{\celsius} yang cerah,
tanpa munculnya genangan? \ bongkahan es yang mengaburkan lemari beku
dengan ``asap''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:3}
Sebuah botol air tertutup rapat \omterm{def:g4:weight-and-mass:weight}{beratnya} \qty{412}{g}. Ia \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}
padat semalaman. Berapa \omterm{def:g4:weight-and-mass:weight}{beratnya} sekarang, dan proposisi mana yang
menjawab?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:4}
Pada \cref{met:g6:water-states:weigh}, mengapa botolnya harus tertutup
--- dan mengapa dikeringkan sebelum tiap penimbangan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:5}
Sepanci air mendidih tanpa tutup selama sepuluh menit dan \omterm{def:g3:measuring-mass:mass}{massanya}
berkurang \qty{200}{g}. Apakah \omterm{def:g3:measuring-mass:mass}{massanya} dimusnahkan? Di mana \omterm{def:g3:measuring-mass:units}{gram} itu
sekarang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:6}
Satu \omterm{def:g6:mass-and-volume:volume}{liter} air masuk ke lemari beku. Kira-kira berapa \omterm{def:g6:mass-and-volume:volume}{volume} es yang
keluar? Pertanyaan yang sama untuk \omterm{def:g3:measuring-mass:mass}{massanya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:7}
Mengapa pipa air pecah pada \omterm{def:g3:sun-days-seasons:year}{musim} dingin yang keras --- dan mengapa
tukang kebun mengosongkan keran luar rumahnya pada \omterm{def:g3:sun-days-seasons:year}{musim} gugur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:8}
Mengapa es \omterm{def:g1:floating-and-sinking:words}{terapung} di atas air? Hubungkan
\cref{prop:g6:water-states:volume} dengan aturan \omterm{def:g1:floating-and-sinking:words}{terapung} --- lebih
ringan atau lebih berat untuk ruang yang sama?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:9}
Ceritakan daur air sebagai buku besar: ikutilah \qty{1}{kg} air laut
mengelilingi lingkar penuhnya, sambil menyebutkan tiap pintu yang
dilewatinya dan menyatakan \omterm{def:g3:measuring-mass:mass}{massanya} pada setiap tahap.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:10}
Sebuah gelas limun diisi sampai ke bibirnya, dengan bongkahan es yang
\omterm{def:g1:floating-and-sinking:words}{terapung}. Seorang tamu khawatir gelasnya akan melimpah ketika esnya
\omterm{def:g2:ice-water-steam:meltfreeze}{mencair}. Pakailah kedua audit itu (\omterm{def:g3:measuring-mass:mass}{massa} tetap, ruang tambahan es
dikembalikan) untuk menenangkan --- atau justru mencemaskan --- tamu
itu, lengkap dengan alasanmu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:11}
Bulu embun beku muncul pada kaca jendela yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} semalaman di
ruangan yang kering, tanpa hujan dan tanpa kaca yang basah menjelang
tidur. Pintu mana yang diambil airnya, dari mana, dan mengapa nama
pintu ini bukan ``\omterm{def:g6:water-states:changes}{pembekuan}''?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:water-states:12}
Kolam \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} dari atas ke bawah. Jelaskan seluruh rantainya: mengapa
es dari air yang paling dingin terbentuk \emph{di permukaan} lalu
tinggal di sana, apa yang dilakukan tutup \omterm{def:g1:floating-and-sinking:words}{terapung} itu terhadap aliran
\omterm{def:g5:heat-and-insulation:heat}{kalor} di bawahnya (gagasan isolasi dari tahun lalu), dan mengapa ikan
di bawahnya berutang \omterm{def:g3:sun-days-seasons:year}{musim} dinginnya kepada penggembungan air yang
aneh itu.
\end{exercise}

\section{Soal: Sekilogram Keliling Dunia}

\begin{problem}[{Soal akhir pekan --- sekilogram air laut dalam
lawatan besarnya; enam pintu, tiga gunung, satu buku besar yang tidak
pernah putus}]\label{pb:g6:water-states:1}
Kita cap sebuah label khayalan pada \qty{1}{kg} air laut lalu
mengikutinya selama setahun.

\medskip\noindent\textbf{Bagian I --- Naik.}
\begin{enumerate}
  \item \omterm{def:g2:moon-in-the-sky:moon}{Bulan} Juni: Matahari menghangatkan permukaan laut dan
        sekilogram kita naik ke \omterm{def:g2:air-around-us:air}{udara}, tanpa terlihat. Sebutkan pintu
        yang diambilnya.
  \item Berapa \omterm{def:g3:measuring-mass:mass}{massa} sekilogram air kita (yang kini tak terlihat)
        itu? Proposisi mana yang menjaminnya?
  \item Pada ketinggian \qty{3000}{m}, \omterm{def:g2:air-around-us:air}{udaranya} dingin: airnya
        menjadi awan berisi tetes. Sebutkan pintu ini.
  \item Makin dingin lagi, tetesnya menjadi kepingan salju. Sebutkan
        pintu itu --- \omterm{def:g6:water-states:changes}{pembekuan} --- lalu nyatakan \omterm{def:g3:temperature-thermometers:temperature}{suhu} ketika tetes
        awan berisi air murni mulai \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Buku besar sang gletser.}
Sekilogram kita mendarat di sebuah gletser lalu dipadatkan menjadi es
gletser.
\begin{enumerate}[resume]
  \item Sebagai air, sekilogram kita memakai \qty{1}{L}. Kira-kira
        berapa \omterm{def:g6:mass-and-volume:volume}{volume} yang dipakainya sebagai es?
  \item Moncong gletsernya melebur pada \omterm{def:g3:sun-days-seasons:year}{musim} panas. Sekilogram kita
        menjadi air lelehan: berapa \omterm{def:g3:measuring-mass:mass}{massanya} sekarang, dan berapa
        \omterm{def:g6:mass-and-volume:volume}{volumenya} (kembali sebagai air cair)?
  \item Dalam perjalanan turun gunung, \qty{250}{g} dari sekilogram
        kita \omterm{def:g2:ice-water-steam:evapcond}{menguap} dari kolam air terjun yang terkena matahari.
        Berapa banyak air berlabel itu yang melanjutkan perjalanan
        sebagai \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} di dalam sungai?
  \item Apakah seperempat yang \omterm{def:g2:ice-water-steam:evapcond}{menguap} itu sudah meninggalkan
        auditnya? Katakan di mana \qty{250}{g}-nya berada, dan pintu
        apa yang paling mungkin diambilnya berikutnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pulang, dan bukunya
diseimbangkan.}
\begin{enumerate}[resume]
  \item \qty{750}{g} milik sungainya mencapai laut pada bulan
        Oktober. \qty{250}{g} yang \omterm{def:g2:ice-water-steam:evapcond}{menguap} tadi jatuh sebagai hujan
        di pesisir pada bulan Agustus lalu mengalir ke laut pada
        bulan September. Jumlahkan berapa banyak sekilogram kita yang
        diterima kembali oleh lautnya.
  \item Di sepanjang seluruh lawatannya, sekilogram kita tidak pernah
        sekali pun kehilangan \omterm{def:g4:weight-and-mass:weight}{beratnya} --- tetapi ia \emph{tak
        terlihat} dua kali. Pada tahap yang mana, dan dalam wujud
        apa?
  \item Seorang teman sekelas membantah: ``Salju itu gembur dan
        ringan --- pasti sekilogram itu \omterm{def:g4:weight-and-mass:weight}{beratnya} berkurang ketika
        menjadi salju.'' Uraikan kekeliruannya dengan sepasang kata
        yang tepat dari tahun ini.
  \item Tulislah lawatan itu sebagai rantai pintu, berurutan, dengan
        hanya memakai keenam nama yang benar pada
        \cref{def:g6:water-states:changes}.
\end{enumerate}
\end{problem}
