\chapter{Sistem Matahari--Bumi--Bulan}\label{ch:g6:sun-earth-moon}

Tiga dunia menjalankan langit kita: Matahari yang memberi siang, Bumi
yang membawa kita, \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang menemani kita. Kamu sudah mengenal
masing-masingnya; sekarang kita memetakan \emph{sistemnya} --- siapa
mengelilingi siapa, secepat apa, sejauh apa --- lalu melunasi utang
lama: rahasia \omterm{def:g3:sun-days-seasons:year}{musim} yang dijanjikan itu ada di bab ini.

\section{Tiga gerak, tiga jam}

\begin{definition}[Sumbu dan orbit]\label{def:g6:sun-earth-moon:axisorbit}
\emph{Sumbu}\index{sumbu} Bumi adalah garis khayal yang melewati kedua
kutubnya, yang menjadi poros putarannya, seperti tangkai sebuah
gasing. Sebuah \emph{orbit}\index{orbit} adalah jalan tertutup yang
ditempuh sebuah benda langit mengelilingi benda langit yang lain ---
nyaris berupa lingkaran bagi Bumi mengelilingi Matahari, dan bagi
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} mengelilingi Bumi.
\end{definition}

\begin{proposition}[Ketiga geraknya]\label{prop:g6:sun-earth-moon:motions}
Seluruh kalender bergantung pada tiga gerak yang ajek:
\begin{enumerate}
  \item Bumi \emph{berputar} pada \omterm{def:g6:sun-earth-moon:axisorbit}{sumbunya}: satu putaran dalam $24$
        \omterm{ex:g2:measuring-time:clock}{jam} --- itulah hari, yang mengarak Matahari dan \omterm{def:g4:solar-system:star}{bintang}
        melintasi langit kita;
  \item Bumi \emph{mengorbit} Matahari: satu putaran dalam kira-kira
        $365$ hari seperempat --- itulah tahun, yang menggeser busur
        Matahari naik dan turun sepanjang \omterm{def:g3:sun-days-seasons:year}{musim};
  \item \omterm{def:g2:moon-in-the-sky:moon}{Bulan} \emph{mengorbit} Bumi: satu putaran dalam kira-kira
        sebulan --- dan, selagi separuhnya yang tersinari matahari
        menghadap kita bergantian, mengarak fasenya sepanjang buku
        harian Bulanmu yang lama.
\end{enumerate}
Tiga gerak, tiga \omterm{ex:g2:measuring-time:clock}{jam}: hari, bulan, tahun --- penunjuk waktu tertua
umat manusia, yang semuanya masih berjalan.
\end{proposition}

\begin{example}[Seperempat hari dan tahun kabisat]\label{ex:g6:sun-earth-moon:leap}
Putaran Bumi memakan $365$ hari \emph{dan kira-kira seperempat} ---
alam tidak berutang kesetiaan kepada bilangan bulat. Kalender melunasi
utangnya dengan menabung seperempat itu: empat tahun berisi
seperempat-hari menjadi satu hari penuh, yang ditambahkan ke kalender
setiap tahun keempat --- itulah \emph{tahun kabisat} yang berumur
$366$ hari, dengan hari tambahannya di akhir Februari. Lalaikan
pembukuan ini selama seabad dan kalendernya melenceng
berminggu-minggu dari \omterm{def:g3:sun-days-seasons:year}{musimnya} --- seperti yang dahulu ditemukan
kalender yang lebih tua, dengan menyakitkan.
\end{example}

\begin{example}[Sistemnya, menurut skala]\label{ex:g6:sun-earth-moon:scale}
Bilangan untuk buku catatan: garis tengah Bumi kira-kira
\qty{12800}{km}; \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, kira-kira seperempatnya, mengorbit pada jarak
kira-kira \qty{384000}{km} --- tiga puluh Bumi yang dijajarkan
berdampingan. Garis tengah Matahari kira-kira $109$ Bumi dan ia duduk
$150$ juta kilometer jauhnya. Susutkan Bumi menjadi kelereng
sesentimeter dan modelnya berkata: \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, sebutir merica $30$
\omterm{def:g2:measuring-length:units}{sentimeter} jauhnya; Matahari, sebuah bola selebar lebih dari satu
\omterm{def:g2:measuring-length:units}{meter}, yang berdiri lebih dari seratus \omterm{def:g2:measuring-length:units}{meter} di ujung jalan. Seluruh
lingkungan kita sebagian besar kekosongan --- seperti sudah
dibisikkan model berskala \omterm{def:g4:solar-system:family}{Tata Surya} di taman itu.
\end{example}

\section{Utang yang dilunasi: mengapa ada musim}

\begin{proposition}[Kemiringan sumbulah yang membuat musim]\label{prop:g6:sun-earth-moon:tilt}
\omterm{def:g6:sun-earth-moon:axisorbit}{Sumbu} Bumi tidak tegak lurus terhadap \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya}: ia condong, kira-kira
seperempat sudut siku-siku --- dan ia tetap condong \emph{ke arah yang
sama} di sepanjang putaran tahunnya, seperti \omterm{def:g4:magnets-and-compass:compass}{kompas} \omterm{def:g4:solar-system:star}{bintang} yang
setia. Jadi selama separuh putarannya, belahan dunia kita condong
\emph{ke arah} Matahari: Matahari menunggangi jalan yang tinggi,
harinya berjalan panjang --- itulah \omterm{def:g3:sun-days-seasons:year}{musim} panas. Setengah putaran
kemudian, kecondongan yang sama dan tidak berubah itu menunjuk kita
\emph{menjauh}: matahari rendah, hari pendek --- \omterm{def:g3:sun-days-seasons:year}{musim} dingin. \omterm{def:g3:sun-days-seasons:year}{Musim}
bukanlah Matahari yang berubah, bukan pula jaraknya: \omterm{def:g3:sun-days-seasons:year}{musim} adalah
\omterm{def:g4:solar-system:planet}{planet} yang miring yang membawa kecondongannya mengelilingi \omterm{def:g4:solar-system:star}{bintangnya}.
\end{proposition}

\begin{method}[Bola dunia dan lampu]\label{met:g6:sun-earth-moon:globe}
Saksikan seluruh mesinnya di atas meja:
\begin{enumerate}
  \item letakkan sebuah lampu di tengah meja sebagai Matahari, lalu
        bawalah sebuah bola dunia yang miring (atau sebuah bola pada
        tusuk sate yang menyerong) pelan-pelan mengelilinginya,
        \emph{sambil menjaga tusuknya tetap menunjuk ke sudut
        ruangan yang sama} sepanjang jalan;
  \item pada tiap seperempat putarannya, berhentilah lalu lihatlah
        garis lintang tempat tinggalmu: amati seberapa banyak
        lingkaran putaran hariannya yang berada di dalam \omterm{def:g1:light-and-shadows:source}{cahaya};
  \item di satu sisi lampunya, belahan bumimu berjemur sambil
        condong masuk --- busur tersinari yang panjang, lampu yang
        tinggi: \omterm{def:g3:sun-days-seasons:year}{musim} panas; di seberangnya, condong keluar ---
        busur yang pendek, lampu yang rendah: \omterm{def:g3:sun-days-seasons:year}{musim} dingin;
  \item pada kedua perhentian di antaranya, tidak ada kutub yang
        condong masuk: siang dan malam berbagi sama rata --- \omterm{def:g3:sun-days-seasons:year}{musim}
        semi dan \omterm{def:g3:sun-days-seasons:year}{musim} gugur.
\end{enumerate}
Satu disiplin yang membuatnya berhasil: jangan pernah biarkan tusuknya
berayun --- kesetiaan \omterm{def:g6:sun-earth-moon:axisorbit}{sumbunya} \emph{itulah} \omterm{def:g3:sun-days-seasons:year}{musimnya}.
\end{method}

\begin{example}[Petunjuk lama, terperhitungkan]\label{ex:g6:sun-earth-moon:clues}
Setiap pengamatan pada tahun \omterm{ex:g2:measuring-time:clock}{jam} mataharimu kini melapor kepada
kemiringan itu. Busur tinggi \omterm{def:g3:sun-days-seasons:year}{musim} panas: belahan bumi kita condong
masuk, sehingga Matahari berdiri lebih tinggi pada tengah hari. Busur
pendek \omterm{def:g3:sun-days-seasons:year}{musim} dingin dan \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} yang panjang: karena condong
keluar, Matahari menyerempet rendah. \omterm{def:g3:sun-days-seasons:year}{Musim} yang berlawanan di seberang
khatulistiwa --- liburan pantai bulan Desember di selatan sementara
utara menyekop salju: ketika satu belahan condong masuk, belahan yang
lain pasti condong keluar. Bahkan hari yang sama panjang pada \omterm{def:g3:sun-days-seasons:year}{musim}
semi dan \omterm{def:g3:sun-days-seasons:year}{musim} gugur pun keluar dari jalan-jalan lampu-dan-bola dunia
itu. Satu kecondongan seperempat sudut siku-siku menjalankan seluruh
pertunjukan.
\end{example}

\begin{remark}[Bukan jarak --- dan buktinya]\label{rem:g6:sun-earth-moon:notdistance}
\omterm{def:g6:sun-earth-moon:axisorbit}{Orbit} Bumi begitu nyaris melingkar sehingga jaraknya ke Matahari
hampir tidak berubah --- dan, menyenangkan sekali, kita justru
menunggangi jarak yang sedikit \emph{lebih dekat} ke Matahari pada
\omterm{def:g3:sun-days-seasons:year}{musim} dingin belahan utara. \omterm{def:g3:sun-days-seasons:year}{Musim} peduli pada \emph{kemiringan}
\omterm{def:g1:light-and-shadows:source}{cahayanya}, bukan pada panjang perjalanannya: ketika condong masuk,
\omterm{def:g1:light-and-shadows:source}{cahaya} matahari menumbuk curam dan terpusat; ketika condong keluar,
\omterm{def:g1:light-and-shadows:source}{cahaya} yang sama tersebar tipis di atas tanah. Senter yang dipegang
tegak lurus dibandingkan dengan yang menyerong pada sebuah dinding
menunjukkan bedanya dalam satu sekon --- \omterm{def:g1:light-and-shadows:source}{cahaya} yang curam menggigit,
\omterm{def:g1:light-and-shadows:source}{cahaya} yang menyerempet nyaris tidak menghangatkan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:1}
Sebutkan ketiga gerak sistem itu dan \omterm{ex:g2:measuring-time:clock}{jam} yang diputar
masing-masingnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:2}
Apa itu \omterm{def:g6:sun-earth-moon:axisorbit}{sumbu} Bumi? Kenyataan seperempat-sudut-siku-siku apa
tentangnya, ditambah kesetiaan yang mana, yang membuat \omterm{def:g3:sun-days-seasons:year}{musim}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:3}
Mengapa kalender menambahkan satu hari setiap tahun keempat? Apa yang
akan tergelincir kalau ia tidak pernah melakukannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:4}
Kira-kira berapa Bumi, yang dijajarkan berdampingan, mencapai \omterm{def:g2:moon-in-the-sky:moon}{Bulan}?
Berapa Bumi yang membentang selebar wajah Matahari?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:5}
Pada jalan-jalan bola dunia dan lampu, mengapa tusuknya harus terus
menunjuk ke sudut ruangan yang sama? Apa yang rusak kalau ia berayun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:6}
Ketika di sini \omterm{def:g3:sun-days-seasons:year}{musim} panas, \omterm{def:g3:sun-days-seasons:year}{musim} apa yang sedang berlangsung bagi
belahan bumi di seberang khatulistiwa, dan mengapa pasti begitu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:7}
Pakailah gambaran senter-pada-dinding di
\cref{rem:g6:sun-earth-moon:notdistance} untuk menjelaskan mengapa
condong ke arah Matahari lebih menghangatkan kita.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:8}
Seorang kawan lama kembali: mengapa \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menampilkan fase sepanjang
bulan? Gerak mana dari ketiganya yang menjalankan pertunjukan itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:9}
Seorang paman yang keras kepala bersikeras: ``\omterm{def:g3:sun-days-seasons:year}{Musim} dingin adalah
ketika Bumi paling jauh dari Matahari.'' Kerahkan dua kenyataan dari
bab ini yang bersama-sama mengalahkannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:10}
Andaikan \omterm{def:g6:sun-earth-moon:axisorbit}{sumbu} Bumi berdiri tegak lurus sempurna terhadap \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} ---
tanpa kecondongan sama sekali. Gambarkan satu tahunnya: apa yang
terjadi pada \omterm{def:g3:sun-days-seasons:year}{musim}, pada tinggi Matahari tengah hari, pada panjang
harinya? (Jalankan bola dunia yang tegak itu mengelilingi lampunya di

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:11}
Satu putaran \omterm{def:g2:moon-in-the-sky:moon}{Bulan} memakan kira-kira $27$ hari, namun dari purnama ke

lagi yang bergerak selagi \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menempuh putarannya? (Sketsalah Bumi
yang maju di sepanjang \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} selama satu putaran \omterm{def:g2:moon-in-the-sky:moon}{Bulan}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:sun-earth-moon:12}
Pada pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} panas, daerah di dekat Kutub Utara menjaga
Matahari tetap di atas cakrawala pada tengah malam --- ``matahari
tengah malam'' yang termasyhur --- dan pada pertengahan \omterm{def:g3:sun-days-seasons:year}{musim} dingin
mereka menunggu berminggu-minggu untuk \omterm{def:g1:day-and-night:daynight}{matahari terbit}. Jelaskan
kedua keajaiban itu dengan lingkaran putaran yang miring pada
\cref{met:g6:sun-earth-moon:globe}: apa yang dilakukan kecondongannya
terhadap lingkaran harian sebuah kota kutub pada bulan Juni, dan pada
bulan Desember?
\end{exercise}

\section{Soal: Model di Lapangan Olahraga}

\begin{problem}[{Soal akhir pekan --- kelas membangun sistem
Matahari--Bumi--Bulan di lapangan olahraga; kelereng, merica, dan
jalan kaki yang sangat panjang; kemiringan sumbu ikut masuk
model}]\label{pb:g6:sun-earth-moon:1}
Kelas itu membangun sistemnya menurut skala: Bumi adalah kelereng
selebar \qty{1}{cm}. Bilangan sungguhan yang berguna: garis tengah
Bumi \qty{12800}{km}; garis tengah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} \qty{3200}{km}, mengorbit
pada jarak \qty{384000}{km}; garis tengah Matahari $109$ Bumi, pada
jarak $150$ juta kilometer.

\medskip\noindent\textbf{Bagian I --- Memilih pemerannya.}
\begin{enumerate}
  \item Pada modelnya, \qty{1}{cm} mewakili berapa kilometer?
  \item Lebar \omterm{def:g2:moon-in-the-sky:moon}{Bulan} kira-kira seperempat lebar Bumi. Selebar apa
        \omterm{def:g2:moon-in-the-sky:moon}{Bulan} modelnya --- dan benda dapur apa yang memerankannya
        dengan baik?
  \item Jarak \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang sebenarnya adalah $30$ lebar Bumi. Di mana
        \omterm{def:g2:moon-in-the-sky:moon}{Bulan} modelnya berdiri dari kelerengnya?
  \item Matahari modelnya selebar $109$ kelereng. Selebar apa itu,
        dalam \omterm{def:g2:measuring-length:units}{sentimeter} --- dan benda apa di halaman bermain yang
        kira-kira sebesar itu?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Jalan kaki yang panjang.}
\begin{enumerate}[resume]
  \item Jarak Mataharinya $150$ juta km. Pada modelnya --- bagilah
        dengan jawaban pertanyaan 1 --- berapa \omterm{def:g2:measuring-length:units}{sentimeter} itu, dan
        berapa \omterm{def:g2:measuring-length:units}{meter}?
  \item Muatkah modelnya di lapangan olahraga \qty{100}{m}? Di mana
        bola Mataharinya harus berdiri?
  \item Seorang murid berjalan menempuh bentangan Bumi--Matahari
        dengan $1$ \omterm{def:g2:measuring-length:units}{meter} tiap sekon. Kira-kira berapa menit jalan
        kakinya?
  \item Sambil berdiri di kelerengnya dan menoleh ke bola Matahari
        selebar semeter yang berjarak $117$ \omterm{def:g2:measuring-length:units}{meter}, kelasnya terkejut:
        bola itu tampak kira-kira sebesar butiran \omterm{def:g2:moon-in-the-sky:moon}{Bulan} seperempat
        \omterm{def:g2:measuring-length:units}{sentimeter} yang berjarak $30$ \omterm{def:g2:measuring-length:units}{sentimeter} dari kelerengnya.
        Periksalah kedua perbandingannya --- lalu hubungkan kebetulan
        ini dengan sesuatu yang terjadi, jarang dan mengagumkan, di
        langit yang sesungguhnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Membuatnya berjalan.}
\begin{enumerate}[resume]
  \item Untuk menghidupkan modelnya dengan jujur, berapa lama
        kelerengnya harus mengelilingi bola Mataharinya, dan
        mericanya mengelilingi kelerengnya, kalau satu hari yang
        sesungguhnya diperankan oleh satu sekon? (Hitunglah secara
        kasar: setahun sebagai $365$ hari, satu putaran \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
        sebagai $27$ hari.)
  \item Kelerengnya juga harus berputar. Pada satu hari sungguhan
        tiap sekon, secepat apa itu pada modelnya --- dan mengapa
        tidak seorang pun akan melihatnya pada kelereng \qty{1}{cm}?
  \item Sebatang tusuk sate yang menembus kelerengnya memerankan
        \omterm{def:g6:sun-earth-moon:axisorbit}{sumbunya}. Pada kemiringan berapa, dan menunjuk ke mana,
        tusuk itu harus dibawa mengelilingi seluruh putarannya?
  \item Tulislah penyangkalan yang jujur untuk modelnya: sebutkan
        satu hal yang ditunjukkan model lapangan olahraga itu dengan
        benar, dan satu hal yang tidak dapat ditunjukkan sekaligus
        oleh model seukuran lapangan mana pun (ukuran dan jarak
        bersama-sama? gerak? \omterm{def:g1:light-and-shadows:source}{cahaya}?). Berikan alasan dengan
        bilanganmu.
\end{enumerate}
\end{problem}
