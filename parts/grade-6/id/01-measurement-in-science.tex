\chapter{Pengukuran dalam Sains: Satuan dan Alat Ukur}\label{ch:g6:measurement-in-science}

Seorang tukang kayu, seorang dokter, dan seorang astronom memulai hari
kerjanya: yang satu \omterm{def:g6:measurement-in-science:measuring}{mengukur} sebuah rak, yang lain demam, yang lain
lagi pengembaraan Mars. Ketiganya memercayai mesin yang sama dan
tenang --- \omterm{def:g6:measurement-in-science:measuring}{satuan} yang disepakati, alat yang jujur, pembacaan yang
cermat. Tahun ini fisika benar-benar berbelok menjadi kuantitatif,
jadi kita mulai dengan mengasah alat yang akan menjadi sandaran setiap
bab berikutnya.

\section{Dari tampaknya menjadi bilangan}

Lima tahun percobaan sudah mengajarkan kita sebuah irama: mata dan
tangan menebak, alat memutuskan. Penggaris melawan
tampak-lebih-panjang; \omterm{def:g3:levers-and-scales:scale}{neraca} melawan tampak-lebih-berat; \omterm{def:g3:temperature-thermometers:temperature}{termometer}
melawan terasa-dingin. Setiap kali, keputusan alatnya adalah
\emph{bilangan dengan \omterm{def:g6:measurement-in-science:measuring}{satuan}} --- dan pasangan itu dapat bepergian,
dibandingkan, dan diperiksa siapa pun, hal yang tidak pernah dapat
dilakukan sebuah perasaan.

\begin{definition}[Mengukur sebuah besaran]\label{def:g6:measurement-in-science:measuring}
Untuk \emph{mengukur}\index{mengukur} sebuah besaran --- panjang,
\omterm{def:g3:measuring-mass:mass}{massa}, lamanya waktu, \omterm{def:g3:temperature-thermometers:temperature}{suhu} --- kita menghitung berapa kali
\emph{satuan}\index{satuan} yang disepakati untuknya termuat di
dalamnya, dengan alat yang dibuat untuk pekerjaan itu. Hasilnya selalu
ditulis sebagai bilangan \emph{beserta} satuannya: \qty{28.5}{cm},
\qty{450}{g}, \qty{37.2}{\celsius}. Bilangan sendirian hanyalah
desas-desus; bilangan beserta satuan adalah pengukuran.
\end{definition}

\begin{example}[Harga satuan yang hilang]\label{ex:g6:measurement-in-science:missing}
``Tambahkan 3 tepung.'' \omterm{def:g3:measuring-mass:units}{Gram}? Sendok? Cangkir? Nasib kuenya
bergantung pada kata yang hilang itu. Bencana yang sungguhan pernah
menyusul \omterm{def:g6:measurement-in-science:measuring}{satuan} yang hilang atau tertukar --- pilot pernah kehabisan
bahan bakar di tengah penerbangan karena \omterm{def:g3:measuring-mass:units}{kilogram} tertukar dengan pon.
Para profesional menulis \omterm{def:g6:measurement-in-science:measuring}{satuannya} setiap kali, tanpa kecuali; mulai
bab ini, kita pun begitu.
\end{example}

\section{Satuan milik dunia}

\begin{definition}[Perlengkapan awal seorang ilmuwan]\label{def:g6:measurement-in-science:units}
Dunia sudah menyepakati \omterm{def:g6:measurement-in-science:measuring}{satuan} bersama, satu keluarga untuk tiap
besaran:
\begin{enumerate}
  \item panjang: \emph{\omterm{def:g2:measuring-length:units}{meter}} (\unit{m}), dengan keturunannya
        kilometer ($\qty{1}{km} = \qty{1000}{m}$), \omterm{def:g2:measuring-length:units}{sentimeter}
        ($\qty{100}{cm} = \qty{1}{m}$), dan milimeter
        ($\qty{1000}{mm} = \qty{1}{m}$);
  \item \omterm{def:g3:measuring-mass:mass}{massa}: \emph{\omterm{def:g3:measuring-mass:units}{kilogram}} (\unit{kg}), dengan \omterm{def:g3:measuring-mass:units}{gram}
        ($\qty{1}{kg} = \qty{1000}{g}$) dan miligram
        ($\qty{1000}{mg} = \qty{1}{g}$);
  \item waktu: \emph{sekon} (\unit{s}), dengan menit
        ($\qty{1}{min} = \qty{60}{s}$) dan \omterm{ex:g2:measuring-time:clock}{jam}
        ($\qty{1}{h} = \qty{60}{min} = \qty{3600}{s}$);
  \item \omterm{def:g3:temperature-thermometers:temperature}{suhu}: \omterm{def:g3:temperature-thermometers:temperature}{derajat Celsius} (\unit{\celsius});
  \item volume \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}: liter (\unit{L}), yang kita takar pada bab
        berikutnya.
\end{enumerate}
\end{definition}

\begin{example}[Mengubah satuan seperti ilmuwan]\label{ex:g6:measurement-in-science:convert}
Pengubahan \omterm{def:g6:measurement-in-science:measuring}{satuan} adalah tumpangan di tangga desimal.
$\qty{2.35}{m}$ dalam \omterm{def:g2:measuring-length:units}{sentimeter}: kalikan dengan $100$ --- titiknya
meluncur dua tempat ke kanan: \qty{235}{cm}. $\qty{4500}{g}$ dalam
\omterm{def:g3:measuring-mass:units}{kilogram}: bagi dengan $1000$: \qty{4.5}{kg}. Waktu adalah pembangkang
keluarga itu --- undakannya $60$, bukan $10$: $\qty{2.5}{min}$ bukan
$250$ sekon melainkan $2 \times 60 + 30 = \qty{150}{s}$. Hormatilah
sang pembangkang.
\end{example}

\begin{remark}[Siapa yang menjaga meter?]\label{rem:g6:measurement-in-science:guard}
Selama dua abad, \omterm{def:g2:measuring-length:units}{meter} milik dunia adalah sebatang logam berharga di
dalam brankas terjaga di dekat sebuah ibu kota besar Eropa, dan
\omterm{def:g3:measuring-mass:units}{kilogramnya} sebuah tabung berkilau di sebelahnya: penggaris dan anak
timbangan setiap bangsa adalah salinan dari salinan benda itu. Hari
ini para penjaganya sudah dipensiunkan: \omterm{def:g2:measuring-length:units}{meter} dan kawan-kawannya kini
ditetapkan dari kenyataan alam yang tidak berubah --- \omterm{def:g1:light-and-shadows:source}{cahaya} dan \omterm{ex:g2:measuring-time:clock}{jam}
di dalam atom --- sehingga laboratorium mana pun di Bumi (atau di
luarnya!) yang berperalatan baik dapat membangunnya kembali dengan
tepat. \omterm{def:g6:measurement-in-science:measuring}{Satuan} adalah kesepakatan damai umat manusia yang paling lama
bertahan.
\end{remark}

\section{Memakai alat dengan baik}

\begin{method}[Membaca skala berjenjang apa pun]\label{met:g6:measurement-in-science:read}
Penggaris, \omterm{def:g3:temperature-thermometers:temperature}{termometer}, gelas ukur, jarum penunjuk: satu keterampilan
membaca semuanya.
\begin{enumerate}
  \item carilah \emph{nilai satu undakan}: ambillah dua tanda
        bernomor yang bertetangga, kurangkan, lalu bagilah dengan
        banyaknya undakan kecil di antaranya --- pada penggaris yang
        bernomor tiap \omterm{def:g2:measuring-length:units}{sentimeter} dengan sepuluh undakan, tiap undakan
        bernilai $1 \div 10 = \qty{0.1}{cm}$, yaitu satu milimeter;
  \item tempatkan matamu tepat di depan tandanya (pandangan yang
        menyerong menggeser bacaannya);
  \item hitunglah terus dari tanda bernomor yang terakhir: $7$ dan
        $3$ undakan kecil berarti $7.3$;
  \item tulislah bilangannya \emph{beserta \omterm{def:g6:measurement-in-science:measuring}{satuannya}}.
\end{enumerate}
\end{method}

\begin{method}[Taksir, ukur, nilai]\label{met:g6:measurement-in-science:estimate}
Para profesional tidak pernah \omterm{def:g6:measurement-in-science:measuring}{mengukur} tanpa persiapan:
\begin{enumerate}
  \item \emph{taksirlah} lebih dahulu --- ``meja ini kira-kira dua
        \omterm{def:g2:measuring-length:units}{meter}'';
  \item \emph{pilihlah} alat yang jangkauan dan undakannya cocok
        dengan pekerjaannya (penggaris untuk perangko, meteran pita
        untuk koridor, timbangan dapur untuk tepung, timbangan kamar
        mandi untuk orang);
  \item \emph{ukurlah}, menurut aturan pembacaan tadi;
  \item \emph{nilailah}: apakah hasilnya kira-kira sesuai dengan
        taksiranmu? Meja yang terukur \qty{21.3}{m} tidak bertumbuh
        --- kamu salah baca dengan faktor sepuluh, dan taksiranmu
        baru saja menangkapnya.
\end{enumerate}
Taksiranmu adalah jaring pengamanmu: ia menangkap titik desimal yang
tergelincir sebelum titik itu menangkapmu.
\end{method}

\begin{example}[Memilih dengan bijak]\label{ex:g6:measurement-in-science:choose}
Untuk menimbang sepucuk surat yang akan dikirim, timbangan kamar mandi
tidak berguna --- \qty{18}{g} suratnya lenyap di antara undakannya;
timbangan dapur, yang berundak satu \omterm{def:g3:measuring-mass:units}{gram}, menjawab dengan sempurna.
Untuk menghitung waktu lari $100$ \omterm{def:g2:measuring-length:units}{meter}, undakan menit pada \omterm{ex:g2:measuring-time:clock}{jam} dapur
tidak ada harapan; stopwatch menghitung dalam perseratus sekon. Tidak
ada alat yang ``lebih baik'': masing-masing punya wilayahnya, yang
ditentukan oleh jangkauan dan undakan terkecilnya.
\end{example}

\section{Bilangan yang jujur}

\begin{remark}[Di antara dua tanda]\label{rem:g6:measurement-in-science:between}
Jarum yang sungguhan senang mendarat di antara dua tanda. Kejujuran
lalu punya dua tingkat: paling sedikit, sebutkan kedua tetangganya ---
``antara $7.3$ dan $7.4$, lebih dekat ke $7.3$''; yang lebih baik,
taksirlah satu angka lagi di luar tandanya --- ``kira-kira
\qty{7.32}{cm}''. Yang dilarang kejujuran adalah mengarang ketelitian:
membaca \qty{7.3214}{cm} dari tanda milimeter adalah dongeng yang
mengenakan jas laboratorium. Tanda pada sebuah alat menetapkan batas
apa yang dapat dijanjikannya.
\end{remark}

\begin{example}[Ukur dua kali, percaya lebih]\label{ex:g6:measurement-in-science:repeat}
Tiga teman menghitung waktu lari mobil mainan yang sama dengan
stopwatch yang sama: \qty{4.6}{s}, \qty{4.8}{s}, \qty{4.7}{s}. Tidak
ada yang berbuat salah --- jari memang menekan mulai dan berhenti
selisih sehelai rambut. Sebaran yang kecil adalah napas biasa
pengukuran yang sungguhan. Kebiasaan yang bijak: ulangi pengukurannya,
amati bacaannya berkerumun, dan curigailah bilangan tunggal yang
sendirian --- terutama bilanganmu sendiri.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:1}
Lengkapi tiap pengukuran dengan \omterm{def:g6:measurement-in-science:measuring}{satuan} yang masuk akal: sebuah pintu
tingginya $2$ \dots; \ sekeping koin bermassa $8$ \dots; \ satu
pelajaran di sekolah berlangsung $55$ \dots; \ demam bersuhu $39$ \dots
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:2}
Ubahlah: \qty{3.2}{m} menjadi \unit{cm}; \ \qty{750}{g} menjadi
\unit{kg}; \ \qty{4}{min} menjadi sekon; \ \qty{5600}{m} menjadi \unit{km}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:3}
Mengapa waktu menjadi pembangkang keluarga \omterm{def:g6:measurement-in-science:measuring}{satuan}? Ubahlah
\qty{3.5}{min} menjadi sekon --- lalu katakan jawaban keliru apa yang
akan diberikan tangga desimal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:4}
Sebuah gelas ukur bernomor tiap \qty{100}{mL} dengan lima undakan
kecil di antara nomornya. Berapa nilai satu undakan? Jusnya berdiri
dua undakan di atas tanda $300$ --- berapa banyak jusnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:5}
Alat yang mana untuk tiap pekerjaan, dan mengapa: menimbang sejumput
ragi; \ menghitung waktu denyut jantung; \ lebar halaman ini;
\ panjang halaman sekolah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:6}
Nadia \omterm{def:g6:measurement-in-science:measuring}{mengukur} pensilnya: \qty{17.4}{m}. Jaring pengaman yang mana
pada \cref{met:g6:measurement-in-science:estimate} yang seharusnya
menangkap ini, dan berapa kira-kira nilai yang sebenarnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:7}
Kolom sebuah \omterm{def:g3:temperature-thermometers:temperature}{termometer} berhenti di antara tanda $18$ dan $19$, lebih
dekat ke $19$. Berikan satu cara melaporkannya dengan jujur --- dan
satu laporan tidak jujur yang harus dihindari.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:8}
Tiga bacaan stopwatch untuk satu kali lari: \qty{6.1}{s},
\qty{6.3}{s}, \qty{6.2}{s}. Adakah yang bersalah? Bagaimana laporan
yang bijak?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:9}
Sebuah resep dari luar negeri meminta ``seperempat pon mentega''.
Timbangan dapurmu berbicara dalam \omterm{def:g3:measuring-mass:units}{gram}, dan satu pon penuh kira-kira
\qty{450}{g}. Berapa \omterm{def:g3:measuring-mass:units}{gram} yang kamu timbang? Apa yang diajarkan
kerepotan ini tentang \omterm{def:g6:measurement-in-science:measuring}{satuan} bersama?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:10}
\omterm{def:g2:measuring-length:units}{Sentimeter} pertama sebuah penggaris sudah aus, sehingga tepinya mulai
pada tanda $1$. Sebuah perangko yang diletakkan dari tepinya terbaca
$3.4$ pada ujung jauhnya. Berapa lebar perangko itu sebenarnya?
Kebiasaan apa yang mengalahkan penggaris yang aus?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:11}
Kelas \omterm{def:g6:measurement-in-science:measuring}{mengukur} koridornya dengan meteran pita \qty{20}{m} sebanyak dua
kali: \qty{18.25}{m} dan \qty{18.35}{m}. Gurunya menulis
``\qty{18.3}{m}''. Jelaskan pilihannya, dengan
\cref{rem:g6:measurement-in-science:between} dan
\cref{ex:g6:measurement-in-science:repeat}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:measurement-in-science:12}
Sebuah kisah lama: seorang raja menetapkan telapak kakinya sendiri
sebagai \omterm{def:g6:measurement-in-science:measuring}{satuan} panjang kerajaannya. Sebutkan dua bencana praktis yang
menyusul (pikirkan tukang kayu di kota yang jauh, dan putra sang raja
yang bertumbuh sebagai pewaris), lalu jelaskan bagaimana batang logam
yang sudah dipensiunkan pada
\cref{rem:g6:measurement-in-science:guard} memecahkan keduanya ---
dan masalah apa yang bahkan batang itu tinggalkan tanpa penyelesaian
sampai akhirnya dibereskan oleh tetapan alam itu sendiri.
\end{exercise}

\section{Soal: Sang Pemeriksa Alat Ukur}

\begin{problem}[{Soal akhir pekan --- sehari bersama pemeriksa alat
ukur kota; timbangan, gelas ukur, dan jam yang diadili; buku catatan
sang murid magang}]\label{pb:g6:measurement-in-science:1}
Setiap tahun pemeriksa alat ukur kota menguji alat milik toko dan
sekolah. Hari ini kamu adalah murid magangnya, buku catatan di tangan.

\medskip\noindent\textbf{Bagian I --- Timbangan toko roti.}
Sang pemeriksa membawa balok bersertifikat: \qty{500}{g},
\qty{200}{g}, \qty{200}{g}, \qty{100}{g}, \qty{50}{g}.
\begin{enumerate}
  \item Ia memeriksa bahwa jarum timbangan yang kosong berdiri tepat
        pada nol. Mengapa ini harus menjadi uji yang pertama dari
        semuanya?
  \item Ia meletakkan balok \qty{500}{g} dan satu balok \qty{200}{g}.
        Berapa yang seharusnya terbaca pada timbangan yang jujur?
  \item Jarumnya justru menunjukkan \qty{680}{g}. Apakah timbangan
        ini menguntungkan si tukang roti atau para pembeli? Jelaskan
        siapa yang merugi pada setiap penjualan.
  \item Gabungan balok mana saja yang membuatnya dapat menguji bacaan
        \qty{750}{g}? Berikan satu.
  \item Timbangan toko rotinya berundak \qty{5}{g}. Dapatkah sang
        pemeriksa memakainya untuk memeriksa perangko surat
        \qty{2}{g}? Mengapa tidak?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Gelas ukur milik sekolah.}
\begin{enumerate}[resume]
  \item Sebuah gelas ukur bernomor tiap \qty{200}{mL}, dengan empat
        undakan di antara nomornya. Berapa nilai tiap undakan?
  \item Airnya berdiri satu undakan di atas tanda \qty{600}{mL}.
        Catatan gurunya berbunyi ``kira-kira \qty{700}{mL}''.
        Nilailah laporan itu: jujur, terlalu kabur, atau dongeng?
  \item Seorang murid membaca gelas ukurnya dari kursi yang rendah,
        matanya jauh di bawah permukaan airnya, lalu mendapat
        bilangan yang berbeda dari sang pemeriksa. Aturan pembacaan
        yang mana yang dilanggar?
  \item Sang pemeriksa menuangkan \qty{650}{mL} yang dinyatakan gelas
        ukur itu ke dalam gelas ukur bersertifikat, yang menunjukkan
        \qty{640}{mL}. Berapa besar melesetnya gelas ukur sekolah
        itu, dan apakah itu lebih buruk atau lebih baik daripada satu
        undakannya sendiri?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- \omterm{ex:g2:measuring-time:clock}{Jam}, dan keputusan sang murid
magang.}
\begin{enumerate}[resume]
  \item \omterm{ex:g2:measuring-time:clock}{Jam} stasiun diperiksa terhadap isyarat waktu radio: ia
        menunjukkan tengah hari isyarat itu pada pukul $12$ lewat $2$
        menit. Cepat atau lambat, dan berapa selisihnya?
  \item Kalau tidak ada yang membetulkannya, seberapa jauh ia akan
        mendahului sesudah seminggu? (Ia maju sama banyaknya tiap
        hari.)
  \item \omterm{ex:g2:measuring-time:clock}{Jam} tiga toko menunjukkan pukul $12$ lewat $2$ menit, pukul
        $11$ lewat $58$ menit, dan tepat pukul $12$ pada tengah hari
        isyarat itu. Sang pemeriksa berkata: ``Merata-ratakan akan
        konyol di sini --- betulkan tiap \omterm{ex:g2:measuring-time:clock}{jamnya}.'' Perbaikan apa yang
        diperlukan masing-masingnya?
  \item Petang hari. Di buku catatanmu, tulislah tiga aturan emas
        sang pemeriksa --- satu tentang nol, satu tentang pembacaan,
        satu tentang \omterm{def:g6:measurement-in-science:measuring}{satuan} --- masing-masing dalam satu kalimat
        karanganmu sendiri.
\end{enumerate}
\end{problem}
