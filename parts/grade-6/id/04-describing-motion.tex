\chapter{Menggambarkan Gerak: Lintasan dan Kelajuan}\label{ch:g6:describing-motion}

Daun yang jatuh, kereta cepat, kuda komidi putar, tendangan penalti:
dunia tidak pernah diam. Untuk mempelajari gerak, fisika mengajukan
dua pertanyaan yang rapi kepada setiap benda yang bergerak:
\emph{menyusuri jalan yang mana?} dan \emph{secepat apa di sepanjang
jalan itu?} Jalan dan langkah --- \omterm{def:g6:describing-motion:trajectory}{lintasan} dan \omterm{def:g5:speed-distance-time:speed}{kelajuan} --- itulah
sepasang kacamata bab ini.

\section{Lintasan: jalan yang ditempuh}

\begin{definition}[Lintasan]\label{def:g6:describing-motion:trajectory}
Yang disebut \emph{lintasan}\index{lintasan} sebuah benda yang
bergerak adalah garis yang digambar oleh perjalanannya --- jalan yang
ditorehkannya melalui ruang. Tiga bentuk sudah meliput hampir seluruh
kehidupan sehari-hari: \emph{lurus}\index{lintasan lurus} (lift, apel
yang jatuh), \emph{melingkar}\index{lintasan lingkaran} (kuda komidi
putar, ujung jarum \omterm{ex:g2:measuring-time:clock}{jam}), dan \emph{melengkung} dengan cara yang lebih
bebas (busur bola yang dilempar, sambaran burung layang-layang).
\end{definition}

\begin{example}[Lintasan milik siapa?]\label{ex:g6:describing-motion:whose}
Berhati-hatilah menyebut \emph{titik yang mana} yang kamu ikuti.
Hidung kuda komidi putar menggambar lingkaran; pusat seluruh komidi
putarnya tidak menggambar apa-apa --- ia tinggal diam. Poros roda yang
menggelinding meluncur pada garis lurus sementara sebuah titik di
pelegnya menorehkan lompatan yang anggun. Satu mesin, banyak \omterm{def:g6:describing-motion:trajectory}{lintasan}:
fisika mengikuti satu titik pada satu waktu.
\end{example}

\begin{remark}[Bergerak dibandingkan dengan apa?]\label{rem:g6:describing-motion:relative}
Penumpang yang tertidur di kereta sama sekali tidak menggambar
\omterm{def:g6:describing-motion:trajectory}{lintasan} bagi penumpang di seberangnya --- dan menggambar garis
seratus kilometer di peta bagi seekor sapi yang mengamati dari ladang.
Kedua penggambaran itu jujur: gerak selalu \emph{gerak dibandingkan
dengan sesuatu}, dan si penggambar harus menyebutkan sesuatu itu.
Untuk sekarang kita diam-diam membandingkan segalanya dengan tanah;
kekuatan penuh dari memilih sudut pandang yang lain adalah kisah
termasyhur yang disimpan untuk jilid Sekolah Menengah.
\end{remark}

\section{Beraturan atau berubah?}

\begin{definition}[Gerak beraturan dan gerak berubah]\label{def:g6:describing-motion:uniform}
Sebuah gerak disebut \emph{beraturan}\index{gerak beraturan} ketika ia
menempuh jarak yang sama dalam waktu yang sama --- langkah ajek
seorang pendengar metronom, jalan tenang kereta di antara dua stasiun.
Ia disebut \emph{berubah}\index{gerak berubah} ketika langkahnya
berganti: melaju lebih cepat (pelari yang meninggalkan baloknya),
melambat (bus yang mengerem), atau keduanya bergantian (lalu lintas
kota yang tiada henti berhenti dan berjalan).
\end{definition}

\begin{proposition}[Membaca film bertitik]\label{prop:g6:describing-motion:dots}
Fotolah sebuah benda yang bergerak pada ketukan waktu yang sama ---
banyak jepretan cepat pada satu gambar --- dan geraknya menandatangani
namanya sendiri dalam titik:
\begin{enumerate}
  \item titiknya berjarak sama: \omterm{def:g6:describing-motion:uniform}{gerak beraturan};
  \item titiknya makin merenggang di sepanjang jalan: melaju lebih
        cepat;
  \item titiknya makin berdesakan: melambat.
\end{enumerate}
Waktu yang sama duduk di antara setiap pasang titik, sehingga celah
antartitik membandingkan jaraknya --- seluruh diagnosisnya dengan
sekali pandang.
\end{proposition}

\begin{method}[Membuat film bertitik]\label{met:g6:describing-motion:film}
Kamera telepon mana pun dapat menghasilkan potret seperti ini:
\begin{enumerate}
  \item rekamlah benda yang bergerak --- seorang teman yang
        bersepeda lewat, sebuah bola yang menggelinding turun dari
        landasan --- sambil memegang kameranya tetap diam;
  \item telusuri videonya langkah demi langkah, sambil berhenti pada
        selang yang sama (misalnya tiap setengah sekon pemutaran);
  \item pada tiap perhentian, tandai kedudukan bendanya pada kertas
        jiplak di atas layarnya, atau catatlah terhadap tiang pagar
        di latar belakangnya;
  \item bentangkan tandamu berderet lalu bacalah dengan
        \cref{prop:g6:describing-motion:dots}.
\end{enumerate}
Satu peringatan dari para profesional: peganglah kameranya tetap diam
--- kamera yang bergeser menambahkan geraknya sendiri ke dalam filmnya
lalu mengaburkan \omterm{def:g6:describing-motion:trajectory}{lintasan} siapa milik siapa (persis peringatan pada
\cref{rem:g6:describing-motion:relative}).
\end{method}

\section{Memberi bilangan pada langkahnya}

\begin{example}[Kelajuan lewat kesebandingan]\label{ex:g6:describing-motion:proportion}
Sebuah trem yang bergerak beraturan menempuh \qty{9}{km} dalam $20$
menit. Berapa \omterm{def:g5:speed-distance-time:speed}{kelajuannya}, dalam kilometer per \omterm{ex:g2:measuring-time:clock}{jam} yang sudah akrab?
Kesebandingan menjawabnya tanpa mesin baru apa pun: satu \omterm{ex:g2:measuring-time:clock}{jam} adalah
tiga jatah $20$ menit, sehingga tremnya menempuh
$3 \times 9 = \qty{27}{km}$ dalam satu \omterm{ex:g2:measuring-time:clock}{jam} --- \omterm{def:g5:speed-distance-time:speed}{kelajuannya} $27$
kilometer per \omterm{ex:g2:measuring-time:clock}{jam}.
Dalam tabel:
\begin{center}
\begin{tabular}{l|cc}
menit & $20$ & $60$ \\
\hline
kilometer & $9$ & $27$
\end{tabular}
\end{center}
\omterm{def:g6:describing-motion:uniform}{Gerak beraturan} \emph{adalah} kesebandingan antara jarak dan waktu ---
pelajaran matematikamu dan pelajaran fisikamu baru saja berjabat
tangan.
\end{example}

\begin{example}[Papan langkah]\label{ex:g6:describing-motion:paces}
Langkah sehari-hari, untuk menilai jawaban apa pun: berjalan santai,
kira-kira $4$ kilometer per \omterm{ex:g2:measuring-time:clock}{jam}; berjalan cepat, $6$; bersepeda
santai, $15$; bus kota dengan perhentian, $20$; mobil di jalan
terbuka, $90$; kereta cepat, sampai $300$. Ketika sebuah hitungan
menyerahkan kepadamu pejalan kaki dengan $40$ kilometer per \omterm{ex:g2:measuring-time:clock}{jam}, papan
langkah itu akan menolaknya.
\end{example}

\begin{remark}[Kelajuan pada suatu saat]\label{rem:g6:describing-motion:instant}
Gerak yang berubah tidak punya satu \omterm{def:g5:speed-distance-time:speed}{kelajuan} tunggal --- bus yang
mengerem cepat, lalu lambat, lalu diam. Jarum spidometer sebuah mobil
menjawab pertanyaan yang lebih halus: \emph{secepat apa saat ini?}
Menetapkan kelajuan-saat-ini itu dengan persis sungguh-sungguh dalam
--- ia memerlukan matematika yang akan kamu temui pada tahun terakhir
sekolah menengah, dan ia membuka salah satu bab terbesar dalam seluruh
sains. Untuk tahun ini: \omterm{def:g6:describing-motion:uniform}{gerak beraturan} mendapat satu bilangan yang
jujur; gerak yang berubah mendapat sebuah cerita.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:1}
Sebutkan bentuk \omterm{def:g6:describing-motion:trajectory}{lintasannya}: setetes hujan yang turun pada kaca
jendela yang diam; \ ujung jarum menit; \ bola basket yang ditembakkan
ke ring; \ slalom seorang pemain ski.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:2}
Pada sepeda yang menggelinding, \omterm{def:g6:describing-motion:trajectory}{lintasan} apa yang digambar poros
rodanya? Dan pentil di pelegnya? (Masing-masing satu kata dan satu
sketsa.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:3}
Beraturan atau berubah: kereta yang melaju tenang di antara dua
stasiun; \ kereta yang sama ketika meninggalkan peron; \ seorang
penerjun payung yang melayang turun dengan laju ajek; \ keping hoki
yang meluncur dan melambat di es yang kasar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:4}
Sebuah film bertitik menunjukkan celah \qty{2}{cm}, \qty{2}{cm},
\qty{2}{cm}, \qty{2}{cm} di antara jepretannya. Diagnosislah geraknya.
Film yang lain menunjukkan \qty{1}{cm}, \qty{2}{cm}, \qty{4}{cm},
\qty{7}{cm}: diagnosislah yang itu.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:5}
Mengapa kameranya harus tetap diam ketika membuat film bertitik?
Catatan mana dalam bab ini yang dipertaruhkan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:6}
Sebuah kapal feri yang bergerak beraturan menempuh \qty{12}{km} dalam
$30$ menit. Bangunlah tabel kesebandingannya lalu sebutkan
\omterm{def:g5:speed-distance-time:speed}{kelajuannya} dalam kilometer per \omterm{ex:g2:measuring-time:clock}{jam}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:7}
Seorang pesepeda mengaku ajek $14$ kilometer per \omterm{ex:g2:measuring-time:clock}{jam}. Sejauh apa ia
bersepeda dalam $30$ menit? Dalam $15$ menit? (Dengan tabel, bukan
rumus.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:describing-motion:8}
Terhadap papan langkah pada \cref{ex:g6:describing-motion:paces},
nilailah pengakuan berikut: seorang pejalan santai menempuh
\qty{12}{km} dalam satu \omterm{ex:g2:measuring-time:clock}{jam}; \ sebuah bus kota menempuh \qty{10}{km}
dalam $30$ menit; \ sebuah kereta cepat menempuh \qty{75}{km} dalam
$15$ menit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:9}
Penumpang yang tertidur pada
\cref{rem:g6:describing-motion:relative}: gambarkan geraknya ---
\omterm{def:g6:describing-motion:trajectory}{lintasan} dan langkahnya --- mula-mula bagi penumpang di seberangnya,
lalu bagi sapi di ladang. Mengapa kedua penggambarannya jujur?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:10}
Sebuah jalur metro panjangnya \qty{9}{km} dari ujung ke ujung.
Jadwalnya memberi waktu $18$ menit termasuk enam perhentian stasiun
selama satu menit. Selama waktu berjalannya yang sesungguhnya, apakah
langkah rata-rata metronya di atas atau di bawah $45$ kilometer per
\omterm{ex:g2:measuring-time:clock}{jam}? (Carilah dulu waktu berjalannya, baru pakai tabel.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:11}
Rancanglah sebuah film bertitik untuk menyelesaikan pertengkaran
keluarga: apakah anjingnya berlari dengan langkah yang ajek ketika
mengambilkan barang, atau berlari kencang lalu meluncur bebas?
Sebutkan langkahmu, apa yang akan kamu ukur, dan keputusan apa yang
akan diberikan tiap pola titiknya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:describing-motion:12}
Sebuah bola menggelinding jatuh dari tepi meja. Sketsalah \omterm{def:g6:describing-motion:trajectory}{lintasannya}
dari tepi meja sampai ke lantai sebagaimana dilihat orang yang berdiri
di dalam ruangan. Lalu beralasanlah --- sudut pandang lagi! ---
\omterm{def:g6:describing-motion:trajectory}{lintasan} apa yang akan dilihat seorang pengamat mungil yang menumpang
gerobak yang menggelinding di bawah bola yang jatuh itu tepat dengan
langkah maju bolanya. (Salah satu dari keduanya melihat sebuah
lengkungan; yang lain, sesuatu yang jauh lebih sederhana. Teka-teki
ini kembali, terpecahkan, di jilid Sekolah Menengah.)
\end{exercise}

\section{Soal: Film Bertitik Jalur 7}

\begin{problem}[{Soal akhir pekan --- seorang insinyur angkutan
membaca film bertitik sebuah jalur trem; mendiagnosis perjalanannya,
menghitung waktu jalurnya, menangkap kebohongan
jadwalnya}]\label{pb:g6:describing-motion:1}
Seorang insinyur angkutan sudah merekam perjalanan trem dengan kamera
di tepi jalan yang menandai kedudukan tremnya pada tiap ketukan (satu
ketukan $=$ $10$ sekon), lalu menyerahkan filmnya kepadamu.

\medskip\noindent\textbf{Bagian I --- Membaca filmnya.}
Film A, di antara dua stasiun, menunjukkan celah (dalam \omterm{def:g2:measuring-length:units}{meter}):
$50, 50, 50, 50, 50$.
\begin{enumerate}
  \item Diagnosislah gerak pada film A.
  \item Sejauh apa tremnya berjalan selama $50$ sekon film A?
  \item Film B, yang meninggalkan sebuah stasiun, menunjukkan celah
        $10, 25, 40, 50, 50$. Ceritakan kisah perjalanannya dengan
        kata-kata.
  \item Pada film B, pada selang ketukan yang mana geraknya (hampir)
        beraturan?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Langkah sesungguhnya jalur itu.}
\begin{enumerate}[resume]
  \item Pada bentangan beraturannya, tremnya menempuh \qty{50}{m}
        tiap $10$ sekon. Bangunlah tabelnya sampai satu menit: sejauh
        apa tiap menit?
  \item Lanjutkan sampai satu \omterm{ex:g2:measuring-time:clock}{jam}: berapa \omterm{def:g5:speed-distance-time:speed}{kelajuan} jelajah tremnya
        dalam kilometer per \omterm{ex:g2:measuring-time:clock}{jam}?
  \item Seluruh jalurnya \qty{12}{km}. Pada \omterm{def:g5:speed-distance-time:speed}{kelajuan} jelajah tanpa
        perhentian, berapa menit dari ujung ke ujung?
  \item Jadwalnya memberi waktu $50$ menit dari ujung ke ujung.
        Berapa menit yang hilang pada jalur yang sesungguhnya untuk
        stasiun, lampu merah, dan daerah lambat seluruhnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Iklan yang diadili.}
Poster perusahaannya membanggakan: ``Jalur 7 --- menyeberangi kota
dengan $18$ kilometer per \omterm{ex:g2:measuring-time:clock}{jam}, dua kali langkah terbaik seorang
pejalan kaki!''
\begin{enumerate}[resume]
  \item Periksalah aritmetika posternya terhadap jadwalnya:
        \qty{12}{km} dalam $50$ menit --- bangunlah tabelnya menuju
        satu \omterm{ex:g2:measuring-time:clock}{jam}. Jujurkah ``$18$ kilometer per \omterm{ex:g2:measuring-time:clock}{jam}'' untuk
        \emph{seluruh} perjalanannya?
  \item Seorang perewel membantah: ``Bilangan semacam itu tidak
        pernah ada di spidometer --- tremnya kalau tidak melaju pada
        $18$ ya berdiri pada $0$!'' Jelaskan apa yang sebenarnya
        digambarkan \omterm{def:g5:speed-distance-time:speed}{kelajuan} seluruh perjalanan, dan mengapa ia tetap
        bilangan yang adil bagi perjalanan yang berlangkah-langkah
        banyak.
  \item Insinyurnya mengusulkan melewatkan dua stasiun untuk
        menghemat empat menit. Berapa waktu ujung-ke-ujung yang baru,
        dan --- dengan tabel lagi --- berapa \omterm{def:g5:speed-distance-time:speed}{kelajuan} seluruh
        perjalanannya yang baru dalam kilometer per \omterm{ex:g2:measuring-time:clock}{jam}? (Bulatkan
        secara masuk akal.)
  \item Tulislah laporan satu kalimat sang insinyur: bilangan mana
        yang sebaiknya dipercaya para penglaju untuk membuat rencana,
        \omterm{def:g5:speed-distance-time:speed}{kelajuan} jelajah atau \omterm{def:g5:speed-distance-time:speed}{kelajuan} seluruh perjalanan, dan
        mengapa?
\end{enumerate}
\end{problem}
