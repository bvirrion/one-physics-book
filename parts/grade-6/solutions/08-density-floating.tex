\section*{Chapter \ref{ch:g6:density-floating} --- Density: Why Things Float}

\begin{solution}{exo:g6:density-floating:1}
The \omterm{def:g3:measuring-mass:mass}{mass} of one \omterm{def:g6:mass-and-volume:volume}{cubic centimetre} of the substance, in \omterm{def:g3:measuring-mass:units}{grams};
water's is \qty{1}{g} per \unit{cm^3}. The rule: less dense than
water \omterm{def:g1:floating-and-sinking:words}{floats}, denser \omterm{def:g1:floating-and-sinking:words}{sinks}.
\end{solution}

\begin{solution}{exo:g6:density-floating:2}
Oak \omterm{def:g1:floating-and-sinking:words}{floats}; stone \omterm{def:g1:floating-and-sinking:words}{sinks}; cork \omterm{def:g1:floating-and-sinking:words}{floats}; iron \omterm{def:g1:floating-and-sinking:words}{sinks}; ice \omterm{def:g1:floating-and-sinking:words}{floats} ---
by a whisker.
\end{solution}

\begin{solution}{exo:g6:density-floating:3}
$120 \div 80 = 1.5$: \omterm{def:g6:density-floating:density}{density} $1.5$ --- denser than water, it
\omterm{def:g1:floating-and-sinking:words}{sinks}.
\end{solution}

\begin{solution}{exo:g6:density-floating:4}
$158 \div 20 = 7.9$: the signature of iron.
\end{solution}

\begin{solution}{exo:g6:density-floating:5}
\omterm{def:g6:density-floating:density}{Density} belongs to the substance, not the piece: every \omterm{def:g6:mass-and-volume:volume}{cubic
centimetre} of oak carries the same share of \omterm{def:g3:measuring-mass:units}{grams}, splinter or
\omterm{def:g6:light-propagation:ray}{beam}. That makes \omterm{def:g6:density-floating:density}{density} an identity card for unknown substances.
\end{solution}

\begin{solution}{exo:g6:density-floating:6}
\omterm{def:g3:solids-liquids-gases:liquid}{Liquids} stack by \omterm{def:g6:density-floating:density}{density}, densest at the bottom: honey $1.4$
below water $1.0$ below oil $0.9$. The grape ($1.05$) \omterm{def:g1:floating-and-sinking:words}{sinks}
through oil and water and rests on the honey floor --- denser
than both upper floors, less dense than honey.
\end{solution}

\begin{solution}{exo:g6:density-floating:7}
Most \omterm{def:g3:solids-liquids-gases:solid}{solids} are denser than their own \omterm{def:g3:solids-liquids-gases:liquid}{liquid} and \omterm{def:g1:floating-and-sinking:words}{sink} in it. Ice
\omterm{def:g1:floating-and-sinking:words}{floats} because water swells on freezing --- the same stuff in a
tenth more room gives \omterm{def:g6:density-floating:density}{density} $0.9 < 1$.
\end{solution}

\begin{solution}{exo:g6:density-floating:8}
Wood signs about $0.6$: every \omterm{def:g6:mass-and-volume:volume}{cubic centimetre} of the log
underbids water's \qty{1}{g}, so the log \omterm{def:g1:floating-and-sinking:words}{floats} however huge.
Bronze signs near $9$: every \omterm{def:g6:mass-and-volume:volume}{cubic centimetre} of the coin
outbids water, so it \omterm{def:g1:floating-and-sinking:words}{sinks} however small. Size never mattered;
the per-room \omterm{def:g3:measuring-mass:mass}{mass} always did.
\end{solution}

\begin{solution}{exo:g6:density-floating:9}
$\qty{1}{L} = \qty{1000}{cm^3}$, so $800 \div 1000 = 0.8$ \omterm{def:g3:measuring-mass:units}{grams}
per \omterm{def:g6:mass-and-volume:volume}{cubic centimetre}. Ice at $0.9$ is \emph{denser} than this
\omterm{def:g3:solids-liquids-gases:liquid}{liquid}: the ice cube \omterm{def:g1:floating-and-sinking:words}{sinks} in it.
\end{solution}

\begin{solution}{exo:g6:density-floating:10}
The floating parcel is the whole hull: steel shell \emph{plus}
the great \omterm{def:g6:mass-and-volume:volume}{volume} of \omterm{def:g2:air-around-us:air}{air} it encloses. Averaged over all that room,
the parcel's \omterm{def:g6:density-floating:density}{density} falls below $1$, and the rule says \omterm{def:g1:floating-and-sinking:words}{float}. A
hole lets water replace the \omterm{def:g2:air-around-us:air}{air}: the parcel's \omterm{def:g6:density-floating:density}{density} climbs
toward steel's own, passes $1$ --- and the rule, unmoved,
says \omterm{def:g1:floating-and-sinking:words}{sink}.
\end{solution}

\begin{solution}{exo:g6:density-floating:11}
The referee changed: against sea water's $1.03$, a swimmer's
near-$1$ body is comfortably on the floating side, so the sea
carries you higher. Entering fresh water ($1.0$), the ship loses
that margin and settles deeper: its waterline rises up the hull.
\end{solution}

\begin{solution}{exo:g6:density-floating:12}
Half-submerged, the sphere sits exactly as deep in the water as a
parcel half water's \omterm{def:g6:density-floating:density}{density} should: shell plus \omterm{def:g2:air-around-us:air}{air} average out to
about $0.5$ \omterm{def:g3:measuring-mass:units}{grams} per \omterm{def:g6:mass-and-volume:volume}{cubic centimetre} --- the hidden \omterm{def:g2:air-around-us:air}{air} doing
most of the lightening. As water leaks in it replaces \omterm{def:g2:air-around-us:air}{air}, the
parcel's average \omterm{def:g6:density-floating:density}{density} climbs, and the sphere settles deeper
and deeper; the moment the average passes water's $1.0$, the
floating rule \omterm{ex:g3:first-electric-circuit:switch}{switches} sides and the sphere goes down.
\end{solution}

\begin{solution}{pb:g6:density-floating:1}
\textbf{1.} Nothing: swapping gold for an equal \emph{\omterm{def:g3:measuring-mass:mass}{mass}} of
silver keeps the balance happy --- \omterm{def:g3:measuring-mass:mass}{mass} alone cannot see the
substitution.
\textbf{2.} $1930 \div 19.3 = \qty{100}{cm^3}$.
\textbf{3.} \qty{130}{cm^3} --- the overflow is the crown's
\omterm{def:g6:mass-and-volume:volume}{volume}, by displacement.
\textbf{4.} The crown takes \qty{30}{cm^3} more room than pure
gold of its \omterm{def:g3:measuring-mass:mass}{mass} should: something bulkier than gold hides
inside.
\textbf{5.} $1930 \div 130 \approx 14.8$.
\textbf{6.} Gold $19.3$ --- crown $14.8$ --- silver $10.5$: the
crown falls between the two signatures.
\textbf{7.} ``Sire, every \omterm{def:g6:mass-and-volume:volume}{cubic centimetre} of pure gold carries
\qty{19.3}{g}; your crown carries only about $15$ --- pure gold
it is not.''
\textbf{8.} The range gives densities $1900 \div 130 \approx
14.6$ up to $1960 \div 130 \approx 15.1$ --- nowhere near
$19.3$: no plausible balance error rescues the goldsmith.
\textbf{9.} Silver is less dense: each stolen \omterm{def:g3:measuring-mass:units}{gram} of gold gave
back a \omterm{def:g3:measuring-mass:units}{gram} of silver that needs \emph{more room} ---
gram-for-gram substitution preserves \omterm{def:g3:measuring-mass:mass}{mass} and inflates \omterm{def:g6:mass-and-volume:volume}{volume}.
\textbf{10.} Gold share: $965 \div 19.3 = \qty{50.0}{cm^3}$;
silver share: $965 \div 10.5 \approx \qty{91.9}{cm^3}$.
\textbf{11.} Total $\approx \qty{142}{cm^3}$ --- more than the
measured $130$: the half-and-half guess used too much silver;
the true theft was smaller (the measured \omterm{def:g6:mass-and-volume:volume}{volume} sits between
$100$ and $142$).
\textbf{12.} ``Proved: the crown is not pure gold. Method: its
\omterm{def:g3:measuring-mass:mass}{mass} shared over its water-measured \omterm{def:g6:mass-and-volume:volume}{volume} gives a \omterm{def:g6:density-floating:density}{density} far
below gold's unchangeable signature. And the crown never needed
so much as a scratch: the water read its secret from the
outside --- that is the triumph.''
\end{solution}
