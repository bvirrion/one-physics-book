\section*{Chapitre \ref{ch:g6:mass-and-volume} --- Masse et volume}

\begin{solution}{exo:g6:mass-and-volume:1}
Le \omterm{def:g6:mass-and-volume:volume}{volume}~: quelle place un objet occupe --- en \unit{cm^3} et en
\omterm{def:g6:mass-and-volume:volume}{litres}. La \omterm{def:g3:measuring-mass:mass}{masse}~: quelle quantité de matière il contient --- en
\omterm{def:g3:measuring-mass:units}{grammes} et en \omterm{def:g3:measuring-mass:units}{kilogrammes}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:2}
\qty{2000}{cm^3}~; \ \qty{350}{cm^3}~; \ \qty{1.5}{L}~; \
\qty{500}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:3}
Sur la table~: une éprouvette penchée penche la surface. À hauteur
d'œil~: un œil trop haut ou trop bas décale le trait apparent
(l'erreur de la vue de biais). Au bas du \omterm{met:g6:mass-and-volume:cylinder}{ménisque}~: le \omterm{def:g3:solids-liquids-gases:liquid}{liquide} grimpe
le long des parois, et seul le bas de la courbe se tient au \omterm{def:g6:mass-and-volume:volume}{volume}
vrai.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:4}
$265 - 200 = 65$~: le \omterm{def:g6:mass-and-volume:volume}{volume} de la prune est de \qty{65}{cm^3}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:5}
L'eau se moule sur chaque creux et chaque bosse~: l'eau chassée égale
donc exactement le \omterm{def:g6:mass-and-volume:volume}{volume} du caillou --- c'est la proposition du
déplacement. La règle a besoin d'arêtes droites~; l'eau n'a besoin de
rien.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:6}
$5 \times 2 \times 1 = \qty{10}{cm^3}$. Vérification~: la descendre
dans une éprouvette --- la lecture doit monter d'exactement
\qty{10}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:7}
$745 - 280 = \qty{465}{g}$ de riz. Le bouton de tare~: remettre la
\omterm{def:g3:levers-and-scales:scale}{balance} à zéro avec le saladier vide dessus, puis verser.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:8}
Des \omterm{def:g3:measuring-mass:mass}{masses} égales --- un \omterm{def:g3:measuring-mass:units}{kilogramme} chacun. Les plumes prennent
beaucoup plus de place~: un \omterm{def:g6:mass-and-volume:volume}{volume} bien plus grand pour la même
matière.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:9}
Immerger le bouchon exprès~: l'enfoncer avec une fine brochette (en
lisant la montée quand seul le bouchon est immergé), ou l'attacher à
un lest --- \omterm{def:g6:measurement-in-science:measuring}{mesurer} d'abord la montée due au lest seul, puis celle du
couple, et retrancher la part du lest.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:10}
Versé~: $3 \times 250 + 200 = \qty{950}{mL}$. L'annonce manque de
\qty{50}{mL} --- un vingtième de \omterm{def:g6:mass-and-volume:volume}{litre} en moins.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:11}
La montée vaut $83.5 - 80.0 = \qty{3.5}{mL}$~: le \omterm{def:g6:mass-and-volume:volume}{volume} total du
collier est de \qty{3.5}{cm^3}. L'eau mesure le \omterm{def:g6:mass-and-volume:volume}{volume} total, métal,
fermoir et tout --- une affirmation de \qty{5}{cm^3} d'or ne peut
donc pas être vraie~: il n'y a même pas \qty{5}{cm^3} de collier.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:12}
La règle promet $4 \times 3 \times 2 = \qty{24}{cm^3}$. Descendre le
bloc (au fil, sans éclaboussure, entièrement immergé, sans toucher
les parois) dans \qty{150}{mL}~: la prédiction est une montée de
\qty{150}{mL} à \qty{174}{mL}. Si l'eau montait d'autre chose que
\qty{24}{mL} --- au-delà d'une ou deux graduations honnêtes de
l'éprouvette --- la proposition du déplacement serait réfutée. (Elle
ne l'a jamais été.)
\end{solution}

\begin{solution}{pb:g6:mass-and-volume:1}
\textbf{1.} $20 \times 1000 = \qty{20000}{cm^3}$.
\textbf{2.} $\qty{20}{L} = \qty{2000}{cL}$~; $2000 \div 75 = 26$
bouteilles pleines (avec un reste).
\textbf{3.} $26 \times 75 = \qty{1950}{cL}$, donc
$2000 - 1950 = \qty{50}{cL}$ restent.
\textbf{4.} $50 \div 25 = 2$ verres, exactement.
\textbf{5.} $\qty{75}{cL} = \qty{0.75}{L}$, et un \omterm{def:g6:mass-and-volume:volume}{litre} pèse environ
\qty{1000}{g}~: le jus pèse environ \qty{750}{g}.
\textbf{6.} $750 + 450 = \qty{1200}{g}$ --- soit \qty{1.2}{kg}.
\textbf{7.} $6 \times 1.2 = \qty{7.2}{kg}$ de bouteilles plus
\qty{2}{kg} de caisse~: \qty{9.2}{kg}.
\textbf{8.} $200 \div 9.2$~: $21$ caisses feraient \qty{193.2}{kg},
$22$ feraient \qty{202.4}{kg} --- trop. La camionnette prend $21$
caisses.
\textbf{9.} Le débordement est le jus déplacé par le caillou~: le
\omterm{def:g6:mass-and-volume:volume}{volume} du caillou est de \qty{40}{cm^3}, par la proposition du
déplacement.
\textbf{10.} $1000 - 40 = \qty{960}{mL}$ de jus restent.
\textbf{11.} L'éprouvette doit monter de \qty{100}{mL} à
\qty{140}{mL} --- une montée de \qty{40}{mL}, qui concorde avec le
débordement. Les poings ne mesurent rien~; l'eau dit la vérité.
\textbf{12.} Ces \qty{40}{cm^3}, remplis de jus, contiendraient
environ \qty{40}{g} de jus --- le caillou tasse \qty{104}{g} dans la
même place~: la matière du caillou est plus lourde que celle du jus à
place égale, plus de deux fois plus. Le chapitre sur la \omterm{def:g3:measuring-mass:mass}{masse}
volumique donnera à cette comparaison son nom et son nombre.
\end{solution}
