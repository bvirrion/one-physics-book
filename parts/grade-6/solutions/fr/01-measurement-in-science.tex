\section*{Chapitre \ref{ch:g6:measurement-in-science} --- La mesure en science~: unités et instruments}

\begin{solution}{exo:g6:measurement-in-science:1}
$2$ \unit{m}~; \ $8$ \unit{g}~; \ $55$ \unit{min}~; \ $39$
\unit{\celsius}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:2}
\qty{320}{cm}~; \ \qty{0.75}{kg}~; \ \qty{240}{s}~; \ \qty{5.6}{km}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:3}
Ses marches valent $60$ et non $10$~: l'escalier décimal ment donc.
$\qty{3.5}{min} = 3 \times 60 + 30 = \qty{210}{s}$, alors que
l'escalier aurait donné à tort \qty{350}{s}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:4}
Chaque pas~: $200 \div 5 = \qty{40}{mL}$. Deux pas au-dessus de
$300$~: $300 + 80 = \qty{380}{mL}$.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:5}
La levure~: la \omterm{def:g3:levers-and-scales:scale}{balance} de cuisine (des pas d'un \omterm{def:g3:measuring-mass:units}{gramme} --- le
pèse-personne ne remarquerait pas une pincée). Les battements de
cœur~: un chronomètre (des pas d'une seconde et plus fins). La
largeur de la page~: une règle (des pas d'un millimètre). La cour~:
un long \omterm{def:g2:measuring-length:units}{mètre} ruban ou une roue d'arpenteur (une étendue de plusieurs
\omterm{def:g2:measuring-length:units}{mètres}).
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:6}
L'étape du jugement~: on estime un crayon à moins de \qty{20}{cm},
et \qty{17.4}{m} échoue donc aussitôt au contrôle de bon sens. Elle a
lu des \omterm{def:g2:measuring-length:units}{centimètres} et écrit des \omterm{def:g2:measuring-length:units}{mètres}~: la valeur vraie est
\qty{17.4}{cm}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:7}
Honnête~: «~entre $18$ et \qty{19}{\celsius}, plus près de $19$~» (ou
«~environ \qty{18.8}{\celsius}~»). Malhonnête~:
«~\qty{18.7643}{\celsius}~» --- une précision que les graduations ne
peuvent pas promettre.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:8}
Personne n'est fautif~: les doigts partent et s'arrêtent à un cheveu
près, et une petite dispersion est la respiration normale de la
mesure. Compte rendu raisonnable~: les lectures se serrent autour de
\qty{6.2}{s} --- «~environ \qty{6.2}{s}~».
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:9}
Un quart de \qty{450}{g}~: environ \qty{112}{g} (\qty{113}{g}
convient). Cette course montre pourquoi les \omterm{def:g6:measurement-in-science:measuring}{unités} partagées
comptent~: une recette écrite dans les \omterm{def:g6:measurement-in-science:measuring}{unités} d'un pays doit être
traduite avant que les instruments d'un autre puissent lui obéir.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:10}
Le timbre s'étend du trait $1$ au trait $3.4$~:
$3.4 - 1 = \qty{2.4}{cm}$. L'habitude~: ne jamais se fier au bord
d'un instrument --- partir d'un trait choisi et soustraire, et les
règles usées perdent tout pouvoir.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:11}
Les deux lectures s'écartent de \qty{0.10}{m}~: le dernier chiffre de
l'une comme de l'autre n'est donc pas fiable, et elles se serrent
autour de $18.3$. Écrire «~\qty{18.3}{m}~» conserve tous les
chiffres que le ruban peut honnêtement promettre, et pas un de plus.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:12}
Désastres~: les «~pieds~» de deux villes ne coïncident jamais, si
bien que les poutres taillées dans une ville ne s'ajustent jamais aux
maisons d'une autre~; et quand le fils hérite, l'\omterm{def:g6:measurement-in-science:measuring}{unité} elle-même
change de longueur --- toutes les mesures du royaume expirent avec le
roi. La barre de métal a réglé les deux~: une longueur unique et
immuable, copiée pour toutes les villes, survivant à tous les rois.
Son propre défaut~: un objet unique peut être rayé, volé ou
lentement altéré, et le monde entier devait faire le voyage jusqu'à
son coffre pour vérifier les copies --- jusqu'à ce que les \omterm{def:g6:measurement-in-science:measuring}{unités}
soient rebâties sur les constantes de la nature, qui ne s'usent pas
et sont de service partout à la fois.
\end{solution}

\begin{solution}{pb:g6:measurement-in-science:1}
\textbf{1.} Chaque pesée ajoute la charge à ce que l'aiguille
indiquait déjà~; un départ non nul empoisonne toutes les lectures de
la journée --- le contrôle du départ honnête doit venir en premier.
\textbf{2.} $500 + 200 = \qty{700}{g}$.
\textbf{3.} Elle indique \qty{20}{g} de moins que la réalité. En
versant de la farine jusqu'à ce que l'aiguille dise \qty{700}{g}, le
boulanger en a réellement versé \qty{720}{g}~: la \omterm{def:g3:levers-and-scales:scale}{balance} lui fait
donner plus de marchandise que le client n'en paie --- elle avantage
les clients, et le boulanger perd à chaque vente.
\textbf{4.} $500 + 200 + 50 = \qty{750}{g}$.
\textbf{5.} Non~: un timbre de \qty{2}{g} disparaît sous les pas de
\qty{5}{g} de la \omterm{def:g3:levers-and-scales:scale}{balance} --- hors de son territoire.
\textbf{6.} $200 \div 4 = \qty{50}{mL}$ par pas.
\textbf{7.} L'eau est à $600 + 50 = \qty{650}{mL}$~; «~environ
\qty{700}{mL}~» arrondit un pas entier --- trop vague pour un
instrument aussi bon.
\textbf{8.} L'œil au niveau du trait~: lire d'en dessous décale la
surface apparente de l'eau --- la règle du regard bien en face a été
enfreinte.
\textbf{9.} Un écart de \qty{10}{mL} --- mieux (plus petit) que son
propre pas de \qty{50}{mL}~: acceptable pour un verre doseur
d'école.
\textbf{10.} Elle avance, de $2$ minutes.
\textbf{11.} $2 \times 7 = 14$ minutes d'avance au bout d'une
semaine.
\textbf{12.} Première horloge~: la retarder de $2$ min~; deuxième~:
l'avancer de $2$ min~; troisième~: rien --- elle est juste. (Faire la
moyenne des trois ne «~réparerait~» rien et fausserait la seule
horloge honnête.)
\textbf{13.} Par exemple~: \emph{Zéro}~: vérifier que l'instrument à
vide indique zéro avant de croire quoi que ce soit de ce qu'il dit.
\emph{Lecture}~: apprendre ce que vaut un pas et lire bien en face,
en restant honnêtement modeste entre deux traits. \emph{\omterm{def:g6:measurement-in-science:measuring}{Unités}}~: un
nombre sans son \omterm{def:g6:measurement-in-science:measuring}{unité} n'est pas une mesure --- écrire l'\omterm{def:g6:measurement-in-science:measuring}{unité},
toujours.
\end{solution}
