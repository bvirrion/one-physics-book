\section*{Chapitre \ref{ch:g6:water-states} --- L'eau dans tous ses états}

\begin{solution}{exo:g6:water-states:1}
La \omterm{def:g6:water-states:changes}{fusion} (le givre du matin qui quitte l'herbe)~; la \omterm{def:g6:water-states:changes}{solidification}
(les glaçons du congélateur)~; la \omterm{def:g6:water-states:changes}{vaporisation} (le linge qui sèche,
la casserole qui bout)~; la \omterm{def:g6:water-states:changes}{condensation} (la buée sur le \omterm{def:g5:mirrors-reflection:reflection}{miroir}
froid)~; la \omterm{def:g6:water-states:changes}{sublimation} (un banc de neige qui rétrécit par gel sec)~;
le passage direct du \omterm{def:g3:solids-liquids-gases:gas}{gaz} au \omterm{def:g3:solids-liquids-gases:solid}{solide} (les fleurs de givre sur la
vitre).
\end{solution}

\begin{solution}{exo:g6:water-states:2}
La rosée~: \omterm{def:g6:water-states:changes}{condensation}. Le bonhomme de neige qui rétrécit sans
flaque~: \omterm{def:g6:water-states:changes}{sublimation} --- du \omterm{def:g3:solids-liquids-gases:solid}{solide} directement à la vapeur, dans
l'\omterm{def:g2:air-around-us:air}{air} froid et sec. La «~fumée~» du congélateur~: \omterm{def:g6:water-states:changes}{condensation} --- la
vapeur de la pièce qui \omterm{def:g2:ice-water-steam:evapcond}{se condense} en petit nuage dans l'\omterm{def:g2:air-around-us:air}{air} froid
(des gouttelettes, pas du \omterm{def:g3:solids-liquids-gases:gas}{gaz}).
\end{solution}

\begin{solution}{exo:g6:water-states:3}
Toujours \qty{412}{g}~: la \omterm{def:g3:measuring-mass:mass}{masse} survit aux changements d'état --- la
bouteille fermée n'a rien perdu ni rien gagné.
\end{solution}

\begin{solution}{exo:g6:water-states:4}
Bouchée, pour qu'aucune vapeur ne parte avec ses \omterm{def:g3:measuring-mass:units}{grammes} (l'audit
doit rester clos)~; séchée, parce que de l'eau ou du givre accrochés
à l'extérieur ajouteraient des \omterm{def:g3:measuring-mass:units}{grammes} étrangers, qui appartiennent
à la pièce et non à l'audit.
\end{solution}

\begin{solution}{exo:g6:water-states:5}
Rien n'a été détruit~: \qty{200}{g} d'eau sont partis sous forme de
vapeur dans l'\omterm{def:g2:air-around-us:air}{air} de la cuisine --- la \omterm{def:g6:water-states:changes}{vaporisation} a fait sortir les
\omterm{def:g3:measuring-mass:units}{grammes} de la casserole, pas du monde.
\end{solution}

\begin{solution}{exo:g6:water-states:6}
Environ \qty{1.1}{L} de glace --- un dixième de place en plus. La
\omterm{def:g3:measuring-mass:mass}{masse} est inchangée~: \qty{1}{kg} à l'entrée, \qty{1}{kg} à la
sortie.
\end{solution}

\begin{solution}{exo:g6:water-states:7}
L'eau enfermée dans un tuyau gonfle en gelant, et le tuyau, comme la
bouteille en verre, cède. Les jardiniers vidangent les robinets à
l'automne pour qu'il ne reste rien à l'intérieur qui puisse gonfler.
\end{solution}

\begin{solution}{exo:g6:water-states:8}
La glace tasse la matière d'un \omterm{def:g6:mass-and-volume:volume}{litre} dans $1.1$ \omterm{def:g6:mass-and-volume:volume}{litre} de place~: plus
légère que l'eau à place égale --- et plus léger à place égale, c'est
exactement ce qui \omterm{def:g1:floating-and-sinking:words}{flotte}.
\end{solution}

\begin{solution}{exo:g6:water-states:9}
La mer (\qty{1}{kg}, \omterm{def:g3:solids-liquids-gases:liquid}{liquide}) --- \omterm{def:g6:water-states:changes}{vaporisation} --- vapeur
(\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{condensation} --- gouttelettes de nuage
(\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{solidification} --- neige (\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{fusion}
--- eau de fonte (\qty{1}{kg}) --- rivière puis mer (\qty{1}{kg}).
Six étapes, une \omterm{def:g3:measuring-mass:mass}{masse} inchangée.
\end{solution}

\begin{solution}{exo:g6:water-states:10}
Il faut rassurer. La glace qui \omterm{def:g1:floating-and-sinking:words}{flotte} déplace déjà son propre
\emph{\omterm{def:g4:weight-and-mass:weight}{poids}} de limonade~; en fondant, elle rend exactement cette
eau, retirée du dixième dont elle avait gonflé --- l'eau se glisse
dans la place que le glaçon occupait déjà sous la surface. Le verre
reste plein à ras bord, sans déborder d'une goutte.
\end{solution}

\begin{solution}{exo:g6:water-states:11}
La porte du \omterm{def:g3:solids-liquids-gases:gas}{gaz} vers le \omterm{def:g3:solids-liquids-gases:solid}{solide}~: la vapeur invisible de l'\omterm{def:g2:air-around-us:air}{air} de la
pièce a touché la vitre glacée et y a construit de la glace
directement, en sautant l'état \omterm{def:g3:solids-liquids-gases:liquid}{liquide}. Ce n'est pas de la
«~\omterm{def:g6:water-states:changes}{solidification}~» parce qu'il n'y a jamais eu de \omterm{def:g3:solids-liquids-gases:liquid}{liquide} sur la
vitre --- l'eau est arrivée directement de l'état de vapeur.
\end{solution}

\begin{solution}{exo:g6:water-states:12}
L'étang perd de la \omterm{def:g5:heat-and-insulation:heat}{chaleur} vers l'\omterm{def:g2:air-around-us:air}{air} d'hiver par sa surface~: l'eau
de surface atteint donc \qty{0}{\celsius} la première et sa glace s'y
forme --- et elle \omterm{def:g1:floating-and-sinking:words}{flotte}, étant plus légère que l'eau à place égale,
si bien qu'elle reste au-dessus. Ce couvercle de glace (et de neige)
est une voie lente~: il isole l'eau du dessous, dont la \omterm{def:g5:heat-and-insulation:heat}{chaleur} ne
s'échappe plus que lentement vers le haut. L'eau profonde reste
légèrement au-dessus de zéro tout l'hiver, et les poissons nagent
sous un toit que leur étrange eau gonflante a bâti pour eux.
\end{solution}

\begin{solution}{pb:g6:water-states:1}
\textbf{1.} La \omterm{def:g6:water-states:changes}{vaporisation} (l'évaporation depuis la surface
chaude).
\textbf{2.} Toujours \qty{1}{kg} --- la \omterm{def:g3:measuring-mass:mass}{masse} survit aux changements
d'état.
\textbf{3.} La \omterm{def:g6:water-states:changes}{condensation}.
\textbf{4.} À \qty{0}{\celsius}.
\textbf{5.} Environ \qty{1.1}{L}.
\textbf{6.} \omterm{def:g3:measuring-mass:mass}{Masse} \qty{1}{kg}~; \omterm{def:g6:mass-and-volume:volume}{volume} revenu à \qty{1}{L}.
\textbf{7.} $1000 - 250 = \qty{750}{g}$ poursuivent dans le torrent.
\textbf{8.} Non~: les \qty{250}{g} chevauchent l'\omterm{def:g2:air-around-us:air}{air} sous forme de
vapeur, intacts~; leur porte suivante la plus probable est la
\omterm{def:g6:water-states:changes}{condensation} --- en nuage, puis en pluie.
\textbf{9.} $750 + 250 = \qty{1000}{g}$~: le \omterm{def:g3:measuring-mass:units}{kilogramme} entier est
rentré à la maison.
\textbf{10.} Invisible deux fois~: sous forme de vapeur juste après
avoir quitté la mer, et de nouveau sous forme de vapeur au-dessus de
la vasque de la cascade --- les deux fois à l'état gazeux.
\textbf{11.} «~Duveteux~» parle de \emph{\omterm{def:g6:mass-and-volume:volume}{volume}}~: la neige est faite
de cristaux de glace lacés d'\omterm{def:g2:air-around-us:air}{air}, qui prennent une place généreuse.
La \emph{\omterm{def:g3:measuring-mass:mass}{masse}} de notre eau étiquetée est restée exactement
\qty{1}{kg} --- léger-pour-la-place n'est pas léger-en-matière~: le
couple de mots, c'est \omterm{def:g3:measuring-mass:mass}{masse} et \omterm{def:g6:mass-and-volume:volume}{volume}.
\textbf{12.} \omterm{def:g6:water-states:changes}{Vaporisation}, \omterm{def:g6:water-states:changes}{condensation}, \omterm{def:g6:water-states:changes}{solidification}, \omterm{def:g6:water-states:changes}{fusion} ---
et, pour le quart évaporé~: \omterm{def:g6:water-states:changes}{vaporisation}, \omterm{def:g6:water-states:changes}{condensation} (la pluie),
puis retour à l'état \omterm{def:g3:solids-liquids-gases:liquid}{liquide}. Quatre portes utilisées~; la
\omterm{def:g6:water-states:changes}{sublimation} et la porte du \omterm{def:g3:solids-liquids-gases:gas}{gaz} vers le \omterm{def:g3:solids-liquids-gases:solid}{solide} n'ont pas servi lors de
ce voyage.
\end{solution}
