\section*{Chapter \ref{ch:g6:water-states} --- Water in All Its States}

\begin{solution}{exo:g6:water-states:1}
\omterm{def:g6:water-states:changes}{Melting} (frost off the morning grass); \omterm{def:g6:water-states:changes}{solidification} (the
freezer's ice cubes); \omterm{def:g6:water-states:changes}{vaporization} (laundry drying, the boiling
pot); \omterm{def:g6:water-states:changes}{condensation} (mist on the cold \omterm{def:g5:mirrors-reflection:reflection}{mirror}); \omterm{def:g6:water-states:changes}{sublimation} (a
snowbank shrinking in dry frost); \omterm{def:g3:solids-liquids-gases:gas}{gas} to \omterm{def:g3:solids-liquids-gases:solid}{solid} directly (frost
feathers on the pane).
\end{solution}

\begin{solution}{exo:g6:water-states:2}
Dew: \omterm{def:g6:water-states:changes}{condensation}. The puddle-less shrinking snowman: \omterm{def:g6:water-states:changes}{sublimation}
--- \omterm{def:g3:solids-liquids-gases:solid}{solid} straight to vapor in the cold dry \omterm{def:g2:air-around-us:air}{air}. The freezer's
``smoke'': \omterm{def:g6:water-states:changes}{condensation} --- room vapor condensing into a little
cloud in the cold \omterm{def:g2:air-around-us:air}{air} (droplets, not \omterm{def:g3:solids-liquids-gases:gas}{gas}).
\end{solution}

\begin{solution}{exo:g6:water-states:4}

\begin{solution}{exo:g6:water-states:5}
Not destroyed: \qty{200}{g} of water left as vapor through the
kitchen \omterm{def:g2:air-around-us:air}{air} --- \omterm{def:g6:water-states:changes}{vaporization} moved the \omterm{def:g3:measuring-mass:units}{grams} out of the pot, not

\end{solution}

\begin{solution}{exo:g6:water-states:6}
About \qty{1.1}{L} of ice --- a tenth more room. The \omterm{def:g3:measuring-mass:mass}{mass} is
unchanged: \qty{1}{kg} going in, \qty{1}{kg} coming out.
\end{solution}

\begin{solution}{exo:g6:water-states:7}
Water trapped in a pipe swells as it \omterm{def:g2:ice-water-steam:meltfreeze}{freezes}, and the pipe, like
the glass bottle, gives way. Gardeners drain autumn taps so there
is nothing left inside to swell.
\end{solution}

\begin{solution}{exo:g6:water-states:8}
Ice packs a \omterm{def:g6:mass-and-volume:volume}{litre}'s stuff into $1.1$ \omterm{def:g6:mass-and-volume:volume}{litres} of room: lighter than
water for the same room --- and lighter-for-the-same-room is
exactly what \omterm{def:g1:floating-and-sinking:words}{floats}.
\end{solution}

\begin{solution}{exo:g6:water-states:9}
Sea (\qty{1}{kg}, \omterm{def:g3:solids-liquids-gases:liquid}{liquid}) --- \omterm{def:g6:water-states:changes}{vaporization} --- vapor
(\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{condensation} --- cloud droplets (\qty{1}{kg})
--- \omterm{def:g6:water-states:changes}{solidification} --- snow (\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{melting} ---
meltwater (\qty{1}{kg}) --- river to sea (\qty{1}{kg}). Six
stages, one unchanging \omterm{def:g3:measuring-mass:mass}{mass}.
\end{solution}

\begin{solution}{exo:g6:water-states:10}
Reassure. The bobbing ice already displaces its own \emph{\omterm{def:g4:weight-and-mass:weight}{weight}}
of lemonade; \omterm{def:g6:water-states:changes}{melting} returns exactly that water, shrunk back by
the tenth it had swollen --- it slips into the room the cube was
already using below the surface. The glass stays brim-full,
overflowing not a drop.
\end{solution}

\begin{solution}{exo:g6:water-states:11}
The gas-to-solid door: invisible vapor from the room's \omterm{def:g2:air-around-us:air}{air} touched
the freezing pane and built ice directly, skipping \omterm{def:g3:solids-liquids-gases:liquid}{liquid}. It is
not ``freezing'' because nothing \omterm{def:g3:solids-liquids-gases:liquid}{liquid} ever existed on the pane
--- the water arrived straight from the vapor state.
\end{solution}

\begin{solution}{exo:g6:water-states:12}
The pond loses \omterm{def:g5:heat-and-insulation:heat}{heat} to the winter \omterm{def:g2:air-around-us:air}{air} at its surface, so surface
water reaches \qty{0}{\celsius} first and its ice forms there ---
and \omterm{def:g1:floating-and-sinking:words}{floats}, being lighter than water for the same room, so it
stays on top. The lid of ice (and snow) is a slow lane: it
insulates the water beneath, whose \omterm{def:g5:heat-and-insulation:heat}{heat} now leaks only slowly
upward. The deep water hovers above freezing all winter, and the
fish swim under a roof their strange, swelling water built for
them.
\end{solution}

\begin{solution}{pb:g6:water-states:1}
\textbf{1.} \omterm{def:g6:water-states:changes}{Vaporization} (evaporation from the warm surface).
\textbf{2.} Still \qty{1}{kg} --- \omterm{def:g3:measuring-mass:mass}{mass} survives changes of state.
\textbf{3.} \omterm{def:g6:water-states:changes}{Condensation}.
\textbf{4.} At \qty{0}{\celsius}.
\textbf{5.} About \qty{1.1}{L}.
\textbf{6.} \omterm{def:g3:measuring-mass:mass}{Mass} \qty{1}{kg}; \omterm{def:g6:mass-and-volume:volume}{volume} back to \qty{1}{L}.
\textbf{7.} $1000 - 250 = \qty{750}{g}$ continue in the stream.
\textbf{8.} No: the \qty{250}{g} ride the \omterm{def:g2:air-around-us:air}{air} as vapor, intact;
their likeliest next door is \omterm{def:g6:water-states:changes}{condensation} --- into cloud, then
rain.
\textbf{9.} $750 + 250 = \qty{1000}{g}$: the whole \omterm{def:g3:measuring-mass:units}{kilogram} came
home.
\textbf{10.} Twice invisible: as vapor just after leaving the sea,
and as vapor again over the waterfall pool --- both in the \omterm{def:g3:solids-liquids-gases:gas}{gas}
state.
\textbf{11.} Fluffy is about \emph{\omterm{def:g6:mass-and-volume:volume}{volume}}: snow is ice crystals
laced with \omterm{def:g2:air-around-us:air}{air}, taking generous room. The \emph{\omterm{def:g3:measuring-mass:mass}{mass}} of our
labeled water stayed exactly \qty{1}{kg} --- light-per-room is not
light-in-stuff: the pair of words is \omterm{def:g3:measuring-mass:mass}{mass} and \omterm{def:g6:mass-and-volume:volume}{volume}.
\textbf{12.} \omterm{def:g6:water-states:changes}{Vaporization}, \omterm{def:g6:water-states:changes}{condensation}, \omterm{def:g6:water-states:changes}{solidification}, \omterm{def:g6:water-states:changes}{melting}
--- and for the evaporated quarter: \omterm{def:g6:water-states:changes}{vaporization}, \omterm{def:g6:water-states:changes}{condensation}
(rain), rejoining as \omterm{def:g3:solids-liquids-gases:liquid}{liquid}. Four doors used; \omterm{def:g6:water-states:changes}{sublimation} and the
gas-to-solid door were not needed on this tour.
\end{solution}
