\section*{Capítulo \ref{ch:g6:mass-and-volume} --- Massa e volume}

\begin{solution}{exo:g6:mass-and-volume:1}
\omterm{def:g6:mass-and-volume:volume}{Volume}: quanto espaço uma coisa ocupa --- \unit{cm^3} e \omterm{def:g6:mass-and-volume:volume}{litros}.
\omterm{def:g3:measuring-mass:mass}{Massa}: quanto material ela contém --- \omterm{def:g3:measuring-mass:units}{gramas} e \omterm{def:g3:measuring-mass:units}{quilogramas}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:2}
\qty{2000}{cm^3}; \ \qty{350}{cm^3}; \ \qty{1.5}{L}; \
\qty{500}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:3}
Sobre a mesa: uma proveta inclinada inclina a superfície. Na altura
do olho: um olho alto ou baixo desloca a marca aparente (o erro da
vista de lado). No fundo do \omterm{met:g6:mass-and-volume:cylinder}{menisco}: o \omterm{def:g3:solids-liquids-gases:liquid}{líquido} sobe pelas paredes, e
só o fundo da curva está no \omterm{def:g6:mass-and-volume:volume}{volume} verdadeiro.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:4}
$265 - 200 = 65$: o \omterm{def:g6:mass-and-volume:volume}{volume} da ameixa é \qty{65}{cm^3}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:5}
A água se molda a cada reentrância e a cada saliência, então a água
empurrada para o lado é exatamente igual ao \omterm{def:g6:mass-and-volume:volume}{volume} do seixo --- a
proposição do deslocamento. A régua precisa de arestas retas; a água
não precisa de nenhuma.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:6}
$5 \times 2 \times 1 = \qty{10}{cm^3}$. Conferência: mergulhe-a numa
proveta --- a leitura deve subir exatamente \qty{10}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:7}
$745 - 280 = \qty{465}{g}$ de arroz. O botão de tara: zere a balança
com a tigela vazia em cima e depois despeje.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:8}
As \omterm{def:g3:measuring-mass:mass}{massas} são iguais --- um \omterm{def:g3:measuring-mass:units}{quilograma} cada. As penas ocupam muito
mais espaço: \omterm{def:g6:mass-and-volume:volume}{volume} bem maior para a mesma quantidade de material.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:9}
Afunde a rolha de propósito: empurre-a para baixo com um espeto fino
(leia a subida com apenas a rolha submersa), ou amarre-a a um \omterm{def:g4:weight-and-mass:weight}{peso}
--- meça primeiro a subida só do \omterm{def:g4:weight-and-mass:weight}{peso}, depois a do conjunto, e
subtraia a parte do \omterm{def:g4:weight-and-mass:weight}{peso}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:10}
Despejado: $3 \times 250 + 200 = \qty{950}{mL}$. A declaração errou
por \qty{50}{mL} --- faltou um vigésimo de \omterm{def:g6:mass-and-volume:volume}{litro}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:11}
A subida é $83.5 - 80.0 = \qty{3.5}{mL}$: o \omterm{def:g6:mass-and-volume:volume}{volume} inteiro do colar é
\qty{3.5}{cm^3}. A água mede o \omterm{def:g6:mass-and-volume:volume}{volume} total, metal, fecho e tudo ---
então uma afirmação de \qty{5}{cm^3} de ouro não pode ser verdadeira:
não há nem \qty{5}{cm^3} de colar.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:12}
A régua promete $4 \times 3 \times 2 = \qty{24}{cm^3}$. Mergulhe o
bloco (com linha, sem respingo, todo submerso, sem encostar nas
paredes) em \qty{150}{mL}: a previsão é uma subida de \qty{150}{mL}
para \qty{174}{mL}. Se a água subisse qualquer coisa diferente de
\qty{24}{mL} --- fora de um ou dois passinhos honestos da proveta ---
a proposição do deslocamento estaria refutada. (Isso nunca
aconteceu.)
\end{solution}

\begin{solution}{pb:g6:mass-and-volume:1}
\textbf{1.} $20 \times 1000 = \qty{20000}{cm^3}$.
\textbf{2.} $\qty{20}{L} = \qty{2000}{cL}$; $2000 \div 75 = 26$
garrafas cheias (com sobra).
\textbf{3.} $26 \times 75 = \qty{1950}{cL}$, então sobram
$2000 - 1950 = \qty{50}{cL}$.
\textbf{4.} $50 \div 25 = 2$ copos, exatamente.
\textbf{5.} $\qty{75}{cL} = \qty{0.75}{L}$, e um \omterm{def:g6:mass-and-volume:volume}{litro} pesa cerca de
\qty{1000}{g}: o suco tem cerca de \qty{750}{g}.
\textbf{6.} $750 + 450 = \qty{1200}{g}$ --- \qty{1.2}{kg}.
\textbf{7.} $6 \times 1.2 = \qty{7.2}{kg}$ de garrafas mais
\qty{2}{kg} de engradado: \qty{9.2}{kg}.
\textbf{8.} $200 \div 9.2$: $21$ engradados dariam \qty{193.2}{kg},
e $22$ dariam \qty{202.4}{kg} --- demais. A perua leva $21$
engradados.
\textbf{9.} O suco derramado é o suco que o seixo deslocou: o \omterm{def:g6:mass-and-volume:volume}{volume}
do seixo é \qty{40}{cm^3}, pela proposição do deslocamento.
\textbf{10.} $1000 - 40 = \qty{960}{mL}$ de suco restam.
\textbf{11.} A proveta deve subir de \qty{100}{mL} para
\qty{140}{mL} --- uma subida de \qty{40}{mL}, batendo com o
derramamento. Punhos não medem nada; a água diz a verdade.
\textbf{12.} Esses \qty{40}{cm^3}, cheios de suco, guardariam cerca
de \qty{40}{g} de suco --- o seixo enfia \qty{104}{g} no mesmo
espaço: o material do seixo é mais pesado que o material do suco para
o mesmo espaço, mais que o dobro. O capítulo da densidade vai dar a
essa comparação o nome e o número próprios dela.
\end{solution}
