\section*{अध्याय \ref{ch:g6:describing-motion} --- गति का वर्णन: गतिपथ और चाल}

\begin{solution}{exo:g6:describing-motion:1}
सरल रेखीय (शीशे पर सीधे नीचे); वृत्तीय; वक्र (एक मेहराब); वक्र
(लहराती हुई रेखा)।
\end{solution}

\begin{solution}{exo:g6:describing-motion:2}
\omterm{def:g3:levers-and-scales:lever}{धुरी}: एक सीधी रेखा, सड़क के समांतर सरकती हुई। वाल्व: सुंदर छलाँगों का
वक्र --- हर चक्कर पर उठता, मेहराब बनाता, और फिर नीचे छूता हुआ।
\end{solution}

\begin{solution}{exo:g6:describing-motion:3}
एकसमान; परिवर्ती (तेज़ होती हुई); एकसमान; परिवर्ती (धीमी होती हुई)।
\end{solution}

\begin{solution}{exo:g6:describing-motion:4}
बराबर फ़ासले: \omterm{def:g6:describing-motion:uniform}{एकसमान गति}। बढ़ते फ़ासले ($1, 2, 4, 7$): गति तेज़ हो रही
है --- हर बराबर टिक में ज़्यादा दूरी।
\end{solution}

\begin{solution}{exo:g6:describing-motion:5}
खिसकता कैमरा अभिलेख में अपनी गति भी जोड़ देता है --- तब बिंदु वस्तु की
यात्रा और कैमरे की यात्रा को मिला देते हैं। गति हमेशा किसी चीज़ के
मुक़ाबले होती है, और फ़िल्म को अपनी वह ``कोई चीज़'' पक्की करनी पड़ती
है: यही दृष्टिकोण वाली टिप्पणी है।
\end{solution}

\begin{solution}{exo:g6:describing-motion:6}
तालिका: $30$ मिनट $\to$ \qty{12}{km}, इसलिए $60$ मिनट $\to$
\qty{24}{km}: नाव की \omterm{def:g5:speed-distance-time:speed}{चाल} $24$ किलोमीटर प्रति घंटा है।
\end{solution}

\begin{solution}{exo:g6:describing-motion:7}
$60$ मिनट $\to$ \qty{14}{km}; $30$ मिनट $\to$ \qty{7}{km}; $15$
मिनट $\to$ \qty{3.5}{km}।
\end{solution}

\begin{solution}{exo:g6:describing-motion:8}
एक घंटे में \qty{12}{km} चलने वाला टहलने वाला: अस्वीकार --- टहलने की
रफ़्तार का तीन गुना। बस: $30$ मिनट में $10$ किलोमीटर यानी $20$
किलोमीटर प्रति घंटा --- ठीक शहर की बस वाली रफ़्तार, स्वीकार।
रेलगाड़ी: $15$ मिनट में $75$ किलोमीटर यानी $300$ किलोमीटर प्रति घंटा
--- तख़्ते के सबसे ऊपर, पर तेज़ रफ़्तार रेलगाड़ी के लिए मुमकिन:
स्वीकार।
\end{solution}

\begin{solution}{exo:g6:describing-motion:9}
सामने बैठे यात्री के लिए: कोई गति ही नहीं --- न \omterm{def:g6:describing-motion:trajectory}{गतिपथ}, रफ़्तार शून्य।
गाय के लिए: रेलगाड़ी की पूरी रफ़्तार पर एक लंबा \omterm{def:g6:describing-motion:trajectory}{सरल रेखीय गतिपथ}।
दोनों ईमानदार हैं: हर एक ऊँघते यात्री का वर्णन अपने ही दृष्टिकोण के
मुक़ाबले करता है, और गति हमेशा किसी चीज़ के मुक़ाबले ही होती है।
\end{solution}

\begin{solution}{exo:g6:describing-motion:10}
चलने का समय: \qty{9}{km} के लिए $18 - 6 = 12$ मिनट। तालिका: $12$
मिनट $\to$ $9$ किलोमीटर, इसलिए $60$ मिनट $\to$ $45$ किलोमीटर ---
यानी ठीक $45$ किलोमीटर प्रति घंटा: न ऊपर, न नीचे, बल्कि बिलकुल वहीं।
\end{solution}

\begin{solution}{exo:g6:describing-motion:11}
स्थिर कैमरे से गेंद लाने का वीडियो बनाओ; हर आधे सेकंड पर रोको; बाड़ के
खंभों के सामने कुत्ते की जगहों पर निशान लगाओ; फ़ासले मापो। पूरे रास्ते
बराबर फ़ासले: एक-सी रफ़्तार। फ़ासले पहले फैलें फिर सिमटें (या चौड़े और
सँकरे बारी-बारी आएँ): दौड़-और-सुस्ती। घर में जो भी चिल्लाहट हुई हो,
फ़ैसला बिंदु सुनाते हैं।
\end{solution}

\begin{solution}{exo:g6:describing-motion:12}
कमरे से: एक वक्र --- गेंद मेज़ के किनारे से आगे और नीचे की ओर मेहराब
बनाती हुई फ़र्श तक जाती है। गेंद की आगे वाली रफ़्तार से लुढ़कती नन्ही
गाड़ी से: गेंद न आगे निकलती है न पीछे रहती है --- वह \emph{सीधे नीचे}
गिरती दिखती है। एक ही गिरना, दो दृष्टिकोण, और सरल नज़ारा चलते हुए
प्रेक्षक का है: एक बड़े विचार का बीज।
\end{solution}

\begin{solution}{pb:g6:describing-motion:1}
\textbf{1.} \qty{50}{m} के बराबर फ़ासले: \omterm{def:g6:describing-motion:uniform}{एकसमान गति}।
\textbf{2.} $5 \times 50 = \qty{250}{m}$।
\textbf{3.} ट्राम ठहरी हुई हालत से चलती है: पहले छोटे फ़ासले, जो
बढ़ते जाते हैं --- पहले तीन टिकों में गति तेज़ होती है --- और फिर
\qty{50}{m} के बराबर फ़ासलों पर जम जाती है: यानी चलती रफ़्तार।
\textbf{4.} आख़िरी दो अंतराल, जिनमें हर एक \qty{50}{m} का है।
\textbf{5.} $10$ सेकंड $\to$ \qty{50}{m}, इसलिए $60$ सेकंड $\to$
प्रति मिनट \qty{300}{m}।
\textbf{6.} $60$ मिनट $\to$ $60 \times 300 = \qty{18000}{m}$:
चलती रफ़्तार $18$ किलोमीटर प्रति घंटा।
\textbf{7.} तालिका: $60$ मिनट में \qty{18}{km}, इसलिए \qty{12}{km}
के लिए $40$ मिनट।
\textbf{8.} $50 - 40 = 10$ मिनट ठहरावों और धीमे इलाक़ों में गँवाए
जाते हैं।
\textbf{9.} $50$ मिनट $\to$ \qty{12}{km}, इसलिए $60$ मिनट $\to$
$12 \times 60 \div 50 = \qty{14.4}{km}$: पूरी यात्रा लगभग $14$
किलोमीटर प्रति घंटा पर चलती है --- पोस्टर का $18$ सिर्फ़ चलते हुए
हिस्सों के लिए ईमानदार है, ``पूरे शहर के पार'' के लिए नहीं।
\textbf{10.} पूरी यात्रा वाली \omterm{def:g5:speed-distance-time:speed}{चाल} यात्रा का वर्णन ऐसे करती है
\emph{मानो} वह एकसमान रही हो: यानी वह एक-सी रफ़्तार जो उन्हीं $50$
मिनटों में उतने ही \qty{12}{km} तय कर देती। कोई चालमापी उसे नहीं
दिखाता, फिर भी योजना बनाने के लिए वही निष्पक्ष संख्या है, क्योंकि
उसमें हर ठहराव और हर धीमा इलाक़ा पहले से शामिल है।
\textbf{11.} $50 - 4 = 46$ मिनट; $12 \times 60 \div 46 \approx
\qty{15.7}{km}$ प्रति घंटा --- यानी लगभग $16$।
\textbf{12.} जैसे: ``सवारियों को अपनी योजना पूरी यात्रा वाली \omterm{def:g5:speed-distance-time:speed}{चाल} से
बनानी चाहिए --- आज लगभग $14$ किलोमीटर प्रति घंटा --- क्योंकि इंतज़ार
को गिनने वाली अकेली संख्या यही है; चलती रफ़्तार वाला $18$ लाइन की
चापलूसी करता है और हर ठहराव भुला देता है।''
\end{solution}
