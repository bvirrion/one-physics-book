\section*{Bab \ref{ch:g6:measurement-in-science} --- Pengukuran dalam Sains: Satuan dan Alat Ukur}

\begin{solution}{exo:g6:measurement-in-science:1}
$2$ \unit{m}; \ $8$ \unit{g}; \ $55$ \unit{min}; \ $39$
\unit{\celsius}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:2}
\qty{320}{cm}; \ \qty{0.75}{kg}; \ \qty{240}{s}; \ \qty{5.6}{km}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:3}
Undakannya berjalan dengan $60$, bukan dengan $10$, sehingga tangga
desimalnya berbohong: $\qty{3.5}{min} = 3 \times 60 + 30 = \qty{210}{s}$,
sementara tangganya akan keliru memberikan \qty{350}{s}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:4}
Tiap undakan: $200 \div 5 = \qty{40}{mL}$. Dua undakan di atas $300$:
$300 + 80 = \qty{380}{mL}$.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:5}
Ragi: timbangan dapur (berundak \omterm{def:g3:measuring-mass:units}{gram} --- timbangan kamar mandi tidak
akan menyadari sejumput). Denyut jantung: stopwatch (berundak sekon
dan lebih halus). Lebar halaman: penggaris (berundak milimeter).
Halaman sekolah: meteran pita yang panjang atau roda ukur (jangkauan
berskala \omterm{def:g2:measuring-length:units}{meter}).
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:6}
Langkah penilaiannya: sebuah pensil ditaksir kurang dari \qty{20}{cm},
sehingga \qty{17.4}{m} langsung gagal pada uji kewarasannya. Ia
membaca \omterm{def:g2:measuring-length:units}{sentimeter} lalu menulis \omterm{def:g2:measuring-length:units}{meter}: nilai yang sebenarnya
\qty{17.4}{cm}.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:7}
Jujur: ``antara $18$ dan \qty{19}{\celsius}, lebih dekat ke $19$''
(atau ``kira-kira \qty{18.8}{\celsius}''). Tidak jujur:
``\qty{18.7643}{\celsius}'' --- ketelitian yang tidak dapat dijanjikan
oleh tandanya.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:8}
Tidak ada yang bersalah: jari memang menekan mulai dan berhenti
selisih sehelai rambut, dan sebaran yang kecil adalah napas biasa
pengukuran. Laporan yang bijak: bacaannya berkerumun di sekitar
\qty{6.2}{s} --- ``kira-kira \qty{6.2}{s}''.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:9}
Seperempat dari \qty{450}{g}: kira-kira \qty{112}{g} (\qty{113}{g}
juga baik). Kerepotan itu menunjukkan mengapa \omterm{def:g6:measurement-in-science:measuring}{satuan} bersama penting:
resep yang ditulis dalam \omterm{def:g6:measurement-in-science:measuring}{satuan} sebuah negeri harus diterjemahkan
sebelum alat negeri yang lain dapat menurutinya.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:10}
Perangkonya membentang dari tanda $1$ sampai tanda $3.4$: $3.4 - 1 =
\qty{2.4}{cm}$. Kebiasaannya: jangan pernah memercayai tepi sebuah
alat --- mulailah dari tanda pilihan lalu kurangkan, dan penggaris
yang aus pun kehilangan kuasanya.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:11}
Kedua bacaannya bersebaran sejauh \qty{0.10}{m}, sehingga angka
terakhir keduanya tidak dapat dipercaya; keduanya berkerumun di
sekitar $18.3$. Menulis ``\qty{18.3}{m}'' mempertahankan setiap angka
yang dapat dijanjikan meteran pitanya dengan jujur, dan tidak lebih.
\end{solution}

\begin{solution}{exo:g6:measurement-in-science:12}
Bencananya: tidak ada dua kota yang ``telapak kakinya'' sepakat,
sehingga balok yang dipotong di satu kota tidak pernah muat di rumah
kota yang lain; dan ketika putranya mewarisi takhta, \omterm{def:g6:measurement-in-science:measuring}{satuannya} sendiri
berubah panjang --- setiap pengukuran di kerajaan itu kedaluwarsa
bersama sang raja. Batang logam membereskan keduanya: satu panjang
yang tidak berubah, yang disalin untuk semua kota, dan yang berumur
lebih panjang daripada semua raja. Cacatnya sendiri: satu benda
tunggal dapat tergores, dicuri, atau berubah perlahan, dan dunia harus
berjalan ke brankasnya untuk memeriksa salinannya --- sampai \omterm{def:g6:measurement-in-science:measuring}{satuannya}
dibangun ulang di atas tetapan alam, yang tidak dapat aus dan bertugas
di mana-mana sekaligus.
\end{solution}

\begin{solution}{pb:g6:measurement-in-science:1}
\textbf{1.} Setiap penimbangan menambahkan bebannya kepada apa pun
yang sudah ditunjukkan jarumnya; permulaan yang bukan nol meracuni
seluruh bacaan hari itu --- pemeriksaan permulaan yang adil harus
datang lebih dahulu.
\textbf{2.} $500 + 200 = \qty{700}{g}$.
\textbf{3.} Bacaannya \qty{20}{g} lebih ringan. Kalau tepungnya
dituang sampai jarumnya berkata \qty{700}{g}, si tukang roti
sebenarnya sudah menuang \qty{720}{g}: timbangan itu membuat si tukang
roti menyerahkan barang lebih banyak daripada yang dibayar pembelinya
--- ia menguntungkan para pembeli, dan si tukang roti merugi pada
setiap penjualan.
\textbf{4.} $500 + 200 + 50 = \qty{750}{g}$.
\textbf{5.} Tidak: perangko \qty{2}{g} lenyap di bawah undakan
\qty{5}{g} timbangannya --- di luar wilayahnya.
\textbf{6.} $200 \div 4 = \qty{50}{mL}$ tiap undakan.
\textbf{7.} Airnya pada $600 + 50 = \qty{650}{mL}$; ``kira-kira
\qty{700}{mL}'' membulatkan satu undakan penuh --- terlalu kabur untuk
alat sebaik ini.
\textbf{8.} Mata sejajar dengan tandanya: membaca dari bawah menggeser
letak permukaan airnya --- aturan tepat-di-depan yang dilanggar.
\textbf{9.} Meleset \qty{10}{mL} --- lebih baik (lebih kecil) daripada
undakan \qty{50}{mL} miliknya sendiri: dapat diterima untuk gelas ukur
sekolah.
\textbf{10.} Cepat, sebanyak $2$ menit.
\textbf{11.} $2 \times 7 = 14$ menit mendahului sesudah seminggu.
\textbf{12.} \omterm{ex:g2:measuring-time:clock}{Jam} pertama: mundurkan $2$ menit; \omterm{ex:g2:measuring-time:clock}{jam} kedua: majukan $2$
menit; \omterm{ex:g2:measuring-time:clock}{jam} ketiga: tidak perlu apa-apa --- ia sudah benar.
(Merata-ratakan ketiganya tidak akan ``membetulkan'' apa pun dan malah
memalsukan satu \omterm{ex:g2:measuring-time:clock}{jam} yang jujur.)
\textbf{13.} Misalnya: \emph{Nol}: periksalah bahwa alat yang kosong
membaca nol sebelum memercayai apa pun yang dikatakannya.
\emph{Pembacaan}: pelajarilah nilai satu undakannya lalu bacalah tepat
di depannya, dan bersikaplah rendah hati dengan jujur untuk bacaan di
antara dua tanda. \emph{\omterm{def:g6:measurement-in-science:measuring}{Satuan}}: bilangan tanpa \omterm{def:g6:measurement-in-science:measuring}{satuannya} bukanlah
sebuah pengukuran --- tulislah \omterm{def:g6:measurement-in-science:measuring}{satuannya}, selalu.
\end{solution}
