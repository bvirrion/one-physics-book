\section*{Bab \ref{ch:g6:water-states} --- Air dalam Segala Wujudnya}

\begin{solution}{exo:g6:water-states:1}
\omterm{def:g6:water-states:changes}{Peleburan} (embun beku yang lepas dari rumput pagi); \omterm{def:g6:water-states:changes}{pembekuan}
(bongkahan es di lemari beku); \omterm{def:g6:water-states:changes}{penguapan} (cucian yang mengering, panci
yang mendidih); \omterm{def:g6:water-states:changes}{pengembunan} (kabut pada \omterm{def:g5:mirrors-reflection:reflection}{cermin} yang dingin); \omterm{def:g6:water-states:changes}{sublimasi}
(timbunan salju yang menyusut pada embun beku yang kering); \omterm{def:g3:solids-liquids-gases:gas}{gas} ke
padat secara langsung (bulu embun beku pada kaca jendela).
\end{solution}

\begin{solution}{exo:g6:water-states:2}
Embun: \omterm{def:g6:water-states:changes}{pengembunan}. Manusia salju yang menyusut tanpa genangan:
\omterm{def:g6:water-states:changes}{sublimasi} --- dari padat langsung menjadi uap di \omterm{def:g2:air-around-us:air}{udara} dingin yang
kering. ``Asap'' lemari beku: \omterm{def:g6:water-states:changes}{pengembunan} --- uap ruangan yang
\omterm{def:g2:ice-water-steam:evapcond}{mengembun} menjadi awan kecil di \omterm{def:g2:air-around-us:air}{udara} yang dingin (tetes, bukan \omterm{def:g3:solids-liquids-gases:gas}{gas}).
\end{solution}

\begin{solution}{exo:g6:water-states:3}
Tetap \qty{412}{g}: \omterm{def:g3:measuring-mass:mass}{massa} selamat melewati perubahan wujud --- botol
yang tertutup rapat itu tidak kehilangan maupun mendapat apa pun.
\end{solution}

\begin{solution}{exo:g6:water-states:4}
Ditutup, supaya tidak ada uap yang melayang pergi bersama \omterm{def:g3:measuring-mass:units}{gramnya}
(auditnya harus tetap tertutup); dikeringkan, karena air atau embun
beku dari luar yang menempel akan menambahkan \omterm{def:g3:measuring-mass:units}{gram} asing yang menjadi
milik ruangan, bukan milik auditnya.
\end{solution}

\begin{solution}{exo:g6:water-states:5}
Tidak dimusnahkan: \qty{200}{g} air pergi sebagai uap lewat \omterm{def:g2:air-around-us:air}{udara}
dapur --- \omterm{def:g6:water-states:changes}{penguapan} memindahkan \omterm{def:g3:measuring-mass:units}{gramnya} keluar dari pancinya, bukan
keluar dari dunia.
\end{solution}

\begin{solution}{exo:g6:water-states:6}
Kira-kira \qty{1.1}{L} es --- ruangnya sepersepuluh lebih banyak.
\omterm{def:g3:measuring-mass:mass}{Massanya} tidak berubah: \qty{1}{kg} masuk, \qty{1}{kg} keluar.
\end{solution}

\begin{solution}{exo:g6:water-states:7}
Air yang terperangkap di dalam pipa menggembung ketika \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}, dan
pipanya, seperti botol kaca itu, menyerah. Tukang kebun mengosongkan
keran pada \omterm{def:g3:sun-days-seasons:year}{musim} gugur supaya tidak ada apa pun yang tertinggal di
dalamnya untuk menggembung.
\end{solution}

\begin{solution}{exo:g6:water-states:8}
Es memampatkan zat seliter ke dalam ruang $1.1$ \omterm{def:g6:mass-and-volume:volume}{liter}: lebih ringan
daripada air untuk ruang yang sama --- dan lebih-ringan-untuk-ruang-
yang-sama itulah yang tepat membuat sesuatu \omterm{def:g1:floating-and-sinking:words}{terapung}.
\end{solution}

\begin{solution}{exo:g6:water-states:9}
Laut (\qty{1}{kg}, cair) --- \omterm{def:g6:water-states:changes}{penguapan} --- uap (\qty{1}{kg}) ---
\omterm{def:g6:water-states:changes}{pengembunan} --- tetes awan (\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{pembekuan} --- salju
(\qty{1}{kg}) --- \omterm{def:g6:water-states:changes}{peleburan} --- air lelehan (\qty{1}{kg}) --- sungai
menuju laut (\qty{1}{kg}). Enam tahap, satu \omterm{def:g3:measuring-mass:mass}{massa} yang tidak berubah.
\end{solution}

\begin{solution}{exo:g6:water-states:10}
Tenangkanlah. Es yang \omterm{def:g1:floating-and-sinking:words}{terapung} itu sudah mendesak limun seberat
\emph{\omterm{def:g4:weight-and-mass:weight}{beratnya}} sendiri; ketika \omterm{def:g2:ice-water-steam:meltfreeze}{mencair} ia mengembalikan tepat air
itu, yang menyusut kembali sebanyak sepersepuluh yang tadi
menggembung --- airnya menyelinap ke dalam ruang yang tadi sudah
dipakai bongkahannya di bawah permukaan. Gelasnya tetap penuh sampai
ke bibirnya, tanpa melimpah setetes pun.
\end{solution}

\begin{solution}{exo:g6:water-states:11}
Pintu gas-ke-padat: uap yang tak terlihat dari \omterm{def:g2:air-around-us:air}{udara} ruangan menyentuh
kaca yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku} lalu membangun es secara langsung, dengan
melewatkan wujud cairnya. Namanya bukan ``\omterm{def:g6:water-states:changes}{pembekuan}'' karena tidak
pernah ada \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} di kaca itu --- airnya tiba langsung dari wujud
uapnya.
\end{solution}

\begin{solution}{exo:g6:water-states:12}
Kolamnya kehilangan \omterm{def:g5:heat-and-insulation:heat}{kalor} ke \omterm{def:g2:air-around-us:air}{udara} \omterm{def:g3:sun-days-seasons:year}{musim} dingin di permukaannya,
sehingga air permukaannya mencapai \qty{0}{\celsius} lebih dahulu dan
esnya terbentuk di sana --- lalu \omterm{def:g1:floating-and-sinking:words}{terapung}, karena lebih ringan
daripada air untuk ruang yang sama, sehingga ia tinggal di atas.
Tutup es (dan salju) itu adalah jalur lambat: ia mengisolasi air di
bawahnya, yang \omterm{def:g5:heat-and-insulation:heat}{kalornya} kini bocor ke atas hanya dengan pelan. Air
yang dalam melayang di atas titik beku sepanjang \omterm{def:g3:sun-days-seasons:year}{musim} dingin, dan
ikannya berenang di bawah atap yang dibangunkan oleh airnya yang aneh
dan menggembung itu.
\end{solution}

\begin{solution}{pb:g6:water-states:1}
\textbf{1.} \omterm{def:g6:water-states:changes}{Penguapan} (dari permukaan yang hangat).
\textbf{2.} Tetap \qty{1}{kg} --- \omterm{def:g3:measuring-mass:mass}{massa} selamat melewati perubahan
wujud.
\textbf{3.} \omterm{def:g6:water-states:changes}{Pengembunan}.
\textbf{4.} Pada \qty{0}{\celsius}.
\textbf{5.} Kira-kira \qty{1.1}{L}.
\textbf{6.} \omterm{def:g3:measuring-mass:mass}{Massanya} \qty{1}{kg}; \omterm{def:g6:mass-and-volume:volume}{volumenya} kembali ke \qty{1}{L}.
\textbf{7.} $1000 - 250 = \qty{750}{g}$ melanjutkan perjalanan di
dalam sungainya.
\textbf{8.} Tidak: \qty{250}{g} itu menunggangi \omterm{def:g2:air-around-us:air}{udara} sebagai uap,
utuh; pintu berikutnya yang paling mungkin adalah \omterm{def:g6:water-states:changes}{pengembunan} ---
menjadi awan, lalu hujan.
\textbf{9.} $750 + 250 = \qty{1000}{g}$: seluruh sekilogramnya pulang.
\textbf{10.} Dua kali tak terlihat: sebagai uap sesaat sesudah
meninggalkan laut, dan sebagai uap lagi di atas kolam air terjunnya
--- keduanya dalam wujud \omterm{def:g3:solids-liquids-gases:gas}{gas}.
\textbf{11.} Gembur itu tentang \emph{\omterm{def:g6:mass-and-volume:volume}{volume}}: salju adalah hablur es
yang berjalin dengan \omterm{def:g2:air-around-us:air}{udara}, sehingga memakai ruang dengan murah hati.
\emph{\omterm{def:g3:measuring-mass:mass}{Massa}} air berlabel kita tetap tepat \qty{1}{kg} ---
ringan-tiap-ruang bukanlah ringan-dalam-zat: sepasang katanya adalah
\omterm{def:g3:measuring-mass:mass}{massa} dan \omterm{def:g6:mass-and-volume:volume}{volume}.
\textbf{12.} \omterm{def:g6:water-states:changes}{Penguapan}, \omterm{def:g6:water-states:changes}{pengembunan}, \omterm{def:g6:water-states:changes}{pembekuan}, \omterm{def:g6:water-states:changes}{peleburan} --- dan
untuk seperempat yang \omterm{def:g2:ice-water-steam:evapcond}{menguap}: \omterm{def:g6:water-states:changes}{penguapan}, \omterm{def:g6:water-states:changes}{pengembunan} (hujan), lalu
bergabung kembali sebagai \omterm{def:g3:solids-liquids-gases:liquid}{zat cair}. Empat pintu dipakai; \omterm{def:g6:water-states:changes}{sublimasi} dan
pintu gas-ke-padat tidak diperlukan pada lawatan ini.
\end{solution}
