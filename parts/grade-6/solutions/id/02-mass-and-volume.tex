\section*{Bab \ref{ch:g6:mass-and-volume} --- Massa dan Volume}

\begin{solution}{exo:g6:mass-and-volume:1}
\omterm{def:g6:mass-and-volume:volume}{Volume}: seberapa banyak ruang yang dipakai sebuah benda ---
\unit{cm^3} dan \omterm{def:g6:mass-and-volume:volume}{liter}. \omterm{def:g3:measuring-mass:mass}{Massa}: seberapa banyak zat yang dikandungnya
--- \omterm{def:g3:measuring-mass:units}{gram} dan \omterm{def:g3:measuring-mass:units}{kilogram}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:2}
\qty{2000}{cm^3}; \ \qty{350}{cm^3}; \ \qty{1.5}{L}; \
\qty{500}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:3}
Di atas meja: gelas ukur yang miring memiringkan permukaannya. Mata
sejajar: mata yang tinggi atau rendah menggeser letak tanda yang
tampak (kesalahan pandangan menyerong). Dasar meniskusnya: \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya}
memanjat dindingnya, dan hanya dasar lengkungannya yang berdiri pada
\omterm{def:g6:mass-and-volume:volume}{volume} yang sebenarnya.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:4}
$265 - 200 = 65$: \omterm{def:g6:mass-and-volume:volume}{volume} premnya \qty{65}{cm^3}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:5}
Airnya membentuk dirinya mengikuti setiap lekuk dan tonjolan, sehingga
air yang terdesak ke samping sama persis dengan \omterm{def:g6:mass-and-volume:volume}{volume} kerikilnya ---
itulah proposisi pendesakan air. Penggaris memerlukan tepi yang lurus;
airnya tidak memerlukan apa-apa.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:6}
$5 \times 2 \times 1 = \qty{10}{cm^3}$. Pemeriksaannya: turunkan ke
dalam gelas ukur --- bacaannya seharusnya naik tepat \qty{10}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:7}
$745 - 280 = \qty{465}{g}$ beras. Tombol taranya: nolkan \omterm{def:g3:levers-and-scales:scale}{neracanya}
dengan mangkuk kosong di atasnya, baru tuangkan.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:8}
\omterm{def:g3:measuring-mass:mass}{Massanya} sama --- masing-masing satu \omterm{def:g3:measuring-mass:units}{kilogram}. Bulunya memakai jauh
lebih banyak ruang: \omterm{def:g6:mass-and-volume:volume}{volumenya} jauh lebih besar untuk zat yang sama.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:9}
Tenggelamkan gabusnya dengan sengaja: tekanlah ke bawah air dengan
tusuk yang tipis (bacalah kenaikannya ketika hanya gabusnya yang
terbenam), atau ikatkan pada pemberat --- ukurlah dulu kenaikan
pemberatnya sendirian, lalu keduanya bersama, dan kurangkan jatah
pemberatnya.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:10}
Yang tertuang: $3 \times 250 + 200 = \qty{950}{mL}$. Pengakuannya
meleset \qty{50}{mL} --- kurang seperdua puluh \omterm{def:g6:mass-and-volume:volume}{liter}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:11}
Kenaikannya $83.5 - 80.0 = \qty{3.5}{mL}$: seluruh \omterm{def:g6:mass-and-volume:volume}{volume} kalungnya
\qty{3.5}{cm^3}. Airnya \omterm{def:g6:measurement-in-science:measuring}{mengukur} \omterm{def:g6:mass-and-volume:volume}{volume} totalnya, logam beserta
kaitnya --- sehingga pengakuan \qty{5}{cm^3} emas tidak mungkin benar:
kalungnya sendiri saja tidak sampai \qty{5}{cm^3}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:12}
Penggarisnya menjanjikan $4 \times 3 \times 2 = \qty{24}{cm^3}$.
Turunkan baloknya (dengan benang, tanpa cipratan, seluruhnya terbenam,
tanpa menyentuh dindingnya) ke dalam \qty{150}{mL}: ramalannya adalah
kenaikan dari \qty{150}{mL} menjadi \qty{174}{mL}. Kalau airnya naik
sebanyak apa pun selain \qty{24}{mL} --- di luar satu dua undakan
jujur gelas ukurnya --- proposisi pendesakan air akan terbantahkan.
(Itu tidak pernah terjadi.)
\end{solution}

\begin{solution}{pb:g6:mass-and-volume:1}
\textbf{1.} $20 \times 1000 = \qty{20000}{cm^3}$.
\textbf{2.} $\qty{20}{L} = \qty{2000}{cL}$; $2000 \div 75 = 26$ botol
penuh (dengan sisa).
\textbf{3.} $26 \times 75 = \qty{1950}{cL}$, sehingga $2000 - 1950 =
\qty{50}{cL}$ tersisa.
\textbf{4.} $50 \div 25 = 2$ gelas, tepat.
\textbf{5.} $\qty{75}{cL} = \qty{0.75}{L}$, dan satu \omterm{def:g6:mass-and-volume:volume}{liter} \omterm{def:g4:weight-and-mass:weight}{beratnya}
kira-kira \qty{1000}{g}: jusnya kira-kira \qty{750}{g}.
\textbf{6.} $750 + 450 = \qty{1200}{g}$ --- yaitu \qty{1.2}{kg}.
\textbf{7.} $6 \times 1.2 = \qty{7.2}{kg}$ botol ditambah \qty{2}{kg}
peti: \qty{9.2}{kg}.
\textbf{8.} $200 \div 9.2$: $21$ peti berarti \qty{193.2}{kg}, $22$
peti berarti \qty{202.4}{kg} --- terlalu banyak. Mobil boksnya
mengangkut $21$ peti.
\textbf{9.} Tumpahannya adalah jus yang didesak kerikilnya: \omterm{def:g6:mass-and-volume:volume}{volume}
kerikilnya \qty{40}{cm^3}, menurut proposisi pendesakan air.
\textbf{10.} $1000 - 40 = \qty{960}{mL}$ jus tersisa.
\textbf{11.} Gelas ukurnya seharusnya memanjat dari \qty{100}{mL}
menjadi \qty{140}{mL} --- kenaikan \qty{40}{mL}, yang cocok dengan
tumpahannya. Kepalan tangan tidak \omterm{def:g6:measurement-in-science:measuring}{mengukur} apa pun; airnya berkata
benar.
\textbf{12.} Ruang \qty{40}{cm^3} itu, kalau diisi jus, akan memuat
kira-kira \qty{40}{g} jus --- kerikilnya memampatkan \qty{104}{g} ke
dalam ruang yang sama: zat kerikil lebih berat daripada zat jus untuk
ruang yang sama, bahkan lebih dari dua kali lipat. Bab \omterm{def:g3:measuring-mass:mass}{massa} jenis
akan memberi perbandingan ini nama dan bilangannya yang benar.
\end{solution}
