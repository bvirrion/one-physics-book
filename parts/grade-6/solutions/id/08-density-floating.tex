\section*{Bab \ref{ch:g6:density-floating} --- Massa Jenis: Mengapa Benda Terapung}

\begin{solution}{exo:g6:density-floating:1}
\omterm{def:g3:measuring-mass:mass}{Massa} satu \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik} zat itu, dalam \omterm{def:g3:measuring-mass:units}{gram}; \omterm{def:g6:density-floating:density}{massa jenis} air
adalah \qty{1}{g} tiap \unit{cm^3}. Aturannya: yang \omterm{def:g3:measuring-mass:mass}{massa} jenisnya
kurang daripada air \omterm{def:g1:floating-and-sinking:words}{terapung}, yang lebih besar \omterm{def:g1:floating-and-sinking:words}{tenggelam}.
\end{solution}

\begin{solution}{exo:g6:density-floating:2}
Kayu ek \omterm{def:g1:floating-and-sinking:words}{terapung}; batu \omterm{def:g1:floating-and-sinking:words}{tenggelam}; gabus \omterm{def:g1:floating-and-sinking:words}{terapung}; besi \omterm{def:g1:floating-and-sinking:words}{tenggelam}; es
\omterm{def:g1:floating-and-sinking:words}{terapung} --- dengan selisih setipis rambut.
\end{solution}

\begin{solution}{exo:g6:density-floating:3}
$120 \div 80 = 1.5$: \omterm{def:g3:measuring-mass:mass}{massa} jenisnya $1.5$ --- lebih besar daripada
air, jadi ia \omterm{def:g1:floating-and-sinking:words}{tenggelam}.
\end{solution}

\begin{solution}{exo:g6:density-floating:4}
$158 \div 20 = 7.9$: itulah tanda tangan besi.
\end{solution}

\begin{solution}{exo:g6:density-floating:5}
\omterm{def:g6:density-floating:density}{Massa jenis} menjadi milik zatnya, bukan milik kepingnya: tiap
\omterm{def:g6:mass-and-volume:volume}{sentimeter kubik} kayu ek membawa jatah \omterm{def:g3:measuring-mass:units}{gram} yang sama, entah serpih
entah balok. Hal itulah yang menjadikan \omterm{def:g6:density-floating:density}{massa jenis} kartu pengenal
bagi zat yang tidak diketahui.
\end{solution}

\begin{solution}{exo:g6:density-floating:6}
\omterm{def:g3:solids-liquids-gases:liquid}{Zat cair} bertumpuk menurut \omterm{def:g3:measuring-mass:mass}{massa} jenisnya, yang terpadat di dasarnya:
madu $1.4$ di bawah air $1.0$ di bawah minyak $0.9$. Anggurnya
($1.05$) \omterm{def:g1:floating-and-sinking:words}{tenggelam} menembus minyak dan air lalu bertumpu di lantai
madunya --- lebih padat daripada kedua lantai di atasnya, kurang padat
daripada madunya.
\end{solution}

\begin{solution}{exo:g6:density-floating:7}
Kebanyakan \omterm{def:g3:solids-liquids-gases:solid}{zat padat} lebih padat daripada \omterm{def:g3:solids-liquids-gases:liquid}{zat cairnya} sendiri dan
\omterm{def:g1:floating-and-sinking:words}{tenggelam} di dalamnya. Es \omterm{def:g1:floating-and-sinking:words}{terapung} karena air menggembung ketika
\omterm{def:g2:ice-water-steam:meltfreeze}{membeku} --- zat yang sama di dalam ruang sepersepuluh lebih banyak
memberi \omterm{def:g6:density-floating:density}{massa jenis} $0.9 < 1$.
\end{solution}

\begin{solution}{exo:g6:density-floating:8}
Kayu menandatangani kira-kira $0.6$: tiap \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik} batangnya
menawar lebih rendah daripada \qty{1}{g} milik air, sehingga batangnya
\omterm{def:g1:floating-and-sinking:words}{terapung} sebesar apa pun. Perunggu menandatangani dekat $9$: tiap
\omterm{def:g6:mass-and-volume:volume}{sentimeter kubik} koinnya menawar lebih tinggi daripada air, sehingga
ia \omterm{def:g1:floating-and-sinking:words}{tenggelam} sekecil apa pun. Ukurannya tidak pernah menentukan;
\omterm{def:g3:measuring-mass:mass}{massanya} tiap ruang selalu menentukan.
\end{solution}

\begin{solution}{exo:g6:density-floating:9}
$\qty{1}{L} = \qty{1000}{cm^3}$, sehingga $800 \div 1000 = 0.8$ \omterm{def:g3:measuring-mass:units}{gram}
tiap \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik}. Es pada $0.9$ \emph{lebih padat} daripada \omterm{def:g3:solids-liquids-gases:liquid}{zat
cair} ini: bongkahan esnya \omterm{def:g1:floating-and-sinking:words}{tenggelam} di dalamnya.
\end{solution}

\begin{solution}{exo:g6:density-floating:10}
Bungkusan yang \omterm{def:g1:floating-and-sinking:words}{terapung} adalah seluruh lambungnya: cangkang baja
\emph{ditambah} \omterm{def:g6:mass-and-volume:volume}{volume} \omterm{def:g2:air-around-us:air}{udara} besar yang dikurungnya. Kalau
dirata-ratakan atas seluruh ruang itu, \omterm{def:g6:density-floating:density}{massa jenis} bungkusannya jatuh
di bawah $1$, dan aturannya berkata \omterm{def:g1:floating-and-sinking:words}{terapung}. Sebuah lubang membiarkan
air menggantikan \omterm{def:g2:air-around-us:air}{udaranya}: \omterm{def:g6:density-floating:density}{massa jenis} bungkusannya memanjat menuju
\omterm{def:g6:density-floating:density}{massa jenis} bajanya sendiri, melewati $1$ --- dan aturannya, yang tidak
bergeming, berkata \omterm{def:g1:floating-and-sinking:words}{tenggelam}.
\end{solution}

\begin{solution}{exo:g6:density-floating:11}
Wasitnya berubah: terhadap $1.03$ milik air laut, tubuh perenang yang
\omterm{def:g3:measuring-mass:mass}{massa} jenisnya dekat $1$ berada dengan nyaman di sisi yang \omterm{def:g1:floating-and-sinking:words}{terapung},
sehingga laut membawamu lebih tinggi. Ketika masuk ke air tawar
($1.0$), kapalnya kehilangan selisih itu lalu turun lebih dalam: garis
airnya naik di lambungnya.
\end{solution}

\begin{solution}{exo:g6:density-floating:12}
Ketika setengah terbenam, bolanya duduk sedalam yang seharusnya untuk
bungkusan yang \omterm{def:g3:measuring-mass:mass}{massa} jenisnya separuh \omterm{def:g6:density-floating:density}{massa jenis} air: cangkang
ditambah \omterm{def:g2:air-around-us:air}{udara} rata-rata menjadi kira-kira $0.5$ \omterm{def:g3:measuring-mass:units}{gram} tiap \omterm{def:g6:mass-and-volume:volume}{sentimeter
kubik} --- \omterm{def:g2:air-around-us:air}{udara} tersembunyilah yang paling banyak meringankannya.
Selagi airnya merembes masuk menggantikan \omterm{def:g2:air-around-us:air}{udaranya}, \omterm{def:g6:density-floating:density}{massa jenis}
rata-rata bungkusannya memanjat, dan bolanya turun makin dalam; pada
saat rata-ratanya melewati $1.0$ milik air, aturan \omterm{def:g1:floating-and-sinking:words}{terapungnya}
berganti pihak dan bolanya pun turun.
\end{solution}

\begin{solution}{pb:g6:density-floating:1}
\textbf{1.} Tidak apa-apa: menukar emas dengan \emph{\omterm{def:g3:measuring-mass:mass}{massa}} perak yang
sama membuat \omterm{def:g3:levers-and-scales:scale}{neracanya} senang --- \omterm{def:g3:measuring-mass:mass}{massa} saja tidak dapat melihat
penukarannya.
\textbf{2.} $1930 \div 19.3 = \qty{100}{cm^3}$.
\textbf{3.} \qty{130}{cm^3} --- limpahannya adalah \omterm{def:g6:mass-and-volume:volume}{volume} mahkotanya,
menurut pendesakan air.
\textbf{4.} Mahkotanya memakai ruang \qty{30}{cm^3} lebih banyak
daripada yang seharusnya dipakai emas murni bermassa sama: ada sesuatu
yang lebih meruah daripada emas bersembunyi di dalamnya.
\textbf{5.} $1930 \div 130 \approx 14.8$.
\textbf{6.} Emas $19.3$ --- mahkotanya $14.8$ --- perak $10.5$:
mahkotanya jatuh di antara kedua tanda tangan itu.
\textbf{7.} ``Baginda, tiap \omterm{def:g6:mass-and-volume:volume}{sentimeter kubik} emas murni membawa
\qty{19.3}{g}; mahkota Baginda hanya membawa kira-kira $15$ --- ia
bukan emas murni.''
\textbf{8.} Jangkauannya memberi \omterm{def:g6:density-floating:density}{massa jenis} $1900 \div 130 \approx
14.6$ sampai $1960 \div 130 \approx 15.1$ --- sama sekali jauh dari
$19.3$: tidak ada kesalahan \omterm{def:g3:levers-and-scales:scale}{neraca} yang masuk akal yang menyelamatkan
si pandai emas.
\textbf{9.} Perak kurang padat: tiap \omterm{def:g3:measuring-mass:units}{gram} emas yang dicuri
dikembalikan sebagai segram perak yang memerlukan \emph{ruang lebih
banyak} --- penukaran \omterm{def:g3:measuring-mass:units}{gram} demi \omterm{def:g3:measuring-mass:units}{gram} mempertahankan \omterm{def:g3:measuring-mass:mass}{massanya} dan
menggembungkan \omterm{def:g6:mass-and-volume:volume}{volumenya}.
\textbf{10.} Bagian emasnya: $965 \div 19.3 = \qty{50.0}{cm^3}$;
bagian peraknya: $965 \div 10.5 \approx \qty{91.9}{cm^3}$.
\textbf{11.} Totalnya $\approx \qty{142}{cm^3}$ --- lebih banyak
daripada $130$ yang terukur: tebakan separuh-separuh itu memakai
terlalu banyak perak; pencuriannya yang sebenarnya lebih kecil (\omterm{def:g6:mass-and-volume:volume}{volume}
terukurnya duduk di antara $100$ dan $142$).
\textbf{12.} ``Terbukti: mahkotanya bukan emas murni. Caranya:
\omterm{def:g3:measuring-mass:mass}{massanya} dibagi atas \omterm{def:g6:mass-and-volume:volume}{volumenya} yang diukur dengan air memberi \omterm{def:g6:density-floating:density}{massa
jenis} yang jauh di bawah tanda tangan emas yang tidak berubah. Dan
mahkotanya sama sekali tidak perlu tergores: airnya membaca rahasianya
dari luar --- itulah kemenangannya.''
\end{solution}
