\section*{Bab \ref{ch:g6:sun-earth-moon} --- Sistem Matahari--Bumi--Bulan}

\begin{solution}{exo:g6:sun-earth-moon:1}
Bumi berputar pada \omterm{def:g6:sun-earth-moon:axisorbit}{sumbunya} --- itulah hari. Bumi mengorbit Matahari
--- itulah tahun. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} mengorbit Bumi --- itulah bulan.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:2}
Garis khayal yang melewati kedua kutubnya, yang menjadi poros putaran
Bumi. Ia condong kira-kira seperempat sudut siku-siku --- dan tetap
condong ke arah yang sama di seluruh \omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya}: kecondongan ditambah
kesetiaan itulah yang membuat \omterm{def:g3:sun-days-seasons:year}{musim}.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:3}
Putaran Bumi memakan $365$ hari seperempat; empat seperempat yang
ditabung menjadi hari tambahan tahun kabisat. Tanpanya kalendernya
akan melenceng menembus \omterm{def:g3:sun-days-seasons:year}{musim} --- kira-kira satu hari tiap empat
tahun, dan berminggu-minggu dalam seabad.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:4}
Kira-kira $30$ Bumi sampai ke \omterm{def:g2:moon-in-the-sky:moon}{Bulan}; kira-kira $109$ Bumi membentang
selebar wajah Matahari.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:5}
Tusuknya memerankan \omterm{def:g6:sun-earth-moon:axisorbit}{sumbunya}, dan \omterm{def:g6:sun-earth-moon:axisorbit}{sumbu} yang sebenarnya menjaga
kecondongannya tetap terhadap \omterm{def:g4:solar-system:star}{bintang}. Biarkan ia berayun --- misalnya
selalu condong ke arah lampunya --- dan modelnya menunjukkan \omterm{def:g3:sun-days-seasons:year}{musim}
panas tanpa akhir: \omterm{def:g3:sun-days-seasons:year}{musimnya} lenyap bersama kesetiaannya.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:6}
Yang berlawanan: ketika belahan bumi kita condong ke arah Matahari,
belahan yang lain pasti condong menjauh --- satu kemiringan tidak
mungkin menguntungkan kedua ujungnya sekaligus.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:7}
Ketika tegak lurus, \omterm{def:g1:light-and-shadows:source}{cahaya} senternya mendarat curam dan terpusat ---
bercak yang kecil, terang, dan menggigit. Ketika menyerong, \omterm{def:g1:light-and-shadows:source}{cahaya}
yang sama mengabur di sepanjang bentangan dinding yang panjang, tipis
dan lemah. Condong masuk memberi kita jenis yang curam dan terpusat.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:8}
\omterm{def:g6:sun-earth-moon:axisorbit}{Orbit} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} sendiri mengelilingi Bumi: selagi ia berputar, kita
memandang separuhnya yang tersinari matahari dari sudut yang
berubah-ubah --- dari \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} sampai purnama lalu kembali. Gerak ketiga
yang menjalankan pertunjukan fasenya.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:9}
Pertama: \omterm{def:g3:sun-days-seasons:year}{musim} dingin belahan utara terjadi selagi Bumi menunggangi
jarak yang sedikit \emph{lebih dekat} ke Matahari --- jaraknya
menunjuk ke arah yang keliru. Kedua: kedua belahan bumi mengalami
\omterm{def:g3:sun-days-seasons:year}{musim} yang berlawanan pada saat yang sama dan pada jarak yang sama ---
tidak ada jarak bersama yang dapat melakukan itu; hanya kemiringannya,
yang menguntungkan satu ujung dan merugikan ujung yang lain, yang
dapat.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:10}
Tanpa kecondongan, setiap perhentian putarannya tampak serupa:
Matahari memanjat ke tinggi tengah hari yang sama sepanjang tahun,
siang dan malam tetap sama panjang di mana-mana, dan tahunnya
kehilangan \omterm{def:g3:sun-days-seasons:year}{musimnya} --- satu \omterm{def:g3:sun-days-seasons:year}{musim} semi abadi yang panjang, busur \omterm{def:g3:sun-days-seasons:sundial}{jam
mataharinya} \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}, dan drama utama kalendernya dibatalkan.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:11}
Selagi \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menempuh putaran $27$ harinya, Bumi maju di sepanjang
\omterm{def:g6:sun-earth-moon:axisorbit}{orbitnya} sendiri --- sehingga arah Mataharinya bergeser, dan \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
memerlukan kira-kira dua setengah hari perjalanan tambahan untuk
menyusul kesejajaran yang sama dengan Matahari yang membuat purnama.
Putaran dan kesejajaran adalah dua garis akhir yang berbeda.
\end{solution}

\begin{solution}{exo:g6:sun-earth-moon:12}
Lingkaran harian sebuah kota kutub itu kecil dan, berkat kecondongan
bulan Juni, dapat terletak \emph{seluruhnya} di belahan yang tersinari:
kotanya berputar terus tanpa pernah meninggalkan \omterm{def:g1:light-and-shadows:source}{cahaya} siang ---
itulah matahari tengah malam. Pada bulan Desember kecondongan yang
sama memiringkan lingkaran kecilnya seluruhnya ke belahan yang gelap:
putaran demi putaran tanpa \omterm{def:g1:day-and-night:daynight}{matahari terbit}. Kemiringannya melakukan
dengan lembut pada tiap garis lintang apa yang dilakukannya secara
mutlak di kutub.
\end{solution}

\begin{solution}{pb:g6:sun-earth-moon:1}
\textbf{1.} \qty{12800}{km} tiap \omterm{def:g2:measuring-length:units}{sentimeter}.
\textbf{2.} Kira-kira \qty{0.25}{cm} --- sebutir merica (atau sebutir
biji-bijian).
\textbf{3.} $30 \times 1 = \qty{30}{cm}$ dari kelerengnya.
\textbf{4.} $109$ cm --- kira-kira satu \omterm{def:g2:measuring-length:units}{meter}: sebuah gelang hula atau
bola pantai yang besar.
\textbf{5.} $150000000 \div 12800 \approx \qty{11700}{cm}$ ---
kira-kira \qty{117}{m}.
\textbf{6.} Nyaris tidak muat: bola Mataharinya harus berdiri kira-kira
\qty{17}{m} di luar garis gawang yang jauh pada lapangan \qty{100}{m}.
\textbf{7.} Kira-kira $117$ sekon --- kasarnya dua menit.
\textbf{8.} Matahari: $109 \div 11700 \approx 0.009$; butiran \omterm{def:g2:moon-in-the-sky:moon}{Bulan}:
$0.25 \div 30 \approx 0.008$ --- hampir sama: keduanya tampak sebesar
itu juga dari kelerengnya. Begitu pula di langit kita --- dan ketika
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} lewat tepat di depan Matahari, ia menutupinya hampir dengan
tepat: itulah gerhana matahari total, alam yang memanfaatkan
kebetulannya.
\textbf{9.} Putaran kelerengnya: kira-kira $365$ sekon --- enam menit
berjalan kaki; putaran butirannya mengelilingi kelereng: kira-kira
$27$ sekon.
\textbf{10.} Satu putaran penuh tiap sekon --- dan pada kelereng
sesentimeter yang licin tanpa tanda apa pun, putarannya tidak
meninggalkan apa pun untuk ditangkap mata: melukis satu titik padanya
adalah perbaikan yang jujur.
\textbf{11.} Menyerong kira-kira seperempat sudut siku-siku dari tegak
--- dan dibawa mengelilingi seluruh putarannya sambil selalu menunjuk
ke sudut lapangan yang jauh dan sama, tidak pernah berbalik menghadap
bola Mataharinya.
\textbf{12.} Misalnya: modelnya menunjukkan jarak dan ukuran
\emph{bersama-sama}, dengan benar --- kekosongan yang luas itulah
kemenangannya. Yang tidak dapat ditunjukkannya sekaligus adalah
geraknya pada langkah yang sebenarnya (``setahun'' enam menit sudah
merupakan percepatan yang liar; kelereng yang sungguhan akan merayap)
--- dan tentu saja juga \omterm{def:g1:light-and-shadows:source}{cahaya} dan kehangatannya: tidak ada bola
selebar semeter yang menerangi lapangan olahraga sebagaimana Matahari
menerangi sebuah \omterm{def:g4:solar-system:planet}{planet}.
\end{solution}
