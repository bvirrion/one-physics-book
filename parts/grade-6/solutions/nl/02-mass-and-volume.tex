\section*{Hoofdstuk \ref{ch:g6:mass-and-volume} --- Massa en volume}

\begin{solution}{exo:g6:mass-and-volume:1}
\omterm{def:g6:mass-and-volume:volume}{Volume}: hoeveel ruimte iets inneemt --- \unit{cm^3} en
\omterm{def:g6:mass-and-volume:volume}{liter}. \omterm{def:g3:measuring-mass:mass}{Massa}: hoeveel stof het bevat --- \omterm{def:g3:measuring-mass:units}{gram} en
\omterm{def:g3:measuring-mass:units}{kilogram}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:2}
\qty{2000}{cm^3}; \ \qty{350}{cm^3}; \ \qty{1.5}{L}; \
\qty{500}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:3}
Op de tafel: een scheve cilinder maakt het oppervlak scheef. Op
ooghoogte: een hoog of laag oog verschuift het schijnbare streepje (de
schuine-blikfout). Onderkant van de \omterm{met:g6:mass-and-volume:cylinder}{meniscus}: de \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} klimt
tegen de wanden op, en alleen de onderkant van de kromming staat op
het echte \omterm{def:g6:mass-and-volume:volume}{volume}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:4}
$265 - 200 = 65$: het \omterm{def:g6:mass-and-volume:volume}{volume} van de pruim is \qty{65}{cm^3}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:5}
Het water vormt zich naar elke deuk en elke bult, dus is het
opzijgeduwde water precies gelijk aan het \omterm{def:g6:mass-and-volume:volume}{volume} van de kiezel ---
de verdringingspropositie. De liniaal heeft rechte randen nodig; het
water niet.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:6}
$5 \times 2 \times 1 = \qty{10}{cm^3}$. Controle: laat hem in een
cilinder zakken --- de aflezing moet met precies \qty{10}{mL} stijgen.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:7}
$745 - 280 = \qty{465}{g}$ rijst. De tarreerknop: zet de balans op nul
met de lege kom erop, en schep dan.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:8}
Gelijke \omterm{def:g3:measuring-mass:mass}{massa}'s --- elk één \omterm{def:g3:measuring-mass:units}{kilogram}. De veren nemen veel meer ruimte
in: een veel groter \omterm{def:g6:mass-and-volume:volume}{volume} voor dezelfde stof.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:9}
Laat de kurk met opzet \omterm{def:g1:floating-and-sinking:words}{zinken}: duw hem met een dun spiesje onder
water (lees de stijging af terwijl alleen de kurk onder is), of bind
hem aan een zwaar verzwaringsstuk --- meet eerst de stijging van het
verzwaringsstuk alleen, dan die van het paar samen, en trek het
aandeel van het verzwaringsstuk eraf.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:10}
Uitgeschonken: $3 \times 250 + 200 = \qty{950}{mL}$. De bewering kwam
\qty{50}{mL} tekort --- een twintigste \omterm{def:g6:mass-and-volume:volume}{liter} te weinig.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:11}
De stijging is $83.5 - 80.0 = \qty{3.5}{mL}$: het hele
\omterm{def:g6:mass-and-volume:volume}{volume} van de ketting is \qty{3.5}{cm^3}. Het water meet het totale
\omterm{def:g6:mass-and-volume:volume}{volume}, metaal, sluiting en al --- dus kan een bewering van
\qty{5}{cm^3} goud niet waar zijn: er is niet eens \qty{5}{cm^3}
ketting.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:12}
De liniaal belooft $4 \times 3 \times 2 = \qty{24}{cm^3}$. Laat het
blokje (aan een draadje, zonder plons, helemaal onder, zonder de
wanden te raken) in \qty{150}{mL} zakken: de voorspelling is een
stijging van \qty{150}{mL} naar \qty{174}{mL}. Steeg het water met
iets anders dan \qty{24}{mL} --- buiten het eerlijke stapje of twee
van de cilinder --- dan zou de verdringingspropositie weerlegd zijn.
(Dat is nooit gebeurd.)
\end{solution}

\begin{solution}{pb:g6:mass-and-volume:1}
\textbf{1.} $20 \times 1000 = \qty{20000}{cm^3}$.
\textbf{2.} $\qty{20}{L} = \qty{2000}{cL}$; $2000 \div 75 = 26$ volle
flessen (met een rest).
\textbf{3.} $26 \times 75 = \qty{1950}{cL}$, dus blijft er
$2000 - 1950 = \qty{50}{cL}$ over.
\textbf{4.} $50 \div 25 = 2$ glazen, precies.
\textbf{5.} $\qty{75}{cL} = \qty{0.75}{L}$, en een \omterm{def:g6:mass-and-volume:volume}{liter} weegt
ongeveer \qty{1000}{g}: het sap is ongeveer \qty{750}{g}.
\textbf{6.} $750 + 450 = \qty{1200}{g}$ --- \qty{1.2}{kg}.
\textbf{7.} $6 \times 1.2 = \qty{7.2}{kg}$ aan flessen plus
\qty{2}{kg} krat: \qty{9.2}{kg}.
\textbf{8.} $200 \div 9.2$: $21$ kratten zouden \qty{193.2}{kg} zijn,
$22$ zouden \qty{202.4}{kg} zijn --- te veel. Het busje neemt $21$
kratten mee.
\textbf{9.} Het gemorste sap is het sap dat de kiezel heeft
verdrongen: het \omterm{def:g6:mass-and-volume:volume}{volume} van de kiezel is \qty{40}{cm^3}, volgens de
verdringingspropositie.
\textbf{10.} $1000 - 40 = \qty{960}{mL}$ sap blijft over.
\textbf{11.} De cilinder moet van \qty{100}{mL} naar \qty{140}{mL}
klimmen --- een stijging van \qty{40}{mL}, die klopt met het gemorste
sap. Vuisten \omterm{def:g6:measurement-in-science:measuring}{meten} niets; water vertelt de waarheid.
\textbf{12.} Die \qty{40}{cm^3}, met sap gevuld, zouden ongeveer
\qty{40}{g} sap bevatten --- de kiezel propt \qty{104}{g} in dezelfde
ruimte: kiezelstof is voor gelijke ruimte zwaarder dan sapstof, meer
dan twee keer zo zwaar. Het hoofdstuk over dichtheid geeft deze
vergelijking haar eigen naam en getal.
\end{solution}
