\section*{Hoofdstuk \ref{ch:g6:density-floating} --- Dichtheid: waarom dingen drijven}

\begin{solution}{exo:g6:density-floating:1}
De \omterm{def:g3:measuring-mass:mass}{massa} van één \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} van de stof, in
\omterm{def:g3:measuring-mass:units}{gram}; die van water is \qty{1}{g} per \unit{cm^3}. De regel: minder
dicht dan water \omterm{def:g1:floating-and-sinking:words}{drijft}, dichter \omterm{def:g1:floating-and-sinking:words}{zinkt}.
\end{solution}

\begin{solution}{exo:g6:density-floating:2}
Eikenhout \omterm{def:g1:floating-and-sinking:words}{drijft}; steen \omterm{def:g1:floating-and-sinking:words}{zinkt}; kurk \omterm{def:g1:floating-and-sinking:words}{drijft}; ijzer
\omterm{def:g1:floating-and-sinking:words}{zinkt}; ijs \omterm{def:g1:floating-and-sinking:words}{drijft} --- met een haar verschil.
\end{solution}

\begin{solution}{exo:g6:density-floating:3}
$120 \div 80 = 1.5$: \omterm{def:g6:density-floating:density}{dichtheid} $1.5$ --- dichter dan water, dus
\omterm{def:g1:floating-and-sinking:words}{zinkt} het.
\end{solution}

\begin{solution}{exo:g6:density-floating:4}
$158 \div 20 = 7.9$: de handtekening van ijzer.
\end{solution}

\begin{solution}{exo:g6:density-floating:5}
\omterm{def:g6:density-floating:density}{Dichtheid} hoort bij de stof, niet bij het stuk: elke
\omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} eikenhout draagt hetzelfde aandeel \omterm{def:g3:measuring-mass:units}{gram},
splinter of balk. Dat maakt \omterm{def:g6:density-floating:density}{dichtheid} tot een identiteitskaart voor
onbekende stoffen.
\end{solution}

\begin{solution}{exo:g6:density-floating:6}
Vloeistoffen stapelen zich naar \omterm{def:g6:density-floating:density}{dichtheid}, de dichtste onderaan:
honing $1.4$ onder water $1.0$ onder olie $0.9$. De druif ($1.05$)
\omterm{def:g1:floating-and-sinking:words}{zinkt} door olie en water en komt op de honingvloer tot rust ---
dichter dan beide bovenverdiepingen, minder dicht dan honing.
\end{solution}

\begin{solution}{exo:g6:density-floating:7}
De meeste vaste stoffen zijn dichter dan hun eigen
\omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} en \omterm{def:g1:floating-and-sinking:words}{zinken} erin. IJs \omterm{def:g1:floating-and-sinking:words}{drijft} doordat water bij het
\omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen} zwelt --- dezelfde stof in een tiende meer ruimte geeft
\omterm{def:g6:density-floating:density}{dichtheid} $0.9 < 1$.
\end{solution}

\begin{solution}{exo:g6:density-floating:8}
Hout tekent ongeveer $0.6$: elke \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} van de stam biedt
onder de \qty{1}{g} van water, dus \omterm{def:g1:floating-and-sinking:words}{drijft} de stam, hoe enorm ook. Brons
tekent bijna $9$: elke \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} van het muntje biedt boven
water, dus \omterm{def:g1:floating-and-sinking:words}{zinkt} het, hoe klein ook. De grootte deed er nooit toe; de
\omterm{def:g3:measuring-mass:mass}{massa} per ruimte altijd.
\end{solution}

\begin{solution}{exo:g6:density-floating:9}
$\qty{1}{L} = \qty{1000}{cm^3}$, dus $800 \div 1000 = 0.8$
\omterm{def:g3:measuring-mass:units}{gram} per \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter}. IJs met $0.9$ is
\emph{dichter} dan deze \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}: het ijsblokje \omterm{def:g1:floating-and-sinking:words}{zinkt} erin.
\end{solution}

\begin{solution}{exo:g6:density-floating:10}
Het drijvende pakket is de hele romp: stalen schaal \emph{plus} het
enorme \omterm{def:g6:mass-and-volume:volume}{volume} \omterm{def:g2:air-around-us:air}{lucht} dat hij omsluit. Uitgemiddeld over al die ruimte
zakt de \omterm{def:g6:density-floating:density}{dichtheid} van het pakket onder $1$, en de regel zegt:
\omterm{def:g1:floating-and-sinking:words}{drijven}. Een gat laat water de \omterm{def:g2:air-around-us:air}{lucht} vervangen: de
\omterm{def:g6:density-floating:density}{dichtheid} van het pakket klimt naar die van staal zelf, passeert $1$
--- en de regel zegt onbewogen: \omterm{def:g1:floating-and-sinking:words}{zinken}.
\end{solution}

\begin{solution}{exo:g6:density-floating:11}
De scheidsrechter is veranderd: tegen de $1.03$ van zeewater staat het
lichaam van een zwemmer, met bijna $1$, comfortabel aan de drijfkant,
dus draagt de zee je hoger. In zoet water ($1.0$) verliest het schip
die marge en zakt het dieper: zijn waterlijn klimt langs de romp
omhoog.
\end{solution}

\begin{solution}{exo:g6:density-floating:12}
Half ondergedompeld ligt de bol precies zo diep in het water als een
pakket met de halve \omterm{def:g6:density-floating:density}{dichtheid} van water zou moeten: schaal plus
\omterm{def:g2:air-around-us:air}{lucht} middelen uit op ongeveer $0.5$ \omterm{def:g3:measuring-mass:units}{gram} per
\omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} --- de verborgen \omterm{def:g2:air-around-us:air}{lucht} doet het meeste
lichtmaakwerk. Terwijl er water naar binnen lekt vervangt dat de
\omterm{def:g2:air-around-us:air}{lucht}, klimt de gemiddelde \omterm{def:g6:density-floating:density}{dichtheid} van het pakket en zakt de bol
dieper en dieper; op het moment dat het gemiddelde de $1.0$ van water
passeert, wisselt de drijfregel van kant en gaat de bol omlaag.
\end{solution}

\begin{solution}{pb:g6:density-floating:1}
\textbf{1.} Niets: goud ruilen voor een gelijke \emph{\omterm{def:g3:measuring-mass:mass}{massa}} zilver
houdt de balans tevreden --- \omterm{def:g3:measuring-mass:mass}{massa} alleen kan de omruil niet zien.
\textbf{2.} $1930 \div 19.3 = \qty{100}{cm^3}$.
\textbf{3.} \qty{130}{cm^3} --- de overloop is het \omterm{def:g6:mass-and-volume:volume}{volume} van de
kroon, door verdringing.
\textbf{4.} De kroon neemt \qty{30}{cm^3} meer ruimte in dan zuiver
goud van haar \omterm{def:g3:measuring-mass:mass}{massa} zou moeten: er zit iets omvangrijkers dan goud
verstopt.
\textbf{5.} $1930 \div 130 \approx 14.8$.
\textbf{6.} Goud $19.3$ --- kroon $14.8$ --- zilver $10.5$: de kroon
valt tussen de twee handtekeningen.
\textbf{7.} ``Sire, elke \omterm{def:g6:mass-and-volume:volume}{kubieke centimeter} zuiver goud draagt
\qty{19.3}{g}; uw kroon draagt er maar ongeveer $15$ --- zuiver goud
is ze niet.''
\textbf{8.} Het bereik geeft dichtheden van
$1900 \div 130 \approx 14.6$ tot $1960 \div 130 \approx 15.1$ ---
nergens in de buurt van $19.3$: geen geloofwaardige balansfout redt de
goudsmid.
\textbf{9.} Zilver is minder dicht: elke gestolen \omterm{def:g3:measuring-mass:units}{gram} goud gaf een
\omterm{def:g3:measuring-mass:units}{gram} zilver terug dat \emph{meer ruimte} nodig heeft --- omruilen
\omterm{def:g3:measuring-mass:units}{gram} voor \omterm{def:g3:measuring-mass:units}{gram} behoudt de \omterm{def:g3:measuring-mass:mass}{massa} en blaast het \omterm{def:g6:mass-and-volume:volume}{volume} op.
\textbf{10.} Gouden deel: $965 \div 19.3 = \qty{50.0}{cm^3}$;
zilveren deel: $965 \div 10.5 \approx \qty{91.9}{cm^3}$.
\textbf{11.} Totaal $\approx \qty{142}{cm^3}$ --- meer dan de gemeten
$130$: de gok half-om-half gebruikte te veel zilver; de werkelijke
diefstal was kleiner (het gemeten \omterm{def:g6:mass-and-volume:volume}{volume} ligt tussen $100$ en $142$).
\textbf{12.} ``Bewezen: de kroon is geen zuiver goud. Methode: haar
\omterm{def:g3:measuring-mass:mass}{massa} verdeeld over haar met water gemeten \omterm{def:g6:mass-and-volume:volume}{volume} geeft een
\omterm{def:g6:density-floating:density}{dichtheid} die ver onder de onveranderlijke handtekening van goud
ligt. En de kroon had geen krasje nodig: het water las haar geheim van
buitenaf --- dat is de triomf.''
\end{solution}
