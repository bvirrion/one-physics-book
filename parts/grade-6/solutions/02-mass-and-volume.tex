\section*{Chapter \ref{ch:g6:mass-and-volume} --- Mass and Volume}

\begin{solution}{exo:g6:mass-and-volume:1}
\omterm{def:g6:mass-and-volume:volume}{Volume}: how much room a thing takes --- \unit{cm^3} and \omterm{def:g6:mass-and-volume:volume}{litres}.
\omterm{def:g3:measuring-mass:mass}{Mass}: how much stuff it holds --- \omterm{def:g3:measuring-mass:units}{grams} and \omterm{def:g3:measuring-mass:units}{kilograms}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:2}
\qty{2000}{cm^3}; \ \qty{350}{cm^3}; \ \qty{1.5}{L}; \
\qty{500}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:3}
On the table: a tilted cylinder tilts the surface. Eye level: a
high or low eye shifts the apparent mark (the slanted-view
mistake). Bottom of the \omterm{met:g6:mass-and-volume:cylinder}{meniscus}: the \omterm{def:g3:solids-liquids-gases:liquid}{liquid} climbs the walls, and
only the curve's bottom stands at the true \omterm{def:g6:mass-and-volume:volume}{volume}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:4}
$265 - 200 = 65$: the plum's \omterm{def:g6:mass-and-volume:volume}{volume} is \qty{65}{cm^3}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:5}
The water molds itself to every dent and bump, so the pushed-aside
water equals the pebble's \omterm{def:g6:mass-and-volume:volume}{volume} exactly --- the displacement
proposition. The ruler needs straight edges; the water needs none.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:6}
$5 \times 2 \times 1 = \qty{10}{cm^3}$. Check: lower it into a
cylinder --- the reading should rise by exactly \qty{10}{mL}.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:7}
$745 - 280 = \qty{465}{g}$ of rice. The tare button: zero the
balance with the empty bowl on it, then pour.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:8}
Equal masses --- one \omterm{def:g3:measuring-mass:units}{kilogram} each. The feathers take far more
room: much greater \omterm{def:g6:mass-and-volume:volume}{volume} for the same stuff.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:9}
\omterm{def:g1:floating-and-sinking:words}{Sink} the cork on purpose: push it under with a thin skewer (read
the rise while only the cork is submerged), or tie it to a heavy
sinker --- \omterm{def:g6:measurement-in-science:measuring}{measure} the sinker's rise alone first, then the pair
together, and subtract the sinker's share.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:10}
Poured out: $3 \times 250 + 200 = \qty{950}{mL}$. The claim missed
by \qty{50}{mL} --- a twentieth of a \omterm{def:g6:mass-and-volume:volume}{litre} short.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:11}
The rise is $83.5 - 80.0 = \qty{3.5}{mL}$: the necklace's whole
\omterm{def:g6:mass-and-volume:volume}{volume} is \qty{3.5}{cm^3}. The water \omterm{def:g6:measurement-in-science:measuring}{measures} total \omterm{def:g6:mass-and-volume:volume}{volume}, metal
and clasp and all --- so a claim of \qty{5}{cm^3} of gold cannot be
true: there is not even \qty{5}{cm^3} of necklace.
\end{solution}

\begin{solution}{exo:g6:mass-and-volume:12}
The ruler promises $4 \times 3 \times 2 = \qty{24}{cm^3}$. Lower
the block (thread, no splash, fully under, not touching the walls)
into \qty{150}{mL}: the prediction is a rise from \qty{150}{mL} to
\qty{174}{mL}. If the water rose by anything other than
\qty{24}{mL} --- outside the cylinder's honest step or two --- the
displacement proposition would stand refuted. (It never has been.)
\end{solution}

\begin{solution}{pb:g6:mass-and-volume:1}
\textbf{1.} $20 \times 1000 = \qty{20000}{cm^3}$.
\textbf{2.} $\qty{20}{L} = \qty{2000}{cL}$; $2000 \div 75 = 26$
full bottles (with remainder).
\textbf{3.} $26 \times 75 = \qty{1950}{cL}$, so $2000 - 1950 =
\qty{50}{cL}$ remain.
\textbf{4.} $50 \div 25 = 2$ glasses, exactly.
\textbf{5.} $\qty{75}{cL} = \qty{0.75}{L}$, and a \omterm{def:g6:mass-and-volume:volume}{litre} weighs
about \qty{1000}{g}: the juice is about \qty{750}{g}.
\textbf{6.} $750 + 450 = \qty{1200}{g}$ --- \qty{1.2}{kg}.
\textbf{7.} $6 \times 1.2 = \qty{7.2}{kg}$ of bottles plus
\qty{2}{kg} of crate: \qty{9.2}{kg}.
\textbf{8.} $200 \div 9.2$: $21$ crates would be
\qty{193.2}{kg}, $22$ would be \qty{202.4}{kg} --- too much. The
van takes $21$ crates.
\textbf{9.} The spill is the juice the pebble displaced: the
pebble's \omterm{def:g6:mass-and-volume:volume}{volume} is \qty{40}{cm^3}, by the displacement
proposition.
\textbf{10.} $1000 - 40 = \qty{960}{mL}$ of juice remain.
\textbf{11.} The cylinder should climb from \qty{100}{mL} to
\qty{140}{mL} --- a \qty{40}{mL} rise, matching the spill. Fists
\omterm{def:g6:measurement-in-science:measuring}{measure} nothing; water tells the truth.
\textbf{12.} Those \qty{40}{cm^3}, filled with juice, would hold
about \qty{40}{g} of juice --- the pebble packs \qty{104}{g} into
the same room: pebble-stuff is heavier than juice-stuff for equal
room, more than twice over. The density chapter will give this
comparison its proper name and number.
\end{solution}
