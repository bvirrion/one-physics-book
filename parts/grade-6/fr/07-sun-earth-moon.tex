\chapter{Le système Soleil--Terre--Lune}\label{ch:g6:sun-earth-moon}

Trois mondes mènent notre ciel~: le Soleil qui donne le jour, la
Terre qui nous porte, la \omterm{def:g2:moon-in-the-sky:moon}{Lune} qui nous tient compagnie. Tu les
connais chacun~; nous dressons maintenant la carte du \emph{système}
--- qui tourne autour de qui, à quelle \omterm{def:g5:speed-distance-time:speed}{vitesse}, à quelle distance ---
et nous payons une vieille dette~: le secret promis des \omterm{def:g3:sun-days-seasons:year}{saisons} se
trouve dans ce chapitre.

\section{Trois mouvements, trois horloges}

\begin{definition}[Axe et orbite]\label{def:g6:sun-earth-moon:axisorbit}
L'\emph{axe}\index{axe} de la Terre est la ligne imaginaire passant
par ses pôles, autour de laquelle elle tourne sur elle-même, comme la
tige d'une toupie. Une \emph{orbite}\index{orbite} est le chemin
fermé qu'un astre parcourt autour d'un autre --- un cercle ou presque
pour la Terre autour du Soleil, et pour la \omterm{def:g2:moon-in-the-sky:moon}{Lune} autour de la Terre.
\end{definition}

\begin{proposition}[Les trois mouvements]\label{prop:g6:sun-earth-moon:motions}
Tout le calendrier tient à trois mouvements réguliers~:
\begin{enumerate}
  \item la Terre \emph{tourne} sur son \omterm{def:g6:sun-earth-moon:axisorbit}{axe}~: un tour en $24$
        \omterm{ex:g2:measuring-time:clock}{heures} --- le jour, qui fait défiler le Soleil et les
        \omterm{def:g4:solar-system:star}{étoiles} dans notre ciel~;
  \item la Terre \emph{parcourt son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite}} autour du Soleil~: un
        tour en environ $365$ jours et un quart --- l'année, qui
        fait monter et descendre l'arc du Soleil au fil des
        \omterm{def:g3:sun-days-seasons:year}{saisons}~;
  \item la \omterm{def:g2:moon-in-the-sky:moon}{Lune} \emph{parcourt son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite}} autour de la Terre~: un
        tour en environ un mois --- et, comme sa moitié ensoleillée
        nous fait face tour à tour, elle fait défiler ses phases
        dans ton vieux journal de la \omterm{def:g2:moon-in-the-sky:moon}{Lune}.
\end{enumerate}
Trois mouvements, trois horloges~: le jour, le mois, l'année --- les
plus vieux garde-temps de l'humanité, tous encore en marche.
\end{proposition}

\begin{example}[Le quart de jour et l'année
bissextile]\label{ex:g6:sun-earth-moon:leap}
Le tour de la Terre dure $365$ jours \emph{et environ un quart} ---
la nature ne doit aucune fidélité aux nombres ronds. Les calendriers
paient la dette en épargnant les quarts~: quatre années de quarts de
jour font un jour entier, ajouté au calendrier tous les quatre ans
--- l'\emph{année bissextile} de $366$ jours, avec son jour
supplémentaire à la fin de février. Néglige cette comptabilité
pendant un siècle et le calendrier dérive de plusieurs semaines par
rapport aux \omterm{def:g3:sun-days-seasons:year}{saisons} --- comme des calendriers plus anciens l'ont
douloureusement découvert.
\end{example}

\begin{example}[Le système, à l'échelle]\label{ex:g6:sun-earth-moon:scale}
Des nombres pour le carnet~: la Terre mesure environ \qty{12800}{km}
de diamètre~; la \omterm{def:g2:moon-in-the-sky:moon}{Lune}, environ le quart de cela, tourne à environ
\qty{384000}{km} de distance --- trente Terres posées côte à côte. Le
Soleil fait environ $109$ Terres de diamètre et se tient à $150$
millions de kilomètres. Réduis la Terre à une bille d'un \omterm{def:g2:measuring-length:units}{centimètre}
et la maquette dit~: la \omterm{def:g2:moon-in-the-sky:moon}{Lune}, un grain de poivre à $30$ \omterm{def:g2:measuring-length:units}{centimètres}~;
le Soleil, une boule de plus d'un \omterm{def:g2:measuring-length:units}{mètre}, à plus de cent \omterm{def:g2:measuring-length:units}{mètres} au
bout de la rue. Tout notre voisinage est surtout du vide --- comme la
maquette du \omterm{def:g4:solar-system:family}{Système solaire} dans le parc le murmurait déjà.
\end{example}

\section{La dette payée~: pourquoi les saisons}

\begin{proposition}[L'inclinaison fait les saisons]\label{prop:g6:sun-earth-moon:tilt}
L'\omterm{def:g6:sun-earth-moon:axisorbit}{axe} de la Terre n'est pas droit sur son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite}~: il penche,
d'environ un quart d'angle droit --- et il continue de pencher
\emph{dans le même sens} tout au long du tour de l'année, comme une
fidèle \omterm{def:g4:magnets-and-compass:compass}{boussole} des \omterm{def:g4:solar-system:star}{étoiles}. Pendant la moitié du tour, notre moitié
du monde penche donc \emph{vers} le Soleil~: le Soleil monte haut,
les journées s'allongent --- c'est l'été. Un demi-tour plus tard, la
même inclinaison inchangée nous pointe à l'\emph{opposé}~: Soleil
bas, journées courtes --- c'est l'hiver. Les \omterm{def:g3:sun-days-seasons:year}{saisons} ne sont ni le
Soleil qui change, ni la distance~: c'est une \omterm{def:g4:solar-system:planet}{planète} inclinée qui
promène son inclinaison inchangée autour de son \omterm{def:g4:solar-system:star}{étoile}.
\end{proposition}

\begin{method}[Le globe et la lampe]\label{met:g6:sun-earth-moon:globe}
Voir toute la machine sur une table~:
\begin{enumerate}
  \item poser une lampe au milieu de la table pour jouer le Soleil,
        et porter lentement un globe incliné (ou une balle sur une
        brochette en biais) tout autour, \emph{en gardant la
        brochette pointée vers le même coin de la pièce} pendant
        tout le trajet~;
  \item à chaque quart de tour, s'arrêter et regarder la latitude
        de chez toi~: observer quelle part de son cercle de
        rotation quotidien se trouve dans la \omterm{def:g1:light-and-shadows:source}{lumière}~;
  \item d'un côté de la lampe, ton hémisphère se prélasse, penché
        vers elle --- de longs arcs éclairés, une lampe haute~:
        c'est l'été~; à l'opposé, penché à l'écart --- des arcs
        courts, une lampe basse~: c'est l'hiver~;
  \item aux deux étapes intermédiaires, aucun pôle ne penche vers
        la lampe~: le jour et la nuit se partagent également ---
        c'est le printemps et l'automne.
\end{enumerate}
La seule discipline qui fait marcher le tout~: ne jamais laisser la
brochette pivoter --- la fidélité de l'\omterm{def:g6:sun-earth-moon:axisorbit}{axe} \emph{est} la cause des
\omterm{def:g3:sun-days-seasons:year}{saisons}.
\end{method}

\begin{example}[Les vieux indices, expliqués]\label{ex:g6:sun-earth-moon:clues}
Chaque observation de tes années de \omterm{def:g3:sun-days-seasons:sundial}{cadran solaire} rend maintenant
des comptes à l'inclinaison. L'arc haut de l'été~: notre hémisphère
penche vers le Soleil, qui se tient donc plus haut à midi. L'arc
trapu de l'hiver et ses longues \omterm{def:g1:light-and-shadows:shadow}{ombres}~: penché à l'écart, le Soleil
rase l'horizon. Les \omterm{def:g3:sun-days-seasons:year}{saisons} opposées de part et d'autre de
l'équateur --- des vacances de plage en décembre au sud pendant que
le nord déblaie la neige~: quand un hémisphère penche vers le Soleil,
l'autre doit pencher à l'opposé. Même les journées égales du
printemps et de l'automne découlent de la promenade du globe autour
de la lampe. Une inclinaison d'un quart d'angle droit mène tout le
spectacle.
\end{example}

\begin{remark}[Pas la distance --- et la preuve]\label{rem:g6:sun-earth-moon:notdistance}
L'\omterm{def:g6:sun-earth-moon:axisorbit}{orbite} de la Terre est si proche d'un cercle que la distance au
Soleil change peu --- et, délicieusement, nous passons un peu
\emph{plus près} du Soleil pendant l'hiver de l'hémisphère nord. Ce
qui intéresse les \omterm{def:g3:sun-days-seasons:year}{saisons}, c'est l'\emph{inclinaison} de la \omterm{def:g1:light-and-shadows:source}{lumière},
et non la longueur du trajet~: penchés vers le Soleil, ses rayons
frappent le sol de plein fouet et concentrés~; penchés à l'écart, la
même \omterm{def:g1:light-and-shadows:source}{lumière} s'étale et se dilue sur le sol. Une lampe de poche
tenue de face, puis en biais, contre un mur montre la différence en
une seconde --- une \omterm{def:g1:light-and-shadows:source}{lumière} droite mord, une \omterm{def:g1:light-and-shadows:source}{lumière} rasante
réchauffe à peine.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:1}
Nomme les trois mouvements du système et l'horloge que chacun
remonte.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:2}
Qu'est-ce que l'\omterm{def:g6:sun-earth-moon:axisorbit}{axe} de la Terre~? Quel fait, à un quart d'angle
droit près, et quelle fidélité font ensemble les \omterm{def:g3:sun-days-seasons:year}{saisons}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:3}
Pourquoi le calendrier ajoute-t-il un jour tous les quatre ans~? Que
se décalerait-il s'il ne le faisait jamais~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:4}
Combien de Terres, posées côte à côte, atteignent à peu près la
\omterm{def:g2:moon-in-the-sky:moon}{Lune}~? Combien de Terres couvrent le disque du Soleil~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:5}
Dans la promenade du globe autour de la lampe, pourquoi la brochette
doit-elle rester pointée vers le même coin de la pièce~? Qu'est-ce
qui se casse si elle pivote~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:6}
Quand c'est l'été ici, quelle \omterm{def:g3:sun-days-seasons:year}{saison} est-ce pour l'hémisphère situé
de l'autre côté de l'équateur, et pourquoi cela doit-il être ainsi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:7}
Sers-toi de l'\omterm{prop:g5:mirrors-reflection:image}{image} de la lampe de poche contre le mur de
\cref{rem:g6:sun-earth-moon:notdistance} pour expliquer pourquoi
pencher vers le Soleil nous réchauffe davantage.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:sun-earth-moon:8}
Une vieille connaissance revient~: pourquoi la \omterm{def:g2:moon-in-the-sky:moon}{Lune} montre-t-elle des
phases au fil du mois~? Lequel des trois mouvements mène ce
spectacle~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:9}
Un oncle têtu insiste~: «~L'hiver, c'est quand la Terre est la plus
éloignée du Soleil.~» Rassemble deux faits de ce chapitre qui,
ensemble, le mettent en déroute.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:10}
Suppose que l'\omterm{def:g6:sun-earth-moon:axisorbit}{axe} de la Terre se tienne parfaitement droit sur son
\omterm{def:g6:sun-earth-moon:axisorbit}{orbite} --- sans la moindre inclinaison. Décris une année~:
qu'advient-il des \omterm{def:g3:sun-days-seasons:year}{saisons}, de la hauteur du Soleil à midi, de la
\omterm{def:g2:measuring-time:duration}{durée} des journées~? (Promène en pensée le globe droit autour de la
lampe.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:11}
Le tour de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} dure environ $27$ jours, et pourtant il s'écoule
environ $29.5$ jours d'une \omterm{ex:g2:moon-in-the-sky:shapes}{pleine lune} à la suivante. Propose la
résolution --- qu'est-ce qui a bougé d'autre pendant que la \omterm{def:g2:moon-in-the-sky:moon}{Lune}
faisait son tour~? (Esquisse la Terre qui avance sur son \omterm{def:g6:sun-earth-moon:axisorbit}{orbite}
pendant un tour de \omterm{def:g2:moon-in-the-sky:moon}{Lune}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:sun-earth-moon:12}
Au cœur de l'été, les régions proches du pôle Nord gardent le Soleil
au-dessus de l'horizon à minuit --- le fameux «~soleil de minuit~»
--- et, au cœur de l'hiver, elles attendent des semaines un \omterm{def:g1:day-and-night:daynight}{lever du
Soleil}. Explique les deux merveilles avec les cercles de rotation
inclinés de \cref{met:g6:sun-earth-moon:globe}~: que fait
l'inclinaison au cercle quotidien d'une ville polaire en juin, et en
décembre~?
\end{exercise}

\section{Problème~: la maquette du terrain de sport}

\begin{problem}[{Problème du week-end --- la classe construit le
système Soleil--Terre--Lune sur le terrain de sport~; billes, grains
de poivre et une très longue marche~; l'inclinaison rejoint la
maquette}]\label{pb:g6:sun-earth-moon:1}
La classe construit le système à l'échelle~: la Terre est une bille
de \qty{1}{cm} de diamètre. Nombres vrais bien commodes~: la Terre
\qty{12800}{km} de diamètre~; la \omterm{def:g2:moon-in-the-sky:moon}{Lune} \qty{3200}{km} de diamètre, en
\omterm{def:g6:sun-earth-moon:axisorbit}{orbite} à \qty{384000}{km}~; le Soleil $109$ Terres de diamètre, à
$150$ millions de kilomètres.

\medskip\noindent\textbf{Partie I --- distribuer les rôles.}
\begin{enumerate}
  \item Dans la maquette, \qty{1}{cm} représente combien de
        kilomètres~?
  \item La \omterm{def:g2:moon-in-the-sky:moon}{Lune} fait environ le quart de la largeur de la Terre.
        Quelle est la largeur de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} de la maquette --- et quel
        objet de cuisine la joue bien~?
  \item La vraie distance de la \omterm{def:g2:moon-in-the-sky:moon}{Lune} vaut $30$ largeurs de Terre. À
        quelle distance de la bille se place la \omterm{def:g2:moon-in-the-sky:moon}{Lune} de la
        maquette~?
  \item Le Soleil de la maquette fait $109$ billes de diamètre.
        Cela fait combien de \omterm{def:g2:measuring-length:units}{centimètres} --- et quel objet de la
        cour de récréation a à peu près cette taille~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- la longue marche.}
\begin{enumerate}[resume]
  \item La distance du Soleil vaut $150$ millions de kilomètres.
        Dans la maquette --- en divisant par la réponse à la
        question 1 --- cela fait combien de \omterm{def:g2:measuring-length:units}{centimètres}, et combien
        de \omterm{def:g2:measuring-length:units}{mètres}~?
  \item La maquette tiendra-t-elle sur un terrain de sport de
        \qty{100}{m}~? Où doit se placer la boule-Soleil~?
  \item Un élève parcourt le trajet Terre--Soleil à $1$ \omterm{def:g2:measuring-length:units}{mètre} par
        seconde. Combien de minutes dure environ la marche~?
  \item Debout près de la bille et regardant la boule-Soleil d'un
        \omterm{def:g2:measuring-length:units}{mètre} à $117$ \omterm{def:g2:measuring-length:units}{mètres} de là, la classe est surprise~: elle
        paraît à peu près de la même taille que le \omterm{def:g2:moon-in-the-sky:moon}{grain-Lune} d'un
        quart de \omterm{def:g2:measuring-length:units}{centimètre} placé à $30$ \omterm{def:g2:measuring-length:units}{centimètres} de la bille.
        Vérifie les deux rapports --- et relie cette coïncidence à
        quelque chose qui se produit, rarement et glorieusement,
        dans le vrai ciel.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- la mettre en marche.}
\begin{enumerate}[resume]
  \item Pour animer honnêtement la maquette, combien de temps la
        bille doit-elle mettre à faire le tour de la boule-Soleil,
        et le pois à faire le tour de la bille, si un vrai jour est
        joué par une seconde~? (Compte grossièrement~: une année
        pour $365$ jours, un tour de \omterm{def:g2:moon-in-the-sky:moon}{Lune} pour $27$ jours.)
  \item La bille doit aussi tourner sur elle-même. À raison d'un
        vrai jour par seconde, cela fait quelle \omterm{def:g5:speed-distance-time:speed}{vitesse} dans la
        maquette --- et pourquoi personne ne le verra-t-il sur une
        bille de \qty{1}{cm}~?
  \item Une brochette traversant la bille joue l'\omterm{def:g6:sun-earth-moon:axisorbit}{axe}. Selon quelle
        inclinaison, et pointée comment, la brochette doit-elle
        être portée tout au long du tour~?
  \item Rédige l'avertissement honnête de la maquette~: nomme une
        chose que la maquette du terrain de sport montre
        fidèlement, et une chose qu'aucune maquette de la taille
        d'un terrain ne peut montrer en même temps (les tailles et
        les distances ensemble~? les mouvements~? la \omterm{def:g1:light-and-shadows:source}{lumière}~?).
        Justifie avec tes nombres.
\end{enumerate}
\end{problem}
