\chapter{La propagation rectiligne de la lumière}\label{ch:g6:light-propagation}

«~La \omterm{def:g1:light-and-shadows:source}{lumière} se propage en ligne droite~» t'a servi pendant deux ans
--- pour éclairer les \omterm{def:g1:light-and-shadows:shadow}{ombres}, viser avec des \omterm{def:g5:mirrors-reflection:reflection}{miroirs}, aligner trois
cartons et une bougie. Cette année, la vieille loi reçoit son nom
officiel, ses règles de dessin, et sa conséquence la plus magique~:
un appareil photo sans objectif, sans verre, sans pièce mobile ---
rien qu'un trou.

\section{La loi, énoncée comme il faut}

\begin{proposition}[Propagation rectiligne de la
lumière]\label{prop:g6:light-propagation:law}
Dans l'\omterm{def:g2:air-around-us:air}{air} limpide --- et dans tout milieu limpide et homogène,
comme l'eau immobile ou le verre --- la \omterm{def:g1:light-and-shadows:source}{lumière} se propage en ligne
droite. C'est la loi de la \emph{propagation
rectiligne}\index{propagation rectiligne}~: \emph{rectiligne} est
simplement la façon savante de dire «~en ligne droite~», et
\emph{propagation} nomme le voyage de la \omterm{def:g1:light-and-shadows:source}{lumière} d'un endroit à un
autre.
\end{proposition}

\begin{definition}[Rayon et faisceau]\label{def:g6:light-propagation:ray}
Un \emph{rayon lumineux}\index{rayon lumineux} est le dessin, par le
physicien, d'un seul fil de \omterm{def:g1:light-and-shadows:source}{lumière}~: une ligne droite fléchée dans
son sens de propagation. Les vraies lampes déversent ensemble
d'innombrables fils~: un paquet de rayons est un
\emph{faisceau}\index{faisceau}. Trois formes de faisceau couvrent la
pratique~: \emph{divergent} (des rayons qui s'écartent d'un point ---
n'importe quelle petite lampe), \emph{parallèle} (des rayons côte à
côte, qui ne se rencontrent ni ne s'écartent --- la \omterm{def:g1:light-and-shadows:source}{lumière} du
Soleil, qui a voyagé si loin), et \emph{convergent} (des rayons qui
se referment vers un point --- de la \omterm{def:g1:light-and-shadows:source}{lumière} rassemblée par des
instruments que tu rencontreras dans deux ans).
\end{definition}

\begin{example}[La loi à l'œuvre, partout]\label{ex:g6:light-propagation:sightings}
Les charpentiers visent le long de l'arête d'une planche pour la
juger --- l'œil, l'arête et l'extrémité lointaine doivent être
alignés, et la rectitude de la \omterm{def:g1:light-and-shadows:source}{lumière} est le serment du juge. Les
géomètres plantent des jalons en ligne à l'œil~; les archers et les
photographes «~alignent leur tir~»~; et l'expérience des trois
cartons d'il y a deux ans a maintenant un nom officiel~: c'est un
test de la \omterm{prop:g6:light-propagation:law}{propagation rectiligne}. Chacun de ces métiers échange une
ligne droite de \omterm{def:g1:light-and-shadows:source}{lumière} contre une ligne droite de matière.
\end{example}

\section{Les ombres, construites}

\begin{method}[Construire une ombre à la règle]\label{met:g6:light-propagation:shadow}
Pour une petite lampe $S$, une balle, et un écran derrière elle~:
\begin{enumerate}
  \item tracer le rayon issu de $S$ qui frôle tout juste le bord
        supérieur de la balle, et le prolonger en ligne droite
        jusqu'à l'écran~;
  \item de même pour le rayon qui frôle le bord inférieur (et,
        dans l'espace, tout autour)~;
  \item entre les deux points d'arrivée sur l'écran se trouve
        l'\emph{ombre}\index{ombre}~: la zone qu'aucun rayon
        rectiligne issu de $S$ ne peut atteindre~;
  \item l'espace entre la balle et l'écran, limité par les rayons
        frôlants, est le \emph{cône d'\omterm{def:g1:light-and-shadows:shadow}{ombre}}~: un œil placé à
        l'intérieur ne peut pas voir la lampe.
\end{enumerate}
Aucune approximation nulle part~: deux traits de règle fixent la
taille et la place exactes de l'\omterm{def:g1:light-and-shadows:shadow}{ombre}.
\end{method}

\begin{example}[Plus grande, plus petite, plus
nette]\label{ex:g6:light-propagation:sizes}
La construction prédit ce que les jeux de jouet et de lampe de poche
de ta première année trouvaient en jouant~: fais glisser la balle
vers la lampe et les rayons frôlants s'écartent davantage --- une
\omterm{def:g1:light-and-shadows:shadow}{ombre} géante~; fais-la glisser vers l'écran et ils arrivent plus
serrés --- une \omterm{def:g1:light-and-shadows:shadow}{ombre} à la taille réelle. Et la prédiction est
exacte~: avec la balle à mi-chemin de la lampe et de l'écran, les
rayons frôlants s'écartent de deux fois la largeur de la balle ---
une \omterm{def:g1:light-and-shadows:shadow}{ombre} exactement deux fois plus grande que la balle. De la
proportionnalité, dessinée en \omterm{def:g1:light-and-shadows:source}{lumière}.
\end{example}

\section{L'appareil photo sans objectif}

\begin{method}[Construire une chambre à sténopé]\label{met:g6:light-propagation:pinhole}
\begin{enumerate}
  \item prendre une boîte (une boîte à chaussures fait l'affaire)~;
        au milieu d'une des extrémités, percer un trou net de la
        taille d'une pointe d'épingle~;
  \item remplacer l'extrémité opposée par du papier calque ---
        l'écran~;
  \item se couvrir la tête et le côté de l'écran avec un tissu, et
        pointer le trou vers une scène bien éclairée --- la
        fenêtre, une rue ensoleillée~;
  \item regarder le papier calque~: la scène apparaît, en couleurs,
        bougeant quand la scène bouge --- \emph{à l'envers}.
\end{enumerate}
\end{method}

\begin{proposition}[Pourquoi l'image est à
l'envers]\label{prop:g6:light-propagation:inverted}
La \omterm{prop:g6:light-propagation:law}{propagation rectiligne} construit l'\omterm{prop:g5:mirrors-reflection:image}{image} et la retourne d'un même
geste. Depuis le haut de la scène, les rayons filent tout droit à
travers le trou --- et des lignes droites issues d'un point haut,
passant par un trou plus bas, doivent continuer \emph{vers le bas}~:
le haut arrive en bas de l'écran. Depuis le bas de la scène, cela
monte à travers le trou jusqu'en haut de l'écran. Chaque point de la
scène envoie son fil par cette unique petite porte, les fils s'y
croisent, et l'\omterm{prop:g5:mirrors-reflection:image}{image} s'assemble retournée --- le haut à la place du
bas, la gauche à la place de la droite.
\end{proposition}

\begin{remark}[Une boîte vieille et honorable]\label{rem:g6:light-propagation:history}
La chambre à sténopé --- \emph{camera obscura}, «~chambre noire~» ---
est de plusieurs siècles antérieure à la photographie~: les
astronomes y observaient les éclipses sans danger sur son écran, et
les peintres y décalquaient les \omterm{prop:g5:mirrors-reflection:image}{images} pour réussir leur perspective.
Tous les appareils photo modernes, et ton propre œil, en conservent
l'architecture~: une petite porte pour la \omterm{def:g1:light-and-shadows:source}{lumière}, un écran pour
l'\omterm{prop:g5:mirrors-reflection:image}{image} --- et oui, l'\omterm{prop:g5:mirrors-reflection:image}{image} au fond de ton œil est elle aussi à
l'envers~; ton cerveau la retourne poliment. Une seule loi de la
ligne droite, patiemment suivie, explique l'appareil photo familial
et la moitié de l'histoire de la fabrication d'\omterm{prop:g5:mirrors-reflection:image}{images}.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:1}
Énonce la loi de la \omterm{prop:g6:light-propagation:law}{propagation rectiligne} avec tes propres mots, et
décortique les deux mots savants.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:2}
Nomme les trois formes de \omterm{def:g6:light-propagation:ray}{faisceau}, avec pour chacune une source ou
une situation réelle. Sous quelle forme la \omterm{def:g1:light-and-shadows:source}{lumière} du Soleil
arrive-t-elle, et pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:3}
Un charpentier vise le long de l'arête d'une étagère pour vérifier
qu'elle est droite. Qu'est-ce qui joue le rôle de la règle dans ce
test~? À quelle loi fait-il confiance~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:4}
Dans la construction de l'\omterm{def:g1:light-and-shadows:shadow}{ombre}, qu'est-ce qui définit le bord de
l'\omterm{def:g1:light-and-shadows:shadow}{ombre} sur l'écran~? Qu'est-il vrai pour un œil placé à l'intérieur
du cône d'\omterm{def:g1:light-and-shadows:shadow}{ombre}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:5}
À l'aide de \cref{ex:g6:light-propagation:sizes}~: la balle se trouve
à mi-chemin de la lampe et de l'écran. La balle fait \qty{6}{cm} de
large --- quelle est la largeur de son \omterm{def:g1:light-and-shadows:shadow}{ombre}~? Et si la balle
s'approche de l'écran~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:6}
Dans la chambre à sténopé, où arrive sur l'écran la \omterm{def:g1:light-and-shadows:source}{lumière} venue du
haut de la scène~? Retrace la raison en une phrase.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:7}
Un camarade agite sa main droite devant ta chambre à sténopé. Sur
l'écran, quel côté s'agite, et dans quel sens en hauteur~?
(Attention --- deux retournements.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:8}
Cite deux utilisateurs, anciens ou modernes, de la chambre noire, et
le service qu'elle a rendu à chacun.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:light-propagation:9}
Construis (croquis à la règle)~: une lampe à gauche, un obstacle en
forme de pièce de monnaie, un écran à droite --- puis éloigne la
\emph{lampe} deux fois plus de la pièce, l'écran restant fixe.
Qu'arrive-t-il à la taille de l'\omterm{def:g1:light-and-shadows:shadow}{ombre}, d'après ton propre dessin~?
Concilie cela avec l'éventail des rayons frôlants.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:light-propagation:10}
Par temps ensoleillé, les \omterm{def:g1:light-and-shadows:shadow}{ombres} des arbres sont nettes~; sous un
ciel couvert, les \omterm{def:g1:light-and-shadows:shadow}{ombres} disparaissent presque. Explique les deux
ciels avec des rayons et des sources --- quelle sorte de «~lampe~»
est une couche de nuages gris tout entière~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:light-propagation:11}
L'\omterm{prop:g5:mirrors-reflection:image}{image} de la chambre à sténopé devient plus lumineuse quand on
élargit le trou --- mais elle devient floue. Explique les deux
effets avec les fils qui passent par la porte~: que laisse peindre
sur l'écran une porte large, pour chaque point de la scène~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:light-propagation:12}
Sous un arbre feuillu, pendant une éclipse partielle de Soleil, le
sol se couvre de centaines de petits croissants. Assemble
l'explication à partir de ce chapitre~: que sont les trouées entre
les feuilles, qu'est chaque tache lumineuse un jour ordinaire, et
pourquoi les taches deviennent-elles des croissants quand le Soleil
en devient un~? (Le chapitre sur les éclipses de l'an prochain sera
fier de toi.)
\end{exercise}

\section{Problème~: le théâtre d'ombres}

\begin{problem}[{Problème du week-end --- le théâtre d'ombres de la
famille~; des marionnettes dimensionnées par proportion, un final au
sténopé~; les lignes droites de la lumière mènent le
spectacle}]\label{pb:g6:light-propagation:1}
Un drap tendu en écran, une petite lampe bien vive, des marionnettes
en carton~: la première est samedi, et l'équipe de scène travaille à
la règle.

\medskip\noindent\textbf{Partie I --- installer la scène.}
La lampe est à \qty{4}{m} de l'écran.
\begin{enumerate}
  \item Une marionnette tenue contre l'écran jette une \omterm{def:g1:light-and-shadows:shadow}{ombre}
        exactement à sa taille. Pourquoi~?
  \item La marionnette du dragon fait \qty{20}{cm} de haut. Tenue à
        mi-chemin --- à \qty{2}{m} de la lampe et de l'écran ---
        quelle est la hauteur de son \omterm{def:g1:light-and-shadows:shadow}{ombre}~?
  \item Pour le final, le dragon doit dominer la scène du haut de
        \qty{80}{cm}. L'équipe le fait glisser vers la lampe~: à
        $1$ \omterm{def:g2:measuring-length:units}{mètre} de la lampe, les rayons frôlants s'écartent de
        quatre fois la taille de la marionnette. Vérifie~: les
        \qty{80}{cm} correspondent-ils, et à quelle fraction de la
        distance lampe--écran la marionnette se tient-elle~?
  \item Pourquoi la lampe du théâtre doit-elle être petite et nue,
        et non un large globe dépoli~? (Que ferait le globe aux
        bords nets du dragon~?)
\end{enumerate}

\medskip\noindent\textbf{Partie II --- mise en scène avec des
rayons.}
\begin{enumerate}[resume]
  \item La marionnette du loup doit grandir de façon menaçante
        pendant sa scène, sans que le bras du marionnettiste
        apparaisse. Dans quel sens la marionnette est-elle
        déplacée, et qu'arrive-t-il aux bords de son \omterm{def:g1:light-and-shadows:shadow}{ombre} à mesure
        qu'elle grandit~? (Les bords s'adoucissent légèrement ---
        accepte-le et explique-le par la taille de la lampe,
        petite mais non ponctuelle.)
  \item Deux marionnettes, tenues à la même distance, ne doivent
        pas se faire de l'\omterm{def:g1:light-and-shadows:shadow}{ombre} l'une à l'autre. Que doit-il être
        vrai des lignes droites issues de la lampe et passant par
        chaque marionnette~?
  \item Une main se glisse entre la lampe et le dragon, au tiers du
        trajet depuis la lampe. Quelle \omterm{def:g1:light-and-shadows:shadow}{ombre} paraît la plus grande
        sur l'écran, celle de la main ou celle du dragon, et
        pourquoi~?
  \item Le méchant doit disparaître instantanément. Rayons en main,
        donne deux façons de faire~: l'une en déplaçant la
        marionnette, l'autre en utilisant un second obstacle placé
        plus près.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- le final au sténopé.}
Pour la dernière scène, l'équipe transforme le théâtre lui-même en
chambre noire~: un réverbère dehors, un sténopé percé dans le volet,
le drap-écran tendu en travers de la pièce.
\begin{enumerate}[resume]
  \item Le réverbère est suspendu haut, dehors. Où son \omterm{prop:g5:mirrors-reflection:image}{image}
        arrive-t-elle sur le drap --- en haut ou en bas~? Pourquoi~?
  \item Un cycliste attardé passe de gauche à droite. Dans quel
        sens l'\omterm{prop:g5:mirrors-reflection:image}{image} du cycliste glisse-t-elle sur le drap~?
  \item Le public réclame une \omterm{prop:g5:mirrors-reflection:image}{image} plus lumineuse~: l'équipe
        agrandit le sténopé jusqu'à la taille d'une pièce de
        monnaie. Que devient le spectacle, et pourquoi~?
  \item Phrase de conclusion du programme~: en une phrase, nomme
        l'unique loi qui a mené toute la soirée --- les deux
        théâtres, toutes les \omterm{def:g1:light-and-shadows:shadow}{ombres}, et la rue à l'envers.
\end{enumerate}
\end{problem}
