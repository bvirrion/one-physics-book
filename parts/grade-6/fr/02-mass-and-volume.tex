\chapter{Masse et volume}\label{ch:g6:mass-and-volume}

Combien de limonade tient dans la bouteille, et quel est le \omterm{def:g4:weight-and-mass:weight}{poids} de
la caisse de bouteilles~? Deux questions différentes --- la place et
la matière --- que le langage courant emmêle allègrement en un seul
mot, «~gros~». La science les démêle~: le \omterm{def:g6:mass-and-volume:volume}{volume} pour la place qu'un
objet occupe, la \omterm{def:g3:measuring-mass:mass}{masse} pour la matière qu'il contient. \omterm{def:g6:measurement-in-science:measuring}{Mesurer} la
seconde, tu es un vieux routier~; aujourd'hui, nous apprenons à
\omterm{def:g6:measurement-in-science:measuring}{mesurer} la première --- même pour un caillou dont aucune règle ne
saurait aimer la forme.

\section{Le volume~: la place occupée}

\begin{definition}[Volume]\label{def:g6:mass-and-volume:volume}
Le \emph{volume}\index{volume} d'un objet est la quantité de place
qu'il occupe. Les \omterm{def:g3:solids-liquids-gases:solid}{solides}, les \omterm{def:g3:solids-liquids-gases:liquid}{liquides} et les \omterm{def:g3:solids-liquids-gases:gas}{gaz} ont tous un
volume~: la brique prend sa place, la limonade prend l'intérieur de
la bouteille, l'\omterm{def:g2:air-around-us:air}{air} prend le reste. Le volume se mesure en
\emph{centimètres cubes}\index{centimètre cube} (\unit{cm^3}) --- la
place d'un petit cube d'un \omterm{def:g2:measuring-length:units}{centimètre} d'arête --- et en
\emph{litres}\index{litre} (\unit{L}).
\end{definition}

\begin{proposition}[Le litre en cubes]\label{prop:g6:mass-and-volume:litre}
Les deux familles d'\omterm{def:g6:measurement-in-science:measuring}{unités} de \omterm{def:g6:mass-and-volume:volume}{volume} n'en font qu'une~:
\[
\qty{1}{L} = \qty{1000}{cm^3},
\qquad
\qty{1}{mL} = \qty{1}{cm^3}.
\]
Un \omterm{def:g6:mass-and-volume:volume}{litre} est exactement la place à l'intérieur d'un cube de dix
\omterm{def:g2:measuring-length:units}{centimètres} d'arête~: $10 \times 10 \times 10 = 1000$ petits cubes.
\end{proposition}

\begin{example}[Des volumes à savoir par cœur]\label{ex:g6:mass-and-volume:byheart}
Une cuillère à café contient environ \qty{5}{mL}~; un verre à boire,
environ \qty{25}{cL}, soit un quart de \omterm{def:g6:mass-and-volume:volume}{litre}~; la brique de lait,
\qty{1}{L}~; une baignoire, autour de \qty{150}{L}~; un morceau de
sucre occupe environ \qty{3}{cm^3}~; un dé à jouer, environ
\qty{4}{cm^3}. Des repères de ce genre rendent possible l'étape du
jugement, dans toute mesure.
\end{example}

\section{Mesurer les liquides}

\begin{method}[Lire une éprouvette graduée]\label{met:g6:mass-and-volume:cylinder}
Le verre doseur du laboratoire est l'\emph{éprouvette graduée}~:
haute, étroite, finement graduée.
\begin{enumerate}
  \item poser l'éprouvette sur une table plane --- ne jamais la
        lire à la main~;
  \item trouver la valeur d'une graduation, comme pour toute
        échelle~;
  \item mettre l'œil au niveau de la surface du \omterm{def:g3:solids-liquids-gases:liquid}{liquide}~: cette
        surface n'est pas plane mais légèrement courbée --- c'est le
        \emph{ménisque}\index{ménisque} --- et grimpe un peu le long
        des parois de verre~;
  \item lire au \emph{bas} de la courbe, bien en face.
\end{enumerate}
L'œil trop haut ou trop bas, et la courbe te ment d'une graduation ou
deux --- l'erreur de la vue de biais, sous son déguisement préféré.
\end{method}

\section{Mesurer les solides --- même biscornus}

Un \omterm{def:g3:solids-liquids-gases:solid}{solide} en forme de brique se rend à la règle~: longueur fois
largeur fois hauteur donne son \omterm{def:g6:mass-and-volume:volume}{volume} en \unit{cm^3}, comme l'a
montré ton cours de mathématiques. Mais que faire d'un caillou, d'une
clé, d'une prune~? Aucune règle ne convient à une forme biscornue. La
réponse a deux mille ans, et elle est sortie d'une baignoire.

\begin{proposition}[Le déplacement]\label{prop:g6:mass-and-volume:displacement}
Un \omterm{def:g3:solids-liquids-gases:solid}{solide} entièrement immergé chasse --- déplace --- exactement son
propre \omterm{def:g6:mass-and-volume:volume}{volume} d'eau. La montée du niveau de l'eau mesure
l'intrus~: biscornu ou lisse, l'eau se moule sur chaque creux et
chaque bosse.
\end{proposition}

\begin{method}[Le volume par la montée d'eau]\label{met:g6:mass-and-volume:rise}
\begin{enumerate}
  \item verser de l'eau dans une éprouvette graduée --- assez pour
        recouvrir l'objet --- et lire le \omterm{def:g6:mass-and-volume:volume}{volume}~: disons
        \qty{120}{mL}~;
  \item attacher un fil à l'objet et le descendre doucement
        jusqu'à immersion complète, sans éclaboussure et sans
        toucher les parois~;
  \item lire de nouveau~: disons \qty{145}{mL}~;
  \item soustraire~: $145 - 120 = 25$ --- le \omterm{def:g6:mass-and-volume:volume}{volume} de l'objet est
        de \qty{25}{cm^3} (des millilitres de montée, des
        \omterm{def:g6:mass-and-volume:volume}{centimètres cubes} de pierre~: c'est la même chose).
\end{enumerate}
\end{method}

\begin{example}[La couronne dans la baignoire]\label{ex:g6:mass-and-volume:archimedes}
La vieille histoire~: un roi soupçonnait son orfèvre d'avoir coupé
d'argent l'or de la couronne royale, et demanda au savant Archimède
de le découvrir sans abîmer la couronne. En entrant dans un bain
plein, Archimède vit l'eau déborder --- et comprit d'un éclair que le
débordement mesurait le \omterm{def:g6:mass-and-volume:volume}{volume} de son propre corps, bosses comprises.
La légende dit qu'il courut dans les rues en criant «~Eurêka~!~» ---
\emph{j'ai trouvé~!} Comment le \omterm{def:g6:mass-and-volume:volume}{volume} de la couronne a démasqué la
fraude demande une idée de plus, deux chapitres plus loin~; la
baignoire, elle, lui a donné le \omterm{def:g6:mass-and-volume:volume}{volume} de \emph{n'importe quoi}.
\end{example}

\section{Masse et volume répondent à deux questions}

\begin{example}[Même volume, masses différentes]\label{ex:g6:mass-and-volume:different}
Remplis deux verres identiques de \qty{25}{cL}, l'un d'eau, l'autre
de miel, et pèse-les (en retranchant les verres)~: le verre de miel
contient nettement plus de \omterm{def:g3:measuring-mass:units}{grammes} dans exactement la même place.
Même \omterm{def:g6:mass-and-volume:volume}{volume}, \omterm{def:g3:measuring-mass:mass}{masses} différentes. Et un \omterm{def:g6:mass-and-volume:volume}{litre} d'\omterm{def:g2:air-around-us:air}{air} pèse à peine plus
d'un \omterm{def:g3:measuring-mass:units}{gramme}, dans la place où un \omterm{def:g6:mass-and-volume:volume}{litre} d'eau pèse un \omterm{def:g3:measuring-mass:units}{kilogramme}
entier. La place et la matière tiennent vraiment deux comptabilités
distinctes --- garde bien cette idée~: elle deviendra une loi, avec un
nom, au dernier chapitre de cette année.
\end{example}

\begin{remark}[Peser ce qui ne tient pas debout tout
seul]\label{rem:g6:mass-and-volume:tare}
Comment peser de la farine sans peser son saladier~? Les \omterm{def:g3:levers-and-scales:scale}{balances}
modernes ont un bouton pour cela~: on pose le saladier vide sur le
plateau, on appuie --- l'affichage se remet à zéro --- puis on verse~:
la \omterm{def:g3:levers-and-scales:scale}{balance} annonce alors la farine seule. La manœuvre s'appelle la
\emph{tare}. Sans le bouton, on pèse deux fois et on soustrait~:
saladier-avec-farine moins saladier vide. Dans les deux cas, c'est de
nouveau la règle du départ honnête~: partir d'un zéro loyal.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:1}
À quelle question le \omterm{def:g6:mass-and-volume:volume}{volume} répond-il, et à quelle question la \omterm{def:g3:measuring-mass:mass}{masse}
répond-elle~? Donne les deux principales \omterm{def:g6:measurement-in-science:measuring}{unités} de chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:2}
Convertis~: \qty{2}{L} en \unit{cm^3}~; \ \qty{350}{mL} en
\unit{cm^3}~; \ \qty{1500}{cm^3} en \omterm{def:g6:mass-and-volume:volume}{litres}~; \ un \omterm{def:g6:mass-and-volume:volume}{demi-litre} en
millilitres.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:3}
Pourquoi une éprouvette graduée doit-elle être lue à hauteur d'œil,
posée sur la table, au bas du \omterm{met:g6:mass-and-volume:cylinder}{ménisque}~? Nomme l'erreur que chaque
règle évite.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:4}
Une éprouvette indique \qty{200}{mL}~; une prune y étant descendue,

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:5}
Pourquoi la méthode de la montée fonctionne-t-elle même pour un
caillou biscornu, qu'aucune règle ne saurait \omterm{def:g6:measurement-in-science:measuring}{mesurer}~? Quelle

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:6}
Une gomme en forme de brique mesure \qty{5}{cm} sur \qty{2}{cm} sur
\qty{1}{cm}. Son \omterm{def:g6:mass-and-volume:volume}{volume}~? Plan de vérification~: comment le
confirmerais-tu par la méthode de la montée~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:7}
Pour peser du riz~: saladier vide \qty{280}{g}~; saladier avec riz
\qty{745}{g}. Combien de riz~? Quel bouton fait le même travail en

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:8}
Qu'est-ce qui a le plus grand \omterm{def:g6:mass-and-volume:volume}{volume}, un \omterm{def:g3:measuring-mass:units}{kilogramme} de plumes ou un

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:mass-and-volume:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:mass-and-volume:10}

\begin{exercise}[$\star\star$]\label{exo:g6:mass-and-volume:11}

\section{Problème~: jour de mise en bouteilles au verger}

\begin{problem}[{Problème du week-end --- jour de pressage au
verger~; des litres dans les bouteilles, des caisses sur la balance~;
le caillou dans le pichet}]\label{pb:g6:mass-and-volume:1}
Le verger presse ses pommes aujourd'hui, et chaque aide reçoit un
travail de mesure.

\medskip\noindent\textbf{Partie I --- du jus au \omterm{def:g6:mass-and-volume:volume}{litre}.}
Le pressoir a rempli un tonneau de \qty{20}{L}.
\begin{enumerate}
  \item Combien de \omterm{def:g6:mass-and-volume:volume}{centimètres cubes} font \qty{20}{L}~?
  \item Le jus est mis en bouteilles de \qty{75}{cL}. Combien de
        bouteilles \emph{pleines} le tonneau donne-t-il~?
  \item Combien de jus reste-t-il après la dernière bouteille
        pleine, en centilitres~?
  \item Le reste est versé dans des verres de \qty{25}{cL}. Combien
        de verres remplit-il~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- les caisses sur la \omterm{def:g3:levers-and-scales:scale}{balance}.}
\begin{enumerate}[resume]
  \item Une bouteille vide a une \omterm{def:g3:measuring-mass:mass}{masse} de \qty{450}{g}. À l'aide de
        la partie I, quelle est la \omterm{def:g3:measuring-mass:mass}{masse} du jus contenu dans une
        bouteille pleine, sachant qu'un \omterm{def:g6:mass-and-volume:volume}{litre} de jus a une \omterm{def:g3:measuring-mass:mass}{masse}
        d'environ \qty{1}{kg}~? (Travaille en \omterm{def:g3:measuring-mass:units}{grammes}.)
  \item Que pèse donc une bouteille pleine au total~?
  \item Une caisse en bois pèse \qty{2}{kg} à vide et porte six
        bouteilles pleines. \omterm{def:g3:measuring-mass:mass}{Masse} totale d'une caisse chargée~?
  \item La camionnette peut transporter \qty{200}{kg} de caisses.
        Combien de caisses chargées peut-elle prendre~? (Des
        caisses entières seulement.)
\end{enumerate}

\medskip\noindent\textbf{Partie III --- le caillou dans le pichet.}
Le petit Tom lâche un caillou dans un pichet de jus plein de
\qty{1}{L}, et du jus déborde sur la table.
\begin{enumerate}[resume]
  \item Le débordement épongé mesure \qty{40}{mL}. Qu'a mesuré ce
        dégât au sujet du caillou, et par quelle proposition~?
  \item Une fois le caillou repêché, combien de jus reste-t-il dans
        le pichet~?
  \item Tom proteste~: «~Le caillou est petit --- regarde, il tient
        dans mon poing~!~» Sa sœur répond avec une éprouvette
        graduée~: elle met le caillou dans \qty{100}{mL} d'eau.
        Quelle lecture prouve que le débordement disait vrai~?
  \item Comptabilité en prime~: le caillou pèse \qty{104}{g} sur la
        \omterm{def:g3:levers-and-scales:scale}{balance} de cuisine. Sans aucune idée nouvelle, peux-tu déjà
        dire si la matière du caillou est plus lourde ou plus
        légère que celle du jus \emph{pour la même place occupée}~?
        Compare \qty{104}{g} de caillou dans \qty{40}{cm^3} à la
        \omterm{def:g3:measuring-mass:mass}{masse} de jus qui remplirait ces mêmes \qty{40}{cm^3} --- et
        garde ta conclusion au chaud pour le chapitre sur la \omterm{def:g3:measuring-mass:mass}{masse}
        volumique.
\end{enumerate}
\end{problem}
