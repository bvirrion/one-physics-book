\chapter{الكتلة والحجم}\label{ch:g6:mass-and-volume}

كم من الليموناضة يسع في القنّينة، وكم يثقل صندوق القنّينات؟
سؤالان مختلفان --- الحيّز والمادة --- تخلطهما لغة الحياة اليومية
راضيةً في كلمة واحدة: ``كبير''. والعلم يفكّهما: \omterm{def:g6:mass-and-volume:volume}{الحجم} للحيّز
الذي يشغله الشيء، \omterm{def:g3:measuring-mass:mass}{والكتلة} للمادة التي يحملها. وأنت في \omterm{def:g6:measurement-in-science:measuring}{قياس}
الثانية قديم الخبرة؛ واليوم نتعلّم \omterm{def:g6:measurement-in-science:measuring}{قياس} الأول --- حتى لحصاة لا
شكل لها تحبّه مسطرة.

\section{الحجم: الحيّز الذي يشغله الشيء}

\begin{definition}[الحجم]\label{def:g6:mass-and-volume:volume}
\emph{حجم}\index{حجم} الشيء هو مقدار الحيّز الذي يشغله. وللأجسام
الصلبة والسوائل والغازات حجم كلها: فالآجرّة تشغل حيّزها،
والليموناضة تشغل داخل القنّينة، \omterm{def:g2:air-around-us:air}{والهواء} يشغل الباقي. ويُقاس
الحجم \emph{بالسنتيمتر
المكعّب}\index{سنتيمتر مكعّب} (\unit{cm^3}) --- وهو حيّز مكعّب
صغير طول حرفه \omterm{def:g2:measuring-length:units}{سنتيمتر} واحد --- \emph{وباللتر}\index{لتر}
(\unit{L}).
\end{definition}

\begin{proposition}[اللتر مكعّبات]\label{prop:g6:mass-and-volume:litre}
أسرتا وحدات \omterm{def:g6:mass-and-volume:volume}{الحجم} أسرة واحدة:
\[
\qty{1}{L} = \qty{1000}{cm^3},
\qquad
\qty{1}{mL} = \qty{1}{cm^3}.
\]
\omterm{def:g6:mass-and-volume:volume}{فاللتر} هو بالضبط الحيّز داخل مكعّب طول حرفه عشرة سنتيمترات:
$10 \times 10 \times 10 = 1000$ مكعّب صغير.
\end{proposition}

ممكنة.
\end{example}

\section{كيل السوائل}

\begin{method}[كيف تقرأ المخبار المدرَّج]\label{met:g6:mass-and-volume:cylinder}
إبريق المختبر هو \emph{المخبار المدرَّج}: طويل ضيّق دقيق
التدريج.
\begin{enumerate}
  \item ضع المخبار على طاولة مستوية --- ولا تقرأه في يدك
        أبدًا؛
  \item وجِد قيمة التدريج، كما في كل سلّم؛
  \item واجعل عينك في مستوى سطح \omterm{def:g3:solids-liquids-gases:liquid}{السائل}: فالسطح ليس مستويًا بل
        منحنيًا قليلًا --- \emph{سطحًا هلاليًّا}\index{سطح هلالي}
        --- يتسلّق قليلًا على جدران الزجاج؛
  \item واقرأ عند \emph{أسفل} الانحناء، أفقيًّا تمامًا.
\end{enumerate}
فإن علت العين أو انخفضت كذب عليك الانحناء بخطوة أو خطوتين ---
وهو خطأ النظرة المائلة في زيّه المفضّل.
\end{method}

\section{كيل الأجسام الصلبة --- حتى المتكتّلة منها}

\omterm{def:g3:solids-liquids-gases:solid}{الجسم الصلب} على شكل آجرّة يستسلم للمسطرة: فالطول في العرض في
الارتفاع يعطي حجمه \omterm{def:g6:measurement-in-science:measuring}{بوحدة} \unit{cm^3}، كما بيّن منهاج الرياضيات. فماذا
عن حصاة أو مفتاح أو حبّة برقوق؟ لا مسطرة تلائم \omterm{def:g3:measuring-mass:mass}{كتلة} غير منتظمة.
والجواب عمره ألفا سنة، وقد خرج من حوض حمّام.

\begin{proposition}[الإزاحة]\label{prop:g6:mass-and-volume:displacement}
\omterm{def:g3:solids-liquids-gases:solid}{الجسم الصلب} المغمور كله تحت الماء يزيح --- أي يدفع جانبًا ---
حجمه هو من الماء بالضبط. وارتفاع مستوى الماء يقيس الداخل
الجديد: متكتّلًا كان أم أملس، فالماء يتشكّل على كل غمّازة
ونتوء.
\end{proposition}

\begin{method}[الحجم بارتفاع الماء]\label{met:g6:mass-and-volume:rise}
\begin{enumerate}
  \item اسكب ماءً في مخبار مدرَّج --- بما يكفي لتغطية الجسم ---
        واقرأ \omterm{def:g6:mass-and-volume:volume}{الحجم}: ولنقل \qty{120}{mL}؛
  \item واربط خيطًا بالجسم وأنزله برفق حتى يغمره الماء كله، بلا
        رشقة ولا ملامسة للجدران؛
  \item واقرأ من جديد: ولنقل \qty{145}{mL}؛
  \item واطرح: $145 - 120 = 25$ --- \omterm{def:g6:mass-and-volume:volume}{فحجم} الجسم \qty{25}{cm^3}
        (مليلترات من الارتفاع، ومكعّبات سنتيمترية من الحجر:
        الشيء نفسه).
\end{enumerate}
\end{method}

\begin{example}[التاج في حوض الحمّام]\label{ex:g6:mass-and-volume:archimedes}
الحكاية القديمة: ملك ارتاب في أن صائغه خلط ذهب التاج الملكي
بفضّة، فطلب من العالِم أرخميدس أن يعرف من غير أن يُتلف التاج.
وبينما كان أرخميدس ينزل في حمّام ممتلئ راقب الماء يفيض عن
الحافة --- فرأى في ومضة أن الفائض يقيس \omterm{def:g6:mass-and-volume:volume}{حجم} جسمه هو بكل تكتّلاته.
وتقول الأسطورة إنه ركض في الشوارع صائحًا ``وجدتها!'' أما كيف
فضح حجمُ التاج الغشَّ فيحتاج إلى فكرة واحدة أخرى، بعد فصلين؛
لكن حوض الحمّام أعطاه \omterm{def:g6:mass-and-volume:volume}{حجم} \emph{أي شيء}.
\end{example}

\section{الكتلة والحجم جوابان مختلفان}

\begin{example}[الحجم نفسه وكتلة مختلفة]\label{ex:g6:mass-and-volume:different}
املأ كأسين متماثلتين سعة \qty{25}{cL}، إحداهما ماءً والأخرى
عسلًا، وزِنهما (طارحًا الكأسين): تجد كأس العسل تحمل غرامات أكثر
بوضوح في الحيّز نفسه بعينه. \omterm{def:g6:mass-and-volume:volume}{الحجم} نفسه \omterm{def:g3:measuring-mass:mass}{وكتلة} مختلفة. \omterm{def:g6:mass-and-volume:volume}{ولتر}
\omterm{def:g2:air-around-us:air}{الهواء} يكاد لا يزن أكثر من \omterm{def:g3:measuring-mass:units}{غرام}، في الحيّز الذي يزن فيه \omterm{def:g6:mass-and-volume:volume}{لتر}
الماء كيلوغرامًا كاملًا. فالحيّز والمادة محاسبان مختلفان حقًّا
--- فأمسك هذه الفكرة بقوة: فستصير قانونًا، له اسم، في آخر فصل
من هذه السنة.
\end{example}

\begin{remark}[وزن ما لا يقف وحده]\label{rem:g6:mass-and-volume:tare}
كيف تزن طحينًا من غير أن تزن وعاءه؟ للموازين الحديثة زرّ لذلك:
ضع الوعاء الفارغ على الكفّة، واضغط --- فتعود الشاشة إلى الصفر
--- ثم اسكب: فيُبلّغ الميزان الآن الطحين وحده. وتُسمّى المناورة
\emph{التصفير}. وبلا الزرّ: زِن مرتين واطرح: الوعاء مع الطحين
ناقصًا الوعاء الفارغ. وفي الحالتين هي قاعدة البداية العادلة من
جديد: ابدأ من صفر أمين.
\end{remark}

عن أيّ سؤال يجيب \omterm{def:g6:mass-and-volume:volume}{الحجم}، وعن أيّ سؤال تجيب \omterm{def:g3:measuring-mass:mass}{الكتلة}؟ أعطِ

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:2}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:3}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:5}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:6}
ممحاة على شكل آجرّة أبعادها \qty{5}{cm} في \qty{2}{cm} في

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:7}
\omterm{def:g4:weight-and-mass:weight}{لوزن} الأرزّ: الوعاء الفارغ \qty{280}{g}؛ والوعاء مع الأرزّ

خطوة واحدة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:mass-and-volume:8}
أيّهما أكبر حجمًا، \omterm{def:g3:measuring-mass:units}{كيلوغرام} من الريش أم \omterm{def:g3:measuring-mass:units}{كيلوغرام} من الحديد؟

\begin{exercise}[$\star\star$]\label{exo:g6:mass-and-volume:9}

\begin{exercise}[$\star\star$]\label{exo:g6:mass-and-volume:10}

\begin{exercise}[$\star\star$]\label{exo:g6:mass-and-volume:11}

\begin{exercise}[$\star\star\star$]\label{exo:g6:mass-and-volume:12}

\end{exercise}

\section{مسألة: يوم التعبئة في البستان}

\begin{problem}[{مسألة نهاية الأسبوع --- يوم العصر في البستان؛
لترات إلى قنّينات، وصناديق على الموازين؛ الحصاة في
الإبريق}]\label{pb:g6:mass-and-volume:1}
يعصر البستان تفّاحه اليوم، وينال كل معين عملًا في \omterm{def:g6:measurement-in-science:measuring}{القياس}.

\medskip\noindent\textbf{الجزء الأول --- العصير \omterm{def:g6:mass-and-volume:volume}{باللتر}.}
ملأت المعصرة برميلًا سعته \qty{20}{L}.
\begin{enumerate}
  \item كم سنتيمترًا مكعّبًا هي \qty{20}{L}؟
  \item ويُعبّأ العصير في قنّينات سعة \qty{75}{cL}. فكم قنّينة
        \emph{ممتلئة} يعطي البرميل؟
  \item وكم يبقى من العصير بعد آخر قنّينة ممتلئة، بالسنتيلترات؟
  \item ويُسكب الباقي في كؤوس سعة \qty{25}{cL}. فكم كأسًا
        يملأ؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- الصناديق على
الميزان.}
\begin{enumerate}[resume]
  \item \omterm{def:g3:measuring-mass:mass}{كتلة} القنّينة الفارغة \qty{450}{g}. باستعمال الجزء
        الأول، ما \omterm{def:g3:measuring-mass:mass}{كتلة} العصير في قنّينة ممتلئة واحدة، إذا كانت
        \omterm{def:g3:measuring-mass:mass}{كتلة} \omterm{def:g6:mass-and-volume:volume}{لتر} العصير نحو \qty{1}{kg}؟ (اعمل بالغرامات.)
  \item فكم تزن قنّينة ممتلئة واحدة إذن كلها؟
  \item وصندوق خشبي يزن \qty{2}{kg} فارغًا ويحمل ستّ قنّينات
        ممتلئة. فما \omterm{def:g3:measuring-mass:mass}{الكتلة} الكلّية لصندوق محمّل؟
  \item والشاحنة الصغيرة تستطيع حمل \qty{200}{kg} من الصناديق.
        فكم صندوقًا محمّلًا تأخذ؟ (صناديق كاملة فقط.)
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- الحصاة في الإبريق.}
يُسقط توم الصغير حصاة في إبريق عصير ممتلئ سعته \qty{1}{L}،
فيفيض العصير على الطاولة.
\begin{enumerate}[resume]
  \item الفائض الممسوح يكيل \qty{40}{mL}. فما الذي قاسه
        الانسكاب عن الحصاة، وبأيّ قضية؟
  \item وبعد انتشال الحصاة، كم يبقى من العصير في الإبريق؟
  \item ويحتجّ توم: ``الحصاة صغيرة --- انظروا، تختبئ في قبضتي!''
        فتجيبه أخته بمخبار مدرَّج: تضع الحصاة في \qty{100}{mL}
        من الماء. فأيّ قراءة تثبت أن الانسكاب قال الصدق؟
  \item محاسبة إضافية: تزن الحصاة \qty{104}{g} على ميزان
        المطبخ. أتستطيع، من غير أي فكرة جديدة، أن تقول أيّهما
        أثقل، مادة الحصاة أم مادة العصير، \emph{للحيّز نفسه
        المشغول}؟ قارن \qty{104}{g} من الحصاة في \qty{40}{cm^3}
        \omterm{def:g3:measuring-mass:mass}{بالكتلة} التي يملأ بها العصير \qty{40}{cm^3} نفسها ---
        واحتفظ بنتيجتك دافئة لفصل الكثافة.
\end{enumerate}
\end{problem}
