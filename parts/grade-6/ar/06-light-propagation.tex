\chapter{الانتشار المستقيم للضوء}\label{ch:g6:light-propagation}

خدمتك عبارة ``\omterm{def:g1:light-and-shadows:source}{الضوء} ينتشر في خطوط مستقيمة'' سنتين --- ففسّرت
الظلال، وصوّبت المرايا، وصفّت ثلاث بطاقات وشمعة. وهذه السنة ينال
القانون القديم اسمه الرسمي، وقواعد رسمه، وأعجب نتائجه: آلة تصوير
بلا عدسة ولا زجاج ولا قطعة متحرّكة --- بثقب لا غير.

\section{القانون، مقولًا على أصوله}

\begin{proposition}[الانتشار المستقيم للضوء]\label{prop:g6:light-propagation:law}
في \omterm{def:g2:air-around-us:air}{الهواء} الصافي --- وفي أي مادة صافية متجانسة كالماء الساكن أو
الزجاج --- ينتشر \omterm{def:g1:light-and-shadows:source}{الضوء} في خطوط مستقيمة. وهذا قانون
\emph{الانتشار المستقيم}\index{انتشار مستقيم}: و\emph{المستقيم}
ببساطة صفة الخطوط التي لا تنحني، و\emph{الانتشار} اسم انتقال
\omterm{def:g1:light-and-shadows:source}{الضوء} من موضع إلى موضع.
\end{proposition}

\begin{definition}[الشعاع والحزمة]\label{def:g6:light-propagation:ray}
\emph{الشعاع الضوئي}\index{شعاع ضوئي} رسم الفيزيائي لخيط واحد من
\omterm{def:g1:light-and-shadows:source}{الضوء}: خطّ مستقيم عليه سهم يبيّن جهة سيره. والمصابيح الحقيقية
تصبّ خيوطًا لا تُحصى معًا: وحزمة من الأشعّة تُسمّى
\emph{حزمة}\index{حزمة}. وثلاثة أشكال للحزم تغطّي العمل:
\emph{متباعدة} (أشعّة تنتشر من نقطة --- أي \omterm{def:g3:first-electric-circuit:bulb}{مصباح} صغير)،
و\emph{متوازية} (أشعّة جنبًا إلى جنب لا تلتقي ولا تفترق --- \omterm{def:g1:light-and-shadows:source}{ضوء}
الشمس، بعد سفره الطويل)، و\emph{متقاربة} (أشعّة تنغلق نحو نقطة
--- \omterm{def:g1:light-and-shadows:source}{ضوء} تجمعه أجهزة ستلقاها بعد سنتين).
\end{definition}

\begin{example}[القانون في عمله، في كل مكان]\label{ex:g6:light-propagation:sightings}
يسدّد البنّاؤون النظر على طول حافة لوح مستقيم ليحكموا عليها ---
فلا بدّ أن تصطفّ العين والحافة والطرف البعيد، واستقامة \omterm{def:g1:light-and-shadows:source}{الضوء} هي
قسم القاضي. والمسّاحون ينصبون الأوتاد على استقامة بالعين؛ والرماة
والمصوّرون ``يصطفّون للتسديدة''؛ وتجربة البطاقات الثلاث قبل
سنتين صار لها اسم رسمي الآن: اختبار \omterm{prop:g6:light-propagation:law}{للانتشار المستقيم}. وكل واحد
من هؤلاء يقايض خطًّا مستقيمًا من \omterm{def:g1:light-and-shadows:source}{الضوء} بخطّ مستقيم من المادة.
\end{example}

\section{الظلال، مبنيّةً بالإنشاء}

\begin{method}[إنشاء ظلّ بالمسطرة]\label{met:g6:light-propagation:shadow}
\omterm{def:g3:first-electric-circuit:bulb}{لمصباح} صغير $S$، وكرة، وحاجز خلفها:
\begin{enumerate}
  \item ارسم الشعاع الخارج من $S$ الذي يمسّ حافة الكرة العليا
        مسًّا، ومُدّه مستقيمًا إلى الحاجز؛
  \item وكذلك الشعاع الماسّ للحافة السفلى (وفي ثلاثة أبعاد:
        حولها كلها)؛
  \item وبين نقطتَي المسّ على الحاجز يقع \emph{الظلّ}\index{ظلّ}:
        وهي المنطقة التي لا يبلغها أي شعاع مستقيم من $S$؛
  \item والحيّز بين الكرة والحاجز المحدود بالشعاعين الماسّين هو
        \emph{مخروط \omterm{def:g1:light-and-shadows:shadow}{الظلّ}}: فالعين داخله لا ترى \omterm{def:g3:first-electric-circuit:bulb}{المصباح}.
\end{enumerate}
ولا حَزْر في أي موضع: فخطّان بالمسطرة يثبّتان \omterm{def:g6:mass-and-volume:volume}{حجم} \omterm{def:g1:light-and-shadows:shadow}{الظلّ} وموضعه
بالضبط.
\end{method}

\begin{example}[أكبر وأصغر وأحدّ]\label{ex:g6:light-propagation:sizes}
يتوقّع الإنشاء ما وجدته ألعاب اللعبة \omterm{def:g3:first-electric-circuit:bulb}{والمصباح} اليدوي في سنتك
الأولى باللهو: أزلِق الكرة نحو \omterm{def:g3:first-electric-circuit:bulb}{المصباح} فينفرج الشعاعان الماسّان
--- \omterm{def:g1:light-and-shadows:shadow}{ظلّ} عملاق؛ وأزلِقها نحو الحاجز فيحطّان أضيق --- \omterm{def:g1:light-and-shadows:shadow}{ظلّ} \omterm{def:g6:mass-and-volume:volume}{بالحجم}
الحقيقي. والتوقّع دقيق: فمع وقوف الكرة في منتصف المسافة بين
\omterm{def:g3:first-electric-circuit:bulb}{المصباح} والحاجز ينفرج الشعاعان الماسّان ضِعف عرض الكرة --- أي
ظلًّا بضِعف \omterm{def:g6:mass-and-volume:volume}{حجم} الكرة بالضبط. تناسب مرسوم \omterm{def:g1:light-and-shadows:source}{بالضوء}.
\end{example}

\section{آلة التصوير بلا عدسة}

\begin{method}[كيف تبني آلة تصوير ذات ثقب]\label{met:g6:light-propagation:pinhole}
\begin{enumerate}
  \item خذ علبة (تفي علبة حذاء بالغرض)؛ واصنع في وسط أحد طرفيها
        ثقبًا نظيفًا \omterm{def:g6:mass-and-volume:volume}{بحجم} رأس دبّوس؛
  \item واستبدل بالطرف المقابل ورق شفّ --- وهو الحاجز؛
  \item وضع قماشة على رأسك وعلى طرف الحاجز، وصوّب الثقب نحو
        مشهد ساطع --- النافذة، أو شارع مشمس؛
  \item وانظر إلى ورق الشفّ: فيظهر المشهد بألوانه، ويتحرّك مع
        حركته --- \emph{مقلوبًا رأسًا على عقب}.
\end{enumerate}
\end{method}

\begin{proposition}[لماذا تكون الصورة مقلوبة]\label{prop:g6:light-propagation:inverted}
\omterm{prop:g6:light-propagation:law}{الانتشار المستقيم} يبني الصورة ويقلبها في الحركة نفسها. فمن أعلى
المشهد تسافر الأشعّة مستقيمة عبر الثقب --- والخطوط المستقيمة
النازلة من نقطة عالية عبر ثقب منخفض لا بدّ أن تواصل
\emph{نازلة}: فيحطّ الأعلى في أسفل الحاجز. ومن أسفل المشهد صاعدةً
عبر الثقب إلى أعلى الحاجز. فكل نقطة من المشهد ترسل خيطها عبر
البوّابة الصغيرة الواحدة، وتتقاطع الخيوط عند البوّابة، وتتركّب
الصورة معكوسة --- الأعلى مكان الأسفل، واليمين مكان اليسار.
\end{proposition}

\begin{remark}[علبة قديمة كريمة]\label{rem:g6:light-propagation:history}
الغرفة ذات الثقب --- \emph{الغرفة المظلمة} --- أقدم من التصوير
الفوتوغرافي بقرون: فقد راقب الفلكيون الكسوف على حاجزها في أمان،
واقتفى الرسّامون صورها ليضبطوا المنظور. وكل آلة تصوير حديثة،
وعينك أنت، تحفظ عمارتها: بوّابة صغيرة \omterm{def:g1:light-and-shadows:source}{للضوء}، وحاجز للصورة ---
ونعم، الصورة على قعر عينك مقلوبة هي أيضًا؛ ودماغك يقلبها في
أدب. فقانون خطّ مستقيم واحد، مُتتبَّعًا بصبر، يفسّر آلة تصوير
الأسرة ونصف تاريخ صناعة الصور.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:1}
قل قانون \omterm{prop:g6:light-propagation:law}{الانتشار المستقيم} بكلماتك، وفكّ معنى الكلمتين
الطويلتين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:2}
سمِّ أشكال الحزم الثلاثة، مع منبع حقيقي أو موقف لكل واحد. وبأيّ
شكل يصل \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس، ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:3}
يسدّد بنّاء النظر على طول حافة رفّ ليتحقّق من استقامته. فما الذي
يقوم بدور المسطرة في هذا الاختبار؟ وبأيّ قانون يوثق؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:4}
في إنشاء \omterm{def:g1:light-and-shadows:shadow}{الظلّ}، ما الذي يحدّد حافة \omterm{def:g1:light-and-shadows:shadow}{الظلّ} على الحاجز؟ وما الذي
يصحّ في حقّ عين موضوعة داخل مخروط \omterm{def:g1:light-and-shadows:shadow}{الظلّ}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:5}
باستعمال \cref{ex:g6:light-propagation:sizes}: تقف الكرة في
منتصف المسافة بين \omterm{def:g3:first-electric-circuit:bulb}{المصباح} والحاجز. وعرض الكرة \qty{6}{cm} ---
فكم عرض ظلّها؟ وماذا لو اقتربت الكرة من الحاجز؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:6}
في آلة التصوير ذات الثقب، أين يحطّ \omterm{def:g1:light-and-shadows:source}{الضوء} الآتي من أعلى المشهد
على الحاجز؟ تتبّع السبب في جملة واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:7}
يلوّح صديق بيده اليمنى أمام آلة تصويرك ذات الثقب. فأيّ جهة
تلوّح على الحاجز، وفي أيّ وضع؟ (وانتبه --- انعكاسان.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:light-propagation:8}
اذكر مستعملَين تاريخيَّين أو حديثَين للغرفة المظلمة، والعمل
الذي أدّته لكل واحد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:light-propagation:9}
أنشِئ (ارسم بالمسطرة): \omterm{def:g3:first-electric-circuit:bulb}{مصباح} على اليسار، وحاجب على شكل قطعة
نقود، وحاجز على اليمين --- ثم أبعِد \emph{\omterm{def:g3:first-electric-circuit:bulb}{المصباح}} إلى ضعف
بُعده عن القطعة، والحاجز ثابت. فماذا يحدث \omterm{def:g6:mass-and-volume:volume}{لحجم} \omterm{def:g1:light-and-shadows:shadow}{الظلّ}، بحسب
رسمك أنت؟ ووفّق ذلك مع مروحة الشعاعين الماسّين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:light-propagation:10}
في يوم مشمس تكون ظلال الأشجار حادّة؛ وتحت سماء ملبّدة تكاد
الظلال تختفي. اشرح السماءين بالأشعّة والمنابع --- أيّ ``\omterm{def:g3:first-electric-circuit:bulb}{مصباح}''
هي طبقة غيوم رمادية كاملة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:light-propagation:11}
تسطع صورة آلة التصوير ذات الثقب حين يُوسَّع الثقب --- لكنها
تُغبَّش. اشرح الأثرين بالخيوط عبر البوّابة: ما الذي تدع بوّابة
واسعة كل نقطة من المشهد ترسمه على الحاجز؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:light-propagation:12}
تحت شجرة وارفة أثناء كسوف جزئي للشمس تمتلئ الأرض بمئات
الأهلّة الصغيرة. ركّب التفسير من هذا الفصل: ما الفجوات بين
الأوراق، وما كل بقعة ساطعة في يوم عادي، ولماذا تصير البقع
أهلّة حين تصير الشمس هلالًا؟ (وسيفخر بك فصل الكسوف في السنة
القادمة.)
\end{exercise}

\section{مسألة: مسرح الظلال}

\begin{problem}[{مسألة نهاية الأسبوع --- مسرح ظلال عائلي؛ دمى
مقيسة بالتناسب، وختام بالثقب؛ خطوط الضوء المستقيمة تدير
العرض}]\label{pb:g6:light-propagation:1}
شرشف يصلح حاجزًا، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} صغير ساطع، ودمى من ورق مقوّى: ليلة
الافتتاح يوم السبت، وطاقم الخشبة يعمل بالمساطر.

\medskip\noindent\textbf{الجزء الأول --- إعداد الخشبة.}
يقف \omterm{def:g3:first-electric-circuit:bulb}{المصباح} على بعد \qty{4}{m} من الحاجز.
\begin{enumerate}
  \item دمية مُسنَدة إلى الحاجز تلقي ظلًّا بحجمها هي بالضبط.
        فلماذا؟
  \item ودمية التنّين ارتفاعها \qty{20}{cm}. فإذا أُمسكت في
        المنتصف --- على \qty{2}{m} من \omterm{def:g3:first-electric-circuit:bulb}{المصباح} ومن الحاجز ---
        فكم يبلغ ارتفاع ظلّها؟
  \item وفي الختام يجب أن يشمخ التنّين \qty{80}{cm}. فيُزلقه
        الطاقم نحو \omterm{def:g3:first-electric-circuit:bulb}{المصباح}: وعلى \omterm{def:g2:measuring-length:units}{متر} واحد من \omterm{def:g3:first-electric-circuit:bulb}{المصباح} ينفرج
        الشعاعان الماسّان أربعة أمثال \omterm{def:g6:mass-and-volume:volume}{حجم} الدمية. تحقّق: أتوافق
        \qty{80}{cm}، وعند أيّ كسر من المسافة بين \omterm{def:g3:first-electric-circuit:bulb}{المصباح}
        والحاجز تقف الدمية؟
  \item ولماذا يجب أن يكون \omterm{def:g3:first-electric-circuit:bulb}{مصباح} المسرح صغيرًا عاريًا لا كرة
        مصنفرة عريضة؟ (وماذا كانت الكرة ستفعل بحواف التنّين
        الحادّة؟)
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- فنّ الخشبة
بالأشعّة.}
\begin{enumerate}[resume]
  \item يجب أن يكبر الذئب كبرًا مخيفًا في مشهده من غير أن تظهر
        ذراع \omterm{ex:g6:circuits-and-safety:motor}{المحرّك}. فإلى أيّ جهة تُحرّك الدمية، وماذا يحدث
        لحواف ظلّها وهو يكبر؟ (تلين الحواف قليلًا --- فاقبل ذلك
        واشرحه \omterm{def:g6:mass-and-volume:volume}{بحجم} \omterm{def:g3:first-electric-circuit:bulb}{المصباح} الصغير لا النقطي.)
  \item ودميتان مُمسكتان على البعد نفسه يجب ألّا تُظلّ إحداهما
        الأخرى. فما الذي يجب أن يصحّ في الخطوط المستقيمة من
        \omterm{def:g3:first-electric-circuit:bulb}{المصباح} عبر كل دمية؟
  \item وتنزلق يد بين \omterm{def:g3:first-electric-circuit:bulb}{المصباح} والتنّين عند ثلث المسافة من
        \omterm{def:g3:first-electric-circuit:bulb}{المصباح}. \omterm{def:g1:light-and-shadows:shadow}{فظلّ} مَن يظهر أكبر على الحاجز، اليد أم
        التنّين، ولماذا؟
  \item ويجب أن يختفي الشرّير في الحال. أعطِ، والأشعّة في يدك،
        طريقتين: واحدة بتحريك الدمية، وأخرى بحاجب ثانٍ
        أقرب.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- ختام الثقب.}
في المشهد الأخير يحوّل الطاقم المسرح نفسه غرفةً مظلمة: \omterm{def:g3:first-electric-circuit:bulb}{مصباح}
شارع في الخارج، وثقب دبّوس في المصراع، وشرشف حاجز عبر الغرفة.
\begin{enumerate}[resume]
  \item يتعلّق \omterm{def:g3:first-electric-circuit:bulb}{مصباح} الشارع عاليًا في الخارج. فأين تحطّ صورته
        على الشرشف --- عاليًا أم منخفضًا؟ ولماذا؟
  \item ويمرّ راكب درّاجة متأخّر من اليسار إلى اليمين. فإلى
        أيّ جهة تنزلق صورته على الشرشف؟
  \item ويريد الجمهور صورة أسطع، فيوسّع الطاقم ثقب الدبّوس حتى
        يصير فتحة \omterm{def:g6:mass-and-volume:volume}{بحجم} قطعة نقود. فماذا يصير العرض، ولماذا؟
  \item وسطر الختام في كرّاسة البرنامج: سمِّ في جملة واحدة
        القانونَ الواحد الذي أدار السهرة كلها --- المسرحين معًا،
        وكل الظلال، والشارع المقلوب.
\end{enumerate}
\end{problem}
