\chapter{وصف الحركة: المسار والسرعة}\label{ch:g6:describing-motion}

ورقة ساقطة، وقطار عالي \omterm{def:g5:speed-distance-time:speed}{السرعة}، وحصان دوّامة، وركلة جزاء: العالم
لا يسكن أبدًا. ولدراسة الحركة تسأل الفيزياء كل متحرّك سؤالين
مرتّبين: \emph{على أيّ طريق؟} و\emph{بأيّ \omterm{def:g5:speed-distance-time:speed}{سرعة} عليه؟} الطريق
والوتيرة --- \omterm{def:g6:describing-motion:trajectory}{المسار} \omterm{def:g5:speed-distance-time:speed}{والسرعة} --- هما نظّارتا هذا الفصل.

\section{المسار: الطريق المسلوك}

\begin{definition}[المسار]\label{def:g6:describing-motion:trajectory}
\emph{مسار}\index{مسار} جسم متحرّك هو الخطّ الذي ترسمه رحلته ---
أي الطريق الذي يخطّه في الفضاء. وثلاثة أشكال تغطّي أكثر الحياة
اليومية: \emph{مستقيم}\index{مسار مستقيم} (المصعد، والتفاحة
الساقطة)، و\emph{دائري}\index{مسار دائري} (حصان الدوّامة، وطرف
عقرب الساعة)، و\emph{منحنٍ} بحرّية أكبر (قوس الكرة المقذوفة،
وانقضاض السنونو).
\end{definition}

\begin{example}[مسار مَن؟]\label{ex:g6:describing-motion:whose}
انتبه إلى تسمية \emph{أيّ نقطة} تتابع. فأنف حصان الدوّامة يرسم
دائرة؛ ومركز الدوّامة كلها لا يرسم شيئًا البتة --- فهو ثابت.
وصُرّة عجلة تتدحرج تنزلق في خطّ مستقيم بينما ترسم نقطة على
إطارها قفزات رشيقة. آلة واحدة ومسارات كثيرة: والفيزياء تتابع
نقطة واحدة في المرة الواحدة.
\end{example}

\begin{remark}[متحرّك بالقياس إلى ماذا؟]\label{rem:g6:describing-motion:relative}
مسافر ينعس في القطار لا يرسم مسارًا البتة عند المسافر الذي
يقابله --- ويرسم خطًّا طوله مئة كيلومتر على الخريطة عند بقرة
تراقب من الحقل. والوصفان كلاهما أمين: فالحركة دائمًا
\emph{حركة \omterm{def:g6:measurement-in-science:measuring}{بالقياس} إلى شيء}، وعلى الواصف أن يقول ما هو. ونحن
الآن نقارن كل شيء بالأرض في هدوء؛ أما \omterm{def:g2:pushes-and-pulls:force}{قوة} اختيار وجهات نظر أخرى
كاملةً فقصة مشهورة محفوظة لمجلّد المرحلة الثانوية.
\end{remark}

\section{منتظمة أم متغيّرة؟}

\begin{definition}[الحركة المنتظمة والمتغيّرة]\label{def:g6:describing-motion:uniform}
تكون الحركة \emph{منتظمة}\index{حركة منتظمة} حين تقطع مسافات
متساوية في أزمنة متساوية --- كمشية ثابتة على وقع ميقاتية، وسير
قطار بين محطّتين. وتكون \emph{متغيّرة}\index{حركة متغيّرة} حين
تتبدّل الوتيرة: إسراعًا (العدّاء عند الانطلاق)، أو إبطاءً
(الحافلة تكبح)، أو الأمرين تناوبًا (وقوف السير في المدينة
وانطلاقه بلا نهاية).
\end{definition}

\begin{proposition}[قراءة الشريط المنقّط]\label{prop:g6:describing-motion:dots}
صوّر جسمًا متحرّكًا عند دقّات زمنية متساوية --- لقطات سريعة
كثيرة في صورة واحدة --- فتوقّع الحركة اسمها بنفسها نقاطًا:
\begin{enumerate}
  \item نقاط متساوية التباعد: \omterm{def:g6:describing-motion:uniform}{حركة منتظمة}؛
  \item نقاط تتباعد على طول الطريق: إسراع؛
  \item نقاط تتزاحم: إبطاء.
\end{enumerate}
فبين كل زوج من النقاط زمن متساوٍ، فالفجوات بين النقاط تقارن
المسافات --- والتشخيص كله بنظرة واحدة.
\end{proposition}

\begin{method}[كيف تصنع شريطًا منقّطًا]\label{met:g6:describing-motion:film}
تستطيع أي آلة تصوير في هاتف أن تنتج هذه الصور:
\begin{enumerate}
  \item صوّر الجسم المتحرّك --- صديقًا يمرّ على درّاجته، أو كرة
        تتدحرج عن منحدر --- وأمسك آلة التصوير ساكنة؛
  \item وتنقّل في المقطع، واقفًا عند خطوات متساوية (كل نصف ثانية
        من العرض مثلًا)؛
  \item وعند كل وقفة علّم موضع الجسم على ورقة شفّافة فوق
        الشاشة، أو سجّله في مواجهة أعمدة سياج في الخلفية؛
  \item وصُفّ علاماتك في صفّ واحد واقرأها
        وفق \cref{prop:g6:describing-motion:dots}.
\end{enumerate}
وتحذير واحد من المحترفين: أمسك آلة التصوير ساكنة --- فآلة
منجرفة تضيف حركتها هي إلى الشريط وتخلط \omterm{def:g6:describing-motion:trajectory}{مسار} مَن \omterm{def:g6:describing-motion:trajectory}{بمسار} مَن (وهو
تحذير \cref{rem:g6:describing-motion:relative} بعينه).
\end{method}

\section{وضع الأعداد على الوتيرة}

\begin{example}[السرعة بالتناسب]\label{ex:g6:describing-motion:proportion}
ترام في \omterm{def:g6:describing-motion:uniform}{حركة منتظمة} يقطع \qty{9}{km} في $20$ دقيقة. فأيّ \omterm{def:g5:speed-distance-time:speed}{سرعة}
هذه، بالكيلومترات في الساعة المألوفة؟ يجيب التناسب بلا آلة
جديدة: فالساعة ثلاث حصص من $20$ دقيقة، فيقطع الترام
$3 \times 9 = \qty{27}{km}$ في ساعة واحدة --- فسرعته $27$
كيلومترًا في الساعة. وفي جدول:
\begin{center}
\begin{tabular}{l|cc}
الدقائق & $20$ & $60$ \\
\hline
الكيلومترات & $9$ & $27$
\end{tabular}
\end{center}
\omterm{def:g6:describing-motion:uniform}{فالحركة المنتظمة} \emph{هي} تناسب بين المسافة والزمن --- وقد
تصافح منهاجُ الرياضيات ومنهاجُ الفيزياء عندك للتوّ.
\end{example}

\begin{example}[لوحة الوتائر]\label{ex:g6:describing-motion:paces}
وتائر يومية للحكم على أي جواب: التنزّه، نحو $4$ كيلومترات في
الساعة؛ والمشي النشيط، $6$؛ وركوب الدرّاجة الهيّن، $15$؛ وحافلة
مدينة بمحطّاتها، $20$؛ وسيارة على طريق مفتوح، $90$؛ وقطار عالي
\omterm{def:g5:speed-distance-time:speed}{السرعة}، حتى $300$. فإذا سلّمك حساب ماشيًا يقطع $40$ كيلومترًا في
الساعة ردّته لوحة الوتائر.
\end{example}

\begin{remark}[السرعة في لحظة]\label{rem:g6:describing-motion:instant}
\omterm{def:g6:describing-motion:uniform}{الحركة المتغيّرة} ليس لها \omterm{def:g5:speed-distance-time:speed}{سرعة} واحدة --- فالحافلة الكابحة سريعة،
ثم بطيئة، ثم واقفة. وإبرة عدّاد \omterm{def:g5:speed-distance-time:speed}{سرعة} السيارة تجيب عن سؤال
أدقّ: \emph{كم \omterm{def:g5:speed-distance-time:speed}{السرعة} الآن؟} وتعريف تلك \omterm{def:g5:speed-distance-time:speed}{السرعة} اللحظية تعريفًا
دقيقًا عميق حقًّا --- إذ يحتاج إلى رياضيات ستلقاها في السنة
الأخيرة من الثانوية، وقد فتح واحدًا من أعظم فصول العلم كله.
أما هذه السنة: فالحركات المنتظمة تنال عددًا أمينًا واحدًا؛
والحركات المتغيّرة تنال قصة.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:10}
خطّ مترو طوله \qty{9}{km} من طرف إلى طرف. ويسمح الجدول الزمني

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:describing-motion:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:describing-motion:12}

المرحلة الثانوية.)
\end{exercise}

\section{مسألة: الشريط المنقّط للخطّ 7}

\begin{problem}[{مسألة نهاية الأسبوع --- مهندس نقل يقرأ الأشرطة
المنقّطة لخطّ ترام؛ تشخيص الرحلة، وتوقيت الخطّ، وضبط كذبة الجدول
الزمني}]\label{pb:g6:describing-motion:1}
صوّر مهندس نقل جولات ترام بآلة تصوير على جانب الطريق تعلّم موضع
الترام عند كل دقّة (الدقّة $=$ $10$ ثوانٍ)، وسلّمك الأشرطة.

\medskip\noindent\textbf{الجزء الأول --- قراءة الأشرطة.}
الشريط A، بين محطّتين، يُظهر فجوات (بالأمتار):
$50, 50, 50, 50, 50$.
\begin{enumerate}
  \item شخّص الحركة في الشريط A.
  \item وكم قطع الترام خلال الخمسين ثانية في الشريط A؟
  \item والشريط B، عند مغادرة محطّة، يُظهر فجوات
        $10, 25, 40, 50, 50$. احكِ قصة الرحلة بالكلمات.
  \item وفي الشريط B، في أيّ الفترات بين الدقّات تكون \omterm{def:g6:describing-motion:uniform}{الحركة
        منتظمة} (أو تكاد)؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- الوتيرة الحقيقية
للخطّ.}
\begin{enumerate}[resume]
  \item على الامتداد المنتظم يقطع الترام \qty{50}{m} كل $10$
        ثوانٍ. ابنِ الجدول حتى الدقيقة الواحدة: فكم يقطع في
        الدقيقة؟
  \item وواصِل إلى الساعة: فما \omterm{def:g5:speed-distance-time:speed}{سرعة} الترام في السير
        بالكيلومترات في الساعة؟
  \item والخطّ كله \qty{12}{km}. فكم دقيقة من طرف إلى طرف \omterm{def:g5:speed-distance-time:speed}{بسرعة}
        السير من غير وقفات؟
  \item والجدول الزمني يسمح \omterm{def:g2:measuring-time:duration}{بمدة} $50$ دقيقة من طرف إلى طرف. فكم
        دقيقة يخسرها الخطّ الحقيقي في المحطّات والإشارات
        الحمراء والمناطق البطيئة مجتمعةً؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- الإعلان في قفص
الاتهام.}
تتباهى ملصقة الشركة: ``الخطّ 7 --- عبر المدينة كلها \omterm{def:g5:speed-distance-time:speed}{بسرعة} $18$
كيلومترًا في الساعة، ضِعف أحسن ما يبلغه ماشٍ!''
\begin{enumerate}[resume]
  \item تحقّق من حساب الملصقة في مواجهة الجدول الزمني:
        \qty{12}{km} في $50$ دقيقة --- ابنِ الجدول نحو الساعة
        الواحدة. فهل ``$18$ كيلومترًا في الساعة'' أمين
        \emph{للرحلة كلها}؟
  \item ويعترض متزمّت: ``لا يظهر عدد من هذا النوع على عدّاد
        \omterm{def:g5:speed-distance-time:speed}{السرعة} أبدًا --- فالترام إمّا يسير عند $18$ وإمّا واقف
        عند $0$!'' اشرح ما الذي تصفه \omterm{def:g5:speed-distance-time:speed}{سرعة} الرحلة كلها حقًّا،
        ولماذا تظلّ العدد العادل لرحلة ذات وتائر كثيرة.
  \item ويقترح المهندس تخطّي محطّتين لتوفير أربع دقائق. فما
        الزمن الجديد من طرف إلى طرف، و--- بالجدول من جديد ---
        ما \omterm{def:g5:speed-distance-time:speed}{سرعة} الرحلة كلها الجديدة بالكيلومترات في الساعة؟
        (قرِّب تقريبًا معقولًا.)
  \item واكتب تقرير المهندس في جملة واحدة: بأيّ عدد ينبغي أن
        يثق الركّاب في التخطيط، \omterm{def:g5:speed-distance-time:speed}{سرعة} السير أم \omterm{def:g5:speed-distance-time:speed}{سرعة} الرحلة كلها،
        ولماذا؟
\end{enumerate}
