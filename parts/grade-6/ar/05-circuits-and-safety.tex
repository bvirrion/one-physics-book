\chapter{الدارات الكهربائية والسلامة}\label{ch:g6:circuits-and-safety}

ثلاث سنوات من الحلقات والمصابيح والفروع جعلتك بانيًا للدارات.
وهذه السنة تصير شيئًا أندر: \emph{كاتبًا} للدارات. فالكهربائيون
في كل بلد يقرأون اللغة الصامتة نفسها ويرسمونها --- وقواعد تلك
اللغة، مع فصل من حكمة السلامة المكتسبة بثمن غالٍ، هي ما يقف بين
التوصيل الذكي والأصابع المحروقة.

\section{اللغة الصامتة}

\begin{definition}[المخطط الكهربائي]\label{def:g6:circuits-and-safety:diagram}
\emph{المخطط الكهربائي}\index{مخطط كهربائي} هو الرسم المعياري
للدارة: يُبيَّن فيه كل عنصر \emph{برمزه}\index{رمز} المتّفق
عليه، وتُبيَّن الأسلاك خطوطًا مستقيمة بزوايا قائمة. والمخطط
يُظهر كيف \emph{تُوصَّل} الدارة --- مَن يغذّي مَن، وأين يتشعّب
ماذا --- ويهمل عن قصد كيف تتلوّى الأسلاك الحقيقية على الطاولة.
\end{definition}

\begin{method}[من الطاولة إلى الورق، والعكس]\label{met:g6:circuits-and-safety:translate}
\emph{لترسم} دارة مبنيّة:
\begin{enumerate}
  \item أحصِ العناصر، واختر \omterm{def:g6:circuits-and-safety:diagram}{رمز} كل واحد؛
  \item وتتبّع الحلقة الحقيقية بإصبعك من قطب \omterm{def:g3:first-electric-circuit:battery}{البطارية} $+$،
        مسجّلًا ترتيب العناصر وكل تشعّب؛
  \item وأعِد الرسم مستطيلًا مرتّبًا: الرموز متباعدة، والأسلاك
        مستقيمة، والزوايا قائمة --- التوصيلات نفسها بلا
        تلوٍّ.
\end{enumerate}
و\emph{لتبني} من مخطط، أجرِ الترجمة بالمقلوب --- ثم تحقّق بعدها
كالعادة: فكل حلقة متتبَّعة من $+$ يجب أن تبلغ $-$ عبر عنصر واحد
على الأقلّ.
\end{method}

\begin{example}[الدارة نفسها، وصورتان]\label{ex:g6:circuits-and-safety:portraits}
على الطاولة: \omterm{def:g3:first-electric-circuit:battery}{بطارية}، وحزمة متشابكة من أسلاك بمشابك تمساحية
تلتفّ فوق مقلمة، \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} مسنَد إلى كتاب، وقاطعة متدلّية. وعلى
الورق: مستطيل هادئ --- \omterm{def:g3:first-electric-circuit:battery}{البطارية} على اليسار، \omterm{ex:g3:first-electric-circuit:switch}{والقاطعة} في الأعلى،
\omterm{def:g3:first-electric-circuit:bulb}{والمصباح} على اليمين. والحزمة المتشابكة والمستطيل هما \emph{الدارة
نفسها}: فكل توصيلة مطابقة. ولتقرّر أالدارتان واحدة أم لا، اسأل
سؤالًا واحدًا لا غير: مَن مُوصَّل بمَن؟
\end{example}

\begin{example}[المحرّكات والطنّانات تدخل اللعبة]\label{ex:g6:circuits-and-safety:motor}
وافدان جديدان إلى عُدّتك يتصرّفان في المخطط تصرّف المصابيح
تمامًا: \emph{المحرّك}\index{محرّك} (ورمزه: حرف M في دائرة)
يدور كلما جرى فيه تيار --- المراوح، وسيارات اللعب، والمثاقب؛
و\emph{الطنّان}\index{طنّان} (دائرة مضروبة بخطّ) يُصوّت كلما
غُذّي --- أجراس الأبواب، وأجهزة الإنذار. وقواعد التسلسل والتفرع
تنطبق عليهما بلا تغيير: فقاطعة \omterm{def:g4:series-circuits:series}{على التسلسل} تأمرهما؛ والفروع
المتفرّعة تجعلهما مستقلّين.
\end{example}

\begin{example}[قراءة مخطط قراءة محترف]\label{ex:g6:circuits-and-safety:reading}
أمام أي مخطط، استنطقه بمشية الإصبع من فصل الدارات المتفرّعة: من
$+$، امشِ كل طريق إلى $-$. فالعناصر كلها على طريق واحد: تسلسل

تغيّرت اللغة؛ ولم تتغيّر القوانين.
\end{example}

\section{فصل الندوب}

قواعد السلامة الكهربائية دُفع ثمنها حرائقَ وأسوأ. وإليكها،
بأسبابها --- فالمحترف يعرف الاثنين.

\begin{definition}[الدارة القصيرة]\label{def:g6:circuits-and-safety:short}
تحدث \emph{الدارة القصيرة}\index{دارة قصيرة} حين يصل طريق \omterm{def:g4:conductors-insulators:conductor}{ناقل}
عارٍ بين جانبَي \omterm{def:g3:first-electric-circuit:battery}{بطارية} (أو مأخذ) \emph{مباشرة}، متخطّيًا كل
عنصر. فبلا شيء نافع في طريقه يندفع التيار اندفاعًا جامحًا ---
فتسخن الأسلاك، \omterm{def:g2:ice-water-steam:meltfreeze}{وينصهر} العزل، وتتلف البطاريات، وقد تشتعل أسلاك
الشبكة. ويدرس فصل الدارات في السنة القادمة هذا الشرّير عن قرب؛
أما هذه السنة فتعلّم وجهه وأبعده عن بناياتك.
\end{definition}

\begin{remark}[حرّاس البيت]\label{rem:g6:circuits-and-safety:guards}
انظر في علبة المصاهر في البيت: صفوف من قواطع صغيرة ---
\emph{قواطع الدارة} --- واحد لكل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} من الدارة المتفرّعة الكبرى
في البيت. ويقف كل واحد حارسًا على فرعه: فإن سمح عطل للتيار
بالاندفاع --- \omterm{def:g6:circuits-and-safety:short}{دارة قصيرة}، أو مأخذ محمَّل فوق طاقته --- قطع
الحارسُ نفسه في طرفة عين، فقطع \omterm{def:g5:series-parallel-circuits:parallel}{الفرع} قبل أن تفرط الأسلاك في
السخونة. وحين ``يقفز'' قاطع فليس ذلك سوء أدب منه: بل هو قد
أدّى عمله الوحيد. فجِد العطل قبل أن تعيده.
\end{remark}

\begin{remark}[القواعد، بأسبابها]\label{rem:g6:circuits-and-safety:rules}
\begin{enumerate}
  \item \emph{تجارب البطاريات وحدها؛ وأما المأخذ فأبدًا.} فشبكة
        الكهرباء تدفع أشدّ بمئات المرات --- وعبر جلد رطب قد
        توقف قلبًا.
  \item \emph{لا سلك عارٍ يمتدّ مباشرة عبر \omterm{def:g3:first-electric-circuit:battery}{بطارية}.} فتلك \omterm{def:g6:circuits-and-safety:short}{دارة
        قصيرة}: خلايا تالفة، وسلك ساخن، وأصابع محروقة.
  \item \emph{الماء وكهرباء الشبكة لا يتقاسمان متناول غرفة
        واحدة:} فلا أجهزة قرب الحمّام، ولا أيدٍ مبتلّة على
        القوابس --- فالجلد الرطب \omterm{def:g4:conductors-insulators:conductor}{ناقل}.
  \item \emph{لا تُدخل شيئًا في مأخذ أبدًا.} فالحارس في علبة
        المصاهر سريع، لكن يجب ألّا تراهن عليه أبدًا.
  \item \emph{الأسلاك المتضرّرة إلى سلّة المهملات}، لا إلى
        الدرج: فالعزل المتشقّق سياج فيه فجوة.
\end{enumerate}
\end{remark}

\section{تمارين}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:2}
ارسم من الذاكرة الرموز: \omterm{def:g3:first-electric-circuit:battery}{بطارية}؛ \ \omterm{def:g3:first-electric-circuit:bulb}{مصباح}؛ \ قاطعة مفتوحة؛
\ قاطعة مغلقة؛ \ محرّك؛ \ طنّان. وأيّ علامة تدلّ على جانب
\omterm{def:g3:first-electric-circuit:battery}{البطارية} $+$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:3}
ارسم مخطط جرس باب: \omterm{def:g3:first-electric-circuit:battery}{بطارية} وقاطعة ضغط وطنّان في حلقة واحدة.
ولماذا يتوقّف الجرس حين تُرفع الإصبع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:4}
بُنيت دارتان من علبة القطع نفسها: أهما ``الدارة نفسها''؟ وأيّ
سؤال واحد يحسم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:5}
في مخطط \omterm{def:g3:first-electric-circuit:bulb}{المصباح} \omterm{ex:g6:circuits-and-safety:motor}{والمحرّك} في الفصل، ماذا يحدث للمروحة إن احترق
\omterm{def:g3:first-electric-circuit:bulb}{المصباح}؟ وماذا يحدث للاثنين إن فُتحت \omterm{ex:g3:first-electric-circuit:switch}{القاطعة}؟ وأيّ قواعد قديمة
تجيب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:6}
ما \omterm{def:g6:circuits-and-safety:short}{الدارة القصيرة}، ولماذا تُسخّن الأسلاك؟ وسمِّ اثنين من
ضحاياها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:7}
قفز قاطع دارة في علبة المصاهر. فما الذي فعله للبيت للتوّ؟ وما
الذي ينبغي فعله قبل إعادته؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:circuits-and-safety:8}
أعطِ السبب وراء كل قاعدة: لا أجهزة قرب الحمّام؛ \ الأسلاك
المتضرّرة إلى سلّة المهملات؛ \ ألعاب البطاريات نعم والمأخذ
أبدًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:circuits-and-safety:9}
ترجِم إلى مخطط (بالكلمات أو بالرسم): \omterm{def:g3:first-electric-circuit:battery}{بطارية}؛ ومن قطبها $+$ سلك
إلى قاطعة مغلقة؛ ومن \omterm{ex:g3:first-electric-circuit:switch}{القاطعة} تشعّب --- \omterm{def:g5:series-parallel-circuits:parallel}{فرع} عبر طنّان، \omterm{def:g5:series-parallel-circuits:parallel}{وفرع} عبر
\omterm{def:g3:first-electric-circuit:bulb}{مصباح}؛ ثم يلتقي الفرعان ويعودان إلى $-$. توقّع ما يُسمع وما
يُرى، وما الذي يتغيّر إن نال \omterm{def:g5:series-parallel-circuits:parallel}{فرع} \omterm{ex:g6:circuits-and-safety:motor}{الطنّان} وحده قاطعة إضافية
خاصة به، مفتوحة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:circuits-and-safety:10}
وُصّلت مروحة لعبة: من قطب \omterm{def:g3:first-electric-circuit:battery}{البطارية} $+$ إلى \omterm{ex:g6:circuits-and-safety:motor}{المحرّك}، ومن \omterm{ex:g6:circuits-and-safety:motor}{المحرّك}
إلى القطب $-$، و--- ``لمتانة التوصيل'' --- سلك عارٍ إضافي واحد
مباشرة من $+$ إلى $-$. فتدور المروحة بالكاد ويسخن السلك
الإضافي. شخّص الحال بما في \cref{def:g6:circuits-and-safety:short}:
أيّ طريق يفضّله التيار، ولماذا كان البناء خطِرًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:circuits-and-safety:11}
حيلة الجدّ في ورشته: قبل أن يلمس أي تجهيزة إنارة يطفئ قاطعتها
الجدارية \emph{ويطفئ} قاطع فرعها كله في علبة المصاهر. اشرح
بتفكير التسلسل ما الذي يحقّقه كل قطع --- ولماذا يفعل الرجل
المتأنّي الاثنين.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:circuits-and-safety:12}
صمّم توصيل مسرح دمى: قاطعة رئيسية واحدة تُظلم كل شيء؛ ومصباحا
خشبة يجب أن يتعطّلا مستقلّين؛ ومحرّك ستارة وطنّان بدء العرض،
لكل واحد قاطعته الخاصة. صِف (أو ارسم) المخطط --- أيّ العناصر
يركب الطريق المشترك، وأيّها يركب الفروع، وأين تجلس كل قاطعة ---
وتحقّق من تصميمك في مواجهة كل قواعد التسلسل والتفرع التي
تملكها.
\end{exercise}

\section{مسألة: توصيل كوخ الحديقة}

\begin{problem}[{مسألة نهاية الأسبوع --- أسرة توصّل كوخ حديقتها؛
المخططات قبل المفكّات؛ قائمة تحقّق
المفتّش}]\label{pb:g6:circuits-and-safety:1}
تخطّط الأسرة شبكة الكوخ الصغيرة العاملة \omterm{def:g3:first-electric-circuit:battery}{بالبطارية} على الورق
أولًا --- كما يفعل المحترفون.

\medskip\noindent\textbf{الجزء الأول --- الخطة على الورق.}
قائمة الأمنيات: \omterm{def:g3:first-electric-circuit:bulb}{مصباح} سقف تأمره قاطعة عند الباب؛ \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} طاولة
عمل بقاطعته الخاصة؛ ومروحة تهوية صغيرة (محرّك) بقاطعتها
الخاصة.
\begin{enumerate}
  \item لماذا يجب أن تركب العناصر الثلاثة \omterm{def:g5:series-parallel-circuits:parallel}{فروعًا} متفرّعة لا
        حلقة تسلسل واحدة؟ اذكر راحتين تنكسران \omterm{def:g4:series-circuits:series}{على التسلسل}.
  \item ويحمل كل \omterm{def:g5:series-parallel-circuits:parallel}{فرع} قاطعته الخاصة. فماذا تأمر كل قاطعة، وما
        الذي لا تأمره أيّ واحدة منها؟
  \item وأين يجب أن تجلس قاطعة رئيسية لتُظلم الكوخ كله وتُسكته
        دفعة واحدة؟
  \item وارسم (أو صِف بدقّة) المخطط كاملًا: \omterm{def:g3:first-electric-circuit:battery}{بطارية}، وقاطعة
        رئيسية، وثلاثة فروع، وأربع قواطع في المجموع.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- البناء، ومفتَّشًا.}
\begin{enumerate}[resume]
  \item بعد البناء والتشغيل يسطع \omterm{def:g3:first-electric-circuit:bulb}{مصباح} السقف --- لكن خافتًا،
        ولا يسطع \omterm{def:g3:first-electric-circuit:bulb}{مصباح} طاولة العمل إلا حين تكون قاطعة
        \emph{المروحة} مغلقة. وتجد مشية الإصبع أن \omterm{def:g3:first-electric-circuit:bulb}{مصباح} الطاولة
        والمروحة يتقاسمان فرعًا واحدًا، واحدًا خلف الآخر. سمِّ
        ترتيبهما، واشرح العرضين بالقواعد القديمة.
  \item أعطِ الإصلاح بسلك واحد الذي يعيد الخطة.
  \item وأثناء الترتيب يستلقي مفكّ لحظة عبر قطبَي \omterm{def:g3:first-electric-circuit:battery}{البطارية}
        العاريين فيسخن. سمِّ الحدث، واذكر ما ينبغي أن تفحصه
        الأسرة في \omterm{def:g3:first-electric-circuit:battery}{البطارية} بعد ذلك.
  \item ويضيف الوالد المفتّش قاعدة: ``كل وصلة ملفوفة، ولا نحاس
        عارٍ في أي مكان.'' فمن أيّ خطرين تحرس تلك القاعدة
        الواحدة؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- ليلة العاصفة.}
\begin{enumerate}[resume]
  \item ليلة ماطرة: يرشح سقف الكوخ على طاولة العمل. فلماذا يجب
        ألّا تلمس الأسرة الأسلاك بأيدٍ مبتلّة، ولو كان الكوخ
        يعمل \omterm{def:g3:first-electric-circuit:battery}{ببطارية} --- ولماذا تكون اللمسة نفسها أخطر بكثير في
        البيت الموصول بالشبكة؟
  \item وتخفت أضواء البيت ويقفز قاطع واحد. فما الذي كشفه
        القاطع، وما ترتيب الأفعال الحكيم؟
  \item وفي الصباح تتناقش الأسرة في إضافة جرس باب --- طنّان عند
        الكوخ، وقاطعة ضغط عند البيت، وسلك مزدوج طويل بينهما.
        أعلى التسلسل يكون مع شبكة الكوخ أم \omterm{def:g5:series-parallel-circuits:parallel}{على التفرع}؟ ومن أين
        يأتي تياره؟ ارسم الحلقة.
  \item واكتب ميثاق سلامة الكوخ عند الأسرة في ثلاثة أسطر، سطرًا
        لكل واحد: المآخذ، والماء، والأسلاك العارية.
\end{enumerate}
\end{problem}
