\chapter{منظومة الشمس والأرض والقمر}\label{ch:g6:sun-earth-moon}

ثلاثة عوالم تدير سماءنا: الشمس التي تمنح النهار، والأرض التي
تحملنا، \omterm{def:g2:moon-in-the-sky:moon}{والقمر} الذي يرافقنا. وأنت تعرف كل واحد منها؛ والآن نرسم
خريطة \emph{المنظومة} --- مَن يدور حول مَن، وبأيّ \omterm{def:g5:speed-distance-time:speed}{سرعة}، وعلى أيّ
بُعد --- ونسدّد دَينًا قديمًا: فسرّ \omterm{def:g3:sun-days-seasons:year}{فصول السنة} الموعود في هذا
الفصل.

\section{ثلاث حركات وثلاث ساعات}

\begin{definition}[المحور والمدار]\label{def:g6:sun-earth-moon:axisorbit}
\emph{محور}\index{محور} الأرض هو الخطّ المتخيَّل المارّ \omterm{def:g3:first-electric-circuit:battery}{بقطبيها}
والذي تدور حوله، مثل سيخ نحلة. و\emph{المدار}\index{مدار} هو
\omterm{def:g6:describing-motion:trajectory}{المسار} المغلق الذي يقطعه جرم سماوي حول آخر --- ويكاد يكون دائرة
للأرض حول الشمس، وللقمر حول الأرض.
\end{definition}

\begin{proposition}[الحركات الثلاث]\label{prop:g6:sun-earth-moon:motions}
التقويم كله معلَّق بثلاث حركات ثابتة:
\begin{enumerate}
  \item الأرض \emph{تدور} حول محورها: دورة واحدة في $24$
        ساعة --- وهو اليوم، ويستعرض الشمس والنجوم عبر سمائنا؛
  \item والأرض \emph{تدور} حول الشمس: دورة واحدة في نحو $365$
        يومًا ورُبع --- وهي السنة، وتزلق قوس الشمس صعودًا ونزولًا
        عبر \omterm{def:g3:sun-days-seasons:year}{فصول السنة}؛
  \item \omterm{def:g2:moon-in-the-sky:moon}{والقمر} \emph{يدور} حول الأرض: دورة واحدة في نحو شهر ---
        وباستدارة نصفه المضيء نحونا تباعًا يستعرض أطواره في
        مفكّرة \omterm{def:g2:moon-in-the-sky:moon}{القمر} القديمة عندك.
\end{enumerate}
ثلاث حركات وثلاث ساعات: اليوم والشهر والسنة --- أقدم موقّتات
البشر، وكلها ما زالت تعمل.
\end{proposition}

\begin{example}[رُبع اليوم والسنة الكبيسة]\label{ex:g6:sun-earth-moon:leap}
دورة الأرض تستغرق $365$ يومًا \emph{ونحو رُبع} --- فالطبيعة لا
تدين بالولاء للأعداد المستديرة. والتقاويم تسدّد الدَّين بادّخار
الأرباع: فأربع سنوات من أرباع الأيام تصنع يومًا كاملًا، يُضاف
إلى التقويم كل سنة رابعة --- وهي \emph{السنة الكبيسة} ذات
$366$ يومًا، ويومها الزائد في آخر شباط. وأهمِل هذه المحاسبة قرنًا
فينحرف التقويم أسابيع عن \omterm{def:g3:sun-days-seasons:year}{فصول السنة} --- كما اكتشفت تقاويم أقدم،
بألم.
\end{example}

\begin{example}[المنظومة بمقياس]\label{ex:g6:sun-earth-moon:scale}
أعداد للدفتر: قطر الأرض نحو \qty{12800}{km}؛ \omterm{def:g2:moon-in-the-sky:moon}{والقمر} نحو ربع
ذلك، ويدور على بُعد نحو \qty{384000}{km} --- أي ثلاثين أرضًا
مصفوفة جنبًا إلى جنب. والشمس قطرها نحو $109$ أرض وتقف على بعد
$150$ مليون كيلومتر. صغّر الأرض حتى تصير كرة زجاجية سنتيمترية
فيقول النموذج: \omterm{def:g2:moon-in-the-sky:moon}{القمر} حبّة فلفل على بعد $30$ سنتيمترًا؛ والشمس
كرة عرضها أكثر من \omterm{def:g2:measuring-length:units}{متر}، تقف على أكثر من مئة \omterm{def:g2:measuring-length:units}{متر} في الشارع.
فجوارنا كله خواء في معظمه --- كما همس أصلًا نموذج \omterm{def:g4:solar-system:family}{المجموعة
الشمسية} في الحديقة.
\end{example}

\section{سداد الدَّين: لماذا فصول السنة}

\begin{proposition}[الميل يصنع الفصول]\label{prop:g6:sun-earth-moon:tilt}
\omterm{def:g6:sun-earth-moon:axisorbit}{محور} الأرض ليس قائمًا على مدارها: بل يميل بنحو رُبع زاوية قائمة
--- ويظلّ مائلًا \emph{في الجهة نفسها} طوال دورة السنة، مثل
\omterm{def:g4:magnets-and-compass:compass}{بوصلة} وفيّة للنجوم. فنصف الدورة يميل نصف عالمنا \emph{نحو}
الشمس: فتعلو الشمس، وتطول النهارات --- وهو الصيف. وبعد نصف دورة
يشير الميل نفسه غير المتبدّل \emph{بعيدًا}: فشمس منخفضة ونهارات
قصيرة --- وهو الشتاء. فالفصول ليست تغيّرًا في الشمس ولا في
المسافة: إنما هي \omterm{def:g4:solar-system:planet}{كوكب} مائل يحمل ميله غير المتبدّل حول نجمه.
\end{proposition}

\begin{method}[الكرة الأرضية والمصباح]\label{met:g6:sun-earth-moon:globe}
شاهِد الآلة كلها على طاولة:
\begin{enumerate}
  \item ضع مصباحًا في وسط الطاولة شمسًا، واحمل كرة أرضية مائلة
        (أو كرة على سيخ مائل) ودُر بها حوله على مهل،
        \emph{مبقيًا السيخ مشيرًا إلى ركن الغرفة نفسه} طوال
        الطريق؛
  \item وعند كل رُبع من الدورة قف وانظر إلى خطّ عرض بيتك: وراقب
        كم يقع من دائرة دورانه اليومية في \omterm{def:g1:light-and-shadows:source}{الضوء}؛
  \item ففي جهة من \omterm{def:g3:first-electric-circuit:bulb}{المصباح} ينعم نصف كرتك بالميل نحو الداخل ---
        أقواس مضيئة طويلة \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} عالٍ: وهو الصيف؛ وفي المقابلة
        يميل بعيدًا --- أقواس قصيرة \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} منخفض: وهو الشتاء؛
  \item وعند المحطّتين اللتين بينهما لا يميل أي قطب نحو الداخل:
        فيتقاسم النهار والليل بالتساوي --- وهما الربيع
        والخريف.
\end{enumerate}
والانضباط الوحيد الذي يجعل ذلك يعمل: ألّا تدع السيخ يتأرجح ---
فوفاء \omterm{def:g6:sun-earth-moon:axisorbit}{المحور} \emph{هو} الفصول.
\end{method}

\begin{example}[القرائن القديمة، وقد حُوسبت]\label{ex:g6:sun-earth-moon:clues}
كل ملاحظة من سنوات مزولتك ترفع تقريرها الآن إلى الميل. فقوس
الصيف العالي: نصف كرتنا مائل نحو الداخل، فتقف الشمس أعلى عند
الظهر. وقوس الشتاء القصير وظلاله الطويلة: مائل بعيدًا، فتلامس
الشمس الأفق منخفضة. والفصول المتعاكسة عبر خطّ الاستواء ---
عطلات الشاطئ في كانون الأول في الجنوب بينما يجرف الشمال الثلج:
فحين يميل نصف كرة نحو الداخل وجب أن يميل الآخر بعيدًا. بل حتى
تساوي النهار والليل في الربيع والخريف يخرج من مشية \omterm{def:g3:first-electric-circuit:bulb}{المصباح}
والكرة. فميل واحد قدره ربع زاوية قائمة يدير العرض كله.
\end{example}

\begin{remark}[لا المسافة --- والبرهان]\label{rem:g6:sun-earth-moon:notdistance}
\omterm{def:g6:sun-earth-moon:axisorbit}{مدار} الأرض قريب من الدائرة إلى حدّ أن البعد عن الشمس يتغيّر
قليلًا --- بل، وهو أمر طريف، نركب أقرب قليلًا إلى الشمس في
الشتاء الشمالي. فالفصول تعنيها \emph{مَيلة} \omterm{def:g1:light-and-shadows:source}{الضوء} لا طول الرحلة:
فبالميل نحو الداخل يصطدم \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس عموديًّا مركَّزًا؛ وبالميل
بعيدًا ينتشر \omterm{def:g1:light-and-shadows:source}{الضوء} نفسه رقيقًا على الأرض. \omterm{def:g3:first-electric-circuit:bulb}{ومصباح} يدوي مسلَّط
مواجهةً ثم مائلًا على جدار يُظهر الفرق في ثانية واحدة ---
\omterm{def:g1:light-and-shadows:source}{فالضوء} العمودي يعضّ، \omterm{def:g1:light-and-shadows:source}{والضوء} الماسّ لا يكاد يُدفئ.
\end{remark}

\section{مسألة: نموذج ملعب الرياضة}

\begin{problem}[{مسألة نهاية الأسبوع --- الصفّ يبني منظومة الشمس
والأرض والقمر على ملعب الرياضة؛ كرات زجاجية وحبّات فلفل ومشية
طويلة جدًّا؛ والميل ينضمّ إلى النموذج}]\label{pb:g6:sun-earth-moon:1}
يبني الصفّ المنظومة بمقياس: الأرض كرة زجاجية قطرها \qty{1}{cm}.
وأعداد حقيقية مفيدة: قطر الأرض \qty{12800}{km}؛ وقطر \omterm{def:g2:moon-in-the-sky:moon}{القمر}
\qty{3200}{km}، ويدور على بعد \qty{384000}{km}؛ وقطر الشمس $109$
أرض، وبعدها $150$ مليون كيلومتر.

\medskip\noindent\textbf{الجزء الأول --- توزيع الأدوار.}
\begin{enumerate}
  \item في النموذج، كم كيلومترًا يمثّل \qty{1}{cm}؟
  \item \omterm{def:g2:moon-in-the-sky:moon}{والقمر} نحو رُبع عرض الأرض. فكم عرض \omterm{def:g2:moon-in-the-sky:moon}{قمر} النموذج --- وأيّ
        شيء من المطبخ يؤدّي دوره جيدًا؟
  \item وبُعد \omterm{def:g2:moon-in-the-sky:moon}{القمر} الحقيقي $30$ عرضًا للأرض. فأين يقف \omterm{def:g2:moon-in-the-sky:moon}{قمر}
        النموذج من الكرة الزجاجية؟
  \item وشمس النموذج $109$ كرة زجاجية عرضًا. فكم يبلغ ذلك
        بالسنتيمترات --- وأيّ شيء في ساحة اللعب بهذا \omterm{def:g6:mass-and-volume:volume}{الحجم}
        تقريبًا؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- المشية الطويلة.}
\begin{enumerate}[resume]
  \item بُعد الشمس $150$ مليون كيلومتر. ففي النموذج --- اقسم على
        جواب السؤال الأول --- كم سنتيمترًا يكون ذلك، وكم
        مترًا؟
  \item وهل يسع النموذج ملعب رياضة طوله \qty{100}{m}؟ وأين يجب
        أن تقف كرة الشمس؟
  \item ويمشي تلميذ المسافة بين الأرض والشمس \omterm{def:g5:speed-distance-time:speed}{بسرعة} \omterm{def:g2:measuring-length:units}{متر} واحد في
        الثانية. فكم دقيقة تستغرق المشية تقريبًا؟
  \item وواقفًا عند الكرة الزجاجية وناظرًا إلى كرة الشمس التي
        عرضها \omterm{def:g2:measuring-length:units}{متر} على بعد $117$ مترًا، يفاجأ الصفّ: فهي تبدو
        \omterm{def:g6:mass-and-volume:volume}{بالحجم} نفسه تقريبًا الذي تبدو به حبّة \omterm{def:g2:moon-in-the-sky:moon}{القمر} التي عرضها
        رُبع \omterm{def:g2:measuring-length:units}{سنتيمتر} على بعد $30$ سنتيمترًا من الكرة الزجاجية.
        تحقّق من النسبتين --- واربط هذه المصادفة بشيء يحدث،
        نادرًا وبهيًّا، في السماء الحقيقية.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- تشغيل النموذج.}
\begin{enumerate}[resume]
  \item لتحريك النموذج تحريكًا أمينًا، كم ينبغي أن تستغرق الكرة
        الزجاجية لتدور حول كرة الشمس، وكم تستغرق البازلّاء لتدور
        حول الكرة الزجاجية، إذا مثّلت الثانيةُ الواحدة يومًا
        حقيقيًّا واحدًا؟ (احسب تقريبًا: السنة $365$ يومًا،
        ودورة \omterm{def:g2:moon-in-the-sky:moon}{القمر} $27$ يومًا.)
  \item ويجب أن تدور الكرة الزجاجية حول نفسها أيضًا. فبأيّ \omterm{def:g5:speed-distance-time:speed}{سرعة}
        يكون ذلك في النموذج بمعدّل يوم حقيقي في الثانية ---
        ولماذا لن يراه أحد على كرة زجاجية قطرها \qty{1}{cm}؟
  \item وسيخ يخترق الكرة الزجاجية يؤدّي دور \omterm{def:g6:sun-earth-moon:axisorbit}{المحور}. فبأيّ ميل،
        ومشيرًا إلى أين، يجب أن يُحمل السيخ في الدورة كلها؟
  \item واكتب إخلاء مسؤولية النموذج الأمين: سمِّ شيئًا واحدًا
        يُظهره نموذج الملعب بصدق، وشيئًا واحدًا لا يستطيع أي
        نموذج \omterm{def:g6:mass-and-volume:volume}{بحجم} ملعب أن يُظهره في آن واحد (الأحجام والمسافات
        معًا؟ الحركات؟ \omterm{def:g1:light-and-shadows:source}{الضوء}؟). وبرّر بأعدادك.
\end{enumerate}
\end{problem}
