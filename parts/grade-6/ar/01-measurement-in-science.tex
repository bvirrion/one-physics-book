\chapter{القياس في العلم: الوحدات والأجهزة}\label{ch:g6:measurement-in-science}

نجّار وطبيب وفلكيّ يمضون إلى أعمالهم: هذا يقيس رفًّا، وهذا حمّى،
وهذا هيام المريخ. وثلاثتهم يثقون بالآلة الهادئة نفسها --- وحدات
متّفق عليها، وأجهزة أمينة، وقراءة متأنّية. وهذه السنة تصير
الفيزياء كمّية على أصولها، فلنبدأ بصقل الأدوات التي سيستند إليها
كل فصل لاحق.

\section{من الظاهر إلى العدد}

علّمتنا خمس سنوات من التجارب إيقاعًا: العيون والأيدي تحزر،
والأجهزة تقرّر. المسطرة في مواجهة ``يبدو أطول''؛ والميزان في
مواجهة ``يبدو أثقل''؛ \omterm{def:g3:temperature-thermometers:temperature}{وميزان الحرارة} في مواجهة ``أحسّه باردًا''.
وفي كل مرة يكون حكم الجهاز \emph{عددًا مع \omterm{def:g6:measurement-in-science:measuring}{وحدة}} --- وذلك الزوج
يسافر ويقارن ويستطيع أي أحد أن يتحقّق منه، وهو ما لم يستطعه
إحساس قطّ.

\begin{definition}[قياس مقدار]\label{def:g6:measurement-in-science:measuring}
أن \emph{تقيس}\index{قياس} مقدارًا --- طولًا أو \omterm{def:g3:measuring-mass:mass}{كتلة} أو \omterm{def:g2:measuring-time:duration}{مدة} أو
\omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} --- هو أن تعدّ كم مرة تدخل فيه \emph{وحدته}\index{وحدة}
المتّفق عليها، مستعملًا جهازًا مبنيًّا لذلك العمل. والنتيجة
تُكتب دائمًا عددًا \emph{مع} وحدته: \qty{28.5}{cm}،
\qty{450}{g}، \qty{37.2}{\celsius}. فالعدد وحده إشاعة؛ والعدد مع
وحدته قياس.
\end{definition}

\begin{example}[ثمن وحدة ناقصة]\label{ex:g6:measurement-in-science:missing}
``أضِف 3 من الطحين.'' غرامات؟ ملاعق؟ أكواب؟ مصير الكعكة معلّق
بالكلمة الناقصة. وقد تبعت كوارث حقيقية وحداتٍ ناقصة أو مختلطة
--- فقد نفد الوقود بطيّارين في وسط الرحلة لأن الكيلوغرامات
خُلطت بالأرطال. والمحترفون يكتبون \omterm{def:g6:measurement-in-science:measuring}{الوحدة} في كل مرة بلا استثناء؛
ومن هذا الفصل فصاعدًا نفعل مثلهم.
\end{example}

\section{وحدات العالم}

\begin{definition}[عُدّة العالِم الأولى]\label{def:g6:measurement-in-science:units}
اتّفق العالم على وحدات مشتركة، أسرة لكل مقدار:
\begin{enumerate}
  \item الطول: \emph{\omterm{def:g2:measuring-length:units}{المتر}} (\unit{m})، وذرّيته الكيلومتر
        ($\qty{1}{km} = \qty{1000}{m}$)، \omterm{def:g2:measuring-length:units}{والسنتيمتر}
        ($\qty{100}{cm} = \qty{1}{m}$)، والمليمتر
        ($\qty{1000}{mm} = \qty{1}{m}$)؛
  \item \omterm{def:g3:measuring-mass:mass}{الكتلة}: \emph{\omterm{def:g3:measuring-mass:units}{الكيلوغرام}} (\unit{kg})، ومعه \omterm{def:g3:measuring-mass:units}{الغرام}
        ($\qty{1}{kg} = \qty{1000}{g}$) والمليغرام
        ($\qty{1000}{mg} = \qty{1}{g}$)؛
  \item الزمن: \emph{الثانية} (\unit{s})، ومعها الدقيقة
        ($\qty{1}{min} = \qty{60}{s}$) والساعة
        ($\qty{1}{h} = \qty{60}{min} = \qty{3600}{s}$)؛
  \item \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة}: \omterm{def:g3:temperature-thermometers:temperature}{الدرجة السلسيوسية} (\unit{\celsius})؛
  \item حجم السوائل: اللتر (\unit{L})، وسنكيله في الفصل
        التالي.
\end{enumerate}
\end{definition}

\begin{example}[التحويل على طريقة العلماء]\label{ex:g6:measurement-in-science:convert}
التحويلات ركوب على السلّم العشري.
فالمقدار $\qty{2.35}{m}$ بالسنتيمترات: اضرب في $100$ --- فتنزلق النقطة
منزلتين إلى اليمين: \qty{235}{cm}. و$\qty{4500}{g}$
بالكيلوغرامات: اقسم على $1000$: \qty{4.5}{kg}. أما الزمن فمتمرّد
الأسرة --- إذ خطواته $60$ لا $10$: فالمقدار $\qty{2.5}{min}$ ليس
$250$ ثانية بل $2 \times 60 + 30 = \qty{150}{s}$. فاحترم
المتمرّد.
\end{example}

\begin{remark}[مَن يحرس المتر؟]\label{rem:g6:measurement-in-science:guard}
كان \omterm{def:g2:measuring-length:units}{متر} العالم قرنين قضيبًا من معدن ثمين في خزانة محروسة قرب
عاصمة أوروبية كبيرة، وكان \omterm{def:g3:measuring-mass:units}{الكيلوغرام} أسطوانة لامعة إلى جانبه:
وكانت مساطر كل أمة وصنجاتها نسخًا عن نسخ عنهما. أما اليوم فقد
أُحيل الحرّاس إلى التقاعد: إذ صار \omterm{def:g2:measuring-length:units}{المتر} وأصحابه معرَّفين من
حقائق الطبيعة نفسها التي لا تتغيّر --- \omterm{def:g1:light-and-shadows:source}{الضوء} وساعات الذرّات
الداخلية --- بحيث يستطيع أي مختبر مجهَّز جيدًا على الأرض (أو
خارجها!) أن يعيد بناءها بالضبط. فالوحدات أطول اتفاق سلمي في
تاريخ البشر.
\end{remark}

\section{حسن استعمال الأجهزة}

\begin{method}[كيف تقرأ أي سلّم مدرَّج]\label{met:g6:measurement-in-science:read}
المساطر وموازين \omterm{def:g5:heat-and-insulation:heat}{الحرارة} وأباريق الكيل والأقراص: مهارة واحدة
تقرؤها كلها.
\begin{enumerate}
  \item جِد \emph{قيمة التدريج}: خذ علامتين مرقّمتين متجاورتين،
        واطرح، واقسم على عدد الخطوات الصغيرة بينهما --- ففي
        مسطرة مرقّمة كل \omterm{def:g2:measuring-length:units}{سنتيمتر} بعشر خطوات تكون كل خطوة
        $1 \div 10 = \qty{0.1}{cm}$، أي مليمترًا واحدًا؛
  \item وضع عينك أمام العلامة تمامًا (فالنظرة المائلة تزيح
        القراءة)؛
  \item وعُدّ من آخر علامة مرقّمة: $7$ وثلاث خطوات صغيرة تعني
        $7.3$؛
  \item واكتب العدد \emph{مع وحدته}.
\end{enumerate}
\end{method}

\begin{method}[قدّر، ثم قِس، ثم احكم]\label{met:g6:measurement-in-science:estimate}
المحترفون لا يقيسون على البارد أبدًا:
\begin{enumerate}
  \item \emph{قدّر} أولًا --- ``هذه الطاولة نحو مترين''؛
  \item \emph{واختر} الجهاز الذي يلائم مداه وخطواته العمل
        (مسطرة لطابع بريد، وشريط \omterm{def:g6:measurement-in-science:measuring}{قياس} لممرّ، وميزان مطبخ
        للطحين، وميزان حمّام للناس)؛
  \item \emph{وقِس}، وفق قواعد القراءة؛
  \item \emph{واحكم}: أتوافق النتيجة التقدير موافقة تقريبية؟
        فطاولة قيسَت \qty{21.3}{m} لم تكبر --- بل أخطأت القراءة
        بعامل عشرة، والتقدير التقطها للتوّ.
\end{enumerate}
والتقدير شبكة أمانك: يلتقط النقاط العشرية المنزلقة قبل أن
تلتقطك.
\end{method}

\begin{example}[الاختيار الحكيم]\label{ex:g6:measurement-in-science:choose}
\omterm{def:g4:weight-and-mass:weight}{لوزن} رسالة للبريد لا ينفع ميزان الحمّام --- فكتلتها \qty{18}{g} تختفي
بين خطواته؛ أما ميزان المطبخ، وخطوته \omterm{def:g3:measuring-mass:units}{الغرام}، فيجيب إجابة تامّة.
ولقياس زمن عَدو $100$ \omterm{def:g2:measuring-length:units}{متر} تكون خطوات ساعة المطبخ الدقيقية بائسة؛
أما ساعة الإيقاف فتعدّ بأجزاء المئة من الثانية. وليس أحد
الجهازين ``أفضل'': فلكل واحد إقليمه، يحدّده مداه وأصغر خطواته.
\end{example}

\section{أعداد أمينة}

\begin{remark}[بين العلامات]\label{rem:g6:measurement-in-science:between}
المؤشّرات الحقيقية تحبّ أن تحطّ بين العلامات. وللأمانة عندئذ
مستويان: أدناهما أن تسمّي العلامتين الجارتين --- ``بين $7.3$
و$7.4$، أقرب إلى $7.3$''؛ وأفضل منه أن تقدّر رقمًا واحدًا وراء
العلامات --- ``نحو \qty{7.32}{cm}''. أما ما تحرّمه الأمانة
فاختلاق الدقّة: فقراءة \qty{7.3214}{cm} من علامات مليمترية خيال
يلبس معطف مختبر. فعلامات الجهاز تحدّ ما يستطيع أن يعد به.
\end{remark}

\begin{example}[قِس مرتين تثق أكثر]\label{ex:g6:measurement-in-science:repeat}
ثلاثة أصدقاء يقيسون زمن جولة سيارة اللعبة نفسها بساعة الإيقاف
نفسها: \qty{4.6}{s}، و\qty{4.8}{s}، و\qty{4.7}{s}. ولم يخطئ
أحد --- إنما تبدأ الأصابع وتقف متفاوتة شعرةً. والتشتّت الصغير
هو النفَس الطبيعي \omterm{def:g6:measurement-in-science:measuring}{للقياس} الحقيقي. والعادات الحكيمة: أن تكرّر
\omterm{def:g6:measurement-in-science:measuring}{القياس}، وأن تراقب القراءات وهي تتجمّع، وأن ترتاب في أي عدد وحيد
معزول --- ولا سيّما عددك أنت.
\end{example}

\section{تمارين}

أكمل كل \omterm{def:g6:measurement-in-science:measuring}{قياس} \omterm{def:g6:measurement-in-science:measuring}{بوحدة} معقولة: ارتفاع باب $2$
\dots؛ \ \omterm{def:g3:measuring-mass:mass}{كتلة} قطعة نقود $8$ \dots؛ \ يدوم درس مدرسي

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:2}
حوّل: \qty{3.2}{m} إلى \unit{cm}؛ \ \qty{750}{g} إلى \unit{kg}؛ \

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:3}
لماذا كان الزمن متمرّد أسرة الوحدات؟ حوّل \qty{3.5}{min} إلى

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:4}

قيمة الخطوة الواحدة؟ والعصير يقف خطوتين فوق علامة $300$ ---
فكم مقدار العصير؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:5}
أيّ جهاز لكل عمل، ولماذا: \omterm{def:g4:weight-and-mass:weight}{وزن} قبضة خميرة؛ \ \omterm{def:g6:measurement-in-science:measuring}{قياس} زمن عدّ نبضات
القلب؛ \ عرض هذه الصفحة؛ \ طول ساحة المدرسة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:6}
\omterm{def:g6:measurement-in-science:measuring}{تقيس} نادية قلمها: \qty{17.4}{m}. فأيّ شبكة أمان في
\cref{met:g6:measurement-in-science:estimate} كان ينبغي أن

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:7}
يقف عمود \omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} بين علامتَي $18$ و$19$، أقرب إلى $19$.
أعطِ طريقة أمينة واحدة لتبليغه --- وتبليغًا غير أمين واحدًا

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:8}
ثلاث قراءات لساعة إيقاف لجولة واحدة: \qty{6.1}{s}، و\qty{6.3}{s}،

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:9}
وصفة من بلد آخر تطلب ``رُبع رطل من الزبدة''. وميزان مطبخك يتكلّم

تعلّم هذه المهمّة عن الوحدات المشتركة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:10}
انمحى \omterm{def:g2:measuring-length:units}{السنتيمتر} الأول من مسطرة، فصارت حافّتها تبدأ عند العلامة
$1$. ووُضع طابع بريد من الحافة فقرأ طرفه البعيد $3.4$. فكم عرض
الطابع حقًّا؟ وأيّ عادة تهزم المساطر البالية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:11}
يقيس الصفّ الممرّ بشريط \qty{20}{m} مرتين: \qty{18.25}{m}
و\qty{18.35}{m}. فتكتب المعلّمة ``\qty{18.3}{m}''. اشرح
اختيارها، مستعملًا
\cref{rem:g6:measurement-in-science:between} و
\cref{ex:g6:measurement-in-science:repeat}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:measurement-in-science:12}
حكاية قديمة: ملك جعل قدمه هو \omterm{def:g6:measurement-in-science:measuring}{وحدة} الطول في المملكة. اذكر
كارثتين عمليتين تتبعان ذلك (فكّر في نجّارين في مدن بعيدة، وفي
ابن الملك الوارث وهو ينمو)، واشرح كيف حلّ القضيب المعدني
المتقاعد في \cref{rem:g6:measurement-in-science:guard} الأمرين
--- وأيّ مشكلة تركها القضيب نفسه بلا حلّ حتى حسمتها ثوابت
الطبيعة أخيرًا.
\end{exercise}

\section{مسألة: مفتّش الأجهزة}

\begin{problem}[{مسألة نهاية الأسبوع --- يوم مع مفتّش أجهزة
البلدة؛ موازين وأباريق وساعات في قفص الاتهام؛ دفتر
المتدرّب}]\label{pb:g6:measurement-in-science:1}
في كل سنة يختبر مفتّش البلدة أجهزة المتاجر والمدارس. واليوم أنت
المتدرّب، ودفترك في يدك.

\medskip\noindent\textbf{الجزء الأول --- ميزان المخبز.}
تحمل المفتّشة قطعًا معتمَدة: \qty{500}{g}،
و\qty{200}{g}، و\qty{200}{g}، و\qty{100}{g}، و\qty{50}{g}.
\begin{enumerate}
  \item تتحقّق من أن مؤشّر الميزان الفارغ يقف على الصفر تمامًا.
        فلماذا يجب أن يكون هذا أول الاختبارات كلها؟
  \item ثم تحمّل قطعة \qty{500}{g} وقطعة \qty{200}{g}. فماذا
        ينبغي أن يقرأ ميزان أمين؟
  \item ويُظهر المؤشّر \qty{680}{g} بدلًا من ذلك. فهل يجامل هذا
        الميزانُ الخبّازَ أم الزبائن؟ اشرح مَن يخسر مالًا في كل
        بيعة.
  \item وأيّ تركيبات من قطعها تتيح لها اختبار القراءة
        \qty{750}{g}؟ أعطِ واحدة.
  \item وخطوة ميزان المخبز \qty{5}{g}. أفتستطيع المفتّشة أن
        تستعمله للتحقّق من طابع بريد كتلته \qty{2}{g}؟
        ولماذا لا؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- أباريق الكيل في
المدرسة.}
\begin{enumerate}[resume]
  \item إبريق مرقّم كل \qty{200}{mL}، بأربع خطوات بين الأرقام.
        فما قيمة كل خطوة؟
  \item ويقف الماء خطوة واحدة فوق علامة \qty{600}{mL}. ويقول
        سجلّ المعلّم ``نحو \qty{700}{mL}''. قيّم ذلك التبليغ:
        أمين، أم مفرط في الإبهام، أم خيال؟
  \item ويقرأ تلميذ الإبريق من كرسيّ منخفض، وعينه دون خطّ الماء
        بكثير، فيحصل على عدد مختلف عن عدد المفتّشة. فأيّ قاعدة
        من قواعد القراءة خُرقت؟
  \item وتسكب المفتّشة \qty{650}{mL} المعلَنة من الإبريق في
        إبريق معتمَد، فيُظهر \qty{640}{mL}. فبكم ينحرف إبريق
        المدرسة، وهل ذلك أسوأ من خطوة من خطواته أم أفضل؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- الساعات، وحكم
المتدرّب.}
\begin{enumerate}[resume]
  \item تُقارن ساعة المحطة بإشارة الوقت الإذاعية: فتُظهر ظهرَ
        الإشارة عند $12$ س و$2$ د. أهي مسرعة أم متأخّرة، وبكم؟
  \item وإن لم يصحّحها أحد، فبكم ستسبق بعد أسبوع؟ (وهي تكسب
        المقدار نفسه كل يوم.)
  \item وثلاث ساعات في متاجر تُظهر $12$ س و$2$ د، و$11$ س
        و$58$ د، و$12$ س تمامًا عند ظهر الإشارة. فتقول
        المفتّشة: ``المتوسّط هنا سخف --- بل صحّح كل ساعة.''
        فأيّ إصلاح تحتاج إليه كل واحدة؟
  \item وفي المساء، اكتب في دفترك قواعد المفتّشة الذهبية الثلاث
        --- واحدة عن الصفر، وواحدة عن القراءة، وواحدة عن
        الوحدات --- كل واحدة في جملة واحدة من عندك.
\end{enumerate}
\end{problem}
