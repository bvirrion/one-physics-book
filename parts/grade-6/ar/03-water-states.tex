\chapter{الماء في حالاته كلها}\label{ch:g6:water-states}

أنت تطارد الماء في حالاته منذ سنتك المدرسية الثانية: ثلج، وماء،
وبخار؛ ورجال ثلج ينصهرون، ونوافذ يعلوها الغبش. وهذه السنة تصير
المطاردة محاسبة. فحين \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} الثلج، أتُفقد مادة؟ وحين \omterm{def:g2:ice-water-steam:meltfreeze}{يتجمّد}
الماء، لماذا تنفجر الأنابيب؟ وبالموازين والمخابير في أيدينا،
ندقّق حسابات حالات الماء الثلاث.

\section{الحالات الثلاث بأسمائها على أصولها}

\begin{definition}[تغيّرات الحالة الستّة]\label{def:g6:water-states:changes}
بين حالاته الثلاث للماء ستّة أبواب، لكل واحد اسمه العلمي:
\begin{enumerate}
  \item من الصلبة $\to$ إلى السائلة:
        \emph{الانصهار}\index{انصهار}؛ ومن السائلة $\to$ إلى
        الصلبة: \emph{التصلّب}\index{تصلّب} (\omterm{def:g2:ice-water-steam:meltfreeze}{التجمّد})؛
  \item ومن السائلة $\to$ إلى الغازية:
        \emph{التبخّر}\index{تبخّر} --- هادرًا سريعًا عند
        الغليان، وهادئًا باردًا عند التبخّر السطحي؛ ومن الغازية
        $\to$ إلى السائلة: \emph{التكاثف}\index{تكاثف}؛
  \item ومن الصلبة $\to$ إلى الغازية مباشرة:
        \emph{التسامي}\index{تسامي}، ومن الغازية $\to$ إلى
        الصلبة مباشرة --- وهما بابان أندر، لكنهما حقيقيان: فريش
        الصقيع على نافذة متجمّدة ينمو من البخار الخفيّ مباشرة،
        متخطّيًا الحالة السائلة كلها.
\end{enumerate}
\end{definition}

صقيع الصباح \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} عن العشب؛ والغسيل \omterm{def:g2:ice-water-steam:evapcond}{يتبخّر} على الحبل؛ \omterm{def:g5:mirrors-reflection:reflection}{ومرآة}

\section{التدقيق: أتُفقد مادة؟}

\begin{method}[وزن تغيّر حالة]\label{met:g6:water-states:weigh}
اطرح السؤال على الميزان:
\begin{enumerate}
  \item اسكب بعض الماء في قنّينة بلاستيكية صغيرة، وأحكِم الغطاء،
        وجفّف ظاهرها؛
  \item وزِنها: ولنقل \qty{368}{g} --- واكتب ذلك؛
  \item وضعها في المجمّدة طوال الليل: وهذا \omterm{def:g6:water-states:changes}{تصلّب}؛
  \item وزِن القنّينة المتجمّدة في الحال (وجفّف أي صقيع
        عليها).
\end{enumerate}
فيجيب الميزان: \qty{368}{g}، حتى \omterm{def:g3:measuring-mass:units}{الغرام}. وأذِبها على حافة
النافذة وزِنها من جديد: \qty{368}{g}. تغيّرت الحالة مرتين؛ ولم
ترمش المادة.
\end{method}

\begin{proposition}[الكتلة تنجو من تغيّرات الحالة]\label{prop:g6:water-states:mass}
تغيّر الحالة يغيّر منظر المادة لا مقدارها: \omterm{def:g3:measuring-mass:mass}{فالكتلة} قبل تساوي
\omterm{def:g3:measuring-mass:mass}{الكتلة} بعد، بالضبط. \omterm{def:g6:water-states:changes}{الانصهار} \omterm{def:g6:water-states:changes}{والتصلّب} والغليان في قِدر مغلق
\omterm{def:g6:water-states:changes}{والتكاثف} --- في كل باب يقرأ الميزان القراءة نفسها. (لكن اترك
الغطاء مفتوحًا فينجرف البخار بغراماته: فالمادة ما زالت موجودة،
لكن تدقيقك لم يعد يحويها.)
\end{proposition}

\begin{example}[أين ذهبت غرامات البركة]\label{ex:g6:water-states:puddle}
بركة كتلتها \qty{300}{g} تجفّ. أاختفت؟ أما كان لها \omterm{def:g4:weight-and-mass:weight}{وزن}؟ لا:
فمقدار \qty{300}{g} من الماء يركب الآن \omterm{def:g2:air-around-us:air}{الهواء} بخارًا خفيًّا، وكل \omterm{def:g3:measuring-mass:units}{غرام}
سليم --- وبعضها سيسقط مطر الأسبوع القادم. \omterm{def:g6:water-states:changes}{فالتبخّر} يخدع العين
لا المحاسبة أبدًا. وخطّ الطباشير حول بركة طفولتك كان تدقيقًا؛
وقد صارت له أعداد الآن.
\end{example}

\section{التدقيق: أيُحفظ الحيّز؟}

\begin{proposition}[الماء ينتفخ حين يتجمّد]\label{prop:g6:water-states:volume}
\omterm{def:g3:measuring-mass:mass}{الكتلة} تنجو من الأبواب --- أما \omterm{def:g6:mass-and-volume:volume}{الحجم} فلا. والماء مادة عجيبة:
فهو حين يصير صلبًا \emph{يتمدّد}. \omterm{def:g6:mass-and-volume:volume}{فلتر} الماء يصنع نحو $1.1$ \omterm{def:g6:mass-and-volume:volume}{لتر}
من الثلج: \omterm{def:g3:measuring-mass:mass}{الكتلة} نفسها، وحيّز أكبر بنحو العشر. وأكثر الموادّ
تنكمش حين تتصلّب؛ أما الماء فينتفخ --- وتلك العجيبة تعيد رسم
المناظر الطبيعية.
\end{proposition}

\begin{example}[الأنبوب المنفجر والجبل المتشقّق]\label{ex:g6:water-states:burst}
قنّينة زجاجية ممتلئة بالماء نُسيت في المجمّدة تتشقّق: فالثلج
المنتفخ يحتاج إلى عُشره الزائد من الحيّز ويأخذه، زجاجًا كان أم
غير زجاج. والشتاء يفعل ذلك بالسباكة --- فالماء المتجمّد في
أنبوب يفجّره كما تفجّره مطرقة --- ويفعله بالجبال: فيرشح المطر
في شقوق الصخر، \omterm{def:g2:ice-water-steam:meltfreeze}{ويتجمّد} ليلًا، وينتفخ، فيوسّع الشقّ. وسنةً بعد
سنة، وتجمّدًا بعد \omterm{def:g2:ice-water-steam:meltfreeze}{تجمّد}، يشقّ الانتفاخ الصغير العجيب في
\cref{prop:g6:water-states:volume} الصخورَ ويفتّت القمم.
\end{example}

\begin{remark}[لماذا يطفو الثلج]\label{rem:g6:water-states:floats}
العجيبة نفسها تجيب عن سؤال سألته قبل سنوات وقت الحمّام. فالثلج
يحمل مادة \omterm{def:g6:mass-and-volume:volume}{لتر} في $1.1$ \omterm{def:g6:mass-and-volume:volume}{لتر} من الحيّز: فهو للحيّز المتساوي
\emph{أخفّ} من الماء --- \omterm{def:g1:floating-and-sinking:words}{فيطفو} الثلج على سائله. فالجبال الجليدية
تركب البحر، والبِرَك تتجمّد من الأعلى إلى الأسفل (فتنجو الأسماك
تحتها)، ومكعّبات ليموناضتك تطفو بدل أن تغرق. وقليل من الموادّ
\omterm{def:g1:floating-and-sinking:words}{يطفو} على نفسه؛ والحياة على البِرَك المتجمّدة سعيدة بأن الماء
يفعل.
\end{remark}

\section{دورة الماء، مدقَّقةً}

\begin{example}[الرحلة الدائرية الكبرى، بالمصطلحات]\label{ex:g6:water-states:cycle}
حكاية الطفولة عن البحر والغيمة والمطر تُقرأ الآن دفترَ حسابات.
فدفء الشمس يقود \emph{\omterm{def:g6:water-states:changes}{التبخّر}} من البحار --- فترتفع أطنان فوق
أطنان من غير أن تُرى. وعاليًا في البرد \emph{\omterm{def:g2:ice-water-steam:evapcond}{يتكاثف}} البخار
قطيرات غيم؛ وإذا اشتدّ البرد \emph{تصلّبت} القطيرات ثلجًا
متساقطًا. ويتكوّم الثلج أنهارًا جليدية؛ و\emph{تنصهر} الأنهار
الجليدية أنهارًا جارية؛ وتردّ الأنهار كل \omterm{def:g3:measuring-mass:units}{غرام} إلى البحر. وعلى
الدارة كلها تصمد \cref{prop:g6:water-states:mass}: \omterm{def:g4:solar-system:planet}{فالكوكب} لا
يكسب قطرة ولا يفقدها --- والماء الذي في كأسك أبحر هذه الدورة من
قبل الديناصورات.
\end{example}

\section{مسألة: كيلوغرام حول العالم}

\begin{problem}[{مسألة نهاية الأسبوع --- كيلوغرام من ماء البحر
في جولته الكبرى؛ ستّة أبواب، وثلاثة جبال، ودفتر حسابات لا
ينكسر}]\label{pb:g6:water-states:1}
نضع لصاقة متخيَّلة على \qty{1}{kg} من ماء البحر ونتابعه سنةً.

\medskip\noindent\textbf{الجزء الأول --- صعودًا.}
\begin{enumerate}
  \item حزيران: تُدفئ الشمس سطح البحر فيرتفع كيلوغرامنا في
        \omterm{def:g2:air-around-us:air}{الهواء} من غير أن يُرى. سمِّ الباب الذي سلكه.
  \item وما \omterm{def:g3:measuring-mass:mass}{كتلة} كيلوغرامنا من الماء (وقد صار خفيًّا)؟ وأيّ
        قضية تضمن ذلك؟
  \item وعند \qty{3000}{m} يكون \omterm{def:g2:air-around-us:air}{الهواء} باردًا: فيصير الماء
        غيمة من قطيرات. سمِّ هذا الباب.
  \item وإذا اشتدّ البرد صارت القطيرات ندفات ثلج. سمِّ ذلك
        الباب --- وهو \omterm{def:g6:water-states:changes}{التصلّب} --- واذكر \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} التي تبدأ
        عندها قطيرات الغيم من ماء نقيّ \omterm{def:g2:ice-water-steam:meltfreeze}{بالتجمّد}.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- دفتر النهر الجليدي.}
يحطّ كيلوغرامنا على نهر جليدي ويُضغط جليدًا نهريًّا.
\begin{enumerate}[resume]
  \item شغل كيلوغرامنا ماءً \qty{1}{L}. فأيّ \omterm{def:g6:mass-and-volume:volume}{حجم} يشغله جليدًا
        تقريبًا؟
  \item \omterm{def:g2:ice-water-steam:meltfreeze}{وينصهر} أنف النهر الجليدي في الصيف. فيصير كيلوغرامنا ماء
        \omterm{def:g6:water-states:changes}{انصهار}: فما كتلته الآن، وما حجمه (وقد عاد ماءً
        سائلًا)؟
  \item وفي النزول من الجبل \omterm{def:g2:ice-water-steam:evapcond}{يتبخّر} \qty{250}{g} من كيلوغرامنا
        من بركة شلّال مشمسة. فكم يبقى من الماء المعلَّم سائلًا
        في الجدول؟
  \item وهل غادر الرُّبع المتبخّر التدقيق؟ قل أين تكون
        \qty{250}{g} تلك، وأيّ باب سيسلكه على الأرجح
        بعد ذلك.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- العودة، والحسابات
متوازنة.}
\begin{enumerate}[resume]
  \item تبلغ \qty{750}{g} من الجدول البحرَ في تشرين الأول. أما
        \qty{250}{g} المتبخّرة فسقطت مطرًا على الساحل في آب
        وجرت إلى البحر في أيلول. فاجمع ما استردّه البحر من
        كيلوغرامنا.
  \item وطوال الجولة لم يكن كيلوغرامنا عديم \omterm{def:g4:weight-and-mass:weight}{الوزن} قطّ --- لكنه
        كان \emph{خفيًّا} مرتين. ففي أيّ مرحلتين، وفي أيّ
        حالة؟
  \item ويعترض زميل: ``الثلج المتساقط منفوش خفيف --- فلا بدّ أن
        \omterm{def:g3:measuring-mass:units}{الكيلوغرام} كان أخفّ ثلجًا.'' فُكّ الالتباس بالزوج
        الصحيح من كلمات هذه السنة.
  \item واكتب الجولة سلسلةً من الأبواب، بالترتيب، مستعملًا
        الأسماء الستّة على أصولها في
        \cref{def:g6:water-states:changes} لا غير.
\end{enumerate}
\end{problem}
