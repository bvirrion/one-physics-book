\chapter{घनत्व: चीज़ें तैरती क्यों हैं}\label{ch:g6:density-floating}

तुम्हारे बिलकुल पहले साल में एक नन्हा सिक्का डूब गया था जबकि एक विशाल
लट्ठा तैर गया था, और हमने वादा किया था कि उसका ``क्यों'' इस पाठ्यक्रम
का ख़ज़ाना है। तुमने वह कमा लिया है। कुंजी दो अध्यायों में गढ़ी गई ---
पदार्थ के लिए \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, और स्थान के लिए \omterm{def:g6:mass-and-volume:volume}{आयतन} --- और आज दोनों मिलकर
ताले में घूमते हैं।

\section{एक घन सेंटीमीटर का द्रव्यमान}

\begin{example}[आख़िरकार निष्पक्ष तुलना]\label{ex:g6:density-floating:fair}
पूरे लट्ठे की तुलना पूरे सिक्के से करना कभी निष्पक्ष था ही नहीं: नाप
अलग, आकार अलग। निष्पक्ष मुक़ाबला \emph{बराबर \omterm{def:g6:mass-and-volume:volume}{आयतन}} लेता है: हर
प्रतियोगी का एक-एक \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर}। एक \unit{cm^3} पानी का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}
\qty{1}{g} होता है। एक \unit{cm^3} बलूत का: लगभग \qty{0.6}{g}।
पत्थर का: लगभग \qty{2.5}{g}। लोहे का: \qty{7.9}{g}। बराबर स्थान, और
पदार्थ में बेतहाशा फ़र्क़ --- तैरने वाला खेल इसी संख्या का इंतज़ार कर
रहा था।
\end{example}

\begin{definition}[घनत्व]\label{def:g6:density-floating:density}
किसी पदार्थ का \emph{घनत्व}\index{घनत्व} उसके एक \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} का
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} है, \omterm{def:g3:measuring-mass:units}{ग्राम} में। पानी का घनत्व प्रति \unit{cm^3}
\qty{1}{g} है; बलूत का लगभग $0.6$; लोहे का $7.9$। घनत्व निकालने के
लिए किसी नए उपकरण की ज़रूरत नहीं: किसी नमूने का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और \omterm{def:g6:mass-and-volume:volume}{आयतन} मापो,
फिर भाग दो --- ग्रामों को घन सेंटीमीटरों में बराबर बाँट दो, और उत्तर
हर \omterm{def:g2:measuring-length:units}{सेंटीमीटर} का हिस्सा है।
\end{definition}

\begin{proposition}[घनत्व एक दस्तख़त है]\label{prop:g6:density-floating:signature}
किसी शुद्ध पदार्थ का \omterm{def:g6:density-floating:density}{घनत्व} उसके हर टुकड़े के लिए एक ही रहता है, चाहे
वह बड़ा हो या छोटा: बलूत की किरच और पूरा शहतीर, दोनों अपने हर \omterm{def:g6:mass-and-volume:volume}{घन
सेंटीमीटर} में लगभग \qty{0.6}{g} लिए चलते हैं। इसलिए \omterm{def:g6:density-floating:density}{घनत्व} पदार्थों की
पहचान करा देता है, ठीक जैसे \omterm{def:g4:changes-of-state:melting-point}{गलनांक} करते हैं: किसी रहस्यमय वस्तु का
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और \omterm{def:g6:mass-and-volume:volume}{आयतन} मापो, भाग दो, और तालिका से मिलाओ --- पदार्थ अकसर
वहीं का वहीं इक़बाल कर लेता है।
\end{proposition}

\begin{method}[घनत्व मापना]\label{met:g6:density-floating:measure}
किसी कंकड़ के लिए --- या किसी मुकुट के लिए:
\begin{enumerate}
  \item \omterm{def:g3:levers-and-scales:scale}{तुला} से उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} मापो: मान लो \qty{75}{g};
  \item पानी की उठान से उसका \omterm{def:g6:mass-and-volume:volume}{आयतन} मापो: मान लो \qty{30}{cm^3};
  \item ग्रामों को घन सेंटीमीटरों से भाग दो: $75 \div 30 = 2.5$ ---
        यानी हर \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} \qty{2.5}{g} लिए चलता है;
  \item तालिका देखो: \omterm{def:g6:density-floating:density}{घनत्व} $2.5$ --- हमारा कंकड़ साधारण पत्थर है।
\end{enumerate}
\end{method}

\section{तैरने का नियम}

\begin{proposition}[चीज़ें तैरती क्यों हैं]\label{prop:g6:density-floating:rule}
कोई वस्तु पानी में तब \omterm{def:g1:floating-and-sinking:words}{तैरती} है जब उसका \omterm{def:g6:density-floating:density}{घनत्व} पानी के \omterm{def:g6:density-floating:density}{घनत्व} से
\emph{कम} हो --- यानी प्रति \omterm{def:g6:mass-and-volume:volume}{घन सेंटीमीटर} \qty{1}{g} से कम --- और तब
\omterm{def:g1:floating-and-sinking:words}{डूबती} है जब उसका \omterm{def:g6:density-floating:density}{घनत्व} ज़्यादा हो। कह लो कि पानी घुसपैठिये की तुलना
अपने आप से करता है, स्थान-दर-स्थान: जो मुझसे प्रति-स्थान हल्का है वह
ऊपर सवार रहता है; जो प्रति-स्थान भारी है वह नीचे चला जाता है। यही नियम
हर \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} पर चलता है, और हर \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} अपने \omterm{def:g6:density-floating:density}{घनत्व} के साथ जज बनकर बैठा है।
\end{proposition}

\begin{example}[छह साल की पहेलियाँ, निपट गईं]\label{ex:g6:density-floating:settled}
लट्ठा ($0.6$): तैरता है, चाहे कितना ही विशाल हो --- उसका हर \omterm{def:g6:mass-and-volume:volume}{घन
सेंटीमीटर} पानी से कम की बोली लगाता है। सिक्का (काँसा, लगभग $9$): डूब
जाता है, चाहे कितना ही छोटा हो। प्लास्टिक की बत्तख, कॉर्क, पेंसिल:
सब $1$ से नीचे, यानी सब तैरने वाले। बर्फ़, $0.9$ पर --- यानी पानी का
अपना फूला हुआ \omterm{def:g3:solids-liquids-gases:solid}{ठोस} --- बाल-बाल अपने ही \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} पर तैर जाती है, और इसीलिए
तालाब ढक्कन पहनते हैं और हिमशैल सफ़र करते हैं। और प्लास्टिसीन की नाव?
कटोरी के आकार में \emph{नाव-और-हवा} वाली पुड़िया अपने हर स्थान में
ख़ूब सारी \omterm{def:g2:air-around-us:air}{हवा} रखती है: पुड़िया का \omterm{def:g6:density-floating:density}{घनत्व} $1$ से नीचे उतर जाता है, और
बंदरगाहों पर यही तरकीब इस्पात के दस हज़ार टन वाले जहाज़ भी उतार देती
है।
\end{example}

\begin{example}[द्रवों की मीनार]\label{ex:g6:density-floating:tower}
किसी ऊँचे गिलास की दीवार के सहारे धीरे-धीरे पहले शहद, फिर स्याही से
रँगा पानी, और फिर तेल उँडेलो: तीन परतें बन जाती हैं, सबसे नीचे शहद
($1.4$) और सबसे ऊपर तेल ($0.9$), और वे अपनी क़तार तोड़ने से इनकार कर
देती हैं। अब मेहमान गिराओ: एक अंगूर ($1.05$) तेल में से डूबता है, पानी
में से भी डूबता है, और शहद पर आकर टिक जाता है --- दो मंज़िलों से घना
और तीसरी से हल्का। यह मीनार तैरने के नियम का रंगमंच पर मंचन है।
\end{example}

\begin{remark}[हिमशैल पढ़ना]\label{rem:g6:density-floating:iceberg}
पानी के $1.0$ के सामने बर्फ़ का $0.9$ सिर्फ़ हिमशैल को तैराता भर नहीं:
वह उसकी जलरेखा भी तय कर देता है। हिमशैल का दसवाँ हिस्सा छोड़कर बाक़ी
सब सतह के \emph{नीचे} सवार रहता है --- यही वह मशहूर छिपा हुआ \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}
है जिसने घमंडी जहाज़ डुबाए। नीचे रहने वाला यह भाग घनत्वों के अनुपात से
ठीक-ठीक मेल क्यों खाता है, यह उसी प्राचीन नहाने-वाले वैज्ञानिक का एक
सुंदर नियम है, जिसे उच्च विद्यालय वाले खंड में ईमानदारी से सिद्ध किया
जाएगा; घनत्वों की तालिका उत्तर पहले ही फुसफुसा देती है।
\end{remark}

\section{और आख़िर में, वह मुकुट}

\begin{example}[आर्किमिडीज़ मुक़दमा बंद करता है]\label{ex:g6:density-floating:crown}
अब दो अध्याय पहले शुरू हुई कहानी पूरी करो। मुकुट का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}: आसान,
\omterm{def:g3:levers-and-scales:scale}{तुला} बता देती है। उसका \omterm{def:g6:mass-and-volume:volume}{आयतन}: गँठीली कलाकृति के लिए नामुमकिन --- जब तक
नहाने के टब वाले सबक़ ने उठान वाली विधि न दे दी। \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} में \omterm{def:g6:mass-and-volume:volume}{आयतन}
से भाग: मुकुट का \emph{\omterm{def:g6:density-floating:density}{घनत्व}} --- और \omterm{def:g6:density-floating:density}{घनत्व} एक दस्तख़त है। शुद्ध सोना
$19.3$ के दस्तख़त करता है; चाँदी $10.5$ के; और कोई गुप्त मिलावट इनके
बीच का कुछ। किंवदंती कहती है कि सुनार के मुकुट ने सोने से काफ़ी कम
\omterm{def:g6:density-floating:density}{घनत्व} का इक़बाल कर लिया, और मुकुट पर एक ख़रोंच आए बिना धोखाधड़ी पकड़ी
गई। सप्ताहांत की समस्या तुम्हें उसके अंक थमा देती है।
\end{example}

\section{अभ्यास}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:density-floating:7}
बर्फ़ पानी पर \omterm{def:g1:floating-and-sinking:words}{तैरती} है --- किसी \omterm{def:g3:solids-liquids-gases:solid}{ठोस} के लिए यह अनोखा क्यों है, और जमते
पानी वाली कौन-सी पुरानी प्रतिज्ञप्ति $0.9$ की व्याख्या करती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:density-floating:8}
तुम्हारे पहले साल का वह विशाल लट्ठा और वह नन्हा सिक्का: अध्याय की
संख्याओं के साथ आख़िरी, पूरा उत्तर दो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:density-floating:9}
किसी \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} के एक \omterm{def:g6:mass-and-volume:volume}{लीटर} का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \qty{800}{g} है। \omterm{def:g3:measuring-mass:units}{ग्राम} प्रति \omterm{def:g6:mass-and-volume:volume}{घन
सेंटीमीटर} में उसका \omterm{def:g6:density-floating:density}{घनत्व} क्या है? क्या बर्फ़ का टुकड़ा ($0.9$) उसमें
तैरेगा या डूबेगा?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:density-floating:10}
इस्पात $7.9$ के दस्तख़त करता है, फिर भी इस्पात के जहाज़ तैरते हैं।
\cref{ex:g6:density-floating:settled} वाले पुड़िया के विचार से इस
विरोधाभास को सुलझाओ --- इस्पात के अलावा जहाज़-पुड़िया में और क्या शामिल
है, और उसका कुल \omterm{def:g6:density-floating:density}{घनत्व} कैसा होना चाहिए? छेद हो चुका जहाज़ इस नियम पर
कब खरा उतरना बंद कर देता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:density-floating:11}
समुद्र का पानी मीठे पानी से ज़रा घना होता है --- लगभग $1.03$। समझाओ कि
तैराक समुद्र में साफ़ तौर पर बेहतर क्यों तैरते हैं, और यह भी बताओ कि
खारे समुद्र से मीठी नदी में आते ही किसी जहाज़ की जलरेखा का क्या होता
है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:density-floating:12}
धातु का एक खोखला गोला पानी में ठीक आधा डूबा हुआ तैरता है।
\emph{पूरे गोला-पुड़िया} (धातु का खोल और भीतर की \omterm{def:g2:air-around-us:air}{हवा}) के \omterm{def:g6:density-floating:density}{घनत्व} का तर्क
से अंदाज़ा लगाओ। फिर बताओ कि भीतर धीरे-धीरे पानी रिसने पर वह किस ओर
जाएगा, और किस पल वह \omterm{def:g1:floating-and-sinking:words}{डूबना} शुरू करेगा --- नियम और पुड़िया, दोनों मिलकर
काम करते हुए।
\end{exercise}

\section{समस्या: राजा का मुकुट}

\begin{problem}[{सप्ताहांत समस्या --- राजा के मुकुट का मुक़दमा,
तुम्हारे उपकरणों के साथ फिर से खोला गया; सुनार की पेशी, और भाग से
मिला फ़ैसला}]\label{pb:g6:density-floating:1}
प्राचीन काल की सबसे मशहूर धोखाधड़ी का मुक़दमा आधुनिक संख्याओं के साथ
फिर से खोलो। दस्तख़त: सोना $19.3$, चाँदी $10.5$ (\omterm{def:g3:measuring-mass:units}{ग्राम} प्रति \omterm{def:g6:mass-and-volume:volume}{घन
सेंटीमीटर})।

\medskip\noindent\textbf{भाग I --- सबूत।}
राजा ने सुनार को \qty{1930}{g} शुद्ध सोना दिया था। तैयार मुकुट, \omterm{def:g3:levers-and-scales:scale}{तुला}
पर: ठीक \qty{1930}{g}।
\begin{enumerate}
  \item \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बिलकुल मेल खाता है। फिर भी अकेला यह तथ्य धोखाधड़ी के
        बारे में कुछ साबित क्यों नहीं करता? (सुनार क्या कर सकता था
        और फिर भी यही \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} पा सकता था?)
  \item \emph{शुद्ध सोने} के \qty{1930}{g} को कितना \omterm{def:g6:mass-and-volume:volume}{आयतन} घेरना
        चाहिए? (ग्रामों को बाँटो: \qty{19.3}{g} वाले कितने \omterm{def:g6:mass-and-volume:volume}{घन
        सेंटीमीटर}?)
  \item आर्किमिडीज़ मुकुट को लबालब भरे बरतन में उतारता है और छलका हुआ
        पानी इकट्ठा करता है: \qty{130}{cm^3}। मुकुट का सच्चा \omterm{def:g6:mass-and-volume:volume}{आयतन}
        कितना है?
  \item दोनों आयतनों की तुलना करो। यह तुलना अभी से क्या चिल्ला रही
        है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दस्तख़त।}
\begin{enumerate}[resume]
  \item मुकुट के \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और मापे गए \omterm{def:g6:mass-and-volume:volume}{आयतन} से उसका \omterm{def:g6:density-floating:density}{घनत्व} निकालो (एक
        दशमलव काफ़ी है)।
  \item तीनों दस्तख़त अगल-बग़ल रखो: सोना, मुकुट, चाँदी। मुकुट कहाँ
        बैठता है?
  \item राजा को एक वाक्य में समझाओ कि मुकुट शुद्ध सोने का हो ही नहीं
        सकता।
  \item सुनार विरोध करता है: ``शायद \omterm{def:g3:levers-and-scales:scale}{तुला} ने ग़लती की हो!'' उसे दोनों
        ओर \qty{30}{g} की उदार छूट दे दो। क्या \qty{1900}{g} और
        \qty{1960}{g} के बीच का कोई भी \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} \qty{130}{cm^3}
        \omterm{def:g6:mass-and-volume:volume}{आयतन} वाले शुद्ध सोने के मुकुट को बचा लेता है? (\omterm{def:g6:density-floating:density}{घनत्व} का
        परास जाँचो।)
\end{enumerate}

\medskip\noindent\textbf{भाग III --- चोरी कितनी हुई?}
मान लो सुनार ने कुछ सोना रख लिया और उसकी जगह ग्राम-दर-ग्राम चाँदी
मिला दी।
\begin{enumerate}[resume]
  \item सोने की जगह ग्राम-दर-ग्राम चाँदी रखने से \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} सही क्यों
        बना रहता है पर \omterm{def:g6:mass-and-volume:volume}{आयतन} \emph{फूल} क्यों जाता है?
  \item एक अंदाज़ा जाँचो: \qty{965}{g} सोना $+$ \qty{965}{g} चाँदी
        वाला मुकुट। सोने वाले हिस्से का \omterm{def:g6:mass-and-volume:volume}{आयतन}? चाँदी वाले हिस्से का?
        (एक दशमलव तक।)
  \item अंदाज़े वाले मुकुट का कुल \omterm{def:g6:mass-and-volume:volume}{आयतन} जोड़ो, और उसे मापे गए
        \qty{130}{cm^3} के सामने परखो: चाँदी ज़्यादा मान ली गई या
        कम?
  \item अदालत को फ़ैसले का एक वाक्य चाहिए: बताओ कि बिना किसी संदेह के
        क्या साबित हुआ (शुद्ध सोना या नहीं), तरीक़ा क्या था, और मुकुट
        को कभी नुक़सान पहुँचाने की ज़रूरत क्यों नहीं पड़ी ---
        आर्किमिडीज़ की असली जीत।
\end{enumerate}
\end{problem}
