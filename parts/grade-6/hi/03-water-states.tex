\chapter{पानी अपनी सभी अवस्थाओं में}\label{ch:g6:water-states}

स्कूल के दूसरे साल से ही तुम पानी का उसकी अवस्थाओं में पीछा करते आ रहे
हो: बर्फ़, पानी, भाप; पिघलते बर्फ़ के पुतले और धुँधली खिड़कियाँ। इस
साल यह पीछा हिसाब-किताब बन जाता है। बर्फ़ पिघलने पर क्या कोई पदार्थ
खोता है? पानी जमने पर पाइप क्यों फट जाते हैं? हाथ में \omterm{def:g3:levers-and-scales:scale}{तुला} और बेलन
लेकर हम पानी की तीनों अवस्थाओं की जाँच-पड़ताल करेंगे।

\section{तीन अवस्थाएँ, ठीक-ठीक नामों के साथ}

\begin{definition}[अवस्था बदलने के छह दरवाज़े]\label{def:g6:water-states:changes}
अपनी तीन अवस्थाओं के बीच पानी के पास छह दरवाज़े हैं, और हर एक का अपना
वैज्ञानिक नाम है:
\begin{enumerate}
  \item \omterm{def:g3:solids-liquids-gases:solid}{ठोस} $\to$ \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}: \emph{गलन}\index{गलन};
        \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} $\to$ \omterm{def:g3:solids-liquids-gases:solid}{ठोस}:
        \emph{ठोसीकरण}\index{ठोसीकरण} (यानी \omterm{def:g2:ice-water-steam:meltfreeze}{जमना});
  \item \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} $\to$ \omterm{def:g3:solids-liquids-gases:gas}{गैस}: \emph{वाष्पन}\index{वाष्पन}
        --- उबाल पर गरजता हुआ तेज़, और सतह से चुपचाप तथा ठंडा हो तो
        वाष्पीकरण; \omterm{def:g3:solids-liquids-gases:gas}{गैस} $\to$ \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}:
        \emph{संघनन}\index{संघनन};
  \item सीधे \omterm{def:g3:solids-liquids-gases:solid}{ठोस} $\to$ \omterm{def:g3:solids-liquids-gases:gas}{गैस}:
        \emph{ऊर्ध्वपातन}\index{ऊर्ध्वपातन}, और सीधे \omterm{def:g3:solids-liquids-gases:gas}{गैस} $\to$ \omterm{def:g3:solids-liquids-gases:solid}{ठोस}
        --- ये दरवाज़े कम ही खुलते हैं, पर सच्चे हैं: जमती हुई
        खिड़की पर पाले के पंख अनदेखी वाष्प से सीधे उगते हैं, और
        \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} अवस्था को पूरी तरह छोड़ देते हैं।
\end{enumerate}
\end{definition}

\begin{example}[चारों ओर दरवाज़े]\label{ex:g6:water-states:doors}
सुबह घास से पिघलता पाला; तार पर सूखते कपड़ों का \omterm{def:g6:water-states:changes}{वाष्पन}; ठंडे \omterm{def:g5:mirrors-reflection:reflection}{दर्पण} पर
\omterm{def:g2:ice-water-steam:evapcond}{संघनित} होती तुम्हारे स्नान की वाष्प; कल के लिए बर्फ़ जमाता फ़्रीज़र; और
रात भर में शीशे पर उगे पाले के पंख। एक ही पदार्थ, छह दरवाज़े, और सब के
सब रोज़ के एक ही दिन के भीतर --- इनके नाम जान लो, तो ये तुम्हारे हुए।
\end{example}

\section{जाँच-पड़ताल: क्या पदार्थ खोता है?}

\begin{method}[अवस्था के बदलाव को तौलना]\label{met:g6:water-states:weigh}
सवाल सीधे \omterm{def:g3:levers-and-scales:scale}{तुला} के सामने रखो:
\begin{enumerate}
  \item प्लास्टिक की छोटी बोतल में थोड़ा पानी डालो, ढक्कन कसकर बंद
        करो, और बाहर से पोंछ लो;
  \item उसे तौलो: मान लो \qty{368}{g} --- लिख लो;
  \item उसे रात भर फ़्रीज़र में रखो: यानी \omterm{def:g6:water-states:changes}{ठोसीकरण};
  \item जमी हुई बोतल को तुरंत तौलो (बाहर लगा पाला पोंछकर)।
\end{enumerate}
\omterm{def:g3:levers-and-scales:scale}{तुला} का उत्तर: \qty{368}{g}, \omterm{def:g3:measuring-mass:units}{ग्राम} तक। उसे खिड़की की चौखट पर वापस
पिघलाओ और फिर तौलो: \qty{368}{g}। अवस्था दो बार बदली; पदार्थ ने पलक
तक नहीं झपकाई।
\end{method}

\begin{proposition}[द्रव्यमान अवस्था के बदलावों को झेल जाता है]\label{prop:g6:water-states:mass}
अवस्था का बदलाव किसी पदार्थ का रूप बदलता है, उसकी मात्रा नहीं: पहले
का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बाद के \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} के ठीक बराबर होता है। \omterm{def:g6:water-states:changes}{गलन}, \omterm{def:g2:ice-water-steam:meltfreeze}{जमना}, बंद
पतीले में उबलना, \omterm{def:g6:water-states:changes}{संघनन} --- हर दरवाज़े से गुज़रने पर \omterm{def:g3:levers-and-scales:scale}{तुला} वही पाठ देती
है। (हाँ, ढक्कन खुला छोड़ दो और वाष्प अपने ग्रामों समेत उड़ जाएगी:
पदार्थ तब भी मौजूद रहता है, पर तुम्हारे हिसाब-किताब में नहीं रहता।)
\end{proposition}

\begin{example}[गड्ढे के ग्राम गए कहाँ]\label{ex:g6:water-states:puddle}
\qty{300}{g} वाला एक गड्ढा सूख जाता है। ग़ायब हो गया? भार-रहित हो गया?
नहीं: \qty{300}{g} पानी अब अनदेखी वाष्प बनकर \omterm{def:g2:air-around-us:air}{हवा} पर सवार है, एक-एक
\omterm{def:g3:measuring-mass:units}{ग्राम} सही-सलामत --- और उसमें से कुछ अगले हफ़्ते बारिश बनकर गिरेगा।
\omterm{def:g6:water-states:changes}{वाष्पन} आँख को धोखा देता है, हिसाब-किताब को कभी नहीं। बचपन के गड्ढे वाले
प्रयोग की खड़िया-रेखा एक जाँच-पड़ताल ही थी; अब उसके पास संख्याएँ भी
हैं।
\end{example}

\section{जाँच-पड़ताल: क्या जगह वही रहती है?}

\begin{proposition}[जमते समय पानी फैलता है]\label{prop:g6:water-states:volume}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} दरवाज़ों को झेल जाता है --- \omterm{def:g6:mass-and-volume:volume}{आयतन} नहीं। पानी एक अनोखा पदार्थ
है: \omterm{def:g3:solids-liquids-gases:solid}{ठोस} बनते समय वह \emph{फैल} जाता है। एक \omterm{def:g6:mass-and-volume:volume}{लीटर} पानी से लगभग $1.1$
\omterm{def:g6:mass-and-volume:volume}{लीटर} बर्फ़ बनती है: \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} वही, जगह लगभग दसवाँ हिस्सा ज़्यादा।
ज़्यादातर पदार्थ \omterm{def:g3:solids-liquids-gases:solid}{ठोस} बनते समय सिकुड़ते हैं; पानी फैलता है --- और यही
अनोखापन इलाक़ों के नक़्शे तक बदल देता है।
\end{proposition}

\begin{example}[फटा पाइप और चटकता पहाड़]\label{ex:g6:water-states:burst}
फ़्रीज़र में भूली हुई, पानी से भरी काँच की बोतल चटक जाती है: फैलती हुई
बर्फ़ को अपना अतिरिक्त दसवाँ हिस्सा चाहिए ही, और वह उसे ले लेती है ---
काँच हो या न हो। जाड़ा यही काम नलों के साथ करता है --- पाइप में \omterm{def:g2:ice-water-steam:meltfreeze}{जमता}
पानी उसे हथौड़े जितनी पक्की तरह फोड़ देता है --- और पहाड़ों के साथ
भी: बारिश चट्टान की दरारों में रिसती है, रात में जमती है, फैलती है,
और दरार को और चौड़ा कर देती है। साल-दर-साल, पाले-दर-पाले,
\cref{prop:g6:water-states:volume} वाला वह छोटा-सा अनोखा फैलाव
चट्टानें चीर देता है और चोटियाँ भुरभुरा देता है।
\end{example}

\begin{remark}[बर्फ़ तैरती क्यों है]\label{rem:g6:water-states:floats}
वही अनोखापन उस सवाल का उत्तर भी दे देता है जो तुमने वर्षों पहले नहाते
समय पूछा था। बर्फ़ एक \omterm{def:g6:mass-and-volume:volume}{लीटर} का पदार्थ $1.1$ \omterm{def:g6:mass-and-volume:volume}{लीटर} जगह में रखती है: बराबर
जगह के लिए बर्फ़ पानी से \emph{हल्की} है --- इसलिए बर्फ़ अपने ही \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}
पर \omterm{def:g1:floating-and-sinking:words}{तैरती} है। हिमशैल समुद्र पर सवार रहते हैं, तालाब ऊपर से नीचे की ओर
जमते हैं (और नीचे की मछलियाँ बच जाती हैं), और तुम्हारी शिकंजी के
टुकड़े डूबने के बजाय डोलते रहते हैं। बहुत कम पदार्थ अपने ही ऊपर तैरते
हैं; जमे हुए तालाबों का जीवन ख़ुश है कि पानी ऐसा करता है।
\end{remark}

\section{जल-चक्र, हिसाब के साथ}

\begin{example}[वह बड़ा फेरा, शब्दावली सहित]\label{ex:g6:water-states:cycle}
समुद्र, बादल और बारिश की बचपन वाली कहानी अब बही-खाते की तरह पढ़ी जाती
है। सूर्य की गरमी समुद्रों से \emph{\omterm{def:g6:water-states:changes}{वाष्पन}} चलाती है --- टनों-टन पानी
अनदेखा ऊपर उठता है। ऊपर की ठंड में वह वाष्प बादल की बूँदों में
\emph{\omterm{def:g2:ice-water-steam:evapcond}{संघनित}} होती है; और ठंड बढ़े तो बूँदें जमकर हिम बन जाती हैं। हिम
जमा होकर हिमनद बनता है; हिमनद \emph{पिघलकर} नदियाँ बनते हैं; और
नदियाँ हर \omterm{def:g3:measuring-mass:units}{ग्राम} समुद्र को लौटा देती हैं। पूरे चक्कर में
\cref{prop:g6:water-states:mass} टिका रहता है: \omterm{def:g4:solar-system:planet}{ग्रह} न एक बूँद कमाता
है, न गँवाता --- तुम्हारे गिलास का पानी यह फेरा डायनासोरों से भी पहले
से लगा रहा है।
\end{example}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:6}
एक \omterm{def:g6:mass-and-volume:volume}{लीटर} पानी फ़्रीज़र में जाता है। बाहर लगभग कितने \omterm{def:g6:mass-and-volume:volume}{आयतन} की बर्फ़ आती
है? वही सवाल \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} के लिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:7}
कड़ाके के जाड़ों में पानी के पाइप क्यों फट जाते हैं --- और माली शरद
में बाहर के नल क्यों ख़ाली करा देते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:8}
बर्फ़ पानी पर क्यों \omterm{def:g1:floating-and-sinking:words}{तैरती} है? \cref{prop:g6:water-states:volume} को
तैरने के नियम से जोड़ो --- बराबर जगह के लिए हल्की या भारी?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:9}
जल-चक्र को बही-खाते की तरह सुनाओ: समुद्र के \qty{1}{kg} पानी का पूरे
चक्कर में पीछा करो, हर दरवाज़े का नाम लेते हुए और हर पड़ाव पर उसका
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बताते हुए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:10}
शिकंजी का गिलास ऊपर किनारे तक भरा है और उसमें बर्फ़ के टुकड़े डोल रहे
हैं। एक मेहमान को चिंता है कि बर्फ़ पिघलते ही वह छलक जाएगा। दोनों
जाँच-पड़तालों (\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बचा रहता है, और बर्फ़ अपनी अतिरिक्त जगह लौटा
देती है) से मेहमान को तसल्ली दो --- या डराओ --- अपने तर्क के साथ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:11}
सूखे कमरे में रात भर जमती खिड़की के शीशे पर पाले के पंख उग आते हैं,
जबकि न बारिश हुई थी और न सोते समय शीशा गीला था। पानी ने कौन-सा
दरवाज़ा लिया, कहाँ से आकर, और इस दरवाज़े का नाम ``\omterm{def:g2:ice-water-steam:meltfreeze}{जमना}'' क्यों नहीं
है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:water-states:12}
तालाब ऊपर से नीचे की ओर जमते हैं। पूरी शृंखला समझाओ: सबसे ठंडे पानी की
बर्फ़ \emph{सतह पर} क्यों बनती है और वहीं क्यों टिकी रहती है, तैरता
हुआ ढक्कन नीचे की \omterm{def:g5:heat-and-insulation:heat}{ऊष्मा} के बहाव के साथ क्या करता है (पिछले साल का
रोधन वाला विचार), और नीचे की मछलियाँ अपने जाड़े पानी के उस अनोखे
फैलाव की क्यों ऋणी हैं।
\end{exercise}

\section{समस्या: दुनिया भर में एक किलोग्राम}

\begin{problem}[{सप्ताहांत समस्या --- समुद्र के एक किलोग्राम पानी की
बड़ी यात्रा; छह दरवाज़े, तीन पहाड़, और एक अटूट बही-खाता}]\label{pb:g6:water-states:1}
हम समुद्र के \qty{1}{kg} पानी पर एक काल्पनिक लेबल चिपकाते हैं और साल
भर उसका पीछा करते हैं।

\medskip\noindent\textbf{भाग I --- ऊपर की ओर।}
\begin{enumerate}
  \item जून: सूर्य समुद्र की सतह को गरम करता है और हमारा \omterm{def:g3:measuring-mass:units}{किलोग्राम}
        अनदेखा होकर \omterm{def:g2:air-around-us:air}{हवा} में उठ जाता है। उसने कौन-सा दरवाज़ा लिया?
  \item हमारे (अब अनदेखे) एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} पानी का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} कितना है?
        कौन-सी प्रतिज्ञप्ति इसकी गारंटी देती है?
  \item \qty{3000}{m} पर \omterm{def:g2:air-around-us:air}{हवा} ठंडी है: पानी बूँदों का बादल बन जाता
        है। इस दरवाज़े का नाम बताओ।
  \item और ठंड बढ़े तो बूँदें हिमकण बन जाती हैं। उस दरवाज़े का नाम
        --- \omterm{def:g6:water-states:changes}{ठोसीकरण} --- बताओ, और वह \omterm{def:g3:temperature-thermometers:temperature}{ताप} भी जिस पर शुद्ध पानी की
        बादल-बूँदें \omterm{def:g2:ice-water-steam:meltfreeze}{जमना} शुरू करती हैं।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- हिमनद का बही-खाता।}
हमारा \omterm{def:g3:measuring-mass:units}{किलोग्राम} किसी हिमनद पर गिरता है और दबकर हिमनद की बर्फ़ बन जाता
है।
\begin{enumerate}[resume]
  \item पानी के रूप में हमारा \omterm{def:g3:measuring-mass:units}{किलोग्राम} \qty{1}{L} जगह घेरता था।
        बर्फ़ के रूप में वह लगभग कितना \omterm{def:g6:mass-and-volume:volume}{आयतन} घेरता है?
  \item गरमियों में हिमनद का सिरा पिघलता है। हमारा \omterm{def:g3:measuring-mass:units}{किलोग्राम} पिघला
        हुआ पानी बन जाता है: अब उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} कितना है, और (फिर से
        \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} पानी के रूप में) उसका \omterm{def:g6:mass-and-volume:volume}{आयतन}?
  \item पहाड़ से नीचे उतरते समय, धूप में चमकते किसी झरने के कुंड से
        हमारे \omterm{def:g3:measuring-mass:units}{किलोग्राम} में से \qty{250}{g} \omterm{def:g2:ice-water-steam:evapcond}{वाष्पित} हो जाता है।
        लेबल लगा कितना पानी धारा में \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} के रूप में आगे चलता है?
  \item क्या \omterm{def:g2:ice-water-steam:evapcond}{वाष्पित} हुआ चौथाई हिसाब से बाहर हो गया? बताओ कि उसके
        \qty{250}{g} कहाँ हैं, और वे सबसे संभव किस दरवाज़े से आगे
        जाएँगे।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- घर वापसी, और बही-खाता
बराबर।}
\begin{enumerate}[resume]
  \item धारा के \qty{750}{g} अक्टूबर में समुद्र पहुँच जाते हैं।
        \omterm{def:g2:ice-water-steam:evapcond}{वाष्पित} हुए \qty{250}{g} अगस्त में तट पर बारिश बनकर गिरे और
        सितंबर में बहकर समुद्र में मिल गए। जोड़ो कि हमारे \omterm{def:g3:measuring-mass:units}{किलोग्राम}
        में से समुद्र को कितना वापस मिला।
  \item पूरी यात्रा में हमारा \omterm{def:g3:measuring-mass:units}{किलोग्राम} कभी भी भार-रहित नहीं हुआ ---
        पर वह दो बार \emph{अनदेखा} ज़रूर हुआ। किन पड़ावों पर, और
        किस अवस्था में?
  \item एक सहपाठी आपत्ति करता है: ``हिम तो फूला हुआ और हल्का होता है
        --- यानी हिम के रूप में \omterm{def:g3:measuring-mass:units}{किलोग्राम} का \omterm{def:g4:weight-and-mass:weight}{भार} कम रहा होगा।'' इस
        साल के सही शब्द-युग्म से यह गड़बड़ी सुलझाओ।
  \item पूरी यात्रा को दरवाज़ों की एक शृंखला के रूप में, क्रम में
        लिखो, और सिर्फ़ \cref{def:g6:water-states:changes} के छह
        ठीक नामों का उपयोग करो।
\end{enumerate}
\end{problem}
