\chapter{विज्ञान में मापन: मात्रक और उपकरण}\label{ch:g6:measurement-in-science}

एक बढ़ई, एक डॉक्टर और एक खगोलविद अपने-अपने काम पर निकलते हैं: एक ताक़
मापता है, दूसरा बुख़ार, और तीसरा मंगल की भटकन। तीनों उसी चुपचाप चलती
मशीनरी पर भरोसा करते हैं --- सबकी मानी हुई मात्रक-प्रणाली, ईमानदार
उपकरण, और ध्यान से पढ़ना। इस साल भौतिकी ठीक-ठीक संख्याओं वाली हो जाती
है, इसलिए हम उन्हीं औज़ारों को चमकाकर शुरू करते हैं जिन पर आगे का हर
अध्याय टिकेगा।

\section{``लगने'' से संख्या तक}

पाँच साल के प्रयोगों ने हमें एक ताल सिखा दी है: आँखें और हाथ अंदाज़ा
लगाते हैं, उपकरण फ़ैसला करते हैं। ``ज़्यादा लंबा लगने'' के सामने रूलर;
``ज़्यादा भारी लगने'' के सामने \omterm{def:g3:levers-and-scales:scale}{तुला}; ``ठंडा लगने'' के सामने \omterm{def:g3:temperature-thermometers:temperature}{तापमापी}।
हर बार उपकरण का फ़ैसला \emph{\omterm{def:g6:measurement-in-science:measuring}{मात्रक} सहित एक संख्या} होता है --- और यह
जोड़ी सफ़र करती है, तुलना करती है, और कोई भी इसे जाँच सकता है, जो
किसी एहसास से कभी नहीं हो सकता था।

\begin{definition}[किसी राशि को मापना]\label{def:g6:measurement-in-science:measuring}
किसी राशि को --- कोई लंबाई, कोई \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}, कोई \omterm{def:g2:measuring-time:duration}{अवधि}, कोई \omterm{def:g3:temperature-thermometers:temperature}{ताप} ---
\emph{मापना}\index{मापना} यानी उस काम के लिए बने उपकरण से यह गिनना कि
उसका माना हुआ \emph{मात्रक}\index{मात्रक} उसमें कितनी बार समाता है।
परिणाम हमेशा संख्या \emph{और} उसके मात्रक के साथ लिखा जाता है:
\qty{28.5}{cm}, \qty{450}{g}, \qty{37.2}{\celsius}। अकेली संख्या
अफ़वाह है; संख्या-सहित-मात्रक मापन है।
\end{definition}

\begin{example}[ग़ायब मात्रक की क़ीमत]\label{ex:g6:measurement-in-science:missing}
``आटा 3 डाल देना।'' \omterm{def:g3:measuring-mass:units}{ग्राम}? चम्मच? कटोरी? केक का भाग्य उसी ग़ायब शब्द
पर टँगा है। ग़ायब या आपस में गड्डमड्ड हुए मात्रकों से सचमुच की
दुर्घटनाएँ हो चुकी हैं --- कुछ विमानों का ईंधन बीच उड़ान में इसलिए
ख़त्म हो गया कि \omterm{def:g3:measuring-mass:units}{किलोग्राम} को पाउंड समझ लिया गया था। पेशेवर लोग \omterm{def:g6:measurement-in-science:measuring}{मात्रक}
हर बार लिखते हैं, बिना किसी अपवाद के; इस अध्याय के बाद से हम भी।
\end{example}

\section{दुनिया के मात्रक}

\begin{definition}[वैज्ञानिक का शुरुआती डिब्बा]\label{def:g6:measurement-in-science:units}
दुनिया ने साझा \omterm{def:g6:measurement-in-science:measuring}{मात्रक} तय कर लिए हैं, हर राशि के लिए एक कुनबा:
\begin{enumerate}
  \item लंबाई: \emph{\omterm{def:g2:measuring-length:units}{मीटर}} (\unit{m}), और उसके बच्चे किलोमीटर
        ($\qty{1}{km} = \qty{1000}{m}$), \omterm{def:g2:measuring-length:units}{सेंटीमीटर}
        ($\qty{100}{cm} = \qty{1}{m}$) तथा मिलीमीटर
        ($\qty{1000}{mm} = \qty{1}{m}$);
  \item \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}: \emph{\omterm{def:g3:measuring-mass:units}{किलोग्राम}} (\unit{kg}), और उसके साथ \omterm{def:g3:measuring-mass:units}{ग्राम}
        ($\qty{1}{kg} = \qty{1000}{g}$) तथा मिलीग्राम
        ($\qty{1000}{mg} = \qty{1}{g}$);
  \item समय: \emph{सेकंड} (\unit{s}), और उसके साथ मिनट
        ($\qty{1}{min} = \qty{60}{s}$) तथा घंटा
        ($\qty{1}{h} = \qty{60}{min} = \qty{3600}{s}$);
  \item \omterm{def:g3:temperature-thermometers:temperature}{ताप}: \omterm{def:g3:temperature-thermometers:temperature}{डिग्री सेल्सियस} (\unit{\celsius});
  \item द्रवों का आयतन: लीटर (\unit{L}), जिसे हम अगले अध्याय में
        नापेंगे।
\end{enumerate}
\end{definition}

\begin{example}[वैज्ञानिक की तरह बदलना]\label{ex:g6:measurement-in-science:convert}
\omterm{def:g6:measurement-in-science:measuring}{मात्रक} बदलना दशमलव की सीढ़ी पर सवारी भर है। $\qty{2.35}{m}$ को
\omterm{def:g2:measuring-length:units}{सेंटीमीटर} में: $100$ से गुणा करो --- बिंदु दो जगह दाईं ओर सरक जाता है:
\qty{235}{cm}। $\qty{4500}{g}$ को \omterm{def:g3:measuring-mass:units}{किलोग्राम} में: $1000$ से भाग दो:
\qty{4.5}{kg}। समय इस कुनबे का बाग़ी है --- उसके पायदान $10$ के नहीं,
$60$ के हैं: $\qty{2.5}{min}$ का मतलब $250$ सेकंड नहीं, बल्कि
$2 \times 60 + 30 = \qty{150}{s}$ है। बाग़ी की इज़्ज़त करो।
\end{example}

\begin{remark}[मीटर की पहरेदारी कौन करता है?]\label{rem:g6:measurement-in-science:guard}
दो सदियों तक दुनिया का \omterm{def:g2:measuring-length:units}{मीटर} किसी बड़ी यूरोपीय राजधानी के पास एक
पहरेदार तहख़ाने में रखी क़ीमती धातु की छड़ था, और उसके बग़ल में
\omterm{def:g3:measuring-mass:units}{किलोग्राम} एक चमकता बेलन: हर देश के रूलर और बाट उन्हीं की नक़लों की
नक़लें थे। आज पहरेदार सेवामुक्त कर दिए गए हैं: \omterm{def:g2:measuring-length:units}{मीटर} और उसके साथी अब
ख़ुद प्रकृति के अटल तथ्यों से परिभाषित होते हैं --- \omterm{def:g1:light-and-shadows:source}{प्रकाश} से, और
परमाणुओं की भीतरी घड़ियों से --- ताकि पृथ्वी की (या उससे बाहर की भी!)
कोई भी अच्छी तरह सजी प्रयोगशाला उन्हें ठीक-ठीक फिर से बना सके। \omterm{def:g6:measurement-in-science:measuring}{मात्रक}
मानव जाति का सबसे लंबा चलने वाला शांतिपूर्ण समझौता हैं।
\end{remark}

\section{उपकरणों को ठीक से काम में लेना}

\begin{method}[किसी भी अंशांकित मापनी को पढ़ना]\label{met:g6:measurement-in-science:read}
रूलर, \omterm{def:g3:temperature-thermometers:temperature}{तापमापी}, मापक जग, डायल: इन सबको पढ़ने का हुनर एक ही है।
\begin{enumerate}
  \item \emph{एक पायदान का मान} निकालो: संख्या लिखे दो पड़ोसी निशान
        चुनो, घटाओ, और उनके बीच के छोटे पायदानों की संख्या से भाग
        दो --- हर \omterm{def:g2:measuring-length:units}{सेंटीमीटर} पर संख्या और बीच में दस पायदान वाले
        रूलर पर हर पायदान $1 \div 10 = \qty{0.1}{cm}$ का, यानी एक
        मिलीमीटर;
  \item अपनी आँख ठीक निशान के सामने रखो (तिरछी नज़र पाठ को खिसका
        देती है);
  \item संख्या लिखे आख़िरी निशान से आगे गिनो: $7$ और $3$ छोटे
        पायदान यानी $7.3$;
  \item संख्या को \emph{उसके \omterm{def:g6:measurement-in-science:measuring}{मात्रक} के साथ} लिखो।
\end{enumerate}
\end{method}

\begin{method}[अंदाज़ा लगाओ, मापो, परखो]\label{met:g6:measurement-in-science:estimate}
पेशेवर लोग कभी बिना तैयारी के मापने नहीं बैठते:
\begin{enumerate}
  \item पहले \emph{अंदाज़ा} लगाओ --- ``यह मेज़ लगभग दो \omterm{def:g2:measuring-length:units}{मीटर} की
        है'';
  \item वह उपकरण \emph{चुनो} जिसका परास और जिसके पायदान इस काम पर
        बैठते हों (टिकट के लिए रूलर, गलियारे के लिए फ़ीता, आटे के
        लिए रसोई की तराज़ू, लोगों के लिए वज़न वाली तराज़ू);
  \item पढ़ने के नियमों के अनुसार \emph{मापो};
  \item फिर \emph{परखो}: क्या परिणाम मोटे तौर पर अंदाज़े से मेल खाता
        है? \qty{21.3}{m} मापी गई मेज़ बड़ी नहीं हो गई --- तुमने
        दस गुने के फ़र्क़ से ग़लत पढ़ा, और अंदाज़े ने अभी-अभी उसे पकड़
        लिया।
\end{enumerate}
अंदाज़ा तुम्हारा सुरक्षा-जाल है: वह फिसले हुए दशमलव बिंदुओं को तुम्हें
पकड़ने से पहले पकड़ लेता है।
\end{method}

\begin{example}[समझदारी से चुनना]\label{ex:g6:measurement-in-science:choose}
डाक के लिए चिट्ठी तौलनी हो तो वज़न वाली तराज़ू बेकार है --- चिट्ठी के
\qty{18}{g} उसके पायदानों के बीच ग़ायब हो जाते हैं; ग्राम-दर-ग्राम
चलने वाली रसोई की तराज़ू सही उत्तर देती है। $100$ \omterm{def:g2:measuring-length:units}{मीटर} की दौड़ का समय
लेना हो तो रसोई की घड़ी के मिनट वाले पायदान बेकार हैं; स्टॉपवॉच सेकंड
के सौवें हिस्से गिनती है। इनमें से कोई उपकरण ``बेहतर'' नहीं है: हर एक
का अपना इलाक़ा है, जो उसके परास और उसके सबसे छोटे पायदान से तय होता
है।
\end{example}

\section{ईमानदार संख्याएँ}

\begin{remark}[निशानों के बीच]\label{rem:g6:measurement-in-science:between}
असली सुइयों को निशानों के बीच गिरना बहुत पसंद है। तब ईमानदारी के दो
स्तर होते हैं: कम से कम, दोनों पड़ोसी बताओ --- ``$7.3$ और $7.4$ के
बीच, $7.3$ के ज़्यादा पास''; और उससे बेहतर, निशानों से एक अंक आगे का
अंदाज़ा लगाओ --- ``लगभग \qty{7.32}{cm}''। ईमानदारी जिस चीज़ को मना
करती है वह है परिशुद्धि गढ़ लेना: मिलीमीटर वाले निशानों से
\qty{7.3214}{cm} पढ़ना प्रयोगशाला का कोट पहने हुआ झूठ है। किसी उपकरण
के निशान ही तय करते हैं कि वह कितना वादा कर सकता है।
\end{remark}

\begin{example}[दो बार मापो, ज़्यादा भरोसा करो]\label{ex:g6:measurement-in-science:repeat}
तीन साथी एक ही खिलौना गाड़ी की एक ही दौड़ का समय एक ही स्टॉपवॉच से
लेते हैं: \qty{4.6}{s}, \qty{4.8}{s}, \qty{4.7}{s}। किसी ने ग़लती
नहीं की --- उँगलियाँ बस बाल भर के फ़र्क़ से चालू और बंद होती हैं।
थोड़ा-सा बिखराव असली मापन की सामान्य साँस है। समझदारी की आदतें: मापन
दोहराओ, देखो कि पाठ कहाँ जुटते हैं, और किसी भी अकेली-अलग संख्या पर
शक करो --- ख़ासकर अपनी ही।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:1}
हर मापन को समझदारी वाले \omterm{def:g6:measurement-in-science:measuring}{मात्रक} से पूरा करो: एक दरवाज़ा $2$ \dots\
ऊँचा है; \ एक सिक्के का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} $8$ \dots\ है; \ स्कूल का एक पाठ
$55$ \dots\ चलता है; \ बुख़ार $39$ \dots\ है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:2}
बदलो: \qty{3.2}{m} को \unit{cm} में; \ \qty{750}{g} को \unit{kg}
में; \ \qty{4}{min} को सेकंड में; \ \qty{5600}{m} को \unit{km} में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:3}
समय मात्रक-कुनबे का बाग़ी क्यों है? \qty{3.5}{min} को सेकंड में
बदलो --- और बताओ कि दशमलव की सीढ़ी कौन-सा ग़लत उत्तर देती।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:4}
एक जग पर हर \qty{100}{mL} पर संख्या लिखी है और दो संख्याओं के बीच
पाँच छोटे पायदान हैं। एक पायदान कितने का है? रस $300$ के निशान से दो
पायदान ऊपर खड़ा है --- रस कितना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:5}
हर काम के लिए कौन-सा उपकरण, और क्यों: चुटकी भर ख़मीर तौलना; \ नाड़ी
गिनने का समय लेना; \ इस पन्ने की चौड़ाई; \ स्कूल के मैदान की लंबाई?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:6}
नादिया अपनी पेंसिल मापती है: \qty{17.4}{m}।
\cref{met:g6:measurement-in-science:estimate} का कौन-सा सुरक्षा-जाल
इसे पकड़ लेता, और असली मान क्या होगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:7}
किसी \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} का स्तंभ $18$ और $19$ के निशानों के बीच, $19$ के
ज़्यादा पास रुकता है। इसे बताने का एक ईमानदार तरीक़ा लिखो --- और एक
बेईमान तरीक़ा भी, जिससे बचना है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measurement-in-science:8}
एक ही दौड़ के तीन स्टॉपवॉच-पाठ: \qty{6.1}{s}, \qty{6.3}{s},
\qty{6.2}{s}। क्या किसी की ग़लती है? समझदारी वाली रिपोर्ट क्या
होगी?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:9}
किसी विदेशी पाक-विधि में ``चौथाई पाउंड मक्खन'' माँगा गया है।
तुम्हारी रसोई की तराज़ू \omterm{def:g3:measuring-mass:units}{ग्राम} की भाषा बोलती है, और पूरा एक पाउंड लगभग
\qty{450}{g} होता है। तुम कितने \omterm{def:g3:measuring-mass:units}{ग्राम} तौलोगे? यह काम साझा मात्रकों
के बारे में क्या सिखाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:10}
किसी रूलर का पहला \omterm{def:g2:measuring-length:units}{सेंटीमीटर} घिस चुका है, इसलिए उसका किनारा $1$ के
निशान से शुरू होता है। किनारे से रखे गए एक टिकट का दूसरा सिरा $3.4$
पर पड़ता है। टिकट असल में कितना चौड़ा है? घिसे हुए रूलरों को कौन-सी
आदत हरा देती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measurement-in-science:11}
कक्षा \qty{20}{m} के फ़ीते से गलियारा दो बार मापती है:
\qty{18.25}{m} और \qty{18.35}{m}। शिक्षिका लिखती हैं
``\qty{18.3}{m}''। \cref{rem:g6:measurement-in-science:between} और
\cref{ex:g6:measurement-in-science:repeat} की मदद से उनका चुनाव
समझाओ।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:measurement-in-science:12}
एक पुरानी कहानी: किसी राजा ने अपने ही पैर को राज्य का लंबाई-मात्रक
घोषित कर दिया। इससे निकलने वाली दो व्यावहारिक आफ़तें गिनाओ (दूर के
क़स्बों के बढ़ई सोचो, और राजा के बड़े होते बेटे को भी, जो उत्तराधिकारी
है), और बताओ कि \cref{rem:g6:measurement-in-science:guard} वाली
सेवामुक्त धातु की छड़ ने दोनों को कैसे हल किया --- और वह छड़ भी
कौन-सी समस्या छोड़ गई, जिसे आख़िरकार प्रकृति के अपने नियतांकों ने ठीक
किया।
\end{exercise}

\section{समस्या: उपकरण-निरीक्षक}

\begin{problem}[{सप्ताहांत समस्या --- क़स्बे की उपकरण-निरीक्षक के साथ
एक दिन; कठघरे में तराज़ू, जग और घड़ियाँ; और नौसिखिए की कॉपी}]\label{pb:g6:measurement-in-science:1}
हर साल क़स्बे की निरीक्षक दुकानों और स्कूलों के उपकरण जाँचती हैं। आज
तुम उनके नौसिखिए हो, हाथ में कॉपी लिए।

\medskip\noindent\textbf{भाग I --- नानबाई की तराज़ू।}
निरीक्षक अपने साथ प्रमाणित बाट लाई हैं: \qty{500}{g},
\qty{200}{g}, \qty{200}{g}, \qty{100}{g}, \qty{50}{g}।
\begin{enumerate}
  \item वे पहले जाँचती हैं कि ख़ाली तराज़ू की सुई ठीक शून्य पर खड़ी
        है। यह सबसे पहली जाँच क्यों होनी चाहिए?
  \item वे \qty{500}{g} और एक \qty{200}{g} का बाट रखती हैं।
        ईमानदार तराज़ू को क्या दिखाना चाहिए?
  \item सुई इसके बजाय \qty{680}{g} दिखाती है। यह तराज़ू नानबाई की
        चापलूसी कर रही है या ग्राहकों की? बताओ कि हर सौदे में पैसा
        कौन गँवाता है।
  \item उनके बाटों के कौन-से जोड़ से \qty{750}{g} का पाठ जाँचा जा
        सकता है? एक तरीक़ा बताओ।
  \item नानबाई की तराज़ू \qty{5}{g} के पायदानों में चलती है। क्या
        निरीक्षक उससे \qty{2}{g} का डाक टिकट जाँच सकती हैं? क्यों
        नहीं?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- स्कूल के मापक जग।}
\begin{enumerate}[resume]
  \item एक जग पर हर \qty{200}{mL} पर संख्या लिखी है, और दो
        संख्याओं के बीच चार पायदान हैं। हर पायदान कितने का है?
  \item पानी \qty{600}{mL} के निशान से एक पायदान ऊपर खड़ा है।
        शिक्षक की डायरी कहती है ``लगभग \qty{700}{mL}''। इस रिपोर्ट
        को परखो: ईमानदार, बहुत ढीली, या गढ़ी हुई?
  \item एक विद्यार्थी नीची कुर्सी पर बैठकर, आँख पानी की सतह से
        काफ़ी नीचे रखकर जग पढ़ता है, और उसे निरीक्षक से अलग संख्या
        मिलती है। पढ़ने का कौन-सा नियम टूटा?
  \item निरीक्षक जग के घोषित \qty{650}{mL} को प्रमाणित जग में
        उँडेलती हैं, जो \qty{640}{mL} दिखाता है। स्कूल का जग कितना
        भटका हुआ है, और यह उसके अपने एक पायदान से बुरा है या बेहतर?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- घड़ियाँ, और नौसिखिए का
फ़ैसला।}
\begin{enumerate}[resume]
  \item स्टेशन की घड़ी रेडियो के समय-संकेत से जाँची जाती है: संकेत
        की दोपहर पर वह $12$ बजकर $2$ मिनट दिखाती है। वह तेज़ है या
        सुस्त, और कितनी?
  \item अगर उसे कोई ठीक न करे, तो हफ़्ते भर बाद वह कितनी आगे निकल
        चुकी होगी? (वह रोज़ उतना ही आगे बढ़ती है।)
  \item संकेत की दोपहर पर तीन दुकानों की घड़ियाँ $12$ बजकर $2$
        मिनट, $11$ बजकर $58$ मिनट, और ठीक $12$ बजे दिखाती हैं।
        निरीक्षक कहती हैं: ``यहाँ औसत निकालना बेवक़ूफ़ी होगी ---
        हर घड़ी अलग से ठीक करो।'' हर एक को कौन-सी मरम्मत चाहिए?
  \item शाम। अपनी कॉपी में निरीक्षक के तीन सुनहरे नियम लिखो ---
        एक शून्य के बारे में, एक पढ़ने के बारे में, और एक मात्रकों
        के बारे में --- हर एक अपने शब्दों के एक ही वाक्य में।
\end{enumerate}
\end{problem}
