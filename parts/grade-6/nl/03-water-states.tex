\chapter{Water in al zijn toestanden}\label{ch:g6:water-states}

Je jaagt al sinds je tweede schooljaar op water in zijn toestanden:
ijs, water, damp; smeltende sneeuwpoppen en beslagen ramen. Dit jaar
wordt de jacht boekhouding. Gaat er stof verloren als ijs
\omterm{def:g2:ice-water-steam:meltfreeze}{smelt}? Waarom barsten leidingen als water \omterm{def:g2:ice-water-steam:meltfreeze}{bevriest}? Met balansen
en maatcilinders in de hand controleren we alle drie de toestanden van
water.

\section{De drie toestanden, met hun officiële etiketten}

\begin{definition}[De zes toestandsovergangen]\label{def:g6:water-states:changes}
Tussen zijn drie toestanden heeft water zes deuren, elk met een
wetenschappelijke naam:
\begin{enumerate}
  \item vast $\to$ vloeibaar: \emph{smelten}\index{smelten};
        vloeibaar $\to$ vast: \emph{stollen}\index{stollen}
        (\omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen});
  \item vloeibaar $\to$ \omterm{def:g3:solids-liquids-gases:gas}{gas}: \emph{verdamping}\index{verdamping}
        --- razendsnel bij het koken, stil en koel als verdamping aan
        het oppervlak; \omterm{def:g3:solids-liquids-gases:gas}{gas} $\to$ vloeibaar:
        \emph{condensatie}\index{condensatie};
  \item vast $\to$ \omterm{def:g3:solids-liquids-gases:gas}{gas} rechtstreeks:
        \emph{sublimatie}\index{sublimatie}, en \omterm{def:g3:solids-liquids-gases:gas}{gas} $\to$
        vast rechtstreeks --- zeldzamere deuren, maar echt: ijsbloemen
        op een vriezend raam groeien rechtstreeks uit onzichtbare damp
        en slaan de \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} helemaal over.
\end{enumerate}
\end{definition}

\begin{example}[Overal deuren om je heen]\label{ex:g6:water-states:doors}
Ochtendrijp die van het gras \omterm{def:g2:ice-water-steam:meltfreeze}{smelt}; was die aan de lijn \omterm{def:g2:ice-water-steam:evapcond}{verdampt}; een
koude \omterm{def:g5:mirrors-reflection:reflection}{spiegel} die de damp van je douche laat \omterm{def:g2:ice-water-steam:evapcond}{condenseren}; de vriezer
die de ijsblokjes van morgen laat \omterm{def:g6:water-states:changes}{stollen}; ijsbloemen die 's nachts op
de ruit zijn gegroeid. Eén stof, zes deuren, allemaal binnen één dag
gewoon leven --- benoem ze en je bezit ze.
\end{example}

\section{De controle: gaat er stof verloren?}

\begin{method}[Een toestandsovergang wegen]\label{met:g6:water-states:weigh}
Leg de vraag voor aan de balans:
\begin{enumerate}
  \item giet wat water in een klein plastic flesje, draai de dop
        stevig dicht en droog de buitenkant af;
  \item weeg het: zeg \qty{368}{g} --- schrijf het op;
  \item leg het een nacht in de vriezer: \omterm{def:g6:water-states:changes}{stollen};
  \item weeg het bevroren flesje meteen (veeg eventuele rijp eraf).
\end{enumerate}
De balans antwoordt: \qty{368}{g}, tot op de \omterm{def:g3:measuring-mass:units}{gram}. Laat het op de
vensterbank weer \omterm{def:g6:water-states:changes}{smelten} en weeg opnieuw: \qty{368}{g}. De toestand
veranderde twee keer; de stof kneep geen oog dicht.
\end{method}

\begin{proposition}[Massa overleeft toestandsovergangen]\label{prop:g6:water-states:mass}
Een toestandsovergang verandert het uiterlijk van een stof, niet haar
hoeveelheid: de \omterm{def:g3:measuring-mass:mass}{massa} ervóór is precies gelijk aan de \omterm{def:g3:measuring-mass:mass}{massa} erna.
\omterm{def:g6:water-states:changes}{Smelten}, \omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen}, koken in een afgesloten pan, \omterm{def:g2:ice-water-steam:evapcond}{condenseren} ---
door elke deur leest de balans hetzelfde. (Laat de dop er echter af,
en damp trekt weg met haar grammen: de stof bestaat nog steeds, maar
je controle bevat haar niet meer.)
\end{proposition}

\begin{example}[Waar de grammen van de plas heen gingen]\label{ex:g6:water-states:puddle}
Een plas van \qty{300}{g} droogt op. Verdwenen? Niets meer waard? Nee:
\qty{300}{g} water rijdt nu als onzichtbare damp op de \omterm{def:g2:air-around-us:air}{lucht}, elke
\omterm{def:g3:measuring-mass:units}{gram} intact --- een deel om volgende week als regen te vallen.
\omterm{def:g6:water-states:changes}{Verdamping} houdt het oog voor de gek, nooit de boekhouding. De
krijtlijn van je plassenproef uit je kindertijd was een controle; nu
heeft ze getallen.
\end{example}

\section{De controle: blijft de ruimte gelijk?}

\begin{proposition}[Water zwelt als het bevriest]\label{prop:g6:water-states:volume}
\omterm{def:g3:measuring-mass:mass}{Massa} overleeft de deuren --- \omterm{def:g6:mass-and-volume:volume}{volume} niet. Water is een merkwaardige
stof: als het vast wordt, \emph{zet het uit}. Een \omterm{def:g6:mass-and-volume:volume}{liter} water maakt
ongeveer $1.1$ \omterm{def:g6:mass-and-volume:volume}{liter} ijs: dezelfde \omterm{def:g3:measuring-mass:mass}{massa}, ongeveer een tiende meer
ruimte. De meeste stoffen krimpen als ze \omterm{def:g6:water-states:changes}{stollen}; water zwelt --- en
die eigenaardigheid hertekent landschappen.
\end{proposition}

\begin{example}[De gesprongen leiding en de gebarsten berg]\label{ex:g6:water-states:burst}
Vergeten in de vriezer barst een volle glazen fles water: het
zwellende ijs heeft zijn extra tiende ruimte nodig en neemt die, glas
of geen glas. De winter doet hetzelfde met leidingwerk --- water dat
in een pijp \omterm{def:g2:ice-water-steam:meltfreeze}{bevriest} laat hem net zo zeker springen als een hamer ---
en met bergen: regen sijpelt in scheuren in het gesteente,
\omterm{def:g2:ice-water-steam:meltfreeze}{bevriest} 's nachts, zwelt op en wigt de scheur wijder. Jaar na jaar,
vorst na vorst, splijt het rare kleine zwellen van
\cref{prop:g6:water-states:volume} zwerfkeien en verkruimelt het
bergtoppen.
\end{example}

\begin{remark}[Waarom ijs drijft]\label{rem:g6:water-states:floats}
Dezelfde eigenaardigheid beantwoordt een vraag die je jaren geleden in
bad stelde. IJs bevat de stof van een \omterm{def:g6:mass-and-volume:volume}{liter} in $1.1$ \omterm{def:g6:mass-and-volume:volume}{liter} ruimte:
voor gelijke ruimte is ijs \emph{lichter} dan water --- dus \omterm{def:g1:floating-and-sinking:words}{drijft}
ijs op zijn eigen \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}. IJsbergen varen op de zee, vijvers
\omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen} van boven naar beneden (wat de vissen eronder redt), en de
blokjes in je limonade dobberen in plaats van te \omterm{def:g1:floating-and-sinking:words}{zinken}. Weinig
stoffen \omterm{def:g1:floating-and-sinking:words}{drijven} op zichzelf; het leven op bevroren vijvers is er blij
mee dat water dat wel doet.
\end{remark}

\section{De waterkringloop, gecontroleerd}

\begin{example}[De grote rondreis, met vaktaal]\label{ex:g6:water-states:cycle}
Het kinderverhaal van zee, wolk en regen leest nu als een grootboek.
De \omterm{def:g5:heat-and-insulation:heat}{warmte} van de Zon drijft \emph{verdamping} vanaf de zeeën aan ---
tonnen op tonnen stijgen onzichtbaar op. Hoog en koud
\emph{\omterm{def:g2:ice-water-steam:evapcond}{condenseert}} de damp tot wolkendruppeltjes; nog kouder
\omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen} de druppeltjes tot sneeuw. Sneeuw stapelt zich op tot
gletsjers; gletsjers \emph{\omterm{def:g6:water-states:changes}{smelten}} tot rivieren; rivieren brengen elke
\omterm{def:g3:measuring-mass:units}{gram} terug naar de zee. Over de hele kringloop geldt
\cref{prop:g6:water-states:mass}: de \omterm{def:g4:solar-system:planet}{planeet} wint noch verliest een
druppel --- het water in jouw glas heeft deze lus al bevaren van vóór
de dinosauriërs.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:water-states:1}
Noem de zes deuren van \cref{def:g6:water-states:changes}, elk met één
alledaagse waarneming.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:2}
Welke deur: dauw die bij het ochtendgloren op koud gras ontstaat? \
een sneeuwpop die op een zonnige dag van \qty{-5}{\celsius} krimpt
zonder dat er een plas verschijnt? \ ijsblokjes die de vriezer met
``rook'' vullen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:3}
Een afgesloten fles water weegt \qty{412}{g}. Ze \omterm{def:g2:ice-water-steam:meltfreeze}{bevriest} 's nachts
tot vast ijs. Wat weegt ze nu, en welke propositie geeft het
antwoord?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:4}
Waarom moet het flesje in \cref{met:g6:water-states:weigh} dicht zitten
--- en waarom moet het vóór elke weging afgedroogd worden?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:5}
Een pan water kookt tien minuten zonder deksel en haar \omterm{def:g3:measuring-mass:mass}{massa} daalt
met \qty{200}{g}. Is er \omterm{def:g3:measuring-mass:mass}{massa} vernietigd? Waar zijn die
grammen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:6}
Eén \omterm{def:g6:mass-and-volume:volume}{liter} water gaat de vriezer in. Ongeveer welk \omterm{def:g6:mass-and-volume:volume}{volume} ijs komt
eruit? Dezelfde vraag voor de \omterm{def:g3:measuring-mass:mass}{massa}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:7}
Waarom springen waterleidingen in strenge winters --- en waarom tappen
tuiniers hun buitenkranen in de herfst af?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:water-states:8}
Waarom \omterm{def:g1:floating-and-sinking:words}{drijft} ijs op water? Verbind
\cref{prop:g6:water-states:volume} met de drijfregel --- lichter of
zwaarder voor dezelfde ruimte?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:9}
Vertel de waterkringloop als een grootboek: volg \qty{1}{kg} zeewater
de hele lus rond, benoem elke deur die het passeert en geef in elk
stadium zijn \omterm{def:g3:measuring-mass:mass}{massa}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:10}
Een limonadeglas is tot de rand gevuld, met dobberende ijsblokjes. Een
gast is bang dat het overloopt als het ijs \omterm{def:g2:ice-water-steam:meltfreeze}{smelt}. Gebruik de twee
controles (\omterm{def:g3:measuring-mass:mass}{massa} blijft behouden, ijs geeft zijn extra ruimte terug) om
de gast gerust te stellen --- of te alarmeren --- met je redenering.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:water-states:11}
Er verschijnen 's nachts ijsbloemen op een vriezende ruit in een droge
kamer, zonder regen en met een droge ruit bij het slapengaan. Welke
deur nam het water, vanwaar, en waarom heet die deur geen ``\omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen}''?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:water-states:12}
Vijvers \omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen} van boven naar beneden. Leg de hele keten uit:
waarom het ijs van het koudste water zich \emph{aan het oppervlak}
vormt en daar blijft, wat het drijvende deksel met de warmtestroom
eronder doet (een isolatie-idee van vorig jaar), en waarom de vissen
eronder hun winters te danken hebben aan het vreemde zwellen van
water.
\end{exercise}

\section{Probleem: een kilogram de wereld rond}

\begin{problem}[{Weekendprobleem --- één kilogram zeewater op grote
tournee; zes deuren, drie bergen, één ononderbroken
grootboek}]\label{pb:g6:water-states:1}
We plakken een denkbeeldig etiket op \qty{1}{kg} zeewater en volgen
het een jaar lang.

\medskip\noindent\textbf{Deel I --- Omhoog.}
\begin{enumerate}
  \item Juni: de Zon verwarmt het zeeoppervlak en onze
        \omterm{def:g3:measuring-mass:units}{kilogram} stijgt onzichtbaar de \omterm{def:g2:air-around-us:air}{lucht} in. Noem de deur die hij
        nam.
  \item Wat is de \omterm{def:g3:measuring-mass:mass}{massa} van onze (nu onzichtbare) \omterm{def:g3:measuring-mass:units}{kilogram} water?
        Welke propositie garandeert dat?
  \item Op \qty{3000}{m} is de \omterm{def:g2:air-around-us:air}{lucht} koud: het water wordt een wolk
        van druppeltjes. Noem deze deur.
  \item Nog kouder worden de druppeltjes sneeuwvlokken. Noem die deur
        --- \omterm{def:g6:water-states:changes}{stollen} --- en geef de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} waarbij
        wolkendruppeltjes zuiver water beginnen te \omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen}.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Het grootboek van de gletsjer.}
Onze \omterm{def:g3:measuring-mass:units}{kilogram} landt op een gletsjer en wordt tot gletsjerijs
samengeperst.
\begin{enumerate}[resume]
  \item Als water nam onze \omterm{def:g3:measuring-mass:units}{kilogram} \qty{1}{L} in. Ongeveer welk
        \omterm{def:g6:mass-and-volume:volume}{volume} neemt hij als ijs in?
  \item De tong van de gletsjer \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} in de zomer. Onze
        \omterm{def:g3:measuring-mass:units}{kilogram} wordt smeltwater: wat is nu zijn \omterm{def:g3:measuring-mass:mass}{massa}, en zijn
        \omterm{def:g6:mass-and-volume:volume}{volume} (weer als vloeibaar water)?
  \item Op weg de berg af \omterm{def:g2:ice-water-steam:evapcond}{verdampt} \qty{250}{g} van onze
        \omterm{def:g3:measuring-mass:units}{kilogram} uit een zonovergoten waterval\-kom. Hoeveel van het
        gemerkte water gaat als \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} verder in de beek?
  \item Heeft het verdampte kwart de controle verlaten? Zeg waar die
        \qty{250}{g} zijn, en welke deur ze hoogstwaarschijnlijk
        hierna nemen.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Thuis, en de boeken sluitend.}
\begin{enumerate}[resume]
  \item De \qty{750}{g} van de beek bereiken de zee in oktober. De
        verdampte \qty{250}{g} viel in augustus als regen boven de
        kust en stroomde in september naar zee. Tel op wat de zee van
        onze \omterm{def:g3:measuring-mass:units}{kilogram} terugkreeg.
  \item Tijdens de hele tournee was onze \omterm{def:g3:measuring-mass:units}{kilogram} precies nooit
        gewichtloos --- maar hij was twee keer \emph{onzichtbaar}. In
        welke stadia, en in welke toestand?
  \item Een klasgenoot werpt tegen: ``Sneeuw is pluizig en \omterm{def:g1:light-and-shadows:source}{licht} ---
        de \omterm{def:g3:measuring-mass:units}{kilogram} woog als sneeuw vast minder.'' Ontwar de
        verwarring met het juiste paar woorden van dit jaar.
  \item Schrijf de tournee op als een keten van deuren, op volgorde,
        met alleen de zes officiële namen uit
        \cref{def:g6:water-states:changes}.
\end{enumerate}
\end{problem}
