\chapter{Het stelsel Zon--Aarde--Maan}\label{ch:g6:sun-earth-moon}

Drie werelden besturen onze hemel: de Zon die de dag geeft, de Aarde
die ons draagt, de \omterm{def:g2:moon-in-the-sky:moon}{Maan} die ons gezelschap houdt. Je kent ze alle
drie; nu brengen we het \emph{stelsel} in kaart --- wie draait om wie,
hoe snel, hoe ver --- en lossen we een oude schuld in: het beloofde
geheim van de \omterm{def:g3:sun-days-seasons:year}{seizoenen} zit in dit hoofdstuk.

\section{Drie bewegingen, drie klokken}

\begin{definition}[As en omloopbaan]\label{def:g6:sun-earth-moon:axisorbit}
De \emph{as}\index{as} van de Aarde is de denkbeeldige lijn door haar
polen waaromheen ze draait, als de spil van een tol. Een
\emph{omloopbaan}\index{omloopbaan} is het gesloten pad dat het ene
hemellichaam om het andere aflegt --- vrijwel een cirkel voor de Aarde
om de Zon, en voor de \omterm{def:g2:moon-in-the-sky:moon}{Maan} om de Aarde.
\end{definition}

\begin{proposition}[De drie bewegingen]\label{prop:g6:sun-earth-moon:motions}
De hele kalender hangt aan drie gelijkmatige bewegingen:
\begin{enumerate}
  \item de Aarde \emph{draait} om haar \omterm{def:g6:sun-earth-moon:axisorbit}{as}: één omwenteling in $24$
        uur --- de dag, die Zon en sterren over onze hemel laat
        paraderen;
  \item de Aarde \emph{doorloopt haar \omterm{def:g6:sun-earth-moon:axisorbit}{omloopbaan}} om de Zon: één
        ronde in ongeveer $365$ dagen en een kwart --- het jaar, dat
        de boog van de Zon door de \omterm{def:g3:sun-days-seasons:year}{seizoenen} omhoog en omlaag laat
        glijden;
  \item de \omterm{def:g2:moon-in-the-sky:moon}{Maan} \emph{doorloopt haar \omterm{def:g6:sun-earth-moon:axisorbit}{omloopbaan}} om de Aarde: één
        ronde in ongeveer een maand --- en terwijl haar zonovergoten
        helft ons om beurten aankijkt, laat ze haar schijngestalten
        paraderen, zoals in je oude maandagboek.
\end{enumerate}
Drie bewegingen, drie klokken: dag, maand, jaar --- de oudste
tijdmeters van de mensheid, en ze lopen alle drie nog.
\end{proposition}

\begin{example}[De kwartdag en het schrikkeljaar]\label{ex:g6:sun-earth-moon:leap}
De ronde van de Aarde duurt $365$ dagen \emph{en ongeveer een kwart}
--- de natuur is geen trouw aan ronde getallen verschuldigd. Kalenders
lossen die schuld af door de kwarten te sparen: vier jaar kwartdagen
maken één hele dag, die elk vierde jaar aan de kalender wordt
toegevoegd --- het \emph{schrikkeljaar} van $366$ dagen, met zijn
extra dag eind februari. Sla die boekhouding een eeuw over en de
kalender \omterm{def:g1:floating-and-sinking:words}{drijft} weken af van de \omterm{def:g3:sun-days-seasons:year}{seizoenen} --- zoals oudere kalenders
pijnlijk hebben ontdekt.
\end{example}

\begin{example}[Het stelsel op schaal]\label{ex:g6:sun-earth-moon:scale}
Getallen voor in je schrift: de Aarde meet ongeveer \qty{12800}{km} in
doorsnede; de \omterm{def:g2:moon-in-the-sky:moon}{Maan} ongeveer een kwart daarvan, en haar
\omterm{def:g6:sun-earth-moon:axisorbit}{omloopbaan} ligt op ongeveer \qty{384000}{km} --- dertig Aardes naast
elkaar. De Zon meet ongeveer $109$ Aardes in doorsnede en staat $150$
miljoen kilometer ver. Krimp de Aarde tot een knikker van een
\omterm{def:g2:measuring-length:units}{centimeter} en het model zegt: \omterm{def:g2:moon-in-the-sky:moon}{Maan}, een peperkorrel op $30$
\omterm{def:g2:measuring-length:units}{centimeter}; Zon, een bal van ruim een \omterm{def:g2:measuring-length:units}{meter} breed, meer dan honderd
\omterm{def:g2:measuring-length:units}{meter} verderop in de straat. Onze hele buurt is vooral leegte ---
zoals het schaalmodel in het park van het \omterm{def:g4:solar-system:family}{zonnestelsel} al fluisterde.
\end{example}

\section{De schuld ingelost: waarom seizoenen}

\begin{proposition}[De scheefstand maakt de seizoenen]\label{prop:g6:sun-earth-moon:tilt}
De \omterm{def:g6:sun-earth-moon:axisorbit}{as} van de Aarde staat niet rechtop op haar \omterm{def:g6:sun-earth-moon:axisorbit}{omloopbaan}: ze helt
over, met ongeveer een kwart rechte hoek --- en ze blijft de hele
jaarronde \emph{dezelfde kant op} hellen, als een trouw \omterm{def:g4:magnets-and-compass:compass}{kompas} van de
sterren. Dus helt onze helft van de wereld gedurende de halve ronde
\emph{naar} de Zon toe: de Zon rijdt hoog, de dagen lopen lang ---
zomer. Een halve ronde later wijst dezelfde onveranderde helling ons
\emph{van} haar af: lage Zon, korte dagen --- winter. De
\omterm{def:g3:sun-days-seasons:year}{seizoenen} zijn niet de Zon die verandert, en ook niet de afstand: ze
zijn een scheefstaande \omterm{def:g4:solar-system:planet}{planeet} die haar onveranderde helling om haar
\omterm{def:g4:solar-system:star}{ster} heen draagt.
\end{proposition}

\begin{method}[De globe en de lamp]\label{met:g6:sun-earth-moon:globe}
Zie de hele machine op tafel:
\begin{enumerate}
  \item zet midden op tafel een lamp als de Zon, en draag een
        scheefstaande globe (of een bal aan een schuin spiesje)
        langzaam om haar heen, terwijl je \emph{het spiesje de hele
        weg naar dezelfde hoek van de kamer laat wijzen};
  \item stop bij elk kwart van de ronde en kijk naar de
        breedtegraad van je woonplaats: kijk hoeveel van haar
        dagelijkse draaicirkel in het \omterm{def:g1:light-and-shadows:source}{licht} ligt;
  \item aan de ene kant van de lamp koestert jouw halfrond zich
        naar binnen gehellend --- lange verlichte bogen, hoge lamp:
        zomer; aan de andere kant, naar buiten gehellend --- korte
        bogen, lage lamp: winter;
  \item op de twee haltes ertussen helt geen van beide polen naar
        binnen: dag en nacht delen gelijk op --- lente en herfst.
\end{enumerate}
De ene discipline die het laat werken: laat het spiesje nooit
meezwaaien --- de trouw van de \omterm{def:g6:sun-earth-moon:axisorbit}{as} \emph{is} de \omterm{def:g3:sun-days-seasons:year}{seizoenen}.
\end{method}

\begin{example}[De oude aanwijzingen, verantwoord]\label{ex:g6:sun-earth-moon:clues}
Elke waarneming uit je zonnewijzerjaren rapporteert nu aan de
scheefstand. De hoge zomerboog: ons halfrond helt naar binnen, dus
staat de Zon rond het middaguur hoger. De stompe winterboog en de
lange \omterm{def:g1:light-and-shadows:shadow}{schaduwen}: naar buiten gehellend scheert de Zon laag.
Tegengestelde \omterm{def:g3:sun-days-seasons:year}{seizoenen} aan weerszijden van de evenaar ---
decemberstrandvakanties in het zuiden terwijl het noorden sneeuw
schept: als het ene halfrond naar binnen helt, moet het andere naar
buiten hellen. Zelfs de gelijke dagen van lente en herfst rollen uit
de wandeling met lamp en globe. Eén helling van een kwart rechte hoek
runt de hele voorstelling.
\end{example}

\begin{remark}[Niet de afstand --- en het bewijs]\label{rem:g6:sun-earth-moon:notdistance}
De \omterm{def:g6:sun-earth-moon:axisorbit}{omloopbaan} van de Aarde is zo bijna cirkelvormig dat de afstand
tot de Zon nauwelijks verandert --- en, aardig genoeg, rijden we in de
noordelijke winter iets \emph{dichter} bij de Zon. De \omterm{def:g3:sun-days-seasons:year}{seizoenen} geven
om de \emph{schuinte} van het \omterm{def:g1:light-and-shadows:source}{licht}, niet om de lengte van de reis:
naar binnen gehellend valt het zonlicht steil en geconcentreerd; naar
buiten gehellend spreidt hetzelfde \omterm{def:g1:light-and-shadows:source}{licht} zich dun over de grond uit.
Een zaklamp recht van voren tegenover schuin tegen een muur laat het
verschil in één \omterm{def:g2:measuring-time:second}{seconde} zien --- steil \omterm{def:g1:light-and-shadows:source}{licht} bijt, scherend
\omterm{def:g1:light-and-shadows:source}{licht} verwarmt nauwelijks.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:10}

\begin{exercise}[$\star\star$]\label{exo:g6:sun-earth-moon:11}

\begin{exercise}[$\star\star\star$]\label{exo:g6:sun-earth-moon:12}

en midden in de winter wachten ze weken op een \omterm{def:g1:day-and-night:daynight}{zonsopgang}. Verklaar
beide wonderen met de scheefstaande draaicirkels van
\cref{met:g6:sun-earth-moon:globe}: wat doet de helling in juni met de
dagcirkel van een poolstad, en wat in december?
\end{exercise}

\section{Probleem: het model op het sportveld}

\begin{problem}[{Weekendprobleem --- de klas bouwt het stelsel
Zon--Aarde--Maan op het sportveld; knikkers, peperkorrels en een heel
lange wandeling; de scheefstand doet mee}]\label{pb:g6:sun-earth-moon:1}
De klas bouwt het stelsel op schaal: de Aarde is een knikker van
\qty{1}{cm} doorsnede. Handige echte getallen: Aarde \qty{12800}{km}
doorsnede; \omterm{def:g2:moon-in-the-sky:moon}{Maan} \qty{3200}{km} doorsnede, op een \omterm{def:g6:describing-motion:trajectory}{baan} van
\qty{384000}{km}; Zon $109$ Aardes in doorsnede, op $150$ miljoen
kilometer.

\medskip\noindent\textbf{Deel I --- De rollen verdelen.}
\begin{enumerate}
  \item Voor hoeveel kilometer staat \qty{1}{cm} in het model?
  \item De \omterm{def:g2:moon-in-the-sky:moon}{Maan} is ongeveer een kwart van de breedte van de Aarde.
        Hoe breed is de modelmaan --- en welk keukenvoorwerp speelt
        haar goed?
  \item De echte maanafstand is $30$ aardbreedtes. Waar staat de
        modelmaan ten opzichte van de knikker?
  \item De modelzon is $109$ knikkers breed. Hoe breed is dat in
        \omterm{def:g2:measuring-length:units}{centimeter} --- en welk voorwerp op het schoolplein is ongeveer
        zo groot?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De lange wandeling.}
\begin{enumerate}[resume]
  \item De afstand tot de Zon is $150$ miljoen km. Hoeveel
        \omterm{def:g2:measuring-length:units}{centimeter} is dat in het model --- deel door het antwoord op
        vraag 1 --- en hoeveel \omterm{def:g2:measuring-length:units}{meter}?
  \item Past het model op een sportveld van \qty{100}{m}? Waar moet
        de zonnebal staan?
  \item Een leerling loopt het stuk Aarde--Zon met $1$ \omterm{def:g2:measuring-length:units}{meter} per
        \omterm{def:g2:measuring-time:second}{seconde}. Ongeveer hoeveel minuten kost die wandeling?
  \item Vanaf de knikker terugkijkend naar de meterbrede zonnebal op
        $117$ \omterm{def:g2:measuring-length:units}{meter} is de klas verrast: hij lijkt ongeveer even groot
        als de maankorrel van een kwart \omterm{def:g2:measuring-length:units}{centimeter} op $30$
        \omterm{def:g2:measuring-length:units}{centimeter} van de knikker. Controleer de twee verhoudingen
        --- en verbind dit toeval met iets wat zelden en luisterrijk
        aan de echte hemel gebeurt.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het model laten lopen.}
\begin{enumerate}[resume]
  \item Om het model eerlijk te animeren: hoe lang moet de knikker
        doen over een ronde om de zonnebal, en de erwt over een
        ronde om de knikker, als één echte dag door één \omterm{def:g2:measuring-time:second}{seconde}
        gespeeld wordt? (Reken ruwweg: een jaar als $365$ dagen, een
        maanronde als $27$ dagen.)
  \item De knikker moet ook draaien. Hoe snel is dat in het model
        bij één echte dag per \omterm{def:g2:measuring-time:second}{seconde} --- en waarom zal niemand het
        aan een knikker van \qty{1}{cm} zien?
  \item Een spiesje door de knikker speelt de \omterm{def:g6:sun-earth-moon:axisorbit}{as}. Onder welke
        schuinte, en waarheen wijzend, moet het spiesje de hele ronde
        worden meegedragen?
  \item Schrijf de eerlijke bijsluiter van het model: noem één ding
        dat het sportveldmodel getrouw laat zien, en één ding dat
        geen enkel model ter grootte van een veld tegelijk kan tonen
        (groottes en afstanden samen? bewegingen? \omterm{def:g1:light-and-shadows:source}{licht}?). Onderbouw
        het met je getallen.
\end{enumerate}
\end{problem}
