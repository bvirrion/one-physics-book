\chapter{Circuitos eléctricos y potencia}\label{ch:g11:circuits-and-power}

Acciona un interruptor: una lámpara se enciende, porque un generador
situado a muchos kilómetros empuja carga por un lazo de cobre que pasa
por tu pared. Los cursos anteriores cartografiaron el circuito ---
intensidad, \omterm{def:g11:circuits-and-power:voltage}{tensión}, \omterm{prop:g11:circuits-and-power:ohm}{ley de Ohm}. Este capítulo lo vuelve a abrir con el
libro de cuentas del \cref{ch:g10:energy-conservation}: qué cobra el
contador, por qué se ponen incandescentes los hilos finos y por qué la
red de transporte trabaja a \qty{400}{kV}.

\section{La carga en movimiento}

\begin{definition}[Intensidad de corriente y culombio]\label{def:g11:circuits-and-power:current}
Una \emph{corriente eléctrica}\index{corriente eléctrica} es un flujo
ordenado de carga (de electrones, en un metal). Su intensidad $I$, en
\emph{amperios}\index{amperio} (\unit{A}), es la carga que atraviesa por
segundo una sección del hilo: una corriente constante $I$ que circula
durante un tiempo $t$ transporta la carga
\[ q = I\,t , \]
en \emph{culombios}\index{culombio} (\unit{C}), con
$\qty{1}{C} = \qty{1}{A.s}$. La carga elemental vale
$e = \qty{1.6e-19}{C}$. La intensidad se mide con un amperímetro,
intercalado en serie.
\end{definition}

\begin{example}[El rayo frente al móvil]\label{ex:g11:circuits-and-power:coulomb}
Un rayo lleva unos $\qty{3.0e4}{A}$ durante unos
$\qty{1.0e-4}{s}$: $q = 3.0\times10^{4} \times 10^{-4} = \qty{3.0}{C}$.
Tu móvil, a \qty{2.0}{A}, mueve esa misma carga cada $1.5$ segundos.
\end{example}

\begin{definition}[Tensión y voltio]\label{def:g11:circuits-and-power:voltage}
La \emph{tensión}\index{tensión} $U$ entre dos puntos de un circuito, en
\emph{voltios}\index{voltio} (\unit{V}), es la \omterm{def:g10:energy-conservation:energy}{energía} intercambiada por
\omterm{def:g10:orders-of-magnitude:unit}{unidad} de carga que viaja entre ellos: $\qty{1}{V} = \qty{1}{J/C}$, de
modo que una carga $q$ que atraviesa un dispositivo bajo una tensión $U$
intercambia con él la \omterm{def:g10:energy-conservation:energy}{energía} $E = qU$. Se mide con un voltímetro,
conectado en paralelo.
\end{definition}

\begin{proposition}[Leyes de los circuitos]\label{prop:g11:circuits-and-power:laws}
En un circuito que funciona en régimen estacionario:
\begin{enumerate}
  \item \emph{ley de los nudos:} las intensidades que llegan a un nudo
        suman lo mismo que las que salen de él;
  \item \emph{ley de las mallas en serie:} las tensiones se suman a lo
        largo de un camino: $U_{AC} = U_{AB} + U_{BC}$;
  \item \emph{ley del paralelo:} dos ramas que unen los mismos dos
        puntos soportan la misma \omterm{def:g11:circuits-and-power:voltage}{tensión}.
\end{enumerate}
\end{proposition}

\begin{proof}
En régimen estacionario la carga no se acumula en ningún sitio, así que
lo que entra en un nudo sale de él. La \omterm{def:g10:energy-conservation:energy}{energía} por \omterm{def:g11:circuits-and-power:current}{culombio} es aditiva a
lo largo de un camino y depende únicamente de sus extremos --- no de la
rama que se tome entre ellos.
\end{proof}

\section{La ley de Ohm y las asociaciones de resistencias}

\begin{definition}[Resistencia]\label{def:g11:circuits-and-power:resistance}
La \emph{resistencia}\index{resistencia} de un conductor es el cociente
$R = U/I$ entre la \omterm{def:g11:circuits-and-power:voltage}{tensión} en sus extremos y la intensidad que lo
atraviesa, medida en \emph{ohmios}\index{ohmio} (\unit{\ohm}):
$\qty{1}{\ohm} = \qty{1}{V/A}$.
\end{definition}

\begin{proposition}[Ley de Ohm]\label{prop:g11:circuits-and-power:ohm}
\index{ley de Ohm}
Para un conductor metálico mantenido a temperatura fija, $R$ es
constante: la \omterm{def:g11:circuits-and-power:voltage}{tensión} y la intensidad son proporcionales,
\[ U = R\,I . \]
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:circuits-and-power:ohm-limits}
Es una ley experimental, no universal: un diodo, el filamento de una
lámpara o tu propia piel no la cumplen. Por qué la cumplen los metales
se deduce en los volúmenes universitarios.
\end{remark}

\begin{proposition}[Resistencias en serie y en paralelo]\label{prop:g11:circuits-and-power:networks}
\index{resistencia equivalente}
Dos \omterm{def:g11:circuits-and-power:resistance}{resistencias} en serie equivalen a una sola \omterm{def:g11:circuits-and-power:resistance}{resistencia}
$R_{\text{s}} = R_1 + R_2$; dos en paralelo, a una \omterm{def:g11:circuits-and-power:resistance}{resistencia}
$R_{\text{p}}$ dada por
\[
\frac{1}{R_{\text{p}}} = \frac{1}{R_1} + \frac{1}{R_2} ,
\qquad\text{es decir}\quad
R_{\text{p}} = \frac{R_1 R_2}{R_1 + R_2} < \min(R_1, R_2) .
\]
\end{proposition}

\begin{proof}
En serie: la misma $I$ atraviesa las dos y las tensiones se suman
(\cref{prop:g11:circuits-and-power:laws}), luego $U = (R_1 + R_2)\,I$.
En paralelo: las dos están bajo la misma $U$ y las intensidades se
suman, luego $I = \frac{U}{R_1} + \frac{U}{R_2}$.
\end{proof}

\begin{example}[Resistencia equivalente]\label{ex:g11:circuits-and-power:networks}
$R_1 = \qty{100}{\ohm}$ y $R_2 = \qty{150}{\ohm}$: en serie,
\qty{250}{\ohm}; en paralelo,
$\frac{100 \times 150}{250} = \qty{60}{\ohm}$ --- menos que cualquiera
de las dos: la corriente dispone de un segundo camino.
\end{example}

\section{Energía y potencia en un circuito}

\begin{proposition}[Potencia eléctrica]\label{prop:g11:circuits-and-power:power}
\index{potencia eléctrica}
Un dispositivo sometido a una \omterm{def:g11:circuits-and-power:voltage}{tensión} $U$ y atravesado por una
intensidad $I$ recibe \omterm{def:g10:energy-conservation:energy}{energía} al ritmo
\[ P = U\,I , \qquad\text{es decir}\quad E = U\,I\,t \ \text{en un tiempo } t . \]
\end{proposition}

\begin{proof}
En un tiempo $t$ atraviesa el dispositivo la carga $q = It$
(\cref{def:g11:circuits-and-power:current}) y cada \omterm{def:g11:circuits-and-power:current}{culombio} le entrega
$U$ \omterm{def:g10:energy-conservation:energy}{julios} (\cref{def:g11:circuits-and-power:voltage}): $E = qU = UIt$.
\end{proof}

\begin{proposition}[Potencia en una resistencia]\label{prop:g11:circuits-and-power:ri2}
Una \omterm{def:g11:circuits-and-power:resistance}{resistencia} $R$ recorrida por una intensidad $I$ disipa
\[ P = R\,I^2 = \frac{U^2}{R} . \]
\end{proposition}

\begin{proof}
Basta sustituir la \omterm{prop:g11:circuits-and-power:ohm}{ley de Ohm} en $P = UI$:
$P = (RI)I = RI^2 = U(U/R) = U^2/R$.
\end{proof}

\begin{example}[Leer una placa de características]\label{ex:g11:circuits-and-power:nameplate}
Un hervidor marcado ``\qty{230}{V} -- \qty{2200}{W}'' consume
$I = P/U = 2200/230 = \qty{9.6}{A}$ y tiene una \omterm{def:g11:circuits-and-power:resistance}{resistencia}
$R = U^2/P = 230^2/2200 = \qty{24}{\ohm}$. Funcionando \qty{150}{s}
gasta $E = 2200 \times 150 = \qty{3.3e5}{J} \approx \qty{0.09}{kW.h}$
--- unos dos céntimos.
\end{example}

\begin{remark}[Julios y kilovatios-hora]\label{rem:g11:circuits-and-power:kwh}
El contador cuenta kilovatios-hora
(\cref{def:g10:energy-conservation:kwh}):
$\qty{1}{kW.h} = \qty{3.6e6}{J}$. Los \omterm{def:g10:energy-conservation:energy}{julios} son la \omterm{def:g10:orders-of-magnitude:unit}{unidad} del físico;
los kilovatios-hora ponen precio a las facturas.
\end{remark}

\section{El efecto Joule: de la tostadora a la torre de alta tensión}

\begin{definition}[Efecto Joule]\label{def:g11:circuits-and-power:joule}
La disipación $P = RI^2$ que convierte \omterm{def:g10:energy-conservation:forms}{energía eléctrica} en calor en
toda \omterm{def:g11:circuits-and-power:resistance}{resistencia} es el \emph{efecto Joule}\index{efecto Joule}. Las
estufas, los hervidores, las tostadoras y las bombillas de antaño son
\omterm{def:g11:circuits-and-power:resistance}{resistencias} por diseño; un \emph{fusible}\index{fusible} es un hilo
fino calibrado para fundirse --- abriendo el circuito --- cuando la
intensidad supera su valor nominal.
\end{definition}

\begin{example}[Por qué las torres van a tensión alta]\label{ex:g11:circuits-and-power:transmission}
Un pueblo necesita $P = \qty{100}{kW}$ por una línea de \omterm{def:g11:circuits-and-power:resistance}{resistencia}
$R_\ell = \qty{5.0}{\ohm}$. Entregada a $U = \qty{500}{V}$:
$I = P/U = \qty{200}{A}$ y la línea derrocha
$R_\ell I^2 = \qty{200}{kW}$ --- el doble de lo entregado: absurdo. A
$U = \qty{20}{kV}$: $I = \qty{5.0}{A}$ y $R_\ell I^2 = \qty{125}{W}$.
Cuarenta veces menos intensidad, $1600$ veces menos pérdida.
\end{example}

\end{circuitikz}

\end{omfigure}

\begin{remark}[La regla de hierro de la red]\label{rem:g11:circuits-and-power:grid}
A \omterm{def:g10:energy-conservation:power}{potencia} entregada fija $P = UI$, la pérdida de línea
$R_\ell I^2 = R_\ell P^2/U^2$ cae como el \emph{cuadrado} de la \omterm{def:g11:circuits-and-power:voltage}{tensión}
de transporte. De ahí la receta: transformar hacia arriba
(\qty{400}{kV}) para viajar y hacia abajo (\qty{230}{V}) para convivir
con ella.
\end{remark}

\begin{remark}[Seguridad doméstica]\label{rem:g11:circuits-and-power:safety}
Una línea de \qty{230}{V} protegida con un \omterm{def:g11:circuits-and-power:joule}{fusible} de \qty{16}{A}
alimenta como mucho $230 \times 16 \approx \qty{3.7}{kW}$: dos
hervidores a la vez funden el \omterm{def:g11:circuits-and-power:joule}{fusible} --- que es, por diseño, el eslabón
más fino y débil, mucho más barato que un incendio dentro de la pared.
\end{remark}

\section{Los generadores reales: la fuerza electromotriz}

\begin{definition}[Fem y resistencia interna]\label{def:g11:circuits-and-power:emf}
Una pila convierte \omterm{def:g10:energy-conservation:forms}{energía química} en \omterm{def:g10:energy-conservation:forms}{energía eléctrica}. Su \emph{fuerza
electromotriz}\index{fuerza electromotriz} (fem) $\mathcal{E}$, en
\omterm{def:g11:circuits-and-power:voltage}{voltios}, es la \omterm{def:g10:energy-conservation:energy}{energía} que entrega a cada \omterm{def:g11:circuits-and-power:current}{culombio} que la atraviesa; su
\emph{resistencia interna}\index{resistencia interna} $r$ da cuenta del
\omterm{def:g11:circuits-and-power:joule}{efecto Joule} dentro de su propia química. Modelo: una fuente ideal
$\mathcal{E}$ en serie con $r$ (la $\mathcal{E}$ caligráfica: $E$ es la
\omterm{def:g10:energy-conservation:energy}{energía}).
\end{definition}

\begin{proposition}[Tensión en bornes]\label{prop:g11:circuits-and-power:terminal}
Una pila que entrega una intensidad $I$ presenta en sus bornes
\[ U = \mathcal{E} - r\,I . \]
\end{proposition}

\begin{proof}
La química suministra $\mathcal{E}I$, la \omterm{def:g11:circuits-and-power:emf}{resistencia interna} se come
$rI^2$ y el circuito recibe $UI$: la \omterm{thm:g10:energy-conservation:conservation}{conservación de la energía}
(\cref{ch:g10:energy-conservation}) da $UI = \mathcal{E}I - rI^2$.
\end{proof}

\begin{example}[Una pila con carga]\label{ex:g11:circuits-and-power:discharge}
Una pila ($\mathcal{E} = \qty{9.0}{V}$, $r = \qty{1.0}{\ohm}$) alimenta
una bombilla de \omterm{def:g11:circuits-and-power:resistance}{resistencia} $R = \qty{8.0}{\ohm}$. El lazo cumple
$\mathcal{E} = rI + RI$, luego $I = \mathcal{E}/(R + r) = \qty{1.0}{A}$
y $U = 9.0 - 1.0 \times 1.0 = \qty{8.0}{V}$: la bombilla recibe
\qty{8.0}{W} mientras la pila derrocha \qty{1.0}{W} en calentarse.
\end{example}

\begin{remark}[Los receptores]\label{rem:g11:circuits-and-power:receivers}
Un motor o una batería que se carga recorren la conversión al revés:
absorben bajo $U = \mathcal{E}' + r'I$, convierten $\mathcal{E}'I$ en
movimiento o en química y pierden $r'I^2$. Los generadores empujan; los
receptores empujan en contra.
\end{remark}

\begin{example}[Rendimiento de una pila]\label{ex:g11:circuits-and-power:efficiency}
El \omterm{def:g10:energy-conservation:efficiency}{rendimiento} (\cref{def:g10:energy-conservation:efficiency}) de la
descarga anterior: lo útil, $UI = \qty{8.0}{W}$, de los
$\mathcal{E}I = \qty{9.0}{W}$ suministrados, luego
$\eta = U/\mathcal{E} = 8/9 \approx 0.89$ --- y baja a medida que sube
la intensidad: las pilas que trabajan mucho se calientan.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:1}
La batería de un móvil se carga con una intensidad constante de
\qty{2.0}{A} durante \qty{90}{min}. ¿Qué carga pasa por el cable, en
\omterm{def:g11:circuits-and-power:current}{culombios}? ¿A cuántas cargas elementales equivale
($e = \qty{1.6e-19}{C}$)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:2}
Una \omterm{def:g11:circuits-and-power:resistance}{resistencia} de \qty{220}{\ohm} sometida a \qty{12}{V}: ¿qué
intensidad circula? Una \omterm{def:g11:circuits-and-power:resistance}{resistencia} calefactora consume \qty{0.50}{A} a
\qty{230}{V}: ¿qué \omterm{def:g11:circuits-and-power:resistance}{resistencia} tiene?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:3}
Calcula la \omterm{prop:g11:circuits-and-power:networks}{resistencia equivalente} de $R_1 = \qty{120}{\ohm}$ y
$R_2 = \qty{60}{\ohm}$ en serie y después en paralelo. ¿Cuál es menor
que las dos, y por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:4}
Una plancha de viaje consume \qty{8.7}{A} a \qty{230}{V}. Calcula su
\omterm{def:g10:energy-conservation:power}{potencia} y después la \omterm{def:g10:energy-conservation:energy}{energía} que gasta en \qty{4.0}{min}, en \omterm{def:g10:energy-conservation:energy}{julios} y
en kilovatios-hora.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:5}
El cargador de un portátil entrega \qty{60}{W} durante \qty{5.0}{h} al
día: ¿\omterm{def:g10:energy-conservation:energy}{energía} diaria en kilovatios-hora y en \omterm{def:g10:energy-conservation:energy}{julios}, y coste a $0.25$
euros el \unit{kW.h}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:6}
Una estufa eléctrica está marcada como \qty{1200}{W} a \qty{230}{V}.
\begin{enumerate}
  \item Calcula su \omterm{def:g11:circuits-and-power:resistance}{resistencia} y la intensidad que consume.
  \item Enchufada (con la misma \omterm{def:g11:circuits-and-power:resistance}{resistencia}) a una red de \qty{115}{V},
        ¿qué \omterm{def:g10:energy-conservation:power}{potencia} entrega? ¿Por qué no la mitad?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:7}
$R_1 = \qty{40}{\ohm}$ y $R_2 = \qty{20}{\ohm}$ en serie bajo
\qty{12}{V}. Calcula la intensidad, la \omterm{def:g11:circuits-and-power:voltage}{tensión} en cada \omterm{def:g11:circuits-and-power:resistance}{resistencia} y la
\omterm{def:g10:energy-conservation:power}{potencia} disipada en cada una; comprueba que las \omterm{def:g10:energy-conservation:power}{potencias} suman $UI$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:8}
$R_1 = \qty{60}{\ohm}$ y $R_2 = \qty{30}{\ohm}$ en paralelo bajo
\qty{12}{V}. Calcula la intensidad de cada rama, la intensidad total y
la \omterm{prop:g11:circuits-and-power:networks}{resistencia equivalente} de dos maneras.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:9}
Una pila entrega $U = \qty{5.5}{V}$ con $I = \qty{0.50}{A}$ y
$U = \qty{4.0}{V}$ con $I = \qty{2.0}{A}$. Halla su fem $\mathcal{E}$ y
su \omterm{def:g11:circuits-and-power:emf}{resistencia interna} $r$, y después la intensidad de cortocircuito
$\mathcal{E}/r$.
\end{exercise}

\section{Problema: de la presa a la tostadora}

\begin{problem}[{Problema de fin de semana --- el viaje de mil
kilómetros de un kilovatio-hora, del agua que cae al pan que se tuesta,
con la factura completa del trayecto}]
\label{pb:g11:circuits-and-power:1}
Una presa de montaña retiene agua \qty{200}{m} por encima de sus
turbinas, que se tragan \qty{50}{m^3} --- \qty{5.0e4}{kg} --- por
segundo; la turbina y el alternador convierten con un \omterm{def:g10:energy-conservation:efficiency}{rendimiento}
$0.90$. Por una línea de \omterm{def:g11:circuits-and-power:resistance}{resistencia} total
$R_\ell = \qty{32}{\ohm}$ (\qty{1000}{km} de ida y vuelta), la central
alimenta una ciudad donde tu tostadora de \qty{900}{W} espera a
\qty{230}{V}. El alternador entrega \qty{20}{kV}; los transformadores
(\omterm{def:g10:energy-conservation:efficiency}{rendimiento} $0.99$ cada uno) convierten las tensiones a voluntad. La
electricidad se vende a $0.25$ euros el \unit{kW.h};
$g = \qty{9.81}{N/kg}$.

\medskip
\noindent\textbf{Parte I --- En la presa.}
\begin{enumerate}
  \item ¿Cuánta \omterm{def:g10:energy-conservation:energy}{energía} potencial libera cada segundo el agua que cae?
        Deduce la \omterm{def:g10:energy-conservation:power}{potencia} disponible para las turbinas.
  \item ¿Qué \omterm{prop:g11:circuits-and-power:power}{potencia eléctrica} sale del alternador?
  \item A \qty{20}{kV} directamente desde el alternador, ¿qué intensidad
        circula?
  \item Calcula la pérdida por \omterm{def:g11:circuits-and-power:joule}{efecto Joule} $R_\ell I^2$ con esa
        intensidad y compárala con lo que produce la central. Concluye.
  \item La \omterm{def:g10:energy-conservation:power}{potencia} de la ciudad $P = UI$ la fija la demanda. ¿Cuál es
        la única manera de encoger $I$, y qué dispositivo lo hace?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- En la línea.}
\begin{enumerate}[resume]
  \item Un transformador, tratado como ideal en esta pregunta, eleva los
        \qty{88}{MW} a \qty{400}{kV}: ¿qué intensidad circula ahora?
  \item Calcula la nueva pérdida de línea, en megavatios.
  \item ¿Qué fracción de la producción de la central se pierde en la
        línea?
  \item Demuestra que la fracción perdida vale $R_\ell P/U^2$ y
        comprueba que explica el factor entre las preguntas 4 y 7.
  \item Calcula la caída de \omterm{def:g11:circuits-and-power:voltage}{tensión} $R_\ell I$ a lo largo de la línea a
        \qty{400}{kV}. ¿Es coherente con la pregunta 8?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- En la cocina.}
\begin{enumerate}[resume]
  \item Convierte \qty{1}{kW.h} a \omterm{def:g10:energy-conservation:energy}{julios}; ¿qué multiplica el contador?
  \item Calcula la intensidad y la \omterm{def:g11:circuits-and-power:resistance}{resistencia} de la tostadora a
        \qty{230}{V}.
  \item Tres minutos de tostada: la \omterm{def:g10:energy-conservation:energy}{energía} en \omterm{def:g10:energy-conservation:energy}{julios} y en
        kilovatios-hora, y su precio.
  \item La tostadora disipa $RI^2$: explica en una frase por qué la
        misma fórmula es la pérdida en la línea y el producto en la
        cocina.
  \item La tostadora comparte un \omterm{def:g11:circuits-and-power:joule}{fusible} de \qty{16}{A} con un hervidor
        de \qty{2.2}{kW}: ¿aguanta el \omterm{def:g11:circuits-and-power:joule}{fusible}? Se suma una estufa de
        \qty{2.0}{kW}: ¿y ahora?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La auditoría.}
\begin{enumerate}[resume]
  \item La instalación de la propia casa, de \omterm{def:g11:circuits-and-power:resistance}{resistencia}
        \qty{0.05}{\ohm}, lleva los \qty{13.5}{A} de la pregunta 15:
        calcula la pérdida y su fracción sobre los aproximadamente
        \qty{3.1}{kW} entregados.
  \item Encadena los \omterm{def:g10:energy-conservation:efficiency}{rendimientos} (turbina-alternador, transformador
        elevador, línea, transformador reductor e instalación de la
        casa) en un único \omterm{def:g10:energy-conservation:efficiency}{rendimiento} global de la red.
  \item ¿Cuántos \omterm{def:g10:orders-of-magnitude:unit}{kilogramos} (y cuántos \omterm{def:g10:orders-of-magnitude:unit}{metros} cúbicos) de agua de la
        presa tienen que caer para que llegue \emph{un} \omterm{def:g10:energy-conservation:kwh}{kilovatio-hora} a
        tu tostadora?
  \item Evalúa la fracción perdida $R_\ell P/U^2$ de la pregunta 9 con
        $U = \qty{20}{kV}$. ¿Qué significa un valor mayor que $1$?
  \item El remate --- \emph{el precio de la distancia}: en dos frases,
        ¿qué le cuesta a la montaña un \omterm{def:g10:energy-conservation:kwh}{kilovatio-hora} tostado, qué
        fracción muere por el camino y qué único truco lo hace posible?
\end{enumerate}
\end{problem}
