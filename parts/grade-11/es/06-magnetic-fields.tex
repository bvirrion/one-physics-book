\chapter{El magnetismo y el campo magnético}\label{ch:g11:magnetic-fields}

Durante mil años los marineros se guiaron por agujas imantadas sin que
nadie supiera por qué señalan al norte. La respuesta llegó en 1820,
cuando una corriente hizo temblar una brújula: el magnetismo lo fabrican
las cargas en movimiento. Este capítulo cartografía el \omterm{def:g11:magnetic-fields:bfield}{campo magnético}
--- el de los imanes, el de la Tierra, el de las corrientes --- y
después cierra el círculo: el \omterm{def:g11:electric-gravitational-fields:field}{campo} devuelve el empujón a las
corrientes, y así es como gira todo motor eléctrico.

\section{Los imanes y el campo magnético}

\begin{definition}[Imán y polos]\label{def:g11:magnetic-fields:magnet}
Un \emph{imán}\index{imán} atrae al hierro y, colgado de un hilo, gira
hasta encararse al norte. Su acción se concentra en dos \emph{polos
magnéticos}\index{polo magnético}, el \emph{norte} (el extremo que busca
el norte geográfico) y el \emph{sur}: los polos iguales se repelen y los
distintos se atraen. Serrar un imán por la mitad da dos imanes completos
--- ningún experimento ha aislado nunca un polo.
\end{definition}

\begin{definition}[Campo magnético]\label{def:g11:magnetic-fields:bfield}
Un \omterm{def:g11:magnetic-fields:magnet}{imán} modifica el espacio que lo rodea
(\cref{ch:g11:electric-gravitational-fields}): crea en todas partes un
\emph{campo magnético}\index{campo magnético} $\vect B$. Una brújula lee
su dirección (la aguja se alinea con $\vect B$, con el extremo norte por
delante); una \emph{sonda Hall}\index{sonda Hall} mide su módulo, en
\emph{teslas}\index{tesla} (\unit{T}).
\end{definition}

\begin{definition}[Líneas de campo magnético]\label{def:g11:magnetic-fields:lines}
Las \emph{líneas de campo magnético}\index{línea de campo magnético} son
curvas tangentes a $\vect B$, que se apiñan donde el \omterm{def:g11:electric-gravitational-fields:field}{campo} es intenso.
Fuera de un \omterm{def:g11:magnetic-fields:magnet}{imán} van del polo norte al sur; a diferencia de las líneas
de \omterm{def:g11:electric-gravitational-fields:efield}{campo eléctrico}, no tienen extremos: cada una se cierra por dentro
del cuerpo del \omterm{def:g11:magnetic-fields:magnet}{imán}.
\end{definition}

\section{La Tierra es un imán}

\begin{definition}[El campo magnético terrestre]\label{def:g11:magnetic-fields:earth}
Las corrientes del \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hierro fundido de la Tierra convierten al
planeta en un \omterm{def:g11:magnetic-fields:magnet}{imán} gigantesco: en la superficie
$B \approx \qty{50}{\micro T}$, y una aguja libre se alinea con la
componente horizontal del \omterm{def:g11:electric-gravitational-fields:field}{campo} --- eso es la brújula. Los polos
geomagnéticos no son los geográficos; el ángulo entre el norte de la
brújula y el norte verdadero es la
\emph{declinación}\index{declinación}, que se corrige en todos los
rumbos.
\end{definition}

\begin{example}[Corregir un rumbo]\label{ex:g11:magnetic-fields:bearing}
Donde la \omterm{def:g11:magnetic-fields:earth}{declinación} es de $\ang{7}$ oeste, un barco que navegue al
``norte de brújula'' durante \qty{5.0}{km} deriva
$5000 \times \sin\ang{7} \approx \qty{6.1e2}{m}$ al oeste de su
derrota: para ir al norte verdadero hay que gobernar $\ang{7}$ al
\emph{este} de la aguja.
\end{example}

\section{El campo magnético de una corriente}

\begin{remark}[Oersted, 1820]\label{rem:g11:magnetic-fields:oersted}
Durante una clase en 1820, Hans Christian Oersted vio oscilar la aguja
de una brújula al cerrar un circuito cercano: la electricidad y el
magnetismo eran una misma materia --- \emph{las corrientes crean \omterm{def:g11:magnetic-fields:bfield}{campos
magnéticos}}.
\end{remark}

\begin{proposition}[Campo de un hilo rectilíneo]\label{prop:g11:magnetic-fields:wire}
Un hilo rectilíneo largo recorrido por una intensidad $I$ crea un \omterm{def:g11:electric-gravitational-fields:field}{campo}
cuyas líneas son circunferencias centradas en el hilo, en planos
perpendiculares a él. A una distancia $d$ del hilo,
\[
B = \frac{\mu_0\, I}{2\pi d},
\qquad \mu_0 = 4\pi\times 10^{-7}\ \unit{T.m/A} \approx \qty{1.26e-6}{T.m/A}.
\]
\end{proposition}
\admitted

\begin{example}[Un hilo es un imán flojo]\label{ex:g11:magnetic-fields:wire}
A $d = \qty{1.0}{cm}$ de un hilo por el que circulan $I = \qty{15}{A}$:
$B = \num{2e-7} \times 15 / 0.010 = \qty{0.30}{mT}$ --- seis \omterm{def:g11:electric-gravitational-fields:field}{campos}
terrestres, a partir de una corriente que haría saltar el automático de
una casa. Los \omterm{def:g11:electric-gravitational-fields:field}{campos} intensos exigen una geometría mejor.
\end{example}

\begin{definition}[Bobina y solenoide]\label{def:g11:magnetic-fields:solenoid}
Enrollar el hilo concentra su \omterm{def:g11:electric-gravitational-fields:field}{campo}. Una \emph{bobina
plana}\index{bobina} ($N$ espiras en un disco) se comporta como un \omterm{def:g11:magnetic-fields:magnet}{imán}
delgado, con una cara norte y la otra sur; un
\emph{solenoide}\index{solenoide} --- un bobinado cilíndrico largo, de
$n$ espiras por \omterm{def:g10:orders-of-magnitude:unit}{metro} --- se comporta como un \omterm{def:g11:magnetic-fields:magnet}{imán} recto.
\end{definition}

\begin{proposition}[Campo dentro de un solenoide]\label{prop:g11:magnetic-fields:solenoidfield}
Dentro de un \omterm{def:g11:magnetic-fields:solenoid}{solenoide} largo, lejos de los extremos, el \omterm{def:g11:electric-gravitational-fields:field}{campo} es
\emph{uniforme}: paralelo al eje y el mismo en todos los puntos
interiores, de módulo $B = \mu_0\, n\, I$, independiente de la anchura
del hueco. Fuera, el \omterm{def:g11:electric-gravitational-fields:field}{campo} es débil.
\end{proposition}
\admitted

\begin{remark}
Las dos fórmulas se deducen honradamente en el volumen del primer año, a
partir de la ley que relaciona la circulación de un \omterm{def:g11:electric-gravitational-fields:field}{campo} con la
corriente que rodea. Conviene fijarse en lo que cuenta: $n$, las espiras
\emph{por \omterm{def:g10:orders-of-magnitude:unit}{metro}} --- hay que enrollar más apretado, no más largo.
\end{remark}

\begin{method}[La mano derecha, dos veces]\label{met:g11:magnetic-fields:righthand}
\begin{itemize}
  \item \emph{Hilo}: pulgar $=$ corriente; dedos curvados $=$ giro de
        las líneas.
  \item \emph{\omterm{def:g11:magnetic-fields:solenoid}{Bobina}, \omterm{def:g11:magnetic-fields:solenoid}{solenoide}}: dedos curvados $=$ corriente por las
        espiras; pulgar $=$ $\vect B$ interior, apuntando a la cara
        norte.
\end{itemize}
\end{method}

\begin{example}[Números de un solenoide]\label{ex:g11:magnetic-fields:solenoid}
Con $10$ espiras por centímetro ($n = \qty{1000}{m^{-1}}$) y
$I = \qty{5.0}{A}$:
$B = \num{1.26e-6} \times 1000 \times 5.0 \approx \qty{6.3}{mT}$ ---
cien \omterm{def:g11:electric-gravitational-fields:field}{campos} terrestres, regulables con un mando.
\end{example}

\section{Los electroimanes}

\begin{definition}[Electroimán]\label{def:g11:magnetic-fields:electromagnet}
Un \emph{electroimán}\index{electroimán} es un \omterm{def:g11:magnetic-fields:solenoid}{solenoide} bobinado sobre
un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hierro dulce: el hierro se imana en el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la \omterm{def:g11:magnetic-fields:solenoid}{bobina} y
lo multiplica, normalmente por $100$ o $1000$. A diferencia de un \omterm{def:g11:magnetic-fields:magnet}{imán}
permanente, se apaga con la corriente.
\end{definition}

\begin{remark}[Tres máquinas]\label{rem:g11:magnetic-fields:machines}
La grúa del desguace levanta un coche con un \omterm{def:g11:magnetic-fields:electromagnet}{electroimán} y --- ahí está
la gracia --- lo \emph{suelta} abriendo un interruptor. El pequeño
\omterm{def:g11:magnetic-fields:electromagnet}{electroimán} de un relé cierra un contacto de un tirón: una corriente
débil conmuta una fuerte (\cref{ch:g11:circuits-and-power}). Un equipo
de resonancia magnética mantiene \qty{1.5}{T} uniformes alrededor del
paciente con un \omterm{def:g11:magnetic-fields:solenoid}{solenoide} superconductor.
\end{remark}

\section{La fuerza de Laplace}

\begin{proposition}[Fuerza de Laplace]\label{prop:g11:magnetic-fields:laplace}
Un hilo rectilíneo de longitud $L$, recorrido por una intensidad $I$ en
un \omterm{def:g11:electric-gravitational-fields:capacitor}{campo uniforme} $\vect B$ \emph{perpendicular} al hilo, nota la
\emph{fuerza de Laplace}\index{fuerza de Laplace} de módulo
$F = I\,L\,B$, perpendicular al hilo y a $\vect B$ a la vez: mano
derecha extendida, pulgar en el sentido de la corriente y dedos
estirados según $\vect B$ --- la palma empuja en el sentido de
$\vect F$. Un hilo \emph{paralelo} al \omterm{def:g11:electric-gravitational-fields:field}{campo} no nota ninguna \omterm{def:g10:inertia:force}{fuerza}.
\end{proposition}
\admitted

\begin{remark}[El raíl que salta]\label{rem:g11:magnetic-fields:rail}
El volumen del primer año deduce esto de la \omterm{def:g10:inertia:force}{fuerza} magnética sobre cada
carga en movimiento. En el laboratorio: una varilla de cobre descansa
sobre dos raíles horizontales entre los polos de un \omterm{def:g11:magnetic-fields:magnet}{imán} (con el \omterm{def:g11:electric-gravitational-fields:field}{campo}
entrando en la página, en la figura); se cierra el interruptor y la
varilla sale disparada a lo largo de los raíles. Si se invierte la
corriente, o se le da la vuelta al \omterm{def:g11:magnetic-fields:magnet}{imán}, sale disparada al otro lado.
\end{remark}

\begin{example}[Números de Laplace]\label{ex:g11:magnetic-fields:laplace}
Una varilla de longitud $L = \qty{5.0}{cm}$ recorrida por
$I = \qty{8.0}{A}$ en $B = \qty{0.50}{T}$:
$F = 8.0 \times 0.050 \times 0.50 = \qty{0.20}{N}$ --- el \omterm{def:g10:universal-gravitation:weight}{peso} de
\qty{20}{g}, sobre un hilo que pesa unos pocos gramos.
\end{example}

\begin{remark}[El motor de corriente continua]\label{rem:g11:magnetic-fields:motor}
Si se dobla el hilo formando una espira dentro del \omterm{def:g11:electric-gravitational-fields:field}{campo}, los dos lados
perpendiculares a $\vect B$ llevan corrientes opuestas, de modo que sus
\omterm{prop:g11:magnetic-fields:laplace}{fuerzas de Laplace} son opuestas --- una pareja que hace girar la espira.
Media vuelta más tarde, esa misma pareja la giraría hacia atrás, así que
un interruptor giratorio (el \emph{conmutador}\index{conmutador})
invierte la corriente cada media vuelta. Todo ventilador y todo
taladro son esta espira, multiplicada.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:1}
Un \omterm{def:g11:magnetic-fields:magnet}{imán} recto se sierra por la mitad, entre sus polos. ¿Qué polos lleva
cada trozo? Si se vuelven a juntar, ¿se atraen o se repelen las caras
recién cortadas? ¿Hay alguna forma de cortarlo que aísle el polo norte?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:2}
Verdadero o falso, con una razón cada vez: (a) dos \omterm{def:g11:magnetic-fields:lines}{líneas de campo
magnético} se cortan donde el \omterm{def:g11:electric-gravitational-fields:field}{campo} es intenso; (b) fuera de un \omterm{def:g11:magnetic-fields:magnet}{imán}, las
\omterm{def:g11:electric-gravitational-fields:lines}{líneas de campo} van del polo norte al polo sur; (c) toda \omterm{def:g11:magnetic-fields:lines}{línea de campo
magnético} es un lazo cerrado; (d) una aguja se asienta perpendicular a
la línea que pasa por ella.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:3}
Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} a \qty{2.0}{cm} de un hilo rectilíneo recorrido por
\qty{10}{A}. ¿Cuántas veces los \qty{50}{\micro T} de la Tierra es eso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:4}
Un \omterm{def:g11:magnetic-fields:solenoid}{solenoide} de $800$ espiras bobinadas sobre \qty{40}{cm} lleva
$I = \qty{1.5}{A}$. Calcula $n$ y después $B$ en su interior; ¿en qué se
convierte $B$ si se duplica la corriente?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:5}
Un tramo de hilo de \qty{10}{cm} lleva \qty{5.0}{A} perpendicular a un
\omterm{def:g11:electric-gravitational-fields:field}{campo} de \qty{0.20}{T}. Calcula la \omterm{prop:g11:magnetic-fields:laplace}{fuerza de Laplace}; ¿al \omterm{def:g10:universal-gravitation:weight}{peso} de qué
masa equivale?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:6}
Donde está una excursionista, la \omterm{def:g11:magnetic-fields:earth}{declinación} es de $\ang{8}$ oeste.
Camina \qty{2.0}{km} ``al norte según la brújula'': ¿cuánto deriva, y
hacia qué lado del norte verdadero? ¿Qué rumbo la habría llevado al
norte verdadero?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:7}
¿A qué distancia de un hilo rectilíneo recorrido por \qty{20}{A} iguala
su \omterm{def:g11:electric-gravitational-fields:field}{campo} a los \qty{50}{\micro T} de la Tierra? ¿Qué dice esto sobre
fiarse de una brújula cerca de cables con corriente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:8}
Diseña un \omterm{def:g11:magnetic-fields:solenoid}{solenoide} que produzca $B = \qty{10}{mT}$ con
$I = \qty{4.0}{A}$: calcula la $n$ necesaria y después el número de
espiras para una \omterm{def:g11:magnetic-fields:solenoid}{bobina} de \qty{25}{cm}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:9}
En el experimento de los raíles, la varilla (masa \qty{20}{g}, longitud
$L = \qty{12}{cm}$ entre los raíles) lleva \qty{10}{A} en un \omterm{def:g11:electric-gravitational-fields:field}{campo}
perpendicular de \qty{0.50}{T}. Calcula la \omterm{prop:g11:magnetic-fields:laplace}{fuerza de Laplace} y después
la aceleración inicial de la varilla; compárala con $g$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:10}
Un hilo horizontal de masa \qty{5.0}{g} y longitud \qty{20}{cm} está en
un \omterm{def:g11:electric-gravitational-fields:field}{campo} horizontal $B = \qty{0.10}{T}$ perpendicular a él. ¿Qué
intensidad hace que la \omterm{prop:g11:magnetic-fields:laplace}{fuerza de Laplace} equilibre el \omterm{def:g10:universal-gravitation:weight}{peso} del hilo, de
modo que levite? ¿Hacia dónde tiene que apuntar la \omterm{def:g10:inertia:force}{fuerza}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:11}
Da la dirección de la \omterm{prop:g11:magnetic-fields:laplace}{fuerza de Laplace} (o di que se anula): (a)
corriente hacia la derecha de la página, $\vect B$ entrando en la
página; (b) corriente hacia la parte superior de la página, $\vect B$
saliendo de la página; (c) corriente paralela a $\vect B$. Usa la mano
derecha extendida.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:12}
Un hilo horizontal va en dirección norte--sur geográfica, \qty{3.0}{cm}
por encima de una brújula; allí la componente horizontal terrestre vale
\qty{20}{\micro T}. El hilo lleva $I = \qty{5.0}{A}$: calcula su \omterm{def:g11:electric-gravitational-fields:field}{campo}
en la brújula, da su dirección y después la desviación de la aguja (los
\omterm{def:g11:electric-gravitational-fields:field}{campos} se suman como vectores).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:13}
El \omterm{def:g11:magnetic-fields:electromagnet}{electroimán} de un desguace es una \omterm{def:g11:magnetic-fields:solenoid}{bobina} de $400$ espiras sobre
\qty{20}{cm}, de \omterm{def:g11:circuits-and-power:resistance}{resistencia} \qty{2.0}{\ohm}, alimentada a \qty{12}{V};
su \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hierro multiplica por $100$ el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la \omterm{def:g11:magnetic-fields:solenoid}{bobina} sola.
Calcula la intensidad, el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la \omterm{def:g11:magnetic-fields:solenoid}{bobina} sola, el \omterm{def:g11:electric-gravitational-fields:field}{campo} con el
\omterm{def:g11:fundamental-interactions:ladder}{núcleo} y la \omterm{def:g10:energy-conservation:power}{potencia} que disipa la \omterm{def:g11:magnetic-fields:solenoid}{bobina}
(\cref{ch:g11:circuits-and-power}). ¿Por qué cae la chatarra en el
instante en que se abre el interruptor?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:14}
La espira rectangular de un motor tiene $N = 50$ vueltas recorridas por
$I = \qty{2.0}{A}$; sus dos lados de \qty{5.0}{cm} son perpendiculares a
un \omterm{def:g11:electric-gravitational-fields:field}{campo} $B = \qty{0.30}{T}$. Calcula la \omterm{def:g10:inertia:force}{fuerza} sobre cada uno de esos
lados; ¿por qué son opuestas las dos \omterm{def:g10:inertia:force}{fuerzas} y qué le hacen a la espira?
¿Por qué los otros dos lados no notan a veces ninguna \omterm{def:g10:inertia:force}{fuerza}? ¿Por qué
hay que invertir la corriente cada media vuelta?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:15}
¿A qué distancia de un cable recorrido por \qty{100}{A} habría que estar
para que su \omterm{def:g11:electric-gravitational-fields:field}{campo} llegase a \qty{1}{T}? Compárala con el radio

se construyen con \omterm{def:g11:magnetic-fields:solenoid}{solenoides} y no con hilos sueltos. Y después: el \omterm{def:g11:electric-gravitational-fields:field}{campo}
de \qty{e8}{T} de la superficie de una estrella de neutrones, ¿cuántas
potencias de diez está por encima del terrestre?
\end{exercise}

\section{Problema: la brújula, la grúa y la resonancia magnética}

\begin{problem}[{Problema de fin de semana --- tres imanes en el
trabajo: una brújula que cruza un océano, un electroimán que levanta un
coche y el solenoide de una resonancia magnética --- un solo campo, a lo
largo de diez potencias de diez, haciendo tres trabajos}]
\label{pb:g11:magnetic-fields:1}
El mismo vector $\vect B$ gobierna un velero con \qty{50}{\micro T},
levanta chatarra con \qty{1}{T} e imagina una rodilla con \qty{1.5}{T}
--- tres máquinas para las tres fórmulas de este capítulo.

\medskip
\noindent\textbf{Parte I --- La brújula.}
En la posición del barco, el \omterm{def:g11:electric-gravitational-fields:field}{campo} terrestre tiene una componente
horizontal de \qty{20}{\micro T} y una vertical de \qty{44}{\micro T};
la \omterm{def:g11:magnetic-fields:earth}{declinación} es de $\ang{6}$ oeste.
\begin{enumerate}
  \item ¿Por qué apunta al norte (magnético) la aguja de una brújula?
  \item Calcula el módulo del \omterm{def:g11:electric-gravitational-fields:field}{campo} total y su ángulo con la horizontal.
  \item Gobernando al ``norte de brújula'' durante \qty{30}{km}, ¿a qué
        distancia al oeste de la derrota al norte verdadero acaba el
        barco?
  \item Una carga de acero desvía la aguja otros $\ang{4}$ al oeste:
        misma pregunta.
  \item Cerca del polo geomagnético una brújula es inservible: ¿qué
        componente tiene la culpa, y por qué?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La grúa.}
El \omterm{def:g11:magnetic-fields:electromagnet}{electroimán} del desguace es un \omterm{def:g11:magnetic-fields:solenoid}{solenoide} de $600$ espiras bobinadas
sobre \qty{30}{cm}, de \omterm{def:g11:circuits-and-power:resistance}{resistencia} \qty{1.5}{\ohm}, alimentado a
\qty{12}{V}; su \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hierro multiplica por $50$ el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la
\omterm{def:g11:magnetic-fields:solenoid}{bobina} sola.
\begin{enumerate}[resume]
  \item Calcula $n$, la densidad de espiras.
  \item Calcula la intensidad en la \omterm{def:g11:magnetic-fields:solenoid}{bobina}
        (\cref{ch:g11:circuits-and-power}).
  \item Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la \omterm{def:g11:magnetic-fields:solenoid}{bobina} sola, $\mu_0 n I$.
  \item Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} con el \omterm{def:g11:fundamental-interactions:ladder}{núcleo}; compáralo con los \qty{0.5}{T}
        del neodimio.
  \item Calcula la \omterm{def:g10:energy-conservation:power}{potencia} disipada en la \omterm{def:g11:magnetic-fields:solenoid}{bobina} y después la \omterm{def:g10:energy-conservation:energy}{energía}
        de un turno de carga de \qty{10}{\minute}.
  \item Existen imanes permanentes así de intensos. ¿Por qué un
        \omterm{def:g11:magnetic-fields:electromagnet}{electroimán} entonces?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El motor del cabrestante.}
El cabrestante de la grúa es un motor de corriente continua: una espira
rectangular de $N = 100$ vueltas, con lados de \qty{4.0}{cm}
perpendiculares a $B = \qty{0.25}{T}$, recorrida por
$I = \qty{3.0}{A}$.
\begin{enumerate}[resume]
  \item Calcula la \omterm{prop:g11:magnetic-fields:laplace}{fuerza de Laplace} sobre cada haz de lados
        perpendicular a $\vect B$.
  \item Las dos \omterm{def:g10:inertia:force}{fuerzas} son iguales y opuestas y, sin embargo, su efecto
        no se cancela. ¿Qué producen, y por qué tienen que ser
        \emph{opuestas}?
  \item ¿Qué \omterm{def:g10:inertia:force}{fuerza} actúa sobre los otros dos lados cuando son paralelos
        a $\vect B$?
  \item Media vuelta después, esas mismas \omterm{def:g10:inertia:force}{fuerzas} deshacerían el giro.
        ¿Qué hace el \omterm{rem:g11:magnetic-fields:motor}{conmutador}, y cuándo?
  \item Da tres cambios de diseño distintos que dupliquen, cada uno por
        separado, la pareja de \omterm{def:g10:inertia:force}{fuerzas}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La resonancia magnética.}
El \omterm{def:g11:magnetic-fields:solenoid}{solenoide} del equipo mantiene \qty{1.5}{T} uniformes con una
intensidad $I = \qty{500}{A}$.
\begin{enumerate}[resume]
  \item ¿Cuántas veces el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la Tierra son \qty{1.5}{T}?
  \item Calcula la densidad de espiras $n$ necesaria \emph{sin} ningún
        \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hierro.
  \item Si el bobinado tuviera una \omterm{def:g11:circuits-and-power:resistance}{resistencia} de \qty{0.20}{\ohm}, ¿qué
        \omterm{def:g10:energy-conservation:power}{potencia} disiparía? ¿Qué propiedad del bobinado real evita esto,
        al precio de un frío extremo?
  \item El remate: sitúa los \omterm{def:g11:electric-gravitational-fields:field}{campos} del barco, de la grúa y de la
        resonancia magnética en la escalera de potencias de diez que va
        del espacio interestelar (\qty{e-10}{T}) a una estrella de
        neutrones (\qty{e8}{T}); ¿qué ha cambiado de una máquina a otra
        --- la física o los \omterm{def:g11:circuits-and-power:current}{amperios}?
\end{enumerate}
\end{problem}
