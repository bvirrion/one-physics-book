\chapter{Campos eléctricos y gravitatorios}\label{ch:g11:electric-gravitational-fields}

¿Cómo tira la Tierra de una manzana a la que nunca toca? El propio
Newton llamó a la acción a distancia ``un absurdo tan grande'' que se
negó a defenderla. La respuesta moderna: una masa, o una carga, modifica
el espacio que la rodea --- crea un \emph{\omterm{def:g11:electric-gravitational-fields:field}{campo}} --- y lo que se
coloque allí responde al \omterm{def:g11:electric-gravitational-fields:field}{campo} en su propia posición. Vamos a construir
los dos grandes ejemplos uno al lado del otro, hasta que la analogía se
convierta en una identidad de forma.

\section{De la acción a distancia al campo}

\begin{definition}[Campo]\label{def:g11:electric-gravitational-fields:field}
Un \emph{campo}\index{campo} asigna un vector a cada punto del espacio.
Una fuente (una carga, una masa) crea su campo en todo su alrededor,
haya o no allí algo que lo sienta; un objeto de prueba nota entonces una
\omterm{def:g10:inertia:force}{fuerza} determinada únicamente por el campo \emph{en su propia posición}.
Esta imagen en dos pasos sustituye a la acción a distancia.
\end{definition}

\begin{remark}
El \omterm{def:g11:electric-gravitational-fields:field}{campo} no es un truco contable: transporta \omterm{def:g10:energy-conservation:energy}{energía}, y los cambios de
la fuente viajan hacia fuera a través de él con velocidad finita --- la
propia luz es un \omterm{def:g11:electric-gravitational-fields:field}{campo} que viaja así
(\cref{ch:g12:light-as-wave}). Aquí estudiamos \omterm{def:g11:electric-gravitational-fields:field}{campos} que no cambian con
el tiempo.
\end{remark}

\section{El campo eléctrico}

\begin{definition}[Campo eléctrico]\label{def:g11:electric-gravitational-fields:efield}
Si una carga de prueba pequeña $q$ colocada en un punto $M$ nota una
\omterm{def:g10:inertia:force}{fuerza} eléctrica $\vect F$, el \emph{campo
eléctrico}\index{campo eléctrico} en $M$ es
\[
\vect E = \frac{\vect F}{q}
\qquad \text{(en \unit{N/C}, que también se escribe \unit{V/m})}.
\]
$\vect E$ depende solo de las fuentes y de $M$, no de la carga de
prueba: duplicar $q$ duplica $\vect F$ y deja el cociente igual.
\end{definition}

\begin{proposition}[Fuerza sobre una carga]\label{prop:g11:electric-gravitational-fields:force}
Una carga $q$ colocada donde el \omterm{def:g11:electric-gravitational-fields:field}{campo} vale $\vect E$ nota la \omterm{def:g10:inertia:force}{fuerza}
$\vect F = q \vect E$: en el sentido de $\vect E$ si $q > 0$ y en
sentido contrario si $q < 0$, de módulo $F = \abs{q}\,E$.
\end{proposition}

\begin{proof}
Basta despejar en la definición; la regla de los signos es el álgebra de
un múltiplo negativo.
\end{proof}

\begin{example}[Leer un campo]\label{ex:g11:electric-gravitational-fields:reading}
Una carga de prueba $q = \qty{+2.0e-8}{C}$ situada en $M$ nota una
\omterm{def:g10:inertia:force}{fuerza} de \qty{6.0e-4}{N} que apunta al este. El \omterm{def:g11:electric-gravitational-fields:field}{campo} en $M$ vale
$E = \num{6.0e-4} / \num{2.0e-8} = \qty{3.0e4}{N/C}$, hacia el este; una
carga $q' = \qty{-1.0e-8}{C}$ colocada en ese mismo punto nota
$F = \num{1.0e-8} \times \num{3.0e4} = \qty{3.0e-4}{N}$ --- hacia el
oeste.
\end{example}

\begin{proposition}[Campo de una carga puntual]\label{prop:g11:electric-gravitational-fields:point}
Una carga puntual $Q$ crea, a una distancia $d$, un \omterm{def:g11:electric-gravitational-fields:field}{campo} dirigido según
la recta que une la carga con el punto --- alejándose de $Q$ si $Q > 0$
y hacia ella si $Q < 0$ --- de módulo
\[
E = k\,\frac{\abs{Q}}{d^2},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{proposition}

\begin{proof}
La ley de Coulomb (\cref{ch:g11:fundamental-interactions}) da la \omterm{def:g10:inertia:force}{fuerza}
sobre una carga de prueba $q$ situada a la distancia $d$: de módulo
$k \abs{qQ} / d^2$, según la recta que las une y repulsiva si los signos
son iguales. Basta dividir entre $q$ y aplicar la
\cref{prop:g11:electric-gravitational-fields:force}.
\end{proof}

\begin{example}[Una esfera cargada]\label{ex:g11:electric-gravitational-fields:sphere}
Una esfera de Van de Graaff lleva $Q = \qty{+5.0e-7}{C}$ (una esfera
pequeña cargada se comporta como una carga puntual en su centro). A
$d = \qty{0.50}{m}$:
\[
E = \num{8.99e9} \times \num{5.0e-7} / 0.50^2 \approx \qty{1.8e4}{N/C},
\]
dirigido radialmente hacia fuera de la esfera. A \qty{1.0}{m} ha bajado
a la cuarta parte: el $1/d^2$ de la \omterm{def:g10:inertia:force}{fuerza} sobrevive en el \omterm{def:g11:electric-gravitational-fields:field}{campo}.
\end{example}

\begin{definition}[Líneas de campo]\label{def:g11:electric-gravitational-fields:lines}
Una \emph{línea de campo}\index{línea de campo} es una curva tangente en
todos sus puntos al \omterm{def:g11:electric-gravitational-fields:field}{campo}, con una flecha que da el sentido del \omterm{def:g11:electric-gravitational-fields:field}{campo}.
Tres reglas:
\begin{itemize}
  \item salen de las cargas positivas y llegan a las negativas;
  \item donde las líneas se apiñan el \omterm{def:g11:electric-gravitational-fields:field}{campo} es intenso; donde se
        separan, débil;
  \item dos líneas no se cortan nunca: el \omterm{def:g11:electric-gravitational-fields:field}{campo} tiene una sola dirección
        en cada punto.
\end{itemize}
\end{definition}

\section{El campo uniforme de un condensador}

\begin{definition}[Condensador plano]\label{def:g11:electric-gravitational-fields:capacitor}
Dos placas metálicas planas y paralelas, enfrentadas, próximas y con
cargas opuestas, forman un \emph{condensador}\index{condensador}. Entre
las placas el \omterm{def:g11:electric-gravitational-fields:field}{campo} es \emph{uniforme}\index{campo uniforme}: el mismo
módulo y la misma dirección en todos los puntos --- perpendicular a las
placas y de la placa positiva a la negativa.
\end{definition}

\begin{proposition}[Campo entre las placas]\label{prop:g11:electric-gravitational-fields:uniform}
Si la tensión entre las placas (\cref{ch:g11:circuits-and-power}) es $U$
y su separación es $d$, el \omterm{def:g11:electric-gravitational-fields:field}{campo} entre ellas tiene módulo
\[
E = \frac{U}{d} \qquad \text{(en \unit{V/m})}.
\]
\end{proposition}
\admitted

\begin{remark}
La uniformidad y $E = U/d$ se deducen honradamente en el volumen del
primer año; la fórmula explica la \omterm{def:g10:orders-of-magnitude:unit}{unidad} \unit{V/m}. Las separaciones
pequeñas dan \omterm{def:g11:electric-gravitational-fields:field}{campos} intensos: \qty{100}{V} en un milímetro ya son
\qty{e5}{V/m}.
\end{remark}

\begin{example}[Desviar un haz de electrones]\label{ex:g11:electric-gravitational-fields:beam}
En un \omterm{def:g10:signals-and-waves:oscilloscope}{osciloscopio}, el haz pasa entre unas placas con
$U = \qty{100}{V}$ y $d = \qty{5.0}{mm}$:
$E = 100 / \num{5.0e-3} = \qty{2.0e4}{V/m}$, y cada electrón nota
$F = \num{1.60e-19} \times \num{2.0e4} = \qty{3.2e-15}{N}$ hacia la
placa positiva --- $\num{4e14}$ veces su \omterm{def:g10:universal-gravitation:weight}{peso} de \qty{8.9e-30}{N}, así
que el haz se desvía de forma visible (la \omterm{def:g10:relative-motion:trajectory}{trayectoria} curva es asunto
del \cref{ch:g12:newtons-laws}).
\end{example}

\section{El campo gravitatorio}

\begin{definition}[Campo gravitatorio]\label{def:g11:electric-gravitational-fields:gfield}
Si una masa de prueba $m$ colocada en un punto nota una \omterm{def:g10:universal-gravitation:force}{fuerza
gravitatoria} (un \omterm{def:g10:universal-gravitation:weight}{peso}) $\vect P$, el \emph{campo
gravitatorio}\index{campo gravitatorio} en ese punto es
\[
\vect g = \frac{\vect P}{m} \qquad \text{(en \unit{N/kg})},
\]
y recíprocamente $\vect P = m \vect g$. Un capítulo anterior introdujo
el número $g$; $\vect g$ le añade la dirección en la que empieza a
moverse una piedra que se suelta.
\end{definition}

\begin{proposition}[Campo de la Tierra]\label{prop:g11:electric-gravitational-fields:earth}
A una distancia $d$ del centro de la Tierra (de masa $M$), el \omterm{def:g11:electric-gravitational-fields:gfield}{campo
gravitatorio} apunta al centro y tiene módulo
\[
g = G\,\frac{M}{d^2},
\qquad G = \qty{6.67e-11}{N.m^2/kg^2}.
\]
\end{proposition}

\begin{proof}
La \omterm{def:g11:fundamental-interactions:four}{gravitación} universal (\cref{ch:g10:universal-gravitation}) da la
\omterm{def:g10:inertia:force}{fuerza} sobre una masa de prueba $m$: de módulo $G M m / d^2$ y dirigida
al centro; basta dividir entre $m$. En la superficie, $d = R$ y esto es
la $g = GM/R^2 \approx \qty{9.81}{N/kg}$ del capítulo anterior.
\end{proof}

\begin{example}[El campo donde vive la Luna]\label{ex:g11:electric-gravitational-fields:moon}
A la distancia de la Luna, $d = \qty{3.84e8}{m}$:
\[
g = \num{6.67e-11} \times \num{5.97e24} / (\num{3.84e8})^2
\approx \qty{2.7e-3}{N/kg}.
\]
El \omterm{def:g11:electric-gravitational-fields:field}{campo} de la Tierra no se acaba nunca --- solo se adelgaza como
$1/d^2$; es este \omterm{def:g11:electric-gravitational-fields:field}{campo} el que mantiene a la Luna en su \omterm{def:g10:universal-gravitation:satellite}{órbita}.
\end{example}

\begin{remark}[Uniforme cerca de la superficie]\label{rem:g11:electric-gravitational-fields:uniformg}
A lo largo de un laboratorio, o incluso de una montaña, $d$ apenas
cambia frente a $R = \qty{6.37e6}{m}$, de modo que $\vect g$ es \omterm{def:g11:electric-gravitational-fields:capacitor}{uniforme}
con gran precisión: líneas verticales y \qty{9.81}{N/kg} constantes ---
el \omterm{def:g11:electric-gravitational-fields:field}{campo} de la Tierra se parece al del \omterm{def:g11:electric-gravitational-fields:capacitor}{condensador}, igual que la Tierra
redonda parece plana desde un jardín.
\end{remark}

\section{Una idea, dos fuerzas}

Los dos \omterm{def:g11:electric-gravitational-fields:field}{campos} de este capítulo son las mismas matemáticas con dos
disfraces:

\begin{center}
\begin{tabular}{lcc}
 & \textbf{eléctrico} & \textbf{gravitatorio} \\
\hline
fuente & carga $Q$ & masa $M$ \\
\omterm{def:g11:electric-gravitational-fields:field}{campo} & $\vect E = \vect F / q$ & $\vect g = \vect P / m$ \\
\omterm{def:g10:orders-of-magnitude:unit}{unidad} & \unit{N/C} = \unit{V/m} & \unit{N/kg} \\
fuente puntual & $E = k \abs{Q} / d^2$ & $g = G M / d^2$ \\
constante & $k = \qty{8.99e9}{N.m^2/C^2}$ & $G = \qty{6.67e-11}{N.m^2/kg^2}$ \\
\omterm{def:g10:inertia:force}{fuerza} sobre la prueba & $\vect F = q \vect E$ & $\vect P = m \vect g$ \\
\omterm{def:g11:electric-gravitational-fields:lines}{líneas de campo} & salen del $+$, entran en el $-$ & siempre hacia la masa \\
\end{tabular}
\end{center}

Cada fórmula de la izquierda se convierte en la de la derecha con el
diccionario $q \leftrightarrow m$, $\vect E \leftrightarrow \vect g$,
$k \leftrightarrow G$.

\begin{remark}[Dónde se rompe la analogía]\label{rem:g11:electric-gravitational-fields:noanalogue}
La carga viene en dos signos; la masa, en uno solo. Por eso los \omterm{def:g11:electric-gravitational-fields:efield}{campos
eléctricos} se pueden cancelar: las cargas reordenadas de una caja
metálica borran dentro cualquier \omterm{def:g11:electric-gravitational-fields:field}{campo} exterior (una \emph{jaula de
Faraday} --- por eso un ascensor mata la cobertura del móvil), y la
materia en bloque, con $+$ y $-$ equilibrados, es eléctricamente muda.
La gravedad no se puede apantallar ni neutralizar: cada \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} suma
su tirón --- y por eso la \omterm{def:g10:inertia:force}{fuerza} que es $\num{e39}$ veces más débil
(\cref{exo:g11:electric-gravitational-fields:10}) gobierna el universo a
gran escala.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:1}
Una carga de prueba $q = \qty{+2.0e-6}{C}$ situada en un punto $M$ nota
una \omterm{def:g10:inertia:force}{fuerza} de \qty{5.0e-3}{N} que apunta al norte. Da el \omterm{def:g11:electric-gravitational-fields:field}{campo} en $M$
(módulo y dirección) y después la \omterm{def:g10:inertia:force}{fuerza} sobre
$q' = \qty{-4.0e-6}{C}$ colocada en $M$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:2}
Una carga puntual $Q = \qty{+5.0}{nC}$ está en $O$.
\begin{enumerate}
  \item Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} a $d = \qty{30}{cm}$: módulo y dirección.
  \item ¿A qué distancia es el \omterm{def:g11:electric-gravitational-fields:field}{campo} la mitad de intenso?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:3}
Un \omterm{def:g11:electric-gravitational-fields:capacitor}{condensador} tiene $U = \qty{600}{V}$ entre placas separadas
\qty{3.0}{mm}. Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} y después la \omterm{def:g10:inertia:force}{fuerza} sobre un electrón
situado allí ($e = \qty{1.60e-19}{C}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:4}
En Marte, una sonda de \qty{250}{kg} pesa \qty{930}{N}. Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo}
marciano en la superficie, $g_{\text{Mars}}$, y después el \omterm{def:g10:universal-gravitation:weight}{peso} de un
astronauta de \qty{80}{kg} allí.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:5}
Verdadero o falso, con una razón cada vez: (a) dos \omterm{def:g11:electric-gravitational-fields:lines}{líneas de campo}
pueden cortarse donde el \omterm{def:g11:electric-gravitational-fields:field}{campo} es intenso; (b) las \omterm{def:g11:electric-gravitational-fields:lines}{líneas de campo} salen
de las cargas negativas y llegan a las positivas; (c) unas \omterm{def:g11:electric-gravitational-fields:lines}{líneas de
campo} apiñadas señalan un \omterm{def:g11:electric-gravitational-fields:field}{campo} intenso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:6}
La Estación Espacial Internacional orbita a una altitud de
\qty{420}{km} ($R = \qty{6.37e6}{m}$, $M = \qty{5.97e24}{kg}$).
\begin{enumerate}
  \item Calcula $g$ en la estación; ¿qué fracción del valor en la
        superficie es?
  \item Los astronautas flotan. Concilia este hecho con tu respuesta en
        una frase.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:7}
¿Qué módulo de \omterm{def:g11:electric-gravitational-fields:field}{campo} equilibraría el \omterm{def:g10:universal-gravitation:weight}{peso} de un electrón
($m_e = \qty{9.11e-31}{kg}$)? Compáralo con los \qty{2.0e4}{V/m} del
\cref{ex:g11:electric-gravitational-fields:beam}; concluye sobre el
papel de la gravedad en la física del electrón.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:8}
Una mota de polvo de masa \qty{2.0e-6}{kg} flota en reposo entre placas
horizontales separadas \qty{2.0}{cm} con $U = \qty{5.0}{kV}$ entre
ellas.
\begin{enumerate}
  \item Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} y después el módulo de la carga de la mota.
  \item La carga de la mota es positiva: ¿cuál es la placa positiva?
  \item ¿Cuántas \omterm{def:g11:fundamental-interactions:elementary}{cargas elementales} lleva la mota?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:9}
Las cargas $+Q$ y $+4Q$ están a \qty{30}{cm}. ¿En qué punto del segmento
que las une es nulo el \omterm{def:g11:electric-gravitational-fields:field}{campo} total? Explica primero por qué el punto
tiene que estar entre las cargas y más cerca de la menor.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:10}
En un \omterm{def:g11:fundamental-interactions:ladder}{átomo} de hidrógeno, el protón ($m_p = \qty{1.67e-27}{kg}$, carga
$+e$) y el electrón ($m_e = \qty{9.11e-31}{kg}$, carga $-e$) están a
$d = \qty{5.3e-11}{m}$. Calcula las \omterm{def:g10:inertia:force}{fuerzas} eléctrica y gravitatoria
entre ellos, y su cociente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:11}
En un punto $M$, una fuente por sí sola crearía un \omterm{def:g11:electric-gravitational-fields:field}{campo} de
\qty{300}{N/C} hacia el este; una segunda, de \qty{400}{N/C} hacia el
norte. Los \omterm{def:g11:electric-gravitational-fields:field}{campos} se suman como vectores: da el \omterm{def:g11:electric-gravitational-fields:field}{campo} total en $M$
(módulo y dirección) y después la \omterm{def:g10:inertia:force}{fuerza} sobre
$q = \qty{-2.0e-6}{C}$ colocada allí.
\end{exercise}

\section{Problema: la gota de Millikan}

\begin{problem}[{Problema de fin de semana --- pesar la carga del
electrón: una gota de aceite aparcada en el aire entre dos placas revela
que la carga viene en escalones indivisibles de $\num{1.60e-19}$
\omterm{def:g11:fundamental-interactions:charge}{culombios}}]
\label{pb:g11:electric-gravitational-fields:1}
En 1909, Robert Millikan pulverizó gotitas de aceite entre las placas
horizontales de un \omterm{def:g11:electric-gravitational-fields:capacitor}{condensador}. Ajustando la tensión hasta que una gota
elegida quedaba inmóvil --- con la \omterm{def:g10:inertia:force}{fuerza} eléctrica equilibrando
exactamente al \omterm{def:g10:universal-gravitation:weight}{peso} ---, podía pesar su carga. Nuestra gota tiene una
masa $m = \qty{4.90e-15}{kg}$ (medida a partir de su lenta caída con el
\omterm{def:g11:electric-gravitational-fields:field}{campo} apagado; la damos por conocida); las placas están separadas
$d = \qty{8.0}{mm}$, con la \emph{superior} positiva, y la carga de la
gota es negativa.

\medskip
\noindent\textbf{Parte I --- El escenario.}
\begin{enumerate}
  \item Describe el \omterm{def:g11:electric-gravitational-fields:field}{campo} entre las placas; ¿qué aporta que sea
        ``\omterm{def:g11:electric-gravitational-fields:capacitor}{uniforme}''?
  \item Para $U = \qty{800}{V}$, calcula $E$.
  \item ¿En qué sentido va la \omterm{def:g10:inertia:force}{fuerza} eléctrica sobre la gota? ¿Por qué
        la placa \emph{superior} tenía que ser la positiva?
  \item Calcula el \omterm{def:g10:universal-gravitation:weight}{peso} de la gota.
  \item Escribe la condición de equilibrio como una igualdad entre los
        módulos de dos \omterm{def:g10:inertia:force}{fuerzas}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El equilibrio.}
\begin{enumerate}[resume]
  \item A partir de ella, expresa $\abs{q}$ en función de $m$, $g$, $d$
        y $U$.
  \item La gota queda en reposo con $U = \qty{800}{V}$: calcula
        $\abs{q}$.
  \item Divide entre $e = \qty{1.60e-19}{C}$. ¿Cuántos electrones en
        exceso lleva la gota?
  \item Se sube la tensión hasta \qty{900}{V}. ¿En qué sentido apunta
        ahora la \omterm{def:g10:inertia:force}{fuerza} neta?
  \item Demuestra que la tensión de equilibrio vale
        $U = mgd/\abs{q}$: cuanto mayor es la carga, menor es la
        tensión.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Los escalones.}
Un destello de radiación puede arrancar electrones de la gota o
depositarlos en ella; tras cada cambio, Millikan vuelve a ajustar la
tensión para equilibrarla.
\begin{enumerate}[resume]
  \item Tras un destello, el equilibrio exige $U = \qty{1200}{V}$.
        Calcula el nuevo $\abs{q}$. ¿La gota ha ganado o ha perdido un
        electrón?
  \item Se encuentran equilibrios sucesivos con $U = 1200$, $800$,
        $600$, $480$ y \qty{400}{V}. Calcula las cinco cargas.
  \item Demuestra que las cinco son múltiplos enteros de una misma
        cantidad; da su valor.
  \item Entre dos equilibrios consecutivos, ¿cuánto cambia la carga?
        Interprétalo.
  \item Explica por qué esta gota nunca podrá quedar en reposo con
        $U = \qty{700}{V}$, por mucha paciencia que se tenga al
        ajustar.
  \item Millikan repitió esto con cientos de gotas: todas las cargas
        medidas eran múltiplos enteros del mismo valor. Enuncia la
        conclusión y nombra la magnitud medida.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Por qué no hay un Millikan gravitatorio.}
\begin{enumerate}[resume]
  \item En el diccionario de este capítulo, ¿qué hace los papeles de
        $q$, $E$ y $qE$ en el equilibrio de la Parte II?
  \item ¿Podría una masa colocada encima sostener la gota en su lugar?
        Calcula el \omterm{def:g11:electric-gravitational-fields:field}{campo} de una esfera de plomo de \qty{1000}{kg}
        situada \qty{1.0}{m} por encima de la gota, la \omterm{def:g10:inertia:force}{fuerza} sobre la
        gota y compárala con su \omterm{def:g10:universal-gravitation:weight}{peso}.
  \item El equilibrio necesita un tirón hacia arriba capaz de pelearse
        con la Tierra entera. ¿Por qué puede darlo una placa cargada y
        no puede darlo ninguna disposición de masas?
  \item La \emph{masa} de la gota también cambia a escalones (una
        \omterm{def:g11:fundamental-interactions:ladder}{molécula} de aceite: unos \qty{5e-25}{kg}). Compara los saltos
        relativos de $m$ y de $\abs{q}$ cuando cae un electrón y
        concluye: ¿por qué ve este equilibrio los \omterm{def:g11:fundamental-interactions:ladder}{átomos} de la
        electricidad y no las \omterm{def:g11:fundamental-interactions:ladder}{moléculas} del aceite? El remate: la carga
        está cuantizada, y su \omterm{def:g11:fundamental-interactions:ladder}{átomo} es $e = \qty{1.60e-19}{C}$.
\end{enumerate}
\end{problem}
