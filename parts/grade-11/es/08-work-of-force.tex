\chapter{El trabajo de una fuerza}\label{ch:g11:work-of-force}

Empuja un armario de un lado a otro de la habitación y habrás
trabajado; empuja contra una pared toda la tarde y, diga lo que diga tu
musculatura, la física cuenta cero. El \omterm{def:g11:work-of-force:work}{trabajo} es lo que una \omterm{def:g10:inertia:force}{fuerza}
entrega \emph{a lo largo de un movimiento} --- \omterm{def:g10:energy-conservation:energy}{energía} en tránsito
(\cref{ch:g10:energy-conservation}), pagada en \omterm{def:g11:work-of-force:work}{julios}. Este capítulo
pone precio a las \omterm{def:g10:inertia:force}{fuerzas} del \cref{ch:g11:forces-and-motion} ---
tirones en ángulo, \omterm{def:g10:universal-gravitation:weight}{peso}, \omterm{def:g10:inertia:inventory}{rozamiento} --- y cronometra el ritmo del pago:
la \omterm{def:g11:work-of-force:power}{potencia}.

\section{El trabajo de una fuerza constante}

\begin{definition}[Trabajo de una fuerza constante]\label{def:g11:work-of-force:work}
Una \omterm{def:g10:inertia:force}{fuerza} constante $\vect F$ aplicada a un objeto que se desplaza a lo
largo de un segmento recto de $A$ a $B$ realiza el
\emph{trabajo}\index{trabajo (de una fuerza)}
\[
W_{AB}(\vect F) = F \times AB \times \cos\theta ,
\]
donde $F$ es el módulo de la \omterm{def:g10:inertia:force}{fuerza} (en \unit{N}), $AB$ la longitud del
desplazamiento (en \unit{m}) y $\theta$ el ángulo entre $\vect F$ y
$\vect{AB}$. El trabajo se mide en \emph{julios}\index{julio}
(\unit{J}): un julio es el trabajo de una \omterm{def:g10:inertia:force}{fuerza} de un \omterm{def:g10:inertia:force}{newton} a lo largo
de un \omterm{def:g10:orders-of-magnitude:unit}{metro} en su propia dirección.
\end{definition}

\begin{remark}\label{rem:g11:work-of-force:scal}
$F \times AB \times \cos\theta$ es exactamente el producto escalar
$\vect F \cdot \vect{AB}$ del volumen de matemáticas de este año. Solo
la componente $F\cos\theta$ \emph{a lo largo} del movimiento se gana
algo.
\end{remark}

\begin{example}[Maleta con correa]\label{ex:g11:work-of-force:suitcase}
Un viajero arrastra una maleta \qty{300}{m} por una terminal, tirando de
la correa con $F = \qty{40}{N}$ a $\theta = \qty{50}{\degree}$ del
suelo: $W = 40 \times 300 \times \cos\qty{50}{\degree} \approx
\qty{7.7}{kJ}$. Tirando en horizontal ($\theta = 0$) esa misma \omterm{def:g10:inertia:force}{fuerza}
entregaría \qty{12}{kJ}: un tercio del tirón no levanta nada.
\end{example}

\begin{definition}[Trabajo motor, resistente y nulo]\label{def:g11:work-of-force:sign}
El signo de $\cos\theta$ clasifica toda \omterm{def:g10:inertia:force}{fuerza} en tres regímenes:
\begin{itemize}
  \item $\theta < \qty{90}{\degree}$: $W > 0$, el \omterm{def:g11:work-of-force:work}{trabajo} es
        \emph{motor}\index{trabajo motor} --- la \omterm{def:g10:inertia:force}{fuerza} alimenta de
        \omterm{def:g10:energy-conservation:energy}{energía} el movimiento;
  \item $\theta = \qty{90}{\degree}$: $W = 0$, la \omterm{def:g10:inertia:force}{fuerza} no trabaja en
        absoluto, por grande que sea;
  \item $\theta > \qty{90}{\degree}$: $W < 0$, el \omterm{def:g11:work-of-force:work}{trabajo} es
        \emph{resistente}\index{trabajo resistente} --- la \omterm{def:g10:inertia:force}{fuerza} drena
        \omterm{def:g10:energy-conservation:energy}{energía}. El caso puro es un freno a $\qty{180}{\degree}$:
        \qty{800}{N} a lo largo de \qty{45}{m} le quitan a un coche
        $800 \times 45 \times (-1) = \qty{-36}{kJ}$, que van a parar a
        unos discos calientes.
\end{itemize}
\end{definition}

\section{El trabajo del peso}

\begin{proposition}[Trabajo del peso]\label{prop:g11:work-of-force:weight}
Cuando un objeto de masa $m$ va de $A$ (altitud $z_A$) a $B$ (altitud
$z_B$) por \emph{cualquier} camino formado por segmentos rectos, su \omterm{def:g10:universal-gravitation:weight}{peso}
$\vect P$ realiza el \omterm{def:g11:work-of-force:work}{trabajo}
\[
W_{AB}(\vect P) = mg\,(z_A - z_B) = \pm\, mgh ,
\]
donde $h = \abs{z_A - z_B}$ es el desnivel: $+mgh$ al bajar y $-mgh$ al
subir --- solo cuenta el desnivel, no la ruta.
\end{proposition}

\begin{proof}
Sobre un segmento recto $AB$, el \omterm{def:g10:universal-gravitation:weight}{peso} (de módulo $mg$, dirigido
verticalmente hacia abajo) forma un ángulo $\theta$ con $\vect{AB}$, y
$AB\cos\theta$ es precisamente la proyección del desplazamiento sobre la
vertical descendente: $z_A - z_B$. Por tanto
$W = mg \times AB\cos\theta = mg\,(z_A - z_B)$. A lo largo de una
cadena de segmentos los \omterm{def:g11:work-of-force:work}{trabajos} se suman y los desniveles se
telescopan: $mg(z_A - z_C) + mg(z_C - z_B) = mg(z_A - z_B)$.
\end{proof}

\begin{proposition}[Cualquier camino]\label{prop:g11:work-of-force:anypath}
La misma fórmula vale a lo largo de cualquier camino curvo de $A$ a $B$.
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:work-of-force:anypath}
Es verosímil --- una curva se parece tanto como se quiera a una línea
quebrada de muchos segmentos cortos ---, pero hacer honrado ese ``tanto
como se quiera'' es cálculo infinitesimal, y se hace en el
\cref{ch:g12:work-and-energy}.
\end{remark}

\begin{example}[Excursionista]\label{ex:g11:work-of-force:hiker}
Un excursionista de \qty{78}{kg} (mochila incluida) sube \qty{450}{m},
por el sendero que sea:
$W(\vect P) = -78 \times 9.81 \times 450 \approx \qty{-344}{kJ}$, y
exactamente $+\qty{344}{kJ}$ a la bajada, con los rodeos que sean.
\end{example}

\section{El trabajo del rozamiento}

\begin{proposition}[Trabajo del rozamiento de deslizamiento]\label{prop:g11:work-of-force:friction}
Una \omterm{def:g10:inertia:force}{fuerza} de \omterm{def:g10:inertia:inventory}{rozamiento} de módulo constante $f$, opuesta al movimiento
en todo instante, realiza el \omterm{def:g11:work-of-force:work}{trabajo} $W = -fL$ a lo largo de un camino
de longitud total $L$: siempre \omterm{def:g11:work-of-force:sign}{resistente} y proporcional a la
\emph{longitud del camino recorrido} --- no al desplazamiento.
\end{proposition}

\begin{proof}
En cada trocito recto la \omterm{def:g10:inertia:force}{fuerza} forma $\qty{180}{\degree}$ con el
movimiento y aporta $-f \times (\text{su longitud})$; los trocitos suman
$-fL$. Un camino curvo se trocea igual, con el paso al límite admitido
como en el caso del \omterm{def:g10:universal-gravitation:weight}{peso} (\cref{rem:g11:work-of-force:anypath}).
\end{proof}

\begin{remark}[Rozamiento frente a peso: la gran divisoria]\label{rem:g11:work-of-force:divide}
El \omterm{def:g11:work-of-force:work}{trabajo} del \omterm{def:g10:universal-gravitation:weight}{peso} se olvida del camino; el del \omterm{def:g10:inertia:inventory}{rozamiento} es un peaje
por \omterm{def:g10:orders-of-magnitude:unit}{metro}. En un viaje de ida y vuelta el \omterm{def:g10:universal-gravitation:weight}{peso} da cero neto --- lo que
cobra en la subida lo devuelve en la bajada ---, mientras que el
\omterm{def:g10:inertia:inventory}{rozamiento} cobra $-fL$ en cada tramo y no devuelve nada: el \omterm{def:g11:work-of-force:work}{trabajo} del
\omterm{def:g10:universal-gravitation:weight}{peso} se puede almacenar y recuperar (\omterm{def:g10:energy-conservation:energy}{energía} potencial,
\cref{ch:g11:mechanical-energy}) y el del \omterm{def:g10:inertia:inventory}{rozamiento} se pierde como
calor.
\end{remark}

\begin{example}[Ida y vuelta]\label{ex:g11:work-of-force:roundtrip}
Una caja desliza \qty{5.0}{m} por un suelo y \qty{5.0}{m} de vuelta,
contra $f = \qty{60}{N}$: el \omterm{def:g10:inertia:inventory}{rozamiento} realiza
$-60 \times 10 = \qty{-600}{J}$ aunque la caja acabe donde empezó. El
\omterm{def:g10:universal-gravitation:weight}{peso} y la \omterm{def:g10:inertia:force}{fuerza} normal, perpendiculares al movimiento: cero en todo
momento.
\end{example}

\begin{remark}[¿Y un muelle?]\label{rem:g11:work-of-force:spring}
Un muelle tira más cuanto más estirado está: su \omterm{def:g10:inertia:force}{fuerza} no es constante y
la \cref{def:g11:work-of-force:work} no se le aplica. Su \omterm{def:g11:work-of-force:work}{trabajo} exige
cortar el estiramiento en pasos infinitesimales y sumar --- cálculo
infinitesimal, servido en el \cref{ch:g12:work-and-energy}.
\end{remark}

\section{La potencia}

\begin{definition}[Potencia]\label{def:g11:work-of-force:power}
La \emph{potencia}\index{potencia!mecánica} de una \omterm{def:g10:inertia:force}{fuerza} es el ritmo al
que trabaja, $P = W/\Delta t$, en \emph{vatios}\index{vatio}
($\unit{W} = \unit{J/s}$) --- el mismo vatio del
\cref{ch:g11:circuits-and-power}. Una \omterm{def:g10:orders-of-magnitude:unit}{unidad} anterior al SI sobrevive en
las placas de los \omterm{def:g11:work-of-force:sign}{motores}: el \emph{caballo de vapor}\index{caballo de
vapor}, $\qty{1}{CV} = \qty{736}{W}$.
\end{definition}

\begin{proposition}[Potencia a velocidad constante]\label{prop:g11:work-of-force:pfv}
Si el objeto se mueve con rapidez constante $v$, una \omterm{def:g10:inertia:force}{fuerza} constante
$\vect F$ que forme un ángulo $\theta$ con el movimiento entrega la
\omterm{def:g11:work-of-force:power}{potencia} $P = F v \cos\theta$ --- en particular, $P = Fv$ para una
\omterm{def:g10:inertia:force}{fuerza} en la dirección del movimiento.
\end{proposition}

\begin{proof}
En un tiempo $\Delta t$ el objeto recorre $d = v\,\Delta t$, de modo que
$W = F \times v\,\Delta t \times \cos\theta$; basta dividir entre
$\Delta t$.
\end{proof}

\begin{example}[Grúa]\label{ex:g11:work-of-force:crane}
Una grúa iza un palé de \qty{600}{kg} a \qty{0.90}{m/s} constantes: la
\omterm{def:g10:inertia:force}{fuerza} del cable iguala al \omterm{def:g10:universal-gravitation:weight}{peso}, $mg = \qty{5.89e3}{N}$, luego
$P = Fv = 5890 \times 0.90 \approx \qty{5.3}{kW} \approx \qty{7.2}{CV}$
--- un tiro de siete caballos dentro de una caja de acero.
\end{example}

\begin{method}[Subir con menos fuerza: el zigzag]\label{met:g11:work-of-force:zigzag}
Para subir una carga una altura $h$ con una \omterm{def:g10:inertia:force}{fuerza} limitada:
\begin{enumerate}
  \item La factura está fijada: por la
        \cref{prop:g11:work-of-force:weight}, cualquier ruta hasta
        arriba cuesta el \omterm{def:g11:work-of-force:work}{trabajo} $mgh$.
  \item Estira el camino: sobre una rampa de longitud $L$, la \omterm{def:g10:inertia:force}{fuerza}
        necesaria (a velocidad constante y sin contar el \omterm{def:g10:inertia:inventory}{rozamiento}) es
        $F = mgh/L$ --- el doble de longitud, la mitad de \omterm{def:g10:inertia:force}{fuerza}.
  \item Paga el recargo: el \omterm{def:g10:inertia:inventory}{rozamiento} cobra por \omterm{def:g10:orders-of-magnitude:unit}{metro}
        (\cref{prop:g11:work-of-force:friction}) y añade $fL$.
\end{enumerate}
Las carreteras de montaña zigzaguean por el paso 2: la misma altura,
menos \omterm{def:g10:inertia:force}{fuerza}, el mismo \omterm{def:g11:work-of-force:work}{trabajo}.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:1}
Una \omterm{def:g10:inertia:force}{fuerza} constante $F = \qty{40}{N}$ actúa a lo largo de un
desplazamiento recto de \qty{50}{m}: calcula su \omterm{def:g11:work-of-force:work}{trabajo} para
$\theta = \qty{0}{\degree}$, $\qty{60}{\degree}$, $\qty{90}{\degree}$ y
$\qty{120}{\degree}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:2}
Una caja de herramientas de \qty{12}{kg} se baja \qty{2.5}{m} desde una
furgoneta y, más tarde, se vuelve a subir. ¿\omterm{def:g11:work-of-force:work}{Trabajo} de su \omterm{def:g10:universal-gravitation:weight}{peso} en cada
caso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:3}
¿\omterm{def:g11:work-of-force:sign}{Motor}, \omterm{def:g11:work-of-force:sign}{resistente} o nulo? (a) el \omterm{def:g10:universal-gravitation:weight}{peso} de una manzana que cae; (b) el
\omterm{def:g10:universal-gravitation:weight}{peso} de una pelota que sube; (c) la \omterm{def:g10:inertia:force}{fuerza} normal sobre una caja que
desliza; (d) el \omterm{def:g10:inertia:inventory}{rozamiento} sobre una bicicleta que frena; (e) la tensión
de la cuerda sobre la masa de un péndulo que oscila.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:4}
Una grúa sube \qty{600}{kg} una altura de \qty{25}{m} a velocidad
constante. ¿\omterm{def:g11:work-of-force:work}{Trabajo} de la \omterm{def:g10:inertia:force}{fuerza} que iza y del \omterm{def:g10:universal-gravitation:weight}{peso}? ¿Cuánto suman, y
por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:5}
El \omterm{def:g11:work-of-force:sign}{motor} de una escalera mecánica realiza \qty{90}{kJ} de \omterm{def:g11:work-of-force:work}{trabajo} cada
minuto. ¿Su \omterm{def:g11:work-of-force:power}{potencia} en \omterm{def:g11:work-of-force:power}{vatios}? ¿Y en \omterm{def:g11:work-of-force:power}{caballos de vapor}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:6}
Un trineo es remolcado \qty{200}{m} por una cuerda de tensión
\qty{120}{N} a $\qty{30}{\degree}$ por encima de la nieve; el \omterm{def:g10:inertia:inventory}{rozamiento}
vale \qty{45}{N}. ¿\omterm{def:g11:work-of-force:work}{Trabajo} de cada una de las cuatro \omterm{def:g10:inertia:force}{fuerzas} (cuerda,
\omterm{def:g10:inertia:inventory}{rozamiento}, \omterm{def:g10:universal-gravitation:weight}{peso} y normal) y total?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:7}
Un ciclista rueda a \qty{25}{km/h} constantes contra una \omterm{def:g11:circuits-and-power:resistance}{resistencia}
total de \qty{18}{N}. ¿\omterm{def:g11:work-of-force:power}{Potencia} entregada? ¿\omterm{def:g11:work-of-force:work}{Trabajo} realizado en una
hora?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:8}
Un esquiador de \qty{65}{kg} desciende \qty{120}{m} de desnivel, por una
pendiente fuerte de \qty{800}{m} o por una pista suave de \qty{1500}{m};
el \omterm{def:g10:inertia:inventory}{rozamiento} sobre los esquís vale \qty{30}{N} en las dos. ¿\omterm{def:g11:work-of-force:work}{Trabajo} del
\omterm{def:g10:universal-gravitation:weight}{peso} y del \omterm{def:g10:inertia:inventory}{rozamiento} en cada ruta? ¿Qué \omterm{def:g11:work-of-force:work}{trabajo} depende de la ruta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:9}
Una caja de \qty{20}{kg} se empuja \qty{4.0}{m} rampa arriba, con una
inclinación de $\qty{25}{\degree}$, a velocidad constante y contra
\qty{30}{N} de \omterm{def:g10:inertia:inventory}{rozamiento}. Calcula los \omterm{def:g11:work-of-force:work}{trabajos} del \omterm{def:g10:universal-gravitation:weight}{peso} y del
\omterm{def:g10:inertia:inventory}{rozamiento} y, a partir de la velocidad constante, el \omterm{def:g11:work-of-force:work}{trabajo} y el módulo
del empujón, paralelo a la rampa.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:10}

\section{Problema: la jornada del mudancero}

\begin{problem}[{Problema de fin de semana --- la física paga por
julio: un día de arrastrar, izar y tomar curvas de herradura, que
termina con el humillante descubrimiento de lo que opina un hervidor del
trabajo honrado}]
\label{pb:g11:work-of-force:1}
Una cuadrilla de mudanzas vacía un almacén, sube la carga a un piso de
un tercero y después conduce la furgoneta hasta un pueblo en lo alto de
una colina. Cada tarea se tasa en \omterm{def:g11:work-of-force:work}{julios}; el hervidor de la cocina
auditará la jornada. Cajas: \qty{30}{kg}.

\medskip
\noindent\textbf{Parte I --- Por el suelo del almacén.}
Una caja se empuja con una \omterm{def:g10:inertia:force}{fuerza} horizontal $F = \qty{110}{N}$ a lo
largo de $d = \qty{8.0}{m}$; el \omterm{def:g10:inertia:inventory}{rozamiento} de deslizamiento vale
$f = \qty{95}{N}$.
\begin{enumerate}
  \item Calcula el \omterm{def:g11:work-of-force:work}{trabajo} del empujón.
  \item Calcula el \omterm{def:g11:work-of-force:work}{trabajo} del \omterm{def:g10:inertia:inventory}{rozamiento}, el del \omterm{def:g10:universal-gravitation:weight}{peso} y el de la \omterm{def:g10:inertia:force}{fuerza}
        normal (justifica los ceros).
  \item ¿\omterm{def:g11:work-of-force:work}{Trabajo} total sobre la caja y significado de su signo (la caja
        partía del reposo)? ¿Qué empujón mantendría constante la
        velocidad (\cref{ch:g11:forces-and-motion})?
  \item A la esquina opuesta se puede llegar por la diagonal de
        \qty{10.0}{m} o pegado a dos paredes (\qty{6.0}{m} y después
        \qty{8.0}{m}). ¿\omterm{def:g11:work-of-force:work}{Trabajo} del \omterm{def:g10:inertia:inventory}{rozamiento} en cada ruta?
  \item En una ida y vuelta por la diagonal, calcula el \omterm{def:g11:work-of-force:work}{trabajo} del
        \omterm{def:g10:inertia:inventory}{rozamiento} y el del \omterm{def:g10:universal-gravitation:weight}{peso}. ¿Qué \omterm{def:g10:inertia:force}{fuerza} devuelve el dinero?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Hasta el tercer piso ($h = \qty{9.0}{m}$).}
\begin{enumerate}[resume]
  \item Calcula el \omterm{def:g11:work-of-force:work}{trabajo} del \omterm{def:g10:universal-gravitation:weight}{peso} de una caja subida al piso. ¿Qué
        \omterm{def:g11:work-of-force:work}{trabajo} tiene que aportar como mínimo la cuadrilla?
  \item Si la sube por las escaleras un mudancero de \qty{70}{kg}, la
        gravedad cobra por el mudancero \emph{y} por la caja: ¿\omterm{def:g11:work-of-force:work}{trabajo}
        total contra la gravedad y fracción realmente invertida en la
        caja?
  \item Una rampa para carretilla mide \qty{30}{m} para el mismo
        desnivel de \qty{9.0}{m}, con \qty{40}{N} de \omterm{def:g10:inertia:inventory}{rozamiento} de
        rodadura. A velocidad constante, calcula el empujón necesario a
        lo largo de la rampa y su \omterm{def:g11:work-of-force:work}{trabajo}.
  \item Compara la rampa con una izada vertical directa: \omterm{def:g10:inertia:force}{fuerza}
        necesaria y \omterm{def:g11:work-of-force:work}{trabajo} realizado. ¿Qué compra la rampa, y a qué
        precio?
  \item Un aparejo de cuerda y polea sube la caja a \qty{0.50}{m/s}.
        ¿Duración y \omterm{def:g11:work-of-force:power}{potencia} mecánica?
  \item Recupera esa \omterm{def:g11:work-of-force:power}{potencia} como $P = Fv$. El \omterm{def:g11:work-of-force:sign}{motor} del aparejo tiene
        un \omterm{def:g10:energy-conservation:efficiency}{rendimiento} del $60\,\%$: ¿\omterm{prop:g11:circuits-and-power:power}{potencia eléctrica} consumida?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La carretera en zigzag.}
La furgoneta cargada (\qty{3500}{kg}) tiene que llegar a un pueblo
\qty{240}{m} por encima del valle: por una pista recta de \qty{1.0}{km}
al $24\,\%$ de pendiente, o por una carretera asfaltada de \qty{8.0}{km}
al $3.0\,\%$ (pendiente $= \sin\alpha$).
\begin{enumerate}[resume]
  \item Comprueba que las dos rutas suben los mismos \qty{240}{m}.
  \item A velocidad constante y despreciando el \omterm{def:g10:inertia:inventory}{rozamiento}, calcula la
        \omterm{def:g10:inertia:force}{fuerza} motriz necesaria en cada ruta.
  \item ¿\omterm{def:g11:work-of-force:work}{Trabajo} de la \omterm{def:g10:inertia:force}{fuerza} motriz en cada ruta? Compáralo con $mgh$.
  \item La \omterm{def:g11:circuits-and-power:resistance}{resistencia} a la rodadura vale \qty{400}{N} en las dos:
        calcula el \omterm{def:g11:work-of-force:work}{trabajo} adicional de cada ruta. ¿Qué cuesta la
        comodidad del zigzag?
  \item A \qty{36}{km/h}, calcula la \omterm{def:g11:work-of-force:power}{potencia} del \omterm{def:g11:work-of-force:sign}{motor} necesaria en
        cada ruta (en \unit{kW} y en \unit{CV}). ¿Por qué zigzaguean las
        carreteras?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El libro de cuentas del día.}
\begin{enumerate}[resume]
  \item La cuadrilla arrastra $60$ cajas por el suelo (con el empujón de
        la pregunta 1) e iza las $60$ los \qty{9.0}{m} (pregunta 6).
        ¿\omterm{def:g11:work-of-force:work}{Trabajo} mecánico total entregado a las cajas?
  \item Añade el propio cuerpo del que sube por la escalera: $20$
        subidas con \qty{70}{kg} (las devoluciones de la bajada van a
        unas rodillas calientes, no al libro de cuentas). ¿Total del
        día?
  \item La cuadrilla trabajó $8.0$ horas: ¿\omterm{def:g11:work-of-force:power}{potencia} mecánica media?
        ¿Cuánto tardaría un hervidor de \qty{2.0}{kW} en gastar esa
        misma \omterm{def:g10:energy-conservation:energy}{energía}?
  \item Final: di en dos frases qué paga el \omterm{def:g11:work-of-force:work}{julio} --- y qué no (la pared
        empujada toda la tarde, la caja sostenida inmóvil) --- y por qué
        los mudanceros tienen cabrestantes, rampas y carreteras en
        zigzag en lugar de músculos más grandes.
\end{enumerate}
\end{problem}
