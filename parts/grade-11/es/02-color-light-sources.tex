\chapter{El color y las fuentes de luz}\label{ch:g11:color-light-sources}

Pregúntale a un físico de qué color es un tomate y te devolverá otra
pregunta: ¿bajo qué luz? El tomate solo \omterm{def:g11:color-light-sources:objectcolor}{difunde} lo que se le da, la
lámpara solo da lo que contiene su \omterm{def:g10:light-spectra:white}{espectro}, y el ojo comprime lo que
llegue en tres números. El color es una negociación entre la fuente, el
objeto y el ojo --- y este capítulo interroga por turnos a las tres
partes.

\section{Ver con tres números}

\begin{definition}[Conos y visión tricromática]\label{def:g11:color-light-sources:cones}
La \omterm{def:g11:lenses-and-eye:eye}{retina} (\cref{ch:g11:lenses-and-eye}) lleva dos familias de células
sensibles a la luz: los \emph{bastones}\index{bastón}, que responden a
la luz débil pero ignoran la \omterm{def:g10:light-spectra:wavelength}{longitud de onda} (visión nocturna, en
grises), y los \emph{conos}\index{cono}, de tres tipos con sensibilidad
máxima cerca de $\qty{420}{nm}$ (azul), $\qty{530}{nm}$ (verde) y
$\qty{560}{nm}$ (rojo). La visión humana del color es
\emph{tricromática}\index{visión tricromática}: sea cual sea el
\omterm{def:g10:light-spectra:white}{espectro}, el cerebro solo recibe tres respuestas de los conos, y el
color es lo que hace con ese trío.
\end{definition}

\begin{definition}[El color como percepción]\label{def:g11:color-light-sources:perception}
El color es, por tanto, una \emph{percepción}, no una propiedad de la
luz sola: un \omterm{def:g10:light-spectra:white}{espectro} especifica una intensidad en cada \omterm{def:g10:light-spectra:wavelength}{longitud de
onda}, y el ojo se queda con tres números. Dos luces de \omterm{def:g10:light-spectra:white}{espectros}
distintos pero con las mismas tres respuestas de los \omterm{def:g11:color-light-sources:cones}{conos} se ven
estrictamente iguales: son \emph{metámeros}\index{metámero}.
\end{definition}

\begin{example}[Dos amarillos]\label{ex:g11:color-light-sources:yellows}
La línea de $\qty{589}{nm}$ de una lámpara de sodio excita casi por
igual los \omterm{def:g11:color-light-sources:cones}{conos} rojo y verde: el cerebro dice \emph{amarillo}. Una
pantalla que encienda sus subpíxeles rojo ($\qty{610}{nm}$) y verde
($\qty{540}{nm}$) da ese mismo amarillo sin nada de luz cerca de
$\qty{589}{nm}$. Un prisma separa las dos al instante; el ojo, nunca.
\end{example}

\section{Sumar luces}

\begin{definition}[Síntesis aditiva]\label{def:g11:color-light-sources:additive}
Superponer haces de luz sobre una pantalla blanca suma sus \omterm{def:g10:light-spectra:white}{espectros}:
esto es la \emph{síntesis aditiva}\index{síntesis aditiva}. Tres luces
bien elegidas, roja, verde y azul --- los \emph{colores primarios}%
\index{colores primarios} aditivos --- se mezclan hasta dar casi
cualquier color perceptible: R $+$ G $=$ amarillo, R $+$ B $=$ magenta,
G $+$ B $=$ cian y R $+$ G $+$ B $=$ blanco.
\end{definition}

\begin{proposition}[Colores complementarios]\label{prop:g11:color-light-sources:complementary}
Dos luces cuya suma se percibe como blanca son
\emph{complementarias}\index{colores complementarios}. Las tres parejas
amarillo/azul, cian/rojo y magenta/verde son \omterm{prop:g11:color-light-sources:complementary}{complementarias}.
\end{proposition}

\begin{proof}
El amarillo es R $+$ G; añadir azul completa el trío hasta el blanco.
Las otras dos parejas se obtienen con el mismo razonamiento, de primario
en primario.
\end{proof}

\begin{example}[Las pantallas]\label{ex:g11:color-light-sources:screen}
Cada píxel de la pantalla de un móvil son tres subpíxeles --- rojo,
verde y azul, de menos de $\qty{0.1}{mm}$ de ancho --- demasiado juntos
para que el ojo los separe, así que sus luces se suman en la \omterm{def:g11:lenses-and-eye:eye}{retina}: el
amarillo es R y G encendidos, el naranja es R a plena \omterm{def:g10:energy-conservation:power}{potencia} y G a
media, y el blanco son los tres. Una \omterm{def:g11:lenses-and-eye:magnifier}{lupa} sobre una zona blanca desvela
el truco: no hay blanco por ninguna parte, solo R, G y B.
\end{example}

\section{Restar a la luz blanca}

\begin{definition}[Síntesis sustractiva]\label{def:g11:color-light-sources:subtractive}
Un \emph{filtro}\index{filtro} (un vidrio coloreado, una gelatina, una
capa de tinta o de pintura) transmite parte del \omterm{def:g10:light-spectra:white}{espectro} que recibe y
absorbe el resto. Producir colores \emph{quitando} bandas a la \omterm{def:g10:light-spectra:white}{luz
blanca} es la \emph{síntesis sustractiva}\index{síntesis sustractiva}.
Sus primarios, complementarios de los aditivos, restan cada uno una
banda: cian $=$ blanco $-$ R, magenta $=$ blanco $-$ G, amarillo $=$
blanco $-$ B. Los filtros apilados restan a la vez: cian $+$ amarillo
deja verde, cian $+$ magenta deja azul, magenta $+$ amarillo deja rojo y
los tres juntos, negro. La imprenta y la pintura funcionan así, con
tintas C, M y Y.
\end{definition}

\begin{method}[Contabilidad de espectros]\label{met:g11:color-light-sources:bookkeeping}
Para predecir un color a lo largo de cualquier cadena de fuentes,
\omterm{def:g11:color-light-sources:subtractive}{filtros} y objetos, hay que seguir las tres bandas primarias: se escribe
la luz de la fuente por su contenido en R, G y B; en cada \omterm{def:g11:color-light-sources:subtractive}{filtro} o
superficie se tachan las bandas absorbidas; y se nombra el color de lo
que sobrevive (si no queda nada, negro).
\end{method}

\begin{example}[Una cadena de filtros]\label{ex:g11:color-light-sources:chain}
\omterm{def:g10:light-spectra:white}{Luz blanca} $(R, G, B)$ atraviesa un \omterm{def:g11:color-light-sources:subtractive}{filtro} amarillo: se absorbe la banda
azul y queda $(R, G)$ --- amarillo. Un \omterm{def:g11:color-light-sources:subtractive}{filtro} cian absorbe después la
banda roja y deja $(G)$: verde. Intercambiar los \omterm{def:g11:color-light-sources:subtractive}{filtros} no cambia nada:
cada uno absorbe su propia banda sin importar el orden.
\end{example}

\section{El color de un objeto}

\begin{definition}[Difusión y color de un objeto]\label{def:g11:color-light-sources:objectcolor}
Un objeto opaco absorbe parte de la luz que recibe y
\emph{difunde}\index{difusión} el resto --- la reemite en todas
direcciones, entre otras hacia nuestros ojos. El \emph{color de un
objeto}\index{color de un objeto} es el color de la luz que difunde
\emph{bajo la iluminación dada}: el objeto pone sus costumbres de
absorción y la fuente, la materia prima. Cambia la lámpara y cambia el
color.
\end{definition}

\begin{example}[La camiseta roja]\label{ex:g11:color-light-sources:redshirt}
Una camiseta se ve roja a la luz del día: absorbe las bandas verde y
azul y \omterm{def:g11:color-light-sources:objectcolor}{difunde} la roja. Bajo una luz verde pura no hay banda roja que
difundir: la camiseta lo absorbe todo y se ve negra. Bajo luz magenta
$(R, B)$ \omterm{def:g11:color-light-sources:objectcolor}{difunde} solo la banda roja: sigue siendo roja. Un objeto blanco
\omterm{def:g11:color-light-sources:objectcolor}{difunde} todas las bandas que recibe (bajo luz verde se ve verde); uno
negro las absorbe todas bajo cualquier luz --- razón por la cual la ropa
negra también se calienta al sol.
\end{example}

\section{Fuentes de luz y temperatura de color}

\begin{definition}[Fuentes incandescentes]\label{def:g11:color-light-sources:incandescent}
Una fuente que brilla porque está caliente --- una llama, un filamento,
una estrella --- luce por \emph{incandescencia}\index{incandescencia}.
Su \omterm{def:g10:light-spectra:white}{espectro} es el \omterm{def:g10:light-spectra:white}{espectro} térmico continuo del
\cref{ch:g10:light-spectra}, y su máximo obedece la ley de Wien
(\cref{prop:g10:light-spectra:wien}),
\[
\lambda_{\max} \times T = \qty{2.90e-3}{m.K},
\]
que será nuestra herramienta cuantitativa de aquí en adelante: máximo
medido, temperatura conocida.
\end{definition}

\begin{example}[Una lámpara pésima]\label{ex:g11:color-light-sources:bulb}
Un filamento de wolframio a $T = \qty{2700}{K}$ tiene su máximo en
$\lambda_{\max} = \qty{2.90e-3}{m.K}/\qty{2700}{K} = \qty{1.07e-6}{m}
= \qty{1070}{nm}$, muy dentro del \omterm{def:g10:light-spectra:wavelength}{infrarrojo}: casi toda la \omterm{def:g10:energy-conservation:power}{potencia}
eléctrica se va como radiación invisible y solo un $5\,\%$ como luz ---
la bombilla incandescente era sobre todo una estufa.
\end{example}

\begin{definition}[Lámparas espectrales y láseres]\label{def:g11:color-light-sources:lamps}
Una \emph{lámpara espectral}\index{lámpara espectral} --- un gas a baja
\omterm{def:g10:pressure:pressure}{presión} excitado por una descarga: sodio, mercurio, neón --- emite un
\omterm{def:g10:light-spectra:emission}{espectro de líneas} (\cref{ch:g10:light-spectra}): unas pocas \omterm{def:g10:light-spectra:wavelength}{longitudes
de onda} aisladas y nada entre ellas. Un \emph{láser}\index{láser} va más
lejos: una sola \omterm{def:g10:light-spectra:wavelength}{longitud de onda}, \omterm{def:g10:light-spectra:white}{luz monocromática}, en un haz estrecho.
\end{definition}

\begin{definition}[Los ledes]\label{def:g11:color-light-sources:led}
Un \emph{diodo emisor de luz}\index{diodo emisor de luz} (led) emite una
banda \emph{cuasimonocromática}\index{cuasimonocromática} de unas
decenas de nanómetros de anchura: lo bastante estrecha para verse de un
color definido, pero mucho más ancha que la línea de un \omterm{def:g11:color-light-sources:lamps}{láser}. Un led
blanco es un led azul (con máximo cerca de $\qty{450}{nm}$) recubierto
de un \emph{fósforo}\index{fósforo}, un material que absorbe parte del
azul y lo reemite como una banda amarilla ancha: como el azul y el
amarillo son complementarios
(\cref{prop:g11:color-light-sources:complementary}), la mezcla se lee
como blanco.
\end{definition}

\begin{definition}[Temperatura de color]\label{def:g11:color-light-sources:colortemp}
La \emph{temperatura de color}\index{temperatura de color} de una fuente
más o menos blanca es la temperatura del cuerpo incandescente cuyo tono
iguala el de su luz: vela $\qty{1800}{K}$, bombilla incandescente
$\qty{2700}{K}$, halógena $\qty{3400}{K}$, sol de mediodía
$\qty{5500}{K}$, luz de día con cielo cubierto $\qty{6500}{K}$. Es una
etiqueta de color, no la lectura de un termómetro --- un led de
$\qty{6500}{K}$ funciona a temperatura ambiente --- y el vocabulario va
al revés: una temperatura de color baja se lee como un naranja
``cálido'' y una alta, como un azulado ``frío''.
\end{definition}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:1}
Da el color percibido cuando una pantalla blanca recibe a la vez: (a)
luz roja y verde; (b) luz verde y azul; (c) los tres primarios. ¿Qué luz
es complementaria del azul?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:2}
¿Qué subpíxeles (R, G, B) enciende una pantalla para mostrar amarillo,
magenta, blanco, negro y naranja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:3}
\omterm{def:g10:light-spectra:white}{Luz blanca} atraviesa un \omterm{def:g11:color-light-sources:subtractive}{filtro} magenta. ¿Qué banda se absorbe y qué
color emerge? Se añade después un \omterm{def:g11:color-light-sources:subtractive}{filtro} cian por detrás: ¿qué emerge
ahora?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:4}
Bajo \omterm{def:g10:light-spectra:white}{luz blanca}, una bandera muestra una franja amarilla. ¿Qué bandas
\omterm{def:g11:color-light-sources:objectcolor}{difunde} y cuáles absorbe? ¿Su color bajo (a) luz roja; (b) luz azul?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:5}
El \omterm{def:g10:light-spectra:continuous}{espectro continuo} de una estrella tiene su máximo en
$\lambda_{\max} = \qty{580}{nm}$. Calcula su temperatura superficial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:6}
La llama de una vela está a unos $\qty{1800}{K}$. (a) Calcula
$\lambda_{\max}$ e indica su dominio. (b) Y sin embargo la llama se ve
naranja: explícalo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:7}
Un ciclista lleva una chaqueta roja y unos guantes cian. Con el
\cref{met:g11:color-light-sources:bookkeeping}, da el color aparente de
cada prenda bajo (a) \omterm{def:g10:light-spectra:white}{luz blanca}; (b) luz roja; (c) luz verde.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:8}
Una calle antigua está iluminada con lámparas de sodio a baja \omterm{def:g10:pressure:pressure}{presión}
(una única línea en $\qty{589}{nm}$). (a) ¿Qué tipo de \omterm{def:g10:light-spectra:white}{espectro} es este
y qué colores puede tomar \emph{cualquier} objeto bajo él? (b) Un coche
se ve negro bajo esas lámparas y azul a la luz del día: explícalo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:9}
Un puntero \omterm{def:g11:color-light-sources:lamps}{láser} verde emite en $\qty{532}{nm}$. (a) Su haz atraviesa un
prisma: compara el resultado con el de un haz blanco y nombra la
propiedad que se ilustra. (b) Después encuentra un \omterm{def:g11:color-light-sources:subtractive}{filtro} rojo: ¿qué
sale?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:10}
Describe el \omterm{def:g10:light-spectra:white}{espectro} de un led blanco (diodo azul más \omterm{def:g11:color-light-sources:led}{fósforo}) y explica
por qué su luz se percibe blanca. ¿Por qué un objeto rojo intenso se ve
más apagado bajo un led blanco barato que a la luz del día?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:11}
Calcula $\lambda_{\max}$ para un cuerpo incandescente a (a) la
\omterm{def:g11:color-light-sources:colortemp}{temperatura de color} de una vela ($\qty{1800}{K}$); (b) la de un día con
cielo cubierto ($\qty{6500}{K}$). ¿Cuál de las dos luces se llama
``cálida''? Coméntalo.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:12}
Una lámpara de sodio ($\qty{589}{nm}$) y una pantalla que muestra
amarillo (subpíxeles en $\qty{610}{nm}$ y $\qty{540}{nm}$) se ven
exactamente del mismo color. (a) ¿Cómo se llaman dos luces así, y qué
demuestra su existencia sobre el color? (b) Propón dos experimentos
distintos que las distingan y predice el resultado de cada uno.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:13}
Un \omterm{def:g11:lenses-and-eye:focal}{foco} halógeno de teatro trabaja a $\qty{3200}{K}$. (a) Calcula
$\lambda_{\max}$: ¿es esta luz más cálida o más fría que la del día? (b)
Para imitar la luz del día se cubre el \omterm{def:g11:lenses-and-eye:focal}{foco} con una gelatina azul
pálido. ¿Qué quita la gelatina, y por qué se oscurece el escenario? (c)
Explica por qué ninguna gelatina podría, en cambio, \emph{reforzar} el
extremo azul del \omterm{def:g10:light-spectra:white}{espectro} del \omterm{def:g11:lenses-and-eye:focal}{foco}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:14}
Un carnicero ilumina el expositor de carne con ledes de \omterm{def:g11:color-light-sources:colortemp}{temperatura de
color} $\qty{2700}{K}$; la pescadería de al lado usa $\qty{6500}{K}$. (a)
Describe el tono de cada iluminación y su efecto sobre la carne roja y
sobre el pescado blanco. (b) ¿Ha cambiado alguno de los dos su
mercancía? Di exactamente dónde ocurre el cambio
(\cref{def:g11:color-light-sources:objectcolor}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:15}
Un pueblo vecino a un observatorio tiene que renovar el alumbrado de sus
calles y duda entre lámparas de sodio a baja \omterm{def:g10:pressure:pressure}{presión} y ledes blancos de
\omterm{def:g10:light-spectra:white}{espectro} ancho. (a) ¿Qué opción prefieren los astrónomos y por qué?
(Piensa en lo que un solo \omterm{def:g11:color-light-sources:subtractive}{filtro} puede quitar de sus imágenes.) (b) ¿Qué
opción prefieren conductores y peatones? Justifícalo con los colores de
los objetos. (c) Propón un compromiso, formulado con la \omterm{def:g11:color-light-sources:colortemp}{temperatura de
color}.
\end{exercise}

\section{Problema: fabricar color}

\begin{problem}[{Problema de fin de semana --- focos de teatro, calles
de sodio y el led blanco: cómo se fabrica el color, desde tres focos
capaces de pintar cualquier tono hasta el chip que ilumina esta página}]
\label{pb:g11:color-light-sources:1}
Un iluminador de teatro solo tiene \omterm{def:g11:lenses-and-eye:focal}{focos} rojos, verdes y azules; un
ingeniero de alumbrado tiene una lámpara que emite una única \omterm{def:g10:light-spectra:wavelength}{longitud de
onda}; un fabricante de chips tiene un diodo que solo luce azul. Los tres
venden color. Este problema recorre sus oficios por turnos --- mezcla
aditiva en el escenario, economía \omterm{def:g10:light-spectra:white}{monocromática} en la calle y alquimia
de \omterm{def:g11:color-light-sources:led}{fósforos} en el chip --- y termina eligiendo la luz de un dormitorio y
la del escaparate de un joyero.

\medskip
\noindent\textbf{Parte I --- El escenario: pintar con tres \omterm{def:g11:lenses-and-eye:focal}{focos}.}
\begin{enumerate}
  \item Los tres \omterm{def:g11:lenses-and-eye:focal}{focos} se superponen sobre un telón blanco. Nombra el
        color de cada superposición dos a dos y el de la superposición
        triple.
  \item ¿Cómo consigue el iluminador un naranja? ¿Y un rosa pálido?
        (Describe cualitativamente las \omterm{def:g10:energy-conservation:power}{potencias} de los \omterm{def:g11:lenses-and-eye:focal}{focos}.)
  \item Una bailarina se sitúa delante del telón, iluminada solo por los
        \omterm{def:g11:lenses-and-eye:focal}{focos} rojo y verde. Aparecen dos sombras de color: ¿de qué color
        es cada una? Explícalo.
  \item Se repinta el telón de azul oscuro. ¿Cómo se ve bajo el \omterm{def:g11:lenses-and-eye:focal}{foco}
        rojo solo? ¿Y bajo los tres?
  \item ¿Por qué pueden tres \omterm{def:g11:lenses-and-eye:focal}{focos} reproducir ``casi cualquier'' color?
        ¿Qué pieza de anatomía fija el número tres?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El vestuario: la resta en acción.}
\begin{enumerate}[resume]
  \item Un traje es amarillo bajo \omterm{def:g10:light-spectra:white}{luz blanca}. ¿Qué banda absorbe su
        tinte?
  \item Predice su color bajo el \omterm{def:g11:lenses-and-eye:focal}{foco} rojo, el verde y el azul.
  \item El iluminador quiere que el traje pase de brillante a negro con
        un solo golpe de interruptor. ¿A qué \omterm{def:g11:lenses-and-eye:focal}{foco} conmuta ese golpe?
  \item Se desliza una gelatina magenta delante de un \omterm{def:g11:lenses-and-eye:focal}{foco} blanco que
        ilumina el traje amarillo. Sigue las bandas
        (\cref{met:g11:color-light-sources:bookkeeping}): ¿color del
        traje?
  \item ¿Por qué las gelatinas de teatro se venden en cian, magenta y
        amarillo en lugar de rojo, verde y azul? (¿Qué le haría una
        gelatina roja a la \omterm{def:g10:energy-conservation:power}{potencia} de un \omterm{def:g11:lenses-and-eye:focal}{foco} blanco?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Calles de sodio: la economía
\omterm{def:g10:light-spectra:white}{monocromática}.}
\begin{enumerate}[resume]
  \item Una lámpara de sodio a baja \omterm{def:g10:pressure:pressure}{presión} emite esencialmente una
        línea en $\qty{589}{nm}$. ¿Qué tipo de \omterm{def:g10:light-spectra:white}{espectro} es, y qué
        aspecto tiene la calle entera bajo él?
  \item Bajo esa lámpara, da el color aparente de una pared blanca, de
        una marca vial amarilla, de un coche rojo y de uno azul.
  \item ¿Por qué adoraban los astrónomos esas lámparas? (¿Qué puede
        hacerle un solo \omterm{def:g11:color-light-sources:subtractive}{filtro} al resplandor de toda la ciudad?)
  \item Se propone en su lugar una bombilla de wolframio de alumbrado a
        $\qty{2700}{K}$. Calcula su $\lambda_{\max}$: ¿adónde va casi
        toda su \omterm{def:g10:energy-conservation:power}{potencia}?
  \item La lámpara de sodio convierte en luz visible cerca del $30\,\%$ de
        su \omterm{def:g10:energy-conservation:power}{potencia} eléctrica, y la bombilla de wolframio, un $5\,\%$.
        ¿Cuántos \omterm{def:g10:energy-conservation:power}{vatios} de luz entrega cada una por cada
        $\qty{100}{W}$ de \omterm{def:g10:energy-conservation:power}{potencia}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El led blanco y la elección de una luz.}
\begin{enumerate}[resume]
  \item Un led blanco es un diodo azul ($\qty{450}{nm}$) bajo un \omterm{def:g11:color-light-sources:led}{fósforo}
        amarillo. Explica, con los \omterm{prop:g11:color-light-sources:complementary}{colores complementarios}, por qué su
        emisión se lee como blanca.
  \item Describe su \omterm{def:g10:light-spectra:white}{espectro} y compáralo con el \omterm{def:g10:light-spectra:continuous}{espectro continuo} de la
        luz del día. ¿Qué banda está infrarrepresentada?
  \item Deduce qué objetos reproduce mal un led así y relaciónalo con el
        carnicero del \cref{exo:g11:color-light-sources:14}.
  \item Una bombilla led se vende como ``$\qty{2700}{K}$''. ¿Hay algo
        dentro de ella a $\qty{2700}{K}$? ¿Qué promete la etiqueta?
        Calcula el $\lambda_{\max}$ del cuerpo incandescente que imita.
  \item Final --- elige, con una frase de justificación cada vez, una
        \omterm{def:g11:color-light-sources:colortemp}{temperatura de color} para la lámpara de un dormitorio y otra
        para el escaparate de un joyero; y enuncia después en una línea
        la ley del color de este capítulo.
\end{enumerate}
\end{problem}
