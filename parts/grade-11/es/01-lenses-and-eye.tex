\chapter{Lentes, imágenes y el ojo}\label{ch:g11:lenses-and-eye}

Toda imagen nítida --- en el sensor de un móvil, en una pantalla de cine
o en el fondo de tu propio ojo --- sale del mismo truco: un medio
transparente curvado desvía los rayos por \omterm{def:g10:refraction:refraction}{refracción}
(\cref{ch:g10:refraction}) de modo que todos los rayos que salen de un
punto vuelven a encontrarse en otro. Este capítulo domestica el truco:
una \omterm{def:g11:lenses-and-eye:lens}{lente}, tres rayos, una fórmula --- y después lo aplica a las
cámaras, a las gafas y a la \omterm{def:g11:lenses-and-eye:lens}{lente} viva que está leyendo esta frase.

\section{Lentes convergentes y divergentes}

\begin{definition}[Lente delgada]\label{def:g11:lenses-and-eye:lens}
Una \emph{lente}\index{lente} es un medio transparente limitado por dos
superficies, al menos una de ellas curva. Si es más gruesa en el centro
que en el borde, es \emph{convergente}\index{lente convergente}: los
rayos paralelos se desvían unos hacia otros; si es más gruesa en el
borde, es \emph{divergente}\index{lente divergente}: los rayos se
separan. Una lente \emph{delgada}\index{lente delgada} tiene un grosor
despreciable: la dibujamos como un segmento con puntas de flecha hacia
fuera (convergente) o hacia dentro (divergente). Su centro $O$ es el
\emph{centro óptico}\index{centro óptico}; la recta perpendicular que
pasa por $O$, el \emph{eje principal}\index{eje principal}.
\end{definition}

\begin{definition}[Focos y distancia focal]\label{def:g11:lenses-and-eye:focal}
Los rayos que llegan paralelos al eje de una \omterm{def:g11:lenses-and-eye:lens}{lente convergente} cortan
todos el eje en un mismo punto situado detrás de ella, el \emph{foco
imagen}\index{foco} $F'$; simétricamente, los rayos que salen del
\emph{foco objeto} $F$ (delante, con $OF = OF'$) emergen paralelos al
eje. La \emph{distancia focal}\index{distancia focal}
$f' = \overline{OF'}$ es positiva para una \omterm{def:g11:lenses-and-eye:lens}{lente convergente}; a través
de una \omterm{def:g11:lenses-and-eye:lens}{lente divergente}, los rayos paralelos salen como si vinieran de
un punto $F'$ situado \emph{delante} de la \omterm{def:g11:lenses-and-eye:lens}{lente}: $f' < 0$.
\end{definition}

\begin{definition}[Potencia óptica]\label{def:g11:lenses-and-eye:vergence}
La \emph{potencia óptica}\index{potencia óptica} de una \omterm{def:g11:lenses-and-eye:lens}{lente} de
\omterm{def:g11:lenses-and-eye:focal}{distancia focal} $f'$ (en \omterm{def:g10:orders-of-magnitude:unit}{metros}) es $C = 1/f'$, medida en
\emph{dioptrías}\index{dioptría} (símbolo D):
$1~\text{D} = \qty{1}{m^{-1}}$; las \omterm{def:g11:lenses-and-eye:lens}{lentes divergentes} tienen $C < 0$.
Los ópticos recetan en dioptrías: las \omterm{def:g10:energy-conservation:power}{potencias} ópticas de \omterm{def:g11:lenses-and-eye:lens}{lentes
delgadas} en contacto simplemente se suman (admitido aquí; deducido en el
volumen del primer año).
\end{definition}

\begin{example}[Leer una receta]\label{ex:g11:lenses-and-eye:vergence}
Una \omterm{def:g11:lenses-and-eye:lens}{lente} con $f' = \qty{0.50}{m}$ tiene $C = 1/0.50 = +2.0$~D:
\omterm{def:g11:lenses-and-eye:lens}{convergente}. Una receta de $-4.0$~D significa $f' = \qty{-0.25}{m}$:
\omterm{def:g11:lenses-and-eye:lens}{divergente}, de \qty{25}{cm} de \omterm{def:g11:lenses-and-eye:focal}{distancia focal}.
\end{example}

\section{Construir la imagen}

\begin{method}[Los tres rayos de construcción]\label{met:g11:lenses-and-eye:rays}
Sea $AB$ un objeto perpendicular al eje en $A$. De todos los rayos que
salen de la punta $B$, hay tres cuyo camino se conoce:
\begin{enumerate}
  \item el rayo que pasa por el \omterm{def:g11:lenses-and-eye:lens}{centro óptico} $O$ sigue recto;
  \item el rayo paralelo al eje sale pasando por $F'$;
  \item el rayo que pasa por $F$ sale paralelo al eje.
\end{enumerate}
Bastan dos cualesquiera: su intersección es el punto imagen $B'$, y la
imagen $A'B'$ queda perpendicular al eje en $A'$. Si los rayos salientes
divergen, sus prolongaciones \emph{hacia atrás} (a trazos) se cortan en
un punto \omterm{def:g11:lenses-and-eye:images}{imagen virtual}.
\end{method}

\begin{definition}[Imágenes reales y virtuales]\label{def:g11:lenses-and-eye:images}
Una imagen es \emph{real}\index{imagen real} cuando los rayos salientes
pasan realmente por ella: una pantalla puesta ahí la recoge; es
\emph{virtual}\index{imagen virtual} cuando solo se cortan sus
prolongaciones hacia atrás: ninguna pantalla la recoge, pero un ojo que
mire a través de la \omterm{def:g11:lenses-and-eye:lens}{lente} la ve perfectamente. Una \omterm{def:g11:lenses-and-eye:lens}{lente convergente} da
una imagen real e invertida de cualquier objeto situado más allá de $F$;
si el objeto se desliza dentro de la \omterm{def:g11:lenses-and-eye:focal}{distancia focal}, la imagen pasa a
ser virtual, derecha y aumentada.
\end{definition}

\section{Una sola fórmula: la relación de conjugación}

Las construcciones muestran \emph{dónde} está la imagen; una fórmula la
calcula. Se orienta el eje en el sentido de la luz y se usan
\emph{medidas algebraicas}: $\overline{OA}$ es la coordenada de $A$
respecto de $O$ (negativa para un objeto \omterm{def:g11:lenses-and-eye:images}{real} situado delante); las
alturas $\overline{AB}$ se cuentan positivas hacia arriba.

\begin{theorem}[Relación de conjugación de las lentes delgadas]\label{thm:g11:lenses-and-eye:conjugate}
A través de una \omterm{def:g11:lenses-and-eye:lens}{lente delgada} de \omterm{def:g11:lenses-and-eye:focal}{distancia focal} $f'$, un punto objeto
$A$ del eje forma su imagen en el punto $A'$ con
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'} = C .
\]
\end{theorem}
\admitted

\begin{remark}
La relación se sigue de la \omterm{thm:g10:refraction:snell}{ley de Snell} aplicada a los rayos próximos al
eje; la deducción --- y los límites del modelo de \omterm{def:g11:lenses-and-eye:lens}{lente delgada} --- se
hacen honradamente en el volumen del primer año.
\end{remark}

\begin{definition}[Aumento]\label{def:g11:lenses-and-eye:magnification}
El \emph{aumento}\index{aumento} de la imagen es
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}}
       = \frac{\overline{OA'}}{\overline{OA}} .
\]
$\abs{\gamma}$ compara los tamaños y su signo, las orientaciones
($\gamma < 0$: invertida); el signo de $\overline{OA'}$ indica la
naturaleza (positivo: \omterm{def:g11:lenses-and-eye:images}{real}, detrás de la \omterm{def:g11:lenses-and-eye:lens}{lente}; negativo: \omterm{def:g11:lenses-and-eye:images}{virtual},
delante).
\end{definition}

\begin{example}[La fórmula en acción]\label{ex:g11:lenses-and-eye:conjugate}
Un objeto está \qty{30}{cm} delante de una \omterm{def:g11:lenses-and-eye:lens}{lente convergente} con
$f' = \qty{10}{cm}$: $\overline{OA} = \qty{-30}{cm}$, luego
\[
\frac{1}{\overline{OA'}} = \frac{1}{10} - \frac{1}{30} = \frac{1}{15},
\qquad \overline{OA'} = +\qty{15}{cm},
\qquad \gamma = \frac{15}{-30} = -0.50 :
\]
una \omterm{def:g11:lenses-and-eye:images}{imagen real} \qty{15}{cm} detrás de la \omterm{def:g11:lenses-and-eye:lens}{lente}, invertida y de tamaño
mitad --- exactamente lo que dibujan los tres rayos.
\end{example}

\section{El ojo}

\begin{definition}[El ojo como sistema óptico]\label{def:g11:lenses-and-eye:eye}
Ópticamente, el ojo es una \omterm{def:g11:lenses-and-eye:lens}{lente convergente} frente a una pantalla: la
\emph{córnea}\index{córnea} y el \emph{cristalino}\index{cristalino}
actúan como una sola \omterm{def:g11:lenses-and-eye:lens}{lente convergente} de \omterm{def:g11:lenses-and-eye:focal}{distancia focal} ajustable; la
\emph{retina}\index{retina}, a unos \qty{17}{mm} por detrás, es la
pantalla fija. El enfoque se consigue apretando la \omterm{def:g11:lenses-and-eye:lens}{lente}, no
desplazándola como en una cámara: la
\emph{acomodación}\index{acomodación} es el cambio de $f'$ que producen
los músculos ciliares. El punto más próximo que se ve nítido con la
acomodación al máximo es el \emph{punto próximo}\index{punto próximo}
--- unos \qty{25}{cm} para el \emph{ojo normal}\index{ojo normal} --- y
el más lejano, en reposo, el \emph{punto remoto}\index{punto remoto}
(el infinito para el ojo normal).
\end{definition}

\begin{example}[La potencia óptica de tu ojo]\label{ex:g11:lenses-and-eye:eyerange}
La \omterm{def:g11:lenses-and-eye:eye}{retina} está en $\overline{OA'} = +\qty{0.017}{m}$. Objeto lejano
($1/\overline{OA} \approx 0$): $C = 1/0.017 \approx 59$~D. \omterm{def:g11:lenses-and-eye:eye}{Punto próximo}
($\overline{OA} = \qty{-0.25}{m}$):
$C = 1/0.017 + 1/0.25 \approx 63$~D. Leer cuesta unas $4$~D: el ojo es
una \omterm{def:g11:lenses-and-eye:lens}{lente} de $59$~D con un ajuste fino incorporado de $+4$~D.
\end{example}

\section{Defectos, correcciones y la lupa}

\begin{definition}[Miopía e hipermetropía]\label{def:g11:lenses-and-eye:defects}
Un ojo \emph{miope}\index{miopía} (corto de vista) es demasiado
\omterm{def:g11:lenses-and-eye:lens}{convergente} para su longitud: los rayos paralelos se enfocan
\emph{delante de} la \omterm{def:g11:lenses-and-eye:eye}{retina}, los objetos lejanos se ven borrosos y el
\omterm{def:g11:lenses-and-eye:eye}{punto remoto} queda a distancia finita; se corrige con una \omterm{def:g11:lenses-and-eye:lens}{lente
divergente} en las gafas. Un ojo \emph{hipermétrope}\index{hipermetropía}
no converge lo suficiente: los objetos cercanos se enfocarían detrás de
la \omterm{def:g11:lenses-and-eye:eye}{retina} y el \omterm{def:g11:lenses-and-eye:eye}{punto próximo} se aleja; se corrige con una \omterm{def:g11:lenses-and-eye:lens}{lente
convergente}. La edad endurece el \omterm{def:g11:lenses-and-eye:eye}{cristalino}
(\emph{presbicia}\index{presbicia}) y encoge la \omterm{def:g11:lenses-and-eye:eye}{acomodación} --- mismo
remedio: gafas de lectura.
\end{definition}

\begin{example}[Recetar para un miope]\label{ex:g11:lenses-and-eye:correction}
Un ojo \omterm{def:g11:lenses-and-eye:defects}{miope} tiene su \omterm{def:g11:lenses-and-eye:eye}{punto remoto} a \qty{50}{cm}: la \omterm{def:g11:lenses-and-eye:lens}{lente} correctora
debe mostrar los objetos lejanos \emph{en el \omterm{def:g11:lenses-and-eye:eye}{punto remoto}} --- objeto en
el infinito e imagen en $\overline{OA'} = \qty{-0.50}{m}$ (\omterm{def:g11:lenses-and-eye:images}{virtual}, por
delante). Entonces $1/f' = 1/\overline{OA'}$: $f' = \qty{-0.50}{m}$,
$C = -2.0$~D; el ojo relajado mira esa \omterm{def:g11:lenses-and-eye:images}{imagen virtual} y la ve nítida.
\end{example}

\begin{definition}[La lupa]\label{def:g11:lenses-and-eye:magnifier}
Una \emph{lupa}\index{lupa} es una \omterm{def:g11:lenses-and-eye:lens}{lente convergente} de \omterm{def:g11:lenses-and-eye:focal}{distancia focal}
corta usada con el objeto \emph{dentro} de la \omterm{def:g11:lenses-and-eye:focal}{distancia focal}: la imagen
es \omterm{def:g11:lenses-and-eye:images}{virtual}, derecha y aumentada
(\cref{def:g11:lenses-and-eye:images}), y el ojo la observa cómodamente
desde lejos.
\end{definition}

\begin{example}[La lupa del joyero]\label{ex:g11:lenses-and-eye:magnifier}
\omterm{def:g11:lenses-and-eye:magnifier}{Lupa} con $f' = \qty{5.0}{cm}$ y piedra a \qty{4.0}{cm}:
$1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$, luego
$\overline{OA'} = \qty{-20}{cm}$ y $\gamma = (-20)/(-4.0) = +5.0$: una
\omterm{def:g11:lenses-and-eye:images}{imagen virtual} y derecha, cinco veces mayor, a unos cómodos \qty{20}{cm}
de la \omterm{def:g11:lenses-and-eye:lens}{lente}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:lenses-and-eye:2}

\qty{40}{cm} de la \omterm{def:g11:lenses-and-eye:lens}{lente}. Calcula $f'$ y $\gamma$. ¿Por qué esta imagen
simétrica tiene que ser exactamente del mismo tamaño que el objeto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:lenses-and-eye:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:lenses-and-eye:11}
Un lector présbita tiene su \omterm{def:g11:lenses-and-eye:eye}{punto próximo} a \qty{80}{cm} pero quiere
leer a \qty{25}{cm}: las gafas tienen que formar, de una página situada
a \qty{25}{cm}, una página \omterm{def:g11:lenses-and-eye:images}{virtual} a \qty{80}{cm}. Calcula la \omterm{def:g11:lenses-and-eye:vergence}{potencia

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:lenses-and-eye:12}
Un proyector tiene que lanzar la imagen de una diapositiva, con un

  \item Deduce la \omterm{def:g11:lenses-and-eye:focal}{distancia focal} del objetivo de proyección.

\begin{exercise}[$\star\star\star$]\label{exo:g11:lenses-and-eye:13}

  \item Calcula la \omterm{def:g11:lenses-and-eye:vergence}{potencia óptica} del ojo enfocado en el infinito y
        después en el \omterm{def:g11:lenses-and-eye:eye}{punto próximo} normal (\qty{25}{cm}). Deduce la
        \omterm{def:g10:signals-and-waves:amplitude}{amplitud} de \omterm{def:g11:lenses-and-eye:eye}{acomodación} del \omterm{def:g11:lenses-and-eye:eye}{ojo normal}.
  \item Un ojo de más edad tiene su \omterm{def:g11:lenses-and-eye:eye}{punto próximo} a \qty{1.0}{m}.
        Calcula su \omterm{def:g10:signals-and-waves:amplitude}{amplitud} y la fracción de la \omterm{def:g10:signals-and-waves:amplitude}{amplitud} normal que le
        queda.
\end{enumerate}

\begin{exercise}[$\star\star\star$]\label{exo:g11:lenses-and-eye:14}
Una \omterm{def:g11:lenses-and-eye:magnifier}{lupa} vendida como ``$\times 3$'' tiene su \omterm{def:g11:lenses-and-eye:focal}{distancia focal} dada por

\begin{enumerate}
  \item Calcula $f'$.
  \item ¿Dónde tiene que estar un objeto para que su \omterm{def:g11:lenses-and-eye:images}{imagen virtual}
        quede \qty{25}{cm} delante de la \omterm{def:g11:lenses-and-eye:lens}{lente}? Calcula $\gamma$ y
        compáralo con el $\times 3$ anunciado.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:lenses-and-eye:15}
Un objeto \omterm{def:g11:lenses-and-eye:images}{real} está a una distancia $d > 0$ delante de una \omterm{def:g11:lenses-and-eye:lens}{lente
convergente} ($\overline{OA} = -d$, $f' > 0$). Demuestra que
$\overline{OA'} = \frac{f'd}{d - f'}$ y
$\gamma = \frac{f'}{f' - d}$. Deduce: la imagen es \omterm{def:g11:lenses-and-eye:images}{real} e invertida
cuando $d > f'$; \omterm{def:g11:lenses-and-eye:images}{virtual}, derecha y \emph{aumentada} ($\gamma > 1$)
cuando $0 < d < f'$ --- toda \omterm{def:g11:lenses-and-eye:lens}{lente convergente} es una \omterm{def:g11:lenses-and-eye:magnifier}{lupa} si se usa
suficientemente cerca.
\end{exercise}

\section{Problema: una fórmula, tres instrumentos}

\begin{problem}[{Problema de fin de semana --- la misma relación de
conjugación enfoca una cámara, receta unas gafas de lectura, diseña una
lupa y mide el zum que llevas en el ojo}]
\label{pb:g11:lenses-and-eye:1}
Un fotógrafo mueve un objetivo de \qty{50}{mm}, una abuela sujeta el
periódico con el brazo estirado, un joyero se inclina sobre un diamante:
una sola línea de álgebra,
$1/\overline{OA'} - 1/\overline{OA} = 1/f'$
(\cref{thm:g11:lenses-and-eye:conjugate}), gobierna los tres oficios ---
y después el propio ojo.

\medskip
\noindent\textbf{Parte I --- Una sola fórmula.} Una \omterm{def:g11:lenses-and-eye:lens}{lente convergente}
tiene $f' = \qty{20}{cm}$.
\begin{enumerate}
  \item Objeto a \qty{60}{cm}: calcula $\overline{OA'}$ y $\gamma$;
        describe la imagen.
  \item Mueve el objeto a \qty{30}{cm}; recalcula. ¿Qué se ha
        intercambiado respecto de la pregunta 1? (Los caminos de la luz
        son reversibles.)
  \item Muévelo a \qty{10}{cm}, dentro de $f'$: recalcula y describe.
  \item Resume: cuando el objeto se acerca desde muy lejos hasta $F$, y
        después lo rebasa, ¿cómo se mueve la imagen y cómo cambia de
        naturaleza?
  \item Demuestra que un objeto muy lejano
        ($1/\overline{OA} \approx 0$) forma siempre su imagen en el
        plano focal: $\overline{OA'} = f'$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El fotógrafo.}
Un objetivo tiene $f' = \qty{50}{mm}$; el sensor, de \qty{24}{mm} de
alto, se puede desplazar respecto del objetivo.
\begin{enumerate}[resume]
  \item ¿Dónde está el sensor cuando se enfoca un paisaje lejano?
  \item Ahora un rostro a \qty{1.0}{m}: calcula la nueva distancia
        objetivo-sensor y el recorrido desde la posición del paisaje.
  \item El mecanismo permite como mucho \qty{55}{mm} de distancia
        objetivo-sensor. Calcula la distancia mínima de enfoque.
  \item En ese enfoque más próximo, calcula $\gamma$. ¿Cabe en el sensor
        la imagen de un rostro de \qty{20}{cm}?
  \item El ``macro'' a tamaño \omterm{def:g11:lenses-and-eye:images}{real} significa $\gamma = -1$: ¿dónde deben
        estar entonces el objeto y el sensor, y por qué exige eso un
        objetivo especial?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La abuela.}
Con la edad, el \omterm{def:g11:lenses-and-eye:eye}{punto próximo} se aleja: el de la abuela está a
\qty{1.0}{m}.
\begin{enumerate}[resume]
  \item Explica lo del periódico con el brazo estirado.
  \item Para leer a \qty{25}{cm}, sus gafas tienen que convertir una
        página situada a \qty{25}{cm} en una \omterm{def:g11:lenses-and-eye:images}{imagen virtual} a
        \qty{1.0}{m}: calcula la \omterm{def:g11:lenses-and-eye:vergence}{potencia óptica} necesaria.
  \item Calcula $\gamma$. La imagen es cuatro veces mayor y, sin
        embargo, no parece mayor: explícalo comparando tamaños \emph{y}
        distancias.
  \item Demuestra que el texto solo se ve nítido entre \qty{25}{cm} y
        $f' \approx \qty{33}{cm}$. ¿Y un objeto \emph{lejano} visto a
        través de las gafas?
  \item Deduce en una frase por qué existen las \omterm{def:g11:lenses-and-eye:lens}{lentes} bifocales.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El joyero y el zum del propio ojo.}
La \omterm{def:g11:lenses-and-eye:magnifier}{lupa} del joyero tiene $f' = \qty{5.0}{cm}$; su \omterm{def:g11:lenses-and-eye:eye}{punto próximo} está a
\qty{25}{cm}; su \omterm{def:g11:lenses-and-eye:eye}{retina}, \qty{17}{mm} por detrás de la \omterm{def:g11:lenses-and-eye:lens}{lente} de su ojo.
\begin{enumerate}[resume]
  \item Una piedra \qty{4.0}{cm} bajo la \omterm{def:g11:lenses-and-eye:magnifier}{lupa}: ¿posición, naturaleza,
        \omterm{def:g11:lenses-and-eye:magnification}{aumento}?
  \item ¿Dónde tiene que estar la piedra para que su imagen caiga en su
        \omterm{def:g11:lenses-and-eye:eye}{punto próximo}, y cuánto vale entonces $\gamma$?
  \item ¿Por qué los joyeros experimentados colocan en cambio la piedra
        exactamente en $F$ (imagen en el infinito)?
  \item Ahora el ojo solo: calcula su \omterm{def:g11:lenses-and-eye:vergence}{potencia óptica} mirando un objeto
        lejano y después uno a \qty{25}{cm}: ¿cuántas \omterm{def:g11:lenses-and-eye:vergence}{dioptrías} de
        \omterm{def:g11:lenses-and-eye:eye}{acomodación}?
  \item Convierte las dos \omterm{def:g11:lenses-and-eye:vergence}{potencias ópticas} en \omterm{def:g11:lenses-and-eye:focal}{distancias focales}: ¿en
        cuánto, en milímetros y en porcentaje, cambia el ojo su propia
        \omterm{def:g11:lenses-and-eye:focal}{distancia focal} --- el zum que la cámara hacía con
        desplazamiento y la abuela con $+3$~D?
\end{enumerate}
\end{problem}
