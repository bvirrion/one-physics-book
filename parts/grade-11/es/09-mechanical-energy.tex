\chapter{La energía mecánica y su conservación}\label{ch:g11:mechanical-energy}

Una montaña rusa no tiene motor a partir de la primera cuesta: cada
rizo, cada arrancada de velocidad posterior se paga de la cuenta abierta
en la subida. Este capítulo convierte esa contabilidad en dos números
--- la \omterm{def:g11:mechanical-energy:kinetic}{energía cinética} y la potencial --- y una regla: mientras solo
trabaje el \omterm{def:g10:universal-gravitation:weight}{peso}, su suma no cambia. El \omterm{def:g10:inertia:inventory}{rozamiento} es la comisión;
aprenderemos a leerla en los libros.

\section{La energía cinética}

\begin{definition}[Energía cinética]\label{def:g11:mechanical-energy:kinetic}
Un cuerpo de masa $m$ (\unit{kg}) que se mueve con rapidez $v$
(\unit{m/s}) lleva la \emph{energía cinética}\index{energía cinética}
\[
E_k = \tfrac12\, m v^2 .
\]
Comprobación de \omterm{def:g10:orders-of-magnitude:unit}{unidades}: $\unit{kg.m^2/s^2} = \unit{N.m} = \unit{J}$
--- \omterm{def:g11:work-of-force:work}{julios}, como toda \omterm{def:g10:energy-conservation:energy}{energía} (\cref{ch:g10:energy-conservation}).
\end{definition}

\begin{example}[Órdenes de magnitud]\label{ex:g11:mechanical-energy:orders}
\begin{itemize}
  \item un peatón, \qty{70}{kg} a \qty{1.4}{m/s}:
        $E_k = \tfrac12 \times 70 \times 1.4^2 \approx \qty{69}{J}$;
  \item un coche, \qty{1300}{kg} a \qty{130}{km/h} $= \qty{36.1}{m/s}$:
        $E_k \approx \qty{8.5e5}{J}$;
  \item un tren de alta velocidad, $m = \qty{3.85e5}{kg}$, a
        \qty{320}{km/h} $= \qty{88.9}{m/s}$: $E_k \approx \qty{1.5e9}{J}$;
  \item una bala de fusil, \qty{8.0}{g} a \qty{800}{m/s}:
        $E_k \approx \qty{2.6}{kJ}$.
\end{itemize}
Siete órdenes de magnitud separan el paseo del tren. Y $E_k$ crece como
el \emph{cuadrado} de la rapidez: a \qty{130}{km/h} el coche lleva $4$
veces la \omterm{def:g10:energy-conservation:energy}{energía} que tiene a \qty{65}{km/h} --- cuatro veces lo que sus
frenos tendrán que retirar.
\end{example}

\section{El teorema de la energía cinética}

\begin{theorem}[Teorema de la energía cinética]\label{thm:g11:mechanical-energy:ketheorem}
\index{teorema de la energía cinética}
Para un cuerpo de masa $m$ que se mueve en línea recta de $A$ a $B$, la
variación de \omterm{def:g11:mechanical-energy:kinetic}{energía cinética} es igual al \omterm{def:g11:work-of-force:work}{trabajo} total
(\cref{ch:g11:work-of-force}) recibido:
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
donde la suma recorre todas las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre el cuerpo. Se
extiende al movimiento por cualquier camino, cosa que también admitimos
aquí.
\end{theorem}
\admitted

\begin{example}[Caída libre, comprobada fotograma a fotograma]\label{ex:g11:mechanical-energy:freefall}
Se deja caer una bola de \qty{0.200}{kg} y se graba; los fotogramas dan:
\begin{center}\small
\begin{tabular}{lcccc}
$t$ (\unit{s}) & 0.10 & 0.20 & 0.30 & 0.40\\
altura caída $h$ (\unit{m}) & 0.049 & 0.196 & 0.441 & 0.785\\
rapidez $v$ (\unit{m/s}) & 0.98 & 1.96 & 2.94 & 3.92\\
$E_k = \tfrac12 mv^2$ (\unit{J}) & 0.096 & 0.384 & 0.864 & 1.54\\
$W = mgh$ (\unit{J}) & 0.096 & 0.385 & 0.865 & 1.54
\end{tabular}
\end{center}
Partiendo de $E_k = 0$, en cada fotograma $E_k$ coincide con el \omterm{def:g11:work-of-force:work}{trabajo}
$mgh$ del \omterm{def:g10:universal-gravitation:weight}{peso}: $\Delta E_k = W(\vect P)$, es decir $v^2 = 2gh$ --- el
teorema, verificado hasta la precisión de las lecturas.
\end{example}

\begin{remark}[Dónde vive la demostración]\label{rem:g11:mechanical-energy:proof}
El enunciado general --- válido por cualquier camino, no solo en línea
recta --- se sigue de la segunda ley de Newton y de un poco de cálculo
infinitesimal, ambos más adelante en este libro
(\cref{ch:g12:newtons-laws,ch:g12:work-and-energy}); hasta entonces, la
comprobación de la caída libre es nuestro aval, y admitimos la versión
para caminos curvos y la usamos con libertad.
\end{remark}

\section{La energía potencial gravitatoria}

\begin{definition}[Energía potencial gravitatoria]\label{def:g11:mechanical-energy:potential}
Un cuerpo de masa $m$ a la altitud $z$, medida hacia arriba desde un
\emph{nivel de referencia}\index{nivel de referencia} elegido donde
$z = 0$, almacena la \emph{energía potencial
gravitatoria}\index{energía potencial!gravitatoria}
\[
E_p = m g z, \qquad g = \qty{9.81}{N/kg}.
\]
\end{definition}

\begin{remark}[Solo cuentan las diferencias]\label{rem:g11:mechanical-energy:reference}
Ir de $A$ a $B$ cambia $E_p$ en
$mg(z_B - z_A) = -W_{AB}(\vect P)$: lo opuesto del \omterm{def:g11:work-of-force:work}{trabajo} del \omterm{def:g10:universal-gravitation:weight}{peso}
(\cref{ch:g11:work-of-force}). El \omterm{def:g11:mechanical-energy:potential}{nivel de referencia} es una elección
libre --- el suelo, el nivel del mar, la mesa ---, que desplaza $E_p$
en todas partes en una misma constante que se cae de toda diferencia.
Ponlo donde los números salgan más sencillos, y no lo muevas a mitad de
problema.
\end{remark}

\begin{example}[Un libro en una estantería]\label{ex:g11:mechanical-energy:shelf}
Un libro de \qty{1.2}{kg} está \qty{1.8}{m} por encima del suelo, que a
su vez está \qty{9.0}{m} por encima de la calle. Con la referencia en el
suelo: $E_p = 1.2 \times 9.81 \times 1.8 \approx \qty{21}{J}$; en la
calle: $E_p \approx \qty{127}{J}$. Caer al suelo libera \qty{21}{J} en
las dos contabilidades.
\end{example}

\section{La energía mecánica y su conservación}

\begin{definition}[Energía mecánica]\label{def:g11:mechanical-energy:em}
La \emph{energía mecánica}\index{energía mecánica} de un cuerpo es
\[
E_m = E_k + E_p = \tfrac12 m v^2 + m g z .
\]
\end{definition}

\begin{theorem}[Conservación de la energía mecánica]\label{thm:g11:mechanical-energy:conservation}
\index{conservación de la energía mecánica}
Si la única \omterm{def:g10:inertia:force}{fuerza} que trabaja sobre un cuerpo es su \omterm{def:g10:universal-gravitation:weight}{peso} (sin
\omterm{def:g10:inertia:inventory}{rozamiento}; y con las demás \omterm{def:g10:inertia:force}{fuerzas}, como la reacción de un carril sin
\omterm{def:g10:inertia:inventory}{rozamiento}, perpendiculares al movimiento), $E_m$ se conserva: para dos
puntos cualesquiera $A$ y $B$ del movimiento,
\[
\tfrac12 m v_A^2 + m g z_A = \tfrac12 m v_B^2 + m g z_B .
\]
\end{theorem}

\begin{proof}
Por el \omterm{thm:g11:mechanical-energy:ketheorem}{teorema de la energía cinética},
$E_k(B) - E_k(A) = W_{AB}(\vect P) = mg(z_A - z_B) = E_p(A) - E_p(B)$:
lo que gana $E_k$ lo pierde $E_p$.
\end{proof}

\begin{example}[Tobogán de agua]\label{ex:g11:mechanical-energy:slide}
Una niña parte del reposo en lo alto de un tobogán sin \omterm{def:g10:inertia:inventory}{rozamiento} de
altura $h = \qty{3.2}{m}$. Con la referencia abajo:
$mgh = \tfrac12 mv^2$, luego $v = \sqrt{2gh} \approx \qty{7.9}{m/s}$
(\qty{29}{km/h}) --- sea cual sea la masa \emph{y sea cual sea la forma
del tobogán}: recto, curvo o en espiral, solo entra el desnivel.
\end{example}

\begin{example}[Péndulo]\label{ex:g11:mechanical-energy:pendulum}
Se aparta la masa de un péndulo hasta que sube $h = \qty{12}{cm}$ por
encima de su punto más bajo y se suelta. En los extremos $v = 0$; abajo,
todo el $mgh$ es cinético, $v = \sqrt{2gh} \approx \qty{1.5}{m/s}$; y al
otro lado la masa vuelve a subir exactamente \qty{12}{cm}.
\end{example}

\section{La disipación por rozamiento}

\begin{definition}[Disipación]\label{def:g11:mechanical-energy:dissipation}
De la \omterm{def:g10:energy-conservation:energy}{energía} que abandona la cuenta mecánica se dice que se ha
\emph{disipado}\index{disipación}: reaparece como \omterm{def:g10:energy-conservation:forms}{energía térmica} en las
superficies que rozan y en el aire. La \omterm{def:g10:energy-conservation:energy}{energía} total se conserva
(\cref{ch:g10:energy-conservation}); la \omterm{def:g11:mechanical-energy:em}{energía mecánica} por sí sola,
no.
\end{definition}

\begin{proposition}[El déficit mide el rozamiento]\label{prop:g11:mechanical-energy:deficit}
Si además trabaja una \omterm{def:g10:inertia:force}{fuerza} de \omterm{def:g10:inertia:inventory}{rozamiento} $\vect f$ a lo largo del
movimiento,
\[
\Delta E_m = E_m(B) - E_m(A) = W_{AB}(\vect f) < 0:
\]
la pérdida de \omterm{def:g11:mechanical-energy:em}{energía mecánica} es igual al \omterm{def:g11:work-of-force:work}{trabajo} del \omterm{def:g10:inertia:inventory}{rozamiento}.
\end{proposition}

\begin{proof}
El \omterm{thm:g11:mechanical-energy:ketheorem}{teorema de la energía cinética} dice ahora
$\Delta E_k = W_{AB}(\vect P) + W_{AB}(\vect f) = -\Delta E_p +
W_{AB}(\vect f)$; se pasa $\Delta E_p$ al otro lado. El \omterm{def:g10:inertia:inventory}{rozamiento} se
opone al movimiento, luego su \omterm{def:g11:work-of-force:work}{trabajo} es negativo.
\end{proof}

\begin{example}[Auditoría de un trineo]\label{ex:g11:mechanical-energy:toboggan}
Una niña con su trineo ($m = \qty{40}{kg}$) parten del reposo, bajan
$h = \qty{5.0}{m}$ a lo largo de \qty{12}{m} de nieve y llegan a
\qty{7.0}{m/s} en lugar de los $\sqrt{2gh} \approx \qty{9.9}{m/s}$ sin
\omterm{def:g10:inertia:inventory}{rozamiento}. Con la referencia abajo:
$\Delta E_m = \tfrac12 \times 40 \times 7.0^2 - 40 \times 9.81
\times 5.0 = 980 - 1962 = \qty{-982}{J}$. Así que
$W(\vect f) = \qty{-982}{J}$ en \qty{12}{m}: una \omterm{def:g10:inertia:force}{fuerza} media de
\omterm{def:g10:inertia:inventory}{rozamiento} $f = 982/12 \approx \qty{82}{N}$, y \qty{982}{J} de calor en
los patines y en la nieve.
\end{example}

\begin{method}[Contabilidad energética]\label{met:g11:mechanical-energy:bookkeeping}
\begin{enumerate}
  \item Elegir un \omterm{def:g11:mechanical-energy:potential}{nivel de referencia} y mantenerlo; nombrar $A$ (los
        datos) y $B$ (la pregunta).
  \item Escribir $E_m = \tfrac12 mv^2 + mgz$ en $A$ y en $B$.
  \item Sin \omterm{def:g10:inertia:inventory}{rozamiento}: $E_m(A) = E_m(B)$; se despeja --- normalmente
        $v_B = \sqrt{v_A^2 + 2g(z_A - z_B)}$.
  \item Con \omterm{def:g10:inertia:inventory}{rozamiento}: $E_m(B) - E_m(A) = -f\ell$ ($\ell$: longitud del
        camino); se despeja la incógnita.
  \item Comprobación: las barras de $E_k$ y $E_p$ se apilan hasta $E_m$,
        constante salvo que haya \omterm{def:g10:inertia:inventory}{rozamiento}.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:1}
Calcula la \omterm{def:g11:mechanical-energy:kinetic}{energía cinética} de (a) una moto con su conductor,
\qty{90}{kg} en total, a \qty{25}{km/h}; (b) un coche de \qty{1200}{kg}
a \qty{130}{km/h}. (c) ¿En qué factor cambia $E_k$ cuando la rapidez se
duplica? ¿Y cuando se triplica?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:2}
Un escalador de \qty{75}{kg} gana \qty{850}{m} de altitud. Calcula el
\omterm{def:g11:lenses-and-eye:magnification}{aumento} de \omterm{def:g10:energy-conservation:energy}{energía} potencial. ¿Depende del \omterm{def:g11:mechanical-energy:potential}{nivel de referencia}? ¿Y de la
ruta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:3}
Una maceta cae \qty{20}{m} desde un balcón. Despreciando la \omterm{def:g11:circuits-and-power:resistance}{resistencia}
del aire, halla su rapidez al llegar al suelo en \unit{m/s} y en
\unit{km/h}. ¿Adónde se ha ido la masa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:4}
Un coche de \qty{1200}{kg} frena de \qty{25}{m/s} hasta pararse en
\qty{50}{m}. Usa el \omterm{thm:g11:mechanical-energy:ketheorem}{teorema de la energía cinética} para hallar el
\omterm{def:g11:work-of-force:work}{trabajo} de los frenos y después la \omterm{def:g10:inertia:force}{fuerza} media de frenado.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:5}
Se suelta la masa de un péndulo \qty{8.0}{cm} por encima de su punto más
bajo. Halla su rapidez abajo y la altura que alcanza al otro lado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:6}
Con el \cref{ex:g11:mechanical-energy:orders}: ¿en qué factor supera la
\omterm{def:g11:mechanical-energy:kinetic}{energía cinética} del tren a la de la bala? ¿A qué altura podría elevar
al tren su propia $E_k$ ($h = v^2/2g$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:7}
Dos toboganes sin \omterm{def:g10:inertia:inventory}{rozamiento} bajan desde la misma plataforma de
\qty{3.0}{m}, uno recto y empinado y otro largo y serpenteante. Compara
las rapideces de llegada (calcúlalas) y los tiempos de llegada (sin
calcular).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:8}
Se lanza una pelota verticalmente hacia arriba a \qty{12}{m/s}. Halla la
altura máxima alcanzada y después la rapidez a \qty{4.0}{m} --- de
subida y de bajada.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:9}
Un saltador de esquí deja el trampolín a \qty{23}{m/s} y aterriza
\qty{35}{m} más abajo. Predice la rapidez de aterrizaje sin \omterm{def:g11:circuits-and-power:resistance}{resistencia}
del aire; ¿el valor real es mayor o menor, y por qué?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:10}
Un trineo de \qty{45}{kg} parte del reposo, baja \qty{6.0}{m} a lo largo
de \qty{15}{m} de pendiente y llega a \qty{8.0}{m/s}. Halla la \omterm{def:g11:mechanical-energy:em}{energía
mecánica} perdida y después la \omterm{def:g10:inertia:force}{fuerza} media de \omterm{def:g10:inertia:inventory}{rozamiento}. ¿Adónde ha ido
la \omterm{def:g10:energy-conservation:energy}{energía}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:11}
El clavo de Galileo: un péndulo soltado \qty{15}{cm} por encima de su
punto más bajo se encuentra en la vertical con un clavo que bloquea la
mitad superior del hilo. Halla la rapidez abajo y la subida más allá del
clavo; ¿qué cambia el clavo y qué no?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mechanical-energy:12}
Un patinador se deja caer en un \emph{half-pipe} desde \qty{2.5}{m};
cada travesía (de un lado al otro) disipa el $10\,\%$ de la \omterm{def:g11:mechanical-energy:em}{energía
mecánica}. Halla la rapidez al fondo del primer descenso y después,
calculando las alturas sucesivas, la primera travesía que termine por
debajo de \qty{1.0}{m}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mechanical-energy:13}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mechanical-energy:14}

\section{Problema: diseñar el rizo}

\begin{problem}[{Problema de fin de semana --- auditoría energética de
una montaña rusa: la altura de lanzamiento que impone un rizo, las
velocidades que promete, el rozamiento que confiesan dos sensores y los
frenos que cierran la cuenta --- todo el viaje colgando de un solo
número}]
\label{pb:g11:mechanical-energy:1}
Un tren de masa $m = \qty{2.0e3}{kg}$ se suelta desde el reposo en una
colina de lanzamiento de altura $H$ y recorre, sin motor, un rizo
circular vertical de radio $R = \qty{8.0}{m}$ cuya cima está a la altura
$2R$. El análisis del movimiento circular (del curso que viene) aporta
el único dato que tomamos prestado: en la cima del rizo el tren aprieta
sobre el carril solo si $v_{\text{top}}^2 \geq gR$. Las Partes I y II
ignoran el \omterm{def:g10:inertia:inventory}{rozamiento}.

\medskip
\noindent\textbf{Parte I --- La colina que impone el rizo.}
\begin{enumerate}
  \item Calcula la rapidez mínima de seguridad en la cima del rizo.
  \item Con el suelo como \omterm{def:g11:mechanical-energy:potential}{nivel de referencia}, calcula la \omterm{def:g11:mechanical-energy:em}{energía
        mecánica} del tren en la cima del rizo con esa rapidez mínima.
  \item Deduce $H_{\min}$ y demuestra que $H_{\min} = 5R/2$ --- sin masa
        y sin $g$.
  \item El equipo toma $H = \qty{24}{m}$: calcula la rapidez real en la
        cima del rizo y el cociente $v_{\text{top}}^2/(gR)$.
  \item ¿Adónde se fue la $m$? ¿Por qué le vale el diseño igual a los
        trenes llenos que a los vacíos?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Lo que promete la altura}
($H = \qty{24}{m}$).
\begin{enumerate}[resume]
  \item ¿Rapidez en la base del rizo ($z = 0$), en \unit{m/s} y en
        \unit{km/h}?
  \item ¿Rapidez en el costado del rizo ($z = R$)?
  \item Calcula $E_p$ y $E_k$ en el lanzamiento, en la base, en el
        costado y en la cima del rizo; comprueba que cada pareja suma
        $mgH$.
  \item Tras el rizo viene un lomo de \qty{12}{m}: ¿rapidez en su
        cresta?
  \item Da la fórmula general de $v(z)$; ¿dónde es más rápido el viaje y
        dónde más lento?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La auditoría del \omterm{def:g10:inertia:inventory}{rozamiento}.}
Dos sensores situados a la misma altura $z = \qty{2.0}{m}$ y separados
\qty{60}{m} de vía marcan $v_1 = \qty{20.4}{m/s}$ y después
$v_2 = \qty{19.2}{m/s}$.
\begin{enumerate}[resume]
  \item Calcula la variación de \omterm{def:g11:mechanical-energy:em}{energía mecánica} entre los sensores.
        ¿Qué tenía de astuto colocarlos a la misma altura?
  \item Deduce el \omterm{def:g11:work-of-force:work}{trabajo} del \omterm{def:g10:inertia:inventory}{rozamiento} y la \omterm{def:g10:inertia:force}{fuerza} media de
        \omterm{def:g10:inertia:inventory}{rozamiento}.
  \item ¿Adónde ha ido la \omterm{def:g10:energy-conservation:energy}{energía} que falta? Nombra la forma y los
        lugares.
  \item ¿Qué fracción de la $E_m$ del tren se pierde cada \qty{60}{m}?
        ¿Por qué sería peligroso el $H_{\min}$ pelado en el mundo real,
        que roza?
  \item Unos \qty{90}{m} de vía separan el lanzamiento de la cima del
        rizo: estima $E_m$ allí; ¿sigue salvando el rizo el margen de la
        pregunta 4?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Cerrar la cuenta.}
El último sensor, a la entrada del tramo recto y horizontal de frenado
($z = 0$), marca \qty{18.0}{m/s}.
\begin{enumerate}[resume]
  \item Calcula allí la \omterm{def:g11:mechanical-energy:kinetic}{energía cinética} del tren.
  \item Los frenos tienen que pararlo en $d = \qty{45}{m}$: calcula la
        \omterm{def:g10:inertia:force}{fuerza} de frenado (constante) necesaria.
  \item Comprobación de confort: calcula la deceleración $a = F/m$;
        compárala con $g$.
  \item ¿Cuánta \omterm{def:g10:energy-conservation:energy}{energía} disipó el \omterm{def:g10:inertia:inventory}{rozamiento} de la vía en todo el
        recorrido? Comprueba que \omterm{def:g10:inertia:inventory}{rozamiento} $+$ frenos $= mgH$, \omterm{def:g11:work-of-force:work}{julio} a
        \omterm{def:g11:work-of-force:work}{julio}.
  \item El remate, en una frase: ¿qué único número fijó la seguridad del
        rizo, todas las velocidades, la factura del \omterm{def:g10:inertia:inventory}{rozamiento} y la
        \omterm{def:g10:inertia:force}{fuerza} de los frenos --- y cuál fue el precio de ignorar el
        \omterm{def:g10:inertia:inventory}{rozamiento}?
\end{enumerate}
\end{problem}
