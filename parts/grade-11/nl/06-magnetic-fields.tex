\chapter{Magnetisme en magnetische velden}\label{ch:g11:magnetic-fields}

Zeelieden voeren duizend jaar lang op gemagnetiseerde naalden voordat
iemand wist waarom die naar het noorden wijzen. Het antwoord kwam in
1820, toen een stroom een kompas deed trillen: magnetisme wordt gemaakt
door bewegende ladingen. Dit hoofdstuk brengt het magnetische \omterm{def:g11:electric-gravitational-fields:field}{veld} in
kaart --- van magneten, van de Aarde, van stromen --- en sluit dan de
kring: het \omterm{def:g11:electric-gravitational-fields:field}{veld} duwt terug op stromen, en zo draait elke elektromotor.

\section{Magneten en het magnetische veld}

\begin{definition}[Magneet en polen]\label{def:g11:magnetic-fields:magnet}
Een \emph{magneet}\index{magneet} trekt ijzer aan en draait, aan een
draad opgehangen, met één uiteinde naar het noorden. Haar werking
concentreert zich in twee \emph{magnetische polen}\index{magnetische
pool}, de \emph{noordpool} (het uiteinde dat het geografische noorden
opzoekt) en de \emph{zuidpool}: gelijke polen stoten elkaar af,
ongelijke trekken elkaar aan. Een magneet doormidden zagen levert twee
volledige magneten op --- geen enkele proef heeft ooit één pool
geïsoleerd.
\end{definition}

\begin{definition}[Magnetisch veld]\label{def:g11:magnetic-fields:bfield}
Een \omterm{def:g11:magnetic-fields:magnet}{magneet} verandert de ruimte om zich heen
(\cref{ch:g11:electric-gravitational-fields}): ze wekt overal een
\emph{magnetisch veld}\index{magnetisch veld} $\vect B$ op. Een kompas
leest de richting ervan af (de naald richt zich naar $\vect B$, met de
noordpool voorop); een \emph{hallsonde}\index{hallsonde} meet de grootte,
in \emph{tesla}\index{tesla} (\unit{T}).
\end{definition}

\begin{definition}[Magnetische veldlijnen]\label{def:g11:magnetic-fields:lines}
\emph{Magnetische veldlijnen}\index{magnetische veldlijn} zijn krommen
die raken aan $\vect B$ en dichter opeen liggen waar het \omterm{def:g11:electric-gravitational-fields:field}{veld} sterk is.
Buiten een \omterm{def:g11:magnetic-fields:magnet}{magneet} lopen ze van de noordpool naar de zuidpool; anders dan
elektrische \omterm{def:g11:electric-gravitational-fields:lines}{veldlijnen} hebben ze geen uiteinden: elke lijn sluit zich
door het lichaam van de \omterm{def:g11:magnetic-fields:magnet}{magneet} heen.
\end{definition}

\section{De Aarde is een magneet}

\begin{definition}[Het magnetische veld van de Aarde]\label{def:g11:magnetic-fields:earth}
Stromen in de gesmolten ijzeren kern van de Aarde maken van de planeet
een reusachtige \omterm{def:g11:magnetic-fields:magnet}{magneet}: aan het oppervlak is
$B \approx \qty{50}{\micro T}$, en een vrij draaiende naald richt zich
naar de horizontale component van het \omterm{def:g11:electric-gravitational-fields:field}{veld} --- dat is het kompas. De
geomagnetische polen zijn niet de geografische; de hoek tussen het
kompasnoorden en het ware noorden is de
\emph{declinatie}\index{declinatie}, die op elke koers wordt
gecorrigeerd.
\end{definition}

\begin{example}[Een koers corrigeren]\label{ex:g11:magnetic-fields:bearing}
Waar de \omterm{def:g11:magnetic-fields:earth}{declinatie} $\ang{7}$ west bedraagt, drijft een schip dat
\qty{5.0}{km} lang op ``kompasnoord'' vaart
$5000 \times \sin\ang{7} \approx \qty{6.1e2}{m}$ ten westen van zijn
koerslijn af: om werkelijk pal noord te varen, stuur je $\ang{7}$
\emph{oostelijker} dan de naald.
\end{example}

\section{Het magnetische veld van een stroom}

\begin{remark}[Oersted, 1820]\label{rem:g11:magnetic-fields:oersted}
Tijdens een college in 1820 zag Hans Christian Oersted een kompasnaald
uitslaan toen hij vlakbij een schakeling sloot: elektriciteit en
magnetisme waren één onderwerp --- \emph{stromen wekken magnetische
\omterm{def:g11:electric-gravitational-fields:field}{velden} op}.
\end{remark}

\begin{proposition}[Veld van een rechte draad]\label{prop:g11:magnetic-fields:wire}
Een lange rechte draad met stroomsterkte $I$ wekt een \omterm{def:g11:electric-gravitational-fields:field}{veld} op waarvan de
lijnen cirkels zijn rond de draad, in vlakken loodrecht erop. Op afstand
$d$ van de draad geldt
\[
B = \frac{\mu_0\, I}{2\pi d},
\qquad \mu_0 = 4\pi\times 10^{-7}\ \unit{T.m/A} \approx \qty{1.26e-6}{T.m/A}.
\]
\end{proposition}
\admitted

\begin{example}[Een draad is een zwakke magneet]\label{ex:g11:magnetic-fields:wire}
Op $d = \qty{1.0}{cm}$ van een draad met $I = \qty{15}{A}$:
$B = \num{2e-7} \times 15 / 0.010 = \qty{0.30}{mT}$ --- zes aardvelden,
uit een stroom die een huisinstallatie al zou uitschakelen. Sterke \omterm{def:g11:electric-gravitational-fields:field}{velden}
vragen om een betere meetkunde.
\end{example}

\begin{definition}[Spoel en solenoïde]\label{def:g11:magnetic-fields:solenoid}
De draad opwinden concentreert haar \omterm{def:g11:electric-gravitational-fields:field}{veld}. Een \emph{vlakke
spoel}\index{vlakke spoel} ($N$ windingen in een schijf) gedraagt zich
als een \omterm{def:g11:lenses-and-eye:lens}{dunne} \omterm{def:g11:magnetic-fields:magnet}{magneet}, met de ene kant noord en de andere zuid; een
\emph{solenoïde}\index{solenoïde} --- een lange cilindrische wikkeling
met $n$ windingen per \omterm{def:g10:orders-of-magnitude:unit}{meter} --- als een staafmagneet.
\end{definition}

\begin{proposition}[Veld in een solenoïde]\label{prop:g11:magnetic-fields:solenoidfield}
Binnen in een lange \omterm{def:g11:magnetic-fields:solenoid}{solenoïde}, ver van de uiteinden, is het \omterm{def:g11:electric-gravitational-fields:field}{veld}
\emph{homogeen}: evenwijdig aan de as en in elk inwendig punt hetzelfde,
met grootte $B = \mu_0\, n\, I$, onafhankelijk van de wijdte van de
boring. Buiten de \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} is het \omterm{def:g11:electric-gravitational-fields:field}{veld} zwak.
\end{proposition}
\admitted

\begin{remark}
Beide formules worden eerlijk afgeleid in het volume van bachelorjaar 1,
uit de wet die de circulatie van een \omterm{def:g11:electric-gravitational-fields:field}{veld} verbindt met de stroom die het
omsluit. Let op wat telt: $n$, windingen \emph{per \omterm{def:g10:orders-of-magnitude:unit}{meter}} --- wikkel
dichter, niet langer.
\end{remark}

\begin{method}[De rechterhand, twee keer]\label{met:g11:magnetic-fields:righthand}
\begin{itemize}
  \item \emph{Draad}: duim $=$ stroom; gekromde vingers $=$ draaizin van
        de lijnen.
  \item \emph{Spoel, \omterm{def:g11:magnetic-fields:solenoid}{solenoïde}}: gekromde vingers $=$ stroom in de
        windingen; duim $= \vect B$ binnenin, wijzend naar de noordkant.
\end{itemize}
\end{method}

\begin{example}[Getallen bij een solenoïde]\label{ex:g11:magnetic-fields:solenoid}
Bij $10$ windingen per centimeter ($n = \qty{1000}{m^{-1}}$) en
$I = \qty{5.0}{A}$: $B = \num{1.26e-6} \times 1000 \times 5.0 \approx
\qty{6.3}{mT}$ --- honderd aardvelden, met een knop instelbaar.
\end{example}

\section{Elektromagneten}

\begin{definition}[Elektromagneet]\label{def:g11:magnetic-fields:electromagnet}
Een \emph{elektromagneet}\index{elektromagneet} is een \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} die om
een kern van zacht ijzer is gewikkeld: het ijzer magnetiseert in het \omterm{def:g11:electric-gravitational-fields:field}{veld}
van de spoel en vermenigvuldigt het, doorgaans met $100$ tot $1000$.
Anders dan een permanente \omterm{def:g11:magnetic-fields:magnet}{magneet} schakelt hij met de stroom uit.
\end{definition}

\begin{remark}[Drie machines]\label{rem:g11:magnetic-fields:machines}
De kraan op het schrootterrein tilt een auto op met een \omterm{def:g11:magnetic-fields:electromagnet}{elektromagneet} en
--- dat is de hele bedoeling --- \emph{laat hem vallen} door een
schakelaar te openen. De kleine \omterm{def:g11:magnetic-fields:electromagnet}{elektromagneet} van een relais trekt een
contact dicht: een zwakke stroom schakelt een sterke
(\cref{ch:g11:circuits-and-power}). Een mri-scanner houdt met een
supergeleidende \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} een homogene \qty{1.5}{T} rond de patiënt in
stand.
\end{remark}

\section{De lorentzkracht op een stroomdraad}

\begin{proposition}[Lorentzkracht op een stroomdraad]\label{prop:g11:magnetic-fields:laplace}
Een rechte draad met lengte $L$ die een stroom $I$ voert in een \omterm{def:g11:electric-gravitational-fields:capacitor}{homogeen
veld} $\vect B$ \emph{loodrecht} op de draad, voelt de
\emph{lorentzkracht}\index{lorentzkracht} met grootte $F = I\,L\,B$,
loodrecht op zowel de draad als $\vect B$: vlakke rechterhand, duim langs
de stroom, gestrekte vingers langs $\vect B$ --- de handpalm duwt langs
$\vect F$. Een draad \emph{evenwijdig} aan het \omterm{def:g11:electric-gravitational-fields:field}{veld} voelt geen \omterm{def:g10:inertia:force}{kracht}.
\end{proposition}
\admitted

\begin{remark}[De springende staaf]\label{rem:g11:magnetic-fields:rail}
Het volume van bachelorjaar 1 leidt dit af uit de magnetische \omterm{def:g10:inertia:force}{kracht} op
elke bewegende lading. In het practicum: een koperen \omterm{def:g11:color-light-sources:cones}{staafje} ligt over
twee horizontale rails tussen de polen van een \omterm{def:g11:magnetic-fields:magnet}{magneet} (in de figuur
wijst het \omterm{def:g11:electric-gravitational-fields:field}{veld} het blad in); sluit de schakelaar en het \omterm{def:g11:color-light-sources:cones}{staafje} schiet
langs de rails weg. Draai de stroom om, of de \omterm{def:g11:magnetic-fields:magnet}{magneet}: het schiet de
andere kant op.
\end{remark}

\begin{example}[Getallen bij de lorentzkracht]\label{ex:g11:magnetic-fields:laplace}
Een \omterm{def:g11:color-light-sources:cones}{staafje} met lengte $L = \qty{5.0}{cm}$ dat $I = \qty{8.0}{A}$ voert
in $B = \qty{0.50}{T}$:
$F = 8.0 \times 0.050 \times 0.50 = \qty{0.20}{N}$ --- het \omterm{def:g10:universal-gravitation:weight}{gewicht} van
\qty{20}{g}, op een draad die een paar gram weegt.
\end{example}

\begin{remark}[De gelijkstroommotor]\label{rem:g11:magnetic-fields:motor}
Buig de draad tot een lus in het \omterm{def:g11:electric-gravitational-fields:field}{veld}: de twee zijden loodrecht op
$\vect B$ voeren tegengestelde stromen, dus zijn hun \omterm{prop:g11:magnetic-fields:laplace}{lorentzkrachten}
tegengesteld --- een koppel dat de lus laat draaien. Een halve slag later
zou dat koppel haar terugdraaien, dus keert een draaiende schakelaar (de
\emph{commutator}\index{commutator}) de stroom elke halve slag om. Elke
ventilator en elke boormachine is deze lus, in het groot.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:1}
Een staafmagneet wordt tussen zijn polen doormidden gezaagd. Welke polen
draagt elk stuk? Trekken de pas gezaagde vlakken elkaar aan of stoten ze
elkaar af als je ze weer bij elkaar brengt? Bestaat er een manier van
zagen die de noordpool isoleert?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:2}
Juist of onjuist, met telkens één reden: (a) twee \omterm{def:g11:magnetic-fields:lines}{magnetische veldlijnen}
snijden elkaar waar het \omterm{def:g11:electric-gravitational-fields:field}{veld} sterk is; (b) buiten een \omterm{def:g11:magnetic-fields:magnet}{magneet} lopen de
\omterm{def:g11:electric-gravitational-fields:lines}{veldlijnen} van de noordpool naar de zuidpool; (c) elke \omterm{def:g11:magnetic-fields:lines}{magnetische
veldlijn} is een gesloten lus; (d) een naald richt zich loodrecht op de
lijn die er doorheen loopt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:3}
Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld} op \qty{2.0}{cm} van een rechte draad met \qty{10}{A}.
Hoeveel keer de \qty{50}{\micro T} van de Aarde is dat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:4}
Een \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} met $800$ windingen over \qty{40}{cm} voert
$I = \qty{1.5}{A}$. Bereken $n$ en daarna $B$ binnenin; wat wordt $B$ als
de stroom verdubbelt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:5}
Een draadstuk van \qty{10}{cm} voert \qty{5.0}{A} loodrecht op een \omterm{def:g11:electric-gravitational-fields:field}{veld}
van \qty{0.20}{T}. Bereken de \omterm{prop:g11:magnetic-fields:laplace}{lorentzkracht}; aan het \omterm{def:g10:universal-gravitation:weight}{gewicht} van welke
massa is die gelijk?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:6}
Waar een wandelaarster staat, is de \omterm{def:g11:magnetic-fields:earth}{declinatie} $\ang{8}$ west. Ze loopt
\qty{2.0}{km} ``noord volgens kompas'': hoever, en naar welke kant van
het ware noorden, drijft ze af? Welke koers zou haar pal noord hebben
gebracht?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:7}
Op welke afstand van een rechte draad met \qty{20}{A} is haar \omterm{def:g11:electric-gravitational-fields:field}{veld} gelijk
aan de \qty{50}{\micro T} van de Aarde? Wat zegt dat over het vertrouwen
in een kompas vlak bij kabels onder \omterm{def:g11:circuits-and-power:voltage}{spanning}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:8}
Ontwerp een \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} die $B = \qty{10}{mT}$ levert bij
$I = \qty{4.0}{A}$: bereken de vereiste $n$ en daarna het aantal
windingen voor een spoel van \qty{25}{cm}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:9}
In de railproef voert het \omterm{def:g11:color-light-sources:cones}{staafje} (massa \qty{20}{g}, lengte
$L = \qty{12}{cm}$ tussen de rails) \qty{10}{A} in een loodrecht \omterm{def:g11:electric-gravitational-fields:field}{veld} van
\qty{0.50}{T}. Bereken de \omterm{prop:g11:magnetic-fields:laplace}{lorentzkracht} en daarna de beginversnelling van
het \omterm{def:g11:color-light-sources:cones}{staafje}; vergelijk met $g$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:10}
Een horizontale draad met massa \qty{5.0}{g} en lengte \qty{20}{cm} ligt
in een horizontaal \omterm{def:g11:electric-gravitational-fields:field}{veld} $B = \qty{0.10}{T}$ dat er loodrecht op staat.
Bij welke stroom houdt de \omterm{prop:g11:magnetic-fields:laplace}{lorentzkracht} het \omterm{def:g10:universal-gravitation:weight}{gewicht} van de draad in
\omterm{def:g10:inertia:compensate}{evenwicht}, zodat hij zweeft? Welke kant moet de \omterm{def:g10:inertia:force}{kracht} op wijzen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:magnetic-fields:11}
Geef de richting van de \omterm{prop:g11:magnetic-fields:laplace}{lorentzkracht} (of zeg dat ze verdwijnt):
(a) stroom naar rechts op het blad, $\vect B$ het blad in; (b) stroom
naar de bovenkant van het blad, $\vect B$ uit het blad; (c) stroom
evenwijdig aan $\vect B$. Gebruik de vlakke rechterhand.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:12}
Een horizontale draad loopt geografisch noord--zuid, \qty{3.0}{cm} boven
een kompas; de horizontale component van het aardveld is daar
\qty{20}{\micro T}. De draad voert $I = \qty{5.0}{A}$: bereken haar \omterm{def:g11:electric-gravitational-fields:field}{veld}
bij het kompas, geef de richting ervan, en daarna de uitslag van de naald
(\omterm{def:g11:electric-gravitational-fields:field}{velden} tellen op als vectoren).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:13}
Een \omterm{def:g11:magnetic-fields:electromagnet}{elektromagneet} op een schrootterrein is een spoel van $400$
windingen over \qty{20}{cm}, met \omterm{def:g11:circuits-and-power:resistance}{weerstand} \qty{2.0}{\ohm}, op een
voeding van \qty{12}{V}; zijn ijzeren kern vermenigvuldigt het kale \omterm{def:g11:electric-gravitational-fields:field}{veld}
met $100$. Bereken de stroom, het \omterm{def:g11:electric-gravitational-fields:field}{veld} van de kale spoel, het \omterm{def:g11:electric-gravitational-fields:field}{veld} met de
kern, en het \omterm{def:g10:energy-conservation:power}{vermogen} dat de spoel omzet
(\cref{ch:g11:circuits-and-power}). Waarom valt het schroot op het moment
dat de schakelaar opengaat?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:14}
De rechthoekige lus van een motor heeft $N = 50$ windingen met
$I = \qty{2.0}{A}$; haar twee zijden van \qty{5.0}{cm} staan loodrecht op
een \omterm{def:g11:electric-gravitational-fields:field}{veld} $B = \qty{0.30}{T}$. Bereken de \omterm{def:g10:inertia:force}{kracht} op elk van die zijden;
waarom zijn de twee \omterm{def:g10:inertia:force}{krachten} tegengesteld, en wat doen ze met de lus?
Waarom voelen de twee andere zijden soms geen \omterm{def:g10:inertia:force}{kracht}? Waarom moet de
stroom elke halve slag worden omgekeerd?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:magnetic-fields:15}
Hoe dicht bij een kabel met \qty{100}{A} zou je moeten komen om haar \omterm{def:g11:electric-gravitational-fields:field}{veld}
\qty{1}{T} te laten bereiken? Vergelijk met de straal van een millimeter
van de kabel en besluit waarom \omterm{def:g11:electric-gravitational-fields:field}{velden} van een \omterm{def:g11:magnetic-fields:bfield}{tesla} uit \omterm{def:g11:magnetic-fields:solenoid}{solenoïden} worden
opgebouwd en niet uit losse draden. En daarna: hoeveel machten van tien
ligt het oppervlakteveld van \qty{e8}{T} van een neutronenster boven dat
van de Aarde?
\end{exercise}

\section{Probleem: Het kompas, de kraan en de mri}

\begin{problem}[{Weekendopgave --- drie magneten aan het werk: een
kompas dat een oceaan oversteekt, een elektromagneet die een auto optilt
en een mri-solenoïde --- één veld, over tien machten van tien, dat drie
klussen klaart}]
\label{pb:g11:magnetic-fields:1}
Dezelfde vector $\vect B$ stuurt een zeilboot met \qty{50}{\micro T},
tilt schroot op met \qty{1}{T} en brengt een knie in beeld met
\qty{1.5}{T} --- drie machines voor de drie formules van dit hoofdstuk.

\medskip
\noindent\textbf{Deel I --- Het kompas.}
Op de positie van het schip heeft het aardveld een horizontale component
van \qty{20}{\micro T} en een verticale van \qty{44}{\micro T}; de
\omterm{def:g11:magnetic-fields:earth}{declinatie} is $\ang{6}$ west.
\begin{enumerate}
  \item Waarom wijst een kompasnaald überhaupt naar het (magnetische)
        noorden?
  \item Bereken de grootte van het totale \omterm{def:g11:electric-gravitational-fields:field}{veld} en zijn hoek met de
        horizontaal.
  \item Als het schip \qty{30}{km} lang op ``kompasnoord'' vaart, hoever
        ten westen van de ware noordkoers komt het dan uit?
  \item Stalen lading doet de naald nog eens $\ang{4}$ westwaarts
        uitslaan: dezelfde vraag.
  \item Vlak bij de geomagnetische pool is een kompas onbruikbaar: welke
        component is daar de schuldige, en waarom?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De kraan.}
De \omterm{def:g11:magnetic-fields:electromagnet}{elektromagneet} op het schrootterrein is een \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} van $600$
windingen over \qty{30}{cm}, met \omterm{def:g11:circuits-and-power:resistance}{weerstand} \qty{1.5}{\ohm}, gevoed door
een bron van \qty{12}{V}; zijn ijzeren kern vermenigvuldigt het kale \omterm{def:g11:electric-gravitational-fields:field}{veld}
met $50$.
\begin{enumerate}[resume]
  \item Bereken $n$, de windingdichtheid.
  \item Bereken de stroom in de spoel
        (\cref{ch:g11:circuits-and-power}).
  \item Bereken het kale \omterm{def:g11:electric-gravitational-fields:field}{veld} $\mu_0 n I$ van de spoel.
  \item Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld} met de kern; vergelijk met de \qty{0.5}{T} van
        neodymium.
  \item Bereken het \omterm{def:g10:energy-conservation:power}{vermogen} dat de spoel omzet, en daarna de \omterm{def:g10:energy-conservation:energy}{energie}
        voor een hijsdienst van \qty{10}{\minute}.
  \item Een permanente \omterm{def:g11:magnetic-fields:magnet}{magneet} die zo sterk is, bestaat. Waarom dan toch
        een \omterm{def:g11:magnetic-fields:electromagnet}{elektromagneet}?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De liermotor.}
De lier van de kraan is een gelijkstroommotor: een rechthoekige lus van
$N = 100$ windingen, met zijden van \qty{4.0}{cm} loodrecht op
$B = \qty{0.25}{T}$, die $I = \qty{3.0}{A}$ voert.
\begin{enumerate}[resume]
  \item Bereken de \omterm{prop:g11:magnetic-fields:laplace}{lorentzkracht} op elke bundel zijden die loodrecht op
        $\vect B$ staat.
  \item De twee \omterm{def:g10:inertia:force}{krachten} zijn even groot en tegengesteld, en toch heffen
        hun gevolgen elkaar niet op. Wat brengen ze voort, en waarom
        moeten ze \emph{tegengesteld} zijn?
  \item Welke \omterm{def:g10:inertia:force}{kracht} werkt er op de twee andere zijden wanneer die
        evenwijdig aan $\vect B$ staan?
  \item Na een halve slag zouden dezelfde \omterm{def:g10:inertia:force}{krachten} de draaiing weer
        ongedaan maken. Wat doet de \omterm{rem:g11:magnetic-fields:motor}{commutator}, en wanneer?
  \item Geef drie afzonderlijke ontwerpwijzigingen die elk het
        krachtenpaar verdubbelen.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De mri.}
De \omterm{def:g11:magnetic-fields:solenoid}{solenoïde} van de scanner houdt een homogene \qty{1.5}{T} in stand met
een stroom $I = \qty{500}{A}$.
\begin{enumerate}[resume]
  \item Hoeveel keer het aardveld is \qty{1.5}{T}?
  \item Bereken de windingdichtheid $n$ die daarvoor \emph{zonder}
        ijzeren kern nodig is.
  \item Als de wikkeling een \omterm{def:g11:circuits-and-power:resistance}{weerstand} van \qty{0.20}{\ohm} had, welk
        \omterm{def:g10:energy-conservation:power}{vermogen} zou ze dan omzetten? Welke eigenschap van de echte
        wikkeling voorkomt dat, tegen de prijs van extreme kou?
  \item De clou: plaats de \omterm{def:g11:electric-gravitational-fields:field}{velden} van het schip, de kraan en de mri op de
        ladder van machten van tien, van de interstellaire ruimte
        (\qty{e-10}{T}) tot een neutronenster (\qty{e8}{T}); wat is er
        tussen de machines veranderd --- de natuurkunde, of de \omterm{def:g11:circuits-and-power:current}{ampères}?
\end{enumerate}
\end{problem}
