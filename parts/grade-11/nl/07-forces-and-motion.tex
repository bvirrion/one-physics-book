\chapter{Krachten en beweging}\label{ch:g11:forces-and-motion}

Gooi een moersleutel tollend door de kamer: de uiteinden zwabberen
wild, en toch beschrijft één punt erin een volmaakte parabool.
\Cref{ch:g10:inertia} regelde wat er gebeurt wanneer \omterm{def:g10:inertia:force}{krachten} in
\omterm{def:g10:inertia:compensate}{evenwicht} zijn --- niets. Dit hoofdstuk vraagt wat er gebeurt wanneer ze
dat niet zijn: de \omterm{def:g10:inertia:force}{kracht} die overblijft \emph{verandert de snelheid} ---
ze versnelt lichamen, vertraagt ze, doet ze afbuigen --- en zegt met
hoeveel, in verhoudingen zo niet in formules.

\section{Het massamiddelpunt}

\begin{definition}[Massamiddelpunt]\label{def:g11:forces-and-motion:com}
Het \emph{massamiddelpunt}\index{massamiddelpunt} van een lichaam of een
stelsel is zijn evenwichtspunt --- de gemiddelde ligging van zijn massa:
het meetkundige midden van een homogeen symmetrisch lichaam; voor twee
puntmassa's het punt van het verbindingsstuk waarvan de afstanden zich
verhouden als de \emph{omgekeerde} verhouding van de massa's,
$\frac{d_1}{d_2} = \frac{m_2}{m_1}$ --- dus dichter bij de zwaarste.
\end{definition}

\begin{example}[Een scheve halter]\label{ex:g11:forces-and-motion:dumbbell}
Bollen van \qty{2.0}{kg} en \qty{1.0}{kg} zitten aan de uiteinden van een
lichte staaf van \qty{0.90}{m}. Het \omterm{def:g11:forces-and-motion:com}{massamiddelpunt} verdeelt de staaf in
de omgekeerde verhouding $d_1/d_2 = 1.0/2.0$: op $\qty{0.30}{m}$ van de
zware bol en $\qty{0.60}{m}$ van de lichte --- daar houd je de staaf in
\omterm{def:g10:inertia:compensate}{evenwicht}.
\end{example}

\begin{remark}[De geworpen moersleutel]\label{rem:g11:forces-and-motion:wrench}
Film een tollende moersleutel: elk punt volgt een ingewikkelde lussende
kromme --- behalve het \omterm{def:g11:forces-and-motion:com}{massamiddelpunt}, dat de nette parabool van een
geworpen bal beschrijft. De beweging van het \omterm{def:g11:forces-and-motion:com}{massamiddelpunt} is
eenvoudig; de rest is draaiing \emph{eromheen} --- en daarom mag de
mechanica een auto, een planeet of een turnster als één punt behandelen.
\end{remark}

\section{Een resulterende kracht verandert de snelheid}

\begin{definition}[Resulterende kracht]\label{def:g11:forces-and-motion:netforce}
De \emph{resulterende kracht}\index{resulterende kracht} op een lichaam
is de vectorsom van alle \omterm{def:g10:inertia:force}{krachten} die erop werken,
$\vect F = \vect F_1 + \vect F_2 + \dots$ (kop aan staart, zoals in het
wiskundevolume); de \omterm{def:g10:inertia:force}{krachten} zijn precies dan in \omterm{def:g10:inertia:compensate}{evenwicht} wanneer
$\vect F = \vect 0$.
\end{definition}

\begin{theorem}[Werking van een resulterende kracht --- half kwantitatief]\label{thm:g11:forces-and-motion:secondlaw}
\index{tweede bewegingswet}
Een \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} $\vect F$ die gedurende een korte tijd $\Delta t$
op een lichaam met massa $m$ werkt, verandert de snelheid van zijn
\omterm{def:g11:forces-and-motion:com}{massamiddelpunt} met een vector $\Delta \vect v$ die de \emph{richting}
van $\vect F$ heeft, en waarvan de grootte evenredig groeit met $F$ en
met $\Delta t$ en omgekeerd evenredig krimpt met de massa:
\[
\Delta v \propto \frac{F \, \Delta t}{m} .
\]
\end{theorem}
\admitted

\begin{remark}
Verdubbel de \omterm{def:g10:inertia:force}{kracht} of haar duur: twee keer zoveel verandering;
verdubbel de massa: de helft. Volgend jaar wordt dit tot een exacte
vergelijking aangescherpt, en het volume van bachelorjaar 1 vraagt in
welke stelsels ze geldt; dit jaar volstaan verhoudingen --- daarmee
werken airbags, remmen en omloopbanen al.
\end{remark}

\begin{proposition}[De drie standaardgevallen]\label{prop:g11:forces-and-motion:cases}
Een lichaam beweegt met snelheid $\vect v$ onder een \omterm{def:g11:forces-and-motion:netforce}{resulterende
kracht} $\vect F$.
\begin{enumerate}
  \item $\vect F$ langs $\vect v$: het \emph{versnelt} langs een rechte
        lijn;
  \item $\vect F$ tegengesteld aan $\vect v$: het \emph{vertraagt} langs
        een rechte lijn;
  \item $\vect F$ loodrecht op $\vect v$: het \emph{buigt af} zonder van
        snelheid te veranderen.
\end{enumerate}
\end{proposition}

\begin{proof}
Tel $\Delta\vect v$, gericht langs $\vect F$
(\cref{thm:g11:forces-and-motion:secondlaw}), kop aan staart bij
$\vect v$ op: langs $\vect v$ stapelen de pijlen zich op (langer);
tegengesteld heffen ze elkaar deels op (korter); loodrecht kantelt elke
kleine $\Delta \vect v$ de pijl zonder hem te verlengen --- alleen de
richting draait.
\end{proof}

\begin{example}[Een tik van opzij]\label{ex:g11:forces-and-motion:kick}
Een puck glijdt met \qty{6.0}{m/s} naar het oosten; een korte klap van
opzij voegt $\Delta\vect v = \qty{8.0}{m/s}$ pal naar het noorden toe. De
nieuwe snelheid is de vectorsom: grootte
$\sqrt{6.0^2 + 8.0^2} = \qty{10.0}{m/s}$, richting
$\tan\theta = 8.0/6.0$, dus $\theta \approx 53^\circ$ ten noorden van het
oosten --- de oude snelheid wordt niet gewist, de verandering wordt
erbij opgeteld.
\end{example}

\section{Veranderingen vergelijken: kracht, tijd, massa}

\begin{method}[Redeneren met verhoudingen]\label{met:g11:forces-and-motion:ratio}
Om twee situaties te vergelijken los je nooit een vergelijking op maar
vorm je verhoudingen: de verhouding van de snelheidsveranderingen is de
verhouding van de $F$'s, maal die van de $\Delta t$'s, gedeeld door die
van de $m$'s. Lees daarna af welke factor is veranderd, en met hoeveel.
\end{method}

\begin{example}[Het beladen karretje]\label{ex:g11:forces-and-motion:cart}
Dezelfde duw, even lang volgehouden, stuurt een lege boodschappenwagen
van \qty{20}{kg} weg met \qty{1.2}{m/s}. Beladen tot \qty{60}{kg} ---
drie keer de massa, dezelfde $F\Delta t$ --- vertrekt hij met een derde
daarvan, \qty{0.40}{m/s}. Massa is de koppigheid van materie.
\end{example}

\begin{remark}[Botsingen, airbags, gebogen knieën]\label{rem:g11:forces-and-motion:airbag}
Bij een botsing zijn $\Delta v$ en $m$ niet onderhandelbaar --- de
snelheid van de inzittende moet nu eenmaal naar nul --- dus ligt
$F\,\Delta t$ vast. De enige overgebleven vrijheid is de \emph{tijd}: rek
$\Delta t$ op en de \omterm{def:g10:inertia:force}{kracht} krimpt in dezelfde verhouding. Een hoofd dat
in \qty{0.08}{s} op een airbag tot stilstand komt in plaats van in
\qty{0.01}{s} op het stuur, voelt een acht keer kleinere gemiddelde
\omterm{def:g10:inertia:force}{kracht}; kreukelzones, helmschuim en gebogen knieën halen dezelfde truc
uit.
\end{remark}

\section{Wrijving: de kracht die het glijden tegenwerkt}

\begin{definition}[Statische en kinetische wrijving]\label{def:g11:forces-and-motion:friction}
\omterm{def:g10:inertia:inventory}{Wrijving} tussen oppervlakken die elkaar raken, werkt hun \emph{onderling
glijden} tegen. De \emph{statische wrijving}\index{wrijving!statische}
werkt zolang ze niet glijden en stelt zichzelf tot een maximum in om op
te heffen wat het glijden probeert te beginnen; de \emph{kinetische
wrijving}\index{wrijving!kinetische}, zodra het glijden begonnen is, is
ruwweg constant, gericht tegen het glijden in, en meestal \emph{zwakker}
dan het statische maximum.
\end{definition}

\begin{example}[De kast verschuiven]\label{ex:g11:forces-and-motion:wardrobe}
Duw zachtjes tegen een kast: hij beweegt niet --- de \omterm{def:g11:forces-and-motion:friction}{statische wrijving}
heft je \qty{150}{N} op. Duw harder: bij zo'n \qty{400}{N} schiet hij in
beweging en schuift daarna met minder moeite door --- de zwakkere
\omterm{def:g11:forces-and-motion:friction}{kinetische wrijving} heeft het overgenomen.
\end{example}

\begin{remark}[Waarom abs werkt]\label{rem:g11:forces-and-motion:abs}
Het contactpunt van een rollend wiel glijdt niet over de weg: de band
grijpt met \emph{statische} \omterm{def:g10:inertia:inventory}{wrijving}. Een geblokkeerd wiel slipt en
krijgt alleen de zwakkere \emph{kinetische} \omterm{def:g10:inertia:inventory}{wrijving}, die tegen de slip
in wijst wat de wielen ook doen --- sturen valt weg. Abs laat de rem los
op het ogenblik dat een wiel blokkeert: meer grip, kortere remweg, en een
bestuurder die kan sturen.
\end{remark}

\section{Afbuigen: een cirkelbeweging vraagt een kracht naar binnen}

\begin{definition}[Eenparige cirkelbeweging]\label{def:g11:forces-and-motion:ucm}
Een lichaam voert een \emph{eenparige cirkelbeweging}\index{eenparige
cirkelbeweging} uit wanneer zijn \omterm{def:g10:relative-motion:trajectory}{baan} een cirkel is die met constante
snelheid wordt doorlopen. De snelheid, rakend aan de cirkel, verandert op
elk ogenblik van richting --- ze is dus nooit constant.
\end{definition}

\begin{proposition}[De kracht naar binnen]\label{prop:g11:forces-and-motion:inward}
Op een lichaam in \omterm{def:g11:forces-and-motion:ucm}{eenparige cirkelbeweging} werkt een \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht}
die nooit nul is: op elk ogenblik wijst ze van het lichaam naar het
middelpunt van de cirkel.
\end{proposition}

\begin{proof}
De snelheid is constant in grootte, dus heeft de \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} geen
deel langs $\vect v$ (gevallen 1 en 2 van
\cref{prop:g11:forces-and-motion:cases}): ze staat loodrecht op de
raaklijn en ligt dus langs de straal --- en wel naar binnen, want elke
$\Delta \vect v$ buigt de \omterm{def:g10:relative-motion:trajectory}{baan} naar het middelpunt toe.
\end{proof}

\begin{example}[Het touw, de bocht, de Maan]\label{ex:g11:forces-and-motion:moon}
Achter elke gelijkmatige bocht schuilt een \omterm{def:g10:inertia:force}{kracht} naar binnen. De
rondgeslingerde bal: de \omterm{def:g10:inertia:inventory}{spankracht} van het touw. De auto in de bocht: de
zijwaartse \omterm{def:g11:forces-and-motion:friction}{statische wrijving} van de weg op de banden --- weg op ijs, en
de auto gaat rechtdoor. De Maan: de gravitatietrek van de Aarde
(\cref{ch:g10:universal-gravitation}), die haar in \num{27.3} dagen
rondvoert --- niet \emph{omhoog} gehouden maar zijwaarts in een
eeuwigdurende bocht getrokken, om de Aarde heen vallend; volgend jaar
wordt dat de mechanica van \omterm{def:g10:universal-gravitation:satellite}{satellieten}.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:1}
\omterm{def:g11:forces-and-motion:netforce}{Resulterende kracht} nul of niet nul? Verantwoord dat vanuit de snelheid:
(a) een auto die rechtdoor met een constante \qty{90}{km/h} rijdt;
(b) een auto die rechtlijnig remt; (c) een auto die met een constante
\qty{50}{km/h} een bocht neemt; (d) een parachutespringer op
eindsnelheid.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:2}
Bollen van \qty{3.0}{kg} en \qty{1.0}{kg} zitten aan de uiteinden van een
lichte staaf van \qty{0.80}{m}. Bepaal het \omterm{def:g11:forces-and-motion:com}{massamiddelpunt}; en waar ligt
het bij gelijke massa's?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:3}
Dezelfde korte duw wordt gegeven aan een lege bagagekar van \qty{15}{kg}
en aan dezelfde kar beladen tot \qty{45}{kg}. Vergelijk de
snelheidsveranderingen; welke duw zou de beladen kar de $\Delta v$ van de
lege geven?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:4}
Een tram die met \qty{10}{m/s} rijdt, remt; een seconde later rijdt hij
met \qty{8}{m/s} in dezelfde richting. Geef $\Delta \vect v$ en de
richting van de \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht}. Welk geval uit
\cref{prop:g11:forces-and-motion:cases} is dit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:5}
Statische of \omterm{def:g11:forces-and-motion:friction}{kinetische wrijving} --- en welke kant op gericht? (a) Een
krat dat je (te zachte) duw weerstaat; (b) hetzelfde krat zodra het
schuift; (c) een boek op een licht gekanteld dienblad dat niet
wegglijdt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:6}
Een \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} $F$ die gedurende $\Delta t$ op een massa $m$
werkt, levert $\Delta v = \qty{4.0}{m/s}$ op. Voorspel $\Delta v$ voor:
(a) $2F, \Delta t, m$; (b) $F, \Delta t/2, m$; (c) $F, \Delta t, 2m$;
(d) $3F, 2\Delta t, 6m$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:7}
Een puck glijdt met \qty{8.0}{m/s} naar het oosten; een klap voegt
$\Delta \vect v = \qty{6.0}{m/s}$ pal naar het noorden toe. Bereken de
nieuwe snelheid en richting. Waarom is de snelheid niet
$8.0 + 6.0 = \qty{14}{m/s}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:8}
Een keepster stopt hetzelfde schot op twee manieren: met stijve armen (de
bal staat in \qty{0.02}{s} stil) of door met de bal ``mee te geven''
(\qty{0.08}{s}). Vergelijk de gemiddelde \omterm{def:g10:inertia:force}{krachten}, en zeg welke
grootheden al vastlagen voordat ze koos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:9}
Welke uitwendige \omterm{def:g10:inertia:force}{kracht} duwt een sprinter bij de start vooruit? Waarom
helpen spikes --- en waarom kan niemand versnellen, of zelfs maar lopen,
op volmaakt wrijvingsloos ijs?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:10}
Waarom staat een auto met abs (wielen die blijven rollen) bij een
noodstop meestal eerder stil dan een auto met geblokkeerde, slippende
wielen --- en waarom valt de slippende auto niet meer te sturen? Verwijs
naar \cref{def:g11:forces-and-motion:friction}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:11}
Een kogelslingeraar slingert de kogel aan haar draad rond. Welke \omterm{def:g10:inertia:force}{kracht}
laat de kogel afbuigen? Schets zijn \omterm{def:g10:relative-motion:trajectory}{baan} na het loslaten. Waarom heeft de
Maan geen draad nodig, en wat zou haar \omterm{def:g10:relative-motion:trajectory}{baan} zonder zwaartekracht worden?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:12}
Een bezem is een homogene stok van \qty{1.0}{kg} (eigen \omterm{def:g11:forces-and-motion:com}{massamiddelpunt}
op \qty{0.60}{m} van de bovenkant) met een kop van \qty{2.0}{kg} waarvan
het \omterm{def:g11:forces-and-motion:com}{massamiddelpunt} \qty{1.20}{m} van de bovenkant ligt. Behandel elk
deel als een puntmassa, bepaal het \omterm{def:g11:forces-and-motion:com}{massamiddelpunt} van de bezem en toets
de regel van de omgekeerde verhouding uit
\cref{def:g11:forces-and-motion:com}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:13}
Een auto met massa $m$ en een beladen bestelwagen met massa $2m$ remmen
vanaf dezelfde snelheid met dezelfde remkracht.
\begin{enumerate}
  \item Vergelijk de tijden die ze nodig hebben om stil te staan.
  \item Elk verliest gelijkmatig snelheid en rijdt dus met dezelfde
        \emph{gemiddelde} snelheid: vergelijk de remwegen.
  \item De auto remt nu vanaf de \emph{dubbele} snelheid, met dezelfde
        \omterm{def:g10:inertia:force}{kracht}: vergelijk tijd en weg met zijn eerste stop. Welke moraal
        volgt daaruit voor snelheidslimieten?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:14}
De Maan draait op $R \approx \qty{3.84e8}{m}$ in $T = \num{27.3}$ dagen
om de Aarde.
\begin{enumerate}
  \item Bereken haar baansnelheid (omtrek gedeeld door \omterm{def:g10:signals-and-waves:period}{periode}).
  \item Welke kant op wijst de verandering van haar snelheid, en welke
        \omterm{def:g10:inertia:force}{kracht} levert die?
  \item Verbeter met \cref{prop:g11:forces-and-motion:inward} de uitspraak
        ``de Maan valt niet omdat ze snel genoeg beweegt''.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:15}
Een ei van \qty{50}{g} raakt de vloer met \qty{4.0}{m/s}: op tegels staat
het in ongeveer \qty{1}{ms} stil, op dik schuim in ongeveer \qty{50}{ms}.
Vergelijk de gemiddelde \omterm{def:g10:inertia:force}{krachten}. Leg met dezelfde redenering uit waarom
je bij het landen na een sprong je knieën buigt, en waarvan de valmat van
een turner in wezen is gemaakt.
\end{exercise}

\section{Probleem: De botsproef}

\begin{problem}[{Weekendopgave --- de zachte botsing ontwerpen: omdat een
botsing de snelheidsverandering vastlegt, is elke veiligheidsvoorziening
in een auto een machine om tijd op te rekken}]
\label{pb:g11:forces-and-motion:1}
Een botsproeflaboratorium rijdt een auto met \qty{50}{km/h} tegen een
starre muur, met een volwassen dummy van \qty{70}{kg} en een kinderdummy
van \qty{20}{kg} aan boord; camera's klokken elke stop. Jouw taak:
uitzoeken waar het geweld van een botsing schuilt, en dat wegontwerpen.

\medskip
\noindent\textbf{Deel I --- De maat van een botsing.}
\begin{enumerate}
  \item Zet \qty{50}{km/h} om naar \unit{m/s}.
  \item Bij elke frontale stop heeft de snelheidsverandering van de
        inzittenden dezelfde grootte. Welke? Waarom kan geen enkele
        voorziening haar veranderen?
  \item Verwijs naar \cref{thm:g11:forces-and-motion:secondlaw}: met vaste
        $\Delta v$ en $m$ ligt $F\,\Delta t$ vast. Aan welke ene knop kan
        de ontwerper dan nog draaien?
  \item De muur brengt de auto in \qty{0.050}{s} tot stilstand; een
        kreukelzone rekt dat op tot \qty{0.150}{s}. Vergelijk de
        gemiddelde \omterm{def:g10:inertia:force}{krachten}.
  \item Welke kant op wijzen $\Delta \vect v$ en de \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} op
        de dummy's tijdens de stop?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De kreukelzone.}
\begin{enumerate}[resume]
  \item Waarom ontwerpen ingenieurs, in één zin, de voorkant van de auto
        zo dat die vernield wordt?
  \item In een oude, stijve auto (stop in \qty{0.050}{s}) is de dummy
        star aan de stoel gegespt. Vergelijk haar \omterm{def:g10:inertia:force}{kracht} met die bij de
        moderne stop van \qty{0.150}{s}.
  \item De snelheid van de stoppende auto zakt gelijkmatig (gemiddeld: de
        helft van de beginwaarde). Hoever komt hij in \qty{0.150}{s}?
        Vergelijk met een echte kreukelzone.
  \item Waarom geen kreukelzone van \qty{10}{m}? En welk deel van de auto
        mag juist \emph{niet} kreukelen?
  \item Vat deel II samen als een leus: ``je kunt $\Delta v$ niet kiezen,
        dus moet je \dots\ kiezen''.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De gordel, de airbag en het kind.}
\begin{enumerate}[resume]
  \item Een dummy zonder gordel houdt zijn \qty{14}{m/s}
        (\cref{ch:g10:inertia}) tot de voorruit hem in \qty{0.005}{s}
        stopt; een dummy met gordel stopt met de auto mee in
        \qty{0.150}{s}. Vergelijk de gemiddelde \omterm{def:g10:inertia:force}{krachten}.
  \item Waarom is een gordel die een beetje uitrekt beter dan een
        onrekbaar stalen harnas?
  \item Het hoofd zou in \qty{0.010}{s} op het stuur tot stilstand komen;
        de airbag rekt dat op tot \qty{0.080}{s}. Vergelijk de \omterm{def:g10:inertia:force}{krachten}.
  \item Volwassene en kind ondergaan dezelfde $\Delta v$ in dezelfde
        $\Delta t$. Vergelijk de \omterm{def:g11:forces-and-motion:netforce}{resulterende krachten} die de twee
        gordels moeten leveren.
  \item Een kind op schoot: de armen moeten de snelheid van een baby van
        \qty{10}{kg} in \qty{0.150}{s} met \qty{14}{m/s} veranderen,
        terwijl de zwaartekracht een snelheid elke seconde met
        \qty{9.8}{m/s} verandert. Hoeveel keer sneller moeten de armen
        werken --- en het \omterm{def:g10:universal-gravitation:weight}{gewicht} van welke massa zou zo zwaar aanvoelen?
        Trek je conclusie.
  \item Kinderen voelen kleinere \omterm{def:g10:inertia:force}{krachten} --- waarom dan toch
        kinderzitjes? Denk aan \emph{waar} de gordel zijn \omterm{def:g10:inertia:force}{kracht}
        aangrijpt, en aan $\Delta t$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De veiligheidskooi, en het vonnis.}
\begin{enumerate}[resume]
  \item De passagierscel is star terwijl beide uiteinden kreukelen.
        Waarom mag de cel niet vervormen, ook al betekent starheid
        gewelddadige stops?
  \item Twee identieke auto's botsen frontaal, elk met \qty{50}{km/h}.
        Waar komt elk tot stilstand? Vergelijk elke botsing met de
        muurproef.
  \item Een kleine auto botst frontaal op een beladen vrachtwagen. De
        \omterm{def:g10:inertia:force}{krachten} die ze op elkaar uitoefenen zijn even groot en
        tegengesteld, wederzijds zoals elke wisselwerking
        (\cref{ch:g11:fundamental-interactions}). Wiens snelheid
        verandert het meest?
  \item Noem de veiligheidsvoorzieningen uit deze opgave en de ene
        grootheid die ze alle oprekken.
  \item Slot: formuleer de ontwerpwet van de zachte botsing in één zin,
        met alleen \emph{massa}, \emph{snelheidsverandering},
        \emph{\omterm{def:g10:inertia:force}{kracht}} en \emph{tijd}.
\end{enumerate}
\end{problem}
