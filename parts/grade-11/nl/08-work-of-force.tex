\chapter{Arbeid van een kracht}\label{ch:g11:work-of-force}

Duw een kast door de kamer en je hebt \omterm{def:g11:work-of-force:work}{arbeid} verricht; duw een hele
middag tegen een muur en de natuurkunde telt, wat je spieren ook
melden, nul. \omterm{def:g11:work-of-force:work}{Arbeid} is wat een \omterm{def:g10:inertia:force}{kracht} \emph{langs een beweging} levert
--- \omterm{def:g10:energy-conservation:energy}{energie} onderweg (\cref{ch:g10:energy-conservation}), betaald in
\omterm{def:g11:work-of-force:work}{joule}. Dit hoofdstuk zet een prijs op de \omterm{def:g10:inertia:force}{krachten} van
\cref{ch:g11:forces-and-motion} --- trekken onder een hoek, \omterm{def:g10:universal-gravitation:weight}{gewicht},
\omterm{def:g10:inertia:inventory}{wrijving} --- en klokt het betaaltempo: het \omterm{def:g11:work-of-force:power}{vermogen}.

\section{Arbeid van een constante kracht}

\begin{definition}[Arbeid van een constante kracht]\label{def:g11:work-of-force:work}
Een constante \omterm{def:g10:inertia:force}{kracht} $\vect F$ die aangrijpt op een voorwerp dat langs
een recht stuk van $A$ naar $B$ beweegt, verricht de
\emph{arbeid}\index{arbeid}
\[
W_{AB}(\vect F) = F \times AB \times \cos\theta ,
\]
waarin $F$ de grootte van de \omterm{def:g10:inertia:force}{kracht} is (in \unit{N}), $AB$ de lengte van
de verplaatsing (in \unit{m}) en $\theta$ de hoek tussen $\vect F$ en
$\vect{AB}$. Arbeid wordt gemeten in \emph{joule}\index{joule}
(\unit{J}): één joule is de arbeid van een \omterm{def:g10:inertia:force}{kracht} van één \omterm{def:g10:inertia:force}{newton} over
één \omterm{def:g10:orders-of-magnitude:unit}{meter} in haar eigen richting.
\end{definition}

\begin{remark}\label{rem:g11:work-of-force:scal}
$F \times AB \times \cos\theta$ is precies het inwendig product
$\vect F \cdot \vect{AB}$ uit het wiskundevolume van dit jaar. Alleen de
component $F\cos\theta$ \emph{langs} de beweging levert iets op.
\end{remark}

\begin{example}[Koffer aan een riem]\label{ex:g11:work-of-force:suitcase}
Een reiziger sleept een koffer \qty{300}{m} door een vertrekhal, met de
riem onder $\theta = \qty{50}{\degree}$ met de vloer getrokken met
$F = \qty{40}{N}$: $W = 40 \times 300 \times \cos\qty{50}{\degree}
\approx \qty{7.7}{kJ}$. Vlak getrokken ($\theta = 0$) zou dezelfde \omterm{def:g10:inertia:force}{kracht}
\qty{12}{kJ} leveren: een derde van de trek tilt niets op.
\end{example}

\begin{definition}[Drijvende, remmende en nul-arbeid]\label{def:g11:work-of-force:sign}
Het teken van $\cos\theta$ deelt elke \omterm{def:g10:inertia:force}{kracht} in drie regimes in:
\begin{itemize}
  \item $\theta < \qty{90}{\degree}$: $W > 0$, de \omterm{def:g11:work-of-force:work}{arbeid} is
        \emph{drijvend}\index{drijvende arbeid} --- de \omterm{def:g10:inertia:force}{kracht} voert
        \omterm{def:g10:energy-conservation:energy}{energie} aan de beweging toe;
  \item $\theta = \qty{90}{\degree}$: $W = 0$, de \omterm{def:g10:inertia:force}{kracht} verricht in het
        geheel geen \omterm{def:g11:work-of-force:work}{arbeid}, hoe groot ze ook is;
  \item $\theta > \qty{90}{\degree}$: $W < 0$, de \omterm{def:g11:work-of-force:work}{arbeid} is
        \emph{remmend}\index{remmende arbeid} --- de \omterm{def:g10:inertia:force}{kracht} tapt \omterm{def:g10:energy-conservation:energy}{energie}
        af. Het zuivere geval is een rem bij $\qty{180}{\degree}$:
        \qty{800}{N} over \qty{45}{m} haalt
        $800 \times 45 \times (-1) = \qty{-36}{kJ}$ uit een auto, en
        stopt die in hete remschijven.
\end{itemize}
\end{definition}

\section{De arbeid van het gewicht}

\begin{proposition}[Arbeid van het gewicht]\label{prop:g11:work-of-force:weight}
Wanneer een voorwerp met massa $m$ van $A$ (hoogte $z_A$) naar $B$
(hoogte $z_B$) beweegt langs \emph{welke} weg van rechte stukken dan ook,
verricht zijn \omterm{def:g10:universal-gravitation:weight}{gewicht} $\vect P$ de \omterm{def:g11:work-of-force:work}{arbeid}
\[
W_{AB}(\vect P) = mg\,(z_A - z_B) = \pm\, mgh ,
\]
waarin $h = \abs{z_A - z_B}$ het hoogteverschil is: $+mgh$ omlaag,
$-mgh$ omhoog --- alleen de daling telt, niet de route.
\end{proposition}

\begin{proof}
Op één recht stuk $AB$ maakt het \omterm{def:g10:universal-gravitation:weight}{gewicht} (grootte $mg$, recht omlaag) een
hoek $\theta$ met $\vect{AB}$, en $AB\cos\theta$ is precies de projectie
van de verplaatsing op de neerwaartse verticaal: $z_A - z_B$. Dus
$W = mg \times AB\cos\theta = mg\,(z_A - z_B)$. Langs een keten van
stukken tellen de \omterm{def:g11:work-of-force:work}{arbeiden} op en vallen de dalingen tegen elkaar weg:
$mg(z_A - z_C) + mg(z_C - z_B) = mg(z_A - z_B)$.
\end{proof}

\begin{proposition}[Elke weg]\label{prop:g11:work-of-force:anypath}
Dezelfde formule geldt langs elke gekromde weg van $A$ naar $B$.
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:work-of-force:anypath}
Aannemelijk --- een kromme ligt zo dicht als we willen bij een gebroken
lijn van vele korte stukken --- maar dat ``zo dicht als we willen''
eerlijk maken is analyse, en die volgt in
\cref{ch:g12:work-and-energy}.
\end{remark}

\begin{example}[Wandelaar]\label{ex:g11:work-of-force:hiker}
Een wandelaar van \qty{78}{kg} (rugzak inbegrepen) klimt \qty{450}{m},
langs welk pad ook:
$W(\vect P) = -78 \times 9.81 \times 450 \approx \qty{-344}{kJ}$, en
precies $+\qty{344}{kJ}$ op de terugweg omlaag, hoeveel omwegen hij ook
maakt.
\end{example}

\section{De arbeid van de wrijving}

\begin{proposition}[Arbeid van glijdende wrijving]\label{prop:g11:work-of-force:friction}
Een wrijvingskracht met constante grootte $f$, op elk ogenblik
tegengesteld aan de beweging, verricht over een weg met totale lengte $L$
de \omterm{def:g11:work-of-force:work}{arbeid} $W = -fL$: altijd \omterm{def:g11:work-of-force:sign}{remmend}, en evenredig met de \emph{lengte van
de afgelegde weg} --- niet met de verplaatsing.
\end{proposition}

\begin{proof}
Op elk kort recht stukje staat de \omterm{def:g10:inertia:force}{kracht} onder $\qty{180}{\degree}$ met
de beweging en levert ze $-f \times (\text{de lengte ervan})$; de stukjes
tellen op tot $-fL$. Een gekromde weg wordt op dezelfde manier in korte
stukjes gehakt, met de limiet aangenomen zoals bij het \omterm{def:g10:universal-gravitation:weight}{gewicht}
(\cref{rem:g11:work-of-force:anypath}).
\end{proof}

\begin{remark}[Wrijving tegen gewicht: de grote scheidslijn]\label{rem:g11:work-of-force:divide}
De \omterm{def:g11:work-of-force:work}{arbeid} van het \omterm{def:g10:universal-gravitation:weight}{gewicht} vergeet de weg; die van de \omterm{def:g10:inertia:inventory}{wrijving} is tol per
\omterm{def:g10:orders-of-magnitude:unit}{meter}. Op een rondreis levert het \omterm{def:g10:universal-gravitation:weight}{gewicht} netto nul op --- wat het
bergopwaarts vraagt, geeft het bergafwaarts terug --- terwijl de \omterm{def:g10:inertia:inventory}{wrijving}
op elk stuk $-fL$ int en niets teruggeeft: de \omterm{def:g11:work-of-force:work}{arbeid} van het \omterm{def:g10:universal-gravitation:weight}{gewicht}
laat zich opslaan en terugwinnen (\omterm{def:g10:energy-conservation:potential}{potentiële energie},
\cref{ch:g11:mechanical-energy}), die van de \omterm{def:g10:inertia:inventory}{wrijving} gaat als warmte
verloren.
\end{remark}

\begin{example}[Heen en terug]\label{ex:g11:work-of-force:roundtrip}
Een krat schuift \qty{5.0}{m} over een vloer en \qty{5.0}{m} terug, tegen
$f = \qty{60}{N}$ in: de \omterm{def:g10:inertia:inventory}{wrijving} verricht
$-60 \times 10 = \qty{-600}{J}$, ook al eindigt het krat waar het
begon. \omterm{def:g10:universal-gravitation:weight}{Gewicht} en \omterm{def:g10:inertia:inventory}{normaalkracht} staan loodrecht op de beweging: overal
nul.
\end{example}

\begin{remark}[En een veer?]\label{rem:g11:work-of-force:spring}
Een veer trekt harder naarmate ze verder is uitgerekt: haar \omterm{def:g10:inertia:force}{kracht} is
niet constant en \cref{def:g11:work-of-force:work} is niet van
toepassing. Haar \omterm{def:g11:work-of-force:work}{arbeid} betekent de uitrekking in oneindig kleine stapjes
opdelen en optellen --- analyse, die in \cref{ch:g12:work-and-energy}
wordt geleverd.
\end{remark}

\section{Vermogen}

\begin{definition}[Vermogen]\label{def:g11:work-of-force:power}
Het \emph{vermogen}\index{vermogen!mechanisch} van een \omterm{def:g10:inertia:force}{kracht} is het
tempo waarin ze \omterm{def:g11:work-of-force:work}{arbeid} verricht, $P = W/\Delta t$, in
\emph{watt}\index{watt} ($\unit{W} = \unit{J/s}$) --- dezelfde watt als
in \cref{ch:g11:circuits-and-power}. Op motorplaatjes leeft een \omterm{def:g10:orders-of-magnitude:unit}{eenheid}
van vóór het SI voort: de \emph{paardenkracht}\index{paardenkracht},
$\qty{1}{hp} = \qty{736}{W}$.
\end{definition}

\begin{proposition}[Vermogen bij constante snelheid]\label{prop:g11:work-of-force:pfv}
Beweegt het voorwerp met constante snelheid $v$, dan levert een constante
\omterm{def:g10:inertia:force}{kracht} $\vect F$ onder een hoek $\theta$ met de beweging het \omterm{def:g11:work-of-force:power}{vermogen}
$P = F v \cos\theta$ --- en in het bijzonder $P = Fv$ voor een \omterm{def:g10:inertia:force}{kracht}
langs de beweging.
\end{proposition}

\begin{proof}
In een tijd $\Delta t$ legt het voorwerp $d = v\,\Delta t$ af, dus
$W = F \times v\,\Delta t \times \cos\theta$; deel door $\Delta t$.
\end{proof}

\begin{example}[Hijskraan]\label{ex:g11:work-of-force:crane}
Een kraan hijst een pallet van \qty{600}{kg} met een constante
\qty{0.90}{m/s}: de \omterm{def:g10:inertia:force}{kracht} in de kabel is gelijk aan het \omterm{def:g10:universal-gravitation:weight}{gewicht},
$mg = \qty{5.89e3}{N}$, dus
$P = Fv = 5890 \times 0.90 \approx \qty{5.3}{kW} \approx \qty{7.2}{hp}$
--- zeven paarden in een stalen kast.
\end{example}

\begin{method}[Klimmen met minder kracht: de zigzag]\label{met:g11:work-of-force:zigzag}
Om een last met een beperkte \omterm{def:g10:inertia:force}{kracht} een hoogte $h$ op te tillen:
\begin{enumerate}
  \item De rekening ligt vast: volgens
        \cref{prop:g11:work-of-force:weight} kost elke route naar boven
        de \omterm{def:g11:work-of-force:work}{arbeid} $mgh$.
  \item Rek de weg op: op een helling met lengte $L$ is de benodigde
        \omterm{def:g10:inertia:force}{kracht} (bij constante snelheid, \omterm{def:g10:inertia:inventory}{wrijving} daargelaten)
        $F = mgh/L$ --- verdubbel de lengte en de \omterm{def:g10:inertia:force}{kracht} halveert.
  \item Betaal de toeslag: de \omterm{def:g10:inertia:inventory}{wrijving} rekent per \omterm{def:g10:orders-of-magnitude:unit}{meter}
        (\cref{prop:g11:work-of-force:friction}) en voegt $fL$ toe.
\end{enumerate}
Bergwegen zigzaggen om stap 2: dezelfde hoogte, minder \omterm{def:g10:inertia:force}{kracht}, dezelfde
\omterm{def:g11:work-of-force:work}{arbeid}.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:1}
Een constante \omterm{def:g10:inertia:force}{kracht} $F = \qty{40}{N}$ werkt over een rechte
verplaatsing van \qty{50}{m}: bereken haar \omterm{def:g11:work-of-force:work}{arbeid} voor
$\theta = \qty{0}{\degree}$, $\qty{60}{\degree}$, $\qty{90}{\degree}$ en
$\qty{120}{\degree}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:2}
Een gereedschapskist van \qty{12}{kg} wordt \qty{2.5}{m} uit een
bestelwagen omlaag gelaten en later weer omhoog getild. Welke \omterm{def:g11:work-of-force:work}{arbeid}
verricht haar \omterm{def:g10:universal-gravitation:weight}{gewicht} in beide gevallen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:3}
\omterm{def:g11:work-of-force:sign}{Drijvend}, \omterm{def:g11:work-of-force:sign}{remmend} of nul? (a) het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een vallende appel; (b) het
\omterm{def:g10:universal-gravitation:weight}{gewicht} van een stijgende bal; (c) de \omterm{def:g10:inertia:inventory}{normaalkracht} op een schuivende
doos; (d) de \omterm{def:g10:inertia:inventory}{wrijving} op een remmende fiets; (e) de \omterm{def:g10:inertia:inventory}{spankracht} van het
touw op een slingerende slingerbol.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:4}
Een kraan tilt \qty{600}{kg} \qty{25}{m} omhoog met constante snelheid.
Welke \omterm{def:g11:work-of-force:work}{arbeid} verrichten de hijskracht en het \omterm{def:g10:universal-gravitation:weight}{gewicht}? Wat is hun som, en
waarom?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:5}
De motor van een roltrap verricht elke minuut \qty{90}{kJ} \omterm{def:g11:work-of-force:work}{arbeid}. Wat is
zijn \omterm{def:g11:work-of-force:power}{vermogen} in \omterm{def:g11:work-of-force:power}{watt}? En in \omterm{def:g11:work-of-force:power}{paardenkracht}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:6}
Een slee wordt \qty{200}{m} voortgetrokken door een touw met \omterm{def:g10:inertia:inventory}{spankracht}
\qty{120}{N} onder $\qty{30}{\degree}$ boven de sneeuw; de \omterm{def:g10:inertia:inventory}{wrijving} is
\qty{45}{N}. Welke \omterm{def:g11:work-of-force:work}{arbeid} verricht elk van de vier \omterm{def:g10:inertia:force}{krachten} (touw,
\omterm{def:g10:inertia:inventory}{wrijving}, \omterm{def:g10:universal-gravitation:weight}{gewicht}, \omterm{def:g10:inertia:inventory}{normaalkracht}), en hoeveel is het totaal?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:7}
Een fietser rijdt met een constante \qty{25}{km/h} tegen een totale
\omterm{def:g11:circuits-and-power:resistance}{weerstand} van \qty{18}{N} in. Welk \omterm{def:g11:work-of-force:power}{vermogen} levert hij? Welke \omterm{def:g11:work-of-force:work}{arbeid}
verricht hij in een uur?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:8}
Een skiër van \qty{65}{kg} daalt \qty{120}{m} in hoogte, via een steile

\omterm{def:g10:inertia:inventory}{wrijving} op de ski's is in beide gevallen \qty{30}{N}. Welke \omterm{def:g11:work-of-force:work}{arbeid}
verrichten het \omterm{def:g10:universal-gravitation:weight}{gewicht} en de \omterm{def:g10:inertia:inventory}{wrijving} op elke route? Welke \omterm{def:g11:work-of-force:work}{arbeid} trekt
zich iets van de route aan?
\end{exercise}

\section{Probleem: De dag van de verhuizer}

\begin{problem}[{Weekendopgave --- de natuurkunde betaalt per joule: een
dag van schuiven, hijsen en haarspeldbochten, met als slot de
ontnuchterende ontdekking van wat een waterkoker van eerlijke arbeid
vindt}]
\label{pb:g11:work-of-force:1}
Een verhuisploeg leegt een loods, sleept de lading naar een woning op de
derde verdieping en rijdt de wagen daarna naar een dorp op een heuvel.
Elke klus wordt in \omterm{def:g11:work-of-force:work}{joule} geprijsd; de waterkoker in de keuken controleert
de dag. Dozen: \qty{30}{kg}.

\medskip
\noindent\textbf{Deel I --- Over de vloer van de loods.}
Een doos wordt met een horizontale \omterm{def:g10:inertia:force}{kracht} $F = \qty{110}{N}$ over
$d = \qty{8.0}{m}$ geduwd; de glijdende \omterm{def:g10:inertia:inventory}{wrijving} is $f = \qty{95}{N}$.
\begin{enumerate}
  \item Bereken de \omterm{def:g11:work-of-force:work}{arbeid} van de duw.
  \item Bereken de \omterm{def:g11:work-of-force:work}{arbeid} van de \omterm{def:g10:inertia:inventory}{wrijving}, van het \omterm{def:g10:universal-gravitation:weight}{gewicht} en van de
        \omterm{def:g10:inertia:inventory}{normaalkracht} (verantwoord de nullen).
  \item Hoe groot is de totale \omterm{def:g11:work-of-force:work}{arbeid} op de doos, en wat betekent het
        teken ervan (de doos begon in rust)? Welke duw houdt de snelheid
        constant (\cref{ch:g11:forces-and-motion})?
  \item De verste hoek is bereikbaar over de diagonaal van \qty{10.0}{m}
        of langs twee wanden (\qty{6.0}{m} en daarna \qty{8.0}{m}).
        Hoeveel \omterm{def:g11:work-of-force:work}{arbeid} verricht de \omterm{def:g10:inertia:inventory}{wrijving} op elke route?
  \item Bereken op een heen-en-terugreis over de diagonaal de \omterm{def:g11:work-of-force:work}{arbeid} van
        de \omterm{def:g10:inertia:inventory}{wrijving} en die van het \omterm{def:g10:universal-gravitation:weight}{gewicht}. Welke \omterm{def:g10:inertia:force}{kracht} geeft terug?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Naar de derde verdieping
($h = \qty{9.0}{m}$).}
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:g11:work-of-force:work}{arbeid} van het \omterm{def:g10:universal-gravitation:weight}{gewicht} op een doos die naar de woning
        wordt getild. Welke \omterm{def:g11:work-of-force:work}{arbeid} moet de ploeg minstens leveren?
  \item Draagt een verhuizer van \qty{70}{kg} de doos de trap op, dan
        rekent de zwaartekracht voor verhuizer \emph{en} doos: hoeveel
        \omterm{def:g11:work-of-force:work}{arbeid} tegen de zwaartekracht in totaal, en welk deel gaat er
        werkelijk naar de doos?
  \item Een steekwagenhelling is \qty{30}{m} lang voor dezelfde
        \qty{9.0}{m} stijging, met \qty{40}{N} rolwrijving. Bereken bij
        constante snelheid de benodigde duw langs de helling en de \omterm{def:g11:work-of-force:work}{arbeid}
        daarvan.
  \item Vergelijk de helling met recht verticaal hijsen: benodigde \omterm{def:g10:inertia:force}{kracht}
        en verrichte \omterm{def:g11:work-of-force:work}{arbeid}. Wat levert de helling op, en tegen welke
        prijs?
  \item Een takel met touw en katrol tilt de doos met \qty{0.50}{m/s}
        omhoog. Hoe lang duurt dat, en hoe groot is het mechanische
        \omterm{def:g11:work-of-force:power}{vermogen}?
  \item Vind dat \omterm{def:g11:work-of-force:power}{vermogen} terug als $P = Fv$. De motor van de takel heeft
        een \omterm{def:g10:energy-conservation:efficiency}{rendement} van $60\%$: welk \omterm{prop:g11:circuits-and-power:power}{elektrisch vermogen} wordt
        opgenomen?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De zigzagweg.}
De beladen wagen (\qty{3500}{kg}) moet een dorp bereiken dat \qty{240}{m}
boven het dal ligt: over een rechte weg van \qty{1.0}{km} met $24\%$
stijging, of over een verharde weg van \qty{8.0}{km} met $3.0\%$
stijging (stijging $= \sin\alpha$).
\begin{enumerate}[resume]
  \item Ga na dat beide routes dezelfde \qty{240}{m} klimmen.
  \item Bereken bij constante snelheid en met verwaarloosbare \omterm{def:g10:inertia:inventory}{wrijving} de
        benodigde aandrijfkracht op elke route.
  \item Hoeveel \omterm{def:g11:work-of-force:work}{arbeid} verricht de aandrijfkracht op elke route?
        Vergelijk met $mgh$.
  \item De rolweerstand is op beide routes \qty{400}{N}: bereken de extra
        \omterm{def:g11:work-of-force:work}{arbeid} op elke route. Wat kost het gemak van de zigzag?
  \item Bereken bij \qty{36}{km/h} het motorvermogen dat op elke route
        nodig is (in \unit{kW} en \unit{hp}). Waarom zigzaggen wegen?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het grootboek van de dag.}
\begin{enumerate}[resume]
  \item De ploeg schuift $60$ dozen over de vloer (met de duw uit vraag
        1) en hijst alle $60$ over de \qty{9.0}{m} omhoog (vraag 6).
        Hoeveel mechanische \omterm{def:g11:work-of-force:work}{arbeid} is er in totaal aan de dozen
        geleverd?
  \item Tel het eigen lichaam van de trapdrager erbij: $20$ keer omhoog
        met \qty{70}{kg} (de teruggave bergafwaarts komt in hete knieën
        terecht en niet in het grootboek). Wat is het dagtotaal?
  \item De ploeg werkte $8.0$ uur: hoe groot is het gemiddelde
        mechanische \omterm{def:g11:work-of-force:power}{vermogen}? Hoe lang zou de waterkoker van \qty{2.0}{kW}
        erover doen om dezelfde \omterm{def:g10:energy-conservation:energy}{energie} uit te geven?
  \item Slot: zeg in twee zinnen waarvoor de \omterm{def:g11:work-of-force:work}{joule} betaalt --- en
        waarvoor niet (de muur waar je een middag lang tegen duwde, de
        doos die je roerloos vasthield) --- en waarom verhuizers eerder
        lieren, hellingen en zigzagwegen bezitten dan grotere spieren.
\end{enumerate}
\end{problem}
