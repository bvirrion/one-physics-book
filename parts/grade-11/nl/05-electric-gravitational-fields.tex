\chapter{Elektrische en gravitatievelden}\label{ch:g11:electric-gravitational-fields}

Hoe trekt de Aarde aan een appel die ze nooit aanraakt? Newton zelf noemde
werking op afstand ``zo grote onzin'' dat hij weigerde haar te
verdedigen. Het moderne antwoord: een massa, of een lading, verandert de
ruimte om zich heen --- ze wekt een \emph{\omterm{def:g11:electric-gravitational-fields:field}{veld}} op --- en wat je daar
neerzet reageert op het \omterm{def:g11:electric-gravitational-fields:field}{veld} op zijn eigen plaats. We bouwen de twee
grote voorbeelden naast elkaar op, tot de analogie een gelijkheid van
vorm wordt.

\section{Van werking op afstand naar het veld}

\begin{definition}[Veld]\label{def:g11:electric-gravitational-fields:field}
Een \emph{veld}\index{veld} kent aan elk punt van de ruimte een vector
toe. Een bron (een lading, een massa) wekt haar veld overal om zich heen
op, of er nu iets is dat het voelt of niet; een testvoorwerp voelt
vervolgens een \omterm{def:g10:inertia:force}{kracht} die alleen door het veld \emph{op zijn eigen
plaats} wordt bepaald. Dat beeld in twee stappen vervangt de werking op
afstand.
\end{definition}

\begin{remark}
Het \omterm{def:g11:electric-gravitational-fields:field}{veld} is geen boekhoudkundige truc: het draagt \omterm{def:g10:energy-conservation:energy}{energie}, en
veranderingen van de bron reizen er met eindige snelheid doorheen naar
buiten --- licht zelf is zo'n reizend \omterm{def:g11:electric-gravitational-fields:field}{veld}
(\cref{ch:g12:light-as-wave}). Hier bestuderen we \omterm{def:g11:electric-gravitational-fields:field}{velden} die niet met de
tijd veranderen.
\end{remark}

\section{Het elektrische veld}

\begin{definition}[Elektrisch veld]\label{def:g11:electric-gravitational-fields:efield}
Als een kleine testlading $q$ in een punt $M$ een elektrische \omterm{def:g10:inertia:force}{kracht}
$\vect F$ voelt, dan is het \emph{elektrische veld}\index{elektrisch veld}
in $M$
\[
\vect E = \frac{\vect F}{q}
\qquad \text{(in \unit{N/C}, ook geschreven als \unit{V/m})}.
\]
$\vect E$ hangt alleen van de bronnen en van $M$ af, niet van de
testlading: $q$ verdubbelen verdubbelt $\vect F$ en laat het quotiënt
onveranderd.
\end{definition}

\begin{proposition}[Kracht op een lading]\label{prop:g11:electric-gravitational-fields:force}
Een lading $q$ op een plaats waar het \omterm{def:g11:electric-gravitational-fields:field}{veld} $\vect E$ is, voelt de \omterm{def:g10:inertia:force}{kracht}
$\vect F = q \vect E$: langs $\vect E$ als $q > 0$, tegengesteld aan
$\vect E$ als $q < 0$, met grootte $F = \abs{q}\,E$.
\end{proposition}

\begin{proof}
Herschik de definitie; de regel voor het teken is de algebra van een
negatief veelvoud.
\end{proof}

\begin{example}[Een veld aflezen]\label{ex:g11:electric-gravitational-fields:reading}
Een testlading $q = \qty{+2.0e-8}{C}$ in $M$ voelt een \omterm{def:g10:inertia:force}{kracht} van
\qty{6.0e-4}{N} naar het oosten. Het \omterm{def:g11:electric-gravitational-fields:field}{veld} in $M$ is
$E = \num{6.0e-4} / \num{2.0e-8} = \qty{3.0e4}{N/C}$, naar het oosten
gericht; een lading $q' = \qty{-1.0e-8}{C}$ in hetzelfde punt voelt
$F = \num{1.0e-8} \times \num{3.0e4} = \qty{3.0e-4}{N}$ --- naar het
westen.
\end{example}

\begin{proposition}[Veld van een puntlading]\label{prop:g11:electric-gravitational-fields:point}
Een puntlading $Q$ wekt op afstand $d$ een \omterm{def:g11:electric-gravitational-fields:field}{veld} op langs de lijn van de
lading naar het punt --- van $Q$ af als $Q > 0$, ernaartoe als $Q < 0$
--- met grootte
\[
E = k\,\frac{\abs{Q}}{d^2},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{proposition}

\begin{proof}
De wet van Coulomb (\cref{ch:g11:fundamental-interactions}) geeft de
\omterm{def:g10:inertia:force}{kracht} op een testlading $q$ op afstand $d$: grootte
$k \abs{qQ} / d^2$, langs de verbindingslijn, afstotend bij gelijke
tekens. Deel door $q$ en pas
\cref{prop:g11:electric-gravitational-fields:force} toe.
\end{proof}

\begin{example}[Een geladen bol]\label{ex:g11:electric-gravitational-fields:sphere}
Een vandegraaffbol draagt $Q = \qty{+5.0e-7}{C}$ (een kleine geladen bol
werkt als een puntlading in haar middelpunt). Op $d = \qty{0.50}{m}$:
\[
E = \num{8.99e9} \times \num{5.0e-7} / 0.50^2 \approx \qty{1.8e4}{N/C},
\]
radiaal van de bol af gericht. Op \qty{1.0}{m} is het tot een kwart
daarvan gezakt: de $1/d^2$ van de \omterm{def:g10:inertia:force}{kracht} leeft voort in het \omterm{def:g11:electric-gravitational-fields:field}{veld}.
\end{example}

\begin{definition}[Veldlijnen]\label{def:g11:electric-gravitational-fields:lines}
Een \emph{veldlijn}\index{veldlijn} is een kromme die overal raakt aan
het \omterm{def:g11:electric-gravitational-fields:field}{veld}, met een pijl die de richting van het \omterm{def:g11:electric-gravitational-fields:field}{veld} aangeeft. Drie
regels:
\begin{itemize}
  \item ze vertrekken uit positieve ladingen en komen aan bij negatieve;
  \item waar de lijnen dicht opeen liggen is het \omterm{def:g11:electric-gravitational-fields:field}{veld} sterk, waar ze
        uitwaaieren zwak;
  \item twee lijnen snijden elkaar nooit: het \omterm{def:g11:electric-gravitational-fields:field}{veld} heeft in elk punt één
        richting.
\end{itemize}
\end{definition}

\section{Het homogene veld van een condensator}

\begin{definition}[Vlakkeplaatcondensator]\label{def:g11:electric-gravitational-fields:capacitor}
Twee evenwijdige metalen platen tegenover elkaar, dicht bij elkaar en met
tegengestelde ladingen, vormen een \emph{condensator}\index{condensator}.
Tussen de platen is het \omterm{def:g11:electric-gravitational-fields:field}{veld} een \emph{homogeen veld}\index{homogeen
veld}: overal even groot en in dezelfde richting --- loodrecht op de
platen, van de positieve plaat naar de negatieve.
\end{definition}

\begin{proposition}[Veld tussen de platen]\label{prop:g11:electric-gravitational-fields:uniform}
Is de \omterm{def:g11:circuits-and-power:voltage}{spanning} tussen de platen (\cref{ch:g11:circuits-and-power}) gelijk
aan $U$ en hun afstand $d$, dan heeft het \omterm{def:g11:electric-gravitational-fields:field}{veld} ertussen de grootte
\[
E = \frac{U}{d} \qquad \text{(in \unit{V/m})}.
\]
\end{proposition}
\admitted

\begin{remark}
De homogeniteit en $E = U/d$ worden eerlijk afgeleid in het volume van
bachelorjaar 1; de formule verklaart de \omterm{def:g10:orders-of-magnitude:unit}{eenheid} \unit{V/m}. \omterm{def:g11:lenses-and-eye:lens}{Dunne}
spleten geven sterke \omterm{def:g11:electric-gravitational-fields:field}{velden}: \qty{100}{V} over een millimeter is al
\qty{e5}{V/m}.
\end{remark}

\begin{example}[Een elektronenbundel sturen]\label{ex:g11:electric-gravitational-fields:beam}
In een \omterm{def:g10:signals-and-waves:oscilloscope}{oscilloscoop} gaat de bundel tussen platen door met
$U = \qty{100}{V}$ en $d = \qty{5.0}{mm}$:
$E = 100 / \num{5.0e-3} = \qty{2.0e4}{V/m}$, en elk elektron voelt
$F = \num{1.60e-19} \times \num{2.0e4} = \qty{3.2e-15}{N}$ naar de
positieve plaat --- $\num{4e14}$ keer zijn \omterm{def:g10:universal-gravitation:weight}{gewicht} van
\qty{8.9e-30}{N}, zodat de bundel zichtbaar wordt afgebogen (de gebogen
\omterm{def:g10:relative-motion:trajectory}{baan} is de zaak van \cref{ch:g12:newtons-laws}).
\end{example}

\section{Het gravitatieveld}

\begin{definition}[Gravitatieveld]\label{def:g11:electric-gravitational-fields:gfield}
Als een testmassa $m$ in een punt een \omterm{def:g10:universal-gravitation:force}{gravitatiekracht} (een \omterm{def:g10:universal-gravitation:weight}{gewicht})
$\vect P$ voelt, dan is het \emph{gravitatieveld}\index{gravitatieveld}
daar
\[
\vect g = \frac{\vect P}{m} \qquad \text{(in \unit{N/kg})},
\]
en omgekeerd $\vect P = m \vect g$. Een eerder hoofdstuk voerde het getal
$g$ in; $\vect g$ voegt daar de richting aan toe waarin een losgelaten
steen begint te bewegen.
\end{definition}

\begin{proposition}[Veld van de Aarde]\label{prop:g11:electric-gravitational-fields:earth}
Op afstand $d$ van het middelpunt van de Aarde (massa $M$) wijst het
\omterm{def:g11:electric-gravitational-fields:gfield}{gravitatieveld} naar het middelpunt en heeft het de grootte
\[
g = G\,\frac{M}{d^2},
\qquad G = \qty{6.67e-11}{N.m^2/kg^2}.
\]
\end{proposition}

\begin{proof}
De algemene \omterm{def:g11:fundamental-interactions:four}{gravitatie} (\cref{ch:g10:universal-gravitation}) geeft de
\omterm{def:g10:inertia:force}{kracht} op een testmassa $m$: grootte $G M m / d^2$, naar het middelpunt;
deel door $m$. Aan het oppervlak is $d = R$ en dit is de
$g = GM/R^2 \approx \qty{9.81}{N/kg}$ uit het eerdere hoofdstuk.
\end{proof}

\begin{example}[Het veld waar de Maan leeft]\label{ex:g11:electric-gravitational-fields:moon}
Op de afstand van de Maan, $d = \qty{3.84e8}{m}$:
\[
g = \num{6.67e-11} \times \num{5.97e24} / (\num{3.84e8})^2
\approx \qty{2.7e-3}{N/kg}.
\]
Het \omterm{def:g11:electric-gravitational-fields:field}{veld} van de Aarde houdt nooit op --- het wordt alleen ijler als
$1/d^2$; dit is het \omterm{def:g11:electric-gravitational-fields:field}{veld} dat de Maan in haar \omterm{def:g10:universal-gravitation:satellite}{omloopbaan} houdt.
\end{example}

\begin{remark}[Homogeen vlak bij het oppervlak]\label{rem:g11:electric-gravitational-fields:uniformg}
Over een practicumlokaal of zelfs een berg verandert $d$ nauwelijks
vergeleken met $R = \qty{6.37e6}{m}$, dus is $\vect g$ tot in hoge
nauwkeurigheid homogeen: verticale lijnen, een constante
\qty{9.81}{N/kg} --- het \omterm{def:g11:electric-gravitational-fields:field}{veld} van de Aarde lijkt op dat van de
\omterm{def:g11:electric-gravitational-fields:capacitor}{condensator}, net zoals de ronde Aarde vanuit een tuin vlak lijkt.
\end{remark}

\section{Eén idee, twee krachten}

De twee \omterm{def:g11:electric-gravitational-fields:field}{velden} van dit hoofdstuk zijn dezelfde wiskunde in twee
vermommingen:

\begin{center}
\begin{tabular}{lcc}
 & \textbf{elektrisch} & \textbf{gravitationeel} \\
\hline
bron & lading $Q$ & massa $M$ \\
\omterm{def:g11:electric-gravitational-fields:field}{veld} & $\vect E = \vect F / q$ & $\vect g = \vect P / m$ \\
\omterm{def:g10:orders-of-magnitude:unit}{eenheid} & \unit{N/C} = \unit{V/m} & \unit{N/kg} \\
puntbron & $E = k \abs{Q} / d^2$ & $g = G M / d^2$ \\
constante & $k = \qty{8.99e9}{N.m^2/C^2}$ & $G = \qty{6.67e-11}{N.m^2/kg^2}$ \\
\omterm{def:g10:inertia:force}{kracht} op een testvoorwerp & $\vect F = q \vect E$ & $\vect P = m \vect g$ \\
\omterm{def:g11:electric-gravitational-fields:lines}{veldlijnen} & uit $+$, naar $-$ & altijd naar de massa toe \\
\end{tabular}
\end{center}

Elke formule links wordt die rechts onder het woordenboek
$q \leftrightarrow m$, $\vect E \leftrightarrow \vect g$,
$k \leftrightarrow G$.

\begin{remark}[Waar de analogie breekt]\label{rem:g11:electric-gravitational-fields:noanalogue}
Lading bestaat in twee tekens, massa in één. \omterm{def:g11:electric-gravitational-fields:efield}{Elektrische velden} laten
zich dus opheffen: de herschikte ladingen van een metalen doos wissen
elk uitwendig \omterm{def:g11:electric-gravitational-fields:field}{veld} erbinnen uit (een \emph{kooi van Faraday} --- daarom
doodt een lift je telefoonsignaal), en gewone materie, met evenveel $+$
als $-$, is elektrisch stil. De zwaartekracht laat zich niet afschermen
en niet neutraliseren: elke \omterm{def:g10:orders-of-magnitude:unit}{kilogram} voegt zijn trek toe --- en daarom
regeert de \omterm{def:g10:inertia:force}{kracht} die $\num{e39}$ keer zwakker is
(\cref{exo:g11:electric-gravitational-fields:10}) het heelal op grote
schaal.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:1}
Een testlading $q = \qty{+2.0e-6}{C}$ in een punt $M$ voelt een \omterm{def:g10:inertia:force}{kracht}
van \qty{5.0e-3}{N} naar het noorden. Geef het \omterm{def:g11:electric-gravitational-fields:field}{veld} in $M$ (grootte en
richting), en daarna de \omterm{def:g10:inertia:force}{kracht} op $q' = \qty{-4.0e-6}{C}$ in $M$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:2}
Een puntlading $Q = \qty{+5.0}{nC}$ zit in $O$.
\begin{enumerate}
  \item Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld} op $d = \qty{30}{cm}$: grootte en richting.
  \item Op welke afstand is het \omterm{def:g11:electric-gravitational-fields:field}{veld} half zo sterk?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:3}
Een \omterm{def:g11:electric-gravitational-fields:capacitor}{condensator} staat onder $U = \qty{600}{V}$ met platen op
\qty{3.0}{mm} afstand. Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld} en daarna de \omterm{def:g10:inertia:force}{kracht} op één
elektron daarin ($e = \qty{1.60e-19}{C}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:4}
Op Mars weegt een sonde van \qty{250}{kg} \qty{930}{N}. Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld}
$g_{\text{Mars}}$ aan het marsoppervlak, en daarna het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een
astronaut van \qty{80}{kg} daar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:5}
Juist of onjuist, met telkens één reden: (a) twee \omterm{def:g11:electric-gravitational-fields:lines}{veldlijnen} mogen elkaar
snijden waar het \omterm{def:g11:electric-gravitational-fields:field}{veld} sterk is; (b) \omterm{def:g11:electric-gravitational-fields:lines}{veldlijnen} vertrekken uit negatieve
ladingen en komen aan bij positieve; (c) opeengedrongen \omterm{def:g11:electric-gravitational-fields:lines}{veldlijnen} wijzen
op een sterk \omterm{def:g11:electric-gravitational-fields:field}{veld}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:6}
Het internationale ruimtestation draait op een hoogte van \qty{420}{km}
($R = \qty{6.37e6}{m}$, $M = \qty{5.97e24}{kg}$).
\begin{enumerate}
  \item Bereken $g$ bij het station; welk deel van de waarde aan het
        oppervlak is dat?
  \item De astronauten zweven. Verzoen dat in één zin met je antwoord.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:7}
Welke veldgrootte zou het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een elektron opheffen
($m_e = \qty{9.11e-31}{kg}$)? Vergelijk met de \qty{2.0e4}{V/m} uit
\cref{ex:g11:electric-gravitational-fields:beam}; trek je conclusie over
de plaats van de zwaartekracht in de elektronenfysica.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:8}
Een stofdeeltje met massa \qty{2.0e-6}{kg} zweeft in rust tussen
horizontale platen op \qty{2.0}{cm} afstand met $U = \qty{5.0}{kV}$
ertussen.
\begin{enumerate}
  \item Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld} en daarna de grootte van de lading van het
        deeltje.
  \item De lading van het deeltje is positief: welke plaat is de
        positieve?
  \item Hoeveel \omterm{def:g11:fundamental-interactions:elementary}{elementaire ladingen} draagt het deeltje?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:9}
De ladingen $+Q$ en $+4Q$ liggen \qty{30}{cm} uit elkaar. Waar op het
lijnstuk ertussen is het totale \omterm{def:g11:electric-gravitational-fields:field}{veld} nul? Leg eerst uit waarom dat punt
tussen de ladingen moet liggen, dichter bij de kleinste.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:10}
In een waterstofatoom liggen het proton ($m_p = \qty{1.67e-27}{kg}$,
lading $+e$) en het elektron ($m_e = \qty{9.11e-31}{kg}$, lading $-e$)
$d = \qty{5.3e-11}{m}$ uit elkaar. Bereken de elektrische \omterm{def:g10:inertia:force}{kracht} en de
\omterm{def:g10:universal-gravitation:force}{gravitatiekracht} ertussen, en hun verhouding.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:11}
In een punt $M$ zou de ene bron alleen een \omterm{def:g11:electric-gravitational-fields:field}{veld} van \qty{300}{N/C} naar
het oosten opwekken; een tweede bron een van \qty{400}{N/C} naar het
noorden. \omterm{def:g11:electric-gravitational-fields:field}{Velden} tellen op als vectoren: geef het totale \omterm{def:g11:electric-gravitational-fields:field}{veld} in $M$
(grootte en richting), en daarna de \omterm{def:g10:inertia:force}{kracht} op
$q = \qty{-2.0e-6}{C}$ die daar wordt neergezet.
\end{exercise}

\section{Probleem: Het druppeltje van Millikan}

\begin{problem}[{Weekendopgave --- de lading van het elektron wegen: een
oliedruppeltje dat tussen twee platen in de lucht blijft hangen, onthult
dat lading in ondeelbare stapjes van $\num{1.60e-19}$ \omterm{def:g11:fundamental-interactions:charge}{coulomb} komt}]
\label{pb:g11:electric-gravitational-fields:1}
In 1909 verstoof Robert Millikan oliedruppeltjes tussen de horizontale
platen van een \omterm{def:g11:electric-gravitational-fields:capacitor}{condensator}. Door de \omterm{def:g11:circuits-and-power:voltage}{spanning} bij te stellen tot een
gekozen druppeltje roerloos hing --- de elektrische \omterm{def:g10:inertia:force}{kracht} precies in
\omterm{def:g10:inertia:compensate}{evenwicht} met het \omterm{def:g10:universal-gravitation:weight}{gewicht} --- kon hij zijn lading wegen. Ons druppeltje
heeft massa $m = \qty{4.90e-15}{kg}$ (gemeten aan zijn trage val met het
\omterm{def:g11:electric-gravitational-fields:field}{veld} uit; we nemen die als gegeven); de platen liggen $d = \qty{8.0}{mm}$
uit elkaar, met de \emph{bovenste} positief; de lading van het druppeltje
is negatief.

\medskip
\noindent\textbf{Deel I --- Het toneel.}
\begin{enumerate}
  \item Beschrijf het \omterm{def:g11:electric-gravitational-fields:field}{veld} tussen de platen; wat levert ``homogeen'' op?
  \item Bereken $E$ voor $U = \qty{800}{V}$.
  \item In welke richting werkt de elektrische \omterm{def:g10:inertia:force}{kracht} op het druppeltje?
        Waarom moest de \emph{bovenste} plaat de positieve zijn?
  \item Bereken het \omterm{def:g10:universal-gravitation:weight}{gewicht} van het druppeltje.
  \item Schrijf de voorwaarde voor rust als een gelijkheid van twee
        krachtgroottes.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het \omterm{def:g10:inertia:compensate}{evenwicht}.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:pressure:pressure}{Druk} daaruit $\abs{q}$ uit in $m$, $g$, $d$ en $U$.
  \item Het druppeltje hangt in rust bij $U = \qty{800}{V}$: bereken
        $\abs{q}$.
  \item Deel door $e = \qty{1.60e-19}{C}$. Hoeveel elektronen te veel
        draagt het druppeltje?
  \item De \omterm{def:g11:circuits-and-power:voltage}{spanning} wordt naar \qty{900}{V} opgeschroefd. In welke
        richting wijst de resulterende \omterm{def:g10:inertia:force}{kracht} nu?
  \item Toon aan dat de evenwichtsspanning $U = mgd/\abs{q}$ is: hoe
        groter de lading, hoe kleiner de \omterm{def:g11:circuits-and-power:voltage}{spanning}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De stapjes.}
Een stralingspuls kan elektronen op het druppeltje slaan of eraf; na elke
verandering stelt Millikan de \omterm{def:g11:circuits-and-power:voltage}{spanning} opnieuw op \omterm{def:g10:inertia:compensate}{evenwicht} af.
\begin{enumerate}[resume]
  \item Na één puls vergt het \omterm{def:g10:inertia:compensate}{evenwicht} $U = \qty{1200}{V}$. Bereken de
        nieuwe $\abs{q}$. Heeft het druppeltje een elektron gewonnen of
        verloren?
  \item Opeenvolgende \omterm{def:g10:inertia:compensate}{evenwichten} worden gevonden bij $U = 1200$, $800$,
        $600$, $480$ en \qty{400}{V}. Bereken de vijf ladingen.
  \item Toon aan dat alle vijf gehele veelvouden van één grootheid zijn;
        geef haar waarde.
  \item Met hoeveel verandert de lading tussen twee opeenvolgende
        \omterm{def:g10:inertia:compensate}{evenwichten}? Interpreteer dat.
  \item Leg uit waarom dit druppeltje nooit in rust kan hangen bij
        $U = \qty{700}{V}$, hoe geduldig je ook afstelt.
  \item Millikan herhaalde dit met honderden druppeltjes: elke gemeten
        lading was een geheel veelvoud van dezelfde waarde. Formuleer de
        conclusie en noem de grootheid die is gemeten.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Waarom er geen Millikan voor de
zwaartekracht bestaat.}
\begin{enumerate}[resume]
  \item Wat speelt in het woordenboek van dit hoofdstuk de rol van $q$,
        $E$ en $qE$ in het \omterm{def:g10:inertia:compensate}{evenwicht} van deel II?
  \item Zou een massa erboven het druppeltje in plaats daarvan omhoog
        kunnen houden? Bereken het \omterm{def:g11:electric-gravitational-fields:field}{veld} van een loden bol van
        \qty{1000}{kg} op \qty{1.0}{m} boven het druppeltje, de \omterm{def:g10:inertia:force}{kracht} op
        het druppeltje, en vergelijk met zijn \omterm{def:g10:universal-gravitation:weight}{gewicht}.
  \item Het \omterm{def:g10:inertia:compensate}{evenwicht} vraagt om een opwaartse trek die het tegen de hele
        Aarde opneemt. Waarom kan een geladen plaat die leveren en geen
        enkele schikking van massa's?
  \item Ook de \emph{massa} van het druppeltje verandert in stapjes (één
        oliemolecuul: ongeveer \qty{5e-25}{kg}). Vergelijk de relatieve
        sprongen in $m$ en in $\abs{q}$ wanneer er één elektron landt, en
        besluit: waarom ziet dit \omterm{def:g10:inertia:compensate}{evenwicht} de atomen van de
        elektriciteit maar niet de \omterm{def:g11:fundamental-interactions:ladder}{moleculen} van de olie? De clou: lading
        is gekwantiseerd, en haar \omterm{def:g11:fundamental-interactions:ladder}{atoom} is $e = \qty{1.60e-19}{C}$.
\end{enumerate}
\end{problem}
