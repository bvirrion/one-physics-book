\chapter{De fundamentele wisselwerkingen}\label{ch:g11:fundamental-interactions}

Wrijf een ballon over je haar en hij pikt snippers papier op --- en
verslaat daarmee, met een paar vierkante centimeter rubber, de
zwaartekracht van de hele planeet. Die makkelijke overwinning verklaart
waarom atomen bijeenblijven en waarom zware \omterm{def:g11:fundamental-interactions:ladder}{kernen} uiteindelijk
uiteenvallen (\cref{ch:g11:nucleus-radioactivity}). Het heelal draait op
precies vier wisselwerkingen; dit hoofdstuk maakt kennis met ze, weegt ze
tegen elkaar af en wijst elk haar eigen verdieping toe.

\section{Materie van quark tot melkwegstelsel}

\begin{definition}[De ladder van de structuur]\label{def:g11:fundamental-interactions:ladder}
Materie is in verdiepingen gebouwd, elk opgetrokken uit de verdieping
eronder: drie \emph{quarks}\index{quark} (puntvormig, kleiner dan
\qty{e-18}{m}) binden tot een \emph{nucleon}\index{nucleon} --- een
proton of een neutron van ongeveer \qty{e-15}{m}; nucleonen pakken zich
samen tot een \emph{kern}\index{kern} (\qty{e-15}{m} tot \qty{e-14}{m});
een kern met haar elektronenwolk is een \emph{atoom}\index{atoom}
(\qty{e-10}{m}); atomen binden tot \emph{moleculen}\index{molecuul} en
kristallen (vanaf \qty{e-9}{m}), die zich samenvoegen tot alledaagse
voorwerpen (\qty{1}{m}), planeten (\qty{e7}{m}), planetenstelsels
(\qty{e11}{m}), melkwegstelsels (\qty{e21}{m}) en het waarneembare
heelal (\qty{e26}{m}).
\end{definition}

\begin{definition}[De vier wisselwerkingen]\label{def:g11:fundamental-interactions:four}
Een \emph{fundamentele wisselwerking}\index{fundamentele wisselwerking}
is een \omterm{def:g10:inertia:force}{kracht} die niet tot andere \omterm{def:g10:inertia:force}{krachten} te herleiden is; er zijn er
precies vier: de \emph{gravitatie}\index{gravitatie}, de
\emph{elektromagnetische wisselwerking}\index{elektromagnetische
wisselwerking}, de \omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking} en de \omterm{def:g11:fundamental-interactions:weak}{zwakke wisselwerking}.
\end{definition}

\begin{remark}\label{rem:g11:fundamental-interactions:empty}
De verdiepingen worden door leegte gescheiden: een \omterm{def:g11:fundamental-interactions:ladder}{atoom} is $10^{5}$ keer
zo breed als zijn \omterm{def:g11:fundamental-interactions:ladder}{kern} --- vergroot de \omterm{def:g11:fundamental-interactions:ladder}{kern} tot een knikker van
\qty{1}{cm} en de elektronenwolk meet een kilometer. Iets moet die ijle
verdiepingen bijeenhouden, en de zwaartekracht is dat vrijwel nooit.
\end{remark}

\section{Elektrische lading}

\begin{definition}[Elektrische lading]\label{def:g11:fundamental-interactions:charge}
\emph{Elektrische lading}\index{elektrische lading} is de eigenschap van
materie waarop de \omterm{def:g11:fundamental-interactions:four}{elektromagnetische wisselwerking} aangrijpt, zoals massa
de eigenschap is waarop de \omterm{def:g11:fundamental-interactions:four}{gravitatie} aangrijpt. Ze wordt gemeten in
\emph{coulomb}\index{coulomb} (\unit{C}) en bestaat in \emph{twee
tekens}, positief en negatief, die elkaar bij optellen opheffen; gelijke
ladingen stoten elkaar af, ongelijke trekken elkaar aan.
\end{definition}

\begin{definition}[Elementaire lading]\label{def:g11:fundamental-interactions:elementary}
Lading is korrelig: elke waargenomen lading is een geheel veelvoud van de
\emph{elementaire lading}\index{elementaire lading}
$e = \qty{1.60e-19}{C}$. Het proton draagt $+e$, het elektron $-e$, het
neutron $0$; een lading $q$ bevat $N = \abs{q}/e$ elementaire ladingen.
\end{definition}

\begin{proposition}[Behoud van lading]\label{prop:g11:fundamental-interactions:conservation}
De totale lading van een geïsoleerd systeem verandert nooit: lading wordt
niet geschapen en niet vernietigd, alleen overgedragen --- in de praktijk
door elektronen, de lichtste dragers.
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:fundamental-interactions:conservation}
Er is nooit een schending waargenomen; de diepe reden, een symmetrie van
het elektromagnetisme, komt in de universitaire volumes aan bod.
\end{remark}

\begin{definition}[Laden door wrijving en door contact]\label{def:g11:fundamental-interactions:charging}
Twee isolatoren over elkaar wrijven rukt elektronen van de ene los en op
de andere, zodat er tegengestelde ladingen van gelijke grootte
achterblijven (\emph{laden door wrijving}\index{laden door wrijving});
een geladen voorwerp tegen een neutraal houden verdeelt de lading
(\emph{laden door contact}\index{laden door contact}). Alleen elektronen
bewegen; de totale lading blijft behouden.
\end{definition}

\begin{example}[Elektronen tellen]\label{ex:g11:fundamental-interactions:balloon}
Een ballon die over haar is gewreven, krijgt $q = \qty{-1.6e-8}{C}$: hij
heeft $N = \num{1.6e-8} / \num{1.6e-19} = \num{1.0e11}$ elektronen
\emph{gewonnen}, en het haar blijft achter met precies
$+\qty{1.6e-8}{C}$.
\end{example}

\section{De wet van Coulomb}

\begin{theorem}[Wet van Coulomb]\label{thm:g11:fundamental-interactions:coulomb}
Twee puntladingen $q_1$ en $q_2$ op een afstand $d$ oefenen op elkaar
\omterm{def:g10:inertia:force}{krachten} uit langs hun verbindingslijn, van gelijke grootte en
tegengestelde richting --- afstotend bij gelijke tekens, aantrekkend bij
ongelijke --- met
\[
F = k\,\frac{\abs{q_1 q_2}}{d^{2}},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{theorem}
\admitted

\begin{remark}\label{rem:g11:fundamental-interactions:torsion}
Coulomb stelde de wet in 1785 vast door de verdraaiing van een \omterm{def:g11:lenses-and-eye:lens}{dunne}
torsiedraad te meten; waarom de \omterm{def:g10:orders-of-magnitude:scinot}{exponent} precies $2$ is, wordt in de
universitaire volumes beantwoord.
\end{remark}

\begin{remark}[Een formele tweelingzus van de gravitatie]\label{rem:g11:fundamental-interactions:twin}
De wet van Coulomb is, symbool voor symbool, de wet van de algemene
\omterm{def:g11:fundamental-interactions:four}{gravitatie} (\cref{ch:g10:universal-gravitation}) met ladingen in plaats
van massa's:
\begin{center}
\begin{tabular}{lll}
bron & massa $m$ & lading $q$ \\
wet & $F = G\,\dfrac{m_1 m_2}{d^2}$ & $F = k\,\dfrac{\abs{q_1 q_2}}{d^2}$ \\[1.5ex]
constante & $G = \qty{6.67e-11}{N.m^2/kg^2}$ & $k = \qty{8.99e9}{N.m^2/C^2}$ \\
teken & altijd aantrekkend & aantrekkend of afstotend \\
\end{tabular}
\end{center}
De twee verschillen sturen alles wat volgt: de elektriciteit is
overweldigend sterker, en zij alleen kan zichzelf opheffen.
\end{remark}

\begin{example}[Twee protonen, beide krachten tegelijk]\label{ex:g11:fundamental-interactions:protons}
Twee protonen ($m_p = \qty{1.67e-27}{kg}$) liggen $d = \qty{1.0e-15}{m}$
uit elkaar --- buren in een \omterm{def:g11:fundamental-interactions:ladder}{kern}. Elektrische afstoting:
\[
F_E = \num{8.99e9} \times
\frac{(\num{1.60e-19})^{2}}{(\num{1.0e-15})^{2}}
\approx \qty{2.3e2}{N}
\]
--- het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een koffer van \qty{23}{kg},
gedragen door een deeltje van $10^{-27}$
\omterm{def:g10:orders-of-magnitude:unit}{kilogram}. Gravitatie-aantrekking:
$F_G = \num{6.67e-11} \times (\num{1.67e-27})^{2} /
(\num{1.0e-15})^{2} \approx \qty{1.9e-34}{N}$. De verhouding
$F_E / F_G = k e^{2} / (G m_p^{2}) \approx \num{1.2e36}$ hangt niet van
de afstand af ($d^{2}$ valt weg): tussen elementaire deeltjes is de
zwaartekracht $10^{36}$ keer te zwak om mee te tellen --- op \emph{elke}
afstand.
\end{example}

\section{De sterke en de zwakke wisselwerking}

Het voorbeeld laat een schandaal achter: een heliumkern bevat twee
protonen die elkaar met tientallen \omterm{def:g10:inertia:force}{newton} afstoten --- en toch bestaat
helium.

\begin{definition}[De sterke wisselwerking]\label{def:g11:fundamental-interactions:strong}
De \emph{sterke wisselwerking}\index{sterke wisselwerking} bindt \omterm{def:g11:fundamental-interactions:ladder}{quarks}
tot \omterm{def:g11:fundamental-interactions:ladder}{nucleonen} en \omterm{def:g11:fundamental-interactions:ladder}{nucleonen} tot \omterm{def:g11:fundamental-interactions:ladder}{kernen}. Op \qty{e-15}{m} is ze aantrekkend
en veel sterker dan de elektrische afstoting (duizenden \omterm{def:g10:inertia:force}{newton} tussen
\omterm{def:g11:fundamental-interactions:ladder}{nucleonen}); voorbij een paar keer $10^{-15}$ \omterm{def:g10:orders-of-magnitude:unit}{meter} verdwijnt ze: haar
\emph{dracht}\index{dracht} is ongeveer \qty{e-15}{m}. Ze grijpt protonen
en neutronen even hard vast en trekt zich van lading niets aan.
\end{definition}

\begin{remark}[Waarom kernen bestaan --- en waarom zware het begeven]\label{rem:g11:fundamental-interactions:nuclei}
De twee \omterm{def:g10:inertia:force}{krachten} in de \omterm{def:g11:fundamental-interactions:ladder}{kern} schalen verschillend. De sterke lijm heeft
een korte \omterm{def:g11:fundamental-interactions:strong}{dracht}: een \omterm{def:g11:fundamental-interactions:ladder}{nucleon} bindt alleen zijn paar buren binnen
\qty{e-15}{m}. De elektrische afstoting reikt ver: elk proton duwt op
\emph{elk ander} proton. Naarmate \omterm{def:g11:fundamental-interactions:ladder}{kernen} groeien, stapelt de afstoting
zich sneller op dan de lijm: zware \omterm{def:g11:fundamental-interactions:ladder}{kernen} hebben extra neutronen nodig
(lijm zonder lading), en voorbij ongeveer $80$ protonen helpt zelfs dat
niet meer --- de zwaarste \omterm{def:g11:fundamental-interactions:ladder}{kernen} leven op geleende tijd
(\cref{ch:g11:nucleus-radioactivity}).
\end{remark}

\begin{definition}[De zwakke wisselwerking]\label{def:g11:fundamental-interactions:weak}
De \emph{zwakke wisselwerking}\index{zwakke wisselwerking}, met een nog
kortere \omterm{def:g11:fundamental-interactions:strong}{dracht}, laat een neutron in een proton veranderen (en omgekeerd):
zij drijft de $\beta$-radioactiviteit uit
\cref{ch:g11:nucleus-radioactivity} aan.
\end{definition}

\section{Wie welke verdieping regeert}

\begin{method}[Een schaal doorlichten]\label{met:g11:fundamental-interactions:audit}
Om te bepalen welke wisselwerking een structuur met afmeting $d$
regeert, kijk je naar \omterm{def:g11:fundamental-interactions:strong}{dracht} en opheffing:
\begin{enumerate}
  \item $d \leq \qty{e-14}{m}$: de \emph{\omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking}} regeert
        --- ze verslaat de elektrische afstoting, en de zwaartekracht
        ligt er $10^{36}$ naast.
  \item van atomen tot bergen ($\qty{e-10}{m}$ tot $\qty{e4}{m}$): de
        sterke \omterm{def:g10:inertia:force}{kracht} is buiten \omterm{def:g11:fundamental-interactions:strong}{dracht}; de \emph{\omterm{def:g11:fundamental-interactions:four}{elektromagnetische
        wisselwerking}} regeert --- bindingen, samenhang, \omterm{def:g10:inertia:inventory}{wrijving},
        contact.
  \item planeten en groter ($d \geq \qty{e6}{m}$): materie is tot in
        fantastische nauwkeurigheid neutraal, dus heft de elektriciteit
        zichzelf op; massa heeft één teken en telt altijd op --- de
        \emph{\omterm{def:g11:fundamental-interactions:four}{gravitatie}} regeert de hemel.
\end{enumerate}
De \omterm{def:g11:fundamental-interactions:weak}{zwakke wisselwerking} bezet geen enkele verdieping: ze bindt niets,
maar beslecht omzettingen ($\beta$-verval).
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:1}
Geef van elke afmeting de \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte} en de verdieping van de ladder:
proton \qty{1.7e-15}{m}; watermolecuul \qty{2.8e-10}{m}; zandkorrel
\qty{2e-4}{m}; Aarde \qty{1.3e7}{m}; Melkweg \qty{9.5e20}{m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:2}
Een ballon die over haar is gewreven, draagt $q = \qty{-4.8e-9}{C}$.
Heeft hij elektronen gewonnen of verloren, en hoeveel? Welke lading heeft
het haar daarna, en welke wet zegt dat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:3}
De ladingen $q_1 = \qty{+2.0e-6}{C}$ en $q_2 = \qty{-3.0e-6}{C}$ liggen
\qty{30}{cm} uit elkaar. Bereken de \omterm{def:g10:inertia:force}{kracht}; aantrekkend of afstotend?
Vergelijk de \omterm{def:g10:inertia:force}{kracht} op $q_1$ met die op $q_2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:4}
Twee ladingen trekken elkaar aan met een \omterm{def:g10:inertia:force}{kracht} $F_0$. Wat wordt die als:
de afstand verdubbelt; de afstand door $3$ wordt gedeeld; één lading
verdubbelt; beide ladingen \emph{en} de afstand verdubbelen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:5}
Noem de wisselwerking die verantwoordelijk is voor: (a) de Maan die om de
Aarde draait; (b) de samenhang van een zoutkristal; (c) de samenhang van
een heliumkern; (d) het $\beta$-verval van koolstof-14; (e) de ballon uit
\cref{exo:g11:fundamental-interactions:2} die aan een muur blijft
plakken.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:6}
De twee protonen van een heliumkern liggen $d = \qty{2.0e-15}{m}$ uit
elkaar ($m_p = \qty{1.67e-27}{kg}$). Bereken hun elektrische afstoting,
hun gravitatie-aantrekking en de verhouding. Wat houdt de \omterm{def:g11:fundamental-interactions:ladder}{kern} heel?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:7}
In een waterstofatoom zit het elektron ($m_e = \qty{9.11e-31}{kg}$) op
$d = \qty{5.3e-11}{m}$ van het proton ($m_p = \qty{1.67e-27}{kg}$).
Bereken de elektrische en de \omterm{def:g10:universal-gravitation:force}{gravitatiekracht} ertussen en hun
verhouding. Wat houdt atomen bijeen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:8}
Bol A draagt $q_A = \qty{+8.0e-9}{C}$; een identieke bol B is neutraal.
Ze worden tegen elkaar gehouden en daarna tot \qty{10}{cm} uit elkaar
gezet. Welke lading draagt elk (controleer het behoud van lading)?
Bereken de \omterm{def:g10:inertia:force}{kracht} ertussen --- aantrekkend of afstotend?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:9}
Twee gelijke ladingen $q = \qty{1.0e-6}{C}$ stoten elkaar af met
\qty{0.90}{N}: hoever liggen ze uit elkaar? Waar zakt de \omterm{def:g10:inertia:force}{kracht} tot
\qty{0.10}{N}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:10}
Twee protonen in naburige \omterm{def:g11:fundamental-interactions:ladder}{moleculen} liggen $d = \qty{1.0e-10}{m}$ uit
elkaar. Bereken hun elektrische afstoting. Wat draagt de \omterm{def:g11:fundamental-interactions:strong}{sterke
wisselwerking} op deze afstand bij? Welke wisselwerking bindt \omterm{def:g11:fundamental-interactions:ladder}{moleculen}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:11}
Elk van twee muntstukken bevat ongeveer $\num{e23}$ elektronen. Verplaats
één elektron op een \emph{miljoen} van het ene naar het andere en zet de
munten \qty{1.0}{m} uit elkaar. Bereken de lading van elke munt en daarna
de \omterm{def:g10:inertia:force}{kracht}. Aan het \omterm{def:g10:universal-gravitation:weight}{gewicht} van welke massa is die gelijk? Trek je
conclusie over de neutraliteit van alledaagse materie.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:12}
Een uraniumkern heeft een diameter van \qty{1.4e-14}{m} en $92$ protonen.
Bereken de afstoting tussen twee protonen aan weerszijden. Kan de \omterm{def:g11:fundamental-interactions:strong}{sterke
wisselwerking} dit paar binden --- en waarom niet? Hoeveel elkaar
afstotende protonenparen zijn er? Waarom hebben zware \omterm{def:g11:fundamental-interactions:ladder}{kernen} extra
neutronen nodig, en waarom kunnen zelfs neutronen ze voorbij een bepaalde
grootte niet meer redden?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:13}
Twee bolletjes met massa \qty{1.0}{g} hangen aan draden van \qty{50}{cm}
die aan hetzelfde punt zijn vastgemaakt. Krijgen ze dezelfde lading $q$,
dan komen ze \qty{6.0}{cm} uit elkaar tot rust. Elk bolletje is in
\omterm{def:g10:inertia:compensate}{evenwicht} onder zijn \omterm{def:g10:universal-gravitation:weight}{gewicht}, de \omterm{def:g10:inertia:inventory}{spankracht} van de draad en de
elektrische \omterm{def:g10:inertia:force}{kracht}: toon aan dat $F = mg\tan\theta$ ($\theta$: de hoek
van de draad met de verticaal) en dat hier $\tan\theta \approx 0.060$.
Bereken $F$, daarna $q$, en daarna het aantal verplaatste \omterm{def:g11:fundamental-interactions:elementary}{elementaire
ladingen}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:14}
Gegevens: $M_E = \qty{5.97e24}{kg}$, $M_M = \qty{7.35e22}{kg}$, afstand
Aarde--Maan $d = \qty{3.84e8}{m}$. Bereken de \omterm{def:g10:universal-gravitation:force}{gravitatiekracht} tussen
Aarde en Maan. De zwaartekracht wordt uitgeschakeld: welke ladingen $+Q$
op de Aarde en $-Q$ op de Maan houden dezelfde aantrekking in stand?
Hoeveel elektronen is $Q$, en hoeveel wegen die
($m_e = \qty{9.11e-31}{kg}$)? Becommentarieer.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:15}
In een zoutkristal liggen Na$^{+}$- en Cl$^{-}$-ionen (met ladingen
$\pm e$) $d = \qty{2.8e-10}{m}$ uit elkaar; een Cl$^{-}$-ion heeft massa
\qty{5.9e-26}{kg}. Bereken de \omterm{def:g10:inertia:force}{kracht} tussen naburige ionen en het \omterm{def:g10:universal-gravitation:weight}{gewicht}
van een Cl$^{-}$-ion; geef de verhouding. Leg daarna uit hoe de
zwaartekracht op grote schaal toch wint --- geen berg komt boven
\qty{e4}{m} uit --- terwijl elke binding haar met vijftien \omterm{def:g10:orders-of-magnitude:oom}{ordegrootten}
verslaat.
\end{exercise}

\section{Probleem: De doorlichting van de vier krachten}

\begin{problem}[{Weekendopgave --- waarom de wereld niet instort en niet
uiteenvliegt: de vier wisselwerkingen verdieping voor verdieping
doorgelicht, tot en met de neutraliteit van materie op twee delen in
$10^{18}$}]
\label{pb:g11:fundamental-interactions:1}
Atomen imploderen niet, \omterm{def:g11:fundamental-interactions:ladder}{kernen} houden meestal stand, de Maan stort niet
neer en ontsnapt niet; deze opgave zoekt uit wie de eer verdient.
Gegevens: $e = \qty{1.60e-19}{C}$, $k = \qty{8.99e9}{N.m^2/C^2}$,
$G = \qty{6.67e-11}{N.m^2/kg^2}$, $m_p = \qty{1.67e-27}{kg}$,
$m_e = \qty{9.11e-31}{kg}$, $M_E = \qty{5.97e24}{kg}$,
$M_M = \qty{7.35e22}{kg}$, afstand Aarde--Maan \qty{3.84e8}{m}.

\medskip
\noindent\textbf{Deel I --- De ladder.}
\begin{enumerate}
  \item Noem de verdiepingen van de ladder van \omterm{def:g11:fundamental-interactions:ladder}{quark} tot melkwegstelsel,
        met telkens één \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte} van afmeting.
  \item Schaalmodel: de \omterm{def:g11:fundamental-interactions:ladder}{kern} wordt een knikker van \qty{1}{cm}; hoe breed
        is het \omterm{def:g11:fundamental-interactions:ladder}{atoom} dan?
  \item Welk deel van het volume van een \omterm{def:g11:fundamental-interactions:ladder}{atoom} neemt zijn \omterm{def:g11:fundamental-interactions:ladder}{kern} in?
  \item Een haar is \qty{70}{\micro m} dik. Hoeveel atomen liggen erover?
  \item Hoeveel machten van tien liggen er tussen de bovengrens van het
        \omterm{def:g11:fundamental-interactions:ladder}{quark} (\qty{e-18}{m}) en een melkwegstelsel (\qty{e21}{m})?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De elektrische verdieping.}
\begin{enumerate}[resume]
  \item In een waterstofatoom zit het elektron op $d = \qty{5.3e-11}{m}$
        van het proton: bereken de elektrische \omterm{def:g10:inertia:force}{kracht} ertussen.
  \item Bereken de \omterm{def:g10:universal-gravitation:force}{gravitatiekracht} ertussen en de verhouding van de
        twee. Welke van beide houdt het \omterm{def:g11:fundamental-interactions:ladder}{atoom} bijeen?
  \item Wat verandert er als de zwaartekracht binnen atomen wordt
        uitgeschakeld? En als de elektriciteit dat wordt?
  \item Waarom zijn de ``\omterm{def:g10:inertia:contact}{contactkrachten}'' van het dagelijks leven --- de
        vloer die tegen je voeten duwt, de \omterm{def:g10:inertia:inventory}{wrijving} --- vermomd
        elektromagnetisch?
  \item Elk van twee muntstukken bevat ongeveer $\num{e23}$ elektronen.
        Verplaats één elektron op een \emph{miljard} van het ene naar het
        andere en zet ze \qty{1.0}{m} uit elkaar: bereken de ladingen en
        de \omterm{def:g10:inertia:force}{kracht}. Conclusie?
  \item Welke wisselwerking regeert de verdiepingen van \qty{e-10}{m} tot
        \qty{e4}{m}, en waarom doet de \omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking} niet mee?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De kernverdieping.}
\begin{enumerate}[resume]
  \item Twee protonen liggen \qty{1.0e-15}{m} uit elkaar in een \omterm{def:g11:fundamental-interactions:ladder}{kern}:
        bereken hun elektrische afstoting.
  \item Welke versnelling zou die \omterm{def:g10:inertia:force}{kracht} een ongeremd proton geven?
        Vergelijk met $g$.
  \item Noem de drie eigenschappen die de \omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking} nodig
        heeft om te verklaren waarom \omterm{def:g11:fundamental-interactions:ladder}{kernen} bestaan, waarom ook neutronen
        worden vastgegrepen, en waarom \omterm{def:g11:fundamental-interactions:ladder}{kernen} zo klein blijven.
  \item Bereken in een uraniumkern (diameter \qty{1.4e-14}{m}, $92$
        protonen) de afstoting tussen twee protonen aan weerszijden. Kan
        de \omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking} dat paar rechtstreeks binden? Leg uit
        waarom de afstoting wint naarmate \omterm{def:g11:fundamental-interactions:ladder}{kernen} groeien, en waar de
        neutronen voor dienen.
  \item In één zin, zoals beloofd: wat doet de \omterm{def:g11:fundamental-interactions:weak}{zwakke wisselwerking}?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De astronomische verdieping, en het
vonnis.}
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:g10:universal-gravitation:force}{gravitatiekracht} tussen de Aarde en de Maan.
  \item De Aarde bevat ongeveer $\num{1.8e51}$ protonen (en evenveel
        elektronen), de Maan ongeveer $\num{2.2e49}$. Welk deel $f$ van
        de protonlading van elk lichaam zou, niet-opgeheven, de
        elektrische \omterm{def:g10:inertia:force}{kracht} gelijk maken aan de \omterm{def:g10:universal-gravitation:force}{gravitatiekracht}?
  \item Waarom regeert de \omterm{def:g11:fundamental-interactions:four}{gravitatie}, de zwakkeling uit vraag 7, de
        sterrenkunde? Twee redenen.
  \item Vonnis --- één zin per verdieping (\omterm{def:g11:fundamental-interactions:ladder}{kern}, \omterm{def:g11:fundamental-interactions:ladder}{atoom} tot berg, planeet
        en groter): wat houdt haar bijeen? Beantwoord daarna de titel:
        waarom stort de wereld niet in en vliegt ze niet uiteen?
\end{enumerate}
\end{problem}
