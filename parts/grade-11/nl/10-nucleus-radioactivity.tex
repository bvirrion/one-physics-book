\chapter{De atoomkern en radioactiviteit}\label{ch:g11:nucleus-radioactivity}

Op dit ogenblik ontploffen er elke seconde zo'n achtduizend atoomkernen in
je lichaam. Niets valt ze aan: sommige kernen worden onstabiel geboren, en
elk daarvan zet zich vroeg of laat om en slingert een brokstuk weg met een
flink deel van de lichtsnelheid. Dit hoofdstuk keert terug naar de
verdieping van $10^{-15}$ \omterm{def:g10:orders-of-magnitude:unit}{meter} uit
\cref{ch:g11:fundamental-interactions} om te zien welke kernen breekbaar
zijn, hoe ze bezwijken, en hoe hun regelmatige ongeduld grotten dateert en
plafonds bewaakt.

\section{Kernen, nucleonen en isotopen}

\begin{definition}[Kernnotatie]\label{def:g11:nucleus-radioactivity:notation}
Een kern bevat $Z$ protonen (elk met lading $+e$) en $N$ neutronen (zonder
lading), in totaal $A = Z + N$ \omterm{def:g11:fundamental-interactions:ladder}{nucleonen}; ze wordt geschreven als
${}^{A}_{Z}\mathrm{X}$, waarin $\mathrm X$ het scheikundige symbool is,
$Z$ het \emph{atoomnummer}\index{atoomnummer} en $A$ het
\emph{massagetal}\index{massagetal}. Een soort die door $A$ en $Z$ is
vastgelegd, heet een \emph{nuclide}\index{nuclide}; nucliden met dezelfde
$Z$ maar verschillende $A$ zijn \emph{isotopen}\index{isotoop} van
hetzelfde element: dezelfde scheikunde (de elektronenwolk ziet alleen
$Ze$), verschillende kernen en verschillende stabiliteit.
\end{definition}

\begin{example}[Waterstof en koolstof]\label{ex:g11:nucleus-radioactivity:hc}
Waterstof heeft drie \omterm{def:g11:nucleus-radioactivity:notation}{isotopen}: ${}^{1}_{1}\mathrm{H}$ (een enkel proton),
${}^{2}_{1}\mathrm{H}$ (deuterium: één proton, één neutron) en het
onstabiele ${}^{3}_{1}\mathrm{H}$ (tritium: twee neutronen). Natuurlijke
koolstof: vooral ${}^{12}_{6}\mathrm{C}$ ($6$ protonen, $6$ neutronen),
$1.1\%$ ${}^{13}_{6}\mathrm{C}$, en één \omterm{def:g11:fundamental-interactions:ladder}{atoom} op $10^{12}$ van het
onstabiele ${}^{14}_{6}\mathrm{C}$ ($8$ neutronen), de held van
\cref{ex:g11:nucleus-radioactivity:dating}.
\end{example}

\section{Stabiele en onstabiele kernen}

\begin{proposition}[Het dal van de stabiliteit]\label{prop:g11:nucleus-radioactivity:valley}
Stabiele \omterm{def:g11:nucleus-radioactivity:notation}{nucliden} bezetten een smalle band in het $(Z, N)$-vlak:
$N \approx Z$ voor lichte kernen, daarna een groeiend neutronenoverschot
tot $N \approx 1.5\,Z$. Elke \omterm{def:g11:nucleus-radioactivity:notation}{nuclide} buiten die band --- te veel
neutronen, te veel protonen, of te groot (voorbij lood, $Z = 82$, is geen
enkele \omterm{def:g11:nucleus-radioactivity:notation}{nuclide} werkelijk stabiel) --- zet zich vroeg of laat om.
\end{proposition}
\admitted

\begin{remark}
De band precies lokaliseren is kwantummechanica --- werk voor de
universitaire volumes. Het patroon is de boekhouding van
\cref{ch:g11:fundamental-interactions}: de sterke lijm bindt alleen buren,
de coulombafstoting overspant de hele kern, dus hebben zware kernen extra
neutronenlijm nodig; maar een overschot aan neutronen is óók onstabiel
(daar zorgt de \omterm{def:g11:fundamental-interactions:weak}{zwakke wisselwerking} voor).
\end{remark}

\begin{definition}[Radioactiviteit]\label{def:g11:nucleus-radioactivity:radioactivity}
\emph{Radioactiviteit}\index{radioactiviteit} is de spontane omzetting van
een onstabiele kern in een andere \omterm{def:g11:nucleus-radioactivity:notation}{nuclide}, met uitzending van straling.
Ze is \emph{willekeurig}: niets kondigt aan welke kern het eerst vervalt,
en geen \omterm{def:g10:pressure:pressure}{druk}, temperatuur of scheikunde kan haar versnellen of vertragen;
alleen grote aantallen laten zich voorspellen
(\cref{def:g11:nucleus-radioactivity:halflife}).
\end{definition}

\begin{remark}[Becquerel en de Curies]\label{rem:g11:nucleus-radioactivity:curie}
In 1896 ontdekte Henri Becquerel dat uraniumzouten een fotografische plaat
beslaan door zwart papier heen, in een gesloten lade, zonder dat er
\omterm{def:g10:energy-conservation:energy}{energie} wordt toegevoerd. Marie en Pierre Curie toonden aan dat het effect
atomair is, maten het, en isoleerden polonium en radium, een miljoen keer
actiever. Het woord \emph{\omterm{def:g11:nucleus-radioactivity:radioactivity}{radioactiviteit}} is van Marie Curie; net als
twee Nobelprijzen.
\end{remark}

\section{Alfa, bèta, gamma}

\begin{definition}[De drie historische stralingen]\label{def:g11:nucleus-radioactivity:abg}
Drie stralingen, benoemd voordat iemand wist wat ze waren:
\begin{itemize}
  \item \emph{alfa}\index{alfastraling} ($\alpha$): een heliumkern
        ${}^{4}_{2}\mathrm{He}$ die in haar geheel wordt uitgestoten ---
        zwaar, dubbel geladen, tegengehouden door een vel papier of enkele
        centimeters lucht;
  \item \emph{bèta}\index{bètastraling} ($\beta^-$): een elektron
        ${}^{\;0}_{-1}\mathrm{e}$ dat in de kern ontstaat en wordt
        uitgestoten --- licht, snel, tegengehouden door enkele
        millimeters aluminium;
  \item \emph{gamma}\index{gammastraling} ($\gamma$): elektromagnetische
        straling, veel energierijker dan licht --- nooit helemaal
        tegengehouden, alleen \emph{verzwakt}\index{verzwakking}: een
        centimeter lood absorbeert de helft.
\end{itemize}
\end{definition}

\begin{proposition}[Behoudswetten in kernvergelijkingen]\label{prop:g11:nucleus-radioactivity:soddy}
Bij elke kernomzetting zijn het totale \omterm{def:g11:nucleus-radioactivity:notation}{massagetal} $A$ en de totale lading
--- de som van de onderste indices, met $-1$ voor een uitgezonden elektron
--- vóór en na de omzetting dezelfde.
\end{proposition}
\admitted

\begin{remark}
Het behoud van lading gold al in \cref{ch:g11:fundamental-interactions};
het behoud van $A$ zegt dat \omterm{def:g11:fundamental-interactions:ladder}{nucleonen} alleen worden omgezet, nooit
geschapen of vernietigd --- eerlijke boekhouding, uit te werken in de
universitaire volumes.
\end{remark}

\begin{method}[Een vervalvergelijking kloppend maken]\label{met:g11:nucleus-radioactivity:balance}
\begin{enumerate}
  \item Schrijf elke speler met beide labels:
        $\alpha = {}^{4}_{2}\mathrm{He}$,
        $\beta^- = {}^{\;0}_{-1}\mathrm{e}$,
        $\beta^+ = {}^{0}_{+1}\mathrm{e}$; een $\gamma$ draagt
        $A = Z = 0$.
  \item Stel de sommen van de $A$'s gelijk, en daarna die van de $Z$'s
        (\cref{prop:g11:nucleus-radioactivity:soddy}); los op en lees het
        dochterelement af aan zijn $Z$ in het \omterm{def:g10:signals-and-waves:period}{periodiek} systeem.
\end{enumerate}
\end{method}

\begin{example}[Een alfaverval]\label{ex:g11:nucleus-radioactivity:alpha}
Radium-226, het radium van de Curies, is een $\alpha$-straler: uit
${}^{226}_{88}\mathrm{Ra} \to {}^{A}_{Z}\mathrm{Y} + {}^{4}_{2}\mathrm{He}$
volgt $A = 226 - 4 = 222$ en $Z = 88 - 2 = 86$; element $86$ is radon, dus
${}^{226}_{88}\mathrm{Ra} \to {}^{222}_{86}\mathrm{Rn} + {}^{4}_{2}\mathrm{He}$
--- het radioactieve gas van slecht geventileerde kelders.
\end{example}

\begin{example}[Een bèta-minverval]\label{ex:g11:nucleus-radioactivity:beta}
Koolstof-14 ligt boven de band ($8$ neutronen tegen $6$ protonen):
${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} + {}^{\;0}_{-1}\mathrm{e}$.
Controle: $14 = 14 + 0$ en $6 = 7 - 1$; een neutron is een proton geworden
plus het uitgestoten elektron --- de \omterm{def:g11:fundamental-interactions:weak}{zwakke wisselwerking} aan het werk.
\end{example}

\begin{remark}[Bèta-plus en gamma]\label{rem:g11:nucleus-radioactivity:betaplus}
Onder de band maakt het spiegelverval $\beta^+$ van een proton een neutron
plus een \emph{positron}\index{positron} ${}^{0}_{+1}\mathrm{e}$, het
antideeltje van het elektron:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$,
het werkpaard van de pet-scanner. En na de meeste vervallen komt de
dochter, die trillend geboren wordt, tot rust door er nog een $\gamma$
bovenop uit te zenden.
\end{remark}

\section{Halveringstijd en activiteit}

\begin{definition}[Halveringstijd]\label{def:g11:nucleus-radioactivity:halflife}
De \emph{halveringstijd}\index{halveringstijd} $T_{1/2}$ van een \omterm{def:g11:nucleus-radioactivity:notation}{nuclide}
is de tijd waarna de helft van elk groot monster is vervallen. Elke
volgende halveringstijd halveert wat er over is: van $N_0$ kernen zijn er
$N_0/2$ over na $T_{1/2}$, $N_0/4$ na $2\,T_{1/2}$, $N_0/8$ na
$3\,T_{1/2}$, en $N_0/2^{\,n}$ na $n$ halveringstijden.
\end{definition}

\begin{remark}
Willekeur voor één kern, uurwerk voor $10^{23}$: een reusachtige menigte
muntgooiers, stuk voor stuk onvoorspelbaar, waarvan elke ronde de helft
afvalt. Interpoleren \emph{tussen} hele \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} vergt de
exponentiële functie --- volgend jaar, met het wiskundevolume.
\end{remark}

\begin{example}[Twintig machten van tien]\label{ex:g11:nucleus-radioactivity:range}
\begin{center}\small\begin{tabular}{llll}
polonium-214 & $\alpha$ & \qty{1.6e-4}{s} & schakel in de radiumketen \\
technetium-99m & $\gamma$ & $6.0$ uur & medische beeldvorming \\
americium-241 & $\alpha$ & $432$ jaar & rookmelders \\
koolstof-14 & $\beta^-$ & \num{5700} jaar & archeologie \\
kalium-40 & $\beta$ & \num{1.25e9} jaar & in elke banaan \\
uranium-238 & $\alpha$ & \num{4.5e9} jaar & inwendige warmte van de Aarde \\
\end{tabular}\end{center}
\end{example}

\begin{example}[Dateren met koolstof-14]\label{ex:g11:nucleus-radioactivity:dating}
Levende materie wisselt koolstof uit met de atmosfeer en bevat dus het
atmosferische aandeel koolstof-14; bij de dood stopt de aanvoer en
halveert dat aandeel elke \num{5700} jaar. Houtskool uit een beschilderde
grot vertoont een kwart van het levende aandeel: een kwart is de helft van
de helft, dus brandde het vuur twee \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden},
$2 \times \num{5700} = \num{11400}$ jaar, geleden.
\end{example}

\begin{definition}[Activiteit]\label{def:g11:nucleus-radioactivity:activity}
De \emph{activiteit}\index{activiteit} $\mathcal A$ van een monster is het
aantal vervallen per seconde, gemeten in
\emph{becquerel}\index{becquerel}: $\qty{1}{Bq} = 1$ verval per seconde.
Minder overgebleven kernen betekent minder vervallen, dus halveert ook de
activiteit bij elke \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}.
\end{definition}

\begin{example}[Ordegrootten]\label{ex:g11:nucleus-radioactivity:orders}
Alles is een beetje radioactief. Een banaan: ongeveer \qty{15}{Bq}
(kalium-40). Een menselijk lichaam: ongeveer \qty{8000}{Bq} --- de
aanhef van dit hoofdstuk, zo'n vijfhonderd bananen waard. Een kubieke
\omterm{def:g10:orders-of-magnitude:unit}{meter} graniet: rond \qty{3e6}{Bq} (uranium en zijn keten, vandaar het
radon in kelders). Eén injectie technetium: ongeveer \qty{5e8}{Bq},
binnen enkele dagen verdwenen.
\end{example}

\section{Dienaren en gevaren}

\begin{example}[Drie loopbanen van een onstabiele kern]\label{ex:g11:nucleus-radioactivity:uses}
\emph{Geneeskunde}: technetium-99m, een zuivere $\gamma$-straler met
$T_{1/2} = 6$ uur, wordt vastgemaakt aan een \omterm{def:g11:fundamental-interactions:ladder}{molecuul} dat het doelorgaan
opneemt; een camera filmt de $\gamma$'s van buitenaf, en de volgende dag is
de tracer grotendeels verdwenen --- terwijl de bestraling de logica
omkeert en bundels bundelt om een tumor te vernietigen. \emph{Dateren}:
koolstof-14 voor hout, bot en textiel tot zo'n \num{50000} jaar terug;
uranium-238 voor gesteenten --- en voor de Aarde, op \num{4.5e9} jaar.
\emph{Rookmelders}: een spikkeltje americium-241 ioniseert de lucht in een
klein kamertje, waardoor er een minieme stroom kan lopen; rook smoort die
stroom en het alarm gaat af --- de $\alpha$'s sterven in enkele centimeters
lucht.
\end{example}

\begin{remark}[Dosis en de sievert]\label{rem:g11:nucleus-radioactivity:dose}
Straling rukt elektronen van \omterm{def:g11:fundamental-interactions:ladder}{moleculen} af en kan het DNA beschadigen. De
biologische schade wordt bijgehouden met de \emph{dosis}\index{dosis}, in
\emph{sievert}\index{sievert} (\unit{Sv}), die de per \omterm{def:g10:orders-of-magnitude:unit}{kilogram} weefsel
afgezette \omterm{def:g10:energy-conservation:energy}{energie} weegt naar de schadelijkheid van elke stralingssoort.
Drie regels: afstand, afscherming ($\alpha$: de huid volstaat --- tenzij
de straler wordt \emph{ingeademd}, de misdaad van radon; $\beta$:
aluminium; $\gamma$: lood) en tijd. De dosimetrie zelf is stof voor de
universiteit.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:1}

${}^{235}_{92}\mathrm{U}$ en ${}^{3}_{1}\mathrm{H}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:2}

${}^{12}_{6}\mathrm{C}$, ${}^{13}_{6}\mathrm{C}$ en

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:6}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:12}
De keten van uranium-238 eindigt vele stappen later bij het stabiele
${}^{206}_{82}\mathrm{Pb}$; elke stap is een $\alpha$ of een $\beta^-$.
Bepaal met alleen het behoud van $A$ het aantal $\alpha$-stappen, en
daarna met het behoud van $Z$ het aantal $\beta^-$-stappen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:13}
Een kobalt-60-bron van een ziekenhuis ($T_{1/2} = 5.3$ jaar) heeft een
\omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van \qty{6.4}{TBq}; de regels staan afvoer toe onder
\qty{0.1}{TBq}. Na hoeveel jaar mag de bron worden afgevoerd?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:14}
De \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van je lichaam is ongeveer \qty{8000}{Bq}. Hoeveel van je
kernen vervallen er per dag? En over een leven van $80$ jaar? Een banaan
voegt er ongeveer \qty{15}{Bq} aan toe zolang je hem verteert:
becommentarieer in één zin de krantenkoppen die elke \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} vrezen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:15}
Een rookmelder bevat \qty{0.2}{\micro\gram} americium-241
($T_{1/2} = 432$ jaar), met een \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van ongeveer \qty{25}{kBq}.
Hoeveel vervallen per dag zijn dat? Is er na tien jaar dienst veel minder
dan de helft, ongeveer de helft of veel meer dan de helft van het
americium vervallen --- en is uitputting van de bron de reden dat
melders worden vervangen? Waarom is deze $\alpha$-bron ten slotte
ongevaarlijk aan het plafond, en toch gevaarlijk als het capsuletje wordt
fijngemalen en het stof wordt ingeademd?
\end{exercise}

\section{Probleem: De mummie, de reactor en de rookmelder}

\begin{problem}[{Weekendopgave --- drie loopbanen van een onstabiele
kern: een mummie gedateerd met halveringen, de stamboom van uranium
kloppend gemaakt, een rookmelder doorgelicht, en waarom de sterren wel
sterfelijk moesten zijn}]
\label{pb:g11:nucleus-radioactivity:1}
Drie onstabiele kernen, drie taken: koolstof-14 zet een tijdstempel op elke
dode plant en elk dood dier, uranium-238 verwarmt een planeet van binnenuit
en americium-241 houdt plafonds in de gaten. Gegevens:
$T_{1/2} = \num{5700}$ jaar (C-14), \num{4.5e9} jaar (U-238) en $432$ jaar
(Am-241); levende koolstof vertoont $13.6$ vervallen per minuut per gram;
\omterm{def:g11:nucleus-radioactivity:notation}{atoomnummers}: N $7$, O $8$, Pb $82$, Bi $83$, Po $84$, Rn $86$, Ra $88$,
Th $90$, Pa $91$, U $92$, Np $93$, Am $95$.

\medskip
\noindent\textbf{Deel I --- De mummie.}
\begin{enumerate}
  \item Geef de samenstelling (protonen, neutronen) van
        ${}^{12}_{6}\mathrm{C}$ en ${}^{14}_{6}\mathrm{C}$. Waarom
        gedragen ze zich in een lichaam scheikundig identiek?
  \item Waarom is het aandeel koolstof-14 in een levend lichaam constant,
        en waarom begint het bij de dood te dalen?
  \item Schrijf de vervalvergelijking van koolstof-14 op. Zou haar
        straling door het linnen van de mummie naar buiten komen?
  \item Een gram koolstof uit het linnen vertoont $6.8$ vervallen per
        minuut. Welk deel van het levende tempo is dat? Dateer de mummie.
  \item Een tweede ``mummie'', het pronkstuk van een privéverzameling,
        vertoont $13.2$ vervallen per minuut per gram. Oordeel?
  \item Schat het tempo van een monster van tien \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} oud, en
        leg uit waarom koolstofdatering voorbij ruwweg \num{57000} jaar
        wegvalt.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De reactor onder de weide.}
\begin{enumerate}[resume]
  \item Geef de samenstelling van ${}^{238}_{92}\mathrm{U}$. Waarom hebben
        zware kernen zo'n neutronenoverschot nodig?
  \item Schrijf haar $\alpha$-vervalvergelijking op.
  \item De dochter vervalt daarna $\beta^-$, en de kleindochter opnieuw
        $\beta^-$. Schrijf beide vergelijkingen op. Welk element duikt
        weer op?
  \item De keten eindigt bij ${}^{206}_{82}\mathrm{Pb}$. Tel haar
        $\alpha$-stappen en daarna haar $\beta^-$-stappen.
  \item De Aarde ontstond \num{4.5e9} jaar geleden. Welk deel van haar
        oorspronkelijke uranium-238 is er vandaag nog over? En over nog
        eens $5$ miljard jaar?
  \item In één zin: wat drijft deze begraven vervalwarmte aan het
        oppervlak aan? (Denk aan vulkanen en drijvende continenten.)
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De rookmelder.}
\begin{enumerate}[resume]
  \item Schrijf de $\alpha$-vervalvergelijking van
        ${}^{241}_{95}\mathrm{Am}$ op.
  \item Haar \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} is \qty{33}{kBq}: hoeveel vervallen per seconde
        is dat? En per dag?
  \item Leg de melder uit: wat doen de $\alpha$-deeltjes met de lucht in
        het kamertje, en wat verandert de rook daaraan?
  \item Waarom is een $\alpha$-straler hier de juiste keuze --- en een
        $\gamma$-straler met dezelfde \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} de slechtst denkbare?
  \item Na één \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd} ($432$ jaar!) zou de \omterm{def:g11:nucleus-radioactivity:activity}{activiteit}
        \qty{16.5}{kBq} zijn; de handleiding vraagt om vervanging na tien
        jaar. Is de bron de zwakke schakel?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Waarom de sterren sterfelijk moesten zijn.}
\begin{enumerate}[resume]
  \item Stervende sterren smeedden en verstrooiden elk element zwaarder
        dan helium. Welk deel van het uranium-238 dat \num{9e9} jaar
        geleden werd gesmeed, is er vandaag nog? Waarom maakt de
        overgebleven hoeveelheid op Aarde de planeet veel jonger dan de
        oudste sterren?
  \item De radioactieve inwendige warmte van de Aarde drijft het
        vulkanisme en de platentektoniek aan, en houdt de kern in beweging
        die het magnetische schild van \cref{ch:g11:magnetic-fields}
        opwekt. Hoe zou een planeet van uitsluitend stabiele atomen eruit
        zien?
  \item In één regel, als poëzie en als boekhouding: waar zijn je atomen
        gesmeed, en wat verwarmt nog altijd de grond onder je voeten?
\end{enumerate}
\end{problem}
