\chapter{Cor e fontes de luz}\label{ch:g11:color-light-sources}

Pergunte a um físico qual é a cor de um tomate e você recebe outra
pergunta: sob que luz? O tomate só \omterm{def:g11:color-light-sources:objectcolor}{difunde} o que recebe, a lâmpada só
oferece o que o seu \omterm{def:g10:light-spectra:white}{espectro} contém, e o olho comprime tudo o que chega
em três números. A cor é uma negociação entre fonte, objeto e olho ---
este capítulo interroga as três partes, uma de cada vez.

\section{Enxergar em três números}

\begin{definition}[Cones e visão tricromática]\label{def:g11:color-light-sources:cones}
A \omterm{def:g11:lenses-and-eye:eye}{retina} (\cref{ch:g11:lenses-and-eye}) carrega duas famílias de células
sensíveis à luz: os \emph{bastonetes}\index{bastonete}, que respondem a
luz fraca mas ignoram o \omterm{def:g10:light-spectra:wavelength}{comprimento de onda} (visão noturna, em cinza), e
os \emph{cones}\index{cone}, de três tipos, mais sensíveis perto de
$\qty{420}{nm}$ (azul), $\qty{530}{nm}$ (verde) e $\qty{560}{nm}$
(vermelho). A visão humana de cores é \emph{tricromática}\index{visão
tricromática}: seja qual for o \omterm{def:g10:light-spectra:white}{espectro}, o cérebro recebe apenas três
respostas dos cones, e a cor é o que ele faz desse trio.
\end{definition}

\begin{definition}[A cor como percepção]\label{def:g11:color-light-sources:perception}
A cor é, portanto, uma \emph{percepção}, e não uma propriedade só da luz:
um \omterm{def:g10:light-spectra:white}{espectro} especifica uma intensidade em cada \omterm{def:g10:light-spectra:wavelength}{comprimento de onda}, e o
olho guarda três números. Luzes de \omterm{def:g10:light-spectra:white}{espectros} diferentes mas com as mesmas
três respostas dos \omterm{def:g11:color-light-sources:cones}{cones} parecem rigorosamente iguais: são
\emph{metâmeras}\index{metâmera}.
\end{definition}

\begin{example}[Dois amarelos]\label{ex:g11:color-light-sources:yellows}
A linha de $\qty{589}{nm}$ de uma lâmpada de sódio excita os \omterm{def:g11:color-light-sources:cones}{cones}
vermelho e verde quase igualmente: o cérebro diz \emph{amarelo}. Uma tela
que acenda os seus subpixels vermelho ($\qty{610}{nm}$) e verde
($\qty{540}{nm}$) dá o mesmo amarelo sem luz alguma perto de
$\qty{589}{nm}$. Um prisma separa as duas na hora; o olho, nunca.
\end{example}

\section{Somar luzes}

\begin{definition}[Síntese aditiva]\label{def:g11:color-light-sources:additive}
Superpor feixes de luz sobre um anteparo branco soma os \omterm{def:g10:light-spectra:white}{espectros} deles:
isso é a \emph{síntese aditiva}\index{síntese aditiva}. Três luzes bem
escolhidas, vermelha, verde e azul --- as \emph{primárias
aditivas}\index{cores primárias} ---, misturam-se em quase toda cor
perceptível: V $+$ Ve $=$ amarelo, V $+$ A $=$ magenta, Ve $+$ A $=$
ciano e V $+$ Ve $+$ A $=$ branco.
\end{definition}

\begin{proposition}[Cores complementares]\label{prop:g11:color-light-sources:complementary}
Duas luzes cuja soma é percebida como branca são
\emph{complementares}\index{cores complementares}. Os três pares
amarelo/azul, ciano/vermelho e magenta/verde são \omterm{prop:g11:color-light-sources:complementary}{complementares}.
\end{proposition}

\begin{proof}
O amarelo é V $+$ Ve; acrescentar o azul completa o trio até o branco. Os
outros dois pares seguem o mesmo argumento, uma primária de cada vez.
\end{proof}

\begin{example}[Telas]\label{ex:g11:color-light-sources:screen}
Cada pixel da tela de um celular é feito de três subpixels ---
vermelho, verde, azul, com menos de $\qty{0.1}{mm}$ de largura ---
próximos demais para o olho separar, de modo que as luzes deles se somam
na \omterm{def:g11:lenses-and-eye:eye}{retina}: o amarelo é V e Ve acesos, o laranja é V a plena \omterm{def:g10:energy-conservation:power}{potência} e Ve
pela metade, e o branco são os três. Uma \omterm{def:g11:lenses-and-eye:magnifier}{lupa} sobre uma área branca
revela o truque: não há branco nenhum ali, só V, Ve e A.
\end{example}

\section{Subtrair da luz branca}

\begin{definition}[Síntese subtrativa]\label{def:g11:color-light-sources:subtractive}
Um \emph{filtro}\index{filtro} (vidro colorido, uma gelatina, uma camada
de tinta) transmite parte do \omterm{def:g10:light-spectra:white}{espectro} que recebe e absorve o resto.
Produzir cores \emph{retirando} faixas da \omterm{def:g10:light-spectra:white}{luz branca} é a \emph{síntese
subtrativa}\index{síntese subtrativa}. As suas primárias, \omterm{prop:g11:color-light-sources:complementary}{complementares}
das aditivas, subtraem cada uma uma faixa: ciano $=$ branco $-$ V,
magenta $=$ branco $-$ Ve, amarelo $=$ branco $-$ A. Filtros empilhados
subtraem em conjunto: ciano $+$ amarelo deixa verde, ciano $+$ magenta
deixa azul, magenta $+$ amarelo deixa vermelho, e os três juntos deixam
preto. A impressão e a pintura funcionam assim, com tintas C, M e Am.
\end{definition}

\begin{method}[Contabilidade de espectros]\label{met:g11:color-light-sources:bookkeeping}
Para prever uma cor ao longo de qualquer cadeia de fontes, \omterm{def:g11:color-light-sources:subtractive}{filtros} e
objetos, acompanhe as três faixas primárias: escreva a luz da fonte pelo
seu conteúdo em V, Ve e A; em cada \omterm{def:g11:color-light-sources:subtractive}{filtro} ou superfície, apague as faixas
absorvidas; nomeie a cor do que sobreviver (nada sobrando $=$ preto).
\end{method}

\begin{example}[Uma cadeia de filtros]\label{ex:g11:color-light-sources:chain}
\omterm{def:g10:light-spectra:white}{Luz branca} $(V, Ve, A)$ atravessa um \omterm{def:g11:color-light-sources:subtractive}{filtro} amarelo: a faixa azul é
absorvida, restando $(V, Ve)$ --- amarelo. Um \omterm{def:g11:color-light-sources:subtractive}{filtro} ciano em seguida
absorve a faixa vermelha, restando $(Ve)$: verde. Trocar a ordem dos
\omterm{def:g11:color-light-sources:subtractive}{filtros} não muda nada: cada um absorve a sua própria faixa, seja qual for
a ordem.
\end{example}

\section{A cor de um objeto}

\begin{definition}[Difusão e cor do objeto]\label{def:g11:color-light-sources:objectcolor}
Um objeto opaco absorve parte da luz que recebe e
\emph{difunde}\index{difusão} o resto --- reemite-o em todas as direções,
inclusive na dos nossos olhos. A \emph{cor de um objeto}\index{cor de um
objeto} é a cor da luz que ele difunde \emph{sob a iluminação dada}: o
objeto fornece os seus hábitos de absorção; a fonte, a matéria-prima.
Troque a lâmpada e você troca a cor.
\end{definition}

\begin{example}[A camisa vermelha]\label{ex:g11:color-light-sources:redshirt}
Uma camisa parece vermelha à luz do dia: ela absorve as faixas verde e
azul e \omterm{def:g11:color-light-sources:objectcolor}{difunde} a vermelha. Sob uma luz verde pura não há faixa vermelha
para difundir: a camisa absorve tudo e aparece preta. Sob luz magenta
$(V, A)$ ela \omterm{def:g11:color-light-sources:objectcolor}{difunde} apenas a faixa vermelha: continua vermelha. Um
objeto branco \omterm{def:g11:color-light-sources:objectcolor}{difunde} todas as faixas que recebe (sob luz verde ele
parece verde); um objeto preto as absorve todas sob qualquer luz --- e é
também por isso que roupa preta esquenta no sol.
\end{example}

\section{Fontes de luz e temperatura de cor}

\begin{definition}[Fontes incandescentes]\label{def:g11:color-light-sources:incandescent}
Uma fonte que brilha por estar quente --- chama, filamento, estrela ---
brilha por \emph{incandescência}\index{incandescência}. O seu \omterm{def:g10:light-spectra:white}{espectro} é
o \omterm{def:g10:light-spectra:white}{espectro} térmico contínuo do \cref{ch:g10:light-spectra}, e o seu pico
obedece à lei de Wien (\cref{prop:g10:light-spectra:wien}),
\[
\lambda_{\max} \times T = \qty{2.90e-3}{m.K},
\]
a nossa ferramenta quantitativa daqui em diante: pico medido,
temperatura conhecida.
\end{definition}

\begin{example}[Uma lâmpada muito ineficiente]\label{ex:g11:color-light-sources:bulb}
Um filamento de tungstênio a $T = \qty{2700}{K}$ tem pico em
$\lambda_{\max} = \qty{2.90e-3}{m.K}/\qty{2700}{K} = \qty{1.07e-6}{m}
= \qty{1070}{nm}$, bem no \omterm{def:g10:light-spectra:wavelength}{infravermelho}: a maior parte da \omterm{def:g10:energy-conservation:power}{potência}
elétrica sai como radiação invisível, e só cerca de $5\%$ como luz --- a
lâmpada incandescente era sobretudo um aquecedor.
\end{example}

\begin{definition}[Lâmpadas espectrais e lasers]\label{def:g11:color-light-sources:lamps}
Uma \emph{lâmpada espectral}\index{lâmpada espectral} --- um gás a baixa
\omterm{def:g10:pressure:pressure}{pressão} excitado por uma descarga: sódio, mercúrio, neônio --- emite um
\omterm{def:g10:light-spectra:emission}{espectro de linhas} (\cref{ch:g10:light-spectra}): alguns \omterm{def:g10:light-spectra:wavelength}{comprimentos de
onda} isolados e nada entre eles. Um \emph{laser}\index{laser} vai além:
um único \omterm{def:g10:light-spectra:wavelength}{comprimento de onda}, \omterm{def:g10:light-spectra:white}{luz monocromática}, num feixe estreito.
\end{definition}

\begin{definition}[LEDs]\label{def:g11:color-light-sources:led}
Um \emph{diodo emissor de luz}\index{diodo emissor de luz} (LED) emite
uma faixa \emph{quase monocromática}\index{quase monocromática} de
algumas dezenas de nanômetros de largura: estreita o bastante para
parecer uma cor definida, e muito mais larga do que a linha de um \omterm{def:g11:color-light-sources:lamps}{laser}.
Um LED branco é um LED azul (pico perto de $\qty{450}{nm}$) recoberto por
um \emph{fósforo}\index{fósforo}, material que absorve parte do azul e o
reemite como uma faixa amarela larga: como azul e amarelo são
\omterm{prop:g11:color-light-sources:complementary}{complementares} (\cref{prop:g11:color-light-sources:complementary}), a
mistura se lê como branco.
\end{definition}

\begin{definition}[Temperatura de cor]\label{def:g11:color-light-sources:colortemp}
A \emph{temperatura de cor}\index{temperatura de cor} de uma fonte
esbranquiçada é a temperatura do corpo incandescente cuja luz tem o mesmo
tom que a dela: vela $\qty{1800}{K}$, lâmpada incandescente
$\qty{2700}{K}$, halógena $\qty{3400}{K}$, sol do meio-dia
$\qty{5500}{K}$, luz do dia com céu encoberto $\qty{6500}{K}$. É um
rótulo de cor, e não uma leitura de termômetro --- um LED de
$\qty{6500}{K}$ funciona à temperatura ambiente --- e o vocabulário corre
ao contrário: temperatura de cor baixa se lê como laranja ``quente'', e
alta como azulada ``fria''.
\end{definition}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:1}
Dê a cor percebida quando um anteparo branco recebe simultaneamente:
(a) luz vermelha e verde; (b) luz verde e azul; (c) as três primárias.
Que luz é complementar do azul?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:2}
Que subpixels (V, Ve, A) uma tela acende para exibir amarelo, magenta,
branco, preto e laranja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:3}
\omterm{def:g10:light-spectra:white}{Luz branca} atravessa um \omterm{def:g11:color-light-sources:subtractive}{filtro} magenta. Que faixa é absorvida e que cor
emerge? Um \omterm{def:g11:color-light-sources:subtractive}{filtro} ciano é então colocado atrás dele: o que emerge agora?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:4}
Sob \omterm{def:g10:light-spectra:white}{luz branca}, uma bandeira mostra uma faixa amarela. Que faixas ela
\omterm{def:g11:color-light-sources:objectcolor}{difunde} e quais absorve? Qual é a cor dela sob (a) luz vermelha; (b) luz
azul?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:5}
O \omterm{def:g10:light-spectra:continuous}{espectro contínuo} de uma estrela tem pico em
$\lambda_{\max} = \qty{580}{nm}$. Calcule a sua temperatura superficial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:6}
A chama de uma vela fica a cerca de $\qty{1800}{K}$. (a) Calcule
$\lambda_{\max}$ e dê o seu domínio. (b) Ainda assim, a chama parece
alaranjada: explique.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:7}
Um ciclista veste uma jaqueta vermelha e luvas cianas. Usando o
\cref{met:g11:color-light-sources:bookkeeping}, dê a cor aparente de cada
peça sob (a) \omterm{def:g10:light-spectra:white}{luz branca}; (b) luz vermelha; (c) luz verde.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:8}
Uma rua antiga é iluminada por lâmpadas de sódio a baixa \omterm{def:g10:pressure:pressure}{pressão} (uma
única linha em $\qty{589}{nm}$). (a) Que tipo de \omterm{def:g10:light-spectra:white}{espectro} é esse, e que
cores \emph{qualquer} objeto pode assumir sob ele? (b) Um carro parece
preto sob essas lâmpadas e azul à luz do dia: explique.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:9}
Uma caneta \omterm{def:g11:color-light-sources:lamps}{laser} verde emite em $\qty{532}{nm}$. (a) O feixe dela
atravessa um prisma: compare o resultado com o de um feixe branco e
nomeie a propriedade ilustrada. (b) Em seguida ele encontra um \omterm{def:g11:color-light-sources:subtractive}{filtro}
vermelho: o que sai?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:10}
Descreva o \omterm{def:g10:light-spectra:white}{espectro} de um LED branco (diodo azul mais \omterm{def:g11:color-light-sources:led}{fósforo}) e explique
por que a luz dele é percebida como branca. Por que um objeto vermelho
intenso parece mais apagado sob um LED branco barato do que à luz do dia?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:11}
Calcule $\lambda_{\max}$ para um corpo incandescente (a) à \omterm{def:g11:color-light-sources:colortemp}{temperatura de
cor} de uma vela ($\qty{1800}{K}$); (b) à da luz do dia com céu encoberto
($\qty{6500}{K}$). Qual das duas luzes é chamada de ``quente''? Comente.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:12}
Uma lâmpada de sódio ($\qty{589}{nm}$) e uma tela exibindo amarelo
(subpixels em $\qty{610}{nm}$ e $\qty{540}{nm}$) têm exatamente a mesma
cor. (a) Como se chamam duas luzes assim, e o que a existência delas
prova sobre a cor? (b) Proponha duas experiências diferentes que as
distingam e preveja o resultado de cada uma.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:13}
Um refletor halógeno de teatro trabalha a $\qty{3200}{K}$. (a) Calcule
$\lambda_{\max}$: essa luz é mais quente ou mais fria do que a luz do
dia? (b) Para imitar a luz do dia, cobre-se o refletor com uma gelatina
azul-clara. O que a gelatina retira, e por que o palco fica mais escuro?
(c) Explique por que gelatina alguma poderia, em vez disso,
\emph{reforçar} a ponta azul do \omterm{def:g10:light-spectra:white}{espectro} do refletor.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:14}
Um açougueiro ilumina o balcão de carnes com LEDs de \omterm{def:g11:color-light-sources:colortemp}{temperatura de cor}
$\qty{2700}{K}$; a peixaria ao lado usa $\qty{6500}{K}$. (a) Descreva o
tom de cada iluminação e o seu efeito sobre a carne vermelha e sobre o
peixe branco. (b) Algum dos dois comerciantes mudou a sua mercadoria?
Diga exatamente onde a mudança acontece
(\cref{def:g11:color-light-sources:objectcolor}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:15}
Uma cidade vizinha a um observatório precisa reformar a iluminação das

\omterm{def:g10:light-spectra:white}{espectro} largo. (a) Que escolha os astrônomos preferem, e por quê? (Pense

escolha motoristas e pedestres preferem? Justifique com as cores dos

\section{Problema: engenharia da cor}

\begin{problem}[{Problema de fim de semana --- luzes de palco, ruas de
sódio e o LED branco: como a cor é projetada, de três refletores capazes
de pintar qualquer tom ao chip que ilumina esta página}]
\label{pb:g11:color-light-sources:1}
Um iluminador de teatro só tem refletores vermelho, verde e azul; um
engenheiro de iluminação pública só tem uma lâmpada que emite um único
\omterm{def:g10:light-spectra:wavelength}{comprimento de onda}; um fabricante de chips só tem um diodo que brilha em
azul. Os três vendem cor. Este problema percorre os ofícios deles, um a
um --- mistura aditiva no palco, economia \omterm{def:g10:light-spectra:white}{monocromática} na rua, alquimia
de \omterm{def:g11:color-light-sources:led}{fósforo} no chip --- e termina escolhendo a luz de um quarto e a de uma
vitrine de joalheria.

\medskip
\noindent\textbf{Parte I --- O palco: pintar com três refletores.}
\begin{enumerate}
  \item Os três refletores se sobrepõem num fundo branco. Diga a cor de
        cada sobreposição de duas luzes e a da sobreposição das três.
  \item Como o iluminador faz laranja? E um rosa-claro? (Descreva as
        \omterm{def:g10:energy-conservation:power}{potências} dos refletores qualitativamente.)
  \item Uma bailarina se coloca diante do fundo, iluminada apenas pelos
        refletores vermelho e verde. Aparecem duas sombras coloridas: qual
        é a cor de cada uma? Explique.
  \item O fundo é repintado de azul-escuro. Como ele fica sob o refletor
        vermelho sozinho? E sob os três?
  \item Por que três refletores conseguem reproduzir ``quase todas'' as
        cores? Que peça da anatomia fixa o número três?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Os figurinos: a subtração em ação.}
\begin{enumerate}[resume]
  \item Um figurino é amarelo sob \omterm{def:g10:light-spectra:white}{luz branca}. Que faixa o corante dele
        absorve?
  \item Preveja a cor dele sob o refletor vermelho, sob o verde e sob o
        azul.
  \item O iluminador quer que o figurino passe de brilhante a preto num
        único toque de interruptor. Para que refletor o toque muda?
  \item Uma gelatina magenta é colocada diante de um refletor branco que
        ilumina o figurino amarelo. Acompanhe as faixas
        (\cref{met:g11:color-light-sources:bookkeeping}): qual é a cor do
        figurino?
  \item Por que as gelatinas de palco vêm em ciano, magenta e amarelo, e
        não em vermelho, verde e azul? (O que uma gelatina vermelha faria
        com a \omterm{def:g10:energy-conservation:power}{potência} de um refletor branco?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ruas de sódio: a economia \omterm{def:g10:light-spectra:white}{monocromática}.}
\begin{enumerate}[resume]
  \item Uma lâmpada de sódio a baixa \omterm{def:g10:pressure:pressure}{pressão} emite essencialmente uma
        linha em $\qty{589}{nm}$. Que tipo de \omterm{def:g10:light-spectra:white}{espectro} é esse, e como a
        rua inteira aparece sob ele?
  \item Sob essa lâmpada, dê a cor aparente de uma parede branca, de uma
        faixa amarela de sinalização, de um carro vermelho e de um carro
        azul.
  \item Por que os astrônomos adoravam essas lâmpadas? (O que um único
        \omterm{def:g11:color-light-sources:subtractive}{filtro} pode fazer com o brilho da cidade inteira?)
  \item Propõe-se, em vez disso, uma lâmpada de rua de tungstênio a
        $\qty{2700}{K}$. Calcule o seu $\lambda_{\max}$: para onde vai a
        maior parte da \omterm{def:g10:energy-conservation:power}{potência} dela?
  \item A lâmpada de sódio converte cerca de $30\%$ da sua \omterm{def:g10:energy-conservation:power}{potência}
        elétrica em luz visível, e a de tungstênio, cerca de $5\%$.
        Quantos \omterm{def:g10:energy-conservation:power}{watts} de luz cada uma entrega por $\qty{100}{W}$ de
        \omterm{def:g10:energy-conservation:power}{potência}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O LED branco e a escolha de uma luz.}
\begin{enumerate}[resume]
  \item Um LED branco é um diodo azul ($\qty{450}{nm}$) sob um \omterm{def:g11:color-light-sources:led}{fósforo}
        amarelo. Explique, com as \omterm{prop:g11:color-light-sources:complementary}{cores complementares}, por que a saída
        dele se lê como branca.
  \item Descreva o \omterm{def:g10:light-spectra:white}{espectro} dele e compare-o com o \omterm{def:g10:light-spectra:continuous}{espectro contínuo} da
        luz do dia. Que faixa está sub-representada?
  \item Deduza que objetos um LED desses reproduz mal e ligue isso ao
        açougueiro do \cref{exo:g11:color-light-sources:14}.
  \item Uma lâmpada de LED é vendida como ``$\qty{2700}{K}$''. Há alguma
        coisa dentro dela a $\qty{2700}{K}$? O que o rótulo promete?
        Calcule o $\lambda_{\max}$ do corpo incandescente que ela imita.
  \item Final --- escolha, com uma frase de justificativa cada, uma
        \omterm{def:g11:color-light-sources:colortemp}{temperatura de cor} para a lâmpada de um quarto e outra para a
        vitrine de uma joalheria; depois enuncie a lei da cor deste
        capítulo em uma linha.
\end{enumerate}
\end{problem}
