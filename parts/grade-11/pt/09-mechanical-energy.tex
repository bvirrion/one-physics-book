\chapter{Energia mecânica e sua conservação}\label{ch:g11:mechanical-energy}

Uma montanha-russa não tem motor depois da primeira subida: cada
\emph{looping}, cada disparo de velocidade dali em diante é gasto da
conta aberta na subida. Este capítulo transforma essa contabilidade em
dois números --- \omterm{def:g11:mechanical-energy:kinetic}{energia cinética} e \omterm{def:g10:energy-conservation:energy}{energia} potencial --- e numa regra:
enquanto só o \omterm{def:g10:universal-gravitation:weight}{peso} realizar \omterm{def:g11:work-of-force:work}{trabalho}, a soma dos dois não muda. O \omterm{def:g10:inertia:inventory}{atrito}
é a taxa; vamos aprender a lê-la nos livros.

\section{Energia cinética}

\begin{definition}[Energia cinética]\label{def:g11:mechanical-energy:kinetic}
Um corpo de massa $m$ (\unit{kg}) que se move com velocidade $v$
(\unit{m/s}) carrega a \emph{energia cinética}\index{energia cinética}
\[
E_k = \tfrac12\, m v^2 .
\]
Conferência de \omterm{def:g10:orders-of-magnitude:unit}{unidades}: $\unit{kg.m^2/s^2} = \unit{N.m} = \unit{J}$ ---
\omterm{def:g11:work-of-force:work}{joules}, como toda \omterm{def:g10:energy-conservation:energy}{energia} (\cref{ch:g10:energy-conservation}).
\end{definition}

\begin{example}[Ordens de grandeza]\label{ex:g11:mechanical-energy:orders}
\begin{itemize}
  \item um pedestre, \qty{70}{kg} a \qty{1.4}{m/s}:
        $E_k = \tfrac12 \times 70 \times 1.4^2 \approx \qty{69}{J}$;
  \item um carro, \qty{1300}{kg} a \qty{130}{km/h} $= \qty{36.1}{m/s}$:
        $E_k \approx \qty{8.5e5}{J}$;
  \item um trem de alta velocidade, $m = \qty{3.85e5}{kg}$, a
        \qty{320}{km/h} $= \qty{88.9}{m/s}$: $E_k \approx \qty{1.5e9}{J}$;
  \item uma bala de fuzil, \qty{8.0}{g} a \qty{800}{m/s}:
        $E_k \approx \qty{2.6}{kJ}$.
\end{itemize}
Sete \omterm{def:g10:orders-of-magnitude:oom}{ordens de grandeza} separam o passeio do trem. E $E_k$ cresce com o
\emph{quadrado} da velocidade: a \qty{130}{km/h} o carro carrega $4$
vezes a \omterm{def:g10:energy-conservation:energy}{energia} que tem a \qty{65}{km/h} --- quatro vezes o que os freios
dele precisam remover.
\end{example}

\section{O teorema da energia cinética}

\begin{theorem}[Teorema da energia cinética]\label{thm:g11:mechanical-energy:ketheorem}
\index{teorema da energia cinética}
Para um corpo de massa $m$ que se move em linha reta de $A$ até $B$, a
variação da \omterm{def:g11:mechanical-energy:kinetic}{energia cinética} é igual ao \omterm{def:g11:work-of-force:work}{trabalho} total
(\cref{ch:g11:work-of-force}) recebido:
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
com a soma percorrendo todas as \omterm{def:g10:inertia:force}{forças} que agem sobre o corpo. Ele se
estende a movimentos ao longo de qualquer caminho, o que também
admitimos aqui.
\end{theorem}
\admitted

\begin{example}[Queda livre, conferida quadro a quadro]\label{ex:g11:mechanical-energy:freefall}
Uma bola de \qty{0.200}{kg} é solta e filmada; os quadros dão:
\begin{center}\small
\begin{tabular}{lcccc}
$t$ (\unit{s}) & 0.10 & 0.20 & 0.30 & 0.40\\
altura de queda $h$ (\unit{m}) & 0.049 & 0.196 & 0.441 & 0.785\\
velocidade $v$ (\unit{m/s}) & 0.98 & 1.96 & 2.94 & 3.92\\
$E_k = \tfrac12 mv^2$ (\unit{J}) & 0.096 & 0.384 & 0.864 & 1.54\\
$W = mgh$ (\unit{J}) & 0.096 & 0.385 & 0.865 & 1.54
\end{tabular}
\end{center}
Partindo de $E_k = 0$, em cada quadro $E_k$ coincide com o \omterm{def:g11:work-of-force:work}{trabalho}
$mgh$ do \omterm{def:g10:universal-gravitation:weight}{peso}: $\Delta E_k = W(\vect P)$, ou seja, $v^2 = 2gh$ --- o
teorema, verificado com a precisão das leituras.
\end{example}

\begin{remark}[Onde mora a demonstração]\label{rem:g11:mechanical-energy:proof}
O enunciado geral --- válido ao longo de qualquer caminho, e não só de
uma reta --- decorre da segunda lei de Newton e de um pouco de cálculo,
ambos mais adiante neste livro (\cref{ch:g12:newtons-laws,%
ch:g12:work-and-energy}); até lá, a conferência da queda livre é a nossa
garantia, e admitimos a versão para caminhos curvos e a usamos à
vontade.
\end{remark}

\section{Energia potencial gravitacional}

\begin{definition}[Energia potencial gravitacional]\label{def:g11:mechanical-energy:potential}
Um corpo de massa $m$ à altitude $z$, medida para cima a partir de um
\emph{nível de referência}\index{nível de referência} escolhido onde
$z = 0$, armazena a \emph{energia potencial
gravitacional}\index{energia potencial!gravitacional}
\[
E_p = m g z, \qquad g = \qty{9.81}{N/kg}.
\]
\end{definition}

\begin{remark}[Só as diferenças importam]\label{rem:g11:mechanical-energy:reference}
Ir de $A$ até $B$ muda $E_p$ em $mg(z_B - z_A) = -W_{AB}(\vect P)$: o
oposto do \omterm{def:g11:work-of-force:work}{trabalho} do \omterm{def:g10:universal-gravitation:weight}{peso} (\cref{ch:g11:work-of-force}). O \omterm{def:g11:mechanical-energy:potential}{nível de
referência} é uma escolha livre --- o piso, o nível do mar, o tampo da
mesa --- e desloca $E_p$ em toda parte por uma constante que some de toda
diferença. Coloque-o onde os números ficarem mais simples, e não o mude
no meio do problema.
\end{remark}

\begin{example}[Um livro numa prateleira]\label{ex:g11:mechanical-energy:shelf}
Um livro de \qty{1.2}{kg} está \qty{1.8}{m} acima do piso, que por sua
vez está \qty{9.0}{m} acima da rua. Referência no piso:
$E_p = 1.2 \times 9.81 \times 1.8 \approx \qty{21}{J}$; na rua:
$E_p \approx \qty{127}{J}$. Cair até o piso libera \qty{21}{J} nas duas
contas.
\end{example}

\section{Energia mecânica e sua conservação}

\begin{definition}[Energia mecânica]\label{def:g11:mechanical-energy:em}
A \emph{energia mecânica}\index{energia mecânica} de um corpo é
\[
E_m = E_k + E_p = \tfrac12 m v^2 + m g z .
\]
\end{definition}

\begin{theorem}[Conservação da energia mecânica]\label{thm:g11:mechanical-energy:conservation}
\index{conservação da energia mecânica}
Se a única \omterm{def:g10:inertia:force}{força} que realiza \omterm{def:g11:work-of-force:work}{trabalho} sobre um corpo é o seu \omterm{def:g10:universal-gravitation:weight}{peso} (sem
\omterm{def:g10:inertia:inventory}{atrito}; as outras \omterm{def:g10:inertia:force}{forças}, como a reação de um trilho sem \omterm{def:g10:inertia:inventory}{atrito},
perpendiculares ao movimento), $E_m$ se conserva: para quaisquer dois
pontos $A$ e $B$ do movimento,
\[
\tfrac12 m v_A^2 + m g z_A = \tfrac12 m v_B^2 + m g z_B .
\]
\end{theorem}

\begin{proof}
Pelo \omterm{thm:g11:mechanical-energy:ketheorem}{teorema da energia cinética},
$E_k(B) - E_k(A) = W_{AB}(\vect P) = mg(z_A - z_B) = E_p(A) - E_p(B)$: o
que $E_k$ ganha, $E_p$ perde.
\end{proof}

\begin{example}[Toboágua]\label{ex:g11:mechanical-energy:slide}
Uma criança parte do repouso no alto de um toboágua sem \omterm{def:g10:inertia:inventory}{atrito}, de altura
$h = \qty{3.2}{m}$. Referência embaixo: $mgh = \tfrac12 mv^2$, logo
$v = \sqrt{2gh} \approx \qty{7.9}{m/s}$ (\qty{29}{km/h}) --- seja qual
for a massa \emph{e seja qual for o formato do escorregador}: reto,
curvo ou em espiral, só o desnível entra.
\end{example}

\begin{example}[Pêndulo]\label{ex:g11:mechanical-energy:pendulum}
A massa de um pêndulo é puxada de lado até subir $h = \qty{12}{cm}$ acima
do seu ponto mais baixo e depois é solta. Nos extremos $v = 0$; no fundo,
todo o $mgh$ é cinético, $v = \sqrt{2gh} \approx \qty{1.5}{m/s}$; e do
outro lado a massa sobe de volta a exatamente \qty{12}{cm}.
\end{example}

\begin{example}[Velocidade de aterrissagem num salto de esqui]\label{ex:g11:mechanical-energy:skijump}
Um saltador de esqui deixa a rampa a $v_A = \qty{25}{m/s}$ e aterrissa
$\Delta z = \qty{40}{m}$ abaixo. Desprezando a resistência do ar,
$v_B = \sqrt{v_A^2 + 2g\,\Delta z} = \sqrt{625 + 785} \approx
\qty{38}{m/s}$ --- sem precisar conhecer a \omterm{def:g10:relative-motion:trajectory}{trajetória} do voo.
\end{example}

\section{Dissipação pelo atrito}

\begin{definition}[Dissipação]\label{def:g11:mechanical-energy:dissipation}
Diz-se que a \omterm{def:g10:energy-conservation:energy}{energia} que sai da conta mecânica é
\emph{dissipada}\index{dissipação}: ela reaparece como \omterm{def:g10:energy-conservation:forms}{energia térmica}
nas superfícies que se esfregam e no ar. A \omterm{def:g10:energy-conservation:energy}{energia} total se conserva
(\cref{ch:g10:energy-conservation}); só a \omterm{def:g11:mechanical-energy:em}{energia mecânica} é que não.
\end{definition}

\begin{proposition}[O déficit mede o atrito]\label{prop:g11:mechanical-energy:deficit}
Se uma \omterm{def:g10:inertia:force}{força} de \omterm{def:g10:inertia:inventory}{atrito} $\vect f$ também realiza \omterm{def:g11:work-of-force:work}{trabalho} ao longo do
movimento,
\[
\Delta E_m = E_m(B) - E_m(A) = W_{AB}(\vect f) < 0:
\]
a perda de \omterm{def:g11:mechanical-energy:em}{energia mecânica} é igual ao \omterm{def:g11:work-of-force:work}{trabalho} do \omterm{def:g10:inertia:inventory}{atrito}.
\end{proposition}

\begin{proof}
O \omterm{thm:g11:mechanical-energy:ketheorem}{teorema da energia cinética} passa a dizer
$\Delta E_k = W_{AB}(\vect P) + W_{AB}(\vect f) = -\Delta E_p +
W_{AB}(\vect f)$; passe $\Delta E_p$ para o outro lado. O \omterm{def:g10:inertia:inventory}{atrito} se opõe
ao movimento, de modo que o \omterm{def:g11:work-of-force:work}{trabalho} dele é negativo.
\end{proof}

\begin{example}[Auditoria do trenó]\label{ex:g11:mechanical-energy:toboggan}
Uma criança com o trenó ($m = \qty{40}{kg}$) parte do repouso, desce
$h = \qty{5.0}{m}$ ao longo de \qty{12}{m} de neve e chega a
\qty{7.0}{m/s}, em vez dos $\sqrt{2gh} \approx \qty{9.9}{m/s}$ sem
\omterm{def:g10:inertia:inventory}{atrito}. Referência embaixo:
$\Delta E_m = \tfrac12 \times 40 \times 7.0^2 - 40 \times 9.81
\times 5.0 = 980 - 1962 = \qty{-982}{J}$. Logo,
$W(\vect f) = \qty{-982}{J}$ ao longo de \qty{12}{m}: uma \omterm{def:g10:inertia:force}{força} de
\omterm{def:g10:inertia:inventory}{atrito} média $f = 982/12 \approx \qty{82}{N}$ e \qty{982}{J} de calor
nos patins e na neve.
\end{example}

\begin{method}[Contabilidade de energia]\label{met:g11:mechanical-energy:bookkeeping}
\begin{enumerate}
  \item Escolha um \omterm{def:g11:mechanical-energy:potential}{nível de referência} e mantenha-o; nomeie $A$ (dados) e
        $B$ (pergunta).
  \item Escreva $E_m = \tfrac12 mv^2 + mgz$ em $A$ e em $B$.
  \item Sem \omterm{def:g10:inertia:inventory}{atrito}: $E_m(A) = E_m(B)$; resolva --- tipicamente
        $v_B = \sqrt{v_A^2 + 2g(z_A - z_B)}$.
  \item Com \omterm{def:g10:inertia:inventory}{atrito}: $E_m(B) - E_m(A) = -f\ell$ ($\ell$ = comprimento do
        caminho); resolva para a incógnita.
  \item Confira: as barras de $E_k$ e $E_p$ se empilham até $E_m$, que
        fica constante a não ser que haja \omterm{def:g10:inertia:inventory}{atrito}.
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:1}
Calcule a \omterm{def:g11:mechanical-energy:kinetic}{energia cinética} (a) de uma motoneta com condutor,
\qty{90}{kg} a \qty{25}{km/h}; (b) de um carro de \qty{1200}{kg} a
\qty{130}{km/h}. (c) Por que fator $E_k$ muda quando a velocidade dobra?
E quando triplica?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:2}
Um alpinista de \qty{75}{kg} ganha \qty{850}{m} de altitude. Calcule o
ganho de \omterm{def:g10:energy-conservation:energy}{energia} potencial. Ele depende do \omterm{def:g11:mechanical-energy:potential}{nível de referência}? E da
rota?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:3}
Um vaso de flores cai \qty{20}{m} de uma sacada. Desprezando a
resistência do ar, encontre a velocidade de chegada ao chão em \unit{m/s}
e em \unit{km/h}. Para onde foi a massa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:4}
Um carro de \qty{1200}{kg} freia de \qty{25}{m/s} até parar em
\qty{50}{m}. Use o \omterm{thm:g11:mechanical-energy:ketheorem}{teorema da energia cinética} para achar o \omterm{def:g11:work-of-force:work}{trabalho} dos
freios e depois a \omterm{def:g10:inertia:force}{força} média de frenagem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:5}
A massa de um pêndulo é solta \qty{8.0}{cm} acima do seu ponto mais
baixo. Encontre a velocidade dela no fundo e a altura alcançada do outro
lado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:6}
Usando o \cref{ex:g11:mechanical-energy:orders}: por que fator a \omterm{def:g11:mechanical-energy:kinetic}{energia
cinética} do trem supera a da bala? A que altura a própria $E_k$ dele
levantaria o trem ($h = v^2/2g$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:7}
Dois escorregadores sem \omterm{def:g10:inertia:inventory}{atrito} descem da mesma plataforma de
\qty{3.0}{m}, um reto e íngreme, outro longo e sinuoso. Compare as
velocidades de chegada (calcule-as) e os tempos de chegada (sem contas).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:8}
Uma bola é lançada verticalmente para cima a \qty{12}{m/s}. Encontre a
altura máxima alcançada e depois a velocidade a \qty{4.0}{m} --- na
subida e na descida.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:9}
Um saltador de esqui deixa a rampa a \qty{23}{m/s} e aterrissa
\qty{35}{m} abaixo. Preveja a velocidade de aterrissagem sem resistência
do ar; o valor real é maior ou menor, e por quê?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:10}
Um trenó de \qty{45}{kg} parte do repouso, desce \qty{6.0}{m} ao longo de
\qty{15}{m} de encosta e chega a \qty{8.0}{m/s}. Encontre a \omterm{def:g11:mechanical-energy:em}{energia
mecânica} perdida e depois a \omterm{def:g10:inertia:force}{força} média de \omterm{def:g10:inertia:inventory}{atrito}. Para onde foi a
\omterm{def:g10:energy-conservation:energy}{energia}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:11}
O pino de Galileu: um pêndulo solto \qty{15}{cm} acima do seu ponto mais
baixo encontra um pino que bloqueia a metade superior do fio na vertical.
Encontre a velocidade no fundo e a subida além do pino; o que o pino muda
e o que não muda?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mechanical-energy:12}
Um skatista entra num half-pipe a partir de \qty{2.5}{m}; cada travessia
(de um lado ao outro) dissipa $10\%$ da \omterm{def:g11:mechanical-energy:em}{energia mecânica}. Encontre a
velocidade no fundo da primeira descida e, em seguida --- calculando as
alturas sucessivas de subida ---, a primeira travessia que termina abaixo
de \qty{1.0}{m}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mechanical-energy:13}

\end{exercise}

\section{Problema: projetar o looping}

\begin{problem}[{Problema de fim de semana --- uma auditoria energética
de uma montanha-russa: a altura de lançamento que um looping impõe, as
velocidades que ela promete, o atrito confessado por dois sensores e os
freios que fecham a conta --- o passeio inteiro pendurado num único
número}]
\label{pb:g11:mechanical-energy:1}
Um trem de massa $m = \qty{2.0e3}{kg}$ é solto do repouso de um morro de
lançamento de altura $H$ e desliza, sem motor, por um \emph{looping}
circular vertical de raio $R = \qty{8.0}{m}$ cujo topo fica à altura
$2R$. A análise do movimento circular (no ano que vem) fornece o único
dado que tomamos emprestado: no topo do \emph{looping}, o trem só
pressiona o trilho se $v_{\text{topo}}^2 \geq gR$. As Partes I e II
ignoram o \omterm{def:g10:inertia:inventory}{atrito}.

\medskip
\noindent\textbf{Parte I --- O morro que o \emph{looping} impõe.}
\begin{enumerate}
  \item Calcule a velocidade mínima segura no topo do \emph{looping}.
  \item Com o solo como \omterm{def:g11:mechanical-energy:potential}{nível de referência}, calcule a \omterm{def:g11:mechanical-energy:em}{energia mecânica}
        do trem no topo do \emph{looping} com essa velocidade mínima.
  \item Deduza $H_{\min}$ e mostre que $H_{\min} = 5R/2$ --- sem massa e
        sem $g$.
  \item A equipe adota $H = \qty{24}{m}$: calcule a velocidade real no
        topo do \emph{looping} e a razão $v_{\text{topo}}^2/(gR)$.
  \item Para onde foi o $m$? Por que o projeto serve igualmente a trens
        lotados e vazios?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O que a altura promete}
($H = \qty{24}{m}$).
\begin{enumerate}[resume]
  \item Qual é a velocidade no fundo do \emph{looping} ($z = 0$), em
        \unit{m/s} e em \unit{km/h}?
  \item Qual é a velocidade na lateral do \emph{looping} ($z = R$)?
  \item Calcule $E_p$ e $E_k$ no lançamento, no fundo, na lateral e no
        topo do \emph{looping}; confira que cada par soma $mgH$.
  \item Uma lombada de \qty{12}{m} vem depois do \emph{looping}: qual é a
        velocidade no alto dela?
  \item Dê a fórmula geral de $v(z)$; onde o passeio é mais rápido e onde
        é mais lento?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A auditoria do \omterm{def:g10:inertia:inventory}{atrito}.}
Dois sensores à mesma altura $z = \qty{2.0}{m}$, separados por
\qty{60}{m} de trilho, leem $v_1 = \qty{20.4}{m/s}$ e depois
$v_2 = \qty{19.2}{m/s}$.
\begin{enumerate}[resume]
  \item Calcule a variação de \omterm{def:g11:mechanical-energy:em}{energia mecânica} entre os sensores. O que
        torna esperta a escolha de colocá-los na mesma altura?
  \item Deduza o \omterm{def:g11:work-of-force:work}{trabalho} do \omterm{def:g10:inertia:inventory}{atrito} e a \omterm{def:g10:inertia:force}{força} média de \omterm{def:g10:inertia:inventory}{atrito}.
  \item Para onde foi a \omterm{def:g10:energy-conservation:energy}{energia} que falta? Nomeie a forma e os lugares.
  \item Que fração da $E_m$ do trem se perde a cada \qty{60}{m}? Por que
        o $H_{\min}$ puro seria inseguro no mundo real, cheio de \omterm{def:g10:inertia:inventory}{atrito}?
  \item Cerca de \qty{90}{m} de trilho separam o lançamento do topo do
        \emph{looping}: estime $E_m$ ali; a margem da questão 4 ainda
        salva o \emph{looping}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Fechar a conta.}
O último sensor, na entrada do trecho horizontal reto de frenagem
($z = 0$), lê \qty{18.0}{m/s}.
\begin{enumerate}[resume]
  \item Calcule a \omterm{def:g11:mechanical-energy:kinetic}{energia cinética} do trem ali.
  \item Os freios precisam pará-lo em $d = \qty{45}{m}$: calcule a \omterm{def:g10:inertia:force}{força}
        de frenagem (constante) necessária.
  \item Conferência de conforto: calcule a desaceleração $a = F/m$;
        compare com $g$.
  \item Quanta \omterm{def:g10:energy-conservation:energy}{energia} o \omterm{def:g10:inertia:inventory}{atrito} do trilho dissipou no passeio inteiro?
        Verifique que \omterm{def:g10:inertia:inventory}{atrito} $+$ freios $= mgH$, \omterm{def:g11:work-of-force:work}{joule} a \omterm{def:g11:work-of-force:work}{joule}.
  \item Desfecho, em uma frase: que único número fixou a segurança do
        \emph{looping}, todas as velocidades, a conta do \omterm{def:g10:inertia:inventory}{atrito} e a \omterm{def:g10:inertia:force}{força}
        dos freios --- e qual foi o preço de ignorar o \omterm{def:g10:inertia:inventory}{atrito}?
\end{enumerate}
\end{problem}
