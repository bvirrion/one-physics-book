\chapter{Lentes, imagens e o olho}\label{ch:g11:lenses-and-eye}

Toda imagem nítida --- no sensor de um celular, numa tela de cinema ou no
fundo do seu próprio olho --- vem do mesmo truque: um meio transparente
curvo desvia os raios por \omterm{def:g10:refraction:refraction}{refração} (\cref{ch:g10:refraction}) de modo que
todos os raios que saem de um ponto voltem a se encontrar em outro. Este
capítulo domina o truque: uma \omterm{def:g11:lenses-and-eye:lens}{lente}, três raios, uma fórmula --- e depois
o aponta para câmeras, óculos e a \omterm{def:g11:lenses-and-eye:lens}{lente} viva que lê esta frase.

\section{Lentes convergentes e divergentes}

\begin{definition}[Lente delgada]\label{def:g11:lenses-and-eye:lens}
Uma \emph{lente}\index{lente} é um meio transparente limitado por duas
superfícies, pelo menos uma delas curva. Mais grossa no centro do que na
borda, ela é \emph{convergente}\index{lente convergente}: raios paralelos
são desviados uns em direção aos outros; mais grossa na borda, é
\emph{divergente}\index{lente divergente}: eles se afastam. Uma lente
\emph{delgada}\index{lente delgada} tem espessura desprezível: nós a
desenhamos como um segmento com pontas de flecha para fora (convergente)
ou para dentro (divergente). O seu centro $O$ é o \emph{centro
óptico}\index{centro óptico}; a reta perpendicular que passa por $O$ é o
\emph{eixo principal}\index{eixo principal}.
\end{definition}

\begin{definition}[Focos e distância focal]\label{def:g11:lenses-and-eye:focal}
Os raios que chegam paralelos ao eixo de uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} cruzam
todos o eixo num único ponto atrás dela, o \emph{foco imagem}\index{foco}
$F'$; simetricamente, os raios que saem do \emph{foco objeto} $F$ (à
frente, com $OF = OF'$) emergem paralelos ao eixo. A \emph{distância
focal}\index{distância focal} $f' = \overline{OF'}$ é positiva para uma
\omterm{def:g11:lenses-and-eye:lens}{lente convergente}; numa \omterm{def:g11:lenses-and-eye:lens}{lente divergente}, os raios paralelos emergem como
se viessem de um ponto $F'$ situado \emph{à frente} da \omterm{def:g11:lenses-and-eye:lens}{lente}: $f' < 0$.
\end{definition}

\begin{definition}[Vergência]\label{def:g11:lenses-and-eye:vergence}
A \emph{vergência}\index{vergência} de uma \omterm{def:g11:lenses-and-eye:lens}{lente} de \omterm{def:g11:lenses-and-eye:focal}{distância focal} $f'$
(em \omterm{def:g10:orders-of-magnitude:unit}{metros}) é $C = 1/f'$, medida em \emph{dioptrias}\index{dioptria}
(símbolo D): $1~\text{D} = \qty{1}{m^{-1}}$; as \omterm{def:g11:lenses-and-eye:lens}{lentes divergentes} têm
$C < 0$. Os oculistas prescrevem em dioptrias: as vergências de \omterm{def:g11:lenses-and-eye:lens}{lentes
delgadas} em contato simplesmente se somam (admitido aqui; deduzido no
volume do primeiro ano de graduação).
\end{definition}

\begin{example}[Ler uma receita]\label{ex:g11:lenses-and-eye:vergence}
Uma \omterm{def:g11:lenses-and-eye:lens}{lente} com $f' = \qty{0.50}{m}$ tem $C = 1/0.50 = +2.0$~D:
\omterm{def:g11:lenses-and-eye:lens}{convergente}. Uma receita de $-4.0$~D significa $f' = \qty{-0.25}{m}$:
\omterm{def:g11:lenses-and-eye:lens}{divergente}, com \omterm{def:g11:lenses-and-eye:focal}{distância focal} de \qty{25}{cm}.
\end{example}

\section{Construir a imagem}

\begin{method}[Os três raios de construção]\label{met:g11:lenses-and-eye:rays}
Seja $AB$ um objeto perpendicular ao eixo em $A$. Entre todos os raios
que saem da ponta $B$, três têm caminho conhecido:
\begin{enumerate}
  \item o raio que passa pelo \omterm{def:g11:lenses-and-eye:lens}{centro óptico} $O$ segue em linha reta;
  \item o raio paralelo ao eixo emerge passando por $F'$;
  \item o raio que passa por $F$ emerge paralelo ao eixo.
\end{enumerate}
Dois quaisquer bastam: a interseção deles é o ponto imagem $B'$, e a
imagem $A'B'$ fica perpendicular ao eixo em $A'$. Se os raios emergentes
divergirem, os seus prolongamentos \emph{para trás} (tracejados) se
encontram num ponto de \omterm{def:g11:lenses-and-eye:images}{imagem virtual}.
\end{method}

\begin{definition}[Imagens reais e virtuais]\label{def:g11:lenses-and-eye:images}
Uma imagem é \emph{real}\index{imagem real} quando os raios emergentes
passam efetivamente por ela: um anteparo colocado ali a recolhe; é
\emph{virtual}\index{imagem virtual} quando só os prolongamentos para
trás se encontram: anteparo nenhum a recolhe, mas um olho que olhe para
dentro da \omterm{def:g11:lenses-and-eye:lens}{lente} a vê perfeitamente bem. Uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} dá uma
imagem real e invertida de qualquer objeto além de $F$; deslize o objeto
para dentro da \omterm{def:g11:lenses-and-eye:focal}{distância focal} e a imagem se torna virtual, direita e
ampliada.
\end{definition}

\section{Uma fórmula: a relação de conjugação}

As construções mostram \emph{onde} está a imagem; uma fórmula a calcula.
Oriente o eixo no sentido da luz e use \emph{medidas algébricas}:
$\overline{OA}$ é a coordenada de $A$ em relação a $O$ (negativa para um
objeto \omterm{def:g11:lenses-and-eye:images}{real} à frente); as alturas $\overline{AB}$ contam positivas para
cima.

\begin{theorem}[Relação de conjugação das lentes delgadas]\label{thm:g11:lenses-and-eye:conjugate}
Através de uma \omterm{def:g11:lenses-and-eye:lens}{lente delgada} de \omterm{def:g11:lenses-and-eye:focal}{distância focal} $f'$, um ponto objeto $A$
do eixo tem imagem no ponto $A'$, com
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'} = C .
\]
\end{theorem}
\admitted

\begin{remark}
A relação decorre da \omterm{thm:g10:refraction:snell}{lei de Snell} aplicada a raios próximos do eixo; a
dedução --- e os limites do modelo de \omterm{def:g11:lenses-and-eye:lens}{lente delgada} --- são feitos com
honestidade no volume do primeiro ano de graduação.
\end{remark}

\begin{definition}[Aumento]\label{def:g11:lenses-and-eye:magnification}
O \emph{aumento}\index{aumento} da imagem é
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}}
       = \frac{\overline{OA'}}{\overline{OA}} .
\]
$\abs{\gamma}$ compara os tamanhos e o seu sinal compara as orientações
($\gamma < 0$: invertida); o sinal de $\overline{OA'}$ informa a natureza
(positivo: \omterm{def:g11:lenses-and-eye:images}{real}, atrás da \omterm{def:g11:lenses-and-eye:lens}{lente}; negativo: \omterm{def:g11:lenses-and-eye:images}{virtual}, à frente).
\end{definition}

\begin{example}[A fórmula em ação]\label{ex:g11:lenses-and-eye:conjugate}
Um objeto está \qty{30}{cm} à frente de uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} com
$f' = \qty{10}{cm}$: $\overline{OA} = \qty{-30}{cm}$, logo
\[
\frac{1}{\overline{OA'}} = \frac{1}{10} - \frac{1}{30} = \frac{1}{15},
\qquad \overline{OA'} = +\qty{15}{cm},
\qquad \gamma = \frac{15}{-30} = -0.50 :
\]
uma \omterm{def:g11:lenses-and-eye:images}{imagem real} \qty{15}{cm} atrás da \omterm{def:g11:lenses-and-eye:lens}{lente}, invertida, com metade do
tamanho --- exatamente o que os três raios desenham.
\end{example}

\section{O olho}

\begin{definition}[O olho como sistema óptico]\label{def:g11:lenses-and-eye:eye}
Opticamente, o olho é uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} diante de um anteparo: a
\emph{córnea}\index{córnea} e o \emph{cristalino}\index{cristalino} agem
como uma única \omterm{def:g11:lenses-and-eye:lens}{lente convergente} de \omterm{def:g11:lenses-and-eye:focal}{distância focal} ajustável; a
\emph{retina}\index{retina}, cerca de \qty{17}{mm} atrás, é o anteparo
fixo. O \omterm{def:g11:lenses-and-eye:focal}{foco} é obtido apertando a \omterm{def:g11:lenses-and-eye:lens}{lente}, e não deslocando-a como numa
câmera: a \emph{acomodação}\index{acomodação} é a variação de $f'$
produzida pelos músculos ciliares. O ponto mais próximo visto com nitidez
com acomodação máxima é o \emph{ponto próximo}\index{ponto próximo} ---
cerca de \qty{25}{cm} para o \emph{olho padrão}\index{olho padrão} --- e
o mais distante, em repouso, é o \emph{ponto remoto}\index{ponto remoto}
(o infinito, para o olho padrão).
\end{definition}

\begin{example}[A vergência do seu olho]\label{ex:g11:lenses-and-eye:eyerange}
A \omterm{def:g11:lenses-and-eye:eye}{retina} fica em $\overline{OA'} = +\qty{0.017}{m}$. Objeto distante
($1/\overline{OA} \approx 0$): $C = 1/0.017 \approx 59$~D. \omterm{def:g11:lenses-and-eye:eye}{Ponto próximo}
($\overline{OA} = \qty{-0.25}{m}$):
$C = 1/0.017 + 1/0.25 \approx 63$~D. Ler custa cerca de $4$~D: o olho é
uma \omterm{def:g11:lenses-and-eye:lens}{lente} de $59$~D com um ajuste fino embutido de $+4$~D.
\end{example}

\section{Defeitos, correções e a lupa}

\begin{definition}[Miopia e hipermetropia]\label{def:g11:lenses-and-eye:defects}
Um olho \emph{míope}\index{miopia} é \omterm{def:g11:lenses-and-eye:lens}{convergente} demais para o seu
comprimento: os raios paralelos focalizam \emph{antes} da \omterm{def:g11:lenses-and-eye:eye}{retina}, os
objetos distantes ficam borrados, e o \omterm{def:g11:lenses-and-eye:eye}{ponto remoto} fica a distância
finita; uma \omterm{def:g11:lenses-and-eye:lens}{lente divergente} de óculos o corrige. Um olho
\emph{hipermetrope}\index{hipermetropia} não é \omterm{def:g11:lenses-and-eye:lens}{convergente} o bastante: os
objetos próximos focalizariam atrás da \omterm{def:g11:lenses-and-eye:eye}{retina} e o \omterm{def:g11:lenses-and-eye:eye}{ponto próximo} se
afasta; uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} o corrige. O envelhecimento enrijece o
\omterm{def:g11:lenses-and-eye:eye}{cristalino} (\emph{presbiopia}\index{presbiopia}), reduzindo a \omterm{def:g11:lenses-and-eye:eye}{acomodação}
--- e o remédio é o mesmo: óculos de leitura.
\end{definition}

\begin{example}[Receitar para um míope]\label{ex:g11:lenses-and-eye:correction}
Um olho \omterm{def:g11:lenses-and-eye:defects}{míope} tem o \omterm{def:g11:lenses-and-eye:eye}{ponto remoto} a \qty{50}{cm}: a \omterm{def:g11:lenses-and-eye:lens}{lente} corretora
precisa mostrar os objetos distantes \emph{no \omterm{def:g11:lenses-and-eye:eye}{ponto remoto}} --- objeto no
infinito, imagem em $\overline{OA'} = \qty{-0.50}{m}$ (\omterm{def:g11:lenses-and-eye:images}{virtual}, à
frente). Então $1/f' = 1/\overline{OA'}$: $f' = \qty{-0.50}{m}$,
$C = -2.0$~D; o olho relaxado olha para essa \omterm{def:g11:lenses-and-eye:images}{imagem virtual} e a vê com
nitidez.
\end{example}

\begin{definition}[A lupa]\label{def:g11:lenses-and-eye:magnifier}
Uma \emph{lupa}\index{lupa} é uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} de \omterm{def:g11:lenses-and-eye:focal}{distância focal}
curta usada com o objeto \emph{dentro} da \omterm{def:g11:lenses-and-eye:focal}{distância focal}: a imagem é
\omterm{def:g11:lenses-and-eye:images}{virtual}, direita e ampliada (\cref{def:g11:lenses-and-eye:images}), e o
olho a observa confortavelmente longe.
\end{definition}

\begin{example}[A lupa do joalheiro]\label{ex:g11:lenses-and-eye:magnifier}
\omterm{def:g11:lenses-and-eye:magnifier}{Lupa} com $f' = \qty{5.0}{cm}$, pedra a \qty{4.0}{cm}:
$1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$, logo
$\overline{OA'} = \qty{-20}{cm}$ e $\gamma = (-20)/(-4.0) = +5.0$: uma
\omterm{def:g11:lenses-and-eye:images}{imagem virtual}, direita, cinco vezes maior, a confortáveis \qty{20}{cm}
da \omterm{def:g11:lenses-and-eye:lens}{lente}.
\end{example}

\section{Problema: uma fórmula, três instrumentos}

\begin{problem}[{Problema de fim de semana --- a mesma relação de
conjugação focaliza uma câmera, receita óculos de leitura, projeta uma
lupa e mede o zoom embutido no seu próprio olho}]
\label{pb:g11:lenses-and-eye:1}
Um fotógrafo ajusta uma objetiva de \qty{50}{mm}, uma avó segura o jornal
a distância de braço, um joalheiro se debruça sobre um diamante: uma
linha de álgebra,
$1/\overline{OA'} - 1/\overline{OA} = 1/f'$
(\cref{thm:g11:lenses-and-eye:conjugate}), governa os três ofícios --- e
depois o próprio olho.

\medskip
\noindent\textbf{Parte I --- Uma fórmula.} Uma \omterm{def:g11:lenses-and-eye:lens}{lente convergente} tem
$f' = \qty{20}{cm}$.
\begin{enumerate}
  \item Um objeto a \qty{60}{cm}: calcule $\overline{OA'}$ e $\gamma$;
        descreva a imagem.
  \item Mova o objeto para \qty{30}{cm}; recalcule. O que foi trocado em
        relação à questão 1? (Os percursos da luz são reversíveis.)
  \item Mova-o para \qty{10}{cm}, dentro de $f'$: recalcule e descreva.
  \item Resuma: à medida que o objeto vem de muito longe até $F$, e
        depois o ultrapassa, como a imagem se desloca e muda de
        natureza?
  \item Mostre que um objeto muito distante ($1/\overline{OA} \approx 0$)
        sempre tem imagem no plano focal: $\overline{OA'} = f'$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O fotógrafo.}
Uma objetiva de câmera tem $f' = \qty{50}{mm}$; o sensor, de \qty{24}{mm}
de altura, pode ser deslocado em relação à \omterm{def:g11:lenses-and-eye:lens}{lente}.
\begin{enumerate}[resume]
  \item Onde fica o sensor quando ela está focalizada numa paisagem
        distante?
  \item Agora um rosto a \qty{1.0}{m}: calcule a nova distância
        lente--sensor e o curso a partir do ajuste da paisagem.
  \item O mecanismo permite no máximo \qty{55}{mm} de distância
        lente--sensor. Calcule a distância mínima de focalização.
  \item Nessa focalização mais próxima, calcule $\gamma$. A imagem de um
        rosto de \qty{20}{cm} cabe no sensor?
  \item ``Macro'' em tamanho \omterm{def:g11:lenses-and-eye:images}{real} significa $\gamma = -1$: onde precisam
        estar o objeto e o sensor, e por que isso exige uma objetiva
        especial?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A avó.}
Com a idade o \omterm{def:g11:lenses-and-eye:eye}{ponto próximo} se afasta: o da avó está a \qty{1.0}{m}.
\begin{enumerate}[resume]
  \item Explique o jornal a distância de braço.
  \item Para ler a \qty{25}{cm}, os óculos dela precisam transformar uma
        página a \qty{25}{cm} numa \omterm{def:g11:lenses-and-eye:images}{imagem virtual} a \qty{1.0}{m}: calcule
        a \omterm{def:g11:lenses-and-eye:vergence}{vergência} necessária.
  \item Calcule $\gamma$. A imagem é quatro vezes maior e, ainda assim,
        não parece maior: explique, comparando tamanhos \emph{e}
        distâncias.
  \item Mostre que a letra impressa só fica nítida entre \qty{25}{cm} e
        $f' \approx \qty{33}{cm}$. E um objeto \emph{distante} visto
        através dos óculos?
  \item Deduza em uma frase por que existem \omterm{def:g11:lenses-and-eye:lens}{lentes} bifocais.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O joalheiro e o zoom do próprio olho.}
A \omterm{def:g11:lenses-and-eye:magnifier}{lupa} do joalheiro tem $f' = \qty{5.0}{cm}$; o \omterm{def:g11:lenses-and-eye:eye}{ponto próximo} dele está a
\qty{25}{cm}; a \omterm{def:g11:lenses-and-eye:eye}{retina} dele fica \qty{17}{mm} atrás da \omterm{def:g11:lenses-and-eye:lens}{lente} do olho.
\begin{enumerate}[resume]
  \item Uma pedra \qty{4.0}{cm} abaixo da \omterm{def:g11:lenses-and-eye:magnifier}{lupa}: posição, natureza e
        \omterm{def:g11:lenses-and-eye:magnification}{aumento}?
  \item Onde a pedra precisa ficar para que a sua imagem caia no \omterm{def:g11:lenses-and-eye:eye}{ponto
        próximo} dele, e qual é então o $\gamma$?
  \item Por que os joalheiros experientes preferem colocar a pedra
        exatamente em $F$ (imagem no infinito)?
  \item Agora só o olho: calcule a sua \omterm{def:g11:lenses-and-eye:vergence}{vergência} olhando um objeto
        distante e depois um objeto a \qty{25}{cm}: quantas \omterm{def:g11:lenses-and-eye:vergence}{dioptrias} de
        \omterm{def:g11:lenses-and-eye:eye}{acomodação}?
  \item Converta as duas \omterm{def:g11:lenses-and-eye:vergence}{vergências} em \omterm{def:g11:lenses-and-eye:focal}{distâncias focais}: de quanto, em
        milímetros e em porcentagem, o olho muda a sua própria \omterm{def:g11:lenses-and-eye:focal}{distância
        focal} --- o zoom que a câmera fez com curso mecânico e que a avó
        remendou com $+3$~D?
\end{enumerate}
\end{problem}
