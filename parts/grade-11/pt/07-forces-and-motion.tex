\chapter{Forças e movimento}\label{ch:g11:forces-and-motion}

Atire uma chave inglesa girando pela sala: as pontas balançam de modo
caótico e, ainda assim, um ponto dentro dela descreve uma parábola
perfeita. O \cref{ch:g10:inertia} resolveu o que acontece quando as
\omterm{def:g10:inertia:force}{forças} \omterm{def:g10:inertia:compensate}{se compensam} --- nada. Este capítulo pergunta o que acontece
quando elas não \omterm{def:g10:inertia:compensate}{se compensam}: a \omterm{def:g10:inertia:force}{força} que sobra \emph{muda a velocidade}
--- acelerando corpos, freando-os, virando-os --- e diz de quanto, em
razões, se ainda não em fórmulas.

\section{O centro de massa}

\begin{definition}[Centro de massa]\label{def:g11:forces-and-motion:com}
O \emph{centro de massa}\index{centro de massa} de um corpo ou de um
sistema é o seu ponto de equilíbrio --- a posição média da sua massa: o
centro geométrico de um corpo homogêneo e simétrico; para duas massas
pontuais, o ponto do segmento que as une cujas distâncias estão na razão
\emph{inversa} das massas, $\frac{d_1}{d_2} = \frac{m_2}{m_1}$ --- mais
perto da mais pesada.
\end{definition}

\begin{example}[Um haltere desequilibrado]\label{ex:g11:forces-and-motion:dumbbell}
Bolas de \qty{2.0}{kg} e \qty{1.0}{kg} estão nas pontas de uma barra leve
de \qty{0.90}{m}. O \omterm{def:g11:forces-and-motion:com}{centro de massa} a divide na razão inversa
$d_1/d_2 = 1.0/2.0$: $\qty{0.30}{m}$ da bola pesada e $\qty{0.60}{m}$ da
leve --- equilibre a barra ali.
\end{example}

\begin{remark}[A chave inglesa arremessada]\label{rem:g11:forces-and-motion:wrench}
Filme uma chave inglesa girando no ar: cada ponto segue uma curva
complicada, cheia de laços --- exceto o \omterm{def:g11:forces-and-motion:com}{centro de massa}, que traça a
parábola limpa de uma bola arremessada. O movimento do \omterm{def:g11:forces-and-motion:com}{centro de massa} é
simples; o resto é rotação \emph{em torno} dele --- e é por isso que a
mecânica pode tratar um carro, um planeta ou um ginasta como um único
ponto.
\end{remark}

\section{Uma força resultante muda a velocidade}

\begin{definition}[Força resultante]\label{def:g11:forces-and-motion:netforce}
A \emph{força resultante}\index{força resultante} sobre um corpo é a soma
vetorial de todas as \omterm{def:g10:inertia:force}{forças} que agem sobre ele,
$\vect F = \vect F_1 + \vect F_2 + \dots$ (ponta com cauda, como no
volume de matemática); as \omterm{def:g10:inertia:force}{forças} \omterm{def:g10:inertia:compensate}{se compensam} exatamente quando
$\vect F = \vect 0$.
\end{definition}

\begin{theorem}[Efeito de uma força resultante --- semiquantitativo]\label{thm:g11:forces-and-motion:secondlaw}
\index{segunda lei do movimento}
Uma \omterm{def:g11:forces-and-motion:netforce}{força resultante} $\vect F$ que age sobre um corpo de massa $m$
durante uma duração curta $\Delta t$ muda a velocidade do \omterm{def:g11:forces-and-motion:com}{centro de massa}
dele por um vetor $\Delta \vect v$ que tem a \emph{direção e o sentido}
de $\vect F$ e cujo módulo cresce em proporção a $F$ e a $\Delta t$ e
diminui em proporção à massa:
\[
\Delta v \propto \frac{F \, \Delta t}{m} .
\]
\end{theorem}
\admitted

\begin{remark}
Dobre a \omterm{def:g10:inertia:force}{força} ou a duração dela: o dobro de variação; dobre a massa: a
metade. No ano que vem isso vira uma equação exata, e o volume do
primeiro ano de graduação pergunta em que referenciais ela vale; neste
ano as razões bastam --- elas já comandam airbags, freios e \omterm{def:g10:universal-gravitation:satellite}{órbitas}.
\end{remark}

\begin{proposition}[Os três casos canônicos]\label{prop:g11:forces-and-motion:cases}
Seja um corpo movendo-se com velocidade $\vect v$ sob uma \omterm{def:g11:forces-and-motion:netforce}{força
resultante} $\vect F$.
\begin{enumerate}
  \item $\vect F$ no sentido de $\vect v$: ele \emph{acelera} em linha
        reta;
  \item $\vect F$ no sentido oposto ao de $\vect v$: ele \emph{freia} em
        linha reta;
  \item $\vect F$ perpendicular a $\vect v$: ele \emph{vira} sem mudar de
        velocidade escalar.
\end{enumerate}
\end{proposition}

\begin{proof}
Some $\Delta\vect v$, dirigido no sentido de $\vect F$
(\cref{thm:g11:forces-and-motion:secondlaw}), ponta com cauda em
$\vect v$: no sentido de $\vect v$ as flechas se empilham (mais longa);
no sentido oposto, elas se cancelam em parte (mais curta); e
perpendicularmente, cada pequeno $\Delta \vect v$ inclina a flecha sem
alongá-la --- só a direção gira.
\end{proof}

\begin{example}[Um golpe lateral]\label{ex:g11:forces-and-motion:kick}
Um disco desliza para leste a \qty{6.0}{m/s}; um golpe lateral breve
acrescenta $\Delta\vect v = \qty{8.0}{m/s}$ rumo ao norte. A nova
velocidade é a soma vetorial: módulo
$\sqrt{6.0^2 + 8.0^2} = \qty{10.0}{m/s}$, direção
$\tan\theta = 8.0/6.0$, ou seja, $\theta \approx 53^\circ$ do leste em
direção ao norte --- a velocidade antiga não é apagada; a variação é
somada a ela.
\end{example}

\section{Comparar variações: força, tempo, massa}

\begin{method}[Raciocinar por razões]\label{met:g11:forces-and-motion:ratio}
Para comparar duas situações, nunca resolva uma equação --- forme razões:
a razão das variações de velocidade é a razão de $F$, vezes a razão de
$\Delta t$, dividida pela razão de $m$. Depois leia que fator mudou e de
quanto.
\end{method}

\begin{example}[O carrinho carregado]\label{ex:g11:forces-and-motion:cart}
O mesmo empurrão, aplicado durante o mesmo segundo, faz um carrinho de
supermercado vazio de \qty{20}{kg} partir a \qty{1.2}{m/s}. Carregado até
\qty{60}{kg} --- massa tripla, mesmo $F\Delta t$ ---, ele parte a um
terço disso, \qty{0.40}{m/s}. A massa é a teimosia da matéria.
\end{example}

\begin{remark}[Colisões, airbags, joelhos dobrados]\label{rem:g11:forces-and-motion:airbag}
Numa colisão, $\Delta v$ e $m$ não são negociáveis --- a velocidade do
ocupante tem de chegar a zero ---, de modo que $F\,\Delta t$ está fixado.
A única liberdade que resta é o \emph{tempo}: estique $\Delta t$ e a
\omterm{def:g10:inertia:force}{força} encolhe na mesma razão. Uma cabeça que para num airbag em
\qty{0.08}{s}, em vez de parar no volante em \qty{0.01}{s}, sente uma
\omterm{def:g10:inertia:force}{força} média oito vezes menor; as zonas de deformação programada, a
espuma do capacete e os joelhos dobrados fazem o mesmo truque.
\end{remark}

\section{Atrito: a força que combate o deslizamento}

\begin{definition}[Atrito estático e atrito cinético]\label{def:g11:forces-and-motion:friction}
O \omterm{def:g10:inertia:inventory}{atrito} entre superfícies em contato se opõe ao \emph{deslizamento
relativo} delas. O \emph{atrito estático}\index{atrito!estático} age
enquanto elas não deslizam, ajustando-se até um valor máximo para
cancelar o que quer que tente iniciar o deslizamento; o \emph{atrito
cinético}\index{atrito!cinético}, uma vez começado o deslizamento, é
aproximadamente constante, dirigido contra o deslizamento e em geral
\emph{mais fraco} do que o máximo estático.
\end{definition}

\begin{example}[Empurrar o guarda-roupa]\label{ex:g11:forces-and-motion:wardrobe}
Empurre um guarda-roupa de leve: ele não se move --- o \omterm{def:g11:forces-and-motion:friction}{atrito estático}
cancela os seus \qty{150}{N}. Empurre com mais \omterm{def:g10:inertia:force}{força}: por volta de
\qty{400}{N} ele arranca, e depois desliza com menos esforço --- o
\omterm{def:g11:forces-and-motion:friction}{atrito cinético}, mais fraco, assumiu.
\end{example}

\begin{remark}[Por que o freio ABS funciona]\label{rem:g11:forces-and-motion:abs}
O ponto de contato de uma roda que gira não desliza sobre a pista: o pneu
agarra por \omterm{def:g10:inertia:inventory}{atrito} \emph{estático}. Uma roda travada derrapa e recebe
apenas o \omterm{def:g10:inertia:inventory}{atrito} \emph{cinético}, mais fraco, que aponta contra a derrapagem
faça a roda o que fizer --- e a direção morre. O ABS solta o freio no
instante em que uma roda trava: mais aderência, frenagem mais curta e um
motorista que consegue esterçar.
\end{remark}

\section{Virar: o movimento circular precisa de uma força para dentro}

\begin{definition}[Movimento circular uniforme]\label{def:g11:forces-and-motion:ucm}
Um corpo está em \emph{movimento circular uniforme}\index{movimento
circular uniforme} quando a sua \omterm{def:g10:relative-motion:trajectory}{trajetória} é um círculo percorrido com
velocidade escalar constante. A velocidade, tangente ao círculo, muda de
direção a cada instante --- de modo que ela nunca é constante.
\end{definition}

\begin{proposition}[A força para dentro]\label{prop:g11:forces-and-motion:inward}
Um corpo em \omterm{def:g11:forces-and-motion:ucm}{movimento circular uniforme} está sujeito a uma \omterm{def:g11:forces-and-motion:netforce}{força
resultante} que nunca é nula: a cada instante ela aponta do corpo para o
centro do círculo.
\end{proposition}

\begin{proof}
A velocidade escalar é constante, de modo que a \omterm{def:g11:forces-and-motion:netforce}{força resultante} não tem
componente ao longo de $\vect v$ (casos 1 e 2 da
\cref{prop:g11:forces-and-motion:cases}): ela é perpendicular à tangente
e, portanto, está ao longo do raio --- e apontando para dentro, já que
cada $\Delta \vect v$ curva a \omterm{def:g10:relative-motion:trajectory}{trajetória} em direção ao centro.
\end{proof}

\begin{example}[O barbante, a curva e a Lua]\label{ex:g11:forces-and-motion:moon}
Toda curva regular esconde uma \omterm{def:g10:inertia:force}{força} para dentro. A bola girada: a tensão
do barbante. O carro numa curva: o \omterm{def:g11:forces-and-motion:friction}{atrito estático} lateral da pista sobre
os pneus --- que some no gelo, e então o carro segue reto. A Lua: o puxão
gravitacional da Terra (\cref{ch:g10:universal-gravitation}), que a faz
dar a volta em \num{27.3} dias --- ela não é sustentada \emph{para
cima}, mas puxada de lado numa curva perpétua, caindo em volta da Terra;
no ano que vem isso vira a mecânica dos \omterm{def:g10:universal-gravitation:satellite}{satélites}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:1}
\omterm{def:g11:forces-and-motion:netforce}{Força resultante} nula ou não nula? Justifique pela velocidade: (a) um
carro seguindo em linha reta a constantes \qty{90}{km/h}; (b) freando em
linha reta; (c) fazendo uma curva a constantes \qty{50}{km/h}; (d) um
paraquedista na velocidade limite.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:2}
Bolas de \qty{3.0}{kg} e \qty{1.0}{kg} estão nas pontas de uma barra leve
de \qty{0.80}{m}. Localize o \omterm{def:g11:forces-and-motion:com}{centro de massa}; e onde ele ficaria com
massas iguais?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:3}
O mesmo empurrão breve é dado a um carrinho de bagagem vazio de
\qty{15}{kg} e ao mesmo carrinho carregado até \qty{45}{kg}. Compare as
variações de velocidade; que empurrão daria ao carrinho carregado o
$\Delta v$ do vazio?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:4}
Um bonde que se move a \qty{10}{m/s} freia; um segundo depois ele se move
a \qty{8}{m/s}, no mesmo sentido. Dê $\Delta \vect v$ e o sentido da
\omterm{def:g11:forces-and-motion:netforce}{força resultante}. Que caso da \cref{prop:g11:forces-and-motion:cases} é
esse?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:5}
\omterm{def:g11:forces-and-motion:friction}{Atrito estático} ou cinético --- e dirigido para onde? (a) Um engradado
que resiste ao seu empurrão (suave demais); (b) o mesmo engradado depois
que passa a deslizar; (c) um livro numa bandeja ligeiramente inclinada,
sem deslizar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:6}
Uma \omterm{def:g11:forces-and-motion:netforce}{força resultante} $F$ aplicada durante $\Delta t$ a uma massa $m$
produz $\Delta v = \qty{4.0}{m/s}$. Preveja $\Delta v$ para: (a)
$2F, \Delta t, m$; (b) $F, \Delta t/2, m$; (c) $F, \Delta t, 2m$;
(d) $3F, 2\Delta t, 6m$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:7}
Um disco desliza para leste a \qty{8.0}{m/s}; um golpe acrescenta
$\Delta \vect v = \qty{6.0}{m/s}$ rumo ao norte. Calcule a nova
velocidade escalar e a nova direção. Por que a velocidade escalar não é
$8.0 + 6.0 = \qty{14}{m/s}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:8}
Uma goleira defende o mesmo chute de dois modos: com os braços rígidos (a
bola para em \qty{0.02}{s}) ou ``acompanhando'' a bola (\qty{0.08}{s}).
Compare as \omterm{def:g10:inertia:force}{forças} médias e diga que grandezas já estavam fixadas antes da
escolha dela.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:9}
Que \omterm{def:g10:inertia:force}{força} externa empurra um velocista para a frente na largada? Por que
as sapatilhas com pregos ajudam --- e por que ninguém consegue acelerar,
nem sequer caminhar, sobre gelo perfeitamente sem \omterm{def:g10:inertia:inventory}{atrito}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:10}
Numa frenagem de emergência, por que um carro com ABS (rodas mantidas
girando) em geral para em menos espaço do que um com as rodas travadas e
derrapando --- e por que o carro em derrapagem não pode mais ser
esterçado? Cite a \cref{def:g11:forces-and-motion:friction}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:11}
Um lançador de martelo gira a bola presa ao cabo. Que \omterm{def:g10:inertia:force}{força} faz a bola
virar? Esboce a \omterm{def:g10:relative-motion:trajectory}{trajetória} dela depois da soltura. Por que a Lua não
precisa de cabo, e em que a \omterm{def:g10:relative-motion:trajectory}{trajetória} dela se transformaria sem
gravidade?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:12}
Uma vassoura é um cabo uniforme de \qty{1.0}{kg} (com \omterm{def:g11:forces-and-motion:com}{centro de massa}
próprio a \qty{0.60}{m} do topo) com uma cabeça de \qty{2.0}{kg} cujo
\omterm{def:g11:forces-and-motion:com}{centro de massa} está a \qty{1.20}{m} do topo. Tratando cada parte como
massa pontual, localize o \omterm{def:g11:forces-and-motion:com}{centro de massa} da vassoura e confira a regra
da razão inversa da \cref{def:g11:forces-and-motion:com}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:13}
Um carro de massa $m$ e uma van carregada de massa $2m$ freiam a partir
da mesma velocidade com a mesma \omterm{def:g10:inertia:force}{força} de frenagem.
\begin{enumerate}
  \item Compare os tempos necessários para parar.
  \item Cada um perde velocidade de modo regular, e portanto percorre o
        trajeto com a mesma velocidade \emph{média}: compare as distâncias
        de frenagem.
  \item Agora o carro freia a partir do \emph{dobro} da velocidade, com a
        mesma \omterm{def:g10:inertia:force}{força}: compare tempo e distância com a sua primeira
        frenagem. Qual é a moral para os limites de velocidade?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:14}
A Lua orbita a $R \approx \qty{3.84e8}{m}$ em $T = \num{27.3}$ dias.
\begin{enumerate}
  \item Calcule a velocidade orbital dela (comprimento da circunferência
        dividido pelo \omterm{def:g10:signals-and-waves:period}{período}).
  \item Para onde aponta a variação da velocidade dela, e que \omterm{def:g10:inertia:force}{força} a
        fornece?
  \item Melhore, usando a \cref{prop:g11:forces-and-motion:inward}, a
        frase: ``a Lua não cai porque se move depressa o bastante''.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:15}
Um ovo de massa \qty{50}{g} atinge o chão a \qty{4.0}{m/s}: no piso de
cerâmica ele para em cerca de \qty{1}{ms}, sobre uma espuma grossa em
cerca de \qty{50}{ms}. Compare as \omterm{def:g10:inertia:force}{forças} médias. Explique, com o mesmo
raciocínio, por que você dobra os joelhos ao aterrissar de um salto e de
que é feito o colchão de queda de um ginasta.
\end{exercise}

\section{Problema: o teste de colisão}

\begin{problem}[{Problema de fim de semana --- projetar a colisão suave:
como uma batida fixa a variação de velocidade, todo dispositivo de
segurança de um carro é uma máquina de esticar o tempo}]
\label{pb:g11:forces-and-motion:1}
Um laboratório de testes de colisão lança um carro a \qty{50}{km/h}
contra uma parede rígida, com um boneco adulto de \qty{70}{kg} e um
boneco infantil de \qty{20}{kg} a bordo; câmeras cronometram cada
parada. A sua tarefa: descobrir onde mora a violência de uma batida e
projetá-la para longe.

\medskip
\noindent\textbf{Parte I --- A medida de uma batida.}
\begin{enumerate}
  \item Converta \qty{50}{km/h} para \unit{m/s}.
  \item Em toda parada frontal, a variação de velocidade dos ocupantes
        tem o mesmo módulo. Qual? Por que dispositivo algum consegue
        alterá-la?
  \item Cite o \cref{thm:g11:forces-and-motion:secondlaw}: com $\Delta v$
        e $m$ fixos, $F\,\Delta t$ está fixado. Qual é o único botão do
        projetista?
  \item A parede para o carro em \qty{0.050}{s}; uma zona de deformação
        programada estica isso para \qty{0.150}{s}. Compare as \omterm{def:g10:inertia:force}{forças}
        médias.
  \item Em que sentido apontam $\Delta \vect v$ e a \omterm{def:g11:forces-and-motion:netforce}{força resultante}
        sobre os bonecos durante a parada?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A zona de deformação.}
\begin{enumerate}[resume]
  \item Por que, em uma frase, os engenheiros projetam a frente do carro
        para ser destruída?
  \item Num carro antigo e rígido (parada em \qty{0.050}{s}) o boneco é
        amarrado rigidamente ao banco. Compare a \omterm{def:g10:inertia:force}{força} sobre ele com a da
        parada moderna de \qty{0.150}{s}.
  \item A velocidade do carro que para cai de modo regular (média: metade
        da inicial). Que distância ele percorre em \qty{0.150}{s}? Compare
        com uma zona de deformação real.
  \item Por que não uma zona de deformação de \qty{10}{m}? E que parte do
        carro \emph{não} pode se deformar?
  \item Resuma a Parte II num lema: ``você não pode escolher $\Delta v$,
        então tem de escolher \dots''.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O cinto, o airbag e a criança.}
\begin{enumerate}[resume]
  \item Um boneco sem cinto conserva os seus \qty{14}{m/s}
        (\cref{ch:g10:inertia}) até que o para-brisa o pare em
        \qty{0.005}{s}; um boneco com cinto para junto com o carro, em
        \qty{0.150}{s}. Compare as \omterm{def:g10:inertia:force}{forças} médias.
  \item Por que um cinto que estica um pouco é melhor do que um arreio de
        aço inextensível?
  \item A cabeça pararia no volante em \qty{0.010}{s}; o airbag estica
        isso para \qty{0.080}{s}. Compare as \omterm{def:g10:inertia:force}{forças}.
  \item O adulto e a criança sofrem o mesmo $\Delta v$ no mesmo
        $\Delta t$. Compare as \omterm{def:g11:forces-and-motion:netforce}{forças resultantes} que os dois cintos
        precisam fornecer.
  \item Uma criança no colo: os braços têm de mudar a velocidade de um
        bebê de \qty{10}{kg} em \qty{14}{m/s} durante \qty{0.150}{s},
        enquanto a gravidade muda uma velocidade em \qty{9.8}{m/s} a cada
        segundo. Quantas vezes mais depressa os braços precisam agir ---
        e segurar que massa seria igualmente pesado? Conclua.
  \item As crianças sentem \omterm{def:g10:inertia:force}{forças} menores --- então por que cadeirinhas?
        Pense em \emph{onde} o cinto aplica a sua \omterm{def:g10:inertia:force}{força} e em $\Delta t$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A célula de sobrevivência e o veredito.}
\begin{enumerate}[resume]
  \item A célula dos passageiros é rígida, enquanto as duas
        extremidades se deformam. Por que a célula não pode se deformar,
        embora rigidez signifique paradas violentas?
  \item Dois carros idênticos colidem de frente, cada um a
        \qty{50}{km/h}. Onde cada um para? Compare cada batida com o
        teste da parede.
  \item Um carro pequeno encontra de frente um caminhão carregado. As
        \omterm{def:g10:inertia:force}{forças} que eles exercem um sobre o outro são iguais e opostas,
        mútuas como toda interação
        (\cref{ch:g11:fundamental-interactions}). Qual dos dois muda mais
        de velocidade?
  \item Liste os dispositivos de segurança deste problema e a única
        grandeza que todos eles esticam.
  \item Final: enuncie a lei de projeto da colisão suave em uma frase,
        usando apenas \emph{massa}, \emph{variação de velocidade},
        \emph{\omterm{def:g10:inertia:force}{força}} e \emph{tempo}.
\end{enumerate}
\end{problem}
