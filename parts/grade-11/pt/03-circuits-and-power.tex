\chapter{Circuitos elétricos e potência}\label{ch:g11:circuits-and-power}

Acione um interruptor: uma lâmpada acende, porque um gerador a muitos
quilômetros dali empurra carga por uma volta de cobre que passa pela sua
parede. Os anos anteriores mapearam o circuito --- corrente, \omterm{def:g11:circuits-and-power:voltage}{tensão}, \omterm{prop:g11:circuits-and-power:ohm}{lei
de Ohm}. Este capítulo o reabre com o livro-caixa do
\cref{ch:g10:energy-conservation}: o que o medidor cobra, por que os fios
finos esquentam, por que a rede trabalha a \qty{400}{kV}.

\section{Carga em movimento}

\begin{definition}[Corrente e o coulomb]\label{def:g11:circuits-and-power:current}
Uma \emph{corrente elétrica}\index{corrente elétrica} é um escoamento
ordenado de carga (elétrons, num metal). A sua intensidade $I$, em
\emph{ampères}\index{ampère} (\unit{A}), é a carga que atravessa uma seção
do fio por segundo: uma corrente constante $I$ durante um tempo $t$
transporta a carga
\[ q = I\,t , \]
em \emph{coulombs}\index{coulomb} (\unit{C}), com
$\qty{1}{C} = \qty{1}{A.s}$. A carga elementar vale
$e = \qty{1.6e-19}{C}$. A corrente é medida por um amperímetro, ligado em
série.
\end{definition}

\begin{example}[Raio contra celular]\label{ex:g11:circuits-and-power:coulomb}
Uma descarga de raio carrega uns $\qty{3.0e4}{A}$ por cerca de
$\qty{1.0e-4}{s}$: $q = 3.0\times10^{4} \times 10^{-4} = \qty{3.0}{C}$. O
seu celular, a \qty{2.0}{A}, move a mesma carga a cada $1.5$ segundo.
\end{example}

\begin{definition}[Tensão e o volt]\label{def:g11:circuits-and-power:voltage}
A \emph{tensão}\index{tensão} $U$ entre dois pontos de um circuito, em
\emph{volts}\index{volt} (\unit{V}), é a \omterm{def:g10:energy-conservation:energy}{energia} trocada por \omterm{def:g10:orders-of-magnitude:unit}{unidade} de
carga que viaja entre eles: $\qty{1}{V} = \qty{1}{J/C}$, de modo que uma
carga $q$ que atravessa um dispositivo sob tensão $U$ troca com ele a
\omterm{def:g10:energy-conservation:energy}{energia} $E = qU$. Um voltímetro, ligado em paralelo, a mede.
\end{definition}

\begin{proposition}[Leis dos circuitos]\label{prop:g11:circuits-and-power:laws}
Num circuito em regime permanente:
\begin{enumerate}
  \item \emph{lei dos nós:} as correntes que chegam a um nó somam o mesmo
        que as correntes que saem dele;
  \item \emph{lei das tensões em série:} as tensões ao longo de um
        caminho se somam: $U_{AC} = U_{AB} + U_{BC}$;
  \item \emph{lei das tensões em paralelo:} dois ramos que ligam os
        mesmos dois pontos suportam a mesma \omterm{def:g11:circuits-and-power:voltage}{tensão}.
\end{enumerate}
\end{proposition}

\begin{proof}
Em regime permanente a carga não se acumula em lugar algum, de modo que o
que entra num nó sai dele. A \omterm{def:g10:energy-conservation:energy}{energia} por \omterm{def:g11:circuits-and-power:current}{coulomb} é aditiva ao longo de um
caminho e depende só dos seus extremos --- e não do ramo escolhido entre
eles.
\end{proof}

\section{Lei de Ohm e associações de resistores}

\begin{definition}[Resistência]\label{def:g11:circuits-and-power:resistance}
A \emph{resistência}\index{resistência} de um condutor é a razão
$R = U/I$ entre a \omterm{def:g11:circuits-and-power:voltage}{tensão} nos seus terminais e a corrente que o atravessa,
medida em \emph{ohms}\index{ohm} (\unit{\ohm}):
$\qty{1}{\ohm} = \qty{1}{V/A}$.
\end{definition}

\begin{proposition}[Lei de Ohm]\label{prop:g11:circuits-and-power:ohm}
\index{lei de Ohm}
Para um condutor metálico mantido a temperatura fixa, $R$ é constante:
\omterm{def:g11:circuits-and-power:voltage}{tensão} e corrente são proporcionais,
\[ U = R\,I . \]
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:circuits-and-power:ohm-limits}
Uma lei experimental, e não universal: um diodo, o filamento de uma
lâmpada aquecida ou a sua própria pele não a obedecem. Por que os metais
a obedecem é deduzido nos volumes de graduação.
\end{remark}

\begin{proposition}[Resistores em série e em paralelo]\label{prop:g11:circuits-and-power:networks}
\index{resistência equivalente}
Dois resistores em série equivalem a um único resistor
$R_{\text{s}} = R_1 + R_2$; dois em paralelo, a um único resistor
$R_{\text{p}}$ dado por
\[
\frac{1}{R_{\text{p}}} = \frac{1}{R_1} + \frac{1}{R_2} ,
\qquad\text{ou seja,}\quad
R_{\text{p}} = \frac{R_1 R_2}{R_1 + R_2} < \min(R_1, R_2) .
\]
\end{proposition}

\begin{proof}
Em série: a mesma corrente $I$ atravessa os dois e as tensões se somam
(\cref{prop:g11:circuits-and-power:laws}), logo $U = (R_1 + R_2)\,I$. Em
paralelo: os dois ficam sob a mesma \omterm{def:g11:circuits-and-power:voltage}{tensão} $U$ e as correntes se somam,
logo $I = \frac{U}{R_1} + \frac{U}{R_2}$.
\end{proof}

\begin{example}[Resistência equivalente]\label{ex:g11:circuits-and-power:networks}
$R_1 = \qty{100}{\ohm}$ e $R_2 = \qty{150}{\ohm}$: em série,
\qty{250}{\ohm}; em paralelo, $\frac{100 \times 150}{250} =
\qty{60}{\ohm}$ --- menos do que qualquer um dos dois: a corrente ganhou
uma segunda estrada.
\end{example}

\section{Energia e potência num circuito}

\begin{proposition}[Potência elétrica]\label{prop:g11:circuits-and-power:power}
\index{potência elétrica}
Um dispositivo sob \omterm{def:g11:circuits-and-power:voltage}{tensão} $U$ atravessado por uma corrente $I$ recebe
\omterm{def:g10:energy-conservation:energy}{energia} à taxa
\[ P = U\,I , \qquad\text{ou seja,}\quad E = U\,I\,t \ \text{num tempo } t . \]
\end{proposition}

\begin{proof}
Num tempo $t$ a carga $q = It$ atravessa o dispositivo
(\cref{def:g11:circuits-and-power:current}) e cada \omterm{def:g11:circuits-and-power:current}{coulomb} lhe entrega
$U$ \omterm{def:g10:energy-conservation:energy}{joules} (\cref{def:g11:circuits-and-power:voltage}): $E = qU = UIt$.
\end{proof}

\begin{proposition}[Potência num resistor]\label{prop:g11:circuits-and-power:ri2}
Um resistor $R$ percorrido por uma corrente $I$ dissipa
\[ P = R\,I^2 = \frac{U^2}{R} . \]
\end{proposition}

\begin{proof}
Substitua a \omterm{prop:g11:circuits-and-power:ohm}{lei de Ohm} em $P = UI$:
$P = (RI)I = RI^2 = U(U/R) = U^2/R$.
\end{proof}

\begin{example}[Ler uma placa de identificação]\label{ex:g11:circuits-and-power:nameplate}
Uma chaleira elétrica com a marcação ``\qty{230}{V} -- \qty{2200}{W}''
puxa $I = P/U = 2200/230 = \qty{9.6}{A}$ e tem \omterm{def:g11:circuits-and-power:resistance}{resistência}
$R = U^2/P = 230^2/2200 = \qty{24}{\ohm}$. Ligada por \qty{150}{s} ela
usa $E = 2200 \times 150 = \qty{3.3e5}{J} \approx \qty{0.09}{kW.h}$ ---
cerca de dois centavos.
\end{example}

\begin{remark}[Joules e quilowatt-horas]\label{rem:g11:circuits-and-power:kwh}
O medidor conta \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}
(\cref{def:g10:energy-conservation:kwh}):
$\qty{1}{kW.h} = \qty{3.6e6}{J}$. Os \omterm{def:g10:energy-conservation:energy}{joules} são a \omterm{def:g10:orders-of-magnitude:unit}{unidade} do físico; os
\omterm{def:g10:energy-conservation:kwh}{quilowatt-horas} dão preço às contas.
\end{remark}

\section{O efeito Joule: da torradeira à torre de transmissão}

\begin{definition}[Efeito Joule]\label{def:g11:circuits-and-power:joule}
A dissipação $P = RI^2$ que transforma \omterm{def:g10:energy-conservation:forms}{energia elétrica} em calor em todo
resistor é o \emph{efeito Joule}\index{efeito Joule}. Aquecedores,
chaleiras, torradeiras e as lâmpadas de antigamente são resistores por
projeto; um \emph{fusível}\index{fusível} é um fio fino calibrado para
derreter --- abrindo o circuito --- quando a corrente ultrapassa o valor
nominal dele.
\end{definition}

\begin{example}[Por que as torres trabalham em alta tensão]\label{ex:g11:circuits-and-power:transmission}
Um povoado precisa de $P = \qty{100}{kW}$ através de uma linha de
\omterm{def:g11:circuits-and-power:resistance}{resistência} $R_\ell = \qty{5.0}{\ohm}$. Entregue a $U = \qty{500}{V}$:
$I = P/U = \qty{200}{A}$ e a linha desperdiça
$R_\ell I^2 = \qty{200}{kW}$ --- o dobro da entrega: absurdo. A
$U = \qty{20}{kV}$: $I = \qty{5.0}{A}$ e $R_\ell I^2 = \qty{125}{W}$.
Quarenta vezes menos corrente, $1600$ vezes menos perda.
\end{example}

\begin{remark}[A regra de ferro da rede]\label{rem:g11:circuits-and-power:grid}
Com a \omterm{def:g10:energy-conservation:power}{potência} entregue $P = UI$ fixa, a perda na linha
$R_\ell I^2 = R_\ell P^2/U^2$ cai com o \emph{quadrado} da \omterm{def:g11:circuits-and-power:voltage}{tensão} de
transporte. Daí: elevar a \omterm{def:g11:circuits-and-power:voltage}{tensão} (\qty{400}{kV}) para viajar e abaixá-la
(\qty{230}{V}) para conviver com ela.
\end{remark}

\begin{remark}[Segurança doméstica]\label{rem:g11:circuits-and-power:safety}
Uma linha de \qty{230}{V} protegida por um \omterm{def:g11:circuits-and-power:joule}{fusível} de \qty{16}{A}
fornece no máximo $230 \times 16 \approx \qty{3.7}{kW}$: duas chaleiras
ao mesmo tempo derretem o \omterm{def:g11:circuits-and-power:joule}{fusível} --- que é, por projeto, o elo mais fino
e mais fraco, e muito mais barato do que um incêndio dentro da parede.
\end{remark}

\section{Baterias reais: força eletromotriz}

\begin{definition}[Fem e resistência interna]\label{def:g11:circuits-and-power:emf}
Uma bateria converte \omterm{def:g10:energy-conservation:forms}{energia química} em \omterm{def:g10:energy-conservation:forms}{energia elétrica}. A sua
\emph{força eletromotriz}\index{força eletromotriz} (fem)
$\mathcal{E}$, em \omterm{def:g11:circuits-and-power:voltage}{volts}, é a \omterm{def:g10:energy-conservation:energy}{energia} que ela entrega a cada \omterm{def:g11:circuits-and-power:current}{coulomb} que a
atravessa; a sua \emph{resistência interna}\index{resistência interna}
$r$ dá conta do \omterm{def:g11:circuits-and-power:joule}{efeito Joule} dentro da sua própria química. Modelo: uma
fonte ideal $\mathcal{E}$ em série com $r$ (o $\mathcal{E}$ cursivo: $E$
é \omterm{def:g10:energy-conservation:energy}{energia}).
\end{definition}

\begin{proposition}[Tensão nos terminais]\label{prop:g11:circuits-and-power:terminal}
Uma bateria que fornece uma corrente $I$ apresenta nos seus terminais
\[ U = \mathcal{E} - r\,I . \]
\end{proposition}

\begin{proof}
A química fornece $\mathcal{E}I$, a \omterm{def:g11:circuits-and-power:emf}{resistência interna} come $rI^2$ e o
circuito recebe $UI$: a \omterm{thm:g10:energy-conservation:conservation}{conservação da energia}
(\cref{ch:g10:energy-conservation}) dá $UI = \mathcal{E}I - rI^2$.
\end{proof}

\begin{example}[Uma bateria sob carga]\label{ex:g11:circuits-and-power:discharge}
Uma bateria ($\mathcal{E} = \qty{9.0}{V}$, $r = \qty{1.0}{\ohm}$)
alimenta uma lâmpada de \omterm{def:g11:circuits-and-power:resistance}{resistência} $R = \qty{8.0}{\ohm}$. A malha
obedece a $\mathcal{E} = rI + RI$, logo
$I = \mathcal{E}/(R + r) = \qty{1.0}{A}$ e
$U = 9.0 - 1.0 \times 1.0 = \qty{8.0}{V}$: a lâmpada recebe \qty{8.0}{W}
enquanto a bateria desperdiça \qty{1.0}{W} se aquecendo.
\end{example}

\begin{remark}[Receptores]\label{rem:g11:circuits-and-power:receivers}
Um motor ou uma bateria em carga fazem a conversão ao contrário: eles
absorvem sob $U = \mathcal{E}' + r'I$, convertendo $\mathcal{E}'I$ em
movimento ou em química e perdendo $r'I^2$. As fontes empurram; os
receptores empurram de volta.
\end{remark}

\begin{example}[Rendimento de uma bateria]\label{ex:g11:circuits-and-power:efficiency}
O \omterm{def:g10:energy-conservation:efficiency}{rendimento} (\cref{def:g10:energy-conservation:efficiency}) da descarga
acima: útil $UI = \qty{8.0}{W}$ de $\mathcal{E}I = \qty{9.0}{W}$, logo
$\eta = U/\mathcal{E} = 8/9 \approx 0.89$ --- e ele cai à medida que a
corrente sobe: bateria que trabalha muito esquenta.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:1}
A bateria de um celular carrega com uma corrente constante de
\qty{2.0}{A} por \qty{90}{min}. Que carga passa pelo cabo, em \omterm{def:g11:circuits-and-power:current}{coulombs}?
Quantas cargas elementares isso representa ($e = \qty{1.6e-19}{C}$)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:2}
Um resistor de \qty{220}{\ohm} sob \qty{12}{V}: que corrente circula? Uma
\omterm{def:g11:circuits-and-power:resistance}{resistência} de aquecimento puxa \qty{0.50}{A} sob \qty{230}{V}: qual é a
sua \omterm{def:g11:circuits-and-power:resistance}{resistência}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:3}
Calcule a \omterm{prop:g11:circuits-and-power:networks}{resistência equivalente} de $R_1 = \qty{120}{\ohm}$ e
$R_2 = \qty{60}{\ohm}$ em série e depois em paralelo. Qual delas é menor
do que as duas, e por quê?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:4}
Um ferro de passar de viagem puxa \qty{8.7}{A} sob \qty{230}{V}. Calcule
a sua \omterm{def:g10:energy-conservation:power}{potência} e, em seguida, a \omterm{def:g10:energy-conservation:energy}{energia} que ele usa em \qty{4.0}{min}, em
\omterm{def:g10:energy-conservation:energy}{joules} e em \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:circuits-and-power:5}
O carregador de um notebook entrega \qty{60}{W} durante \qty{5.0}{h} por
dia: qual é a \omterm{def:g10:energy-conservation:energy}{energia} diária em \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas} e em \omterm{def:g10:energy-conservation:energy}{joules}, e o custo a
$0.25$ euros por \unit{kW.h}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:6}
Um aquecedor elétrico tem valores nominais de \qty{1200}{W} sob
\qty{230}{V}.
\begin{enumerate}
  \item Calcule a sua \omterm{def:g11:circuits-and-power:resistance}{resistência} e a corrente que ele puxa.
  \item Ligado (com a mesma \omterm{def:g11:circuits-and-power:resistance}{resistência}) a uma rede de \qty{115}{V}, que
        \omterm{def:g10:energy-conservation:power}{potência} ele entrega? Por que não a metade?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:7}
$R_1 = \qty{40}{\ohm}$ e $R_2 = \qty{20}{\ohm}$ em série sob \qty{12}{V}.
Calcule a corrente, a \omterm{def:g11:circuits-and-power:voltage}{tensão} em cada resistor e a \omterm{def:g10:energy-conservation:power}{potência} dissipada em
cada um; confira que as \omterm{def:g10:energy-conservation:power}{potências} somam $UI$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:8}
$R_1 = \qty{60}{\ohm}$ e $R_2 = \qty{30}{\ohm}$ em paralelo sob
\qty{12}{V}. Calcule a corrente de cada ramo, a corrente total e a
\omterm{prop:g11:circuits-and-power:networks}{resistência equivalente} de dois modos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:9}
Uma bateria fornece $U = \qty{5.5}{V}$ com $I = \qty{0.50}{A}$ e
$U = \qty{4.0}{V}$ com $I = \qty{2.0}{A}$. Encontre a sua fem
$\mathcal{E}$ e a sua \omterm{def:g11:circuits-and-power:emf}{resistência interna} $r$, e depois a corrente de
curto-circuito $\mathcal{E}/r$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:circuits-and-power:10}
Uma linha de cozinha de \qty{230}{V} é protegida por um \omterm{def:g11:circuits-and-power:joule}{fusível} de
\qty{16}{A}. Uma chaleira de \qty{2.2}{kW} e uma torradeira de
\qty{1.0}{kW} funcionam juntas: o \omterm{def:g11:circuits-and-power:joule}{fusível} aguenta? Alguém acrescenta um
aquecedor de \qty{1.5}{kW}: mostre que ele queima.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:circuits-and-power:13}

\begin{enumerate}

  \item Mostre que a \omterm{def:g10:energy-conservation:power}{potência} recebida por um $R$ variável,

        $(R+r)^2/R = (R-r)^2/R + 4r$), e calcule esse máximo.
\end{enumerate}
\end{exercise}

\section{Problema: da barragem à torradeira}

\begin{problem}[{Problema de fim de semana --- a viagem de mil
quilômetros de um quilowatt-hora, da água que cai ao pão que doura, com a
conta completa do trajeto}]
\label{pb:g11:circuits-and-power:1}
Uma barragem de montanha guarda água \qty{200}{m} acima das suas
turbinas, que engolem \qty{50}{m^3} --- \qty{5.0e4}{kg} --- por segundo;
turbina e alternador convertem com \omterm{def:g10:energy-conservation:efficiency}{rendimento} $0.90$. Através de uma
linha de \omterm{def:g11:circuits-and-power:resistance}{resistência} total $R_\ell = \qty{32}{\ohm}$ (\qty{1000}{km} de
ida e volta), a usina alimenta uma cidade onde a sua torradeira de
\qty{900}{W} espera sob \qty{230}{V}. O alternador entrega
\qty{20}{kV}; transformadores (\omterm{def:g10:energy-conservation:efficiency}{rendimento} $0.99$ cada) convertem tensões
à vontade. A eletricidade é vendida a $0.25$ euros por \unit{kW.h};
$g = \qty{9.81}{N/kg}$.

\medskip
\noindent\textbf{Parte I --- Na barragem.}
\begin{enumerate}
  \item Quanta \omterm{def:g10:energy-conservation:energy}{energia} potencial a água em queda libera a cada segundo?
        Deduza a \omterm{def:g10:energy-conservation:power}{potência} disponível para as turbinas.
  \item Que \omterm{prop:g11:circuits-and-power:power}{potência elétrica} sai do alternador?
  \item A \qty{20}{kV}, direto do alternador, que corrente circula?
  \item Calcule a perda por \omterm{def:g11:circuits-and-power:joule}{efeito Joule} $R_\ell I^2$ com essa corrente e
        compare-a com a saída da usina. Conclua.
  \item A \omterm{def:g10:energy-conservation:power}{potência} da cidade $P = UI$ é fixada pela demanda. Qual é o
        único modo de encolher $I$, e que dispositivo faz isso?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Na linha.}
\begin{enumerate}[resume]
  \item Um transformador, tratado como sem perdas nesta questão, eleva os
        \qty{88}{MW} a \qty{400}{kV}: que corrente circula agora?
  \item Calcule a nova perda na linha, em megawatts.
  \item Que fração da saída da usina se perde na linha?
  \item Mostre que a fração perdida vale $R_\ell P/U^2$ e verifique que
        isso explica o fator entre as questões 4 e 7.
  \item Calcule a queda de \omterm{def:g11:circuits-and-power:voltage}{tensão} $R_\ell I$ ao longo da linha a
        \qty{400}{kV}. Isso é coerente com a questão 8?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Na cozinha.}
\begin{enumerate}[resume]
  \item Converta \qty{1}{kW.h} em \omterm{def:g10:energy-conservation:energy}{joules}; o que o medidor multiplica?
  \item Calcule a corrente e a \omterm{def:g11:circuits-and-power:resistance}{resistência} da torradeira sob \qty{230}{V}.
  \item Três minutos de torrada: a \omterm{def:g10:energy-conservation:energy}{energia} em \omterm{def:g10:energy-conservation:energy}{joules} e em
        \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}, e o preço dela.
  \item A torradeira dissipa $RI^2$: explique em uma frase por que a mesma
        fórmula é a perda na linha e o produto na cozinha.
  \item A torradeira divide um \omterm{def:g11:circuits-and-power:joule}{fusível} de \qty{16}{A} com uma chaleira de
        \qty{2.2}{kW}: o \omterm{def:g11:circuits-and-power:joule}{fusível} aguenta? Um aquecedor de \qty{2.0}{kW}
        se junta: e agora?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A auditoria.}
\begin{enumerate}[resume]
  \item A fiação da própria casa, de \omterm{def:g11:circuits-and-power:resistance}{resistência} \qty{0.05}{\ohm},
        conduz os \qty{13.5}{A} da questão 15: calcule a perda e a fração
        dela nos cerca de \qty{3.1}{kW} entregues.
  \item Encadeie os \omterm{def:g10:energy-conservation:efficiency}{rendimentos} (turbina--alternador, elevação, linha,
        abaixamento, fiação da casa) num único \omterm{def:g10:energy-conservation:efficiency}{rendimento} global da rede.
  \item Quantos \omterm{def:g10:orders-of-magnitude:unit}{quilogramas} (e quantos \omterm{def:g10:orders-of-magnitude:unit}{metros} cúbicos) de água da
        barragem precisam cair para que \emph{um} \omterm{def:g10:energy-conservation:kwh}{quilowatt-hora} chegue à
        sua torradeira?
  \item Avalie a fração perdida $R_\ell P/U^2$ da questão 9 em
        $U = \qty{20}{kV}$. O que significa um valor acima de $1$?
  \item O desfecho --- \emph{o preço da distância}: em duas frases, quanto
        um \omterm{def:g10:energy-conservation:kwh}{quilowatt-hora} de torrada custa à montanha, que fração morre
        no caminho e qual é o único truque que torna tudo possível.
\end{enumerate}
\end{problem}
