\chapter{As interações fundamentais}\label{ch:g11:fundamental-interactions}

Esfregue um balão no cabelo e ele recolhe pedacinhos de papel ---
vencendo, com poucos centímetros quadrados de borracha, a atração
gravitacional do planeta inteiro. Essa vitória fácil é a razão de os
\omterm{def:g11:fundamental-interactions:ladder}{átomos} se manterem inteiros e de os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} pesados acabarem se
despedaçando (\cref{ch:g11:nucleus-radioactivity}). O universo funciona
com exatamente quatro interações; este capítulo as apresenta, pesa umas
contra as outras e atribui a cada uma o seu andar.

\section{A matéria, do quark às galáxias}

\begin{definition}[A escada das estruturas]\label{def:g11:fundamental-interactions:ladder}
A matéria é construída em andares, cada um montado a partir do de baixo:
três \emph{quarks}\index{quark} (pontuais, menores que \qty{e-18}{m}) se
ligam num \emph{núcleon}\index{núcleon} --- próton ou nêutron, com cerca
de \qty{e-15}{m} de diâmetro; os núcleons se empacotam num
\emph{núcleo}\index{núcleo} (de \qty{e-15}{m} a \qty{e-14}{m}); um núcleo
mais a sua nuvem de elétrons é um \emph{átomo}\index{átomo}
(\qty{e-10}{m}); os átomos se ligam em \emph{moléculas}\index{molécula} e
cristais (de \qty{e-9}{m} para cima), que se juntam em objetos do dia a
dia (\qty{1}{m}), planetas (\qty{e7}{m}), sistemas planetários
(\qty{e11}{m}), galáxias (\qty{e21}{m}) e no universo observável
(\qty{e26}{m}).
\end{definition}

\begin{definition}[As quatro interações]\label{def:g11:fundamental-interactions:four}
Uma \emph{interação fundamental}\index{interação fundamental} é uma \omterm{def:g10:inertia:force}{força}
não redutível a outras \omterm{def:g10:inertia:force}{forças}; existem exatamente quatro: a
\emph{gravitação}\index{gravitação}, a \emph{interação
eletromagnética}\index{interação eletromagnética}, a \omterm{def:g11:fundamental-interactions:strong}{interação forte} e a
\omterm{def:g11:fundamental-interactions:weak}{interação fraca}.
\end{definition}

\begin{remark}\label{rem:g11:fundamental-interactions:empty}
Os andares são separados por vazios: um \omterm{def:g11:fundamental-interactions:ladder}{átomo} é $10^{5}$ vezes mais largo
do que o seu \omterm{def:g11:fundamental-interactions:ladder}{núcleo} --- amplie o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} até uma bolinha de \qty{1}{cm} e
a nuvem de elétrons terá um quilômetro de diâmetro. Alguma coisa precisa
manter esses andares rarefeitos unidos, e quase nunca é a gravidade.
\end{remark}

\section{Carga elétrica}

\begin{definition}[Carga elétrica]\label{def:g11:fundamental-interactions:charge}
A \emph{carga elétrica}\index{carga elétrica} é a propriedade da matéria
sobre a qual a \omterm{def:g11:fundamental-interactions:four}{interação eletromagnética} age, assim como a massa é a
propriedade sobre a qual a \omterm{def:g11:fundamental-interactions:four}{gravitação} age. Ela é medida em
\emph{coulombs}\index{coulomb} (\unit{C}) e vem em \emph{dois sinais},
positivo e negativo, que se cancelam quando somados; cargas de mesmo
sinal se repelem, e cargas de sinais opostos se atraem.
\end{definition}

\begin{definition}[Carga elementar]\label{def:g11:fundamental-interactions:elementary}
A carga é granulada: toda carga observada é um múltiplo inteiro da
\emph{carga elementar}\index{carga elementar} $e = \qty{1.60e-19}{C}$. O
próton carrega $+e$, o elétron $-e$ e o nêutron $0$; uma carga $q$ contém
$N = \abs{q}/e$ cargas elementares.
\end{definition}

\begin{proposition}[Conservação da carga]\label{prop:g11:fundamental-interactions:conservation}
A carga total de um sistema isolado nunca muda: a carga não é criada nem
destruída, apenas transferida --- na prática pelos elétrons, os
portadores mais leves.
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:fundamental-interactions:conservation}
Nunca se observou violação alguma; a razão profunda, uma simetria do
eletromagnetismo, é dada nos volumes de graduação.
\end{remark}

\begin{definition}[Eletrização por atrito e por contato]\label{def:g11:fundamental-interactions:charging}
Esfregar dois isolantes arranca elétrons de um e os deposita no outro,
deixando cargas opostas de mesmo módulo (\emph{eletrização por
atrito}\index{eletrização por atrito}); encostar um objeto carregado num
neutro reparte a carga (\emph{eletrização por contato}\index{eletrização
por contato}). Só os elétrons se movem; a carga total se conserva.
\end{definition}

\begin{example}[Contar elétrons]\label{ex:g11:fundamental-interactions:balloon}
Um balão esfregado no cabelo adquire $q = \qty{-1.6e-8}{C}$: ele
\emph{ganhou} $N = \num{1.6e-8} / \num{1.6e-19} = \num{1.0e11}$ elétrons,
e o cabelo ficou com exatamente $+\qty{1.6e-8}{C}$.
\end{example}

\section{A lei de Coulomb}

\begin{theorem}[Lei de Coulomb]\label{thm:g11:fundamental-interactions:coulomb}
Duas cargas pontuais $q_1$ e $q_2$ à distância $d$ exercem uma sobre a
outra \omterm{def:g10:inertia:force}{forças} ao longo da reta que as une, de módulos iguais e sentidos
opostos --- repulsivas para sinais iguais, atrativas para sinais opostos
--- com
\[
F = k\,\frac{\abs{q_1 q_2}}{d^{2}},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{theorem}
\admitted

\begin{remark}\label{rem:g11:fundamental-interactions:torsion}
Coulomb estabeleceu a lei em 1785 medindo a torção de um fio fino de
torção; por que o \omterm{def:g10:orders-of-magnitude:scinot}{expoente} é exatamente $2$ é respondido nos volumes de
graduação.
\end{remark}

\begin{remark}[Uma gêmea formal da gravitação]\label{rem:g11:fundamental-interactions:twin}
A lei de Coulomb é, símbolo por símbolo, a lei da \omterm{def:g11:fundamental-interactions:four}{gravitação} universal
(\cref{ch:g10:universal-gravitation}) com cargas no lugar das massas:
\begin{center}
\begin{tabular}{lll}
fonte & massa $m$ & carga $q$ \\
lei & $F = G\,\dfrac{m_1 m_2}{d^2}$ & $F = k\,\dfrac{\abs{q_1 q_2}}{d^2}$ \\[1.5ex]
constante & $G = \qty{6.67e-11}{N.m^2/kg^2}$ & $k = \qty{8.99e9}{N.m^2/C^2}$ \\
sinal & sempre atrativa & atrativa ou repulsiva \\
\end{tabular}
\end{center}
As duas diferenças comandam tudo o que vem abaixo: a eletricidade é
esmagadoramente mais forte, e só ela consegue se cancelar.
\end{remark}

\begin{example}[Dois prótons, as duas forças ao mesmo tempo]\label{ex:g11:fundamental-interactions:protons}
Dois prótons ($m_p = \qty{1.67e-27}{kg}$) estão a
$d = \qty{1.0e-15}{m}$ --- vizinhos nucleares. Repulsão elétrica:
\[
F_E = \num{8.99e9} \times
\frac{(\num{1.60e-19})^{2}}{(\num{1.0e-15})^{2}}
\approx \qty{2.3e2}{N}
\]
--- o \omterm{def:g10:universal-gravitation:weight}{peso} de uma mala de $23$ quilos, carregado por uma partícula de
$10^{-27}$ \omterm{def:g10:orders-of-magnitude:unit}{quilograma}. Atração gravitacional:
$F_G = \num{6.67e-11} \times (\num{1.67e-27})^{2} /
(\num{1.0e-15})^{2} \approx \qty{1.9e-34}{N}$. A razão
$F_E / F_G = k e^{2} / (G m_p^{2}) \approx \num{1.2e36}$ não depende da
distância ($d^{2}$ se cancela): entre partículas elementares, a
gravidade é $10^{36}$ vezes fraca demais para contar --- a
\emph{qualquer} distância.
\end{example}

\section{As interações forte e fraca}

O exemplo deixa um escândalo: um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hélio guarda dois prótons que
se repelem com dezenas de \omterm{def:g10:inertia:force}{newtons} --- e, ainda assim, o hélio existe.

\begin{definition}[A interação forte]\label{def:g11:fundamental-interactions:strong}
A \emph{interação forte}\index{interação forte} liga os \omterm{def:g11:fundamental-interactions:ladder}{quarks} em
\omterm{def:g11:fundamental-interactions:ladder}{núcleons} e os \omterm{def:g11:fundamental-interactions:ladder}{núcleons} em \omterm{def:g11:fundamental-interactions:ladder}{núcleos}. A \qty{e-15}{m} ela é atrativa e muito
mais forte do que a repulsão elétrica (milhares de \omterm{def:g10:inertia:force}{newtons} entre
\omterm{def:g11:fundamental-interactions:ladder}{núcleons}); além de alguns $10^{-15}$ \omterm{def:g10:orders-of-magnitude:unit}{metros} ela desaparece: o seu
\emph{alcance}\index{alcance (de uma interação)} é de cerca de
\qty{e-15}{m}. Ela agarra prótons e nêutrons do mesmo jeito e ignora a
carga.
\end{definition}

\begin{remark}[Por que os núcleos existem --- e por que os pesados cedem]\label{rem:g11:fundamental-interactions:nuclei}
As duas \omterm{def:g10:inertia:force}{forças} do \omterm{def:g11:fundamental-interactions:ladder}{núcleo} escalam de modos diferentes. A cola forte tem
\omterm{def:g11:fundamental-interactions:strong}{alcance} curto: um \omterm{def:g11:fundamental-interactions:ladder}{núcleon} se liga apenas aos seus poucos vizinhos dentro
de \qty{e-15}{m}. A repulsão elétrica tem \omterm{def:g11:fundamental-interactions:strong}{alcance} longo: cada próton
empurra \emph{todos os outros} prótons. À medida que os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} crescem,
a repulsão se acumula mais depressa do que a cola: \omterm{def:g11:fundamental-interactions:ladder}{núcleos} pesados
precisam de nêutrons extras (cola sem carga) e, acima de uns $80$
prótons, nem isso basta --- os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} mais pesados vivem de tempo
emprestado (\cref{ch:g11:nucleus-radioactivity}).
\end{remark}

\begin{definition}[A interação fraca]\label{def:g11:fundamental-interactions:weak}
A \emph{interação fraca}\index{interação fraca}, de \omterm{def:g11:fundamental-interactions:strong}{alcance} ainda mais
curto, permite que um nêutron se transforme em próton (e vice-versa): é
ela que comanda a radioatividade $\beta$ do
\cref{ch:g11:nucleus-radioactivity}.
\end{definition}

\section{Quem manda em cada andar}

\begin{method}[Auditar uma escala]\label{met:g11:fundamental-interactions:audit}
Para descobrir que interação manda numa estrutura de tamanho $d$, confira
o \omterm{def:g11:fundamental-interactions:strong}{alcance} e o cancelamento:
\begin{enumerate}
  \item $d \leq \qty{e-14}{m}$: manda a \emph{\omterm{def:g11:fundamental-interactions:strong}{interação forte}} --- ela
        vence a repulsão elétrica, e a gravidade está $10^{36}$ fora de
        combate.
  \item dos \omterm{def:g11:fundamental-interactions:ladder}{átomos} às montanhas (de $\qty{e-10}{m}$ a $\qty{e4}{m}$): a
        \omterm{def:g10:inertia:force}{força} forte está fora de \omterm{def:g11:fundamental-interactions:strong}{alcance}; manda a \emph{\omterm{def:g11:fundamental-interactions:four}{interação
        eletromagnética}} --- ligações, coesão, \omterm{def:g10:inertia:inventory}{atrito}, contato.
  \item planetas e além ($d \geq \qty{e6}{m}$): a matéria é neutra com
        precisão fantástica, de modo que a eletricidade se cancela
        sozinha; a massa tem um único sinal e sempre se soma --- a
        \emph{\omterm{def:g11:fundamental-interactions:four}{gravitação}} manda no céu.
\end{enumerate}
A \omterm{def:g11:fundamental-interactions:weak}{interação fraca} não domina andar algum: ela não liga nada, mas arbitra
transformações (o decaimento $\beta$).
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:1}
Dê a \omterm{def:g10:orders-of-magnitude:oom}{ordem de grandeza} de cada tamanho e o seu andar na escada: próton
\qty{1.7e-15}{m}; \omterm{def:g11:fundamental-interactions:ladder}{molécula} de água \qty{2.8e-10}{m}; grão de areia
\qty{2e-4}{m}; Terra \qty{1.3e7}{m}; Via Láctea \qty{9.5e20}{m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:2}
Um balão esfregado no cabelo carrega $q = \qty{-4.8e-9}{C}$. Ele ganhou
ou perdeu elétrons, e quantos? Qual é a carga do cabelo depois disso, e
que lei diz isso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:3}
As cargas $q_1 = \qty{+2.0e-6}{C}$ e $q_2 = \qty{-3.0e-6}{C}$ estão a
\qty{30}{cm} uma da outra. Calcule a \omterm{def:g10:inertia:force}{força}; atrativa ou repulsiva?
Compare as \omterm{def:g10:inertia:force}{forças} sobre $q_1$ e sobre $q_2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:4}
Duas cargas se atraem com uma \omterm{def:g10:inertia:force}{força} $F_0$. Em que ela se transforma se: a
distância dobra; a distância é dividida por $3$; uma das cargas dobra; as
duas cargas \emph{e} a distância dobram?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:fundamental-interactions:5}
Nomeie a interação responsável por: (a) a Lua orbitando a Terra; (b) a
coesão de um cristal de sal; (c) a coesão de um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hélio; (d) o
decaimento $\beta$ do carbono-14; (e) o balão do
\cref{exo:g11:fundamental-interactions:2} grudando numa parede.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:6}
Os dois prótons de um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hélio estão a $d = \qty{2.0e-15}{m}$
($m_p = \qty{1.67e-27}{kg}$). Calcule a repulsão elétrica entre eles, a
atração gravitacional e a razão. O que mantém o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} inteiro?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:7}
Num \omterm{def:g11:fundamental-interactions:ladder}{átomo} de hidrogênio, o elétron ($m_e = \qty{9.11e-31}{kg}$) fica a
$d = \qty{5.3e-11}{m}$ do próton ($m_p = \qty{1.67e-27}{kg}$). Calcule as
\omterm{def:g10:inertia:force}{forças} elétrica e gravitacional entre eles e a razão. O que mantém os
\omterm{def:g11:fundamental-interactions:ladder}{átomos} unidos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:8}
A esfera A carrega $q_A = \qty{+8.0e-9}{C}$; uma esfera B idêntica está
neutra. Elas são encostadas uma na outra e depois separadas por
\qty{10}{cm}. Que carga cada uma leva (confira a conservação da carga)?
Calcule a \omterm{def:g10:inertia:force}{força} entre elas --- atrativa ou repulsiva?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:9}
Duas cargas idênticas $q = \qty{1.0e-6}{C}$ se repelem com \qty{0.90}{N}:
a que distância estão? Onde a \omterm{def:g10:inertia:force}{força} cai para \qty{0.10}{N}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:10}
Dois prótons em \omterm{def:g11:fundamental-interactions:ladder}{moléculas} vizinhas estão a $d = \qty{1.0e-10}{m}$.
Calcule a repulsão elétrica entre eles. Com o que a \omterm{def:g11:fundamental-interactions:strong}{interação forte}
contribui a essa distância? Que interação liga as \omterm{def:g11:fundamental-interactions:ladder}{moléculas}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:fundamental-interactions:11}
Cada uma de duas moedas contém cerca de $\num{e23}$ elétrons. Transfira
um elétron em cada \emph{milhão} de uma para a outra e coloque as moedas
a \qty{1.0}{m} uma da outra. Calcule a carga de cada moeda e depois a
\omterm{def:g10:inertia:force}{força}. Ela é igual ao \omterm{def:g10:universal-gravitation:weight}{peso} de que massa? Conclua sobre a neutralidade da
matéria do dia a dia.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:12}
Um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de urânio tem diâmetro \qty{1.4e-14}{m} e $92$ prótons.
Calcule a repulsão entre dois prótons em extremidades opostas. A
\omterm{def:g11:fundamental-interactions:strong}{interação forte} consegue ligar esse par --- por que não? Quantos pares de
prótons se repelem? Por que os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} pesados precisam de nêutrons
extras, e por que, acima de certo tamanho, nem os nêutrons os salvam?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:13}
Duas bolinhas de massa \qty{1.0}{g} pendem de fios de \qty{50}{cm}
presos no mesmo ponto. Recebendo a mesma carga $q$, elas se acomodam a
\qty{6.0}{cm} uma da outra. Cada bolinha está em equilíbrio sob o seu
\omterm{def:g10:universal-gravitation:weight}{peso}, a tensão do fio e a \omterm{def:g10:inertia:force}{força} elétrica: mostre que
$F = mg\tan\theta$ ($\theta$: ângulo do fio com a vertical) e que, aqui,
$\tan\theta \approx 0.060$. Calcule $F$, depois $q$ e depois o número de
\omterm{def:g11:fundamental-interactions:elementary}{cargas elementares} transferidas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:14}
Dados: $M_T = \qty{5.97e24}{kg}$, $M_L = \qty{7.35e22}{kg}$, distância

Terra--Lua. A gravidade é desligada: que cargas $+Q$ na Terra e $-Q$ na

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:fundamental-interactions:15}

\section{Problema: a auditoria das quatro forças}

\begin{problem}[{Problema de fim de semana --- por que o mundo não
desaba nem se desfaz: as quatro interações auditadas andar por andar, até
a neutralidade da matéria em duas partes em $10^{18}$}]
\label{pb:g11:fundamental-interactions:1}
Os \omterm{def:g11:fundamental-interactions:ladder}{átomos} não implodem, os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} em geral se seguram e a Lua nem cai
nem escapa; este problema audita quem merece o crédito. Dados:
$e = \qty{1.60e-19}{C}$, $k = \qty{8.99e9}{N.m^2/C^2}$,
$G = \qty{6.67e-11}{N.m^2/kg^2}$, $m_p = \qty{1.67e-27}{kg}$,
$m_e = \qty{9.11e-31}{kg}$, $M_T = \qty{5.97e24}{kg}$,
$M_L = \qty{7.35e22}{kg}$, distância Terra--Lua \qty{3.84e8}{m}.

\medskip
\noindent\textbf{Parte I --- A escada.}
\begin{enumerate}
  \item Liste os andares da escada, do \omterm{def:g11:fundamental-interactions:ladder}{quark} à galáxia, com uma \omterm{def:g10:orders-of-magnitude:oom}{ordem de
        grandeza} de tamanho para cada um.
  \item Maquete: o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} vira uma bolinha de \qty{1}{cm}; qual é a
        largura do \omterm{def:g11:fundamental-interactions:ladder}{átomo}?
  \item Que fração do volume de um \omterm{def:g11:fundamental-interactions:ladder}{átomo} o seu \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ocupa?
  \item Um fio de cabelo tem \qty{70}{\micro m} de espessura. Quantos
        \omterm{def:g11:fundamental-interactions:ladder}{átomos} o atravessam?
  \item Quantas potências de dez separam o teto do \omterm{def:g11:fundamental-interactions:ladder}{quark}
        (\qty{e-18}{m}) de uma galáxia (\qty{e21}{m})?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O andar elétrico.}
\begin{enumerate}[resume]
  \item Num \omterm{def:g11:fundamental-interactions:ladder}{átomo} de hidrogênio, o elétron fica a $d = \qty{5.3e-11}{m}$
        do próton: calcule a \omterm{def:g10:inertia:force}{força} elétrica entre eles.
  \item Calcule a \omterm{def:g10:universal-gravitation:force}{força gravitacional} entre eles e a razão entre as duas.
        Qual delas mantém o \omterm{def:g11:fundamental-interactions:ladder}{átomo} unido?
  \item O que muda se a gravidade for desligada dentro dos \omterm{def:g11:fundamental-interactions:ladder}{átomos}? E se
        for a eletricidade?
  \item Por que as \omterm{def:g10:inertia:force}{forças} de ``contato'' do dia a dia --- o piso
        empurrando os seus pés, o \omterm{def:g10:inertia:inventory}{atrito} --- são eletromagnéticas
        disfarçadas?
  \item Cada uma de duas moedas contém cerca de $\num{e23}$ elétrons.
        Transfira um elétron em cada \emph{bilhão} de uma para a outra e
        coloque-as a \qty{1.0}{m}: calcule as cargas e a \omterm{def:g10:inertia:force}{força}. Conclusão?
  \item Que interação manda nos andares de \qty{e-10}{m} a \qty{e4}{m}, e
        por que a \omterm{def:g11:fundamental-interactions:strong}{interação forte} não concorre?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O andar nuclear.}
\begin{enumerate}[resume]
  \item Dois prótons estão a \qty{1.0e-15}{m} num \omterm{def:g11:fundamental-interactions:ladder}{núcleo}: calcule a
        repulsão elétrica entre eles.
  \item Sem nada a conter, que aceleração essa \omterm{def:g10:inertia:force}{força} daria a um próton?
        Compare com $g$.
  \item Liste as três propriedades de que a \omterm{def:g11:fundamental-interactions:strong}{interação forte} precisa para
        explicar por que os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} existem, por que os nêutrons também
        são agarrados e por que os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} são minúsculos.
  \item Num \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de urânio (diâmetro \qty{1.4e-14}{m}, $92$ prótons),
        calcule a repulsão entre dois prótons em extremidades opostas. A
        \omterm{def:g11:fundamental-interactions:strong}{interação forte} consegue ligar esse par diretamente? Explique por
        que a repulsão vence à medida que os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} crescem e para que
        servem os nêutrons.
  \item Em uma frase, como prometido: o que faz a \omterm{def:g11:fundamental-interactions:weak}{interação fraca}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O andar astronômico e o veredito.}
\begin{enumerate}[resume]
  \item Calcule a \omterm{def:g10:universal-gravitation:force}{força gravitacional} entre a Terra e a Lua.
  \item A Terra contém cerca de $\num{1.8e51}$ prótons (e outros tantos
        elétrons), e a Lua, cerca de $\num{2.2e49}$. Que fração $f$ da
        carga de prótons de cada corpo, deixada sem cancelamento, faria a
        \omterm{def:g10:inertia:force}{força} elétrica igualar a gravitacional?
  \item Por que a \omterm{def:g11:fundamental-interactions:four}{gravitação}, a fraquinha da questão 7, manda na
        astronomia? Duas razões.
  \item Veredito --- uma frase por andar (\omterm{def:g11:fundamental-interactions:ladder}{núcleo}, do \omterm{def:g11:fundamental-interactions:ladder}{átomo} à montanha,
        planeta e acima): o que o mantém unido? Depois responda ao
        título: por que o mundo não desaba nem se desfaz?
\end{enumerate}
\end{problem}
