\chapter{Campos elétrico e gravitacional}\label{ch:g11:electric-gravitational-fields}

Como a Terra puxa uma maçã em que nunca toca? O próprio Newton chamou a
ação a distância de ``um absurdo tão grande'' que se recusou a defendê-la.
A resposta moderna: uma massa, ou uma carga, modifica o espaço à sua
volta --- ela cria um \emph{\omterm{def:g11:electric-gravitational-fields:field}{campo}} --- e o que quer que seja colocado ali
responde ao \omterm{def:g11:electric-gravitational-fields:field}{campo} na sua própria posição. Construímos os dois grandes
exemplos lado a lado, até a analogia virar uma identidade de forma.

\section{Da ação a distância ao campo}

\begin{definition}[Campo]\label{def:g11:electric-gravitational-fields:field}
Um \emph{campo}\index{campo} atribui um vetor a cada ponto do espaço. Uma
fonte (uma carga, uma massa) cria o seu campo em toda a sua volta, haja
ou não alguma coisa ali para senti-lo; um objeto de prova sente então uma
\omterm{def:g10:inertia:force}{força} determinada apenas pelo campo \emph{na sua própria posição}. Essa
imagem em duas etapas substitui a ação a distância.
\end{definition}

\begin{remark}
O \omterm{def:g11:electric-gravitational-fields:field}{campo} não é um truque de contabilidade: ele carrega \omterm{def:g10:energy-conservation:energy}{energia}, e as
mudanças na fonte se propagam para fora através dele com velocidade
finita --- a própria luz é um \omterm{def:g11:electric-gravitational-fields:field}{campo} assim, em propagação
(\cref{ch:g12:light-as-wave}). Aqui estudamos \omterm{def:g11:electric-gravitational-fields:field}{campos} que não variam com
o tempo.
\end{remark}

\section{O campo elétrico}

\begin{definition}[Campo elétrico]\label{def:g11:electric-gravitational-fields:efield}
Se uma pequena carga de prova $q$ colocada num ponto $M$ sente uma \omterm{def:g10:inertia:force}{força}
elétrica $\vect F$, o \emph{campo elétrico}\index{campo elétrico} em $M$ é
\[
\vect E = \frac{\vect F}{q}
\qquad \text{(em \unit{N/C}, também escrito \unit{V/m})}.
\]
$\vect E$ depende só das fontes e de $M$, e não da carga de prova: dobrar
$q$ dobra $\vect F$ e deixa o quociente inalterado.
\end{definition}

\begin{proposition}[Força sobre uma carga]\label{prop:g11:electric-gravitational-fields:force}
Uma carga $q$ colocada onde o \omterm{def:g11:electric-gravitational-fields:field}{campo} vale $\vect E$ sente a \omterm{def:g10:inertia:force}{força}
$\vect F = q \vect E$: no sentido de $\vect E$ se $q > 0$, no sentido
oposto se $q < 0$, de módulo $F = \abs{q}\,E$.
\end{proposition}

\begin{proof}
Basta reorganizar a definição; a regra dos sinais é a álgebra de um
múltiplo negativo.
\end{proof}

\begin{example}[Ler um campo]\label{ex:g11:electric-gravitational-fields:reading}
Uma carga de prova $q = \qty{+2.0e-8}{C}$ em $M$ sente uma \omterm{def:g10:inertia:force}{força} de
\qty{6.0e-4}{N} apontando para leste. O \omterm{def:g11:electric-gravitational-fields:field}{campo} em $M$ vale
$E = \num{6.0e-4} / \num{2.0e-8} = \qty{3.0e4}{N/C}$, apontando para
leste; uma carga $q' = \qty{-1.0e-8}{C}$ colocada no mesmo ponto sente
$F = \num{1.0e-8} \times \num{3.0e4} = \qty{3.0e-4}{N}$ --- apontando
para oeste.
\end{example}

\begin{proposition}[Campo de uma carga pontual]\label{prop:g11:electric-gravitational-fields:point}
Uma carga pontual $Q$ cria, à distância $d$, um \omterm{def:g11:electric-gravitational-fields:field}{campo} ao longo da reta
que une a carga ao ponto --- afastando-se de $Q$ se $Q > 0$,
aproximando-se dela se $Q < 0$ --- de módulo
\[
E = k\,\frac{\abs{Q}}{d^2},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{proposition}

\begin{proof}
A lei de Coulomb (\cref{ch:g11:fundamental-interactions}) dá a \omterm{def:g10:inertia:force}{força}
sobre uma carga de prova $q$ à distância $d$: módulo $k \abs{qQ} / d^2$,
ao longo da reta que as une, repulsiva para sinais iguais. Divida por $q$
e aplique a \cref{prop:g11:electric-gravitational-fields:force}.
\end{proof}

\begin{example}[Uma esfera carregada]\label{ex:g11:electric-gravitational-fields:sphere}
A esfera de um gerador de Van de Graaff carrega $Q = \qty{+5.0e-7}{C}$
(uma pequena esfera carregada age como uma carga pontual no seu centro).
A $d = \qty{0.50}{m}$:
\[
E = \num{8.99e9} \times \num{5.0e-7} / 0.50^2 \approx \qty{1.8e4}{N/C},
\]
apontando radialmente para longe da esfera. A \qty{1.0}{m} ele caiu para
um quarto disso: o $1/d^2$ da \omterm{def:g10:inertia:force}{força} sobrevive no \omterm{def:g11:electric-gravitational-fields:field}{campo}.
\end{example}

\begin{definition}[Linhas de campo]\label{def:g11:electric-gravitational-fields:lines}
Uma \emph{linha de campo}\index{linha de campo} é uma curva tangente ao
\omterm{def:g11:electric-gravitational-fields:field}{campo} em todos os seus pontos, com uma flecha dando o sentido do \omterm{def:g11:electric-gravitational-fields:field}{campo}.
Três regras:
\begin{itemize}
  \item elas saem das cargas positivas e chegam às cargas negativas;
  \item onde as linhas se apertam o \omterm{def:g11:electric-gravitational-fields:field}{campo} é forte; onde se espalham, é
        fraco;
  \item duas linhas nunca se cruzam: o \omterm{def:g11:electric-gravitational-fields:field}{campo} tem um único sentido em cada
        ponto.
\end{itemize}
\end{definition}

\section{O campo uniforme de um capacitor}

\begin{definition}[Capacitor de placas paralelas]\label{def:g11:electric-gravitational-fields:capacitor}
Duas placas metálicas paralelas, frente a frente, próximas uma da outra e
com cargas opostas, formam um \emph{capacitor}\index{capacitor}. Entre as
placas o \omterm{def:g11:electric-gravitational-fields:field}{campo} é \emph{uniforme}\index{campo uniforme}: mesmo módulo e
mesmo sentido em toda parte --- perpendicular às placas, da placa
positiva para a negativa.
\end{definition}

\begin{proposition}[Campo entre as placas]\label{prop:g11:electric-gravitational-fields:uniform}
Se a tensão entre as placas (\cref{ch:g11:circuits-and-power}) vale $U$ e
a separação delas é $d$, o \omterm{def:g11:electric-gravitational-fields:field}{campo} entre elas tem módulo
\[
E = \frac{U}{d} \qquad \text{(em \unit{V/m})}.
\]
\end{proposition}
\admitted

\begin{remark}
A uniformidade e $E = U/d$ são deduzidas com honestidade no volume do
primeiro ano de graduação; a fórmula explica a \omterm{def:g10:orders-of-magnitude:unit}{unidade} \unit{V/m}.
Intervalos finos dão \omterm{def:g11:electric-gravitational-fields:field}{campos} fortes: \qty{100}{V} através de um milímetro
já são \qty{e5}{V/m}.
\end{remark}

\begin{example}[Desviar um feixe de elétrons]\label{ex:g11:electric-gravitational-fields:beam}
Num \omterm{def:g10:signals-and-waves:oscilloscope}{osciloscópio}, o feixe passa entre placas com $U = \qty{100}{V}$ e
$d = \qty{5.0}{mm}$: $E = 100 / \num{5.0e-3} = \qty{2.0e4}{V/m}$, e cada
elétron sente $F = \num{1.60e-19} \times \num{2.0e4} = \qty{3.2e-15}{N}$
em direção à placa positiva --- $\num{4e14}$ vezes o seu \omterm{def:g10:universal-gravitation:weight}{peso} de
\qty{8.9e-30}{N}, de modo que o feixe se desvia visivelmente (a
\omterm{def:g10:relative-motion:trajectory}{trajetória} curva é assunto do \cref{ch:g12:newtons-laws}).
\end{example}

\section{O campo gravitacional}

\begin{definition}[Campo gravitacional]\label{def:g11:electric-gravitational-fields:gfield}
Se uma massa de prova $m$ colocada num ponto sente uma \omterm{def:g10:universal-gravitation:force}{força
gravitacional} (um \omterm{def:g10:universal-gravitation:weight}{peso}) $\vect P$, o \emph{campo
gravitacional}\index{campo gravitacional} ali é
\[
\vect g = \frac{\vect P}{m} \qquad \text{(em \unit{N/kg})},
\]
e, reciprocamente, $\vect P = m \vect g$. Um capítulo anterior
introduziu o número $g$; $\vect g$ acrescenta a direção em que uma pedra
solta começa a se mover.
\end{definition}

\begin{proposition}[Campo da Terra]\label{prop:g11:electric-gravitational-fields:earth}
À distância $d$ do centro da Terra (de massa $M$), o \omterm{def:g11:electric-gravitational-fields:gfield}{campo gravitacional}
aponta para o centro e tem módulo
\[
g = G\,\frac{M}{d^2},
\qquad G = \qty{6.67e-11}{N.m^2/kg^2}.
\]
\end{proposition}

\begin{proof}
A \omterm{def:g11:fundamental-interactions:four}{gravitação} universal (\cref{ch:g10:universal-gravitation}) dá a \omterm{def:g10:inertia:force}{força}
sobre uma massa de prova $m$: módulo $G M m / d^2$, em direção ao centro;
divida por $m$. Na superfície, $d = R$ e isso é o
$g = GM/R^2 \approx \qty{9.81}{N/kg}$ do capítulo anterior.
\end{proof}

\begin{example}[O campo onde a Lua vive]\label{ex:g11:electric-gravitational-fields:moon}
À distância da Lua, $d = \qty{3.84e8}{m}$:
\[
g = \num{6.67e-11} \times \num{5.97e24} / (\num{3.84e8})^2
\approx \qty{2.7e-3}{N/kg}.
\]
O \omterm{def:g11:electric-gravitational-fields:field}{campo} da Terra nunca acaba --- ele apenas rarefaz como $1/d^2$; é esse
\omterm{def:g11:electric-gravitational-fields:field}{campo} que mantém a Lua na sua \omterm{def:g10:universal-gravitation:satellite}{órbita}.
\end{example}

\begin{remark}[Uniforme perto da superfície]\label{rem:g11:electric-gravitational-fields:uniformg}
Ao longo de um laboratório, ou mesmo de uma montanha, $d$ quase não muda
em comparação com $R = \qty{6.37e6}{m}$, de modo que $\vect g$ é \omterm{def:g11:electric-gravitational-fields:capacitor}{uniforme}
com grande precisão: linhas verticais, \qty{9.81}{N/kg} constantes --- o
\omterm{def:g11:electric-gravitational-fields:field}{campo} da Terra parece o do \omterm{def:g11:electric-gravitational-fields:capacitor}{capacitor}, assim como a Terra redonda parece
plana vista de um jardim.
\end{remark}

\section{Uma ideia, duas forças}

Os dois \omterm{def:g11:electric-gravitational-fields:field}{campos} deste capítulo são a mesma matemática em dois trajes:

\begin{center}
\begin{tabular}{lcc}
 & \textbf{elétrico} & \textbf{gravitacional} \\
\hline
fonte & carga $Q$ & massa $M$ \\
\omterm{def:g11:electric-gravitational-fields:field}{campo} & $\vect E = \vect F / q$ & $\vect g = \vect P / m$ \\
\omterm{def:g10:orders-of-magnitude:unit}{unidade} & \unit{N/C} = \unit{V/m} & \unit{N/kg} \\
fonte pontual & $E = k \abs{Q} / d^2$ & $g = G M / d^2$ \\
constante & $k = \qty{8.99e9}{N.m^2/C^2}$ & $G = \qty{6.67e-11}{N.m^2/kg^2}$ \\
\omterm{def:g10:inertia:force}{força} sobre a prova & $\vect F = q \vect E$ & $\vect P = m \vect g$ \\
\omterm{def:g11:electric-gravitational-fields:lines}{linhas de campo} & saem do $+$, entram no $-$ & sempre entram na massa \\
\end{tabular}
\end{center}

Toda fórmula da esquerda vira a da direita sob o dicionário
$q \leftrightarrow m$, $\vect E \leftrightarrow \vect g$,
$k \leftrightarrow G$.

\begin{remark}[Onde a analogia se rompe]\label{rem:g11:electric-gravitational-fields:noanalogue}
A carga vem em dois sinais; a massa, em um só. Por isso os \omterm{def:g11:electric-gravitational-fields:efield}{campos
elétricos} podem ser cancelados: as cargas rearranjadas de uma caixa
metálica apagam qualquer \omterm{def:g11:electric-gravitational-fields:field}{campo} externo dentro dela (uma \emph{gaiola de
Faraday} --- é por isso que um elevador mata o \omterm{def:g10:signals-and-waves:signal}{sinal} do seu celular), e a
matéria em bloco, com $+$ e $-$ equilibrados, é eletricamente silenciosa.
A gravidade não pode ser blindada nem neutralizada: cada \omterm{def:g10:orders-of-magnitude:unit}{quilograma}
acrescenta o seu puxão --- e é por isso que a \omterm{def:g10:inertia:force}{força} mais fraca por
$\num{e39}$ (\cref{exo:g11:electric-gravitational-fields:10}) comanda o
universo em grande escala.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:1}
Uma carga de prova $q = \qty{+2.0e-6}{C}$ num ponto $M$ sente uma \omterm{def:g10:inertia:force}{força}
de \qty{5.0e-3}{N} apontando para o norte. Dê o \omterm{def:g11:electric-gravitational-fields:field}{campo} em $M$ (módulo e
direção) e depois a \omterm{def:g10:inertia:force}{força} sobre $q' = \qty{-4.0e-6}{C}$ colocada em $M$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:2}
Uma carga pontual $Q = \qty{+5.0}{nC}$ está em $O$.
\begin{enumerate}
  \item Calcule o \omterm{def:g11:electric-gravitational-fields:field}{campo} a $d = \qty{30}{cm}$: módulo e direção.
  \item A que distância o \omterm{def:g11:electric-gravitational-fields:field}{campo} tem metade dessa intensidade?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:3}
Um \omterm{def:g11:electric-gravitational-fields:capacitor}{capacitor} tem $U = \qty{600}{V}$ entre placas separadas por
\qty{3.0}{mm}. Calcule o \omterm{def:g11:electric-gravitational-fields:field}{campo} e depois a \omterm{def:g10:inertia:force}{força} sobre um elétron ali
($e = \qty{1.60e-19}{C}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:4}
Em Marte, uma sonda de \qty{250}{kg} pesa \qty{930}{N}. Calcule o \omterm{def:g11:electric-gravitational-fields:field}{campo}
na superfície marciana $g_{\text{Marte}}$ e depois o \omterm{def:g10:universal-gravitation:weight}{peso} de um
astronauta de \qty{80}{kg} lá.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:5}
Verdadeiro ou falso, com uma razão para cada: (a) duas \omterm{def:g11:electric-gravitational-fields:lines}{linhas de campo}
podem se cruzar onde o \omterm{def:g11:electric-gravitational-fields:field}{campo} é forte; (b) as \omterm{def:g11:electric-gravitational-fields:lines}{linhas de campo} saem das
cargas negativas e chegam às positivas; (c) \omterm{def:g11:electric-gravitational-fields:lines}{linhas de campo} agrupadas
indicam \omterm{def:g11:electric-gravitational-fields:field}{campo} forte.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:6}
A Estação Espacial Internacional orbita a uma altitude de \qty{420}{km}
($R = \qty{6.37e6}{m}$, $M = \qty{5.97e24}{kg}$).
\begin{enumerate}
  \item Calcule $g$ na estação; que fração do valor na superfície?
  \item Os astronautas flutuam. Concilie isso com a sua resposta em uma
        frase.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:7}
Que módulo de \omterm{def:g11:electric-gravitational-fields:field}{campo} equilibraria o \omterm{def:g10:universal-gravitation:weight}{peso} de um elétron
($m_e = \qty{9.11e-31}{kg}$)? Compare com os \qty{2.0e4}{V/m} do
\cref{ex:g11:electric-gravitational-fields:beam}; conclua sobre o lugar
da gravidade na física do elétron.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:8}
Um grão de poeira de massa \qty{2.0e-6}{kg} flutua em repouso entre
placas horizontais separadas por \qty{2.0}{cm} com $U = \qty{5.0}{kV}$
entre elas.
\begin{enumerate}
  \item Calcule o \omterm{def:g11:electric-gravitational-fields:field}{campo} e depois o módulo da carga do grão.
  \item A carga do grão é positiva: qual das placas é a positiva?
  \item Quantas \omterm{def:g11:fundamental-interactions:elementary}{cargas elementares} o grão carrega?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:9}
As cargas $+Q$ e $+4Q$ estão a \qty{30}{cm} uma da outra. Em que ponto do
segmento que as une o \omterm{def:g11:electric-gravitational-fields:field}{campo} total é nulo? Explique primeiro por que esse
ponto tem de estar entre as cargas e mais perto da menor.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:10}
Num \omterm{def:g11:fundamental-interactions:ladder}{átomo} de hidrogênio, o próton ($m_p = \qty{1.67e-27}{kg}$, carga
$+e$) e o elétron ($m_e = \qty{9.11e-31}{kg}$, carga $-e$) estão a
$d = \qty{5.3e-11}{m}$. Calcule as \omterm{def:g10:inertia:force}{forças} elétrica e gravitacional entre
eles e a razão entre elas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:11}
Num ponto $M$, uma fonte sozinha criaria um \omterm{def:g11:electric-gravitational-fields:field}{campo} de \qty{300}{N/C}
apontando para leste; uma segunda, de \qty{400}{N/C} apontando para o
norte. Os \omterm{def:g11:electric-gravitational-fields:field}{campos} se somam como vetores: dê o \omterm{def:g11:electric-gravitational-fields:field}{campo} total em $M$ (módulo e
direção) e depois a \omterm{def:g10:inertia:force}{força} sobre $q = \qty{-2.0e-6}{C}$ colocada ali.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:electric-gravitational-fields:12}
A que altitude o \omterm{def:g11:electric-gravitational-fields:field}{campo} da Terra caiu à metade do seu valor na superfície?
E na altitude $h = R$, a que fração?
\end{exercise}

\section{Problema: a gotícula de Millikan}

\begin{problem}[{Problema de fim de semana --- pesar a carga do elétron:
uma gotícula de óleo estacionada no ar entre duas placas revela que a
carga vem em degraus indivisíveis de $\num{1.60e-19}$ \omterm{def:g11:fundamental-interactions:charge}{coulomb}}]
\label{pb:g11:electric-gravitational-fields:1}
Em 1909, Robert Millikan pulverizou gotículas de óleo entre as placas
horizontais de um \omterm{def:g11:electric-gravitational-fields:capacitor}{capacitor}. Ajustando a tensão até que uma gotícula
escolhida ficasse imóvel --- com a \omterm{def:g10:inertia:force}{força} elétrica equilibrando
exatamente o \omterm{def:g10:universal-gravitation:weight}{peso} ---, ele conseguia pesar a carga dela. A nossa gotícula
tem massa $m = \qty{4.90e-15}{kg}$ (medida a partir da sua queda lenta
com o \omterm{def:g11:electric-gravitational-fields:field}{campo} desligado; tomamos esse valor como dado); as placas estão a
$d = \qty{8.0}{mm}$ uma da outra, com a placa \emph{de cima} positiva; a
carga da gotícula é negativa.

\medskip
\noindent\textbf{Parte I --- O cenário.}
\begin{enumerate}
  \item Descreva o \omterm{def:g11:electric-gravitational-fields:field}{campo} entre as placas; o que ``\omterm{def:g11:electric-gravitational-fields:capacitor}{uniforme}'' garante?
  \item Para $U = \qty{800}{V}$, calcule $E$.
  \item Em que sentido aponta a \omterm{def:g10:inertia:force}{força} elétrica sobre a gotícula? Por que a
        placa \emph{de cima} tinha de ser a positiva?
  \item Calcule o \omterm{def:g10:universal-gravitation:weight}{peso} da gotícula.
  \item Escreva a condição de repouso como uma igualdade entre dois
        módulos de \omterm{def:g10:inertia:force}{força}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O equilíbrio.}
\begin{enumerate}[resume]
  \item A partir dela, exprima $\abs{q}$ em função de $m$, $g$, $d$ e $U$.
  \item A gotícula fica em repouso para $U = \qty{800}{V}$: calcule
        $\abs{q}$.
  \item Divida por $e = \qty{1.60e-19}{C}$. Quantos elétrons em excesso a
        gotícula carrega?
  \item A tensão é ligeiramente elevada até \qty{900}{V}. Em que sentido
        aponta agora a \omterm{def:g10:inertia:force}{força} resultante?
  \item Mostre que a tensão de equilíbrio vale $U = mgd/\abs{q}$: quanto
        maior a carga, menor a tensão.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Os degraus.}
Um pulso de radiação pode arrancar elétrons da gotícula ou depositá-los
nela; depois de cada mudança, Millikan reajusta a tensão para
reequilibrá-la.
\begin{enumerate}[resume]
  \item Depois de um pulso, o equilíbrio exige $U = \qty{1200}{V}$.
        Calcule o novo $\abs{q}$. A gotícula ganhou ou perdeu um elétron?
  \item Equilíbrios sucessivos são encontrados em $U = 1200$, $800$,
        $600$, $480$ e \qty{400}{V}. Calcule as cinco cargas.
  \item Mostre que as cinco são múltiplos inteiros de uma mesma
        quantidade; dê o valor dela.
  \item Entre equilíbrios consecutivos, de quanto a carga muda?
        Interprete.
  \item Explique por que essa gotícula jamais consegue ficar em repouso
        com $U = \qty{700}{V}$, por mais paciência que se tenha no ajuste.
  \item Millikan repetiu isso com centenas de gotículas: toda carga
        medida era múltiplo inteiro do mesmo valor. Enuncie a conclusão e
        nomeie a grandeza medida.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Por que não existe um Millikan gravitacional.}
\begin{enumerate}[resume]
  \item No dicionário deste capítulo, o que faz o papel de $q$, de $E$ e
        de $qE$ no equilíbrio da Parte II?
  \item Uma massa colocada acima poderia sustentar a gotícula? Calcule o
        \omterm{def:g11:electric-gravitational-fields:field}{campo} de uma esfera de chumbo de \qty{1000}{kg} a \qty{1.0}{m}
        acima da gotícula, a \omterm{def:g10:inertia:force}{força} sobre a gotícula e compare com o \omterm{def:g10:universal-gravitation:weight}{peso}
        dela.
  \item O equilíbrio precisa de um puxão para cima forte o bastante para
        enfrentar a Terra inteira. Por que uma placa carregada consegue
        fornecê-lo e arranjo algum de massas consegue?
  \item A \emph{massa} da gotícula também muda em degraus (uma \omterm{def:g11:fundamental-interactions:ladder}{molécula}
        de óleo: cerca de \qty{5e-25}{kg}). Compare os saltos relativos em
        $m$ e em $\abs{q}$ quando um elétron pousa nela e conclua: por que
        esse equilíbrio enxerga os \omterm{def:g11:fundamental-interactions:ladder}{átomos} da eletricidade, mas não as
        \omterm{def:g11:fundamental-interactions:ladder}{moléculas} do óleo? Desfecho: a carga é quantizada, e o \omterm{def:g11:fundamental-interactions:ladder}{átomo} dela
        é $e = \qty{1.60e-19}{C}$.
\end{enumerate}
\end{problem}
