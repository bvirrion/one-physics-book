\chapter{O núcleo e a radioatividade}\label{ch:g11:nucleus-radioactivity}

Neste exato momento, uns oito mil núcleos atômicos explodem dentro do seu
corpo a cada segundo. Nada os ataca: certos \omterm{def:g11:fundamental-interactions:ladder}{núcleos} nascem instáveis, e
cada um deles, cedo ou tarde, se transforma, arremessando um fragmento a
uma boa fração da velocidade da luz. Este capítulo volta ao andar dos
$10^{-15}$ \omterm{def:g10:orders-of-magnitude:unit}{metros} do \cref{ch:g11:fundamental-interactions} para ver que
\omterm{def:g11:fundamental-interactions:ladder}{núcleos} são frágeis, como eles se rompem e como a impaciência regular
deles data cavernas e vigia tetos.

\section{Núcleos, núcleons e isótopos}

\begin{definition}[Notação nuclear]\label{def:g11:nucleus-radioactivity:notation}
Um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} contém $Z$ prótons (de carga $+e$ cada) e $N$ nêutrons (sem
carga), num total de $A = Z + N$ \omterm{def:g11:fundamental-interactions:ladder}{núcleons}; ele se escreve
${}^{A}_{Z}\mathrm{X}$, em que $\mathrm X$ é o símbolo químico, $Z$ é o
\emph{número atômico}\index{número atômico} e $A$, o \emph{número de
massa}\index{número de massa}. Uma espécie especificada por $A$ e $Z$ é um
\emph{nuclídeo}\index{nuclídeo}; nuclídeos de mesmo $Z$ e $A$ diferentes
são \emph{isótopos}\index{isótopo} de um mesmo elemento: mesma química (a
nuvem de elétrons só enxerga $Ze$), \omterm{def:g11:fundamental-interactions:ladder}{núcleos} e estabilidade diferentes.
\end{definition}

\begin{example}[Hidrogênio e carbono]\label{ex:g11:nucleus-radioactivity:hc}
O hidrogênio tem três \omterm{def:g11:nucleus-radioactivity:notation}{isótopos}: ${}^{1}_{1}\mathrm{H}$ (um próton
solitário), ${}^{2}_{1}\mathrm{H}$ (deutério: um próton e um nêutron) e o
instável ${}^{3}_{1}\mathrm{H}$ (trítio: dois nêutrons). O carbono
natural: quase todo ${}^{12}_{6}\mathrm{C}$ ($6$ prótons, $6$ nêutrons),
$1.1\%$ de ${}^{13}_{6}\mathrm{C}$ e um \omterm{def:g11:fundamental-interactions:ladder}{átomo} em $10^{12}$ do instável
${}^{14}_{6}\mathrm{C}$ ($8$ nêutrons), herói do
\cref{ex:g11:nucleus-radioactivity:dating}.
\end{example}

\section{Núcleos estáveis e instáveis}

\begin{proposition}[O vale de estabilidade]\label{prop:g11:nucleus-radioactivity:valley}
Os \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeos} estáveis ocupam uma faixa estreita do plano $(Z, N)$:
$N \approx Z$ para os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} leves e depois um excesso crescente de
nêutrons, até $N \approx 1.5\,Z$. Todo \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo} fora da faixa ---
nêutrons demais, prótons demais ou tamanho demais (acima do chumbo,
$Z = 82$, \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo} algum é verdadeiramente estável) --- se transforma
cedo ou tarde.
\end{proposition}
\admitted

\begin{remark}
Localizar a faixa com exatidão é mecânica quântica --- assunto dos volumes
de graduação. O padrão é o balanço do
\cref{ch:g11:fundamental-interactions}: a cola forte liga apenas
vizinhos, a repulsão de Coulomb atravessa o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} inteiro, e por isso os
\omterm{def:g11:fundamental-interactions:ladder}{núcleos} pesados precisam de cola extra de nêutrons; e, no entanto, um
excesso de nêutrons também é instável (a \omterm{def:g11:fundamental-interactions:weak}{interação fraca} cuida disso).
\end{remark}

\begin{definition}[Radioatividade]\label{def:g11:nucleus-radioactivity:radioactivity}
A \emph{radioatividade}\index{radioatividade} é a transformação espontânea
de um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} instável em outro \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo}, com emissão de radiação. Ela é
\emph{aleatória}: nada anuncia qual \omterm{def:g11:fundamental-interactions:ladder}{núcleo} decai em seguida, e \omterm{def:g10:pressure:pressure}{pressão},
temperatura ou química alguma consegue apressá-la ou atrasá-la; só
populações grandes são previsíveis
(\cref{def:g11:nucleus-radioactivity:halflife}).
\end{definition}

\begin{remark}[Becquerel e os Curie]\label{rem:g11:nucleus-radioactivity:curie}
Em 1896, Henri Becquerel descobriu que sais de urânio velam uma chapa
fotográfica através de papel preto, dentro de uma gaveta fechada, sem
\omterm{def:g10:energy-conservation:energy}{energia} fornecida alguma. Marie e Pierre Curie mostraram que o efeito é
atômico, mediram-no e isolaram o polônio e o rádio, um milhão de vezes
mais ativos. A palavra \emph{\omterm{def:g11:nucleus-radioactivity:radioactivity}{radioatividade}} é de Marie Curie; e dois
prêmios Nobel também.
\end{remark}

\section{Alfa, beta, gama}

\begin{definition}[As três radiações históricas]\label{def:g11:nucleus-radioactivity:abg}
Três radiações, batizadas antes que alguém soubesse o que elas eram:
\begin{itemize}
  \item \emph{alfa}\index{radiação alfa} ($\alpha$): um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de hélio
        ${}^{4}_{2}\mathrm{He}$ ejetado inteiro --- pesado, com carga
        dupla, barrado por uma folha de papel ou por centímetros de ar;
  \item \emph{beta}\index{radiação beta} ($\beta^-$): um elétron
        ${}^{\;0}_{-1}\mathrm{e}$ criado no \omterm{def:g11:fundamental-interactions:ladder}{núcleo} e ejetado --- leve,
        rápido, barrado por milímetros de alumínio;
  \item \emph{gama}\index{radiação gama} ($\gamma$): radiação
        eletromagnética, muito mais energética do que a luz --- nunca
        barrada por completo, apenas
        \emph{atenuada}\index{atenuação}: um centímetro de chumbo absorve
        metade.
\end{itemize}
\end{definition}

\begin{proposition}[Conservação nas equações nucleares]\label{prop:g11:nucleus-radioactivity:soddy}
Em toda transformação nuclear, o \omterm{def:g11:nucleus-radioactivity:notation}{número de massa} total $A$ e a carga
total --- a soma dos índices inferiores, contando $-1$ para um elétron
emitido --- são os mesmos antes e depois.
\end{proposition}
\admitted

\begin{remark}
A conservação da carga já governava o
\cref{ch:g11:fundamental-interactions}; a conservação de $A$ diz que os
\omterm{def:g11:fundamental-interactions:ladder}{núcleons} são apenas convertidos, nunca criados nem destruídos --- uma
contabilidade honesta, para os volumes de graduação.
\end{remark}

\begin{method}[Equilibrar uma equação de decaimento]\label{met:g11:nucleus-radioactivity:balance}
\begin{enumerate}
  \item Escreva todos os atores com os dois rótulos:
        $\alpha = {}^{4}_{2}\mathrm{He}$,
        $\beta^- = {}^{\;0}_{-1}\mathrm{e}$,
        $\beta^+ = {}^{0}_{+1}\mathrm{e}$; um $\gamma$ carrega
        $A = Z = 0$.
  \item Iguale as somas dos $A$ e depois as dos $Z$
        (\cref{prop:g11:nucleus-radioactivity:soddy}); resolva e leia o
        elemento-filho pelo seu $Z$ na tabela periódica.
\end{enumerate}
\end{method}

\begin{example}[Um decaimento alfa]\label{ex:g11:nucleus-radioactivity:alpha}
O rádio-226, o rádio dos Curie, é um emissor $\alpha$:
${}^{226}_{88}\mathrm{Ra} \to {}^{A}_{Z}\mathrm{Y} + {}^{4}_{2}\mathrm{He}$
obriga $A = 226 - 4 = 222$ e $Z = 88 - 2 = 86$; o elemento $86$ é o
radônio, logo
${}^{226}_{88}\mathrm{Ra} \to {}^{222}_{86}\mathrm{Rn} + {}^{4}_{2}\mathrm{He}$
--- o gás radioativo dos porões sem ventilação.
\end{example}

\begin{example}[Um decaimento beta-menos]\label{ex:g11:nucleus-radioactivity:beta}
O carbono-14 fica acima da faixa ($8$ nêutrons contra $6$ prótons):
${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} + {}^{\;0}_{-1}\mathrm{e}$.
Conferindo: $14 = 14 + 0$ e $6 = 7 - 1$; um nêutron virou um próton mais o
elétron ejetado --- a \omterm{def:g11:fundamental-interactions:weak}{interação fraca} em ação.
\end{example}

\begin{remark}[Beta-mais e gama]\label{rem:g11:nucleus-radioactivity:betaplus}
Abaixo da faixa, o decaimento espelhado $\beta^+$ transforma um próton em
um nêutron mais um \emph{pósitron}\index{pósitron}
${}^{0}_{+1}\mathrm{e}$, a antipartícula do elétron:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$,
o burro de carga dos exames PET. E, depois da maioria dos decaimentos, o
núcleo-filho, que nasce trepidando, se acomoda emitindo um $\gamma$ por
cima.
\end{remark}

\section{Meia-vida e atividade}

\begin{definition}[Meia-vida]\label{def:g11:nucleus-radioactivity:halflife}
A \emph{meia-vida}\index{meia-vida} $T_{1/2}$ de um \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo} é o tempo ao
fim do qual metade de qualquer amostra grande já decaiu. Cada meia-vida
adicional reduz o restante à metade: de $N_0$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}, sobram $N_0/2$ em
$T_{1/2}$, $N_0/4$ em $2\,T_{1/2}$, $N_0/8$ em $3\,T_{1/2}$ e
$N_0/2^{\,n}$ depois de $n$ meias-vidas.
\end{definition}

\begin{remark}
Aleatoriedade para um \omterm{def:g11:fundamental-interactions:ladder}{núcleo}, relojoaria para $10^{23}$: uma multidão
imensa de jogadores de cara ou coroa, imprevisíveis um a um, com metade
eliminada a cada rodada. Interpolar \emph{entre} meias-vidas inteiras
exige a função exponencial --- no ano que vem, junto com o volume de
matemática.
\end{remark}

\begin{example}[Vinte potências de dez]\label{ex:g11:nucleus-radioactivity:range}
\begin{center}\small\begin{tabular}{llll}
polônio-214 & $\alpha$ & \qty{1.6e-4}{s} & elo da cadeia do rádio \\
tecnécio-99m & $\gamma$ & $6.0$ horas & exames de imagem \\
amerício-241 & $\alpha$ & $432$ anos & detectores de fumaça \\
carbono-14 & $\beta^-$ & \num{5700} anos & arqueologia \\
potássio-40 & $\beta$ & \num{1.25e9} anos & em toda banana \\
urânio-238 & $\alpha$ & \num{4.5e9} anos & calor interno da Terra \\
\end{tabular}\end{center}
\end{example}

\begin{example}[Datação por carbono-14]\label{ex:g11:nucleus-radioactivity:dating}
A matéria viva troca carbono com a atmosfera e, por isso, guarda a
proporção atmosférica de carbono-14; na morte a reposição para e a
proporção cai à metade a cada \num{5700} anos. O carvão de uma caverna
pintada mostra um quarto da proporção viva: um quarto é a metade da
metade, de modo que o fogo queimou duas meias-vidas atrás, ou seja,
$2 \times \num{5700} = \num{11400}$ anos.
\end{example}

\begin{definition}[Atividade]\label{def:g11:nucleus-radioactivity:activity}
A \emph{atividade}\index{atividade} $\mathcal A$ de uma amostra é o seu
número de decaimentos por segundo, medida em
\emph{becquerels}\index{becquerel}: $\qty{1}{Bq} = 1$ decaimento por
segundo. Menos \omterm{def:g11:fundamental-interactions:ladder}{núcleos} sobreviventes significam menos decaimentos, de modo
que a atividade também cai à metade a cada \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}.
\end{definition}

\begin{example}[Ordens de grandeza]\label{ex:g11:nucleus-radioactivity:orders}
Tudo é ligeiramente radioativo. Uma banana: cerca de \qty{15}{Bq}
(potássio-40). Um corpo humano: cerca de \qty{8000}{Bq} --- o gancho
deste capítulo, o equivalente a umas quinhentas bananas. Um \omterm{def:g10:orders-of-magnitude:unit}{metro} cúbico
de granito: em torno de \qty{3e6}{Bq} (urânio e a cadeia dele, de onde
vem o radônio dos porões). Uma injeção de tecnécio: cerca de
\qty{5e8}{Bq}, que desaparecem em poucos dias.
\end{example}

\section{Serventias e perigos}

\begin{example}[Três carreiras de um núcleo instável]\label{ex:g11:nucleus-radioactivity:uses}
\emph{Medicina}: o tecnécio-99m, emissor $\gamma$ puro com
$T_{1/2} = 6$ horas, é fixado a uma \omterm{def:g11:fundamental-interactions:ladder}{molécula} que o órgão-alvo absorve;
uma câmera filma os $\gamma$ de fora, e no dia seguinte o marcador já
praticamente sumiu --- enquanto a radioterapia inverte a lógica,
concentrando feixes para destruir um tumor. \emph{Datação}: o carbono-14
para madeira, osso e tecido até uns \num{50000} anos; o urânio-238 para
rochas --- e para a Terra, com \num{4.5e9} anos. \emph{Detectores de
fumaça}: uma pontinha de amerício-241 ioniza o ar de uma câmara pequena,
deixando circular uma corrente minúscula; a fumaça sufoca a corrente e o
alarme dispara --- e as partículas $\alpha$ morrem em centímetros de ar.
\end{example}

\begin{remark}[Dose e o sievert]\label{rem:g11:nucleus-radioactivity:dose}
A radiação arranca elétrons de \omterm{def:g11:fundamental-interactions:ladder}{moléculas} e pode danificar o DNA. O dano
biológico é acompanhado pela \emph{dose}\index{dose}, em
\emph{sieverts}\index{sievert} (\unit{Sv}), que pondera a \omterm{def:g10:energy-conservation:energy}{energia}
depositada por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de tecido pelo quanto cada radiação é danosa.
Três regras: distância, blindagem ($\alpha$: a pele basta --- a não ser
que o emissor seja \emph{inalado}, o crime do radônio; $\beta$: alumínio;
$\gamma$: chumbo) e tempo. A dosimetria propriamente dita é material de
graduação.
\end{remark}

\end{exercise}

\section{Problema: a múmia, o reator e o detector de fumaça}

\begin{problem}[{Problema de fim de semana --- três carreiras de um
núcleo instável: uma múmia datada por metades, a árvore genealógica do
urânio equilibrada, um detector de fumaça auditado e por que as estrelas
tinham de ser mortais}]
\label{pb:g11:nucleus-radioactivity:1}
Três \omterm{def:g11:fundamental-interactions:ladder}{núcleos} instáveis, três empregos: o carbono-14 carimba a data de
toda planta e todo animal morto, o urânio-238 aquece um planeta por
dentro e o amerício-241 vigia tetos. Dados:
$T_{1/2} = \num{5700}$ anos (C-14), \num{4.5e9} anos (U-238) e $432$ anos
(Am-241); o carbono vivo mostra $13.6$ decaimentos por minuto por grama;
\omterm{def:g11:nucleus-radioactivity:notation}{números atômicos}: N $7$, O $8$, Pb $82$, Bi $83$, Po $84$, Rn $86$, Ra
$88$, Th $90$, Pa $91$, U $92$, Np $93$, Am $95$.

\medskip
\noindent\textbf{Parte I --- A múmia.}
\begin{enumerate}
  \item Dê a composição (prótons, nêutrons) de ${}^{12}_{6}\mathrm{C}$ e
        de ${}^{14}_{6}\mathrm{C}$. Por que a química deles é idêntica num
        corpo?
  \item Por que a proporção de carbono-14 é constante num corpo vivo, e
        por que ela começa a cair na morte?
  \item Escreva a equação de decaimento do carbono-14. A radiação dele
        escaparia através das faixas de linho da múmia?
  \item Um grama de carbono do linho mostra $6.8$ decaimentos por minuto.
        Que fração da taxa viva é essa? Date a múmia.
  \item Uma segunda ``múmia'', joia de uma coleção particular, mostra
        $13.2$ decaimentos por minuto por grama. Veredito?
  \item Estime a taxa de uma amostra com dez meias-vidas de idade e
        explique por que a datação por carbono se apaga além de uns
        \num{57000} anos.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O reator sob o \omterm{def:g11:electric-gravitational-fields:field}{campo}.}
\begin{enumerate}[resume]
  \item Dê a composição de ${}^{238}_{92}\mathrm{U}$. Por que os \omterm{def:g11:fundamental-interactions:ladder}{núcleos}
        pesados precisam de um excedente tão grande de nêutrons?
  \item Escreva a equação do decaimento $\alpha$ dele.
  \item O núcleo-filho decai então por $\beta^-$, e o neto por $\beta^-$
        de novo. Escreva as duas equações. Que elemento reaparece?
  \item A cadeia termina em ${}^{206}_{82}\mathrm{Pb}$. Conte os passos
        $\alpha$ dela e depois os passos $\beta^-$.
  \item A Terra se formou há \num{4.5e9} anos. Que fração do urânio-238
        primordial dela resta hoje? E daqui a mais $5$ bilhões de anos?
  \item Em uma frase: o que esse calor de decaimento enterrado alimenta na
        superfície? (Pense em vulcões e em continentes à deriva.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O detector de fumaça.}
\begin{enumerate}[resume]
  \item Escreva a equação do decaimento $\alpha$ de
        ${}^{241}_{95}\mathrm{Am}$.
  \item A \omterm{def:g11:nucleus-radioactivity:activity}{atividade} dele é \qty{33}{kBq}: quantos decaimentos por
        segundo? E por dia?
  \item Explique o detector: o que as partículas $\alpha$ fazem com o ar
        da câmara, e o que a fumaça muda?
  \item Por que um emissor $\alpha$ é a escolha certa aqui --- e um
        emissor $\gamma$ de mesma \omterm{def:g11:nucleus-radioactivity:activity}{atividade} seria a pior de todas?
  \item Depois de uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida} ($432$ anos!) a \omterm{def:g11:nucleus-radioactivity:activity}{atividade} seria
        \qty{16.5}{kBq}; o manual pede a substituição depois de dez anos.
        A fonte é o elo fraco?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Por que as estrelas tinham de ser mortais.}
\begin{enumerate}[resume]
  \item Estrelas moribundas forjaram e espalharam todos os elementos mais
        pesados do que o hélio. Que fração do urânio-238 forjado há
        \num{9e9} anos sobrevive hoje? Por que a abundância que resta na
        Terra torna o planeta muito mais jovem do que as estrelas mais
        antigas?
  \item O calor interno radioativo da Terra move o vulcanismo e a
        tectônica de placas, e mantém o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} em agitação produzindo o
        escudo magnético do \cref{ch:g11:magnetic-fields}. Como seria um
        planeta feito só de \omterm{def:g11:fundamental-interactions:ladder}{átomos} estáveis?
  \item Em uma linha, como poesia e como contabilidade: onde os seus
        \omterm{def:g11:fundamental-interactions:ladder}{átomos} foram forjados, e o que ainda aquece o chão sob os seus
        pés?
\end{enumerate}
\end{problem}
