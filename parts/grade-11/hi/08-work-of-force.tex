\chapter{बल का कार्य}\label{ch:g11:work-of-force}

अलमारी को कमरे के उस पार धकेलिए तो आपने कार्य किया; पर दीवार को दोपहर भर
धकेलते रहिए और आपकी माँसपेशियाँ चाहे जो कहें, भौतिकी शून्य गिनेगी।
कार्य वह है जो कोई बल \emph{गति के अनुदिश} पहुँचाता है ---
यानी रास्ते में चलती ऊर्जा (\cref{ch:g10:energy-conservation}), जिसका भुगतान \omterm{def:g11:work-of-force:work}{जूल} में होता
है। यह अध्याय \cref{ch:g11:forces-and-motion} के बलों का दाम लगाता है --- तिरछे खिंचाव,
भार, \omterm{def:g10:inertia:inventory}{घर्षण} --- और भुगतान की दर भी नापता है: शक्ति।

\section{अचर बल का कार्य}

\begin{definition}[अचर बल का कार्य]\label{def:g11:work-of-force:work}
$A$ से $B$ तक सरल रेखाखंड पर चलती किसी वस्तु पर लगता अचर बल
$\vect F$ इतना \emph{कार्य}\index{कार्य (बल का)} करता है:
\[
W_{AB}(\vect F) = F \times AB \times \cos\theta ,
\]
जहाँ $F$ बल का परिमाण है (\unit{N} में), $AB$ विस्थापन की लंबाई
(\unit{m} में) और $\theta$ $\vect F$ तथा $\vect{AB}$ के बीच का कोण। कार्य
\emph{जूल}\index{जूल} (\unit{J}) में मापा जाता है: एक जूल वह कार्य है जो
एक \omterm{def:g10:inertia:force}{न्यूटन} का बल अपनी ही दिशा में एक \omterm{def:g10:orders-of-magnitude:unit}{मीटर} पर करता
है।
\end{definition}

\begin{remark}\label{rem:g11:work-of-force:scal}
$F \times AB \times \cos\theta$ ठीक वही अदिश गुणनफल $\vect F \cdot \vect{AB}$ है जो इसी वर्ष के गणित के खंड में आया।
कमाई केवल गति के \emph{अनुदिश} पड़ते घटक $F\cos\theta$ की होती है।
\end{remark}

\begin{example}[पट्टे पर सूटकेस]\label{ex:g11:work-of-force:suitcase}
कोई यात्री सूटकेस को टर्मिनल में \qty{300}{m} घसीटता है, और पट्टे को फ़र्श से
$\theta = \qty{50}{\degree}$ पर $F = \qty{40}{N}$ से खींचता है: $W = 40 \times 300 \times \cos\qty{50}{\degree} \approx
\qty{7.7}{kJ}$। चपटा खींचने पर ($\theta = 0$) वही
बल \qty{12}{kJ} पहुँचाता: खिंचाव का तिहाई हिस्सा कुछ नहीं उठाता।
\end{example}

\begin{definition}[प्रेरक, प्रतिरोधी और शून्य कार्य]\label{def:g11:work-of-force:sign}
$\cos\theta$ का चिह्न हर बल को तीन ढर्रों में बाँट देता है:
\begin{itemize}
  \item $\theta < \qty{90}{\degree}$: $W > 0$, कार्य \emph{प्रेरक}\index{प्रेरक कार्य} है
        --- बल गति में ऊर्जा डालता है;
  \item $\theta = \qty{90}{\degree}$: $W = 0$, बल चाहे जितना बड़ा हो, कार्य बिलकुल
        नहीं करता;
  \item $\theta > \qty{90}{\degree}$: $W < 0$, कार्य \emph{प्रतिरोधी}\index{प्रतिरोधी कार्य}
        है --- बल ऊर्जा खींच लेता है। शुद्ध उदाहरण $\qty{180}{\degree}$ पर
        लगा ब्रेक है: \qty{45}{m} पर \qty{800}{N} कार से $800 \times 45 \times (-1)
        = \qty{-36}{kJ}$ निकालकर गरम चक्रिकाओं
        में डाल देता है।
\end{itemize}
\end{definition}

\section{भार का कार्य}

\begin{proposition}[भार का कार्य]\label{prop:g11:work-of-force:weight}
जब द्रव्यमान $m$ की कोई वस्तु सरल रेखाखंडों से बने \emph{किसी भी} पथ पर
$A$ (ऊँचाई $z_A$) से $B$ (ऊँचाई $z_B$) तक जाती है, तो उसका
भार $\vect P$ यह कार्य करता है:
\[
W_{AB}(\vect P) = mg\,(z_A - z_B) = \pm\, mgh ,
\]
जहाँ $h = \abs{z_A - z_B}$ ऊँचाई की गिरावट है: नीचे जाने पर $+mgh$, ऊपर जाने पर $-mgh$ ---
यानी केवल गिरावट मायने रखती है, रास्ता नहीं।
\end{proposition}

\begin{proof}
एक सरल रेखाखंड $AB$ पर भार (परिमाण $mg$, सीधा नीचे) $\vect{AB}$ के
साथ $\theta$ का कोण बनाता है, और $AB\cos\theta$ ठीक विस्थापन का नीचे की ओर की
\omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} पर प्रक्षेप है: $z_A - z_B$। इसलिए $W = mg \times AB\cos\theta = mg\,(z_A - z_B)$। रेखाखंडों की शृंखला पर
कार्य जुड़ते जाते हैं और गिरावटें दूरबीन की तरह सिकुड़ जाती हैं:
$mg(z_A - z_C) + mg(z_C - z_B) = mg(z_A - z_B)$।
\end{proof}

\begin{proposition}[कोई भी पथ]\label{prop:g11:work-of-force:anypath}
वही सूत्र $A$ से $B$ तक के किसी भी वक्र पथ पर लागू होता है।
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:work-of-force:anypath}
बात मानने लायक़ है --- कोई भी वक्र कई छोटे रेखाखंडों की टूटी रेखा के जितना
पास चाहें उतना पास है --- पर ``जितना पास चाहें'' को ईमानदारी से कहने के लिए
कलन चाहिए, जो \cref{ch:g12:work-and-energy} में किया गया है।
\end{remark}

\begin{example}[पदयात्री]\label{ex:g11:work-of-force:hiker}
\qty{78}{kg} का पदयात्री (पिट्ठू समेत) किसी भी पगडंडी से \qty{450}{m} चढ़ता है:
$W(\vect P) = -78 \times 9.81 \times 450 \approx \qty{-344}{kJ}$, और उतरते समय ठीक $+\qty{344}{kJ}$, चाहे रास्ते में कितने ही चक्कर काटे हों।
\end{example}

\section{घर्षण का कार्य}

\begin{proposition}[सरकन-घर्षण का कार्य]\label{prop:g11:work-of-force:friction}
अचर परिमाण $f$ का घर्षण-बल, जो हर क्षण गति के विपरीत
रहता है, कुल लंबाई $L$ के पथ पर $W = -fL$ कार्य करता है: सदा
\omterm{def:g11:work-of-force:sign}{प्रतिरोधी}, और \emph{तय की गई पथ-लंबाई} के समानुपाती --- विस्थापन के
नहीं।
\end{proposition}

\begin{proof}
हर छोटे सीधे टुकड़े पर बल गति से $\qty{180}{\degree}$ पर है और $-f \times (\text{इसकी लंबाई})$ का योगदान
देता है; टुकड़े जुड़कर $-fL$ बनाते हैं। वक्र पथ को भी इसी तरह छोटे टुकड़ों
में काटा जाता है, और सीमा भार की तरह मान ली जाती है (\cref{rem:g11:work-of-force:anypath})।
\end{proof}

\begin{remark}[घर्षण बनाम भार: बड़ी दरार]\label{rem:g11:work-of-force:divide}
भार का कार्य पथ भूल जाता है; \omterm{def:g10:inertia:inventory}{घर्षण} का हर
\omterm{def:g10:orders-of-magnitude:unit}{मीटर} पर चुंगी है। आने-जाने के चक्कर में भार का शुद्ध हिसाब
शून्य रहता है --- चढ़ाई पर जो लेता है, उतार पर लौटा देता है --- जबकि
\omterm{def:g10:inertia:inventory}{घर्षण} हर टुकड़े पर $-fL$ वसूलता है और कुछ नहीं लौटाता:
भार का कार्य संचित करके वापस पाया जा सकता है (स्थितिज
ऊर्जा, \cref{ch:g11:mechanical-energy}), \omterm{def:g10:inertia:inventory}{घर्षण} का ऊष्मा बनकर खो जाता है।
\end{remark}

\begin{example}[जाना और लौटना]\label{ex:g11:work-of-force:roundtrip}
एक संदूक फ़र्श पर \qty{5.0}{m} सरकता है और \qty{5.0}{m} लौट आता है, $f = \qty{60}{N}$ के
विरुद्ध: संदूक वहीं ख़त्म होता है जहाँ से चला था, फिर भी \omterm{def:g10:inertia:inventory}{घर्षण}
$-60 \times 10 =
\qty{-600}{J}$ करता है। भार और \omterm{def:g10:refraction:angles}{अभिलंब} बल गति के लंबवत हैं: पूरे
सफ़र में शून्य।
\end{example}

\begin{remark}[और स्प्रिंग का क्या?]\label{rem:g11:work-of-force:spring}
स्प्रिंग जितनी अधिक खिंचती है उतना ही ज़ोर से खींचती है: उसका बल अचर
नहीं है और \cref{def:g11:work-of-force:work} लागू नहीं होता। उसका कार्य निकालने के लिए खिंचाव
को अत्यंत सूक्ष्म क़दमों में काटकर जोड़ना पड़ता है --- यानी कलन, जो \cref{ch:g12:work-and-energy}
में दिया गया है।
\end{remark}

\section{शक्ति}

\begin{definition}[शक्ति]\label{def:g11:work-of-force:power}
किसी बल की \emph{शक्ति}\index{शक्ति!यांत्रिक} वह दर है जिससे वह
कार्य करता है, $P = W/\Delta t$, और उसे \emph{वाट}\index{वाट} ($\unit{W} = \unit{J/s}$) में
मापते हैं --- वही वाट जो \cref{ch:g11:circuits-and-power} में था। इंजनों की तख़्तियों पर SI से पहले
का एक मात्रक आज भी टिका है: \emph{अश्वशक्ति}\index{अश्वशक्ति},
$\qty{1}{hp} = \qty{736}{W}$।
\end{definition}

\begin{proposition}[अचर वेग पर शक्ति]\label{prop:g11:work-of-force:pfv}
यदि वस्तु अचर चाल $v$ से चले, तो गति से $\theta$ कोण पर लगता अचर
बल $\vect F$ शक्ति $P = F v \cos\theta$ पहुँचाता है --- और गति के अनुदिश
लगते बल के लिए विशेष रूप से $P = Fv$।
\end{proposition}

\begin{proof}
समय $\Delta t$ में वस्तु $d = v\,\Delta t$ तय करती है, इसलिए $W = F \times v\,\Delta t \times \cos\theta$; $\Delta t$ से भाग
दीजिए।
\end{proof}

\begin{example}[क्रेन]\label{ex:g11:work-of-force:crane}
कोई क्रेन \qty{600}{kg} का फट्टा स्थिर \qty{0.90}{m/s} से ऊपर उठाती है: तार का
बल भार के बराबर है, $mg = \qty{5.89e3}{N}$, इसलिए
$P = Fv = 5890 \times 0.90 \approx \qty{5.3}{kW} \approx \qty{7.2}{hp}$
--- यानी इस्पात के डिब्बे में सात घोड़ों की जोड़ी।
\end{example}

\begin{method}[कम बल से चढ़ना: घुमावदार रास्ता]\label{met:g11:work-of-force:zigzag}
किसी बोझ को सीमित बल से $h$ ऊँचाई तक उठाने के लिए:
\begin{enumerate}
  \item बिल तय है: \cref{prop:g11:work-of-force:weight} के अनुसार ऊपर तक का हर रास्ता $mgh$
        कार्य माँगता है।
  \item पथ खींच दीजिए: लंबाई $L$ की ढलान पर ज़रूरी बल (अचर चाल,
        \omterm{def:g10:inertia:inventory}{घर्षण} छोड़कर) $F = mgh/L$ है --- लंबाई दुगुनी, बल
        आधा।
  \item अधिभार चुकाइए: \omterm{def:g10:inertia:inventory}{घर्षण} \omterm{def:g10:orders-of-magnitude:unit}{मीटर} के हिसाब से वसूलता है
        (\cref{prop:g11:work-of-force:friction}), और $fL$ जोड़ देता है।
\end{enumerate}
पहाड़ी सड़कें क़दम 2 के लिए घुमावदार होती हैं: वही ऊँचाई, कम बल,
वही कार्य।
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:1}
अचर बल $F = \qty{40}{N}$ सीधे \qty{50}{m} विस्थापन पर लगता है: $\theta = \qty{0}{\degree}$, $\qty{60}{\degree}$,
$\qty{90}{\degree}$ और $\qty{120}{\degree}$ के लिए उसका कार्य निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:2}
\qty{12}{kg} के औज़ार-बक्से को वैन से \qty{2.5}{m} नीचे उतारा जाता है, और बाद में फिर
ऊपर चढ़ाया जाता है। दोनों बार उसके भार का कार्य?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:3}
\omterm{def:g11:work-of-force:sign}{प्रेरक}, \omterm{def:g11:work-of-force:sign}{प्रतिरोधी} या शून्य? (क) गिरते सेब का भार;
(ख) ऊपर जाती गेंद का भार; (ग) सरकते डिब्बे पर \omterm{def:g10:refraction:angles}{अभिलंब} बल;
(घ) ब्रेक लगाती साइकिल पर \omterm{def:g10:inertia:inventory}{घर्षण}; (ङ) झूलते लोलक के गोलक पर डोरी का
\omterm{def:g10:inertia:inventory}{तनाव}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:4}
कोई क्रेन \qty{600}{kg} को अचर चाल से \qty{25}{m} ऊपर उठाती है। उठाने वाले
बल का और भार का कार्य? उनका योग क्या है, और
क्यों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:5}
एस्केलेटर की मोटर हर मिनट \qty{90}{kJ} कार्य करती है।
\omterm{def:g11:work-of-force:power}{वाट} में उसकी शक्ति? \omterm{def:g11:work-of-force:power}{अश्वशक्ति} में?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:6}
एक स्लेज को बर्फ़ से $\qty{30}{\degree}$ ऊपर \qty{120}{N} \omterm{def:g10:inertia:inventory}{तनाव} वाली रस्सी से
\qty{200}{m} खींचा जाता है; \omterm{def:g10:inertia:inventory}{घर्षण} \qty{45}{N} है। चारों बलों (रस्सी,
\omterm{def:g10:inertia:inventory}{घर्षण}, भार, \omterm{def:g10:refraction:angles}{अभिलंब}) में से हर एक का कार्य और कुल?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:7}
कोई साइकिल-सवार \qty{18}{N} के कुल कर्षण के विरुद्ध अचर \qty{25}{km/h} से चलता है। दी
गई शक्ति? एक घंटे में किया गया कार्य?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:8}
\qty{65}{kg} का स्कीबाज़ \qty{120}{m} ऊँचाई उतरता है, या तो \qty{800}{m} की खड़ी ढलान से या
\qty{1500}{m} के हलके रास्ते से; दोनों पर स्कियों पर \omterm{def:g10:inertia:inventory}{घर्षण} \qty{30}{N} है।
दोनों रास्तों पर भार का और \omterm{def:g10:inertia:inventory}{घर्षण} का कार्य? कौन-सा
कार्य रास्ते की परवाह करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:9}
\qty{20}{kg} का डिब्बा $\qty{25}{\degree}$ की ढलान पर \omterm{def:g10:inertia:inventory}{घर्षण} के \qty{30}{N} के विरुद्ध
अचर चाल से \qty{4.0}{m} ऊपर धकेला जाता है। भार और \omterm{def:g10:inertia:inventory}{घर्षण} के
कार्य निकालिए, फिर (अचर चाल से) ढलान के समांतर लगते धक्के का
कार्य और परिमाण।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:10}
कोई कार \qty{130}{km/h} से चलती है और उसका इंजन पहियों तक \qty{50}{kW} पहुँचाता है। वह
कुल कितने \omterm{def:g11:work-of-force:sign}{प्रतिरोधी} बल से लड़ रही है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:11}
नहर का एक घोड़ा नाव को \qty{1.0}{m/s} से खींचता है, और \qty{800}{N} \omterm{def:g10:inertia:inventory}{तनाव} वाली
रस्सी पगडंडी से $\qty{20}{\degree}$ का कोण बनाती है। घोड़े की शक्ति
\omterm{def:g11:work-of-force:power}{वाट} में और \omterm{def:g11:work-of-force:power}{अश्वशक्ति} में निकालिए। टिप्पणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:work-of-force:12}
\qty{1200}{kg} की कार दर्रे तक $h = \qty{300}{m}$ चढ़ती है, या तो $20\%$ ढाल वाले सीधे रास्ते
से या $6\%$ ढाल वाली घुमावदार सड़क से (ढाल $=
\sin\alpha$)। \omterm{def:g10:inertia:inventory}{घर्षण} छोड़कर,
अचर चाल पर, हर रास्ते के लिए लंबाई, ढलान के अनुदिश ज़रूरी बल और
उसका कार्य निकालिए। एक वाक्य में निष्कर्ष लिखिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:work-of-force:13}
एक चरखी \qty{250}{kg} खपरैल को मचान पर स्थिर \qty{0.40}{m/s} से \qty{15}{m} ऊपर उठाती है। तार
के बल का कार्य, लगा समय, यांत्रिक शक्ति (दो बार:
$W/\Delta t$ और $Fv$) और $70\%$ मोटर-दक्षता पर खींची जाती
\omterm{prop:g11:circuits-and-power:power}{विद्युत शक्ति} निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:work-of-force:14}
एक अलमारी को \omterm{def:g10:inertia:inventory}{घर्षण} के $f = \qty{140}{N}$ के विरुद्ध कमरे के एक कोने से सामने
वाले कोने तक सरकाया जाता है: या तो सीधे \qty{6.0}{m} के विकर्ण पर, या दीवारों के
सहारे \qty{3.6}{m} और फिर \qty{4.8}{m}। हर रास्ते पर \omterm{def:g10:inertia:inventory}{घर्षण} का कार्य,
और फिर विकर्ण पर आने-जाने के चक्कर में? उस चक्कर में भार क्या करता
है, और इससे दोनों बलों के बीच का कौन-सा गहरा अंतर उजागर होता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:work-of-force:15}
एक साइकिल-सवार (साइकिल समेत \qty{80}{kg}) दर्रा चढ़ता है: \qty{12}{km} सड़क पर \qty{600}{m}
ऊँचाई, अचर \qty{15}{km/h} पर, कर्षण और लोटनी प्रतिरोध के \qty{14}{N} के
विरुद्ध। गुरुत्व के विरुद्ध, \omterm{def:g10:inertia:inventory}{घर्षण} के विरुद्ध और कुल दिया गया
कार्य निकालिए; चढ़ाई की अवधि; और औसत शक्ति, \omterm{def:g11:work-of-force:power}{वाट}
तथा \omterm{def:g11:work-of-force:power}{अश्वशक्ति} में।
\end{exercise}

\section{समस्या: सामान ढोने वालों का दिन}

\begin{problem}[{सप्ताहांत समस्या --- भौतिकी जूल के हिसाब से भुगतान करती है:
सरकाने, ऊपर चढ़ाने और हेयरपिन मोड़ों से भरा एक दिन, और अंत में यह विनम्र
करती खोज कि ईमानदार मेहनत के बारे में एक केतली क्या सोचती है}]
\label{pb:g11:work-of-force:1}
सामान ढोने वालों की एक टोली गोदाम ख़ाली करती है, बोझ को तीसरी मंज़िल के फ़्लैट
तक चढ़ाती है, और फिर वैन को पहाड़ी की चोटी के गाँव तक ले जाती है। हर काम का
दाम \omterm{def:g11:work-of-force:work}{जूल} में लगेगा; और रसोई की केतली दिन भर का हिसाब जाँचेगी।
डिब्बे: \qty{30}{kg}।

\medskip
\noindent\textbf{भाग I --- गोदाम के फ़र्श पर।}
एक डिब्बे को क्षैतिज बल $F = \qty{110}{N}$ से $d = \qty{8.0}{m}$ धकेला जाता है; सरकन
\omterm{def:g10:inertia:inventory}{घर्षण} $f = \qty{95}{N}$ है।
\begin{enumerate}
  \item धक्के का कार्य निकालिए।
  \item \omterm{def:g10:inertia:inventory}{घर्षण}, भार और \omterm{def:g10:refraction:angles}{अभिलंब} बल का कार्य
        निकालिए (शून्यों का कारण दीजिए)।
  \item डिब्बे पर कुल कार्य और उसके चिह्न का अर्थ क्या है (डिब्बा
        \omterm{def:g10:relative-motion:frame}{विराम} से चला था)? किस धक्के पर चाल अचर बनी रहेगी (\cref{ch:g11:forces-and-motion})?
  \item दूर वाले कोने तक \qty{10.0}{m} के विकर्ण से पहुँचा जा सकता है, या दो
        दीवारों के सहारे (\qty{6.0}{m} और फिर \qty{8.0}{m})। हर रास्ते पर
        \omterm{def:g10:inertia:inventory}{घर्षण} का कार्य?
  \item विकर्ण पर आने-जाने के चक्कर में \omterm{def:g10:inertia:inventory}{घर्षण} का कार्य और
        भार का कार्य निकालिए। कौन-सा बल लौटाता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तीसरी मंज़िल तक ($h = \qty{9.0}{m}$)।}
\begin{enumerate}[resume]
  \item फ़्लैट तक उठाए गए डिब्बे पर भार का कार्य निकालिए।
        टोली को कम से कम कितना कार्य देना ही होगा?
  \item \qty{70}{kg} का मज़दूर उसे सीढ़ियों से ऊपर ले जाता है और गुरुत्व मज़दूर
        \emph{और} डिब्बा, दोनों का दाम वसूलता है: गुरुत्व के विरुद्ध कुल
        कार्य, और उसका कितना अंश सचमुच डिब्बे पर ख़र्च हुआ?
  \item उसी \qty{9.0}{m} चढ़ाई के लिए ठेली की ढलान \qty{30}{m} लंबी है, और उस पर
        लोटनी \omterm{def:g10:inertia:inventory}{घर्षण} \qty{40}{N} है। अचर चाल पर ढलान के अनुदिश ज़रूरी
        धक्का और उसका कार्य निकालिए।
  \item ढलान की तुलना सीधे \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} उठाव से कीजिए: ज़रूरी बल, किया
        गया कार्य। ढलान से क्या मिलता है, और किस क़ीमत पर?
  \item रस्सी-घिरनी वाला उठाव डिब्बे को \qty{0.50}{m/s} से उठाता है। अवधि और
        यांत्रिक शक्ति?
  \item उस शक्ति को $P = Fv$ में भी लिखिए। उठाव की मोटर
        $60\%$ दक्ष है: खींची गई \omterm{prop:g11:circuits-and-power:power}{विद्युत शक्ति}?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- घुमावदार सड़क।}
लदी वैन (\qty{3500}{kg}) को घाटी से \qty{240}{m} ऊपर बसे गाँव तक पहुँचना है: या तो सीधी
पगडंडी, $24\%$ ढाल पर \qty{1.0}{km}, या पक्की सड़क, $3.0\%$ ढाल पर \qty{8.0}{km} (ढाल
$= \sin\alpha$)।
\begin{enumerate}[resume]
  \item जाँचिए कि दोनों रास्ते वही \qty{240}{m} चढ़ते हैं।
  \item अचर चाल पर और \omterm{def:g10:inertia:inventory}{घर्षण} छोड़कर, हर रास्ते पर ज़रूरी चालक
        बल निकालिए।
  \item हर रास्ते पर चालक बल का कार्य? $mgh$ से तुलना
        कीजिए।
  \item दोनों पर लोटनी प्रतिरोध \qty{400}{N} है: हर रास्ते पर अतिरिक्त
        कार्य निकालिए। घुमावदार रास्ते का आराम किस दाम पर मिलता है?
  \item \qty{36}{km/h} पर हर रास्ते के लिए ज़रूरी इंजन-शक्ति निकालिए
        (\unit{kW} और \unit{hp} में)। सड़कें घुमावदार क्यों होती हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दिन भर का बहीखाता।}
\begin{enumerate}[resume]
  \item टोली $60$ डिब्बे फ़र्श पर सरकाती है (प्रश्न 1 वाला धक्का) और
        सभी $60$ को \qty{9.0}{m} ऊपर चढ़ाती है (प्रश्न 6)। डिब्बों को दिया गया
        कुल यांत्रिक कार्य?
  \item सीढ़ी चढ़ने वाले का अपना शरीर भी जोड़िए: \qty{70}{kg} पर $20$ चढ़ाइयाँ
        (उतरने पर लौटी ऊर्जा गरम घुटनों में जाती है, बहीखाते में नहीं)। दिन
        का कुल?
  \item टोली ने $8.0$ घंटे काम किया: औसत यांत्रिक शक्ति? उतनी ही
        ऊर्जा ख़र्च करने में \qty{2.0}{kW} की केतली को कितना समय लगेगा?
  \item समापन: दो वाक्यों में लिखिए कि \omterm{def:g11:work-of-force:work}{जूल} किसका दाम चुकाता है ---
        और किसका नहीं (दोपहर भर धकेली गई दीवार, निश्चल थामा गया डिब्बा) ---
        और सामान ढोने वाले बड़ी माँसपेशियों के बजाय चरखियाँ, ढलानें और
        घुमावदार सड़कें क्यों रखते हैं।
\end{enumerate}
\end{problem}
