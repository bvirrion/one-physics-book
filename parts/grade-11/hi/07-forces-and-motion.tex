\chapter{बल और गति}\label{ch:g11:forces-and-motion}

पाना घुमाकर कमरे के उस पार उछालिए: उसके सिरे बेतहाशा डोलते हैं, फिर भी उसके
भीतर का एक बिंदु बिलकुल साफ़ परवलय खींचता जाता है। \Cref{ch:g10:inertia} ने यह तय कर दिया था
कि बलों के संतुलित होने पर क्या होता है --- कुछ नहीं। यह
अध्याय पूछता है कि जब वे संतुलित न हों तब क्या होता है: बचा हुआ बल
\emph{वेग बदल देता है} --- पिंडों को तेज़ करता है, धीमा करता है, मोड़ देता
है --- और बताता है कि कितना, भले अभी सूत्रों में नहीं, अनुपातों में।

\section{द्रव्यमान केंद्र}

\begin{definition}[द्रव्यमान केंद्र]\label{def:g11:forces-and-motion:com}
किसी पिंड या निकाय का \emph{द्रव्यमान केंद्र}\index{द्रव्यमान केंद्र} उसका
संतुलन-बिंदु है --- उसके द्रव्यमान की औसत स्थिति: समांगी सममित पिंड में उसका
ज्यामितीय केंद्र; और दो बिंदु-द्रव्यमानों के लिए, जोड़ते रेखाखंड का वह बिंदु
जिसकी दूरियाँ द्रव्यमानों के \emph{व्युत्क्रम} अनुपात में हों, $\frac{d_1}{d_2} = \frac{m_2}{m_1}$ ---
यानी भारी वाले के पास।
\end{definition}

\begin{example}[बेढब डंबल]\label{ex:g11:forces-and-motion:dumbbell}
\qty{0.90}{m} लंबी हल्की छड़ के सिरों पर \qty{2.0}{kg} और \qty{1.0}{kg} की गेंदें लगी हैं।
\omterm{def:g11:forces-and-motion:com}{द्रव्यमान केंद्र} उसे व्युत्क्रम अनुपात $d_1/d_2 = 1.0/2.0$ में बाँटता है: भारी
गेंद से $\qty{0.30}{m}$, हल्की से $\qty{0.60}{m}$ --- छड़ को वहीं टिकाइए।
\end{example}

\begin{remark}[उछाला हुआ पाना]\label{rem:g11:forces-and-motion:wrench}
लुढ़कते-घूमते पाने का चलचित्र बनाइए: उसका हर बिंदु उलझा हुआ, फंदेदार वक्र
खींचता है --- सिवाय \omterm{def:g11:forces-and-motion:com}{द्रव्यमान केंद्र} के, जो वही साफ़ परवलय खींचता है
जो उछाली हुई गेंद खींचती। \omterm{def:g11:forces-and-motion:com}{द्रव्यमान केंद्र} की गति सरल है; बाक़ी सब
उसके \emph{चारों ओर} का घूमना है --- और इसीलिए यांत्रिकी कार, ग्रह या
जिमनास्ट को एक ही बिंदु मान सकती है।
\end{remark}

\section{परिणामी बल वेग बदल देता है}

\begin{definition}[परिणामी बल]\label{def:g11:forces-and-motion:netforce}
किसी पिंड पर लगता \emph{परिणामी बल}\index{परिणामी बल} उस पर लगते सभी
बलों का सदिश योग है, $\vect F = \vect F_1 + \vect F_2 + \dots$ (सिरे से पूँछ जोड़कर, जैसे गणित के खंड
में); और बल ठीक तभी संतुलित होते हैं जब $\vect F = \vect 0$।
\end{definition}

\begin{theorem}[परिणामी बल का असर --- अर्ध-मात्रात्मक]\label{thm:g11:forces-and-motion:secondlaw}
\index{गति का दूसरा नियम}
द्रव्यमान $m$ के पिंड पर छोटी अवधि $\Delta t$ तक लगता \omterm{def:g11:forces-and-motion:netforce}{परिणामी बल}
$\vect F$ उसके \omterm{def:g11:forces-and-motion:com}{द्रव्यमान केंद्र} के वेग को एक सदिश $\Delta \vect v$ जितना बदल देता
है, जिसकी \emph{दिशा} $\vect F$ की होती है और जिसका परिमाण $F$ तथा $\Delta t$ के
समानुपात में बढ़ता है और द्रव्यमान के समानुपात में घटता है:
\[
\Delta v \propto \frac{F \, \Delta t}{m} .
\]
\end{theorem}
\admitted

\begin{remark}
बल या उसकी अवधि दुगुनी कीजिए: बदलाव दुगुना; द्रव्यमान दुगुना कीजिए:
आधा। अगला वर्ष इसे ठीक समीकरण में कस देगा, और वर्ष 1 का खंड पूछेगा कि यह किन
निर्देश तंत्रों में सही उतरता है; इस वर्ष अनुपात ही काफ़ी हैं --- वे पहले ही
वायु-थैले, ब्रेक और कक्षाएँ चला रहे हैं।
\end{remark}

\begin{proposition}[तीन मानक स्थितियाँ]\label{prop:g11:forces-and-motion:cases}
मान लीजिए कोई पिंड \omterm{def:g11:forces-and-motion:netforce}{परिणामी बल} $\vect F$ के नीचे वेग $\vect v$ से चल रहा
है।
\begin{enumerate}
  \item $\vect F$ $\vect v$ के अनुदिश: वह सरल रेखा में \emph{तेज़} होता जाता है;
  \item $\vect F$ $\vect v$ के विपरीत: वह सरल रेखा में \emph{धीमा} पड़ता जाता है;
  \item $\vect F$ $\vect v$ के लंबवत: वह चाल बदले बिना \emph{मुड़} जाता है।
\end{enumerate}
\end{proposition}

\begin{proof}
$\vect F$ के अनुदिश (\cref{thm:g11:forces-and-motion:secondlaw}) दिशा वाले $\Delta\vect v$ को $\vect v$ पर सिरे से पूँछ जोड़ते
जाइए: $\vect v$ के अनुदिश तीर लंबा होता जाता है; विपरीत होने पर वे आंशिक रूप से
कट जाते हैं (छोटा); और लंबवत होने पर हर छोटा $\Delta \vect v$ तीर को लंबा किए बिना
झुका देता है --- केवल दिशा मुड़ती है।
\end{proof}

\begin{example}[बग़ल से एक ठोकर]\label{ex:g11:forces-and-motion:kick}
कोई चक्रिका \qty{6.0}{m/s} से पूरब की ओर सरकती है; बग़ल से लगी एक छोटी चोट ठीक उत्तर
की ओर $\Delta\vect v = \qty{8.0}{m/s}$ जोड़ देती है। नया वेग सदिश योग है: परिमाण $\sqrt{6.0^2 + 8.0^2} = \qty{10.0}{m/s}$, दिशा $\tan\theta = 8.0/6.0$,
यानी पूरब से $\theta \approx 53^\circ$ उत्तर की ओर --- पुराना वेग मिटता नहीं, बदलाव उसमें जुड़
जाता है।
\end{example}

\section{बदलावों की तुलना: बल, समय, द्रव्यमान}

\begin{method}[अनुपात से सोचना]\label{met:g11:forces-and-motion:ratio}
दो स्थितियों की तुलना के लिए समीकरण कभी मत हल कीजिए --- अनुपात बनाइए:
वेग-बदलाव का अनुपात $F$ के अनुपात गुणा $\Delta t$ के अनुपात, भाग $m$ का
अनुपात है। फिर पढ़ लीजिए कि कौन-सा गुणक कितना बदला।
\end{method}

\begin{example}[लदी हुई ठेली]\label{ex:g11:forces-and-motion:cart}
वही धक्का, उतने ही एक \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} तक, ख़ाली \qty{20}{kg} की ठेली को \qty{1.2}{m/s} से रवाना कर
देता है। \qty{60}{kg} तक लादने पर --- यानी तिगुना द्रव्यमान, वही $F\Delta t$ --- वह
उसका तिहाई, \qty{0.40}{m/s}, लेकर चलती है। द्रव्यमान पदार्थ की ज़िद है।
\end{example}

\begin{remark}[टक्करें, वायु-थैले, मुड़े हुए घुटने]\label{rem:g11:forces-and-motion:airbag}
टक्कर में $\Delta v$ और $m$ पर कोई मोल-भाव नहीं होता --- सवार की चाल को शून्य
तक पहुँचना ही है --- इसलिए $F\,\Delta t$ तय है। बची हुई एकमात्र छूट \emph{समय} की
है: $\Delta t$ खींच दीजिए और बल उसी अनुपात में सिकुड़ जाएगा। जो सिर
स्टीयरिंग पर \qty{0.01}{s} में रुकने के बजाय वायु-थैले पर \qty{0.08}{s} में रुकता है, वह
आठ गुना कम औसत बल झेलता है; कुचल जाने वाले हिस्से, हेलमेट का झाग और
मुड़े हुए घुटने यही चाल चलते हैं।
\end{remark}

\section{घर्षण: सरकने से लड़ता बल}

\begin{definition}[स्थैतिक और गतिज घर्षण]\label{def:g11:forces-and-motion:friction}
छूती हुई सतहों के बीच \omterm{def:g10:inertia:inventory}{घर्षण} उनके \emph{सापेक्ष सरकने} का विरोध करता
है। \emph{स्थैतिक घर्षण}\index{घर्षण!स्थैतिक} तब तक लगता है जब तक वे सरकती
नहीं, और जो कुछ सरकाना शुरू करना चाहे उसे काटने के लिए वह एक अधिकतम मान तक
अपने को बढ़ाता जाता है; और सरकना शुरू हो जाने पर \emph{गतिज
घर्षण}\index{घर्षण!गतिज} लगता है, जो मोटे तौर पर अचर रहता है, सरकने के
विरुद्ध होता है और आम तौर पर स्थैतिक अधिकतम से \emph{कमज़ोर} होता है।
\end{definition}

\begin{example}[अलमारी को धकेलना]\label{ex:g11:forces-and-motion:wardrobe}
अलमारी को धीरे से धकेलिए: वह हिलती नहीं --- \omterm{def:g11:forces-and-motion:friction}{स्थैतिक घर्षण} आपके
\qty{150}{N} को काट देता है। और ज़ोर लगाइए: किसी \qty{400}{N} पर वह झटके से चल पड़ती है
और फिर कम मेहनत में सरकती जाती है --- कमज़ोर \omterm{def:g11:forces-and-motion:friction}{गतिज घर्षण} ने कमान
सँभाल ली है।
\end{example}

\begin{remark}[ABS ब्रेक काम क्यों करते हैं]\label{rem:g11:forces-and-motion:abs}
लुढ़कते पहिये का संपर्क-बिंदु सड़क पर सरकता नहीं: टायर \emph{स्थैतिक}
\omterm{def:g10:inertia:inventory}{घर्षण} से पकड़ता है। जकड़ा हुआ पहिया घिसटता है और उसे केवल कमज़ोर
\emph{गतिज} \omterm{def:g10:inertia:inventory}{घर्षण} मिलता है, जो पहिए चाहे जो करें, घिसटने के विरुद्ध
ही रहता है --- और दिशा बदलने की सुविधा मर जाती है। ABS पहिया जकड़ते ही उसी
क्षण ब्रेक ढीला कर देता है: अधिक पकड़, छोटा रुकाव, और चालक जो गाड़ी मोड़
सकता है।
\end{remark}

\section{मुड़ना: वृत्तीय गति के लिए भीतर की ओर बल चाहिए}

\begin{definition}[एकसमान वृत्तीय गति]\label{def:g11:forces-and-motion:ucm}
कोई पिंड \emph{एकसमान वृत्तीय गति}\index{एकसमान वृत्तीय गति} में तब है जब
उसका \omterm{def:g10:relative-motion:trajectory}{गतिपथ} वृत्त हो और वह उस पर अचर चाल से चले। वृत्त का स्पर्शी वेग
हर क्षण अपनी दिशा बदलता है --- इसलिए वह कभी अचर नहीं होता।
\end{definition}

\begin{proposition}[भीतर की ओर बल]\label{prop:g11:forces-and-motion:inward}
\omterm{def:g11:forces-and-motion:ucm}{एकसमान वृत्तीय गति} में चलते पिंड पर ऐसा \omterm{def:g11:forces-and-motion:netforce}{परिणामी बल} लगता है
जो कभी शून्य नहीं होता: हर क्षण वह पिंड से वृत्त के केंद्र की ओर संकेत करता
है।
\end{proposition}

\begin{proof}
चाल अचर है, इसलिए \omterm{def:g11:forces-and-motion:netforce}{परिणामी बल} का $\vect v$ के अनुदिश कोई हिस्सा नहीं
है (\cref{prop:g11:forces-and-motion:cases} की स्थितियाँ 1 और 2): वह स्पर्शी के लंबवत है, यानी त्रिज्या के
अनुदिश --- और भीतर की ओर, क्योंकि हर $\Delta \vect v$ पथ को केंद्र की ओर मोड़ता है।
\end{proof}

गेंद में: डोरी का \omterm{def:g10:inertia:inventory}{तनाव}। मोड़ पर कार में: सड़क का टायरों पर लगता
बग़ली \omterm{def:g11:forces-and-motion:friction}{स्थैतिक घर्षण} --- बर्फ़ पर वह ग़ायब, और कार सीधी निकल जाती
है। चंद्रमा में: पृथ्वी का गुरुत्वीय खिंचाव (\cref{ch:g10:universal-gravitation}), जो उसे \num{27.3} दिनों
में एक चक्कर घुमा देता है --- वह \emph{ऊपर} थामा नहीं गया, बग़ल से खींचकर
एक अनंत मोड़ में डाल दिया गया है, और पृथ्वी के चारों ओर गिरता रहता है; अगला
वर्ष इसी को उपग्रहों की यांत्रिकी बना देगा।
\end{example}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:5}

धक्के का विरोध करता संदूक; (ख) वही संदूक, जब वह सरकने लगे; (ग) थोड़ी झुकी
तश्तरी पर रखी किताब, जो सरक नहीं रही।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:6}

पैदा करता है। इनके लिए $\Delta v$ का पूर्वानुमान लगाइए: (क) $2F, \Delta t, m$;

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:7}

जोड़ देती है। नई चाल और दिशा निकालिए। चाल $8.0 + 6.0 = \qty{14}{m/s}$ क्यों नहीं है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:8}
गोलरक्षक वही शॉट दो तरह से रोकती है: बाँहें कड़ी रखकर (गेंद \qty{0.02}{s} में रुक
जाती है) या गेंद के साथ ``ढील देकर'' (\qty{0.08}{s})। औसत बलों की तुलना
कीजिए, और बताइए कि चुनाव से पहले कौन-सी राशियाँ तय हो चुकी थीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:9}
दौड़ शुरू करते समय धावक को कौन-सा बाहरी बल आगे धकेलता है? काँटेदार
जूते किस काम आते हैं --- और बिलकुल घर्षणहीन बर्फ़ पर कोई त्वरित होना तो दूर,
चल तक क्यों नहीं सकता?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:10}
आपात-स्थिति में रुकते समय ABS वाली कार (जिसके पहिए लुढ़कते रहते हैं) आम तौर
पर जकड़े, घिसटते पहियों वाली कार से पहले क्यों रुक जाती है --- और घिसटती कार
को मोड़ा क्यों नहीं जा सकता? \cref{def:g11:forces-and-motion:friction} का हवाला दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:11}

मोड़ता है? छोड़ने के बाद उसका पथ खींचिए। चंद्रमा को तार क्यों नहीं चाहिए, और
गुरुत्व के बिना उसका पथ क्या हो जाता?
\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:13}

  \item अब कार उसी बल के साथ \emph{दुगुनी} चाल से रुकती है: पहले

  \item \cref{prop:g11:forces-and-motion:inward} का उपयोग करके इस वाक्य को सुधारिए: ``चंद्रमा इसलिए नहीं

\section{समस्या: टक्कर की जाँच}

\begin{problem}[{सप्ताहांत समस्या --- कोमल टक्कर की डिज़ाइन: चूँकि टक्कर
वेग-बदलाव को तय कर देती है, इसलिए कार का हर सुरक्षा-उपकरण असल में समय खींचने
की मशीन है}]
\label{pb:g11:forces-and-motion:1}
टक्कर-जाँच की एक प्रयोगशाला कार को \qty{50}{km/h} पर एक कठोर दीवार से भिड़ा देती
है, और उसमें \qty{70}{kg} का वयस्क पुतला तथा \qty{20}{kg} का बच्चा-पुतला बैठे हैं;
कैमरे हर रुकाव का समय नापते हैं। आपका काम: खोजिए कि टक्कर की हिंसा रहती कहाँ
है, और उसे डिज़ाइन से मिटा दीजिए।

\medskip
\noindent\textbf{भाग I --- टक्कर की माप।}
\begin{enumerate}
  \item \qty{50}{km/h} को \unit{m/s} में बदलिए।
  \item सामने से होते किसी भी रुकाव में सवारों के वेग-बदलाव का परिमाण एक ही
        रहता है। कौन-सा? कोई उपकरण उसे बदल क्यों नहीं सकता?
  \item \cref{thm:g11:forces-and-motion:secondlaw} का हवाला दीजिए: $\Delta v$ और $m$ तय होने पर $F\,\Delta t$ भी तय
        है। तो डिज़ाइनर के हाथ में बची एकमात्र घुंडी?
  \item दीवार कार को \qty{0.050}{s} में रोक देती है; कुचल जाने वाला अगला हिस्सा इसे
        खींचकर \qty{0.150}{s} कर देता है। औसत बलों की तुलना कीजिए।
  \item रुकाव के दौरान $\Delta \vect v$ और पुतलों पर लगता \omterm{def:g11:forces-and-motion:netforce}{परिणामी बल} किस
        दिशा में हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- कुचल जाने वाला हिस्सा।}
\begin{enumerate}[resume]
  \item एक वाक्य में: अभियंता कार का अगला हिस्सा नष्ट हो जाने लायक़ क्यों
        बनाते हैं?
  \item पुराने ज़माने की कठोर कार में (रुकाव \qty{0.050}{s} में) पुतला सीट से कसकर
        बँधा है। उसके बल की तुलना आधुनिक \qty{0.150}{s} वाले रुकाव से
        कीजिए।
  \item रुकती कार की चाल एकसार घटती है (औसत: आरंभिक चाल की आधी)। वह \qty{0.150}{s}
        में कितनी दूर चलती है? असली कुचल-हिस्से से तुलना कीजिए।
  \item \qty{10}{m} का कुचल-हिस्सा क्यों नहीं? और कार का कौन-सा हिस्सा
        \emph{नहीं} कुचलना चाहिए?
  \item भाग II को एक नारे में समेटिए: ``$\Delta v$ आप चुन नहीं सकते, इसलिए आपको
        चुनना पड़ेगा \dots''।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- पेटी, वायु-थैला और बच्चा।}
\begin{enumerate}[resume]
  \item बिना पेटी बँधा पुतला अपना \qty{14}{m/s} (\cref{ch:g10:inertia}) तब तक बनाए रखता है जब तक
        सामने का काँच उसे \qty{0.005}{s} में न रोक दे; पेटी बँधा पुतला कार के साथ
        \qty{0.150}{s} में रुक जाता है। औसत बलों की तुलना कीजिए।
  \item थोड़ी खिंच जाने वाली पेटी न खिंचने वाले इस्पात के साज़ से बेहतर क्यों
        है?
  \item सिर स्टीयरिंग पर \qty{0.010}{s} में रुकता; वायु-थैला इसे खींचकर \qty{0.080}{s} कर
        देता है। बलों की तुलना कीजिए।
  \item वयस्क और बच्चा उसी $\Delta t$ में वही $\Delta v$ झेलते हैं। दोनों पेटियों
        को देने पड़ते परिणामी बलों की तुलना कीजिए।
  \item गोद में बैठा बच्चा: बाँहों को \qty{10}{kg} के शिशु का वेग \qty{0.150}{s} में
        \qty{14}{m/s} जितना बदलना होगा, जबकि गुरुत्व हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} वेग को \qty{9.8}{m/s}
        जितना बदलता है। बाँहों को कितने गुना तेज़ काम करना पड़ेगा --- और
        कितना द्रव्यमान थामना उतना ही भारी लगेगा? निष्कर्ष लिखिए।
  \item बच्चे कम बल झेलते हैं --- फिर बच्चों की सीट क्यों? सोचिए कि
        पेटी अपना बल \emph{कहाँ} लगाती है, और $\Delta t$ पर भी विचार
        कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सुरक्षा-पिंजरा, और फ़ैसला।}
\begin{enumerate}[resume]
  \item सवारी वाला कक्ष कठोर है जबकि दोनों सिरे कुचल जाते हैं। कठोरता का
        मतलब हिंसक रुकाव होने पर भी कक्ष का विरूपित न होना ज़रूरी क्यों है?
  \item दो एक जैसी कारें आमने-सामने भिड़ती हैं, हर एक \qty{50}{km/h} पर। हर एक कहाँ
        रुकती है? हर टक्कर की तुलना दीवार वाली जाँच से कीजिए।
  \item एक छोटी कार लदे ट्रक से आमने-सामने भिड़ती है। वे एक-दूसरे पर जो
        बल लगाते हैं वे बराबर और विपरीत हैं, हर अंतःक्रिया की तरह
        परस्पर (\cref{ch:g11:fundamental-interactions})। किसका वेग अधिक बदलता है?
  \item इस समस्या के सुरक्षा-उपकरण गिनाइए और वह एक राशि बताइए जिसे उनमें से
        हर एक खींचता है।
  \item समापन: कोमल टक्कर का डिज़ाइन-नियम केवल \emph{द्रव्यमान},
        \emph{वेग-बदलाव}, \emph{बल} और \emph{समय} का उपयोग करते हुए
        एक वाक्य में लिखिए।
\end{enumerate}
\end{problem}
