\chapter{नाभिक और रेडियोसक्रियता}\label{ch:g11:nucleus-radioactivity}

अभी इसी क्षण आपके शरीर के भीतर हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कोई आठ हज़ार परमाणु-नाभिक फट रहे हैं।
उन पर कोई हमला नहीं करता: कुछ नाभिक अस्थिर ही जन्म लेते हैं, और हर एक देर-सबेर
अपने को बदल डालता है, प्रकाश की चाल के अच्छे-ख़ासे अंश पर एक टुकड़ा बाहर उछालते
हुए। यह अध्याय \cref{ch:g11:fundamental-interactions} की $10^{-15}$-मीटर वाली मंज़िल पर लौटता है, यह देखने कि कौन-से
नाभिक कमज़ोर हैं, वे टूटते कैसे हैं, और उनकी नियमित बेसब्री गुफ़ाओं की उम्र और
छतों की रखवाली कैसे करती है।

\section{नाभिक, न्यूक्लियॉन और समस्थानिक}

\begin{definition}[नाभिकीय संकेतन]\label{def:g11:nucleus-radioactivity:notation}
किसी नाभिक में $Z$ प्रोटॉन (हर एक पर आवेश $+e$) और $N$ न्यूट्रॉन
(कोई आवेश नहीं) होते हैं, कुल $A = Z + N$ \omterm{def:g11:fundamental-interactions:ladder}{न्यूक्लियॉन}; उसे ${}^{A}_{Z}\mathrm{X}$ लिखते हैं,
जहाँ $\mathrm X$ रासायनिक संकेत है, $Z$ \emph{परमाणु क्रमांक}\index{परमाणु
क्रमांक} और $A$ \emph{द्रव्यमान संख्या}\index{द्रव्यमान संख्या}। $A$
और $Z$ से तय होने वाली जाति \emph{न्यूक्लाइड}\index{न्यूक्लाइड} कहलाती
है; और जिन न्यूक्लाइडों का $Z$ एक हो पर $A$ अलग, वे एक ही तत्व के
\emph{समस्थानिक}\index{समस्थानिक} हैं: रसायन वही (इलेक्ट्रॉन-बादल को केवल
$Ze$ दिखता है), नाभिक और स्थिरता अलग।
\end{definition}

\begin{example}[हाइड्रोजन और कार्बन]\label{ex:g11:nucleus-radioactivity:hc}
हाइड्रोजन के तीन \omterm{def:g11:nucleus-radioactivity:notation}{समस्थानिक} हैं: ${}^{1}_{1}\mathrm{H}$ (अकेला प्रोटॉन), ${}^{2}_{1}\mathrm{H}$
(ड्यूटीरियम: एक प्रोटॉन, एक न्यूट्रॉन), और अस्थिर ${}^{3}_{1}\mathrm{H}$ (ट्रिटियम: दो
न्यूट्रॉन)। प्राकृतिक कार्बन: ज़्यादातर ${}^{12}_{6}\mathrm{C}$ ($6$ प्रोटॉन, $6$
न्यूट्रॉन), $1.1\%$ ${}^{13}_{6}\mathrm{C}$, और $10^{12}$ में एक \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} अस्थिर ${}^{14}_{6}\mathrm{C}$
($8$ न्यूट्रॉन) का, जो \cref{ex:g11:nucleus-radioactivity:dating} का नायक है।
\end{example}

\section{स्थिर और अस्थिर नाभिक}

\begin{proposition}[स्थिरता की घाटी]\label{prop:g11:nucleus-radioactivity:valley}
स्थिर \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} $(Z, N)$ तल में एक सँकरी पट्टी घेरते हैं: हलके
नाभिकों के लिए $N \approx Z$, फिर बढ़ता हुआ न्यूट्रॉन-आधिक्य, $N \approx 1.5\,Z$ तक। पट्टी से
हटा हर \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} --- चाहे न्यूट्रॉन बहुत अधिक हों, चाहे प्रोटॉन,
चाहे वह बहुत बड़ा हो (सीसे, $Z = 82$, से आगे कोई \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} सचमुच स्थिर
नहीं) --- देर-सबेर बदल जाता है।
\end{proposition}
\admitted

\begin{remark}
पट्टी की ठीक-ठीक जगह बताना क्वांटम यांत्रिकी का काम है --- यानी विश्वविद्यालय
के खंडों का। पर उसका ढर्रा \cref{ch:g11:fundamental-interactions} का बहीखाता है: प्रबल गोंद केवल पड़ोसियों को
बाँधती है, \omterm{def:g11:fundamental-interactions:charge}{कूलॉम} प्रतिकर्षण पूरे नाभिक पर फैला रहता है, इसलिए भारी
नाभिकों को अतिरिक्त न्यूट्रॉन-गोंद चाहिए; फिर भी न्यूट्रॉनों का आधिक्य भी
अस्थिर है (\omterm{def:g11:fundamental-interactions:weak}{दुर्बल अंतःक्रिया} इसका ध्यान रखती है)।
\end{remark}

\begin{definition}[रेडियोसक्रियता]\label{def:g11:nucleus-radioactivity:radioactivity}
\emph{रेडियोसक्रियता}\index{रेडियोसक्रियता} किसी अस्थिर नाभिक का
स्वतःस्फूर्त रूप से दूसरे \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} में बदल जाना है, जिसके साथ
विकिरण भी निकलता है। यह \emph{यादृच्छिक} है: कौन-सा नाभिक अगली बार
\omterm{def:g11:mechanical-energy:dissipation}{क्षय} करेगा, इसकी कोई घोषणा नहीं होती, और कोई दाब, ताप या रसायन उसे न
जल्दी करा सकता है न टाल सकता है; पूर्वानुमेय केवल बड़ी आबादियाँ हैं (\cref{def:g11:nucleus-radioactivity:halflife})।
\end{definition}

\begin{remark}[बेक्रेल और क्यूरी दंपती]\label{rem:g11:nucleus-radioactivity:curie}
सन 1896 में आँरी \omterm{def:g11:nucleus-radioactivity:activity}{बेक्रेल} ने पाया कि यूरेनियम के लवण बंद दराज़ में, काले काग़ज़
के पार, बिना कोई ऊर्जा दिए फ़ोटो-प्लेट को धुँधला कर देते हैं। मारी
और पिएर क्यूरी ने दिखाया कि यह असर \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} का है, उसे नापा, और पोलोनियम तथा
रेडियम अलग किए, जो दस लाख गुना अधिक सक्रिय हैं। \emph{\omterm{def:g11:nucleus-radioactivity:radioactivity}{रेडियोसक्रियता}}
शब्द मारी क्यूरी का दिया हुआ है; और दो नोबेल पुरस्कार भी उन्हीं के।
\end{remark}

\section{अल्फ़ा, बीटा, गामा}

\begin{definition}[तीन ऐतिहासिक विकिरण]\label{def:g11:nucleus-radioactivity:abg}
तीन विकिरण, जिनके नाम तब पड़े जब कोई नहीं जानता था कि वे हैं क्या:
\begin{itemize}
  \item \emph{अल्फ़ा}\index{अल्फ़ा विकिरण} ($\alpha$): पूरा-का-पूरा उछाला गया
        हीलियम का नाभिक ${}^{4}_{2}\mathrm{He}$ --- भारी, दुहरा आवेशित, और काग़ज़ की
        एक परत या कुछ सेंटीमीटर हवा से ही रुक जाने वाला;
  \item \emph{बीटा}\index{बीटा विकिरण} ($\beta^-$): नाभिक में बना और
        बाहर उछाला गया इलेक्ट्रॉन ${}^{\;0}_{-1}\mathrm{e}$ --- हलका, तेज़, कुछ मिलीमीटर
        ऐलुमिनियम से रुक जाने वाला;
  \item \emph{गामा}\index{गामा विकिरण} ($\gamma$): विद्युत्-चुंबकीय विकिरण,
        प्रकाश से कहीं अधिक ऊर्जावान --- कभी पूरी तरह रुकता नहीं, केवल
        \emph{क्षीण}\index{क्षीणन} होता है: सीसे का एक सेंटीमीटर उसका आधा
        सोख लेता है।
\end{itemize}
\end{definition}

\begin{proposition}[नाभिकीय समीकरणों में संरक्षण]\label{prop:g11:nucleus-radioactivity:soddy}
हर नाभिकीय रूपांतरण में कुल \omterm{def:g11:nucleus-radioactivity:notation}{द्रव्यमान संख्या} $A$ और कुल आवेश ---
यानी नीचे लिखे सूचकांकों का योग, जिसमें उछले इलेक्ट्रॉन के लिए $-1$ गिना
जाता है --- पहले और बाद में एक ही रहते हैं।
\end{proposition}
\admitted

\begin{remark}
आवेश का संरक्षण \cref{ch:g11:fundamental-interactions} में पहले ही राज कर चुका था; और $A$ का संरक्षण कहता
है कि \omterm{def:g11:fundamental-interactions:ladder}{न्यूक्लियॉन} केवल बदले जाते हैं, न बनते हैं न नष्ट होते हैं ---
इसका ईमानदार बहीखाता विश्वविद्यालय के खंडों में है।
\end{remark}

\begin{method}[क्षय-समीकरण संतुलित करना]\label{met:g11:nucleus-radioactivity:balance}
\begin{enumerate}
  \item हर पात्र को उसके दोनों सूचकांकों के साथ लिखिए: $\alpha = {}^{4}_{2}\mathrm{He}$, $\beta^- = {}^{\;0}_{-1}\mathrm{e}$,
        $\beta^+ = {}^{0}_{+1}\mathrm{e}$; और $\gamma$ पर $A = Z = 0$ होता है।
  \item पहले $A$ के योग बराबर कीजिए, फिर $Z$ के (\cref{prop:g11:nucleus-radioactivity:soddy}); हल कीजिए,
        और आवर्त सारणी में उसके $Z$ से पुत्री तत्व पढ़ लीजिए।
\end{enumerate}
\end{method}

\begin{example}[एक अल्फ़ा क्षय]\label{ex:g11:nucleus-radioactivity:alpha}
रेडियम-226, यानी क्यूरी दंपती का रेडियम, $\alpha$ उत्सर्जक है:
${}^{226}_{88}\mathrm{Ra} \to {}^{A}_{Z}\mathrm{Y} + {}^{4}_{2}\mathrm{He}$
बल $A = 226 - 4 = 222$ और $Z = 88 - 2 = 86$; तत्व $86$ रेडॉन है, इसलिए
${}^{226}_{88}\mathrm{Ra} \to {}^{222}_{86}\mathrm{Rn} + {}^{4}_{2}\mathrm{He}$
--- यानी बिना हवादारी वाले तहख़ानों की वह रेडियोसक्रिय गैस।
\end{example}

\begin{example}[एक बीटा-ऋण क्षय]\label{ex:g11:nucleus-radioactivity:beta}
कार्बन-14 पट्टी के ऊपर बैठता है ($6$ प्रोटॉनों के मुक़ाबले $8$
न्यूट्रॉन):
${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} + {}^{\;0}_{-1}\mathrm{e}$।
जाँच: $14 = 14 + 0$, $6 = 7 - 1$; एक न्यूट्रॉन प्रोटॉन और उछले इलेक्ट्रॉन में बदल गया
--- यानी \omterm{def:g11:fundamental-interactions:weak}{दुर्बल अंतःक्रिया} कार्य पर।
\end{example}

\begin{remark}[बीटा-धन और गामा]\label{rem:g11:nucleus-radioactivity:betaplus}
पट्टी के नीचे उलटा \omterm{def:g11:mechanical-energy:dissipation}{क्षय} $\beta^+$ प्रोटॉन को न्यूट्रॉन और एक
\emph{पॉज़िट्रॉन}\index{पॉज़िट्रॉन} ${}^{0}_{+1}\mathrm{e}$ में बदल देता है, जो इलेक्ट्रॉन
का प्रतिकण है:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$,
और यही PET यंत्रों का घोड़ा है। और अधिकांश क्षयों के बाद काँपती हुई जन्मी
पुत्री ऊपर से एक $\gamma$ छोड़कर शांत हो जाती है।
\end{remark}

\section{अर्धायु और सक्रियता}

\begin{definition}[अर्धायु]\label{def:g11:nucleus-radioactivity:halflife}
किसी \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} की \emph{अर्धायु}\index{अर्धायु} $T_{1/2}$ वह समय है
जिसके बाद किसी भी बड़े प्रतिदर्श का आधा हिस्सा \omterm{def:g11:mechanical-energy:dissipation}{क्षय} कर चुका होता है। हर आगे
की अर्धायु बचे हुए को फिर आधा कर देती है: $N_0$ नाभिकों में से $T_{1/2}$ पर
$N_0/2$ बचते हैं, $2\,T_{1/2}$ पर $N_0/4$, $3\,T_{1/2}$ पर $N_0/8$, और $n$
अर्धायुओं के बाद $N_0/2^{\,n}$।
\end{definition}

\begin{remark}
एक नाभिक के लिए यादृच्छिकता, $10^{23}$ के लिए घड़ी की चाल: सिक्के
उछालने वालों की विशाल भीड़, जिसका हर सदस्य अलग-अलग अनिश्चित है पर हर दौर में
आधे बाहर हो जाते हैं। पूरी अर्धायुओं के \emph{बीच} में मान निकालने के लिए
चरघातांकी फलन चाहिए --- अगले वर्ष, गणित के खंड के साथ।
\end{remark}

\begin{example}[दस की बीस घातें]\label{ex:g11:nucleus-radioactivity:range}
\begin{center}\small\begin{tabular}{llll}

\begin{remark}[मात्रा और सीवर्ट]\label{rem:g11:nucleus-radioactivity:dose}

\omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} जमा हुई ऊर्जा को इस बात से तौलती है कि हर विकिरण
कितना हानिकारक है। तीन नियम: दूरी, ढाल ($\alpha$: त्वचा ही काफ़ी है ---
जब तक उत्सर्जक \emph{साँस के साथ भीतर} न चला जाए, जो रेडॉन का अपराध है;
$\beta$: ऐलुमिनियम; $\gamma$: सीसा), और समय। असली मात्रा-मापन विश्वविद्यालय की

\end{remark}

\section{समस्या: ममी, रिएक्टर और धुआँ-संसूचक}

\begin{problem}[{सप्ताहांत समस्या --- अस्थिर नाभिक के तीन पेशे: आधी होती
गिनतियों से आँकी गई एक ममी, यूरेनियम का संतुलित वंश-वृक्ष, एक धुआँ-संसूचक का
लेखा, और यह कि तारों का नश्वर होना क्यों ज़रूरी था}]
\label{pb:g11:nucleus-radioactivity:1}
तीन अस्थिर नाभिक, तीन काम: कार्बन-14 हर मरे हुए पौधे और जानवर पर तारीख़ की
मुहर लगाता है, यूरेनियम-238 एक ग्रह को भीतर से गरम रखता है, और अमेरिशियम-241
छतों पर पहरा देता है। आँकड़े: $T_{1/2} = \num{5700}$ वर्ष (कार्बन-14), \num{4.5e9} वर्ष (यूरेनियम-238),
$432$ वर्ष (अमेरिशियम-241); जीवित कार्बन प्रति ग्राम प्रति मिनट $13.6$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय}
दिखाता है; \omterm{def:g11:nucleus-radioactivity:notation}{परमाणु क्रमांक}: N $7$, O $8$, Pb $82$,
Bi $83$, Po $84$, Rn $86$, Ra $88$, Th $90$, Pa $91$,
U $92$, Np $93$, Am $95$।

\medskip
\noindent\textbf{भाग I --- ममी।}
\begin{enumerate}
  \item ${}^{12}_{6}\mathrm{C}$ और ${}^{14}_{6}\mathrm{C}$ की बनावट (प्रोटॉन, न्यूट्रॉन) बताइए। शरीर में
        दोनों का रसायन एक-सा क्यों है?
  \item जीवित शरीर में कार्बन-14 का अनुपात अचर क्यों रहता है, और मृत्यु पर
        वह गिरने क्यों लगता है?
  \item कार्बन-14 का क्षय-समीकरण लिखिए। क्या उसका विकिरण ममी की सूती
        पट्टियों के पार निकल आएगा?
  \item पट्टियों के एक ग्राम कार्बन में प्रति मिनट $6.8$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} मिलते हैं।
        यह जीवित दर का कितना अंश है? ममी की तिथि निकालिए।
  \item किसी निजी संग्रह की शान, एक दूसरी ``ममी'', प्रति ग्राम प्रति मिनट
        $13.2$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} दिखाती है। फ़ैसला?
  \item दस \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} पुराने प्रतिदर्श की दर का अनुमान लगाइए, और समझाइए कि
        मोटे तौर पर \num{57000} वर्ष से आगे कार्बन-तिथि क्यों धुँधला जाती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- घास के मैदान के नीचे का रिएक्टर।}
\begin{enumerate}[resume]
  \item ${}^{238}_{92}\mathrm{U}$ की बनावट बताइए। भारी नाभिकों को इतना न्यूट्रॉन-आधिक्य क्यों
        चाहिए?
  \item उसका $\alpha$ क्षय-समीकरण लिखिए।
  \item फिर पुत्री $\beta^-$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} करती है, और पौत्री दुबारा $\beta^-$। दोनों
        समीकरण लिखिए। कौन-सा तत्व फिर लौट आता है?
  \item शृंखला ${}^{206}_{82}\mathrm{Pb}$ पर ख़त्म होती है। उसके $\alpha$ क़दम गिनिए, फिर
        $\beta^-$ क़दम।
  \item पृथ्वी \num{4.5e9} वर्ष पहले बनी थी। उसके आदि यूरेनियम-238 का कितना अंश
        आज बचा है? और $5$ अरब वर्ष और बीतने पर?
  \item एक वाक्य में: यह दबी हुई क्षय-ऊष्मा सतह पर किसे शक्ति देती
        है? (ज्वालामुखियों और खिसकते महाद्वीपों के बारे में सोचिए।)
\end{enumerate}

\noindent\textbf{भाग III --- धुआँ-संसूचक।}
\begin{enumerate}[resume]
  \item ${}^{241}_{95}\mathrm{Am}$ का $\alpha$ क्षय-समीकरण लिखिए।
  \item उसकी \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} \qty{33}{kBq} है: प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कितने \omterm{def:g11:mechanical-energy:dissipation}{क्षय}? प्रतिदिन
        कितने?
  \item संसूचक को समझाइए: $\alpha$ कण कक्ष की हवा का क्या करते हैं, और धुआँ
        क्या बदल देता है?
  \item यहाँ $\alpha$ उत्सर्जक ही सही चुनाव क्यों है --- और उतनी ही
        \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} वाला $\gamma$ उत्सर्जक सबसे बुरा क्यों?
  \item एक \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} (यानी $432$ वर्ष!) के बाद \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता}

        कमज़ोर कड़ी स्रोत है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- तारों का नश्वर होना क्यों ज़रूरी था।}
\begin{enumerate}[resume]
  \item मरते तारों ने हीलियम से भारी हर तत्व गढ़ा और बिखेरा। \num{9e9} वर्ष
        पहले गढ़े गए यूरेनियम-238 का कितना अंश आज बचा है? पृथ्वी पर बची \omterm{rem:g11:nucleus-radioactivity:dose}{मात्रा}
        इस ग्रह को सबसे पुराने तारों से कहीं छोटा क्यों साबित करती है?
  \item पृथ्वी की रेडियोसक्रिय भीतरी ऊष्मा ज्वालामुखी और पट्ट-विवर्तनिकी
        चलाती है, और \omterm{def:g10:refraction:fiber}{क्रोड} को मथती रहती है, जिससे \cref{ch:g11:magnetic-fields} वाली चुंबकीय ढाल
        बनी रहती है। केवल स्थिर परमाणुओं वाला ग्रह कैसा होता?
  \item एक पंक्ति में, कविता की तरह भी और बहीखाते की तरह भी: आपके
        \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} कहाँ गढ़े गए थे, और आपके पैरों तले की ज़मीन को आज भी
        गरम कौन रखे है?
\end{enumerate}
\end{problem}
