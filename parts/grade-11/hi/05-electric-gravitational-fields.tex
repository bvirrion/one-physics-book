\chapter{विद्युत और गुरुत्वीय क्षेत्र}\label{ch:g11:electric-gravitational-fields}

पृथ्वी सेब को छुए बिना खींचती कैसे है? दूरी पर होती क्रिया को स्वयं न्यूटन ने
``इतनी बड़ी बेतुकी बात'' कहा था कि उसका बचाव करने से इनकार कर दिया। आधुनिक
उत्तर यह है: कोई द्रव्यमान या आवेश अपने चारों ओर की जगह को बदल देता है ---
वह एक \emph{क्षेत्र} रचता है --- और वहाँ जो कुछ रखा जाए, वह ठीक अपनी
जगह के क्षेत्र के अनुसार प्रतिक्रिया करता है। हम दोनों बड़े उदाहरण
साथ-साथ खड़े करेंगे, तब तक, जब तक यह सादृश्य रूप की पहचान न बन जाए।

\section{दूरी पर क्रिया से क्षेत्र तक}

\begin{definition}[क्षेत्र]\label{def:g11:electric-gravitational-fields:field}
\emph{क्षेत्र}\index{क्षेत्र} अवकाश के हर बिंदु को एक सदिश सौंप देता है। कोई
स्रोत (आवेश, द्रव्यमान) अपने चारों ओर हर जगह अपना क्षेत्र रचता है, चाहे उसे
महसूस करने वाला वहाँ कुछ हो या न हो; और फिर कोई परीक्षण वस्तु \emph{केवल अपनी
जगह के} क्षेत्र से तय हुआ बल महसूस करती है। दो क़दमों का यही चित्र
दूरी पर होती क्रिया की जगह ले लेता है।
\end{definition}

\begin{remark}
क्षेत्र कोई बहीखाते की चाल नहीं है: वह ऊर्जा ढोता है, और स्रोत
में हुए बदलाव उसी के भीतर से सीमित चाल पर बाहर की ओर चलते हैं --- प्रकाश
स्वयं ऐसा ही चलता हुआ क्षेत्र है (\cref{ch:g12:light-as-wave})। यहाँ हम ऐसे क्षेत्र
पढ़ेंगे जो समय के साथ नहीं बदलते।
\end{remark}

\section{विद्युत क्षेत्र}

\begin{definition}[विद्युत क्षेत्र]\label{def:g11:electric-gravitational-fields:efield}
यदि बिंदु $M$ पर रखा छोटा परीक्षण आवेश $q$ विद्युत बल $\vect F$
महसूस करता है, तो $M$ पर \emph{विद्युत क्षेत्र}\index{विद्युत क्षेत्र} है
\[
\vect E = \frac{\vect F}{q}
\qquad \text{(\unit{N/C} में, इसे \unit{V/m} भी लिखते हैं)}.
\]
$\vect E$ केवल स्रोतों और $M$ पर निर्भर करता है, परीक्षण आवेश पर नहीं:
$q$ दुगुना करने पर $\vect F$ भी दुगुना हो जाता है और भागफल वही रहता है।
\end{definition}

\begin{proposition}[आवेश पर बल]\label{prop:g11:electric-gravitational-fields:force}
जहाँ क्षेत्र $\vect E$ है, वहाँ रखा आवेश $q$ बल $\vect F = q \vect E$
महसूस करता है: $q > 0$ हो तो $\vect E$ के अनुदिश, $q < 0$ हो तो $\vect E$ के
विपरीत, और परिमाण $F = \abs{q}\,E$।
\end{proposition}

\begin{proof}
परिभाषा को फिर से लिखिए; चिह्न का नियम ऋणात्मक गुणज का बीजगणित भर है।
\end{proof}

\begin{example}[क्षेत्र पढ़ना]\label{ex:g11:electric-gravitational-fields:reading}
$M$ पर रखा परीक्षण आवेश $q = \qty{+2.0e-8}{C}$ पूरब की ओर संकेत करता \qty{6.0e-4}{N} का
बल महसूस करता है। $M$ पर क्षेत्र है $E = \num{6.0e-4} / \num{2.0e-8} = \qty{3.0e4}{N/C}$, पूरब की ओर;
उसी बिंदु पर रखा $q' = \qty{-1.0e-8}{C}$ $F = \num{1.0e-8} \times \num{3.0e4} = \qty{3.0e-4}{N}$ महसूस करता है --- पश्चिम की ओर।
\end{example}

\begin{proposition}[बिंदु आवेश का क्षेत्र]\label{prop:g11:electric-gravitational-fields:point}
बिंदु आवेश $Q$ दूरी $d$ पर आवेश और बिंदु को जोड़ती रेखा के अनुदिश एक
क्षेत्र रचता है --- $Q > 0$ हो तो $Q$ से बाहर की ओर, $Q < 0$ हो तो
उसकी ओर --- जिसका परिमाण है
\[
E = k\,\frac{\abs{Q}}{d^2},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{proposition}

\begin{proof}
\omterm{def:g11:fundamental-interactions:charge}{कूलॉम} का नियम (\cref{ch:g11:fundamental-interactions}) दूरी $d$ पर रखे परीक्षण आवेश $q$ पर लगता
बल देता है: परिमाण $k \abs{qQ} / d^2$, जोड़ती रेखा के अनुदिश, और समान चिह्नों के
लिए प्रतिकर्षी। $q$ से भाग दीजिए और \cref{prop:g11:electric-gravitational-fields:force} लगा दीजिए।
\end{proof}

\begin{example}[एक आवेशित गोला]\label{ex:g11:electric-gravitational-fields:sphere}
वान डे ग्राफ़ का गोला $Q = \qty{+5.0e-7}{C}$ रखता है (छोटा आवेशित गोला अपने केंद्र पर रखे
बिंदु आवेश जैसा बरतता है)। $d = \qty{0.50}{m}$ पर:
\[
E = \num{8.99e9} \times \num{5.0e-7} / 0.50^2 \approx \qty{1.8e4}{N/C},
\]
जो गोले से त्रिज्यीय दिशा में बाहर की ओर संकेत करता है। \qty{1.0}{m} पर वह गिरकर
इसका चौथाई रह जाता है: बल का $1/d^2$ क्षेत्र में भी बचा रहता
है।
\end{example}

\begin{definition}[क्षेत्र रेखाएँ]\label{def:g11:electric-gravitational-fields:lines}
\emph{क्षेत्र रेखा}\index{क्षेत्र रेखा} वह वक्र है जो हर जगह क्षेत्र
के स्पर्शी हो, और उस पर लगा तीर क्षेत्र की दिशा बताता है। तीन नियम:
\begin{itemize}
  \item वे धनात्मक आवेशों से निकलती हैं और ऋणात्मक आवेशों पर पहुँचती हैं;
  \item जहाँ रेखाएँ सटती हैं वहाँ क्षेत्र प्रबल है; जहाँ फैलती हैं,
        दुर्बल;
  \item दो रेखाएँ कभी नहीं कटतीं: हर बिंदु पर क्षेत्र की एक ही दिशा
        होती है।
\end{itemize}
\end{definition}

\section{संधारित्र का एकसमान क्षेत्र}

\begin{definition}[समांतर-पट्ट संधारित्र]\label{def:g11:electric-gravitational-fields:capacitor}
आमने-सामने रखी धातु की दो समांतर पट्टियाँ, जो पास-पास हों और विपरीत आवेश
रखती हों, \emph{संधारित्र}\index{संधारित्र} बनाती हैं। पट्टियों के बीच
क्षेत्र \emph{एकसमान}\index{एकसमान क्षेत्र} रहता है: हर जगह वही
परिमाण और वही दिशा --- पट्टियों के लंबवत, धनात्मक पट्टी से ऋणात्मक पट्टी की
ओर।
\end{definition}

\begin{proposition}[पट्टियों के बीच का क्षेत्र]\label{prop:g11:electric-gravitational-fields:uniform}
यदि पट्टियों के बीच का \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} (\cref{ch:g11:circuits-and-power}) $U$ हो और उनके बीच का
अंतराल $d$, तो उनके बीच क्षेत्र का परिमाण है
\[
E = \frac{U}{d} \qquad \text{(\unit{V/m} में)}.
\]
\end{proposition}
\admitted

\begin{remark}
एकसमानता और $E = U/d$ दोनों वर्ष 1 के खंड में ईमानदारी से निकाले गए हैं; सूत्र
से मात्रक \unit{V/m} समझ में आ जाता है। \omterm{def:g11:lenses-and-eye:lens}{पतले} अंतराल प्रबल
क्षेत्र बनाते हैं: एक मिलीमीटर पर \qty{100}{V} ही \qty{e5}{V/m} है।
\end{remark}

\begin{example}[इलेक्ट्रॉन-पुंज को मोड़ना]\label{ex:g11:electric-gravitational-fields:beam}
\omterm{def:g10:signals-and-waves:oscilloscope}{दोलनदर्शी} में पुंज ऐसी पट्टियों के बीच से गुज़रता है जिन पर $U = \qty{100}{V}$ और
$d = \qty{5.0}{mm}$ हैं: $E = 100 / \num{5.0e-3} = \qty{2.0e4}{V/m}$, और हर इलेक्ट्रॉन धनात्मक पट्टी की ओर $F = \num{1.60e-19} \times \num{2.0e4} = \qty{3.2e-15}{N}$ महसूस करता
है --- यानी \qty{8.9e-30}{N} वाले अपने भार का $\num{4e14}$ गुना, इसलिए पुंज साफ़
दिखता हुआ मुड़ जाता है (मुड़े हुए पथ का हिसाब \cref{ch:g12:newtons-laws} का काम है)।
\end{example}

\section{गुरुत्वीय क्षेत्र}

\begin{definition}[गुरुत्वीय क्षेत्र]\label{def:g11:electric-gravitational-fields:gfield}
यदि किसी बिंदु पर रखा परीक्षण द्रव्यमान $m$ \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण बल}
(यानी भार) $\vect P$ महसूस करता है, तो वहाँ \emph{गुरुत्वीय
क्षेत्र}\index{गुरुत्वीय क्षेत्र} है
\[
\vect g = \frac{\vect P}{m} \qquad \text{(\unit{N/kg} में)},
\]
और उलटे, $\vect P = m \vect g$। पिछले किसी अध्याय ने संख्या $g$ दी थी; $\vect g$ उसमें
वह दिशा जोड़ देता है जिधर छोड़ा हुआ पत्थर चलना शुरू करता है।
\end{definition}

\begin{proposition}[पृथ्वी का क्षेत्र]\label{prop:g11:electric-gravitational-fields:earth}
पृथ्वी के केंद्र (द्रव्यमान $M$) से दूरी $d$ पर \omterm{def:g11:electric-gravitational-fields:gfield}{गुरुत्वीय
क्षेत्र} केंद्र की ओर संकेत करता है और उसका परिमाण है
\[
g = G\,\frac{M}{d^2},
\qquad G = \qty{6.67e-11}{N.m^2/kg^2}.
\]
\end{proposition}

\begin{proof}
सार्वत्रिक \omterm{def:g11:fundamental-interactions:four}{गुरुत्वाकर्षण} (\cref{ch:g10:universal-gravitation}) परीक्षण द्रव्यमान $m$ पर
लगता बल देता है: परिमाण $G M m / d^2$, केंद्र की ओर; $m$ से भाग
दीजिए। सतह पर $d = R$ होता है और यही पिछले अध्याय का $g = GM/R^2 \approx \qty{9.81}{N/kg}$ है।
\end{proof}

\begin{example}[जहाँ चंद्रमा रहता है, वहाँ का क्षेत्र]\label{ex:g11:electric-gravitational-fields:moon}
चंद्रमा की दूरी $d = \qty{3.84e8}{m}$ पर:
\[
g = \num{6.67e-11} \times \num{5.97e24} / (\num{3.84e8})^2
\approx \qty{2.7e-3}{N/kg}.
\]
पृथ्वी का क्षेत्र कहीं ख़त्म नहीं होता --- वह बस पतला पड़ता
जाता है, $1/d^2$ के हिसाब से; यही क्षेत्र चंद्रमा को उसकी
कक्षा पर थामे रखता है।
\end{example}

\begin{remark}[सतह के पास एकसमान]\label{rem:g11:electric-gravitational-fields:uniformg}
प्रयोगशाला भर पर, बल्कि पहाड़ भर पर भी, $d$ $R = \qty{6.37e6}{m}$ की तुलना में
मुश्किल से बदलता है, इसलिए $\vect g$ बहुत अच्छी परिशुद्धता तक
\omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान} है: \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} रेखाएँ, अचर \qty{9.81}{N/kg} --- पृथ्वी का
क्षेत्र \omterm{def:g11:electric-gravitational-fields:capacitor}{संधारित्र} जैसा दिखने लगता है, ठीक जैसे गोल पृथ्वी
बग़ीचे से चपटी दिखती है।
\end{remark}

\end{tabular}
\end{center}

बाईं ओर का हर सूत्र $q \leftrightarrow m$, $\vect E \leftrightarrow \vect g$, $k \leftrightarrow G$ की शब्दावली रखते ही दाईं ओर
वाला बन जाता है।

\begin{remark}[सादृश्य कहाँ टूटता है]\label{rem:g11:electric-gravitational-fields:noanalogue}
आवेश दो चिह्नों में आता है; द्रव्यमान एक ही में। इसलिए \omterm{def:g11:electric-gravitational-fields:efield}{विद्युत
क्षेत्र} काटे जा सकते हैं: धातु के डिब्बे के आवेश फिर से जमकर उसके भीतर हर
बाहरी क्षेत्र को मिटा देते हैं (यही \emph{फैराडे पिंजरा} है ---
इसीलिए लिफ़्ट में आपके फ़ोन का संकेत मर जाता है), और थोक पदार्थ,
जिसमें $+$ और $-$ बराबर हैं, विद्युत् की दृष्टि से चुप रहता है।
गुरुत्व को न ढका जा सकता है न उदासीन किया जा सकता है: हर \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम}
अपना खिंचाव जोड़ देता है --- और इसीलिए $\num{e39}$ गुना कमज़ोर बल
(\cref{exo:g11:electric-gravitational-fields:10}) बड़े पैमानों पर ब्रह्मांड को चलाता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:1}
बिंदु $M$ पर रखा परीक्षण आवेश $q = \qty{+2.0e-6}{C}$ उत्तर की ओर संकेत करता \qty{5.0e-3}{N} का
बल महसूस करता है। $M$ पर क्षेत्र (परिमाण और दिशा)
बताइए, फिर $M$ पर रखे $q' = \qty{-4.0e-6}{C}$ पर लगता बल।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:2}
बिंदु आवेश $Q = \qty{+5.0}{nC}$ $O$ पर रखा है।
\begin{enumerate}
  \item $d = \qty{30}{cm}$ पर क्षेत्र निकालिए: परिमाण, दिशा।
  \item किस दूरी पर क्षेत्र आधा रह जाता है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:3}
किसी \omterm{def:g11:electric-gravitational-fields:capacitor}{संधारित्र} की \qty{3.0}{mm} दूर रखी पट्टियों पर $U = \qty{600}{V}$ है।
क्षेत्र निकालिए, फिर वहाँ एक इलेक्ट्रॉन पर लगता बल
($e = \qty{1.60e-19}{C}$)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:4}
मंगल पर \qty{250}{kg} के किसी यान का भार \qty{930}{N} है। मंगल की सतह का
क्षेत्र $g_{\text{मंगल}}$ निकालिए, फिर वहाँ \qty{80}{kg} के अंतरिक्ष यात्री का
भार।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:5}
सही या ग़लत, हर एक के लिए एक कारण के साथ: (क) जहाँ क्षेत्र प्रबल
हो, वहाँ दो \omterm{def:g11:electric-gravitational-fields:lines}{क्षेत्र रेखाएँ} कट सकती हैं; (ख) \omterm{def:g11:electric-gravitational-fields:lines}{क्षेत्र
रेखाएँ} ऋणात्मक आवेशों से निकलती हैं और धनात्मक आवेशों पर पहुँचती हैं;
(ग) सटी हुई \omterm{def:g11:electric-gravitational-fields:lines}{क्षेत्र रेखाएँ} प्रबल क्षेत्र का
संकेत हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:6}
अंतरराष्ट्रीय अंतरिक्ष स्थानक \qty{420}{km} की ऊँचाई पर कक्षा में घूमता
है ($R = \qty{6.37e6}{m}$, $M = \qty{5.97e24}{kg}$)।
\begin{enumerate}
  \item स्थानक पर $g$ निकालिए; यह सतह के मान का कितना अंश है?
  \item अंतरिक्ष यात्री तैरते हैं। एक वाक्य में अपने उत्तर से इसका मेल
        बिठाइए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:7}
किस परिमाण का क्षेत्र एक इलेक्ट्रॉन के भार ($m_e = \qty{9.11e-31}{kg}$) को
संतुलित कर देगा? \cref{ex:g11:electric-gravitational-fields:beam} के \qty{2.0e4}{V/m} से तुलना कीजिए; और इलेक्ट्रॉन की भौतिकी
में गुरुत्व की जगह पर निष्कर्ष लिखिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:8}
\qty{2.0e-6}{kg} द्रव्यमान का धूल का एक कण \qty{2.0}{cm} दूर रखी क्षैतिज पट्टियों के बीच
\omterm{def:g10:relative-motion:frame}{विराम} में तैरता है, और पट्टियों पर $U = \qty{5.0}{kV}$ है।
\begin{enumerate}
  \item क्षेत्र निकालिए, फिर कण के आवेश का परिमाण।
  \item कण का आवेश धनात्मक है: कौन-सी पट्टी धनात्मक है?
  \item कण कितने \omterm{def:g11:fundamental-interactions:elementary}{मूल आवेश} रखता है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:9}
आवेश $+Q$ और $+4Q$ \qty{30}{cm} की दूरी पर हैं। उन्हें जोड़ते रेखाखंड पर
किस जगह कुल क्षेत्र शून्य है? पहले समझाइए कि वह बिंदु दोनों आवेशों
के बीच ही क्यों होगा, और छोटे वाले के पास क्यों।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:10}
हाइड्रोजन के \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} में प्रोटॉन ($m_p = \qty{1.67e-27}{kg}$, आवेश $+e$) और
इलेक्ट्रॉन ($m_e = \qty{9.11e-31}{kg}$, आवेश $-e$) $d = \qty{5.3e-11}{m}$ की दूरी पर हैं। उनके बीच का
विद्युत बल और \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण बल} तथा उनका अनुपात निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:11}
बिंदु $M$ पर अकेला एक स्रोत पूरब की ओर \qty{300}{N/C} का क्षेत्र
बनाता; दूसरा अकेला, उत्तर की ओर \qty{400}{N/C} का। क्षेत्र सदिशों की तरह
जुड़ते हैं: $M$ पर कुल क्षेत्र (परिमाण और दिशा) बताइए, फिर वहाँ
रखे $q = \qty{-2.0e-6}{C}$ पर लगता बल।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:electric-gravitational-fields:12}
किस ऊँचाई पर पृथ्वी का क्षेत्र गिरकर सतह के मान का आधा रह जाता है?
और $h = R$ की ऊँचाई पर कितना अंश?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:electric-gravitational-fields:13}
$\qty{+2.0}{nC}$ के आवेश $A$ और $B$ पर रखे हैं, आपस में \qty{6.0}{cm} दूर।
$M$ $[AB]$ के लंब समद्विभाजक पर है, उसके मध्यबिंदु से \qty{4.0}{cm} दूर।
\begin{enumerate}
  \item दूरी $AM$ निकालिए, फिर $M$ पर हर आवेश द्वारा बने
        क्षेत्र का परिमाण।
  \item दोनों क्षेत्रों को सदिशों की तरह जोड़िए (सममिति का उपयोग
        कीजिए) और $M$ पर कुल क्षेत्र का परिमाण तथा दिशा बताइए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:electric-gravitational-fields:14}
आवेश के दो चिह्नों के सहारे समझाइए कि धातु के डिब्बे के मुक्त आवेश उसके भीतर
हर जगह बाहरी \omterm{def:g11:electric-gravitational-fields:efield}{विद्युत क्षेत्र} को कैसे काट देते हैं --- और
द्रव्यमानों की कोई सजावट गुरुत्व के लिए वही काम क्यों नहीं कर सकती।
गुरुत्व-ढाल के लिए कौन-सी असंभव सामग्री चाहिए होगी?
\end{exercise}

एक छोटी गेंद ($m = \qty{0.50}{g}$, $q = \qty{+2.0e-7}{C}$) धागे से लटकी है और क्षैतिज
\omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान क्षेत्र} $E = \qty{1.0e4}{V/m}$ में है। \omterm{def:g10:relative-motion:frame}{विराम} में धागा \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} से
$\theta$ का कोण बनाता है।
\begin{enumerate}
  \item तीनों बल गिनाइए; समझाइए कि $\tan\theta = qE / (mg)$ क्यों।
  \item $\theta$ निकालिए, फिर धागे का \omterm{def:g10:inertia:inventory}{तनाव}।
\end{enumerate}
\end{exercise}

\section{समस्या: मिलिकन की बूँद}

\begin{problem}[{सप्ताहांत समस्या --- इलेक्ट्रॉन का आवेश तौलना: दो पट्टियों
के बीच हवा में ठहरी तेल की एक बूँद दिखा देती है कि आवेश $\num{1.60e-19}$
\omterm{def:g11:fundamental-interactions:charge}{कूलॉम} के अविभाज्य क़दमों में आता है}]
\label{pb:g11:electric-gravitational-fields:1}
सन 1909 में रॉबर्ट मिलिकन ने किसी \omterm{def:g11:electric-gravitational-fields:capacitor}{संधारित्र} की क्षैतिज पट्टियों के
बीच तेल की बूँदें छिड़कीं। \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} को तब तक सुर में बिठाकर, जब तक
चुनी हुई बूँद निश्चल न लटक जाए --- यानी विद्युत बल ठीक
भार को काट दे --- वे उसका आवेश तौल सकते थे। हमारी बूँद का द्रव्यमान
$m = \qty{4.90e-15}{kg}$ है (क्षेत्र बंद रखकर उसके धीमे गिरने से नापा गया; हम उसे
दिया हुआ मान लेते हैं); पट्टियाँ $d = \qty{8.0}{mm}$ दूर हैं, \emph{ऊपर} वाली धनात्मक
है; और बूँद का आवेश ऋणात्मक है।

\medskip
\noindent\textbf{भाग I --- मंच।}
\begin{enumerate}
  \item पट्टियों के बीच के क्षेत्र का वर्णन कीजिए; ``\omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान}''
        होने से क्या मिल जाता है?
  \item $U = \qty{800}{V}$ के लिए $E$ निकालिए।
  \item बूँद पर लगता विद्युत बल किस दिशा में है? \emph{ऊपर} वाली
        पट्टी को ही धनात्मक क्यों होना पड़ा?
  \item बूँद का भार निकालिए।
  \item \omterm{def:g10:relative-motion:frame}{विराम} की शर्त को दो बल-परिमाणों की समता के रूप में लिखिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तराज़ू।}
\begin{enumerate}[resume]
  \item उसी से $\abs{q}$ को $m$, $g$, $d$ और $U$ के पदों में
        लिखिए।
  \item बूँद $U = \qty{800}{V}$ पर \omterm{def:g10:relative-motion:frame}{विराम} में लटकी रहती है: $\abs{q}$ निकालिए।
  \item $e = \qty{1.60e-19}{C}$ से भाग दीजिए। बूँद पर कितने अतिरिक्त इलेक्ट्रॉन हैं?
  \item \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} को थोड़ा बढ़ाकर \qty{900}{V} कर दिया जाता है। अब कुल
        बल किस दिशा में है?
  \item दिखाइए कि संतुलन का \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} $U = mgd/\abs{q}$ है: आवेश जितना बड़ा,
        \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} उतना छोटा।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- क़दम।}
विकिरण का एक झोंका बूँद पर इलेक्ट्रॉन चढ़ा या उतार सकता है; हर बदलाव के बाद
मिलिकन संतुलन के लिए \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} फिर से सुर में बिठाते हैं।
\begin{enumerate}[resume]
  \item एक झोंके के बाद संतुलन के लिए $U = \qty{1200}{V}$ चाहिए। नया $\abs{q}$ निकालिए।
        बूँद ने इलेक्ट्रॉन पाया या खोया?
  \item क्रमशः संतुलन $U = 1200$, $800$, $600$, $480$ और \qty{400}{V} पर मिलते
        हैं। पाँचों आवेश निकालिए।
  \item दिखाइए कि पाँचों एक ही राशि के पूर्णांक गुणज हैं; उसका मान दीजिए।
  \item लगातार दो संतुलनों के बीच आवेश कितना बदलता है? व्याख्या कीजिए।
  \item समझाइए कि चाहे जितने धीरज से सुर बिठाया जाए, यह बूँद $U = \qty{700}{V}$ पर
        \omterm{def:g10:relative-motion:frame}{विराम} में कभी क्यों नहीं लटक सकती।
  \item मिलिकन ने यही सैकड़ों बूँदों पर दोहराया: हर मापा गया आवेश उसी एक मान
        का पूर्णांक गुणज निकला। निष्कर्ष लिखिए, और मापी गई राशि का नाम
        बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- गुरुत्वीय मिलिकन क्यों नहीं होता।}
\begin{enumerate}[resume]
  \item इस अध्याय की शब्दावली में, भाग II के संतुलन में $q$, $E$ और
        $qE$ की भूमिकाएँ कौन निभाता है?
  \item क्या इसके बदले ऊपर रखा कोई द्रव्यमान बूँद को थामे रख सकता था? बूँद
        से \qty{1.0}{m} ऊपर रखे \qty{1000}{kg} के सीसे के गोले का क्षेत्र,
        बूँद पर लगता बल निकालिए और उसके भार से तुलना
        कीजिए।
  \item संतुलन के लिए ऊपर की ओर इतना खिंचाव चाहिए कि वह पूरी पृथ्वी से लड़
        सके। आवेश वाली पट्टी वह क्यों दे सकती है और द्रव्यमानों की कोई सजावट
        क्यों नहीं?
  \item बूँद का \emph{द्रव्यमान} भी क़दमों में बदलता है (तेल का एक
        \omterm{def:g11:fundamental-interactions:ladder}{अणु}: लगभग \qty{5e-25}{kg})। एक इलेक्ट्रॉन के आ बैठने पर $m$ और
        $\abs{q}$ में हुई सापेक्ष छलाँगों की तुलना कीजिए और निष्कर्ष निकालिए:
        यह तराज़ू बिजली के \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} तो देख लेता है पर तेल के
        \omterm{def:g11:fundamental-interactions:ladder}{अणु} क्यों नहीं? और अंतिम चोट: आवेश क्वांटित है, और उसका
        \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} $e = \qty{1.60e-19}{C}$ है।
\end{enumerate}
\end{problem}
