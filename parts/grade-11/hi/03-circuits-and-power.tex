\chapter{विद्युत परिपथ और शक्ति}\label{ch:g11:circuits-and-power}

स्विच दबाइए और लैंप जल उठता है: कई किलोमीटर दूर बैठा जनित्र आपकी दीवार में
गड़े ताँबे के फंदे पर आवेश को घुमा रहा है। पिछले वर्षों में परिपथ का नक्शा
बना था --- धारा, \omterm{def:g11:circuits-and-power:voltage}{विभवांतर}, \omterm{prop:g11:circuits-and-power:ohm}{ओम का नियम}। यह अध्याय उसी परिपथ को
\cref{ch:g10:energy-conservation} के बहीखाते के साथ फिर खोलता है: \omterm{def:g10:orders-of-magnitude:unit}{मीटर} किस चीज़ का पैसा लेता है,
\omterm{def:g11:lenses-and-eye:lens}{पतले} तार क्यों दहकते हैं, और ग्रिड \qty{400}{kV} पर क्यों चलता है।

\section{चलता हुआ आवेश}

\begin{definition}[धारा और कूलॉम]\label{def:g11:circuits-and-power:current}
\emph{विद्युत धारा}\index{विद्युत धारा} आवेश का व्यवस्थित बहाव है (धातु में,
इलेक्ट्रॉनों का)। उसकी तीव्रता $I$, \emph{ऐंपियर}\index{ऐंपियर} (\unit{A}) में,
वह आवेश है जो प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} तार के अनुप्रस्थ काट से गुज़रता है: स्थिर धारा
$I$ समय $t$ तक बहकर आवेश
\[ q = I\,t , \]
पहुँचाती है, जो \emph{कूलॉम}\index{कूलॉम} (\unit{C}) में मापा जाता है और
$\qty{1}{C} =
\qty{1}{A.s}$। मूल आवेश $e = \qty{1.6e-19}{C}$ है। धारा को ऐमीटर से मापते हैं, जिसे श्रेणीक्रम में
जोड़ा जाता है।
\end{definition}

\begin{example}[तड़ित बनाम फ़ोन]\label{ex:g11:circuits-and-power:coulomb}
तड़ित का एक झटका लगभग $\qty{3.0e4}{A}$ लगभग $\qty{1.0e-4}{s}$ तक ले जाता है: $q = 3.0\times10^{4} \times 10^{-4} = \qty{3.0}{C}$।
आपका फ़ोन \qty{2.0}{A} पर वही आवेश हर $1.5$ \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में सरका देता है।
\end{example}

\begin{definition}[विभवांतर और वोल्ट]\label{def:g11:circuits-and-power:voltage}
परिपथ के दो बिंदुओं के बीच का \emph{विभवांतर}\index{विभवांतर} $U$,
\emph{वोल्ट}\index{वोल्ट} (\unit{V}) में, उन बिंदुओं के बीच चलने वाले आवेश के
प्रति मात्रक विनिमय हुई ऊर्जा है: $\qty{1}{V} = \qty{1}{J/C}$, इसलिए विभवांतर
$U$ वाले उपकरण को पार करता आवेश $q$ उससे ऊर्जा $E = qU$ का विनिमय
करता है। समांतर क्रम में जुड़ा वोल्टमीटर इसे मापता है।
\end{definition}

\begin{proposition}[परिपथ के नियम]\label{prop:g11:circuits-and-power:laws}
स्थायी रूप से चलते परिपथ में:
\begin{enumerate}
  \item \emph{संधि नियम:} किसी संधि पर आती धाराओं का योग वहाँ से निकलती
        धाराओं के योग के बराबर होता है;
  \item \emph{श्रेणी नियम:} किसी पथ के साथ-साथ \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} जुड़ते जाते
        हैं: $U_{AC} = U_{AB} + U_{BC}$;
  \item \emph{समांतर नियम:} एक ही दो बिंदुओं को जोड़ती दो शाखाओं पर एक ही
        \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} होता है।
\end{enumerate}
\end{proposition}

\begin{proof}
स्थायी अवस्था में आवेश कहीं जमा नहीं होता, इसलिए जो संधि में आता है वही उससे
निकलता भी है। प्रति \omterm{def:g11:circuits-and-power:current}{कूलॉम} ऊर्जा पथ के साथ जुड़ती है और केवल
पथ के सिरों पर निर्भर करती है --- बीच में कौन-सी शाखा ली गई, इस पर नहीं।
\end{proof}

\section{ओम का नियम और प्रतिरोधकों के जाल}

\begin{definition}[प्रतिरोध]\label{def:g11:circuits-and-power:resistance}
किसी चालक का \emph{प्रतिरोध}\index{प्रतिरोध} उसके सिरों के \omterm{def:g11:circuits-and-power:voltage}{विभवांतर}
और उसमें बहती धारा का अनुपात $R = U/I$ है, जिसे \emph{ओम}\index{ओम} (\unit{\ohm}) में
मापते हैं: $\qty{1}{\ohm} = \qty{1}{V/A}$।
\end{definition}

\begin{proposition}[ओम का नियम]\label{prop:g11:circuits-and-power:ohm}
\index{ओम का नियम}
स्थिर ताप पर रखे धातु के चालक में $R$ अचर रहता है: \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} और
धारा समानुपाती होते हैं,
\[ U = R\,I . \]
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:circuits-and-power:ohm-limits}
यह प्रायोगिक नियम है, सार्वत्रिक नहीं: डायोड, बल्ब का तंतु या आपकी अपनी
त्वचा इसे नहीं मानते। धातुएँ इसे क्यों मानती हैं, यह विश्वविद्यालय के खंडों
में निकाला गया है।
\end{remark}

\begin{proposition}[श्रेणी और समांतर प्रतिरोधक]\label{prop:g11:circuits-and-power:networks}
\index{तुल्य प्रतिरोध}
श्रेणीक्रम में जुड़े दो प्रतिरोधक एक प्रतिरोधक $R_{\text{s}} =
R_1 + R_2$ के तुल्य हैं; समांतर
क्रम में जुड़े दो, एक प्रतिरोधक $R_{\text{p}}$ के, जो इस प्रकार मिलता है:
\[
\frac{1}{R_{\text{p}}} = \frac{1}{R_1} + \frac{1}{R_2} ,
\qquad\text{अर्थात्}\quad
R_{\text{p}} = \frac{R_1 R_2}{R_1 + R_2} < \min(R_1, R_2) .
\]
\end{proposition}

\begin{proof}
श्रेणीक्रम: वही $I$ दोनों से गुज़रती है और \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} जुड़ते हैं
(\cref{prop:g11:circuits-and-power:laws}), इसलिए $U = (R_1 + R_2)\,I$।
समांतर क्रम: दोनों एक ही $U$ के नीचे हैं और धाराएँ जुड़ती हैं, इसलिए
$I = \frac{U}{R_1} + \frac{U}{R_2}$।
\end{proof}

\begin{example}[तुल्य प्रतिरोध]\label{ex:g11:circuits-and-power:networks}
$R_1 = \qty{100}{\ohm}$ और $R_2 = \qty{150}{\ohm}$: श्रेणीक्रम में \qty{250}{\ohm}; समांतर क्रम में $\frac{100 \times 150}{250} =
\qty{60}{\ohm}$ --- दोनों में
से किसी से भी कम: धारा को दूसरा रास्ता मिल जाता है।
\end{example}

\section{परिपथ में ऊर्जा और शक्ति}

\begin{proposition}[विद्युत शक्ति]\label{prop:g11:circuits-and-power:power}
\index{विद्युत शक्ति}
\omterm{def:g11:circuits-and-power:voltage}{विभवांतर} $U$ के नीचे रखा वह उपकरण, जिससे धारा $I$ गुज़रती है,
इस दर से ऊर्जा लेता है:
\[ P = U\,I , \qquad\text{अर्थात्}\quad E = U\,I\,t \ \text{समय } t \text{ में} . \]
\end{proposition}

\begin{proof}
समय $t$ में आवेश $q = It$ उपकरण को पार करता है (\cref{def:g11:circuits-and-power:current}) और हर \omterm{def:g11:circuits-and-power:current}{कूलॉम}
$U$ \omterm{def:g10:energy-conservation:energy}{जूल} सौंप देता है (\cref{def:g11:circuits-and-power:voltage}): $E = qU = UIt$।
\end{proof}

\begin{proposition}[प्रतिरोधक में शक्ति]\label{prop:g11:circuits-and-power:ri2}
धारा $I$ ले जाता प्रतिरोधक $R$ इतनी शक्ति क्षय करता है:
\[ P = R\,I^2 = \frac{U^2}{R} . \]
\end{proposition}

\begin{proof}
$P = UI$ में \omterm{prop:g11:circuits-and-power:ohm}{ओम का नियम} रखिए: $P = (RI)I = RI^2 = U(U/R) = U^2/R$।
\end{proof}

\begin{example}[नामपट्ट पढ़ना]\label{ex:g11:circuits-and-power:nameplate}
``\qty{230}{V} -- \qty{2200}{W}'' अंकित केतली $I = P/U =
2200/230 = \qty{9.6}{A}$ खींचती है और उसका प्रतिरोध
$R = U^2/P = 230^2/2200 =
\qty{24}{\ohm}$ है। \qty{150}{s} चलाने पर वह $E = 2200 \times 150 =
\qty{3.3e5}{J} \approx \qty{0.09}{kW.h}$ खर्च करती है --- लगभग दो सेंट।
\end{example}

\begin{remark}[जूल और किलोवाट-घंटे]\label{rem:g11:circuits-and-power:kwh}
\omterm{def:g10:orders-of-magnitude:unit}{मीटर} किलोवाट-घंटे (\cref{def:g10:energy-conservation:kwh}) गिनता है: $\qty{1}{kW.h} = \qty{3.6e6}{J}$। \omterm{def:g10:energy-conservation:energy}{जूल} भौतिक विज्ञानी का
मात्रक है; बिल की क़ीमत किलोवाट-घंटे तय करते हैं।
\end{remark}

\section{जूल प्रभाव: टोस्टर से खंभे तक}

\begin{definition}[जूल प्रभाव]\label{def:g11:circuits-and-power:joule}
हर प्रतिरोधक में \omterm{def:g10:energy-conservation:forms}{विद्युत ऊर्जा} को ऊष्मा में बदल देने वाला क्षय
$P = RI^2$ \emph{जूल प्रभाव}\index{जूल प्रभाव} कहलाता है। हीटर, केतली, टोस्टर और
पुराने ढंग के बल्ब जान-बूझकर प्रतिरोधक बनाए जाते हैं; और \emph{फ़्यूज़}\index{फ़्यूज़}
वह पतला तार है जिसे इस तरह नापा जाता है कि धारा अपनी सीमा पार करते
ही वह पिघल जाए --- और परिपथ खुल जाए।
\end{definition}

\begin{example}[खंभे ऊँचे विभव पर क्यों चलते हैं]\label{ex:g11:circuits-and-power:transmission}
एक गाँव को प्रतिरोध $R_\ell = \qty{5.0}{\ohm}$ वाली लाइन से $P = \qty{100}{kW}$ चाहिए। $U = \qty{500}{V}$ पर
भेजें तो $I = P/U =
\qty{200}{A}$ और लाइन $R_\ell I^2 = \qty{200}{kW}$ बरबाद करती है --- पहुँचाई गई शक्ति की दुगुनी:
बेतुका। $U = \qty{20}{kV}$ पर: $I = \qty{5.0}{A}$ और $R_\ell I^2 = \qty{125}{W}$। चालीस गुना कम धारा, $1600$ गुना कम
हानि।
\end{example}

\begin{remark}[ग्रिड का लोहे-सा नियम]\label{rem:g11:circuits-and-power:grid}
पहुँचाई गई शक्ति $P = UI$ स्थिर हो तो लाइन की हानि $R_\ell I^2 = R_\ell
P^2/U^2$ संवहन
\omterm{def:g11:circuits-and-power:voltage}{विभवांतर} के \emph{वर्ग} के अनुपात में घटती है। इसलिए: यात्रा के लिए
ऊपर चढ़ाइए (\qty{400}{kV}), बरतने के लिए नीचे उतारिए (\qty{230}{V})।
\end{remark}

\begin{remark}[घर की सुरक्षा]\label{rem:g11:circuits-and-power:safety}
\qty{16}{A} पर \omterm{def:g11:circuits-and-power:joule}{फ़्यूज़} लगी \qty{230}{V} लाइन अधिक से अधिक $230 \times 16
\approx \qty{3.7}{kW}$ दे सकती है: एक साथ दो
केतलियाँ \omterm{def:g11:circuits-and-power:joule}{फ़्यूज़} पिघला देती हैं --- वह जान-बूझकर सबसे पतली, सबसे
कमज़ोर कड़ी है, और दीवार के भीतर लगी आग से कहीं सस्ती।
\end{remark}

\section{असली सेल: विद्युत वाहक बल}

\begin{definition}[विद्युत वाहक बल और आंतरिक प्रतिरोध]\label{def:g11:circuits-and-power:emf}
सेल \omterm{def:g10:energy-conservation:forms}{रासायनिक ऊर्जा} को \omterm{def:g10:energy-conservation:forms}{विद्युत ऊर्जा} में बदलता है। उसका
\emph{विद्युत वाहक बल}\index{विद्युत वाहक बल} $\mathcal{E}$, \omterm{def:g11:circuits-and-power:voltage}{वोल्ट} में, वह
ऊर्जा है जो वह अपने से गुज़रते हर \omterm{def:g11:circuits-and-power:current}{कूलॉम} को सौंपता है; और
उसका \emph{आंतरिक प्रतिरोध}\index{आंतरिक प्रतिरोध} $r$ उसकी अपनी
रसायन-क्रिया के भीतर होते \omterm{def:g11:circuits-and-power:joule}{जूल प्रभाव} का हिसाब रखता है। प्रतिरूप:
आदर्श स्रोत $\mathcal{E}$, $r$ के साथ श्रेणीक्रम में (गोलाकार $\mathcal{E}$: $E$
ऊर्जा है)।
\end{definition}

\begin{proposition}[सिरों का विभवांतर]\label{prop:g11:circuits-and-power:terminal}
धारा $I$ देता सेल अपने सिरों पर यह दिखाता है:
\[ U = \mathcal{E} - r\,I . \]
\end{proposition}

\begin{proof}
रसायन-क्रिया $\mathcal{E}I$ देती है, \omterm{def:g11:circuits-and-power:emf}{आंतरिक प्रतिरोध} $rI^2$ खा जाता है,
परिपथ को $UI$ मिलता है: \omterm{thm:g10:energy-conservation:conservation}{ऊर्जा संरक्षण} (\cref{ch:g10:energy-conservation}) से $UI = \mathcal{E}I - rI^2$ मिलता
है।
\end{proof}

\begin{example}[भार के नीचे एक सेल]\label{ex:g11:circuits-and-power:discharge}
एक सेल ($\mathcal{E} = \qty{9.0}{V}$, $r = \qty{1.0}{\ohm}$) प्रतिरोध $R = \qty{8.0}{\ohm}$ वाले बल्ब को खिलाता है।
फंदा $\mathcal{E} =
rI + RI$ मानता है, इसलिए $I = \mathcal{E}/(R + r) = \qty{1.0}{A}$ और $U = 9.0 -
1.0 \times 1.0 = \qty{8.0}{V}$: बल्ब को \qty{8.0}{W} मिलता है
जबकि सेल स्वयं गरम होकर \qty{1.0}{W} बरबाद करता है।
\end{example}

\begin{remark}[ग्राहक]\label{rem:g11:circuits-and-power:receivers}
मोटर या चार्ज होता सेल यही रूपांतरण उलटी दिशा में चलाता है: वह $U = \mathcal{E}' + r'I$ के
नीचे ऊर्जा सोखता है, $\mathcal{E}'I$ को गति या रसायन में बदलता है और $r'I^2$ गँवाता
है। स्रोत धकेलते हैं, ग्राहक पीछे धकेलते हैं।
\end{remark}

\begin{example}[सेल की दक्षता]\label{ex:g11:circuits-and-power:efficiency}
ऊपर के विसर्जन की \omterm{def:g10:energy-conservation:efficiency}{दक्षता} (\cref{def:g10:energy-conservation:efficiency}): $\mathcal{E}I =
\qty{9.0}{W}$ में से $UI = \qty{8.0}{W}$ काम का,
इसलिए $\eta = U/\mathcal{E} = 8/9 \approx 0.89$ --- और धारा बढ़ने के साथ यह गिरती जाती है: कड़ी मेहनत करते सेल
गरम रहते हैं।
\end{example}

\end{enumerate}
\end{exercise}

\section{समस्या: बाँध से टोस्टर तक}

\begin{problem}[{सप्ताहांत समस्या --- एक किलोवाट-घंटे की हज़ार किलोमीटर
लंबी यात्रा, गिरते पानी से लेकर सिंकती रोटी तक, और सफ़र का पूरा
हिसाब}]
\label{pb:g11:circuits-and-power:1}
पहाड़ का एक बाँध पानी को अपनी टरबाइनों से \qty{200}{m} ऊपर रोके रखता है, और वे
टरबाइनें हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} \qty{50}{m^3} --- यानी \qty{5.0e4}{kg} --- निगलती हैं; टरबाइन और
प्रत्यावर्तित्र \omterm{def:g10:energy-conservation:efficiency}{दक्षता} $0.90$ से रूपांतरण करते हैं। कुल
प्रतिरोध $R_\ell = \qty{32}{\ohm}$ (\qty{1000}{km} जाने और आने का) वाली एक लाइन से संयंत्र
उस शहर को खिलाता है जहाँ आपका \qty{900}{W} का टोस्टर \qty{230}{V} पर प्रतीक्षा कर रहा
है। प्रत्यावर्तित्र \qty{20}{kV} देता है; ट्रांसफ़ॉर्मर (हर एक की \omterm{def:g10:energy-conservation:efficiency}{दक्षता}
$0.99$) \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} जैसा चाहें बदल देते हैं। बिजली \unit{kW.h} के
$0.25$ यूरो के भाव पर बिकती है; $g = \qty{9.81}{N/kg}$।

\medskip
\noindent\textbf{भाग I --- बाँध पर।}
\begin{enumerate}
  \item गिरता पानी हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कितनी स्थितिज ऊर्जा छोड़ता है? इससे
        टरबाइनों को उपलब्ध शक्ति निकालिए।
  \item प्रत्यावर्तित्र से कितनी \omterm{prop:g11:circuits-and-power:power}{विद्युत शक्ति} निकलती है?
  \item सीधे प्रत्यावर्तित्र से \qty{20}{kV} पर भेजने पर कितनी धारा बहती है?
  \item उस धारा पर जूल-हानि $R_\ell I^2$ निकालिए और उसकी तुलना संयंत्र के उत्पादन
        से कीजिए। निष्कर्ष लिखिए।
  \item शहर की शक्ति $P = UI$ माँग से तय होती है। $I$ घटाने का
        एकमात्र उपाय क्या है, और वह काम कौन-सा उपकरण करता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- लाइन पर।}
\begin{enumerate}[resume]
  \item इस प्रश्न भर के लिए हानिरहित मानकर, ट्रांसफ़ॉर्मर \qty{88}{MW} को चढ़ाकर
        \qty{400}{kV} कर देता है: अब कितनी धारा बहती है?
  \item नई लाइन-हानि मेगावाट में निकालिए।
  \item संयंत्र के उत्पादन का कितना अंश लाइन में गँवाया जाता है?
  \item दिखाइए कि गँवाया गया अंश $R_\ell P/U^2$ के बराबर है, और जाँचिए कि यह
        प्रश्न 4 और 7 के बीच के गुणक को समझा देता है।
  \item \qty{400}{kV} पर लाइन के साथ-साथ \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} की गिरावट $R_\ell I$
        निकालिए। क्या यह प्रश्न 8 से मेल खाती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- रसोई में।}
\begin{enumerate}[resume]
  \item \qty{1}{kW.h} को \omterm{def:g10:energy-conservation:energy}{जूल} में बदलिए; \omterm{def:g10:orders-of-magnitude:unit}{मीटर} किससे गुणा करता है?
  \item \qty{230}{V} पर टोस्टर की धारा और प्रतिरोध निकालिए।
  \item तीन मिनट सेंकने पर: \omterm{def:g10:energy-conservation:energy}{जूल} और किलोवाट-घंटे में
        ऊर्जा, और उसका दाम।
  \item टोस्टर $RI^2$ क्षय करता है: एक वाक्य में समझाइए कि वही सूत्र लाइन
        पर हानि क्यों है और रसोई में उत्पाद।
  \item टोस्टर \qty{2.2}{kW} की केतली के साथ एक ही \qty{16}{A} का \omterm{def:g11:circuits-and-power:joule}{फ़्यूज़}
        बाँटता है: क्या \omterm{def:g11:circuits-and-power:joule}{फ़्यूज़} टिकेगा? अब \qty{2.0}{kW} का हीटर भी आ
        जुड़ता है: और अब?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- लेखा-जोखा।}
\begin{enumerate}[resume]
  \item घर की अपनी तारबंदी, प्रतिरोध \qty{0.05}{\ohm}, प्रश्न 15 की
        \qty{13.5}{A} ले जाती है: हानि निकालिए और मोटे तौर पर पहुँचाए गए \qty{3.1}{kW}
        में उसका अंश।
  \item सारी दक्षताओं (टरबाइन--प्रत्यावर्तित्र, ऊपर चढ़ाना, लाइन, नीचे
        उतारना, घर की तारबंदी) को जोड़कर एक कुल ग्रिड \omterm{def:g10:energy-conservation:efficiency}{दक्षता}
        बनाइए।
  \item \emph{एक} किलोवाट-घंटे को आपके टोस्टर तक पहुँचाने के लिए
        बाँध का कितने \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} (और कितने घन \omterm{def:g10:orders-of-magnitude:unit}{मीटर}) पानी
        गिरना चाहिए?
  \item प्रश्न 9 वाले गँवाए गए अंश $R_\ell P/U^2$ का मान $U = \qty{20}{kV}$ पर निकालिए।
        $1$ से बड़े मान का अर्थ क्या है?
  \item और अंतिम चोट --- \emph{दूरी की क़ीमत}: दो वाक्यों में लिखिए कि एक
        सिंका हुआ \omterm{def:g10:energy-conservation:kwh}{किलोवाट-घंटा} पहाड़ को कितना पड़ता है, रास्ते में
        उसका कितना अंश मर जाता है, और कौन-सी एक तरकीब इसे संभव बनाती है।
\end{enumerate}
\end{problem}
