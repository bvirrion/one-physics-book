\chapter{लेंस, प्रतिबिंब और नेत्र}\label{ch:g11:lenses-and-eye}

हर साफ़ चित्र --- फ़ोन के संवेदक पर, सिनेमा के परदे पर, या आपकी अपनी आँख
के पिछले हिस्से पर --- एक ही करतब से बनता है: कोई वक्र पारदर्शी माध्यम
किरणों को \omterm{def:g10:refraction:refraction}{अपवर्तन} (\cref{ch:g10:refraction}) से इस तरह मोड़ देता है कि एक बिंदु से
निकली सारी किरणें दूसरे बिंदु पर फिर मिल जाएँ। यह अध्याय उसी करतब को
साधता है: एक लेंस, तीन किरणें, एक सूत्र --- और फिर उसी को
कैमरों, चश्मों और उस जीवित लेंस पर तान देता है जो यह वाक्य पढ़
रहा है।

\section{अभिसारी और अपसारी लेंस}

\begin{definition}[पतला लेंस]\label{def:g11:lenses-and-eye:lens}
\emph{लेंस}\index{लेंस} वह पारदर्शी माध्यम है जो दो सतहों से घिरा हो और
जिसकी कम से कम एक सतह वक्र हो। जो किनारों की तुलना में बीच में मोटा हो
वह \emph{अभिसारी}\index{अभिसारी लेंस} है: समांतर किरणें एक-दूसरे की ओर
मुड़ जाती हैं; जो किनारों पर मोटा हो वह
\emph{अपसारी}\index{अपसारी लेंस}: किरणें फैल जाती हैं।
\emph{पतले}\index{पतला लेंस} लेंस की मोटाई नगण्य होती है: हम उसे बाहर की
ओर (अभिसारी) या भीतर की ओर (अपसारी) नोकों वाले एक खंड के रूप में खींचते
हैं। उसका केंद्र $O$ \emph{प्रकाशिक केंद्र}\index{प्रकाशिक केंद्र} है,
और $O$ से गुज़रती लंब रेखा \emph{मुख्य अक्ष}\index{मुख्य अक्ष}।
\end{definition}

\begin{definition}[फोकस और फोकस दूरी]\label{def:g11:lenses-and-eye:focal}
किसी \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} के अक्ष के समांतर आती किरणें उसके पीछे एक ही
बिंदु पर अक्ष को काटती हैं, जो \emph{प्रतिबिंब फोकस}\index{फोकस} $F'$
है; और सममित रूप से, \emph{वस्तु फोकस} $F$ (जो आगे है, और $OF = OF'$) से
निकलती किरणें अक्ष के समांतर बाहर जाती हैं। \emph{फोकस दूरी}\index{फोकस
दूरी} $f' = \overline{OF'}$ \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} के लिए धनात्मक होती है; किसी
\omterm{def:g11:lenses-and-eye:lens}{अपसारी लेंस} से गुज़रकर समांतर किरणें ऐसे निकलती हैं मानो वे
लेंस के \emph{आगे} पड़े किसी बिंदु $F'$ से आ रही हों: $f' < 0$।
\end{definition}

\begin{definition}[क्षमता]\label{def:g11:lenses-and-eye:vergence}
$f'$ (मीटरों में) \omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} वाले लेंस की
\emph{क्षमता}\index{क्षमता} $C = 1/f'$ होती है, और उसे
\emph{डायॉप्टर}\index{डायॉप्टर} (संकेत D) में नापा जाता है: $1~\text{D} = \qty{1}{m^{-1}}$;
\omterm{def:g11:lenses-and-eye:lens}{अपसारी} लेंसों की $C < 0$ होती है। नेत्र-विशेषज्ञ डायॉप्टर में
नुस्ख़ा लिखते हैं: सटे हुए \omterm{def:g11:lenses-and-eye:lens}{पतले} लेंसों की क्षमताएँ बस जुड़ जाती
हैं (यहाँ स्वीकृत; व्युत्पत्ति स्नातक वर्ष 1 के खंड में)।
\end{definition}

\begin{example}[नुस्ख़ा पढ़ना]\label{ex:g11:lenses-and-eye:vergence}
$f' = \qty{0.50}{m}$ वाले लेंस की क्षमता $C = 1/0.50 = +2.0$~D है: \omterm{def:g11:lenses-and-eye:lens}{अभिसारी}।
$-4.0$~D का नुस्ख़ा $f' = \qty{-0.25}{m}$ का अर्थ रखता है: \omterm{def:g11:lenses-and-eye:lens}{अपसारी},
\omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} \qty{25}{cm}।
\end{example}

\section{प्रतिबिंब की रचना}

\begin{method}[रचना की तीन किरणें]\label{met:g11:lenses-and-eye:rays}
मान लीजिए $AB$ अक्ष के लंबवत, $A$ पर रखी कोई वस्तु है। नोक $B$ से
निकलती सारी किरणों में से तीन के रास्ते पहले से पता हैं:
\begin{enumerate}
  \item \omterm{def:g11:lenses-and-eye:lens}{प्रकाशिक केंद्र} $O$ से गुज़रती किरण सीधी चली जाती है;
  \item अक्ष के समांतर आती किरण $F'$ से होकर निकलती है;
  \item $F$ से गुज़रती किरण अक्ष के समांतर निकलती है।
\end{enumerate}
इनमें से कोई दो काफ़ी हैं: उनका कटान-बिंदु ही प्रतिबिंब-बिंदु $B'$ है,
और प्रतिबिंब $A'B'$ अक्ष के लंबवत $A'$ पर खड़ा होता है। यदि निकलती
किरणें फैल जाएँ तो उनकी \emph{पीछे की ओर} बढ़ाई गई रेखाएँ (बिंदुदार) एक
\omterm{def:g11:lenses-and-eye:images}{आभासी प्रतिबिंब} बिंदु पर मिलती हैं।
\end{method}

\begin{definition}[वास्तविक और आभासी प्रतिबिंब]\label{def:g11:lenses-and-eye:images}
प्रतिबिंब \emph{वास्तविक}\index{वास्तविक प्रतिबिंब} तब होता है जब
निकलती किरणें सचमुच उससे होकर गुज़रें: वहाँ रखा परदा उसे पकड़ लेता है;
और \emph{आभासी}\index{आभासी प्रतिबिंब} तब, जब केवल उनकी पीछे बढ़ाई गई
रेखाएँ मिलें: तब कोई परदा उसे नहीं पकड़ता, पर लेंस में झाँकती
आँख उसे बख़ूबी देख लेती है। \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} $F$ से परे रखी हर
वस्तु का वास्तविक, उलटा प्रतिबिंब देता है; वस्तु को \omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} के
भीतर सरका दीजिए और प्रतिबिंब आभासी, सीधा और बड़ा हो जाता है।
\end{definition}

\section{एक ही सूत्र: संयुग्मी संबंध}

रचनाएँ बताती हैं कि प्रतिबिंब \emph{कहाँ} है; सूत्र उसे निकाल देता है।
अक्ष को प्रकाश की दिशा में लीजिए और \emph{बीजीय मापों} का उपयोग कीजिए:
$\overline{OA}$ $O$ के सापेक्ष $A$ का निर्देशांक है (आगे रखी
\omterm{def:g11:lenses-and-eye:images}{वास्तविक} वस्तु के लिए ऋणात्मक); ऊँचाइयाँ $\overline{AB}$ ऊपर की ओर
धनात्मक गिनी जाती हैं।

\begin{theorem}[पतले लेंस का संयुग्मी संबंध]\label{thm:g11:lenses-and-eye:conjugate}
$f'$ \omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} वाले \omterm{def:g11:lenses-and-eye:lens}{पतले} लेंस से गुज़रकर अक्ष पर रखा
वस्तु-बिंदु $A$ बिंदु $A'$ पर प्रतिबिंबित होता है, जहाँ
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'} = C .
\]
\end{theorem}
\admitted

\begin{remark}
यह संबंध अक्ष के पास वाली किरणों पर \omterm{thm:g10:refraction:snell}{स्नेल का नियम} लगाने से निकलता
है; व्युत्पत्ति --- और \omterm{def:g11:lenses-and-eye:lens}{पतले} लेंस वाले प्रतिरूप की सीमाएँ ---
स्नातक वर्ष 1 के खंड में ईमानदारी से रखी गई हैं।
\end{remark}

\begin{definition}[आवर्धन]\label{def:g11:lenses-and-eye:magnification}
प्रतिबिंब का \emph{आवर्धन}\index{आवर्धन}
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}}
       = \frac{\overline{OA'}}{\overline{OA}} .
\]
होता है। $\abs{\gamma}$ आकारों की तुलना करता है और उसका चिह्न दिशाओं की ($\gamma < 0$:
उलटा); जबकि $\overline{OA'}$ का चिह्न प्रकृति बताता है (धनात्मक:
\omterm{def:g11:lenses-and-eye:images}{वास्तविक}, लेंस के पीछे; ऋणात्मक: \omterm{def:g11:lenses-and-eye:images}{आभासी}, आगे)।
\end{definition}

\begin{example}[सूत्र काम पर]\label{ex:g11:lenses-and-eye:conjugate}
कोई वस्तु $f' = \qty{10}{cm}$ वाले \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} के \qty{30}{cm} आगे खड़ी है:
$\overline{OA} = \qty{-30}{cm}$, इसलिए
\[
\frac{1}{\overline{OA'}} = \frac{1}{10} - \frac{1}{30} = \frac{1}{15},
\qquad \overline{OA'} = +\qty{15}{cm},
\qquad \gamma = \frac{15}{-30} = -0.50 :
\]
यानी लेंस के \qty{15}{cm} पीछे एक \omterm{def:g11:lenses-and-eye:images}{वास्तविक प्रतिबिंब}, उलटा और
आधे आकार का --- ठीक वही जो तीनों किरणें खींचती हैं।
\end{example}

\section{नेत्र}

\begin{definition}[एक प्रकाशिक तंत्र के रूप में नेत्र]\label{def:g11:lenses-and-eye:eye}
प्रकाशिकी की दृष्टि से नेत्र परदे के सामने रखा एक \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस}
है: \emph{कॉर्निया}\index{कॉर्निया} और
\emph{नेत्र-लेंस}\index{नेत्र-लेंस} मिलकर बदली जा सकने वाली
\omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} का एक \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} बनाते हैं; और लगभग \qty{17}{mm}
पीछे रखा \emph{दृष्टिपटल}\index{दृष्टिपटल} स्थिर परदा है। \omterm{def:g11:lenses-and-eye:focal}{फोकस} करना
लेंस को दबाकर होता है, कैमरे की तरह उसे खिसकाकर नहीं:
\emph{समंजन}\index{समंजन} वही $f'$ का बदलाव है जो पक्ष्माभी
माँसपेशियाँ पैदा करती हैं। पूरे समंजन पर साफ़ दिखने वाला सबसे पास का
बिंदु \emph{निकट बिंदु}\index{निकट बिंदु} है --- \emph{मानक
नेत्र}\index{मानक नेत्र} के लिए लगभग \qty{25}{cm} --- और विश्राम में सबसे दूर
वाला \emph{दूर बिंदु}\index{दूर बिंदु} (मानक नेत्र के लिए अनंत)।
\end{definition}

\end{example}

\section{दोष, संशोधन और आवर्धक}

\begin{definition}[निकट-दृष्टि और दूर-दृष्टि दोष]\label{def:g11:lenses-and-eye:defects}
\emph{निकट-दृष्टि दोष}\index{निकट-दृष्टि दोष} वाला नेत्र अपनी लंबाई के
हिसाब से अधिक \omterm{def:g11:lenses-and-eye:lens}{अभिसारी} होता है: समांतर किरणें
\omterm{def:g11:lenses-and-eye:eye}{दृष्टिपटल} के \emph{आगे} \omterm{def:g11:lenses-and-eye:focal}{फोकस} होती हैं, दूर की वस्तुएँ धुँधली
पड़ जाती हैं, और \omterm{def:g11:lenses-and-eye:eye}{दूर बिंदु} किसी परिमित दूरी पर आ जाता है; इसे
\omterm{def:g11:lenses-and-eye:lens}{अपसारी} चश्मे का लेंस सुधार देता है। \emph{दूर-दृष्टि
दोष}\index{दूर-दृष्टि दोष} वाला नेत्र पर्याप्त \omterm{def:g11:lenses-and-eye:lens}{अभिसारी} नहीं
होता: पास की वस्तुएँ \omterm{def:g11:lenses-and-eye:eye}{दृष्टिपटल} के पीछे \omterm{def:g11:lenses-and-eye:focal}{फोकस} होतीं, और
\omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु} पीछे हट जाता है; इसे \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} सुधारता
है। उम्र के साथ \omterm{def:g11:lenses-and-eye:eye}{नेत्र-लेंस} कड़ा पड़ जाता है (\emph{जरा-दृष्टि
दोष}\index{जरा-दृष्टि दोष}) और \omterm{def:g11:lenses-and-eye:eye}{समंजन} सिकुड़ जाता है --- इलाज
वही: पढ़ने का चश्मा।
\end{definition}

\begin{example}[निकट-दृष्टि दोष का नुस्ख़ा]\label{ex:g11:lenses-and-eye:correction}
\omterm{def:g11:lenses-and-eye:defects}{निकट-दृष्टि दोष} वाले नेत्र का \omterm{def:g11:lenses-and-eye:eye}{दूर बिंदु} \qty{50}{cm} पर है:
सुधारने वाले लेंस को दूर की वस्तुएँ \emph{\omterm{def:g11:lenses-and-eye:eye}{दूर बिंदु} पर}
दिखानी होंगी --- यानी वस्तु अनंत पर और प्रतिबिंब $\overline{OA'} = \qty{-0.50}{m}$ पर
(\omterm{def:g11:lenses-and-eye:images}{आभासी}, आगे)। तब $1/f' = 1/\overline{OA'}$: $f' = \qty{-0.50}{m}$, $C = -2.0$~D; और विश्राम में
रखा नेत्र उसी \omterm{def:g11:lenses-and-eye:images}{आभासी प्रतिबिंब} को देखकर उसे साफ़ देख लेता है।
\end{example}

\begin{definition}[आवर्धक]\label{def:g11:lenses-and-eye:magnifier}
\emph{आवर्धक}\index{आवर्धक} छोटी \omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} वाला एक
\omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} है, जिसे वस्तु को \omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} के \emph{भीतर}
रखकर काम में लिया जाता है: तब प्रतिबिंब \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा और बड़ा
(\cref{def:g11:lenses-and-eye:images}) होता है, और नेत्र उसे आराम से दूर रखकर देख लेता है।
\end{definition}

\begin{example}[जौहरी का आवर्धक]\label{ex:g11:lenses-and-eye:magnifier}
$f' = \qty{5.0}{cm}$ वाला \omterm{def:g11:lenses-and-eye:magnifier}{आवर्धक}, और \qty{4.0}{cm} पर रखा नगीना: $1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$, इसलिए $\overline{OA'} = \qty{-20}{cm}$ और
$\gamma = (-20)/(-4.0) = +5.0$: यानी एक \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा और पाँच गुना बड़ा प्रतिबिंब, जो
लेंस से आरामदेह \qty{20}{cm} पर बनता है।
\end{example}

\begin{exercise}[$\star\star\star$]\label{exo:g11:lenses-and-eye:15}
कोई \omterm{def:g11:lenses-and-eye:images}{वास्तविक} वस्तु किसी \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} ($\overline{OA} = -d$, $f' > 0$)
के आगे $d > 0$ दूरी पर खड़ी है। दिखाइए कि $\overline{OA'} = \frac{f'd}{d - f'}$ और $\gamma = \frac{f'}{f' - d}$। इससे

और $0 < d < f'$ होने पर \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा तथा \emph{बड़ा} ($\gamma > 1$) ---
यानी हर \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} पर्याप्त पास से देखने पर एक
\omterm{def:g11:lenses-and-eye:magnifier}{आवर्धक} है।
\end{exercise}

\section{समस्या: एक सूत्र, तीन उपकरण}

\begin{problem}[{सप्ताहांत समस्या --- वही संयुग्मी संबंध एक कैमरा फोकस
करता है, पढ़ने का चश्मा लिखता है, एक जौहरी का आवर्धक गढ़ता है, और आपकी
अपनी आँख में लगे ज़ूम को भी नाप देता है}]
\label{pb:g11:lenses-and-eye:1}
एक छायाकार \qty{50}{mm} का लेंस सधाता है, एक दादी अख़बार हाथ भर दूर
पकड़ती है, एक जौहरी हीरे पर झुकता है: बीजगणित की एक ही पंक्ति, $1/\overline{OA'} - 1/\overline{OA} = 1/f'$
(\cref{thm:g11:lenses-and-eye:conjugate}), तीनों पेशे चलाती है --- और फिर स्वयं नेत्र भी।

\medskip
\noindent\textbf{भाग I --- एक ही सूत्र।} किसी \omterm{def:g11:lenses-and-eye:lens}{अभिसारी लेंस} की
$f' = \qty{20}{cm}$ है।
\begin{enumerate}
  \item वस्तु \qty{60}{cm} पर: $\overline{OA'}$ और $\gamma$ निकालिए; प्रतिबिंब बताइए।
  \item वस्तु को \qty{30}{cm} पर ले जाइए; फिर से निकालिए। प्रश्न 1 के साथ
        क्या आपस में बदल गया? (प्रकाश-पथ प्रतिवर्ती होते हैं।)
  \item उसे \qty{10}{cm} पर, यानी $f'$ के भीतर ले जाइए: फिर से निकालिए और
        बताइए।
  \item सार में कहिए: जैसे-जैसे वस्तु बहुत दूर से $F$ की ओर और फिर
        उसके पार सरकती है, प्रतिबिंब कैसे चलता है और उसकी प्रकृति कैसे
        बदलती है?
  \item दिखाइए कि बहुत दूर की वस्तु ($1/\overline{OA} \approx 0$) का प्रतिबिंब हमेशा \omterm{def:g11:lenses-and-eye:focal}{फोकस}
        तल में बनता है: $\overline{OA'} = f'$।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- छायाकार।}
किसी कैमरे के लेंस की $f' = \qty{50}{mm}$ है; और \qty{24}{mm} ऊँचे संवेदक को
लेंस के सापेक्ष खिसकाया जा सकता है।
\begin{enumerate}[resume]
  \item दूर के दृश्य पर \omterm{def:g11:lenses-and-eye:focal}{फोकस} करने पर संवेदक कहाँ है?
  \item अब \qty{1.0}{m} दूर एक चेहरा: नई लेंस--संवेदक दूरी और दृश्य वाली
        सेटिंग से सरकाव निकालिए।
  \item यंत्र-रचना अधिक से अधिक \qty{55}{mm} की लेंस--संवेदक दूरी देती है।
        सबसे कम फोकस-दूरी निकालिए।
  \item उस सबसे पास वाले \omterm{def:g11:lenses-and-eye:focal}{फोकस} पर $\gamma$ निकालिए। क्या \qty{20}{cm} के चेहरे
        का प्रतिबिंब संवेदक पर समा जाएगा?
  \item असली आकार वाले ``मैक्रो'' का अर्थ है $\gamma = -1$: तब वस्तु और संवेदक
        कहाँ होने चाहिए, और इसके लिए ख़ास लेंस क्यों चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दादी।}
उम्र के साथ \omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु} पीछे हटता जाता है: दादी का \qty{1.0}{m} पर है।
\begin{enumerate}[resume]
  \item हाथ भर दूर पकड़े अख़बार की व्याख्या कीजिए।
  \item \qty{25}{cm} पर पढ़ने के लिए उनके चश्मे को \qty{25}{cm} पर रखे पन्ने को
        \qty{1.0}{m} पर एक \omterm{def:g11:lenses-and-eye:images}{आभासी प्रतिबिंब} में बदलना होगा: ज़रूरी
        क्षमता निकालिए।
  \item $\gamma$ निकालिए। प्रतिबिंब चार गुना बड़ा है, फिर भी बड़ा नहीं
        दिखता: आकारों \emph{और} दूरियों, दोनों की तुलना करके समझाइए।
  \item दिखाइए कि छपाई केवल \qty{25}{cm} और $f' \approx \qty{33}{cm}$ के बीच ही साफ़ रहती है।
        और चश्मे से देखी किसी \emph{दूर} की वस्तु का क्या?
  \item एक वाक्य में निकालिए कि द्विफोकसी लेंस क्यों बने।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- जौहरी, और नेत्र का अपना ज़ूम।}
जौहरी के \omterm{def:g11:lenses-and-eye:magnifier}{आवर्धक} की $f' = \qty{5.0}{cm}$ है; उसका \omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु} \qty{25}{cm} पर है;
और उसका \omterm{def:g11:lenses-and-eye:eye}{दृष्टिपटल} उसके नेत्र के लेंस से \qty{17}{mm} पीछे
है।
\begin{enumerate}[resume]
  \item \omterm{def:g11:lenses-and-eye:magnifier}{आवर्धक} के नीचे \qty{4.0}{cm} पर रखा नगीना: स्थिति, प्रकृति,
        \omterm{def:g11:lenses-and-eye:magnification}{आवर्धन}?
  \item नगीना कहाँ रखा जाए ताकि उसका प्रतिबिंब उसके \omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु}
        पर पड़े, और तब $\gamma$ कितना है?
  \item अनुभवी जौहरी नगीने को इसके बजाय ठीक $F$ पर क्यों रखते हैं
        (प्रतिबिंब अनंत पर)?
  \item अब अकेला नेत्र: दूर की वस्तु देखते समय उसकी क्षमता
        निकालिए, फिर \qty{25}{cm} पर रखी वस्तु के लिए: \omterm{def:g11:lenses-and-eye:eye}{समंजन} के
        कितने \omterm{def:g11:lenses-and-eye:vergence}{डायॉप्टर} लगे?
  \item दोनों क्षमताओं को \omterm{def:g11:lenses-and-eye:focal}{फोकस} दूरियों में बदलिए: नेत्र
        अपनी \omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} कितनी बदलता है, मिलीमीटरों और प्रतिशत
        में --- यानी वही ज़ूम जो कैमरे ने सरकाव से और दादी ने $+3$~D
        से किया?
\end{enumerate}
\end{problem}
