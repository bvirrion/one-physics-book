\chapter{यांत्रिक ऊर्जा और उसका संरक्षण}\label{ch:g11:mechanical-energy}

पहली चढ़ाई के बाद रोलर कोस्टर के पास कोई इंजन नहीं होता: उसके बाद का हर फंदा,
हर तेज़ी उसी खाते से निकलती है जो चढ़ते समय खोला गया था। यह अध्याय उस बहीखाते
को दो संख्याओं में बदल देता है --- गतिज और स्थितिज ऊर्जा --- और एक
नियम में: जब तक केवल भार कार्य करता है, उनका योग नहीं बदलता।
\omterm{def:g10:inertia:inventory}{घर्षण} उसकी फ़ीस है; और हम उसे बहीखाते से पढ़ना सीखेंगे।

\section{गतिज ऊर्जा}

\begin{definition}[गतिज ऊर्जा]\label{def:g11:mechanical-energy:kinetic}
द्रव्यमान $m$ (\unit{kg}) का पिंड चाल $v$ (\unit{m/s}) से चलते हुए
\emph{गतिज ऊर्जा}\index{गतिज ऊर्जा} रखता है:
\[
E_k = \tfrac12\, m v^2 .
\]
मात्रकों की जाँच: $\unit{kg.m^2/s^2} = \unit{N.m} = \unit{J}$ --- यानी \omterm{def:g11:work-of-force:work}{जूल}, हर ऊर्जा की
तरह (\cref{ch:g10:energy-conservation})।
\end{definition}

\begin{example}[परिमाण की कोटियाँ]\label{ex:g11:mechanical-energy:orders}
\begin{itemize}
  \item पैदल चलता व्यक्ति, \qty{1.4}{m/s} पर \qty{70}{kg}: $E_k = \tfrac12 \times 70 \times 1.4^2 \approx \qty{69}{J}$;
  \item कार, \qty{130}{km/h} $= \qty{36.1}{m/s}$ पर \qty{1300}{kg}: $E_k \approx \qty{8.5e5}{J}$;
  \item तेज़ रफ़्तार रेलगाड़ी, $m = \qty{3.85e5}{kg}$, \qty{320}{km/h} $= \qty{88.9}{m/s}$ पर: $E_k \approx \qty{1.5e9}{J}$;
  \item राइफ़ल की गोली, \qty{800}{m/s} पर \qty{8.0}{g}: $E_k \approx \qty{2.6}{kJ}$।
\end{itemize}
टहलने और रेलगाड़ी के बीच सात कोटियाँ हैं। और $E_k$ चाल के \emph{वर्ग} की
तरह बढ़ता है: \qty{130}{km/h} पर कार \qty{65}{km/h} वाली अपनी ऊर्जा का $4$ गुना ले
चलती है --- यानी ब्रेकों को चार गुना निकालना पड़ेगा।
\end{example}

\section{गतिज-ऊर्जा प्रमेय}

\begin{theorem}[गतिज-ऊर्जा प्रमेय]\label{thm:g11:mechanical-energy:ketheorem}
\index{गतिज-ऊर्जा प्रमेय}
सरल रेखा पर $A$ से $B$ तक चलते द्रव्यमान $m$ के पिंड के लिए
\omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} का बदलाव उसे मिले कुल कार्य (\cref{ch:g11:work-of-force}) के बराबर
होता है:
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
जहाँ योग पिंड पर लगते सभी बलों पर है। यह किसी भी पथ पर होती गति तक
फैल जाता है, और हम उसे भी यहीं मान लेते हैं।
\end{theorem}
\admitted

\begin{example}[मुक्त पतन, फ़्रेम-दर-फ़्रेम जाँचा हुआ]\label{ex:g11:mechanical-energy:freefall}
\qty{0.200}{kg} की एक गेंद छोड़ी जाती है और उसका चलचित्र बनाया जाता है; फ़्रेम ये
आँकड़े देते हैं:
\begin{center}\small
\begin{tabular}{lcccc}
$t$ (\unit{s}) & 0.10 & 0.20 & 0.30 & 0.40\\
गिरी हुई ऊँचाई $h$ (\unit{m}) & 0.049 & 0.196 & 0.441 & 0.785\\
चाल $v$ (\unit{m/s}) & 0.98 & 1.96 & 2.94 & 3.92\\
$E_k = \tfrac12 mv^2$ (\unit{J}) & 0.096 & 0.384 & 0.864 & 1.54\\
$W = mgh$ (\unit{J}) & 0.096 & 0.385 & 0.865 & 1.54
\end{tabular}
\end{center}
$E_k = 0$ से चलकर हर फ़्रेम पर $E_k$ भार के कार्य $mgh$ से मेल
खाता है: $\Delta E_k = W(\vect P)$, यानी $v^2 = 2gh$ --- यही प्रमेय है, और वह पाठ्यांकों की
परिशुद्धता तक सत्यापित हो जाती है।
\end{example}

\begin{remark}[उपपत्ति कहाँ रहती है]\label{rem:g11:mechanical-energy:proof}
सामान्य कथन --- जो केवल सरल रेखा पर नहीं, हर पथ पर लागू है ---
गति के दूसरे नियम और थोड़े कलन से निकलता है, और दोनों इसी पुस्तक में आगे
(\cref{ch:g12:newtons-laws,%
ch:g12:work-and-energy}); तब तक मुक्त पतन की यह जाँच ही हमारा प्रमाणपत्र है, और हम वक्र-पथ
वाला रूप मानकर उसका खुलकर उपयोग करते हैं।
\end{remark}

\section{गुरुत्वीय स्थितिज ऊर्जा}

\begin{definition}[गुरुत्वीय स्थितिज ऊर्जा]\label{def:g11:mechanical-energy:potential}
ऊँचाई $z$ पर रखा द्रव्यमान $m$ का पिंड, जहाँ ऊँचाई किसी चुने हुए
\emph{संदर्भ स्तर}\index{संदर्भ स्तर} (जहाँ $z = 0$) से ऊपर की ओर नापी जाती
है, \emph{गुरुत्वीय स्थितिज ऊर्जा}\index{स्थितिज ऊर्जा!गुरुत्वीय} संचित
रखता है:
\[
E_p = m g z, \qquad g = \qty{9.81}{N/kg}.
\]
\end{definition}

\begin{remark}[मायने केवल अंतर रखते हैं]\label{rem:g11:mechanical-energy:reference}
$A$ से $B$ तक जाने पर $E_p$ $mg(z_B - z_A) =
-W_{AB}(\vect P)$ जितना बदलता है: यानी भार
के कार्य का ठीक उलटा (\cref{ch:g11:work-of-force})। \omterm{def:g11:mechanical-energy:potential}{संदर्भ स्तर} मन-मरज़ी का
चुनाव है --- फ़र्श, समुद्र-तल, मेज़ की सतह --- और वह हर जगह $E_p$ को एक अचर
जितना खिसका देता है, जो हर अंतर में से कट जाता है। उसे वहाँ रखिए जहाँ
संख्याएँ सबसे सरल हों, और बीच समस्या में उसे मत हिलाइए।
\end{remark}

\begin{example}[ताक़ पर रखी किताब]\label{ex:g11:mechanical-energy:shelf}
\qty{1.2}{kg} की किताब फ़र्श से \qty{1.8}{m} ऊपर रखी है, और फ़र्श ख़ुद सड़क से \qty{9.0}{m}
ऊपर है। संदर्भ फ़र्श पर हो तो $E_p = 1.2 \times 9.81 \times 1.8 \approx \qty{21}{J}$; सड़क पर हो तो $E_p \approx \qty{127}{J}$। फ़र्श तक गिरने पर
दोनों खातों में \qty{21}{J} ही छूटता है।
\end{example}

\section{यांत्रिक ऊर्जा और उसका संरक्षण}

\begin{definition}[यांत्रिक ऊर्जा]\label{def:g11:mechanical-energy:em}
किसी पिंड की \emph{यांत्रिक ऊर्जा}\index{यांत्रिक ऊर्जा} है
\[
E_m = E_k + E_p = \tfrac12 m v^2 + m g z .
\]
\end{definition}

\begin{theorem}[यांत्रिक ऊर्जा का संरक्षण]\label{thm:g11:mechanical-energy:conservation}
\index{यांत्रिक ऊर्जा का संरक्षण}
यदि पिंड पर कार्य करता एकमात्र बल उसका भार हो (कोई
\omterm{def:g10:inertia:inventory}{घर्षण} नहीं; बाक़ी बल, जैसे घर्षणहीन पटरी की प्रतिक्रिया, गति
के लंबवत हों), तो $E_m$ संरक्षित रहता है: गति के किन्हीं दो बिंदुओं $A$
और $B$ के लिए
\[
\tfrac12 m v_A^2 + m g z_A = \tfrac12 m v_B^2 + m g z_B .
\]
\end{theorem}

\begin{proof}
गतिज-ऊर्जा प्रमेय से $E_k(B) - E_k(A) = W_{AB}(\vect P) =
mg(z_A - z_B) = E_p(A) - E_p(B)$: जो $E_k$ कमाता है, वही $E_p$
गँवाता है।
\end{proof}

\begin{example}[पानी की फिसलपट्टी]\label{ex:g11:mechanical-energy:slide}
कोई बच्चा $h = \qty{3.2}{m}$ ऊँची घर्षणहीन फिसलपट्टी के सिरे पर \omterm{def:g10:relative-motion:frame}{विराम} से चलता है। संदर्भ
नीचे हो तो $mgh = \tfrac12 mv^2$, इसलिए $v = \sqrt{2gh} \approx \qty{7.9}{m/s}$ (\qty{29}{km/h}) --- द्रव्यमान चाहे जो हो
\emph{और फिसलपट्टी का आकार चाहे जो हो}: सीधी, मुड़ी या सर्पिल, हिसाब में
केवल ऊँचाई की गिरावट आती है।
\end{example}

\begin{example}[लोलक]\label{ex:g11:mechanical-energy:pendulum}
लोलक के गोलक को एक ओर तब तक खींचा जाता है जब तक वह अपने सबसे निचले बिंदु से
$h = \qty{12}{cm}$ ऊपर न उठ जाए, फिर छोड़ दिया जाता है। दोनों छोरों पर $v = 0$; सबसे
नीचे पूरा $mgh$ गतिज है, $v = \sqrt{2gh} \approx \qty{1.5}{m/s}$; और दूसरी ओर गोलक ठीक \qty{12}{cm} तक चढ़ जाता
है।
\end{example}

\end{example}

\section{घर्षण से क्षय}

\begin{definition}[क्षय]\label{def:g11:mechanical-energy:dissipation}
यांत्रिक खाते से निकल जाती ऊर्जा के बारे में कहा जाता है कि वह
\emph{क्षय}\index{क्षय} हो गई: वह रगड़ती सतहों और हवा में \omterm{def:g10:energy-conservation:forms}{तापीय
ऊर्जा} बनकर फिर प्रकट हो जाती है। कुल ऊर्जा संरक्षित रहती है
(\cref{ch:g10:energy-conservation}); अकेली \omterm{def:g11:mechanical-energy:em}{यांत्रिक ऊर्जा} नहीं।
\end{definition}

\begin{proposition}[घाटा घर्षण को नाप देता है]\label{prop:g11:mechanical-energy:deficit}
यदि गति के अनुदिश घर्षण-बल $\vect f$ भी कार्य करे,
तो
\[
\Delta E_m = E_m(B) - E_m(A) = W_{AB}(\vect f) < 0:
\]
यानी \omterm{def:g11:mechanical-energy:em}{यांत्रिक ऊर्जा} की हानि \omterm{def:g10:inertia:inventory}{घर्षण} के कार्य के
बराबर होती है।
\end{proposition}

\begin{proof}
गतिज-ऊर्जा प्रमेय अब $\Delta E_k = W_{AB}(\vect P) +
W_{AB}(\vect f) = -\Delta E_p + W_{AB}(\vect f)$ कहता है; $\Delta E_p$ को दूसरी ओर ले
जाइए। \omterm{def:g10:inertia:inventory}{घर्षण} गति का विरोध करता है, इसलिए उसका कार्य ऋणात्मक
है।
\end{proof}

\begin{example}[बर्फ़-गाड़ी का लेखा]\label{ex:g11:mechanical-energy:toboggan}
बच्चा और स्लेज ($m = \qty{40}{kg}$) \omterm{def:g10:relative-motion:frame}{विराम} से चलते हैं, \qty{12}{m} बर्फ़ पर $h = \qty{5.0}{m}$ गिरते
हैं और घर्षणहीन $\sqrt{2gh} \approx \qty{9.9}{m/s}$ के बजाय \qty{7.0}{m/s} पर पहुँचते हैं। संदर्भ नीचे रखने पर:
$\Delta E_m = \tfrac12 \times 40 \times 7.0^2 - 40 \times 9.81
\times 5.0 = 980 - 1962 = \qty{-982}{J}$। तो \qty{12}{m} पर $W(\vect f) =
\qty{-982}{J}$: यानी औसत घर्षण-बल $f = 982/12 \approx \qty{82}{N}$,
और फिसलपट्टियों तथा बर्फ़ में \qty{982}{J} की गरमाहट।
\end{example}

घटता हुआ --- हर क्षण की गिरावट वहीं तक पैदा हुई ऊष्मा है।}
\end{omfigure}

\begin{method}[ऊर्जा का बहीखाता]\label{met:g11:mechanical-energy:bookkeeping}
\begin{enumerate}
  \item एक \omterm{def:g11:mechanical-energy:potential}{संदर्भ स्तर} चुनिए और उसे बनाए रखिए; $A$ (आँकड़े) और
        $B$ (प्रश्न) के नाम रखिए।
  \item $A$ पर और $B$ पर $E_m = \tfrac12 mv^2 + mgz$ लिखिए।
  \item \omterm{def:g10:inertia:inventory}{घर्षण} न हो: $E_m(A) = E_m(B)$; हल कीजिए --- आम तौर पर $v_B = \sqrt{v_A^2 + 2g(z_A - z_B)}$।
  \item \omterm{def:g10:inertia:inventory}{घर्षण} हो: $E_m(B) - E_m(A) = -f\ell$ ($\ell$ = पथ-लंबाई); अज्ञात के लिए हल
        कीजिए।
  \item जाँच: $E_k$/$E_p$ के स्तंभ जुड़कर $E_m$ बनते हैं, और
        \omterm{def:g10:inertia:inventory}{घर्षण} न हो तो समतल रहते हैं।
\end{enumerate}
\end{method}

\section{समस्या: फंदे की डिज़ाइन}

\begin{problem}[{सप्ताहांत समस्या --- एक रोलर कोस्टर का ऊर्जा-लेखा: वह
छोड़-ऊँचाई जो फंदा तय कर देता है, वे चालें जिनका वह वादा करता है, वह घर्षण
जिसे दो संवेदक क़बूल कर लेते हैं, और वे ब्रेक जो खाता बंद करते हैं --- पूरी
सवारी एक ही संख्या से लटकी हुई}]
\label{pb:g11:mechanical-energy:1}
द्रव्यमान $m = \qty{2.0e3}{kg}$ की एक गाड़ी $H$ ऊँची छोड़-पहाड़ी से \omterm{def:g10:relative-motion:frame}{विराम} से छोड़ी
जाती है और बिना इंजन के त्रिज्या $R = \qty{8.0}{m}$ वाले उस \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} वृत्ताकार फंदे से
गुज़रती है जिसका शिखर ऊँचाई $2R$ पर है। वृत्तीय गति का विश्लेषण (अगला
वर्ष) हमें केवल एक आँकड़ा उधार देता है: फंदे के शिखर पर गाड़ी पटरी पर तभी दबाव
डालती है जब $v_{\text{शिखर}}^2 \geq gR$। भाग I और II में \omterm{def:g10:inertia:inventory}{घर्षण} छोड़ दिया गया है।

\medskip
\noindent\textbf{भाग I --- वह पहाड़ी जो फंदा तय करता है।}
\begin{enumerate}
  \item फंदे के शिखर पर न्यूनतम सुरक्षित चाल निकालिए।
  \item ज़मीन को \omterm{def:g11:mechanical-energy:potential}{संदर्भ स्तर} मानकर उस न्यूनतम चाल पर फंदे के शिखर
        पर गाड़ी की \omterm{def:g11:mechanical-energy:em}{यांत्रिक ऊर्जा} निकालिए।
  \item $H_{\min}$ निकालिए और दिखाइए कि $H_{\min} = 5R/2$ --- यानी न द्रव्यमान, न
        $g$।
  \item दल $H = \qty{24}{m}$ लेता है: फंदे के शिखर की असली चाल और अनुपात $v_{\text{शिखर}}^2/(gR)$
        निकालिए।
  \item $m$ कहाँ गया? यह डिज़ाइन भरी और ख़ाली, दोनों गाड़ियों पर एक-सा
        क्यों बैठता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ऊँचाई किसका वादा करती है} ($H = \qty{24}{m}$)।
\begin{enumerate}[resume]
  \item फंदे के तल ($z = 0$) पर चाल, \unit{m/s} और \unit{km/h} में?
  \item फंदे की बग़ल ($z = R$) पर चाल?
  \item छोड़-बिंदु, फंदे के तल, बग़ल और शिखर पर $E_p$ तथा $E_k$
        निकालिए; जाँचिए कि हर जोड़ी का योग $mgH$ बनता है।
  \item फंदे के बाद \qty{12}{m} का ऊँट-कूबड़ आता है: उसकी चोटी पर चाल?
  \item $v(z)$ का सामान्य सूत्र दीजिए; सवारी कहाँ सबसे तेज़ है, और कहाँ सबसे
        धीमी?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g10:inertia:inventory}{घर्षण} का लेखा।}
एक ही ऊँचाई $z = \qty{2.0}{m}$ पर, पटरी के \qty{60}{m} की दूरी पर लगे दो संवेदक पहले
$v_1 = \qty{20.4}{m/s}$ और फिर $v_2 = \qty{19.2}{m/s}$ पढ़ते हैं।
\begin{enumerate}[resume]
  \item दोनों संवेदकों के बीच \omterm{def:g11:mechanical-energy:em}{यांत्रिक ऊर्जा} का बदलाव निकालिए।
        दोनों को एक ही ऊँचाई पर रखना चतुराई क्यों थी?
  \item \omterm{def:g10:inertia:inventory}{घर्षण} का कार्य और औसत घर्षण-बल
        निकालिए।
  \item ग़ायब ऊर्जा कहाँ गई? उसका रूप और उसकी जगहें बताइए।
  \item प्रति \qty{60}{m} गाड़ी का $E_m$ का कितना अंश खो जाता है? असली, रगड़ती
        दुनिया में नंगा $H_{\min}$ असुरक्षित क्यों होगा?
  \item छोड़-बिंदु और फंदे के शिखर के बीच लगभग \qty{90}{m} पटरी है: वहाँ $E_m$
        का अनुमान लगाइए; क्या प्रश्न 4 वाला हाशिया अब भी फंदे को बचा लेता
        है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- खाता बंद करना।}
सीधे क्षैतिज ब्रेक-पथ के मुहाने ($z = 0$) पर लगा आख़िरी संवेदक \qty{18.0}{m/s} पढ़ता
है।
\begin{enumerate}[resume]
  \item वहाँ गाड़ी की \omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} निकालिए।
  \item ब्रेकों को उसे $d = \qty{45}{m}$ में रोकना है: ज़रूरी (अचर) ब्रेक-बल
        निकालिए।
  \item आराम की जाँच: मंदन $a = F/m$ निकालिए; $g$ से तुलना कीजिए।
  \item पूरी सवारी में पटरी के \omterm{def:g10:inertia:inventory}{घर्षण} ने कितनी ऊर्जा \omterm{def:g11:mechanical-energy:dissipation}{क्षय}
        की? जाँचिए कि \omterm{def:g10:inertia:inventory}{घर्षण} $+$ ब्रेक $= mgH$, \omterm{def:g11:work-of-force:work}{जूल} तक
        सही बैठता है।
  \item अंतिम चोट, एक वाक्य में: वह कौन-सी अकेली संख्या थी जिसने फंदे की
        सुरक्षा, हर चाल, \omterm{def:g10:inertia:inventory}{घर्षण} का बिल और ब्रेक-बल, सब तय
        कर दिए --- और \omterm{def:g10:inertia:inventory}{घर्षण} को अनदेखा करने का दाम क्या रहा?
\end{enumerate}
\end{problem}
