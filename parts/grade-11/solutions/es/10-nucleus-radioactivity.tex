\section*{Capítulo \ref{ch:g11:nucleus-radioactivity} --- El núcleo y la radiactividad}

\begin{solution}{exo:g11:nucleus-radioactivity:1}
${}^{16}_{8}\mathrm{O}$: $8$ p, $8$ n; ${}^{27}_{13}\mathrm{Al}$: $13$
p, $14$ n; ${}^{60}_{27}\mathrm{Co}$: $27$ p, $33$ n;
${}^{235}_{92}\mathrm{U}$: $92$ p, $143$ n; ${}^{3}_{1}\mathrm{H}$: $1$
p, $2$ n. ($N = A - Z$ en todos los casos.)
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:2}
Los tres carbonos ($Z = 6$) son \omterm{def:g11:nucleus-radioactivity:notation}{isótopos} entre sí.
${}^{14}_{6}\mathrm{C}$ y ${}^{14}_{7}\mathrm{N}$ comparten $A = 14$
pero no $Z$: son elementos distintos (con nubes de electrones
distintas), así que no son \omterm{def:g11:nucleus-radioactivity:notation}{isótopos}.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:3}
$A = 210 - 4 = 206$, $Z = 84 - 2 = 82$:
${}^{210}_{84}\mathrm{Po} \to {}^{206}_{82}\mathrm{Pb} +
{}^{4}_{2}\mathrm{He}$ --- el hijo es el plomo-206.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:4}
${}^{60}_{27}\mathrm{Co} \to {}^{60}_{28}\mathrm{Ni} +
{}^{\;0}_{-1}\mathrm{e}$ y
${}^{131}_{53}\mathrm{I} \to {}^{131}_{54}\mathrm{Xe} +
{}^{\;0}_{-1}\mathrm{e}$: $A$ no cambia y $Z$ sube en una \omterm{def:g10:orders-of-magnitude:unit}{unidad}.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:5}
$16$ días $= 2\,T_{1/2}$: $800/4 = \qty{200}{MBq}$. $32$ días
$= 4\,T_{1/2}$: $800/16 = \qty{50}{MBq}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:6}
(a) $\alpha$ (pesada, y por eso apenas se desvía); (b) $\beta^-$; (c)
$\gamma$ (neutra y penetrante); (d) $\beta^+$ (un \omterm{rem:g11:nucleus-radioactivity:betaplus}{positrón}: tan ligero
como (b), con carga opuesta).
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:7}

{}^{4}_{2}\mathrm{He}$; \
${}^{218}_{84}\mathrm{Po} \to {}^{214}_{82}\mathrm{Pb} +
{}^{4}_{2}\mathrm{He}$; \
${}^{214}_{82}\mathrm{Pb} \to {}^{214}_{83}\mathrm{Bi} +

\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:8}
El flúor-18 tiene $9$ neutrones donde el flúor-19 estable tiene $10$:
pobre en neutrones, por debajo de la banda, así que $\beta^+$:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:9}
$24$ horas $= 4\,T_{1/2}$: $500/16 \approx \qty{31}{MBq}$. Como
$2^9 = 512$, nueve \omterm{def:g10:signals-and-waves:period}{periodos} llevan a $500/512 < \qty{1}{MBq}$: al cabo
de $9 \times 6 = \qty{54}{h}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:10}
$\frac18 = \frac1{2^3}$: tres \omterm{def:g10:signals-and-waves:period}{periodos},
$3 \times \num{5700} = \num{17100}$ años. Un hueso de dinosaurio tiene
unos $\num{17500}$ \omterm{def:g10:signals-and-waves:period}{periodos} de antigüedad: queda una fracción
$1/2^{17500}$ --- ni un solo \omterm{def:g11:fundamental-interactions:ladder}{átomo} de carbono-14 (un gramo de carbono
contiene apenas unos $\num{5e22}$ \omterm{def:g11:fundamental-interactions:ladder}{átomos}).
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:11}
${}^{14}_{6}\mathrm{C}$: $8$ neutrones frente a los $6$--$7$ del carbono
estable, por encima de la banda, $\beta^-$. ${}^{11}_{6}\mathrm{C}$: $5$
neutrones, pobre en neutrones, $\beta^+$. ${}^{238}_{92}\mathrm{U}$:
$Z = 92 > 83$, demasiado pesado, $\alpha$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:12}
Solo la $\alpha$ cambia $A$: $238 - 206 = 32 = 4x$, luego $x = 8$
partículas \omterm{def:g11:nucleus-radioactivity:abg}{alfa}. Estas le quitan $16$ a $Z$: de $92 - 16 + y = 82$ sale
$y = 6$ desintegraciones \omterm{def:g11:nucleus-radioactivity:abg}{beta} menos.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:13}
$6.4/0.1 = 64 = 2^{6}$: seis \omterm{def:g10:signals-and-waves:period}{periodos},
$6 \times 5.3 = 31.8 \approx 32$ años en almacenamiento blindado.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:14}
Al día: $8000 \times 86400 \approx \num{6.9e8}$ desintegraciones. En
$80$ años: $\num{6.9e8} \times 365 \times 80 \approx \num{2e13}$. La
vida siempre ha funcionado con este fondo --- un \omterm{def:g11:nucleus-radioactivity:activity}{becquerelio} es un \omterm{def:g11:fundamental-interactions:ladder}{átomo}
por segundo de entre los $\sim 10^{27}$ del cuerpo; la \omterm{def:g10:orders-of-magnitude:unit}{unidad} es
diminuta, y un recuento de \omterm{def:g11:nucleus-radioactivity:activity}{becquerelios} que suene aterrador puede
equivaler a unos pocos plátanos.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:15}
$\num{25000} \times 86400 \approx \num{2.2e9}$ desintegraciones al día.
Diez años son $10/432 \approx 2\,\%$ de un \omterm{def:g10:signals-and-waves:period}{periodo}: se ha desintegrado
mucho menos de la mitad, así que lo que obliga a cambiarlo es la
electrónica y el polvo, no la fuente. Por fuera, las $\alpha$ mueren en
el aire de la cámara y en la carcasa; el polvo inhalado aparca el emisor
en los pulmones, donde cada $\alpha$ vuelca su \omterm{def:g10:energy-conservation:energy}{energía} en tejido vivo
--- una \omterm{rem:g11:nucleus-radioactivity:dose}{dosis} grande (en \omterm{rem:g11:nucleus-radioactivity:dose}{sieverts}) a partir de una \omterm{def:g11:nucleus-radioactivity:activity}{actividad} modesta.
\end{solution}

\begin{solution}{pb:g11:nucleus-radioactivity:1}
\textbf{1.} ${}^{12}_{6}\mathrm{C}$: $6$ p, $6$ n;
${}^{14}_{6}\mathrm{C}$: $6$ p, $8$ n. La química solo ve la nube de
electrones, fijada por $Z = 6$: comportamiento idéntico.

\textbf{2.} Comer y respirar renuevan constantemente el carbono de un
cuerpo vivo con la proporción atmosférica; al morir cesa la entrada y la
desintegración actúa sin oposición.

\textbf{3.} ${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} +
{}^{\;0}_{-1}\mathrm{e}$ ($\beta^-$). Esas $\beta$ son blandas: unos
\omterm{def:g10:orders-of-magnitude:unit}{metros} de lino enrollado las detienen, así que poco escapa de la momia
--- por eso se mide una pequeña muestra de carbono tomada del propio
objeto.

\textbf{4.} $6.8/13.6 = \frac12$: un \omterm{def:g10:signals-and-waves:period}{periodo}. La momia tiene unos
\num{5700} años.

\textbf{5.} $13.2/13.6 \approx 0.97$: el lino es esencialmente moderno
--- como mucho de unos pocos siglos. Una falsificación.

\textbf{6.} $13.6/2^{10} = 13.6/1024 \approx 0.013$ desintegraciones por
minuto --- un recuento cada $75$ minutos y gramo, ahogado en el fondo
natural. Diez \omterm{def:g10:signals-and-waves:period}{periodos}, unos \num{57000} años, es el horizonte
práctico.

\textbf{7.} $92$ protones y $146$ neutrones. La repulsión de Coulomb
actúa entre todas las parejas de protones del \omterm{def:g11:fundamental-interactions:ladder}{núcleo}, mientras que el
pegamento fuerte solo liga a los vecinos: los \omterm{def:g11:fundamental-interactions:ladder}{núcleos} pesados solo
sobreviven con un excedente de neutrones, que son pegamento sin
repulsión.

\textbf{8.} ${}^{238}_{92}\mathrm{U} \to {}^{234}_{90}\mathrm{Th} +
{}^{4}_{2}\mathrm{He}$.

\textbf{9.} ${}^{234}_{90}\mathrm{Th} \to {}^{234}_{91}\mathrm{Pa} +
{}^{\;0}_{-1}\mathrm{e}$ y después
${}^{234}_{91}\mathrm{Pa} \to {}^{234}_{92}\mathrm{U} +
{}^{\;0}_{-1}\mathrm{e}$: reaparece el uranio, como uranio-234.

\textbf{10.} Por $A$: $238 - 206 = 32$, luego $8$ pasos $\alpha$; por
$Z$: $92 - 16 + y = 82$, luego $6$ pasos $\beta^-$.

\textbf{11.} $\num{4.5e9}$ años son un \omterm{def:g10:signals-and-waves:period}{periodo}: queda $\frac12$. Cinco
mil millones de años más $\approx$ otro \omterm{def:g10:signals-and-waves:period}{periodo}: alrededor de
$\frac14$.

\textbf{12.} Impulsa los volcanes, la tectónica de placas (la deriva de
los continentes, los terremotos) y el calor geotérmico: la geología
superficial de un planeta calentado desde dentro.

\textbf{13.} ${}^{241}_{95}\mathrm{Am} \to {}^{237}_{93}\mathrm{Np} +
{}^{4}_{2}\mathrm{He}$.

\textbf{14.} $\num{3.3e4}$ desintegraciones por segundo;
$\num{33000} \times 86400 \approx \num{2.9e9}$ al día.

\textbf{15.} Cada $\alpha$ ioniza el aire que atraviesa; los iones
transportan una corriente diminuta entre los electrodos de la cámara.
Las partículas de humo capturan los iones, la corriente cae y salta la
alarma.

\textbf{16.} Las $\alpha$ gastan toda su \omterm{def:g10:energy-conservation:energy}{energía} dentro de la cámara y
ninguna escapa de la carcasa: ionización máxima y fuga nula. Una fuente
$\gamma$ de la misma \omterm{def:g11:nucleus-radioactivity:activity}{actividad} apenas ionizaría el aire de la cámara y,
en cambio, irradiaría la habitación entera.

\textbf{17.} Diez años son $10/432 \approx 2\,\%$ de un \omterm{def:g10:signals-and-waves:period}{periodo}: la
\omterm{def:g11:nucleus-radioactivity:activity}{actividad} no ha cambiado prácticamente nada. Los eslabones débiles son
el polvo, los insectos y la electrónica --- el \omterm{def:g11:fundamental-interactions:ladder}{núcleo} sobrevive al
aparato.

\textbf{18.} $\num{9e9}$ años $= 2$ \omterm{def:g10:signals-and-waves:period}{periodos}: sobrevive $\frac14$. El
uranio de la Tierra sigue siendo abundante, así que se forjó como mucho
hace unos pocos \omterm{def:g10:signals-and-waves:period}{periodos} --- el planeta se condensó a partir de ceniza
estelar fresca, miles de millones de años después de las primeras
estrellas.

\textbf{19.} Geológicamente muerto: un interior frío, sin vulcanismo ni
tectónica de placas que reciclen aire y roca, y sin un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} agitado
--- y, por tanto, sin escudo magnético frente al viento solar.

\textbf{20.} Todos tus \omterm{def:g11:fundamental-interactions:ladder}{átomos} más pesados que el helio se forjaron en
una estrella moribunda, y los restos inestables de aquella fragua
todavía calientan el suelo que pisas: polvo de estrellas, mantenido vivo
por su propio \omterm{def:g11:nucleus-radioactivity:halflife}{periodo de semidesintegración}.
\end{solution}
