\section*{الفصل \ref{ch:g11:circuits-and-power} --- الدارات الكهربائية والاستطاعة}

\begin{solution}{exo:g11:circuits-and-power:1}
$q = It = 2.0 \times 90 \times 60 = \qty{1.1e4}{C}$. وعدد الشحنات
الأولية: $q/e = 1.08\times10^{4}/1.6\times10^{-19} \approx \num{6.8e22}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:2}
$I = U/R = 12/220 = \qty{0.055}{A} = \qty{55}{mA}$؛
$R = U/I = 230/0.50 = \qty{460}{\ohm}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:3}
على التسلسل: $120 + 60 = \qty{180}{\ohm}$. وعلى التفرع: $\frac{120 \times 60}{180}
= \qty{40}{\ohm}$ --- أي أصغر من كلتيهما: \omterm{def:g11:circuits-and-power:voltage}{فالتوتر} نفسه، لكن التيار
يجد مسارًا ثانيًا، فيزداد التيار الكلي وتقلّ \omterm{def:g11:circuits-and-power:resistance}{المقاومة}.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:4}
$P = UI = 230 \times 8.7 = \qty{2.0}{kW}$. وفي $t = \qty{240}{s}$:
$E = Pt = 2000 \times 240 = \qty{4.8e5}{J} = 4.8\times10^{5}/3.6
\times10^{6} \approx \qty{0.13}{kW.h}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:5}
$E = 60 \times 5.0 = \qty{300}{W.h} = \qty{0.30}{kW.h} = 0.30 \times
3.6\times10^{6} = \qty{1.1e6}{J}$. والكلفة: $0.30 \times 0.25 = 0.075$
يورو في اليوم.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:6}
\emph{1.} $R = U^2/P = 230^2/1200 = \qty{44}{\ohm}$؛
$I = P/U = 1200/230 = \qty{5.2}{A}$.

\emph{2.} $P' = U'^2/R = 115^2/44.1 = \qty{300}{W}$: أي الربع، لا
النصف، لأن تنصيف $U$ ينصّف كذلك $I$، و$P = UI \propto U^2$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:7}
$R_{\text{s}} = \qty{60}{\ohm}$، ومنه $I = 12/60 = \qty{0.20}{A}$.
و$U_1 = R_1 I = \qty{8.0}{V}$، $U_2 = \qty{4.0}{V}$ (والمجموع \qty{12}{V})؛
و$P_1 = U_1 I = \qty{1.6}{W}$، $P_2 = \qty{0.8}{W}$: والمجموع \qty{2.4}{W}
$= UI = 12 \times 0.20$، كما يجب.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:8}
$I_1 = 12/60 = \qty{0.20}{A}$، $I_2 = 12/30 = \qty{0.40}{A}$، والمجموع
$I = \qty{0.60}{A}$. والمكافئة: $R = U/I = 12/0.60 = \qty{20}{\ohm}$،
وفعلًا $\frac{60 \times 30}{60 + 30} = \qty{20}{\ohm}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:9}
$\mathcal{E} - 0.50\,r = 5.5$ و$\mathcal{E} - 2.0\,r = 4.0$؛
وبالطرح، $1.5\,r = 1.5$، ومنه $r = \qty{1.0}{\ohm}$ و$\mathcal{E} =
\qty{6.0}{V}$. وفي القصر: $\mathcal{E}/r = \qty{6.0}{A}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:10}
الغلاية $2200/230 = \qty{9.6}{A}$، والمحمصة $1000/230 = \qty{4.3}{A}$:
والمجموع \qty{13.9}{A} $< \qty{16}{A}$، فيصمد \omterm{def:g11:circuits-and-power:joule}{المصهر}. ومع المدفأة $1500/230 =
\qty{6.5}{A}$: يصير المجموع \qty{20.4}{A} $> \qty{16}{A}$، فينصهر
(أو بالمكافئ، $\qty{4.7}{kW} > 230 \times 16 \approx \qty{3.7}{kW}$).
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:11}
عند \qty{200}{V}: $I = 10^4/200 = \qty{50}{A}$، والضياع $RI^2 = 2.0 \times
50^2 = \qty{5.0}{kW}$ --- أي نصف ما يُوصَّل. وعند \qty{4.0}{kV}: $I =
\qty{2.5}{A}$، والضياع $2.0 \times 2.5^2 = \qty{12.5}{W}$. والنسبة $400 =
20^2$: فعشرون ضعف \omterm{def:g11:circuits-and-power:voltage}{التوتر} تعني تيارًا أقلّ عشرين مرة، و
الضياع يتغير مثل $I^2$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:12}
أربعة تركيبات: الثلاث على التسلسل، $3 \times 60 = \qty{180}{\ohm}$؛ والثلاث على
التفرع، $60/3 = \qty{20}{\ohm}$؛ وواحدة على التسلسل مع اثنتين على التفرع،
$60 + 30 = \qty{90}{\ohm}$؛ وواحدة على التفرع مع اثنتين على التسلسل،
$\frac{60 \times 120}{180} = \qty{40}{\ohm}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:13}
\emph{1.} $I = \mathcal{E}/(R + r) = 4.5/7.5 = \qty{0.60}{A}$؛
$U = RI = \qty{3.6}{V}$؛ والمفيدة $P = UI = \qty{2.2}{W}$؛ والضائعة $rI^2 =
1.5 \times 0.60^2 = \qty{0.54}{W}$؛ و$\eta = U/\mathcal{E} = 3.6/4.5 =
0.80$.

\emph{2.} بالتلميح، $P = \mathcal{E}^2 \big/ \big[(R - r)^2/R +
4r\big]$؛ والمقدار بين القوسين أصغر ما يكون ($= 4r$) حين $R = r$ بالضبط، ومنه
$P_{\max} = \mathcal{E}^2/(4r) = 4.5^2/6.0 = \qty{3.4}{W}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:14}
الحرارة: $Q = 1.0 \times 4180 \times (100 - 20) = \qty{3.3e5}{J}$.
والكهربائية: $E = Q/0.90 = \qty{3.7e5}{J}$. والزمن: $t = E/P =
3.7\times10^{5}/2200 \approx \qty{170}{s}$، أي دون ثلاث دقائق. والكلفة:
$3.7\times10^{5}/3.6\times10^{6} \approx \qty{0.10}{kW.h}$، أي نحو $2.6$
سنتات.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:15}
\emph{1.} $q = 60 \times 3600 = \qty{2.2e5}{C}$؛ $E \approx \mathcal{E}q
= 12.7 \times 2.16\times10^{5} = \qty{2.7e6}{J} \approx \qty{0.76}{kW.h}$.

\emph{2.} $U = 12.7 - 0.020 \times 150 = \qty{9.7}{V}$؛ والموصَّلة
$UI = \qty{1.5}{kW}$؛ والضائعة $rI^2 = 0.020 \times 150^2 = \qty{450}{W}$.

\emph{3.} تقع المصابيح الأمامية بين المربطين: وأثناء الإدارة يهبط
\omterm{def:g11:circuits-and-power:voltage}{التوتر} بين المربطين من $12.7$ إلى $\qty{9.7}{V}$، فتتلقى
\omterm{def:g10:energy-conservation:power}{استطاعة} أقلّ وتخفت.
\end{solution}

\begin{solution}{pb:g11:circuits-and-power:1}
\textbf{1.} في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}، $E_p = mgh = 5.0\times10^{4} \times 9.81 \times
200 = \qty{9.8e7}{J}$: \omterm{def:g10:energy-conservation:power}{فالاستطاعة} المتاحة \qty{98}{MW}.

\textbf{2.} $0.90 \times 98 = \qty{88}{MW}$.

\textbf{3.} $I = P/U = 8.8\times10^{7}/2.0\times10^{4} = \qty{4.4e3}{A}$.

\textbf{4.} $R_\ell I^2 = 32 \times (4.4\times10^{3})^2 \approx
\qty{6.2e8}{W} = \qty{620}{MW}$ --- أي سبعة أضعاف خرج المحطة:
فالنقل عند \qty{20}{kV} مستحيل.

\textbf{5.} مع بقاء $P = UI$ ثابتة، لا يقلّص $I$ إلا رفع $U$: أي
محوّلة.

\textbf{6.} $I = 8.8\times10^{7}/4.0\times10^{5} = \qty{220}{A}$.

\textbf{7.} $R_\ell I^2 = 32 \times 220^2 \approx \qty{1.6}{MW}$.

\textbf{8.} $1.6/88 \approx 0.018$: أي نحو $1.8\%$.

\textbf{9.} الجزء $= R_\ell I^2/P = R_\ell (P/U)^2/P = R_\ell P/U^2
\propto 1/U^2$: فمن \qty{20}{kV} إلى \qty{400}{kV} يهبط بمقدار
$20^2 = 400$، وفعلًا $\qty{620}{MW}/400 \approx \qty{1.6}{MW}$.

\textbf{10.} $R_\ell I = 32 \times 220 \approx \qty{7.1}{kV}$: أي $1.8\%$
من \qty{400}{kV}، وهو الجزء نفسه في السؤال 8، لأن
$R_\ell I/U = R_\ell I^2/UI$.

\textbf{11.} $\qty{1}{kW.h} = \qty{3.6e6}{J}$: فالعدّاد يضرب
\omterm{def:g10:energy-conservation:power}{الاستطاعة} المسحوبة في زمن سحبها، ثم يجمع.

\textbf{12.} $I = 900/230 = \qty{3.9}{A}$؛
$R = U^2/P = 230^2/900 \approx \qty{59}{\ohm}$.

\textbf{13.} $E = 900 \times 180 = \qty{1.6e5}{J} = \qty{0.045}{kW.h}$:
أي نحو $1.1$ سنتات.

\textbf{14.} فيزياء $RI^2$ واحدة؛ إنما تختلف الغاية
--- فعلى الخط تكون الحرارة غير مرغوبة، وفي المحمصة تكون الحرارة
\emph{هي} المنتَج.

\textbf{15.} $3.9 + 9.6 = \qty{13.5}{A} < \qty{16}{A}$: يصمد. ومع المدفأة
$2000/230 = \qty{8.7}{A}$: يصير المجموع $\qty{22.2}{A} > \qty{16}{A}$، فينصهر
\omterm{def:g11:circuits-and-power:joule}{المصهر}.

\textbf{16.} $0.05 \times 13.5^2 = \qty{9.1}{W}$: أي $9.1/3100 \approx
0.3\%$ من \omterm{def:g10:energy-conservation:power}{الاستطاعة} الموصَّلة --- وهو مهمَل على أسلاك قصيرة.

\textbf{17.} $\eta = 0.90 \times 0.99 \times 0.982 \times 0.99 \times
0.997 \approx 0.86$.

\textbf{18.} يحتاج \omterm{def:g10:energy-conservation:kwh}{كيلوواط ساعة} محمَّص واحد $3.6\times10^{6}/0.86
\approx \qty{4.2e6}{J}$ في المنبع: أي $m = E/(gh) = 4.2\times10^{6}/(9.81
\times 200) \approx \qty{2.1e3}{kg}$، ونحو \qty{2.1}{m^3} من الماء.

\textbf{19.} $R_\ell P/U^2 = 32 \times 8.8\times10^{7}/(2.0\times
10^{4})^2 \approx 7$. و``جزء'' يفوق $1$ يعني أن الخط سيبدّد
أكثر مما يحمل: أي أن التوصيل مستحيل عند ذلك
\omterm{def:g11:circuits-and-power:voltage}{التوتر} أصلًا.

\textbf{20.} يكلّف \omterm{def:g10:energy-conservation:kwh}{كيلوواط ساعة} محمَّص واحد الجبلَ نحو طنّين
--- أي \qty{2.1}{m^3} --- من الماء الساقط من \qty{200}{m}، ويموت نحو
$14\%$ من \omterm{def:g10:energy-conservation:energy}{الطاقة} في الطريق، ومعظمه في العنفات والمحوّلات
(ودون $2\%$ في \qty{1000}{km} من الخط). أما الحيلة الواحدة التي تجعل
الرحلة ممكنة فهي المحوّلة: فالرفع إلى \qty{400}{kV}
يقسم التيار على $20$ ويقسم ضياع جول على $400$.
\end{solution}
