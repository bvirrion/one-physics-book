\begin{solution}{exo:g11:work-of-force:1}
$W = 40 \times 50 \times \cos\theta$: $+\qty{2000}{J}$؛
$+\qty{1000}{J}$؛ $0$؛ $-\qty{1000}{J}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:2}
$mgh = 12 \times 9.81 \times 2.5 = \qty{294}{J}$: عند الإنزال،
$+\qty{294}{J}$ (\omterm{def:g10:universal-gravitation:weight}{فالوزن} في جهة الهبوط)؛ وعند الرفع، $-\qty{294}{J}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:3}
(أ) \omterm{def:g11:work-of-force:sign}{محرِّك}؛ (ب) \omterm{def:g11:work-of-force:sign}{مقاوم}؛ (ج) معدوم (لأنه عمودي)؛ (د) \omterm{def:g11:work-of-force:sign}{مقاوم}؛
(ه) معدوم --- \omterm{def:g10:inertia:inventory}{فشدّ} الخيط يبقى عموديًّا على حركة

\end{solution}

\begin{solution}{exo:g11:work-of-force:4}
عند السرعة الثابتة: \omterm{def:g10:inertia:force}{قوة} الرفع $= mg = \qty{5.89e3}{N}$، ومنه
$W_{\text{الرفع}} = 5886 \times 25 \approx +\qty{147}{kJ}$ و
$W(\vect P) = -\qty{147}{kJ}$. والمجموع منعدم: فعند السرعة الثابتة
تتوازن القوى (\cref{ch:g11:forces-and-motion})، ومن ثم يتلاشى
شغلاهما.
\end{solution}

\begin{solution}{exo:g11:work-of-force:5}
$P = \num{9.0e4}/60 = \qty{1.5}{kW}$؛ $1500/736 \approx \qty{2.0}{hp}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:6}
الحبل: $120 \times 200 \times \cos\qty{30}{\degree} \approx
+\qty{20.8}{kJ}$؛ \omterm{def:g10:inertia:inventory}{والاحتكاك}: $-45 \times 200 = -\qty{9.0}{kJ}$؛ \omterm{def:g10:universal-gravitation:weight}{والوزن}
\omterm{def:g10:inertia:inventory}{ورد الفعل الناظمي}، وهما عموديان: $0$. والمجموع $+\qty{11.8}{kJ}$: فتكتسب
الزحّافة سرعة.
\end{solution}

\begin{solution}{exo:g11:work-of-force:7}
$v = 25/3.6 = \qty{6.94}{m/s}$؛ $P = Fv = 18 \times 6.94 \approx
\qty{125}{W}$. وفي ساعة واحدة، $W = P\,\Delta t = 125 \times 3600 =
\qty{450}{kJ}$ (أو بالمكافئ $18 \times \qty{25000}{m}$).
\end{solution}

\begin{solution}{exo:g11:work-of-force:8}
\omterm{def:g10:universal-gravitation:weight}{الوزن}، على الطريقين: $+65 \times 9.81 \times 120 \approx
+\qty{76.5}{kJ}$ (فالهبوط وحده). \omterm{def:g10:inertia:inventory}{والاحتكاك}: $-30 \times 800 =
-\qty{24}{kJ}$ على المنحدر، و$-30 \times 1500 = -\qty{45}{kJ}$ على
الدرب. \omterm{def:g10:inertia:inventory}{فالاحتكاك} وحده يهتمّ بالطريق.
\end{solution}

\begin{solution}{exo:g11:work-of-force:9}
الصعود $h = 4.0\sin\qty{25}{\degree} = \qty{1.69}{m}$:
$W(\vect P) = -20 \times 9.81 \times 1.69 \approx -\qty{332}{J}$؛
$W_f = -30 \times 4.0 = -\qty{120}{J}$. وعند السرعة الثابتة يكون \omterm{def:g11:work-of-force:work}{الشغل} الكلي
منعدمًا، ومنه $W_{\text{الدفعة}} = +\qty{452}{J}$ و
$F = 452/4.0 \approx \qty{113}{N}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:10}
$v = 130/3.6 = \qty{36.1}{m/s}$؛ وبسرعة ثابتة، ومنه
$F = P/v = \num{5.0e4}/36.1 \approx \qty{1.4}{kN}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:11}
$P = Fv\cos\theta = 800 \times 1.0 \times \cos\qty{20}{\degree}
\approx \qty{752}{W}$؛ وبالحصان، $752/736 \approx \num{1.02}$ ---
أي تقريبًا بالضبط ما يحافظ عليه \omterm{def:g11:work-of-force:power}{حصان} قوي، وهي المقارنة التي بنى
\omterm{def:g11:work-of-force:power}{واط} عليها وحدته.
\end{solution}

\begin{solution}{exo:g11:work-of-force:12}
الأطوال $L = h/\text{الميل}$: أي \qty{1500}{m} و\qty{5000}{m}. والقوتان
$F = mg \times \text{الميل}$: أي $1200 \times 9.81 \times 0.20 \approx
\qty{2.35}{kN}$ و$\qty{0.71}{kN}$. والشغلان $FL$: أي $2354 \times 1500
\approx \qty{3.53}{MJ}$ و$706 \times 5000 \approx \qty{3.53}{MJ}$
--- وكلاهما يساوي $mgh$: فالطريق تقايض \omterm{def:g10:inertia:force}{القوة} بالطول، لا \omterm{def:g11:work-of-force:work}{الشغل} أبدًا.
\end{solution}

\begin{solution}{exo:g11:work-of-force:13}
$F = mg = \qty{2.45e3}{N}$؛ $W = 2453 \times 15 \approx \qty{36.8}{kJ}$؛
$\Delta t = 15/0.40 = \qty{37.5}{s}$؛
$P = 36800/37.5 \approx \qty{981}{W} = 2453 \times 0.40$ (ويوافق $Fv$)؛
والكهربائية: $981/0.70 \approx \qty{1.4}{kW}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:14}
القُطر: $-140 \times 6.0 = -\qty{840}{J}$؛ والجداران:
$-140 \times 8.4 \approx -\qty{1.18}{kJ}$؛ ورحلة الذهاب والإياب:
$-140 \times 12 = -\qty{1.68}{kJ}$. أما \omterm{def:g10:universal-gravitation:weight}{الوزن} فينجز صفرًا في رحلة الذهاب
والإياب (فلا هبوط محصّل). فشغل \omterm{def:g10:inertia:inventory}{الاحتكاك} يتوقف على \omterm{def:g10:relative-motion:trajectory}{المسار} ولا يردّ
أبدًا؛ وشغل \omterm{def:g10:universal-gravitation:weight}{الوزن} لا يتوقف إلا على الطرفين
(\cref{rem:g11:work-of-force:divide}).
\end{solution}

\begin{solution}{exo:g11:work-of-force:15}
\omterm{def:g11:fundamental-interactions:four}{الجاذبية}: $80 \times 9.81 \times 600 \approx \qty{471}{kJ}$؛ \omterm{def:g10:inertia:inventory}{والاحتكاك}:
$14 \times 12000 = \qty{168}{kJ}$؛ والمجموع $\qty{639}{kJ}$.
و$v = \qty{4.17}{m/s}$، ومنه $\Delta t = 12000/4.17 = \qty{2880}{s} =
\qty{48}{min}$ و$P = \num{6.39e5}/2880 \approx \qty{222}{W} \approx
\qty{0.30}{hp}$ --- أي هاوٍ قوي، بثلث \omterm{def:g11:work-of-force:power}{حصان}.
\end{solution}

\begin{solution}{pb:g11:work-of-force:1}
\textbf{1.} $W = 110 \times 8.0 = +\qty{880}{J}$.

\textbf{2.} \omterm{def:g10:inertia:inventory}{الاحتكاك}: $-95 \times 8.0 = -\qty{760}{J}$. أما \omterm{def:g10:universal-gravitation:weight}{الوزن} و
\omterm{def:g10:inertia:inventory}{رد الفعل الناظمي} فعموديان على الحركة: صفر لكلٍّ.

\textbf{3.} المجموع $+\qty{120}{J}$: أي \omterm{def:g11:work-of-force:sign}{شغل محرِّك} محصّل، فيتسارع الصندوق.
ودفعة قدرها \qty{95}{N} توازن \omterm{def:g10:inertia:inventory}{الاحتكاك} --- \omterm{def:g11:forces-and-motion:netforce}{فقوة محصّلة} منعدمة، وسرعة
ثابتة، وشغل كلي منعدم.

\textbf{4.} القُطر: $-95 \times 10.0 = -\qty{950}{J}$؛ والجداران:
$-95 \times 14.0 = -\qty{1.33}{kJ}$: \omterm{def:g10:inertia:inventory}{فالاحتكاك} يحاسب \omterm{def:g10:orders-of-magnitude:unit}{بالمتر}.

\textbf{5.} \omterm{def:g10:inertia:inventory}{الاحتكاك}: $-95 \times 20.0 = -\qty{1.9}{kJ}$؛ \omterm{def:g10:universal-gravitation:weight}{والوزن}:
$0$ (فلا هبوط محصّل). \omterm{def:g10:universal-gravitation:weight}{والوزن} وحده يردّ.

\textbf{6.} $W(\vect P) = -30 \times 9.81 \times 9.0 =
-\qty{2.65}{kJ}$؛ ويجب أن يوفّر الطاقم $+\qty{2.65}{kJ}$ على الأقل.

\textbf{7.} $(70 + 30) \times 9.81 \times 9.0 = \qty{8.83}{kJ}$؛ وحصة
الصندوق $2.65/8.83 \approx 30\%$ --- فالحمل نقل ذاتي في معظمه.

\textbf{8.} $\sin\alpha = 9.0/30 = 0.30$:
$F = mg\sin\alpha + f = 88.3 + 40 = \qty{128}{N}$؛
$W = 128 \times 30 = \qty{3.85}{kJ}$ ($= 2.65 + 1.2$).

\textbf{9.} الرفع العمودي: $F = mg = \qty{294}{N}$،
$W = \qty{2.65}{kJ}$. فيشتري المنحدر \omterm{def:g10:inertia:force}{قوة} أصغر ($128$ مقابل
\qty{294}{N}) بثمن شغل أكبر: أي رسم \omterm{def:g10:inertia:inventory}{الاحتكاك} البالغ \qty{1.2}{kJ}.

\textbf{10.} $\Delta t = 9.0/0.50 = \qty{18}{s}$؛
$P = 2649/18 \approx \qty{147}{W}$.

\textbf{11.} $P = Fv = 294 \times 0.50 = \qty{147}{W}$ --- وهي نفسها.
والكهربائية: $147/0.60 \approx \qty{245}{W}$.

\textbf{12.} $1000 \times 0.24 = \qty{240}{m}$ و
$8000 \times 0.030 = \qty{240}{m}$: أي القمة نفسها.

\textbf{13.} $F = mg \times \text{الميل}$: الممرّ
$3500 \times 9.81 \times 0.24 \approx \qty{8.2}{kN}$؛ والطريق
$\approx \qty{1.0}{kN}$.

\textbf{14.} $8240 \times 1000 \approx \qty{8.24}{MJ}$ و
$1030 \times 8000 \approx \qty{8.24}{MJ}$ --- وكلاهما $= mgh = 3500
\times 9.81 \times 240$.

\textbf{15.} \omterm{def:g11:circuits-and-power:resistance}{مقاومة} التدحرج: $400 \times 1000 = \qty{0.40}{MJ}$
مقابل $400 \times 8000 = \qty{3.2}{MJ}$: فراحة الطريق اللطيف
تكلّف \qty{2.8}{MJ} من \omterm{def:g10:inertia:inventory}{الاحتكاك} الإضافي.

\textbf{16.} $v = \qty{10}{m/s}$. الطريق:
$(1030 + 400) \times 10 = \qty{14.3}{kW} \approx \qty{19}{hp}$؛ والممرّ:
$(8240 + 400) \times 10 = \qty{86.4}{kW} \approx \qty{117}{hp}$ ---
أي فوق \omterm{def:g10:energy-conservation:energy}{طاقة} شاحنة محمَّلة. وتتعرّج الطرق حتى تقدر المحركات المتواضعة (والإطارات)
على الصعود: \omterm{def:g11:work-of-force:work}{الشغل} نفسه، موزّعًا رقيقًا.

\textbf{17.} لكل صندوق: $880 + 2649 = \qty{3.53}{kJ}$؛ ومن أجل $60$ صندوقًا،
$60 \times 3529 \approx \qty{212}{kJ}$.

\textbf{18.} الدرج: $20 \times 70 \times 9.81 \times 9.0 \approx
\qty{124}{kJ}$. ومجموع اليوم $\approx \qty{336}{kJ} = \qty{0.34}{MJ}$.

\textbf{19.} $P = \num{3.36e5}/28800 \approx \qty{12}{W}$. والغلاية:
$\num{3.36e5}/2000 = \qty{168}{s}$ --- أي دون ثلاث دقائق لكدح
اليوم كله.

\textbf{20.} يدفع \omterm{def:g11:work-of-force:work}{الجول} ثمن \omterm{def:g10:inertia:force}{قوة} موصَّلة \emph{على امتداد} حركة
--- ولا يدفع شيئًا للجدار المدفوع ولا للصندوق المحمول، مهما يكن ذلك مرهقًا
(فالعضلات تحرق \omterm{def:g10:energy-conservation:forms}{طاقة كيميائية} حتى وهي تنجز شغلًا ميكانيكيًّا معدومًا). وبما أن
\omterm{def:g11:work-of-force:power}{استطاعة} الإنسان بضع عشرات من الواطات، يدع عمال النقل الآلات تعيد تشكيل
\omterm{def:g10:inertia:force}{القوة} --- ملفافًا ومنحدرًا وتعرّجًا --- بينما لا تتغير الفاتورة $mgh$
أبدًا.
\end{solution}
