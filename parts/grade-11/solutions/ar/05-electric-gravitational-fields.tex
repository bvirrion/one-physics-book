\section*{الفصل \ref{ch:g11:electric-gravitational-fields} --- المجال الكهربائي ومجال الجاذبية}

\begin{solution}{exo:g11:electric-gravitational-fields:1}
$E = \num{5.0e-3} / \num{2.0e-6} = \qty{2.5e3}{N/C}$، متجهًا شمالًا.
وعلى $q'$: $F = \num{4.0e-6} \times \num{2.5e3} = \qty{1.0e-2}{N}$،
متجهة \emph{جنوبًا} ($q' < 0$).
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:2}
\emph{1.} $E = \num{8.99e9} \times \num{5.0e-9} / 0.30^2
\approx \qty{5.0e2}{N/C}$، متجهًا بعيدًا عن $O$ ($Q > 0$).

\emph{2.} تنصيف $k\abs{Q}/d^2$ يقتضي مضاعفة $d^2$:
$d' = 0.30\sqrt{2} \approx \qty{0.42}{m}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:3}
$E = 600 / \num{3.0e-3} = \qty{2.0e5}{V/m}$؛
$F = eE = \num{1.60e-19} \times \num{2.0e5} = \qty{3.2e-14}{N}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:4}
$g_{\text{المريخ}} = P/m = 930/250 = \qty{3.7}{N/kg}$. ورائد الفضاء:
$P = 80 \times 3.72 \approx \qty{3.0e2}{N}$ (أي نحو $\qty{780}{N}$ على
الأرض).
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:5}
(أ) خطأ: فللمجال جهة واحدة في كل نقطة، ومن ثم لا تتقاطع الخطوط
أبدًا. (ب) خطأ: بل تغادر $+$ وتصل إلى $-$. (ج) صحيح: فالتزاحم
يشفّر الشدة.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:6}
\emph{1.} $d = \num{6.37e6} + \num{4.2e5} = \qty{6.79e6}{m}$، ومنه
$g = \num{6.67e-11} \times \num{5.97e24} / (\num{6.79e6})^2
\approx \qty{8.6}{N/kg}$ --- أي $88\%$ من القيمة السطحية.

\emph{2.} يطفون لأن المحطة والرائد يسقطان معًا (سقوطًا
حرًّا)، لا لأن \omterm{def:g11:fundamental-interactions:four}{الجاذبية} غائبة.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:7}
$E = m_e g / e = \num{9.11e-31} \times 9.81 / \num{1.60e-19}
\approx \qty{5.6e-11}{V/m}$: أي أضعف نحو $\num{4e14}$ مرة من
\omterm{def:g11:electric-gravitational-fields:field}{مجال} \omterm{def:g10:signals-and-waves:oscilloscope}{راسم الاهتزاز}. \omterm{def:g11:fundamental-interactions:four}{فالجاذبية} مهمَلة تمامًا في فيزياء الإلكترون.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:8}
\emph{1.} $E = \num{5.0e3} / \num{2.0e-2} = \qty{2.5e5}{V/m}$؛ وفي السكون
$qE = mg$، ومنه
$q = \num{2.0e-6} \times 9.81 / \num{2.5e5} \approx \qty{7.8e-11}{C}$.

\emph{2.} يجب أن تتجه \omterm{def:g10:inertia:force}{القوة} على الحبة الموجبة إلى أعلى، ومن ثم يتجه $\vect E$
إلى أعلى: فالصفيحة \emph{السفلى} هي الموجبة.

\emph{3.} $n = \num{7.85e-11} / \num{1.60e-19} \approx \num{4.9e8}$
\omterm{def:g11:fundamental-interactions:elementary}{شحنة أولية}.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:9}
خارج القطعة يتجه المجالان في الجهة نفسها؛ وبينهما يتعاكسان
--- ويقتضي التوازن أن تكون النقطة أقرب إلى المنبع الأضعف $+Q$.
وبأخذ $x$ مسافةً إلى $+Q$: $kQ/x^2 = 4kQ/(0.30 - x)^2$، ومنه
$0.30 - x = 2x$ و$x = \qty{0.10}{m}$ من $+Q$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:10}
$F_e = ke^2/d^2 = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$؛
$F_g = G m_p m_e / d^2 = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31} / (\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$. والنسبة
$F_e / F_g \approx \num{2.3e39}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:11}
$E = \sqrt{300^2 + 400^2} = \qty{500}{N/C}$، بزاوية
$\tan^{-1}(400/300) \approx \ang{53}$ شمال الشرق. وعلى $q < 0$:
$F = \num{2.0e-6} \times 500 = \qty{1.0e-3}{N}$، عكس \omterm{def:g11:electric-gravitational-fields:field}{المجال} ---
أي $\ang{53}$ جنوب الغرب.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:12}
يعطي $GM/(R+h)^2 = \tfrac12\,GM/R^2$ العلاقة $R + h = R\sqrt{2}$، ومنه
$h = (\sqrt{2} - 1)R \approx \qty{2.6e6}{m}$ (أي نحو \qty{2600}{km}). وعند
$h = R$، $d = 2R$: أي ربع القيمة السطحية.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:13}
\emph{1.} $AM = \sqrt{3.0^2 + 4.0^2} = \qty{5.0}{cm}$؛ وتنشئ كل شحنة
$E_1 = \num{8.99e9} \times \num{2.0e-9} / 0.050^2
\approx \qty{7.2e3}{N/C}$.

\emph{2.} تتلاشى المركّبتان على $AB$ بالتناظر؛ ويسهم كل \omterm{def:g11:electric-gravitational-fields:field}{مجال}
بالمقدار $E_1 \times 4/5$ على الواسط، بعيدًا عن القطعة:
$E = 2 \times \num{7.19e3} \times 0.80 \approx \qty{1.2e4}{N/C}$،
متجهًا على الواسط، بعيدًا عن $[AB]$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:14}
تتحرك الشحن الحرة في الصندوق تحت \omterm{def:g11:electric-gravitational-fields:field}{المجال} الخارجي: فتتراكم $+$ على
وجه، و$-$ على الآخر، وتنشئ هذه الشحن المزاحة مجالًا
داخليًّا يعاكس الخارجي. وتظل تتحرك حتى ينعدم \omterm{def:g11:electric-gravitational-fields:field}{المجال}
الكلي في الداخل تمامًا --- فيحدث التوازن. أما \omterm{def:g11:fundamental-interactions:four}{الجاذبية} فلا تعرض
إلا \omterm{def:g10:signals-and-waves:signal}{إشارة} واحدة للمنبع: فكل كتلة \emph{تضيف} مجالًا جاذبًا،
ولا شيء يعاكسه. والدرع الثقالي كان سيحتاج كتلة سالبة، وهي
غير موجودة.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:15}
\emph{1.} \omterm{def:g10:universal-gravitation:weight}{الوزن} $m\vect g$ (إلى أسفل)، \omterm{def:g10:inertia:inventory}{وشدّ} الخيط $\vect T$ (على
الخيط)، \omterm{def:g10:inertia:force}{والقوة} الكهربائية $q\vect E$ (أفقية). وفي السكون،
$T\sin\theta = qE$ و$T\cos\theta = mg$؛ وبالقسمة:
$\tan\theta = qE/(mg)$.

\emph{2.} $\tan\theta = \num{2.0e-7} \times \num{1.0e4} /
(\num{5.0e-4} \times 9.81) \approx 0.41$، ومنه $\theta \approx \ang{22}$؛
و$T = mg/\cos\theta \approx \qty{5.3e-3}{N}$.
\end{solution}

\begin{solution}{pb:g11:electric-gravitational-fields:1}
\textbf{1.} منتظم، عمودي على الصفيحتين، متجه إلى أسفل (من
الصفيحة العليا $+$ إلى السفلى $-$)؛ والانتظام يعني أن \omterm{def:g10:inertia:force}{القوة}
على القطيرة هي نفسها أينما انجرفت.
\textbf{2.} $E = U/d = 800 / \num{8.0e-3} = \qty{1.0e5}{V/m}$.
\textbf{3.} الشحنة سالبة، ومن ثم تكون \omterm{def:g10:inertia:force}{القوة} \emph{عكس}
$\vect E$: أي إلى أعلى. ولا يتجه $\vect E$ إلى أسفل ولا تصارع \omterm{def:g10:inertia:force}{القوة} الكهربائية الوزنَ
إلا والصفيحة $+$ في الأعلى.
\textbf{4.} $P = mg = \num{4.90e-15} \times 9.81 \approx
\qty{4.8e-14}{N}$.
\textbf{5.} $\abs{q}E = mg$.
\textbf{6.} $\abs{q} = mg/E = mgd/U$.
\textbf{7.} $\abs{q} = \num{4.81e-14} \times \num{8.0e-3} / 800
\approx \qty{4.8e-19}{C}$.
\textbf{8.} $n = \num{4.8e-19} / \num{1.60e-19} = 3$ إلكترونًا زائدًا.
\textbf{9.} تكبر $E$، ومنه $\abs{q}E > mg$: فتتجه \omterm{def:g10:inertia:force}{القوة} المحصّلة إلى أعلى.
\textbf{10.} من $\abs{q}\,U/d = mg$، نجد $U = mgd/\abs{q}$ --- أي متناسبًا
عكسًا مع الشحنة.
\textbf{11.} $\abs{q} = mgd/U = \num{3.85e-16}/1200 \approx
\qty{3.2e-19}{C} = 2e$: أي انخفاضًا من $3e$، فقد \emph{فقدت} القطيرة
إلكترونًا.
\textbf{12.} $\abs{q} = \num{3.85e-16}/U$: \qty{3.2e-19}{C}،
\qty{4.8e-19}{C}، \qty{6.4e-19}{C}، \qty{8.0e-19}{C}،
\qty{9.6e-19}{C}.
\textbf{13.} هي $2e, 3e, 4e, 5e, 6e$ حيث
$e = \qty{1.60e-19}{C}$.
\textbf{14.} كل خطوة تغيّر $\abs{q}$ بمقدار \qty{1.6e-19}{C}
بالضبط: أي أن إلكترونًا واحدًا يهبط في كل مرة.
\textbf{15.} كان سيقتضي ذلك $\abs{q} = \num{3.85e-16}/700 \approx
\qty{5.5e-19}{C} = 3.4\,e$ --- وهو ليس عددًا صحيحًا من الإلكترونات، فلا
دفقة إشعاع تقدر أن تنتجه أبدًا.
\textbf{16.} الشحنة \emph{مكمَّاة}: فهي تأتي مضاعفات صحيحة
\omterm{def:g11:fundamental-interactions:elementary}{للشحنة الأولية} $e = \qty{1.60e-19}{C}$ --- وهو المقدار الذي
قاسه ميليكان.
\textbf{17.} $q \leftrightarrow m$، $E \leftrightarrow g$، و
$\abs{q}E \leftrightarrow mg$: أي \omterm{def:g10:universal-gravitation:weight}{الوزن} نفسه.
\textbf{18.} $g_{\text{الكرة}} = GM/d^2 = \num{6.67e-11} \times 1000 /
1.0^2 = \qty{6.7e-8}{N/kg}$؛ \omterm{def:g10:inertia:force}{والقوة} على القطيرة
$\num{4.90e-15} \times \num{6.7e-8} \approx \qty{3.3e-22}{N}$ --- أي أصغر نحو
$\num{1.5e8}$ مرة من وزنها. فالأمر ميؤوس منه.
\textbf{19.} لأن $G$ صغير جدًّا، فلا يقدر على منافسة جذب الأرض إلا كتلة
بحجم كوكب فوق الرأس --- ولا كتلة تنفر. أما $k$ الهائل
وإشارتا الشحنة فيتيحان لصفيحة على طاولة أن تغلب الكوكب
في أي من الجهتين.
\textbf{20.} الإلكترون الواحد: $\Delta\abs{q}/\abs{q} = e/3e \approx 33\%$؛
\omterm{def:g11:fundamental-interactions:ladder}{وجزيء} الزيت الواحد: $\Delta m/m = \num{5e-25} / \num{4.90e-15}
\approx \num{e-10}$. فسُلّم الشحنة يحرّك \omterm{def:g11:circuits-and-power:voltage}{توتر} التوازن
مئات الفولطات؛ وسُلّم الكتلة غير مرئي عند أي \omterm{def:g11:circuits-and-power:voltage}{توتر}.
ولذلك يفرّق هذا التوازن \omterm{def:g11:fundamental-interactions:ladder}{ذرة} الكهرباء ---
$e = \qty{1.60e-19}{C}$ --- بينما تتشوّش \omterm{def:g11:fundamental-interactions:ladder}{جزيئات} الزيت في
اتصال.
\end{solution}
