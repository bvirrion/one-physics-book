\section*{الفصل \ref{ch:g11:magnetic-fields} --- المغناطيسية والمجالات المغناطيسية}

\begin{solution}{exo:g11:magnetic-fields:1}
كل قطعة \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} كامل له قطب شمالي وقطب جنوبي: فيظهر قطب
جديد على كل وجه مقطوع. والوجهان الجديدان أحدهما N والآخر S، ومن ثم
يتجاذبان --- فتحاول القطعتان أن تلتئما. ولا طريقة تعزل
قطبًا: فكل قطع ينشئ زوجًا.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:2}
(أ) خطأ: فللمجال جهة واحدة في كل نقطة. (ب) صحيح.
(ج) صحيح: فكل خط ينغلق عبر جسم \omterm{def:g11:magnetic-fields:magnet}{المغناطيس}. (د) خطأ: بل تستقرّ
الإبرة \emph{على} الخط، مماسّة له.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:3}
$B = \num{2e-7} \times 10 / 0.020 = \qty{1.0e-4}{T}$ --- أي ضعف \omterm{def:g11:electric-gravitational-fields:field}{مجال}
الأرض.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:4}
$n = 800/0.40 = \qty{2000}{m^{-1}}$؛
$B = \num{1.26e-6} \times 2000 \times 1.5 \approx \qty{3.8}{mT}$.
و$B \propto I$: فمضاعفة التيار تعطي \qty{7.5}{mT}.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:5}
$F = ILB = 5.0 \times 0.10 \times 0.20 = \qty{0.10}{N}$ --- أي \omterm{def:g10:universal-gravitation:weight}{وزن}
$m = F/g = 0.10/9.81 \approx \qty{10}{g}$.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:6}
\emph{1.} يتجه مسارها $\ang{8}$ غرب الشمال الحقيقي:
أي $2000 \times \sin\ang{8} \approx \qty{2.8e2}{m}$ إلى الغرب.

\emph{2.} اتجه $\ang{8}$ \emph{شرق} شمال الإبرة.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:7}
$d = \mu_0 I / (2\pi B) = \num{2e-7} \times 20 / \num{5.0e-5}
= \qty{8.0}{cm}$. فضمن ديسيمتر من كابل كهذا تقرأ البوصلة
السلك لا الكوكب --- فأبعِدها عن النواقل الحيّة.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:8}
$n = B/(\mu_0 I) = 0.010 / (\num{1.26e-6} \times 4.0)
\approx \qty{2.0e3}{m^{-1}}$؛ وعلى \qty{25}{cm}،
$N = 1989 \times 0.25 \approx 500$ لفّة.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:9}
$F = 10 \times 0.12 \times 0.50 = \qty{0.60}{N}$؛
$a = F/m = 0.60/0.020 = \qty{30}{m/s^2}$ --- أي نحو $3g$: فالقضيب
يقفز فعلًا.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:10}
التوازن: $ILB = mg$، ومنه
$I = \num{5.0e-3} \times 9.81 / (0.20 \times 0.10) \approx
\qty{2.5}{A}$. ويجب أن تتجه \omterm{def:g10:inertia:force}{القوة} إلى أعلى؛ ومع كون $\vect B$ أفقيًّا و
عموديًّا على السلك، تحدّد اليد اليمنى المبسوطة (والراحة إلى أعلى)
جهة التيار المطلوبة.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:11}
اليد اليمنى المبسوطة: (أ) نحو أعلى الصفحة؛ (ب) إلى اليمين؛
(ج) لا \omterm{def:g10:inertia:force}{قوة} --- فالسلك موازٍ \omterm{def:g11:electric-gravitational-fields:field}{للمجال}.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:12}
\emph{1.} $B = \num{2e-7} \times 5.0 / 0.030 \approx
\qty{33}{\micro T}$؛ وتدور الخطوط حول السلك، فيكون \omterm{def:g11:electric-gravitational-fields:field}{المجال} تحته مباشرة
أفقيًّا عموديًّا على السلك: أي شرقًا-غربًا.

\emph{2.} تتبع الإبرة المجموع المتجهي: $\tan\theta = 33/20 =
1.67$، ومنه $\theta \approx \ang{59}$ عن الشمال.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:13}
\emph{1.} $I = U/R = 12/2.0 = \qty{6.0}{A}$؛
$n = 400/0.20 = \qty{2000}{m^{-1}}$؛
$B_0 = \num{1.26e-6} \times 2000 \times 6.0 \approx \qty{15}{mT}$؛
ومع القلب، $100 B_0 \approx \qty{1.5}{T}$.

\emph{2.} $P = UI = 12 \times 6.0 = \qty{72}{W}$.

\emph{3.} لا تيار فلا \omterm{def:g11:electric-gravitational-fields:field}{مجال} --- والحديد الليّن لا يبقى
ممغنطًا: فتُطلَق الحمولة فورًا، بالتصميم.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:14}
\emph{1.} $F = NILB = 50 \times 2.0 \times 0.050 \times 0.30 =
\qty{1.5}{N}$ على كل ضلع. ويجري التيار في الضلعين في جهتين
متعاكستين، ومن ثم تكون القوتان متعاكستين: أي مزدوجة تدير الحلقة.

\emph{2.} حين يكون السلك موازيًا للمتجهة $\vect B$، لا يحسّ \omterm{prop:g11:magnetic-fields:laplace}{قوة لابلاس}.

\emph{3.} بعد نصف دورة كانت المزدوجة ستنعكس وتنقض
الدوران؛ فيقلب \omterm{rem:g11:magnetic-fields:motor}{المبدّل} التيار كل نصف دورة حتى تظل الحلقة
تدور في جهة واحدة.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:15}
\emph{1.} $d = \num{2e-7} \times 100 / 1 = \qty{2.0e-5}{m} =
\qty{20}{\micro m}$ --- أي عميقًا \emph{داخل} النحاس ذي السُّمك المِلّيمتري.
فلا نقطة قابلة للوصول قرب سلك مفرد تبلغ \qty{1}{T}: فالمجالات
القوية تكدّس آلاف اللفّات (أي ملفًّا لولبيًّا)، مع حديد أو
نواقل فائقة.

\emph{2.} $\num{e8} / \num{5e-5} = \num{2e12}$: أي نحو اثنتي عشرة \omterm{def:g10:inertia:force}{قوة} من قوى
العشرة.
\end{solution}

\begin{solution}{pb:g11:magnetic-fields:1}
\textbf{1.} الإبرة نفسها \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} صغير؛ \omterm{def:g11:electric-gravitational-fields:field}{ومجال} الأرض
يصفّها، وطرفها الشمالي على المركّبة الأفقية للمتجهة $\vect B$،
أي\ نحو الشمال المغناطيسي.

\textbf{2.} $B = \sqrt{20^2 + 44^2} = \sqrt{2336} \approx
\qty{48}{\micro T}$، مائلًا $\tan^{-1}(44/20) \approx \ang{66}$ تحت
الأفقي.

\textbf{3.} يمتد \omterm{def:g10:relative-motion:trajectory}{المسار} $\ang{6}$ غرب الشمال الحقيقي:
أي $30 \times \sin\ang{6} \approx \qty{3.1}{km}$ غربًا.

\textbf{4.} تتجمع الأخطاء: $\ang{10}$، ومنه $30 \times \sin\ang{10} \approx
\qty{5.2}{km}$ --- أي خمسة كيلومترات لعشر درجات.

\textbf{5.} المركّبة الأفقية: فقرب القطب يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال}
عموديًّا تقريبًا، وينكمش الجزء الأفقي نحو الصفر، ولا
يبقى للإبرة ما تصطفّ معه.

\textbf{6.} $n = 600/0.30 = \qty{2000}{m^{-1}}$.

\textbf{7.} $I = U/R = 12/1.5 = \qty{8.0}{A}$.

\textbf{8.} $B_0 = \mu_0 n I = \num{1.26e-6} \times 2000 \times 8.0
\approx \qty{20}{mT}$.

\textbf{9.} $50 \times \qty{20}{mT} = \qty{1.0}{T}$ --- أي ضعف
\omterm{def:g11:electric-gravitational-fields:field}{المجال} عند قطب النيوديميوم، وعلى وجه أكبر بكثير.

\textbf{10.} $P = UI = 12 \times 8.0 = \qty{96}{W}$؛ وفي
\qty{10}{\minute}: $E = 96 \times 600 \approx \qty{58}{kJ}$.

\textbf{11.} لأنه يترك: فافتح المفتاح يمُت \omterm{def:g11:electric-gravitational-fields:field}{المجال} وتسقط
السيارة حيث يُراد بالضبط. أما \omterm{def:g11:magnetic-fields:magnet}{المغناطيس} الدائم فيمسك إلى الأبد.

\textbf{12.} $F = NILB = 100 \times 3.0 \times 0.040 \times 0.25 =
\qty{3.0}{N}$.

\textbf{13.} مزدوجة: فالقوتان المتعاكستان، مطبَّقتين على جانبي
المحور، تديران الحلقة في الجهة نفسها. أما قوتان متساويتان \emph{متوازيتان}
فلن تفعلا إلا دفع الحلقة جانبيًّا، لا إدارتها.

\textbf{14.} لا \omterm{def:g10:inertia:force}{قوة}: فالسلك الموازي \omterm{def:g11:electric-gravitational-fields:field}{للمجال} لا يحسّ \omterm{prop:g11:magnetic-fields:laplace}{قوة
لابلاس}.

\textbf{15.} يعكس التيار في الحلقة عند كل نصف دورة،
تمامًا حين كانت المزدوجة ستغيّر إشارتها --- فيقود العزم
الدورانَ نفسه دائمًا.

\textbf{16.} ضاعِف $N$، أو ضاعِف $I$، أو ضاعِف $B$: فمزدوجة القوتين
$F = NILB$ تتناسب مع كلٍّ منها.

\textbf{17.} $1.5 / \num{5.0e-5} = \num{3.0e4}$: أي ثلاثين ألف
أرض.

\textbf{18.} $n = B/(\mu_0 I) = 1.5 / (\num{1.26e-6} \times 500)
\approx \qty{2.4e3}{m^{-1}}$ --- وكل لفّة من تلك اللفّات تحمل
\qty{500}{A}.

\textbf{19.} $P = RI^2 = 0.20 \times 500^2 = \qty{50}{kW}$ --- أي ما
يكفي لتدفئة حيّ. أما اللفّ الحقيقي فهو
فائق الناقلية: \omterm{def:g11:circuits-and-power:resistance}{بمقاومة} صفرية وتبديد صفري، شرط أن
يبقى على بضع درجات فوق الصفر \omterm{met:g10:pressure:dive}{المطلق}.

\textbf{20.} البوصلة $\num{5e-5}$، والرافعة $1$، وجهاز الرنين $1.5$: فالسفينة
تقع خمس درجات تحت الآلتين، وهما تتقاسمان عقدًا واحدًا ---
بين $\num{e-10}$ في الفضاء بين النجوم و$\num{e8}$ على نجم نيوتروني. ولم يتغير
شيء إلا الأمبيرات (واللفّات والحديد والناقل الفائق التي
تضاعفها): \omterm{def:g11:electric-gravitational-fields:field}{مجال} واحد، وثلاث وظائف.
\end{solution}
