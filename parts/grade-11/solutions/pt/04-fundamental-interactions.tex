\begin{solution}{exo:g11:fundamental-interactions:6}
$F_E = \num{8.99e9} \times (\num{1.60e-19})^2/(\num{2.0e-15})^2
= \num{2.30e-28}/\num{4.0e-30} \approx \qty{58}{N}$;

\omterm{def:g11:fundamental-interactions:strong}{interação forte} segura o \omterm{def:g11:fundamental-interactions:ladder}{núcleo}; a gravidade é irrelevante.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:7}

\approx \qty{8.2e-8}{N}$;
$F_G = \num{6.67e-11} \times \num{1.67e-27} \times \num{9.11e-31} /

\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:8}

$F = \num{8.99e9} \times (\num{4.0e-9})^2/0.10^2 = \qty{1.4e-5}{N}$,

\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:9}

\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:10}

forte} não contribui com nada: $10^{-10}$~m é $10^{5}$ vezes o \omterm{def:g11:fundamental-interactions:strong}{alcance}

\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:11}
$q = \num{e23}/\num{e6} \times \num{1.6e-19} = \qty{1.6e-2}{C}$ em cada
moeda. $F = \num{8.99e9} \times (\num{1.6e-2})^2/1.0^2 \approx
\qty{2.3e6}{N}$ --- o \omterm{def:g10:universal-gravitation:weight}{peso} de
$\num{2.3e6}/9.81 \approx \qty{2.3e5}{kg}$, um trem de carga carregado,
a partir de um milionésimo dos elétrons. A matéria do dia a dia precisa
ser (e é) neutra com precisão muito melhor do que uma parte em um
milhão.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:12}
$F = \num{2.30e-28}/(\num{1.4e-14})^2 = \num{2.30e-28}/\num{1.96e-28}
\approx \qty{1.2}{N}$ --- ainda enorme nessa escala. Não: eles estão a
$14$ vezes o \omterm{def:g11:fundamental-interactions:strong}{alcance} da \omterm{def:g10:inertia:force}{força} forte; cada um só está colado aos seus
vizinhos mais próximos. Pares: $\frac{92 \times 91}{2} = 4186$,
\emph{todos} se repelindo a qualquer distância, enquanto a cola não
cresce com o tamanho. Os nêutrons acrescentam atração sem repulsão e
diluem os prótons; mas a repulsão cresce com o quadrado do número de
prótons e a cola só com o número de vizinhos, de modo que, acima de uns
$80$ prótons, orçamento de nêutron algum fecha as contas --- esses
\omterm{def:g11:fundamental-interactions:ladder}{núcleos} decaem (\cref{ch:g11:nucleus-radioactivity}).
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:13}
Equilíbrio: na vertical $T\cos\theta = mg$, na horizontal
$T\sin\theta = F$, logo $F = mg\tan\theta$. Aqui
$\sin\theta = 3.0/50 = 0.060$ e, para ângulos pequenos,
$\tan\theta \approx \sin\theta = 0.060$. Então
$F = \num{1.0e-3} \times 9.81 \times 0.060 \approx \qty{5.9e-4}{N}$;
$q = d\sqrt{F/k} = 0.060 \times \sqrt{\num{5.9e-4}/\num{8.99e9}}
\approx \qty{1.54e-8}{C}$;
$N = \num{1.54e-8}/\num{1.6e-19} \approx \num{9.6e10}$ \omterm{def:g11:fundamental-interactions:elementary}{cargas
elementares}.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:14}
$F = \num{6.67e-11} \times \num{5.97e24} \times \num{7.35e22} /
(\num{3.84e8})^2 = \num{2.93e37}/\num{1.47e17} \approx \qty{2.0e20}{N}$.
Impondo $kQ^2/d^2 = F$:
$Q = d\sqrt{F/k} = \num{3.84e8} \times \sqrt{\num{2.0e20}/\num{8.99e9}}
\approx \qty{5.7e13}{C}$. Isso são
$N = \num{5.7e13}/\num{1.6e-19} \approx \num{3.6e32}$ elétrons, de massa
$\num{3.6e32} \times \num{9.11e-31} \approx \qty{3.2e2}{kg}$: algumas
centenas de \omterm{def:g10:orders-of-magnitude:unit}{quilogramas} de elétrons fariam o trabalho de
$\qty{7.35e22}{kg}$ de Lua --- a \omterm{def:g10:inertia:force}{força} da eletricidade e a fraqueza da
gravidade num único número.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:15}
$F = \num{2.30e-28}/(\num{2.8e-10})^2 = \num{2.30e-28}/\num{7.84e-20}
\approx \qty{2.9e-9}{N}$; \omterm{def:g10:universal-gravitation:weight}{peso}
$P = \num{5.9e-26} \times 9.81 \approx \qty{5.8e-25}{N}$; razão
$F/P \approx \num{5e15}$. Cada ligação tem uma \omterm{def:g10:inertia:force}{força} \emph{fixa}, mas o
\omterm{def:g10:universal-gravitation:weight}{peso} cresce com toda a massa que está por cima: empilhe rocha e a carga
sobre a camada de baixo cresce sem limite, enquanto as ligações dela
não. Por volta de \qty{e4}{m} de rocha a carga esmaga as ligações
elétricas --- a gravidade vence não pela \omterm{def:g10:inertia:force}{força} por partícula, mas por
\emph{se somar} quando a carga se cancela.
\end{solution}

\begin{solution}{pb:g11:fundamental-interactions:1}
\textbf{1.} \omterm{def:g11:fundamental-interactions:ladder}{Quark} $< \qty{e-18}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{núcleon} $\qty{e-15}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{núcleo}
$10^{-15}$--$\qty{e-14}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{átomo} $\qty{e-10}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{molécula}
$\qty{e-9}{m}$; objeto do dia a dia $\qty{1}{m}$; planeta $\qty{e7}{m}$;
sistema planetário $\qty{e11}{m}$; galáxia $\qty{e21}{m}$.

\textbf{2.} Fator de escala $10^{5}$: o \omterm{def:g11:fundamental-interactions:ladder}{átomo} fica com
$\qty{1}{cm} \times 10^{5} = \qty{1}{km}$ de largura.

\textbf{3.} $(10^{-15}/10^{-10})^3 = 10^{-15}$: a matéria é
$\num{99.9999999999999}\,\%$ vazia.

\textbf{4.} $\num{7e-5}/\num{e-10} = \num{7e5}$ \omterm{def:g11:fundamental-interactions:ladder}{átomos} --- cerca de um
milhão ao longo de um fio de cabelo.

\textbf{5.} De $10^{-18}$ a $10^{21}$: $39$ potências de dez.

\textbf{6.} $F_E = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$.

\textbf{7.} $F_G = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31}/(\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$; razão
$F_E/F_G \approx \num{2.3e39}$. A eletricidade segura o \omterm{def:g11:fundamental-interactions:ladder}{átomo}; a
gravidade é espectadora.

\textbf{8.} Sem gravidade: nada de mensurável muda (ela está $10^{39}$
abaixo). Sem eletricidade: nem \omterm{def:g11:fundamental-interactions:ladder}{átomos}, nem \omterm{def:g11:fundamental-interactions:ladder}{moléculas}, nem matéria
alguma.

\textbf{9.} O ``contato'' é a repulsão elétrica entre as nuvens de
elétrons de \omterm{def:g11:fundamental-interactions:ladder}{átomos} aproximados até cerca de $\qty{e-10}{m}$: nada nunca
se toca de fato. A reação do piso e o \omterm{def:g10:inertia:inventory}{atrito} são a lei de Coulomb em
escala macroscópica.

\textbf{10.} $q = \num{e23} \times \num{e-9} \times \num{1.6e-19} =
\qty{1.6e-5}{C}$ em cada uma;
$F = \num{8.99e9} \times (\num{1.6e-5})^2 \approx \qty{2.3}{N}$ --- uma
\omterm{def:g10:inertia:force}{força} claramente visível a partir de um elétron em cada bilhão. A matéria
do dia a dia é neutra com precisão muito melhor do que $10^{-9}$.

\textbf{11.} A \omterm{def:g11:fundamental-interactions:four}{interação eletromagnética}. A \omterm{def:g10:inertia:force}{força} forte está fora de
\omterm{def:g11:fundamental-interactions:strong}{alcance} além de cerca de $\qty{e-14}{m}$ --- $10^{4}$ vezes menos do que
um único \omterm{def:g11:fundamental-interactions:ladder}{átomo}.

\textbf{12.} $F = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{1.0e-15})^2 \approx \qty{2.3e2}{N}$.

\textbf{13.} $a = F/m_p = 230/\num{1.67e-27} \approx
\qty{1.4e29}{m/s^2}$, cerca de $\num{1.4e28}$ vezes $g$: \omterm{def:g10:inertia:force}{força} alguma do
dia a dia chega perto.

\textbf{14.} Mais forte do que \qty{230}{N} a $\qty{e-15}{m}$ (os \omterm{def:g11:fundamental-interactions:ladder}{núcleos}
existem); agindo igualmente sobre prótons e nêutrons, cega à carga (os
nêutrons também são agarrados); \omterm{def:g11:fundamental-interactions:strong}{alcance} de cerca de $\qty{e-15}{m}$ (os
\omterm{def:g11:fundamental-interactions:ladder}{núcleos} continuam minúsculos --- a cola não alcança mais longe).

\textbf{15.} $F = \num{2.30e-28}/(\num{1.4e-14})^2 \approx \qty{1.2}{N}$.
Não: são $14$ vezes o \omterm{def:g11:fundamental-interactions:strong}{alcance}. A repulsão age em \emph{todos} os
$\frac{92\times91}{2} = 4186$ pares de prótons; a cola liga só os
vizinhos mais próximos, de modo que o crescimento favorece a repulsão. Os
nêutrons acrescentam cola e espaçamento sem acrescentar repulsão --- até
que, acima de uns $80$ prótons, mistura alguma equilibra e o \omterm{def:g11:fundamental-interactions:ladder}{núcleo}
decai.

\textbf{16.} A \omterm{def:g11:fundamental-interactions:weak}{interação fraca} transforma um nêutron em próton (e
vice-versa), provocando o decaimento $\beta$ --- ela transforma; nunca
liga.

\textbf{17.} $F = \num{6.67e-11} \times \num{5.97e24} \times
\num{7.35e22}/(\num{3.84e8})^2 \approx \qty{2.0e20}{N}$.

\textbf{18.} $Q_T = \num{1.8e51} \times \num{1.6e-19} =
\qty{2.9e32}{C}$, $Q_L = \qty{3.5e30}{C}$. Igualar as \omterm{def:g10:inertia:force}{forças} significa
$f^2\,k\,Q_T Q_L = G M_T M_L$:
\[
f = \sqrt{\frac{\num{2.93e37}}{\num{8.99e9} \times \num{2.9e32}
\times \num{3.5e30}}} = \sqrt{\num{3.2e-36}} \approx \num{1.8e-18} :
\]
um desequilíbrio de \emph{duas partes em $10^{18}$} cancelaria a
gravidade.

\textbf{19.} A carga vem em dois sinais e se cancela (a questão 18 mostra
com que perfeição), de modo que a \omterm{def:g10:inertia:force}{força} mais forte se desliga sozinha em
grande escala; a massa tem um único sinal, soma para sempre e nada a
blinda.

\textbf{20.} \omterm{def:g11:fundamental-interactions:ladder}{Núcleo}: a \omterm{def:g11:fundamental-interactions:strong}{interação forte} vence no braço a repulsão dos
prótons a $\qty{e-15}{m}$. Do \omterm{def:g11:fundamental-interactions:ladder}{átomo} à montanha: a \omterm{def:g11:fundamental-interactions:four}{interação
eletromagnética} liga os elétrons aos \omterm{def:g11:fundamental-interactions:ladder}{núcleos} e os \omterm{def:g11:fundamental-interactions:ladder}{átomos} uns aos outros.
Planeta e acima: a \omterm{def:g11:fundamental-interactions:four}{gravitação}, sem oposição depois que a carga se
cancela, segura planetas, \omterm{def:g10:universal-gravitation:satellite}{órbitas} e galáxias. O mundo não desaba porque
cada andar é escorado por um ligante mais rígido do que a carga, e não se
desfaz porque a matéria é neutra a cerca de $10^{-18}$ --- de modo que
cada escala tem exatamente uma \omterm{def:g10:inertia:force}{força} no comando, e ela é atrativa onde
precisa ser.
\end{solution}
