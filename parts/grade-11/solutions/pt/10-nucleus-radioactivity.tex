\section*{Capítulo \ref{ch:g11:nucleus-radioactivity} --- O núcleo e a radioatividade}

\begin{solution}{exo:g11:nucleus-radioactivity:1}
${}^{16}_{8}\mathrm{O}$: $8$ p, $8$ n; ${}^{27}_{13}\mathrm{Al}$: $13$ p,
$14$ n; ${}^{60}_{27}\mathrm{Co}$: $27$ p, $33$ n;
${}^{235}_{92}\mathrm{U}$: $92$ p, $143$ n; ${}^{3}_{1}\mathrm{H}$: $1$ p,
$2$ n. (Em cada caso, $N = A - Z$.)
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:2}
Os três carbonos ($Z = 6$) são \omterm{def:g11:nucleus-radioactivity:notation}{isótopos} uns dos outros.
${}^{14}_{6}\mathrm{C}$ e ${}^{14}_{7}\mathrm{N}$ compartilham $A = 14$,
mas não $Z$: são elementos diferentes (nuvens de elétrons diferentes),
logo não são \omterm{def:g11:nucleus-radioactivity:notation}{isótopos}.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:3}
$A = 210 - 4 = 206$, $Z = 84 - 2 = 82$:
${}^{210}_{84}\mathrm{Po} \to {}^{206}_{82}\mathrm{Pb} +
{}^{4}_{2}\mathrm{He}$ --- o núcleo-filho é o chumbo-206.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:8}
O flúor-18 tem $9$ nêutrons onde o flúor-19 estável tem $10$: é pobre em
nêutrons, está abaixo da faixa, logo $\beta^+$:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:9}
$24$ horas $= 4\,T_{1/2}$: $500/16 \approx \qty{31}{MBq}$. Como
$2^9 = 512$, nove meias-vidas levam a $500/512 < \qty{1}{MBq}$: depois de
$9 \times 6 = \qty{54}{h}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:10}
$\frac18 = \frac1{2^3}$: três meias-vidas,
$3 \times \num{5700} = \num{17100}$ anos. Um osso de dinossauro tem cerca
de $\num{17500}$ meias-vidas de idade: resta uma fração $1/2^{17500}$ ---
nem um \omterm{def:g11:fundamental-interactions:ladder}{átomo} de carbono-14 (um grama de carbono guarda apenas cerca de
$\num{5e22}$ \omterm{def:g11:fundamental-interactions:ladder}{átomos}).
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:11}
${}^{14}_{6}\mathrm{C}$: $8$ nêutrons contra $6$ ou $7$ no carbono
estável, acima da faixa, logo $\beta^-$. ${}^{11}_{6}\mathrm{C}$: $5$
nêutrons, pobre em nêutrons, logo $\beta^+$. ${}^{238}_{92}\mathrm{U}$:
$Z = 92 > 83$, pesado demais, logo $\alpha$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:12}
Só o $\alpha$ muda $A$: $238 - 206 = 32 = 4x$, logo $x = 8$ decaimentos
\omterm{def:g11:nucleus-radioactivity:abg}{alfa}. Eles retiram $16$ de $Z$: $92 - 16 + y = 82$ dá $y = 6$ decaimentos
beta-menos.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:13}
$6.4/0.1 = 64 = 2^{6}$: seis meias-vidas,
$6 \times 5.3 = 31.8 \approx 32$ anos de armazenamento blindado.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:14}
Por dia: $8000 \times 86400 \approx \num{6.9e8}$ decaimentos. Em $80$
anos: $\num{6.9e8} \times 365 \times 80 \approx \num{2e13}$. A vida
sempre funcionou com esse pano de fundo --- um \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} é um \omterm{def:g11:fundamental-interactions:ladder}{átomo} por
segundo entre os $\sim 10^{27}$ do corpo; a \omterm{def:g10:orders-of-magnitude:unit}{unidade} é minúscula, e uma
contagem de \omterm{def:g11:nucleus-radioactivity:activity}{becquerels} de aparência assustadora pode valer umas poucas
bananas.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:15}
$\num{25000} \times 86400 \approx \num{2.2e9}$ decaimentos por dia. Dez
anos são $10/432 \approx 2\%$ de uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}: bem menos do que a
metade decaiu, de modo que são a eletrônica e a poeira, e não a fonte,
que obrigam à substituição. Do lado de fora, as partículas $\alpha$
morrem no ar da câmara e no invólucro; já o pó inalado estaciona o
emissor nos pulmões, onde cada $\alpha$ despeja a sua \omterm{def:g10:energy-conservation:energy}{energia} em tecido
vivo --- uma \omterm{rem:g11:nucleus-radioactivity:dose}{dose} grande (em \omterm{rem:g11:nucleus-radioactivity:dose}{sieverts}) a partir de uma \omterm{def:g11:nucleus-radioactivity:activity}{atividade}
modesta.
\end{solution}

\begin{solution}{pb:g11:nucleus-radioactivity:1}
\textbf{1.} ${}^{12}_{6}\mathrm{C}$: $6$ p, $6$ n;
${}^{14}_{6}\mathrm{C}$: $6$ p, $8$ n. A química só enxerga a nuvem de
elétrons, fixada por $Z = 6$: comportamento idêntico.

\textbf{2.} Comer e respirar renovam constantemente o carbono de um corpo
vivo na proporção atmosférica; na morte a reposição para e o decaimento
segue sem oposição.

\textbf{3.} ${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} +
{}^{\;0}_{-1}\mathrm{e}$ ($\beta^-$). Esses $\beta$ são fracos: as
camadas de linho os barram, e por isso quase nada escapa da múmia --- e é
por isso que se mede uma pequena amostra de carbono retirada do próprio
objeto.

\textbf{4.} $6.8/13.6 = \frac12$: uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}. A múmia tem cerca de
\num{5700} anos.

\textbf{5.} $13.2/13.6 \approx 0.97$: o linho é essencialmente moderno
--- no máximo com alguns séculos. Uma falsificação.

\textbf{6.} $13.6/2^{10} = 13.6/1024 \approx 0.013$ decaimento por minuto
--- uma contagem a cada $75$ minutos por grama, afogada na radiação de
fundo natural. Dez meias-vidas, cerca de \num{57000} anos, é o horizonte
prático.

\textbf{7.} $92$ prótons e $146$ nêutrons. A repulsão de Coulomb age
entre todos os pares de prótons, de ponta a ponta do \omterm{def:g11:fundamental-interactions:ladder}{núcleo}, enquanto a
cola forte liga só os vizinhos: os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} pesados só sobrevivem com um
excedente de nêutrons, que é cola sem repulsão.

\textbf{8.} ${}^{238}_{92}\mathrm{U} \to {}^{234}_{90}\mathrm{Th} +
{}^{4}_{2}\mathrm{He}$.

\textbf{9.} ${}^{234}_{90}\mathrm{Th} \to {}^{234}_{91}\mathrm{Pa} +
{}^{\;0}_{-1}\mathrm{e}$ e depois ${}^{234}_{91}\mathrm{Pa} \to
{}^{234}_{92}\mathrm{U} + {}^{\;0}_{-1}\mathrm{e}$: o urânio reaparece,
como urânio-234.

\textbf{10.} Pelo $A$: $238 - 206 = 32$, logo $8$ passos $\alpha$; pelo
$Z$: $92 - 16 + y = 82$, logo $6$ passos $\beta^-$.

\textbf{11.} $\num{4.5e9}$ anos são uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}: resta $\frac12$. Mais
cinco bilhões de anos $\approx$ mais uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}: cerca de $\frac14$.

\textbf{12.} Ele move vulcões, a tectônica de placas (continentes à
deriva, terremotos) e o calor geotérmico: a geologia de superfície de um
planeta aquecido por dentro.

\textbf{13.} ${}^{241}_{95}\mathrm{Am} \to {}^{237}_{93}\mathrm{Np} +
{}^{4}_{2}\mathrm{He}$.

\textbf{14.} $\num{3.3e4}$ decaimentos por segundo;
$\num{33000} \times 86400 \approx \num{2.9e9}$ por dia.

\textbf{15.} Cada $\alpha$ ioniza o ar que atravessa; os íons conduzem
uma corrente minúscula entre os eletrodos da câmara. As partículas de
fumaça capturam os íons, a corrente cai e o alarme dispara.

\textbf{16.} As partículas $\alpha$ gastam toda a sua \omterm{def:g10:energy-conservation:energy}{energia} dentro da
câmara e nenhuma escapa do invólucro: ionização máxima, vazamento nulo.
Uma fonte $\gamma$ de mesma \omterm{def:g11:nucleus-radioactivity:activity}{atividade} quase não ionizaria o ar da câmara
e, em compensação, irradiaria o cômodo inteiro.

\textbf{17.} Dez anos são $10/432 \approx 2\%$ de uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}: a
\omterm{def:g11:nucleus-radioactivity:activity}{atividade} fica essencialmente inalterada. Os elos fracos são a poeira, os
insetos e a eletrônica --- o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} sobrevive ao aparelho.

\textbf{18.} $\num{9e9}$ anos $= 2$ meias-vidas: sobrevive $\frac14$. O
urânio da Terra ainda é abundante, de modo que ele foi forjado no máximo
há algumas meias-vidas --- o planeta se condensou de cinzas estelares
recentes, bilhões de anos depois das primeiras estrelas.

\textbf{19.} Geologicamente morto: interior frio, sem vulcanismo nem
tectônica de placas para reciclar ar e rocha, e sem um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} em
agitação --- e, portanto, sem escudo magnético contra o vento solar.

\textbf{20.} Cada \omterm{def:g11:fundamental-interactions:ladder}{átomo} seu além do hélio foi forjado numa estrela
moribunda, e os restos instáveis daquela forja ainda aquecem o chão em
que você pisa: poeira de estrelas, mantida viva pela própria \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}.
\end{solution}
