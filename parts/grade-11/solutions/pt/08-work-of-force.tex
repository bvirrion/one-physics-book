\section*{Capítulo \ref{ch:g11:work-of-force} --- Trabalho de uma força}

\begin{solution}{exo:g11:work-of-force:1}
$W = 40 \times 50 \times \cos\theta$: $+\qty{2000}{J}$;
$+\qty{1000}{J}$; $0$; $-\qty{1000}{J}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:2}
$mgh = 12 \times 9.81 \times 2.5 = \qty{294}{J}$: baixada,
$+\qty{294}{J}$ (\omterm{def:g10:universal-gravitation:weight}{peso} no sentido da descida); içada, $-\qty{294}{J}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:3}
(a) \omterm{def:g11:work-of-force:sign}{motor}; (b) \omterm{def:g11:work-of-force:sign}{resistente}; (c) nulo (perpendicular); (d) \omterm{def:g11:work-of-force:sign}{resistente};
(e) nulo --- a tensão do fio permanece perpendicular ao movimento da
massa.
\end{solution}

\begin{solution}{exo:g11:work-of-force:4}
Velocidade constante: \omterm{def:g10:inertia:force}{força} de içamento $= mg = \qty{5.89e3}{N}$, logo
$W_{\text{içamento}} = 5886 \times 25 \approx +\qty{147}{kJ}$ e
$W(\vect P) = -\qty{147}{kJ}$. Soma nula: com velocidade constante as
\omterm{def:g10:inertia:force}{forças} se equilibram (\cref{ch:g11:forces-and-motion}), de modo que os
\omterm{def:g11:work-of-force:work}{trabalhos} delas se cancelam.
\end{solution}

\begin{solution}{exo:g11:work-of-force:5}
$P = \num{9.0e4}/60 = \qty{1.5}{kW}$; $1500/736 \approx \qty{2.0}{hp}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:6}
Corda: $120 \times 200 \times \cos\qty{30}{\degree} \approx
+\qty{20.8}{kJ}$; \omterm{def:g10:inertia:inventory}{atrito}: $-45 \times 200 = -\qty{9.0}{kJ}$; \omterm{def:g10:universal-gravitation:weight}{peso} e \omterm{def:g10:inertia:force}{força}
normal, perpendiculares: $0$. Total $+\qty{11.8}{kJ}$: o trenó ganha
velocidade.
\end{solution}

\begin{solution}{exo:g11:work-of-force:7}
$v = 25/3.6 = \qty{6.94}{m/s}$;
$P = Fv = 18 \times 6.94 \approx \qty{125}{W}$. Em uma hora,
$W = P\,\Delta t = 125 \times 3600 = \qty{450}{kJ}$ (o que dá no mesmo:
$18 \times \qty{25000}{m}$).
\end{solution}

\begin{solution}{exo:g11:work-of-force:8}
\omterm{def:g10:universal-gravitation:weight}{Peso}, nas duas rotas:
$+65 \times 9.81 \times 120 \approx +\qty{76.5}{kJ}$ (só o desnível).
\omterm{def:g10:inertia:inventory}{Atrito}: $-30 \times 800 = -\qty{24}{kJ}$ na pista e
$-30 \times 1500 = -\qty{45}{kJ}$ na trilha. Só o \omterm{def:g10:inertia:inventory}{atrito} se importa com a
rota.
\end{solution}

\begin{solution}{exo:g11:work-of-force:9}
Subida $h = 4.0\sin\qty{25}{\degree} = \qty{1.69}{m}$:
$W(\vect P) = -20 \times 9.81 \times 1.69 \approx -\qty{332}{J}$;
$W_f = -30 \times 4.0 = -\qty{120}{J}$. Velocidade constante: \omterm{def:g11:work-of-force:work}{trabalho}
total nulo, logo $W_{\text{empurrão}} = +\qty{452}{J}$ e
$F = 452/4.0 \approx \qty{113}{N}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:10}
$v = 130/3.6 = \qty{36.1}{m/s}$; velocidade constante, logo
$F = P/v = \num{5.0e4}/36.1 \approx \qty{1.4}{kN}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:11}
$P = Fv\cos\theta = 800 \times 1.0 \times \cos\qty{20}{\degree}
\approx \qty{752}{W}$; em cavalos-vapor, $752/736 \approx \num{1.02}$ ---
quase exatamente o que um cavalo forte sustenta, a comparação sobre a
qual Watt construiu a sua \omterm{def:g10:orders-of-magnitude:unit}{unidade}.
\end{solution}

\begin{solution}{exo:g11:work-of-force:12}
Comprimentos $L = h/\text{inclinação}$: \qty{1500}{m} e \qty{5000}{m}.
\omterm{def:g10:inertia:force}{Forças} $F = mg \times \text{inclinação}$:
$1200 \times 9.81 \times 0.20 \approx \qty{2.35}{kN}$ e
$\qty{0.71}{kN}$. \omterm{def:g11:work-of-force:work}{Trabalhos} $FL$:
$2354 \times 1500 \approx \qty{3.53}{MJ}$ e
$706 \times 5000 \approx \qty{3.53}{MJ}$ --- os dois iguais a $mgh$: a
estrada troca \omterm{def:g10:inertia:force}{força} por comprimento, jamais por \omterm{def:g11:work-of-force:work}{trabalho}.
\end{solution}

\begin{solution}{exo:g11:work-of-force:13}
$F = mg = \qty{2.45e3}{N}$; $W = 2453 \times 15 \approx \qty{36.8}{kJ}$;
$\Delta t = 15/0.40 = \qty{37.5}{s}$;
$P = 36800/37.5 \approx \qty{981}{W} = 2453 \times 0.40$ ($Fv$
concorda); elétrica: $981/0.70 \approx \qty{1.4}{kW}$.
\end{solution}

\begin{solution}{exo:g11:work-of-force:14}
Diagonal: $-140 \times 6.0 = -\qty{840}{J}$; paredes:
$-140 \times 8.4 \approx -\qty{1.18}{kJ}$; ida e volta:
$-140 \times 12 = -\qty{1.68}{kJ}$. O \omterm{def:g10:universal-gravitation:weight}{peso} realiza \omterm{def:g11:work-of-force:work}{trabalho} nulo na ida e
volta (sem desnível líquido). O \omterm{def:g11:work-of-force:work}{trabalho} do \omterm{def:g10:inertia:inventory}{atrito} depende do caminho e
nunca devolve; o do \omterm{def:g10:universal-gravitation:weight}{peso} depende só dos extremos
(\cref{rem:g11:work-of-force:divide}).
\end{solution}

\begin{solution}{exo:g11:work-of-force:15}
Gravidade: $80 \times 9.81 \times 600 \approx \qty{471}{kJ}$; \omterm{def:g10:inertia:inventory}{atrito}:
$14 \times 12000 = \qty{168}{kJ}$; total \qty{639}{kJ}.
$v = \qty{4.17}{m/s}$, logo
$\Delta t = 12000/4.17 = \qty{2880}{s} = \qty{48}{min}$ e
$P = \num{6.39e5}/2880 \approx \qty{222}{W} \approx \qty{0.30}{hp}$ ---
um amador forte, um terço de cavalo.
\end{solution}

\begin{solution}{pb:g11:work-of-force:1}
\textbf{1.} $W = 110 \times 8.0 = +\qty{880}{J}$.

\textbf{2.} \omterm{def:g10:inertia:inventory}{Atrito}: $-95 \times 8.0 = -\qty{760}{J}$. \omterm{def:g10:universal-gravitation:weight}{Peso} e \omterm{def:g10:inertia:force}{força} normal
são perpendiculares ao movimento: zero cada um.

\textbf{3.} Total $+\qty{120}{J}$: \omterm{def:g11:work-of-force:sign}{trabalho motor} líquido, e a caixa
ganha velocidade. Um empurrão de \qty{95}{N} equilibra o \omterm{def:g10:inertia:inventory}{atrito} ---
\omterm{def:g11:forces-and-motion:netforce}{força resultante} nula, velocidade constante, \omterm{def:g11:work-of-force:work}{trabalho} total nulo.

\textbf{4.} Diagonal: $-95 \times 10.0 = -\qty{950}{J}$; paredes:
$-95 \times 14.0 = -\qty{1.33}{kJ}$: o \omterm{def:g10:inertia:inventory}{atrito} cobra por \omterm{def:g10:orders-of-magnitude:unit}{metro}.

\textbf{5.} \omterm{def:g10:inertia:inventory}{Atrito}: $-95 \times 20.0 = -\qty{1.9}{kJ}$; \omterm{def:g10:universal-gravitation:weight}{peso}: $0$ (sem
desnível líquido). Só o \omterm{def:g10:universal-gravitation:weight}{peso} devolve.

\textbf{6.} $W(\vect P) = -30 \times 9.81 \times 9.0 = -\qty{2.65}{kJ}$;
a equipe precisa fornecer pelo menos $+\qty{2.65}{kJ}$.

\textbf{7.} $(70 + 30) \times 9.81 \times 9.0 = \qty{8.83}{kJ}$; a fatia
da caixa é $2.65/8.83 \approx 30\%$ --- carregar é, sobretudo,
transportar a si mesmo.

\textbf{8.} $\sin\alpha = 9.0/30 = 0.30$:
$F = mg\sin\alpha + f = 88.3 + 40 = \qty{128}{N}$;
$W = 128 \times 30 = \qty{3.85}{kJ}$ ($= 2.65 + 1.2$).

\textbf{9.} Içamento vertical: $F = mg = \qty{294}{N}$,
$W = \qty{2.65}{kJ}$. A rampa compra uma \omterm{def:g10:inertia:force}{força} menor ($128$ contra
\qty{294}{N}) ao preço de mais \omterm{def:g11:work-of-force:work}{trabalho}: o pedágio de \qty{1.2}{kJ} do
\omterm{def:g10:inertia:inventory}{atrito}.

\textbf{10.} $\Delta t = 9.0/0.50 = \qty{18}{s}$;
$P = 2649/18 \approx \qty{147}{W}$.

\textbf{11.} $P = Fv = 294 \times 0.50 = \qty{147}{W}$ --- o mesmo.
Elétrica: $147/0.60 \approx \qty{245}{W}$.

\textbf{12.} $1000 \times 0.24 = \qty{240}{m}$ e
$8000 \times 0.030 = \qty{240}{m}$: o mesmo cume.

\textbf{13.} $F = mg \times \text{inclinação}$: ladeira,
$3500 \times 9.81 \times 0.24 \approx \qty{8.2}{kN}$; estrada,
$\approx \qty{1.0}{kN}$.

\textbf{14.} $8240 \times 1000 \approx \qty{8.24}{MJ}$ e
$1030 \times 8000 \approx \qty{8.24}{MJ}$ --- os dois iguais a
$mgh = 3500 \times 9.81 \times 240$.

\textbf{15.} \omterm{def:g11:circuits-and-power:resistance}{Resistência} de rolamento:
$400 \times 1000 = \qty{0.40}{MJ}$ contra
$400 \times 8000 = \qty{3.2}{MJ}$: o conforto da estrada suave custa
\qty{2.8}{MJ} de \omterm{def:g10:inertia:inventory}{atrito} extra.

\textbf{16.} $v = \qty{10}{m/s}$. Estrada:
$(1030 + 400) \times 10 = \qty{14.3}{kW} \approx \qty{19}{hp}$; ladeira:
$(8240 + 400) \times 10 = \qty{86.4}{kW} \approx \qty{117}{hp}$ --- fora
do \omterm{def:g11:fundamental-interactions:strong}{alcance} de uma van carregada. As estradas ziguezagueiam para que
\omterm{def:g11:work-of-force:sign}{motores} (e pneus) modestos consigam subir: o mesmo \omterm{def:g11:work-of-force:work}{trabalho}, espalhado.

\textbf{17.} Por caixa: $880 + 2649 = \qty{3.53}{kJ}$; para $60$ caixas,
$60 \times 3529 \approx \qty{212}{kJ}$.

\textbf{18.} Escada: $20 \times 70 \times 9.81 \times 9.0 \approx
\qty{124}{kJ}$. Total do dia $\approx \qty{336}{kJ} = \qty{0.34}{MJ}$.

\textbf{19.} $P = \num{3.36e5}/28800 \approx \qty{12}{W}$. A chaleira:
$\num{3.36e5}/2000 = \qty{168}{s}$ --- menos de três minutos para o
\omterm{def:g11:work-of-force:work}{trabalho} do dia inteiro.

\textbf{20.} O \omterm{def:g11:work-of-force:work}{joule} paga pela \omterm{def:g10:inertia:force}{força} entregue \emph{ao longo} de um
movimento --- e nada pela parede empurrada ou pela caixa segurada, por
mais extenuantes que sejam (os músculos queimam \omterm{def:g10:energy-conservation:forms}{energia química} mesmo
realizando \omterm{def:g11:work-of-force:work}{trabalho} mecânico nulo). Como a \omterm{def:g11:work-of-force:power}{potência} humana é de algumas
dezenas de \omterm{def:g11:work-of-force:power}{watts}, as equipes de mudança deixam que as máquinas remodelem
a \omterm{def:g10:inertia:force}{força} --- guincho, rampa, ziguezague --- enquanto a conta $mgh$ nunca
muda.
\end{solution}
