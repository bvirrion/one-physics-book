\section*{Capítulo \ref{ch:g11:electric-gravitational-fields} --- Campos elétrico e gravitacional}

\begin{solution}{exo:g11:electric-gravitational-fields:1}
$E = \num{5.0e-3} / \num{2.0e-6} = \qty{2.5e3}{N/C}$, apontando para o
norte. Sobre $q'$:
$F = \num{4.0e-6} \times \num{2.5e3} = \qty{1.0e-2}{N}$, apontando para
o \emph{sul} ($q' < 0$).
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:2}
\emph{1.} $E = \num{8.99e9} \times \num{5.0e-9} / 0.30^2
\approx \qty{5.0e2}{N/C}$, apontando para longe de $O$ ($Q > 0$).

\emph{2.} Reduzir $k\abs{Q}/d^2$ à metade exige $d^2$ dobrado:
$d' = 0.30\sqrt{2} \approx \qty{0.42}{m}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:3}
$E = 600 / \num{3.0e-3} = \qty{2.0e5}{V/m}$;
$F = eE = \num{1.60e-19} \times \num{2.0e5} = \qty{3.2e-14}{N}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:4}
$g_{\text{Marte}} = P/m = 930/250 = \qty{3.7}{N/kg}$. Astronauta:
$P = 80 \times 3.72 \approx \qty{3.0e2}{N}$ (cerca de \qty{780}{N} na
Terra).
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:5}
(a) Falso: o \omterm{def:g11:electric-gravitational-fields:field}{campo} tem um único sentido em cada ponto, de modo que as
linhas nunca se cruzam. (b) Falso: elas saem do $+$ e chegam ao $-$.
(c) Verdadeiro: o aperto codifica a intensidade.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:6}
\emph{1.} $d = \num{6.37e6} + \num{4.2e5} = \qty{6.79e6}{m}$, logo
$g = \num{6.67e-11} \times \num{5.97e24} / (\num{6.79e6})^2
\approx \qty{8.6}{N/kg}$ --- $88\%$ do valor na superfície.

\emph{2.} Eles flutuam porque estação e astronauta caem juntos (queda
livre), e não porque a gravidade esteja ausente.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:7}
$E = m_e g / e = \num{9.11e-31} \times 9.81 / \num{1.60e-19}
\approx \qty{5.6e-11}{V/m}$: uns $\num{4e14}$ vezes mais fraco do que o
\omterm{def:g11:electric-gravitational-fields:field}{campo} do \omterm{def:g10:signals-and-waves:oscilloscope}{osciloscópio}. A gravidade é absolutamente desprezível na física
do elétron.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:8}
\emph{1.} $E = \num{5.0e3} / \num{2.0e-2} = \qty{2.5e5}{V/m}$; em
repouso $qE = mg$, logo
$q = \num{2.0e-6} \times 9.81 / \num{2.5e5} \approx \qty{7.8e-11}{C}$.

\emph{2.} A \omterm{def:g10:inertia:force}{força} sobre o grão positivo tem de apontar para cima, de modo
que $\vect E$ aponta para cima: a placa \emph{de baixo} é a positiva.

\emph{3.} $n = \num{7.85e-11} / \num{1.60e-19} \approx \num{4.9e8}$
\omterm{def:g11:fundamental-interactions:elementary}{cargas elementares}.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:9}
Fora do segmento os dois \omterm{def:g11:electric-gravitational-fields:field}{campos} apontam no mesmo sentido; entre elas, se
opõem --- e o equilíbrio exige o ponto mais perto da fonte mais fraca,
$+Q$. Com $x$ sendo a distância até $+Q$: $kQ/x^2 = 4kQ/(0.30 - x)^2$,
logo $0.30 - x = 2x$ e $x = \qty{0.10}{m}$ a partir de $+Q$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:10}
$F_e = ke^2/d^2 = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$;
$F_g = G m_p m_e / d^2 = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31} / (\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$. Razão
$F_e / F_g \approx \num{2.3e39}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:11}
$E = \sqrt{300^2 + 400^2} = \qty{500}{N/C}$, a
$\tan^{-1}(400/300) \approx \ang{53}$ do leste em direção ao norte. Sobre
$q < 0$: $F = \num{2.0e-6} \times 500 = \qty{1.0e-3}{N}$, no sentido
oposto ao do \omterm{def:g11:electric-gravitational-fields:field}{campo} --- a $\ang{53}$ do oeste em direção ao sul.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:12}
$GM/(R+h)^2 = \tfrac12\,GM/R^2$ dá $R + h = R\sqrt{2}$, logo
$h = (\sqrt{2} - 1)R \approx \qty{2.6e6}{m}$ (cerca de \qty{2600}{km}).
Em $h = R$, temos $d = 2R$: um quarto do valor na superfície.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:13}
\emph{1.} $AM = \sqrt{3.0^2 + 4.0^2} = \qty{5.0}{cm}$; cada carga cria
$E_1 = \num{8.99e9} \times \num{2.0e-9} / 0.050^2
\approx \qty{7.2e3}{N/C}$.

\emph{2.} As componentes ao longo de $AB$ se cancelam por simetria; cada
\omterm{def:g11:electric-gravitational-fields:field}{campo} contribui com $E_1 \times 4/5$ ao longo da mediatriz, afastando-se
do segmento: $E = 2 \times \num{7.19e3} \times 0.80 \approx
\qty{1.2e4}{N/C}$, dirigido ao longo da mediatriz, para longe de $[AB]$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:14}
As cargas livres da caixa se movem sob o \omterm{def:g11:electric-gravitational-fields:field}{campo} externo: o $+$ se acumula
numa face e o $-$ na outra, e essas cargas deslocadas criam um \omterm{def:g11:electric-gravitational-fields:field}{campo}
interno que se opõe ao externo. Elas continuam se movendo até que o \omterm{def:g11:electric-gravitational-fields:field}{campo}
total no interior seja exatamente nulo --- e então há equilíbrio. A
gravidade oferece um único sinal de fonte: toda massa \emph{acrescenta}
um \omterm{def:g11:electric-gravitational-fields:field}{campo} atrativo, e nenhuma consegue se opor a ele. Um escudo de
gravidade precisaria de massa negativa, que não existe.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:15}
\emph{1.} \omterm{def:g10:universal-gravitation:weight}{Peso} $m\vect g$ (para baixo), tensão $\vect T$ (ao longo do
fio) e \omterm{def:g10:inertia:force}{força} elétrica $q\vect E$ (horizontal). Em repouso,
$T\sin\theta = qE$ e $T\cos\theta = mg$; dividindo:
$\tan\theta = qE/(mg)$.

\emph{2.} $\tan\theta = \num{2.0e-7} \times \num{1.0e4} /
(\num{5.0e-4} \times 9.81) \approx 0.41$, logo
$\theta \approx \ang{22}$; $T = mg/\cos\theta \approx \qty{5.3e-3}{N}$.
\end{solution}

\begin{solution}{pb:g11:electric-gravitational-fields:1}
\textbf{1.} \omterm{def:g11:electric-gravitational-fields:capacitor}{Uniforme}, perpendicular às placas, apontando para baixo (da
placa $+$ de cima para a $-$ de baixo); a uniformidade garante que a
\omterm{def:g10:inertia:force}{força} sobre a gotícula é a mesma para onde quer que ela derive.
\textbf{2.} $E = U/d = 800 / \num{8.0e-3} = \qty{1.0e5}{V/m}$.
\textbf{3.} A carga é negativa, de modo que a \omterm{def:g10:inertia:force}{força} é \emph{oposta} a
$\vect E$: para cima. Só com a placa $+$ em cima é que $\vect E$ aponta
para baixo e a \omterm{def:g10:inertia:force}{força} elétrica enfrenta o \omterm{def:g10:universal-gravitation:weight}{peso}.
\textbf{4.} $P = mg = \num{4.90e-15} \times 9.81 \approx
\qty{4.8e-14}{N}$.
\textbf{5.} $\abs{q}E = mg$.
\textbf{6.} $\abs{q} = mg/E = mgd/U$.
\textbf{7.} $\abs{q} = \num{4.81e-14} \times \num{8.0e-3} / 800
\approx \qty{4.8e-19}{C}$.
\textbf{8.} $n = \num{4.8e-19} / \num{1.60e-19} = 3$ elétrons em
excesso.
\textbf{9.} $E$ cresce, de modo que $\abs{q}E > mg$: a \omterm{def:g10:inertia:force}{força} resultante
aponta para cima.
\textbf{10.} De $\abs{q}\,U/d = mg$ vem $U = mgd/\abs{q}$ ---
inversamente proporcional à carga.
\textbf{11.} $\abs{q} = mgd/U = \num{3.85e-16}/1200 \approx
\qty{3.2e-19}{C} = 2e$: de $3e$ para baixo, a gotícula \emph{perdeu} um
elétron.
\textbf{12.} $\abs{q} = \num{3.85e-16}/U$: \qty{3.2e-19}{C},
\qty{4.8e-19}{C}, \qty{6.4e-19}{C}, \qty{8.0e-19}{C},
\qty{9.6e-19}{C}.
\textbf{13.} Elas são $2e, 3e, 4e, 5e, 6e$ com
$e = \qty{1.60e-19}{C}$.
\textbf{14.} Cada degrau muda $\abs{q}$ em exatamente
\qty{1.6e-19}{C}: um elétron pousa de cada vez.
\textbf{15.} Isso exigiria $\abs{q} = \num{3.85e-16}/700 \approx
\qty{5.5e-19}{C} = 3.4\,e$ --- que não é um número inteiro de elétrons,
de modo que pulso de radiação algum consegue produzi-lo.
\textbf{16.} A carga é \emph{quantizada}: ela vem em múltiplos inteiros
da \omterm{def:g11:fundamental-interactions:elementary}{carga elementar} $e = \qty{1.60e-19}{C}$ --- a grandeza que Millikan
mediu.
\textbf{17.} $q \leftrightarrow m$, $E \leftrightarrow g$ e
$\abs{q}E \leftrightarrow mg$: o próprio \omterm{def:g10:universal-gravitation:weight}{peso}.
\textbf{18.} $g_{\text{esfera}} = GM/d^2 = \num{6.67e-11} \times 1000 /
1.0^2 = \qty{6.7e-8}{N/kg}$; \omterm{def:g10:inertia:force}{força} sobre a gotícula
$\num{4.90e-15} \times \num{6.7e-8} \approx \qty{3.3e-22}{N}$ --- cerca
de $\num{1.5e8}$ vezes menor do que o \omterm{def:g10:universal-gravitation:weight}{peso} dela. Sem chance.
\textbf{19.} Como $G$ é minúsculo, só uma massa do tamanho de um planeta
colocada acima rivalizaria com o puxão da Terra --- e massa alguma
repele. Já o enorme $k$ e os dois sinais de carga permitem que uma placa
de bancada supere o planeta, em qualquer dos dois sentidos.
\textbf{20.} Um elétron: $\Delta\abs{q}/\abs{q} = e/3e \approx 33\%$; uma
\omterm{def:g11:fundamental-interactions:ladder}{molécula} de óleo:
$\Delta m/m = \num{5e-25} / \num{4.90e-15} \approx \num{e-10}$. A escada
de carga desloca a tensão de equilíbrio em centenas de \omterm{def:g11:circuits-and-power:voltage}{volts}; a escada de
massa é invisível em qualquer tensão. O equilíbrio, portanto, resolve o
\omterm{def:g11:fundamental-interactions:ladder}{átomo} da eletricidade --- $e = \qty{1.60e-19}{C}$ --- enquanto as
\omterm{def:g11:fundamental-interactions:ladder}{moléculas} de óleo se diluem num contínuo.
\end{solution}
