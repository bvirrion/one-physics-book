\section*{Hoofdstuk \ref{ch:g11:nucleus-radioactivity} --- De atoomkern en radioactiviteit}

\begin{solution}{exo:g11:nucleus-radioactivity:1}
${}^{16}_{8}\mathrm{O}$: $8$ p, $8$ n; ${}^{27}_{13}\mathrm{Al}$: $13$ p,
$14$ n; ${}^{60}_{27}\mathrm{Co}$: $27$ p, $33$ n;
${}^{235}_{92}\mathrm{U}$: $92$ p, $143$ n; ${}^{3}_{1}\mathrm{H}$: $1$ p,
$2$ n. (Telkens $N = A - Z$.)
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:2}
De drie koolstofsoorten ($Z = 6$) zijn \omterm{def:g11:nucleus-radioactivity:notation}{isotopen} van elkaar.
${}^{14}_{6}\mathrm{C}$ en ${}^{14}_{7}\mathrm{N}$ delen $A = 14$ maar
niet $Z$: het zijn verschillende elementen (met verschillende
elektronenwolken), dus geen \omterm{def:g11:nucleus-radioactivity:notation}{isotopen}.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:3}
$A = 210 - 4 = 206$, $Z = 84 - 2 = 82$:
${}^{210}_{84}\mathrm{Po} \to {}^{206}_{82}\mathrm{Pb} +
{}^{4}_{2}\mathrm{He}$ --- de dochter is lood-206.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:4}
${}^{60}_{27}\mathrm{Co} \to {}^{60}_{28}\mathrm{Ni} +
{}^{\;0}_{-1}\mathrm{e}$ en
${}^{131}_{53}\mathrm{I} \to {}^{131}_{54}\mathrm{Xe} +
{}^{\;0}_{-1}\mathrm{e}$: $A$ blijft gelijk, $Z$ gaat één omhoog.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:5}
$16$ dagen $= 2\,T_{1/2}$: $800/4 = \qty{200}{MBq}$. $32$ dagen
$= 4\,T_{1/2}$: $800/16 = \qty{50}{MBq}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:6}
(a) $\alpha$ (zwaar, dus nauwelijks afgebogen); (b) $\beta^-$;
(c) $\gamma$ (neutraal, doordringend); (d) $\beta^+$ (een \omterm{rem:g11:nucleus-radioactivity:betaplus}{positron}: even
licht als (b), met tegengestelde lading).
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:7}
${}^{222}_{86}\mathrm{Rn} \to {}^{218}_{84}\mathrm{Po} +
{}^{4}_{2}\mathrm{He}$; \
${}^{218}_{84}\mathrm{Po} \to {}^{214}_{82}\mathrm{Pb} +
{}^{4}_{2}\mathrm{He}$; \
${}^{214}_{82}\mathrm{Pb} \to {}^{214}_{83}\mathrm{Bi} +

\begin{solution}{exo:g11:nucleus-radioactivity:8}
Fluor-18 heeft $9$ neutronen waar het stabiele fluor-19 er $10$ heeft:
neutronenarm, onder de band, dus $\beta^+$:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:9}
$24$ uur $= 4\,T_{1/2}$: $500/16 \approx \qty{31}{MBq}$. Omdat
$2^9 = 512$, brengen negen \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} $500/512 < \qty{1}{MBq}$: na
$9 \times 6 = \qty{54}{h}$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:10}
$\frac18 = \frac1{2^3}$: drie \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}, dus
$3 \times \num{5700} = \num{17100}$ jaar. Een dinosaurusbot is ongeveer
$\num{17500}$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} oud: er blijft een deel $1/2^{17500}$ over
--- geen enkel \omterm{def:g11:fundamental-interactions:ladder}{atoom} koolstof-14 (een gram koolstof bevat maar zo'n
$\num{5e22}$ atomen).
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:11}
${}^{14}_{6}\mathrm{C}$: $8$ neutronen tegen $6$--$7$ in stabiele
koolstof, dus boven de band: $\beta^-$. ${}^{11}_{6}\mathrm{C}$: $5$
neutronen, neutronenarm: $\beta^+$. ${}^{238}_{92}\mathrm{U}$:
$Z = 92 > 83$, te zwaar: $\alpha$.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:12}
Alleen $\alpha$ verandert $A$: $238 - 206 = 32 = 4x$, dus $x = 8$
alfastappen. Die halen $16$ van $Z$ af: uit $92 - 16 + y = 82$ volgt
$y = 6$ bèta-minvervallen.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:13}
$6.4/0.1 = 64 = 2^{6}$: zes \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}, dus
$6 \times 5.3 = 31.8 \approx 32$ jaar in afgeschermde opslag.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:14}
Per dag: $8000 \times 86400 \approx \num{6.9e8}$ vervallen. Over $80$
jaar: $\num{6.9e8} \times 365 \times 80 \approx \num{2e13}$. Het leven
draait al altijd op deze achtergrond --- één \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} is één \omterm{def:g11:fundamental-interactions:ladder}{atoom} per
seconde van de $\sim 10^{27}$ in het lichaam; de \omterm{def:g10:orders-of-magnitude:unit}{eenheid} is minuscuul, en
een angstaanjagend klinkend aantal \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} kan de waarde van een paar
bananen zijn.
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:15}
$\num{25000} \times 86400 \approx \num{2.2e9}$ vervallen per dag. Tien
jaar is $10/432 \approx 2\%$ van een \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}: er is veel minder
dan de helft vervallen, dus is niet de bron maar zijn de elektronica en
het stof de reden voor vervanging. Aan het plafond sterven de $\alpha$'s
in de lucht van het kamertje en in de behuizing; ingeademd stof parkeert
de straler in de longen, waar elke $\alpha$ zijn \omterm{def:g10:energy-conservation:energy}{energie} in levend
weefsel dumpt --- een grote \omterm{rem:g11:nucleus-radioactivity:dose}{dosis} (in \omterm{rem:g11:nucleus-radioactivity:dose}{sievert}) uit een bescheiden
\omterm{def:g11:nucleus-radioactivity:activity}{activiteit}.
\end{solution}

\begin{solution}{pb:g11:nucleus-radioactivity:1}
\textbf{1.} ${}^{12}_{6}\mathrm{C}$: $6$ p, $6$ n;
${}^{14}_{6}\mathrm{C}$: $6$ p, $8$ n. De scheikunde ziet alleen de
elektronenwolk, vastgelegd door $Z = 6$: identiek gedrag.

\textbf{2.} Eten en ademen vernieuwen de koolstof van een levend lichaam
voortdurend op het atmosferische aandeel; bij de dood stopt de aanvoer en
loopt het verval ongehinderd door.

\textbf{3.} ${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} +
{}^{\;0}_{-1}\mathrm{e}$ ($\beta^-$). Die $\beta$'s zijn zacht: \omterm{def:g10:orders-of-magnitude:unit}{meters}
gewikkeld linnen houden ze tegen, dus ontsnapt er weinig uit de mummie
--- en daarom meet men een klein koolstofmonster dat uit het voorwerp
zelf is genomen.

\textbf{4.} $6.8/13.6 = \frac12$: één \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}. De mummie is
ongeveer \num{5700} jaar oud.

\textbf{5.} $13.2/13.6 \approx 0.97$: het linnen is in wezen modern ---
hoogstens een paar eeuwen oud. Een vervalsing.

\textbf{6.} $13.6/2^{10} = 13.6/1024 \approx 0.013$ vervallen per minuut
--- één telling per $75$ minuten per gram, verdronken in de natuurlijke
achtergrond. Tien \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}, ongeveer \num{57000} jaar, is de
praktische horizon.

\textbf{7.} $92$ protonen, $146$ neutronen. De coulombafstoting werkt
tussen alle protonenparen door de hele kern heen, terwijl de sterke lijm
alleen buren bindt: zware kernen overleven alleen met overtollige
neutronen, lijm zonder afstoting.

\textbf{8.} ${}^{238}_{92}\mathrm{U} \to {}^{234}_{90}\mathrm{Th} +
{}^{4}_{2}\mathrm{He}$.

\textbf{9.} ${}^{234}_{90}\mathrm{Th} \to {}^{234}_{91}\mathrm{Pa} +
{}^{\;0}_{-1}\mathrm{e}$, en daarna
${}^{234}_{91}\mathrm{Pa} \to {}^{234}_{92}\mathrm{U} +
{}^{\;0}_{-1}\mathrm{e}$: uranium duikt weer op, als uranium-234.

\textbf{10.} Uit $A$: $238 - 206 = 32$, dus $8$ $\alpha$-stappen; uit
$Z$: $92 - 16 + y = 82$, dus $6$ $\beta^-$-stappen.

\textbf{11.} $\num{4.5e9}$ jaar is één \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}: de helft is over.
Nog vijf miljard jaar $\approx$ nog één \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}: ongeveer een
kwart.

\textbf{12.} Ze drijft vulkanen, platentektoniek (drijvende continenten,
aardbevingen) en aardwarmte aan: de oppervlaktegeologie van een planeet
die van binnenuit wordt verwarmd.

\textbf{13.} ${}^{241}_{95}\mathrm{Am} \to {}^{237}_{93}\mathrm{Np} +
{}^{4}_{2}\mathrm{He}$.

\textbf{14.} $\num{3.3e4}$ vervallen per seconde;
$\num{33000} \times 86400 \approx \num{2.9e9}$ per dag.

\textbf{15.} Elke $\alpha$ ioniseert de lucht die hij doorkruist; de ionen
dragen een minieme stroom tussen de elektroden van het kamertje.
Rookdeeltjes vangen de ionen weg, de stroom zakt, en het alarm gaat af.

\textbf{16.} De $\alpha$'s besteden al hun \omterm{def:g10:energy-conservation:energy}{energie} binnen in het kamertje
en er ontsnapt niets uit de behuizing: maximale ionisatie, nul lekkage.
Een $\gamma$-bron met dezelfde \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} zou de lucht in het kamertje
nauwelijks ioniseren en in plaats daarvan de hele kamer bestralen.

\textbf{17.} Tien jaar is $10/432 \approx 2\%$ van een \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}: de
\omterm{def:g11:nucleus-radioactivity:activity}{activiteit} is vrijwel onveranderd. De zwakke schakels zijn stof, insecten
en elektronica --- de kern overleeft het apparaat.

\textbf{18.} $\num{9e9}$ jaar $= 2$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}: een kwart is over.
Het uranium van de Aarde is nog volop aanwezig, dus is het hoogstens een
paar \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} geleden gesmeed --- de planeet is uit verse
sterrenas gecondenseerd, miljarden jaren na de eerste sterren.

\textbf{19.} Geologisch dood: een koud binnenste, geen vulkanisme of
platentektoniek om lucht en gesteente te vernieuwen, en geen kolkende
kern --- en dus geen magnetisch schild tegen de zonnewind.

\textbf{20.} Elk \omterm{def:g11:fundamental-interactions:ladder}{atoom} in jou zwaarder dan helium is in een stervende ster
gesmeed, en de onstabiele restanten van die smidse verwarmen nog altijd de
grond waarop je staat: sterrenstof, in leven gehouden door zijn eigen
\omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}.
\end{solution}
