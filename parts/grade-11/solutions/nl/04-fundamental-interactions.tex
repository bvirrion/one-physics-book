\begin{solution}{exo:g11:fundamental-interactions:1}
Proton: $10^{-15}$ (verdieping van de \omterm{def:g11:fundamental-interactions:ladder}{nucleonen}); watermolecuul:
$10^{-10}$ (verdieping van de \omterm{def:g11:fundamental-interactions:ladder}{moleculen} --- losse \omterm{def:g11:fundamental-interactions:ladder}{moleculen} liggen op

$10^{7}$ (planeet); Melkweg: $10^{21}$ (melkwegstelsel). Alles in \omterm{def:g10:orders-of-magnitude:unit}{meter}.
\end{solution}

\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:6}
$F_E = \num{8.99e9} \times (\num{1.60e-19})^2/(\num{2.0e-15})^2
= \num{2.30e-28}/\num{4.0e-30} \approx \qty{58}{N}$;
$F_G = \num{6.67e-11} \times (\num{1.67e-27})^2/\num{4.0e-30}
\approx \qty{4.7e-35}{N}$. Verhouding $F_E/F_G \approx \num{1.2e36}$. De
\omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking} houdt de \omterm{def:g11:fundamental-interactions:ladder}{kern} bijeen; de zwaartekracht doet niet
mee.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:7}
$F_E = \num{2.30e-28}/(\num{5.3e-11})^2 = \num{2.30e-28}/\num{2.81e-21}
\approx \qty{8.2e-8}{N}$;
$F_G = \num{6.67e-11} \times \num{1.67e-27} \times \num{9.11e-31} /
\num{2.81e-21} \approx \qty{3.6e-47}{N}$. Verhouding

bijeen.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:8}
Door het contact wordt de lading verdeeld:
$q_A = q_B = \qty{+4.0e-9}{C}$ (samen nog altijd $\qty{8.0e-9}{C}$).
Daarna is $F = \num{8.99e9} \times (\num{4.0e-9})^2/0.10^2 =
\qty{1.4e-5}{N}$, afstotend (gelijke tekens).
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:9}
$d = \sqrt{k q^2/F} = \sqrt{\num{8.99e9} \times \num{1.0e-12}/0.90}
\approx \qty{0.10}{m}$. Een $9$ keer kleinere \omterm{def:g10:inertia:force}{kracht} vergt een $3$ keer
grotere afstand: $d = \qty{0.30}{m}$.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:10}
$F = \num{2.30e-28}/(\num{1.0e-10})^2 = \qty{2.3e-8}{N}$. De \omterm{def:g11:fundamental-interactions:strong}{sterke
wisselwerking} draagt niets bij: $10^{-10}$~m is $10^{5}$ keer haar
\omterm{def:g11:fundamental-interactions:strong}{dracht}. \omterm{def:g11:fundamental-interactions:ladder}{Moleculen} worden gebonden door de \omterm{def:g11:fundamental-interactions:four}{elektromagnetische
wisselwerking}.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:11}
$q = \num{e23}/\num{e6} \times \num{1.6e-19} = \qty{1.6e-2}{C}$ op elke
munt. $F = \num{8.99e9} \times (\num{1.6e-2})^2/1.0^2 \approx
\qty{2.3e6}{N}$ --- het \omterm{def:g10:universal-gravitation:weight}{gewicht} van $\num{2.3e6}/9.81 \approx
\qty{2.3e5}{kg}$, een volgeladen goederentrein, uit één miljoenste van
de elektronen. Alledaagse materie moet (en blijkt) veel nauwkeuriger dan
één deel op een miljoen neutraal te zijn.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:12}
$F = \num{2.30e-28}/(\num{1.4e-14})^2 = \num{2.30e-28}/\num{1.96e-28}
\approx \qty{1.2}{N}$ --- op deze schaal nog altijd enorm. Nee: ze liggen
$14$ keer de sterke \omterm{def:g11:fundamental-interactions:strong}{dracht} uit elkaar; elk is alleen aan zijn naaste
buren gelijmd. Paren: $\frac{92 \times 91}{2} = 4186$, die \emph{alle}
op elke afstand afstoten, terwijl de lijm niet met de grootte meegroeit.
Neutronen voegen aantrekking zonder afstoting toe en verdunnen de
protonen; maar de afstoting groeit met het kwadraat van het aantal
protonen en de lijm alleen met het aantal buren, zodat voorbij ongeveer
$80$ protonen geen enkele neutronenbegroting de boeken sluitend krijgt
--- zulke \omterm{def:g11:fundamental-interactions:ladder}{kernen} vervallen
(\cref{ch:g11:nucleus-radioactivity}).
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:13}
\omterm{def:g10:inertia:compensate}{Evenwicht}: verticaal $T\cos\theta = mg$, horizontaal $T\sin\theta = F$,
dus $F = mg\tan\theta$. Hier is $\sin\theta = 3.0/50 = 0.060$, en voor
kleine hoeken $\tan\theta \approx \sin\theta = 0.060$. Dan is
$F = \num{1.0e-3} \times 9.81 \times 0.060 \approx \qty{5.9e-4}{N}$;
$q = d\sqrt{F/k} = 0.060 \times \sqrt{\num{5.9e-4}/\num{8.99e9}}
\approx \qty{1.54e-8}{C}$; $N = \num{1.54e-8}/\num{1.6e-19} \approx
\num{9.6e10}$ \omterm{def:g11:fundamental-interactions:elementary}{elementaire ladingen}.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:14}
$F = \num{6.67e-11} \times \num{5.97e24} \times \num{7.35e22} /
(\num{3.84e8})^2 = \num{2.93e37}/\num{1.47e17} \approx \qty{2.0e20}{N}$.
Stel $kQ^2/d^2 = F$:
$Q = d\sqrt{F/k} = \num{3.84e8} \times \sqrt{\num{2.0e20}/\num{8.99e9}}
\approx \qty{5.7e13}{C}$. Dat zijn
$N = \num{5.7e13}/\num{1.6e-19} \approx \num{3.6e32}$ elektronen, met
massa $\num{3.6e32} \times \num{9.11e-31} \approx \qty{3.2e2}{kg}$: een
paar honderd \omterm{def:g10:orders-of-magnitude:unit}{kilogram} elektronen zou het werk van $\qty{7.35e22}{kg}$
Maan kunnen doen --- de \omterm{def:g10:inertia:force}{kracht} van de elektriciteit en de zwakte van de
zwaartekracht in één getal.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:15}
$F = \num{2.30e-28}/(\num{2.8e-10})^2 = \num{2.30e-28}/\num{7.84e-20}
\approx \qty{2.9e-9}{N}$; \omterm{def:g10:universal-gravitation:weight}{gewicht}
$P = \num{5.9e-26} \times 9.81 \approx \qty{5.8e-25}{N}$; verhouding
$F/P \approx \num{5e15}$. Elke binding heeft een \emph{vaste} sterkte,
maar het \omterm{def:g10:universal-gravitation:weight}{gewicht} groeit met de hele massa erboven: stapel gesteente op en
de last op de onderste laag groeit onbegrensd terwijl haar bindingen dat
niet doen. Rond \qty{e4}{m} gesteente verplettert de last de elektrische
bindingen --- de zwaartekracht wint niet door \omterm{def:g10:inertia:force}{kracht} per deeltje, maar
doordat ze \emph{optelt} zodra de lading zichzelf opheft.
\end{solution}

\begin{solution}{pb:g11:fundamental-interactions:1}
\textbf{1.} \omterm{def:g11:fundamental-interactions:ladder}{Quark} $< \qty{e-18}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{nucleon} $\qty{e-15}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{kern}
$10^{-15}$--$\qty{e-14}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{atoom} $\qty{e-10}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{molecuul}
$\qty{e-9}{m}$; alledaags voorwerp $\qty{1}{m}$; planeet $\qty{e7}{m}$;
planetenstelsel $\qty{e11}{m}$; melkwegstelsel $\qty{e21}{m}$.

\textbf{2.} Schaalfactor $10^{5}$: het \omterm{def:g11:fundamental-interactions:ladder}{atoom} is
$\qty{1}{cm} \times 10^{5} = \qty{1}{km}$ breed.

\textbf{3.} $(10^{-15}/10^{-10})^3 = 10^{-15}$: materie is voor
$\num{99.9999999999999}\,\%$ leeg.

\textbf{4.} $\num{7e-5}/\num{e-10} = \num{7e5}$ atomen --- ongeveer een
miljoen over de breedte van één haar.

\textbf{5.} Van $10^{-18}$ tot $10^{21}$: $39$ machten van tien.

\textbf{6.} $F_E = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$.

\textbf{7.} $F_G = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31}/(\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$; verhouding
$F_E/F_G \approx \num{2.3e39}$. De elektriciteit houdt het \omterm{def:g11:fundamental-interactions:ladder}{atoom} bijeen;
de zwaartekracht kijkt toe.

\textbf{8.} Zonder zwaartekracht: er verandert niets meetbaars (ze ligt
er $10^{39}$ onder). Zonder elektriciteit: geen atomen, geen \omterm{def:g11:fundamental-interactions:ladder}{moleculen},
helemaal geen materie.

\textbf{9.} ``Contact'' is de elektrische afstoting tussen de
elektronenwolken van atomen die tot ongeveer $\qty{e-10}{m}$ bij elkaar
zijn gebracht: niets raakt ooit iets aan. De \omterm{def:g10:inertia:inventory}{normaalkracht} van de vloer
en de \omterm{def:g10:inertia:inventory}{wrijving} zijn de wet van Coulomb in het groot.

\textbf{10.} $q = \num{e23} \times \num{e-9} \times \num{1.6e-19} =
\qty{1.6e-5}{C}$ per munt;
$F = \num{8.99e9} \times (\num{1.6e-5})^2 \approx \qty{2.3}{N}$ --- een
duidelijk zichtbare \omterm{def:g10:inertia:force}{kracht} uit één elektron op een miljard. Alledaagse
materie is veel nauwkeuriger dan $10^{-9}$ neutraal.

\textbf{11.} De \omterm{def:g11:fundamental-interactions:four}{elektromagnetische wisselwerking}. De sterke \omterm{def:g10:inertia:force}{kracht} is
buiten \omterm{def:g11:fundamental-interactions:strong}{dracht} voorbij ongeveer $\qty{e-14}{m}$ --- $10^{4}$ keer kleiner
dan één \omterm{def:g11:fundamental-interactions:ladder}{atoom}.

\textbf{12.} $F = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{1.0e-15})^2 \approx \qty{2.3e2}{N}$.

\textbf{13.} $a = F/m_p = 230/\num{1.67e-27} \approx
\qty{1.4e29}{m/s^2}$, ongeveer $\num{1.4e28}$ maal $g$: geen alledaagse
\omterm{def:g10:inertia:force}{kracht} komt er in de buurt.

\textbf{14.} Sterker dan \qty{230}{N} op $\qty{e-15}{m}$ (\omterm{def:g11:fundamental-interactions:ladder}{kernen}
bestaan); even hard werkend op protonen en neutronen, blind voor lading
(neutronen worden ook vastgehouden); een \omterm{def:g11:fundamental-interactions:strong}{dracht} van ongeveer
$\qty{e-15}{m}$ (\omterm{def:g11:fundamental-interactions:ladder}{kernen} blijven klein --- de lijm reikt niet verder).

\textbf{15.} $F = \num{2.30e-28}/(\num{1.4e-14})^2 \approx
\qty{1.2}{N}$. Nee: $14$ keer de \omterm{def:g11:fundamental-interactions:strong}{dracht}. De afstoting werkt over
\emph{alle} $\frac{92\times91}{2} = 4186$ protonenparen; de lijm bindt
alleen naaste buren, dus komt groei de afstoting ten goede. Neutronen
voegen lijm en ruimte toe zonder afstoting toe te voegen --- tot voorbij
ongeveer $80$ protonen geen enkel mengsel meer in \omterm{def:g10:inertia:compensate}{evenwicht} is en de \omterm{def:g11:fundamental-interactions:ladder}{kern}
vervalt.

\textbf{16.} De \omterm{def:g11:fundamental-interactions:weak}{zwakke wisselwerking} verandert een neutron in een proton
(en omgekeerd) en veroorzaakt zo het $\beta$-verval --- ze zet om; ze
bindt nooit.

\textbf{17.} $F = \num{6.67e-11} \times \num{5.97e24} \times
\num{7.35e22}/(\num{3.84e8})^2 \approx \qty{2.0e20}{N}$.

\textbf{18.} $Q_E = \num{1.8e51} \times \num{1.6e-19} =
\qty{2.9e32}{C}$, $Q_M = \qty{3.5e30}{C}$. Gelijke \omterm{def:g10:inertia:force}{krachten} betekent
$f^2\,k\,Q_E Q_M = G M_E M_M$:
\[
f = \sqrt{\frac{\num{2.93e37}}{\num{8.99e9} \times \num{2.9e32}
\times \num{3.5e30}}} = \sqrt{\num{3.2e-36}} \approx \num{1.8e-18} :
\]
een onbalans van \emph{twee delen op $10^{18}$} zou de zwaartekracht
opheffen.

\textbf{19.} Lading bestaat in twee tekens en heft zichzelf op (vraag 18
laat zien hoe volmaakt), zodat de sterkere \omterm{def:g10:inertia:force}{kracht} zichzelf op grote
schaal uitschakelt; massa heeft één teken, telt eeuwig op, en niets
schermt haar af.

\textbf{20.} \omterm{def:g11:fundamental-interactions:ladder}{Kern}: de \omterm{def:g11:fundamental-interactions:strong}{sterke wisselwerking} overtroeft op $\qty{e-15}{m}$
de afstoting tussen de protonen. \omterm{def:g11:fundamental-interactions:ladder}{Atoom} tot berg: de \omterm{def:g11:fundamental-interactions:four}{elektromagnetische
wisselwerking} bindt elektronen aan \omterm{def:g11:fundamental-interactions:ladder}{kernen} en atomen aan elkaar. Planeet
en groter: de \omterm{def:g11:fundamental-interactions:four}{gravitatie}, onbestreden zodra de lading zich opheft, houdt
planeten, omloopbanen en melkwegstelsels vast. De wereld stort niet in
doordat elke verdieping wordt geschraagd door een bindmiddel dat stijver
is dan de last, en vliegt niet uiteen doordat materie tot op ongeveer
$10^{-18}$ neutraal is --- zodat op elke schaal precies één \omterm{def:g10:inertia:force}{kracht} de
leiding heeft, en aantrekkend is waar dat moet.
\end{solution}
