\begin{solution}{exo:g11:work-of-force:2}

\end{solution}

\begin{solution}{exo:g11:work-of-force:3}
(क) \omterm{def:g11:work-of-force:sign}{प्रेरक}; (ख) \omterm{def:g11:work-of-force:sign}{प्रतिरोधी}; (ग) शून्य (लंबवत);
(घ) \omterm{def:g11:work-of-force:sign}{प्रतिरोधी}; (ङ) शून्य --- डोरी का \omterm{def:g10:inertia:inventory}{तनाव} गोलक की गति के

\end{solution}

\begin{solution}{exo:g11:work-of-force:4}
अचर चाल: उठाने वाला बल $= mg = \qty{5.89e3}{N}$, इसलिए $W_{\text{उठाव}} = 5886 \times 25 \approx +\qty{147}{kJ}$ और $W(\vect P) = -\qty{147}{kJ}$। योग शून्य:
अचर वेग पर बल संतुलित रहते हैं (\cref{ch:g11:forces-and-motion}), इसलिए उनके कार्य भी
कट जाते हैं।
\end{solution}

\begin{solution}{exo:g11:work-of-force:5}
$P = \num{9.0e4}/60 = \qty{1.5}{kW}$; $1500/736 \approx \qty{2.0}{hp}$।
\end{solution}

\begin{solution}{exo:g11:work-of-force:6}
रस्सी: $120 \times 200 \times \cos\qty{30}{\degree} \approx
+\qty{20.8}{kJ}$; \omterm{def:g10:inertia:inventory}{घर्षण}: $-45 \times 200 = -\qty{9.0}{kJ}$; भार और \omterm{def:g10:refraction:angles}{अभिलंब} बल,
दोनों लंबवत: $0$। कुल $+\qty{11.8}{kJ}$: स्लेज की चाल बढ़ती है।
\end{solution}

\begin{solution}{exo:g11:work-of-force:7}
$v = 25/3.6 = \qty{6.94}{m/s}$; $P = Fv = 18 \times 6.94 \approx
\qty{125}{W}$। एक घंटे में $W = P\,\Delta t = 125 \times 3600 =
\qty{450}{kJ}$ (या यों कहिए, $18 \times \qty{25000}{m}$)।
\end{solution}

\begin{solution}{exo:g11:work-of-force:8}
भार, दोनों रास्तों पर: $+65 \times 9.81 \times 120 \approx
+\qty{76.5}{kJ}$ (केवल गिरावट)। \omterm{def:g10:inertia:inventory}{घर्षण}: ढलान पर
$-30 \times 800 =
-\qty{24}{kJ}$, रास्ते पर $-30 \times 1500 = -\qty{45}{kJ}$। रास्ते की परवाह केवल \omterm{def:g10:inertia:inventory}{घर्षण} करता है।
\end{solution}

\begin{solution}{exo:g11:work-of-force:9}
चढ़ाई $h = 4.0\sin\qty{25}{\degree} = \qty{1.69}{m}$: $W(\vect P) = -20 \times 9.81 \times 1.69 \approx -\qty{332}{J}$; $W_f = -30 \times 4.0 = -\qty{120}{J}$। अचर चाल: कुल कार्य शून्य, इसलिए
$W_{\text{धक्का}} = +\qty{452}{J}$ और $F = 452/4.0 \approx \qty{113}{N}$।
\end{solution}

\begin{solution}{exo:g11:work-of-force:10}
$v = 130/3.6 = \qty{36.1}{m/s}$; अचर चाल है, इसलिए $F = P/v = \num{5.0e4}/36.1 \approx \qty{1.4}{kN}$।
\end{solution}

\begin{solution}{exo:g11:work-of-force:11}
$P = Fv\cos\theta = 800 \times 1.0 \times \cos\qty{20}{\degree}
\approx \qty{752}{W}$; \omterm{def:g11:work-of-force:power}{अश्वशक्ति} में $752/736 \approx \num{1.02}$ --- लगभग ठीक उतना ही जितना कोई मज़बूत
घोड़ा लगातार दे सकता है, यानी वही तुलना जिस पर \omterm{def:g11:work-of-force:power}{वाट} ने अपना मात्रक
गढ़ा था।
\end{solution}

\begin{solution}{exo:g11:work-of-force:12}
लंबाइयाँ $L = h/\text{ढाल}$: \qty{1500}{m} और \qty{5000}{m}। बल $F = mg \times \text{ढाल}$: $1200 \times 9.81 \times 0.20 \approx
\qty{2.35}{kN}$ और $\qty{0.71}{kN}$।
कार्य $FL$: $2354 \times 1500
\approx \qty{3.53}{MJ}$ और $706 \times 5000 \approx \qty{3.53}{MJ}$ --- दोनों $mgh$ के बराबर: सड़क
बल के बदले लंबाई देती है, कार्य के बदले कभी नहीं।
\end{solution}

\begin{solution}{exo:g11:work-of-force:13}
$F = mg = \qty{2.45e3}{N}$; $W = 2453 \times 15 \approx \qty{36.8}{kJ}$; $\Delta t = 15/0.40 = \qty{37.5}{s}$; $P = 36800/37.5 \approx \qty{981}{W} = 2453 \times 0.40$ ($Fv$ से मेल खाता है); विद्युत: $981/0.70 \approx \qty{1.4}{kW}$।
\end{solution}

\begin{solution}{exo:g11:work-of-force:14}
विकर्ण: $-140 \times 6.0 = -\qty{840}{J}$; दीवारें: $-140 \times 8.4 \approx -\qty{1.18}{kJ}$; आने-जाने का चक्कर: $-140 \times 12 = -\qty{1.68}{kJ}$। चक्कर में
भार शून्य करता है (कुल गिरावट कोई नहीं)। \omterm{def:g10:inertia:inventory}{घर्षण} का
कार्य पथ पर निर्भर करता है और कभी लौटाता नहीं; भार का केवल
सिरों पर निर्भर करता है (\cref{rem:g11:work-of-force:divide})।
\end{solution}

\begin{solution}{exo:g11:work-of-force:15}
गुरुत्व: $80 \times 9.81 \times 600 \approx \qty{471}{kJ}$; \omterm{def:g10:inertia:inventory}{घर्षण}: $14 \times 12000 = \qty{168}{kJ}$; कुल $\qty{639}{kJ}$। $v = \qty{4.17}{m/s}$, इसलिए $\Delta t = 12000/4.17 = \qty{2880}{s} =
\qty{48}{min}$
और $P = \num{6.39e5}/2880 \approx \qty{222}{W} \approx
\qty{0.30}{hp}$ --- यानी एक मज़बूत शौक़िया सवार, घोड़े का तिहाई।
\end{solution}

\begin{solution}{pb:g11:work-of-force:1}
\textbf{1.} $W = 110 \times 8.0 = +\qty{880}{J}$।

\textbf{2.} \omterm{def:g10:inertia:inventory}{घर्षण}: $-95 \times 8.0 = -\qty{760}{J}$। भार और \omterm{def:g10:refraction:angles}{अभिलंब} बल गति
के लंबवत हैं: दोनों शून्य।

\textbf{3.} कुल $+\qty{120}{J}$: शुद्ध \omterm{def:g11:work-of-force:sign}{प्रेरक कार्य}, इसलिए डिब्बे की चाल
बढ़ती है। \qty{95}{N} का धक्का \omterm{def:g10:inertia:inventory}{घर्षण} को काट देता है --- \omterm{def:g11:forces-and-motion:netforce}{परिणामी
बल} शून्य, चाल अचर, कुल कार्य शून्य।

\textbf{4.} विकर्ण: $-95 \times 10.0 = -\qty{950}{J}$; दीवारें: $-95 \times 14.0 = -\qty{1.33}{kJ}$: \omterm{def:g10:inertia:inventory}{घर्षण} \omterm{def:g10:orders-of-magnitude:unit}{मीटर}
के हिसाब से वसूलता है।

\textbf{5.} \omterm{def:g10:inertia:inventory}{घर्षण}: $-95 \times 20.0 = -\qty{1.9}{kJ}$; भार: $0$ (कुल गिरावट कोई
नहीं)। लौटाता केवल भार है।

\textbf{6.} $W(\vect P) = -30 \times 9.81 \times 9.0 =
-\qty{2.65}{kJ}$; टोली को कम से कम $+\qty{2.65}{kJ}$ देना ही होगा।

\textbf{7.} $(70 + 30) \times 9.81 \times 9.0 = \qty{8.83}{kJ}$; डिब्बे का हिस्सा $2.65/8.83 \approx 30\%$ है --- यानी ढोना असल में
ज़्यादातर अपने आप को ढोना है।

\textbf{8.} $\sin\alpha = 9.0/30 = 0.30$: $F = mg\sin\alpha + f = 88.3 + 40 = \qty{128}{N}$; $W = 128 \times 30 = \qty{3.85}{kJ}$ ($= 2.65 + 1.2$)।

\textbf{9.} \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} उठाव: $F = mg = \qty{294}{N}$, $W = \qty{2.65}{kJ}$। ढलान छोटा बल ख़रीद
देती है (\qty{294}{N} के मुक़ाबले $128$), पर उसका दाम अधिक कार्य है:
\omterm{def:g10:inertia:inventory}{घर्षण} की \qty{1.2}{kJ} चुंगी।

\textbf{10.} $\Delta t = 9.0/0.50 = \qty{18}{s}$; $P = 2649/18 \approx \qty{147}{W}$।

\textbf{11.} $P = Fv = 294 \times 0.50 = \qty{147}{W}$ --- वही। विद्युत: $147/0.60 \approx \qty{245}{W}$।

\textbf{12.} $1000 \times 0.24 = \qty{240}{m}$ और $8000 \times 0.030 = \qty{240}{m}$: वही चोटी।

\textbf{13.} $F = mg \times \text{ढाल}$: पगडंडी $3500 \times 9.81 \times 0.24 \approx \qty{8.2}{kN}$; सड़क $\approx \qty{1.0}{kN}$।

\textbf{14.} $8240 \times 1000 \approx \qty{8.24}{MJ}$ और $1030 \times 8000 \approx \qty{8.24}{MJ}$ --- दोनों $= mgh = 3500
\times 9.81 \times 240$।

\textbf{15.} लोटनी प्रतिरोध: $400 \times 8000 = \qty{3.2}{MJ}$ के मुक़ाबले $400 \times 1000 = \qty{0.40}{MJ}$: हलकी
सड़क का आराम \omterm{def:g10:inertia:inventory}{घर्षण} के \qty{2.8}{MJ} अतिरिक्त के दाम पर मिलता है।

\textbf{16.} $v = \qty{10}{m/s}$। सड़क: $(1030 + 400) \times 10 = \qty{14.3}{kW} \approx \qty{19}{hp}$; पगडंडी: $(8240 + 400) \times 10 = \qty{86.4}{kW} \approx \qty{117}{hp}$ --- यानी लदी वैन के बस
से बाहर। सड़कें इसलिए घुमावदार होती हैं कि मामूली इंजन (और टायर) चढ़ सकें:
वही कार्य, बस पतला फैलाकर।

\textbf{17.} प्रति डिब्बा: $880 + 2649 = \qty{3.53}{kJ}$; $60$ डिब्बों के लिए $60 \times 3529 \approx \qty{212}{kJ}$।

\textbf{18.} सीढ़ियाँ: $20 \times 70 \times 9.81 \times 9.0 \approx
\qty{124}{kJ}$। दिन का कुल $\approx \qty{336}{kJ} = \qty{0.34}{MJ}$।

\textbf{19.} $P = \num{3.36e5}/28800 \approx \qty{12}{W}$। केतली: $\num{3.36e5}/2000 = \qty{168}{s}$ --- यानी पूरे दिन की मेहनत के बराबर
तीन मिनट से भी कम।

\textbf{20.} \omterm{def:g11:work-of-force:work}{जूल} उसी बल का दाम चुकाता है जो गति के
\emph{अनुदिश} लगे --- धकेली गई दीवार या थामे रखे डिब्बे का कुछ नहीं, चाहे
वह कितना ही थकाऊ हो (माँसपेशियाँ शून्य यांत्रिक कार्य करते हुए भी
\omterm{def:g10:energy-conservation:forms}{रासायनिक ऊर्जा} जलाती रहती हैं)। और चूँकि मनुष्य की शक्ति
कुछ दर्जन \omterm{def:g11:work-of-force:power}{वाट} भर है, इसलिए ढोने वाले बल को नए साँचे में
ढालने का काम मशीनों को सौंप देते हैं --- चरखी, ढलान, घुमावदार सड़क ---
जबकि बिल $mgh$ कभी नहीं बदलता।
\end{solution}
