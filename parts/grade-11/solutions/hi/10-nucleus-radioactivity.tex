\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:12}
$A$ केवल $\alpha$ से बदलता है: $238 - 206 = 32 = 4x$, इसलिए $x = 8$ \omterm{def:g11:nucleus-radioactivity:abg}{अल्फ़ा}। वे
$Z$ में से $16$ निकाल ले जाते हैं: $92 - 16 + y = 82$ से $y = 6$ बीटा-ऋण \omterm{def:g11:mechanical-energy:dissipation}{क्षय} मिलते

\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:13}

\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:14}
प्रतिदिन: $8000 \times 86400 \approx \num{6.9e8}$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय}। $80$ वर्ष में: $\num{6.9e8} \times 365 \times 80 \approx \num{2e13}$। जीवन हमेशा से इसी पृष्ठभूमि
पर चलता आया है --- एक \omterm{def:g11:nucleus-radioactivity:activity}{बेक्रेल} का अर्थ है शरीर के $\sim 10^{27}$ में से
प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} एक \omterm{def:g11:fundamental-interactions:ladder}{परमाणु}; मात्रक बेहद छोटा है, और
\omterm{def:g11:nucleus-radioactivity:activity}{बेक्रेल} की डरावनी लगती गिनती असल में कुछ केलों भर की हो सकती है।
\end{solution}

\begin{solution}{exo:g11:nucleus-radioactivity:15}
प्रतिदिन $\num{25000} \times 86400 \approx \num{2.2e9}$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय}। दस वर्ष \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} का $10/432 \approx 2\%$ है: आधे से कहीं
कम \omterm{def:g11:mechanical-energy:dissipation}{क्षय} हुआ है, इसलिए बदलने पर बाध्य स्रोत नहीं, बल्कि इलेक्ट्रॉनिकी
और धूल करती है। बाहर से $\alpha$ कक्ष की हवा और खोल में ही मर जाते हैं; पर
साँस में गई धूल उत्सर्जक को फेफड़ों में बिठा देती है, जहाँ हर $\alpha$ अपनी
ऊर्जा जीवित ऊतक में उड़ेल देता है --- मामूली \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} से
बड़ी \omterm{rem:g11:nucleus-radioactivity:dose}{मात्रा} (\omterm{rem:g11:nucleus-radioactivity:dose}{सीवर्ट})।
\end{solution}

\begin{solution}{pb:g11:nucleus-radioactivity:1}
\textbf{1.} ${}^{12}_{6}\mathrm{C}$: $6$ प्रोटॉन, $6$ न्यूट्रॉन; ${}^{14}_{6}\mathrm{C}$: $6$
प्रोटॉन, $8$ न्यूट्रॉन। रसायन को केवल इलेक्ट्रॉन-बादल दिखता है, जो
$Z = 6$ से तय होता है: इसलिए बरताव बिलकुल एक-सा।

\textbf{2.} खाना और साँस जीवित शरीर के कार्बन को लगातार वायुमंडलीय अनुपात पर
नया करते रहते हैं; मृत्यु पर आवक रुक जाती है और \omterm{def:g11:mechanical-energy:dissipation}{क्षय} निर्विरोध चलने लगता है।

\textbf{3.} ${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} +
{}^{\;0}_{-1}\mathrm{e}$ ($\beta^-$)। ये $\beta$ नरम हैं: लिपटी सूती पट्टियों के
\omterm{def:g10:orders-of-magnitude:unit}{मीटर} उन्हें रोक लेते हैं, इसलिए ममी से बहुत कम बाहर निकलता है ---
और इसीलिए वस्तु से ही लिया गया छोटा कार्बन-प्रतिदर्श नापा जाता है।

\textbf{4.} $6.8/13.6 = \frac12$: यानी एक \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}। ममी लगभग \num{5700} वर्ष पुरानी
है।

\textbf{5.} $13.2/13.6 \approx 0.97$: वह कपड़ा असल में आधुनिक है --- ज़्यादा से ज़्यादा कुछ
सदी पुराना। यानी जालसाज़ी।

\textbf{6.} प्रति मिनट $13.6/2^{10} = 13.6/1024 \approx 0.013$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} --- यानी प्रति ग्राम हर $75$ मिनट में
एक गिनती, जो प्राकृतिक पृष्ठभूमि में डूब जाती है। दस \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}, यानी लगभग
\num{57000} वर्ष, व्यावहारिक क्षितिज है।

\textbf{7.} $92$ प्रोटॉन, $146$ न्यूट्रॉन। \omterm{def:g11:fundamental-interactions:charge}{कूलॉम} प्रतिकर्षण
नाभिक भर में हर प्रोटॉन-युग्म पर लगता है जबकि प्रबल गोंद केवल
पड़ोसियों को बाँधती है: इसलिए भारी नाभिक केवल अतिरिक्त न्यूट्रॉनों के सहारे
बचते हैं, यानी बिना प्रतिकर्षण वाली गोंद के सहारे।

\textbf{8.} ${}^{238}_{92}\mathrm{U} \to {}^{234}_{90}\mathrm{Th} +
{}^{4}_{2}\mathrm{He}$।

\textbf{9.} ${}^{234}_{90}\mathrm{Th} \to {}^{234}_{91}\mathrm{Pa} +
{}^{\;0}_{-1}\mathrm{e}$, फिर ${}^{234}_{91}\mathrm{Pa} \to
{}^{234}_{92}\mathrm{U} + {}^{\;0}_{-1}\mathrm{e}$: यूरेनियम फिर लौट आता है, यूरेनियम-234 के
रूप में।

\textbf{10.} $A$: $238 - 206 = 32$, इसलिए $8$ $\alpha$ क़दम; $Z$: $92 - 16 + y = 82$,
इसलिए $6$ $\beta^-$ क़दम।

\textbf{11.} $\num{4.5e9}$ वर्ष एक \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} है: $\frac12$ बचा है। पाँच अरब
वर्ष और $\approx$ एक और \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}: लगभग $\frac14$।

\textbf{12.} वह ज्वालामुखी, पट्ट-विवर्तनिकी (खिसकते महाद्वीप, भूकंप) और
भूतापीय ऊष्मा चलाती है: यानी भीतर से गरम रखे गए ग्रह का सतही भूविज्ञान।

\textbf{13.} ${}^{241}_{95}\mathrm{Am} \to {}^{237}_{93}\mathrm{Np} +
{}^{4}_{2}\mathrm{He}$।

\textbf{14.} प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $\num{3.3e4}$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय}; प्रतिदिन $\num{33000} \times 86400 \approx \num{2.9e9}$।

\textbf{15.} हर $\alpha$ अपने रास्ते की हवा को आयनित कर देता है; और वे आयन
कक्ष के इलेक्ट्रोडों के बीच एक बहुत छोटी धारा ढोते हैं। धुएँ के कण आयनों को
पकड़ लेते हैं, धारा गिर जाती है, और घंटी बज उठती है।

\textbf{16.} $\alpha$ अपनी सारी ऊर्जा कक्ष के भीतर ही ख़र्च कर देते
हैं और खोल से कुछ बाहर नहीं जाता: अधिकतम आयनन, शून्य रिसाव। उतनी ही
\omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} वाला $\gamma$ स्रोत कक्ष की हवा को लगभग आयनित ही नहीं करता
और बदले में पूरे कमरे को विकिरण दे देता।

\textbf{17.} दस वर्ष \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} का $10/432 \approx 2\%$ है: \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} लगभग
अछूती रहती है। कमज़ोर कड़ी धूल, कीड़े और इलेक्ट्रॉनिकी हैं ---
नाभिक तो उपकरण से भी अधिक जीता है।

\textbf{18.} $\num{9e9}$ वर्ष $= 2$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}: $\frac14$ बचता है। पृथ्वी का
यूरेनियम आज भी भरपूर है, इसलिए वह ज़्यादा से ज़्यादा कुछ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} पहले ही गढ़ा
गया था --- यानी यह ग्रह पहले तारों के अरबों वर्ष बाद, ताज़ी तारा-राख से
जमकर बना।

\textbf{19.} भूवैज्ञानिक रूप से मरा हुआ: ठंडा भीतरी भाग, हवा और चट्टान को
फिर से चलाने वाला न कोई ज्वालामुखी न पट्ट-विवर्तनिकी, और मथता हुआ \omterm{def:g10:refraction:fiber}{क्रोड} भी
नहीं --- इसलिए सौर पवन के विरुद्ध कोई चुंबकीय ढाल भी नहीं।

\textbf{20.} हीलियम से आगे का आपका हर \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} किसी मरते तारे में गढ़ा
गया था, और उस भट्ठी के अस्थिर अवशेष आज भी उस ज़मीन को गरम रखे हैं जिस पर आप
खड़े हैं: तारा-धूल, जो अपनी ही \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} के सहारे जी रही है।
\end{solution}
