\begin{solution}{exo:g11:lenses-and-eye:2}
\omterm{def:g11:lenses-and-eye:lens}{अपसारी} ($f' < 0$)। नहीं: वस्तु की किसी भी दूरी $d$ के लिए
$\gamma = f'/(f' - d)$ (जो \cref{exo:g11:lenses-and-eye:15} में दिखाया गया है) $0$ और $1$ के बीच पड़ता है
--- यानी प्रतिबिंब हमेशा \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा और \emph{छोटा} होता है।
छपाई सीधी और सिकुड़ी हुई दिखती है, कभी बड़ी नहीं।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:3}

उलटा, असली आकार का --- जैसा खींचा गया है।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:5}

\omterm{def:g11:lenses-and-eye:focal}{फोकस दूरी} (यानी क्षमता) बदलता है; लेंस--दृष्टिपटल दूरी
बदल ही नहीं सकती। \omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु}: पूरे \omterm{def:g11:lenses-and-eye:eye}{समंजन} पर साफ़ दिखने

\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:6}

$1/\overline{OA'} = 1/50 - 1/3000 = 59/3000$, इसलिए $\overline{OA'} \approx \qty{50.8}{mm}$। सरकाव: लगभग \qty{0.8}{mm}।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:7}
$1/\overline{OA'} = 1/6.0 - 1/5.0 = -1/30$: $\overline{OA'} = \qty{-30}{cm}$ --- \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा, $\gamma = (-30)/(-5.0) = +6.0$: यानी डाक-टिकट से छह
गुना, और लेंस से \qty{30}{cm} ऊपर।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:8}
$1/f' = 1/40 + 1/40 = 1/20$: $f' = \qty{20}{cm}$; $\gamma = 40/(-40) = -1$। सममिति $\abs{\overline{OA'}} = \abs{\overline{OA}}$ को बाध्य करती है, इसलिए
$\abs{\gamma} = 1$; प्रतिबिंब \omterm{def:g11:lenses-and-eye:images}{वास्तविक} और उलटा है, इसलिए $\gamma = -1$: यानी ठीक
असली आकार का।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:9}
$1/\overline{OA'} = -1/25 - 1/50 = -3/50$: $\overline{OA'} \approx \qty{-16.7}{cm}$, $\gamma = (-16.7)/(-50) = +0.33$: \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा, और आकार का एक-तिहाई।
\omterm{def:g11:lenses-and-eye:defects}{निकट-दृष्टि दोष} वाले का चश्मा \omterm{def:g11:lenses-and-eye:lens}{अपसारी} होता है: उससे होकर दिखने
वाली हर चीज़ --- और बाहर से देखने पर पहनने वाले की आँखें भी --- छोटी हो
जाती है।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:10}
अनंत का प्रतिबिंब \omterm{def:g11:lenses-and-eye:eye}{दूर बिंदु} पर: $f' = \qty{-0.40}{m}$, इसलिए $C = -2.5$~D। पहाड़ का

\omterm{def:g11:lenses-and-eye:eye}{दूर बिंदु} पर, जिसे विश्राम में रखा नेत्र शून्य \omterm{def:g11:lenses-and-eye:eye}{समंजन} के

\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:12}
\emph{1.} $\gamma = \overline{OA'}/\overline{OA} = -50$ से $\overline{OA'} = -50\,\overline{OA}$ मिलता है; और $\overline{OA'} - \overline{OA} = \qty{5.1}{m}$ लेकर $-51\,\overline{OA} = \qty{5.1}{m}$: $\overline{OA} = \qty{-0.10}{m}$,
$\overline{OA'} = +\qty{5.0}{m}$।

\emph{2.} $1/f' = 1/5.0 + 1/0.10 = 10.2$: $f' \approx \qty{9.8}{cm}$।

\emph{3.} $50 \times \qty{24}{mm} = \qty{1.2}{m}$, उलटा --- और इसीलिए स्लाइडें उलटी लगाई जाती हैं।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:13}
\emph{1.} $C_\infty = 1/0.017 \approx 59$~D; $C_{25} = 1/0.017 + 1/0.25 \approx 63$~D। आयाम: $1/0.25 = 4.0$~D।

\emph{2.} आयाम $= 1/1.0 = 1.0$~D: यानी \omterm{def:g11:lenses-and-eye:eye}{मानक नेत्र} के $4.0$~D
का एक-चौथाई ($25\%$) ही बचा है।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:14}
\emph{1.} $f' = 0.25/3 \approx \qty{0.083}{m} = \qty{8.3}{cm}$।

\emph{2.} $\overline{OA'} = \qty{-25}{cm}$: $1/\overline{OA} = -1/25 - 3/25 = -4/25$, इसलिए $\overline{OA} = \qty{-6.25}{cm}$ और $\gamma = (-25)/(-6.25) = +4.0$ --- यानी विज्ञापित
$\times 3$ से बेहतर, जो प्रतिबिंब को अनंत पर मानता है; उसे
\omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु} तक खींच लाने से एक मात्रक और मिल जाता है।
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:15}
$1/\overline{OA'} = 1/f' - 1/d = (d - f')/(f'd)$, इसलिए $\overline{OA'} = f'd/(d - f')$ और $\gamma = \overline{OA'}/(-d) = f'/(f' - d)$। यदि $d > f'$: $\overline{OA'} > 0$
(\omterm{def:g11:lenses-and-eye:images}{वास्तविक}) और $\gamma < 0$ (उलटा)। यदि $0 < d < f'$: $\overline{OA'} < 0$
(\omterm{def:g11:lenses-and-eye:images}{आभासी}) और $\gamma = f'/(f' - d) > 1$: सीधा और बड़ा --- यानी \omterm{def:g11:lenses-and-eye:magnifier}{आवर्धक} वाली
व्यवस्था।
\end{solution}

\begin{solution}{pb:g11:lenses-and-eye:1}
\textbf{1.} $1/\overline{OA'} = 1/20 - 1/60 = 1/30$: $\overline{OA'} = +\qty{30}{cm}$, $\gamma = -0.50$ --- \omterm{def:g11:lenses-and-eye:images}{वास्तविक}, उलटा, आधे
आकार का।

\textbf{2.} $\overline{OA'} = +\qty{60}{cm}$, $\gamma = -2$: \qty{30}{cm} और \qty{60}{cm} ने भूमिकाएँ आपस में बदल
लीं, और \omterm{def:g11:lenses-and-eye:magnification}{आवर्धन} एक-दूसरे के व्युत्क्रम हैं --- उलटी दिशा में
चलता प्रकाश वही पथ लेता है।

\textbf{3.} $1/\overline{OA'} = 1/20 - 1/10 = -1/20$: $\overline{OA'} = \qty{-20}{cm}$, $\gamma = +2$ --- \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा, दुगुना।

\textbf{4.} अनंत से \omterm{def:g11:lenses-and-eye:focal}{प्रतिबिंब फोकस} तल में शुरू होता है; वस्तु के $F$
के पास आते-आते \omterm{def:g11:lenses-and-eye:images}{वास्तविक}, उलटा प्रतिबिंब अनंत की ओर हटता और
बढ़ता जाता है; और $f'$ के भीतर वह लेंस के आगे
\omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा और बड़ा हो जाता है।

\textbf{5.} $1/\overline{OA} \approx 0$ से $1/\overline{OA'} = 1/f'$ बचता है: $\overline{OA'} = f'$, यानी \omterm{def:g11:lenses-and-eye:focal}{फोकस} तल।

\textbf{6.} \omterm{def:g11:lenses-and-eye:focal}{फोकस} तल में, लेंस के \qty{50}{mm} पीछे।

\textbf{7.} $1/\overline{OA'} = 1/50 - 1/1000 = 19/1000$: $\overline{OA'} \approx \qty{52.6}{mm}$; सरकाव $\approx \qty{2.6}{mm}$।

\textbf{8.} $1/\overline{OA} = 1/55 - 1/50 = -1/550$: सबसे कम फोकस-दूरी \qty{550}{mm} = \qty{55}{cm}।

\textbf{9.} $\gamma = 55/(-550) = -0.10$: चेहरे का प्रतिबिंब \qty{20}{mm} बनता है, यानी \qty{24}{mm} के
संवेदक में बस-बस समा जाता है।

\textbf{10.} $\gamma = -1$ $\overline{OA'} = -\overline{OA}$ को बाध्य करता है, इसलिए $2/\overline{OA'} = 1/f'$:
वस्तु \qty{100}{mm} ($= 2f'$) आगे, संवेदक \qty{100}{mm} पीछे --- यानी दृश्य वाली
सेटिंग का दुगुना, जो \qty{55}{mm} वाली यंत्र-रचना से कहीं आगे है; मैक्रो लेंस
यही अतिरिक्त सरकाव देते हैं।

\textbf{11.} उनके \omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु} (\qty{1.0}{m}) से पास की हर चीज़
धुँधली पड़ती है: फैली हुई बाँहें पन्ने को \qty{1.0}{m} के जितना पास हो सके
उतना पास ले आती हैं --- साफ़, भले छोटा।

\textbf{12.} $C = 1/(-1.0) - 1/(-0.25) = -1.0 + 4.0 = +3.0$~D।

\textbf{13.} $\gamma = (-1.0)/(-0.25) = +4.0$: चार गुना बड़ा, पर चार गुना दूर भी --- यानी वही
कोणीय आकार; लाभ आकार का नहीं, \emph{साफ़ी} का है, क्योंकि प्रतिबिंब अब
उनके \omterm{def:g11:lenses-and-eye:eye}{निकट बिंदु} पर पड़ता है।

\textbf{14.} साफ़ दिखने के लिए प्रतिबिंब \qty{1.0}{m} से परे होना चाहिए:
यानी पन्ना \qty{25}{cm} पर (प्रतिबिंब \qty{1.0}{m} पर) से लेकर पन्ना $f' = 1/3.0 \approx \qty{0.33}{m}$ पर
(प्रतिबिंब अनंत पर) तक। किसी दूर की वस्तु का प्रतिबिंब चश्मे के \qty{33}{cm}
\emph{पीछे} बनता है --- यानी एक \omterm{def:g11:lenses-and-eye:images}{वास्तविक प्रतिबिंब} जिसे नेत्र
काम में नहीं ले सकता: दूर की दुनिया धुँधली पड़ जाती है।

\textbf{15.} इसीलिए द्विफोकसी लेंस: लेंस के निचले आधे हिस्से में
पढ़ने की क्षमता, और ऊपरी आधे में कोई नहीं (या दूर का सुधार)।

\textbf{16.} $1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$: $\overline{OA'} = \qty{-20}{cm}$, \omterm{def:g11:lenses-and-eye:images}{आभासी}, सीधा, $\gamma = +5.0$।

\textbf{17.} $1/\overline{OA} = -1/25 - 1/5.0 = -6/25$: $\overline{OA} \approx \qty{-4.2}{cm}$; $\gamma = (-25)/(-25/6) = +6.0$।

\textbf{18.} नगीना $F$ पर: प्रतिबिंब अनंत पर, और उसे शून्य
\omterm{def:g11:lenses-and-eye:eye}{समंजन} के साथ देखा जाता है --- यानी घंटों की जाँच बिना आँख थके।

\textbf{19.} $C_\infty = 1/0.017 \approx 59$~D; $C_{25} = 1/0.017 + 1/0.25 \approx 63$~D: यानी लगभग $4$~D का
\omterm{def:g11:lenses-and-eye:eye}{समंजन}।

\textbf{20.} $1/62.8 \approx \qty{15.9}{mm}$ के सामने $f' = 1/58.8 \approx \qty{17.0}{mm}$: लगभग \qty{1.1}{mm}, यानी क़रीब $6\%$
--- वही अंदरूनी ज़ूम, जिसकी नक़ल कैमरे ने \qty{2.6}{mm} के सरकाव से की और
दादी ने $+3$~D से।
\end{solution}
