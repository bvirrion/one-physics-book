\section*{अध्याय \ref{ch:g11:circuits-and-power} --- विद्युत परिपथ और शक्ति}

\begin{solution}{exo:g11:circuits-and-power:1}
$q = It = 2.0 \times 90 \times 60 = \qty{1.1e4}{C}$। मूल आवेशों की संख्या: $q/e = 1.08\times10^{4}/1.6\times10^{-19} \approx \num{6.8e22}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:2}
$I = U/R = 12/220 = \qty{0.055}{A} = \qty{55}{mA}$;
$R = U/I = 230/0.50 = \qty{460}{\ohm}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:3}
श्रेणीक्रम: $120 + 60 = \qty{180}{\ohm}$। समांतर क्रम: $\frac{120 \times 60}{180}
= \qty{40}{\ohm}$ --- दोनों से छोटा: \omterm{def:g11:circuits-and-power:voltage}{विभवांतर}
वही रहता है, पर धारा को दूसरा रास्ता मिल जाता है, इसलिए कुल धारा अधिक और
प्रतिरोध कम।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:4}
$P = UI = 230 \times 8.7 = \qty{2.0}{kW}$। $t = \qty{240}{s}$ में: $E = Pt = 2000 \times 240 = \qty{4.8e5}{J} = 4.8\times10^{5}/3.6
\times10^{6} \approx \qty{0.13}{kW.h}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:5}
$E = 60 \times 5.0 = \qty{300}{W.h} = \qty{0.30}{kW.h} = 0.30 \times
3.6\times10^{6} = \qty{1.1e6}{J}$। दाम: दिन के $0.30 \times 0.25 = 0.075$ यूरो।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:6}
\emph{1.} $R = U^2/P = 230^2/1200 = \qty{44}{\ohm}$; $I = P/U = 1200/230 = \qty{5.2}{A}$।

\emph{2.} $P' = U'^2/R = 115^2/44.1 = \qty{300}{W}$: चौथाई, आधी नहीं, क्योंकि $U$ आधा करने पर $I$ भी आधा
हो जाता है, और $P = UI \propto U^2$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:7}
$R_{\text{s}} = \qty{60}{\ohm}$, इसलिए $I = 12/60 = \qty{0.20}{A}$। $U_1 = R_1 I = \qty{8.0}{V}$, $U_2 = \qty{4.0}{V}$ (योग \qty{12}{V}); $P_1 = U_1 I = \qty{1.6}{W}$, $P_2 = \qty{0.8}{W}$: कुल
\qty{2.4}{W} $= UI = 12 \times 0.20$, जैसा होना ही चाहिए।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:8}
$I_1 = 12/60 = \qty{0.20}{A}$, $I_2 = 12/30 = \qty{0.40}{A}$, कुल $I = \qty{0.60}{A}$। तुल्य: $R = U/I = 12/0.60 = \qty{20}{\ohm}$, और सचमुच $\frac{60 \times 30}{60 + 30} = \qty{20}{\ohm}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:9}
$\mathcal{E} - 0.50\,r = 5.5$ और $\mathcal{E} - 2.0\,r = 4.0$; घटाने पर $1.5\,r = 1.5$, इसलिए $r = \qty{1.0}{\ohm}$ और $\mathcal{E} =
\qty{6.0}{V}$। लघुपथ में:
$\mathcal{E}/r = \qty{6.0}{A}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:10}
केतली $2200/230 = \qty{9.6}{A}$, टोस्टर $1000/230 = \qty{4.3}{A}$: कुल \qty{13.9}{A} $< \qty{16}{A}$, \omterm{def:g11:circuits-and-power:joule}{फ़्यूज़} टिक जाता है।
हीटर $1500/230 =
\qty{6.5}{A}$: कुल \qty{20.4}{A} $> \qty{16}{A}$, वह उड़ जाता है (या यों कहिए, $\qty{4.7}{kW} > 230 \times 16 \approx \qty{3.7}{kW}$)।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:11}
\qty{200}{V} पर: $I = 10^4/200 = \qty{50}{A}$, हानि $RI^2 = 2.0 \times
50^2 = \qty{5.0}{kW}$ --- पहुँचाई गई शक्ति की आधी। \qty{4.0}{kV} पर:
$I =
\qty{2.5}{A}$, हानि $2.0 \times 2.5^2 = \qty{12.5}{W}$। अनुपात $400 =
20^2$: बीस गुना \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} का अर्थ है बीस
गुना कम धारा, और हानि $I^2$ के साथ चलती है।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:12}
चार सजावटें: सभी श्रेणीक्रम में, $3 \times 60 = \qty{180}{\ohm}$; सभी समांतर क्रम में, $60/3 = \qty{20}{\ohm}$; एक
प्रतिरोधक समांतर क्रम में जुड़े दो के साथ श्रेणीक्रम में, $60 + 30 = \qty{90}{\ohm}$; एक
प्रतिरोधक श्रेणीक्रम में जुड़े दो के साथ समांतर क्रम में, $\frac{60 \times 120}{180} = \qty{40}{\ohm}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:13}
\emph{1.} $I = \mathcal{E}/(R + r) = 4.5/7.5 = \qty{0.60}{A}$; $U = RI = \qty{3.6}{V}$; काम का $P = UI = \qty{2.2}{W}$; गँवाया गया $rI^2 =
1.5 \times 0.60^2 = \qty{0.54}{W}$; $\eta = U/\mathcal{E} = 3.6/4.5 =
0.80$।

\emph{2.} संकेत से $P = \mathcal{E}^2 \big/ \big[(R - r)^2/R +
4r\big]$; कोष्ठक का मान ठीक तभी सबसे कम ($= 4r$) होता है जब
$R = r$, इसलिए $P_{\max} = \mathcal{E}^2/(4r) = 4.5^2/6.0 = \qty{3.4}{W}$।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:14}
ऊष्मा: $Q = 1.0 \times 4180 \times (100 - 20) = \qty{3.3e5}{J}$। विद्युत: $E = Q/0.90 = \qty{3.7e5}{J}$। समय: $t = E/P =
3.7\times10^{5}/2200 \approx \qty{170}{s}$, तीन मिनट से कम। दाम: $3.7\times10^{5}/3.6\times10^{6} \approx \qty{0.10}{kW.h}$,
लगभग $2.6$ सेंट।
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:15}
\emph{1.} $q = 60 \times 3600 = \qty{2.2e5}{C}$; $E \approx \mathcal{E}q
= 12.7 \times 2.16\times10^{5} = \qty{2.7e6}{J} \approx \qty{0.76}{kW.h}$।

\emph{2.} $U = 12.7 - 0.020 \times 150 = \qty{9.7}{V}$; दिया गया $UI = \qty{1.5}{kW}$; गँवाया गया $rI^2 = 0.020 \times 150^2 = \qty{450}{W}$।

\emph{3.} हेडलाइटें सेल के सिरों पर ही जुड़ी हैं: स्टार्ट करते समय सिरों का
\omterm{def:g11:circuits-and-power:voltage}{विभवांतर} $12.7$ से गिरकर $\qty{9.7}{V}$ रह जाता है, इसलिए उन्हें कम
शक्ति मिलती है और वे मंद पड़ जाती हैं।
\end{solution}

\begin{solution}{pb:g11:circuits-and-power:1}
\textbf{1.} हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $E_p = mgh = 5.0\times10^{4} \times 9.81 \times
200 = \qty{9.8e7}{J}$: उपलब्ध शक्ति \qty{98}{MW}।

\textbf{2.} $0.90 \times 98 = \qty{88}{MW}$।

\textbf{3.} $I = P/U = 8.8\times10^{7}/2.0\times10^{4} = \qty{4.4e3}{A}$।

\textbf{4.} $R_\ell I^2 = 32 \times (4.4\times10^{3})^2 \approx
\qty{6.2e8}{W} = \qty{620}{MW}$ --- संयंत्र के उत्पादन का सात गुना: \qty{20}{kV} पर संचरण
असंभव है।

\textbf{5.} $P = UI$ स्थिर रहते हुए $I$ केवल $U$ बढ़ाने से घटता है:
यानी ट्रांसफ़ॉर्मर से।

\textbf{6.} $I = 8.8\times10^{7}/4.0\times10^{5} = \qty{220}{A}$।

\textbf{7.} $R_\ell I^2 = 32 \times 220^2 \approx \qty{1.6}{MW}$।

\textbf{8.} $1.6/88 \approx 0.018$: लगभग $1.8\%$।

\textbf{9.} अंश $= R_\ell I^2/P = R_\ell (P/U)^2/P = R_\ell P/U^2
\propto 1/U^2$: \qty{20}{kV} से \qty{400}{kV} तक जाने पर वह $20^2 = 400$ से घटता है,
और सचमुच $\qty{620}{MW}/400 \approx \qty{1.6}{MW}$।

\textbf{10.} $R_\ell I = 32 \times 220 \approx \qty{7.1}{kV}$: \qty{400}{kV} का $1.8\%$, वही अंश जो प्रश्न 8 में था,
क्योंकि $R_\ell I/U = R_\ell I^2/UI$।

\textbf{11.} $\qty{1}{kW.h} = \qty{3.6e6}{J}$: \omterm{def:g10:orders-of-magnitude:unit}{मीटर} खींची गई शक्ति को उस समय से गुणा करता है
जितनी देर वह खींची गई, और सब जोड़ता जाता है।

\textbf{12.} $I = 900/230 = \qty{3.9}{A}$; $R = U^2/P = 230^2/900 \approx \qty{59}{\ohm}$।

\textbf{13.} $E = 900 \times 180 = \qty{1.6e5}{J} = \qty{0.045}{kW.h}$: लगभग $1.1$ सेंट।

\textbf{14.} $RI^2$ की भौतिकी बिलकुल वही है; फ़र्क़ केवल मक़सद का है ---
लाइन पर ऊष्मा अनचाही है, टोस्टर में ऊष्मा \emph{ही} उत्पाद है।

\textbf{15.} $3.9 + 9.6 = \qty{13.5}{A} < \qty{16}{A}$: टिक जाता है। हीटर $2000/230 = \qty{8.7}{A}$: कुल $\qty{22.2}{A} > \qty{16}{A}$, \omterm{def:g11:circuits-and-power:joule}{फ़्यूज़}
उड़ जाता है।

\textbf{16.} $0.05 \times 13.5^2 = \qty{9.1}{W}$: पहुँचाई गई शक्ति का $9.1/3100 \approx
0.3\%$ --- छोटे तारों पर
नगण्य।

\textbf{17.} $\eta = 0.90 \times 0.99 \times 0.982 \times 0.99 \times
0.997 \approx 0.86$।

\textbf{18.} एक सिंके हुए किलोवाट-घंटे के लिए ऊपर की ओर $3.6\times10^{6}/0.86
\approx \qty{4.2e6}{J}$
चाहिए: $m = E/(gh) = 4.2\times10^{6}/(9.81
\times 200) \approx \qty{2.1e3}{kg}$, यानी लगभग \qty{2.1}{m^3} पानी।

\textbf{19.} $R_\ell P/U^2 = 32 \times 8.8\times10^{7}/(2.0\times
10^{4})^2 \approx 7$। $1$ से बड़े ``अंश'' का अर्थ है कि लाइन उतने से भी
अधिक क्षय कर देगी जितना वह ले जाती है: उस \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} पर सुपुर्दगी हो
ही नहीं सकती।

\textbf{20.} एक सिंका हुआ \omterm{def:g10:energy-conservation:kwh}{किलोवाट-घंटा} पहाड़ को लगभग दो टन ---
\qty{2.1}{m^3} --- पानी का पड़ता है, जो \qty{200}{m} गिरता है, और रास्ते में
ऊर्जा का लगभग $14\%$ मर जाता है, ज़्यादातर टरबाइनों और
ट्रांसफ़ॉर्मरों में (\qty{1000}{km} लंबी लाइन में $2\%$ से भी कम)। इस सफ़र को
संभव बनाने वाली एकमात्र तरकीब ट्रांसफ़ॉर्मर है: \qty{400}{kV} तक चढ़ा देने से धारा
$20$ से बँट जाती है और जूल-हानि $400$ से।
\end{solution}
