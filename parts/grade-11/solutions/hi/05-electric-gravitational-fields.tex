\begin{solution}{exo:g11:electric-gravitational-fields:8}
\emph{1.} $E = \num{5.0e3} / \num{2.0e-2} = \qty{2.5e5}{V/m}$; \omterm{def:g10:relative-motion:frame}{विराम} में $qE = mg$, इसलिए $q = \num{2.0e-6} \times 9.81 / \num{2.5e5} \approx \qty{7.8e-11}{C}$।

\emph{2.} धनात्मक कण पर बल ऊपर की ओर होना चाहिए, इसलिए $\vect E$ ऊपर की
ओर है: \emph{नीचे} वाली पट्टी धनात्मक है।

\emph{3.} $n = \num{7.85e-11} / \num{1.60e-19} \approx \num{4.9e8}$ \omterm{def:g11:fundamental-interactions:elementary}{मूल आवेश}।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:9}
रेखाखंड के बाहर दोनों क्षेत्र एक ही ओर संकेत करते हैं; बीच में वे
एक-दूसरे के विरुद्ध हैं --- और संतुलन के लिए वह बिंदु कमज़ोर स्रोत $+Q$ के
पास चाहिए। $+Q$ तक की दूरी को $x$ मानने पर: $kQ/x^2 = 4kQ/(0.30 - x)^2$, इसलिए $0.30 - x = 2x$ और
$+Q$ से $x = \qty{0.10}{m}$।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:10}
$F_e = ke^2/d^2 = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$; $F_g = G m_p m_e / d^2 = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31} / (\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$। अनुपात $F_e / F_g \approx \num{2.3e39}$।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:11}
$E = \sqrt{300^2 + 400^2} = \qty{500}{N/C}$, पूरब से $\tan^{-1}(400/300) \approx \ang{53}$ उत्तर की ओर। $q < 0$ पर: $F = \num{2.0e-6} \times 500 = \qty{1.0e-3}{N}$, क्षेत्र के
विपरीत --- पश्चिम से $\ang{53}$ दक्षिण की ओर।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:12}
$GM/(R+h)^2 = \tfrac12\,GM/R^2$ से $R + h = R\sqrt{2}$ मिलता है, इसलिए $h = (\sqrt{2} - 1)R \approx \qty{2.6e6}{m}$ (लगभग \qty{2600}{km})। $h = R$ पर $d = 2R$:
सतह के मान का चौथाई।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:13}
\emph{1.} $AM = \sqrt{3.0^2 + 4.0^2} = \qty{5.0}{cm}$; हर आवेश $E_1 = \num{8.99e9} \times \num{2.0e-9} / 0.050^2
\approx \qty{7.2e3}{N/C}$ बनाता है।

\emph{2.} $AB$ के अनुदिश घटक सममिति से कट जाते हैं; समद्विभाजक के अनुदिश
हर क्षेत्र $E_1 \times 4/5$ का योगदान देता है, रेखाखंड से दूर की ओर:
$E = 2 \times \num{7.19e3} \times 0.80 \approx \qty{1.2e4}{N/C}$, समद्विभाजक के अनुदिश, $[AB]$ से दूर।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:14}
डिब्बे के मुक्त आवेश बाहरी क्षेत्र के नीचे खिसक जाते हैं: एक फलक पर
$+$ जमा होता है, दूसरे पर $-$, और ये खिसके हुए आवेश भीतर एक ऐसा
क्षेत्र बनाते हैं जो बाहरी क्षेत्र के विरुद्ध है। वे तब तक खिसकते
रहते हैं जब तक भीतर कुल क्षेत्र ठीक शून्य न हो जाए --- तभी
साम्यावस्था आती है। गुरुत्व के पास स्रोत का एक ही चिह्न है: हर द्रव्यमान एक
आकर्षी क्षेत्र \emph{जोड़ता} है, कोई उसका विरोध नहीं कर सकता।
गुरुत्व-ढाल के लिए ऋणात्मक द्रव्यमान चाहिए होगा, जो होता ही नहीं।
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:15}
\emph{1.} भार $m\vect g$ (नीचे), \omterm{def:g10:inertia:inventory}{तनाव} $\vect T$ (धागे के अनुदिश),
विद्युत बल $q\vect E$ (क्षैतिज)। \omterm{def:g10:relative-motion:frame}{विराम} में $T\sin\theta = qE$ और $T\cos\theta = mg$; भाग देने पर
$\tan\theta = qE/(mg)$।

\emph{2.} $\tan\theta = \num{2.0e-7} \times \num{1.0e4} /
(\num{5.0e-4} \times 9.81) \approx 0.41$, इसलिए $\theta \approx \ang{22}$; $T = mg/\cos\theta \approx \qty{5.3e-3}{N}$।
\end{solution}

\begin{solution}{pb:g11:electric-gravitational-fields:1}
\textbf{1.} \omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान}, पट्टियों के लंबवत, नीचे की ओर संकेत करता ($+$
ऊपरी पट्टी से $-$ निचली पट्टी की ओर); \omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान} होने का अर्थ है कि बूँद
जहाँ भी बहे, उस पर लगता बल वही रहेगा।
\textbf{2.} $E = U/d = 800 / \num{8.0e-3} = \qty{1.0e5}{V/m}$।
\textbf{3.} आवेश ऋणात्मक है, इसलिए बल $\vect E$ के \emph{विपरीत} है:
ऊपर की ओर। केवल $+$ पट्टी ऊपर होने पर ही $\vect E$ नीचे की ओर होता है और
विद्युत बल भार से लड़ पाता है।
\textbf{4.} $P = mg = \num{4.90e-15} \times 9.81 \approx
\qty{4.8e-14}{N}$।
\textbf{5.} $\abs{q}E = mg$।
\textbf{6.} $\abs{q} = mg/E = mgd/U$।
\textbf{7.} $\abs{q} = \num{4.81e-14} \times \num{8.0e-3} / 800
\approx \qty{4.8e-19}{C}$।
\textbf{8.} $n = \num{4.8e-19} / \num{1.60e-19} = 3$ अतिरिक्त इलेक्ट्रॉन।
\textbf{9.} $E$ बढ़ता है, इसलिए $\abs{q}E > mg$: कुल बल ऊपर की ओर है।
\textbf{10.} $\abs{q}\,U/d = mg$ से $U = mgd/\abs{q}$ --- यानी आवेश के व्युत्क्रमानुपाती।
\textbf{11.} $\abs{q} = mgd/U = \num{3.85e-16}/1200 \approx
\qty{3.2e-19}{C} = 2e$: $3e$ से नीचे, यानी बूँद ने एक इलेक्ट्रॉन \emph{खोया}
है।
\textbf{12.} $\abs{q} = \num{3.85e-16}/U$: \qty{3.2e-19}{C}, \qty{4.8e-19}{C}, \qty{6.4e-19}{C}, \qty{8.0e-19}{C}, \qty{9.6e-19}{C}।
\textbf{13.} वे $e = \qty{1.60e-19}{C}$ के साथ $2e, 3e, 4e, 5e, 6e$ हैं।
\textbf{14.} हर क़दम $\abs{q}$ को ठीक \qty{1.6e-19}{C} बदलता है: एक बार में एक ही
इलेक्ट्रॉन आ बैठता है।
\textbf{15.} उसके लिए $\abs{q} = \num{3.85e-16}/700 \approx
\qty{5.5e-19}{C} = 3.4\,e$ चाहिए होगा --- और यह इलेक्ट्रॉनों की पूरी
संख्या नहीं है, इसलिए विकिरण का कोई झोंका उसे कभी नहीं बना सकता।
\textbf{16.} आवेश \emph{क्वांटित} है: वह \omterm{def:g11:fundamental-interactions:elementary}{मूल आवेश} $e = \qty{1.60e-19}{C}$ के
पूर्णांक गुणजों में आता है --- और यही वह राशि है जो मिलिकन ने नापी।
\textbf{17.} $q \leftrightarrow m$, $E \leftrightarrow g$ और $\abs{q}E \leftrightarrow mg$: यानी स्वयं भार।
\textbf{18.} $g_{\text{गोला}} = GM/d^2 = \num{6.67e-11} \times 1000 /
1.0^2 = \qty{6.7e-8}{N/kg}$; बूँद पर बल $\num{4.90e-15} \times \num{6.7e-8} \approx \qty{3.3e-22}{N}$ --- उसके भार से
लगभग $\num{1.5e8}$ गुना छोटा। बात बनने वाली नहीं।
\textbf{19.} क्योंकि $G$ इतना छोटा है कि पृथ्वी के खिंचाव की बराबरी ऊपर
रखा कोई ग्रह जितना बड़ा द्रव्यमान ही कर सकता है --- और कोई द्रव्यमान
प्रतिकर्षित नहीं करता। विशाल $k$ और आवेश के दो चिह्न मिलकर मेज़ पर रखी
एक पट्टी को ग्रह से दोनों दिशाओं में जीत जाने देते हैं।
\textbf{20.} एक इलेक्ट्रॉन: $\Delta\abs{q}/\abs{q} = e/3e \approx 33\%$; तेल का एक \omterm{def:g11:fundamental-interactions:ladder}{अणु}: $\Delta m/m = \num{5e-25} / \num{4.90e-15}
\approx \num{e-10}$। आवेश
की सीढ़ी संतुलन के \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} को सैकड़ों \omterm{def:g11:circuits-and-power:voltage}{वोल्ट} खिसका देती है;
द्रव्यमान की सीढ़ी किसी भी \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} पर दिखती ही नहीं। इसीलिए यह
तराज़ू बिजली के \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} --- $e = \qty{1.60e-19}{C}$ --- को अलग-अलग पहचान लेता है,
जबकि तेल के \omterm{def:g11:fundamental-interactions:ladder}{अणु} धुँधलाकर एक सतत धारा बन जाते हैं।
\end{solution}
