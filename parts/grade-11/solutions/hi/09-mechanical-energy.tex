\end{solution}

\begin{solution}{exo:g11:mechanical-energy:5}

\begin{solution}{exo:g11:mechanical-energy:6}

$400$ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} ऊपर उठा सकती थी।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:7}
चाल दोनों पर वही, $v = \sqrt{2 \times 9.81 \times 3.0} \approx \qty{7.7}{m/s}$ (हिसाब में केवल गिरावट आती है)। समय में खड़ी
फिसलपट्टी जीतती है: वह तेज़ चाल जल्दी पा लेती है और उसका पथ छोटा भी है।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:8}
$h = v^2/2g = 144/19.62 \approx \qty{7.3}{m}$। \qty{4.0}{m} पर: $v = \sqrt{144 - 2 \times 9.81 \times 4.0} \approx \qty{8.1}{m/s}$ --- और दोनों ओर वही: $E_m$ ऊँचाई पर निर्भर करता
है, चलने की दिशा पर नहीं।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:9}
$v = \sqrt{23^2 + 2 \times 9.81 \times 35} = \sqrt{1216} \approx
\qty{35}{m/s}$। असल में इससे कम: वायु-प्रतिरोध $E_m$ का कुछ हिस्सा \omterm{def:g11:mechanical-energy:dissipation}{क्षय} कर देता है।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:10}

\end{solution}

\begin{solution}{exo:g11:mechanical-energy:11}

वृत्त), ऊर्जा नहीं: ऊँचाइयाँ और चालें अछूती रहती हैं।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:12}
$v = \sqrt{2 \times 9.81 \times 2.5} = \qty{7.0}{m/s}$। चढ़ाई की ऊँचाइयाँ $E_m$ के साथ चलती हैं, इसलिए हर फेरे पर $\times 0.9$:
$2.25$, $2.03$, $1.82$, $1.64$, $1.48$, $1.33$, $1.20$, $1.08$, $0.97$ ---
यानी $9$वें फेरे पर वह पहली बार \qty{1.0}{m} से नीचे रहता है।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:13}
कोण चाहे जो हो, आरंभिक $E_m = \tfrac12 m \times 15^2 + m \times 9.81 \times 45$ वही रहता है, इसलिए ज़मीन पर चाल $v = \sqrt{225 + 2 \times 9.81 \times 45} = \sqrt{1108} \approx
\qty{33}{m/s}$ है।
कोण उतरते वेग की \emph{दिशा}, उड़ान का समय और \omterm{def:g11:fundamental-interactions:strong}{परास} तय करता है ---
उतरने की चाल नहीं।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:14}
$\tfrac12 mv^2 = mgh - f\ell = 80 \times 9.81 \times 120 - 65 \times
800 = \num{94176} - \num{52000} = \qty{4.22e4}{J}$, इसलिए $v = \sqrt{2 \times \num{42176}/80} \approx \qty{32}{m/s}$ (\qty{117}{km/h})। \omterm{def:g11:mechanical-energy:dissipation}{क्षय} हुआ अंश: $52000/94176 \approx 55\%$।
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:15}
$m = \qty{2.0e9}{kg}$: $E_p = \num{2.0e9} \times 9.81 \times 300 \approx \qty{5.9e12}{J}
= \num{5.9e12}/\num{3.6e6} \approx \qty{1.6e6}{kW.h}$। वापस पाया गया: $0.85 \times \num{1.64e6} \approx \qty{1.4e6}{kW.h}$ --- यानी एक दिन के लिए लगभग $\num{1.4e5}$
घर।
\end{solution}

\begin{solution}{pb:g11:mechanical-energy:1}
\textbf{1.} $v_{\text{शिखर}} = \sqrt{gR} = \sqrt{9.81 \times 8.0}
\approx \qty{8.9}{m/s}$।

\textbf{2.} $E_m = mg\,(2R) + \tfrac12 m gR = \num{3.14e5} +
\num{7.8e4} \approx \qty{3.92e5}{J}$।

\textbf{3.} $mgH_{\min} = E_m$ से $H_{\min} = \qty{20}{m}$ मिलता है; बीजगणित में $H_{\min} = 2R + \frac{gR}{2g} = \frac{5R}{2}$।

\textbf{4.} $v_{\text{शिखर}} = \sqrt{2g(H - 2R)} =
\sqrt{2 \times 9.81 \times 8.0} \approx \qty{12.5}{m/s}$; $v_{\text{शिखर}}^2/(gR) = 2.0$ --- यानी न्यूनतम का दुगुना।

\textbf{5.} हर ऊर्जा $m$ के समानुपाती है, इसलिए $m$ कट जाता है:
गाड़ी भरी हो या ख़ाली, चालें वही और सुरक्षा वही।

\textbf{6.} $v = \sqrt{2 \times 9.81 \times 24} \approx \qty{21.7}{m/s}
\approx \qty{78}{km/h}$।

\textbf{7.} $v = \sqrt{2 \times 9.81 \times (24 - 8.0)} \approx
\qty{17.7}{m/s}$।

\textbf{8.} $mgH = \qty{4.71e5}{J}$। $(E_p, E_k)$: छोड़-बिंदु $(\num{4.71e5}, 0)$; तल $(0, \num{4.71e5})$; बग़ल $(\num{1.57e5}, \num{3.14e5})$; शिखर
$(\num{3.14e5}, \num{1.57e5})$ --- और हर जोड़ी का योग \qty{4.71e5}{J}।

\textbf{9.} $v = \sqrt{2 \times 9.81 \times (24 - 12)} \approx
\qty{15.3}{m/s}$।

\textbf{10.} $v(z) = \sqrt{2g(H - z)}$: सबसे तेज़ पटरी के सबसे निचले बिंदु पर, सबसे धीमी फंदे
के शिखर पर।

\textbf{11.} ऊँचाई एक ही है, इसलिए $\Delta E_p = 0$ और $\Delta E_m = \Delta E_k = \tfrac12 \times 2000 \times (19.2^2 -
20.4^2) \approx \qty{-4.75e4}{J}$: इस जगह-चुनाव से घाटा
शुद्ध गतिज बन जाता है, जिसे दो चालमापियों से सीधे पढ़ा जा सकता है।

\textbf{12.} $W(\vect f) = \qty{-4.75e4}{J}$; $f = \num{4.75e4}/60 \approx \qty{7.9e2}{N}$।

\textbf{13.} \omterm{def:g10:energy-conservation:forms}{तापीय ऊर्जा}, पहियों में, पटरियों में और हवा में।

\textbf{14.} पहले संवेदक पर $E_m$: $\num{4.16e5} + \num{3.9e4} =
\qty{4.55e5}{J}$; प्रति \qty{60}{m} हानि $\num{4.75e4}/\num{4.55e5} \approx 10\%$।
$H_{\min}$ पर फंदे का शिखर \emph{ठीक} $\sqrt{gR}$ पर पहुँचता है; कोई भी हानि उसे
सुरक्षित चाल से नीचे गिरा देती है।

\textbf{15.} प्रति \qty{60}{m} $\approx 10\%$ गँवाते हुए $mgH$ का लगभग $15\%$
($\approx \qty{7.1e4}{J}$) चला जाता है: $E_m \approx \qty{4.00e5}{J}$, जो ज़रूरी \qty{3.92e5}{J} से अब भी ऊपर है --- सुरक्षित,
पर केवल $2\%$ के हाशिये से। वह हाशिया कोई विलासिता नहीं था।

\textbf{16.} $E_k = \tfrac12 \times 2000 \times 18.0^2 =
\qty{3.24e5}{J}$।

\textbf{17.} $F = \num{3.24e5}/45 = \qty{7.2e3}{N}$।

\textbf{18.} $a = 7200/2000 = \qty{3.6}{m/s^2} \approx 0.37\,g$: कड़ा, पर आरामदेह।

\textbf{19.} पटरी का \omterm{def:g10:inertia:inventory}{घर्षण}: $mgH - \num{3.24e5} = \num{470880} -
\num{324000} = \qty{1.47e5}{J}$। लेखा: $\num{1.47e5} + \num{3.24e5} =
\num{4.71e5} = mgH$ --- हर
\omterm{def:g11:work-of-force:work}{जूल} का हिसाब मिल जाता है।

\textbf{20.} छोड़-ऊँचाई $H = \qty{24}{m}$ ने ही फंदे की सुरक्षा, हर चाल $\sqrt{2g(H - z)}$,
\omterm{def:g10:inertia:inventory}{घर्षण} का बिल और ब्रेक-बल, सब तय कर दिए; और \omterm{def:g10:inertia:inventory}{घर्षण}
को अनदेखा करने का दाम वही हाशिया है --- पहाड़ी का वह $\qty{4}{m}$ जो रगड़ती दुनिया
चुपचाप खा जाती है।
\end{solution}
