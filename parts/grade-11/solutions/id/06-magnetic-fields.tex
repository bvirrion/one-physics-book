\section*{Bab \ref{ch:g11:magnetic-fields} --- Kemagnetan dan Medan Magnet}

\begin{solution}{exo:g11:magnetic-fields:1}
Tiap potongnya menjadi \omterm{def:g11:magnetic-fields:magnet}{magnet} yang lengkap dengan sebuah kutub utara
dan sebuah kutub selatan: kutub baru muncul pada tiap muka potongnya.
Kedua muka yang baru terpotong itu adalah U dan S, sehingga keduanya
tarik-menarik --- potongannya berusaha merakit dirinya kembali. Tidak
ada cara yang mengasingkan satu kutub: setiap potongan menciptakan
sepasang.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:2}
(a) Salah: \omterm{def:g11:electric-gravitational-fields:field}{medannya} memiliki satu arah pada tiap titik. (b) Benar.
(c) Benar: tiap garisnya menutup melalui tubuh \omterm{def:g11:magnetic-fields:magnet}{magnetnya}. (d) Salah:
jarumnya berhenti \emph{sejajar} garisnya, menyinggung garis itu.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:3}
$B = \num{2e-7} \times 10 / 0.020 = \qty{1.0e-4}{T}$ --- dua kali \omterm{def:g11:electric-gravitational-fields:field}{medan}
Bumi.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:4}
$n = 800/0.40 = \qty{2000}{m^{-1}}$;
$B = \num{1.26e-6} \times 2000 \times 1.5 \approx \qty{3.8}{mT}$.
Karena $B \propto I$: menggandakan arusnya memberikan \qty{7.5}{mT}.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:5}
$F = ILB = 5.0 \times 0.10 \times 0.20 = \qty{0.10}{N}$ --- setara
\omterm{def:g10:universal-gravitation:weight}{berat} $m = F/g = 0.10/9.81 \approx \qty{10}{g}$.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:6}
\emph{1.} Jalurnya menunjuk $\ang{8}$ ke barat dari utara sejati:
$2000 \times \sin\ang{8} \approx \qty{2.8e2}{m}$ ke barat.

\emph{2.} Kemudikanlah $\ang{8}$ ke \emph{timur} dari utara jarumnya.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:7}
$d = \mu_0 I / (2\pi B) = \num{2e-7} \times 20 / \num{5.0e-5}
= \qty{8.0}{cm}$. Di dalam satu desimeter dari kabel semacam itu,
kompasnya membaca kawatnya, bukan planetnya --- jauhkanlah kompas dari
penghantar yang bertegangan.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:8}
$n = B/(\mu_0 I) = 0.010 / (\num{1.26e-6} \times 4.0)
\approx \qty{2.0e3}{m^{-1}}$; sepanjang \qty{25}{cm},
$N = 1989 \times 0.25 \approx 500$ lilitan.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:9}
$F = 10 \times 0.12 \times 0.50 = \qty{0.60}{N}$;
$a = F/m = 0.60/0.020 = \qty{30}{m/s^2}$ --- sekitar $3g$: batang itu
memang benar-benar meloncat.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:10}
Keseimbangannya: $ILB = mg$, sehingga
$I = \num{5.0e-3} \times 9.81 / (0.20 \times 0.10) \approx
\qty{2.5}{A}$. \omterm{def:g10:inertia:force}{Gayanya} harus menunjuk ke atas; dengan $\vect B$ yang
mendatar dan tegak lurus kawatnya, telapak tangan kanan yang terbuka
(telapaknya menghadap ke atas) menetapkan arah arus yang diperlukan.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:11}
Telapak tangan kanan yang terbuka: (a) ke arah atas halaman; (b) ke
kanan; (c) tidak ada \omterm{def:g10:inertia:force}{gaya} --- kawatnya sejajar \omterm{def:g11:electric-gravitational-fields:field}{medannya}.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:12}
\emph{1.} $B = \num{2e-7} \times 5.0 / 0.030 \approx
\qty{33}{\micro T}$; garisnya melingkari kawatnya, sehingga tepat di
bawah kawat itu \omterm{def:g11:electric-gravitational-fields:field}{medannya} mendatar dan tegak lurus kawatnya:
timur--barat.

\emph{2.} Jarumnya mengikuti jumlah vektornya: $\tan\theta = 33/20 =
1.67$, sehingga $\theta \approx \ang{59}$ dari utara.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:13}
\emph{1.} $I = U/R = 12/2.0 = \qty{6.0}{A}$;
$n = 400/0.20 = \qty{2000}{m^{-1}}$;
$B_0 = \num{1.26e-6} \times 2000 \times 6.0 \approx \qty{15}{mT}$;
beserta intinya, $100 B_0 \approx \qty{1.5}{T}$.

\emph{2.} $P = UI = 12 \times 6.0 = \qty{72}{W}$.

\emph{3.} Tanpa arus, tidak ada \omterm{def:g11:electric-gravitational-fields:field}{medan} --- dan besi lunak tidak tetap
termagnetkan: bebannya terlepas seketika, dan memang begitulah
dirancang.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:14}
\emph{1.} $F = NILB = 50 \times 2.0 \times 0.050 \times 0.30 =
\qty{1.5}{N}$ pada tiap sisinya. Arusnya berjalan ke arah yang
berlawanan pada kedua sisinya, sehingga \omterm{def:g10:inertia:force}{gayanya} berlawanan: sepasang
kopel yang memutar untainya.

\emph{2.} Ketika sejajar $\vect B$, sebuah kawat tidak merasakan \omterm{prop:g11:magnetic-fields:laplace}{gaya
Laplace}.

\emph{3.} Setelah setengah putaran kopelnya akan berbalik lalu
membatalkan putarannya; \omterm{rem:g11:magnetic-fields:motor}{komutatornya} membalikkan arusnya tiap setengah
putaran sehingga untainya terus berputar ke satu arah.
\end{solution}

\begin{solution}{exo:g11:magnetic-fields:15}
\emph{1.} $d = \num{2e-7} \times 100 / 1 = \qty{2.0e-5}{m} =
\qty{20}{\micro m}$ --- jauh \emph{di dalam} tembaga setebal milimeter
itu. Tidak ada titik terjangkau di dekat kawat tunggal yang mencapai
\qty{1}{T}: \omterm{def:g11:electric-gravitational-fields:field}{medan} yang kuat menumpuk ribuan lilitan (sebuah
\omterm{def:g11:magnetic-fields:solenoid}{solenoida}), ditambah besi atau bahan adihantar.

\emph{2.} $\num{e8} / \num{5e-5} = \num{2e12}$: sekitar dua belas
pangkat sepuluh.
\end{solution}

\begin{solution}{pb:g11:magnetic-fields:1}
\textbf{1.} Jarumnya sendiri adalah \omterm{def:g11:magnetic-fields:magnet}{magnet} kecil; \omterm{def:g11:electric-gravitational-fields:field}{medan} Bumi
membariskannya, dengan ujung utaranya sejajar komponen mendatar
$\vect B$, yaitu ke arah utara \omterm{def:g11:magnetic-fields:magnet}{magnet}.

\textbf{2.} $B = \sqrt{20^2 + 44^2} = \sqrt{2336} \approx
\qty{48}{\micro T}$, miring $\tan^{-1}(44/20) \approx \ang{66}$ di
bawah arah mendatar.

\textbf{3.} Jalurnya berjalan $\ang{6}$ ke barat dari utara sejati:
$30 \times \sin\ang{6} \approx \qty{3.1}{km}$ ke barat.

\textbf{4.} Galatnya berjumlah: $\ang{10}$, sehingga
$30 \times \sin\ang{10} \approx
\qty{5.2}{km}$ --- lima kilometer untuk
sepuluh derajat.

\textbf{5.} Komponen mendatarnya: di dekat kutubnya \omterm{def:g11:electric-gravitational-fields:field}{medan} itu hampir
tegak, bagian mendatarnya menyusut menuju nol, dan jarumnya tidak lagi
memiliki apa pun untuk dijadikan acuan berbaris.

\textbf{6.} $n = 600/0.30 = \qty{2000}{m^{-1}}$.

\textbf{7.} $I = U/R = 12/1.5 = \qty{8.0}{A}$.

\textbf{8.} $B_0 = \mu_0 n I = \num{1.26e-6} \times 2000 \times 8.0
\approx \qty{20}{mT}$.

\textbf{9.} $50 \times \qty{20}{mT} = \qty{1.0}{T}$ --- dua kali \omterm{def:g11:electric-gravitational-fields:field}{medan}
pada kutub neodimium, pada muka yang jauh lebih luas.

\textbf{10.} $P = UI = 12 \times 8.0 = \qty{96}{W}$; dalam
\qty{10}{\minute}: $E = 96 \times 600 \approx \qty{58}{kJ}$.

\textbf{11.} Karena \omterm{def:g11:magnetic-fields:electromagnet}{elektromagnet} dapat melepaskan: bukalah sakelarnya,
\omterm{def:g11:electric-gravitational-fields:field}{medannya} mati, dan mobilnya jatuh tepat di tempat yang diinginkan.
\omterm{def:g11:magnetic-fields:magnet}{Magnet} tetap mencengkeram selamanya.

\textbf{12.} $F = NILB = 100 \times 3.0 \times 0.040 \times 0.25 =
\qty{3.0}{N}$.

\textbf{13.} Sepasang kopel: karena bekerja pada sisi yang berseberangan
terhadap sumbunya, kedua \omterm{def:g10:inertia:force}{gaya} yang berlawanan itu sama-sama memutar
untainya ke arah yang sama. \omterm{def:g10:inertia:force}{Gaya} yang sama besar dan \emph{sejajar}
hanya akan mendorong untainya ke samping, bukan memutarnya.

\textbf{14.} Tidak ada: kawat yang sejajar \omterm{def:g11:electric-gravitational-fields:field}{medannya} tidak merasakan
\omterm{prop:g11:magnetic-fields:laplace}{gaya Laplace}.

\textbf{15.} \omterm{rem:g11:magnetic-fields:motor}{Komutator} membalikkan arus pada untainya tiap setengah
putaran, tepat ketika kopelnya akan berganti tanda --- sehingga momen
\omterm{def:g10:inertia:force}{gayanya} selalu menggerakkan putaran yang sama.

\textbf{16.} Gandakan $N$, atau gandakan $I$, atau gandakan $B$:
pasangan \omterm{def:g10:inertia:force}{gaya} $F = NILB$ sebanding dengan masing-masingnya.

\textbf{17.} $1.5 / \num{5.0e-5} = \num{3.0e4}$: tiga puluh ribu kali
\omterm{def:g11:electric-gravitational-fields:field}{medan} Bumi.

\textbf{18.} $n = B/(\mu_0 I) = 1.5 / (\num{1.26e-6} \times 500)
\approx \qty{2.4e3}{m^{-1}}$ --- dan tiap lilitannya membawa
\qty{500}{A}.

\textbf{19.} $P = RI^2 = 0.20 \times 500^2 = \qty{50}{kW}$ --- setara
pemanasan satu permukiman. Gulungan yang sebenarnya bersifat adihantar:
\omterm{def:g11:circuits-and-power:resistance}{hambatannya} nol, pembuangan \omterm{def:g10:energy-conservation:power}{dayanya} nol, asalkan gulungan itu dijaga
beberapa derajat di atas nol mutlak.

\textbf{20.} Kompas $\num{5e-5}$, derek $1$, MRI $1.5$: kapalnya berada
lima anak tangga di bawah kedua mesin itu, yang berbagi satu dekade ---
di antara ruang antarbintang $\num{e-10}$ dan bintang neutron
$\num{e8}$. Tidak ada yang berubah selain \omterm{def:g11:circuits-and-power:current}{amperenya} (beserta lilitan,
besi dan bahan adihantar yang melipatgandakannya): satu \omterm{def:g11:electric-gravitational-fields:field}{medan}, tiga
pekerjaan.
\end{solution}
