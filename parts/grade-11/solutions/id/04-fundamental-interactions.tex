\section*{Bab \ref{ch:g11:fundamental-interactions} --- Interaksi Fundamental}

\begin{solution}{exo:g11:fundamental-interactions:1}
Proton: $10^{-15}$ (lantai \omterm{def:g11:fundamental-interactions:ladder}{nukleon}); \omterm{def:g11:fundamental-interactions:ladder}{molekul} air: $10^{-10}$ (lantai
\omterm{def:g11:fundamental-interactions:ladder}{molekul} --- \omterm{def:g11:fundamental-interactions:ladder}{molekul} tunggal berukuran seperti \omterm{def:g11:fundamental-interactions:ladder}{atom}); pasir: $10^{-4}$
(materi sehari-hari); Bumi: $10^{7}$ (planet); Bima Sakti: $10^{21}$
(galaksi). Semuanya dalam \omterm{def:g10:orders-of-magnitude:unit}{meter}.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:2}

$N = \num{4.8e-9}/\num{1.6e-19} = \num{3.0e10}$ elektron. Rambutnya
membawa $+\qty{4.8e-9}{C}$, menurut kekekalan muatan
(\cref{prop:g11:fundamental-interactions:conservation}).
\end{solution}

\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:6}
$F_E = \num{8.99e9} \times (\num{1.60e-19})^2/(\num{2.0e-15})^2
= \num{2.30e-28}/\num{4.0e-30} \approx \qty{58}{N}$;
$F_G = \num{6.67e-11} \times (\num{1.67e-27})^2/\num{4.0e-30}
\approx \qty{4.7e-35}{N}$. Nisbahnya
$F_E/F_G \approx \num{1.2e36}$. Interaksi kuatlah yang menahan \omterm{def:g11:fundamental-interactions:ladder}{inti
atomnya}; \omterm{def:g11:fundamental-interactions:four}{gravitasinya} tidak berarti.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:7}
$F_E = \num{2.30e-28}/(\num{5.3e-11})^2 = \num{2.30e-28}/\num{2.81e-21}
\approx \qty{8.2e-8}{N}$;
$F_G = \num{6.67e-11} \times \num{1.67e-27} \times \num{9.11e-31} /
\num{2.81e-21} \approx \qty{3.6e-47}{N}$. Nisbahnya
$\approx \num{2.3e39}$: interaksi elektromagnetiklah yang menahan \omterm{def:g11:fundamental-interactions:ladder}{atom}
tetap utuh.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:8}
Melalui sentuhan, muatannya dibagi:
$q_A = q_B = \qty{+4.0e-9}{C}$ (seluruhnya tetap
$\qty{8.0e-9}{C}$). Lalu
$F = \num{8.99e9} \times (\num{4.0e-9})^2/0.10^2 =
\qty{1.4e-5}{N}$, tolak-menolak (tandanya sejenis).
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:9}
$d = \sqrt{k q^2/F} = \sqrt{\num{8.99e9} \times \num{1.0e-12}/0.90}
\approx \qty{0.10}{m}$. \omterm{def:g10:inertia:force}{Gaya} yang $9$ kali lebih kecil menuntut jarak
$3$ kali lebih besar: $d = \qty{0.30}{m}$.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:10}
$F = \num{2.30e-28}/(\num{1.0e-10})^2 = \qty{2.3e-8}{N}$. \omterm{def:g11:fundamental-interactions:strong}{Interaksi
kuatnya} tidak menyumbang apa pun: $10^{-10}$~m adalah $10^{5}$ kali
\omterm{def:g11:fundamental-interactions:strong}{jangkauannya}. \omterm{def:g11:fundamental-interactions:ladder}{Molekul} terikat oleh \omterm{def:g11:fundamental-interactions:four}{interaksi elektromagnetik}.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:11}
$q = \num{e23}/\num{e6} \times \num{1.6e-19} = \qty{1.6e-2}{C}$ pada
tiap keping. $F = \num{8.99e9} \times (\num{1.6e-2})^2/1.0^2 \approx
\qty{2.3e6}{N}$ --- setara \omterm{def:g10:universal-gravitation:weight}{berat} $\num{2.3e6}/9.81 \approx
\qty{2.3e5}{kg}$, yaitu sebuah kereta barang bermuatan, hanya dari
sepersejuta elektronnya. Materi sehari-hari tentu (dan memang) netral
jauh lebih baik daripada satu bagian dari sejuta.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:12}
$F = \num{2.30e-28}/(\num{1.4e-14})^2 = \num{2.30e-28}/\num{1.96e-28}
\approx \qty{1.2}{N}$ --- tetap raksasa pada skala ini. Tidak: keduanya
terpisah $14$ kali \omterm{def:g11:fundamental-interactions:strong}{jangkauan} \omterm{def:g10:inertia:force}{gaya} kuatnya; masing-masing hanya
terekatkan pada tetangga terdekatnya. Pasangannya:
$\frac{92 \times 91}{2} = 4186$, dan \emph{semuanya} tolak-menolak pada
jarak berapa pun, sementara perekatnya tidak tumbuh bersama ukurannya.
Neutron menambahkan tarikan tanpa tolakan lalu mengencerkan protonnya;
tetapi tolakannya tumbuh bersama kuadrat banyaknya proton sedangkan
perekatnya hanya tumbuh bersama banyaknya tetangga, sehingga di luar
sekitar $80$ proton tidak ada anggaran neutron yang menyeimbangkan
pembukuannya --- \omterm{def:g11:fundamental-interactions:ladder}{inti atom} semacam itu meluruh
(\cref{ch:g11:nucleus-radioactivity}).
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:13}
Kesetimbangannya: pada arah tegak $T\cos\theta = mg$, pada arah
mendatar $T\sin\theta = F$, sehingga $F = mg\tan\theta$. Di sini
$\sin\theta = 3.0/50 = 0.060$, dan untuk sudut yang kecil
$\tan\theta \approx \sin\theta = 0.060$. Lalu
$F = \num{1.0e-3} \times 9.81 \times 0.060 \approx \qty{5.9e-4}{N}$;
$q = d\sqrt{F/k} = 0.060 \times \sqrt{\num{5.9e-4}/\num{8.99e9}}
\approx \qty{1.54e-8}{C}$; $N = \num{1.54e-8}/\num{1.6e-19} \approx
\num{9.6e10}$ \omterm{def:g11:fundamental-interactions:elementary}{muatan dasar}.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:14}
$F = \num{6.67e-11} \times \num{5.97e24} \times \num{7.35e22} /
(\num{3.84e8})^2 = \num{2.93e37}/\num{1.47e17} \approx \qty{2.0e20}{N}$.
Dengan mengambil $kQ^2/d^2 = F$:
$Q = d\sqrt{F/k} = \num{3.84e8} \times \sqrt{\num{2.0e20}/\num{8.99e9}}
\approx \qty{5.7e13}{C}$. Nilai itu setara
$N = \num{5.7e13}/\num{1.6e-19} \approx \num{3.6e32}$ elektron, yang
bermassa $\num{3.6e32} \times \num{9.11e-31} \approx \qty{3.2e2}{kg}$:
beberapa ratus \omterm{def:g10:orders-of-magnitude:unit}{kilogram} elektron sanggup mengerjakan pekerjaan
$\qty{7.35e22}{kg}$ Bulan --- kekuatan listrik dan kelemahan \omterm{def:g11:fundamental-interactions:four}{gravitasi}
di dalam satu bilangan.
\end{solution}

\begin{solution}{exo:g11:fundamental-interactions:15}
$F = \num{2.30e-28}/(\num{2.8e-10})^2 = \num{2.30e-28}/\num{7.84e-20}
\approx \qty{2.9e-9}{N}$; \omterm{def:g10:universal-gravitation:weight}{beratnya}
$P = \num{5.9e-26} \times 9.81 \approx \qty{5.8e-25}{N}$; nisbahnya
$F/P \approx \num{5e15}$. Tiap ikatannya berkekuatan \emph{tetap},
tetapi \omterm{def:g10:universal-gravitation:weight}{beratnya} tumbuh bersama seluruh massa di atasnya: tumpuklah
batuan dan beban pada lapisan terbawahnya tumbuh tanpa batas sementara
ikatannya tidak. Di sekitar \qty{e4}{m} batuan, bebannya meremukkan
ikatan listriknya --- \omterm{def:g11:fundamental-interactions:four}{gravitasi} menang bukan karena kekuatannya tiap
partikel melainkan karena ia \emph{menjumlah} ketika muatannya saling
menghapus.
\end{solution}

\begin{solution}{pb:g11:fundamental-interactions:1}
\textbf{1.} \omterm{def:g11:fundamental-interactions:ladder}{Kuark} $< \qty{e-18}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{nukleon} $\qty{e-15}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{inti atom}
$10^{-15}$--$\qty{e-14}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{atom} $\qty{e-10}{m}$; \omterm{def:g11:fundamental-interactions:ladder}{molekul}
$\qty{e-9}{m}$; benda sehari-hari $\qty{1}{m}$; planet $\qty{e7}{m}$;
sistem keplanetan $\qty{e11}{m}$; galaksi $\qty{e21}{m}$.

\textbf{2.} Faktor skalanya $10^{5}$: \omterm{def:g11:fundamental-interactions:ladder}{atomnya} selebar
$\qty{1}{cm} \times 10^{5} = \qty{1}{km}$.

\textbf{3.} $(10^{-15}/10^{-10})^3 = 10^{-15}$: materi kosong sebesar
$\num{99.9999999999999}\,\%$.

\textbf{4.} $\num{7e-5}/\num{e-10} = \num{7e5}$ \omterm{def:g11:fundamental-interactions:ladder}{atom} --- sekitar sejuta
\omterm{def:g11:fundamental-interactions:ladder}{atom} melintangi sehelai rambut.

\textbf{5.} Dari $10^{-18}$ sampai $10^{21}$: $39$ pangkat sepuluh.

\textbf{6.} $F_E = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$.

\textbf{7.} $F_G = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31}/(\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$; nisbahnya
$F_E/F_G \approx \num{2.3e39}$. Listriklah yang menahan \omterm{def:g11:fundamental-interactions:ladder}{atomnya};
\omterm{def:g11:fundamental-interactions:four}{gravitasi} hanya penonton.

\textbf{8.} Tanpa \omterm{def:g11:fundamental-interactions:four}{gravitasi}: tidak ada apa pun yang berubah secara
terukur (\omterm{def:g11:fundamental-interactions:four}{gravitasinya} $10^{39}$ di bawah). Tanpa listrik: tidak ada
\omterm{def:g11:fundamental-interactions:ladder}{atom}, tidak ada \omterm{def:g11:fundamental-interactions:ladder}{molekul}, tidak ada materi sama sekali.

\textbf{9.} ``Sentuhan'' adalah tolakan listrik antara awan elektron
\omterm{def:g11:fundamental-interactions:ladder}{atom} yang didekatkan sampai sekitar $\qty{e-10}{m}$: tidak pernah ada
apa pun yang benar-benar bersentuhan. \omterm{def:g10:inertia:inventory}{Gaya normal} lantai dan \omterm{def:g10:inertia:inventory}{gesekannya}
adalah hukum Coulomb dalam jumlah besar.

\textbf{10.} $q = \num{e23} \times \num{e-9} \times \num{1.6e-19} =
\qty{1.6e-5}{C}$ pada masing-masing;
$F = \num{8.99e9} \times (\num{1.6e-5})^2 \approx \qty{2.3}{N}$ ---
\omterm{def:g10:inertia:force}{gaya} yang jelas kasatmata dari satu elektron tiap semiliar. Materi
sehari-hari netral jauh lebih baik daripada $10^{-9}$.

\textbf{11.} \omterm{def:g11:fundamental-interactions:four}{Interaksi elektromagnetik}. \omterm{def:g10:inertia:force}{Gaya} kuatnya sudah di luar
\omterm{def:g11:fundamental-interactions:strong}{jangkauan} melampaui sekitar $\qty{e-14}{m}$ --- $10^{4}$ kali lebih
kecil daripada satu \omterm{def:g11:fundamental-interactions:ladder}{atom}.

\textbf{12.} $F = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{1.0e-15})^2 \approx \qty{2.3e2}{N}$.

\textbf{13.} $a = F/m_p = 230/\num{1.67e-27} \approx
\qty{1.4e29}{m/s^2}$, sekitar $\num{1.4e28}$ kali $g$: tidak ada \omterm{def:g10:inertia:force}{gaya}
sehari-hari yang mendekatinya.

\textbf{14.} Lebih kuat daripada \qty{230}{N} pada $\qty{e-15}{m}$
(\omterm{def:g11:fundamental-interactions:ladder}{inti atom} ada); bekerja pada proton dan neutron sama saja, buta
terhadap muatan (neutronnya ikut tercengkeram); \omterm{def:g11:fundamental-interactions:strong}{jangkauannya} sekitar
$\qty{e-15}{m}$ (\omterm{def:g11:fundamental-interactions:ladder}{inti atom} tetap mungil --- perekatnya tidak sanggup
menjangkau lebih jauh).

\textbf{15.} $F = \num{2.30e-28}/(\num{1.4e-14})^2 \approx
\qty{1.2}{N}$. Tidak: jaraknya $14$ kali \omterm{def:g11:fundamental-interactions:strong}{jangkauannya}. Tolakannya
bekerja pada \emph{seluruh} $\frac{92\times91}{2} = 4186$ pasangan
protonnya; perekatnya hanya mengikat tetangga terdekat, sehingga
pertumbuhannya berpihak kepada tolakan. Neutron menambahkan perekat dan
jarak tanpa menambahkan tolakan --- sampai, di luar sekitar $80$ proton,
tidak ada campuran yang menyeimbangkan dan \omterm{def:g11:fundamental-interactions:ladder}{inti atomnya} pun meluruh.

\textbf{16.} \omterm{def:g11:fundamental-interactions:weak}{Interaksi lemah} mengubah sebuah neutron menjadi proton
(dan sebaliknya), sehingga menimbulkan peluruhan $\beta$ --- interaksi
itu mengubah; ia tidak pernah mengikat.

\textbf{17.} $F = \num{6.67e-11} \times \num{5.97e24} \times
\num{7.35e22}/(\num{3.84e8})^2 \approx \qty{2.0e20}{N}$.

\textbf{18.} $Q_E = \num{1.8e51} \times \num{1.6e-19} =
\qty{2.9e32}{C}$, $Q_M = \qty{3.5e30}{C}$. Menyamakan \omterm{def:g10:inertia:force}{gayanya} berarti
$f^2\,k\,Q_E Q_M = G M_E M_M$:
\[
f = \sqrt{\frac{\num{2.93e37}}{\num{8.99e9} \times \num{2.9e32}
\times \num{3.5e30}}} = \sqrt{\num{3.2e-36}} \approx \num{1.8e-18} :
\]
ketakseimbangan sebesar \emph{dua bagian dari $10^{18}$} sudah akan
menghapus \omterm{def:g11:fundamental-interactions:four}{gravitasinya}.

\textbf{19.} Muatan hadir dalam dua tanda dan saling menghapus
(pertanyaan 18 menunjukkan betapa sempurnanya), sehingga \omterm{def:g10:inertia:force}{gaya} yang
lebih kuat itu mematikan dirinya sendiri pada skala yang besar; massa
hanya bertanda satu, menjumlah selamanya, dan tidak ada yang
menudunginya.

\textbf{20.} \omterm{def:g11:fundamental-interactions:ladder}{Inti atom}: \omterm{def:g11:fundamental-interactions:strong}{interaksi kuat} mengalahkan tolakan protonnya
pada $\qty{e-15}{m}$. \omterm{def:g11:fundamental-interactions:ladder}{Atom} sampai gunung: \omterm{def:g11:fundamental-interactions:four}{interaksi elektromagnetik}
mengikat elektron pada \omterm{def:g11:fundamental-interactions:ladder}{inti atomnya} dan \omterm{def:g11:fundamental-interactions:ladder}{atom} pada \omterm{def:g11:fundamental-interactions:ladder}{atom} yang lain.
Planet dan seterusnya: \omterm{def:g11:fundamental-interactions:four}{gravitasi}, yang tanpa lawan setelah muatannya
saling menghapus, menahan planet, \omterm{def:g10:universal-gravitation:satellite}{orbit} dan galaksi. Dunia tidak runtuh
karena tiap lantainya ditopang perekat yang lebih kaku daripada
bebannya, dan tidak berhamburan karena materi netral sampai sekitar
$10^{-18}$ --- sehingga tiap skala memiliki tepat satu \omterm{def:g10:inertia:force}{gaya} yang
bertanggung jawab, dan \omterm{def:g10:inertia:force}{gaya} itu bersifat tarik-menarik di tempat ia
memang harus demikian.
\end{solution}
