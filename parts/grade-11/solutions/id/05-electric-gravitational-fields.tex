\section*{Bab \ref{ch:g11:electric-gravitational-fields} --- Medan Listrik dan Medan Gravitasi}

\begin{solution}{exo:g11:electric-gravitational-fields:1}
$E = \num{5.0e-3} / \num{2.0e-6} = \qty{2.5e3}{N/C}$, menunjuk ke utara.
Pada $q'$: $F = \num{4.0e-6} \times \num{2.5e3} = \qty{1.0e-2}{N}$,
menunjuk ke \emph{selatan} ($q' < 0$).
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:2}
\emph{1.} $E = \num{8.99e9} \times \num{5.0e-9} / 0.30^2
\approx \qty{5.0e2}{N/C}$, menunjuk menjauhi $O$ ($Q > 0$).

\emph{2.} Menyeparuhkan $k\abs{Q}/d^2$ menuntut $d^2$ digandakan:
$d' = 0.30\sqrt{2} \approx \qty{0.42}{m}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:3}
$E = 600 / \num{3.0e-3} = \qty{2.0e5}{V/m}$;
$F = eE = \num{1.60e-19} \times \num{2.0e5} = \qty{3.2e-14}{N}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:4}
$g_{\text{Mars}} = P/m = 930/250 = \qty{3.7}{N/kg}$. Antariksawannya:
$P = 80 \times 3.72 \approx \qty{3.0e2}{N}$ (sekitar $\qty{780}{N}$ di
Bumi).
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:5}
(a) Salah: \omterm{def:g11:electric-gravitational-fields:field}{medannya} memiliki satu arah pada tiap titik, sehingga \omterm{def:g11:electric-gravitational-fields:lines}{garis
medannya} tidak pernah berpotongan. (b) Salah: \omterm{def:g11:electric-gravitational-fields:lines}{garis medannya}
meninggalkan $+$ dan tiba pada $-$. (c) Benar: berdesakannya garis
menyandikan kekuatannya.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:6}
\emph{1.} $d = \num{6.37e6} + \num{4.2e5} = \qty{6.79e6}{m}$, sehingga
$g = \num{6.67e-11} \times \num{5.97e24} / (\num{6.79e6})^2
\approx \qty{8.6}{N/kg}$ --- $88\%$ dari nilainya di permukaan.

\emph{2.} Mereka melayang karena stasiunnya dan antariksawannya jatuh
bersama-sama (jatuh bebas), bukan karena gravitasinya tidak ada.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:7}
$E = m_e g / e = \num{9.11e-31} \times 9.81 / \num{1.60e-19}
\approx \qty{5.6e-11}{V/m}$: sekitar $\num{4e14}$ kali lebih lemah
daripada \omterm{def:g11:electric-gravitational-fields:field}{medan} \omterm{def:g10:signals-and-waves:oscilloscope}{osiloskopnya}. Gravitasi benar-benar dapat diabaikan di
dalam fisika elektron.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:8}
\emph{1.} $E = \num{5.0e3} / \num{2.0e-2} = \qty{2.5e5}{V/m}$; ketika
diam $qE = mg$, sehingga
$q = \num{2.0e-6} \times 9.81 / \num{2.5e5} \approx \qty{7.8e-11}{C}$.

\emph{2.} \omterm{def:g10:inertia:force}{Gaya} pada butir debu yang positif itu harus menunjuk ke atas,
sehingga $\vect E$ menunjuk ke atas: keping \emph{bawah}nya yang
positif.

\emph{3.} $n = \num{7.85e-11} / \num{1.60e-19} \approx \num{4.9e8}$
\omterm{def:g11:fundamental-interactions:elementary}{muatan dasar}.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:9}
Di luar ruas garisnya kedua \omterm{def:g11:electric-gravitational-fields:field}{medan} menunjuk ke arah yang sama; di
antaranya keduanya berlawanan --- dan keseimbangannya menuntut titik
itu lebih dekat ke sumber yang lebih lemah, yaitu $+Q$. Dengan $x$
adalah jaraknya ke $+Q$: $kQ/x^2 = 4kQ/(0.30 - x)^2$, sehingga
$0.30 - x = 2x$ dan $x = \qty{0.10}{m}$ dari $+Q$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:10}
$F_e = ke^2/d^2 = \num{8.99e9} \times (\num{1.60e-19})^2 /
(\num{5.3e-11})^2 \approx \qty{8.2e-8}{N}$;
$F_g = G m_p m_e / d^2 = \num{6.67e-11} \times \num{1.67e-27} \times
\num{9.11e-31} / (\num{5.3e-11})^2 \approx \qty{3.6e-47}{N}$. Nisbahnya
$F_e / F_g \approx \num{2.3e39}$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:11}
$E = \sqrt{300^2 + 400^2} = \qty{500}{N/C}$, pada
$\tan^{-1}(400/300) \approx \ang{53}$ dari timur ke arah utara. Pada
$q < 0$: $F = \num{2.0e-6} \times 500 = \qty{1.0e-3}{N}$, berlawanan
dengan \omterm{def:g11:electric-gravitational-fields:field}{medannya} --- $\ang{53}$ dari barat ke arah selatan.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:12}
$GM/(R+h)^2 = \tfrac12\,GM/R^2$ memberikan $R + h = R\sqrt{2}$,
sehingga $h = (\sqrt{2} - 1)R \approx \qty{2.6e6}{m}$ (sekitar
\qty{2600}{km}). Pada $h = R$, $d = 2R$: seperempat nilainya di
permukaan.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:13}
\emph{1.} $AM = \sqrt{3.0^2 + 4.0^2} = \qty{5.0}{cm}$; tiap muatannya
menciptakan $E_1 = \num{8.99e9} \times \num{2.0e-9} / 0.050^2
\approx \qty{7.2e3}{N/C}$.

\emph{2.} Komponen sepanjang $AB$ saling menghapus karena setangkup;
tiap \omterm{def:g11:electric-gravitational-fields:field}{medannya} menyumbang $E_1 \times 4/5$ sepanjang sumbu tegak
lurusnya, menjauhi ruas garisnya:
$E = 2 \times \num{7.19e3} \times 0.80 \approx \qty{1.2e4}{N/C}$,
terarah sepanjang sumbu tegak lurusnya, menjauhi $[AB]$.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:14}
Muatan bebas pada kotak itu bergerak di bawah \omterm{def:g11:electric-gravitational-fields:field}{medan} luarnya: $+$
menumpuk pada satu sisinya, $-$ pada sisi yang lain, dan muatan yang
tergeser itu menciptakan \omterm{def:g11:electric-gravitational-fields:field}{medan} di dalam yang melawan \omterm{def:g11:electric-gravitational-fields:field}{medan} luarnya.
Muatan itu terus bergerak sampai \omterm{def:g11:electric-gravitational-fields:field}{medan} seluruhnya di dalamnya tepat
bernilai nol --- lalu setimbang. Gravitasi hanya menawarkan satu tanda
sumber: setiap massa \emph{menambahkan} \omterm{def:g11:electric-gravitational-fields:field}{medan} yang menarik, dan tidak
satu pun dapat melawannya. Tudung gravitasi akan memerlukan massa
negatif, yang tidak ada.
\end{solution}

\begin{solution}{exo:g11:electric-gravitational-fields:15}
\emph{1.} \omterm{def:g10:universal-gravitation:weight}{Beratnya} $m\vect g$ (ke bawah), \omterm{def:g11:circuits-and-power:voltage}{tegangan} benangnya $\vect T$
(sepanjang benangnya), \omterm{def:g10:inertia:force}{gaya} listriknya $q\vect E$ (mendatar). Ketika
diam, $T\sin\theta = qE$ dan $T\cos\theta = mg$; bagilah keduanya:
$\tan\theta = qE/(mg)$.

\emph{2.} $\tan\theta = \num{2.0e-7} \times \num{1.0e4} /
(\num{5.0e-4} \times 9.81) \approx 0.41$, sehingga
$\theta \approx \ang{22}$; $T = mg/\cos\theta \approx \qty{5.3e-3}{N}$.
\end{solution}

\begin{solution}{pb:g11:electric-gravitational-fields:1}
\textbf{1.} \omterm{def:g11:electric-gravitational-fields:capacitor}{Seragam}, tegak lurus kepingnya, menunjuk ke bawah (dari
keping atas yang $+$ ke keping bawah yang $-$); keseragamannya berarti
\omterm{def:g10:inertia:force}{gaya} pada tetes itu sama ke mana pun tetes itu bergeser.
\textbf{2.} $E = U/d = 800 / \num{8.0e-3} = \qty{1.0e5}{V/m}$.
\textbf{3.} Muatannya negatif, sehingga \omterm{def:g10:inertia:force}{gayanya} \emph{berlawanan}
dengan $\vect E$: ke atas. Hanya dengan keping $+$ di atas, $\vect E$
menunjuk ke bawah dan \omterm{def:g10:inertia:force}{gaya} listriknya melawan \omterm{def:g10:universal-gravitation:weight}{beratnya}.
\textbf{4.} $P = mg = \num{4.90e-15} \times 9.81 \approx
\qty{4.8e-14}{N}$.
\textbf{5.} $\abs{q}E = mg$.
\textbf{6.} $\abs{q} = mg/E = mgd/U$.
\textbf{7.} $\abs{q} = \num{4.81e-14} \times \num{8.0e-3} / 800
\approx \qty{4.8e-19}{C}$.
\textbf{8.} $n = \num{4.8e-19} / \num{1.60e-19} = 3$ kelebihan
elektron.
\textbf{9.} $E$ bertambah, sehingga $\abs{q}E > mg$: \omterm{def:g10:inertia:force}{gaya} totalnya
menunjuk ke atas.
\textbf{10.} Dari $\abs{q}\,U/d = mg$ diperoleh $U = mgd/\abs{q}$ ---
berbanding terbalik dengan muatannya.
\textbf{11.} $\abs{q} = mgd/U = \num{3.85e-16}/1200 \approx
\qty{3.2e-19}{C} = 2e$: turun dari $3e$, sehingga tetes itu
\emph{kehilangan} satu elektron.
\textbf{12.} $\abs{q} = \num{3.85e-16}/U$: \qty{3.2e-19}{C},
\qty{4.8e-19}{C}, \qty{6.4e-19}{C}, \qty{8.0e-19}{C},
\qty{9.6e-19}{C}.
\textbf{13.} Kelimanya adalah $2e, 3e, 4e, 5e, 6e$ dengan
$e = \qty{1.60e-19}{C}$.
\textbf{14.} Tiap langkahnya mengubah $\abs{q}$ tepat sebesar
\qty{1.6e-19}{C}: satu elektron mendarat setiap kali.
\textbf{15.} Hal itu akan menuntut $\abs{q} = \num{3.85e-16}/700 \approx
\qty{5.5e-19}{C} = 3.4\,e$ --- bukan bilangan bulat elektron,
sehingga tidak ada semburan radiasi yang pernah dapat menghasilkannya.
\textbf{16.} Muatan bersifat \emph{terkuantisasi}: muatan hadir sebagai
kelipatan bulat \omterm{def:g11:fundamental-interactions:elementary}{muatan dasar} $e = \qty{1.60e-19}{C}$ --- itulah besaran
yang diukur Millikan.
\textbf{17.} $q \leftrightarrow m$, $E \leftrightarrow g$, dan
$\abs{q}E \leftrightarrow mg$: yaitu \omterm{def:g10:universal-gravitation:weight}{beratnya} sendiri.
\textbf{18.} $g_{\text{bola}} = GM/d^2 = \num{6.67e-11} \times 1000 /
1.0^2 = \qty{6.7e-8}{N/kg}$; \omterm{def:g10:inertia:force}{gayanya} pada tetes itu
$\num{4.90e-15} \times \num{6.7e-8} \approx \qty{3.3e-22}{N}$ ---
sekitar $\num{1.5e8}$ kali lebih kecil daripada \omterm{def:g10:universal-gravitation:weight}{beratnya}. Mustahil.
\textbf{19.} Karena $G$ begitu kecil, hanya massa seukuran planet di
atasnya yang dapat menandingi tarikan Bumi --- dan tidak ada massa yang
menolak. Nilai $k$ yang raksasa dan kedua tanda muatan memungkinkan
sebuah keping di atas meja mengalahkan tarikan planetnya, ke arah mana
pun.
\textbf{20.} Satu elektron: $\Delta\abs{q}/\abs{q} = e/3e \approx 33\%$;
satu \omterm{def:g11:fundamental-interactions:ladder}{molekul} minyak: $\Delta m/m = \num{5e-25} / \num{4.90e-15}
\approx \num{e-10}$. Tangga muatannya menggeser \omterm{def:g11:circuits-and-power:voltage}{tegangan} pengimbangnya
sebesar ratusan \omterm{def:g11:circuits-and-power:voltage}{volt}; tangga massanya tidak terlihat pada \omterm{def:g11:circuits-and-power:voltage}{tegangan}
berapa pun. Karena itu keseimbangan ini memisahkan \omterm{def:g11:fundamental-interactions:ladder}{atom} kelistrikan ---
$e = \qty{1.60e-19}{C}$ --- sementara \omterm{def:g11:fundamental-interactions:ladder}{molekul} minyaknya melebur menjadi
sesuatu yang sinambung.
\end{solution}
