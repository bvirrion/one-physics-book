\begin{solution}{exo:g11:lenses-and-eye:2}
\omterm{def:g11:lenses-and-eye:lens}{Cekung} ($f' < 0$). Tidak: untuk jarak benda $d$ berapa pun,
$\gamma = f'/(f' - d)$ (ditunjukkan pada
\cref{exo:g11:lenses-and-eye:15}) terletak antara $0$ dan $1$ ---
bayangannya selalu \omterm{def:g11:lenses-and-eye:images}{maya}, tegak, dan \emph{diperkecil}. Cetakannya
tampak tegak dan menyusut, tidak pernah membesar.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:3}
Bendanya berada pada $2f'$; sinar pusat dan sinar sejajarnya
berpotongan \qty{20}{cm} di belakang \omterm{def:g11:lenses-and-eye:lens}{lensanya}. Pemeriksaannya:
$1/\overline{OA'} = 1/10 - 1/20 = 1/20$, sehingga
$\overline{OA'} = +\qty{20}{cm}$ dan $\gamma = 20/(-20) = -1$: \omterm{def:g11:lenses-and-eye:images}{nyata},
terbalik, seukuran aslinya --- persis seperti yang dilukis.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:4}
$1/\overline{OA'} = 1/8.0 - 1/24 = 1/12$:
$\overline{OA'} = +\qty{12}{cm}$, $\gamma = 12/(-24) = -0.50$. \omterm{def:g11:lenses-and-eye:images}{Bayangan
nyata} \qty{12}{cm} di belakang \omterm{def:g11:lenses-and-eye:lens}{lensanya}, terbalik, setengah ukurannya.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:5}
\omterm{def:g11:lenses-and-eye:eye}{Kornea} dan \omterm{def:g11:lenses-and-eye:eye}{lensa mata} membentuk \omterm{def:g11:lenses-and-eye:lens}{lensa cembungnya}; \omterm{def:g11:lenses-and-eye:eye}{retina} adalah
layarnya. \omterm{def:g11:lenses-and-eye:eye}{Akomodasi} mengubah \omterm{def:g11:lenses-and-eye:focal}{jarak fokus} (kekuatan) \omterm{def:g11:lenses-and-eye:eye}{lensa matanya};
jarak lensa--retina tidak dapat berubah. \omterm{def:g11:lenses-and-eye:eye}{Titik dekat}: titik terdekat
yang terlihat tajam pada \omterm{def:g11:lenses-and-eye:eye}{akomodasi} penuh --- \qty{25}{cm} untuk \omterm{def:g11:lenses-and-eye:eye}{mata
normal}.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:6}
Bentang alam: pengindranya berada pada bidang fokus, \qty{50}{mm} dari
\omterm{def:g11:lenses-and-eye:lens}{lensanya}. Pada \qty{3.0}{m}:
$1/\overline{OA'} = 1/50 - 1/3000 = 59/3000$, sehingga
$\overline{OA'} \approx \qty{50.8}{mm}$. Pergeserannya sekitar
\qty{0.8}{mm}.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:7}
$1/\overline{OA'} = 1/6.0 - 1/5.0 = -1/30$:
$\overline{OA'} = \qty{-30}{cm}$ --- \omterm{def:g11:lenses-and-eye:images}{maya}, tegak, dengan
$\gamma = (-30)/(-5.0) = +6.0$: enam kali perangkonya, \qty{30}{cm} di
atas \omterm{def:g11:lenses-and-eye:lens}{lensanya}.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:8}
$1/f' = 1/40 + 1/40 = 1/20$: $f' = \qty{20}{cm}$;
$\gamma = 40/(-40) = -1$. Kesetangkupannya memaksa
$\abs{\overline{OA'}} = \abs{\overline{OA}}$, sehingga
$\abs{\gamma} = 1$; bayangannya \omterm{def:g11:lenses-and-eye:images}{nyata} dan terbalik, sehingga
$\gamma = -1$: tepat seukuran aslinya.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:9}
$1/\overline{OA'} = -1/25 - 1/50 = -3/50$:
$\overline{OA'} \approx \qty{-16.7}{cm}$,
$\gamma = (-16.7)/(-50) = +0.33$: \omterm{def:g11:lenses-and-eye:images}{maya}, tegak, sepertiga ukurannya.
Kacamata seorang penderita \omterm{def:g11:lenses-and-eye:defects}{miopi} bersifat \omterm{def:g11:lenses-and-eye:lens}{cekung}: segala sesuatu yang
dipandang melaluinya --- termasuk, dari luar, mata si pemakainya ---
tampak diperkecil.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:10}
Bayangan benda di tak terhingga jatuh di \omterm{def:g11:lenses-and-eye:eye}{titik jauhnya}:
$f' = \qty{-0.40}{m}$, sehingga $C = -2.5$~D. Bayangan gunungnya berada
\qty{40}{cm} di depan \omterm{def:g11:lenses-and-eye:lens}{lensanya} --- persis di \omterm{def:g11:lenses-and-eye:eye}{titik jauhnya}, yang
dilihat tajam oleh mata yang beristirahat tanpa \omterm{def:g11:lenses-and-eye:eye}{akomodasi} sama sekali.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:11}
$C = 1/\overline{OA'} - 1/\overline{OA} = 1/(-0.80) - 1/(-0.25)

\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:12}
\emph{1.} $\gamma = \overline{OA'}/\overline{OA} = -50$ memberikan
$\overline{OA'} = -50\,\overline{OA}$; dengan
$\overline{OA'} - \overline{OA} = \qty{5.1}{m}$,
$-51\,\overline{OA} = \qty{5.1}{m}$: $\overline{OA} = \qty{-0.10}{m}$,
$\overline{OA'} = +\qty{5.0}{m}$.

\emph{2.} $1/f' = 1/5.0 + 1/0.10 = 10.2$: $f' \approx \qty{9.8}{cm}$.

\emph{3.} $50 \times \qty{24}{mm} = \qty{1.2}{m}$, terbalik --- dan
itulah sebabnya salindia dipasang terbalik.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:13}
\emph{1.} $C_\infty = 1/0.017 \approx 59$~D;
$C_{25} = 1/0.017 + 1/0.25 \approx 63$~D. Lebar \omterm{def:g11:lenses-and-eye:eye}{akomodasinya}:
$1/0.25 = 4.0$~D.

\emph{2.} Lebar \omterm{def:g11:lenses-and-eye:eye}{akomodasinya} $= 1/1.0 = 1.0$~D: seperempat ($25\%$)
dari $4.0$~D milik \omterm{def:g11:lenses-and-eye:eye}{mata normal} yang tersisa.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:14}
\emph{1.} $f' = 0.25/3 \approx \qty{0.083}{m} = \qty{8.3}{cm}$.

\emph{2.} $\overline{OA'} = \qty{-25}{cm}$:
$1/\overline{OA} = -1/25 - 3/25 = -4/25$, sehingga
$\overline{OA} = \qty{-6.25}{cm}$ dan
$\gamma = (-25)/(-6.25) = +4.0$ --- lebih baik daripada $\times 3$ yang
diiklankan, yang menganggap bayangannya di tak terhingga; menariknya ke
\omterm{def:g11:lenses-and-eye:eye}{titik dekat} menambah satu \omterm{def:g10:orders-of-magnitude:unit}{satuan}.
\end{solution}

\begin{solution}{exo:g11:lenses-and-eye:15}
$1/\overline{OA'} = 1/f' - 1/d = (d - f')/(f'd)$, sehingga
$\overline{OA'} = f'd/(d - f')$ dan
$\gamma = \overline{OA'}/(-d) = f'/(f' - d)$. Jika $d > f'$:
$\overline{OA'} > 0$ (\omterm{def:g11:lenses-and-eye:images}{nyata}) dan $\gamma < 0$ (terbalik). Jika
$0 < d < f'$: $\overline{OA'} < 0$ (\omterm{def:g11:lenses-and-eye:images}{maya}) dan
$\gamma = f'/(f' - d) > 1$: tegak dan diperbesar --- itulah susunan
sebuah \omterm{def:g11:lenses-and-eye:magnifier}{lup}.
\end{solution}

\begin{solution}{pb:g11:lenses-and-eye:1}
\textbf{1.} $1/\overline{OA'} = 1/20 - 1/60 = 1/30$:
$\overline{OA'} = +\qty{30}{cm}$, $\gamma = -0.50$ --- \omterm{def:g11:lenses-and-eye:images}{nyata}, terbalik,
setengah ukurannya.

\textbf{2.} $\overline{OA'} = +\qty{60}{cm}$, $\gamma = -2$: kedudukan
\qty{30}{cm} dan \qty{60}{cm} bertukar peran, dan \omterm{def:g11:lenses-and-eye:magnification}{perbesarannya} saling
berkebalikan --- cahaya yang dibalik menempuh \omterm{def:g10:relative-motion:trajectory}{lintasan} yang sama.

\textbf{3.} $1/\overline{OA'} = 1/20 - 1/10 = -1/20$:
$\overline{OA'} = \qty{-20}{cm}$, $\gamma = +2$ --- \omterm{def:g11:lenses-and-eye:images}{maya}, tegak,
menjadi dua kali lipat.

\textbf{4.} Dari tak terhingga bayangannya bermula pada bidang
fokusnya; ketika bendanya mendekati $F$, \omterm{def:g11:lenses-and-eye:images}{bayangan nyata} yang terbalik
itu menjauh ke tak terhingga sambil membesar; di dalam $f'$ bayangannya
berubah menjadi \omterm{def:g11:lenses-and-eye:images}{maya}, tegak, diperbesar, dan berada di depan \omterm{def:g11:lenses-and-eye:lens}{lensanya}.

\textbf{5.} $1/\overline{OA} \approx 0$ menyisakan
$1/\overline{OA'} = 1/f'$: $\overline{OA'} = f'$, yaitu bidang
fokusnya.

\textbf{6.} Pada bidang fokusnya, \qty{50}{mm} di belakang \omterm{def:g11:lenses-and-eye:lens}{lensanya}.

\textbf{7.} $1/\overline{OA'} = 1/50 - 1/1000 = 19/1000$:
$\overline{OA'} \approx \qty{52.6}{mm}$; pergeserannya
$\approx \qty{2.6}{mm}$.

\textbf{8.} $1/\overline{OA} = 1/55 - 1/50 = -1/550$: jarak pemfokusan
terkecilnya \qty{550}{mm} = \qty{55}{cm}.

\textbf{9.} $\gamma = 55/(-550) = -0.10$: wajahnya membentuk bayangan
setinggi \qty{20}{mm}, tepat muat di dalam pengindra \qty{24}{mm} itu.

\textbf{10.} $\gamma = -1$ memaksa $\overline{OA'} = -\overline{OA}$,
sehingga $2/\overline{OA'} = 1/f'$: bendanya \qty{100}{mm} ($= 2f'$) di
depan, pengindranya \qty{100}{mm} di belakang --- dua kali jarak
setelan bentang alamnya, jauh melampaui mekanisme \qty{55}{mm} tadi;
\omterm{def:g11:lenses-and-eye:lens}{lensa} makro menyediakan pergeseran tambahannya.

\textbf{11.} Apa pun yang lebih dekat daripada \omterm{def:g11:lenses-and-eye:eye}{titik dekatnya}
(\qty{1.0}{m}) menjadi kabur: lengan yang terentang membawa halamannya
sedekat mungkin ke \qty{1.0}{m} --- tajam, sekalipun kecil.

\textbf{12.} $C = 1/(-1.0) - 1/(-0.25) = -1.0 + 4.0 = +3.0$~D.

\textbf{13.} $\gamma = (-1.0)/(-0.25) = +4.0$: empat kali lebih besar
tetapi juga empat kali lebih jauh --- ukuran sudutnya sama; yang
diperolehnya bukan ukuran melainkan \emph{ketajaman}, sebab bayangannya
kini terletak di \omterm{def:g11:lenses-and-eye:eye}{titik dekatnya}.

\textbf{14.} Agar tajam, bayangannya harus berada di luar \qty{1.0}{m}:
dari halaman pada \qty{25}{cm} (bayangannya pada \qty{1.0}{m}) sampai
halaman pada $f' = 1/3.0 \approx \qty{0.33}{m}$ (bayangannya di tak
terhingga). Benda yang jauh membentuk bayangan \qty{33}{cm} \emph{di
belakang} kacamatanya --- \omterm{def:g11:lenses-and-eye:images}{bayangan nyata} yang tidak dapat dipakai
matanya: dunia yang jauh menjadi kabur.

\textbf{15.} Dari sinilah \omterm{def:g11:lenses-and-eye:lens}{lensa} dwifokus: kekuatan bacanya pada paruh
bawah \omterm{def:g11:lenses-and-eye:lens}{lensanya}, dan tanpa kekuatan (atau dengan koreksi jarak jauh)
pada paruh atasnya.

\textbf{16.} $1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$:
$\overline{OA'} = \qty{-20}{cm}$, \omterm{def:g11:lenses-and-eye:images}{maya}, tegak, $\gamma = +5.0$.

\textbf{17.} $1/\overline{OA} = -1/25 - 1/5.0 = -6/25$:
$\overline{OA} \approx \qty{-4.2}{cm}$;
$\gamma = (-25)/(-25/6) = +6.0$.

\textbf{18.} Batunya di $F$: bayangannya di tak terhingga, yang
dipandang tanpa \omterm{def:g11:lenses-and-eye:eye}{akomodasi} sama sekali --- berjam-jam memeriksa tanpa
mata yang lelah.

\textbf{19.} $C_\infty = 1/0.017 \approx 59$~D;
$C_{25} = 1/0.017 + 1/0.25 \approx 63$~D: sekitar $4$~D \omterm{def:g11:lenses-and-eye:eye}{akomodasi}.

\textbf{20.} $f' = 1/58.8 \approx \qty{17.0}{mm}$ berbanding
$1/62.8 \approx \qty{15.9}{mm}$: sekitar \qty{1.1}{mm}, kira-kira
$6\%$ --- zum bawaan yang ditirukan kamera dengan pergeseran
\qty{2.6}{mm} dan ditambal sang nenek dengan $+3$~D.
\end{solution}
