\section*{Bab \ref{ch:g11:mechanical-energy} --- Energi Mekanik dan Kekekalannya}

\begin{solution}{exo:g11:mechanical-energy:1}
(a) $v = \qty{6.94}{m/s}$:
$E_k = \tfrac12 \times 90 \times 6.94^2 \approx \qty{2.2e3}{J}$.
(b) $v = \qty{36.1}{m/s}$:
$E_k = \tfrac12 \times 1200 \times 36.1^2 \approx \qty{7.8e5}{J}$.
(c) $\times 4$; $\times 9$ --- karena kuadratnya.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:2}
$\Delta E_p = 75 \times 9.81 \times 850 \approx \qty{6.3e5}{J}$. Tidak:
\omterm{def:g11:mechanical-energy:potential}{aras acuannya} menggeser kedua nilainya sama banyak. Tidak juga: hanya
selisih ketinggiannya yang masuk hitungan.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:3}
$v = \sqrt{2 \times 9.81 \times 20} = \qty{19.8}{m/s} \approx
\qty{71}{km/h}$. Karena $mgh = \tfrac12 mv^2$: massanya saling
menghapus.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:4}
$W = \Delta E_k = 0 - \tfrac12 \times 1200 \times 25^2 =
\qty{-3.75e5}{J}$; $F = \num{3.75e5}/50 = \qty{7.5e3}{N}$.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:5}
$v = \sqrt{2 \times 9.81 \times 0.080} \approx \qty{1.3}{m/s}$; bandul
itu naik kembali sampai \qty{8.0}{cm} (kekekalannya).
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:6}
$\num{1.5e9}/\num{2.6e3} \approx \num{5.9e5}$: sekitar enam ratus ribu
peluru. $h = 88.9^2/(2 \times 9.81) \approx \qty{4.0e2}{m}$ --- lajunya
sendiri dapat mengangkat kereta itu setinggi $400$ \omterm{def:g10:orders-of-magnitude:unit}{meter}.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:7}
Lajunya sama, yaitu
$v = \sqrt{2 \times 9.81 \times 3.0} \approx \qty{7.7}{m/s}$ (hanya
selisih ketinggiannya yang masuk hitungan). Seluncuran yang curam
menang dalam waktu: ia mencapai laju tinggi lebih cepat dan \omterm{def:g10:relative-motion:trajectory}{lintasannya}
lebih pendek.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:8}
$h = v^2/2g = 144/19.62 \approx \qty{7.3}{m}$. Pada \qty{4.0}{m}:
$v = \sqrt{144 - 2 \times 9.81 \times 4.0} \approx \qty{8.1}{m/s}$ ---
sama pada kedua arahnya: $E_m$ bergantung pada ketinggiannya, bukan
pada arah perjalanannya.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:9}
$v = \sqrt{23^2 + 2 \times 9.81 \times 35} = \sqrt{1216} \approx
\qty{35}{m/s}$. Pada kenyataannya lebih kecil: hambatan udara
mendisipasikan sebagian $E_m$.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:10}
$\Delta E_m = \tfrac12 \times 45 \times 8.0^2 - 45 \times 9.81 \times
6.0 = 1440 - 2649 \approx \qty{-1.2e3}{J}$;
$f = 1209/15 \approx \qty{81}{N}$. \omterm{def:g10:energy-conservation:energy}{Energinya} menjadi kalor, pada
peluncur dan saljunya.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:11}
$v = \sqrt{2 \times 9.81 \times 0.15} \approx \qty{1.7}{m/s}$; bandul
itu naik lagi sampai \qty{15}{cm}. Pasaknya mengubah \omterm{def:g10:relative-motion:trajectory}{lintasannya}
(lingkaran yang lebih rapat), bukan \omterm{def:g10:energy-conservation:energy}{energinya}: ketinggian dan lajunya
tidak tersentuh.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:12}
$v = \sqrt{2 \times 9.81 \times 2.5} = \qty{7.0}{m/s}$. Tinggi
kenaikannya berskala dengan $E_m$, sehingga $\times 0.9$ tiap
penyeberangan: $2.25$, $2.03$, $1.82$, $1.64$, $1.48$, $1.33$, $1.20$,
$1.08$, $0.97$ --- pertama kali di bawah \qty{1.0}{m} pada
penyeberangan ke-$9$.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:13}
\omterm{def:g10:energy-conservation:energy}{Energi} awalnya $E_m = \tfrac12 m \times 15^2 + m \times 9.81 \times 45$
sama berapa pun sudutnya, sehingga laju di tanahnya adalah
$v = \sqrt{225 + 2 \times 9.81 \times 45} = \sqrt{1108} \approx
\qty{33}{m/s}$. Sudutnya menetapkan \emph{arah} kecepatan
pendaratannya, lama terbangnya dan \omterm{def:g11:fundamental-interactions:strong}{jangkauannya} --- bukan laju
pendaratannya.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:14}
$\tfrac12 mv^2 = mgh - f\ell = 80 \times 9.81 \times 120 - 65 \times
800 = \num{94176} - \num{52000} = \qty{4.22e4}{J}$, sehingga
$v = \sqrt{2 \times \num{42176}/80} \approx \qty{32}{m/s}$
(\qty{117}{km/h}). Bagian yang \omterm{def:g11:mechanical-energy:dissipation}{terdisipasi}:
$52000/94176 \approx 55\%$.
\end{solution}

\begin{solution}{exo:g11:mechanical-energy:15}
$m = \qty{2.0e9}{kg}$:
$E_p = \num{2.0e9} \times 9.81 \times 300 \approx \qty{5.9e12}{J}
= \num{5.9e12}/\num{3.6e6} \approx \qty{1.6e6}{kW.h}$. Yang diperoleh
kembali: $0.85 \times \num{1.64e6} \approx \qty{1.4e6}{kW.h}$ ---
sekitar $\num{1.4e5}$ rumah selama satu hari.
\end{solution}

\begin{solution}{pb:g11:mechanical-energy:1}
\textbf{1.} $v_{\text{puncak}} = \sqrt{gR} = \sqrt{9.81 \times 8.0}
\approx \qty{8.9}{m/s}$.

\textbf{2.} $E_m = mg\,(2R) + \tfrac12 m gR = \num{3.14e5} +
\num{7.8e4} \approx \qty{3.92e5}{J}$.

\textbf{3.} $mgH_{\min} = E_m$ memberikan $H_{\min} = \qty{20}{m}$;
secara aljabar $H_{\min} = 2R + \frac{gR}{2g} = \frac{5R}{2}$.

\textbf{4.} $v_{\text{puncak}} = \sqrt{2g(H - 2R)} =
\sqrt{2 \times 9.81 \times 8.0} \approx \qty{12.5}{m/s}$;
$v_{\text{puncak}}^2/(gR) = 2.0$ --- dua kali nilai terkecilnya.

\textbf{5.} Setiap \omterm{def:g10:energy-conservation:energy}{energinya} sebanding dengan $m$, sehingga $m$ saling
menghapus: lajunya sama dan keamanannya sama, penuh atau kosong.

\textbf{6.} $v = \sqrt{2 \times 9.81 \times 24} \approx \qty{21.7}{m/s}
\approx \qty{78}{km/h}$.

\textbf{7.} $v = \sqrt{2 \times 9.81 \times (24 - 8.0)} \approx
\qty{17.7}{m/s}$.

\textbf{8.} $mgH = \qty{4.71e5}{J}$. Pasangan $(E_p, E_k)$: pada
peluncurannya $(\num{4.71e5}, 0)$; di dasarnya $(0, \num{4.71e5})$; di
sisinya $(\num{1.57e5}, \num{3.14e5})$; di puncaknya
$(\num{3.14e5}, \num{1.57e5})$ --- tiap pasangnya berjumlah
\qty{4.71e5}{J}.

\textbf{9.} $v = \sqrt{2 \times 9.81 \times (24 - 12)} \approx
\qty{15.3}{m/s}$.

\textbf{10.} $v(z) = \sqrt{2g(H - z)}$: paling cepat di titik terendah
relnya, paling lambat di puncak putarannya.

\textbf{11.} Ketinggiannya sama, sehingga $\Delta E_p = 0$ dan
$\Delta E_m = \Delta E_k = \tfrac12 \times 2000 \times (19.2^2 -
20.4^2) \approx \qty{-4.75e4}{J}$: penempatannya membuat selisihnya
murni kinetik, yang cukup dibaca dari dua pengukur laju.

\textbf{12.} $W(\vect f) = \qty{-4.75e4}{J}$;
$f = \num{4.75e4}/60 \approx \qty{7.9e2}{N}$.

\textbf{13.} Menjadi \omterm{def:g10:energy-conservation:forms}{energi termal}, pada rodanya, relnya dan udaranya.

\textbf{14.} $E_m$ pada pengindra pertamanya:
$\num{4.16e5} + \num{3.9e4} =
\qty{4.55e5}{J}$; kehilangannya
$\num{4.75e4}/\num{4.55e5} \approx 10\%$ tiap \qty{60}{m}. Pada
$H_{\min}$, puncak putarannya tercapai \emph{tepat} pada $\sqrt{gR}$;
kehilangan berapa pun menjatuhkannya di bawah laju amannya.

\textbf{15.} Karena kehilangan $\approx 10\%$ tiap \qty{60}{m}, sekitar
$15\%$ dari $mgH$ ($\approx \qty{7.1e4}{J}$) sudah lenyap:
$E_m \approx \qty{4.00e5}{J}$, masih di atas \qty{3.92e5}{J} yang
dituntut --- aman, dengan kelebihan hanya $2\%$. Kelebihan tadi bukan
kemewahan.

\textbf{16.} $E_k = \tfrac12 \times 2000 \times 18.0^2 =
\qty{3.24e5}{J}$.

\textbf{17.} $F = \num{3.24e5}/45 = \qty{7.2e3}{N}$.

\textbf{18.} $a = 7200/2000 = \qty{3.6}{m/s^2} \approx 0.37\,g$: tegas
tetapi masih nyaman.

\textbf{19.} \omterm{def:g10:inertia:inventory}{Gesekan} relnya: $mgH - \num{3.24e5} = \num{470880} -
\num{324000} = \qty{1.47e5}{J}$. Auditnya:
$\num{1.47e5} + \num{3.24e5} =
\num{4.71e5} = mgH$ --- setiap \omterm{def:g11:work-of-force:work}{joulenya}
terhitung.

\textbf{20.} Tinggi peluncuran $H = \qty{24}{m}$ menetapkan keamanan
putarannya, setiap lajunya $\sqrt{2g(H - z)}$, tagihan \omterm{def:g10:inertia:inventory}{gesekannya} dan
\omterm{def:g10:inertia:force}{gaya} remnya; harga mengabaikan \omterm{def:g10:inertia:inventory}{gesekan} adalah kelebihannya --- yaitu
$\qty{4}{m}$ bukit yang diam-diam dilahap dunia yang bergesekan.
\end{solution}
