\section*{Bab \ref{ch:g11:circuits-and-power} --- Rangkaian Listrik dan Daya}

\begin{solution}{exo:g11:circuits-and-power:1}
$q = It = 2.0 \times 90 \times 60 = \qty{1.1e4}{C}$. Banyaknya muatan
dasar: $q/e = 1.08\times10^{4}/1.6\times10^{-19} \approx \num{6.8e22}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:2}
$I = U/R = 12/220 = \qty{0.055}{A} = \qty{55}{mA}$;
$R = U/I = 230/0.50 = \qty{460}{\ohm}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:3}
Seri: $120 + 60 = \qty{180}{\ohm}$. Paralel:
$\frac{120 \times 60}{180}
= \qty{40}{\ohm}$ --- lebih kecil daripada
keduanya: \omterm{def:g11:circuits-and-power:voltage}{tegangannya} sama, tetapi arusnya memperoleh \omterm{def:g10:relative-motion:trajectory}{lintasan} kedua,
sehingga arus seluruhnya lebih besar dan \omterm{def:g11:circuits-and-power:resistance}{hambatannya} lebih kecil.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:4}
$P = UI = 230 \times 8.7 = \qty{2.0}{kW}$. Dalam $t = \qty{240}{s}$:
$E = Pt = 2000 \times 240 = \qty{4.8e5}{J} = 4.8\times10^{5}/3.6
\times10^{6} \approx \qty{0.13}{kW.h}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:5}
$E = 60 \times 5.0 = \qty{300}{W.h} = \qty{0.30}{kW.h} = 0.30 \times
3.6\times10^{6} = \qty{1.1e6}{J}$. Biayanya:
$0.30 \times 0.25 = 0.075$ euro tiap hari.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:6}
\emph{1.} $R = U^2/P = 230^2/1200 = \qty{44}{\ohm}$;
$I = P/U = 1200/230 = \qty{5.2}{A}$.

\emph{2.} $P' = U'^2/R = 115^2/44.1 = \qty{300}{W}$: seperempat, bukan
separuh, sebab menyeparuhkan $U$ juga menyeparuhkan $I$, dan
$P = UI \propto U^2$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:7}
$R_{\text{s}} = \qty{60}{\ohm}$, sehingga $I = 12/60 = \qty{0.20}{A}$.
$U_1 = R_1 I = \qty{8.0}{V}$, $U_2 = \qty{4.0}{V}$ (jumlahnya
\qty{12}{V}); $P_1 = U_1 I = \qty{1.6}{W}$, $P_2 = \qty{0.8}{W}$:
seluruhnya \qty{2.4}{W} $= UI = 12 \times 0.20$, sebagaimana
seharusnya.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:8}
$I_1 = 12/60 = \qty{0.20}{A}$, $I_2 = 12/30 = \qty{0.40}{A}$,
seluruhnya $I = \qty{0.60}{A}$. Penggantinya:
$R = U/I = 12/0.60 = \qty{20}{\ohm}$, dan memang
$\frac{60 \times 30}{60 + 30} = \qty{20}{\ohm}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:9}
$\mathcal{E} - 0.50\,r = 5.5$ dan $\mathcal{E} - 2.0\,r = 4.0$; dengan
mengurangkannya, $1.5\,r = 1.5$, sehingga $r = \qty{1.0}{\ohm}$ dan
$\mathcal{E} =
\qty{6.0}{V}$. Hubung singkatnya:
$\mathcal{E}/r = \qty{6.0}{A}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:10}
Ketelnya $2200/230 = \qty{9.6}{A}$, pemanggang rotinya
$1000/230 = \qty{4.3}{A}$: seluruhnya \qty{13.9}{A} $< \qty{16}{A}$,
\omterm{def:g11:circuits-and-power:joule}{sekringnya} bertahan. Pemanasnya $1500/230 =
\qty{6.5}{A}$: seluruhnya
\qty{20.4}{A} $> \qty{16}{A}$, \omterm{def:g11:circuits-and-power:joule}{sekringnya} putus (setara dengan
$\qty{4.7}{kW} > 230 \times 16 \approx \qty{3.7}{kW}$).
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:11}
Pada \qty{200}{V}: $I = 10^4/200 = \qty{50}{A}$, rugi \omterm{def:g10:energy-conservation:power}{dayanya}
$RI^2 = 2.0 \times
50^2 = \qty{5.0}{kW}$ --- separuh yang diserahkan.
Pada \qty{4.0}{kV}: $I =
\qty{2.5}{A}$, rugi \omterm{def:g10:energy-conservation:power}{dayanya}
$2.0 \times 2.5^2 = \qty{12.5}{W}$. Nisbahnya $400 =
20^2$: \omterm{def:g11:circuits-and-power:voltage}{tegangan}
dua puluh kali lipat berarti arus dua puluh kali lebih kecil, dan rugi
\omterm{def:g10:energy-conservation:power}{dayanya} berjalan seperti $I^2$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:12}
Ada empat susunan: ketiganya seri, $3 \times 60 = \qty{180}{\ohm}$;
ketiganya paralel, $60/3 = \qty{20}{\ohm}$; satu seri dengan dua yang
paralel, $60 + 30 = \qty{90}{\ohm}$; satu paralel dengan dua yang seri,
$\frac{60 \times 120}{180} = \qty{40}{\ohm}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:13}
\emph{1.} $I = \mathcal{E}/(R + r) = 4.5/7.5 = \qty{0.60}{A}$;
$U = RI = \qty{3.6}{V}$; \omterm{def:g10:energy-conservation:power}{daya} bergunanya $P = UI = \qty{2.2}{W}$; yang
hilang $rI^2 =
1.5 \times 0.60^2 = \qty{0.54}{W}$;
$\eta = U/\mathcal{E} = 3.6/4.5 =
0.80$.

\emph{2.} Menurut petunjuknya, $P = \mathcal{E}^2 \big/ \big[(R - r)^2/R +
4r\big]$; kurungannya bernilai terkecil ($= 4r$) tepat ketika
$R = r$, sehingga
$P_{\max} = \mathcal{E}^2/(4r) = 4.5^2/6.0 = \qty{3.4}{W}$.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:14}
Kalornya: $Q = 1.0 \times 4180 \times (100 - 20) = \qty{3.3e5}{J}$.
Listriknya: $E = Q/0.90 = \qty{3.7e5}{J}$. Waktunya:
$t = E/P =
3.7\times10^{5}/2200 \approx \qty{170}{s}$, kurang dari tiga
menit. Biayanya:
$3.7\times10^{5}/3.6\times10^{6} \approx \qty{0.10}{kW.h}$, sekitar
$2.6$ sen.
\end{solution}

\begin{solution}{exo:g11:circuits-and-power:15}
\emph{1.} $q = 60 \times 3600 = \qty{2.2e5}{C}$;
$E \approx \mathcal{E}q
= 12.7 \times 2.16\times10^{5} = \qty{2.7e6}{J} \approx \qty{0.76}{kW.h}$.

\emph{2.} $U = 12.7 - 0.020 \times 150 = \qty{9.7}{V}$; yang diserahkan
$UI = \qty{1.5}{kW}$; yang hilang
$rI^2 = 0.020 \times 150^2 = \qty{450}{W}$.

\emph{3.} Lampu depannya terpasang pada kutub akinya: selagi distarter,
\omterm{def:g11:circuits-and-power:voltage}{tegangan} jepitnya turun dari $12.7$ menjadi $\qty{9.7}{V}$, sehingga
lampu itu menerima \omterm{def:g10:energy-conservation:power}{daya} yang lebih kecil lalu meredup.
\end{solution}

\begin{solution}{pb:g11:circuits-and-power:1}
\textbf{1.} Tiap sekon, $E_p = mgh = 5.0\times10^{4} \times 9.81 \times
200 = \qty{9.8e7}{J}$: \omterm{def:g10:energy-conservation:power}{daya} yang tersedia \qty{98}{MW}.

\textbf{2.} $0.90 \times 98 = \qty{88}{MW}$.

\textbf{3.} $I = P/U = 8.8\times10^{7}/2.0\times10^{4} = \qty{4.4e3}{A}$.

\textbf{4.} $R_\ell I^2 = 32 \times (4.4\times10^{3})^2 \approx
\qty{6.2e8}{W} = \qty{620}{MW}$ --- tujuh kali keluaran pembangkitnya:
mengangkut \omterm{def:g10:energy-conservation:power}{daya} pada \qty{20}{kV} tidak mungkin dilakukan.

\textbf{5.} Dengan $P = UI$ yang tetap, hanya menaikkan $U$ yang
menyusutkan $I$: dengan sebuah transformator.

\textbf{6.} $I = 8.8\times10^{7}/4.0\times10^{5} = \qty{220}{A}$.

\textbf{7.} $R_\ell I^2 = 32 \times 220^2 \approx \qty{1.6}{MW}$.

\textbf{8.} $1.6/88 \approx 0.018$: sekitar $1.8\%$.

\textbf{9.} Bagiannya $= R_\ell I^2/P = R_\ell (P/U)^2/P = R_\ell P/U^2
\propto 1/U^2$: dari \qty{20}{kV} ke \qty{400}{kV} bagian itu turun
sebesar $20^2 = 400$, dan memang
$\qty{620}{MW}/400 \approx \qty{1.6}{MW}$.

\textbf{10.} $R_\ell I = 32 \times 220 \approx \qty{7.1}{kV}$: $1.8\%$
dari \qty{400}{kV}, bagian yang sama seperti pada pertanyaan 8, sebab
$R_\ell I/U = R_\ell I^2/UI$.

\textbf{11.} $\qty{1}{kW.h} = \qty{3.6e6}{J}$: meterannya mengalikan
\omterm{def:g10:energy-conservation:power}{daya} yang ditarik dengan lamanya \omterm{def:g10:energy-conservation:power}{daya} itu ditarik, lalu menjumlahkannya.

\textbf{12.} $I = 900/230 = \qty{3.9}{A}$;
$R = U^2/P = 230^2/900 \approx \qty{59}{\ohm}$.

\textbf{13.} $E = 900 \times 180 = \qty{1.6e5}{J} = \qty{0.045}{kW.h}$:
sekitar $1.1$ sen.

\textbf{14.} Fisika $RI^2$-nya serupa; hanya tujuannya yang berbeda ---
di salurannya kalor itu tidak diinginkan, di pemanggang rotinya kalor
itu \emph{adalah} barang jadinya.

\textbf{15.} $3.9 + 9.6 = \qty{13.5}{A} < \qty{16}{A}$: bertahan.
Pemanasnya $2000/230 = \qty{8.7}{A}$: seluruhnya
$\qty{22.2}{A} > \qty{16}{A}$, \omterm{def:g11:circuits-and-power:joule}{sekringnya} putus.

\textbf{16.} $0.05 \times 13.5^2 = \qty{9.1}{W}$:
$9.1/3100 \approx
0.3\%$ dari \omterm{def:g10:energy-conservation:power}{daya} yang diserahkan --- dapat diabaikan
pada kawat yang pendek.

\textbf{17.} $\eta = 0.90 \times 0.99 \times 0.982 \times 0.99 \times
0.997 \approx 0.86$.

\textbf{18.} Satu \omterm{def:g10:energy-conservation:kwh}{kilowatt-jam} yang memanggang roti memerlukan
$3.6\times10^{6}/0.86
\approx \qty{4.2e6}{J}$ di hulunya:
$m = E/(gh) = 4.2\times10^{6}/(9.81
\times 200) \approx \qty{2.1e3}{kg}$,
yaitu sekitar \qty{2.1}{m^3} air.

\textbf{19.} $R_\ell P/U^2 = 32 \times 8.8\times10^{7}/(2.0\times
10^{4})^2 \approx 7$. ``Bagian'' yang lebih besar daripada $1$ berarti
salurannya akan membuang lebih banyak daripada yang dibawanya:
penyerahan \omterm{def:g10:energy-conservation:power}{dayanya} sama sekali tidak dapat terjadi pada \omterm{def:g11:circuits-and-power:voltage}{tegangan} itu.

\textbf{20.} Satu \omterm{def:g10:energy-conservation:kwh}{kilowatt-jam} yang memanggang roti berharga sekitar dua
ton --- \qty{2.1}{m^3} --- air yang jatuh setinggi \qty{200}{m} bagi
gunungnya, dan sekitar $14\%$ \omterm{def:g10:energy-conservation:energy}{energinya} mati di perjalanan, sebagian
besar di turbin dan transformatornya (kurang dari $2\%$ pada
\qty{1000}{km} salurannya). Satu muslihat yang membuat perjalanan itu
mungkin adalah transformator: menaikkan \omterm{def:g11:circuits-and-power:voltage}{tegangannya} ke \qty{400}{kV}
membagi arusnya dengan $20$ dan rugi Joule-nya dengan $400$.
\end{solution}
