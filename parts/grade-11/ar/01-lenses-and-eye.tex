\section{العدسات المجمِّعة والمفرِّقة}

\begin{definition}[العدسة الرقيقة]\label{def:g11:lenses-and-eye:lens}
\emph{العدسة}\index{عدسة} وسط شفاف يحدّه سطحان،
أحدهما منحنٍ على الأقل. فإن كانت أسمك في المركز منها عند الحافة، كانت
\emph{مجمِّعة}\index{عدسة مجمِّعة}: فتُحرف الأشعة المتوازية
بعضها نحو بعض؛ وإن كانت أسمك عند الحافة، كانت \emph{مفرِّقة}\index{عدسة
مفرِّقة}: فتتباعد الأشعة. والعدسة \emph{الرقيقة}\index{عدسة رقيقة} عدسة
سُمكها مهمَل: فنرسمها قطعة ذات رؤوس سهام إلى الخارج (للمجمِّعة) أو
إلى الداخل (للمفرِّقة). ومركزها $O$ هو \emph{المركز
البصري}\index{المركز البصري}؛ والمستقيم العمودي المارّ بالنقطة $O$ هو
\emph{المحور الرئيسي}\index{المحور الرئيسي}.
\end{definition}

\begin{definition}[البؤرتان والبعد البؤري]\label{def:g11:lenses-and-eye:focal}
الأشعة الواصلة موازية لمحور \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة} تعبر كلها
المحورَ عند نقطة واحدة خلفها، هي \emph{بؤرة الصورة}\index{بؤرة}
$F'$؛ وبالتناظر، الأشعة المغادرة من \emph{بؤرة الجسم}
$F$ (أمامها، حيث $OF = OF'$) تخرج موازية للمحور. و\emph{البعد
البؤري}\index{البعد البؤري} $f' = \overline{OF'}$
موجب في \omterm{def:g11:lenses-and-eye:lens}{العدسة المجمِّعة}؛ أما عبر \omterm{def:g11:lenses-and-eye:lens}{عدسة مفرِّقة} فتخرج الأشعة المتوازية
كأنها آتية من نقطة $F'$ \emph{أمام} \omterm{def:g11:lenses-and-eye:lens}{العدسة}: أي $f' < 0$.
\end{definition}

\begin{definition}[التقارب]\label{def:g11:lenses-and-eye:vergence}
\emph{تقارب}\index{تقارب} عدسةٍ بعدها البؤري $f'$
(\omterm{def:g10:orders-of-magnitude:unit}{بالمتر}) هو $C = 1/f'$،
ووحدته \emph{الديوبتر}\index{ديوبتر} (رمزه D):
$1~\text{D} = \qty{1}{m^{-1}}$؛ والعدسات \omterm{def:g11:lenses-and-eye:lens}{المفرِّقة} لها $C < 0$.
ويصف اختصاصيو البصريات بالديوبتر: فتقاربات العدسات \omterm{def:g11:lenses-and-eye:lens}{الرقيقة} المتلامسة
تتجمع ببساطة (نقبله هنا؛ ويُشتقّ في مجلد السنة الأولى).
\end{definition}

\begin{example}[قراءة وصفة طبية]\label{ex:g11:lenses-and-eye:vergence}
\omterm{def:g11:lenses-and-eye:lens}{عدسة} لها $f' = \qty{0.50}{m}$ تقاربها $C = 1/0.50 = +2.0$~D:
فهي \omterm{def:g11:lenses-and-eye:lens}{مجمِّعة}. ووصفة $-4.0$~D تعني $f' = \qty{-0.25}{m}$:
أي \omterm{def:g11:lenses-and-eye:lens}{مفرِّقة}، وبعدها البؤري \qty{25}{cm}.
\end{example}

\section{بناء الصورة}

\begin{method}[أشعة البناء الثلاثة]\label{met:g11:lenses-and-eye:rays}
ليكن $AB$ جسمًا عموديًّا على المحور عند $A$. ومن بين كل الأشعة
المغادرة للرأس $B$، ثلاثة مساراتها معروفة:
\begin{enumerate}
  \item الشعاع المارّ \omterm{def:g11:lenses-and-eye:lens}{بالمركز البصري} $O$ يمضي مستقيمًا؛
  \item والشعاع الموازي للمحور يخرج مارًّا \omterm{def:g11:lenses-and-eye:focal}{بالبؤرة} $F'$؛
  \item والشعاع المارّ \omterm{def:g11:lenses-and-eye:focal}{بالبؤرة} $F$ يخرج موازيًا للمحور.
\end{enumerate}
ويكفي أيّ اثنين منها: فتقاطعهما هو نقطة الصورة $B'$، وتقف
الصورة $A'B'$ عمودية على المحور عند $A'$. وإذا تباعدت الأشعة
الخارجة، التقت امتداداتها \emph{الخلفية} (المتقطعة) عند
نقطة \omterm{def:g11:lenses-and-eye:images}{صورة خيالية}.
\end{method}

\begin{definition}[الصور الحقيقية والصور الخيالية]\label{def:g11:lenses-and-eye:images}
تكون الصورة \emph{حقيقية}\index{صورة حقيقية} إذا مرّت بها الأشعة
الخارجة فعلًا: فتلتقطها شاشة توضع هناك؛ وتكون
\emph{خيالية}\index{صورة خيالية} إذا لم تلتقِ إلا امتداداتها الخلفية:
فلا تلتقطها شاشة، لكن عينًا تنظر في \omterm{def:g11:lenses-and-eye:lens}{العدسة} تراها
تمام الرؤية. وتعطي \omterm{def:g11:lenses-and-eye:lens}{العدسة المجمِّعة} صورة حقيقية مقلوبة لأي
جسم وراء $F$؛ وأزلِق الجسم داخل \omterm{def:g11:lenses-and-eye:focal}{البعد البؤري} فتنقلب
الصورة خيالية معتدلة مكبَّرة.
\end{definition}

\section{صيغة واحدة: علاقة الاقتران}

يبيّن البناء \emph{أين} تقع الصورة؛ أما الصيغة فتحسبها.
وجّه المحور في جهة الضوء واستعمل \emph{القياسات الجبرية}:
و$\overline{OA}$ فاصلة $A$ بالنسبة إلى $O$ (وهي سالبة في حالة
جسم حقيقي أمام \omterm{def:g11:lenses-and-eye:lens}{العدسة})؛ والارتفاعات $\overline{AB}$ تُعدّ موجبة إلى أعلى.

\begin{theorem}[علاقة الاقتران للعدسة الرقيقة]\label{thm:g11:lenses-and-eye:conjugate}
عبر \omterm{def:g11:lenses-and-eye:lens}{عدسة رقيقة} بعدها البؤري $f'$، تُصوَّر نقطة جسم $A$ على
المحور عند النقطة $A'$ حيث
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'} = C .
\]
\end{theorem}
\admitted

\begin{remark}
تنتج العلاقة عن تطبيق \omterm{thm:g10:refraction:snell}{قانون سنيل} على الأشعة القريبة من
المحور؛ ويُجعل الاشتقاق --- وحدود نموذج \omterm{def:g11:lenses-and-eye:lens}{العدسة الرقيقة} ---
أمينًا في مجلد السنة الأولى.
\end{remark}

\begin{definition}[التكبير]\label{def:g11:lenses-and-eye:magnification}
\emph{تكبير}\index{تكبير} الصورة هو
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}}
       = \frac{\overline{OA'}}{\overline{OA}} .
\]
ويقارن $\abs{\gamma}$ الأحجام، وتقارن إشارته الاتجاهات ($\gamma < 0$:
مقلوبة)؛ أما \omterm{def:g10:signals-and-waves:signal}{إشارة} $\overline{OA'}$ فتقرأ الطبيعة (موجبة:
\omterm{def:g11:lenses-and-eye:images}{حقيقية} خلف \omterm{def:g11:lenses-and-eye:lens}{العدسة}؛ وسالبة: \omterm{def:g11:lenses-and-eye:images}{خيالية} أمامها).
\end{definition}

\begin{example}[الصيغة في العمل]\label{ex:g11:lenses-and-eye:conjugate}
يقف جسم على بعد \qty{30}{cm} أمام \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة} لها
$f' = \qty{10}{cm}$: فيكون $\overline{OA} = \qty{-30}{cm}$، ومنه
\[
\frac{1}{\overline{OA'}} = \frac{1}{10} - \frac{1}{30} = \frac{1}{15},
\qquad \overline{OA'} = +\qty{15}{cm},
\qquad \gamma = \frac{15}{-30} = -0.50 :
\]
أي \omterm{def:g11:lenses-and-eye:images}{صورة حقيقية} على بعد \qty{15}{cm} خلف \omterm{def:g11:lenses-and-eye:lens}{العدسة}، مقلوبة، بنصف الحجم ---
وهو بالضبط ما ترسمه الأشعة الثلاثة.
\end{example}

\section{العين}

\begin{definition}[العين جملةً بصرية]\label{def:g11:lenses-and-eye:eye}
بصريًّا، العين \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة} تواجه شاشة: فتعمل
\emph{القرنية}\index{قرنية} و\emph{العدسة البلورية}\index{عدسة
بلورية} عمل \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة} واحدة بعدها البؤري قابل للضبط؛ أما
\emph{الشبكية}\index{شبكية}، على بعد نحو \qty{17}{mm} خلفها، فهي الشاشة
الثابتة. ويجري التبؤر بعصر \omterm{def:g11:lenses-and-eye:lens}{العدسة} لا بتحريكها كما في
آلة التصوير: و\emph{التكيّف}\index{تكيّف} هو تغيّر
$f'$ الذي تُحدثه العضلات الهدبية. وأقرب نقطة تُرى حادّة عند تمام
التكيّف هي \emph{النقطة القريبة}\index{النقطة القريبة} --- ونحو
\qty{25}{cm} في \emph{العين المعيارية}\index{العين المعيارية} --- وأبعدها،
في الراحة، هي \emph{النقطة البعيدة}\index{النقطة البعيدة} (وهي اللانهاية في
العين المعيارية).
\end{definition}

\begin{example}[تقارب عينك]\label{ex:g11:lenses-and-eye:eyerange}
تقع \omterm{def:g11:lenses-and-eye:eye}{الشبكية} عند $\overline{OA'} = +\qty{0.017}{m}$. وفي حالة جسم بعيد
($1/\overline{OA} \approx 0$): $C = 1/0.017 \approx 59$~D. وعند \omterm{def:g11:lenses-and-eye:eye}{النقطة القريبة}

ضبط دقيق مدمج قدره $+4$~D.
\end{example}

\section{العيوب وتصحيحها والمكبّرة}

\begin{definition}[قصر النظر وطول النظر]\label{def:g11:lenses-and-eye:defects}
العين \emph{القصيرة النظر}\index{قصر النظر} مفرطة التجميع
بالنسبة إلى طولها: فتتبأر الأشعة المتوازية \emph{أمام} \omterm{def:g11:lenses-and-eye:eye}{الشبكية}،
وتتشوّش الأجسام البعيدة، وتقع \omterm{def:g11:lenses-and-eye:eye}{النقطة البعيدة} على مسافة منتهية؛ وتصححها
\omterm{def:g11:lenses-and-eye:lens}{عدسة} نظارة \omterm{def:g11:lenses-and-eye:lens}{مفرِّقة}. أما العين \emph{الطويلة النظر}\index{طول النظر}
فليست \omterm{def:g11:lenses-and-eye:lens}{مجمِّعة} بما يكفي: إذ ستتبأر الأجسام القريبة
خلف \omterm{def:g11:lenses-and-eye:eye}{الشبكية}، وتتراجع \omterm{def:g11:lenses-and-eye:eye}{النقطة القريبة}؛ وتصححها \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة}.
والتقدم في السنّ يصلّب \omterm{def:g11:lenses-and-eye:eye}{العدسة البلورية}
(\emph{قصو البصر}\index{قصو البصر})، فيتقلص \omterm{def:g11:lenses-and-eye:eye}{التكيّف} --- والعلاج
نفسه: نظارات القراءة.
\end{definition}

\begin{example}[وصفة لقصير النظر]\label{ex:g11:lenses-and-eye:correction}
عين قصيرة النظر نقطتها البعيدة على \qty{50}{cm}: فيجب أن تُظهر \omterm{def:g11:lenses-and-eye:lens}{عدسة}
التصحيح الأجسام البعيدة \emph{عند \omterm{def:g11:lenses-and-eye:eye}{النقطة البعيدة}} --- أي جسم في
اللانهاية وصورة على $\overline{OA'} = \qty{-0.50}{m}$ (\omterm{def:g11:lenses-and-eye:images}{خيالية}،
أمامها). ومنه $1/f' = 1/\overline{OA'}$: أي $f' = \qty{-0.50}{m}$،
و$C = -2.0$~D؛ فتنظر العين المرتخية إلى تلك \omterm{def:g11:lenses-and-eye:images}{الصورة الخيالية} فتراها
حادّة.
\end{example}

\begin{definition}[المكبّرة]\label{def:g11:lenses-and-eye:magnifier}
\emph{المكبّرة}\index{مكبّرة} \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة} قصيرة \omterm{def:g11:lenses-and-eye:focal}{البعد
البؤري} تُستعمل والجسم \emph{داخل} \omterm{def:g11:lenses-and-eye:focal}{البعد البؤري}: فتكون \omterm{def:g11:lenses-and-eye:images}{الصورة
خيالية} معتدلة مكبَّرة (\cref{def:g11:lenses-and-eye:images})،
وتشاهدها العين مرتاحة على مسافة بعيدة.
\end{definition}

\begin{example}[مكبّرة جواهري]\label{ex:g11:lenses-and-eye:magnifier}
\omterm{def:g11:lenses-and-eye:magnifier}{مكبّرة} لها $f' = \qty{5.0}{cm}$، والحجر على \qty{4.0}{cm}:
$1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$، ومنه
$\overline{OA'} = \qty{-20}{cm}$ و$\gamma = (-20)/(-4.0) = +5.0$: أي
\omterm{def:g11:lenses-and-eye:images}{صورة خيالية} معتدلة، أكبر خمس مرات، على مسافة مريحة قدرها \qty{20}{cm}
من \omterm{def:g11:lenses-and-eye:lens}{العدسة}.
\end{example}

\section{مسألة: صيغة واحدة وثلاث أدوات}

\begin{problem}[{مسألة نهاية الأسبوع --- علاقة الاقتران نفسها
تبئّر آلة تصوير، وتصف نظارة قراءة، وتصمّم مكبّرة،
وتقيس التقريب المدمج في عينك أنت}]
\label{pb:g11:lenses-and-eye:1}
يحرّك مصوّر \omterm{def:g11:lenses-and-eye:lens}{عدسة} \qty{50}{mm}، وتمسك جدّة الجريدة
على مدّ ذراعها، وينحني جواهري على ماسة: فسطر واحد من
الجبر، $1/\overline{OA'} - 1/\overline{OA} = 1/f'$
(\cref{thm:g11:lenses-and-eye:conjugate})، يدير المهن الثلاث ---
ثم يدير العين نفسها.

\medskip
\noindent\textbf{الجزء I --- صيغة واحدة.} \omterm{def:g11:lenses-and-eye:lens}{عدسة مجمِّعة} لها
$f' = \qty{20}{cm}$.
\begin{enumerate}
  \item جسم على \qty{60}{cm}: احسب $\overline{OA'}$ و
        $\gamma$؛ وصِف الصورة.
  \item انقل الجسم إلى \qty{30}{cm}؛ وأعِد الحساب. فما الذي تبادل مواضعه
        مع السؤال 1؟ (فمسارات الضوء عكوسة.)
  \item وانقله إلى \qty{10}{cm}، داخل $f'$: أعِد الحساب وصِف.
  \item لخّص: كيف تتحرك الصورة وتتغير طبيعتها كلما انزلق الجسم من
        بعيد جدًّا نحو $F$، ثم تجاوزها؟
  \item بيّن أن الجسم البعيد جدًّا ($1/\overline{OA} \approx 0$)
        يُصوَّر دائمًا في المستوي البؤري: $\overline{OA'} = f'$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- المصوّر.}
\omterm{def:g11:lenses-and-eye:lens}{لعدسة} آلة تصوير $f' = \qty{50}{mm}$؛ والحسّاس، وارتفاعه \qty{24}{mm}،
يمكن تحريكه بالنسبة إلى \omterm{def:g11:lenses-and-eye:lens}{العدسة}.
\begin{enumerate}[resume]
  \item أين يقع الحسّاس عند التبؤر على منظر بعيد؟
  \item والآن وجه على بعد \qty{1.0}{m}: احسب المسافة الجديدة بين \omterm{def:g11:lenses-and-eye:lens}{العدسة}
        والحسّاس والانتقال من وضع المنظر.
  \item وتسمح الآلية بمسافة بين \omterm{def:g11:lenses-and-eye:lens}{العدسة} والحسّاس أقصاها \qty{55}{mm}.
        احسب أقرب مسافة تبؤر.
  \item وعند ذلك التبؤر الأقرب، احسب $\gamma$. فهل تسع صورة
        وجه ارتفاعه \qty{20}{cm} على الحسّاس؟
  \item والتصوير ``الماكرو'' بالحجم الطبيعي يعني $\gamma = -1$: فأين يجب أن يقع الجسم
        والحسّاس عندئذٍ، ولماذا يقتضي ذلك \omterm{def:g11:lenses-and-eye:lens}{عدسة} خاصة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الجدّة.}
تتراجع \omterm{def:g11:lenses-and-eye:eye}{النقطة القريبة} مع السنّ: فنقطة الجدّة على \qty{1.0}{m}.
\begin{enumerate}[resume]
  \item اشرح مسألة الجريدة على مدّ الذراع.
  \item ولتقرأ على \qty{25}{cm}، يجب أن تحوّل نظارتها صفحة على
        \qty{25}{cm} إلى \omterm{def:g11:lenses-and-eye:images}{صورة خيالية} على \qty{1.0}{m}: احسب
        \omterm{def:g11:lenses-and-eye:vergence}{التقارب} المطلوب.
  \item احسب $\gamma$. والصورة أكبر أربع مرات، ومع ذلك لا تبدو
        أكبر: اشرح، مقارنًا الأحجام \emph{والمسافات}.
  \item بيّن أن المطبوع لا يكون حادًّا إلا بين \qty{25}{cm} و
        $f' \approx \qty{33}{cm}$. وماذا عن جسم \emph{بعيد} يُرى
        عبر النظارة؟
  \item استنتج في جملة واحدة لماذا توجد العدسات ثنائية \omterm{def:g11:lenses-and-eye:focal}{البؤرة}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الجواهري، وتقريب العين نفسها.}
\omterm{def:g11:lenses-and-eye:magnifier}{لمكبّرة} الجواهري $f' = \qty{5.0}{cm}$؛ ونقطته القريبة على
\qty{25}{cm}؛ وشبكيته على بعد \qty{17}{mm} خلف \omterm{def:g11:lenses-and-eye:lens}{عدسة} عينه.
\begin{enumerate}[resume]
  \item حجر على \qty{4.0}{cm} تحت \omterm{def:g11:lenses-and-eye:magnifier}{المكبّرة}: فما الموضع والطبيعة
        \omterm{def:g11:lenses-and-eye:magnification}{والتكبير}؟
  \item وأين يجب أن يقع الحجر لتقع صورته عند نقطته
        القريبة، وما $\gamma$ عندئذٍ؟
  \item ولماذا يضع الجواهريون المتمرسون الحجر عند $F$ بالضبط
        بدل ذلك (فتقع الصورة في اللانهاية)؟
  \item والآن العين وحدها: احسب تقاربها وهي تنظر إلى جسم
        بعيد، ثم إلى جسم على \qty{25}{cm}: فكم ديوبترًا من
        \omterm{def:g11:lenses-and-eye:eye}{التكيّف}؟
  \item حوّل التقاربين إلى بعدين بؤريين: فبكم، بالمِلّيمتر
        وبالمئة، تغيّر العين بعدها البؤري
        --- أي التقريب الذي فعلته آلة التصوير بالانتقال وفعلته الجدّة
        بمقدار $+3$~D؟
\end{enumerate}
\end{problem}
