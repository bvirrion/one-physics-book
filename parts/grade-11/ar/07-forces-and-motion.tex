\chapter{القوى والحركة}\label{ch:g11:forces-and-motion}

اقذف مفتاح ربط يدور عبر الغرفة: فتتمايل طرفاه تمايلًا عنيفًا، ومع ذلك
ترسم نقطة واحدة فيه قطعًا مكافئًا تامًّا. وقد حسم \Cref{ch:g10:inertia}
ما يحدث حين تتعادل القوى --- لا شيء. ويسأل هذا الفصل
عمّا يحدث حين لا تتعادل: \omterm{def:g10:inertia:force}{فالقوة} الفائضة \emph{تغيّر
السرعة} --- فتسرّع الأجسام أو تبطّئها أو تديرها ---
ويقول بكم، بالنِّسب إن لم يكن بالصيغ بعد.

\section{مركز الكتلة}

\begin{definition}[مركز الكتلة]\label{def:g11:forces-and-motion:com}
\emph{مركز الكتلة}\index{مركز الكتلة} لجسم أو لجملة هو
نقطة توازنها --- أي الموضع المتوسط لكتلتها: فهو المركز
الهندسي في الجسم المتجانس المتناظر؛ وفي كتلتين نقطيتين هو نقطة
القطعة الواصلة على مسافتين بالنسبة \emph{العكسية} للكتلتين،
$\frac{d_1}{d_2} = \frac{m_2}{m_1}$ --- أي أقرب إلى الأثقل.
\end{definition}

\begin{example}[قضيب أثقال غير متوازن]\label{ex:g11:forces-and-motion:dumbbell}
كرتان كتلتاهما \qty{2.0}{kg} و\qty{1.0}{kg} عند طرفي قضيب خفيف
طوله \qty{0.90}{m}. ويقسمه \omterm{def:g11:forces-and-motion:com}{مركز الكتلة} بالنسبة العكسية
$d_1/d_2 = 1.0/2.0$: أي على بعد $\qty{0.30}{m}$ من الكرة الثقيلة، و
$\qty{0.60}{m}$ من الخفيفة --- فوازِن القضيب هناك.
\end{example}

\begin{remark}[مفتاح الربط المقذوف]\label{rem:g11:forces-and-motion:wrench}
صوّر مفتاح ربط يتقلّب: فيتبع كل نقطة منه منحنًى معقّدًا ملتفًّا
--- إلا \omterm{def:g11:forces-and-motion:com}{مركز الكتلة}، فيرسم القطع المكافئ النظيف الذي ترسمه كرة
مقذوفة. فحركة \omterm{def:g11:forces-and-motion:com}{مركز الكتلة} بسيطة؛ وما بقي دوران
\emph{حوله} --- ولهذا يجوز للميكانيك أن يعامل سيارة أو كوكبًا أو
لاعب جمباز نقطةً واحدة.
\end{remark}

\section{القوة المحصّلة تغيّر السرعة}

\begin{definition}[القوة المحصّلة]\label{def:g11:forces-and-motion:netforce}
\emph{القوة المحصّلة}\index{قوة محصّلة} على جسم هي المجموع المتجهي
لكل القوى المؤثرة فيه، $\vect F = \vect F_1 + \vect F_2 + \dots$
(رأسًا إلى ذيل، كما في مجلد الرياضيات)؛ وتتعادل القوى
تمامًا حين $\vect F = \vect 0$.
\end{definition}

\begin{theorem}[أثر القوة المحصّلة --- شبه كمّي]\label{thm:g11:forces-and-motion:secondlaw}
\index{القانون الثاني للحركة}
\omterm{def:g11:forces-and-motion:netforce}{القوة المحصّلة} $\vect F$ المؤثرة في جسم كتلته $m$ خلال مدة قصيرة
$\Delta t$ تغيّر سرعة مركز كتلته بمتجهة
$\Delta \vect v$ لها \emph{جهة} $\vect F$، وتنمو شدتها
تناسبًا مع $F$ ومع $\Delta t$ وتتقلص
تناسبًا عكسيًّا مع الكتلة:
\[
\Delta v \propto \frac{F \, \Delta t}{m} .
\]
\end{theorem}
\admitted

\begin{remark}
ضاعِف \omterm{def:g10:inertia:force}{القوة} أو مدتها: يتضاعف التغير؛ وضاعِف الكتلة: ينتصف.
وستشحذ السنة القادمة ذلك ليصير معادلة تامة، ويسأل مجلد السنة
الأولى في أي مراجع يصحّ؛ أما هذه السنة فالنِّسب تكفي ---
وهي تدير الوسائد الهوائية والمكابح والمدارات أصلًا.
\end{remark}

\begin{proposition}[الحالات النموذجية الثلاث]\label{prop:g11:forces-and-motion:cases}
ليتحرك جسم بالسرعة $\vect v$ تحت \omterm{def:g11:forces-and-motion:netforce}{قوة محصّلة} $\vect F$.
\begin{enumerate}
  \item $\vect F$ في جهة $\vect v$: فيتسارع \emph{مسرعًا} في خط مستقيم؛
  \item $\vect F$ عكس $\vect v$: فيتباطأ \emph{مبطئًا} في خط مستقيم؛
  \item $\vect F$ عمودية على $\vect v$: فينعطف \emph{منعطفًا} دون
        تغيّر السرعة.
\end{enumerate}
\end{proposition}

\begin{proof}
أضِف $\Delta\vect v$، الموجَّهة في جهة $\vect F$
(\cref{thm:g11:forces-and-motion:secondlaw})، رأسًا إلى ذيل على
$\vect v$: ففي جهة $\vect v$ تتراكم السهام (فتطول)؛ وفي الجهة المعاكسة
تتلاشى جزئيًّا (فتقصر)؛ وفي الجهة العمودية، تميل كل $\Delta \vect v$
صغيرة السهمَ دون أن تطيله --- فلا يدور إلا الاتجاه.
\end{proof}

\begin{example}[ركلة جانبية]\label{ex:g11:forces-and-motion:kick}
ينزلق قرص شرقًا بسرعة \qty{6.0}{m/s}؛ وتضيف ضربة جانبية قصيرة
$\Delta\vect v = \qty{8.0}{m/s}$ نحو الشمال تمامًا. فتكون السرعة الجديدة
المجموع المتجهي: شدتها $\sqrt{6.0^2 + 8.0^2} = \qty{10.0}{m/s}$،
واتجاهها $\tan\theta = 8.0/6.0$، أي\ $\theta \approx 53^\circ$ شمال
الشرق --- فالسرعة القديمة لا تُمحى، إنما يُضاف إليها التغير.
\end{example}

\section{مقارنة التغيرات: القوة والزمن والكتلة}

\begin{method}[الاستدلال بالنِّسب]\label{met:g11:forces-and-motion:ratio}
لمقارنة وضعين، لا تحلّ معادلة أبدًا --- بل كوّن نسبًا: فنسبة
تغير السرعة هي نسبة $F$ مضروبة في نسبة $\Delta t$،
مقسومة على نسبة $m$. ثم اقرأ أيّ عامل تغيّر، وبكم.
\end{method}

\begin{example}[العربة المحمَّلة]\label{ex:g11:forces-and-motion:cart}
الدفعة نفسها، مطبَّقة \omterm{def:g10:orders-of-magnitude:unit}{الثانية} نفسها، تُطلق عربة تسوّق فارغة
كتلتها \qty{20}{kg} بسرعة \qty{1.2}{m/s}. وإذا حُمّلت إلى
\qty{60}{kg} --- أي ثلاثة أضعاف الكتلة، وبالمقدار $F\Delta t$ نفسه --- غادرت
بثلث ذلك، أي \qty{0.40}{m/s}. فالكتلة عناد المادة.
\end{example}

\begin{remark}[التصادمات والوسائد الهوائية والركبتان المثنيتان]\label{rem:g11:forces-and-motion:airbag}
في تصادم، لا يقبل $\Delta v$ و$m$ التفاوض --- إذ يجب أن تبلغ
سرعة الراكب الصفر --- ومن ثم يكون $F\,\Delta t$ ثابتًا. والحرية الوحيدة
الباقية هي \emph{الزمن}: فمدّ $\Delta t$ تتقلص \omterm{def:g10:inertia:force}{القوة} بالنسبة
نفسها. فالرأس المتوقف على وسادة هوائية في \qty{0.08}{s} بدل توقفه على
المِقود في \qty{0.01}{s} يحسّ \omterm{def:g10:inertia:force}{قوة} متوسطة أصغر ثماني مرات؛
ومناطق الانسحاق ورغوة الخوذة والركبتان المثنيتان تلعب الحيلة نفسها.
\end{remark}

\section{الاحتكاك: القوة التي تصارع الانزلاق}

\begin{definition}[الاحتكاك السكوني والحركي]\label{def:g11:forces-and-motion:friction}
يعاكس \omterm{def:g10:inertia:inventory}{الاحتكاك} بين السطحين المتلامسين \emph{انزلاقهما
النسبي}. و\emph{الاحتكاك السكوني}\index{احتكاك!سكوني} يؤثر ما داما
لا ينزلقان، ويضبط نفسه حتى حدّ أقصى ليلغي كل ما يحاول
أن يبدأ الانزلاق؛ أما \emph{الاحتكاك الحركي}\index{احتكاك!حركي}،
متى بدأ الانزلاق، فهو ثابت تقريبًا، موجَّه ضد
الانزلاق، وهو عادةً \emph{أضعف} من الحدّ السكوني الأقصى.
\end{definition}

\begin{example}[دفع الخزانة]\label{ex:g11:forces-and-motion:wardrobe}
ادفع خزانة برفق: فلا تتحرك --- إذ يلغي \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني}
دفعتك \qty{150}{N}. وادفع أقوى: فعند \qty{400}{N} ما تنطلق فجأة
في الحركة، ثم تنزلق بجهد أقل --- إذ تولّى \omterm{def:g11:forces-and-motion:friction}{الاحتكاك الحركي}
الأضعف الأمر.
\end{example}

\begin{remark}[لماذا تعمل مكابح ABS]\label{rem:g11:forces-and-motion:abs}
نقطة تماس العجلة الدائرة لا تنزلق على الطريق: فيمسك الإطار
\omterm{def:g10:inertia:inventory}{باحتكاك} \emph{سكوني}. أما العجلة المقفلة فتنزلق ولا تنال إلا
\omterm{def:g10:inertia:inventory}{الاحتكاك} \emph{الحركي} الأضعف، وهو يتجه ضد الانزلاق
مهما تفعل العجلات --- فيموت التوجيه. ويحرّر نظام ABS المكبح
لحظةَ تُقفل عجلة: فمزيد من التماسك، وتوقف أقصر، وسائق يقدر على التوجيه.
\end{remark}

\section{الانعطاف: الحركة الدائرية تحتاج قوة نحو الداخل}

\begin{definition}[الحركة الدائرية المنتظمة]\label{def:g11:forces-and-motion:ucm}
يكون الجسم في \emph{حركة دائرية منتظمة}\index{حركة دائرية
منتظمة} إذا كان مساره دائرة تُقطع بسرعة ثابتة. والسرعة،
وهي مماسّة للدائرة، تغيّر اتجاهها في كل لحظة ---
ومن ثم فهي ليست ثابتة أبدًا.
\end{definition}

\begin{proposition}[القوة نحو الداخل]\label{prop:g11:forces-and-motion:inward}
الجسم في \omterm{def:g11:forces-and-motion:ucm}{حركة دائرية منتظمة} تؤثر فيه \omterm{def:g11:forces-and-motion:netforce}{قوة محصّلة}
لا تنعدم أبدًا: فهي تتجه في كل لحظة من الجسم نحو مركز
الدائرة.
\end{proposition}

\begin{proof}
السرعة ثابتة، ومن ثم ليس \omterm{def:g11:forces-and-motion:netforce}{للقوة المحصّلة} جزء على $\vect v$
(أي الحالتان 1 و2 في \cref{prop:g11:forces-and-motion:cases}): فهي
عمودية على المماسّ، ومن ثم على نصف القطر --- ونحو الداخل،
لأن كل $\Delta \vect v$ تحني \omterm{def:g10:relative-motion:trajectory}{المسار} نحو المركز.
\end{proof}

\begin{example}[الخيط والمنعطف والقمر]\label{ex:g11:forces-and-motion:moon}
كل انعطاف ثابت يخفي \omterm{def:g10:inertia:force}{قوة} نحو الداخل. ففي الكرة المدارة: \omterm{def:g10:inertia:inventory}{شدّ}
الخيط. وفي السيارة على منعطف: \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني} الجانبي بين الطريق و
الإطارات --- وهو يزول على الجليد، فتمضي السيارة مستقيمة. وفي القمر: جذب الأرض
الثقالي (\cref{ch:g10:universal-gravitation})، وهو يديره
في \num{27.3} يومًا --- فهو ليس ممسوكًا \emph{من أعلى} بل مجذوبًا جانبيًّا إلى
انعطاف دائم، ساقطًا حول الأرض؛ وستجعل السنة القادمة من ذلك
ميكانيك التوابع.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:1}
أالقوة المحصّلة منعدمة أم لا؟ برّر انطلاقًا من السرعة: (أ) سيارة تنساب
مستقيمة بسرعة ثابتة \qty{90}{km/h}؛ (ب) تكبح في خط مستقيم؛
(ج) تدور في منعطف بسرعة ثابتة \qty{50}{km/h}؛ (د) قافز بمظلة عند
السرعة الحدية.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:2}
كرتان كتلتاهما \qty{3.0}{kg} و\qty{1.0}{kg} عند طرفي قضيب خفيف
طوله \qty{0.80}{m}. عيّن \omterm{def:g11:forces-and-motion:com}{مركز الكتلة}؛ وأين يقع لو تساوت الكتلتان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:3}
تُعطى الدفعة القصيرة نفسها لعربة أمتعة فارغة كتلتها \qty{15}{kg}
وللعربة نفسها محمَّلة إلى \qty{45}{kg}. قارن تغيري
السرعة؛ وأيّ دفعة تعطي العربة المحمَّلة $\Delta v$ الخاصة بالفارغة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:4}
ترام يتحرك بسرعة \qty{10}{m/s} يكبح؛ وبعد \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة يتحرك بسرعة
\qty{8}{m/s} في الاتجاه نفسه. أعطِ $\Delta \vect v$ وجهة
\omterm{def:g11:forces-and-motion:netforce}{القوة المحصّلة}. وأيّ حالة من حالات \cref{prop:g11:forces-and-motion:cases} هذه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:5}
\omterm{def:g10:inertia:inventory}{احتكاك} سكوني أم حركي --- وفي أي جهة؟ (أ) صندوق يقاوم
دفعتك (اللطيفة أكثر مما ينبغي)؛ (ب) الصندوق نفسه متى انزلق؛
(ج) كتاب على صينية مائلة قليلًا، لا ينزلق.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:6}
\omterm{def:g11:forces-and-motion:netforce}{قوة محصّلة} $F$ مطبَّقة مدة $\Delta t$ على كتلة $m$ تنتج
$\Delta v = \qty{4.0}{m/s}$. تنبّأ بالمقدار $\Delta v$ من أجل: (أ) $2F, \Delta t, m$؛
(ب) $F, \Delta t/2, m$؛ (ج) $F, \Delta t, 2m$؛ (د) $3F, 2\Delta t, 6m$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:7}
ينزلق قرص شرقًا بسرعة \qty{8.0}{m/s}؛ وتضيف ضربة
$\Delta \vect v = \qty{6.0}{m/s}$ نحو الشمال تمامًا. احسب السرعة الجديدة و
اتجاهها. ولماذا ليست السرعة $8.0 + 6.0 = \qty{14}{m/s}$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:8}
يوقف حارس مرمى التسديدة نفسها بطريقتين: بذراعين متصلبتين (فتتوقف الكرة في
\qty{0.02}{s}) أو ``بالتراخي'' مع الكرة (\qty{0.08}{s}). قارن
القوتين المتوسطتين، واذكر أيّ المقادير كانت مثبَّتة قبل اختياره.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:9}
أيّ \omterm{def:g10:inertia:force}{قوة} خارجية تدفع عدّاءً إلى الأمام عند الانطلاق؟ ولماذا تساعد
الأحذية ذات المسامير --- ولماذا لا يقدر أحد على التسارع، بل ولا على المشي، على
جليد بلا \omterm{def:g10:inertia:inventory}{احتكاك} تمامًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:10}
في توقف طارئ، لماذا تتوقف سيارة فيها ABS (وعجلاتها تظل تدور)
في مسافة أقصر عادةً من سيارة عجلاتها مقفلة منزلقة --- ولماذا
لا تعود السيارة المنزلقة قابلة للتوجيه؟ استشهد
بما جاء في \cref{def:g11:forces-and-motion:friction}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:11}
يدير رامي مطرقة الكرةَ على سلكها. فأيّ \omterm{def:g10:inertia:force}{قوة} تدير
الكرة؟ ارسم مسارها بعد الإطلاق. ولماذا لا يحتاج القمر سلكًا،
وماذا كان سيصير مساره بلا جاذبية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:12}
المكنسة عصا منتظمة كتلتها \qty{1.0}{kg} (ومركز كتلتها
على بعد \qty{0.60}{m} من الأعلى) مع رأس كتلته \qty{2.0}{kg} ومركز
كتلته على بعد \qty{1.20}{m} من الأعلى. بمعاملة كل جزء كتلةً نقطية،
عيّن مركز كتلة المكنسة وتحقّق من قاعدة النسبة العكسية في
\cref{def:g11:forces-and-motion:com}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:13}
سيارة كتلتها $m$ وشاحنة صغيرة محمَّلة كتلتها $2m$ تكبحان من السرعة
نفسها \omterm{def:g10:inertia:force}{بقوة} الكبح نفسها.
\begin{enumerate}
  \item قارن الزمنين اللازمين للتوقف.
  \item تفقد كلٌّ منهما سرعتها بانتظام، ومن ثم تنتقل بالسرعة
        \emph{المتوسطة} نفسها: قارن مسافتي التوقف.
  \item والآن تكبح السيارة من \emph{ضعف} السرعة، \omterm{def:g10:inertia:force}{بالقوة} نفسها:
        قارن الزمن والمسافة بتوقفها الأول. فما الحكمة في حدود السرعة؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:14}
يدور القمر على بعد $R \approx \qty{3.84e8}{m}$ في $T = \num{27.3}$ يومًا.
\begin{enumerate}
  \item احسب سرعته المدارية (المحيط على \omterm{def:g10:signals-and-waves:period}{الدور}).
  \item وفي أي جهة يتجه تغير سرعته، وأيّ \omterm{def:g10:inertia:force}{قوة}
        توفّره؟
  \item حسّن، باستعمال \cref{prop:g11:forces-and-motion:inward}، العبارة:
        ``لا يسقط القمر لأنه يتحرك بسرعة كافية''.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:15}
بيضة كتلتها \qty{50}{g} تصطدم بالأرضية بسرعة \qty{4.0}{m/s}: فعلى البلاط
تتوقف في نحو \qty{1}{ms}، وعلى رغوة سميكة في نحو \qty{50}{ms}. قارن
القوتين المتوسطتين. واشرح، بالاستدلال نفسه، لماذا تثني
ركبتيك عند الهبوط من قفزة، وممّ تُصنع حصيرة لاعب الجمباز.
\end{exercise}

\section{مسألة: اختبار التصادم}

\begin{problem}[{مسألة نهاية الأسبوع --- تصميم التصادم اللطيف:
بما أن التصادم يثبّت تغير السرعة، فكل جهاز أمان في سيارة
آلةٌ لمدّ الزمن}]
\label{pb:g11:forces-and-motion:1}
يقود مختبر لاختبار التصادم سيارة بسرعة \qty{50}{km/h} نحو جدار
صلب، وفيها دمية بالغ كتلتها \qty{70}{kg} ودمية طفل كتلتها \qty{20}{kg}؛
وتوقّت الكاميرات كل توقف. ومهمتك: أن تجد أين يسكن عنف
التصادم، وأن تصمّمه بعيدًا.

\medskip
\noindent\textbf{الجزء I --- مقياس التصادم.}
\begin{enumerate}
  \item حوّل \qty{50}{km/h} إلى \unit{m/s}.
  \item في أي توقف أمامي يكون لتغير سرعة الركاب الشدة
        نفسها. فأيّها؟ ولماذا لا يقدر أي جهاز على تغييرها؟
  \item استشهد بما جاء في \cref{thm:g11:forces-and-motion:secondlaw}: فمع تثبيت $\Delta v$
        و$m$ يكون $F\,\Delta t$ ثابتًا. فما المقبض الوحيد للمصمّم؟
  \item يوقف الجدار السيارة في \qty{0.050}{s}؛ وتمدّ منطقة انسحاق
        ذلك إلى \qty{0.150}{s}. قارن القوتين المتوسطتين.
  \item وفي أي جهة يتجه $\Delta \vect v$ \omterm{def:g11:forces-and-motion:netforce}{والقوة المحصّلة} على
        الدميتين أثناء التوقف؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- منطقة الانسحاق.}
\begin{enumerate}[resume]
  \item لماذا، في جملة واحدة، يصمّم المهندسون مقدمة السيارة
        لتُدمَّر؟
  \item في سيارة قديمة صلبة (والتوقف في \qty{0.050}{s}) تُربط الدمية
        بصلابة إلى المقعد. قارن قوتها بتوقف \qty{0.150}{s} الحديث.
  \item تهبط سرعة السيارة المتوقفة بانتظام (والمتوسط: نصف السرعة الابتدائية).
        فكم تقطع في \qty{0.150}{s}؟ قارن ذلك بمنطقة انسحاق \omterm{def:g11:lenses-and-eye:images}{حقيقية}.
  \item ولماذا لا نصنع منطقة انسحاق طولها \qty{10}{m}؟ وأيّ جزء من السيارة
        يجب أن \emph{لا} ينسحق؟
  \item لخّص الجزء II شعارًا: ``لا تقدر على اختيار $\Delta v$،
        فعليك أن تختار \dots''.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الحزام والوسادة الهوائية والطفل.}
\begin{enumerate}[resume]
  \item تحفظ دمية غير مربوطة بالحزام \qty{14}{m/s} (\cref{ch:g10:inertia})
        حتى يوقفها الزجاج الأمامي في \qty{0.005}{s}؛ أما المربوطة بالحزام
        فتتوقف مع السيارة في \qty{0.150}{s}. قارن القوتين المتوسطتين.
  \item ولماذا يكون حزام يتمطط قليلًا أفضل من
        طقم فولاذي غير قابل للتمطط؟
  \item كان الرأس سيتوقف على المِقود في \qty{0.010}{s}؛ وتمدّ الوسادة الهوائية
        ذلك إلى \qty{0.080}{s}. قارن القوتين.
  \item يخضع البالغ والطفل للتغير $\Delta v$ نفسه في الزمن $\Delta t$
        نفسه. قارن القوتين المحصّلتين اللتين يجب أن يوفّرهما الحزامان.
  \item طفل على حِجر: يجب أن تغيّر الذراعان سرعة رضيع كتلته \qty{10}{kg}
        بمقدار \qty{14}{m/s} في \qty{0.150}{s}، بينما تغيّر \omterm{def:g11:fundamental-interactions:four}{الجاذبية}
        سرعةً بمقدار \qty{9.8}{m/s} في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}. فكم مرة يجب أن تعمل
        الذراعان أسرع --- وحمل أيّ كتلة كان سيبدو بذلك الثِّقل؟
        استنتج.
  \item يحسّ الأطفال قوى أصغر --- فلماذا مقاعد الأطفال إذن؟ فكّر في
        \emph{أين} يطبّق الحزام قوته، وفي $\Delta t$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- قفص الأمان، والحكم.}
\begin{enumerate}[resume]
  \item حجرة الركاب صلبة بينما ينسحق الطرفان. فلماذا يجب ألّا
        تتشوّه الحجرة، مع أن الصلابة تعني توقفات عنيفة؟
  \item تتصادم سيارتان متطابقتان وجهًا لوجه، كلٌّ بسرعة \qty{50}{km/h}.
        فأين تتوقف كلٌّ منهما؟ قارن كل تصادم باختبار الجدار.
  \item تلاقي سيارة صغيرة شاحنة محمَّلة وجهًا لوجه. والقوتان اللتان تمارسانهما
        إحداهما على الأخرى متساويتان متعاكستان، متبادلتان كأي تفاعل
        (\cref{ch:g11:fundamental-interactions}). فأيّهما تتغير سرعتها
        أكثر؟
  \item أحصِ أجهزة الأمان في هذه المسألة والمقدار الواحد
        الذي يمدّه كلٌّ منها.
  \item الخاتمة: اذكر قانون تصميم التصادم اللطيف في جملة
        واحدة لا تستعمل فيها إلا \emph{الكتلة} و\emph{تغير السرعة}
        و\emph{\omterm{def:g10:inertia:force}{القوة}} و\emph{الزمن}.
\end{enumerate}
\end{problem}
