\chapter{المجال الكهربائي ومجال الجاذبية}\label{ch:g11:electric-gravitational-fields}

كيف تجذب الأرض تفاحة لا تلمسها قط؟ لقد سمّى نيوتن نفسه
التأثير عن بعد ``سخفًا عظيمًا'' حتى إنه رفض الدفاع
عنه. أما الجواب الحديث: فكتلة، أو شحنة، تعدّل الفضاء حولها
--- أي تنشئ \emph{مجالًا} --- ويستجيب كل ما يوضع هناك
\omterm{def:g11:electric-gravitational-fields:field}{للمجال} في موضعه هو. ونبني المثالين الكبيرين جنبًا
إلى جنب، حتى يصير التماثل تطابقًا في الصورة.

\section{من التأثير عن بعد إلى المجال}

\begin{definition}[المجال]\label{def:g11:electric-gravitational-fields:field}
\emph{المجال}\index{مجال} يسند إلى كل نقطة من الفضاء متجهة. و
المنبع (شحنة أو كتلة) ينشئ مجاله في كل مكان حوله،
سواء وُجد هناك ما يحسّه أم لا؛ ثم يحسّ جسم اختبار
قوةً يحدّدها المجال \emph{في موضعه هو} وحده. وهذه
الصورة ذات الخطوتين تحلّ محلّ التأثير عن بعد.
\end{definition}

\begin{remark}
\omterm{def:g11:electric-gravitational-fields:field}{المجال} ليس حيلة محاسبية: فهو يحمل \omterm{def:g10:energy-conservation:energy}{طاقة}، وتنتقل التغيرات في
المنبع إلى الخارج خلاله بسرعة منتهية --- والضوء نفسه
\omterm{def:g11:electric-gravitational-fields:field}{مجال} منتقل كهذا (\cref{ch:g12:light-as-wave}). وندرس هنا
مجالات لا تتغير مع الزمن.
\end{remark}

\section{المجال الكهربائي}

\begin{definition}[المجال الكهربائي]\label{def:g11:electric-gravitational-fields:efield}
إذا أحسّت شحنة اختبار صغيرة $q$ موضوعة في نقطة $M$ \omterm{def:g10:inertia:force}{قوة} كهربائية
$\vect F$، فإن \emph{المجال الكهربائي}\index{مجال كهربائي} عند $M$ هو
\[
\vect E = \frac{\vect F}{q}
\qquad \text{(بوحدة \unit{N/C}، وتُكتب أيضًا \unit{V/m})}.
\]
ولا يتوقف $\vect E$ إلا على المنابع وعلى $M$، لا على شحنة
الاختبار: فمضاعفة $q$ تضاعف $\vect F$، ويبقى الخارج على حاله.
\end{definition}

\begin{proposition}[القوة على شحنة]\label{prop:g11:electric-gravitational-fields:force}
الشحنة $q$ الموضوعة حيث \omterm{def:g11:electric-gravitational-fields:field}{المجال} $\vect E$ تحسّ \omterm{def:g10:inertia:force}{القوة}
$\vect F = q \vect E$: أي في جهة $\vect E$ إذا كان $q > 0$، وعكس $\vect E$
إذا كان $q < 0$، وشدتها $F = \abs{q}\,E$.
\end{proposition}

\begin{proof}
أعِد ترتيب التعريف؛ وقاعدة \omterm{def:g10:signals-and-waves:signal}{الإشارة} جبر مضاعفٍ سالب.
\end{proof}

\begin{example}[قراءة مجال]\label{ex:g11:electric-gravitational-fields:reading}
شحنة اختبار $q = \qty{+2.0e-8}{C}$ عند $M$ تحسّ \omterm{def:g10:inertia:force}{قوة} قدرها
\qty{6.0e-4}{N} متجهة شرقًا. \omterm{def:g11:electric-gravitational-fields:field}{فالمجال} عند $M$ هو
$E = \num{6.0e-4} / \num{2.0e-8} = \qty{3.0e4}{N/C}$، متجهًا شرقًا؛
والشحنة $q' = \qty{-1.0e-8}{C}$ الموضوعة في النقطة نفسها تحسّ
$F = \num{1.0e-8} \times \num{3.0e4} = \qty{3.0e-4}{N}$ --- متجهة غربًا.
\end{example}

\begin{proposition}[مجال شحنة نقطية]\label{prop:g11:electric-gravitational-fields:point}
الشحنة النقطية $Q$ تنشئ، على المسافة $d$، مجالًا على المستقيم
الواصل بين الشحنة والنقطة --- مبتعدًا عن $Q$ إذا كان $Q > 0$، ونحوها إذا كان
$Q < 0$ --- وشدته
\[
E = k\,\frac{\abs{Q}}{d^2},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{proposition}

\begin{proof}
يعطي قانون كولوم (\cref{ch:g11:fundamental-interactions}) القوةَ
على شحنة اختبار $q$ على المسافة $d$: شدتها $k \abs{qQ} / d^2$،
على المستقيم الواصل، نافرة في حالة الإشارتين المتماثلتين. اقسم على $q$ و
طبّق \cref{prop:g11:electric-gravitational-fields:force}.
\end{proof}

\begin{example}[كرة مشحونة]\label{ex:g11:electric-gravitational-fields:sphere}
كرة فان دي غراف تحمل $Q = \qty{+5.0e-7}{C}$ (والكرة المشحونة الصغيرة
تعمل عمل شحنة نقطية في مركزها). وعند $d = \qty{0.50}{m}$:
\[
E = \num{8.99e9} \times \num{5.0e-7} / 0.50^2 \approx \qty{1.8e4}{N/C},
\]
متجهًا قطريًّا بعيدًا عن الكرة. وعند \qty{1.0}{m} يكون قد هبط
إلى ربع ذلك: فقانون $1/d^2$ الذي \omterm{def:g10:inertia:force}{للقوة} يبقى في \omterm{def:g11:electric-gravitational-fields:field}{المجال}.
\end{example}

\begin{definition}[خطوط المجال]\label{def:g11:electric-gravitational-fields:lines}
\emph{خط المجال}\index{خط مجال} منحنٍ مماسّ \omterm{def:g11:electric-gravitational-fields:field}{للمجال} في كل
نقطة، وعليه سهم يعطي جهة \omterm{def:g11:electric-gravitational-fields:field}{المجال}. وثلاث قواعد:
\begin{itemize}
  \item تغادر الشحن الموجبة وتصل إلى الشحن السالبة؛
  \item وحيث تتزاحم الخطوط يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال} قويًّا، وحيث تتباعد يكون ضعيفًا؛
  \item ولا يتقاطع خطان أبدًا: فللمجال جهة واحدة في كل نقطة.
\end{itemize}
\end{definition}

\section{المجال المنتظم في مكثّفة}

\begin{definition}[المكثّفة ذات الصفيحتين المتوازيتين]\label{def:g11:electric-gravitational-fields:capacitor}
صفيحتان معدنيتان متوازيتان متقابلتان، متقاربتان وتحملان شحنتين
متعاكستين، تشكّلان \emph{مكثّفة}\index{مكثّفة}. وبين الصفيحتين
يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال} \emph{منتظمًا}\index{مجال منتظم}: أي بالشدة والجهة
نفسيهما في كل مكان --- عموديًّا على الصفيحتين، من الصفيحة الموجبة
إلى السالبة.
\end{definition}

\begin{proposition}[المجال بين الصفيحتين]\label{prop:g11:electric-gravitational-fields:uniform}
إذا كان \omterm{def:g11:circuits-and-power:voltage}{التوتر} بين الصفيحتين (\cref{ch:g11:circuits-and-power})
هو $U$ والمسافة بينهما $d$، فإن شدة \omterm{def:g11:electric-gravitational-fields:field}{المجال} بينهما
\[
E = \frac{U}{d} \qquad \text{(بوحدة \unit{V/m})}.
\]
\end{proposition}
\admitted

\begin{remark}
يُشتقّ الانتظام والعلاقة $E = U/d$ اشتقاقًا أمينًا في مجلد السنة الأولى؛ و
الصيغة تفسّر \omterm{def:g10:orders-of-magnitude:unit}{الوحدة} \unit{V/m}. والفجوات \omterm{def:g11:lenses-and-eye:lens}{الرقيقة} تصنع مجالات قوية:
إذ إنّ \qty{100}{V} عبر مِلّيمتر تعطي أصلًا \qty{e5}{V/m}.
\end{remark}

\begin{example}[توجيه حزمة إلكترونات]\label{ex:g11:electric-gravitational-fields:beam}
في \omterm{def:g10:signals-and-waves:oscilloscope}{راسم الاهتزاز}، تمرّ الحزمة بين صفيحتين حيث
$U = \qty{100}{V}$ و$d = \qty{5.0}{mm}$:
فيكون $E = 100 / \num{5.0e-3} = \qty{2.0e4}{V/m}$، ويحسّ كل إلكترون
$F = \num{1.60e-19} \times \num{2.0e4} = \qty{3.2e-15}{N}$ نحو الصفيحة
الموجبة --- أي $\num{4e14}$ ضعف وزنه البالغ \qty{8.9e-30}{N}،
فتنحرف الحزمة انحرافًا مرئيًّا (أما \omterm{def:g10:relative-motion:trajectory}{المسار} المنحني
فشأن \cref{ch:g12:newtons-laws}).
\end{example}

\section{مجال الجاذبية}

\begin{definition}[مجال الجاذبية]\label{def:g11:electric-gravitational-fields:gfield}
إذا أحسّت كتلة اختبار $m$ موضوعة في نقطة \omterm{def:g10:inertia:force}{قوة} جاذبية (أي
وزنًا) $\vect P$، فإن \emph{مجال الجاذبية}\index{مجال
الجاذبية} هناك هو
\[
\vect g = \frac{\vect P}{m} \qquad \text{(بوحدة \unit{N/kg})},
\]
وعكسًا $\vect P = m \vect g$. وقد أدخل فصل سابق
العدد $g$؛ أما $\vect g$ فتضيف الجهة التي يبدأ حجر مسقَط بالتحرك فيها.
\end{definition}

\begin{proposition}[مجال الأرض]\label{prop:g11:electric-gravitational-fields:earth}
على المسافة $d$ من مركز الأرض (وكتلتها $M$)، يتجه \omterm{def:g11:electric-gravitational-fields:gfield}{مجال
الجاذبية} نحو المركز وشدته
\[
g = G\,\frac{M}{d^2},
\qquad G = \qty{6.67e-11}{N.m^2/kg^2}.
\]
\end{proposition}

\begin{proof}
تعطي \omterm{def:g11:fundamental-interactions:four}{الجاذبية} الكونية (\cref{ch:g10:universal-gravitation}) القوةَ
على كتلة اختبار $m$: شدتها $G M m / d^2$، نحو المركز؛
فاقسم على $m$. وعند السطح يكون $d = R$ وهذا هو
$g = GM/R^2 \approx \qty{9.81}{N/kg}$ في الفصل السابق.
\end{proof}

\begin{example}[المجال حيث يعيش القمر]\label{ex:g11:electric-gravitational-fields:moon}
عند مسافة القمر $d = \qty{3.84e8}{m}$:
\[
g = \num{6.67e-11} \times \num{5.97e24} / (\num{3.84e8})^2
\approx \qty{2.7e-3}{N/kg}.
\]
\omterm{def:g11:electric-gravitational-fields:field}{فمجال} الأرض لا يتوقف أبدًا --- إنما يخفّ وفق $1/d^2$؛ وهو \omterm{def:g11:electric-gravitational-fields:field}{المجال}
الذي يمسك القمر على \omterm{def:g10:universal-gravitation:satellite}{مداره}.
\end{example}

\begin{remark}[منتظم قرب السطح]\label{rem:g11:electric-gravitational-fields:uniformg}
فوق مختبر أو حتى فوق جبل، لا تكاد $d$ تتغير أمام
$R = \qty{6.37e6}{m}$، ومن ثم يكون $\vect g$ \omterm{def:g11:electric-gravitational-fields:capacitor}{منتظمًا} بدقة عالية:
خطوط عمودية، و\qty{9.81}{N/kg} ثابتة --- \omterm{def:g11:electric-gravitational-fields:field}{فمجال} الأرض يبدو
\omterm{def:g11:electric-gravitational-fields:field}{كمجال} \omterm{def:g11:electric-gravitational-fields:capacitor}{المكثّفة}، تمامًا كما تبدو الأرض الكروية مسطحة من حديقة.
\end{remark}

\section{فكرة واحدة وقوتان}

مجالا هذا الفصل هما الرياضيات نفسها في ثوبين:

\begin{center}
\begin{tabular}{lcc}
 & \textbf{الكهربائي} & \textbf{الثقالي} \\
\hline
المنبع & الشحنة $Q$ & الكتلة $M$ \\
\omterm{def:g11:electric-gravitational-fields:field}{المجال} & $\vect E = \vect F / q$ & $\vect g = \vect P / m$ \\
\omterm{def:g10:orders-of-magnitude:unit}{الوحدة} & \unit{N/C} = \unit{V/m} & \unit{N/kg} \\
المنبع النقطي & $E = k \abs{Q} / d^2$ & $g = G M / d^2$ \\
الثابت & $k = \qty{8.99e9}{N.m^2/C^2}$ & $G = \qty{6.67e-11}{N.m^2/kg^2}$ \\
\omterm{def:g10:inertia:force}{القوة} على جسم اختبار & $\vect F = q \vect E$ & $\vect P = m \vect g$ \\
\omterm{def:g11:electric-gravitational-fields:lines}{خطوط المجال} & خارجة من $+$، داخلة إلى $-$ & داخلة إلى الكتلة دائمًا \\
\end{tabular}
\end{center}

وكل صيغة في اليمين تصير صيغة اليسار بالقاموس
$q \leftrightarrow m$، $\vect E \leftrightarrow \vect g$، $k \leftrightarrow G$.

\begin{remark}[أين ينكسر التماثل]\label{rem:g11:electric-gravitational-fields:noanalogue}
تأتي الشحنة بإشارتين؛ والكتلة \omterm{def:g10:signals-and-waves:signal}{بإشارة} واحدة. ولذلك يمكن إلغاء
المجالات الكهربائية: فالشحن المعادة ترتيبها في صندوق معدني تمحو أي \omterm{def:g11:electric-gravitational-fields:field}{مجال}
خارجي في داخله (وهو \emph{قفص فاراداي} --- ولهذا يقتل المصعد \omterm{def:g10:signals-and-waves:signal}{إشارة}
هاتفك)، والمادة بالجملة، بتوازن $+$ و$-$، صامتة
كهربائيًّا. أما \omterm{def:g11:fundamental-interactions:four}{الجاذبية} فلا تُحجب ولا تُعادَل:
فكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} يضيف جذبه --- ولهذا تحكم \omterm{def:g10:inertia:force}{القوة} الأضعف بمقدار
$\num{e39}$ (\cref{exo:g11:electric-gravitational-fields:10})
الكون على المقاييس الكبيرة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:1}
شحنة اختبار $q = \qty{+2.0e-6}{C}$ في نقطة $M$ تحسّ \omterm{def:g10:inertia:force}{قوة} قدرها
\qty{5.0e-3}{N} متجهة شمالًا. أعطِ \omterm{def:g11:electric-gravitational-fields:field}{المجال} عند $M$ (شدته و
جهته)، ثم \omterm{def:g10:inertia:force}{القوة} على $q' = \qty{-4.0e-6}{C}$ الموضوعة عند $M$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:2}
شحنة نقطية $Q = \qty{+5.0}{nC}$ تقع عند $O$.
\begin{enumerate}
  \item احسب \omterm{def:g11:electric-gravitational-fields:field}{المجال} عند $d = \qty{30}{cm}$: شدته وجهته.
  \item وعلى أي مسافة تكون شدة \omterm{def:g11:electric-gravitational-fields:field}{المجال} نصف ذلك؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:3}
\omterm{def:g11:electric-gravitational-fields:capacitor}{مكثّفة} توترها $U = \qty{600}{V}$ بين صفيحتين تفصل بينهما \qty{3.0}{mm}.
احسب \omterm{def:g11:electric-gravitational-fields:field}{المجال}، ثم \omterm{def:g10:inertia:force}{القوة} على إلكترون واحد هناك
($e = \qty{1.60e-19}{C}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:4}
على المريخ، يزن مسبار كتلته \qty{250}{kg} مقدار \qty{930}{N}. احسب \omterm{def:g11:electric-gravitational-fields:field}{مجال}
سطح المريخ $g_{\text{المريخ}}$، ثم \omterm{def:g10:universal-gravitation:weight}{وزن} رائد فضاء كتلته \qty{80}{kg}
هناك.
\end{exercise}

\section{مسألة: قطيرة ميليكان}

\begin{problem}[{مسألة نهاية الأسبوع --- \omterm{def:g10:universal-gravitation:weight}{وزن} شحنة الإلكترون: قطيرة
زيت تُركن في الهواء بين صفيحتين تكشف أن الشحنة
تأتي بخطوات غير قابلة للقسمة قدرها $\num{1.60e-19}$ \omterm{def:g11:fundamental-interactions:charge}{كولوم}}]
\label{pb:g11:electric-gravitational-fields:1}
سنة 1909 رشّ روبرت ميليكان قطيرات زيت بين الصفيحتين
الأفقيتين \omterm{def:g11:electric-gravitational-fields:capacitor}{لمكثّفة}. وبضبط \omterm{def:g11:circuits-and-power:voltage}{التوتر} حتى تتعلّق قطيرة مختارة
بلا حركة --- فتوازن \omterm{def:g10:inertia:force}{القوة} الكهربائية الوزنَ تمامًا --- استطاع
أن يزن شحنتها. وكتلة قطيرتنا $m = \qty{4.90e-15}{kg}$
(مقيسة من سقوطها البطيء \omterm{def:g11:electric-gravitational-fields:field}{والمجال} مطفأ؛ ونأخذها مسلَّمة)؛
وتفصل بين الصفيحتين $d = \qty{8.0}{mm}$، والصفيحة \emph{العليا}
موجبة؛ وشحنة القطيرة سالبة.

\medskip
\noindent\textbf{الجزء I --- المسرح.}
\begin{enumerate}
  \item صِف \omterm{def:g11:electric-gravitational-fields:field}{المجال} بين الصفيحتين؛ وماذا يشتري لنا ``الانتظام''؟
  \item من أجل $U = \qty{800}{V}$، احسب $E$.
  \item وفي أي جهة تكون \omterm{def:g10:inertia:force}{القوة} الكهربائية على القطيرة؟ ولماذا وجب أن تكون
        الصفيحة \emph{العليا} هي الموجبة؟
  \item احسب \omterm{def:g10:universal-gravitation:weight}{وزن} القطيرة.
  \item اكتب شرط السكون مساواةً بين شدتي
        قوتين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- التوازن.}
\begin{enumerate}[resume]
  \item ومنه، عبّر عن $\abs{q}$ بدلالة $m$ و$g$ و$d$ و$U$.
  \item تتعلّق القطيرة ساكنة عند $U = \qty{800}{V}$: احسب
        $\abs{q}$.
  \item اقسم على $e = \qty{1.60e-19}{C}$. فكم إلكترونًا زائدًا
        تحمل القطيرة؟
  \item يُرفع \omterm{def:g11:circuits-and-power:voltage}{التوتر} قليلًا إلى \qty{900}{V}. فإلى أي جهة تتجه
        \omterm{def:g10:inertia:force}{القوة} المحصّلة الآن؟
  \item بيّن أن \omterm{def:g11:circuits-and-power:voltage}{توتر} التوازن هو $U = mgd/\abs{q}$: فكلما كبرت
        الشحنة صغر \omterm{def:g11:circuits-and-power:voltage}{التوتر}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الخطوات.}
دفقة من الإشعاع تقدر أن تُلقي إلكترونات على القطيرة أو تنزعها منها؛ وبعد
كل تغير، يعيد ميليكان ضبط \omterm{def:g11:circuits-and-power:voltage}{التوتر} ليوازن.
\begin{enumerate}[resume]
  \item بعد دفقة واحدة، يقتضي التوازن $U = \qty{1200}{V}$. احسب
        $\abs{q}$ الجديدة. فهل كسبت القطيرة إلكترونًا أم فقدته؟
  \item وتُوجَد توازنات متتالية عند $U = 1200$ و$800$ و$600$
        و$480$ و\qty{400}{V}. احسب الشحن الخمس.
  \item بيّن أن الخمس جميعها مضاعفات صحيحة لمقدار واحد؛ وأعطِ
        قيمته.
  \item وبين توازنين متتاليين، بكم تتغير الشحنة؟
        فسّر ذلك.
  \item اشرح لماذا لا تقدر هذه القطيرة على التعلّق ساكنةً عند
        $U = \qty{700}{V}$ أبدًا، مهما طال الضبط بصبر.
  \item كرّر ميليكان ذلك على مئات القطيرات: فكانت كل شحنة مقيسة
        مضاعفًا صحيحًا للقيمة نفسها. اذكر
        الاستنتاج، وسمِّ المقدار المقيس.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- لماذا لا يوجد ميليكان ثقالي.}
\begin{enumerate}[resume]
  \item في قاموس هذا الفصل، ما الذي يقوم بأدوار $q$ و$E$
        و$qE$ في توازن الجزء II؟
  \item وهل تقدر كتلة فوق الرأس أن تمسك القطيرة مرفوعة بدل ذلك؟ احسب
        \omterm{def:g11:electric-gravitational-fields:field}{مجال} كرة رصاص كتلتها \qty{1000}{kg} على ارتفاع \qty{1.0}{m} فوق
        القطيرة، \omterm{def:g10:inertia:force}{والقوة} على القطيرة، وقارنها بوزنها.
  \item يحتاج التوازن جذبًا صاعدًا قويًّا بما يكفي لمصارعة
        الأرض كلها. فلماذا تقدر صفيحة مشحونة على توفيره ولا يقدر
        أي ترتيب للكتل؟
  \item و\emph{كتلة} القطيرة تتغير هي أيضًا بخطوات (\omterm{def:g11:fundamental-interactions:ladder}{فجزيء} زيت
        واحد نحو \qty{5e-25}{kg}). قارن القفزات النسبية في
        $m$ وفي $\abs{q}$ حين يهبط إلكترون واحد، واستنتج: لماذا
        يرى هذا التوازن ذرات الكهرباء ولا يرى \omterm{def:g11:fundamental-interactions:ladder}{جزيئات} الزيت؟ وبيت القصيد:
        الشحنة مكمَّاة، وذرّتها $e = \qty{1.60e-19}{C}$.
\end{enumerate}
\end{problem}
