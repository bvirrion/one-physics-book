خريطة الدارة --- التيار \omterm{def:g11:circuits-and-power:voltage}{والتوتر} \omterm{prop:g11:circuits-and-power:ohm}{وقانون أوم}. ويعيد هذا الفصل فتحها

\section{الشحنة في حركة}

\begin{definition}[التيار والكولوم]\label{def:g11:circuits-and-power:current}
\emph{التيار الكهربائي}\index{تيار كهربائي} تدفق منظَّم
للشحنة (وهي الإلكترونات في المعدن). وشدته $I$، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\emph{الأمبير}\index{أمبير} (\unit{A})، هي الشحنة العابرة
لمقطع من السلك في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}: فالتيار الثابت $I$ الجاري مدة
$t$ ينقل الشحنة
\[ q = I\,t , \]
\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \emph{الكولوم}\index{كولوم} (\unit{C})، حيث $\qty{1}{C} =
\qty{1}{A.s}$. والشحنة الأولية $e = \qty{1.6e-19}{C}$. ويقاس التيار
بمقياس تيار يُدرج على التسلسل.
\end{definition}

\begin{example}[الصاعقة في مقابل الهاتف]\label{ex:g11:circuits-and-power:coulomb}
تحمل ضربة برق نحو $\qty{3.0e4}{A}$ مدة نحو
$\qty{1.0e-4}{s}$: أي $q = 3.0\times10^{4} \times 10^{-4} = \qty{3.0}{C}$.
أما هاتفك، عند \qty{2.0}{A}، فينقل الشحنة نفسها كل $1.5$ \omterm{def:g10:orders-of-magnitude:unit}{ثانية}.
\end{example}

\begin{definition}[التوتر والفولط]\label{def:g11:circuits-and-power:voltage}
\emph{التوتر}\index{توتر} $U$ بين نقطتين من دارة، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\emph{الفولط}\index{فولط} (\unit{V})، هو \omterm{def:g10:energy-conservation:energy}{الطاقة} المتبادَلة لكل \omterm{def:g10:orders-of-magnitude:unit}{وحدة}
شحنة تنتقل بينهما: أي $\qty{1}{V} = \qty{1}{J/C}$، ومن ثم فإن شحنة
$q$ تعبر جهازًا تحت التوتر $U$ تتبادل معه \omterm{def:g10:energy-conservation:energy}{الطاقة} $E = qU$.
ويقيسه مقياس توتر يوصَّل على التفرع.
\end{definition}

\begin{proposition}[قوانين الدارة]\label{prop:g11:circuits-and-power:laws}
في دارة تشتغل في نظام دائم:
\begin{enumerate}
  \item \emph{قانون العقدة:} مجموع التيارات الواصلة إلى عقدة يساوي
        مجموع التيارات المغادرة لها؛
  \item \emph{قانون التسلسل:} التوترات على \omterm{def:g10:relative-motion:trajectory}{مسار} تتجمع:
        $U_{AC} = U_{AB} + U_{BC}$؛
  \item \emph{قانون التفرع:} فرعان يلتقيان عند النقطتين نفسيهما
        يحملان \omterm{def:g11:circuits-and-power:voltage}{التوتر} نفسه.
\end{enumerate}
\end{proposition}

\begin{proof}
في النظام الدائم لا تتراكم الشحنة في أي موضع، فما يدخل
عقدةً يخرج منها. \omterm{def:g10:energy-conservation:energy}{والطاقة} لكل \omterm{def:g11:circuits-and-power:current}{كولوم} تجميعية على \omterm{def:g10:relative-motion:trajectory}{مسار} وهي
لا تتوقف إلا على طرفيه --- لا على الفرع المسلوك بينهما.
\end{proof}

\section{قانون أوم وشبكات المقاومات}

\begin{definition}[المقاومة]\label{def:g11:circuits-and-power:resistance}
\emph{مقاومة}\index{مقاومة} ناقل هي النسبة
$R = U/I$ بين \omterm{def:g11:circuits-and-power:voltage}{التوتر} بين طرفيه والتيار المارّ فيه، وتقاس \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\emph{الأوم}\index{أوم} (\unit{\ohm}): أي $\qty{1}{\ohm} = \qty{1}{V/A}$.
\end{definition}

\begin{proposition}[قانون أوم]\label{prop:g11:circuits-and-power:ohm}
\index{قانون أوم}
في ناقل معدني محفوظ عند درجة حرارة ثابتة، تكون $R$ ثابتة:
أي أن \omterm{def:g11:circuits-and-power:voltage}{التوتر} والتيار متناسبان،
\[ U = R\,I . \]
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:circuits-and-power:ohm-limits}
هو قانون تجريبي لا قانون كوني: فالثنائي، أو شعيرة مصباح متوهّج،
أو جلدك أنت، لا تخضع له. أما لماذا تخضع له المعادن فيُشتقّ في
المجلدات الجامعية.
\end{remark}

\begin{proposition}[المقاومات على التسلسل وعلى التفرع]\label{prop:g11:circuits-and-power:networks}
\index{مقاومة مكافئة}
مقاومتان على التسلسل تكافئان \omterm{def:g11:circuits-and-power:resistance}{مقاومة} واحدة $R_{\text{s}} =
R_1 + R_2$؛ ومقاومتان على التفرع تكافئان \omterm{def:g11:circuits-and-power:resistance}{مقاومة} واحدة $R_{\text{p}}$ تُعطى بالعلاقة
\[
\frac{1}{R_{\text{p}}} = \frac{1}{R_1} + \frac{1}{R_2} ,
\qquad\text{أي}\quad
R_{\text{p}} = \frac{R_1 R_2}{R_1 + R_2} < \min(R_1, R_2) .
\]
\end{proposition}

\begin{proof}
على التسلسل: يعبر التيار $I$ نفسه كلتيهما وتتجمع التوترات
(\cref{prop:g11:circuits-and-power:laws})، ومنه $U = (R_1 + R_2)\,I$.
وعلى التفرع: تقع كلتاهما تحت \omterm{def:g11:circuits-and-power:voltage}{التوتر} $U$ نفسه وتتجمع التيارات، ومنه
$I = \frac{U}{R_1} + \frac{U}{R_2}$.
\end{proof}

\begin{example}[المقاومة المكافئة]\label{ex:g11:circuits-and-power:networks}
$R_1 = \qty{100}{\ohm}$ و$R_2 = \qty{150}{\ohm}$: على التسلسل،
\qty{250}{\ohm}؛ وعلى التفرع، $\frac{100 \times 150}{250} =
\qty{60}{\ohm}$ --- أي أقل من كلتيهما: فالتيار يجد طريقًا ثانيًا.
\end{example}

\section{الطاقة والاستطاعة في دارة}

\begin{proposition}[الاستطاعة الكهربائية]\label{prop:g11:circuits-and-power:power}
\index{استطاعة كهربائية}
الجهاز الذي يقع تحت \omterm{def:g11:circuits-and-power:voltage}{التوتر} $U$ ويعبره تيار $I$ يتلقى \omterm{def:g10:energy-conservation:energy}{الطاقة} بالمعدل
\[ P = U\,I , \qquad\text{أي}\quad E = U\,I\,t \ \text{خلال زمن } t . \]
\end{proposition}

\begin{proof}
خلال زمن $t$ تعبر الشحنة $q = It$ الجهازَ
(\cref{def:g11:circuits-and-power:current}) ويسلّم كل \omterm{def:g11:circuits-and-power:current}{كولوم}
$U$ جولًا (\cref{def:g11:circuits-and-power:voltage}): ومنه $E = qU = UIt$.
\end{proof}

\begin{proposition}[الاستطاعة في مقاومة]\label{prop:g11:circuits-and-power:ri2}
\omterm{def:g11:circuits-and-power:resistance}{المقاومة} $R$ التي يجري فيها تيار $I$ تبدّد
\[ P = R\,I^2 = \frac{U^2}{R} . \]
\end{proposition}

\begin{proof}
عوّض \omterm{prop:g11:circuits-and-power:ohm}{قانون أوم} في $P = UI$: فتحصل على $P = (RI)I = RI^2 = U(U/R) = U^2/R$.
\end{proof}

\begin{example}[قراءة لوحة بيانات]\label{ex:g11:circuits-and-power:nameplate}
غلاية مكتوب عليها ``\qty{230}{V} -- \qty{2200}{W}'' تسحب $I = P/U =
2200/230 = \qty{9.6}{A}$ ومقاومتها $R = U^2/P = 230^2/2200 =
\qty{24}{\ohm}$. وإذا اشتغلت \qty{150}{s} استعملت $E = 2200 \times 150 =
\qty{3.3e5}{J} \approx \qty{0.09}{kW.h}$ --- أي نحو سنتين.
\end{example}

\begin{remark}[الجول والكيلوواط ساعة]\label{rem:g11:circuits-and-power:kwh}
يعدّ العدّاد \omterm{def:g10:energy-conservation:kwh}{الكيلوواط ساعة} (\cref{def:g10:energy-conservation:kwh}):
$\qty{1}{kW.h} = \qty{3.6e6}{J}$. \omterm{def:g10:energy-conservation:energy}{فالجول} \omterm{def:g10:orders-of-magnitude:unit}{وحدة} الفيزيائي؛
أما \omterm{def:g10:energy-conservation:kwh}{الكيلوواط ساعة} فيسعّر الفواتير.
\end{remark}

\section{مفعول جول: من المحمصة إلى برج النقل}

\begin{definition}[مفعول جول]\label{def:g11:circuits-and-power:joule}
التبديد $P = RI^2$ الذي يحوّل \omterm{def:g10:energy-conservation:forms}{الطاقة الكهربائية} حرارةً في كل
\omterm{def:g11:circuits-and-power:resistance}{مقاومة} هو \emph{مفعول جول}\index{مفعول جول}. فالمدافئ
والغلايات والمحمصات والمصابيح القديمة مقاومات بالتصميم؛ و
\emph{المِصهر}\index{مصهر} سلك رفيع معايَر لينصهر --- فيفتح
الدارة --- حين يتجاوز التيار حدّه.
\end{definition}

\begin{example}[لماذا تشتغل أبراج النقل على توتر عالٍ]\label{ex:g11:circuits-and-power:transmission}
تحتاج قرية $P = \qty{100}{kW}$ عبر خط مقاومته
$R_\ell = \qty{5.0}{\ohm}$. فإذا وُصّلت عند $U = \qty{500}{V}$: $I = P/U =
\qty{200}{A}$ ويبدّد الخط $R_\ell I^2 = \qty{200}{kW}$ --- أي ضعف
ما يُوصَّل: وهذا سخف. وعند $U = \qty{20}{kV}$: $I = \qty{5.0}{A}$ و
$R_\ell I^2 = \qty{125}{W}$. أي تيار أقلّ أربعين مرة، وضياع أقلّ $1600$
مرة.
\end{example}

\begin{remark}[قاعدة الشبكة الحديدية]\label{rem:g11:circuits-and-power:grid}
عند \omterm{def:g10:energy-conservation:power}{استطاعة} موصَّلة ثابتة $P = UI$، يهبط ضياع الخط $R_\ell I^2 = R_\ell
P^2/U^2$ مثل \emph{مربع} \omterm{def:g11:circuits-and-power:voltage}{توتر} النقل. ومن هنا:
حوّل إلى أعلى (\qty{400}{kV}) للسفر، وإلى أسفل (\qty{230}{V}) للعيش معه.
\end{remark}

\begin{remark}[سلامة المنزل]\label{rem:g11:circuits-and-power:safety}
خط قدره \qty{230}{V} محميّ \omterm{def:g11:circuits-and-power:joule}{بمصهر} \qty{16}{A} يغذّي على الأكثر $230 \times 16
\approx \qty{3.7}{kW}$: فغلايتان معًا تصهران \omterm{def:g11:circuits-and-power:joule}{المصهر} --- وهو بالتصميم
الحلقة الأرفع والأضعف، وهو أرخص بكثير من حريق داخل الجدار.
\end{remark}

\section{البطاريات الحقيقية: القوة المحركة الكهربائية}

\begin{definition}[القوة المحركة والمقاومة الداخلية]\label{def:g11:circuits-and-power:emf}
تحوّل البطارية \omterm{def:g10:energy-conservation:forms}{الطاقة الكيميائية} إلى كهربائية. و
\emph{قوتها المحركة الكهربائية}\index{قوة محركة كهربائية}
$\mathcal{E}$، \omterm{def:g11:circuits-and-power:voltage}{بالفولط}، هي \omterm{def:g10:energy-conservation:energy}{الطاقة} التي تسلّمها لكل \omterm{def:g11:circuits-and-power:current}{كولوم} يعبرها؛
أما \emph{مقاومتها الداخلية}\index{مقاومة داخلية} $r$ فتفسّر
مفعول جول داخل كيميائها. والنموذج: منبع مثالي
$\mathcal{E}$ على التسلسل مع $r$ (والحرف المزخرف $\mathcal{E}$: لأن $E$ \omterm{def:g10:energy-conservation:energy}{للطاقة}).
\end{definition}

\begin{proposition}[التوتر بين المربطين]\label{prop:g11:circuits-and-power:terminal}
البطارية التي توصّل تيارًا $I$ تُظهر بين مربطيها
\[ U = \mathcal{E} - r\,I . \]
\end{proposition}

\begin{proof}
يوفّر الكيمياء $\mathcal{E}I$، وتأكل \omterm{def:g11:circuits-and-power:emf}{المقاومة الداخلية}
$rI^2$، وتتلقى الدارة $UI$: فيعطي \omterm{thm:g10:energy-conservation:conservation}{انحفاظ الطاقة}
(\cref{ch:g10:energy-conservation}) العلاقة $UI = \mathcal{E}I - rI^2$.
\end{proof}

\begin{example}[بطارية تحت حمل]\label{ex:g11:circuits-and-power:discharge}
بطارية ($\mathcal{E} = \qty{9.0}{V}$، $r = \qty{1.0}{\ohm}$) تغذّي
مصباحًا مقاومته $R = \qty{8.0}{\ohm}$. وتخضع الحلقة للعلاقة $\mathcal{E} =
rI + RI$، ومنه $I = \mathcal{E}/(R + r) = \qty{1.0}{A}$ و$U = 9.0 -
1.0 \times 1.0 = \qty{8.0}{V}$: فيتلقى المصباح \qty{8.0}{W} بينما
تبدّد البطارية \qty{1.0}{W} في تسخين نفسها.
\end{example}

\begin{remark}[المستقبِلات]\label{rem:g11:circuits-and-power:receivers}
المحرك أو البطارية أثناء الشحن يجريان التحويل معكوسًا: فيمتصّان
تحت $U = \mathcal{E}' + r'I$، ويحوّلان $\mathcal{E}'I$ إلى حركة أو
كيمياء ويضيّعان $r'I^2$. فالمنابع تدفع، والمستقبِلات تدفع مقابلةً.
\end{remark}

\begin{example}[مردود بطارية]\label{ex:g11:circuits-and-power:efficiency}
\omterm{def:g10:energy-conservation:efficiency}{مردود} (\cref{def:g10:energy-conservation:efficiency}) التفريغ
أعلاه: أي $UI = \qty{8.0}{W}$ مفيدة من $\mathcal{E}I =
\qty{9.0}{W}$، ومنه $\eta = U/\mathcal{E} = 8/9 \approx 0.89$ --- وهو يهبط
كلما تسلّق التيار: فالبطاريات الكادحة تشتغل ساخنة.
\end{example}

\section{مسألة: من السدّ إلى المحمصة}

\begin{problem}[{مسألة نهاية الأسبوع --- رحلة الألف كيلومتر لكيلوواط
ساعة، من الماء الساقط إلى الخبز المحمّر، مع فاتورة الرحلة
كاملة}]
\label{pb:g11:circuits-and-power:1}
سدّ جبلي يحبس الماء على ارتفاع \qty{200}{m} فوق عنفاته، وهي تبتلع
\qty{50}{m^3} --- أي \qty{5.0e4}{kg} --- في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}؛ وتحوّل العنفة والمولّد التناوبي
\omterm{def:g10:energy-conservation:efficiency}{بمردود} $0.90$. وعبر خط مقاومته الكلية
$R_\ell = \qty{32}{\ohm}$ (\qty{1000}{km} ذهابًا وإيابًا) تغذّي المحطة
مدينةً تنتظر فيها محمصتك \qty{900}{W} عند \qty{230}{V}. ويعطي
المولّد \qty{20}{kV}؛ وتحوّل المحوّلات (\omterm{def:g10:energy-conservation:efficiency}{ومردود} كلٍّ منها $0.99$)
التوترات كما نشاء. وتُباع الكهرباء بسعر $0.25$ يورو لكل
\unit{kW.h}؛ و$g = \qty{9.81}{N/kg}$.

\medskip
\noindent\textbf{الجزء I --- عند السدّ.}
\begin{enumerate}
  \item كم \omterm{def:g10:energy-conservation:energy}{طاقة} كامنة يحرّرها الماء الساقط في كل
        \omterm{def:g10:orders-of-magnitude:unit}{ثانية}؟ استنتج \omterm{def:g10:energy-conservation:power}{الاستطاعة} المتاحة للعنفات.
  \item وأيّ \omterm{prop:g11:circuits-and-power:power}{استطاعة كهربائية} تغادر المولّد؟
  \item وعند \qty{20}{kV} مباشرة من المولّد، أيّ تيار يجري؟
  \item احسب ضياع جول $R_\ell I^2$ عند ذلك التيار وقارنه
        بخرج المحطة. واستنتج.
  \item \omterm{def:g10:energy-conservation:power}{استطاعة} المدينة $P = UI$ يحدّدها الطلب. فما السبيل الوحيد
        إلى تقليص $I$، وأيّ جهاز يؤديه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- على الخط.}
\begin{enumerate}[resume]
  \item محوّلة، تُعامَل في هذا السؤال بلا ضياع، ترفع
        \qty{88}{MW} إلى \qty{400}{kV}: فأيّ تيار يجري الآن؟
  \item احسب ضياع الخط الجديد، بالميغاواط.
  \item وأيّ جزء من خرج المحطة يضيع في الخط؟
  \item بيّن أن الجزء الضائع يساوي $R_\ell P/U^2$، وتحقّق من أنه
        يفسّر العامل بين السؤالين 4 و7.
  \item احسب هبوط \omterm{def:g11:circuits-and-power:voltage}{التوتر} $R_\ell I$ على طول الخط عند
        \qty{400}{kV}. أيتوافق مع السؤال 8؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- في المطبخ.}
\begin{enumerate}[resume]
  \item حوّل \qty{1}{kW.h} إلى \omterm{def:g10:energy-conservation:energy}{الجول}؛ فماذا يضرب العدّاد؟
  \item احسب تيار المحمصة ومقاومتها عند \qty{230}{V}.
  \item ثلاث دقائق من التحميص: \omterm{def:g10:energy-conservation:energy}{الطاقة} \omterm{def:g10:energy-conservation:energy}{بالجول} و
        \omterm{def:g10:energy-conservation:kwh}{بالكيلوواط ساعة}، وثمنها.
  \item تبدّد المحمصة $RI^2$: اشرح في جملة واحدة لماذا تكون
        الصيغة نفسها ضياعًا على الخط ومنتَجًا في المطبخ.
  \item تتقاسم المحمصة \omterm{def:g11:circuits-and-power:joule}{مصهر} \qty{16}{A} مع غلاية \qty{2.2}{kW}:
        فهل يصمد \omterm{def:g11:circuits-and-power:joule}{المصهر}؟ ثم تنضمّ مدفأة \qty{2.0}{kW}: فماذا الآن؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- التدقيق.}
\begin{enumerate}[resume]
  \item تحمل أسلاك البيت نفسها، ومقاومتها \qty{0.05}{\ohm}،
        التيارَ \qty{13.5}{A} في السؤال 15: احسب الضياع وجزأه
        من نحو \qty{3.1}{kW} الموصَّلة.
  \item سلسِل المردودات (العنفة-المولّد، والرفع، والخط،
        والخفض، وأسلاك البيت) في \omterm{def:g10:energy-conservation:efficiency}{مردود} شبكة كلي واحد.
  \item وكم كيلوغرامًا (وكم مترًا مكعبًا) من ماء السدّ يجب أن يسقط ليبلغ
        \emph{\omterm{def:g10:energy-conservation:kwh}{كيلوواط ساعة} واحد} محمصتك؟
  \item احسب الجزء الضائع $R_\ell P/U^2$ في السؤال 9 عند
        $U = \qty{20}{kV}$. وماذا تعني قيمة تفوق $1$؟
  \item بيت القصيد --- \emph{ثمن المسافة}: في جملتين،
        كم يكلّف \omterm{def:g10:energy-conservation:kwh}{كيلوواط ساعة} محمَّص الجبلَ، وأيّ جزء
        منه يموت في الطريق، وأيّ حيلة واحدة تجعل ذلك ممكنًا؟
\end{enumerate}
\end{problem}
