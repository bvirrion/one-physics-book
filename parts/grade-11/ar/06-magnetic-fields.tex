\chapter{المغناطيسية والمجالات المغناطيسية}\label{ch:g11:magnetic-fields}

اهتدى البحّارة بإبر ممغنطة ألف سنة قبل أن
يعرف أحد لماذا تشير شمالًا. وجاء الجواب سنة 1820، حين
جعل تيارٌ بوصلةً ترتجف: فالمغناطيسية تصنعها الشحن المتحركة.
ويرسم هذا الفصل خريطة \omterm{def:g11:magnetic-fields:bfield}{المجال المغناطيسي} --- للمغانط، وللأرض، و
للتيارات --- ثم يغلق الحلقة: \omterm{def:g11:electric-gravitational-fields:field}{فالمجال} يدفع التيارات بدوره،
وبذلك يدور كل محرك كهربائي.

\section{المغانط والمجال المغناطيسي}

\begin{definition}[المغناطيس والقطبان]\label{def:g11:magnetic-fields:magnet}
\emph{المغناطيس}\index{مغناطيس} يجذب الحديد، وإذا عُلّق بخيط
استدار ليواجه الشمال. ويتركّز تأثيره في \emph{قطبين
مغناطيسيين}\index{قطب مغناطيسي}، \emph{شمالي} (وهو الطرف الذي يطلب
الشمال الجغرافي) و\emph{جنوبي}: فالقطبان المتماثلان يتنافران، والمختلفان يتجاذبان.
ونشر مغناطيس نصفين يعطي مغناطيسين كاملين --- ولم تعزل
تجربة قط قطبًا واحدًا.
\end{definition}

\begin{definition}[المجال المغناطيسي]\label{def:g11:magnetic-fields:bfield}
يعدّل \omterm{def:g11:magnetic-fields:magnet}{المغناطيس} الفضاء حوله
(\cref{ch:g11:electric-gravitational-fields}): فينشئ في كل مكان
\emph{مجالًا مغناطيسيًّا}\index{مجال مغناطيسي} $\vect B$. وتقرأ البوصلة
جهته (فتصطفّ الإبرة مع $\vect B$، وطرفها الشمالي إلى الأمام)؛ ويقيس
\emph{مسبار هول}\index{مسبار هول} شدته، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\emph{التسلا}\index{تسلا} (\unit{T}).
\end{definition}

\begin{definition}[خطوط المجال المغناطيسي]\label{def:g11:magnetic-fields:lines}
\emph{خطوط المجال المغناطيسي}\index{خط مجال مغناطيسي} منحنيات
مماسّة للمتجهة $\vect B$، تتزاحم حيث يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال} قويًّا. وخارج
\omterm{def:g11:magnetic-fields:magnet}{المغناطيس} تسير من القطب الشمالي إلى الجنوبي؛ وخلافًا لخطوط \omterm{def:g11:electric-gravitational-fields:efield}{المجال الكهربائي}
ليست لها نهايات: فكلٌّ ينغلق عبر جسم \omterm{def:g11:magnetic-fields:magnet}{المغناطيس}.
\end{definition}

\section{الأرض مغناطيس}

\begin{definition}[مجال الأرض المغناطيسي]\label{def:g11:magnetic-fields:earth}
تجعل التيارات في \omterm{def:g11:fundamental-interactions:ladder}{نواة} الأرض الحديدية المنصهرة الكوكبَ مغناطيسًا
عملاقًا: فعند السطح $B \approx \qty{50}{\micro T}$، وتصطفّ إبرة حرة
مع المركّبة الأفقية \omterm{def:g11:electric-gravitational-fields:field}{للمجال} --- وتلك هي البوصلة. والقطبان
المغناطيسيان الأرضيان ليسا الجغرافيين؛ والزاوية بين
شمال البوصلة والشمال الحقيقي هي
\emph{الانحراف المغناطيسي}\index{انحراف مغناطيسي}، ويُصحَّح في كل اتجاه.
\end{definition}

\begin{example}[تصحيح اتجاه]\label{ex:g11:magnetic-fields:bearing}
حيث يكون الانحراف $\ang{7}$ غربًا، فإن سفينة تتجه ``شمال
البوصلة'' مسافة \qty{5.0}{km} تنجرف $5000 \times \sin\ang{7} \approx
\qty{6.1e2}{m}$ غرب مسارها: فلتبحر شمالًا حقيقيًّا، اتجه $\ang{7}$
\emph{شرق} الإبرة.
\end{example}

\section{المجال المغناطيسي لتيار}

\begin{remark}[أورستد، 1820]\label{rem:g11:magnetic-fields:oersted}
خلال محاضرة سنة 1820، رأى هانز كريستيان أورستد إبرة بوصلة
تنزاح حين أغلق دارة قريبة: فالكهرباء والمغناطيسية
موضوع واحد --- \emph{فالتيارات تنشئ مجالات مغناطيسية}.
\end{remark}

\begin{proposition}[مجال سلك مستقيم]\label{prop:g11:magnetic-fields:wire}
السلك المستقيم الطويل الذي يحمل تيارًا $I$ ينشئ مجالًا
خطوطه دوائر مركزها السلك، في مستويات عمودية عليه.
وعلى المسافة $d$ من السلك،
\[
B = \frac{\mu_0\, I}{2\pi d},
\qquad \mu_0 = 4\pi\times 10^{-7}\ \unit{T.m/A} \approx \qty{1.26e-6}{T.m/A}.
\]
\end{proposition}
\admitted

\begin{example}[السلك مغناطيس ضعيف]\label{ex:g11:magnetic-fields:wire}
على بعد $d = \qty{1.0}{cm}$ من سلك يحمل $I = \qty{15}{A}$:
$B = \num{2e-7} \times 15 / 0.010 = \qty{0.30}{mT}$ --- أي ستة أضعاف \omterm{def:g11:electric-gravitational-fields:field}{مجال}
الأرض، من تيار يقطع قاطعًا منزليًّا. فالمجالات القوية
تحتاج هندسة أفضل.
\end{example}

\begin{definition}[الوشيعة والملف اللولبي]\label{def:g11:magnetic-fields:solenoid}
لفّ السلك يركّز مجاله. و\emph{الوشيعة
المسطحة}\index{وشيعة} ($N$ لفّة في قرص) تسلك سلوك \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} رقيق،
أحد وجهيها شمالي والآخر جنوبي؛ أما \emph{الملف اللولبي}\index{ملف لولبي} ---
وهو لفّ أسطواني طويل، فيه $n$ لفّة في \omterm{def:g10:orders-of-magnitude:unit}{المتر} --- فيسلك سلوك \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} قضيبي.
\end{definition}

\begin{proposition}[المجال داخل ملف لولبي]\label{prop:g11:magnetic-fields:solenoidfield}
داخل \omterm{def:g11:magnetic-fields:solenoid}{ملف لولبي} طويل، بعيدًا عن الطرفين، يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال}
\emph{\omterm{def:g11:electric-gravitational-fields:capacitor}{منتظمًا}}: أي موازيًا للمحور ونفسه في كل نقطة
داخلية، وشدته $B = \mu_0\, n\, I$، مستقلة عن عرض
التجويف. وخارجه يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال} ضعيفًا.
\end{proposition}
\admitted

\begin{remark}
تُشتقّ الصيغتان كلتاهما اشتقاقًا أمينًا في مجلد السنة الأولى، من القانون
الذي يربط دوران \omterm{def:g11:electric-gravitational-fields:field}{مجال} بالتيار الذي يطوّقه. ولاحظ ما
يهمّ: $n$، أي اللفّات \emph{في \omterm{def:g10:orders-of-magnitude:unit}{المتر}} --- فلُفّ أكثف لا أطول.
\end{remark}

\begin{method}[اليد اليمنى مرتين]\label{met:g11:magnetic-fields:righthand}
\begin{itemize}
  \item \emph{السلك}: الإبهام = التيار؛ والأصابع الملتفّة = دوران الخطوط.
  \item \emph{\omterm{def:g11:magnetic-fields:solenoid}{الوشيعة} \omterm{def:g11:magnetic-fields:solenoid}{والملف اللولبي}}: الأصابع الملتفّة = التيار في اللفّات؛
        والإبهام = $\vect B$ في الداخل، مشيرًا إلى الوجه الشمالي.
\end{itemize}
\end{method}

\begin{example}[أرقام ملف لولبي]\label{ex:g11:magnetic-fields:solenoid}
عند $10$ لفّة في السنتيمتر ($n = \qty{1000}{m^{-1}}$) و
$I = \qty{5.0}{A}$: $B = \num{1.26e-6} \times 1000 \times 5.0 \approx
\qty{6.3}{mT}$ --- أي مئة ضعف \omterm{def:g11:electric-gravitational-fields:field}{مجال} الأرض، قابلة للضبط بمقبض.
\end{example}

\section{المغانط الكهربائية}

\begin{definition}[المغناطيس الكهربائي]\label{def:g11:magnetic-fields:electromagnet}
\emph{المغناطيس الكهربائي}\index{مغناطيس كهربائي} \omterm{def:g11:magnetic-fields:solenoid}{ملف لولبي} ملفوف على
قلب من حديد ليّن: فيتمغنط الحديد في \omterm{def:g11:electric-gravitational-fields:field}{مجال} \omterm{def:g11:magnetic-fields:solenoid}{الوشيعة} ويضاعفه،
عادةً بعامل من $100$ إلى $1000$. وخلافًا \omterm{def:g11:magnetic-fields:magnet}{للمغناطيس} الدائم،
ينطفئ مع انقطاع التيار.
\end{definition}

\begin{remark}[ثلاث آلات]\label{rem:g11:magnetic-fields:machines}
ترفع رافعة الخردة سيارة \omterm{def:g11:magnetic-fields:electromagnet}{بمغناطيس كهربائي}، ثم --- وهذا هو
المقصود كله --- \emph{تُسقطها} بفتح مفتاح. ويشدّ \omterm{def:g11:magnetic-fields:electromagnet}{مغناطيس كهربائي} صغير في
مرحّل تماسًّا فيغلقه: فيبدّل تيار ضعيف تيارًا
قويًّا (\cref{ch:g11:circuits-and-power}). ويمسك جهاز الرنين المغناطيسي مجالًا \omterm{def:g11:electric-gravitational-fields:capacitor}{منتظمًا} قدره
\qty{1.5}{T} حول المريض \omterm{def:g11:magnetic-fields:solenoid}{بملف لولبي} فائق الناقلية.
\end{remark}

\section{قوة لابلاس}

\begin{proposition}[قوة لابلاس]\label{prop:g11:magnetic-fields:laplace}
السلك المستقيم الذي طوله $L$، والذي يحمل تيارًا $I$ في \omterm{def:g11:electric-gravitational-fields:capacitor}{مجال منتظم}
$\vect B$ \emph{عمودي} على السلك، يحسّ
\emph{قوة لابلاس}\index{قوة لابلاس} وشدتها $F = I\,L\,B$،
عمودية على السلك وعلى $\vect B$ معًا: اليد اليمنى مبسوطة، والإبهام في جهة
التيار، والأصابع الممدودة في جهة $\vect B$ --- فتدفع الراحة في جهة
$\vect F$. أما السلك \emph{الموازي} \omterm{def:g11:electric-gravitational-fields:field}{للمجال} فلا يحسّ \omterm{def:g10:inertia:force}{قوة}.
\end{proposition}
\admitted

\begin{remark}[السكة القافزة]\label{rem:g11:magnetic-fields:rail}
يشتقّ مجلد السنة الأولى ذلك من \omterm{def:g10:inertia:force}{القوة} المغناطيسية على كل شحنة
متحركة. وفي المختبر: يستقرّ قضيب نحاسي عبر سكتين أفقيتين
بين قطبَي \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} (\omterm{def:g11:electric-gravitational-fields:field}{والمجال} داخل إلى الصفحة في
الشكل)؛ أغلِق المفتاح فينطلق القضيب على السكتين. واعكس
التيار، أو اقلب \omterm{def:g11:magnetic-fields:magnet}{المغناطيس}: فينطلق في الجهة الأخرى.
\end{remark}

\begin{example}[أرقام لابلاس]\label{ex:g11:magnetic-fields:laplace}
قضيب طوله $L = \qty{5.0}{cm}$ يحمل $I = \qty{8.0}{A}$ عبر
$B = \qty{0.50}{T}$: $F = 8.0 \times 0.050 \times 0.50 = \qty{0.20}{N}$
--- أي \omterm{def:g10:universal-gravitation:weight}{وزن} \qty{20}{g}، على سلك يزن بضعة غرامات.
\end{example}

\begin{remark}[محرك التيار المستمر]\label{rem:g11:magnetic-fields:motor}
اثنِ السلك حلقةً في \omterm{def:g11:electric-gravitational-fields:field}{المجال}: فيحمل الضلعان العموديان على
$\vect B$ تيارين متعاكسين، ومن ثم تكون قوتا لابلاس عليهما متعاكستين
--- أي مزدوجة تدير الحلقة. وبعد نصف دورة كانت المزدوجة ستديرها
عكسًا، فيعكس مفتاح دوّار (هو \emph{المبدّل}\index{مبدّل})
التيار كل نصف دورة. فكل مروحة ومثقاب هي هذه الحلقة، مضاعَفة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:1}
يُنشر \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} قضيبي نصفين بين قطبيه. فأيّ قطبين تحمل كل
قطعة؟ وإذا أُعيدتا معًا، أتتجاذب الوجوه المقطوعة حديثًا
أم تتنافر؟ وهل تقدر أي طريقة في القطع على عزل القطب الشمالي؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:magnetic-fields:2}
صحيح أم خطأ، مع سبب واحد لكلٍّ: (أ) يتقاطع خطا \omterm{def:g11:magnetic-fields:bfield}{مجال مغناطيسي}
حيث يكون \omterm{def:g11:electric-gravitational-fields:field}{المجال} قويًّا؛ (ب) خارج \omterm{def:g11:magnetic-fields:magnet}{المغناطيس} تسير \omterm{def:g11:electric-gravitational-fields:lines}{خطوط المجال}
من القطب الشمالي إلى الجنوبي؛ (ج) كل \omterm{def:g11:magnetic-fields:lines}{خط مجال مغناطيسي}
حلقة منغلقة؛ (د) تستقرّ الإبرة عمودية على الخط المارّ بها.
\end{exercise}

عموديان على \omterm{def:g11:electric-gravitational-fields:field}{مجال} قدره $B = \qty{0.30}{T}$. احسب \omterm{def:g10:inertia:force}{القوة} على كل

\section{مسألة: البوصلة والرافعة وجهاز الرنين}

\begin{problem}[{مسألة نهاية الأسبوع --- ثلاثة مغانط في العمل: بوصلة
تعبر محيطًا، ومغناطيس كهربائي يرفع سيارة، وملف لولبي في جهاز رنين
--- مجال واحد يمتد على عشر قوى من قوى العشرة، يؤدي ثلاث وظائف}]
\label{pb:g11:magnetic-fields:1}
المتجهة $\vect B$ نفسها توجّه قاربًا شراعيًّا بمقدار \qty{50}{\micro T}،
وترفع خردة حديد بمقدار \qty{1}{T}، وتصوّر ركبة بمقدار \qty{1.5}{T}
--- أي ثلاث آلات لصيغ هذا الفصل الثلاث.

\medskip
\noindent\textbf{الجزء I --- البوصلة.}
عند موضع السفينة، \omterm{def:g11:electric-gravitational-fields:field}{لمجال} الأرض مركّبة أفقية
\qty{20}{\micro T} ومركّبة عمودية \qty{44}{\micro T}؛
والانحراف $\ang{6}$ غربًا.
\begin{enumerate}
  \item لماذا تشير إبرة البوصلة إلى الشمال (المغناطيسي) أصلًا؟
  \item احسب شدة \omterm{def:g11:electric-gravitational-fields:field}{المجال} الكلي وزاويته مع الأفقي.
  \item وبالاتجاه ``شمال البوصلة'' مسافة \qty{30}{km}، كم تنتهي السفينة
        غرب \omterm{def:g10:relative-motion:trajectory}{مسار} الشمال الحقيقي؟
  \item وتزيح حمولة فولاذية الإبرة $\ang{4}$ أخرى غربًا: السؤال نفسه.
  \item وقرب \omterm{def:g11:magnetic-fields:magnet}{القطب المغناطيسي} الأرضي تكون البوصلة عديمة الفائدة: فأيّ مركّبة
        الملومة، ولماذا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الرافعة.}
\omterm{def:g11:magnetic-fields:magnet}{مغناطيس} مخزن الخردة الكهربائي \omterm{def:g11:magnetic-fields:solenoid}{ملف لولبي} فيه $600$ لفّة ملفوفة على
\qty{30}{cm}، ومقاومته \qty{1.5}{\ohm}، ويغذّيه منبع \qty{12}{V}؛
ويضاعف قلبه الحديدي \omterm{def:g11:electric-gravitational-fields:field}{المجال} العاري $50$ مرة.
\begin{enumerate}[resume]
  \item احسب $n$، أي كثافة اللفّات.
  \item احسب التيار في \omterm{def:g11:magnetic-fields:solenoid}{الوشيعة} (\cref{ch:g11:circuits-and-power}).
  \item احسب \omterm{def:g11:electric-gravitational-fields:field}{مجال} \omterm{def:g11:magnetic-fields:solenoid}{الوشيعة} العارية $\mu_0 n I$.
  \item احسب \omterm{def:g11:electric-gravitational-fields:field}{المجال} مع القلب؛ وقارنه \omterm{def:g11:electric-gravitational-fields:field}{بمجال} النيوديميوم \qty{0.5}{T}.
  \item احسب \omterm{def:g10:energy-conservation:power}{الاستطاعة} المبدَّدة في \omterm{def:g11:magnetic-fields:solenoid}{الوشيعة}، ثم \omterm{def:g10:energy-conservation:energy}{الطاقة} في
        وردية رفع مدتها \qty{10}{\minute}.
  \item ويوجد \omterm{def:g11:magnetic-fields:magnet}{مغناطيس} دائم بهذه \omterm{def:g10:inertia:force}{القوة}. فلماذا \omterm{def:g11:magnetic-fields:electromagnet}{مغناطيس كهربائي} إذن؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- محرك الملفاف.}
ملفاف الرافعة محرك تيار مستمر: حلقة مستطيلة فيها $N = 100$ لفّة،
وضلعاها اللذان طول كلٍّ منهما \qty{4.0}{cm} عموديان على $B = \qty{0.25}{T}$،
وتحمل $I = \qty{3.0}{A}$.
\begin{enumerate}[resume]
  \item احسب \omterm{prop:g11:magnetic-fields:laplace}{قوة لابلاس} على كل حزمة ضلع عمودية على $\vect B$.
  \item القوتان متساويتان متعاكستان، ومع ذلك لا يتلاشى أثرهما.
        فماذا تنتجان، ولماذا يجب أن تكونا \emph{متعاكستين}؟
  \item وأيّ \omterm{def:g10:inertia:force}{قوة} تؤثر في الضلعين الآخرين حين يكونان موازيين للمتجهة $\vect B$؟
  \item وبعد نصف دورة، كانت القوتان نفساهما ستنقضان الدوران.
        فماذا يفعل \omterm{rem:g11:magnetic-fields:motor}{المبدّل}، ومتى؟
  \item أعطِ ثلاثة تغييرات تصميمية منفصلة يضاعف كلٌّ منها مزدوجة القوتين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- جهاز الرنين.}
يمسك \omterm{def:g11:magnetic-fields:solenoid}{الملف اللولبي} في الجهاز مجالًا \omterm{def:g11:electric-gravitational-fields:capacitor}{منتظمًا} قدره \qty{1.5}{T} بتيار
$I = \qty{500}{A}$.
\begin{enumerate}[resume]
  \item كم ضعفًا \omterm{def:g11:electric-gravitational-fields:field}{لمجال} الأرض يبلغ \qty{1.5}{T}؟
  \item احسب كثافة اللفّات $n$ اللازمة \emph{دون} أي قلب حديدي.
  \item ولو كانت \omterm{def:g11:circuits-and-power:resistance}{مقاومة} اللفّ \qty{0.20}{\ohm}، فأيّ \omterm{def:g10:energy-conservation:power}{استطاعة}
        كان سيبدّدها؟ وأيّ خاصية في اللفّ الحقيقي تتجنب
        ذلك، بثمن برودة قصوى؟
  \item بيت القصيد: ضع مجالات السفينة والرافعة وجهاز الرنين
        على سُلّم قوى العشرة من الفضاء بين النجوم
        (\qty{e-10}{T}) إلى نجم نيوتروني (\qty{e8}{T})؛ فما الذي تغيّر
        بين الآلات --- الفيزياء أم الأمبيرات؟
\end{enumerate}
\end{problem}
