\chapter{اللون ومنابع الضوء}\label{ch:g11:color-light-sources}

اسأل فيزيائيًّا عن لون طماطمة تحصل على سؤال مضادّ:
تحت أي ضوء؟ فالطماطمة لا تنثر إلا ما يُعطى لها، والمصباح
لا يعطي إلا ما يحويه طيفه، والعين تضغط كل ما يصل
إلى ثلاثة أعداد. فاللون مفاوضة بين المنبع
والجسم والعين --- ويستجوب هذا الفصل الأطراف الثلاثة واحدًا واحدًا.

\section{الرؤية بثلاثة أعداد}

\begin{definition}[المخاريط والرؤية ثلاثية اللون]\label{def:g11:color-light-sources:cones}
تحمل \omterm{def:g11:lenses-and-eye:eye}{الشبكية} (\cref{ch:g11:lenses-and-eye}) أسرتين من الخلايا
الحسّاسة للضوء: \emph{العصيّ}\index{عصيّة}، وهي تستجيب
للضوء الخافت وتتجاهل \omterm{def:g10:light-spectra:wavelength}{طول الموجة} (رؤية ليلية بالرمادي)، و
\emph{المخاريط}\index{مخروط}، في ثلاثة أنواع أشدّ حساسية قرب
$\qty{420}{nm}$ (أزرق) و$\qty{530}{nm}$ (أخضر) و$\qty{560}{nm}$
(أحمر). ورؤية الإنسان للألوان
\emph{ثلاثية اللون}\index{رؤية ثلاثية اللون}: فمهما يكن \omterm{def:g10:light-spectra:white}{الطيف}،
لا يتلقى الدماغ إلا ثلاث استجابات مخروطية، واللون هو ما
يصنعه من ذلك الثلاثي.
\end{definition}

\begin{definition}[اللون إدراكًا]\label{def:g11:color-light-sources:perception}
فاللون إذن \emph{إدراك}، لا خاصية للضوء وحده:
\omterm{def:g10:light-spectra:white}{فالطيف} يحدّد شدة عند كل طول \omterm{def:g10:signals-and-waves:wave}{موجة}، والعين تحتفظ
بثلاثة أعداد. والأضواء المختلفة الأطياف ذات الاستجابات المخروطية الثلاث
نفسها تبدو متطابقة تمامًا: وتسمّى \emph{متماثلات لونية}\index{متماثل لوني}.
\end{definition}

\begin{example}[أصفران]\label{ex:g11:color-light-sources:yellows}
يثير خط $\qty{589}{nm}$ لمصباح صوديوم المخروطين الأحمر والأخضر
بالقدر نفسه تقريبًا: فيقول الدماغ \emph{أصفر}. وشاشة تضيء
بكسلاتها الفرعية الحمراء ($\qty{610}{nm}$) والخضراء ($\qty{540}{nm}$) تعطي
الأصفر نفسه دون أي ضوء قرب $\qty{589}{nm}$. ويفصل موشور
بين الاثنين فورًا؛ أما العين فلا تقدر أبدًا.
\end{example}

\section{جمع الأضواء}

\begin{definition}[التركيب الجمعي]\label{def:g11:color-light-sources:additive}
تركيب حزم الضوء على شاشة بيضاء يجمع أطيافها: وهذا هو
\emph{التركيب الجمعي}\index{تركيب جمعي}. وثلاثة أضواء مختارة جيدًا،
أحمر وأخضر وأزرق --- وهي \emph{الألوان الأساسية
الجمعية}\index{ألوان أساسية} --- تمتزج لتعطي كل لون يمكن إدراكه
تقريبًا: R $+$ G $=$ أصفر، وR $+$ B $=$ أرجواني، وG $+$ B $=$ سماوي، و
R $+$ G $+$ B $=$ أبيض.
\end{definition}

\begin{proposition}[الألوان المتممة]\label{prop:g11:color-light-sources:complementary}
الضوءان اللذان يُدرك مجموعهما أبيض
\emph{متمّمان}\index{ألوان متممة}. والأزواج الثلاثة
أصفر/أزرق وسماوي/أحمر وأرجواني/أخضر متممة.
\end{proposition}

\begin{proof}
الأصفر هو R $+$ G؛ وإضافة الأزرق تكمل الثلاثي إلى الأبيض. والزوجان
الآخران الحجّة نفسها، بلون أساسي واحد في كل مرة.
\end{proof}

\begin{example}[الشاشات]\label{ex:g11:color-light-sources:screen}
كل بكسل في شاشة هاتف ثلاثة بكسلات فرعية --- أحمر وأخضر وأزرق،
عرضها دون $\qty{0.1}{mm}$ --- أقرب من أن تفصلها العين، فتتجمع
أضواؤها على \omterm{def:g11:lenses-and-eye:eye}{الشبكية}: فالأصفر هو إضاءة R وG، والبرتقالي هو R
بكامل \omterm{def:g10:energy-conservation:power}{الاستطاعة} وG بنصفها، والأبيض هو الثلاثة معًا. وتكشف \omterm{def:g11:lenses-and-eye:lens}{عدسة} \omterm{def:g11:lenses-and-eye:magnifier}{مكبّرة}
على مساحة بيضاء الحيلةَ: فلا أبيض في أي موضع، بل R وG وB لا غير.
\end{example}

\section{الطرح من الضوء الأبيض}

\begin{definition}[التركيب الطرحي]\label{def:g11:color-light-sources:subtractive}
\emph{المرشّح}\index{مرشّح} (زجاج ملوّن، أو رقاقة ملوّنة، أو طبقة حبر أو
دهان) ينفذ جزءًا من \omterm{def:g10:light-spectra:white}{الطيف} الذي يتلقاه ويمتصّ الباقي.
وإنتاج الألوان \emph{بنزع} حزم من \omterm{def:g10:light-spectra:white}{الضوء الأبيض} هو
\emph{التركيب الطرحي}\index{تركيب طرحي}. وألوانه
الأساسية، وهي متممات الألوان الجمعية، يطرح كلٌّ منها حزمة واحدة:
فالسماوي $=$ أبيض $-$ R، والأرجواني $=$ أبيض $-$ G، والأصفر $=$ أبيض $-$ B.
والمرشّحات المكدَّسة تطرح مجتمعة: فالسماوي $+$ الأصفر يترك الأخضر، والسماوي
$+$ الأرجواني يترك الأزرق، والأرجواني $+$ الأصفر يترك الأحمر، والثلاثة معًا تعطي الأسود. والطباعة
والتلوين يعملان هكذا، بأحبار C وM وY.
\end{definition}

\begin{method}[محاسبة الطيف]\label{met:g11:color-light-sources:bookkeeping}
للتنبؤ بلون عبر أي سلسلة من المنابع والمرشّحات والأجسام،
تتبّع الحزم الأساسية الثلاث: اكتب ضوء المنبع بمحتواه
من R وG وB؛ واحذف عند كل \omterm{def:g11:color-light-sources:subtractive}{مرشّح} أو سطح الحزمَ الممتصّة؛
ثم سمِّ لون ما بقي (فإن لم يبقَ شيء $=$ أسود).
\end{method}

\begin{example}[سلسلة مرشّحات]\label{ex:g11:color-light-sources:chain}
\omterm{def:g10:light-spectra:white}{ضوء أبيض} $(R, G, B)$ يعبر مرشّحًا أصفر: فتُمتصّ الحزمة الزرقاء،
ويبقى $(R, G)$ --- أي أصفر. ثم يمتصّ \omterm{def:g11:color-light-sources:subtractive}{مرشّح} سماوي
الحزمة الحمراء، فيبقى $(G)$: أي أخضر. وتبديل المرشّحين لا يغيّر شيئًا:
فكلٌّ يمتصّ حزمته الخاصة مهما يكن الترتيب.
\end{example}

\section{لون الجسم}

\begin{definition}[النثر ولون الجسم]\label{def:g11:color-light-sources:objectcolor}
الجسم المعتم يمتصّ جزءًا من الضوء الذي يتلقاه و
\emph{ينثر}\index{نثر} الباقي --- أي يعيد إصداره في كل
الاتجاهات، ومنها اتجاه أعيننا. و\emph{لون
الجسم}\index{لون الجسم} هو لون الضوء الذي ينثره
\emph{تحت الإضاءة المعطاة}: فالجسم يوفّر عاداته في الامتصاص،
والمنبع يوفّر المادة الخام. فغيّر المصباح يتغير اللون.
\end{definition}

\begin{example}[القميص الأحمر]\label{ex:g11:color-light-sources:redshirt}
يبدو قميص أحمر في ضوء النهار: فهو يمتصّ الحزمتين الخضراء والزرقاء
\omterm{def:g11:color-light-sources:objectcolor}{وينثر} الحمراء. وتحت ضوء أخضر نقي لا توجد حزمة حمراء
تُنثر: فيمتصّ القميص كل شيء ويبدو أسود. وتحت ضوء
أرجواني $(R, B)$ \omterm{def:g11:color-light-sources:objectcolor}{ينثر} الحزمة الحمراء وحدها: فيبقى أحمر. والجسم الأبيض
\omterm{def:g11:color-light-sources:objectcolor}{ينثر} كل حزمة يتلقاها (فيبدو أخضر تحت ضوء أخضر)؛
والجسم الأسود يمتصّها كلها تحت كل ضوء --- ولهذا أيضًا
تسخن الملابس السوداء في الشمس.
\end{example}

\section{منابع الضوء ودرجة حرارة اللون}

\begin{definition}[المنابع المتوهّجة]\label{def:g11:color-light-sources:incandescent}
المنبع الذي يضيء لأنه ساخن --- لهب أو شعيرة أو نجم --- يشعّ
بفعل \emph{التوهّج}\index{توهّج}. وطيفه هو
\omterm{def:g10:light-spectra:white}{الطيف} الحراري المستمر في \cref{ch:g10:light-spectra}، وتخضع ذروته
لقانون فين (\cref{prop:g10:light-spectra:wien})،
\[
\lambda_{\max} \times T = \qty{2.90e-3}{m.K},
\]
وهي أداتنا الكمّية من الآن فصاعدًا: فالذروة مقيسة، ودرجة الحرارة معلومة.
\end{definition}

\begin{example}[مصباح شديد الرداءة]\label{ex:g11:color-light-sources:bulb}
شعيرة تنغستن عند $T = \qty{2700}{K}$ ذروتها عند
$\lambda_{\max} = \qty{2.90e-3}{m.K}/\qty{2700}{K} = \qty{1.07e-6}{m}
= \qty{1070}{nm}$، أي عميقًا في \omterm{def:g10:light-spectra:wavelength}{تحت الأحمر}: فمعظم \omterm{def:g10:energy-conservation:power}{الاستطاعة} الكهربائية
يغادر إشعاعًا غير مرئي، ونحو $5\%$ فقط ضوءًا --- فقد كان
المصباح المتوهّج مدفأة في المقام الأول.
\end{example}

\begin{definition}[المصابيح الطيفية والليزر]\label{def:g11:color-light-sources:lamps}
\emph{المصباح الطيفي}\index{مصباح طيفي} --- وهو غاز تحت \omterm{def:g10:pressure:pressure}{ضغط} منخفض
يثيره تفريغ: صوديوم أو زئبق أو نيون --- يصدر طيفًا خطيًّا
(\cref{ch:g10:light-spectra}): أي بضعة أطوال \omterm{def:g10:signals-and-waves:wave}{موجة} معزولة، ولا شيء
بينها. أما \emph{الليزر}\index{ليزر} فيمضي أبعد: طول \omterm{def:g10:signals-and-waves:wave}{موجة}
واحد لا غير، أي \omterm{def:g10:light-spectra:white}{ضوء أحادي اللون}، في حزمة ضيقة واحدة.
\end{definition}

\begin{definition}[الثنائيات الباعثة للضوء]\label{def:g11:color-light-sources:led}
\emph{الثنائي الباعث للضوء}\index{ثنائي باعث للضوء} (LED) يصدر حزمة
\emph{شبه أحادية اللون}\index{شبه أحادي اللون} عرضها بضع عشرات من النانومترات: ضيقة
بما يكفي لتبدو لونًا محدّدًا، وأعرض بكثير من خط \omterm{def:g11:color-light-sources:lamps}{ليزر}. والثنائي
الأبيض ثنائي أزرق (ذروته قرب $\qty{450}{nm}$) مغطًّى
بمادة \emph{فوسفور}\index{فوسفور}، وهو مادة تمتصّ جزءًا من الأزرق
وتعيد إصداره حزمة صفراء عريضة: وبما أن الأزرق والأصفر
متممان (\cref{prop:g11:color-light-sources:complementary})، يُقرأ
المزيج أبيض.
\end{definition}

\begin{definition}[درجة حرارة اللون]\label{def:g11:color-light-sources:colortemp}
\emph{درجة حرارة اللون}\index{درجة حرارة اللون} لمنبع مائل
إلى البياض هي درجة حرارة الجسم المتوهّج الذي توافق صبغةُ ضوئه صبغةَ
ضوء المنبع: فالشمعة $\qty{1800}{K}$، والمصباح المتوهّج $\qty{2700}{K}$،
والهالوجين $\qty{3400}{K}$، وشمس الظهيرة $\qty{5500}{K}$، وضوء النهار الغائم
$\qty{6500}{K}$. وهي لافتة لون لا قراءة ميزان حرارة --- فثنائي
$\qty{6500}{K}$ يشتغل في درجة حرارة الغرفة --- والمفردات تسير
معكوسة: فدرجة حرارة اللون المنخفضة تُقرأ برتقاليًّا ``دافئًا''، والمرتفعة
مزرقًّا ``باردًا''.
\end{definition}

\section{مسألة: هندسة اللون}

\begin{problem}[{مسألة نهاية الأسبوع --- أضواء المسرح وشوارع الصوديوم
وثنائي LED الأبيض: كيف يُهندَس اللون، من ثلاثة كشّافات تقدر
أن ترسم أي صبغة إلى الرقاقة التي تضيء هذه الصفحة}]
\label{pb:g11:color-light-sources:1}
لا يملك مصمّم إضاءة إلا كشّافات حمراء وخضراء وزرقاء؛ ويملك مهندس
شوارع مصباحًا يصدر طول \omterm{def:g10:signals-and-waves:wave}{موجة} واحدًا لا غير؛ ويملك صانع رقائق
ثنائيًّا لا يضيء إلا أزرق. والثلاثة يبيعون اللون. وتجري هذه المسألة
مِهنهم واحدة واحدة --- المزج الجمعي على المسرح، واقتصاد اللون الواحد
في الشارع، وكيمياء \omterm{def:g11:color-light-sources:led}{الفوسفور} في الرقاقة --- وتنتهي باختيار
الضوء لغرفة نوم ولواجهة جواهري.

\medskip
\noindent\textbf{الجزء I --- المسرح: الرسم بثلاثة كشّافات.}
\begin{enumerate}
  \item تتراكب الكشّافات الثلاثة على خلفية بيضاء. سمِّ لون
        كل تراكب ثنائي ولون التراكب الثلاثي.
  \item وكيف يصنع المصمّم البرتقالي؟ والوردي الفاتح؟ (صِف
        استطاعات الكشّافات وصفًا نوعيًّا.)
  \item تقف راقصة أمام الخلفية، ولا يضيئها إلا الكشّافان
        الأحمر والأخضر. فيظهر ظلّان ملوّنان: ما لون كلٍّ منهما؟ اشرح.
  \item تُعاد دهانة الخلفية بأزرق داكن. فكيف تبدو
        تحت الكشّاف الأحمر وحده؟ وتحت الثلاثة معًا؟
  \item ولماذا تقدر ثلاثة كشّافات على إعادة إنتاج ``كل لون تقريبًا''؟ وأيّ
        جزء من التشريح يثبّت العدد ثلاثة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الأزياء: الطرح في العمل.}
\begin{enumerate}[resume]
  \item زيّ أصفر في \omterm{def:g10:light-spectra:white}{الضوء الأبيض}. فأيّ حزمة تمتصّها
        صِبغته؟
  \item تنبّأ بلونه تحت الكشّاف الأحمر، ثم الأخضر، ثم
        الأزرق.
  \item يريد المصمّم أن ينقلب الزيّ من ساطع إلى أسود بنقرة
        مفتاح واحدة. فإلى أي كشّاف تحوّل النقرة؟
  \item يُدسّ \omterm{def:g11:color-light-sources:subtractive}{مرشّح} أرجواني أمام كشّاف أبيض يضيء
        الزيّ الأصفر. تتبّع الحزم
        (\cref{met:g11:color-light-sources:bookkeeping}): فما لون الزيّ؟
  \item ولماذا تأتي مرشّحات المسرح سماوية وأرجوانية وصفراء بدل
        الحمراء والخضراء والزرقاء؟ (ماذا كان \omterm{def:g11:color-light-sources:subtractive}{مرشّح} أحمر سيفعل \omterm{def:g10:energy-conservation:power}{باستطاعة} كشّاف أبيض؟)
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- شوارع الصوديوم: اقتصاد اللون الواحد.}
\begin{enumerate}[resume]
  \item يصدر مصباح صوديوم منخفض \omterm{def:g10:pressure:pressure}{الضغط} خطًّا واحدًا أساسًا عند
        $\qty{589}{nm}$. فأيّ نوع من الأطياف هذا، وكيف يبدو
        الشارع كله تحته؟
  \item وتحت هذا المصباح، أعطِ اللون الظاهري لجدار أبيض، ولعلامة
        طريق صفراء، ولسيارة حمراء، ولسيارة زرقاء.
  \item ولماذا أحبّ الفلكيون هذه المصابيح؟ (ماذا يقدر \omterm{def:g11:color-light-sources:subtractive}{مرشّح} واحد أن يفعل
        بوهج المدينة كله؟)
  \item ويُقترح بدلًا منها مصباح شارع بالتنغستن عند $\qty{2700}{K}$.
        احسب $\lambda_{\max}$: فأين تذهب معظم استطاعته؟
  \item يحوّل مصباح الصوديوم نحو $30\%$ من استطاعته الكهربائية
        إلى ضوء مرئي، ويحوّل مصباح التنغستن نحو $5\%$. فكم
        واطًا من الضوء يوصّل كلٌّ منهما لكل $\qty{100}{W}$ من \omterm{def:g10:energy-conservation:power}{الاستطاعة}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- ثنائي LED الأبيض واختيار الضوء.}
\begin{enumerate}[resume]
  \item ثنائي LED الأبيض ثنائي أزرق ($\qty{450}{nm}$) تحت
        \omterm{def:g11:color-light-sources:led}{فوسفور} أصفر. اشرح، \omterm{prop:g11:color-light-sources:complementary}{بالألوان المتممة}، لماذا يُقرأ خرجه
        أبيض.
  \item صِف طيفه وقارنه \omterm{def:g10:light-spectra:continuous}{بالطيف المستمر}
        لضوء النهار. وأيّ حزمة ناقصة التمثيل؟
  \item استنتج أيّ الأجسام يعرضها ثنائي كهذا عرضًا رديئًا، واربط
        ذلك بجزّار \cref{exo:g11:color-light-sources:14}.
  \item يُباع مصباح LED على أنه ``$\qty{2700}{K}$''. فهل يوجد شيء بداخله
        عند $\qty{2700}{K}$؟ وبماذا تَعِد اللافتة؟ احسب
        $\lambda_{\max}$ للجسم المتوهّج الذي يحاكيه.
  \item الخاتمة --- اختر، مع جملة تبرير لكلٍّ، درجة
        حرارة لون لمصباح غرفة نوم وأخرى لواجهة جواهري؛
        ثم اذكر قانون اللون في هذا الفصل في سطر واحد.
\end{enumerate}
\end{problem}
