\chapter{النواة والنشاط الإشعاعي}\label{ch:g11:nucleus-radioactivity}

في هذه اللحظة تنفجر نحو ثمانية آلاف \omterm{def:g11:fundamental-interactions:ladder}{نواة} ذرية داخل جسمك في كل
\omterm{def:g10:orders-of-magnitude:unit}{ثانية}. ولا شيء يهاجمها: إنما تُولد نوى بعينها غير مستقرة، ويتحول كلٌّ
منها عاجلًا أو آجلًا، قاذفًا شظية بجزء معتبر من
سرعة الضوء. ويعود هذا الفصل إلى الطابق ذي الرتبة $10^{-15}$ \omterm{def:g10:orders-of-magnitude:unit}{متر} في
\cref{ch:g11:fundamental-interactions} ليرى أيّ النوى هشّ، وكيف
تتكسّر، وكيف يؤرّخ نفاد صبرها المنتظم الكهوفَ ويحرس الأسقف.

\section{النوى والنوكليونات والنظائر}

\begin{definition}[الترميز النووي]\label{def:g11:nucleus-radioactivity:notation}
تحوي \omterm{def:g11:fundamental-interactions:ladder}{النواة} $Z$ بروتونًا (وشحنة كلٍّ $+e$) و$N$ نيوترونًا (بلا
شحنة)، أي $A = Z + N$ نوكليونًا في المجموع؛ وتُكتب ${}^{A}_{Z}\mathrm{X}$،
حيث $\mathrm X$ الرمز الكيميائي، و$Z$ \emph{العدد
الذري}\index{العدد الذري}، و$A$ \emph{العدد الكتلي}\index{العدد
الكتلي}. والنوع المحدَّد بالعددين $A$ و$Z$ هو \emph{نوكليد}\index{نوكليد}؛
والنوكليدات ذات $Z$ نفسه و$A$ مختلف هي \emph{نظائر}\index{نظير}
لعنصر واحد: فالكيمياء نفسها (إذ لا ترى سحابة الإلكترونات إلا $Ze$)،
والنوى والاستقرار مختلفان.
\end{definition}

\begin{example}[الهيدروجين والكربون]\label{ex:g11:nucleus-radioactivity:hc}
للهيدروجين ثلاثة \omterm{def:g11:nucleus-radioactivity:notation}{نظائر}: ${}^{1}_{1}\mathrm{H}$ (بروتون وحيد)،
و${}^{2}_{1}\mathrm{H}$ (الديوتيريوم: بروتون ونيوترون)، وغير المستقر
${}^{3}_{1}\mathrm{H}$ (التريتيوم: نيوترونان). والكربون الطبيعي: معظمه
${}^{12}_{6}\mathrm{C}$ ($6$ بروتونات، $6$ نيوترونات)، و$1.1\%$ من
${}^{13}_{6}\mathrm{C}$، \omterm{def:g11:fundamental-interactions:ladder}{وذرة} واحدة في $10^{12}$ من الكربون
${}^{14}_{6}\mathrm{C}$ غير المستقر ($8$ نيوترونات)، بطل \cref{ex:g11:nucleus-radioactivity:dating}.
\end{example}

\section{النوى المستقرة وغير المستقرة}

\begin{proposition}[وادي الاستقرار]\label{prop:g11:nucleus-radioactivity:valley}
تشغل النوكليدات المستقرة شريطًا ضيقًا في مستوي $(Z, N)$: إذ إنّ $N \approx Z$
في النوى الخفيفة، ثم فائض نيوتروني متزايد، حتى $N \approx 1.5\,Z$.
وكل \omterm{def:g11:nucleus-radioactivity:notation}{نوكليد} خارج الشريط --- بنيوترونات زائدة، أو ببروتونات زائدة، أو
بحجم مفرط (فوراء الرصاص، $Z = 82$، لا \omterm{def:g11:nucleus-radioactivity:notation}{نوكليد} مستقر حقًّا) --- يتحول عاجلًا أو آجلًا.
\end{proposition}
\admitted

\begin{remark}
تعيين الشريط بالضبط ميكانيك كمومي --- أي المجلدات الجامعية.
والنمط دفتر \cref{ch:g11:fundamental-interactions}: فالغراء القوي
لا يربط إلا الجيران، وتنافر كولوم يمتد على \omterm{def:g11:fundamental-interactions:ladder}{النواة} كلها، ومن ثم
تحتاج النوى الثقيلة غراءً نيوترونيًّا إضافيًّا؛ ومع ذلك فإن فائض النيوترونات غير مستقر
هو أيضًا (والتفاعل الضعيف يتكفّل بذلك).
\end{remark}

\begin{definition}[النشاط الإشعاعي]\label{def:g11:nucleus-radioactivity:radioactivity}
\emph{النشاط الإشعاعي}\index{نشاط إشعاعي} هو التحول التلقائي
\omterm{def:g11:fundamental-interactions:ladder}{لنواة} غير مستقرة إلى \omterm{def:g11:nucleus-radioactivity:notation}{نوكليد} آخر، مع إصدار إشعاع. وهو
\emph{عشوائي}: فلا شيء يعلن أيّ \omterm{def:g11:fundamental-interactions:ladder}{نواة} تتفكك تاليًا، ولا
\omterm{def:g10:pressure:pressure}{ضغط} ولا درجة حرارة ولا كيمياء تقدر أن تعجّله أو تؤخّره؛ ولا يمكن التنبؤ
إلا بالمجموعات الكبيرة (\cref{def:g11:nucleus-radioactivity:halflife}).
\end{definition}

\begin{remark}[بيكريل وآل كوري]\label{rem:g11:nucleus-radioactivity:curie}
سنة 1896 وجد هنري \omterm{def:g11:nucleus-radioactivity:activity}{بيكريل} أن أملاح اليورانيوم تُضبّب لوحًا فوتوغرافيًّا
عبر ورق أسود، في درج مغلق، دون أي \omterm{def:g10:energy-conservation:energy}{طاقة} مزوَّدة. وبيّن ماري
وبيير كوري أن الأثر ذرّي، وقاساه، وعزلا البولونيوم
والراديوم، وهو أنشط بمليون مرة. وكلمة \emph{\omterm{def:g11:nucleus-radioactivity:radioactivity}{النشاط الإشعاعي}}
من وضع ماري كوري؛ وكذلك جائزتا نوبل.
\end{remark}

\section{ألفا وبيتا وغاما}

\begin{definition}[الإشعاعات التاريخية الثلاثة]\label{def:g11:nucleus-radioactivity:abg}
ثلاثة إشعاعات، سُمّيت قبل أن يعرف أحد ما هي:
\begin{itemize}
  \item \emph{ألفا}\index{إشعاع ألفا} ($\alpha$): \omterm{def:g11:fundamental-interactions:ladder}{نواة} هيليوم
        ${}^{4}_{2}\mathrm{He}$ تُقذف كاملة --- ثقيلة، مضاعفة الشحنة،
        يوقفها ورقة أو سنتيمترات من الهواء؛
  \item \emph{بيتا}\index{إشعاع بيتا} ($\beta^-$): إلكترون
        ${}^{\;0}_{-1}\mathrm{e}$ يُنشأ في \omterm{def:g11:fundamental-interactions:ladder}{النواة} ويُقذف ---
        خفيف سريع، توقفه مِلّيمترات من الألمنيوم؛
  \item \emph{غاما}\index{إشعاع غاما} ($\gamma$): إشعاع
        كهرمغناطيسي، أشدّ طاقةً من الضوء بكثير --- لا يُوقَف تمامًا أبدًا، إنما
        \emph{يُوهَن}\index{توهين}: فسنتيمتر من الرصاص يمتصّ النصف.
\end{itemize}
\end{definition}

\begin{proposition}[الانحفاظ في المعادلات النووية]\label{prop:g11:nucleus-radioactivity:soddy}
في كل تحول نووي، يكون \omterm{def:g11:nucleus-radioactivity:notation}{العدد الكتلي} الكلي $A$ والشحنة
الكلية --- أي مجموع الدلائل السفلى، مع عدّ $-1$ للإلكترون
المقذوف --- هما نفسهما قبل وبعد.
\end{proposition}
\admitted

\begin{remark}
انحفاظ الشحنة حكم أصلًا \cref{ch:g11:fundamental-interactions}؛
وانحفاظ $A$ يقول إن النوكليونات لا تُحوَّل إلا تحويلًا، ولا تُخلق ولا
تُفنى --- وهي محاسبة أمينة للمجلدات الجامعية.
\end{remark}

\begin{method}[موازنة معادلة تفكك]\label{met:g11:nucleus-radioactivity:balance}
\begin{enumerate}
  \item اكتب كل فاعل بدليليه: $\alpha = {}^{4}_{2}\mathrm{He}$،
        $\beta^- = {}^{\;0}_{-1}\mathrm{e}$، $\beta^+ = {}^{0}_{+1}\mathrm{e}$؛ ويحمل $\gamma$ الدليلين $A = Z = 0$.
  \item ساوِ بين مجموعي $A$، ثم بين مجموعي $Z$
        (\cref{prop:g11:nucleus-radioactivity:soddy})؛ وحُلّ، ثم اقرأ
        العنصر الوليد من $Z$ الخاص به في الجدول الدوري.
\end{enumerate}
\end{method}

\begin{example}[تفكك ألفا]\label{ex:g11:nucleus-radioactivity:alpha}
الراديوم-226، راديوم آل كوري، مُصدِر $\alpha$:
${}^{226}_{88}\mathrm{Ra} \to {}^{A}_{Z}\mathrm{Y} + {}^{4}_{2}\mathrm{He}$
فيفرض $A = 226 - 4 = 222$ و$Z = 88 - 2 = 86$؛ والعنصر $86$ هو الرادون، ومنه
${}^{226}_{88}\mathrm{Ra} \to {}^{222}_{86}\mathrm{Rn} + {}^{4}_{2}\mathrm{He}$
--- أي الغاز المشعّ في الأقبية غير المهوّاة.
\end{example}

\begin{example}[تفكك بيتا-ناقص]\label{ex:g11:nucleus-radioactivity:beta}
يقع الكربون-14 فوق الشريط ($8$ نيوترونات مقابل $6$ بروتونات):
${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} + {}^{\;0}_{-1}\mathrm{e}$.
وللتحقق: $14 = 14 + 0$، $6 = 7 - 1$؛ فقد صار نيوترون بروتونًا مع
الإلكترون المقذوف --- أي التفاعل الضعيف في العمل.
\end{example}

\begin{remark}[بيتا-زائد وغاما]\label{rem:g11:nucleus-radioactivity:betaplus}
تحت الشريط، يحوّل التفكك المرآتي $\beta^+$ بروتونًا إلى نيوترون
مع \emph{بوزيترون}\index{بوزيترون} ${}^{0}_{+1}\mathrm{e}$، وهو الجسيم المضادّ للإلكترون:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$،
وهو \omterm{def:g11:work-of-force:power}{حصان} عمل ماسحات PET. وبعد معظم التفككات تستقرّ الوليدة، وقد وُلدت مرتجّة،
بإصدار $\gamma$ فوق ذلك.
\end{remark}

\section{عمر النصف والنشاط}

\begin{definition}[عمر النصف]\label{def:g11:nucleus-radioactivity:halflife}
\emph{عمر النصف}\index{عمر النصف} $T_{1/2}$ \omterm{def:g11:nucleus-radioactivity:notation}{لنوكليد} هو الزمن
الذي بعده يكون نصف أي عيّنة كبيرة قد تفكك. وكل عمر نصف إضافي
ينصّف ما بقي: فمن $N_0$ \omterm{def:g11:fundamental-interactions:ladder}{نواة}، يبقى $N_0/2$ عند $T_{1/2}$، و$N_0/4$
عند $2\,T_{1/2}$، و$N_0/8$ عند $3\,T_{1/2}$، و$N_0/2^{\,n}$ بعد $n$ عمر نصف.
\end{definition}

\begin{remark}
عشوائية \omterm{def:g11:fundamental-interactions:ladder}{لنواة} واحدة، وانتظام ساعة لأجل $10^{23}$: أي حشد هائل
من رامي العملات، لا يمكن التنبؤ بهم واحدًا واحدًا، ويُقصى نصفهم في كل جولة.
أما الاستيفاء \emph{بين} أعمار النصف الكاملة فيقتضي الدالة
الأسّية --- في السنة القادمة، مع مجلد الرياضيات.
\end{remark}

\begin{example}[عشرون قوة من قوى العشرة]\label{ex:g11:nucleus-radioactivity:range}
\begin{center}\small\begin{tabular}{llll}
البولونيوم-214 & $\alpha$ & \qty{1.6e-4}{s} & حلقة في سلسلة الراديوم \\
التكنيشيوم-99m & $\gamma$ & $6.0$ ساعة & التصوير الطبي \\
الأمريسيوم-241 & $\alpha$ & $432$ سنة & كواشف الدخان \\
الكربون-14 & $\beta^-$ & \num{5700} سنة & علم الآثار \\
البوتاسيوم-40 & $\beta$ & \num{1.25e9} سنة & في كل موزة \\
اليورانيوم-238 & $\alpha$ & \num{4.5e9} سنة & حرارة باطن الأرض \\
\end{tabular}\end{center}
\end{example}

\begin{example}[التأريخ بالكربون-14]\label{ex:g11:nucleus-radioactivity:dating}
تتبادل المادة الحيّة الكربون مع الغلاف الجوي، فتحمل
نسبة الكربون-14 الجوية؛ وعند الموت يتوقف الأخذ وتنتصف
النسبة كل \num{5700} سنة. ويُظهر فحم من كهف مرسوم
ربع النسبة الحيّة: والربع نصف النصف، فقد اشتعلت
النار قبل عمرَي نصف، أي $2 \times \num{5700} = \num{11400}$ سنة.
\end{example}

\begin{definition}[النشاط]\label{def:g11:nucleus-radioactivity:activity}
\emph{نشاط}\index{نشاط} عيّنةٍ هو عدد
تفككاتها في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}، ويرمز له $\mathcal A$، ووحدته \emph{البيكريل}\index{بيكريل}:
أي $\qty{1}{Bq} = 1$ تفكك في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}. وكلما قلّت النوى الباقية قلّت
التفككات، ومن ثم ينتصف النشاط هو أيضًا عند كل عمر نصف.
\end{definition}

\section{خدم ومخاطر}

\begin{example}[ثلاث مِهَن لنواة غير مستقرة]\label{ex:g11:nucleus-radioactivity:uses}
\emph{الطب}: التكنيشيوم-99m، وهو مُصدِر $\gamma$ نقي عمر نصفه
$T_{1/2} = 6$ ساعة، يُثبَّت على \omterm{def:g11:fundamental-interactions:ladder}{جزيء} يمتصّه العضو الهدف؛ وتصوّر
كاميرا أشعة $\gamma$ من الخارج، وبحلول الغد يكون الكاشف قد
زال في معظمه --- بينما يعكس العلاج الإشعاعي المنطق، فيركّز حزمًا
لتدمير ورم. و\emph{التأريخ}: الكربون-14 للخشب والعظم والقماش إلى
نحو \num{50000} سنة مضت؛ واليورانيوم-238 للصخور --- وللأرض، عند
\num{4.5e9} سنة. و\emph{كواشف الدخان}: ذرّة من الأمريسيوم-241 تؤيّن
هواء حجرة صغيرة، فتدع تيارًا ضئيلًا يجري؛ ويخنق الدخان
التيار فينطلق الإنذار --- وتموت أشعة $\alpha$ في سنتيمترات من الهواء.
\end{example}

\begin{remark}[الجرعة والسيفرت]\label{rem:g11:nucleus-radioactivity:dose}
يقتلع الإشعاع الإلكترونات من \omterm{def:g11:fundamental-interactions:ladder}{الجزيئات} ويقدر أن يضرّ الحمض النووي. ويُتابَع
الضرر البيولوجي بما يسمّى \emph{الجرعة}\index{جرعة}، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\emph{السيفرت}\index{سيفرت} (\unit{Sv})، وهي
تزن \omterm{def:g10:energy-conservation:energy}{الطاقة} المودَعة في كل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من النسيج بمقدار ضرر كل
إشعاع. وثلاث قواعد: المسافة، والتدريع ($\alpha$: يكفي الجلد ---
ما لم يُستنشق \emph{المُصدِر}، وتلك جريمة الرادون؛ و$\beta$: الألمنيوم؛
و$\gamma$: الرصاص)، والزمن. أما قياس الجرعات بحقّه فمادة جامعية.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:1}
أعطِ عدد البروتونات وعدد النيوترونات في ${}^{16}_{8}\mathrm{O}$
و${}^{27}_{13}\mathrm{Al}$ و${}^{60}_{27}\mathrm{Co}$ و${}^{235}_{92}\mathrm{U}$
و${}^{3}_{1}\mathrm{H}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:2}
من بين ${}^{14}_{6}\mathrm{C}$ و${}^{14}_{7}\mathrm{N}$ و${}^{12}_{6}\mathrm{C}$
و${}^{13}_{6}\mathrm{C}$ و${}^{16}_{8}\mathrm{O}$، أيّها \omterm{def:g11:nucleus-radioactivity:notation}{نظائر}
بعضها لبعض؟ ولماذا لا يكون ${}^{14}_{6}\mathrm{C}$ و${}^{14}_{7}\mathrm{N}$
\emph{نظيرين}، مع تساوي عدديهما الكتليين؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:3}
البولونيوم-210، ${}^{210}_{84}\mathrm{Po}$، مُصدِر $\alpha$. اكتب
معادلة التفكك وسمِّ الوليدة ($Z = 81$: الثاليوم، $82$: الرصاص، $83$:
البزموت).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:4}
اكتب معادلتي تفكك $\beta^-$ للكوبالت-60 (${}^{60}_{27}\mathrm{Co}$)
ولليود-131 (${}^{131}_{53}\mathrm{I}$)؛ والوليدتان: $Z = 28$ النيكل،
و$Z = 54$ الزينون.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:5}
منبع يود-131 ($T_{1/2} = 8.0$ يومًا) نشاطه \qty{800}{MBq}
اليوم. فما نشاطه بعد $16$ يومًا؟ وبعد $32$ يومًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:6}
تعبر أربعة إشعاعات \omterm{def:g11:magnetic-fields:bfield}{مجالًا مغناطيسيًّا} قويًّا، ثم حواجز. عيّن كلًّا منها:
(أ) يوقفه الورق، ولا يكاد \omterm{def:g11:electric-gravitational-fields:field}{المجال} يزيحه؛ (ب) يعبر الورق،
ويوقفه \qty{4}{mm} من الألمنيوم، ويُزاح إزاحة قوية؛ (ج) يعبر الاثنين،
ولا يوهنه إلا الرصاص، ولا يُزاح؛ (د) مثل (ب)، لكنه يُزاح في الجهة الأخرى.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:7}
الرادون-222 (${}^{222}_{86}\mathrm{Rn}$) يفتح سلسلة سريعة: تفكك $\alpha$،
ثم $\alpha$ آخر، ثم $\beta^-$. اكتب المعادلات الثلاث
($Z = 82$: الرصاص، $83$: البزموت، $84$: البولونيوم).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:8}
الفلور-18، ${}^{18}_{9}\mathrm{F}$، هو كاشف ماسحات PET؛ والفلور
المستقر ${}^{19}_{9}\mathrm{F}$. فعلى أي جانب من شريط الاستقرار يقع؟
تنبّأ بنمط تفككه واكتب المعادلة ($Z = 8$: الأكسجين).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:9}
يتلقى مريض \qty{500}{MBq} من التكنيشيوم-99m ($T_{1/2} = 6.0$ ساعة)
في الساعة 08:00. فما \omterm{def:g11:nucleus-radioactivity:activity}{النشاط} عند الساعة 08:00 صباح الغد؟ وبعد كم
ساعة يهبط أول مرة دون \qty{1}{MBq}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:10}
عظم من مستنقع خثّ يُظهر نسبة كربون-14 تساوي ثمن نسبة عظم
حيّ ($T_{1/2} = \num{5700}$ سنة). فما عمر العظم؟ ولماذا لا يفيد
الكربون-14 في تأريخ عظام الديناصورات (وعمرها نحو $10^{8}$ سنة)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:11}
باستعمال وادي الاستقرار، تنبّأ بنمط التفكك ($\alpha$ أو $\beta^-$ أو
$\beta^+$) لكلٍّ من ${}^{14}_{6}\mathrm{C}$ و${}^{11}_{6}\mathrm{C}$ و
${}^{238}_{92}\mathrm{U}$، مبرّرًا كلًّا في سطر واحد (والكربون المستقر:
${}^{12}_{6}\mathrm{C}$، ${}^{13}_{6}\mathrm{C}$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:12}
تنتهي سلسلة اليورانيوم-238، بعد خطوات كثيرة، عند
${}^{206}_{82}\mathrm{Pb}$ المستقر؛ وكل خطوة $\alpha$ أو $\beta^-$. باستعمال
انحفاظ $A$ وحده، جِد عدد خطوات $\alpha$؛ ثم، بانحفاظ
$Z$، عدد خطوات $\beta^-$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:13}
منبع كوبالت-60 في مستشفى ($T_{1/2} = 5.3$ سنة) نشاطه
\qty{6.4}{TBq}؛ وتسمح الأنظمة بالتخلص منه دون \qty{0.1}{TBq}. فبعد كم
سنة يجوز التخلص منه؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:14}
\omterm{def:g11:nucleus-radioactivity:activity}{نشاط} جسمك نحو \qty{8000}{Bq}. فكم \omterm{def:g11:fundamental-interactions:ladder}{نواة} من نواك تتفكك
في اليوم؟ وعلى مدى عمر $80$ سنة؟ وتضيف موزة نحو \qty{15}{Bq} بينما
تهضمها: علّق، في جملة، على العناوين التي تخاف كل \omterm{def:g11:nucleus-radioactivity:activity}{بيكريل}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:15}
يحوي كاشف دخان \qty{0.2}{\micro\gram} من الأمريسيوم-241
($T_{1/2} = 432$ سنة)، ونشاطه نحو \qty{25}{kBq}. فكم تفككًا
في اليوم؟ وبعد عشر سنوات من الخدمة، هل تفكك الأمريسيوم بأقلّ
بكثير من النصف، أم بنحو النصف، أم بأكثر بكثير --- وهل نفاد المنبع هو سبب
استبدال الكواشف؟ وأخيرًا، لماذا يكون منبع $\alpha$ هذا غير ضارّ على
السقف، وخطرًا إذا سُحقت الكبسولة واستُنشق الغبار؟
\end{exercise}

\section{مسألة: المومياء والمفاعل وكاشف الدخان}

\begin{problem}[{مسألة نهاية الأسبوع --- ثلاث مِهَن لنواة غير مستقرة:
مومياء مؤرَّخة بالتنصيفات، وشجرة عائلة اليورانيوم موازَنة، وكاشف دخان
مدقَّق، ولماذا كان لا بدّ للنجوم أن تكون فانية}]
\label{pb:g11:nucleus-radioactivity:1}
ثلاث نوى غير مستقرة، وثلاث وظائف: فالكربون-14 يختم بالوقت كل نبات
وحيوان ميت، واليورانيوم-238 يسخّن كوكبًا من الداخل، والأمريسيوم-241 يراقب
الأسقف. المعطيات: $T_{1/2} = \num{5700}$ سنة (C-14)، و\num{4.5e9} سنة
(U-238)، و$432$ سنة (الأمريسيوم-241)؛ ويُظهر الكربون الحيّ $13.6$ تفككًا في الدقيقة
لكل غرام؛ والأعداد الذرية: N $7$، O $8$، Pb $82$، Bi $83$، Po $84$، Rn $86$،
Ra $88$، Th $90$، Pa $91$، U $92$، Np $93$، Am $95$.

\medskip
\noindent\textbf{الجزء I --- المومياء.}
\begin{enumerate}
  \item أعطِ تركيب (البروتونات والنيوترونات) كلٍّ من ${}^{12}_{6}\mathrm{C}$
        و${}^{14}_{6}\mathrm{C}$. ولماذا تتطابق كيمياؤهما في جسم؟
  \item ولماذا تكون نسبة الكربون-14 ثابتة في جسم حيّ، ولماذا
        تبدأ بالهبوط عند الموت؟
  \item اكتب معادلة تفكك الكربون-14. وهل كان إشعاعه سيفلت
        عبر لفائف الكتّان في المومياء؟
  \item غرام من كربون اللفائف يُظهر $6.8$ تفككًا في الدقيقة.
        فأيّ جزء من المعدل الحيّ هو ذلك؟ وأرّخ المومياء.
  \item و``مومياء'' \omterm{def:g10:orders-of-magnitude:unit}{ثانية}، جوهرة مجموعة خاصة، تُظهر
        $13.2$ تفككًا في الدقيقة لكل غرام. فما الحكم؟
  \item قدّر المعدل من عيّنة عمرها عشرة أعمار نصف، واشرح
        لماذا يخبو التأريخ بالكربون وراء نحو \num{57000} سنة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- المفاعل تحت المرج.}
\begin{enumerate}[resume]
  \item أعطِ تركيب ${}^{238}_{92}\mathrm{U}$. ولماذا تحتاج النوى الثقيلة
        هذا الفائض النيوتروني؟
  \item اكتب معادلة تفككه $\alpha$.
  \item ثم تتفكك الوليدة $\beta^-$، وتتفكك الحفيدة
        $\beta^-$ مرة أخرى. اكتب المعادلتين. وأيّ عنصر يعاود الظهور؟
  \item وتنتهي السلسلة عند ${}^{206}_{82}\mathrm{Pb}$. عُدّ خطوات $\alpha$
        فيها، ثم خطوات $\beta^-$.
  \item تشكّلت الأرض قبل \num{4.5e9} سنة. فأيّ جزء من يورانيوم-238
        البدئي يبقى اليوم؟ وبعد $5$ مليار سنة أخرى؟
  \item وفي جملة واحدة: ماذا تشغّل حرارة التفكك المدفونة هذه عند
        السطح؟ (فكّر في البراكين والقارات الجارفة.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- كاشف الدخان.}
\begin{enumerate}[resume]
  \item اكتب معادلة تفكك $\alpha$ \omterm{def:g11:fundamental-interactions:ladder}{للنواة} ${}^{241}_{95}\mathrm{Am}$.
  \item ونشاطه \qty{33}{kBq}: فكم تفككًا في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}؟ وفي اليوم؟
  \item اشرح الكاشف: ماذا تفعل جسيمات $\alpha$ بهواء
        الحجرة، وما الذي يغيّره الدخان؟
  \item ولماذا يكون مُصدِر $\alpha$ هو الخيار الصحيح هنا --- ويكون
        مُصدِر $\gamma$ \omterm{def:g11:nucleus-radioactivity:activity}{بالنشاط} نفسه أسوأ خيار؟
  \item وبعد عمر نصف واحد ($432$ سنة!) سيكون \omterm{def:g11:nucleus-radioactivity:activity}{النشاط}
        \qty{16.5}{kBq}؛ ويطلب الدليل الاستبدال بعد عشر سنوات.
        فهل المنبع هو الحلقة الأضعف؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- لماذا كان لا بدّ للنجوم أن تكون فانية.}
\begin{enumerate}[resume]
  \item صنعت النجوم المحتضرة كل عنصر أثقل من الهيليوم ونثرته.
        فأيّ جزء من يورانيوم-238 المصنوع قبل \num{9e9} سنة
        يبقى اليوم؟ ولماذا تجعل وفرته الباقية على الأرض
        الكوكبَ أصغر بكثير من أقدم النجوم؟
  \item تقود حرارة باطن الأرض المشعّة البركنة وتكتونية
        الصفائح، وتبقي \omterm{def:g11:fundamental-interactions:ladder}{النواة} تخضّ الدرع المغناطيسي في
        \cref{ch:g11:magnetic-fields}. فكيف كان سيكون كوكب من ذرات
        مستقرة وحدها؟
  \item في سطر واحد، شعرًا ومحاسبةً: أين صُنعت ذراتك،
        وما الذي ما يزال يدفّئ الأرض تحت قدميك؟
\end{enumerate}
\end{problem}
