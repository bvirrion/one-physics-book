\chapter{شغل قوة}\label{ch:g11:work-of-force}

ادفع خزانة عبر الغرفة تكن قد اشتغلت؛ وادفع جدارًا
طوال العصر فلن تحسب الفيزياء، مهما تقل عضلاتك، إلا
صفرًا. \omterm{def:g11:work-of-force:work}{فالشغل} هو ما توصّله \omterm{def:g10:inertia:force}{قوة} \emph{على امتداد حركة} --- أي \omterm{def:g10:energy-conservation:energy}{طاقة}
في العبور (\cref{ch:g10:energy-conservation})، تُدفع \omterm{def:g11:work-of-force:work}{بالجول}. ويسعّر
هذا الفصل قوى \cref{ch:g11:forces-and-motion} --- من سحب
بزاوية، \omterm{def:g10:universal-gravitation:weight}{ووزن}، \omterm{def:g10:inertia:inventory}{واحتكاك} --- ويوقّت معدل الدفع: أي \omterm{def:g10:energy-conservation:power}{الاستطاعة}.

\section{شغل قوة ثابتة}

\begin{definition}[شغل قوة ثابتة]\label{def:g11:work-of-force:work}
\omterm{def:g10:inertia:force}{القوة} الثابتة $\vect F$ المطبَّقة على جسم يتحرك على قطعة
مستقيمة من $A$ إلى $B$ تنجز \emph{الشغل}\index{شغل (قوة)}
\[
W_{AB}(\vect F) = F \times AB \times \cos\theta ,
\]
حيث $F$ شدة \omterm{def:g10:inertia:force}{القوة} (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{N})، و$AB$ طول
الانتقال (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{m})، و$\theta$ الزاوية بين
$\vect F$ و$\vect{AB}$. \omterm{def:g10:orders-of-magnitude:unit}{ووحدة} الشغل هي
\emph{الجول}\index{جول} (\unit{J}): فالجول الواحد شغل \omterm{def:g10:inertia:force}{قوة}
قدرها \omterm{def:g10:inertia:force}{نيوتن} واحد على امتداد \omterm{def:g10:orders-of-magnitude:unit}{متر} واحد في جهتها هي.
\end{definition}

\begin{remark}\label{rem:g11:work-of-force:scal}
المقدار $F \times AB \times \cos\theta$ هو بالضبط الجداء السلّمي
$\vect F \cdot \vect{AB}$ في مجلد رياضيات هذه السنة. فلا يكسب شيئًا
إلا المركّبة $F\cos\theta$ \emph{على امتداد} الحركة.
\end{remark}

\begin{example}[حقيبة على حزام]\label{ex:g11:work-of-force:suitcase}
يجرّ مسافر حقيبة \qty{300}{m} عبر صالة، والحزام
مسحوب \omterm{def:g10:inertia:force}{بقوة} $F = \qty{40}{N}$ بزاوية $\theta = \qty{50}{\degree}$ مع
الأرضية: أي $W = 40 \times 300 \times \cos\qty{50}{\degree} \approx
\qty{7.7}{kJ}$. ولو سُحبت أفقيًّا ($\theta = 0$) لأوصلت \omterm{def:g10:inertia:force}{القوة} نفسها
\qty{12}{kJ}: فثلث السحب لا يرفع شيئًا.
\end{example}

\begin{definition}[الشغل المحرِّك والمقاوم والمعدوم]\label{def:g11:work-of-force:sign}
\omterm{def:g10:signals-and-waves:signal}{إشارة} $\cos\theta$ تصنّف كل \omterm{def:g10:inertia:force}{قوة} في ثلاثة أنظمة:
\begin{itemize}
  \item $\theta < \qty{90}{\degree}$: فيكون $W > 0$، \omterm{def:g11:work-of-force:work}{والشغل}
        \emph{محرِّك}\index{شغل محرِّك} --- أي أن \omterm{def:g10:inertia:force}{القوة} تغذّي
        الحركة \omterm{def:g10:energy-conservation:energy}{بالطاقة}؛
  \item $\theta = \qty{90}{\degree}$: فيكون $W = 0$، ولا تشتغل \omterm{def:g10:inertia:force}{القوة}
        البتة، مهما كبرت؛
  \item $\theta > \qty{90}{\degree}$: فيكون $W < 0$، \omterm{def:g11:work-of-force:work}{والشغل}
        \emph{مقاوم}\index{شغل مقاوم} --- أي أن \omterm{def:g10:inertia:force}{القوة} تستنزف
        \omterm{def:g10:energy-conservation:energy}{الطاقة}. والحالة النقية مكبح عند $\qty{180}{\degree}$:
        إذ إنّ \qty{800}{N} على \qty{45}{m} تسحب $800 \times 45 \times (-1)
        = \qty{-36}{kJ}$ من سيارة، إلى أقراص ساخنة.
\end{itemize}
\end{definition}

\section{شغل الوزن}

\begin{proposition}[شغل الوزن]\label{prop:g11:work-of-force:weight}
حين ينتقل جسم كتلته $m$ من $A$ (على الارتفاع $z_A$) إلى $B$
(على الارتفاع $z_B$) على \emph{أي} \omterm{def:g10:relative-motion:trajectory}{مسار} مؤلَّف من قطع مستقيمة، ينجز
وزنه $\vect P$ \omterm{def:g11:work-of-force:work}{الشغل}
\[
W_{AB}(\vect P) = mg\,(z_A - z_B) = \pm\, mgh ,
\]
حيث $h = \abs{z_A - z_B}$ هو الهبوط في الارتفاع: أي $+mgh$ عند النزول،
و$-mgh$ عند الصعود --- فالهبوط وحده يهمّ، لا الطريق.
\end{proposition}

\begin{proof}
على قطعة مستقيمة واحدة $AB$، يصنع \omterm{def:g10:universal-gravitation:weight}{الوزن} (وشدته $mg$، وهو شاقولي
إلى أسفل) زاوية $\theta$ مع $\vect{AB}$، ويكون $AB\cos\theta$
بالضبط مسقط الانتقال على العمودي النازل:
أي $z_A - z_B$. ومنه $W = mg \times AB\cos\theta = mg\,(z_A - z_B)$.
وعلى سلسلة من القطع تتجمع الأشغال وتتداخل الهبوطات:
$mg(z_A - z_C) + mg(z_C - z_B) = mg(z_A - z_B)$.
\end{proof}

\begin{proposition}[أي مسار]\label{prop:g11:work-of-force:anypath}
تصحّ الصيغة نفسها على أي \omterm{def:g10:relative-motion:trajectory}{مسار} منحنٍ من $A$ إلى $B$.
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:work-of-force:anypath}
وهو معقول --- فالمنحني قريب كما نشاء من خط مكسور من قطع
قصيرة كثيرة --- لكن جعل ``كما نشاء'' أمينًا يقتضي
التحليل، ويُجرى في \cref{ch:g12:work-and-energy}.
\end{remark}

إلى $B$ يهبط $h$ نفسه: فينجز \omterm{def:g10:universal-gravitation:weight}{الوزن} $+mgh$ على كليهما.}

\begin{example}[متنزّه]\label{ex:g11:work-of-force:hiker}
متنزّه كتلته \qty{78}{kg} (مع الحقيبة) يصعد \qty{450}{m}، بأي
درب كان: $W(\vect P) = -78 \times 9.81 \times 450 \approx \qty{-344}{kJ}$،
و$+\qty{344}{kJ}$ بالضبط في النزول، مهما تكن الالتفافات.
\end{example}

\section{شغل الاحتكاك}

\begin{proposition}[شغل احتكاك الانزلاق]\label{prop:g11:work-of-force:friction}
\omterm{def:g10:inertia:force}{قوة} \omterm{def:g10:inertia:inventory}{احتكاك} شدتها ثابتة $f$، معاكسة للحركة في كل لحظة،
تنجز \omterm{def:g11:work-of-force:work}{الشغل} $W = -fL$ على \omterm{def:g10:relative-motion:trajectory}{مسار} طوله الكلي $L$:
فهو \omterm{def:g11:work-of-force:sign}{مقاوم} دائمًا، ومتناسب مع \emph{طول \omterm{def:g10:relative-motion:trajectory}{المسار}
المقطوع} --- لا مع الانتقال.
\end{proposition}

\begin{proof}
كل قطعة مستقيمة قصيرة تكون فيها \omterm{def:g10:inertia:force}{القوة} بزاوية $\qty{180}{\degree}$ مع
الحركة، فتسهم بالمقدار $-f \times (\text{طوله})$؛ وتتجمع القطع
لتعطي $-fL$. \omterm{def:g10:relative-motion:trajectory}{والمسار} المنحني يُقطَّع قطعًا قصيرة كذلك، ويُقبَل
النهاية كما في حالة \omterm{def:g10:universal-gravitation:weight}{الوزن} (\cref{rem:g11:work-of-force:anypath}).
\end{proof}

\begin{remark}[الاحتكاك في مقابل الوزن: الفارق الكبير]\label{rem:g11:work-of-force:divide}
شغل \omterm{def:g10:universal-gravitation:weight}{الوزن} ينسى \omterm{def:g10:relative-motion:trajectory}{المسار}؛ أما شغل \omterm{def:g10:inertia:inventory}{الاحتكاك} فرسم على كل \omterm{def:g10:orders-of-magnitude:unit}{متر}. وفي
رحلة ذهاب وإياب يصفّر \omterm{def:g10:universal-gravitation:weight}{الوزن} حسابه --- فما يأخذه صعودًا يردّه
نزولًا --- بينما يفرض \omterm{def:g10:inertia:inventory}{الاحتكاك} $-fL$ في كل مرحلة ولا يردّ
شيئًا: فشغل \omterm{def:g10:universal-gravitation:weight}{الوزن} يمكن تخزينه واسترجاعه (طاقةً كامنة،
\cref{ch:g11:mechanical-energy})، أما شغل \omterm{def:g10:inertia:inventory}{الاحتكاك} فيضيع حرارةً.
\end{remark}

\begin{example}[ذهابًا وإيابًا]\label{ex:g11:work-of-force:roundtrip}
ينزلق صندوق \qty{5.0}{m} على أرضية ثم \qty{5.0}{m} عائدًا،
في مواجهة $f = \qty{60}{N}$: فينجز \omterm{def:g10:inertia:inventory}{الاحتكاك} $-60 \times 10 =
\qty{-600}{J}$ مع أن الصندوق ينتهي حيث بدأ. أما \omterm{def:g10:universal-gravitation:weight}{الوزن} و
\omterm{def:g10:inertia:inventory}{رد الفعل الناظمي}، وهما عموديان على الحركة، فصفر طول الوقت.
\end{example}

\begin{remark}[وماذا عن زنبرك؟]\label{rem:g11:work-of-force:spring}
يسحب الزنبرك أقوى كلما ازدادت استطالته: فقوته ليست
ثابتة ولا ينطبق \cref{def:g11:work-of-force:work}.
وحساب شغله يعني تقطيع الاستطالة خطوات متناهية في الصغر
وجمعها --- أي التحليل، ويأتي في \cref{ch:g12:work-and-energy}.
\end{remark}

\section{الاستطاعة}

\begin{definition}[الاستطاعة]\label{def:g11:work-of-force:power}
\emph{استطاعة}\index{استطاعة!ميكانيكية} قوةٍ هي المعدل الذي
تشتغل به، $P = W/\Delta t$، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \emph{الواط}\index{واط}
($\unit{W} = \unit{J/s}$) --- وهو الواط نفسه في
\cref{ch:g11:circuits-and-power}. وتبقى \omterm{def:g10:orders-of-magnitude:unit}{وحدة} سابقة للنظام الدولي على شارات
المحركات: هي \emph{الحصان}\index{حصان}،
$\qty{1}{hp} = \qty{736}{W}$.
\end{definition}

\begin{proposition}[الاستطاعة عند سرعة ثابتة]\label{prop:g11:work-of-force:pfv}
إذا تحرك الجسم بسرعة ثابتة $v$، فإن \omterm{def:g10:inertia:force}{قوة} ثابتة $\vect F$
بزاوية $\theta$ مع الحركة توصّل \omterm{def:g10:energy-conservation:power}{الاستطاعة}
$P = F v \cos\theta$ --- وخاصةً $P = Fv$ في حالة \omterm{def:g10:inertia:force}{قوة} على امتداد
الحركة.
\end{proposition}

\begin{proof}
في زمن $\Delta t$ يقطع الجسم $d = v\,\Delta t$، ومنه
$W = F \times v\,\Delta t \times \cos\theta$؛ فاقسم على $\Delta t$.
\end{proof}

\begin{example}[رافعة]\label{ex:g11:work-of-force:crane}
ترفع رافعة منصة كتلتها \qty{600}{kg} بسرعة ثابتة \qty{0.90}{m/s}: فتساوي
\omterm{def:g10:inertia:force}{قوة} الكابل الوزنَ، $mg = \qty{5.89e3}{N}$، ومنه
$P = Fv = 5890 \times 0.90 \approx \qty{5.3}{kW} \approx \qty{7.2}{hp}$
--- أي فريق من سبعة أحصنة في صندوق فولاذي.
\end{example}

\begin{method}[الصعود بقوة أقل: التعرّج]\label{met:g11:work-of-force:zigzag}
لرفع حمل إلى ارتفاع $h$ \omterm{def:g10:inertia:force}{بقوة} محدودة:
\begin{enumerate}
  \item الفاتورة ثابتة: فبحسب \cref{prop:g11:work-of-force:weight}،
        يكلّف أي طريق إلى القمة الشغلَ $mgh$.
  \item مدّ \omterm{def:g10:relative-motion:trajectory}{المسار}: فعلى منحدر طوله $L$ تكون \omterm{def:g10:inertia:force}{القوة} اللازمة
        (بسرعة ثابتة، وبإهمال \omterm{def:g10:inertia:inventory}{الاحتكاك}) هي $F = mgh/L$ --- فضاعِف
        الطول ينتصف الجهد.
  \item ادفع الرسم الإضافي: \omterm{def:g10:inertia:inventory}{فالاحتكاك} يحاسب \omterm{def:g10:orders-of-magnitude:unit}{بالمتر}
        (\cref{prop:g11:work-of-force:friction})، فيضيف $fL$.
\end{enumerate}
وتتعرّج الطرق الجبلية من أجل الخطوة 2: الارتفاع نفسه، \omterm{def:g10:inertia:force}{وقوة} أقل، \omterm{def:g11:work-of-force:work}{والشغل} نفسه.
\end{method}

\section{مسألة: يوم عامل النقل}

\begin{problem}[{مسألة نهاية الأسبوع --- الفيزياء تدفع بالجول:
يوم من الإزلاق والرفع والمنعطفات الحادّة، ينتهي بالاكتشاف
المذلّ لما تراه الغلاية في العمل الشريف}]
\label{pb:g11:work-of-force:1}
يفرّغ طاقم نقل مستودعًا، ويجرّ الحمولة إلى شقة في الطابق
الثالث، ثم يقود الشاحنة إلى قرية على قمة تلّ. وكل مهمة مسعَّرة
\omterm{def:g11:work-of-force:work}{بالجول}؛ وستدقّق غلاية المطبخ اليوم. والصناديق: \qty{30}{kg}.

\medskip
\noindent\textbf{الجزء I --- عبر أرضية المستودع.}
يُدفع صندوق \omterm{def:g10:inertia:force}{بقوة} أفقية $F = \qty{110}{N}$ على مدى
$d = \qty{8.0}{m}$؛ \omterm{def:g10:inertia:inventory}{واحتكاك} الانزلاق $f = \qty{95}{N}$.
\begin{enumerate}
  \item احسب شغل الدفعة.
  \item احسب شغل \omterm{def:g10:inertia:inventory}{الاحتكاك}، وشغل \omterm{def:g10:universal-gravitation:weight}{الوزن}، وشغل \omterm{def:g10:inertia:inventory}{رد الفعل
        الناظمي} (وبرّر الأصفار).
  \item ما \omterm{def:g11:work-of-force:work}{الشغل} الكلي على الصندوق، وما معنى إشارته (وقد بدأ
        الصندوق ساكنًا)؟ وأيّ دفعة تبقي السرعة ثابتة
        (\cref{ch:g11:forces-and-motion})؟
  \item يمكن بلوغ الزاوية البعيدة على القُطر \qty{10.0}{m}
        أو على جدارين (\qty{6.0}{m} ثم \qty{8.0}{m}). فما شغل \omterm{def:g10:inertia:inventory}{الاحتكاك}
        على كل طريق؟
  \item وفي رحلة ذهاب وإياب قطرية، احسب شغل
        \omterm{def:g10:inertia:inventory}{الاحتكاك} وشغل \omterm{def:g10:universal-gravitation:weight}{الوزن}. فأيّ القوتين تردّ؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- إلى الطابق الثالث ($h = \qty{9.0}{m}$).}
\begin{enumerate}[resume]
  \item احسب شغل \omterm{def:g10:universal-gravitation:weight}{الوزن} على صندوق يُرفع إلى الشقة.
        وأيّ شغل يجب أن يوفّره الطاقم على الأقل؟
  \item وإذا حُمل على الدرج بعامل كتلته \qty{70}{kg}، حاسبت \omterm{def:g11:fundamental-interactions:four}{الجاذبية}
        العامل \emph{والصندوق}: فما \omterm{def:g11:work-of-force:work}{الشغل} الكلي ضد \omterm{def:g11:fundamental-interactions:four}{الجاذبية}، وما
        الجزء المبذول فعلًا على الصندوق؟
  \item ومنحدر عربة طوله \qty{30}{m} لصعود \qty{9.0}{m}
        نفسه، فيه \qty{40}{N} من \omterm{def:g10:inertia:inventory}{احتكاك} التدحرج. بسرعة ثابتة،
        احسب الدفعة اللازمة على امتداد المنحدر وشغلها.
  \item قارن المنحدر برفع عمودي مستقيم: \omterm{def:g10:inertia:force}{القوة} اللازمة، و
        \omterm{def:g11:work-of-force:work}{الشغل} المنجَز. فماذا يشتري المنحدر، وبأي ثمن؟
  \item رافعة بحبل وبكرة ترفع الصندوق بسرعة \qty{0.50}{m/s}.
        فما المدة \omterm{def:g10:energy-conservation:power}{والاستطاعة} الميكانيكية؟
  \item استخرج تلك \omterm{def:g10:energy-conservation:power}{الاستطاعة} بالعلاقة $P = Fv$. \omterm{def:g10:energy-conservation:efficiency}{ومردود} محرك الرافعة
        $60\%$: فما \omterm{prop:g11:circuits-and-power:power}{الاستطاعة الكهربائية} المسحوبة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الطريق المتعرّج.}
يجب أن تبلغ الشاحنة المحمَّلة (\qty{3500}{kg}) قرية على ارتفاع \qty{240}{m}
فوق الوادي: إما ممرّ مستقيم طوله \qty{1.0}{km} بميل $24\%$، وإما
طريق معبّد طوله \qty{8.0}{km} بميل $3.0\%$ (والميل $= \sin\alpha$).
\begin{enumerate}[resume]
  \item تحقّق من أن الطريقين يصعدان \qty{240}{m} نفسه.
  \item بسرعة ثابتة وبإهمال \omterm{def:g10:inertia:inventory}{الاحتكاك}، احسب \omterm{def:g10:inertia:force}{قوة}
        القيادة اللازمة على كل طريق.
  \item ما شغل \omterm{def:g10:inertia:force}{قوة} القيادة على كل طريق؟ قارنه بالمقدار $mgh$.
  \item \omterm{def:g11:circuits-and-power:resistance}{ومقاومة} التدحرج \qty{400}{N} على كليهما: احسب \omterm{def:g11:work-of-force:work}{الشغل}
        الإضافي على كل طريق. فكم تكلّف راحة التعرّج؟
  \item وعند \qty{36}{km/h}، احسب \omterm{def:g11:work-of-force:power}{استطاعة} المحرك اللازمة على كل
        طريق (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{kW} وبوحدة \unit{hp}). ولماذا تتعرّج الطرق؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- دفتر اليوم.}
\begin{enumerate}[resume]
  \item يُزلق الطاقم $60$ صندوقًا عبر الأرضية (بدفعة
        السؤال 1) ويرفعها كلها البالغة $60$ إلى \qty{9.0}{m}
        (السؤال 6). فما \omterm{def:g11:work-of-force:work}{الشغل} الميكانيكي الكلي الموصَّل إلى الصناديق؟
  \item أضِف جسم حامل الدرج نفسه: $20$ رحلة صعود بكتلة \qty{70}{kg}
        (فمردودات النزول تذهب إلى ركبتين ساخنتين، لا إلى الدفتر). فما
        مجموع اليوم؟
  \item اشتغل الطاقم $8.0$ ساعات: فما \omterm{def:g10:energy-conservation:power}{الاستطاعة} الميكانيكية المتوسطة؟ وكم
        كانت غلاية \qty{2.0}{kW} ستستغرق لتنفق \omterm{def:g10:energy-conservation:energy}{الطاقة} نفسها؟
  \item الخاتمة: قل في جملتين ما الذي يدفع \omterm{def:g11:work-of-force:work}{الجول} ثمنه --- وما
        لا يدفعه (الجدار المدفوع طوال العصر، والصندوق المحمول
        بلا حركة) --- ولماذا يملك عمال النقل ملافيف ومنحدرات وطرقًا
        متعرّجة بدل عضلات أكبر.
\end{enumerate}
\end{problem}
