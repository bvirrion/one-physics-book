\chapter{Usaha sebuah Gaya}\label{ch:g11:work-of-force}

Doronglah sebuah lemari pakaian melintasi ruangan dan kamu sudah
melakukan \omterm{def:g11:work-of-force:work}{usaha}; doronglah sebuah dinding sepanjang sore dan, apa pun
laporan ototmu, fisika menghitungnya nol. \omterm{def:g11:work-of-force:work}{Usaha} adalah apa yang
diserahkan sebuah \omterm{def:g10:inertia:force}{gaya} \emph{sepanjang sebuah gerak} --- yaitu \omterm{def:g10:energy-conservation:energy}{energi}
yang sedang berpindah (\cref{ch:g10:energy-conservation}), yang dibayar
dalam \omterm{def:g11:work-of-force:work}{joule}. Bab ini menghargai \omterm{def:g10:inertia:force}{gaya} pada
\cref{ch:g11:forces-and-motion} --- tarikan bersudut, \omterm{def:g10:universal-gravitation:weight}{berat}, \omterm{def:g10:inertia:inventory}{gesekan} ---
lalu mencatat laju pembayarannya: \omterm{def:g11:work-of-force:power}{daya}.

\section{Usaha sebuah gaya tetap}

\begin{definition}[Usaha sebuah gaya tetap]\label{def:g11:work-of-force:work}
Sebuah \omterm{def:g10:inertia:force}{gaya} tetap $\vect F$ yang dikerjakan pada benda yang bergerak
sepanjang ruas garis lurus dari $A$ ke $B$ melakukan
\emph{usaha}\index{usaha (sebuah gaya)}
\[
W_{AB}(\vect F) = F \times AB \times \cos\theta ,
\]
dengan $F$ adalah besar \omterm{def:g10:inertia:force}{gayanya} (dalam \unit{N}), $AB$ panjang
perpindahannya (dalam \unit{m}) dan $\theta$ sudut antara $\vect F$ dan
$\vect{AB}$. Usaha diukur dalam \emph{joule}\index{joule} (\unit{J}):
satu joule adalah usaha \omterm{def:g10:inertia:force}{gaya} satu \omterm{def:g10:inertia:force}{newton} sepanjang satu \omterm{def:g10:orders-of-magnitude:unit}{meter} pada
arahnya sendiri.
\end{definition}

\begin{remark}\label{rem:g11:work-of-force:scal}
$F \times AB \times \cos\theta$ tepat sama dengan hasil kali skalar
$\vect F \cdot \vect{AB}$ pada jilid matematika tahun ini. Hanya
komponen $F\cos\theta$ yang \emph{searah} geraknya yang menghasilkan
sesuatu.
\end{remark}

\begin{example}[Koper pada tali seret]\label{ex:g11:work-of-force:suitcase}
Seorang pelancong menyeret koper sejauh \qty{300}{m} melintasi
terminal, dengan talinya ditarik $F = \qty{40}{N}$ pada
$\theta = \qty{50}{\degree}$ terhadap lantainya:
$W = 40 \times 300 \times \cos\qty{50}{\degree} \approx
\qty{7.7}{kJ}$.
Jika ditarik mendatar ($\theta = 0$), \omterm{def:g10:inertia:force}{gaya} yang sama akan menyerahkan
\qty{12}{kJ}: sepertiga tarikannya tidak mengangkat apa-apa.
\end{example}

\begin{definition}[Usaha pendorong, penghambat, dan nol]\label{def:g11:work-of-force:sign}
Tanda $\cos\theta$ memilah setiap \omterm{def:g10:inertia:force}{gaya} ke dalam tiga golongan:
\begin{itemize}
  \item $\theta < \qty{90}{\degree}$: $W > 0$, \omterm{def:g11:work-of-force:work}{usahanya} bersifat
        \emph{pendorong}\index{usaha pendorong} --- \omterm{def:g10:inertia:force}{gayanya} menyuapkan
        \omterm{def:g10:energy-conservation:energy}{energi} ke dalam geraknya;
  \item $\theta = \qty{90}{\degree}$: $W = 0$, \omterm{def:g10:inertia:force}{gayanya} sama sekali
        tidak melakukan \omterm{def:g11:work-of-force:work}{usaha}, sebesar apa pun ia;
  \item $\theta > \qty{90}{\degree}$: $W < 0$, \omterm{def:g11:work-of-force:work}{usahanya} bersifat
        \emph{penghambat}\index{usaha penghambat} --- \omterm{def:g10:inertia:force}{gayanya} menguras
        \omterm{def:g10:energy-conservation:energy}{energi}. Kasus murninya adalah rem pada $\qty{180}{\degree}$:
        \qty{800}{N} sepanjang \qty{45}{m} mengambil
        $800 \times 45 \times (-1)
        = \qty{-36}{kJ}$ dari sebuah mobil,
        lalu memindahkannya ke cakram yang panas.
\end{itemize}
\end{definition}

\section{Usaha oleh berat}

\begin{proposition}[Usaha oleh berat]\label{prop:g11:work-of-force:weight}
Ketika sebuah benda bermassa $m$ berpindah dari $A$ (ketinggian $z_A$)
ke $B$ (ketinggian $z_B$) melalui \omterm{def:g10:relative-motion:trajectory}{lintasan} \emph{apa pun} yang tersusun
dari ruas garis lurus, \omterm{def:g10:universal-gravitation:weight}{beratnya} $\vect P$ melakukan \omterm{def:g11:work-of-force:work}{usaha}
\[
W_{AB}(\vect P) = mg\,(z_A - z_B) = \pm\, mgh ,
\]
dengan $h = \abs{z_A - z_B}$ adalah selisih ketinggiannya: $+mgh$
ketika turun, $-mgh$ ketika naik --- hanya selisihnya yang berperan,
bukan rutenya.
\end{proposition}

\begin{proof}
Pada satu ruas garis lurus $AB$, \omterm{def:g10:universal-gravitation:weight}{beratnya} (yang besarnya $mg$, lurus ke
bawah) membentuk sudut $\theta$ dengan $\vect{AB}$, dan $AB\cos\theta$
tepat sama dengan proyeksi perpindahannya pada arah tegak ke bawah:
$z_A - z_B$. Karena itu $W = mg \times AB\cos\theta = mg\,(z_A - z_B)$.
Sepanjang rangkaian ruas garisnya, \omterm{def:g11:work-of-force:work}{usahanya} berjumlah dan selisih
ketinggiannya \omterm{def:g10:inertia:compensate}{saling meniadakan}:
$mg(z_A - z_C) + mg(z_C - z_B) = mg(z_A - z_B)$.
\end{proof}

\begin{proposition}[Lintasan apa pun]\label{prop:g11:work-of-force:anypath}
Rumus yang sama berlaku sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} melengkung apa pun dari $A$
ke $B$.
\end{proposition}
\admitted

\begin{remark}\label{rem:g11:work-of-force:anypath}
Masuk akal --- sebuah lengkung sedekat yang kita kehendaki dengan garis
patah yang tersusun dari banyak ruas pendek --- tetapi menjujurkan
``sedekat yang kita kehendaki'' adalah kalkulus, yang dikerjakan pada
\cref{ch:g12:work-and-energy}.
\end{remark}

\begin{example}[Pendaki]\label{ex:g11:work-of-force:hiker}
Seorang pendaki \qty{78}{kg} (beserta ranselnya) mendaki \qty{450}{m},
lewat jalur mana pun:
$W(\vect P) = -78 \times 9.81 \times 450 \approx \qty{-344}{kJ}$, dan
tepat $+\qty{344}{kJ}$ ketika turun kembali, seberapa pun berkeloknya.
\end{example}

\section{Usaha oleh gesekan}

\begin{proposition}[Usaha oleh gesekan peluncuran]\label{prop:g11:work-of-force:friction}
Sebuah \omterm{def:g10:inertia:inventory}{gaya gesek} yang besarnya tetap $f$, yang pada setiap saat
berlawanan dengan geraknya, melakukan \omterm{def:g11:work-of-force:work}{usaha} $W = -fL$ sepanjang
\omterm{def:g10:relative-motion:trajectory}{lintasan} yang panjang seluruhnya $L$: selalu menghambat, dan sebanding
dengan \emph{panjang \omterm{def:g10:relative-motion:trajectory}{lintasan} yang ditempuh} --- bukan dengan
perpindahannya.
\end{proposition}

\begin{proof}
Tiap potongan lurus yang pendek memiliki \omterm{def:g10:inertia:force}{gayanya} pada
$\qty{180}{\degree}$ terhadap geraknya, sehingga menyumbang
$-f \times (\text{panjangnya})$; potongannya berjumlah menjadi $-fL$.
\omterm{def:g10:relative-motion:trajectory}{Lintasan} yang melengkung dicincang menjadi potongan pendek dengan cara
yang sama, dan limitnya diterima seperti pada \omterm{def:g10:universal-gravitation:weight}{beratnya}
(\cref{rem:g11:work-of-force:anypath}).
\end{proof}

\begin{remark}[Gesekan melawan berat: pemisah yang besar]\label{rem:g11:work-of-force:divide}
\omterm{def:g11:work-of-force:work}{Usaha} oleh \omterm{def:g10:universal-gravitation:weight}{berat} melupakan \omterm{def:g10:relative-motion:trajectory}{lintasannya}; \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekan} adalah
retribusi tiap \omterm{def:g10:orders-of-magnitude:unit}{meter}. Pada perjalanan pulang pergi, \omterm{def:g10:universal-gravitation:weight}{beratnya} berjumlah
nol --- apa yang diambilnya saat mendaki dikembalikannya saat menuruni
--- sedangkan \omterm{def:g10:inertia:inventory}{gesekannya} menagih $-fL$ pada tiap penggalnya dan tidak
mengembalikan apa pun: \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:universal-gravitation:weight}{berat} dapat disimpan dan diperoleh
kembali (\omterm{def:g10:energy-conservation:energy}{energi} potensial, \cref{ch:g11:mechanical-energy}), sedangkan
\omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekan} hilang sebagai kalor.
\end{remark}

\begin{example}[Pergi dan kembali]\label{ex:g11:work-of-force:roundtrip}
Sebuah peti meluncur \qty{5.0}{m} melintasi lantai lalu \qty{5.0}{m}
kembali, melawan $f = \qty{60}{N}$: \omterm{def:g10:inertia:inventory}{gesekannya} melakukan
$-60 \times 10 =
\qty{-600}{J}$ walaupun peti itu berakhir di tempat ia
bermula. \omterm{def:g10:universal-gravitation:weight}{Berat} dan \omterm{def:g10:inertia:inventory}{gaya normalnya}, yang tegak lurus geraknya: nol
sepanjang perjalanan.
\end{example}

\begin{remark}[Bagaimana dengan pegas?]\label{rem:g11:work-of-force:spring}
Sebuah pegas menarik makin kuat ketika ia makin diregangkan: \omterm{def:g10:inertia:force}{gayanya}
tidak tetap dan \cref{def:g11:work-of-force:work} tidak berlaku.
\omterm{def:g11:work-of-force:work}{Usahanya} berarti mencincang regangannya menjadi langkah yang sangat
kecil lalu menjumlahkannya --- yaitu kalkulus, yang diserahkan pada
\cref{ch:g12:work-and-energy}.
\end{remark}

\section{Daya}

\begin{definition}[Daya]\label{def:g11:work-of-force:power}
Yang disebut \emph{daya}\index{daya!mekanis} sebuah \omterm{def:g10:inertia:force}{gaya} adalah laju \omterm{def:g10:inertia:force}{gaya}
itu melakukan \omterm{def:g11:work-of-force:work}{usaha}, $P = W/\Delta t$, dalam \emph{watt}\index{watt}
($\unit{W} = \unit{J/s}$) --- watt yang sama seperti pada
\cref{ch:g11:circuits-and-power}. \omterm{def:g10:orders-of-magnitude:unit}{Satuan} sebelum SI masih bertahan pada
lencana mesin: \emph{daya kuda}\index{daya kuda},
$\qty{1}{hp} = \qty{736}{W}$.
\end{definition}

\begin{proposition}[Daya pada kecepatan tetap]\label{prop:g11:work-of-force:pfv}
Jika bendanya bergerak dengan kelajuan tetap $v$, \omterm{def:g10:inertia:force}{gaya} tetap $\vect F$
yang bersudut $\theta$ terhadap geraknya menyerahkan \omterm{def:g11:work-of-force:power}{daya}
$P = F v \cos\theta$ --- khususnya $P = Fv$ untuk \omterm{def:g10:inertia:force}{gaya} yang searah
geraknya.
\end{proposition}

\begin{proof}
Dalam waktu $\Delta t$ bendanya menempuh $d = v\,\Delta t$, sehingga
$W = F \times v\,\Delta t \times \cos\theta$; bagilah dengan
$\Delta t$.
\end{proof}

\begin{example}[Derek]\label{ex:g11:work-of-force:crane}
Sebuah derek mengangkat palet \qty{600}{kg} dengan laju tetap
\qty{0.90}{m/s}: \omterm{def:g10:inertia:force}{gaya} kabelnya sama dengan \omterm{def:g10:universal-gravitation:weight}{beratnya},
$mg = \qty{5.89e3}{N}$, sehingga
$P = Fv = 5890 \times 0.90 \approx \qty{5.3}{kW} \approx \qty{7.2}{hp}$
--- setara tujuh ekor kuda di dalam kotak baja.
\end{example}

\begin{method}[Mendaki dengan gaya yang lebih kecil: berkelok]\label{met:g11:work-of-force:zigzag}
Untuk menaikkan sebuah beban setinggi $h$ dengan \omterm{def:g10:inertia:force}{gaya} yang terbatas:
\begin{enumerate}
  \item Tagihannya sudah tertetapkan: menurut
        \cref{prop:g11:work-of-force:weight}, rute mana pun ke puncaknya
        berharga \omterm{def:g11:work-of-force:work}{usaha} sebesar $mgh$.
  \item Panjangkanlah \omterm{def:g10:relative-motion:trajectory}{lintasannya}: pada lereng sepanjang $L$, \omterm{def:g10:inertia:force}{gaya} yang
        diperlukan (dengan laju tetap, \omterm{def:g10:inertia:inventory}{gesekannya} dikesampingkan)
        adalah $F = mgh/L$ --- gandakan panjangnya, separuhkan \omterm{def:g10:inertia:force}{gayanya}.
  \item Bayarlah biaya tambahannya: \omterm{def:g10:inertia:inventory}{gesekan} menagih tiap \omterm{def:g10:orders-of-magnitude:unit}{meter}
        (\cref{prop:g11:work-of-force:friction}), dan menambahkan $fL$.
\end{enumerate}
Jalan gunung berkelok demi langkah 2: tinggi yang sama, \omterm{def:g10:inertia:force}{gaya} yang lebih
kecil, \omterm{def:g11:work-of-force:work}{usaha} yang sama.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:1}
Sebuah \omterm{def:g10:inertia:force}{gaya} tetap $F = \qty{40}{N}$ bekerja sepanjang perpindahan lurus
\qty{50}{m}: hitunglah \omterm{def:g11:work-of-force:work}{usahanya} untuk $\theta = \qty{0}{\degree}$,
$\qty{60}{\degree}$, $\qty{90}{\degree}$ dan $\qty{120}{\degree}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:2}
Sebuah kotak perkakas \qty{12}{kg} diturunkan \qty{2.5}{m} dari sebuah
van, lalu kemudian dinaikkan kembali. Berapa \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:universal-gravitation:weight}{beratnya} pada
masing-masing kasus?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:3}
\omterm{def:g11:work-of-force:sign}{Pendorong}, \omterm{def:g11:work-of-force:sign}{penghambat}, atau nol? (a) \omterm{def:g10:universal-gravitation:weight}{berat} sebuah apel yang jatuh;
(b) \omterm{def:g10:universal-gravitation:weight}{berat} sebuah bola yang naik; (c) \omterm{def:g10:inertia:inventory}{gaya normal} pada kotak yang
meluncur; (d) \omterm{def:g10:inertia:inventory}{gesekan} pada sepeda yang mengerem; (e) tegangan tali pada
bandul yang berayun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:4}
Sebuah derek mengangkat \qty{600}{kg} setinggi \qty{25}{m} dengan laju
tetap. Berapa \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:force}{gaya} pengangkatnya dan \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:universal-gravitation:weight}{beratnya}?
Berapa jumlah keduanya, dan mengapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:work-of-force:5}
Motor sebuah eskalator melakukan \omterm{def:g11:work-of-force:work}{usaha} \qty{90}{kJ} tiap menit. Berapa
\omterm{def:g11:work-of-force:power}{dayanya} dalam \omterm{def:g11:work-of-force:power}{watt}? Dan dalam \omterm{def:g11:work-of-force:power}{daya kuda}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:work-of-force:6}
Sebuah kereta luncur ditarik \qty{200}{m} oleh tali bertegangan
\qty{120}{N} pada $\qty{30}{\degree}$ di atas saljunya; \omterm{def:g10:inertia:inventory}{gesekannya}
\qty{45}{N}. Berapa \omterm{def:g11:work-of-force:work}{usaha} oleh masing-masing dari keempat \omterm{def:g10:inertia:force}{gayanya}
(tali, \omterm{def:g10:inertia:inventory}{gesekan}, \omterm{def:g10:universal-gravitation:weight}{berat}, normal) dan berapa jumlahnya?
\end{exercise}

\section{Soal: Sehari bersama pengangkut barang}

\begin{problem}[{Soal akhir pekan --- fisika membayar tiap joule:
sehari menggeser, mengerek dan menempuh tikungan tajam, yang berakhir
dengan penemuan yang merendahkan hati tentang pendapat sebuah ketel
terhadap kerja keras yang jujur}]
\label{pb:g11:work-of-force:1}
Sebuah regu pengangkut barang mengosongkan sebuah gudang, menaikkan
muatannya ke sebuah rumah susun di lantai tiga, lalu mengemudikan
vannya ke sebuah desa di puncak bukit. Setiap pekerjaannya dihargai
dalam \omterm{def:g11:work-of-force:work}{joule}; ketel dapurlah yang akan mengaudit hari itu. Kotaknya:
\qty{30}{kg}.

\medskip
\noindent\textbf{Bagian I --- Melintasi lantai gudang.}
Sebuah kotak didorong dengan \omterm{def:g10:inertia:force}{gaya} mendatar $F = \qty{110}{N}$ sepanjang
$d = \qty{8.0}{m}$; \omterm{def:g10:inertia:inventory}{gesekan} peluncurannya $f = \qty{95}{N}$.
\begin{enumerate}
  \item Hitunglah \omterm{def:g11:work-of-force:work}{usaha} oleh dorongannya.
  \item Hitunglah \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekannya}, oleh \omterm{def:g10:universal-gravitation:weight}{beratnya}, dan oleh \omterm{def:g10:inertia:inventory}{gaya
        normalnya} (berikan alasan bagi yang bernilai nol).
  \item Berapa \omterm{def:g11:work-of-force:work}{usaha} seluruhnya pada kotak itu, dan apa arti tandanya
        (kotak itu bermula dalam keadaan diam)? Dorongan berapa yang
        menjaga lajunya tetap (\cref{ch:g11:forces-and-motion})?
  \item Sudut yang jauh dapat dicapai sepanjang diagonal \qty{10.0}{m}
        atau menyusuri dua dinding (\qty{6.0}{m} lalu \qty{8.0}{m}).
        Berapa \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekannya} pada tiap rute?
  \item Pada perjalanan pulang pergi lewat diagonalnya (pergi dan
        kembali), hitunglah \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekannya} dan \omterm{def:g11:work-of-force:work}{usaha} oleh
        \omterm{def:g10:universal-gravitation:weight}{beratnya}. \omterm{def:g10:inertia:force}{Gaya} yang mana yang mengembalikan?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Naik ke lantai tiga ($h = \qty{9.0}{m}$).}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:universal-gravitation:weight}{berat} sebuah kotak yang dinaikkan ke rumah
        susun itu. \omterm{def:g11:work-of-force:work}{Usaha} berapa yang sekurang-kurangnya harus dipasok
        regunya?
  \item Ketika dipanggul menaiki tangga oleh pengangkut \qty{70}{kg},
        gravitasi menagih pengangkut \emph{dan} kotaknya: berapa \omterm{def:g11:work-of-force:work}{usaha}
        seluruhnya melawan gravitasi, dan berapa bagiannya yang
        sungguh dibelanjakan untuk kotaknya?
  \item Sebuah bidang miring untuk troli panjangnya \qty{30}{m} untuk
        kenaikan \qty{9.0}{m} yang sama, dengan \omterm{def:g10:inertia:inventory}{gesekan} gelinding
        \qty{40}{N}. Pada laju tetap, hitunglah dorongan yang
        diperlukan sepanjang bidang miringnya dan \omterm{def:g11:work-of-force:work}{usahanya}.
  \item Bandingkan bidang miringnya dengan pengerekan tegak lurus:
        \omterm{def:g10:inertia:force}{gaya} yang diperlukan, \omterm{def:g11:work-of-force:work}{usaha} yang dilakukan. Apa yang dibelikan
        bidang miring itu, dan dengan harga berapa?
  \item Sebuah pengerek bertali dan berkatrol menaikkan kotaknya dengan
        \qty{0.50}{m/s}. Berapa lamanya dan berapa \omterm{def:g11:work-of-force:power}{daya} mekanisnya?
  \item Perolehlah kembali \omterm{def:g11:work-of-force:power}{daya} itu sebagai $P = Fv$. Motor pengereknya
        berefisiensi $60\%$: berapa \omterm{prop:g11:circuits-and-power:power}{daya listrik} yang ditariknya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Jalan yang berkelok.}
Van yang bermuatan (\qty{3500}{kg}) harus mencapai sebuah desa
\qty{240}{m} di atas lembahnya: jalur lurus, \qty{1.0}{km} bertanjakan
$24\%$, atau jalan beraspal, \qty{8.0}{km} bertanjakan $3.0\%$
(tanjakan $= \sin\alpha$).
\begin{enumerate}[resume]
  \item Periksalah bahwa kedua rutenya mendaki \qty{240}{m} yang sama.
  \item Pada laju tetap dan dengan mengabaikan \omterm{def:g10:inertia:inventory}{gesekan}, hitunglah \omterm{def:g10:inertia:force}{gaya}
        penggerak yang diperlukan sepanjang tiap rutenya.
  \item Berapa \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:force}{gaya} penggeraknya pada tiap rute? Bandingkan
        dengan $mgh$.
  \item \omterm{def:g11:circuits-and-power:resistance}{Hambatan} gelindingnya \qty{400}{N} pada keduanya: hitunglah
        \omterm{def:g11:work-of-force:work}{usaha} tambahannya pada tiap rute. Berapa harga kenyamanan
        jalan yang berkelok itu?
  \item Pada \qty{36}{km/h}, hitunglah \omterm{def:g11:work-of-force:power}{daya} mesin yang diperlukan pada
        tiap rute (dalam \unit{kW} dan \unit{hp}). Mengapa jalan
        berkelok?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Buku besar hari itu.}
\begin{enumerate}[resume]
  \item Regunya menggeser $60$ kotak melintasi lantai (dengan dorongan
        pada pertanyaan 1) lalu mengerek seluruh $60$ kotak itu setinggi
        \qty{9.0}{m} (pertanyaan 6). Berapa \omterm{def:g11:work-of-force:work}{usaha} mekanis seluruhnya
        yang diserahkan kepada kotaknya?
  \item Tambahkan tubuh si pemanggul tangga sendiri: $20$ perjalanan
        naik dengan \qty{70}{kg} (pengembalian saat menuruni jatuh ke
        lutut yang panas, bukan ke buku besarnya). Berapa jumlah
        seluruhnya untuk hari itu?
  \item Regunya bekerja $8.0$ jam: berapa \omterm{def:g11:work-of-force:power}{daya} mekanis rata-ratanya?
        Berapa lama ketel \qty{2.0}{kW} membelanjakan \omterm{def:g10:energy-conservation:energy}{energi} yang sama?
  \item Penutup: dalam dua kalimat, nyatakan apa yang dibayar \omterm{def:g11:work-of-force:work}{joule} ---
        dan apa yang tidak dibayarnya (dinding yang didorong sepanjang
        sore, kotak yang ditahan tak bergerak) --- serta mengapa
        pengangkut barang memiliki derek gulung, bidang miring dan
        jalan berkelok, dan bukan otot yang lebih besar.
\end{enumerate}
\end{problem}
