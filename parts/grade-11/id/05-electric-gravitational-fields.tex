\chapter{Medan Listrik dan Medan Gravitasi}\label{ch:g11:electric-gravitational-fields}

Bagaimana Bumi menarik sebuah apel yang tidak pernah disentuhnya? Newton
sendiri menyebut pengaruh dari kejauhan sebagai ``kemustahilan yang
begitu besar'' sehingga ia menolak membelanya. Jawaban modernnya: sebuah
massa, atau sebuah muatan, mengubah ruang di sekelilingnya --- ia
menciptakan sebuah \emph{\omterm{def:g11:electric-gravitational-fields:field}{medan}} --- dan apa pun yang ditempatkan di sana
menanggapi \omterm{def:g11:electric-gravitational-fields:field}{medan} itu di kedudukannya sendiri. Kita membangun kedua
contoh besarnya berdampingan, sampai keserupaannya menjadi kesamaan
bentuk.

\section{Dari pengaruh jarak jauh ke medan}

\begin{definition}[Medan]\label{def:g11:electric-gravitational-fields:field}
Sebuah \emph{medan}\index{medan} memasangkan sebuah vektor pada setiap
titik ruang. Sebuah sumber (muatan, massa) menciptakan medannya di mana
pun di sekelilingnya, ada atau tidak ada sesuatu di sana yang
merasakannya; sebuah benda uji lalu merasakan \omterm{def:g10:inertia:force}{gaya} yang ditentukan oleh
medan \emph{di kedudukannya sendiri} saja. Gambaran dua langkah ini
menggantikan pengaruh dari kejauhan.
\end{definition}

\begin{remark}
\omterm{def:g11:electric-gravitational-fields:field}{Medan} bukan muslihat pembukuan: \omterm{def:g11:electric-gravitational-fields:field}{medan} membawa \omterm{def:g10:energy-conservation:energy}{energi}, dan perubahan pada
sumbernya merambat ke luar melaluinya dengan laju yang berhingga ---
cahaya sendiri adalah \omterm{def:g11:electric-gravitational-fields:field}{medan} yang merambat semacam itu
(\cref{ch:g12:light-as-wave}). Di sini kita menelaah \omterm{def:g11:electric-gravitational-fields:field}{medan} yang tidak
berubah terhadap waktu.
\end{remark}

\section{Medan listrik}

\begin{definition}[Medan listrik]\label{def:g11:electric-gravitational-fields:efield}
Jika sebuah muatan uji kecil $q$ yang ditempatkan pada titik $M$
merasakan \omterm{def:g10:inertia:force}{gaya} listrik $\vect F$, maka \emph{medan
listrik}\index{medan listrik} di $M$ adalah
\[
\vect E = \frac{\vect F}{q}
\qquad \text{(dalam \unit{N/C}, ditulis juga \unit{V/m})}.
\]
$\vect E$ hanya bergantung pada sumbernya dan pada $M$, bukan pada
muatan ujinya: menggandakan $q$ menggandakan $\vect F$, sehingga
hasil baginya tidak berubah.
\end{definition}

\begin{proposition}[Gaya pada sebuah muatan]\label{prop:g11:electric-gravitational-fields:force}
Muatan $q$ yang ditempatkan di tempat \omterm{def:g11:electric-gravitational-fields:field}{medannya} $\vect E$ merasakan \omterm{def:g10:inertia:force}{gaya}
$\vect F = q \vect E$: searah $\vect E$ jika $q > 0$, berlawanan dengan
$\vect E$ jika $q < 0$, dengan besar $F = \abs{q}\,E$.
\end{proposition}

\begin{proof}
Susunlah ulang definisinya; aturan tandanya adalah aljabar sebuah
kelipatan negatif.
\end{proof}

\begin{example}[Membaca sebuah medan]\label{ex:g11:electric-gravitational-fields:reading}
Muatan uji $q = \qty{+2.0e-8}{C}$ di $M$ merasakan \omterm{def:g10:inertia:force}{gaya}
\qty{6.0e-4}{N} yang menunjuk ke timur. \omterm{def:g11:electric-gravitational-fields:field}{Medan} di $M$ adalah
$E = \num{6.0e-4} / \num{2.0e-8} = \qty{3.0e4}{N/C}$, menunjuk ke
timur; $q' = \qty{-1.0e-8}{C}$ yang ditempatkan di titik yang sama
merasakan $F = \num{1.0e-8} \times \num{3.0e4} = \qty{3.0e-4}{N}$ ---
menunjuk ke barat.
\end{example}

\begin{proposition}[Medan sebuah muatan titik]\label{prop:g11:electric-gravitational-fields:point}
Sebuah muatan titik $Q$ menciptakan, pada jarak $d$, sebuah \omterm{def:g11:electric-gravitational-fields:field}{medan}
sepanjang garis penghubung muatannya dengan titiknya --- menjauhi $Q$
jika $Q > 0$, menuju muatan itu jika $Q < 0$ --- dengan besar
\[
E = k\,\frac{\abs{Q}}{d^2},
\qquad k = \qty{8.99e9}{N.m^2/C^2}.
\]
\end{proposition}

\begin{proof}
Hukum Coulomb (\cref{ch:g11:fundamental-interactions}) memberikan \omterm{def:g10:inertia:force}{gaya}
pada muatan uji $q$ pada jarak $d$: besarnya $k \abs{qQ} / d^2$,
sepanjang garis penghubungnya, tolak-menolak untuk tanda yang sejenis.
Bagilah dengan $q$ lalu terapkan
\cref{prop:g11:electric-gravitational-fields:force}.
\end{proof}

\begin{example}[Bola bermuatan]\label{ex:g11:electric-gravitational-fields:sphere}
Sebuah bola Van de Graaff membawa $Q = \qty{+5.0e-7}{C}$ (bola kecil
yang bermuatan berlaku seperti muatan titik di pusatnya). Pada
$d = \qty{0.50}{m}$:
\[
E = \num{8.99e9} \times \num{5.0e-7} / 0.50^2 \approx \qty{1.8e4}{N/C},
\]
yang menunjuk menjauhi bolanya secara radial. Pada \qty{1.0}{m} nilainya
sudah turun menjadi seperempatnya: $1/d^2$ milik \omterm{def:g10:inertia:force}{gayanya} bertahan pada
\omterm{def:g11:electric-gravitational-fields:field}{medannya}.
\end{example}

\begin{definition}[Garis medan]\label{def:g11:electric-gravitational-fields:lines}
Sebuah \emph{garis medan}\index{garis medan} adalah kurva yang di mana
pun menyinggung \omterm{def:g11:electric-gravitational-fields:field}{medannya}, dengan anak panah yang memberikan arah
\omterm{def:g11:electric-gravitational-fields:field}{medannya}. Ada tiga aturannya:
\begin{itemize}
  \item garis \omterm{def:g11:electric-gravitational-fields:field}{medan} meninggalkan muatan positif dan tiba pada muatan
        negatif;
  \item di tempat garisnya berdesakan \omterm{def:g11:electric-gravitational-fields:field}{medannya} kuat; di tempat garisnya
        merenggang, \omterm{def:g11:electric-gravitational-fields:field}{medannya} lemah;
  \item dua garis \omterm{def:g11:electric-gravitational-fields:field}{medan} tidak pernah berpotongan: \omterm{def:g11:electric-gravitational-fields:field}{medannya} memiliki
        satu arah pada tiap titik.
\end{itemize}
\end{definition}

\section{Medan seragam sebuah kapasitor}

\begin{definition}[Kapasitor keping sejajar]\label{def:g11:electric-gravitational-fields:capacitor}
Dua keping logam sejajar yang saling berhadapan, berdekatan dan membawa
muatan yang berlawanan, membentuk sebuah
\emph{kapasitor}\index{kapasitor}. Di antara kedua kepingnya \omterm{def:g11:electric-gravitational-fields:field}{medannya}
\emph{seragam}\index{medan seragam}: besar dan arahnya sama di mana pun
--- tegak lurus kepingnya, dari keping yang positif ke keping yang
negatif.
\end{definition}

\begin{proposition}[Medan di antara kedua keping]\label{prop:g11:electric-gravitational-fields:uniform}
Jika \omterm{def:g11:circuits-and-power:voltage}{tegangan} antara kedua kepingnya
(\cref{ch:g11:circuits-and-power}) adalah $U$ dan jarak keduanya $d$,
maka \omterm{def:g11:electric-gravitational-fields:field}{medan} di antara keduanya berbesar
\[
E = \frac{U}{d} \qquad \text{(dalam \unit{V/m})}.
\]
\end{proposition}
\admitted

\begin{remark}
Keseragamannya dan $E = U/d$ diturunkan secara jujur pada jilid Tahun
ke-1; rumus itu menjelaskan \omterm{def:g10:orders-of-magnitude:unit}{satuan} \unit{V/m}. Celah yang \omterm{def:g11:lenses-and-eye:lens}{tipis}
menghasilkan \omterm{def:g11:electric-gravitational-fields:field}{medan} yang kuat: \qty{100}{V} pada jarak satu milimeter
sudah menjadi \qty{e5}{V/m}.
\end{remark}

\begin{example}[Mengemudikan berkas elektron]\label{ex:g11:electric-gravitational-fields:beam}
Di dalam \omterm{def:g10:signals-and-waves:oscilloscope}{osiloskop}, berkasnya melewati keping dengan
$U = \qty{100}{V}$ dan $d = \qty{5.0}{mm}$:
$E = 100 / \num{5.0e-3} = \qty{2.0e4}{V/m}$, dan tiap elektron
merasakan $F = \num{1.60e-19} \times \num{2.0e4} = \qty{3.2e-15}{N}$ ke
arah keping yang positif --- $\num{4e14}$ kali \omterm{def:g10:universal-gravitation:weight}{beratnya} yang
\qty{8.9e-30}{N}, sehingga berkasnya membelok secara kasatmata
(\omterm{def:g10:relative-motion:trajectory}{lintasannya} yang melengkung menjadi urusan
\cref{ch:g12:newtons-laws}).
\end{example}

\section{Medan gravitasi}

\begin{definition}[Medan gravitasi]\label{def:g11:electric-gravitational-fields:gfield}
Jika massa uji $m$ yang ditempatkan pada sebuah titik merasakan \omterm{def:g10:universal-gravitation:force}{gaya
gravitasi} (yaitu sebuah \omterm{def:g10:universal-gravitation:weight}{berat}) $\vect P$, maka \emph{medan
gravitasi}\index{medan gravitasi} di sana adalah
\[
\vect g = \frac{\vect P}{m} \qquad \text{(dalam \unit{N/kg})},
\]
dan sebaliknya $\vect P = m \vect g$. Bab terdahulu memperkenalkan
bilangan $g$; $\vect g$ menambahkan arah mulai bergeraknya sebuah batu
yang dijatuhkan.
\end{definition}

\begin{proposition}[Medan Bumi]\label{prop:g11:electric-gravitational-fields:earth}
Pada jarak $d$ dari pusat Bumi (yang bermassa $M$), \omterm{def:g11:electric-gravitational-fields:gfield}{medan gravitasinya}
menunjuk ke pusatnya dan berbesar
\[
g = G\,\frac{M}{d^2},
\qquad G = \qty{6.67e-11}{N.m^2/kg^2}.
\]
\end{proposition}

\begin{proof}
Gravitasi semesta (\cref{ch:g10:universal-gravitation}) memberikan \omterm{def:g10:inertia:force}{gaya}
pada massa uji $m$: besarnya $G M m / d^2$, menuju pusatnya; bagilah
dengan $m$. Di permukaannya, $d = R$ dan inilah
$g = GM/R^2 \approx \qty{9.81}{N/kg}$ pada bab terdahulu.
\end{proof}

\begin{example}[Medan di tempat Bulan tinggal]\label{ex:g11:electric-gravitational-fields:moon}
Pada jarak Bulan $d = \qty{3.84e8}{m}$:
\[
g = \num{6.67e-11} \times \num{5.97e24} / (\num{3.84e8})^2
\approx \qty{2.7e-3}{N/kg}.
\]
\omterm{def:g11:electric-gravitational-fields:field}{Medan} Bumi tidak pernah berhenti --- \omterm{def:g11:electric-gravitational-fields:field}{medan} itu hanya menipis sebagai
$1/d^2$; inilah \omterm{def:g11:electric-gravitational-fields:field}{medan} yang menahan Bulan pada \omterm{def:g10:universal-gravitation:satellite}{orbitnya}.
\end{example}

\begin{remark}[Seragam di dekat permukaan]\label{rem:g11:electric-gravitational-fields:uniformg}
Sepanjang sebuah laboratorium atau bahkan sebuah gunung, $d$ nyaris
tidak berubah dibandingkan $R = \qty{6.37e6}{m}$, sehingga $\vect g$
\omterm{def:g11:electric-gravitational-fields:capacitor}{seragam} dengan ketelitian yang tinggi: garisnya tegak, tetap
\qty{9.81}{N/kg} --- \omterm{def:g11:electric-gravitational-fields:field}{medan} Bumi tampak seperti \omterm{def:g11:electric-gravitational-fields:field}{medan} \omterm{def:g11:electric-gravitational-fields:capacitor}{kapasitornya},
sebagaimana Bumi yang bulat tampak datar dari sebuah kebun.
\end{remark}

\section{Satu gagasan, dua gaya}

Kedua \omterm{def:g11:electric-gravitational-fields:field}{medan} pada bab ini adalah matematika yang sama dalam dua busana:

\begin{center}
\begin{tabular}{lcc}
 & \textbf{listrik} & \textbf{gravitasi} \\
\hline
sumber & muatan $Q$ & massa $M$ \\
\omterm{def:g11:electric-gravitational-fields:field}{medan} & $\vect E = \vect F / q$ & $\vect g = \vect P / m$ \\
\omterm{def:g10:orders-of-magnitude:unit}{satuan} & \unit{N/C} = \unit{V/m} & \unit{N/kg} \\
sumber titik & $E = k \abs{Q} / d^2$ & $g = G M / d^2$ \\
tetapan & $k = \qty{8.99e9}{N.m^2/C^2}$ & $G = \qty{6.67e-11}{N.m^2/kg^2}$ \\
\omterm{def:g10:inertia:force}{gaya} pada benda uji & $\vect F = q \vect E$ & $\vect P = m \vect g$ \\
\omterm{def:g11:electric-gravitational-fields:lines}{garis medan} & keluar dari $+$, masuk ke $-$ & selalu masuk ke massanya \\
\end{tabular}
\end{center}

Setiap rumus di sebelah kiri menjadi rumus di sebelah kanan menurut
kamus $q \leftrightarrow m$, $\vect E \leftrightarrow \vect g$, $k \leftrightarrow G$.

\begin{remark}[Di mana keserupaannya patah]\label{rem:g11:electric-gravitational-fields:noanalogue}
Muatan hadir dalam dua tanda; massa hanya dalam satu. Karena itu \omterm{def:g11:electric-gravitational-fields:efield}{medan
listrik} dapat dihapuskan: muatan yang tersusun ulang pada kotak logam
menghapus \omterm{def:g11:electric-gravitational-fields:field}{medan} luar apa pun di dalamnya (sebuah \emph{sangkar Faraday}
--- sebabnya lift mematikan \omterm{def:g10:signals-and-waves:signal}{sinyal} teleponmu), dan materi dalam jumlah
besar, dengan $+$ dan $-$ yang setimbang, bersifat bisu secara listrik.
Gravitasi tidak dapat ditudungi dan tidak dapat dinetralkan: tiap
\omterm{def:g10:orders-of-magnitude:unit}{kilogram} menambahkan tarikannya --- dan itulah sebabnya \omterm{def:g10:inertia:force}{gaya} yang
$\num{e39}$ kali lebih lemah
(\cref{exo:g11:electric-gravitational-fields:10}) justru menjalankan
alam semesta pada skala yang besar.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:1}
Sebuah muatan uji $q = \qty{+2.0e-6}{C}$ pada titik $M$ merasakan \omterm{def:g10:inertia:force}{gaya}
\qty{5.0e-3}{N} yang menunjuk ke utara. Sebutkan \omterm{def:g11:electric-gravitational-fields:field}{medan} di $M$ (besar dan
arahnya), lalu \omterm{def:g10:inertia:force}{gaya} pada $q' = \qty{-4.0e-6}{C}$ yang ditempatkan di
$M$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:2}
Sebuah muatan titik $Q = \qty{+5.0}{nC}$ berada di $O$.
\begin{enumerate}
  \item Hitunglah \omterm{def:g11:electric-gravitational-fields:field}{medannya} pada $d = \qty{30}{cm}$: besar dan arahnya.
  \item Pada jarak berapa \omterm{def:g11:electric-gravitational-fields:field}{medannya} tinggal separuhnya?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:3}
Sebuah \omterm{def:g11:electric-gravitational-fields:capacitor}{kapasitor} memiliki $U = \qty{600}{V}$ pada keping yang berjarak
\qty{3.0}{mm}. Hitunglah \omterm{def:g11:electric-gravitational-fields:field}{medannya}, lalu \omterm{def:g10:inertia:force}{gaya} pada satu elektron di sana
($e = \qty{1.60e-19}{C}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:4}
Di Mars, sebuah wahana \qty{250}{kg} \omterm{def:g10:universal-gravitation:weight}{beratnya} \qty{930}{N}. Hitunglah
\omterm{def:g11:electric-gravitational-fields:field}{medan} di permukaan Mars $g_{\text{Mars}}$, lalu \omterm{def:g10:universal-gravitation:weight}{berat} seorang
antariksawan \qty{80}{kg} di sana.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:electric-gravitational-fields:5}
Benar atau salah, masing-masing dengan satu alasan: (a) dua \omterm{def:g11:electric-gravitational-fields:lines}{garis medan}
boleh berpotongan di tempat \omterm{def:g11:electric-gravitational-fields:field}{medannya} kuat; (b) \omterm{def:g11:electric-gravitational-fields:lines}{garis medan} meninggalkan
muatan negatif dan tiba pada muatan positif; (c) \omterm{def:g11:electric-gravitational-fields:lines}{garis medan} yang
berdesakan menandakan \omterm{def:g11:electric-gravitational-fields:field}{medan} yang kuat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:6}
Stasiun antariksa internasional mengorbit pada ketinggian \qty{420}{km}
($R = \qty{6.37e6}{m}$, $M = \qty{5.97e24}{kg}$).
\begin{enumerate}
  \item Hitunglah $g$ di stasiun itu; berapa bagian dari nilainya di
        permukaan?
  \item Antariksawan di sana melayang. Damaikan hal itu dengan
        jawabanmu dalam satu kalimat.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:7}
\omterm{def:g11:electric-gravitational-fields:field}{Medan} berbesar berapa yang akan mengimbangi \omterm{def:g10:universal-gravitation:weight}{berat} sebuah elektron
($m_e = \qty{9.11e-31}{kg}$)? Bandingkan dengan \qty{2.0e4}{V/m} pada
\cref{ex:g11:electric-gravitational-fields:beam}; simpulkan tentang
tempat gravitasi di dalam fisika elektron.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:8}
Sebutir debu bermassa \qty{2.0e-6}{kg} melayang diam di antara keping
mendatar yang berjarak \qty{2.0}{cm} dengan $U = \qty{5.0}{kV}$ pada
keduanya.
\begin{enumerate}
  \item Hitunglah \omterm{def:g11:electric-gravitational-fields:field}{medannya}, lalu besar muatan butir debu itu.
  \item Muatan butir debu itu positif: keping yang mana yang positif?
  \item Berapa \omterm{def:g11:fundamental-interactions:elementary}{muatan dasar} yang dibawa butir debu itu?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:electric-gravitational-fields:9}
Muatan $+Q$ dan $+4Q$ berjarak \qty{30}{cm}. Di titik mana pada ruas

\section{Soal: Tetes minyak Millikan}

\begin{problem}[{Soal akhir pekan --- menimbang muatan elektron: setetes
minyak yang diparkir di udara di antara dua keping menyingkapkan bahwa
muatan hadir dalam langkah tak terbagi sebesar $\num{1.60e-19}$
\omterm{def:g11:fundamental-interactions:charge}{coulomb}}]
\label{pb:g11:electric-gravitational-fields:1}
Pada tahun 1909, Robert Millikan menyemprotkan tetes minyak di antara
keping mendatar sebuah \omterm{def:g11:electric-gravitational-fields:capacitor}{kapasitor}. Dengan menala \omterm{def:g11:circuits-and-power:voltage}{tegangannya} sampai
sebuah tetes yang dipilihnya tergantung tanpa bergerak --- \omterm{def:g10:inertia:force}{gaya}
listriknya tepat mengimbangi \omterm{def:g10:universal-gravitation:weight}{beratnya} --- ia dapat menimbang muatan
tetes itu. Tetes kita bermassa $m = \qty{4.90e-15}{kg}$ (terukur dari
jatuhnya yang lambat ketika \omterm{def:g11:electric-gravitational-fields:field}{medannya} dimatikan; kita menerimanya sebagai
diketahui); kepingnya berjarak $d = \qty{8.0}{mm}$, dan keping
\emph{atas}nya positif; muatan tetesnya negatif.

\medskip
\noindent\textbf{Bagian I --- Panggungnya.}
\begin{enumerate}
  \item Lukiskan \omterm{def:g11:electric-gravitational-fields:field}{medan} di antara kedua kepingnya; apa yang dibelikan
        kata ``\omterm{def:g11:electric-gravitational-fields:capacitor}{seragam}''?
  \item Untuk $U = \qty{800}{V}$, hitunglah $E$.
  \item Ke arah mana \omterm{def:g10:inertia:force}{gaya} listrik pada tetes itu? Mengapa keping
        \emph{atas}nya harus menjadi yang positif?
  \item Hitunglah \omterm{def:g10:universal-gravitation:weight}{berat} tetes itu.
  \item Tulislah syarat diamnya sebagai kesamaan dua besar \omterm{def:g10:inertia:force}{gaya}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Keseimbangannya.}
\begin{enumerate}[resume]
  \item Dari syarat itu, nyatakan $\abs{q}$ dalam $m$, $g$, $d$ dan $U$.
  \item Tetes itu tergantung diam untuk $U = \qty{800}{V}$: hitunglah
        $\abs{q}$.
  \item Bagilah dengan $e = \qty{1.60e-19}{C}$. Berapa kelebihan
        elektron yang dibawa tetes itu?
  \item \omterm{def:g11:circuits-and-power:voltage}{Tegangannya} dinaikkan sedikit menjadi \qty{900}{V}. Ke arah
        mana \omterm{def:g10:inertia:force}{gaya} totalnya kini menunjuk?
  \item Tunjukkan bahwa \omterm{def:g11:circuits-and-power:voltage}{tegangan} pengimbangnya $U = mgd/\abs{q}$: makin
        besar muatannya, makin kecil \omterm{def:g11:circuits-and-power:voltage}{tegangannya}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Langkah-langkahnya.}
Semburan radiasi dapat memukulkan elektron ke atas atau ke luar dari
tetes itu; sesudah tiap perubahan, Millikan menala ulang \omterm{def:g11:circuits-and-power:voltage}{tegangannya}
untuk mengimbangi.
\begin{enumerate}[resume]
  \item Setelah satu semburan, keseimbangannya menuntut
        $U = \qty{1200}{V}$. Hitunglah $\abs{q}$ yang baru. Apakah tetes
        itu memperoleh atau kehilangan satu elektron?
  \item Keseimbangan yang berturut-turut ditemukan pada $U = 1200$,
        $800$, $600$, $480$ dan \qty{400}{V}. Hitunglah kelima
        muatannya.
  \item Tunjukkan bahwa kelimanya merupakan kelipatan bulat satu
        besaran; sebutkan nilainya.
  \item Di antara dua keseimbangan yang berurutan, sebanyak apa
        muatannya berubah? Tafsirkanlah.
  \item Jelaskan mengapa tetes ini tidak akan pernah dapat tergantung
        diam pada $U = \qty{700}{V}$, sesabar apa pun penalaannya.
  \item Millikan mengulanginya pada ratusan tetes: setiap muatan yang
        terukur merupakan kelipatan bulat dari nilai yang sama.
        Nyatakan kesimpulannya, dan sebutkan besaran yang terukur itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Mengapa tidak ada Millikan gravitasi.}
\begin{enumerate}[resume]
  \item Pada kamus bab ini, apa yang memainkan peran $q$, $E$ dan $qE$
        pada keseimbangan Bagian II?
  \item Dapatkah sebuah massa di atasnya menahan tetes itu sebagai
        gantinya? Hitunglah \omterm{def:g11:electric-gravitational-fields:field}{medan} sebuah bola timbal \qty{1000}{kg} yang
        berada \qty{1.0}{m} di atas tetes itu, \omterm{def:g10:inertia:force}{gayanya} pada tetes itu,
        lalu bandingkan dengan \omterm{def:g10:universal-gravitation:weight}{beratnya}.
  \item Keseimbangannya memerlukan tarikan ke atas yang cukup kuat
        untuk melawan seluruh Bumi. Mengapa sebuah keping bermuatan
        sanggup menyediakannya sedangkan tidak ada susunan massa yang
        sanggup?
  \item \emph{Massa} tetes itu pun berubah dalam langkah (satu \omterm{def:g11:fundamental-interactions:ladder}{molekul}
        minyak: sekitar \qty{5e-25}{kg}). Bandingkan lompatan relatif
        pada $m$ dan pada $\abs{q}$ ketika satu elektron mendarat, lalu
        simpulkan: mengapa keseimbangan ini melihat \omterm{def:g11:fundamental-interactions:ladder}{atom} kelistrikan
        tetapi tidak melihat \omterm{def:g11:fundamental-interactions:ladder}{molekul} minyaknya? Puncak ceritanya: muatan
        terkuantisasi, dan \omterm{def:g11:fundamental-interactions:ladder}{atomnya} adalah $e = \qty{1.60e-19}{C}$.
\end{enumerate}
\end{problem}
