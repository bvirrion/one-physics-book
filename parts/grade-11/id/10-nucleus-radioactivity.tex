\chapter{Inti Atom dan Keradioaktifan}\label{ch:g11:nucleus-radioactivity}

Saat ini juga, sekitar delapan ribu \omterm{def:g11:fundamental-interactions:ladder}{inti atom} meledak di dalam tubuhmu tiap
sekon. Tidak ada yang menyerangnya: inti tertentu memang lahir tidak stabil,
dan masing-masing cepat atau lambat mengubah dirinya sendiri sambil
melemparkan sebuah serpihan pada sebagian besar laju cahaya. Bab ini kembali
ke lantai $10^{-15}$ \omterm{def:g10:orders-of-magnitude:unit}{meter} pada
\cref{ch:g11:fundamental-interactions} untuk melihat inti yang mana yang
rapuh, bagaimana inti itu berkeping, dan bagaimana ketidaksabarannya yang
teratur menanggali gua serta menjaga langit-langit.

\section{Inti atom, nukleon dan isotop}

\begin{definition}[Notasi inti atom]\label{def:g11:nucleus-radioactivity:notation}
Sebuah \omterm{def:g11:fundamental-interactions:ladder}{inti atom} memuat $Z$ proton (masing-masing bermuatan $+e$) dan $N$
neutron (tanpa muatan), yaitu $A = Z + N$ \omterm{def:g11:fundamental-interactions:ladder}{nukleon} seluruhnya; inti itu
ditulis ${}^{A}_{Z}\mathrm{X}$, dengan $\mathrm X$ adalah lambang kimianya,
$Z$ adalah \emph{nomor atom}\index{nomor atom} dan $A$ adalah \emph{nomor
massa}\index{nomor massa}. Jenis yang ditentukan oleh $A$ dan $Z$ disebut
\emph{nuklida}\index{nuklida}; nuklida yang $Z$-nya sama tetapi $A$-nya
berbeda merupakan \emph{isotop}\index{isotop} satu unsur yang sama: kimianya
sama (awan elektronnya hanya melihat $Ze$), inti dan kestabilannya berbeda.
\end{definition}

\begin{example}[Hidrogen dan karbon]\label{ex:g11:nucleus-radioactivity:hc}
Hidrogen memiliki tiga \omterm{def:g11:nucleus-radioactivity:notation}{isotop}: ${}^{1}_{1}\mathrm{H}$ (sebuah proton
tunggal), ${}^{2}_{1}\mathrm{H}$ (deuterium: satu proton, satu neutron), dan
${}^{3}_{1}\mathrm{H}$ yang tidak stabil (tritium: dua neutron). Karbon
alami: sebagian besar ${}^{12}_{6}\mathrm{C}$ ($6$ proton, $6$ neutron),
$1.1\%$ berupa ${}^{13}_{6}\mathrm{C}$, dan satu \omterm{def:g11:fundamental-interactions:ladder}{atom} dari tiap $10^{12}$
berupa ${}^{14}_{6}\mathrm{C}$ yang tidak stabil ($8$ neutron), sang jagoan
pada \cref{ex:g11:nucleus-radioactivity:dating}.
\end{example}

\section{Inti atom yang stabil dan yang tidak stabil}

\begin{proposition}[Lembah kestabilan]\label{prop:g11:nucleus-radioactivity:valley}
\omterm{def:g11:nucleus-radioactivity:notation}{Nuklida} yang stabil menempati pita yang sempit pada bidang $(Z, N)$:
$N \approx Z$ untuk inti yang ringan, lalu kelebihan neutron yang makin
besar, sampai $N \approx 1.5\,Z$. Setiap \omterm{def:g11:nucleus-radioactivity:notation}{nuklida} di luar pita itu ---
neutronnya terlalu banyak, protonnya terlalu banyak, atau ukurannya terlalu
besar (di luar timbal, $Z = 82$, tidak ada \omterm{def:g11:nucleus-radioactivity:notation}{nuklida} yang benar-benar stabil)
--- cepat atau lambat berubah.
\end{proposition}
\admitted

\begin{remark}
Menemukan letak pitanya secara tepat adalah mekanika kuantum --- yaitu jilid
perguruan tinggi. Polanya adalah buku besar
\cref{ch:g11:fundamental-interactions}: perekat kuatnya hanya mengikat
tetangga, sedangkan tolakan Coulomb-nya membentang di seluruh inti, sehingga
inti yang berat memerlukan perekat neutron tambahan; namun kelebihan neutron
pun tidak stabil (interaksi lemahlah yang mengurusnya).
\end{remark}

\begin{definition}[Keradioaktifan]\label{def:g11:nucleus-radioactivity:radioactivity}
\emph{Keradioaktifan}\index{keradioaktifan} adalah perubahan spontan sebuah
\omterm{def:g11:fundamental-interactions:ladder}{inti atom} yang tidak stabil menjadi \omterm{def:g11:nucleus-radioactivity:notation}{nuklida} yang lain, disertai pancaran
radiasi. Perubahan itu bersifat \emph{acak}: tidak ada yang mengumumkan inti
mana yang meluruh berikutnya, dan tidak ada \omterm{def:g10:pressure:pressure}{tekanan}, suhu atau kimia yang
dapat mempercepat atau menundanya; hanya jumlah besar yang dapat diramalkan
(\cref{def:g11:nucleus-radioactivity:halflife}).
\end{definition}

\begin{remark}[Becquerel dan keluarga Curie]\label{rem:g11:nucleus-radioactivity:curie}
Pada tahun 1896 Henri Becquerel menemukan bahwa garam uranium mengaburkan
pelat foto menembus kertas hitam, di dalam laci yang tertutup, tanpa \omterm{def:g10:energy-conservation:energy}{energi}
yang dipasok. Marie dan Pierre Curie menunjukkan bahwa gejalanya bersifat
keatoman, mengukurnya, lalu memisahkan polonium dan radium yang sejuta kali
lebih aktif. Kata \emph{\omterm{def:g11:nucleus-radioactivity:radioactivity}{keradioaktifan}} adalah kata Marie Curie; begitu pula
dua hadiah Nobel.
\end{remark}

\section{Alfa, beta, gama}

\begin{definition}[Ketiga radiasi bersejarahnya]\label{def:g11:nucleus-radioactivity:abg}
Ada tiga radiasi, yang dinamai sebelum ada yang tahu apa sebenarnya
ketiganya:
\begin{itemize}
  \item \emph{alfa}\index{radiasi alfa} ($\alpha$): sebuah inti helium
        ${}^{4}_{2}\mathrm{He}$ yang terlempar utuh --- \omterm{def:g10:universal-gravitation:weight}{berat}, bermuatan
        rangkap, dan dihentikan selembar kertas atau beberapa sentimeter
        udara;
  \item \emph{beta}\index{radiasi beta} ($\beta^-$): sebuah elektron
        ${}^{\;0}_{-1}\mathrm{e}$ yang tercipta di dalam intinya lalu
        terlempar --- ringan, cepat, dan dihentikan beberapa milimeter
        aluminium;
  \item \emph{gama}\index{radiasi gama} ($\gamma$): radiasi
        elektromagnetik yang jauh lebih berenergi daripada cahaya --- tidak
        pernah benar-benar terhenti, hanya
        \emph{terlemahkan}\index{pelemahan}: satu sentimeter timbal
        menyerap separuhnya.
\end{itemize}
\end{definition}

\begin{proposition}[Kekekalan pada persamaan inti]\label{prop:g11:nucleus-radioactivity:soddy}
Pada setiap perubahan inti, \omterm{def:g11:nucleus-radioactivity:notation}{nomor massa} seluruhnya $A$ dan muatan
seluruhnya --- yaitu jumlah indeks bawahnya, dengan menghitung $-1$ untuk
elektron yang terpancar --- sama sebelum dan sesudahnya.
\end{proposition}
\admitted

\begin{remark}
Kekekalan muatan sudah berlaku pada
\cref{ch:g11:fundamental-interactions}; kekekalan $A$ menyatakan bahwa
\omterm{def:g11:fundamental-interactions:ladder}{nukleon} hanya diubah, tidak pernah diciptakan atau dimusnahkan ---
pembukuan yang jujur bagi jilid perguruan tinggi.
\end{remark}

\begin{method}[Menyetimbangkan persamaan peluruhan]\label{met:g11:nucleus-radioactivity:balance}
\begin{enumerate}
  \item Tulislah setiap pelakunya dengan kedua labelnya:
        $\alpha = {}^{4}_{2}\mathrm{He}$,
        $\beta^- = {}^{\;0}_{-1}\mathrm{e}$,
        $\beta^+ = {}^{0}_{+1}\mathrm{e}$; sebuah $\gamma$ membawa
        $A = Z = 0$.
  \item Samakan jumlah $A$-nya, lalu jumlah $Z$-nya
        (\cref{prop:g11:nucleus-radioactivity:soddy}); selesaikan, lalu
        bacalah unsur anaknya dari $Z$-nya pada tabel berkala.
\end{enumerate}
\end{method}

\begin{example}[Sebuah peluruhan alfa]\label{ex:g11:nucleus-radioactivity:alpha}
Radium-226, radium milik keluarga Curie, adalah pemancar $\alpha$:
${}^{226}_{88}\mathrm{Ra} \to {}^{A}_{Z}\mathrm{Y} + {}^{4}_{2}\mathrm{He}$
memaksa $A = 226 - 4 = 222$ dan $Z = 88 - 2 = 86$; unsur $86$ adalah radon,
sehingga
${}^{226}_{88}\mathrm{Ra} \to {}^{222}_{86}\mathrm{Rn} + {}^{4}_{2}\mathrm{He}$
--- gas radioaktif di ruang bawah tanah yang tidak berventilasi.
\end{example}

\begin{example}[Sebuah peluruhan beta-minus]\label{ex:g11:nucleus-radioactivity:beta}
Karbon-14 berada di atas pitanya ($8$ neutron berbanding $6$ proton):
${}^{14}_{6}\mathrm{C} \to {}^{14}_{7}\mathrm{N} + {}^{\;0}_{-1}\mathrm{e}$.
Periksalah: $14 = 14 + 0$, $6 = 7 - 1$; sebuah neutron menjadi sebuah proton
ditambah elektron yang terlempar --- \omterm{def:g11:fundamental-interactions:weak}{interaksi lemah} sedang bekerja.
\end{example}

\begin{remark}[Beta-plus dan gama]\label{rem:g11:nucleus-radioactivity:betaplus}
Di bawah pitanya, peluruhan cerminnya $\beta^+$ mengubah sebuah proton
menjadi sebuah neutron ditambah sebuah \emph{positron}\index{positron}
${}^{0}_{+1}\mathrm{e}$, yaitu antipartikel elektron:
${}^{18}_{9}\mathrm{F} \to {}^{18}_{8}\mathrm{O} + {}^{0}_{+1}\mathrm{e}$,
kuda beban pemindai PET. Dan sesudah sebagian besar peluruhannya, inti
anaknya, yang lahir bergetar, menenangkan diri dengan memancarkan sebuah
$\gamma$ sebagai tambahan.
\end{remark}

\section{Waktu paruh dan aktivitas}

\begin{definition}[Waktu paruh]\label{def:g11:nucleus-radioactivity:halflife}
\emph{Waktu paruh}\index{waktu paruh} $T_{1/2}$ sebuah \omterm{def:g11:nucleus-radioactivity:notation}{nuklida} adalah waktu
yang membuat separuh cuplikan besar mana pun sudah meluruh. Tiap waktu paruh
berikutnya menyeparuhkan sisanya: dari $N_0$ inti, tersisa $N_0/2$ pada
$T_{1/2}$, $N_0/4$ pada $2\,T_{1/2}$, $N_0/8$ pada $3\,T_{1/2}$, dan
$N_0/2^{\,n}$ sesudah $n$ waktu paruh.
\end{definition}

\begin{remark}
Acak bagi satu inti, seperti jam bagi $10^{23}$ inti: sebuah kerumunan besar
pelempar uang logam, yang tidak dapat diramalkan satu per satu, dan
separuhnya tersingkir tiap babak. Menyisipkan nilai \emph{di antara} \omterm{def:g11:nucleus-radioactivity:halflife}{waktu
paruh} yang bulat memerlukan fungsi eksponensial --- tahun depan, bersama
jilid matematika.
\end{remark}

\begin{example}[Dua puluh pangkat sepuluh]\label{ex:g11:nucleus-radioactivity:range}
\begin{center}\small\begin{tabular}{llll}
polonium-214 & $\alpha$ & \qty{1.6e-4}{s} & mata rantai radium \\
teknesium-99m & $\gamma$ & $6.0$ jam & pencitraan kedokteran \\
amerisium-241 & $\alpha$ & $432$ tahun & pengindra asap \\
karbon-14 & $\beta^-$ & \num{5700} tahun & arkeologi \\
kalium-40 & $\beta$ & \num{1.25e9} tahun & di dalam tiap pisang \\
uranium-238 & $\alpha$ & \num{4.5e9} tahun & kalor dalam Bumi \\
\end{tabular}\end{center}
\end{example}

\begin{example}[Penanggalan karbon-14]\label{ex:g11:nucleus-radioactivity:dating}
Makhluk hidup menukarkan karbon dengan atmosfer, sehingga ia memuat
perbandingan karbon-14 yang sama dengan atmosfernya; pada saat mati
asupannya berhenti dan perbandingannya menjadi separuh tiap \num{5700}
tahun. Arang dari sebuah gua berlukisan menunjukkan seperempat perbandingan
makhluk hidup: seperempat adalah separuh dari separuh, sehingga apinya
menyala dua \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}, yaitu $2 \times \num{5700} = \num{11400}$ tahun
yang lalu.
\end{example}

\begin{definition}[Aktivitas]\label{def:g11:nucleus-radioactivity:activity}
\emph{Aktivitas}\index{aktivitas} $\mathcal A$ sebuah cuplikan adalah
banyaknya peluruhan tiap sekon, yang diukur dalam
\emph{becquerel}\index{becquerel}: $\qty{1}{Bq} = 1$ peluruhan tiap sekon.
Makin sedikit inti yang bertahan, makin sedikit peluruhannya, sehingga
aktivitasnya pun menjadi separuh pada tiap \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}.
\end{definition}

\begin{example}[Orde besar]\label{ex:g11:nucleus-radioactivity:orders}
Segala sesuatu sedikit banyak bersifat radioaktif. Sebuah pisang: sekitar
\qty{15}{Bq} (kalium-40). Tubuh manusia: sekitar \qty{8000}{Bq} --- kaitan
pembuka bab ini, setara sekitar lima ratus pisang. Satu \omterm{def:g10:orders-of-magnitude:unit}{meter} kubik granit:
sekitar \qty{3e6}{Bq} (uranium beserta rantainya, sumber radon ruang bawah
tanah). Satu suntikan teknesium: sekitar \qty{5e8}{Bq}, yang lenyap dalam
beberapa hari.
\end{example}

\section{Pelayan dan bahaya}

\begin{example}[Tiga karier sebuah inti yang tidak stabil]\label{ex:g11:nucleus-radioactivity:uses}
\emph{Kedokteran}: teknesium-99m, pemancar $\gamma$ murni dengan
$T_{1/2} = 6$ jam, dilekatkan pada \omterm{def:g11:fundamental-interactions:ladder}{molekul} yang diserap organ sasarannya;
sebuah kamera merekam $\gamma$-nya dari luar, dan pada hari berikutnya
perunutnya sebagian besar sudah lenyap --- sedangkan terapi radiasi
membalikkan alasannya, memusatkan berkasnya untuk memusnahkan sebuah tumor.
\emph{Penanggalan}: karbon-14 untuk kayu, tulang dan kain sampai sekitar
\num{50000} tahun ke belakang; uranium-238 untuk batuan --- dan untuk Bumi,
pada \num{4.5e9} tahun. \emph{Pengindra asap}: sebutir kecil amerisium-241
mengionkan udara sebuah kamar kecil sehingga arus yang mungil dapat
mengalir; asap mencekik arusnya dan alarmnya pun berbunyi --- $\alpha$-nya
mati di dalam beberapa sentimeter udara.
\end{example}

\begin{remark}[Dosis dan sievert]\label{rem:g11:nucleus-radioactivity:dose}
Radiasi merenggut elektron dari \omterm{def:g11:fundamental-interactions:ladder}{molekul} dan dapat merusak DNA. Kerugian
hayatinya dilacak oleh \emph{dosis}\index{dosis}, dalam
\emph{sievert}\index{sievert} (\unit{Sv}), yang menimbang \omterm{def:g10:energy-conservation:energy}{energi} yang
diendapkan tiap \omterm{def:g10:orders-of-magnitude:unit}{kilogram} jaringan menurut seberapa merusak tiap radiasinya.
Ada tiga aturannya: jarak, penudungan ($\alpha$: kulit sudah memadai ---
kecuali jika pemancarnya \emph{terhirup}, itulah kejahatan radon; $\beta$:
aluminium; $\gamma$: timbal), dan waktu. Dosimetri yang sebenarnya adalah
bahan perguruan tinggi.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:1}
Sebutkan banyaknya proton dan neutron pada ${}^{16}_{8}\mathrm{O}$,
${}^{27}_{13}\mathrm{Al}$, ${}^{60}_{27}\mathrm{Co}$,
${}^{235}_{92}\mathrm{U}$ dan ${}^{3}_{1}\mathrm{H}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:2}
Di antara ${}^{14}_{6}\mathrm{C}$, ${}^{14}_{7}\mathrm{N}$,
${}^{12}_{6}\mathrm{C}$, ${}^{13}_{6}\mathrm{C}$ dan
${}^{16}_{8}\mathrm{O}$, yang mana yang saling merupakan \omterm{def:g11:nucleus-radioactivity:notation}{isotop}? Mengapa
${}^{14}_{6}\mathrm{C}$ dan ${}^{14}_{7}\mathrm{N}$ \emph{bukan} \omterm{def:g11:nucleus-radioactivity:notation}{isotop},
walaupun \omterm{def:g11:nucleus-radioactivity:notation}{nomor massanya} sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:3}
Polonium-210, ${}^{210}_{84}\mathrm{Po}$, adalah pemancar $\alpha$. Tulislah
persamaan peluruhannya lalu sebutkan inti anaknya ($Z = 81$: talium, $82$:
timbal, $83$: bismut).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:4}
Tulislah persamaan peluruhan $\beta^-$ kobalt-60
(${}^{60}_{27}\mathrm{Co}$) dan iodin-131 (${}^{131}_{53}\mathrm{I}$); inti
anaknya: $Z = 28$ nikel, $Z = 54$ xenon.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:nucleus-radioactivity:5}
Sebuah sumber iodin-131 ($T_{1/2} = 8.0$ hari) beraktivitas \qty{800}{MBq}
hari ini. Berapa aktivitasnya sesudah $16$ hari? Dan sesudah $32$ hari?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:6}
Empat radiasi melintasi \omterm{def:g11:magnetic-fields:bfield}{medan magnet} yang kuat, lalu beberapa tabir.
Kenalilah masing-masing: (a) dihentikan kertas, nyaris tidak disimpangkan
\omterm{def:g11:electric-gravitational-fields:field}{medannya}; (b) menembus kertas, dihentikan \qty{4}{mm} aluminium,
disimpangkan dengan kuat; (c) menembus keduanya, hanya \omterm{def:g11:nucleus-radioactivity:abg}{terlemahkan} oleh
timbal, tidak disimpangkan; (d) seperti (b), tetapi disimpangkan ke arah
yang lain.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:7}
Radon-222 (${}^{222}_{86}\mathrm{Rn}$) membuka sebuah rantai yang cepat:
sebuah peluruhan $\alpha$, lalu $\alpha$ lagi, lalu sebuah $\beta^-$.
Tulislah ketiga persamaannya ($Z = 82$: timbal, $83$: bismut, $84$:
polonium).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:8}
Fluorin-18, ${}^{18}_{9}\mathrm{F}$, adalah perunut pemindai PET; fluorin
yang stabil adalah ${}^{19}_{9}\mathrm{F}$. Berada di sisi mana dari pita
kestabilannya? Ramalkan cara peluruhannya lalu tulislah persamaannya
($Z = 8$: oksigen).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:9}
Seorang pasien menerima \qty{500}{MBq} teknesium-99m ($T_{1/2} = 6.0$ jam)
pada pukul 08:00. Berapa aktivitasnya pada pukul 08:00 keesokan paginya?
Sesudah berapa jam aktivitasnya pertama kali turun di bawah \qty{1}{MBq}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:10}
Sebuah tulang dari rawa gambut menunjukkan perbandingan karbon-14
seperdelapan perbandingan tulang yang hidup ($T_{1/2} = \num{5700}$ tahun).
Berapa umur tulang itu? Dan mengapa karbon-14 tidak berguna untuk
menanggali tulang dinosaurus (umurnya sekitar $10^{8}$ tahun)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:nucleus-radioactivity:11}
Dengan memakai lembah kestabilannya, ramalkan cara peluruhan ($\alpha$,
$\beta^-$ atau $\beta^+$) ${}^{14}_{6}\mathrm{C}$,
${}^{11}_{6}\mathrm{C}$ dan ${}^{238}_{92}\mathrm{U}$, masing-masing
dengan alasan satu baris (karbon yang stabil: ${}^{12}_{6}\mathrm{C}$,
${}^{13}_{6}\mathrm{C}$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:12}
Rantai uranium-238 berakhir, setelah banyak langkah, pada
${}^{206}_{82}\mathrm{Pb}$ yang stabil; setiap langkahnya adalah $\alpha$
atau $\beta^-$. Dengan memakai kekekalan $A$ saja, tentukan banyaknya
langkah $\alpha$; lalu, dengan kekekalan $Z$, banyaknya langkah $\beta^-$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:13}
Sumber kobalt-60 sebuah rumah sakit ($T_{1/2} = 5.3$ tahun) beraktivitas
\qty{6.4}{TBq}; peraturannya mengizinkan pembuangan di bawah
\qty{0.1}{TBq}. Sesudah berapa tahun sumber itu boleh dibuang?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:14}
\omterm{def:g11:nucleus-radioactivity:activity}{Aktivitas} tubuhmu sekitar \qty{8000}{Bq}. Berapa banyak inti atommu yang
meluruh tiap hari? Dan sepanjang hidup $80$ tahun? Sebuah pisang menambahkan
sekitar \qty{15}{Bq} selagi kamu mencernanya: berikan ulasanmu, dalam satu
kalimat, tentang tajuk berita yang menakuti setiap \omterm{def:g11:nucleus-radioactivity:activity}{becquerel}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:nucleus-radioactivity:15}
Sebuah pengindra asap memuat \qty{0.2}{\micro\gram} amerisium-241
($T_{1/2} = 432$ tahun), yang beraktivitas sekitar \qty{25}{kBq}. Setara
berapa peluruhan tiap hari? Sesudah sepuluh tahun bertugas, apakah
amerisiumnya sudah meluruh jauh kurang dari separuh, kira-kira separuh, atau
jauh lebih banyak --- dan apakah habisnya sumber itu yang menjadi alasan
pengindranya diganti? Terakhir, mengapa sumber $\alpha$ ini tidak berbahaya
di langit-langit, namun berbahaya jika kapsulnya digerus dan debunya
terhirup?
\end{exercise}

\section{Soal: Mumi, reaktor, dan pengindra asap}

\begin{problem}[{Soal akhir pekan --- tiga karier sebuah inti yang tidak
stabil: sebuah mumi ditanggali dengan penyeparuhan, silsilah uranium
disetimbangkan, sebuah pengindra asap diaudit, dan mengapa bintang harus
fana}]
\label{pb:g11:nucleus-radioactivity:1}
Tiga inti yang tidak stabil, tiga pekerjaan: karbon-14 menandai waktu setiap
tumbuhan dan hewan yang mati, uranium-238 memanaskan sebuah planet dari
dalam, amerisium-241 mengawasi langit-langit. Datanya:
$T_{1/2} = \num{5700}$ tahun (C-14), \num{4.5e9} tahun (U-238), $432$ tahun
(Am-241); karbon yang hidup menunjukkan $13.6$ peluruhan tiap menit tiap
gram; \omterm{def:g11:nucleus-radioactivity:notation}{nomor atomnya}: nitrogen $7$, oksigen $8$, timbal $82$, bismut $83$,
polonium $84$, radon $86$, radium $88$, torium $90$, protaktinium $91$,
uranium $92$, neptunium $93$, amerisium $95$.

\medskip
\noindent\textbf{Bagian I --- Muminya.}
\begin{enumerate}
  \item Sebutkan susunan (proton, neutron) ${}^{12}_{6}\mathrm{C}$ dan
        ${}^{14}_{6}\mathrm{C}$. Mengapa kimianya serupa di dalam tubuh?
  \item Mengapa perbandingan karbon-14 tetap di dalam tubuh yang hidup, dan
        mengapa perbandingan itu mulai turun pada saat mati?
  \item Tulislah persamaan peluruhan karbon-14. Apakah radiasinya lolos
        menembus kain pembungkus muminya?
  \item Satu gram karbon dari kainnya menunjukkan $6.8$ peluruhan tiap
        menit. Berapa bagian dari laju makhluk hidupnya? Tanggalilah mumi
        itu.
  \item ``Mumi'' kedua, kebanggaan sebuah koleksi pribadi, menunjukkan
        $13.2$ peluruhan tiap menit tiap gram. Bagaimana putusannya?
  \item Taksirlah lajunya dari cuplikan yang berumur sepuluh \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh},
        lalu jelaskan mengapa penanggalan karbon memudar di luar kira-kira
        \num{57000} tahun.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Reaktor di bawah padang rumput.}
\begin{enumerate}[resume]
  \item Sebutkan susunan ${}^{238}_{92}\mathrm{U}$. Mengapa inti yang berat
        memerlukan kelebihan neutron sebanyak itu?
  \item Tulislah persamaan peluruhan $\alpha$-nya.
  \item Inti anaknya lalu meluruh $\beta^-$, dan inti cucunya meluruh
        $\beta^-$ lagi. Tulislah kedua persamaannya. Unsur yang mana yang
        muncul kembali?
  \item Rantainya berakhir pada ${}^{206}_{82}\mathrm{Pb}$. Hitunglah
        banyaknya langkah $\alpha$-nya, lalu banyaknya langkah
        $\beta^-$-nya.
  \item Bumi terbentuk \num{4.5e9} tahun yang lalu. Berapa bagian
        uranium-238 purbanya yang tersisa hari ini? Dan $5$ miliar tahun
        lagi?
  \item Dalam satu kalimat: apa yang dinyalakan kalor peluruhan yang
        terkubur itu di permukaan? (Pikirkanlah gunung api dan benua yang
        hanyut.)
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Pengindra asapnya.}
\begin{enumerate}[resume]
  \item Tulislah persamaan peluruhan $\alpha$ ${}^{241}_{95}\mathrm{Am}$.
  \item Aktivitasnya \qty{33}{kBq}: berapa peluruhan tiap sekon? Dan tiap
        hari?
  \item Jelaskan pengindranya: apa yang dilakukan partikel $\alpha$-nya
        terhadap udara di dalam kamarnya, dan apa yang diubah asapnya?
  \item Mengapa pemancar $\alpha$ merupakan pilihan yang tepat di sini ---
        dan pemancar $\gamma$ dengan \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} yang sama merupakan pilihan
        yang terburuk?
  \item Sesudah satu \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} ($432$ tahun!) aktivitasnya akan menjadi
        \qty{16.5}{kBq}; buku petunjuknya meminta penggantian sesudah
        sepuluh tahun. Apakah sumbernya yang menjadi mata rantai terlemah?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Mengapa bintang harus fana.}
\begin{enumerate}[resume]
  \item Bintang yang sekarat menempa lalu menaburkan setiap unsur yang
        lebih berat daripada helium. Berapa bagian uranium-238 yang ditempa
        \num{9e9} tahun yang lalu yang bertahan hari ini? Mengapa kelimpahan
        sisanya di Bumi menjadikan planet itu jauh lebih muda daripada
        bintang yang tertua?
  \item Kalor radioaktif di dalam Bumi menggerakkan kegunungapian dan
        lempeng tektonik, dan menjaga intinya tetap bergolak sehingga
        menghasilkan tameng \omterm{def:g11:magnetic-fields:magnet}{magnet} pada
        \cref{ch:g11:magnetic-fields}. Akan seperti apa
        sebuah planet yang hanya berisi \omterm{def:g11:fundamental-interactions:ladder}{atom} yang stabil?
  \item Satu baris, sebagai puisi sekaligus sebagai pembukuan: di manakah
        atom-atommu ditempa, dan apa yang masih menghangatkan tanah di bawah
        kakimu?
\end{enumerate}
\end{problem}
