\chapter{Gaya dan Gerak}\label{ch:g11:forces-and-motion}

Lemparkan sebuah kunci pas sambil berputar melintasi ruangan: kedua
ujungnya bergoyang liar, namun satu titik di dalamnya melukis parabola
yang sempurna. \Cref{ch:g10:inertia} sudah menuntaskan apa yang terjadi
ketika \omterm{def:g10:inertia:compensate}{gaya saling meniadakan} --- tidak terjadi apa-apa. Bab ini
bertanya apa yang terjadi ketika \omterm{def:g10:inertia:force}{gayanya} tidak meniadakan: \omterm{def:g10:inertia:force}{gaya} yang
tersisa \emph{mengubah kecepatannya} --- mempercepat benda,
memperlambatnya, membelokkannya --- lalu menyatakan seberapa banyak,
dalam nisbah kalaupun belum dalam rumus.

\section{Pusat massa}

\begin{definition}[Pusat massa]\label{def:g11:forces-and-motion:com}
\emph{Pusat massa}\index{pusat massa} sebuah benda atau sistem adalah
titik setimbangnya --- kedudukan rata-rata massanya: pusat ilmu ukur
sebuah benda homogen yang setangkup; untuk dua massa titik, titik pada
ruas garis penghubungnya yang jaraknya berbanding \emph{terbalik}
dengan massanya, $\frac{d_1}{d_2} = \frac{m_2}{m_1}$ --- lebih dekat ke
yang lebih berat.
\end{definition}

\begin{example}[Barbel yang berat sebelah]\label{ex:g11:forces-and-motion:dumbbell}
Bola \qty{2.0}{kg} dan \qty{1.0}{kg} berada di kedua ujung sebatang
ringan sepanjang \qty{0.90}{m}. \omterm{def:g11:forces-and-motion:com}{Pusat massanya} membagi batang itu
dengan nisbah terbalik $d_1/d_2 = 1.0/2.0$: $\qty{0.30}{m}$ dari bola
yang berat, $\qty{0.60}{m}$ dari yang ringan --- setimbangkanlah
batangnya di sana.
\end{example}

\begin{remark}[Kunci pas yang dilemparkan]\label{rem:g11:forces-and-motion:wrench}
Rekamlah sebuah kunci pas yang berjungkir balik: setiap titiknya
menempuh kurva berlekuk yang rumit --- kecuali \omterm{def:g11:forces-and-motion:com}{pusat massanya}, yang
melukis parabola bersih seperti yang ditempuh sebuah bola yang
dilemparkan. Gerak \omterm{def:g11:forces-and-motion:com}{pusat massanya} sederhana; sisanya adalah putaran
\emph{mengelilinginya} --- dan itulah sebabnya mekanika boleh
memperlakukan sebuah mobil, sebuah planet atau seorang pesenam sebagai
satu titik.
\end{remark}

\section{Gaya total mengubah kecepatan}

\begin{definition}[Gaya total]\label{def:g11:forces-and-motion:netforce}
\emph{Gaya total}\index{gaya total} pada sebuah benda adalah jumlah
vektor semua \omterm{def:g10:inertia:force}{gaya} yang bekerja padanya,
$\vect F = \vect F_1 + \vect F_2 + \dots$ (ujung ke pangkal, seperti
pada jilid matematika); \omterm{def:g10:inertia:force}{gayanya} \omterm{def:g10:inertia:compensate}{saling meniadakan} tepat ketika
$\vect F = \vect 0$.
\end{definition}

\begin{theorem}[Akibat sebuah gaya total --- setengah kuantitatif]\label{thm:g11:forces-and-motion:secondlaw}
\index{hukum kedua gerak}
\omterm{def:g11:forces-and-motion:netforce}{Gaya total} $\vect F$ yang bekerja pada benda bermassa $m$ selama selang
waktu singkat $\Delta t$ mengubah kecepatan \omterm{def:g11:forces-and-motion:com}{pusat massanya} sebesar
vektor $\Delta \vect v$ yang \emph{arahnya} searah $\vect F$, dan yang
besarnya tumbuh sebanding dengan $F$ dan dengan $\Delta t$ serta
menyusut berbanding terbalik dengan massanya:
\[
\Delta v \propto \frac{F \, \Delta t}{m} .
\]
\end{theorem}
\admitted

\begin{remark}
Gandakan \omterm{def:g10:inertia:force}{gayanya} atau lamanya: perubahannya menjadi dua kali lipat;
gandakan massanya: perubahannya menjadi separuh. Tahun depan
mempertajamnya menjadi persamaan yang tepat, dan jilid Tahun ke-1
bertanya di dalam kerangka yang mana persamaan itu berlaku; tahun ini
nisbahnya sudah cukup --- nisbah itu sudah menjalankan kantong udara,
rem dan \omterm{def:g10:universal-gravitation:satellite}{orbit}.
\end{remark}

\begin{proposition}[Tiga kasus bakunya]\label{prop:g11:forces-and-motion:cases}
Misalkan sebuah benda bergerak dengan kecepatan $\vect v$ di bawah \omterm{def:g11:forces-and-motion:netforce}{gaya
total} $\vect F$.
\begin{enumerate}
  \item $\vect F$ searah $\vect v$: benda itu \emph{bertambah cepat}
        pada garis lurus;
  \item $\vect F$ berlawanan dengan $\vect v$: benda itu
        \emph{melambat} pada garis lurus;
  \item $\vect F$ tegak lurus $\vect v$: benda itu \emph{berbelok}
        tanpa berubah lajunya.
\end{enumerate}
\end{proposition}

\begin{proof}
Jumlahkan $\Delta\vect v$, yang terarah searah $\vect F$
(\cref{thm:g11:forces-and-motion:secondlaw}), ujung ke pangkal pada
$\vect v$: searah $\vect v$ anak panahnya bertumpuk (menjadi lebih
panjang); berlawanan, keduanya saling menghapus sebagian (menjadi lebih
pendek); tegak lurus, tiap $\Delta \vect v$ yang kecil memiringkan anak
panahnya tanpa memanjangkannya --- hanya arahnya yang berputar.
\end{proof}

\begin{example}[Sebuah tendangan menyamping]\label{ex:g11:forces-and-motion:kick}
Sebuah keping meluncur ke timur dengan \qty{6.0}{m/s}; sebuah pukulan
singkat menyamping menambahkan $\Delta\vect v = \qty{8.0}{m/s}$ tepat
ke utara. Kecepatannya yang baru adalah jumlah vektornya: besarnya
$\sqrt{6.0^2 + 8.0^2} = \qty{10.0}{m/s}$, arahnya
$\tan\theta = 8.0/6.0$, yaitu $\theta \approx 53^\circ$ dari timur ke
arah utara --- kecepatan lamanya tidak terhapus, perubahannya
dijumlahkan padanya.
\end{example}

\section{Membandingkan perubahan: gaya, waktu, massa}

\begin{method}[Penalaran dengan nisbah]\label{met:g11:forces-and-motion:ratio}
Untuk membandingkan dua keadaan, jangan pernah menyelesaikan sebuah
persamaan --- bentuklah nisbah: nisbah perubahan kecepatannya adalah
nisbah $F$ dikali nisbah $\Delta t$ lalu dibagi nisbah $m$. Lalu bacalah
faktor yang mana yang berubah, dan sebanyak apa.
\end{method}

\begin{example}[Kereta belanja yang bermuatan]\label{ex:g11:forces-and-motion:cart}
Dorongan yang sama, yang diberikan selama sekon yang sama, membuat
kereta belanja \qty{20}{kg} yang kosong melaju \qty{1.2}{m/s}. Ketika
bermuatan sampai \qty{60}{kg} --- massanya tiga kali lipat, $F\Delta t$
yang sama --- kereta itu pergi dengan sepertiganya, yaitu
\qty{0.40}{m/s}. Massa adalah kekeraskepalaan materi.
\end{example}

\begin{remark}[Tabrakan, kantong udara, lutut yang ditekuk]\label{rem:g11:forces-and-motion:airbag}
Pada sebuah tabrakan, $\Delta v$ dan $m$ tidak dapat ditawar ---
kecepatan penumpangnya harus mencapai nol --- sehingga $F\,\Delta t$
sudah tertetapkan. Satu-satunya kebebasan yang tersisa adalah
\emph{waktu}: panjangkanlah $\Delta t$ dan \omterm{def:g10:inertia:force}{gayanya} menyusut dengan
nisbah yang sama. Kepala yang berhenti pada kantong udara dalam
\qty{0.08}{s} dan bukan pada setirnya dalam \qty{0.01}{s} merasakan
\omterm{def:g10:inertia:force}{gaya} rata-rata delapan kali lebih kecil; zona remuk, busa helm dan lutut
yang ditekuk memainkan muslihat yang sama.
\end{remark}

\section{Gesekan: gaya yang melawan peluncuran}

\begin{definition}[Gaya gesek statis dan kinetis]\label{def:g11:forces-and-motion:friction}
\omterm{def:g10:inertia:inventory}{Gesekan} antara permukaan yang bersentuhan melawan \emph{peluncuran
relatif}nya. \emph{Gaya gesek statis}\index{gaya gesek!statis} bekerja
selama keduanya tidak meluncur, dan menyesuaikan dirinya sampai suatu
nilai terbesar untuk menghapus apa pun yang berusaha memulai
peluncurannya; \emph{gaya gesek kinetis}\index{gaya gesek!kinetis},
begitu peluncurannya dimulai, kira-kira bernilai tetap, terarah melawan
peluncurannya, dan biasanya \emph{lebih lemah} daripada nilai terbesar
\omterm{def:g10:inertia:inventory}{gesekan} statisnya.
\end{definition}

\begin{example}[Mendorong lemari pakaian]\label{ex:g11:forces-and-motion:wardrobe}
Dorong sebuah lemari pakaian perlahan-lahan: lemari itu tidak bergerak
--- \omterm{def:g10:inertia:inventory}{gaya gesek} statisnya menghapus \qty{150}{N} milikmu. Doronglah lebih
kuat: pada sekitar \qty{400}{N} lemari itu tersentak bergerak, lalu
meluncur dengan usaha yang lebih kecil --- \omterm{def:g11:forces-and-motion:friction}{gaya gesek kinetis} yang lebih
lemah sudah mengambil alih.
\end{example}

\begin{remark}[Mengapa rem ABS bekerja]\label{rem:g11:forces-and-motion:abs}
Titik sentuh roda yang menggelinding tidak meluncur di atas jalannya:
bannya mencengkeram dengan \omterm{def:g10:inertia:inventory}{gaya gesek} \emph{statis}. Roda yang terkunci
tergelincir dan hanya memperoleh \omterm{def:g10:inertia:inventory}{gaya gesek} \emph{kinetis} yang lebih
lemah, yang menunjuk melawan gelincirnya apa pun yang dilakukan
rodanya --- kemudinya pun mati. ABS melepaskan remnya seketika ketika
sebuah roda terkunci: cengkeramannya lebih kuat, berhentinya lebih
pendek, dan pengemudinya masih dapat mengemudi.
\end{remark}

\section{Berbelok: gerak melingkar memerlukan gaya ke dalam}

\begin{definition}[Gerak melingkar beraturan]\label{def:g11:forces-and-motion:ucm}
Sebuah benda berada dalam \emph{gerak melingkar
beraturan}\index{gerak melingkar beraturan} apabila \omterm{def:g10:relative-motion:trajectory}{lintasannya} berupa
lingkaran yang ditempuh dengan kelajuan tetap. Kecepatannya, yang
menyinggung lingkarannya, berubah arah pada setiap saat --- sehingga
kecepatan itu tidak pernah tetap.
\end{definition}

\begin{proposition}[Gaya ke dalam]\label{prop:g11:forces-and-motion:inward}
Benda dalam \omterm{def:g11:forces-and-motion:ucm}{gerak melingkar beraturan} dikenai \omterm{def:g11:forces-and-motion:netforce}{gaya total} yang tidak
pernah bernilai nol: pada setiap saat \omterm{def:g10:inertia:force}{gaya} itu menunjuk dari bendanya
menuju pusat lingkarannya.
\end{proposition}

\begin{proof}
Lajunya tetap, sehingga \omterm{def:g11:forces-and-motion:netforce}{gaya totalnya} tidak memiliki bagian sepanjang
$\vect v$ (kasus 1 dan 2 pada
\cref{prop:g11:forces-and-motion:cases}): \omterm{def:g10:inertia:force}{gaya} itu tegak lurus
garis singgungnya, jadi sepanjang jari-jarinya --- dan mengarah ke
dalam, sebab tiap $\Delta \vect v$ membengkokkan \omterm{def:g10:relative-motion:trajectory}{lintasannya} menuju
pusatnya.
\end{proof}

\begin{example}[Tali, tikungan, dan Bulan]\label{ex:g11:forces-and-motion:moon}
Setiap belokan yang tetap menyembunyikan sebuah \omterm{def:g10:inertia:force}{gaya} ke dalam. Bola yang
diputar: tegangan talinya. Mobil pada tikungan: \omterm{def:g10:inertia:inventory}{gaya gesek} statis
menyamping antara jalan dan bannya --- yang lenyap di atas es, dan
mobilnya pun berjalan lurus. Bulan: tarikan gravitasi Bumi
(\cref{ch:g10:universal-gravitation}), yang memutarnya dalam
\num{27.3} hari --- Bulan tidak ditahan \emph{ke atas} melainkan ditarik
menyamping ke dalam belokan yang abadi, jatuh mengitari Bumi; tahun
depan menjadikan hal ini mekanika \omterm{def:g10:universal-gravitation:satellite}{satelit}.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:1}
\omterm{def:g11:forces-and-motion:netforce}{Gaya totalnya} nol atau bukan nol? Berikan alasannya dari kecepatannya:
(a) mobil yang melaju lurus dengan tetap \qty{90}{km/h}; (b) mobil yang
mengerem pada garis lurus; (c) mobil yang membelok pada tikungan dengan
tetap \qty{50}{km/h}; (d) penerjun payung pada laju terminalnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:2}
Bola \qty{3.0}{kg} dan \qty{1.0}{kg} berada di kedua ujung sebatang
ringan sepanjang \qty{0.80}{m}. Tentukan letak \omterm{def:g11:forces-and-motion:com}{pusat massanya}; di mana
letaknya jika massanya sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:3}
Dorongan singkat yang sama diberikan kepada troli bagasi \qty{15}{kg}
yang kosong dan kepada troli yang sama ketika bermuatan sampai
\qty{45}{kg}. Bandingkan perubahan kecepatannya; dorongan berapa yang
akan memberikan troli bermuatan itu $\Delta v$ milik troli yang kosong?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:4}
Sebuah trem yang bergerak \qty{10}{m/s} mengerem; satu sekon kemudian
trem itu bergerak \qty{8}{m/s} pada arah yang sama. Sebutkan
$\Delta \vect v$ dan arah \omterm{def:g11:forces-and-motion:netforce}{gaya totalnya}. Kasus yang mana pada
\cref{prop:g11:forces-and-motion:cases} ini?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:forces-and-motion:5}
\omterm{def:g11:forces-and-motion:friction}{Gaya gesek statis} atau kinetis --- dan terarah ke mana? (a) Sebuah peti
yang melawan doronganmu yang (terlalu lembut); (b) peti yang sama
setelah meluncur; (c) sebuah buku di atas nampan yang sedikit miring,
yang tidak meluncur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:6}
\omterm{def:g11:forces-and-motion:netforce}{Gaya total} $F$ yang bekerja selama $\Delta t$ pada massa $m$
menghasilkan $\Delta v = \qty{4.0}{m/s}$. Ramalkan $\Delta v$ untuk:
(a) $2F, \Delta t, m$; (b) $F, \Delta t/2, m$; (c) $F, \Delta t, 2m$;
(d) $3F, 2\Delta t, 6m$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:7}
Sebuah keping meluncur ke timur dengan \qty{8.0}{m/s}; sebuah pukulan
menambahkan $\Delta \vect v = \qty{6.0}{m/s}$ tepat ke utara. Hitunglah
laju dan arahnya yang baru. Mengapa lajunya bukan
$8.0 + 6.0 = \qty{14}{m/s}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:8}
Seorang penjaga gawang menghentikan tendangan yang sama dengan dua
cara: lengannya kaku (bolanya berhenti dalam \qty{0.02}{s}) atau
lengannya ``mengalah'' bersama bolanya (\qty{0.08}{s}). Bandingkan \omterm{def:g10:inertia:force}{gaya}
rata-ratanya, lalu nyatakan besaran mana yang sudah tertetapkan sebelum
ia memilih.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:9}
\omterm{def:g10:inertia:force}{Gaya} luar yang mana yang mendorong seorang pelari cepat ke depan pada
saat mulai? Mengapa sepatu berpaku membantu --- dan mengapa tidak ada
yang dapat mempercepat, atau bahkan berjalan, di atas es yang
benar-benar tanpa \omterm{def:g10:inertia:inventory}{gesekan}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:10}
Pada pengereman darurat, mengapa mobil dengan ABS (rodanya dijaga tetap
menggelinding) biasanya berhenti lebih pendek daripada mobil dengan roda
terkunci yang tergelincir --- dan mengapa mobil yang tergelincir itu
tidak lagi dapat dikemudikan? Kutiplah
\cref{def:g11:forces-and-motion:friction}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:forces-and-motion:11}
Seorang pelempar martil memutar bolanya pada kawatnya. \omterm{def:g10:inertia:force}{Gaya} apa yang
memutar bola itu? Sketsalah \omterm{def:g10:relative-motion:trajectory}{lintasannya} setelah dilepaskan. Mengapa
Bulan tidak memerlukan kawat, dan \omterm{def:g10:relative-motion:trajectory}{lintasannya} akan menjadi apa tanpa
gravitasi?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:12}
Sebuah sapu berupa tongkat seragam \qty{1.0}{kg} (\omterm{def:g11:forces-and-motion:com}{pusat massanya}
sendiri \qty{0.60}{m} dari ujung atasnya) dengan kepala \qty{2.0}{kg}
yang \omterm{def:g11:forces-and-motion:com}{pusat massanya} \qty{1.20}{m} dari ujung atasnya. Dengan
memperlakukan tiap bagiannya sebagai massa titik, tentukan letak \omterm{def:g11:forces-and-motion:com}{pusat
massa} sapu itu lalu periksalah aturan nisbah terbalik pada
\cref{def:g11:forces-and-motion:com}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:13}
Sebuah mobil bermassa $m$ dan sebuah van bermuatan bermassa $2m$
mengerem dari laju yang sama dengan \omterm{def:g10:inertia:force}{gaya} pengereman yang sama.
\begin{enumerate}
  \item Bandingkan waktu yang diperlukan keduanya untuk berhenti.
  \item Masing-masing kehilangan lajunya secara tetap, sehingga
        menempuh perjalanan dengan laju \emph{rata-rata} yang sama:
        bandingkan jarak pengeremannya.
  \item Mobilnya kini mengerem dari laju \emph{dua kali lipat}, dengan
        \omterm{def:g10:inertia:force}{gaya} yang sama: bandingkan waktu dan jaraknya dengan
        pengeremannya yang pertama. Apa pesan moralnya bagi batas
        kecepatan?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:14}
Bulan mengorbit pada $R \approx \qty{3.84e8}{m}$ dalam
$T = \num{27.3}$ hari.
\begin{enumerate}
  \item Hitunglah laju \omterm{def:g10:universal-gravitation:satellite}{orbitnya} (kelilingnya dibagi \omterm{def:g10:signals-and-waves:period}{periodenya}).
  \item Ke arah mana perubahan kecepatannya menunjuk, dan \omterm{def:g10:inertia:force}{gaya} yang
        mana yang menyediakannya?
  \item Perbaikilah, dengan memakai
        \cref{prop:g11:forces-and-motion:inward}: ``Bulan tidak jatuh
        karena ia bergerak cukup cepat''.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:forces-and-motion:15}
Sebutir telur bermassa \qty{50}{g} membentur lantai dengan
\qty{4.0}{m/s}: di atas ubin telur itu berhenti dalam sekitar
\qty{1}{ms}, di atas busa tebal dalam sekitar \qty{50}{ms}. Bandingkan
\omterm{def:g10:inertia:force}{gaya} rata-ratanya. Jelaskan, dengan penalaran yang sama, mengapa kamu
menekuk lutut ketika mendarat dari lompatan, dan dari apa matras
pesenam dibuat.
\end{exercise}

\section{Soal: Uji tabrak}

\begin{problem}[{Soal akhir pekan --- merancang tumbukan yang lembut:
karena sebuah tabrakan sudah menetapkan perubahan kecepatannya, setiap
alat keselamatan di dalam mobil adalah mesin untuk memanjangkan waktu}]
\label{pb:g11:forces-and-motion:1}
Sebuah laboratorium uji tabrak menabrakkan mobil pada \qty{50}{km/h} ke
dinding yang kaku, dengan boneka dewasa \qty{70}{kg} dan boneka anak
\qty{20}{kg} di dalamnya; kamera mencatat waktu setiap perhentian.
Tugasmu: menemukan di mana kekejaman sebuah tabrakan bersemayam, lalu
merancangnya agar hilang.

\medskip
\noindent\textbf{Bagian I --- Ukuran sebuah tabrakan.}
\begin{enumerate}
  \item Ubahlah \qty{50}{km/h} ke \unit{m/s}.
  \item Pada setiap perhentian dari depan, perubahan kecepatan
        penumpangnya sama besarnya. Yang mana itu? Mengapa tidak ada
        alat yang dapat mengubahnya?
  \item Kutiplah \cref{thm:g11:forces-and-motion:secondlaw}: dengan
        $\Delta v$ dan $m$ yang tertetapkan, $F\,\Delta t$ pun
        tertetapkan. Lalu apa satu-satunya kenop bagi perancangnya?
  \item Dindingnya menghentikan mobil itu dalam \qty{0.050}{s}; zona
        remuknya memanjangkannya menjadi \qty{0.150}{s}. Bandingkan \omterm{def:g10:inertia:force}{gaya}
        rata-ratanya.
  \item Ke arah mana $\Delta \vect v$ dan \omterm{def:g11:forces-and-motion:netforce}{gaya total} pada bonekanya
        menunjuk selama perhentian itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Zona remuk.}
\begin{enumerate}[resume]
  \item Mengapa, dalam satu kalimat, para perekayasa merancang bagian
        depan mobilnya agar hancur?
  \item Pada mobil kuno yang kaku (berhenti dalam \qty{0.050}{s})
        bonekanya diikat kaku pada kursinya. Bandingkan \omterm{def:g10:inertia:force}{gayanya} dengan
        perhentian modern yang \qty{0.150}{s}.
  \item Laju mobil yang berhenti itu turun secara tetap (rata-ratanya:
        separuh laju awalnya). Sejauh apa mobil itu menempuh perjalanan
        dalam \qty{0.150}{s}? Bandingkan dengan zona remuk yang
        sebenarnya.
  \item Mengapa tidak dibuat zona remuk sepanjang \qty{10}{m}? Dan
        bagian mobil yang mana yang \emph{tidak} boleh remuk?
  \item Rangkumlah Bagian II sebagai sebuah semboyan: ``kamu tidak
        dapat memilih $\Delta v$, jadi kamu harus memilih \dots''.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Sabuk, kantong udara, dan si anak.}
\begin{enumerate}[resume]
  \item Boneka yang tidak bersabuk mempertahankan \qty{14}{m/s} miliknya
        (\cref{ch:g10:inertia}) sampai kaca depannya menghentikannya
        dalam \qty{0.005}{s}; boneka yang bersabuk berhenti bersama
        mobilnya dalam \qty{0.150}{s}. Bandingkan \omterm{def:g10:inertia:force}{gaya} rata-ratanya.
  \item Mengapa sabuk yang sedikit meregang lebih baik daripada tali
        baja yang tidak dapat meregang?
  \item Kepalanya akan berhenti pada setirnya dalam \qty{0.010}{s};
        kantong udaranya memanjangkannya menjadi \qty{0.080}{s}.
        Bandingkan \omterm{def:g10:inertia:force}{gayanya}.
  \item Si dewasa dan si anak mengalami $\Delta v$ yang sama dalam
        $\Delta t$ yang sama. Bandingkan \omterm{def:g11:forces-and-motion:netforce}{gaya total} yang harus dipasok
        kedua sabuknya.
  \item Seorang anak di atas pangkuan: lengannya harus mengubah
        kecepatan bayi \qty{10}{kg} sebesar \qty{14}{m/s} dalam
        \qty{0.150}{s}, sedangkan gravitasi mengubah kecepatan sebesar
        \qty{9.8}{m/s} tiap sekon. Berapa kali lebih cepat lengan itu
        harus bertindak --- dan memangku massa berapa yang akan terasa
        seberat itu? Simpulkan.
  \item Anak merasakan \omterm{def:g10:inertia:force}{gaya} yang lebih kecil --- lalu mengapa harus ada
        kursi anak? Pikirkanlah \emph{di mana} sabuknya mengerjakan
        \omterm{def:g10:inertia:force}{gayanya}, dan pikirkan $\Delta t$.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Sangkar keselamatan, dan putusannya.}
\begin{enumerate}[resume]
  \item Ruang penumpangnya kaku sementara kedua ujungnya remuk. Mengapa
        ruang itu tidak boleh berubah bentuk, padahal kekakuan berarti
        perhentian yang keras?
  \item Dua mobil yang serupa bertabrakan dari depan, masing-masing
        pada \qty{50}{km/h}. Di mana masing-masing berhenti? Bandingkan
        tiap tabrakannya dengan uji dindingnya.
  \item Sebuah mobil kecil bertemu truk bermuatan dari depan. \omterm{def:g10:inertia:force}{Gaya} yang
        dikerjakan keduanya satu sama lain sama besar dan berlawanan,
        timbal balik sebagaimana setiap interaksi
        (\cref{ch:g11:fundamental-interactions}). Kecepatan siapa yang
        lebih banyak berubah?
  \item Daftarkan alat keselamatan pada soal ini beserta satu besaran
        yang dipanjangkan oleh setiap alat itu.
  \item Penutup: nyatakan hukum perancangan tumbukan yang lembut dalam
        satu kalimat, hanya dengan memakai \emph{massa}, \emph{perubahan
        kecepatan}, \emph{\omterm{def:g10:inertia:force}{gaya}} dan \emph{waktu}.
\end{enumerate}
\end{problem}
