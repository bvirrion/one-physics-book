\chapter{Energi Mekanik dan Kekekalannya}\label{ch:g11:mechanical-energy}

Sebuah kereta luncur pasar malam tidak lagi bermesin setelah bukit
pertamanya: setiap putaran, setiap semburan laju sesudahnya dibelanjakan
dari rekening yang dibuka ketika naik. Bab ini mengubah pembukuannya
menjadi dua bilangan --- \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} dan \omterm{def:g10:energy-conservation:energy}{energi} potensial --- serta
satu aturan: selama hanya \omterm{def:g10:universal-gravitation:weight}{berat} yang melakukan \omterm{def:g11:work-of-force:work}{usaha}, jumlah keduanya
tidak berubah. \omterm{def:g10:inertia:inventory}{Gesekan} adalah biayanya; kita akan belajar membacanya
dari bukunya.

\section{Energi kinetik}

\begin{definition}[Energi kinetik]\label{def:g11:mechanical-energy:kinetic}
Sebuah benda bermassa $m$ (\unit{kg}) yang bergerak dengan laju $v$
(\unit{m/s}) membawa \emph{energi kinetik}\index{energi kinetik}
\[
E_k = \tfrac12\, m v^2 .
\]
Pemeriksaan \omterm{def:g10:orders-of-magnitude:unit}{satuannya}: $\unit{kg.m^2/s^2} = \unit{N.m} = \unit{J}$ ---
\omterm{def:g11:work-of-force:work}{joule}, seperti setiap \omterm{def:g10:energy-conservation:energy}{energi} (\cref{ch:g10:energy-conservation}).
\end{definition}

\begin{example}[Orde besar]\label{ex:g11:mechanical-energy:orders}
\begin{itemize}
  \item seorang pejalan kaki, \qty{70}{kg} pada \qty{1.4}{m/s}:
        $E_k = \tfrac12 \times 70 \times 1.4^2 \approx \qty{69}{J}$;
  \item sebuah mobil, \qty{1300}{kg} pada \qty{130}{km/h} $= \qty{36.1}{m/s}$: $E_k \approx \qty{8.5e5}{J}$;
  \item sebuah kereta cepat, $m = \qty{3.85e5}{kg}$, pada
        \qty{320}{km/h} $= \qty{88.9}{m/s}$:
        $E_k \approx \qty{1.5e9}{J}$;
  \item sebutir peluru senapan, \qty{8.0}{g} pada \qty{800}{m/s}:
        $E_k \approx \qty{2.6}{kJ}$.
\end{itemize}
Tujuh \omterm{def:g10:orders-of-magnitude:oom}{orde besar} memisahkan langkah santai dari keretanya. Dan $E_k$
tumbuh seperti \emph{kuadrat} lajunya: pada \qty{130}{km/h} mobil itu
membawa $4$ kali \omterm{def:g10:energy-conservation:energy}{energi} yang dimilikinya pada \qty{65}{km/h} --- empat
kali yang harus dibuang remnya.
\end{example}

\section{Teorema energi kinetik}

\begin{theorem}[Teorema energi kinetik]\label{thm:g11:mechanical-energy:ketheorem}
\index{teorema energi kinetik}
Untuk benda bermassa $m$ yang bergerak sepanjang garis lurus dari $A$ ke
$B$, perubahan \omterm{def:g11:mechanical-energy:kinetic}{energi kinetiknya} sama dengan \omterm{def:g11:work-of-force:work}{usaha} seluruhnya
(\cref{ch:g11:work-of-force}) yang diterimanya:
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
dengan jumlahannya meliputi semua \omterm{def:g10:inertia:force}{gaya} yang bekerja pada benda itu.
Pernyataan itu meluas ke gerak sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} apa pun, yang juga
kita terima di sini.
\end{theorem}
\admitted

\begin{example}[Jatuh bebas, diperiksa bingkai demi bingkai]\label{ex:g11:mechanical-energy:freefall}
Sebuah bola \qty{0.200}{kg} dijatuhkan lalu direkam; bingkainya
memberikan:
\begin{center}\small
\begin{tabular}{lcccc}
$t$ (\unit{s}) & 0.10 & 0.20 & 0.30 & 0.40\\
tinggi jatuh $h$ (\unit{m}) & 0.049 & 0.196 & 0.441 & 0.785\\
laju $v$ (\unit{m/s}) & 0.98 & 1.96 & 2.94 & 3.92\\
$E_k = \tfrac12 mv^2$ (\unit{J}) & 0.096 & 0.384 & 0.864 & 1.54\\
$W = mgh$ (\unit{J}) & 0.096 & 0.385 & 0.865 & 1.54
\end{tabular}
\end{center}
Bermula dari $E_k = 0$, pada tiap bingkainya $E_k$ sepadan dengan \omterm{def:g11:work-of-force:work}{usaha}
$mgh$ oleh \omterm{def:g10:universal-gravitation:weight}{beratnya}: $\Delta E_k = W(\vect P)$, yaitu $v^2 = 2gh$ ---
teoremanya, terbukti sampai ketelitian bacaannya.
\end{example}

\begin{remark}[Di mana buktinya bersemayam]\label{rem:g11:mechanical-energy:proof}
Pernyataan umumnya --- yang berlaku sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} apa pun, bukan
hanya garis lurus --- mengikuti dari hukum kedua Newton beserta sedikit
kalkulus, yang keduanya datang kemudian di dalam buku ini
(\cref{ch:g12:newtons-laws,%
ch:g12:work-and-energy}); sampai saat itu pemeriksaan jatuh bebasnyalah
jaminan kita, dan kita menerima versi \omterm{def:g10:relative-motion:trajectory}{lintasan} melengkungnya lalu
memakainya dengan bebas.
\end{remark}

\section{Energi potensial gravitasi}

\begin{definition}[Energi potensial gravitasi]\label{def:g11:mechanical-energy:potential}
Sebuah benda bermassa $m$ pada ketinggian $z$, yang diukur ke atas dari
\emph{aras acuan}\index{aras acuan} yang dipilih dengan $z = 0$,
menyimpan \emph{energi potensial gravitasi}\index{energi
potensial!gravitasi}
\[
E_p = m g z, \qquad g = \qty{9.81}{N/kg}.
\]
\end{definition}

\begin{remark}[Hanya selisihnya yang berperan]\label{rem:g11:mechanical-energy:reference}
Berpindah dari $A$ ke $B$ mengubah $E_p$ sebesar
$mg(z_B - z_A) =
-W_{AB}(\vect P)$: kebalikan \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:universal-gravitation:weight}{beratnya}
(\cref{ch:g11:work-of-force}). \omterm{def:g11:mechanical-energy:potential}{Aras acuannya} bebas dipilih --- lantai,
permukaan laut, atas meja --- dan menggesernya menggeser $E_p$ di mana
pun dengan sebuah tetapan yang hilang pada setiap selisih. Letakkanlah
aras itu di tempat yang angkanya paling sederhana, dan jangan
memindahkannya di tengah soal.
\end{remark}

\begin{example}[Buku di rak]\label{ex:g11:mechanical-energy:shelf}
Sebuah buku \qty{1.2}{kg} terletak \qty{1.8}{m} di atas lantai, yang
lantainya sendiri \qty{9.0}{m} di atas jalan. Dengan acuan pada
lantainya: $E_p = 1.2 \times 9.81 \times 1.8 \approx \qty{21}{J}$;
dengan acuan pada jalannya: $E_p \approx \qty{127}{J}$. Jatuh ke
lantainya melepaskan \qty{21}{J} pada kedua rekening itu.
\end{example}

\section{Energi mekanik dan kekekalannya}

\begin{definition}[Energi mekanik]\label{def:g11:mechanical-energy:em}
\emph{Energi mekanik}\index{energi mekanik} sebuah benda adalah
\[
E_m = E_k + E_p = \tfrac12 m v^2 + m g z .
\]
\end{definition}

\begin{theorem}[Kekekalan energi mekanik]\label{thm:g11:mechanical-energy:conservation}
\index{kekekalan energi mekanik}
Jika satu-satunya \omterm{def:g10:inertia:force}{gaya} yang melakukan \omterm{def:g11:work-of-force:work}{usaha} pada sebuah benda adalah
\omterm{def:g10:universal-gravitation:weight}{beratnya} (tanpa \omterm{def:g10:inertia:inventory}{gesekan}; \omterm{def:g10:inertia:force}{gaya} yang lain, seperti \omterm{def:g10:inertia:inventory}{gaya normal} dari \omterm{def:g10:relative-motion:trajectory}{lintasan}
tanpa \omterm{def:g10:inertia:inventory}{gesekan}, tegak lurus geraknya), maka $E_m$ bersifat kekal: untuk
dua titik $A$, $B$ mana pun pada geraknya,
\[
\tfrac12 m v_A^2 + m g z_A = \tfrac12 m v_B^2 + m g z_B .
\]
\end{theorem}

\begin{proof}
Menurut teorema \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik}, $E_k(B) - E_k(A) = W_{AB}(\vect P) =
mg(z_A - z_B) = E_p(A) - E_p(B)$: apa yang diperoleh $E_k$ dilepaskan
$E_p$.
\end{proof}

\begin{example}[Seluncuran air]\label{ex:g11:mechanical-energy:slide}
Seorang anak bermula dalam keadaan diam di puncak seluncuran tanpa
\omterm{def:g10:inertia:inventory}{gesekan} setinggi $h = \qty{3.2}{m}$. Dengan acuan pada dasarnya:
$mgh = \tfrac12 mv^2$, sehingga $v = \sqrt{2gh} \approx \qty{7.9}{m/s}$
(\qty{29}{km/h}) --- berapa pun massanya \emph{dan apa pun bentuk
seluncurannya}: lurus, melengkung atau berpilin, hanya selisih
ketinggiannya yang masuk hitungan.
\end{example}

\begin{example}[Bandul]\label{ex:g11:mechanical-energy:pendulum}
Sebuah bandul ditarik ke samping sampai naik $h = \qty{12}{cm}$ di atas
titik terendahnya, lalu dilepaskan. Pada kedua ujungnya $v = 0$; di
dasarnya seluruh $mgh$ menjadi kinetik,
$v = \sqrt{2gh} \approx \qty{1.5}{m/s}$; pada sisi seberangnya bandul
itu naik kembali tepat setinggi \qty{12}{cm}.
\end{example}

\begin{example}[Laju pendaratan lompat ski]\label{ex:g11:mechanical-energy:skijump}

\end{example}

\section{Disipasi oleh gesekan}

\begin{definition}[Disipasi]\label{def:g11:mechanical-energy:dissipation}
\omterm{def:g10:energy-conservation:energy}{Energi} yang meninggalkan rekening mekanisnya dikatakan
\emph{terdisipasi}\index{disipasi}: \omterm{def:g10:energy-conservation:energy}{energi} itu muncul kembali sebagai
\omterm{def:g10:energy-conservation:forms}{energi termal} pada permukaan yang bergesekan dan pada udaranya. \omterm{def:g10:energy-conservation:energy}{Energi}
seluruhnya tetap kekal (\cref{ch:g10:energy-conservation}); \omterm{def:g11:mechanical-energy:em}{energi
mekaniknya} sendirilah yang tidak.
\end{definition}

\begin{proposition}[Selisihnya mengukur gesekan]\label{prop:g11:mechanical-energy:deficit}
Jika sebuah \omterm{def:g10:inertia:inventory}{gaya gesek} $\vect f$ ikut melakukan \omterm{def:g11:work-of-force:work}{usaha} sepanjang
geraknya, maka
\[
\Delta E_m = E_m(B) - E_m(A) = W_{AB}(\vect f) < 0:
\]
kehilangan \omterm{def:g11:mechanical-energy:em}{energi mekaniknya} sama dengan \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekannya}.
\end{proposition}

\begin{proof}
Teorema \omterm{def:g11:mechanical-energy:kinetic}{energi kinetiknya} kini berbunyi
$\Delta E_k = W_{AB}(\vect P) +
W_{AB}(\vect f) = -\Delta E_p + W_{AB}(\vect f)$; pindahkanlah $\Delta E_p$ ke ruas yang lain. \omterm{def:g10:inertia:inventory}{Gesekan}
melawan geraknya, sehingga \omterm{def:g11:work-of-force:work}{usahanya} negatif.
\end{proof}

\begin{example}[Audit kereta luncur salju]\label{ex:g11:mechanical-energy:toboggan}
Seorang anak beserta kereta luncurnya ($m = \qty{40}{kg}$) bermula
dalam keadaan diam, turun $h = \qty{5.0}{m}$ sepanjang \qty{12}{m}
salju, lalu tiba pada \qty{7.0}{m/s} dan bukan pada
$\sqrt{2gh} \approx \qty{9.9}{m/s}$ milik keadaan tanpa \omterm{def:g10:inertia:inventory}{gesekan}. Dengan
acuan pada dasarnya:
$\Delta E_m = \tfrac12 \times 40 \times 7.0^2 - 40 \times 9.81
\times 5.0 = 980 - 1962 = \qty{-982}{J}$. Jadi
$W(\vect f) =
\qty{-982}{J}$ sepanjang \qty{12}{m}: \omterm{def:g10:inertia:inventory}{gaya gesek}
rata-ratanya $f = 982/12 \approx \qty{82}{N}$, dan \qty{982}{J}
kehangatan pada peluncur dan saljunya.
\end{example}

\begin{method}[Pembukuan energi]\label{met:g11:mechanical-energy:bookkeeping}
\begin{enumerate}
  \item Pilihlah sebuah \omterm{def:g11:mechanical-energy:potential}{aras acuan} lalu pertahankan; namailah $A$
        (datanya) dan $B$ (pertanyaannya).
  \item Tulislah $E_m = \tfrac12 mv^2 + mgz$ di $A$ dan di $B$.
  \item Tanpa \omterm{def:g10:inertia:inventory}{gesekan}: $E_m(A) = E_m(B)$; selesaikan --- biasanya
        $v_B = \sqrt{v_A^2 + 2g(z_A - z_B)}$.
  \item Dengan \omterm{def:g10:inertia:inventory}{gesekan}: $E_m(B) - E_m(A) = -f\ell$ ($\ell$ = panjang
        \omterm{def:g10:relative-motion:trajectory}{lintasannya}); selesaikan untuk yang belum diketahui.
  \item Periksalah: batang $E_k$/$E_p$ bertumpuk menjadi $E_m$, yang
        rata kecuali jika ada \omterm{def:g10:inertia:inventory}{gesekan}.
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:1}
Hitunglah \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} (a) sebuah skuter beserta pengendaranya
\qty{90}{kg} pada \qty{25}{km/h}; (b) sebuah mobil \qty{1200}{kg} pada
\qty{130}{km/h}. (c) Dengan faktor berapa $E_k$ berubah ketika lajunya
digandakan? Dan ketika lajunya menjadi tiga kali lipat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:2}
Seorang pendaki \qty{75}{kg} memperoleh ketinggian \qty{850}{m}.
Hitunglah pertambahan \omterm{def:g10:energy-conservation:energy}{energi} potensialnya. Apakah nilainya bergantung
pada \omterm{def:g11:mechanical-energy:potential}{aras acuannya}? Pada rutenya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:3}
Sebuah pot bunga jatuh \qty{20}{m} dari balkon. Dengan mengabaikan
hambatan udara, tentukan lajunya saat mencapai tanah dalam \unit{m/s}
dan \unit{km/h}. Ke mana massanya pergi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:4}
Sebuah mobil \qty{1200}{kg} mengerem dari \qty{25}{m/s} sampai berhenti
dalam \qty{50}{m}. Pakailah teorema \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} untuk menemukan \omterm{def:g11:work-of-force:work}{usaha}
oleh remnya, lalu \omterm{def:g10:inertia:force}{gaya} pengereman rata-ratanya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:mechanical-energy:5}
Sebuah bandul dilepaskan \qty{8.0}{cm} di atas titik terendahnya.
Tentukan lajunya di dasarnya dan tinggi yang dicapainya di sisi
seberangnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:6}
Dengan memakai \cref{ex:g11:mechanical-energy:orders}: dengan faktor
berapa \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} keretanya melampaui \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} pelurunya?
Setinggi apa $E_k$ miliknya sendiri dapat mengangkat kereta itu
($h = v^2/2g$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:7}
Dua seluncuran tanpa \omterm{def:g10:inertia:inventory}{gesekan} turun dari peron \qty{3.0}{m} yang sama,
yang satu lurus dan curam, yang lain panjang dan berkelok. Bandingkan
laju tibanya (hitunglah keduanya) dan waktu tibanya (tanpa
perhitungan).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mechanical-energy:8}
Sebuah bola dilemparkan lurus ke atas dengan \qty{12}{m/s}. Tentukan
tinggi terbesar yang dicapainya, lalu lajunya pada \qty{4.0}{m} ---

\section{Soal: Merancang putarannya}

\begin{problem}[{Soal akhir pekan --- audit energi sebuah kereta luncur
pasar malam: tinggi peluncuran yang dituntut putarannya, laju yang
dijanjikannya, gesekan yang diakui dua pengindra, rem yang menutup
rekeningnya --- seluruh wahananya bergantung pada satu bilangan}]
\label{pb:g11:mechanical-energy:1}
Sebuah rangkaian kereta bermassa $m = \qty{2.0e3}{kg}$ dilepaskan dalam
keadaan diam dari bukit peluncuran setinggi $H$ lalu meluncur tanpa
mesin menembus putaran lingkaran tegak berjari-jari $R = \qty{8.0}{m}$
yang puncaknya berada pada ketinggian $2R$. Telaah gerak melingkar
(tahun depan) memasok satu data yang kita pinjam: pada puncak
putarannya, keretanya baru menekan relnya apabila
$v_{\text{puncak}}^2 \geq gR$. Bagian I dan II mengabaikan \omterm{def:g10:inertia:inventory}{gesekan}.

\medskip
\noindent\textbf{Bagian I --- Bukit yang dituntut putarannya.}
\begin{enumerate}
  \item Hitunglah laju aman terkecil di puncak putarannya.
  \item Dengan tanah sebagai \omterm{def:g11:mechanical-energy:potential}{aras acuannya}: hitunglah \omterm{def:g11:mechanical-energy:em}{energi mekanik}
        keretanya di puncak putaran pada laju terkecil itu.
  \item Simpulkan $H_{\min}$ lalu tunjukkan bahwa $H_{\min} = 5R/2$ ---
        tanpa massa, tanpa $g$.
  \item Timnya mengambil $H = \qty{24}{m}$: hitunglah laju yang
        sebenarnya di puncak putarannya dan nisbah
        $v_{\text{puncak}}^2/(gR)$.
  \item Ke mana $m$ pergi? Mengapa rancangannya cocok baik untuk kereta
        yang penuh maupun yang kosong?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Apa yang dijanjikan tingginya} ($H = \qty{24}{m}$).
\begin{enumerate}[resume]
  \item Berapa lajunya di dasar putarannya ($z = 0$), dalam \unit{m/s}
        dan \unit{km/h}?
  \item Berapa lajunya di sisi putarannya ($z = R$)?
  \item Hitunglah $E_p$ dan $E_k$ pada peluncurannya, di dasar, di sisi
        dan di puncak putarannya; periksalah bahwa tiap pasangnya
        berjumlah $mgH$.
  \item Sebuah gundukan punuk unta \qty{12}{m} menyusul putarannya:
        berapa lajunya di puncak gundukan itu?
  \item Berikan rumus umum $v(z)$; di mana wahana itu paling cepat, dan
        di mana paling lambat?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Audit \omterm{def:g10:inertia:inventory}{gesekannya}.}
Dua pengindra pada ketinggian yang sama $z = \qty{2.0}{m}$, yang
terpisah \qty{60}{m} rel, membaca $v_1 = \qty{20.4}{m/s}$ lalu
$v_2 = \qty{19.2}{m/s}$.
\begin{enumerate}[resume]
  \item Hitunglah perubahan \omterm{def:g11:mechanical-energy:em}{energi mekanik} antara kedua pengindranya.
        Apa yang membuat penempatan pada ketinggian yang sama itu
        cerdik?
  \item Simpulkan \omterm{def:g11:work-of-force:work}{usaha} oleh \omterm{def:g10:inertia:inventory}{gesekannya} dan \omterm{def:g10:inertia:inventory}{gaya gesek} rata-ratanya.
  \item Ke mana \omterm{def:g10:energy-conservation:energy}{energi} yang hilang itu pergi? Sebutkan bentuknya dan
        tempatnya.
  \item Berapa bagian $E_m$ keretanya yang hilang tiap \qty{60}{m}?
        Mengapa $H_{\min}$ yang telanjang akan berbahaya di dunia \omterm{def:g11:lenses-and-eye:images}{nyata}
        yang bergesekan?
  \item Sekitar \qty{90}{m} rel memisahkan peluncurannya dari puncak
        putarannya: taksirlah $E_m$ di sana; apakah kelebihan pada
        pertanyaan 4 masih menyelamatkan putarannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Menutup rekeningnya.}
Pengindra terakhir, pada pintu masuk \omterm{def:g10:relative-motion:trajectory}{lintasan} pengereman mendatar yang
lurus ($z = 0$), membaca \qty{18.0}{m/s}.
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} keretanya di sana.
  \item Remnya harus menghentikannya dalam $d = \qty{45}{m}$: hitunglah
        \omterm{def:g10:inertia:force}{gaya} pengereman (tetap) yang diperlukan.
  \item Pemeriksaan kenyamanannya: hitunglah perlambatannya $a = F/m$;
        bandingkan dengan $g$.
  \item Berapa \omterm{def:g10:energy-conservation:energy}{energi} yang didisipasikan \omterm{def:g10:inertia:inventory}{gesekan} rel sepanjang seluruh
        wahananya? Periksalah bahwa \omterm{def:g10:inertia:inventory}{gesekan} $+$ rem $= mgH$, sampai ke
        \omterm{def:g11:work-of-force:work}{joulenya}.
  \item Puncak ceritanya, dalam satu kalimat: bilangan tunggal yang mana
        yang menetapkan keamanan putarannya, setiap lajunya, tagihan
        \omterm{def:g10:inertia:inventory}{gesekannya} dan \omterm{def:g10:inertia:force}{gaya} remnya --- dan berapa harga mengabaikan
        \omterm{def:g10:inertia:inventory}{gesekan}?
\end{enumerate}
\end{problem}
