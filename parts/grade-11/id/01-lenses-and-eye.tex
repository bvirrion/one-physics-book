\chapter{Lensa, Bayangan, dan Mata}\label{ch:g11:lenses-and-eye}

Setiap gambar yang tajam --- pada pengindra telepon, layar bioskop, atau
di bagian belakang matamu sendiri --- lahir dari satu muslihat yang
sama: medium bening yang melengkung membelokkan sinar melalui \omterm{def:g10:refraction:refraction}{pembiasan}
(\cref{ch:g10:refraction}) sehingga semua sinar yang meninggalkan satu
titik bertemu kembali di titik yang lain. Bab ini menjinakkan muslihat
itu: satu \omterm{def:g11:lenses-and-eye:lens}{lensa}, tiga sinar, satu rumus --- lalu mengarahkannya ke
kamera, kacamata, dan \omterm{def:g11:lenses-and-eye:lens}{lensa} hidup yang sedang membaca kalimat ini.

\section{Lensa cembung dan lensa cekung}

\begin{definition}[Lensa tipis]\label{def:g11:lenses-and-eye:lens}
Sebuah \emph{lensa}\index{lensa} adalah medium bening yang dibatasi dua
permukaan, sekurang-kurangnya satu di antaranya melengkung. Jika
bagian tengahnya lebih tebal daripada tepinya, lensa itu
\emph{cembung}\index{lensa cembung}: sinar sejajar dibelokkan saling
mendekat; jika tepinya lebih tebal, lensa itu
\emph{cekung}\index{lensa cekung}: sinarnya memencar. Lensa
\emph{tipis}\index{lensa tipis} adalah lensa yang tebalnya dapat
diabaikan: kita menggambarnya sebagai sebuah ruas dengan mata panah
mengarah ke luar (cembung) atau ke dalam (cekung). Pusatnya $O$ disebut
\emph{pusat optik}\index{pusat optik}; garis yang tegak lurus melalui
$O$ disebut \emph{sumbu utama}\index{sumbu utama}.
\end{definition}

\begin{definition}[Titik fokus dan jarak fokus]\label{def:g11:lenses-and-eye:focal}
Sinar yang tiba sejajar sumbu sebuah \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} semuanya memotong
sumbunya di satu titik di belakangnya, yaitu \emph{titik fokus
bayangan}\index{titik fokus} $F'$; secara setangkup, sinar yang
meninggalkan \emph{titik fokus benda} $F$ (di depan, dengan
$OF = OF'$) keluar sejajar sumbunya. \emph{Jarak fokus}\index{jarak
fokus} $f' = \overline{OF'}$ bernilai positif untuk \omterm{def:g11:lenses-and-eye:lens}{lensa cembung};
melalui \omterm{def:g11:lenses-and-eye:lens}{lensa cekung}, sinar sejajar keluar seolah-olah berasal dari
sebuah titik $F'$ \emph{di depan} \omterm{def:g11:lenses-and-eye:lens}{lensanya}: $f' < 0$.
\end{definition}

\begin{definition}[Kekuatan lensa]\label{def:g11:lenses-and-eye:vergence}
\emph{Kekuatan lensa}\index{kekuatan lensa} yang berjarak fokus $f'$
(dalam \omterm{def:g10:orders-of-magnitude:unit}{meter}) adalah $C = 1/f'$, yang diukur dalam
\emph{dioptri}\index{dioptri} (lambang D):
$1~\text{D} = \qty{1}{m^{-1}}$; \omterm{def:g11:lenses-and-eye:lens}{lensa cekung} memiliki $C < 0$. Ahli
kacamata meresepkan dalam dioptri: kekuatan \omterm{def:g11:lenses-and-eye:lens}{lensa tipis} yang saling
bersentuhan cukup dijumlahkan (diterima di sini; diturunkan pada jilid
Tahun ke-1).
\end{definition}

\begin{example}[Membaca sebuah resep]\label{ex:g11:lenses-and-eye:vergence}
\omterm{def:g11:lenses-and-eye:lens}{Lensa} dengan $f' = \qty{0.50}{m}$ memiliki $C = 1/0.50 = +2.0$~D:
\omterm{def:g11:lenses-and-eye:lens}{cembung}. Resep $-4.0$~D berarti $f' = \qty{-0.25}{m}$: \omterm{def:g11:lenses-and-eye:lens}{cekung}, dengan
\omterm{def:g11:lenses-and-eye:focal}{jarak fokus} \qty{25}{cm}.
\end{example}

\section{Melukis bayangannya}

\begin{method}[Tiga sinar istimewa]\label{met:g11:lenses-and-eye:rays}
Misalkan $AB$ adalah sebuah benda yang tegak lurus sumbu di titik $A$.
Di antara semua sinar yang meninggalkan ujung $B$, ada tiga yang
\omterm{def:g10:relative-motion:trajectory}{lintasannya} sudah diketahui:
\begin{enumerate}
  \item sinar yang melewati \omterm{def:g11:lenses-and-eye:lens}{pusat optik} $O$ diteruskan lurus;
  \item sinar yang sejajar sumbu keluar melalui $F'$;
  \item sinar yang melewati $F$ keluar sejajar sumbunya.
\end{enumerate}
Dua di antaranya sudah memadai: titik potongnya adalah titik bayangan
$B'$, dan bayangan $A'B'$ berdiri tegak lurus sumbu di titik $A'$.
Apabila sinar yang keluar memencar, perpanjangan \emph{ke belakang}
sinar itu (garis putus-putus) bertemu di sebuah titik \omterm{def:g11:lenses-and-eye:images}{bayangan maya}.
\end{method}

\begin{definition}[Bayangan nyata dan bayangan maya]\label{def:g11:lenses-and-eye:images}
Sebuah bayangan disebut \emph{nyata}\index{bayangan nyata} apabila sinar
yang keluar benar-benar melewatinya: layar yang ditempatkan di sana
menangkapnya; disebut \emph{maya}\index{bayangan maya} apabila hanya
perpanjangan ke belakangnya yang bertemu: tidak ada layar yang
menangkapnya, tetapi mata yang memandang ke dalam \omterm{def:g11:lenses-and-eye:lens}{lensanya} melihatnya
dengan sangat baik. \omterm{def:g11:lenses-and-eye:lens}{Lensa cembung} memberikan bayangan nyata dan
terbalik untuk setiap benda di luar $F$; geserlah bendanya ke dalam
jarak fokusnya dan bayangannya berubah menjadi maya, tegak, dan
diperbesar.
\end{definition}

\section{Satu rumus: hubungan benda dan bayangan}

Lukisan menunjukkan \emph{di mana} bayangannya berada; sebuah rumus
menghitungnya. Arahkanlah sumbunya searah cahaya lalu pakailah
\emph{ukuran beraljabar}: $\overline{OA}$ adalah koordinat $A$ terhadap
$O$ (bernilai negatif untuk benda \omterm{def:g11:lenses-and-eye:images}{nyata} di depan); tinggi
$\overline{AB}$ dihitung positif ke atas.

\begin{theorem}[Rumus lensa tipis]\label{thm:g11:lenses-and-eye:conjugate}
Melalui \omterm{def:g11:lenses-and-eye:lens}{lensa tipis} berjarak fokus $f'$, sebuah titik benda $A$ pada
sumbunya membentuk bayangan di titik $A'$ dengan
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'} = C .
\]
\end{theorem}
\admitted

\begin{remark}
Hubungan itu mengikuti dari \omterm{thm:g10:refraction:snell}{hukum Snell} yang diterapkan pada sinar yang
dekat dengan sumbunya; penurunannya --- beserta batas berlakunya model
\omterm{def:g11:lenses-and-eye:lens}{lensa tipis} --- dibuat jujur pada jilid Tahun ke-1.
\end{remark}

\begin{definition}[Perbesaran]\label{def:g11:lenses-and-eye:magnification}
\emph{Perbesaran}\index{perbesaran} sebuah bayangan adalah
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}}
       = \frac{\overline{OA'}}{\overline{OA}} .
\]
$\abs{\gamma}$ membandingkan ukurannya, sedangkan tandanya
membandingkan arahnya ($\gamma < 0$: terbalik); tanda
$\overline{OA'}$ membaca sifatnya (positif: \omterm{def:g11:lenses-and-eye:images}{nyata}, di belakang \omterm{def:g11:lenses-and-eye:lens}{lensanya};
negatif: \omterm{def:g11:lenses-and-eye:images}{maya}, di depannya).
\end{definition}

\begin{example}[Rumus itu bekerja]\label{ex:g11:lenses-and-eye:conjugate}
Sebuah benda berdiri \qty{30}{cm} di depan \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} dengan
$f' = \qty{10}{cm}$: $\overline{OA} = \qty{-30}{cm}$, sehingga
\[
\frac{1}{\overline{OA'}} = \frac{1}{10} - \frac{1}{30} = \frac{1}{15},
\qquad \overline{OA'} = +\qty{15}{cm},
\qquad \gamma = \frac{15}{-30} = -0.50 :
\]
sebuah \omterm{def:g11:lenses-and-eye:images}{bayangan nyata} \qty{15}{cm} di belakang \omterm{def:g11:lenses-and-eye:lens}{lensanya}, terbalik,
setengah ukurannya --- persis seperti yang dilukis ketiga sinar tadi.
\end{example}

\section{Mata}

\begin{definition}[Mata sebagai sistem optik]\label{def:g11:lenses-and-eye:eye}
Secara optis, mata adalah \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} yang menghadap sebuah layar:
\emph{kornea}\index{kornea} dan \emph{lensa mata}\index{lensa mata}
bekerja sebagai satu \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} yang jarak fokusnya dapat diatur;
\emph{retina}\index{retina}, sekitar \qty{17}{mm} di belakangnya,
adalah layarnya yang tetap. Pemfokusannya dilakukan dengan memencet
\omterm{def:g11:lenses-and-eye:lens}{lensanya}, bukan dengan menggesernya seperti pada kamera:
\emph{akomodasi}\index{akomodasi} adalah perubahan $f'$ yang dihasilkan
otot siliar. Titik terdekat yang terlihat tajam pada akomodasi penuh
disebut \emph{titik dekat}\index{titik dekat} --- sekitar \qty{25}{cm}
untuk \emph{mata normal}\index{mata normal} --- dan yang terjauh, saat
mata beristirahat, disebut \emph{titik jauh}\index{titik jauh} (tak
terhingga untuk mata normal).
\end{definition}

\omterm{def:g11:lenses-and-eye:eye}{Retinanya} berada pada $\overline{OA'} = +\qty{0.017}{m}$. Untuk benda

\end{example}

\section{Cacat mata, koreksinya, dan lup}

\begin{definition}[Miopi dan hipermetropi]\label{def:g11:lenses-and-eye:defects}
Mata \emph{miopi}\index{miopi} (rabun jauh) terlalu kuat untuk panjang
bola matanya: sinar sejajar memusat \emph{di depan} \omterm{def:g11:lenses-and-eye:eye}{retinanya}, benda
jauh menjadi kabur, dan \omterm{def:g11:lenses-and-eye:eye}{titik jauhnya} berada pada jarak berhingga;
\omterm{def:g11:lenses-and-eye:lens}{lensa} kacamata yang \omterm{def:g11:lenses-and-eye:lens}{cekung} mengoreksinya. Mata
\emph{hipermetropi}\index{hipermetropi} (rabun dekat) kurang kuat:
benda dekat akan memusat di belakang \omterm{def:g11:lenses-and-eye:eye}{retinanya} dan \omterm{def:g11:lenses-and-eye:eye}{titik dekatnya}
menjauh; \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} mengoreksinya. Bertambahnya usia mengakukan
\omterm{def:g11:lenses-and-eye:eye}{lensa matanya} (\emph{presbiopi}\index{presbiopi}) dan menyusutkan
\omterm{def:g11:lenses-and-eye:eye}{akomodasinya} --- dengan obat yang sama: kacamata baca.
\end{definition}

\begin{example}[Meresepkan untuk mata miopi]\label{ex:g11:lenses-and-eye:correction}
Sebuah mata \omterm{def:g11:lenses-and-eye:defects}{miopi} memiliki \omterm{def:g11:lenses-and-eye:eye}{titik jauh} pada \qty{50}{cm}: \omterm{def:g11:lenses-and-eye:lens}{lensa}
pengoreksinya harus menampilkan benda jauh \emph{di \omterm{def:g11:lenses-and-eye:eye}{titik jauh} itu} ---
bendanya di tak terhingga, bayangannya di
$\overline{OA'} = \qty{-0.50}{m}$ (\omterm{def:g11:lenses-and-eye:images}{maya}, di depan). Maka
$1/f' = 1/\overline{OA'}$: $f' = \qty{-0.50}{m}$, $C = -2.0$~D; mata
yang beristirahat memandang \omterm{def:g11:lenses-and-eye:images}{bayangan maya} itu dan melihatnya dengan
tajam.
\end{example}

\begin{definition}[Lup]\label{def:g11:lenses-and-eye:magnifier}
Sebuah \emph{lup}\index{lup} adalah \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} berjarak fokus pendek
yang dipakai dengan bendanya \emph{di dalam} jarak fokusnya: bayangannya
\omterm{def:g11:lenses-and-eye:images}{maya}, tegak, dan diperbesar (\cref{def:g11:lenses-and-eye:images}), dan
mata memandangnya dengan nyaman dari jauh.
\end{definition}

\begin{example}[Lup seorang pengasah permata]\label{ex:g11:lenses-and-eye:magnifier}
\omterm{def:g11:lenses-and-eye:magnifier}{Lup} dengan $f' = \qty{5.0}{cm}$, batunya pada \qty{4.0}{cm}:
$1/\overline{OA'} = 1/5.0 - 1/4.0 = -1/20$, sehingga
$\overline{OA'} = \qty{-20}{cm}$ dan $\gamma = (-20)/(-4.0) = +5.0$:
\omterm{def:g11:lenses-and-eye:images}{bayangan maya}, tegak, lima kali lebih besar, pada jarak \qty{20}{cm}
dari \omterm{def:g11:lenses-and-eye:lens}{lensanya} yang nyaman dipandang.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:lenses-and-eye:1}
Hitunglah \omterm{def:g11:lenses-and-eye:vergence}{kekuatan lensa} untuk $f' = \qty{50}{cm}$, $\qty{-20}{cm}$ dan
$\qty{4.0}{mm}$, lalu \omterm{def:g11:lenses-and-eye:focal}{jarak fokus} sebuah \omterm{def:g11:lenses-and-eye:lens}{lensa} $+8.0$~D.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:lenses-and-eye:2}
Sebuah \omterm{def:g11:lenses-and-eye:lens}{lensa} tepinya lebih tebal daripada tengahnya. \omterm{def:g11:lenses-and-eye:lens}{Cembung} ataukah
\omterm{def:g11:lenses-and-eye:lens}{cekung}? Dapatkah \omterm{def:g11:lenses-and-eye:lens}{lensa} itu dipakai sebagai \omterm{def:g11:lenses-and-eye:magnifier}{lup}? Apa yang kamu lihat
ketika memegangnya di atas halaman cetakan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:lenses-and-eye:3}
Sebuah \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} memiliki $f' = \qty{10}{cm}$; sebuah benda berdiri
\qty{20}{cm} di depannya. Lukislah bayangannya dengan dua dari ketiga
sinar itu, lalu periksalah kedudukan dan ukurannya dengan rumus
\omterm{def:g11:lenses-and-eye:lens}{lensanya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:lenses-and-eye:4}
Sebuah benda berdiri \qty{24}{cm} di depan \omterm{def:g11:lenses-and-eye:lens}{lensa cembung} dengan
$f' = \qty{8.0}{cm}$. Hitunglah $\overline{OA'}$ dan $\gamma$, lalu
lukiskan bayangannya (sifat, arah, ukuran).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:lenses-and-eye:5}
Di dalam mata: apa yang membentuk \omterm{def:g11:lenses-and-eye:lens}{lensa cembungnya}? Apa yang berperan
sebagai layarnya? Apa yang berubah selama \omterm{def:g11:lenses-and-eye:eye}{akomodasi} --- dan apa yang
tidak dapat berubah? Definisikan \omterm{def:g11:lenses-and-eye:eye}{titik dekat} \omterm{def:g11:lenses-and-eye:eye}{mata normal}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:lenses-and-eye:15}

\section{Soal: Satu rumus, tiga alat}

\begin{problem}[{Soal akhir pekan --- rumus lensa yang sama memfokuskan
sebuah kamera, meresepkan kacamata baca, merancang sebuah lup, dan
mengukur zum yang sudah terpasang di dalam matamu sendiri}]
\label{pb:g11:lenses-and-eye:1}
Seorang juru foto memutar \omterm{def:g11:lenses-and-eye:lens}{lensa} \qty{50}{mm}, seorang nenek memegang
surat kabar sepanjang lengannya, seorang pengasah permata membungkuk di
atas sebutir intan: satu baris aljabar,
$1/\overline{OA'} - 1/\overline{OA} = 1/f'$
(\cref{thm:g11:lenses-and-eye:conjugate}), menjalankan ketiga pekerjaan
itu --- lalu mata itu sendiri.

\medskip
\noindent\textbf{Bagian I --- Satu rumus.} Sebuah \omterm{def:g11:lenses-and-eye:lens}{lensa cembung}
memiliki $f' = \qty{20}{cm}$.
\begin{enumerate}
  \item Sebuah benda pada \qty{60}{cm}: hitunglah $\overline{OA'}$ dan
        $\gamma$; lukiskan bayangannya.
  \item Pindahkan bendanya ke \qty{30}{cm}; hitunglah kembali. Apa yang
        bertukar dengan pertanyaan 1? (\omterm{def:g10:relative-motion:trajectory}{Lintasan} cahaya dapat dibalik.)
  \item Pindahkan ke \qty{10}{cm}, di dalam $f'$: hitunglah kembali dan
        lukiskan.
  \item Rangkumlah: ketika bendanya meluncur masuk dari jauh sekali
        menuju $F$, lalu melewatinya, bagaimana bayangannya bergerak
        dan berubah sifat?
  \item Tunjukkan bahwa benda yang sangat jauh
        ($1/\overline{OA} \approx 0$) selalu membentuk bayangan pada
        bidang fokusnya: $\overline{OA'} = f'$.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Juru foto.}
Sebuah \omterm{def:g11:lenses-and-eye:lens}{lensa} kamera memiliki $f' = \qty{50}{mm}$; pengindranya, yang
tingginya \qty{24}{mm}, dapat digeser terhadap \omterm{def:g11:lenses-and-eye:lens}{lensanya}.
\begin{enumerate}[resume]
  \item Di mana pengindranya ketika difokuskan pada bentang alam yang
        jauh?
  \item Kini sebuah wajah sejauh \qty{1.0}{m}: hitunglah jarak
        lensa--pengindra yang baru dan pergeserannya dari setelan
        bentang alam tadi.
  \item Mekanismenya hanya mengizinkan jarak lensa--pengindra paling
        besar \qty{55}{mm}. Hitunglah jarak pemfokusan terkecilnya.
  \item Pada pemfokusan terdekat itu, hitunglah $\gamma$. Apakah
        bayangan wajah selebar \qty{20}{cm} muat pada pengindranya?
  \item ``Makro'' seukuran aslinya berarti $\gamma = -1$: di mana
        bendanya dan pengindranya harus berada, dan mengapa hal itu
        menuntut \omterm{def:g11:lenses-and-eye:lens}{lensa} yang khusus?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Sang nenek.}
Seiring bertambahnya usia \omterm{def:g11:lenses-and-eye:eye}{titik dekat} menjauh: \omterm{def:g11:lenses-and-eye:eye}{titik dekat} sang nenek
berada pada \qty{1.0}{m}.
\begin{enumerate}[resume]
  \item Jelaskan surat kabar sepanjang lengan itu.
  \item Untuk membaca pada \qty{25}{cm}, kacamatanya harus mengubah
        halaman pada \qty{25}{cm} menjadi \omterm{def:g11:lenses-and-eye:images}{bayangan maya} pada
        \qty{1.0}{m}: hitunglah \omterm{def:g11:lenses-and-eye:vergence}{kekuatan lensa} yang diperlukan.
  \item Hitunglah $\gamma$. Bayangannya empat kali lebih besar, namun
        tidak tampak lebih besar: jelaskan, dengan membandingkan
        ukurannya \emph{dan} jaraknya.
  \item Tunjukkan bahwa cetakannya hanya tajam antara \qty{25}{cm} dan
        $f' \approx \qty{33}{cm}$. Bagaimana dengan benda yang
        \emph{jauh} jika dipandang melalui kacamata itu?
  \item Simpulkan dalam satu kalimat mengapa ada \omterm{def:g11:lenses-and-eye:lens}{lensa} dwifokus.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pengasah permata, dan zum milik mata.}
\omterm{def:g11:lenses-and-eye:magnifier}{Lup} si pengasah permata memiliki $f' = \qty{5.0}{cm}$; \omterm{def:g11:lenses-and-eye:eye}{titik dekatnya}
pada \qty{25}{cm}; \omterm{def:g11:lenses-and-eye:eye}{retinanya} \qty{17}{mm} di belakang \omterm{def:g11:lenses-and-eye:eye}{lensa matanya}.
\begin{enumerate}[resume]
  \item Sebutir batu \qty{4.0}{cm} di bawah \omterm{def:g11:lenses-and-eye:magnifier}{lupnya}: berapa kedudukan,
        sifat, dan \omterm{def:g11:lenses-and-eye:magnification}{perbesarannya}?
  \item Di mana batunya harus diletakkan agar bayangannya jatuh di
        \omterm{def:g11:lenses-and-eye:eye}{titik dekatnya}, dan berapa $\gamma$ pada saat itu?
  \item Mengapa pengasah permata yang berpengalaman justru meletakkan
        batunya tepat di $F$ (bayangannya di tak terhingga)?
  \item Kini matanya sendiri: hitunglah kekuatannya ketika memandang
        benda yang jauh, lalu benda pada \qty{25}{cm}: berapa \omterm{def:g11:lenses-and-eye:vergence}{dioptri}
        \omterm{def:g11:lenses-and-eye:eye}{akomodasinya}?
  \item Ubahlah kedua kekuatannya menjadi \omterm{def:g11:lenses-and-eye:focal}{jarak fokus}: sebanyak apa,
        dalam milimeter dan dalam persen, mata mengubah jarak fokusnya
        sendiri --- yaitu zum yang dilakukan kamera dengan pergeseran
        dan sang nenek dengan $+3$~D?
\end{enumerate}
\end{problem}
