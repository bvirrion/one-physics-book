\chapter{Warna dan Sumber Cahaya}\label{ch:g11:color-light-sources}

Tanyalah kepada seorang fisikawan apa warna sebuah tomat dan kamu akan
menerima pertanyaan balik: di bawah cahaya yang mana? Tomat itu hanya
\omterm{def:g11:color-light-sources:objectcolor}{menghamburkan} apa yang diberikan kepadanya, lampunya hanya memberikan
apa yang termuat pada \omterm{def:g10:light-spectra:white}{spektrumnya}, dan mata memampatkan apa pun yang
tiba menjadi tiga bilangan. Warna adalah perundingan antara sumber,
benda dan mata --- bab ini menyidik ketiga pihaknya satu per satu.

\section{Melihat dalam tiga bilangan}

\begin{definition}[Sel kerucut dan penglihatan trikromatik]\label{def:g11:color-light-sources:cones}
\omterm{def:g11:lenses-and-eye:eye}{Retina} (\cref{ch:g11:lenses-and-eye}) membawa dua keluarga sel yang peka
cahaya: \emph{sel batang}\index{sel batang}, yang menanggapi cahaya
redup tetapi tidak menghiraukan \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} (penglihatan malam,
berwarna kelabu), dan \emph{sel kerucut}\index{sel kerucut}, yang
terdiri atas tiga jenis dan paling peka di dekat $\qty{420}{nm}$
(biru), $\qty{530}{nm}$ (hijau) dan $\qty{560}{nm}$ (merah).
Penglihatan warna manusia bersifat
\emph{trikromatik}\index{penglihatan trikromatik}: apa pun \omterm{def:g10:light-spectra:white}{spektrumnya},
otak hanya menerima tiga tanggapan sel kerucut, dan warna adalah apa
yang dibuatnya dari ketiga bilangan itu.
\end{definition}

\begin{definition}[Warna sebagai tanggapan indra]\label{def:g11:color-light-sources:perception}
Karena itu warna adalah sebuah \emph{tanggapan indra}, bukan sifat
cahaya semata: \omterm{def:g10:light-spectra:white}{spektrum} menyebutkan intensitas pada setiap \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang}, sedangkan mata hanya menyimpan tiga bilangan. Cahaya yang
\omterm{def:g10:light-spectra:white}{spektrumnya} berbeda tetapi tanggapan ketiga \omterm{def:g11:color-light-sources:cones}{sel kerucutnya} sama tampak
benar-benar serupa: keduanya disebut \emph{metamer}\index{metamer}.
\end{definition}

\begin{example}[Dua warna kuning]\label{ex:g11:color-light-sources:yellows}
Garis $\qty{589}{nm}$ sebuah lampu natrium merangsang \omterm{def:g11:color-light-sources:cones}{sel kerucut} merah
dan hijau hampir sama kuatnya: otak berkata \emph{kuning}. Sebuah layar
yang menyalakan subpiksel merahnya ($\qty{610}{nm}$) dan hijaunya
($\qty{540}{nm}$) memberikan kuning yang sama tanpa cahaya apa pun di
dekat $\qty{589}{nm}$. Prisma memisahkan keduanya seketika; mata tidak
akan pernah bisa.
\end{example}

\section{Menjumlahkan cahaya}

\begin{definition}[Perpaduan aditif]\label{def:g11:color-light-sources:additive}
Menumpangkan berkas cahaya pada layar putih berarti menjumlahkan
\omterm{def:g10:light-spectra:white}{spektrumnya}: itulah \emph{perpaduan aditif}\index{perpaduan aditif}.
Tiga cahaya yang dipilih dengan baik, merah, hijau dan biru --- yaitu
\emph{warna primer aditif}\index{warna primer} --- bercampur menjadi
hampir setiap warna yang dapat ditangkap indra: R $+$ G $=$ kuning,
R $+$ B $=$ magenta, G $+$ B $=$ sian, dan R $+$ G $+$ B $=$ putih.
\end{definition}

\begin{proposition}[Warna komplementer]\label{prop:g11:color-light-sources:complementary}
Dua cahaya yang jumlahnya ditangkap sebagai putih disebut saling
\emph{komplementer}\index{warna komplementer}. Ketiga pasangannya
kuning/biru, sian/merah dan magenta/hijau saling \omterm{prop:g11:color-light-sources:complementary}{komplementer}.
\end{proposition}

\begin{proof}
Kuning adalah R $+$ G; menambahkan biru melengkapi ketiganya menjadi
putih. Kedua pasangan yang lain memakai alasan yang sama, satu \omterm{def:g11:color-light-sources:additive}{warna
primer} setiap kali.
\end{proof}

\begin{example}[Layar]\label{ex:g11:color-light-sources:screen}
Tiap piksel layar telepon terdiri atas tiga subpiksel --- merah, hijau,
biru, dengan lebar di bawah $\qty{0.1}{mm}$ --- yang terlalu rapat
untuk dipisahkan mata, sehingga cahayanya berjumlah pada \omterm{def:g11:lenses-and-eye:eye}{retina}: kuning
berarti R dan G menyala, jingga berarti R menyala penuh dan G separuh,
putih berarti ketiganya menyala. Sebuah kaca pembesar di atas daerah
yang putih menyingkap muslihatnya: tidak ada putih di mana pun, hanya R,
G, B.
\end{example}

\section{Mengurangkan dari cahaya putih}

\begin{definition}[Perpaduan subtraktif]\label{def:g11:color-light-sources:subtractive}
Sebuah \emph{filter}\index{filter} (kaca berwarna, lembar gel, lapisan
tinta atau cat) meneruskan sebagian \omterm{def:g10:light-spectra:white}{spektrum} yang diterimanya dan
menyerap sisanya. Menghasilkan warna dengan \emph{membuang} pita dari
\omterm{def:g10:light-spectra:white}{cahaya putih} disebut \emph{perpaduan
subtraktif}\index{perpaduan subtraktif}. \omterm{def:g11:color-light-sources:additive}{Warna primernya}, yang
\omterm{prop:g11:color-light-sources:complementary}{komplementer} terhadap \omterm{def:g11:color-light-sources:additive}{warna primer aditif}, masing-masing mengurangkan
satu pita: sian $=$ putih $-$ R, magenta $=$ putih $-$ G, kuning $=$
putih $-$ B. Filter yang bertumpuk mengurangkan bersama-sama: sian $+$
kuning akan menyisakan hijau, sian $+$ magenta menyisakan biru,
lalu magenta $+$ kuning menyisakan merah, sedangkan ketiganya
menyisakan hitam. Percetakan dan lukisan bekerja dengan cara ini,
dengan memakai tinta C, M, Y.
\end{definition}

\begin{method}[Pembukuan spektrum]\label{met:g11:color-light-sources:bookkeeping}
Untuk meramalkan warna melalui rangkaian sumber, \omterm{def:g11:color-light-sources:subtractive}{filter} dan benda apa
pun, ikutilah ketiga pita primernya: tulislah cahaya sumbernya sebagai
kandungan R, G, B-nya; pada tiap \omterm{def:g11:color-light-sources:subtractive}{filter} atau permukaan, hapuslah pita
yang terserap; lalu sebutkan warna dari apa yang bertahan (tidak ada
yang tersisa $=$ hitam).
\end{method}

\begin{example}[Rangkaian filter]\label{ex:g11:color-light-sources:chain}
\omterm{def:g10:light-spectra:white}{Cahaya putih} $(R, G, B)$ melewati \omterm{def:g11:color-light-sources:subtractive}{filter} kuning: pita birunya terserap
dan menyisakan $(R, G)$ --- kuning. Sebuah \omterm{def:g11:color-light-sources:subtractive}{filter} sian kemudian
menyerap pita merahnya dan menyisakan $(G)$: hijau. Menukar urutan
\omterm{def:g11:color-light-sources:subtractive}{filternya} tidak mengubah apa pun: masing-masing menyerap pitanya
sendiri tanpa menghiraukan urutannya.
\end{example}

\section{Warna sebuah benda}

\begin{definition}[Penghamburan dan warna benda]\label{def:g11:color-light-sources:objectcolor}
Benda tak tembus cahaya menyerap sebagian cahaya yang diterimanya lalu
\emph{menghamburkan}\index{penghamburan} sisanya --- memancarkannya
kembali ke segala arah, di antaranya ke mata kita. \emph{Warna sebuah
benda}\index{warna benda} adalah warna cahaya yang dihamburkannya
\emph{di bawah penerangan yang diberikan}: bendanya menyumbangkan
kebiasaan menyerapnya, sumbernya menyumbangkan bahan mentahnya. Gantilah
lampunya, gantilah warnanya.
\end{definition}

\begin{example}[Kemeja merah]\label{ex:g11:color-light-sources:redshirt}
Sebuah kemeja tampak merah di bawah cahaya siang: kemeja itu menyerap
pita hijau dan birunya lalu \omterm{def:g11:color-light-sources:objectcolor}{menghamburkan} pita merahnya. Di bawah
cahaya hijau murni tidak ada pita merah untuk dihamburkan: kemeja itu
menyerap semuanya lalu tampak hitam. Di bawah cahaya magenta $(R, B)$
kemeja itu \omterm{def:g11:color-light-sources:objectcolor}{menghamburkan} pita merahnya saja: tetap merah. Benda putih
\omterm{def:g11:color-light-sources:objectcolor}{menghamburkan} setiap pita yang diterimanya (di bawah cahaya hijau ia
tampak hijau); benda hitam menyerap semuanya di bawah cahaya apa pun
--- dan itu pula sebabnya pakaian hitam menjadi panas di bawah
matahari.
\end{example}

\section{Sumber cahaya dan suhu warna}

\begin{definition}[Sumber pijar]\label{def:g11:color-light-sources:incandescent}
Sumber yang bercahaya karena panas --- nyala api, kawat pijar, bintang
--- bersinar melalui \emph{pemijaran}\index{pemijaran}. \omterm{def:g10:light-spectra:white}{Spektrumnya}
adalah \omterm{def:g10:light-spectra:white}{spektrum} termal kontinu pada \cref{ch:g10:light-spectra}, dan
puncaknya menaati hukum Wien
(\cref{prop:g10:light-spectra:wien}),
\[
\lambda_{\max} \times T = \qty{2.90e-3}{m.K},
\]
mulai sekarang menjadi perkakas berangka kita: puncaknya terukur,
suhunya diketahui.
\end{definition}

\begin{example}[Lampu yang sangat tidak efisien]\label{ex:g11:color-light-sources:bulb}
Kawat pijar wolfram pada $T = \qty{2700}{K}$ memuncak pada
$\lambda_{\max} = \qty{2.90e-3}{m.K}/\qty{2700}{K} = \qty{1.07e-6}{m}
= \qty{1070}{nm}$, jauh di daerah \omterm{def:g10:light-spectra:wavelength}{inframerah}: sebagian besar \omterm{def:g10:energy-conservation:power}{daya}
listriknya pergi sebagai radiasi yang tak terlihat, hanya sekitar $5\%$
sebagai cahaya --- bohlam pijar pada dasarnya sebuah pemanas.
\end{example}

\begin{definition}[Lampu spektrum dan laser]\label{def:g11:color-light-sources:lamps}
Sebuah \emph{lampu spektrum}\index{lampu spektrum} --- gas bertekanan
rendah yang dieksitasi oleh lucutan listrik: natrium, raksa, neon ---
memancarkan \omterm{def:g10:light-spectra:emission}{spektrum garis} (\cref{ch:g10:light-spectra}): beberapa
\omterm{def:g10:light-spectra:wavelength}{panjang gelombang} terpencil, dan tidak ada apa pun di antaranya. Sebuah
\emph{laser}\index{laser} melangkah lebih jauh: satu \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}
tunggal, \omterm{def:g10:light-spectra:white}{cahaya monokromatik}, di dalam satu berkas yang sempit.
\end{definition}

\begin{definition}[LED]\label{def:g11:color-light-sources:led}
Sebuah \emph{dioda pemancar cahaya}\index{dioda pemancar cahaya} (LED)
memancarkan pita \emph{kuasi-monokromatik}\index{kuasi-monokromatik}
selebar beberapa puluh nanometer: cukup sempit untuk tampak berwarna
tertentu, jauh lebih lebar daripada garis sebuah \omterm{def:g11:color-light-sources:lamps}{laser}. LED putih adalah
LED biru (puncaknya di dekat $\qty{450}{nm}$) yang dilapisi
\emph{fosfor}\index{fosfor}, yaitu bahan yang menyerap sebagian birunya
lalu memancarkannya kembali sebagai pita kuning yang lebar: karena biru
dan kuning saling \omterm{prop:g11:color-light-sources:complementary}{komplementer}
(\cref{prop:g11:color-light-sources:complementary}), campurannya
terbaca sebagai putih.
\end{definition}

\begin{definition}[Suhu warna]\label{def:g11:color-light-sources:colortemp}
\emph{Suhu warna}\index{suhu warna} sebuah sumber yang keputihan adalah
suhu benda pijar yang rona cahayanya sepadan dengan cahaya sumber itu:
lilin $\qty{1800}{K}$, bohlam pijar $\qty{2700}{K}$, halogen
$\qty{3400}{K}$, matahari tengah hari $\qty{5500}{K}$, cahaya siang yang
mendung $\qty{6500}{K}$. Suhu warna adalah label warna, bukan bacaan
termometer --- LED $\qty{6500}{K}$ bekerja pada suhu ruangan --- dan
kosakatanya berjalan terbalik: suhu warna yang rendah terbaca jingga
``hangat'', yang tinggi kebiruan ``sejuk''.
\end{definition}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:1}
Sebutkan warna yang tertangkap indra ketika sebuah layar putih menerima
secara serentak: (a) cahaya merah dan hijau; (b) cahaya hijau dan biru;
(c) ketiga \omterm{def:g11:color-light-sources:additive}{warna primernya}. Cahaya mana yang \omterm{prop:g11:color-light-sources:complementary}{komplementer} terhadap
biru?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:2}
Subpiksel yang mana (R, G, B) yang dinyalakan sebuah layar untuk
menampilkan: kuning, magenta, putih, hitam, jingga?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:3}
\omterm{def:g10:light-spectra:white}{Cahaya putih} melewati \omterm{def:g11:color-light-sources:subtractive}{filter} magenta. Pita yang mana yang terserap, dan
warna apa yang muncul? Sebuah \omterm{def:g11:color-light-sources:subtractive}{filter} sian lalu ditambahkan di
belakangnya: apa yang muncul sekarang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:4}
Di bawah \omterm{def:g10:light-spectra:white}{cahaya putih} sebuah bendera menampilkan jalur kuning. Pita
yang mana yang dihamburkannya, dan yang mana yang diserapnya? Apa
warnanya di bawah (a) cahaya merah; (b) cahaya biru?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:color-light-sources:5}
\omterm{def:g10:light-spectra:continuous}{Spektrum kontinu} sebuah bintang memuncak pada
$\lambda_{\max} = \qty{580}{nm}$. Hitunglah suhu permukaannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:6}
Nyala sebuah lilin berada pada sekitar $\qty{1800}{K}$. (a) Hitunglah
$\lambda_{\max}$ dan sebutkan daerahnya. (b) Nyalanya toh tampak
jingga: jelaskan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:7}
Seorang pengendara sepeda memakai jaket merah dan sarung tangan sian.
Dengan memakai \cref{met:g11:color-light-sources:bookkeeping},
sebutkan warna tampak tiap barangnya di bawah (a) \omterm{def:g10:light-spectra:white}{cahaya putih};
(b) cahaya merah; (c) cahaya hijau.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:8}
Sebuah jalan tua diterangi lampu natrium bertekanan rendah (satu garis
tunggal pada $\qty{589}{nm}$). (a) Jenis \omterm{def:g10:light-spectra:white}{spektrum} apa itu, dan warna
apa saja yang dapat diambil benda \emph{mana pun} di bawahnya? (b)
Sebuah mobil tampak hitam di bawah lampu itu dan biru di bawah cahaya
siang: jelaskan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:9}
Sebuah penunjuk \omterm{def:g11:color-light-sources:lamps}{laser} hijau memancar pada $\qty{532}{nm}$. (a) Berkasnya
melewati sebuah prisma: bandingkan hasilnya dengan hasil bagi berkas
putih, lalu sebutkan sifat yang diperagakannya. (b) Berkas itu kemudian
menemui \omterm{def:g11:color-light-sources:subtractive}{filter} merah: apa yang keluar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:10}
Lukiskan \omterm{def:g10:light-spectra:white}{spektrum} sebuah LED putih (dioda biru ditambah \omterm{def:g11:color-light-sources:led}{fosfor}), lalu
jelaskan mengapa cahayanya ditangkap sebagai putih. Mengapa benda
berwarna merah tua tampak lebih kusam di bawah LED putih yang murah
daripada di bawah cahaya siang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:color-light-sources:11}
Hitunglah $\lambda_{\max}$ untuk benda pijar pada (a) \omterm{def:g11:color-light-sources:colortemp}{suhu warna} sebuah
lilin ($\qty{1800}{K}$); (b) \omterm{def:g11:color-light-sources:colortemp}{suhu warna} cahaya siang yang mendung
($\qty{6500}{K}$). Cahaya yang mana dari keduanya yang disebut
``hangat''? Berikan ulasanmu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:12}
Sebuah lampu natrium ($\qty{589}{nm}$) dan sebuah layar yang
menampilkan kuning (subpikselnya pada $\qty{610}{nm}$ dan
$\qty{540}{nm}$) tampak berwarna persis sama. (a) Kedua cahaya semacam
itu disebut apa, dan apa yang dibuktikan keberadaannya tentang warna?
(b) Usulkan dua percobaan yang berbeda untuk membedakan keduanya, lalu
ramalkan hasil masing-masing.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:13}
Sebuah lampu sorot halogen panggung bekerja pada $\qty{3200}{K}$.
(a) Hitunglah $\lambda_{\max}$: apakah cahaya ini lebih hangat atau
lebih sejuk daripada cahaya siang? (b) Untuk menirukan cahaya siang,
lampu sorot itu ditutup dengan gel biru pucat. Apa yang dibuang gel itu,
dan mengapa panggungnya menjadi lebih redup? (c) Jelaskan mengapa tidak
ada gel yang justru dapat \emph{menguatkan} ujung biru \omterm{def:g10:light-spectra:white}{spektrum} lampu
sorot itu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:14}
Seorang tukang daging menerangi etalase dagingnya dengan LED bersuhu
warna $\qty{2700}{K}$; penjual ikan di sebelahnya memakai
$\qty{6500}{K}$. (a) Lukiskan rona tiap penerangannya dan akibatnya
pada daging merah dan pada ikan yang putih. (b) Adakah salah satu
pedagang itu mengubah dagangannya? Sebutkan dengan tepat di mana
perubahannya terjadi (\cref{def:g11:color-light-sources:objectcolor}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:color-light-sources:15}
Sebuah kota di samping sebuah observatorium harus menerangi kembali
jalannya dan ragu antara lampu natrium bertekanan rendah dan LED putih
berspektrum lebar. (a) Pilihan yang mana yang lebih disukai para
astronom, dan mengapa? (Pikirkanlah apa yang dapat dibuang satu \omterm{def:g11:color-light-sources:subtractive}{filter}
dari citra mereka.) (b) Pilihan yang mana yang lebih disukai pengemudi
dan pejalan kaki? Berikan alasannya dengan \omterm{def:g11:color-light-sources:objectcolor}{warna benda}. (c) Usulkan
jalan tengahnya, yang dinyatakan dengan \omterm{def:g11:color-light-sources:colortemp}{suhu warna}.
\end{exercise}

\section{Soal: Merekayasa warna}

\begin{problem}[{Soal akhir pekan --- lampu panggung, jalan bernatrium
dan LED putih: bagaimana warna direkayasa, dari tiga lampu sorot yang
dapat melukiskan rona apa pun sampai keping yang menyalakan halaman
ini}]
\label{pb:g11:color-light-sources:1}
Seorang penata cahaya hanya memiliki lampu sorot merah, hijau dan biru;
seorang perekayasa jalan memiliki lampu yang memancarkan satu \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang} tunggal; seorang pembuat keping memiliki dioda yang hanya
bersinar biru. Ketiganya menjual warna. Soal ini menjalankan pekerjaan
mereka satu per satu --- pencampuran aditif di panggung, penghematan
monokrom di jalan, ilmu sihir \omterm{def:g11:color-light-sources:led}{fosfor} di dalam kepingnya --- lalu
berakhir dengan memilih cahaya untuk sebuah kamar tidur dan untuk
etalase seorang pengasah permata.

\medskip
\noindent\textbf{Bagian I --- Panggung: melukis dengan tiga lampu sorot.}
\begin{enumerate}
  \item Ketiga lampu sorot itu bertindihan pada latar putih. Sebutkan
        warna tiap tindihan berpasangan dan warna tindihan ketiganya.
  \item Bagaimana penata cahaya itu membuat jingga? Dan merah muda yang
        pucat? (Lukiskan \omterm{def:g10:energy-conservation:power}{daya} tiap lampu sorotnya secara kualitatif.)
  \item Seorang penari berdiri di depan latar itu, disinari lampu sorot
        merah dan hijau saja. Muncul dua bayang-bayang berwarna: apa
        warna masing-masing? Jelaskan.
  \item Latarnya dicat ulang menjadi biru tua. Bagaimana tampaknya di
        bawah lampu sorot merah saja? Dan di bawah ketiganya?
  \item Mengapa tiga lampu sorot dapat menghasilkan kembali ``hampir
        setiap'' warna? Bagian tubuh yang mana yang menetapkan angka
        tiga itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Kostum: pengurangan sedang bekerja.}
\begin{enumerate}[resume]
  \item Sebuah kostum berwarna kuning di bawah \omterm{def:g10:light-spectra:white}{cahaya putih}. Pita yang
        mana yang diserap zat warnanya?
  \item Ramalkan warnanya di bawah lampu sorot merah, lampu sorot
        hijau, dan lampu sorot biru.
  \item Penata cahaya itu ingin kostumnya berbalik dari terang menjadi
        hitam dengan sekali sentak sakelar. Ke lampu sorot yang mana
        sentakan itu berpindah?
  \item Sebuah gel magenta diselipkan di depan lampu sorot putih yang
        menyinari kostum kuning itu. Ikutilah pitanya
        (\cref{met:g11:color-light-sources:bookkeeping}): apa warna
        kostumnya?
  \item Mengapa gel panggung tersedia dalam warna sian, magenta dan
        kuning dan bukan merah, hijau dan biru? (Apa yang akan
        dilakukan gel merah terhadap \omterm{def:g10:energy-conservation:power}{daya} sebuah lampu sorot putih?)
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Jalan bernatrium: penghematan monokrom.}
\begin{enumerate}[resume]
  \item Lampu natrium bertekanan rendah pada dasarnya memancarkan satu
        garis pada $\qty{589}{nm}$. Jenis \omterm{def:g10:light-spectra:white}{spektrum} apa itu, dan
        bagaimana seluruh jalannya tampak di bawahnya?
  \item Di bawah lampu itu, sebutkan warna tampak sebuah dinding putih,
        marka jalan kuning, mobil merah, dan mobil biru.
  \item Mengapa para astronom menyukai lampu semacam itu? (Apa yang
        dapat dilakukan satu \omterm{def:g11:color-light-sources:subtractive}{filter} terhadap seluruh pendar kotanya?)
  \item Sebagai gantinya diusulkan bohlam jalan berkawat wolfram pada
        $\qty{2700}{K}$. Hitunglah $\lambda_{\max}$-nya: ke mana
        sebagian besar \omterm{def:g10:energy-conservation:power}{dayanya} pergi?
  \item Lampu natrium itu mengubah sekitar $30\%$ \omterm{def:g10:energy-conservation:power}{daya} listriknya
        menjadi cahaya tampak, sedangkan bohlam wolframnya sekitar
        $5\%$. Berapa \omterm{def:g10:energy-conservation:power}{watt} cahaya yang diserahkan masing-masing tiap
        $\qty{100}{W}$ \omterm{def:g10:energy-conservation:power}{daya}?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- LED putih dan pemilihan sebuah cahaya.}
\begin{enumerate}[resume]
  \item LED putih adalah dioda biru ($\qty{450}{nm}$) di bawah lapisan
        \omterm{def:g11:color-light-sources:led}{fosfor} kuning. Jelaskan, dengan \omterm{prop:g11:color-light-sources:complementary}{warna komplementer}, mengapa
        keluarannya terbaca sebagai putih.
  \item Lukiskan \omterm{def:g10:light-spectra:white}{spektrumnya} lalu bandingkan dengan \omterm{def:g10:light-spectra:continuous}{spektrum kontinu}
        cahaya siang. Pita yang mana yang kurang terwakili?
  \item Simpulkan benda yang mana yang ditampilkan LED semacam itu
        dengan buruk, lalu kaitkan dengan tukang daging pada
        \cref{exo:g11:color-light-sources:14}.
  \item Sebuah bohlam LED dijual sebagai ``$\qty{2700}{K}$''. Adakah
        sesuatu di dalamnya yang bersuhu $\qty{2700}{K}$? Apa yang
        dijanjikan labelnya? Hitunglah $\lambda_{\max}$ benda pijar
        yang ditirukannya.
  \item Penutup --- pilihlah, masing-masing dengan satu kalimat
        alasan, sebuah \omterm{def:g11:color-light-sources:colortemp}{suhu warna} untuk lampu kamar tidur dan sebuah
        lagi untuk etalase seorang pengasah permata; lalu nyatakan
        hukum warna bab ini dalam satu baris.
\end{enumerate}
\end{problem}
