\chapter{Het waterstofatoom}\label{ch:b3:hydrogen-atom}

Negen tiende van de atomen in het heelal is waterstof: één proton dat één
elektron vasthoudt, het enige atoom dat de natuurkunde exact kan oplossen
--- en de sleutel die elk ander heeft geopend. De oude kwantumtheorie van
\cref{ch:b3:hamiltonian-mechanics} raadde zijn energieën; het gereedschap
staat nu klaar om ze te \emph{leiden}: de coulombpotentiaal gaat de
\omterm{prop:b3:schrodinger-three-dimensions:swave}{radiale vergelijking} van het vorige hoofdstuk in, en eruit komen de
niveaus $-E_{\text{I}}/n^2$, de \omterm{thm:b3:hydrogen-atom:levels}{bohrstraal}, de schillen en subschillen van
het periodiek systeem uit de scheikunde, en de spectraalreeksen die
astronomen in elke nevel lezen. Dezelfde oplossing beschrijft, herschaald,
geïoniseerd helium, muonische atomen, positronium en de ``atomen'' van
elektron en gat binnen in halfgeleiders; opgerekt tot $n
\approx 100$
beschrijft zij rydbergatomen van een halve micrometer breed, wier
radiogefluister de geïoniseerde wolken van de Melkweg in kaart brengt en
wier wisselwerkingen inmiddels kwantumcomputers aandrijven.

% exactly; emphasis on scaling laws and applications.

\section{Het coulombvraagstuk opgelost}

\begin{theorem}[Niveaus van waterstof]\label{thm:b3:hydrogen-atom:levels}
\index{waterstofatoom}\index{hoofdkwantumgetal}\index{bohrstraal}
Voor $V(r) = -e^2/4\pi\varepsilon_0 r$ (schrijf $k = e^2/4\pi
\varepsilon_0$) worden de gebonden toestanden geëtiketteerd met $(n, l, m)$, met
\[
E_n = -\frac{E_{\text{I}}}{n^2} , \qquad
E_{\text{I}} = \frac{m_{\text{e}}k^2}{2\hbar^2} = \qty{13.6}{eV} ,
\qquad n = 1, 2, 3, \dots
\]
en, voor elke $n$, de baangetallen $l = 0, 1, \dots, n-1$ en
$m = -l, \dots, l$. De natuurlijke lengte is de \omterm{thm:b3:hydrogen-atom:levels}{bohrstraal} $a_0 =
\hbar^2/m_{\text{e}}k = \qty{52.9}{pm}$; de grondtoestand is
\[
\psi_{100} = \frac{1}{\sqrt{\pi a_0^3}}\;\eu^{-r/a_0} .
\]
De energie hangt \emph{uitsluitend} van $n$ af: tellen we de $m$'s en de
$l$'s, dan is niveau $n$ $n^2$-voudig ontaard (verdubbeld door de spin,
\cref{ch:b3:spin-two-level}) --- ver voorbij de $(2l+1)$-voudige
ontaarding die de isotropie verklaart.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Probeer voor de grondtoestand $u = Cr\,\eu^{-r/a}$ in de \omterm{prop:b3:schrodinger-three-dimensions:swave}{radiale
vergelijking} met $l = 0$: $u'' = (1/a^2 - 2/ar)u$, zodat
$-\tfrac{\hbar^2}{2m_{\text{e}}}u'' - \tfrac kr u = Eu$ vergt dat
$\hbar^2/m_{\text{e}}a = k$ (het gelijkstellen van de termen in $1/r$) ---
dat wil zeggen $a = a_0$ --- en $E = -\hbar^2/2m_{\text{e}}a_0^2 = -E_{\text{I}}$. De algemene oplossing (laguerreveeltermen, $n - l - 1$
radiale knopen) wordt aangenomen; elke stap van de constructie is
elementair maar lang. De ``toevallige'' ontaarding in $l$ weerspiegelt een
klassiek geheim van de zuivere kracht $1/r$ --- keplerellipsen precederen
niet, en een extra behouden vector (Laplace--Runge--Lenz) wijst langs de
vaste lange as; haar kwantumversie is wat verschillende $l$ aan één
energie bindt (\cref{exo:b3:hydrogen-atom:12}).
\end{proof}

\section{Waar het elektron is}

\begin{proposition}[Radiale verdelingen]\label{prop:b3:hydrogen-atom:radial}
\index{radiale kans}
De kans om het elektron tussen $r$ en $r + \dd r$ aan te treffen is
$P(r)\,\dd r$ met $P(r) = |u(r)|^2$, het kwadraat van de genormeerde
radiale functie van \cref{thm:b3:quantum-angular-momentum:radial}. Voor de laagste toestanden:
$P_{1s} \propto r^2\eu^{-2r/a_0}$, met een piek precies bij $r = a_0$ en
$\langle r\rangle = \tfrac32 a_0$; $P_{2s}$ toont twee bulten gescheiden
door een bolvormige \emph{knoop}; $P_{2p} \propto
r^4\eu^{-r/a_0}$, met
een piek bij $4a_0$. In het algemeen is $\langle r\rangle
\approx n^2a_0$:
atomen groeien \emph{kwadratisch} met de aanslag, terwijl hun binding als
$1/n^2$ krimpt --- de hefboom achter de rydbergreuzen van
\cref{pb:b3:hydrogen-atom:1}.
\end{proposition}
\admitted

\begin{example}[De afmetingen lezen]\label{ex:b3:hydrogen-atom:sizes}
Grondtoestand: \qty{0.1}{nm} breed, \qty{13.6}{eV} diep --- de schalen van
de hele scheikunde. Bij $n = 10$: straal $\sim\qty{5}{nm}$, binding
\qty{0.136}{eV} --- losjes vastgehouden. Bij $n = 100$: een atoom op de
schaal van een \emph{micrometer}, gebonden met \qty{1.4}{meV} en door het
zwakste veld gesloopt --- en toch is de interstellaire ruimte leeg genoeg
opdat zulke atomen leven en uitzenden
(\cref{pb:b3:hydrogen-atom:1}).
\end{example}

\section{Het spectrum}

\begin{proposition}[Reeksen en selectieregels]\label{prop:b3:hydrogen-atom:series}
\index{balmerreeks}\index{lymanreeks}
Een sprong $n' \to n$ zendt de golflengte
\[
\frac{1}{\lambda} = R_{\text{H}}\Big(\frac{1}{n^2} -
\frac{1}{n'^2}\Big) , \qquad
R_{\text{H}} = \frac{E_{\text{I}}}{hc} = \qty{1.097e7}{m^{-1}} ,
\]
uit, onder de dipoolselectieregel $\Delta l = \pm1$ (het foton draagt
$\hbar$). De \omterm{prop:b3:hydrogen-atom:series}{lymanreeks} ($\to n = 1$) ligt in het verre ultraviolet vanaf
\qty{121.6}{nm}; de \omterm{prop:b3:hydrogen-atom:series}{balmerreeks} ($\to n = 2$) begint bij de rode lijn
$H\alpha$, \qty{656.3}{nm} --- de kleur van emissienevels en
zonneprotuberansen; de paschenreeks en latere reeksen wijken het
infrarood in, en tussen $n = 110$ en $109$ landt dezelfde formule op
\qty{5.0}{GHz}: waterstof spreekt van het ultraviolet tot aan de
radioschaal.
\end{proposition}

\begin{proof}
Energiebehoud met \cref{thm:b3:hydrogen-atom:levels};
$R_{\text{H}}$ als in \cref{pb:b3:hamiltonian-mechanics:1}, nu afgeleid in
plaats van gepostuleerd.
\end{proof}

\begin{example}[Waterstofachtige atomen: één oplossing, vele atomen]\label{ex:b3:hydrogen-atom:scaling}
Vervang de lading van het proton door $Ze$ en het elektron door een
willekeurige omlopende massa $\mu$: elke formule herschaalt als
\[
a \to a_0\,\frac{m_{\text{e}}}{\mu}\,\frac1Z , \qquad
E_{\text{I}} \to E_{\text{I}}\,\frac{\mu}{m_{\text{e}}}\,Z^2 .
\]
He$^+$ ($Z = 2$): \qty{54.4}{eV}. De binnenelektronen van zware atomen
($Z \sim 30$): K-schilenergieën in de keV --- de karakteristieke
röntgenstraling waarmee Moseley de elementen ordende
(\cref{exo:b3:hydrogen-atom:5}). Muonische waterstof ($\mu = 207
m_{\text{e}}$): een atoom op de schaal van een femtometer dat het proton
zelf aftast. Positronium ($e^+e^-$, $\mu = m_{\text{e}}/2$): de helft van
de binding, tweemaal de afmeting, en een kort leven dat op
annihilatiefotonen uitloopt. En in een halfgeleider draaien een elektron
en een gat om elkaar met kleine effectieve massa's in een afschermend
diëlektricum: een \emph{exciton}, hetzelfde atoom uitgegroeid tot tien
nanometer en millielektronvolts (\cref{exo:b3:hydrogen-atom:7}) ---
waterstof is minder een atoom dan een sjabloon.
\end{example}

\begin{method}[Werken met waterstofachtige stelsels]\label{met:b3:hydrogen-atom:method}
(1) Schaal eerst: $a \propto 1/\mu Z$ en $E \propto \mu Z^2$ maken van elk
antwoord voor waterstof een antwoord voor elk waterstofachtig stelsel. (2)
Afmetingen: $\langle
r\rangle \sim n^2a$; afstanden nabij niveau $n$:
$\dd E/\dd n =
2E_{\text{I}}/n^3$. (3) Spectra: de formule van Rydberg
plus $\Delta l =
\pm1$. (4) Doordringing: $s$-toestanden voelen de kern,
toestanden met hoge $l$ draaien erbuiten --- de hefboom van de
meerelektronenscheikunde. (5) Ankers voor het gezond verstand:
$\qty{13.6}{eV}$, $\qty{52.9}{pm}$, $\qty{121.6}{nm}$ en
$\qty{656.3}{nm}$ --- vier getallen die het waard zijn levenslang te
onthouden.
\end{method}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:1}
(a) Ga door invullen na dat $u = Cr\eu^{-r/a_0}$ de \omterm{prop:b3:schrodinger-three-dimensions:swave}{radiale vergelijking}
met $l = 0$ oplost bij $E = -E_{\text{I}}$, mits $a_0 =
\hbar^2/m_{\text{e}}k$. (b) Normeer haar ($\int_0^\infty
x^2\eu^{-x}\dd x = 2$). (c) Controleer de dimensies van $a_0$ en
$E_{\text{I}}$. (d) Waarom is er geen toestand onder $-E_{\text{I}}$
(beredeneer met \cref{exo:b3:schrodinger-three-dimensions:12})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:2}
Bereken (a) de golflengten van Lyman $\alpha$ en van de lymanlimiet; (b)
de eerste drie balmerlijnen en de balmerlimiet --- welke zijn zichtbaar,
en in welke kleuren? (c) het bereik van de paschenreeks; (d) de frequentie
van de overgang $n = 110 \to 109$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:3}
Voor de grondtoestand: (a) bepaal het maximum van $P_{1s}(r)$; (b)
bereken $\langle r\rangle$; (c) bereken de kans om het elektron voorbij
$2a_0$ aan te treffen ($\int_x^\infty t^2\eu^{-t}\dd t = (x^2 +
2x + 2)\eu^{-x}$); (d) waarom misleiden ``baan''plaatjes met straal precies
$a_0$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:4}
(a) Noem de paren $(l, m)$ van $n = 3$ en ga de telling $n^2 =
9$ na. (b)
Hoeveel toestanden bevatten de schillen $n = 1, 2, 3$ met spin? (c) Breng
die in verband met de lengten $2, 8, 18$ van de rijen van het periodiek
systeem. (d) Welke ontaarding (in $m$ of in $l$) overleeft bij het
buitenste elektron van natrium, en waarom breekt de andere?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:5}
De ladder van Moseley. Een binnenelektron (K-schil) van een element $Z$
beweegt in een vrijwel kale kernveld dat door het ene andere K-elektron
wordt afgeschermd: effectieve lading $\approx Z - 1$. (a) Toon aan dat de
röntgenenergie van $2p \to 1s$ bij benadering
$\tfrac34(Z-1)^2E_{\text{I}}$ is. (b) Reken dit uit voor koper ($Z = 29$)
en vergelijk met de gemeten K$\alpha$ bij \qty{8.05}{keV}. (c) Moseley
(1913) zette $\sqrt{\nu}$ tegen $Z$ uit en kreeg rechte lijnen: wat
bewees dit over de betekenis van het atoomnummer? (d) Voorspel de
K$\alpha$-energie van het toen nog ontbrekende element $Z = 43$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:6}
Positronium. (a) Rechtvaardig $\mu = m_{\text{e}}/2$ en geef de
bindingsenergie en de \omterm{thm:b3:hydrogen-atom:levels}{bohrstraal} ervan. (b) Zijn golflengte voor Lyman
$\alpha$. (c) Parapositronium annihileert in twee fotonen: hun energieën
en onderlinge richtingen (uit \cref{ch:b3:relativistic-dynamics}). (d)
Waar in de geneeskunde wordt precies deze handtekening dagelijks
opgemerkt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:7}
Excitonen. In galliumarsenide is $\varepsilon_{\text{r}} = 12.9$ en de
gereduceerde effectieve massa $\mu = 0.058\,m_{\text{e}}$. (a) Toon aan
dat $a_{\text{ex}} = a_0\,\varepsilon_{\text{r}}\,m_{\text{e}}/\mu$ en
reken dit uit. (b) De bindingsenergie van het exciton. (c) Vergelijk
$a_{\text{ex}}$ met de stralen van de kwantumstippen van
\cref{pb:b3:schrodinger-three-dimensions:1} en rechtvaardig eindelijk het
verwaarlozen van de coulombterm in kleine stippen. (d) Waarom verschijnen
excitonlijnen alleen in \emph{zuivere, koude} halfgeleiders
($k_{\text{B}}T$ tegen uw antwoord op (b))?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:8}
Doordringing en de alkalimetalen. Natrium is een waterstofachtig
valentie-elektron buiten een gesloten romp met effectieve lading $+e$.
(a) Welke van $3s$, $3p$ en $3d$ voelt de onvolledig afgeschermde kern het
sterkst, en waarom (denk aan de radiale figuur)? (b) De gemeten niveaus
zijn $E_{3s} = \qty{-5.14}{eV}$, $E_{3p} = \qty{-3.04}{eV},
E_{3d} = \qty{-1.52}{eV}$: ga na dat $3d$ bijna waterstofachtig is ($n = 3$) terwijl $3s$ veel dieper ligt, en verklaar dat. (c) Bereken de
golflengte van $3p \to 3s$: welke beroemde kleur is dat? (d) Schrijf de
alkaliniveaus als $-E_{\text{I}}/(n - \delta_l)^2$ en haal de
``kwantumdefecten'' $\delta_s$ en $\delta_p$ voor natrium eruit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:9}
Een proton in een nevel vangt een elektron in $n = 100$. (a) Hoeveel
energie komt er bij de vangst in het foton vrij als het elektron vrijwel
vrij aankwam? (b) Het atoom cascadeert omlaag, bij voorkeur met
stappen $\Delta n = 1$ bij hoge $n$: in welke band vallen die fotonen? (c) De
laatste stap van de cascade is Lyman $\alpha$; het zichtbare licht van
nevels wordt door H$\alpha$ overheerst: spoor op welke stap van de cascade
dat is. (d) Leg uit waarom emissienevels \emph{rood} gloeien hoewel de
sterkste lijn van waterstof (Lyman $\alpha$) ultraviolet is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:10}
Gemiddelden en het viriaal. Bereken voor de grondtoestand (a)
$\langle 1/r\rangle$ en ga $\langle E_p\rangle = -2E_{\text{I}}

na; (b) $\langle E_k\rangle = +E_{\text{I}}$, waarmee de viriaalverhouding

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:11}
Hoe fijn is fijnstructuur. (a) Toon aan dat de snelheidsschaal van de
grondtoestand $\langle v^2\rangle^{1/2}/c = \alpha \approx 1/137$ is, en
voor waterstofachtige $Z$: $Z\alpha/n$. (b) Relativistische correcties
komen binnen op relatieve orde $(v/c)^2$: schat de schaal van de
fijnstructuur $\alpha^2E_{\text{I}}$ in meV, en de orde van de splitsing
voor $n = 2$ in GHz. (c) Voor waterstofachtig uranium ($Z = 92$): $v/c$
bij $n = 1$ --- is de niet-relativistische behandeling houdbaar? (d) Wat
kondigt de formele divergentie bij $Z\alpha \to 1$ ($Z \approx 137$)
natuurkundig aan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:12}
De onredelijke ontaarding. (a) Toon in de klassieke mechanica aan dat de
baan voor $V = -k/r$ sluit (geen precessie) door de behouden vector van
Laplace--Runge--Lenz $\vect A = \vect p\wedge\vect
L - mk\,\vect e_r$ aan
te halen --- ga $\dd\vect A/\dd t = 0$ na met de wet van Newton. (b)
Beredeneer: een behouden vector die de as van de ellips vastlegt is een
extra symmetrie boven op de rotaties --- en extra symmetrie betekent extra
ontaarding (denk aan \cref{exo:b3:quantum-formalism:9}). (c) Voeg een
kleine correctie $1/r^2$ aan de potentiaal toe en toon aan dat de ellips
precedeert: de as draait, en $\vect A$ blijft niet meer behouden. (d) Leg
het verband: in atomen met meer elektronen is de effectieve potentiaal
niet zuiver $1/r$, en breekt de ontaarding in $l$ (natrium!); in waterstof
zelf levert de relativiteitstheorie de kleine correctie --- welk
waargenomen kenmerk uit \cref{exo:b3:hydrogen-atom:11} is dat?
\end{exercise}

\section{Vraagstuk: Reuzenatomen}

\begin{problem}[{Weekendvraagstuk --- rydbergatomen, van de radiogloed
van de Melkweg tot kwantumcomputers}]\label{pb:b3:hydrogen-atom:1}
Rek waterstof op tot $n \approx 100$ en het wordt een ander soort
voorwerp: micrometers breed, gebonden met millielektronvolts, absurd
gevoelig --- en absurd nuttig. Dit vraagstuk schaalt de oplossing voor
waterstof omhoog, eerst naar de interstellaire wolken die op
centimetergolflengten uitzenden, dan naar de laboratoriumroosters waar
zulke reuzen elkaar tot processoren verstrengelen. Gegevens:
$E_{\text{I}} = \qty{13.6}{eV}$, $a_0 = \qty{52.9}{pm}$,
$R_{\text{H}}c = \qty{3.29e15}{Hz}$, en $k_{\text{B}}T$ bij
\qty{300}{K} is \qty{25.9}{meV}.

\medskip\noindent\textbf{Deel I --- De schalingswetten.}
\begin{enumerate}
  \item Geef de vier basisschalingen met $n$: de afmeting $\langle
        r\rangle$, de binding $|E_n|$, de afstand $E_{n+1} - E_n$ (voor
        grote $n$) en de klassieke omloopfrequentie van de
        overeenkomstige bohrbaan.
  \item Toon aan dat voor grote $n$ de overgangsfrequentie $n+1 \to
        n$ naar $2R_{\text{H}}c/n^3$ gaat, en ga na dat zij gelijk is aan de
        klassieke omloopfrequentie (het correspondentiebeginsel van
        \cref{pb:b3:hamiltonian-mechanics:1}, voltooid).
  \item Reken de afmeting en de binding uit bij $n = 50$ en $n = 100$.
  \item Het elektrische dipoolmoment van een rydbergatoom schaalt als
        $n^2ea_0$: reken het uit bij $n = 50$ in debye ($1\,\text{D} =
        \qty{3.34e-30}{C.m}$) en vergelijk met de \qty{1.85}{D} van een
        watermolecuul.
  \item Schat het elektrische veld dat het atoom bij $n = 50$ uiteenrukt,
        als $F \sim |E_n|/(e\,\langle r\rangle)$, in V/cm.
  \item Waarom kunnen zulke atomen de gewone chemie van een
        laboratoriumvacuüm niet overleven --- en overleven zij in de
        interstellaire ruimte wel prima (vergelijk de botsingstempo's)?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De recombinatielijnen van de
Melkweg.}
In geïoniseerde nevels vangen protonen elektronen in hoge $n$; de cascade
straalt een kam van ``radiorecombinatielijnen'' uit.
\begin{enumerate}[resume]
  \item Bereken de frequentie van de overgang $n = 110 \to 109$
        (H109$\alpha$ genoemd).
  \item In welke band valt zij, en wat voor telescoop ontvangt haar?
  \item Bereken de golflengte van H109$\alpha$ en vergelijk haar met de
        afmeting van het uitzendende atoom: hoeveel atomen passen er in
        één golflengte?
  \item Deze lijnen meten de temperatuur van de nevel via hun
        dopplerbreedte: bij $T = \qty{e4}{K}$ is de thermische snelheid
        van waterstof $\sim\qty{12.8}{km/s}$ --- bereken de relatieve
        lijnbreedte $\Delta\nu/\nu$ en de breedte van H109$\alpha$ in
        kHz.
  \item Radiogolven doorkruisen het stof dat de optische hemel zwart
        maakt: wat kunnen astronomen met recombinatielijnen daardoor in
        kaart brengen wat met balmerlijnen niet lukt?
  \item Boven ruwweg $n \sim 1000$ (atomen van $0.05$ millimeter breed!)
        vervagen de niveaus zelfs in de ruimte tot een continuüm: noem
        twee effecten die zulke toestanden verbreden of vernietigen.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Reuzen in het laboratorium.}
\begin{enumerate}[resume]
  \item Lasers drijven rubidiumatomen naar $n \approx 70$ in roosters van
        optische pincetten met een onderlinge afstand van een micrometer.
        Twee zulke atomen op afstand $R$ wisselwerken via hun
        geïnduceerde dipolen (denk aan
        \cref{exo:b3:harmonic-oscillator:10}): toon met een dipoolmoment
        $\propto n^2$ aan dat de vanderwaalssterkte als een kolossale
        $n^{11}$ schaalt (gebruik $C_6 \propto d^4/\Delta E$ met
        niveauafstand $\Delta E \propto n^{-3}$).
  \item De ``blokkade'': binnen een straal $R_{\text{b}}$ verschuift de
        wisselwerking de dubbel aangeslagen toestand uit de laserresonantie,
        zodat er geen \emph{twee} atomen tegelijk aangeslagen kunnen zijn.
        Leg in één zin uit waarom dit een poort met twee qubits
        verwezenlijkt.
  \item Blokkadestralen halen enkele micrometers --- duizenden keren de
        afmeting van de atomen in hun grondtoestand: waarom is dat lange
        bereik (vergelijk het nanometerbereik van de scheikunde) precies
        wat een schaalbare processor wil?
  \item Een rydbergtoestand bij $n = 70$ leeft $\sim\qty{100}{\micro s}$
        terwijl een poort $\sim\qty{0.5}{\micro s}$ kost: hoeveel
        bewerkingen passen er ruwweg in één levensduur, en waarom doet
        die verhouding ertoe en niet de levensduur alleen?
  \item Zwartestraling bij kamertemperatuur, die nabij \qty{100}{meV}
        piekt maar rijk is aan een staart bij lage energie, drijft
        overgangen tussen naburige rydbergniveaus aan die slechts
        $\sim\qty{0.1}{meV}$ uit elkaar liggen: waarom moeten
        precisieproeven de atomen in koude schilden opsluiten?
  \item Schat hoeveel antenneachtige dipoolovergangen ($\propto n^2$) een
        fotonenbad van \qty{300}{K} per seconde aandrijft, vergeleken met
        een atoom bij $n = 1$: welke schaling maakt rydbergatomen tot
        voortreffelijke \emph{sensoren} voor microgolf- en
        terahertzvelden?
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Het grote plaatje.}
\begin{enumerate}[resume]
  \item Eén formule, $E_n = -E_{\text{I}}\mu Z^2/m_{\text{e}}n^2$,
        overspant hoeveel ordes van grootte aan bindingsenergie van
        waterstofachtig uranium ($Z = 92$, $n = 1$) tot een
        rydbergtoestand bij $n = 1000$? Bereken beide uiteinden.
  \item De stralingslevensduur van toestanden met lage $l$ schaalt als
        $n^3$: schat uit de levensduur van $2p$ van \qty{1.6}{ns} de
        levensduur van een toestand $100p$ --- en vergelijk met het
        probleem van de zwartestraling bij kamertemperatuur uit deel III.
  \item Bij CERN worden antiatomen van antiwaterstof inmiddels op het
        interval $1s \to 2s$ tot op vijftien cijfers spectroscopisch
        gemeten: welke fundamentele symmetrie toetst overeenstemming met
        gewone waterstof, en waarom is waterstof het juiste atoom voor de
        vergelijking?
  \item Bij $n \sim 100$ wikkelt de debrogliegolf van het elektron zich
        om een bijna klassieke baan; bij $n = 1$ bestaat er in het geheel
        geen baan: zeg in één zin waar het klassieke beeld aangaat.
  \item Rydbergconstanten worden tot op vijftien cijfers gemeten: waarom
        is waterstof, van alle stelsels, het natuurlijke precisieanker van
        de atoomfysica?
  \item De Nobelprijs van 2012 (Haroche) gebruikte rydbergatomen als
        fotontellers die het foton niet vernietigen: welke twee
        eigenschappen uit de delen I en III maken ze tot ideale sondes
        zonder afbraak voor microgolfvelden?
  \item Vat het genoemde resultaat samen: de schalingen $n^2$, $n^{-2}$,
        $n^{-3}$ en $n^{11}$ van één exact opgelost atoom rekken waterstof
        van \qty{52.9}{pm} tot een halve micrometer, stemmen zijn stem van
        \qty{121.6}{nm} tot \qty{6}{cm}, en maken het tegelijk tot het
        radiobaken van de Melkweg en tot de poort met twee qubits van
        kwantumcomputers met neutrale atomen.
\end{enumerate}
\end{problem}
