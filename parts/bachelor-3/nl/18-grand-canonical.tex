\chapter{Het grootkanonieke ensemble}\label{ch:b3:grand-canonical}

Een waterdruppel in vochtige lucht groeit noch krimpt; zuurstof bindt in
de longen aan hemoglobine en laat in de spieren weer los; een batterij
drijft elektronen door een telefoon omdat zij in de ene elektrode
``duurder'' zijn dan in de andere. In elk geval is de munt die in
evenwicht wordt gebracht niet de temperatuur maar de \emph{\omterm{thm:b3:grand-canonical:distribution}{chemische
potentiaal}} $\mu$: de prijs in \omterm{prop:b3:canonical-ensemble:machine}{vrije energie} van één deeltje erbij. Dit
hoofdstuk opent de tweede kraan van het reservoir --- deeltjes zowel als
energie --- en bouwt het grootkanonieke ensemble, waarvan de beloning een
verbluffende doelmatigheid is: de bezetting van één enkel kwantumniveau,
fermion of boson, volgt in twee regels, en daarmee de moederformules van
het volgende hoofdstuk. De toepassingen lopen van de
adsorptie-isothermen van gasmaskers en katalysatoren tot de \omterm{prop:b3:grand-canonical:saha}{vergelijking
van Saha}, waarmee een promovenda in Harvard in 1925 de temperaturen van
de sterren aflas en, tegen elke verwachting in, ontdekte dat het heelal
uit waterstof bestaat.

\section{Deeltjes doen mee aan de onderhandeling}

\begin{theorem}[Grootkanonieke verdeling]\label{thm:b3:grand-canonical:distribution}
\index{grootkanoniek ensemble}\index{chemische potentiaal}
Een stelsel dat energie \emph{en deeltjes} uitwisselt met een reservoir op
temperatuur $T$ en \omterm{thm:b3:grand-canonical:distribution}{chemische potentiaal} $\mu$ bezet zijn toestand $s$
(energie $E_s$, deeltjesaantal $N_s$) met kans
\[
\mathcal P_s = \frac{\eu^{-\beta(E_s - \mu N_s)}}{\Xi} , \qquad
\Xi = \sum_s\eu^{-\beta(E_s - \mu N_s)} ,
\]
de \emph{grote \omterm{thm:b3:canonical-ensemble:boltzmann}{toestandssom}}. Uit $\Xi$: $\langle N\rangle
= k_{\text{B}}T\,\partial\ln\Xi/\partial\mu$, en de \emph{grote potentiaal}
$\Phi = -k_{\text{B}}T\ln\Xi$ voldoet aan $\Phi = -PV$.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Ontwikkel, net als in \cref{thm:b3:canonical-ensemble:boltzmann}, de
entropie van het reservoir in beide onttrekkingen:
$S_{\text{res}}(E - E_s, N - N_s) \approx S_{\text{res}} - E_s/T +
\mu N_s/T$, met $\partial S/\partial N = -\mu/T$
(\cref{thm:b3:microcanonical-ensemble:temperature}). De gemiddelden volgen
door differentiëren; $\Phi = -PV$ wordt aangenomen (het volgt uit de
extensiviteit).
\end{proof}

\begin{definition}[Wat $\mu$ betekent]\label{def:b3:grand-canonical:mu}
\index{chemische potentiaal!betekenis}
$\mu$ is de prijs, in \omterm{prop:b3:canonical-ensemble:machine}{vrije energie}, van één deeltje erbij bij vaste $T$
en $V$. Deeltjes stromen spontaan van hoge $\mu$ naar lage $\mu$ ---
diffusie, osmose, verdamping, chemische reacties en elektrische stromen
zijn alle $\mu$-gradiënten die ontspannen. In evenwicht tussen twee fasen
of plaatsen is $\mu$ aan beide kanten gelijk --- de derde gelijkheid
(naast $T$ en $P$) die de coëxistentie beheerst. Voor een klassiek ideaal
gas geldt
\[
\mu = k_{\text{B}}T\,\ln\!\big(n\lambda_T^3\big) < 0
\quad\text{(ijl gas)} :
\]
een deeltje aan een ijl gas toevoegen \emph{verlaagt} de \omterm{prop:b3:canonical-ensemble:machine}{vrije energie},
omdat de entropiewinst elke energieprijs overtreft --- het teken dat de
verdamping in droge lucht aandrijft. En een batterij is een
$\mu$-machine: haar spanning is de val van de \omterm{thm:b3:grand-canonical:distribution}{chemische potentiaal} per
elektron tussen haar elektroden, $U = \Delta\mu/e$ --- volts zijn de
wisselkoers van de elektronenchemie, en daarom leveren cellen spanningen
van één cijfer.
\end{definition}

\section{Eén niveau tegelijk: de moedertruc}

\begin{theorem}[Bezetting van één kwantumniveau]\label{thm:b3:grand-canonical:occupation}
\index{verdeling van Fermi--Dirac}\index{verdeling van Bose--Einstein}
Neem als ``het stelsel'' één eendeeltjesniveau met energie $\epsilon$, in
groot contact met het gas van alle andere. \emph{Fermionen} (bezetting $0$
of $1$):
\[
\Xi = 1 + \eu^{-\beta(\epsilon - \mu)}
\;\Longrightarrow\;
\bar n_{\text{FD}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} + 1} .
\]
\emph{Bosonen} (bezetting $0, 1, 2, \dots$):
\[
\Xi = \frac{1}{1 - \eu^{-\beta(\epsilon - \mu)}}
\;\Longrightarrow\;
\bar n_{\text{BE}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} - 1} ,
\]
wat $\mu < \epsilon_{\min}$ vergt. Beide herleiden zich tot de
boltzmannfactor $\eu^{-\beta(\epsilon - \mu)}$ wanneer de bezettingen
zeldzaam zijn. Deze twee formules, elk in twee regels verkregen, zijn de
volledige invoer van de kwantumstatistiek: het volgende hoofdstuk is hun
oogst.
\end{theorem}

\begin{proof}
Fermionen: twee termen. Bosonen: een meetkundige reeks, alleen convergent
voor $\mu < \epsilon$. In beide gevallen geeft $\bar n = k_{\text{B}}T\,
\partial\ln\Xi/\partial\mu$ de aangegeven vormen.
\end{proof}

\section{Adsorptie: plaatsen te huur}

\begin{proposition}[De isotherm van Langmuir]\label{prop:b3:grand-canonical:langmuir}
\index{isotherm van Langmuir}\index{adsorptie}
Een oppervlak biedt onafhankelijke plaatsen aan, elk leeg of met één
gasmolecuul met bindingsenergie $-\epsilon_0$; het gas erboven, bij druk
$P$, is het reservoir. Elke plaats is een groot stelsel met twee
toestanden, en haar bezetting is
\[
\theta = \frac{1}{\eu^{\beta(-\epsilon_0 - \mu)} + 1}
 = \frac{P}{P + P_0(T)} , \qquad
P_0(T) = \frac{k_{\text{B}}T}{\lambda_T^3}\,
\eu^{-\epsilon_0/k_{\text{B}}T} ,
\]
met de $\mu$ van het ideale gas. De bedekking stijgt bij lage druk lineair
en verzadigt bij één monolaag --- de \emph{\omterm{prop:b3:grand-canonical:langmuir}{isotherm van Langmuir}}, de
werkkromme van katalysatoren, actieve kool, gasmaskers en de zuurstofbinding
van bloed (\cref{exo:b3:grand-canonical:6}). De overgangsdruk $P_0$ daalt
\emph{exponentieel} met de bindingsenergie en stijgt steil met de
temperatuur: een oppervlak verwarmen schudt zijn gasten los --- daarom is
glaswerk onder vacuüm uitbakken de enige echte reiniging.
\end{proposition}

\begin{proof}
De bezetting van de plaats is
\cref{thm:b3:grand-canonical:occupation} (in de fermionvorm: een plaats
bevat hoogstens één) met $\epsilon = -\epsilon_0$; vul $\eu^{\beta\mu} = n\lambda_T^3 = P\lambda_T^3/k_{\text{B}}T$ in.
\end{proof}

\section{Ionisatie-evenwichten: de vergelijking van Saha}

\begin{proposition}[De vergelijking van Saha]\label{prop:b3:grand-canonical:saha}
\index{vergelijking van Saha}
In een heet gas komt de reactie $\text{H} \rightleftharpoons \text{p} +
\text{e}$ in evenwicht wanneer $\mu_{\text{H}} = \mu_{\text{p}} +
\mu_{\text{e}}$. Elke $\mu$ met de vorm van het ideale gas schrijven (met
het atoom zijn binding $-E_{\text{I}}$ toegerekend) geeft
\[
\frac{n_{\text{e}}\,n_{\text{p}}}{n_{\text{H}}}
 = \frac{1}{\lambda_{T,\text{e}}^3}\,
\eu^{-E_{\text{I}}/k_{\text{B}}T}
\qquad\Big(\lambda_{T,\text{e}} =
h/\sqrt{2\pi m_{\text{e}}k_{\text{B}}T}\Big)
\]
(op ontaardingsfactoren van de orde één na). Opnieuw twee concurrerende
munten: de boltzmannfactor belast de ionisatie met $E_{\text{I}} = \qty{13.6}{eV}$, terwijl de voorfactor --- de \emph{entropie van het
bevrijde elektron}, met haar $\lambda_T^{-3}$ aan toegankelijke toestanden
--- haar subsidieert. Bij sterdichtheden is die subsidie zo groot dat
waterstof al nabij \qty{10000}{K} merkbaar ioniseert, waar $k_{\text{B}}T$
slechts \qty{0.9}{eV} is: evenwichten worden om \omterm{prop:b3:canonical-ensemble:machine}{vrije energie} uitgevochten
en niet om energie --- en de spectra van de sterren zijn het grootboek
(\cref{pb:b3:grand-canonical:1}).
\end{proposition}

\begin{proof}[Gedeeltelijk bewijs]
Stel $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$ met $\mu_i
= k_{\text{B}}T\ln(n_i\lambda_{T,i}^3) + \varepsilon_i$
($\varepsilon_{\text{H}} = -E_{\text{I}}$, de andere nul) en neem de
exponentiële functie. De thermische golflengten van het proton en het
atoom heffen elkaar vrijwel op (gelijke massa's); de ontaardingsfactoren
worden aangenomen.
\end{proof}

\begin{method}[Grootkanoniek ambacht]\label{met:b3:grand-canonical:method}
(1) Alles wat deeltjes met een grote partner uitwisselt: geef die partner
een $\mu$ en weeg de toestanden met $\eu^{-\beta(E - \mu N)}$. (2)
Onafhankelijke plaatsen of niveaus factoriseren: los er \emph{één} op en
vermenigvuldig. (3) Voor een reservoir van klassiek gas zet
$\eu^{\beta\mu} =
n\lambda_T^3$ de $\mu$ om in een meetbare druk. (4)
Reacties en faseovergangen: breng de $\mu$'s in evenwicht, één per soort,
bindingsenergieën inbegrepen --- verwacht dat entropievoorfactoren de
evenwichten ver van naïeve boltzmanngissingen verschuiven. (5) Elektronen
in vaste stoffen: hun $\mu$ is het ferminiveau, en verschillen in $\mu$
zijn spanningen --- de brug naar
\cref{ch:b3:electrons-in-solids}.
\end{method}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:1}
(a) Twee dozen met hetzelfde gas bij dezelfde $T$ maar verschillende
dichtheden worden verbonden: welke kant stromen de deeltjes op, in de taal
van $\mu$? (b) Toon met de $\mu$ van het ideale gas aan dat de stroom bij
gelijke dichtheden stopt. (c) Water staat onder vochtige lucht: formuleer
de evenwichtsvoorwaarde in $\mu$'s. (d) Waarom droogt de was ook onder
$100\,^\circ$C wanneer de lucht droog is (vergelijk $\mu$'s, geen
temperaturen)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:2}
De \omterm{thm:b3:grand-canonical:distribution}{chemische potentiaal} van lucht. (a) Bereken $\lambda_T$ voor N$_2$ bij
\qty{300}{K}. (b) Bereken $n\lambda_T^3$ bij \qty{1}{bar} en de
resulterende $\mu$ in eV. (c) Waarom is $\mu < 0$ geen paradox (welke term
van $F = E - TS$ overheerst bij het toevoegen van een molecuul)? (d) Bij
welke dichtheid zou $\mu$ nul bereiken, en wat kondigt $n\lambda_T^3 \sim 1$ aan (\cref{exo:b3:grand-canonical:11})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:3}
Leid de bezetting van Fermi en Dirac stap voor stap uit haar $\Xi$ met
twee termen af; toets de limieten $\epsilon \ll \mu$ en $\epsilon \gg
\mu$, en de waarde bij $\epsilon = \mu$. Waar is deze functie in dit boek
al eerder opgedoken
(\cref{exo:b3:microcanonical-ensemble:8})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:4}
Volts zijn scheikunde. (a) Een lithiumcel levert \qty{3.0}{V}: druk het
verschil in \omterm{thm:b3:grand-canonical:distribution}{chemische potentiaal} per elektron in eV uit en per mol in kJ.
(b) Waarom hebben alle alledaagse batterijen spanningen van één cijfer
(wat legt de schaal van chemische verschillen in $\mu$ vast)? (c) Een
loodaccu in een auto stapelt zes cellen: waarom stapelen in plaats van een
reactie van \qty{12}{eV} zoeken? (d) Wat doet een batterij in de taal van
$\mu$ wanneer zij ``leeg'' is?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:5}
Bosonen op een niveau. (a) Sommeer de meetkundige reeks voor $\Xi$ en leid
$\bar n_{\text{BE}}$ af. (b) Waarom moet $\mu$ onder het laagste niveau
blijven --- wat divergeert er anders, en wat kondigt die divergentie aan
(\cref{ch:b3:quantum-statistics})? (c) Bereken $\bar n$ voor $(\epsilon - \mu)/k_{\text{B}}T = 0.1$, $1$ en $3$ en vergelijk met Boltzmann. (d)
Fotonen hebben $\mu = 0$: geef de natuurkundige reden (hun aantal blijft
niet behouden --- wanden absorberen en zenden vrijelijk uit).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:6}
Langmuir aan het werk. (a) Welke druk haalt volgens de isotherm $\theta = 0.5$? En $0.9$? (b) Myoglobine bindt O$_2$ op één plaats: de verzadiging
volgt Langmuir in de partiële druk van zuurstof --- waarom ziet men bij
myoglobine een \emph{hyperbolische} (en geen sigmoïdale) kromme, maar bij
hemoglobine met vier plaatsen een coöperatieve S-kromme? (c)
Koolstofmonoxide bindt aan dezelfde plaatsen met een $\epsilon_0$ die
$\approx\qty{0.13}{eV}$ groter is: bereken bij gelijke partiële drukken en
\qty{310}{K} de bezettingsverhouding CO:O$_2$ en verklaar
CO-vergiftiging. (d) Waarom werkt hyperbare zuurstoftherapie, in de taal
van de isotherm?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:7}
Fluctuaties in het aantal. (a) Toon aan dat $\operatorname{Var}N =
k_{\text{B}}T\,\partial\langle N\rangle/\partial\mu$. (b) Leid voor het
klassieke ideale gas af dat $\operatorname{Var}N = \langle
N\rangle$:
poisson. (c) Breng dit in overeenstemming met de dichtheidsfluctuaties van
\cref{exo:b3:microcanonical-ensemble:12}. (d) Bereken voor één fermionisch
niveau $\operatorname{Var}n = \bar n(1 - \bar
n)$: waar zijn
fermionaantallen het stilst, en waarom?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:8}
Osmose uit $\mu$. Een membraan laat water door maar geen opgeloste stof
(concentratie $c$ per volume, ijl). (a) Toon door de $\mu$ van het water
aan weerszijden gelijk te stellen aan dat de kant met zuiver water met
$\Pi = ck_{\text{B}}T$ moet worden overdrukt (Van 't Hoff) --- neem aan dat
opgeloste stof de $\mu$ van water per watermolecuul met
$-k_{\text{B}}T\,c/n_{\text{w}}$ verlaagt. (b) Bereken $\Pi$ voor
fysiologisch zout ($c \approx \qty{3e26}{m^{-3}}$ deeltjes in totaal). (c)
Rode bloedcellen in zuiver water: welke kant stroomt het water op, en met
welk gevolg? (d) Waarom hebben bomen pompen noch wonderen nodig om sap
tientallen meters op te tillen (osmotische en capillaire $\mu$-gradiënten)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:9}
Evenwicht in een veld. In de zwaartekracht vergt het evenwicht van een
gaskolom dat de \emph{totale} \omterm{thm:b3:grand-canonical:distribution}{chemische potentiaal} $\mu(h) =
k_{\text{B}}T\ln(n\lambda_T^3) + mgh$ uniform is. (a) Leid hier de
barometrische wet in één regel uit af. (b) Veralgemeen naar een centrifuge
met hoeksnelheid $\omega$ (potentiaal $-\tfrac12 m\omega^2r^2$) en verkrijg
het radiale profiel. (c) Centrifuges voor uraniumverrijking draaien met
$\omega r \sim
\qty{600}{m/s}$: bereken de evenwichtsverhouding van de
verrijking van $^{238}$UF$_6$ tot $^{235}$UF$_6$ per trap, $\exp[\Delta m\,\omega^2
r^2/2k_{\text{B}}T]$ met $\Delta m = \qty{5e-27}{kg}$ en $T =
\qty{320}{K}$. (d) Waarom worden er vele trappen in cascade gezet?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:10}
Saha met de hand. (a) Voer het bewijs met de $\mu$-balans van

Zonnefotosfeer: $T = \qty{5800}{K}$, $n \approx

aan dat de geïoniseerde fractie van waterstof slechts $\sim\num{e-4}$ is

\qty{10000}{K}$
(dezelfde $n$): herbereken en geef commentaar op de steilheid.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:11}
Wanneer gassen kwantummechanisch worden. De ontaarding zet in bij
$n\lambda_T^3 \sim
1$. (a) Voor lucht bij \qty{300}{K}: hoe ver onder de

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:12}
Het heelal verliest zijn positronen. In het hete jonge heelal hield
$\eu^+ + \eu^- \rightleftharpoons 2\gamma$ de paren in evenwicht met
$\mu_{e^+} + \mu_{e^-} = 0$ (fotonen: $\mu = 0$). (a) Voor een
symmetrisch plasma geeft dit $\mu_{e^\pm} \approx 0$: toon aan dat de
paardichtheid dan $n_\pm \approx
\lambda_T^{-3}\eu^{-m_{\text{e}}c^2/k_{\text{B}}T}$ is zodra
$k_{\text{B}}T \ll m_{\text{e}}c^2$. (b) Bereken de temperatuur waarbij
$m_{\text{e}}c^2 = k_{\text{B}}T$ en de kosmische tijdschaal die daarbij
hoort (neem $t \sim \qty{1}{s}$ bij $\qty{e10}{K}$ aan). (c) Toon aan dat
een bescheiden verdere afkoeling (een factor van enkele eenheden) de
paardichtheid met vele ordes van grootte laat instorten: de positronen
annihileren weg en laten alleen het minieme oerbeschot aan elektronen
achter. (d) De vrijgekomen fotonen verwarmen het stralingsbad opnieuw:
welke reststraling draagt daarvan het bewijs
(\cref{ch:b3:astrophysics})?
\end{exercise}

\section{Vraagstuk: De temperaturen van de sterren lezen}

\begin{problem}[{Weekendvraagstuk --- Saha, Payne en waaruit het
heelal bestaat}]\label{pb:b3:grand-canonical:1}
Sterspectra werden in de jaren 1890 in de klassen O, B, A, F, G, K en M
gecatalogiseerd naar de sterkte van hun absorptielijnen --- waarbij de
balmerlijnen van waterstof in klasse A het sterkst waren en zowel in
hetere als in koelere sterren raadselachtig zwak. In 1925 paste Cecilia
Payne de gloednieuwe \omterm{prop:b3:grand-canonical:saha}{vergelijking van Saha} toe en toonde zij aan dat de
hele dierentuin één stof bij verschillende temperaturen was --- en dat
sterren overweldigend uit waterstof bestaan, een conclusie die zo ketters
was dat men haar dwong die in haar proefschrift af te zwakken. Dit
vraagstuk volgt haar redenering. Gegevens: de balmerabsorptie vertrekt
vanaf $n = 2$, op $E_2 - E_1 =
\qty{10.2}{eV}$ boven de grondtoestand;
$E_{\text{I}} =
\qty{13.6}{eV}$; fotosferische elektrondichtheden
$n_{\text{e}}
\sim \qty{e20}{m^{-3}}$.

\medskip\noindent\textbf{Deel I --- Twee concurrerende factoren.}
\begin{enumerate}
  \item Balmerabsorptie vergt neutrale waterstof \emph{in} $n =
        2$:
        schrijf de boltzmannfactor voor de bezetting van $n = 2$ op
        (ontaardingen: $g_2/g_1 = 4$).
  \item Reken haar uit bij $5000$, $10000$ en \qty{20000}{K}: welke kant
        duwt deze factor de lijnsterkte op naarmate $T$ stijgt?
  \item Nu de sahafactor: wat gebeurt er met de \emph{neutrale} fractie
        naarmate $T$ stijgt, en dus met de balmerlijnen?
  \item Leg kwalitatief uit waarom het product --- aangeslagen \emph{en}
        nog neutraal --- bij een of andere tussentemperatuur moet pieken.
  \item Schat met de \omterm{prop:b3:grand-canonical:saha}{vergelijking van Saha} en $n_{\text{e}} =
        \qty{e20}{m^{-3}}$ de temperatuur van halve ionisatie van
        waterstof.
  \item Zet de piek in elkaar: nabij welke temperatuur zouden de
        balmerlijnen het sterkst moeten zijn, en welke spectraalklasse
        (A-sterren, $\sim\qty{10000}{K}$) pikt dat eruit?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De spectraalreeks
ontcijferd.}
\begin{enumerate}[resume]
  \item M-sterren (\qty{3000}{K}) vertonen sterke molecuulbanden (TiO) en
        zwakke waterstof: verklaar beide met de twee factoren.
  \item O-sterren (\qty{40000}{K}) vertonen heliumlijnen en opnieuw zwakke
        waterstof: verklaar dat (wat is er met de waterstof gebeurd, en
        waarom komt helium, $E_{\text{I}} = \qty{24.6}{eV}$, nu aan de
        beurt?).
  \item De zon (G, \qty{5800}{K}) vertoont sterke lijnen van
        \emph{geïoniseerd calcium} ($E_{\text{I,Ca}} = \qty{6.1}{eV}$) en
        toch zwakke waterstof: toon met de logica van Saha aan waarom
        calcium al geïoniseerd is terwijl waterstof nog neutraal is en
        grotendeels in $n = 1$ zit.
  \item Astronomen van vóór Payne lazen lijnsterkte als hoeveelheid: wat
        deden de machtige calciumlijnen en de zwakke balmerlijnen van de
        zon hen over haar samenstelling besluiten?
  \item Het inzicht van Payne: corrigeer de sterkte van elke lijn voor
        haar ``zichtbaarheidsfactor'' van Boltzmann en Saha. Welke
        rangorde van ware hoeveelheden kwam tevoorschijn toen zij dat deed?
  \item Haar resultaat --- waterstof een miljoen keer talrijker dan
        calcium, sterren grotendeels waterstof en helium --- waarom stuitte
        het op verzet (wat suggereerde de op de aarde gerichte scheikunde),
        en wanneer kreeg zij gelijk?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Getallen op de piek.}
\begin{enumerate}[resume]
  \item Bereken de boltzmannfactor voor $n = 2$ bij precies \qty{9600}{K}.
  \item Neem met de sahaschatting uit punt 5 de neutrale fractie bij
        $9600$, $15000$ en \qty{25000}{K} ruwweg gelijk aan $0.5$,
        $\num{e-3}$ en $\num{e-5}$: vorm het product (aangeslagen fractie
        $\times$ neutrale fractie) bij uw drie temperaturen en bepaal het
        maximum.
  \item Waarom is de balmerpiek aan de hete kant \emph{scherp} (welke
        exponentiële functie wint) en aan de koele kant \emph{geleidelijk}?
  \item Een fotosfeer van een witte dwerg heeft een $n_{\text{e}}$ die
        duizend keer groter is: welke kant schuift de temperatuur van halve
        ionisatie op, en waarom (Le Chatelier in de taal van $\mu$)?
  \item Twee sterren met dezelfde temperatuur maar verschillende drukken
        vertonen dus verschillende lijnsterkten: welke stereigenschap
        (reus tegen dwerg --- de zwaartekracht aan het oppervlak) laat dat
        spectroscopisten aflezen?
  \item Moderne surveys leggen stertemperaturen tot op $\pm1\%$ vast uit
        lijnverhoudingen: formuleer de keten postulaat $\to$ Boltzmann
        $\to$ Saha $\to$ thermometer.
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Wat er is beslecht.}
\begin{enumerate}[resume]
  \item De reeks OBAFGKM was alfabetische chaos die empirisch was
        herordend: welke ene natuurkundige grootheid ordent haar na Saha?
  \item Zeg waarom \emph{hetzelfde} element bij de ene temperatuur een
        spectrum kan overheersen en bij de andere kan verdwijnen zonder
        enige verandering in hoeveelheid --- de kernles die Payne de
        astronomie leerde.
  \item Waterstof op driekwart van alle baryonische massa: noem twee
        andere pijlers van dit boek die op die hoeveelheid rusten (de
        kernfusie in sterren, \cref{ch:b3:astrophysics}; de hemel op 21 cm,
        \cref{ch:b3:spin-two-level}).
  \item Otto Struve noemde het proefschrift van Payne ``het schitterendste
        proefschrift dat ooit in de astronomie is geschreven'':
        rechtvaardig die lof in één zin vanuit dit vraagstuk.
  \item Dezelfde sahafysica, op kosmische schaal achterstevoren
        uitgevoerd, voorspelt de temperatuur waarbij de waterstof van het
        hele heelal zich voor het eerst verbond ($\sim\qty{3000}{K}$ bij
        de veel lagere kosmische dichtheid): noem de gebeurtenis en haar
        relict (\cref{ch:b3:astrophysics}).
  \item Ionisatie-evenwichten drijven ook tl-lampen en plasmabewerking
        aan: zeg in één zin waarom een geïoniseerde toestand in de stijl
        van \qty{10000}{K} kan bestaan in een buis waarvan het glas koel
        blijft (welke twee deelstelsels komen niet in evenwicht?).
  \item Vat het genoemde resultaat samen: één evenwichtsvoorwaarde,
        $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$, maakt van het
        spectraalalfabet een thermometer die van $3000$ tot \qty{40000}{K}
        loopt, legt de balmerpiek bij de A-sterren en onthult een heelal
        dat uit waterstof bestaat.
\end{enumerate}
\end{problem}
