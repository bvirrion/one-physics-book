\chapter{Kernfysica}\label{ch:b3:nuclear-physics}

% Provenance (not for print): first half of the "physique subatomique"

Het volume voor de middelbare school vertelde de hoofdlijn: atomen hebben kernen;
sommige kernen brokkelen af, op schema's van microseconden tot
miljarden jaren; en grammen materie verbergen megatonnen energie. Dit
hoofdstuk opent de kern opnieuw met het gereedschap van jaar~3. Een druppelmodel met
vijf termen voorspelt de binding van honderden \omterm{def:b3:nuclear-physics:nuclide}{nucliden} tot op een
procent; kwantumtunneling --- het barrièreprobleem uit het volume van jaar 2,
bevorderd --- verklaart hoe één formule vijfentwintig ordegrootten
aan $\alpha$-vervaltijden overspant; en het stille maximum van de
bindingsenergiekromme bij ijzer verdeelt de hele kerntechnologie in twee
kampen: splijt het zware (reactoren) of voeg het lichte samen (sterren, en
ooit elektriciteitscentrales). Het hoofdstuk eindigt naast een reactorbassin,
gloeiend in cherenkovblauw.

\section{De kern: een verzadigde druppel}

\begin{definition}[Nucliden, grootte en dichtheid]\label{def:b3:nuclear-physics:nuclide}
\index{nuclide}\index{isotoop}\index{kernstraal}
Een \emph{nuclide} $^{A}_{Z}\text{X}$ bevat $Z$ protonen en $N = A -
Z$ neutronen, gebonden door de \emph{sterke wisselwerking}: intens
(bij contact ${\sim}\qty{100}{}\times$ Coulomb), van korte dracht
(${\sim}\qty{1}{fm}$, alleen gevoeld door rakende buren) en blind
voor lading (zij grijpt protonen en neutronen even goed). Verstrooiings\-experimenten
geven elke kern hetzelfde recept: straal
\[
R = r_0A^{1/3} , \qquad r_0 \approx \qty{1.2}{fm} ,
\]
zodat het volume als $A$ groeit en de dichtheid een universele
$\rho \approx \qty{2.3e17}{kg/m^3}$ is --- een regendruppel ervan zou
wegen als een vloot tankers, en
\cref{ch:b3:astrophysics} zal hele sterren met deze dichtheid tegenkomen.
\emph{Isotopen} delen $Z$ maar verschillen in $N$: dezelfde chemie,
andere nucleaire lotgevallen.
\end{definition}

\begin{proposition}[Massadefect en bindingsenergie]\label{prop:b3:nuclear-physics:binding}
\index{massadefect}\index{bindingsenergie}\index{ijzerpiek}
Een gebonden kern weegt \emph{minder} dan zijn onderdelen; volgens
\cref{ch:b3:relativistic-dynamics} is de ontbrekende massa de
\omterm{prop:b3:nuclear-physics:binding}{bindingsenergie},
\[
B = \big(Zm_{\text{p}} + Nm_{\text{n}} - m_{\text{kern}}\big)c^2 ,
\]
doorgaans ${\sim}\qty{8}{MeV}$ \emph{per nucleon} --- een miljoen
keer de chemie. Per nucleon uitgezet klimt $B/A$ steil door de lichte kernen
(de oppervlakte-effecten verbleken), piekt bij $\approx\qty{8.8}{MeV}$ vlak bij
ijzer-56, en zakt dan weg naarmate de coulombafstoting, van lange dracht en
onverzadigbaar, de zwaargewichten belast. Beide hellingen zijn
energiemijnen: \emph{fusie} beklimt de linkerhelling (het bedrijf van de Zon),
\emph{splijting} rolt de rechter af (dat van de reactor) --- en ijzer,
op de top, is voor beide nucleaire as.
\end{proposition}

\begin{proposition}[De semi-empirische massaformule]\label{prop:b3:nuclear-physics:semf}
\index{semi-empirische massaformule}\index{stabiliteitsvallei}
Modelleer de kern als een geladen vloeistofdruppel en de hele kromme
volgt uit vijf termen:
\[
B = a_{\text{V}}A - a_{\text{S}}A^{2/3}
- a_{\text{C}}\frac{Z^2}{A^{1/3}}
- a_{\text{A}}\frac{(A - 2Z)^2}{A} + \delta(A) ,
\]
met (in \unit{MeV}) $a_{\text{V}} = 15.8$ (elk nucleon bindt zijn
buren: volume), $a_{\text{S}} = 17.8$ (nucleonen aan het oppervlak binden
er minder: de ``oppervlaktespanning'' van de druppel), $a_{\text{C}} = 0.71$
(Coulomb, de saboteur van lange dracht), $a_{\text{A}} = 23.7$
(asymmetrie: volgens \cref{ch:b3:quantum-statistics} vullen protonen en
neutronen twee aparte fermiladders, het goedkoopst als ze even
vol zijn), en $\delta$ een kleine paringsbonus voor even aantallen
(\cref{ch:b3:identical-particles}). Minimaliseren bij vaste $A$ tekent
de \emph{\omterm{prop:b3:nuclear-physics:semf}{stabiliteitsvallei}}: $N \approx Z$ voor lichte kernen,
neutronenrijk buigend ($N/Z \to 1.5$) naarmate Coulomb de protonen straft
--- en \omterm{def:b3:nuclear-physics:nuclide}{nucliden} buiten de vallei rollen er via $\beta$-verval weer in.
\end{proposition}
\admitted

\section{Radioactiviteit, met de klok in de hand}

\begin{theorem}[De vervalwet]\label{thm:b3:nuclear-physics:decay}
\index{wet van het radioactieve verval}\index{halveringstijd}\index{activiteit}
Een onstabiele kern heeft geen geheugen en veroudert niet: in elke seconde dat hij
leeft, vervalt hij met dezelfde kans $\lambda$. Voor $N$
kernen geldt $\dd N = -\lambda N\,\dd t$, dus
\[
N(t) = N_0\,\eu^{-\lambda t} = N_0\,2^{-t/T_{1/2}} ,
\qquad T_{1/2} = \frac{\ln2}{\lambda} ,
\]
en de \emph{activiteit} $\mathcal A = \lambda N$ (vervallen per
seconde, \unit{Bq}) dooft op dezelfde exponentiële uit. Halveringstijden lopen
van microseconden tot miljarden jaren; de statistiek van
\cref{ch:b3:microcanonical-ensemble} garandeert dat één
kern volstrekt onvoorspelbaar is, terwijl een mol ervan
exponentieel beter de tijd houdt dan welke klok ook --- de grondslag van de
radiometrische datering.
\end{theorem}

\begin{proof}
Een constante kans per seconde betekent $N(t + \dd t) = N(t)(1 -
\lambda\dd t)$; integreer. $N/N_0 = \tfrac12$ stellen geeft
$T_{1/2}$.
\end{proof}

\begin{remark}[Drie deuren uit een kern]\label{rem:b3:nuclear-physics:modes}
\index{alfaverval}\index{bètaverval}\index{gammaverval}\index{model van Gamow}
$\boldsymbol\alpha$: een zware kern zendt een cluster $^4$He uit. Klassiek
onmogelijk --- de coulombbarrière reikt tot \qty{25}{MeV} en het
$\alpha$-deeltje vertrekt met \qty{5}{}--\qty{9}{MeV} --- gebeurt het door
\emph{tunneling} (de barrière uit het volume van jaar 2, gekromd): de
exponentiële gevoeligheid van Gamow maakt van een factor twee in energie twintig
ordegrootten in levensduur, precies het patroon van Geiger--Nuttall
(\cref{exo:b3:nuclear-physics:8}).
$\boldsymbol\beta$: een neutron wordt een proton (of omgekeerd), waarbij het
een elektron (of positron) en een neutrino uitzendt --- de \emph{zwakke
wisselwerking} aan het werk, de kracht die de vallei herstelt; het verhaal van het
neutrino wacht op \cref{ch:b3:particle-physics}.
$\boldsymbol\gamma$: een aangeslagen kern werpt fotonen van \unit{MeV} af, het
nucleaire analogon van de spectraallijnen uit \cref{ch:b3:hydrogen-atom}.
Vervalstappen ketenen tot de valleibodem is bereikt: de ladder van uranium
eindigt, veertien sporten later, bij stabiel lood.
\end{remark}

\section{De hellingen af: splijting en fusie}

\begin{proposition}[Splijting en de kettingreactie]\label{prop:b3:nuclear-physics:fission}
\index{splijting}\index{kettingreactie}\index{kritieke massa}
Een langzaam neutron dat door $^{235}$U wordt opgenomen maakt een wiebelende druppel die
splijt: twee fragmenten van middelmatige massa, twee tot drie verse neutronen, en
$\approx\qty{200}{MeV}$ --- het gat in $B/A$ tussen de
\qty{7.6}{MeV} van uranium en de \qty{8.5}{MeV} van de fragmenten, maal 235.
De vrijgekomen neutronen kunnen verdere kernen splijten: met
vermenigvuldigingsfactor $k$ (neutronen per neutron, één generatie
later) groeit een populatie als $k^n$ --- subkritisch ($k < 1$)
dooft uit, kritisch ($k = 1$) blijft gelijk, superkritisch explodeert.
Een reactor remt neutronen tot lage snelheden af (het voorkeursdieet van de splijting),
leunt op de 0.7\,\% van de neutronen die \emph{seconden}
te laat uit fragmentverval komen --- de vertraging die
$k \approx 1$ menselijk regelbaar maakt --- en laat regelstaven het
overschot opeten (\cref{pb:b3:nuclear-physics:1}).
\end{proposition}
\admitted

\begin{remark}[Fusie: de moeilijkere, betere helling]\label{rem:b3:nuclear-physics:fusion}
\index{fusie}
Lichte kernen samenvoegen betaalt per kilogram $\sim4\times$ meer dan
zware splijten, zonder langlevende fragmenten --- maar de
reactanten, allebei positief, moeten door hun onderlinge coulombbarrière
tunnelen, wat temperaturen van $10^7$--$10^8\,\unit{K}$ vergt
(\cref{exo:b3:nuclear-physics:11}). Sterren doen het door enorm te zijn
(\cref{ch:b3:astrophysics}: de Zon maakt elke seconde \qty{600}{} miljoen
ton waterstof tot helium); laboratoria doen het
met magnetisch ingesloten plasma's en, tot dusver, een energiegrootboek
dat nog achterblijft bij zijn ambities om quitte te spelen.
\end{remark}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:1}
Boekhouding. (a) Geef $Z$, $N$, $A$ voor $^{12}$C, $^{14}$C,
$^{235}$U, $^{238}$U. (b) Welke paren zijn isotopen? (c) Waarom delen
isotopen de chemie maar niet de nucleaire stabiliteit? (d) $^{14}$C
en $^{14}$N delen $A = 14$: hoe heten zulke \omterm{def:b3:nuclear-physics:nuclide}{nucliden}, en
welk verval verbindt ze?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:2}
De universele dichtheid. (a) Toon vanuit $R = r_0A^{1/3}$ aan dat de
kerndichtheid onafhankelijk is van $A$ en evalueer haar. (b)
Bereken de massa van een theelepel (\qty{5}{mL}) kernmaterie.
(c) Welke fractie van het volume van een atoom vult zijn kern (neem
$R_{\text{atoom}} = \qty{1e-10}{m}$, $A = 60$)? (d) Wat zegt de
wet $A^{1/3}$ zelf over de dracht van de sterke kracht?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:3}
De lijm wegen. Kernmassa's: $m_{\text{p}} =
\qty{1.007276}{u}$, $m_{\text{n}} = \qty{1.008665}{u}$,
$m(^4\text{He}) = \qty{4.001506}{u}$;
$\qty{1}{u}\,c^2 = \qty{931.5}{MeV}$. (a) Bereken het \omterm{thm:b3:relativistic-dynamics:emc2}{massadefect} en
de \omterm{prop:b3:nuclear-physics:binding}{bindingsenergie} van $^4$He. (b) Geef $B/A$. (c) Vergelijk $B$ met
de \qty{13.6}{eV} van \cref{ch:b3:hydrogen-atom}: hoeveel ordegrootten
scheiden de nucleaire van de atomaire lijm? (d) Waarom maakt
de uitzonderlijke binding van $^4$He (de piek in de figuur) hem tot de
munt van het $\alpha$-verval?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:4}
De eerste curie. Radium-226: $T_{1/2} = 1600$ jaar, molaire massa
\qty{226}{g/mol}. Voor één gram: (a) tel de kernen; (b) bereken
$\lambda$; (c) bereken de activiteit en vergelijk met de historische
eenheid $1\,\text{Ci} = \qty{3.7e10}{Bq}$ (gedefinieerd als precies deze
gram); (d) hoeveel van de gram is er na 4800 jaar over?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:5}
De formule aan het werk. Met $a_{\text{V}} = 15.75$,
$a_{\text{S}} = 17.8$, $a_{\text{C}} = 0.711$,
$a_{\text{A}} = 23.7$ (\unit{MeV}) en paring
$+12/\sqrt A$ voor even-even kernen: (a) bereken de vier hoofdtermen
van $B$ voor $^{56}$Fe ($Z = 26$). (b) Tel op (met paring) en
vergelijk $B/A$ met de gemeten \qty{8.79}{MeV}. (c) Welke twee
termen vechten het hardst, en wat legt hun evenwicht vast? (d) Waarom
moet juist de coulombterm \emph{sneller} groeien dan $A$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:6}
De valleibodem. (a) Houd alleen de van $Z$ afhangende termen over,
minimaliseer de massa bij vaste $A$ en leid
$Z^* = \dfrac{A/2}{1 + a_{\text{C}}A^{2/3}/4a_{\text{A}}}$ af. (b)
Evalueer voor $A = 101$ en vergelijk met ruthenium ($Z = 44$).
(c) Toon aan dat de limiet voor lichte $A$ gelijk is aan $Z^* = A/2$ en interpreteer dat met
de twee fermiladders. (d) Een splijtingsfragment heeft $A = 95$,
$Z = 36$: hoe ver ligt het van de vallei, en welke reeks
vervalstappen volgt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:7}
Radiokoolstof. Levende materie houdt haar $^{14}$C
($T_{1/2} = 5730$ jaar) op peil bij 1 atoom per $10^{12}$
koolstofatomen; de dood stopt de aanvulling. (a) Een houtskoolmonster vertoont
30\,\% van de levende activiteit: hoe oud is het vuur? (b) Eén gram
levende koolstof: bereken zijn $^{14}$C-activiteit
(verwacht $\approx\qty{0.23}{Bq}$). (c) Waarom is de methode nutteloos
voorbij $\sim\qty{50000}{}$ jaar? (d) En waarom is zij nutteloos voor het
dateren van gesteenten (welke klokken nemen het over)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:8}
De klif van Gamow. Polonium-212 zendt een $\alpha$ van \qty{8.78}{MeV} uit;
zijn barrière reikt tot $\approx\qty{26}{MeV}$. (a) Toon aan dat klassieke
ontsnapping onmogelijk is, en klassieke binnenkomst ook --- en toch vervalt het
in \qty{0.3}{\micro s}. (b) Het $\alpha$-deeltje van uranium-238 draagt
\qty{4.2}{MeV} en zijn halveringstijd is $4.5\times10^9$ jaar:
bereken de verhouding van de twee vervalconstanten. (c) Leg met
de tunnelexponentiële uit hoe een factor twee in energie
$\sim10^{24}$ in tempo koopt. (d) Waarom legt dezelfde fysica de eis aan de
kerntemperatuur van de Zon vast?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:9}
Tweehonderd miljoen elektronvolt. (a) Onderbouw de
$\qty{200}{MeV}$ per splijting vanuit de $B/A$-kromme. (b) Bereken
de energie in één kilogram volledig gespleten $^{235}$U, in
joule en in ton olie-equivalent
(\qty{42}{GJ/t}). (c) Een centrale van \qty{1}{GW_e} met 33\,\%
rendement: hoeveel kilogram $^{235}$U per jaar? (d) Waarom dragen
de fragmenten, en niet de neutronen, het meeste van de
\qty{200}{MeV}, en in welke vorm gaat die onmiddellijk over?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:nuclear-physics:10}
$k$ temmen. In een reactor planten prompte neutronen zich voort in $\ell
\approx \qty{e-4}{s}$. (a) Bereken met $k = 1.001$ op alleen prompte
neutronen de vermogensvermenigvuldiging in één seconde --- en het
oordeel over mechanische regeling. (b) Een fractie $\beta =
0.7\,\%$ van de neutronen is $\sim\qty{10}{s}$ \emph{vertraagd}:
leg uit waarom de effectieve klok van de keten voor $k - 1 < \beta$
seconden wordt. (c) Bereken dezelfde vermenigvuldiging over één seconde
met de effectieve generatietijd $\sim\qty{0.1}{s}$. (d) Zeg
in één zin wat de operators van Tsjernobyl kwijtraakten toen hun
reactor prompt-superkritisch werd.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:nuclear-physics:11}
Het rekenwerk van de Zon. In totaal zet $4\,^1\text{H} \to {}^4\text{He}$
\qty{0.71}{\percent} van de massa om in energie
(\qty{26.7}{MeV}, neutrino's inbegrepen). (a) Bereken uit $L_\odot =
\qty{3.8e26}{W}$ de massa die per seconde wordt omgezet, en de
waterstof die per seconde wordt verbruikt. (b) Schat met $\qty{2e30}{kg}$ zon,
waarvan een tiende brandbare kernwaterstof (neem de waterstoffractie
$\approx0.75$), de levensduur. (c) Twee protonen bij
\qty{1.5e7}{K}: vergelijk $k_{\text{B}}T$ met hun coulombbarrière
bij \qty{1}{fm} en concludeer wie de oversteek maakt
(\cref{exo:b3:nuclear-physics:8}). (d) Waarom laat fusie, anders dan
splijting, vrijwel geen radioactieve as achter?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:nuclear-physics:12}
Nucleaire geneeskunde. (a) Technetium-99m ($T_{1/2} = \qty{6}{h}$,
$\gamma$ van \qty{140}{keV}) is het werkpaard van de geneeskunde: bereken het
aantal kernen, en hun massa, achter een geïnjecteerde
\qty{500}{MBq}. (b) Leg uit waarom een halveringstijd van zes \emph{uur} en een
zuivere $\gamma$ elk precies zijn wat een diagnose wil. (c) PET:
een positron uit $^{18}$F annihileert met een elektron ---
gebruik \cref{ch:b3:relativistic-dynamics} om de energie en de
meetkunde van de twee fotonen te geven, en daarmee het principe van de detector.
(d) Radiotherapie levert \qty{2}{Gy} (\unit{J/kg}) aan een tumor:
vergelijk de energie met een slok warm water, en breng de
onschadelijkheid van de joules in overeenstemming met de dodelijkheid van de ionisatie.
\end{exercise}

\section{Vraagstuk: Open dag bij de onderzoeksreactor}

\begin{problem}\label{pb:b3:nuclear-physics:1}
\emph{Open dag bij de onderzoeksreactor.} Het nationale laboratorium
stelt zijn onderzoeksreactor van \qty{20}{MW} van het bassintype open voor bezoekers, en
u hebt uzelf het balkon op gepraat: beneden, door acht
meter helder water, gloeit de kern bovenaards blauw. De
gids --- een reactorfysicus die met pensioen gaat --- heeft u toegestaan
het rekenwerk te doen.

\medskip\noindent\textbf{Deel I --- Het energiegrootboek.}

\begin{enumerate}
  \item Elke splijting van $^{235}$U komt op ongeveer \qty{200}{MeV}.
        Reken om naar joule en bereken de splijtingen per seconde
        die \qty{20}{MW} in stand houden.
  \item Reken om naar grammen $^{235}$U die per dag worden verbruikt.
  \item Dezelfde \qty{20}{MW} uit steenkool (\qty{30}{MJ/kg}): tonnen
        per dag? Vorm de massaverhouding en verbind die met de
        schalen \qty{}{MeV} tegenover \qty{}{eV} van dit hoofdstuk en
        van de chemie.
  \item De fragmenten worden geboren bij een binding van $\sim\qty{8.5}{MeV}$ per
        nucleon, uranium bij \qty{7.6}{MeV}: controleer dat
        $235\times0.9 \approx \qty{200}{MeV}$ consistent is.
  \item De fragmenten komen binnen micrometers van hun geboorteplaats tot stilstand.
        Waarin zetten zij hun \qty{200}{MeV} om, en op welke
        tijdschaal bereikt die het koelwater?
  \item Waarom zijn de fragmenten (bijv.\ $A = 95$, $Z = 36$)
        onvermijdelijk radioactief? Gebruik de \omterm{prop:b3:nuclear-physics:semf}{stabiliteitsvallei}.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- $k = 1$ vasthouden.}

\begin{enumerate}[resume]
  \item Definieer de vermenigvuldigingsfactor $k$ en geef aan wat
        $k = 0.999$, $1.000$ en $1.001$ voor de neutronenpopulatie
        betekenen.
  \item Het bassinwater remt af. Leg in twee zinnen uit waarom
        neutronen tot thermische snelheden afremmen de splijting van
        $^{235}$U \emph{helpt}, en waarom waterstof de beste moderator is
        (denk aan de elastische botsingen uit het volume van jaar 1).
  \item Regelstaven zijn van boorstaal. Wat doet boor, en
        hoe stuurt het heffen of laten zakken van de staven $k$?
  \item Bereken met alleen prompte neutronen ($\ell = \qty{e-4}{s}$) en
        $k = 1.0005$ de vermogensgroei over één seconde
        en trek uw conclusie.
  \item De 0.7\,\% vertraagde neutronen rekken de effectieve
        generatietijd tot $\sim\qty{0.1}{s}$: reken opnieuw, en
        formuleer de ontwerpregel die $k - 1$ veilig onder
        de vertraagde fractie houdt.
  \item Het water is ook het koelmiddel. Leg de intrinsieke
        veiligheidseigenschap uit: wat gebeurt er met het afremmen --- en dus
        met $k$ --- als het water wegkookt?
  \item Waarom heeft de reactor na uitschakeling toch nog noodkoeling
        nodig? (Noem de warmtebron waar regelstaven niets aan
        kunnen doen.)
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het blauwe licht.}

\begin{enumerate}[resume]
  \item De gloed is cherenkovstraling: de lichtsnelheid in water
        is $c/n$ met $n = 1.33$. Wat moet een geladen deeltje doen
        om haar uit te zenden?
  \item Bereken de drempelsnelheid, en met
        \cref{ch:b3:relativistic-dynamics} de drempelwaarde van de kinetische
        energie voor een elektron.
  \item Welke deeltjes in de reactor komen daarvoor in aanmerking? Volg de keten van
        splijtingsfragment tot snel $\beta$-elektron in het water.
  \item De intensiteit van het spectrum groeit naar korte golflengten toe:
        waarom gloeit het bassin blauw en niet rood?
  \item De gids zegt: ``de gloed \emph{is} de afscherming die
        werkt.'' Leg uit --- wat doet acht meter water
        met de straling van de kern, en waarom is het balkon ruwweg
        veilig?
  \item Na uitschakeling dooft het blauw in minuten, maar het verdwijnt pas
        na dagen. Leg het verband met vraag 6 van deel I.
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Waar de reactor voor is.}

\begin{enumerate}[resume]
  \item Het dagelijkse werk van deze reactor is molybdeen-99 maken
        ($T_{1/2} = \qty{66}{h}$), de ouder van het
        technetium-99m van de geneeskunde. Waarom moeten de ziekenhuizen ter wereld
        wekelijks worden bevoorraad, en waarom kan geen enkel magazijn het inslaan?
  \item Een zending Mo-99 van \qty{100}{GBq} vertrekt op maandag; het
        ziekenhuis elueert Tc-99m op vrijdag (96 uur later). Welke
        activiteit van de ouder is er over?
  \item Neutronenbundels uit de kern voeden ook de diffractie-instrumenten
        van \cref{ch:b3:crystalline-solids}: waarom worden reactorneutronen,
        eenmaal thermisch, met de juiste golflengte geboren?
  \item Het afscheidsvraagstuk van de gids: een splijtstofelement brengt vijf
        jaar in het bassin door voordat het wordt verscheept. Leg met de
        mengeling van halveringstijden van de fragmenten de logica uit --- wat hebben vijf
        jaar bassintijd gekocht?
  \item Schat de vervalwarmte één uur na uitschakeling
        ($\approx1\,\%$ van \qty{20}{MW}) en vergelijk die met een
        elektrische kachel in huis: is passieve bassinkoeling
        aannemelijk?
  \item Sluit het grootboek in vier regels: $B/A$ betaalt
        \qty{200}{MeV} per splijting; vertraagde neutronen lenen de
        seconden die $k$ stuurbaar maken; water remt af,
        koelt, schermt af en gloeit; en het echte product van de dag
        vertrekt in medische flesjes, niet in megawatts.
\end{enumerate}
\end{problem}
