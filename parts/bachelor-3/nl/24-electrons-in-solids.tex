\chapter{Elektronen in vaste stoffen}\label{ch:b3:electrons-in-solids}

% Provenance (not for print): second half of the "physique du solide"
% module (Drude, Sommerfeld, bands, semiconductors) common to the
% surveyed third-year curricula.

Koper geleidt elektriciteit een miljoen miljard miljard keer beter
dan kwarts --- het grootste bereik van welke fysische eigenschap ook,
gecommandeerd door materialen die er hoe dan ook als grijze klompen uitzien. Het geraamte van
het vorige hoofdstuk verklaart er niets van; de \emph{elektronen} die erin
worden gegoten verklaren alles. Dit hoofdstuk loopt de drie aanvallen van de eeuw op
het probleem door, die elk het wrak van de vorige erven: het klassieke flipperspel
van Drude (1900), dat de wet van Ohm goed krijgt en de warmtecapaciteit
schandalig fout; het fermigas van Sommerfeld, dat
\cref{ch:b3:quantum-statistics} zijn cheque laat innen; en het inzicht van Bloch
dat een elektrongolf in een periodiek \omterm{def:b3:crystalline-solids:lattice}{rooster} net als elke andere golf
braggreflectie ondergaat --- waardoor \emph{bandkloven} opengaan die alle vaste stoffen
sorteren in metalen, \omterm{def:b3:electrons-in-solids:classes}{isolatoren} en de middenkinderen met een smalle kloof,
de \omterm{def:b3:electrons-in-solids:classes}{halfgeleiders}, waarop het hele elektronische tijdperk is gebouwd. Het
hoofdstuk eindigt in een zonnepaneel.

\section{Het flippermetaal van Drude}

\begin{proposition}[De geleidbaarheid van Drude]\label{prop:b3:electrons-in-solids:drude}
\index{model van Drude}\index{geleidbaarheid!microscopisch}\index{driftsnelheid}
Behandel de valentie-elektronen van een \omterm{def:b3:electrons-in-solids:classes}{metaal} als een klassiek gas met dichtheid $n$,
dat om de $\tau$ seconden tegen hindernissen botst. Een veld $\vect E$
versnelt elk elektron tussen de botsingen; de gemiddelde \emph{\omterm{prop:b3:electrons-in-solids:drude}{driftsnelheid}}
is $\vect v_{\text{d}} = -e\tau\vect E/m_{\text{e}}$,
piepklein naast de thermische beweging. De stroomdichtheid $\vect j =
-ne\vect v_{\text{d}}$ gehoorzaamt dan plaatselijk aan de wet van Ohm,
$\vect j = \sigma\vect E$, met
\[
\sigma = \frac{ne^2\tau}{m_{\text{e}}} .
\]
De gemeten $\sigma$ van koper eist $\tau \approx \qty{2.5e-14}{s}$
--- en de twee glorierijke resultaten van het model volgen: het verklaart waarom weerstanden
warm worden (de arbeid van het veld wordt bij elke botsing thermisch), en het
voorspelt de wet van Wiedemann--Franz, dat $\kappa/\sigma T$ voor alle metalen
dezelfde constante is --- elektronen dragen zowel lading als
warmte (\cref{exo:b3:electrons-in-solids:5}).
\end{proposition}

\begin{proof}
Tussen de botsingen geldt $\dot{\vect v} = -e\vect E/m_{\text{e}}$;
als het elektron bij elke botsing opnieuw begint, is de gemiddelde snelheid die het in een
gemiddelde tijd $\tau$ opdoet $-e\tau\vect E/m_{\text{e}}$. Dan is $\vect j =
-ne\vect v_{\text{d}} = (ne^2\tau/m_{\text{e}})\vect E$.
\end{proof}

\begin{remark}[De redding door Sommerfeld]\label{rem:b3:electrons-in-solids:sommerfeld}
\index{model van Sommerfeld}\index{fermi-energie!in metalen}
Het gas van Drude zou $\tfrac32k_{\text{B}}$ per elektron aan de
warmtecapaciteit van een \omterm{def:b3:electrons-in-solids:classes}{metaal} moeten toevoegen; het experiment vindt honderd keer minder.
\Cref{ch:b3:quantum-statistics} sprak de elektronen al vrij:
zij vormen een \emph{ontaard fermigas}, $T_{\text{F}} \sim
\qty{8e4}{K}$, en alleen de thermische schil binnen
$k_{\text{B}}T$ van het fermioppervlak kan reageren --- op warmte en
op verstrooiing. Sommerfelds herhaling van Drude met fermi-dirac\-statistiek
behoudt de formule voor de geleidbaarheid (met $\tau$ nu de
levensduur van de elektronen aan het fermioppervlak, waarvan de \qty{e6}{m/s}
$v_{\text{F}}$ is en niet de thermische snelheid), zet de warmtecapaciteit terug op haar
gemeten flintertje $T/T_{\text{F}}$, en repareert zelfs de
getalsconstante van Wiedemann--Franz. Eén raadsel overleeft elk
vrije-elektronmodel: waarom kwarts helemaal niet geleidt.
\end{remark}

\section{Bloch: het rooster spreekt}

\begin{theorem}[Blochtoestanden en bandkloven]\label{thm:b3:electrons-in-solids:bloch}
\index{theorema van Bloch}\index{bandkloof}\index{bandenstructuur}
In een perfect periodieke potentiaal zijn de stationaire toestanden
\emph{blochgolven} $\psi_k(x) = u_k(x)\eu^{\iu kx}$ ---
vlakke golven aangekleed met de eigen periodiciteit $u_k$ van het \omterm{def:b3:crystalline-solids:lattice}{rooster} ---
die zich \emph{zonder verstrooiing} voortplanten: een perfect \omterm{def:b3:crystalline-solids:lattice}{kristal} heeft
weerstand nul, en echte weerstand komt van onvolkomenheden
(trillingen, onzuiverheden). Maar niet alle energieën overleven. Bij
$k = \pi/a$ is de golflengte van het elektron $2a$: de braggvoorwaarde
op het \omterm{def:b3:crystalline-solids:lattice}{rooster} zelf (\cref{thm:b3:crystalline-solids:dispersion}
deed dezelfde ontdekking voor geluid). De weerkaatste en de invallende
golf vormen twee staande patronen --- lading opgestapeld op de ionen
(lagere energie) of ertussen (hogere) --- zodat de parabool
$E = \hbar^2k^2/2m_{\text{e}}$ van het vrije elektron openscheurt: een interval van
verboden energieën, de \emph{\omterm{thm:b3:electrons-in-solids:bloch}{bandkloof}} $E_{\text{g}}$, scheidt
een volgemaakte \emph{band} van toegestane toestanden van de volgende. De
elektronische identiteit van een \omterm{def:b3:crystalline-solids:lattice}{kristal} is zijn ladder van banden en kloven.
\end{theorem}
\admitted

\begin{definition}[Metalen, isolatoren, halfgeleiders]\label{def:b3:electrons-in-solids:classes}
\index{metaal}\index{isolator}\index{halfgeleider}
Vul de banden met de elektronen van het \omterm{def:b3:crystalline-solids:lattice}{kristal}, twee per toestand
(\cref{ch:b3:identical-particles}), de koudste eerst. Als de bovenste
bezette band \emph{gedeeltelijk} gevuld is, kan een oneindig kleine energie
elektronen in netto beweging brengen: een \emph{metaal} (natrium: één
valentie-elektron, een halve band; tweewaardige metalen geleiden via
overlappende banden). Als hij \emph{precies} vol is, met een brede kloof
erboven, kan geen klein duwtje iets herschikken: een \emph{isolator}
(diamant, $E_{\text{g}} = \qty{5.5}{eV}$). Is de kloof klein ---
$\qty{1.12}{eV}$ in silicium --- dan hijst de thermische onrust een paar
elektronen erover, die er elk een beweeglijk \emph{\omterm{def:b3:electrons-in-solids:doping}{gat}} onder achterlaten: een
\emph{halfgeleider}, waarvan het aantal ladingsdragers
$\propto\eu^{-E_{\text{g}}/2k_{\text{B}}T}$ om de paar graden verdubbelt.
Vandaar de grote scheidslijn: metalen geleiden slechter als ze warm worden
(meer trillingen om aan te verstrooien), halfgeleiders beter
(exponentieel meer ladingsdragers) --- één tekenwissel die de klasse van een
materiaal in één meting vaststelt.
\end{definition}

\section{De kloof ontwerpen: dotering en de junctie}

\begin{definition}[Dotering]\label{def:b3:electrons-in-solids:doping}
\index{dotering}\index{donor}\index{acceptor}\index{gat}
Vervang één siliciumatoom op een miljoen door fosfor (vijf
valentie-elektronen): vier bindingen zijn verzadigd en het vijfde elektron,
dat met een schamele \qty{45}{meV} gebonden is
(\cref{exo:b3:electrons-in-solids:10}), is bij kamertemperatuur
vrij. Zulke \emph{donoren} maken een \emph{n-type}
\omterm{def:b3:electrons-in-solids:classes}{halfgeleider}, met geleiding door elektronen; boor (drie elektronen)
is een \emph{acceptor}, die een bindingselektron grijpt en een
beweeglijk \emph{gat} vrijmaakt --- een ontbrekend elektron dat beweegt, op
velden reageert en net zo echt positieve lading draagt als een bel
opwaartse kracht draagt: \emph{p-type}. Dotering laat de ladingsdragersdichtheid
--- en daarmee de geleidbaarheid --- naar believen over zes ordegrootten zwaaien:
de knop die zand in schakelingen verandert.
\end{definition}

\begin{proposition}[De p--n-junctie]\label{prop:b3:electrons-in-solids:junction}
\index{p--n-junctie}\index{diode}\index{depletiezone}
Voeg p-type aan n-type. Elektronen stromen naar de \omterm{def:b3:electrons-in-solids:doping}{gaten} toe en
vernietigen ze vlak bij het grensvlak, waardoor een \emph{\omterm{prop:b3:electrons-in-solids:junction}{depletiezone}} van
vaste geïoniseerde doteringsatomen bloot komt te liggen, waarvan de dubbellaag een
inwendig veld opbouwt --- evenwicht bij de ingebouwde spanning
$V_{\text{bi}} = (k_{\text{B}}T/e)
\ln(N_{\text{a}}N_{\text{d}}/n_{\text{i}}^2) \approx
\qty{0.7}{V}$ in silicium. De junctie gelijkricht dan: voorwaartse
spanning verlaagt de barrière en de stroom groeit als
$\eu^{eV/k_{\text{B}}T}$; tegenspanning verhoogt hem en er vloeit alleen een
lekstroom $I_0$:
\[
I = I_0\big(\eu^{eV/k_{\text{B}}T} - 1\big) .
\]
Deze asymmetrie is de \emph{diode} --- en in haar varianten
de led (recombinerende paren zenden fotonen met de kloofenergie uit), de
zonnecel (fotonen met de kloofenergie maken paren die het ingebouwde veld
naar buiten veegt: \cref{pb:b3:electrons-in-solids:1}), en, verdubbeld
tot sandwiches, de transistor: de schakelaar waarvan een moderne
chip er honderden miljarden drukt.
\end{proposition}
\admitted

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:1}
Het koper van Drude. $n = \qty{8.5e28}{m^{-3}}$, soortelijke weerstand
$\rho = \qty{1.7e-8}{\ohm.m}$. Bereken (a) $\tau =
m_{\text{e}}/ne^2\rho$; (b) de gemiddelde vrije weglengte met de
fermisnelheid $v_{\text{F}} = \qty{1.6e6}{m/s}$; (c) die weglengte in
\omterm{def:b3:crystalline-solids:lattice}{roosterconstanten} ($a = \qty{3.61}{\angstrom}$); (d) leg uit waarom
zo'n lange vlucht er al op wijst dat de ionen zelf niet voor de
verstrooiing zorgen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:2}
De slak in de draad. Een koperdraad van \qty{1}{mm^2} voert
\qty{10}{A}. (a) Bereken de \omterm{prop:b3:electrons-in-solids:drude}{driftsnelheid}. (b) Hoe lang zou
een elektron erover doen om een lampsnoer van \qty{1}{m} over te steken? (c) Waarom gaat
de lamp toch onmiddellijk aan? (d) Vergelijk
$v_{\text{d}}$ met $v_{\text{F}}$ en becommentarieer het
flipperbeeld van Drude.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:3}
Sorteren op banden. Classificeer met de bandvullingsregel: (a)
natrium (één valentie-elektron); (b) magnesium (twee --- en toch een
\omterm{def:b3:electrons-in-solids:classes}{metaal}: wat moeten zijn banden doen?); (c) diamant
($E_{\text{g}} = \qty{5.5}{eV}$); (d) silicium
($\qty{1.12}{eV}$) --- en geef voor elk het teken van
$\dd\rho/\dd T$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:4}
Kloven en fotonen. (a) Welke fotongolflengte past bij de kloof van
silicium, en in welk spectraalgebied wordt silicium
doorzichtig? (b) Idem voor diamant: waarom is hij helder in het
zichtbare? (c) Een led van galliumnitride heeft $E_{\text{g}} =
\qty{3.4}{eV}$: welke kleurgrens legt dat vast, en waarom hadden blauwe
leds dit materiaal nodig? (d) Waarom sterft een rode led donker in plaats van
flauw wit te gloeien als u hem te ver opdrijft?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:5}
Wiedemann--Franz. (a) Bereken uit de $\kappa =
\qty{400}{W/(m.K)}$ en $\sigma = \qty{5.9e7}{S/m}$ van koper bij
\qty{300}{K} de waarde van $\kappa/\sigma T$. (b) Vergelijk met de
sommerfeldwaarde $L = \pi^2k_{\text{B}}^2/3e^2 =
\qty{2.44e-8}{W.\ohm/K^2}$. (c) Leg in één zin uit waarom één
soort ladingsdrager de twee geleidbaarheden aan elkaar knoopt. (d) Kookpannen
en hun handvatten: gebruik de wet om de materiaalkeuzes in een keuken
te verklaren.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:6}
De vrijspraak van de warmtecapaciteit. (a) Drude voorspelt
$\tfrac32nk_{\text{B}}$ van de elektronen: welke fractie zou dat met de $n$ van
koper toevoegen aan de $3nk_{\text{B}}$ van het \omterm{def:b3:crystalline-solids:lattice}{rooster}
(één geleidingselektron per atoom)? (b)
\Cref{ch:b3:quantum-statistics} geeft in plaats daarvan $C_{\text{el}}
\sim \tfrac{\pi^2}{2}nk_{\text{B}}(T/T_{\text{F}})$: evalueer de
onderdrukking $T/T_{\text{F}}$ bij \qty{300}{K}
($T_{\text{F}} = \qty{8.1e4}{K}$). (c) Waarom \emph{wint} de elektronische
term toch bij vloeibaarheliumtemperaturen
(denk aan de $T^3$ van het \omterm{def:b3:crystalline-solids:lattice}{rooster})? (d) Welke meting, uitgezet als
$C/T$ tegen $T^2$, ontwart de twee?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:7}
Het fermioppervlak ontmoet de zone. (a) Bereken voor koper de
fermigolflengte $\lambda_{\text{F}} = 2\pi/k_{\text{F}}$ met
$k_{\text{F}} = (3\pi^2n)^{1/3}$. (b) Vergelijk met $2a$: hoe
dicht ligt het fermioppervlak bij de braggvoorwaarde? (c) Leg
fysisch uit waarom de twee staande golven bij $k = \pi/a$ (lading op de
ionen tegenover ertussen) in energie moeten verschillen. (d) Welk
experimentele feit uit \cref{exo:b3:electrons-in-solids:1} verklaart
het niet-verstrooiingstheorema van Bloch eindelijk?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:8}
De exponentiële thermometer. Intrinsiek silicium heeft $n_{\text{i}}
\propto \eu^{-E_{\text{g}}/2k_{\text{B}}T}$. (a) Bereken de
verhouding van de ladingsdragersdichtheden tussen \qty{400}{K} en \qty{300}{K}. (b)
Schets daaruit de weerstand van een thermistor tegen de temperatuur en
zet die af tegen die van een platinadraad. (c) Waarom de factor 2 in de
exponent (wat wordt er \emph{in paren} gemaakt)? (d) Schat de
temperatuur waarbij silicium-elektronica faalt doordat intrinsieke
ladingsdragers een \omterm{def:b3:electrons-in-solids:doping}{dotering} van \qty{5e21}{m^{-3}} overspoelen
($n_{\text{i}}(300\,\text{K}) \approx \qty{1e16}{m^{-3}}$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:9}

fosforatoom per miljoen siliciumatomen: welke ladingsdragersdichtheid, en

Leg uit waarom de halfgeleiderfabricage eerst tot delen per

gemiddelde afstand tussen \omterm{def:b3:electrons-in-solids:doping}{donoren} en vergelijk die met de
baan van \qty{2.4}{nm} uit \cref{exo:b3:electrons-in-solids:10}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electrons-in-solids:10}
De \omterm{def:b3:electrons-in-solids:doping}{donor} als \omterm{thm:b3:hydrogen-atom:levels}{waterstofatoom}. Het vijfde fosforelektron
draait om zijn ion met lading $+e$ \emph{binnen} silicium: scherm de coulombkracht af met
$\varepsilon_{\text{r}} = 11.7$
(\cref{ch:b3:electromagnetism-in-matter}) en maak het elektron lichter
tot zijn effectieve bandmassa $m^* = 0.26\,m_{\text{e}}$.
(a) Schaal de waterstofresultaten van \cref{ch:b3:hydrogen-atom}:
toon aan dat $E = \qty{13.6}{eV}\times(m^*/m_{\text{e}})/
\varepsilon_{\text{r}}^2$ en evalueer dat. (b) Vergelijk met
$k_{\text{B}}T$ bij \qty{300}{K}: zijn de \omterm{def:b3:electrons-in-solids:doping}{donoren} geïoniseerd? (c) Schaal
de \omterm{thm:b3:hydrogen-atom:levels}{bohrstraal} op dezelfde manier. (d) De baan omspant tientallen
roostercellen: leg uit waarom dat zelfconsistent rechtvaardigt dat we
$\varepsilon_{\text{r}}$ en $m^*$ überhaupt gebruiken.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electrons-in-solids:11}
Junctiegetallen. Een siliciumdiode heeft $N_{\text{a}} =
N_{\text{d}} = \qty{1e22}{m^{-3}}$, $n_{\text{i}} =
\qty{1e16}{m^{-3}}$, $T = \qty{300}{K}$. (a) Bereken
$V_{\text{bi}} = (k_{\text{B}}T/e)\ln(N_{\text{a}}N_{\text{d}}/
n_{\text{i}}^2)$. (b) Bereken uit de diodewet de verhouding van de
stromen bij $+\qty{0.5}{V}$ en $-\qty{0.5}{V}$. (c) Waarom verdubbelt
$I_0$ --- en daarmee de lekstroom in de sperrichting --- ruwweg om de
\qty{10}{K}? (d) Een brug van vier zulke diodes maakt van wisselstroom gelijkstroom:
schets het idee van de schakeling in woorden.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electrons-in-solids:12}
De zonnekloof ontwerpen. (a) Fotonen onder $E_{\text{g}}$ gaan er

\end{exercise}

\section{Vraagstuk: Het dakreferendum}

\begin{problem}\label{pb:b3:electrons-in-solids:1}
\emph{Het dakreferendum.} Je buurman weegt af of hij
zijn huis met zonnepanelen zal bedekken en heeft u, de fysicus,
tot scheidsrechter benoemd. Het kandidaat-paneel: silicium,
\qty{2}{m^2}, zestig cellen in serie, met een nominaal vermogen van \qty{400}{W} onder
het standaard zonlicht van \qty{1000}{W/m^2} op het middaguur.

\medskip\noindent\textbf{Deel I --- Zonlicht ontmoet de kloof.}

\begin{enumerate}
  \item Zonlicht is ruwweg een thermisch spectrum van \qty{5800}{K}
        (\cref{ch:b3:photons-phonons}): bereken de energie van zijn
        piekfotonen (Wien) in eV.
  \item De kloof van silicium is \qty{1.12}{eV}: wat is de langste
        golflengte die een siliciumcel kan oogsten?
  \item Ruwweg een vijfde van het zonnevermogen komt aan in fotonen
        onder de kloof. Wat gebeurt daarmee in het paneel?
  \item Een groen foton van \qty{2.5}{eV} wordt geabsorbeerd: hoeveel van zijn
        energie overleeft als energie van een elektron-gatpaar, en waar
        gaat de rest heen, en hoe snel?
  \item Voeg de punten 3 en 4 samen tot het oordeel over het spectrum: ongeveer
        de helft van het invallende vermogen is weg voordat er ook maar elektronica
        begint. Noem de twee verliesroutes in telkens één zin.
  \item Waarom heeft een foton überhaupt $\ge E_{\text{g}}$ nodig --- wat
        verbiedt het om in gewoon silicium twee fotonen van een halve kloof
        snel na elkaar te absorberen?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De junctie als motor.}

\begin{enumerate}[resume]
  \item Elke cel is een \omterm{prop:b3:electrons-in-solids:junction}{p--n-junctie}. Leg in drie zinnen uit
        hoe het ingebouwde veld van de \omterm{prop:b3:electrons-in-solids:junction}{depletiezone} een gemaakt
        paar in uitwendige stroom omzet --- welke ladingsdrager gaat welke
        kant op, en waarom recombineren ze niet gewoon.
  \item Bereken met $N_{\text{a}} = N_{\text{d}} = \qty{1e22}{m^{-3}}$
        en $n_{\text{i}} = \qty{1e16}{m^{-3}}$ de waarde van
        $V_{\text{bi}}$ bij \qty{300}{K}.
  \item Een werkende cel levert ongeveer \qty{0.6}{V}. Zestig in
        serie: de bedrijfsspanning van het paneel?
  \item Leid uit het nominale vermogen van \qty{400}{W} de bedrijfsstroom
        af.
  \item Schat de fotonenflux (fotonen per seconde) die het paneel
        nuttig absorbeert, uitgaande van $\sim\qty{2}{eV}$ per geabsorbeerd
        foton, en vergelijk die met de elektronenflux die uw stroom
        impliceert: welke fractie van de geabsorbeerde fotonen levert een
        verzameld elektron op?
  \item Het nominale rendement: \qty{400}{W} uit
        \qty{2000}{W} invallend. Breng 20\,\% in overeenstemming met het
        ``de helft verloren vóór de elektronica'' van deel I: waar gaan de resterende
        dertig punten heen?
  \item Waarom levert de cel \qty{0.6}{V} en niet de volle
        \qty{1.12}{V} van de kloof? (Noem de boosdoener in de
        diodewet.)
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Echte daken.}

\begin{enumerate}[resume]
  \item Een wolkeloos wintermiddaguur op breedtegraad $50^\circ$ levert
        licht onder een hoogte van $30^\circ$. Bereken de meetkundige
        factor ten opzichte van loodrechte inval op een horizontaal
        paneel.
  \item Het gegevensblad van het paneel vermeldt $-0.4\,\%/\text{K}$ van de opbrengst:
        een zwart dakpaneel bereikt in de zomer \qty{65}{\celsius}.
        Hoeveel van het nominale vermogen overleeft?
  \item Verklaar het temperatuurverlies met
        \cref{exo:b3:electrons-in-solids:12}(d).
  \item Een schoorsteen beschaduwt één van de zestig cellen. Waarom kan dat
        de hele serieketen wurgen --- en wat doet de
        diodenatuur van de cel zelf met de arme beschaduwde cel?
  \item Fabrikanten schakelen een \emph{omloopdiode} over elke
        deelketen: leg haar taak in één zin uit.
  \item Gemiddeld over dagen en weer levert het dak
        \qty{15}{\percent} van het nominale vermogen. Schat de jaarlijkse
        energie van het paneel van \qty{400}{W} in kilowattuur.
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Het oordeel.}

\begin{enumerate}[resume]
  \item Wat verdient het paneel bij \qty{0.25}{euro} per kWh
        per jaar? Wat is bij aanschaf- en installatiekosten van \qty{300}{euro}
        de terugverdientijd?
  \item Het paneel kostte ruwweg \qty{500}{kWh} om te maken
        (silicium wordt door smelten gezuiverd): hoe lang tot het
        zijn eigen energie heeft terugbetaald?
  \item Je buurman vraagt waarom het paneel niet
        ``gewoon zwart gemaakt'' kan worden om het vijfde deel onder de kloof te vangen. Antwoord
        met de bandentheorie in twee zinnen.
  \item Vervolgens vraagt hij waarom er niet een tweede paneel met kleinere kloof
        onder wordt gestapeld: antwoord met
        \cref{exo:b3:electrons-in-solids:12}(c) --- en noem de
        praktische adder onder het gras.
  \item Dertig jaar later is het paneel vervaagd tot 85\,\% van zijn
        nominale vermogen. Welke microscopische boosdoeners uit dit hoofdstuk
        (denk aan wat $\tau$ begrenst en waar juncties bang voor zijn)
        laten een cel plausibel verouderen?
  \item Geef de samenvatting van de scheidsrechter in vier regels: de fysica van de kloof
        bepaalt de oogst, de junctie zet die om, de seriebedrading
        en de temperatuur belasten haar, en het grootboek --- energie
        en euro's --- sluit in het voordeel van het paneel.
\end{enumerate}
\end{problem}
