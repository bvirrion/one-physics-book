\chapter{Storingsrekening}\label{ch:b3:perturbation-theory}

Slechts een handvol kwantumvraagstukken lost exact op: de doos, de
oscillator, waterstof en het stelsel met twee niveaus. Al het overige ---
een atom in een veld, een molecuulbinding die verstijft, een niveau dat
door een buur wordt aangeduwd --- is een van die oplosbare skeletten met
een kleine extra term aan. De \omterm{thm:b3:perturbation-theory:corrections}{storingsrekening} is de kunst die extra term
te behandelen voor wat hij is: een correctie, orde voor orde in zijn
kleinheid berekend. Haar regel van eerste orde is een gemiddelde van één
regel; haar regel van tweede orde verklaart waarom niveaus elkaar
\emph{afstoten}, waarom elk atoom polariseerbaar is en waarom de
vanderwaalsaantrekking universeel is; haar ontaarde variant behandelt de
kiese gevallen waarin bijna gelijke niveaus zich volledig herordenen; en
haar tijdafhankelijke vorm --- de \omterm{thm:b3:perturbation-theory:golden}{gouden regel van Fermi} --- is de
vergelijking achter elke absorptielijn, elk vervaltempo en elke werkzame
doorsnede die ooit in een laboratorium is gemeten. Als beloning onderweg
berekent het hoofdstuk aan het slot de kleur van de hemel.

\section{Kleine verschuivingen: de niet-ontaarde theorie}

\begin{theorem}[Correcties uit de storingsrekening]\label{thm:b3:perturbation-theory:corrections}
\index{storingsrekening}
Zij $\hat H = \hat H_0 + \hat V$ met $\hat H_0$ opgelost ($\hat H_0\ket n = E_n\ket n$, met een niet-ontaard spectrum) en $\hat V$ klein. Dan
geldt, orde voor orde in $\hat V$,
\[
E_n' = E_n + \bra n\hat V\ket n
 + \sum_{m \neq n}\frac{|\bra m\hat V\ket n|^2}{E_n - E_m}
 + \cdots
\]
en krijgt de toestand bijmengingen $\ket{n'} = \ket n + \sum_{m\neq
n}\dfrac{\bra m\hat V\ket n}{E_n - E_m}\,\ket m + \cdots$
\emph{Eerste orde}: de verschuiving is gewoon het gemiddelde van de
storing in de ongestoorde toestand. \emph{Tweede orde}: koppelingen aan
andere niveaus duwen $E_n$ van hen \emph{weg} --- elke $m$ erboven duwt
omlaag, elke eronder omhoog --- zodat de verschuiving van tweede orde van
de grondtoestand altijd negatief is. Geldigheid: de mengfracties
$|\bra m\hat V\ket n/(E_n - E_m)|$ moeten klein zijn.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Ontwikkel $E' = E + \epsilon_1 + \epsilon_2 + \cdots$ en $\ket{\psi} =
\ket n + \ket{\psi_1} + \cdots$ in machten van $\hat V$, vul in
$(\hat H_0 + \hat V)\ket\psi = E'\ket\psi$ in en stel de ordes gelijk.
De vergelijking van eerste orde op $\bra n$ projecteren geeft $\epsilon_1 =
\bra n\hat V\ket n$; op $\bra m$ ($m \neq n$) projecteren geeft de
mengcoëfficiënten; die in de vergelijking van tweede orde meenemen en
opnieuw op $\bra n$ projecteren geeft $\epsilon_2$. De uitspraak over het
teken voor de grondtoestand: elke noemer $E_0 - E_m$ is negatief terwijl
de tellers kwadraten zijn.
\end{proof}

\begin{proposition}[Een storing met een exact antwoord]\label{prop:b3:perturbation-theory:oscillator-field}
Een oscillator met lading $q$ in een uniform veld $\mathcal E$ heeft
$\hat V = -q\mathcal E\hat x$. Kwadraatafsplitsing lost hem exact op: de
potentiaal is dezelfde parabool, verschoven, met alle niveaus verlaagd met
$q^2\mathcal E^2/2m\omega^2$. De \omterm{thm:b3:perturbation-theory:corrections}{storingsrekening} moet daarmee
overeenstemmen --- en doet dat: de eerste orde verdwijnt ($\langle\hat
x\rangle_n = 0$), en de som van tweede orde, waarin $\hat x$ de $n$ alleen
aan $n \pm 1$ koppelt, geeft voor elk niveau precies
$-q^2\mathcal E^2/2m\omega^2$
(\cref{exo:b3:perturbation-theory:2}). Eén oplosbaar geval, langs beide
wegen nagegaan, is de beste ijking die een methode kan krijgen.
\end{proposition}

\begin{proof}
De twee matrixelementen en hun noemers geven
\[
\frac{|\bra{n+1}\hat x\ket{n}|^2}{-\hbar\omega}
 + \frac{|\bra{n-1}\hat x\ket n|^2}{+\hbar\omega}
 = \frac{\hbar}{2m\omega}\,\frac{-(n+1) + n}{\hbar\omega}
 = -\frac{1}{2m\omega^2} ;
\]
vermenigvuldig met $q^2\mathcal E^2$.
\end{proof}

\section{Wanneer niveaus ontaard zijn}

\begin{theorem}[Storingsrekening bij ontaarding]\label{thm:b3:perturbation-theory:degenerate}
\index{storingsrekening bij ontaarding}
Is $E_n$ ontaard, dan deelt de bovenstaande formule door nul --- de
storing mag de ontaarde toestanden volledig herordenen, en de remedie is
haar dat te laten doen: \emph{diagonaliseer $\hat V$ beperkt tot de
ontaarde deelruimte}. Haar \omterm{def:b3:quantum-formalism:observable}{eigenwaarden} zijn de verschuivingen van eerste
orde; haar eigenvectoren zijn de juiste toestanden van nulde orde, de
combinaties die de storing zelf uitkiest.
\end{theorem}
\admitted

\begin{example}[Het lineaire starkeffect]\label{ex:b3:perturbation-theory:stark}
\index{starkeffect}
Het niveau $n = 2$ van waterstof bevat de ontaarde toestanden $2s$ en
$2p$. In een veld $\mathcal E\vect e_z$ koppelt $\hat V = e\mathcal E\hat z$ de $2s$ aan de $2p_0$ met $\bra{2s}\hat z\ket{2p_0} = -3a_0$: het
diagonaliseren van het $2\times2$-blok geeft de verschuivingen
\[
\Delta E = \pm\,3e a_0\mathcal E ,
\]
\emph{lineair} in het veld --- terwijl de niet-ontaarde grondtoestand
alleen kwadratisch verschuift. Bij $\mathcal E = \qty{e7}{V/m}$: $\pm
\qty{1.6}{meV}$, een moeiteloos op te lossen splitsing. De toestanden die
de splitsing doen zijn $(\ket{2s} \mp \ket{2p_0})/\sqrt2$ --- hybriden
met een permanent elektrisch dipoolmoment, alleen mogelijk omdat de
ontaarding het atoom in nulde orde vrijliet om te polariseren. (Dezelfde
hybridisering van $s$ en $p$, aangedreven door buren in plaats van door
een veld, is de koolstofchemie van elk organisch molecuul.)
\end{example}

\section{Overgangen: de gouden regel}

\begin{theorem}[Tijdafhankelijke storingen]\label{thm:b3:perturbation-theory:golden}
\index{gouden regel van Fermi}
Schakel $\hat V(t) = \hat V\cos\omega t$ in op $t = 0$. In eerste orde is
de kans om het stelsel in $\ket f$ aan te treffen op tijdstip $t$, na
vanuit $\ket i$ te zijn gestart,
\[
\mathcal P_{i\to f}(t) = \frac{|\bra f\hat V\ket i|^2}{4\hbar^2}\;
\frac{\sin^2\!\big[(\omega_{fi} - \omega)\,t/2\big]}
{\big[(\omega_{fi} - \omega)/2\big]^2} ,
\qquad \omega_{fi} = \frac{E_f - E_i}{\hbar} :
\]
een resonantie scherp gepiekt op $\hbar\omega = E_f - E_i$ ---
absorptie wanneer $E_f > E_i$, \emph{gestimuleerde emissie} voor de
begeleidende term met $E_f < E_i$. Vormen de eindtoestanden een continuüm
met dichtheid $\rho(E)$, dan wordt de groei van de piek een constant
\emph{tempo}:
\[
\Gamma_{i\to f} = \frac{2\pi}{\hbar}\,
|\bra f\hat V\ket i|^2\,\rho(E_f)
\]
--- de \emph{\omterm{thm:b3:perturbation-theory:golden}{gouden regel van Fermi}}, de moederformule van
vervaltempo's, absorptiecoëfficiënten en werkzame doorsneden.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Amplitude in eerste orde: $c_f(t) = -\tfrac{\iu}{\hbar}\int_0^t\bra
f\hat V(t')\ket i\,\eu^{\iu\omega_{fi}t'}\dd t'$; de bijna-resonante
exponentiële behouden en integreren geeft de aangegeven vorm in
$\sin^2$. Voor het continuüm: als functie van de ontstemming $x$ heeft de
resonantiefactor hoogte $t^2$ en breedte $\sim 2\pi/t$ --- oppervlakte
$2\pi t$ --- zodat sommeren over een glad continuüm haar door
$2\pi t\,\rho$ vervangt: de kans is lineair in $t$, dus een tempo.
\end{proof}

\begin{example}[Wat de gouden regel aandrijft]\label{ex:b3:perturbation-theory:applications}
Selectieregels: een overgang vindt alleen plaats als het matrixelement
$\bra f\hat V\ket i$ overleeft --- voor licht is $\hat V \propto \hat
z$,
en diens pariteit en impulsmoment doden alles behalve $\Delta l = \pm1$,
$\Delta m = 0, \pm1$: de regels die sinds
\cref{ch:b3:quantum-angular-momentum} zijn ingeroepen, nu stellingen.
Tempo's: het spontane verval van de toestand $2p$ van waterstof rekent uit
op \qty{1.6}{ns} (\cref{exo:b3:perturbation-theory:8}); de maserovergang
van ammoniak, met haar $\omega^3$ op microgolven, duurt maanden --- de
$\omega^3$ in stralingstempo's is de reden dat radiolijnen
(\cref{ex:b3:spin-two-level:hyperfine}) miljoenen jaren leven terwijl
ultraviolette lijnen in nanoseconden opflitsen. Absorptiespectra,
foto-ionisatie, het bètaverval van kernen
(\cref{ch:b3:nuclear-physics}) en werkzame doorsneden bij verstrooiing
(\cref{ch:b3:scattering-theory}): alle zijn deze ene formule met andere
matrixelementen en andere toestandstellingen.
\end{example}

\begin{method}[Het juiste gereedschap uit de storingsrekening kiezen]\label{met:b3:perturbation-theory:method}
(1) Statische vraag, niet-ontaard niveau: middel $\hat V$ (eerste orde);
verdwijnt dat door symmetrie, dan de tweede orde --- verwacht
niveauafstoting. (2) Ontaard niveau: diagonaliseer $\hat V$ eerst in het
ontaarde blok; de aan de symmetrie aangepaste combinaties doen de
splitsing. (3) Overgangstempo's: de gouden regel --- één matrixelement,
één \omterm{prop:b3:schrodinger-three-dimensions:counting}{toestandsdichtheid}; toets de selectieregels voordat u iets berekent.
(4) Toets de ontwikkeling altijd: de mengfractie $|V_{mn}/(E_n - E_m)| \ll 1$, anders exact diagonaliseren (de formule met twee niveaus). (5)
Exact oplosbare grensgevallen (de oscillator in een veld) zijn gratis
ijkingen: gebruik ze.
\end{method}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:1}
Een deeltje in een doos $[0, a]$ ondervindt de extra potentiaal $V(x) =
\lambda x/a$. (a) Bereken de verschuiving van eerste orde van elk niveau
(gebruik $\langle x\rangle = a/2$ in elke doostoestand --- rechtvaardig
dat). (b) Waarom is de verschuiving hier voor alle niveaus dezelfde? (c)
Wat is de verandering van eerste orde van de \emph{afstanden}, en waarom
zou een meting van overgangsfrequenties deze storing volledig kunnen
missen? (d) Wanneer houdt de eerste orde op betrouwbaar te zijn?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:2}
Voer de berekening van tweede orde uit
\cref{prop:b3:perturbation-theory:oscillator-field} volledig uit: de
matrixelementen van $\hat x$, de twee termen, het wegvallen van $n$, en de
exacte overeenstemming met de kwadraatafsplitsing. Waarom verdwijnt de
correctie van eerste orde op de \emph{toestand} niet, terwijl die op de
energie dat wel doet?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:3}
De \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniaan} met twee niveaus $\begin{pmatrix}E_1 & v\\ v &
E_2\end{pmatrix}$, met $E_1 < E_2$ en reële $v$. (a) Bepaal de exacte
\omterm{def:b3:quantum-formalism:observable}{eigenwaarden}. (b) Ontwikkel voor $|v| \ll E_2 - E_1$ en herken de formule
van tweede orde. (c) Toon aan dat de niveaus elkaar afstoten: de kloof
overtreft altijd $E_2 - E_1$. (d) Bij $E_1 = E_2$: wat geeft de
\omterm{thm:b3:perturbation-theory:corrections}{storingsrekening}, wat geeft het exacte antwoord, en welke stelling uit het
hoofdstuk verzoent ze?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:4}
Voor \cref{thm:b3:perturbation-theory:corrections}: (a) toon aan dat de
correctie van eerste orde op de toestand loodrecht op $\ket n$ staat; (b)
toon aan dat de mengcoëfficiënt van $\ket m$ gelijk is aan $V_{mn}/(E_n - E_m)$ en formuleer het geldigheidscriterium; (c) een niveau dat
$\qty{1}{eV}$ van zijn naaste buur ligt is er met \qty{0.1}{eV} aan
gekoppeld: schat de bijmengkans; (d) hetzelfde met een kloof van
\qty{10}{meV} --- welk gereedschap moet de ontwikkeling dan vervangen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:5}
Anharmonische bindingen. Voeg $\hat V = \beta\hat x^3 + \gamma\hat x^4$
aan de oscillator toe. (a) Toon aan dat de term in $x^3$ in eerste orde
geen enkel niveau verschuift (pariteit). (b) Toon aan dat de term in $x^4$
$\Delta E_n =
3\gamma(\hbar/2m\omega)^2(2n^2 + 2n + 1)$ geeft (ontwikkel
$\hat x^4$ in \omterm{def:b3:harmonic-oscillator:ladder}{ladderoperatoren}; alleen de ``uitgebalanceerde'' termen
overleven). (c) Voor een echte binding is de effectieve $\gamma$ negatief (de put wordt naar
buiten toe zachter): toon aan dat de niveauafstand dan met $n$
\emph{krimpt}.
(d) Welk waargenomen kenmerk van moleculaire spectra
(\cref{exo:b3:harmonic-oscillator:4}) geeft dit weer?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:6}
In de vierkante tweedimensionale doos wordt het ontaarde paar $\ket{1,2},
\ket{2,1}$ gestoord door $\hat V = \lambda\,xy$. (a) Leg uit waarom de
niet-ontaarde theorie faalt. (b) Bereken de $2\times2$-matrix van $\hat V$
in het paar (de integralen splitsen in factoren; $\int_0^a
x\sin(\pi x/a)\sin(2\pi x/a)\dd x = -8a^2/9\pi^2$). (c) Bepaal de
splitsingen en de juiste toestanden van nulde orde. (d) Vergelijk met de
symmetrieanalyse van \cref{exo:b3:quantum-formalism:9}: welk antwoord gaf
de symmetrie gratis?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:7}
\omterm{ex:b3:perturbation-theory:stark}{Starkeffecten}, lineair en kwadratisch. (a) Bereken met
\cref{ex:b3:perturbation-theory:stark} de splitsing van $n = 2$ bij
$\mathcal E = \qty{2.5e6}{V/m}$. (b) Waarom vertoont de grondtoestand
\emph{geen} lineaire verschuiving (twee redenen: pariteit, en geen
ontaarde partner)? (c) Haar kwadratische verschuiving definieert de
polariseerbaarheid, $\Delta E = -\tfrac12\alpha\mathcal E^2$: schat
$\alpha \sim e^2a_0^2/E_{\text{I}}$ en reken haar uit in eenheden
$a_0^3$ (het exacte antwoord is $4.5\,a_0^3 \times
4\pi\varepsilon_0$). (d) Van welke alledaagse eigenschap van materie ---
haar diëlektrische respons --- hebt u zojuist de atomaire kiem berekend?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:8}
Spontane emissie (met één aangenomen gegeven): een aangeslagen toestand
vervalt met $A = \omega^3|d_{fi}|^2/3\pi\varepsilon_0\hbar c^3$, waarin
matrixelementen van het type $d_{fi} = e\bra f\hat z\ket i$ het
dipoolmoment bepalen. (a) Voor waterstof $2p \to 1s$: bereken met $|d| \approx 0.74\,
ea_0$ en $\lambda = \qty{121.6}{nm}$ de waarde van $A$ en de
levensduur. (b) Schaal naar de D-lijn van natrium. (c) Schaal naar de
hyperfijnovergang op 21 cm ($|d| \to$ een magnetisch moment, met een
tempo dat nog eens met een factor van de orde $\alpha^2$ omlaag gaat; neem
$A \approx
\qty{2.9e-15}{s^{-1}}$ aan): ga de ``tien miljoen jaar'' na.
(d) Uit de $\omega^3$: waarom zijn ultraviolette lijnen snel en radiolijnen
eeuwig, en waarom maakte dat de \omterm{ex:b3:spin-two-level:hyperfine}{21 cm-lijn} \emph{voorspelbaar} maar in
een laboratorium vrijwel niet waarneembaar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:9}
Selectieregels als integralen. (a) Toon met pariteit aan dat
$\bra{100}\hat z\ket{200} =
0$. (b) Toon aan dat $\bra{100}\hat z\ket{21m} = 0$ tenzij $m =

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:10}
Polariseerbaarheid, eerlijk. De verschuiving van tweede orde van de
grondtoestand van waterstof in een veld is $\Delta E = -e^2\mathcal E^2\sum_{m\neq
0}|\bra m\hat z\ket 0|^2/(E_m - E_0)$. (a) Begrens de som
door elke noemer door de kleinste kloof $E_2 - E_1 =
\tfrac34 E_{\text{I}}$ te vervangen en de volledigheidsbetrekking $\sum_m|\bra
m\hat z\ket0|^2 = \langle z^2\rangle_0 = a_0^2$ te gebruiken: verkrijg
$\alpha
\le \tfrac{16}3\,a_0^3$ (in eenheden $4\pi\varepsilon_0$ naar
gaussiaans model). (b) Vergelijk met de exacte $\tfrac92 a_0^3$. (c) Schat
uit $\alpha$ de brekingsindex van waterstofgas bij atmosferische dichtheid
($n - 1 \approx N\alpha/2\varepsilon_0
\cdot e^2\dots$ --- gebruik $n - 1 = N\alpha_{\text{SI}}/2
\varepsilon_0$) en vergelijk met de gemeten
$\num{1.3e-4}$. (d) Formuleer in één zin de keten atoom $\to$
polariseerbaarheid $\to$ breking waarop deel II van het weekendvraagstuk
zal meeliften.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:11}
Vermeden kruisingen en trage sweeps. Een stelsel met twee niveaus heeft
kale energieën $\pm\lambda t$ die op $t = 0$ kruisen, gekoppeld door $v$.
(a) Schets de exacte niveaus tegen $t$: een vermeden kruising met kloof
$2|v|$. (b) Is de sweep \emph{traag}, dan volgt het stelsel de onderste
kromme (adiabatische stelling, aangenomen): in welke toestand eindigt het?
(c) Het criterium van Landau en Zener vergelijkt de tijd om de klofzone te
doorlopen, $\sim v/\lambda$, met de inwendige klok $\hbar/v$: geef de
voorwaarde voor adiabatisch volgen. (d) Twee toepassingen: de toestand van
een qubit die door een trage chirp wordt overgedragen; een atomaire
botsing die lading tussen kernen laat hoppen --- leg er één in twee zinnen
uit.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:12}
De tweede orde is een pessimist, en andere stellingen. (a) Bewijs dat de
correctie van tweede orde op de \emph{grond}toestand altijd negatief is.
(b) Concludeer (met \cref{exo:b3:harmonic-oscillator:10}) waarom
vanderwaalskrachten tussen atomen in hun grondtoestand altijd aantrekkend
zijn. (c) Toon aan dat de theorie van eerste orde elke grondtoestandsenergie
\emph{overschat}: het is een variatiegrens (evalueer $\bra{\psi_0}\hat
H\ket{\psi_0}$ met de ongestoorde toestand als proeffunctie). (d) Een
consistentietoets op \cref{ex:b3:perturbation-theory:stark}: waarom is het
\emph{lineaire} \omterm{ex:b3:perturbation-theory:stark}{starkeffect} geen schending van (a)?
\end{exercise}

\section{Vraagstuk: De kleur van de hemel}

\begin{problem}[{Weekendvraagstuk --- rayleighverstrooiing aan gestoorde
atomen}]\label{pb:b3:perturbation-theory:1}
Waarom is de daghemel blauw, de zonsondergang rood en de wolk wit? Het
volledige antwoord is een keten door dit hoofdstuk: een lichtgolf stoort
elk luchtmolecuul; het geïnduceerde dipoolmoment straalt opnieuw uit (de
dipoolstraling van het volume van jaar 2); en de interferentie beslist wat
er overleeft. Gegevens: de moleculaire polariseerbaarheid van lucht
$\alpha/4\pi\varepsilon_0 \approx \qty{2.2e-30}{m^3}$; aantalsdichtheid
$N = \qty{2.5e25}{m^{-3}}$; zichtbaar bereik
$450$--\qty{700}{nm}.

\medskip\noindent\textbf{Deel I --- Het gestoorde molecuul.}
\begin{enumerate}
  \item Een statisch veld $\mathcal E$ verschuift de grondtoestand van
        een molecuul met $-\tfrac12\alpha\mathcal E^2$: breng deze
        uitspraak in verband met de \omterm{thm:b3:perturbation-theory:corrections}{storingsrekening} van tweede orde
        (welke formule, en waarom negatief?).
  \item Dezelfde $\alpha$ geeft het geïnduceerde dipoolmoment $p =
        \alpha\mathcal E$: bereken voor het veld van het zonlicht,
        $\mathcal E \sim
        \qty{800}{V/m}$, het geïnduceerde dipoolmoment
        van één stikstofmolecuul en vergelijk het met $ea_0$.
  \item Waarom mag het optische veld (frequentie $\num{5e14}$ Hz) met de
        \emph{statische} polariseerbaarheid worden behandeld voor
        moleculen waarvan de elektronische overgangen in het ultraviolet
        liggen? (Vergelijk $\hbar\omega$ met de kloof; dit is de limiet
        ver van resonantie van de noemer in de gouden regel.)
  \item Een trillend dipoolmoment $p_0\cos\omega t$ straalt het
        gemiddelde vermogen $P = p_0^2\omega^4/12\pi\varepsilon_0c^3$
        uit (volume van jaar 2). Hoe hangt het verstrooide vermogen,
        samen met $p_0 = \alpha\mathcal
        E_0$, bij gelijke belichting van
        $\omega$ af?
  \item Bereken de verhouding van de verstrooide vermogens bij
        \qty{450}{nm} en \qty{700}{nm}.
  \item In één zin: waarom is de hemel blauw en niet violet (twee
        ingrediënten: het spectrum van de zon dat naar het violet toe
        afneemt, en de gevoeligheid van het oog)?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Van één molecuul naar de
hemel.}
\begin{enumerate}[resume]
  \item De werkzame doorsnede voor verstrooiing komt uit op $\sigma =
        \dfrac{8\pi}{3}\Big(\dfrac{\alpha}{4\pi\varepsilon_0}
        \Big)^2\dfrac{\omega^4}{c^4}$: ga haar dimensies na en reken
        haar uit bij \qty{550}{nm}.
  \item De verzwakkingslengte is $\ell = 1/N\sigma$: reken haar uit bij
        \qty{550}{nm} en vergelijk met de dikte van de atmosfeer
        ($\sim\qty{8}{km}$ equivalent bij de dichtheid op zeeniveau).
  \item Welk deel van een verticale zonnestraal wordt bij \qty{450}{nm}
        weggestrooid, en welk bij \qty{700}{nm}? (Gebruik $1 -
        \eu^{-L/\ell}$ met $L = \qty{8}{km}$.)
  \item Bij zonsondergang is de weg dertig keer langer: bereken de
        overleving van het blauw opnieuw en verklaar de kleur van de lage
        zon --- en van het licht dat de maan bij een maansverduistering
        rood kleurt.
  \item Waarom is het licht van de hemel op $90^\circ$ van de zon
        \emph{gepolariseerd} (denk aan het stralingspatroon van een
        dipool: geen emissie langs de as van de dipool)?
  \item Bijen en Vikingen navigeerden naar verluidt op deze polarisatie:
        wat doet een bewolkte dag ermee, en waarom?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Waarom de hemel niet helderder
is.}
\begin{enumerate}[resume]
  \item Een paradox: in een volmaakt uniform medium interfereren de
        golfjes van naburige volume-elementen zijwaarts destructief, en
        zou er \emph{geen} licht worden verstrooid (alleen de breking
        overleeft). Wat doorbreekt de uniformiteit van lucht in
        werkelijkheid --- wat is er willekeurig verdeeld?
  \item Dichtheidsfluctuaties van een ideaal gas maken dat de
        verstrooide intensiteiten van onafhankelijke moleculen
        \emph{optellen}: zeg waarom die onafhankelijkheid (willekeurige
        posities, afstanden op de schaal van de golflengte) de
        destructieve interferentie doodt.
  \item Welk deel van het zonlicht verstrooit één atmosfeer op een
        heldere dag, volgens uw $\ell$ bij \qty{550}{nm} --- komt het
        getal overeen met de alledaagse helderheid van de hemel naast de
        zon?
  \item Een wolkendruppel (\qty{10}{\micro m}) bevat $\sim10^{12}$
        moleculen binnen één golflengte: hun golfjes tellen
        \emph{coherent} op. Hoe schaalt het verstrooide vermogen dan met
        het aantal moleculen, en waarom verdwijnt de kleurselectie door
        $\omega^4$ (alle golflengten verstrooien sterk): welke kleur
        heeft de wolk?
  \item Melk, mist en witte verf zijn om dezelfde reden wit: formuleer
        de algemene regel --- verstrooiers die klein zijn tegen $\lambda$
        kleuren het licht, verstrooiers die groot zijn tegen $\lambda$
        maken het wit.
  \item Kritische opalescentie: nabij het kritische punt van een
        vloeistof (de hoofdstukken over faseovergangen van het volume van
        jaar 1, en \cref{ch:b3:phase-transitions} verderop) groeien de dichtheidsfluctuaties tot optische
        schalen en wordt de vloeistof melkachtig: leg het verband met punt
        13.
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- De keten, voltooid.}
\begin{enumerate}[resume]
  \item De voorwaarts verstrooide golfjes heffen elkaar \emph{niet} op:
        zij interfereren met de bundel en vertragen haar fase --- de
        brekingsindex. Reken met $n - 1 = N\alpha/2\varepsilon_0$ (met
        $\alpha$ in SI) de waarde $n - 1$ voor lucht uit en vergelijk met
        de gemeten \num{2.9e-4}.
  \item Eén constante $\alpha$, door de \omterm{thm:b3:perturbation-theory:corrections}{storingsrekening} van tweede orde
        berekend, legt zo drie verschijnselen tegelijk vast: noem ze (de
        verschuiving in een statisch veld; het blauw van de hemel; het
        buigen van licht in lucht).
  \item Ozonabsorptie, aerosolen en meervoudige verstrooiing maken echte
        hemels ingewikkelder: noem bij elk één waarneembaar verschijnsel
        (de kleuren van de schemering; een melkachtige horizon; de
        resterende helderheid van de hemel in het zenit na zonsondergang).
  \item Waarom heeft de maan, zonder lucht, een \emph{zwarte} daghemel
        --- en wat bevestigde elke foto van haar oppervlak daarmee over de
        oorsprong van de onze?
  \item De daghemel van Mars is karamelkleurig en haar zonsondergangen
        zijn \emph{blauw}: haar ijle lucht verstrooit weinig, en zwevende
        stofkorrels van een micrometer doen in plaats daarvan het werk.
        Leg met de regel uit punt 18 uit hoe stof de kleurlogica omkeert.
  \item Voor röntgenstraling overtreft $\hbar\omega$ elke moleculaire
        overgang ruimschoots: de elektronen reageren alsof zij vrij zijn
        en de verstrooiing (Thomson) verliest haar $\omega^4$. Wat gebeurt
        er in dat regime met het mechanisme van het ``hemelsblauw'' --- en
        waarom is die vlakke respons precies wat röntgenstralen tot
        zuivere sondes van de elektronendichtheid maakt?
  \item Vat het genoemde resultaat samen: een polariseerbaarheid van
        $\qty{2.2e-30}{m^3}$, gekwadrateerd en met $\omega^4$
        vermenigvuldigd, geeft in het groen een verstrooiingslengte op de
        schaal van \qty{60}{km} --- lang genoeg opdat de middaghemel
        zachtaardig is, kort genoeg opdat zonsondergangen roden en het
        firmament blauw is, met het polarisatiepatroon van een woud
        aangedreven dipolen.
\end{enumerate}
\end{problem}
