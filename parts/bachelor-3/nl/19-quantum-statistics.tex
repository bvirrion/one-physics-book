\chapter{Kwantumstatistiek}\label{ch:b3:quantum-statistics}

De geleidingselektronen van koper vormen een gas dat duizend keer dichter
is dan lucht --- en toch dragen zij vrijwel niets bij aan de
warmtecapaciteit van koper, een schandaal dat de klassieke natuurkunde
vijftig jaar achtervolgde. Een wolk rubidiumatomen, tot honderd
miljardsten van een kelvin gekoeld, stort plotseling in tot één enkele
kwantumtoestand van tienduizend atomen diep. Een dode ster met de massa
van de zon en de afmeting van de aarde weigert verder te krimpen,
overeind gehouden door niets dan het uitsluitingsbeginsel. Alle drie zijn
hetzelfde hoofdstuk natuurkunde: wat er met een gas gebeurt wanneer
$n\lambda_T^3 \sim 1$ en de bezettingsformules van
\cref{ch:b3:grand-canonical} van Boltzmann afwijken. Fermionen bouwen
zeeën --- koud, star, onder druk; bosonen bouwen condensaten ---
gezellig, coherent, supervloeibaar. Tussen de zwijgende meerderheid van de
fermizee en de stormloop van het condensaat ligt het grootste deel van de
moderne fysica van de gecondenseerde materie en van de lage temperaturen,
en het lot van de sterren.

\section{Gassen van bezettingen}

\begin{proposition}[Het raderwerk van het kwantumgas]\label{prop:b3:quantum-statistics:machinery}
\index{toestandsdichtheid}
Voor niet-wisselwerkende deeltjes in een doos worden sommen over niveaus
integralen over de \omterm{prop:b3:schrodinger-three-dimensions:counting}{toestandsdichtheid} van
\cref{prop:b3:schrodinger-three-dimensions:counting} (per spintoestand
$g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$):
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
met $\bar n$ de bezetting van Fermi--Dirac of Bose--Einstein. De eerste
vergelijking legt $\mu(T, n)$ vast; de tweede levert de thermodynamica. In
de ijle limiet $n\lambda_T^3 \ll 1$ vallen beide statistieken op Boltzmann
samen en keert het klassieke gas terug; de kwantumregimes beginnen waar de
golfpakketten elkaar raken.
\end{proposition}
\admitted

\section{De fermizee}

\begin{theorem}[Het ontaarde fermigas]\label{thm:b3:quantum-statistics:fermi}
\index{fermi-energie}\index{ontaardingsdruk}
Bij $T = 0$ vullen fermionen elk niveau tot aan de \emph{\omterm{thm:b3:quantum-statistics:fermi}{fermi-energie}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{spin } \tfrac12) ,
\]
met gemiddelde energie $\tfrac35 E_{\text{F}}$ per deeltje en de
\emph{\omterm{thm:b3:quantum-statistics:fermi}{ontaardingsdruk}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
een druk bij het absolute nulpunt, van zuiver kwantummechanische
oorsprong --- de stijfheid van het elektronengas van elk metaal en de
steun van dode sterren. Bij $0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$ kunnen alleen de elektronen binnen $\sim k_{\text{B}}T$ van het oppervlak van deze \emph{fermizee} ergens op
reageren: de fractie $\sim T/T_{\text{F}}$ is de hoofdsleutel tot het
gedrag van metalen.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Vul $g(\epsilon)$ tot aan $E_{\text{F}}$: het omkeren van $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ geeft de aangegeven $E_{\text{F}}$
(\cref{exo:b3:schrodinger-three-dimensions:8}); $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ volgt uit nog een integraal. Druk: de doos
samendrukken tilt elk niveau als $V^{-2/3}$ op, dus $E
\propto V^{-2/3}$
en $P = -\partial E/\partial V = \tfrac23 E/V$ --- met $E = \tfrac35 NE_{\text{F}}$ geeft dat de aangegeven wet.
\end{proof}

\begin{example}[Het warmtecapaciteitsschandaal, opgelost]\label{ex:b3:quantum-statistics:heat}
\index{elektronische warmtecapaciteit}
Klassiek zou het vrije elektron per atoom van koper $\tfrac32
k_{\text{B}}$ aan de warmtecapaciteit moeten toevoegen --- $50\%$ boven op
Dulong--Petit. Gemeten: ongeveer $1\%$ bij kamertemperatuur. De zee
verklaart dat: met $T_{\text{F}} \approx \qty{8e4}{K}$ kan alleen de
fractie $\sim T/T_{\text{F}} \approx 0.4\%$ van de elektronen thermisch
worden aangeslagen, elk met $\sim k_{\text{B}}T$:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(de exacte coëfficiënt van Sommerfeld;
\cref{exo:b3:quantum-statistics:12}). Lineair in $T$ ligt zij bij gewone
temperaturen onder de $T^3$ van het rooster begraven, maar
\emph{overheerst} zij onder enkele kelvin --- precies zoals calorimeters
vinden, waarmee in één klap het raadsel is opgelost van de elektronen die
wel stroom dragen maar geen warmtecapaciteit.
\end{example}

\begin{example}[Sterren die door uitsluiting overeind blijven]\label{ex:b3:quantum-statistics:dwarf}
\index{witte dwerg}
Een \omterm{ex:b3:quantum-statistics:dwarf}{witte dwerg} (\cref{exo:b3:identical-particles:11}) houdt de
zwaartekracht in evenwicht met de \omterm{thm:b3:quantum-statistics:fermi}{ontaardingsdruk} van de elektronen. Omdat
$P \propto
n^{5/3}$ sneller verstijft dan de zwaartekracht eist naarmate
de ster krimpt, bestaat er bij elke bescheiden massa een evenwichtsstraal
--- met de griezelige schaling $R \propto M^{-1/3}$: \emph{zwaardere
dwergen zijn kleiner}. Harder samengeperst worden de elektronen
relativistisch, verzacht de drukwet tot $n^{4/3}$ en faalt het evenwicht
bij één unieke massa, de grens van Chandrasekhar $M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) --- daarvoorbij volgt
ineenstorting tot een neutronenster: een fermizee van neutronen bij
kerndichtheid, dezelfde natuurkunde bij $10^{14}$ keer de dichtheid.
\end{example}

\section{De stormloop van de bosonen}

\begin{theorem}[Bose-einsteincondensatie]\label{thm:b3:quantum-statistics:bec}
\index{bose-einsteincondensatie}
Voor een gas van $N$ behouden bosonen kunnen de aangeslagen toestanden
hoogstens $N_{\max}(T) = 2.612\,V/\lambda_T^3$ deeltjes bevatten (de
integraal van $g\bar n_{\text{BE}}$ op haar plafond $\mu \to 0$). Onder de
kritische temperatuur
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(equivalent } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
kan het overschot nergens heen dan naar de ene grondtoestand: een
\emph{macroscopische} bezetting
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
condenseert in één kwantumtoestand --- materie die zich als één
reusachtige golffunctie gedraagt. In 1925 door Einstein voorspeld uit de
fotonstatistiek van Bose; in 1995 in ijl rubidium verwezenlijkt bij
$T_{\text{c}} \approx \qty{100}{nK}$ (Nobelprijs 2001), aangekondigd door
de veelzeggende \emph{bimodale} snelheidsverdeling: een thermische wolk
met een naald condensaat zonder snelheid die uit haar midden oprijst.
\end{theorem}
\admitted

\begin{example}[Helium, de veteraan onder de supervloeistoffen]\label{ex:b3:quantum-statistics:helium}
\index{supervloeibaarheid}
Vloeibaar $^4$He (bosonisch) gaat bij $T_\lambda = \qty{2.17}{K}$ over in
een \emph{supervloeistof}: geen viscositeit door de fijnste haarvaten,
filmstroming over wanden van vaten, fonteineffecten, en circulatie
gekwantiseerd in eenheden $h/m$ --- de macroscopische fase van een
condensaat aan het werk, door de wisselwerkingen verschoven ten opzichte
van de voorspelling voor een ideaal gas ($\qty{3.1}{K}$ bij de dichtheid
van helium) maar onmiskenbaar hetzelfde verschijnsel. Fermionisch $^3$He
kan niet in zijn eentje condenseren: zijn atomen moeten eerst \emph{paren}
(zoals de elektronen van een supergeleider,
\cref{ch:b3:electrons-in-solids}) tot samengestelde bosonen --- en de
supervloeibaarheid komt dan ook, maar bij \qty{2.6}{mK}, duizend keer
kouder: twee isotopen, één les --- de statistiek is het lot.
\end{example}

\begin{method}[Welke statistiek, welk regime]\label{met:b3:quantum-statistics:method}
(1) Bereken eerst $n\lambda_T^3$: $\ll1$ betekent Boltzmann, en hoofdstuk
17 volstaat. (2) Ontaarde fermionen: denk in $E_{\text{F}}$ en
$T/T_{\text{F}}$; responsen (warmtecapaciteit, susceptibiliteit,
verstrooiing) dragen de factor $T/T_{\text{F}}$ of bemonsteren alleen het
fermioppervlak. (3) Ontaarde bosonen: vergelijk $T$ met $T_{\text{c}}$;
daaronder splitst u het gas in een condensaat plus een thermische wolk.
(4) Drukken: de fermionische stijfheid $n^{5/3}$ (of $n^{4/3}$
relativistisch) tegen wat er ook knijpt. (5) Samengestelde deeltjes: tel de
fermionen waaruit ze bestaan om de statistiek te kiezen --- en onthoud dat
paarvorming de ene statistiek in de andere kan omzetten.
\end{method}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:1}
Bereken $E_{\text{F}}$ en $T_{\text{F}}$ voor (a) koper ($n = \qty{8.5e28}{m^{-3}}$); (b) natrium ($\qty{2.5e28}{m^{-3}}$); (c)
vloeibaar $^3$He ($\qty{1.6e28}{m^{-3}}$, $m = 3u$); (d) rangschik ze ten

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:2}
(a) Toon aan dat $\langle\epsilon\rangle = \tfrac35 E_{\text{F}}$ bij $T =
0$. (b) Bereken de effectieve snelheid van de elektronen in koper. (c) Met
welke temperatuur komt die snelheid klassiek overeen? (d) Waarom bevat een
koud metaal elektronen die met een honderdste van de lichtsnelheid bewegen
zonder die energie weg te stralen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:3}
De \omterm{thm:b3:quantum-statistics:fermi}{ontaardingsdruk} van de elektronen in koper: (a) reken $P =
\tfrac25 nE_{\text{F}}$ uit in pascal en in atmosfeer. (b) Waarom ontploft het
metaal niet (wat trekt terug)? (c) Deze druk weerstaat samendrukking:
breng dat in verband met de lage samendrukbaarheid van metalen. (d) Schat
de compressiemodulus $K \sim \tfrac53 P$ en vergelijk met de gemeten
\qty{140}{GPa} van koper.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:4}

millikelvins (wat gebeurt er scheikundig bij hoge dichtheid en lage

= 2.612$ op
uw getallen voor rubidium.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:5}
Twee warmtecapaciteiten. In een metaal is $C = \gamma T +
\beta T^3$
(elektronen; rooster). (a) Schat met
\cref{ex:b3:quantum-statistics:heat} en de debye-achtige roosterterm
$\beta T^3 = 234\,Nk_{\text{B}}(T/\theta_{\text{D}})^3$ de
omslagtemperatuur voor koper ($\theta_{\text{D}} = \qty{343}{K}$,
$T_{\text{F}} =
\qty{8.2e4}{K}$). (b) Het aandeel van de elektronen bij
\qty{300}{K}, in procenten. (c) En bij \qty{0.1}{K}? (d) Waarom zetten
experimentatoren $C/T$ tegen $T^2$ uit, en wat leveren het snijpunt en de
helling van die rechte?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}
Een \omterm{ex:b3:quantum-statistics:dwarf}{witte dwerg} met één zonsmassa ($M = \qty{2e30}{kg}$) en de straal van
de aarde ($R = \qty{6.4e6}{m}$), met één elektron per twee nucleonen. (a)
Bereken $n_{\text{e}}$ en $E_{\text{F}}$ (niet-relativistische formule).
(b) Vergelijk $E_{\text{F}}$ met $m_{\text{e}}c^2$ en met $k_{\text{B}}T$
bij de \qty{e7}{K} van het inwendige: rechtvaardig ``koud'' en ``bijna
relativistisch'' tegelijk. (c) Vergelijk de \omterm{thm:b3:quantum-statistics:fermi}{ontaardingsdruk} $\tfrac25 nE_{\text{F}}$ met de schatting van de zwaartekrachtsdruk $\sim GM^2/R^4$:
dezelfde orde? (d) Wat gebeurt er kwalitatief met elke kant van het
evenwicht als $M$ groeit?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:7}
Neutronenmaterie. Een neutronenster propt $M = 1.4\,M_\odot$ in $R
\approx \qty{11}{km}$. (a) Bereken de neutronendichtheid en vergelijk met de

= \qty{939}{MeV}$) en ga
na hoe relativistisch zij zijn. (c) Waarom zijn de elektronen van de ster

--- wat liet de $E_{\text{F}}$ van de elektronen ervoor betalen)? (d)
Hoeveel weegt één theelepel van deze materie?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:8}
Bosonen zonder aantal. (a) Fotonen hebben altijd $\mu = 0$: waarom kunnen
zij de condensatie van \cref{thm:b3:quantum-statistics:bec} niet ondergaan
(wat doet het gas in plaats daarvan bij afkoelen)? (b) Welke behoudswet
hebben rubidiumatomen die fotonen missen? (c) In 2010 \emph{is} er een
fotonen-BEC gemaakt, in een holte gevuld met kleurstof waarin fotonen
thermaliseren en hun aantal in de praktijk vastligt: welk ingrediënt
herstelde de kleurstofholte? (d) Fononen in een kristal: condensatie of
niet, en waarom?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
De twee isotopen van helium. (a) Bereken de $T_{\text{c}}$ voor een ideaal
gas voor vloeibaar $^4$He ($n = \qty{2.2e28}{m^{-3}}$, $m = 4u$) en
vergelijk met $T_\lambda = \qty{2.17}{K}$. (b) Waarom is de
overeenstemming maar ruw (wat verwaarloost ``ideaal'' in een vloeistof)?
(c) $^3$He blijft bij vrijwel dezelfde dichtheid tot in de millikelvins
normaal: noem het obstakel en de omweg die de natuur vond. (d) Toets de
logica van het samengestelde deeltje: een \emph{paar} $^3$He bevat een
even aantal fermionen --- boson; $^6$Li bevat $3\text{p} + 3\text{n}
+ 3\text{e} = 9$ fermionen: deel het in, en noem de befaamde proeven met
koude gassen die precies dit contrast tussen de lithiumisotopen uitbuiten.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ eerlijk. (a) Schrijf $N_{\text{exc}} = \int_0^\infty

condensaatfractie $1 -
(T/T_{\text{c}})^{3/2}$ af. (d) Waarom brengt
hetzelfde argument in een tweedimensionale doos \emph{geen} condensatie
voort (hoe verandert $g(
\epsilon)$, en wat gebeurt er met de convergentie
van de integraal)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
Waarom metalen bijeen blijven. Modelleer een metaal als elektronen met

(coulombwinst per elektron $\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (a)
Schrijf de energie per elektron $\epsilon(n) = An^{2/3} - Bn^{1/3}$ op en
toon aan dat zij een minimum heeft. (b) Toon aan dat de
evenwichtsafstand tussen de elektronen van de orde $a_0$ is --- de
dichtheden van metalen zijn geen toeval. (c) Schat de schaal van de
cohesie-energie in het minimum, in eV. (d) Welke rol speelt de
\omterm{thm:b3:quantum-statistics:fermi}{ontaardingsdruk} in deze gebonden toestand (wat stopt de coulombineenstorting)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
De schil van Sommerfeld, gekwantificeerd. (a) Beredeneer: van $N$
elektronen zitten er maar $\sim g(E_{\text{F}})k_{\text{B}}T$ in de
actieve schil, elk met een extra energie $\sim k_{\text{B}}T$: leid
$C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ op een constante na
af. (b) Vind met $g(E_{\text{F}}) = 3N/2E_{\text{F}}$ de uitdrukking
$C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ terug (de exacte
constante is $\pi^2/2$). (c) Dezelfde schillogica geeft de van de
temperatuur onafhankelijke spinsusceptibiliteit van Pauli: alleen
elektronen in de schil kunnen omklappen --- maar de schil groeit als $T$
terwijl elke bijdrage als $1/T$ daalt: toon de opheffing aan. (d)
Formuleer de algemene moraal voor ontaarde fermionen: \emph{elke} respons
is de klassieke maal $\sim T/T_{\text{F}}$ of haar analogon dat alleen het
oppervlak betreft.
\end{exercise}

\section{Vraagstuk: De zwaarst mogelijke witte dwerg}

\begin{problem}[{Weekendvraagstuk --- de grens van Chandrasekhar en de
kaarsen die zij aanstak}]\label{pb:b3:quantum-statistics:1}
In 1930 berekende Subrahmanyan Chandrasekhar, negentien jaar oud, op de
boot naar Engeland dat de ontaarding van de elektronen een dode ster
slechts tot een welbepaalde massa kan dragen. Het resultaat --- door
Eddington verbeten bestreden --- ligt aan de basis van neutronensterren,
zwarte gaten en de supernova's die als ``standaardkaarsen'' het versnellend
uitdijende heelal onthulden. Dit vraagstuk bouwt het met schalingen op.
Gegevens: $M_\odot =
\qty{2e30}{kg}$; één elektron per $\mu_e = 2$
nucleonen; $m_{\text{n}} = \qty{1.67e-27}{kg}$; $\hbar c = \qty{3.16e-26}{J.m}$; $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{Deel I --- Evenwicht van twee energieën.}
Behandel voor een ster met massa $M$ en straal $R$ de energieën per
eenheid van massa met schalingen.
\begin{enumerate}
  \item Zwaartekrachtsenergie: $E_{\text{g}} \sim -GM^2/R$ --- haal haar
        oorsprong op.
  \item Aantal elektronen $N_{\text{e}} = M/\mu_e m_{\text{n}}$ en
        dichtheid $n \sim N_{\text{e}}/R^3$: schrijf de schaal van de
        niet-relativistische \omterm{thm:b3:quantum-statistics:fermi}{fermi-energie} als functie van $M$ en $R$ op.
  \item Toon aan dat de totale kinetische (ontaardings)energie schaalt als
        $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item Minimaliseer $E_{\text{k}} + E_{\text{g}}$ over $R$ en toon aan
        dat de evenwichtsstraal $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$ is.
  \item Lees de schaling $R \propto M^{-1/3}$ af: zwaardere dwergen zijn
        \emph{kleiner}. Waarom ruikt dat al naar moeilijkheden?
  \item Reken $R$ uit voor $M = 1\,M_\odot$ en vergelijk met de gemeten
        $\approx\qty{5800}{km}$ van Sirius B.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De relativiteitstheorie trekt de vloer
weg.}
\begin{enumerate}[resume]
  \item Naarmate $R$ krimpt groeit $E_{\text{F}}$: schat de dichtheid
        waarbij $E_{\text{F}} \sim m_{\text{e}}c^2$, en het bijbehorende
        massagebied van de dwerg.
  \item In het ultrarelativistische regime is de energie van elk elektron
        $\sim\hbar c\,n^{1/3}$: toon aan dat de kinetische energie
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$ wordt.
  \item Beide energieën schalen nu als $1/R$: concludeer dat het evenwicht
        geen straal meer uitkiest --- vergelijk in plaats daarvan de twee
        coëfficiënten.
  \item Leid uit $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$ het kritische aantal
        elektronen en de massa
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item Reken de dimensieloze verhouding $\hbar c/Gm_{\text{n}}^2$ uit ---
        een van de grote zuivere getallen van de natuurkunde --- en daarna
        $M_{\text{Ch}}$ in zonsmassa's (de exacte behandeling geeft
        $1.4\,M_\odot$).
  \item Duid de formule: welke drie constanten spannen samen, en waarom
        draagt een \emph{kwantum}grens aan een \emph{ster} een $\hbar$?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Voorbij de grens.}
\begin{enumerate}[resume]
  \item Een dwerg die voorbij $M_{\text{Ch}}$ wordt geduwd (aanwas van een
        begeleider): wat draagt hem daarna, en op welke straalschaal
        (vervang $m_{\text{e}}$ door $m_{\text{n}}$ en $\mu_e$ door 1 in de
        straal van deel I)?
  \item Toon aan dat de eigen relativistische grens van de neutronenster
        van dezelfde orde $M_{\text{Ch}}$ is --- de natuur staat maar een
        uitstel met een factor twee toe (waargenomen maximum $\approx
        2.2\,M_\odot$). Wat wacht daarvoorbij?
  \item Tijdens de ineenstorting tot een neutronenster komt
        $\sim GM^2/R_{
        \text{ns}}$ aan zwaartekrachtsenergie vrij: reken
        haar uit voor $M = 1.4\,M_\odot$ en $R = \qty{11}{km}$, en
        vergelijk met de volledige opbrengst van de zon over
        \qty{10}{Gyr} ($\sim\qty{1.2e44}{J}$).
  \item Vrijwel alles vertrekt binnen seconden als neutrino's: welke
        waarneming uit 1987 bevestigde dit beeld op spectaculaire wijze?
  \item De terugkaatsing van de instortende kern en de duw van de
        neutrino's blazen de mantel weg: een supernova door
        kerninstorting. Onderscheid haar, elk in één zin, van de
        \emph{thermonucleaire} supernova (type Ia) van een \omterm{ex:b3:quantum-statistics:dwarf}{witte dwerg} die
        op de drempel van Chandrasekhar ontbrandt.
  \item Waarom maakt een ontbranding \emph{bij een universele massa}
        supernova's van type Ia vrijwel identiek --- en daarmee bruikbaar
        als standaardkaars?
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- De kaarsen en de
kosmos.}
\begin{enumerate}[resume]
  \item De schijnbare helderheid van een standaardkaars geeft haar
        afstand: formuleer de logica van de omgekeerde kwadratenwet en
        waarom \emph{standaard} het beslissende woord is.
  \item In 1998 vonden twee teams verre supernova's van type Ia
        \emph{zwakker} dan verwacht: formuleer de gevolgtrekking (de
        uitdijing versnelt --- donkere energie) en waarom de
        betrouwbaarheid van de kaars de kern van de zaak was.
  \item Geef de keten van dit vraagstuk in één regel: Pauli $\to$
        \omterm{thm:b3:quantum-statistics:fermi}{ontaardingsdruk} $\to$ $M_{\text{Ch}}$ $\to$ eenvormige explosies
        $\to$ kosmische versnelling.
  \item Eddington spotte met het resultaat (``een ster die
        ongerijmdheden begaat''); Chandrasekhar kreeg drieënvijftig jaar
        later de Nobelprijs: wat leert die episode over het volgen van
        vergelijkingen voorbij comfortabele conclusies?
  \item Het spectrum van Sirius B impliceerde in 1915 een ton per kubieke
        centimeter --- ``onzin'' totdat Fowler, binnen enkele maanden na
        de statistiek van Fermi in 1926, de ster als een ontaard gas
        verklaarde: zeg waarom het raadsel het ene jaar onoplosbaar was en
        het volgende routine.
  \item Overzichten van duizenden \omterm{ex:b3:quantum-statistics:dwarf}{witte dwergen} vinden massa's die zich
        rond $0.6\,M_\odot$ ophopen en geen enkele boven $\approx1.4\,M_\odot$: welke twee voorspellingen van dit vraagstuk bevestigt dat
        ene histogram?
  \item Vat het genoemde resultaat samen: $(\hbar c/Gm_{\text{n}}^2)^{
        3/2}m_{\text{n}} \approx 1.4\,M_\odot$ --- drie natuurconstanten
        die de zwaarst mogelijke \omterm{ex:b3:quantum-statistics:dwarf}{witte dwerg} vastleggen, de ontstekingsmassa
        van de geijkte explosies van het heelal.
\end{enumerate}
\end{problem}
