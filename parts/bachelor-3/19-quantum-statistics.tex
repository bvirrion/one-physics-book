\chapter{Quantum Statistics}\label{ch:b3:quantum-statistics}

Copper's conduction electrons are a gas a thousand times denser than
air --- yet they add almost nothing to copper's heat capacity, a
scandal that haunted classical physics for fifty years. A cloud of
rubidium atoms, cooled to a hundred billionths of a kelvin,
suddenly collapses into a single quantum state ten thousand atoms
deep. A dead star the mass of the Sun and the size of the Earth
refuses to shrink further, held up by nothing but the exclusion
principle. All three are the same chapter of physics: what happens
to a gas when $n\lambda_T^3 \sim 1$ and the occupation formulas of
\cref{ch:b3:grand-canonical} depart from Boltzmann. Fermions build
seas --- cold, rigid, pressurised; bosons build condensates ---
gregarious, coherent, superfluid. Between the Fermi sea's silent
majority and the condensate's stampede lies most of modern
condensed-matter and low-temperature physics, and the fate of the
stars.

\section{Gases of occupations}

\begin{proposition}[The quantum gas machinery]\label{prop:b3:quantum-statistics:machinery}
\index{density of states}
For non-interacting particles in a box, sums over levels become
integrals over the \omterm{prop:b3:schrodinger-three-dimensions:counting}{density of states} of
\cref{prop:b3:schrodinger-three-dimensions:counting} (per spin
state, $g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$):
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
with $\bar n$ the Fermi--Dirac or Bose--Einstein occupation. The
first equation fixes $\mu(T, n)$; the second delivers the
thermodynamics. In the dilute limit $n\lambda_T^3 \ll 1$ both
statistics collapse onto Boltzmann and the classical gas returns;
the quantum regimes begin where the wave packets touch.
\end{proposition}
\admitted

\section{The Fermi sea}

\begin{theorem}[The degenerate Fermi gas]\label{thm:b3:quantum-statistics:fermi}
\index{Fermi energy}\index{degeneracy pressure}
At $T = 0$, fermions fill every level up to the \emph{\omterm{thm:b3:quantum-statistics:fermi}{Fermi
energy}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{spin } \tfrac12) ,
\]
with mean energy $\tfrac35 E_{\text{F}}$ per particle and the
\emph{\omterm{thm:b3:quantum-statistics:fermi}{degeneracy pressure}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
a pressure at absolute zero, of purely quantum origin --- the
stiffness of every metal's electron gas and the support of dead
stars. At $0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$,
only the electrons within $\sim k_{\text{B}}T$ of the surface of
this \emph{Fermi sea} can respond to anything: the fraction
$\sim T/T_{\text{F}}$ is the master key to metallic behaviour.
\end{theorem}

\begin{proof}[Partial proof]
Fill $g(\epsilon)$ up to $E_{\text{F}}$: inverting $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ gives the stated $E_{\text{F}}$
(\cref{exo:b3:schrodinger-three-dimensions:8}); $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ by one more integral. Pressure:
compressing the box raises every level as $V^{-2/3}$, so $E
\propto V^{-2/3}$ and $P = -\partial E/\partial V = \tfrac23 E/V$
--- with $E = \tfrac35 NE_{\text{F}}$, the stated law.
\end{proof}

\begin{example}[The heat-capacity scandal, resolved]\label{ex:b3:quantum-statistics:heat}
\index{electronic heat capacity}
Classically, copper's free electron per atom should add $\tfrac32
k_{\text{B}}$ to the heat capacity --- $50\%$ on top of
Dulong--Petit. Measured: about $1\%$ at room temperature. The sea
explains: with $T_{\text{F}} \approx \qty{8e4}{K}$, only the
fraction $\sim T/T_{\text{F}} \approx 0.4\%$ of electrons can be
thermally excited, each by $\sim k_{\text{B}}T$:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(the exact Sommerfeld coefficient;
\cref{exo:b3:quantum-statistics:12}). Linear in $T$, it is buried
under the lattice's $T^3$ at ordinary temperatures but
\emph{dominates} below a few kelvin --- exactly as calorimeters
find, resolving in one stroke the puzzle of the electrons that
carry current but not heat capacity.
\end{example}

\begin{example}[Stars held up by exclusion]\label{ex:b3:quantum-statistics:dwarf}
\index{white dwarf}
A \omterm{ex:b3:quantum-statistics:dwarf}{white dwarf} (\cref{exo:b3:identical-particles:11}) balances
gravity against electron \omterm{thm:b3:quantum-statistics:fermi}{degeneracy pressure}. Since $P \propto
n^{5/3}$ stiffens faster than gravity's demands as the star
shrinks, an equilibrium radius exists at any modest mass --- with
the eerie scaling $R \propto M^{-1/3}$: \emph{heavier dwarfs are
smaller}. Squeezed harder, the electrons turn relativistic, the
pressure law softens to $n^{4/3}$, and the balance fails at a
unique mass, the Chandrasekhar limit $M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) --- beyond it,
collapse to a neutron star: a Fermi sea of neutrons at nuclear
density, the same physics at $10^{14}$ times the density.
\end{example}

\section{The Bose stampede}

\begin{theorem}[Bose--Einstein condensation]\label{thm:b3:quantum-statistics:bec}
\index{Bose--Einstein condensation}
For a gas of $N$ conserved bosons, the excited states can hold at
most $N_{\max}(T) = 2.612\,V/\lambda_T^3$ particles (the integral
of $g\bar n_{\text{BE}}$ at its ceiling $\mu \to 0$). Below the
critical temperature
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(equivalently } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
the surplus has nowhere to go but the single ground state: a
\emph{macroscopic} population
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
condenses into one quantum state --- matter behaving as one giant
wave function. Predicted by Einstein in 1925 from Bose's photon
statistics; realised in dilute rubidium in 1995 at
$T_{\text{c}} \approx \qty{100}{nK}$ (Nobel 2001), announced by
the tell-tale \emph{bimodal} velocity distribution: a thermal cloud
with a needle of zero-velocity condensate rising from its centre.
\end{theorem}
\admitted

\begin{example}[Helium, the veteran superfluid]\label{ex:b3:quantum-statistics:helium}
\index{superfluidity}
Liquid $^4$He (bosonic) transforms at $T_\lambda = \qty{2.17}{K}$
into a \emph{superfluid}: zero viscosity through the finest
capillaries, film flow over container walls, fountain effects, and
circulation quantised in units of $h/m$ --- a condensate's
macroscopic phase at work, shifted from the ideal-gas prediction
($\qty{3.1}{K}$ for helium's density) by interactions but
unmistakably the same phenomenon. Fermionic $^3$He cannot condense
alone: its atoms must first \emph{pair} (like the electrons of a
superconductor, \cref{ch:b3:electrons-in-solids}) into composite
bosons --- and superfluidity duly arrives, but at \qty{2.6}{mK}, a
thousand times colder: two isotopes, one lesson --- statistics is
destiny.
\end{example}

\begin{method}[Which statistics, which regime]\label{met:b3:quantum-statistics:method}
(1) Compute $n\lambda_T^3$ first: $\ll1$ means Boltzmann and
chapter 17 suffices. (2) Degenerate fermions: think in
$E_{\text{F}}$ and $T/T_{\text{F}}$; responses (heat capacity,
susceptibility, scattering) carry the factor $T/T_{\text{F}}$ or
sample only the Fermi surface. (3) Degenerate bosons: compare $T$
with $T_{\text{c}}$; below it, split the gas into condensate plus
thermal cloud. (4) Pressures: $n^{5/3}$ fermionic stiffness (or
$n^{4/3}$ relativistic) against whatever squeezes. (5) Composite
particles: count constituent fermions to choose the statistics ---
and remember pairing can convert one statistics into the other.
\end{method}

($n = \qty{8.5e28}{m^{-3}}$); (b) sodium ($\qty{2.5e28}{m^{-3}}$);

0$. (b) Compute the electrons' r.m.s.\ speed in copper. (c) That

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}

\end{exercise}

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
Helium's two isotopes. (a) Compute the ideal-gas $T_{\text{c}}$

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ honestly. (a) Write $N_{\text{exc}} = \int_0^\infty
g(\epsilon)\dd\epsilon/(\eu^{\beta\epsilon} - 1)$ at $\mu = 0$ and
reduce it, with $x = \beta\epsilon$, to $N_{\text{exc}} =
(V/\lambda_T^3)\cdot\frac{2}{\sqrt\pi}\int_0^\infty\frac{\sqrt
x\,\dd x}{\eu^x - 1}$. (b) Given $\int_0^\infty\sqrt x\,\dd x/(

\emph{no} condensation in a two-dimensional box (how does $g(
\epsilon)$ change, and what happens to the integral's
convergence)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
Why metals hold together. Model a metal as electrons of density
$n$ (\omterm{thm:b3:quantum-statistics:fermi}{Fermi energy} cost $\propto n^{2/3}$ per electron) attracted
to a neutralising ion background (Coulomb gain per electron
$\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (a) Write the energy per
electron $\epsilon(n) = An^{2/3} - Bn^{1/3}$ and show it has a
minimum. (b) Show the equilibrium electron spacing is of order
$a_0$ --- metallic densities are not accidents. (c) Estimate the
cohesive energy scale at the minimum, in eV. (d) What role does
\omterm{thm:b3:quantum-statistics:fermi}{degeneracy pressure} play in this bound state (what stops the
Coulomb collapse)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
Sommerfeld's shell, quantified. (a) Argue: of $N$ electrons, only
$\sim g(E_{\text{F}})k_{\text{B}}T$ sit in the active shell, each
holding extra energy $\sim k_{\text{B}}T$: derive $C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ up to a constant. (b) With
$g(E_{\text{F}}) = 3N/2E_{\text{F}}$, recover $C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ (the exact constant is $\pi^2/2$).
(c) The same shell logic gives Pauli's temperature-independent
spin susceptibility: only shell electrons can flip --- but the
shell grows as $T$ while each contribution falls as $1/T$: show
the cancellation. (d) State the general moral for degenerate
fermions: \emph{every} response is the classical one times
$\sim T/T_{\text{F}}$ or its surface-only analogue.
\end{exercise}

\section{Problem: The heaviest possible white dwarf}

\begin{problem}[{Weekend problem --- the Chandrasekhar limit, and
the candles it lit}]\label{pb:b3:quantum-statistics:1}
In 1930, aged nineteen, Subrahmanyan Chandrasekhar computed on the
boat to England that electron degeneracy can support a dead star
only up to a definite mass. The result --- resisted bitterly by
Eddington --- underlies neutron stars, black holes and the
supernova ``standard candles'' that revealed the accelerating
universe. This problem builds it by scaling. Data: $M_\odot =
\qty{2e30}{kg}$; one electron per $\mu_e = 2$ nucleons; $m_{\text{
n}} = \qty{1.67e-27}{kg}$; $\hbar c = \qty{3.16e-26}{J.m}$; $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{Part I --- Balance of two energies.}
For a star of mass $M$, radius $R$, treat the energies per unit
mass by scaling.
\begin{enumerate}
  \item Gravitational energy: $E_{\text{g}} \sim -GM^2/R$ ---
        recall its origin.
  \item Electron count $N_{\text{e}} = M/\mu_e m_{\text{n}}$ and
        density $n \sim N_{\text{e}}/R^3$: write the
        non-relativistic \omterm{thm:b3:quantum-statistics:fermi}{Fermi energy} scale as a function of $M$
        and $R$.
  \item Show the total kinetic (degeneracy) energy scales as
        $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item Minimise $E_{\text{k}} + E_{\text{g}}$ over $R$ and show
        the equilibrium radius is
        $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$.
  \item Read off the scaling $R \propto M^{-1/3}$: heavier dwarfs
        are \emph{smaller}. Why does that already smell of
        trouble?
  \item Evaluate $R$ for $M = 1\,M_\odot$ and compare with Sirius
        B's measured $\approx\qty{5800}{km}$.
\end{enumerate}

\medskip\noindent\textbf{Part II --- Relativity pulls the floor
away.}
\begin{enumerate}[resume]
  \item As $R$ shrinks, $E_{\text{F}}$ grows: estimate the density
        at which $E_{\text{F}} \sim m_{\text{e}}c^2$, and the
        corresponding dwarf mass region.
  \item In the ultrarelativistic regime each electron's energy is
        $\sim\hbar c\,n^{1/3}$: show the kinetic energy becomes
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$.
  \item Both energies now scale as $1/R$: conclude that
        equilibrium no longer selects a radius --- compare the two
        coefficients instead.
  \item Setting $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$, derive the
        critical electron number and the mass
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item Evaluate the dimensionless ratio $\hbar c/Gm_{\text{n}}^2$
        --- one of physics' great pure numbers --- and then
        $M_{\text{Ch}}$ in solar masses (the exact treatment gives
        $1.4\,M_\odot$).
  \item Interpret the formula: which three constants conspire, and
        why does a \emph{quantum} limit on a \emph{star} carry
        $\hbar$?
\end{enumerate}

\medskip\noindent\textbf{Part III --- Beyond the limit.}
\begin{enumerate}[resume]
  \item A dwarf pushed past $M_{\text{Ch}}$ (accretion from a
        companion): what supports it next, and at what radius
        scale (replace $m_{\text{e}}$ by $m_{\text{n}}$ and
        $\mu_e$ by 1 in Part I's radius)?
  \item Show the neutron star's own relativistic limit is of the
        same order $M_{\text{Ch}}$ --- nature allows only a
        factor-two reprieve (observed maximum $\approx
        2.2\,M_\odot$). What waits beyond?
  \item During the collapse to a neutron star, $\sim GM^2/R_{
        \text{ns}}$ of gravitational energy is released: evaluate
        it for $M = 1.4\,M_\odot$, $R = \qty{11}{km}$, and
        compare with the Sun's entire \qty{10}{Gyr} output
        ($\sim\qty{1.2e44}{J}$).
  \item Almost all of it leaves as neutrinos in seconds: which
        1987 observation confirmed this picture spectacularly?
  \item The collapsing core's rebound and neutrino push blow off
        the envelope: a core-collapse supernova. Distinguish it,
        in one sentence each, from the \emph{thermonuclear}
        (type Ia) supernova of a \omterm{ex:b3:quantum-statistics:dwarf}{white dwarf} igniting at the
        Chandrasekhar threshold.
  \item Why does detonation \emph{at a universal mass} make
        type Ia supernovae nearly identical --- and hence usable
        as standard candles?
\end{enumerate}
