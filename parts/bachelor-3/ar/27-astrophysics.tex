\section{النجم توازن}

\begin{proposition}[التوازن الهيدروستاتيكي]\label{prop:b3:astrophysics:hydrostatic}
\index{التوازن الهيدروستاتيكي (النجمي)}\index{البنية النجمية}
النجم لا ينهار ولا يتبدد: فعند كل نصف قطر، يحمل
تدرّج الضغط وزن الغاز فوقه. ومن أجل
الكتلة $m(r)$ داخل نصف القطر $r$،
\[
\frac{\dd P}{\dd r} = -\frac{Gm(r)\rho(r)}{r^2} .
\]
ومقدَّرًا على الشمس كلها، يكون $P_{\text{c}} \sim GM^2/R^4 \sim
\qty{e15}{Pa}$ ويكون، عبر قانون الغاز المثالي، $T_{\text{c}} \sim
10^7\,\unit{K}$ (\cref{exo:b3:astrophysics:1}) --- وهي بالضبط
درجة الحرارة التي تبدأ عندها بروتونات \cref{exo:b3:nuclear-physics:11}
بالنفق. وليس ذلك مصادفةً بل منظِّم حرارة: فالنجم
\emph{لا بدّ} أن ينكمش ويسخن حتى يشتعل لبّه، ثم يمسك الاندماج
بالتوازن ملايين إلى مليارات السنين. وللنجوم
سعة حرارية سالبة --- فَقْدُ الطاقة يعني الانكماش والازدياد
\emph{سخونةً} (\cref{exo:b3:astrophysics:5}) --- ولذلك
تفوز الثقالة دائمًا في النهاية.
\end{proposition}

\begin{proof}
القشرة $[r, r + \dd r]$ ذات المساحة $A$ تحسّ بضغط $P(r)A$ من
الأسفل، و $P(r + \dd r)A$ من الأعلى، ووزن $g\rho A\,\dd r$
مع $g = Gm(r)/r^2$ (فلا تجذب إلا الكتلة الداخلية، بمبرهنة
غاوس للثقالة). ويعطي توازن الثلاثة المعادلةَ
المذكورة.
\end{proof}

\section{حيوات النجوم}

\begin{definition}[مخطط هرتزشبرونغ--راسل]\label{def:b3:astrophysics:hr}
\index{مخطط هرتزشبرونغ--راسل}\index{المتتالية الرئيسية}
ارسم النجوم بدلالة درجة حرارة السطح (والساخن على \emph{اليسار}،
اصطلاحًا) في مقابل الإضاءة، وكلتاهما من قوانين الجسم الأسود في
\cref{ch:b3:photons-phonons}، فتأبى أن
تتبعثر: إذ تتزاحم تسعون بالمئة على \emph{المتتالية الرئيسية}، وهي
العصابة القطرية لحارقات الهدروجين، مرتَّبةً بالكتلة وحدها ---
فالأثقل أسخن \emph{وأسطع} بغير تناسب
($L \propto M^{3.5}$)، ومن ثمّ أقصر عمرًا
($t \propto M/L \propto M^{-2.5}$:
\cref{exo:b3:astrophysics:4}). وخارج العصابة تعيش
استثناءات المخطط ونهاياته: \emph{العمالقة} الباردة الهائلة
(أعلى اليمين)، و\emph{الأقزام البيض} الساخنة الضئيلة (أسفل اليسار)
--- لأن الإضاءة، عند درجة حرارة ثابتة وبحسب $L = 4\pi R^2\sigma T^4$،
مقياس \emph{للحجم}. ومخطط هرتزشبرونغ--راسل ليس
خريطة أنواع بل خريطة \emph{أعمار}: فنقطة النجم تنجرف عبره كلما
نفد وقوده.
\end{definition}

\begin{remark}[كيف تشيخ النجوم، وماذا تخلّف]\label{rem:b3:astrophysics:evolution}
\index{العملاق الأحمر}\index{المستعر الأعظم}\index{التخليق النوويّ (النجمي)}
حين ينفد هدروجين اللبّ يعاد تشغيل منطق منظِّم الحرارة:
فينكمش اللبّ ويسخن، ويشتعل الهليوم (فيندمج إلى كربون
وأكسجين)، وينتفخ الغلاف مئة ضعف --- أي عملاق أحمر.
والنجم الشبيه بالشمس يتوقف عند ذلك: فينفض غلافه ويتقاعد
قزمًا أبيض. أما النجم فوق $\approx8M_\odot$ فيظلّ يشعل
وقودًا جديدًا في قشور بصلية حتى يصير اللبّ \emph{حديدًا} ---
أي قمّة \cref{prop:b3:nuclear-physics:binding}، التي لا يعيد
منها أيّ اندماج طاقة. فينهار اللبّ الميت في ثانية؛ و
يعصف الارتداد وطوفان النيوترينوات بالنجم فيمزقانه: أي
\emph{مستعر أعظم}، يفوق مجرّته سطوعًا لبرهة، ويصوغ
العناصر وراء الحديد باقتناص النيوترونات ويبثّ ذلك كله
في السحب التي تبني الجيل التالي من النجوم و
الكواكب. والكالسيوم في عظامك والحديد في دمك
سلكا هذا الطريق (\cref{pb:b3:astrophysics:1}): فالكيمياء
فيزياء فلكية معاد تدويرها.
\end{remark}

\section{الجثث: المادة عند حدودها}

\begin{proposition}[الأقزام البيض والنجوم النيوترونية]\label{prop:b3:astrophysics:compact}
\index{القزم الأبيض}\index{النجم النيوتروني}\index{كتلة تشاندراسيخار}\index{الثقب الأسود}\index{نصف قطر شفارتزشيلد}
ما الذي يمسك نجمًا لم يعد يحترق؟ الميكانيك الكمومي. و\emph{\omterm{ex:b3:quantum-statistics:dwarf}{القزم
الأبيض}} كتلة شمسية من الكربون والأكسجين يسندها
\emph{ضغط انحلال إلكتروناتها} --- أي ضغط فيرمي عند $T = 0$
في \cref{ch:b3:quantum-statistics} --- بحجم
الأرض وبطنّ لكل ملعقة صغيرة. ولسند باولي
سقف: ففوق \emph{\omterm{prop:b3:astrophysics:compact}{كتلة تشاندراسيخار}} $M_{\text{Ch}}
\approx 1.4\,M_\odot$، تصير الإلكترونات نسبوية ويخسر
الضغط السباق (وهي نتيجة نذكرها ولا نستنتجها). ووراءها
تُسحق الإلكترونات في البروتونات فيتكوّن \emph{نجم نيوتروني}:
أي انحلال نيوتروني زائدَ تنافر نوويّ يمسك
$\sim1.4\,M_\odot$ في عشرات الكيلومترات عند الكثافة
النووية في \cref{def:b3:nuclear-physics:nuclide} --- أي
نواة بحجم مدينة، تدور حتى الميلي ثانية وتشعّ حزمةً مثل
منارة (أي \emph{نابض}). ووراء نحو $3\,M_\odot$
لا يقاوم شيء معلوم: فيسقط النجم داخل نصف القطر الذي يحتاج
الإفلات عنده إلى سرعة الضوء،
\[
R_{\text{s}} = \frac{2GM}{c^2}
\qquad (\qty{3}{km}\text{ لكل كتلة شمسية}) ,
\]
أي \emph{ثقب أسود} --- وهو كلمة الثقالة الأخيرة، وعتبة
النسبية العامة، على بعد كتاب واحد من هذا.
\end{proposition}
\admitted

\section{الكون المتمدد}

\begin{definition}[قانون هابل والبداية الساخنة]\label{def:b3:astrophysics:hubble}
\index{قانون هابل}\index{خلفية الكون الميكروية}\index{التخليق النوويّ في الانفجار العظيم}
كل مجرّة بعيدة تبتعد، بسرعة تتناسب مع البعد:
\[
v = H_0d , \qquad H_0 \approx \qty{70}{km/s}
\text{ لكل ميغافرسخ} .
\]
وهذا ليس انفجارًا \emph{في} الفضاء بل تمددًا
\emph{للفضاء} --- فأجرِه إلى الوراء يكن كل شيء يومًا
كثيفًا وساخنًا: أي قبل $1/H_0 \approx 14$ مليار سنة
(\cref{exo:b3:astrophysics:9}). والكون الفتيّ هو هذا الكتاب
معادًا تشغيله بسرعة. فعند الثانية الأولى يكون حساء جسيمات بمقياس \unit{MeV}
(\cref{ch:b3:particle-physics})؛ وفي الدقائق الأولى، تجري
\cref{ch:b3:nuclear-physics} جامحةً وتجمّد
\emph{ربعًا هليومًا بالكتلة} --- متنبَّأً به بعامل
بولتزمان، ومشاهَدًا في كل مكان
(\cref{exo:b3:astrophysics:11}). وعند 380\,000 سنة و
\qty{3000}{K}، تتيح \omterm{prop:b3:grand-canonical:saha}{معادلة ساها}
(\cref{prop:b3:grand-canonical:saha}) للإلكترونات أن تستقر أخيرًا
على البروتونات؛ فينقشع الضباب، وذلك الضوء المنتثر آخر مرة ---
ممطوطًا ألف ضعف بالتمدد --- يصل اليوم
\emph{خلفيةً كونية ميكروية}: أي طيف بلانك عند \qty{2.725}{K}
(\cref{ch:b3:photons-phonons})، وهو أقدم صورة
فوتوغرافية موجودة. وما تُظهره الصورة أيضًا: أن المجرّات تدور وتتعنقد
كأنما توجّهها \emph{مادة مظلمة} لم يلتقطها كاشف، و
أن التمدد \emph{يتسارع} تحت \emph{طاقة مظلمة}
لم يشرحها أحد --- أي الخمسة والتسعين بالمئة من الميزانية
التي لا يستطيع هذا الكتاب كتابتها بعد.
\end{definition}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:1}
الشمس، مقدَّرةً. $M = \qty{2e30}{kg}$ و $R = \qty{7e8}{m}$.
(أ) قدّر $P_{\text{c}} \sim GM^2/R^4$. (ب) ومع متوسط
الكثافة وقانون الغاز المثالي ($P = \rho k_{\text{B}}T/
m_{\text{p}}$)، قدّر $T_{\text{c}}$. (ج) وقارن مع
عتبة الاشتعال $10^7\,\unit{K}$ في
\cref{exo:b3:nuclear-physics:11}. (د) ولماذا يكون هذا التوافق
منظِّم حرارة لا مصادفة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:2}
وزن الضوء. (أ) من الثابت الشمسي
\qty{1360}{W/m^2} عند $d = \qty{1.5e11}{m}$، احسب
$L_\odot$. (ب) ومن $L = 4\pi R^2\sigma T^4$، احسب
درجة حرارة السطح. (ج) وتحقق من ذروة فين في مقابل أصفر
الشمس المائل إلى الأخضر. (د) وأيّ قانون من أيّ فصل استعملته كل خطوة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:3}
قراءة المخطط. سيريوس B: $T \approx \qty{25000}{K}$ و $L
\approx 0.026\,L_\odot$؛ وبيتلجوز: $T \approx \qty{3500}{K}$ و
$L \approx 10^5\,L_\odot$. (أ) ضع الاثنين على المخطط. (ب)
ومن $L = 4\pi R^2\sigma T^4$، احسب نصف قطر كلٍّ بالوحدات
الشمسية. (ج) ولو وُضع بيتلجوز في المجموعة الشمسية: فأيّ كواكب
تقع داخله؟ (د) وما الذي لا بدّ أنه يمسك سيريوس B؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:4}
عش سريعًا ومت شابًّا. مع $L \propto M^{3.5}$ والعمر
$t \propto M/L$، ومع عشرة مليارات سنة للشمس: (أ) احسب عمر
نجم $10\,M_\odot$؛ (ب) وعمر قزم أحمر $0.5\,M_\odot$ ---
قارنه بعمر الكون؛ (ج) ولماذا تموت النجوم الأثقل، ولها
وقود \emph{أكثر}، \emph{أبكر}؟ (د) وبمَ يوحي قصر
حيوات النجوم الضخمة عن مواضع المستعرات العظمى في مجرّة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:5}
ترموديناميك الثقالة الغريب. من أجل كرة غاز
تجذب نفسها ثقاليًّا تعطي مبرهنة الفيريال (\cref{ch:b3:lagrangian-mechanics})
طاقة كلية $E = -K$ (أي ناقص الطاقة الحركية/الحرارية).
(أ) استنتج: أن إشعاع الطاقة بعيدًا يجعل النجم \emph{أسخن}
--- أي سعة حرارية سالبة. (ب) وقدّر الطاقة الثقالية
$GM^2/R$ للشمس. (ج) وعند $L_\odot$ اليوم، كم كان الانكماش
وحده ليمدّها بالطاقة (وهو زمن كلفن--هلمهولتز)؟ (د) والجيولوجيا
تؤرخ الأرض بأربعة مليارات ونصف: اذكر مفارقة
القرن التاسع عشر وحلّها النووي.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:6}
تشريح قزم أبيض. خذ $M = 1\,M_\odot$ عند
$R = R_{\text{الأرض}} = \qty{6.4e6}{m}$. (أ) احسب متوسط
الكثافة، وكتلة ملعقة صغيرة (\qty{5}{mL}). (ب) واحسب
ثقالة السطح. (ج) \omterm{thm:b3:quantum-statistics:fermi}{وضغط الانحلال} في
\cref{ch:b3:quantum-statistics} يتحجّم وفق $n^{5/3}$، وحاجة الثقالة وفق
$M^2/R^4 \propto n^{4/3}M^{2/3}$: بيّن أن هذا يقتضي
العلاقة الغريبة $R \propto M^{-1/3}$ --- فالأقزام الأثقل
\emph{أصغر}. (د) واشرح فيزيائيًّا لماذا يكمش تكديس الكتلة
النجمَ، وبمَ ينذر ذلك عند \omterm{prop:b3:astrophysics:compact}{كتلة
تشاندراسيخار}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:7}
تشريح نجم نيوتروني. (أ) عند الكثافة النووية
$\qty{2.3e17}{kg/m^3}$، احسب نصف القطر الذي يمسك
$1.4\,M_\odot$. (ب) واللبّ قبل الانهيار كان يدور مرة كل 25 يومًا
عند $R \sim 10^5\,\unit{km}$: استعمل حفظ كمية الحركة الزاوية
لتقدير دور الدوران النهائي. (ج) واحسب ثقالة
السطح. (د) وتجميد التدفق يضخّم المجال المغناطيسي وفق
$B \propto 1/R^2$ (\cref{ch:b3:electromagnetism-in-matter}):
فمن \qty{e-2}{T} نجمية، أيّ مجال سطحي ينتج، و
لماذا تشعّ النوابض حزمًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:8}
نقطة اللاعودة. (أ) استنتج $R_{\text{s}} = 2GM/c^2$ من
شرط سرعة الإفلات النيوتوني $v_{\text{إفلات}} = c$ (فالاستنتاج
الأمين يحتاج إلى النسبية العامة؛ والجواب، على نحو
شهير، متوافق). (ب) واحسبه من أجل الشمس ومن أجل الأرض.
(ج) \omterm{prop:b3:astrophysics:compact}{والثقب الأسود} المركزي في المجرّة: $M = 4\times10^6\,M_\odot$
--- احسب $R_{\text{s}}$ وقارنه بمسافة الشمس إلى عطارد
($\qty{5.8e10}{m}$). (د) وبيّن أن متوسط الكثافة داخل
$R_{\text{s}}$ يهبط وفق $1/M^2$ --- \omterm{prop:b3:astrophysics:compact}{فالثقب الأسود} الكبير بما يكفي
أقلّ كثافة من الماء: فبمَ يخبرك ذلك عن حدوس ``السحق''؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:9}
حساب هابل. $H_0 = \qty{70}{km/s}$ لكل ميغافرسخ؛ و $1\,
\text{ميغافرسخ} = \qty{3.09e22}{m}$. (أ) حوّل $H_0$ إلى النظام الدولي. (ب)
واحسب $1/H_0$ بالسنوات. (ج) ومجرّة على \qty{200}{Mpc}: احسب
سرعة ابتعادها، والعامل الذي مُطّ به طول موجة ضوئها
(بدوبلر عند $z$ الصغيرة،
\cref{ch:b3:relativistic-kinematics}). (د) ولماذا يكون $1/H_0$
عمر الكون تقريبًا لا غير --- فماذا كان معدّل التمدد
يفعل في تلك الأثناء؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:10}
أقدم ضوء. الخلفية الميكروية جسم أسود عند \qty{2.725}{K}. (أ)
احسب طولها الموجي الذروي (\cref{ch:b3:photons-phonons}) و
برّر تسميتها ``ميكروية''. (ب) وقد أُصدرت ضوءًا عند
\qty{3000}{K}: فأيّ عامل تمدد مطّها،
ولماذا \qty{3000}{K} (تذكّر ساها،
\cref{prop:b3:grand-canonical:saha})؟ (ج) وكثافة فوتوناتها
$\sim\qty{4e8}{m^{-3}}$: قارنها بكثافة الباريونات،
$\sim\qty{0.25}{m^{-3}}$، واذكر نسبة الفوتونات إلى الباريونات.
(د) وتدقيق كلاسيكي على الطاولة: بضعة بالمئة من تشويش تلفاز
تناظري غير مضبوط كانت الخلفية --- فلماذا يستقبلها هوائي
مصوَّب إلى أيّ مكان؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:11}
ربع هليوم، من عامل بولتزمان. قرب $t \sim
\qty{1}{s}$ و $T \sim \qty{e10}{K}$ (مع $k_{\text{B}}T \approx
\qty{0.8}{MeV}$)، تركت التفاعلات الضعيفة وهي تتجمّد
نسبةَ النيوترونات إلى البروتونات عند قيمتها البولتزمانية
(\cref{ch:b3:canonical-ensemble}). (أ) ومع $\Delta mc^2 =
\qty{1.29}{MeV}$، احسب $n/p$ عند التجمّد. (ب) وقد قلّمها تفكك النيوترونات
في الدقائق الأولى إلى $\approx1/7$: بيّن أن ربط كل ناجٍ
في $^4$He يعطي نسبة \emph{كتلة} هليوم
$2(n/p)/(1 + n/p) = 25\,\%$. (ج) واذكر لماذا يكون هذا العدد،
مقيسًا في النجوم في كل مكان، برهانًا قويًّا على البداية
الساخنة. (د) ولماذا توقف الطبخ البدئي عند الهليوم
(مع آثار من الليثيوم)، تاركًا الكربون وكل ما عداه إلى
\cref{rem:b3:astrophysics:evolution}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:12}
المادة المفقودة. سرعة دوران مجرّة حلزونية تبقى مسطَّحة
عند $v = \qty{220}{km/s}$ حتى $r = \qty{50}{kpc}$
($\qty{1.5e21}{m}$)، أي وراء حافتها المرئية بكثير. (أ) ومن أجل
مدار دائري، بيّن أن $M(r) = v^2r/G$. (ب) واحسبها عند
\qty{50}{kpc}، بالكتل الشمسية. (ج) ومجموع النجوم والغاز المرئي
$\sim5\times10^{10}\,M_\odot$: فأيّ نسبة تكون
محسوبة؟ (د) والتوقع الكبلري وراء الحافة المرئية
هو $v \propto r^{-1/2}$ (وهو نمط المجموعة الشمسية): فبمَ يوحي
التسطّح عن توزّع الكتلة غير المرئية؟
\end{exercise}

\section{مسألة: الليلة التي مات فيها النجم}

\begin{problem}\label{pb:b3:astrophysics:1}
\emph{الليلة التي مات فيها النجم.} في 23 شباط 1987، صار عملاق
أزرق عملاق في سحابة ماجلان الكبرى --- $d \approx
\qty{50}{kpc} \approx \qty{1.5e21}{m}$ --- هو SN~1987A، أقرب
مستعر أعظم في أربعة قرون. و\emph{قبل} أن يرى أيّ مقراب سطوعه
بساعات، عدّت ثلاثة كواشف تحت الأرض
أربعة وعشرين نيوترينو. وأنت تعيد بناء تلك الليلة من المبادئ
الأولى.

\medskip\noindent\textbf{الجزء الأول --- النجم الذي كان.}

\begin{enumerate}
  \item وزن السلف ${\sim}20\,M_\odot$. باستعمال
        تحجيم \cref{exo:b3:astrophysics:4}، قدّر
        عمره وقارنه بعمر الشمس.
  \item وفي يومه الأخير كان النجم بصلةً: قشور من الهدروجين والهليوم والكربون والأكسجين والسيليكون
        حول لبّ حديدي. فلماذا يقتضي كل وقود تالٍ لبًّا أسخن؟
  \item ولماذا تتوقف المتتالية، توقفًا دائمًا، عند الحديد؟
        (\cref{prop:b3:nuclear-physics:binding}.)
  \item واللبّ الحديدي الخامل ينمو نحو $1.4\,M_\odot$. فما
        المميَّز في ذلك العدد، وأيّ ضغط يخفق
        هناك؟
  \item وحرق السيليكون يسند النجم نحو
        \emph{يوم} واحد: فبمَ يوحي ذلك عن تسارع ساعة
        النجم نحو النهاية؟
  \item وما إن يخفق سند الضغط، فكم يستغرق السقوط
        الحرّ تقريبًا، من أجل لبّ كثافته
        $\sim\qty{e12}{kg/m^3}$ (مع $t \sim 1/\sqrt{G\rho}$)؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- ومضة النيوترينوات.}

\begin{enumerate}[resume]
  \item اللبّ ($M \approx 1.4\,M_\odot = \qty{2.8e30}{kg}$)
        ينهار إلى $R \approx \qty{12}{km}$: احسب
        الطاقة الثقالية المحرَّرة، $E \sim GM^2/R$.
  \item وقارنها بضوء مليار شمس تشعّ
        شهرًا ($L \sim 10^{9}L_\odot$ على مدى $\qty{2.6e6}{s}$):
        بيّن أن الفوتونات خطأ تدوير --- فأين تذهب
        نسبة 99\,\%، ولماذا لا تستطيع مغادرة لبّ منهار
        بسرعة إلا النيوترينوات؟
  \item وانشر $E \approx \qty{3e46}{J}$ على كرة نصف قطرها
        \qty{1.5e21}{m}: احسب دفق الطاقة عند
        الأرض.
  \item ومع $\sim\qty{15}{MeV}$ لكل نيوترينو، احسب
        دفق العدد (أي النيوترينوات لكل \unit{m^2}).
  \item وكاشف يحمل \qty{2000}{t} من الماء:
        أي ${\sim}1.5\times10^{32}$ بروتونًا حرًّا. ومع
        احتمال تفاعل لكل بروتون مقداره
        $\sigma \approx \qty{1e-46}{m^2}$ لكل نقيض نيوترينو
        (معطًى)، قدّر العدّة المتوقَّعة --- وقارنها
        بالإحدى عشرة التي سجّلها كاشف واحد.
  \item والنيوترينوات سبقت الضوء بساعات. اشرح
        الساعتين: فما الذي يؤخر الفوتونات (وأين يولد
        السطوع فعلًا؟)، وبمَ يوحي \omterm{def:b3:relativistic-kinematics:event}{التزامن}
        شبه التامّ عن كتلة النيوترينو وسرعته؟
  \item وخمسة وعشرون نيوترينو معدودًا: اذكر، في جملة واحدة
        لكلٍّ، أمرين أكداهما عن نظرية المستعرات العظمى.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- ما الذي بقي.}

\begin{enumerate}[resume]
  \item احسب متوسط كثافة البقيّة البالغة \qty{12}{km} و
        $1.4\,M_\odot$ وقارنه بالكثافة النووية في
        \cref{def:b3:nuclear-physics:nuclide}.
  \item ولبّ السلف كان يدور مرة كل 30 يومًا عند
        $R \sim \qty{e5}{km}$: قدّر دور البقيّة.
  \item ومجاله المغناطيسي، وهو مجمَّد التدفق، يُضخَّم بمقدار
        $(R_0/R)^2$: فمن \qty{e-2}{T}، احسب المجال
        السطحي.
  \item واشرح المنارة: ما الذي يمسح، وما الذي يشعّ حزمةً، و
        لماذا لا يزال نابض السرطان (المولود سنة 1054، والمدوَّن ``نجمًا
        ضيفًا'' نهاريًّا) يومض ثلاثين مرة في الثانية.
  \item ولو تجاوزت البقيّة ${\sim}3\,M_\odot$: احسب
        $R_{\text{s}}$ من أجل $3\,M_\odot$ واذكر مصيرها.
  \item وتوقيت النوابض مستقر حتى $10^{-15}$: سمِّ استعمالًا
        فيزيائيًّا واحدًا لساعة بهذه الجودة، تدور في السماء.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- ماذا يعني ذلك.}

\begin{enumerate}[resume]
  \item الأكسجين الذي تتنفسه والحديد الذي في دمك:
        تتبّع سيرة كل ذرة في جملتين، من
        هدروجين الانفجار العظيم إلى مجرى دمك.
  \item والمستعرات العظمى من النمط Ia (وهي أقزام بيض دُفعت وراء
        $M_{\text{Ch}}$) تنفجر كلها عند الكتلة نفسها، ومن ثمّ
        بسطوع شبه واحد. اشرح لماذا يجعلها ذلك
        \emph{شموعًا معيارية}، وكيف يُبنى مخطط
        \cref{def:b3:astrophysics:hubble}
        منها.
  \item وفي سنة 1998 خرجت مستعرات Ia البعيدة \emph{أخفت}
        مما يتنبأ به خط هابل المستقيم: اذكر
        النتيجة التي نالت جائزة نوبل 2011.
  \item وقدّر كم سافر ضوء SN~1987A حين
        وصل، بالسنوات --- ولاحظ ما الذي كان يحدث على
        الأرض حين انطلق.
  \item ونيوترينوات سحابة ماجلان مرّت بك أيضًا (وبكل من كان
        حيًّا ذلك اليوم). فكم منها عبر كل إنسان
        (${\sim}\qty{0.03}{m^2}$) تقريبًا؟ وهل أحسّ به أحد
        (\cref{exo:b3:particle-physics:10})؟
  \item وأغلق دفتر الكتاب في أربعة أسطر: نجم متوازن
        عشرة ملايين سنة بالنفق؛ وانهيار دُفع
        نيوترينوات؛ وجثة يمسكها باولي؛ و
        الحطام --- ومنه القارئ --- لا يزال يركب
        تمدد هابل.
\end{enumerate}
\end{problem}
