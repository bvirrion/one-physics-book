\chapter{ميكانيك هاميلتون}\label{ch:b3:hamiltonian-mechanics}

صوّر بندولًا ألف مرة وارسم، من أجل كل لقطة، زاويته في مقابل عزمه
الحركي: تقع النقاط على منحنٍ مغلق، وكل حركة ممكنة للبندول هي منحنٍ
من هذا القبيل --- تأرجحات صغيرة على بيضاويات متداخلة، ودورات
كاملة على خطوط مموّجة فوقها وتحتها، وبينهما منحنٍ متقاطع وحيد
يفصل النظامين. وهذه الصورة، \emph{\omterm{def:b3:hamiltonian-mechanics:portrait}{المخطط الطوري}}، هي قلب الصياغة
التي أعطاها هاميلتون للميكانيك سنة 1833: فحالة الجملة نقطة في
فضاء الإحداثيات \emph{وكميات الحركة}، وتطورها تدفق في ذلك
الفضاء، والتدفق مولَّد بدالة واحدة هي الطاقة، عبر معادلتين من
المرتبة الأولى بديعتي التناظر. والمكسب ليس حسابات أسهل --- إذ
يفوز لاغرانج هناك عادةً --- بل \emph{الهندسة} الصحيحة: فالتدفق
يحفظ حجم \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور}، وهو ما ستقوم عليه الفيزياء الإحصائية؛ أما
هيكله الجبري، أي \omterm{def:b3:hamiltonian-mechanics:poisson}{قوس بواسون}، فهو بالضبط ما سيرقّيه الميكانيك
الكمومي إلى مبدّلة. وهذا الفصل هو المفصل بين ميكانيك الأشياء
وفيزياء القرن العشرين.

\section{من لاغرانج إلى هاميلتون}

\begin{definition}[الهاميلتونية والمعادلات القانونية]\label{def:b3:hamiltonian-mechanics:hamiltonian}
\index{هاميلتونية}\index{معادلات قانونية}\index{فضاء الطور}
من أجل جملة لاغرانجيتها $L(q, \dot q, t)$، نعبّر عن السرعات
بدلالة كميات الحركة $p_i = \partial L/\partial\dot q_i$ ونعرّف
\emph{الهاميلتونية} بأنها تحويل لوجاندر
\[
H(q, p, t) = \sum_i p_i\dot q_i - L ,
\]
وهي دالة الإحداثيات و\emph{كميات الحركة}. والفضاء ذو البعد
$2n$ الذي تعيش فيه $(q_1, \dots, q_n, p_1, \dots, p_n)$ هو
\emph{فضاء الطور}؛ ونقطة واحدة منه --- أي \emph{حالة} واحدة ---
تحدّد مستقبل الجملة وماضيها بأكمله عبر
\emph{المعادلات القانونية} الواردة في
\cref{thm:b3:hamiltonian-mechanics:canonical}.
\end{definition}

\begin{theorem}[معادلات هاميلتون]\label{thm:b3:hamiltonian-mechanics:canonical}
تكافئ \omterm{thm:b3:lagrangian-mechanics:euler-lagrange}{معادلات أويلر--لاغرانج} المعادلات $2n$ من المرتبة
الأولى
\[
\dot q_i = \frac{\partial H}{\partial p_i} , \qquad
\dot p_i = -\frac{\partial H}{\partial q_i} .
\]
\end{theorem}

\begin{proof}
نشتقّ $H = \sum p_i\dot q_i - L$ بوصفها دالة في $(q, p, t)$:
$\dd H = \sum(\dot q_i\,\dd p_i + p_i\,\dd\dot q_i) - \sum(
\partial_{q_i}L\,\dd q_i + \partial_{\dot q_i}L\,\dd\dot q_i) -
\partial_tL\,\dd t$. وتتلاشى حدود $\dd\dot q_i$ بحكم تعريف
$p_i$ --- وهو مغزى تحويل لوجاندر كله --- فيبقى
$\dd H = \sum(\dot q_i\,\dd p_i - \partial_{q_i}L\,\dd q_i) -
\partial_tL\,\dd t$. وبقراءة المشتقات الجزئية:
$\partial H/\partial p_i = \dot q_i$ و $\partial H/\partial q_i =
-\partial L/\partial q_i$، وهي بحسب أويلر--لاغرانج $-\dot p_i$.
(وكذلك $\partial_tH = -\partial_tL$.)
\end{proof}

\begin{proposition}[ما هي $H$]\label{prop:b3:hamiltonian-mechanics:energy}
تنطبق $H$ على \omterm{prop:b3:lagrangian-mechanics:energy}{دالة الطاقة} $h$ في الفصل السابق: فعلى طول حركة
لدينا $\dd H/\dd t = \partial H/\partial t$، ومن ثمّ تنحفظ $H$
كلما خلت من تعلق صريح بالزمن؛ وحين تكون الطاقة الحركية تربيعية
في السرعات مع قيود لا تتعلق بالزمن، يكون $H = E_k + E_p$، أي
الطاقة الميكانيكية --- مكتوبةً الآن بالمتغيرين $(q, p)$.
\end{proposition}

\begin{proof}
$\dd H/\dd t = \sum(\partial_{q_i}H\,\dot q_i + \partial_{p_i}H\,\dot
p_i) + \partial_tH = \sum(\partial_{q_i}H\,\partial_{p_i}H -
\partial_{p_i}H\,\partial_{q_i}H) + \partial_tH = \partial_tH$: إذ
يجعل تناظر \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات القانونية} المجموع يتلاشى تطابقيًا. وأما
التماهي مع $E_k + E_p$ فهو
\cref{prop:b3:lagrangian-mechanics:energy}.
\end{proof}

\begin{example}[هاميلتونيتان]\label{ex:b3:hamiltonian-mechanics:examples}
كتلة على نابض: $p = m\dot x$، ومنه
\[
H = \frac{p^2}{2m} + \frac12 kx^2 ;
\]
والمعادلتان القانونيتان $\dot x = p/m$ و $\dot p = -kx$ هما الزوج
المألوف. وبندول: $p = m\ell^2\dot\theta$ ومنه
\[
H = \frac{p^2}{2m\ell^2} - mg\ell\cos\theta .
\]
وفي الحالتين تكون $H$ هي الطاقة، وهي ثابتة على كل حركة: فالحركات
\emph{هي} خطوط المستوى للدالة $H$ في المستوي $(q,p)$.
\end{example}

\begin{method}[وصفة هاميلتون]\label{met:b3:hamiltonian-mechanics:recipe}
(1) من $L$ نحسب كميات الحركة $p_i = \partial L/\partial\dot q_i$
ونقلبها لإيجاد $\dot q_i$. (2) $H = \sum p_i\dot q_i - L$،
معبَّرًا عنها بالمتغيرين $(q, p)$ وحدهما --- وفي جملة طبيعية هي ببساطة $E_k + E_p$
مع كتابة $E_k$ بدلالة كميات الحركة. (3) نكتب \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات
القانونية}. (4) نرسم خطوط المستوى للدالة $H$: فمن أجل درجة حرية
واحدة هي المسارات نفسها، وتُقرأ منها الحركة النوعية كلها --- من
اهتزازات ودورات ومواضع توازن ومنحنيات فاصلة --- من دون حل أيّ
شيء.
\end{method}

\section{فضاء الطور}

\begin{definition}[المخطط الطوري والنقاط الثابتة والمنحني الفاصل]\label{def:b3:hamiltonian-mechanics:portrait}
\index{مخطط طوري}\index{نقطة ثابتة}\index{منحنٍ فاصل}
\emph{المخطط الطوري} لجملة هو عائلة مساراتها في \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور}.
و\emph{النقطة الثابتة} حالة تنعدم عندها المعادلتان القانونيتان
معًا --- أي وضع توازن: فتظهر أصغرية الكمون \emph{مركزًا} تحيط به
منحنيات مغلقة، وتظهر الأعظمية \emph{سرجًا} يمرّ به \emph{منحنٍ
فاصل}، وهو المسار الذي يقسم \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور} إلى مناطق ذات حركات
مختلفة نوعيًّا.
\end{definition}

\begin{example}[المخطط الطوري للبندول]\label{ex:b3:hamiltonian-mechanics:pendulum-portrait}
من أجل $H = p^2/2m\ell^2 - mg\ell\cos\theta$: بيضاويات مغلقة حول
$(0, 0)$ --- وهي \emph{التأرجحات}، أي الاهتزازات العادية؛ ومنحنيات
مموّجة عند $|p|$ الكبيرة --- وهي \emph{الدورات}، إذ يقلب البندول
فوق القمة؛ ويمرّ بالسرجين عند $\theta = \pm\pi$ المنحني الفاصل ذو
الطاقة $E = +mg\ell$، وعليه يستغرق البندول زمنًا لانهائيًا ليبلغ
القمة. والحالة الواقعة على المنحني الفاصل هي تأرجح أُطلق بقوة
\emph{تكفي بالضبط} لبلوغ الوضع المقلوب من دون فائض.
\end{example}

\begin{theorem}[مبرهنة ليوفيل]\label{thm:b3:hamiltonian-mechanics:liouville}
\index{مبرهنة ليوفيل}
يحفظ التدفق الهاميلتوني حجم \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور}: فإذا حملت \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات
القانونية} منطقة من الحالات الابتدائية، لم يتغير حجمها (أي
مساحتها من أجل درجة حرية واحدة) أبدًا --- مهما يصر شكلها.
\end{theorem}

\begin{proof}
للتدفق في \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور} مجال سرعات $(\dot q_i, \dot p_i) =
(\partial_{p_i}H, -\partial_{q_i}H)$. وتباعده هو
\[
\sum_i\Big(\frac{\partial\dot q_i}{\partial q_i}
 + \frac{\partial\dot p_i}{\partial p_i}\Big)
 = \sum_i\Big(\frac{\partial^2H}{\partial q_i\partial p_i}
 - \frac{\partial^2H}{\partial p_i\partial q_i}\Big) = 0 :
\]
أي إن التدفق غير قابل للانضغاط، والتدفق غير القابل للانضغاط ينقل
الأحجام من دون تغيير، كما في موائع كتاب السنة الثانية. (أما
الخطوة الصورية من انعدام التباعد إلى انحفاظ الحجم فهي مبرهنة
النقل المبرهنة هناك.)
\end{proof}

\begin{remark}[لماذا تهمّ مبرهنة ليوفيل]\label{rem:b3:hamiltonian-mechanics:liouville-matters}
ليس في صياغة نيوتن ما يوحي بأن شيئًا غير قابل للانضغاط. أما في
الصورة \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} فذلك تلقائي --- وهو رخصة الفيزياء الإحصائية:
فحين نصف غازًا من $10^{23}$ جزيئة بسحابة من النقاط في \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء
الطور}، تقول \omterm{thm:b3:hamiltonian-mechanics:liouville}{مبرهنة ليوفيل} إن السحابة تسيل كمائع غير قابل
للانضغاط، ومن ثمّ فإن ``عدد الحالات في حجم من \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور}'' مقدار
لا تستطيع الديناميك نفسها أن تخلقه أو تفنيه. وعلى هذا بالضبط
تقوم المسلَّمة الميكروقانونية للفيزياء الإحصائية، في موضع لاحق
من هذا الكتاب.
\end{remark}

\section{أقواس بواسون}

\begin{definition}[قوس بواسون]\label{def:b3:hamiltonian-mechanics:poisson}
\index{قوس بواسون}
\emph{قوس بواسون} لدالتين $f(q, p, t)$ و
$g(q, p, t)$ على \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور} هو
\[
\{f, g\} = \sum_i\Big(
 \frac{\partial f}{\partial q_i}\frac{\partial g}{\partial p_i}
 - \frac{\partial f}{\partial p_i}\frac{\partial g}{\partial q_i}\Big) .
\]
وهو متخالف التناظر، وخطي في كل وسيط، ويحقق قاعدة الجداء
$\{f, gh\} = \{f, g\}h + g\{f, h\}$، ويحقق من أجل الإحداثيات نفسها
\emph{العلاقات القانونية}
\[
\{q_i, p_j\} = \delta_{ij} , \qquad
\{q_i, q_j\} = \{p_i, p_j\} = 0 .
\]
\end{definition}

\begin{theorem}[التطور بوصفه قوسًا]\label{thm:b3:hamiltonian-mechanics:evolution}
على طول أيّ حركة،
\[
\frac{\dd f}{\dd t} = \{f, H\} + \frac{\partial f}{\partial t} .
\]
وعلى الخصوص، ينحفظ مقدار خالٍ من التعلق الصريح بالزمن إذا وفقط
إذا انعدم قوسه مع \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية}؛ والمعادلتان القانونيتان نفساهما
هما $\dot q_i = \{q_i, H\}$ و $\dot p_i
= \{p_i, H\}$: أي إن \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} \emph{تولّد} التطور الزمني.
\end{theorem}

\begin{proof}
قاعدة السلسلة مع \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات القانونية}: $\dd f/\dd t = \sum(
\partial_{q_i}f\,\dot q_i + \partial_{p_i}f\,\dot p_i) + \partial_tf =
\sum(\partial_{q_i}f\,\partial_{p_i}H - \partial_{p_i}f\,
\partial_{q_i}H) + \partial_tf$.
\end{proof}

\begin{example}[أقواس العزم الحركي]\label{ex:b3:hamiltonian-mechanics:brackets}
من أجل جسيم واحد، $L_z = xp_y - yp_x$ وقرينتاها بالتبديل الدوري.
ويعطي حساب مباشر انطلاقًا من العلاقات القانونية
\[
\{L_x, L_y\} = L_z , \qquad
\{L_y, L_z\} = L_x , \qquad
\{L_z, L_x\} = L_y ,
\]
و $\{L^2, L_z\} = 0$: أي إن مركبات العزم الحركي لا ``تتبادل''
فيما بينها، لكن كل واحدة منها تتبادل مع المربع الكلي. واحفظ شكل
هذه العلاقات --- فستعود، حرفيًا، مبدّلات في الميكانيك الكمومي،
حيث تملي كل شيء عن البنية الذرّية.
\end{example}

\begin{remark}[الباب المفضي إلى الميكانيك الكمومي]\label{rem:b3:hamiltonian-mechanics:doorway}
لاحظ ديراك سنة 1925 أن الميكانيك الكمومي كله يُنال بالإبقاء على
جبر الميكانيك الهاميلتوني واستبدال مبدّلة المؤثرات مقسومة على
$\iu\hbar$ \omterm{def:b3:hamiltonian-mechanics:poisson}{بقوس بواسون}: فتصير $\{q, p\} = 1$ هي $[\hat x, \hat p] = \iu\hbar$،
ويبقى الانحفاظ ``انعدام القوس مع $H$''، ويبقى التطور الزمني
مولَّدًا \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{بالهاميلتونية}. فالنظرية الكلاسيكية تحمل في عظامها هيكل
النظرية الكمومية؛ وستُظهر فصول الميكانيك الكمومي هذا التقابل
صراحةً.
\end{remark}

\section{الفعل والثوابت الكظومة}

\begin{definition}[متغير الفعل]\label{def:b3:hamiltonian-mechanics:action-variable}
\index{متغير الفعل}
من أجل جملة ذات درجة حرية واحدة تهتز على مسار طوري مغلق، يكون
\emph{متغير \omterm{def:b3:lagrangian-mechanics:action}{الفعل}} هو المساحة المحصورة مقسومة على
$2\pi$:
\[
I = \frac{1}{2\pi}\oint p\,\dd q .
\]
\end{definition}

\begin{proposition}[فعل المذبذب التوافقي]\label{prop:b3:hamiltonian-mechanics:sho-action}
من أجل $H = p^2/2m + \tfrac12 m\omega^2q^2$ عند الطاقة $E$، يكون
المسار قطعًا ناقصًا نصفا محوريه $\sqrt{2E/m\omega^2}$ و $\sqrt{2mE}$،
ويحصر المساحة $2\pi E/\omega$: ومن ثمّ
\[
I = \frac{E}{\omega} , \qquad E = \omega I ,
\]
وتواتر الاهتزاز هو $\partial E/\partial I = \omega$، كما ينبغي أن
يكون.
\end{proposition}

\begin{proof}
مساحة قطع ناقص، $\pi ab = \pi\sqrt{2E/m\omega^2}\sqrt{2mE} =
2\pi E/\omega$.
\end{proof}

\begin{proposition}[الثبات الكظوم]\label{prop:b3:hamiltonian-mechanics:adiabatic}
\index{ثابت كظوم}
إذا غُيّر وسيط من وسائط الجملة (طول أو صلابة أو مجال) تغييرًا
\emph{بطيئًا} --- على مدى أدوار اهتزاز كثيرة --- تغيّرت الطاقة
وتغيّر التواتر، لكن \omterm{def:b3:lagrangian-mechanics:action}{الفعل} $I$ يبقى ثابتًا بتقريب ممتاز: أي إن
$I$ \emph{\omterm{prop:b3:hamiltonian-mechanics:adiabatic}{ثابت كظوم}}. ومن ثمّ ينحفظ $E/\omega$ في المذبذب
المعدَّل ببطء: فصلّب النابض ببطء حتى يتضاعف $\omega$، تتضاعف
الطاقة معه.
\end{proposition}

\begin{proof}[برهان جزئي]
من أجل مذبذب تواتره $\omega(t)$ متغير ببطء: يمكن حساب شغل الوسيط
المتغير على دور واحد بالتوسيط؛ ويعطي الحساب (الموجَّه في
\cref{exo:b3:hamiltonian-mechanics:11})
$\dot E/E = \dot\omega/\omega$، أي $\dd(E/\omega)/\dd t = 0$
في المرتبة الغالبة. أما العبارة العامة، من أجل أيّ جملة مهتزة
مشوَّهة ببطء، فمُسلَّم بها --- وهي سبب صمود المدارات الكوكبية
أمام الاضطرابات البطيئة، ونقطة انطلاق ``النظرية الكمومية
القديمة'' أدناه.
\end{proof}

\begin{remark}[النظرية الكمومية القديمة]\label{rem:b3:hamiltonian-mechanics:old-quantum}
لماذا تكون طاقات الذرات متقطعة؟ كُتب الجواب الكمّي الأول (بور
1913، وسومرفيلد 1915) بلغة هذا الفصل: فمن بين كل الحركات
الكلاسيكية، تحتفظ الطبيعة بتلك التي يكون تكامل فعلها عددًا
صحيحًا من ثوابت بلانك،
\[
\oint p\,\dd q = nh .
\]
وبتطبيق ذلك على المذبذب التوافقي نجد $E_n = n\hbar\omega$
(ولا ينقصه إلا النصف من القيمة الحقيقية $\big(n + \tfrac12\big)\hbar
\omega$)؛ وبتطبيقه على جسيم في علبة نجد \emph{بالضبط} مستويات
كتاب السنة الثانية؛ وبتطبيقه على ذرة الهدروجين
(\cref{pb:b3:hamiltonian-mechanics:1}) نجد الطيف المقيس بأربعة
أرقام. فالقاعدة كانت صحيحة عدديًا ومؤقتة مفهوميًا --- ولمّا كان
\omterm{def:b3:lagrangian-mechanics:action}{الفعل} ثابتًا كظومًا، فإن العدد الكمومي $n$ لا يتغير تحت
الاضطرابات البطيئة، وهذا وحده ما جعل قاعدة كهذه قادرة على العمل
أصلًا. أما النظرية الحقيقية فتبدأ بعد خمسة فصول من هنا.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:1}
من أجل الكتلة على النابض: (أ) أنشئ $H$ من $L$ وتحقق من أن $H =
E_k + E_p$؛ (ب) اكتب \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات القانونية} وتحقق من أنها تعطي
$\ddot x = -\omega^2x$؛ (ج) بيّن أن المسارات قطوع ناقصة في
المستوي $(x, p)$ وأعطِ نصفَي محوريها عند الطاقة $E$؛
(د) بأيّ معنًى تتحرك النقطة الممثِّلة في اتجاه عقارب الساعة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:2}
للخرزة على الحلقة الدوّارة في الفصل السابق \omterm{def:b3:lagrangian-mechanics:action}{لاغرانجية} $L =
\tfrac12 mR^2\dot\theta^2 + \tfrac12 m\omega^2R^2\sin^2\theta +
mgR\cos\theta$. (أ) احسب $p$ و $H$. (ب) هل تنحفظ $H$؟ وهل هي
الطاقة الميكانيكية؟ (ج) ارسم تقريبيًا خطوط المستوى للدالة $H$ من أجل
$\omega^2 < g/R$ ومن أجل $\omega^2 > g/R$، مستعملًا الكمون
الفعّال الوارد في ذلك الفصل. (د) عيّن المراكز والسروج والمنحنيات
الفاصلة في الحالة السريعة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:3}
كرة ترتدّ ارتدادًا مرنًا على الأرض، $H = p^2/2m + mgz$ (حيث $z >
0$). (أ) ارسم المسار الطوري لطيران واحد ولحركة الارتداد كلها.
(ب) ما الذي يفعله الارتداد المرن بالنقطة الممثِّلة؟ (ج) احسب
المساحة المحصورة من أجل ارتفاع أقصى $z_{\max}$، واحسب \omterm{def:b3:lagrangian-mechanics:action}{الفعل} $I$.
(د) قاعدة بور--سومرفيلد $\oint p\,\dd z = nh$ مطبَّقة على نترون مرتدّ ($m =
\qty{1.67e-27}{kg}$) تعطي ارتفاعات ارتداد مكمّاة: قدّر أدناها،
وقارنه بالمقدار \qty{15}{\micro m} المقيس في تجربة الحالات
الكمومية الثقالية سنة 2002.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:4}
انطلاقًا من العلاقات القانونية وحدها، احسب (أ) $\{x^2, p\}$؛ (ب)
$\{xp, H\}$ من أجل $H = p^2/2m + E_p(x)$، وفسّر الحدّين
(فهذا القوس يقود \emph{مبرهنة الفيريال})؛ (ج) $\{L_z, x\}$ و
$\{L_z, p_x\}$؛ (د) بيّن أنه إذا كان $\{f, H\} = 0$ و $\{g, H\} = 0$
فإن $\{f, g\}$ ينحفظ أيضًا (استعمل متطابقة جاكوبي المُسلَّم بها:
$\{f,\{g,h\}\} + \{g,\{h,f\}\} + \{h,\{f,g\}\} = 0$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:5}
البندول بجوار منحنيه الفاصل. (أ) أعطِ طاقة المنحني الفاصل وأكبر
قيمة $|p|$ عليه. (ب) بيّن أنه على المنحني الفاصل $p =
\pm 2m\ell^2\omega_0\cos(\theta/2)$ مع $\omega_0 = \sqrt{g/\ell}$.
(ج) باستعمال $\dot\theta = p/m\ell^2$، بيّن أن زمن الانتقال من $\theta$
إلى القمة يتباعد لوغاريتميًا. (د) بندول حقيقي يُطلق تحت المنحني
الفاصل بقليل يبقى قرب الوضع المقلوب لحظة طويلة قبل أن يعود
متأرجحًا --- اربط ذلك بالسؤال (ج)، وبالتباطؤ المشاهَد بجوار كل
سرج.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:6}
لجسيم مشحون في مجال مغناطيسي \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} $H = (\vect p -
q\vect A)^2/2m$ حيث $\vect p$ كمية الحركة القانونية في الفصل
السابق. (أ) اكتب \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات القانونية} من أجل $\vect A = \tfrac12
B(-y, x, 0)$ وتحقق من أنها تعطي حركة السيكلوترون. (ب) بيّن أن $H =
\tfrac12 m\vect v^{\,2}$: أي إن المجال المغناطيسي لا يؤدي شغلًا.
(ج) احسب قوس المقدارين المحفوظين $\pi_x = p_x - \tfrac12
qBy$ و $\pi_y = p_y + \tfrac12 qBx$ (وتحقق أولًا من انحفاظ كل
منهما)، وبيّن أن $\{\pi_x, \pi_y\} = -qB$: أي مقداران محفوظان
قوسهما ثابت. (د) استعمل
\cref{exo:b3:hamiltonian-mechanics:4}(د) لتفسير لماذا لم يكن
يُنتظر منهما مقدار محفوظ ثالث مستقل.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:7}
ليوفيل باليد. (أ) من أجل الجسيم الحر، التدفق هو $q \to q +
pt/m$ و $p \to p$: بيّن أن مستطيلًا يصير متوازي أضلاع له المساحة
نفسها. (ب) من أجل المذبذب التوافقي، بيّن أن التدفق دوران
(بوحدات مناسبة) واستنتج. (ج) من أجل المذبذب \emph{المخمَّد}
$\ddot x = -\omega^2x - \gamma\dot x$، اكتب تباعد التدفق في
المستوي $(x, v)$ وبيّن أن المساحات تتقلص وفق $\eu^{-\gamma t}$: أي
إن التخميد ليس هاميلتونيًا. (د) إلى أين ``تذهب'' المساحة الضائعة
فيزيائيًّا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:8}
حركة مستوية في كمون مركزي، $H = (p_r^2 + p_\varphi^2/r^2)/2m
+ E_p(r)$. (أ) اكتب \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات القانونية} الأربع. (ب) بيّن أن $\{
p_\varphi, H\} = 0$ وعيّن قانون الانحفاظ. (ج) أرجع المسألة إلى
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{هاميلتونية} قطرية أحادية البعد بكمون فعّال. (د)
من أجل $E_p = -k/r$، عيّن المدار الدائري عند $p_\varphi$ معطى
وأعطِ طاقته --- على أن تُكمّى في
\cref{pb:b3:hamiltonian-mechanics:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:9}
(أ) تحقق من $\{L_x, L_y\} = L_z$ انطلاقًا من العلاقات القانونية.
(ب) استنتج القوسين الآخرين بالتبديل الدوري. (ج) بيّن أن
$\{L^2, L_z\} = 0$. (د) جسم صلب يدور بحرية: بأخذ $H =
L^2/2J$ (عطالة كروية التناظر)، بيّن أن المقادير $L_i$ الثلاثة تنحفظ؛
فماذا عن جسم عزوم عطالته غير متساوية (أجب
نوعيًّا انطلاقًا من الأقواس)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:10}
مستويات النظرية الكمومية القديمة. (أ) بيّن أن $\oint p\,\dd q = nh$
مطبَّقة على المذبذب التوافقي تعطي $E_n = n\hbar\omega$، مستعملًا
\cref{prop:b3:hamiltonian-mechanics:sho-action}. (ب) طبّقها على
جسيم في علبة طولها $a$ واستعد بالضبط $E_n =
n^2h^2/8ma^2$. (ج) من أجل جزيء ثنائي الذرة منمذَج بمذبذب
صلابته $k = \qty{1.9e3}{N/m}$ وكتلته المرجعة $\qty{1.14e-26}
{kg}$ (أحادي أكسيد الكربون)، احسب $\hbar\omega$ بالإلكترون-فولط وطول
موجة الإصدار $n = 1 \to 0$؛ وفي أيّ مجال طيفي يقع؟
(د) المستويات الحقيقية هي $(n + \tfrac12)\hbar\omega$: فهل يغيّر
النصف الناقص أطوال الموجات \emph{المصدَرة}؟ وأيّ تجربة تكشف نصف
طاقة النقطة الصفرية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:11}
اشتقاق الثبات الكظوم للمقدار $E/\omega$. كتلة على نابض تتزايد
صلابته $k(t)$ ببطء. (أ) بيّن أن $\dot E = \tfrac12\dot k\,
x^2$ على طول الحركة المضبوطة. (ب) خذ المتوسط على دور واحد عند $k$ ثابتة:
باستعمال $\langle\tfrac12 kx^2\rangle = E/2$، بيّن أن $\langle\dot E\rangle
= \dot k\,E/2k$. (ج) استنتج $\dd\ln E = \tfrac12\dd\ln k = \dd\ln
\omega$، ومن ثمّ ثبات $E/\omega$. (د) بندول يتأرجح بسعة
$\theta_0 = 5^\circ$ يُقصَّر خيطه ببطء من
\qty{1.0}{m} إلى \qty{0.5}{m}: أوجد السعة الجديدة والعامل الذي
نمت به طاقته، وقل من أين جاءت الطاقة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:12}
المساحة داخل المنحني الفاصل للبندول هي عدد من الحالات الكمومية.
(أ) بيّن أن $\oint p\,\dd\theta$ حول المنحني الفاصل يساوي
$16m\ell^2\omega_0$. (ب) بحسب قاعدة بور--سومرفيلد، يكون عدد
الحالات الكمومية ذات الطاقات دون المنحني الفاصل $N \approx
16m\ell^2\omega_0/h$: قدّره من أجل كتلة قدرها غرام على خيط طوله
\qty{10}{cm}. (ج) قدّره من أجل مذبذب جزيئي شبيه بالنشادر:
$m \sim \qty{1e-26}{kg}$ و $\ell \sim \qty{1e-10}{m}$ و
$\omega_0 \sim \qty{1e14}{rad/s}$. (د) استنتج: أين الحدّ بين
ميكانيك يحتاج إلى $\hbar$ وميكانيك لا يحتاج إليه؟
\end{exercise}

\section{مسألة: النظرية الكمومية القديمة وذرة الهدروجين}

\begin{problem}[{مسألة نهاية الأسبوع --- تكميم فضاء الطور، ووزن
ثابت ريدبرغ}]\label{pb:b3:hamiltonian-mechanics:1}
حسب بور سنة 1913 طيف الهدروجين انطلاقًا من ميكانيك كوكبي مضافًا
إليه قاعدة تكميم واحدة؛ وأدرك سومرفيلد أن هذه القاعدة عبارة عن
مساحة في \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء الطور}. وتعيد هذه المسألة بناء حسابهما بأدوات هذا
الفصل. المعطيات: $m_{\text{e}} =
\qty{9.11e-31}{kg}$ و $e = \qty{1.60e-19}{C}$ و $1/4\pi\varepsilon_0 =
\qty{8.99e9}{N.m^2/C^2}$ و $h = \qty{6.63e-34}{J.s}$ و $c =
\qty{3.00e8}{m/s}$. ونكتب $k = e^2/4\pi\varepsilon_0$.

\medskip\noindent\textbf{الجزء الأول --- \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{هاميلتونية} كبلر.}
يتحرك إلكترون في مستوٍ حول بروتون ثابت.
\begin{enumerate}
  \item برّر معاملة البروتون على أنه ثابت (نسبة الكتلتين)، واكتب
        \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} $H = \dfrac{p_r^2}{2m_{\text{e}}} +
        \dfrac{p_\varphi^2}{2m_{\text{e}}r^2} - \dfrac{k}{r}$.
  \item اكتب \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{المعادلات القانونية} الأربع.
  \item بيّن أن $p_\varphi$ ينحفظ وسمِّه.
  \item أرجع الحركة القطرية إلى الكمون الفعّال
        $U_{\text{فعّال}}(r) = p_\varphi^2/2m_{\text{e}}r^2 - k/r$ وارسمه
        تقريبيًا.
  \item عيّن المدار الدائري: $r_{\text{c}} =
        p_\varphi^2/m_{\text{e}}k$.
  \item بيّن أن طاقته $E = -k/2r_{\text{c}} =
        -m_{\text{e}}k^2/2p_\varphi^2$، وتحقق من أنها نصف الطاقة
        الكامنة (وهي نسبة الفيريال لمدارات الثقالة في كتاب السنة
        الأولى).
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- المدار الكلاسيكي.}
\begin{enumerate}[resume]
  \item من أجل مدار دائري نصف قطره $r$، أوجد السرعة $v$ والتواتر
        المداري $f_{\text{مداري}}$ بدلالة $r$.
  \item مدار بحجم ذري، $r = \qty{0.05}{nm}$: احسب $v$
        (والنسبة $v/c$) و $f_{\text{مداري}}$ والطاقة بالإلكترون-فولط.
  \item احسب العزم الحركي $p_\varphi$ لذلك المدار
        وقارنه بالمقدار $\hbar = h/2\pi$: بم يوحي هذا التقارب؟
  \item كلاسيكيًّا، الإلكترون الدائر ثنائي قطب متذبذب فيشعّ
        (كتاب السنة الثانية) عند $f_{\text{مداري}}$؛ وتضمحل
        طاقته في نحو \qty{1e-11}{s}. اذكر التنبؤين القاتلين
        اللذين يترتبان على ذلك في شأن الذرات، وما الذي يُشاهَد
        بدلًا منهما.
  \item أيّ سمة من سمات الأطياف المشاهَدة (الخطوط المتقطعة،
        وقاعدة التأليف $1/\lambda = R_H(1/n^2 - 1/n'^2)$) أوحت
        بوجود \emph{مستويات} طاقة متقطعة؟
  \item فسّر لماذا يكون \emph{\omterm{prop:b3:hamiltonian-mechanics:adiabatic}{الثابت الكظوم}} المرشَّح الطبيعي
        لمقدار يأخذ قيمًا كونية ثابتة
        (تذكّر \cref{prop:b3:hamiltonian-mechanics:adiabatic}).
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- التكميم.}
افرض قاعدة سومرفيلد على الحركة الزاوية للمدار الدائري:
$\oint p_\varphi\,\dd\varphi = nh$ مع $n = 1, 2, 3, \dots$
\begin{enumerate}[resume]
  \item بيّن أن القاعدة تُكتب $p_\varphi = n\hbar$.
  \item استنتج أنصاف الأقطار المسموحة $r_n = n^2a_0$ مع $a_0 =
        \hbar^2/m_{\text{e}}k$، واحسب $a_0$.
  \item استنتج الطاقات المسموحة $E_n = -E_{\text{I}}/n^2$ مع
        $E_{\text{I}} = m_{\text{e}}k^2/2\hbar^2$، واحسب
        $E_{\text{I}}$ بالجول وبالإلكترون-فولط.
  \item احسب السرعة على المدار الأول وتحقق من أن $v_1/c =
        k/\hbar c \approx 1/137$ (وهو ثابت البنية الدقيقة).
  \item يحمل الفوتون طاقة الانتقال: اشتقّ صيغة ريدبرغ وقيمة
        $R_H = E_{\text{I}}/hc$؛ وقارنها بالقيمة المقيسة
        \qty{1.097e7}{m^{-1}}.
  \item احسب أطوال موجات الانتقالات $2 \to 1$ و
        $3 \to 2$ و $\infty \to 2$؛ أيّها مرئي، وما لونه؟
  \item أيون الهليوم المتأين He$^+$ هدروجين شحنته النووية
        $2e$: كيف يتغير $a_0$ و $E_{\text{I}}$ بالمقياس، وأين
        يقع خطه $2 \to 1$؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- المواجهة.}
\begin{enumerate}[resume]
  \item احسب التواتر المداري $f_{\text{مداري}}(n)$ وتواتر
        الانتقال $\nu_{n \to n-1}$ من أجل $n = 2$ و $10$ و
        $100$، وبيّن أن نسبتهما تؤول إلى $1$ كلما كبر $n$.
  \item هذا هو \emph{مبدأ التقابل} عند بور: اذكره في
        جملة واحدة.
  \item تبدأ القاعدة $p_\varphi = n\hbar$ من $n = 1$: فأيّ عبث
        يعنيه $n = 0$ من أجل مدار دائري؟
  \item سيحتفظ الميكانيك الكمومي بالعلاقة $E_n = -E_{\text{I}}/n^2$ بالضبط،
        لكنه سيطرح المدارات: سمِّ واقعتين قابلتين للقياس تخطئ
        فيهما صورة المدارات (مقدار العزم الحركي للحالة
        الأساسية؛ ووجود حالات لها $n$ نفسه وأشكال
        مختلفة).
  \item الميون إلكترون أثقل $207$ مرة: من أجل الهدروجين
        الميوني، احسب $a_0^\mu$ و $E_{\text{I}}^\mu$، وفسّر
        لماذا تسبر الذرات الميونية \emph{النواة}.
  \item لخّص النتيجة الموسومة: \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{ثابت كظوم} واحد، يُجعل مساويًا
        لأعداد صحيحة من $h$، يعطي $a_0 = \qty{52.9}{pm}$ و
        $E_{\text{I}} = \qty{13.6}{eV}$ وطيف الهدروجين بأربعة
        أرقام --- ويسلّم النظرية الحقيقية ثابتيها المركزيين.
\end{enumerate}
\end{problem}
