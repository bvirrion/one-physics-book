\chapter{السبين والجمل ذات المستويين}\label{ch:b3:spin-two-level}

سنة 1922 أرسل شتيرن وغيرلاخ حزمة من ذرات الفضة عبر مغنط
غير متناظر، وهما يتوقعان إما انعدام الانحراف وإما لطخة
متصلة. فانشطرت الحزمة نظيفةً \emph{شطرين}. ولا عزم حركي
مداري يفعل ذلك: فالعدد $2l + 1$ فردي دائمًا. وكانت الذرات
تعلن ملكية كمومية صرفة جديدة --- هي \emph{\omterm{def:b3:spin-two-level:spin}{السبين}}، وهو عزم
حركي ذاتي مقداره $j = \tfrac12$، أي الدرجة نصف الصحيحة
التي أبقاها جبر \cref{ch:b3:quantum-angular-momentum} جاهزة ولم
تستطع المدارات استعمالها. \omterm{def:b3:spin-two-level:spin}{والسبين} يضاعف كل حالة إلكترونية (فيكمل
الإحصاء وراء الجدول الدوري)، ويحمل مغناطيسية الحديد
والبروتون، وهو أنظف جملة ذات مستويين تقدّمها الطبيعة:
ففيزياء هذا الفصل تدير الماسح \omterm{prop:b3:spin-two-level:rabi}{بالرنين المغناطيسي} في كل مستشفى،
وساعة السيزيوم التي تعرّف الثانية،
والهمسة عند واحد وعشرين سنتيمترًا التي تُرسم بها المجرّات، والكيوبت.

\section{التجربة التي وجدته}

\begin{example}[شتيرن--غيرلاخ]\label{ex:b3:spin-two-level:stern-gerlach}
العزم المغناطيسي $\vect\mu$ في مجال \emph{غير منتظم} يحسّ
القوة $F_z = \mu_z\,\partial B_z/\partial z$: فالانحراف
يقيس $\mu_z$. والتوقع الكلاسيكي: عزوم موجَّهة عشوائيًا،
أي مروحة متصلة. والتوقع الكمومي من أجل العزوم المدارية: $2l + 1$
بقعة: واحدة أو ثلاث أو خمس --- فردية دائمًا. والمشاهَد في الفضة (وفي
الهدروجين): \emph{بقعتان} متناظرتان ولا شيء بينهما. فالمركبة
المقيسة تأخذ قيمتين بالضبط --- وهو توقيع $j
= \tfrac12$ الممنوع على المدارات، والعرض المباشر
للتكميم: فالجهاز أداة قياس لمركبة \omterm{def:b3:spin-two-level:spin}{سبين} واحدة،
وتنشطر الحزمة إلى قيمتيها الذاتيتين.
\end{example}

\begin{definition}[سبين النصف]\label{def:b3:spin-two-level:spin}
\index{سبين}\index{مصفوفات باولي}\index{سبينور}
يحمل الإلكترون (وكذلك البروتون والنترون والكواركات) عزمًا
حركيًا ذاتيًا مقداره $j = \tfrac12$: أي فضاء حالات ثنائي البعد
يولّده $\ket\uparrow, \ket\downarrow$ (وهما الحالتان الذاتيتان
للمؤثر $\hat S_z$ بالقيمتين $\pm\hbar/2$)، وعليه
\[
\hat{\vect S} = \frac\hbar2\,\hat{\vect\sigma} , \qquad
\sigma_x = \begin{pmatrix}0&1\\1&0\end{pmatrix} ,\;
\sigma_y = \begin{pmatrix}0&-\iu\\\iu&0\end{pmatrix} ,\;
\sigma_z = \begin{pmatrix}1&0\\0&-1\end{pmatrix}
\]
--- وهي \emph{مصفوفات باولي}، التي تحقق جبر العزم
الحركي في أصغر بيت ممكن له. والحالة العامة
$\alpha\ket\uparrow + \beta\ket\downarrow$ \emph{سبينور}. أما
العزم المغناطيسي للإلكترون فهو
\[
\hat{\vect\mu} = -g\,\frac{\mu_{\text{B}}}{\hbar}\,\hat{\vect S} ,
\qquad g \approx 2 :
\]
أي ضعف المعدل المداري لكل وحدة عزم حركي --- وهو الشذوذ
الذي قاسه أينشتاين ودي هاس
(\cref{exo:b3:quantum-angular-momentum:12}) والذي ستتنبأ به
معادلة ديراك لاحقًا. والسبين ليس دورانًا لشيء: فلا نصف قطر
ولا ``كرة دوّارة'' يصمد أمام التدقيق --- بل هو ذاتي، كالشحنة.
\end{definition}

\begin{example}[مرشّحات شتيرن--غيرلاخ المتسلسلة]\label{ex:b3:spin-two-level:chained}
انتقِ الحزمة $S_z = +\hbar/2$، ثم قِس $S_z$ من جديد: تجيب كل
الذرات بالقيمة $+$. وقِس $S_x$ بدلًا منه: فتجد النصف والنصف --- إذ إن الحالة
$\ket\uparrow$ هي التراكب $(\ket{+x} + \ket{-x})/\sqrt2$.
والآن أبقِ الحزمة $S_x = +$ وقِس $S_z$ مرة أخرى: فتجد النصف والنصف
\emph{من جديد} --- إذ محا قياس $S_x$ حدّةَ $S_z$
السابقة. فثلاثة مغانط تكفي لعرض اللاتوافق
والانهيار \omterm{thm:b3:quantum-formalism:postulates}{وقاعدة بورن}؛ وهذه السلسلة هي
\cref{ex:b3:quantum-formalism:threepolarizers} مؤدّاةً بالذرات.
\end{example}

\section{كرة بلوخ}

\begin{proposition}[السبين على أيّ محور]\label{prop:b3:spin-two-level:bloch}
\index{كرة بلوخ}
من أجل المنحى الواحدي $\vect n = (\sin\theta\cos\varphi,
\sin\theta\sin\varphi, \cos\theta)$، يكون للمؤثر $\vect
n\cdot\hat{\vect\sigma}$ قيمتان ذاتيتان $\pm1$، مع
\[
\ket{+\vect n} = \cos\tfrac\theta2\,\ket\uparrow +
\eu^{\iu\varphi}\sin\tfrac\theta2\,\ket\downarrow .
\]
وكل \omterm{def:b3:spin-two-level:spin}{سبينور} صافٍ هو $\ket{+\vect n}$ من أجل منحى واحد بالضبط:
فينطبق فضاء الحالات على كرة --- هي \emph{\omterm{prop:b3:spin-two-level:bloch}{كرة بلوخ}} --- وتقع
الحالات المتعامدة في \emph{طرفين متقابلين} (ولاحظ أنصاف الزوايا:
فالمنحيان المتعاكسان، لا المتعامدان، هما المتعامدان في الحالات).
ودورانات الكرة هي بالضبط التطورات التي يولّدها مجال مغناطيسي:
أي هندسة كل تلاعب بكيوبت أُجري يومًا.
\end{proposition}

\begin{proof}[برهان جزئي]
نقطّر المصفوفة $2\times2$ التالية $\vect n\cdot\vect\sigma =
\begin{pmatrix}\cos\theta & \sin\theta\,\eu^{-\iu\varphi}\\
\sin\theta\,\eu^{\iu\varphi} & -\cos\theta\end{pmatrix}$: فأثرها $0$
ومحدّدها $-1$ وقيمتاها الذاتيتان $\pm1$؛ ويُتحقق من \omterm{def:b3:spin-two-level:spin}{السبينور} المعطى
بالتعويض (بمتطابقات نصف الزاوية). وأما الطرفان المتقابلان: فالتبديل $\theta \to \pi -
\theta$ و $\varphi \to \varphi + \pi$ يعطي $\braket{+\vect
n'}{+\vect n} = 0$.
\end{proof}

\section{السبين في مجال: المبادرة والرنين}

\begin{proposition}[مبادرة لارمور]\label{prop:b3:spin-two-level:larmor}
\index{مبادرة لارمور}\index{النسبة الجيرومغناطيسية}
في مجال ساكن $\vect B_0 = B_0\vect e_z$، تفصم \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} $\hat
H = -\hat{\vect\mu}\cdot\vect B_0$ المستويين بالمقدار
$\hbar\omega_0$ وتجعل \omterm{def:b3:spin-two-level:spin}{السبين} المتوسط \emph{يبادر} حول
المجال:
\[
\frac{\dd\langle\hat{\vect S}\rangle}{\dd t} =
\gamma\,\langle\hat{\vect S}\rangle\wedge\vect B_0 , \qquad
\omega_0 = |\gamma|B_0 ,
\]
مع \omterm{prop:b3:spin-two-level:larmor}{النسبة الجيرومغناطيسية} $\gamma$ (حيث $\mu = \gamma S$). فتدور
متجهة بلوخ حول $z$ عند $\omega_0$، وزاويتها إلى المجال مجمَّدة
--- أي صورة الجيروسكوب الكلاسيكية بالضبط، وهي مضبوطة هنا لأن
المبدّلات خطية. ومن أجل الإلكترون: $\qty{28}{GHz/T}$؛ ومن أجل
البروتون (عزمه $2.79$ مغنطونًا نوويًا): $\gamma_p/2\pi =
\qty{42.6}{MHz/T}$ --- وهي لوحة ترخيص كل جهاز تصوير \omterm{prop:b3:spin-two-level:rabi}{بالرنين المغناطيسي}.
\end{proposition}

\begin{proof}
نطبّق \cref{prop:b3:quantum-formalism:evolution} مع $\hat H =
-\gamma B_0\hat S_z$: فتعطي المبدّلات $\dd\langle\hat
S_x\rangle/\dd t = -\gamma B_0\langle\hat S_y\rangle$ و
$\dd\langle\hat S_y\rangle/\dd t = +\gamma B_0\langle\hat
S_x\rangle$ و $\dd\langle\hat S_z\rangle/\dd t = 0$، وهي
تتجمع في الجداء الشعاعي المذكور.
\end{proof}

\begin{proposition}[الرنين المغناطيسي]\label{prop:b3:spin-two-level:rabi}
\index{الرنين المغناطيسي}\index{اهتزاز رابي}
أضف مجالًا صغيرًا $B_1$ يدور في المستوي المستعرض بالتواتر
$\omega$. ففي المرجع الدائر معه يرجع $B_0$ فعليًا
إلى $(\omega_0 - \omega)/|\gamma|$؛ و\emph{عند الرنين}
($\omega = \omega_0$) لا يبقى غير $B_1$، ويبادر \omterm{def:b3:spin-two-level:spin}{السبين}
حوله عند \emph{تواتر رابي} $\Omega_1 =
|\gamma|B_1$: فابتداءً من $\ket\uparrow$، يكون احتمال
الانقلاب
\[
\mathcal P_\downarrow(t) = \sin^2\Big(\frac{\Omega_1t}{2}\Big) .
\]
والنبضة التي مدتها $\pi/\Omega_1$ (أي ``نبضة $\pi$'') تقلب
\omterm{def:b3:spin-two-level:spin}{السبين}؛ ونصفها (أي ``نبضة $\pi/2$'') تضعه في المستوي الاستوائي،
حيث يبادر عند $\omega_0$ و\emph{يبثّ}، بالتحريض الفاراداي،
تواترَه في أيّ وشيعة قريبة. وخارج الرنين
تنهار سعة التقلب وفق $\Omega_1^2/[\Omega_1^2 +
(\omega - \omega_0)^2]$: فالاستجابة انتقائية بحدّة --- أي
رنين --- وهو ما يجعل السبينات قابلة للعنونة،
نوعًا نوعًا، وبتدرج مجال، موضعًا موضعًا
(\cref{pb:b3:spin-two-level:1}).
\end{proposition}
\admitted

\begin{example}[البنية فوق الدقيقة وخط 21 سنتيمترًا]\label{ex:b3:spin-two-level:hyperfine}
\index{البنية فوق الدقيقة}\index{خط 21 سنتيمترًا}
في الحالة الأساسية للهدروجين يتفاعل سبينا الإلكترون والبروتون
عبر عزميهما المغناطيسيين: فتنفصم حالات \omterm{def:b3:spin-two-level:spin}{السبين} الأربع
إلى ثلاثي وأحادي يفصل بينهما $\Delta E =
\qty{5.9}{\micro eV}$ لا غير --- وهو الانفصام \emph{فوق الدقيق}، مع $\nu =
\qty{1420}{MHz}$ و $\lambda = \qty{21}{cm}$. والانتقال بطيء
بطئًا عبثيًا (انقلاب واحد كل عشرة ملايين سنة)، لكن المجرّة تحوي
$10^{66}$ ذرة هدروجين: \omterm{ex:b3:spin-two-level:hyperfine}{فخط 21 سنتيمترًا} ساطع بما يكفي لأن يكون قد
رسم الأذرع الحلزونية والتواء القرص، و--- عبر انزياحات
دوبلر فيه --- منحنيات الدوران المسطحة التي تحاجّ للمادة
المظلمة. والفيزياء نفسها، في الحالة الأساسية للسيزيوم، تفصم المستويين
بالمقدار $\qty{9192631770}{Hz}$ بالضبط: فمنذ 1967 صار \emph{تعريف
الثانية} انقلابَ \omterm{def:b3:spin-two-level:spin}{سبين} فوق دقيق معدودًا
(\cref{exo:b3:spin-two-level:12}).
\end{example}

\begin{method}[حرفة المستويين]\label{met:b3:spin-two-level:method}
(1) اكتب أيّ \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{هاميلتونية} ذات مستويين \omterm{def:b3:spin-two-level:spin}{بمصفوفات باولي}: $\hat H =
\epsilon_0\mathbb 1 + \vect h\cdot\hat{\vect\sigma}$ --- فتبادر
عندئذٍ متجهة بلوخ حول $\vect h$ بالتواتر $2|\vect h|/\hbar$؛
وكل ما عدا ذلك هندسة. (2) والحالات الذاتية على المنحى $\vect n$:
استعمل أنصاف الزوايا. (3) ومسائل الرنين: انتقل إلى المرجع الدائر؛
فعند الرنين لا يبقى غير $B_1$. (4) والنبضات: $\pi/2$ لبدء
إشارة مبادرة، و $\pi$ للقلب (وللإعادة إلى البؤرة --- أي صدى
\omterm{def:b3:spin-two-level:spin}{السبين}). (5) ومقاييس الطاقة: $\mu_{\text{B}}B$ للإلكترونات
($\qty{58}{\micro eV/T}$)، وأقلّ بثلاث رتب للنوى --- أي
تواترات راديوية، ومطيافية بلا كلفنات.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:1}
(أ) تحقق من $\sigma_x^2 = \sigma_y^2 = \sigma_z^2 = \mathbb 1$ ومن
$\sigma_x\sigma_y = \iu\sigma_z$. (ب) استنتج $[\hat S_x, \hat S_y]
= \iu\hbar\hat S_z$: أي جبر العزم الحركي في البعد الثاني.
(ج) بيّن أن $\hat S^2 = \tfrac34\hbar^2\,\mathbb 1$ وتحقق من $j(j+1)$
من أجل $j = \tfrac12$. (د) بيّن أن أيّ مصفوفة هرميتية $2\times2$
تركيب حقيقي من $\mathbb 1$ ومن المصفوفات الثلاث $\sigma_i$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:2}
(أ) أوجد الحالات الذاتية المنظَّمة للمصفوفتين $\sigma_x$ و $\sigma_y$.
(ب) \omterm{def:b3:spin-two-level:spin}{سبين} محضَّر في $\ket{+x}$ يُقاس على المحور $z$: فما احتمالات
النتائج؟ (ج) وعلى المحور $y$؟ (د) تحقق من الاثنين أمام صورة
بلوخ (فأيّ زاويتين $\theta, \varphi$؟).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:3}
ثلاثة مغانط شتيرن--غيرلاخ متسلسلة، محاورها $z$ ثم $\vect n$ بزاوية
$\theta$ في المستوي $xz$ ثم $z$ من جديد؛ وتُبقى الحزمة $+$
في كل مرة. (أ) احسب احتمال النجاة من المغنط الثاني. (ب)
واحتمال النجاة من الثلاثة. (ج) قدّرهما عند $\theta = 90^\circ$ و
قارن مع \cref{ex:b3:spin-two-level:chained}. (د) من أجل أيّ
$\theta$ تكون النجاة من المغانط الثلاثة أعظمية، وبمَ يذكّر الجواب
من \cref{exo:b3:quantum-formalism:5}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:4}
احسب انفصام المستويين $2|\mu|B$ وتواتر الرنين
في مجال \qty{1}{T} من أجل (أ) الإلكترون ($g = 2$)؛ (ب)
البروتون ($\mu_p = 2.79\,\mu_{\text{N}}$ مع $\mu_{\text{N}} =
e\hbar/2m_{\text{p}}$)؛ (ج) قارن كليهما مع $k_{\text{B}}T$ عند
\qty{300}{K}: فما مدى استقطاب سبينات الإلكترونات والنوى في درجة
حرارة الغرفة؟ (د) فأيّ نطاقات الطيف يسكنها إذًا الرنين
البارامغناطيسي الإلكتروني \omterm{prop:b3:spin-two-level:rabi}{والرنين المغناطيسي} النووي؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:5}
استنتج \omterm{prop:b3:spin-two-level:larmor}{مبادرة لارمور}: مع $\hat H = -\gamma B_0\hat S_z$، (أ)
احسب $\dd\langle\hat S_x\rangle/\dd t$ و $\dd\langle\hat
S_y\rangle/\dd t$ و $\dd\langle\hat S_z\rangle/\dd t$ انطلاقًا من
المبدّلات؛ (ب) حلّ من أجل $\langle\hat{\vect S}\rangle(t)$ مع
\omterm{def:b3:spin-two-level:spin}{سبين} ابتدائي على المحور $x$؛ (ج) بيّن أن $\|\langle\hat{\vect S}\rangle\|$
والزاوية إلى $\vect B_0$ ثابتان؛ (د) لماذا تصف صورة
\emph{الحالات الذاتية للطاقة} (أي مستويان مستقران) وصورة
\emph{المبادرة} (أي متجهة تدور) الفيزياء نفسها ---
فأيّ حالة تبادر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:6}
(أ) تحقق من أن $\ket{+\vect n}$ في
\cref{prop:b3:spin-two-level:bloch} هي المتجهة الذاتية المزعومة. (ب)
بيّن أن الحالتين في الطرفين المتقابلين متعامدتان. (ج) أين تقع على الكرة
الحالات الذاتية للمصفوفتين $\sigma_x$ و $\sigma_y$؟ (د) بوابة ``النفي''
في الكيوبت تنقل $\ket\uparrow \leftrightarrow \ket\downarrow$: فأيّ
دوران للكرة هي، وأيّ نبضة من
\cref{prop:b3:spin-two-level:rabi} تؤديها؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:7}
المرجع الدائر بأمانة. مع $\hat H(t) = -\gamma(B_0\hat S_z
+ B_1[\hat S_x\cos\omega t - \hat S_y\sin\omega t])$: (أ) اشرح
لماذا يستبدل الانتقال إلى المرجع الدائر عند $\omega$ حول $z$ بالمقدار
$B_0$ المقدارَ $B_0 - \omega/\gamma$ ويجمّد $B_1$؛ (ب) عند الرنين،
صف حركة متجهة بلوخ في المرجع الدائر؛ (ج)
استنتج مدة نبضة $\pi$ من أجل بروتون مع $B_1 =
\qty{10}{\micro T}$؛ (د) ارسم تقريبيًا ما يراه راصد في مرجع المخبر
في أثناء تلك النبضة (أي دورانان متداخلان).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:8}
\omterm{ex:b3:spin-two-level:hyperfine}{خط 21 سنتيمترًا}. (أ) تحقق من أن $\Delta E = \qty{5.9}{\micro eV}$
يعطي $\nu = \qty{1420}{MHz}$ و $\lambda = \qty{21.1}{cm}$. (ب)
تعيش الحالة المثارة $\qty{e7}{yr}$: فأيّ عرض خط هذا،
ولماذا تكون العروض المشاهَدة ($\sim$ كيلوهرتز فما فوق) دوبلرية صرفة؟ (ج)
تبتعد حافة قرص مجرّة بسرعة \qty{220}{km/s} بالنسبة إلى
مركزها: احسب مدى التواتر في جانبية \omterm{ex:b3:spin-two-level:hyperfine}{خط 21 سنتيمترًا} فيها. (د) لماذا
يتقفّى \omterm{ex:b3:spin-two-level:hyperfine}{خط 21 سنتيمترًا} الغازَ \emph{الذرّي المتعادل}، بينما يتقفّى أحادي
أكسيد الكربون (\cref{pb:b3:quantum-angular-momentum:1}) الغازَ
\emph{الجزيئي} --- ولماذا يحتاج الفلكيون إليهما معًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:9}
تنفق ذرة النتروجين في جزيء النشادر عبر مستوي H$_3$:
فيمتزج التشكيلان ``المظلّيان'' في حالتين متناظرة ومتخالفة
التناظر ينفصمان بالمقدار $\Delta E = \qty{e-4}{eV}$ (وهو ثنائي
الانقلاب في كتاب السنة الثانية). (أ) برّر معاملة
النشادر جملةً ذات مستويين. (ب) احسب تواتر الانتقال و
طول موجته. (ج) في الميزر (1954)، تضخّم الجزيئات المنتقاة بحالتها
والعابرة تجويفًا رنينيًا تلك الموجة الميكروية: فأيّ شرط على
الجمهرة يجب أن تحققه الحزمة الداخلة، وكيف يختلف عن
الشرط الحراري؟ (د) بمَ برهن ميزر النشادر قبل ليزر
كتاب السنة الثانية الضوئي بثلاث سنوات؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:10}
التقارن السبيني--المداري، مقدَّرًا. في مرجع سكون الإلكترون
تدور النواة حوله: فيجلس الإلكترون في مجال مغناطيسي $B_{\text
{int}}$. (أ) انطلاقًا من \cref{exo:b3:hydrogen-atom:10}(د)، خذ
$B_{\text{int}} \sim \qty{10}{T}$ من أجل الهدروجين عند $n = 2$ وقدّر
انزياح المستوي $\mu_{\text{B}}B_{\text{int}}$: وقارنه
بمقياس البنية الدقيقة في \cref{exo:b3:hydrogen-atom:11}. (ب) ينفصم
خط الصوديوم D بالمقدار \qty{0.6}{nm} عند \qty{589}{nm}: حوّله إلى ميلي إلكترون-فولط
وإلى مجال داخلي. (ج) لماذا ينمو الانفصام نموًا حادًّا
مع $Z$ (فالمجال الداخلي يتغير تقريبًا وفق $Z^4$ على $n^3$)؟
(د) تُوسم الحالات بالعدد الكلي $j = l \pm \tfrac12$: أحصِ
حالات مستوٍ $p$ وتحقق من أن شيئًا لم يضع.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:11}
سبينان معًا. من أجل جسيمين \omterm{def:b3:spin-two-level:spin}{سبين} كل منهما $\tfrac12$، يكون \omterm{def:b3:spin-two-level:spin}{السبين}
الكلي $\hat{\vect S} = \hat{\vect S}_1 + \hat{\vect S}_2$. (أ) بيّن
أن حالات الجداء الأربع تعيد تنظيمها في \emph{ثلاثي} ($S = 1$:
$\ket{\uparrow\uparrow}$ و $(\ket{\uparrow\downarrow} +
\ket{\downarrow\uparrow})/\sqrt2$ و $\ket{\downarrow\downarrow}$)
و\emph{أحادي} ($S = 0$: $(\ket{\uparrow\downarrow} -
\ket{\downarrow\uparrow})/\sqrt2$) --- وتحقق من قيم $S_z$ و،
من أجل الحالتين الوسطيتين، من أثر $\hat S^2$ (واستعمل $\hat
S^2 = \hat S_1^2 + \hat S_2^2 + 2\hat{\vect S}_1\cdot\hat{\vect
S}_2$ و $\hat{\vect S}_1\cdot\hat{\vect S}_2 = \tfrac12(\hat
S_{1+}\hat S_{2-} + \hat S_{1-}\hat S_{2+}) + \hat S_{1z}\hat
S_{2z}$). (ب) أيّ الحالات الأربع متشابكة --- أي لا تُكتب
جداءً؟ (ج) زوج الهدروجين فوق الدقيق
(\cref{ex:b3:spin-two-level:hyperfine}) هو بالضبط هذا
الثلاثي والأحادي: فأيّهما أعلى طاقة، ما دام الخط
\emph{يُصدَر} عند 21 سنتيمترًا؟ (د) سبينا الأحادي متضادّان تمامًا
على \emph{كل} محور: فقِس أحدهما على أيّ
$\vect n$ يجب الآخر بالعكس. فلماذا لا ينقل هذا
الترابط، على قوته، أيّ إشارة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:12}
الساعة التي تعرّف الثانية. الانفصام فوق الدقيق للحالة الأساسية
في السيزيوم هو $\qty{9192631770}{Hz}$ بالضبط (بتعريف
الثانية). وفي ساعة النافورة، تنال الذرات نبضة $\pi/2$، وتطير بحرية
مدة $T = \qty{0.5}{s}$، ثم تنال نبضة $\pi/2$ ثانية؛ و
تتذبذب النسبة المنتقلة وفق $\cos^2(\pi\,\delta\,T)$ بانحراف
$\delta$ (وهي أهداب رامزي --- اقبلها). (أ) اشرح
بلغة بلوخ ما تفعله كل نبضة $\pi/2$ وما يفعله الطيران الحر. (ب)
احسب عرض الهدب المركزي. (ج) وبنسبة إشارة إلى ضجيج تتيح
شطر مركز الهدب $10^4$ مرة، فأيّ دقة تواترية نسبية تنتج؟
(د) لماذا تتفوق الساعات الضوئية (بانتقالات بالبيتاهرتز) على
ساعات الميكروويف عند قدرة شطر أهداب ثابتة --- وبأيّ عامل تقريبًا؟
\end{exercise}

\section{مسألة: الرؤية داخل الجسم}

\begin{problem}[{مسألة نهاية الأسبوع --- التصوير بالرنين المغناطيسي،
من السبين إلى الصورة}]\label{pb:b3:spin-two-level:1}
ثلثا الإنسان ماء؛ وكل جزيء ماء يحمل بروتونين، كل منهما
مغنط سبينه $\tfrac12$. فضع إنسانًا في مجال قوي، ودغدغ
البروتونات عند تواتر لارمور، وأنصت: ذاك هو
التصوير \omterm{prop:b3:spin-two-level:rabi}{بالرنين المغناطيسي}، أي فيزياء \omterm{def:b3:spin-two-level:spin}{السبين} طبًّا. المعطيات:
$\gamma_{\text{p}}/2\pi = \qty{42.58}{MHz/T}$؛ و $B_0 = \qty{3.0}{T}$؛
وكثافة بروتونات النسيج $n \approx \qty{6.6e28}{m^{-3}}$؛
و $k_{\text{B}}T$ عند \qty{310}{K} يساوي \qty{26.7}{meV}؛ و $h =
\qty{4.14e-15}{eV.s}$.

\medskip\noindent\textbf{الجزء الأول --- السبينات في المغنط.}
\begin{enumerate}
  \item احسب تواتر لارمور عند \qty{3.0}{T} وطاقة الفوتون
        $h\nu$ بالإلكترون-فولط.
  \item احسب اختلال التوازن بين جمهرتَي مستويَي البروتون،
        $\Delta n/n \approx h\nu/2k_{\text{B}}T$.
  \item لا يسهم في الإشارة غير هذا الفائض --- وهو أجزاء من
        المليون: فكم بروتونًا ``نافعًا'' في السنتيمتر المكعب من
        النسيج؟
  \item لماذا يكون الاستقطاب الحراري بهذا الوهن هنا في حين انشطرت
        حزمة شتيرن--غيرلاخ انشطارًا تامًّا؟ (فما الذي تقيسه كل
        تجربة: قيمًا ذاتية لذرة مفردة، أم متوسطًا حراريًّا؟)
  \item ماذا تفعل مضاعفة $B_0$ بالإشارة (وهما أثران:
        الاستقطاب والقوة المحركة المحرَّضة عند تواتر أعلى)؟ ولماذا
        تدفع المستشفيات ثمن مغانط أكبر؟
  \item المغنط فائق الناقلية ومشتغل دائمًا: فلماذا تكون قارورة
        أكسجين فولاذية طليقة في الغرفة قذيفةً قاتلة
        (فأيّ قوة من \cref{ex:b3:spin-two-level:stern-gerlach}
        تؤثر فيها)؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- النبضات والأصداء.}
\begin{enumerate}[resume]
  \item تطبّق وشيعة مستعرضة $B_1 = \qty{10}{\micro T}$ عند
        الرنين: احسب تواتر رابي ومدتَي
        نبضتَي $\pi/2$ و $\pi$.
  \item بعد نبضة $\pi/2$، ماذا تفعل المغنطة،
        وماذا يحرّض قانون فاراداي في وشيعة الاستقبال (وعند
        أيّ تواتر)؟
  \item تضمحل الإشارة المستعرضة بثابت زمن $T_2$
        (إذ تتباعد أطوار السبينات في مجالات بعضها بعضًا وفي
        اللاتجانسات الموضعية)؛ وتنمو المغنطة الطولية من جديد
        بالثابت $T_1$ (إذ تسيل الطاقة إلى النسيج). والقيم النموذجية:
        $T_1 \approx \qty{1}{s}$ و $T_2 \approx \qty{0.1}{s}$. فلماذا
        لا يمكن لزمن الترابط أن يتجاوز زمن استرخاء الطاقة
        بكثير (فما الذي يفعله كل انقلاب متبادِل للطاقة
        بالطور)؟
  \item صدى \omterm{def:b3:spin-two-level:spin}{السبين}: بعد نبضة $\pi/2$ ومهلة $\tau$،
        تُطبَّق نبضة $\pi$، وعند $2\tau$ تعود السبينات المتباعدة
        الأطوار إلى الاصطفاف وتعود الإشارة. اشرح الحيلة
        بعدّائين على مضمار يُطلب إليهم أن يستديروا.
  \item أيّ اضمحلال يبطله الصدى --- التباعد الطوري الناتج عن
        اللاتجانسات \emph{الساكنة}، أم عن المجالات الجزيئية
        المتذبذبة --- ولماذا ذاك وحده؟
  \item للأنسجة المختلفة قيم $T_1$ و $T_2$ مختلفة (فالشحم:
        $T_1$ قصير؛ والماء والسوائل: طويل؛ وكثير من الأورام: $T_2$
        أطول من نسيجها المضيف). في جملة واحدة: كيف يحوّل توقيت
        متتالية النبضات أزمنةَ الاسترخاء إلى \emph{تباين} في
        الصورة؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- من الإشارة إلى الصورة.}
\begin{enumerate}[resume]
  \item راكِب تدرجًا $G = \qty{40}{mT/m}$ على المحور $z$: فيصير
        تواتر لارمور متعلقًا بالموضع. احسب
        $\dd\nu/\dd z$ بالكيلوهرتز في الملّيمتر.
  \item نبضة راديوية مشكَّلة لا تحوي غير النطاق $\nu_0 \pm
        \qty{1}{kHz}$ تثير أيّ شريحة من الجسم؟ (أي انتقاء
        الشريحة.)
  \item في أثناء القراءة، يجعل تدرج على المحور $x$ كل عمود من
        الشريحة يبثّ تواتره الخاص: فأيّ عملية رياضية
        تحوّل الإشارة الزمنية المستقبَلة إلى جانبية فضائية
        (تذكّر عدّة فورييه في كتاب السنة الثانية)؟
  \item قدّر مجموع معلومات الفحص: صورة $256 \times 256$
        بعمق 12 بتًا، ولماذا يجعل اكتسابها سطرًا سطرًا
        (صدى واحد لكل سطر، وزمن تكرار $\sim T_1$) الفحصَ
        يستغرق دقائق.
  \item يسمع المريض طرقًا عاليًا: فما الذي يطرق
        ميكانيكيًا (فكّر في وشائع التدرج وهي تُقلَّب في المجال
        \qty{3}{T} --- فأيّ قوة)؟
  \item يباين التصوير بالأشعة السينية كثافة الإلكترونات؛ ويباين \omterm{prop:b3:spin-two-level:rabi}{الرنين
        المغناطيسي} كثافة البروتونات \emph{وأزمنة} الاسترخاء: فلماذا يكون
        الرنين أداة الاختيار للأنسجة الرخوة والدماغ، وأيّ جرعة
        مؤيِّنة يوصلها؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- أعمال \omterm{def:b3:spin-two-level:spin}{السبين} الأخرى.}
\begin{enumerate}[resume]
  \item يجري الكيميائيون التجربة نفسها بتمييز \qty{10}{ppm}:
        فسحب الإلكترونات تحجب كل نواة قليلًا،
        فتزيح رنينها (وهو ``الانزياح الكيميائي''). فلماذا يحوّل
        هذا الرنين النوويّ إلى بصمة جزيئية؟
  \item يرسم \omterm{prop:b3:spin-two-level:rabi}{الرنين المغناطيسي} الوظيفي التفكير: فالهيموغلوبين
        منزوع الأكسجين بارامغناطيسي ويقصّر $T_2^*$ الموضعي.
        تقصَّ السلسلة من النشاط العصبي إلى سطوع الصورة.
  \item فيزياء لارمور نفسها بعزم \emph{الإلكترون} تعمل
        عند \qty{84}{GHz} في \qty{3}{T}: فلماذا يكون رنين
        الإلكترون عديم الفائدة داخل إنسان لكنه ثمين لدراسة
        الجذور الحرة والعيوب؟
  \item قارن طاقة فوتون البند 1 بطاقات الروابط
        الكيميائية النموذجية: وبرّر وصف ``غير المؤيِّن''،
        وقابله بفوتون أشعة سينية طاقته \qty{60}{keV}.
  \item يمكن رفع الاستقطاب البالغ عشرة أجزاء من المليون:
        فغاز الزينون المستقطَب بالليزر يبلغ استقطابًا من رتبة الواحد ويُستنشق
        لتصوير أجواف الرئة. فبأيّ عامل تقريبًا ترتفع الإشارة
        لكل نواة، ولماذا تبقى كثافة الغاز المنخفضة تجعل
        الحيلة ضرورية؟
  \item يُقرأ اليوم \omterm{def:b3:spin-two-level:spin}{سبين} إلكترون مفرد في مصيدة شبيهة بترنزستور
        سيليكوني ويُقاد كيوبتًا بنبضات هذا الفصل
        بالضبط: سمِّ خاصيتين من خواصّ \omterm{def:b3:spin-two-level:spin}{السبين} (حجم
        \omterm{def:b3:quantum-formalism:state}{فضاء هيلبرت} الخاص به؛ وضعف تقارنه مع الضجيج) توصيان به.
  \item لخّص النتيجة الموسومة: مبادرة سبينية عند \qty{128}{MHz}،
        واستقطاب حراري بعشرة أجزاء من المليون،
        وساعتا استرخاء، وثلاثة تدرجات مجال تكفي
        لتصوير دماغ حيّ شرائح --- أي بقعتا شتيرن وغيرلاخ،
        وقد صارتا مستشفى.
\end{enumerate}
\end{problem}
