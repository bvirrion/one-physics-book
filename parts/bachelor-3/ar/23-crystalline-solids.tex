\section{الشبكات والخلايا والبنى}

\begin{definition}[الشبكة والأساس]\label{def:b3:crystalline-solids:lattice}
\index{بلورة}\index{شبكة}\index{أساس (علم البلورات)}\index{خلية الوحدة}
\emph{البلورة} ترتيب متكرر: أي \emph{شبكة} (وهي
شبكة النقاط الرياضية $\vect r = n_1\vect a_1 + n_2\vect a_2
+ n_3\vect a_3$، مع $n_i$ أعدادًا صحيحة) مزيَّنة بما يسمّى \emph{أساسًا} (وهو
الذرة أو المجموعة المعلَّقة تعليقًا متطابقًا على كل نقطة). ومتوازي السطوح
الذي تمدّه $\vect a_1, \vect a_2, \vect a_3$ هو \emph{خلية وحدة}؛
وحرفه $a$ هو \emph{ثابت الشبكة}، وهو بضعة
\unit{\angstrom}. وثلاثة أكداس مكعبية تحمل أكثر هذا المقرّر:
\emph{المكعبي البسيط} (بنقاط عند زوايا المكعب --- وهو نادر: بذرة واحدة لكل
خلية)؛ و\emph{المكعبي المركزي الجسم} (BCC: زوايا $+$ مركز، بذرتين
لكل خلية --- الحديد والكروم)؛ و\emph{المكعبي المركزي الوجوه} (FCC:
زوايا $+$ مراكز أوجه، بأربع ذرات لكل خلية --- النحاس والألومنيوم و
الفضة والذهب: وهو أكثف رصّ مكعبي، يملأ 74\,\% من الفضاء).
والألماس FCC بأساس من ذرتين؛ وملح الطعام FCC بزوج من الصوديوم
والكلور.
\end{definition}

\begin{example}[وزن خلية وحدة]\label{ex:b3:crystalline-solids:density}
النحاس FCC مع $a = \qty{3.61}{\angstrom}$. وخليته تحمل 4
ذرات كتلتها المولية \qty{63.5}{g/mol}، ومنه
\[
\rho = \frac{4M/N_{\text{A}}}{a^3}
= \frac{4\times\qty{1.055e-25}{kg}}{\qty{4.71e-29}{m^3}}
\approx \qty{8960}{kg/m^3} ,
\]
وهي قيمة الكتيّب. ومجرًى إلى الوراء، يحوّل الحساب نفسه
$a$ مقيسًا وكثافةً على الطاولة إلى $N_{\text{A}}$ --- أي
الطريق الذي عدّت به الأشعة السينية الذرات أول مرة
(\cref{exo:b3:crystalline-solids:6}).
\end{example}

\section{رؤية الكدس: انعراج الأشعة السينية}

\begin{proposition}[قانون براغ]\label{prop:b3:crystalline-solids:bragg}
\index{قانون براغ}\index{انعراج الأشعة السينية}\index{دلائل ميلر}
تحوي \omterm{def:b3:crystalline-solids:lattice}{البلورة} عائلات من المستويات الذرّية المتوازية. وحزمة
أشعة سينية أحادية اللون طولها الموجي $\lambda$، تصيب عائلة
تباعدها $d$ بزاوية مماسّة $\theta$، لا تنعكس انعكاسًا قويًّا إلا
حين تخطو الأمواج من المستويات المتعاقبة على الطور نفسه:
\[
2d\sin\theta = n\lambda , \qquad n = 1, 2, \dots
\]
وفي \omterm{def:b3:crystalline-solids:lattice}{البلورة} المكعبية تكون العائلة التي \emph{\omterm{prop:b3:crystalline-solids:bragg}{دلائل ميلر}} لها
$(hkl)$ --- أي مستويات تقطع حروف الخلية إلى $h$ و $k$ و $l$ جزءًا
--- بتباعد $d = a/\sqrt{h^2 + k^2 + l^2}$، ومنه يبصم نمط
زوايا الانعكاس نوعَ \omterm{def:b3:crystalline-solids:lattice}{الشبكة} وثابتها معًا:
أي علم البلورات في صيغتين.
\end{proposition}

\begin{proof}
الشعاع المنعكس عن المستوي الأسفل يقطع مسيرًا زائدًا $2d
\sin\theta$ (وهما ساقان مماسّتان للمثلث القائم الذي عمقه $d$).
ويقتضي التداخل البنّاء أن يكون عددًا صحيحًا من
الأطوال الموجية. وأما صيغة التباعد للعائلات المكعبية $(hkl)$ فهي
هندسة مستوية: إذ تقطع المستويات المتعاقبة حرف المكعب $a$ عند
فترات $a/h$ على $x$، وهكذا، فتعطي $d$ المذكورة.
\end{proof}

\section{ما الذي يماسكها}

\begin{proposition}[التماسك الأيوني: مجموع مادلونغ]\label{prop:b3:crystalline-solids:madelung}
\index{طاقة التماسك}\index{ثابت مادلونغ}
\emph{\omterm{prop:b3:crystalline-solids:madelung}{طاقة التماسك}} هي ما يكلّف تفكيك \omterm{def:b3:crystalline-solids:lattice}{بلورة}
إلى ذرات (أو أيونات) بعيدة. وفي ملح الطعام يجلس كل أيون شحنته
$\pm e$ بين جيران متناوبين؛ وجمع طاقة كولوم لكل
\omterm{def:b3:crystalline-solids:lattice}{الشبكة} لكل زوج أيونات يعطي
\[
U = -\alpha\,\frac{e^2}{4\pi\varepsilon_0 r_0} ,
\qquad \alpha_{\text{NaCl}} = 1.7476 ,
\]
مع $r_0$ مسافة أقرب جار و $\alpha$
\emph{\omterm{prop:b3:crystalline-solids:madelung}{ثابت مادلونغ}} --- أي هندسة \omterm{def:b3:crystalline-solids:lattice}{البلورة} كلها
مضغوطة في عدد واحد (ويجب جمع المتسلسلة في قشور
متعادلة متمددة؛ فهي متباعدة حدًّا بحدّ). ومع $r_0 =
\qty{2.82}{\angstrom}$ يعطي هذا \qty{8.9}{eV} لكل زوج؛ و
يردّ التنافر الكمومي في اللبّ الصلب $\sim12\,\%$، فتقع النتيجة في حدود
بضعة بالمئة من القيمة المقيسة \qty{7.9}{eV}
(\cref{exo:b3:crystalline-solids:7}).
\end{proposition}
\admitted

\begin{remark}[رباعي الروابط]\label{rem:b3:crystalline-solids:bonds}
\index{رابطة فان دير فالس}\index{رابطة تساهمية}\index{رابطة فلزية}
أربعة أصماغ تبني كل البلورات. \emph{الأيونية} (NaCl): بنقل
إلكترون، ثم كهرباء مادلونغ الساكنة --- فتكون عوازل صلبة قصفة
شفافة. و\emph{التساهمية} (الألماس والسيليكون): بأزواج مشتركة
في الروابط الاتجاهية في
\cref{ch:b3:identical-particles} --- وهي أصلب الكل.
و\emph{الفلزية} (النحاس): أيونات مغمورة في إلكترونات غير موضعية ---
وهي موضوع الفصل التالي، ومطواعة لأن الصمغ لا يعبأ
بأيّ أيون في أيّ مكان. و\emph{فان دير فالس} (الأرغون الصلب والبلورات
الجزيئية): بجذب ثنائيات أقطاب متذبذبة $\sim r^{-6}$ في مقابل
تنافر كمومي في اللبّ الصلب، مغلَّفًا في كمون
لينارد--جونز $U = 4\epsilon[(\sigma/r)^{12} - (\sigma/r)^6]$ --- وهي ضعيفة
($\epsilon \sim \qty{0.01}{eV}$)، ولذلك تنصهر الأجسام الصلبة من الغازات النبيلة
عند عشرات الكلفنات فوق الصفر المطلق
(\cref{exo:b3:crystalline-solids:8}).
\end{remark}

\section{الكدس يهتزّ: التبدد}

\begin{theorem}[تبدد السلسلة أحادية الذرة]\label{thm:b3:crystalline-solids:dispersion}
\index{علاقة التبدد (الشبكة)}\index{الفرع الصوتي}
نمذج صفًّا بلوريًّا بكتل $m$ متباعدة بمقدار $a$، موصولة بنوابض
$K$ (وهي صلابة الرابطة --- أي تحدّب الأشكال في الأعلى).
والبحث عن أمواج $u_n = A\eu^{\iu(kna - \omega t)}$ في قانون نيوتن
$m\ddot u_n = K(u_{n+1} + u_{n-1} - 2u_n)$ يعطي
\[
\omega(k) = 2\sqrt{\frac{K}{m}}\,\Big|\sin\frac{ka}{2}\Big| .
\]
وعند الطول الموجي الطويل ($ka \ll 1$) يكون هذا صوتًا، $\omega = vk$ مع
$v = a\sqrt{K/m}$ --- أي كيلومترات في الثانية من نوابض ذرّية.
لكن المنحني \emph{ينثني}: فعند حافة المنطقة $k = \pi/a$
(بطول موجي $2a$، والجيران في طور متعاكس) تنعدم سرعة المجموعة
--- إذ تنعكس الموجة انعكاس براغ عن \omterm{def:b3:crystalline-solids:lattice}{الشبكة} نفسها التي
تحملها --- ولا ينتشر أيّ تواتر أعلى البتة: \omterm{def:b3:crystalline-solids:lattice}{فالبلورة}
مرشّح تمرير منخفض بقطع عند التيراهرتز. وهذه الأمواج المكمَّمة هي
الفونونات التي أحصت \cref{ch:b3:photons-phonons} إحصاءها سلفًا؛
والجزء الخطي من هذا المنحني هو بالضبط ما أبقاه \omterm{thm:b3:photons-phonons:debye}{نموذج
ديباي}.
\end{theorem}

\begin{proof}
بتعويض الموجة في معادلة الحركة: $-m\omega^2 =
K(\eu^{\iu ka} + \eu^{-\iu ka} - 2) = -2K(1 - \cos ka) =
-4K\sin^2(ka/2)$.
\end{proof}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:crystalline-solids:1}
مسك دفاتر الخلية. (أ) تحقق من أعداد الذرات: المكعبي البسيط واحدة، والمركزي الجسم اثنتان، والمركزي الوجوه أربع.
(ب) وفي BCC تتلامس الذرات على قطر المكعب: عبّر عن نصف القطر
الذرّي بدلالة $a$. (ج) والمثل من أجل FCC، حيث تتلامس على
قطر وجه. (د) واحسب نسبتَي الرصّ
($\pi\sqrt3/8 \approx 0.68$ و $\pi/\sqrt{18} \approx 0.74$) و
اذكر أيّ بنية تفضّلها الفلزات حين يكون الربط
أعمى عن الاتجاه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:crystalline-solids:2}
النحاس بالأعداد. FCC مع $a = \qty{3.61}{\angstrom}$ و $M =
\qty{63.5}{g/mol}$. احسب (أ) مسافة أقرب جار؛
(ب) عدد أقرب الجيران؛ (ج) الكثافة؛ (د)
عدد الذرات في مكعب من سلك وصل ضلعه \qty{1}{\micro m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:crystalline-solids:3}
زوايا براغ الأولى. أشعة سينية من نحاس K$\alpha$
($\lambda = \qty{1.54}{\angstrom}$) تصيب مستويات NaCl تباعدها
$d = \qty{2.82}{\angstrom}$. (أ) جد زاوية المرتبة الأولى
$\theta_1$. (ب) وكم مرتبةً توجد؟ (ج) ولماذا يجب أن يكون
$\lambda \le 2d$ من أجل أيّ انعكاس --- ولماذا لا أمل في الضوء
المرئي؟ (د) وماذا يصيب $\theta_1$ إذا سُخّنت \omterm{def:b3:crystalline-solids:lattice}{البلورة}
حتى يتمدد $d$ بمقدار 1\,\%؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:crystalline-solids:4}

\begin{exercise}[$\star\star$]\label{exo:b3:crystalline-solids:5}

\begin{exercise}[$\star\star$]\label{exo:b3:crystalline-solids:7}

\begin{exercise}[$\star\star$]\label{exo:b3:crystalline-solids:8}

\begin{exercise}[$\star\star\star$]\label{exo:b3:crystalline-solids:10}

\section{مسألة: برادة المتحف}

\begin{problem}\label{pb:b3:crystalline-solids:1}
\emph{برادة المتحف.} متحف بحري ينتشل سبيكة متآكلة
من حطام سفينة من القرن الثامن عشر ويرسل إلى مخبرك
بضعة ميليغرامات من البرادة: معدن نفيس، لكن أيّها؟ ويستعمل مقياس
الانعراج المسحوقي عندك أشعة سينية من نحاس K$\alpha$،
$\lambda = \qty{1.5406}{\angstrom}$. والكاشف، وهو يمسح زاوية
الانحراف $2\theta$، يجد حلقات قوية عند $2\theta =
38.1^\circ$ و $44.3^\circ$ و $64.4^\circ$ و $77.5^\circ$.

\medskip\noindent\textbf{الجزء الأول --- قراءة الحلقات.}

\begin{enumerate}
  \item اذكر \omterm{prop:b3:crystalline-solids:bragg}{قانون براغ} واشرح، في جملة واحدة لكلٍّ، دورَي
        الضوء أحادي اللون وحبيبات المسحوق العشوائية
        التوجّه.
  \item حوّل الحلقات الأربع إلى $\theta$ واحسب
        $\sin^2\theta$ لكل منها.
  \item ومن أجل \omterm{def:b3:crystalline-solids:lattice}{بلورة} مكعبية، بيّن أن \omterm{prop:b3:crystalline-solids:bragg}{قانون براغ} يعطي
        $\sin^2\theta = (\lambda^2/4a^2)(h^2 + k^2 + l^2)$.
  \item واقسم قيم $\sin^2\theta$ الأربع على أصغرها:
        بيّن أن النسب قريبة من $1 : 1.33 : 2.67 : 3.67$،
        أي $3 : 4 : 8 : 11$ على 3.
  \item وبلورات FCC لا تعكس إلا حين تكون $h, k, l$ كلها زوجية أو
        كلها فردية: تحقق من أن $(111)$ و $(200)$ و $(220)$ و $(311)$ ---
        ومجاميعها 3 و 4 و 8 و 11 --- تلائم، وأن المجاميع الغائبة
        (1 و 2 و 5 و 6 و 7) تؤكد FCC في مقابل المكعبي البسيط.
  \item ولماذا يُظهر فلز BCC (بمجاميع مسموحة 2 و 4 و 6 و 8 و\dots)
        بصمةً مختلفة --- ولماذا تكون لعبة الزوجية هذه، لا
        الزوايا المطلقة، هي المعرّف المتين؟
  \item واشرح لماذا تحدّ الحلقات كلما كبرت الحبيبات: فماذا تفعل
        بلوريّة عرضها خمس ذرات بانعكاس براغ؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- تسمية المعدن.}

\begin{enumerate}[resume]
  \item من كل حلقة، احسب \omterm{def:b3:crystalline-solids:lattice}{ثابت الشبكة} $a$ عبر
        $a = \lambda\sqrt{h^2+k^2+l^2}/2\sin\theta$.
  \item وسّط قيمك الأربع وأعطِ $a$ بثلاثة
        أرقام معنوية.
  \item ومع 4 ذرات لكل خلية FCC، عبّر عن الكثافة بدلالة
        $a$ والكتلة المولية $M$.
  \item والمرشحون النفيسون من FCC: الفضة
        ($M = \qty{107.9}{g/mol}$، $\rho = \qty{10.5}{g/cm^3}$)، و
        الذهب ($197.0$، $19.3$)، والبلاتين ($195.1$، $21.4$).
        احسب الكثافة التي يتنبأ بها $a$ عندك من أجل كل $M$
        مرشَّح وعيّن السبيكة.
  \item وثابت \omterm{def:b3:crystalline-solids:lattice}{شبكة} الذهب $\qty{4.08}{\angstrom}$ ---
        ويكاد يطابق ثابت الفضة. فأيّ قياس واحد في
        هذه المسألة يفصل بينهما مع ذلك، ولماذا يكون محصَّنًا
        من المصادفة؟
  \item ويطلب المتحف فحصًا غير متلف للسبيكة كلها.
        اقترح واحدًا (فالكثافة بأرخميدس تصلح) و
        وفّق بينه وبين جوابك المجهري.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- \omterm{def:b3:crystalline-solids:lattice}{البلورة} في حركة.}

\begin{enumerate}[resume]
  \item الفضة الدافئة تهتزّ: فتخشخش كل ذرة حول موقعها
        بمقدار $\langle x^2\rangle \propto T$ (بالتوزّع المتساوي على
        نوابض الروابط). فلماذا \emph{لا} يعرّض هذا حلقات براغ؟
  \item لكنه \emph{يضعفها}: فالأمواج المنتثرة عن الذرات المزاحة
        تفقد التزامنها. اذكر نوعيًّا كيف ينبغي أن تسلك
        الشدّة مع ارتفاع $T$ (وهو أثر
        ديباي--فالر).
  \item وقدّر نابض رابطة الفضة: $K \approx Ea$ مع
        $E = \qty{8.3e10}{Pa}$ و $a = \qty{2.89}{\angstrom}$
        بين أقرب جارين.
  \item ومع $m = \qty{1.79e-25}{kg}$، احسب مقياس سرعة
        الصوت $v = a\sqrt{K/m}$ وقارنه بقيمة الفضة المقيسة
        $\sim\qty{2700}{m/s}$.
  \item وقدّر الانزياح الحراري الجذري المتوسط عند \qty{300}{K}
        من $\tfrac12K\langle x^2\rangle \sim
        \tfrac32k_{\text{B}}T$ وقارنه بطول
        الرابطة: فأيّ نسبة هو؟
  \item وقاعدة لينديمان تصهر بلورةً حين تبلغ تلك النسبة
        $\sim10\,\%$: تحقق من الاتساق مع انصهار الفضة
        عند \qty{1235}{K}.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- ما وراء السبيكة.}

\begin{enumerate}[resume]
  \item الكروم BCC مع $a = \qty{2.88}{\angstrom}$: تنبّأ
        بالزاوية $2\theta$ لحلقته الأولى (العائلة (110)) تحت
        الأشعة السينية نفسها.
  \item ولماذا تعرج المجاهر الإلكترونية عن السطوح و
        الرقائق الرقيقة بينما تسبر الأشعة السينية والنيوترونات الكتلة؟ (جملة
        واحدة عن مدى قوة اقتران كلٍّ منها بالمادة.)
  \item وفي سنة 1952 كشفت صورة انعراج لليف مسحوب --- أي
        نمط من بقع ملطّخة على شكل X --- عن تكرار مقداره
        \qty{3.4}{\angstrom} مكدَّس على لولب: فأيّ جزيء،
        ولماذا نجح الانعراج حيث عجزت المجاهر؟
  \item وتصهر المسبكة قطعة اختبار: صف ما تصير إليه الحلقات
        الحادّة في نمط انعراج السائل، وما الذي
        يقوله ذلك عن الترتيب الذي يحفظه سائل.
  \item وشبه \omterm{def:b3:crystalline-solids:lattice}{البلورة} تعرج بقعًا حادّة بتناظر
        خماسي --- وهو مستحيل على أيّ \omterm{def:b3:crystalline-solids:lattice}{شبكة} متكررة. فماذا
        يشهد نمطها الحادّ به مع ذلك عن ترتيبها؟
  \item لخّص التعيين في ثلاثة أسطر: نسب الحلقات
        $\to$ FCC؛ و $a = \qty{4.09}{\angstrom}$ $+$ الكثافة $\to$
        الفضة؛ واهتزازات \omterm{def:b3:crystalline-solids:lattice}{البلورة} نفسها $\to$ لماذا لا ينبغي
        للمتحف أن يطلب منك تصويرها بالأشعة السينية وهي منصهرة.
\end{enumerate}
\end{problem}
