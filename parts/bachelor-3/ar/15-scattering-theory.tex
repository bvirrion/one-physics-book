\chapter{نظرية الانتثار}\label{ch:b3:scattering-theory}

كل ما نعرفه تقريبًا عن الصغير تعلّمناه برمي أشياء
عليه ومراقبة ما يرتدّ. فقد اكتشف رذرفورد النواة بإحصاء
ومضات جسيمات ألفا على شاشة من كبريتيد التوتياء؛ وبعد
خمسين سنة وجدت التجربة نفسها، مُجراةً بإلكترونات ذات طاقة أعلى،
كواركات داخل البروتون؛ وبينهما ومنذئذٍ، قُرئ كل \omterm{def:b3:scattering-theory:cross-section}{مقطع}
في الفيزياء النووية وفيزياء الجسيمات والذرات والمادة المكثفة
عبر نظرية هذا الفصل. واللغة هي
\emph{\omterm{def:b3:scattering-theory:cross-section}{المقطع}} --- أي مساحة هدف فعّالة؛
والغرض الكمومي هو \emph{\omterm{thm:b3:scattering-theory:amplitude}{سعة الانتثار}}، ويقول \omterm{thm:b3:scattering-theory:born}{تقريب
بورن} لها شيئًا لا يُنسى: أي إن هدفًا منتشرًا يُسبَر
في فضاء فورييه، زاويةً زاوية. وعند الطاقات المنخفضة تتولّى \omterm{thm:b3:scattering-theory:partial}{الأمواج
الجزئية}، فتطوي كل كمون معقد في عدد
واحد --- هو \omterm{thm:b3:scattering-theory:partial}{طول الانتثار} --- الذي يوالف اليوم الغازات الكمومية
فائقة البرودة؛ وحيث يستطيع القذيف أن يتلكأ في حالة شبه مقيدة،
يقفز \omterm{def:b3:scattering-theory:cross-section}{المقطع} رنينات، وهي مطيافية ما لا يُرى غير ذلك.

\section{المقاطع}

\begin{definition}[المقطع]\label{def:b3:scattering-theory:cross-section}
\index{مقطع}\index{مقطع تفاضلي}
أرسل حزمة منتظمة تدفقها $\Phi$ (أي جسيمات في وحدة المساحة في
الثانية) على جسيم هدف واحد. فمعدل القذائف المنتثرة
في الزاوية المجسمة $\dd\Omega$ حول المنحى $(\theta,
\varphi)$ يعرّف \emph{المقطع التفاضلي}،
\[
\dd\dot N = \Phi\,\frac{\dd\sigma}{\dd\Omega}\,\dd\Omega ,
\]
وتكامله $\sigma = \int(\dd\sigma/\dd\Omega)\,\dd\Omega$ هو
\emph{المقطع الكلي}: أي المساحة التي يعرضها الهدف فعليًا.
ومن أجل شريحة رقيقة فيها $n$ هدفًا في وحدة الحجم، تتوهّن حزمة
وفق $\eu^{-n\sigma x}$: فالمقاطع تُقاس
بإحصاء ما يخرج. والوحدات: تستعمل الفيزياء النووية \emph{البارن}،
$\qty{1}{b} = \qty{e-28}{m^2}$ --- وهو نحو الحجم الهندسي لنواة
يورانيوم، أي ``بحجم حظيرة'' بالنسبة إلى نترون.
\end{definition}

\section{السعة وتقريب بورن}

\begin{theorem}[حالات الانتثار والسعة]\label{thm:b3:scattering-theory:amplitude}
\index{سعة الانتثار}
من أجل جسيم طاقته $E = \hbar^2k^2/2m$ يلاقي كمونًا
موضعيًا $V(\vect r)$، تسلك حالة الانتثار المستقرة على المسافات
الكبيرة سلوك
\[
\psi(\vect r) \;\longrightarrow\;
\eu^{\iu kz} + f(\theta)\,\frac{\eu^{\iu kr}}{r} ,
\]
أي موجة مستوية واردة مضافًا إليها موجة كروية خارجة معدَّلة
بمقدار \emph{\omterm{thm:b3:scattering-theory:amplitude}{سعة الانتثار}} $f(\theta)$ (وهي طول)، مع
\[
\frac{\dd\sigma}{\dd\Omega} = |f(\theta)|^2 .
\]
\end{theorem}

\begin{proof}[برهان جزئي]
يجعل هبوط $1/r$ التدفقَ الخارج عبر أيّ كرة كبيرة
منتهيًا؛ وبحساب تيارَي الاحتمال
(\cref{prop:b3:schrodinger-three-dimensions:current}) للحدّين،
يكون التيار القطري الخارج في وحدة الزاوية المجسمة
$(\hbar k/m)|f|^2$ في مقابل التدفق الوارد $\hbar k/m$: والنسبة
هي تعريف $\dd\sigma/\dd\Omega$. أما وجود حلول بهذه
الصورة التقاربية فمُسلَّم به.
\end{proof}

\begin{theorem}[تقريب بورن]\label{thm:b3:scattering-theory:born}
\index{تقريب بورن}\index{انتقال كمية الحركة}
من أجل كمون ضعيف (أو قذيف سريع)، تعطي \omterm{thm:b3:perturbation-theory:corrections}{نظرية الاضطرابات}
في المرتبة الأولى في $V$
\[
f(\theta) = -\frac{m}{2\pi\hbar^2}
\int V(\vect r)\,\eu^{\iu\vect q\cdot\vect r}\,\dd^3r ,
\qquad
\vect q = \vect k - \vect k' , \quad
q = 2k\sin\frac\theta2 :
\]
أي إن السعة، إلى ثوابت، هي \emph{تحويل فورييه للكمون}
مقدَّرًا عند \omterm{thm:b3:scattering-theory:born}{انتقال كمية الحركة} $\hbar\vect q$.
فتجارب الانتثار محلّلات فورييه للمادة: إذ تقرأ الزوايا
الصغيرة الأطوال الموجية الطويلة (أي البنية الخشنة)، وتقرأ الزوايا الكبيرة
التفصيل الدقيق --- وهو أعمق سبب لكون ``طاقة أعلى'' تعني
``مسافات أصغر''.
\end{theorem}

\begin{proof}[برهان جزئي]
هذه هي \omterm{thm:b3:perturbation-theory:golden}{قاعدة فيرمي الذهبية}
(\cref{thm:b3:perturbation-theory:golden}) مطبَّقة بين موجتين مستويتين
$\ket{\vect k} \to \ket{\vect k'}$: فالعنصر المصفوفي هو
$\int V\eu^{\iu(\vect k - \vect k')\cdot\vect r}\dd^3r$، وتقدّم
\omterm{prop:b3:schrodinger-three-dimensions:counting}{كثافة الحالات} النهائية العوامل التحريكية، وتجمعها
المحاسبة (المُسلَّم بها) في $f$. وتعطي المرونة
$|\vect k'| = |\vect k|$، ومنها $q = 2k\sin(\theta/2)$ بمثلث
$\vect k, \vect k'$ المتساوي الساقين.
\end{proof}

\begin{example}[من يوكاوا إلى رذرفورد]\label{ex:b3:scattering-theory:rutherford}
\index{انتثار رذرفورد}
من أجل كمون كولوم المحجوب $V = \dfrac{Q_1Q_2}{4\pi
\varepsilon_0 r}\,\eu^{-r/a}$ يكون تكامل بورن أوليًا
(\cref{exo:b3:scattering-theory:5})، وعندما يؤول نصف قطر الحجب
$a \to \infty$،
\[
\frac{\dd\sigma}{\dd\Omega} =
\Big(\frac{Q_1Q_2}{16\pi\varepsilon_0 E}\Big)^2
\frac{1}{\sin^4(\theta/2)} :
\]
وهو \emph{\omterm{def:b3:scattering-theory:cross-section}{مقطع} رذرفورد}. وبمصادفة تخصّ
الكمون $1/r$ وحده، يتفق الحساب الكلاسيكي \omterm{thm:b3:scattering-theory:born}{وتقريب بورن}
والجواب الكمومي المضبوط جميعًا --- ولهذا نال رذرفورد،
وهو يحسب كلاسيكيًّا سنة 1911، الصيغة الصحيحة، ونال، من
التحقق منها ومضةً ومضة، النواةَ
(\cref{pb:b3:scattering-theory:1}).
\end{example}

\section{الأمواج الجزئية وطول الانتثار}

\begin{theorem}[انزياحات الطور]\label{thm:b3:scattering-theory:partial}
\index{أمواج جزئية}\index{انزياح الطور}\index{طول الانتثار}
من أجل كمون مركزي، حلّل حالة الانتثار على العزوم
الحركية: فتغادر كل موجة جزئية $l$ منطقةَ الكمون
واهتزازها القطري \emph{منزاح} بطور $\delta_l(k)$
--- إذ يقدّمه التجاذب ويؤخره التنافر --- وتعاد السعة
القابلة للرصد تركيبها في الصورة
\[
f(\theta) = \frac1k\sum_{l=0}^\infty(2l+1)\,\eu^{\iu\delta_l}
\sin\delta_l\,P_l(\cos\theta) , \qquad
\sigma = \frac{4\pi}{k^2}\sum_l(2l+1)\sin^2\delta_l .
\]
وعند الطاقات المنخفضة، يقول الحدس الكلاسيكي ($l \sim kb$ من أجل وسيط
الاصطدام $b$) إن $l = 0$ وحدها تنفذ: فيصير الانتثار
متماثل المناحي ويحكم عدد واحد،
\[
f \to -a , \qquad \sigma \to 4\pi a^2 ,
\]
حيث $a = -\lim_{k\to0}\delta_0/k$ هو \emph{\omterm{thm:b3:scattering-theory:partial}{طول
الانتثار}}. ومهما يكن تعقيده الداخلي، فإن الهدف عند الطاقة المنخفضة
\emph{هو} طول انتثاره --- وهو التبسيط العظيم الذي
تقوم عليه فيزياء الذرات الباردة.
\end{theorem}
\admitted

\begin{example}[نترون يلاقي بروتونًا]\label{ex:b3:scattering-theory:np}
تنتثر النترونات البطيئة عن البروتونات \omterm{def:b3:scattering-theory:cross-section}{بمقطع} $\sigma \approx
\qty{20.4}{b}$ --- وهي مساحة تفوق أربعين مرة الحجم الهندسي
للديوترون. والتفسير: أن التصادم عند الطاقة المنخفضة موجة $s$ صرفة
بقناتَي سبين، وطولا انتثارهما المقيسان،
$a_{\text{t}} = \qty{5.4}{fm}$ (للثلاثي) و $a_{\text{s}} =
\qty{-23.7}{fm}$ (للأحادي)، أكبر بكثير من مدى القوة البالغ
\qty{2}{fm}. والقيمة الكبيرة للمقدار $|a|$ تعني حالة \emph{تكاد} تكون
عند الطاقة الصفرية: أي ديوترون الثلاثي المربوط بالكاد
(\cref{ex:b3:schrodinger-three-dimensions:deuteron})، وفي
الأحادي شريك ``افتراضي'' يخفق في الارتباط بالكاد --- وإخفاقه
مكتوب في $a_{\text{s}} < 0$. أي لغز من عشرين بارنًا،
حلّته حالتان مقيدتان تكادان تكونان
(\cref{exo:b3:scattering-theory:7}).
\end{example}

\section{الرنينات}

\begin{proposition}[رنينات برايت--فيغنر]\label{prop:b3:scattering-theory:resonance}
\index{الرنين!في الانتثار}\index{صيغة برايت--فيغنر}
إذا أمكن أسر القذيف مؤقتًا في حالة شبه مقيدة طاقتها
$E_{\text{R}}$ وعمرها $\tau = \hbar/\Gamma$،
اجتاح \omterm{thm:b3:scattering-theory:partial}{انزياح الطور} المقابل القيمة $\pi/2$ وقفز \omterm{def:b3:scattering-theory:cross-section}{المقطع}
الجزئي:
\[
\sigma_l(E) \approx \frac{4\pi}{k^2}\,(2l+1)\,
\frac{\Gamma^2/4}{(E - E_{\text{R}})^2 + \Gamma^2/4} ,
\]
أي ذروة لورنتزية عرضها $\Gamma$ عند السقف الذي تستطيع الموجة
بلوغه (وهو ``حدّ الواحدية'' $4\pi(2l+1)/k^2$). ومقروءًا بالعكس،
يكون الرنين في مقطعٍ \emph{هو} اكتشاف حالة:
فموضعه مستوى طاقة وعرضه عمر. ومعظم
حديقة الجسيمات في \cref{ch:b3:particle-physics} اكتُشف بوصفه
نتوءات كهذه بالضبط.
\end{proposition}
\admitted

\begin{method}[قراءة مسألة انتثار]\label{met:b3:scattering-theory:method}
(1) التحريك أولًا: $k$، ومجال $q$ المتاح $0 \le q
\le 2k$ --- فأيّ مقاييس بنية مرئية؟ (2) وسريع وضعيف:
بورن --- حوّل الكمون بفورييه؛ وتحقق من الصغر. (3)
وبطيء: \omterm{thm:b3:scattering-theory:partial}{أمواج جزئية}، وعادةً $s$ وحدها؛ ففكّر \omterm{thm:b3:scattering-theory:partial}{بطول الانتثار}،
وتوقّع $\sigma = 4\pi a^2$ (ومن أجل بوزونات متطابقة:
$8\pi a^2$). (4) والذرى في $\sigma(E)$: لائم برايت--فيغنر، وأبلغ عن
مستوٍ وعمر؛ والهبوطات: طور يمرّ بالقيمة $\pi$. (5) والأهداف
الممتدة: اضرب سعة النقطة في عامل الشكل --- أي
تحويل فورييه للكثافة
(\cref{exo:b3:scattering-theory:12}).
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:1}
حزمة بروتونات شدّتها \qty{1}{\micro A} (أي $\num{6.2e12}$ بروتون في الثانية) ومقطعها
\qty{1}{cm^2} تعبر رقاقة ذهب سماكتها \qty{1}{\micro m} (مع $n =
\qty{5.9e28}{m^{-3}}$). (أ) احسب كثافة الأهداف السطحية. (ب) ومع
$\sigma = \qty{10}{b}$ لكل نواة، أيّ نسبة من الحزمة
تنتثر؟ (ج) احسب المعدل في كاشف زاويته المجسمة
\qty{e-3}{sr} لو كان الانتثار متماثل المناحي. (د) لماذا يكون
``\omterm{def:b3:scattering-theory:cross-section}{المقطع}'' خاصية محددة تخصّ \emph{الثلاثي}
(القذيف والهدف والطاقة) لا الهدف وحده؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:2}
كرات صلبة كلاسيكية أنصاف أقطارها $R_1, R_2$: (أ) بيّن أن $\sigma =
\pi(R_1 + R_2)^2$. (ب) من أجل جزيئات الهواء ($R \approx
\qty{0.15}{nm}$ و $n = \qty{2.5e25}{m^{-3}}$)، قدّر المسار الحر
المتوسط $1/n\sigma$، وقارنه بقيمة النظرية الحركية
في كتاب السنة الأولى. (ج) لماذا لا تُظهر كرة صلبة كلاسيكية أيّ
بنية زاوية (أي $\dd\sigma/\dd\Omega$ ثابتًا --- اقبل أو
استنتج)؟ (د) أيّ معلم من معالم انتثار الكرة الصلبة \emph{الكمومي}
(\cref{exo:b3:scattering-theory:6}) لا نظير كلاسيكي له؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:3}
\omterm{thm:b3:scattering-theory:born}{انتقال كمية الحركة}. (أ) ارسم مثلث $\vect k$ و $\vect k'$
واستنتج $q = 2k\sin(\theta/2)$. (ب) من أجل جسيمات ألفا بطاقة \qty{5.5}{MeV}
(أي $k = \qty{1.0e15}{m^{-1}}$): أعطِ مجال $q$ على $\theta \in
[0, \pi]$، وأصغر بنية قابلة للتمييز $\sim 1/q_{\max}$.
(ج) والسؤال نفسه من أجل إلكترونات بطاقة \qty{500}{MeV} (مع $k \approx E/\hbar c$).
(د) اقرأ العبرة التي تربط طاقة الحزمة بقدرة المجهر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:4}
انطلاقًا من \cref{thm:b3:scattering-theory:amplitude}: (أ) تحقق من أن
$|f|^2$ له أبعاد مساحة في وحدة الزاوية المجسمة؛ (ب) بيّن
أن التدفق المنتثر عبر كرة هو $\Phi\sigma$؛ (ج) لماذا يجب أن يكون
$f$ عقديًا في العموم (فأيّ قانون انحفاظ يحرسه طوره)؟ (د)
اذكر المبرهنة الضوئية $\sigma_{\text{كلي}} =
(4\pi/k)\operatorname{Im}f(0)$ ومعناها (فالموجة الأمامية
يجب أن تُستنزف بالضبط بمقدار ما ينتثر بعيدًا).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:5}
بورن من أجل يوكاوا. مع $V(r) = \dfrac{g\,\eu^{-\mu r}}{r}$: (أ)
نفّذ تكامل بورن (الجزء الزاوي أولًا، ثم $\int_0^
\infty\sin(qr)\eu^{-\mu r}\dd r$) لتجد $f = -\dfrac{2mg}
{\hbar^2(q^2 + \mu^2)}$. (ب) دع $\mu \to 0$ مع $g =
Q_1Q_2/4\pi\varepsilon_0$ واستعد رذرفورد. (ج) لماذا يتباعد
\omterm{def:b3:scattering-theory:cross-section}{مقطع} رذرفورد \emph{الكلي}، وأيّ مكوّن فيزيائي
(أي الحجب بإلكترونات الذرة) يعيد جوابًا منتهيًا؟ (د) في
فيزياء الجسيمات تنمذج صيغة يوكاوا مع $\mu =
m_\pi c/\hbar$ القوةَ النووية: احسب مداها
$1/\mu$ من أجل $m_\pi c^2 = \qty{140}{MeV}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:6}
الكرة الصلبة الكمومية (نصف قطرها $a$)، الموجة $s$: في الخارج $u_0 =
\sin(kr + \delta_0)$ مع $u_0(a) = 0$. (أ) بيّن أن $\delta_0 = -ka$.
(ب) بيّن أن $\sigma \to 4\pi a^2$ لمّا $k \to 0$: أي \emph{أربعة}
أضعاف الظلّ الهندسي. (ج) أعطِ التفسير الموجي لهذا العامل
(فلا ظلّ حادّ في العراج عند الأطوال الموجية الطويلة). (د) وعند الطاقات
العالية يؤول الجواب إلى $2\pi a^2$ --- أي ضعف الهندسي مع ذلك
(بعراج الظلّ): فلماذا لا يكون $\pi a^2$ ببساطة أبدًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:7}
عشرون بارنًا للهدروجين. النترونات البطيئة غير المستقطبة على
البروتونات تعاين قناة الثلاثي بوزن $3/4$ وقناة
الأحادي بوزن $1/4$: $\sigma = \pi(3a_{\text{t}}^2 +
a_{\text{s}}^2)$. (أ) قدّرها مع $a_{\text{t}} = \qty{5.4}{fm}$ و
$a_{\text{s}} = \qty{-23.7}{fm}$ وقارنها بالقيمة المقيسة
\qty{20.4}{b}. (ب) أيّ قناة تسود، رغم وزنها الإحصائي
الأصغر؟ (ج) اربط حجم $a_{\text{t}}$ وإشارته
بربط الديوترون شبه المعدوم؛ وبم ينبئ $a_{\text{s}} < 0$
عن الجملة الأحادية؟ (د) لماذا يكون الماء مهدّئًا فعّالًا
لنترونات المفاعلات --- ولماذا يهمّ هذا \omterm{def:b3:scattering-theory:cross-section}{المقطع} في ذلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:8}
رامزاور--تاونسند. نمذج ذرة الأرغون، أمام إلكترون
وارد، ببئر مربعة جاذبة مداها $R =
\qty{0.2}{nm}$. (أ) في الداخل يكون عدد الموجة $K =
\sqrt{2m(E + V_0)}/\hbar$: ويمكن للوصل أن يجعل الموجة الخارجية
تخرج بالقيمة $\delta_0 = \pi$ بالضبط --- فما \omterm{def:b3:scattering-theory:cross-section}{مقطع} الموجة $s$
عندئذٍ؟ (ب) لماذا تصير الذرة شبه غير مرئية عند تلك
الطاقة مع أن البئر قوية؟ (ج) قدّر $V_0$
الذي يضع الشفافية قرب $E = \qty{1}{eV}$ (اجعل البئر
تسع نصف طول موجة: $KR \approx \pi$). (د) لماذا \emph{لا} يُظهر
الهليوم والنيون الأثرَ عند طاقات مماثلة (فآبارهما أضحل
من أن يبلغ $\delta_0$ فيها $\pi$) --- وبمَ
يبرهن مجرد وجود الأثر في شأن أمواج المادة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:9}
من أجل سعة موجة $s$ صرفة $f = \eu^{\iu\delta_0}\sin\delta_0/
k$: (أ) احسب $\sigma$؛ (ب) احسب $(4\pi/k)\operatorname{Im}
f(0)$ وتحقق من المبرهنة الضوئية؛ (ج) بيّن أن أعظم \omterm{def:b3:scattering-theory:cross-section}{مقطع}
للموجة $s$ هو $4\pi/k^2$ (وهو حدّ الواحدية) و
قدّره من أجل النترونات الحرارية ($k = \qty{3.1e10}{m^{-1}}$)
بالبارن؛ (د) لماذا لا يستطيع رنين أن يدفع $\sigma_0$ وراء ذلك
السقف، مهما يكن التفاعل قويًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:10}
رنين نتروني. يُظهر اليورانيوم-238 رنينًا شهيرًا من أجل
النترونات عند $E_{\text{R}} = \qty{6.67}{eV}$ بعرض كلي
$\Gamma = \qty{27}{meV}$. (أ) احسب عمر الرنين.
(ب) احسب $4\pi/k^2$ عند تلك الطاقة بالبارن، وقارنه بالذروة
المقيسة $\sim\qty{2e4}{b}$ (والفرق هو عامل التفرّع بين
قناتَي المرونة والأسر --- علّق). (ج) عبّر عن العرض
بوحدات الحرارة: فلماذا يهمّ تعريض دوبلر لهذا الرنين
مع درجة حرارة الوقود في استقرار المفاعل (وأيّ إشارة للتغذية الراجعة)؟ (د) في جملتين:
كيف تجعل رنينات كهذه $^{238}$U ماصًّا للنترونات في المجال
فوق الحراري، ولماذا يكون ذلك مركزيًا في تصميم المفاعلات.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:11}
الذرات الباردة تعيش على عدد واحد. (أ) من أجل بوزونات متطابقة يكون
\omterm{def:b3:scattering-theory:cross-section}{المقطع} عند الطاقة المنخفضة $8\pi a^2$ (بتداخل تبديلي
بنّاء --- اقبله): قدّره من أجل الروبيديوم مع $a =
100\,a_0$. (ب) عند $T = \qty{100}{nK}$، تحقق من أن $ka \ll 1$
(احسب $k$ من $k_{\text{B}}T = \hbar^2k^2/2m$ مع $m =
\qty{1.4e-25}{kg}$): فالغاز ينسى فعلًا كل شيء عدا $a$.
(ج) بجوار رنين ``فيشباخ'' يجرّ مجال مغناطيسي مستوًى
جزيئيًا عبر الطاقة الصفرية فيتباعد $a$ ويغيّر
إشارته، تمامًا كما في \cref{ex:b3:scattering-theory:np}: فماذا
يصير تفاعل الغاز عندئذٍ رهن الإرادة؟ (د) لماذا تكون هذه
القابلية للموالفة --- أي قوة تفاعل على قرص --- أداةَ أحلام
الفيزيائي (سمِّ استعمالًا واحدًا: صنع الجزيئات، أو محاكاة
المادة قوية التقارن، أو إنهيار المكثفات)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:12}
عوامل الشكل: وزن سحب الشحنة. من أجل كثافة شحنة
ممتدة $\rho(\vect r)$، تضرب سعة بورن جوابَ النقطة
في $F(q) = \int\rho(\vect r)\eu^{\iu\vect q\cdot\vect r}
\dd^3r/Q$. (أ) بيّن أن $F(0) = 1$. (ب) انشره من أجل $q$ الصغيرة: $F
\approx 1 - q^2\langle r^2\rangle/6$: فقياس الهبوط عند $q$
الصغيرة يزن $\langle r^2\rangle$. (ج) يلائم انتثار
الإلكترون على البروتون القيمة $\sqrt{\langle r^2\rangle} \approx \qty{0.84}{fm}$:
فأيّ قياس مستقل من
\cref{pb:b3:hydrogen-atom:1} (أي الهدروجين الميوني) يتحقق منها،
ولماذا أُخذ خلافهما التاريخي (``لغز نصف قطر البروتون'') على هذا القدر من الجدّ؟ (د)
عند $q \gg 1/r_{\text{p}}$ ينهار عامل الشكل المرن لكن يبقى الانتثار
العنيف عن \emph{مكوّنات نقطية}: اذكر في جملة واحدة ماذا وجدت
نسخةُ تجربة رذرفورد العميقة غير المرنة سنة 1968 داخل
البروتون.
\end{exercise}

\section{مسألة: عدّادو الومضات الذين وجدوا النواة}

\begin{problem}[{مسألة نهاية الأسبوع --- انتثار رذرفورد، من
الهندسة إلى الكواركات}]\label{pb:b3:scattering-theory:1}
سنة 1909 جلس غايغر ومارسدن في الظلام يحصيان ومضات
الوميض: جسيمات ألفا مقذوفة عبر رقاقة ذهب، وقليل منها يرتدّ
إلى الخلف تقريبًا --- ``كأن قذيفة ارتدّت عن ورق
مناديل''. وقد خلق تحليل رذرفورد لتلك العدّات سنة 1911
الذرة النووية. وتعيد هذه المسألة بناءه. المعطيات: طاقة ألفا $E =
\qty{5.5}{MeV}$، وشحنتها $z = 2$؛ وللذهب $Z = 79$، وسماكة الرقاقة $t
= \qty{0.4}{\micro m}$، و $n = \qty{5.9e28}{atoms/m^3}$؛ و $k_C =
1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$.

\medskip\noindent\textbf{الجزء الأول --- ألفا واحدة ونواة واحدة.}
\begin{enumerate}
  \item في صورة حلوى البرقوق (أي شحنة منتشرة على الذرة،
        \qty{e-10}{m})، قدّر أقصى انحراف تستطيع ذرة
        واحدة أن تعطيه لألفا طاقتها \qty{5.5}{MeV} (بمجال كرة
        مطلية: $\theta \sim$ القوة $\times$ الزمن / كمية الحركة
        $\sim 2Zze^2k_C/(E\cdot\qty{e-10}{m})$ بالراديان). فهل يستطيع
        أيّ تراكم لركلات كهذه أن يرسل جسيمات ألفا إلى الخلف؟
  \item والآن لتكن الشحنة نقطية. اكتب مسافة أقرب
        اقتراب $d_0$ في تصادم وجهًا لوجه (أي بتحويل كل الطاقة
        الحركية إلى طاقة كولوم) وقدّرها.
  \item فبم يقتضي إذًا معدل قابل للقياس من الانتثار شبه الخلفي
        \emph{فورًا} في شأن مدى تركّز الشحنة الموجبة
        في الذرة؟
  \item ومن أجل وسيط اصطدام عام $b$، يعطي المدار الكلاسيكي
        $\tan(\theta/2) = d_0/2b$ (وهو مُسلَّم به --- أي
        قطع زائد لفصل الثقالة في كتاب السنة الأولى، مع
        التنافر): تحقق من حالتيه الحديتين عند $b \to 0$ وعند $b \to
        \infty$.
  \item احسب وسيط الاصطدام الذي ينثر ألفا بأكثر
        من $90^\circ$، ونسبة جسيمات ألفا العابرة
        للرقاقة التي تقترب من نواة إلى تلك المسافة ($n t\,\pi
        b^2$).
  \item وجد غايغر ومارسدن نحو ألفا واحدة من 8000 منحرفةً
        وراء $90^\circ$: قارن.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- \omterm{def:b3:scattering-theory:cross-section}{المقطع}.}
\begin{enumerate}[resume]
  \item جسيمات ألفا ذات وسيط اصطدام في $[b, b + \dd b]$ تنتثر
        في $[\theta, \theta + \dd\theta]$: فانطلاقًا من $\dd\sigma =
        2\pi b\,|\dd b|$ ومن علاقة المدار، استنتج
        \[
        \frac{\dd\sigma}{\dd\Omega} =
        \Big(\frac{d_0}{4}\Big)^2\frac{1}{\sin^4(\theta/2)} .
        \]
  \item تحقق من أن هذا يطابق جواب بورن الكمومي في
        \cref{ex:b3:scattering-theory:rutherford} (أعد كتابة $d_0$
        بدلالة $E$).
  \item قدّر $\dd\sigma/\dd\Omega$ عند $\theta = 30^\circ$ و
        $60^\circ$ و $120^\circ$ (بالبارن في الستراديان).
  \item تتغير عدّات غايغر ومارسدن سنة 1913 عند هندسة ثابتة
        وفق $1{:}\,\sin^{-4}(\theta/2)$: احسب نسبة العدّات
        المتنبَّأ بها بين $30^\circ$ و $120^\circ$ و
        لاحظ أن معطياتهما تبعت نسبًا كهذه على مدى خمس رتب
        من المعدل.
  \item لماذا يتيح العامل $Z^2$ في الصيغة نفسها للتجربة أن
        \emph{تزن} الشحنة النووية --- وكيف ساعدت ملاءمات كهذه
        على تثبيت $Z$ (وللذهب 79) بوصفه العدد الذرّي؟
  \item والعامل $1/E^2$: ماذا يحدث للنقش الزاوي كله
        حين تتضاعف طاقة ألفا؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- حجم النواة.}
\begin{enumerate}[resume]
  \item أقرب اقتراب عند الزاوية $\theta$: $d(\theta) =
        \tfrac{d_0}2[1 + 1/\sin(\theta/2)]$ (وهو مُسلَّم به). قدّره
        من أجل $\theta = 180^\circ$ ومن أجل \qty{60}{\degree}.
  \item ما دام $d(\theta)$ يفوق نصف قطر النواة، صحّت
        صيغة الشحنة النقطية: فأيّ حدّ أعلى لحجم نواة
        الذهب أرسته نقاط رذرفورد المتحقَّق منها عند $180^\circ$؟
  \item ومع قذائف أعلى طاقة تهبط العدّات عند الزوايا الكبيرة
        \emph{دون} رذرفورد: فبم ينبئ هذا الحيود
        (أيّ قوة جديدة، وعند أيّ مسافة)؟
  \item يعطي انتثار الإلكترونات الحديث أنصاف أقطار نووية $R
        \approx r_0A^{1/3}$ مع $r_0 = \qty{1.2}{fm}$: قدّرها
        من أجل الذهب ($A = 197$) وقارنها بحدّك.
  \item والأسّ $A^{1/3}$: بمَ ينبئ عن كثافة المادة النووية
        (أثابتة أم متزايدة؟) --- وهي واقعة سيبني عليها
        \cref{ch:b3:nuclear-physics}.
  \item لماذا مات نموذج حلوى البرقوق (الأعمى عن الشحنة) من
        \emph{هذه} التجربة لا من المطيافية؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- أحفاد رذرفورد.}
\begin{enumerate}[resume]
  \item اذكر جدول الترجمة الدقيق بين 1911 و1968:
        فألفا $\to$ حزمة إلكترونات، وذرة الذهب $\to$ بروتون، و
        النواة $\to$ كواركات --- فما الذي قام بدور
        ``الأحداث غير المتوقعة عند الزوايا الكبيرة'' في سلاك؟
  \item لماذا تكون \emph{الإلكترونات} المسبار الأنظف (فأيّ
        تعقيد في انتثار ألفا على النواة لا تعانيه)؟
  \item بلغة بورن: يقيس الانتثار المرن عند $q$ الكبيرة
        انهيار عامل الشكل (أي سحابة ليّنة)، بينما بقيت
        معدلات الانتثار العميق غير المرن كبيرة وشبه نقطية:
        اذكر النتيجة المستخلصة.
  \item قوس قرن في ثلاثة صفوف: أعطِ (طاقة المسبار، والمسافة
        الممكَّن تمييزها) من أجل جسيمات ألفا سنة 1911 ($\sim\qty{e-14}{m}$)،
        وسلاك سنة 1968 (\qty{20}{GeV} و $\sim\qty{e-16}{m}$)،
        والمصادم الهدروني الكبير (بمقياس التيرا إلكترون-فولط و $\sim\qty{e-19}{m}$) --- باستعمال
        القاعدة ``التمييز $\sim \hbar c/(qc)$''.
  \item النترونات، وهي غير مشحونة، لا تنتثر إلا عن النوى (وعن
        العزوم المغناطيسية): سمِّ شيئين يرسمهما إذًا انتثار
        النترونات في المواد ممّا تراه الأشعة السينية رؤيةً رديئة
        (كالذرات الخفيفة مثل الهدروجين؛ والترتيب المغناطيسي).
  \item تُقاس مقاطع الاكتشاف في المصادم الهدروني الكبير
        \emph{بالفمتوبارن} ($\qty{e-43}{m^2}$): فمن البارن إلى
        الفمتوبارن خمس عشرة رتبة --- فبمَ ينبئ ذلك
        عن ندرة التصادمات المثيرة، ولماذا تكون
        الإضاءة (أي التدفق المسلَّم) ثمينة قدر الطاقة؟
  \item لخّص النتيجة الموسومة: قانون $\sin^{-4}(\theta/2)$،
        متحقَّقًا منه ومضةً ومضة، وضع $99.98\%$ من كتلة الذرة
        في حجم مقداره $10^{-14}$ من حجمها --- والتجربة نفسها،
        معادةً بقوة أكبر أبدًا، لم تكفّ قط عن إيجاد الطبقة
        التالية.
\end{enumerate}
\end{problem}
