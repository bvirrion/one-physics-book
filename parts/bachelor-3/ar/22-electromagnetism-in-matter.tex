\section{الاستقطاب ومجال الإزاحة}

\begin{definition}[الاستقطاب والشحنات المرتبطة]\label{def:b3:electromagnetism-in-matter:polarisation}
\index{الاستقطاب}\index{شحنة مرتبطة}\index{عازل كهربائي}
\emph{العازل الكهربائي} عازل تكتسب جزيئاته (أو
تملك أصلًا) عزوم ثنائي قطب كهربائي. واستجابة المادة
يلخّصها \emph{الاستقطاب} $\vect P$: أي عزم ثنائي القطب
لكل وحدة حجم، $\vect P = n\langle\vect p\rangle$ من أجل $n$ جزيء
لكل وحدة حجم. والكتلة المستقطبة تحمل \emph{شحنات مرتبطة} ---
لا تحرّ لها في المغادرة، لكنها حقيقية تمامًا: أي كثافة سطحية
$\sigma_{\text{b}} = \vect P\cdot\vect n$ حيث يلقى الاستقطاب
السطح، وكثافة حجمية $\rho_{\text{b}} =
-\nabla\cdot\vect P$ حيثما كان غير منتظم. والصفيحة
المستقطبة استقطابًا منتظمًا تكافئ بالضبط لوحين من الشحنة $\pm P$
على وجهيها: إذ تتلاشى كل رؤوس ثنائيات أقطابها وأذيالها في الداخل.
\end{definition}

\begin{proposition}[مجال الإزاحة]\label{prop:b3:electromagnetism-in-matter:displacement}
\index{الإزاحة الكهربائية}\index{القابلية الكهربائية}\index{السماحية النسبية}
الشحنة الكلية حرة زائدَ مرتبطة؛ وقانون غاوس مع $\rho =
\rho_{\text{f}} + \rho_{\text{b}}$ يعاد ترتيبه قانونًا يخصّ
\emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{الإزاحة الكهربائية}} $\vect D = \varepsilon_0\vect E +
\vect P$:
\[
\nabla\cdot\vect D = \rho_{\text{f}} :
\]
منابع $\vect D$ هي الشحنات الحرة وحدها --- أي التي على
لبوسنا وأسلاكنا. وفي عازل \emph{خطي} تكون الاستجابة
تناسبية، $\vect P = \varepsilon_0\chi_{\text{e}}\vect E$، ومنه
$\vect D = \varepsilon_0\varepsilon_{\text{r}}\vect E$ مع
$\varepsilon_{\text{r}} = 1 + \chi_{\text{e}}$ وهي \emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{السماحية
النسبية}}: فالهواء $1.0006$، والبولي إتيلين $2.3$، والزجاج $\sim 5$، و
الماء $80$ الهائلة. وبين لبوسَي مكثفة عند شحنة
ثابتة لا يتغير $\vect D$ بالعازل، ومنه \emph{تهبط} $\vect E =
\vect D/\varepsilon_0\varepsilon_{\text{r}}$ بعامل
$\varepsilon_{\text{r}}$: فتواجه الشحنات السطحية المرتبطة الحرةَ
وتلغي أكثر مجالها. وتُضرب السعة في
$\varepsilon_{\text{r}}$ --- أي صناعة المكثفات كلها في حرف
يوناني واحد.
\end{proposition}

\begin{proof}
$\nabla\cdot(\varepsilon_0\vect E) = \rho_{\text{f}} +
\rho_{\text{b}} = \rho_{\text{f}} - \nabla\cdot\vect P$؛ ثم انقل حدّ
\omterm{def:b3:electromagnetism-in-matter:polarisation}{الاستقطاب} إلى اليسار.
\end{proof}

\section{من أين تأتي السماحية}

\begin{proposition}[الاستقطاب المستحثّ وكلاوزيوس--موسوتي]\label{prop:b3:electromagnetism-in-matter:mossotti}
\index{القابلية للاستقطاب}\index{علاقة كلاوزيوس--موسوتي}
الذرة في مجال $\vect E_{\text{موضعي}}$ تمتدّ ثنائيَ قطب
$\vect p = \alpha\vect E_{\text{موضعي}}$، مع $\alpha$
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{القابلية للاستقطاب}} (وبعدها: $\varepsilon_0\times$حجم ---
أي نحو $\varepsilon_0$ مضروبًا في الحجم الذرّي). وفي غاز مخلخل
يكون $E_{\text{موضعي}} \approx E$ وتكون $\chi_{\text{e}} =
n\alpha/\varepsilon_0$ صغيرة. وفي وسط كثيف يحسّ كل جزيء أيضًا
بجيرانه المستقطبين: فحفر تجويف كروي صغير
حوله يعطي $\vect E_{\text{موضعي}} = \vect E + \vect
P/3\varepsilon_0$، ويعطي الاتساق الذاتي
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{علاقة كلاوزيوس--موسوتي}}
\[
\frac{\varepsilon_{\text{r}} - 1}{\varepsilon_{\text{r}} + 2}
= \frac{n\alpha}{3\varepsilon_0} :
\]
فقِس قابلية بخار الضئيلة وتنبّأ بقابلية السائل ---
وهي تعمل في حدود بضعة بالمئة من أجل السوائل غير القطبية.
\end{proposition}
\admitted

\begin{remark}
مجال التجويف $\vect P/3\varepsilon_0$ هو نتيجة الكرة المستقطبة
استقطابًا منتظمًا، مذكورةً هنا بلا برهان؛ والاستنتاج الأمين من شروط
الحدّ يخصّ مقرّرًا مكرَّسًا للديناميك الكهربائي. وكل ما
عدا ذلك في الأعلى مسك دفاتر.
\end{remark}

\begin{proposition}[الجزيئات القطبية: دالة لانجفان]\label{prop:b3:electromagnetism-in-matter:langevin}
\index{دالة لانجفان}\index{استقطاب التوجيه}
الجزيء ذو ثنائي القطب \emph{الدائم} $p$ (وللماء:
$p = \qty{6.2e-30}{C.m}$) لا يحتاج إلى تمديد --- بل إلى محاذاة
لا غير، في وجه الاهتياج الحراري. ويعطي المتوسط القانوني في
\cref{ch:b3:canonical-ensemble} على التوجّهات، بطاقة
$-pE\cos\theta$،
\[
\langle\cos\theta\rangle = L(x) = \coth x - \frac1x ,
\qquad x = \frac{pE}{k_{\text{B}}T} ,
\]
وهي \emph{\omterm{prop:b3:electromagnetism-in-matter:langevin}{دالة لانجفان}}: خطية ($L \approx x/3$) عند $x$ الصغيرة، و
متشبّعة عند $1$ حين يفوز المجال فوزًا صريحًا. وفي النظام الخطي
تكون $\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ --- أي قانون
كوري كهربائي، يهبط وفق $1/T$: فسخّن سائلًا قطبيًّا \emph{تهبطْ}
سماحيته، وهو التوقيع الذي يميّز ثنائيات الأقطاب الدائمة من المستحثّة.
ومن أجل الماء عند حرارة الغرفة يتنبأ هذا بالقيمة
$\chi_{\text{e}} \sim 12$ --- أي القدر الصحيح، مع تعاون الروابط
الهدروجينية الذي يدفع القيمة الحقيقية إلى $80$.
\end{proposition}

\begin{proof}[برهان جزئي]
$Z = \int_0^\pi \eu^{x\cos\theta}\,2\pi\sin\theta\,\dd\theta \propto
\sinh x/x$، و $\langle\cos\theta\rangle = \dd\ln Z/\dd x = \coth x
- 1/x$. ونشر $\coth x = 1/x + x/3 - \dots$ من أجل $x$ الصغيرة يعطي
$L \approx x/3$، ومنه $P = np\langle\cos\theta\rangle =
np^2E/3k_{\text{B}}T$.
\end{proof}

\section{المغنطة والمجال \texorpdfstring{$H$}{H}}

\begin{definition}[المغنطة والتيارات المرتبطة والمجال H]\label{def:b3:electromagnetism-in-matter:magnetisation}
\index{المغنطة}\index{المجال المغناطيسي H}\index{القابلية المغناطيسية}
والحكاية المغناطيسية تجري بالتوازي. فالمادة مملوءة حلقات تيار
مجهرية --- أي إلكترونات تدور مداريًّا ومغزليًّا، وكلٌّ منها ثنائي قطب
مغناطيسي $\vect\mu$؛ وكثافتها هي \emph{المغنطة}
$\vect M = n\langle\vect\mu\rangle$. والكتلة الممغنطة انتظامًا
تكافئ \emph{تيارًا سطحيًّا مرتبطًا} يدور حول
جوانبها (فالحلقات الداخلية تتلاشى زوجًا زوجًا؛ والمغنط القضيبي \emph{هو}
ملفّ من تيار مرتبط). وبقسمة التيارات إلى حرة ومرتبطة،
يعاد ترتيب قانون أمبير حول المجال المساعد
\[
\vect H = \frac{\vect B}{\mu_0} - \vect M , \qquad
\oint \vect H\cdot\dd\vect\ell = I_{\text{حرّ}} :
\]
ينبع $\vect H$ من التيارات في أسلاكنا (وحدته
\unit{A/m})، بينما يحتفظ $\vect B$ بالفيزياء --- أي القوى والتدفق و
التحريض. وللأوساط الخطية $\vect M = \chi_{\text{m}}\vect H$ و
$\vect B = \mu_0(1 + \chi_{\text{m}})\vect H =
\mu_0\mu_{\text{r}}\vect H$.
\end{definition}

\begin{remark}[ثلاث شخصيات مغناطيسية]\label{rem:b3:electromagnetism-in-matter:zoo}
\index{المغناطيسية المعاكسة}\index{المغناطيسية المسايرة}\index{المغناطيسية الحديدية}
تجيب المواد $\vect H$ بثلاث طرق.
و\emph{المعاكسات} ($\chi_{\text{m}} \sim -10^{-5}$: الماء والنحاس و
الغرافيت) تستجيب كأن قانون لنز صار دائمًا: فالمجال المطبَّق
يضطرب كل مدار إلكتروني اضطرابًا يعاكسه --- وهي استجابة ضعيفة شاملة
مستقلة عن درجة الحرارة، وتنفرها المغانط (وقد رُفعت ضفدعة
هكذا في تجويف عند \qty{16}{T}).
و\emph{المسايرات} ($\chi_{\text{m}} \sim +10^{-5}$ إلى $10^{-3}$:
الألومنيوم و O$_2$) تملك عزومًا دائمة --- أي سبينات غير مزدوجة من
\cref{ch:b3:identical-particles} --- تتحاذى مثل ثنائيات أقطاب
لانجفان: $\chi_{\text{m}} = \mu_0 n\mu^2/3k_{\text{B}}T$، وهو قانون
كوري $1/T$ (والصورة الكمومية ذات المستويين هي
\cref{exo:b3:canonical-ensemble:11}).
و\emph{الحديديات} (الحديد والكوبالت والنيكل: $\mu_{\text{r}}$ بالآلاف)
مسايرات تتحادث سبيناتها أيضًا ---
أي اقتران التبديل في \cref{exo:b3:identical-particles:12} ---
وتحت \omterm{thm:b3:phase-transitions:meanfield}{درجة حرارة كوري} تتحاذى \emph{تلقائيًّا}:
أي \cref{thm:b3:phase-transitions:meanfield}، وقد وُضعت في العمل.
\end{remark}

\section{المغناطيسية الحديدية في العمل: المناطق والتخلّف}

\begin{definition}[المناطق والتخلّف]\label{def:b3:electromagnetism-in-matter:hysteresis}
\index{منطقة مغناطيسية}\index{التخلّف}\index{المغنطة البقيّة}\index{القسر}
الكتلة الخام من الحديد \emph{ليست} ممغنطة: بل تتشظى إلى
\emph{مناطق}، وهي أقاليم بمقياس الميكرومتر كلٌّ منها ممغنط تمامًا لكنه
يشير إلى جهة مختلفة، فيكاد المجال الخارجي (وكلفة طاقته)
يتلاشى. والمجال المطبَّق $\vect H$ يحرّك جدران المناطق ---
فتنمو المناطق المؤاتية --- ثم يدير مناطق كاملة إلى
الصفّ. وتعلق الجدران بالعيوب، فتصير العملية غير عكوسة: فأدر
$H$ صعودًا ونزولًا يرسمْ $B(H)$ حلقةً، هي \emph{دورة
التخلّف}. وأطفئ التيار يبقَ \emph{\omterm{def:b3:electromagnetism-in-matter:magnetisation}{المغنطة} البقيّة}
$B_{\text{r}}$؛ وإلغاؤه يحتاج إلى مجال \emph{القسر}
العكسي $H_{\text{c}}$. والمساحة التي تحيطها الحلقة هي
طاقة مبدَّدة لكل دورة ولكل وحدة حجم. والمواد \emph{الليّنة}
(الفولاذ السيليكوني والفرّيتات: حلقة رفيعة و
$H_{\text{c}}$ صغيرة) تصنع قلوب المحوّلات والمحركات؛ و\emph{الصلبة}
(الألنيكو و Nd$_2$Fe$_{14}$B: حلقة سمينة و $H_{\text{c}}$ هائلة) تصنع
المغانط الدائمة --- والذاكرة المغناطيسية: فمغنط الثلاجة
والقرص الصلب حلقتا تخلّف تأبيان النسيان.
\end{definition}

\begin{example}[المغنط الكهربائي ذو القلب الحديدي]\label{ex:b3:electromagnetism-in-matter:electromagnet}
لُفّ $N$ لفة تحمل $I$ حول حلقة حديدية ($\mu_{\text{r}}
\sim 5000$) فيها فجوة هواء صغيرة $e$. فقانون أمبير من أجل $\vect H$
حول الحلقة: $H_{\text{حديد}}\ell + H_{\text{فجوة}}e = NI$، و
استمرار التدفق يجعل $B$ مشتركًا، ومنه
\[
B\Big(\frac{\ell}{\mu_0\mu_{\text{r}}} + \frac{e}{\mu_0}\Big) = NI .
\]
ومع $\ell = \qty{1}{m}$ و $e = \qty{1}{cm}$، يكون حدّ الفجوة
خمسين ضعف حدّ الحديد: فيُنفَق كل جهد الوشيعة تقريبًا على
دفع المجال عبر سنتيمتر واحد من الهواء. وتلك هي الدارة
المغناطيسية في سطر واحد --- فالحديد ناقل شبه تامّ للتدفق، و
الهواء مقاومة --- ولذلك تبقي المحركات والمرحّلات ورافعات ساحة الخردة
فجوات هوائها رقيقة بلا هوادة
(\cref{pb:b3:electromagnetism-in-matter:1}).
\end{example}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:1}
مسك دفاتر \omterm{def:b3:electromagnetism-in-matter:polarisation}{الشحنة المرتبطة}. صفيحة سماكتها $d$ مستقطبة انتظامًا
على ناظمها، ومقدارها $P$. (أ) أعطِ كثافتَي \omterm{def:b3:electromagnetism-in-matter:polarisation}{الشحنة

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:4}
$B$ و $H$ و $M$. ملفّ طويل ($n = \num{2000}$ لفة في المتر، و

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:5}

لكل وحدة طول تُضرب في $\varepsilon_{\text{r}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:6}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:8}
\omterm{rem:b3:electromagnetism-in-matter:zoo}{المغناطيسية المعاكسة} مقدَّرةً. عامل إلكترونًا ذرّيًّا شحنةً على
حلقة نصف قطرها $r$؛ والمجال المطبَّق $B$ يغيّر تواتره الزاوي بمقدار
إزاحة لارمور $\Delta\omega = eB/2m_{\text{e}}$. (أ) بيّن أن
العزم المستحثّ هو $\Delta\mu = -e^2r^2B/4m_{\text{e}}$ (معاكسًا
$B$). (ب) وبتوسيط التوجّهات وجمع $Z$ إلكترونًا لكل
ذرة، يُوجد $\chi_{\text{m}} = -\mu_0ne^2Z\langle
r^2\rangle/6m_{\text{e}}$: احسبه من أجل الماء ($n =
\qty{3.3e28}{m^{-3}}$ جزيء، و $Z = 10$، و $\langle r^2\rangle
\approx (\qty{0.7}{\angstrom})^2$) وقارنه بالقيمة المقيسة
$-9\times10^{-6}$. (ج) ولماذا تكون \omterm{rem:b3:electromagnetism-in-matter:zoo}{المغناطيسية المعاكسة} مستقلة عن درجة الحرارة
بينما لا تكون المسايرة كذلك؟ (د) ولماذا ترتفع حتى ضفدعة عند
\qty{16}{T}، ولماذا يجب أن يكون مجال المغنط \emph{غير منتظم}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:10}
الدارة المغناطيسية. حلقة حديدية ($\ell = \qty{60}{cm}$ و

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:11}
فقد \omterm{def:b3:electromagnetism-in-matter:hysteresis}{التخلّف}. قلب محوّل (حجمه \qty{4e-3}{m^3})
يعمل عند \qty{50}{Hz}. وتحيط حلقة فولاذه السيليكوني
بمقدار $\sim\qty{40}{J/m^3}$ لكل دورة. (أ) احسب قدرة فقد
\omterm{def:b3:electromagnetism-in-matter:hysteresis}{التخلّف}. (ب) والقلب نفسه من فولاذ صلب
($\sim\qty{6e3}{J/m^3}$): احسب الفقد، واحكم. (ج) وتيارات إيدي تضيف
فقدًا $\propto f^2$: اشرح لماذا تُصفَّح القلوب ولماذا
تتولى الفرّيتات عند تواترات الراديو. (د) وبِتّ القرص الصلب يجب أن
\emph{يحفظ} مغنطته في وجه الركلات الحرارية عشر سنوات:
اربط مطالب $H_{\text{c}}$ من أجل الذاكرة بمطالبها من أجل
المحوّل، واستنتج أن لا مادة واحدة تصلح للعملين معًا.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:12}
مجال إزالة \omterm{def:b3:electromagnetism-in-matter:magnetisation}{المغنطة}. داخل جسم ممغنط انتظامًا وبلا
تيار حرّ، يفرض $\oint\vect H\cdot\dd\vect\ell = 0$ على
$\vect H$ في الداخل أن \emph{يعاكس} $\vect M$ ($\vect H =
-N_{\text{d}}\vect M$، مع $N_{\text{d}}$ عامل شكل: $1/3$
من أجل كرة، و $\approx 0$ على طول إبرة طويلة، و $\approx 1$ عبر
صفيحة رقيقة). (أ) برّر نهايتَي الإبرة والصفيحة بصورة
التيار المرتبط (الملفّ). (ب) ولماذا يزيل مغنط قصير جزءًا من
مغنطة نفسه بينما تحفظ الإبرة
مغنطتها؟ (ج) واربط بإبر البوصلة وبالحبيبات المستطيلة
في الشريط المغناطيسي. (د) وكرة من حديد ليّن تجلس في مجال خارجي
منتظم $B_0$: اشرح لماذا تتشبّع استجابتها عند $\vect H
\approx 0$ في الداخل، أي $M \approx 3B_0/\mu_0$ على الأكثر، مهما
كبرت $\mu_{\text{r}}$.
\end{exercise}

\section{مسألة: رافعة ساحة الخردة}

\begin{problem}\label{pb:b3:electromagnetism-in-matter:1}
\emph{رافعة ساحة الخردة.} ساحة إعادة تدوير تطلب مغنطًا
كهربائيًّا لرافعة: أي قِدرًا حديديًّا مسطَّح الوجه، عرضه متر، يجب أن
يرفع بالات سيارات مهروسة زنتها طنّان، ثم --- وهو لا يقلّ أهمية ---
\emph{يسقطها} عند الأمر. وأنت مهندس التصميم.

\medskip\noindent\textbf{الجزء الأول --- لماذا الحديد.}

\begin{enumerate}
  \item الوشيعة العارية: 400 لفة تحمل \qty{25}{A} موزَّعة على
        مسير مغناطيسي طوله $\sim\qty{1}{m}$. قدّر $B \sim
        \mu_0NI/\ell$ بلا حديد.
  \item اشرح في جملتين، بالمناطق، لماذا يضرب ملء
        الوشيعة بالحديد هذا المقدارَ في $\sim\mu_{\text{r}}$.
  \item ومغنطة الحديد تتشبّع عند $M_{\text{s}} \approx
        \qty{1.7e6}{A/m}$. احسب السقف $\mu_0M_{\text{s}}$
        الذي يضعه هذا على مجال القطب، مهما فعلت الوشيعة.
  \item وكل ذرة حديد تسهم بمقدار $\approx 2.2$ \omterm{prop:b3:quantum-angular-momentum:magnetic}{مغنطون بور}.
        تحقق: مع $n = \qty{8.5e28}{m^{-3}}$، استعد
        $M_{\text{s}}$.
  \item ولماذا يجب خرط وجهَي القطبين مسطَّحين وإبقاؤهما خاليين من
        الصدأ والحصى؟ (فكّر في
        \cref{ex:b3:electromagnetism-in-matter:electromagnet}.)
  \item والحمل نفسه يصير جزءًا من الدارة المغناطيسية.
        ارسم مسير التدفق: قلب القدر، والوجه الشمالي، وبالة الفولاذ،
        والوجه الجنوبي، والعودة عبر النير.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- الدارة المغناطيسية والقوة.}

\begin{enumerate}[resume]
  \item نمذج الدارة: مسير حديدي $\ell = \qty{1.2}{m}$، و
        $\mu_{\text{r}} = 2000$، وفجوتا هواء فعّالتان
        (تماسّ الوجه والبالة) مقدار كلٍّ منهما $e = \qty{1.5}{mm}$. اكتب
        قانون أمبير من أجل $H$ حول الحلقة.
  \item احسب حدود الممانعة الثلاثة لكل وحدة مساحة
        ($\ell/\mu_{\text{r}}$ في مقابل $2e$) وبيّن أن الفجوات
        الميليمترية تستهلك مع ذلك أكثر جهد الوشيعة.
  \item ومع $NI = \num{10000}$ أمبير-لفة، احسب $B$ في
        الفجوتين.
  \item وضغط الرفع على كل وجه قطب هو $B^2/2\mu_0$
        (أي كثافة الطاقة المغناطيسية المحرَّرة لكل متر تقارب).
        احسبه بوحدة \unit{kPa} من أجل $B$ الذي وجدته.
  \item ومساحة وجهَي القطبين الكلية $A = \qty{0.12}{m^2}$: احسب
        قوة الرفع وحوّلها إلى أطنان.
  \item فهل تلبّي مواصفة الطنّين بهامش أمان مقداره
        ضعفان؟ وإن لم تفعل، عدّل $NI$ واذكر التيار الجديد.
  \item والبالات المهروسة تلمس الوجهين على ثلث مساحتها ربما،
        بفجوات فعّالة أوسع. أعد حساب القوة من أجل
        $e = \qty{4}{mm}$ و $A_{\text{فعّال}} = \qty{0.04}{m^2}$ و
        علّق على سبب ذكر الرفعات المقنَّنة لسعة ``الصفيحة المسطَّحة''.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- التذكّر والنسيان.}

\begin{enumerate}[resume]
  \item تقلب الساحة المفتاح لإسقاط بالة --- فتبقى
        معلَّقة. سمِّ الجاني، بحلقة \omterm{def:b3:electromagnetism-in-matter:hysteresis}{التخلّف}.
  \item ولماذا يجب أن يكون القِدر من حديد \emph{ليّن} (وفيه
        $H_{\text{c}}$ صغيرة و $B_{\text{r}}$ صغيرة) لا من فولاذ
        مغناطيسي صلب؟
  \item وحتى الحديد الليّن يحفظ بقيّة قليلة. اقترح العلاج
        المعياري: نبضة تيار \emph{معكوسة} قصيرة ---
        فأيّ نقطة من الحلقة تستهدفها؟
  \item وبعض المتحكمات تطبّق بدلًا من ذلك تيارًا متناوبًا متلاشيًا.
        ارسم ما يفعله مسار $B$--$H$ ولماذا
        ينتهي بإزالة \omterm{def:b3:electromagnetism-in-matter:magnetisation}{المغنطة}.
  \item والرافعة تناول أيضًا صفائح ساخنة خارجة توًّا من فرن،
        عند \qty{800}{\celsius}. ونقطة كوري للحديد
        \qty{1043}{K}: فماذا يصيب قوة الرفع، ولماذا؟
        (\cref{thm:b3:phase-transitions:meanfield}.)
  \item ويقترح متدرب صيفي توفير النحاس بمضاعفة
        الفجوة ومضاعفة $NI$. استعمل صيغ الجزء الثاني
        لتشرح اللاتناظر: فما الذي ينتصف وما الذي يثبت لا غير؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- القدرة والحرارة والفاتورة.}

\begin{enumerate}[resume]
  \item الوشيعة: 400 لفة من النحاس، وطول اللفة الوسطي
        \qty{2.5}{m}، ومساحة السلك \qty{16}{mm^2}، والمقاومية
        $\rho = \qty{1.7e-8}{\Omega.m}$. احسب مقاومتها.
  \item وعند $I = \qty{25}{A}$: احسب القدرة المبدَّدة والطاقة
        اليومية لنوبة من ثماني ساعات بدورة عمل \qty{60}{\percent}.
  \item والطاقة المغناطيسية المخزونة في الفجوتين
        ($B^2/2\mu_0 \times$ الحجم): احسبها وقارنها مع
        \emph{ثانية} واحدة من تبديد الوشيعة. فأين تذهب كل
        بقية الطاقة الكهربائية؟
  \item وإسقاط الحمل لا يعيد من الطاقة المخزونة إلى
        الشبكة شيئًا يُذكر. اشرح، بمساحة الحلقة و
        بالنبضة التحريضية، لماذا يحتاج المتحكم إلى مسير
        دوران حرّ (ارتدادي) عبر الوشيعة.
  \item وتنقطع القدرة وبالة في الهواء. فماذا تفعل البالة،
        وما الذي تمسكه البقيّة \emph{وحدها}؟ برّر
        قاعدة الساحة بألا يمشي أحد تحت مغنط مشغَّل ---
        والحجّة التسويقية للرافعات المدعومة ببطارية.
  \item لخّص التصميم في أربعة أسطر: سقف المجال يضعه
        $M_{\text{s}}$، والقوة $B^2A/2\mu_0$، والقابلية للتحكم
        تخلّفُ الحديد الليّن، وكلفة التشغيل $RI^2$ --- أي
        الفصل كله معلَّقًا برافعة واحدة.
\end{enumerate}
\end{problem}
