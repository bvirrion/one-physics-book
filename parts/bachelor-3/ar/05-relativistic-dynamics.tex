\chapter{الديناميك النسبي}\label{ch:b3:relativistic-dynamics}

سنة 1964 صوّر ويليام برتوتزي إلكترونات تنطلق في مسرّع خطي: إذ
كانت كل دفقة تُعطى طاقة مقيسة، وتُقاس سرعتها على مدى \qty{8.4}{m}
من الطيران. وقد تنبأ نيوتن بأن إلكترونات طاقتها \qty{15}{MeV}
ينبغي أن تطير بثمانية أضعاف سرعة الضوء؛ لكن الميقاتة قالت
$0.9999c$ وأبت أن تتجاوز ذلك مهما دُفعت الإلكترونات. لقد أخبرنا
التحريك في الفصل السابق أن $c$ حدّ؛ وعلى الديناميك الآن أن يفسّر
ما يحلّ بالدفع --- أي أين يذهب الشغل حين تكفّ السرعة عن النمو.
والجواب يعيد تنظيم الميكانيك حول كمية حركة جديدة وطاقة جديدة،
تجمعهما أشهر معادلة في الفيزياء: فالطاقة المخزَّنة كتلة، والكتلة
طاقة مقيمة، $E = mc^2$. ويبني هذا الفصل هذا الديناميك، ثم يشغّله
حيث يعيش كل يوم --- في التفككات المشعّة وتصادمات الجسيمات
والمسرّعات التي تحوّل الحركة إلى مادة جديدة.

\section{إعادة بناء كمية الحركة والطاقة}

\begin{remark}[لماذا وجب أن يذهب $m\vect v$]\label{rem:b3:relativistic-dynamics:why}
كمية الحركة الكلاسيكية $m\vect v$ محفوظة في كل مرجع \emph{إذا}
تحوّلت السرعات على طريقة غاليلي. وهي لا تفعل
(\cref{prop:b3:relativistic-kinematics:addition}): فتصادم يحفظ
$\sum m\vect v$ في مرجع لا يحفظه في مرجع آخر، فلا يمكن أن تكون
كمية الحركة القديمة قانونًا للطبيعة --- بل هي في أحسن الأحوال
ظلّها عند السرعات الصغيرة. والإصلاح هو أن نقيس السرعة بساعة
الجسيم \emph{هو}: فالمقدار $\dd\vect r/\dd\tau =
\gamma\,\vect v$ يتحوّل تحوّلًا نظيفًا، والمقدار $m\,\dd\vect r/\dd\tau$
يتبيّن أنه محفوظ في كل المراجع في آن واحد.
\end{remark}

\begin{definition}[كمية الحركة والطاقة النسبيتان]\label{def:b3:relativistic-dynamics:momentum-energy}
\index{كمية الحركة النسبية}\index{الطاقة النسبية}\index{طاقة السكون}
جسيم كتلته $m$ وسرعته $\vect v$ (مع $\gamma = 1/\sqrt{1 -
v^2/c^2}$) يحمل كمية الحركة والطاقة
\[
\vect p = \gamma m\vect v , \qquad
E = \gamma mc^2 .
\]
وفي السكون $E_0 = mc^2$: وهي \emph{طاقة السكون}. والطاقة الحركية
هي ما تضيفه الحركة، $E_k = E - mc^2 = (\gamma - 1)mc^2$، وهي من أجل $v
\ll c$ ترجع إلى المقدار المألوف $\tfrac12 mv^2$ (انشر $\gamma$).
وفي كل سيرورة معزولة ينحفظ مجموع $\vect p$ ومجموع $E$ --- بما
فيه طاقات السكون.
\end{definition}

\begin{theorem}[علاقة الطاقة--كمية الحركة]\label{thm:b3:relativistic-dynamics:relation}
\index{علاقة الطاقة--كمية الحركة}
من أجل أيّ جسيم،
\[
E^2 = (pc)^2 + (mc^2)^2 ,
\qquad
\vect v = \frac{\vect p\,c^2}{E} :
\]
وللتركيب $E^2 - p^2c^2 = m^2c^4$ القيمة نفسها في كل
مرجع --- وهو للطاقة وكمية الحركة ما الفاصل للزمن
والفضاء. وثمة نظامان: عند السرعات الصغيرة $E \approx mc^2 + p^2/2m$؛
وعند الطاقات العالية ($E \gg mc^2$) يكون $E \approx pc$ و $v \to c$: فالدفع
الأقوى يضيف طاقة وكمية حركة، ولا يكاد يضيف سرعة.
\end{theorem}

\begin{proof}
$E^2 - p^2c^2 = \gamma^2m^2c^4(1 - v^2/c^2) = m^2c^4$؛ و $pc^2/E =
\gamma mvc^2/\gamma mc^2 = v$. وينتج ثبات المقدار بتغيير المرجع لأن $m$
و $c$ لا يتوقفان على المرجع.
\end{proof}

\begin{proposition}[الجسيمات عديمة الكتلة]\label{prop:b3:relativistic-dynamics:massless}
\index{الفوتون!كمية حركته}
للجسيم المعدوم الكتلة $E = pc$ ويتحرك بالسرعة $c$ بالضبط في كل
مرجع؛ ولا يمكن إبطاؤه، بل إزاحته نحو الأحمر فقط. والفوتون هو
الحالة النموذجية: $E = h\nu$ و $p = h\nu/c = h/\lambda$ --- وهما
علاقتا بلانك--أينشتاين في كتاب السنة الأولى، منظورًا إليهما الآن
بوصفهما ركن $m = 0$ من الميكانيك.
\end{proposition}

\begin{proof}
نضع $m = 0$ في \cref{thm:b3:relativistic-dynamics:relation}: فنجد $E = pc$
و $v = pc^2/E = c$.
\end{proof}

\begin{example}[رتب المقادير]\label{ex:b3:relativistic-dynamics:orders}
تحصي فيزياء الجسيمات الطاقة بالإلكترون-فولط والكتل بواحدة
$\unit{MeV}/c^2$: فالإلكترون $0.511$، والبروتون $938.3$، والميون $105.7$،
والبيون $139.6$. وللإلكترون ذي الطاقة الحركية \qty{1}{MeV} عامل $\gamma =
2.96$ وسرعة $v = 0.94c$ --- أي إنه نسبي تمامًا، ولهذا تبقى
الإلكترونيات كلاسيكية (\unit{eV}) ولا تبقى الفيزياء النووية كذلك
(\unit{MeV}). ولبروتون المصادم الهدروني الكبير ذي الطاقة \qty{6.8}{TeV} عامل $\gamma = 7250$،
وهو يتخلف عن فوتون بمقدار \qty{2.9}{m/s} لا غير.
\end{example}

\section{الكتلة طاقة مقيمة}

\begin{theorem}[تكافؤ الكتلة والطاقة]\label{thm:b3:relativistic-dynamics:emc2}
\index{تكافؤ الكتلة والطاقة}\index{النقص الكتلي}
كتلة جسم هي الطاقة الكلية لمحتوياته، مقسومة على
$c^2$، مقيسةً في مرجع سكونه: فسخّن جسمًا، أو اشدد نابضه، أو
أثِر ذراته، يزدد وزنه بالمقدار $\Delta E/c^2$؛ واربط أجزاءه
بعضها ببعض يصر وزنه \emph{أقلّ} بمقدار طاقة الربط على $c^2$
--- وهو \emph{\omterm{thm:b3:relativistic-dynamics:emc2}{النقص الكتلي}}. وبالعكس يمكن تحرير \omterm{def:b3:relativistic-dynamics:momentum-energy}{طاقة السكون}:
فكلما كان مجموع الكتل النهائية أقلّ من مجموع الكتل الابتدائية،
ظهر الفرق $\Delta m\,c^2$ طاقةً حركية أو إشعاعًا.
\end{theorem}
\admitted

\begin{example}[دفتر حساب $c^2$]\label{ex:b3:relativistic-dynamics:ledger}
$c^2 = \qty{9e16}{J/kg}$: فغرام واحد من فرق الكتلة هو
\qty{9e13}{J}، أي طاقة مدينة صغيرة ليوم كامل. والروابط الكيميائية
(بضعة إلكترون-فولط للجزيء) تزيح الكتل بأجزاء من $10^{10}$ ---
وهو ما لا يمكن وزنه أبدًا، ولهذا لم تلحظه الكيمياء قط. أما الربط
النوويّ فيزيحها بنحو $1\%$: فالهليوم أخفّ بمقدار $0.7\%$ من
مكوّناته الهدروجينية الأربعة، وتلك النسبة المفقودة، إذ تنساب
من الشمس ضوءًا، تكلّفها $\Delta m = L_\odot/c^2 =
\qty{4.3e9}{kg}$ في كل ثانية --- أي أربعة ملايين طن من ضوء الشمس.
ويُصفّي الإفناء الحساب كله: فإلكترون يلاقي بوزيترونًا لا يترك
غير فوتونين طاقة كل منهما \qty{511}{keV}، وهي الإشارة التي
تراقب بها ماسحات الإصدار البوزيتروني دماغًا حيًا.
\end{example}

\begin{remark}[ما معنى ``التحوّل'']\label{rem:b3:relativistic-dynamics:conversion}
لا شيء مادي ``يتحوّل إلى'' طاقة: فالطاقة الكلية كانت محفوظة
طوال الوقت. وإنما الذي يتغير هو \emph{مقامها} --- من طاقة سكون
تزن، إلى طاقة حركية وإشعاع يطيران. والقول العميق في
$E = mc^2$ هو أن العطالة نفسها محتوًى من الطاقة:
فعلبة غاز ساخن تقاوم التسارع أكثر من العلبة نفسها باردة.
\end{remark}

\section{التصادمات والتفككات}

\begin{definition}[رباعي كمية الحركة والكتلة الثابتة]\label{def:b3:relativistic-dynamics:four-momentum}
\index{رباعي كمية الحركة}\index{الكتلة الثابتة}
نحزم طاقة الجسيم وكمية حركته في \emph{رباعي كمية حركته}
$P = (E/c, \vect p)$. ومن أجل جملة من الجسيمات نجمع مركبةً مركبةً:
$E_{\text{كلي}} = \sum E_i$ و $\vect p_{\text{كلي}} = \sum\vect p_i$.
وتُعرَّف \emph{الكتلة الثابتة} $M$ للجملة بالعلاقة
\[
M^2c^4 = E_{\text{كلي}}^2 - \|\vect p_{\text{كلي}}\|^2c^2 :
\]
وهي العدد نفسه في كل مرجع، ويساوي الطاقة الكلية (مقسومة على
$c^2$) في \emph{مرجع مركز كميات الحركة} حيث
$\vect p_{\text{كلي}} = \vect 0$. ولاحظ أن $M$ تفوق مجموع
كتل الأجزاء كلما تحركت بعضها بالنسبة إلى بعض --- ففوتونان
يتباعدان لهما $M > 0$ وإن كان كل منهما عديم الكتلة.
\end{definition}

\begin{method}[حلّ تصادم نسبي]\label{met:b3:relativistic-dynamics:method}
(1) نكتب انحفاظ \omterm{def:b3:relativistic-dynamics:four-momentum}{رباعي كمية الحركة}: أي $E$ و $\vect p$ معًا،
لا أحدهما وحده. (2) نحسب الثابت $E^2 - p^2c^2$ لأيّ
حزمة تناسبنا --- إذ يمكن تقديره في أسهل
مرجع واستعماله في أيّ مرجع آخر. (3) العتبات: يكون التفاعل ممكنًا
حين تبلغ \omterm{def:b3:relativistic-dynamics:four-momentum}{الكتلة الثابتة} للحالة الابتدائية مجموع كتل السكون
للحالة النهائية. (4) التفككات في السكون: تكون كميتا حركة
الناتجين متعاكستين ومتساويتين في القيمة؛ وتتبع الطاقات من الكتل وحدها. (5)
ولا نحوّل إلى سرعات، إن طُلب ذلك، إلا في النهاية عبر $v = pc^2/E$.
\end{method}

\begin{example}[بصمة البيون]\label{ex:b3:relativistic-dynamics:pion}
يتفكك بيون مشحون ساكن وفق $\pi^+ \to \mu^+ + \nu$ (والنيوترينو
عديم الكتلة عمليًّا). ويجعل انحفاظ كمية الحركة الناتجين
متعاكسين بقيمة $p$ واحدة؛ ويُكتب انحفاظ الطاقة $m_\pi c^2 =
\sqrt{p^2c^2 + m_\mu^2c^4} + pc$. وبالحل:
\[
pc = \frac{(m_\pi^2 - m_\mu^2)c^2}{2m_\pi} = \qty{29.8}{MeV} ,
\]
فيولد كل ميون آتٍ من بيون متفكك ساكن بطاقة حركية مقدارها
\qty{4.1}{MeV} بالضبط --- أي خط أحادي الطاقة، وقد كانت مشاهدته
(باول، 1947) هي الطريقة التي وُزنت بها كتلة البيون أول مرة.
والتفككات الثنائية تعطي دائمًا خطوطًا كهذه؛ أما التفككات
الثلاثية فتلطّخها أطيافًا، وهي بالضبط الطريقة التي شُكّ بها أول
مرة في وجود النيوترينو في التفكك النوويّ $\beta$.
\end{example}

\begin{proposition}[العتبات: استبداد الهدف الثابت]\label{prop:b3:relativistic-dynamics:threshold}
\index{طاقة العتبة}\index{مصادم}
لخلق جسيمات جديدة لا تتوفر إلا طاقة \emph{مركز كميات الحركة}
$Mc^2$. وحزمة طاقتها $E$ تضرب هدفًا كتلته
$m_{\text{t}}$ ساكنًا تعطي
\[
M^2c^4 = m_{\text{b}}^2c^4 + m_{\text{t}}^2c^4 +
2\,E\,m_{\text{t}}c^2 :
\]
فالطاقة النافعة لا تنمو إلا وفق $\sqrt E$ --- والباقي يُهدر في
دفع الشظايا إلى الأمام. أما حزمتان متطابقتان تتصادمان وجهًا لوجه
فتعطيان $Mc^2 = 2E$: أي كلها نافعة. وهذه الصيغة الواحدة هي سبب
كون آلات الحدود القصوى مصادمات
(\cref{pb:b3:relativistic-dynamics:1}).
\end{proposition}

\begin{proof}
نقدّر $E_{\text{كلي}}^2 - p_{\text{كلي}}^2c^2$ في المخبر:
$(E + m_{\text{t}}c^2)^2 - p^2c^2$ حيث $p$ كمية حركة الحزمة؛ ننشر
ونستعمل $E^2 - p^2c^2 = m_{\text{b}}^2c^4$. وأما في المصادم فيكون
$\vect p_{\text{كلي}} = \vect 0$ و $Mc^2 = E_{\text{كلي}}$.
\end{proof}

\section{القوة والآلات}

\begin{proposition}[معادلة الحركة النسبية]\label{prop:b3:relativistic-dynamics:force}
\index{سنكروترون}
يبقى قانون نيوتن صالحًا في الصورة
\[
\vect F = \frac{\dd\vect p}{\dd t} = \frac{\dd(\gamma m\vect v)}
{\dd t} , \qquad
\frac{\dd E}{\dd t} = \vect F\cdot\vect v .
\]
والمجال المغناطيسي، وهو عمودي على $\vect v$ دائمًا، لا يغيّر أيّ
طاقة ويثني المسار إلى دائرة نصف قطرها
\[
r = \frac{p}{qB}
\]
--- وهي الصيغة نفسها الواردة في كتاب السنة الأولى، مع
$p$ النسبية. لكن تواتر الدوران $qB/\gamma m$ صار الآن \emph{يهبط}
كلما كسب الجسيم طاقة: فدفعة السيكلوترون ذات التواتر الثابت
تخرج عن الطور قرب مقياس الميغا إلكترون-فولط، وآلات الطاقات العالية --- أي
\emph{السنكروترونات} --- ترفع بدلًا من ذلك مجالها وتواترها
جنبًا إلى جنب مع $\gamma$، فتُبقي الحزمة على حلقة واحدة.
\end{proposition}

\begin{proof}[برهان جزئي]
قانون القوة هو تعريف الديناميك المتسق مع كمية الحركة
الجديدة (ويتبع $\dd E/\dd t$ من
$E^2 = p^2c^2 + m^2c^4$: إذ $E\,\dot E = c^2\vect p\cdot\dot{\vect p}$).
وأما الدائرة: فإن $|\dd\vect p/\dd t| = qvB$ مع ثبات $|\vect p|$
يعني أن متجهة كمية الحركة تدور بالمعدل $qvB/p$، فيكون نصف القطر
$v/(qvB/p) = p/qB$.
\end{proof}

\begin{example}[قراءة المصادم الهدروني الكبير كتمرين]\label{ex:b3:relativistic-dynamics:lhc}
بروتونات طاقتها $E = \qty{6.8}{TeV}$ في ثنائيات قطب مجالها $B = \qty{8.3}{T}$:
لدينا $p \approx E/c$ (فوق نسبي)، ومنه $r = p/qB =
\qty{2.7}{km}$ --- وبالفعل فإن نصف قطر الثني المغناطيسي للحلقة
\qty{2.8}{km}، ومحيطها \qty{27}{km} معظمه مغانط.
وتوفّر التصادمات وجهًا لوجه $Mc^2 = \qty{13.6}{TeV}$؛ ولبلوغ ذلك على
هدف ثابت لزمت حزمة طاقتها $2E^2/m_{\text{p}}c^2 \approx
\qty{e17}{eV}$ --- أي مئة ضعف مدى أيّ آلة بُنيت يومًا.
فصيغة المصادم، لا مغانط أقوى، هي التي اشترت
حدود الطاقة الحديثة.
\end{example}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:9}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:10}

(ج) قدّر $h/m_{\text{e}}c$ والانزياح الأقصى؛ ولماذا يكون
الأثر غير مرئي بالضوء المرئي وحاسمًا في الأشعة السينية؟ (د)

(ج) فوق تلك العتبة يصير الكون معتمًا أمام البروتونات وراء نحو
\qty{100}{Mly}: فبم يتنبأ ذلك في شأن طيف الأشعة الكونية

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:12}
صاروخ الفوتونات. صاروخ كتلته الابتدائية $m_{\text{i}}$ يقذف
عادمه حزمةَ ضوء موجَّهة فيبلغ السرعة $\beta c$. (أ)
بانحفاظ الطاقة وكمية الحركة بين البداية والنهاية، بيّن أن
\[
\frac{m_{\text{i}}}{m_{\text{f}}} =
\sqrt{\frac{1 + \beta}{1 - \beta}} = \eu^{\varphi} ,
\]
وهو أسّ السرعة الزائدية --- أي معادلة تسيولكوفسكي النسبية
بأفضل عادم ممكن. (ب) ما نسبة الكتلتين لبلوغ
$0.9c$؟ ولبلوغ $0.9c$ ثم \emph{التوقف} عند الوجهة؟ (ج) لا بدّ

الحمولة في رحلة ذهاب واحدة عند $0.9c$؟ (د) إنتاج العالم من
مضادات البروتون نانوغرامات في السنة: استنتج، في جملة واحدة، حكمًا على صواريخ الفوتونات.
\end{exercise}

\section{مسألة: صنع المادة --- مضاد البروتون والبيفاترون}

\begin{problem}[{مسألة نهاية الأسبوع --- كيف سعّر انحفاظ كمية
الحركة العالمَ المضاد}]\label{pb:b3:relativistic-dynamics:1}
اقتضت معادلات ديراك، منذ 1931، أن يكون للبروتون توأم مرآوي
معاكس الشحنة. وصنع واحد منه كان يعني شراء طاقة سكونه بطاقة
الحزمة --- وقد حدّد انحفاظ كمية الحركة الثمن. وتسعّر هذه المسألة
ذلك الثمن، وتصمّم الآلة التي دفعته، وتتابع اكتشاف
1955. المعطيات: $m_{\text{p}}c^2 = \qty{938.3}{MeV}$؛ وتنحفظ الشحنة
و\emph{العدد الباريوني} (وهو $+1$ للبروتونات والنترونات، و $-1$ لمضادات البروتون)
في كل تفاعل.

\medskip\noindent\textbf{الجزء الأول --- عدّة \omterm{def:b3:relativistic-dynamics:four-momentum}{رباعي كمية الحركة}.}
\begin{enumerate}
  \item من أجل جسيم واحد، بيّن أن $E^2 - p^2c^2 = m^2c^4$ انطلاقًا من
        تعريفَي $E$ و $\vect p$.
  \item من أجل جملة، لماذا يكون $E_{\text{كلي}}^2 -
        p_{\text{كلي}}^2c^2$ هو نفسه في كل المراجع، وبم
        يساوي في مرجع مركز كميات الحركة؟
  \item حزمة بروتونات طاقتها $E$ تصدم بروتونًا ساكنًا: بيّن أن
        $M^2c^4 = 2m_{\text{p}}^2c^4 + 2E\,m_{\text{p}}c^2$.
  \item تحقق من الحالتين الحديتين: عند الطاقة المنخفضة $Mc^2 \to
        2m_{\text{p}}c^2$؛ وعند الطاقة العالية $Mc^2 \approx
        \sqrt{2E\,m_{\text{p}}c^2}$ --- أي قانون الجذر التربيعي.
  \item لا يُسمح بالتفاعل إلا إذا بلغ $Mc^2$ مجموع كتل
        السكون للنواتج، مع سكون النواتج كلها في
        مرجع مركز كميات الحركة عند العتبة: برّر هذا الشرط
        الأخير.
  \item لماذا لا يمكن أبدًا استعادة الطاقة الحركية للنواتج في
        المخبر من أجل خلق الجسيمات على هدف ثابت؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- تسعير مضاد البروتون.}
\begin{enumerate}[resume]
  \item في تصادمات $p + p$، يجب على أرخص تفاعل يصنع
        مضاد بروتون $\bar p$ أن يصنع أيضًا بروتونًا \emph{إضافيًا}:
        اكتبه، وبرّره بالشحنة والعدد الباريوني.
  \item بيّن أن العتبة تقتضي $Mc^2 = 4m_{\text{p}}c^2$.
  \item استنتج طاقة الحزمة اللازمة $E = 7m_{\text{p}}c^2$ والطاقة
        الحركية $T = 6m_{\text{p}}c^2$؛ وقدّر $T$.
  \item عند العتبة، أيّ نسبة من الطاقة الحركية للحزمة
        صارت فعلًا كتلة سكون جديدة (منها $2m_{\text{p}}c^2$)؟
        وأين الباقي؟
  \item في مصادم $p$--$p$ وجهًا لوجه، أيّ طاقة حركية
        \emph{لكل حزمة} يقتضيها التفاعل نفسه؟ قارن
        التصميمين.
  \item سنة 1954 لم يكن ثمة مصادم (فالحزم كانت أرقّ من أن
        تتلاقى): فأيّ واقعة عملية فرضت اختيار الهدف الثابت،
        وكم كلّف ذلك من طاقة الحزمة؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- تصميم البيفاترون.}
بلغت الآلة المبنية في بيركلي $T = \qty{6.2}{GeV}$ ---
أي فوق العتبة براحة --- بمجالات ثنائيات قطب قدرها $B =
\qty{1.56}{T}$.
\begin{enumerate}[resume]
  \item احسب طاقة الحزمة الكلية وكمية حركتها عند الطاقة القصوى.
  \item احسب نصف قطر الثني $r = p/qB$ وقارنه بالقيمة
        \qty{15.2}{m} للآلة.
  \item احسب سرعة البروتون وتواتر الدوران على
        الحلقة $2\pi r$.
  \item عند الحقن تصل البروتونات بطاقة حركية \qty{10}{MeV}:
        احسب سرعتها، وفسّر لماذا وجب أن \emph{يمسح} تواتر
        التسريع في أثناء كل دورة (فأيّ مقدارين يتغيران كلما نمت $E$؟).
  \item بكسب نحو \qty{1.5}{kV} في كل لفة، كم لفة تلزم وكم
        تستغرق دورة تسريع واحدة؟
  \item جاء اسم ``البيفاترون'' من BeV، أي مليارات
        الإلكترون-فولط: اذكر في سطر واحد أيّ فيزياء حدّدت
        رقم التصميم عند $6$ وكسر من هذه المليارات.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- الاكتشاف، 1955.}
\begin{enumerate}[resume]
  \item يثني مغنط كل مرشَّح على دائرة: فسّر لماذا ينتقي هذا
        الجسيمات بحسب \emph{كمية الحركة} ($r = p/qB$)، لا بحسب
        الطاقة أو السرعة --- ولماذا تبقى حزمة منتقاة بكمية
        الحركة خليطًا من الأصناف.
  \item أنتجت الحزمة على هدف نحاسي سيولًا من البيونات
        السالبة ($m_\pi c^2 = \qty{139.6}{MeV}$) اختبأت بينها
        بضعة مضادات بروتون. وانتقى المطياف السوالب ذات
        كمية الحركة $p = \qty{1.19}{GeV}/c$: احسب سرعة
        مضاد البروتون وسرعة البيون عند كمية الحركة تلك.
  \item على مسار طيران طوله \qty{12}{m}، احسب زمنَي
        الطيران: فأيّ تمييز زمني اقتضاه الاكتشاف؟
  \item استعملت بصمة ثانية مستقلة عدّادات تشيرنكوف،
        لا تشتغل إلا فوق عتبة سرعة: فأيّ جسيم رُتّب ليشغّلها،
        ولماذا تهمّ الوفرة في اكتشاف؟
  \item أحصى تشيمبرلين وسيغري نحو مضاد بروتون واحد لكل
        $\num{44000}$ بيون: فلماذا ضمنت حجة العتبة أن يبقى
        المعدل ضئيلًا حتى فوق العتبة؟
  \item يلاقي مضاد البروتون في النهاية بروتونًا فيفنيان:
        فكم من الطاقة تتحرر في كل حدث، وإلى ماذا،
        نموذجيًا؟
  \item لخّص النتيجة الموسومة: سعّر انحفاظ \omterm{def:b3:relativistic-dynamics:four-momentum}{رباعي كمية الحركة}
        مضادَ البروتون عند $T = 6m_{\text{p}}c^2 =
        \qty{5.6}{GeV}$ على هدف ثابت؛ ودفعه سنكروترون طاقته
        \qty{6.2}{GeV} ونصف قطره \qty{15}{m}، فصار العالم المضاد
        واقعة مخبرية --- وجائزة نوبل لسنة 1959.
\end{enumerate}
\end{problem}
