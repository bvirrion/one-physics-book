\chapter{المذبذب التوافقي الكمومي}\label{ch:b3:harmonic-oscillator}

لا يكاد شيء في الطبيعة يكون مذبذبًا توافقيًا بالضبط، ويكاد كل
شيء يكون كذلك بالتقريب: فأيّ جملة تُزاح عن توازن مستقر ---
رابطة جزيء، أو ذرة في بلورة، أو جسر، أو نمط من أنماط المجال
الكهرطيسي --- تحسّ قوة إرجاع تتناسب مع الانزياح، لأن كل كمون
أملس قطع مكافئ في قاع بئره. ومن حلّ المذبذب الكمومي مرة واحدة
فقد حلّ اهتزازات العالم الصغيرة كلها. ويحلّه هذا الفصل بالأسلوب
الجبري الذي صار توقيع الميكانيك الكمومي: إذ يصعد مؤثرا سلّم
وينزلان درجًا متساوي الدرجات تمامًا من المستويات $\big(n +
\tfrac12\big)\hbar\omega$، ونصف الدرجة في الأسفل --- أي
\omterm{thm:b3:harmonic-oscillator:spectrum}{طاقة النقطة الصفرية} --- مبرهنة لا خيار. وتمتدّ
النتائج من سبب امتناع الهليوم عن التجمد إلى كيفية اعتراض جزيء
ثاني أكسيد الكربون، وهو يرنّ عند $\hbar\omega$ الخاص به، حرارةَ
الأرض الخارجة.

\section{السلّم}

\begin{definition}[مؤثرا السلّم]\label{def:b3:harmonic-oscillator:ladder}
\index{مؤثرا السلّم}\index{مؤثر الإفناء}\index{مؤثر الإنشاء}
من أجل $\hat H = \hat p^2/2m + \tfrac12 m\omega^2\hat x^2$، نُدخل
الزوج عديم الأبعاد وغير الهرميتي
\[
\hat a = \sqrt{\frac{m\omega}{2\hbar}}\Big(\hat x +
\frac{\iu\hat p}{m\omega}\Big) , \qquad
\hat a^\dagger = \sqrt{\frac{m\omega}{2\hbar}}\Big(\hat x -
\frac{\iu\hat p}{m\omega}\Big) .
\]
انطلاقًا من $[\hat x, \hat p] = \iu\hbar$:
\[
[\hat a, \hat a^\dagger] = 1 , \qquad
\hat H = \hbar\omega\Big(\hat N + \tfrac12\Big) , \quad
\hat N = \hat a^\dagger\hat a .
\]
والمؤثر $\hat N$ هرميتي؛ وستحصي قيمه الذاتية \emph{الكمّات}، و
$\hat a$ و $\hat a^\dagger$ --- أي مؤثرا \emph{الإفناء} و
\emph{الإنشاء} --- سيحذفان واحدًا ويضيفان واحدًا.
\end{definition}

\begin{theorem}[الطيف، بالجبر وحده]\label{thm:b3:harmonic-oscillator:spectrum}
\index{طاقة النقطة الصفرية}
القيم الذاتية للمؤثر $\hat H$ هي بالضبط
\[
E_n = \Big(n + \tfrac12\Big)\hbar\omega , \qquad n = 0, 1, 2, \dots
\]
--- أي سلّم لانهائي متساوي الدرجات $\hbar\omega$ فوق مستوٍ
أساسي \emph{ليس} معدومًا: هو \emph{\omterm{thm:b3:harmonic-oscillator:spectrum}{طاقة النقطة الصفرية}}
$\tfrac12\hbar\omega$. وتربط بين الحالات الذاتية المنظَّمة $\ket n$
العلاقاتُ
\[
\hat a\ket n = \sqrt n\,\ket{n - 1} , \qquad
\hat a^\dagger\ket n = \sqrt{n + 1}\,\ket{n + 1} ,
\qquad
\ket n = \frac{(\hat a^\dagger)^n}{\sqrt{n!}}\,\ket0 .
\]
\end{theorem}

\begin{proof}
انطلاقًا من \omterm{def:b3:quantum-formalism:commutator}{المبدّلة}، $\hat N\hat a = \hat a(\hat N - 1)$: فإذا كانت
$\ket\nu$ ذات \omterm{def:b3:quantum-formalism:observable}{قيمة ذاتية} للمؤثر $\hat N$ مقدارها $\nu$، كانت $\hat a\ket\nu$
متجهةً ذاتية بالقيمة $\nu - 1$ (أو المتجهة المعدومة). ولا يُسمح بكل
نزول إلا ما دام $\nu \ge 0$، لأن $\nu = \braket{\nu}{\hat
N\nu} = \|\hat a\ket\nu\|^2 \ge 0$؛ فوجب أن ينتهي النزول،
وهو لا ينتهي إلا على حالة فيها $\hat a\ket{\nu_0}
= 0$، ومنها $\nu_0 = 0$. فيمسح $\nu$ إذًا الأعداد الصحيحة
غير السالبة. وتتبع التنظيمات من $\|\hat a\ket n\|^2 = n$
و $\|\hat a^\dagger\ket n\|^2 = n + 1$.
\end{proof}

\begin{proposition}[الحالات في الفضاء]\label{prop:b3:harmonic-oscillator:wavefunctions}
الحالة الأساسية هي الغاوسية
\[
\varphi_0(x) = \Big(\frac{m\omega}{\pi\hbar}\Big)^{1/4}
\eu^{-m\omega x^2/2\hbar} ,
\]
وتُنال بحلّ المعادلة \emph{من المرتبة الأولى} $\hat a\varphi_0
= 0$؛ وهي تشبّع متراجحة هايزنبرغ، $\Delta x\,\Delta p =
\hbar/2$، مع $\Delta x = \sqrt{\hbar/2m\omega}$. وتطبيق
$\hat a^\dagger$ مرارًا يولّد $\varphi_n$: أي كثير حدود من
الدرجة $n$ (وله $n$ عقدة) مضروبًا في الغاوسية نفسها. ومن أجل $n$ الكبيرة
تتذبذب $|\varphi_n|^2$ تذبذبًا سريعًا حول توزيع زمن المكوث
الكلاسيكي، وهي أكبر ما تكون قرب نقطتَي الرجوع --- أي مبدأ
التقابل في صورة.
\end{proposition}

\begin{proof}[برهان جزئي]
تُكتب $\hat a\varphi_0 = 0$ في الصورة $\varphi_0' = -(m\omega/\hbar)x
\varphi_0$: فتعطي الغاوسية، منظَّمةً. وأما التشبّع: فالغاوسية هي
حالة المساواة في \cref{thm:b3:quantum-formalism:uncertainty}.
وتتبع بنية كثير الحدود من كون $\hat a^\dagger$ من المرتبة
الأولى في $x$ و $\dd/\dd x$؛ وتُتحقق عبارة $n$ الكبيرة في
\cref{exo:b3:harmonic-oscillator:12}.
\end{proof}

\section{طاقة النقطة الصفرية حقيقية}

\begin{remark}[لماذا لا يمكن أن تكون الأرضية أدنى]\label{rem:b3:harmonic-oscillator:zeropoint}
حالة الطاقة المعدومة كانت ستحتاج إلى $\langle\hat p^2\rangle =
\langle\hat x^2\rangle = 0$: أي ساكنة تمامًا \emph{ومركَّزة}
تمامًا، وهو ما يمنعه $\Delta x\,\Delta p \ge \hbar/2$. فالحالة
الأساسية أفضل تسوية تبيحها المتراجحة، و
$\tfrac12\hbar\omega$ هي الأجرة. وليست ثابتًا حسابيًا:
فحركة النقطة الصفرية تلطّخ نقوش انعراج الأشعة السينية عند الصفر
المطلق؛ وتعطي النظائر الأخفّ روابط فعّالة أضعف (فجزيئا H$_2$ و
D$_2$ يتفككان بطاقتين مختلفتين قابلتين للقياس من البئر
الإلكترونية نفسها)؛ وهي في الهليوم عنيفة إلى حدّ --- ذرات خفيفة
وتجاذب واهن --- يجعل السائل \emph{لا} يتجمد أبدًا تحت ضغط
بخاره: فهو العنصر الوحيد الذي يبقى سائلًا عند الصفر
المطلق، ولا يتصلب إلا بعون 25 جوًا.
\end{remark}

\begin{example}[مقاييس $\hbar\omega$]\label{ex:b3:harmonic-oscillator:scales}
رابطة أحادي أكسيد الكربون ($\omega = \qty{4.1e14}{rad/s}$): لها $\hbar\omega =
\qty{0.27}{eV}$، أي عشرة أضعاف $k_{\text{B}}T$ في درجة حرارة الغرفة ---
فاهتزازات الجزيئات مجمَّدة في هواء كل يوم، ولهذا حيّرت
السعات الحرارية للغازات الثنائية القرنَ التاسع عشر. وأما بندول
($\omega = \qty{5}{rad/s}$): فله $\hbar\omega = \qty{3e-15}{eV}$،
وهو بعيد عن التمييز بلا أمل --- أي نهاية التقابل. ولمرآة من مرايا
مرصد لايغو كتلتها \qty{40}{kg}، معلَّقة بحيث تتأرجح عند
\qty{100}{Hz}، بوصفها نمط مذبذب سعةُ نقطة صفرية $\Delta x =
\sqrt{\hbar/2m\omega} \approx \qty{5e-20}{m}$ --- ويميّز المرصد
روتينيًا انتقالات \emph{عند} هذه الأرضية الكمومية: فحركة
النقطة الصفرية لجسم كتلته أربعون كيلوغرامًا صارت اليوم قيدًا
هندسيًا (\cref{exo:b3:harmonic-oscillator:3}).
\end{example}

\section{الحالات المترابطة: الوجه الكلاسيكي للمذبذب الكمومي}

\begin{definition}[الحالات المترابطة]\label{def:b3:harmonic-oscillator:coherent}
\index{حالة مترابطة}
\emph{الحالة المترابطة} $\ket\alpha$ متجهة ذاتية \omterm{def:b3:harmonic-oscillator:ladder}{لمؤثر
الإفناء}، $\hat a\ket\alpha = \alpha\ket\alpha$، مع
$\alpha$ عددًا عقديًا كيفما كان. وبنشرها على السلّم،
\[
\ket\alpha = \eu^{-|\alpha|^2/2}\sum_{n=0}^\infty
\frac{\alpha^n}{\sqrt{n!}}\,\ket n :
\]
يكون الإحصاء الكمومي موزَّعًا وفق بواسون بمتوسط $\langle\hat
N\rangle = |\alpha|^2$ وانتشار $\Delta N = |\alpha|$.
\end{definition}

\begin{proposition}[لماذا الليزرات كلاسيكية]\label{prop:b3:harmonic-oscillator:coherent-motion}
\omterm{def:b3:harmonic-oscillator:coherent}{الحالة المترابطة} هي غاوسية الحالة الأساسية مزاحةً في \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{فضاء
الطور}؛ وتحت تطور المذبذب \emph{تبقى} مترابطة، مع
$\alpha(t) = \alpha\,\eu^{-\iu\omega t}$: فينفّذ مركزها بالضبط
الحركة الكلاسيكية، $\langle\hat x\rangle(t) =
x_0\cos\omega t + \cdots$، بينما يبقى عرضاها في الحدّ الأدنى
$\Delta x\,\Delta p = \hbar/2$ إلى الأبد --- أي حزمة موجية لا
تنتشر أبدًا. والحالات المترابطة هي كيف ينتحل مذبذب كمومي شخصية
مذبذب كلاسيكي؛ وضوء الليزر \omterm{def:b3:harmonic-oscillator:coherent}{حالة مترابطة} لنمط من أنماط المجال،
وعدد فوتوناته بواسوني (وهو \emph{الضجيج الحبيبي} في كل
كاشف ضوئي)، ومجاله يتذبذب كما قال ماكسويل.
\end{proposition}

\begin{proof}[برهان جزئي]
يتبع النشر بكتابة $\ket\alpha = \sum c_n\ket n$ في
$\hat a\ket\alpha = \alpha\ket\alpha$: فنجد $c_n = \alpha c_{n-1}/\sqrt
n$. وأما التطور: فتلتقط كل $\ket n$ الطور $\eu^{-\iu(n + 1/2)\omega t}$،
وهو يُجمع من جديد فيعطي \omterm{def:b3:harmonic-oscillator:coherent}{حالة مترابطة} للمقدار $\alpha\eu^{-\iu\omega t}$
(إلى طور شامل). ويأتي $\langle\hat x\rangle \propto
\operatorname{Re}\alpha(t)$ من $\hat x \propto \hat a +
\hat a^\dagger$. وأما كون الحالة هي الغاوسية المزاحة فمُسلَّم
به هنا.
\end{proof}

\begin{method}[جبر المذبذب]\label{met:b3:harmonic-oscillator:method}
(1) عبّر عمّا يُطلب بدلالة $\hat a$ و $\hat a^\dagger$: $\hat x
= \sqrt{\hbar/2m\omega}\,(\hat a + \hat a^\dagger)$ و $\hat p =
\iu\sqrt{m\hbar\omega/2}\,(\hat a^\dagger - \hat a)$. (2) وانقل
مؤثرات $\hat a$ إلى اليمين بالعلاقة $[\hat a, \hat a^\dagger] = 1$؛ واستعمل
$\hat a\ket0 = 0$. (3) والعناصر المصفوفية: $\hat x$ لا يصل إلا بين
الدرجات المتجاورة --- وهي قاعدة الانتقاء $\Delta n = \pm1$ في
مطيافية الاهتزاز. (4) وكل \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{هاميلتونية} تربيعية (مذبذبات متقارنة
أو دارات أو أنماط مجال) تتقطّر إلى سلالم مستقلة: فأوجد
\omterm{prop:b3:lagrangian-mechanics:modes}{الأنماط الذاتية} أولًا، وكمِّ كلًّا منها. (5) والأعداد
أولًا: فمقارنة $\hbar\omega$ مع $k_{\text{B}}T$ تحسم أهي جملة
كمومية أم كلاسيكية قبل أيّ جبر.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:1}
(أ) تحقق من $[\hat a, \hat a^\dagger] = 1$ انطلاقًا من $[\hat x, \hat p] =
\iu\hbar$. (ب) تحقق من $\hat H = \hbar\omega(\hat a^\dagger\hat a +
\tfrac12)$ بالنشر المباشر. (ج) اقلب العلاقتين لتعبّر عن $\hat x$ و
$\hat p$. (د) بيّن أن $\hat N$ هرميتي واشرح لماذا لا يمكن أن يكون $\hat a$
وحده مرصودة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:2}
على الحالة الذاتية $\ket n$: (أ) بيّن أن $\langle\hat x\rangle =
\langle\hat p\rangle = 0$؛ (ب) احسب $\langle\hat x^2\rangle$ و
$\langle\hat p^2\rangle$؛ (ج) استنتج $\Delta x\,\Delta p = (n +
\tfrac12)\hbar$؛ (د) تحقق من نسبة الفيريال $\langle E_k\rangle =
\langle E_p\rangle$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:3}
سعات النقطة الصفرية $\Delta x = \sqrt{\hbar/2m\omega}$: احسبها
من أجل (أ) جزيء أحادي أكسيد الكربون ($\mu = \qty{1.14e-26}{kg}$ و $\omega =
\qty{4.1e14}{rad/s}$)، مقارنةً بطول الرابطة
\qty{0.11}{nm}؛ (ب) ذرة هدروجين في جسم صلب ($m =
\qty{1.7e-27}{kg}$ و $\hbar\omega = \qty{0.1}{eV}$)؛ (ج) مرآة من مرايا
لايغو ($m = \qty{40}{kg}$ و $f = \qty{100}{Hz}$)؛ (د) رتّب الثلاثة
بوصفها نسبًا من أحجام جملها وعلّق على من هو
``الكمومي''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:4}
(أ) باستعمال علاقات السلّم، احسب $\bra m\hat x\ket n$ و
بيّن أنه ينعدم ما لم يكن $m = n \pm 1$. (ب) استنتج قاعدة
الانتقاء الاهتزازية $\Delta n = \pm1$ من أجل امتصاص الضوء (فالتقارن
$\propto\hat x$). (ج) لماذا يمتصّ جزيء غير متجانس النوى (مثل أحادي أكسيد الكربون)
الأشعة تحت الحمراء بينما لا يمتصّها N$_2$؟ (د) تُظهر الجزيئات الحقيقية
خطوطًا ``توافقية علوية'' ضعيفة عند $\approx 2\hbar\omega$: فبم يكشف ذلك
عن الكمون؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:5}
(أ) حلّ $\hat a\varphi_0 = 0$ بوصفها معادلة تفاضلية و
نظّم الحلّ. (ب) ولّد $\varphi_1$ و $\varphi_2$ بتطبيق
$\hat a^\dagger$. (ج) تحقق من $\varphi_1 \perp \varphi_0$ بالزوجية
وحدها. (د) ارسم الكثافات الثلاث تقريبيًا وتحقق من عدد العقد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:6}
تمهيد بولتزمان. مجموعة من مذبذبات متطابقة عند
درجة الحرارة $T$ تعمر المستوي $n$ باحتمال $\propto
\eu^{-E_n/k_{\text{B}}T}$ (كتاب السنة الثانية، عامل بولتزمان). (أ)
بيّن أن العدد الكمومي المتوسط هو $\langle n\rangle =
1/(\eu^{\hbar\omega/k_{\text{B}}T} - 1)$. (ب) قدّره من أجل اهتزاز
أحادي أكسيد الكربون عند \qty{300}{K} وعند \qty{2000}{K}. (ج) بيّن أن الطاقة
المتوسطة تؤول إلى $k_{\text{B}}T$ عند درجات الحرارة العالية (فنستعيد
التوزيع المتساوي) وإلى $\tfrac12\hbar\omega$ عند المنخفضة. (د) عند أيّ
درجة حرارة يكون لاهتزاز عند \qty{15}{\micro m} (وهو انحناء ثاني
أكسيد الكربون) عددٌ $\langle n\rangle = 0.1$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:7}
إحصاء الحالات المترابطة. (أ) انطلاقًا من نشر $\ket\alpha$،
بيّن أن $\mathcal P(n)$ بواسوني بمتوسط $|\alpha|^2$. (ب) بيّن أن
$\langle\hat N\rangle = |\alpha|^2$ و $\Delta N = |\alpha|$. (ج)
ليزر قدرته \qty{1}{mW} عند \qty{633}{nm}: احسب عدد الفوتونات في الثانية و
التذبذب النسبي $\Delta N/\langle N\rangle$ في ثانية واحدة. (د)
الضجيج الحبيبي: بيّن أن نسبة الضجيج إلى الإشارة في التيار الضوئي تهبط وفق
$1/\sqrt{\langle N\rangle}$ --- ولماذا تبدو الحزم الساطعة ملساء.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:8}
تطور \omterm{def:b3:harmonic-oscillator:coherent}{حالة مترابطة}. (أ) طبّق أطوار التطور على
النشر وبيّن أن الحالة $\ket{\alpha(t)}$ مع $\alpha(t) =
\alpha\eu^{-\iu\omega t}$ (إلى طور شامل). (ب) استنتج
$\langle\hat x\rangle(t)$ و $\langle\hat p\rangle(t)$ وقارنهما
بالحلّ الكلاسيكي. (ج) لماذا \emph{لا} تصف حالة ذاتية للطاقة،
رغم كونها مستقرة، بندولًا يتأرجح --- وأيّ سمة من سمات \omterm{def:b3:harmonic-oscillator:coherent}{الحالة
المترابطة} تفعل؟ (د) قدّر $|\alpha|$
من أجل بندول حقيقي (\qty{10}{g}، وسعة \qty{10}{cm}،
و\qty{1}{Hz}) وعلّق.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:9}
الدارة الكمومية. حلقة فائقة الناقلية فيها $L =
\qty{10}{nH}$ و $C = \qty{0.4}{pF}$ تذبذب الشحنة والتدفق
كما يفعل $x$ و $p$. (أ) احسب تواتر رنينها (كتاب السنة الأولى) و
الكمّ $\hbar\omega$ بواحدة $\unit{\micro eV}$. (ب) تحت أيّ
درجة حرارة يكون $k_{\text{B}}T \ll \hbar\omega$، فتجلس الدارة
في حالتها الأساسية؟ (ج) لماذا تُشغَّل الكيوبتات الفائقة الناقلية في
برادات تخفيف عند \qty{20}{mK}؟ (د) تذبذب جهد النقطة الصفرية
$\Delta V = \Delta q/C$ مع $\Delta q =
\sqrt{\hbar\omega C/2}$: قدّره.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:10}
فان دير فالس من حركة النقطة الصفرية. نمذج ذرتين متعادلتين على
المسافة $R$ بمذبذبَي ثنائي قطب متطابقين (الشحنة $e$ والكتلة
$m$ والتواتر $\omega_0$) يتقارن انزياحاهما بطاقة ثنائي
القطب $\lambda\,x_1x_2$ مع $\lambda = e^2/2\pi\varepsilon_0R^3$.
(أ) بيّن أن الإحداثيين الذاتيين $(x_1 \pm x_2)/\sqrt2$ يذبذبان عند
$\omega_\pm = \omega_0\sqrt{1 \pm \lambda/m\omega_0^2}$. (ب) الطاقة
الأساسية هي $\tfrac\hbar2(\omega_+ + \omega_-)$: انشرها حتى
المرتبة الثانية في $\lambda$ وبيّن أن طاقة التفاعل هي
\[
\Delta E = -\frac{\hbar\lambda^2}{8m^2\omega_0^3}
\ \propto\ -\frac{1}{R^6} .
\]
(ج) لماذا يكون هذا التجاذب كونيًا --- حاضرًا بين ذرات
لا ثنائيات قطب دائمة لها البتة؟ (د) قانون $1/R^6$ هو قوة فان
دير فالس في تصحيحات الغاز الحقيقي بكتاب السنة الأولى: فما الذي
``يتذبذب'' مجهريًا --- وماذا كان سيحلّ بهذه القوة
في عالم $\hbar = 0$؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:11}
النطاقات الجانبية لأيون محصور. أيون في مصيدة توافقية ($f =
\qty{1}{MHz}$) يمتصّ ضوء ليزر. ولمّا كان الأيون يتحرك، يُظهر
طيف امتصاصه الخطَّ الإلكتروني عند $\nu_0$ محفوفًا
بخطين عند $\nu_0 \pm f$: أي انتقالات تغيّر العدد الكمومي
\emph{الحركي} بالمقدار $\mp1$ إلى جانب الإلكتروني. (أ) كم يبلغ
$\hbar\omega_{\text{مصيدة}}$ بواحدة $\unit{neV}$، ولماذا يحتاج تمييز
النطاقات الجانبية إلى خط ضيق جدًا؟ (ب) قيادة الخط $\nu_0 - f$
تحذف كمًّا حركيًا واحدًا في كل دورة: اشرح \emph{التبريد
بالنطاق الجانبي} حتى الحالة الأساسية. (ج) وحين يصير $\langle n\rangle \approx
0$، يختفي النطاق الجانبي السفلي تمامًا --- فلماذا يكون ذلك
اللاتناظر برهانًا على بلوغ الحالة الكمومية الأساسية؟ (د) هكذا
تُشهَد الحالة الحركية الأساسية لذرة مفردة --- ولمرايا لايغو
ذات الكيلوغرامات، بوسائل أخرى: اذكر أيّ ``درجة حرارة'' بلغها
الأيون من أجل $f = \qty{1}{MHz}$ و
$\langle n\rangle = 0.05$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:12}
النهاية الكلاسيكية كمّيًا. مذبذب كلاسيكي سعته
$A$ يقضي في $[x, x + \dd x]$ النسبة $\dd
t/T = \dd x/\pi\sqrt{A^2 - x^2}$. (أ) استنتج توزيع زمن
المكوث هذا. (ب) من أجل الحالة الكمومية $n$، خذ $A_n$ من $E_n =
\tfrac12 m\omega^2A_n^2$ وقارن التوزيع الكلاسيكي
بالمقدار $|\varphi_n|^2$ الموسَّط موضعيًا (وهو معطى): أين يتفقان
وأين يجب أن يختلفا؟ (ج) بيّن أن التباعد النسبي بين
المستويين المتجاورين، $\Delta E/E$، ينعدم وفق $1/n$: فتصير الطاقة
متصلة فعليًا. (د) من أجل بندول
\cref{exo:b3:harmonic-oscillator:8}(د)، قدّر $n$ و $\Delta
E/E$، واستنتج حجة التقابل في جملة واحدة.
\end{exercise}

\section{مسألة: الجزيء الذي يدفئ الأرض}

\begin{problem}[{مسألة نهاية الأسبوع --- سلّم ثاني أكسيد الكربون
الكمومي وأثر الدفيئة}]\label{pb:b3:harmonic-oscillator:1}
جزيء ثاني أكسيد الكربون سلسلة خطية O=C=O: أي بضعة
مذبذبات مكمّاة. وكون $\hbar\omega$ لنمط انحنائه يقع صدفةً في
منتصف توهج الأرض الحراري الخارج هو سبب توجيه هذا الغاز النزر
--- أربعة جزيئات في عشرة آلاف --- مناخَ الكوكب. المعطيات:
عدد موجة الانحناء $\tilde\nu_2 = \qty{667}{cm^{-1}}$
(وهي وحدة المطيافيين: $E = hc\tilde\nu$، و $\qty{1}{cm^{-1}} =
\qty{1.24e-4}{eV}$)؛ والتمدد غير المتناظر $\tilde\nu_3 =
\qty{2349}{cm^{-1}}$؛ والتمدد المتناظر $\tilde\nu_1 =
\qty{1388}{cm^{-1}}$؛ و $k_{\text{B}}T$ عند $\qty{288}{K}$ يساوي
\qty{24.8}{meV}؛ وقانون فين (كتاب السنة الثانية): $\lambda_{\max}T =
\qty{2898}{\micro m.K}$.

\medskip\noindent\textbf{الجزء الأول --- أنماط جزيء خطي.}
\begin{enumerate}
  \item للذرات الثلاث تسع درجات حرية: فكم منها انسحابات
        للجزيء كله، وكم دورانات (والجزيء \emph{خطي})،
        وكم اهتزازًا يبقى؟
  \item صف الاهتزازات الأربعة: التمدد المتناظر والتمدد
        غير المتناظر وانحناء منحلّ مرتين --- ولماذا يأتي الانحناء
        مرتين؟
  \item يتقارن الضوء مع \emph{ثنائي قطب كهربائي} مهتز
        (\cref{exo:b3:harmonic-oscillator:4}). فأيّ أنماط
        O=C=O تعدّل عزم ثنائي القطب، وأيّها
        صامت في تحت الأحمر؟
  \item حوّل أعداد الموجة الثلاثة إلى أطوال موجات الفوتونات، و
        عيّن مواقعها: أيّها في تحت الأحمر الحراري؟
  \item لماذا لا يمتصّ غازا الهواء الرئيسان، N$_2$ و O$_2$،
        شيئًا يُذكر من تحت الأحمر --- وبأيّ نتيجة على
        شفافية الغلاف الجوّي؟
  \item يمتصّ بخار الماء، وهو منحنٍ وثنائي القطب، عبر معظم
        تحت الأحمر: ففي أيّ ``نافذة'' طيفية يعمل انحناء ثاني أكسيد
        الكربون وحده تقريبًا (قارن \qty{15}{\micro m} بنطاقات
        الماء القوية دون \qty{8}{\micro m} وفوق
        \qty{20}{\micro m})؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- السلّم الكمومي للانحناء.}
\begin{enumerate}[resume]
  \item احسب $\hbar\omega_2$ بالميلي إلكترون-فولط، والسلّم $E_n$.
  \item ما نسبة الجزيئات التي تعمر $n = 1$ عند
        \qty{288}{K} (بالنسبة إلى $n = 0$، بعامل بولتزمان)؟ وما نسبة
        $n = 2$؟
  \item أيّ طول موجة فوتونية يقود $n = 0 \to 1$؟ ولماذا يسود
        طول الموجة \emph{نفسه} في الإصدار؟
  \item برّر انطلاقًا من السلّم التوافقي أن طول موجة واحدًا يخدم
        السلّم كله ($n \to n + 1$ من أجل كل $n$): فأيّ
        خاصية من خواصّ تباعد المستويات تعمل هنا؟
  \item يدور الجزيء أيضًا فيضيف بنية دقيقة: فمعلم
        \qty{15}{\micro m} في الحقيقة \emph{نطاق} عرضه نحو
        \qty{1}{\micro m}. فلماذا يهمّ \emph{عرض} النطاق
        في غاز دفيئة (فكّر فيما يحدث متى صار مركز النطاق
        معتمًا)؟
  \item جزيء CO$_2$ مثار اهتزازيًا في الهواء أرجح كثيرًا أن
        يفقد كمّه بالتصادم لا بالإشعاع (فعمر الإشعاع
        $\sim\qty{1}{s}$ وزمن التصادم $\sim\qty{e-9}{s}$):
        فأين تذهب الطاقة المشعّة الممتصّة فعلًا؟
  \item وبالعكس، يبقي الهواء عند \qty{288}{K} جمهرة حرارية
        في $n = 1$ (السؤال 8): فماذا تفعل تلك الجمهرة ممّا
        يهمّ ميزان الطاقة؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- دفتر إشعاع الكوكب.}
\begin{enumerate}[resume]
  \item تشعّ الشمس كجسم عند \qty{5800}{K}: احسب ذروة فين
        لها. فهل يعترض انحناء CO$_2$ كثيرًا من ضوء الشمس؟
  \item تشعّ الأرض كجسم عند \qty{288}{K}: احسب ذروة فين
        لها، وعيّن موقع \qty{15}{\micro m} على ذلك المنحني
        الحراري.
  \item اشرح آلية الدفيئة في أربع جمل:
        دخول ضوء الشمس، وخروج تحت الأحمر الحراري، والاعتراض عند
        \qty{15}{\micro m}، وإعادة الإصدار صعودًا \emph{ونزولًا}.
  \item إعادة الإصدار التي تفلت إلى الفضاء تأتي من طبقات عالية
        باردة ($\sim\qty{220}{K}$): فلماذا يخفّض الإصدار من طبقة
        أبرد الاستطاعة الخارجة من الكوكب عند تلك
        الأطوال (وتذكّر أن الإصدار الحراري ينمو مع $T$)؟
  \item ويدفع مزيد من CO$_2$ الطبقة المصدِرة أعلى وأبرد:
        اذكر في جملة واحدة لماذا يجب أن يسخن السطح عندئذٍ
        لإعادة توازن الدفاتر.
  \item مركز النطاق معتم أصلًا؛ وأثر CO$_2$ المضاف
        يعمل في \emph{جناحَي} النطاق، فيعطي نموًا لوغاريتميًا
        للتأثير مع التركيز: اربط هذا بالسؤال 11.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- النظائر: البصمة
الكمومية.}
\begin{enumerate}[resume]
  \item يتغير تواتر المذبذب وفق $\sqrt{k/\mu}$: ففي التمدد
        غير المتناظر، إذا استُبدل بالنظير $^{12}$C النظيرُ $^{13}$C
        تغيّرت الكتلة الفعّالة؛ وينزاح الخط المشاهَد من
        \qty{2349}{cm^{-1}} إلى نحو \qty{2283}{cm^{-1}}. تحقق من
        رتبة هذا الانزياح البالغ $\approx 3\%$ انطلاقًا من
        الكتل.
  \item ليزران موالفان على هذين الخطين يحصيان $^{13}$CO$_2$ و
        $^{12}$CO$_2$ منفصلين في عينة غازية: اشرح لماذا يجعل
        السلّم المكمّى هذا الكشف المميِّز للنظائر ممكنًا أصلًا.
  \item تفضّل النباتات النظير الأخفّ، فالكربون الأحفوري
        فقير في $^{13}$C: فماذا فعلت النسبة النظيرية المقيسة في
        CO$_2$ الجوّي مع ارتفاع تركيزه --- و
        بم يبرهن ذلك في شأن \emph{مصدر} الغاز المضاف؟
  \item فيزياء \qty{15}{\micro m} نفسها تعمل على الزهرة
        ($96\%$ من CO$_2$، وعند \qty{90}{bar}): فبم يوضّح
        سطحها عند \qty{737}{K}؟
  \item والمريخ يتنفس أيضًا CO$_2$ شبه صافٍ، لكن عند \qty{6}{mbar}،
        وهو قارس البرودة: فبم يبرهن ثلاثي الزهرة والأرض والمريخ
        في شأن أيّ متغير يتحكم في قوة الأثر؟
  \item لخّص النتيجة الموسومة: كمّ مقداره \qty{83}{meV} ---
        أي $\hbar\omega$ لجزيء ثلاثي الذرات منحنٍ --- موضوعًا عند
        \qty{15}{\micro m} على الطيف الحراري لكوكب عند \qty{288}{K}،
        ممتصًّا ومحوَّلًا إلى حرارة في نانوثوانٍ ومعادًا
        إصداره من ارتفاعات باردة، هو الآلية التي يضبط بها
        $0.04\%$ من الهواء درجةَ حرارة الأرض.
\end{enumerate}
\end{problem}
