\chapter{الجمهور القانوني}\label{ch:b3:canonical-ensemble}

الجمل المعزولة خيال نظري: فالعينات الحقيقية تجلس في
منظِّمات حرارية وغرف ومحيطات من هواء --- على تماسّ مع خزان
لا يثبّت طاقتها بل \emph{درجة حرارتها}. وقد لقي كتاب السنة
الثانية القانونَ الناتج تجريبيًا: أي أن احتمال حالة
طاقتها $E$ يحمل العامل $\eu^{-E/k_{\text{B}}T}$. ويستنتج
هذا الفصل عامل بولتزمان ذاك في أربعة أسطر من
إحصاء \cref{ch:b3:microcanonical-ensemble}، ثم يبني
الآلة التي تجعل الفيزياء الإحصائية علمًا صناعيًا:
\emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{دالة التقسيم}} $Z$، أي مجموع واحد تسقط منه الطاقة والإنتروبيا
والضغط والسعة الحرارية والتذبذبات كلها
بالاشتقاق. وتُروى هنا انتصارات الآلة الأولى:
لماذا تموت السعات الحرارية عند درجة الحرارة المنخفضة (أينشتاين، 1907 --- أول
نظرية كمومية للمادة)، ولماذا تتسلق سعة الهدروجين الحرارية
درجًا، ولماذا تخفّ السماء أُسّيًا --- وكيف
عدّ بيران، وهو يراقب حبيبات مجهرية تترسب في قطرة ماء،
عددَ أفوغادرو وأنهى الجدل حول واقعية
الذرات.

\section{توزيع بولتزمان، مستنتَجًا}

\begin{theorem}[التوزيع القانوني]\label{thm:b3:canonical-ensemble:boltzmann}
\index{توزيع بولتزمان}\index{الجمهور القانوني}\index{دالة التقسيم}
الجملة التي تتبادل الطاقة مع خزان كبير عند درجة الحرارة $T$
تشغل حالتها المجهرية $s$ (وطاقتها $E_s$) باحتمال
\[
\mathcal P_s = \frac{\eu^{-E_s/k_{\text{B}}T}}{Z} , \qquad
Z = \sum_s \eu^{-E_s/k_{\text{B}}T}
\]
(مع $\beta = 1/k_{\text{B}}T$ فيما يلي). و $Z$، وهي \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{دالة
التقسيم}}، تحجّم الاحتمالات --- ويتبيّن أنها تحمل
ترموديناميك الجملة بأسرها.
\end{theorem}

\begin{proof}
الجملة في الحالة $s$ تترك للخزان الطاقة $E - E_s$:
فبتطبيق \omterm{thm:b3:microcanonical-ensemble:postulate}{المسلَّمة الأساسية} على المجموع يكون $\mathcal P_s
\propto \Omega_{\text{خزان}}(E - E_s)$. وننشر إنتروبيا
الخزان، وهي هائلة وملساء، حتى المرتبة الأولى: $S_{\text{خزان}}(E - E_s)
= S_{\text{خزان}}(E) - E_s\,\partial S_{\text{خزان}}/\partial E =
S_{\text{خزان}}(E) - E_s/T$ بتعريف درجة الحرارة. ومنه
$\Omega_{\text{خزان}} \propto \eu^{-E_s/k_{\text{B}}T}$. (وتموت المراتب
الأعلى مع حجم الخزان.)
\end{proof}

\begin{proposition}[الآلة]\label{prop:b3:canonical-ensemble:machine}
\index{الطاقة الحرة}\index{السعة الحرارية!والتذبذبات}
انطلاقًا من $Z(T, V, N)$:
\[
\langle E\rangle = -\frac{\partial\ln Z}{\partial\beta} , \qquad
F = -k_{\text{B}}T\ln Z , \qquad
S = -\frac{\partial F}{\partial T} , \qquad
P = -\frac{\partial F}{\partial V} ,
\]
حيث $F = \langle E\rangle - TS$ هي \emph{\omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة}}؛ وتكون
السعة الحرارية \emph{تذبذبًا}:
\[
C = \frac{\partial\langle E\rangle}{\partial T}
 = \frac{\langle E^2\rangle - \langle E\rangle^2}
{k_{\text{B}}T^2} .
\]
ومن أجل جمل جزئية مستقلة وقابلة للتمييز يكون $Z = z^N$؛ ومن أجل $N$
جسيمًا متطابقًا في علبة مشتركة يكون $Z = z^N/N!$. وعند $T$ و
$V$ ثابتين \emph{يصغّر} التوازن $F$: فالطبيعة تقايض الطاقة
بالإنتروبيا بسعر الصرف $T$ --- وهو المبدأ الأنفع
في فيزياء المادة.
\end{proposition}

\begin{proof}[برهان جزئي]
$-\partial_\beta\ln Z = \sum E_s\eu^{-\beta E_s}/Z = \langle
E\rangle$؛ ومشتق آخر يعطي $\langle E^2\rangle - \langle
E\rangle^2$، وتحوّل قاعدة السلسلة $\partial_\beta$ إلى
$\partial_T$. وأما تعيين $F$ و $S$ و $P$ فيتبع
مقارنةَ $\dd(-k_{\text{B}}T\ln Z)$ بالعبارة الترموديناميكية
$\dd F = -S\dd T - P\dd V$؛ وأما التحليل إلى عوامل فأُسّي
مجموع. وأما مبدأ الأصغرية: فتناقص $F_{\text{جملة}}$ هو
تزايد $S_{\text{كلي}}$، ما دام $\Delta S_{\text{خزان}} =
-\Delta E_{\text{جملة}}/T$.
\end{proof}

\section{الانتصارات الأولى}

\begin{example}[مستويان، في سطرين]\label{ex:b3:canonical-ensemble:twolevel}
من أجل المستويين $0, \epsilon$: يكون $z = 1 + \eu^{-\beta\epsilon}$، ومنه
$\langle E\rangle = N\epsilon/(\eu^{\beta\epsilon} + 1)$ --- وهي
النتيجة التي كلّفت الطريق الميكروقانوني صفحةً من ستيرلنغ
(\cref{exo:b3:microcanonical-ensemble:8}). فالصياغة القانونية
هي الميكروقانونية والتوافيق فيها مهضومة سلفًا؛ وتتبع
حدبة شوتكي في $C(T)$ باشتقاق واحد.
\end{example}

\begin{example}[جسم أينشتاين الصلب، وموت ديولونغ--بتي]\label{ex:b3:canonical-ensemble:einstein}
\index{جسم أينشتاين الصلب}\index{قانون ديولونغ--بتي}
نمذج بلورةً بوصفها $3N$ مذبذبًا كموميًا تواترها $\omega$
(\cref{ch:b3:harmonic-oscillator}). فلكل مذبذب،
\[
z = \sum_n\eu^{-\beta\hbar\omega(n + 1/2)}
 = \frac{1}{2\sinh(\beta\hbar\omega/2)} , \qquad
\langle E\rangle = \frac{\hbar\omega}{2} +
\frac{\hbar\omega}{\eu^{\beta\hbar\omega} - 1} .
\]
وعند $T$ المرتفعة: $\langle E\rangle \to k_{\text{B}}T$ لكل مذبذب،
و $C \to 3Nk_{\text{B}}$ --- أي \omterm{ex:b3:canonical-ensemble:einstein}{قانون ديولونغ--بتي} في كتاب السنة
الأولى، مشروحًا. وعند $T$ المنخفضة يصير الكمّ $\hbar\omega$
فوق الطاقة، فتنهار $C$ انهيارًا أُسّيًا --- كما قيس، وكما
تعذّر شرحه كلاسيكيًّا. وكان منحني أينشتاين لسنة 1907، بوسيط واحد لكل
عنصر، أول تطبيق للكمّات على المادة العادية؛
ولا يزال الألماس، بروابطه الصلبة وذراته الخفيفة ($\hbar\omega/k_{\text{B}}
\approx \qty{1300}{K}$)، ``مجمَّدًا'' عند حرارة الغرفة ---
وهو الشذوذ الذي حيّر الكيميائيين ثمانين سنة
(\cref{exo:b3:canonical-ensemble:6}).
\end{example}

\begin{theorem}[التوزّع المتساوي، ورخصته]\label{thm:b3:canonical-ensemble:equipartition}
\index{مبرهنة التوزّع المتساوي}
كل إحداثي أو كمية حركة تدخل الطاقة دخولًا \emph{تربيعيًا}
تسهم، في النظام الكلاسيكي (ذي درجة الحرارة المرتفعة)، بالمقدار
\[
\langle\epsilon\rangle = \tfrac12 k_{\text{B}}T
\]
في متوسط الطاقة: أي $\tfrac32 k_{\text{B}}T$ لذرة غاز
وحيدة الذرة، و $\tfrac52$ لجزيء ثنائي الذرة دوّار، و $3k_{\text{B}}T$
لمذبذب. وتنتهي صلاحية الرخصة حين يهبط $k_{\text{B}}T$ تحت
تباعد مستويات النمط: فيتجمّد النمط وتنعدم مساهمته
--- وهو حلّ فضائح السعة الحرارية في القرن التاسع عشر،
مرسومًا درجًا في الأعلى.
\end{theorem}

\begin{proof}[برهان جزئي]
من أجل $\epsilon = ax^2$،
\[
\langle\epsilon\rangle
 = \frac{\int ax^2\,\eu^{-\beta ax^2}\dd x}
        {\int\eu^{-\beta ax^2}\dd x}
 = -\partial_\beta\ln\!\int\eu^{-\beta ax^2}\dd x
 = -\partial_\beta\ln\beta^{-1/2} = \frac{1}{2\beta} .
\] والتجمّد هو
حساب \cref{ex:b3:canonical-ensemble:einstein}، نمطًا
نمطًا.
\end{proof}

\begin{example}[الغاز، قانونيًا]\label{ex:b3:canonical-ensemble:gas}
ذرة واحدة في علبة: $z = V/\lambda_T^3$ (أي إحصاء الحالات في
\cref{prop:b3:schrodinger-three-dimensions:counting} موزونًا
ببولتزمان)؛ ومن أجل $N$ ذرة متطابقة: $Z = z^N/N!$. ومنه $F =
-Nk_{\text{B}}T[\ln(V/N\lambda_T^3) + 1]$، ويخرج الاشتقاق
$PV = Nk_{\text{B}}T$، وإنتروبيا ساكور--تترود، و
$\langle E\rangle = \tfrac32 Nk_{\text{B}}T$ --- أي الغاز المثالي كله
من تكامل غاوسي واحد. وعامل بولتزمان مطبَّقًا على الطاقة
الحركية لجزيء يعطي توزيع ماكسويل للسرعات في
كتاب السنة الأولى، مستنتَجًا الآن؛ ومطبَّقًا على طاقته الكامنة
$mgh$ يعطي الجوّ الأُسّي --- ويعطي، في قطرة
ماء، سلّم بيران للحبيبات
(\cref{pb:b3:canonical-ensemble:1}).
\end{example}

\begin{method}[الحرفة القانونية]\label{met:b3:canonical-ensemble:method}
(1) اذكر حالات \emph{وحدة} واحدة وطاقاتها؛ واحسب $z$.
(2) وحلّل إلى عوامل: $Z = z^N$ (أو $z^N/N!$ من أجل \omterm{def:b3:identical-particles:exchange}{جسيمات متطابقة}
تتقاسم الفضاء)؛ وخذ $\ln$ باكرًا. (3) واشتقّ: بالنسبة إلى $\beta$ من أجل
الطاقة، وإلى $T$ من أجل الإنتروبيا عبر $F$، وإلى $V$ من أجل الضغط؛ ومشتق
ثانٍ من أجل $C$ والتذبذبات. (4) وتحقق من الطرفين: فدرجة الحرارة $T$ المرتفعة
يجب أن تعيد التوزّع المتساوي، و $T$ المنخفضة يجب أن تتجمّد بذيل
$\eu^{-\Delta/k_{\text{B}}T}$. (5) وأما المنافسات (الانطواء، و
الارتباط، والمحاذاة): فاكتب $F = E - TS$ لكل بديل و
دع الأصغر يفوز --- ويقع الانتقال عند $T \approx \Delta
E/\Delta S$.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:1}
(أ) اكتب نسبة جمهرتَي مستويين ينفصمان بالمقدار $\epsilon$ عند
درجة الحرارة $T$. (ب) في لهب عند \qty{2500}{K}، ما نسبة
ذرات الصوديوم الجالسة في الحالة المثارة عند \qty{2.1}{eV} لخط D
(بانحلال المستوي الأساسي 2 والمثار 6: أي نسبة جمهرتين
$3\eu^{-\beta\epsilon}$)؟
(ج) ولماذا يتوهج اللهب أصفر مع ذلك (وكم ذرة في
السنتيمتر المكعب cm$^3$ تكفي)؟ (د) وعند أيّ درجة حرارة تبلغ النسبة المثارة
$10\%$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:2}
جملة ذات ثلاثة مستويات: $0, \epsilon, 2\epsilon$. (أ) اكتب $z$. (ب)
احسب $\langle E\rangle$ وتحقق من نهايتَي درجة الحرارة. (ج)
عند أيّ $T$ يكون المستوي الأوسط أكثر ما يكون جمهرةً \emph{بالقيمة
المطلقة}؟ (د) بيّن أنه لا درجة حرارة، مهما ارتفعت، تجعل
مستوًى أعلى أكثر جمهرةً من مستوًى أدنى --- وأيّ مفهوم من الفصل 16 يلزم لذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:3}
الجوّ متساوي الحرارة. (أ) طبّق عامل بولتزمان على
الطاقة الكامنة $mgh$ واستنتج $n(h) = n_0\eu^{-mgh/k_{\text{B}}
T}$. (ب) احسب ارتفاع المقياس للهواء ($m = \qty{4.8e-26}{kg}$)
عند \qty{288}{K}. (ج) احسب ضغط قمة إفرست كنسبة من ضغط سطح
البحر. (د) ولماذا يكون هبوط الجوّ الحقيقي قريبًا من الأُسّي لا
أُسّيًا بالضبط (فما الذي ثبّتناه وهو ليس بثابت)؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:4}
مسك دفاتر التوزّع المتساوي. أحصِ الحدود التربيعية وتنبّأ
بالمقدار المولي $C_V$ من أجل (أ) الأرغون؛ (ب) N$_2$ عند حرارة الغرفة
(بدوران يعمل واهتزاز مجمَّد)؛ (ج) N$_2$ عند \qty{3000}{K}؛ (د)
جسم صلب بلوري (ديولونغ--بتي). فأين توافق القيم المقيسة
$12.5$ و $20.8$ و $\approx26$ و $\approx\qty{25}{J/(mol.K)}$،
وما الذي يعلّمه كل فرق؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:5}
المذبذب الكمومي، قانونيًا. (أ) اجمع المتتالية الهندسية
من أجل $z$. (ب) استنتج $\langle E\rangle$ وعيّن $\langle
n\rangle = 1/(\eu^{\beta\hbar\omega} - 1)$ --- وهي نتيجة
\cref{exo:b3:harmonic-oscillator:6}، بلا كلفة الآن. (ج)
اشتقّ من أجل $C(T)$ وتحقق من النهايتين. (د) ولماذا توشك صورة
$\langle n\rangle$ أن تصير مشهورة
(\cref{ch:b3:photons-phonons})؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:6}
أينشتاين في مواجهة المعطيات. (أ) من صيغة الشكل، عند أيّ
$T/\theta_{\text{E}}$ تهبط $C$ إلى نصف ديولونغ--بتي؟ (ب)
الألماس: $\theta_{\text{E}} \approx \qty{1300}{K}$ --- احسب
$C/3Nk_{\text{B}}$ عند \qty{300}{K} واشرح ``شذوذ الألماس'' في
القرن التاسع عشر. (ج) الرصاص: $\theta_{\text{E}}
\approx \qty{90}{K}$ --- فلماذا كان الرصاص ``حسن السلوك'' دائمًا؟ (د)
تُظهر القياسات عند $T$ المنخفضة جدًّا أن $C \propto T^3$، لا
هبوطًا أُسّيًا: فأيّ فرضيات أينشتاين تخفق (أي أن للمذبذبات كلها
تواترًا \emph{واحدًا})، ومن أصلحها
(\cref{ch:b3:photons-phonons})؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:7}
ذيل ماكسويل والهدروجين المفقود. (أ) من عامل
بولتزمان على الطاقة الحركية، اكتب توزيع السرعات وعيّن
السرعة الأرجح من أجل N$_2$ و H$_2$ عند \qty{288}{K}. (ب)
سرعة الإفلات من الأرض \qty{11.2}{km/s}: احسب
$mv_{\text{إفلات}}^2/2k_{\text{B}}T$ للغازين. (ج) النسبة المفلتة
تجري وفق $\eu^{-mv_{\text{إفلات}}^2/2k_{\text{B}}T}$: قارن
الأُسّين واستنتج أيّ غاز يتسرب على المدى
الجيولوجي. (د) اربط بالتركيب المشاهَد لجوّ الأرض (بلا
H$_2$ حرّ) مقابل جوّ المشتري (وأكثره H$_2$) --- فأيّ وسيطين
يقلبان الحكم؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:8}
التذبذبات تلقى الاستجابة. (أ) برهن $C = (\langle E^2\rangle -
\langle E\rangle^2)/k_{\text{B}}T^2$ من مشتقين للمقدار $\ln
Z$. (ب) تحقق منها صراحةً على الجملة ذات المستويين. (ج) من أجل $N$
وحدة مستقلة، بيّن أن تذبذب الطاقة \emph{النسبي}
يهبط وفق $1/\sqrt N$. (د) اذكر العبرة: فما علاقة استعداد
جملة لامتصاص الحرارة (وهو استجابة) بمقدار ارتجاف
طاقتها (وهو تذبذب) --- أي مقايضة الفيزياء الإحصائية
المتكررة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:9}
بولتزمان في مخبر الكيمياء. تحمل سرعات التفاعل العامل
$\eu^{-E_a/k_{\text{B}}T}$ (أي عبور حاجز تنشيط $E_a$
--- أرينيوس). (أ) بيّن أن القاعدة التقريبية ``تتضاعف السرعة كل
\qty{10}{K} قرب حرارة الغرفة'' توافق $E_a \approx
\qty{0.55}{eV}$. (ب) وبأيّ عامل يبطؤ ذلك التفاعل في
ثلاجة (\qty{5}{\celsius})؟ (ج) وسلق بيضة في الارتفاع:
يغلي الماء عند \qty{93}{\celsius} على ممرّ عند \qty{2000}{m} ---
فقدّر زمن السلق الإضافي. (د) ولماذا يحكم ذيل الحاجز،
لا متوسط الطاقة، الكيمياءَ (أيّ الجزيئات يتفاعل)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:10}
انطواء ذو حالتين. الجزيء الحيوي إما منطوٍ (طاقته $0$، وله تشكيلة
واحدة) وإما منشور (طاقته $\Delta E > 0$، وله $\Omega_u =
\eu^{\Delta S/k_{\text{B}}}$ تشكيلة). (أ) اكتب النسبة المنطوية
بدلالة $T$. (ب) بيّن أن منتصف ``الانصهار'' هو
$T_{\text{m}} = \Delta E/\Delta S$ وفسّره تعادلًا في $F = E -
TS$. (ج) ومع $\Delta E = \qty{3.0}{eV}$ و $\Delta S =
100\,k_{\text{B}}$ (وهي وحدة تعاونية لبروتين صغير): احسب
$T_{\text{m}}$ وعرض الانتقال. (د) ولماذا تحدّ
\emph{التعاونيةُ} (بانكسار تماسات كثيرة معًا، أي $\Delta E$ كبيرة
\emph{و} $\Delta S$ كبيرة) الانصهارَ --- وكيف تستغل
آلات التدوير الحراري للحمض النوويّ (PCR) هذا بالضبط؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:11}
المغناطيسية المسايرة والتبريد المغناطيسي. $N$ سبينًا مقداره $\tfrac12$ وعزمه
$\mu$ في مجال $B$. (أ) بيّن أن $M = N\mu\tanh(\mu B/k_{\text{B}}T)$
وانشره إلى قانون كوري $M \approx N\mu^2B/k_{\text{B}}T$. (ب)
احسب المحاذاة $\mu B/k_{\text{B}}T$ من أجل عزوم الإلكترون
عند $B = \qty{1}{T}$ و $T = \qty{300}{K}$ وعند \qty{1}{K}. (ج)
إزالة المغنطة الكظومة: مغنِط عند \qty{1}{K}، ثم اعزل، ثم
أنقص $B$ عشرة أضعاف --- بيّن أن ثبات الإنتروبيا يعني ثبات
$\mu B/k_{\text{B}}T$، ومنه تهبط $T$ عشرة أضعاف. (د) وما الذي يضع
أرضية هذه الثلاجة (أي التفاعلات بين السبينات ---
قدّر المقياس ثنائي القطب $\mu_0\mu^2/4\pi a^3$ من أجل $a =
\qty{0.5}{nm}$، بوحدات درجة الحرارة)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:12}
الدرج الدوراني، كمّيًا. مستويات جزيء ثنائي الذرة
الدورانية هي $E_J = BJ(J+1)$، وانحلالها $2J + 1$
(\cref{ch:b3:quantum-angular-momentum}). (أ) اكتب $z_{\text{دوران}}$
وبيّن أنه من أجل $k_{\text{B}}T \gg B$ يصير المجموع
التكامل $k_{\text{B}}T/B$: فيُستعاد $k_{\text{B}}$ التوزّع المتساوي في
$C_V$. (ب) عرّف $\theta_{\text{دوران}} =
B/k_{\text{B}}$ واحسبه من أجل H$_2$ ($B = \qty{7.5}{meV}$) و
N$_2$ ($B = \qty{0.25}{meV}$): فأيّ غاز يظهر الدرجة الدورانية
عند درجات حرارة متاحة؟ (ج) ارسم تقريبيًا درج $C_V(T)$ الكامل للهدروجين
بمصاطبه الثلاث وقائمتيه، واضعًا الأعداد على الاثنتين. (د)
يزيد تعقيدَ السلوك المقيس عند $T$ المنخفضة للغاز H$_2$ مزيجُ
الأورثو--بارا بنسبة 3:1 في
\cref{exo:b3:identical-particles:8}: اذكر في جملة واحدة كيف
يمتدّ إحصاء السبين النوويّ إلى سعة غاز الحرارية.
\end{exercise}

\section{مسألة: عدّ أفوغادرو في قطرة ماء}

\begin{problem}[{مسألة نهاية الأسبوع --- حبيبات بيران و
واقعية الذرات}]\label{pb:b3:canonical-ensemble:1}
في سنة 1908 علّق جان بيران حبيبات راتنج مجهرية في الماء،
وتركها ترسب، وعدّها طبقةً طبقة تحت
مجهر. وكان السلّم الأُسّي الذي وجده هو عامل بولتزمان
مرئيًّا --- ومن ارتفاع مقياسه استخرج
عدد أفوغادرو، فأقنع آخر المتشككين بأن الذرات موجودة.
جائزة نوبل، 1926. والمعطيات: حبيبات غمغم نصف قطرها $r =
\qty{0.212}{\micro m}$، وكثافتها $\rho_{\text{g}} =
\qty{1207}{kg/m^3}$؛ والماء $\rho_{\text{w}} = \qty{999}{kg/m^3}$؛
و $T = \qty{293}{K}$؛ و $g = \qty{9.81}{m/s^2}$.

\medskip\noindent\textbf{الجزء الأول --- عامل بولتزمان المرئي.}
\begin{enumerate}
  \item الحبيبة في الماء تحسّ بالثقالة ناقصةً الدفع: احسب كتلتها
        \emph{الفعّالة} $m' = \tfrac43\pi r^3(\rho_{\text{g}}
        - \rho_{\text{w}})$ ووزنها.
  \item اكتب منحني التركيز التوازني $n(h)$ من
        عامل بولتزمان.
  \item احسب ارتفاع المقياس $h_0 = k_{\text{B}}T/m'g$ بقيمة
        $k_{\text{B}}$ الحديثة.
  \item ولماذا يجب أن تكون الحبيبات بهذا القدر من تساوي الحجم (فكيف
        يتوقف $h_0$ على $r$)؟
  \item قارن $h_0$ بالصيغة نفسها من أجل جزيئات الهواء:
        فلماذا يكون جوّ الحبيبات بارتفاع ميكرومترات بينما
        جوّ الهواء بارتفاع كيلومترات؟
  \item عدّ بيران (في تجربة واحدة) تراكيز نسبية
        $100 : 55 : 30 : 17$ عند أربعة أعماق متساوية التباعد
        بمقدار \qty{30}{\micro m}: تحقق من أن هذا سلّم
        أُسّي واستخرج $h_0$ الخاص به.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- وزن ما لا يُرى.}
\begin{enumerate}[resume]
  \item اعكس: من $h_0$ المقيس في البند 6 ومن $m'g$
        المعلوم، استخرج $k_{\text{B}}$.
  \item كان ثابت الغازات $R = \qty{8.314}{J/(mol.K)}$ معلومًا
        من الكيمياء العيانية: فاجمعه مع
        $k_{\text{B}}$ الذي وجدته لتحصل على عدد أفوغادرو $N_{\text{A}} =
        R/k_{\text{B}}$.
  \item أعطت تجارب بيران $N_{\text{A}}$ بين $5.5$ و
        $\num{7.2e23}$: قارن بالقيمة الحديثة
        \num{6.022e23} وعلّق على الإنجاز بالنظر إلى
        مجهره وساعته.
  \item من $N_{\text{A}}$، احسب كتلة ذرة هدروجين
        واحدة --- وهو العدد الذي أراده أنصار الذرة
        قرنًا كاملًا.
  \item اشرح البنية المنطقية: فأيّ قياسات \emph{عيانية}
        ($R$؛ وحجم الحبيبات وكثافتها؛ والسلّم)
        تجتمع لوزن ذرة واحدة، من دون أن تُرى ذرة قط؟
  \item ولماذا تطيع الحبيبات --- وكتلة كلٍّ منها $10^{10}$ كتلة ذرية ---
        إحصاء بولتزمان نفسه الذي تطيعه الجزيئات (فما الذي في
        استنتاج \cref{thm:b3:canonical-ensemble:boltzmann}
        يعبأ بالحجم)؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- الارتجاف الذي يختم الأمر.}
\begin{enumerate}[resume]
  \item الحبيبات نفسها ترتجف: فصيغة أينشتاين لسنة 1905 في
        الحركة البراونية تعطي $\langle x^2\rangle = 2Dt$ مع $D =
        k_{\text{B}}T/6\pi\eta r$ (وللماء: $\eta =
        \qty{e-3}{Pa.s}$). احسب $D$ من أجل حبيبات بيران.
  \item وكم تسير الحبيبة في دقيقة واحدة؟ وهل كان بيران يستطيع
        قياس ذلك بعدسة عينية ميكرومترية وساعة؟
  \item تحقق بيران من أن $\langle x^2\rangle \propto t$ و
        استخرج $k_{\text{B}}$ \emph{من جديد}، استخراجًا مستقلًا:
        فلماذا حمل طريقان لا صلة بينهما (سلّم ساكن؛ وارتجاف
        متحرك) إلى عدد واحد مثل هذا الثقل البرهاني؟
  \item الارتجاف هو التوزّع المتساوي مطبَّقًا على الحبيبة: فكل
        مركبة سرعة تحمل $\tfrac12 k_{\text{B}}T$.
        قدّر السرعة الحرارية الجذرية المتوسطة للحبيبة ($m \approx
        \qty{4.8e-17}{kg}$).
  \item ولماذا لا تُرى تلك السرعة مباشرةً قط (فما الذي يقطع
        الطيران الحرّ بعد نانومترات)، وما الذي \emph{يُرى}
        بدلًا منها؟
  \item اذكر أيّ فصلين من هذا الكتاب يلتقيان في
        المشاهدة: ميكانيك السحب (كتاب السنة الثانية،
        اللزوجة) وإحصاء هذا الفصل.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- ما الذي حُسم.}
\begin{enumerate}[resume]
  \item كان أوستفالد وماخ يعدّان الذرات خيالات في مسك
        الدفاتر: اذكر في جملة واحدة لماذا أنهى $N_{\text{A}}$
        \emph{معدودًا} من سلالم الحبيبات ذلك الموقف.
  \item اذكر ثلاثة طرق أخرى من سنوات 1900 تقاربت على $N_{\text{A}}$
        نفسه (زرقة السماء،
        \cref{exo:b3:microcanonical-ensemble:12}؛ والتحليل الكهربائي
        مع شحنة الإلكترون؛ وإنتاج الهليوم في النشاط
        الإشعاعي) --- ولماذا كان \emph{التقارب} أهمّ
        من أيّ قيمة مفردة؟
  \item سلّم بيران توازن بين أيّ عملتين من عملات هذا الفصل
        (طاقة تجذب إلى أسفل، وإنتروبيا تنشر إلى أعلى)، وبأيّ سعر؟
  \item الاستعمالات الحديثة للفيزياء نفسها: تدير النابذات
        الفائقة التحليلية البروتينات عند $10^5g$ لتضغط
        ``أجواءها'' في سلالم قابلة للقياس --- بيّن
        أن ضرب $g$ في $10^5$ يقسم $h_0$ على العامل
        نفسه، وقدّر $h_0$ من أجل بروتين كتلته الفعّالة
        \qty{e-22}{kg} عند $10^5g$ و \qty{293}{K}.
  \item مضاعفة نصف قطر الحبيبة تقسم $h_0$ على ثمانية: فحدّد
        النافذة العملية لأحجام الحبيبات بين ``سلّم أطول
        من أن يُرى فيه تدرّج'' و``سلّم أرقّ من حبيبة واحدة''
        من أجل حقل مجهر عمقه \qty{100}{\micro m}.
  \item منذ سنة 2019 \emph{يعرّف} النظام الدولي $k_{\text{B}}$ و
        $N_{\text{A}}$ تعريفًا مضبوطًا: اذكر ما الذي تقيسه تجربة
        على طريقة بيران \emph{اليوم} (أهو تحقق من الاتساق، أم
        معايرة للجهاز والحبيبات) --- ولماذا تبقى
        الفيزياء على حالها.
  \item لخّص النتيجة الموسومة: تركيز حبيبة قطرها \qty{0.2}{\micro m}
        ينتصف كل $\sim\qty{35}{\micro m}$ من
        الارتفاع؛ ومقروءًا بالعلاقة $n(h) = n_0\eu^{-m'gh/k_{\text{B}}T}$،
        أعطى ذلك السلّم $N_{\text{A}} \approx \num{6e23}$ ---
        أي ذرات معدودة لا مفترضة، في قطرة ماء على
        مسرح مجهر.
\end{enumerate}
\end{problem}
