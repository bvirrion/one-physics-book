\chapter{الجسيمات المتطابقة}\label{ch:b3:identical-particles}

كل إلكترون في الكون مطابق تمامًا لكل إلكترون
آخر --- لا شبيه به كما تتشابه قطعتا نقد من دار سكّ واحدة، بل
غير مميَّز عنه مبدئيًا: فلا وسم ولا خدش ولا تاريخ يميّز
واحدًا منها. والميكانيك الكمومي يأخذ ذلك على محمل الجدّ، وتدير
نتائجه العالم. فتبديل جسيمين متطابقين يجب ألّا يغيّر شيئًا
قابلًا للرصد، وهذا لا يترك للدالة الموجية المتعددة الجسيمات غير
خيارين: أن تبقى على حالها (\emph{البوزونات}) أو أن تغيّر
إشارتها (\emph{الفرميونات}). ومن إشارة الناقص وحدها يتبع \omterm{cor:b3:identical-particles:pauli}{مبدأ
الاستبعاد لباولي}، وبنية الأغلفة \omterm{prop:b3:identical-particles:periodic}{والجدول الدوري}
كله، وصلابة المادة تحت أقدامنا، والضغط الذي يحمل
النجوم الميتة؛ ومن إشارة الزائد تتبع اجتماعية
الفوتونات --- ومن ثمّ الليزر --- ثم، بعد ثلاثة فصول،
تكثّف بوز--أينشتاين. وقلّ في الفيزياء أن حملت إشارة واحدة
هذا القدر.

\section{مسلَّمة المناظرة}

\begin{definition}[التبديل]\label{def:b3:identical-particles:exchange}
\index{جسيمات متطابقة}\index{مؤثر التبديل}
من أجل جسيمين متطابقين فضاء حالاتهما مولَّد بالجداءات
$\ket{a}\otimes\ket{b}$ (فالجسيم 1 في $\ket a$ والجسيم 2 في
$\ket b$ --- بالمواضع والسبينات وكل شيء)، يبدّل \emph{مؤثر التبديل}
$\hat P_{12}$ الدورين: $\hat P_{12}\ket a\otimes\ket b =
\ket b\otimes\ket a$. وهو هرميتي، ومربعه المطابقة، ويتبادل
مع أيّ \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{هاميلتونية} تعامل الجسيمين معاملة واحدة:
فقيمتاه الذاتيتان $\pm1$ وسمان محفوظان للعالم.
\end{definition}

\begin{theorem}[البوزونات والفرميونات]\label{thm:b3:identical-particles:postulate}
\index{بوزون}\index{فرميون}\index{مبرهنة السبين والإحصاء}
حالات \omterm{def:b3:identical-particles:exchange}{الجسيمات المتطابقة} ليست مسموحة فحسب بل \emph{ملزَمة}
بأن تكون حالات ذاتية لكل تبديل: أي \emph{متناظرة} تناظرًا تامًّا
من أجل صنف من الجسيمات --- هي \emph{البوزونات} ---
و\emph{متخالفة التناظر} تخالفًا تامًّا (أي تغيّر الإشارة عند كل تبديل)
من أجل الصنف الآخر --- هي \emph{الفرميونات}. ويحدّد سبينُ النوع
الصنفَ الذي ينتمي إليه (وهي \emph{\omterm{thm:b3:identical-particles:postulate}{مبرهنة السبين والإحصاء}}، ولا تُبرهن
إلا في نظرية الحقول الكمومية النسبية وهي مُسلَّم بها هنا):
\[
\text{سبين صحيح} \Rightarrow \text{بوزون} , \qquad
\text{سبين نصف صحيح} \Rightarrow \text{فرميون} .
\]
فالفوتونات والبيونات وذرات $^4$He بوزونات؛ والإلكترونات والبروتونات
والنترونات و$^3$He فرميونات.
والمركَّب يسلك سلوك مجموع أجزائه: فعدد زوجي من
الفرميونات المرتبطة بوزون ($^4$He: بروتونان ونترونان وإلكترونان)، وعدد
فردي فرميون ($^3$He) --- ولذلك نتائج ظاهرة إلى حدّ وجود
سائلَي هليوم مختلفين.
\end{theorem}
\admitted

\begin{corollary}[مبدأ الاستبعاد لباولي]\label{cor:b3:identical-particles:pauli}
\index{مبدأ الاستبعاد لباولي}
لا يمكن أبدًا لفرميونين أن يشغلا الحالة الأحادية الجسيم نفسها: فوضع
$\ket a = \ket b$ في التركيب المتخالف التناظر
$(\ket a\otimes\ket b - \ket b\otimes\ket a)/\sqrt2$ يعطي المتجهة
المعدومة --- فلا وجود لحالة كهذه. ومن أجل الإلكترونات (سبينها $\tfrac12$)،
يقبل كل مدار اثنين على الأكثر، بإسقاطَي سبين متعاكسين.
\end{corollary}

\begin{example}[إحصاء مختلف]\label{ex:b3:identical-particles:counting}
جسيمان بين حالتين $a, b$: يعطي الإحصاء الكلاسيكي (أي بجسيمات
مميَّزة) أربعة تشكيلات؛ وللبوزونات ثلاثة ($aa$ و $bb$
وحالة مختلطة \emph{واحدة})؛ وللفرميونات واحد بالضبط (وهو المختلط
المتخالف التناظر). وسترث الفيزياء الإحصائية هذه الأعداد
جملةً (\cref{ch:b3:quantum-statistics})؛ وهي هنا سلفًا تحلّ
إحراجًا من القرن التاسع عشر --- أي إنتروبيا ``مزج''
عيّنتين من الغاز نفسه
(\cref{exo:b3:identical-particles:10}).
\end{example}

\section{الجدول الدوري}

\begin{proposition}[الأغلفة، ولماذا توجد الكيمياء]\label{prop:b3:identical-particles:periodic}
\index{الجدول الدوري}\index{مبدأ البناء}
املأ المدارات الشبيهة بالهدروجينية $(n, l, m, m_s)$ بإلكترون
واحد لكل منها، من أدنى الطاقات أولًا، بترتيب المستويات الذي يحدّده النفاذ
والحجب ($1s$ و $2s$ و $2p$ و $3s$ و $3p$ و $4s$ و $3d$ \dots ---
\cref{exo:b3:hydrogen-atom:8}). ويجعل مبدأ باولي الملء
\emph{يتشبّع}: فيغلق $2$ إلكترون الغلاف الأول، ويغلق $8$
يغلقون الصف الثاني والثالث، فيكتسب جدول العناصر
دوراته. والكيمياء هي فيزياء الإلكترونات التي تركها مبدأ
الاستبعاد في الخارج: فالأغلفة الممتلئة (الغازات النبيلة)
خاملة؛ وإلكترون واحد وراء غلاف ممتلئ (القلويات) أو نقصان
إلكترون واحد عن الامتلاء (الهالوجينات) نشطان في اتجاهين متعاكسين. ولولا إشارة
الناقص لغرق كل إلكترون في $1s$، ولكانت كل ذرة كرة
منكمشة شبيهة بالهليوم، ولما كانت روابط ولا
جزيئات ولا كيميائيون يأسفون على ذلك.
\end{proposition}
\admitted

\section{تفاعل التبديل}

\begin{proposition}[تناظر حالة الزوج الفضائية]\label{prop:b3:identical-particles:exchange-force}
\index{تفاعل التبديل}
من أجل إلكترونين، يجب أن تكون الحالة الكلية (الفضاء $\times$ السبين)
متخالفة التناظر: فالحالة \emph{الأحادية} (وسبينها متخالف التناظر،
\cref{exo:b3:spin-two-level:11}) تفرض دالة موجية فضائية
\emph{متناظرة}، والحالة \emph{الثلاثية} تفرض متخالفة التناظر. وتختلف الحالتان
فيزيائيًّا: ففي الحالة الفضائية المتخالفة التناظر لا يمكن للإلكترونين
أن يتطابقا أبدًا ($\psi(\vect r, \vect r) = 0$) فيبقيان متباعدين، بينما
تدع الحالة المتناظرة لهما أن يتكدّسا. ولمّا كان القرب يكلّف طاقة
كولوم، اكتسب ترتيب السبينين فرقًا في الطاقة --- هو
\emph{طاقة التبديل} --- بحجم كهرستاتيكي (أي بالإلكترون-فولط!)،
مع أنه لا وجود لقوة مغناطيسية بتلك القوة: فالسبينات تصطفّ أو
تتعاكس بسبب الهندسة مع التنافر. وهذا هو محرك
قاعدة هوند، وطيفَي الهليوم، والرابطة التساهمية، و---
مكبَّرًا عبر بلورة --- المغناطيسية الحديدية نفسها
(\cref{ch:b3:electromagnetism-in-matter}).
\end{proposition}

\begin{proof}[برهان جزئي]
المحاسبة التناظرية هي
\cref{thm:b3:identical-particles:postulate} مع تراكيب
السبين؛ والانعدام عند التطابق هو تخالف التناظر مقروءًا عند
$\vect r_1 = \vect r_2$. وأما نتيجة الطاقة فهي \omterm{thm:b3:perturbation-theory:corrections}{نظرية الاضطرابات}
في المرتبة الأولى في التنافر: فالحدّ المباشر مشترك بين
التناظرين ويدخل التكامل المتقاطع (``التبديلي'') بإشارتين
متعاكستين (\cref{exo:b3:identical-particles:6}).
\end{proof}

\begin{example}[الهليومان في ذرة واحدة]\label{ex:b3:identical-particles:helium}
تأتي حالات الهليوم المثارة $1s\,2s$ في عائلتين لا تكادان
تتداخلان: \emph{البارا}هليوم (أحادي، بفضاء متناظر) و\emph{الأورثو}هليوم
(ثلاثي، بفضاء متخالف التناظر)، والثلاثي \emph{أدنى}
بنحو $\approx\qty{0.8}{eV}$ --- إذ يوفّر ثقب تبديله
تنافر كولوم. والحالة الأساسية بارا بالضرورة (فالتشكيل $1s^2$
يفرض الحالة الفضائية المتناظرة)؛ وأدنى حالة أورثو،
إذ لا ثلاثي أدنى تتفكك إليه وانقلابات السبين ممنوعة، تعيش
ساعات --- أي ذرة شبه مستقرة تُستعمل حصانَ عملٍ في الحزم
الذرّية. وقد فهرس مطيافيو القرن التاسع عشر ``هليومين''؛
والحلّ إشارة ناقص واحدة.
\end{example}

\begin{example}[الرابطة التساهمية]\label{ex:b3:identical-particles:bond}
قرّب ذرتَي هدروجين إحداهما من الأخرى: يستطيع الإلكترونان أن يدخلا
التركيب الفضائي المتناظر للمدارين الذرّيين (بسبينين
أحاديين) أو التركيب المتخالف التناظر (بسبينين ثلاثيين). فالحالة المتناظرة
تكدّس الشحنة \emph{بين} البروتونين، حيث تجذبهما معًا:
أي ربط --- وهو الرابطة التساهمية في H$_2$، وعمقها \qty{4.5}{eV}. أما
الحالة المتخالفة التناظر فتخلي المستوي الأوسط: فهي دافعة صرفة. وكل
سطر في الكيمياء العضوية مسك دفاتر لإلكترونات مزدوجة السبين
مشتركة تناظريًا --- فمبدأ الاستبعاد، مقروءًا بالعكس، هو
صمغ الجزيئات.
\end{example}

\section{البوزونات: كلما كثروا كان أفضل}

\begin{proposition}[التعزيز البوزوني]\label{prop:b3:identical-particles:enhancement}
\index{الإصدار المستحثّ!أصله البوزوني}
الانتقالات التي تضيف بوزونًا إلى حالة تحوي $n$ منها سلفًا
لها سعات معزَّزة بالعامل $\sqrt{n + 1}$ --- ومعدلات بالعامل $n + 1$
(بجبر السلّم في \cref{ch:b3:harmonic-oscillator}، إذ إن النمط
البوزوني \emph{هو} مذبذب). والحدّ ``$1$'' هو الإصدار
التلقائي؛ والحدّ ``$n$'' هو الإصدار \emph{المستحثّ}، المتناسب
مع الجمهور الحاضر سلفًا: أي الانهيال وراء كل ليزر
(كتاب السنة الثانية)، وقد انكشف الآن تناظرَ تبديل صرفًا. والإحصاء
نفسه، مطبَّقًا على ذرات مبرَّدة حتى تتراكب حزمها الموجية،
يتنبأ بتدافع نحو حالة كمومية واحدة ---
أي تكثّف بوز--أينشتاين، المحفوظ إلى
\cref{ch:b3:quantum-statistics}.
\end{proposition}
\admitted

\begin{method}[معالجة الجسيمات المتطابقة]\label{met:b3:identical-particles:method}
(1) حدّد الإحصاء من السبين (وفي المركَّبات: اجمع عدد
الفرميونات). (2) ومن أجل إلكترونين، حلّل الحالة إلى فضاء $\times$ سبين
واقرن الأحادي بالفضاء المتناظر والثلاثي بالمتخالف التناظر.
(3) وأحصِ الحالات لا الجسيمات بأسمائها: اذكر الإشغالات.
(4) ومسائل الطاقة مع التنافر: تكامل مباشر واحد، وتكامل
تبديلي بإشارة التناظر. (5) وتذكّر الشعارين ---
الفرميونات: واحد على الأكثر لكل حالة، وثقب تبديل حول كل منها؛
والبوزونات: معدلات معزَّزة بالإشغال.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:identical-particles:1}
(أ) بيّن أن $\hat P_{12}$ هرميتي وأن $\hat P_{12}^2 = \mathbb 1$:
فما قيمه الذاتية الممكنة؟ (ب) بيّن أن $[\hat P_{12}, \hat H] =
0$ من أجل \omterm{def:b3:identical-particles:exchange}{جسيمات متطابقة} واستنتج انحفاظ صنف التناظر.
(ج) لماذا لا يمكن لحالة جسيمين متطابقين أن ``تكون'' ببساطة
$\ket a\otimes\ket b$ بلا تناظر؟ (د) صنّف:
$^1$H و$^2$H (الديوتريوم) و$^4$He و$^3$He، وجزيء هدروجين، و
فوتونًا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:identical-particles:2}
(أ) اكتب الحالة المتخالفة التناظر المنظَّمة المبنية من مدارين
متعامدين $\ket a, \ket b$. (ب) بيّن أنها تنعدم لمّا $a = b$. (ج)
ومن أجل ثلاثة فرميونات في $a, b, c$: اكتب التركيب المتخالف
التناظر تمامًا (ستة حدود بإشارات متناوبة --- أي محدد
$3\times3$) وتحقق من أن تبديلًا واحدًا يقلب إشارته. (د) لماذا تجعل
صورة المحدد مبدأَ باولي تلقائيًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:identical-particles:3}
جسيمان يتقاسمان ثلاث حالات متعامدة. أحصِ حالات الجسيمين
من أجل (أ) جسيمين مميَّزين؛ (ب) بوزونين؛ (ج) فرميونين
(بلا سبين). (د) وأعد (ج) من أجل إلكترونين بسبين: فكم حالة
الآن؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:identical-particles:4}
(أ) أعطِ التشكيلات الأساسية للكربون والنتروجين والأكسجين و
النيون. (ب) اذكر قاعدة هوند وطبّقها على إلكترونات النتروجين الثلاثة في $2p$.
(ج) برّر قاعدة هوند في جملة واحدة انطلاقًا من
\cref{prop:b3:identical-particles:exchange-force}. (د) أيّ
عنصر تجعله أغلفته الممتلئة أقلّ المواد المعروفة نشاطًا،
وأيّ إلكترون واحد ناقص يجعل الفلور أشرهها؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:identical-particles:5}
جسيمان لا يتفاعلان في علبة مستوياتها $E_n = n^2E_1$.
(أ) احسب الطاقة الأساسية من أجل بوزونين؛ ومن أجل فرميونين بلا سبين؛ ومن أجل
إلكترونين. (ب) احسب المستوي المثار الأول لكل حالة. (ج) الحالة
الأساسية الفرميونية أغلى: وهذه ``الترقية الفرميونية'' ضغط
جنيني --- اشرح في جملة واحدة كيف تمنع المادة من
الانهيار. (د) عمّم: $N$ فرميونًا بلا سبين في العلبة ---
بيّن أن الطاقة الأساسية تنمو وفق $N^3$ لا وفق $N$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:identical-particles:6}
مع مدارين حقيقيين متعامدين $\varphi_a, \varphi_b$ وحالتَي
زوج $\psi_\pm \propto \varphi_a(1)\varphi_b(2) \pm
\varphi_b(1)\varphi_a(2)$: (أ) بيّن أن $\psi_-(\vect r, \vect r) = 0$؛
(ب) استنتج
$\langle(\vect r_1 - \vect r_2)^2\rangle_\pm =
(\text{حدود مشتركة}) \mp 2\,|\bra{a}\hat{\vect r}\ket{b}|^2$، و
عيّن أيّ تناظر يبقي الجسيمين أبعد؛ (ج) من أجل
تفاعل دافع $V(|\vect r_1 - \vect r_2|)$، حاجج في
إشارة $E_+ - E_-$؛ (د) اربط ذلك بترتيب الأورثو والبارا في الهليوم.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:identical-particles:7}
سلّم الهليوم. (أ) لماذا يجب أن تكون الحالة الأساسية أحادية السبين؟
(ب) يقع ثلاثي $1s2s$ تحت أحادي $1s2s$ بمقدار \qty{0.80}{eV}:
عبّر عن ذلك بتكامل تبديلي وقارن حجمه
بطاقات السبين--سبين المغناطيسية ($\sim\mu_{\text{B}}^2\mu_0/4\pi
a_0^3$، وقدّرها). (ج) لماذا تكون الحالة $2\,^3S$ شبه مستقرة
(فأيّ قاعدتَي انتقاء يجب أن تُخرقا معًا)؟ (د) عمرها المقيس
$\approx\qty{8000}{s}$: قارنه مع \qty{1.6}{ns} في الحالة $2p$ من
الهدروجين واعجب بمسؤولية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:identical-particles:8}
الأورثو والبارا هدروجين. بروتونا الجزيء H$_2$ فرميونان:
فالحالة \emph{الكلية} من سبين نوويّ $\times$ دوران يجب أن تكون
متخالفة التناظر تحت تبديلهما؛ وتبديل البروتونين يقلب أيضًا
محور الجزيء، فيضرب الحالة الدورانية في
$(-1)^J$. (أ) بيّن أن الثلاثي النوويّ (الأورثو) يقترن مع $J$ الفردي،
والأحادي (البارا) مع $J$ الزوجي. (ب) عند درجات الحرارة العالية، أيّ
نسبة أورثو إلى بارا تمليها تعدديات السبين؟ (ج) تقع أدنى
حالة أورثو ($J = 1$) أعلى بالمقدار $2B \approx \qty{15}{meV}$ من حالة بارا $J = 0$،
ويستغرق تحوّل السبين أيامًا: فماذا يحدث
داخل خزان هدروجين مسيَّل حديثًا كلما تحوّلت نسبة الأورثو
ببطء، ولماذا يجب أن تحفّز معامل التسييل التحوّل
\emph{قبل} التخزين؟ (د) أيّ برنامج وقود مبرَّد يومي
(الصواريخ!) يلقى هذه المسألة بالضبط؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:identical-particles:9}
التدافع البوزوني. نمط يحوي $n$ فوتونًا؛ وتتغير العناصر المصفوفية
لإضافة واحد وفق $\sqrt{n+1}$. (أ) بيّن أن معدلات الإصدار في
النمط تتغير وفق $n + 1$ والامتصاص وفق $n$. (ب) في تجويف يحوي
أحد أنماطه $10^{10}$ فوتون وجيرانه لا شيء،
قارن معدلَي إصدار ذرة مثارة في كل منهما. (ج) اشرح في
جملتين كيف يجعل هذا ضوء الليزر \emph{ينمو} في نمط
واحد. (د) أين ظهر العامل $\sqrt{n+1}$ سلفًا في
\cref{prop:b3:harmonic-oscillator:coherent-motion}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:identical-particles:10}
مفارقة غيبس، محلولةً بالتطابق. حجمان متساويان $V$ من
غاز مثالي، في كل منهما $N$ جزيئة، وبدرجة الحرارة نفسها، يُوصلان. (أ) إذا
كان الغازان \emph{مختلفين} (كالأرغون والنيون)، ارتفعت الإنتروبيا
بالمقدار $2Nk_{\text{B}}\ln2$ --- تذكّر لماذا (فكل غاز يضاعف
حجمه). (ب) وإذا كانا الغاز \emph{نفسه}، لم يغيّر فتح الصنبور
شيئًا عيانيًا: فأيّ عبث يترتب لو ظهرت إنتروبيا
المزج مع ذلك (انظر في الإغلاق ثم الفتح من جديد)؟
(ج) بيّن أن معاملة الجزيئات على أنها مميَّزة \emph{تنتج}
فعلًا تلك الإنتروبيا الزائفة، وأن قسمة إحصاء
الحالات على $N!$ --- وهو مشروع فقط لأن الجزيئات
متطابقة --- تحذفها. (د) العبرة في جملة واحدة:
فالفيزياء الإحصائية الكلاسيكية تستعير في صمت واقعة كمومية؛
وستستعيرها الفصول التالية جهرًا.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:identical-particles:11}
باولي يحمل الأموات. يحشد قزم أبيض نحو
$\qty{e9}{kg/m^3}$، أي كثافة إلكترونات $n_{\text{e}} \approx
\qty{3e35}{m^{-3}}$ (بإلكترون واحد لكل كتلتَي نويّة). (أ)
احسب طاقة فيرمي انطلاقًا من
\cref{exo:b3:schrodinger-three-dimensions:8}
--- ولاحظ أن الصيغة اللانسبية تتوتّر. (ب) قارن
$E_{\text{F}}$ مع $k_{\text{B}}T$ عند $T = \qty{e7}{K}$: فبأيّ
معنًى يكون هذا النجم الأبيض المتوهج \emph{باردًا}؟ (ج) اشرح في
جملتين كيف يقدّم الاستبعاد، لا الحرارة، الضغطَ الذي
يوازن الثقالة. (د) ماذا يجب أن يحدث كلما أُضيفت كتلة و
صارت $E_{\text{F}}$ نسبية --- وأيّ حدّ شهير يُنذر بذلك
(\cref{ch:b3:astrophysics})؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:identical-particles:12}
من التبديل إلى المغانط. نمذج إلكترونين متجاورين في
جسم صلب وفق \cref{prop:b3:identical-particles:exchange-force}: فطاقتهما
هي $E_{\text{أحادي}}$ أو $E_{\text{ثلاثي}} =
E_{\text{أحادي}} - 2J$ بثابت تبديل $J$. (أ) بيّن أن هذا
تعيد إنتاجه \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} السبينية الفعّالة $\hat H =
-2J\,\hat{\vect S}_1\cdot\hat{\vect S}_2/\hbar^2$ (باستعمال
$\hat{\vect S}_1\cdot\hat{\vect S}_2 = \tfrac12[\hat S^2 - \hat
S_1^2 - \hat S_2^2]$). (ب) من أجل $J > 0$ تصطفّ السبينات المتجاورة:
فأيّ ظاهرة عيانية تنتجها شبكة من أزواج كهذه؟
(ج) قدّر درجة حرارة الترتيب $k_{\text{B}}T_{\text{C}}
\sim J$ من أجل $J \approx \qty{0.1}{eV}$ وقارنها بقيمة الحديد
\qty{1043}{K}. (د) لماذا لا يستطيع أيّ تفاعل \emph{مغناطيسي}
بين ثنائيَي قطب (تذكّر حجمه من
\cref{exo:b3:identical-particles:7}ب) أن يفسّر مغنطًا عند ألف
كلفن --- وما الذي يفسّره، في جملة واحدة؟
\end{exercise}

\section{مسألة: منطق الجدول الدوري}

\begin{problem}[{مسألة نهاية الأسبوع --- ثلاث قواعد كمومية تبني
الكيمياء كلها}]\label{pb:b3:identical-particles:1}
أعطِ فيزيائيًّا مدارات الهدروجين $(n, l, m)$ وسبين الإلكترون
ومبدأ باولي، يتبع \omterm{prop:b3:identical-particles:periodic}{الجدول الدوري} ---
بدوراته وعائلاته وأحجامه، وتعرّج طاقات التأيّن، بل حتى
لون الذهب. وتبني هذه المسألة ذلك. المعطيات: المستويات
الهدروجينية $-13.6\,Z_{\text{فعّال}}^2/n^2$ إلكترون-فولط؛ وطاقات التأيّن
الأولى المقيسة (بالإلكترون-فولط): H $13.6$، He $24.6$، Li $5.4$، Be $9.3$، B $8.3$،
C $11.3$، N $14.5$، O $13.6$، F $17.4$، Ne $21.6$، Na $5.1$.

\medskip\noindent\textbf{الجزء الأول --- القواعد الثلاث.}
\begin{enumerate}
  \item اذكر المكوّنات الثلاثة: فهرس المدارات
        $(n, l, m)$ بانحلالاته، والمضاعفة بالسبين، و
        مبدأ الاستبعاد. فكم إلكترونًا يسع $n = 1$ و
        $2$ و $3$؟
  \item لماذا يختلف \emph{ترتيب طاقات} الأغلفة الجزئية في
        ذرة متعددة الإلكترونات عن ترتيب الهدروجين المبني على $n$ وحده
        (تذكّر النفاذ والحجب،
        \cref{exo:b3:hydrogen-atom:8})؟
  \item اكتب ترتيب الملء حتى $4s$ و $3d$، وأعطِ
        تشكيلات Na ($Z = 11$) و Cl ($17$) و K ($19$)
        و Fe ($26$).
  \item أطوال الصفوف: فسّر القيم المشاهَدة $2, 8, 8, 18$ في
        دورات الجدول الأربع الأولى انطلاقًا من ترتيب الملء (ولماذا
        يكون الصف الثالث 8 لا 18؟).
  \item أيّ سمة واحدة تجعل الغازات النبيلة صامتة كيميائيًا؟
  \item عرّف الشحنة الفعّالة $Z_{\text{فعّال}}$ التي يراها إلكترون
        التكافؤ، واشرح لماذا تكون نحو $1$ لإلكترون
        الصوديوم الخارجي بينما يرى كل من إلكتروني الهليوم
        الشحنةَ الكاملة $Z = 2$ تقريبًا ناقصًا جزءًا من حجب
        أحدهما للآخر.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- قراءة الصف الثاني.}
\begin{enumerate}[resume]
  \item انطلاقًا من المعطيات، تتسلق طاقة التأيّن من Li إلى Ne لكنها
        تهبط عند B وعند O: عيّن الهبوطين.
  \item فسّر هبوط البورون (فأيّ غلاف جزئي يُفتتح عند B، ولماذا
        يكون $2p$ أغلى إمساكًا من $2s$؟).
  \item فسّر هبوط الأكسجين (فما الذي يجب أن يفعله اثنان من
        إلكترونات الأكسجين في $2p$ لأول مرة، وأيّ قاعدة تُفقد
        راحتها؟).
  \item قدّر طاقة تأين الهليوم تقديرًا ساذجًا بالمقدار
        $13.6 \times Z^2 = \qty{54.4}{eV}$: فلماذا تكون القيمة المقيسة
        \qty{24.6}{eV} أصغر بكثير، وأيّ عدد
        $Z_{\text{فعّال}} \approx 1.34$ يخبرك به هذا الملاءمة؟
  \item قيمة الليثيوم \qty{5.4}{eV} في مقابل قيمة الهدروجين \qty{13.6}{eV}:
        فأيّ أثرين (الحجب؛ و $n$ الأعلى) يتآمران؟
  \item لماذا ينكمش \emph{حجم} الذرة عبر الصف
        (Li $\to$ Ne) مع أن الإلكترونات تُضاف؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- العائلات والروابط.}
\begin{enumerate}[resume]
  \item القلويات: إلكترون $s$ واحد فوق قلب نبيل --- استنتج
        سماتها العائلية (تأين منخفض، وأيونات $+1$، وكيمياء
        عنيفة مع الماء) من التشكيل.
  \item الهالوجينات: فجوة واحدة في غلاف $p$ --- استنتج سماتها
        (الألفة الإلكترونية، وأيونات $-1$، وجزيئات ثنائية الذرة).
  \item كربونات أربع: لماذا تكون ذرة $2s^22p^2$ رباعية التكافؤ
        عمليًّا (فأيّ شبه انحلال يتيح لها الترقية و
        التهجين --- تذكّر هجائن شتارك في
        \cref{ex:b3:perturbation-theory:stark})؟
  \item استعمل \cref{ex:b3:identical-particles:bond}: لماذا لا تكوّن
        ذرتان مغلقتا الغلاف (هليومان) \emph{أيّ} رابطة
        تساهمية؟
  \item الفلزات الانتقالية: شبه الانحلال بين $4s$ و $3d$ يعطي عشرة
        أعمدة من كيمياء متشابهة: فلماذا يغيّر ملء غلاف
        $d$ \emph{داخلي} الكيمياء تغييرًا ضئيلًا هكذا؟
  \item تنبّأ، من التشكيلات وحدها، بأيّ العناصر K و Ca و Sc
        يُظهر أول إلكترونات $d$ غير مزدوجة صالحة للمغناطيسية.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- وراء القواعد التقريبية.}
\begin{enumerate}[resume]
  \item تسند قاعدة هوند إلى النتروجين ثلاثة سبينات متوازية في $2p$:
        فأيّ خاصية ذرية قابلة للقياس (أي عزمه المغناطيسي)
        يحدّدها ذلك، وأيّ آلية من آليات الفصل الرابع عشر تدفع ثمن
        الاصطفاف؟
  \item يقتضي وجود الجدول نفسه تطابق الإلكترونات تطابقًا
        صارمًا: فماذا كانت إلكترونات ``مميَّزة قليلًا''
        ستفعل بمبدأ باولي وبإغلاق الأغلفة؟
  \item في الذرات الثقيلة تتحرك الإلكترونات الداخلية بسرعة $Z\alpha c$:
        فمن أجل الذهب ($Z = 79$)، قدّر $v/c$ لزوج $1s$ و
        اذكر نوعيًّا ما تفعله النسبية بطاقة المدار
        $6s$ --- وهو التفسير المقبول للون الذهب
        ولسيولة الزئبق.
  \item يمتلئ الغلاف الجزئي $4s$ قبل $3d$ ومع ذلك يتأين
        \emph{أولًا} في الفلزات الانتقالية، ويستعير الكروم والنحاس
        إلكترون $s$ ليبلغا $3d^5$ و $3d^{10}$: فبم
        تنبئ هذه الشذوذات عن مدى تقارب طاقتَي الغلافين
        الجزئيين؟
  \item نظيران لعنصر واحد متطابقان كيميائيًا تقريبًا:
        فانطلاقًا من القواعد الثلاث، لماذا لا تعبأ الكيمياء إلا
        بالعدد $Z$ ولا تكاد تعبأ بكتلة النواة؟
  \item تنتهي الدورة السابعة قرب $Z \approx 118$ وتحيد كيمياء
        العناصر فائقة الثقل عن منطق أعمدة الجدول الساذج:
        سمِّ المقياسين المتنافسين (بنية الأغلفة في مقابل
        الآثار النسبية والكولومية عند $Z$ الكبير).
  \item لخّص النتيجة الموسومة: فهرس واحد من المدارات، ومضاعفة
        واحدة بالسبين، وإشارة ناقص واحدة --- والحصيلة هي
        القانون الدوري: صفوف $2, 8, 8, 18$، والهبوطان عند B و O،
        وعائلتا النشط والنبيل، مقيسةً كلها في
        سلّم التأيّن من \qty{5.1}{eV} (Na) إلى
        \qty{24.6}{eV} (He).
\end{enumerate}
\end{problem}
