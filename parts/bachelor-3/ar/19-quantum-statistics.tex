\chapter{الإحصاء الكمومي}\label{ch:b3:quantum-statistics}

إلكترونات الناقلية في النحاس غاز أكثف من الهواء بألف
مرة --- ومع ذلك لا تكاد تضيف شيئًا إلى سعة النحاس الحرارية، وهي
فضيحة طاردت الفيزياء الكلاسيكية خمسين سنة. وسحابة من
ذرات الروبيديوم، مبرَّدةً إلى مئة جزء من مليار من الكلفن،
تنهار فجأةً إلى حالة كمومية واحدة عمقها عشرة آلاف
ذرة. ونجم ميت بكتلة الشمس وحجم الأرض
يأبى أن يزداد انكماشًا، ولا يسنده إلا مبدأ
الاستبعاد. والثلاثة كلها فصل واحد من الفيزياء: أي ما يحدث
لغاز حين $n\lambda_T^3 \sim 1$ وتفارق صيغ الإشغال في
\cref{ch:b3:grand-canonical} بولتزمانَ. فالفرميونات تبني
بحارًا --- باردة صلبة مضغوطة؛ والبوزونات تبني تكاثفات ---
اجتماعية مترابطة فائقة الميوعة. وبين أغلبية بحر فيرمي الصامتة
وهرولة التكاثف يقع أكثر فيزياء المادة المكثفة
ودرجات الحرارة المنخفضة الحديثة، ومصير
النجوم.

\section{غازات من الإشغالات}

\begin{proposition}[آلة الغاز الكمومي]\label{prop:b3:quantum-statistics:machinery}
\index{كثافة الحالات}
من أجل جسيمات غير متفاعلة في علبة، تصير المجاميع على المستويات
تكاملات على \omterm{prop:b3:schrodinger-three-dimensions:counting}{كثافة الحالات} في
\cref{prop:b3:schrodinger-three-dimensions:counting} (لكل حالة
سبين، $g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$):
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
مع $\bar n$ إشغالَ فيرمي--ديراك أو بوز--أينشتاين. وتثبّت
المعادلة الأولى $\mu(T, n)$؛ وتخرج الثانية
الترموديناميك. وفي النهاية المخلخلة $n\lambda_T^3 \ll 1$ ينهار
الإحصاءان معًا على بولتزمان ويعود الغاز الكلاسيكي؛
وتبدأ الأنظمة الكمومية حيث تتلامس الحزم الموجية.
\end{proposition}
\admitted

\section{بحر فيرمي}

\begin{theorem}[غاز فيرمي المنحلّ]\label{thm:b3:quantum-statistics:fermi}
\index{طاقة فيرمي}\index{ضغط الانحلال}
عند $T = 0$، تملأ الفرميونات كل مستوٍ حتى \emph{\omterm{thm:b3:quantum-statistics:fermi}{طاقة
فيرمي}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{سبين } \tfrac12) ,
\]
بمتوسط طاقة $\tfrac35 E_{\text{F}}$ لكل جسيم، ومع \emph{\omterm{thm:b3:quantum-statistics:fermi}{ضغط
الانحلال}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
أي ضغط عند الصفر المطلق، أصله كمومي محض --- وهو
صلابة غاز الإلكترونات في كل فلز وسند النجوم
الميتة. وعند $0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$،
لا يستطيع الاستجابة لأيّ شيء إلا الإلكترونات التي تقع في حدود
$\sim k_{\text{B}}T$ من سطح \emph{بحر فيرمي} هذا: فالنسبة
$\sim T/T_{\text{F}}$ هي المفتاح الرئيسي للسلوك الفلزي.
\end{theorem}

\begin{proof}[برهان جزئي]
املأ $g(\epsilon)$ حتى $E_{\text{F}}$: فيعطي عكس $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ قيمةَ $E_{\text{F}}$ المذكورة
(\cref{exo:b3:schrodinger-three-dimensions:8})؛ و $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ بتكامل آخر. وأما الضغط:
فضغط العلبة يرفع كل مستوٍ وفق $V^{-2/3}$، ومنه $E
\propto V^{-2/3}$ و $P = -\partial E/\partial V = \tfrac23 E/V$
--- ومع $E = \tfrac35 NE_{\text{F}}$ ينتج القانون المذكور.
\end{proof}

\begin{example}[فضيحة السعة الحرارية، محلولةً]\label{ex:b3:quantum-statistics:heat}
\index{السعة الحرارية الإلكترونية}
كلاسيكيًّا، ينبغي لإلكترون النحاس الحرّ لكل ذرة أن يضيف $\tfrac32
k_{\text{B}}$ إلى السعة الحرارية --- أي $50\%$ فوق
ديولونغ--بتي. والمقيس: نحو $1\%$ عند حرارة الغرفة. والبحر
يشرح: فمع $T_{\text{F}} \approx \qty{8e4}{K}$، لا يمكن إثارة
إلا النسبة $\sim T/T_{\text{F}} \approx 0.4\%$ من الإلكترونات
حراريًّا، كلٌّ منها بمقدار $\sim k_{\text{B}}T$:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(وهو معامل زومرفلد المضبوط؛
\cref{exo:b3:quantum-statistics:12}). وهي خطية في $T$، ومدفونة
تحت $T^3$ الشبكة عند درجات الحرارة العادية لكنها
\emph{تهيمن} تحت بضعة كلفنات --- كما تجده المسعّرات
بالضبط، فتحلّ بضربة واحدة لغز الإلكترونات التي
تحمل التيار لا السعة الحرارية.
\end{example}

\begin{example}[نجوم يسندها الاستبعاد]\label{ex:b3:quantum-statistics:dwarf}
\index{القزم الأبيض}
\omterm{ex:b3:quantum-statistics:dwarf}{القزم الأبيض} (\cref{exo:b3:identical-particles:11}) يوازن
الثقالة بضغط انحلال الإلكترونات. وما دام $P \propto
n^{5/3}$ يتصلّب أسرع من مطالب الثقالة كلما انكمش
النجم، فثمة نصف قطر توازني عند أيّ كتلة متواضعة --- مع
التحجيم المخيف $R \propto M^{-1/3}$: أي إن \emph{الأقزام الأثقل
أصغر}. وإذا زاد العصر، صارت الإلكترونات نسبوية، و
ليّن قانون الضغط إلى $n^{4/3}$، وأخفق التوازن عند
كتلة وحيدة، هي حدّ تشاندراسيخار $M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) --- وما وراءه
انهيار إلى نجم نيوتروني: أي بحر فيرمي من النيوترونات عند الكثافة
النووية، وهي الفيزياء نفسها عند كثافة أكبر $10^{14}$ مرة.
\end{example}

\section{هرولة بوز}

\begin{theorem}[تكاثف بوز--أينشتاين]\label{thm:b3:quantum-statistics:bec}
\index{تكاثف بوز--أينشتاين}
من أجل غاز من $N$ بوزونًا محفوظًا، لا تستطيع الحالات المثارة أن تحمل
أكثر من $N_{\max}(T) = 2.612\,V/\lambda_T^3$ جسيمًا (وهو تكامل
$g\bar n_{\text{BE}}$ عند سقفه $\mu \to 0$). وتحت درجة
الحرارة الحرجة
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(أو بالمكافئ } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
لا يجد الفائض مذهبًا إلا الحالة الأساسية المفردة: أي جمهرة
\emph{عيانية}
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
تتكاثف في حالة كمومية واحدة --- أي مادة تسلك سلوك دالة موجية
عملاقة واحدة. تنبّأ بها أينشتاين سنة 1925 من إحصاء بوز
للفوتونات؛ وتحققت في روبيديوم مخلخل سنة 1995 عند
$T_{\text{c}} \approx \qty{100}{nK}$ (نوبل 2001)، وأعلنها
توزيع السرعات \emph{ثنائي النمط} الفارق: أي سحابة حرارية
ينهض من مركزها إبرة من تكاثف عديم السرعة.
\end{theorem}
\admitted

\begin{example}[الهليوم، المائع الفائق المخضرم]\label{ex:b3:quantum-statistics:helium}
\index{الميوعة الفائقة}
الهليوم $^4$He السائل (وهو بوزوني) يتحول عند $T_\lambda = \qty{2.17}{K}$
إلى \emph{مائع فائق}: بلزوجة معدومة عبر أدقّ
الشعيرات، وسيلان غشائي فوق جدران الأوعية، وآثار نافورية، و
دوران مكمَّم بوحدات $h/m$ --- أي طور تكاثف
عياني في العمل، مزاحًا عن تنبؤ الغاز المثالي
($\qty{3.1}{K}$ من أجل كثافة الهليوم) بالتفاعلات لكنه
الظاهرة نفسها بلا لبس. أما $^3$He الفرميوني فلا يستطيع التكاثف
وحده: إذ يجب أن \emph{تتزاوج} ذراته أولًا (مثل إلكترونات
الناقل الفائق، \cref{ch:b3:electrons-in-solids}) في بوزونات
مركّبة --- فتأتي \omterm{ex:b3:quantum-statistics:helium}{الميوعة الفائقة} كما ينبغي، لكن عند \qty{2.6}{mK}، أي
أبرد بألف مرة: نظيران، ودرس واحد --- الإحصاء
قدر.
\end{example}

\begin{method}[أيّ إحصاء، وأيّ نظام]\label{met:b3:quantum-statistics:method}
(1) احسب $n\lambda_T^3$ أولًا: فقيمة $\ll1$ تعني بولتزمان و
يكفي الفصل 17. (2) والفرميونات المنحلّة: فكّر بدلالة
$E_{\text{F}}$ و $T/T_{\text{F}}$؛ فالاستجابات (السعة الحرارية، و
القابلية، والانتثار) تحمل العامل $T/T_{\text{F}}$ أو
لا تعاين إلا سطح فيرمي. (3) والبوزونات المنحلّة: قارن $T$
مع $T_{\text{c}}$؛ وتحته، اقسم الغاز إلى تكاثف زائد
سحابة حرارية. (4) والضغوط: صلابة فرميونية $n^{5/3}$ (أو
$n^{4/3}$ نسبوية) في وجه ما يعصر. (5) والجسيمات
المركّبة: أحصِ الفرميونات المكوِّنة لاختيار الإحصاء ---
وتذكّر أن التزاوج يستطيع تحويل إحصاء إلى آخر.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:1}
احسب $E_{\text{F}}$ و $T_{\text{F}}$ من أجل (أ) النحاس
($n = \qty{8.5e28}{m^{-3}}$)؛ (ب) الصوديوم ($\qty{2.5e28}{m^{-3}}$)؛
(ج) الهليوم $^3$He السائل ($\qty{1.6e28}{m^{-3}}$، $m = 3u$)؛ (د) رتّبها
قياسًا إلى حرارة الغرفة وعلّق على مَن يكون منحلًّا ومتى.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:2}
(أ) بيّن أن $\langle\epsilon\rangle = \tfrac35 E_{\text{F}}$ عند $T =
0$. (ب) احسب السرعة الجذرية المتوسطة للإلكترونات في النحاس. (ج) وتلك
السرعة توافق كلاسيكيًّا أيّ درجة حرارة؟ (د) ولماذا يحوي الفلز
البارد إلكترونات تتحرك بجزء من مئة من سرعة الضوء
من دون أن تشعّ تلك الطاقة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:3}
ضغط انحلال إلكترونات النحاس: (أ) احسب $P =
\tfrac25 nE_{\text{F}}$ بالباسكال وبالجوّ. (ب) ولماذا لا
ينفجر الفلز (فما الذي يشدّ إلى الوراء)؟ (ج) وهذا الضغط يقاوم
الانضغاط: اربطه بانضغاطية الفلزات المنخفضة. (د) قدّر
معامل الحجم $K \sim \tfrac53 P$ وقارنه بقيمة النحاس المقيسة \qty{140}{GPa}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:4}
عتبات \omterm{thm:b3:quantum-statistics:bec}{تكاثف بوز--أينشتاين}. (أ) احسب $T_{\text{c}}$ من أجل الروبيديوم-87 ($m =
\qty{1.44e-25}{kg}$) عند $n = \qty{e20}{m^{-3}}$. (ب) ومن أجل
$n$ نفسها، أيّ $T_{\text{c}}$ تكون لذرات الهدروجين؟ (ج) ولماذا
تستعمل التجارب غازات مخلخلة عند النانوكلفنات بدل غازات
كثيفة عند الميليكلفنات (فما الذي يحدث كيميائيًّا عند الكثافة العالية و
الحرارة المنخفضة)؟ (د) تحقق من قاعدة الاتساق $n\lambda_{T_c}^3
= 2.612$ على أعداد الروبيديوم عندك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:5}
سعتان حراريتان. في الفلز، $C = \gamma T +
\beta T^3$ (للإلكترونات؛ وللشبكة). (أ) باستعمال
\cref{ex:b3:quantum-statistics:heat} وحدّ الشبكة الشبيه
بديباي $\beta T^3 = 234\,Nk_{\text{B}}(T/\theta_{\text{D}})^3$،
قدّر درجة حرارة العبور من أجل النحاس
($\theta_{\text{D}} = \qty{343}{K}$ و $T_{\text{F}} =
\qty{8.2e4}{K}$). (ب) وعند \qty{300}{K}، احسب حصة الإلكترونات
بالمئة. (ج) وعند \qty{0.1}{K}؟ (د) ولماذا يرسم المجرّبون $C/T$
بدلالة $T^2$، وماذا يعطي تقاطع ذلك الخط المستقيم وميله؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}
قزم أبيض بكتلة شمسية واحدة ($M = \qty{2e30}{kg}$) وبنصف قطر
الأرض ($R = \qty{6.4e6}{m}$)، بإلكترون واحد لكل نوويّين. (أ)
احسب $n_{\text{e}}$ و $E_{\text{F}}$ (بالصيغة
اللانسبوية). (ب) قارن $E_{\text{F}}$ مع $m_{\text{e}}c^2$ و
مع $k_{\text{B}}T$ عند \qty{e7}{K} في الداخل: وبرّر
``البارد'' و``شبه النسبوي'' في آن. (ج) قارن
\omterm{thm:b3:quantum-statistics:fermi}{ضغط الانحلال} $\tfrac25 nE_{\text{F}}$ بتقدير الضغط
الثقالي $\sim GM^2/R^4$: أهما من الرتبة نفسها؟ (د) وماذا يحدث نوعيًّا
لكل طرف من طرفي التوازن إذا نمت $M$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:7}
مادة نيوترونية. نجم نيوتروني يحشر $M = 1.4\,M_\odot$ في $R
\approx \qty{11}{km}$. (أ) احسب كثافة النيوترونات وقارنها
بالكثافة النووية ($\qty{2.3e17}{kg/m^3}$). (ب) احسب
$E_{\text{F}}$ للنيوترونات (لانسبويًّا؛ مع $m_{\text{n}}c^2
= \qty{939}{MeV}$) وتحقق من مقدار نسبويتها. (ج) ولماذا
اختفت إلكترونات النجم (بتفكك بيتا العكسي:
$\text{e} + \text{p} \to \text{n} + \nu$ --- فما الذي جعل
$E_{\text{F}}$ الإلكترونات تدفع ثمنه)؟ (د) وملعقة صغيرة من هذه
المادة، كم تزن؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:8}
بوزونات بلا عدد. (أ) للفوتونات $\mu = 0$ دائمًا: فلماذا لا
تستطيع أن تخضع للتكاثف في
\cref{thm:b3:quantum-statistics:bec} (وماذا يفعل الغاز بدلًا من ذلك
عند التبريد)؟ (ب) وأيّ قانون حفظ تملكه ذرات الروبيديوم و
يفتقده الفوتون؟ (ج) وفي سنة 2010 \emph{صُنع} تكاثف فوتوني، في
جوف مملوء بصبغة تتوازن فيه الفوتونات حراريًّا ويكون عددها
ثابتًا فعليًّا: فأيّ مكوّن أعاده جوف الصبغة؟
(د) والفونونات في بلورة: أتكاثف أم لا، ولماذا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
نظيرا الهليوم. (أ) احسب $T_{\text{c}}$ للغاز المثالي
من أجل $^4$He السائل ($n = \qty{2.2e28}{m^{-3}}$، $m = 4u$) و
قارنها مع $T_\lambda = \qty{2.17}{K}$. (ب) ولماذا يكون التوافق
تقريبيًّا لا غير (فما الذي يهمله ``المثالي'' في سائل)؟ (ج) و $^3$He عند
الكثافة نفسها تقريبًا يبقى عاديًّا نزولًا إلى الميليكلفنات: اذكر
العائق والمخرج الذي وجدته الطبيعة. (د) تدقيق منطقي مركّب: \emph{زوج} $^3$He يحوي عددًا
زوجيًّا من الفرميونات --- فهو بوزون؛ و $^6$Li يحمل $3\text{p} + 3\text{n}
+ 3\text{e} = 9$ فرميونات: صنّفه، وسمِّ التجارب الشهيرة
على الغازات الباردة التي تستغل تباين الليثيوم--الليثيوم هذا بالضبط.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ بأمانة. (أ) اكتب $N_{\text{مثار}} = \int_0^\infty
g(\epsilon)\dd\epsilon/(\eu^{\beta\epsilon} - 1)$ عند $\mu = 0$ و
اختزله، مع $x = \beta\epsilon$، إلى $N_{\text{مثار}} =
(V/\lambda_T^3)\cdot\frac{2}{\sqrt\pi}\int_0^\infty\frac{\sqrt
x\,\dd x}{\eu^x - 1}$. (ب) وبمعلومية $\int_0^\infty\sqrt x\,\dd x/(
\eu^x - 1) = 2.612\,\sqrt\pi/2$، احصل على السقف وعلى
$T_{\text{c}}$. (ج) استنتج نسبة التكاثف $1 -
(T/T_{\text{c}})^{3/2}$. (د) ولماذا لا تنتج الحجة نفسها
\emph{أيّ} تكاثف في علبة ثنائية البعد (فكيف تتغير $g(
\epsilon)$، وماذا يصيب تقارب
التكامل)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
لماذا تتماسك الفلزات. نمذج فلزًا بإلكترونات كثافتها
$n$ (وكلفة \omterm{thm:b3:quantum-statistics:fermi}{طاقة فيرمي} $\propto n^{2/3}$ لكل إلكترون) منجذبة
إلى خلفية أيونية معادِلة (وكسب كولوم لكل إلكترون
$\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (أ) اكتب الطاقة لكل
إلكترون $\epsilon(n) = An^{2/3} - Bn^{1/3}$ وبيّن أن لها
أصغرية. (ب) بيّن أن تباعد الإلكترونات التوازني من رتبة
$a_0$ --- فكثافات الفلزات ليست مصادفات. (ج) قدّر
مقياس طاقة التماسك عند الأصغرية، بوحدة eV. (د) وأيّ دور
يؤديه \omterm{thm:b3:quantum-statistics:fermi}{ضغط الانحلال} في هذه الحالة المرتبطة (فما الذي يوقف
انهيار كولوم)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
قشرة زومرفلد، مكمَّاةً. (أ) احتجّ: من $N$ إلكترونًا، لا يجلس في
القشرة النشطة إلا $\sim g(E_{\text{F}})k_{\text{B}}T$، كلٌّ منها
يحمل طاقة زائدة $\sim k_{\text{B}}T$: فاستنتج $C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ في حدود ثابت. (ب) ومع
$g(E_{\text{F}}) = 3N/2E_{\text{F}}$، استعد $C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ (والثابت المضبوط هو $\pi^2/2$).
(ج) ومنطق القشرة نفسه يعطي قابلية باولي السبينية المستقلة عن
درجة الحرارة: فلا يستطيع القلب إلا إلكترونات القشرة --- لكن
القشرة تنمو مثل $T$ بينما تهبط كل مساهمة مثل $1/T$: بيّن
التلاشي. (د) اذكر العبرة العامة للفرميونات المنحلّة:
\emph{كل} استجابة هي الكلاسيكية مضروبةً في
$\sim T/T_{\text{F}}$ أو في نظيرها السطحي.
\end{exercise}

\section{مسألة: أثقل قزم أبيض ممكن}

\begin{problem}[{مسألة نهاية الأسبوع --- حدّ تشاندراسيخار، و
الشموع التي أوقدها}]\label{pb:b3:quantum-statistics:1}
في سنة 1930، وعمره تسع عشرة سنة، حسب سوبراهمانيان تشاندراسيخار على
متن الباخرة إلى إنكلترا أن انحلال الإلكترونات لا يسند نجمًا ميتًا
إلا حتى كتلة محددة. والنتيجة --- وقد قاومها إدنغتون
بمرارة --- تقوم عليها النجوم النيوترونية والثقوب السود و
``الشموع المعيارية'' من المستعرات العظمى التي كشفت تسارع
الكون. وتبني هذه المسألة النتيجة بالتحجيم. والمعطيات: $M_\odot =
\qty{2e30}{kg}$؛ وإلكترون واحد لكل $\mu_e = 2$ نوويّ؛ و $m_{\text{
n}} = \qty{1.67e-27}{kg}$؛ و $\hbar c = \qty{3.16e-26}{J.m}$؛ و $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{الجزء الأول --- توازن طاقتين.}
من أجل نجم كتلته $M$ ونصف قطره $R$، عالج الطاقات لكل وحدة
كتلة بالتحجيم.
\begin{enumerate}
  \item الطاقة الثقالية: $E_{\text{g}} \sim -GM^2/R$ ---
        تذكّر أصلها.
  \item عدد الإلكترونات $N_{\text{e}} = M/\mu_e m_{\text{n}}$ و
        الكثافة $n \sim N_{\text{e}}/R^3$: اكتب مقياس
        \omterm{thm:b3:quantum-statistics:fermi}{طاقة فيرمي} اللانسبوي بدلالة $M$
        و $R$.
  \item بيّن أن الطاقة الحركية (طاقة الانحلال) الكلية تُحجَّم وفق
        $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item صغّر $E_{\text{k}} + E_{\text{g}}$ بدلالة $R$ وبيّن أن
        نصف القطر التوازني هو
        $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$.
  \item اقرأ التحجيم $R \propto M^{-1/3}$: فالأقزام الأثقل
        \emph{أصغر}. ولماذا تفوح من ذلك رائحة
        المشكلة أصلًا؟
  \item احسب $R$ من أجل $M = 1\,M_\odot$ وقارنه بالقيمة المقيسة
        للنجم سيريوس B البالغة $\approx\qty{5800}{km}$.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- النسبوية تسحب الأرضية
من تحت.}
\begin{enumerate}[resume]
  \item مع انكماش $R$ تنمو $E_{\text{F}}$: قدّر الكثافة
        التي عندها $E_{\text{F}} \sim m_{\text{e}}c^2$، ومنطقة
        كتل الأقزام الموافقة.
  \item وفي النظام فوق النسبوي تكون طاقة كل إلكترون
        $\sim\hbar c\,n^{1/3}$: بيّن أن الطاقة الحركية تصير
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$.
  \item وتُحجَّم الطاقتان الآن معًا وفق $1/R$: استنتج أن
        التوازن لم يعد يختار نصف قطر --- بل قارن
        المعاملين بدلًا من ذلك.
  \item وبوضع $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$، استنتج
        عدد الإلكترونات الحرج والكتلة
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item احسب النسبة عديمة الأبعاد $\hbar c/Gm_{\text{n}}^2$
        --- وهي من أعظم الأعداد الصرفة في الفيزياء --- ثم
        احسب $M_{\text{Ch}}$ بالكتل الشمسية (والمعالجة المضبوطة تعطي
        $1.4\,M_\odot$).
  \item فسّر الصيغة: فأيّ ثلاثة ثوابت تتآمر، و
        لماذا يحمل حدّ \emph{كمومي} على \emph{نجم}
        الثابتَ $\hbar$؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- ما وراء الحدّ.}
\begin{enumerate}[resume]
  \item قزم دُفع وراء $M_{\text{Ch}}$ (بتراكم من
        رفيق): فما الذي يسنده بعدئذٍ، وعند أيّ مقياس
        لنصف القطر (ضع $m_{\text{n}}$ مكان $m_{\text{e}}$ و
        الواحدَ مكان $\mu_e$ في نصف قطر الجزء الأول)؟
  \item بيّن أن الحدّ النسبوي للنجم النيوتروني نفسه من
        رتبة $M_{\text{Ch}}$ نفسها --- فلا تسمح الطبيعة إلا
        بمهلة بعامل اثنين (والأعظمية المشاهَدة $\approx
        2.2\,M_\odot$). فما الذي ينتظر وراء ذلك؟
  \item وأثناء الانهيار إلى نجم نيوتروني، تُحرَّر طاقة ثقالية
        مقدارها $\sim GM^2/R_{
        \text{ns}}$: احسبها
        من أجل $M = 1.4\,M_\odot$ و $R = \qty{11}{km}$، و
        قارنها بكامل خرج الشمس على مدى \qty{10}{Gyr}
        ($\sim\qty{1.2e44}{J}$).
  \item وأكثرها كله يغادر نيوترينوهات في ثوانٍ: فأيّ
        مشاهدة سنة 1987 أكدت هذه الصورة تأكيدًا مذهلًا؟
  \item وارتداد اللبّ المنهار ودفع النيوترينوات ينسفان
        الغلاف: أي مستعر أعظم بانهيار اللبّ. ميّزه،
        في جملة واحدة لكلٍّ، عن المستعر الأعظم
        \emph{النوويّ الحراري} (من النمط Ia) لقزم أبيض يشتعل عند
        عتبة تشاندراسيخار.
  \item ولماذا يجعل الانفجار \emph{عند كتلة كونية واحدة}
        مستعرات النمط Ia شبه متطابقة --- ومن ثمّ صالحة
        شموعًا معيارية؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- الشموع و
الكون.}
\begin{enumerate}[resume]
  \item سطوع الشمعة المعيارية الظاهري يعطي
        بعدها: اذكر منطق التربيع العكسي ولماذا تكون
        \emph{المعيارية} هي الكلمة الفاعلة.
  \item وفي سنة 1998 وجد فريقان أن مستعرات النمط Ia البعيدة
        \emph{أخفت} مما هو متوقَّع: اذكر الاستنتاج
        (تسارع التمدد --- أي الطاقة المظلمة) ولماذا كانت
        موثوقية الشمعة هي عقدة الأمر.
  \item اذكر سلسلة هذه المسألة في سطر واحد: باولي $\to$
        \omterm{thm:b3:quantum-statistics:fermi}{ضغط الانحلال} $\to$ $M_{\text{Ch}}$ $\to$ انفجارات
        منتظمة $\to$ تسارع كوني.
  \item سخر إدنغتون من النتيجة (``نجم يرتكب
        سخافات'')؛ ونال تشاندراسيخار جائزة نوبل
        بعد ثلاث وخمسين سنة: فبمَ تعلّم الحادثة عن
        اتّباع المعادلات إلى ما وراء النتائج المريحة؟
  \item طيف سيريوس B سنة 1915 كان يقتضي طنًّا في السنتيمتر
        المكعب --- وهو ``هراء'' إلى أن شرح فاولر، في غضون أشهر من
        إحصاء فيرمي سنة 1926، النجمَ غازًا
        منحلًّا: اذكر لماذا كان اللغز مستعصيًا في سنة
        وروتينيًّا في التي تليها.
  \item مسوح آلاف الأقزام البيض تجد كتلًا
        متجمّعة قرب $0.6\,M_\odot$ ولا شيء فوق
        $\approx1.4\,M_\odot$: فأيّ تنبؤين من تنبؤات هذه
        المسألة يؤكدهما ذلك المدرَّج وحده؟
  \item لخّص النتيجة الموسومة: $(\hbar c/Gm_{\text{n}}^2)^{
        3/2}m_{\text{n}} \approx 1.4\,M_\odot$ --- أي ثلاثة ثوابت
        طبيعية تضع أثقل قزم أبيض ممكن، وهي
        كتلة زناد انفجارات الكون المعايَرة.
\end{enumerate}
\end{problem}
