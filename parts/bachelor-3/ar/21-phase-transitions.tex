\chapter{انتقالات الطور}\label{ch:b3:phase-transitions}

سخّن الماء درجةً واحدة فلا يحدث شيء يُذكر --- إلى أن يمزّق السائل
نفسه بخارًا، عند درجة حرارة محددة تحديدًا حادًّا.
وبرّد الحديد عبر \qty{1043}{K} فيتمغنط تلقائيًّا، من دون
أيّ تغير في التركيب. ولا شيء في جزيء مفرد أو
سبين مفرد ينذر بهذه الثورات: فهي \emph{جماعية}
--- $10^{23}$ فاعلًا، يتفاعل كلٌّ منها مع جيران قليلين،
فيتآمرون على تغيير طور مجتمعهم دفعةً واحدة. وقد فهرس كتاب
السنة الأولى الانتقالات؛ ويشرح هذا الفصل آليتها
بأدوات الفصول الخمسة الماضية: منافسة \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة}، و
وسطاء الترتيب، \omterm{def:b3:phase-transitions:order}{وكسر التناظر}، والاكتشاف
المذهل أن التفاصيل المجهرية تكفّ عن الأهمية قرب نقطة
حرجة --- فيتقاسم مغنط ومائع يغلي الأُسسَ الحرجة نفسها.
وأفكار هذا الفصل تجري اليوم من قدور الضغط
إلى النواقل الفائقة إلى مجال هيغز في
\cref{ch:b3:particle-physics}.

\section{وسطاء الترتيب ومنافسة الطاقة الحرة}

\begin{definition}[الأطوار ووسطاء الترتيب]\label{def:b3:phase-transitions:order}
\index{انتقال الطور}\index{وسيط الترتيب}\index{كسر التناظر}
\emph{انتقال الطور} تغيّر غير أملس في حالة توازن
جملة حين يعبر وسيط تحكّم ($T$ أو $P$ أو $B$)
حدًّا. وماسك دفاتره هو \emph{وسيط الترتيب}: أي مقدار
منعدم في الطور غير المرتَّب وغير منعدم في المرتَّب --- أي
مغنطة $m$ للمغنط الحديدي، وفرق الكثافة
$\rho_{\text{سائل}} - \rho_{\text{غاز}}$ للمائع، ونسبة
التكاثف في \cref{ch:b3:quantum-statistics}. والانتقالات على
نوعين: \emph{من المرتبة الأولى} (فيقفز وسيط الترتيب؛ وثمة حرارة كامنة؛ و
تتعايش الأطوار --- كالغليان والانصهار) و\emph{متصلة} (فينمو وسيط
الترتيب من الصفر؛ ولا حرارة كامنة؛ وثمة تذبذبات جامحة ---
كنقطة كوري \omterm{prop:b3:phase-transitions:vdw}{والنقطة الحرجة} والهليوم الفائق الميوعة). وحين يجب على
الحالة المرتَّبة أن \emph{تختار} بين خيارات متكافئة --- فشمال
المغنط لا بدّ أن يشير إلى مكان ما --- يكون الانتقال
\emph{كاسرًا لتناظر} تحترمه القوانين الأساسية: وهو موضوع
سيعود، وعلى أعلى رهان، مع آلية هيغز.
والمحرك وراء كل حالة هو \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة} $F = E - TS$
(\cref{prop:b3:canonical-ensemble:machine}): فالطاقة تفضّل الترتيب، و
الإنتروبيا تفضّل الفوضى، و $T$ يضع سعر الصرف --- فحدود
الأطوار حيث يتعادل الدفتر.
\end{definition}

\section{الانتقال سائل--غاز: فان دير فالس}

\begin{proposition}[متساويات حرارة فان دير فالس]\label{prop:b3:phase-transitions:vdw}
\index{معادلة فان دير فالس}\index{النقطة الحرجة}
معادلة الغاز الحقيقي في كتاب السنة الأولى،
\[
\Big(P + \frac{aN^2}{V^2}\Big)(V - Nb) = Nk_{\text{B}}T ,
\]
ترمّز الجذب ($a$) واللبوب الصلبة ($b$). وفوق درجة حرارة
حرجة $T_{\text{c}} = 8a/27bk_{\text{B}}$ تكون متساويات حرارتها
رتيبة: أي طور مائع واحد. وتحتها تنشأ حلقة تحوي
امتدادًا \emph{غير مستقر} ميكانيكيًّا ($\partial P/\partial V >
0$): فينقسم المائع هناك إلى سائل وغاز متعايشين، عند
الضغط الذي يثبّته تساوي الكمونات الكيميائية (وهي قاعدة
المساحتين المتساويتين لماكسويل). وتنغلق منطقة التعايش عند \omterm{prop:b3:phase-transitions:vdw}{النقطة
الحرجة} ($P_{\text{c}}, V_{\text{c}}, T_{\text{c}}$)، حيث يصير السائل
والغاز \emph{غير متمايزين}: فباقترابنا منها ينعدم
فرق الكثافة، وتتباعد الانضغاطية، وتنمو
تذبذبات الكثافة إلى حجم ضوئي --- أي الانتهاء الحرج
في \cref{pb:b3:perturbation-theory:1}.
\end{proposition}
\admitted

\begin{proposition}[كلاوزيوس--كلابيرون]\label{prop:b3:phase-transitions:clapeyron}
\index{علاقة كلاوزيوس--كلابيرون}
على طول أيّ خط تعايش من المرتبة الأولى، يفرض تساوي الكمونات
الكيميائية على الجانبين
(\cref{def:b3:grand-canonical:mu}) أن يكون ميل الخط
\[
\frac{\dd P}{\dd T} = \frac{L}{T\,\Delta v} ,
\]
مع $L$ الحرارة الكامنة و $\Delta v$ تغيّر الحجم لكل
جسيم (أو لكل كيلوغرام، باتساق). وكل ما يخصّ
خطوط التعايش في هذه الصيغة الواحدة: فخط غليان الماء
يتسلق بمقدار $\approx\qty{3.6}{kPa/K}$ قرب \qty{100}{\celsius}
(ومنه قدور الضغط والطبخ في الارتفاع ---
\cref{pb:b3:phase-transitions:1})؛ وخط \emph{انصهار} الماء
يميل إلى الوراء ($\Delta v < 0$: فالجليد يطفو)، فيصهر الضغط
الجليدَ --- وهي ندرة بين المواد لها نتائج من الأنهار الجليدية
إلى حلبات التزلج.
\end{proposition}

\begin{proof}
على الخط، $\mu_1(T, P) = \mu_2(T, P)$؛ وتحرّك على طوله:
$-s_1\dd T + v_1\dd P = -s_2\dd T + v_2\dd P$ (لكل جسيم، باستعمال
$\dd\mu = -s\,\dd T + v\,\dd P$). ثم حلّ من أجل $\dd P/\dd T$ واستعمل
$L = T(s_2 - s_1)$.
\end{proof}

\section{المغنط الحديدي: نظرية المجال المتوسط}

\begin{theorem}[انتقال إيزينغ بالمجال المتوسط]\label{thm:b3:phase-transitions:meanfield}
\index{نموذج إيزينغ}\index{نظرية المجال المتوسط}\index{درجة حرارة كوري}
نمذج مغنطًا بسبينات $s_i = \pm1$ على شبكة، والجيران
مقترنون بطاقة التبديل $-J\,s_is_j$
(\cref{exo:b3:identical-particles:12})، ولكل سبين $q$
جارًا. واستبدل بجيران كل سبين متوسطهم $m =
\langle s\rangle$ (وهو تقريب \emph{المجال المتوسط}): فيجلس السبين
عندئذٍ في المجال الفعّال $qJm$، وتعطي
\cref{exo:b3:canonical-ensemble:11} شرط
الاتساق الذاتي
\[
m = \tanh\!\Big(\frac{T_{\text{c}}}{T}\,m\Big) , \qquad
k_{\text{B}}T_{\text{c}} = qJ .
\]
وفوق $T_{\text{c}}$ يكون الحلّ الوحيد $m = 0$؛ وتحتها يظهر حلّان
متناظران $\pm m(T)$ وهما المستقران: فلا بدّ للمغنط \emph{أن}
يتمغنط، مختارًا اتجاهًا عشوائيًّا ---
أي \omterm{def:b3:phase-transitions:order}{كسر التناظر} التلقائي. وقرب $T_{\text{c}}$ تكون $m \propto
(T_{\text{c}} - T)^{1/2}$، وفوقها تطيع القابلية
قانون كوري--فايس $\chi \propto 1/(T - T_{\text{c}})$:
وهو متباعد عند الانتقال --- أي مغنط لا حدّ لتلبيته.
\end{theorem}

\begin{proof}[برهان جزئي]
السبين في المجال $B_{\text{فعّال}} = qJm/\mu$ له $\langle s\rangle =
\tanh(\beta qJm)$؛ واشتراط $\langle s\rangle = m$ يعطي
المعادلة المذكورة. وبيانيًّا: يقطع الخط $y = m$ المنحني $y =
\tanh(T_{\text{c}}m/T)$ عند المبدأ وحده حين يكون ميل المنحني
الابتدائي $T_{\text{c}}/T < 1$، ويقطعه أيضًا عند $\pm m^* \neq 0$ حين
$T < T_{\text{c}}$. ونشر $\tanh$ من أجل $m$ الصغيرة
يعطي $m^2 = 3(1 - T/T_{\text{c}})$، أي النموّ الجذري؛ و
إضافة مجال مطبَّق صغير والنشر كذلك يعطيان
كوري--فايس (\cref{exo:b3:phase-transitions:6}).
\end{proof}

\section{وجهة نظر لانداو والشمولية}

\begin{proposition}[نظرية لانداو]\label{prop:b3:phase-transitions:landau}
\index{نظرية لانداو}\index{الأُسس الحرجة}\index{الشمولية}
قرب انتقال متصل، انشر \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة} بدلالة \omterm{def:b3:phase-transitions:order}{وسيط
الترتيب} الصغير، مبقيًا ما يسمح به التناظر ($m \to -m$
هنا):
\[
F(m) = a\,(T - T_{\text{c}})\,m^2 + b\,m^4 , \qquad a, b > 0 .
\]
فوق $T_{\text{c}}$: بئر واحدة عند $m = 0$. وتحتها: يصير المبدأ
قمّة وتظهر بئران متناظرتان عند $m = \pm
\sqrt{a(T_{\text{c}} - T)/2b}$ --- أي الجذر التربيعي نفسه الذي أعطاه المجال
المتوسط، آتيًا الآن من التناظر وحده. ولهذا تسلك جمل
مختلفة اختلافًا جامحًا سلوكًا متطابقًا قرب نقاطها الحرجة:
أي \emph{الشمولية}. والأُسس المقيسة (مثل $m \propto |T -
T_{\text{c}}|^\beta$ مع $\beta \approx 0.33$ من أجل الموائع
\emph{ومن أجل} المغانط أحادية المحور) تختلف عن قيمة المجال المتوسط $1/2$
لأن التذبذبات تهيج على كل المقاييس قرب $T_{\text{c}}$ ---
وقد بيّن ترويضها، أي زمرة إعادة التنظيم (ويلسون، نوبل 1982)،
أن البعدية والتناظر وحدهما، لا التفاصيل المجهرية،
تقرران الأُسس. ولصورة البئرين المزدوجة نفسها
مسيرة تتجاوز المغانط: فأعطِ \omterm{def:b3:phase-transitions:order}{وسيط الترتيب} دور
مجال يملأ الفضاء تكن بئره الكاسرة للتناظر آليةَ
هيغز (\cref{ch:b3:particle-physics}).
\end{proposition}
\admitted

\begin{method}[قراءة انتقال]\label{met:b3:phase-transitions:method}
(1) عيّن \omterm{def:b3:phase-transitions:order}{وسيط الترتيب} وهل يقفز
(من المرتبة الأولى: حرارة كامنة، وتعايش، وتخلّف، وتنوٍّ) أم
ينمو نموًّا متصلًا (نقطة حرجة: تباعدات، وانتهاء،
وشمولية). (2) وخطوط المرتبة الأولى: كلاوزيوس--كلابيرون من أجل
الميل؛ وتساوي $\mu$ من أجل التعايش. (3) والمتصلة: المجال المتوسط
أولًا ($\tanh$ أو لانداو) من أجل شكل مخطط الأطوار؛ وثِق بما هو
\emph{طوبولوجي} فيه لا بأُسسه. (4) وتحت $T_{\text{c}}$،
تذكّر الاختيار: فالتناظر المكسور يعني مناطق وعيوبًا و
توقفًا على التاريخ. (5) والتقديرات: $k_{\text{B}}T_{\text{c}} \sim$
(الاقتران لكل جار) $\times$ (عدد الجيران).
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:2}
الحرارات الكامنة إنتروبيا. (أ) الماء: $L_{\text{تبخر}} =
\qty{2.26e6}{J/kg}$ عند \qty{373}{K}: احسب $\Delta S$ لكل
جزيء بوحدة $k_{\text{B}}$. (ب) والمثل من أجل الانصهار
($L_{\text{انصهار}} = \qty{3.3e5}{J/kg}$ عند \qty{273}{K}). (ج) وأكثر
السوائل تتبخر مع $\Delta S \approx 10$--$11\,k_{\text{B}}$
(وهي قاعدة تروتون): فماذا يحصي هذا العدد تقريبًا (أي
$\ln$ أيّ نسبة حجوم)؟ (د) ولماذا تكون قفزة إنتروبيا الانصهار أصغر
بكثير من قفزة الغليان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:3}
الثوابت الحرجة عند فان دير فالس ($V_{\text{c}} = 3Nb$ و
$k_{\text{B}}T_{\text{c}} = 8a/27b$ و $P_{\text{c}} = a/27b^2$ ---
اقبلها). (أ) بيّن أن $P_{\text{c}}V_{\text{c}}/Nk_{\text{B}}T_{\text{
c}} = 3/8$، وهو عدد شامل للنموذج؛ والقيم المقيسة
تتجمّع قرب $0.29$ --- علّق. (ب) ومن أجل CO$_2$
($T_{\text{c}} = \qty{304}{K}$ و $P_{\text{c}} = \qty{73.8}{bar}$):
استخرج $a$ و $b$. (ج) ومن $b$، قدّر قطر
الجزيء. (د) ولماذا يتسيّل CO$_2$ عند حرارة الغرفة تحت
الضغط بينما لا يتسيّل N$_2$ عند حرارة الغرفة ($T_{\text{c}} =
\qty{126}{K}$) أبدًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:4}
وسطاء الترتيب، بسرعة: أعطِ واحدًا من أجل (أ) مغنط حديدي؛ (ب)
الانتقال سائل--غاز؛ (ج) \omterm{thm:b3:quantum-statistics:bec}{تكاثف بوز--أينشتاين}؛ (د)
مغنط مضادّ (بشبكتين جزئيتين --- فأيّ مقدار يترتّب بينما
تبقى المغنطة الكلية منعدمة؟).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:5}
المجال المتوسط باليد. (أ) استنتج $m = \tanh(\beta qJm)$ من سبين
في متوسط مجال جيرانه $q$. (ب) بيّن أن حلًّا غير منعدم
يوجد بالضبط حين $T < qJ/k_{\text{B}}$. (ج) وانشر حتى
المرتبة الثالثة واستنتج $m(T) \approx \sqrt{3(1 -
T/T_{\text{c}})}$. (د) والحديد: $T_{\text{c}} = \qty{1043}{K}$ و $q =
8$: استخرج $J$ بوحدة eV وتحقق من مقياس التبديل في
\cref{exo:b3:identical-particles:12}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:6}
كوري--فايس. أضف مجالًا صغيرًا: $m = \tanh[\beta(qJm + \mu B)]$.
(أ) خطّط فوق $T_{\text{c}}$ واستنتج $\chi = \partial
m/\partial B \propto 1/(T - T_{\text{c}})$. (ب) وقارن بقانون
كوري للسبينات الحرة: فماذا يعطي رسم $1/\chi$ بدلالة $T$
في كل حالة، وكيف يشخّص التقاطع الاقتران
المغناطيسي الحديدي؟ (ج) وخط $1/\chi$ للنيكل يبلغ الصفر عند \qty{631}{K}:
فسّر. (د) وللمغنط المضادّ يكون التقاطع
\emph{سالبًا}: فأيّ إشارة للاقتران $J$ يفضح ذلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:7}
كلاوزيوس--كلابيرون في العمل. (أ) الماء عند \qty{100}{\celsius}
(مع $\Delta v = \qty{1.67}{m^3/kg}$): احسب $\dd P/\dd T$. (ب)
واستعمل $\dd\ln P/\dd T = Lm_{\text{w}}/k_{\text{B}}T^2$ لتقدير
درجة حرارة الغليان عند $P = \qty{0.7}{atm}$ (أي قمة عند \qty{3000}{m}).
(ج) والانصهار: مع $\Delta v = \qty{-9.1e-5}{m^3/kg}$ و
$L = \qty{3.3e5}{J/kg}$، احسب $\dd T/\dd P$ والضغط
اللازم لخفض نقطة انصهار الجليد بمقدار \qty{5}{K}. (د) ومتزلج
كتلته \qty{70}{kg} على \qty{30}{mm^2} من النصل: احسب الضغط و
إزاحته لنقطة الانصهار --- فهل يشرح الانصهار بالضغط
التزلج؟ (وما الذي يشرحه: الاحتكاك والانصهار السطحي المسبق.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:8}
حسابات لانداو. (أ) صغّر $F = a(T - T_{\text{c}})m^2 +
bm^4$ تحت $T_{\text{c}}$. (ب) واحسب $F$ عند الأصغرية و
بيّن أن للسعة الحرارية \emph{قفزة} منتهية $\Delta C =
a^2T_{\text{c}}/2b$ عند الانتقال. (ج) وارسم $C(T)$:
درجة المجال المتوسط في مقابل الذروة شبه اللوغاريتمية المقيسة عند
نقطة $\lambda$ للهليوم --- فماذا تضيف التذبذبات الحقيقية؟ (د)
وأضف حدّ مجال $-hm$: بيّن أن الانتقال يستدير ---
فلا بدّ للتناظر أن يكون مضبوطًا كي ينكسر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:9}
التنوّي. قطيرة نصف قطرها $r$ في بخار فوق مشبَع
تكسب طاقة حرة حجمية $-\tfrac43\pi r^3\,n\,\delta\mu$ لكنها تدفع
توترًا سطحيًّا $4\pi r^2\sigma$. (أ) بيّن أن الحاجز يبلغ ذروته عند
$r^* = 2\sigma/n\,\delta\mu$. (ب) واشرح: لماذا تبقى الأبخرة
النظيفة فوق المشبَعة والسوائل فوق المسخَّنة
(شبه الاستقرار)، وما الذي يقدّمه الغبار والأيونات والخدوش؟
(ج) وغرف السحاب (وهي كواشف أسلاف في
\cref{ch:b3:particle-physics}) وغرف الفقاعات تعمل \emph{على} هذه
الفيزياء: فأيّ دور يؤديه أثر أيونات الجسيم العابر؟ (د) ولماذا
تُوضع ``حبيبات الغليان'' في دوارق المخبر؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:phase-transitions:10}
لا انتقال في بعد واحد. خذ سلسلة إيزينغ أحادية البعد من $N$
سبينًا، كلها إلى أعلى، وأدخل ``جدار منطقة'' واحدًا (بقلب كل السبينات وراء
رابطة ما). (أ) كلفة طاقة الجدار: $2J$،
مستقلة عن الموضع. (ب) وكسب الإنتروبيا: يستطيع الجدار الجلوس عند
أيّ من $\sim N$ رابطة --- $\Delta S = k_{\text{B}}\ln N$. (ج)
بيّن أن $\Delta F = 2J - k_{\text{B}}T\ln N < 0$ من أجل أيّ $T > 0$ عند
$N$ الكبيرة: فتتكاثر الجدران دائمًا، ويستحيل الترتيب البعيد
المدى --- أي إن \omterm{thm:b3:phase-transitions:meanfield}{لنموذج إيزينغ} أحادي البعد $T_{\text{c}} = 0$. (د) وفي
بعدين يكون الجدار خطًّا طوله $\ell$ يكلّف $2J\ell$ لكن له
$\sim3^\ell$ شكل: أعد التقدير وبيّن أن الترتيب
\emph{ينجو} تحت $T_{\text{c}}$ منتهية --- وهي
حجة بايرلس وراء قيمة أونساغر المضبوطة $2.269\,J/k_{\text{B}}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:phase-transitions:11}
الشمولية مدقَّقةً. يتنبأ المجال المتوسط بالقيم $\beta = 1/2$ و $\gamma
= 1$ (مع $\chi \sim |t|^{-\gamma}$) و $\delta = 3$ (مع $m \sim h^{1/3}$
عند $T_{\text{c}}$). والمقيس، من أجل \omterm{prop:b3:phase-transitions:vdw}{النقطة الحرجة} سائل--غاز
\emph{ومن أجل} المغانط أحادية المحور ثلاثية البعد: $\beta
\approx 0.326$ و $\gamma \approx 1.237$ و $\delta \approx 4.79$.
(أ) فما الذي يبرّر مقارنة مائع بمغنط أصلًا (فما الذي
يتقاسمه وسيطا ترتيبهما --- أي اختيار إشارة واحدة)؟ (ب)
ولماذا يخفق المجال المتوسط قرب $T_{\text{c}}$ (فما الذي يتباعد،
فيبطل إهمال التذبذبات)؟ (ج) ولماذا يخفق
\emph{أقلّ} في الأبعاد الأعلى (عدد الجيران في مقابل مدى
التذبذب؛ ويصير المجال المتوسط مضبوطًا فوق أربعة
أبعاد)؟ (د) اذكر عبرة زمرة إعادة التنظيم في جملتين:
ما الذي يُتوسَّط مقياسًا بعد مقياس، ولماذا لا ينجو من التوسيط إلا
التناظر والبعدية.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:phase-transitions:12}
من المغانط إلى هيغز. ارفع \omterm{def:b3:phase-transitions:order}{وسيط الترتيب} إلى مجال
$\phi(\vect r)$ بكثافة طاقة لانداو $\alpha(T -
T_{\text{c}})\phi^2 + b\phi^4$ زائدَ حدّ تدرّج. (أ) تحت
$T_{\text{c}}$ يجلس المجال عند $\pm\phi_0$: وللاهتزازات الصغيرة
\emph{على طول} البئر تحدّب مُرجِع --- فاحسب
$F''(\phi_0)$ وفسّره، بلغة المجالات، \emph{كتلةً}. (ب) ومن أجل وسيط
ترتيب \emph{مركّب} (بدائرة من الأصغريات --- وهي ``قبّعة
المكسيك'')، لا تكلّف الاهتزازات \emph{حول} الحافة طاقة كامنة:
أي نمط عديم الكتلة --- سمِّ نظائره في المادة
المكثفة (أمواج السبين، وفونونات المائع الفائق). (ج)
وفي النظرية الكهرضعيفة يجلس الخلاء نفسه في بئر كاسرة
للتناظر مثل هذه؛ وتكتسب الجسيمات كتلتها من قيمة المجال
غير المنعدمة، والاهتزاز القطري \emph{هو} بوزون هيغز:
فارسم كل عنصر على (أ)--(ب). (د) وفي جملة واحدة: بمَ
علّمت فيزياء المادة المكثفة فيزياءَ الجسيمات هنا (أي تصدير
\omterm{def:b3:phase-transitions:order}{كسر التناظر}، عبر أندرسون)؟
\end{exercise}

\section{مسألة: مخطط أطوار المطبخ}

\begin{problem}[{مسألة نهاية الأسبوع --- قدور الضغط، وشاي
الجبل، ونهاية الغليان}]\label{pb:b3:phase-transitions:1}
مخطط أطوار الماء معلَّق، من غير أن يُلحظ، فوق كل موقد. وتمشي
هذه المسألة خطوطه الثلاثة وكلاوزيوس--كلابيرون في اليد ---
من سبب مناصفة قدر الضغط لأزمنة الطبخ، إلى سبب خيبة الشاي
في الارتفاع، إلى العالم الغريب وراء \omterm{prop:b3:phase-transitions:vdw}{النقطة
الحرجة} حيث يكفّ الغليان نفسه عن الوجود. والمعطيات: $L_{\text{تبخر}}
= \qty{2.26e6}{J/kg}$ و $L_{\text{انصهار}} = \qty{3.34e5}{J/kg}$؛ و
بخار الماء عند \qty{373}{K} و\qty{1}{atm}: $\Delta v_{\text{تبخر}}
= \qty{1.67}{m^3/kg}$؛ والانصهار: $\Delta v = \qty{-9.1e-5}{m^3/kg}$؛ و
الكتلة الجزيئية $m_{\text{w}} = \qty{3.0e-26}{kg}$؛ \omterm{prop:b3:phase-transitions:vdw}{والنقطة الحرجة}:
$T_{\text{c}} = \qty{647}{K}$ و $P_{\text{c}} = \qty{221}{bar}$.

\medskip\noindent\textbf{الجزء الأول --- خط الغليان.}
\begin{enumerate}
  \item ما الذي يعرّف ``الغليان'' عند ضغط معطًى (فأيّ
        كمونين كيميائيين يتعادلان)؟
  \item احسب ميل $\dd P/\dd T$ لخط التبخر
        عند \qty{100}{\celsius}.
  \item قدر ضغط يحمل \qty{2}{atm}: كامل الميل
        (أو استعمل الصورة الأُسّية مع $Lm_{\text{w}}/
        k_{\text{B}} \approx \qty{4900}{K}$) لإيجاد درجة حرارة
        غليانه.
  \item ومع قاعدة أرينيوس المطبخية في
        \cref{exo:b3:canonical-ensemble:9} (بتضاعف السرعات لكل
        \qty{10}{K})، قدّر التسريع على الطبخ في قدر
        مفتوح.
  \item وفي لاباز (\qty{3600}{m}، $P \approx \qty{0.65}{atm}$):
        احسب درجة حرارة الغليان، ولماذا يُنقَع الشاي نقعًا رديئًا.
  \item ولماذا يرفع إضافة الملح نقطة الغليان (فأيّ كمون $\mu$
        يخفضه المذاب --- وتذكّر
        \cref{exo:b3:grand-canonical:8})؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- خط الانصهار.}
\begin{enumerate}[resume]
  \item احسب $\dd T/\dd P$ للانصهار ولاحظ الإشارة.
  \item وأيّ مشاهدة يومية ترمّز $\Delta v < 0$ (فماذا
        يفعل الجليد في كأسك)، ولماذا يكون ذلك الشذوذ نادرًا
        بين المواد؟
  \item ونهر جليدي سماكته \qty{300}{m}: احسب الضغط عند قاعدته
        وخفض نقطة الانصهار --- فهل يهمّ الانصهار
        بالضغط عند القاعدة في جريان النهر الجليدي؟
  \item وعُد إلى المتزلج في
        \cref{exo:b3:phase-transitions:7}(د): فما الذي يزيّت
        النصل حقًّا؟
  \item ولو غاص الجليد (بميل موجب، مثل كل جسم صلب
        آخر تقريبًا)، فماذا كان الشتاء ليفعل بالبحيرات وأسماكها ---
        تتبّع سلسلة النتائج.
  \item والميل السالب ينتهي عند الضغط العالي حيث تتولّى أشكال
        بلورية أخرى للجليد (الجليد III و V و VI): فماذا يقول
        وجود عشرة أنواع من الجليد عن مشهد \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة}
        لجزيء بسيط واحد؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- نهاية الخط.}
\begin{enumerate}[resume]
  \item ينتهي خط الغليان عند ($T_{\text{c}},
        P_{\text{c}}$): فماذا يصيب الحرارة الكامنة و
        فرق الكثافة كلما اقتربنا من النهاية؟
  \item ووراء \omterm{prop:b3:phase-transitions:vdw}{النقطة الحرجة} يمكن نقل الماء من
        ``شبه السائل'' إلى ``شبه الغاز'' بلا \emph{أيّ} انتقال
        البتة: صف المسار في المستوي $(P, T)$.
  \item وباقترابنا من \omterm{prop:b3:phase-transitions:vdw}{النقطة الحرجة} يصير المائع حليبيًّا:
        اربط هذا الانتهاء مع
        \cref{exo:b3:microcanonical-ensemble:12} ومع
        \cref{pb:b3:perturbation-theory:1}.
  \item وثاني أكسيد الكربون CO$_2$ فوق الحرج ($T_{\text{c}} = \qty{304}{K}$ و
        $P_{\text{c}} = \qty{74}{bar}$) يذيب كالسائل
        وينفذ كالغاز: سمِّ الاستعمال الصناعي الذي
        ينزع الكافيين من البنّ، ولماذا لا يترك المذيب
        بقايا.
  \item والماء فوق الحرج (في مراجل محطات الطاقة فوق
        \qty{221}{bar}): فلماذا يغري إلغاء الغليان ---
        بالفقاعات والأغشية والاحتراق --- مهندسي المراجل؟
  \item والفوّهات الحرارية في أعماق المحيط تصدر ماءً عند
        \qty{400}{\celsius} لا يغلي: تحقق قياسًا إلى
        الضغط المحلي ($\sim\qty{300}{bar}$ عند \qty{3}{km}) و
        مخطط الأطوار.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- قراءة الخريطة كلها.}
\begin{enumerate}[resume]
  \item ارسم (بالكلمات) مخطط $(P, T)$ للماء: الخطوط
        الثلاثة، والنقطة الثلاثية ($\qty{0.01}{\celsius}$ و
        \qty{611}{Pa})، \omterm{prop:b3:phase-transitions:vdw}{والنقطة الحرجة}.
  \item وتحت \qty{611}{Pa} يتسامى الجليد من دون انصهار: فأيّ
        جهاز (التجفيف بالتجميد) وأيّ قلنسوتَي قطبَي كوكب
        (المريخ، CO$_2$ و H$_2$O) يعيشان على هذه الحقيقة؟
  \item ولماذا كانت النقطة الثلاثية للماء --- وهي
        \qty{273.16}{K} بالتعريف بالضبط حتى سنة 2019 --- معيارًا
        جيدًا هكذا لدرجة الحرارة (فكم طورًا متعايشًا يثبّت كم
        درجة حرية)؟
  \item وكل خط من خطوط الخريطة انتقال من المرتبة الأولى؛ ولا يكون
        حرجًا إلا طرفه: أعد ذكر التمييز بوسطاء
        الترتيب وبالحرارة الكامنة.
  \item وخط ندى الهواء الرطب هو الفيزياء نفسها معكوسة:
        اشرح الندى والضباب وشبورة الصباح على نافذة باردة
        بجملة واحدة عن عبور $\mu$.
  \item وكل غطاء قدر تجربة: فالقطيرات تتكاثف عليه
        وتمطر راجعةً --- عيّن انتقالَي الطور في دورة الماء
        المصغَّرة ومكوك الحرارة الكامنة الذي
        يجعل القدر المغطى يغلي أبكر.
  \item لخّص النتيجة الموسومة: صيغة ميل واحدة، $\dd P/
        \dd T = L/T\Delta v$، تشرح \qty{121}{\celsius} لقدر
        الضغط، وشاي الجبل عند \qty{88}{\celsius}، وطفو الجليد
        صاعدًا خط انصهاره إلى الوراء، و
        الغلاية فوق الحرجة حيث ينتهي الغليان عند
        $\qty{647}{K}, \qty{221}{bar}$.
\end{enumerate}
\end{problem}
