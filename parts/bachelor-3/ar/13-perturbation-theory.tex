\chapter{نظرية الاضطرابات}\label{ch:b3:perturbation-theory}

لا تنحلّ بالضبط إلا حفنة من المسائل الكمومية: العلبة
والمذبذب والهدروجين والجملة ذات المستويين. وكل ما عداها --- ذرة
في مجال، أو رابطة جزيء تتصلّب، أو مستوٍ يدفعه
جار --- هيكلٌ من تلك الهياكل المحلولة يرتدي حدًّا
إضافيًا صغيرًا. \omterm{thm:b3:perturbation-theory:corrections}{ونظرية الاضطرابات} فنّ معاملة الحدّ
الإضافي على ما هو: تصحيح، يُحسب مرتبةً مرتبة في
صغره. فقاعدتها في المرتبة الأولى متوسط في سطر واحد؛ وقاعدتها في
المرتبة الثانية تفسّر لماذا \emph{تتنافر} المستويات، ولماذا تكون
كل ذرة قابلة للاستقطاب، ولماذا يكون تجاذب فان دير فالس كونيًا؛
وصورتها المنحلّة تعالج الحالات الدقيقة التي تعيد فيها مستويات
شبه متساوية تنظيمها كليًا؛ وصورتها المتعلقة بالزمن --- أي \omterm{thm:b3:perturbation-theory:golden}{قاعدة
فيرمي الذهبية} --- هي المعادلة وراء كل خط امتصاص ومعدل
تفكك ومقطع انتثار يُقاس في مخبر. وثوابًا جاريًا،
يختم الفصل بحساب لون السماء.

\section{انزياحات صغيرة: النظرية غير المنحلّة}

\begin{theorem}[التصحيحات الاضطرابية]\label{thm:b3:perturbation-theory:corrections}
\index{نظرية الاضطرابات}
ليكن $\hat H = \hat H_0 + \hat V$ مع $\hat H_0$ محلولة
($\hat H_0\ket n = E_n\ket n$، وطيفها غير منحلّ) و $\hat V$
صغيرة. عندئذٍ، مرتبةً مرتبة في $\hat V$،
\[
E_n' = E_n + \bra n\hat V\ket n
 + \sum_{m \neq n}\frac{|\bra m\hat V\ket n|^2}{E_n - E_m}
 + \cdots
\]
وتكتسب الحالة خلائط $\ket{n'} = \ket n + \sum_{m\neq
n}\dfrac{\bra m\hat V\ket n}{E_n - E_m}\,\ket m + \cdots$
\emph{في المرتبة الأولى}: الانزياح هو مجرد متوسط
الاضطراب في الحالة غير المضطربة. \emph{وفي المرتبة الثانية}:
تدفع التقارنات مع المستويات الأخرى $E_n$ \emph{بعيدًا} عنها ---
فكل $m$ فوقها يدفعها نزولًا وكل ما تحتها يدفعها صعودًا --- ومن ثمّ
يكون انزياح الحالة الأساسية في المرتبة الثانية سالبًا دائمًا. وأما الصلاحية:
فيجب أن تكون نسب الخلط $|\bra m\hat V\ket n/(E_n - E_m)|$ صغيرة.
\end{theorem}

\begin{proof}[برهان جزئي]
ننشر $E' = E + \epsilon_1 + \epsilon_2 + \cdots$ و $\ket{\psi} =
\ket n + \ket{\psi_1} + \cdots$ بقوى $\hat V$، وندخلهما في
$(\hat H_0 + \hat V)\ket\psi = E'\ket\psi$، ونطابق المراتب.
وإسقاط معادلة المرتبة الأولى على $\bra n$ يعطي $\epsilon_1 =
\bra n\hat V\ket n$؛ وإسقاطها على $\bra m$ (مع $m \neq n$) يعطي
معاملات الخلط؛ وحمل تلك إلى معادلة المرتبة الثانية
وإسقاطها على $\bra n$ من جديد يعطي $\epsilon_2$. وأما عبارة
الإشارة من أجل الحالة الأساسية: فكل مقام $E_0 - E_m$
سالب في حين أن البسوط مربعات.
\end{proof}

\begin{proposition}[اضطراب له جواب مضبوط]\label{prop:b3:perturbation-theory:oscillator-field}
مذبذب شحنته $q$ في مجال منتظم $\mathcal E$ له
$\hat V = -q\mathcal E\hat x$. وإكمال المربع يحلّه
بالضبط: فالكمون هو القطع المكافئ نفسه، مزاحًا، وكل
المستويات مخفوضة بالمقدار $q^2\mathcal E^2/2m\omega^2$. ويجب أن
توافق \omterm{thm:b3:perturbation-theory:corrections}{نظرية الاضطرابات} --- وهي توافق: إذ تنعدم المرتبة الأولى (لأن $\langle\hat
x\rangle_n = 0$)، ويعطي مجموع المرتبة الثانية، الذي لا يقارن فيه $\hat x$
العددَ $n$ إلا مع $n \pm 1$، بالضبط
$-q^2\mathcal E^2/2m\omega^2$ لكل مستوٍ
(\cref{exo:b3:perturbation-theory:2}). فحالة محلولة واحدة، متحقَّق منها
بالطريقتين، أفضل معايرة يمكن أن تنالها طريقة.
\end{proposition}

\begin{proof}
يعطي العنصران المصفوفيان ومقاماهما
\[
\frac{|\bra{n+1}\hat x\ket{n}|^2}{-\hbar\omega}
 + \frac{|\bra{n-1}\hat x\ket n|^2}{+\hbar\omega}
 = \frac{\hbar}{2m\omega}\,\frac{-(n+1) + n}{\hbar\omega}
 = -\frac{1}{2m\omega^2} ;
\]
ثم نضرب في $q^2\mathcal E^2$.
\end{proof}

\section{حين تكون المستويات منحلّة}

\begin{theorem}[نظرية الاضطرابات المنحلّة]\label{thm:b3:perturbation-theory:degenerate}
\index{نظرية الاضطرابات المنحلّة}
إذا كان $E_n$ منحلًّا، قسمت الصيغة أعلاه على الصفر --- فقد
يعيد الاضطراب تنظيم الحالات المنحلّة كليًا،
والعلاج أن ندعه يفعل: أي \emph{نقطّر $\hat V$ مقصورًا على
الفضاء الجزئي المنحلّ}. فقيمه الذاتية هي انزياحات المرتبة الأولى؛
ومتجهاته الذاتية هي حالات المرتبة الصفرية الصحيحة، أي
التراكيب التي ينتقيها الاضطراب نفسه.
\end{theorem}
\admitted

\begin{example}[أثر شتارك الخطي]\label{ex:b3:perturbation-theory:stark}
\index{أثر شتارك}
يحوي المستوي $n = 2$ في الهدروجين الحالتين المنحلّتين $2s$ و $2p$.
وفي مجال $\mathcal E\vect e_z$، يقارن $\hat V = e\mathcal E\hat z$
الحالة $2s$ مع $2p_0$ بالعنصر $\bra{2s}\hat z\ket{2p_0} = -3a_0$:
فيعطي تقطير القالب $2\times2$ الانزياحين
\[
\Delta E = \pm\,3e a_0\mathcal E ,
\]
وهما \emph{خطيان} في المجال --- في حين لا تنزاح الحالة الأساسية غير
المنحلّة إلا تربيعيًا. وعند $\mathcal E = \qty{e7}{V/m}$: يكون الانزياح $\pm
\qty{1.6}{meV}$، وهو انفصام سهل التمييز. والحالتان اللتان تقومان
بالانفصام هما $(\ket{2s} \mp \ket{2p_0})/\sqrt2$ --- أي هجينان لهما
ثنائي قطب كهربائي دائم، ولا يمكنان إلا لأن الانحلال ترك
الذرة حرة في أن تستقطب في المرتبة الصفرية. (والتهجين $s$--$p$
نفسه، مقودًا بالجيران بدل المجال، هو كيمياء الكربون في كل
جزيء عضوي.)
\end{example}

\section{الانتقالات: القاعدة الذهبية}

\begin{theorem}[الاضطرابات المتعلقة بالزمن]\label{thm:b3:perturbation-theory:golden}
\index{قاعدة فيرمي الذهبية}
شغّل $\hat V(t) = \hat V\cos\omega t$ عند $t = 0$. عندئذٍ، حتى المرتبة
الأولى، يكون احتمال إيجاد الجملة في $\ket f$ عند الزمن
$t$، وقد بدأت في $\ket i$، هو
\[
\mathcal P_{i\to f}(t) = \frac{|\bra f\hat V\ket i|^2}{4\hbar^2}\;
\frac{\sin^2\!\big[(\omega_{fi} - \omega)\,t/2\big]}
{\big[(\omega_{fi} - \omega)/2\big]^2} ,
\qquad \omega_{fi} = \frac{E_f - E_i}{\hbar} :
\]
أي رنين حادّ الذروة عند $\hbar\omega = E_f - E_i$ ---
وهو امتصاص لمّا $E_f > E_i$، و\emph{إصدار مستحثّ} من أجل
الحدّ القرين مع $E_f < E_i$. وحين تؤلف الحالات النهائية
سلسلة مستمرة كثافتها $\rho(E)$، يصير نموّ الذروة \emph{معدلًا}
ثابتًا:
\[
\Gamma_{i\to f} = \frac{2\pi}{\hbar}\,
|\bra f\hat V\ket i|^2\,\rho(E_f)
\]
--- وهي \emph{\omterm{thm:b3:perturbation-theory:golden}{قاعدة فيرمي الذهبية}}، الصيغة الأمّ لمعدلات التفكك
ومعاملات الامتصاص والمقاطع.
\end{theorem}

\begin{proof}[برهان جزئي]
سعة المرتبة الأولى: $c_f(t) = -\tfrac{\iu}{\hbar}\int_0^t\bra
f\hat V(t')\ket i\,\eu^{\iu\omega_{fi}t'}\dd t'$؛ وبإبقاء الأسّي
القريب من الرنين ومكاملته نجد صورة $\sin^2$
المذكورة. وأما السلسلة المستمرة: فبدلالة الانحراف
$x$، يكون لعامل الرنين ارتفاع $t^2$ وعرض $\sim 2\pi/t$
--- ومساحة $2\pi t$ --- فيستبدل جمعُه على سلسلة مستمرة ملساء
به المقدارَ $2\pi t\,\rho$: أي احتمال خطي في $t$، وهو معدل.
\end{proof}

\begin{example}[ما الذي تديره القاعدة الذهبية]\label{ex:b3:perturbation-theory:applications}
قواعد الانتقاء: لا يقع الانتقال إلا إذا نجا العنصر المصفوفي
$\bra f\hat V\ket i$ --- ومن أجل الضوء، $\hat V \propto \hat
z$، وتقتل زوجيته وعزمه الحركي كل شيء عدا
$\Delta l = \pm1$ و $\Delta m = 0, \pm1$: أي القواعد التي استُدعيت منذ
\cref{ch:b3:quantum-angular-momentum}، وقد صارت الآن مبرهنات. والمعدلات:
يُحسب التفكك التلقائي للحالة $2p$ في الهدروجين فيعطي
\qty{1.6}{ns} (\cref{exo:b3:perturbation-theory:8})؛ أما انتقال
ميزر النشادر، بتواتره الميكروي $\omega^3$، فيستغرق شهورًا ---
والعامل $\omega^3$ في معدلات الإشعاع هو سبب عيش الخطوط الراديوية
(\cref{ex:b3:spin-two-level:hyperfine}) ملايين السنين بينما تومض
الخطوط فوق البنفسجية في نانوثوانٍ. وأطياف الامتصاص،
والتأيّن الضوئي، وتفكك بيتا النووي
(\cref{ch:b3:nuclear-physics})، ومقاطع الانتثار
(\cref{ch:b3:scattering-theory}): كلها هذه الصيغة الواحدة بعناصر
مصفوفية وإحصاءات حالات مختلفة.
\end{example}

\begin{method}[اختيار الأداة الاضطرابية الصحيحة]\label{met:b3:perturbation-theory:method}
(1) سؤال ساكن ومستوٍ غير منحلّ: توسيط $\hat V$ (في المرتبة
الأولى)؛ فإن انعدم ذلك بالتناظر، فالمرتبة الثانية --- وتوقّع
تنافر المستويات. (2) ومستوٍ منحلّ: قطّر $\hat V$ في
القالب المنحلّ أولًا؛ فالتراكيب المكيَّفة مع التناظر هي التي تقوم
بالانفصام. (3) ومعدلات الانتقال: القاعدة الذهبية --- عنصر مصفوفي
واحد وكثافة حالات واحدة؛ وتحقق من قواعد الانتقاء قبل
حساب أيّ شيء. (4) واختبر النشر دائمًا: فيجب أن تكون نسبة الخلط
$|V_{mn}/(E_n - E_m)| \ll 1$، وإلا فقطّر بالضبط (بصيغة
المستويين). (5) والحالات الحدية المحلولة بالضبط (كالمذبذب في مجال)
معايرات مجانية: فاستعملها.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:1}
جسيم في علبة $[0, a]$ يحسّ الكمون الإضافي $V(x) =
\lambda x/a$. (أ) احسب انزياح كل مستوٍ في المرتبة الأولى
(استعمل $\langle x\rangle = a/2$ في أيّ حالة من حالات العلبة --- وبرّر ذلك).
(ب) لماذا يكون الانزياح واحدًا لكل المستويات هنا؟ (ج) ما التغير
في \emph{التباعدات} في المرتبة الأولى، ولماذا قد يفوت قياسَ
تواترات الانتقال هذا الاضطرابُ كليًا؟ (د) متى تكف المرتبة الأولى عن أن تكون موثوقة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:2}
نفّذ حساب المرتبة الثانية في
\cref{prop:b3:perturbation-theory:oscillator-field} كاملًا: العناصر
المصفوفية للمؤثر $\hat x$، والحدّان، وتقاصّ $n$، و
التوافق التامّ مع إكمال المربع. ولماذا لا ينعدم تصحيح
\emph{الحالة} في المرتبة الأولى مع أن تصحيح الطاقة ينعدم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:3}
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{الهاميلتونية} ذات المستويين $\begin{pmatrix}E_1 & v\\ v &
E_2\end{pmatrix}$ مع $E_1 < E_2$ و $v$ حقيقي. (أ) أوجد القيم
الذاتية المضبوطة. (ب) انشرها من أجل $|v| \ll E_2 - E_1$ وعيّن
صيغة المرتبة الثانية. (ج) بيّن أن المستويين يتنافران: فالفجوة تفوق
$E_2 - E_1$ دائمًا. (د) وعند $E_1 = E_2$: ماذا تعطي \omterm{thm:b3:perturbation-theory:corrections}{نظرية
الاضطرابات}، وماذا يعطي الجواب المضبوط، وأيّ مبرهنة من
مبرهنات الفصل توفّق بينهما؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:4}
من أجل \cref{thm:b3:perturbation-theory:corrections}: (أ) بيّن أن
تصحيح الحالة في المرتبة الأولى متعامد مع $\ket n$؛ (ب) بيّن أن
معامل خلط $\ket m$ هو $V_{mn}/(E_n - E_m)$ و
اذكر معيار الصلاحية؛ (ج) مستوٍ يبعد $\qty{1}{eV}$ عن
أقرب جيرانه ويتقارن معه بالمقدار \qty{0.1}{eV}: قدّر
احتمال الخلط؛ (د) والسؤال نفسه مع فجوة \qty{10}{meV} ---
فأيّ أداة ينبغي أن تحلّ محل النشر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:5}
الروابط اللاتوافقية. أضف $\hat V = \beta\hat x^3 + \gamma\hat x^4$ إلى
المذبذب. (أ) بيّن أن الحدّ $x^3$ لا يزيح أيّ مستوٍ في المرتبة
الأولى (بالزوجية). (ب) بيّن أن الحدّ $x^4$ يعطي $\Delta E_n =
3\gamma(\hbar/2m\omega)^2(2n^2 + 2n + 1)$ (انشر $\hat x^4$ بمؤثري
السلّم؛ فلا تبقى غير الحدود ``المتوازنة''). (ج) من أجل رابطة حقيقية يكون $\gamma$ الفعّال سالبًا (إذ
تليّن البئر إلى الخارج): بيّن أن تباعد المستويات \emph{يتقلص} عندئذٍ مع
$n$.
(د) أيّ معلم مشاهَد في أطياف الجزيئات
(\cref{exo:b3:harmonic-oscillator:4}) يعيد هذا إنتاجه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:6}
في العلبة المربعة الثنائية البعد، يُضطرب الزوج المنحلّ $\ket{1,2},
\ket{2,1}$ بالمؤثر $\hat V = \lambda\,xy$. (أ) اشرح لماذا
تخفق النظرية غير المنحلّة. (ب) احسب مصفوفة $2\times2$ للمؤثر
$\hat V$ في الزوج (فالتكاملات تتحلل إلى عوامل؛ ولدينا $\int_0^a
x\sin(\pi x/a)\sin(2\pi x/a)\dd x = -8a^2/9\pi^2$). (ج) أوجد
الانفصامات وحالات المرتبة الصفرية الصحيحة. (د) قارن مع
تحليل التناظر في \cref{exo:b3:quantum-formalism:9}: فأيّ
جواب أعطاه التناظر مجانًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:7}
أثرا شتارك، الخطي والتربيعي. (أ) باستعمال
\cref{ex:b3:perturbation-theory:stark}، احسب انفصام $n = 2$
عند $\mathcal E = \qty{2.5e6}{V/m}$. (ب) لماذا \emph{لا} تُظهر
الحالة الأساسية انزياحًا خطيًا (ولسببين: الزوجية، وانعدام
شريك منحلّ)؟ (ج) يعرّف انزياحها التربيعي القابلية
للاستقطاب، $\Delta E = -\tfrac12\alpha\mathcal E^2$: قدّر
$\alpha \sim e^2a_0^2/E_{\text{I}}$ وعبّر عنه بوحدة
$a_0^3$ (والجواب المضبوط $4.5\,a_0^3 \times
4\pi\varepsilon_0$). (د) أيّ خاصية يومية للمادة ---
أي استجابتها العازلة --- حسبتَ لتوّك بذرتها الذرّية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:8}
الإصدار التلقائي (بمعطًى واحد مُسلَّم به): تتفكك حالة مثارة
بالمعدل $A = \omega^3|d_{fi}|^2/3\pi\varepsilon_0\hbar c^3$،
حيث تحدّد العناصر المصفوفية من نمط $d_{fi} = e\bra f\hat z\ket i$
ثنائي القطب. (أ) من أجل الانتقال $2p \to 1s$ في الهدروجين: مع $|d| \approx 0.74\,
ea_0$ و $\lambda = \qty{121.6}{nm}$، احسب $A$ و
العمر. (ب) احجّم النتيجة لخط الصوديوم D. (ج) واحجّمها إلى
الانتقال فوق الدقيق عند 21 سنتيمترًا (حيث يصير $|d| \to$ عزمًا مغناطيسيًّا، ويهبط المعدل
بعامل آخر من رتبة $\alpha^2$؛ واقبل $A \approx
\qty{2.9e-15}{s^{-1}}$): وتحقق من ``عشرة ملايين سنة''. (د)
انطلاقًا من العامل $\omega^3$: لماذا تكون الخطوط فوق البنفسجية سريعة والخطوط
الراديوية أبدية، ولماذا جعل ذلك \omterm{ex:b3:spin-two-level:hyperfine}{خط 21 سنتيمترًا}
\emph{قابلًا للتنبؤ} لكنه يكاد لا يُكشف في مخبر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:9}
قواعد الانتقاء تكاملاتٍ. (أ) بيّن أن $\bra{100}\hat z\ket{200} =
0$ بالزوجية. (ب) بيّن أن $\bra{100}\hat z\ket{21m} = 0$ ما لم يكن $m =
0$ (بتكامل $\varphi$). (ج) اذكر القاعدتين الناتجتين $\Delta
l = \pm1$ و $\Delta m = 0, \pm1$ وقراءتهما الفيزيائية (أي
سبين الفوتون). (د) لا تستطيع الحالة $2s$ بلوغ $1s$ بأيّ
طريق ثنائي القطب: فبم يتنبأ ذلك في شأن عمرها، وأيّ
طريق ``ممنوع'' بفوتونين تسلكه الطبيعة فعلًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:10}
القابلية للاستقطاب بأمانة. انزياح الحالة الأساسية للهدروجين
في مجال، في المرتبة الثانية، هو $\Delta E = -e^2\mathcal E^2\sum_{m\neq
0}|\bra m\hat z\ket 0|^2/(E_m - E_0)$. (أ) حاصر المجموع باستبدال
أصغر فجوة $E_2 - E_1 =
\tfrac34 E_{\text{I}}$ بكل مقام واستعمال علاقة التمام $\sum_m|\bra
m\hat z\ket0|^2 = \langle z^2\rangle_0 = a_0^2$: فتجد $\alpha
\le \tfrac{16}3\,a_0^3$ (بوحدات من نمط غاوس أي
$4\pi\varepsilon_0$). (ب) قارنه بالقيمة المضبوطة $\tfrac92 a_0^3$.
(ج) انطلاقًا من $\alpha$، قدّر قرينة انكسار غاز الهدروجين
عند الكثافة الجوّية ($n - 1 \approx N\alpha/2\varepsilon_0
\cdot e^2\dots$ --- استعمل $n - 1 = N\alpha_{\text{SI}}/2
\varepsilon_0$) وقارنها بالقيمة المقيسة $\num{1.3e-4}$.
(د) اذكر في جملة واحدة السلسلة ذرة $\to$ قابلية للاستقطاب
$\to$ انكسار التي سيركبها الجزء الثاني من مسألة نهاية الأسبوع.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:11}
التقاطعات المتجنَّبة والمسح البطيء. جملة ذات مستويين لها طاقتان
عاريتان $\pm\lambda t$ تتقاطعان عند $t = 0$، وتتقارنان بالمقدار $v$. (أ)
ارسم المستويين المضبوطين تقريبيًا بدلالة $t$: أي تقاطع متجنَّب فجوته
$2|v|$. (ب) إذا كان المسح \emph{بطيئًا}، تتبع الجملة
المنحني الأدنى (بمبرهنة الكظومية، وهي مُسلَّم بها): ففي أيّ حالة
تنتهي؟ (ج) يقارن معيار لانداو--زينر زمن المسح
عبر منطقة الفجوة، $\sim v/\lambda$، بالساعة الداخلية
$\hbar/v$: أعطِ شرط التتبع الكظوم. (د) واستعمالان:
حالة كيوبت تُنقل بزقزقة بطيئة؛ وتصادم ذري تقفز فيه شحنة بين
نواتين --- اشرح أحدهما في
جملتين.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:12}
المرتبة الثانية متشائمة، ومبرهنات أخرى. (أ) برهن على أن
تصحيح المرتبة الثانية للحالة \emph{الأساسية} سالب
دائمًا. (ب) استنتج (مع
\cref{exo:b3:harmonic-oscillator:10}) لماذا تكون قوى فان دير فالس
جاذبة دائمًا بين ذرات في حالتها الأساسية. (ج) بيّن أن
نظرية المرتبة الأولى \emph{تبالغ} في تقدير كل طاقة
أساسية: فهي حدّ تغايري (قدّر $\bra{\psi_0}\hat
H\ket{\psi_0}$ بالحالة غير المضطربة دالةً تجريبية). (د)
تحقق من الاتساق مع \cref{ex:b3:perturbation-theory:stark}: فلماذا
لا يكون \omterm{ex:b3:perturbation-theory:stark}{أثر شتارك} \emph{الخطي} خرقًا للسؤال (أ)؟
\end{exercise}

\section{مسألة: لون السماء}

\begin{problem}[{مسألة نهاية الأسبوع --- انتثار رايلي من ذرات
مضطربة}]\label{pb:b3:perturbation-theory:1}
لماذا تكون سماء النهار زرقاء والغروب أحمر والسحابة بيضاء؟
والجواب الكامل سلسلة عبر هذا الفصل: فموجة ضوئية
تضطرب كل جزيء هواء؛ ويعيد ثنائي القطب المستحثّ الإشعاع (وهو
إشعاع ثنائي القطب في كتاب السنة الثانية)؛ ويقرّر التداخل
ما الذي ينجو. المعطيات: القابلية الجزيئية للاستقطاب في الهواء
$\alpha/4\pi\varepsilon_0 \approx \qty{2.2e-30}{m^3}$؛ والكثافة
العددية $N = \qty{2.5e25}{m^{-3}}$؛ والمجال المرئي
$450$--\qty{700}{nm}.

\medskip\noindent\textbf{الجزء الأول --- الجزيء المضطرب.}
\begin{enumerate}
  \item مجال ساكن $\mathcal E$ يزيح الحالة الأساسية لجزيء
        بالمقدار $-\tfrac12\alpha\mathcal E^2$: اربط هذه
        العبارة \omterm{thm:b3:perturbation-theory:corrections}{بنظرية الاضطرابات} في المرتبة الثانية (أيّ
        صيغة، ولماذا سالبة؟).
  \item ويعطي $\alpha$ نفسه ثنائي القطب المستحثّ $p =
        \alpha\mathcal E$: فمن أجل مجال ضوء الشمس $\mathcal E \sim
        \qty{800}{V/m}$، احسب ثنائي القطب المستحثّ في جزيء
        نتروجين واحد وقارنه بالمقدار $ea_0$.
  \item لماذا يجوز معالجة المجال الضوئي (تواتره $\num{5e14}$ هرتز)
        بالقابلية \emph{الساكنة} للاستقطاب من أجل
        جزيئات تقع انتقالاتها الإلكترونية في
        فوق البنفسجي؟ (قارن $\hbar\omega$ بالفجوة؛ وهذه هي
        نهاية الابتعاد عن الرنين في مقام القاعدة
        الذهبية.)
  \item ثنائي قطب متذبذب $p_0\cos\omega t$ يشعّ الاستطاعة
        المتوسطة $P = p_0^2\omega^4/12\pi\varepsilon_0c^3$
        (كتاب السنة الثانية). وبضمّ ذلك إلى $p_0 = \alpha\mathcal
        E_0$: كيف تتوقف الاستطاعة المنتثرة على $\omega$ عند
        إضاءة ثابتة؟
  \item احسب نسبة الاستطاعتين المنتثرتين عند \qty{450}{nm}
        وعند \qty{700}{nm}.
  \item في جملة واحدة: لماذا تكون السماء زرقاء لا بنفسجية
        (وثمة مكوّنان: طيف الشمس الهابط نحو
        البنفسجي، وحساسية العين)؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- من جزيء واحد إلى
السماء.}
\begin{enumerate}[resume]
  \item يخرج مقطع الانتثار بالصورة $\sigma =
        \dfrac{8\pi}{3}\Big(\dfrac{\alpha}{4\pi\varepsilon_0}
        \Big)^2\dfrac{\omega^4}{c^4}$: تحقق من أبعاده و
        قدّره عند \qty{550}{nm}.
  \item طول التوهين هو $\ell = 1/N\sigma$: قدّره
        عند \qty{550}{nm} وقارنه بسماكة الغلاف
        الجوّي ($\sim\qty{8}{km}$ مكافئة عند كثافة مستوى
        البحر).
  \item عند \qty{450}{nm} في مقابل \qty{700}{nm}، ما نسبة
        حزمة شمسية شاقولية تُنتثر خارجًا؟ (استعمل $1 -
        \eu^{-L/\ell}$ مع $L = \qty{8}{km}$.)
  \item عند الغروب يكون المسار أطول ثلاثين مرة: أعد حساب
        نجاة الأزرق واشرح لون الشمس المنخفضة ---
        ولون الضوء الذي يحمرّ به القمر في خسوف
        قمري.
  \item لماذا يكون ضوء السماء \emph{مستقطبًا} عند $90^\circ$
        من الشمس (تذكّر نقش إشعاع ثنائي القطب: فلا
        إصدار على محور ثنائي القطب)؟
  \item يقال إن النحل والفايكنغ اهتدوا بهذا
        الاستقطاب: فماذا يفعل به يوم غائم، و
        لماذا؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- لماذا ليست السماء
أسطع.}
\begin{enumerate}[resume]
  \item مفارقة: في وسط منتظم تمامًا، تتداخل الموجيات
        الآتية من عناصر الحجم المتجاورة تداخلًا هدّامًا جانبيًا،
        فلا ينتثر ضوء \emph{البتة} (ولا يبقى غير الانكسار). فما
        الذي يكسر انتظام الهواء فعلًا --- وما الموزَّع عشوائيًا؟
  \item تجعل تذبذبات الكثافة في غاز مثالي شدّات الانتثار
        من الجزيئات المستقلة \emph{تُجمع}: اذكر
        لماذا يقتل الاستقلال (أي المواضع العشوائية والتباعدات
        بمقياس طول الموجة) التداخلَ الهدّام.
  \item انطلاقًا من $\ell$ عندك عند \qty{550}{nm}: ما نسبة
        ضوء الشمس التي ينتثرها غلاف جوي واحد في يوم صافٍ --- وهل
        يطابق العدد سطوع السماء اليومي في مقابل الشمس؟
  \item قطرة سحاب (\qty{10}{\micro m}) تحوي $\sim10^{12}$
        جزيء داخل طول موجة: فتُجمع موجياتها
        \emph{ترابطيًا}. فكيف تتغير الاستطاعة المنتثرة عندئذٍ
        مع عدد الجزيئات، ولماذا يختفي انتقاء اللون بالعامل
        $\omega^4$ (إذ تنتثر كل أطوال الموجات بقوة): فما لون السحابة؟
  \item الحليب والضباب والطلاء الأبيض بيض للسبب نفسه:
        اذكر القاعدة العامة --- فالمنتثرات الصغيرة أمام
        $\lambda$ تلوّن الضوء، والكبيرة أمام $\lambda$ تبيّضه.
  \item الانتهاب الحرج: بجوار النقطة الحرجة لسائل
        (في فصول تغير الطور بكتاب السنة الأولى، وفي
        \cref{ch:b3:phase-transitions} قادمًا)، تنمو تذبذبات الكثافة إلى مقاييس ضوئية
        فيصير المائع حليبيًا: اربط ذلك بالبند 13.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- السلسلة، مكتملةً.}
\begin{enumerate}[resume]
  \item الموجيات المنتثرة إلى الأمام \emph{لا} تتقاصّ: بل
        تتداخل مع الحزمة وتبطئ طورها --- وهي
        قرينة الانكسار. وباستعمال $n - 1 = N\alpha/2\varepsilon_0$
        (مع $\alpha$ بالنظام الدولي)، قدّر $n - 1$ للهواء و
        قارنه بالقيمة المقيسة \num{2.9e-4}.
  \item فثابت واحد $\alpha$، محسوب \omterm{thm:b3:perturbation-theory:corrections}{بنظرية الاضطرابات} في
        المرتبة الثانية، يحدّد ثلاث ظواهر دفعةً واحدة:
        سمّها (الانزياح في مجال ساكن؛ وزرقة السماء؛ وانحناء
        الضوء في الهواء).
  \item ويعقّد امتصاصُ الأوزون والهباء والانتثار المتعدد
        السماواتِ الحقيقية: سمّ مشاهَدة واحدة لكل منها
        (ألوان الشفق؛ والأفق الحليبي؛ وسطوع السماء المتبقي
        عند السمت بعد الغروب).
  \item لماذا يحمل القمر، وهو بلا هواء، سماء نهار
        \emph{سوداء} --- وبم أكدت كل صورة من سطحه
        في شأن أصل سمائنا؟
  \item سماء المريخ نهارًا بلون الكراميل وغروبه \emph{أزرق}:
        فهواؤه الرقيق ينتثر قليلًا، ويقوم بالعمل غبار معلّق
        بحبيبات ميكرومترية. وباستعمال قاعدة
        البند 18، اشرح كيف يعكس الغبار منطق الألوان.
  \item ومن أجل الأشعة السينية، يفوق $\hbar\omega$ كل انتقال جزيئي
        بكثير: فتستجيب الإلكترونات كأنها حرة ويفقد الانتثار
        (انتثار طومسون) عاملَه $\omega^4$. فماذا يحلّ
        بآلية ``زرقة السماء'' في ذلك النظام --- ولماذا تكون تلك
        الاستجابة المسطحة بالضبط ما يجعل الأشعة السينية مسابر نظيفة
        لكثافة الإلكترونات؟
  \item لخّص النتيجة الموسومة: قابلية للاستقطاب مقدارها $\qty{2.2e-30}{m^3}$،
        مربَّعة ومضروبة في $\omega^4$،
        تعطي طول انتثار بمقياس \qty{60}{km} في الأخضر
        --- طويلًا بما يكفي لتكون سماء الظهيرة لطيفة، وقصيرًا بما
        يكفي ليحمرّ الغروب وتزرقّ السماوات، بنقش
        استقطاب غابة من ثنائيات قطب مقودة.
\end{enumerate}
\end{problem}
