\chapter{الجمهور القانوني الكبير}\label{ch:b3:grand-canonical}

قطرة الماء في هواء رطب لا تنمو ولا تنكمش؛ والأكسجين
يرتبط بالهيموغلوبين في الرئتين وينفكّ عنه في العضلات؛ و
البطارية تسوق الإلكترونات عبر هاتف لأنها ``أغلى''
في مسرى منها في الآخر. وفي كل حالة لا تكون العملة
الموازَنة درجةَ الحرارة بل \emph{\omterm{thm:b3:grand-canonical:distribution}{الكمون
الكيميائي}} $\mu$: أي كلفة \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة} لجسيم واحد إضافي. ويفتح
هذا الفصل صمّام الخزان الثاني --- أي الجسيمات مع
الطاقة --- ويبني \omterm{thm:b3:grand-canonical:distribution}{الجمهور القانوني الكبير}، وجزاؤه
مردود مذهل: فإشغال مستوٍ كمومي واحد،
سواء أكان فرميونًا أم بوزونًا، يتبع في سطرين، وتتبعه
صيغ الفصل التالي الأساسية. وتمتد تطبيقاته من
متساويات حرارة الامتزاز في الكمامات والمحوّلات الحفّازة إلى
\omterm{prop:b3:grand-canonical:saha}{معادلة ساها}، التي قرأت بها طالبة دكتوراه في هارفارد سنة 1925
درجات حرارة النجوم واكتشفت، على غير كل
توقع، أن الكون مصنوع من الهدروجين.

\section{الجسيمات تنضمّ إلى المفاوضة}

\begin{theorem}[التوزيع القانوني الكبير]\label{thm:b3:grand-canonical:distribution}
\index{الجمهور القانوني الكبير}\index{الكمون الكيميائي}
الجملة التي تتبادل الطاقة \emph{والجسيمات} مع خزان عند
درجة الحرارة $T$ \omterm{thm:b3:grand-canonical:distribution}{والكمون الكيميائي} $\mu$ تشغل حالتها $s$
(بطاقة $E_s$ وعدد جسيمات $N_s$) باحتمال
\[
\mathcal P_s = \frac{\eu^{-\beta(E_s - \mu N_s)}}{\Xi} , \qquad
\Xi = \sum_s\eu^{-\beta(E_s - \mu N_s)} ,
\]
وهي \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{دالة التقسيم} الكبرى}. ومن $\Xi$: $\langle N\rangle
= k_{\text{B}}T\,\partial\ln\Xi/\partial\mu$، ويطيع \emph{الكمون
الكبير} $\Phi = -k_{\text{B}}T\ln\Xi$ العلاقةَ $\Phi = -PV$.
\end{theorem}

\begin{proof}[برهان جزئي]
كما في \cref{thm:b3:canonical-ensemble:boltzmann}، ننشر
إنتروبيا الخزان في السحبين معًا:
$S_{\text{خزان}}(E - E_s, N - N_s) \approx S_{\text{خزان}} - E_s/T +
\mu N_s/T$، باستعمال $\partial S/\partial N = -\mu/T$
(\cref{thm:b3:microcanonical-ensemble:temperature}). وتتبع
المتوسطات بالاشتقاق؛ وأما $\Phi = -PV$ فمقبولة (إذ تتبع
من الشمول).
\end{proof}

\begin{definition}[ما معنى $\mu$]\label{def:b3:grand-canonical:mu}
\index{الكمون الكيميائي!معناه}
$\mu$ هي كلفة إضافة جسيم واحد، \omterm{prop:b3:canonical-ensemble:machine}{بالطاقة الحرة}، عند
$T$ و $V$ ثابتين. وتسيل الجسيمات تلقائيًّا من $\mu$ العالية إلى $\mu$
المنخفضة --- فالانتشار والتناضح والتبخر والتفاعلات الكيميائية و
التيارات الكهربائية كلها تدرّجات في $\mu$ تسترخي. وعند التوازن
بين أيّ طورين أو موضعين تتساوى $\mu$ على الجانبين ---
وهي المساواة الثالثة (مع $T$ و $P$) التي تحكم التعايش. ومن أجل
غاز مثالي كلاسيكي،
\[
\mu = k_{\text{B}}T\,\ln\!\big(n\lambda_T^3\big) < 0
\quad\text{(غاز مخلخل)} :
\]
فإضافة جسيم إلى غاز مخلخل \emph{تخفض} \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة}،
لأن كسب الإنتروبيا يرجح على أيّ كلفة طاقية --- وهي الإشارة التي
تسوق التبخر في هواء جافّ. والبطارية آلة $\mu$:
فجهدها هو هبوط \omterm{thm:b3:grand-canonical:distribution}{الكمون الكيميائي} لكل إلكترون بين
مسريَيها، $U = \Delta\mu/e$ --- فالفولتات سعر صرف كيمياء
الإلكترون، ولذلك تعطي الخلايا فولتات من رقم واحد.
\end{definition}

\section{مستوًى واحدًا في كل مرة: الحيلة الأساسية}

\begin{theorem}[إشغال مستوٍ كمومي مفرد]\label{thm:b3:grand-canonical:occupation}
\index{توزيع فيرمي--ديراك}\index{توزيع بوز--أينشتاين}
خذ ``الجملة'' مستوًى واحدًا لجسيم مفرد طاقته
$\epsilon$، على تماسّ كبير مع غاز كل ما عداه.
\emph{الفرميونات} (وإشغالها $0$ أو $1$):
\[
\Xi = 1 + \eu^{-\beta(\epsilon - \mu)}
\;\Longrightarrow\;
\bar n_{\text{FD}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} + 1} .
\]
\emph{البوزونات} (وإشغالها $0, 1, 2, \dots$):
\[
\Xi = \frac{1}{1 - \eu^{-\beta(\epsilon - \mu)}}
\;\Longrightarrow\;
\bar n_{\text{BE}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} - 1} ,
\]
ويشترط $\mu < \epsilon_{\min}$. ويؤول كلاهما إلى عامل
بولتزمان $\eu^{-\beta(\epsilon - \mu)}$ حين تندر الإشغالات.
وهاتان الصيغتان، ومأخذ كلٍّ منهما سطران، هما كامل
مدخلات الإحصاء الكمومي: والفصل التالي حصادهما.
\end{theorem}

\begin{proof}
الفرميونات: حدّان. والبوزونات: متتالية هندسية، لا تتقارب إلا
من أجل $\mu < \epsilon$. وفي الحالتين يعطي $\bar n = k_{\text{B}}T\,
\partial\ln\Xi/\partial\mu$ الصورتين المذكورتين.
\end{proof}

\section{الامتزاز: مواقع للإيجار}

\begin{proposition}[متساوية حرارة لانغموير]\label{prop:b3:grand-canonical:langmuir}
\index{متساوية حرارة لانغموير}\index{الامتزاز}
يعرض سطح مواقع مستقلة، كل موقع إما خالٍ وإما يحمل جزيء
غاز واحدًا بطاقة ارتباط $-\epsilon_0$؛ والغاز فوقه، عند
الضغط $P$، هو الخزان. فكل موقع جملة كبيرة ذات حالتين،
وإشغاله هو
\[
\theta = \frac{1}{\eu^{\beta(-\epsilon_0 - \mu)} + 1}
 = \frac{P}{P + P_0(T)} , \qquad
P_0(T) = \frac{k_{\text{B}}T}{\lambda_T^3}\,
\eu^{-\epsilon_0/k_{\text{B}}T} ,
\]
باستعمال $\mu$ الغاز المثالي. وترتفع التغطية خطيًّا عند الضغط
المنخفض وتتشبّع عند طبقة أحادية واحدة --- وهي \emph{\omterm{prop:b3:grand-canonical:langmuir}{متساوية حرارة
لانغموير}}، أي منحني عمل المحوّلات الحفّازة والفحم
المنشَّط والكمامات وارتباط الأكسجين بالدم
(\cref{exo:b3:grand-canonical:6}). ويهبط ضغط العبور $P_0$
هبوطًا \emph{أُسّيًا} مع طاقة الارتباط ويرتفع ارتفاعًا حادًّا
مع درجة الحرارة: فتسخين سطح يهزّ ضيوفه فيفلتون ---
ولذلك يكون خبز الزجاجيات تحت الخلاء هو تنظيفها حقًّا.
\end{proposition}

\begin{proof}
إشغال الموقع هو
\cref{thm:b3:grand-canonical:occupation} (بالصورة الفرميونية: فالموقع
يحمل واحدًا على الأكثر) مع $\epsilon = -\epsilon_0$؛ ثم عوّض
$\eu^{\beta\mu} = n\lambda_T^3 = P\lambda_T^3/k_{\text{B}}T$.
\end{proof}

\section{توازنات التأيّن: معادلة ساها}

\begin{proposition}[معادلة ساها]\label{prop:b3:grand-canonical:saha}
\index{معادلة ساها}
في غاز ساخن يتوازن التفاعل $\text{H} \rightleftharpoons \text{p} +
\text{e}$ حين $\mu_{\text{H}} = \mu_{\text{p}} +
\mu_{\text{e}}$. وكتابة كل $\mu$ بصورة الغاز المثالي (مع
حساب ارتباط الذرة $-E_{\text{I}}$) تعطي
\[
\frac{n_{\text{e}}\,n_{\text{p}}}{n_{\text{H}}}
 = \frac{1}{\lambda_{T,\text{e}}^3}\,
\eu^{-E_{\text{I}}/k_{\text{B}}T}
\qquad\Big(\lambda_{T,\text{e}} =
h/\sqrt{2\pi m_{\text{e}}k_{\text{B}}T}\Big)
\]
(في حدود عوامل انحلال سبيني من رتبة الواحد). وعملتان
متنافستان من جديد: فعامل بولتزمان يفرض على التأيّن ضريبة
$E_{\text{I}} = \qty{13.6}{eV}$، بينما يدعمه المعامل الأمامي --- أي
\emph{إنتروبيا الإلكترون المحرَّر}، بما لها من حالات متاحة
تعادل $\lambda_T^{-3}$. والدعم كبير إلى حدّ أن الهدروجين، عند
كثافات النجوم، يتأيّن تأيّنًا ملحوظًا
قرب \qty{10000}{K} أصلًا، حيث لا يزيد $k_{\text{B}}T$ على
\qty{0.9}{eV}: فالتوازنات يُتنازع فيها على \omterm{prop:b3:canonical-ensemble:machine}{الطاقة الحرة} لا على الطاقة
--- وأطياف النجوم هي الدفتر
(\cref{pb:b3:grand-canonical:1}).
\end{proposition}

\begin{proof}[برهان جزئي]
ضع $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$ مع $\mu_i
= k_{\text{B}}T\ln(n_i\lambda_{T,i}^3) + \varepsilon_i$
(مع $\varepsilon_{\text{H}} = -E_{\text{I}}$، والباقي أصفار)؛
ثم خذ الأُسّ. وأطوال الموجة الحرارية للبروتون وللذرة تكاد
تتلاشى (لتساوي الكتلتين)؛ وعوامل الانحلال مقبولة.
\end{proof}

\begin{method}[الحرفة القانونية الكبرى]\label{met:b3:grand-canonical:method}
(1) كل ما يتبادل الجسيمات مع شريك كبير: أعطِ
الشريك $\mu$ وزِن الحالات بالعامل $\eu^{-\beta(E - \mu N)}$.
(2) والمواقع أو المستويات المستقلة تتحلل إلى عوامل: فحلّ \emph{واحدًا}
واضربه. (3) ومن أجل خزان غاز كلاسيكي، يحوّل $\eu^{\beta\mu} =
n\lambda_T^3$ الكمونَ $\mu$ إلى ضغط قابل للقياس. (4) وأما التفاعلات
وتغيرات الطور: فوازن قيم $\mu$، واحدةً لكل نوع، بما فيها طاقات
الارتباط --- وتوقّع من المعاملات الإنتروبية الأمامية أن تزيح
التوازنات بعيدًا عن تخمينات بولتزمان الساذجة. (5) وأما الإلكترونات في
الأجسام الصلبة: فالكمون $\mu$ لها هو مستوي فيرمي، وفروق $\mu$
جهود --- وهو الجسر إلى
\cref{ch:b3:electrons-in-solids}.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:1}
(أ) علبتان من الغاز نفسه عند $T$ نفسها لكن بكثافتين
مختلفتين تُوصلان: فبأيّ اتجاه تسيل الجسيمات، بلغة
$\mu$؟ (ب) بيّن من $\mu$ الغاز المثالي أن السيلان يتوقف
عند تساوي الكثافتين. (ج) الماء تحت هواء رطب: اذكر
شرط التوازن بقيم $\mu$. (د) ولماذا يجفّ الغسيل حتى
تحت $100\,^\circ$C حين يكون الهواء جافًّا (قارن قيم $\mu$، لا
درجات الحرارة)؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:2}
\omterm{thm:b3:grand-canonical:distribution}{الكمون الكيميائي} للهواء. (أ) احسب $\lambda_T$ من أجل N$_2$
عند \qty{300}{K}. (ب) احسب $n\lambda_T^3$ عند \qty{1}{bar} و
$\mu$ الناتجة بوحدة eV. (ج) ولماذا لا تكون $\mu < 0$ مفارقة
(فأيّ حدّ من $F = E - TS$ يغلب عند إضافة جزيء)؟ (د)
عند أيّ كثافة تبلغ $\mu$ الصفر، وبمَ ينذر
$n\lambda_T^3 \sim 1$
(\cref{exo:b3:grand-canonical:11})؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:3}
استنتج إشغال فيرمي--ديراك من $\Xi$ ذات الحدّين،
خطوةً خطوة؛ وتحقق من النهايتين $\epsilon \ll \mu$ و $\epsilon \gg
\mu$، ومن القيمة عند $\epsilon = \mu$. وأين ظهرت هذه الدالة
من قبل في هذا الكتاب
(\cref{exo:b3:microcanonical-ensemble:8})؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:4}
الفولتات كيمياء. (أ) خلية ليثيوم تحمل \qty{3.0}{V}:
عبّر عن فرق \omterm{thm:b3:grand-canonical:distribution}{الكمون الكيميائي} لكل إلكترون بوحدة eV و
لكل مول بوحدة kJ. (ب) ولماذا تكون كل بطاريات الحياة اليومية ذات
فولتات من رقم واحد (فما الذي يضع مقياس فروق $\mu$ الكيميائية)؟ (ج)
بطارية السيارة الرصاصية--الحمضية تكدّس ست خلايا: فلمَ التكديس بدل
إيجاد تفاعل عند \qty{12}{eV}؟ (د) وبلغة $\mu$، ماذا تفعل
البطارية حين ``تفرغ''؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:5}
البوزونات عند مستوٍ. (أ) اجمع المتتالية الهندسية من أجل $\Xi$ و
استنتج $\bar n_{\text{BE}}$. (ب) ولماذا يجب أن تبقى $\mu$ تحت
أدنى مستوٍ --- فما الذي يتباعد لولا ذلك، وبمَ ينبئ
التباعد (\cref{ch:b3:quantum-statistics})؟ (ج)
احسب $\bar n$ من أجل $(\epsilon - \mu)/k_{\text{B}}T = 0.1$ و $1$
و $3$ وقارن ببولتزمان. (د) للفوتونات $\mu = 0$: اذكر
السبب الفيزيائي (فعددها غير محفوظ --- إذ تمتصّ الجدران
وتصدر بحرية).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:6}
لانغموير في العمل. (أ) من متساوية الحرارة، أيّ ضغط يحقق
$\theta = 0.5$؟ و $0.9$؟ (ب) يرتبط الميوغلوبين بالأكسجين O$_2$ على موقع واحد:
فيتبع الإشباع لانغموير بدلالة ضغط الأكسجين الجزئي ---
فلماذا يُشاهَد منحنٍ \emph{زائدي} (لا سيني) من أجل
الميوغلوبين ومنحنى S تعاوني من أجل الهيموغلوبين ذي المواقع الأربعة؟
(ج) يرتبط أول أكسيد الكربون بالمواقع نفسها بطاقة $\epsilon_0$
أكبر بمقدار $\approx\qty{0.13}{eV}$: فعند ضغطين جزئيين متساويين و
\qty{310}{K}، احسب نسبة الإشغال CO:O$_2$ واشرح تسمّم
أول أكسيد الكربون. (د) ولماذا ينفع العلاج بالأكسجين المضغوط، بلغة
متساوية الحرارة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:7}
تذبذبات العدد. (أ) بيّن أن $\operatorname{Var}N =
k_{\text{B}}T\,\partial\langle N\rangle/\partial\mu$. (ب) ومن أجل
الغاز المثالي الكلاسيكي، استنتج $\operatorname{Var}N = \langle
N\rangle$: أي بواسون. (ج) وفّق ذلك مع تذبذبات الكثافة في
\cref{exo:b3:microcanonical-ensemble:12}. (د) ومن أجل مستوٍ
فرميوني مفرد، احسب $\operatorname{Var}n = \bar n(1 - \bar
n)$: فأين تكون أعداد الفرميونات أهدأ، ولماذا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:8}
التناضح من $\mu$. غشاء يمرّر الماء لا المذاب
(بتركيز $c$ لكل حجم، مخلخل). (أ) بمساواة $\mu$ الماء
عبر الغشاء، بيّن أن جانب الماء النقي يجب أن يكون
مضغوطًا زيادةً بالمقدار $\Pi = ck_{\text{B}}T$ (فانت هوف) --- واقبل
أن المذاب المنحلّ يخفض $\mu$ الماء بالمقدار
$-k_{\text{B}}T\,c/n_{\text{w}}$ لكل جزيء ماء. (ب) احسب
$\Pi$ من أجل محلول ملحي فيزيولوجي ($c \approx \qty{3e26}{m^{-3}}$
جسيمًا كليًّا). (ج) الكريات الحمر في ماء نقي: فبأيّ اتجاه يسيل
الماء، وبأيّ نتيجة؟ (د) ولماذا لا تحتاج الأشجار إلى مضخات
ولا معجزات لرفع النسغ عشرات الأمتار (تدرّجات $\mu$ التناضحية
والشعرية)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:9}
التوازن في مجال. في الثقالة، يتطلب توازن عمود غاز
أن يكون \omterm{thm:b3:grand-canonical:distribution}{الكمون الكيميائي} \emph{الكلي} $\mu(h) =
k_{\text{B}}T\ln(n\lambda_T^3) + mgh$ منتظمًا. (أ) استنتج
القانون البارومتري من هذا في سطر واحد. (ب) عمّم على
نابذة بسرعة زاوية $\omega$ (وكمونها
$-\tfrac12 m\omega^2r^2$) واحصل على التوزّع القطري. (ج)
تدور نابذات تخصيب اليورانيوم عند $\omega r \sim
\qty{600}{m/s}$: احسب نسبة التوازن لتعزيز $^{238}$UF$_6$
على $^{235}$UF$_6$ في كل مرحلة، $\exp[\Delta m\,\omega^2
r^2/2k_{\text{B}}T]$ مع $\Delta m = \qty{5e-27}{kg}$ و $T =
\qty{320}{K}$. (د) ولماذا تُسلسَل مراحل كثيرة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:10}
ساها، باليد. (أ) نفّذ برهان موازنة $\mu$ في
\cref{prop:b3:grand-canonical:saha}. (ب) عرّف النسبة
المتأيّنة $x$ عند كثافة هدروجين كلية $n$ وبيّن أن $x^2/(1 - x)
= \eu^{-E_{\text{I}}/k_{\text{B}}T}/n\lambda_{T,\text{e}}^3$. (ج)
الغلاف الضوئي الشمسي: $T = \qty{5800}{K}$ و $n \approx
\qty{e23}{m^{-3}}$: بيّن أن النسبة المتأيّنة من الهدروجين لا تتجاوز
$\sim\num{e-4}$ --- فسطح الشمس متعادل. (د) وعند $T =
\qty{10000}{K}$ (بالكثافة $n$ نفسها): أعد الحساب وعلّق على
الحدّة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:11}
حين تصير الغازات كمومية. يبدأ الانحلال عند $n\lambda_T^3 \sim
1$. (أ) من أجل الهواء عند \qty{300}{K}: كم يبعد عن العتبة
(\cref{exo:b3:grand-canonical:2})؟ (ب) ومن أجل إلكترونات الناقلية
في النحاس ($n = \qty{8.5e28}{m^{-3}}$): جد درجة الحرارة التي
تحتها $n\lambda_T^3 > 1$ --- واستنتج أن إلكترونات الفلز
منحلّة عند \emph{أيّ} درجة حرارة مخبرية.
(ج) ومن أجل ذرات الروبيديوم عند $n = \qty{e20}{m^{-3}}$: احسب درجة
حرارة العتبة الكمومية (وهي مقياس تكاثف بوز--أينشتاين في
\cref{ch:b3:quantum-statistics}). (د) رتّب الثلاثة واذكر
القاعدة: الجسيمات الخفيفة والكثافات العالية تصير كمومية أولًا.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:12}
الكون يفقد بوزتروناته. في الكون الفتيّ الساخن، أبقى التفاعل
$\eu^+ + \eu^- \rightleftharpoons 2\gamma$ الأزواجَ في
توازن مع $\mu_{e^+} + \mu_{e^-} = 0$ (وللفوتونات $\mu = 0$).
(أ) ومن أجل بلازما متناظرة يعطي هذا $\mu_{e^\pm} \approx 0$:
بيّن أن كثافة الأزواج تكون عندئذٍ $n_\pm \approx
\lambda_T^{-3}\eu^{-m_{\text{e}}c^2/k_{\text{B}}T}$ متى صار
$k_{\text{B}}T \ll m_{\text{e}}c^2$. (ب) احسب درجة الحرارة
التي عندها $m_{\text{e}}c^2 = k_{\text{B}}T$ والمقياس الزمني
الكوني الموافق لها (واقبل $t \sim \qty{1}{s}$ عند
$\qty{e10}{K}$). (ج) بيّن أن تبريدًا إضافيًّا متواضعًا (بعامل
قليل) يهوي بكثافة الأزواج مراتب عديدة:
فتفنى البوزترونات، ولا يبقى إلا الفائض البدئي الضئيل
من الإلكترونات. (د) والفوتونات المحرَّرة تعيد تسخين حمّام
الإشعاع: فأيّ إشعاع أثريّ يحمل الوصل
(\cref{ch:b3:astrophysics})؟
\end{exercise}

\section{مسألة: قراءة درجات حرارة النجوم}

\begin{problem}[{مسألة نهاية الأسبوع --- ساها وباين، وممّ
صُنع الكون}]\label{pb:b3:grand-canonical:1}
فُهرست أطياف النجوم عبر تسعينيات القرن التاسع عشر في أصناف
O و B و A و F و G و K و M بحسب شدّة خطوط امتصاصها ---
وكانت خطوط بالمر للهدروجين أشدّ ما تكون في الصنف A وضعيفة على نحو
غامض في النجوم الأسخن والأبرد معًا. وفي سنة 1925 طبّقت سيسيليا باين
\omterm{prop:b3:grand-canonical:saha}{معادلة ساها} الوليدة وبيّنت أن الحديقة كلها
مادة واحدة عند درجات حرارة مختلفة --- وأن النجوم
هدروجين في أغلبها الساحق، وهي نتيجة من الهرطقة بحيث حُملت على
تليينها في أطروحتها. وتعيد هذه المسألة تتبّع استدلالها.
والمعطيات: امتصاص بالمر يبدأ من $n = 2$، عند $E_2 - E_1 =
\qty{10.2}{eV}$ فوق الحالة الأساسية؛ و $E_{\text{I}} =
\qty{13.6}{eV}$؛ وكثافات إلكترونات الغلاف الضوئي $n_{\text{e}}
\sim \qty{e20}{m^{-3}}$.

\medskip\noindent\textbf{الجزء الأول --- عاملان متنافسان.}
\begin{enumerate}
  \item امتصاص بالمر يحتاج إلى هدروجين متعادل \emph{في} $n =
        2$: اكتب عامل بولتزمان لجمهرة $n = 2$
        (بانحلالات $g_2/g_1 = 4$).
  \item احسبه عند $5000$ و $10000$ و \qty{20000}{K}: فبأيّ
        اتجاه يدفع هذا العامل شدّةَ الخط مع ارتفاع $T$؟
  \item والآن عامل ساها: مع ارتفاع $T$، ماذا يحدث للنسبة
        \emph{المتعادلة}، ومن ثمّ لخطوط بالمر؟
  \item اشرح نوعيًّا لماذا يجب أن يبلغ الجداء --- أي المثار
        \emph{والمتعادل بعد} --- ذروته عند درجة حرارة
        وسطى.
  \item باستعمال \omterm{prop:b3:grand-canonical:saha}{معادلة ساها} مع $n_{\text{e}} =
        \qty{e20}{m^{-3}}$، قدّر درجة حرارة
        نصف تأيّن الهدروجين.
  \item اجمع الذروة: فقرب أيّ درجة حرارة ينبغي أن تكون
        خطوط بالمر أشدّ، وأيّ صنف طيفي
        (نجوم A، $\sim\qty{10000}{K}$) يختار ذلك؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- المتتالية الطيفية
مفكوكة.}
\begin{enumerate}[resume]
  \item نجوم M (\qty{3000}{K}) تُظهر عصابات جزيئية قوية (TiO)
        وهدروجينًا ضعيفًا: اشرح الأمرين بالعاملين.
  \item ونجوم O (\qty{40000}{K}) تُظهر خطوط الهليوم وهدروجينًا
        ضعيفًا من جديد: اشرح (فما الذي أصاب الهدروجين، ولماذا
        يأتي دور الهليوم الآن، وله $E_{\text{I}} = \qty{24.6}{eV}$؟).
  \item والشمس (G، \qty{5800}{K}) تُظهر خطوطًا قوية
        \emph{للكالسيوم المتأيّن} ($E_{\text{I,Ca}} = \qty{6.1}{eV}$)
        وهدروجينًا ضعيفًا مع ذلك: بيّن بمنطق ساها لماذا يكون الكالسيوم
        متأيّنًا سلفًا بينما يبقى الهدروجين متعادلًا وأكثره
        في $n = 1$.
  \item كان فلكيّو ما قبل باين يقرؤون شدّة الخط وفرةً:
        فبمَ جعلتهم خطوط الشمس الكالسيومية الجبارة وخطوط بالمر
        الضعيفة يستنتجون في تركيبها؟
  \item وبصيرة باين: صحّح شدّة كل خط وفق ``عامل الرؤية''
        البولتزماني--الساهوي الخاص به. فحين فعلت، أيّ
        تراتب من الوفرات الحقيقية ظهر؟
  \item ونتيجتها --- أن الهدروجين أوفر من الكالسيوم بمليون
        مرة، وأن النجوم أكثرها هدروجين وهليوم --- لماذا
        قوبلت بالمقاومة (فبمَ كانت توحي الكيمياء المتمركزة
        حول الأرض)، ومتى أُنصفت؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- أعداد على الذروة.}
\begin{enumerate}[resume]
  \item احسب عامل بولتزمان للمستوي $n = 2$ عند
        \qty{9600}{K} بالضبط.
  \item ومع تقدير ساها في البند 5، خذ النسبة المتعادلة
        عند $9600$ و $15000$ و \qty{25000}{K} لتكون
        نحو $0.5$ و $\num{e-3}$ و $\num{e-5}$: كوّن الجداء
        (النسبة المثارة $\times$ النسبة المتعادلة) عند درجات
        الحرارة الثلاث وحدّد أعظميته.
  \item ولماذا تكون ذروة بالمر \emph{حادّة} من جهة الحرارة العالية
        (فأيّ أُسّي يغلب) و\emph{متدرجة} من الجهة
        الباردة؟
  \item للغلاف الضوئي لقزم أبيض $n_{\text{e}}$ أكبر بألف
        مرة: فبأيّ اتجاه تتحرك درجة حرارة نصف
        التأيّن، ولماذا (لوشاتلييه بلغة $\mu$)؟
  \item ونجمان بدرجة حرارة واحدة لكن بضغطين مختلفين
        يُظهران من ثمّ شدّتَي خطّ مختلفتين: فأيّ خاصة
        نجمية (عملاق أم قزم --- أي ثقالة السطح)
        يتيح ذلك لعلماء الطيف قراءتها؟
  \item تثبّت المسوح الحديثة درجات حرارة النجوم في حدود $\pm1\%$ من
        نسب الخطوط: اذكر السلسلة: مسلَّمة $\to$ بولتزمان
        $\to$ ساها $\to$ ميزان حرارة.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- ما الذي حُسم.}
\begin{enumerate}[resume]
  \item كانت متتالية OBAFGKM فوضى أبجدية أُعيد
        ترتيبها تجريبيًّا: فبعد ساها، أيّ متغير فيزيائي
        واحد يرتّبها؟
  \item اذكر لماذا يستطيع العنصر \emph{نفسه} أن يهيمن على
        طيف عند درجة حرارة وأن يختفي عند أخرى من دون
        أيّ تغير في الوفرة --- وهو الدرس الجوهري الذي علّمته باين
        علمَ الفلك.
  \item الهدروجين عند ثلاثة أرباع كل الكتلة الباريونية: سمِّ
        ركيزتين أخريين من ركائز هذا الكتاب تقومان على تلك الوفرة
        (الاندماج النجمي، \cref{ch:b3:astrophysics}؛ وسماء
        21 سنتيمترًا، \cref{ch:b3:spin-two-level}).
  \item وصف أوتو ستروفه أطروحة باين بأنها ``ألمع
        أطروحة دكتوراه كُتبت في علم الفلك'': فمن هذه
        المسألة، برّر الثناء في جملة واحدة.
  \item وفيزياء ساها نفسها، مُجراةً عكسيًّا على المقياس الكوني،
        تتنبأ بدرجة الحرارة التي عندها اتّحد هدروجين الكون
        كله أول مرة ($\sim\qty{3000}{K}$ عند الكثافة الكونية
        الأدنى بكثير): سمِّ الحدث وأثره
        (\cref{ch:b3:astrophysics}).
  \item وتوازنات التأيّن تدير أيضًا مصابيح الفلورسنت و
        معالجة البلازما: اذكر في جملة واحدة لماذا تستطيع
        حالة مؤيَّنة على طراز \qty{10000}{K} أن توجد في أنبوب
        يبقى زجاجه باردًا (فأيّ جملتين جزئيتين تخفقان في التوازن؟).
  \item لخّص النتيجة الموسومة: شرط توازن واحد،
        $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$، يحوّل
        الأبجدية الطيفية إلى ميزان حرارة يجري من
        $3000$ إلى \qty{40000}{K}، ويضع ذروة بالمر عند نجوم
        A، ويكشف كونًا مصنوعًا من الهدروجين.
\end{enumerate}
\end{problem}
