\chapter{ميكانيك الأوساط المتصلة والمرونة}\label{ch:b3:continuum-elasticity}

ألصق أذنك بسكة فولاذية طويلة بينما يطرقها عامل بعيد: تسمع
رنّتين --- واحدة عبر الفولاذ وأخرى عبر الهواء، بينهما ثانية أو
أكثر. فالصخر والفولاذ والعظم والمطاط تحمل القوى والأمواج كما
يحمل المائع الضغط، لكن معها ما ليس لأيّ مائع: إذ تقاوم
\emph{تغيرات الشكل}، لا تغيرات الحجم وحدها. وقد بنى كتاب السنة
الثانية ميكانيك الموائع على فكرتين، هما الجسيم المادي ومجال
الضغط؛ وهذا الفصل يمدّهما إلى الأجسام الصلبة. فيصير انتقال كل
جسيم مجالًا، ويصير تشويهه الموضعي \emph{تشوّهًا}، وتصير القوى
الداخلية \emph{إجهادًا}، وبينهما تقف بطاقة هوية الجسم الصلب، أي
\omterm{thm:b3:continuum-elasticity:hooke}{قانون هوك} بثابتَي مرونته. والمردود يمتدّ من اليومي --- لماذا
تستطيل الكابلات، ولماذا تنفجر القوارير، وكيف ينحني لوح الغطس ---
إلى الكوكبي: فالزلازل ترسل عبر الأرض صنفَي الموجة المرنة اللذين
يتنبأ بهما هذا الفصل بالضبط، وكانت أزمنة وصولهما أول إصغاء
لباطن كوكبنا.

\section{التشوّه: وصف التغير في الشكل}

\begin{definition}[مجال الانتقال وموتّر التشوّه]\label{def:b3:continuum-elasticity:strain}
\index{مجال الانتقال}\index{موتّر التشوّه}\index{التمدد الحجمي}
حين يتشوّه جسم صلب، ينتقل الجسيم الذي كان في $\vect r$ إلى
$\vect r + \vect u(\vect r)$: فالمجال $\vect u$ هو \emph{مجال
الانتقال}. ومن أجل التشوّهات الصغيرة يُقاس التشويه الموضعي
بواسطة \emph{موتّر التشوّه}، وهو الجدول المتناظر
\[
\varepsilon_{ij} = \frac12\Big(
 \frac{\partial u_i}{\partial x_j} + \frac{\partial u_j}{\partial x_i}
\Big) , \qquad i, j \in \{x, y, z\} .
\]
والمركبة القطرية $\varepsilon_{xx}$ هي الاستطالة النسبية للمادة
على المنحى $x$ (وهي عديمة الأبعاد، ومقدارها مثلًا $\num{e-3}$ في
كابل فولاذي في الخدمة)؛ أما المركبة غير القطرية $\varepsilon_{xy}$
فهي نصف انغلاق الزاوية بين المنحيين الماديين $x$ و $y$
--- أي \emph{قصّ}. وأثر الموتّر هو التغير النسبي في الحجم، أي
\emph{التمدد الحجمي}:
\[
\frac{\delta V}{V} = \varepsilon_{xx} + \varepsilon_{yy} +
\varepsilon_{zz} = \operatorname{div}\vect u .
\]
\end{definition}

\begin{example}[ثلاثة تشوّهات أولية]\label{ex:b3:continuum-elasticity:elementary}
تمدد منتظم $\vect u = \alpha\vect r$: فيكون $\varepsilon_{ij} =
\alpha\,\delta_{ij}$، والتغير الحجمي $3\alpha$، ولا قصّ. وقصّ بسيط
$\vect u = (\gamma y, 0, 0)$: فيكون $\varepsilon_{xy} = \gamma/2$
وتنعدم المركبات القطرية كلها --- فيتغير الشكل دون الحجم. ودوران
صلب $\vect u = (-\omega y, \omega x, 0)$: فيكون $\varepsilon_{ij} = 0$
تطابقيًا --- إذ إن الجزء المتخالف التناظر من $\partial u_i/\partial x_j$،
الذي تطرحه المناظرة، هو بالضبط الجزء الذي يدير من دون أن يشوّه.
فالتشوّه لا يقيس إلا التشوّه.
\end{example}

\section{الإجهاد: وصف القوى الداخلية}

\begin{definition}[الشدّ وموتّر الإجهاد]\label{def:b3:continuum-elasticity:stress}
\index{موتّر الإجهاد}\index{شعاع الشدّ}
اقطع الجسم الصلب، بالفكر، على سطح صغير $\dd S$ ناظمه
$\vect n$: تجذب المادة الواقعة في جهة $+\vect n$ الجهة الأخرى
بقوة $\dd\vect F = \boldsymbol\sigma(\vect n)\,\dd S$ --- وهي
\emph{شعاع الشدّ}. وتتعلق هذه القوة خطيًا بالناظم $\vect n$ (كوشي)،
فيرمّزها \emph{موتّر الإجهاد} $\sigma_{ij}$:
\[
\dd F_i = \sum_j \sigma_{ij}\,n_j\,\dd S ,
\]
مقيسًا بالباسكال. فالمركبة $\sigma_{xx}$ جذب ($> 0$: شدّ) أو دفع
($< 0$: ضغط) عبر وجه ناظمه $x$؛ والمركبة $\sigma_{xy}$ قوة على
المنحى $x$ يحملها وجه ناظمه $y$ --- أي إجهاد قصّ. والمائع
الساكن هو الحالة الخاصة $\sigma_{ij} = -p\,\delta_{ij}$: فالضغط
يدفع كل وجه بالمقدار نفسه ولا يقصّ أيّ وجه، ولهذا تسيل الموائع
--- إذ لا تستطيع حمل قصّ ساكن.
\end{definition}

\begin{proposition}[التوازن والتناظر]\label{prop:b3:continuum-elasticity:equilibrium}
في جسم صلب متوازن تحت قوة حجمية $\vect f$ في وحدة
الحجم (مثل $\rho\vect g$)،
\[
\sum_j\frac{\partial\sigma_{ij}}{\partial x_j} + f_i = 0
\]
في كل نقطة داخلية؛ ويكون \omterm{def:b3:continuum-elasticity:stress}{موتّر الإجهاد} متناظرًا، أي
$\sigma_{ij} = \sigma_{ji}$.
\end{proposition}

\begin{proof}[برهان جزئي]
نوازن القوى على مكعب صغير ضلعه $a$: فتختلف قوى الشدّ على
الوجهين الناظميين للمنحى $x$ بالمقدار $a\,\partial_x\sigma_{ix}$ في
وحدة المساحة، وكذلك من أجل الزوجين الآخرين؛ فتكون القوة السطحية
المحصلة في وحدة الحجم $\sum_j\partial_j\sigma_{ij}$، ويجب أن
تقاصّ $f_i$ --- وهي المحاسبة نفسها التي أعطت
$-\vect\nabla p + \rho\vect g = \vect 0$ في سكون الموائع في كتاب
السنة الأولى. وأما التناظر: فعزم إجهادات القصّ حول مركز المكعب
هو $(\sigma_{xy} -
\sigma_{yx})a^3$ في المرتبة الغالبة، بينما
يتغير عزم عطالته وفق $a^5$؛ فإجهاد غير متناظر يجعل المكعبات
الصغيرة تدور بسرعة لانهائية. أما الحجة المتصلة الكاملة فمُسلَّم
بها.
\end{proof}

\section{قانون هوك وثوابت المرونة}

\begin{theorem}[المرونة الخطية المتماثلة المناحي]\label{thm:b3:continuum-elasticity:hooke}
\index{قانون هوك}\index{معامل يونغ}\index{نسبة بواسون}
من أجل التشوّهات الصغيرة، يستجيب جسم صلب متماثل المناحي استجابة
خطية: فالإجهاد يتناسب مع التشوّه،
\[
\sigma_{ij} = \lambda\,(\varepsilon_{xx} + \varepsilon_{yy} +
\varepsilon_{zz})\,\delta_{ij} + 2\mu\,\varepsilon_{ij} ,
\]
بثابتين ماديين هما \emph{معاملا لامي} $\lambda$ و
$\mu$ (ويُكتب $\mu$ أيضًا $G$، وهو \emph{معامل القصّ}). وفي
تجربة الشدّ البسيط --- قضيب يُجذب على المنحى $x$ وجانباه حرّان ---
يأخذ القانون نفسه صيغة المهندس
\[
\varepsilon_{xx} = \frac{\sigma_{xx}}{E} , \qquad
\varepsilon_{yy} = \varepsilon_{zz} = -\nu\,\varepsilon_{xx} ,
\]
فتُعرَّف \emph{\omterm{thm:b3:continuum-elasticity:hooke}{معامل يونغ}} $E$ (الصلابة أمام الاستطالة) و
\emph{\omterm{thm:b3:continuum-elasticity:hooke}{نسبة بواسون}} $\nu$ (التقلص الجانبي)، مع
\[
\mu = \frac{E}{2(1 + \nu)} , \qquad
\lambda = \frac{E\nu}{(1 + \nu)(1 - 2\nu)} .
\]
وهذا هو قول هوك ``كما الاستطالة تكون القوة'' (1678)، مرقّى إلى
موتّر. وهو صالح حتى \emph{حدّ مرونة} يتوقف على المادة.
\end{theorem}
\admitted

\begin{remark}[رتب المقادير]\label{rem:b3:continuum-elasticity:orders}
$E \approx \qty{200}{GPa}$ في الفولاذ، و\qty{70}{GPa} في الألمنيوم
وزجاج النوافذ، و\qty{50}{GPa} في الغرانيت، و\qty{15}{GPa} في العظم
والخرسانة، و\qty{10}{GPa} في الخشب على منحى الليف، وبضعة
ميغاباسكالات فقط في المطاط --- أي خمس رتب من المقدار، وهي المسافة
بين جسر ورباط مطاطي. وتقع $\nu$ بين $0$ (الفلين، تقريبًا) و $1/2$
(المطاط): فالقيمة $\nu = 1/2$ تعني تشوّهًا يحفظ الحجم، والقيمة
$\nu \approx 0.3$ نموذجية في الفلزات. وتشوّه قدره $\num{e-3}$ في
الفولاذ يعني حقًّا $\sigma = \qty{200}{MPa}$، وهو قريب من إجهاد
الخضوع في الأصناف العادية: فالمرونة اليومية تعيش دون جزء من ألف.
\end{remark}

\begin{example}[كابل مصعد]\label{ex:b3:continuum-elasticity:cable}
كابل فولاذي طوله \qty{60}{m} ومقطعه \qty{2.0}{cm^2} يحمل مقصورة
كتلتها \qty{1000}{kg}: فيكون $\sigma = mg/S = \qty{49}{MPa}$ و $\varepsilon =
\sigma/E = \num{2.5e-4}$، والاستطالة $\varepsilon L = \qty{15}{mm}$ ---
ويسلك الكابل سلوك نابض صلابته $k = ES/L =
\qty{6.7e5}{N/m}$: والصيغة $k = ES/L$ هي الطريقة التي يعيد بها
الوسط المتصل نوابض كتاب السنة الأولى.
\end{example}

\begin{proposition}[الطاقة المرنة]\label{prop:b3:continuum-elasticity:energy}
\index{طاقة مرنة}
يخزّن جسم صلب مشوَّه، في وحدة الحجم، كثافة الطاقة
\[
w = \frac12\sum_{i,j}\sigma_{ij}\,\varepsilon_{ij}
\qquad\text{(شدّ بسيط: } w = \tfrac12 E\varepsilon^2 =
\sigma^2/2E\text{)} ,
\]
وهي المقابل الثلاثي البعد للمقدار $\tfrac12 kx^2$.
\end{proposition}

\begin{proof}[برهان جزئي]
في الشدّ البسيط، يؤدي رفع الإجهاد من $0$ إلى $\sigma$ في وحدة
الحجم الشغل $\int_0^\varepsilon\sigma'\,\dd\varepsilon' =
\int_0^\varepsilon E\varepsilon'\,\dd\varepsilon' = \tfrac12
E\varepsilon^2$ --- وهي المساحة تحت خط الإجهاد--التشوّه، تمامًا
كما في نابض. أما الشكل التربيعي العام فينتج بتراكب المركبات؛
وهو مُسلَّم به.
\end{proof}

\begin{method}[حل مسائل المرونة الصغيرة]\label{met:b3:continuum-elasticity:method}
(1) نعيّن الهندسة والتحميل؛ ونخمّن أيّ مركبات الإجهاد غير
معدومة (قضيب مجذوب: $\sigma_{xx}$ وحدها؛ سلك مفتول: قصّ؛ غلاف
مضغوط: شدّ مماسّي). (2) نكتب موازنة القوى على قطعة مختارة جيدًا
--- نصف أسطوانة أو شريحة أو مكعب. (3) نحوّل الإجهاد إلى تشوّه
\omterm{thm:b3:continuum-elasticity:hooke}{بقانون هوك}، والتشوّه إلى انتقال بالمكاملة. (4) نحسب الطاقات
بالمقدار $w = \sigma^2/2E$ (أو $\sigma^2/2G$ في القصّ). (5) نتحقق
دائمًا من بقاء التشوّه صغيرًا والإجهاد دون حدّ المرونة ---
وإلا كان الجواب يصف جسمًا صلبًا لم يعد موجودًا.
\end{method}

\section{الأمواج المرنة}

\begin{proposition}[صوتا الجسم الصلب]\label{prop:b3:continuum-elasticity:waves}
\index{أمواج مرنة}\index{الموجة P}\index{الموجة S}
في جسم صلب مرن غير محدود كثافته $\rho$، تنتشر الاضطرابات الصغيرة
على شكل صنفين مستقلين من الأمواج: أمواج \emph{طولية} (P) تهتز
فيها المادة على منحى الانتشار بالانضغاط والتمدد، وسرعتها
\[
c_{\text{P}} = \sqrt{\frac{\lambda + 2\mu}{\rho}} ,
\]
وأمواج \emph{مستعرضة} (S) تنقصّ فيها المادة جانبيًا، وسرعتها
\[
c_{\text{S}} = \sqrt{\frac{\mu}{\rho}} < c_{\text{P}} .
\]
وفي المائع $\mu = 0$: فلا أمواج S --- إذ لا يمكن نقل القصّ ---
وترجع سرعة P إلى سرعة الصوت في كتاب السنة الثانية. ومن أجل
$\nu = 1/4$ (وهي قيمة نموذجية للصخر) يكون $\lambda = \mu$ و $c_{\text{P}} =
\sqrt3\,c_{\text{S}}$.
\end{proposition}

\begin{proof}[برهان جزئي]
لنأخذ اضطرابًا مستويًا $\vect u = \vect u(x, t)$. قانون نيوتن في
وحدة الحجم هو $\rho\,\partial_t^2u_i = \sum_j\partial_j\sigma_{ij}$
(أي معادلة التوازن مع إعادة العطالة). ومن أجل المركبة الطولية،
يعطي \omterm{thm:b3:continuum-elasticity:hooke}{قانون هوك} $\sigma_{xx} = (\lambda +
2\mu)\,\partial_xu_x$ (إذ تنعدم التشوّهات الجانبية في موجة
مستوية، لأن المادة المجاورة تمنع التحرر الجانبي)، ومنه
$\rho\,\partial_t^2u_x = (\lambda + 2\mu)\,\partial_x^2u_x$: أي
دالامبير بالسرعة $c_{\text{P}}$. ومن أجل المركبة المستعرضة،
$\sigma_{yx} =
2\mu\varepsilon_{yx} = \mu\,\partial_xu_y$، ومنه
$\rho\,\partial_t^2u_y = \mu\,\partial_x^2u_y$: أي السرعة $c_{\text{S}}$.
أما انفصال اضطراب عام إلى هاتين العائلتين فمُسلَّم به.
\end{proof}

\begin{example}[السكة والزلزال]\label{ex:b3:continuum-elasticity:rail}
في الفولاذ $c_{\text{P}} \approx \qty{5.9}{km/s}$ --- فطرقة على سكة
تبعد \qty{2}{km} تصل عبر الفولاذ في \qty{0.34}{s} وعبر الهواء
(بالسرعة $\qty{340}{m/s}$) في \qty{5.9}{s}: أي رنّتان. وفي
الغرانيت ($E = \qty{75}{GPa}$ و $\nu = 1/4$ و $\rho =
\qty{2.7e3}{kg/m^3}$): $c_{\text{P}} \approx \qty{5.8}{km/s}$ و
$c_{\text{S}} \approx \qty{3.3}{km/s}$. ولذلك يعلن الزلزال عن
نفسه مرتين: هزّة P حادّة، ثم، بعد ثوانٍ، رجّة S الأبطأ والأقوى
--- والتأخر يقيس المسافة
(\cref{pb:b3:continuum-elasticity:1}). وتعيش أنظمة الإنذار
المبكر بالزلازل داخل ذلك التأخر: إذ تسبق \omterm{prop:b3:continuum-elasticity:waves}{الموجة P}، والرسالة
اللاسلكية التي تطلقها، الموجةَ S التي تُحدث الدمار.
\end{example}

\begin{remark}[استعادة الأوتار والقضبان والصوت]\label{rem:b3:continuum-elasticity:limits}
أمواج كتاب السنة الثانية كلها حالات حدية لهذا الفصل. فالقضيب
الرفيع، الحرّ في الترقق جانبيًا، يحمل أمواج انضغاط بالسرعة
$\sqrt{E/\rho}$ (وهي أبطأ من $c_{\text{P}}$: إذ يليّن التحرر
الجانبي الاستجابة)؛ والوتر المشدود يحمل أمواجًا مستعرضة بالسرعة
$\sqrt{T/\rho_\ell}$، حيث تحلّ الشدّة محل الصلابة؛ والمائع لا
يبقي إلا $\lambda$ --- أي معامل انضغاطه الحجمي --- وسرعة الصوت
الوحيدة $\sqrt{\lambda/\rho}$.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:1}
(أ) تحقق من الوحدات: ما الباسكال بدلالة الكيلوغرام والمتر
والثانية، وما وحدة التشوّه؟ (ب) كابل فولاذي يعمل عند $\sigma =
\qty{200}{MPa}$: احسب تشوّهه. (ج) رباط مطاطي ($E \approx
\qty{2}{MPa}$) يُمدّ إلى ضعف طوله: أيّ ``تشوّه'' هذا، ولماذا لا
ينطبق إطار هذا الفصل عليه حقًا؟ (د)
رتّب بحسب كثافة \omterm{prop:b3:continuum-elasticity:energy}{الطاقة المرنة} المخزَّنة عند إجهاد التشغيل:
الفولاذ عند \qty{200}{MPa}، والمطاط عند \qty{1}{MPa}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:2}
احسب \omterm{def:b3:continuum-elasticity:strain}{موتّر التشوّه} لكل مجال انتقال وصف
التشوّه: (أ) $\vect u = \alpha(x, y, z)$؛ (ب) $\vect u =
(\gamma y, 0, 0)$؛ (ج) $\vect u = (-\omega y, \omega x, 0)$؛ (د)
$\vect u = (\varepsilon x, -\nu\varepsilon y, -\nu\varepsilon z)$ ---
وأيّ تجربة تحققه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:3}
تحمل قاعدة عمود غرانيتي ارتفاعه $h$ الإجهاد $\sigma
= \rho gh$. (أ) استنتج ذلك من معادلة التوازن مع
الثقالة. (ب) ينسحق الغرانيت عند نحو \qty{200}{MPa}: فما أطول
عمود ممكن؟ (ج) ارتفاع إفرست \qty{8.8}{km}: علّق. (د) لماذا
يمكن أن يكون الجبل أعلى قليلًا من عمود من صخره نفسه
(فكّر في الشكل)؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:4}
قضيب فولاذي ($E = \qty{200}{GPa}$ و $\nu = 0.30$) يُمدّ بالمقدار
$\varepsilon = \num{e-3}$. (أ) ما التشوّه الجانبي؟ (ب) ما التغير
الحجمي النسبي؟ (ج) من أجل أيّ $\nu$ لا يتغير الحجم البتة، وأيّ
مادة شائعة تقترب من ذلك؟ (د) لماذا يستحيل $\nu > 1/2$
(انظر في الانضغاط الهدروستاتيكي)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:5}
إجهاد الطوق. أسطوانة رقيقة الجدار (نصف قطرها $R$ وسمك جدارها $t \ll
R$) تحوي ضغطًا $p$. (أ) بموازنة القوى على نصف أسطوانة طوله
واحدي، بيّن أن الجدار يحمل الإجهاد المماسّي
$\sigma_\theta = pR/t$. (ب) بيّن أن الإجهاد الطولي (بالموازنة على
مقطع عرضي) هو $pR/2t$: فالأسطوانة مجهدة حول محيطها ضعف ما هي
مجهدة على طولها --- فبأيّ اتجاه تنشقّ النقانق؟ (ج) قارورة غطس:
$p =
\qty{200}{bar}$ و $R = \qty{9}{cm}$ و $t = \qty{5}{mm}$: احسب
$\sigma_\theta$ وقارنه بإجهاد خضوع فولاذي قدره
\qty{700}{MPa}. (د) لماذا تكون نهايات الخزانات عالية الضغط نصف
كروية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:6}
كتلة مطاطية ($E = \qty{3.0}{MPa}$ و $\nu \approx 0.5$)، قاعدتها $10
\times \qty{10}{cm}$ وارتفاعها \qty{2}{cm}، ملصوقة بين صفيحتين؛
وتُدفع الصفيحة العليا جانبيًا بقوة \qty{150}{N}. (أ) احسب $G$.
(ب) احسب إجهاد القصّ وزاوية القصّ $\gamma = \sigma_{xy}/G$؛ ثم
الانتقال الجانبي للصفيحة العليا. (ج) لماذا تكون $\nu \approx 1/2$
في المطاط (ما الصعب وما السهل على كومة متشابكة من السلاسل
البوليميرية)؟ (د) تحمل كتل مطاطية كهذه مبانيَ بأكملها في مناطق
الزلازل: فأيّ خاصية من خواصّ هذا السند تحمي المبنى، الصلابة في
الضغط أم الليونة في القصّ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:7}
حبل تسلق طوله $L = \qty{20}{m}$ ومقطعه $S =
\qty{80}{mm^2}$ ومعامله الفعّال $E = \qty{1.2}{GPa}$. (أ) احسب
صلابته $k = ES/L$. (ب) يسقط متسلق كتلته \qty{80}{kg} سقوطًا حرًا
\qty{4}{m} قبل أن يشتغل الحبل: بمساواة الطاقات، أوجد أقصى
استطالة للحبل (حل المعادلة من الدرجة الثانية؛ وأهمل في المحاولة
الأولى استمرار السقوط أثناء الكبح). (ج) استنتج القوة القصوى
والتباطؤ الأقصى بوحدة $g$. (د) لماذا يجب إحالة حبل تحمّل سقطة
عنيفة إلى التقاعد (أين ذهبت الطاقة، وماذا يقول منحني
الإجهاد--التشوّه)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:8}
الفتل. سلك نصف قطره $a$ وطوله $L$ ومعامل قصّه $G$ يُفتل
بزاوية $\theta$. (أ) بيّن أن أنبوبًا نصف قطره $r$ داخله يعاني
زاوية القصّ $\gamma(r) = r\theta/L$. (ب) إجهاد القصّ فيه
$G\gamma$: كامل $r \times$ الإجهاد على المقطع لتجد عزم
الإرجاع $\Gamma = (\pi Ga^4/2L)\,\theta$. (ج) الأسّ $a^4$:
انصف نصف القطر، فبأيّ عامل تهبط صلابة الفتل؟ (د) هذه الليونة
الشديدة هي سبب تعليق كافنديش (1798) ميزانه بسلك رفيع: من أجل $a = \qty{25}{\micro m}$ و $L =
\qty{1}{m}$ و $G = \qty{40}{GPa}$، احسب $C = \pi Ga^4/2L$ والدور
من أجل قضيب عزم عطالته $I = \qty{5e-4}{kg.m^2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:9}
(أ) انطلاقًا من اشتقاق الموجة المستوية في
\cref{prop:b3:continuum-elasticity:waves}، احسب $c_{\text{P}}$ و
$c_{\text{S}}$ في الفولاذ ($E = \qty{200}{GPa}$ و $\nu = 0.29$ و $\rho =
\qty{7.85e3}{kg/m^3}$). (ب) قارن $c_{\text{P}}$ بسرعة القضيب
الرفيع $\sqrt{E/\rho}$ وفسّر الفرق في جملة واحدة. (ج)
من أجل الماء ($\mu = 0$، ومعامل انضغاطه الحجمي \qty{2.2}{GPa}): ما سرعة S
وما سرعة P؟ (د) بأيّ زاوية تختلف حركة المادة في موجة P عنها في
موجة S، وكيف يميّز بينهما مقياس زلازل بثلاث مركبات؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:continuum-elasticity:10}
الانحناء. تُحنى عارضة إلى نصف قطر تقعّر $R$؛ فتقصر أليافها
الداخلية وتستطيل أليافها الخارجية، ويحفظ \emph{سطح محايد} بينهما
طوله. (أ) بيّن أن الليف الواقع على المسافة $y$ من السطح المحايد
تشوّهه $\varepsilon = y/R$، ومن ثمّ إجهاده $Ey/R$.
(ب) بجمع العزوم على المقطع العرضي، بيّن أن عزم الانحناء هو
$M = EI/R$ مع $I = \int y^2\,\dd S$ (وهو \emph{العزم الثاني})؛
واحسب $I$ من أجل مستطيل عرضه $b$ وارتفاعه $h$. (ج) الأسّ
$h^3$: لماذا يتدلى لوح محنيّ على وجهه العريض تدليًا ظاهرًا بينما
يبدو اللوح نفسه على حرفه صلبًا؟ احسب النسبة من أجل $b/h = 5$. (د) من أجل
عارضة ناتئة طولها $L$ محمَّلة بقوة $F$ في طرفها، يكون تدلي الطرف
$\delta = FL^3/3EI$ (وهو مُسلَّم به): قدّر $\delta$ من أجل لوح غطس
($L = \qty{1.8}{m}$ و $b = \qty{50}{cm}$ و $h = \qty{3.5}{cm}$، خشب $E
= \qty{12}{GPa}$) تحت غطّاس كتلته \qty{75}{kg}، وعلّق.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:continuum-elasticity:11}
الانبعاج. عمود نحيل (طوله $L$ وصلابته الانحنائية $EI$)،
مفصلي الطرفين، يحمل حمولة محورية $P$. لنفترض أنه ينحني
جانبيًا وفق $y(x)$. (أ) بيّن أن الحمولة تمارس حينئذٍ عزم الانحناء
$M(x) = -Py(x)$ حول المحور المنزاح، ومنه $EI\,y'' = -Py$. (ب) مع
$y(0) = y(L) = 0$، بيّن أن انحناءً غير معدوم يصير ممكنًا أول ما
يصير عند حمولة أويلر الحرجة
$P_{\text{c}} = \pi^2EI/L^2$. (ج) احسب $P_{\text{c}}$ من أجل مسطرة
مترية ($b = \qty{3}{cm}$ و $h = \qty{1.5}{mm}$ و $E = \qty{12}{GPa}$)
وقارنها بدفعة يدك. (د) فسّر انطلاقًا من $I$ لماذا تكون العظام
والخيزران وهياكل الدراجات \emph{أنابيب}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:continuum-elasticity:12}
سرعة الصوت في جسم صلب، انطلاقًا من الذرات. نمذج الجسم الصلب
سلاسلَ من الذرات تباعدها $a$، موصولة بواسطة ``نوابض'' صلابتها $\kappa$؛
وكتلة الذرة $m$. (أ) بيّن أن مدّ السلسلة يقابل شدّ الوسط
المتصل مع $E = \kappa/a$ (أحصِ النوابض في وحدة المساحة،
والاستطالة في كل نابض). (ب) بيّن أن $\rho = m/a^3$ واستنتج $c =
\sqrt{E/\rho} = a\sqrt{\kappa/m}$. (ج) قدّر $\kappa$ من عمق
رابطة بين ذرتين ($\sim\qty{3}{eV}$ على مدى $\sim
\qty{0.1}{nm}$): تجد $\kappa \sim \qty{50}{N/m}$؛ ومع $a =
\qty{2.5e-10}{m}$ و $m = \qty{1e-25}{kg}$، قدّر $c$. (د) قارن
بسرعات الصوت المقيسة في الفلزات، واستنتج ممّ صُنعت المرونة
اليومية.
\end{exercise}

\section{مسألة: الإصغاء إلى الأرض}

\begin{problem}[{مسألة نهاية الأسبوع --- كيف كشفت الزلازل عن
اللبّ السائل}]\label{pb:b3:continuum-elasticity:1}
زلزال واحد يرنّ الكوكب كله، وموجتا هذا الفصل المرنتان، إذ
تعبرانه، تصوّرانه بالأشعة. وتتابع هذه المسألة الفيزياء من \omterm{thm:b3:continuum-elasticity:hooke}{قانون
هوك} في الغرانيت إلى اكتشاف أولدهام سنة 1906 أن للأرض لبًّا ---
مع تقدير حجمه بالأشعة المستقيمة. خذ من أجل صخر القشرة والوشاح
$E = \qty{75}{GPa}$ و $\nu = 1/4$ و
$\rho = \qty{2.7e3}{kg/m^3}$؛ ونصف قطر الأرض $R_{\text{E}} =
\qty{6371}{km}$.

\medskip\noindent\textbf{الجزء الأول --- الصخر بوصفه جسمًا صلبًا مرنًا.}
\begin{enumerate}
  \item احسب معاملَي لامي $\mu$ و $\lambda$ لهذا
        الصخر، ولاحظ أن $\nu = 1/4$ يجعلهما متساويين.
  \item احسب $c_{\text{P}}$ و $c_{\text{S}}$، وتحقق من أن
        $c_{\text{P}} = \sqrt3\,c_{\text{S}}$.
  \item ينزلق الزلزال بوجهين صخريين أمتارًا في ثوانٍ:
        قدّر التشوّه المحرَّر إذا أرخى انزلاق قدره \qty{2}{m}
        الصخر على مقياس \qty{50}{km}، وقدّر الإجهاد الذي كان قد
        تراكم.
  \item انطلاقًا من كثافة الطاقة $w = \tfrac12 E\varepsilon^2$،
        قدّر \omterm{prop:b3:continuum-elasticity:energy}{الطاقة المرنة} في المتر المكعب، ثم الطاقة
        في حجم $50 \times 50 \times \qty{20}{km^3}$؛ وقارنها
        بميغاطن (\qty{4.2e15}{J}).
  \item لماذا ترجّ \omterm{prop:b3:continuum-elasticity:waves}{الموجة S} المباني عادةً أشدّ ممّا ترجّها
        \omterm{prop:b3:continuum-elasticity:waves}{الموجة P} (فكّر في اتجاه حركة الأرض من أجل موجة
        قادمة من الأسفل)؟
  \item في أيّ الموجتين يتغير حجم الأرض؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- وصولا مقياس الزلازل.}
\begin{enumerate}[resume]
  \item تسجّل محطة وصول \omterm{prop:b3:continuum-elasticity:waves}{الموجة P}، ثم وصول \omterm{prop:b3:continuum-elasticity:waves}{الموجة S} بعدها
        بالمدة $\Delta t$. من أجل مصدر على المسافة $d$ (قصيرة بما
        يكفي لاعتماد الأشعة المستقيمة)، بيّن أن
        \[
        d = \frac{\Delta t}{\dfrac{1}{c_{\text{S}}} -
        \dfrac{1}{c_{\text{P}}}} .
        \]
  \item قدّر المعامل: كم كيلومترًا لكل ثانية من تأخر
        S عن P؟
  \item تقيس محطة $\Delta t = \qty{28}{s}$: فكم يبعد
        الزلزال؟
  \item محطة واحدة تعطي مسافة لا موضعًا: فكم محطة يلزم
        لتحديد مركز الزلزال السطحي، وكيف، هندسيًا؟
  \item تدوّي صفارات الإنذار المبكر في اليابان قبل الرجّة
        القوية بثوانٍ: من أجل زلزال على عمق \qty{80}{km} تحت
        مدينة، كم من الإنذار يتيحه تأخر S عن P وحده؟
  \item تضيف الإنذارات الحديثة سرعة الضوء: إذ تُذاع \omterm{prop:b3:continuum-elasticity:waves}{الموجة P}
        المكشوفة قرب المصدر سبقًا للموجتين معًا. من أجل مدينة
        تبعد \qty{200}{km} عن مركز الزلزال، متى تصل \omterm{prop:b3:continuum-elasticity:waves}{الموجة S} بعد
        التمزق، وأيّ إنذار تتيحه رسالة لاسلكية تُرسل عند أول
        وصول للموجة P على مسافة \qty{20}{km} من المصدر؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- أمواج تعبر الكوكب.}
\begin{enumerate}[resume]
  \item عند ضغوط الوشاح العميق تكبر المعاملات: فتبلغ سرعات P
        \qty{13.7}{km/s} عند قاعدة الوشاح. بأخذ متوسط تقريبي
        $c_{\text{P}} \approx \qty{10}{km/s}$، كم يلزم موجة P
        لتعبر الأرض على قطرها؟
  \item تسجّل المحطات زلازل من الجهة المقابلة من الكرة:
        فماذا يقول مجرد وجود هذه الوصولات عن باطن الأرض
        العميق (أهو مرن؟ أهو منصهر بأكمله؟)؟
  \item عرّف الزاوية المركزية $\Delta$ (وهي الزاوية عند مركز
        الأرض بين المصدر والمحطة). من أجل الأشعة المستقيمة، بيّن
        أن طول الوتر هو $2R_{\text{E}}\sin(\Delta/2)$.
  \item قدّر الوتر وزمن انتقاله بالشعاع المستقيم من أجل
        $\Delta = 60^\circ$ عند المتوسط $c_{\text{P}} \approx
        \qty{10}{km/s}$.
  \item نحو سنة 1900 لاحظ أولدهام أن أمواج S تُسجَّل حتى
        $\Delta \approx 103^\circ$ ثم \emph{تختفي}: فلا موجة S
        مباشرة وراء ذلك. فأيّ خاصية لمنطقة عميقة داخل الأرض
        تقتل أمواج S، وأيّ حالة من حالات المادة لها تلك
        الخاصية؟
  \item أمواج P وراء $103^\circ$ ليست غائبة بل ضعيفة
        ومتأخرة ومزاحة (وهي ``منطقة ظل'' تمتد حتى
        $\approx 142^\circ$): فلماذا تؤخّر منطقة سائلة أمواج P
        وتكسرها من دون أن توقفها؟
  \item استنتج في جملة واحدة ما الذي يقبع في مركز
        الأرض.
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- وزن اللبّ بمسطرة.}
\begin{enumerate}[resume]
  \item شعاع S مستقيم يغادر المصدر يماسّ اللبّ إذا كان أقرب
        اقتراب له من المركز يساوي نصف قطر اللبّ
        $R_{\text{c}}$. بيّن أن هذا الشعاع المماسّ يبلغ
        السطح عند الزاوية المركزية $\Delta^* =
        2\arccos(R_{\text{c}}/R_{\text{E}})$.
  \item انطلاقًا من $\Delta^* = 103^\circ$، احسب $R_{\text{c}}$.
  \item القيمة الحديثة \qty{3480}{km}: احسب خطأك، وفسّر
        \emph{إشارته}: فالأشعة الحقيقية تنحني عائدة نحو
        السطح كلما كبرت السرعة مع العمق --- ناقش هل تبالغ
        الأشعة المستقيمة في تقدير اللبّ أم تقلّل منه.
  \item سنة 1936 وجدت إنغه ليمان وصولات P ضعيفة \emph{داخل}
        منطقة الظل: فماذا استنتجت أنه يقبع داخل اللبّ
        السائل؟
  \item نصف قطر اللبّ الداخلي \qty{1220}{km} وهو صلب:
        اقترح المشاهدة الموجية التي تؤكد صلابته
        (أيّ موجة توجد هناك ويمنعها اللبّ الخارجي؟).
  \item لخّص النتيجة الموسومة: سرعتان مرنتان في الصخر
        ($5.8$ و\qty{3.3}{km/s})، وموجة مفقودة وراء
        $103^\circ$، ومسطرة، تعطي لبًّا سائلًا نصف قطره
        $\approx \qty{4000}{km}$ --- في حدود $15\%$ من القيمة
        الحديثة \qty{3480}{km}، مقيسًا عبر \qty{6000}{km} من
        الصخر الصلب.
\end{enumerate}
\end{problem}
