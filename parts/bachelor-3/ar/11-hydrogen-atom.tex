\chapter{ذرة الهدروجين}\label{ch:b3:hydrogen-atom}

تسعة أعشار ذرات الكون هدروجين: بروتون واحد
يمسك إلكترونًا واحدًا، وهي الذرة الوحيدة التي تستطيع الفيزياء حلّها
بالضبط --- والمفتاح الذي فتح كل ذرة سواها. وقد خمّنت النظرية
الكمومية القديمة في \cref{ch:b3:hamiltonian-mechanics} طاقاتها؛
والآلة الآن جاهزة \emph{لاستنتاجها}: فيدخل كمون
كولوم في \omterm{prop:b3:schrodinger-three-dimensions:swave}{المعادلة القطرية} من الفصل السابق، فتخرج
المستويات $-E_{\text{I}}/n^2$، \omterm{thm:b3:hydrogen-atom:levels}{ونصف قطر بور}، والأغلفة
والأغلفة الجزئية في جدول الكيمياء الدوري، والسلاسل الطيفية
التي يقرأها الفلكيون في كل سديم. والحلّ نفسه، معادَ التحجيم،
يصف الهليوم المتأين والذرات الميونية والبوزيترونيوم و``ذرات''
الإلكترون--الفجوة داخل أنصاف النواقل؛ وممدودًا إلى $n
\approx 100$ يصف ذرات ريدبرغ التي يبلغ عرضها نصف ميكرومتر،
والتي ترسم همساتها الراديوية خرائط سحب المجرّة المتأينة وتقود
تفاعلاتها اليوم حواسيب كمومية.

\section{حلّ مسألة كولوم}

\begin{theorem}[مستويات الهدروجين]\label{thm:b3:hydrogen-atom:levels}
\index{ذرة الهدروجين}\index{العدد الكمومي الرئيسي}\index{نصف قطر بور}
من أجل $V(r) = -e^2/4\pi\varepsilon_0 r$ (ونكتب $k = e^2/4\pi
\varepsilon_0$)، تُوسم الحالات المقيدة بالثلاثي $(n, l, m)$ مع
\[
E_n = -\frac{E_{\text{I}}}{n^2} , \qquad
E_{\text{I}} = \frac{m_{\text{e}}k^2}{2\hbar^2} = \qty{13.6}{eV} ,
\qquad n = 1, 2, 3, \dots
\]
ومن أجل كل $n$، تكون الأعداد المدارية $l = 0, 1, \dots, n-1$ و
$m = -l, \dots, l$. والطول الطبيعي هو \omterm{thm:b3:hydrogen-atom:levels}{نصف قطر بور} $a_0 =
\hbar^2/m_{\text{e}}k = \qty{52.9}{pm}$؛ والحالة الأساسية هي
\[
\psi_{100} = \frac{1}{\sqrt{\pi a_0^3}}\;\eu^{-r/a_0} .
\]
وتتوقف الطاقة على $n$ \emph{وحده}: فبإحصاء قيم $m$ و
$l$، يكون المستوي $n$ منحلًّا $n^2$ مرة (ويتضاعف بالسبين،
\cref{ch:b3:spin-two-level}) --- أي أبعد بكثير من الانحلال
$(2l+1)$ الذي يفسّره تماثل المناحي.
\end{theorem}

\begin{proof}[برهان جزئي]
من أجل الحالة الأساسية، نجرّب $u = Cr\,\eu^{-r/a}$ في \omterm{prop:b3:schrodinger-three-dimensions:swave}{المعادلة
القطرية} مع $l = 0$: فنجد $u'' = (1/a^2 - 2/ar)u$، ومن ثمّ يقتضي
$-\tfrac{\hbar^2}{2m_{\text{e}}}u'' - \tfrac kr u = Eu$ أن يكون
$\hbar^2/m_{\text{e}}a = k$ (بمطابقة حدود $1/r$) --- أي
$a = a_0$ --- و $E = -\hbar^2/2m_{\text{e}}a_0^2 = -E_{\text{I}}$.
أما الحلّ العام (بكثيرات حدود لاغير، مع $n - l - 1$ عقدة قطرية)
فمُسلَّم به؛ وكل خطوة من بنائه أولية لكنها
طويلة. والانحلال ``العرضي'' في $l$ يعكس سرًّا كلاسيكيًّا
للقوة $1/r$ الصرفة --- فإهليلجيات كبلر لا تتقدم، وثمة
متجهة محفوظة إضافية (لابلاس--رونغه--لنز) تشير على المحور الكبير
الثابت؛ وصورتها الكمومية هي ما يربط $l$ مختلفة بطاقة
واحدة (\cref{exo:b3:hydrogen-atom:12}).
\end{proof}

\section{أين يقع الإلكترون}

\begin{proposition}[التوزيعات القطرية]\label{prop:b3:hydrogen-atom:radial}
\index{الاحتمال القطري}
احتمال إيجاد الإلكترون بين $r$ و $r + \dd r$
هو $P(r)\,\dd r$ مع $P(r) = |u(r)|^2$، أي مربع
الدالة القطرية المنظَّمة في \cref{thm:b3:quantum-angular-momentum:radial}. ومن أجل أدنى الحالات:
$P_{1s} \propto r^2\eu^{-2r/a_0}$، وذروتها عند $r = a_0$ بالضبط مع
$\langle r\rangle = \tfrac32 a_0$؛ وتُظهر $P_{2s}$ حدبتين
تفصل بينهما \emph{عقدة} كروية؛ و $P_{2p} \propto
r^4\eu^{-r/a_0}$، وذروتها عند $4a_0$. وعمومًا $\langle r\rangle
\approx n^2a_0$: فتكبر الذرات \emph{تربيعيًا} مع الإثارة،
بينما يتقلص ربطها وفق $1/n^2$ --- وهي الرافعة وراء
عمالقة ريدبرغ في \cref{pb:b3:hydrogen-atom:1}.
\end{proposition}
\admitted

\section{الطيف}

\begin{proposition}[السلاسل وقواعد الانتقاء]\label{prop:b3:hydrogen-atom:series}
\index{سلسلة بالمر}\index{سلسلة لايمان}
تُصدر القفزة $n' \to n$ طول الموجة
\[
\frac{1}{\lambda} = R_{\text{H}}\Big(\frac{1}{n^2} -
\frac{1}{n'^2}\Big) , \qquad
R_{\text{H}} = \frac{E_{\text{I}}}{hc} = \qty{1.097e7}{m^{-1}} ,
\]
شريطة قاعدة الانتقاء ثنائية القطب $\Delta l = \pm1$ (فالفوتون
يحمل $\hbar$). وتقع \omterm{prop:b3:hydrogen-atom:series}{سلسلة لايمان} ($\to n = 1$) في فوق البنفسجي
البعيد ابتداءً من \qty{121.6}{nm}؛ وتبدأ \omterm{prop:b3:hydrogen-atom:series}{سلسلة بالمر} ($\to n = 2$)
عند الخط الأحمر $H\alpha$ عند \qty{656.3}{nm} --- وهو لون
السدم المصدِرة والشواظ الشمسي؛ وتتراجع سلسلة باشن وما بعدها
إلى تحت الأحمر، وبين $n = 110$ و $109$ تهبط الصيغة نفسها
عند \qty{5.0}{GHz}: فالهدروجين يتكلم من
فوق البنفسجي إلى القرص الراديوي.
\end{proposition}

\begin{proof}
بانحفاظ الطاقة مع \cref{thm:b3:hydrogen-atom:levels}؛
والثابت $R_{\text{H}}$ كما في \cref{pb:b3:hamiltonian-mechanics:1}، مستنتَجًا
الآن لا مفروضًا.
\end{proof}

\begin{example}[الذرات الشبيهة بالهدروجين: حلّ واحد وذرات كثيرة]\label{ex:b3:hydrogen-atom:scaling}
استبدل بشحنة البروتون الشحنةَ $Ze$ وبالإلكترون أيّ كتلة
دائرة $\mu$: فتُعاد كل صيغة تحجيمها وفق
\[
a \to a_0\,\frac{m_{\text{e}}}{\mu}\,\frac1Z , \qquad
E_{\text{I}} \to E_{\text{I}}\,\frac{\mu}{m_{\text{e}}}\,Z^2 .
\]
He$^+$ (مع $Z = 2$): \qty{54.4}{eV}. والإلكترونات الداخلية للذرات الثقيلة
($Z \sim 30$): طاقات الغلاف K بالكيلو إلكترون-فولط --- وهي الأشعة السينية
المميِّزة التي رتّب بها موزلي العناصر
(\cref{exo:b3:hydrogen-atom:5}). والهدروجين الميوني ($\mu = 207
m_{\text{e}}$): ذرة بمقياس الفمتومتر تسبر البروتون نفسه.
والبوزيترونيوم ($e^+e^-$ مع $\mu = m_{\text{e}}/2$): نصف الربط،
وضعف الحجم، وعمر قصير ينتهي بفوتونات إفناء.
وفي نصف ناقل، يدور إلكترون وفجوة أحدهما حول الآخر
بكتلتين فعّالتين صغيرتين في عازل حاجب: فينشأ
\emph{إكسيتون}، وهي الذرة نفسها وقد كبرت إلى عشرة نانومترات و
مليات إلكترون-فولط (\cref{exo:b3:hydrogen-atom:7}) --- فالهدروجين
قالب أكثر منه ذرة.
\end{example}

\begin{method}[العمل مع الجمل الهدروجينية]\label{met:b3:hydrogen-atom:method}
(1) أعد التحجيم أولًا: فالعلاقتان $a \propto 1/\mu Z$ و $E \propto \mu Z^2$ تحوّلان أيّ
جواب هدروجيني إلى أيّ جواب شبيه بالهدروجين. (2) والأحجام: $\langle
r\rangle \sim n^2a$؛ والتباعدات بجوار المستوي $n$: $\dd E/\dd n =
2E_{\text{I}}/n^3$. (3) والأطياف: صيغة ريدبرغ مع $\Delta l =
\pm1$. (4) والنفاذ: الحالات $s$ تحسّ النواة، والحالات ذات $l$
الكبيرة تدور في الخارج --- وهي رافعة كيمياء الذرات المتعددة الإلكترونات. (5)
ومراسي التعقّل: $\qty{13.6}{eV}$ و $\qty{52.9}{pm}$ و
$\qty{121.6}{nm}$ و $\qty{656.3}{nm}$ --- أربعة أعداد تستحق
الحفظ مدى العمر.
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:1}
(أ) تحقق بالتعويض من أن $u = Cr\eu^{-r/a_0}$ يحلّ \omterm{prop:b3:schrodinger-three-dimensions:swave}{المعادلة
القطرية} من أجل $l = 0$ مع $E = -E_{\text{I}}$، شريطة أن يكون $a_0 =
\hbar^2/m_{\text{e}}k$. (ب) نظّمه (مع $\int_0^\infty
x^2\eu^{-x}\dd x = 2$). (ج) تحقق من أبعاد $a_0$ و
$E_{\text{I}}$. (د) لماذا لا توجد حالة دون $-E_{\text{I}}$
(حاجج وفق \cref{exo:b3:schrodinger-three-dimensions:12})؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:2}
احسب (أ) طولَي موجة لايمان $\alpha$ وحدّ لايمان؛ (ب) خطوط
بالمر الثلاثة الأولى وحدّ بالمر ---
أيّها مرئي، وبأيّ ألوان؟ (ج) مدى سلسلة باشن؛
(د) تواتر الانتقال $n = 110 \to 109$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:3}
من أجل الحالة الأساسية: (أ) عيّن أعظمية $P_{1s}(r)$؛ (ب)
احسب $\langle r\rangle$؛ (ج) احسب احتمال إيجاد
الإلكترون وراء $2a_0$ (مع $\int_x^\infty t^2\eu^{-t}\dd t = (x^2 +
2x + 2)\eu^{-x}$)؛ (د) لماذا تضلّل صور ``المدار'' عند نصف قطر
$a_0$ بالضبط؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hydrogen-atom:4}
(أ) اذكر أزواج $(l, m)$ من أجل $n = 3$ وتحقق من العدد $n^2 =
9$. (ب) مع السبين، كم حالة في الأغلفة $n = 1, 2, 3$؟ (ج)
طابقها بأطوال صفوف الجدول الدوري $2, 8, 18$. (د)
أيّ انحلال (في $m$ أم في $l$) يبقى في الإلكترون الخارجي
للصوديوم، ولماذا ينكسر الآخر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:5}
سلّم موزلي. إلكترون داخلي (من الغلاف K) في عنصر عدده $Z$
يتحرك في مجال نوويّ شبه عارٍ يحجبه إلكترون K
الآخر: فالشحنة الفعّالة $\approx Z - 1$. (أ) بيّن أن طاقة الأشعة السينية
للانتقال $2p \to 1s$ هي تقريبًا $\tfrac34(Z-1)^2E_{\text{I}}$. (ب)
قدّرها من أجل النحاس ($Z = 29$) وقارنها بالقيمة المقيسة للخط
K$\alpha$ عند \qty{8.05}{keV}. (ج) رسم موزلي (1913)
$\sqrt{\nu}$ في مقابل $Z$ فحصل على خطوط مستقيمة: فبم برهن ذلك
على معنى العدد الذرّي؟ (د) تنبّأ بطاقة الخط K$\alpha$
للعنصر المفقود حينذاك $Z = 43$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:8}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:10}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:11}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:12}
الانحلال غير المعقول. (أ) في الميكانيك الكلاسيكي، بيّن أنه

بمتجهة لابلاس--رونغه--لنز المحفوظة $\vect A = \vect p\wedge\vect

إضافي وراء الدورانات --- والتناظر الإضافي يعني انحلالًا
إضافيًا (تذكّر \cref{exo:b3:quantum-formalism:9}). (ج) أضف

\end{exercise}

\section{مسألة: ذرات عملاقة}

\begin{problem}[{مسألة نهاية الأسبوع --- ذرات ريدبرغ، من التوهج
الراديوي للمجرّة إلى الحواسيب الكمومية}]\label{pb:b3:hydrogen-atom:1}
مُدّ الهدروجين إلى $n \approx 100$ يصر غرضًا من صنف آخر:
عرضه ميكرومترات، وربطه مليات إلكترون-فولط، وحساسيته
عبثية --- ونفعه عبثي كذلك. وتكبّر هذه المسألة
حلّ الهدروجين، أولًا إلى السحب بين النجمية التي تبثّ
عند أطوال موجات سنتيمترية، ثم إلى مصفوفات المخبر
حيث يشابك مثل هؤلاء العمالقة بعضهم بعضًا فيصيرون معالجات. المعطيات:
$E_{\text{I}} = \qty{13.6}{eV}$ و $a_0 = \qty{52.9}{pm}$ و
$R_{\text{H}}c = \qty{3.29e15}{Hz}$ و $k_{\text{B}}T$ عند
\qty{300}{K} يساوي \qty{25.9}{meV}.

\medskip\noindent\textbf{الجزء الأول --- قوانين التحجيم.}
\begin{enumerate}
  \item أعطِ التحجيمات الأربعة الأساسية مع $n$: الحجم $\langle
        r\rangle$، والربط $|E_n|$، والتباعد $E_{n+1} - E_n$ (من أجل
        $n$ الكبيرة)، والتواتر المداري الكلاسيكي لمدار
        بور المقابل.
  \item بيّن أنه من أجل $n$ الكبيرة يقارب تواتر الانتقال $n+1 \to
        n$ المقدارَ $2R_{\text{H}}c/n^3$، وتحقق من أنه يساوي
        التواتر المداري الكلاسيكي (أي مبدأ
        التقابل في \cref{pb:b3:hamiltonian-mechanics:1}،
        مكتملًا).
  \item قدّر الحجم والربط عند $n = 50$ وعند $n = 100$.
  \item يتغير ثنائي القطب الكهربائي لذرة ريدبرغ وفق $n^2ea_0$:
        قدّره عند $n = 50$ بالديباي ($1\,\text{D} =
        \qty{3.34e-30}{C.m}$) وقارنه بقيمة جزيء الماء
        \qty{1.85}{D}.
  \item قدّر المجال الكهربائي الذي يمزّق الذرة عند $n = 50$،
        بالصيغة $F \sim |E_n|/(e\,\langle r\rangle)$، بالفولط في السنتيمتر.
  \item لماذا لا تستطيع ذرات كهذه أن تنجو في كيمياء الخلاء
        المخبري العادي --- ومع ذلك تنجو تمامًا في الفضاء بين النجوم
        (قارن معدلات التصادم)؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- خطوط إعادة التركيب في
المجرّة.}
في السدم المتأينة، تأسر البروتونات إلكترونات في $n$ عالية؛ ويشعّ
التدحرج مشطًا من ``خطوط إعادة التركيب الراديوية''.
\begin{enumerate}[resume]
  \item احسب تواتر الانتقال $n = 110 \to 109$
        (المسمّى H109$\alpha$).
  \item في أيّ نطاق يقع، وأيّ صنف من المقاريب
        يستقبله؟
  \item احسب طول موجة H109$\alpha$ وقارنه بحجم
        الذرة المصدِرة: كم ذرة تسع في طول موجة واحد؟
  \item تقيس هذه الخطوط درجة حرارة السديم بعرضها
        الدوبلري: فعند $T = \qty{e4}{K}$، تكون سرعة الهدروجين
        الحرارية $\sim\qty{12.8}{km/s}$ --- احسب عرض الخط النسبي
        $\Delta\nu/\nu$ وعرض H109$\alpha$
        بالكيلوهرتز.
  \item تعبر الأمواج الراديوية الغبار الذي يُسوّد السماء الضوئية:
        فما الذي يتيح ذلك لفلكيي خطوط إعادة التركيب أن يرسموه ولا
        يستطيعه فلكيو خطوط بالمر؟
  \item فوق $n \sim 1000$ تقريبًا (أي ذرات عرضها $0.05$
        ملّيمتر!)، تتلطخ المستويات فتصير متصلة حتى في الفضاء:
        سمِّ أثرين يعرّضان مثل هذه الحالات أو يدمّرانها.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- عمالقة في المخبر.}
\begin{enumerate}[resume]
  \item تقود الليزرات ذرات الروبيديوم إلى $n \approx 70$ في
        مصفوفات ملاقط ضوئية تباعدها ميكرومترات. وتتفاعل ذرتان
        كهاتين تفصلهما مسافة $R$ بثنائيَي قطبهما المستحثين
        (تذكّر \cref{exo:b3:harmonic-oscillator:10}): فمع ثنائي قطب
        $\propto n^2$، بيّن أن قوة فان دير فالس تتغير وفق
        $n^{11}$ الهائل (استعمل $C_6 \propto d^4/\Delta E$ مع
        تباعد المستويات $\Delta E \propto n^{-3}$).
  \item ``الحصار'': داخل نصف قطر $R_{\text{b}}$، يزيح
        التفاعل الحالةَ المثارة مرتين خارج رنين الليزر،
        فلا تستطيع ذرتان \emph{كلتاهما} أن تُثارا.
        اشرح في جملة واحدة لماذا يحقق هذا بوابة ذات كيوبتين.
  \item تبلغ أنصاف أقطار الحصار عدة ميكرومترات --- أي آلاف
        أضعاف حجم الذرات في حالتها الأساسية: فلماذا يكون هذا
        المدى الطويل (قارن مدى الكيمياء النانومتري)
        هو بالضبط ما يريده معالج قابل للتوسيع؟
  \item تعيش حالة ريدبرغ عند $n = 70$ نحو $\sim\qty{100}{\micro s}$
        بينما تستغرق البوابة $\sim\qty{0.5}{\micro s}$: فكم عملية
        تسع تقريبًا في عمر واحد، ولماذا تهمّ تلك
        النسبة لا العمر وحده؟
  \item إشعاع الجسم الأسود في درجة حرارة الغرفة، وذروته قرب
        \qty{100}{meV} لكنه غنيّ بذيل منخفض الطاقة، يقود
        انتقالات بين مستويات ريدبرغ المتجاورة التي لا يتباعد بعضها عن بعض إلا
        $\sim\qty{0.1}{meV}$: فلماذا يجب أن تحيط التجارب الدقيقة الذرات بدروع باردة؟
  \item قدّر كم انتقالًا ثنائي قطب شبيهًا بالهوائي
        ($\propto n^2$) يقوده حمام فوتونات عند \qty{300}{K} في
        الثانية بالمقارنة مع ذرة عند $n = 1$: فأيّ تحجيم يجعل
        ذرات ريدبرغ \emph{مجسّات} بديعة للمجالات الميكروية
        والتيراهرتزية؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- الصورة الكبرى.}
\begin{enumerate}[resume]
  \item صيغة واحدة، $E_n = -E_{\text{I}}\mu Z^2/m_{\text{e}}n^2$،
        تمسح كم رتبة من رتب طاقة الربط، من
        اليورانيوم الشبيه بالهدروجين ($Z = 92$ و $n = 1$) إلى
        حالة ريدبرغ عند $n = 1000$؟ احسب الطرفين.
  \item تتغير أعمار الإشعاع للحالات ذات $l$ الصغيرة وفق $n^3$:
        فانطلاقًا من عمر $2p$ البالغ \qty{1.6}{ns}، قدّر عمر
        الحالة $100p$ --- وقارنه بمسألة إشعاع الجسم الأسود في
        درجة حرارة الغرفة في الجزء الثالث.
  \item في سيرن، تُدرَس اليوم أطياف ذرات الهدروجين المضادّة
        على الفاصل $1s \to 2s$ حتى خمسة عشر رقمًا: فأيّ
        تناظر أساسي يختبره التوافق مع الهدروجين العادي،
        ولماذا يكون الهدروجين الذرة الصحيحة لتلك
        المقارنة؟
  \item عند $n \sim 100$ تلفّ موجة دي بروي للإلكترون مدارًا
        شبه كلاسيكي؛ وعند $n = 1$ لا وجود لمدار البتة:
        ففي جملة واحدة، أين تُشعَل الصورة الكلاسيكية؟
  \item تُقاس ثوابت ريدبرغ حتى خمسة عشر رقمًا: فلماذا يكون
        الهدروجين، من بين كل الجمل، مرساة الدقة الطبيعية لفيزياء
        الذرات؟
  \item استعملت جائزة نوبل لسنة 2012 (هاروش) ذرات ريدبرغ
        عدّادات فوتونات لا تدمّر الفوتون: فأيّ خاصيتين
        من الجزأين الأول والثالث تجعلانها مسابر
        مثالية غير مدمِّرة للمجالات الميكروية؟
  \item لخّص النتيجة الموسومة: تحجيمات $n^2$ و $n^{-2}$ و $n^{-3}$
        و $n^{11}$ لذرة واحدة محلولة بالضبط تمدّ
        الهدروجين من \qty{52.9}{pm} إلى نصف ميكرومتر، وتوالف
        صوته من \qty{121.6}{nm} إلى \qty{6}{cm}، وتجعله
        منارة المجرّة الراديوية وبوابة الكيوبتين في
        الحواسيب الكمومية بالذرات المتعادلة معًا.
\end{enumerate}
\end{problem}
