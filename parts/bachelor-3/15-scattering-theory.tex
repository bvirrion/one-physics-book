\chapter{Scattering Theory}\label{ch:b3:scattering-theory}

Almost everything we know about the small is learned by throwing
things at it and watching what bounces. Rutherford discovered the
nucleus by counting alpha-particle flashes on a zinc-sulfide screen;
fifty years later the same experiment, run with electrons at higher
energy, found quarks inside the proton; between and since, every
\omterm{def:b3:scattering-theory:cross-section}{cross-section} in nuclear, particle, atomic and condensed-matter
physics has been read through the theory of this chapter. The
language is the \emph{\omterm{def:b3:scattering-theory:cross-section}{cross-section}} --- an effective target area;
the quantum object is the \emph{\omterm{thm:b3:scattering-theory:amplitude}{scattering amplitude}}, whose \omterm{thm:b3:scattering-theory:born}{Born
approximation} says something unforgettable: a diffuse target is
probed in Fourier space, angle by angle. At low energy the \omterm{thm:b3:scattering-theory:partial}{partial
waves} take over, collapsing every complicated potential into a
single number --- the \omterm{thm:b3:scattering-theory:partial}{scattering length} --- that today tunes
ultracold quantum gases; and where a projectile can linger in a
quasi-bound state, the \omterm{def:b3:scattering-theory:cross-section}{cross-section} spikes into resonances, the
spectroscopy of the otherwise invisible.

\section{Cross-sections}

\begin{definition}[Cross-section]\label{def:b3:scattering-theory:cross-section}
\index{cross-section}\index{differential cross-section}
Send a uniform beam of flux $\Phi$ (particles per unit area per
second) onto one target particle. The rate of projectiles scattered
into the solid angle $\dd\Omega$ around the direction $(\theta,
\varphi)$ defines the \emph{differential cross-section},
\[
\dd\dot N = \Phi\,\frac{\dd\sigma}{\dd\Omega}\,\dd\Omega ,
\]
and its integral $\sigma = \int(\dd\sigma/\dd\Omega)\,\dd\Omega$ is
the \emph{total cross-section}: the area the target effectively
presents. For a thin slab with $n$ targets per unit volume, a beam
attenuates as $\eu^{-n\sigma x}$: cross-sections are measured by
counting what emerges. Units: nuclear physics uses the \emph{barn},
$\qty{1}{b} = \qty{e-28}{m^2}$ --- about a uranium nucleus's
geometric size, ``as big as a barn'' to a neutron.
\end{definition}

\section{The amplitude and the Born approximation}

\begin{theorem}[Scattering states and the amplitude]\label{thm:b3:scattering-theory:amplitude}
\index{scattering amplitude}
For a particle of energy $E = \hbar^2k^2/2m$ meeting a localised
potential $V(\vect r)$, the stationary scattering state behaves at
large distance as
\[
\psi(\vect r) \;\longrightarrow\;
\eu^{\iu kz} + f(\theta)\,\frac{\eu^{\iu kr}}{r} ,
\]
an incident plane wave plus an outgoing spherical wave modulated by
the \emph{\omterm{thm:b3:scattering-theory:amplitude}{scattering amplitude}} $f(\theta)$ (a length), and
\[
\frac{\dd\sigma}{\dd\Omega} = |f(\theta)|^2 .
\]
\end{theorem}

\begin{proof}[Partial proof]
The $1/r$ fall-off makes the outgoing flux through any large sphere
finite; computing the \omterm{prop:b3:schrodinger-three-dimensions:current}{probability currents}
(\cref{prop:b3:schrodinger-three-dimensions:current}) of the two
terms, the radial outgoing current per solid angle is
$(\hbar k/m)|f|^2$ against the incident flux $\hbar k/m$: the ratio
is the definition of $\dd\sigma/\dd\Omega$. That solutions of this
asymptotic form exist is admitted.
\end{proof}

\begin{theorem}[Born approximation]\label{thm:b3:scattering-theory:born}
\index{Born approximation}\index{momentum transfer}
For a weak potential (or fast projectile), first-order \omterm{thm:b3:perturbation-theory:corrections}{perturbation
theory} in $V$ gives
\[
f(\theta) = -\frac{m}{2\pi\hbar^2}
\int V(\vect r)\,\eu^{\iu\vect q\cdot\vect r}\,\dd^3r ,
\qquad
\vect q = \vect k - \vect k' , \quad
q = 2k\sin\frac\theta2 :
\]
the amplitude is, up to constants, the \emph{Fourier transform of
the potential} evaluated at the \omterm{thm:b3:scattering-theory:born}{momentum transfer} $\hbar\vect q$.
Scattering experiments are Fourier analysers of matter: small
angles read long wavelengths (coarse structure), large angles the
fine detail --- the deepest reason why ``higher energy'' means
``smaller distances''.
\end{theorem}

\begin{proof}[Partial proof]
This is Fermi's golden rule
(\cref{thm:b3:perturbation-theory:golden}) run between plane waves
$\ket{\vect k} \to \ket{\vect k'}$: the matrix element is
$\int V\eu^{\iu(\vect k - \vect k')\cdot\vect r}\dd^3r$, the
density of final states supplies the kinematic factors, and the
bookkeeping (admitted) assembles them into $f$. Elasticity gives
$|\vect k'| = |\vect k|$, whence $q = 2k\sin(\theta/2)$ by the
isoceles triangle of $\vect k, \vect k'$.
\end{proof}

\begin{example}[From Yukawa to Rutherford]\label{ex:b3:scattering-theory:rutherford}
\index{Rutherford scattering}
For the screened Coulomb potential $V = \dfrac{Q_1Q_2}{4\pi
\varepsilon_0 r}\,\eu^{-r/a}$ the Born integral is elementary
(\cref{exo:b3:scattering-theory:5}) and, as the screening radius
$a \to \infty$,
\[
\frac{\dd\sigma}{\dd\Omega} =
\Big(\frac{Q_1Q_2}{16\pi\varepsilon_0 E}\Big)^2
\frac{1}{\sin^4(\theta/2)} :
\]
the \emph{Rutherford \omterm{def:b3:scattering-theory:cross-section}{cross-section}}. By a coincidence unique to
$1/r$, the classical calculation, the \omterm{thm:b3:scattering-theory:born}{Born approximation} and the
exact quantum answer all agree --- which is why Rutherford,
computing classically in 1911, got the right formula and, from its
verification flash by flash, the nucleus
(\cref{pb:b3:scattering-theory:1}).
\end{example}

\section{Partial waves and the scattering length}

\begin{theorem}[Phase shifts]\label{thm:b3:scattering-theory:partial}
\index{partial waves}\index{phase shift}\index{scattering length}
For a central potential, decompose the scattering state over
angular momenta: each partial wave $l$ leaves the potential region
with its radial oscillation \emph{shifted} by a phase $\delta_l(k)$
--- attraction advances it, repulsion delays it --- and the
observable amplitude reassembles as
\[
f(\theta) = \frac1k\sum_{l=0}^\infty(2l+1)\,\eu^{\iu\delta_l}
\sin\delta_l\,P_l(\cos\theta) , \qquad
\sigma = \frac{4\pi}{k^2}\sum_l(2l+1)\sin^2\delta_l .
\]
At low energy, classical intuition ($l \sim kb$ for impact
parameter $b$) says only $l = 0$ penetrates: scattering becomes
isotropic and one number rules,
\[
f \to -a , \qquad \sigma \to 4\pi a^2 ,
\]
where $a = -\lim_{k\to0}\delta_0/k$ is the \emph{\omterm{thm:b3:scattering-theory:partial}{scattering
length}}. Whatever its inner complexity, a target at low energy
\emph{is} its \omterm{thm:b3:scattering-theory:partial}{scattering length} --- the great simplification on
which cold-atom physics runs.
\end{theorem}
\admitted

\begin{example}[Neutron meets proton]\label{ex:b3:scattering-theory:np}
Slow neutrons scatter off protons with $\sigma \approx
\qty{20.4}{b}$ --- an area forty times the deuteron's geometric
size. The explanation: the low-energy collision is pure $s$ wave
with two spin channels, whose measured \omterm{thm:b3:scattering-theory:partial}{scattering lengths},
$a_{\text{t}} = \qty{5.4}{fm}$ (triplet) and $a_{\text{s}} =
\qty{-23.7}{fm}$ (singlet), are both far larger than the
\qty{2}{fm} force range. Large $|a|$ means a state \emph{almost}
at zero energy: the triplet's barely bound deuteron
(\cref{ex:b3:schrodinger-three-dimensions:deuteron}), and in the
singlet a ``virtual'' partner that just fails to bind --- its
failure written in $a_{\text{s}} < 0$. A twenty-barn puzzle,
solved by two nearly-there bound states
(\cref{exo:b3:scattering-theory:7}).
\end{example}

\section{Resonances}

\begin{proposition}[Breit--Wigner resonances]\label{prop:b3:scattering-theory:resonance}
\index{resonance!scattering}\index{Breit--Wigner formula}
If the projectile can be temporarily captured into a quasi-bound
state of energy $E_{\text{R}}$ and lifetime $\tau = \hbar/\Gamma$,
the corresponding \omterm{thm:b3:scattering-theory:partial}{phase shift} sweeps through $\pi/2$ and the
partial \omterm{def:b3:scattering-theory:cross-section}{cross-section} spikes:
\[
\sigma_l(E) \approx \frac{4\pi}{k^2}\,(2l+1)\,
\frac{\Gamma^2/4}{(E - E_{\text{R}})^2 + \Gamma^2/4} ,
\]
a Lorentzian peak of width $\Gamma$ at the ceiling the wave can
reach (the ``unitarity limit'' $4\pi(2l+1)/k^2$). Read backwards, a
resonance in a \omterm{def:b3:scattering-theory:cross-section}{cross-section} \emph{is} the discovery of a state:
its position is an energy level, its width a lifetime. Most of the
particle zoo of \cref{ch:b3:particle-physics} was discovered as
exactly such bumps.
\end{proposition}
\admitted

\begin{method}[Reading a scattering problem]\label{met:b3:scattering-theory:method}
(1) Kinematics first: $k$, and the accessible $q$-range $0 \le q
\le 2k$ --- what structure scales are visible? (2) Fast and weak:
Born --- Fourier-transform the potential; check the smallness. (3)
Slow: \omterm{thm:b3:scattering-theory:partial}{partial waves}, usually $s$ only; think \omterm{thm:b3:scattering-theory:partial}{scattering length},
and expect $\sigma = 4\pi a^2$ (bosonic \omterm{def:b3:identical-particles:exchange}{identical particles}:
$8\pi a^2$). (4) Peaks in $\sigma(E)$: fit Breit--Wigner, report a
level and a lifetime; dips: a phase passing $\pi$. (5) Extended
targets: multiply the point amplitude by the form factor --- the
Fourier transform of the density
(\cref{exo:b3:scattering-theory:12}).
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:1}
A \qty{1}{\micro A} proton beam ($\num{6.2e12}$ protons/s) of
\qty{1}{cm^2} section crosses a \qty{1}{\micro m} gold foil ($n =
\qty{5.9e28}{m^{-3}}$). (a) The areal target density. (b) With
$\sigma = \qty{10}{b}$ per nucleus, what fraction of the beam
scatters? (c) The rate into a detector of solid angle
\qty{e-3}{sr} if scattering were isotropic. (d) Why is
``\omterm{def:b3:scattering-theory:cross-section}{cross-section}'' a well-defined property of the \emph{pair}
(projectile, target, energy) rather than of the target alone?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:2}
Classical hard spheres of radii $R_1, R_2$: (a) show $\sigma =
\pi(R_1 + R_2)^2$. (b) For air molecules ($R \approx
\qty{0.15}{nm}$, $n = \qty{2.5e25}{m^{-3}}$), the mean free path
$1/n\sigma$: evaluate, and compare with the kinetic-theory value
of the Year 1 volume. (c) Why does a classical hard sphere show no
angular structure ($\dd\sigma/\dd\Omega$ constant --- accept or
derive)? (d) Which feature of \emph{quantum} hard-sphere
scattering (\cref{exo:b3:scattering-theory:6}) has no classical
counterpart?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:3}
\omterm{thm:b3:scattering-theory:born}{Momentum transfer}. (a) Draw the triangle of $\vect k$, $\vect k'$
and derive $q = 2k\sin(\theta/2)$. (b) For \qty{5.5}{MeV} alphas
($k = \qty{1.0e15}{m^{-1}}$): the range of $q$ over $\theta \in
[0, \pi]$, and the smallest resolvable structure $\sim 1/q_{\max}$.
(c) For \qty{500}{MeV} electrons ($k \approx E/\hbar c$): the same.
(d) Read off the moral connecting beam energy to microscope power.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:4}
From \cref{thm:b3:scattering-theory:amplitude}: (a) check that
$|f|^2$ has the dimensions of an area per solid angle; (b) show
the scattered flux through a sphere is $\Phi\sigma$; (c) why must
$f$ generally be complex (which conservation law does its phase
guard)? (d) State the optical theorem $\sigma_{\text{tot}} =
(4\pi/k)\operatorname{Im}f(0)$ and its meaning (the forward wave
must be depleted by exactly what scatters away).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:5}
Born for Yukawa. With $V(r) = \dfrac{g\,\eu^{-\mu r}}{r}$: (a)
carry out the Born integral (angular part first, then $\int_0^
\infty\sin(qr)\eu^{-\mu r}\dd r$) to get $f = -\dfrac{2mg}
{\hbar^2(q^2 + \mu^2)}$. (b) Let $\mu \to 0$ with $g =
Q_1Q_2/4\pi\varepsilon_0$ and recover Rutherford. (c) Why does the
\emph{total} Rutherford \omterm{def:b3:scattering-theory:cross-section}{cross-section} diverge, and which physical
ingredient (screening by atomic electrons) restores a finite
answer? (d) In particle physics the Yukawa form with $\mu =
m_\pi c/\hbar$ models the nuclear force: compute its range
$1/\mu$ for $m_\pi c^2 = \qty{140}{MeV}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:6}
The quantum hard sphere (radius $a$), $s$ wave: outside, $u_0 =
\sin(kr + \delta_0)$ with $u_0(a) = 0$. (a) Show $\delta_0 = -ka$.
(b) Show $\sigma \to 4\pi a^2$ as $k \to 0$: \emph{four} times the
geometric shadow. (c) Give the wave explanation of the factor
(diffraction has no sharp shadow at long wavelength). (d) At high
energy the answer tends to $2\pi a^2$ --- still twice geometric
(shadow diffraction): why never simply $\pi a^2$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:7}
The twenty barns of hydrogen. Slow unpolarised neutrons on
protons sample the triplet channel with weight $3/4$ and the
singlet with $1/4$: $\sigma = \pi(3a_{\text{t}}^2 +
a_{\text{s}}^2)$. (a) Evaluate with $a_{\text{t}} = \qty{5.4}{fm}$,
$a_{\text{s}} = \qty{-23.7}{fm}$ and compare with the measured
\qty{20.4}{b}. (b) Which channel dominates, despite its smaller
statistical weight? (c) Relate $a_{\text{t}}$'s size and sign to
the deuteron's near-zero binding; what does $a_{\text{s}} < 0$
say about the singlet system? (d) Why is water such an effective
moderator of reactor neutrons --- and why does this \omterm{def:b3:scattering-theory:cross-section}{cross-section}
matter for it?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:8}
Ramsauer--Townsend. Model the argon atom, for an incoming
electron, as an attractive square well of range $R =
\qty{0.2}{nm}$. (a) Inside, the wave number is $K =
\sqrt{2m(E + V_0)}/\hbar$: matching can make the outside wave
emerge with $\delta_0 = \pi$ exactly --- what is then the $s$-wave
\omterm{def:b3:scattering-theory:cross-section}{cross-section}? (b) Why does the atom become nearly invisible at
that energy although the well is strong? (c) Estimate the $V_0$
that puts the transparency near $E = \qty{1}{eV}$ (make the well
hold half a wavelength: $KR \approx \pi$). (d) Why do helium and
neon \emph{not} show the effect at comparable energies (their
wells are too shallow for $\delta_0$ to reach $\pi$) --- and what
does the effect's mere existence prove about matter waves?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:9}
For a pure $s$-wave amplitude $f = \eu^{\iu\delta_0}\sin\delta_0/
k$: (a) compute $\sigma$; (b) compute $(4\pi/k)\operatorname{Im}
f(0)$ and verify the optical theorem; (c) show the maximal
$s$-wave \omterm{def:b3:scattering-theory:cross-section}{cross-section} is $4\pi/k^2$ (the unitarity limit) and
evaluate it for thermal neutrons ($k = \qty{3.1e10}{m^{-1}}$) in
barns; (d) why can a resonance never push $\sigma_0$ beyond that
ceiling, however strong the interaction?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:10}
A neutron resonance. Uranium-238 shows a famous resonance for
neutrons at $E_{\text{R}} = \qty{6.67}{eV}$ with total width
$\Gamma = \qty{27}{meV}$. (a) Compute the resonance's lifetime.
(b) Compute $4\pi/k^2$ at that energy, in barns, and compare with
the measured peak of $\sim\qty{2e4}{b}$ (the difference is the
branching factor between elastic and capture channels ---
comment). (c) The width in temperature units: why does Doppler
broadening of this resonance with fuel temperature matter for
reactor stability (which sign of feedback)? (d) In two sentences:
how do such resonances make $^{238}$U a neutron absorber in the
epithermal range, and why is that central to reactor design.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:11}
Cold atoms live on one number. (a) For identical bosons the
low-energy \omterm{def:b3:scattering-theory:cross-section}{cross-section} is $8\pi a^2$ (constructive exchange
interference --- accept): evaluate for rubidium with $a =
100\,a_0$. (b) At $T = \qty{100}{nK}$, check that $ka \ll 1$
(compute $k$ from $k_{\text{B}}T = \hbar^2k^2/2m$, $m =
\qty{1.4e-25}{kg}$): the gas genuinely forgets everything but $a$.
(c) Near a ``Feshbach'' resonance a magnetic field drags a
molecular level through zero energy and $a$ diverges and changes
sign, exactly as in \cref{ex:b3:scattering-theory:np}: what
becomes of the gas's interactions at will? (d) Why is this
tunability --- interaction strength on a dial --- a physicist's
dream instrument (name one use: making molecules, simulating
strongly-coupled matter, collapsing condensates)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:12}
Form factors: weighing charge clouds. For an extended charge
density $\rho(\vect r)$, the Born amplitude multiplies the point
answer by $F(q) = \int\rho(\vect r)\eu^{\iu\vect q\cdot\vect r}
\dd^3r/Q$. (a) Show $F(0) = 1$. (b) Expand for small $q$: $F
\approx 1 - q^2\langle r^2\rangle/6$: measuring the low-$q$
fall-off weighs $\langle r^2\rangle$. (c) Electron--proton
scattering fits $\sqrt{\langle r^2\rangle} \approx \qty{0.84}{fm}$:
which independent measurement from
\cref{pb:b3:hydrogen-atom:1} (muonic hydrogen) cross-checks it,
and why was their historical disagreement (``the proton radius
puzzle'') taken so seriously? (d) At $q \gg 1/r_{\text{p}}$ the
elastic form factor collapses but hard scattering persists off
\emph{point-like constituents}: state in one sentence what 1968's
deep-inelastic version of Rutherford's experiment found inside
the proton.
\end{exercise}

\section{Problem: The flash counters who found the nucleus}

\begin{problem}[{Weekend problem --- Rutherford scattering, from
geometry to quarks}]\label{pb:b3:scattering-theory:1}
In 1909 Geiger and Marsden sat in the dark counting scintillation
flashes: alpha particles fired through gold foil, a few bouncing
nearly backwards --- ``as if a shell had rebounded off tissue
paper''. Rutherford's 1911 analysis of those counts created the
nuclear atom. This problem rebuilds it. Data: alpha energy $E =
\qty{5.5}{MeV}$, charge $z = 2$; gold $Z = 79$, foil thickness $t
= \qty{0.4}{\micro m}$, $n = \qty{5.9e28}{atoms/m^3}$; $k_C =
1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$.

\medskip\noindent\textbf{Part I --- One alpha, one nucleus.}
\begin{enumerate}
  \item In the plum-pudding picture (charge spread over the atom,
        \qty{e-10}{m}), estimate the maximum deflection a single
        atom could give a \qty{5.5}{MeV} alpha (field of a smeared
        sphere: $\theta \sim$ force $\times$ time / momentum
        $\sim 2Zze^2k_C/(E\cdot\qty{e-10}{m})$ in radians). Could
        any accumulation of such kicks send alphas backwards?
  \item Now let the charge be a point. Write the distance of
        closest approach $d_0$ for a head-on collision (all
        kinetic energy into Coulomb energy) and evaluate it.
  \item What does a measurable rate of near-backward scattering
        therefore \emph{immediately} imply about how concentrated
        the atom's positive charge is?
  \item For a general impact parameter $b$, the classical orbit
        gives $\tan(\theta/2) = d_0/2b$ (admitted --- the
        hyperbola of the Year 1 volume's gravity chapter, with
        repulsion): check its limits at $b \to 0$ and $b \to
        \infty$.
  \item Compute the impact parameter that scatters an alpha by
        more than $90^\circ$, and the fraction of alphas passing
        the foil that come within it of some nucleus ($n t\,\pi
        b^2$).
  \item Geiger and Marsden found about 1 in 8000 alphas deflected
        beyond $90^\circ$: compare.
\end{enumerate}

\medskip\noindent\textbf{Part II --- The \omterm{def:b3:scattering-theory:cross-section}{cross-section}.}
\begin{enumerate}[resume]
  \item Alphas with impact parameter in $[b, b + \dd b]$ scatter
        into $[\theta, \theta + \dd\theta]$: from $\dd\sigma =
        2\pi b\,|\dd b|$ and the orbit relation, derive
        \[
        \frac{\dd\sigma}{\dd\Omega} =
        \Big(\frac{d_0}{4}\Big)^2\frac{1}{\sin^4(\theta/2)} .
        \]
  \item Verify this matches the Born/quantum answer of
        \cref{ex:b3:scattering-theory:rutherford} (rewrite $d_0$
        in terms of $E$).
  \item Evaluate $\dd\sigma/\dd\Omega$ at $\theta = 30^\circ$,
        $60^\circ$, $120^\circ$ (in barns per steradian).
  \item Geiger and Marsden's 1913 counts at fixed geometry scale
        as $1{:}\,\sin^{-4}(\theta/2)$: compute the predicted
        count ratio between $30^\circ$ and $120^\circ$ and
        note that
        their data followed such ratios over five orders of
        magnitude of rate.
  \item Why does the same formula's $Z^2$ let the experiment
        \emph{weigh} the nuclear charge --- and how did such fits
        help pin $Z$ (gold: 79) as the atomic number?
  \item The $1/E^2$: what happens to the whole angular pattern
        when the alpha energy is doubled?
\end{enumerate}

\medskip\noindent\textbf{Part III --- The size of the nucleus.}
\begin{enumerate}[resume]
  \item Closest approach at angle $\theta$: $d(\theta) =
        \tfrac{d_0}2[1 + 1/\sin(\theta/2)]$ (admitted). Evaluate
        for $\theta = 180^\circ$ and \qty{60}{\degree}.
  \item As long as $d(\theta)$ exceeds the nuclear radius, the
        point-charge formula holds: what upper bound on the gold
        nucleus's size did Rutherford's verified $180^\circ$
        points establish?
  \item With higher-energy projectiles the counts at large angle
        \emph{fall below} Rutherford: what does the departure
        signal (which new force, at which distance)?
  \item Modern electron scattering gives nuclear radii $R
        \approx r_0A^{1/3}$ with $r_0 = \qty{1.2}{fm}$: evaluate
        for gold ($A = 197$) and compare with your bound.
  \item The $A^{1/3}$: what does it say about nuclear matter's
        density (constant? growing?) --- a fact
        \cref{ch:b3:nuclear-physics} will build on.
  \item Why did the (charge-blind) plum-pudding model die from
        \emph{this} experiment rather than from spectroscopy?
\end{enumerate}

\medskip\noindent\textbf{Part IV --- Rutherford's grandchildren.}
\begin{enumerate}[resume]
  \item List the exact translation table between 1911 and 1968:
        alpha $\to$ electron beam, gold atom $\to$ proton,
        nucleus $\to$ quarks --- what played the role of the
        ``unexpected large-angle events'' at SLAC?
  \item Why are \emph{electrons} the cleaner probe (what
        complication of alpha--nucleus scattering do they not
        have)?
  \item In Born language: elastic scattering at high $q$ measures
        the form factor's collapse (a soft cloud), while
        deep-inelastic rates stayed large and quasi-point-like:
        state the conclusion drawn.
  \item A century's arc in three rows: give (probe energy,
        distance resolved) for 1911 alphas ($\sim\qty{e-14}{m}$),
        1968 SLAC (\qty{20}{GeV}, $\sim\qty{e-16}{m}$), and the
        LHC (TeV scale, $\sim\qty{e-19}{m}$) --- using
        the rule ``resolution $\sim \hbar c/(qc)$''.
  \item Neutrons, uncharged, scatter only off nuclei (and off
        magnetic moments): name two things neutron scattering
        therefore maps in materials that X-rays see poorly
        (light atoms such as hydrogen; magnetic order).
  \item Discovery \omterm{def:b3:scattering-theory:cross-section}{cross-sections} at the LHC are measured in
        \emph{femtobarns} ($\qty{e-43}{m^2}$): from the barn to
        the femtobarn is fifteen orders of magnitude --- what does
        that say about how rare the interesting collisions are,
        and why luminosity (delivered flux) is as precious as
        energy?
  \item Summarise the named result: a $\sin^{-4}(\theta/2)$ law,
        verified flash by flash, put $99.98\%$ of the atom's mass
        into a volume $10^{-14}$ of its size --- and the same
        experiment, repeated ever harder, has never stopped
        finding the next layer.
\end{enumerate}
\end{problem}
