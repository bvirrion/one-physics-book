\chapter{सूक्ष्मविहित समुदाय}\label{ch:b3:microcanonical-ensemble}

वायु के एक घन सेंटीमीटर में $2.5\times10^{19}$ अणु होते हैं। कोई
संगणक कभी उनके गति-समीकरण समाकलित नहीं करेगा, और कोई प्रयोग आरंभिक
शर्तें नहीं जुटा सकता --- फिर भी गैस अद्भुत सरलता के नियम मानती है,
और वर्ष 1 के खंड की ऊष्मागतिकी ने उन्हें बिना कभी किसी अणु का नाम लिए
वर्णित किया। यह अध्याय वही पुल बनाता है: \emph{सांख्यिकीय भौतिकी}, जो
ऊष्मा के नियम यांत्रिकी और ईमानदार गिनती से व्युत्पन्न करती है। पूरी
इमारत एक ही अभिगृहीत पर खड़ी है --- कोई विलगित तंत्र अपनी ऊर्जा के
अनुरूप हर \omterm{def:b3:microcanonical-ensemble:states}{सूक्ष्म अवस्था} में समान संभावना से मिलता है ---
जिसे \omterm{thm:b3:hamiltonian-mechanics:liouville}{ल्यूविल की प्रमेय}
(\cref{ch:b3:hamiltonian-mechanics}) वैध बनाती है और संख्याओं का
निरा आकार शक्तिशाली: जब विन्यास
$10^{10^{20}}$ की इकाइयों में गिने जाएँ, तब ``अत्यधिक संभावित'' और
``निश्चित'' में कोई भेद नहीं रह जाता, और प्रायिकता कठोर होकर नियम बन
जाती है। गिनती से ही निकलते हैं एन्ट्रॉपी, ताप, दाब और दूसरा नियम ---
और, अध्याय के अंत में, वह कारण भी कि रबड़ का छल्ला पीछे क्यों खींचता है।

\section{सूक्ष्म अवस्थाएँ, स्थूल अवस्थाएँ, और अभिगृहीत}

\begin{definition}[सूक्ष्म और स्थूल अवस्थाएँ]\label{def:b3:microcanonical-ensemble:states}
\index{सूक्ष्म अवस्था}\index{स्थूल अवस्था}\index{बहुलता}
कोई \emph{सूक्ष्म अवस्था} किसी तंत्र का पूरा सूक्ष्म विनिर्देश है ---
हर कण की हर क्वांटम संख्या (या, चिरसम्मत ढंग से, हर स्थिति और हर
संवेग, जो प्रति कण $h^3$ की कला-समष्टि कोठरियों में गिने जाते हैं,
\cref{prop:b3:schrodinger-three-dimensions:counting})।
कोई \emph{स्थूल अवस्था} वह है जो ऊष्मागतिकी देख सकती है: ऊर्जा $E$,
आयतन $V$, कण संख्या $N$, चुंबकन\dots\ और
\emph{बहुलता} $\Omega(E, V, N)$ उन सूक्ष्म अवस्थाओं की संख्या है
जो एक ही स्थूल अवस्था ओढ़े हुए हैं।
\end{definition}

\begin{theorem}[मूल अभिगृहीत]\label{thm:b3:microcanonical-ensemble:postulate}
\index{सूक्ष्मविहित समुदाय}\index{मूल अभिगृहीत}
संतुलन में कोई विलगित तंत्र अपनी $\Omega(E, V, N)$ सुलभ सूक्ष्म
अवस्थाओं में से प्रत्येक में समान प्रायिकता से पाया जाता है। सारी
संतुलन सांख्यिकीय भौतिकी इसी एक वाक्य से निकलती है।
उसके प्रमाणपत्र: \omterm{thm:b3:hamiltonian-mechanics:liouville}{ल्यूविल की प्रमेय} दिखाती है कि ऊर्जा-कोश पर एकसमान
वितरण ही वह है जिसे गतिकी बनाए रखती है --- कोई प्रवाह कला-समष्टि की
प्रायिकता को कभी गुच्छित नहीं करता --- और डेढ़ सदी के परिणाम कभी किसी
प्रयोग से असहमत नहीं हुए।
\end{theorem}
\admitted

\begin{example}[$N$ सिक्कों का एक चुंबक]\label{ex:b3:microcanonical-ensemble:spins}
$N$ स्वतंत्र चक्रण, हर एक ऊपर या नीचे: $2^N$ सूक्ष्म अवस्थाएँ। \omterm{def:b3:microcanonical-ensemble:states}{स्थूल
अवस्था} ``$n$ ऊपर'' की \omterm{def:b3:microcanonical-ensemble:states}{बहुलता} $\Omega(n) = \binom Nn$ है,
जो $n = N/2$ पर अत्यधिक शिखरित है और जिसकी सापेक्ष चौड़ाई
$\sim1/\sqrt N$ है। $N = 100$ के लिए: सब-ऊपर वाली अवस्था एक है; और
संतुलित \omterm{def:b3:microcanonical-ensemble:states}{स्थूल अवस्था} में $\num{1e29}$ हैं। $N = 10^{20}$ के लिए
ऊपर-अंश में $10^{-8}$ जितने विचलन भी
$\eu^{-10^4}$ जैसे गुणकों से दब जाते हैं: तंत्र अपने शिखर पर \emph{है},
और जिसे हम संतुलन कहते हैं वह वही \omterm{def:b3:microcanonical-ensemble:states}{स्थूल अवस्था} है जिसमें सबसे अधिक
सूक्ष्म अवस्थाएँ हैं।
\end{example}

\section{एन्ट्रॉपी एक गिनती है}

\begin{definition}[बोल्ट्ज़मान एन्ट्रॉपी]\label{def:b3:microcanonical-ensemble:entropy}
\index{एन्ट्रॉपी!सांख्यिकीय}
किसी \omterm{def:b3:microcanonical-ensemble:states}{स्थूल अवस्था} की \emph{सांख्यिकीय एन्ट्रॉपी} है
\[
S = k_{\text{B}}\ln\Omega , \qquad
k_{\text{B}} = \qty{1.38e-23}{J/K} .
\]
लघुगणक एन्ट्रॉपी को वहाँ \emph{योज्य} बना देता है जहाँ बहुलताएँ
\emph{गुणित} होती हैं: स्वतंत्र तंत्रों के लिए $\Omega = \Omega_1\Omega_2$
और $S = S_1 + S_2$; और बोल्ट्ज़मान का नियतांक इस गिनती को उन्हीं
केल्विन-और-जूल इकाइयों पर अंशांकित करता है जिन्हें ऊष्मागतिकी पहले ही
चुन चुकी थी। यह
$S$, जैसा हम सत्यापित करेंगे, वर्ष 1 के खंड के दूसरे नियम की
एन्ट्रॉपी है --- अब एक सूक्ष्म अर्थ के साथ: एन्ट्रॉपी लघुगणकीय
मुद्रा में यह नापती है कि वह \omterm{def:b3:microcanonical-ensemble:states}{स्थूल अवस्था} \emph{कितने ढंगों से}
साकार हो सकती है।
\end{definition}

\begin{proposition}[आदर्श गैस की एन्ट्रॉपी]\label{prop:b3:microcanonical-ensemble:gas}
\index{ज़ाकुर--टेट्रोड सूत्र}
किसी बक्से में $N$ सर्वसम परमाणुओं की क्वांटम अवस्थाएँ गिनने पर
(\cref{prop:b3:schrodinger-three-dimensions:counting}, \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{कला समष्टि} में
प्रति परमाणु $h^3$ की एक कोठरी, और सर्वसमता के लिए $N!$ से भाग ---
\cref{exo:b3:identical-particles:10}) मिलता है
\[
S = Nk_{\text{B}}\Big[\ln\Big(\frac VN\,\frac1{\lambda_T^3}\Big)
 + \frac52\Big] , \qquad
\lambda_T = \frac{h}{\sqrt{2\pi mk_{\text{B}}T}}
\]
(अर्थात् \emph{\omterm{prop:b3:microcanonical-ensemble:gas}{ज़ाकुर--टेट्रोड सूत्र}}; $\lambda_T$, तापीय दे
ब्रॉय तरंगदैर्घ्य, ताप $T$ पर किसी परमाणु के तरंग-पुंज का आकार है)।
दो उल्लेखनीय जाँचें: यह सूत्र \emph{केवल} $N!$ की कृपा से विस्तीर्ण
है; और उसका निरपेक्ष मान --- जिसमें $h$ भी है --- वास्तविक गैसों की
कैलोरीमितीय एन्ट्रॉपियों से मेल खाता है
(\cref{exo:b3:microcanonical-ensemble:11}): उन्नीसवीं सदी में मापी
गई चिरसम्मत ऊष्मागतिकी प्लांक का नियतांक बिना जाने ही जानती थी।
\end{proposition}
\admitted

\section{ताप गिनती का एक अवकलज है}

\begin{theorem}[संतुलन और ताप]\label{thm:b3:microcanonical-ensemble:temperature}
\index{ताप!सांख्यिकीय परिभाषा}
दो तंत्रों को ऊर्जा का विनिमय करने दीजिए, जहाँ कुल $E = E_1 + E_2$
स्थिर है। संयुक्त \omterm{def:b3:microcanonical-ensemble:states}{बहुलता} $\Omega_1(E_1)\,\Omega_2(E - E_1)$
उस विभाजन पर अधिकतम है --- और अत्यधिक रूप से --- जहाँ
\[
\frac{\partial S_1}{\partial E_1} =
\frac{\partial S_2}{\partial E_2} .
\]
परिभाषित कीजिए
\[
\frac1T = \frac{\partial S}{\partial E}\bigg|_{V,N} ,
\]
संतुलन का अर्थ है \emph{तापों की समता}, और ऊर्जा स्वतःस्फूर्त रूप से
उच्च $T$ से निम्न $T$ की ओर बहती है, क्योंकि वही दिशा कुल गिनती
बढ़ाती है। यही तर्क आयतन और कण-विनिमय पर लगाने से
$P/T = \partial S/\partial V$ और $-\mu/T
= \partial S/\partial N$ मिलते हैं: ऊष्मागतिकी की सारी गहन
राशियाँ गिनती के अवकलज हैं।
\end{theorem}

\begin{proof}
$\ln\Omega_1 + \ln\Omega_2$ को $E_1$ पर अधिकतम कीजिए: अचरत्व की
शर्त वही कथित समता है। तीखापन: शिखर की चौड़ाई
$\sim E/\sqrt N$ है (\cref{exo:b3:microcanonical-ensemble:7}), अतः
अधिकतम ही प्रेक्षित अवस्था है। यदि $\partial S_1/\partial E_1 >
\partial S_2/\partial E_2$, तो ऊर्जा को 1 के \emph{भीतर} ले जाना
$S_{\text{कु}}$ बढ़ाता है: ठंडा पिंड (जिसका $\partial S/\partial E$
बड़ा है) अवशोषित करता है --- अर्थात् ऊष्मा का गरम से ठंडे की ओर बहना
एक गिनती का कथन है। आदर्श गैस पर जाँच: $S \propto \tfrac32 Nk_{\text{B}}\ln E + \cdots$
देता है $1/T = \tfrac32 Nk_{\text{B}}/E$, अर्थात् $E = \tfrac32
Nk_{\text{B}}T$ --- वर्ष 1 के खंड का गति-सिद्धांत वाला परिणाम,
विशुद्ध गिनती से पुनः प्राप्त।
\end{proof}

\section{दूसरा नियम, और उससे भी अजीब बातें}

\begin{proposition}[अनुत्क्रमणीयता अंकगणित है]\label{prop:b3:microcanonical-ensemble:second-law}
\index{दूसरा नियम!सांख्यिकीय}
कोई \omterm{def:b3:lagrangian-mechanics:coordinates}{प्रतिबंध} हटाया जाना (कोई दीवार हटी, कोई वाल्व खुला) सुलभ गिनती
को केवल बढ़ा ही सकता है: $\Omega$ बढ़ता है, $S$ बढ़ता है --- अर्थात्
वर्ष 1 के खंड का दूसरा नियम, अब प्रायिकता के कथन के रूप में।
उलटाव वर्जित नहीं, पर \emph{छूट पर} हैं: $N$ अणुओं की कोई गैस फिर से
बाईं आधी में सिमट आए, इसकी प्रायिकता
$2^{-N}$ है, और वायु के एक घन सेंटीमीटर के लिए प्रतीक्षा-काल
ब्रह्मांड की आयु से $10^{19}$ अंकों वाले गुणक जितना अधिक है।
समय का तीर, इस स्तर पर, वही दिशा है जिसमें गिनतियाँ बढ़ती हैं।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
किसी \omterm{def:b3:lagrangian-mechanics:coordinates}{प्रतिबंध} को हटाने से पहले से सुलभ हर \omterm{def:b3:microcanonical-ensemble:states}{सूक्ष्म अवस्था}
सुलभ बनी रहती है और नई भी जुड़ जाती हैं: $\Omega$ घट नहीं सकता। मुक्त
प्रसार की गिनती: हर अणु अपना सुलभ आयतन दुगना कर लेता है,
$\Omega \to 2^N\Omega$, अतः $\Delta S = Nk_{\text{B}}\ln2$ ---
ठीक वही ऊष्मागतिकीय परिणाम जो समतापी मुक्त दुगुनेपन का है।
\end{proof}

\begin{example}[ऋणात्मक ताप]\label{ex:b3:microcanonical-ensemble:negative}
\index{ऋणात्मक ताप}
जिस तंत्र की ऊर्जा \emph{ऊपर से परिबद्ध} है --- किसी क्षेत्र में $N$
चक्रण, ऊर्जा $-N\epsilon$ से $+N\epsilon$ तक --- उसका $\Omega(E)$
पहले चढ़ता है, फिर \emph{गिरता} है: संतुलित बिंदु के आगे ऊर्जा जोड़ना
गिनती घटा देता है। वहाँ $\partial S/\partial E < 0$: अर्थात् ताप
\emph{ऋणात्मक} है। ऐसी अवस्थाएँ होती हैं (जनसंख्या-प्रतिलोमन --- हर
लेज़र माध्यम की कार्यशील अवस्था --- और क्षेत्र-उलटाव से तैयार किए गए
चक्रण-तंत्र), और वे ठंडी नहीं बल्कि \emph{हर धनात्मक ताप से अधिक गरम}
हैं: ऊर्जा किसी भी $T < 0$ पिंड से किसी भी $T > 0$ पिंड में बहती है।
तापों का ईमानदार क्रम है $+0, \dots, +\infty \equiv -\infty, \dots, -0$:
जिसे $1/T$ स्वाभाविक ढंग से क्रमित करता है, उसे $T$ गड़बड़ा देता है
(\cref{exo:b3:microcanonical-ensemble:9})।
\end{example}

\begin{method}[सूक्ष्मविहित कारीगरी]\label{met:b3:microcanonical-ensemble:method}
(1) सूचीबद्ध कीजिए: यहाँ \omterm{def:b3:microcanonical-ensemble:states}{सूक्ष्म अवस्था} क्या है, और \omterm{def:b3:microcanonical-ensemble:states}{स्थूल अवस्था}
क्या नियत करती है? (2) $\Omega$ गिनिए --- विविक्त इकाइयों के लिए
संचय-गणित, और गैसों के लिए $h^{3N}N!$ पर कला-समष्टि का आयतन। (3)
लघुगणक जल्दी लीजिए: स्टर्लिंग ($\ln N! \approx N\ln N - N$) गुणनफलों
को साधने योग्य योगों में बदल देता है। (4) $S$ का अवकलन कीजिए: $T$ के
लिए ऊर्जा-अवकलज, $P$ के लिए आयतन, $\mu$ के लिए संख्या; और तनावों तथा
क्षेत्रों के लिए लंबाई या चुंबकन के अवकलज
(\cref{pb:b3:microcanonical-ensemble:1})। (5) शिखर पर भरोसा कीजिए:
उतार-चढ़ाव $1/\sqrt N$ से दबे हैं, अतः ``सबसे संभावित'' की जगह
``बराबर'' रखिए और किसी से क्षमा मत माँगिए।
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:microcanonical-ensemble:3}

\begin{exercise}[$\star$]\label{exo:b3:microcanonical-ensemble:4}
मुक्त प्रसार, जाँचा हुआ। एक मोल निर्वात में अपना आयतन दुगना करता है।
(क) गिनती के तर्क से $\Delta S$। (ख) गैस को फिर उसी आधे भाग में

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:5}
आदर्श-गैस की गिनती।
$\Omega \propto \dfrac{V^N}{N!\,h^{3N}}\times$ (संवेग-कोश का
आयतन $\propto E^{3N/2}$) से आरंभ करके: (क) दिखाइए कि $S = Nk_{\text{B}}[\ln V +
\tfrac32\ln E] - k_{\text{B}}\ln N! + $ नियतांक। (ख) दिखाइए कि
$N!$ के बिना तंत्र दुगना करने पर ($V$ और $N$ दोनों)
$S$ दुगना \emph{नहीं} होता --- अर्थात्
\cref{exo:b3:identical-particles:10} वाली विस्तीर्णता की विफलता। (ग)
स्टर्लिंग से परिणाम को इस रूप में रखिए: $S = Nk_{\text{B}}[\ln(V/N) + \tfrac32\ln(E/N)]
+ $ नियतांक$\cdot N$। (घ) कौन-से भौतिक निवेशों ने उन दो
``नियतांकों'' को तय किया जिन्हें इतिहास ने अनिर्धारित छोड़ा था
(क्वांटम कोठरी का आकार; कणों की सर्वसमता)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:6}
अवकलजों से ऊष्मागतिकी।
\cref{exo:b3:microcanonical-ensemble:5}(ग) के रूप से: (क) $1/T =
\partial S/\partial E$ निकालिए और $E = \tfrac32 Nk_{\text{B}}T$ पुनः पाइए;
(ख) $P/T = \partial S/\partial V$ निकालिए और $PV =
Nk_{\text{B}}T$ पुनः पाइए; (ग) \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{रुद्धोष्म निश्चर} व्युत्पन्न कीजिए: अचर
$S$ और $N$ पर दिखाइए कि $VE^{3/2}$ स्थिर है, अर्थात् $TV^{2/3}$ --- और
उसे वर्ष 1 के खंड के $PV^{5/3}$ से मिलाइए; (घ) बताइए कि क्या हासिल
हुआ: कौन-से आनुभविक नियम अभी-अभी गिनती की प्रमेय बन गए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:7}
संतुलन का तीखापन। आदर्श गैस के दो बराबर खंड कुल
ऊर्जा $E$ बाँटते हैं। (क) दिखाइए कि $\ln[\Omega_1\Omega_2]$ बराबर
बँटवारे पर अधिकतम है। (ख) असंतुलन $x =
(E_1 - E_2)/E$ में द्वितीय कोटि तक प्रसार कीजिए और दिखाइए कि प्रायिकता
$\propto\eu^{-3Nx^2/4}$ है (हर खंड की $\tfrac32 N$ स्वातंत्र्य
कोटियाँ --- गुणकों को ढीला रहने दीजिए)। (ग) $N = 10^{22}$ के लिए:
वर्ग-माध्य-मूल असंतुलन। (घ) \qty{300}{K} की गैस के लिए वह असंतुलन
कितना ताप-अंतर दर्शाता है: क्या कोई तापमापी उसे देख सकता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:8}
द्वि-स्तरीय (शॉट्की) ठोस: $N$ स्थल, प्रत्येक की ऊर्जा $0$ या
$\epsilon$; और $n$ उत्तेजित। (क) $\Omega(n)$ और $S(n)$। (ख) $E =
n\epsilon$ तथा स्टर्लिंग के साथ व्युत्पन्न कीजिए
\[
\frac1T = \frac{k_{\text{B}}}{\epsilon}\,
\ln\frac{N - n}{n} ,
\qquad\text{अतः}\qquad
\frac nN = \frac{1}{\eu^{\epsilon/k_{\text{B}}T} + 1} .
\]
(ग) $E(T)$ और ऊष्माधारिता का रेखाचित्र बनाइए: $k_{\text{B}}T \sim \epsilon/2$
के पास एक उभार (\emph{शॉट्की विषमता}) ---
$C$ दोनों सिरों पर क्यों लुप्त हो जाता है? (घ) अधिभोग-सूत्र की यह
आकृति पहले कहाँ प्रकट हो चुकी है, और कहाँ लौटेगी
(\cref{ch:b3:quantum-statistics})?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:microcanonical-ensemble:10}

\section{समस्या: एन्ट्रॉपी वाली कमानी}

\begin{problem}[{सप्ताहांत समस्या --- रबड़ पीछे क्यों खींचता है}]\label{pb:b3:microcanonical-ensemble:1}
रबड़ का छल्ला तानिए और वह वापस सिमट आता है --- जबकि उसके अणुओं के
बंध मुश्किल से विरूपित होते हैं और उसकी आंतरिक ऊर्जा मुश्किल से
बदलती है। रबड़ का प्रत्यानयन बल न्यूनतम खोजती ऊर्जा नहीं, बल्कि
\emph{अधिकतम खोजती एन्ट्रॉपी} है: क्रियाविधि पर विशुद्ध गिनती की
पहली महान विजय, और वही गणित आज एकल-अणु DNA प्रयोगों में चलता है।
प्रतिरूप: $N$ दृढ़ कड़ियों की एक शृंखला,
जिनमें हर कड़ी की लंबाई $b$ है और हर कड़ी खिंचाव-अक्ष के अनुदिश बाएँ
या दाएँ इशारा करती है; और सिरे-से-सिरे विस्तार $x = (n_+ - n_-)\,b$।

\medskip\noindent\textbf{भाग I --- विन्यास गिनना।}
\begin{enumerate}
  \item $n_\pm$ को $N$ और $x$ के पदों में व्यक्त कीजिए, और
        \omterm{def:b3:microcanonical-ensemble:states}{बहुलता} $\Omega(x)$ लिखिए।
  \item $\Omega$ कहाँ अधिकतम है, और वह किसी मुक्त शृंखला के पसंदीदा
        विस्तार के बारे में क्या कहता है?
  \item स्टर्लिंग से, $x \ll Nb$ के लिए दिखाइए:
        \[
        S(x) \approx S(0) - \frac{k_{\text{B}}x^2}{2Nb^2} .
        \]
  \item शृंखला की ऊर्जा (इस प्रतिरूप में) $x$ से स्वतंत्र है:
        किसी भी प्रत्यानयन बल को ``एन्ट्रॉपिक'' कहना उचित ठहराइए।
  \item मुक्त शृंखला का वर्ग-माध्य-मूल विस्तार $b\sqrt N$ है
        (यादृच्छिक चाल): $N = 10^4$ और $b = \qty{0.5}{nm}$ के लिए
        कुंडली के आकार की तनी हुई लंबाई $Nb$ से तुलना कीजिए।
  \item असली रबड़ ऐसी ही शृंखलाओं का जाल है, जो आड़े-बंधों के बीच
        फैला रहता है: वल्कनीकरण (अधिक आड़े-बंध) प्रतिरूप का कौन-सा
        एक प्राचल बदलता है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- गिनती का बल।}
\begin{enumerate}[resume]
  \item ताप $T$ पर रखे किसी तंत्र के लिए, विस्तार $x$ बनाए रखने
        को आवश्यक तनाव $f = -T\,\partial S/\partial x$ है
        (अचर ऊर्जा पर $\dd E = T\dd S + f\dd x$ से इसे मान
        लीजिए): व्युत्पन्न कीजिए
        \[
        f = \frac{k_{\text{B}}T\,x}{Nb^2} .
        \]
  \item \omterm{thm:b3:continuum-elasticity:hooke}{हुक का नियम} कमानी नियतांक $k =
        k_{\text{B}}T/Nb^2$ के साथ उभर आया है: \qty{300}{K} पर $N =
        \num{e4}$, $b = \qty{0.5}{nm}$ वाली एक शृंखला के लिए उसका मान निकालिए।
  \item चौंकाने वाला गुणक है $T$: स्थिर भार के नीचे रखे किसी रबड़
        के छल्ले को \emph{गरम} करने पर क्या करना चाहिए? इस्पात की
        कमानी से इसका विरोधाभास दिखाइए।
  \item किसी एक शृंखला को उसकी पूरी लंबाई की आधी तक तानने का बल
        आकलिए, और वह बल-पैमाना $k_{\text{B}}T/b$ भी, जिस पर रैखिक
        प्रतिरूप विफल हो जाता है।
  \item \qty{1}{mm^2} \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} वाले किसी रबड़ के छल्ले में
        समांतर में $\sim\num{e14}$ प्रभावी शृंखलाएँ होती हैं: उसका
        कमानी नियतांक आकलिए और अनुभव से तुलिए।
  \item ग्राम-दर-ग्राम, रबड़ ताप के साथ कड़ा (नरम नहीं) क्यों होता
        है, जबकि धातुएँ नरम पड़ जाती हैं?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- गफ़--जूल की रसोई।}
\begin{enumerate}[resume]
  \item किसी रबड़ के छल्ले को अपने होंठ से लगाकर तेज़ी से (रुद्धोष्म
        ढंग से) तानिए: वह गरम हो जाता है। एन्ट्रॉपी के बजट से
        समझाइए: तानना विन्यास-एन्ट्रॉपी \emph{घटाता} है, अतः कुल
        स्थिर रहते हुए एन्ट्रॉपी (ऊष्मा) को कहाँ जाना चाहिए?
  \item उसे तेज़ी से छोड़िए: वह ठंडा हो जाता है। सममित तर्क पूरा
        कीजिए।
  \item किसी भार लटके रबड़ के छल्ले को धीरे-धीरे गरम किया जाता है
        (बाल सुखाने वाला यंत्र): भार किस ओर चलता है, और यह
        एन्ट्रॉपिक प्रत्यास्थता का स्वच्छ चिह्न क्यों है?
  \item इस्पात की कमानी से प्रति-प्रयोग गढ़िए: इसके बदले क्या होता
        है, और क्यों (किस प्रकार की प्रत्यास्थता)?
  \item एक इंजन: रबड़ की तीलियों वाला कोई पहिया, जिसे एक ओर से गरम
        किया जाए, लगातार घूमता है। एक चक्र का तर्क बताइए (गरम
        तीलियाँ सिकुड़ती हैं और पहिये का संतुलन बिगाड़ देती हैं)।
  \item वही $f = k_{\text{B}}Tx/Nb^2$ नियम, जो एकल DNA अणुओं पर
        प्रकाशिक चिमटियों से नापा गया (किसी जीवाणु-जीनोम खंड के लिए
        $b \approx
        \qty{100}{nm}$, $N \approx 500$), किस परास के बल देता है?
        ($k_{\text{B}}T/b$ निकालिए।) इससे यह प्रतिरूप एक ही अणु पर
        \emph{सीधे} परखने योग्य क्यों बन गया?
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- सार।}
\begin{enumerate}[resume]
  \item शृंखला-प्रतिरूप में न कोई अन्योन्यक्रिया है, न कोई
        ऊर्जा-पैमाना, फिर भी वह एक बल-नियम देता है: इस अध्याय की
        सूक्ष्मविहित भाषा में ठीक-ठीक बताइए कि वह बल ``कहाँ से
        आता'' है।
  \item $T = 0$ पर वह बल क्यों लुप्त हो जाता है --- और यह किसी ऐसे
        संसार में प्रत्यास्थता के स्वभाव के बारे में क्या कहता है
        जिसमें तापीय हलचल ही न हो?
  \item एन्ट्रॉपिक कमानी नियतांक की $T$-रैखिकता की तुलना आदर्श गैस
        के दाब की $T$-रैखिकता से कीजिए: दिखाइए कि दोनों एक ही
        परिघटना हैं (किसी \omterm{def:b3:lagrangian-mechanics:coordinates}{प्रतिबंध} के सामने विन्यास गिनना), बस
        अलग-अलग वेश में।
  \item गरम किया गया छल्ला भार उठाता है और सचमुच कार्य करता है:
        ऊर्जा का स्रोत और मार्ग बताइए (सुखाने वाले यंत्र से भीतर
        ऊष्मा, एन्ट्रॉपी का हिसाब, बाहर कार्य) और पुष्टि कीजिए कि
        पहला नियम अछूता है।
  \item आदर्श शृंखला स्व-परिहार और कड़ी-मुड़ाई की ऊर्जा दोनों की
        उपेक्षा करती है: असली रबड़ का एक-एक मापा गया लक्षण बताइए
        जिसे हर चूक छिपा देती है (पूरे विस्तार के पास तीखा कड़ापन;
        विकृति-प्रेरित क्रिस्टलन और शैथिल्य)।
  \item प्रोटीन मुड़ते हैं, झिल्लियाँ थरथराती हैं, बहुलक कुंडलित
        होते हैं: शरीर के ताप पर $k_{\text{B}}T$ सारे कोमल और सजीव
        पदार्थ की स्वाभाविक बल-मुद्रा क्यों है
        (पिकोन्यूटन में $k_{\text{B}}T/\unit{nm}$ --- निकालिए)?
  \item नामित परिणाम का सार दीजिए: $10^4$ अंधी कड़ियों की एक
        शृंखला, ईमानदारी से गिनी हुई, \omterm{thm:b3:continuum-elasticity:hooke}{हुक का नियम} $k =
        k_{\text{B}}T/Nb^2$ के साथ देती है, और यह भविष्यवाणी करती
        है कि रबड़ तानने पर गरम होता है और गरम करने पर भार उठाता है
        --- अर्थात् एन्ट्रॉपी का बल की तरह काम करना, जो साइकिल की
        भीतरी ट्यूबों से लेकर $\sim\qty{0.04}{pN}$ पैमाने पर एकल
        DNA अणुओं तक सत्यापित है।
\end{enumerate}
\end{problem}
