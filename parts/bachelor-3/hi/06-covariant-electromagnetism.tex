\chapter{सहचर विद्युत्चुंबकत्व}\label{ch:b3:covariant-electromagnetism}

कोई ताँबे का तार दस ऐंपियर ढोता है। उसके इलेक्ट्रॉन प्रति सेकंड
मिलीमीटर के दसवें भाग जितना अपवाहित होते हैं --- घोंघे से भी धीमे ---
और तार अद्भुत यथार्थता तक विद्युत्-उदासीन रहता है। फिर भी उसके साथ-साथ
चलता कोई आवेश एक मापने योग्य खिंचाव अनुभव करता है। उसी आवेश के अपने
तंत्र से देखने पर ``चुंबकीय'' व्याख्या भाप बन जाती है: आवेश विराम में
है, और विराम में रखा आवेश कोई चुंबकीय बल अनुभव करता ही नहीं। उस तंत्र
में उसे जो खींचता है, वह एक \emph{विद्युत्} क्षेत्र है --- जिसे स्वयं
आपेक्षिकता ने बुला लिया, क्योंकि लंबाई-संकुचन तार के आवेश की दोनों
धाराओं को $10^{24}$ में एक भाग जितना असंतुलित कर देता है।
चुंबकत्व वही विद्युत् है, जिसे किसी चलती सीट से देखा गया हो। यह अध्याय
वर्ष 2 के खंड के विद्युत्चुंबकत्व को पिछले तीन अध्यायों की भाषा में
फिर से लिखता है: आवेश और विभव के लिए \omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश}, एक प्रतिसममित
प्रदिश जो $\vect E$ और $\vect B$ को साथ बाँधे रखता है, और मैक्सवेल के
चार समीकरण दो पंक्तियों में सिमटते हुए, जो हर \omterm{thm:b3:relativistic-kinematics:postulates}{जड़त्वीय निर्देश तंत्र} में
एक-सी पढ़ी जाती हैं। सुंदरता से आगे, इसका लाभ काम की भौतिकी है: क्षेत्र
कैसे रूपांतरित होते हैं, किसी तेज़ आवेश का क्षेत्र चपटा होकर पूए-सा
क्यों हो जाता है, और किसी आपेक्षिकीय इलेक्ट्रॉन का प्रकाश आगे की ओर
अग्रदीप की तरह क्यों बहता है --- यही उन सिंक्रोट्रॉन प्रकाश-स्रोतों का
सिद्धांत है जो आज प्रोटीनों और बैटरियों का एक्स-किरण चित्र लेते हैं।

\section{आवेश और धारा के लिए चतुःसदिश}

\begin{definition}[चतुःसदिश]\label{def:b3:covariant-electromagnetism:fourvector}
\index{चतुःसदिश}
\emph{चतुःसदिश} एक चतुष्क $a^\mu = (a^0, \vect a)$ है, $\mu =
0, 1, 2, 3$, जिसके घटक किसी संवर्धन के अंतर्गत ठीक वैसे मिलते हैं जैसे
$(ct,
\vect r)$ के मिलते हैं:
\[
a'^0 = \gamma(a^0 - \beta a^1) , \qquad
a'^1 = \gamma(a^1 - \beta a^0) , \qquad
a'^{2,3} = a^{2,3}
\]
जब संवर्धन $\beta c$ पर $x$ के अनुदिश हो। किन्हीं दो चतुःसदिशों से
निश्चर अदिश गुणनफल $a\cdot b = a^0b^0 - \vect a\cdot\vect b$ मिलता है,
जो हर तंत्र में एक ही संख्या है --- वही ढाँचा जो $c^2t^2 - x^2$
और $E^2 - p^2c^2$ के पीछे है। स्थिति $(ct, \vect r)$ और \omterm{def:b3:relativistic-dynamics:four-momentum}{चतुःसंवेग}
$(E/c, \vect p)$, ये दो हमारे पास हैं; विद्युत्चुंबकत्व अब तीन और
देता है: धारा, विभव और तरंग-सदिश।
\end{definition}

\begin{proposition}[चतुःधारा और आवेश-संरक्षण]\label{prop:b3:covariant-electromagnetism:fourcurrent}
\index{चतुःधारा}
आवेश-घनत्व $\rho$ और धारा-घनत्व $\vect\jmath$ मिलकर चतुःधारा बनाते
हैं
\[
J^\mu = (\rho c,\ \vect\jmath\,) :
\]
निज घनत्व $\rho_0$ का कोई आवेश-बादल, जो $\vect u$ से चलता हो, उसका
$J^\mu = \rho_0\gamma_u(c, \vect u)$ होता है --- घनत्व $\gamma_u$ गुना
बढ़ जाता है, क्योंकि बादल का आयतन संकुचित होता है, जबकि आवेश स्वयं
निश्चर है (एक निर्णायक प्रायोगिक तथ्य: तेज़ भीतरी इलेक्ट्रॉनों वाले
परमाणु ठीक-ठीक उदासीन बने रहते हैं)। आवेश-संरक्षण यह निश्चर कथन है
\[
\frac{\partial(\rho c)}{\partial(ct)} +
\operatorname{div}\vect\jmath = 0 ,
\]
अर्थात् वर्ष 2 के खंड का सांतत्य समीकरण, जो अब दिखने में भी सभी
प्रेक्षकों के लिए एक ही नियम है।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
चलता बादल: निज आयतन $V_0$ का कोई संदूक आवेश $\rho_0
V_0$ रखता है; प्रयोगशाला में संदूक संकुचित होकर $V_0/\gamma_u$ रह जाता
है, अतः $\rho =
\gamma_u\rho_0$, और धारा-घनत्व की परिभाषा से
$\vect\jmath = \rho\vect u$। फिर $(\rho c, \vect\jmath)$ इसलिए
\omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} की तरह रूपांतरित होता है कि वह $\rho_0/c$ गुना
चतुर्वेग $\gamma_u(c, \vect u)$ है, जो रचना से ही \omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} है।
आवेश की निश्चरता एक प्रायोगिक निवेश है।
\end{proof}

\section{दोनों क्षेत्रों के लिए एक ही प्रदिश}

\begin{definition}[चतुःविभव और क्षेत्र प्रदिश]\label{def:b3:covariant-electromagnetism:tensor}
\index{चतुःविभव}\index{क्षेत्र प्रदिश}
वर्ष 2 के खंड के विभव मिलकर चतुःविभव बनाते हैं
\[
A^\mu = \Big(\frac{V}{c},\ \vect A\Big) ,
\]
और मापने योग्य क्षेत्र $\vect E = -\vect\nabla V -
\partial_t\vect A$, $\vect B = \operatorname{\vect{curl}}\vect A$ एक
ही प्रतिसममित सारणी के छह स्वतंत्र घटक हैं, अर्थात्
\emph{विद्युत्चुंबकीय क्षेत्र प्रदिश} के
\[
F^{\mu\nu} =
\begin{pmatrix}
0 & -E_x/c & -E_y/c & -E_z/c\\
E_x/c & 0 & -B_z & B_y\\
E_y/c & B_z & 0 & -B_x\\
E_z/c & -B_y & B_x & 0
\end{pmatrix} .
\]
$\vect E$ और $\vect B$ दो क्षेत्र नहीं, एक ही वस्तु के छह मुख हैं; और
किस मुख को कोई प्रेक्षक ``विद्युत्'' कहेगा, यह प्रेक्षक की गति पर
निर्भर करता है।
\end{definition}

\begin{theorem}[मैक्सवेल, सहचर रूप में]\label{thm:b3:covariant-electromagnetism:maxwell}
\index{मैक्सवेल के समीकरण!सहचर रूप}
वर्ष 2 के खंड के चारों मैक्सवेल समीकरण दो तंत्र-निरपेक्ष कथनों का
घटक-रूप हैं: स्रोत वाला युग्म
(गाउस तथा ऐंपियर--मैक्सवेल) है
\[
\sum_\mu \partial_\mu F^{\mu\nu} = \mu_0\,J^\nu ,
\]
और स्रोतहीन युग्म वह सर्वसमिका है जो सुनिश्चित करती है कि
$F$ किसी \omterm{def:b3:covariant-electromagnetism:tensor}{चतुःविभव} से व्युत्पन्न है। लोरेंत्स
बल $\dd p^\mu/\dd\tau = qF^{\mu\nu}u_\nu$ है --- बल-नियम,
चुंबकीय भाग सहित, एक ही पंक्ति में। चूँकि हर समीकरण के दोनों पक्ष
एक ही तरह रूपांतरित होते हैं, \emph{मैक्सवेल के सिद्धांत को आपेक्षिकीय
बनने के लिए किसी संशोधन की आवश्यकता नहीं}: वह आपेक्षिकता का पहला पूरा
हुआ टुकड़ा था, जो 1865 में जन्मा।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
$\nu = 0$ घटक का प्रसार कीजिए: $\sum_i\partial_i(E_i/c) =
\mu_0\rho c$, अर्थात् $\operatorname{div}\vect E = \rho/\varepsilon_0$
($c^2 = 1/\mu_0\varepsilon_0$ काम में लाते हुए)। $\nu = 1$ घटक
$-\partial_t(E_x/c^2) + (\partial_yB_z - \partial_zB_y)
= \mu_0 j_x$ बटोरता है: अर्थात् ऐंपियर--मैक्सवेल का $x$ घटक। शेष
घटक वही ढाँचा दोहराते हैं; और स्रोतहीन युग्म $A^\mu$ के आड़े-तिरछे
दूसरे अवकलजों की समता है (जाँच
\cref{exo:b3:covariant-electromagnetism:3} में)। $F^{\mu\nu}$
किसी (द्वि-सूचकांक) प्रदिश की तरह रूपांतरित होता है --- हर सूचकांक किसी
\omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} की तरह --- यह स्वीकृत है; उसके परिणाम अगली प्रतिज्ञप्ति हैं।
\end{proof}

\section{क्षेत्र किस तरह रूपांतरित होते हैं}

\begin{proposition}[क्षेत्र रूपांतरण]\label{prop:b3:covariant-electromagnetism:transform}
\index{क्षेत्र रूपांतरण}
वेग $\vect v = v\,\vect e_x$ पर संवर्धन के लिए: गति के \emph{अनुदिश}
घटक अछूते रहते हैं, और अनुप्रस्थ घटक आपस में मिल जाते हैं,
\[
E'_x = E_x , \qquad
E'_y = \gamma(E_y - vB_z) , \qquad
E'_z = \gamma(E_z + vB_y) ,
\]
\[
B'_x = B_x , \qquad
B'_y = \gamma\Big(B_y + \frac{v}{c^2}E_z\Big) , \qquad
B'_z = \gamma\Big(B_z - \frac{v}{c^2}E_y\Big) .
\]
संक्षेप में, संवर्धन के अनुप्रस्थ: $\vect E' = \gamma(\vect E +
\vect v\wedge\vect B)_\perp$ और $\vect B' = \gamma(\vect B - \vect
v\wedge\vect E/c^2)_\perp$। किसी एक तंत्र का शुद्ध $\vect B$ दूसरे में $\vect
E$ और $\vect B$ होता है: दोनों क्षेत्र एक ही परिघटना हैं।
\end{proposition}
\admitted

\begin{example}[उदासीन तार से चुंबकत्व]\label{ex:b3:covariant-electromagnetism:wire}
कोई उदासीन तार धारा ढोता है: धनात्मक जालक विराम में, इलेक्ट्रॉन
अपवाहित। साथ-साथ चलते किसी परीक्षण-आवेश के लिए उसके तंत्र में संवर्धन
कीजिए: दोनों आवेश-धाराएँ \emph{भिन्न-भिन्न} संकुचित होती हैं (उनके वेग
अलग हैं), तार पर कुल रेखीय आवेश $\lambda' =
-\gamma vI/c^2$ आ जाता है, और उसका त्रिज्य विद्युत् क्षेत्र आवेश को
ठीक उतने ही बल से खींचता है जिसे प्रयोगशाला ने $q\vect v\wedge\vect B$
कहा था। चुंबकत्व वही है जो आपेक्षिकता का द्वितीय-कोटि पद $v u/c^2$ तब
दिखता है जब प्रति मीटर
$10^{28}$ मूल आवेश मिलकर षड्यंत्र करें: हर प्रभाव अद्भुत रूप से छोटा
है, पर तार उससे भी अधिक यथार्थता तक उदासीन है, अतः वह नन्हा असंतुलन ही
पूरी कहानी है (\cref{pb:b3:covariant-electromagnetism:1})।
\end{example}

\begin{example}[तेज़ आवेश का चपटा क्षेत्र]\label{ex:b3:covariant-electromagnetism:pancake}
किसी आवेश के कूलॉम क्षेत्र को उस तंत्र में रूपांतरित कीजिए जिसमें वह
$\gamma \gg 1$ से चलता है: अनुदैर्घ्य क्षेत्र अपरिवर्तित रहता है, जबकि
अनुप्रस्थ क्षेत्र $\gamma$ गुना हो जाता है। कभी गोलीय रहा क्षेत्र
चपटा होकर आवेश से गुज़रते तल के चारों ओर $\sim 1/\gamma$ खुलाव-कोण की
चकती बन जाता है --- और गति के अनुप्रस्थ, उसके साथ एक चुंबकीय वलय
$B' = vE'/c^2$ भी। किसी स्थिर प्रेक्षक के लिए किसी LHC प्रोटॉन का
गुज़रना ($\gamma \approx 7000$) लगभग परस्पर लंबवत $\vect E$ और
$\vect B$ का एक उप-पिकोसेकंड थप्पड़ है: लगभग प्रकाश की एक स्पंद ---
और यही कारण है कि तेज़ पुंज पदार्थ से वैसे ही बात करते हैं जैसे फ़ोटॉन।
\end{example}

\section{निश्चर, प्रकाश और अग्रदीप प्रभाव}

\begin{proposition}[दो क्षेत्र-निश्चर]\label{prop:b3:covariant-electromagnetism:invariants}
\index{क्षेत्र निश्चर}
$F^{\mu\nu}$ से ठीक दो स्वतंत्र अदिश बनाए जा सकते हैं:
\[
E^2 - c^2B^2
\qquad\text{और}\qquad
\vect E\cdot\vect B ,
\]
जो हर \omterm{thm:b3:relativistic-kinematics:postulates}{जड़त्वीय निर्देश तंत्र} में एक ही हैं। परिणाम: यदि कहीं $E > cB$ हो
($\vect E\cdot\vect B = 0$ के साथ), तो कोई तंत्र वहाँ शुद्ध विद्युत्
क्षेत्र देखता है; यदि $E < cB$ हो, तो कोई तंत्र शुद्ध चुंबकीय देखता है;
और किसी समतल प्रकाश-तरंग के, जिसमें $E = cB$ और $\vect E\perp\vect B$
हो, \emph{दोनों} निश्चर शून्य हैं --- वह हर तंत्र में प्रकाश ही है:
कोई संवर्धन उसे स्थैतिक क्षेत्र नहीं बना सकता, केवल लाल-विस्थापित कर
सकता है। किसी प्रकाश-तरंग को पकड़ा नहीं जा सकता; आइंस्टाइन का किशोर
प्रश्न दो निश्चरों में अपना उत्तर आप दे देता है।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
\cref{prop:b3:covariant-electromagnetism:transform} के सीधे
प्रतिस्थापन से मानक संवर्धन के अंतर्गत निश्चरता जाँचिए
(\cref{exo:b3:covariant-electromagnetism:6}); कोई तीसरा
स्वतंत्र निश्चर नहीं है --- यह स्वीकृत है। तरंग के लिए: $E = cB$ और
लंबकोणता दोनों को शून्य कर देती हैं; और जिस तंत्र में स्थैतिक क्षेत्र
होता, उसमें कोई निश्चर शून्येतर चाहिए होता।
\end{proof}

\begin{proposition}[चतुःतरंग-सदिश, डॉप्लर और विपथन]\label{prop:b3:covariant-electromagnetism:wavevector}
\index{विपथन}\index{अग्रदीप प्रभाव}
किसी समतल तरंग की आवृत्ति और दिशा मिलकर वह अशून्य-रहित \omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} बनाती
हैं जो $k^\mu = (\omega/c, \vect k)$ है, $\|\vect k\| = \omega/c$।
उसे रूपांतरित करने पर एक साथ
\cref{ch:b3:relativistic-kinematics} का आपेक्षिकीय डॉप्लर सूत्र और
दिशाओं का \emph{विपथन} मिल जाते हैं:
\[
\cos\theta' = \frac{\cos\theta - \beta}{1 - \beta\cos\theta} .
\]
उलटा पढ़िए: जो स्रोत अपने विराम-तंत्र में समदैशिक विकिरण करता है, वह
प्रयोगशाला में अपना आधा प्रकाश आगे के शंकु $\theta
\lesssim 1/\gamma$ में भेज देता है: यही \emph{\omterm{prop:b3:covariant-electromagnetism:wavevector}{अग्रदीप प्रभाव}} है।
किसी संचयन-वलय में $\gamma \sim 10^4$ पर चक्कर काटता इलेक्ट्रॉन
वलय के चारों ओर एक्स-किरणों का
$\qty{0.1}{mrad}$ का खोजी-दीप घुमाता है --- वही सिंक्रोट्रॉन प्रकाश,
जो प्रोटीन-क्रिस्टलिकी की किरण-रेखाएँ भरता है।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
कला $\omega t - \vect k\cdot\vect r = k^\mu x_\mu$ घटनाओं से गुज़रते
तरंग-शिखर गिनती है --- अर्थात् एक निश्चर --- अतः $k^\mu$ को
\omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} की तरह ही रूपांतरित होना चाहिए। $(\omega/c, k\cos\theta,
k\sin\theta, 0)$ पर संवर्धन लगाइए: काल-घटक डॉप्लर देता है, और
दिक्-घटकों का अनुपात विपथन का सूत्र। $\theta' =
90^\circ$ रखने पर (अर्थात् स्रोत-तंत्र की बगल वाली किरण): $\cos\theta =
\beta$, अर्थात् $\theta \approx 1/\gamma$, जब $\gamma \gg 1$ हो ---
आधा गोला आगे के शंकु में मुड़ जाता है।
\end{proof}

\begin{method}[सहचर ढंग से काम करना]\label{met:b3:covariant-electromagnetism:method}
(1) खेल में लगे \omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} पहचानिए ($x^\mu$, $P^\mu$, $J^\mu$,
$A^\mu$, $k^\mu$) और घटकों के बजाय उनके निश्चर गुणनफलों को वरीयता
दीजिए। (2) क्षेत्र रूपांतरित करने के लिए उन्हें संवर्धन के अनुदिश और
अनुप्रस्थ घटकों में बाँटिए और
\cref{prop:b3:covariant-electromagnetism:transform} लगाइए। (3) पहले
और बाद में दोनों निश्चर जाँचिए --- इस विषय का सबसे तेज़ त्रुटि-संसूचक।
(4) जब कोई चुंबकीय समस्या रहस्यमय लगे, तो आवेश के साथ सवार हो
जाइए: उसके तंत्र में केवल $\vect E'$ काम करता है। (5) मैक्सवेल पर
भरोसा रखिए: समीकरणों को कभी आपेक्षिकीय ``संशोधन'' नहीं चाहिए, केवल
आपेक्षिकीय पाठ।
\end{method}

निकालिए। (ग) विपथन का सूत्र निकालिए। (घ) तारों के प्रकाश का
विपथन: पृथ्वी \qty{29.8}{km/s} से परिक्रमा करती है; किसी तारे की
आभासी स्थिति एक वर्ष में कितने कोण पर घूम जाती है? (ब्रैडली ने 1728 में
$20.5''$ नापा --- यह पहला सीधा प्रमाण कि पृथ्वी चलती है।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:covariant-electromagnetism:9}
कोई आवेश $q$ अचर $\vect v = v\vect e_x$ से चलता है, $\gamma \gg
1$। (क) कूलॉम के क्षेत्र का रूपांतरण करके दिखाइए कि आवेश से गुज़रते
अनुप्रस्थ तल में क्षेत्र संवर्धित होकर $\gamma
q/4\pi\varepsilon_0b^2$ हो जाता है, दूरी $b$ पर, जबकि ठीक आगे और
पीछे वह $1/\gamma^2$ जितना कुचल जाता है। (ख) दिखाइए कि दूरी $b$ पर का
कोई स्थिर प्रेक्षक $\sim b/\gamma v$ अवधि की एक क्षेत्र-स्पंद देखता है। (ग) $b = \qty{1}{cm}$ पर गुज़रते किसी LHC प्रोटॉन के लिए:
शिखर क्षेत्र और स्पंद-अवधि। (घ) यह स्पंद ठीक किस अर्थ में ``लगभग
प्रकाश'' है? ($E' \approx cB'$ और निश्चर जाँचिए।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:covariant-electromagnetism:10}
किसी प्रकाश-तरंग का पीछा। किसी समतल तरंग के $\vect E = E_0\vect e_y
\cos(\omega(t - x/c))$, $\vect B = (E_0/c)\vect e_z\cos(\omega(t -
x/c))$ हैं। कोई प्रेक्षक उसका पीछा $\beta$ से करता है। (क) क्षेत्रों
का रूपांतरण कीजिए: दिखाइए कि $E'_0 = E_0\sqrt{(1-\beta)/(1+\beta)}$ और $B'_0 = E'_0/c$। (ख)
दिखाइए कि आवृत्ति भी उसी गुणक से रूपांतरित होती है: तरंग तरंग ही रहती
है, लाल-विस्थापित, और सदा $E' = cB'$ के साथ। (ग) तरंग के ऊर्जा-घनत्व
($E^2$ के अनुक्रमानुपाती) और फ़ोटॉनों की संख्या का क्या होता है? (घ)
निष्कर्ष निकालिए: ``किसी प्रकाश-पुंज के साथ-साथ सवारी'' के लिए क्या
चाहिए होता, और कौन-सा निश्चर उसे मना करता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:covariant-electromagnetism:11}
सिंक्रोट्रॉन प्रकाश और वृत्ताकार इलेक्ट्रॉन-मशीनों की मृत्यु। त्रिज्या
$r$ के वृत्त पर कोई अति-आपेक्षिकीय आवेश शक्ति
$P = q^2c\gamma^4/6\pi\varepsilon_0r^2$ विकिरित करता है (स्वीकृत ---
वर्ष 2 के खंड का लार्मर सूत्र, संवर्धित)। (क) दिखाइए कि प्रति चक्कर
खोई ऊर्जा $\Delta E = q^2\gamma^4/3\varepsilon_0r$ है। (ख) LEP:
$r = \qty{3.1}{km}$ पर \qty{100}{GeV} के इलेक्ट्रॉन: $\gamma$ और
प्रति चक्कर $\Delta E$ निकालिए --- हर चक्कर पर पुंज-ऊर्जा का कितना
अंश फिर से खरीदना पड़ता है? (ग) LHC: उसी सुरंग में \qty{6.8}{TeV} के
प्रोटॉन: प्रति चक्कर $\Delta E$ निकालिए और तुलना कीजिए। (घ) $\gamma^4 =
(E/mc^2)^4$ गुणक की सहायता से समझाइए कि इलेक्ट्रॉनों का उत्तराधिकारी
कोई \emph{रैखिक} संघट्टक या कहीं बड़ा वलय क्यों है, और प्रोटॉन-वलय
क्यों बचे रहते हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:covariant-electromagnetism:12}
आपेक्षिकीय साइक्लोट्रॉन। किसी एकसमान $\vect B$ में कोई आवेश,
आपेक्षिकीय चाल पर। (क) $\dd\vect p/\dd t = q\vect v\wedge\vect
B$ से, $E$ के अचर रहते, दिखाइए कि कक्षा एक वृत्त है, जिस पर
$\omega = qB/\gamma m$ से चला जाता है: साइक्लोट्रॉन-आवृत्ति ऊर्जा के
साथ घटती है।
(ख) कोई चिरसम्मत साइक्लोट्रॉन नियत $\omega_0 = qB/m$ पर धकेलता है:
दिखाइए कि $N$ चक्करों के बाद, जबकि $\gamma - 1$ रैखिक रूप से अपने
अंतिम मान $\Gamma$ तक बढ़े, संचित कला-सर्पण किसी RF आवर्तकाल के चौथाई
तक तब पहुँचता है जब $\Gamma \approx 1/2N$; $N = 100$ चक्करों के लिए वह
प्रोटॉन की कितनी गतिज ऊर्जा है, और चिरसम्मत साइक्लोट्रॉनों की
ऐतिहासिक अधिकतम सीमा (कोई \qty{20}{MeV}, अतिरिक्त गुंजाइश और वोल्टता
से खरीदी गई) से उसकी तुलना कैसी है? (ग) दो
इलाज हैं: आवृत्ति को ढालना (सिंक्रोसाइक्लोट्रॉन), या $B(r)$ को इस तरह
गढ़ना कि वह $\gamma$ की तरह बढ़े (समकालिक साइक्लोट्रॉन): हर एक को एक
वाक्य में समझाइए। (घ) PSI का समकालिक साइक्लोट्रॉन \qty{590}{MeV} के
प्रोटॉन देता है: उसका नेमि-क्षेत्र केंद्रीय क्षेत्र से कितने गुना
अधिक है?
\end{exercise}

\section{समस्या: घोंघे की चाल पर चुंबकत्व}

\begin{problem}[{सप्ताहांत समस्या --- $10^{-24}$ का असंतुलन हर मोटर
कैसे चलाता है}]\label{pb:b3:covariant-electromagnetism:1}
इलेक्ट्रॉन किसी लैंप की डोरी में शहद के रेंगने से भी धीमे अपवाहित होते
हैं, और उस अपवाह के लिए $v^2/c^2$ मात्र $10^{-24}$ है --- फिर भी उससे
बनता चुंबकीय बल कबाड़खानों में कारें उठा लेता है। यह समस्या किसी सीधे
तार और किसी चलते आवेश का पूरा दो-तंत्र हिसाब करती है, और हर
विद्युत्चुंबक में छिपी आपेक्षिकता ढूँढ़ निकालती है। आँकड़े: ताँबे का
तार, काट $S = \qty{1.0}{mm^2}$, धारा $I = \qty{10}{A}$,
चालन-इलेक्ट्रॉन घनत्व $n = \qty{8.5e28}{m^{-3}}$; परीक्षण-आवेश
$q = \qty{1}{\micro C}$, दूरी $d = \qty{1.0}{cm}$ पर, तार के समांतर
$v = \qty{e5}{m/s}$ से चलता हुआ (कोई आयन-पुंज चाल, ताकि संख्याएँ दिखती
रहें); $\mu_0 = 4\pi\times10^{-7}\,\unit{H/m}$।

\medskip\noindent\textbf{भाग I --- प्रयोगशाला-तंत्र का हिसाब।}
\begin{enumerate}
  \item इलेक्ट्रॉनों की अपवाह-चाल $u$, और $u^2/c^2$ निकालिए।
  \item तार का निदर्शन दो अध्यारोपित रेखीय आवेशों से कीजिए: जालक
        विराम में $+\lambda$, और इलेक्ट्रॉन $-\lambda$, जो
        $u$ से अपवाहित हों, तथा $I = \lambda u$। $\lambda$ निकालिए।
  \item आवेश की स्थिति पर $B$ और उस पर लगता चुंबकीय बल
        $F = qvB$ निकालिए --- दिशा सहित, जब $v$ पारंपरिक धारा
        $I$ के समांतर हो (स्मरण रहे: समांतर धाराएँ आकर्षित करती हैं)।
  \item प्रयोगशाला में कोई \emph{विद्युत्} बल क्यों नहीं है?
  \item यदि उसी तार पर प्रति मीटर केवल एक अतिरिक्त इलेक्ट्रॉन जितना
        कुल आवेश होता, तो उसका विद्युत् क्षेत्र $q$ पर जो बल लगाता,
        उसकी तुलना प्रश्न 3 के चुंबकीय बल से कीजिए,
        और निष्कर्ष निकालिए कि चुंबकत्व को श्रेय देने से पहले
        ``उदासीन'' को कितनी यथार्थता से नापना पड़ेगा।
  \item \qty{e5}{m/s} पर \qty{1}{\micro C} का आवेश: क्या प्रश्न 3 का
        बल मापने योग्य है? (रेत के एक कण के भार
        $\sim\qty{e-6}{N}$ से तुलना कीजिए।)
\end{enumerate}

\medskip\noindent\textbf{भाग II --- सीट बदलना।}
परीक्षण-आवेश के तंत्र में संवर्धन कीजिए (तार के अनुदिश चाल $v$)।
\begin{enumerate}[resume]
  \item उस तंत्र में जालक का और इलेक्ट्रॉनों का वेग क्या है
        (इलेक्ट्रॉन $I$ के विपरीत अपवाहित होते हैं, अतः इस संवर्धन में
        चाल पा जाते हैं --- उन्हें ठीक से संयोजित कीजिए)?
  \item तार की काट पर समाकलित करने पर प्रति एकांक लंबाई चतुःधारा
        $(c\lambda_{\text{कुल}}, I, 0, 0)$ है, जहाँ यहाँ
        $\lambda_{\text{कुल}} = 0$ है। उसका रूपांतरण कीजिए और दिखाइए
        कि नए तंत्र में तार का रेखीय आवेश यह है
        \[
        \lambda' = -\gamma_v\,\frac{vI}{c^2} .
        \]
  \item $\lambda'$ का मान निकालिए, प्रति मीटर कूलॉम में और प्रति मीटर
        इलेक्ट्रॉन-आवेशों में। कौन-सी धारा सघन हुई, और क्यों
        (चाहें तो हर धारा को अलग-अलग संकुचित कीजिए)?
  \item आवेश पर विद्युत् क्षेत्र $E' = \lambda'/2\pi\varepsilon_0d$
        और उस पर विद्युत् बल $qE'$ निकालिए।
  \item $qE'$ की तुलना भाग I के प्रयोगशाला वाले $qvB$ से कीजिए, और
        बचे हुए गुणक $\gamma_v \approx 1 + \num{5.6e-8}$ को समझाइए
        (किस \omterm{def:b3:covariant-electromagnetism:fourvector}{चतुःसदिश} का रूपांतरण-नियम इन बलों को जोड़ता है?)।
  \item इस तंत्र में एक चुंबकीय क्षेत्र भी है (तार अब भी धारा ढोता
        है): वह यहाँ कोई बल क्यों नहीं लगाता?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- संख्याओं का सार।}
\begin{enumerate}[resume]
  \item आपेक्षिक असंतुलन $|\lambda'|/\lambda$ निकालिए और उसे
        $\gamma_v vu/c^2$ के रूप में लिखिए।
  \item तार के आवेश के $10^{-19}$ कोटि का प्रभाव कोई स्थूल बल कैसे
        बना सकता है? (वह किस विशाल संख्या से गुणा होता है?)
  \item \qty{1}{cm} की दूरी पर \qty{10}{A} वाले दो समांतर तार:
        प्रति मीटर बल निकालिए, और उसे परिभाषा-कोटि के मान
        $2 \times 10^{-7}\,I_1I_2/d$ न्यूटन प्रति मीटर से जाँचिए।
  \item कोई विद्युत्चुंबक इसी कहानी के दस हज़ार फेरे है:
        \qty{10}{A} ढोती प्रति मीटर \qty{e4}{} फेरों की कुंडली का
        क्षेत्र आँकिए (वर्ष 1 के खंड का परिनालिका सूत्र), और वह
        प्रति वर्ग सेंटीमीटर लोहे पर जो बल लगाती है, वह भी
        ($\sim B^2/2\mu_0$)।
  \item यदि आपेक्षिकता बंद कर दी जाए (रूपांतरण में $c \to \infty$),
        तो $\lambda'$ का, $B$ के बल का, तथा मोटरों और कबाड़खाने के
        चुंबकों का क्या बचता?
  \item तार के पास \emph{विराम} में रखा परीक्षण-आवेश कुछ भी क्यों
        अनुभव नहीं करता, यद्यपि इलेक्ट्रॉन उसके पास से बहते रहते हैं?
        (कौन-सा निरसन उसकी रक्षा करता है, और किस यथार्थता तक?)
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- तार से आगे।}
\begin{enumerate}[resume]
  \item वर्ष 2 के खंड ने चुंबकीय क्षेत्र ऐंपियर के नियम से निकाले
        थे, और धाराओं को दिए हुए स्रोत माना था। यह अध्याय उस
        हिसाब में क्या जोड़ता है --- इस समस्या से देखने पर चुंबकीय
        क्षेत्र \emph{है} क्या?
  \item लोहे के चुंबकों के पास कोई बैटरी नहीं होती: स्थायी चुंबकत्व
        में धारा की भूमिका कौन निभाता है (एक वाक्य में; ईमानदार
        सूक्ष्मदर्शीय उत्तर क्वांटम है और पाँच अध्याय आगे प्रतीक्षा
        करता है)?
  \item निर्वात में कोई अकेला इलेक्ट्रॉन-पुंज (कोई जालक नहीं): क्या
        साथ-साथ चलता कोई प्रेक्षक उसे प्रयोगशाला के प्रेक्षक से अधिक
        आकर्षित होता देखता है या अधिक प्रतिकर्षित? पुंज के अपने
        सिकुड़ने के दोनों तंत्रों वाले हिसाब मिलाइए।
  \item अंतिम बात को संख्याओं में लाइए: दिखाइए कि साथ-साथ $\beta$ से
        चलते समान आवेश वाले दो समांतर पुंज कुल बल (विद्युत् प्रतिकर्षण
        ऋण चुंबकीय आकर्षण) से प्रतिकर्षित होते हैं, जो अपने
        विराम-मान से ठीक $1/\gamma^2$ जितना घटा हुआ है --- और
        बताइए कि किस तंत्र का हिसाब इस गुणक को स्पष्ट कर देता है।
  \item किसी लैंप की ऊर्जा लगभग $c$ से पहुँचती है, जबकि उसके
        इलेक्ट्रॉन प्रति घंटा एक मीटर अपवाहित होते हैं: डोरी के
        अनुदिश ऊर्जा वास्तव में कौन ले जाता है (वर्ष 2 के खंड का
        पॉयंटिंग सदिश स्मरण कीजिए)?
  \item इस समस्या के इलेक्ट्रॉनों और किसी पैदल चलते परीक्षण-आवेश के
        लिए $vu/c^2$ आँकिए ($v = \qty{1}{m/s}$): वहाँ भी
        बल किसी दिक्सूचक सुई से प्रथम कोटि पर मापा जा सकता है ---
        किसने धारा से दिक्सूचक का विक्षेपण दिखाया, और किस वर्ष?
  \item नामित परिणाम का सार लिखिए: कोई उदासीन \qty{10}{A} तार,
        \qty{e5}{m/s} से चलती सीट से देखने पर,
        $-\qty{e-11}{C/m}$ का आपेक्षिकीय आवेश ढोता है --- और
        लंबाई-संकुचन की वही $10^{-24}$ कोटि की हिसाब-त्रुटि,
        $10^{28}$ इलेक्ट्रॉनों से गुणा होकर, प्रयोगशाला का पूरा
        चुंबकीय बल है।
\end{enumerate}
\end{problem}
