\chapter{क्वांटम यांत्रिकी की औपचारिक रचना}\label{ch:b3:quantum-formalism}

तीन ध्रुवक छन्ने लीजिए। उनमें से दो, जो $90^\circ$ पर आड़े रखे हों,
प्रकाश को पूरी तरह रोक देते हैं; अब तीसरा उनके बीच $45^\circ$ पर सरका
दीजिए और प्रकाश \emph{पार} आने लगता है --- एक और बाधा जोड़ने से रास्ता
खुल जाता है। छन्नों को छलनी मानने वाला कोई चित्र इस प्रयोग को नहीं झेलता;
जो झेलता है वह रैखिक बीजगणित है: किसी फ़ोटॉन का ध्रुवण एक
\emph{सदिश} है, हर छन्ना उसे किसी अक्ष के अनुदिश नापकर प्रक्षिप्त कर
देता है, और प्रायिकताएँ घटकों के वर्ग हैं। यह अध्याय उसी बीजगणित को
क्वांटम यांत्रिकी की असली नींव के रूप में बिठा देता है। पिछले अध्यायों
के तरंग फलन किसी अधिक अमूर्त शरीर की एक ठोस पोशाक भर थे: अवस्थाएँ
\omterm{def:b3:quantum-formalism:state}{हिल्बर्ट समष्टि} के सदिश हैं,
\omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} \omterm{def:b3:quantum-formalism:observable}{हर्मिटीय संकारक} हैं, मापे गए मान \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} हैं,
प्रायिकताएँ अदिश गुणनफलों के वर्ग हैं, और विकास
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} से उत्पन्न होता है। पाँच अभिगृहीत, जो अब तक की हर गणना में
पहले से ही काम कर रहे थे --- और एक बार साफ़-साफ़ कह देने पर इतने समर्थ
कि वे ऐसे तंत्र भी सँभाल लें जिनका वर्णन किसी रेखा पर की कोई तरंग नहीं
कर सकती: फ़ोटॉन का ध्रुवण, इलेक्ट्रॉन का चक्रण, और वे न्यूट्रिनो जिनकी
पहचान पृथ्वी को पार करते-करते दोलन करती रहती है।

\section{अवस्थाएँ सदिश हैं}

\begin{definition}[अवस्था-समष्टि, केट और कोष्ठक]\label{def:b3:quantum-formalism:state}
\index{हिल्बर्ट समष्टि}\index{केट}\index{ब्रा}
किसी क्वांटम तंत्र की अवस्थाएँ एक आंतरिक गुणनफल वाली सम्मिश्र सदिश
समष्टि बनाती हैं --- अर्थात् कोई \emph{हिल्बर्ट समष्टि} $\mathcal H$
(गणित वर्ष 3 के गणित-खंड में विकसित है)। कोई अवस्था
एक सदिश है, जिसे \emph{केट} $\ket\psi$ लिखते हैं, और जो प्रसामान्यित है:
$\braket\psi\psi = 1$। दो अवस्थाओं का आंतरिक गुणनफल सम्मिश्र
संख्या $\braket\varphi\psi$ है, जो पहले खाने में संयुग्मी-रैखिक है और
जिसमें $\braket\varphi\psi = \braket\psi\varphi^*$। किसी लांबिक
प्रसामान्य आधार $\{\ket{e_i}\}$ में,
\[
\ket\psi = \sum_i c_i\ket{e_i} , \qquad
c_i = \braket{e_i}\psi , \qquad
\sum_i|c_i|^2 = 1 .
\]
पिछले अध्यायों का तरंग फलन स्थिति के अनुदिश
$\ket\psi$ के घटकों का कुल ही है: $\psi(x) =
\braket{x}{\psi}$; इसमें कुछ भी नहीं खोता, और \emph{परिमित}
अवस्था-समष्टि वाले तंत्र --- जिनकी तरंगों के रूप में कल्पना भी नहीं की
जा सकती --- वर्णनीय हो जाते हैं।
\end{definition}

\begin{example}[फ़ोटॉन का ध्रुवण: एक द्विविमीय संसार]\label{ex:b3:quantum-formalism:polarization}
$z$ अक्ष के अनुदिश जाता कोई फ़ोटॉन अपना ध्रुवण किसी
\emph{द्वि}विमीय \omterm{def:b3:quantum-formalism:state}{हिल्बर्ट समष्टि} में ढोता है, जिसका आधार $\ket H$
(क्षैतिज) और $\ket V$ (ऊर्ध्वाधर) है। कोण
$\theta$ पर ध्रुवित प्रकाश यह अध्यारोपण है
\[
\ket\theta = \cos\theta\,\ket H + \sin\theta\,\ket V ,
\]
और वृत्तीय ध्रुवण वह सम्मिश्र संयोजन $(\ket H \pm
\iu\ket V)/\sqrt2$ है: गुणांकों का सम्मिश्र होना कोई सजावट नहीं,
भौतिकी है। हर क्वांटम द्वि-स्तरीय तंत्र --- चक्रण, अमोनिया का
अणु, कोई अतिचालक क्यूबिट --- वही सदिश समष्टि है, बस दूसरे कपड़ों में।
\end{example}

\section{प्रेक्ष्य संकारक हैं}

\begin{definition}[प्रेक्ष्य]\label{def:b3:quantum-formalism:observable}
\index{प्रेक्ष्य}\index{हर्मिटीय संकारक}\index{अभिलक्षणिक मान}
\emph{प्रेक्ष्य} कोई ऐसा रैखिक संकारक $\hat A$ है, जो $\mathcal H$ पर
हो और \emph{हर्मिटीय} हो: सभी अवस्थाओं के लिए $\braket{\varphi}{\hat A\psi} =
\braket{\hat A\varphi}{\psi}$ (आव्यूह की भाषा में,
$A = A^{*\top}$)। उसके \emph{अभिलक्षणिक सदिश} और \emph{अभिलक्षणिक मान},
$\hat A\ket{a} = a\ket{a}$, भौतिकी ढोते हैं: $a$ ही संभव मापे गए मान
हैं। परिचित उदाहरण: स्थिति ($\hat x$:
$x$ से गुणा), संवेग ($\hat p = -\iu\hbar\,\dd/\dd x$),
ऊर्जा ($\hat H = \hat p^2/2m + V(\hat x)$) --- और दो
विमाओं में, कोई भी हर्मिटीय $2\times2$ आव्यूह।
\end{definition}

\begin{theorem}[हर्मिटीय क्यों]\label{thm:b3:quantum-formalism:hermitian}
किसी \omterm{def:b3:quantum-formalism:observable}{हर्मिटीय संकारक} के \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} वास्तविक होते हैं, और भिन्न
अभिलक्षणिक मानों वाले अभिलक्षणिक सदिश लांबिक होते हैं; इस पुस्तक की
समष्टियों पर उसके अभिलक्षणिक सदिश $\mathcal H$ का लांबिक प्रसामान्य
आधार बनाते हैं (अर्थात्
\emph{वर्णक्रमीय प्रमेय}, जो परिमित विमा में वर्ष 2 के गणित-खंड में
सिद्ध है; अपरिबद्ध संकारकों $\hat x$, $\hat p$,
$\hat H$ के लिए पूरा कथन वर्ष 3 के वर्णक्रम-सिद्धांत का है और यहाँ
स्वीकृत है)। मापे गए मान वास्तविक होने ही चाहिए और अलग-अलग पहचाने जाने
वाले परिणाम लांबिक: हर्मिटीयता ठीक वही शर्त है जिससे कोई
संकारक किसी माप का प्रतिनिधित्व कर सके।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
$a\braket aa = \braket{a}{\hat Aa} = \braket{\hat Aa}{a} =
a^*\braket aa$: अतः $a$ वास्तविक। $\hat A\ket a = a\ket a$, $\hat A\ket b
= b\ket b$ के लिए: $a\braket ba = \braket{b}{\hat Aa} = \braket{\hat
Ab}{a} = b\braket ba$, अतः $(a - b)\braket ba = 0$।
\end{proof}

\begin{example}[एक द्वि-स्तरीय प्रेक्ष्य]\label{ex:b3:quantum-formalism:twolevel}
ध्रुवण-समष्टि पर, $(\ket H, \ket V)$ आधार में, यह लीजिए
\[
\hat A = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix} :
\]
हर्मिटीय; \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm1$; अभिलक्षणिक सदिश $(\ket H \pm
\ket V)/\sqrt2$ --- अर्थात् $\pm45^\circ$ के ध्रुवण।
$\hat A$ को नापने का \emph{अर्थ} है यह पूछना कि ``विकर्ण या
प्रति-विकर्ण?'', और अभिलक्षणिक आधार उन्हीं दो प्रश्नों के उत्तर हैं।
आरंभिक प्रयोग का हर $45^\circ$ ध्रुवक काँच में ढला यही आव्यूह है।
\end{example}

\section{अभिगृहीत}

\begin{theorem}[क्वांटम यांत्रिकी के नियम]\label{thm:b3:quantum-formalism:postulates}
\index{क्वांटम यांत्रिकी के अभिगृहीत}\index{बोर्न का नियम}\index{संकुचन}
\emph{(अ1)} किसी तंत्र की अवस्था उसकी \omterm{def:b3:quantum-formalism:state}{हिल्बर्ट समष्टि} में एक
प्रसामान्यित \omterm{def:b3:quantum-formalism:state}{केट} $\ket\psi$ है।
\emph{(अ2)} हर मापनीय राशि कोई \omterm{def:b3:quantum-formalism:observable}{हर्मिटीय संकारक}
$\hat A$ है।
\emph{(अ3)} $\hat A$ को नापने के एकमात्र संभव परिणाम उसके
\omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} हैं।
\emph{(अ4)} किसी अवस्था $\ket\psi$ पर परिणाम $a$ प्रायिकता
$\mathcal P(a) = |\braket a\psi|^2$ से आता है (\omterm{thm:b3:quantum-formalism:postulates}{बोर्न का नियम}; किसी
अपह्रासी \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} के लिए उसके अभिलक्षणिक अवकाश पर घटकों के वर्ग
जोड़ लीजिए)। अनेक प्रयासों का माध्य $\langle\hat A\rangle =
\bra\psi\hat A\ket\psi$ है।
\emph{(अ5)} $a$ देने वाली माप के तुरंत बाद अवस्था
$\ket\psi$ का $a$ वाले अभिलक्षणिक अवकाश पर (प्रसामान्यित) प्रक्षेप होती
है --- यही \emph{संकुचन} है: माप एक ऐसी अन्योन्यक्रिया है जो तंत्र को
उसी अवस्था में छोड़ जाती है जो उसके अपने उत्तर से मेल खाती हो।
\emph{(अ6)} मापों के बीच अवस्था श्रोडिंगर समीकरण
$\iu\hbar\,\dfrac{\dd}{\dd t}\ket{\psi(t)} = \hat H\ket{\psi(t)}$
के अंतर्गत एकात्मक रूप से विकसित होती है।
\end{theorem}
\admitted

\begin{example}[तीन ध्रुवक, गणना सहित]\label{ex:b3:quantum-formalism:threepolarizers}
ऊर्ध्वाधर प्रकाश $90^\circ$ पर आड़े रखे किसी विश्लेषक से मिलता है: $|\braket
HV|^2 = 0$ --- अर्थात् पूर्ण विलोपन। अब बीच में $45^\circ$ का ध्रुवक
रखिए: अवस्था $\ket V$ उससे प्रायिकता $|\braket{45^\circ}{V}|^2 =
\tfrac12$ से गुज़रती है और \emph{संकुचित} होकर $\ket{45^\circ}$ बन जाती
है (अ5); फिर वही अवस्था क्षैतिज विश्लेषक से प्रायिकता
$|\braket{H}{45^\circ}|^2 = \tfrac12$ से गुज़रती है। कुल संचरण
शून्य के बदले $\tfrac14$: बीच वाला छन्ना कोई ``छेद नहीं खोलता'' ---
वह एक माप करता है, और संकुचन फ़ोटॉन को फिर से तैयार कर देता है। कोई
चिरसम्मत छलनी ऐसा नहीं करती; और कोई प्रक्षेप इसके सिवा कुछ करता ही नहीं।
\end{example}

\begin{remark}[संकुचन किसकी अनुमति नहीं देता]\label{rem:b3:quantum-formalism:nosignal}
औपचारिक रचना में संकुचन तात्क्षणिक है, और दूर-दूर के कणों के बीच क्वांटम
सहसंबंध वास्तविक हैं और नापे गए हैं --- पर उन पर कोई
\emph{संदेश} सवार नहीं होता: किसी एक संसूचक के परिणाम, अकेले पढ़े जाएँ
तो सिक्कों की उछाल से अभेद्य रहते हैं, चाहे दूर कुछ भी किया जाए।
क्वांटम यादृच्छिकता अपह्रास्य भी नहीं है: \omterm{thm:b3:quantum-formalism:postulates}{बोर्न का नियम} तब भी
प्रायिकताएँ ही देता है जब अवस्था \emph{पूरी तरह} ज्ञात हो --- जानने को
और कुछ बचता ही नहीं। दोनों कथन इस औपचारिक रचना की प्रमेय हैं, जो ऊँची
यथार्थता तक परखे जा चुके हैं; उनके प्रति असहजता आदरणीय है और उसने एक
सदी के प्रयोग चलाए हैं, जिनमें से हर एक क्वांटम यांत्रिकी ने जीता है।
\end{remark}

\section{क्रमविनिमेयक और अनिश्चितता}

\begin{definition}[क्रमविनिमेयक; संगत प्रेक्ष्य]\label{def:b3:quantum-formalism:commutator}
\index{क्रमविनिमेयक}\index{संगत प्रेक्ष्य}
दो संकारकों का \emph{क्रमविनिमेयक} $[\hat A, \hat B] = \hat
A\hat B - \hat B\hat A$ है। संस्थापक उदाहरण, $\hat p =
-\iu\hbar\,\dd/\dd x$ से:
\[
[\hat x, \hat p] = \iu\hbar .
\]
दो \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} तब \emph{संगत} कहलाते हैं जब $[\hat A, \hat B] = 0$ हो:
तब वे कोई उभयनिष्ठ अभिलक्षणिक आधार स्वीकार करते हैं और एक साथ जाने जा
सकते हैं; एक को नापने से वह अवस्था विक्षुब्ध नहीं होती जो दूसरे में
तीखी हो। ऐसे क्रमविनिमेय प्रेक्ष्यों का समुच्चय, जिसका उभयनिष्ठ
अभिलक्षणिक आधार अद्वितीय हो (कोई
\emph{CSCO}), वही है जिसे ``किसी अवस्था पर पूरा नाम लगाना'' कहते हैं ---
संदूक के नाम $(n_1, n_2, n_3)$ ठीक यही थे।
\end{definition}

\begin{theorem}[अनिश्चितता संबंध, व्यापक रूप में]\label{thm:b3:quantum-formalism:uncertainty}
\index{अनिश्चितता संबंध}
किसी भी अवस्था में दो प्रेक्ष्यों के मानक विचलन इसका पालन करते हैं
\[
\Delta A\;\Delta B \ \ge\ \tfrac12\,\big|\langle[\hat A, \hat
B]\rangle\big| .
\]
$\hat x$ और $\hat p$ के लिए: $\Delta x\,\Delta p \ge \hbar/2$ ---
अर्थात् हाइज़ेनबर्ग का संबंध, जो अब किसी अनुमान के बदले रैखिक बीजगणित
की प्रमेय है। असंगति मात्रात्मक है: \omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} का आकार
संयुक्त तीक्ष्णता के नीचे का फ़र्श तय कर देता है।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
मान लीजिए $\hat\alpha = \hat A - \langle A\rangle$, $\hat\beta = \hat B -
\langle B\rangle$। वर्ष 3 के गणित-खंड की कोशी--श्वार्त्स असमिका
देती है $\Delta A^2\Delta B^2 =
\|\hat\alpha\psi\|^2\|\hat\beta\psi\|^2 \ge
|\braket{\hat\alpha\psi}{\hat\beta\psi}|^2$; उस गुणनफल का काल्पनिक भाग
$\tfrac1{2\iu}\langle[\hat A, \hat B]\rangle$ है, और
$|z|^2 \ge (\operatorname{Im}z)^2$।
\end{proof}

\begin{remark}[चिरसम्मत छाया]\label{rem:b3:quantum-formalism:poisson}
$\iu\hbar$ से भाग दीजिए और $\hbar \to 0$ कीजिए: \omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक}
\cref{ch:b3:hamiltonian-mechanics} के \omterm{def:b3:hamiltonian-mechanics:poisson}{प्वासों कोष्ठक} बन जाते हैं, $\{x, p\} =
1$ में $[\hat x, \hat p] = \iu\hbar$ की गूँज है, और वहाँ निकाले गए
कोणीय-संवेग कोष्ठक \cref{ch:b3:quantum-angular-momentum} में
क्रमविनिमेयकों के रूप में, अपरिवर्तित, लौट आएँगे। डिराक का नियम ---
चिरसम्मत कोष्ठक गुणा $\iu\hbar$ --- वही तरीका है जिससे यांत्रिकी का
ढाँचा क्रांति को झेल गया।
\end{remark}

\section{विकास और संरक्षण}

\begin{proposition}[औसतों का विकास; संरक्षित राशियाँ]\label{prop:b3:quantum-formalism:evolution}
\index{एरेनफ़ेस्ट प्रमेय}
स्पष्ट काल-निर्भरता रहित किसी \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} के लिए,
\[
\frac{\dd\langle\hat A\rangle}{\dd t}
 = \frac{\iu}{\hbar}\,\big\langle[\hat H, \hat A]\big\rangle .
\]
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} के साथ क्रमविनिमेय कोई \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} संरक्षित है --- और
केवल उसका माध्य नहीं, उसकी प्रायिकताएँ तक जमी रहती हैं; सममितियाँ
फिर से संरक्षण नियम देती हैं, अब क्रमविनिमेयन के रूप में। स्थायी
अवस्थाएँ $\hat H$ के अभिलक्षणिक सदिश हैं, जो केवल कला
$\eu^{-\iu Et/\hbar}$ से विकसित होती हैं; और दो स्तरों का कोई
अध्यारोपण बोर आवृत्ति $(E_2 - E_1)/h$ पर विस्पंदित होता है, जैसा
वर्ष 2 के खंड के कूप पहले ही दिखा चुके हैं।
\end{proposition}

\begin{proof}
$\bra\psi\hat A\ket\psi$ का अवकलन कीजिए और अ6 तथा उसका संयुग्मी
रखिए:
\[
\frac{\dd\langle\hat A\rangle}{\dd t}
 = \frac{\iu}{\hbar}\bra\psi\hat H\hat A\ket\psi
 - \frac{\iu}{\hbar}\bra\psi\hat A\hat H\ket\psi
 = \frac{\iu}{\hbar}\,\langle[\hat H, \hat A]\rangle .
\qedhere
\]
\end{proof}

\begin{method}[आव्यूह क्वांटम यांत्रिकी]\label{met:b3:quantum-formalism:method}
किसी भी परिमित-स्तर समस्या के लिए: (1) प्रश्न के अनुकूल कोई आधार
चुनिए (विश्लेषक की अक्षें, ऊर्जा की अभिलक्षणिक अवस्थाएँ); (2)
अवस्थाओं को स्तंभ-सदिश और प्रेक्ष्यों को हर्मिटीय आव्यूह लिखिए; (3)
जो नापा जा रहा हो उसका विकर्णन कीजिए --- \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} परिणाम हैं,
और घटकों के वर्ग प्रायिकताएँ; (4) ऊर्जा की अभिलक्षणिक अवस्थाओं को
कलाओं $\eu^{-\iu E_it/\hbar}$ से विकसित कीजिए और माप के आधार में फिर
से व्यक्त कीजिए; (5) किसी माप के बाद प्रक्षिप्त अवस्था से नए सिरे से
आरंभ कीजिए। पूरी \cref{pb:b3:quantum-formalism:1} किसी द्वि-स्तरीय
ब्रह्मांड पर चलाई गई यही विधि है।
\end{method}

\includegraphics[width=0.58\linewidth]{images/book5/ai/b5-08-three-polarizers.jpg}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b3:quantum-formalism:1}
किसी लांबिक प्रसामान्य आधार $(\ket1, \ket2)$ में मान लीजिए $\ket\psi = (2\ket1 +
\iu\ket2)/\sqrt5$ और $\ket\varphi = (\ket1 - \ket2)/\sqrt2$। (क)
प्रसामान्यन जाँचिए। (ख) $\braket\varphi\psi$ और
$\braket\psi\varphi$ निकालिए। (ग) $\ket\psi$ को
अवस्था $\ket\varphi$ में पाने की प्रायिकता। (घ) $\ket\psi$ के लांबिक
अवस्था की रचना कीजिए (कला को छोड़कर)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-formalism:2}
इनमें से कौन-से आव्यूह हर्मिटीय हैं, और जो हैं उनके \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान}
तथा प्रसामान्यित अभिलक्षणिक सदिश क्या हैं?
\[
\begin{pmatrix}0 & 1\\ 1 & 0\end{pmatrix} , \quad
\begin{pmatrix}0 & -\iu\\ \iu & 0\end{pmatrix} , \quad
\begin{pmatrix}1 & 1\\ -1 & 1\end{pmatrix} , \quad
\begin{pmatrix}2 & 1-\iu\\ 1+\iu & 3\end{pmatrix} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-formalism:3}
कोण $\alpha$ पर ध्रुवित प्रकाश कोण $\beta$ के किसी विश्लेषक से मिलता
है। (क) दोनों अवस्थाएँ $(\ket H, \ket V)$ आधार में लिखिए और
संचरण-प्रायिकता निकालिए। (ख) मैलस का नियम पुनः प्राप्त कीजिए। (ग)
$N$ फ़ोटॉनों की धारा के लिए संचरित संख्या का प्रसरण क्या है? (घ)
किसी भी कोण के रैखिक विश्लेषक पर वृत्तीय प्रकाश $(\ket H + \iu\ket V)/\sqrt2$:
संचरण? उत्तर की कोण से स्वतंत्रता समझाइए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-formalism:4}
(क) किसी परीक्षण-फलन पर लगाकर $[\hat x, \hat p] = \iu\hbar$ सिद्ध कीजिए।
(ख) $[\hat x^2, \hat p]$ और $[\hat x, \hat p^2]$ निकालिए। (ग) दिखाइए कि
$[\hat A, \hat B\hat C] = [\hat A, \hat B]\hat C + \hat B[\hat A,
\hat C]$। (घ) $[\hat x, \hat p^n] = \iu\hbar\,n\hat p^{n-1}$ निकालिए
और व्याख्या कीजिए: $[\hat x, \cdot\,]$ किस चिरसम्मत संक्रिया की नकल
करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-formalism:5}
क्रमिक माप। कोई फ़ोटॉन $\ket V$ से आरंभ करता है। (क) वह
$45^\circ$ और फिर $0^\circ$ (क्षैतिज) के ध्रुवकों से मिलता है: दोनों
से बचने की प्रायिकता निकालिए, और बीच के संकुचन को स्पष्ट रखिए। (ख)
बीच वाले ध्रुवक के स्थान पर $30^\circ$,
$60^\circ$ वाले रखिए: संचरण? (ग) $N - 1$ मध्यवर्ती ध्रुवक, जो
$90^\circ/N$ के पगों से बढ़ें: दिखाइए कि बचने की प्रायिकता
$\cos^{2N}(\pi/2N)$ है, और $N = 2, 5, 20$ के लिए मान निकालिए। (घ)
सीमा $N \to \infty$ ध्रुवण को \emph{बिना} किसी हानि के घुमा देती है:
टिप्पणी कीजिए (यही ``क्वांटम ज़ीनो'' घूर्णन असली क्यूबिटों पर काम में लाया जाता है)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-formalism:6}
द्वि-स्तरीय समष्टि पर $\hat A = \begin{pmatrix}0&1\\1&0
\end{pmatrix}$, $\hat B = \begin{pmatrix}1&0\\0&-1\end{pmatrix}$।
(क) $[\hat A, \hat B]$ निकालिए: संगत? (ख) अवस्था
$\ket1$ है: $\hat B$ नापने के परिणामों की सांख्यिकी दीजिए, और फिर
$\hat A$ नापने की भी --- $\hat B$ की माप के $+1$ देने के \emph{बाद}।
(ग) पहले $\hat A$ नापिए (परिणाम $+1$), फिर $\hat B$, फिर
$\hat A$ दोबारा: अंतिम $\hat A$ पहले वाले का \emph{खंडन}
किस प्रायिकता से करता है? (घ) शून्य \omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} होने पर (ग) का क्या
अर्थ होता?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-formalism:7}
(क) \cref{exo:b3:schrodinger-three-dimensions:1} की गाउसीय अवस्था को
एक विमा तक सीमित करके
$\Delta x$ और $\Delta p$ निकालिए (फूरिये युग्म काम में लाइए या
समाकलन कीजिए) और हाइज़ेनबर्ग के संबंध में समता जाँचिए। (ख) कौन-सी
अवस्थाएँ व्यापक अनिश्चितता प्रमेय को संतृप्त करती हैं (कोशी--श्वार्त्स
की समता-स्थिति से शर्त बताइए)? (ग) $\Delta x = \qty{0.1}{nm}$ तक बद्ध
किसी इलेक्ट्रॉन के लिए: न्यूनतम गतिज ऊर्जा का पैमाना? (घ) किसी
कंचे ($\qty{10}{g}$) के लिए, जो एक माइक्रोन तक स्थानीकृत हो: निष्कर्ष दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-formalism:8}
(क) \cref{prop:b3:quantum-formalism:evolution} से $\hat x$ और
$\hat p$ के लिए एरेनफ़ेस्ट का युग्म पुनः प्राप्त कीजिए, $\hat H = \hat
p^2/2m + V$ के साथ। (ख) दिखाइए कि सम-विषमता $\hat\Pi$ ($\hat\Pi\psi(x) =
\psi(-x)$) हर्मिटीय है, उसका वर्ग तत्समक है, और उसके
\omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm1$ हैं। (ग) किसी सममित विभव के लिए दिखाइए कि
$[\hat H, \hat\Pi] = 0$ और निष्कर्ष निकालिए कि अ-अपह्रासी स्तरों की
सम-विषमता निश्चित होती है। (घ) किसी सममित द्विकूप के स्तरों का
कौन-सा प्रेक्षित तथ्य इससे समझ में आता है (वर्ष 2 के खंड का
अमोनिया-द्विक स्मरण कीजिए)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-formalism:9}
काम पर एक CSCO। किसी वर्गाकार द्विविमीय संदूक में स्तर $E
\propto 5$ को $\ket{1,2}$ और $\ket{2,1}$ फैलाते हैं। (क) दिखाइए कि
अकेली ऊर्जा अवस्थाओं पर नाम नहीं लगाती। (ख) मान लीजिए $\hat S$, $x
\leftrightarrow y$ की अदला-बदली करता है: दिखाइए कि $\hat S$ हर्मिटीय
है, $\hat
H$ के साथ क्रमविनिमेय है, और उस स्तर के भीतर उसकी
अभिलक्षणिक अवस्थाएँ ज्ञात कीजिए। (ग) जाँचिए कि युग्म $(\hat H, \hat S)$ इस स्तर की
हर अवस्था पर अद्वितीय नाम लगा देता है। (घ) व्यापक सार बताइए:
अपह्रासिता का अर्थ है कि नामों का समुच्चय अभी पूरा नहीं था, और
सममिति छूटा हुआ नाम दे देती है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-formalism:10}
ऊर्जा--काल अनिश्चितता, ईमानदारी से। किसी भी \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} $\hat A$ के लिए
विकास-काल $\tau_A = \Delta A\,/\,|\dd\langle\hat
A\rangle/\dd t|$ परिभाषित कीजिए --- अर्थात् वह समय, जिसमें माध्य एक
मानक विचलन जितना खिसक जाए। (क) \cref{thm:b3:quantum-formalism:uncertainty} और
\cref{prop:b3:quantum-formalism:evolution} से $\Delta
E\;\tau_A \ge \hbar/2$ सिद्ध कीजिए। (ख) यह दो प्रेक्ष्यों के बीच की
अनिश्चितता \emph{क्यों नहीं} है (औपचारिक रचना में काल है क्या)? (ग)
\qty{10}{ns} जीवनकाल वाली किसी उत्तेजित परमाणु-अवस्था पर लगाइए:
रेखा-चौड़ाई। (घ) अपनी कलाई-घड़ी पर लगाइए: यदि किसी तंत्र में हर सेकंड
कुछ दिखाई देने लायक बदलता हो, तो उसकी ऊर्जा कितनी तीखी हो सकती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-formalism:11}
अनिश्चितता प्रमेय की उपपत्ति। पाठ वाले $\hat\alpha, \hat\beta$ के साथ:
(क) हर्मिटीयता का उपयोग करके $\Delta A^2 = \|\hat\alpha\ket\psi\|^2$
का औचित्य दीजिए; (ख) कोशी--श्वार्त्स लगाइए और
$\braket{\hat\alpha\psi}{\hat\beta\psi}$ को हर्मिटीय तथा
प्रति-हर्मिटीय भागों में बाँटिए, और उन्हें प्रति-क्रमविनिमेयक तथा
\omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} के औसतों से पहचानिए; (ग) निष्कर्ष निकालिए, और बताइए कि
समता कब होती है; (घ) दिखाइए कि $[\hat A, \hat B] = \iu\hbar$ के लिए
कोई भी अवस्था दोनों विचलनों को परिमित रखते हुए किसी एक \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} की
अभिलक्षणिक सदिश नहीं हो सकती --- और इसे समतल तरंगों से मिलाइए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-formalism:12}
देखी जाती हाँडी। कोई द्वि-स्तरीय तंत्र $\ket1$ से आरंभ करता है और
उसका \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} राबी-जैसा दोलन चलाता है: समय $t$ के बाद अवस्था
$\cos(\omega t)\ket1 + \sin(\omega t)\ket2$ है (इसे दिया हुआ मानिए)।
(क) बिना किसी माप के, $\ket2$ में अंतरण कब पूरा होता है?
(ख) समयों $T/N, 2T/N, \dots$ पर ``कौन-सी अवस्था?'' नापिए, जहाँ $T =
\pi/2\omega$: दिखाइए कि हर जाँच पर तंत्र को
\emph{अब भी} $\ket1$ में पाने की प्रायिकता $[\cos^2(\pi/2N)]^N$ है। (ग)
$N = 1, 4, 20, 100$ के लिए मान निकालिए और दिखाइए कि वह $1$ की ओर जाता
है: बार-बार का प्रेक्षण विकास को जमा देता है (क्वांटम ज़ीनो प्रभाव,
जो 1990 में बद्ध आयनों के साथ देखा गया)। (घ) एक वाक्य में समझाइए कि
यह जमाव कौन-सा अभिगृहीत करता है।
\end{exercise}

\includegraphics[width=0.62\linewidth]{images/book5/photo-solvay-1927.jpg}

\section{समस्या: न्यूट्रिनो उड़ान के बीच में भेस बदल लेते हैं}

\begin{problem}[{सप्ताहांत समस्या --- पृथ्वी भर में द्वि-स्तरीय
दोलन}]\label{pb:b3:quantum-formalism:1}
न्यूट्रिनो नाभिकीय अभिक्रियाओं में \emph{स्वाद} अवस्थाओं के रूप में
जन्मते हैं --- इलेक्ट्रॉन-न्यूट्रिनो $\ket{\nu_e}$ या
म्युऑन-न्यूट्रिनो $\ket{\nu_\mu}$ --- पर वे ऊर्जा (द्रव्यमान) की
अभिलक्षणिक अवस्थाओं $\ket{\nu_1},
\ket{\nu_2}$ के रूप में \emph{संचरित} होते हैं। दोनों आधार मेल नहीं
खाते: वे किसी \emph{मिश्रण कोण} $\theta$ से घूमे हुए हैं,
\[
\ket{\nu_e} = \cos\theta\,\ket{\nu_1} + \sin\theta\,\ket{\nu_2} ,
\qquad
\ket{\nu_\mu} = -\sin\theta\,\ket{\nu_1} + \cos\theta\,\ket{\nu_2} .
\]
इस खोज को कि न्यूट्रिनो इसलिए उड़ान में स्वादों के बीच \emph{दोलन}
करते हैं --- और अतः उनका द्रव्यमान है --- 2015 का नोबेल
पुरस्कार मिला। यह समस्या इस प्रभाव को इस अध्याय से आगे कुछ भी लिए बिना
निकाल देती है। संवेग $p$ और द्रव्यमान
$m_i$ वाले किसी अति-आपेक्षिकीय न्यूट्रिनो की ऊर्जा $E_i \approx pc + m_i^2c^4/2pc \approx E +
m_i^2c^4/2E$ होती है।

\medskip\noindent\textbf{भाग I --- औपचारिक रचना का ढाँचा।}
\begin{enumerate}
  \item जाँचिए कि $(\ket{\nu_1}, \ket{\nu_2})$ के लांबिक प्रसामान्य
        होने से $(\ket{\nu_e}, \ket{\nu_\mu})$ भी लांबिक प्रसामान्य
        हो जाते हैं।
  \item संचरण का आधार \emph{ऊर्जा} का अभिलक्षणिक आधार ही क्यों होना
        चाहिए, चाहे न्यूट्रिनो किसी भी आधार में जन्मा हो? (मुक्त
        उड़ान को कौन-सा अभिगृहीत चलाता है?)
  \item कोई म्युऑन-न्यूट्रिनो $t = 0$ पर जन्मता है। $\ket{\psi(0)}$
        द्रव्यमान-आधार में लिखिए।
  \item $\ket{\psi(t)}$ लिखिए, जिसमें हर द्रव्यमान-घटक अपनी कला
        $\eu^{-\iu E_it/\hbar}$ ढोता हो।
  \item दिखाइए कि कोई \emph{वैश्विक} कला अप्रासंगिक है, और
        $\eu^{-\iu E_1t/\hbar}$ बाहर निकालिए: केवल \emph{आपेक्षिक}
        कला $\Delta\phi = (E_2 - E_1)t/\hbar$ ही भौतिकी चलाती है।
  \item अति-आपेक्षिकीय न्यूट्रिनो के लिए $E_2 - E_1$ को
        $\Delta m^2 = m_2^2 - m_1^2$ और $E$ के पदों में व्यक्त कीजिए।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दोलन का सूत्र।}
\begin{enumerate}[resume]
  \item आयाम $\braket{\nu_e}{\psi(t)}$ निकालिए।
  \item दिखाइए कि प्राकट्य-प्रायिकता यह है
        \[
        \mathcal P_{\nu_\mu \to \nu_e}(t) =
        \sin^2(2\theta)\,\sin^2\Big(\frac{\Delta\phi}{2}\Big) .
        \]
        ($2\sin\theta\cos\theta = \sin2\theta$ काम में लाइए।)
  \item दोनों विवेक-सीमाएँ जाँचिए: $\theta = 0$ और $m_1 = m_2$।
        अतः कोई भी प्रेक्षित दोलन क्या सिद्ध कर देता है?
  \item $t \approx L/c$ के साथ दिखाइए कि
        \[
        \mathcal P = \sin^2(2\theta)\,
        \sin^2\Big(\frac{\Delta m^2c^4}{4\hbar c}\,\frac{L}{E}\Big) ,
        \]
        और दोलन-लंबाई $L_{\text{दो}} =
        4\pi\hbar cE/\Delta m^2c^4$ परिभाषित कीजिए।
  \item दिखाइए कि उत्तरजीविता-प्रायिकता $\mathcal P_{\nu_\mu \to
        \nu_\mu} = 1 - \mathcal P_{\nu_\mu\to\nu_e}$: यहाँ
        एकात्मकता का उपयोग कहाँ हुआ?
  \item दोलन केवल $\Delta m^2$ ही क्यों नापता है, कभी स्वयं
        द्रव्यमान क्यों नहीं?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- प्रयोगों को पढ़ना।}
वायुमंडलीय म्युऑन-न्यूट्रिनो ($E \approx \qty{1}{GeV}$) किसी संसूचक पर
ऊपर से बरसते हैं ($L \approx \qty{15}{km}$) और नीचे से भी, पृथ्वी को
पार करके ($L \approx \qty{12800}{km}$)। सूपर-कामिओकांदे
(1998) ने पाया कि नीचे से आता अभिवाह आधा रह गया, और ऊपर से आता अभिवाह अछूता।
\begin{enumerate}[resume]
  \item $\hbar c = \qty{197}{MeV.fm}$ से यह सुविधाजनक रूप दिखाइए
        $\dfrac{\Delta m^2c^4\,L}{4\hbar cE} = 1.27\,
        \dfrac{\Delta m^2c^4\,[\unit{eV^2}]\ L\,[\unit{km}]}
        {E\,[\unit{GeV}]}$।
  \item यदि दोलन $L =
        \qty{12800}{km}$ पर \emph{भली-भाँति विकसित} हो पर
        \qty{15}{km} पर \emph{नगण्य}, $E = \qty{1}{GeV}$ के लिए, तो
        $\Delta m^2c^4$ की मोटी सीमाएँ बाँधिए।
  \item मापा गया मान $\Delta m^2c^4 \approx
        \qty{2.5e-3}{eV^2}$ है: \qty{1}{GeV} पर दोलन-लंबाई
        निकालिए और दोनों आधार-रेखाओं से जाँचिए।
  \item नीचे से आता दमन $1/2$ के पास है, $0$ के पास नहीं:
        दिखाइए कि अनेक दोलन-लंबाइयों पर (और ऊर्जाओं पर) $\sin^2$
        का \emph{औसत} $\tfrac12\sin^22\theta$ देता है, और
        निष्कर्ष निकालिए कि वायुमंडलीय मिश्रण लगभग
        \emph{अधिकतम} है ($\theta \approx 45^\circ$)।
  \item यदि हलका द्रव्यमान नगण्य हो, तो $\Delta m^2c^4 = \qty{2.5e-3}{eV^2}$
        अकेले भारी द्रव्यमान के लिए क्या देता है --- और
        उसकी तुलना इलेक्ट्रॉन के द्रव्यमान से कीजिए: न्यूट्रिनो कितने
        अजीब ढंग से हलके हैं?
  \item रिएक्टर-प्रतिन्यूट्रिनो की $E \approx \qty{4}{MeV}$ होती है।
        उसी $1.27$ वाले सूत्र से दिखाइए कि एक-दो किलोमीटर की
        आधार-रेखा $\qty{2.5e-3}{eV^2}$ के विपाटन पर सुर में है
        (दाया बे प्रयोग), जबकि $L \approx \qty{180}{km}$
        (कामलैंड) छोटे ``सौर'' विपाटन
        $\qty{7.5e-5}{eV^2}$ पर --- दोनों संख्याओं से जाँचिए।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- इसका अर्थ क्या है।}
\begin{enumerate}[resume]
  \item सूर्य $\nu_e$ उत्सर्जित करता है; दशकों तक संसूचक भविष्यवाणी
        का केवल एक तिहाई गिनते रहे। ``सौर न्यूट्रिनो समस्या'' और
        उसका समाधान, एक-एक वाक्य में समझाइए।
  \item दोलनों ने मानक निदर्श के मूल बहीखाते के विरुद्ध यह निष्कर्ष
        क्यों मनवा लिया कि न्यूट्रिनो का \emph{द्रव्यमान है}?
  \item कोई क्वांटम तंत्र $12800$ किलोमीटर तक कला-संसक्ति बनाए रखता
        है: इससे न्यूट्रिनो की अन्योन्यक्रिया की दुर्बलता के विषय में
        क्या पता चलता है, और संसूचक को विशाल क्यों होना पड़ता है?
  \item स्वाद एक \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} है: उसका संकारक मुक्त
        \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} के साथ क्रमविनिमेय क्यों नहीं है, और इसलिए
        स्वाद के लिए कौन-सा संरक्षण नियम \emph{उपलब्ध नहीं} है
        (जबकि ऊर्जा और संवेग संरक्षित बने रहते हैं)?
  \item प्रकृति में तीन स्वाद और तीन द्रव्यमान हैं: फिर भी
        द्वि-स्तरीय विवेचन हर प्रयोग का इतना अच्छा वर्णन क्यों कर
        देता है? (दोनों $\Delta m^2$ के अनुक्रम पर विचार कीजिए, और
        इस पर कि कौन-सी आधार-रेखा किसे सुलझाती है।)
  \item इस अध्याय की औपचारिक रचना में ठीक-ठीक नाम लीजिए कि दोलन किन
        अवयवों से बना: कौन-सा आधार-बेमेल, कौन-सा अभिगृहीत, कौन-सी कला।
  \item नामित परिणाम का सार लिखिए: $45^\circ$ के पास का एक घूर्णन-कोण
        और विपाटन $\Delta m^2c^4 =
        \qty{2.5e-3}{eV^2}$ मिलकर किसी GeV म्युऑन-न्यूट्रिनो को
        $\sim\qty{1000}{km}$ दोलन-लंबाई के साथ ग़ायब कर देते हैं ---
        द्वि-स्तरीय रैखिक बीजगणित, जिसकी पुष्टि पृथ्वी के शरीर से
        होकर हुई।
\end{enumerate}
\end{problem}
