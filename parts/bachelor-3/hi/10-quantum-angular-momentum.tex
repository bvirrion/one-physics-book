\chapter{क्वांटम कोणीय संवेग}\label{ch:b3:quantum-angular-momentum}

किसी रेडियो दूरदर्शी को आकाश के किसी अँधेरे टुकड़े की ओर कीजिए और उसे
\qty{115}{GHz} पर सुर में लाइए: एक चमकीली, सुई-सी तीखी रेखा प्रकट होती
है --- कार्बन मोनोऑक्साइड के अणु, परम शून्य से दस केल्विन ऊपर,
\emph{घूर्णन} अवस्थाओं की किसी सीढ़ी का एक पायदान उतरते हुए। उनका
लुढ़कना, क्वांटम यांत्रिकी की हर घूमती वस्तु की तरह, दो बार क्वांटीकृत
है: कोणीय संवेग का परिमाण केवल
$\sqrt{l(l+1)}\,\hbar$ हो सकता है, और किसी भी चुनी हुई अक्ष के अनुदिश
उसका घटक केवल $m\hbar$। यह अध्याय वही दुहरा क्वांटन निकालता है ---
एक बार बीजगणितीय ढंग से, उन क्रमविनिमेयकों से जो
\cref{ch:b3:hamiltonian-mechanics} के प्वासों कोष्ठकों से उत्तराधिकार
में मिले, और एक बार ठोस ढंग से, उन गोलीय संनादियों के रूप में जो हर
परमाणु-कक्षक को आकार देते हैं। बीजगणित हमें माँगे से अधिक दे देगा: वह
ऐसे अर्ध-पूर्णांक मान भी अनुमत करता है जिन्हें कोई कक्षकीय गति साकार
नहीं कर सकती --- एक रिक्ति, जिसे प्रकृति यहाँ से दो अध्याय आगे चक्रण
से भर देती है। रास्ते में हम अणुओं की घूर्णन-सीढ़ियों से मिलते हैं ---
वे मिलीमीटर-तरंग रेखाएँ, जिनसे खगोलविद मंदाकिनियों की ठंडी गैस तौलते
हैं --- और उन चुंबकीय आघूर्णों से भी जिनके द्वारा कोणीय संवेग ने पहली
बार अपना विपाटन दिखाया।

\section{घूर्णन का बीजगणित}

\begin{definition}[कोणीय संवेग के संकारक]\label{def:b3:quantum-angular-momentum:operators}
\index{कोणीय संवेग!क्वांटम}
कक्षकीय कोणीय संवेग वह संकारक $\hat{\vect L} =
\hat{\vect r}\wedge\hat{\vect p}$ है, जिसके घटक $\hat L_z = \hat
x\hat p_y - \hat y\hat p_x$ और चक्रीय हैं। $[\hat x, \hat p_x] =
\iu\hbar$ से:
\[
[\hat L_x, \hat L_y] = \iu\hbar\,\hat L_z
\quad\text{(और चक्रीय)} , \qquad
[\hat L^2, \hat L_z] = 0 :
\]
घटक परस्पर असंगत हैं --- किसी भी अवस्था में उनमें से दो तीखे नहीं होते
--- पर कुल वर्ग उनमें से किसी एक के साथ संगत है: युग्म
$(\hat L^2, \hat L_z)$ नामों का मानक चुनाव है। ये \omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} ठीक
$\iu\hbar$ गुना वे \omterm{def:b3:hamiltonian-mechanics:poisson}{प्वासों कोष्ठक} हैं जो
\cref{ex:b3:hamiltonian-mechanics:brackets} में निकाले गए थे:
अर्थात् डिराक का शब्दकोश काम पर।
\end{definition}

\begin{theorem}[वर्णक्रम, केवल बीजगणित से]\label{thm:b3:quantum-angular-momentum:spectrum}
\index{चुंबकीय क्वांटम संख्या}
मान लीजिए $\hat J_x, \hat J_y, \hat J_z$ \emph{कोई भी} तीन \omterm{def:b3:quantum-formalism:observable}{हर्मिटीय
संकारक} हैं जो ऊपर के क्रमविनिमय-संबंधों का पालन करते हैं। तब
$(\hat J^2, \hat J_z)$ के संयुक्त \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} ये हैं
\[
\hat J^2:\ j(j+1)\hbar^2 , \qquad
\hat J_z:\ m\hbar , \quad m = -j, -j+1, \dots, +j ,
\]
जहाँ $j$ कोई अ-ऋणात्मक \emph{पूर्णांक या अर्ध-पूर्णांक} है: $2j + 1$
मान $m$ के, हर $j$ के लिए। \omterm{def:b3:harmonic-oscillator:ladder}{सोपान संकारक} $\hat J_\pm = \hat
J_x \pm \iu\hat J_y$, $m$ को $\pm1$ जितना बढ़ाते-घटाते हैं, नियत $j$ पर:
\[
\hat J_\pm\ket{j,m} = \hbar\sqrt{j(j+1) - m(m\pm1)}\,\ket{j,m\pm1} .
\]
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
$[\hat J_z, \hat J_\pm] = \pm\hbar\hat J_\pm$: अतः $\hat
J_\pm$ लगाने से $\hat J_z$ का \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm\hbar$ जितना खिसक
जाता है, और $\hat J^2$ नियत रहता है (जो $\hat
J_i$ से बनी हर वस्तु के साथ क्रमविनिमेय है)। प्रसामान्य
$\|\hat J_\pm\ket{j,m}\|^2 = \hbar^2[j(j+1) -
m(m\pm1)]$ ($\hat J_\mp\hat J_\pm = \hat J^2 - \hat
J_z^2 \mp \hbar\hat J_z$ से निकाला हुआ) अ-ऋणात्मक बना ही रहना चाहिए:
अतः सीढ़ी को ऊपर किसी $m_{\max}$ पर समाप्त होना पड़ेगा, जहाँ $m_{\max}(m_{\max}+1) =
j(j+1)$, अर्थात् $m_{\max} = j$, और नीचे $m_{\min} = -j$ पर।
$-j$ से $+j$ तक एकांक पगों में चढ़ना $2j$ को
अ-ऋणात्मक पूर्णांक होने पर विवश कर देता है। और इस तरह \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} का
संकेतन $j(j+1)\hbar^2$ बाद में उचित ठहर जाता है।
\end{proof}

\begin{remark}[सूची में एक रिक्ति]\label{rem:b3:quantum-angular-momentum:halfinteger}
बीजगणित $j = \tfrac12, \tfrac32, \dots$ की अनुमति देता है --- अर्थात्
ऐसी सीढ़ियाँ जिनमें पायदानों की संख्या सम हो। कक्षकीय गति, जैसा हम आगे
दिखाएँगे, केवल पूर्णांक काम में लाती है। पर प्रकृति कुछ भी बरबाद नहीं
करती: इलेक्ट्रॉन स्वयं $j = \tfrac12$ ढोता है, और उसके पीछे कोई कक्षा
नहीं (\cref{ch:b3:spin-two-level}); यहाँ निकाला गया बीजगणित,
अपरिवर्तित, परमाणु-चुंबकत्व, नाभिकीय चक्रण और क्यूबिट चलाएगा।
\end{remark}

\section{कक्षकीय कोणीय संवेग: गोलीय संनादी}

\begin{proposition}[गोलीय संनादी]\label{prop:b3:quantum-angular-momentum:harmonics}
\index{गोलीय संनादी}\index{कक्षकीय क्वांटम संख्या}
गोलीय निर्देशांकों में $\hat L_z = -\iu\hbar\,\partial_\varphi$,
और गोले पर $(\hat L^2, \hat L_z)$ के संयुक्त अभिलक्षणिक फलन
\emph{\omterm{prop:b3:quantum-angular-momentum:harmonics}{गोलीय संनादी}} $Y_l^m(\theta, \varphi)$ हैं:
\[
\hat L^2\,Y_l^m = l(l+1)\hbar^2\,Y_l^m , \qquad
\hat L_z\,Y_l^m = m\hbar\,Y_l^m ,
\]
जहाँ $l = 0, 1, 2, \dots$ \emph{पूर्णांक} है ---
$\eu^{\iu m\varphi}$ की एकमानता $m$ को पूर्णांक होने पर विवश करती है,
और इसलिए $l$ को भी। प्रसामान्यन को छोड़कर, पहले कुछ ये हैं: $Y_0^0 = \text{अचर}$ ($s$
आकार, अर्थात् एक गोला); $Y_1^0 \propto \cos\theta$ और $Y_1^{\pm1}
\propto \sin\theta\,\eu^{\pm\iu\varphi}$ ($p$ आकार, दो पालियाँ);
$Y_2^0 \propto 3\cos^2\theta - 1$ और उसके साथी ($d$ कुल)।
सम-विषमता: $Y_l^m(-\vect r$ दिशा$) = (-1)^l\,Y_l^m$।
\end{proposition}
\admitted

\begin{example}[दृढ़ घूर्णक और उसकी सीढ़ी]\label{ex:b3:quantum-angular-momentum:rotor}
जड़त्व-आघूर्ण $I$ के साथ लुढ़कते किसी द्विपरमाणुक अणु का $\hat H
= \hat L^2/2I$ होता है: ऊर्जाएँ
\[
E_J = \frac{\hbar^2}{2I}\,J(J+1) = B\,J(J+1) ,
\qquad J = 0, 1, 2, \dots
\]
जिनमें से हर एक $(2J+1)$ गुना अपह्रासी है। फ़ोटॉन-अवशोषण $\Delta J =
\pm1$ का पालन करता है, अतः अवशोषण-आवृत्तियाँ $\nu_{J+1\leftarrow J} =
2B(J+1)/h$ हैं: अर्थात् \emph{बराबर दूरी} पर रेखाओं की एक कंघी --- वही
पहचान, जिससे कोई घूर्णन-वर्णक्रम एक ही नज़र में पहचान लिया जाता है। कार्बन
मोनोऑक्साइड के लिए $B/h = \qty{57.6}{GHz}$: मूल रेखा
\qty{115}{GHz} पर पड़ती है ($\lambda = \qty{2.6}{mm}$), जो मिलीमीटर
रेडियो-खगोलिकी की सबसे काम की रेखा है
(\cref{pb:b3:quantum-angular-momentum:1})।
\end{example}

\section{केंद्रीय विभव, पूरे किए हुए}

\begin{theorem}[किसी भी केंद्रीय विभव में पृथक्करण]\label{thm:b3:quantum-angular-momentum:radial}
\index{त्रिज्य समीकरण}\index{अपकेंद्री रोधिका}
$V(r)$ के लिए स्थायी अवस्थाएँ इस रूप में ली जा सकती हैं
\[
\psi(\vect r) = \frac{u(r)}{r}\,Y_l^m(\theta, \varphi) ,
\]
जहाँ $u$ एकविमीय \omterm{prop:b3:schrodinger-three-dimensions:swave}{त्रिज्य समीकरण} हल करता है
\[
-\frac{\hbar^2}{2m}\,u'' + \Big[V(r) +
\frac{\hbar^2\,l(l+1)}{2mr^2}\Big]u = E\,u , \qquad u(0) = 0 .
\]
कोणीय समस्या $Y_l^m$ एक ही बार में, सदा के लिए हल कर देते हैं; और हर
$l$ प्रतिकर्षी \emph{\omterm{thm:b3:quantum-angular-momentum:radial}{अपकेंद्री रोधिका}}
$\hbar^2l(l+1)/2mr^2$ जोड़ देता है --- अर्थात्
\cref{ex:b3:lagrangian-mechanics:central} के प्रभावी विभव का क्वांटम
रूप --- और दिए गए $l$ का हर स्तर $(2l+1)$ गुना अपह्रासी है, क्योंकि
किसी भी केंद्रीय बल को $z$ अक्ष के अभिविन्यास की परवाह नहीं। और
\cref{prop:b3:schrodinger-three-dimensions:swave} की $s$-तरंग वाली
युक्ति इसी प्रमेय की $l = 0$ वाली पंक्ति थी।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
लाप्लासीय इस तरह बँटता है: $\Delta = \tfrac1r\partial_r^2(r\,\cdot) -
\hat L^2/\hbar^2r^2$ (स्वीकृत; यह वही कथन है कि $\hat L^2$
$-\hbar^2\Delta$ का कोणीय भाग है)। $\psi = (u/r)Y_l^m$ रखिए
और $\hat L^2$ का \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} काम में लाइए: कथित समीकरण मिल जाता है।
अपह्रासिता इसलिए है कि $\hat H$ तीनों $\hat
L_i$ के साथ क्रमविनिमेय है, जिनके \omterm{def:b3:harmonic-oscillator:ladder}{सोपान संकारक} ऊर्जा बदले बिना $m$ बदल देते हैं।
\end{proof}

\section{चुंबकीय आघूर्ण: कोणीय संवेग, दिखाई देता हुआ}

\begin{proposition}[कक्षकीय चुंबकीय आघूर्ण और ज़ीमान प्रभाव]\label{prop:b3:quantum-angular-momentum:magnetic}
\index{बोर मैग्नेटॉन}\index{ज़ीमान प्रभाव}
आवेश $q$ और द्रव्यमान $m$ का कोई कण, जिसका कक्षकीय कोणीय संवेग
$\hat{\vect L}$ हो, यह चुंबकीय आघूर्ण ढोता है
\[
\hat{\vect\mu} = \frac{q}{2m}\,\hat{\vect L} ;
\]
इलेक्ट्रॉन के लिए स्वाभाविक मात्रक \emph{\omterm{prop:b3:quantum-angular-momentum:magnetic}{बोर मैग्नेटॉन}}
$\mu_{\text{B}} = e\hbar/2m_{\text{e}} = \qty{5.79e-5}{eV/T}$ है
(तुलना कीजिए \cref{exo:b3:schrodinger-three-dimensions:5})। क्षेत्र
$B\vect e_z$ में ऊर्जा $-\hat{\vect\mu}\cdot\vect B$ हर स्तर को
$+m\,\mu_{\text{B}}B$ जितना खिसका देती है (इलेक्ट्रॉन का आवेश ऋणात्मक
है): दिए गए $l$ का कोई स्तर अपने $2l + 1$ घटकों में बँट जाता है ---
यही \emph{\omterm{prop:b3:quantum-angular-momentum:magnetic}{ज़ीमान प्रभाव}} है, अर्थात् $m$ के क्वांटन का पहला सीधा
प्रदर्शन, और यही कारण है कि $m$ को \omterm{thm:b3:quantum-angular-momentum:spectrum}{चुंबकीय क्वांटम संख्या} कहते हैं।
$B = \qty{1}{T}$ पर विपाटन $\qty{58}{\micro eV}$ है: प्रकाशिक
ऊर्जाओं के आगे छोटा, पर वर्णक्रमीय रेखाओं के खिसकाव के रूप में सहज ही
सुलझने वाला --- और उलटा पढ़ें तो एक चुंबकत्वमापी: सूर्य-प्रकाश की
रेखाओं का ज़ीमान विपाटन सूर्य-कलंकों के चुंबकीय क्षेत्र का नक्शा बना देता है।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
चिरसम्मत रूप से किसी कक्षा पर आवेश एक धारा-पाश है: $\mu = IA =
(qv/2\pi r)(\pi r^2) = qL/2m$; और संकारक वाला कथन उसे उत्तराधिकार में
पाता है (तथा \cref{prop:b3:lagrangian-mechanics:charge} के
$\hat{\vect A}$ युग्मन से निकलता है)। ऊर्जा-खिसकाव प्रथम-कोटि
विक्षोभ सिद्धांत है, जिसकी झलक यहाँ है और औचित्य
\cref{ch:b3:perturbation-theory} में।
\end{proof}

\begin{method}[कोणीय संवेग, व्यवहार में]\label{met:b3:quantum-angular-momentum:method}
(1) अवस्थाओं पर $(l, m)$ के नाम लगाइए --- या अमूर्त बीजगणित के लिए
$(j, m)$ --- और कभी एक साथ दो घटक मत माँगिए। (2) आव्यूह-अवयव:
सोपान-सूत्र काम में लाइए; $\hat L_x = (\hat L_+ + \hat
L_-)/2$। (3) केंद्रीय विभव: $Y_l^m$ उद्धृत कीजिए, और केवल
\omterm{thm:b3:quantum-angular-momentum:radial}{अपकेंद्री रोधिका} वाला \omterm{prop:b3:schrodinger-three-dimensions:swave}{त्रिज्य समीकरण} हल कीजिए। (4) घूर्णन-ऊर्जाएँ:
$BJ(J+1)$, रेखाएँ $2B(J+1)$ पर --- बराबर अंतरालों से $B$, और उससे
बंध-लंबाइयाँ निकालिए। (5) चुंबकीय प्रश्न: कोणीय संवेगों को आघूर्णों में
बदलिए, $\mu_{\text{B}}$ प्रति $\hbar$ की दर से, और ऊर्जाओं को
$\mu_{\text{B}}B$ प्रति एकांक $m$ की दर से।
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:quantum-angular-momentum:4}
कोई कण $l = 2$ की अवस्था में है। (क) $L_z$ नापने के संभव परिणाम

$\cos\theta = m/\sqrt{l(l+1)}$ से: उसका मान निकालिए। (ग) वह कोण
कभी शून्य क्यों नहीं हो सकता? (घ) दिखाइए कि $l \to \infty$ पर न्यूनतम
कोण शून्य की ओर जाता है: अर्थात् चिरसम्मत सदिश लौट आते हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:5}
$-\hbar^2\Delta$ का कोणीय भाग $\hat L^2$ है, जहाँ

\hat L^2 = -\hbar^2\Big[\frac{1}{\sin\theta}\,\partial_\theta
(\sin\theta\,\partial_\theta) +
\frac{1}{\sin^2\theta}\,\partial_\varphi^2\Big] .

(क) जाँचिए कि $Y_1^0 \propto \cos\theta$ का $\hat L^2$-अभिलक्षणिक मान

\eu^{\pm\iu\varphi}$ के लिए भी वही जाँचिए, और उनके $\hat L_z$
\omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} भी। (ग)
सम-विषमताएँ जाँचिए। (घ) कोई भी समाकल निकाले बिना

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:6}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:7}

(घ) अधिकांश वास्तविक रेखाएँ तीन से अधिक घटकों में बँटती हैं

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:9}
$\ket{l, m}$ पर: (क) दिखाइए कि $\langle\hat L_x\rangle = \langle\hat

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-angular-momentum:10}
कंपन--घूर्णन बैंड। किसी द्विपरमाणुक का कोई अवरक्त कंपन-संक्रमण

दिखाइए कि अवशोषण-रेखाएँ $h\nu =

बैंड का रेखाचित्र बनाइए: किसी लुप्त केंद्रीय रेखा के दोनों ओर दो कंघियाँ,

\cref{pb:b3:harmonic-oscillator:1} के \qty{15}{\micro m} बैंड की
ठीक यही शरीर-रचना है: उसकी कुल चौड़ाई कौन तय करता है, और इस तरह वे
``पंख'' भी जो जुड़ी हुई कार्बन डाइऑक्साइड को प्रभावी बनाते हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-angular-momentum:11}
बीजगणित पूरा कीजिए। (क) $\hat J_\mp\hat J_\pm = \hat J^2 -

(क) संरक्षण से छड़ को प्रतिक्षेप करना ही पड़ेगा: प्रति एकांक आयतन

\section{समस्या: 2.6 मिलीमीटर पर मंदाकिनियाँ तौलना}

\begin{problem}[{सप्ताहांत समस्या --- कार्बन मोनोऑक्साइड, रेडियो
खगोलिकी और ठंडा ब्रह्मांड}]\label{pb:b3:quantum-angular-momentum:1}
किसी मंदाकिनी का अधिकांश तारा-निर्माता पदार्थ ठंडा आणविक हाइड्रोजन है
--- और वह अदृश्य है: H$_2$, जो सममित है, न कोई द्विध्रुव रखता है न
ठंडे बादलों के तापों पर कोई घूर्णन-रेखाएँ। खगोलिकी का समाधान है

\qty{115}{GHz} की सीढ़ी के साथ, आकाश के हर आणविक बादल को जगमगा देता
है। आँकड़े: CO के लिए $B/h = \qty{57.6}{GHz}$ ($B =
\qty{0.238}{meV}$); बंध-लंबाई $r_0 = \qty{0.113}{nm}$; समानीत
द्रव्यमान $\mu = \qty{1.14e-26}{kg}$; \qty{10}{K} पर $k_{\text{B}}T$
\qty{0.862}{meV} है।

\medskip\noindent\textbf{भाग I --- क्वांटम घूर्णक।}
\begin{enumerate}
  \item $\hat H = \hat L^2/2I$ से ऊर्जाएँ $E_J$ और उनकी
        अपह्रासिताएँ दीजिए।
  \item CO के लिए $B = \hbar^2/2I$ जाँचिए, $r_0$ और $\mu$ से।
  \item चयन-नियम $\Delta J = \pm1$ के साथ अवशोषण-आवृत्तियाँ
        $\nu_{J+1\leftarrow J} = 2B(J+1)/h$ निकालिए
        और पहली तीन गिनाइए।
  \item रेखाएँ \emph{बराबर दूरी} पर क्यों हैं --- और मापा गया
        अंतराल ($I$ के द्वारा) खगोलविद को क्या दे देता है?
  \item मूल रेखा की तरंगदैर्घ्य निकालिए, और समझाइए कि उसे देखने के
        लिए ऊँचे, सूखे स्थान क्यों चाहिए (हमारे अपने वायुमंडल में
        मिलीमीटर तरंगें कौन सोखता है --- स्मरण कीजिए
        \cref{pb:b3:harmonic-oscillator:1} के अणु का पड़ोसी,
        अर्थात् जल)।
  \item $1 \to 0$ रेखा का फ़ोटॉन $\hbar$ जितना कोणीय
        संवेग ढोता है: $\Delta J = \pm1$ केवल अंगूठे का नियम क्यों
        नहीं, बल्कि एक संरक्षण नियम क्यों है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- CO क्यों और H$_2$ क्यों नहीं।}
\begin{enumerate}[resume]
  \item H$_2$ समनाभिकीय है: बताइए कि वह कोई घूर्णन-द्विध्रुव
        विकिरण क्यों नहीं करता।
  \item H$_2$ \emph{हलका} भी है: H$_2$ का घूर्णन-नियतांक
        ($\mu = m_{\text{p}}/2$, $r_0 = \qty{0.074}{nm}$)
        और उसका पहला सुलभ संक्रमण-ऊर्जा निकालिए; और उसकी तुलना
        \qty{10}{K} पर $k_{\text{B}}T$ से कीजिए: चतुर्ध्रुवी
        पिछले दरवाज़ों से भी, क्या ठंडा H$_2$ विकिरण कर सकता है?
  \item CO का पहला पायदान $2B = \qty{0.48}{meV}$ का पड़ता है, जबकि
        $k_{\text{B}}T = \qty{0.86}{meV}$: क्या सीढ़ी
        \qty{10}{K} पर जीवित है?
  \item \qty{10}{K} पर जनसंख्या-अनुपात $n_1/n_0$ निकालिए
        (अपह्रासिता सहित)।
  \item \qty{10}{K} पर जनसंख्या किस $J$ के पास शिखर पर होती है
        ($J_{\max} \approx \sqrt{k_{\text{B}}T/2B} - \tfrac12$ लीजिए)?
  \item इस सौदे की शर्त एक वाक्य में बताइए: CO क्या देता है, और
        CO-से-H$_2$ अनुपात के विषय में क्या मान लेना पड़ता है।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- आकाश को पढ़ना।}
\begin{enumerate}[resume]
  \item किसी बादल की रेखा प्रयोगशाला की \qty{115.271}{GHz} के बजाय
        \qty{115.156}{GHz} पर पहुँचती है: दृष्टि-रेखा के अनुदिश
        बादल का वेग और उसकी दिशा निकालिए।
  \item वह रेखा भीतरी गतियों $\pm\qty{2}{km/s}$ से डॉप्लर-\emph{चौड़ी}
        हो जाती है: रेखा-चौड़ाई MHz में निकालिए, और उसकी तुलना
        \qty{10}{K} पर अपेक्षित तापीय चौड़ाई से कीजिए
        ($v_{\text{th}} \sim \sqrt{k_{\text{B}}T/m_{\text{CO}}}
        \approx \qty{55}{m/s}$): चौड़ाई पर किसका प्रभुत्व है?
  \item किसी सर्पिल मंदाकिनी की चकती भर CO के डॉप्लर खिसकाव का नक्शा
        बनाने से उसका घूर्णन-वक्र $v(r)$ मिल जाता है। वे वक्र दृश्य
        चकती से बहुत परे तक \emph{सपाट} बने रहते हैं: स्मरण कीजिए
        (वर्ष 1 का खंड, गुरुत्वाकर्षण) कि किसी केंद्र पर संकेंद्रित
        द्रव्यमान के चारों ओर $v(r)$ को क्या करना चाहिए, और बताइए कि
        इस सपाटपन का क्या अर्थ है।
  \item $2\to1$ रेखा की चमक और $1\to0$ की चमक का अनुपात स्तरों की
        जनसंख्या नापता है: वह बादल की कौन-सी एक भौतिक राशि दे देता है?
  \item ALMA उन मंदाकिनियों में CO सुलझा लेता है जिनका प्रकाश तब
        चला था जब ब्रह्मांड युवा था; किसी मंदाकिनी से
        $1 \to 0$ रेखा की प्रेक्षित आवृत्ति, जब वह मंदाकिनी
        ``लाल-विस्थापन $z = 1$'' पर हो, अपने विराम-मान की आधी आती है
        --- किस आवृत्ति पर, और
        (\cref{ex:b3:relativistic-kinematics:redshift} स्मरण करते
        हुए) उसे किसने खींचा?
  \item मापी गई CO रेखा-ज्योति से खगोलविद बादलों के द्रव्यमान
        सौर-द्रव्यमानों में बताते हैं: रूपांतरणों की शृंखला गिनाइए
        (रेखा के फ़ोटॉन $\to$ CO की गिनती $\to$ H$_2$ द्रव्यमान)
        और उसकी सबसे कमज़ोर कड़ी भी।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- ठंडी नर्सरियाँ।}
\begin{enumerate}[resume]
  \item तारा-निर्माता क्रोड \qty{10}{K} पर बैठे हैं: वीन के नियम से
        उनकी तापीय चमक का शिखर \qty{290}{\micro m} के पास होता है ---
        जो हर प्रकाशिक दूरदर्शी के लिए अदृश्य है। एक वाक्य में: ऐसा
        कोई क्रोड जो \emph{एकमात्र} तापमापी और तराज़ू देता है, वह
        घूर्णन-सीढ़ी ही क्यों है?
  \item स्तर $J = 1$ मूल अवस्था से \qty{5.5}{K} (ताप के मात्रकों में)
        ऊपर पड़ता है: इससे $1\to0$ रेखा ठीक \qty{10}{K} वाले परिसर
        में इतना उत्तम तापमापी क्यों बन जाती है (और, मान लीजिए,
        \qty{2}{eV} $\sim \qty{23000}{K}$ की कोई प्रकाशिक रेखा
        क्यों नहीं)?
  \item कोई \emph{ढहता} क्रोड अंततः CO ($1\to0$) में दिखना क्यों बंद
        कर देता है --- सोचिए कि ऊँचा घनत्व और धूल उस रेखा तथा उसके
        बाहर निकलने के साथ क्या करते हैं।
  \item आणविक बादल NH$_3$, HCN और दर्जनों अन्य अणुओं की रेखाएँ भी
        दिखाते हैं, जिनमें से हर एक का अपना $B$ और अपना द्विध्रुव है:
        इस ``रासायनिक रेडियो डायल'' की समृद्धि खगोलविदों को क्या
        अलग-अलग करके देखने देती है?
  \item पूरा आकाश \qty{2.7}{K} पर चमकता है (ब्रह्मांडीय सूक्ष्मतरंग
        पृष्ठभूमि): जब किसी बादल का अपना ताप \qty{2.7}{K} के पास आने
        लगे, तब उसकी CO रेखाओं के विपर्यास का क्या होता है, और यह इस
        तापमापी के लिए कठोर फ़र्श क्यों है?
  \item किसी मिलीमीटर-थाली को अपना आकार लगभग
        $\lambda/20$ तक बनाए रखना पड़ता है: \qty{115}{GHz} पर वह
        सहनशीलता निकालिए, और उसकी तुलना उस सहनशीलता से कीजिए जो किसी
        प्रकाशिक दर्पण (\qty{500}{nm}) को झेलनी पड़ती है --- ALMA की
        बारह-मीटर थालियाँ खुली हवा में क्यों बन सकीं?
  \item नामित परिणाम का सार लिखिए: सीढ़ी $BJ(J+1)$, जिसका $B/h =
        \qty{57.6}{GHz}$ है, और जो दस केल्विन पर
        जनसंख्या पाती है और \qty{2.6}{mm} पर पढ़ी जाती है, वही तरीका
        है जिससे ठंडे ब्रह्मांड का द्रव्यमान, ताप, वेग और घूर्णन ---
        तथा सर्पिल मंदाकिनियों के चारों ओर श्याम पदार्थ का प्रमाण ---
        नापे जाते हैं।
\end{enumerate}
\end{problem}
