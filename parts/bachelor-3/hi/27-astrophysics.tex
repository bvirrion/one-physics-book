\section{तारा एक संतुलन है}

\begin{proposition}[द्रवस्थैतिक संतुलन]\label{prop:b3:astrophysics:hydrostatic}
\index{द्रवस्थैतिक संतुलन (तारकीय)}\index{तारकीय संरचना}
कोई तारा न ढहता है न बिखरता: हर त्रिज्या पर दाब-प्रवणता अपने ऊपर की
गैस का भार ढोती है। द्रव्यमान $m(r)$ के लिए, जो त्रिज्या
$r$ के भीतर है,
\[
\frac{\dd P}{\dd r} = -\frac{Gm(r)\rho(r)}{r^2} .
\]
पूरे सूर्य पर आकलित करने पर $P_{\text{c}} \sim GM^2/R^4 \sim
\qty{e15}{Pa}$, और आदर्श गैस के नियम से $T_{\text{c}} \sim
10^7\,\unit{K}$ (\cref{exo:b3:astrophysics:1}) --- यानी ठीक वही
ताप जिस पर \cref{exo:b3:nuclear-physics:11} के प्रोटॉन
सुरंग बनाने लगते हैं। यह कोई संयोग नहीं, एक तापस्थापी है: किसी तारे को
सिकुड़ना और गरम होना \emph{ही पड़ता} है जब तक उसका क्रोड न सुलग जाए,
और तब संलयन लाखों से अरबों वर्षों तक वह संतुलन थामे रहता है। तारों की
ऊष्माधारिता ऋणात्मक होती है --- ऊर्जा खोइए, सिकुड़िए, और
\emph{अधिक गरम} हो जाइए (\cref{exo:b3:astrophysics:5}) --- और
इसीलिए अंततः गुरुत्व ही जीतता है।
\end{proposition}

\begin{proof}
कोई खोल $[r, r + \dd r]$, जिसका क्षेत्रफल $A$ है, नीचे से दाब $P(r)A$
अनुभव करता है, ऊपर से $P(r + \dd r)A$, और भार $g\rho A\,\dd r$,
जहाँ $g = Gm(r)/r^2$ (गुरुत्व के लिए गाउस की प्रमेय से केवल भीतरी
द्रव्यमान खींचता है)। इन तीनों का संतुलन वही कथित
समीकरण देता है।
\end{proof}

\section{तारों का जीवन}

\begin{definition}[हर्ट्ज़्स्प्रुंग--रसेल आरेख]\label{def:b3:astrophysics:hr}
\index{हर्ट्ज़्स्प्रुंग--रसेल आरेख}\index{मुख्य अनुक्रम}
तारों को सतही ताप (परंपरा से गरम \emph{बाईं} ओर) के सामने
ज्योति के साथ आलेखित कीजिए, और दोनों
\cref{ch:b3:photons-phonons} के कृष्णिका-नियमों से लीजिए, तो वे
बिखरने से इनकार कर देते हैं: नब्बे प्रतिशत \emph{मुख्य अनुक्रम} पर
भीड़ लगा देते हैं, यानी हाइड्रोजन जलाने वालों की उस विकर्ण पट्टी पर,
जो विशुद्ध रूप से द्रव्यमान से क्रमित है --- भारी का अर्थ है अधिक गरम
\emph{और} अनुपात से कहीं अधिक चमकीला
($L \propto M^{3.5}$), और इस तरह अल्पजीवी
($t \propto M/L \propto M^{-2.5}$:
\cref{exo:b3:astrophysics:4})। उस पट्टी से बाहर आरेख के
अपवाद और अंत रहते हैं: ठंडे पर विशाल \emph{दानव}
(ऊपर दाएँ), और गरम पर नन्हे \emph{\omterm{ex:b3:quantum-statistics:dwarf}{श्वेत वामन}} (नीचे बाएँ)
--- क्योंकि स्थिर ताप पर, $L = 4\pi R^2\sigma T^4$ से,
ज्योति दरअसल \emph{आकार} की नाप है। HR आरेख किस्मों का नहीं,
\emph{आयु} का नक्शा है: किसी तारे का बिंदु ईंधन चुकते-चुकते उस पर
सरकता रहता है।
\end{definition}

\begin{remark}[तारे कैसे बूढ़े होते हैं, और क्या छोड़ जाते हैं]\label{rem:b3:astrophysics:evolution}
\index{लाल दानव}\index{अधिनवतारा}\index{तारकीय नाभिक-संश्लेषण}
जब क्रोड का हाइड्रोजन चुक जाता है, तब तापस्थापी का तर्क फिर चलता है:
क्रोड सिकुड़कर गरम होता है, हीलियम सुलग उठती है (कार्बन और ऑक्सीजन में
संलयित होकर), और आवरण सौ गुना फूल जाता है --- यानी कोई \omterm{rem:b3:astrophysics:evolution}{लाल दानव}।
सूर्य जैसा तारा वहीं रुक जाता है: वह अपना आवरण झाड़ देता है और
\omterm{ex:b3:quantum-statistics:dwarf}{श्वेत वामन} बनकर सेवानिवृत्त हो जाता है। पर $\approx8M_\odot$ से ऊपर का
कोई तारा प्याज़-खोलों में नए-नए ईंधन सुलगाता जाता है, जब तक क्रोड
\emph{लोहा} न बन जाए --- यानी \cref{prop:b3:nuclear-physics:binding}
की वह चोटी, जहाँ से कोई संलयन ऊर्जा नहीं लौटाता। वह मरा हुआ क्रोड एक
सेकंड में ढह जाता है; और पलटाव तथा न्यूट्रिनो की बाढ़ उस तारे को उड़ा
देती है: एक \emph{अधिनवतारा}, जो थोड़ी देर को अपनी पूरी आकाशगंगा से
अधिक चमकता है, न्यूट्रॉन-परिग्रहण से लोहे के आगे के तत्त्व गढ़ता है,
और वह सब उन बादलों में बिखेर देता है जो तारों तथा ग्रहों की अगली पीढ़ी
बनाते हैं। आपकी हड्डियों का कैल्सियम और आपके रक्त का लोहा
इसी रास्ते आए (\cref{pb:b3:astrophysics:1}): रसायन दरअसल
पुनर्चक्रित खगोल-भौतिकी है।
\end{remark}

\section{लाशें: अपनी सीमाओं पर पदार्थ}

\begin{proposition}[श्वेत वामन और न्यूट्रॉन तारे]\label{prop:b3:astrophysics:compact}
\index{श्वेत वामन}\index{न्यूट्रॉन तारा}\index{चंद्रशेखर द्रव्यमान}\index{कृष्ण विवर}\index{श्वार्ज़्शिल्ड त्रिज्या}
जो तारा अब नहीं जलता, उसे थामे कौन रखता है? क्वांटम यांत्रिकी। कोई
\emph{\omterm{ex:b3:quantum-statistics:dwarf}{श्वेत वामन}} कार्बन और ऑक्सीजन का एक सौर द्रव्यमान है, जिसे उसका
\emph{इलेक्ट्रॉन \omterm{thm:b3:quantum-statistics:fermi}{अपह्रासन दाब}} थामे रहता है --- यानी
\cref{ch:b3:quantum-statistics} का $T = 0$ वाला फ़र्मी दाब --- और वह
भी पृथ्वी के आकार पर और प्रति चम्मच एक टन के घनत्व पर। पर पाउली के
सहारे की एक छत है: \emph{\omterm{prop:b3:astrophysics:compact}{चंद्रशेखर द्रव्यमान}} $M_{\text{Ch}}
\approx 1.4\,M_\odot$ के ऊपर वे इलेक्ट्रॉन आपेक्षिकीय हो जाते हैं और
वह दाब दौड़ हार जाता है (यह परिणाम हम बताते हैं, व्युत्पन्न नहीं
करते)। उसके आगे इलेक्ट्रॉन कुचलकर प्रोटॉनों में घुस जाते हैं और कोई
\emph{\omterm{prop:b3:astrophysics:compact}{न्यूट्रॉन तारा}} बनता है: यानी न्यूट्रॉन अपह्रासन और नाभिकीय
प्रतिकर्षण, जो $\sim1.4\,M_\odot$ को
\cref{def:b3:nuclear-physics:nuclide} वाले नाभिकीय घनत्व पर दर्जन भर
किलोमीटरों में थामे रखते हैं --- यानी शहर के आकार का कोई नाभिक, जो
मिलीसेकंडों तक घूमता है और किसी प्रकाश-स्तंभ की तरह किरण फेंकता है
(कोई \emph{पल्सर})। और लगभग $3\,M_\odot$ से आगे कोई ज्ञात चीज़ नहीं
टिकती: वह तारा उस त्रिज्या के भीतर गिर जाता है जिस पर
पलायन के लिए प्रकाश की गति चाहिए,
\[
R_{\text{s}} = \frac{2GM}{c^2}
\qquad (\qty{3}{km}\text{ प्रति सौर द्रव्यमान}) ,
\]
यानी कोई \emph{\omterm{prop:b3:astrophysics:compact}{कृष्ण विवर}} --- गुरुत्व का अंतिम शब्द, और सामान्य
आपेक्षिकता की देहरी, जो इस खंड से एक खंड आगे है।
\end{proposition}
\admitted

\section{फैलता ब्रह्मांड}

\begin{definition}[हबल का नियम और तप्त आरंभ]\label{def:b3:astrophysics:hubble}
\index{हबल का नियम}\index{ब्रह्मांडीय सूक्ष्मतरंग पृष्ठभूमि}\index{महाविस्फोट नाभिक-संश्लेषण}
हर दूर की आकाशगंगा पीछे हटती है, और वह भी दूरी के अनुक्रमानुपाती चाल से:
\[
v = H_0d , \qquad H_0 \approx \qty{70}{km/s}
\text{ प्रति मेगापारसेक} .
\]
यह समष्टि \emph{में} कोई विस्फोट नहीं, बल्कि स्वयं समष्टि का
\emph{प्रसार} है --- उसे उलटा चलाइए और कभी सब कुछ
सघन तथा तप्त था: $1/H_0 \approx 14$ अरब वर्ष पहले
(\cref{exo:b3:astrophysics:9})। आरंभिक ब्रह्मांड यही पुस्तक है,
तेज़ी से दोहराई हुई। एक सेकंड पर वह \unit{MeV} का कण-शोरबा है
(\cref{ch:b3:particle-physics}); पहले मिनटों में
\cref{ch:b3:nuclear-physics} उन्मत्त होकर चलता है और
\emph{द्रव्यमान से एक चौथाई हीलियम} जमा देता है --- जिसकी भविष्यवाणी
किसी बोल्ट्ज़मान गुणक ने की और जो हर जगह प्रेक्षित है
(\cref{exo:b3:astrophysics:11})। 380\,000 वर्ष और
\qty{3000}{K} पर साहा का समीकरण
(\cref{prop:b3:grand-canonical:saha}) इलेक्ट्रॉनों को आख़िरकार
प्रोटॉनों पर बैठने देता है; कुहरा छँट जाता है, और अंतिम बार प्रकीर्ण
हुआ वह प्रकाश --- प्रसार से हज़ार गुना खिंचकर --- आज
\emph{ब्रह्मांडीय सूक्ष्मतरंग पृष्ठभूमि} के रूप में आता है: यानी
\qty{2.725}{K} का प्लांक वर्णक्रम (\cref{ch:b3:photons-phonons}),
जो अब तक का सबसे पुराना छायाचित्र है। वह चित्र यह भी दिखाता है:
आकाशगंगाएँ ऐसे घूमती और गुच्छित होती हैं मानो उन्हें ऐसा
\emph{\omterm{rem:b3:particle-physics:higgs}{कृष्ण पदार्थ}} चला रहा हो जिसे कोई संसूचक पकड़ नहीं सका, और
वह प्रसार किसी ऐसी \emph{कृष्ण ऊर्जा} के नीचे \emph{त्वरित} हो रहा है
जिसे किसी ने समझाया नहीं --- यानी उस बजट का पंचानबे प्रतिशत,
जिसे यह पुस्तक अभी लिख नहीं सकती।
\end{definition}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:1}
सूर्य, आकलित। $M = \qty{2e30}{kg}$, $R = \qty{7e8}{m}$।
(क) $P_{\text{c}} \sim GM^2/R^4$ आकलिए। (ख) माध्य
घनत्व और आदर्श गैस के नियम ($P = \rho k_{\text{B}}T/
m_{\text{p}}$) से $T_{\text{c}}$ आकलिए। (ग)
\cref{exo:b3:nuclear-physics:11} की
$10^7\,\unit{K}$ वाली प्रज्वलन-देहली से तुलिए। (घ) यह सहमति कोई
संयोग नहीं बल्कि तापस्थापी क्यों है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:2}
प्रकाश को तौलना। (क) $d = \qty{1.5e11}{m}$ पर सौर नियतांक
\qty{1360}{W/m^2} से
$L_\odot$ निकालिए। (ख) $L = 4\pi R^2\sigma T^4$ से
सतही ताप निकालिए। (ग) वीन के शिखर को सूर्य के
दृश्य पीले-हरे के सामने जाँचिए। (घ) हर चरण ने किस अध्याय का नियम काम में लिया?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:3}
आरेख पढ़ना। सीरियस B: $T \approx \qty{25000}{K}$, $L
\approx 0.026\,L_\odot$; बेटेल्ज़्यूस: $T \approx \qty{3500}{K}$,
$L \approx 10^5\,L_\odot$। (क) दोनों को HR आरेख पर रखिए। (ख)
$L = 4\pi R^2\sigma T^4$ से हर एक की त्रिज्या सौर
इकाइयों में निकालिए। (ग) बेटेल्ज़्यूस को सौर मंडल में रखिए: कौन-से
ग्रह उसके भीतर आ जाते हैं? (घ) सीरियस B को क्या थामे होना चाहिए?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:astrophysics:4}
तेज़ जियो, जल्दी मरो। $L \propto M^{3.5}$ और जीवनकाल
$t \propto M/L$ के साथ, तथा सूर्य के दस अरब वर्षों से: (क) किसी
$10\,M_\odot$ तारे का जीवनकाल; (ख) किसी $0.5\,M_\odot$ लाल वामन का ---
और उसकी तुलना ब्रह्मांड की आयु से; (ग) भारी तारे, जिनके पास
\emph{अधिक} ईंधन है, \emph{जल्दी} क्यों मरते हैं? (घ) भारी जीवनों का
छोटा होना इस बारे में क्या कहता है कि किसी आकाशगंगा में अधिनवतारे
कहाँ होते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:5}
गुरुत्व की अजीब ऊष्मागतिकी। किसी स्वयं-गुरुत्वित गैस-गोले के लिए
विरियल प्रमेय (\cref{ch:b3:lagrangian-mechanics})
कुल ऊर्जा $E = -K$ देती है (यानी ऋण गतिज/तापीय ऊर्जा)।
(क) निष्कर्ष निकालिए: ऊर्जा विकिरित करना उस तारे को \emph{अधिक गरम}
कर देता है --- यानी ऋणात्मक ऊष्माधारिता। (ख) सूर्य की गुरुत्वीय
ऊर्जा $GM^2/R$ आकलिए। (ग) आज के $L_\odot$ पर अकेला संकुचन उसे कब तक
शक्ति दे सकता था (केल्विन--हेल्महोल्ट्ज़
काल)? (घ) भूविज्ञान पृथ्वी की आयु 4.5 अरब वर्ष बताता है:
उन्नीसवीं सदी का विरोधाभास और उसका नाभिकीय समाधान बताइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:6}
किसी \omterm{ex:b3:quantum-statistics:dwarf}{श्वेत वामन} की शरीर-रचना। $M = 1\,M_\odot$ लीजिए,
$R = R_{\text{पृथ्वी}} = \qty{6.4e6}{m}$ पर। (क) माध्य
घनत्व और एक चम्मच (\qty{5}{mL}) का द्रव्यमान निकालिए। (ख) सतही
गुरुत्व निकालिए। (ग) \Cref{ch:b3:quantum-statistics} का
\omterm{thm:b3:quantum-statistics:fermi}{अपह्रासन दाब} $n^{5/3}$ की तरह चलता है और गुरुत्व की माँग
$M^2/R^4 \propto n^{4/3}M^{2/3}$ की तरह: दिखाइए कि इससे वह
विचित्र $R \propto M^{-1/3}$ निकलता है --- यानी भारी वामन
\emph{छोटे} होते हैं। (घ) भौतिक रूप से समझाइए कि द्रव्यमान लादने से
वह तारा क्यों सिकुड़ता है, और \omterm{prop:b3:astrophysics:compact}{चंद्रशेखर द्रव्यमान} पर वह किस बात की
आहट है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:7}
किसी न्यूट्रॉन तारे की शरीर-रचना। (क) नाभिकीय घनत्व
$\qty{2.3e17}{kg/m^3}$ पर $1.4\,M_\odot$ थामने वाली त्रिज्या
निकालिए। (ख) पतन से पहले वह क्रोड $R \sim 10^5\,\unit{km}$ पर 25 दिन
में एक बार घूमता था: कोणीय-संवेग के संरक्षण से
अंतिम घूर्णन-आवर्त आकलिए। (ग) सतही
गुरुत्व निकालिए। (घ) अभिवाह-जमाव चुंबकीय क्षेत्र को
$B \propto 1/R^2$ की तरह बढ़ा देता है
(\cref{ch:b3:electromagnetism-in-matter}):
किसी तारकीय \qty{e-2}{T} से कौन-सा सतही क्षेत्र निकलता है, और
पल्सर किरण क्यों फेंकते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:astrophysics:8}
वापसी का बिंदु नहीं। (क) $R_{\text{s}} = 2GM/c^2$ व्युत्पन्न कीजिए,
न्यूटनी पलायन-चाल की शर्त $v_{\text{पल}} = c$ से
(ईमानदार व्युत्पत्ति को सामान्य आपेक्षिकता चाहिए; पर उत्तर,
प्रसिद्ध रूप से, वही रहता है)। (ख) सूर्य और पृथ्वी के लिए मान निकालिए।
(ग) आकाशगंगा का केंद्रीय \omterm{prop:b3:astrophysics:compact}{कृष्ण विवर}: $M = 4\times10^6\,M_\odot$
--- $R_{\text{s}}$ निकालिए और सूर्य--बुध
दूरी ($\qty{5.8e10}{m}$) से तुलिए। (घ) दिखाइए कि
$R_{\text{s}}$ के भीतर माध्य घनत्व $1/M^2$ की तरह गिरता है --- यानी
पर्याप्त बड़ा कोई \omterm{prop:b3:astrophysics:compact}{कृष्ण विवर} पानी से भी कम सघन है: यह
``कुचल देने'' वाले अंतर्ज्ञानों के बारे में क्या कहता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:9}
हबल का अंकगणित। $H_0 = \qty{70}{km/s}$ प्रति \unit{Mpc}; $1\,
\text{मेगापारसेक} = \qty{3.09e22}{m}$। (क) $H_0$ को SI में बदलिए। (ख)
$1/H_0$ को वर्षों में निकालिए। (ग) \qty{200}{Mpc} पर कोई आकाशगंगा: उसकी
पीछे हटने की चाल, और वह गुणक भी जिससे उसके प्रकाश की तरंगदैर्घ्य
खिंच चुकी है (छोटे-$z$ का डॉप्लर,
\cref{ch:b3:relativistic-kinematics})। (घ) $1/H_0$ ब्रह्मांड की आयु
केवल लगभग ही क्यों है --- इस बीच वह प्रसार-दर
क्या करती रही है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:10}
सबसे पुराना प्रकाश। CMB एक \qty{2.725}{K} की कृष्णिका है। (क)
उसकी शिखर-तरंगदैर्घ्य निकालिए (\cref{ch:b3:photons-phonons}) और
``सूक्ष्मतरंग'' कहना उचित ठहराइए। (ख) वह
\qty{3000}{K} के प्रकाश के रूप में उत्सर्जित हुई थी: किस प्रसार-गुणक
ने उसे खींचा है, और \qty{3000}{K} ही क्यों (साहा याद कीजिए,
\cref{prop:b3:grand-canonical:saha})? (ग) उसका फ़ोटॉन घनत्व
$\sim\qty{4e8}{m^{-3}}$ है: बैरिऑनों के घनत्व
$\sim\qty{0.25}{m^{-3}}$ से तुलिए और फ़ोटॉन-से-बैरिऑन अनुपात बताइए।
(घ) मेज़ पर की एक चिरसम्मत जाँच: किसी बेसुरे अनुरूप
TV के स्थैतिक शोर का कुछ प्रतिशत CMB था --- कोई ऐंटेना, कहीं भी
मोड़ा जाए, उसे क्यों पकड़ लेता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:11}
किसी बोल्ट्ज़मान गुणक से एक चौथाई हीलियम। लगभग $t \sim
\qty{1}{s}$, $T \sim \qty{e10}{K}$ ($k_{\text{B}}T \approx
\qty{0.8}{MeV}$) पर जमती दुर्बल अभिक्रियाओं ने
न्यूट्रॉन-से-प्रोटॉन अनुपात अपने बोल्ट्ज़मान मान पर छोड़ दिया
(\cref{ch:b3:canonical-ensemble})। (क) $\Delta mc^2 =
\qty{1.29}{MeV}$ के साथ जमाव पर $n/p$ निकालिए। (ख) पहले मिनटों में
न्यूट्रॉन का क्षय उसे $\approx1/7$ तक छाँट गया: दिखाइए कि हर बचे हुए
को $^4$He में बाँधने पर हीलियम का \emph{द्रव्यमान}-अंश
$2(n/p)/(1 + n/p) = 25\,\%$ मिलता है। (ग) बताइए कि हर जगह मापी गई
यह संख्या तप्त आरंभ का इतना प्रबल साक्ष्य क्यों है। (घ) आदिम पकाई
हीलियम पर ही क्यों रुक गई (लीथियम के अंशों के साथ), और
कार्बन तथा बाकी सब \cref{rem:b3:astrophysics:evolution} के लिए क्यों छूट गए?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:astrophysics:12}
लापता पदार्थ। किसी सर्पिल आकाशगंगा की घूर्णन-चाल
$v = \qty{220}{km/s}$ पर सपाट बनी रहती है, और वह भी
$r = \qty{50}{kpc}$ ($\qty{1.5e21}{m}$) तक, यानी उसके दृश्य किनारे से
कहीं आगे तक। (क) किसी वृत्तीय कक्षा के लिए दिखाइए कि $M(r) = v^2r/G$।
(ख) \qty{50}{kpc} पर उसका मान सौर द्रव्यमानों में निकालिए। (ग)
दृश्य तारे और गैस मिलकर $\sim5\times10^{10}\,M_\odot$ हैं: कितना अंश
समझाया जा सका? (घ) दृश्य किनारे के आगे केप्लरी अपेक्षा
$v \propto r^{-1/2}$ है (यानी सौर मंडल का ढाँचा): वह सपाटपन इस बारे
में क्या कहता है कि वह अदृश्य द्रव्यमान कैसे बँटा है?
\end{exercise}

\section{समस्या: जिस रात वह तारा मरा}

\begin{problem}\label{pb:b3:astrophysics:1}
\emph{जिस रात वह तारा मरा।} 23 फ़रवरी 1987 को बृहत् मैगेलानी मेघ का
एक नीला महादानव --- $d \approx
\qty{50}{kpc} \approx \qty{1.5e21}{m}$ --- SN~1987A बन गया, यानी
चार सदियों का सबसे निकटतम अधिनवतारा। किसी भी दूरदर्शी के उसे चमकते
देखने से घंटों \emph{पहले} तीन भूमिगत संसूचकों ने
दो दर्जन न्यूट्रिनो गिन लिए। आप उस रात को मूल सिद्धांतों से फिर
गढ़ रहे हैं।

\medskip\noindent\textbf{भाग I --- वह तारा, जो था।}

\begin{enumerate}
  \item पूर्वज का भार ${\sim}20\,M_\odot$ था।
        \cref{exo:b3:astrophysics:4} के मापन से उसका
        जीवनकाल आकलिए और सूर्य से तुलिए।
  \item अपने अंतिम दिन तक वह तारा एक प्याज़ था: लोहे के क्रोड के
        चारों ओर H, He, C, O, Si के खोल। हर अगले
        ईंधन को अधिक गरम क्रोड क्यों चाहिए?
  \item वह क्रम लोहे पर हमेशा के लिए क्यों रुक जाता है?
        (\cref{prop:b3:nuclear-physics:binding}।)
  \item वह निष्क्रिय लोहे का क्रोड $1.4\,M_\odot$ की ओर बढ़ता है।
        उस संख्या में क्या खास है, और वहाँ कौन-सा दाब विफल
        हो जाता है?
  \item सिलिकॉन का जलना उस तारे को लगभग एक \emph{दिन} तक थामे
        रखता है: यह इस बारे में क्या कहता है कि अंत की ओर किसी तारे
        की घड़ी कैसे तेज़ हो जाती है?
  \item एक बार दाब का सहारा विफल हो जाए, तो घनत्व
        $\sim\qty{e12}{kg/m^3}$ वाले किसी क्रोड के लिए मुक्त पतन में
        मोटे तौर पर कितना समय लगता है ($t \sim 1/\sqrt{G\rho}$)?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- न्यूट्रिनो की कौंध।}

\begin{enumerate}[resume]
  \item वह क्रोड ($M \approx 1.4\,M_\odot = \qty{2.8e30}{kg}$)
        ढहकर $R \approx \qty{12}{km}$ हो जाता है: मुक्त हुई
        गुरुत्वीय ऊर्जा $E \sim GM^2/R$ निकालिए।
  \item उसकी तुलना अरब सूर्यों के महीने भर चमकने से कीजिए
        ($L \sim 10^{9}L_\odot$, $\qty{2.6e6}{s}$ के लिए):
        दिखाइए कि फ़ोटॉन एक पूर्णांकन-त्रुटि भर हैं --- वे
        99\,\% कहाँ जाते हैं, और किसी ढहते क्रोड से तुरंत केवल
        न्यूट्रिनो ही क्यों निकल सकते हैं?
  \item $E \approx \qty{3e46}{J}$ को त्रिज्या
        \qty{1.5e21}{m} के किसी गोले पर फैलाइए: पृथ्वी पर
        ऊर्जा-प्रवाह निकालिए।
  \item प्रति न्यूट्रिनो $\sim\qty{15}{MeV}$ के साथ
        संख्या-प्रवाह (प्रति \unit{m^2} न्यूट्रिनो) निकालिए।
  \item किसी संसूचक में \qty{2000}{t} पानी है, यानी
        ${\sim}1.5\times10^{32}$ मुक्त प्रोटॉन। प्रति
        प्रति-न्यूट्रिनो प्रति प्रोटॉन अन्योन्यक्रिया की प्रायिकता
        $\sigma \approx \qty{1e-46}{m^2}$ दी हो, तो
        अपेक्षित गिनती आकलिए --- और उसकी तुलना उन ग्यारह से कीजिए
        जो एक संसूचक ने दर्ज कीं।
  \item वे न्यूट्रिनो प्रकाश से घंटों आगे निकल गए। दोनों घड़ियाँ
        समझाइए: फ़ोटॉनों को कौन रोकता है (वह चमक असल में जन्मती कहाँ
        है?), और वह
        निकट-समकालिकता न्यूट्रिनो के द्रव्यमान तथा
        चाल के बारे में क्या कहती है?
  \item गिने गए पच्चीस न्यूट्रिनो: एक-एक वाक्य में दो बातें बताइए
        जिनकी उन्होंने अधिनवतारा-सिद्धांत के बारे में पुष्टि की।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- जो बचा रहा।}

\begin{enumerate}[resume]
  \item \qty{12}{km}, $1.4\,M_\odot$ वाले उस अवशेष का माध्य घनत्व
        निकालिए और \cref{def:b3:nuclear-physics:nuclide} के
        नाभिकीय घनत्व से
        तुलिए।
  \item पूर्वज का क्रोड $R \sim \qty{e5}{km}$ पर 30 दिन में एक बार
        घूमता था: उस अवशेष का आवर्त आकलिए।
  \item उसका चुंबकीय क्षेत्र, अभिवाह-जमा, $(R_0/R)^2$ से बढ़ जाता
        है: \qty{e-2}{T} से सतही
        क्षेत्र निकालिए।
  \item प्रकाश-स्तंभ समझाइए: क्या घूमता है, क्या किरण फेंकता है, और
        क्रैब पल्सर (जो 1054 में जन्मा और दिन में दिखते ``अतिथि
        तारे'' के रूप में दर्ज हुआ) आज भी प्रति सेकंड तीस बार क्यों
        कौंधता है।
  \item यदि वह अवशेष ${\sim}3\,M_\odot$ से अधिक होता: $R_{\text{s}}$
        निकालिए, $3\,M_\odot$ के लिए, और उसकी नियति बताइए।
  \item पल्सर का समय $10^{-15}$ तक स्थिर रहता है: आकाश में घूमती
        इतनी अच्छी किसी घड़ी का एक भौतिकी-उपयोग बताइए।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- इसका अर्थ।}

\begin{enumerate}[resume]
  \item जिस ऑक्सीजन में आप साँस लेते हैं और आपके रक्त का लोहा:
        महाविस्फोट की हाइड्रोजन से लेकर आपकी रक्त-धारा तक हर परमाणु
        की जीवनी दो वाक्यों में बताइए।
  \item Ia प्रकार के अधिनवतारे ($M_{\text{Ch}}$ के पार धकेले गए
        \omterm{ex:b3:quantum-statistics:dwarf}{श्वेत वामन}) सब एक ही द्रव्यमान पर फटते हैं, और इसलिए
        लगभग एक ही चमक के होते हैं। समझाइए कि इससे वे
        \emph{मानक दीपक} क्यों बन जाते हैं, और
        \cref{def:b3:astrophysics:hubble} का आरेख उन्हीं से कैसे
        बनता है।
  \item 1998 में दूर के Ia अधिनवतारे हबल की सीधी रेखा की
        भविष्यवाणी से \emph{अधिक धुँधले} निकले: वह निष्कर्ष बताइए
        जिसे 2011 का नोबेल पुरस्कार मिला।
  \item आकलिए कि SN~1987A का प्रकाश पहुँचने तक कितने वर्ष चल चुका
        था --- और यह भी बताइए कि जब वह चला था तब पृथ्वी पर क्या
        हो रहा था।
  \item वे मैगेलानी न्यूट्रिनो आपके भीतर से भी गुज़रे (उस दिन जीवित
        हर व्यक्ति के भीतर से)। हर मनुष्य
        (${\sim}\qty{0.03}{m^2}$) से मोटे तौर पर कितने पार हुए? क्या
        किसी को कुछ महसूस हुआ (\cref{exo:b3:particle-physics:10})?
  \item पुस्तक का बहीखाता चार पंक्तियों में बंद कीजिए: कोई तारा, जो
        एक करोड़ वर्ष तक सुरंगन से संतुलित रहा; कोई पतन, जिसका
        भुगतान न्यूट्रिनो में हुआ; कोई लाश, जिसे पाउली थामे है; और
        वह मलबा --- जिसमें पाठक भी शामिल है --- जो आज भी
        हबल के प्रसार पर सवार है।
\end{enumerate}
\end{problem}
