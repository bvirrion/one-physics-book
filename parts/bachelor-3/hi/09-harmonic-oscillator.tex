\chapter{क्वांटम हरात्मक दोलक}\label{ch:b3:harmonic-oscillator}

प्रकृति में लगभग कुछ भी ठीक-ठीक हरात्मक दोलक नहीं है, और लगभग सब कुछ
लगभग वही है: स्थायी साम्य से ज़रा हटाया गया कोई भी तंत्र --- किसी अणु का
बंध, किसी क्रिस्टल में कोई परमाणु, कोई पुल, विद्युत्चुंबकीय क्षेत्र की
कोई विधा --- विस्थापन के अनुक्रमानुपाती प्रत्यानयन बल अनुभव करता है,
क्योंकि हर चिकना विभव अपने कूप के तल पर एक परवलय होता है। जो एक बार
क्वांटम दोलक हल कर लेता है, वह इस तरह पूरे संसार के लघु कंपन हल कर
लेता है। यह अध्याय उसे उसी बीजगणितीय शैली में हल करता है जो क्वांटम
यांत्रिकी की पहचान बन चुकी है: दो \omterm{def:b3:harmonic-oscillator:ladder}{सोपान संकारक} स्तरों की एक पूरी तरह
समान सीढ़ी $\big(n +
\tfrac12\big)\hbar\omega$ पर चढ़ते और उतरते हैं, और तल पर का वह आधा पग
--- \omterm{thm:b3:harmonic-oscillator:spectrum}{शून्य-बिंदु ऊर्जा} --- कोई विकल्प नहीं, एक प्रमेय है।
इसके परिणाम इस बात तक पहुँचते हैं कि हीलियम कभी जमता क्यों नहीं और
कोई कार्बन डाइऑक्साइड अणु, अपनी ही $\hbar\omega$ पर बजता हुआ, पृथ्वी की
बाहर जाती ऊष्मा को कैसे रोक लेता है।

\section{सीढ़ी}

\begin{definition}[सोपान संकारक]\label{def:b3:harmonic-oscillator:ladder}
\index{सोपान संकारक}\index{विनाशन संकारक}\index{सृजन संकारक}
$\hat H = \hat p^2/2m + \tfrac12 m\omega^2\hat x^2$ के लिए यह
विमाहीन, अ-हर्मिटीय युग्म लीजिए
\[
\hat a = \sqrt{\frac{m\omega}{2\hbar}}\Big(\hat x +
\frac{\iu\hat p}{m\omega}\Big) , \qquad
\hat a^\dagger = \sqrt{\frac{m\omega}{2\hbar}}\Big(\hat x -
\frac{\iu\hat p}{m\omega}\Big) .
\]
$[\hat x, \hat p] = \iu\hbar$ से:
\[
[\hat a, \hat a^\dagger] = 1 , \qquad
\hat H = \hbar\omega\Big(\hat N + \tfrac12\Big) , \quad
\hat N = \hat a^\dagger\hat a .
\]
$\hat N$ हर्मिटीय है; उसके \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} \emph{क्वांटा} गिनेंगे, और
$\hat a$, $\hat a^\dagger$ --- अर्थात् \emph{विनाशन} और
\emph{सृजन} संकारक --- एक-एक क्वांटम हटाएँगे और जोड़ेंगे।
\end{definition}

\begin{theorem}[वर्णक्रम, केवल बीजगणित से]\label{thm:b3:harmonic-oscillator:spectrum}
\index{शून्य-बिंदु ऊर्जा}
$\hat H$ के \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} ठीक ये हैं
\[
E_n = \Big(n + \tfrac12\Big)\hbar\omega , \qquad n = 0, 1, 2, \dots
\]
--- अर्थात् बराबर पगों $\hbar\omega$ की एक अनंत सीढ़ी, जो ऐसी मूल
अवस्था पर खड़ी है जो शून्य \emph{नहीं} है: यही \emph{\omterm{thm:b3:harmonic-oscillator:spectrum}{शून्य-बिंदु ऊर्जा}}
$\tfrac12\hbar\omega$ है। प्रसामान्यित अभिलक्षणिक अवस्थाएँ $\ket n$
इनसे जुड़ी हैं
\[
\hat a\ket n = \sqrt n\,\ket{n - 1} , \qquad
\hat a^\dagger\ket n = \sqrt{n + 1}\,\ket{n + 1} ,
\qquad
\ket n = \frac{(\hat a^\dagger)^n}{\sqrt{n!}}\,\ket0 .
\]
\end{theorem}

\begin{proof}
\omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} से $\hat N\hat a = \hat a(\hat N - 1)$: अतः यदि
$\ket\nu$ का $\hat N$-अभिलक्षणिक मान $\nu$ हो, तो $\hat a\ket\nu$
$\nu - 1$ वाला अभिलक्षणिक सदिश है (या शून्य सदिश)। हर उतार
तभी तक अनुमत है जब तक $\nu \ge 0$, क्योंकि $\nu = \braket{\nu}{\hat
N\nu} = \|\hat a\ket\nu\|^2 \ge 0$; अतः उतार को कभी न कभी
रुकना ही पड़ेगा, और वह केवल ऐसी अवस्था पर रुकता है जिसके लिए $\hat a\ket{\nu_0}
= 0$, जिससे $\nu_0 = 0$। अतः $\nu$ अ-ऋणात्मक
पूर्णांकों पर चलता है। प्रसामान्यन $\|\hat a\ket n\|^2 = n$
और $\|\hat a^\dagger\ket n\|^2 = n + 1$ से निकलते हैं।
\end{proof}

\begin{proposition}[अवस्थाएँ, दिक् में]\label{prop:b3:harmonic-oscillator:wavefunctions}
मूल अवस्था यह गाउसीय है
\[
\varphi_0(x) = \Big(\frac{m\omega}{\pi\hbar}\Big)^{1/4}
\eu^{-m\omega x^2/2\hbar} ,
\]
जो \emph{प्रथम-कोटि} समीकरण $\hat a\varphi_0
= 0$ हल करने से मिलती है; वह हाइज़ेनबर्ग की असमिका को संतृप्त कर देती
है, $\Delta x\,\Delta p =
\hbar/2$, जहाँ $\Delta x = \sqrt{\hbar/2m\omega}$। बार-बार
$\hat a^\dagger$ लगाने से $\varphi_n$ बनते हैं: घात $n$ का कोई बहुपद
($n$ निस्पंदों के साथ) गुणा वही गाउसीय। बड़े $n$ के लिए
$|\varphi_n|^2$ चिरसम्मत ठहराव-काल बंटन के चारों ओर तेज़ी से दोलन करता
है, और पलट-बिंदुओं के पास सबसे बड़ा होता है --- अर्थात् संगतता
सिद्धांत, एक चित्र में।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
$\hat a\varphi_0 = 0$ का अर्थ है $\varphi_0' = -(m\omega/\hbar)x
\varphi_0$: अर्थात् वही गाउसीय, प्रसामान्यित। संतृप्ति: गाउसीय ही
\cref{thm:b3:quantum-formalism:uncertainty} की समता-स्थिति है।
बहुपद वाली संरचना इससे निकलती है कि $\hat a^\dagger$, $x$ और
$\dd/\dd x$ में प्रथम कोटि का है; और बड़े $n$ वाला कथन
\cref{exo:b3:harmonic-oscillator:12} में जाँचा गया है।
\end{proof}

\section{शून्य-बिंदु ऊर्जा वास्तविक है}

\begin{remark}[फ़र्श और नीचा क्यों नहीं हो सकता]\label{rem:b3:harmonic-oscillator:zeropoint}
शून्य ऊर्जा वाली किसी अवस्था के लिए $\langle\hat p^2\rangle =
\langle\hat x^2\rangle = 0$ चाहिए होता: पूरी तरह ठहरी \emph{और} पूरी
तरह केंद्रित, जिसे $\Delta x\,\Delta p \ge \hbar/2$ मना करता है। मूल
अवस्था वही सर्वोत्तम समझौता है जो यह असमिका देती है, और
$\tfrac12\hbar\omega$ उसका किराया है। यह कोई हिसाबी नियतांक नहीं है:
शून्य-बिंदु गति परम शून्य पर भी एक्स-किरण विवर्तन के प्रतिरूप धुँधले कर
देती है; वह हलके समस्थानिकों को दुर्बल प्रभावी बंध देती है (H$_2$ और
D$_2$ एक ही इलेक्ट्रॉनिक कूप से मापने योग्य रूप से भिन्न ऊर्जाओं पर
वियोजित होते हैं); और हीलियम में वह इतनी प्रचंड है --- हलके परमाणु,
दुर्बल आकर्षण --- कि द्रव अपने ही वाष्प-दाब के अंतर्गत \emph{कभी} नहीं
जमता: यही अकेला तत्व है जो परम शून्य पर भी द्रव रहता है, और केवल
25 वायुमंडल की सहायता से जमता है।
\end{remark}

\begin{example}[$\hbar\omega$ के पैमाने]\label{ex:b3:harmonic-oscillator:scales}
कार्बन मोनोऑक्साइड का बंध ($\omega = \qty{4.1e14}{rad/s}$): $\hbar\omega =
\qty{0.27}{eV}$, अर्थात् कमरे के ताप के $k_{\text{B}}T$ से दस गुना ---
रोज़मर्रा की हवा में आणविक कंपन जमे रहते हैं, और इसीलिए द्विपरमाणुक
ऊष्मा धारिताओं ने उन्नीसवीं सदी को उलझाए रखा। कोई लोलक
($\omega = \qty{5}{rad/s}$): $\hbar\omega = \qty{3e-15}{eV}$,
जो विभेदन से निराशाजनक रूप से परे है --- अर्थात् संगतता की सीमा। कोई
\qty{40}{kg} का LIGO दर्पण, जो इस तरह लटकाया गया हो कि
\qty{100}{Hz} पर झूले, दोलक-विधा के रूप में शून्य-बिंदु आयाम $\Delta x =
\sqrt{\hbar/2m\omega} \approx \qty{5e-20}{m}$ रखता है --- और वह
वेधशाला इस क्वांटम फ़र्श \emph{पर} के विस्थापन नियमित रूप से सुलझा लेती
है: चालीस किलोग्राम की किसी वस्तु की शून्य-बिंदु गति अब एक अभियांत्रिक
\omterm{def:b3:lagrangian-mechanics:coordinates}{प्रतिबंध} है (\cref{exo:b3:harmonic-oscillator:3})।
\end{example}

\section{संसक्त अवस्थाएँ: क्वांटम दोलक का चिरसम्मत चेहरा}

\begin{definition}[संसक्त अवस्थाएँ]\label{def:b3:harmonic-oscillator:coherent}
\index{संसक्त अवस्था}
\emph{संसक्त अवस्था} $\ket\alpha$ \omterm{def:b3:harmonic-oscillator:ladder}{विनाशन संकारक} का अभिलक्षणिक सदिश है,
$\hat a\ket\alpha = \alpha\ket\alpha$, जहाँ
$\alpha$ कोई भी सम्मिश्र संख्या है। सीढ़ी पर प्रसार करने पर,
\[
\ket\alpha = \eu^{-|\alpha|^2/2}\sum_{n=0}^\infty
\frac{\alpha^n}{\sqrt{n!}}\,\ket n :
\]
क्वांटम गिनती प्वासों-बंटित है, जिसका माध्य $\langle\hat
N\rangle = |\alpha|^2$ और फैलाव $\Delta N = |\alpha|$ है।
\end{definition}

\begin{proposition}[लेज़र चिरसम्मत क्यों हैं]\label{prop:b3:harmonic-oscillator:coherent-motion}
कोई \omterm{def:b3:harmonic-oscillator:coherent}{संसक्त अवस्था} \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{कला समष्टि} में खिसकाई गई मूल-अवस्था वाली गाउसीय ही
है; दोलक के विकास के अंतर्गत वह संसक्त \emph{बनी रहती} है,
$\alpha(t) = \alpha\,\eu^{-\iu\omega t}$: उसका केंद्र ठीक
चिरसम्मत गति करता है, $\langle\hat x\rangle(t) =
x_0\cos\omega t + \cdots$, जबकि उसकी चौड़ाइयाँ सदा न्यूनतम
$\Delta x\,\Delta p = \hbar/2$ बनी रहती हैं --- ऐसा तरंग-पुंज जो कभी
फैलता ही नहीं। संसक्त अवस्थाएँ वही तरीका हैं जिससे कोई क्वांटम दोलक
किसी चिरसम्मत दोलक का स्वाँग भरता है; किसी लेज़र का प्रकाश किसी
क्षेत्र-विधा की \omterm{def:b3:harmonic-oscillator:coherent}{संसक्त अवस्था} है, जिसकी फ़ोटॉन-संख्या प्वासोंी है
(हर प्रकाश-संसूचक का \emph{छर्रा कोलाहल}), और जिसका क्षेत्र वैसे ही
दोलन करता है जैसा मैक्सवेल ने कहा था।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
प्रसार इस तरह निकलता है: $\ket\alpha = \sum c_n\ket n$ को
$\hat a\ket\alpha = \alpha\ket\alpha$ में रखिए: $c_n = \alpha c_{n-1}/\sqrt
n$। विकास: हर $\ket n$ को $\eu^{-\iu(n + 1/2)\omega t}$ मिलता है,
जो फिर से $\alpha\eu^{-\iu\omega t}$ की \omterm{def:b3:harmonic-oscillator:coherent}{संसक्त अवस्था} में जुड़ जाता है
(वैश्विक कला को छोड़कर)। $\langle\hat x\rangle \propto
\operatorname{Re}\alpha(t)$, क्योंकि $\hat x \propto \hat a +
\hat a^\dagger$। यह कि वह अवस्था खिसकाई हुई गाउसीय है,
यहाँ स्वीकृत है।
\end{proof}

\begin{method}[दोलक-बीजगणित]\label{met:b3:harmonic-oscillator:method}
(1) जो भी पूछा जाए उसे $\hat a$, $\hat a^\dagger$ में व्यक्त कीजिए: $\hat x
= \sqrt{\hbar/2m\omega}\,(\hat a + \hat a^\dagger)$, $\hat p =
\iu\sqrt{m\hbar\omega/2}\,(\hat a^\dagger - \hat a)$। (2)
$\hat a$ को दाईं ओर सरकाइए, $[\hat a, \hat a^\dagger] = 1$ से; और
$\hat a\ket0 = 0$ काम में लाइए। (3) आव्यूह-अवयव: $\hat x$ केवल
पड़ोसी पायदान जोड़ता है --- अर्थात् कंपन-वर्णक्रममापी का चयन-नियम
$\Delta n = \pm1$। (4) कोई भी द्विघात \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} (युग्मित
दोलक, परिपथ, क्षेत्र-विधाएँ) स्वतंत्र सीढ़ियों में विकर्णित हो जाता है:
पहले प्रसामान्य विधाएँ ज्ञात कीजिए, फिर हर एक का क्वांटन कीजिए। (5) पहले
संख्याएँ: $\hbar\omega$, $k_{\text{B}}T$ के सामने, ही किसी बीजगणित से
पहले तय कर देता है कि तंत्र क्वांटम है या चिरसम्मत।
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:1}
(क) $[\hat a, \hat a^\dagger] = 1$ जाँचिए, $[\hat x, \hat p] =
\iu\hbar$ से। (ख) सीधे प्रसार से $\hat H = \hbar\omega(\hat a^\dagger\hat a +
\tfrac12)$ जाँचिए। (ग) उलटकर $\hat x$ और
$\hat p$ व्यक्त कीजिए। (घ) दिखाइए कि $\hat N$ हर्मिटीय है और समझाइए
कि अकेला $\hat a$ कोई प्रेक्ष्य क्यों नहीं हो सकता।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:2}
अभिलक्षणिक अवस्था $\ket n$ पर: (क) दिखाइए कि $\langle\hat x\rangle =
\langle\hat p\rangle = 0$; (ख) $\langle\hat x^2\rangle$ और
$\langle\hat p^2\rangle$ निकालिए; (ग) $\Delta x\,\Delta p = (n +
\tfrac12)\hbar$ निकालिए; (घ) विरियल अनुपात $\langle E_k\rangle =
\langle E_p\rangle$ जाँचिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:3}
शून्य-बिंदु आयाम $\Delta x = \sqrt{\hbar/2m\omega}$ निकालिए:
(क) CO अणु के लिए ($\mu = \qty{1.14e-26}{kg}$, $\omega =
\qty{4.1e14}{rad/s}$), और उसकी तुलना बंध-लंबाई
\qty{0.11}{nm} से कीजिए; (ख) किसी ठोस में हाइड्रोजन परमाणु के लिए ($m =
\qty{1.7e-27}{kg}$, $\hbar\omega = \qty{0.1}{eV}$); (ग) किसी LIGO
दर्पण के लिए ($m = \qty{40}{kg}$, $f = \qty{100}{Hz}$); (घ) तीनों को
अपने-अपने तंत्र के आकार के अंशों के रूप में क्रम दीजिए और टिप्पणी
कीजिए कि ``क्वांटम'' कौन है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:4}
(क) सोपान-संबंधों से $\bra m\hat x\ket n$ निकालिए और
दिखाइए कि वह $m = n \pm 1$ को छोड़कर लुप्त हो जाता है। (ख) प्रकाश के
अवशोषण के लिए कंपन-चयन नियम $\Delta n = \pm1$ निकालिए (युग्मन
$\propto\hat x$ है)। (ग) कोई विषम-नाभिकीय अणु (CO)
अवरक्त प्रकाश क्यों सोखता है, जबकि N$_2$ नहीं? (घ) वास्तविक अणु
$\approx 2\hbar\omega$ पर दुर्बल ``अधिस्वरक'' रेखाएँ दिखाते हैं: इससे
विभव के विषय में क्या पता चलता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:5}
(क) $\hat a\varphi_0 = 0$ को अवकल समीकरण की तरह हल कीजिए और
प्रसामान्यित कीजिए। (ख) $\varphi_1$ और $\varphi_2$ बनाइए,
$\hat a^\dagger$ लगाकर। (ग) केवल सम-विषमता से जाँचिए कि $\varphi_1 \perp \varphi_0$।
(घ) तीनों घनत्वों का रेखाचित्र बनाइए और निस्पंदों की गिनती जाँचिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:6}
बोल्ट्समान की झलक। ताप $T$ पर सर्वसम दोलकों का कोई संग्रह स्तर
$n$ पर प्रायिकता $\propto
\eu^{-E_n/k_{\text{B}}T}$ से रहता है (वर्ष 2 का खंड, बोल्ट्समान गुणक)। (क)
दिखाइए कि माध्य क्वांटम संख्या $\langle n\rangle =
1/(\eu^{\hbar\omega/k_{\text{B}}T} - 1)$ है। (ख) CO के
कंपन के लिए \qty{300}{K} और \qty{2000}{K} पर मान निकालिए। (ग) दिखाइए कि माध्य
ऊर्जा ऊँचे ताप पर $k_{\text{B}}T$ की ओर जाती है (ऊर्जा-समविभाजन
पुनः प्राप्त) और नीचे ताप पर $\tfrac12\hbar\omega$ की ओर। (घ) किस
ताप पर \qty{15}{\micro m} का कोई कंपन (कार्बन डाइऑक्साइड का
नमन) $\langle n\rangle = 0.1$ रखता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:7}
संसक्त-अवस्था की सांख्यिकी। (क) $\ket\alpha$ के प्रसार से
दिखाइए कि $\mathcal P(n)$ प्वासों है, जिसका माध्य $|\alpha|^2$ है। (ख) दिखाइए कि
$\langle\hat N\rangle = |\alpha|^2$ और $\Delta N = |\alpha|$। (ग)
\qty{633}{nm} पर \qty{1}{mW} का कोई लेज़र: प्रति सेकंड फ़ोटॉन, और एक
सेकंड में आपेक्षिक उतार-चढ़ाव $\Delta N/\langle N\rangle$। (घ)
छर्रा कोलाहल: दिखाइए कि प्रकाश-धारा का कोलाहल-से-संकेत अनुपात
$1/\sqrt{\langle N\rangle}$ की तरह घटता है --- अतः चमकीले पुंज चिकने क्यों दिखते हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:8}
किसी \omterm{def:b3:harmonic-oscillator:coherent}{संसक्त अवस्था} का विकास। (क) प्रसार पर विकास-कलाएँ लगाइए और
$\ket{\alpha(t)}$ दिखाइए, जहाँ $\alpha(t) =
\alpha\eu^{-\iu\omega t}$ (किसी वैश्विक कला को छोड़कर)। (ख)
$\langle\hat x\rangle(t)$ और $\langle\hat p\rangle(t)$ निकालिए और
चिरसम्मत हल से तुलना कीजिए। (ग) कोई ऊर्जा-अभिलक्षणिक अवस्था स्थायी
होते हुए भी किसी झूलते लोलक का वर्णन \emph{क्यों नहीं} करती ---
और \omterm{def:b3:harmonic-oscillator:coherent}{संसक्त अवस्था} की कौन-सी विशेषता करती है? (घ) किसी वास्तविक लोलक
(\qty{10}{g}, \qty{10}{cm} आयाम, \qty{1}{Hz}) के लिए $|\alpha|$ आँकिए
और टिप्पणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:9}
क्वांटम LC परिपथ। कोई अतिचालक पाश, जिसमें $L =
\qty{10}{nH}$ और $C = \qty{0.4}{pF}$ है, आवेश और अभिवाह को
$x$ और $p$ की तरह दोलित करता है। (क) उसकी अनुनाद-आवृत्ति (वर्ष 1 का
खंड) और क्वांटम $\hbar\omega$, $\unit{\micro eV}$ में। (ख) किस
ताप से नीचे $k_{\text{B}}T \ll \hbar\omega$ होता है, ताकि परिपथ
अपनी मूल अवस्था में बैठा रहे? (ग) अतिचालक क्यूबिट \qty{20}{mK} पर
तनुकरण-प्रशीतकों में क्यों चलाए जाते हैं? (घ) शून्य-बिंदु वोल्टता का
उतार-चढ़ाव $\Delta V = \Delta q/C$, जहाँ $\Delta q =
\sqrt{\hbar\omega C/2}$: उसका मान निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:10}
शून्य-बिंदु गति से वान डर वाल्स। दूरी $R$ पर दो उदासीन परमाणुओं का
निदर्शन दो सर्वसम द्विध्रुव-दोलकों (आवेश $e$, द्रव्यमान
$m$, आवृत्ति $\omega_0$) से कीजिए, जिनके विस्थापन द्विध्रुव-ऊर्जा
$\lambda\,x_1x_2$ से युग्मित हों, जहाँ $\lambda = e^2/2\pi\varepsilon_0R^3$।
(क) दिखाइए कि प्रसामान्य निर्देशांक $(x_1 \pm x_2)/\sqrt2$
$\omega_\pm = \omega_0\sqrt{1 \pm \lambda/m\omega_0^2}$ पर दोलन करते हैं। (ख) मूल
ऊर्जा $\tfrac\hbar2(\omega_+ + \omega_-)$ है: उसे $\lambda$ में
द्वितीय कोटि तक प्रसारित कीजिए और दिखाइए कि अन्योन्यक्रिया-ऊर्जा यह है
\[
\Delta E = -\frac{\hbar\lambda^2}{8m^2\omega_0^3}
\ \propto\ -\frac{1}{R^6} .
\]
(ग) यह आकर्षण सार्वत्रिक क्यों है --- उन परमाणुओं के बीच भी उपस्थित,
जिनके कोई स्थायी द्विध्रुव हैं ही नहीं? (घ) $1/R^6$ नियम वर्ष 1 के
खंड के वास्तविक-गैस संशोधनों का वान डर
वाल्स बल है: सूक्ष्मदर्शीय दृष्टि से ``उतार-चढ़ाव'' किसमें हो रहा है ---
और $\hbar = 0$ वाले संसार में इस बल का क्या होता?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:11}
किसी बद्ध आयन के पार्श्वबैंड। किसी हरात्मक पाश में कोई आयन ($f =
\qty{1}{MHz}$) लेज़र-प्रकाश सोखता है। चूँकि आयन चलता है, उसके
अवशोषण-वर्णक्रम में $\nu_0$ पर की इलेक्ट्रॉनिक रेखा के दोनों ओर
$\nu_0 \pm f$ पर रेखाएँ दिखती हैं: वे संक्रमण, जो इलेक्ट्रॉनिक
क्वांटम संख्या के साथ-साथ \emph{गति की} क्वांटम संख्या भी $\mp1$ जितनी
बदल देते हैं। (क) $\hbar\omega_{\text{पाश}}$, $\unit{neV}$ में क्या है,
और पार्श्वबैंड सुलझाने के लिए बहुत सँकरी रेखा क्यों चाहिए? (ख)
$\nu_0 - f$ रेखा चलाने से प्रति चक्र गति का एक क्वांटम हट जाता है:
मूल अवस्था तक \emph{पार्श्वबैंड शीतलन} समझाइए। (ग) जब एक बार
$\langle n\rangle \approx
0$ हो जाए, तब नीचे वाला पार्श्वबैंड पूरी तरह
लुप्त हो जाता है --- यह असममिति क्वांटम मूल अवस्था तक पहुँचने का प्रमाण
क्यों है? (घ) इसी तरह किसी अकेले परमाणु की --- और दूसरे उपायों से,
किलोग्राम-पैमाने के LIGO दर्पणों की --- गति की मूल अवस्था प्रमाणित की
जाती है: बताइए कि $f = \qty{1}{MHz}$ और
$\langle n\rangle = 0.05$ के लिए आयन किस ``ताप'' तक पहुँचा है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:12}
चिरसम्मत सीमा, संख्याओं में। आयाम $A$ का कोई चिरसम्मत दोलक
$[x, x + \dd x]$ में अंश $\dd
t/T = \dd x/\pi\sqrt{A^2 - x^2}$ जितना समय बिताता है। (क) यह
ठहराव-काल बंटन निकालिए। (ख) क्वांटम अवस्था $n$ के लिए $A_n$ लीजिए, $E_n =
\tfrac12 m\omega^2A_n^2$ से, और चिरसम्मत बंटन की तुलना
(दिए हुए) स्थानिक रूप से औसत किए गए $|\varphi_n|^2$ से कीजिए: वे कहाँ
सहमत होते हैं और कहाँ उन्हें भिन्न होना ही पड़ेगा? (ग) दिखाइए कि
पड़ोसी स्तरों के बीच का आंशिक अंतराल $\Delta E/E$, $1/n$ की तरह लुप्त
हो जाता है: अतः ऊर्जा प्रभावी रूप से संतत हो जाती है। (घ)
\cref{exo:b3:harmonic-oscillator:8}(घ) के लोलक के लिए $n$ और $\Delta
E/E$ आँकिए, और संगतता के तर्क का निष्कर्ष एक वाक्य में दीजिए।
\end{exercise}

\section{समस्या: वह अणु जो पृथ्वी को गरम रखता है}

\begin{problem}[{सप्ताहांत समस्या --- कार्बन डाइऑक्साइड की क्वांटम
सीढ़ी और हरितगृह प्रभाव}]\label{pb:b3:harmonic-oscillator:1}
कोई कार्बन डाइऑक्साइड अणु एक रैखिक O=C=O शृंखला है: अर्थात् कुछ
क्वांटीकृत दोलक। उसकी नमन-विधा का $\hbar\omega$ संयोग से ठीक पृथ्वी की
बाहर जाती तापीय चमक के बीच में पड़ता है, और इसीलिए यह अल्प गैस ---
दस हज़ार में चार अणु --- ग्रह की जलवायु चलाती है। आँकड़े: नमन का
तरंग-संख्यांक $\tilde\nu_2 = \qty{667}{cm^{-1}}$
(वर्णक्रममापी का मात्रक: $E = hc\tilde\nu$, $\qty{1}{cm^{-1}} =
\qty{1.24e-4}{eV}$); असममित खिंचाव $\tilde\nu_3 =
\qty{2349}{cm^{-1}}$; सममित खिंचाव $\tilde\nu_1 =
\qty{1388}{cm^{-1}}$; $k_{\text{B}}T$, $\qty{288}{K}$ पर,
\qty{24.8}{meV} है; वीन का नियम (वर्ष 2 का खंड): $\lambda_{\max}T =
\qty{2898}{\micro m.K}$।

\medskip\noindent\textbf{भाग I --- किसी रैखिक अणु की विधाएँ।}
\begin{enumerate}
  \item तीन परमाणुओं की नौ \omterm{def:b3:lagrangian-mechanics:coordinates}{स्वातंत्र्य कोटि} होती हैं: उनमें कितनी पूरे
        अणु के स्थानांतरण हैं, कितनी घूर्णन (अणु \emph{रैखिक} है), और
        कितने कंपन बचते हैं?
  \item चारों कंपनों का वर्णन कीजिए: सममित खिंचाव, असममित
        खिंचाव, और एक द्विगुण अपह्रासी नमन --- नमन दो बार क्यों आता है?
  \item प्रकाश किसी कंपमान \emph{विद्युत् द्विध्रुव} से युग्मित होता
        है (\cref{exo:b3:harmonic-oscillator:4})। O=C=O की कौन-सी
        विधाएँ द्विध्रुव-आघूर्ण को माडुलित करती हैं, और कौन-सी
        अवरक्त-मौन है?
  \item तीनों तरंग-संख्यांकों को फ़ोटॉन-तरंगदैर्घ्यों में बदलिए, और
        उन्हें रखिए: इनमें से कौन-सी तापीय अवरक्त में हैं?
  \item हवा की दोनों मुख्य गैसें, N$_2$ और O$_2$, लगभग कोई अवरक्त
        क्यों नहीं सोखतीं --- और वायुमंडल की पारदर्शिता पर उसका क्या
        परिणाम होता है?
  \item जल-वाष्प, जो मुड़ा हुआ और द्विध्रुवी है, अवरक्त के बड़े भाग
        पर सोखता है: कार्बन डाइऑक्साइड का नमन किस वर्णक्रमीय
        ``खिड़की'' में लगभग अकेला काम करता है (\qty{15}{\micro m} की
        तुलना \qty{8}{\micro m} से नीचे और \qty{20}{\micro m} से ऊपर
        के जल के प्रबल बैंडों से कीजिए)?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- नमन की क्वांटम सीढ़ी।}
\begin{enumerate}[resume]
  \item $\hbar\omega_2$ को \unit{meV} में और सीढ़ी $E_n$ निकालिए।
  \item \qty{288}{K} पर अणुओं का कितना अंश $n = 1$ पर रहता है
        ($n = 0$ के सापेक्ष, बोल्ट्समान गुणक)? और
        $n = 2$ पर?
  \item कौन-सी फ़ोटॉन-तरंगदैर्घ्य $n = 0 \to 1$ चलाती है? \emph{वही}
        तरंगदैर्घ्य उत्सर्जन में भी प्रधान क्यों है?
  \item हरात्मक सीढ़ी से औचित्य दीजिए कि एक ही तरंगदैर्घ्य पूरी सीढ़ी
        के काम आती है ($n \to n + 1$, हर $n$ के लिए): इसमें
        स्तर-अंतराल का कौन-सा गुण काम कर रहा है?
  \item अणु घूमता भी है, जिससे सूक्ष्म संरचना जुड़ जाती है:
        \qty{15}{\micro m} वाली विशेषता सचमुच लगभग
        \qty{1}{\micro m} चौड़ा एक \emph{बैंड} है। किसी हरितगृह गैस
        के लिए बैंड की \emph{चौड़ाई} क्यों मायने रखती है (सोचिए कि
        बैंड का केंद्र अपारदर्शी हो जाने पर क्या होता है)?
  \item हवा में कंपन से उत्तेजित कोई CO$_2$ अपना क्वांटम विकिरित
        करने की तुलना में संघट्ट से खोने की कहीं अधिक संभावना रखता है
        (विकिरणी जीवनकाल $\sim\qty{1}{s}$, संघट्ट-काल $\sim\qty{e-9}{s}$):
        सोखी हुई विकिरण-ऊर्जा वास्तव में कहाँ जाती है?
  \item और उलटे, \qty{288}{K} की हवा $n = 1$ में एक तापीय जनसंख्या
        बनाए रखती है (प्रश्न 8): वह जनसंख्या ऐसा क्या करती है जो
        ऊर्जा-बजट के लिए मायने रखता है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- ग्रह का विकिरण-बहीखाता।}
\begin{enumerate}[resume]
  \item सूर्य \qty{5800}{K} के पिंड की तरह विकिरित करता है: उसका
        वीन शिखर निकालिए। क्या CO$_2$ का नमन बहुत सूर्य-प्रकाश रोकता है?
  \item भूमि \qty{288}{K} के पिंड की तरह विकिरित करती है: उसका
        वीन शिखर निकालिए, और उस तापीय वक्र पर \qty{15}{\micro m}
        का स्थान बताइए।
  \item हरितगृह क्रियाविधि चार वाक्यों में समझाइए:
        सूर्य-प्रकाश भीतर, तापीय अवरक्त बाहर, \qty{15}{\micro m} पर
        रोक, और ऊपर \emph{तथा} नीचे, दोनों ओर पुनः उत्सर्जन।
  \item जो पुनः उत्सर्जन अंतरिक्ष तक बच निकलता है, वह ऊँची और ठंडी
        परतों ($\sim\qty{220}{K}$) से आता है: किसी ठंडी परत से
        उत्सर्जन उन तरंगदैर्घ्यों पर ग्रह की बाहर जाती शक्ति क्यों
        घटा देता है (याद रखिए कि तापीय उत्सर्जन $T$ के साथ बढ़ता है)?
  \item अधिक CO$_2$ उत्सर्जक परत को और ऊपर तथा और ठंडा कर देती है:
        एक वाक्य में बताइए कि तब बहीखाता फिर से संतुलित करने के लिए
        पृष्ठ को गरम क्यों होना पड़ता है।
  \item बैंड का केंद्र पहले से ही अपारदर्शी है; जुड़ी हुई CO$_2$ का
        प्रभाव बैंड के \emph{पंखों} में काम करता है, जिससे प्रणोदन
        सांद्रता के साथ लघुगणकीय रूप से बढ़ता है: इसे प्रश्न 11 से
        जोड़िए।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- समस्थानिक: क्वांटम
छाप।}
\begin{enumerate}[resume]
  \item किसी दोलक की आवृत्ति $\sqrt{k/\mu}$ की तरह मापित होती है:
        असममित खिंचाव के लिए $^{12}$C के स्थान पर $^{13}$C रखने से
        प्रभावी द्रव्यमान बदल जाता है; और प्रेक्षित रेखा
        \qty{2349}{cm^{-1}} से खिसककर लगभग \qty{2283}{cm^{-1}} पर आ
        जाती है। द्रव्यमानों से इस $\approx 3\%$ खिसकाव की
        परिमाण-कोटि जाँचिए।
  \item इन दोनों रेखाओं पर सधे लेज़र किसी गैस-नमूने में
        $^{13}$CO$_2$ और $^{12}$CO$_2$ अलग-अलग गिन लेते हैं: समझाइए
        कि क्वांटीकृत सीढ़ी ऐसा समस्थानिक-विभेदी संसूचन संभव ही कैसे
        बनाती है।
  \item पौधे हलके समस्थानिक को पसंद करते हैं, अतः जीवाश्म कार्बन में
        $^{13}$C कम होता है: वायुमंडलीय CO$_2$ की सांद्रता बढ़ने के
        साथ उसके मापे गए समस्थानिक अनुपात ने क्या किया --- और उससे
        जुड़ी हुई गैस के \emph{स्रोत} के विषय में क्या सिद्ध होता है?
  \item वही \qty{15}{\micro m} वाली भौतिकी शुक्र पर भी काम करती है
        ($96\%$ CO$_2$, \qty{90}{bar}): उसका
        \qty{737}{K} पृष्ठ क्या दिखाता है?
  \item मंगल भी लगभग शुद्ध CO$_2$ में साँस लेता है, पर \qty{6}{mbar}
        पर, और वह ठिठुरता रहता है: शुक्र--पृथ्वी--मंगल की यह तिकड़ी
        इस विषय में क्या दिखाती है कि प्रभाव की प्रबलता कौन-सा चर तय
        करता है?
  \item नामित परिणाम का सार लिखिए: \qty{83}{meV} का एक क्वांटम ---
        किसी नमन करते त्रिपरमाणुक का $\hbar\omega$ --- जो
        \qty{288}{K} के किसी ग्रह के तापीय वर्णक्रम पर
        \qty{15}{\micro m} पर खड़ा है, सोखा जाता है, नैनोसेकंडों में
        तापीकृत होता है और ठंडी ऊँचाइयों से फिर उत्सर्जित होता है ---
        यही वह क्रियाविधि है जिससे हवा का $0.04\%$ पृथ्वी का ताप तय
        कर देता है।
\end{enumerate}
\end{problem}
