\chapter{प्रकीर्णन सिद्धांत}\label{ch:b3:scattering-theory}

सूक्ष्म के बारे में हम जो कुछ जानते हैं, वह लगभग सारा उस पर चीज़ें
फेंककर और यह देखकर सीखा गया है कि क्या टकराकर लौटता है। रदरफ़ोर्ड ने
जस्ता-सल्फ़ाइड के परदे पर अल्फ़ा-कणों की चमकें गिनकर नाभिक खोजा; पचास
वर्ष बाद वही प्रयोग, ऊँची ऊर्जा पर इलेक्ट्रॉनों से चलाकर, प्रोटॉन के
भीतर क्वार्क पा गया; और बीच में तथा उसके बाद से नाभिकीय, कण, परमाणविक
और संघनित-पदार्थ भौतिकी का हर \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} इसी अध्याय के सिद्धांत से
पढ़ा गया है। यहाँ की भाषा है \emph{\omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट}} --- अर्थात् कोई
प्रभावी लक्ष्य-क्षेत्रफल; और क्वांटम वस्तु है \emph{\omterm{thm:b3:scattering-theory:amplitude}{प्रकीर्णन आयाम}},
जिसका \omterm{thm:b3:scattering-theory:born}{बॉर्न सन्निकटन} एक अविस्मरणीय बात कहता है: किसी फैले हुए लक्ष्य
की जाँच फ़ूर्ये समष्टि में होती है, कोण-दर-कोण। कम ऊर्जा पर \omterm{thm:b3:scattering-theory:partial}{आंशिक
तरंगें} कमान सँभाल लेती हैं और हर उलझे हुए विभव को एक ही संख्या में
समेट देती हैं --- \omterm{thm:b3:scattering-theory:partial}{प्रकीर्णन लंबाई} --- जो आज अतिशीत क्वांटम गैसें
सुरीली करती है; और जहाँ कोई प्रक्षेप्य किसी अर्ध-बद्ध अवस्था में ठहर
सकता है, वहाँ \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} अनुनादों में उछल पड़ता है --- अन्यथा
अदृश्य की वर्णक्रममिति।

\section{अनुप्रस्थ काट}

\begin{definition}[अनुप्रस्थ काट]\label{def:b3:scattering-theory:cross-section}
\index{अनुप्रस्थ काट}\index{अवकल अनुप्रस्थ काट}
एक लक्ष्य कण पर अभिवाह $\Phi$ (प्रति एकांक क्षेत्रफल प्रति सेकंड कण)
का एकसमान पुंज भेजिए। घन कोण $\dd\Omega$ में, दिशा $(\theta,
\varphi)$ के चारों ओर, प्रकीर्ण होने वाले प्रक्षेप्यों की दर
\emph{अवकल अनुप्रस्थ काट} को परिभाषित करती है,
\[
\dd\dot N = \Phi\,\frac{\dd\sigma}{\dd\Omega}\,\dd\Omega ,
\]
और उसका समाकल $\sigma = \int(\dd\sigma/\dd\Omega)\,\dd\Omega$
\emph{कुल अनुप्रस्थ काट} है: वह क्षेत्रफल जो लक्ष्य प्रभावी रूप से
प्रस्तुत करता है। प्रति एकांक आयतन $n$ लक्ष्यों वाली किसी पतली पट्टी
में कोई पुंज $\eu^{-n\sigma x}$ की तरह क्षीण होता है: अनुप्रस्थ काट
जो निकलता है उसे गिनकर नापे जाते हैं। इकाइयाँ: नाभिकीय भौतिकी
\emph{बार्न} काम में लेती है,
$\qty{1}{b} = \qty{e-28}{m^2}$ --- लगभग किसी यूरेनियम नाभिक का
ज्यामितीय आकार, जो न्यूट्रॉन के लिए ``खलिहान जितना बड़ा'' है।
\end{definition}

\section{आयाम और बॉर्न सन्निकटन}

\begin{theorem}[प्रकीर्णन अवस्थाएँ और आयाम]\label{thm:b3:scattering-theory:amplitude}
\index{प्रकीर्णन आयाम}
ऊर्जा $E = \hbar^2k^2/2m$ वाले किसी कण के लिए, जो किसी स्थानिक
विभव $V(\vect r)$ से मिलता है, अचर प्रकीर्णन अवस्था बड़ी दूरी पर
इस तरह बरतती है
\[
\psi(\vect r) \;\longrightarrow\;
\eu^{\iu kz} + f(\theta)\,\frac{\eu^{\iu kr}}{r} ,
\]
अर्थात् एक आपाती समतल तरंग और उस पर \emph{\omterm{thm:b3:scattering-theory:amplitude}{प्रकीर्णन आयाम}}
$f(\theta)$ (एक लंबाई) से मॉडुलित एक निर्गत गोलीय तरंग, तथा
\[
\frac{\dd\sigma}{\dd\Omega} = |f(\theta)|^2 .
\]
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
$1/r$ का ह्रास किसी भी बड़े गोले से गुज़रते निर्गत अभिवाह को परिमित
बना देता है; दोनों पदों की प्रायिकता धाराएँ
(\cref{prop:b3:schrodinger-three-dimensions:current}) निकालने पर
प्रति घन कोण त्रिज्य निर्गत धारा
$(\hbar k/m)|f|^2$ है, जबकि आपाती अभिवाह $\hbar k/m$: यही अनुपात
$\dd\sigma/\dd\Omega$ की परिभाषा है। इस अनन्तस्पर्शी रूप के हल
अस्तित्व में हैं, यह मान लिया गया है।
\end{proof}

\begin{theorem}[बॉर्न सन्निकटन]\label{thm:b3:scattering-theory:born}
\index{बॉर्न सन्निकटन}\index{संवेग अंतरण}
किसी दुर्बल विभव (या तेज़ प्रक्षेप्य) के लिए $V$ में प्रथम-कोटि
\omterm{thm:b3:perturbation-theory:corrections}{विक्षोभ सिद्धांत} देता है
\[
f(\theta) = -\frac{m}{2\pi\hbar^2}
\int V(\vect r)\,\eu^{\iu\vect q\cdot\vect r}\,\dd^3r ,
\qquad
\vect q = \vect k - \vect k' , \quad
q = 2k\sin\frac\theta2 :
\]
आयाम, नियतांकों को छोड़कर, \emph{विभव का फ़ूर्ये रूपांतर} है, जिसका
मूल्यांकन \omterm{thm:b3:scattering-theory:born}{संवेग अंतरण} $\hbar\vect q$ पर होता है।
प्रकीर्णन प्रयोग पदार्थ के फ़ूर्ये विश्लेषक हैं: छोटे कोण लंबी
तरंगदैर्घ्य (मोटी संरचना) पढ़ते हैं, और बड़े कोण महीन ब्योरा --- यही
सबसे गहरा कारण है कि ``ऊँची ऊर्जा'' का अर्थ ``छोटी दूरियाँ'' होता है।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
यह \omterm{thm:b3:perturbation-theory:golden}{फ़र्मी का स्वर्ण नियम} है
(\cref{thm:b3:perturbation-theory:golden}), जो समतल तरंगों
$\ket{\vect k} \to \ket{\vect k'}$ के बीच चलाया गया है: आव्यूह-अवयव
$\int V\eu^{\iu(\vect k - \vect k')\cdot\vect r}\dd^3r$ है, अंतिम
अवस्थाओं का घनत्व गतिक गुणक देता है, और (मान लिया गया) हिसाब उन्हें
$f$ में जोड़ देता है। प्रत्यास्थता देती है
$|\vect k'| = |\vect k|$, जिससे $q = 2k\sin(\theta/2)$ निकलता है ---
$\vect k, \vect k'$ के समद्विबाहु त्रिभुज से।
\end{proof}

\begin{example}[युकावा से रदरफ़ोर्ड तक]\label{ex:b3:scattering-theory:rutherford}
\index{रदरफ़ोर्ड प्रकीर्णन}
परिरक्षित कूलॉम विभव $V = \dfrac{Q_1Q_2}{4\pi
\varepsilon_0 r}\,\eu^{-r/a}$ के लिए बॉर्न समाकल प्रारंभिक है
(\cref{exo:b3:scattering-theory:5}), और जैसे-जैसे परिरक्षण त्रिज्या
$a \to \infty$,
\[
\frac{\dd\sigma}{\dd\Omega} =
\Big(\frac{Q_1Q_2}{16\pi\varepsilon_0 E}\Big)^2
\frac{1}{\sin^4(\theta/2)} :
\]
अर्थात् \emph{रदरफ़ोर्ड \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट}}। $1/r$ के लिए अनन्य एक संयोग
से चिरसम्मत गणना, \omterm{thm:b3:scattering-theory:born}{बॉर्न सन्निकटन} और
यथार्थ क्वांटम उत्तर सब सहमत हैं --- इसीलिए रदरफ़ोर्ड, जिसने 1911 में
चिरसम्मत ढंग से गणना की, सही सूत्र पा गया और, चमक-दर-चमक उसके सत्यापन
से, नाभिक भी
(\cref{pb:b3:scattering-theory:1})।
\end{example}

\section{आंशिक तरंगें और प्रकीर्णन लंबाई}

\begin{theorem}[कला-खिसकाव]\label{thm:b3:scattering-theory:partial}
\index{आंशिक तरंगें}\index{कला-खिसकाव}\index{प्रकीर्णन लंबाई}
किसी केंद्रीय विभव के लिए प्रकीर्णन अवस्था को कोणीय संवेगों पर
विघटित कीजिए: हर आंशिक तरंग $l$ विभव-क्षेत्र से अपने त्रिज्य दोलन के
साथ निकलती है, जो किसी कला $\delta_l(k)$ से \emph{खिसका} हुआ है
--- आकर्षण उसे आगे बढ़ाता है, प्रतिकर्षण पीछे करता है --- और
प्रेक्ष्य आयाम इस तरह फिर से जुड़ता है
\[
f(\theta) = \frac1k\sum_{l=0}^\infty(2l+1)\,\eu^{\iu\delta_l}
\sin\delta_l\,P_l(\cos\theta) , \qquad
\sigma = \frac{4\pi}{k^2}\sum_l(2l+1)\sin^2\delta_l .
\]
कम ऊर्जा पर चिरसम्मत अंतर्ज्ञान ($l \sim kb$, जहाँ संघट्ट
प्राचल $b$) कहता है कि केवल $l = 0$ भीतर घुसता है: प्रकीर्णन
समदैशिक हो जाता है और एक ही संख्या राज करती है,
\[
f \to -a , \qquad \sigma \to 4\pi a^2 ,
\]
जहाँ $a = -\lim_{k\to0}\delta_0/k$ \emph{\omterm{thm:b3:scattering-theory:partial}{प्रकीर्णन
लंबाई}} है। भीतरी जटिलता जो भी हो, कम ऊर्जा पर कोई लक्ष्य
अपनी \omterm{thm:b3:scattering-theory:partial}{प्रकीर्णन लंबाई} \emph{ही} है --- वह महान सरलीकरण जिस पर
शीत-परमाणु भौतिकी चलती है।
\end{theorem}
\admitted

\begin{example}[न्यूट्रॉन प्रोटॉन से मिलता है]\label{ex:b3:scattering-theory:np}
धीमे न्यूट्रॉन प्रोटॉनों से $\sigma \approx
\qty{20.4}{b}$ के साथ प्रकीर्ण होते हैं --- यह क्षेत्रफल ड्यूटरॉन के
ज्यामितीय आकार का चालीस गुना है। व्याख्या: कम ऊर्जा का संघट्ट विशुद्ध
$s$ तरंग है, जिसमें दो चक्रण-वाहिकाएँ हैं, और जिनकी मापी गई प्रकीर्णन
लंबाइयाँ, $a_{\text{t}} = \qty{5.4}{fm}$ (त्रिक) और $a_{\text{s}} =
\qty{-23.7}{fm}$ (एकल), दोनों \qty{2}{fm} की बल-परास से कहीं बड़ी
हैं। बड़ा $|a|$ का अर्थ है शून्य ऊर्जा के \emph{लगभग} पर कोई अवस्था:
त्रिक का मुश्किल से बँधा ड्यूटरॉन
(\cref{ex:b3:schrodinger-three-dimensions:deuteron}), और एकल में एक
``आभासी'' साथी, जो बँधने से ज़रा-सा चूक जाता है --- उसकी चूक
$a_{\text{s}} < 0$ में लिखी है। बीस बार्न की एक पहेली,
जिसे दो लगभग-बन चुकी बद्ध अवस्थाएँ सुलझा देती हैं
(\cref{exo:b3:scattering-theory:7})।
\end{example}

\section{अनुनाद}

\begin{proposition}[ब्राइट--विग्नर अनुनाद]\label{prop:b3:scattering-theory:resonance}
\index{अनुनाद!प्रकीर्णन}\index{ब्राइट--विग्नर सूत्र}
यदि प्रक्षेप्य अस्थायी रूप से ऊर्जा $E_{\text{R}}$ और जीवनकाल
$\tau = \hbar/\Gamma$ वाली किसी अर्ध-बद्ध अवस्था में पकड़ा जा सके,
तो संगत कला-खिसकाव $\pi/2$ से होकर बह जाता है और
आंशिक \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} उछल पड़ता है:
\[
\sigma_l(E) \approx \frac{4\pi}{k^2}\,(2l+1)\,
\frac{\Gamma^2/4}{(E - E_{\text{R}})^2 + \Gamma^2/4} ,
\]
अर्थात् चौड़ाई $\Gamma$ का एक लोरेंत्ज़ीय शिखर, उस छत पर जहाँ तक तरंग
पहुँच सकती है (``एकात्मकता सीमा'' $4\pi(2l+1)/k^2$)। उलटा पढ़िए, तो
किसी \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} में कोई अनुनाद किसी अवस्था की खोज \emph{ही} है:
उसकी स्थिति एक ऊर्जा स्तर है, उसकी चौड़ाई एक जीवनकाल।
\cref{ch:b3:particle-physics} का अधिकांश कण-चिड़ियाघर ठीक ऐसे ही
उभारों के रूप में खोजा गया था।

\begin{method}[किसी प्रकीर्णन समस्या को पढ़ना]\label{met:b3:scattering-theory:method}
(1) पहले गतिकी: $k$, और सुलभ $q$-परास $0 \le q
\le 2k$ --- कौन-से संरचना-पैमाने दिखते हैं? (2) तेज़ और दुर्बल:
बॉर्न --- विभव का फ़ूर्ये रूपांतर लीजिए; लघुता जाँचिए। (3)
धीमा: \omterm{thm:b3:scattering-theory:partial}{आंशिक तरंगें}, प्रायः केवल $s$; \omterm{thm:b3:scattering-theory:partial}{प्रकीर्णन लंबाई} सोचिए,
और $\sigma = 4\pi a^2$ की अपेक्षा कीजिए (बोसॉनी \omterm{def:b3:identical-particles:exchange}{सर्वसम कण}:
$8\pi a^2$)। (4) $\sigma(E)$ में शिखर: ब्राइट--विग्नर अनुरूपित
कीजिए, और एक स्तर तथा एक जीवनकाल बताइए; गर्त: $\pi$ से गुज़रती कोई
कला। (5) विस्तृत लक्ष्य: बिंदु-आयाम को रूप गुणक से गुणा कीजिए ---
अर्थात् घनत्व का फ़ूर्ये रूपांतर
(\cref{exo:b3:scattering-theory:12})।
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:1}
\qty{1}{cm^2} परिच्छेद वाला \qty{1}{\micro A} का प्रोटॉन पुंज
($\num{6.2e12}$ प्रोटॉन/s) \qty{1}{\micro m} की सोने की पन्नी से
गुज़रता है ($n =
\qty{5.9e28}{m^{-3}}$)। (क) क्षेत्रीय लक्ष्य घनत्व। (ख) प्रति नाभिक
$\sigma = \qty{10}{b}$ के साथ पुंज का कितना अंश प्रकीर्ण होता है?
(ग) यदि प्रकीर्णन समदैशिक होता, तो \qty{e-3}{sr} घन कोण वाले किसी
संसूचक में दर क्या होती? (घ) ``\omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट}'' अकेले लक्ष्य का नहीं,
बल्कि \emph{युग्म} (प्रक्षेप्य, लक्ष्य, ऊर्जा) का सुपरिभाषित गुण

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:2}
$R_1, R_2$ त्रिज्याओं वाले चिरसम्मत कठोर गोले: (क) दिखाइए कि $\sigma =
\pi(R_1 + R_2)^2$। (ख) वायु के अणुओं ($R \approx
\qty{0.15}{nm}$, $n = \qty{2.5e25}{m^{-3}}$) के लिए माध्य मुक्त पथ
$1/n\sigma$: मान निकालिए, और वर्ष 1 के खंड के गति-सिद्धांत वाले मान
से तुलिए। (ग) कोई चिरसम्मत कठोर गोला कोई कोणीय संरचना क्यों नहीं
दिखाता ($\dd\sigma/\dd\Omega$ अचर --- मान लीजिए या व्युत्पन्न
कीजिए)? (घ) \emph{क्वांटम} कठोर-गोला प्रकीर्णन
(\cref{exo:b3:scattering-theory:6}) का कौन-सा लक्षण चिरसम्मत
प्रतिरूप नहीं रखता?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:3}
\omterm{thm:b3:scattering-theory:born}{संवेग अंतरण}। (क) $\vect k$, $\vect k'$ का त्रिभुज खींचिए
और $q = 2k\sin(\theta/2)$ व्युत्पन्न कीजिए। (ख) \qty{5.5}{MeV} के
अल्फ़ा कणों ($k = \qty{1.0e15}{m^{-1}}$) के लिए: $q$ का परास $\theta \in
[0, \pi]$ पर, और सबसे छोटी वियोजनीय संरचना $\sim 1/q_{\max}$।
(ग) \qty{500}{MeV} के इलेक्ट्रॉनों ($k \approx E/\hbar c$) के लिए: वही।
(घ) पुंज-ऊर्जा को सूक्ष्मदर्शी की शक्ति से जोड़ने वाली सीख पढ़िए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:4}
\cref{thm:b3:scattering-theory:amplitude} से: (क) जाँचिए कि
$|f|^2$ की विमाएँ प्रति घन कोण क्षेत्रफल की हैं; (ख) दिखाइए कि
किसी गोले से गुज़रता प्रकीर्ण अभिवाह $\Phi\sigma$ है; (ग)
$f$ को सामान्यतः सम्मिश्र क्यों होना चाहिए (उसकी कला किस
संरक्षण-नियम की रखवाली करती है)? (घ) प्रकाशिक प्रमेय $\sigma_{\text{कु}} =
(4\pi/k)\operatorname{Im}f(0)$ और उसका अर्थ बताइए (अग्र तरंग को ठीक
उतना ही क्षीण होना चाहिए जितना प्रकीर्ण होकर चला जाता है)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:5}
युकावा के लिए बॉर्न। $V(r) = \dfrac{g\,\eu^{-\mu r}}{r}$ के साथ: (क)
बॉर्न समाकल कीजिए (पहले कोणीय भाग, फिर $\int_0^
\infty\sin(qr)\eu^{-\mu r}\dd r$) और पाइए $f = -\dfrac{2mg}
{\hbar^2(q^2 + \mu^2)}$। (ख) $\mu \to 0$ लीजिए, जहाँ $g =
Q_1Q_2/4\pi\varepsilon_0$, और रदरफ़ोर्ड पुनः पाइए। (ग) \emph{कुल}
रदरफ़ोर्ड \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} क्यों अपसरित होता है, और कौन-सा भौतिक
अवयव (परमाणविक इलेक्ट्रॉनों द्वारा परिरक्षण) परिमित
उत्तर लौटाता है? (घ) कण भौतिकी में $\mu =
m_\pi c/\hbar$ वाला युकावा रूप नाभिकीय बल का प्रतिरूप है: उसकी परास
$1/\mu$ निकालिए, $m_\pi c^2 = \qty{140}{MeV}$ के लिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:6}
क्वांटम कठोर गोला (त्रिज्या $a$), $s$ तरंग: बाहर, $u_0 =
\sin(kr + \delta_0)$ जहाँ $u_0(a) = 0$। (क) दिखाइए कि $\delta_0 = -ka$।
(ख) दिखाइए कि $\sigma \to 4\pi a^2$, जब $k \to 0$: \emph{चार} गुना
ज्यामितीय छाया। (ग) इस गुणक की तरंग-व्याख्या दीजिए
(लंबी तरंगदैर्घ्य पर विवर्तन कोई तीखी छाया नहीं बनाता)। (घ) ऊँची
ऊर्जा पर उत्तर $2\pi a^2$ की ओर जाता है --- फिर भी ज्यामितीय का दुगना
(छाया-विवर्तन): कभी बस $\pi a^2$ क्यों नहीं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:7}
हाइड्रोजन के बीस बार्न। प्रोटॉनों पर पड़ते धीमे अध्रुवित न्यूट्रॉन
त्रिक वाहिका को भार $3/4$ के साथ और
एकल को $1/4$ के साथ नमूने में लेते हैं: $\sigma = \pi(3a_{\text{t}}^2 +
a_{\text{s}}^2)$। (क) $a_{\text{t}} = \qty{5.4}{fm}$,
$a_{\text{s}} = \qty{-23.7}{fm}$ के साथ मान निकालिए और मापे गए
\qty{20.4}{b} से तुलिए। (ख) अपने छोटे सांख्यिकीय भार के बावजूद
कौन-सी वाहिका हावी रहती है? (ग) $a_{\text{t}}$ के आकार और चिह्न को
ड्यूटरॉन के लगभग-शून्य बंधन से जोड़िए; और $a_{\text{s}} < 0$ एकल
निकाय के बारे में क्या कहता है? (घ) जल रिएक्टर न्यूट्रॉनों का इतना
प्रभावी मंदक क्यों है --- और उसके लिए यह \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} क्यों मायने
रखता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:8}
रामसाउर--टाउनसेंड। किसी आते हुए इलेक्ट्रॉन के लिए आर्गन परमाणु का
प्रतिरूप परास $R =
\qty{0.2}{nm}$ के आकर्षी वर्ग कूप से बनाइए। (क) भीतर तरंग संख्या $K =
\sqrt{2m(E + V_0)}/\hbar$ है: मिलान ऐसा हो सकता है कि बाहरी तरंग ठीक
$\delta_0 = \pi$ के साथ निकले --- तब $s$-तरंग
\omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} क्या है? (ख) कूप के प्रबल होते हुए भी परमाणु उस ऊर्जा
पर लगभग अदृश्य क्यों हो जाता है? (ग) वह $V_0$ आकलिए
जो पारदर्शिता को $E = \qty{1}{eV}$ के पास ले आए (कूप में आधी
तरंगदैर्घ्य समाइए: $KR \approx \pi$)। (घ) हीलियम और
नियॉन तुलनीय ऊर्जाओं पर यह प्रभाव \emph{नहीं} दिखाते --- क्यों (उनके
कूप इतने उथले हैं कि $\delta_0$ $\pi$ तक नहीं पहुँचता) --- और इस
प्रभाव का बस होना ही पदार्थ-तरंगों के बारे में क्या सिद्ध करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:9}
किसी विशुद्ध $s$-तरंग आयाम $f = \eu^{\iu\delta_0}\sin\delta_0/
k$ के लिए: (क) $\sigma$ निकालिए; (ख) $(4\pi/k)\operatorname{Im}
f(0)$ निकालिए और प्रकाशिक प्रमेय सत्यापित कीजिए; (ग) दिखाइए कि अधिकतम
$s$-तरंग \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} $4\pi/k^2$ है (एकात्मकता सीमा) और

बार्न में निकालिए; (घ) अन्योन्यक्रिया कितनी भी प्रबल हो, कोई अनुनाद
$\sigma_0$ को उस छत से आगे क्यों नहीं धकेल सकता?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:10}
एक न्यूट्रॉन अनुनाद। यूरेनियम-238 न्यूट्रॉनों के लिए
$E_{\text{R}} = \qty{6.67}{eV}$ पर एक प्रसिद्ध अनुनाद दिखाता है,
जिसकी कुल चौड़ाई $\Gamma = \qty{27}{meV}$ है। (क) अनुनाद का जीवनकाल
निकालिए। (ख) उस ऊर्जा पर $4\pi/k^2$ बार्न में निकालिए, और मापे गए
शिखर $\sim\qty{2e4}{b}$ से तुलिए (अंतर प्रत्यास्थ और परिग्रहण
वाहिकाओं के बीच का शाखन गुणक है --- टिप्पणी कीजिए)। (ग) ताप-इकाइयों
में चौड़ाई: ईंधन के ताप के साथ इस अनुनाद का डॉप्लर चौड़ीकरण रिएक्टर
की स्थिरता के लिए क्यों मायने रखता है (प्रतिपुष्टि का कौन-सा चिह्न)?
(घ) दो वाक्यों में: ऐसे अनुनाद $^{238}$U को अधितापीय परास में
न्यूट्रॉन-अवशोषक कैसे बना देते हैं, और वह रिएक्टर की अभिकल्पना के लिए
क्यों केंद्रीय है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:11}
शीत परमाणु एक ही संख्या पर जीते हैं। (क) सर्वसम बोसॉनों के लिए
कम-ऊर्जा \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} $8\pi a^2$ है (संपोषी विनिमय
व्यतिकरण --- मान लीजिए): रुबिडियम के लिए $a =
100\,a_0$ के साथ मान निकालिए। (ख) $T = \qty{100}{nK}$ पर जाँचिए कि $ka \ll 1$
($k$ को $k_{\text{B}}T = \hbar^2k^2/2m$, $m =
\qty{1.4e-25}{kg}$ से निकालिए): गैस सचमुच $a$ के सिवा सब भूल जाती है।
(ग) किसी ``फ़ेशबाख'' अनुनाद के पास कोई चुंबकीय क्षेत्र किसी
आणविक स्तर को शून्य ऊर्जा से खींच ले जाता है और $a$ अपसरित होकर चिह्न
बदल लेता है, ठीक \cref{ex:b3:scattering-theory:np} की तरह: तब गैस की
अन्योन्यक्रियाओं का इच्छानुसार क्या होता है? (घ) यह
सुर-मिलान --- डायल पर अन्योन्यक्रिया की प्रबलता --- किसी भौतिकीविद् का
स्वप्न-उपकरण क्यों है (एक उपयोग बताइए: अणु बनाना, प्रबल-युग्मित
पदार्थ का अनुकरण, संघनितों का ढहना)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:12}
रूप गुणक: आवेश-मेघ तौलना। किसी विस्तृत आवेश
घनत्व $\rho(\vect r)$ के लिए बॉर्न आयाम बिंदु-उत्तर को
$F(q) = \int\rho(\vect r)\eu^{\iu\vect q\cdot\vect r}
\dd^3r/Q$ से गुणा कर देता है। (क) दिखाइए कि $F(0) = 1$। (ख) छोटे $q$ के लिए प्रसार कीजिए: $F
\approx 1 - q^2\langle r^2\rangle/6$: निम्न-$q$ ह्रास नापना
$\langle r^2\rangle$ को तौलता है। (ग) इलेक्ट्रॉन--प्रोटॉन
प्रकीर्णन में $\sqrt{\langle r^2\rangle} \approx \qty{0.84}{fm}$ बैठता है:
\cref{pb:b3:hydrogen-atom:1} का कौन-सा स्वतंत्र मापन
(म्यूऑनी हाइड्रोजन) उसकी जाँच करता है,
और उनकी ऐतिहासिक असहमति (``प्रोटॉन-त्रिज्या पहेली'')
इतनी गंभीरता से क्यों ली गई? (घ) $q \gg 1/r_{\text{p}}$ पर
प्रत्यास्थ रूप गुणक ढह जाता है, पर कठोर प्रकीर्णन
\emph{बिंदु-सदृश अवयवों} से बना रहता है: एक वाक्य में बताइए कि
रदरफ़ोर्ड के प्रयोग के 1968 वाले गहन-अप्रत्यास्थ संस्करण ने प्रोटॉन
के भीतर क्या पाया।
\end{exercise}

\section{समस्या: वे चमक गिनने वाले, जिन्होंने नाभिक पाया}

\begin{problem}[{सप्ताहांत समस्या --- रदरफ़ोर्ड प्रकीर्णन, ज्यामिति
से क्वार्क तक}]\label{pb:b3:scattering-theory:1}
1909 में गाइगर और मार्सडेन अँधेरे में बैठकर संदीप्ति की चमकें गिनते
थे: सोने की पन्नी से दागे गए अल्फ़ा कण, जिनमें से कुछ लगभग पीछे को
लौट आते थे --- ``मानो कोई गोला टिशू पेपर से टकराकर लौट आया हो''।
1911 में उन गिनतियों के रदरफ़ोर्ड-कृत विश्लेषण ने नाभिकीय परमाणु
गढ़ा। यह समस्या उसे फिर से बनाती है। आँकड़े: अल्फ़ा ऊर्जा $E =
\qty{5.5}{MeV}$, आवेश $z = 2$; सोना $Z = 79$, पन्नी की मोटाई $t
= \qty{0.4}{\micro m}$, $n = \qty{5.9e28}{m^{-3}}$; $k_C =
1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$।

\medskip\noindent\textbf{भाग I --- एक अल्फ़ा, एक नाभिक।}
\begin{enumerate}
  \item प्लम-पुडिंग चित्र में (आवेश पूरे परमाणु पर फैला,
        \qty{e-10}{m}) आकलिए कि कोई एक परमाणु किसी \qty{5.5}{MeV}
        अल्फ़ा को अधिक से अधिक कितना विक्षेपित कर सकता था (फैले हुए
        गोले का क्षेत्र: $\theta \sim$ बल $\times$ समय / संवेग
        $\sim 2Zze^2k_C/(E\cdot\qty{e-10}{m})$ रेडियन में)। क्या
        ऐसे धक्कों का कोई भी संचय अल्फ़ा कणों को पीछे भेज सकता था?
  \item अब आवेश को बिंदु मान लीजिए। सम्मुख संघट्ट के लिए निकटतम
        पहुँच की दूरी $d_0$ लिखिए (सारी गतिज ऊर्जा कूलॉम ऊर्जा में)
        और उसका मान निकालिए।
  \item अतः लगभग-पश्च प्रकीर्णन की कोई मापनीय दर परमाणु के धन आवेश
        की सांद्रता के बारे में \emph{तत्काल} क्या संकेत देती है?
  \item किसी व्यापक संघट्ट प्राचल $b$ के लिए चिरसम्मत कक्षा
        $\tan(\theta/2) = d_0/2b$ देती है (मान लिया गया --- वर्ष 1
        के खंड के गुरुत्व अध्याय का अतिपरवलय, प्रतिकर्षण के साथ):
        $b \to 0$ और $b \to
        \infty$ पर उसकी सीमाएँ जाँचिए।
  \item वह संघट्ट प्राचल निकालिए जो किसी अल्फ़ा को $90^\circ$ से
        अधिक प्रकीर्ण करता है, और पन्नी से गुज़रते अल्फ़ा कणों का वह
        अंश जो किसी नाभिक के उतने भीतर आ जाते हैं ($n t\,\pi
        b^2$)।
  \item गाइगर और मार्सडेन ने पाया कि लगभग 8000 में से 1 अल्फ़ा
        $90^\circ$ से आगे विक्षेपित होता है: तुलना कीजिए।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट}।}
\begin{enumerate}[resume]
  \item जिन अल्फ़ा कणों का संघट्ट प्राचल $[b, b + \dd b]$ में है, वे
        $[\theta, \theta + \dd\theta]$ में प्रकीर्ण होते हैं: $\dd\sigma =
        2\pi b\,|\dd b|$ और कक्षा-संबंध से व्युत्पन्न कीजिए
        \[
        \frac{\dd\sigma}{\dd\Omega} =
        \Big(\frac{d_0}{4}\Big)^2\frac{1}{\sin^4(\theta/2)} .
        \]
  \item सत्यापित कीजिए कि यह
        \cref{ex:b3:scattering-theory:rutherford} के बॉर्न/क्वांटम
        उत्तर से मेल खाता है ($d_0$ को $E$ के पदों में फिर लिखिए)।
  \item $\dd\sigma/\dd\Omega$ का मान $\theta = 30^\circ$,
        $60^\circ$, $120^\circ$ पर निकालिए (बार्न प्रति स्टेरेडियन में)।
  \item स्थिर ज्यामिति पर गाइगर और मार्सडेन की 1913 की गिनतियाँ
        $1{:}\,\sin^{-4}(\theta/2)$ की तरह बढ़ती हैं: $30^\circ$ और
        $120^\circ$ के बीच अनुमानित गिनती-अनुपात निकालिए और
        ध्यान दीजिए कि
        उनके आँकड़े दर की पाँच कोटियों तक ऐसे ही अनुपातों का
        अनुसरण करते थे।
  \item उसी सूत्र का $Z^2$ प्रयोग को नाभिकीय आवेश \emph{तौलने}
        क्यों देता है --- और ऐसे अनुरूपणों ने $Z$ (सोना: 79) को
        परमाणु क्रमांक के रूप में पक्का करने में कैसे मदद की?
  \item वह $1/E^2$: अल्फ़ा ऊर्जा दुगनी करने पर पूरे कोणीय प्रतिरूप
        का क्या होता है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- नाभिक का आकार।}
\begin{enumerate}[resume]
  \item कोण $\theta$ पर निकटतम पहुँच: $d(\theta) =
        \tfrac{d_0}2[1 + 1/\sin(\theta/2)]$ (मान लिया गया)।
        $\theta = 180^\circ$ और \qty{60}{\degree} के लिए मान निकालिए।
  \item जब तक $d(\theta)$ नाभिकीय त्रिज्या से बड़ा है, तब तक
        बिंदु-आवेश सूत्र चलता है: रदरफ़ोर्ड के सत्यापित $180^\circ$
        बिंदुओं ने सोने के नाभिक के आकार पर कौन-सी ऊपरी सीमा
        स्थापित की?
  \item ऊँची ऊर्जा वाले प्रक्षेप्यों के साथ बड़े कोण पर गिनतियाँ
        रदरफ़ोर्ड से \emph{नीचे गिर जाती हैं}: यह विचलन किसका संकेत
        है (कौन-सा नया बल, किस दूरी पर)?
  \item आधुनिक इलेक्ट्रॉन प्रकीर्णन नाभिकीय त्रिज्याएँ $R
        \approx r_0A^{1/3}$ देता है, जहाँ $r_0 = \qty{1.2}{fm}$:
        सोने ($A = 197$) के लिए मान निकालिए और अपनी सीमा से तुलिए।
  \item वह $A^{1/3}$: वह नाभिकीय पदार्थ के घनत्व के बारे में क्या
        कहता है (अचर? बढ़ता?) --- एक तथ्य, जिस पर
        \cref{ch:b3:nuclear-physics} खड़ा होगा।
  \item (आवेश-अंधा) प्लम-पुडिंग प्रतिरूप वर्णक्रममिति से नहीं,
        बल्कि \emph{इसी} प्रयोग से क्यों मरा?
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- रदरफ़ोर्ड के नाती-पोते।}
\begin{enumerate}[resume]
  \item 1911 और 1968 के बीच ठीक-ठीक अनुवाद-सारणी दीजिए:
        अल्फ़ा $\to$ इलेक्ट्रॉन पुंज, सोने का परमाणु $\to$ प्रोटॉन,
        नाभिक $\to$ क्वार्क --- SLAC में
        ``अप्रत्याशित बड़े-कोण की घटनाओं'' की भूमिका किसने निभाई?
  \item \emph{इलेक्ट्रॉन} अधिक स्वच्छ जाँच-कण क्यों हैं
        (अल्फ़ा--नाभिक प्रकीर्णन की कौन-सी उलझन उनमें नहीं होती)?
  \item बॉर्न की भाषा में: ऊँचे $q$ पर प्रत्यास्थ प्रकीर्णन रूप
        गुणक का ढहना नापता है (एक नरम मेघ), जबकि गहन-अप्रत्यास्थ
        दरें बड़ी और लगभग-बिंदुवत बनी रहीं: जो निष्कर्ष निकला वह
        बताइए।
  \item तीन पंक्तियों में एक सदी का चाप: (जाँच-ऊर्जा, वियोजित
        दूरी) दीजिए --- 1911 के अल्फ़ा ($\sim\qty{e-14}{m}$),
        1968 का SLAC (\qty{20}{GeV}, $\sim\qty{e-16}{m}$), और
        LHC (\unit{TeV} पैमाना, $\sim\qty{e-19}{m}$) --- यह
        नियम काम में लेते हुए: ``वियोजन $\sim \hbar c/(qc)$''।
  \item न्यूट्रॉन, जो अनावेशित हैं, केवल नाभिकों से (और चुंबकीय
        आघूर्णों से) प्रकीर्ण होते हैं: दो चीज़ें बताइए जिन्हें
        न्यूट्रॉन प्रकीर्णन इसलिए पदार्थों में मानचित्रित कर लेता है
        और एक्स-किरणें ठीक से नहीं देखतीं (हाइड्रोजन जैसे हलके
        परमाणु; चुंबकीय क्रम)।
  \item LHC पर खोज के \omterm{def:b3:scattering-theory:cross-section}{अनुप्रस्थ काट} \emph{फ़ेम्टोबार्न} में नापे
        जाते हैं ($\qty{e-43}{m^2}$): बार्न से
        फ़ेम्टोबार्न तक पंद्रह कोटियाँ हैं --- यह इस बारे में क्या
        कहता है कि दिलचस्प संघट्ट कितने विरल हैं, और प्रदीप्ति
        (पहुँचाया गया अभिवाह) ऊर्जा जितनी ही बहुमूल्य क्यों है?
  \item नामित परिणाम का सार दीजिए: एक $\sin^{-4}(\theta/2)$ नियम,
        जो चमक-दर-चमक सत्यापित हुआ, ने परमाणु के द्रव्यमान का
        $99.98\%$ उसके आकार के $10^{-14}$ जितने आयतन में रख दिया ---
        और वही प्रयोग, बार-बार और कठिन ढंग से दोहराया जाकर, अगली
        परत खोजना कभी नहीं रुका।
\end{enumerate}
\end{problem}
