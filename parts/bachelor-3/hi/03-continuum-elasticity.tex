\chapter{सतत माध्यम यांत्रिकी और प्रत्यास्थता}\label{ch:b3:continuum-elasticity}

किसी लंबी इस्पात-पटरी पर कान लगाइए, जब दूर खड़ा कोई कारीगर उस पर चोट
करे: आपको दो ठनक सुनाई देती हैं --- एक इस्पात से होकर, एक हवा से होकर,
और उनके बीच एक सेकंड या उससे अधिक का अंतर। चट्टान, इस्पात, हड्डी और
रबर --- सब बलों और तरंगों को उसी तरह ढोते हैं जैसे कोई तरल दाब को ढोता
है, पर उनके पास कुछ ऐसा है जो किसी तरल के पास नहीं: वे \emph{आकार के
परिवर्तन} का प्रतिरोध करते हैं, केवल आयतन के परिवर्तन का नहीं। वर्ष 2
के खंड ने तरलों की यांत्रिकी दो विचारों पर खड़ी की थी, अर्थात्
भौतिक कण और दाब-क्षेत्र पर; यह अध्याय उन्हें ठोसों तक बढ़ाता है। हर कण
का विस्थापन एक क्षेत्र बन जाता है, उसका स्थानिक विरूपण एक
\emph{विकृति}, भीतरी बल एक \emph{प्रतिबल}, और उनके बीच ठोस का पहचान-पत्र
खड़ा होता है, अर्थात् \omterm{thm:b3:continuum-elasticity:hooke}{हुक का नियम}, अपने दो
प्रत्यास्थ नियतांकों के साथ। इसका लाभ रोज़मर्रा से --- केबल क्यों खिंचते
हैं, बोतलें क्यों फटती हैं, गोताखोरी का तख़्ता कैसे झुकता है --- लेकर
ग्रह तक फैला है: भूकंप पृथ्वी के भीतर ठीक वे दो प्रकार की \omterm{prop:b3:continuum-elasticity:waves}{प्रत्यास्थ
तरंगें} भेजते हैं जिनकी यह अध्याय भविष्यवाणी करता है, और उनके पहुँचने के
समय ही हमारे ग्रह के अंतरतम की पहली आवाज़ थे।

\section{विकृति: विरूपण का वर्णन}

\begin{definition}[विस्थापन क्षेत्र और विकृति प्रदिश]\label{def:b3:continuum-elasticity:strain}
\index{विस्थापन क्षेत्र}\index{विकृति प्रदिश}\index{विस्फार}
जब कोई ठोस विरूपित होता है, तब आरंभ में $\vect r$ पर स्थित कण
$\vect r + \vect u(\vect r)$ पर चला जाता है: $\vect u$ ही
\emph{विस्थापन क्षेत्र} है। छोटे विरूपणों के लिए स्थानिक विरूपण
\emph{विकृति प्रदिश} से नापा जाता है, जो यह सममित सारणी है
\[
\varepsilon_{ij} = \frac12\Big(
 \frac{\partial u_i}{\partial x_j} + \frac{\partial u_j}{\partial x_i}
\Big) , \qquad i, j \in \{x, y, z\} .
\]
कोई विकर्ण घटक $\varepsilon_{xx}$ पदार्थ का $x$ के अनुदिश आपेक्षिक
दीर्घीकरण है (विमाहीन; प्रचालन में इस्पात-केबल के लिए लगभग
$\num{e-3}$); कोई अ-विकर्ण घटक $\varepsilon_{xy}$ पदार्थ की
$x$ और $y$ दिशाओं के बीच के कोण के घटने का आधा है
--- अर्थात् \emph{अपरूपण}। अनुरेख आयतन का आपेक्षिक परिवर्तन है, अर्थात्
\emph{विस्फार}:
\[
\frac{\delta V}{V} = \varepsilon_{xx} + \varepsilon_{yy} +
\varepsilon_{zz} = \operatorname{div}\vect u .
\]
\end{definition}

\begin{example}[तीन प्रारंभिक विरूपण]\label{ex:b3:continuum-elasticity:elementary}
एकसमान \omterm{def:b3:continuum-elasticity:strain}{विस्फार} $\vect u = \alpha\vect r$: $\varepsilon_{ij} =
\alpha\,\delta_{ij}$, आयतन-परिवर्तन $3\alpha$, अपरूपण कोई नहीं। सरल
अपरूपण $\vect u = (\gamma y, 0, 0)$: $\varepsilon_{xy} = \gamma/2$, सभी
विकर्ण घटक शून्य --- आकार बदलता है, आयतन नहीं। दृढ़
घूर्णन $\vect u = (-\omega y, \omega x, 0)$: $\varepsilon_{ij} = 0$
सर्वसमतः --- $\partial u_i/\partial x_j$ का प्रतिसममित भाग,
जिसे सममितीकरण छोड़ देता है, ठीक वही भाग है जो बिना विरूपण किए
घुमाता है। विकृति केवल विरूपण नापती है।
\end{example}

\section{प्रतिबल: भीतरी बलों का वर्णन}

\begin{definition}[कर्षण और प्रतिबल प्रदिश]\label{def:b3:continuum-elasticity:stress}
\index{प्रतिबल प्रदिश}\index{कर्षण}
ठोस को मन ही मन किसी छोटे पृष्ठ $\dd S$ के अनुदिश काटिए, जिसका अभिलंब
$\vect n$ हो: $+\vect n$ की ओर का पदार्थ दूसरी ओर के पदार्थ को
बल $\dd\vect F = \boldsymbol\sigma(\vect n)\,\dd S$ से खींचता है ---
यही \emph{कर्षण} है। यह बल $\vect n$ पर रैखिक रूप से निर्भर है (कोशी),
अतः उसे \emph{प्रतिबल प्रदिश} $\sigma_{ij}$ में बाँधा जा सकता है:
\[
\dd F_i = \sum_j \sigma_{ij}\,n_j\,\dd S ,
\]
पास्कल में। $\sigma_{xx}$ किसी ऐसे फलक के पार का खिंचाव ($> 0$: तनाव)
या धक्का ($< 0$: संपीडन) है जिसका अभिलंब $x$ हो; $\sigma_{xy}$ वह बल है
जो $x$ के अनुदिश हो और जिसे $y$ के अभिलंब वाला फलक ढो रहा हो ---
अर्थात् अपरूपण प्रतिबल। विराम में कोई तरल विशेष स्थिति
$\sigma_{ij} = -p\,\delta_{ij}$ है: दाब हर फलक पर बराबर धकेलता है और
किसी पर अपरूपण नहीं करता, और इसीलिए तरल बहते हैं --- वे स्थैतिक अपरूपण
ढो ही नहीं सकते।
\end{definition}

\begin{proposition}[साम्य और सममिति]\label{prop:b3:continuum-elasticity:equilibrium}
किसी ठोस में, जो प्रति एकांक आयतन के पिंड-बल $\vect f$ (जैसे
$\rho\vect g$) के अंतर्गत साम्य में हो,
\[
\sum_j\frac{\partial\sigma_{ij}}{\partial x_j} + f_i = 0
\]
हर भीतरी बिंदु पर; और \omterm{def:b3:continuum-elasticity:stress}{प्रतिबल प्रदिश} सममित है,
$\sigma_{ij} = \sigma_{ji}$।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
भुजा $a$ के किसी छोटे घन पर बल संतुलित कीजिए: $x$ के अभिलंब वाले दो
फलकों के \omterm{def:b3:continuum-elasticity:stress}{कर्षण} प्रति एकांक क्षेत्रफल $a\,\partial_x\sigma_{ix}$ जितने
भिन्न होते हैं, और शेष युग्मों के लिए भी वैसे ही; प्रति एकांक आयतन कुल
पृष्ठ-बल $\sum_j\partial_j\sigma_{ij}$ है, जिसे $f_i$ को काटना ही चाहिए
--- वही हिसाब जिसने वर्ष 1 के खंड के तरल-स्थैतिकी में
$-\vect\nabla p + \rho\vect g = \vect 0$ दिया था। सममिति: घन के केंद्र
के परितः अपरूपण प्रतिबलों का बल-आघूर्ण अग्रणी कोटि पर $(\sigma_{xy} -
\sigma_{yx})a^3$ है, जबकि उसका जड़त्व-आघूर्ण
$a^5$ की तरह मापित होता है; असममित प्रतिबल छोटे घनों को अपरिमित तेज़ी
से घुमा देता। पूरा सतत-माध्यम तर्क स्वीकृत है।
\end{proof}

\section{हुक का नियम और प्रत्यास्थ नियतांक}

\begin{theorem}[रैखिक समदैशिक प्रत्यास्थता]\label{thm:b3:continuum-elasticity:hooke}
\index{हुक का नियम}\index{यंग मापांक}\index{प्वासों अनुपात}
छोटी विकृतियों के लिए कोई समदैशिक ठोस रैखिक रूप से अनुक्रिया करता है:
प्रतिबल विकृति के अनुक्रमानुपाती होता है,
\[
\sigma_{ij} = \lambda\,(\varepsilon_{xx} + \varepsilon_{yy} +
\varepsilon_{zz})\,\delta_{ij} + 2\mu\,\varepsilon_{ij} ,
\]
जिसमें दो पदार्थ-नियतांक हैं, अर्थात् \emph{लामे गुणांक} $\lambda$ और
$\mu$ ($\mu$ को $G$ भी लिखते हैं, अर्थात् \emph{अपरूपण मापांक})। सरल
\omterm{def:b3:continuum-elasticity:stress}{कर्षण} परीक्षण में --- $x$ के अनुदिश खींची गई छड़, जिसकी बगलें मुक्त हों
--- वही नियम अभियंता का यह रूप ले लेता है
\[
\varepsilon_{xx} = \frac{\sigma_{xx}}{E} , \qquad
\varepsilon_{yy} = \varepsilon_{zz} = -\nu\,\varepsilon_{xx} ,
\]
जो \emph{\omterm{thm:b3:continuum-elasticity:hooke}{यंग मापांक}} $E$ (खिंचाव के विरुद्ध दृढ़ता) और
\emph{\omterm{thm:b3:continuum-elasticity:hooke}{प्वासों अनुपात}} $\nu$ (पार्श्व संकुचन) परिभाषित करता है, जहाँ
\[
\mu = \frac{E}{2(1 + \nu)} , \qquad
\lambda = \frac{E\nu}{(1 + \nu)(1 - 2\nu)} .
\]
यही हुक का ``जैसा विस्तार, वैसा बल'' (1678) है, जिसे प्रदिश तक उठा
दिया गया है। यह किसी पदार्थ-निर्भर \emph{प्रत्यास्थता सीमा} तक चलता है।
\end{theorem}
\admitted

\begin{remark}[परिमाण की कोटियाँ]\label{rem:b3:continuum-elasticity:orders}
इस्पात के लिए $E \approx \qty{200}{GPa}$, ऐलुमिनियम और खिड़की के काँच
के लिए \qty{70}{GPa}, ग्रेनाइट के लिए \qty{50}{GPa}, हड्डी और
कंक्रीट के लिए \qty{15}{GPa}, रेशे के अनुदिश लकड़ी के लिए
\qty{10}{GPa}, और रबर के लिए केवल कुछ MPa --- परिमाण की पाँच कोटियाँ,
अर्थात् किसी पुल और किसी रबर-पट्टी के बीच का फैलाव। $\nu$, $0$ (लगभग
काग) और $1/2$ (रबर) के बीच रहता है: $\nu = 1/2$ का अर्थ है
आयतन-रक्षी विरूपण, और $\nu \approx 0.3$ धातुओं के लिए प्रारूपिक है।
इस्पात में $\num{e-3}$ की विकृति का अर्थ ही
$\sigma = \qty{200}{MPa}$ है, जो साधारण श्रेणियों के पराभव-प्रतिबल के
पास है: रोज़मर्रा की प्रत्यास्थता एक हज़ारवें भाग से नीचे ही रहती है।
\end{remark}

\begin{example}[लिफ़्ट का केबल]\label{ex:b3:continuum-elasticity:cable}
\qty{2.0}{cm^2} अनुप्रस्थ काट का \qty{60}{m} लंबा इस्पात-केबल
\qty{1000}{kg} की डिबिया ढोता है: $\sigma = mg/S = \qty{49}{MPa}$, $\varepsilon =
\sigma/E = \num{2.5e-4}$, खिंचाव $\varepsilon L = \qty{15}{mm}$ ---
और केबल दृढ़ता $k = ES/L =
\qty{6.7e5}{N/m}$ की स्प्रिंग की तरह व्यवहार करता है: सूत्र $k = ES/L$
वह तरीका है जिससे कोई सतत माध्यम वर्ष 1 के खंड की स्प्रिंगें लौटा देता है।
\end{example}

\begin{proposition}[प्रत्यास्थ ऊर्जा]\label{prop:b3:continuum-elasticity:energy}
\index{प्रत्यास्थ ऊर्जा}
कोई विकृत ठोस प्रति एकांक आयतन यह ऊर्जा-घनत्व संचित करता है
\[
w = \frac12\sum_{i,j}\sigma_{ij}\,\varepsilon_{ij}
\qquad\text{(सरल कर्षण: } w = \tfrac12 E\varepsilon^2 =
\sigma^2/2E\text{)} ,
\]
अर्थात् $\tfrac12 kx^2$ का त्रिविमीय रूप।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
सरल \omterm{def:b3:continuum-elasticity:stress}{कर्षण} में प्रतिबल को $0$ से $\sigma$ तक लाने पर प्रति एकांक आयतन
कार्य $\int_0^\varepsilon\sigma'\,\dd\varepsilon' =
\int_0^\varepsilon E\varepsilon'\,\dd\varepsilon' = \tfrac12
E\varepsilon^2$ होता है --- अर्थात् प्रतिबल--विकृति रेखा के नीचे का
क्षेत्रफल, ठीक वैसे ही जैसे किसी स्प्रिंग के लिए। व्यापक द्विघात रूप
घटकों के अध्यारोपण से निकलता है; स्वीकृत।
\end{proof}

\begin{method}[लघु-प्रत्यास्थता की समस्याएँ हल करना]\label{met:b3:continuum-elasticity:method}
(1) ज्यामिति और भार पहचानिए; अनुमान लगाइए कि कौन-से प्रतिबल-घटक
शून्येतर हैं (खींची गई छड़: केवल $\sigma_{xx}$; ऐंठा गया
तार: अपरूपण; दाबित खोल: स्पर्शरेखीय तनाव)। (2) किसी सुविचारित टुकड़े पर
बल-संतुलन लिखिए --- अर्ध-बेलन, फाँक या घन। (3) हुक के नियम से प्रतिबल
को विकृति में बदलिए, और समाकलन करके विकृति को
विस्थापन में। (4) ऊर्जाएँ $w = \sigma^2/2E$ से (या अपरूपण में
$\sigma^2/2G$ से)। (5) सदा जाँचिए कि विकृति छोटी बनी हुई है और
प्रतिबल प्रत्यास्थता सीमा से नीचे --- अन्यथा उत्तर ऐसे ठोस का वर्णन
करता है जो अब बचा ही नहीं।
\end{method}

\section{प्रत्यास्थ तरंगें}

\begin{proposition}[किसी ठोस की दो आवाज़ें]\label{prop:b3:continuum-elasticity:waves}
\index{प्रत्यास्थ तरंगें}\index{P तरंग}\index{S तरंग}
घनत्व $\rho$ के किसी असीम प्रत्यास्थ ठोस में छोटे विक्षोभ दो स्वतंत्र
प्रकार की तरंगों के रूप में संचरित होते हैं:
\emph{अनुदैर्घ्य} (P) तरंगें, जिनमें पदार्थ संचरण-दिशा के अनुदिश
संपीडन और \omterm{def:b3:continuum-elasticity:strain}{विस्फार} से दोलन करता है, और जिनकी चाल है
\[
c_{\text{P}} = \sqrt{\frac{\lambda + 2\mu}{\rho}} ,
\]
तथा \emph{अनुप्रस्थ} (S) तरंगें, जिनमें पदार्थ बगल की ओर अपरूपित होता है, और जिनकी चाल है
\[
c_{\text{S}} = \sqrt{\frac{\mu}{\rho}} < c_{\text{P}} .
\]
किसी तरल में $\mu = 0$ होता है: कोई \omterm{prop:b3:continuum-elasticity:waves}{S तरंग} नहीं --- अपरूपण संचरित नहीं
हो सकता --- और P चाल घटकर वर्ष 2 के खंड की ध्वनि-चाल रह जाती है।
$\nu = 1/4$ (प्रारूपिक चट्टान) के लिए $\lambda = \mu$ और $c_{\text{P}} =
\sqrt3\,c_{\text{S}}$।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
कोई समतल विक्षोभ $\vect u = \vect u(x, t)$ लीजिए। प्रति एकांक आयतन
न्यूटन का नियम $\rho\,\partial_t^2u_i = \sum_j\partial_j\sigma_{ij}$ है
(साम्य-समीकरण, जिसमें जड़त्व लौटा दिया गया हो)। अनुदैर्घ्य
घटक के लिए \omterm{thm:b3:continuum-elasticity:hooke}{हुक का नियम} $\sigma_{xx} = (\lambda +
2\mu)\,\partial_xu_x$ देता है (समतल तरंग में पार्श्व विकृतियाँ लुप्त
रहती हैं, क्योंकि पड़ोसी पदार्थ पार्श्व छूट नहीं देता), अतः
$\rho\,\partial_t^2u_x = (\lambda + 2\mu)\,\partial_x^2u_x$: चाल
$c_{\text{P}}$ पर दालंबेर। अनुप्रस्थ घटक के लिए $\sigma_{yx} =
2\mu\varepsilon_{yx} = \mu\,\partial_xu_y$, जिससे
$\rho\,\partial_t^2u_y = \mu\,\partial_x^2u_y$: चाल $c_{\text{S}}$।
कोई व्यापक विक्षोभ इन्हीं दो कुलों में बँट जाता है --- यह स्वीकृत है।
\end{proof}

\begin{example}[पटरी और भूकंप]\label{ex:b3:continuum-elasticity:rail}
इस्पात: $c_{\text{P}} \approx \qty{5.9}{km/s}$ --- \qty{2}{km} दूर
पटरी पर की गई चोट इस्पात से होकर \qty{0.34}{s} में और
हवा से होकर ($\qty{340}{m/s}$) \qty{5.9}{s} में पहुँचती है: दो ठनक।
ग्रेनाइट ($E = \qty{75}{GPa}$, $\nu = 1/4$, $\rho =
\qty{2.7e3}{kg/m^3}$): $c_{\text{P}} \approx \qty{5.8}{km/s}$,
$c_{\text{S}} \approx \qty{3.3}{km/s}$। इसलिए कोई भूकंप अपनी घोषणा दो
बार करता है: पहले तीखा P झटका, फिर कुछ सेकंड बाद धीमी और अधिक प्रबल
S कँपकँपी --- और यह विलंब दूरी नाप देता है
(\cref{pb:b3:continuum-elasticity:1})। भूकंप की पूर्व-चेतावनी की
प्रणालियाँ उसी विलंब के भीतर जीती हैं: \omterm{prop:b3:continuum-elasticity:waves}{P तरंग}, और जो रेडियो-संदेश वह
छेड़ती है, दोनों उस \omterm{prop:b3:continuum-elasticity:waves}{S तरंग} से आगे निकल जाते हैं जो नुक़सान करती है।
\end{example}

\begin{remark}[डोरियाँ, छड़ें और ध्वनि, फिर से]\label{rem:b3:continuum-elasticity:limits}
वर्ष 2 के खंड की सारी तरंगें इसी अध्याय की सीमाएँ हैं। कोई पतली
छड़, जो बगल में पतली होने को स्वतंत्र हो, संपीडन तरंगें
$\sqrt{E/\rho}$ पर ढोती है ($c_{\text{P}}$ से धीमी: पार्श्व छूट
अनुक्रिया को नरम कर देती है); कोई तनी डोरी अनुप्रस्थ तरंगें
$\sqrt{T/\rho_\ell}$ पर ढोती है, जहाँ दृढ़ता का स्थान तनाव ले लेता है;
और कोई तरल केवल $\lambda$ --- अपना आयतन-मापांक --- रखता है और उसके साथ
अकेली ध्वनि-चाल $\sqrt{\lambda/\rho}$।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:1}
(क) मात्रक जाँचिए: kg, m, s के पदों में पास्कल क्या है, और
विकृति का मात्रक क्या है? (ख) कोई इस्पात-केबल $\sigma =
\qty{200}{MPa}$ पर काम करता है: उसकी विकृति निकालिए। (ग) कोई रबर-पट्टी ($E \approx
\qty{2}{MPa}$) खिंचकर दुगुनी लंबी कर दी जाती है: वह कैसी
``विकृति'' है, और इस अध्याय का ढाँचा उस पर सचमुच लागू क्यों नहीं होता?
(घ) अपने कार्यकारी प्रतिबल पर संचित प्रत्यास्थ ऊर्जा-घनत्व के अनुसार
क्रम दीजिए: इस्पात \qty{200}{MPa} पर, रबर \qty{1}{MPa} पर।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:2}
हर \omterm{def:b3:continuum-elasticity:strain}{विस्थापन क्षेत्र} का \omterm{def:b3:continuum-elasticity:strain}{विकृति प्रदिश} निकालिए और विरूपण का वर्णन
कीजिए: (क) $\vect u = \alpha(x, y, z)$; (ख) $\vect u =
(\gamma y, 0, 0)$; (ग) $\vect u = (-\omega y, \omega x, 0)$; (घ)
$\vect u = (\varepsilon x, -\nu\varepsilon y, -\nu\varepsilon z)$ ---
इसे कौन-सा प्रयोग साकार करता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:3}
ऊँचाई $h$ के किसी ग्रेनाइट-स्तंभ का आधार प्रतिबल $\sigma

निकालिए। (ख) ग्रेनाइट लगभग \qty{200}{MPa} पर चूर हो जाता है: सबसे
ऊँचा स्तंभ कितना होगा? (ग) एवरेस्ट \qty{8.8}{km} ऊँचा है: टिप्पणी
कीजिए। (घ) कोई पर्वत अपनी ही चट्टान के स्तंभ से थोड़ा ऊँचा क्यों हो
सकता है (आकार के विषय में सोचिए)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:continuum-elasticity:4}
कोई इस्पात-छड़ ($E = \qty{200}{GPa}$, $\nu = 0.30$)
$\varepsilon = \num{e-3}$ जितनी खींची जाती है। (क) पार्श्व विकृति?
(ख) आयतन का आपेक्षिक परिवर्तन? (ग) किस $\nu$ के लिए आयतन बिलकुल न
बदलता, और कौन-सा सामान्य पदार्थ उसके पास है? (घ) $\nu > 1/2$
असंभव क्यों है (द्रवस्थैतिक संपीडन पर विचार कीजिए)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:5}
परिधीय प्रतिबल। कोई पतली दीवार वाला बेलन (त्रिज्या $R$, दीवार की मोटाई $t \ll
R$) दाब $p$ रोके है। (क) एकांक लंबाई के अर्ध-बेलन पर बल संतुलित करके
दिखाइए कि दीवार स्पर्शरेखीय प्रतिबल
$\sigma_\theta = pR/t$ ढोती है। (ख) दिखाइए कि अनुदैर्घ्य प्रतिबल
(किसी अनुप्रस्थ काट पर संतुलन) $pR/2t$ है: बेलन पर चारों ओर का ज़ोर
लंबाई के ज़ोर से दुगुना पड़ता है --- सॉसेज किस दिशा में फटते हैं?
(ग) कोई गोताखोरी-सिलिंडर: $p =
\qty{200}{bar}$, $R = \qty{9}{cm}$, $t = \qty{5}{mm}$:
$\sigma_\theta$ निकालिए और \qty{700}{MPa} के इस्पात पराभव-प्रतिबल से
तुलना कीजिए। (घ) उच्च-दाब टंकियों के सिरे अर्धगोलाकार क्यों होते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:6}
कोई रबर-गुटका ($E = \qty{3.0}{MPa}$, $\nu \approx 0.5$), आधार $10
\times \qty{10}{cm}$, ऊँचाई \qty{2}{cm}, दो पट्टिकाओं के बीच चिपकाया
गया है; ऊपरी पट्टिका को \qty{150}{N} से बगल में धकेला जाता है।
(क) $G$ निकालिए। (ख) अपरूपण प्रतिबल और अपरूपण कोण
$\gamma = \sigma_{xy}/G$; ऊपरी पट्टिका का पार्श्व
विस्थापन। (ग) रबर के लिए $\nu \approx 1/2$ क्यों है (बहुलक-शृंखलाओं
की उलझन के लिए क्या कठिन है और क्या आसान)? (घ) भूकंप-क्षेत्रों में ऐसे
ही रबर-गुटके पूरी इमारतें ढोते हैं: इस आधार का कौन-सा गुण इमारत की
रक्षा करता है, संपीडन में दृढ़ता या अपरूपण में नरमी?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:7}
कोई चढ़ाई-रस्सी, लंबाई $L = \qty{20}{m}$, अनुप्रस्थ काट $S =
\qty{80}{mm^2}$, प्रभावी मापांक $E = \qty{1.2}{GPa}$। (क) उसकी
दृढ़ता $k = ES/L$। (ख) कोई \qty{80}{kg} का पर्वतारोही
\qty{4}{m} मुक्त रूप से गिरता है और तब रस्सी पकड़ती है: ऊर्जाएँ बराबर
करके रस्सी का अधिकतम खिंचाव ज्ञात कीजिए (द्विघात हल कीजिए; पहली बार में
रोकने के दौरान जारी गिरावट की उपेक्षा कीजिए)। (ग) शिखर बल और
$g$ में शिखर मंदन निकालिए। (घ) कोई रस्सी, जिसने एक कठिन गिरावट झेल ली
हो, सेवामुक्त क्यों कर देनी चाहिए (ऊर्जा कहाँ गई, और
प्रतिबल--विकृति वक्र क्या कहता है)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:8}
ऐंठन। त्रिज्या $a$, लंबाई $L$ और अपरूपण मापांक $G$ वाला कोई तार
कोण $\theta$ जितना ऐंठा जाता है। (क) दिखाइए कि उसके भीतर त्रिज्या $r$
की कोई नली अपरूपण कोण $\gamma(r) = r\theta/L$ झेलती है। (ख) उसका अपरूपण
प्रतिबल $G\gamma$ है: काट पर $r \times$ प्रतिबल का समाकलन करके
प्रत्यानयन बलाघूर्ण $\Gamma = (\pi Ga^4/2L)\,\theta$ निकालिए। (ग) यह $a^4$:
त्रिज्या आधी कीजिए, ऐंठन-दृढ़ता कितने गुना घट जाती है? (घ) इसी अत्यंत
नरमी के कारण कैवेंडिश (1798) ने अपना तुला किसी महीन तार से लटकाया था:
$a = \qty{25}{\micro m}$, $L =
\qty{1}{m}$, $G = \qty{40}{GPa}$ के लिए $C = \pi Ga^4/2L$ निकालिए और
जड़त्व-आघूर्ण $I = \qty{5e-4}{kg.m^2}$ वाली छड़ के साथ आवर्तकाल भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:continuum-elasticity:9}
(क) \cref{prop:b3:continuum-elasticity:waves} की समतल-तरंग व्युत्पत्ति
से $c_{\text{P}}$ और
$c_{\text{S}}$ निकालिए, इस्पात के लिए ($E = \qty{200}{GPa}$, $\nu = 0.29$, $\rho =
\qty{7.85e3}{kg/m^3}$)। (ख) $c_{\text{P}}$ की तुलना पतली छड़ की चाल
$\sqrt{E/\rho}$ से कीजिए और अंतर एक वाक्य में समझाइए। (ग)
जल के लिए ($\mu = 0$, आयतन-मापांक \qty{2.2}{GPa}): S चाल और
P चाल। (घ) किसी \omterm{prop:b3:continuum-elasticity:waves}{P तरंग} की पदार्थ-गति किसी \omterm{prop:b3:continuum-elasticity:waves}{S तरंग} की पदार्थ-गति से
कितने कोण पर भिन्न है, और तीन घटकों वाला भूकंपमापी उन्हें कैसे अलग
पहचानता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:continuum-elasticity:10}
नमन। किसी धरन को वक्रता-त्रिज्या $R$ तक झुकाया जाता है; उसके भीतरी
रेशे छोटे होते हैं, बाहरी रेशे खिंचते हैं, और बीच का एक
\emph{उदासीन पृष्ठ} अपनी लंबाई बनाए रखता है। (क) दिखाइए कि उदासीन
पृष्ठ से दूरी $y$ पर के रेशे की विकृति $\varepsilon = y/R$ है, अतः
प्रतिबल $Ey/R$।
(ख) अनुप्रस्थ काट पर आघूर्ण जोड़कर दिखाइए कि नमन-आघूर्ण
$M = EI/R$ है, जहाँ $I = \int y^2\,\dd S$ (\emph{द्वितीय आघूर्ण});
$I$ को चौड़ाई $b$ और ऊँचाई $h$ के आयत के लिए निकालिए। (ग) यह
$h^3$: चौड़ाई के बल पर झुकाया गया तख़्ता दिखते-दिखते क्यों झुक जाता है,
जबकि वही तख़्ता किनारे पर खड़ा दृढ़ लगता है? $b/h = 5$ के लिए अनुपात
निकालिए। (घ) लंबाई $L$ के किसी उत्थापक के लिए, जिसकी नोक पर $F$ लदा हो,
नोक का झुकाव $\delta = FL^3/3EI$ है (स्वीकृत): $\delta$ का आकलन उस
गोताखोरी-तख़्ते के लिए कीजिए
($L = \qty{1.8}{m}$, $b = \qty{50}{cm}$, $h = \qty{3.5}{cm}$, लकड़ी $E
= \qty{12}{GPa}$) जिस पर \qty{75}{kg} का गोताखोर हो, और टिप्पणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:continuum-elasticity:11}
बंकन। कोई पतला स्तंभ (लंबाई $L$, नमन-दृढ़ता $EI$),
दोनों सिरों पर कीलित, अक्षीय भार $P$ ढोता है। मान लीजिए वह
$y(x)$ जितना बगल में झुक जाता है। (क) दिखाइए कि तब भार खिसकी हुई अक्ष के
परितः नमन-आघूर्ण $M(x) = -Py(x)$ लगाता है, अतः $EI\,y'' = -Py$। (ख)
$y(0) = y(L) = 0$ के साथ दिखाइए कि शून्येतर झुकाव पहली बार ऑयलर के
क्रांतिक भार
$P_{\text{c}} = \pi^2EI/L^2$ पर संभव होता है। (ग) $P_{\text{c}}$ को किसी मीटर-पैमाने
($b = \qty{3}{cm}$, $h = \qty{1.5}{mm}$, $E = \qty{12}{GPa}$) के लिए
निकालिए और अपने हाथ के धक्के से तुलना कीजिए। (घ)
$I$ से समझाइए कि हड्डियाँ, बाँस और साइकिल के ढाँचे \emph{नलियाँ} क्यों होते हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:continuum-elasticity:12}
किसी ठोस में ध्वनि की चाल, परमाणुओं से। किसी ठोस का निदर्शन परमाणुओं की
शृंखलाओं के रूप में कीजिए, जिनका अंतराल $a$ हो और जो दृढ़ता $\kappa$ की
``स्प्रिंगों'' से जुड़े हों; परमाणु का द्रव्यमान $m$। (क) दिखाइए कि
शृंखला को खींचना सतत-माध्यम \omterm{def:b3:continuum-elasticity:stress}{कर्षण} पर $E = \kappa/a$ के साथ प्रतिचित्रित
होता है (प्रति एकांक क्षेत्रफल स्प्रिंगें गिनिए, और प्रति स्प्रिंग
खिंचाव)। (ख) दिखाइए कि $\rho = m/a^3$ और निष्कर्ष निकालिए कि $c =
\sqrt{E/\rho} = a\sqrt{\kappa/m}$। (ग) $\kappa$ का आकलन किसी अंतरा-परमाणुक
बंध की गहराई से कीजिए ($\sim\qty{3}{eV}$, $\sim
\qty{0.1}{nm}$ पर): $\kappa \sim \qty{50}{N/m}$; $a =
\qty{2.5e-10}{m}$ और $m = \qty{1e-25}{kg}$ के साथ $c$ का आकलन
कीजिए। (घ) मापी गई धातुओं की ध्वनि-चालों से तुलना कीजिए, और निष्कर्ष
निकालिए कि रोज़मर्रा की प्रत्यास्थता किस चीज़ से बनी है।
\end{exercise}

\section{समस्या: पृथ्वी को सुनना}

\begin{problem}[{सप्ताहांत समस्या --- भूकंपों ने द्रव क्रोड कैसे
उजागर किया}]\label{pb:b3:continuum-elasticity:1}
कोई एक भूकंप पूरे ग्रह को घंटे-सा बजा देता है, और इस अध्याय की दो
\omterm{prop:b3:continuum-elasticity:waves}{प्रत्यास्थ तरंगें} उसे पार करती हुई उसका एक्स-किरण चित्र ले लेती हैं।
यह समस्या भौतिकी का पीछा ग्रेनाइट में हुक के नियम से लेकर ओल्डम की
1906 की उस खोज तक करती है कि पृथ्वी के भीतर क्रोड है --- और सीधी-किरण
आकलन से उसका आकार भी निकालती है। पर्पटी और प्रावार की चट्टान के लिए
$E = \qty{75}{GPa}$, $\nu = 1/4$,
$\rho = \qty{2.7e3}{kg/m^3}$ लीजिए; पृथ्वी की त्रिज्या $R_{\text{E}} =
\qty{6371}{km}$।

\medskip\noindent\textbf{भाग I --- चट्टान एक प्रत्यास्थ ठोस के रूप में।}
\begin{enumerate}
  \item इस चट्टान के लामे गुणांक $\mu$ और $\lambda$ निकालिए, और
        देखिए कि $\nu = 1/4$ उन्हें बराबर कर देता है।
  \item $c_{\text{P}}$ और $c_{\text{S}}$ निकालिए, और जाँचिए कि
        $c_{\text{P}} = \sqrt3\,c_{\text{S}}$।
  \item कोई भूकंप चट्टान के दो फलकों को कुछ सेकंडों में कई मीटर सरका
        देता है: यदि \qty{2}{m} का सर्पण चट्टान को \qty{50}{km} के
        पैमाने पर ढीला करता हो, तो मुक्त हुई विकृति और जो प्रतिबल जमा
        हुआ था, उनका आकलन कीजिए।
  \item ऊर्जा-घनत्व $w = \tfrac12 E\varepsilon^2$ से प्रति घन मीटर
        \omterm{prop:b3:continuum-elasticity:energy}{प्रत्यास्थ ऊर्जा} का आकलन कीजिए, फिर
        $50 \times 50 \times \qty{20}{km^3}$ के आयतन में ऊर्जा का;
        और उसकी तुलना एक मेगाटन (\qty{4.2e15}{J}) से कीजिए।
  \item \omterm{prop:b3:continuum-elasticity:waves}{S तरंग} प्रायः इमारतों को \omterm{prop:b3:continuum-elasticity:waves}{P तरंग} से अधिक ज़ोर से क्यों हिलाती
        है (नीचे से आती तरंग के लिए भूमि की गति की दिशा के विषय में
        सोचिए)?
  \item दोनों तरंगों में से किसमें भूमि का आयतन बदलता है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- भूकंपमापी की दो आगमन।}
\begin{enumerate}[resume]
  \item कोई केंद्र पहले P का आगमन दर्ज करता है, फिर $\Delta t$ बाद
        S का। दूरी $d$ पर के स्रोत के लिए (जो सीधी किरणों के लिए
        पर्याप्त निकट हो) दिखाइए कि
        \[
        d = \frac{\Delta t}{\dfrac{1}{c_{\text{S}}} -
        \dfrac{1}{c_{\text{P}}}} .
        \]
  \item गुणांक का मान निकालिए: S--P विलंब के प्रति सेकंड कितने
        किलोमीटर?
  \item कोई केंद्र $\Delta t = \qty{28}{s}$ नापता है: भूकंप कितनी दूर
        है?
  \item एक केंद्र दूरी देता है, स्थान नहीं: अधिकेंद्र निश्चित करने के
        लिए कितने केंद्र चाहिए, और ज्यामितीय रूप से किस तरह?
  \item जापान के पूर्व-चेतावनी भोंपू प्रबल कँपकँपी से कुछ सेकंड पहले
        बज उठते हैं: किसी नगर के \qty{80}{km} नीचे आए भूकंप के लिए
        अकेला P--S विलंब कितनी चेतावनी देता है?
  \item आधुनिक चेतावनियाँ उसमें प्रकाश की चाल जोड़ देती हैं: स्रोत के
        पास पकड़ी गई \omterm{prop:b3:continuum-elasticity:waves}{P तरंग} दोनों तरंगों से आगे रेडियो से भेज दी जाती
        है। अधिकेंद्र से \qty{200}{km} दूर के किसी नगर के लिए, विदर के
        कितने समय बाद \omterm{prop:b3:continuum-elasticity:waves}{S तरंग} पहुँचती है, और स्रोत से \qty{20}{km} दूर
        P के पहले आगमन पर भेजा गया रेडियो-संदेश कितनी चेतावनी दे सकता
        है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- ग्रह को पार करती तरंगें।}
\begin{enumerate}[resume]
  \item गहरे प्रावार के दाबों पर मापांक बढ़ जाते हैं: प्रावार के तल पर
        P चाल \qty{13.7}{km/s} तक पहुँच जाती है। मोटा औसत
        $c_{\text{P}} \approx \qty{10}{km/s}$ लेकर बताइए कि किसी
        \omterm{prop:b3:continuum-elasticity:waves}{P तरंग} को पृथ्वी को व्यास के अनुदिश पार करने में कितना समय लगता है।
  \item केंद्र विश्व के दूसरी ओर से आए भूकंप दर्ज करते हैं: इन आगमनों
        का होना भर ही गहरी पृथ्वी के विषय में क्या कहता है (क्या वह
        प्रत्यास्थ है? क्या वह पूरी पिघली है?)?
  \item अधिकेंद्रीय कोण $\Delta$ परिभाषित कीजिए (पृथ्वी के केंद्र पर
        स्रोत और केंद्र के बीच का कोण)। सीधी किरणों के लिए दिखाइए
        कि जीवा की लंबाई $2R_{\text{E}}\sin(\Delta/2)$ है।
  \item $\Delta = 60^\circ$ के लिए जीवा और उसका सीधी-किरण यात्रा-काल
        निकालिए, औसत $c_{\text{P}} \approx
        \qty{10}{km/s}$ पर।
  \item 1900 के आसपास ओल्डम ने देखा कि S तरंगें
        $\Delta \approx 103^\circ$ तक दर्ज होती हैं और फिर
        \emph{लुप्त} हो जाती हैं: उससे आगे कोई सीधी S नहीं। पृथ्वी के
        भीतर गहराई में किसी क्षेत्र का कौन-सा गुण S तरंगों को मार
        डालता है, और पदार्थ की किस अवस्था में वह गुण होता है?
  \item $103^\circ$ से परे P तरंगें अनुपस्थित नहीं, पर क्षीण,
        विलंबित और खिसकी हुई होती हैं ($\approx 142^\circ$ तक का एक
        ``छाया क्षेत्र''): कोई द्रव क्षेत्र P तरंगों को विलंबित और
        अपवर्तित क्यों करता है, पर रोकता क्यों नहीं?
  \item एक वाक्य में निष्कर्ष निकालिए कि पृथ्वी के केंद्र में क्या
        बैठा है।
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- एक पैमाने से क्रोड को तौलना।}
\begin{enumerate}[resume]
  \item स्रोत से चली कोई सीधी S किरण क्रोड को स्पर्श करती है यदि
        केंद्र से उसकी निकटतम दूरी क्रोड की त्रिज्या
        $R_{\text{c}}$ के बराबर हो। दिखाइए कि यह स्पर्शी किरण
        अधिकेंद्रीय कोण $\Delta^* =
        2\arccos(R_{\text{c}}/R_{\text{E}})$ पर पृष्ठ तक पहुँचती है।
  \item $\Delta^* = 103^\circ$ से $R_{\text{c}}$ निकालिए।
  \item आधुनिक मान \qty{3480}{km} है: अपनी त्रुटि निकालिए, और उसका
        \emph{चिह्न} समझाइए: वास्तविक किरणें पृष्ठ की ओर मुड़ जाती हैं,
        क्योंकि गहराई के साथ चाल बढ़ती है --- तर्क दीजिए कि सीधी
        किरणें क्रोड को बढ़ाकर आँकती हैं या घटाकर।
  \item 1936 में इंगे लेमान को छाया क्षेत्र के \emph{भीतर} क्षीण
        P आगमन मिले: उन्होंने निष्कर्ष निकाला कि द्रव क्रोड के भीतर
        क्या बैठा है?
  \item भीतरी क्रोड की त्रिज्या \qty{1220}{km} है और वह ठोस है:
        ऐसा तरंग-प्रेक्षण सुझाइए जो उसकी ठोसता की पुष्टि कर सके
        (वहाँ कौन-सी तरंग होती है जिसे बाहरी क्रोड नहीं होने देता?)।
  \item नामित परिणाम का सार लिखिए: चट्टान में दो प्रत्यास्थ तरंग-चालें
        ($5.8$ और \qty{3.3}{km/s}), $103^\circ$ से परे एक गायब तरंग,
        और एक पैमाना --- ये मिलकर $\approx \qty{4000}{km}$ त्रिज्या
        का द्रव क्रोड दे देते हैं --- जो आधुनिक \qty{3480}{km} के
        $15\%$ के भीतर है, और यह सब \qty{6000}{km} ठोस चट्टान से होकर
        नापा गया।
\end{enumerate}
\end{problem}
