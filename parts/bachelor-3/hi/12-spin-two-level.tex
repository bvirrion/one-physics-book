\chapter{चक्रण और द्वि-स्तरीय तंत्र}\label{ch:b3:spin-two-level}

1922 में श्टर्न और गेरलाख ने चाँदी के परमाणुओं का एक पुंज किसी बेढब
चुंबक से होकर भेजा, यह अपेक्षा करते हुए कि या तो कोई विक्षेपण नहीं होगा
या कोई संतत धब्बा बनेगा। पुंज साफ़-साफ़ \emph{दो} में बँट गया। कोई भी
कक्षकीय कोणीय संवेग ऐसा नहीं कर सकता: $2l + 1$ सदा विषम होता है। वे
परमाणु एक नई, विशुद्ध क्वांटम संपदा की घोषणा कर रहे थे ---
\emph{\omterm{def:b3:spin-two-level:spin}{चक्रण}}, अर्थात् $j = \tfrac12$ का कोई अंतर्जात कोणीय संवेग, वही
अर्ध-पूर्णांक पायदान जिसे \cref{ch:b3:quantum-angular-momentum} का
बीजगणित तैयार रखे था और जिसे कक्षाएँ कभी काम में नहीं ला सकती थीं।
\omterm{def:b3:spin-two-level:spin}{चक्रण} हर इलेक्ट्रॉन-अवस्था को दुगुना कर देता है (और आवर्त सारणी के
पीछे की गिनती पूरी कर देता है), लोहे तथा प्रोटॉन का चुंबकत्व ढोता है,
और प्रकृति का सबसे स्वच्छ द्वि-स्तरीय तंत्र है: इस अध्याय की भौतिकी
हर अस्पताल का \omterm{prop:b3:spin-two-level:rabi}{चुंबकीय अनुनाद}
क्रमवीक्षक चलाती है, वह सीज़ियम घड़ी भी जो सेकंड को परिभाषित करती है,
वह 21 सेंटीमीटर की फुसफुसाहट भी जिससे मंदाकिनियों का नक्शा बनता है, और क्यूबिट भी।

\section{वह प्रयोग जिसने उसे खोजा}

\begin{example}[श्टर्न--गेरलाख]\label{ex:b3:spin-two-level:stern-gerlach}
किसी \emph{असमांगी} क्षेत्र में कोई चुंबकीय आघूर्ण $\vect\mu$
बल $F_z = \mu_z\,\partial B_z/\partial z$ अनुभव करता है: अतः विक्षेपण
$\mu_z$ नाप देता है। चिरसम्मत अपेक्षा: आघूर्ण यादृच्छिक दिशाओं में,
और एक संतत पंखा। कक्षकीय संवेगों के लिए क्वांटम अपेक्षा: $2l + 1$
धब्बे: एक, तीन, पाँच --- सदा विषम। चाँदी के लिए (और
हाइड्रोजन के लिए) प्रेक्षित: \emph{दो} धब्बे, सममित, और बीच में कुछ
नहीं। नापा गया घटक ठीक दो ही मान लेता है --- $j
= \tfrac12$ की छाप, जो कक्षाओं के लिए वर्जित है, और क्वांटन का सीधा
प्रदर्शन: यंत्र किसी एक चक्रण-घटक का माप-उपकरण है,
और पुंज उसके दो अभिलक्षणिक मानों में बँट जाता है।
\end{example}

\begin{definition}[चक्रण एक-बटा-दो]\label{def:b3:spin-two-level:spin}
\index{चक्रण}\index{पाउली आव्यूह}\index{स्पिनर}
इलेक्ट्रॉन (और प्रोटॉन, न्यूट्रॉन तथा क्वार्क) $j = \tfrac12$ का कोई
अंतर्जात कोणीय संवेग ढोता है: अर्थात् एक द्विविमीय
अवस्था-समष्टि, जिसे $\ket\uparrow, \ket\downarrow$ फैलाते हैं
($\hat S_z$ की अभिलक्षणिक अवस्थाएँ, $\pm\hbar/2$ के साथ), और जिस पर
\[
\hat{\vect S} = \frac\hbar2\,\hat{\vect\sigma} , \qquad
\sigma_x = \begin{pmatrix}0&1\\1&0\end{pmatrix} ,\;
\sigma_y = \begin{pmatrix}0&-\iu\\\iu&0\end{pmatrix} ,\;
\sigma_z = \begin{pmatrix}1&0\\0&-1\end{pmatrix}
\]
--- अर्थात् \emph{पाउली आव्यूह}, जो कोणीय संवेग के बीजगणित को उसके
सबसे छोटे संभव घर में साकार करते हैं। कोई व्यापक अवस्था
$\alpha\ket\uparrow + \beta\ket\downarrow$ एक \emph{स्पिनर} है।
इलेक्ट्रॉन का चुंबकीय आघूर्ण है
\[
\hat{\vect\mu} = -g\,\frac{\mu_{\text{B}}}{\hbar}\,\hat{\vect S} ,
\qquad g \approx 2 :
\]
अर्थात् प्रति एकांक कोणीय संवेग कक्षकीय दर का दुगुना --- वही विषमता,
जिसे आइंस्टाइन--दे हास ने नापा था
(\cref{exo:b3:quantum-angular-momentum:12}) और जिसकी भविष्यवाणी बाद
में डिराक का समीकरण करेगा। चक्रण किसी वस्तु का घूमना नहीं है: न कोई
त्रिज्या, न कोई ``घूमती गेंद'' जाँच के आगे टिकती है --- वह अंतर्जात
है, आवेश की तरह।
\end{definition}

\begin{example}[शृंखलाबद्ध श्टर्न--गेरलाख छन्ने]\label{ex:b3:spin-two-level:chained}
$S_z = +\hbar/2$ वाला पुंज छाँटिए और $S_z$ फिर नापिए: सारे
परमाणु $+$ ही उत्तर देते हैं। इसके बदले $S_x$ नापिए: आधे-आधे --- अर्थात्
अवस्था $\ket\uparrow$ ही अध्यारोपण $(\ket{+x} + \ket{-x})/\sqrt2$ है।
अब $S_x = +$ वाला पुंज रखिए और $S_z$ एक बार और नापिए: \emph{फिर} आधे
और आधे --- $S_x$ की माप ने पहले से तीखे $S_z$ को मिटा दिया।
असंगति, संकुचन और \omterm{thm:b3:quantum-formalism:postulates}{बोर्न का नियम} दिखाने के लिए तीन चुंबक बहुत हैं; यह
शृंखला परमाणुओं से किया गया
\cref{ex:b3:quantum-formalism:threepolarizers} ही है।
\end{example}

\section{ब्लॉख गोला}

\begin{proposition}[किसी भी अक्ष के अनुदिश चक्रण]\label{prop:b3:spin-two-level:bloch}
\index{ब्लॉख गोला}
एकांक दिशा $\vect n = (\sin\theta\cos\varphi,
\sin\theta\sin\varphi, \cos\theta)$ के लिए संकारक $\vect
n\cdot\hat{\vect\sigma}$ के \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm1$ हैं, जहाँ
\[
\ket{+\vect n} = \cos\tfrac\theta2\,\ket\uparrow +
\eu^{\iu\varphi}\sin\tfrac\theta2\,\ket\downarrow .
\]
हर शुद्ध \omterm{def:b3:spin-two-level:spin}{स्पिनर} ठीक एक ही दिशा के लिए $\ket{+\vect n}$ है: अतः
अवस्था-समष्टि किसी गोले पर प्रतिचित्रित हो जाती है --- वही
\emph{\omterm{prop:b3:spin-two-level:bloch}{ब्लॉख गोला}} --- जिसमें लांबिक अवस्थाएँ \emph{प्रतिध्रुवों} पर
बैठती हैं (आधे-कोणों पर ध्यान दीजिए: विपरीत दिशाएँ लांबिक हैं, लंबवत
दिशाएँ नहीं)। इस गोले के घूर्णन ठीक वही विकास हैं जो कोई चुंबकीय क्षेत्र
उत्पन्न करता है: अर्थात् अब तक किए गए हर क्यूबिट-प्रचालन की ज्यामिति।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
$2\times2$ आव्यूह $\vect n\cdot\vect\sigma =
\begin{pmatrix}\cos\theta & \sin\theta\,\eu^{-\iu\varphi}\\
\sin\theta\,\eu^{\iu\varphi} & -\cos\theta\end{pmatrix}$ का विकर्णन
कीजिए: अनुरेख $0$,
सारणिक $-1$, \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm1$; और दिया हुआ \omterm{def:b3:spin-two-level:spin}{स्पिनर}
प्रतिस्थापन से जाँच लिया जाता है (आधे-कोण की सर्वसमिकाएँ)। प्रतिध्रुव:
$\theta \to \pi -
\theta$, $\varphi \to \varphi + \pi$ से $\braket{+\vect
n'}{+\vect n} = 0$ मिलता है।
\end{proof}

\section{क्षेत्र में चक्रण: पुरस्सरण और अनुनाद}

\begin{proposition}[लार्मर पुरस्सरण]\label{prop:b3:spin-two-level:larmor}
\index{लार्मर पुरस्सरण}\index{घूर्णचुंबकीय अनुपात}
किसी स्थैतिक क्षेत्र $\vect B_0 = B_0\vect e_z$ में \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} $\hat
H = -\hat{\vect\mu}\cdot\vect B_0$ दोनों स्तरों को
$\hbar\omega_0$ जितना विपाटित कर देता है और माध्य \omterm{def:b3:spin-two-level:spin}{चक्रण} को क्षेत्र के
परितः \emph{पुरस्सरण} कराता है:
\[
\frac{\dd\langle\hat{\vect S}\rangle}{\dd t} =
\gamma\,\langle\hat{\vect S}\rangle\wedge\vect B_0 , \qquad
\omega_0 = |\gamma|B_0 ,
\]
जहाँ \omterm{prop:b3:spin-two-level:larmor}{घूर्णचुंबकीय अनुपात} $\gamma$ है ($\mu = \gamma S$)। ब्लॉख
सदिश $z$ के परितः $\omega_0$ पर घूमता है, और क्षेत्र से उसका कोण जमा
रहता है --- ठीक वही चिरसम्मत घूर्णदर्शी वाला चित्र, जो यहाँ इसलिए
यथार्थ है कि \omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} रैखिक हैं। इलेक्ट्रॉन के लिए: $\qty{28}{GHz/T}$; और
प्रोटॉन के लिए (आघूर्ण $2.79$ नाभिकीय मैग्नेटॉन): $\gamma_p/2\pi =
\qty{42.6}{MHz/T}$ --- हर MRI मशीन का नंबर-प्लेट।
\end{proposition}

\begin{proof}
यहाँ \cref{prop:b3:quantum-formalism:evolution} को $\hat H =
-\gamma B_0\hat S_z$ के साथ लगाइए: \omterm{def:b3:quantum-formalism:commutator}{क्रमविनिमेयक} देते हैं $\dd\langle\hat
S_x\rangle/\dd t = -\gamma B_0\langle\hat S_y\rangle$,
$\dd\langle\hat S_y\rangle/\dd t = +\gamma B_0\langle\hat
S_x\rangle$ और $\dd\langle\hat S_z\rangle/\dd t = 0$, जो मिलकर
कथित सदिश गुणनफल बना देते हैं।
\end{proof}

\begin{proposition}[चुंबकीय अनुनाद]\label{prop:b3:spin-two-level:rabi}
\index{चुंबकीय अनुनाद}\index{राबी दोलन}
अब एक छोटा क्षेत्र $B_1$ जोड़िए, जो अनुप्रस्थ तल में आवृत्ति
$\omega$ से घूमता हो। उसके साथ घूमते तंत्र में $B_0$
प्रभावी रूप से घटकर $(\omega_0 - \omega)/|\gamma|$ रह जाता है;
\emph{अनुनाद पर} ($\omega = \omega_0$) केवल $B_1$ बचता है, और \omterm{def:b3:spin-two-level:spin}{चक्रण}
उसी के परितः \emph{राबी आवृत्ति} $\Omega_1 =
|\gamma|B_1$ पर पुरस्सरण करता है: $\ket\uparrow$ से आरंभ करने पर
पलट जाने की प्रायिकता है
\[
\mathcal P_\downarrow(t) = \sin^2\Big(\frac{\Omega_1t}{2}\Big) .
\]
$\pi/\Omega_1$ अवधि की कोई स्पंद (कोई ``$\pi$ स्पंद'')
\omterm{def:b3:spin-two-level:spin}{चक्रण} को उलट देती है; और उसकी आधी (कोई ``$\pi/2$ स्पंद'') उसे
विषुवत् तल में लिटा देती है, जहाँ वह $\omega_0$ पर पुरस्सरण करता है और
फ़ैराडे प्रेरण से अपनी आवृत्ति किसी भी पास की कुंडली में
\emph{प्रसारित} कर देता है। अनुनाद से हटने पर
फ़्लॉपिंग का आयाम $\Omega_1^2/[\Omega_1^2 +
(\omega - \omega_0)^2]$ की तरह गिर जाता है: अतः अनुक्रिया तीखे ढंग से
चयनी है --- एक अनुनाद --- और यही चक्रणों को संबोधनीय बनाता है,
जाति-दर-जाति और, किसी क्षेत्र-प्रवणता के साथ, स्थान-दर-स्थान
(\cref{pb:b3:spin-two-level:1})।
\end{proposition}
\admitted

\begin{example}[अतिसूक्ष्म संरचना और 21 cm रेखा]\label{ex:b3:spin-two-level:hyperfine}
\index{अतिसूक्ष्म संरचना}\index{21 cm रेखा}
हाइड्रोजन की मूल अवस्था में इलेक्ट्रॉन और प्रोटॉन के \omterm{def:b3:spin-two-level:spin}{चक्रण}
अपने चुंबकीय आघूर्णों से अन्योन्यक्रिया करते हैं: चारों चक्रण-अवस्थाएँ
एक त्रिक और एक एकक में बँट जाती हैं, जिनके बीच केवल $\Delta E =
\qty{5.9}{\micro eV}$ का अंतर है --- वही \emph{अतिसूक्ष्म} विपाटन, $\nu =
\qty{1420}{MHz}$, $\lambda = \qty{21}{cm}$। यह संक्रमण
हास्यास्पद रूप से धीमा है (प्रति एक करोड़ वर्ष में एक पलटाव), पर मंदाकिनी
में $10^{66}$ \omterm{thm:b3:hydrogen-atom:levels}{हाइड्रोजन परमाणु} हैं: अतः \omterm{ex:b3:spin-two-level:hyperfine}{21 cm रेखा} इतनी चमकीली है कि
उसने सर्पिल भुजाएँ, चकती का मोड़, और --- अपने डॉप्लर खिसकावों से ---
वे सपाट घूर्णन-वक्र भी नक्शे पर ला दिए जो श्याम पदार्थ का तर्क देते
हैं। वही भौतिकी सीज़ियम की मूल अवस्था में स्तरों को
ठीक $\qty{9192631770}{Hz}$ जितना विपाटित करती है: और 1967 से
\emph{सेकंड की परिभाषा} गिनकर पूरा किया गया कोई अतिसूक्ष्म चक्रण-पलटाव
ही है (\cref{exo:b3:spin-two-level:12})।
\end{example}

\begin{method}[द्वि-स्तरीय कारीगरी]\label{met:b3:spin-two-level:method}
(1) किसी भी द्वि-स्तरीय \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{हैमिल्टोनियन} को पाउली आव्यूहों में लिखिए: $\hat H =
\epsilon_0\mathbb 1 + \vect h\cdot\hat{\vect\sigma}$ --- तब ब्लॉख
सदिश $\vect h$ के परितः $2|\vect h|/\hbar$ पर पुरस्सरण करता है;
बाकी सब ज्यामिति है। (2) $\vect n$ के अनुदिश अभिलक्षणिक अवस्थाएँ:
आधे-कोण काम में लाइए। (3) अनुनाद की समस्याएँ: घूमते तंत्र में जाइए;
अनुनाद पर केवल $B_1$ बचता है। (4) स्पंद: पुरस्सरण का संकेत आरंभ करने
के लिए $\pi/2$, उलटने के लिए $\pi$ (और फिर से केंद्रित करने के लिए भी
--- वही चक्रण-प्रतिध्वनि)। (5) ऊर्जा-पैमाने: इलेक्ट्रॉनों के लिए
$\mu_{\text{B}}B$ ($\qty{58}{\micro eV/T}$), और नाभिकों के लिए उससे
तीन कोटि कम --- अर्थात् रेडियो आवृत्तियाँ, केल्विन-मुक्त वर्णक्रममिति।
\end{method}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:11}
दो \omterm{def:b3:spin-two-level:spin}{चक्रण} साथ-साथ। चक्रण-$\tfrac12$ के दो कणों के लिए कुल
\omterm{def:b3:spin-two-level:spin}{चक्रण} $\hat{\vect S} = \hat{\vect S}_1 + \hat{\vect S}_2$ है। (क) दिखाइए
कि चारों गुणनफल-अवस्थाएँ फिर से एक \emph{त्रिक} ($S = 1$:
$\ket{\uparrow\uparrow}$, $(\ket{\uparrow\downarrow} +
\ket{\downarrow\uparrow})/\sqrt2$, $\ket{\downarrow\downarrow}$)
और एक \emph{एकक} ($S = 0$: $(\ket{\uparrow\downarrow} -
\ket{\downarrow\uparrow})/\sqrt2$) में बँट जाती हैं --- $S_z$ की
गिनतियाँ जाँचिए और, बीच की दोनों अवस्थाओं के लिए, $\hat S^2$ का प्रभाव
भी ($\hat
S^2 = \hat S_1^2 + \hat S_2^2 + 2\hat{\vect S}_1\cdot\hat{\vect
S}_2$ और $\hat{\vect S}_1\cdot\hat{\vect S}_2 = \tfrac12(\hat
S_{1+}\hat S_{2-} + \hat S_{1-}\hat S_{2+}) + \hat S_{1z}\hat
S_{2z}$ काम में लाइए)। (ख) चारों में से कौन-सी उलझी हुई है ---
अर्थात् किसी गुणनफल के रूप में नहीं लिखी जा सकती? (ग) हाइड्रोजन का
अतिसूक्ष्म युग्म (\cref{ex:b3:spin-two-level:hyperfine}) ठीक यही
त्रिक/एकक है: कौन ऊर्जा में ऊपर है, यह देखते हुए कि रेखा
21 cm पर \emph{उत्सर्जित} होती है? (घ) एकक के \omterm{def:b3:spin-two-level:spin}{चक्रण} \emph{हर}
अक्ष के अनुदिश पूरी तरह प्रति-सहसंबंधित हैं: किसी एक को किसी भी
$\vect n$ के अनुदिश नापिए और दूसरा उलटा उत्तर देता है। यह
सहसंबंध, चाहे कितना ही चौंकाने वाला हो, कोई संकेत क्यों नहीं भेजता?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:12}
वह घड़ी जो सेकंड परिभाषित करती है। सीज़ियम का मूल-अवस्था अतिसूक्ष्म
विपाटन ठीक $\qty{9192631770}{Hz}$ है (सेकंड की परिभाषा से)।
किसी फ़व्वारा-घड़ी में परमाणुओं को एक $\pi/2$ स्पंद मिलती है, वे
$T = \qty{0.5}{s}$ तक स्वतंत्र उड़ते हैं, और उन्हें दूसरी $\pi/2$ स्पंद
मिलती है; अंतरित अंश $\cos^2(\pi\,\delta\,T)$ की तरह दोलन करता है,
जहाँ विचलन $\delta$ है (रैमज़ी झालर --- इसे स्वीकार कीजिए)। (क) ब्लॉख
की भाषा में समझाइए कि हर $\pi/2$ स्पंद और मुक्त उड़ान क्या करती है। (ख)
केंद्रीय झालर की चौड़ाई। (ग) यदि संकेत-से-कोलाहल अनुपात झालर के केंद्र
को $10^4$ से बाँटने दे, तो कितनी आंशिक आवृत्ति-यथार्थता मिलती है? (घ)
नियत झालर-विभाजन क्षमता पर प्रकाशिक घड़ियाँ (पेटाहर्ट्ज़ संक्रमण)
सूक्ष्मतरंग घड़ियों को क्यों हरा देती हैं --- और मोटे तौर पर कितने गुना?
\end{exercise}

\section{समस्या: शरीर के भीतर देखना}

\begin{problem}[{सप्ताहांत समस्या --- चुंबकीय अनुनाद प्रतिबिंबन,
चक्रण से क्रमवीक्षण तक}]\label{pb:b3:spin-two-level:1}
किसी मनुष्य का दो तिहाई जल है; और हर जल-अणु दो प्रोटॉन ढोता है, जिनमें
से हर एक चक्रण-$\tfrac12$ का चुंबक है। किसी व्यक्ति को प्रबल
क्षेत्र में रखिए, प्रोटॉनों को उनकी लार्मर आवृत्ति पर गुदगुदाइए, और
सुनिए: यही \omterm{prop:b3:spin-two-level:rabi}{चुंबकीय अनुनाद} प्रतिबिंबन है, अर्थात् चक्रण-भौतिकी, चिकित्सा
के रूप में। आँकड़े:
$\gamma_{\text{p}}/2\pi = \qty{42.58}{MHz/T}$; $B_0 = \qty{3.0}{T}$;
ऊतक का प्रोटॉन-घनत्व $n \approx \qty{6.6e28}{m^{-3}}$;
\qty{310}{K} पर $k_{\text{B}}T$ \qty{26.7}{meV} है; $h =
\qty{4.14e-15}{eV.s}$।

\medskip\noindent\textbf{भाग I --- चुंबक में \omterm{def:b3:spin-two-level:spin}{चक्रण}।}
\begin{enumerate}
  \item \qty{3.0}{T} पर लार्मर आवृत्ति और फ़ोटॉन की ऊर्जा
        $h\nu$ eV में निकालिए।
  \item प्रोटॉन के दोनों स्तरों के बीच जनसंख्या-असंतुलन निकालिए,
        $\Delta n/n \approx h\nu/2k_{\text{B}}T$।
  \item केवल यही अतिरेक --- दस लाख में कुछ भाग --- संकेत देता है:
        ऊतक के प्रति घन सेंटीमीटर कितने ``उपयोगी'' प्रोटॉन हैं?
  \item तापीय ध्रुवण यहाँ इतना दुर्बल क्यों है, जबकि
        श्टर्न--गेरलाख का पुंज पूरा बँट गया था? (हर प्रयोग क्या नापता
        है: एक-परमाणु \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान}, या कोई तापीय
        औसत?)
  \item $B_0$ दुगुना करने से संकेत का क्या होता है (दो प्रभाव:
        ध्रुवण, और ऊँची आवृत्ति पर प्रेरित विद्युत्वाहक बल)? अस्पताल
        बड़े चुंबकों के लिए पैसे क्यों चुकाते हैं?
  \item चुंबक अतिचालक है और सदा चालू रहता है: कमरे में पड़ी कोई
        ढीली इस्पात की ऑक्सीजन-बोतल घातक प्रक्षेप्य क्यों बन जाती है
        (\cref{ex:b3:spin-two-level:stern-gerlach} का कौन-सा बल
        उस पर लगता है)?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- स्पंद और प्रतिध्वनियाँ।}
\begin{enumerate}[resume]
  \item कोई अनुप्रस्थ कुंडली अनुनाद पर $B_1 = \qty{10}{\micro T}$
        लगाती है: राबी आवृत्ति और
        $\pi/2$ तथा $\pi$ स्पंदों की अवधियाँ निकालिए।
  \item $\pi/2$ स्पंद के बाद चुंबकन क्या करता है, और फ़ैराडे का नियम
        ग्राही कुंडली में क्या प्रेरित करता है (किस
        आवृत्ति पर)?
  \item अनुप्रस्थ संकेत काल-नियतांक $T_2$ से क्षीण होता है
        (\omterm{def:b3:spin-two-level:spin}{चक्रण} एक-दूसरे के क्षेत्रों और स्थानिक
        असमांगताओं में विकलित हो जाते हैं); जबकि अनुदैर्घ्य चुंबकन
        $T_1$ के साथ फिर से बढ़ता है (ऊर्जा ऊतक में बहती है)।
        प्रारूपिक मान: $T_1 \approx \qty{1}{s}$, $T_2 \approx \qty{0.1}{s}$।
        संसक्ति-काल ऊर्जा-शिथिलन काल से बहुत अधिक क्यों नहीं हो
        सकता (हर ऊर्जा-विनिमयी पलटाव कला के साथ क्या करता है)?
  \item चक्रण-प्रतिध्वनि: $\pi/2$ स्पंद और $\tau$ के विलंब के बाद
        एक $\pi$ स्पंद लगाई जाती है और $2\tau$ पर विकलित
        \omterm{def:b3:spin-two-level:spin}{चक्रण} फिर पंक्तिबद्ध हो जाते हैं और संकेत लौट आता है। इस
        युक्ति को किसी दौड़-पथ के उन धावकों से समझाइए जिन्हें मुड़ने
        को कहा जाता है।
  \item यह प्रतिध्वनि कौन-सा क्षय पलट देती है --- \emph{स्थैतिक}
        क्षेत्र-असमांगताओं से हुआ विकलन, या उतार-चढ़ाव करते आणविक
        क्षेत्रों से हुआ --- और केवल वही क्यों?
  \item भिन्न ऊतकों के $T_1$, $T_2$ भिन्न होते हैं (चर्बी:
        छोटा $T_1$; जल/तरल: लंबा; अनेक अर्बुद: अपने आश्रयी ऊतक से
        लंबा $T_2$)। एक वाक्य में: स्पंद-अनुक्रम का समय साधने से
        शिथिलन-काल चित्र के \emph{विपर्यास} में कैसे बदल जाते हैं?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- संकेत से चित्र तक।}
\begin{enumerate}[resume]
  \item कोई प्रवणता $G = \qty{40}{mT/m}$ अध्यारोपित कीजिए, $z$ के
        अनुदिश: तब लार्मर आवृत्ति स्थान-निर्भर हो जाती है।
        $\dd\nu/\dd z$ को kHz प्रति मिलीमीटर में निकालिए।
  \item केवल बैंड $\nu_0 \pm
        \qty{1}{kHz}$ रखने वाली कोई गढ़ी हुई
        रेडियो-आवृत्ति स्पंद शरीर की कौन-सी परत उत्तेजित करती है?
        (परत-चयन।)
  \item पाठ के दौरान $x$ के अनुदिश कोई प्रवणता उस परत के हर स्तंभ से
        उसकी अपनी आवृत्ति प्रसारित कराती है: ग्रहण किए गए काल-संकेत
        को किसी स्थानिक रूपरेखा में कौन-सी गणितीय संक्रिया बदल देती
        है (वर्ष 2 के खंड की फूरिये औज़ार-पेटी स्मरण कीजिए)?
  \item पूरे क्रमवीक्षण की सूचना आँकिए: 12 बिट पर कोई
        $256 \times 256$ चित्र, और यह भी कि उसे पंक्ति-दर-पंक्ति
        (प्रति पंक्ति एक प्रतिध्वनि, पुनरावृत्ति-काल $\sim T_1$)
        बटोरने से क्रमवीक्षण में मिनट क्यों लग जाते हैं।
  \item रोगी को ज़ोर की ठक-ठक सुनाई देती है: यांत्रिक रूप से क्या
        पीट रहा है (सोचिए कि \qty{3}{T} के क्षेत्र में प्रवणता-कुंडलियाँ
        चालू-बंद हो रही हैं --- कौन-सा बल)?
  \item एक्स-किरण प्रतिबिंबन इलेक्ट्रॉन-घनत्व का विपर्यास दिखाता है;
        जबकि MRI प्रोटॉन-घनत्व \emph{और} शिथिलन का: MRI मृदु ऊतक और
        मस्तिष्क के लिए पसंदीदा औज़ार क्यों है, और वह कितनी आयनकारी
        मात्रा देता है?
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- \omterm{def:b3:spin-two-level:spin}{चक्रण} के दूसरे काम।}
\begin{enumerate}[resume]
  \item रसायनज्ञ वही प्रयोग \qty{10}{ppm} विभेदन पर चलाते हैं:
        इलेक्ट्रॉन-बादल हर नाभिक को थोड़ा परिरक्षित कर देते हैं, और
        उसका अनुनाद खिसका देते हैं (वही ``रासायनिक खिसकाव'')। इससे
        NMR किसी आणविक छाप में क्यों बदल जाती है?
  \item क्रियात्मक MRI सोच का नक्शा बनाती है: विऑक्सीजनित हीमोग्लोबिन
        अनुचुंबकीय है और स्थानिक $T_2^*$ छोटा कर देता है। तंत्रिकीय
        सक्रियता से चित्र की चमक तक की शृंखला बताइए।
  \item वही लार्मर भौतिकी \emph{इलेक्ट्रॉन} के आघूर्ण के साथ
        \qty{3}{T} पर \qty{84}{GHz} पर चलती है: इलेक्ट्रॉन-अनुनाद
        किसी मनुष्य के भीतर बेकार क्यों है, पर मुक्त मूलकों और
        दोषों के अध्ययन के लिए बहुमूल्य?
  \item मद 1 की फ़ोटॉन-ऊर्जा की तुलना प्रारूपिक रासायनिक
        बंध-ऊर्जाओं से कीजिए: ``अनायनकारी'' शब्द का औचित्य दीजिए और
        \qty{60}{keV} की किसी एक्स-किरण फ़ोटॉन से अंतर बताइए।
  \item दस भाग-प्रति-दस-लाख का ध्रुवण बढ़ाया जा सकता है: लेज़र-ध्रुवित
        ज़ेनॉन गैस कोटि-एक ध्रुवण तक पहुँचती है और उसे फेफड़ों की
        वायु-जगह का चित्र लेने के लिए साँस में लिया जाता है। प्रति
        नाभिक संकेत मोटे तौर पर कितने गुना बढ़ता है, और गैस का नीचा
        घनत्व यह युक्ति फिर भी क्यों आवश्यक बना देता है?
  \item सिलिकॉन के किसी ट्रांज़िस्टर-जैसे पाश में कोई अकेला
        इलेक्ट्रॉन-चक्रण अब ठीक इसी अध्याय की स्पंदों से पढ़ा और चलाया
        जाता है: \omterm{def:b3:spin-two-level:spin}{चक्रण} के वे दो गुण बताइए (उसकी
        \omterm{def:b3:quantum-formalism:state}{हिल्बर्ट समष्टि} का आकार; कोलाहल से उसके युग्मन की दुर्बलता)
        जो उसकी सिफ़ारिश करते हैं।
  \item नामित परिणाम का सार लिखिए: \qty{128}{MHz} का कोई
        चक्रण-पुरस्सरण, दस-भाग-प्रति-दस-लाख का तापीय ध्रुवण,
        दो शिथिलन-घड़ियाँ और तीन क्षेत्र-प्रवणताएँ किसी जीवित मस्तिष्क
        का परत-दर-परत चित्र लेने के लिए बहुत हैं --- श्टर्न और
        गेरलाख के दो धब्बे, जो एक अस्पताल बन गए।
\end{enumerate}
\end{problem}
