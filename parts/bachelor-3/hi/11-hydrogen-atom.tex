\chapter{हाइड्रोजन परमाणु}\label{ch:b3:hydrogen-atom}

ब्रह्मांड के दस में से नौ परमाणु हाइड्रोजन हैं: एक प्रोटॉन, जो एक
इलेक्ट्रॉन थामे है, और यही अकेला परमाणु है जिसे भौतिकी ठीक-ठीक हल कर
सकती है --- तथा वही कुंजी है जिसने बाकी सब खोले। \cref{ch:b3:hamiltonian-mechanics}
की पुरानी क्वांटम सिद्धांत ने उसकी ऊर्जाओं का अनुमान लगाया था; अब वह
मशीनरी तैयार है जिससे उन्हें \emph{निकाला} जा सके: कूलॉम विभव पिछले
अध्याय के \omterm{prop:b3:schrodinger-three-dimensions:swave}{त्रिज्य समीकरण} में जाता है, और बाहर
निकलते हैं स्तर $-E_{\text{I}}/n^2$, \omterm{thm:b3:hydrogen-atom:levels}{बोर त्रिज्या}, रसायन की आवर्त
सारणी के कोश और उपकोश, तथा वे वर्णक्रमीय श्रेणियाँ जो खगोलविद हर
नीहारिका में पढ़ते हैं। वही हल, फिर से मापित करके,
आयनित हीलियम, म्युऑनी परमाणुओं, पॉज़िट्रोनियम, और अर्धचालकों के भीतर
इलेक्ट्रॉन--विवर ``परमाणुओं'' का भी वर्णन करता है; और $n
\approx 100$ तक खींचने पर वह आधे माइक्रोमीटर चौड़े रिडबर्ग परमाणुओं
का वर्णन करता है, जिनकी रेडियो-फुसफुसाहटें मंदाकिनी के आयनित बादलों का
नक्शा बनाती हैं और जिनकी अन्योन्यक्रियाएँ अब क्वांटम संगणक चलाती हैं।

\section{कूलॉम समस्या, हल की हुई}

\begin{theorem}[हाइड्रोजन के स्तर]\label{thm:b3:hydrogen-atom:levels}
\index{हाइड्रोजन परमाणु}\index{मुख्य क्वांटम संख्या}\index{बोर त्रिज्या}
$V(r) = -e^2/4\pi\varepsilon_0 r$ के लिए ($k = e^2/4\pi
\varepsilon_0$ लिखिए) बद्ध अवस्थाओं के नाम $(n, l, m)$ हैं, जहाँ
\[
E_n = -\frac{E_{\text{I}}}{n^2} , \qquad
E_{\text{I}} = \frac{m_{\text{e}}k^2}{2\hbar^2} = \qty{13.6}{eV} ,
\qquad n = 1, 2, 3, \dots
\]
और हर $n$ के लिए कक्षकीय संख्याएँ $l = 0, 1, \dots, n-1$ तथा
$m = -l, \dots, l$। स्वाभाविक लंबाई \omterm{thm:b3:hydrogen-atom:levels}{बोर त्रिज्या} $a_0 =
\hbar^2/m_{\text{e}}k = \qty{52.9}{pm}$ है; और मूल अवस्था है
\[
\psi_{100} = \frac{1}{\sqrt{\pi a_0^3}}\;\eu^{-r/a_0} .
\]
ऊर्जा \emph{केवल} $n$ पर निर्भर करती है: $m$ और
$l$ गिनने पर स्तर $n$ $n^2$ गुना अपह्रासी है (चक्रण से दुगुना,
\cref{ch:b3:spin-two-level}) --- अर्थात् उस $(2l+1)$ गुनी
अपह्रासिता से कहीं आगे जिसे समदैशिकता समझाती है।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
मूल अवस्था के लिए \omterm{prop:b3:schrodinger-three-dimensions:swave}{त्रिज्य समीकरण} में $u = Cr\,\eu^{-r/a}$ रखकर
देखिए, $l = 0$ के साथ: $u'' = (1/a^2 - 2/ar)u$, अतः
$-\tfrac{\hbar^2}{2m_{\text{e}}}u'' - \tfrac kr u = Eu$ के लिए
$\hbar^2/m_{\text{e}}a = k$ चाहिए ($1/r$ वाले पदों का मिलान) --- अर्थात्
$a = a_0$ --- और $E = -\hbar^2/2m_{\text{e}}a_0^2 = -E_{\text{I}}$।
व्यापक हल (लागेर बहुपद, $n - l - 1$ त्रिज्य
निस्पंद) स्वीकृत है; उस रचना का हर कदम प्रारंभिक है, पर लंबा।
``आकस्मिक'' $l$-अपह्रासिता शुद्ध $1/r$ बल के एक चिरसम्मत रहस्य का
दर्पण है --- केप्लर के दीर्घवृत्त पुरस्सरण नहीं करते, और एक अतिरिक्त
संरक्षित सदिश (लाप्लास--रुंगे--लेंज़) नियत दीर्घ अक्ष के अनुदिश संकेत
करता है; उसी का क्वांटम रूप भिन्न $l$ को एक ही ऊर्जा से बाँधता है
(\cref{exo:b3:hydrogen-atom:12})।
\end{proof}

\section{इलेक्ट्रॉन कहाँ है}

\begin{proposition}[त्रिज्य बंटन]\label{prop:b3:hydrogen-atom:radial}
\index{त्रिज्य प्रायिकता}
इलेक्ट्रॉन को $r$ और $r + \dd r$ के बीच पाने की प्रायिकता
$P(r)\,\dd r$ है, जहाँ $P(r) = |u(r)|^2$, अर्थात्
\cref{thm:b3:quantum-angular-momentum:radial} के प्रसामान्यित त्रिज्य फलन का वर्ग। सबसे निचली अवस्थाओं के लिए:
$P_{1s} \propto r^2\eu^{-2r/a_0}$, जिसका शिखर ठीक $r = a_0$ पर है और
$\langle r\rangle = \tfrac32 a_0$; $P_{2s}$ दो कूबड़ दिखाता है,
जिनके बीच एक गोलीय \emph{निस्पंद} है; और $P_{2p} \propto
r^4\eu^{-r/a_0}$, जिसका शिखर $4a_0$ पर है। व्यापक रूप से $\langle r\rangle
\approx n^2a_0$: अर्थात् उत्तेजन के साथ परमाणु \emph{द्विघाती} रूप से
बढ़ते हैं, जबकि उनका बंधन $1/n^2$ की तरह सिकुड़ता है --- यही वह उत्तोलन
है जो \cref{pb:b3:hydrogen-atom:1} के रिडबर्ग दानवों के पीछे है।
\end{proposition}
\admitted

\begin{example}[आकार पढ़ना]\label{ex:b3:hydrogen-atom:sizes}
मूल अवस्था: \qty{0.1}{nm} चौड़ी, \qty{13.6}{eV} गहरी --- अर्थात्
पूरी रसायन-विद्या के पैमाने। $n = 10$ पर: त्रिज्या $\sim\qty{5}{nm}$,
बंधन \qty{0.136}{eV} --- ढीली पकड़। $n = 100$ पर: कोई
\emph{माइक्रोमीटर} पैमाने का परमाणु, जो \qty{1.4}{meV} से बँधा है और
जिसे सबसे दुर्बल क्षेत्र भी तोड़ दे --- फिर भी अंतरतारकीय दिक् इतना
ख़ाली है कि ऐसे परमाणु जी लेते हैं और प्रसारण भी करते हैं
(\cref{pb:b3:hydrogen-atom:1})।
\end{example}

\section{वर्णक्रम}

\begin{proposition}[श्रेणियाँ और चयन-नियम]\label{prop:b3:hydrogen-atom:series}
\index{बामर श्रेणी}\index{लाइमन श्रेणी}
कोई छलाँग $n' \to n$ यह तरंगदैर्घ्य उत्सर्जित करती है
\[
\frac{1}{\lambda} = R_{\text{H}}\Big(\frac{1}{n^2} -
\frac{1}{n'^2}\Big) , \qquad
R_{\text{H}} = \frac{E_{\text{I}}}{hc} = \qty{1.097e7}{m^{-1}} ,
\]
और उस पर द्विध्रुव चयन-नियम $\Delta l = \pm1$ लगा है (फ़ोटॉन
$\hbar$ ढोता है)। \omterm{prop:b3:hydrogen-atom:series}{लाइमन श्रेणी} ($\to n = 1$) \qty{121.6}{nm} से
दूर पराबैंगनी में पड़ती है; \omterm{prop:b3:hydrogen-atom:series}{बामर श्रेणी} ($\to n = 2$)
लाल $H\alpha$ रेखा, \qty{656.3}{nm}, से आरंभ होती है --- वही रंग जो
उत्सर्जन-नीहारिकाओं और सौर उभारों का है; पाशन तथा बाद की श्रेणियाँ
अवरक्त में हट जाती हैं, और $n = 110$ तथा $109$ के बीच वही
सूत्र \qty{5.0}{GHz} पर उतरता है: हाइड्रोजन
पराबैंगनी से लेकर रेडियो डायल तक बोलता है।
\end{proposition}

\begin{proof}
\cref{thm:b3:hydrogen-atom:levels} के साथ ऊर्जा-संरक्षण;
और $R_{\text{H}}$ वैसा ही जैसा \cref{pb:b3:hamiltonian-mechanics:1} में था, पर
अब अभिगृहीत के बजाय निकाला हुआ।
\end{proof}

\begin{example}[हाइड्रोजन-सदृश परमाणु: एक हल, अनेक परमाणु]\label{ex:b3:hydrogen-atom:scaling}
प्रोटॉन के आवेश के स्थान पर $Ze$ रखिए और इलेक्ट्रॉन के स्थान पर कोई भी
परिक्रमा करता द्रव्यमान $\mu$: तब हर सूत्र इस तरह फिर से मापित होता है
\[
a \to a_0\,\frac{m_{\text{e}}}{\mu}\,\frac1Z , \qquad
E_{\text{I}} \to E_{\text{I}}\,\frac{\mu}{m_{\text{e}}}\,Z^2 .
\]
He$^+$ ($Z = 2$): \qty{54.4}{eV}। भारी परमाणुओं के भीतरी इलेक्ट्रॉन
($Z \sim 30$): K-कोश की ऊर्जाएँ \unit{keV} में --- वही अभिलक्षणिक
एक्स-किरणें, जिनसे मोज़ली ने तत्वों को क्रम में रखा
(\cref{exo:b3:hydrogen-atom:5})। म्युऑनी हाइड्रोजन ($\mu = 207
m_{\text{e}}$): फ़र्मीमीटर-पैमाने का परमाणु, जो स्वयं प्रोटॉन की जाँच
करता है। पॉज़िट्रोनियम ($e^+e^-$, $\mu = m_{\text{e}}/2$): आधा बंधन,
दुगुना आकार, और छोटा जीवन, जो विनाशन-फ़ोटॉनों में समाप्त होता है।
और किसी अर्धचालक में एक इलेक्ट्रॉन तथा एक विवर, अपने छोटे प्रभावी
द्रव्यमानों के साथ किसी परिरक्षक परावैद्युत में एक-दूसरे की परिक्रमा
करते हुए: अर्थात् कोई \emph{उत्तेजक}, वही परमाणु दस नैनोमीटर और
मिलीइलेक्ट्रॉनवोल्ट तक बढ़ा हुआ (\cref{exo:b3:hydrogen-atom:7}) ---
\omterm{thm:b3:hydrogen-atom:levels}{हाइड्रोजन परमाणु} कम, ढाँचा अधिक है।
\end{example}

\begin{method}[हाइड्रोजन-सदृश तंत्रों के साथ काम करना]\label{met:b3:hydrogen-atom:method}
(1) पहले मापन: $a \propto 1/\mu Z$, $E \propto \mu Z^2$ किसी भी
हाइड्रोजन-उत्तर को किसी भी हाइड्रोजन-सदृश उत्तर में बदल देते हैं।
(2) आकार: $\langle
r\rangle \sim n^2a$; स्तर $n$ के पास अंतराल: $\dd E/\dd n =
2E_{\text{I}}/n^3$। (3) वर्णक्रम: रिडबर्ग सूत्र और $\Delta l =
\pm1$। (4) प्रवेशन: $s$ अवस्थाएँ नाभिक को अनुभव करती हैं, ऊँचे $l$
वाली अवस्थाएँ बाहर परिक्रमा करती हैं --- यही बहु-इलेक्ट्रॉन रसायन का
उत्तोलक है। (5) विवेक के लंगर: $\qty{13.6}{eV}$, $\qty{52.9}{pm}$,
$\qty{121.6}{nm}$, $\qty{656.3}{nm}$ --- जीवन भर के लिए याद रखने
योग्य चार संख्याएँ।
\end{method}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:7}
उत्तेजक। गैलियम आर्सेनाइड में $\varepsilon_{\text{r}} = 12.9$ है और

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:8}
प्रवेशन और क्षारीय धातुएँ। सोडियम एक हाइड्रोजन-सदृश
संयोजी इलेक्ट्रॉन है, जो प्रभावी आवेश $+e$ के किसी बंद क्रोड के बाहर है।
(क) $3s$, $3p$, $3d$ में से कौन अपूर्ण परिरक्षित नाभिक को
सबसे अधिक अनुभव करता है, और क्यों (त्रिज्य चित्र स्मरण कीजिए)? (ख) मापे गए
स्तर $E_{3s} = \qty{-5.14}{eV}$, $E_{3p} = \qty{-3.04}{eV},
E_{3d} = \qty{-1.52}{eV}$ हैं: जाँचिए कि $3d$ लगभग हाइड्रोजनी है

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hydrogen-atom:9}
किसी नीहारिका में कोई प्रोटॉन किसी इलेक्ट्रॉन को $n = 100$ में
पकड़ लेता है। (क) यदि इलेक्ट्रॉन लगभग मुक्त आया हो, तो पकड़ के फ़ोटॉन
में कितनी ऊर्जा मुक्त होती है? (ख) परमाणु नीचे झरता है, अधिमानतः
$\Delta n = 1$ पगों से, ऊँचे $n$ पर: वे फ़ोटॉन किस बैंड में
पड़ते हैं? (ग) झरने का अंतिम पग लाइमन $\alpha$ है; और नीहारिकाओं का
दृश्य प्रकाश H$\alpha$ से प्रधान रहता है: बताइए कि वह झरने का
कौन-सा पग है। (घ) समझाइए कि उत्सर्जन-नीहारिकाएँ \emph{लाल} क्यों
चमकती हैं, यद्यपि हाइड्रोजन की सबसे प्रबल रेखा (लाइमन $\alpha$)
पराबैंगनी है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:10}
औसत और विरियल। मूल अवस्था के लिए निकालिए: (क)
$\langle 1/r\rangle$, और जाँचिए कि $\langle E_p\rangle = -2E_{\text{I}}
= 2E_1$; (ख) $\langle E_k\rangle = +E_{\text{I}}$, जिससे
\cref{exo:b3:hamiltonian-mechanics:4} का विरियल अनुपात जाँचा जाए; (ग)
इलेक्ट्रॉन की वर्ग-माध्य-मूल चाल और $v/c$; (घ) उस चुंबकीय क्षेत्र की परिमाण-कोटि, जिसे
इलेक्ट्रॉन की गति से \emph{प्रोटॉन} अनुभव करता है (कोई धारा-पाश
$ev/2\pi a_0$, दूरी $a_0$ से देखा हुआ) --- वही संख्या, जो अतिसूक्ष्म
संरचना को चाहिए होगी।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:11}
सूक्ष्म संरचना कितनी सूक्ष्म है। (क) दिखाइए कि मूल अवस्था का चाल-पैमाना
$\langle v^2\rangle^{1/2}/c = \alpha \approx 1/137$ है, और
हाइड्रोजन-सदृश $Z$ के लिए: $Z\alpha/n$। (ख) आपेक्षिकीय संशोधन
आपेक्षिक कोटि $(v/c)^2$ पर आते हैं: सूक्ष्म-संरचना का पैमाना
$\alpha^2E_{\text{I}}$ \unit{meV} में आँकिए, और
$n = 2$ के लिए विपाटन की कोटि GHz में। (ग) हाइड्रोजन-सदृश यूरेनियम ($Z = 92$) के लिए: $v/c$,
$n = 1$ पर --- क्या अनापेक्षिकीय विवेचन टिक पाता है? (घ) $Z\alpha \to 1$
($Z \approx 137$) पर औपचारिक अपसरण भौतिक रूप से किसकी घोषणा करता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:12}
अनुचित-सी अपह्रासिता। (क) चिरसम्मत यांत्रिकी में दिखाइए कि
$V = -k/r$ के लिए कक्षा बंद हो जाती है (कोई पुरस्सरण नहीं), और इसके
लिए संरक्षित लाप्लास--रुंगे--लेंज़ सदिश $\vect A = \vect p\wedge\vect
L - mk\,\vect e_r$ का उल्लेख कीजिए --- न्यूटन के नियम से
$\dd\vect A/\dd t = 0$ जाँचिए। (ख) तर्क दीजिए: दीर्घवृत्त की अक्ष तय
करने वाला कोई संरक्षित सदिश घूर्णनों से परे एक अतिरिक्त सममिति है ---
और अतिरिक्त सममिति का अर्थ अतिरिक्त अपह्रासिता है (स्मरण कीजिए
\cref{exo:b3:quantum-formalism:9})। (ग) विभव में कोई छोटा
$1/r^2$ संशोधन जोड़िए और दिखाइए कि दीर्घवृत्त
पुरस्सरण करने लगता है: अक्ष घूम जाती है, और $\vect A$ अब संरक्षित नहीं। (घ)
जोड़िए: बहु-इलेक्ट्रॉन परमाणुओं में प्रभावी विभव शुद्ध $1/r$ नहीं
होता, और $l$-अपह्रासिता टूट जाती है (सोडियम!); जबकि स्वयं हाइड्रोजन
में वह छोटा संशोधन आपेक्षिकता देती है --- \cref{exo:b3:hydrogen-atom:11}
की कौन-सी प्रेक्षित विशेषता वही है?
\end{exercise}

\section{समस्या: दानव परमाणु}

\begin{problem}[{सप्ताहांत समस्या --- रिडबर्ग परमाणु, मंदाकिनी की
रेडियो चमक से क्वांटम संगणकों तक}]\label{pb:b3:hydrogen-atom:1}
हाइड्रोजन को $n \approx 100$ तक खींचिए और वह बिलकुल दूसरी ही तरह की
वस्तु बन जाता है: माइक्रोमीटरों चौड़ा, मिलीइलेक्ट्रॉनवोल्टों से बँधा,
हास्यास्पद रूप से सुग्राही --- और हास्यास्पद रूप से उपयोगी भी। यह
समस्या हाइड्रोजन के हल को ऊपर की ओर मापित करती है, पहले उन अंतरतारकीय
बादलों तक जो सेंटीमीटर तरंगदैर्घ्यों पर प्रसारण करते हैं, फिर उन
प्रयोगशाला-सरणियों तक जहाँ ऐसे दानव एक-दूसरे को उलझाकर संसाधक बना देते
हैं। आँकड़े:
$E_{\text{I}} = \qty{13.6}{eV}$, $a_0 = \qty{52.9}{pm}$,
$R_{\text{H}}c = \qty{3.29e15}{Hz}$, \qty{300}{K} पर
$k_{\text{B}}T$ \qty{25.9}{meV} है।

\medskip\noindent\textbf{भाग I --- मापन-नियम।}
\begin{enumerate}
  \item $n$ के साथ चार बुनियादी मापन दीजिए: आकार $\langle
        r\rangle$, बंधन $|E_n|$, अंतराल $E_{n+1} - E_n$ (बड़े $n$
        के लिए), और संगत बोर कक्षा की चिरसम्मत कक्षीय आवृत्ति।
  \item दिखाइए कि बड़े $n$ के लिए संक्रमण-आवृत्ति $n+1 \to
        n$ $2R_{\text{H}}c/n^3$ की ओर जाती है, और जाँचिए कि वह
        चिरसम्मत कक्षीय आवृत्ति के बराबर है
        (\cref{pb:b3:hamiltonian-mechanics:1} का संगतता
        सिद्धांत, पूरा किया हुआ)।
  \item $n = 50$ और $n = 100$ पर आकार और बंधन निकालिए।
  \item किसी रिडबर्ग परमाणु का विद्युत् द्विध्रुव $n^2ea_0$ की तरह
        मापित होता है: $n = 50$ पर उसे डिबाई में निकालिए
        ($1\,\text{D} =
        \qty{3.34e-30}{C.m}$) और किसी जल-अणु के
        \qty{1.85}{D} से तुलना कीजिए।
  \item वह विद्युत् क्षेत्र आँकिए जो $n = 50$ के परमाणु को चीर देता
        है, $F \sim |E_n|/(e\,\langle r\rangle)$ के रूप में, V/cm में।
  \item ऐसे परमाणु साधारण प्रयोगशाला-निर्वात की रसायन-विद्या क्यों
        नहीं झेल पाते --- फिर भी अंतरतारकीय दिक् में ठीक-ठाक जी लेते
        हैं (संघट्ट-दरों की तुलना कीजिए)?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- मंदाकिनी की पुनर्संयोजन
रेखाएँ।}
आयनित नीहारिकाओं में प्रोटॉन इलेक्ट्रॉनों को ऊँचे $n$ में पकड़ लेते हैं;
और झरना ``रेडियो पुनर्संयोजन रेखाओं'' की एक कंघी विकिरित करता है।
\begin{enumerate}[resume]
  \item $n = 110 \to 109$ संक्रमण (जिसे H109$\alpha$ कहते हैं) की
        आवृत्ति निकालिए।
  \item वह किस बैंड में पड़ती है, और उसे कैसा दूरदर्शी ग्रहण करता
        है?
  \item H109$\alpha$ की तरंगदैर्घ्य निकालिए और उसकी तुलना
        उत्सर्जक परमाणु के आकार से कीजिए: एक तरंगदैर्घ्य में कितने
        परमाणु समाते हैं?
  \item ये रेखाएँ अपनी डॉप्लर चौड़ाई से नीहारिका का ताप नाप देती हैं:
        $T = \qty{e4}{K}$ पर हाइड्रोजन की तापीय
        चाल $\sim\qty{12.8}{km/s}$ है --- आंशिक
        रेखा-चौड़ाई $\Delta\nu/\nu$ और H109$\alpha$ की चौड़ाई
        kHz में निकालिए।
  \item रेडियो तरंगें उस धूल को पार कर जाती हैं जो प्रकाशिक आकाश को
        काला कर देती है: इससे पुनर्संयोजन-रेखा वाले खगोलविद क्या
        नक्शा बना पाते हैं, जो बामर-रेखा वाले नहीं बना पाते?
  \item मोटे तौर पर $n \sim 1000$ से ऊपर (परमाणु $0.05$ मिलीमीटर
        चौड़े!), स्तर दिक् में भी धुँधले होकर सांतत्यक बन जाते हैं:
        ऐसी अवस्थाओं को चौड़ा करने या नष्ट करने वाले दो प्रभाव बताइए।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- प्रयोगशाला में दानव।}
\begin{enumerate}[resume]
  \item लेज़र रूबिडियम परमाणुओं को माइक्रोमीटर-अंतराल वाली
        प्रकाशिक-चिमटी सरणियों में $n \approx 70$ तक चढ़ा देते हैं।
        दूरी $R$ पर के ऐसे दो परमाणु अपने प्रेरित द्विध्रुवों से
        अन्योन्यक्रिया करते हैं (स्मरण कीजिए
        \cref{exo:b3:harmonic-oscillator:10}): द्विध्रुव
        $\propto n^2$ के साथ दिखाइए कि वान डर वाल्स प्रबलता किसी
        विशाल $n^{11}$ की तरह मापित होती है ($C_6 \propto d^4/\Delta E$
        लीजिए, जहाँ स्तर-अंतराल $\Delta E \propto n^{-3}$)।
  \item ``अवरोध'': किसी त्रिज्या $R_{\text{b}}$ के भीतर
        अन्योन्यक्रिया दुहरी उत्तेजित अवस्था को लेज़र-अनुनाद से बाहर
        खिसका देती है, अतः \emph{दो} परमाणु एक साथ उत्तेजित नहीं हो
        सकते। एक वाक्य में समझाइए कि इससे कोई द्वि-क्यूबिट फाटक कैसे
        बन जाता है।
  \item अवरोध-त्रिज्याएँ कई माइक्रोमीटर तक पहुँचती हैं --- परमाणुओं
        के मूल-अवस्था आकार से हज़ारों गुना: यह लंबी पहुँच
        (रसायन की नैनोमीटर पहुँच से तुलना कीजिए)
        किसी मापनीय संसाधक को ठीक क्यों चाहिए?
  \item $n = 70$ की कोई रिडबर्ग अवस्था $\sim\qty{100}{\micro s}$
        जीती है, जबकि कोई फाटक $\sim\qty{0.5}{\micro s}$ लेता है:
        एक जीवनकाल में मोटे तौर पर कितनी संक्रियाएँ समाती हैं, और
        अकेले जीवनकाल के बजाय वही अनुपात क्यों मायने रखता है?
  \item कमरे के ताप का कृष्णिका विकिरण, जिसका शिखर
        \qty{100}{meV} के पास है पर जिसमें नीची ऊर्जा की समृद्ध पुच्छ
        है, केवल $\sim\qty{0.1}{meV}$ के अंतराल वाले पड़ोसी रिडबर्ग
        स्तरों के बीच संक्रमण चला देता है: यथार्थ प्रयोगों को
        परमाणुओं को ठंडे कवचों में क्यों बंद करना पड़ता है?
  \item आँकिए कि \qty{300}{K} का कोई फ़ोटॉन-कुंड प्रति सेकंड कितने
        ऐंटेना-जैसे द्विध्रुव संक्रमण ($\propto n^2$) चलाता है,
        किसी $n = 1$ परमाणु की तुलना में: कौन-सा मापन रिडबर्ग
        परमाणुओं को सूक्ष्मतरंग और टेराहर्ट्ज़ क्षेत्रों का उत्तम
        \emph{संवेदक} बना देता है?
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- बड़ा चित्र।}
\begin{enumerate}[resume]
  \item एक ही सूत्र, $E_n = -E_{\text{I}}\mu Z^2/m_{\text{e}}n^2$,
        हाइड्रोजन-सदृश यूरेनियम ($Z = 92$, $n = 1$) से लेकर किसी
        $n = 1000$ रिडबर्ग अवस्था तक बंधन ऊर्जा की कितनी कोटियाँ
        समेटता है? दोनों सिरे निकालिए।
  \item नीचे $l$ वाली अवस्थाओं के विकिरणी जीवनकाल $n^3$ की तरह मापित
        होते हैं: \qty{1.6}{ns} के $2p$ जीवनकाल से किसी
        $100p$ अवस्था का जीवनकाल आँकिए --- और भाग III की
        कमरे-ताप कृष्णिका समस्या से तुलना कीजिए।
  \item CERN में प्रतिहाइड्रोजन के प्रति-परमाणुओं का वर्णक्रम अब
        $1s \to 2s$ अंतराल पर पंद्रह अंकों तक लिया जाता है: साधारण
        हाइड्रोजन से मेल किस मूल सममिति की परीक्षा करता है, और इस
        तुलना के लिए हाइड्रोजन ही सही परमाणु क्यों है?
  \item $n \sim 100$ पर इलेक्ट्रॉन की दे ब्रॉग्ली तरंग लगभग किसी
        चिरसम्मत कक्षा को लपेट लेती है; जबकि $n = 1$ पर कोई कक्षा है
        ही नहीं: एक वाक्य में बताइए कि चिरसम्मत चित्र कहाँ चालू होता
        है।
  \item रिडबर्ग नियतांक पंद्रह अंकों तक नापे जाते हैं: सारे तंत्रों
        में हाइड्रोजन ही परमाणु-भौतिकी का स्वाभाविक यथार्थता-लंगर
        क्यों है?
  \item 2012 के नोबेल पुरस्कार (आरोश) में रिडबर्ग परमाणु ऐसे
        फ़ोटॉन-गणकों की तरह काम में लाए गए जो फ़ोटॉन को नष्ट नहीं
        करते: भाग I और भाग III के कौन-से दो गुण उन्हें
        सूक्ष्मतरंग क्षेत्रों की आदर्श अ-विध्वंसक जाँच बनाते हैं?
  \item नामित परिणाम का सार लिखिए: एक ठीक-ठीक हल किए गए परमाणु के
        $n^2$, $n^{-2}$, $n^{-3}$ और $n^{11}$ मापन
        हाइड्रोजन को \qty{52.9}{pm} से आधे माइक्रोमीटर तक खींच देते
        हैं, उसकी आवाज़ \qty{121.6}{nm} से \qty{6}{cm} तक सुर में
        बाँध देते हैं, और उसे मंदाकिनी का रेडियो प्रकाश-स्तंभ तथा
        उदासीन-परमाणु क्वांटम संगणकों का द्वि-क्यूबिट फाटक, दोनों
        बना देते हैं।
\end{enumerate}
\end{problem}
