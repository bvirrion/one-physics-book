\chapter{Electromagnetism in Matter}\label{ch:b3:electromagnetism-in-matter}

% Provenance (not for print): completes the "électromagnétisme dans la
% matière" module (dielectrics, magnetism, ferromagnetism) common to the
% surveyed third-year curricula; the Year 2 volume covered only wave
% propagation in media.

Slide a sheet of plastic between a charged capacitor's plates and the
voltage drops, as if charge had appeared from nowhere. Wind a coil
around an iron nail and its magnetic field grows a thousandfold. Peel
a decorative magnet off the refrigerator and it \emph{remembers} which
way it was magnetised --- for decades, with no power supply. The Year 2
volume treated fields in matter wholesale, hiding the material inside
an index of refraction; this chapter opens the material up. The plan
is the same for electricity and for magnetism: matter is a crowd of
microscopic dipoles; sum them into a density ($\vect P$ or $\vect M$);
invent an auxiliary field ($\vect D$ or $\vect H$) that sees only the
charges and currents we control; then descend to the microscopic scale
--- where \cref{ch:b3:spin-two-level}'s spins, the Boltzmann factor of
\cref{ch:b3:canonical-ensemble} and the mean-field transition of
\cref{ch:b3:phase-transitions} take over --- to compute what the
material does. The chapter ends at the scrapyard, designing the
electromagnet that lifts cars.

\section{Polarisation and the displacement field}

\begin{definition}[Polarisation and bound charges]\label{def:b3:electromagnetism-in-matter:polarisation}
\index{polarisation}\index{bound charge}\index{dielectric}
A \emph{dielectric} is an insulator whose molecules acquire (or
already own) electric dipole moments. The material's response is
summarised by the \emph{polarisation} $\vect P$: the dipole moment
per unit volume, $\vect P = n\langle\vect p\rangle$ for $n$ molecules
per unit volume. A polarised block carries \emph{bound charges} ---
not free to leave, but perfectly real: a surface density
$\sigma_{\text{b}} = \vect P\cdot\vect n$ where the polarisation
meets the surface, and a volume density $\rho_{\text{b}} =
-\nabla\cdot\vect P$ wherever it is non-uniform. A uniformly
polarised slab is exactly equivalent to two sheets of charge $\pm P$
on its faces: all its interior dipole heads and tails cancel.
\end{definition}

\begin{proposition}[The displacement field]\label{prop:b3:electromagnetism-in-matter:displacement}
\index{electric displacement}\index{electric susceptibility}\index{relative permittivity}
Total charge is free plus bound; Gauss's law with $\rho =
\rho_{\text{f}} + \rho_{\text{b}}$ rearranges into a law for the
\emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{electric displacement}} $\vect D = \varepsilon_0\vect E +
\vect P$:
\[
\nabla\cdot\vect D = \rho_{\text{f}} :
\]
$\vect D$'s sources are the free charges only --- the ones on our
plates and wires. In a \emph{linear} \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielectric} the response is
proportional, $\vect P = \varepsilon_0\chi_{\text{e}}\vect E$, so
$\vect D = \varepsilon_0\varepsilon_{\text{r}}\vect E$ with
$\varepsilon_{\text{r}} = 1 + \chi_{\text{e}}$ the \emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{relative
permittivity}}: air $1.0006$, polyethylene $2.3$, glass $\sim 5$,
water an enormous $80$. Between capacitor plates held at fixed
charge, $\vect D$ is unchanged by the \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielectric}, so $\vect E =
\vect D/\varepsilon_0\varepsilon_{\text{r}}$ \emph{drops} by
$\varepsilon_{\text{r}}$: the bound surface charges face the free
ones and cancel most of their field. Capacitance is multiplied by
$\varepsilon_{\text{r}}$ --- the entire capacitor industry in one
Greek letter.
\end{proposition}

\begin{proof}
$\nabla\cdot(\varepsilon_0\vect E) = \rho_{\text{f}} +
\rho_{\text{b}} = \rho_{\text{f}} - \nabla\cdot\vect P$; move the
\omterm{def:b3:electromagnetism-in-matter:polarisation}{polarisation} term to the left.
\end{proof}

\section{Where permittivity comes from}

\begin{proposition}[Induced polarisation and Clausius--Mossotti]\label{prop:b3:electromagnetism-in-matter:mossotti}
\index{polarisability}\index{Clausius--Mossotti relation}
An atom in a field $\vect E_{\text{loc}}$ stretches into a dipole
$\vect p = \alpha\vect E_{\text{loc}}$, with $\alpha$ the
\emph{polarisability} (dimension: $\varepsilon_0\times$volume ---
roughly $\varepsilon_0$ times the atomic volume). In a dilute gas
$E_{\text{loc}} \approx E$ and $\chi_{\text{e}} =
n\alpha/\varepsilon_0$ is small. In a dense medium each molecule also
feels its polarised neighbours: carving a small spherical cavity
around it gives $\vect E_{\text{loc}} = \vect E + \vect
P/3\varepsilon_0$, and self-consistency yields the
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{Clausius--Mossotti relation}}
\[
\frac{\varepsilon_{\text{r}} - 1}{\varepsilon_{\text{r}} + 2}
= \frac{n\alpha}{3\varepsilon_0} :
\]
measure a vapour's tiny susceptibility and predict the liquid's ---
it works to a few percent for nonpolar liquids.
\end{proposition}
\admitted

\begin{remark}
The cavity field $\vect P/3\varepsilon_0$ is the uniformly-polarised-%
sphere result quoted here without proof; the honest boundary-value
derivation belongs to a dedicated electrodynamics course. Everything
else above is bookkeeping.
\end{remark}

\begin{proposition}[Polar molecules: the Langevin function]\label{prop:b3:electromagnetism-in-matter:langevin}
\index{Langevin function}\index{orientation polarisation}
A molecule with a \emph{permanent} dipole $p$ (water:
$p = \qty{6.2e-30}{C.m}$) needs no stretching --- only aligning,
against thermal agitation. The canonical average of
\cref{ch:b3:canonical-ensemble} over orientations, with energy
$-pE\cos\theta$, gives
\[
\langle\cos\theta\rangle = L(x) = \coth x - \frac1x ,
\qquad x = \frac{pE}{k_{\text{B}}T} ,
\]
the \emph{\omterm{prop:b3:electromagnetism-in-matter:langevin}{Langevin function}}: linear ($L \approx x/3$) at small $x$,
saturating at $1$ when the field wins outright. In the linear regime
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ --- an electric
Curie law, falling as $1/T$: heat a polar liquid and its permittivity
\emph{drops}, the signature that distinguishes permanent from induced
dipoles. For water at room temperature this predicts
$\chi_{\text{e}} \sim 12$ --- the right magnitude, with hydrogen-bond
cooperation pushing the real value to $80$.
\end{proposition}

\begin{proof}[Partial proof]
$Z = \int_0^\pi \eu^{x\cos\theta}\,2\pi\sin\theta\,\dd\theta \propto
\sinh x/x$, and $\langle\cos\theta\rangle = \dd\ln Z/\dd x = \coth x
- 1/x$. Expanding $\coth x = 1/x + x/3 - \dots$ for small $x$ gives
$L \approx x/3$, hence $P = np\langle\cos\theta\rangle =
np^2E/3k_{\text{B}}T$.
\end{proof}

\section{Magnetisation and the field \texorpdfstring{$H$}{H}}

\begin{definition}[Magnetisation, bound currents, and H]\label{def:b3:electromagnetism-in-matter:magnetisation}
\index{magnetisation}\index{magnetic field H}\index{magnetic susceptibility}
The magnetic story runs in parallel. Matter is full of microscopic
current loops --- orbiting and spinning electrons, each a magnetic
dipole $\vect\mu$; their density is the \emph{magnetisation}
$\vect M = n\langle\vect\mu\rangle$. A uniformly magnetised block is
equivalent to a \emph{bound surface current} circulating around its
sides (interior loops cancel pairwise; a bar magnet \emph{is} a
solenoid of bound current). Splitting currents into free and bound,
Ampère's law rearranges around the auxiliary field
\[
\vect H = \frac{\vect B}{\mu_0} - \vect M , \qquad
\oint \vect H\cdot\dd\vect\ell = I_{\text{free}} :
\]
$\vect H$ is sourced by the currents in our wires (units
\unit{A/m}), while $\vect B$ keeps the physics --- forces, flux,
induction. Linear media have $\vect M = \chi_{\text{m}}\vect H$ and
$\vect B = \mu_0(1 + \chi_{\text{m}})\vect H =
\mu_0\mu_{\text{r}}\vect H$.
\end{definition}

\begin{remark}[Three magnetic personalities]\label{rem:b3:electromagnetism-in-matter:zoo}
\index{diamagnetism}\index{paramagnetism}\index{ferromagnetism}
Materials answer $\vect H$ in three ways.
\emph{Diamagnets} ($\chi_{\text{m}} \sim -10^{-5}$: water, copper,
graphite) respond like Lenz's law made permanent: the applied field
perturbs every electron orbit so as to oppose it --- weak, universal,
temperature-independent, and repelled by magnets (a frog has been
levitated this way in a \qty{16}{T} bore).
\emph{Paramagnets} ($\chi_{\text{m}} \sim +10^{-5}$ to $10^{-3}$:
aluminium, O$_2$) own permanent moments --- unpaired spins from
\cref{ch:b3:identical-particles} --- that align like Langevin's
dipoles: $\chi_{\text{m}} = \mu_0 n\mu^2/3k_{\text{B}}T$, a Curie
$1/T$ law (the quantum two-level version is
\cref{exo:b3:canonical-ensemble:11}).
\emph{Ferromagnets} (iron, cobalt, nickel: $\mu_{\text{r}}$ in the
thousands) are paramagnets whose spins also talk to each other ---
the exchange coupling of \cref{exo:b3:identical-particles:12} ---
and below the \omterm{thm:b3:phase-transitions:meanfield}{Curie temperature} they align \emph{spontaneously}:
\cref{thm:b3:phase-transitions:meanfield}, now put to work.
\end{remark}

\section{Ferromagnetism at work: domains and hysteresis}

\begin{definition}[Domains and hysteresis]\label{def:b3:electromagnetism-in-matter:hysteresis}
\index{magnetic domain}\index{hysteresis}\index{remanence}\index{coercivity}
A raw lump of iron is \emph{not} magnetised: it shatters into
\emph{domains}, micron-scale regions each fully magnetised but
pointing differently, so the exterior field (and its energy cost)
nearly cancels. An applied $\vect H$ moves the domain walls ---
favourable domains grow --- and then rotates whole domains into
line. Walls snag on defects, so the process is irreversible: sweep
$H$ up and down and $B(H)$ traces a loop, the \emph{hysteresis
cycle}. Switch the current off and a \emph{remanent} field
$B_{\text{r}}$ survives; cancelling it needs the reverse
\emph{coercive} field $H_{\text{c}}$. The loop's enclosed area is
energy dissipated per cycle and per unit volume. \emph{Soft}
materials (silicon steel, ferrites: thin loop, small
$H_{\text{c}}$) make transformer and motor cores; \emph{hard} ones
(alnico, Nd$_2$Fe$_{14}$B: fat loop, huge $H_{\text{c}}$) make
permanent magnets --- and magnetic memory: the refrigerator magnet
and the hard disk are hysteresis loops that refuse to forget.
\end{definition}

\begin{example}[The iron-core electromagnet]\label{ex:b3:electromagnetism-in-matter:electromagnet}
Wind $N$ turns carrying $I$ around an iron ring ($\mu_{\text{r}}
\sim 5000$) with a small air gap $e$. Ampère's law for $\vect H$
around the loop: $H_{\text{iron}}\ell + H_{\text{gap}}e = NI$, and
flux continuity makes $B$ common, so
\[
B\Big(\frac{\ell}{\mu_0\mu_{\text{r}}} + \frac{e}{\mu_0}\Big) = NI .
\]
With $\ell = \qty{1}{m}$ and $e = \qty{1}{cm}$, the gap term is
fifty times the iron term: nearly all the coil's effort is spent
pushing field across one centimetre of air. That is the magnetic
circuit in one line --- iron is a near-perfect conductor of flux,
air the resistor --- and it is why motors, relays and scrapyard
lifters keep their air gaps ruthlessly thin
(\cref{pb:b3:electromagnetism-in-matter:1}).
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:1}
Bound-charge bookkeeping. A slab of thickness $d$ is uniformly
polarised along its normal, magnitude $P$. (a) Give the \omterm{def:b3:electromagnetism-in-matter:polarisation}{bound
charge} densities on each face. (b) Show the field \emph{inside} due
to those sheets is $E = P/\varepsilon_0$, opposing $\vect P$. (c)
What is $\vect D$ inside, with no free charge anywhere? (d) A bar
electret (frozen-in $\vect P$) is the electric analogue of which
magnetic object?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:2}
Capacitor with \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielectric}. A parallel-plate capacitor
($C_0 = \qty{100}{pF}$) is charged to \qty{12}{V} and disconnected.
A slab with $\varepsilon_{\text{r}} = 4$ fills it. Find (a) the new
capacitance; (b) the new voltage; (c) the energy before and after
--- where did the difference go? (d) Repeat (b)–(c) with the
battery kept connected.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:3}
The susceptibility zoo. Classify (dia-, para-, or ferromagnetic)
and justify from electronic structure where you can: water
($\chi_{\text{m}} = -9\times10^{-6}$), aluminium
($+2.2\times10^{-5}$), liquid oxygen ($+3.5\times10^{-3}$ ---
why is O$_2$ magnetic at all?), nickel ($\mu_{\text{r}} \sim 600$
at small field). Which of the four would a strong magnet visibly
attract, and which ever so slightly repel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:4}
$B$, $H$, $M$. A long solenoid ($n = \num{2000}$ turns per metre,
$I = \qty{1.5}{A}$) is filled with iron, and $B = \qty{1.6}{T}$ is
measured. Compute (a) $H$; (b) $M$; (c) the amplification
$B/\mu_0H$; (d) the bound surface current per metre equivalent to
$M$ --- compare it with the coil's own $nI$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:5}
Coaxial cable, insulated. A coaxial line (inner radius $a$, outer
$b$) carries free charge $\lambda$ per unit length on the core,
with \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielectric} $\varepsilon_{\text{r}}$ between. (a) Find
$\vect D(r)$ from symmetry. (b) Deduce $\vect E$ and $\vect P$.
(c) Compute the \omterm{def:b3:electromagnetism-in-matter:polarisation}{bound charge} densities at $r = a$ and $r = b$ and
check they sum to zero per unit length. (d) Show the capacitance
per unit length is multiplied by $\varepsilon_{\text{r}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:6}
Clausius--Mossotti at work. Gaseous argon at \qty{1}{bar},
\qty{300}{K} has $\varepsilon_{\text{r}} = 1.00052$. (a) Extract
$\alpha/\varepsilon_0$ (a volume) and compare with the atomic
volume. (b) Liquid argon has $n = \qty{2.1e28}{m^{-3}}$: predict
$\varepsilon_{\text{r}}$ from Clausius--Mossotti (measured: 1.53).
(c) Why does the naive $\chi = n\alpha/\varepsilon_0$ overshoot
less badly here than for water? (d) For which molecules must the
whole scheme fail, and what replaces it?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:7}
Water's giant permittivity. (a) From
\cref{prop:b3:electromagnetism-in-matter:langevin}, compute
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ for water
($n = \qty{3.3e28}{m^{-3}}$, $p = \qty{6.2e-30}{C.m}$,
$T = \qty{300}{K}$). (b) Estimate the field needed for $x = 1$
(serious saturation) and compare with water's \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielectric} strength
$\sim\qty{7e7}{V/m}$. (c) Microwave ovens run at \qty{2.45}{GHz}:
why does a \emph{lossy} rotating-dipole response heat food, and
why does ice heat far more slowly than liquid water? (d) Predict
the sign of $\dd\varepsilon_{\text{r}}/\dd T$ for water and for
argon.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:8}
Diamagnetism estimated. Treat an atomic electron as a charge on a
ring of radius $r$; an applied $B$ changes its angular frequency by
the Larmor shift $\Delta\omega = eB/2m_{\text{e}}$. (a) Show the
induced moment is $\Delta\mu = -e^2r^2B/4m_{\text{e}}$ (opposing
$B$). (b) Averaging orientations and summing $Z$ electrons per
atom, one finds $\chi_{\text{m}} = -\mu_0ne^2Z\langle
r^2\rangle/6m_{\text{e}}$: evaluate for water ($n =
\qty{3.3e28}{m^{-3}}$ molecules, $Z = 10$, $\langle r^2\rangle
\approx (\qty{0.7}{\angstrom})^2$) and compare with the measured
$-9\times10^{-6}$. (c) Why is diamagnetism temperature-independent
while paramagnetism is not? (d) Why does even a frog levitate in
\qty{16}{T}, and why must the magnet's field be \emph{non-uniform}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:9}
Curie paramagnetism and cooling. A salt carries $n =
\qty{2e27}{m^{-3}}$ ions of moment $\mu \approx \mu_{\text{B}}$.
(a) Compute $\chi_{\text{m}} = \mu_0n\mu^2/3k_{\text{B}}T$ at
\qty{300}{K} and at \qty{1}{K}. (b) At what temperature would
$\chi_{\text{m}}$ reach $1$ --- and what physics (neglected here)
intervenes first in most salts? (c) Adiabatic demagnetisation:
magnetise at \qty{1}{K}, isolate, then remove the field slowly ---
explain with the spin entropy of
\cref{ch:b3:microcanonical-ensemble} why the sample cools. (d) Why
does the method need a \emph{para}magnet rather than a ferromagnet?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:10}
The magnetic circuit. An iron torus ($\ell = \qty{60}{cm}$,
$\mu_{\text{r}} = 4000$, section $A = \qty{16}{cm^2}$) carries
$N = 500$ turns and a gap $e = \qty{4}{mm}$. (a) Show $NI =
B(\ell/\mu_0\mu_{\text{r}} + e/\mu_0)$ and compute the current for
$B = \qty{1.2}{T}$. (b) What fraction of $NI$ is spent on the gap?
(c) Define the reluctance $\mathcal R = \ell/\mu A$ of each
segment and restate (a) as a series ``Ohm's law'' for flux. (d)
The iron saturates near \qty{1.8}{T}: explain what happens to the
circuit model --- and to your motor --- beyond it.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:11}
\omterm{def:b3:electromagnetism-in-matter:hysteresis}{Hysteresis} losses. A transformer core (volume \qty{4e-3}{m^3})
runs at \qty{50}{Hz}. Its silicon-steel loop encloses
$\sim\qty{40}{J/m^3}$ per cycle. (a) Compute the \omterm{def:b3:electromagnetism-in-matter:hysteresis}{hysteresis} power
loss. (b) The same core in hard steel
($\sim\qty{6e3}{J/m^3}$): loss, and verdict. (c) Eddy currents add
a loss $\propto f^2$: explain why cores are laminated and why
ferrites take over at radio frequencies. (d) A hard disk bit must
\emph{keep} its \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetisation} against thermal kicks for ten years:
relate the demands on $H_{\text{c}}$ for memory to those for a
transformer, and conclude that no single material can do both jobs.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:12}
The demagnetising field. Inside a uniformly magnetised body with no
free current, $\oint\vect H\cdot\dd\vect\ell = 0$ forces
$\vect H$ inside to \emph{oppose} $\vect M$ ($\vect H =
-N_{\text{d}}\vect M$, with $N_{\text{d}}$ a shape factor: $1/3$
for a sphere, $\approx 0$ along a long needle, $\approx 1$ across a
thin plate). (a) Justify the needle and plate limits with the
bound-current (solenoid) picture. (b) Why does a stubby magnet
partially demagnetise itself while a needle keeps its
\omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetisation}? (c) Connect to compass needles and to the elongated
grains of magnetic tape. (d) A soft-iron sphere sits in a uniform
external $B_0$: explain why its response saturates at $\vect H
\approx 0$ inside, i.e.\ $M \approx 3B_0/\mu_0$ at most, however
large $\mu_{\text{r}}$ is.
\end{exercise}

\section{Problem: The scrapyard lifter}

\begin{problem}\label{pb:b3:electromagnetism-in-matter:1}
\emph{The scrapyard lifter.} A recycling yard orders a crane
electromagnet: a flat-faced iron pot, one metre across, that must
lift crushed-car bales of two tonnes, then --- just as important ---
\emph{drop} them on command. You are the design engineer.

\medskip\noindent\textbf{Part I --- Why iron.}

\begin{enumerate}
  \item The bare coil: 400 turns carrying \qty{25}{A} spread over
        a magnetic path of $\sim\qty{1}{m}$. Estimate $B \sim
        \mu_0NI/\ell$ with no iron.
  \item Explain in two sentences, with domains, why filling the
        coil with iron multiplies this by $\sim\mu_{\text{r}}$.
  \item Iron's \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetisation} saturates at $M_{\text{s}} \approx
        \qty{1.7e6}{A/m}$. Compute the ceiling $\mu_0M_{\text{s}}$
        this puts on the pole field, whatever the coil does.
  \item Each iron atom contributes $\approx 2.2$ \omterm{prop:b3:quantum-angular-momentum:magnetic}{Bohr magnetons}.
        Check: with $n = \qty{8.5e28}{m^{-3}}$, recover
        $M_{\text{s}}$.
  \item Why must the pole faces be machined flat and kept free of
        rust and grit? (Think of
        \cref{ex:b3:electromagnetism-in-matter:electromagnet}.)
  \item The load itself becomes part of the magnetic circuit.
        Sketch the flux path: pot core, north face, steel bale,
        south face, back through the yoke.
\end{enumerate}

\medskip\noindent\textbf{Part II --- The magnetic circuit and the force.}

\begin{enumerate}[resume]
  \item Model the circuit: iron path $\ell = \qty{1.2}{m}$,
        $\mu_{\text{r}} = 2000$, and two effective air gaps
        (face--bale contact) of $e = \qty{1.5}{mm}$ each. Write
        Ampère's law for $H$ around the loop.
  \item Compute the three reluctance terms per unit area
        ($\ell/\mu_{\text{r}}$ versus $2e$) and show the
        millimetre gaps still consume most of the coil's effort.
  \item With $NI = \num{10000}$ ampere-turns, compute $B$ in the
        gaps.
  \item The lifting pressure on each pole face is $B^2/2\mu_0$
        (magnetic energy density released per metre of approach).
        Evaluate it in \unit{kPa} for your $B$.
  \item Total pole-face area $A = \qty{0.12}{m^2}$: compute the
        lifting force and convert to tonnes.
  \item Does it meet the two-tonne specification with a factor-two
        margin? If not, adjust $NI$ and state the new current.
  \item Crushed bales touch the faces on perhaps a third of their
        area, with wider effective gaps. Recompute the force for
        $e = \qty{4}{mm}$, $A_{\text{eff}} = \qty{0.04}{m^2}$ and
        comment on why rated lifts quote ``flat plate'' capacity.
\end{enumerate}

\medskip\noindent\textbf{Part III --- Remembering and forgetting.}

\begin{enumerate}[resume]
  \item The yard flips the switch to drop a bale --- and it hangs
        on. Name the culprit, with the \omterm{def:b3:electromagnetism-in-matter:hysteresis}{hysteresis} loop.
  \item Why must the pot be \emph{soft} iron (small
        $H_{\text{c}}$, small $B_{\text{r}}$) rather than hard
        magnet steel?
  \item Even soft iron keeps a little \omterm{def:b3:electromagnetism-in-matter:hysteresis}{remanence}. Propose the
        standard cure: a brief \emph{reversed} current pulse ---
        which point of the loop is it aiming for?
  \item Some controllers instead apply a decaying alternating
        current. Sketch what the $B$--$H$ trajectory does and why
        it ends demagnetised.
  \item The crane also handles hot slabs straight from a furnace,
        at \qty{800}{\celsius}. Iron's Curie point is
        \qty{1043}{K}: what happens to the lift force, and why?
        (\cref{thm:b3:phase-transitions:meanfield}.)
  \item A summer apprentice suggests saving copper by doubling
        the gap and doubling $NI$. Use your Part II formulas to
        explain the asymmetry: which halves and which merely
        holds?
\end{enumerate}

\medskip\noindent\textbf{Part IV --- Power, heat, and the bill.}

\begin{enumerate}[resume]
  \item The coil: 400 turns of copper, mean turn length
        \qty{2.5}{m}, wire section \qty{16}{mm^2}, resistivity
        $\rho = \qty{1.7e-8}{\Omega.m}$. Compute its resistance.
  \item At $I = \qty{25}{A}$: the dissipated power, and the daily
        energy for an eight-hour shift at \qty{60}{\percent} duty.
  \item The magnetic energy stored in the two gaps
        ($B^2/2\mu_0 \times$ volume): compute it and compare with
        one \emph{second} of coil dissipation. Where does all the
        rest of the electrical energy go?
  \item Dropping the load returns almost none of the stored
        energy to the grid. Explain, with the loop area and the
        inductive spike, why the controller needs a freewheel
        (flyback) path across the coil.
  \item Power fails with a bale in the air. What does the bale
        do, and what does the \omterm{def:b3:electromagnetism-in-matter:hysteresis}{remanence} \emph{alone} hold? Justify
        the yard rule that nobody walks under a powered magnet ---
        and the sales pitch for battery-backed lifters.
  \item Sum up the design in four lines: field ceiling set by
        $M_{\text{s}}$, force by $B^2A/2\mu_0$, controllability by
        soft-iron \omterm{def:b3:electromagnetism-in-matter:hysteresis}{hysteresis}, operating cost by $RI^2$ --- the
        whole chapter hanging from one crane.
\end{enumerate}
\end{problem}
