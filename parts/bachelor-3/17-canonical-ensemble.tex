\chapter{The Canonical Ensemble}\label{ch:b3:canonical-ensemble}

Isolated systems are a theorist's fiction: real samples sit in
thermostats, rooms, oceans of air --- in contact with a reservoir
that fixes not their energy but their \emph{temperature}. The Year 2
volume met the resulting law empirically: the probability of a state
of energy $E$ carries the factor $\eu^{-E/k_{\text{B}}T}$. This
chapter derives that Boltzmann factor in four lines from the
counting of \cref{ch:b3:microcanonical-ensemble}, then builds the
machine that makes statistical physics an industrial discipline: the
\emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{partition function}} $Z$, one sum from which energy, entropy,
pressure, heat capacity and fluctuations all fall by
differentiation. The machine's first victories are recounted here:
why heat capacities die at low temperature (Einstein, 1907 --- the
first quantum theory of matter), why hydrogen's heat capacity
climbs a staircase, why the sky thins exponentially --- and how
Perrin, watching microscopic grains settle in a drop of water,
counted Avogadro's number and ended the debate about the reality of
atoms.

\section{The Boltzmann distribution, derived}

\begin{theorem}[Canonical distribution]\label{thm:b3:canonical-ensemble:boltzmann}
\index{Boltzmann distribution}\index{canonical ensemble}\index{partition function}
A system exchanging energy with a large reservoir at temperature $T$
occupies its \omterm{def:b3:microcanonical-ensemble:states}{microstate} $s$ (energy $E_s$) with probability
\[
\mathcal P_s = \frac{\eu^{-E_s/k_{\text{B}}T}}{Z} , \qquad
Z = \sum_s \eu^{-E_s/k_{\text{B}}T}
\]
($\beta = 1/k_{\text{B}}T$ hereafter). $Z$, the \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{partition
function}}, normalises the probabilities --- and turns out to hold
the entire thermodynamics of the system.
\end{theorem}

\begin{proof}
The system in state $s$ leaves the reservoir the energy $E - E_s$:
by the \omterm{thm:b3:microcanonical-ensemble:postulate}{fundamental postulate} applied to the whole, $\mathcal P_s
\propto \Omega_{\text{res}}(E - E_s)$. Expand the reservoir's
entropy, huge and smooth, to first order: $S_{\text{res}}(E - E_s)
= S_{\text{res}}(E) - E_s\,\partial S_{\text{res}}/\partial E =
S_{\text{res}}(E) - E_s/T$ by the definition of temperature. Hence
$\Omega_{\text{res}} \propto \eu^{-E_s/k_{\text{B}}T}$. (Higher
orders die with the reservoir's size.)
\end{proof}

\begin{proposition}[The machine]\label{prop:b3:canonical-ensemble:machine}
\index{free energy}\index{heat capacity!and fluctuations}
From $Z(T, V, N)$:
\[
\langle E\rangle = -\frac{\partial\ln Z}{\partial\beta} , \qquad
F = -k_{\text{B}}T\ln Z , \qquad
S = -\frac{\partial F}{\partial T} , \qquad
P = -\frac{\partial F}{\partial V} ,
\]
where $F = \langle E\rangle - TS$ is the \emph{\omterm{prop:b3:canonical-ensemble:machine}{free energy}}; and
the heat capacity is a \emph{fluctuation}:
\[
C = \frac{\partial\langle E\rangle}{\partial T}
 = \frac{\langle E^2\rangle - \langle E\rangle^2}
{k_{\text{B}}T^2} .
\]
For independent, distinguishable subsystems $Z = z^N$; for $N$
\omterm{def:b3:identical-particles:exchange}{identical particles} in a common box, $Z = z^N/N!$. At fixed $T$ and
$V$, equilibrium \emph{minimises} $F$: nature trades energy against
entropy at the exchange rate $T$ --- the single most useful
principle in the physics of matter.
\end{proposition}

\begin{proof}[Partial proof]
$-\partial_\beta\ln Z = \sum E_s\eu^{-\beta E_s}/Z = \langle
E\rangle$; one more derivative gives $\langle E^2\rangle - \langle
E\rangle^2$, and the chain rule converts $\partial_\beta$ to
$\partial_T$. The identifications of $F$, $S$, $P$ follow by
comparing $\dd(-k_{\text{B}}T\ln Z)$ with the thermodynamic
$\dd F = -S\dd T - P\dd V$; factorisation is the exponential of a
sum. The minimum principle: $F_{\text{system}}$ decreasing is
$S_{\text{total}}$ increasing, since $\Delta S_{\text{res}} =
-\Delta E_{\text{sys}}/T$.
\end{proof}

\section{First victories}

\begin{example}[Two levels, in two lines]\label{ex:b3:canonical-ensemble:twolevel}
For levels $0, \epsilon$: $z = 1 + \eu^{-\beta\epsilon}$, giving
$\langle E\rangle = N\epsilon/(\eu^{\beta\epsilon} + 1)$ --- the
result that cost the microcanonical route a page of Stirling
(\cref{exo:b3:microcanonical-ensemble:8}). The canonical formalism
is the microcanonical one with the combinatorics pre-digested; the
Schottky bump in $C(T)$ follows by one differentiation.
\end{example}

\begin{example}[Einstein's solid, and the death of Dulong--Petit]\label{ex:b3:canonical-ensemble:einstein}
\index{Einstein solid}\index{Dulong--Petit law}
Model a crystal as $3N$ quantum oscillators of frequency $\omega$
(\cref{ch:b3:harmonic-oscillator}). Per oscillator,
\[
z = \sum_n\eu^{-\beta\hbar\omega(n + 1/2)}
 = \frac{1}{2\sinh(\beta\hbar\omega/2)} , \qquad
\langle E\rangle = \frac{\hbar\omega}{2} +
\frac{\hbar\omega}{\eu^{\beta\hbar\omega} - 1} .
\]
At high $T$: $\langle E\rangle \to k_{\text{B}}T$ per oscillator,
$C \to 3Nk_{\text{B}}$ --- the \omterm{ex:b3:canonical-ensemble:einstein}{Dulong--Petit law} of the Year 1
volume, explained. At low $T$ the quantum $\hbar\omega$ becomes
unaffordable and $C$ collapses exponentially --- as measured, and
inexplicable classically. Einstein's 1907 curve, one parameter per
element, was the first application of quanta to ordinary matter;
diamond, with stiff bonds and light atoms ($\hbar\omega/k_{\text{B}}
\approx \qty{1300}{K}$), is still ``frozen'' at room temperature ---
the anomaly that had puzzled chemists for eighty years
(\cref{exo:b3:canonical-ensemble:6}).
\end{example}

\begin{theorem}[Equipartition, with its licence]\label{thm:b3:canonical-ensemble:equipartition}
\index{equipartition theorem}
Every coordinate or momentum entering the energy \emph{quadratically}
contributes, in the classical (high-temperature) regime,
\[
\langle\epsilon\rangle = \tfrac12 k_{\text{B}}T
\]
to the mean energy: $\tfrac32 k_{\text{B}}T$ for a monatomic gas
atom, $\tfrac52$ for a rotating diatomic, $3k_{\text{B}}T$ for an
oscillator. The licence expires when $k_{\text{B}}T$ falls below the
mode's level spacing: the mode freezes out and its contribution
vanishes --- the resolution of the nineteenth century's
heat-capacity scandals, drawn as the staircase above.
\end{theorem}

\begin{proof}[Partial proof]
For $\epsilon = ax^2$,
\[
\langle\epsilon\rangle
 = \frac{\int ax^2\,\eu^{-\beta ax^2}\dd x}
        {\int\eu^{-\beta ax^2}\dd x}
 = -\partial_\beta\ln\!\int\eu^{-\beta ax^2}\dd x
 = -\partial_\beta\ln\beta^{-1/2} = \frac{1}{2\beta} .
\] The freezing is
\cref{ex:b3:canonical-ensemble:einstein}'s computation, mode by
mode.
\end{proof}

\begin{example}[The gas, canonically]\label{ex:b3:canonical-ensemble:gas}
One atom in a box: $z = V/\lambda_T^3$ (the state count of
\cref{prop:b3:schrodinger-three-dimensions:counting} weighted by
Boltzmann); $N$ identical atoms: $Z = z^N/N!$. Then $F =
-Nk_{\text{B}}T[\ln(V/N\lambda_T^3) + 1]$, and differentiation
delivers $PV = Nk_{\text{B}}T$, the Sackur--Tetrode entropy, and
$\langle E\rangle = \tfrac32 Nk_{\text{B}}T$ --- the whole ideal gas
from one Gaussian integral. The Boltzmann factor applied to a
molecule's kinetic energy gives the Maxwell speed distribution of
the Year 1 volume, now derived; applied to its potential energy
$mgh$ it gives the exponential atmosphere --- and, in a drop of
water, Perrin's ladder of grains
(\cref{pb:b3:canonical-ensemble:1}).
\end{example}

\begin{method}[Canonical craft]\label{met:b3:canonical-ensemble:method}
(1) List the states and energies of \emph{one} unit; compute $z$.
(2) Factorise: $Z = z^N$ (or $z^N/N!$ for \omterm{def:b3:identical-particles:exchange}{identical particles}
sharing space); take $\ln$ early. (3) Differentiate: $\beta$ for
energy, $T$ for entropy via $F$, $V$ for pressure; a second
derivative for $C$ and fluctuations. (4) Check both ends: high $T$
must reproduce equipartition, low $T$ must freeze with an
$\eu^{-\Delta/k_{\text{B}}T}$ tail. (5) Competitions (folding,
binding, alignment): write $F = E - TS$ for each alternative and
let the smaller win --- the crossover sits at $T \approx \Delta
E/\Delta S$.
\end{method}

(a) Write the population ratio of two levels split by $\epsilon$ at

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:11}

\end{exercise}

\section{Problem: Counting Avogadro in a drop of water}

\begin{problem}[{Weekend problem --- Perrin's grains and the
reality of atoms}]\label{pb:b3:canonical-ensemble:1}
In 1908 Jean Perrin suspended microscopic resin grains in water,
let them settle, and counted them layer by layer under a
microscope. The exponential ladder he found was the Boltzmann
factor made visible --- and from its scale height he extracted
Avogadro's number, convincing the last sceptics that atoms exist.
Nobel Prize, 1926. Data: gamboge grains of radius $r =
\qty{0.212}{\micro m}$, density $\rho_{\text{g}} =
\qty{1207}{kg/m^3}$; water $\rho_{\text{w}} = \qty{999}{kg/m^3}$;
$T = \qty{293}{K}$; $g = \qty{9.81}{m/s^2}$.

\medskip\noindent\textbf{Part I --- The visible Boltzmann factor.}
\begin{enumerate}
  \item A grain in water feels gravity minus buoyancy: compute its
        \emph{effective} mass $m' = \tfrac43\pi r^3(\rho_{\text{g}}
        - \rho_{\text{w}})$ and weight.
  \item Write the equilibrium concentration profile $n(h)$ from
        the Boltzmann factor.
  \item Compute the scale height $h_0 = k_{\text{B}}T/m'g$ with
        the modern $k_{\text{B}}$.
  \item Why must the grains be so precisely mono-sized (how does
        $h_0$ depend on $r$)?
  \item Compare $h_0$ with the same formula for air molecules:
        why is the grains' atmosphere micrometres tall while the
        air's is kilometres?
  \item Perrin counted (in one run) relative concentrations
        $100 : 55 : 30 : 17$ at four equally spaced depths
        \qty{30}{\micro m} apart: check that this is an
        exponential ladder and extract its $h_0$.
\end{enumerate}

\medskip\noindent\textbf{Part II --- Weighing the invisible.}
\begin{enumerate}[resume]
  \item Invert: from the measured $h_0$ of item 6 and the known
        $m'g$, extract $k_{\text{B}}$.
  \item The gas constant $R = \qty{8.314}{J/(mol.K)}$ was known
        from macroscopic chemistry: combine with your
        $k_{\text{B}}$ to obtain Avogadro's number $N_{\text{A}} =
        R/k_{\text{B}}$.
  \item Perrin's runs gave $N_{\text{A}}$ between $5.5$ and
        $\num{7.2e23}$: compare with the modern
        \num{6.022e23} and comment on the achievement given his
        microscope and stopwatch.
  \item From $N_{\text{A}}$, compute the mass of a single
        \omterm{thm:b3:hydrogen-atom:levels}{hydrogen atom} --- the number the atomists had wanted for a
        century.
  \item Explain the logical structure: which \emph{macroscopic}
        measurements ($R$; grain size and density; the ladder)
        combine to weigh one atom, with no atom ever seen?
  \item Why do the grains --- $10^{10}$ atomic masses each ---
        obey the same Boltzmann statistics as molecules (what in
        the derivation of
        \cref{thm:b3:canonical-ensemble:boltzmann} cares about
        size)?
\end{enumerate}

\medskip\noindent\textbf{Part III --- The jitter that seals it.}
\begin{enumerate}[resume]
  \item The same grains jitter: Einstein's 1905 formula for
        Brownian motion gives $\langle x^2\rangle = 2Dt$ with $D =
        k_{\text{B}}T/6\pi\eta r$ (water: $\eta =
        \qty{e-3}{Pa.s}$). Compute $D$ for Perrin's grains.
  \item How far does a grain wander in one minute? Could Perrin
        measure it with a micrometer eyepiece and a stopwatch?
  \item Perrin verified $\langle x^2\rangle \propto t$ and
        extracted $k_{\text{B}}$ \emph{again}, independently:
        why did two unrelated routes (a static ladder; a dynamic
        jitter) to one number carry such evidential weight?
  \item The jitter is equipartition applied to the grain: each
        velocity component carries $\tfrac12 k_{\text{B}}T$.
        Estimate the grain's r.m.s.\ thermal speed ($m \approx
        \qty{4.8e-17}{kg}$).
  \item Why is that speed never seen directly (what interrupts
        the free flight after nanometres), and what \emph{is}
        seen instead?
  \item State which two chapters of this book meet in the
        observation: the mechanics of drag (Year 2 volume,
        viscosity) and the statistics of this chapter.
\end{enumerate}

\medskip\noindent\textbf{Part IV --- What was settled.}
\begin{enumerate}[resume]
  \item Ostwald and Mach had held atoms to be bookkeeping
        fictions: state in one sentence why a \emph{counted}
        $N_{\text{A}}$ from grain ladders ended that position.
  \item List three other 1900s routes that converged on the same
        $N_{\text{A}}$ (blue of the sky,
        \cref{exo:b3:microcanonical-ensemble:12}; electrolysis
        plus the electron charge; radioactivity's helium
        production) --- why did \emph{convergence} matter more
        than any single value?
