\chapter{Photons et phonons}\label{ch:b3:photons-phonons}

La théorie quantique est née du sujet de ce chapitre. En 1900, le
spectre des corps incandescents refusait d'obéir à la physique
classique --- pire, la physique classique prédisait une énergie
\emph{infinie} aux courtes longueurs d'onde, la «~catastrophe
ultraviolette~» --- et le remède désespéré de Planck, l'énergie par
paquets de $h\nu$, s'est révélé la clé maîtresse du siècle. Avec la
machinerie désormais assemblée --- le comptage des modes du
\cref{ch:b3:schrodinger-three-dimensions}, la thermodynamique de
l'oscillateur du \cref{ch:b3:canonical-ensemble} --- la \omterm{thm:b3:photons-phonons:planck}{loi de Planck}
tient en une demi-page~: un gaz de photons, un oscillateur occupé
selon Bose par mode de cavité. La même demi-page, avec du son à la
place de la lumière, donne la théorie des solides de Debye et la
capacité thermique en $T^3$ que le modèle d'Einstein manquait. Et
emplissant chaque centimètre cube de l'univers se trouve le
chef-d'œuvre du sujet~: le fond diffus cosmologique à 2,725 kelvins,
le corps noir le plus parfait jamais mesuré --- le rayonnement
thermique du jeune univers lui-même, qui nous parvient encore.

\section{La loi de Planck, démontrée}

\begin{theorem}[Le spectre du corps noir]\label{thm:b3:photons-phonons:planck}
\index{loi de Planck}\index{rayonnement du corps noir}
Une cavité à la température $T$ supporte des modes
électromagnétiques --- des ondes stationnaires comptées exactement
comme dans \cref{prop:b3:schrodinger-three-dimensions:counting}, deux
fois pour la polarisation~:
\[
g(\nu)\,\dd\nu = \frac{8\pi V}{c^3}\,\nu^2\,\dd\nu .
\]
Chaque mode est un oscillateur quantique portant, à l'équilibre,
$\bar n = 1/(\eu^{h\nu/k_{\text{B}}T} - 1)$ photons
(\cref{exo:b3:canonical-ensemble:5}~; de façon équivalente, des bosons à $\mu =
0$). L'énergie par unité de volume et de fréquence vaut donc
\[
u(\nu, T) = \frac{8\pi h\nu^3}{c^3}\,
\frac{1}{\eu^{h\nu/k_{\text{B}}T} - 1} :
\]
la \emph{\omterm{thm:b3:photons-phonons:planck}{loi de Planck}} (1900). Ses conséquences, toutes conformes à
la mesure~: le maximum en $\lambda_{\max}T = \qty{2898}{\micro m.K}$
(la loi du déplacement de Wien du volume de deuxième année,
maintenant démontrée)~; le flux total issu d'une surface noire
$\Phi = \sigma T^4$ avec
\[
\sigma = \frac{2\pi^5k_{\text{B}}^4}{15h^3c^2}
 = \qty{5.67e-8}{W.m^{-2}.K^{-4}}
\]
(Stefan--Boltzmann, sa constante désormais bâtie sur $h$, $c$,
$k_{\text{B}}$)~; et la densité de photons $n_\gamma \approx
\num{2.03e7}\,(T/\unit{K})^3$ par mètre cube.
\end{theorem}

\begin{proof}[Démonstration partielle]
Multiplions le comptage des modes par $h\nu\,\bar n$. L'intégrale
d'énergie totale, avec $x = h\nu/k_{\text{B}}T$ et $\int_0^\infty
x^3\dd x/(\eu^x - 1) = \pi^4/15$ (admis), donne $u_{\text{tot}}
= aT^4$ et, avec le facteur de flux standard $c/4$, le $\sigma$
annoncé. La loi de Wien est la maximisation de
\cref{exo:b3:photons-phonons:1}. À basse fréquence ($h\nu \ll
k_{\text{B}}T$) chaque mode porte $k_{\text{B}}T$ ---
Rayleigh--Jeans, correct pour la radio, catastrophique si on
l'étend à tout $\nu$~: la coupure de Planck, c'est le permis de
l'équipartition
(\cref{thm:b3:canonical-ensemble:equipartition}) révoqué mode par
mode.
\end{proof}

\begin{example}[Trois thermomètres]\label{ex:b3:photons-phonons:thermometers}
Le Soleil ($T = \qty{5800}{K}$)~: maximum à \qty{0.50}{\micro m} ---
nos yeux ont évolué sur son maximum. Un corps à \qty{310}{K}~:
maximum vers \qty{9.4}{\micro m}, rayonnant $\sigma T^4 \approx
\qty{520}{W/m^2}$ par mètre carré de peau (heureusement, la pièce
rayonne en retour~;
\cref{exo:b3:photons-phonons:2}). L'univers ($T =
\qty{2.725}{K}$)~: maximum vers \qty{1.9}{mm}, soit \qty{160}{GHz} en
fréquence, $411$ photons par centimètre cube --- le fond diffus
cosmologique, le grand exemple de ce chapitre
(\cref{pb:b3:photons-phonons:1}).
\end{example}

\begin{proposition}[Thermodynamique de la lumière]\label{prop:b3:photons-phonons:thermo}
\index{pression de radiation!thermique}
Le gaz de photons a une densité d'énergie $u = aT^4$ ($a =
4\sigma/c$), une pression
\[
P = \frac u3 ,
\]
une densité d'entropie $\tfrac43 aT^3$, et un nombre de photons
$\propto T^3$. Conséquences~: une boîte emplie de rayonnement qui se
dilate adiabatiquement refroidit comme $T \propto 1/L$ --- les
photons de l'univers en expansion se sont refroidis de $3000$ K aux
$2.725$ K d'aujourd'hui par un facteur d'étirement $\approx1100$ tout
en \emph{gardant} un spectre de Planck parfait~; et dans les étoiles
massives la croissance en $T^4$ permet à la pression de radiation de
rivaliser avec la pression du gaz, ce qui fixe une limite supérieure
aux masses stellaires (\cref{ch:b3:astrophysics}).
\end{proposition}

\begin{proof}[Démonstration partielle]
$P = u/3$~: un gaz isotrope de particules à la vitesse $c$ délivre
en pression le tiers de sa densité d'énergie (le facteur $\langle
\cos^2\theta\rangle = 1/3$, comme dans la théorie cinétique du
volume de première année). L'entropie vient de $\dd U = T\dd S - P\dd V$ appliqué à $U =
aT^4V$. Adiabatique~: $S \propto T^3V$ constant donne $T \propto
V^{-1/3}$.
\end{proof}

\section{Phonons~: l'achèvement de Debye}

\begin{theorem}[Le modèle de Debye]\label{thm:b3:photons-phonons:debye}
\index{modèle de Debye}\index{phonon}
Les vibrations d'un cristal sont des \omterm{prop:b3:continuum-elasticity:waves}{ondes élastiques} quantifiées
exactement comme la lumière~: des \emph{phonons}, trois
polarisations à la vitesse du son $c_{\text{s}}$, comptés comme des
photons mais avec un total plafonné à $3N$ modes --- ce plafond
définissant la \emph{température de Debye}
\[
k_{\text{B}}\theta_{\text{D}} = \hbar c_{\text{s}}
(6\pi^2n)^{1/3} .
\]
La capacité thermique interpole entre les deux lois classiques~:
\[
C \to 3Nk_{\text{B}} \;(T \gg \theta_{\text{D}})
\qquad\text{(Dulong--Petit)} ,
\qquad
C \to \frac{12\pi^4}{5}\,Nk_{\text{B}}
\Big(\frac{T}{\theta_{\text{D}}}\Big)^3
\;(T \ll \theta_{\text{D}}) :
\]
la loi en $T^3$ qu'exigeait la calorimétrie des basses températures
et que la fréquence unique d'Einstein ne pouvait donner
(\cref{exo:b3:canonical-ensemble:6}). La raison est la même que pour
la lumière~: si froid que soit le cristal, il existe des quanta
sonores de grande longueur d'onde arbitrairement bon marché, et leur
comptage à la manière des photons donne le $T^3$ des photons.
\end{theorem}

\begin{proof}[Démonstration partielle]
Refaites le calcul des photons avec $c \to c_{\text{s}}$, trois
polarisations, et le total de modes $\int g\,\dd\nu = 3N$ fixant la
coupure. À $T \ll \theta_{\text{D}}$ la coupure est invisible et
l'intégrale d'énergie en $T^4$ donne $C \propto T^3$~; à haute $T$,
chacun des $3N$ modes porte $k_{\text{B}}T$. L'intégrale
d'interpolation complète est admise.
\end{proof}

\begin{example}[Les phonons sont de vraies particules]\label{ex:b3:photons-phonons:phonon-real}
Les phonons diffusent les neutrons et les rayons X avec conservation
de l'énergie et de l'impulsion (cristalline) --- les courbes de
dispersion mesurées $\omega(\vect k)$ du
\cref{ch:b3:crystalline-solids} en sont la spectroscopie. Ils
portent la chaleur~: la conductivité thermique d'un isolant est le
$\kappa = \tfrac13 Cc_{\text{s}}\ell$ d'un gaz de phonons, et le
diamant --- rigide, léger, pur --- conduit cinq fois mieux que le
cuivre à température ambiante, par les seuls phonons
(\cref{exo:b3:photons-phonons:12}). Et ils s'éteignent à basse
température comme $T^3$, d'où les cristaux devenus
«~acoustiquement silencieux~» des cryogénistes.
\end{example}

\begin{method}[L'art du gaz de rayonnement]\label{met:b3:photons-phonons:method}
(1) Compter les modes ($\propto\nu^2$ par polarisation en 3D)~;
les occuper avec $1/(\eu^{\beta h\nu} - 1)$~; intégrer --- attendre
des énergies en $T^4$ et des nombres et capacités en $T^3$. (2) Les
maximums par Wien~; les totaux par Stefan~; le $k_{\text{B}}T$
classique par mode seulement au-dessous de
$h\nu \sim k_{\text{B}}T$. (3) Pour les solides, plafonner les modes
à $3N$ ($\theta_{\text{D}}$)~; la forme d'Einstein pour les modes de
branche optique, celle de Debye pour l'acoustique. (4) Rayonnement
adiabatique~: $T \propto
V^{-1/3}$. (5) Vérifier les deux constantes du métier~: $\lambda_{
\max}T = \qty{2898}{\micro m.K}$ et $\sigma =
\qty{5.67e-8}{W/m^2K^4}$.
\end{method}

\section{Exercices}

\begin{exercise}[$\star\star\star$]\label{exo:b3:photons-phonons:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:photons-phonons:12}

isolant~: $\kappa = \tfrac13 C_v c_{\text{s}}\ell$ par unité de
volume, avec $\ell$ le libre parcours moyen des phonons. (a) Pour
le diamant à \qty{300}{K} ($C_v \approx \qty{1.8e6}{J.m^{-3}K^{-1}}$,
donc partiellement gelé~;
$c_{\text{s}} = \qty{13000}{m/s}$~; $\ell

paraît froid sur la lèvre --- expliquer la sensation.
\end{exercise}

\section{Problème~: la plus ancienne lumière}

\begin{problem}[{Problème du week-end --- lire le fond diffus
cosmologique}]\label{pb:b3:photons-phonons:1}
Pointer n'importe quel récepteur millimétrique vers n'importe quelle
région du ciel et il rapporte le même signal~: un spectre de Planck
à $T =
\qty{2.72548(57)}{K}$, plus lisse que tout four jamais construit par
l'humanité. C'est l'éclair de la recombinaison de l'univers
(\cref{exo:b3:grand-canonical:12}), étiré onze cents fois, et il
porte le recensement du cosmos. Données~: $T_0 =
\qty{2.725}{K}$~; $n_\gamma = \num{2.03e7}\,T^3\,\unit{m^{-3}}$~;
$u = aT^4$, $a = \qty{7.56e-16}{J.m^{-3}.K^{-4}}$~; densité de
baryons aujourd'hui $n_{\text{b}} \approx \qty{0.25}{m^{-3}}$~;
densité critique $\rho_{\text{c}} \approx \qty{8.6e-27}{kg/m^3}$.

\medskip\noindent\textbf{Partie I --- Le spectre.}
\begin{enumerate}
  \item Calculer la longueur d'onde du maximum (Wien) du fond et sa
        densité de photons par centimètre cube.
  \item Calculer sa densité d'énergie, en $\unit{J/m^3}$ et en
        $\unit{eV/cm^3}$, et la fraction de la densité critique que
        représente son équivalent en masse.
  \item Le spectre de COBE (1990) collait à Planck à quelques
        parties sur $10^4$ --- l'auditoire a applaudi le graphique.
        Pourquoi un spectre \emph{thermique parfait} venu d'un ciel
        vide est-il si difficile à expliquer autrement que par un
        passé chaud et dense~?
  \item À l'émission, ces mêmes photons formaient un corps noir à
        \qty{3000}{K}~: vérifier le facteur d'étirement $1 + z =
        T_{\text{alors}}/T_0$.
  \item Pourquoi l'univers est-il devenu transparent précisément à
        cette température (rappeler quel équilibre du
        \cref{ch:b3:grand-canonical} s'est éteint)~?
  \item Le ciel nocturne est noir (le paradoxe d'Olbers de
        l'ancienne astronomie)~: en quel sens précis est-il en
        réalité \emph{éclatant} --- à quelle longueur d'onde et à
        quelle température~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Le recensement.}
\begin{enumerate}[resume]
  \item Calculer le rapport photons sur baryons $\eta^{-1} =
        n_\gamma/n_{\text{b}}$.
  \item Ce nombre --- environ deux milliards de photons par proton
        --- est l'une des données fondamentales de la cosmologie~:
        dites ce qu'il mesure de l'asymétrie matière--antimatière
        (rappeler \cref{exo:b3:grand-canonical:12}~: les photons
        sont surtout des produits d'annihilation).
  \item Comparer la densité d'énergie du fond à celle de la lumière
        des étoiles ($\sim\qty{2e-15}{J/m^3}$ moyennée sur
        l'univers)~: quelle lumière domine le cosmos~?
  \item Le fond l'emporte, en photons, sur tout ce que les étoiles
        ont jamais rayonné~: pourquoi cela rend-il l'univers
        primordial «~dominé par le rayonnement~», et qu'a pris le
        relais ensuite (les lois en $T^4$ contre $T^3$ face au
        $a^{-3}$ de la matière)~?
  \item Estimer l'époque de l'égalité matière--rayonnement~: avec
        une densité de rayonnement variant comme $(1+z)^4$ et une
        densité de matière comme $(1+z)^3$, et le rapport actuel
        $u_{\text{rad}}/\rho_{\text{m}}c^2 \approx 1/3400$,
        trouver $z_{\text{eq}}$.
  \item Les neutrinos se sont découplés plus tôt et ont refroidi un
        peu plus (ils ont manqué le réchauffement par annihilation
        $e^+e^-$ de \cref{exo:b3:grand-canonical:12})~: la
        prédiction $T_\nu = (4/11)^{1/3}T_\gamma$ --- l'évaluer,
        et s'émerveiller d'une prédiction de mer de neutrinos à
        \qty{1.95}{K} que personne n'a encore détectée directement.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Les anisotropies.}
\begin{enumerate}[resume]
  \item Le fond est plus chaud de \qty{3.4}{mK} dans une direction
        et plus froid à l'opposé~: interpréter par la physique
        Doppler du \cref{ch:b3:relativistic-kinematics} et extraire
        notre vitesse ($\Delta T/T = v/c$).
  \item Une fois ce dipôle retiré, il reste des rides de $\Delta T/T \sim
        \num{e-5}$ (COBE 1992, puis WMAP, Planck)~: que
        cartographient-elles à $z = 1100$~?
  \item Ces germes de densité à $10^{-5}$ sont devenus des
        galaxies~: pourquoi la gravité amplifie-t-elle les
        surdensités (quelle instabilité), et pourquoi la croissance
        a-t-elle dû attendre l'égalité matière--rayonnement pour
        démarrer vraiment~?
  \item La taille de ride la plus marquée (environ un degré sur le
        ciel) est une longueur d'onde acoustique --- du son dans le
        plasma photons--baryons (le gaz de ce chapitre plus les
        ondes du \cref{ch:b3:continuum-elasticity}~!)~: qu'est-ce
        qui fixe son échelle (la distance parcourue par le son
        avant la recombinaison)~?
  \item Mesurer cet angle face à l'échelle physique connue
        triangule la géométrie de l'univers~: énoncer le résultat
        (plate, au pour cent près).
  \item Le monopôle à \qty{2.7}{K}, le dipôle à \qty{3}{mK}, les
        rides à \num{e-5}~: classer ce que chacun a enseigné (un
        passé chaud~; notre mouvement~; les germes et la géométrie
        de tout).
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- L'héritage.}
\begin{enumerate}[resume]
  \item Penzias et Wilson ont trouvé le signal en 1965 comme un
        bruit d'antenne impossible à supprimer (après avoir expulsé
        les pigeons)~: pourquoi un excès \emph{isotrope} et
        indépendant de la saison à \qty{3}{K} était-il inexplicable
        par une source locale ou galactique~?
  \item Environ $1\%$ de la neige d'un téléviseur analogique non
        réglé était ce signal~: y réfléchir en une phrase.
  \item Le fond traverse les amas de galaxies et se fait légèrement
        pousser vers les hautes fréquences par les électrons chauds
        (l'effet Sunyaev--Zel'dovitch)~: pourquoi cela rend-il les
        amas visibles comme des \emph{ombres} à basse fréquence, et
        quelle est la vertu particulière d'une telle ombre
        (indépendante de la distance)~?
  \item La frontière d'aujourd'hui traque la polarisation du fond
        pour y trouver l'empreinte des ondes gravitationnelles
        primordiales~: quels chapitres de ce livre une telle
        découverte marierait-elle
        (la polarisation du \cref{ch:b3:covariant-electromagnetism},
        les ondes de la gravitation)~?
  \item Calculer le flux de photons du fond à travers votre corps
        ($n_\gamma c/4$ par unité d'aire, \omterm{def:b3:scattering-theory:cross-section}{section efficace} du corps
        $\sim\qty{0.5}{m^2}$)~: combien de vestiges du Big Bang
        vous traversent chaque seconde~?
  \item Le fond distingue un référentiel de repos (celui où le
        dipôle s'annule)~: expliquer pourquoi cela ne contredit
        \emph{pas} la relativité --- quel genre de référentiel est
        privilégié par le contenu matériel plutôt que par les
        lois~?
  \item Résumer le résultat nommé~: une courbe de Planck à $2.725$
        kelvins emplissant le ciel --- $411$ photons par centimètre
        cube, deux milliards par baryon, maximum à
        \qty{160}{GHz} --- est le reçu thermique du commencement
        chaud~: la physique statistique comme pierre de fondation
        de la cosmologie.
\end{enumerate}
\end{problem}
