\chapter{Électromagnétisme dans la matière}\label{ch:b3:electromagnetism-in-matter}

% Provenance (not for print): completes the "électromagnétisme dans la
% matière" module (dielectrics, magnetism, ferromagnetism) common to the
% surveyed third-year curricula; the Year 2 volume covered only wave
% propagation in media.

Glissez une feuille de plastique entre les armatures d'un condensateur
chargé et la tension chute, comme si de la charge était apparue de
nulle part. Enroulez une bobine autour d'un clou de fer et son champ
magnétique croît d'un facteur mille. Décollez un aimant décoratif du
réfrigérateur et il \emph{se souvient} du sens dans lequel il a été
aimanté --- pendant des décennies, sans aucune alimentation. Le volume
de deuxième année traitait les champs dans la matière en bloc, en
cachant le matériau dans un indice de réfraction~; ce chapitre ouvre
le matériau. Le plan est le même pour l'électricité et pour le
magnétisme~: la matière est une foule de dipôles microscopiques~;
les sommer en une densité ($\vect P$ ou $\vect M$)~; inventer un champ
auxiliaire ($\vect D$ ou $\vect H$) qui ne voit que les charges et les
courants que nous contrôlons~; puis descendre à l'échelle
microscopique --- là où les spins du \cref{ch:b3:spin-two-level}, le
facteur de Boltzmann du \cref{ch:b3:canonical-ensemble} et la
transition de champ moyen du \cref{ch:b3:phase-transitions} prennent
le relais --- pour calculer ce que fait le matériau. Le chapitre
s'achève à la casse, en concevant l'électroaimant qui soulève les
voitures.

\section{Polarisation et champ de déplacement}

\begin{definition}[Polarisation et charges liées]\label{def:b3:electromagnetism-in-matter:polarisation}
\index{polarisation}\index{charge liée}\index{diélectrique}
Un \emph{diélectrique} est un isolant dont les molécules acquièrent
(ou possèdent déjà) des moments dipolaires électriques. La réponse du
matériau se résume par la \emph{polarisation} $\vect P$~: le moment
dipolaire par unité de volume, $\vect P = n\langle\vect p\rangle$ pour
$n$ molécules par unité de volume. Un bloc polarisé porte des
\emph{charges liées} --- non libres de partir, mais parfaitement
réelles~: une densité surfacique
$\sigma_{\text{b}} = \vect P\cdot\vect n$ là où la polarisation
rencontre la surface, et une densité volumique $\rho_{\text{b}} =
-\nabla\cdot\vect P$ partout où elle n'est pas uniforme. Une lame
uniformément polarisée équivaut exactement à deux nappes de charge
$\pm P$ sur ses faces~: à l'intérieur, toutes les têtes et queues de
dipôles se compensent.
\end{definition}

\begin{proposition}[Le champ de déplacement]\label{prop:b3:electromagnetism-in-matter:displacement}
\index{déplacement électrique}\index{susceptibilité électrique}\index{permittivité relative}
La charge totale est libre plus liée~; le théorème de Gauss avec $\rho =
\rho_{\text{f}} + \rho_{\text{b}}$ se réarrange en une loi pour le
\emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{déplacement électrique}} $\vect D = \varepsilon_0\vect E +
\vect P$~:
\[
\nabla\cdot\vect D = \rho_{\text{f}} :
\]
les sources de $\vect D$ sont les seules charges libres --- celles de
nos armatures et de nos fils. Dans un \omterm{def:b3:electromagnetism-in-matter:polarisation}{diélectrique} \emph{linéaire} la
réponse est proportionnelle,
$\vect P = \varepsilon_0\chi_{\text{e}}\vect E$, donc
$\vect D = \varepsilon_0\varepsilon_{\text{r}}\vect E$ avec
$\varepsilon_{\text{r}} = 1 + \chi_{\text{e}}$ la \emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{permittivité
relative}}~: l'air $1.0006$, le polyéthylène $2.3$, le verre
$\sim 5$, l'eau un énorme $80$. Entre des armatures maintenues à
charge fixe, $\vect D$ est inchangé par le \omterm{def:b3:electromagnetism-in-matter:polarisation}{diélectrique}, si bien que $\vect E =
\vect D/\varepsilon_0\varepsilon_{\text{r}}$ \emph{chute} d'un facteur
$\varepsilon_{\text{r}}$~: les \omterm{def:b3:electromagnetism-in-matter:polarisation}{charges liées} de surface font face aux
charges libres et annulent l'essentiel de leur champ. La capacité est
multipliée par $\varepsilon_{\text{r}}$ --- toute l'industrie du
condensateur en une lettre grecque.
\end{proposition}

\begin{proof}
$\nabla\cdot(\varepsilon_0\vect E) = \rho_{\text{f}} +
\rho_{\text{b}} = \rho_{\text{f}} - \nabla\cdot\vect P$~; faites
passer le terme de \omterm{def:b3:electromagnetism-in-matter:polarisation}{polarisation} à gauche.
\end{proof}

\section{D'où vient la permittivité}

\begin{proposition}[Polarisation induite et Clausius--Mossotti]\label{prop:b3:electromagnetism-in-matter:mossotti}
\index{polarisabilité}\index{relation de Clausius--Mossotti}
Un atome dans un champ $\vect E_{\text{loc}}$ s'étire en un dipôle
$\vect p = \alpha\vect E_{\text{loc}}$, avec $\alpha$ la
\emph{polarisabilité} (dimension~: $\varepsilon_0\times$volume ---
environ $\varepsilon_0$ fois le volume atomique). Dans un gaz dilué
$E_{\text{loc}} \approx E$ et $\chi_{\text{e}} =
n\alpha/\varepsilon_0$ est petit. Dans un milieu dense, chaque
molécule ressent aussi ses voisines polarisées~: creuser autour d'elle
une petite cavité sphérique donne $\vect E_{\text{loc}} = \vect E + \vect
P/3\varepsilon_0$, et l'auto-cohérence livre la
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{relation de Clausius--Mossotti}}
\[
\frac{\varepsilon_{\text{r}} - 1}{\varepsilon_{\text{r}} + 2}
= \frac{n\alpha}{3\varepsilon_0} :
\]
mesurer la minuscule susceptibilité d'une vapeur et prédiser celle du
liquide --- l'accord est de quelques pour cent pour les liquides
apolaires.
\end{proposition}
\admitted

\begin{remark}
Le champ de cavité $\vect P/3\varepsilon_0$ est le résultat de la
sphère uniformément polarisée, cité ici sans démonstration~; la
dérivation honnête, par conditions aux limites, relève d'un cours
d'électrodynamique dédié. Tout le reste ci-dessus est de la
comptabilité.
\end{remark}

\begin{proposition}[Molécules polaires~: la fonction de Langevin]\label{prop:b3:electromagnetism-in-matter:langevin}
\index{fonction de Langevin}\index{polarisation d'orientation}
Une molécule à dipôle \emph{permanent} $p$ (l'eau~:
$p = \qty{6.2e-30}{C.m}$) n'a pas besoin d'être étirée --- seulement
alignée, contre l'agitation thermique. La moyenne canonique du
\cref{ch:b3:canonical-ensemble} sur les orientations, avec l'énergie
$-pE\cos\theta$, donne
\[
\langle\cos\theta\rangle = L(x) = \coth x - \frac1x ,
\qquad x = \frac{pE}{k_{\text{B}}T} ,
\]
la \emph{\omterm{prop:b3:electromagnetism-in-matter:langevin}{fonction de Langevin}}~: linéaire ($L \approx x/3$) aux petits
$x$, saturant à $1$ quand le champ l'emporte franchement. Dans le
régime linéaire $\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$
--- une loi de Curie électrique, décroissant en $1/T$~: chauffer un
liquide polaire et sa permittivité \emph{baisse}, signature qui
distingue les dipôles permanents des dipôles induits. Pour l'eau à
température ambiante, cela prédit $\chi_{\text{e}} \sim 12$ --- le bon
ordre de grandeur, la coopération des liaisons hydrogène poussant la
valeur réelle à $80$.
\end{proposition}

\begin{proof}[Démonstration partielle]
$Z = \int_0^\pi \eu^{x\cos\theta}\,2\pi\sin\theta\,\dd\theta \propto
\sinh x/x$, et $\langle\cos\theta\rangle = \dd\ln Z/\dd x = \coth x
- 1/x$. Développer $\coth x = 1/x + x/3 - \dots$ aux petits $x$ donne
$L \approx x/3$, d'où $P = np\langle\cos\theta\rangle =
np^2E/3k_{\text{B}}T$.
\end{proof}

\section{Aimantation et champ \texorpdfstring{$H$}{H}}

\begin{definition}[Aimantation, courants liés et H]\label{def:b3:electromagnetism-in-matter:magnetisation}
\index{aimantation}\index{champ magnétique H}\index{susceptibilité magnétique}
L'histoire magnétique est parallèle. La matière est pleine de boucles
de courant microscopiques --- des électrons en orbite et en rotation
propre, chacun un dipôle magnétique $\vect\mu$~; leur densité est
l'\emph{aimantation} $\vect M = n\langle\vect\mu\rangle$. Un bloc
uniformément aimanté équivaut à un \emph{courant lié de surface}
circulant autour de ses flancs (les boucles intérieures se compensent
deux à deux~; un aimant droit \emph{est} un solénoïde de courant lié).
En séparant les courants en libres et liés, le théorème d'Ampère se
réarrange autour du champ auxiliaire
\[
\vect H = \frac{\vect B}{\mu_0} - \vect M , \qquad
\oint \vect H\cdot\dd\vect\ell = I_{\text{libre}} :
\]
$\vect H$ a pour sources les courants de nos fils (unités
\unit{A/m}), tandis que $\vect B$ garde la physique --- forces, flux,
induction. Les milieux linéaires ont $\vect M = \chi_{\text{m}}\vect H$
et $\vect B = \mu_0(1 + \chi_{\text{m}})\vect H =
\mu_0\mu_{\text{r}}\vect H$.
\end{definition}

\begin{remark}[Trois tempéraments magnétiques]\label{rem:b3:electromagnetism-in-matter:zoo}
\index{diamagnétisme}\index{paramagnétisme}\index{ferromagnétisme}
Les matériaux répondent à $\vect H$ de trois façons.
Les \emph{diamagnétiques} ($\chi_{\text{m}} \sim -10^{-5}$~: eau,
cuivre, graphite) répondent comme une loi de Lenz rendue permanente~:
le champ appliqué perturbe chaque orbite électronique de façon à s'y
opposer --- faible, universel, indépendant de la température, et
repoussé par les aimants (une grenouille a été ainsi lévitée dans un
alésage de \qty{16}{T}).
Les \emph{paramagnétiques} ($\chi_{\text{m}} \sim +10^{-5}$ à
$10^{-3}$~: aluminium, O$_2$) possèdent des moments permanents --- des
spins non appariés du \cref{ch:b3:identical-particles} --- qui
s'alignent comme les dipôles de Langevin~:
$\chi_{\text{m}} = \mu_0 n\mu^2/3k_{\text{B}}T$, une loi de Curie en
$1/T$ (la version quantique à deux niveaux est
\cref{exo:b3:canonical-ensemble:11}).
Les \emph{ferromagnétiques} (fer, cobalt, nickel~: $\mu_{\text{r}}$
par milliers) sont des paramagnétiques dont les spins se parlent aussi
--- le couplage d'échange de \cref{exo:b3:identical-particles:12} ---
et au-dessous de la \omterm{thm:b3:phase-transitions:meanfield}{température de Curie} ils s'alignent
\emph{spontanément}~: \cref{thm:b3:phase-transitions:meanfield}, mise
au travail.
\end{remark}

\section{Le ferromagnétisme à l'œuvre~: domaines et hystérésis}

\begin{definition}[Domaines et hystérésis]\label{def:b3:electromagnetism-in-matter:hysteresis}
\index{domaine magnétique}\index{hystérésis}\index{rémanence}\index{coercitivité}
Un bloc de fer brut n'est \emph{pas} aimanté~: il se brise en
\emph{domaines}, régions de taille micrométrique chacune pleinement
aimantée mais pointant différemment, si bien que le champ extérieur
(et son coût en énergie) s'annule presque. Un $\vect H$ appliqué
déplace les parois de domaine --- les domaines favorables croissent
--- puis fait tourner des domaines entiers dans l'alignement. Les
parois s'accrochent aux défauts, si bien que le processus est
irréversible~: balayer $H$ de haut en bas et $B(H)$ trace une boucle,
le \emph{cycle d'hystérésis}. Coupez le courant et un champ
\emph{rémanent} $B_{\text{r}}$ survit~; l'annuler demande le champ
\emph{coercitif} inverse $H_{\text{c}}$. L'aire enfermée par la boucle
est l'énergie dissipée par cycle et par unité de volume. Les
matériaux \emph{doux} (acier au silicium, ferrites~: boucle mince,
petit $H_{\text{c}}$) font les noyaux de transformateurs et de
moteurs~; les \emph{durs} (alnico, Nd$_2$Fe$_{14}$B~: boucle grasse,
$H_{\text{c}}$ énorme) font les aimants permanents --- et la mémoire
magnétique~: l'aimant du réfrigérateur et le disque dur sont des
cycles d'hystérésis qui refusent d'oublier.
\end{definition}

\begin{example}[L'électroaimant à noyau de fer]\label{ex:b3:electromagnetism-in-matter:electromagnet}
Enroulez $N$ spires parcourues par $I$ autour d'un anneau de fer ($\mu_{\text{r}}
\sim 5000$) percé d'un petit entrefer $e$. Le théorème d'Ampère pour
$\vect H$ le long de la boucle donne
$H_{\text{fer}}\ell + H_{\text{gap}}e = NI$, et la continuité du flux
rend $B$ commun, donc
\[
B\Big(\frac{\ell}{\mu_0\mu_{\text{r}}} + \frac{e}{\mu_0}\Big) = NI .
\]
Avec $\ell = \qty{1}{m}$ et $e = \qty{1}{cm}$, le terme d'entrefer
vaut cinquante fois le terme de fer~: presque tout l'effort de la
bobine sert à pousser le champ à travers un centimètre d'air. Voilà le
circuit magnétique en une ligne --- le fer est un conducteur de flux
presque parfait, l'air la résistance --- et c'est pourquoi moteurs,
relais et grues de casse gardent leurs entrefers impitoyablement
minces (\cref{pb:b3:electromagnetism-in-matter:1}).
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:1}
Comptabilité des \omterm{def:b3:electromagnetism-in-matter:polarisation}{charges liées}. Une lame d'épaisseur $d$ est
uniformément polarisée selon sa normale, d'amplitude $P$. (a) Donner
les densités de \omterm{def:b3:electromagnetism-in-matter:polarisation}{charge liée} sur chaque face. (b) Montrer que le champ
\emph{intérieur} dû à ces nappes vaut $E = P/\varepsilon_0$, opposé à
$\vect P$. (c) Que vaut $\vect D$ à l'intérieur, en l'absence de toute
charge libre~? (d) Un électret droit ($\vect P$ figé) est l'analogue
électrique de quel objet magnétique~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:2}
Condensateur à \omterm{def:b3:electromagnetism-in-matter:polarisation}{diélectrique}. Un condensateur plan
($C_0 = \qty{100}{pF}$) est chargé sous \qty{12}{V} puis déconnecté.
Une lame de $\varepsilon_{\text{r}} = 4$ le remplit. Trouver (a) la
nouvelle capacité~; (b) la nouvelle tension~; (c) l'énergie avant et
après --- où est passée la différence~? (d) Reprendre (b)–(c) avec la
pile restée branchée.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:3}
Le zoo des susceptibilités. Classer (dia-, para- ou ferromagnétique)
et justifier par la structure électronique quand vous le pouvez~:
l'eau ($\chi_{\text{m}} = -9\times10^{-6}$), l'aluminium
($+2.2\times10^{-5}$), l'oxygène liquide ($+3.5\times10^{-3}$ ---
pourquoi O$_2$ est-il magnétique~?), le nickel
($\mu_{\text{r}} \sim 600$ en champ faible). Lequel des quatre un
aimant puissant attirerait-il visiblement, et lequel
repousserait-il très légèrement~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:4}
$B$, $H$, $M$. Un long solénoïde ($n = \num{2000}$ spires par mètre,

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:5}
Câble coaxial, isolé. Une ligne coaxiale (rayon intérieur $a$,
extérieur $b$) porte une charge libre $\lambda$ par unité de longueur

deux. (a) Trouver $\vect D(r)$ par symétrie. (b) En déduire $\vect E$
et $\vect P$. (c) Calculer les densités de \omterm{def:b3:electromagnetism-in-matter:polarisation}{charge liée} en $r = a$ et
$r = b$ et vérifier qu'elles se somment à zéro par unité de longueur.
(d) Montrer que la capacité par unité de longueur est multipliée par
$\varepsilon_{\text{r}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:6}
Clausius--Mossotti à l'œuvre. L'argon gazeux à \qty{1}{bar} et
\qty{300}{K} a $\varepsilon_{\text{r}} = 1.00052$. (a) Extraire
$\alpha/\varepsilon_0$ (un volume) et comparer au volume atomique.
(b) L'argon liquide a $n = \qty{2.1e28}{m^{-3}}$~: prédiser
$\varepsilon_{\text{r}}$ par Clausius--Mossotti (mesuré~: 1,53).
(c) Pourquoi le naïf $\chi = n\alpha/\varepsilon_0$ surestime-t-il
moins gravement ici que pour l'eau~? (d) Pour quelles molécules tout
le schéma doit-il échouer, et qu'est-ce qui le remplace~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:7}
La permittivité géante de l'eau. (a) À partir de
\cref{prop:b3:electromagnetism-in-matter:langevin}, calculer
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ pour l'eau

$\sim\qty{7e7}{V/m}$. (c) Les fours à micro-ondes fonctionnent à
\qty{2.45}{GHz}~: pourquoi une réponse \emph{dissipative} de dipôles

chauffe-t-elle bien plus lentement que l'eau liquide~? (d) Prédiser le
signe de $\dd\varepsilon_{\text{r}}/\dd T$ pour l'eau et pour

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:8}
Le diamagnétisme estimé. Traiter un électron atomique comme une charge
sur un anneau de rayon $r$~; un $B$ appliqué change sa pulsation du

le moment induit vaut $\Delta\mu = -e^2r^2B/4m_{\text{e}}$ (opposé à
$B$). (b) En moyennant sur les orientations et en sommant $Z$

$-9\times10^{-6}$ mesuré. (c) Pourquoi le diamagnétisme est-il
indépendant de la température alors que le paramagnétisme ne l'est
pas~? (d) Pourquoi même une grenouille lévite-t-elle dans
\qty{16}{T}, et pourquoi le champ de l'aimant doit-il être
\emph{non uniforme}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:9}

\emph{para}magnétique plutôt qu'un ferromagnétique~?

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:11}
Pertes par \omterm{def:b3:electromagnetism-in-matter:hysteresis}{hystérésis}. Un noyau de transformateur (volume

\section{Problème~: la grue de casse}

\begin{problem}\label{pb:b3:electromagnetism-in-matter:1}
\emph{La grue de casse.} Une déchetterie commande un électroaimant de
grue~: un pot de fer à face plate, d'un mètre de diamètre, qui doit
soulever des balles de voitures compressées de deux tonnes, puis ---
tout aussi important --- les \emph{lâcher} sur commande. Vous êtes
l'ingénieur de conception.

\medskip\noindent\textbf{Partie I --- Pourquoi le fer.}

\begin{enumerate}
  \item La bobine nue~: 400 spires parcourues par \qty{25}{A}
        réparties sur un trajet magnétique de $\sim\qty{1}{m}$.
        Estimer $B \sim
        \mu_0NI/\ell$ sans fer.
  \item Expliquer en deux phrases, avec les domaines, pourquoi
        remplir la bobine de fer multiplie cela par
        $\sim\mu_{\text{r}}$.
  \item L'\omterm{def:b3:electromagnetism-in-matter:magnetisation}{aimantation} du fer sature à $M_{\text{s}} \approx
        \qty{1.7e6}{A/m}$. Calculer le plafond $\mu_0M_{\text{s}}$
        que cela impose au champ des pôles, quoi que fasse la
        bobine.
  \item Chaque atome de fer apporte $\approx 2.2$ \omterm{prop:b3:quantum-angular-momentum:magnetic}{magnétons de
        Bohr}. Vérification~: avec $n = \qty{8.5e28}{m^{-3}}$,
        retrouver $M_{\text{s}}$.
  \item Pourquoi les faces polaires doivent-elles être usinées
        planes et tenues à l'abri de la rouille et des gravillons~?
        (Penser à
        \cref{ex:b3:electromagnetism-in-matter:electromagnet}.)
  \item La charge elle-même devient partie du circuit magnétique.
        Esquisser le trajet du flux~: noyau du pot, face nord,
        balle d'acier, face sud, retour par la culasse.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Le circuit magnétique et la force.}

\begin{enumerate}[resume]
  \item Modéliser le circuit~: trajet de fer $\ell = \qty{1.2}{m}$,
        $\mu_{\text{r}} = 2000$, et deux entrefers effectifs
        (contact face--balle) de $e = \qty{1.5}{mm}$ chacun.
        Écrire le théorème d'Ampère pour $H$ le long de la boucle.
  \item Calculer les trois termes de réluctance par unité d'aire
        ($\ell/\mu_{\text{r}}$ contre $2e$) et montrer que les
        entrefers millimétriques consomment tout de même
        l'essentiel de l'effort de la bobine.
  \item Avec $NI = \num{10000}$ ampères-tours, calculer $B$ dans
        les entrefers.
  \item La pression de levage sur chaque face polaire vaut
        $B^2/2\mu_0$ (densité d'énergie magnétique libérée par
        mètre de rapprochement). L'évaluer en \unit{kPa} pour
        votre $B$.
  \item Aire polaire totale $A = \qty{0.12}{m^2}$~: calculer la
        force de levage et la convertir en tonnes.
  \item Satisfait-elle la spécification de deux tonnes avec une
        marge d'un facteur deux~? Sinon, ajuster $NI$ et donner le
        nouveau courant.
  \item Les balles compressées ne touchent les faces que sur un
        tiers de leur aire environ, avec des entrefers effectifs
        plus larges. Recalculer la force pour
        $e = \qty{4}{mm}$, $A_{\text{eff}} = \qty{0.04}{m^2}$ et
        commenter pourquoi les capacités nominales sont citées en
        «~plaque plane~».
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Se souvenir et oublier.}

\begin{enumerate}[resume]
  \item La déchetterie coupe l'interrupteur pour lâcher une balle
        --- et elle reste accrochée. Nommer le coupable, avec le
        \omterm{def:b3:electromagnetism-in-matter:hysteresis}{cycle d'hystérésis}.
  \item Pourquoi le pot doit-il être en fer \emph{doux} (petit
        $H_{\text{c}}$, petit $B_{\text{r}}$) plutôt qu'en acier
        à aimants dur~?
  \item Même le fer doux garde un peu de \omterm{def:b3:electromagnetism-in-matter:hysteresis}{rémanence}. Proposer le
        remède standard~: une brève impulsion de courant
        \emph{inversé} --- quel point de la boucle vise-t-elle~?
  \item Certains contrôleurs appliquent plutôt un courant
        alternatif décroissant. Esquisser ce que fait la
        trajectoire $B$--$H$ et pourquoi elle finit désaimantée.
  \item La grue manipule aussi des brames chaudes sortant d'un
        four, à \qty{800}{\celsius}. Le point de Curie du fer vaut
        \qty{1043}{K}~: qu'advient-il de la force de levage, et
        pourquoi~? (\cref{thm:b3:phase-transitions:meanfield}.)
  \item Un apprenti d'été propose d'économiser du cuivre en
        doublant l'entrefer et en doublant $NI$. Utiliser vos
        formules de la partie II pour expliquer l'asymétrie~:
        qu'est-ce qui est divisé par deux et qu'est-ce qui tient
        seulement bon~?
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- Puissance, chaleur et facture.}

\begin{enumerate}[resume]
  \item La bobine~: 400 spires de cuivre, longueur moyenne de
        spire \qty{2.5}{m}, section de fil \qty{16}{mm^2},
        résistivité $\rho = \qty{1.7e-8}{\Omega.m}$. Calculer sa
        résistance.
  \item À $I = \qty{25}{A}$~: la puissance dissipée, et l'énergie
        quotidienne pour un poste de huit heures à
        \qty{60}{\percent} de service.
  \item L'énergie magnétique stockée dans les deux entrefers
        ($B^2/2\mu_0 \times$ volume)~: la calculer et comparer à
        \emph{une} seconde de dissipation dans la bobine. Où passe
        tout le reste de l'énergie électrique~?
  \item Lâcher la charge ne rend presque rien de l'énergie stockée
        au réseau. Expliquer, avec l'aire de la boucle et la
        surtension inductive, pourquoi le contrôleur a besoin d'un
        chemin de roue libre aux bornes de la bobine.
  \item L'alimentation tombe avec une balle en l'air. Que fait la
        balle, et que retient la \omterm{def:b3:electromagnetism-in-matter:hysteresis}{rémanence} \emph{seule}~? Justifier
        la règle de la déchetterie selon laquelle nul ne passe sous
        un aimant sous tension --- et l'argument de vente des
        grues à secours par batterie.
  \item Résumer la conception en quatre lignes~: plafond de champ
        fixé par $M_{\text{s}}$, force par $B^2A/2\mu_0$,
        contrôlabilité par l'\omterm{def:b3:electromagnetism-in-matter:hysteresis}{hystérésis} du fer doux, coût
        d'exploitation par $RI^2$ --- tout le chapitre suspendu à
        une grue.
\end{enumerate}
\end{problem}
