\chapter{Le spin et les systèmes à deux niveaux}\label{ch:b3:spin-two-level}

En 1922, Stern et Gerlach envoyèrent un jet d'atomes d'argent à
travers un aimant dissymétrique, s'attendant soit à aucune déviation,
soit à une tache continue. Le jet se scinda nettement en \emph{deux}.
Aucun moment cinétique orbital ne peut faire cela~: $2l + 1$ est
toujours impair. Les atomes annonçaient une possession nouvelle et
purement quantique --- le \emph{\omterm{def:b3:spin-two-level:spin}{spin}}, un moment cinétique intrinsèque
de $j = \tfrac12$, le barreau demi-entier que l'algèbre du
\cref{ch:b3:quantum-angular-momentum} tenait prêt et que les orbites ne
pouvaient jamais employer. Le \omterm{def:b3:spin-two-level:spin}{spin} double chaque état électronique (et
complète ainsi le compte derrière le tableau périodique), porte le
magnétisme du fer et du proton, et constitue le système à deux niveaux
le plus net que la nature offre~: la physique de ce chapitre fait
tourner l'appareil de \omterm{prop:b3:spin-two-level:rabi}{résonance magnétique} de chaque hôpital,
l'horloge au césium qui définit la seconde, le murmure à
21 centimètres par lequel on cartographie les galaxies, et le qubit.

\section{L'expérience qui l'a découvert}

\begin{example}[Stern--Gerlach]\label{ex:b3:spin-two-level:stern-gerlach}
Un moment magnétique $\vect\mu$ dans un champ \emph{inhomogène} subit
la force $F_z = \mu_z\,\partial B_z/\partial z$~: la déviation mesure
$\mu_z$. Attente classique~: des moments orientés au hasard, un
éventail continu. Attente quantique pour des moments orbitaux~:
$2l + 1$ taches~: une, trois, cinq --- toujours un nombre impair.
Observé pour l'argent (et pour l'hydrogène)~: \emph{deux} taches,
symétriques, rien entre elles. La composante mesurée ne prend
exactement que deux valeurs --- la signature de $j
= \tfrac12$, interdite aux orbites, et l'exposition directe de la
quantification~: l'appareil est un instrument de mesure d'une
composante du \omterm{def:b3:spin-two-level:spin}{spin}, et le jet se scinde selon ses deux \omterm{def:b3:quantum-formalism:observable}{valeurs propres}.
\end{example}

\begin{definition}[Le spin un demi]\label{def:b3:spin-two-level:spin}
\index{spin}\index{matrices de Pauli}\index{spineur}
L'électron (comme le proton, le neutron, les quarks) porte un moment
cinétique intrinsèque de $j = \tfrac12$~: un espace d'états à deux
dimensions engendré par $\ket\uparrow, \ket\downarrow$ (états propres
de $\hat S_z$ pour $\pm\hbar/2$), sur lequel
\[
\hat{\vect S} = \frac\hbar2\,\hat{\vect\sigma} , \qquad
\sigma_x = \begin{pmatrix}0&1\\1&0\end{pmatrix} ,\;
\sigma_y = \begin{pmatrix}0&-\iu\\\iu&0\end{pmatrix} ,\;
\sigma_z = \begin{pmatrix}1&0\\0&-1\end{pmatrix}
\]
--- les \emph{matrices de Pauli}, qui réalisent l'algèbre du moment
cinétique dans son plus petit logement possible. Un état général
$\alpha\ket\uparrow + \beta\ket\downarrow$ est un \emph{spineur}. Le
moment magnétique de l'électron vaut
\[
\hat{\vect\mu} = -g\,\frac{\mu_{\text{B}}}{\hbar}\,\hat{\vect S} ,
\qquad g \approx 2 :
\]
le double du taux orbital par unité de moment cinétique --- l'anomalie
qu'Einstein et de Haas avaient mesurée
(\cref{exo:b3:quantum-angular-momentum:12}) et que l'équation de Dirac
allait prédire. Le spin n'est la rotation de rien~: aucun rayon,
aucune «~boule qui tourne~» ne résiste à l'examen --- il est
intrinsèque, comme la charge.
\end{definition}

\begin{example}[Filtres de Stern--Gerlach en cascade]\label{ex:b3:spin-two-level:chained}
Sélectionnez le faisceau $S_z = +\hbar/2$ et mesurez $S_z$ de
nouveau~: tous les atomes répondent $+$. Mesurez plutôt $S_x$~:
moitié-moitié --- l'état $\ket\uparrow$ est la superposition
$(\ket{+x} + \ket{-x})/\sqrt2$. Gardez maintenant le faisceau
$S_x = +$ et mesurez $S_z$ une fois encore~: moitié-moitié
\emph{de nouveau} --- la mesure de $S_x$ a effacé le $S_z$ auparavant
bien défini. Trois aimants suffisent à exhiber incompatibilité,
réduction et \omterm{thm:b3:quantum-formalism:postulates}{règle de Born}~; cette cascade est
\cref{ex:b3:quantum-formalism:threepolarizers} jouée avec des atomes.
\end{example}

\section{La sphère de Bloch}

\begin{proposition}[Le spin selon un axe quelconque]\label{prop:b3:spin-two-level:bloch}
\index{sphère de Bloch}
Pour la direction unitaire $\vect n = (\sin\theta\cos\varphi,
\sin\theta\sin\varphi, \cos\theta)$, l'opérateur $\vect
n\cdot\hat{\vect\sigma}$ a pour \omterm{def:b3:quantum-formalism:observable}{valeurs propres} $\pm1$, avec
\[
\ket{+\vect n} = \cos\tfrac\theta2\,\ket\uparrow +
\eu^{\iu\varphi}\sin\tfrac\theta2\,\ket\downarrow .
\]
Tout \omterm{def:b3:spin-two-level:spin}{spineur} pur est $\ket{+\vect n}$ pour exactement une direction~:
l'espace des états s'envoie sur une sphère --- la \emph{\omterm{prop:b3:spin-two-level:bloch}{sphère de
Bloch}} --- où les états orthogonaux sont \emph{antipodaux} (noter les
demi-angles~: ce sont les directions opposées, non les
perpendiculaires, qui sont orthogonales). Les rotations de la sphère
sont exactement les évolutions qu'engendre un champ magnétique~: la
géométrie de toute manipulation de qubit jamais réalisée.
\end{proposition}

\begin{proof}[Démonstration partielle]
Diagonalisons la matrice $2\times2$ $\vect n\cdot\vect\sigma =
\begin{pmatrix}\cos\theta & \sin\theta\,\eu^{-\iu\varphi}\\
\sin\theta\,\eu^{\iu\varphi} & -\cos\theta\end{pmatrix}$~: trace $0$,
déterminant $-1$, \omterm{def:b3:quantum-formalism:observable}{valeurs propres} $\pm1$~; le \omterm{def:b3:spin-two-level:spin}{spineur} donné se
vérifie par substitution (identités de l'angle moitié). Antipodes~: $\theta \to \pi -
\theta$, $\varphi \to \varphi + \pi$ donne $\braket{+\vect
n'}{+\vect n} = 0$.
\end{proof}

\section{Le spin dans un champ~: précession et résonance}

\begin{proposition}[Précession de Larmor]\label{prop:b3:spin-two-level:larmor}
\index{précession de Larmor}\index{rapport gyromagnétique}
Dans un champ statique $\vect B_0 = B_0\vect e_z$, le \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltonien} $\hat
H = -\hat{\vect\mu}\cdot\vect B_0$ sépare les deux niveaux de
$\hbar\omega_0$ et fait \emph{précesser} le \omterm{def:b3:spin-two-level:spin}{spin} moyen autour du
champ~:
\[
\frac{\dd\langle\hat{\vect S}\rangle}{\dd t} =
\gamma\,\langle\hat{\vect S}\rangle\wedge\vect B_0 , \qquad
\omega_0 = |\gamma|B_0 ,
\]
avec le \omterm{prop:b3:spin-two-level:larmor}{rapport gyromagnétique} $\gamma$ ($\mu = \gamma S$). Le vecteur
de Bloch tourne autour de $z$ à $\omega_0$, son angle avec le champ
restant figé --- exactement l'image classique du gyroscope, exacte ici
parce que les \omterm{def:b3:quantum-formalism:commutator}{commutateurs} sont linéaires. Pour l'électron~:
$\qty{28}{GHz/T}$~; pour le proton (moment $2.79$ magnétons nucléaires)~: $\gamma_p/2\pi =
\qty{42.6}{MHz/T}$ --- la plaque d'immatriculation de toute IRM.
\end{proposition}

\begin{proof}
Appliquons \cref{prop:b3:quantum-formalism:evolution} avec $\hat H =
-\gamma B_0\hat S_z$~: les \omterm{def:b3:quantum-formalism:commutator}{commutateurs} donnent $\dd\langle\hat
S_x\rangle/\dd t = -\gamma B_0\langle\hat S_y\rangle$,
$\dd\langle\hat S_y\rangle/\dd t = +\gamma B_0\langle\hat
S_x\rangle$ et $\dd\langle\hat S_z\rangle/\dd t = 0$, qui
s'assemblent en le produit vectoriel annoncé.
\end{proof}

\begin{proposition}[Résonance magnétique]\label{prop:b3:spin-two-level:rabi}
\index{résonance magnétique}\index{oscillation de Rabi}
Ajoutons un petit champ $B_1$ tournant dans le plan transverse à la
fréquence $\omega$. Dans le référentiel qui tourne avec lui, $B_0$ est
effectivement réduit à $(\omega_0 - \omega)/|\gamma|$~; \emph{à la
résonance} ($\omega = \omega_0$) seul $B_1$ subsiste, et le \omterm{def:b3:spin-two-level:spin}{spin}
précesse autour de lui à la \emph{fréquence de Rabi} $\Omega_1 =
|\gamma|B_1$~: partant de $\ket\uparrow$, la probabilité d'avoir
basculé vaut
\[
\mathcal P_\downarrow(t) = \sin^2\Big(\frac{\Omega_1t}{2}\Big) .
\]
Une impulsion de durée $\pi/\Omega_1$ (une «~impulsion $\pi$~»)
retourne le \omterm{def:b3:spin-two-level:spin}{spin}~; sa moitié (une «~impulsion $\pi/2$~») le couche
dans le plan équatorial, où il précesse à $\omega_0$ et, par induction
de Faraday, \emph{émet} sa fréquence dans toute bobine voisine. Hors
résonance l'amplitude du basculement s'effondre comme $\Omega_1^2/[\Omega_1^2 +
(\omega - \omega_0)^2]$~: la réponse est fortement sélective --- une
résonance --- et c'est ce qui rend les \omterm{def:b3:spin-two-level:spin}{spins} adressables, espèce par
espèce et, avec un gradient de champ, lieu par lieu
(\cref{pb:b3:spin-two-level:1}).
\end{proposition}
\admitted

\begin{example}[Structure hyperfine et raie à 21 cm]\label{ex:b3:spin-two-level:hyperfine}
\index{structure hyperfine}\index{raie a 21 cm@raie à 21 cm}
Dans l'état fondamental de l'hydrogène, les \omterm{def:b3:spin-two-level:spin}{spins} de l'électron et du
proton interagissent par leurs moments magnétiques~: les quatre états
de \omterm{def:b3:spin-two-level:spin}{spin} se séparent en un triplet et un singulet distants de $\Delta E =
\qty{5.9}{\micro eV}$ seulement --- la séparation \emph{hyperfine}, $\nu =
\qty{1420}{MHz}$, $\lambda = \qty{21}{cm}$. La transition est
absurdement lente (un basculement tous les dix millions d'années),
mais la Galaxie contient $10^{66}$ \omterm{thm:b3:hydrogen-atom:levels}{atomes d'hydrogène}~: la \omterm{ex:b3:spin-two-level:hyperfine}{raie à
21 cm} est assez brillante pour avoir cartographié les bras spiraux, le
gauchissement du disque et --- par ses décalages Doppler --- les
courbes de rotation plates qui plaident pour la matière noire. La même
physique, dans l'état fondamental du césium, sépare des niveaux
d'exactement $\qty{9192631770}{Hz}$~: depuis 1967, la \emph{définition
de la seconde} est un basculement de \omterm{def:b3:spin-two-level:spin}{spin} hyperfin que l'on compte
(\cref{exo:b3:spin-two-level:12}).
\end{example}

\begin{method}[L'art des deux niveaux]\label{met:b3:spin-two-level:method}
(1) Écrire tout \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltonien} à deux niveaux avec les \omterm{def:b3:spin-two-level:spin}{matrices de Pauli}~: $\hat H =
\epsilon_0\mathbb 1 + \vect h\cdot\hat{\vect\sigma}$ --- alors le
vecteur de Bloch précesse autour de $\vect h$ à $2|\vect h|/\hbar$~;
tout le reste est de la géométrie. (2) États propres selon $\vect n$~:
utiliser les demi-angles. (3) Problèmes de résonance~: passer au
référentiel tournant~; à la résonance, seul $B_1$ demeure. (4)
Impulsions~: $\pi/2$ pour lancer un signal de précession, $\pi$ pour
retourner (et pour refocaliser --- l'écho de \omterm{def:b3:spin-two-level:spin}{spin}). (5) Échelles
d'énergie~: $\mu_{\text{B}}B$ pour les électrons
($\qty{58}{\micro eV/T}$), trois ordres de moins pour les noyaux ---
des radiofréquences, une spectroscopie sans kelvins.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:1}
(a) Vérifier $\sigma_x^2 = \sigma_y^2 = \sigma_z^2 = \mathbb 1$ et
$\sigma_x\sigma_y = \iu\sigma_z$. (b) En déduire $[\hat S_x, \hat S_y]
= \iu\hbar\hat S_z$~: l'algèbre du moment cinétique en dimension deux.
(c) Montrer que $\hat S^2 = \tfrac34\hbar^2\,\mathbb 1$ et vérifier
$j(j+1)$ pour $j = \tfrac12$. (d) Montrer que toute matrice
hermitienne $2\times2$ est une combinaison réelle de $\mathbb 1$ et
des trois $\sigma_i$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:2}
(a) Trouver les états propres normés de $\sigma_x$ et de $\sigma_y$.
(b) Un \omterm{def:b3:spin-two-level:spin}{spin} préparé dans $\ket{+x}$ est mesuré selon $z$~:
probabilités des issues~? (c) Selon $y$~? (d) Vérifier les deux dans
l'image de Bloch (quels angles $\theta, \varphi$~?).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:3}
Trois aimants de Stern--Gerlach en cascade, d'axes $z$, puis $\vect n$
à l'angle $\theta$ dans le plan $xz$, puis $z$ de nouveau~; on garde
le faisceau $+$ à chaque fois. (a) Probabilité de survivre au deuxième
aimant. (b) De survivre aux trois. (c) Évaluer à $\theta = 90^\circ$
et comparer à \cref{ex:b3:spin-two-level:chained}. (d) Pour quel
$\theta$ la survie aux trois aimants est-elle maximale, et de quoi la
réponse est-elle l'écho, tiré de \cref{exo:b3:quantum-formalism:5}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:4}
Calculer l'écart de niveaux $2|\mu|B$ et la fréquence de résonance
dans un champ de \qty{1}{T} pour (a) l'électron ($g = 2$)~; (b) le
proton ($\mu_p = 2.79\,\mu_{\text{N}}$, $\mu_{\text{N}} =
e\hbar/2m_{\text{p}}$)~; (c) comparer les deux à $k_{\text{B}}T$ à
\qty{300}{K}~: à quel point les \omterm{def:b3:spin-two-level:spin}{spins} électroniques et nucléaires
sont-ils polarisés à température ambiante~? (d) Quelles bandes du
spectre la RPE et la RMN occupent-elles donc~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:5}
Obtenir la \omterm{prop:b3:spin-two-level:larmor}{précession de Larmor}~: avec $\hat H = -\gamma B_0\hat S_z$, (a)
calculer $\dd\langle\hat S_x\rangle/\dd t$, $\dd\langle\hat
S_y\rangle/\dd t$, $\dd\langle\hat S_z\rangle/\dd t$ à partir des
\omterm{def:b3:quantum-formalism:commutator}{commutateurs}~; (b) résoudre pour $\langle\hat{\vect S}\rangle(t)$ avec
un \omterm{def:b3:spin-two-level:spin}{spin} initialement selon $x$~; (c) montrer que
$\|\langle\hat{\vect S}\rangle\|$ et l'angle avec $\vect B_0$ sont
constants~; (d) pourquoi l'image des \emph{états propres de l'énergie}
(deux niveaux stationnaires) et celle de la \emph{précession} (un
vecteur qui tourne) décrivent-elles la même physique --- quel état
précesse~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:6}
(a) Vérifier que $\ket{+\vect n}$ de
\cref{prop:b3:spin-two-level:bloch} est bien le vecteur propre
annoncé. (b) Montrer que les états antipodaux sont orthogonaux. (c)
Où, sur la sphère, se trouvent les états propres de $\sigma_x$ et de
$\sigma_y$~? (d) Une porte «~NON~» de qubit envoie
$\ket\uparrow \leftrightarrow \ket\downarrow$~: quelle rotation de la
sphère est-ce, et quelle impulsion de
\cref{prop:b3:spin-two-level:rabi} la réalise~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:7}
Le référentiel tournant, honnêtement. Avec $\hat H(t) = -\gamma(B_0\hat S_z
+ B_1[\hat S_x\cos\omega t - \hat S_y\sin\omega t])$~: (a) expliquer
pourquoi passer au référentiel tournant à $\omega$ autour de $z$
remplace $B_0$ par $B_0 - \omega/\gamma$ et fige $B_1$~; (b) à la
résonance, décrire le mouvement du vecteur de Bloch dans ce
référentiel~; (c) obtenir la durée de l'impulsion $\pi$ pour un proton avec $B_1 =
\qty{10}{\micro T}$~; (d) représenter ce que voit un observateur du
laboratoire pendant cette impulsion (deux rotations emboîtées).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:9}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:10}
Couplage spin--orbite, estimé. Dans le référentiel propre de

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:11}
Deux \omterm{def:b3:spin-two-level:spin}{spins} ensemble. Pour deux particules de \omterm{def:b3:spin-two-level:spin}{spin} $\tfrac12$, le \omterm{def:b3:spin-two-level:spin}{spin}
total vaut $\hat{\vect S} = \hat{\vect S}_1 + \hat{\vect S}_2$. (a)
Montrer que les quatre états produits se réorganisent en un
\emph{triplet} ($S = 1$~:
$\ket{\uparrow\uparrow}$, $(\ket{\uparrow\downarrow} +
\ket{\downarrow\uparrow})/\sqrt2$, $\ket{\downarrow\downarrow}$)
et un \emph{singulet} ($S = 0$~: $(\ket{\uparrow\downarrow} -
\ket{\downarrow\uparrow})/\sqrt2$) --- vérifier les comptes de $S_z$
et, pour les deux états du milieu, l'effet de $\hat S^2$ (utiliser $\hat
S^2 = \hat S_1^2 + \hat S_2^2 + 2\hat{\vect S}_1\cdot\hat{\vect
S}_2$ et $\hat{\vect S}_1\cdot\hat{\vect S}_2 = \tfrac12(\hat
S_{1+}\hat S_{2-} + \hat S_{1-}\hat S_{2+}) + \hat S_{1z}\hat
S_{2z}$). (b) Lequel des quatre est intriqué --- non factorisable en
produit~? (c) La paire hyperfine de l'hydrogène
(\cref{ex:b3:spin-two-level:hyperfine}) est exactement ce
triplet/singulet~: lequel est le plus haut en énergie, sachant que la
raie est \emph{émise} à 21 cm~? (d) Les \omterm{def:b3:spin-two-level:spin}{spins} du singulet sont
parfaitement anticorrélés selon \emph{tout} axe~: mesurer l'un selon
un $\vect n$ quelconque et l'autre répond à l'opposé. Pourquoi cette
corrélation, si frappante soit-elle, ne transmet-elle aucun signal~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:12}
L'horloge qui définit la seconde. La séparation hyperfine de l'état

\end{exercise}

\section{Problème~: voir à l'intérieur du corps}

\begin{problem}[{Problème du week-end --- l'imagerie par résonance
magnétique, du spin à l'image}]\label{pb:b3:spin-two-level:1}
Les deux tiers d'un être humain sont de l'eau~; chaque molécule d'eau
porte deux protons, chacun un aimant de \omterm{def:b3:spin-two-level:spin}{spin} $\tfrac12$. Placer une
personne dans un champ intense, chatouiller les protons à leur
fréquence de Larmor, et écouter~: voilà l'imagerie par \omterm{prop:b3:spin-two-level:rabi}{résonance
magnétique}, la physique du \omterm{def:b3:spin-two-level:spin}{spin} devenue médecine. Données~:
$\gamma_{\text{p}}/2\pi = \qty{42.58}{MHz/T}$~; $B_0 = \qty{3.0}{T}$~;
densité de protons du tissu $n \approx \qty{6.6e28}{m^{-3}}$~;
$k_{\text{B}}T$ à \qty{310}{K} vaut \qty{26.7}{meV}~; $h =
\qty{4.14e-15}{eV.s}$.

\medskip\noindent\textbf{Partie I --- Les \omterm{def:b3:spin-two-level:spin}{spins} dans l'aimant.}
\begin{enumerate}
  \item Calculer la fréquence de Larmor à \qty{3.0}{T} et l'énergie
        du photon $h\nu$ en eV.
  \item Calculer le déséquilibre de population entre les deux niveaux
        du proton, $\Delta n/n \approx h\nu/2k_{\text{B}}T$.
  \item Seul cet excédent --- quelques parties par million ---
        contribue au signal~: combien de protons «~utiles~» par
        centimètre cube de tissu~?
  \item Pourquoi la polarisation thermique est-elle si faible ici,
        alors que le jet de Stern et Gerlach se scindait
        entièrement~? (Que mesure chaque expérience~: des \omterm{def:b3:quantum-formalism:observable}{valeurs
        propres} d'un atome unique, ou une moyenne thermique~?)
  \item Que fait au signal le doublement de $B_0$ (deux effets~: la
        polarisation et la f.é.m. induite à plus haute fréquence)~?
        Pourquoi les hôpitaux paient-ils pour des aimants plus gros~?
  \item L'aimant est supraconducteur et toujours en service~:
        pourquoi une bouteille d'oxygène en acier laissée libre dans
        la salle est-elle un projectile mortel (quelle force de
        \cref{ex:b3:spin-two-level:stern-gerlach} agit sur elle)~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Impulsions et échos.}
\begin{enumerate}[resume]
  \item Une bobine transverse applique $B_1 = \qty{10}{\micro T}$ à
        la résonance~: calculer la fréquence de Rabi et les durées
        des impulsions $\pi/2$ et $\pi$.
  \item Après l'impulsion $\pi/2$, que fait l'aimantation, et
        qu'induit la loi de Faraday dans la bobine réceptrice (à
        quelle fréquence)~?
  \item Le signal transverse décroît avec la constante de temps $T_2$
        (les \omterm{def:b3:spin-two-level:spin}{spins} se déphasent dans le champ les uns des autres et
        dans les inhomogénéités locales)~; l'aimantation
        longitudinale repousse avec $T_1$ (l'énergie s'écoule vers le
        tissu). Valeurs typiques~: $T_1 \approx \qty{1}{s}$,
        $T_2 \approx \qty{0.1}{s}$. Pourquoi le temps de cohérence
        ne peut-il jamais beaucoup dépasser le temps de relaxation
        de l'énergie (que fait à la phase chaque basculement qui
        échange de l'énergie)~?
  \item L'écho de \omterm{def:b3:spin-two-level:spin}{spin}~: après l'impulsion $\pi/2$ et un délai
        $\tau$, on applique une impulsion $\pi$ et, à $2\tau$, les
        \omterm{def:b3:spin-two-level:spin}{spins} déphasés se réalignent et le signal revient. Expliquer
        l'astuce avec des coureurs sur une piste que l'on fait faire
        demi-tour.
  \item Quelle décroissance l'écho défait-il --- le déphasage dû aux
        inhomogénéités de champ \emph{statiques}, ou celui dû aux
        champs moléculaires fluctuants --- et pourquoi celui-là
        seulement~?
  \item Les tissus ont des $T_1$, $T_2$ différents (la graisse~:
        $T_1$ court~; l'eau et les liquides~: long~; bien des
        tumeurs~: $T_2$ plus long que leur tissu hôte). En une
        phrase~: comment le minutage de la séquence d'impulsions
        change-t-il des temps de relaxation en \emph{contraste}
        d'image~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Du signal à l'image.}
\begin{enumerate}[resume]
  \item Superposons un gradient $G = \qty{40}{mT/m}$ selon $z$~: la
        fréquence de Larmor devient dépendante de la position.
        Calculer $\dd\nu/\dd z$ en kHz par millimètre.
  \item Une impulsion radiofréquence mise en forme ne contenant que
        la seule bande $\nu_0 \pm
        \qty{1}{kHz}$ excite quelle tranche du corps~? (Sélection de
        coupe.)
  \item Pendant la lecture, un gradient selon $x$ fait émettre à
        chaque colonne de la coupe sa propre fréquence~: quelle
        opération mathématique change le signal temporel reçu en un
        profil spatial (rappeler la boîte à outils de Fourier du
        volume de deuxième année)~?
  \item Estimer l'information totale d'un examen~: une image
        $256 \times 256$ sur 12 bits, et pourquoi l'acquérir ligne
        par ligne (un écho par ligne, temps de répétition
        $\sim T_1$) fait durer un examen plusieurs minutes.
  \item Le patient entend de forts coups~: qu'est-ce qui cogne
        mécaniquement (penser aux bobines de gradient qui commutent
        dans le champ de \qty{3}{T} --- quelle force)~?
  \item L'imagerie par rayons X contraste la densité électronique~;
        l'IRM contraste la densité de protons \emph{et} la
        relaxation~: pourquoi l'IRM est-elle l'outil de choix pour
        les tissus mous et le cerveau, et quelle dose ionisante
        délivre-t-elle~?
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- Les autres métiers du \omterm{def:b3:spin-two-level:spin}{spin}.}
\begin{enumerate}[resume]
  \item Les chimistes font la même expérience à une résolution de
        \qty{10}{ppm}~: les nuages électroniques écrantent légèrement
        chaque noyau et décalent sa résonance (le «~déplacement
        chimique~»). Pourquoi cela transforme-t-il la RMN en
        empreinte moléculaire~?
  \item L'IRM fonctionnelle cartographie la pensée~: l'hémoglobine
        désoxygénée est paramagnétique et raccourcit le $T_2^*$
        local. Retracer la chaîne de l'activité neuronale à la
        luminosité de l'image.
  \item La même physique de Larmor avec le moment de l'\emph{électron}
        opère à \qty{84}{GHz} sous \qty{3}{T}~: pourquoi la résonance
        électronique est-elle inutile à l'intérieur d'un être humain
        mais précieuse pour étudier radicaux et défauts~?
  \item Comparer l'énergie du photon de la question 1 aux énergies de
        liaison chimiques typiques~: justifier l'adjectif
        «~non ionisant~» et l'opposer à un photon X de \qty{60}{keV}.
  \item La polarisation de dix ppm peut être augmentée~: du xénon
        polarisé par laser atteint une polarisation d'ordre un et est
        inhalé pour imager les espaces aériens du poumon. De quel
        facteur environ le signal par noyau croît-il, et pourquoi la
        faible densité du gaz rend-elle malgré tout l'astuce
        nécessaire~?
  \item Un \omterm{def:b3:spin-two-level:spin}{spin} d'électron unique dans un piège de type transistor au
        silicium est aujourd'hui lu et piloté comme un qubit avec
        exactement les impulsions de ce chapitre~: nommer les deux
        propriétés du \omterm{def:b3:spin-two-level:spin}{spin} (dimension de son \omterm{def:b3:quantum-formalism:state}{espace de Hilbert}~;
        faiblesse de son couplage au bruit) qui le recommandent.
  \item Résumer le résultat nommé~: une précession de \omterm{def:b3:spin-two-level:spin}{spin} à
        \qty{128}{MHz}, une polarisation thermique de dix parties par
        million, deux horloges de relaxation et trois gradients de
        champ suffisent à photographier un cerveau vivant en coupes
        --- les deux taches de Stern et Gerlach, devenues un hôpital.
\end{enumerate}
\end{problem}
