\chapter{L'ensemble grand-canonique}\label{ch:b3:grand-canonical}

Une gouttelette d'eau dans un air humide ne grossit ni ne rétrécit~;
l'oxygène se fixe à l'hémoglobine dans les poumons et s'en détache
dans les muscles~; une pile pousse des électrons à travers un
téléphone parce qu'ils y sont «~plus chers~» dans une électrode que
dans l'autre. Dans chaque cas, la monnaie que l'on équilibre n'est pas
la température mais le \emph{\omterm{thm:b3:grand-canonical:distribution}{potentiel chimique}} $\mu$~: le coût en
\omterm{prop:b3:canonical-ensemble:machine}{énergie libre} d'une particule de plus. Ce chapitre ouvre la seconde
vanne du réservoir --- les particules autant que l'énergie --- et
bâtit l'\omterm{thm:b3:grand-canonical:distribution}{ensemble grand-canonique}, dont la récompense est une
efficacité stupéfiante~: l'occupation d'un unique niveau quantique,
fermion ou boson, s'obtient en deux lignes, et avec elle les formules
maîtresses du chapitre suivant. Ses applications vont des isothermes
d'adsorption des masques à gaz et des pots catalytiques à l'\omterm{prop:b3:grand-canonical:saha}{équation
de Saha}, avec laquelle une doctorante de Harvard a lu, en 1925, les
températures des étoiles et découvert, contre toute attente, que
l'univers est fait d'hydrogène.

\section{Les particules entrent dans la négociation}

\begin{theorem}[Distribution grand-canonique]\label{thm:b3:grand-canonical:distribution}
\index{ensemble grand-canonique}\index{potentiel chimique}
Un système qui échange de l'énergie \emph{et des particules} avec un
réservoir à la température $T$ et au \omterm{thm:b3:grand-canonical:distribution}{potentiel chimique} $\mu$ occupe
son état $s$ (d'énergie $E_s$, de nombre de particules $N_s$) avec la
probabilité
\[
\mathcal P_s = \frac{\eu^{-\beta(E_s - \mu N_s)}}{\Xi} , \qquad
\Xi = \sum_s\eu^{-\beta(E_s - \mu N_s)} ,
\]
la \emph{grande \omterm{thm:b3:canonical-ensemble:boltzmann}{fonction de partition}}. À partir de $\Xi$~: $\langle N\rangle
= k_{\text{B}}T\,\partial\ln\Xi/\partial\mu$, et le \emph{grand
potentiel} $\Phi = -k_{\text{B}}T\ln\Xi$ vérifie $\Phi = -PV$.
\end{theorem}

\begin{proof}[Démonstration partielle]
Comme dans \cref{thm:b3:canonical-ensemble:boltzmann}, développons
l'entropie du réservoir pour les deux prélèvements~:
$S_{\text{res}}(E - E_s, N - N_s) \approx S_{\text{res}} - E_s/T +
\mu N_s/T$, avec $\partial S/\partial N = -\mu/T$
(\cref{thm:b3:microcanonical-ensemble:temperature}). Les moyennes
suivent par dérivation~; $\Phi = -PV$ est admis (il découle de
l'extensivité).
\end{proof}

\begin{definition}[Ce que signifie $\mu$]\label{def:b3:grand-canonical:mu}
\index{potentiel chimique!signification}
$\mu$ est le coût, en \omterm{prop:b3:canonical-ensemble:machine}{énergie libre}, de l'ajout d'une particule à $T$
et $V$ fixés. Les particules s'écoulent spontanément du $\mu$ élevé
vers le $\mu$ bas --- diffusion, osmose, évaporation, réactions
chimiques et courants électriques sont autant de gradients de $\mu$
qui se relâchent. À l'équilibre entre deux phases ou deux lieux
quelconques, $\mu$ est égal des deux côtés --- la troisième égalité
(avec $T$ et $P$) qui régit la coexistence. Pour un gaz parfait
classique,
\[
\mu = k_{\text{B}}T\,\ln\!\big(n\lambda_T^3\big) < 0
\quad\text{(gaz dilué)} :
\]
ajouter une particule à un gaz dilué \emph{abaisse} l'\omterm{prop:b3:canonical-ensemble:machine}{énergie libre},
parce que le gain d'entropie l'emporte sur tout coût énergétique ---
le signe qui pousse l'évaporation dans un air sec. Et une pile est une
machine à $\mu$~: sa tension est la chute de \omterm{thm:b3:grand-canonical:distribution}{potentiel chimique} par
électron entre ses électrodes, $U = \Delta\mu/e$ --- les volts sont le
taux de change de la chimie des électrons, et c'est pourquoi les
éléments délivrent des tensions à un chiffre.
\end{definition}

\section{Un niveau à la fois~: l'astuce maîtresse}

\begin{theorem}[Occupation d'un unique niveau quantique]\label{thm:b3:grand-canonical:occupation}
\index{distribution de Fermi--Dirac}\index{distribution de Bose--Einstein}
Prenons pour «~système~» un unique niveau à une particule d'énergie
$\epsilon$, en contact grand-canonique avec le gaz de tous les autres.
\emph{Fermions} (occupation $0$ ou $1$)~:
\[
\Xi = 1 + \eu^{-\beta(\epsilon - \mu)}
\;\Longrightarrow\;
\bar n_{\text{FD}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} + 1} .
\]
\emph{Bosons} (occupation $0, 1, 2, \dots$)~:
\[
\Xi = \frac{1}{1 - \eu^{-\beta(\epsilon - \mu)}}
\;\Longrightarrow\;
\bar n_{\text{BE}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} - 1} ,
\]
ce qui exige $\mu < \epsilon_{\min}$. Les deux se réduisent au facteur
de Boltzmann $\eu^{-\beta(\epsilon - \mu)}$ quand les occupations sont
rares. Ces deux formules, obtenues en deux lignes chacune, sont toute
la donnée de la statistique quantique~: le chapitre suivant en est la
récolte.
\end{theorem}

\begin{proof}
Fermions~: deux termes. Bosons~: une série géométrique, convergente
seulement pour $\mu < \epsilon$. Dans les deux cas $\bar n = k_{\text{B}}T\,
\partial\ln\Xi/\partial\mu$ donne les formes annoncées.
\end{proof}

\section{Adsorption~: des sites à louer}

\begin{proposition}[L'isotherme de Langmuir]\label{prop:b3:grand-canonical:langmuir}
\index{isotherme de Langmuir}\index{adsorption}
Une surface offre des sites indépendants, chacun vide ou occupé par
une molécule de gaz d'énergie de liaison $-\epsilon_0$~; le gaz
au-dessus, à la pression $P$, est le réservoir. Chaque site est un
système grand-canonique à deux états, et son occupation vaut
\[
\theta = \frac{1}{\eu^{\beta(-\epsilon_0 - \mu)} + 1}
 = \frac{P}{P + P_0(T)} , \qquad
P_0(T) = \frac{k_{\text{B}}T}{\lambda_T^3}\,
\eu^{-\epsilon_0/k_{\text{B}}T} ,
\]
en utilisant le $\mu$ du gaz parfait. Le taux de recouvrement croît
linéairement à basse pression et sature à une monocouche ---
l'\emph{\omterm{prop:b3:grand-canonical:langmuir}{isotherme de Langmuir}}, la courbe de travail des pots
catalytiques, du charbon actif, des masques à gaz et de la fixation de
l'oxygène par le sang (\cref{exo:b3:grand-canonical:6}). La pression
de transition $P_0$ décroît \emph{exponentiellement} avec l'énergie de
liaison et croît fortement avec la température~: chauffer une surface
en déloge les hôtes --- et c'est pourquoi cuire la verrerie sous vide
est la seule façon de la nettoyer vraiment.
\end{proposition}

\begin{proof}
L'occupation du site est celle de
\cref{thm:b3:grand-canonical:occupation} (forme fermionique~: un site
en tient au plus un) avec $\epsilon = -\epsilon_0$~; substituer
$\eu^{\beta\mu} = n\lambda_T^3 = P\lambda_T^3/k_{\text{B}}T$.
\end{proof}

\section{Équilibres d'ionisation~: l'équation de Saha}

\begin{proposition}[L'équation de Saha]\label{prop:b3:grand-canonical:saha}
\index{équation de Saha}
Dans un gaz chaud, la réaction $\text{H} \rightleftharpoons \text{p} +
\text{e}$ s'équilibre quand $\mu_{\text{H}} = \mu_{\text{p}} +
\mu_{\text{e}}$. En écrivant chaque $\mu$ sous la forme du gaz parfait
(l'atome se voyant créditer sa liaison $-E_{\text{I}}$) on obtient
\[
\frac{n_{\text{e}}\,n_{\text{p}}}{n_{\text{H}}}
 = \frac{1}{\lambda_{T,\text{e}}^3}\,
\eu^{-E_{\text{I}}/k_{\text{B}}T}
\qquad\Big(\lambda_{T,\text{e}} =
h/\sqrt{2\pi m_{\text{e}}k_{\text{B}}T}\Big)
\]
(aux facteurs de dégénérescence de spin d'ordre un près). Deux
monnaies rivales, encore~: le facteur de Boltzmann taxe l'ionisation
de $E_{\text{I}} = \qty{13.6}{eV}$, tandis que le préfacteur ---
l'\emph{entropie de l'électron libéré}, avec ses $\lambda_T^{-3}$
états accessibles --- la subventionne. La subvention est si forte aux
densités stellaires que l'hydrogène s'ionise sensiblement dès
\qty{10000}{K}, où $k_{\text{B}}T$ ne vaut que \qty{0.9}{eV}~: les
équilibres se disputent en \omterm{prop:b3:canonical-ensemble:machine}{énergie libre}, non en énergie --- et les
spectres des étoiles en sont le grand livre
(\cref{pb:b3:grand-canonical:1}).
\end{proposition}

\begin{proof}[Démonstration partielle]
Posons $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$ avec $\mu_i
= k_{\text{B}}T\ln(n_i\lambda_{T,i}^3) + \varepsilon_i$
($\varepsilon_{\text{H}} = -E_{\text{I}}$, les autres nuls)~;
exponentions. Les longueurs d'onde thermiques du proton et de l'atome
se compensent presque (masses égales)~; les facteurs de dégénérescence
sont admis.
\end{proof}

\begin{method}[L'art du grand-canonique]\label{met:b3:grand-canonical:method}
(1) Tout ce qui échange des particules avec un grand partenaire~:
donner au partenaire un $\mu$ et pondérer les états par
$\eu^{-\beta(E - \mu N)}$. (2) Des sites ou des niveaux indépendants
se factorisent~: en résoudre \emph{un}, multiplier. (3) Pour un
réservoir de gaz classique, $\eu^{\beta\mu} =
n\lambda_T^3$ convertit $\mu$ en une pression mesurable. (4) Réactions et changements de
phase~: équilibrer les $\mu$, un par espèce, énergies de liaison
comprises --- s'attendre à ce que les préfacteurs entropiques
déplacent les équilibres loin des estimations naïves de Boltzmann. (5)
Électrons dans les solides~: leur $\mu$ est le niveau de Fermi, et les
différences de $\mu$ sont des tensions --- le pont vers le
\cref{ch:b3:electrons-in-solids}.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:1}
(a) Deux boîtes du même gaz à la même $T$ mais de densités
différentes sont mises en communication~: dans quel sens les
particules s'écoulent-elles, en langage de $\mu$~? (b) Montrer, à
partir du $\mu$ du gaz parfait, que l'écoulement s'arrête à densités
égales. (c) De l'eau sous un air humide~: énoncer la condition
d'équilibre en $\mu$. (d) Pourquoi le linge sèche-t-il même
au-dessous de $100\,^\circ$C quand l'air est sec (comparer les $\mu$,
non les températures)~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:2}
Le \omterm{thm:b3:grand-canonical:distribution}{potentiel chimique} de l'air. (a) Calculer $\lambda_T$ pour N$_2$
à \qty{300}{K}. (b) Calculer $n\lambda_T^3$ à \qty{1}{bar} et le $\mu$
qui en résulte, en eV. (c) Pourquoi $\mu < 0$ n'est-il pas un paradoxe
(quel terme de $F = E - TS$ domine lorsqu'on ajoute une molécule)~? (d)
À quelle densité $\mu$ atteindrait-il zéro, et qu'annonce
$n\lambda_T^3 \sim 1$ (\cref{exo:b3:grand-canonical:11})~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:3}
Obtenir l'occupation de Fermi--Dirac à partir de son $\Xi$ à deux
termes, pas à pas~; vérifier les limites $\epsilon \ll \mu$, $\epsilon \gg
\mu$, et la valeur en $\epsilon = \mu$. Où cette
fonction est-elle déjà apparue dans ce livre
(\cref{exo:b3:microcanonical-ensemble:8})~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:4}
Les volts sont de la chimie. (a) Un élément au lithium tient
\qty{3.0}{V}~: exprimer la différence de \omterm{thm:b3:grand-canonical:distribution}{potentiel chimique} par
électron en eV et par mole en kJ. (b) Pourquoi toutes les piles du
quotidien font-elles quelques volts (qu'est-ce qui fixe l'échelle des
différences de $\mu$ chimiques)~? (c) Une batterie de voiture au plomb
empile six éléments~: pourquoi empiler plutôt que chercher une
réaction à \qty{12}{eV}~? (d) En langage de $\mu$, que fait une pile
quand elle est «~à plat~»~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:5}
Des bosons sur un niveau. (a) Sommer la série géométrique pour $\Xi$
et obtenir $\bar n_{\text{BE}}$. (b) Pourquoi $\mu$ doit-il rester
sous le niveau le plus bas --- qu'est-ce qui diverge sinon, et
qu'annonce cette divergence (\cref{ch:b3:quantum-statistics})~? (c)
Calculer $\bar n$ pour $(\epsilon - \mu)/k_{\text{B}}T = 0.1$, $1$,
$3$ et comparer à Boltzmann. (d) Les photons ont $\mu = 0$~: donner
la raison physique (leur nombre n'est pas conservé --- les parois en
absorbent et en émettent librement).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:6}
Langmuir à l'œuvre. (a) D'après l'isotherme, quelle pression atteint
$\theta = 0.5$~? $0.9$~? (b) La myoglobine fixe O$_2$ sur un seul
site~: la saturation suit Langmuir en pression partielle d'oxygène ---
pourquoi observe-t-on une courbe \emph{hyperbolique} (et non
sigmoïde) pour la myoglobine, mais une courbe en S coopérative pour
l'hémoglobine à quatre sites~? (c) Le monoxyde de carbone se fixe sur
les mêmes sites avec un $\epsilon_0$ supérieur de
$\approx\qty{0.13}{eV}$~: à pressions partielles égales et
\qty{310}{K}, calculer le rapport d'occupation CO:O$_2$ et expliquer
l'intoxication au CO. (d) Pourquoi l'oxygénothérapie hyperbare
fonctionne-t-elle, en langage d'isotherme~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:7}
Fluctuations du nombre. (a) Montrer que $\operatorname{Var}N =
k_{\text{B}}T\,\partial\langle N\rangle/\partial\mu$. (b) Pour le gaz
parfait classique, déduire $\operatorname{Var}N = \langle
N\rangle$~: du Poisson. (c) Réconcilier avec les fluctuations de
densité de \cref{exo:b3:microcanonical-ensemble:12}. (d) Pour un
unique niveau fermionique, calculer $\operatorname{Var}n = \bar n(1 - \bar
n)$~: où les nombres de fermions sont-ils les plus calmes, et
pourquoi~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:8}
L'osmose à partir de $\mu$. Une membrane laisse passer l'eau mais non
le soluté (concentration $c$ par unité de volume, dilué). (a) En
égalant le $\mu$ de l'eau de part et d'autre, montrer que le côté
d'eau pure doit être surpressé de $\Pi = ck_{\text{B}}T$ (van 't
Hoff) --- admettre qu'un soluté dissous abaisse le $\mu$ de l'eau de
$-k_{\text{B}}T\,c/n_{\text{w}}$ par molécule d'eau. (b) Calculer
$\Pi$ pour un sérum physiologique ($c \approx \qty{3e26}{m^{-3}}$
particules au total). (c) Des globules rouges dans de l'eau pure~:
dans quel sens l'eau s'écoule-t-elle, et avec quel résultat~? (d)
Pourquoi les arbres n'ont-ils besoin ni de pompes ni de miracles pour
élever la sève de plusieurs dizaines de mètres (gradients osmotiques
et capillaires de $\mu$)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:9}
Équilibre dans un champ. Dans la pesanteur, l'équilibre d'une colonne
de gaz exige que le \omterm{thm:b3:grand-canonical:distribution}{potentiel chimique} \emph{total} $\mu(h) =
k_{\text{B}}T\ln(n\lambda_T^3) + mgh$ soit uniforme. (a) En obtenir la
loi barométrique en une ligne. (b) Généraliser à une centrifugeuse de
vitesse angulaire $\omega$ (potentiel $-\tfrac12 m\omega^2r^2$) et
obtenir le profil radial. (c) Les centrifugeuses d'enrichissement de
l'uranium tournent à $\omega r \sim
\qty{600}{m/s}$~: calculer l'enrichissement d'équilibre par étage de
$^{238}$UF$_6$ contre $^{235}$UF$_6$, $\exp[\Delta m\,\omega^2
r^2/2k_{\text{B}}T]$ avec $\Delta m = \qty{5e-27}{kg}$, $T =
\qty{320}{K}$. (d) Pourquoi cascade-t-on de nombreux étages~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:10}
Saha, à la main. (a) Mener la démonstration par équilibre des $\mu$ de
\cref{prop:b3:grand-canonical:saha}. (b) Définir la fraction
ionisée $x$ à densité totale d'hydrogène $n$ et montrer que $x^2/(1 - x)
= \eu^{-E_{\text{I}}/k_{\text{B}}T}/n\lambda_{T,\text{e}}^3$. (c)
Photosphère solaire~: $T = \qty{5800}{K}$, $n \approx
\qty{e23}{m^{-3}}$~: montrer que la fraction ionisée de l'hydrogène
n'est que $\sim\num{e-4}$ --- la surface du Soleil est neutre. (d) À $T =
\qty{10000}{K}$ (même $n$)~: recalculer et commenter la raideur.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:11}
Quand les gaz deviennent quantiques. La dégénérescence s'installe vers $n\lambda_T^3 \sim
1$. (a) Pour l'air à \qty{300}{K}~: à quelle
distance du seuil (\cref{exo:b3:grand-canonical:2})~? (b) Pour les
électrons de conduction du cuivre ($n = \qty{8.5e28}{m^{-3}}$)~:
trouver la température au-dessous de laquelle $n\lambda_T^3 > 1$ ---
et conclure que les électrons d'un métal sont dégénérés à
\emph{toute} température de laboratoire. (c) Pour des atomes de
rubidium à $n = \qty{e20}{m^{-3}}$~: la température seuil quantique
(l'échelle du condensat de \cref{ch:b3:quantum-statistics}). (d)
Classer les trois et énoncer la règle~: particules légères et fortes
densités passent au quantique les premières.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:12}
L'univers perd ses positons. Dans l'univers primordial chaud,
$\eu^+ + \eu^- \rightleftharpoons 2\gamma$ maintenait les paires en
équilibre avec $\mu_{e^+} + \mu_{e^-} = 0$ (photons~: $\mu = 0$).
(a) Pour un plasma symétrique cela donne $\mu_{e^\pm} \approx 0$~:
montrer que la densité de paires vaut alors $n_\pm \approx
\lambda_T^{-3}\eu^{-m_{\text{e}}c^2/k_{\text{B}}T}$ dès que
$k_{\text{B}}T \ll m_{\text{e}}c^2$. (b) Calculer la température à
laquelle $m_{\text{e}}c^2 = k_{\text{B}}T$ et l'échelle de temps
cosmique correspondante (admettre $t \sim \qty{1}{s}$ à
$\qty{e10}{K}$). (c) Montrer qu'un modeste refroidissement
supplémentaire (un facteur de quelques unités) effondre la densité de
paires de nombreux ordres de grandeur~: les positons s'annihilent, ne
laissant que le minuscule excès primordial d'électrons. (d) Les
photons libérés réchauffent le bain de rayonnement~: quel rayonnement
fossile en porte le reçu (\cref{ch:b3:astrophysics})~?
\end{exercise}

\section{Problème~: lire les températures des étoiles}

\begin{problem}[{Problème du week-end --- Saha, Payne et ce dont
l'univers est fait}]\label{pb:b3:grand-canonical:1}
Les spectres stellaires furent catalogués au fil des années 1890 en
classes O, B, A, F, G, K, M selon l'intensité de leurs raies
d'absorption --- les raies de Balmer de l'hydrogène étant les plus
fortes en classe A et mystérieusement faibles chez les étoiles plus
chaudes comme plus froides. En 1925, Cecilia Payne appliqua la toute
nouvelle \omterm{prop:b3:grand-canonical:saha}{équation de Saha} et montra que tout ce zoo n'était qu'une
seule substance à des températures différentes --- et que les étoiles
sont massivement de l'hydrogène, conclusion si hérétique qu'on la lui
fit adoucir dans sa thèse. Ce problème refait son raisonnement.
Données~: l'absorption de Balmer part de $n = 2$, à $E_2 - E_1 =
\qty{10.2}{eV}$ au-dessus du fondamental~; $E_{\text{I}} =
\qty{13.6}{eV}$~; densités électroniques photosphériques $n_{\text{e}}
\sim \qty{e20}{m^{-3}}$.

\medskip\noindent\textbf{Partie I --- Deux facteurs rivaux.}
\begin{enumerate}
  \item L'absorption de Balmer exige de l'hydrogène neutre \emph{dans} $n =
        2$~: écrire le facteur de Boltzmann de la population de $n = 2$
        (dégénérescences~: $g_2/g_1 = 4$).
  \item L'évaluer à $5000$, $10000$ et \qty{20000}{K}~: dans quel
        sens ce facteur pousse-t-il l'intensité de raie quand $T$
        monte~?
  \item Maintenant le facteur de Saha~: quand $T$ monte, qu'advient-il
        de la fraction \emph{neutre}, et donc des raies de Balmer~?
  \item Expliquer qualitativement pourquoi le produit --- excité
        \emph{et} encore neutre --- doit passer par un maximum à une
        température intermédiaire.
  \item À l'aide de l'\omterm{prop:b3:grand-canonical:saha}{équation de Saha} avec $n_{\text{e}} =
        \qty{e20}{m^{-3}}$, estimer la température de demi-ionisation
        de l'hydrogène.
  \item Rassembler le maximum~: vers quelle température les raies de
        Balmer devraient-elles être les plus fortes, et quelle classe
        spectrale (les étoiles A, $\sim\qty{10000}{K}$) cela
        désigne-t-il~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- La séquence spectrale
décodée.}
\begin{enumerate}[resume]
  \item Les étoiles M (\qty{3000}{K}) montrent de fortes bandes
        moléculaires (TiO) et un hydrogène faible~: expliquer les deux
        par les deux facteurs.
  \item Les étoiles O (\qty{40000}{K}) montrent des raies d'hélium et
        de nouveau un hydrogène faible~: expliquer (qu'est-il advenu
        de l'hydrogène, et pourquoi l'hélium,
        $E_{\text{I}} = \qty{24.6}{eV}$, a-t-il maintenant son tour~?).
  \item Le Soleil (G, \qty{5800}{K}) montre de fortes raies de
        \emph{calcium ionisé} ($E_{\text{I,Ca}} = \qty{6.1}{eV}$) et
        pourtant un hydrogène faible~: montrer par la logique de Saha
        pourquoi le calcium est déjà ionisé quand l'hydrogène est
        encore neutre et surtout dans $n = 1$.
  \item Avant Payne, les astronomes lisaient l'intensité de raie comme
        une abondance~: qu'ont conclu de la composition du Soleil ses
        puissantes raies de calcium et ses faibles raies de Balmer~?
  \item L'intuition de Payne~: corriger l'intensité de chaque raie par
        son facteur de «~visibilité~» de Boltzmann--Saha. Quand elle
        l'a fait, quelle hiérarchie d'abondances vraies a émergé~?
  \item Son résultat --- l'hydrogène un million de fois plus abondant
        que le calcium, des étoiles surtout d'hydrogène et d'hélium
        --- pourquoi a-t-il été refusé (que suggérait une chimie
        centrée sur la Terre), et quand a-t-elle été confirmée~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Des nombres sur le maximum.}
\begin{enumerate}[resume]
  \item Calculer le facteur de Boltzmann de $n = 2$ exactement à
        \qty{9600}{K}.
  \item Avec l'estimation de Saha de la question 5, prendre la fraction
        neutre à $9600$, $15000$ et \qty{25000}{K} égale
        approximativement à $0.5$, $\num{e-3}$, $\num{e-5}$~: former
        le produit (fraction excitée $\times$ fraction neutre) à vos
        trois températures et situer son maximum.
  \item Pourquoi le maximum de Balmer est-il \emph{net} du côté chaud
        (quelle exponentielle l'emporte) et \emph{progressif} du côté
        froid~?
  \item Une photosphère de naine blanche a un $n_{\text{e}}$ mille
        fois plus grand~: dans quel sens la température de
        demi-ionisation se déplace-t-elle, et pourquoi (Le Chatelier
        en langage de $\mu$)~?
  \item Deux étoiles de même température mais de pressions
        différentes montrent donc des intensités de raie
        différentes~: quelle propriété stellaire (géante contre naine
        --- la gravité de surface) cela permet-il aux spectroscopistes
        de lire~?
  \item Les grands relevés modernes fixent les températures
        stellaires à $\pm1\%$ à partir de rapports de raies~: énoncer
        la chaîne postulat $\to$ Boltzmann $\to$ Saha $\to$
        thermomètre.
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- Ce que cela a réglé.}
\begin{enumerate}[resume]
  \item La séquence OBAFGKM était un chaos alphabétique réordonné
        empiriquement~: après Saha, quelle unique variable physique
        l'ordonne~?
  \item Dites pourquoi un \emph{même} élément peut dominer un spectre
        à une température et disparaître à une autre sans le moindre
        changement d'abondance --- la leçon centrale que Payne a
        enseignée à l'astronomie.
  \item L'hydrogène à trois quarts de toute la masse baryonique~:
        nommer deux autres piliers de ce livre qui reposent sur cette
        abondance (la fusion stellaire, \cref{ch:b3:astrophysics}~; le
        ciel à 21 cm, \cref{ch:b3:spin-two-level}).
  \item Otto Struve a qualifié la thèse de Payne de «~thèse de
        doctorat la plus brillante jamais écrite en astronomie~»~: à
        partir de ce problème, justifier l'éloge en une phrase.
  \item La même physique de Saha, jouée à l'envers à l'échelle
        cosmique, prédit la température à laquelle l'hydrogène de tout
        l'univers s'est recombiné pour la première fois
        ($\sim\qty{3000}{K}$ à la densité cosmique bien plus faible)~:
        nommer l'événement et son vestige
        (\cref{ch:b3:astrophysics}).
  \item Les équilibres d'ionisation font aussi tourner les lampes
        fluorescentes et les procédés plasma~: dites en une phrase
        pourquoi un état ionisé de type \qty{10000}{K} peut exister
        dans un tube dont le verre reste froid (quels deux
        sous-systèmes ne s'équilibrent pas~?).
  \item Résumer le résultat nommé~: une seule condition d'équilibre,
        $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$, change
        l'alphabet spectral en un thermomètre allant de $3000$ à
        \qty{40000}{K}, place le maximum de Balmer sur les étoiles A,
        et révèle un univers fait d'hydrogène.
\end{enumerate}
\end{problem}
