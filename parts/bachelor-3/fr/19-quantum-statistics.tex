\chapter{Statistique quantique}\label{ch:b3:quantum-statistics}

Les électrons de conduction du cuivre forment un gaz mille fois plus
dense que l'air --- et pourtant ils n'ajoutent presque rien à la
capacité thermique du cuivre, scandale qui a hanté la physique
classique pendant cinquante ans. Un nuage d'atomes de rubidium,
refroidi à cent milliardièmes de kelvin, s'effondre soudain dans un
unique état quantique profond de dix mille atomes. Une étoile morte
de la masse du Soleil et de la taille de la Terre refuse de se
contracter davantage, soutenue par rien d'autre que le principe
d'exclusion. Ces trois faits sont un seul chapitre de physique~: ce
qui arrive à un gaz quand $n\lambda_T^3 \sim 1$ et que les formules
d'occupation du \cref{ch:b3:grand-canonical} s'écartent de
Boltzmann. Les fermions bâtissent des mers --- froides, rigides,
sous pression~; les bosons bâtissent des condensats --- grégaires,
cohérents, superfluides. Entre la majorité silencieuse de la mer de
Fermi et la ruée du condensat tient l'essentiel de la physique
moderne de la matière condensée et des basses températures, et le
destin des étoiles.

\section{Des gaz d'occupations}

\begin{proposition}[La machinerie du gaz quantique]\label{prop:b3:quantum-statistics:machinery}
\index{densité d'états}
Pour des particules sans interaction dans une boîte, les sommes sur
les niveaux deviennent des intégrales sur la \omterm{prop:b3:schrodinger-three-dimensions:counting}{densité d'états} de
\cref{prop:b3:schrodinger-three-dimensions:counting} (par état de
spin, $g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$)~:
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
avec $\bar n$ l'occupation de Fermi--Dirac ou de Bose--Einstein. La
première équation fixe $\mu(T, n)$~; la seconde livre la
thermodynamique. Dans la limite diluée $n\lambda_T^3 \ll 1$, les deux
statistiques s'effondrent sur Boltzmann et le gaz classique revient~;
les régimes quantiques commencent là où les paquets d'ondes se
touchent.
\end{proposition}
\admitted

\section{La mer de Fermi}

\begin{theorem}[Le gaz de fermions dégénéré]\label{thm:b3:quantum-statistics:fermi}
\index{énergie de Fermi}\index{pression de dégénérescence}
À $T = 0$, les fermions remplissent tous les niveaux jusqu'à
l'\emph{\omterm{thm:b3:quantum-statistics:fermi}{énergie de Fermi}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{spin } \tfrac12) ,
\]
avec une énergie moyenne $\tfrac35 E_{\text{F}}$ par particule et la
\emph{\omterm{thm:b3:quantum-statistics:fermi}{pression de dégénérescence}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
une pression au zéro absolu, d'origine purement quantique --- la
raideur du gaz d'électrons de tout métal et le soutien des étoiles
mortes. À $0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$, seuls
les électrons situés à $\sim k_{\text{B}}T$ de la surface de cette
\emph{mer de Fermi} peuvent répondre à quoi que ce soit~: la fraction
$\sim T/T_{\text{F}}$ est la clé maîtresse du comportement métallique.
\end{theorem}

\begin{proof}[Démonstration partielle]
Remplissons $g(\epsilon)$ jusqu'à $E_{\text{F}}$~: inverser $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ donne le $E_{\text{F}}$ annoncé
(\cref{exo:b3:schrodinger-three-dimensions:8})~; $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ par une intégrale de plus. Pression~:
comprimer la boîte élève chaque niveau comme $V^{-2/3}$, donc $E
\propto V^{-2/3}$ et $P = -\partial E/\partial V = \tfrac23 E/V$
--- avec $E = \tfrac35 NE_{\text{F}}$, la loi annoncée.
\end{proof}

\begin{example}[Le scandale de la capacité thermique, résolu]\label{ex:b3:quantum-statistics:heat}
\index{capacité thermique électronique}
Classiquement, l'électron libre par atome du cuivre devrait ajouter $\tfrac32
k_{\text{B}}$ à la capacité thermique --- $50\%$ par-dessus
Dulong--Petit. Mesuré~: environ $1\%$ à température ambiante. La mer
l'explique~: avec $T_{\text{F}} \approx \qty{8e4}{K}$, seule la
fraction $\sim T/T_{\text{F}} \approx 0.4\%$ des électrons peut être
excitée thermiquement, chacun de $\sim k_{\text{B}}T$~:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(le coefficient exact de Sommerfeld~;
\cref{exo:b3:quantum-statistics:12}). Linéaire en $T$, elle est
ensevelie sous le $T^3$ du réseau aux températures ordinaires mais
\emph{domine} au-dessous de quelques kelvins --- exactement ce que
trouvent les calorimètres, ce qui résout d'un coup l'énigme des
électrons qui portent le courant mais non la capacité thermique.
\end{example}

\begin{example}[Des étoiles soutenues par l'exclusion]\label{ex:b3:quantum-statistics:dwarf}
\index{naine blanche}
Une \omterm{ex:b3:quantum-statistics:dwarf}{naine blanche} (\cref{exo:b3:identical-particles:11}) équilibre la
gravité par la \omterm{thm:b3:quantum-statistics:fermi}{pression de dégénérescence} électronique. Comme $P \propto
n^{5/3}$ se raidit plus vite que ne l'exige la gravité à mesure que
l'étoile se contracte, un rayon d'équilibre existe pour toute masse
modeste --- avec cette loi d'échelle troublante $R \propto M^{-1/3}$~:
les \emph{naines lourdes sont plus petites}. Comprimés davantage, les
électrons deviennent relativistes, la loi de pression s'adoucit en
$n^{4/3}$, et l'équilibre échoue à une masse unique, la limite de
Chandrasekhar $M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) --- au-delà,
effondrement en étoile à neutrons~: une mer de Fermi de neutrons à la
densité nucléaire, la même physique à $10^{14}$ fois la densité.
\end{example}

\section{La ruée des bosons}

\begin{theorem}[Condensation de Bose--Einstein]\label{thm:b3:quantum-statistics:bec}
\index{condensation de Bose--Einstein}
Pour un gaz de $N$ bosons conservés, les états excités ne peuvent
contenir au plus que $N_{\max}(T) = 2.612\,V/\lambda_T^3$ particules
(l'intégrale de $g\bar n_{\text{BE}}$ à son plafond $\mu \to 0$).
Au-dessous de la température critique
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(de façon équivalente } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
l'excédent n'a nulle part où aller que l'unique état fondamental~:
une population \emph{macroscopique}
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
se condense dans un seul état quantique --- de la matière se
comportant comme une seule fonction d'onde géante. Prédite par
Einstein en 1925 à partir de la statistique de Bose pour les
photons~; réalisée dans du rubidium dilué en 1995 à
$T_{\text{c}} \approx \qty{100}{nK}$ (Nobel 2001), annoncée par la
distribution de vitesses \emph{bimodale} révélatrice~: un nuage
thermique d'où s'élève, en son centre, une aiguille de condensat à
vitesse nulle.
\end{theorem}
\admitted

\begin{example}[L'hélium, le superfluide vétéran]\label{ex:b3:quantum-statistics:helium}
\index{superfluidité}
L'hélium $^4$He liquide (bosonique) se transforme à
$T_\lambda = \qty{2.17}{K}$ en \emph{superfluide}~: viscosité nulle à
travers les capillaires les plus fins, écoulement en film sur les
parois, effets de fontaine, et circulation quantifiée en unités de
$h/m$ --- la phase macroscopique d'un condensat à l'œuvre, décalée de
la prédiction du gaz parfait ($\qty{3.1}{K}$ pour la densité de
l'hélium) par les interactions, mais indubitablement le même
phénomène. Le $^3$He fermionique ne peut condenser seul~: ses atomes
doivent d'abord s'\emph{apparier} (comme les électrons d'un
supraconducteur, \cref{ch:b3:electrons-in-solids}) en bosons
composites --- et la superfluidité arrive bel et bien, mais à
\qty{2.6}{mK}, mille fois plus froid~: deux isotopes, une leçon ---
la statistique est un destin.
\end{example}

\begin{method}[Quelle statistique, quel régime]\label{met:b3:quantum-statistics:method}
(1) Calculer d'abord $n\lambda_T^3$~: $\ll1$ signifie Boltzmann, et
le chapitre 17 suffit. (2) Fermions dégénérés~: penser en
$E_{\text{F}}$ et $T/T_{\text{F}}$~; les réponses (capacité
thermique, susceptibilité, diffusion) portent le facteur
$T/T_{\text{F}}$ ou n'échantillonnent que la surface de Fermi. (3)
Bosons dégénérés~: comparer $T$ à $T_{\text{c}}$~; au-dessous,
séparer le gaz en condensat plus nuage thermique. (4) Pressions~:
raideur fermionique en $n^{5/3}$ (ou $n^{4/3}$ relativiste) contre
ce qui comprime. (5) Particules composites~: compter les fermions
constituants pour choisir la statistique --- et rappeler que
l'appariement peut convertir une statistique en l'autre.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:1}
Calculer $E_{\text{F}}$ et $T_{\text{F}}$ pour (a) le cuivre
($n = \qty{8.5e28}{m^{-3}}$)~; (b) le sodium ($\qty{2.5e28}{m^{-3}}$)~;
(c) le $^3$He liquide ($\qty{1.6e28}{m^{-3}}$, $m = 3u$)~; (d)
les classer face à la température ambiante et dites qui est dégénéré
et quand.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:2}
(a) Montrer que $\langle\epsilon\rangle = \tfrac35 E_{\text{F}}$ à $T =
0$. (b) Calculer la vitesse quadratique moyenne des électrons dans le
cuivre. (c) À quelle température cette vitesse correspondrait-elle
classiquement~? (d) Pourquoi un métal froid contient-il des électrons
allant au centième de la vitesse de la lumière sans rayonner cette
énergie~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:3}
\omterm{thm:b3:quantum-statistics:fermi}{Pression de dégénérescence} des électrons du cuivre~: (a) évaluer $P =
\tfrac25 nE_{\text{F}}$ en pascals et en atmosphères. (b) Pourquoi le
métal n'explose-t-il pas (qu'est-ce qui retient)~? (c) Cette pression
résiste à la compression~: la relier à la faible compressibilité des
métaux. (d) Estimer le module de compressibilité $K \sim \tfrac53 P$
et comparer aux \qty{140}{GPa} mesurés du cuivre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:4}
Seuils de condensation. (a) Calculer $T_{\text{c}}$ pour le
rubidium-87 ($m =
\qty{1.44e-25}{kg}$) à $n = \qty{e20}{m^{-3}}$. (b) Pour le même $n$,
quel $T_{\text{c}}$ auraient des \omterm{thm:b3:hydrogen-atom:levels}{atomes d'hydrogène}~? (c) Pourquoi
les expériences emploient-elles des gaz dilués aux nanokelvins plutôt
que des gaz denses aux millikelvins (que se passe-t-il chimiquement à
haute densité et basse température)~? (d) Vérifier la règle de
cohérence $n\lambda_{T_c}^3
= 2.612$ sur vos nombres du rubidium.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:5}
Deux capacités thermiques. Dans un métal, $C = \gamma T +
\beta T^3$ (électrons~; réseau). (a) À l'aide de
\cref{ex:b3:quantum-statistics:heat} et du terme de réseau de type
Debye $\beta T^3 = 234\,Nk_{\text{B}}(T/\theta_{\text{D}})^3$,
estimer la température de croisement pour le cuivre
($\theta_{\text{D}} = \qty{343}{K}$, $T_{\text{F}} =
\qty{8.2e4}{K}$). (b) À \qty{300}{K}, la part électronique en
pourcentage. (c) À \qty{0.1}{K}~? (d) Pourquoi les expérimentateurs
tracent-ils $C/T$ contre $T^2$, et que livrent l'ordonnée à l'origine
et la pente de cette droite~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}
Une \omterm{ex:b3:quantum-statistics:dwarf}{naine blanche} d'une masse solaire ($M = \qty{2e30}{kg}$) et de
rayon terrestre ($R = \qty{6.4e6}{m}$), un électron pour deux
nucléons. (a) Calculer $n_{\text{e}}$ et $E_{\text{F}}$ (formule non
relativiste). (b) Comparer $E_{\text{F}}$ à $m_{\text{e}}c^2$ et à
$k_{\text{B}}T$ aux \qty{e7}{K} de l'intérieur~: justifier
simultanément «~froid~» et «~presque relativiste~». (c) Comparer la
\omterm{thm:b3:quantum-statistics:fermi}{pression de dégénérescence} $\tfrac25 nE_{\text{F}}$ à l'estimation de
la pression gravitationnelle $\sim GM^2/R^4$~: même ordre~? (d) Qu'arrive-t-il qualitativement à
chaque membre de l'équilibre si $M$ croît~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:7}
Matière neutronique. Une étoile à neutrons tasse $M = 1.4\,M_\odot$ dans $R
\approx \qty{11}{km}$. (a) Calculer la densité de neutrons et
comparer à la densité nucléaire ($\qty{2.3e17}{kg/m^3}$). (b)
Calculer $E_{\text{F}}$ pour les neutrons (non relativiste~; $m_{\text{n}}c^2
= \qty{939}{MeV}$) et vérifier à quel point ils sont relativistes.
(c) Pourquoi les électrons de l'étoile ont-ils disparu
(désintégration bêta inverse~: $\text{e} + \text{p} \to \text{n} + \nu$
--- qu'est-ce qui, dans le $E_{\text{F}}$ des électrons, a payé la
note)~? (d) Combien pèse une cuillerée de cette matière~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:8}
Des bosons sans nombre. (a) Les photons ont toujours $\mu = 0$~:
pourquoi ne peuvent-ils subir la condensation de
\cref{thm:b3:quantum-statistics:bec} (que fait le gaz à la place
quand on le refroidit)~? (b) Quelle loi de conservation les atomes de
rubidium possèdent-ils, que les photons n'ont pas~? (c) En 2010 un
condensat de photons a \emph{bel et bien} été réalisé, dans une
cavité remplie de colorant où les photons se thermalisent et où leur
nombre est effectivement fixé~: quel ingrédient la cavité à colorant
a-t-elle restauré~? (d) Les phonons d'un cristal~: condensation ou
non, et pourquoi~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
Les deux isotopes de l'hélium. (a) Calculer le $T_{\text{c}}$ du gaz
parfait pour le $^4$He liquide ($n = \qty{2.2e28}{m^{-3}}$, $m = 4u$)
et comparer à $T_\lambda = \qty{2.17}{K}$. (b) Pourquoi l'accord
n'est-il que grossier (qu'ignore le mot «~parfait~» dans un
liquide)~? (c) Le $^3$He, à densité presque égale, reste normal
jusqu'aux millikelvins~: énoncer l'obstacle et le contournement
qu'a trouvé la nature. (d) Vérification de logique composite~: une \emph{paire} de $^3$He contient un nombre
pair de fermions --- un boson~; le $^6$Li tient $3\text{p} + 3\text{n}
+ 3\text{e} = 9$ fermions~: le classer, et nommer les fameuses
expériences de gaz froids qui exploitent exactement ce contraste
Li--Li.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ honnêtement. (a) Écrire $N_{\text{exc}} = \int_0^\infty
g(\epsilon)\dd\epsilon/(\eu^{\beta\epsilon} - 1)$ à $\mu = 0$ et
le réduire, avec $x = \beta\epsilon$, à $N_{\text{exc}} =
(V/\lambda_T^3)\cdot\frac{2}{\sqrt\pi}\int_0^\infty\frac{\sqrt

\eu^x - 1) = 2.612\,\sqrt\pi/2$, obtenir le plafond et
$T_{\text{c}}$. (c) Obtenir la fraction condensée $1 -
(T/T_{\text{c}})^{3/2}$. (d) Pourquoi le même argument ne produit-il
\emph{aucune} condensation dans une boîte à deux dimensions (comment
$g(
\epsilon)$ change-t-il, et qu'advient-il de la convergence de
l'intégrale)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
Pourquoi les métaux tiennent. Modéliser un métal par des électrons de
densité $n$ (coût en \omterm{thm:b3:quantum-statistics:fermi}{énergie de Fermi} $\propto n^{2/3}$ par électron)
attirés par un fond d'ions neutralisant (gain coulombien par électron
$\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (a) Écrire l'énergie par
électron $\epsilon(n) = An^{2/3} - Bn^{1/3}$ et montrer qu'elle
possède un minimum. (b) Montrer que l'espacement électronique
d'équilibre est de l'ordre de $a_0$ --- les densités métalliques ne
sont pas un hasard. (c) Estimer l'échelle d'énergie de cohésion au
minimum, en eV. (d) Quel rôle joue la \omterm{thm:b3:quantum-statistics:fermi}{pression de dégénérescence} dans
cet état lié (qu'est-ce qui arrête l'effondrement coulombien)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
La couche de Sommerfeld, quantifiée. (a) Argumenter~: sur $N$
électrons, seuls $\sim g(E_{\text{F}})k_{\text{B}}T$ occupent la
couche active, chacun portant une énergie supplémentaire de
$\sim k_{\text{B}}T$~: obtenir $C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ à une constante près. (b) Avec
$g(E_{\text{F}}) = 3N/2E_{\text{F}}$, retrouver $C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ (la constante exacte vaut $\pi^2/2$).
(c) La même logique de couche donne la susceptibilité de spin de
Pauli, indépendante de la température~: seuls les électrons de la
couche peuvent basculer --- mais la couche croît comme $T$ tandis que
chaque contribution décroît comme $1/T$~: montrer la compensation.
(d) Énoncer la morale générale pour des fermions dégénérés~:
\emph{toute} réponse est la réponse classique multipliée par
$\sim T/T_{\text{F}}$ ou son analogue de surface.
\end{exercise}

\section{Problème~: la naine blanche la plus lourde possible}

\begin{problem}[{Problème du week-end --- la limite de Chandrasekhar
et les chandelles qu'elle a allumées}]\label{pb:b3:quantum-statistics:1}
En 1930, à dix-neuf ans, Subrahmanyan Chandrasekhar calcula, sur le
bateau qui l'emmenait en Angleterre, que la dégénérescence
électronique ne peut soutenir une étoile morte que jusqu'à une masse
bien définie. Le résultat --- combattu âprement par Eddington ---
fonde les étoiles à neutrons, les trous noirs et les «~chandelles
standard~» de supernova qui ont révélé l'accélération de l'univers.
Ce problème le construit par lois d'échelle. Données~: $M_\odot =
\qty{2e30}{kg}$~; un électron pour $\mu_e = 2$ nucléons~; $m_{\text{
n}} = \qty{1.67e-27}{kg}$~; $\hbar c = \qty{3.16e-26}{J.m}$~; $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{Partie I --- Équilibre de deux énergies.}
Pour une étoile de masse $M$ et de rayon $R$, traiter les énergies
par unité de masse en lois d'échelle.
\begin{enumerate}
  \item Énergie gravitationnelle~: $E_{\text{g}} \sim -GM^2/R$ ---
        rappeler son origine.
  \item Nombre d'électrons $N_{\text{e}} = M/\mu_e m_{\text{n}}$ et
        densité $n \sim N_{\text{e}}/R^3$~: écrire l'échelle
        d'\omterm{thm:b3:quantum-statistics:fermi}{énergie de Fermi} non relativiste en fonction de $M$
        et de $R$.
  \item Montrer que l'énergie cinétique totale (de dégénérescence)
        varie comme
        $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item Minimiser $E_{\text{k}} + E_{\text{g}}$ sur $R$ et montrer
        que le rayon d'équilibre vaut
        $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$.
  \item Lire la loi d'échelle $R \propto M^{-1/3}$~: les naines
        lourdes sont \emph{plus petites}. Pourquoi cela sent-il
        déjà les ennuis~?
  \item Évaluer $R$ pour $M = 1\,M_\odot$ et comparer aux
        $\approx\qty{5800}{km}$ mesurés de Sirius B.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- La relativité dérobe le
plancher.}
\begin{enumerate}[resume]
  \item À mesure que $R$ diminue, $E_{\text{F}}$ croît~: estimer la
        densité à laquelle $E_{\text{F}} \sim m_{\text{e}}c^2$, et
        la région de masse de naine correspondante.
  \item Dans le régime ultrarelativiste, l'énergie de chaque
        électron vaut $\sim\hbar c\,n^{1/3}$~: montrer que l'énergie
        cinétique devient
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$.
  \item Les deux énergies varient maintenant comme $1/R$~: conclure
        que l'équilibre ne sélectionne plus de rayon --- comparer
        plutôt les deux coefficients.
  \item En posant $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$, obtenir le
        nombre critique d'électrons et la masse
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item Évaluer le rapport sans dimension $\hbar c/Gm_{\text{n}}^2$
        --- l'un des grands nombres purs de la physique --- puis
        $M_{\text{Ch}}$ en masses solaires (le traitement exact
        donne $1.4\,M_\odot$).
  \item Interpréter la formule~: quelles trois constantes
        conspirent, et pourquoi une limite \emph{quantique} sur une
        \emph{étoile} porte-t-elle $\hbar$~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Au-delà de la limite.}
\begin{enumerate}[resume]
  \item Une naine poussée au-delà de $M_{\text{Ch}}$ (accrétion
        depuis une compagne)~: qu'est-ce qui la soutient ensuite, et
        à quelle échelle de rayon (remplacer $m_{\text{e}}$ par
        $m_{\text{n}}$ et $\mu_e$ par 1 dans le rayon de la partie
        I)~?
  \item Montrer que la limite relativiste propre à l'étoile à
        neutrons est du même ordre que $M_{\text{Ch}}$ --- la nature
        n'accorde qu'un sursis d'un facteur deux (maximum observé
        $\approx
        2.2\,M_\odot$). Qu'attend au-delà~?
  \item Lors de l'effondrement en étoile à neutrons, $\sim GM^2/R_{
        \text{ns}}$ d'énergie gravitationnelle est libéré~: évaluer
        cette énergie pour $M = 1.4\,M_\odot$, $R = \qty{11}{km}$,
        et comparer à toute la production du Soleil sur
        \qty{10}{Gyr} ($\sim\qty{1.2e44}{J}$).
  \item Presque tout part en neutrinos en quelques secondes~:
        quelle observation de 1987 a confirmé ce tableau de manière
        spectaculaire~?
  \item Le rebond du cœur en effondrement et la poussée des
        neutrinos soufflent l'enveloppe~: une supernova à
        effondrement de cœur. La distinguer, en une phrase chacune,
        de la supernova \emph{thermonucléaire} (de type Ia) d'une
        \omterm{ex:b3:quantum-statistics:dwarf}{naine blanche} qui s'enflamme au seuil de Chandrasekhar.
  \item Pourquoi une détonation \emph{à une masse universelle}
        rend-elle les supernovae de type Ia presque identiques ---
        et donc utilisables comme chandelles standard~?
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- Les chandelles et le
cosmos.}
\begin{enumerate}[resume]
  \item L'éclat apparent d'une chandelle standard donne sa
        distance~: énoncer la logique en carré inverse et pourquoi
        \emph{standard} est le mot décisif.
  \item En 1998, deux équipes ont trouvé des supernovae de type Ia
        lointaines \emph{plus faibles} que prévu~: énoncer
        l'inférence (expansion en accélération --- énergie sombre)
        et pourquoi la fiabilité de la chandelle était le point
        crucial.
  \item Énumérer en une ligne la chaîne de ce problème~: Pauli
        $\to$ \omterm{thm:b3:quantum-statistics:fermi}{pression de dégénérescence} $\to$ $M_{\text{Ch}}$ $\to$
        explosions uniformes $\to$ accélération cosmique.
  \item Eddington a raillé le résultat («~une étoile commettant des
        absurdités~»)~; Chandrasekhar a reçu le prix Nobel
        cinquante-trois ans plus tard~: que nous enseigne
        l'épisode sur le fait de suivre les équations au-delà des
        conclusions confortables~?
  \item Le spectre de Sirius B, en 1915, impliquait une tonne par
        centimètre cube --- «~un non-sens~» jusqu'à ce que Fowler,
        quelques mois après la statistique de Fermi de 1926,
        explique l'étoile comme un gaz dégénéré~: dites pourquoi
        l'énigme était insoluble une année et banale la suivante.
  \item Les relevés de milliers de \omterm{ex:b3:quantum-statistics:dwarf}{naines blanches} trouvent des
        masses groupées près de $0.6\,M_\odot$, aucune au-dessus de
        $\approx1.4\,M_\odot$~: quelles deux prédictions de ce
        problème ce seul histogramme confirme-t-il~?
  \item Résumer le résultat nommé~: $(\hbar c/Gm_{\text{n}}^2)^{
        3/2}m_{\text{n}} \approx 1.4\,M_\odot$ --- trois constantes
        de la nature fixant la \omterm{ex:b3:quantum-statistics:dwarf}{naine blanche} la plus lourde
        possible, la masse-gâchette des explosions calibrées de
        l'univers.
\end{enumerate}
\end{problem}
