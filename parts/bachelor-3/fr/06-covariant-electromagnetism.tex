\chapter{Électromagnétisme covariant}\label{ch:b3:covariant-electromagnetism}

Un fil de cuivre porte dix ampères. Ses électrons dérivent d'un
dixième de millimètre par seconde --- plus lentement qu'un escargot ---
et le fil est électriquement neutre à une précision fantastique. Et
pourtant une charge qui longe le fil subit une attraction mesurable. Vu
du référentiel propre de cette charge, l'explication «~magnétique~»
s'évapore~: la charge est au repos, et une charge au repos ne subit
aucune force magnétique. Ce qui l'attire, dans ce référentiel, est un
champ \emph{électrique} --- suscité par la relativité elle-même, parce
que la \omterm{prop:b3:relativistic-kinematics:contraction}{contraction des longueurs} déséquilibre, d'une part sur
$10^{24}$, les deux flots de charge du fil. Le magnétisme, c'est
l'électricité vue d'un siège en mouvement. Ce chapitre réécrit
l'électromagnétisme du volume de deuxième année dans la langue des
trois chapitres précédents~: des \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteurs} pour la charge et le
potentiel, un unique tenseur antisymétrique tenant ensemble $\vect E$
et $\vect B$, les quatre équations de Maxwell repliées en deux lignes
qui se lisent de même dans tout \omterm{thm:b3:relativistic-kinematics:postulates}{référentiel galiléen}. Au-delà de
l'élégance, le bénéfice est de la physique effective~: comment les
champs se transforment, pourquoi le champ d'une charge rapide s'aplatit
en galette, et pourquoi la lumière d'un électron relativiste balaie
vers l'avant comme un phare --- le principe des sources de rayonnement
synchrotron qui radiographient aujourd'hui protéines et batteries.

\section{Quadrivecteurs pour la charge et le courant}

\begin{definition}[Quadrivecteurs]\label{def:b3:covariant-electromagnetism:fourvector}
\index{quadrivecteur}
Un \emph{quadrivecteur} est un quadruplet $a^\mu = (a^0, \vect a)$, $\mu =
0, 1, 2, 3$, dont les composantes se mélangent sous un changement de
référentiel exactement comme celles de $(ct,
\vect r)$~:
\[
a'^0 = \gamma(a^0 - \beta a^1) , \qquad
a'^1 = \gamma(a^1 - \beta a^0) , \qquad
a'^{2,3} = a^{2,3}
\]
pour un changement de référentiel à $\beta c$ selon $x$. Deux
quadrivecteurs quelconques donnent le produit scalaire invariant
$a\cdot b = a^0b^0 - \vect a\cdot\vect b$, le même nombre dans tout
référentiel --- le motif derrière $c^2t^2 - x^2$ et $E^2 - p^2c^2$. La
position $(ct, \vect r)$ et le \omterm{def:b3:relativistic-dynamics:four-momentum}{quadrivecteur énergie--impulsion}
$(E/c, \vect p)$ sont les deux que nous avons~; l'électromagnétisme en
fournit trois de plus~: courant, potentiel et vecteur d'onde.
\end{definition}

\begin{proposition}[Le quadricourant et la conservation de la charge]\label{prop:b3:covariant-electromagnetism:fourcurrent}
\index{quadricourant}
La densité de charge $\rho$ et la densité de courant $\vect\jmath$
forment le quadricourant
\[
J^\mu = (\rho c,\ \vect\jmath\,) :
\]
un nuage de charge de densité propre $\rho_0$ se déplaçant à $\vect u$
a $J^\mu = \rho_0\gamma_u(c, \vect u)$ --- la densité croît de
$\gamma_u$ parce que le volume du nuage se contracte, tandis que la
charge elle-même est invariante (un fait expérimental décisif~: les
atomes dont les électrons internes sont rapides restent exactement
neutres). La conservation de la charge est l'énoncé invariant
\[
\frac{\partial(\rho c)}{\partial(ct)} +
\operatorname{div}\vect\jmath = 0 ,
\]
l'équation de continuité du volume de deuxième année, désormais
visiblement la même loi pour tous les observateurs.
\end{proposition}

\begin{proof}[Démonstration partielle]
Le nuage en mouvement~: une boîte de volume propre $V_0$ contient la charge $\rho_0
V_0$~; au laboratoire la boîte est contractée à $V_0/\gamma_u$, donc $\rho =
\gamma_u\rho_0$, et $\vect\jmath = \rho\vect u$ par définition d'une
densité de courant. Que $(\rho c, \vect\jmath)$ se transforme alors
comme un \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur} suit de ce qu'il vaut $\rho_0/c$ fois la
quadrivitesse $\gamma_u(c, \vect u)$, elle-même \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur} par
construction. L'invariance de la charge est une donnée expérimentale.
\end{proof}

\section{Un seul tenseur pour les deux champs}

\begin{definition}[Quadripotentiel et tenseur du champ]\label{def:b3:covariant-electromagnetism:tensor}
\index{quadripotentiel}\index{tenseur du champ}
Les potentiels du volume de deuxième année s'assemblent en le quadripotentiel
\[
A^\mu = \Big(\frac{V}{c},\ \vect A\Big) ,
\]
et les champs mesurables $\vect E = -\vect\nabla V -
\partial_t\vect A$, $\vect B = \operatorname{\vect{curl}}\vect A$ sont
les six composantes indépendantes d'un unique tableau antisymétrique,
le \emph{tenseur du champ électromagnétique}
\[
F^{\mu\nu} =
\begin{pmatrix}
0 & -E_x/c & -E_y/c & -E_z/c\\
E_x/c & 0 & -B_z & B_y\\
E_y/c & B_z & 0 & -B_x\\
E_z/c & -B_y & B_x & 0
\end{pmatrix} .
\]
$\vect E$ et $\vect B$ ne sont pas deux champs mais six faces d'un
seul objet~; la face qu'un observateur nomme «~électrique~» dépend du
mouvement de cet observateur.
\end{definition}

\begin{theorem}[Maxwell, sous forme covariante]\label{thm:b3:covariant-electromagnetism:maxwell}
\index{équations de Maxwell!forme covariante}
Les quatre équations de Maxwell du volume de deuxième année sont la
forme en composantes de deux énoncés indépendants du référentiel~: la
paire avec sources (Gauss et Ampère--Maxwell) s'écrit
\[
\sum_\mu \partial_\mu F^{\mu\nu} = \mu_0\,J^\nu ,
\]
et la paire sans source (absence de monopôle, Faraday) est l'identité
qui garantit que $F$ dérive d'un \omterm{def:b3:covariant-electromagnetism:tensor}{quadripotentiel}. La force de Lorentz
s'écrit $\dd p^\mu/\dd\tau = qF^{\mu\nu}u_\nu$ --- la loi de force,
partie magnétique comprise, en une ligne. Les deux membres de chaque
équation se transformant à l'identique, \emph{la théorie de Maxwell
n'a besoin d'aucune correction pour être relativiste}~: elle fut la
première pièce achevée de la relativité, née en 1865.
\end{theorem}

\begin{proof}[Démonstration partielle]
Développons la composante $\nu = 0$~: $\sum_i\partial_i(E_i/c) =
\mu_0\rho c$, c'est-à-dire $\operatorname{div}\vect E = \rho/\varepsilon_0$
(avec $c^2 = 1/\mu_0\varepsilon_0$). La composante $\nu = 1$ rassemble $-\partial_t(E_x/c^2) + (\partial_yB_z - \partial_zB_y)
= \mu_0 j_x$~: la composante $x$ d'Ampère--Maxwell. Les autres
composantes répètent le motif~; la paire sans source est l'égalité des
dérivées secondes croisées de $A^\mu$ (vérifiée dans
\cref{exo:b3:covariant-electromagnetism:3}). Que $F^{\mu\nu}$ se
transforme comme un tenseur (à deux indices) --- chaque indice comme un
\omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur} --- est admis~; ses conséquences sont l'objet de la
proposition suivante.
\end{proof}

\section{Comment les champs se transforment}

\begin{proposition}[Transformation des champs]\label{prop:b3:covariant-electromagnetism:transform}
\index{transformation des champs}
Pour un changement de référentiel à la vitesse $\vect v = v\,\vect e_x$~:
les composantes \emph{selon} le mouvement sont intactes, les
transverses se mélangent,
\[
E'_x = E_x , \qquad
E'_y = \gamma(E_y - vB_z) , \qquad
E'_z = \gamma(E_z + vB_y) ,
\]
\[
B'_x = B_x , \qquad
B'_y = \gamma\Big(B_y + \frac{v}{c^2}E_z\Big) , \qquad
B'_z = \gamma\Big(B_z - \frac{v}{c^2}E_y\Big) .
\]
De façon compacte, transversalement au mouvement~: $\vect E' = \gamma(\vect E +
\vect v\wedge\vect B)_\perp$ et $\vect B' = \gamma(\vect B - \vect
v\wedge\vect E/c^2)_\perp$. Un $\vect B$ pur dans un référentiel est $\vect
E$ et $\vect B$ dans un autre~: les champs sont un seul phénomène.
\end{proposition}
\admitted

\begin{example}[Le magnétisme d'un fil neutre]\label{ex:b3:covariant-electromagnetism:wire}
Un fil neutre porte un courant~: réseau positif au repos, électrons en
dérive. Pour une charge d'épreuve qui longe le fil, passons dans son
référentiel~: les deux flots de charge se contractent
\emph{différemment} (ils n'ont pas la même vitesse), le fil acquiert la
densité linéique nette $\lambda' =
-\gamma vI/c^2$, et son champ électrique radial attire la charge avec
exactement la force que le laboratoire appelait $q\vect v\wedge\vect B$.
Le magnétisme est ce à quoi ressemble le terme du second ordre $v u/c^2$
de la relativité quand $10^{28}$ charges élémentaires par mètre
conspirent~: chaque effet est fantastiquement petit, mais le fil est
neutre à une précision plus grande encore, si bien que le minuscule
déséquilibre fait toute l'histoire
(\cref{pb:b3:covariant-electromagnetism:1}).
\end{example}

\begin{example}[Le champ en galette d'une charge rapide]\label{ex:b3:covariant-electromagnetism:pancake}
Transformons le champ coulombien d'une charge dans le référentiel où
elle se déplace à $\gamma \gg 1$~: le champ longitudinal est inchangé
tandis que le transverse est multiplié par $\gamma$. Le champ,
naguère sphérique, s'aplatit en un disque d'angle d'ouverture
$\sim 1/\gamma$ autour du plan passant par la charge --- accompagné,
transversalement au mouvement, d'un anneau magnétique $B' = vE'/c^2$.
Pour un observateur immobile, le passage d'un proton du LHC
($\gamma \approx 7000$) est une gifle sub-picoseconde de $\vect E$ et
$\vect B$ presque croisés~: presque une impulsion de lumière --- la
raison pour laquelle les faisceaux rapides parlent à la matière comme
le font les photons.
\end{example}

\section{Invariants, lumière et effet phare}

\begin{proposition}[Les deux invariants du champ]\label{prop:b3:covariant-electromagnetism:invariants}
\index{invariants du champ}
À partir de $F^{\mu\nu}$ on ne peut construire que deux scalaires indépendants~:
\[
E^2 - c^2B^2
\qquad\text{et}\qquad
\vect E\cdot\vect B ,
\]
les mêmes dans tout \omterm{thm:b3:relativistic-kinematics:postulates}{référentiel galiléen}. Conséquences~: si $E > cB$
quelque part (avec $\vect E\cdot\vect B = 0$), un référentiel y voit un
champ purement électrique~; si $E < cB$, un référentiel y voit un champ
purement magnétique~; et une onde plane lumineuse, avec $E = cB$ et
$\vect E\perp\vect B$, a ses \emph{deux} invariants nuls --- c'est de
la lumière dans tout référentiel~: aucun changement de référentiel ne
peut la changer en champ statique, seulement la décaler vers le rouge.
On ne peut rattraper une onde lumineuse~; la question d'adolescent
d'Einstein se répond d'elle-même en deux invariants.
\end{proposition}

\begin{proof}[Démonstration partielle]
Vérifions l'invariance sous le changement de référentiel standard par
substitution directe de
\cref{prop:b3:covariant-electromagnetism:transform}
(\cref{exo:b3:covariant-electromagnetism:6})~; qu'aucun troisième
invariant indépendant n'existe est admis. Pour l'onde~: $E = cB$ et
l'orthogonalité donnent les deux nuls~; un référentiel où le champ
serait statique exigerait un invariant non nul.
\end{proof}

\begin{proposition}[Quadrivecteur d'onde, Doppler et aberration]\label{prop:b3:covariant-electromagnetism:wavevector}
\index{aberration}\index{effet phare}
La fréquence et la direction d'une onde plane forment le \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur}
isotrope $k^\mu = (\omega/c, \vect k)$, $\|\vect k\| = \omega/c$. Le
transformer donne d'un coup la formule Doppler relativiste du
\cref{ch:b3:relativistic-kinematics} et l'\emph{aberration} des
directions~:
\[
\cos\theta' = \frac{\cos\theta - \beta}{1 - \beta\cos\theta} .
\]
Lue à l'envers, une source rayonnant de façon isotrope dans son
référentiel propre envoie, au laboratoire, la moitié de sa lumière dans le cône avant $\theta
\lesssim 1/\gamma$~: l'\emph{\omterm{prop:b3:covariant-electromagnetism:wavevector}{effet phare}}. Un électron tournant à
$\gamma \sim 10^4$ dans un anneau de stockage promène autour de
l'anneau un projecteur de rayons X de $\qty{0.1}{mrad}$ --- le
rayonnement synchrotron qui alimente les lignes de
cristallographie des protéines.
\end{proposition}

\begin{proof}[Démonstration partielle]
La phase $\omega t - \vect k\cdot\vect r = k^\mu x_\mu$ compte les
crêtes qui passent par des événements --- un invariant --- donc $k^\mu$
doit se transformer comme un \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur}. Appliquons le changement de référentiel à $(\omega/c, k\cos\theta,
k\sin\theta, 0)$~: la composante temporelle donne le Doppler, le
rapport des composantes spatiales la formule d'aberration. En posant $\theta' =
90^\circ$ (le rayon transversal du référentiel de la source)~: $\cos\theta =
\beta$, c'est-à-dire $\theta \approx 1/\gamma$ pour $\gamma \gg 1$ ---
la moitié de la sphère se replie dans le cône avant.
\end{proof}

\begin{method}[Travailler de façon covariante]\label{met:b3:covariant-electromagnetism:method}
(1) Repérer les \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteurs} en jeu ($x^\mu$, $P^\mu$, $J^\mu$,
$A^\mu$, $k^\mu$) et préférer leurs produits invariants aux
composantes. (2) Pour transformer les champs, séparer les composantes
selon le mouvement et transversalement, puis appliquer
\cref{prop:b3:covariant-electromagnetism:transform}. (3) Contrôler les
deux invariants avant et après --- le détecteur d'erreur le plus rapide
du domaine. (4) Quand un problème magnétique paraît mystérieux, monter
avec la charge~: dans son référentiel, seul $\vect E'$ agit. (5) Faire
confiance à Maxwell~: les équations n'ont jamais besoin de
«~corrections~» relativistes, seulement d'une lecture relativiste.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:covariant-electromagnetism:1}
(a) Transformer $a^\mu = (5, 3, 0, 0)$ (en unités d'un certain $a_0$) à $\beta =
0.6$~: calculer $a'^\mu$ et vérifier que $a\cdot a$ est inchangé. (b)
Montrer que la somme de deux \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteurs} est un \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur}. (c) $(c,
\vect v)$ est-il un \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur} pour une particule~? Et
$\gamma(c, \vect v)$~? (d) Pourquoi «~le champ électrique~» seul ne
fait-il partie d'aucun \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:covariant-electromagnetism:2}
Un fil de cuivre de section \qty{1.0}{mm^2} porte \qty{10}{A}~;
densité d'électrons de conduction $n = \qty{8.5e28}{m^{-3}}$. (a)
Calculer la vitesse de dérive. (b) Écrire $J^\mu$ pour le fluide
d'électrons et pour le réseau. (c) Vérifier l'équation de continuité
pour chacun. (d) Le fil est neutre~: que vaut $J^\mu$ pour le fil
entier, et quelle composante survit~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:covariant-electromagnetism:3}
(a) Dans la matrice de $F^{\mu\nu}$, repérer quelles composantes
donnent $E_y$ et $B_x$. (b) Écrire $F^{\mu\nu}$ pour un champ
uniforme pur $\vect B = B\vect e_z$, puis pour le champ d'une onde
plane ($\vect E = E\vect e_y$, $\vect B = (E/c)\vect e_z$). (c)
Vérifier sur les composantes que $F^{\mu\nu} = \partial^\mu A^\nu - \partial^\nu
A^\mu$ redonne $\vect B = \operatorname{\vect{curl}}\vect A$ pour
$\mu\nu = 12$. (d) Pourquoi $F$ doit-il être antisymétrique pour que la
force $qF^{\mu\nu}u_\nu$ ne travaille pas dans le référentiel propre de
la particule~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:covariant-electromagnetism:4}
Un condensateur plan au repos porte $\vect E = E_0\vect e_y$, sans

(b) selon $\vect

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:covariant-electromagnetism:5}

sur des rails à travers un $\vect B$ uniforme. (a) Compte rendu au
laboratoire~: quelle force pousse les électrons le long de la tige, et
quelle f.é.m. en résulte~? (b) Compte rendu dans le référentiel de la
tige~: quel champ les pousse, et d'où vient-il dans
\cref{prop:b3:covariant-electromagnetism:transform}~? (c) Montrer que
les deux f.é.m. s'accordent à l'ordre $v/c$. (d) La règle du flux de
Faraday du volume de première année couvrait d'une seule formule les
cas «~circuit mobile~» et «~champ variable~»~: que dit la relativité
sur les raisons pour lesquelles cette unification devait fonctionner~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:covariant-electromagnetism:6}
(a) À l'aide de \cref{prop:b3:covariant-electromagnetism:transform},

référentiel ne peut-il rendre le champ d'une onde plane lumineuse
purement électrique ou purement magnétique~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:covariant-electromagnetism:7}
Champs croisés. Au laboratoire, $\vect E = E\vect e_y$ et $\vect B =
B\vect e_z$ avec $E < cB$. (a) Montrer que le référentiel se déplaçant à
$\vect v_{\text{d}} = (E/B)\,\vect e_x$ voit un champ purement magnétique.
(b) Décrire le mouvement d'une charge lâchée au repos, vu depuis ce
référentiel puis de retour au laboratoire (une cycloïde dérivant à
$v_{\text{d}}$). (c) Le filtre de vitesse du volume de première année
laissait passer sans déviation les particules de vitesse $E/B$~:
le retrouver en une ligne à partir de (a). (d) Que se passe-t-il,
qualitativement, quand $E > cB$ --- et pourquoi l'astuce du référentiel
de dérive est-elle alors impossible~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:covariant-electromagnetism:8}

Aberration de la lumière des étoiles~: la Terre orbite à
\qty{29.8}{km/s}~; de quel angle la position apparente d'une étoile

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:covariant-electromagnetism:9}
Une charge $q$ va à $\vect v = v\vect e_x$ constant, $\gamma \gg
1$. (a) En transformant le champ de Coulomb, montrer que dans le plan
transverse passant par la charge le champ est amplifié à $\gamma
q/4\pi\varepsilon_0b^2$ à la distance $b$, tandis que droit devant et
droit derrière il est écrasé d'un facteur $1/\gamma^2$. (b) Montrer
qu'un observateur immobile à la distance $b$ voit une impulsion de
champ de durée $\sim b/\gamma v$. (c) Pour un proton du LHC passant à $b = \qty{1}{cm}$~:
champ maximal et durée de l'impulsion. (d) En quel sens précis cette
impulsion est-elle «~presque de la lumière~»~? (Vérifier $E' \approx cB'$
et les invariants.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:covariant-electromagnetism:10}
Poursuivre une onde lumineuse. Une onde plane a $\vect E = E_0\vect e_y
\cos(\omega(t - x/c))$, $\vect B = (E_0/c)\vect e_z\cos(\omega(t -
x/c))$. Un observateur la poursuit à $\beta$. (a) Transformer les
champs~: montrer que $E'_0 = E_0\sqrt{(1-\beta)/(1+\beta)}$ et
$B'_0 = E'_0/c$. (b) Montrer que la fréquence se transforme du même
facteur~: l'onde reste une onde, décalée vers le rouge, avec toujours
$E' = cB'$. (c) Que deviennent la densité d'énergie de l'onde
(proportionnelle à $E^2$) et le nombre de photons~? (d) Conclure~: que
faudrait-il pour «~chevaucher un faisceau lumineux~», et quel invariant
l'interdit~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:covariant-electromagnetism:11}
Le rayonnement synchrotron et la mort des machines circulaires à
électrons. Une charge ultrarelativiste sur un cercle de rayon $r$
rayonne la puissance $P = q^2c\gamma^4/6\pi\varepsilon_0r^2$ (admis ---
la formule de Larmor du volume de deuxième année, transformée). (a)
Montrer que l'énergie perdue par tour vaut
$\Delta E = q^2\gamma^4/3\varepsilon_0r$. (b) LEP~: des électrons à
\qty{100}{GeV} sur $r = \qty{3.1}{km}$~: calculer $\gamma$ et
$\Delta E$ par tour --- quelle fraction de l'énergie du faisceau est
rachetée à chaque tour~? (c) LHC~: des protons à \qty{6.8}{TeV}, même
tunnel~: calculer $\Delta E$ par tour et comparer. (d) Expliquer, avec
le facteur $\gamma^4 =
(E/mc^2)^4$, pourquoi le successeur des électrons est un collisionneur
\emph{linéaire} ou un anneau bien plus grand, et pourquoi les anneaux à
protons survivent.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:covariant-electromagnetism:12}
Le cyclotron relativiste. Une charge dans un $\vect B$ uniforme, à
vitesse relativiste. (a) À partir de $\dd\vect p/\dd t = q\vect v\wedge\vect
B$ avec $E$ constante, montrer que l'orbite est un cercle parcouru à
$\omega = qB/\gamma m$~: la fréquence cyclotron chute avec l'énergie.
(b) Un cyclotron classique pousse à $\omega_0 = qB/m$ fixé~: montrer
qu'après $N$ tours, $\gamma - 1$ croissant linéairement jusqu'à sa
valeur finale $\Gamma$, le déphasage accumulé atteint le quart de
période de la radiofréquence quand $\Gamma \approx 1/2N$~; pour
$N = 100$ tours, à quelle énergie cinétique de proton cela
correspond-il, et comment cela se compare-t-il au plafond historique
des cyclotrons classiques (quelque \qty{20}{MeV}, obtenus avec de la
marge et de la tension en plus)~? (c) Deux remèdes existent~: faire
varier la fréquence (synchrocyclotron) ou façonner $B(r)$ pour qu'il
croisse comme $\gamma$ (cyclotron isochrone)~: expliquer chacun en une
phrase. (d) Le cyclotron isochrone du PSI délivre des protons de
\qty{590}{MeV}~: de quel facteur son champ au bord dépasse-t-il le
champ central~?
\end{exercise}

\section{Problème~: le magnétisme à pas d'escargot}

\begin{problem}[{Problème du week-end --- comment un déséquilibre de
$10^{-24}$ fait tourner tous les moteurs}]\label{pb:b3:covariant-electromagnetism:1}
Les électrons dérivent dans le fil d'une lampe plus lentement que ne
coule le miel, et le $v^2/c^2$ de cette dérive vaut $10^{-24}$ --- et
pourtant la force magnétique qu'elle produit soulève des voitures dans
les casses. Ce problème fait la comptabilité complète, dans les deux
référentiels, d'un fil rectiligne et d'une charge en mouvement, et
débusque la relativité dans chaque électroaimant. Données~: fil de
cuivre, section $S = \qty{1.0}{mm^2}$, courant $I = \qty{10}{A}$,
densité d'électrons de conduction $n = \qty{8.5e28}{m^{-3}}$~; charge
d'épreuve $q = \qty{1}{\micro C}$ à la distance $d = \qty{1.0}{cm}$, se
déplaçant parallèlement au fil à $v = \qty{e5}{m/s}$ (une vitesse de
faisceau d'ions, pour garder des nombres visibles)~;
$\mu_0 = 4\pi\times10^{-7}\,\unit{H/m}$.

\medskip\noindent\textbf{Partie I --- La comptabilité au laboratoire.}
\begin{enumerate}
  \item Calculer la vitesse de dérive $u$ des électrons, et $u^2/c^2$.
  \item Modéliser le fil par deux densités linéiques superposées~: le
        réseau $+\lambda$ au repos et les électrons $-\lambda$ dérivant
        à $u$, avec $I = \lambda u$. Calculer $\lambda$.
  \item Calculer $B$ à la position de la charge et la force magnétique
        $F = qvB$ qu'elle subit --- direction comprise, pour $v$
        parallèle au courant conventionnel $I$ (rappel~: des courants
        parallèles s'attirent).
  \item Pourquoi n'y a-t-il aucune force \emph{électrique} au
        laboratoire~?
  \item Le champ électrique de ce même fil s'il portait une charge
        nette d'un seul électron en excès par mètre~: comparer la force
        qu'il exercerait sur $q$ à la force magnétique de la question
        3, et conclure avec quelle précision la neutralité doit être
        mesurée avant de pouvoir attribuer un effet au magnétisme.
  \item Une charge de \qty{1}{\micro C} à \qty{e5}{m/s}~: la force de
        la question 3 est-elle mesurable~? (Comparer au poids d'un
        grain de sable, $\sim\qty{e-6}{N}$.)
\end{enumerate}

\medskip\noindent\textbf{Partie II --- On change de siège.}
Passons au référentiel de la charge d'épreuve (vitesse $v$ le long du fil).
\begin{enumerate}[resume]
  \item Dans ce référentiel, quelles sont les vitesses du réseau et des
        électrons (qui dérivent à l'opposé de $I$ et gagnent donc de la
        vitesse dans ce changement de référentiel --- les composer
        correctement)~?
  \item Intégré sur la section du fil, le quadricourant par unité de
        longueur vaut $(c\lambda_{\text{tot}}, I, 0, 0)$ avec
        $\lambda_{\text{tot}} = 0$ ici. Le transformer et montrer que
        la densité linéique du fil dans le nouveau référentiel vaut
        \[
        \lambda' = -\gamma_v\,\frac{vI}{c^2} .
        \]
  \item Évaluer $\lambda'$, en coulombs par mètre et en charges
        d'électron par mètre. Quel flot est devenu plus dense, et
        pourquoi (contracter chaque flot séparément si vous préférez)~?
  \item Calculer le champ électrique $E' = \lambda'/2\pi\varepsilon_0d$
        à la charge, et la force électrique $qE'$ qu'elle subit.
  \item Comparer $qE'$ au $qvB$ du laboratoire de la partie I, et
        expliquer le facteur résiduel $\gamma_v \approx 1 + \num{5.6e-8}$
        (la loi de transformation de quel \omterm{def:b3:covariant-electromagnetism:fourvector}{quadrivecteur} relie les
        forces~?).
  \item Dans ce référentiel il y a aussi un champ magnétique (le fil
        porte toujours un courant)~: pourquoi n'exerce-t-il aucune
        force ici~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La morale des nombres.}
\begin{enumerate}[resume]
  \item Le déséquilibre relatif $|\lambda'|/\lambda$~: le calculer et
        l'écrire sous la forme $\gamma_v vu/c^2$.
  \item Comment un effet valant $10^{-19}$ de la charge du fil peut-il
        produire une force macroscopique~? (Par quel nombre énorme
        est-il multiplié~?)
  \item Deux fils parallèles portant \qty{10}{A} chacun à \qty{1}{cm}~:
        calculer la force par mètre et la comparer à la valeur de
        référence $2 \times 10^{-7}\,I_1I_2/d$ newtons par mètre.
  \item Un électroaimant, c'est dix mille fois cette histoire~:
        estimer le champ d'une bobine de \qty{e4}{} spires par mètre
        parcourue par \qty{10}{A} (formule du solénoïde du volume de
        première année), et la force par centimètre carré qu'elle
        exerce sur du fer ($\sim B^2/2\mu_0$).
  \item Si l'on coupait la relativité ($c \to \infty$ dans la
        transformation), que resterait-il de $\lambda'$, de la force de
        $B$, des moteurs et des aimants de casse~?
  \item Pourquoi la charge d'épreuve \emph{au repos} près du fil ne
        ressent-elle rien, alors même que les électrons défilent devant
        elle~? (Quelle compensation la protège, et à quelle précision~?)
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- Au-delà du fil.}
\begin{enumerate}[resume]
  \item Le volume de deuxième année tirait les champs magnétiques du
        théorème d'Ampère, les courants étant des sources données. Que
        ce chapitre ajoute-t-il à ce récit --- qu'\emph{est} le champ
        magnétique, vu de ce problème~?
  \item Les aimants de fer n'ont pas de pile~: qu'est-ce qui tient le
        rôle du courant dans le magnétisme permanent (une phrase~; la
        réponse microscopique honnête est quantique et attend cinq
        chapitres)~?
  \item Un faisceau d'électrons seul dans le vide (sans réseau)~: un
        observateur qui l'accompagne le voit-il s'attirer ou se
        repousser plus qu'un observateur du laboratoire~? Réconcilier
        les récits des deux référentiels sur l'auto-pincement d'un
        faisceau.
  \item Rendre ce dernier point quantitatif~: montrer que deux
        faisceaux parallèles de même charge se déplaçant ensemble à
        $\beta$ se repoussent avec une force nette (répulsion
        électrique moins attraction magnétique) réduite exactement de
        $1/\gamma^2$ par rapport à sa valeur au repos --- et dites
        quel référentiel rend le facteur évident.
  \item L'énergie d'une lampe arrive à une vitesse proche de $c$ alors
        que ses électrons dérivent d'un mètre par heure~: qu'est-ce qui
        transporte réellement l'énergie le long du cordon (rappeler
        le vecteur de Poynting du volume de deuxième année)~?
  \item Estimer $vu/c^2$ pour les électrons de ce problème et une
        charge d'épreuve à allure de piéton ($v = \qty{1}{m/s}$)~: même
        là, la force est mesurable au premier ordre avec une aiguille
        aimantée --- qui a montré qu'un courant dévie une boussole, et
        en quelle année~?
  \item Résumer le résultat nommé~: un fil neutre de \qty{10}{A},
        observé depuis un siège se déplaçant à \qty{e5}{m/s}, porte
        $-\qty{e-11}{C/m}$ de charge relativiste --- et cette erreur de
        comptabilité d'ordre $10^{-24}$ due à la \omterm{prop:b3:relativistic-kinematics:contraction}{contraction des
        longueurs}, multipliée par $10^{28}$ électrons, fait toute la
        force magnétique du laboratoire.
\end{enumerate}
\end{problem}
