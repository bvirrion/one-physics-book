\chapter{La théorie des perturbations}\label{ch:b3:perturbation-theory}

Une poignée seulement de problèmes quantiques se dissolvent
exactement~: la boîte, l'oscillateur, l'hydrogène, le système à deux
niveaux. Tout le reste --- un atome dans un champ, une liaison
moléculaire qui se raidit, un niveau bousculé par un voisin --- est
l'un de ces squelettes solubles portant un petit terme
supplémentaire. La \omterm{thm:b3:perturbation-theory:corrections}{théorie des perturbations} est l'art de traiter ce
terme pour ce qu'il est~: une correction, calculée ordre par ordre en
sa petitesse. Sa règle du premier ordre est une moyenne en une ligne~;
celle du second ordre explique pourquoi les niveaux se
\emph{repoussent}, pourquoi tout atome est polarisable et pourquoi
l'attraction de van der Waals est universelle~; sa variante dégénérée
traite les cas délicats où des niveaux presque égaux se réorganisent
entièrement~; et sa forme dépendante du temps --- la \omterm{thm:b3:perturbation-theory:golden}{règle d'or de
Fermi} --- est l'équation qui gouverne toute raie d'absorption, tout
taux de désintégration et toute section efficace mesurés en
laboratoire. En récompense, le chapitre s'achève en calculant la
couleur du ciel.

\section{Petits décalages~: la théorie non dégénérée}

\begin{theorem}[Corrections perturbatives]\label{thm:b3:perturbation-theory:corrections}
\index{théorie des perturbations}
Soit $\hat H = \hat H_0 + \hat V$ avec $\hat H_0$ résolu
($\hat H_0\ket n = E_n\ket n$, spectre non dégénéré) et $\hat V$
petit. Alors, ordre par ordre en $\hat V$,
\[
E_n' = E_n + \bra n\hat V\ket n
 + \sum_{m \neq n}\frac{|\bra m\hat V\ket n|^2}{E_n - E_m}
 + \cdots
\]
et l'état acquiert les admixtures $\ket{n'} = \ket n + \sum_{m\neq
n}\dfrac{\bra m\hat V\ket n}{E_n - E_m}\,\ket m + \cdots$
\emph{Premier ordre}~: le décalage n'est que la moyenne de la
perturbation dans l'état non perturbé. \emph{Second ordre}~: les
couplages aux autres niveaux poussent $E_n$ \emph{loin} d'eux ---
chaque $m$ au-dessus pousse vers le bas, chacun au-dessous vers le
haut --- de sorte que le décalage au second ordre de l'état
fondamental est toujours négatif. Validité~: les fractions de mélange
$|\bra m\hat V\ket n/(E_n - E_m)|$ doivent être petites.
\end{theorem}

\begin{proof}[Démonstration partielle]
Développons $E' = E + \epsilon_1 + \epsilon_2 + \cdots$ et $\ket{\psi} =
\ket n + \ket{\psi_1} + \cdots$ en puissances de $\hat V$, reportons
dans $(\hat H_0 + \hat V)\ket\psi = E'\ket\psi$ et identifions les
ordres. Projeter l'équation du premier ordre sur $\bra n$ donne $\epsilon_1 =
\bra n\hat V\ket n$~; la projeter sur $\bra m$ ($m \neq n$) donne les
coefficients d'admixture~; les reporter dans l'équation du second
ordre puis projeter de nouveau sur $\bra n$ donne $\epsilon_2$.
L'énoncé sur le signe pour l'état fondamental~: tout dénominateur
$E_0 - E_m$ est négatif tandis que les numérateurs sont des carrés.
\end{proof}

\begin{proposition}[Une perturbation à réponse exacte]\label{prop:b3:perturbation-theory:oscillator-field}
Un oscillateur de charge $q$ dans un champ uniforme $\mathcal E$ a
$\hat V = -q\mathcal E\hat x$. Compléter le carré le résout
exactement~: le potentiel est la même parabole, translatée, avec tous
les niveaux abaissés de $q^2\mathcal E^2/2m\omega^2$. La \omterm{thm:b3:perturbation-theory:corrections}{théorie des
perturbations} doit s'accorder --- et elle s'accorde~: le premier ordre
s'annule ($\langle\hat
x\rangle_n = 0$), et la somme du second ordre, où $\hat x$ ne couple
$n$ qu'à $n \pm 1$, donne exactement
$-q^2\mathcal E^2/2m\omega^2$ pour chaque niveau
(\cref{exo:b3:perturbation-theory:2}). Un cas soluble, vérifié des
deux façons, est le meilleur étalonnage qu'une méthode puisse avoir.
\end{proposition}

\begin{proof}
Les deux éléments de matrice et leurs dénominateurs donnent
\[
\frac{|\bra{n+1}\hat x\ket{n}|^2}{-\hbar\omega}
 + \frac{|\bra{n-1}\hat x\ket n|^2}{+\hbar\omega}
 = \frac{\hbar}{2m\omega}\,\frac{-(n+1) + n}{\hbar\omega}
 = -\frac{1}{2m\omega^2} ;
\]
qu'il faut multiplier par $q^2\mathcal E^2$.
\end{proof}

\section{Quand les niveaux sont dégénérés}

\begin{theorem}[Théorie des perturbations dégénérée]\label{thm:b3:perturbation-theory:degenerate}
\index{théorie des perturbations dégénérée}
Si $E_n$ est dégénéré, la formule ci-dessus divise par zéro --- la
perturbation peut réorganiser complètement les états dégénérés, et le
remède est de la laisser faire~: \emph{diagonaliser $\hat V$ restreint
au sous-espace dégénéré}. Ses \omterm{def:b3:quantum-formalism:observable}{valeurs propres} sont les décalages du
premier ordre~; ses vecteurs propres sont les bons états d'ordre zéro,
les combinaisons que la perturbation choisit elle-même.
\end{theorem}
\admitted

\begin{example}[L'effet Stark linéaire]\label{ex:b3:perturbation-theory:stark}
\index{effet Stark}
Le niveau $n = 2$ de l'hydrogène abrite les états dégénérés $2s$ et
$2p$. Dans un champ $\mathcal E\vect e_z$, $\hat V = e\mathcal E\hat z$
couple $2s$ à $2p_0$ avec $\bra{2s}\hat z\ket{2p_0} = -3a_0$~:
diagonaliser le bloc $2\times2$ donne les décalages
\[
\Delta E = \pm\,3e a_0\mathcal E ,
\]
\emph{linéaires} en le champ --- alors que l'état fondamental, non
dégénéré, ne se décale qu'au second ordre. À $\mathcal E = \qty{e7}{V/m}$~: $\pm
\qty{1.6}{meV}$, un écart aisément résolu. Les états qui produisent la
séparation sont $(\ket{2s} \mp \ket{2p_0})/\sqrt2$ --- des hybrides
dotés d'un dipôle électrique permanent, possibles seulement parce que
la dégénérescence laissait l'atome libre de se polariser à l'ordre
zéro. (La même hybridation $s$--$p$, provoquée par des voisins plutôt
que par un champ, est la chimie du carbone de toute molécule
organique.)
\end{example}

\section{Transitions~: la règle d'or}

\begin{theorem}[Perturbations dépendant du temps]\label{thm:b3:perturbation-theory:golden}
\index{règle d'or de Fermi}
Branchons $\hat V(t) = \hat V\cos\omega t$ à $t = 0$. Au premier
ordre, la probabilité de trouver le système dans $\ket f$ à l'instant
$t$, étant parti de $\ket i$, vaut
\[
\mathcal P_{i\to f}(t) = \frac{|\bra f\hat V\ket i|^2}{4\hbar^2}\;
\frac{\sin^2\!\big[(\omega_{fi} - \omega)\,t/2\big]}
{\big[(\omega_{fi} - \omega)/2\big]^2} ,
\qquad \omega_{fi} = \frac{E_f - E_i}{\hbar} :
\]
une résonance fortement piquée en $\hbar\omega = E_f - E_i$ ---
absorption quand $E_f > E_i$, \emph{émission stimulée} pour le terme
compagnon avec $E_f < E_i$. Lorsque les états finaux forment un
continuum de densité $\rho(E)$, la croissance du pic devient un
\emph{taux} constant~:
\[
\Gamma_{i\to f} = \frac{2\pi}{\hbar}\,
|\bra f\hat V\ket i|^2\,\rho(E_f)
\]
--- la \emph{\omterm{thm:b3:perturbation-theory:golden}{règle d'or de Fermi}}, formule maîtresse des taux de
désintégration, des coefficients d'absorption et des sections
efficaces.
\end{theorem}

\begin{proof}[Démonstration partielle]
Amplitude au premier ordre~: $c_f(t) = -\tfrac{\iu}{\hbar}\int_0^t\bra
f\hat V(t')\ket i\,\eu^{\iu\omega_{fi}t'}\dd t'$~; en gardant
l'exponentielle quasi résonnante et en intégrant, on obtient la forme
en $\sin^2$ annoncée. Pour le continuum~: en fonction du désaccord
$x$, le facteur de résonance a pour hauteur $t^2$ et pour largeur
$\sim 2\pi/t$ --- d'aire $2\pi t$ --- si bien que sommer sur un
continuum régulier le remplace par $2\pi t\,\rho$~: une probabilité
linéaire en $t$, c'est-à-dire un taux.
\end{proof}

\begin{example}[Ce que fait tourner la règle d'or]\label{ex:b3:perturbation-theory:applications}
Règles de sélection~: une transition n'a lieu que si l'élément de
matrice $\bra f\hat V\ket i$ survit --- pour la lumière, $\hat V \propto \hat
z$, dont la parité et le moment cinétique tuent tout sauf
$\Delta l = \pm1$, $\Delta m = 0, \pm1$~: les règles invoquées depuis
le \cref{ch:b3:quantum-angular-momentum}, désormais théorèmes. Taux~:
la désintégration spontanée de l'état $2p$ de l'hydrogène se calcule à
\qty{1.6}{ns} (\cref{exo:b3:perturbation-theory:8})~; la transition
maser de l'ammoniac, avec son $\omega^3$ micro-onde, prend des mois
--- le $\omega^3$ des taux de rayonnement explique que les raies radio
(\cref{ex:b3:spin-two-level:hyperfine}) vivent des millions d'années
quand les raies ultraviolettes éclatent en nanosecondes. Spectres
d'absorption, photoionisation, désintégration bêta nucléaire
(\cref{ch:b3:nuclear-physics}), sections efficaces de diffusion
(\cref{ch:b3:scattering-theory})~: tout cela est cette unique formule
avec d'autres éléments de matrice et d'autres comptes d'états.
\end{example}

\begin{method}[Choisir le bon outil perturbatif]\label{met:b3:perturbation-theory:method}
(1) Question statique, niveau non dégénéré~: moyenner $\hat V$
(premier ordre)~; si cela s'annule par symétrie, passer au second
ordre --- s'attendre à une répulsion de niveaux. (2) Niveau
dégénéré~: diagonaliser d'abord $\hat V$ dans le bloc dégénéré~; les
combinaisons adaptées à la symétrie produisent la séparation. (3) Taux
de transition~: règle d'or --- un élément de matrice, une \omterm{prop:b3:schrodinger-three-dimensions:counting}{densité
d'états}~; vérifier les règles de sélection avant tout calcul. (4)
Toujours tester le développement~: fraction de mélange
$|V_{mn}/(E_n - E_m)| \ll 1$, sinon diagonaliser exactement (formule à
deux niveaux). (5) Les limites solubles exactement (oscillateur dans
un champ) sont des étalonnages gratuits~: s'en servir.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:1}
Une particule dans une boîte $[0, a]$ subit le potentiel supplémentaire $V(x) =
\lambda x/a$. (a) Calculer le décalage au premier ordre de chaque
niveau (utiliser $\langle x\rangle = a/2$ dans tout état de la boîte
--- le justifier). (b) Pourquoi le décalage est-il ici le même pour
tous les niveaux~? (c) Quelle est la variation au premier ordre des
\emph{écarts}, et pourquoi une mesure de fréquences de transition
pourrait-elle manquer entièrement cette perturbation~? (d) Quand le
premier ordre cesse-t-il d'être digne de confiance~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:2}
Mener en entier le calcul du second ordre de
\cref{prop:b3:perturbation-theory:oscillator-field}~: éléments de
matrice de $\hat x$, les deux termes, la disparition de $n$, et
l'accord exact avec la complétion du carré. Pourquoi la correction du
premier ordre de l'\emph{état} ne s'annule-t-elle pas alors que celle
de l'énergie s'annule~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:3}
Le \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltonien} à deux niveaux $\begin{pmatrix}E_1 & v\\ v &
E_2\end{pmatrix}$, $E_1 < E_2$, $v$ réel. (a) Trouver les \omterm{def:b3:quantum-formalism:observable}{valeurs
propres} exactes. (b) Développer pour $|v| \ll E_2 - E_1$ et
identifier la formule du second ordre. (c) Montrer que les niveaux se
repoussent~: l'écart dépasse toujours $E_2 - E_1$. (d) À $E_1 = E_2$~:
que donne la \omterm{thm:b3:perturbation-theory:corrections}{théorie des perturbations}, que donne la réponse exacte,
et quel théorème du chapitre les réconcilie~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:4}
Pour \cref{thm:b3:perturbation-theory:corrections}~: (a) montrer que
la correction du premier ordre de l'état est orthogonale à $\ket n$~;
(b) montrer que le coefficient de mélange de $\ket m$ vaut
$V_{mn}/(E_n - E_m)$ et énoncer le critère de validité~; (c) un niveau
distant de $\qty{1}{eV}$ de son plus proche voisin lui est couplé par
\qty{0.1}{eV}~: estimer la probabilité d'admixture~; (d) même question
avec un écart de \qty{10}{meV} --- quel outil doit remplacer le
développement~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:5}
Liaisons anharmoniques. Ajouter $\hat V = \beta\hat x^3 + \gamma\hat x^4$ à
l'oscillateur. (a) Montrer que le terme en $x^3$ ne décale aucun
niveau au premier ordre (parité). (b) Montrer que le terme en $x^4$
donne $\Delta E_n =
3\gamma(\hbar/2m\omega)^2(2n^2 + 2n + 1)$ (développer $\hat x^4$ en
\omterm{def:b3:harmonic-oscillator:ladder}{opérateurs d'échelle}~; seuls les termes «~équilibrés~» survivent).
(c) Pour une liaison réelle le $\gamma$ effectif est négatif (le puits
s'assouplit vers l'extérieur)~: montrer que l'écart entre niveaux
\emph{diminue} alors avec $n$.
(d) Quel trait observé des spectres moléculaires
(\cref{exo:b3:harmonic-oscillator:4}) cela reproduit-il~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:6}
Dans la boîte carrée à deux dimensions, la paire dégénérée $\ket{1,2},
\ket{2,1}$ est perturbée par $\hat V = \lambda\,xy$. (a) Expliquer
pourquoi la théorie non dégénérée échoue. (b) Calculer la matrice
$2\times2$ de $\hat V$ dans la paire (les intégrales se factorisent~; $\int_0^a
x\sin(\pi x/a)\sin(2\pi x/a)\dd x = -8a^2/9\pi^2$). (c) Trouver les
séparations et les bons états d'ordre zéro. (d) Comparer à l'analyse
de symétrie de \cref{exo:b3:quantum-formalism:9}~: quelle réponse la
symétrie donnait-elle gratuitement~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:7}
\omterm{ex:b3:perturbation-theory:stark}{Effets Stark}, linéaire et quadratique. (a) À l'aide de
\cref{ex:b3:perturbation-theory:stark}, calculer la séparation de
$n = 2$ à $\mathcal E = \qty{2.5e6}{V/m}$. (b) Pourquoi l'état
fondamental ne montre-t-il \emph{aucun} décalage linéaire (deux
raisons~: la parité, et l'absence de partenaire dégénéré)~? (c) Son
décalage quadratique définit la polarisabilité,
$\Delta E = -\tfrac12\alpha\mathcal E^2$~: estimer
$\alpha \sim e^2a_0^2/E_{\text{I}}$ et l'évaluer en unités de
$a_0^3$ (la réponse exacte est $4.5\,a_0^3 \times
4\pi\varepsilon_0$). (d) De quelle propriété quotidienne de la matière
--- sa réponse diélectrique --- venir-vous de calculer le germe
atomique~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:8}
Émission spontanée (avec une donnée admise)~: un état excité se
désintègre au taux $A = \omega^3|d_{fi}|^2/3\pi\varepsilon_0\hbar c^3$,
où des éléments de matrice du type $d_{fi} = e\bra f\hat z\ket i$
fixent le dipôle. (a) Pour l'hydrogène $2p \to 1s$~: avec $|d| \approx 0.74\,
ea_0$ et $\lambda = \qty{121.6}{nm}$, calculer $A$ et la durée de vie.
(b) Transposer à la raie D du sodium. (c) Transposer à la transition
hyperfine à 21 cm ($|d| \to$ un moment magnétique, taux abaissé d'un
facteur en $\alpha^2$ supplémentaire~; admettre $A \approx
\qty{2.9e-15}{s^{-1}}$)~: vérifier les «~dix millions d'années~». (d)
D'après le $\omega^3$~: pourquoi les raies ultraviolettes sont-elles
rapides, les raies radio éternelles, et pourquoi cela a-t-il rendu la
\omterm{ex:b3:spin-two-level:hyperfine}{raie à 21 cm} \emph{prévisible} mais presque indétectable en
laboratoire~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:9}
Les règles de sélection comme intégrales. (a) Montrer que $\bra{100}\hat z\ket{200} =
0$ par parité. (b) Montrer que $\bra{100}\hat z\ket{21m} = 0$ sauf si $m =
0$ (l'intégrale sur $\varphi$). (c) Énoncer les règles qui en résultent, $\Delta
l = \pm1$, $\Delta m = 0, \pm1$, et leur lecture physique (le spin du
photon). (d) L'état $2s$ ne peut atteindre $1s$ par aucune voie
dipolaire~: que cela prédit-il pour sa durée de vie, et quelle voie
«~interdite~» à deux photons la nature emprunte-t-elle réellement~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:10}
La polarisabilité, honnêtement. Le décalage au second ordre de l'état
fondamental de l'hydrogène dans un champ vaut $\Delta E = -e^2\mathcal E^2\sum_{m\neq
0}|\bra m\hat z\ket 0|^2/(E_m - E_0)$. (a) Majorer la somme en
remplaçant tout dénominateur par le plus petit écart $E_2 - E_1 =
\tfrac34 E_{\text{I}}$ et en utilisant la relation de fermeture $\sum_m|\bra
m\hat z\ket0|^2 = \langle z^2\rangle_0 = a_0^2$~: obtenir $\alpha
\le \tfrac{16}3\,a_0^3$ (en unités de type gaussien de
$4\pi\varepsilon_0$). (b) Comparer à la valeur exacte $\tfrac92 a_0^3$.
(c) À partir de $\alpha$, estimer l'indice de réfraction de
l'hydrogène gazeux à la densité atmosphérique ($n - 1 \approx N\alpha/2\varepsilon_0
\cdot e^2\dots$ --- utiliser $n - 1 = N\alpha_{\text{SI}}/2
\varepsilon_0$) et comparer à la valeur mesurée $\num{1.3e-4}$.
(d) Énoncer en une phrase la chaîne atome $\to$ polarisabilité
$\to$ réfraction que la partie II du problème du week-end empruntera.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:11}
Croisements évités et balayages lents. Un système à deux niveaux a
pour énergies nues $\pm\lambda t$, qui se croisent en $t = 0$, couplées
par $v$. (a) Représenter les niveaux exacts en fonction de $t$~: un
croisement évité d'écart $2|v|$. (b) Si le balayage est \emph{lent},
le système suit la courbe inférieure (théorème adiabatique, admis)~:
dans quel état finit-il~? (c) Le critère de Landau--Zener compare la
durée du balayage à travers la région d'écart, $\sim v/\lambda$, à
l'horloge interne $\hbar/v$~: donner la condition de suivi
adiabatique. (d) Deux usages~: l'état d'un qubit transféré par un
balayage lent en fréquence~; une collision atomique qui fait sauter
une charge d'un noyau à l'autre --- expliquer l'un des deux en deux
phrases.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:12}
Le second ordre est pessimiste, et autres théorèmes. (a) Démontrer que
la correction au second ordre de l'état \emph{fondamental} est
toujours négative. (b) Conclure (avec
\cref{exo:b3:harmonic-oscillator:10}) pourquoi les forces de van der
Waals entre atomes dans leur état fondamental sont toujours
attractives. (c) Montrer que la théorie du premier ordre
\emph{surestime} toute énergie d'état fondamental~: c'est une borne
variationnelle (évaluer $\bra{\psi_0}\hat
H\ket{\psi_0}$ avec l'état non perturbé comme fonction d'essai). (d)
Un contrôle de cohérence sur \cref{ex:b3:perturbation-theory:stark}~:
pourquoi l'\omterm{ex:b3:perturbation-theory:stark}{effet Stark} \emph{linéaire} ne viole-t-il pas (a)~?
\end{exercise}

\section{Problème~: la couleur du ciel}

\begin{problem}[{Problème du week-end --- la diffusion Rayleigh par
des atomes perturbés}]\label{pb:b3:perturbation-theory:1}
Pourquoi le ciel du jour est-il bleu, le couchant rouge et le nuage
blanc~? La réponse complète est une chaîne à travers ce chapitre~: une
onde lumineuse perturbe chaque molécule de l'air~; le dipôle induit
rayonne à son tour (le rayonnement dipolaire du volume de deuxième
année)~; et l'interférence décide de ce qui survit. Données~:
polarisabilité moléculaire de l'air
$\alpha/4\pi\varepsilon_0 \approx \qty{2.2e-30}{m^3}$~; densité en
nombre $N = \qty{2.5e25}{m^{-3}}$~; domaine visible
$450$--\qty{700}{nm}.

\medskip\noindent\textbf{Partie I --- La molécule perturbée.}
\begin{enumerate}
  \item Un champ statique $\mathcal E$ décale de
        $-\tfrac12\alpha\mathcal E^2$ l'état fondamental d'une
        molécule~: relier cet énoncé à la \omterm{thm:b3:perturbation-theory:corrections}{théorie des perturbations} au
        second ordre (quelle formule, et pourquoi négatif~?).
  \item Le même $\alpha$ donne le dipôle induit $p =
        \alpha\mathcal E$~: pour un champ solaire $\mathcal E \sim
        \qty{800}{V/m}$, calculer le dipôle induit d'une molécule
        d'azote et le comparer à $ea_0$.
  \item Pourquoi peut-on traiter le champ optique (de fréquence
        $\num{5e14}$ Hz) avec la polarisabilité \emph{statique} pour
        des molécules dont les transitions électroniques se situent
        dans l'ultraviolet~? (Comparer $\hbar\omega$ à l'écart~; c'est
        la limite très hors résonance du dénominateur de la règle
        d'or.)
  \item Un dipôle oscillant $p_0\cos\omega t$ rayonne la puissance
        moyenne $P = p_0^2\omega^4/12\pi\varepsilon_0c^3$ (volume de
        deuxième année). En combinant avec $p_0 = \alpha\mathcal
        E_0$~: comment la puissance diffusée dépend-elle de $\omega$
        à éclairement fixé~?
  \item Calculer le rapport des puissances diffusées à \qty{450}{nm}
        et \qty{700}{nm}.
  \item En une phrase~: pourquoi le ciel est-il bleu et non violet
        (deux ingrédients~: le spectre du Soleil qui décroît vers le
        violet, et la sensibilité de l'œil)~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- D'une molécule au ciel
entier.}
\begin{enumerate}[resume]
  \item La section efficace de diffusion vaut $\sigma =
        \dfrac{8\pi}{3}\Big(\dfrac{\alpha}{4\pi\varepsilon_0}
        \Big)^2\dfrac{\omega^4}{c^4}$~: vérifier ses dimensions et
        l'évaluer à \qty{550}{nm}.
  \item La longueur d'atténuation vaut $\ell = 1/N\sigma$~:
        l'évaluer à \qty{550}{nm} et comparer à l'épaisseur de
        l'atmosphère ($\sim\qty{8}{km}$ équivalents à la densité du
        niveau de la mer).
  \item À \qty{450}{nm} contre \qty{700}{nm}, quelle fraction d'un
        rayon solaire vertical est diffusée hors du faisceau~? (Prendre $1 -
        \eu^{-L/\ell}$ avec $L = \qty{8}{km}$.)
  \item Au couchant le trajet est trente fois plus long~: recalculer
        la survie du bleu et expliquer la couleur du Soleil bas ---
        et celle de la lumière qui rougit la Lune lors d'une éclipse.
  \item Pourquoi la lumière du ciel est-elle \emph{polarisée} à
        $90^\circ$ du Soleil (rappeler le diagramme de
        rayonnement dipolaire~: aucune émission le long de l'axe du
        dipôle)~?
  \item Les abeilles et, dit-on, les Vikings naviguaient à cette
        polarisation~: que lui fait un ciel couvert, et pourquoi~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Pourquoi le ciel n'est pas
plus lumineux.}
\begin{enumerate}[resume]
  \item Un paradoxe~: dans un milieu parfaitement uniforme, les
        ondelettes émises par des éléments de volume voisins
        interfèrent destructivement sur les côtés, et \emph{aucune}
        lumière ne serait diffusée (seule la réfraction survivrait).
        Qu'est-ce qui brise réellement l'uniformité de l'air ---
        qu'est-ce qui y est réparti au hasard~?
  \item Les fluctuations de densité d'un gaz parfait font
        \emph{s'ajouter} les intensités diffusées par des molécules
        indépendantes~: dites pourquoi l'indépendance (positions
        aléatoires, séparations de l'ordre de la longueur d'onde) tue
        l'interférence destructive.
  \item À partir de votre $\ell$ à \qty{550}{nm}~: quelle fraction de
        la lumière solaire une atmosphère de beau temps diffuse-t-elle
        --- ce nombre s'accorde-t-il avec la luminosité quotidienne du
        ciel face au Soleil~?
  \item Une gouttelette de nuage (\qty{10}{\micro m}) contient
        $\sim10^{12}$ molécules dans une longueur d'onde~: leurs
        ondelettes s'ajoutent \emph{en cohérence}. Comment la
        puissance diffusée varie-t-elle alors avec le nombre de
        molécules, et pourquoi la sélection de couleur en $\omega^4$
        disparaît-elle (toutes les longueurs d'onde diffusent
        fortement)~: de quelle couleur est le nuage~?
  \item Le lait, le brouillard et la peinture blanche sont blancs
        pour la même raison~: énoncer la règle générale --- les
        diffuseurs petits devant $\lambda$ colorent la lumière, ceux
        qui sont grands devant $\lambda$ la blanchissent.
  \item Opalescence critique~: près du point critique d'un liquide
        (les chapitres sur les changements d'état du volume de
        première année, et le \cref{ch:b3:phase-transitions} plus
        loin), les fluctuations de densité atteignent des échelles
        optiques et le fluide devient laiteux~: relier à la
        question 13.
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- La chaîne, achevée.}
\begin{enumerate}[resume]
  \item Les ondelettes diffusées vers l'avant, elles, ne se
        compensent \emph{pas}~: elles interfèrent avec le faisceau et
        ralentissent sa phase --- c'est l'indice de réfraction. Avec
        $n - 1 = N\alpha/2\varepsilon_0$ (le $\alpha$ étant en SI),
        évaluer $n - 1$ pour l'air et comparer à la valeur mesurée
        \num{2.9e-4}.
  \item Une seule constante $\alpha$, calculée par la \omterm{thm:b3:perturbation-theory:corrections}{théorie des
        perturbations} au second ordre, fixe ainsi trois phénomènes à
        la fois~: les nommer (décalage dans un champ statique~; le
        bleu du ciel~; la courbure de la lumière dans l'air).
  \item L'absorption par l'ozone, les aérosols et la diffusion
        multiple compliquent les ciels réels~: nommer un observable
        pour chacun (les couleurs du crépuscule~; l'horizon laiteux~;
        la luminosité résiduelle du ciel au zénith après le coucher).
  \item Pourquoi la Lune, sans air, porte-t-elle un ciel diurne
        \emph{noir} --- et qu'ont ainsi confirmé toutes les
        photographies prises depuis sa surface sur l'origine du
        nôtre~?
  \item Le ciel diurne de Mars est caramel et ses couchers de soleil
        sont \emph{bleus}~: son air ténu diffuse peu, et ce sont les
        grains de poussière micrométriques en suspension qui font le
        travail. À l'aide de la règle de la question 18, expliquer
        comment la poussière renverse la logique des couleurs.
  \item Pour les rayons X, $\hbar\omega$ dépasse de loin toute
        transition moléculaire~: les électrons répondent comme s'ils
        étaient libres et la diffusion (Thomson) perd son $\omega^4$.
        Qu'advient-il du mécanisme du «~bleu du ciel~» dans ce
        régime --- et pourquoi cette réponse plate fait-elle
        justement des rayons X des sondes propres de la densité
        électronique~?
  \item Résumer le résultat nommé~: une polarisabilité de
        $\qty{2.2e-30}{m^3}$, élevée au carré et multipliée par
        $\omega^4$, donne une longueur de diffusion de l'ordre de
        \qty{60}{km} dans le vert --- assez longue pour que le ciel de
        midi soit doux, assez courte pour que les couchants
        rougissent et que les cieux soient bleus, avec le motif de
        polarisation d'une forêt de dipôles excités.
\end{enumerate}
\end{problem}
