\chapter{L'ensemble canonique}\label{ch:b3:canonical-ensemble}

Les systèmes isolés sont une fiction de théoricien~: les échantillons
réels siègent dans des thermostats, des pièces, des océans d'air ---
en contact avec un réservoir qui fixe non leur énergie mais leur
\emph{température}. Le volume de deuxième année a rencontré
empiriquement la loi qui en résulte~: la probabilité d'un état
d'énergie $E$ porte le facteur $\eu^{-E/k_{\text{B}}T}$. Ce chapitre
obtient ce facteur de Boltzmann en quatre lignes à partir du comptage
du \cref{ch:b3:microcanonical-ensemble}, puis bâtit la machine qui
fait de la physique statistique une discipline industrielle~: la
\emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{fonction de partition}} $Z$, une somme unique d'où l'énergie,
l'entropie, la pression, la capacité thermique et les fluctuations
tombent toutes par dérivation. Les premières victoires de la machine
sont racontées ici~: pourquoi les capacités thermiques meurent à basse
température (Einstein, 1907 --- la première théorie quantique de la
matière), pourquoi la capacité thermique de l'hydrogène gravit un
escalier, pourquoi le ciel s'amincit exponentiellement --- et comment
Perrin, en regardant des grains microscopiques sédimenter dans une
goutte d'eau, a compté le nombre d'Avogadro et clos le débat sur la
réalité des atomes.

\section{La distribution de Boltzmann, obtenue}

\begin{theorem}[Distribution canonique]\label{thm:b3:canonical-ensemble:boltzmann}
\index{distribution de Boltzmann}\index{ensemble canonique}\index{fonction de partition}
Un système qui échange de l'énergie avec un grand réservoir à la
température $T$ occupe son \omterm{def:b3:microcanonical-ensemble:states}{micro-état} $s$ (d'énergie $E_s$) avec la
probabilité
\[
\mathcal P_s = \frac{\eu^{-E_s/k_{\text{B}}T}}{Z} , \qquad
Z = \sum_s \eu^{-E_s/k_{\text{B}}T}
\]
($\beta = 1/k_{\text{B}}T$ dans toute la suite). $Z$, la
\emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{fonction de partition}}, normalise les probabilités --- et se
trouve contenir toute la thermodynamique du système.
\end{theorem}

\begin{proof}
Le système dans l'état $s$ laisse au réservoir l'énergie $E - E_s$~:
par le \omterm{thm:b3:microcanonical-ensemble:postulate}{postulat fondamental} appliqué à l'ensemble, $\mathcal P_s
\propto \Omega_{\text{res}}(E - E_s)$. Développons l'entropie du
réservoir, énorme et régulière, au premier ordre~: $S_{\text{res}}(E - E_s)
= S_{\text{res}}(E) - E_s\,\partial S_{\text{res}}/\partial E =
S_{\text{res}}(E) - E_s/T$ par la définition de la température. Donc
$\Omega_{\text{res}} \propto \eu^{-E_s/k_{\text{B}}T}$. (Les ordres
supérieurs meurent avec la taille du réservoir.)
\end{proof}

\begin{proposition}[La machine]\label{prop:b3:canonical-ensemble:machine}
\index{énergie libre}\index{capacité thermique!et fluctuations}
À partir de $Z(T, V, N)$~:
\[
\langle E\rangle = -\frac{\partial\ln Z}{\partial\beta} , \qquad
F = -k_{\text{B}}T\ln Z , \qquad
S = -\frac{\partial F}{\partial T} , \qquad
P = -\frac{\partial F}{\partial V} ,
\]
où $F = \langle E\rangle - TS$ est l'\emph{\omterm{prop:b3:canonical-ensemble:machine}{énergie libre}}~; et la
capacité thermique est une \emph{fluctuation}~:
\[
C = \frac{\partial\langle E\rangle}{\partial T}
 = \frac{\langle E^2\rangle - \langle E\rangle^2}
{k_{\text{B}}T^2} .
\]
Pour des sous-systèmes indépendants et discernables $Z = z^N$~; pour
$N$ \omterm{def:b3:identical-particles:exchange}{particules identiques} dans une même boîte, $Z = z^N/N!$. À $T$ et
$V$ fixés, l'équilibre \emph{minimise} $F$~: la nature troque énergie
contre entropie au taux de change $T$ --- le principe le plus utile de
toute la physique de la matière.
\end{proposition}

\begin{proof}[Démonstration partielle]
$-\partial_\beta\ln Z = \sum E_s\eu^{-\beta E_s}/Z = \langle
E\rangle$~; une dérivée de plus donne $\langle E^2\rangle - \langle
E\rangle^2$, et la règle de dérivation composée convertit
$\partial_\beta$ en $\partial_T$. Les identifications de $F$, $S$, $P$
suivent de la comparaison de $\dd(-k_{\text{B}}T\ln Z)$ avec le
$\dd F = -S\dd T - P\dd V$ thermodynamique~; la factorisation est
l'exponentielle d'une somme. Le principe de minimum~: que
$F_{\text{syst}}$ décroisse, c'est que $S_{\text{total}}$ croisse,
puisque $\Delta S_{\text{res}} =
-\Delta E_{\text{sys}}/T$.
\end{proof}

\section{Premières victoires}

\begin{example}[Deux niveaux, en deux lignes]\label{ex:b3:canonical-ensemble:twolevel}
Pour les niveaux $0, \epsilon$~: $z = 1 + \eu^{-\beta\epsilon}$, d'où
$\langle E\rangle = N\epsilon/(\eu^{\beta\epsilon} + 1)$ --- le
résultat qui coûtait une page de Stirling par la voie microcanonique
(\cref{exo:b3:microcanonical-ensemble:8}). Le formalisme canonique est
le microcanonique avec la combinatoire prédigérée~; la bosse de
Schottky dans $C(T)$ suit d'une seule dérivation.
\end{example}

\begin{example}[Le solide d'Einstein, et la mort de Dulong--Petit]\label{ex:b3:canonical-ensemble:einstein}
\index{solide d'Einstein}\index{loi de Dulong--Petit}
Modélisons un cristal par $3N$ oscillateurs quantiques de pulsation
$\omega$ (\cref{ch:b3:harmonic-oscillator}). Par oscillateur,
\[
z = \sum_n\eu^{-\beta\hbar\omega(n + 1/2)}
 = \frac{1}{2\sinh(\beta\hbar\omega/2)} , \qquad
\langle E\rangle = \frac{\hbar\omega}{2} +
\frac{\hbar\omega}{\eu^{\beta\hbar\omega} - 1} .
\]
À haute $T$~: $\langle E\rangle \to k_{\text{B}}T$ par oscillateur,
$C \to 3Nk_{\text{B}}$ --- la \omterm{ex:b3:canonical-ensemble:einstein}{loi de Dulong--Petit} du volume de
première année, expliquée. À basse $T$ le quantum $\hbar\omega$
devient inabordable et $C$ s'effondre exponentiellement --- comme
mesuré, et inexplicable classiquement. La courbe d'Einstein de 1907,
un paramètre par élément, fut la première application des quanta à la
matière ordinaire~; le diamant, aux liaisons raides et aux atomes
légers ($\hbar\omega/k_{\text{B}}
\approx \qty{1300}{K}$), est encore «~gelé~» à température ambiante
--- l'anomalie qui intriguait les chimistes depuis quatre-vingts ans
(\cref{exo:b3:canonical-ensemble:6}).
\end{example}

\begin{theorem}[L'équipartition, et sa licence]\label{thm:b3:canonical-ensemble:equipartition}
\index{théorème d'équipartition}
Toute coordonnée ou quantité de mouvement qui entre
\emph{quadratiquement} dans l'énergie contribue, dans le régime
classique (haute température), pour
\[
\langle\epsilon\rangle = \tfrac12 k_{\text{B}}T
\]
à l'énergie moyenne~: $\tfrac32 k_{\text{B}}T$ pour un atome de gaz
monoatomique, $\tfrac52$ pour une diatomique qui tourne,
$3k_{\text{B}}T$ pour un oscillateur. La licence expire lorsque
$k_{\text{B}}T$ tombe sous l'écart de niveaux du mode~: le mode gèle
et sa contribution disparaît --- la résolution des scandales du
dix-neuvième siècle sur les capacités thermiques, dessinée en escalier
ci-dessus.
\end{theorem}

\begin{proof}[Démonstration partielle]
Pour $\epsilon = ax^2$,
\[
\langle\epsilon\rangle
 = \frac{\int ax^2\,\eu^{-\beta ax^2}\dd x}
        {\int\eu^{-\beta ax^2}\dd x}
 = -\partial_\beta\ln\!\int\eu^{-\beta ax^2}\dd x
 = -\partial_\beta\ln\beta^{-1/2} = \frac{1}{2\beta} .
\] Le gel est le calcul de
\cref{ex:b3:canonical-ensemble:einstein}, mode par mode.
\end{proof}

\begin{example}[Le gaz, canoniquement]\label{ex:b3:canonical-ensemble:gas}
Un atome dans une boîte~: $z = V/\lambda_T^3$ (le comptage d'états de
\cref{prop:b3:schrodinger-three-dimensions:counting} pondéré par
Boltzmann)~; $N$ atomes identiques~: $Z = z^N/N!$. Alors $F =
-Nk_{\text{B}}T[\ln(V/N\lambda_T^3) + 1]$, et la dérivation livre
$PV = Nk_{\text{B}}T$, l'entropie de Sackur--Tetrode et
$\langle E\rangle = \tfrac32 Nk_{\text{B}}T$ --- tout le gaz parfait
à partir d'une intégrale gaussienne. Le facteur de Boltzmann appliqué
à l'énergie cinétique d'une molécule donne la distribution des
vitesses de Maxwell du volume de première année, désormais obtenue~;
appliqué à son énergie potentielle $mgh$, il donne l'atmosphère
exponentielle --- et, dans une goutte d'eau, l'échelle de grains de
Perrin (\cref{pb:b3:canonical-ensemble:1}).
\end{example}

\begin{method}[L'art du canonique]\label{met:b3:canonical-ensemble:method}
(1) Dresser la liste des états et des énergies d'\emph{une} unité~;
calculer $z$. (2) Factoriser~: $Z = z^N$ (ou $z^N/N!$ pour des
\omterm{def:b3:identical-particles:exchange}{particules identiques} qui partagent l'espace)~; prendre le $\ln$ tôt.
(3) Dériver~: en $\beta$ pour l'énergie, en $T$ pour l'entropie via
$F$, en $V$ pour la pression~; une dérivée seconde pour $C$ et les
fluctuations. (4) Vérifier les deux bouts~: la haute $T$ doit
reproduire l'équipartition, la basse $T$ doit geler avec une queue en
$\eu^{-\Delta/k_{\text{B}}T}$. (5) Compétitions (repliement, fixation,
alignement)~: écrire $F = E - TS$ pour chaque possibilité et laisser
gagner la plus petite --- la transition se situe vers $T \approx \Delta
E/\Delta S$.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:1}
(a) Écrire le rapport de populations de deux niveaux séparés de
$\epsilon$ à la température $T$. (b) Dans une flamme à \qty{2500}{K},
quelle fraction des atomes de sodium siège dans l'état excité à
\qty{2.1}{eV} de la raie D (dégénérescence 2 pour le fondamental, 6
pour l'excité~: rapport de populations $3\eu^{-\beta\epsilon}$)~?
(c) Pourquoi la flamme flamboie-t-elle malgré tout en jaune (combien
d'atomes par cm$^3$ suffisent)~? (d) À quelle température la fraction
excitée atteindrait-elle $10\%$~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:2}
Un système à trois niveaux~: $0, \epsilon, 2\epsilon$. (a) Écrire
$z$. (b) Calculer $\langle E\rangle$ et vérifier les deux limites de
température. (c) À quelle $T$ le niveau intermédiaire est-il le plus
peuplé \emph{en valeur absolue}~? (d) Montrer qu'aucune température,
si haute soit-elle, ne rend un niveau supérieur plus peuplé qu'un
niveau inférieur --- quelle notion du chapitre 16 faudrait-il pour
cela~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:3}
L'atmosphère isotherme. (a) Appliquer le facteur de Boltzmann à
l'énergie potentielle $mgh$ et obtenir $n(h) = n_0\eu^{-mgh/k_{\text{B}}
T}$. (b) Calculer la hauteur caractéristique pour l'air
($m = \qty{4.8e-26}{kg}$) à \qty{288}{K}. (c) La pression au sommet de
l'Everest en fraction de celle du niveau de la mer. (d) Pourquoi la
décroissance de l'atmosphère réelle est-elle proche d'une
exponentielle sans l'être exactement (qu'avons-nous tenu pour constant
qui ne l'est pas)~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:4}
Comptabilité d'équipartition. Compter les termes quadratiques et
prédiser le $C_V$ molaire (a) de l'argon~; (b) de N$_2$ à température
ambiante (rotation active, vibration gelée)~; (c) de N$_2$ à
\qty{3000}{K}~; (d) d'un solide cristallin (Dulong--Petit). Où les
valeurs mesurées $12.5$, $20.8$, $\approx26$,
$\approx\qty{25}{J/(mol.K)}$ s'accordent-elles, et qu'enseigne chaque
écart~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:5}
L'oscillateur quantique, canoniquement. (a) Sommer la série
géométrique pour $z$. (b) Obtenir $\langle E\rangle$ et identifier $\langle
n\rangle = 1/(\eu^{\beta\hbar\omega} - 1)$ --- le résultat de
\cref{exo:b3:harmonic-oscillator:6}, désormais sans effort. (c)
Dériver pour $C(T)$ et vérifier les deux limites. (d) Pourquoi la
forme de $\langle n\rangle$ est-elle sur le point de devenir célèbre
(\cref{ch:b3:photons-phonons})~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:6}
Einstein face aux données. (a) D'après la formule de la figure, à quel
$T/\theta_{\text{E}}$ $C$ est-elle tombée à la moitié de
Dulong--Petit~? (b) Le diamant~: $\theta_{\text{E}} \approx \qty{1300}{K}$
--- calculer $C/3Nk_{\text{B}}$ à \qty{300}{K} et expliquer
l'«~anomalie du diamant~» du dix-neuvième siècle. (c) Le plomb~: $\theta_{\text{E}}
\approx \qty{90}{K}$ --- pourquoi le plomb a-t-il toujours été
«~bien élevé~»~? (d) Les mesures à très basse $T$ montrent
$C \propto T^3$, non une exponentielle~: laquelle des hypothèses
d'Einstein tombe (tous les oscillateurs à \emph{une} seule
fréquence), et qui l'a réparée (\cref{ch:b3:photons-phonons})~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:7}
La queue de Maxwell et l'hydrogène manquant. (a) À partir du facteur
de Boltzmann sur l'énergie cinétique, écrire la distribution des
vitesses et situer la vitesse la plus probable pour N$_2$ et H$_2$ à
\qty{288}{K}. (b) La vitesse de libération terrestre est
\qty{11.2}{km/s}~: calculer $mv_{\text{esc}}^2/2k_{\text{B}}T$ pour
les deux gaz. (c) La fraction qui s'échappe varie comme
$\eu^{-mv_{\text{esc}}^2/2k_{\text{B}}T}$~: comparer les deux
exposants et conclure quel gaz fuit à l'échelle géologique. (d)
Relier à la composition observée de l'atmosphère terrestre (sans
H$_2$ libre) et de celle de Jupiter (surtout H$_2$) --- quels deux
paramètres renversent le verdict~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:8}
Les fluctuations rencontrent la réponse. (a) Démontrer que $C = (\langle E^2\rangle -
\langle E\rangle^2)/k_{\text{B}}T^2$ à partir de deux dérivées de $\ln
Z$. (b) Le vérifier explicitement sur le système à deux niveaux. (c)
Pour $N$ unités indépendantes, montrer que la fluctuation
\emph{relative} d'énergie décroît comme $1/\sqrt N$. (d) Énoncer la
morale~: ce que la disposition d'un système à absorber de la chaleur
(une réponse) a à voir avec le tremblement de son énergie (une
fluctuation) --- le marché récurrent de la physique statistique.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:9}
Boltzmann au laboratoire de chimie. Les vitesses de réaction portent
le facteur $\eu^{-E_a/k_{\text{B}}T}$ (franchir une barrière
d'activation $E_a$ --- Arrhenius). (a) Montrer que la règle empirique
«~la vitesse double tous les \qty{10}{K} près de la température
ambiante~» correspond à $E_a \approx
\qty{0.55}{eV}$. (b) De quel facteur cette réaction ralentit-elle dans
un réfrigérateur (\qty{5}{\celsius})~? (c) Cuire un œuf en altitude~:
l'eau bout à \qty{93}{\celsius} sur un col à \qty{2000}{m} --- estimer
le temps de cuisson supplémentaire. (d) Pourquoi la queue au-delà
d'une barrière, et non l'énergie moyenne, gouverne-t-elle la chimie
(quelles molécules réagissent)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:10}
Repliement à deux états. Une biomolécule est repliée (énergie $0$, une
seule configuration) ou dépliée (énergie $\Delta E > 0$, $\Omega_u =
\eu^{\Delta S/k_{\text{B}}}$ configurations). (a) Écrire la fraction
repliée en fonction de $T$. (b) Montrer que le milieu de la «~fusion~»
est $T_{\text{m}} = \Delta E/\Delta S$ et l'interpréter comme un
partage $F = E -
TS$. (c) Avec $\Delta E = \qty{3.0}{eV}$ et $\Delta S =
100\,k_{\text{B}}$ (l'unité coopérative d'une petite protéine)~:
calculer $T_{\text{m}}$ et la largeur de la transition. (d) Pourquoi
la \emph{coopérativité} (de nombreux contacts qui lâchent ensemble,
$\Delta E$ \emph{et} $\Delta S$ grands) affine-t-elle la fusion ---
et comment les machines de cyclage thermique de l'ADN (PCR) en
tirent-elles exactement parti~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:11}
Paramagnétisme et refroidissement magnétique. $N$ spins $\tfrac12$ de
moment $\mu$ dans un champ $B$. (a) Montrer que
$M = N\mu\tanh(\mu B/k_{\text{B}}T)$ et développer en la loi de Curie
$M \approx N\mu^2B/k_{\text{B}}T$. (b) Évaluer l'alignement
$\mu B/k_{\text{B}}T$ pour des moments électroniques à
$B = \qty{1}{T}$, $T = \qty{300}{K}$, puis à \qty{1}{K}. (c)
Désaimantation adiabatique~: aimanter à \qty{1}{K}, isoler, réduire
$B$ d'un facteur dix --- montrer qu'une entropie constante signifie un
$\mu B/k_{\text{B}}T$ constant, donc que $T$ chute d'un facteur dix.
(d) Qu'est-ce qui fixe le plancher de ce réfrigérateur (les
interactions entre spins --- estimer l'échelle dipolaire
$\mu_0\mu^2/4\pi a^3$ pour $a =
\qty{0.5}{nm}$, en unités de température)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:12}
L'escalier de rotation, quantitativement. Les niveaux de rotation
d'une diatomique sont $E_J = BJ(J+1)$, de dégénérescence $2J + 1$
(\cref{ch:b3:quantum-angular-momentum}). (a) Écrire $z_{\text{rot}}$
et montrer que pour $k_{\text{B}}T \gg B$ la somme devient
l'intégrale $k_{\text{B}}T/B$~: le $k_{\text{B}}$ d'équipartition
retrouvé dans $C_V$. (b) Définir $\theta_{\text{rot}} =
B/k_{\text{B}}$ et l'évaluer pour H$_2$ ($B = \qty{7.5}{meV}$) et
N$_2$ ($B = \qty{0.25}{meV}$)~: quel gaz montre la marche de rotation
à des températures accessibles~? (c) Représenter tout l'escalier
$C_V(T)$ de l'hydrogène avec ses trois paliers et ses deux
contremarches, en chiffrant les deux. (d) Le comportement mesuré à
basse $T$ de H$_2$ est encore compliqué par le mélange ortho--para 3:1
de \cref{exo:b3:identical-particles:8}~: dites en une phrase comment
la statistique des spins nucléaires atteint la capacité thermique d'un
gaz.
\end{exercise}

\section{Problème~: compter Avogadro dans une goutte d'eau}

\begin{problem}[{Problème du week-end --- les grains de Perrin et la
réalité des atomes}]\label{pb:b3:canonical-ensemble:1}
En 1908, Jean Perrin mit en suspension dans l'eau des grains de résine
microscopiques, les laissa sédimenter et les compta couche par couche
au microscope. L'échelle exponentielle qu'il trouva était le facteur
de Boltzmann rendu visible --- et de sa hauteur caractéristique il
tira le nombre d'Avogadro, convainquant les derniers sceptiques que
les atomes existent. Prix Nobel, 1926. Données~: grains de gomme-gutte
de rayon $r =
\qty{0.212}{\micro m}$, masse volumique $\rho_{\text{g}} =
\qty{1207}{kg/m^3}$~; eau $\rho_{\text{w}} = \qty{999}{kg/m^3}$~;
$T = \qty{293}{K}$~; $g = \qty{9.81}{m/s^2}$.

\medskip\noindent\textbf{Partie I --- Le facteur de Boltzmann visible.}
\begin{enumerate}
  \item Un grain dans l'eau subit la pesanteur diminuée de la poussée
        d'Archimède~: calculer sa masse effective $m' = \tfrac43\pi r^3(\rho_{\text{g}}
        - \rho_{\text{w}})$ et son \emph{poids}.
  \item Écrire le profil de concentration d'équilibre $n(h)$ à
        partir du facteur de Boltzmann.
  \item Calculer la hauteur caractéristique $h_0 = k_{\text{B}}T/m'g$
        avec le $k_{\text{B}}$ moderne.
  \item Pourquoi les grains doivent-ils être si précisément
        calibrés (comment $h_0$ dépend-elle de $r$)~?
  \item Comparer $h_0$ à la même formule pour les molécules d'air~:
        pourquoi l'atmosphère des grains fait-elle quelques
        micromètres quand celle de l'air fait des kilomètres~?
  \item Perrin compta (dans une série) les concentrations relatives
        $100 : 55 : 30 : 17$ à quatre profondeurs équidistantes de
        \qty{30}{\micro m}~: vérifier que c'est bien une échelle
        exponentielle et en extraire $h_0$.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Peser l'invisible.}
\begin{enumerate}[resume]
  \item Inverser~: à partir du $h_0$ mesuré à la question 6 et du
        $m'g$ connu, extraire $k_{\text{B}}$.
  \item La constante des gaz $R = \qty{8.314}{J/(mol.K)}$ était
        connue de la chimie macroscopique~: la combiner à votre
        $k_{\text{B}}$ pour obtenir le nombre d'Avogadro $N_{\text{A}} =
        R/k_{\text{B}}$.
  \item Les séries de Perrin donnèrent $N_{\text{A}}$ entre $5.5$ et
        $\num{7.2e23}$~: comparer à la valeur moderne
        \num{6.022e23} et commenter l'exploit, vu son microscope et
        son chronomètre.
  \item À partir de $N_{\text{A}}$, calculer la masse d'un unique
        \omterm{thm:b3:hydrogen-atom:levels}{atome d'hydrogène} --- le nombre que les atomistes voulaient
        depuis un siècle.
  \item Expliquer la structure logique~: quelles mesures
        \emph{macroscopiques} ($R$~; taille et masse volumique des
        grains~; l'échelle) se combinent pour peser un atome, sans
        qu'aucun atome n'ait jamais été vu~?
  \item Pourquoi les grains --- $10^{10}$ masses atomiques chacun ---
        obéissent-ils à la même statistique de Boltzmann que les
        molécules (qu'est-ce qui, dans la démonstration de
        \cref{thm:b3:canonical-ensemble:boltzmann}, se soucie de la
        taille)~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le tremblement qui scelle l'affaire.}
\begin{enumerate}[resume]
  \item Ces mêmes grains tremblent~: la formule d'Einstein de 1905
        pour le mouvement brownien donne $\langle x^2\rangle = 2Dt$ avec $D =
        k_{\text{B}}T/6\pi\eta r$ (eau~: $\eta =
        \qty{e-3}{Pa.s}$). Calculer $D$ pour les grains de Perrin.
  \item Jusqu'où un grain vagabonde-t-il en une minute~? Perrin
        pouvait-il le mesurer avec un oculaire micrométrique et un
        chronomètre~?
  \item Perrin vérifia $\langle x^2\rangle \propto t$ et en extrait
        $k_{\text{B}}$ \emph{de nouveau}, indépendamment~: pourquoi
        deux voies sans rapport (une échelle statique~; un
        tremblement dynamique) vers un même nombre ont-elles porté un
        tel poids de preuve~?
  \item Le tremblement est l'équipartition appliquée au grain~:
        chaque composante de vitesse porte $\tfrac12 k_{\text{B}}T$.
        Estimer la vitesse thermique quadratique moyenne du grain ($m \approx
        \qty{4.8e-17}{kg}$).
  \item Pourquoi cette vitesse n'est-elle jamais vue directement
        (qu'est-ce qui interrompt le vol libre après quelques
        nanomètres), et que voit-on \emph{à la place}~?
  \item Dites quels deux chapitres de ce livre se rencontrent dans
        cette observation~: la mécanique de la traînée (volume de
        deuxième année, viscosité) et la statistique de ce chapitre.
\end{enumerate}

\medskip\noindent\textbf{Partie IV --- Ce qui fut réglé.}
\begin{enumerate}[resume]
  \item Ostwald et Mach tenaient les atomes pour des fictions
        comptables~: dites en une phrase pourquoi un $N_{\text{A}}$
        \emph{compté} à partir d'échelles de grains a mis fin à cette
        position.
  \item Citer trois autres voies des années 1900 qui ont convergé
        vers le même $N_{\text{A}}$ (le bleu du ciel,
        \cref{exo:b3:microcanonical-ensemble:12}~; l'électrolyse
        jointe à la charge de l'électron~; la production d'hélium par
        radioactivité) --- pourquoi la \emph{convergence} a-t-elle
        compté plus qu'aucune valeur isolée~?
  \item L'échelle de Perrin est un équilibre entre lesquelles deux
        monnaies de ce chapitre (l'énergie qui tire vers le bas,
        l'entropie qui étale vers le haut), tarifé à quel taux~?
  \item Usages modernes de la même physique~: les
        ultracentrifugeuses analytiques font tourner des protéines à
        $10^5g$ pour comprimer leurs «~atmosphères~» en échelles
        mesurables --- montrer que multiplier $g$ par $10^5$ divise
        $h_0$ d'autant, et estimer $h_0$ pour une protéine de masse
        effective \qty{e-22}{kg} à $10^5g$, \qty{293}{K}.
  \item Doubler le rayon du grain divise $h_0$ par huit~: encadrer la
        fenêtre pratique de tailles de grains entre «~échelle trop
        haute pour voir un gradient~» et «~échelle plus mince qu'un
        grain~» pour un champ de microscope profond de
        \qty{100}{\micro m}.
  \item Depuis 2019 le SI \emph{définit} exactement $k_{\text{B}}$ et
        $N_{\text{A}}$~: dites ce que mesure \emph{aujourd'hui} une
        expérience de Perrin exacte (un contrôle de cohérence, ou un
        étalonnage de l'appareil et des grains) --- et pourquoi la
        physique est inchangée.
  \item Résumer le résultat nommé~: la concentration d'un grain de
        \qty{0.2}{\micro m} est divisée par deux tous les
        $\sim\qty{35}{\micro m}$ de hauteur~; lue avec
        $n(h) = n_0\eu^{-m'gh/k_{\text{B}}T}$, cette échelle a livré
        $N_{\text{A}} \approx \num{6e23}$ --- des atomes comptés, non
        conjecturés, dans une goutte d'eau sur une platine de
        microscope.
\end{enumerate}
\end{problem}
