\chapter{O oscilador harmônico quântico}\label{ch:b3:harmonic-oscillator}

Quase nada na natureza é exatamente um oscilador harmônico, e quase
tudo é aproximadamente um: qualquer sistema afastado de um equilíbrio
estável --- a ligação de uma molécula, um átomo num cristal, uma ponte,
um modo do campo eletromagnético --- sente uma força restauradora
proporcional ao deslocamento, porque todo potencial suave é uma
parábola no fundo do seu poço. Quem resolve o oscilador quântico uma
vez resolve, portanto, as pequenas vibrações do mundo inteiro. Este
capítulo o resolve no estilo algébrico que se tornou a assinatura da
mecânica quântica: dois \omterm{def:b3:harmonic-oscillator:ladder}{operadores de escada} sobem e descem uma escada
perfeitamente regular de níveis $\big(n +
\tfrac12\big)\hbar\omega$, sendo o meio degrau de baixo --- a \omterm{thm:b3:harmonic-oscillator:spectrum}{energia
de ponto zero} --- um teorema, e não uma opção. As consequências vão de
por que o hélio nunca congela até como uma molécula de dióxido de
carbono, ressoando no seu próprio $\hbar\omega$, intercepta o calor que
a Terra emite.

\section{A escada}

\begin{definition}[Operadores de escada]\label{def:b3:harmonic-oscillator:ladder}
\index{operadores de escada}\index{operador de aniquilação}\index{operador de criação}
Para $\hat H = \hat p^2/2m + \tfrac12 m\omega^2\hat x^2$, introduza o
par adimensional e não hermitiano
\[
\hat a = \sqrt{\frac{m\omega}{2\hbar}}\Big(\hat x +
\frac{\iu\hat p}{m\omega}\Big) , \qquad
\hat a^\dagger = \sqrt{\frac{m\omega}{2\hbar}}\Big(\hat x -
\frac{\iu\hat p}{m\omega}\Big) .
\]
A partir de $[\hat x, \hat p] = \iu\hbar$:
\[
[\hat a, \hat a^\dagger] = 1 , \qquad
\hat H = \hbar\omega\Big(\hat N + \tfrac12\Big) , \quad
\hat N = \hat a^\dagger\hat a .
\]
$\hat N$ é hermitiano; seus \omterm{def:b3:quantum-formalism:observable}{autovalores} contarão \emph{quanta}, e
$\hat a$, $\hat a^\dagger$ --- os operadores de \emph{aniquilação} e de
\emph{criação} --- retirarão e acrescentarão um.
\end{definition}

\begin{theorem}[O espectro, só por álgebra]\label{thm:b3:harmonic-oscillator:spectrum}
\index{energia de ponto zero}
Os \omterm{def:b3:quantum-formalism:observable}{autovalores} de $\hat H$ são exatamente
\[
E_n = \Big(n + \tfrac12\Big)\hbar\omega , \qquad n = 0, 1, 2, \dots
\]
--- uma escada infinita de degraus iguais $\hbar\omega$ acima de um
nível fundamental que \emph{não} é nulo: a \emph{\omterm{thm:b3:harmonic-oscillator:spectrum}{energia de ponto
zero}} $\tfrac12\hbar\omega$. Os autoestados normalizados $\ket n$ são
ligados por
\[
\hat a\ket n = \sqrt n\,\ket{n - 1} , \qquad
\hat a^\dagger\ket n = \sqrt{n + 1}\,\ket{n + 1} ,
\qquad
\ket n = \frac{(\hat a^\dagger)^n}{\sqrt{n!}}\,\ket0 .
\]
\end{theorem}

\begin{proof}
Do \omterm{def:b3:quantum-formalism:commutator}{comutador}, $\hat N\hat a = \hat a(\hat N - 1)$: se
$\ket\nu$ é autovetor de $\hat N$ com \omterm{def:b3:quantum-formalism:observable}{autovalor} $\nu$, então $\hat a\ket\nu$ é um
autovetor de \omterm{def:b3:quantum-formalism:observable}{autovalor} $\nu - 1$ (ou o vetor nulo). Cada descida só é
permitida enquanto $\nu \ge 0$, pois $\nu = \braket{\nu}{\hat
N\nu} = \|\hat a\ket\nu\|^2 \ge 0$; a descida tem, portanto, de
terminar, e só termina num estado com $\hat a\ket{\nu_0}
= 0$, donde $\nu_0 = 0$. Assim, $\nu$ percorre os inteiros não
negativos. As normalizações seguem de $\|\hat a\ket n\|^2 = n$
e $\|\hat a^\dagger\ket n\|^2 = n + 1$.
\end{proof}

\begin{proposition}[Os estados no espaço]\label{prop:b3:harmonic-oscillator:wavefunctions}
O estado fundamental é a gaussiana
\[
\varphi_0(x) = \Big(\frac{m\omega}{\pi\hbar}\Big)^{1/4}
\eu^{-m\omega x^2/2\hbar} ,
\]
obtida resolvendo a equação de \emph{primeira ordem} $\hat a\varphi_0
= 0$; ela satura a desigualdade de Heisenberg, $\Delta x\,\Delta p =
\hbar/2$, com $\Delta x = \sqrt{\hbar/2m\omega}$. Aplicar
$\hat a^\dagger$ repetidamente gera $\varphi_n$: um polinômio de grau
$n$ (com $n$ nós) vezes a mesma gaussiana. Para $n$ grande,
$|\varphi_n|^2$ oscila rapidamente em torno da distribuição clássica do
tempo de permanência, máxima perto dos pontos de retorno --- o
princípio da correspondência em imagem.
\end{proposition}

\begin{proof}[Demonstração parcial]
$\hat a\varphi_0 = 0$ se lê $\varphi_0' = -(m\omega/\hbar)x
\varphi_0$: a gaussiana, normalizada. Saturação: a gaussiana é o caso
de igualdade do \cref{thm:b3:quantum-formalism:uncertainty}. A
estrutura polinomial decorre de $\hat a^\dagger$ ser de primeira ordem
em $x$ e $\dd/\dd x$; o enunciado para $n$ grande é verificado no
\cref{exo:b3:harmonic-oscillator:12}.
\end{proof}

\section{A energia de ponto zero é real}

\begin{remark}[Por que o piso não pode ser mais baixo]\label{rem:b3:harmonic-oscillator:zeropoint}
Um estado de energia nula exigiria $\langle\hat p^2\rangle =
\langle\hat x^2\rangle = 0$: perfeitamente parado \emph{e}
perfeitamente centrado, o que $\Delta x\,\Delta p \ge \hbar/2$ proíbe.
O estado fundamental é o melhor compromisso que a desigualdade
permite, e $\tfrac12\hbar\omega$ é o aluguel. Não se trata de uma
constante de contabilidade: o movimento de ponto zero borra os padrões
de difração de raios X no zero absoluto; ele dá aos isótopos mais leves
ligações efetivas mais fracas (H$_2$ e D$_2$ se dissociam a energias
mensuravelmente diferentes a partir do mesmo poço eletrônico); e no
hélio ele é tão violento --- átomos leves, atração débil --- que o
líquido \emph{nunca} congela sob sua própria pressão de vapor: é o
único elemento ainda líquido no zero absoluto, solidificando-se apenas
com a ajuda de 25 atmosferas.
\end{remark}

\begin{example}[Escalas de $\hbar\omega$]\label{ex:b3:harmonic-oscillator:scales}
Ligação do monóxido de carbono ($\omega = \qty{4.1e14}{rad/s}$): $\hbar\omega =
\qty{0.27}{eV}$, dez vezes o $k_{\text{B}}T$ da
temperatura ambiente --- as vibrações moleculares estão paradas no
ar do dia a dia, e é por isso que as capacidades térmicas dos
diatômicos intrigaram o século XIX. Um pêndulo
($\omega = \qty{5}{rad/s}$): $\hbar\omega = \qty{3e-15}{eV}$,
irremediavelmente além da resolução --- o limite da correspondência. Um
espelho do LIGO de \qty{40}{kg}, suspenso de modo a oscilar a
\qty{100}{Hz}, tem como modo de oscilador a amplitude de ponto zero $\Delta x =
\sqrt{\hbar/2m\omega} \approx \qty{5e-20}{m}$ --- e o observatório
resolve rotineiramente deslocamentos \emph{nesse} piso quântico: o
movimento de ponto zero de um objeto de quarenta quilogramas é hoje um
\omterm{def:b3:lagrangian-mechanics:coordinates}{vínculo} de engenharia (\cref{exo:b3:harmonic-oscillator:3}).
\end{example}

\section{Estados coerentes: a face clássica do oscilador quântico}

\begin{definition}[Estados coerentes]\label{def:b3:harmonic-oscillator:coherent}
\index{estado coerente}
Um \emph{estado coerente} $\ket\alpha$ é um autovetor do \omterm{def:b3:harmonic-oscillator:ladder}{operador de
aniquilação}, $\hat a\ket\alpha = \alpha\ket\alpha$, com $\alpha$ um
número complexo qualquer. Expandido na escada,
\[
\ket\alpha = \eu^{-|\alpha|^2/2}\sum_{n=0}^\infty
\frac{\alpha^n}{\sqrt{n!}}\,\ket n :
\]
a contagem quântica segue uma distribuição de Poisson, de média $\langle\hat
N\rangle = |\alpha|^2$ e dispersão $\Delta N = |\alpha|$.
\end{definition}

\begin{proposition}[Por que os lasers são clássicos]\label{prop:b3:harmonic-oscillator:coherent-motion}
Um \omterm{def:b3:harmonic-oscillator:coherent}{estado coerente} é a gaussiana do estado fundamental deslocada no
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}; sob a evolução do oscilador ele \emph{permanece}
coerente, com $\alpha(t) = \alpha\,\eu^{-\iu\omega t}$: seu centro
executa exatamente o movimento clássico, $\langle\hat x\rangle(t) =
x_0\cos\omega t + \cdots$, ao passo que suas larguras permanecem para
sempre no mínimo $\Delta x\,\Delta p = \hbar/2$ --- um pacote de onda
que nunca se espalha. Os \omterm{def:b3:harmonic-oscillator:coherent}{estados coerentes} são o modo como um oscilador
quântico imita um clássico; a luz de um laser é um \omterm{def:b3:harmonic-oscillator:coherent}{estado coerente} de
um modo do campo, com número de fótons poissoniano (o \emph{ruído de
disparo} de todo fotodetector) e campo oscilando como Maxwell dizia.
\end{proposition}

\begin{proof}[Demonstração parcial]
A expansão decorre de escrever $\ket\alpha = \sum c_n\ket n$ em
$\hat a\ket\alpha = \alpha\ket\alpha$: $c_n = \alpha c_{n-1}/\sqrt
n$. Evolução: cada $\ket n$ ganha $\eu^{-\iu(n + 1/2)\omega t}$, o que
se ressoma num \omterm{def:b3:harmonic-oscillator:coherent}{estado coerente} de $\alpha\eu^{-\iu\omega t}$ (a menos
de fase global). $\langle\hat x\rangle \propto
\operatorname{Re}\alpha(t)$, a partir de $\hat x \propto \hat a +
\hat a^\dagger$. Que o estado seja a gaussiana deslocada é aqui
admitido.
\end{proof}

\begin{method}[Álgebra do oscilador]\label{met:b3:harmonic-oscillator:method}
(1) Exprima o que se pede em $\hat a$ e $\hat a^\dagger$: $\hat x
= \sqrt{\hbar/2m\omega}\,(\hat a + \hat a^\dagger)$, $\hat p =
\iu\sqrt{m\hbar\omega/2}\,(\hat a^\dagger - \hat a)$. (2) Empurre os
$\hat a$ para a direita com $[\hat a, \hat a^\dagger] = 1$; use
$\hat a\ket0 = 0$. (3) Elementos de matriz: $\hat x$ liga apenas
degraus vizinhos --- a regra de seleção $\Delta n = \pm1$ da
espectroscopia vibracional. (4) Toda \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} quadrática
(osciladores acoplados, circuitos, modos de campo) se diagonaliza em
escadas independentes: encontre primeiro os \omterm{prop:b3:lagrangian-mechanics:modes}{modos normais} e quantize
cada um. (5) Números antes de tudo: $\hbar\omega$ contra
$k_{\text{B}}T$ decide se o sistema é quântico ou clássico antes de
qualquer álgebra.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:1}
(a) Verifique $[\hat a, \hat a^\dagger] = 1$ a partir de $[\hat x, \hat p] =
\iu\hbar$. (b) Verifique $\hat H = \hbar\omega(\hat a^\dagger\hat a +
\tfrac12)$ por expansão direta. (c) Inverta para exprimir $\hat x$ e
$\hat p$. (d) Mostre que $\hat N$ é hermitiano e explique por que
$\hat a$ sozinho não poderia ser um observável.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:2}
No autoestado $\ket n$: (a) mostre que $\langle\hat x\rangle =
\langle\hat p\rangle = 0$; (b) calcule $\langle\hat x^2\rangle$ e
$\langle\hat p^2\rangle$; (c) deduza $\Delta x\,\Delta p = (n +
\tfrac12)\hbar$; (d) verifique a razão do virial $\langle E_k\rangle =
\langle E_p\rangle$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:3}
Amplitudes de ponto zero $\Delta x = \sqrt{\hbar/2m\omega}$: calcule-as
para (a) a molécula de CO ($\mu = \qty{1.14e-26}{kg}$, $\omega =
\qty{4.1e14}{rad/s}$), comparando com o comprimento de ligação de
\qty{0.11}{nm}; (b) um átomo de hidrogênio num sólido ($m =
\qty{1.7e-27}{kg}$, $\hbar\omega = \qty{0.1}{eV}$); (c) um espelho do
LIGO ($m = \qty{40}{kg}$, $f = \qty{100}{Hz}$); (d) ordene os três como
frações do tamanho de seus sistemas e comente quem é ``quântico''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:harmonic-oscillator:4}
(a) Usando as relações da escada, calcule $\bra m\hat x\ket n$ e
mostre que se anula a menos que $m = n \pm 1$. (b) Deduza a regra de
seleção vibracional $\Delta n = \pm1$ para a absorção de luz (o
acoplamento é $\propto\hat x$). (c) Por que uma molécula heteronuclear
(CO) absorve luz infravermelha e a N$_2$ não? (d) As moléculas reais
mostram linhas fracas de ``sobretom'' em $\approx 2\hbar\omega$: o que
isso revela sobre o potencial?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:5}
(a) Resolva $\hat a\varphi_0 = 0$ como equação diferencial e
normalize. (b) Gere $\varphi_1$ e $\varphi_2$ aplicando
$\hat a^\dagger$. (c) Verifique $\varphi_1 \perp \varphi_0$ só pela
paridade. (d) Esboce as três densidades e confira a contagem de nós.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:6}
Antecipando Boltzmann. Um conjunto de osciladores idênticos à
temperatura $T$ ocupa o nível $n$ com probabilidade $\propto
\eu^{-E_n/k_{\text{B}}T}$ (volume do segundo ano, fator de Boltzmann).
(a) Mostre que o número quântico médio é $\langle n\rangle =
1/(\eu^{\hbar\omega/k_{\text{B}}T} - 1)$. (b) Avalie-o para a vibração
do CO a \qty{300}{K} e a \qty{2000}{K}. (c) Mostre que a energia média
tende a $k_{\text{B}}T$ em alta temperatura (equipartição recuperada) e
a $\tfrac12\hbar\omega$ em baixa. (d) A que temperatura uma vibração de
\qty{15}{\micro m} (a flexão do dióxido de carbono) tem
$\langle n\rangle = 0.1$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:7}
Estatística dos \omterm{def:b3:harmonic-oscillator:coherent}{estados coerentes}. (a) A partir da expansão de
$\ket\alpha$, mostre que $\mathcal P(n)$ é uma poissoniana de média
$|\alpha|^2$. (b) Mostre que $\langle\hat N\rangle = |\alpha|^2$ e
$\Delta N = |\alpha|$. (c) Um laser de \qty{1}{mW} a \qty{633}{nm}:
fótons por segundo e a flutuação relativa $\Delta N/\langle N\rangle$
em um segundo. (d) Ruído de disparo: mostre que a razão ruído-sinal da
fotocorrente cai como $1/\sqrt{\langle N\rangle}$ --- por que os feixes
intensos parecem lisos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:8}
Evolução de um \omterm{def:b3:harmonic-oscillator:coherent}{estado coerente}. (a) Aplique as fases de evolução à
expansão e mostre que se obtém $\ket{\alpha(t)}$ com $\alpha(t) =
\alpha\eu^{-\iu\omega t}$ (a menos de uma fase global). (b) Deduza
$\langle\hat x\rangle(t)$ e $\langle\hat p\rangle(t)$ e compare com a
solução clássica. (c) Por que um autoestado de energia, apesar de
estacionário, \emph{não} descreve um pêndulo que oscila --- e que
característica do \omterm{def:b3:harmonic-oscillator:coherent}{estado coerente} descreve? (d) Estime $|\alpha|$ para
um pêndulo real (\qty{10}{g}, amplitude de \qty{10}{cm},
\qty{1}{Hz}) e comente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:harmonic-oscillator:9}
O circuito LC quântico. Uma malha supercondutora com $L =
\qty{10}{nH}$ e $C = \qty{0.4}{pF}$ faz a carga e o fluxo oscilarem
como $x$ e $p$. (a) Sua frequência de ressonância (volume do primeiro
ano) e o quantum $\hbar\omega$ em $\unit{\micro eV}$. (b) Abaixo de que
temperatura vale $k_{\text{B}}T \ll \hbar\omega$, de modo que o
circuito fique no estado fundamental? (c) Por que os qubits
supercondutores são operados em refrigeradores de diluição a
\qty{20}{mK}? (d) A flutuação de tensão de ponto zero
$\Delta V = \Delta q/C$ com $\Delta q =
\sqrt{\hbar\omega C/2}$: avalie-a.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:10}
Van der Waals a partir do movimento de ponto zero. Modele dois átomos
neutros à distância $R$ como dois osciladores dipolares idênticos
(carga $e$, massa $m$, frequência $\omega_0$) cujos deslocamentos se
acoplam pela energia dipolar $\lambda\,x_1x_2$, com
$\lambda = e^2/2\pi\varepsilon_0R^3$. (a) Mostre que as coordenadas
normais $(x_1 \pm x_2)/\sqrt2$ oscilam a
$\omega_\pm = \omega_0\sqrt{1 \pm \lambda/m\omega_0^2}$. (b) A energia
fundamental é $\tfrac\hbar2(\omega_+ + \omega_-)$: expanda até segunda
ordem em $\lambda$ e mostre que a energia de interação vale
\[
\Delta E = -\frac{\hbar\lambda^2}{8m^2\omega_0^3}
\ \propto\ -\frac{1}{R^6} .
\]
(c) Por que essa atração é universal --- presente mesmo entre átomos
sem dipolo permanente algum? (d) A lei $1/R^6$ é a força de van der
Waals das correções de gás real do volume do primeiro ano: o que,
microscopicamente, está ``flutuando'' --- e o que aconteceria com essa
força num mundo com $\hbar = 0$?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:11}
Bandas laterais de um íon aprisionado. Um íon numa armadilha harmônica ($f =
\qty{1}{MHz}$) absorve luz de laser. Como o íon se move, seu
espectro de absorção mostra a linha eletrônica em $\nu_0$ ladeada por
linhas em $\nu_0 \pm f$: transições que mudam o número quântico
\emph{de movimento} de $\mp1$ ao mesmo tempo que o eletrônico. (a)
Quanto vale $\hbar\omega_{\text{arm}}$ em $\unit{neV}$, e por que
resolver bandas laterais exige uma linha muito estreita? (b) Excitar a
linha $\nu_0 - f$ retira um quantum de movimento por ciclo: explique o
\emph{resfriamento por banda lateral} até o estado fundamental. (c) Uma
vez que $\langle n\rangle \approx
0$, a banda lateral inferior desaparece por completo --- por que essa
assimetria é uma prova de se ter alcançado o estado fundamental
quântico? (d) É assim que o estado fundamental de movimento de um único
átomo --- e o dos espelhos de escala quilograma do LIGO, por outros
meios --- é certificado: diga que ``temperatura'' o íon alcançou para
$f = \qty{1}{MHz}$ e $\langle n\rangle = 0.05$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:harmonic-oscillator:12}
O limite clássico, quantitativamente. Um oscilador clássico de
amplitude $A$ passa em $[x, x + \dd x]$ a fração $\dd
t/T = \dd x/\pi\sqrt{A^2 - x^2}$. (a) Deduza essa distribuição do
tempo de permanência. (b) Para o estado quântico $n$, tome $A_n$ de $E_n =
\tfrac12 m\omega^2A_n^2$ e compare a distribuição clássica com o
(dado) $|\varphi_n|^2$ promediado localmente: onde as duas concordam e
onde precisam diferir? (c) Mostre que o espaçamento relativo entre
níveis vizinhos, $\Delta E/E$, se anula como $1/n$: a energia se torna
efetivamente contínua. (d) Para o pêndulo do
\cref{exo:b3:harmonic-oscillator:8}(d), estime $n$ e $\Delta
E/E$ e conclua o argumento da correspondência numa frase.
\end{exercise}

\section{Problema: A molécula que aquece a Terra}

\begin{problem}[{Problema de fim de semana --- a escada quântica do
dióxido de carbono e o efeito estufa}]\label{pb:b3:harmonic-oscillator:1}
Uma molécula de dióxido de carbono é uma cadeia linear O=C=O: alguns
osciladores quantizados. O fato de o $\hbar\omega$ do seu modo de
flexão calhar de ficar no meio do brilho térmico que a Terra emite é a
razão de esse gás traço --- quatro moléculas em dez mil --- governar o
clima do planeta. Dados: número de onda da flexão
$\tilde\nu_2 = \qty{667}{cm^{-1}}$ (a unidade do espectroscopista:
$E = hc\tilde\nu$, $\qty{1}{cm^{-1}} =
\qty{1.24e-4}{eV}$); estiramento assimétrico $\tilde\nu_3 =
\qty{2349}{cm^{-1}}$; estiramento simétrico $\tilde\nu_1 =
\qty{1388}{cm^{-1}}$; $k_{\text{B}}T$ a $\qty{288}{K}$ vale
\qty{24.8}{meV}; lei de Wien (volume do segundo ano): $\lambda_{\max}T =
\qty{2898}{\micro m.K}$.

\medskip\noindent\textbf{Parte I --- Os modos de uma molécula linear.}
\begin{enumerate}
  \item Três átomos têm nove \omterm{def:b3:lagrangian-mechanics:coordinates}{graus de liberdade}: quantos são
        translações da molécula inteira, quantos são rotações (a
        molécula é \emph{linear}) e quantas vibrações restam?
  \item Descreva as quatro vibrações: estiramento simétrico,
        estiramento assimétrico e uma flexão duplamente degenerada ---
        por que a flexão vem duas vezes?
  \item A luz se acopla a um \emph{dipolo elétrico} que vibra
        (\cref{exo:b3:harmonic-oscillator:4}). Que modos do
        O=C=O modulam o momento de dipolo, e qual deles é silencioso
        no infravermelho?
  \item Converta os três números de onda em comprimentos de onda de
        fóton e situe-os: quais estão no infravermelho térmico?
  \item Por que os dois principais gases do ar, N$_2$ e O$_2$, quase
        não absorvem infravermelho algum --- e com que consequência
        para a transparência da atmosfera?
  \item O vapor de água, angular e dipolar, absorve em boa parte do
        infravermelho: em que ``janela'' espectral a flexão do dióxido
        de carbono atua praticamente sozinha (compare os
        \qty{15}{\micro m} com as bandas fortes da água abaixo de
        \qty{8}{\micro m} e acima de \qty{20}{\micro m})?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A escada quântica da flexão.}
\begin{enumerate}[resume]
  \item Calcule $\hbar\omega_2$ em meV e a escada $E_n$.
  \item Que fração das moléculas ocupa $n = 1$ a \qty{288}{K}
        (relativamente a $n = 0$, pelo fator de Boltzmann)? E
        $n = 2$?
  \item Que comprimento de onda de fóton conduz $n = 0 \to 1$? Por que
        o \emph{mesmo} comprimento de onda domina a emissão?
  \item Justifique, a partir da escada harmônica, que um único
        comprimento de onda serve a escada inteira ($n \to n + 1$ para
        todo $n$): que propriedade do espaçamento entre níveis está em
        ação?
  \item A molécula também gira, acrescentando estrutura fina: o traço
        de \qty{15}{\micro m} é, na verdade, uma \emph{banda} de cerca
        de \qty{1}{\micro m} de largura. Por que a \emph{largura} da
        banda importa para um gás de efeito estufa (pense no que
        acontece depois que o centro da banda fica opaco)?
  \item Um CO$_2$ vibracionalmente excitado no ar tem muito mais
        chance de perder seu quantum por colisão do que por radiação
        (vida radiativa $\sim\qty{1}{s}$, tempo de colisão
        $\sim\qty{e-9}{s}$): para onde vai, de fato, a energia
        radiante absorvida?
  \item Inversamente, o ar a \qty{288}{K} mantém uma população térmica
        em $n = 1$ (questão 8): o que essa população faz de importante
        para o balanço de energia?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O balanço radiativo do planeta.}
\begin{enumerate}[resume]
  \item O Sol irradia como um corpo a \qty{5800}{K}: calcule seu pico
        de Wien. A flexão do CO$_2$ intercepta muita luz solar?
  \item O solo irradia como um corpo a \qty{288}{K}: calcule seu pico
        de Wien e localize os \qty{15}{\micro m} nessa curva térmica.
  \item Explique o mecanismo do efeito estufa em quatro frases: a luz
        solar que entra, o infravermelho térmico que sai, a
        interceptação em \qty{15}{\micro m} e a reemissão tanto para
        cima \emph{quanto} para baixo.
  \item A reemissão que escapa para o espaço vem de camadas altas e
        frias ($\sim\qty{220}{K}$): por que emitir de uma camada mais
        fria reduz a potência que o planeta emite nesses comprimentos
        de onda (lembre que a emissão térmica cresce com $T$)?
  \item Mais CO$_2$ empurra a camada emissora para cima e para o frio:
        diga numa frase por que a superfície precisa então se aquecer
        para reequilibrar as contas.
  \item O centro da banda já está opaco; o efeito do CO$_2$ adicional
        atua nas \emph{asas} da banda, o que dá um crescimento
        logarítmico do forçamento com a concentração: relacione isso
        com a questão 11.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- Isótopos: a impressão digital
quântica.}
\begin{enumerate}[resume]
  \item A frequência de um oscilador escala como $\sqrt{k/\mu}$: para
        o estiramento assimétrico, substituir o $^{12}$C pelo $^{13}$C
        muda a massa efetiva; a linha observada se desloca de
        \qty{2349}{cm^{-1}} para cerca de \qty{2283}{cm^{-1}}.
        Verifique a ordem de grandeza desse deslocamento de
        $\approx 3\%$ a partir das massas.
  \item Lasers sintonizados nessas duas linhas contam separadamente o
        $^{13}$CO$_2$ e o $^{12}$CO$_2$ numa amostra de gás: explique
        por que a escada quantizada torna possível uma detecção
        resolvida por isótopo.
  \item As plantas preferem o isótopo mais leve, de modo que o carbono
        fóssil é pobre em $^{13}$C: o que a razão isotópica medida do
        CO$_2$ atmosférico fez à medida que sua concentração subiu ---
        e o que isso prova sobre a \emph{fonte} do gás acrescentado?
  \item A mesma física dos \qty{15}{\micro m} opera em Vênus
        ($96\%$ de CO$_2$, \qty{90}{bar}): o que a sua superfície a
        \qty{737}{K} ilustra?
  \item Marte também respira CO$_2$ quase puro, mas a \qty{6}{mbar}, e
        é gélido: o que o trio Vênus--Terra--Marte demonstra sobre qual
        variável controla a intensidade do efeito?
  \item Resuma o resultado central: um quantum de \qty{83}{meV} ---
        o $\hbar\omega$ de um triatômico que se dobra --- estacionado
        em \qty{15}{\micro m} no espectro térmico de um planeta a
        \qty{288}{K}, absorvido, termalizado em nanossegundos e
        reemitido de altitudes frias, é o mecanismo pelo qual $0.04\%$
        do ar fixa a temperatura da Terra.
\end{enumerate}
\end{problem}
