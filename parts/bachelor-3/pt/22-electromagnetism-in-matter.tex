\chapter{Eletromagnetismo na matéria}\label{ch:b3:electromagnetism-in-matter}

% Provenance (not for print): completes the "électromagnétisme dans la
% matière" module (dielectrics, magnetism, ferromagnetism) common to the
% surveyed third-year curricula; the Year 2 volume covered only wave
% propagation in media.

Deslize uma folha de plástico entre as placas de um capacitor carregado
e a tensão cai, como se carga tivesse surgido do nada. Enrole uma bobina
em torno de um prego de ferro e o campo magnético dele cresce mil vezes.
Descole um ímã decorativo da geladeira e ele \emph{se lembra} de que
lado foi magnetizado --- por décadas, sem fonte de alimentação. O volume
do segundo ano tratou os campos na matéria por atacado, escondendo o
material dentro de um índice de refração; este capítulo abre o material.
O plano é o mesmo para a eletricidade e para o magnetismo: a matéria é
uma multidão de dipolos microscópicos; some-os numa densidade ($\vect P$
ou $\vect M$); invente um campo auxiliar ($\vect D$ ou $\vect H$) que só
enxergue as cargas e correntes que controlamos; depois desça à escala
microscópica --- onde os spins do \cref{ch:b3:spin-two-level}, o fator de
Boltzmann do \cref{ch:b3:canonical-ensemble} e a transição de campo médio
do \cref{ch:b3:phase-transitions} assumem --- para calcular o que o
material faz. O capítulo termina no ferro-velho, projetando o
eletroímã que levanta carros.

\section{Polarização e o campo de deslocamento}

\begin{definition}[Polarização e cargas ligadas]\label{def:b3:electromagnetism-in-matter:polarisation}
\index{polarização}\index{carga ligada}\index{dielétrico}
Um \emph{dielétrico} é um isolante cujas moléculas adquirem (ou já
possuem) momentos de dipolo elétrico. A resposta do material se
resume na \emph{polarização} $\vect P$: o momento de dipolo
por unidade de volume, $\vect P = n\langle\vect p\rangle$ para $n$ moléculas
por unidade de volume. Um bloco polarizado carrega \emph{cargas ligadas}
--- não livres para sair, mas perfeitamente reais: uma densidade
superficial $\sigma_{\text{b}} = \vect P\cdot\vect n$ onde a polarização
encontra a superfície, e uma densidade volumétrica $\rho_{\text{b}} =
-\nabla\cdot\vect P$ onde quer que ela não seja uniforme. Uma laje
uniformemente polarizada equivale exatamente a duas folhas de carga $\pm P$
nas faces dela: no interior, todas as cabeças e caudas dos dipolos se cancelam.
\end{definition}

\begin{proposition}[O campo de deslocamento]\label{prop:b3:electromagnetism-in-matter:displacement}
\index{deslocamento elétrico}\index{suscetibilidade elétrica}\index{permissividade relativa}
A carga total é a livre mais a ligada; a lei de Gauss com $\rho =
\rho_{\text{f}} + \rho_{\text{b}}$ se rearranja numa lei para o
\emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{deslocamento elétrico}} $\vect D = \varepsilon_0\vect E +
\vect P$:
\[
\nabla\cdot\vect D = \rho_{\text{f}} :
\]
as fontes de $\vect D$ são apenas as cargas livres --- as que estão nas
nossas placas e nos nossos fios. Num \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielétrico} \emph{linear} a resposta é
proporcional, $\vect P = \varepsilon_0\chi_{\text{e}}\vect E$, e assim
$\vect D = \varepsilon_0\varepsilon_{\text{r}}\vect E$ com
$\varepsilon_{\text{r}} = 1 + \chi_{\text{e}}$ a \emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{permissividade
relativa}}: ar $1.0006$, polietileno $2.3$, vidro $\sim 5$,
água uns enormes $80$. Entre placas de capacitor mantidas a carga
fixa, o $\vect D$ não muda com o \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielétrico}, e assim $\vect E =
\vect D/\varepsilon_0\varepsilon_{\text{r}}$ \emph{cai} por
$\varepsilon_{\text{r}}$: as cargas superficiais ligadas encaram as
livres e cancelam a maior parte do campo delas. A capacitância fica
multiplicada por $\varepsilon_{\text{r}}$ --- toda a indústria de
capacitores numa letra grega.
\end{proposition}

\begin{proof}
$\nabla\cdot(\varepsilon_0\vect E) = \rho_{\text{f}} +
\rho_{\text{b}} = \rho_{\text{f}} - \nabla\cdot\vect P$; passe o
termo de \omterm{def:b3:electromagnetism-in-matter:polarisation}{polarização} para a esquerda.
\end{proof}

\section{De onde vem a permissividade}

\begin{proposition}[Polarização induzida e Clausius--Mossotti]\label{prop:b3:electromagnetism-in-matter:mossotti}
\index{polarizabilidade}\index{relação de Clausius--Mossotti}
Um átomo num campo $\vect E_{\text{loc}}$ se estica num dipolo
$\vect p = \alpha\vect E_{\text{loc}}$, com $\alpha$ a
\emph{polarizabilidade} (dimensão: $\varepsilon_0\times$volume ---
grosso modo $\varepsilon_0$ vezes o volume atômico). Num gás diluído,
$E_{\text{loc}} \approx E$ e $\chi_{\text{e}} =
n\alpha/\varepsilon_0$ é pequeno. Num meio denso, cada molécula também
sente os vizinhos polarizados dela: escavar uma pequena cavidade esférica
em torno dela dá $\vect E_{\text{loc}} = \vect E + \vect
P/3\varepsilon_0$, e a autoconsistência produz a
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{relação de Clausius--Mossotti}}
\[
\frac{\varepsilon_{\text{r}} - 1}{\varepsilon_{\text{r}} + 2}
= \frac{n\alpha}{3\varepsilon_0} :
\]
meça a suscetibilidade minúscula de um vapor e preveja a do líquido ---
funciona com alguns por cento para líquidos apolares.
\end{proposition}
\admitted

\begin{remark}
O campo da cavidade $\vect P/3\varepsilon_0$ é o resultado da esfera
uniformemente polarizada, citado aqui sem demonstração; a dedução
honesta, por valores de contorno, cabe a um curso dedicado de
eletrodinâmica. Todo o resto acima é contabilidade.
\end{remark}

\begin{proposition}[Moléculas polares: a função de Langevin]\label{prop:b3:electromagnetism-in-matter:langevin}
\index{função de Langevin}\index{polarização de orientação}
Uma molécula com um dipolo \emph{permanente} $p$ (água:
$p = \qty{6.2e-30}{C.m}$) não precisa ser esticada --- só alinhada,
contra a agitação térmica. A média canônica do
\cref{ch:b3:canonical-ensemble} sobre as orientações, com energia
$-pE\cos\theta$, dá
\[
\langle\cos\theta\rangle = L(x) = \coth x - \frac1x ,
\qquad x = \frac{pE}{k_{\text{B}}T} ,
\]
a \emph{\omterm{prop:b3:electromagnetism-in-matter:langevin}{função de Langevin}}: linear ($L \approx x/3$) em $x$ pequeno e
saturando em $1$ quando o campo vence de vez. No regime linear,
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ --- uma lei de
Curie elétrica, caindo como $1/T$: aqueça um líquido polar e a
permissividade dele \emph{cai}, a assinatura que distingue dipolos
permanentes de induzidos. Para a água à temperatura ambiente, isso prevê
$\chi_{\text{e}} \sim 12$ --- a magnitude certa, com a cooperação das
ligações de hidrogênio empurrando o valor real até $80$.
\end{proposition}

\begin{proof}[Demonstração parcial]
$Z = \int_0^\pi \eu^{x\cos\theta}\,2\pi\sin\theta\,\dd\theta \propto
\sinh x/x$, e $\langle\cos\theta\rangle = \dd\ln Z/\dd x = \coth x
- 1/x$. Expandir $\coth x = 1/x + x/3 - \dots$ para $x$ pequeno dá
$L \approx x/3$, e portanto $P = np\langle\cos\theta\rangle =
np^2E/3k_{\text{B}}T$.
\end{proof}

\section{Magnetização e o campo \texorpdfstring{$H$}{H}}

\begin{definition}[Magnetização, correntes ligadas e H]\label{def:b3:electromagnetism-in-matter:magnetisation}
\index{magnetização}\index{campo magnético H}\index{suscetibilidade magnética}
A história magnética corre em paralelo. A matéria é cheia de espiras
microscópicas de corrente --- elétrons que orbitam e giram, cada um um
dipolo magnético $\vect\mu$; a densidade deles é a \emph{magnetização}
$\vect M = n\langle\vect\mu\rangle$. Um bloco uniformemente magnetizado
equivale a uma \emph{corrente superficial ligada} circulando pelos
lados dele (as espiras interiores se cancelam aos pares; um ímã de barra
\emph{é} um solenoide de corrente ligada). Separando as correntes em
livres e ligadas, a lei de Ampère se rearranja em torno do campo auxiliar
\[
\vect H = \frac{\vect B}{\mu_0} - \vect M , \qquad
\oint \vect H\cdot\dd\vect\ell = I_{\text{livre}} :
\]
o $\vect H$ tem por fonte as correntes dos nossos fios (unidades
\unit{A/m}), enquanto o $\vect B$ guarda a física --- forças, fluxo,
indução. Os meios lineares têm $\vect M = \chi_{\text{m}}\vect H$ e
$\vect B = \mu_0(1 + \chi_{\text{m}})\vect H =
\mu_0\mu_{\text{r}}\vect H$.
\end{definition}

\begin{remark}[Três personalidades magnéticas]\label{rem:b3:electromagnetism-in-matter:zoo}
\index{diamagnetismo}\index{paramagnetismo}\index{ferromagnetismo}
Os materiais respondem ao $\vect H$ de três maneiras.
Os \emph{diamagnetos} ($\chi_{\text{m}} \sim -10^{-5}$: água, cobre,
grafite) respondem como a lei de Lenz tornada permanente: o campo
aplicado perturba toda órbita eletrônica de modo a se opor a ele ---
fraco, universal, independente da temperatura e repelido pelos ímãs (uma
rã já foi levitada assim num furo de \qty{16}{T}).
Os \emph{paramagnetos} ($\chi_{\text{m}} \sim +10^{-5}$ a $10^{-3}$:
alumínio, O$_2$) possuem momentos permanentes --- spins desemparelhados
do \cref{ch:b3:identical-particles} --- que se alinham como os dipolos de
Langevin: $\chi_{\text{m}} = \mu_0 n\mu^2/3k_{\text{B}}T$, uma lei de
Curie em $1/T$ (a versão quântica de dois níveis é o
\cref{exo:b3:canonical-ensemble:11}).
Os \emph{ferromagnetos} (ferro, cobalto, níquel: $\mu_{\text{r}}$ na casa
dos milhares) são paramagnetos cujos spins também conversam entre si ---
o acoplamento de troca do \cref{exo:b3:identical-particles:12} ---
e abaixo da \omterm{thm:b3:phase-transitions:meanfield}{temperatura de Curie} eles se alinham \emph{espontaneamente}:
o \cref{thm:b3:phase-transitions:meanfield}, agora posto a trabalhar.
\end{remark}

\section{Ferromagnetismo na prática: domínios e histerese}

\begin{definition}[Domínios e histerese]\label{def:b3:electromagnetism-in-matter:hysteresis}
\index{domínio magnético}\index{histerese}\index{remanência}\index{coercividade}
Um bloco bruto de ferro \emph{não} está magnetizado: ele se estilhaça em
\emph{domínios}, regiões da escala do micrômetro, cada uma plenamente
magnetizada, mas apontando para lados diferentes, de modo que o campo
exterior (e o custo em energia dele) quase se cancela. Um $\vect H$
aplicado move as paredes de domínio --- os domínios favoráveis crescem
--- e depois gira domínios inteiros para a linha. As paredes engancham
em defeitos, e por isso o processo é irreversível: varra o
$H$ para cima e para baixo e o $B(H)$ traça um laço, o \emph{ciclo de
histerese}. Desligue a corrente e um campo \emph{remanente}
$B_{\text{r}}$ sobrevive; cancelá-lo exige o campo \emph{coercivo}
reverso $H_{\text{c}}$. A área encerrada pelo laço é a
energia dissipada por ciclo e por unidade de volume. Os materiais
\emph{moles} (aço-silício, ferrites: laço fino, $H_{\text{c}}$ pequeno)
fazem os núcleos de transformadores e motores; os \emph{duros}
(alnico, Nd$_2$Fe$_{14}$B: laço gordo, $H_{\text{c}}$ enorme) fazem os
ímãs permanentes --- e a memória magnética: o ímã de geladeira
e o disco rígido são laços de histerese que se recusam a esquecer.
\end{definition}

\begin{example}[O eletroímã de núcleo de ferro]\label{ex:b3:electromagnetism-in-matter:electromagnet}
Enrole $N$ espiras que carregam $I$ num anel de ferro ($\mu_{\text{r}}
\sim 5000$) com um pequeno entreferro $e$. A lei de Ampère para
$\vect H$ ao longo do laço dá $H_{\text{ferro}}\ell + H_{\text{ent}}e = NI$,
e a continuidade do fluxo torna o $B$ comum, logo
\[
B\Big(\frac{\ell}{\mu_0\mu_{\text{r}}} + \frac{e}{\mu_0}\Big) = NI .
\]
Com $\ell = \qty{1}{m}$ e $e = \qty{1}{cm}$, o termo do entreferro é
cinquenta vezes o do ferro: quase todo o esforço da bobina é gasto
empurrando campo através de um centímetro de ar. Esse é o circuito
magnético numa linha --- o ferro é um condutor quase perfeito de fluxo,
e o ar, o resistor --- e é por isso que motores, relés e guindastes de
ferro-velho mantêm os entreferros deles impiedosamente finos
(\cref{pb:b3:electromagnetism-in-matter:1}).
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:1}
Contabilidade de \omterm{def:b3:electromagnetism-in-matter:polarisation}{cargas ligadas}. Uma laje de espessura $d$ está
uniformemente polarizada ao longo da normal dela, com módulo $P$. (a) Dê
as densidades de \omterm{def:b3:electromagnetism-in-matter:polarisation}{carga ligada} em cada face. (b) Mostre que o campo
\emph{interno} devido a essas folhas é $E = P/\varepsilon_0$, opondo-se
a $\vect P$. (c) Quanto vale o $\vect D$ dentro, sem carga livre em
lugar algum? (d) Um eletreto de barra ($\vect P$ congelado) é o análogo
elétrico de que objeto magnético?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:2}
Capacitor com \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielétrico}. Um capacitor de placas paralelas
($C_0 = \qty{100}{pF}$) é carregado a \qty{12}{V} e desconectado.
Uma laje com $\varepsilon_{\text{r}} = 4$ o preenche. Ache (a) a nova
capacitância; (b) a nova tensão; (c) a energia antes e depois
--- para onde foi a diferença? (d) Repita (b)--(c) com a
bateria mantida ligada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:3}
O zoológico das suscetibilidades. Classifique (dia-, para- ou
ferromagnético) e justifique pela estrutura eletrônica onde puder: água
($\chi_{\text{m}} = -9\times10^{-6}$), alumínio
($+2.2\times10^{-5}$), oxigênio líquido ($+3.5\times10^{-3}$ ---
por que o O$_2$ é magnético, para começar?), níquel ($\mu_{\text{r}} \sim 600$
em campo pequeno). Qual dos quatro um ímã forte atrairia visivelmente,
e qual repeliria de leve?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:4}
$B$, $H$, $M$. Um solenoide longo ($n = \num{2000}$ espiras por metro,
$I = \qty{1.5}{A}$) é preenchido com ferro, e mede-se $B = \qty{1.6}{T}$.
Calcule (a) o $H$; (b) o $M$; (c) a amplificação
$B/\mu_0H$; (d) a corrente superficial ligada por metro equivalente ao
$M$ --- compare-a com o $nI$ da própria bobina.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:5}
Cabo coaxial, isolado. Uma linha coaxial (raio interno $a$, externo
$b$) carrega carga livre $\lambda$ por unidade de comprimento no núcleo,
com \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielétrico} $\varepsilon_{\text{r}}$ entre eles. (a) Ache
$\vect D(r)$ pela simetria. (b) Deduza $\vect E$ e $\vect P$.
(c) Calcule as densidades de \omterm{def:b3:electromagnetism-in-matter:polarisation}{carga ligada} em $r = a$ e $r = b$ e
verifique que somam zero por unidade de comprimento. (d) Mostre que a
capacitância por unidade de comprimento fica multiplicada por $\varepsilon_{\text{r}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:6}
Clausius--Mossotti na prática. O argônio gasoso a \qty{1}{bar} e
\qty{300}{K} tem $\varepsilon_{\text{r}} = 1.00052$. (a) Extraia
$\alpha/\varepsilon_0$ (um volume) e compare com o volume
atômico. (b) O argônio líquido tem $n = \qty{2.1e28}{m^{-3}}$: preveja
o $\varepsilon_{\text{r}}$ por Clausius--Mossotti (medido: 1.53).
(c) Por que o ingênuo $\chi = n\alpha/\varepsilon_0$ erra para cima
menos feio aqui do que para a água? (d) Para que moléculas o esquema
inteiro tem de falhar, e o que o substitui?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:7}
A permissividade gigante da água. (a) A partir da
\cref{prop:b3:electromagnetism-in-matter:langevin}, calcule
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ para a água
($n = \qty{3.3e28}{m^{-3}}$, $p = \qty{6.2e-30}{C.m}$,
$T = \qty{300}{K}$). (b) Estime o campo necessário para $x = 1$
(saturação séria) e compare com a rigidez dielétrica da água,
$\sim\qty{7e7}{V/m}$. (c) Os fornos de micro-ondas operam a \qty{2.45}{GHz}:
por que uma resposta \emph{dissipativa} de dipolos girantes aquece a
comida, e por que o gelo aquece bem mais devagar que a água líquida? (d)
Preveja o sinal de $\dd\varepsilon_{\text{r}}/\dd T$ para a água e para o
argônio.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:8}
Diamagnetismo estimado. Trate um elétron atômico como uma carga num
anel de raio $r$; um $B$ aplicado muda a frequência angular dele pelo
deslocamento de Larmor $\Delta\omega = eB/2m_{\text{e}}$. (a) Mostre que o
momento induzido é $\Delta\mu = -e^2r^2B/4m_{\text{e}}$ (opondo-se a
$B$). (b) Fazendo a média das orientações e somando $Z$ elétrons por
átomo, acha-se $\chi_{\text{m}} = -\mu_0ne^2Z\langle
r^2\rangle/6m_{\text{e}}$: avalie para a água ($n =
\qty{3.3e28}{m^{-3}}$ moléculas, $Z = 10$, $\langle r^2\rangle
\approx (\qty{0.7}{\angstrom})^2$) e compare com o valor medido
$-9\times10^{-6}$. (c) Por que o diamagnetismo independe da temperatura
e o paramagnetismo não? (d) Por que até uma rã levita em
\qty{16}{T}, e por que o campo do ímã tem de ser \emph{não uniforme}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:9}
Paramagnetismo de Curie e resfriamento. Um sal carrega $n =
\qty{2e27}{m^{-3}}$ íons de momento $\mu \approx \mu_{\text{B}}$.
(a) Calcule $\chi_{\text{m}} = \mu_0n\mu^2/3k_{\text{B}}T$ a
\qty{300}{K} e a \qty{1}{K}. (b) A que temperatura o
$\chi_{\text{m}}$ chegaria a $1$ --- e que física (aqui desprezada)
intervém antes na maioria dos sais? (c) Desmagnetização adiabática:
magnetize a \qty{1}{K}, isole e então retire o campo devagar ---
explique, com a entropia de spin do
\cref{ch:b3:microcanonical-ensemble}, por que a amostra esfria. (d) Por
que o método precisa de um \emph{para}magneto, e não de um ferromagneto?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:10}
O circuito magnético. Um toro de ferro ($\ell = \qty{60}{cm}$,
$\mu_{\text{r}} = 4000$, seção $A = \qty{16}{cm^2}$) carrega
$N = 500$ espiras e um entreferro $e = \qty{4}{mm}$. (a) Mostre que $NI =
B(\ell/\mu_0\mu_{\text{r}} + e/\mu_0)$ e calcule a corrente para
$B = \qty{1.2}{T}$. (b) Que fração do $NI$ é gasta no entreferro?
(c) Defina a relutância $\mathcal R = \ell/\mu A$ de cada
segmento e reformule (a) como uma ``lei de Ohm'' em série para o fluxo. (d)
O ferro satura perto de \qty{1.8}{T}: explique o que acontece com o
modelo de circuito --- e com o seu motor --- para além disso.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:11}
Perdas por \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histerese}. Um núcleo de transformador (volume \qty{4e-3}{m^3})
opera a \qty{50}{Hz}. O laço de aço-silício dele encerra
$\sim\qty{40}{J/m^3}$ por ciclo. (a) Calcule a potência perdida por
\omterm{def:b3:electromagnetism-in-matter:hysteresis}{histerese}. (b) O mesmo núcleo em aço duro
($\sim\qty{6e3}{J/m^3}$): perda e veredito. (c) As correntes de Foucault
acrescentam uma perda $\propto f^2$: explique por que os núcleos são
laminados e por que as ferrites assumem nas radiofrequências. (d) Um bit
de disco rígido tem de \emph{guardar} a \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetização} dele contra os
chutes térmicos por dez anos: relacione as exigências sobre
$H_{\text{c}}$ para memória com as de um transformador e conclua que
material algum, sozinho, faz os dois trabalhos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:12}
O campo desmagnetizante. Dentro de um corpo uniformemente magnetizado
sem corrente livre, o $\oint\vect H\cdot\dd\vect\ell = 0$ obriga
o $\vect H$ interno a \emph{se opor} ao $\vect M$ ($\vect H =
-N_{\text{d}}\vect M$, com $N_{\text{d}}$ um fator de forma: $1/3$
para uma esfera, $\approx 0$ ao longo de uma agulha comprida, $\approx 1$
através de uma placa fina). (a) Justifique os limites da agulha e da
placa com a imagem das correntes ligadas (o solenoide). (b) Por que um ímã
atarracado se desmagnetiza em parte enquanto uma agulha guarda a
\omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetização} dela? (c) Conecte com as agulhas de bússola e com os grãos
alongados da fita magnética. (d) Uma esfera de ferro doce fica num
$B_0$ externo uniforme: explique por que a resposta dela satura em $\vect H
\approx 0$ por dentro, ou seja, $M \approx 3B_0/\mu_0$ no máximo, por
maior que seja o $\mu_{\text{r}}$.
\end{exercise}

\section{Problema: o guindaste do ferro-velho}

\begin{problem}\label{pb:b3:electromagnetism-in-matter:1}
\emph{O guindaste do ferro-velho.} Um pátio de reciclagem encomenda um
eletroímã de guindaste: um caldeirão de ferro de face plana, com um metro
de diâmetro, que precisa levantar fardos de carros prensados de duas
toneladas e então --- igualmente importante --- \emph{largá-los} sob
comando. Você é o engenheiro de projeto.

\medskip\noindent\textbf{Parte I --- Por que ferro.}

\begin{enumerate}
  \item A bobina nua: 400 espiras carregando \qty{25}{A} espalhadas
        por um caminho magnético de $\sim\qty{1}{m}$. Estime $B \sim
        \mu_0NI/\ell$ sem ferro.
  \item Explique em duas frases, com domínios, por que encher a
        bobina com ferro multiplica isso por $\sim\mu_{\text{r}}$.
  \item A \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetização} do ferro satura em $M_{\text{s}} \approx
        \qty{1.7e6}{A/m}$. Calcule o teto $\mu_0M_{\text{s}}$
        que isso impõe ao campo no polo, faça a bobina o que fizer.
  \item Cada átomo de ferro contribui com $\approx 2.2$ \omterm{prop:b3:quantum-angular-momentum:magnetic}{magnétons de Bohr}.
        Confira: com $n = \qty{8.5e28}{m^{-3}}$, recupere
        $M_{\text{s}}$.
  \item Por que as faces polares têm de ser usinadas planas e mantidas
        livres de ferrugem e de sujeira? (Pense no
        \cref{ex:b3:electromagnetism-in-matter:electromagnet}.)
  \item A própria carga passa a fazer parte do circuito magnético.
        Esboce o caminho do fluxo: núcleo do caldeirão, face norte, fardo
        de aço, face sul, de volta pela culatra.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O circuito magnético e a força.}

\begin{enumerate}[resume]
  \item Modele o circuito: caminho de ferro $\ell = \qty{1.2}{m}$,
        $\mu_{\text{r}} = 2000$, e dois entreferros efetivos
        (contato face--fardo) de $e = \qty{1.5}{mm}$ cada. Escreva
        a lei de Ampère para $H$ ao longo do laço.
  \item Calcule os três termos de relutância por unidade de área
        ($\ell/\mu_{\text{r}}$ contra $2e$) e mostre que os
        entreferros milimétricos ainda consomem a maior parte do esforço da bobina.
  \item Com $NI = \num{10000}$ ampère-espiras, calcule o $B$ nos
        entreferros.
  \item A pressão de levantamento em cada face polar é $B^2/2\mu_0$
        (densidade de energia magnética liberada por metro de aproximação).
        Avalie-a em \unit{kPa} para o seu $B$.
  \item Área total das faces polares $A = \qty{0.12}{m^2}$: calcule a
        força de levantamento e converta em toneladas.
  \item Ela atende à especificação de duas toneladas com margem de
        fator dois? Se não, ajuste o $NI$ e enuncie a nova corrente.
  \item Os fardos prensados tocam as faces em talvez um terço da
        área deles, com entreferros efetivos mais largos. Recalcule a força para
        $e = \qty{4}{mm}$, $A_{\text{eff}} = \qty{0.04}{m^2}$ e
        comente por que os levantamentos nominais citam capacidade de ``placa plana''.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Lembrar e esquecer.}

\begin{enumerate}[resume]
  \item O pátio aciona a chave para largar um fardo --- e ele fica
        pendurado. Nomeie o culpado, com o laço de \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histerese}.
  \item Por que o caldeirão tem de ser de ferro \emph{doce} (com
        $H_{\text{c}}$ pequeno e $B_{\text{r}}$ pequeno) e não de aço
        duro de ímã?
  \item Mesmo o ferro doce guarda um pouco de \omterm{def:b3:electromagnetism-in-matter:hysteresis}{remanência}. Proponha a
        cura padrão: um breve pulso de corrente \emph{reversa} ---
        que ponto do laço ele mira?
  \item Alguns controladores aplicam, em vez disso, uma corrente
        alternada decrescente. Esboce o que a trajetória $B$--$H$ faz e por que
        ela termina desmagnetizada.
  \item O guindaste também manipula chapas quentes saídas direto do forno,
        a \qty{800}{\celsius}. O ponto de Curie do ferro é
        \qty{1043}{K}: o que acontece com a força de levantamento, e por quê?
        (\cref{thm:b3:phase-transitions:meanfield}.)
  \item Um estagiário de verão sugere economizar cobre dobrando
        o entreferro e dobrando o $NI$. Use as fórmulas da Parte II para
        explicar a assimetria: o que se reduz à metade e o que apenas
        se mantém?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- Potência, calor e a conta.}

\begin{enumerate}[resume]
  \item A bobina: 400 espiras de cobre, comprimento médio por espira de
        \qty{2.5}{m}, seção do fio de \qty{16}{mm^2}, resistividade
        $\rho = \qty{1.7e-8}{\Omega.m}$. Calcule a resistência dela.
  \item Em $I = \qty{25}{A}$: a potência dissipada e a energia
        diária de um turno de oito horas com \qty{60}{\percent} de uso.
  \item A energia magnética armazenada nos dois entreferros
        ($B^2/2\mu_0 \times$ volume): calcule-a e compare com
        um \emph{segundo} de dissipação na bobina. Para onde vai todo
        o resto da energia elétrica?
  \item Largar a carga devolve quase nada da energia armazenada
        à rede. Explique, com a área do laço e o pico
        indutivo, por que o controlador precisa de um caminho de roda-livre
        (flyback) atravessando a bobina.
  \item Falta energia com um fardo no ar. O que o fardo
        faz, e o que a \omterm{def:b3:electromagnetism-in-matter:hysteresis}{remanência} \emph{sozinha} segura? Justifique
        a regra do pátio de que ninguém anda sob um ímã energizado ---
        e o argumento de venda dos guindastes com bateria de reserva.
  \item Resuma o projeto em quatro linhas: teto de campo fixado por
        $M_{\text{s}}$, força por $B^2A/2\mu_0$, controlabilidade pela
        \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histerese} do ferro doce e custo de operação por $RI^2$ --- o
        capítulo inteiro pendurado num guindaste.
\end{enumerate}
\end{problem}
