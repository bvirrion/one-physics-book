\chapter{O ensemble canônico}\label{ch:b3:canonical-ensemble}

Os sistemas isolados são uma ficção de teórico: as amostras reais ficam
em termostatos, em salas, em oceanos de ar --- em contato com um
reservatório que não fixa a energia delas, mas a \emph{temperatura}. O
volume do segundo ano encontrou empiricamente a lei resultante: a
probabilidade de um estado de energia $E$ carrega o fator
$\eu^{-E/k_{\text{B}}T}$. Este capítulo deduz esse fator de Boltzmann
em quatro linhas a partir da contagem do
\cref{ch:b3:microcanonical-ensemble} e depois constrói a máquina que
faz da física estatística uma disciplina industrial: a \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{função de
partição}} $Z$, uma única soma da qual a energia, a entropia, a pressão,
a capacidade térmica e as flutuações caem todas por derivação. As
primeiras vitórias da máquina são recontadas aqui: por que as
capacidades térmicas morrem em baixa temperatura (Einstein, 1907 --- a
primeira teoria quântica da matéria), por que a capacidade térmica do
hidrogênio sobe uma escadaria, por que o céu rarefaz exponencialmente
--- e como Perrin, observando grãos microscópicos assentarem numa gota
d'água, contou o número de Avogadro e encerrou o debate sobre a
realidade dos átomos.

\section{A distribuição de Boltzmann, deduzida}

\begin{theorem}[Distribuição canônica]\label{thm:b3:canonical-ensemble:boltzmann}
\index{distribuição de Boltzmann}\index{ensemble canônico}\index{função de partição}
Um sistema que troca energia com um grande reservatório à temperatura
$T$ ocupa seu \omterm{def:b3:microcanonical-ensemble:states}{microestado} $s$ (de energia $E_s$) com probabilidade
\[
\mathcal P_s = \frac{\eu^{-E_s/k_{\text{B}}T}}{Z} , \qquad
Z = \sum_s \eu^{-E_s/k_{\text{B}}T}
\]
($\beta = 1/k_{\text{B}}T$ daqui em diante). O $Z$, a \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{função de
partição}}, normaliza as probabilidades --- e acaba por conter toda a
termodinâmica do sistema.
\end{theorem}

\begin{proof}
O sistema no estado $s$ deixa ao reservatório a energia $E - E_s$:
pelo \omterm{thm:b3:microcanonical-ensemble:postulate}{postulado fundamental} aplicado ao todo, $\mathcal P_s
\propto \Omega_{\text{res}}(E - E_s)$. Expanda a entropia do
reservatório, enorme e suave, até primeira ordem:
$S_{\text{res}}(E - E_s)
= S_{\text{res}}(E) - E_s\,\partial S_{\text{res}}/\partial E =
S_{\text{res}}(E) - E_s/T$, pela definição de temperatura. Logo,
$\Omega_{\text{res}} \propto \eu^{-E_s/k_{\text{B}}T}$. (As ordens
superiores morrem com o tamanho do reservatório.)
\end{proof}

\begin{proposition}[A máquina]\label{prop:b3:canonical-ensemble:machine}
\index{energia livre}\index{capacidade térmica!e flutuações}
A partir de $Z(T, V, N)$:
\[
\langle E\rangle = -\frac{\partial\ln Z}{\partial\beta} , \qquad
F = -k_{\text{B}}T\ln Z , \qquad
S = -\frac{\partial F}{\partial T} , \qquad
P = -\frac{\partial F}{\partial V} ,
\]
em que $F = \langle E\rangle - TS$ é a \emph{\omterm{prop:b3:canonical-ensemble:machine}{energia livre}}; e a
capacidade térmica é uma \emph{flutuação}:
\[
C = \frac{\partial\langle E\rangle}{\partial T}
 = \frac{\langle E^2\rangle - \langle E\rangle^2}
{k_{\text{B}}T^2} .
\]
Para subsistemas independentes e distinguíveis, $Z = z^N$; para $N$
\omterm{def:b3:identical-particles:exchange}{partículas idênticas} numa caixa comum, $Z = z^N/N!$. A $T$ e $V$
fixos, o equilíbrio \emph{minimiza} $F$: a natureza troca energia por
entropia à taxa de câmbio $T$ --- o princípio isolado mais útil da
física da matéria.
\end{proposition}

\begin{proof}[Demonstração parcial]
$-\partial_\beta\ln Z = \sum E_s\eu^{-\beta E_s}/Z = \langle
E\rangle$; mais uma derivada dá $\langle E^2\rangle - \langle
E\rangle^2$, e a regra da cadeia converte $\partial_\beta$ em
$\partial_T$. As identificações de $F$, $S$ e $P$ decorrem de comparar
$\dd(-k_{\text{B}}T\ln Z)$ com o $\dd F = -S\dd T - P\dd V$
termodinâmico; a fatoração é a exponencial de uma soma. O princípio de
mínimo: $F_{\text{sistema}}$ decrescer é $S_{\text{total}}$ crescer, pois
$\Delta S_{\text{res}} =
-\Delta E_{\text{sist}}/T$.
\end{proof}

\section{As primeiras vitórias}

\begin{example}[Dois níveis, em duas linhas]\label{ex:b3:canonical-ensemble:twolevel}
Para os níveis $0, \epsilon$: $z = 1 + \eu^{-\beta\epsilon}$, o que dá
$\langle E\rangle = N\epsilon/(\eu^{\beta\epsilon} + 1)$ --- o
resultado que custou uma página de Stirling à rota microcanônica
(\cref{exo:b3:microcanonical-ensemble:8}). O formalismo canônico é o
microcanônico com a combinatória pré-digerida; a corcova de Schottky em
$C(T)$ decorre de uma única derivação.
\end{example}

\begin{example}[O sólido de Einstein e a morte de Dulong--Petit]\label{ex:b3:canonical-ensemble:einstein}
\index{sólido de Einstein}\index{lei de Dulong--Petit}
Modele um cristal como $3N$ osciladores quânticos de frequência
$\omega$ (\cref{ch:b3:harmonic-oscillator}). Por oscilador,
\[
z = \sum_n\eu^{-\beta\hbar\omega(n + 1/2)}
 = \frac{1}{2\sinh(\beta\hbar\omega/2)} , \qquad
\langle E\rangle = \frac{\hbar\omega}{2} +
\frac{\hbar\omega}{\eu^{\beta\hbar\omega} - 1} .
\]
Em $T$ alto: $\langle E\rangle \to k_{\text{B}}T$ por oscilador e
$C \to 3Nk_{\text{B}}$ --- a \omterm{ex:b3:canonical-ensemble:einstein}{lei de Dulong--Petit} do volume do primeiro
ano, explicada. Em $T$ baixo, o quantum $\hbar\omega$ fica inacessível
e $C$ desaba exponencialmente --- como se mede, e inexplicável
classicamente. A curva de Einstein de 1907, com um parâmetro por
elemento, foi a primeira aplicação dos quanta à matéria comum; o
diamante, de ligações duras e átomos leves ($\hbar\omega/k_{\text{B}}
\approx \qty{1300}{K}$), ainda está
``congelado'' à temperatura ambiente --- a anomalia que intrigara os
químicos por oitenta anos (\cref{exo:b3:canonical-ensemble:6}).
\end{example}

\begin{theorem}[A equipartição, com sua licença]\label{thm:b3:canonical-ensemble:equipartition}
\index{teorema da equipartição}
Toda coordenada ou momento que entre \emph{quadraticamente} na energia
contribui, no regime clássico (de alta temperatura), com
\[
\langle\epsilon\rangle = \tfrac12 k_{\text{B}}T
\]
para a energia média: $\tfrac32 k_{\text{B}}T$ para um átomo de gás
monoatômico, $\tfrac52$ para uma diatômica que gira, $3k_{\text{B}}T$
para um oscilador. A licença expira quando o $k_{\text{B}}T$ cai abaixo
do espaçamento de níveis do modo: o modo congela e sua contribuição
some --- a solução dos escândalos das capacidades térmicas do século
XIX, desenhada como a escadaria acima.
\end{theorem}

\begin{proof}[Demonstração parcial]
Para $\epsilon = ax^2$,
\[
\langle\epsilon\rangle
 = \frac{\int ax^2\,\eu^{-\beta ax^2}\dd x}
        {\int\eu^{-\beta ax^2}\dd x}
 = -\partial_\beta\ln\!\int\eu^{-\beta ax^2}\dd x
 = -\partial_\beta\ln\beta^{-1/2} = \frac{1}{2\beta} .
\] O congelamento é o cálculo do
\cref{ex:b3:canonical-ensemble:einstein}, modo a modo.
\end{proof}

\begin{example}[O gás, canonicamente]\label{ex:b3:canonical-ensemble:gas}
Um átomo numa caixa: $z = V/\lambda_T^3$ (a contagem de estados da
\cref{prop:b3:schrodinger-three-dimensions:counting} ponderada por
Boltzmann); com $N$ átomos idênticos: $Z = z^N/N!$. Então $F =
-Nk_{\text{B}}T[\ln(V/N\lambda_T^3) + 1]$, e a derivação entrega
$PV = Nk_{\text{B}}T$, a entropia de Sackur--Tetrode e
$\langle E\rangle = \tfrac32 Nk_{\text{B}}T$ --- o gás ideal inteiro,
a partir de uma só integral gaussiana. O fator de Boltzmann aplicado à
energia cinética de uma molécula dá a distribuição de velocidades de
Maxwell do volume do primeiro ano, agora deduzida; aplicado à energia
potencial $mgh$, dá a atmosfera exponencial --- e, numa gota d'água, a
escada de grãos de Perrin (\cref{pb:b3:canonical-ensemble:1}).
\end{example}

\begin{method}[Ofício canônico]\label{met:b3:canonical-ensemble:method}
(1) Liste os estados e as energias de \emph{uma} unidade; calcule $z$.
(2) Fatore: $Z = z^N$ (ou $z^N/N!$ para \omterm{def:b3:identical-particles:exchange}{partículas idênticas} que
compartilham o espaço); tome o $\ln$ cedo. (3) Derive: em $\beta$ para
a energia, em $T$ para a entropia via $F$, em $V$ para a pressão; uma
segunda derivada para $C$ e para as flutuações. (4) Confira as duas
pontas: o $T$ alto tem de reproduzir a equipartição, e o $T$ baixo tem
de congelar com uma cauda $\eu^{-\Delta/k_{\text{B}}T}$. (5)
Competições (enovelamento, ligação, alinhamento): escreva
$F = E - TS$ para cada alternativa e deixe vencer a menor --- a
transição fica em $T \approx \Delta
E/\Delta S$.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:1}
(a) Escreva a razão de populações de dois níveis separados por
$\epsilon$ à temperatura $T$. (b) Numa chama a \qty{2500}{K}, que
fração dos átomos de sódio fica no estado excitado de \qty{2.1}{eV} da
linha D (degenerescência 2 no fundamental e 6 no excitado: razão de
populações $3\eu^{-\beta\epsilon}$)?
(c) Por que a chama, ainda assim, arde em amarelo (quantos átomos por
cm$^3$ bastam)? (d) A que temperatura a fração excitada chegaria a
$10\%$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:2}
Um sistema de três níveis: $0, \epsilon, 2\epsilon$. (a) Escreva $z$.
(b) Calcule $\langle E\rangle$ e confira os dois limites de
temperatura. (c) A que $T$ o nível do meio fica maximamente povoado
\emph{em termos absolutos}? (d) Mostre que temperatura alguma, por mais
alta, torna um nível superior mais povoado que um inferior --- que
conceito do capítulo 16 seria necessário para isso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:3}
A atmosfera isotérmica. (a) Aplique o fator de Boltzmann à energia
potencial $mgh$ e deduza $n(h) = n_0\eu^{-mgh/k_{\text{B}}
T}$. (b) Calcule a altura de escala do ar ($m = \qty{4.8e-26}{kg}$) a
\qty{288}{K}. (c) A pressão no cume do Everest como fração da do nível
do mar. (d) Por que o decaimento da atmosfera real é próximo, mas não
exatamente exponencial (o que mantivemos constante e não é)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:4}
Contabilidade da equipartição. Conte os termos quadráticos e preveja o
$C_V$ molar (a) do argônio; (b) do N$_2$ à temperatura ambiente
(rotação ligada, vibração congelada); (c) do N$_2$ a \qty{3000}{K}; (d)
de um sólido cristalino (Dulong--Petit). Onde os valores medidos
$12.5$, $20.8$, $\approx26$ e $\approx\qty{25}{J/(mol.K)}$ concordam, e
o que cada discrepância ensina?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:5}
O oscilador quântico, canonicamente. (a) Some a série geométrica de
$z$. (b) Deduza $\langle E\rangle$ e identifique $\langle
n\rangle = 1/(\eu^{\beta\hbar\omega} - 1)$ --- o resultado do
\cref{exo:b3:harmonic-oscillator:6}, agora sem esforço. (c) Derive para
obter $C(T)$ e verifique os dois limites. (d) Por que a forma de
$\langle n\rangle$ está prestes a ficar famosa
(\cref{ch:b3:photons-phonons})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:6}
Einstein contra os dados. (a) Pela fórmula da figura, em que
$T/\theta_{\text{E}}$ o $C$ caiu à metade do valor de Dulong--Petit?
(b) Diamante: $\theta_{\text{E}} \approx \qty{1300}{K}$ --- calcule
$C/3Nk_{\text{B}}$ a \qty{300}{K} e explique a ``anomalia do diamante''
do século XIX. (c) Chumbo: $\theta_{\text{E}}
\approx \qty{90}{K}$ --- por que o chumbo sempre foi ``bem-comportado''?
(d) As medidas em $T$ muito baixo mostram $C \propto T^3$, e não
exponencial: qual das hipóteses de Einstein falha (todos os
osciladores com \emph{uma} só frequência), e quem a consertou
(\cref{ch:b3:photons-phonons})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:7}
A cauda de Maxwell e o hidrogênio ausente. (a) A partir do fator de
Boltzmann sobre a energia cinética, escreva a distribuição de
velocidades e localize a velocidade mais provável do N$_2$ e do H$_2$ a
\qty{288}{K}. (b) A velocidade de escape da Terra é \qty{11.2}{km/s}:
calcule $mv_{\text{esc}}^2/2k_{\text{B}}T$ para os dois gases. (c) A
fração que escapa vai como $\eu^{-mv_{\text{esc}}^2/2k_{\text{B}}T}$:
compare os dois expoentes e conclua que gás vaza em tempo geológico.
(d) Conecte com a composição observada das atmosferas da Terra (sem
H$_2$ livre) e de Júpiter (quase toda H$_2$) --- que dois parâmetros
invertem o veredito?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:8}
As flutuações encontram a resposta. (a) Demonstre $C = (\langle E^2\rangle -
\langle E\rangle^2)/k_{\text{B}}T^2$ a partir de duas derivadas de $\ln
Z$. (b) Verifique isso explicitamente no sistema de dois níveis. (c)
Para $N$ unidades independentes, mostre que a flutuação de energia
\emph{relativa} cai como $1/\sqrt N$. (d) Enuncie a moral: o que a
disposição de um sistema a absorver calor (uma resposta) tem a ver com
o quanto sua energia vibra (uma flutuação) --- a barganha recorrente da
física estatística.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:9}
Boltzmann no laboratório de química. As taxas de reação carregam o
fator $\eu^{-E_a/k_{\text{B}}T}$ (a travessia de uma barreira de
ativação $E_a$ --- Arrhenius). (a) Mostre que a regra prática ``a taxa
dobra a cada \qty{10}{K} perto da temperatura ambiente'' corresponde a $E_a \approx
\qty{0.55}{eV}$. (b) Por que fator essa reação desacelera
numa geladeira (\qty{5}{\celsius})? (c) Cozinhar um ovo em altitude: a
água ferve a \qty{93}{\celsius} num passo de \qty{2000}{m} --- estime o
tempo extra de cozimento. (d) Por que é a cauda além da barreira, e não
a energia média, que governa a química (quais moléculas reagem)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:10}
Enovelamento de dois estados. Uma biomolécula está enovelada (energia
$0$, uma configuração) ou desenovelada (energia $\Delta E > 0$, $\Omega_u =
\eu^{\Delta S/k_{\text{B}}}$ configurações). (a) Escreva a fração enovelada
em função de $T$. (b) Mostre que o ponto médio da ``fusão'' é
$T_{\text{m}} = \Delta E/\Delta S$ e interprete-o como um empate em $F = E -
TS$. (c) Com $\Delta E = \qty{3.0}{eV}$ e $\Delta S =
100\,k_{\text{B}}$ (a unidade cooperativa de uma proteína
pequena): calcule $T_{\text{m}}$ e a largura da transição. (d) Por que
a \emph{cooperatividade} (muitos contatos que se rompem juntos, com
$\Delta E$ \emph{e} $\Delta S$ grandes) torna a fusão mais abrupta ---
e como as máquinas de ciclagem térmica de DNA (PCR) exploram
exatamente isso?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:11}
Paramagnetismo e resfriamento magnético. $N$ spins $\tfrac12$ de
momento $\mu$ num campo $B$. (a) Mostre que
$M = N\mu\tanh(\mu B/k_{\text{B}}T)$ e expanda até a lei de Curie
$M \approx N\mu^2B/k_{\text{B}}T$. (b) Avalie o alinhamento
$\mu B/k_{\text{B}}T$ para momentos eletrônicos em $B = \qty{1}{T}$ e
$T = \qty{300}{K}$, e depois a \qty{1}{K}. (c) Desmagnetização
adiabática: magnetize a \qty{1}{K}, isole e reduza $B$ dez vezes ---
mostre que entropia constante significa $\mu B/k_{\text{B}}T$
constante, de modo que $T$ cai dez vezes. (d) O que fixa o piso dessa
geladeira (as interações entre os spins --- estime a escala dipolar
$\mu_0\mu^2/4\pi a^3$ para $a =
\qty{0.5}{nm}$, em unidades de temperatura)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:12}
A escadaria rotacional, quantitativamente. Os níveis rotacionais de uma
diatômica são $E_J = BJ(J+1)$, com degenerescência $2J + 1$
(\cref{ch:b3:quantum-angular-momentum}). (a) Escreva
$z_{\text{rot}}$ e mostre que, para $k_{\text{B}}T \gg B$, a soma vira
a integral $k_{\text{B}}T/B$: o $k_{\text{B}}$ da equipartição
recuperado em $C_V$. (b) Defina $\theta_{\text{rot}} =
B/k_{\text{B}}$ e avalie-o para o H$_2$ ($B = \qty{7.5}{meV}$) e o
N$_2$ ($B = \qty{0.25}{meV}$): que gás mostra o degrau rotacional em
temperaturas acessíveis? (c) Esboce a escadaria completa do $C_V(T)$ do
hidrogênio, com seus três patamares e dois degraus, pondo números nos
dois. (d) O comportamento medido em $T$ baixo do H$_2$ é ainda
complicado pela mistura orto--para 3:1 do
\cref{exo:b3:identical-particles:8}: diga numa frase como a estatística
de spin nuclear chega até a capacidade térmica de um gás.
\end{exercise}

\section{Problema: Contar Avogadro numa gota d'água}

\begin{problem}[{Problema de fim de semana --- os grãos de Perrin e a
realidade dos átomos}]\label{pb:b3:canonical-ensemble:1}
Em 1908, Jean Perrin suspendeu grãos microscópicos de resina em água,
deixou-os assentar e os contou camada a camada ao microscópio. A escada
exponencial que ele encontrou era o fator de Boltzmann tornado visível
--- e, da altura de escala dela, ele extraiu o número de Avogadro,
convencendo os últimos céticos de que os átomos existem. Prêmio Nobel,
1926. Dados: grãos de goma-guta de raio $r =
\qty{0.212}{\micro m}$ e densidade $\rho_{\text{g}} =
\qty{1207}{kg/m^3}$; água com $\rho_{\text{w}} = \qty{999}{kg/m^3}$;
$T = \qty{293}{K}$; $g = \qty{9.81}{m/s^2}$.

\medskip\noindent\textbf{Parte I --- O fator de Boltzmann visível.}
\begin{enumerate}
  \item Um grão na água sente a gravidade menos o empuxo: calcule sua
        massa \emph{efetiva} $m' = \tfrac43\pi r^3(\rho_{\text{g}}
        - \rho_{\text{w}})$ e o peso dela.
  \item Escreva o perfil de concentração de equilíbrio $n(h)$ a partir
        do fator de Boltzmann.
  \item Calcule a altura de escala $h_0 = k_{\text{B}}T/m'g$ com o
        $k_{\text{B}}$ moderno.
  \item Por que os grãos precisam ter tamanho tão precisamente único
        (como $h_0$ depende de $r$)?
  \item Compare $h_0$ com a mesma fórmula para as moléculas do ar: por
        que a atmosfera dos grãos tem micrômetros de altura, e a do ar
        tem quilômetros?
  \item Perrin contou (numa das rodadas) concentrações relativas de
        $100 : 55 : 30 : 17$ em quatro profundidades igualmente
        espaçadas, a \qty{30}{\micro m} uma da outra: verifique que
        isso é uma escada exponencial e extraia o $h_0$ dela.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Pesar o invisível.}
\begin{enumerate}[resume]
  \item Inverta: a partir do $h_0$ medido do item 6 e do $m'g$
        conhecido, extraia $k_{\text{B}}$.
  \item A constante dos gases $R = \qty{8.314}{J/(mol.K)}$ já era
        conhecida da química macroscópica: combine-a com o seu
        $k_{\text{B}}$ para obter o número de Avogadro $N_{\text{A}} =
        R/k_{\text{B}}$.
  \item As rodadas de Perrin deram $N_{\text{A}}$ entre $5.5$ e
        $\num{7.2e23}$: compare com o valor moderno
        \num{6.022e23} e comente o feito, dados o microscópio e o
        cronômetro dele.
  \item A partir de $N_{\text{A}}$, calcule a massa de um único \omterm{thm:b3:hydrogen-atom:levels}{átomo
        de hidrogênio} --- o número que os atomistas queriam havia um
        século.
  \item Explique a estrutura lógica: que medidas \emph{macroscópicas}
        ($R$; o tamanho e a densidade dos grãos; a escada) se combinam
        para pesar um átomo, sem que átomo algum jamais seja visto?
  \item Por que os grãos --- de $10^{10}$ massas atômicas cada ---
        obedecem à mesma estatística de Boltzmann que as moléculas (o
        que, na dedução do
        \cref{thm:b3:canonical-ensemble:boltzmann}, se importa com o
        tamanho)?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O tremor que sela o caso.}
\begin{enumerate}[resume]
  \item Os mesmos grãos tremem: a fórmula de Einstein de 1905 para o
        movimento browniano dá $\langle x^2\rangle = 2Dt$ com $D =
        k_{\text{B}}T/6\pi\eta r$ (água: $\eta =
        \qty{e-3}{Pa.s}$). Calcule $D$ para os grãos de Perrin.
  \item Quanto um grão vagueia em um minuto? Perrin poderia medi-lo
        com uma ocular micrométrica e um cronômetro?
  \item Perrin verificou $\langle x^2\rangle \propto t$ e extraiu
        $k_{\text{B}}$ \emph{de novo}, de modo independente: por que
        duas rotas sem relação (uma escada estática; um tremor
        dinâmico) até um mesmo número tiveram tanto peso probatório?
  \item O tremor é a equipartição aplicada ao grão: cada componente da
        velocidade carrega $\tfrac12 k_{\text{B}}T$. Estime a
        velocidade térmica quadrática média do grão ($m \approx
        \qty{4.8e-17}{kg}$).
  \item Por que essa velocidade nunca é vista diretamente (o que
        interrompe o voo livre depois de nanômetros), e o que se vê
        \emph{no lugar dela}?
  \item Diga que dois capítulos deste livro se encontram na
        observação: a mecânica do arrasto (volume do segundo ano,
        viscosidade) e a estatística deste capítulo.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- O que ficou resolvido.}
\begin{enumerate}[resume]
  \item Ostwald e Mach sustentavam que os átomos eram ficções
        contábeis: diga numa frase por que um $N_{\text{A}}$
        \emph{contado} a partir de escadas de grãos encerrou essa
        posição.
  \item Liste outras três rotas dos anos 1900 que convergiram para o
        mesmo $N_{\text{A}}$ (o azul do céu,
        \cref{exo:b3:microcanonical-ensemble:12}; a eletrólise mais a
        carga do elétron; a produção de hélio pela radioatividade) ---
        por que a \emph{convergência} importou mais que qualquer valor
        isolado?
  \item A escada de Perrin é um equilíbrio entre quais duas moedas
        deste capítulo (a energia puxando para baixo, a entropia
        espalhando para cima), cotadas a que taxa?
  \item Usos modernos da mesma física: as ultracentrífugas analíticas
        giram proteínas a $10^5g$ para comprimir as ``atmosferas''
        delas em escadas mensuráveis --- mostre que multiplicar $g$ por
        $10^5$ divide $h_0$ pelo mesmo fator, e estime $h_0$ para uma
        proteína de massa efetiva \qty{e-22}{kg} a $10^5g$ e
        \qty{293}{K}.
  \item Dobrar o raio do grão divide $h_0$ por oito: delimite a janela
        prática de tamanhos de grão entre ``escada alta demais para se
        ver um gradiente'' e ``escada mais fina que um grão'', para um
        campo de microscópio de \qty{100}{\micro m} de profundidade.
  \item Desde 2019, o SI \emph{define} exatamente $k_{\text{B}}$ e
        $N_{\text{A}}$: diga o que um experimento à moda de Perrin mede
        \emph{hoje} (uma verificação de coerência, ou uma calibração do
        aparelho e dos grãos) --- e por que a física não mudou.
  \item Resuma o resultado central: a concentração de um grão de
        \qty{0.2}{\micro m} cai à metade a cada $\sim\qty{35}{\micro m}$
        de altura; lida com $n(h) = n_0\eu^{-m'gh/k_{\text{B}}T}$,
        essa escada rendeu $N_{\text{A}} \approx \num{6e23}$ --- átomos
        contados, e não conjecturados, numa gota d'água sobre a platina
        de um microscópio.
\end{enumerate}
\end{problem}
