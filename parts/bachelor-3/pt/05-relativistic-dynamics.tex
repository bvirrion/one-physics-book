\chapter{Dinâmica relativística}\label{ch:b3:relativistic-dynamics}

Em 1964, William Bertozzi filmou elétrons disparados por um acelerador
linear: cada pulso recebia uma energia medida, e sua velocidade era
cronometrada ao longo de \qty{8.4}{m} de voo. Newton previa que
elétrons de \qty{15}{MeV} deveriam voar a oito vezes a velocidade da
luz; o cronômetro dizia $0.9999c$ e se recusava a ir além, por mais
forte que se empurrassem os elétrons. A cinemática nos disse, no
capítulo anterior, que $c$ é um limite; a dinâmica deve agora explicar o
que acontece com o empurrão --- para onde vai o trabalho quando a
velocidade não cresce mais. A resposta reorganiza a mecânica em torno de
um novo momento e de uma nova energia, unidos na equação mais famosa da
física: a energia armazenada é massa, a massa é energia residente,
$E = mc^2$. Este capítulo constrói a dinâmica e depois a põe a
trabalhar onde ela vive todo dia --- decaimentos radioativos, colisões
de partículas e os aceleradores que transformam movimento em matéria
nova.

\section{Momento e energia, reconstruídos}

\begin{remark}[Por que $m\vect v$ teve de sair]\label{rem:b3:relativistic-dynamics:why}
O momento clássico $m\vect v$ se conserva em todo referencial \emph{se}
as velocidades se transformarem à maneira de Galileu. Elas não se
transformam assim (\cref{prop:b3:relativistic-kinematics:addition}):
uma colisão que conserva $\sum m\vect v$ num referencial deixa de
conservá-lo em outro, de modo que o momento antigo não pode ser uma lei
da natureza --- no máximo, a sombra de uma, a baixas velocidades. O
conserto é medir a velocidade com o relógio \emph{próprio} da
partícula: $\dd\vect r/\dd\tau =
\gamma\,\vect v$ se transforma de modo limpo, e $m\,\dd\vect r/\dd\tau$
acaba por se conservar em todos os referenciais de uma vez.
\end{remark}

\begin{definition}[Momento e energia relativísticos]\label{def:b3:relativistic-dynamics:momentum-energy}
\index{momento relativístico}\index{energia relativística}\index{energia de repouso}
Uma partícula de massa $m$ e velocidade $\vect v$ ($\gamma = 1/\sqrt{1 -
v^2/c^2}$) carrega o momento e a energia
\[
\vect p = \gamma m\vect v , \qquad
E = \gamma mc^2 .
\]
Em repouso, $E_0 = mc^2$: a \emph{energia de repouso}. A energia
cinética é o que o movimento acrescenta,
$E_k = E - mc^2 = (\gamma - 1)mc^2$, que para $v
\ll c$ se reduz ao familiar $\tfrac12 mv^2$ (expanda $\gamma$). Em todo
processo isolado, o $\vect p$ total e a $E$ total --- energias de
repouso incluídas --- se conservam.
\end{definition}

\begin{theorem}[A relação energia--momento]\label{thm:b3:relativistic-dynamics:relation}
\index{relação energia--momento}
Para qualquer partícula,
\[
E^2 = (pc)^2 + (mc^2)^2 ,
\qquad
\vect v = \frac{\vect p\,c^2}{E} :
\]
a combinação $E^2 - p^2c^2 = m^2c^4$ tem o mesmo valor em todo
referencial --- ela é, para a energia e o momento, o que o intervalo é
para o tempo e o espaço. Dois regimes: a baixa velocidade,
$E \approx mc^2 + p^2/2m$; a alta energia ($E \gg mc^2$),
$E \approx pc$ e $v \to c$: empurrar mais forte acrescenta energia e
momento, e quase nenhuma velocidade.
\end{theorem}

\begin{proof}
$E^2 - p^2c^2 = \gamma^2m^2c^4(1 - v^2/c^2) = m^2c^4$; e $pc^2/E =
\gamma mvc^2/\gamma mc^2 = v$. A invariância de referencial decorre de
$m$ e $c$ serem independentes do referencial.
\end{proof}

\begin{proposition}[Partículas sem massa]\label{prop:b3:relativistic-dynamics:massless}
\index{fóton!momento}
Uma partícula de massa nula tem $E = pc$ e se move exatamente a $c$ em
todo referencial; não pode ser desacelerada, apenas desviada para o
vermelho. O fóton é o caso padrão: $E = h\nu$ e $p = h\nu/c = h/\lambda$
--- as relações de Planck--Einstein do volume do primeiro ano, vistas
agora como o canto $m = 0$ da mecânica.
\end{proposition}

\begin{proof}
Faça $m = 0$ no \cref{thm:b3:relativistic-dynamics:relation}: $E = pc$
e $v = pc^2/E = c$.
\end{proof}

\begin{example}[Ordens de grandeza]\label{ex:b3:relativistic-dynamics:orders}
A física de partículas conta a energia em elétron-volts e as massas em
$\unit{MeV}/c^2$: elétron $0.511$, próton $938.3$, múon $105.7$, píon
$139.6$. Um elétron de energia cinética \qty{1}{MeV} tem $\gamma =
2.96$ e $v = 0.94c$ --- já plenamente relativístico, e é por isso que a
eletrônica permanece clássica (\unit{eV}) e a física nuclear não
(\unit{MeV}). Um próton de \qty{6.8}{TeV} do LHC tem $\gamma = 7250$ e
fica atrás de um fóton por apenas \qty{2.9}{m/s}.
\end{example}

\section{A massa é energia residente}

\begin{theorem}[Equivalência massa--energia]\label{thm:b3:relativistic-dynamics:emc2}
\index{equivalência massa--energia}\index{defeito de massa}
A massa de um corpo é a energia total do seu conteúdo, dividida por
$c^2$, medida no referencial em que ele repousa: aqueça um corpo, dê
corda à sua mola, excite seus átomos, e ele pesa mais de $\Delta E/c^2$;
ligue suas partes entre si e ele pesa \emph{menos}, da energia de
ligação sobre $c^2$ --- o \emph{\omterm{thm:b3:relativistic-dynamics:emc2}{defeito de massa}}. Reciprocamente, a
\omterm{def:b3:relativistic-dynamics:momentum-energy}{energia de repouso} pode ser liberada: sempre que as massas finais somam
menos que as iniciais, a diferença $\Delta m\,c^2$ emerge como energia
cinética ou radiação.
\end{theorem}
\admitted

\begin{example}[O balanço de $c^2$]\label{ex:b3:relativistic-dynamics:ledger}
$c^2 = \qty{9e16}{J/kg}$: um grama de diferença de massa são
\qty{9e13}{J}, a energia de uma cidade pequena por um dia. As ligações
químicas (alguns eV por molécula) deslocam as massas em partes em
$10^{10}$ --- para sempre impesáveis, e é por isso que a química nunca
percebeu. A ligação nuclear as desloca em quase $1\%$: o hélio pesa
$0.7\%$ menos que seus quatro ingredientes de hidrogênio, e essa fração
que falta, escoando do Sol como luz, lhe custa $\Delta m = L_\odot/c^2 =
\qty{4.3e9}{kg}$ a cada segundo --- quatro milhões de toneladas de sol.
A aniquilação acerta a conta inteira: um elétron que encontra um
pósitron nada deixa senão dois fótons de \qty{511}{keV} cada, o sinal
com que os tomógrafos PET observam um cérebro vivo.
\end{example}

\begin{remark}[O que significa ``conversão'']\label{rem:b3:relativistic-dynamics:conversion}
Nada material ``se transforma'' em energia: a energia total esteve
conservada o tempo todo. O que muda é a sua \emph{residência} --- da
\omterm{def:b3:relativistic-dynamics:momentum-energy}{energia de repouso}, que pesa, para a energia cinética e a radiação, que
voam. O enunciado profundo de $E = mc^2$ é que a própria inércia é um
conteúdo de energia: uma caixa de gás quente resiste mais à aceleração
que a mesma caixa fria.
\end{remark}

\section{Colisões e decaimentos}

\begin{definition}[Quadrimomento e massa invariante]\label{def:b3:relativistic-dynamics:four-momentum}
\index{quadrimomento}\index{massa invariante}
Reúna a energia e o momento de uma partícula em seu
\emph{quadrimomento} $P = (E/c, \vect p)$. Para um sistema de
partículas, some componente a componente:
$E_{\text{tot}} = \sum E_i$, $\vect p_{\text{tot}} = \sum\vect p_i$.
A \emph{massa invariante} $M$ do sistema é definida por
\[
M^2c^4 = E_{\text{tot}}^2 - \|\vect p_{\text{tot}}\|^2c^2 :
\]
o mesmo número em todo referencial, igual à energia total (dividida
por $c^2$) no \emph{referencial do centro de momento}, onde
$\vect p_{\text{tot}} = \vect 0$. Note que $M$ excede a soma das
massas das partes sempre que estas se movem umas em relação às outras
--- dois fótons que se afastam têm $M > 0$, embora nenhum deles tenha
massa.
\end{definition}

\begin{method}[Resolver uma colisão relativística]\label{met:b3:relativistic-dynamics:method}
(1) Escreva a conservação do \omterm{def:b3:relativistic-dynamics:four-momentum}{quadrimomento}: $E$ e $\vect p$ juntos,
nunca um só. (2) Calcule o invariante $E^2 - p^2c^2$ do conjunto que
for conveniente --- ele pode ser avaliado no referencial mais fácil e
usado em qualquer outro. (3) Limiares: uma reação é possível quando a
\omterm{def:b3:relativistic-dynamics:four-momentum}{massa invariante} do estado inicial alcança a soma das massas de
repouso do estado final. (4) Decaimentos em repouso: os momentos dos
produtos são opostos e iguais; as energias seguem só das massas. (5)
Só no fim, se for pedido, converta em velocidades por $v = pc^2/E$.
\end{method}

\begin{example}[A assinatura do píon]\label{ex:b3:relativistic-dynamics:pion}
Um píon carregado em repouso decai, $\pi^+ \to \mu^+ + \nu$ (com o
neutrino efetivamente sem massa). A conservação do momento põe os
produtos em sentidos opostos com $p$ igual; a conservação da energia se
lê $m_\pi c^2 =
\sqrt{p^2c^2 + m_\mu^2c^4} + pc$. Resolvendo:
\[
pc = \frac{(m_\pi^2 - m_\mu^2)c^2}{2m_\pi} = \qty{29.8}{MeV} ,
\]
de modo que todo múon vindo de um píon que decai em repouso nasce com
exatamente \qty{4.1}{MeV} de energia cinética --- uma linha
monoenergética, cuja observação (Powell, 1947) foi o modo como a massa
do píon foi pesada pela primeira vez. Os decaimentos de dois corpos
sempre produzem linhas assim; os de três corpos as espalham em
espectros, que foi precisamente como se suspeitou pela primeira vez do
neutrino no decaimento nuclear $\beta$.
\end{example}

\begin{proposition}[Limiares: a tirania do alvo fixo]\label{prop:b3:relativistic-dynamics:threshold}
\index{energia de limiar}\index{colisor}
Para criar novas partículas, só a energia do \emph{centro de momento},
$Mc^2$, está disponível. Um feixe de energia $E$ que atinge um alvo de
massa $m_{\text{t}}$ em repouso fornece
\[
M^2c^4 = m_{\text{b}}^2c^4 + m_{\text{t}}^2c^4 +
2\,E\,m_{\text{t}}c^2 :
\]
a energia útil só cresce como $\sqrt E$ --- o resto se desperdiça
empurrando os destroços para a frente. Dois feixes idênticos que
colidem de frente fornecem $Mc^2 = 2E$: tudo aproveitável. Essa única
fórmula é a razão de as máquinas de fronteira serem colisores
(\cref{pb:b3:relativistic-dynamics:1}).
\end{proposition}

\begin{proof}
Avalie $E_{\text{tot}}^2 - p_{\text{tot}}^2c^2$ no laboratório:
$(E + m_{\text{t}}c^2)^2 - p^2c^2$, com $p$ o momento do feixe;
expanda e use $E^2 - p^2c^2 = m_{\text{b}}^2c^4$. Para o colisor,
$\vect p_{\text{tot}} = \vect 0$ e $Mc^2 = E_{\text{tot}}$.
\end{proof}

\section{A força e as máquinas}

\begin{proposition}[Equação de movimento relativística]\label{prop:b3:relativistic-dynamics:force}
\index{síncrotron}
A lei de Newton sobrevive na forma
\[
\vect F = \frac{\dd\vect p}{\dd t} = \frac{\dd(\gamma m\vect v)}
{\dd t} , \qquad
\frac{\dd E}{\dd t} = \vect F\cdot\vect v .
\]
Um campo magnético, sempre perpendicular a $\vect v$, não altera
energia alguma e curva a trajetória num círculo de raio
\[
r = \frac{p}{qB}
\]
--- a mesma fórmula do volume do primeiro ano, com o $p$
relativístico. Mas a frequência de circulação $qB/\gamma m$ agora
\emph{cai} à medida que a partícula ganha energia: o empurrão de
frequência fixa do cíclotron sai do compasso na escala do MeV, e as
máquinas de alta energia --- os \emph{síncrotrons} --- em vez disso
elevam seu campo e sua frequência em sincronia com $\gamma$, mantendo o
feixe num único anel.
\end{proposition}

\begin{proof}[Demonstração parcial]
A lei da força é a definição de dinâmica coerente com o novo momento
($\dd E/\dd t$ decorre de
$E^2 = p^2c^2 + m^2c^4$: $E\,\dot E = c^2\vect p\cdot\dot{\vect p}$).
Quanto ao círculo: $|\dd\vect p/\dd t| = qvB$ com $|\vect p|$ constante
significa que o vetor momento gira à taxa $qvB/p$, de modo que o raio
vale $v/(qvB/p) = p/qB$.
\end{proof}

\begin{example}[Ler o LHC como um exercício]\label{ex:b3:relativistic-dynamics:lhc}
Prótons de $E = \qty{6.8}{TeV}$ em dipolos de $B = \qty{8.3}{T}$:
$p \approx E/c$ (ultrarrelativístico), logo $r = p/qB =
\qty{2.7}{km}$ --- e, de fato, o raio de curvatura magnética do anel é
\qty{2.8}{km}, sendo os \qty{27}{km} de circunferência quase todos de
ímãs. As colisões frontais fornecem $Mc^2 = \qty{13.6}{TeV}$; alcançar
isso num alvo fixo exigiria um feixe de $2E^2/m_{\text{p}}c^2 \approx
\qty{e17}{eV}$ --- cem vezes o alcance de qualquer máquina já
construída. Foi a fórmula do colisor, e não ímãs mais fortes, que
comprou a fronteira moderna de energia.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:1}
Um elétron ($mc^2 = \qty{0.511}{MeV}$) se move a $0.99c$. Calcule
$\gamma$, seu momento em $\unit{MeV}/c$ e suas energias total e
cinética. As mesmas perguntas para um próton ($mc^2 = \qty{938}{MeV}$)
de energia cinética \qty{1}{GeV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:2}
(a) Quanta energia dorme num grama de matéria? (b) A explosão de
Hiroshima liberou cerca de \qty{6e13}{J}: que diferença de massa é
essa? (c) O Sol irradia $L = \qty{3.8e26}{W}$: quanta massa ele perde
por segundo, e que fração de seus $\qty{2e30}{kg}$ em dez bilhões de
anos? (d) A bateria do seu telefone armazena \qty{5e4}{J}: de quanto um
telefone carregado é mais pesado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:3}
(a) Que momento o feixe de um ponteiro laser de \qty{5}{mW} carrega por
segundo, e que força o ponteiro sente? (b) Que força a luz do Sol
(\qty{1.4}{kW/m^2}) exerce sobre uma vela solar perfeitamente refletora
de \qty{100}{m^2}? (c) A partir do repouso, com que velocidade uma
nave-vela de \qty{10}{kg} se move depois de um mês? (d) Por que a cauda
de poeira de um cometa aponta para longe do Sol?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:4}
Prática com unidades. (a) Mostre que $\unit{MeV}/c^2$ é uma unidade de
massa e converta a massa do elétron em quilogramas. (b) Quanto vale
\qty{1}{GeV/c} em $\unit{kg.m/s}$? (c) O quadrado de uma \omterm{def:b3:relativistic-dynamics:four-momentum}{massa
invariante} resulta em $s = \qty{16}{GeV^2}$ (em unidades com $c = 1$):
que energia de centro de momento é essa? (d) Por que os físicos de
partículas fazem $c = 1$, e o que um leitor precisa reinserir para
obter números no SI?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:5}
Os elétrons de Bertozzi tinham energias cinéticas de $0.5$, $1$, $4.5$
e \qty{15}{MeV}. (a) Calcule a previsão de Newton $v^2 = 2E_k/m$ para
cada uma, em unidades de $c^2$. (b) Calcule o $v^2/c^2$ relativístico.
(c) O tempo de voo dele sobre \qty{8.4}{m} a \qty{15}{MeV}: o que
marcou o relógio, e o que Newton teria previsto? (d) A energia dos
elétrons também foi medida calorimetricamente --- pelo calor no
impacto: por que esse passo era o verdadeiro ponto do experimento?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:6}
O píon carregado decai em repouso: $\pi^+ \to \mu^+\nu$, $m_\pi c^2 =
\qty{139.6}{MeV}$, $m_\mu c^2 = \qty{105.7}{MeV}$, $m_\nu \approx 0$.
(a) Mostre que $pc = (m_\pi^2 - m_\mu^2)c^4/2m_\pi c^2$ e avalie-o.
(b) A energia cinética e a velocidade do múon. (c) A energia do
neutrino. (d) Em voo a $\gamma_\pi = 50$, quais são as energias máxima
e mínima do múon de decaimento no laboratório (decaimento para a frente
e para trás)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:7}
\omterm{def:b3:relativistic-dynamics:four-momentum}{Massa invariante} em ação. (a) Dois fótons de \qty{200}{MeV} cada voam
a $60^\circ$ um do outro: calcule a \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invariante} do par.
(b) Um píon neutro ($m_{\pi^0}c^2 = \qty{135}{MeV}$) decai em dois
fótons: qual é a \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invariante} do par de fótons, em qualquer
referencial? (c) Explique como um experimento ``descobre'' uma
partícula na forma de uma protuberância na distribuição de \omterm{def:b3:relativistic-dynamics:four-momentum}{massa
invariante} de seus produtos de decaimento. (d) Dois fótons de energias
iguais voam exatamente paralelos: \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invariante}? O que isso diz
sobre o referencial de repouso de um feixe de luz?
\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:9}
Fótons que colidem. (a) Mostre que um único fóton no vácuo não pode

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:10}
Compton, por \omterm{def:b3:relativistic-dynamics:four-momentum}{quadrimomentos}. Um fóton de comprimento de onda $\lambda$

\lambda' - \lambda = \frac{h}{m_{\text{e}}c}\,(1 - \cos\theta) .
\]
(c) Avalie $h/m_{\text{e}}c$ e o desvio máximo; por que o efeito é

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:11}
O fim do espectro dos raios cósmicos. O universo se banha em fótons de

(c) Acima dele, o universo é opaco aos prótons além de uns

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:12}
O foguete de fótons. Um foguete de massa inicial $m_{\text{i}}$ emite

\[
\frac{m_{\text{i}}}{m_{\text{f}}} =

\]

\end{exercise}

\section{Problema: Fabricar matéria --- o antipróton e o Bevatron}

\begin{problem}[{Problema de fim de semana --- como a conservação do
momento pôs preço no antimundo}]\label{pb:b3:relativistic-dynamics:1}
As equações de Dirac exigiam, desde 1931, que o próton tivesse um gêmeo
espelhado de carga oposta. Fabricar um significava comprar sua \omterm{def:b3:relativistic-dynamics:momentum-energy}{energia
de repouso} com energia de feixe --- e a conservação do momento fixou o
preço. Este problema calcula esse preço, projeta a máquina que o pagou
e acompanha a descoberta de 1955. Dados:
$m_{\text{p}}c^2 = \qty{938.3}{MeV}$; a carga e o \emph{número
bariônico} (prótons e nêutrons $+1$, antiprótons $-1$) se conservam em
toda reação.

\medskip\noindent\textbf{Parte I --- O ferramental do \omterm{def:b3:relativistic-dynamics:four-momentum}{quadrimomento}.}
\begin{enumerate}
  \item Para uma partícula, mostre que $E^2 - p^2c^2 = m^2c^4$ a
        partir das definições de $E$ e $\vect p$.
  \item Para um sistema, por que $E_{\text{tot}}^2 -
        p_{\text{tot}}^2c^2$ é o mesmo em todos os referenciais, e a
        que ele é igual no referencial do centro de momento?
  \item Um feixe de prótons de energia $E$ atinge um próton em
        repouso: mostre que
        $M^2c^4 = 2m_{\text{p}}^2c^4 + 2E\,m_{\text{p}}c^2$.
  \item Verifique os dois limites: a baixa energia, $Mc^2 \to
        2m_{\text{p}}c^2$; a alta energia, $Mc^2 \approx
        \sqrt{2E\,m_{\text{p}}c^2}$ --- a lei da raiz quadrada.
  \item Uma reação só é permitida se $Mc^2$ alcançar a soma das
        massas de repouso dos produtos, com todos os produtos em
        repouso no referencial do centro de momento no limiar:
        justifique esta última cláusula.
  \item Por que a energia cinética dos produtos no laboratório jamais
        pode ser recuperada para a criação de partículas num alvo
        fixo?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O preço do antipróton.}
\begin{enumerate}[resume]
  \item Em colisões $p + p$, a reação mais barata que produz um
        antipróton $\bar p$ tem de produzir também um próton
        \emph{extra}: escreva-a e justifique com a carga e o número
        bariônico.
  \item Mostre que o limiar exige $Mc^2 = 4m_{\text{p}}c^2$.
  \item Deduza a energia de feixe necessária $E = 7m_{\text{p}}c^2$ e
        a energia cinética $T = 6m_{\text{p}}c^2$; avalie $T$.
  \item No limiar, que fração da energia cinética do feixe de fato se
        tornou massa de repouso nova (dela, $2m_{\text{p}}c^2$)? Onde
        está o resto?
  \item Num colisor frontal $p$--$p$, que energia cinética \emph{por
        feixe} a mesma reação exigiria? Compare os dois projetos.
  \item Em 1954 não existia colisor algum (os feixes eram finos
        demais para se acertarem): que fato prático impôs a escolha do
        alvo fixo, e quanto isso custou em energia de feixe?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Projetar o Bevatron.}
A máquina construída em Berkeley alcançou $T = \qty{6.2}{GeV}$ ---
confortavelmente acima do limiar --- com campos dipolares de $B =
\qty{1.56}{T}$.
\begin{enumerate}[resume]
  \item Calcule a energia total e o momento do feixe na energia
        máxima.
  \item Calcule o raio de curvatura $r = p/qB$ e compare com os
        \qty{15.2}{m} da máquina.
  \item Calcule a velocidade do próton e a frequência de revolução no
        anel de $2\pi r$.
  \item Na injeção, os prótons chegam com \qty{10}{MeV} de energia
        cinética: calcule sua velocidade e explique por que a
        frequência de aceleração tinha de \emph{varrer} durante cada
        ciclo (que duas grandezas mudam quando $E$ cresce?).
  \item Com cerca de \qty{1.5}{kV} ganhos por volta, quantas voltas e
        quanto tempo dura um ciclo de aceleração?
  \item O nome ``Bevatron'' veio de BeV, bilhões de elétron-volts:
        diga numa linha que física fixou a cifra de projeto de $6$ e
        pouco BeV.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- A descoberta, 1955.}
\begin{enumerate}[resume]
  \item Um ímã curva cada candidato num círculo: explique por que isso
        seleciona as partículas por \emph{momento} ($r = p/qB$), e não
        por energia ou velocidade --- e por que um feixe selecionado
        em momento ainda mistura espécies.
  \item O feixe sobre um alvo de cobre produzia torrentes de píons
        negativos ($m_\pi c^2 = \qty{139.6}{MeV}$), entre os quais se
        escondiam alguns antiprótons. O espectrômetro selecionava
        negativos de momento $p = \qty{1.19}{GeV}/c$: calcule a
        velocidade de um antipróton e a de um píon nesse momento.
  \item Ao longo do percurso de voo de \qty{12}{m}, calcule os dois
        tempos de voo: de que resolução temporal a descoberta
        precisava?
  \item Uma segunda assinatura, independente, usava contadores
        Cherenkov, que disparam apenas acima de um limiar de
        velocidade: qual partícula foi arranjada para dispará-los, e
        por que a redundância importa para uma descoberta?
  \item Chamberlain e Segrè contaram cerca de um antipróton para cada
        $\num{44000}$ píons: por que o argumento do limiar garantia
        que a taxa seria minúscula mesmo acima do limiar?
  \item Um antipróton acaba por encontrar um próton e se aniquila:
        quanta energia é liberada por evento, e tipicamente sob que
        forma?
  \item Resuma o resultado central: a conservação do \omterm{def:b3:relativistic-dynamics:four-momentum}{quadrimomento} pôs
        no antipróton o preço de $T = 6m_{\text{p}}c^2 =
        \qty{5.6}{GeV}$ num alvo fixo; um síncrotron de \qty{6.2}{GeV}
        e \qty{15}{m} o pagou, e o antimundo tornou-se fato de
        laboratório --- o Prêmio Nobel de 1959.
\end{enumerate}
\end{problem}
