\chapter{Mecânica hamiltoniana}\label{ch:b3:hamiltonian-mechanics}

Fotografe um pêndulo mil vezes e marque, para cada foto, seu ângulo
contra seu momento angular: os pontos caem sobre uma curva fechada, e
todo movimento possível do pêndulo é uma curva dessas --- \omterm{prop:b3:lagrangian-mechanics:modes}{pequenas
oscilações} em ovais encaixados, voltas completas em linhas onduladas
acima e abaixo e, entre elas, uma única curva cruzada que separa os
dois regimes. Essa imagem, o \emph{\omterm{def:b3:hamiltonian-mechanics:portrait}{retrato de fases}}, é o coração da
reformulação que Hamilton deu à mecânica em 1833: o estado de um
sistema é um ponto no espaço das coordenadas \emph{e} dos momentos, sua
evolução é um escoamento nesse espaço, e o escoamento é gerado por uma
única função, a energia, por meio de duas equações de primeira ordem
belamente simétricas. A recompensa não são cálculos mais fáceis ---
nisso Lagrange costuma vencer --- mas a \emph{geometria} certa: o
escoamento conserva o volume do \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}, o que fundará a física
estatística; e seu esqueleto algébrico, o \omterm{def:b3:hamiltonian-mechanics:poisson}{colchete de Poisson}, é
exatamente o que a mecânica quântica promoverá a comutador. Este
capítulo é a dobradiça entre a mecânica das coisas e a física do
século XX.

\section{De Lagrange a Hamilton}

\begin{definition}[Hamiltoniana e equações canônicas]\label{def:b3:hamiltonian-mechanics:hamiltonian}
\index{hamiltoniana}\index{equações canônicas}\index{espaço de fases}
Para um sistema de \omterm{def:b3:lagrangian-mechanics:action}{lagrangiana} $L(q, \dot q, t)$, exprima as
velocidades em função dos momentos $p_i = \partial L/\partial\dot q_i$
e defina a \emph{hamiltoniana} como a transformada de Legendre
\[
H(q, p, t) = \sum_i p_i\dot q_i - L ,
\]
uma função das coordenadas e dos \emph{momentos}. O espaço de dimensão
$2n$ dos $(q_1, \dots, q_n, p_1, \dots, p_n)$ é o
\emph{espaço de fases}; um só ponto dele --- um \emph{estado} ---
determina todo o futuro e todo o passado do sistema pelas
\emph{equações canônicas} da
\cref{thm:b3:hamiltonian-mechanics:canonical}.
\end{definition}

\begin{theorem}[Equações de Hamilton]\label{thm:b3:hamiltonian-mechanics:canonical}
As \omterm{thm:b3:lagrangian-mechanics:euler-lagrange}{equações de Euler--Lagrange} são equivalentes às $2n$ equações de
primeira ordem
\[
\dot q_i = \frac{\partial H}{\partial p_i} , \qquad
\dot p_i = -\frac{\partial H}{\partial q_i} .
\]
\end{theorem}

\begin{proof}
Derive $H = \sum p_i\dot q_i - L$ como função de $(q, p, t)$:
$\dd H = \sum(\dot q_i\,\dd p_i + p_i\,\dd\dot q_i) - \sum(
\partial_{q_i}L\,\dd q_i + \partial_{\dot q_i}L\,\dd\dot q_i) -
\partial_tL\,\dd t$. Os termos em $\dd\dot q_i$ se cancelam pela
definição de $p_i$ --- o objetivo mesmo da transformada de Legendre ---
restando $\dd H = \sum(\dot q_i\,\dd p_i - \partial_{q_i}L\,\dd q_i) -
\partial_tL\,\dd t$. Lendo as derivadas parciais:
$\partial H/\partial p_i = \dot q_i$ e $\partial H/\partial q_i =
-\partial L/\partial q_i$, que por Euler--Lagrange vale $-\dot p_i$.
(Também $\partial_tH = -\partial_tL$.)
\end{proof}

\begin{proposition}[O que é $H$]\label{prop:b3:hamiltonian-mechanics:energy}
$H$ coincide com a \omterm{prop:b3:lagrangian-mechanics:energy}{função energia} $h$ do capítulo anterior: ao longo de
um movimento, $\dd H/\dd t = \partial H/\partial t$, de modo que $H$ se
conserva sempre que não depende explicitamente do tempo; e quando a
energia cinética é quadrática nas velocidades com \omterm{def:b3:lagrangian-mechanics:coordinates}{vínculos}
independentes do tempo, $H = E_k + E_p$, a energia mecânica --- agora
escrita nas variáveis $(q, p)$.
\end{proposition}

\begin{proof}
$\dd H/\dd t = \sum(\partial_{q_i}H\,\dot q_i + \partial_{p_i}H\,\dot
p_i) + \partial_tH = \sum(\partial_{q_i}H\,\partial_{p_i}H -
\partial_{p_i}H\,\partial_{q_i}H) + \partial_tH = \partial_tH$: a
simetria das \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas} faz a soma cancelar-se identicamente.
A identificação com $E_k + E_p$ é a
\cref{prop:b3:lagrangian-mechanics:energy}.
\end{proof}

\begin{example}[Duas hamiltonianas]\label{ex:b3:hamiltonian-mechanics:examples}
Massa numa mola: $p = m\dot x$, logo
\[
H = \frac{p^2}{2m} + \frac12 kx^2 ;
\]
as \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas} $\dot x = p/m$, $\dot p = -kx$ são o par
familiar. Pêndulo: $p = m\ell^2\dot\theta$ e
\[
H = \frac{p^2}{2m\ell^2} - mg\ell\cos\theta .
\]
Nos dois casos $H$ é a energia, constante em cada movimento: os
movimentos \emph{são} as curvas de nível de $H$ no plano $(q,p)$.
\end{example}

\begin{method}[A receita hamiltoniana]\label{met:b3:hamiltonian-mechanics:recipe}
(1) A partir de $L$, calcule os momentos $p_i = \partial L/\partial\dot q_i$
e inverta para obter os $\dot q_i$. (2) $H = \sum p_i\dot q_i - L$,
expressa apenas em $(q, p)$ --- para um sistema natural, simplesmente
$E_k + E_p$ com $E_k$ reescrita nos momentos. (3) Escreva as \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações
canônicas}. (4) Trace as curvas de nível de $H$: para um grau de
liberdade elas são as trajetórias, e todo o movimento qualitativo ---
oscilações, rotações, equilíbrios, \omterm{def:b3:hamiltonian-mechanics:portrait}{separatrizes} --- se lê sem resolver
nada.
\end{method}

\section{O espaço de fases}

\begin{definition}[Retrato de fases, pontos fixos, separatriz]\label{def:b3:hamiltonian-mechanics:portrait}
\index{retrato de fases}\index{ponto fixo}\index{separatriz}
O \emph{retrato de fases} de um sistema é a família de suas
trajetórias no \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}. Um \emph{ponto fixo} é um estado em que
as duas \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas} se anulam --- um equilíbrio: um mínimo do
potencial aparece como um \emph{centro} cercado de curvas fechadas, e
um máximo como uma \emph{sela} pela qual passa uma \emph{separatriz}, a
trajetória que divide o \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases} em regiões de movimento
qualitativamente diferente.
\end{definition}

\begin{example}[O retrato de fases do pêndulo]\label{ex:b3:hamiltonian-mechanics:pendulum-portrait}
Para $H = p^2/2m\ell^2 - mg\ell\cos\theta$: ovais fechados em torno de
$(0, 0)$ --- as \emph{librações}, as oscilações comuns; curvas
onduladas com $|p|$ grande --- as \emph{rotações}, o pêndulo dando
voltas por cima; e, pelas selas em $\theta = \pm\pi$, a \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz}, de
energia $E = +mg\ell$, sobre a qual o pêndulo leva um tempo infinito
para chegar ao topo. Um estado sobre a \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz} é a oscilação lançada
\emph{exatamente} com o vigor necessário para chegar à posição
invertida sem nada de sobra.
\end{example}

\begin{theorem}[Teorema de Liouville]\label{thm:b3:hamiltonian-mechanics:liouville}
\index{teorema de Liouville}
O escoamento hamiltoniano preserva o volume do \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}: se uma
região de estados iniciais é arrastada pelas \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas}, seu
volume (sua área, para um grau de liberdade) nunca muda --- por mais
que sua forma se deforme.
\end{theorem}

\begin{proof}
O escoamento no \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases} tem campo de velocidades $(\dot q_i, \dot p_i) =
(\partial_{p_i}H, -\partial_{q_i}H)$. Sua divergência é
\[
\sum_i\Big(\frac{\partial\dot q_i}{\partial q_i}
 + \frac{\partial\dot p_i}{\partial p_i}\Big)
 = \sum_i\Big(\frac{\partial^2H}{\partial q_i\partial p_i}
 - \frac{\partial^2H}{\partial p_i\partial q_i}\Big) = 0 :
\]
o escoamento é incompressível, e um escoamento incompressível
transporta volumes sem alterá-los, como para os fluidos do volume do
segundo ano. (A passagem formal da divergência nula ao volume
conservado é o teorema de transporte demonstrado lá.)
\end{proof}

\begin{remark}[Por que Liouville importa]\label{rem:b3:hamiltonian-mechanics:liouville-matters}
Nada na formulação de Newton sugere que algo seja incompressível. Na
imagem \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} isso é automático --- e é a licença para a física
estatística: quando descrevemos um gás de $10^{23}$ moléculas por uma
nuvem de pontos no \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}, o teorema de Liouville diz que a
nuvem escoa como um fluido incompressível, de modo que o ``número de
estados num volume do \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}'' é uma grandeza que a própria
dinâmica não pode criar nem destruir. O postulado microcanônico da
física estatística, mais adiante neste volume, apoia-se exatamente
nisso.
\end{remark}

\section{Colchetes de Poisson}

\begin{definition}[Colchete de Poisson]\label{def:b3:hamiltonian-mechanics:poisson}
\index{colchete de Poisson}
O \emph{colchete de Poisson} de duas funções $f(q, p, t)$ e
$g(q, p, t)$ no \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases} é
\[
\{f, g\} = \sum_i\Big(
 \frac{\partial f}{\partial q_i}\frac{\partial g}{\partial p_i}
 - \frac{\partial f}{\partial p_i}\frac{\partial g}{\partial q_i}\Big) .
\]
Ele é antissimétrico, linear em cada argumento, obedece à regra do
produto $\{f, gh\} = \{f, g\}h + g\{f, h\}$ e satisfaz, para as
próprias coordenadas, as \emph{relações canônicas}
\[
\{q_i, p_j\} = \delta_{ij} , \qquad
\{q_i, q_j\} = \{p_i, p_j\} = 0 .
\]
\end{definition}

\begin{theorem}[A evolução como colchete]\label{thm:b3:hamiltonian-mechanics:evolution}
Ao longo de qualquer movimento,
\[
\frac{\dd f}{\dd t} = \{f, H\} + \frac{\partial f}{\partial t} .
\]
Em particular, uma grandeza sem dependência explícita do tempo se
conserva se e somente se seu colchete com a \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} se anula; e as
próprias \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas} são $\dot q_i = \{q_i, H\}$, $\dot p_i
= \{p_i, H\}$: a \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} \emph{gera} a evolução temporal.
\end{theorem}

\begin{proof}
Regra da cadeia mais as \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas}: $\dd f/\dd t = \sum(
\partial_{q_i}f\,\dot q_i + \partial_{p_i}f\,\dot p_i) + \partial_tf =
\sum(\partial_{q_i}f\,\partial_{p_i}H - \partial_{p_i}f\,
\partial_{q_i}H) + \partial_tf$.
\end{proof}

\begin{example}[Colchetes do momento angular]\label{ex:b3:hamiltonian-mechanics:brackets}
Para uma partícula, $L_z = xp_y - yp_x$ e suas companheiras cíclicas.
Um cálculo direto a partir das relações canônicas dá
\[
\{L_x, L_y\} = L_z , \qquad
\{L_y, L_z\} = L_x , \qquad
\{L_z, L_x\} = L_y ,
\]
e $\{L^2, L_z\} = 0$: as componentes do momento angular não
``comutam'' entre si, mas cada uma comuta com o quadrado total. Guarde
a forma dessas relações --- elas voltarão, literalmente, como
comutadores na mecânica quântica, onde ditam tudo a respeito da
estrutura atômica.
\end{example}

\begin{remark}[A porta de entrada da mecânica quântica]\label{rem:b3:hamiltonian-mechanics:doorway}
Dirac observou em 1925 que toda a mecânica quântica se obtém mantendo a
álgebra da mecânica \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} e substituindo o \omterm{def:b3:hamiltonian-mechanics:poisson}{colchete de Poisson}
pelo comutador de operadores dividido por
$\iu\hbar$: $\{q, p\} = 1$ torna-se $[\hat x, \hat p] = \iu\hbar$, a
conservação continua a ser ``o colchete com $H$ se anula'' e a evolução
temporal continua a ser gerada pela \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana}. A teoria clássica
carrega, em seus ossos, o esqueleto da quântica; os capítulos sobre
mecânica quântica tornarão essa correspondência explícita.
\end{remark}

\section{Ação e invariantes adiabáticos}

\begin{definition}[Variável de ação]\label{def:b3:hamiltonian-mechanics:action-variable}
\index{variável de ação}
Para um sistema de um grau de liberdade que oscila sobre uma trajetória
de fase fechada, a \emph{variável de \omterm{def:b3:lagrangian-mechanics:action}{ação}} é a área delimitada dividida
por $2\pi$:
\[
I = \frac{1}{2\pi}\oint p\,\dd q .
\]
\end{definition}

\begin{proposition}[Ação do oscilador harmônico]\label{prop:b3:hamiltonian-mechanics:sho-action}
Para $H = p^2/2m + \tfrac12 m\omega^2q^2$ à energia $E$, a trajetória é
uma elipse de semieixos $\sqrt{2E/m\omega^2}$ e $\sqrt{2mE}$, que
delimita a área $2\pi E/\omega$: portanto
\[
I = \frac{E}{\omega} , \qquad E = \omega I ,
\]
e a frequência de oscilação é $\partial E/\partial I = \omega$, como
deve ser.
\end{proposition}

\begin{proof}
Área de uma elipse, $\pi ab = \pi\sqrt{2E/m\omega^2}\sqrt{2mE} =
2\pi E/\omega$.
\end{proof}

\begin{proposition}[Invariância adiabática]\label{prop:b3:hamiltonian-mechanics:adiabatic}
\index{invariante adiabático}
Se um parâmetro do sistema (um comprimento, uma constante elástica, um
campo) varia \emph{lentamente} --- ao longo de muitos períodos de
oscilação --- a energia muda, a frequência muda, mas a \omterm{def:b3:lagrangian-mechanics:action}{ação} $I$
permanece constante com excelente aproximação: $I$ é um
\emph{\omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invariante adiabático}}. Para o oscilador lentamente modificado,
$E/\omega$ se conserva: enrijeça a mola devagar até dobrar $\omega$ e a
energia dobra junto.
\end{proposition}

\begin{proof}[Demonstração parcial]
Para o oscilador com $\omega(t)$ lentamente variável: ao longo de um
período, o trabalho realizado pelo parâmetro em mudança pode ser
calculado por média; o cálculo (guiado no
\cref{exo:b3:hamiltonian-mechanics:11}) dá $\dot E/E = \dot\omega/\omega$,
isto é, $\dd(E/\omega)/\dd t = 0$ em ordem dominante. O enunciado
geral, para qualquer sistema oscilante lentamente deformado, é
admitido --- é a razão pela qual as órbitas planetárias sobrevivem a
perturbações lentas, e o ponto de partida da ``velha teoria quântica''
adiante.
\end{proof}

dobrada, a elipse de fase muda de forma a área constante, e por isso
$E = \omega I$ dobra. À direita: içar o fio de um pêndulo que oscila
injeta energia na oscilação exatamente na proporção que mantém
$I = E/\omega$ fixo.}
\end{omfigure}

\begin{remark}[A velha teoria quântica]\label{rem:b3:hamiltonian-mechanics:old-quantum}
Por que os átomos têm energias discretas? A primeira resposta
quantitativa (Bohr, 1913; Sommerfeld, 1915) foi escrita na linguagem
deste capítulo: entre todos os movimentos clássicos, a natureza guarda
aqueles cuja integral de \omterm{def:b3:lagrangian-mechanics:action}{ação} vale um número inteiro de constantes de
Planck,
\[
\oint p\,\dd q = nh .
\]
Aplicada ao oscilador harmônico, ela dá $E_n = n\hbar\omega$ (faltando
apenas a metade do verdadeiro $\big(n + \tfrac12\big)\hbar
\omega$); aplicada a uma partícula numa caixa, dá \emph{exatamente} os
níveis do volume do segundo ano; aplicada ao átomo de hidrogênio
(\cref{pb:b3:hamiltonian-mechanics:1}), dá o espectro medido com quatro
algarismos. A regra era quantitativamente certa e conceitualmente
provisória --- e, como a \omterm{def:b3:lagrangian-mechanics:action}{ação} é um \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invariante adiabático}, o número
quântico $n$ não muda sob perturbações lentas, e é por isso que uma
regra assim podia funcionar. A verdadeira teoria começa cinco capítulos
adiante.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:1}
Para a massa na mola: (a) construa $H$ a partir de $L$ e verifique $H =
E_k + E_p$; (b) escreva as \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas} e verifique que
reproduzem $\ddot x = -\omega^2x$; (c) mostre que as trajetórias são
elipses no plano $(x, p)$ e dê seus semieixos à energia $E$; (d) em que
sentido o ponto representativo se move no sentido horário?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:2}
A conta no aro que gira do capítulo anterior tem $L =
\tfrac12 mR^2\dot\theta^2 + \tfrac12 m\omega^2R^2\sin^2\theta +
mgR\cos\theta$. (a) Calcule $p$ e $H$. (b) $H$ se conserva? É a
energia mecânica? (c) Esboce as curvas de nível de $H$ para
$\omega^2 < g/R$ e $\omega^2 > g/R$, usando a \omterm{ex:b3:lagrangian-mechanics:hoop}{energia potencial efetiva}
daquele capítulo. (d) Identifique os centros, as selas e as
\omterm{def:b3:hamiltonian-mechanics:portrait}{separatrizes} no caso rápido.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:3}
Uma bola quica elasticamente no chão, $H = p^2/2m + mgz$ ($z >
0$). (a) Desenhe a trajetória de fase de um voo e a do movimento
completo de quiques. (b) O que o quique elástico faz ao ponto
representativo? (c) Calcule a área delimitada para uma altura máxima
$z_{\max}$ e a \omterm{def:b3:lagrangian-mechanics:action}{ação} $I$. (d) A regra de Bohr--Sommerfeld
$\oint p\,\dd z = nh$ aplicada a um nêutron que quica ($m =
\qty{1.67e-27}{kg}$) dá alturas de quique quantizadas: estime a mais
baixa e compare com os \qty{15}{\micro m} medidos no experimento de
estados quânticos gravitacionais de 2002.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:4}
Só a partir das relações canônicas, calcule (a) $\{x^2, p\}$; (b)
$\{xp, H\}$ para $H = p^2/2m + E_p(x)$, e interprete os dois termos
(esse colchete conduz ao \emph{teorema do virial}); (c) $\{L_z, x\}$ e
$\{L_z, p_x\}$; (d) mostre que, se $\{f, H\} = 0$ e $\{g, H\} = 0$,
então $\{f, g\}$ também se conserva (use a identidade de Jacobi,
admitida: $\{f,\{g,h\}\} + \{g,\{h,f\}\} + \{h,\{f,g\}\} = 0$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:5}
O pêndulo perto de sua \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz}. (a) Dê a energia da \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz} e o
$|p|$ máximo sobre ela. (b) Mostre que sobre a \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz} $p =
\pm 2m\ell^2\omega_0\cos(\theta/2)$ com $\omega_0 = \sqrt{g/\ell}$.
(c) Usando $\dot\theta = p/m\ell^2$, mostre que o tempo para ir de
$\theta$ até o topo diverge logaritmicamente. (d) Um pêndulo real
solto logo abaixo da \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz} fica um longo instante perto da posição
invertida antes de voltar --- relacione isso a (c) e à lentidão
observada perto de toda sela.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:6}
Uma partícula carregada num campo magnético tem $H = (\vect p -
q\vect A)^2/2m$, com $\vect p$ o momento canônico do capítulo
anterior. (a) Escreva as \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas} para $\vect A = \tfrac12
B(-y, x, 0)$ e verifique que dão o movimento ciclotrônico. (b) Mostre que $H =
\tfrac12 m\vect v^{\,2}$: o campo magnético não realiza trabalho. (c)
Calcule o colchete das duas grandezas conservadas $\pi_x = p_x - \tfrac12
qBy$ e $\pi_y = p_y + \tfrac12 qBx$ (verifique antes que cada uma se
conserva) e mostre que $\{\pi_x, \pi_y\} = -qB$: duas grandezas
conservadas cujo colchete é uma constante. (d) Use o
\cref{exo:b3:hamiltonian-mechanics:4}(d) para explicar por que não se
devia esperar delas uma terceira grandeza conservada independente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:7}
Liouville, na mão. (a) Para a partícula livre, o escoamento é $q \to q +
pt/m$, $p \to p$: mostre que um retângulo se torna um paralelogramo de
mesma área. (b) Para o oscilador harmônico, mostre que o escoamento é
uma rotação (em unidades adequadas) e conclua. (c) Para o oscilador
\emph{amortecido} $\ddot x = -\omega^2x - \gamma\dot x$, escreva a
divergência do escoamento no plano $(x, v)$ e mostre que as áreas
encolhem como $\eu^{-\gamma t}$: o amortecimento não é hamiltoniano.
(d) Para onde ``vai'' fisicamente a área perdida?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:8}
Movimento plano num potencial central, $H = (p_r^2 + p_\varphi^2/r^2)/2m
+ E_p(r)$. (a) Escreva as quatro \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas}. (b) Mostre que $\{
p_\varphi, H\} = 0$ e identifique a lei de conservação. (c) Reduza a
uma \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} radial unidimensional com um potencial efetivo. (d)
Para $E_p = -k/r$, localize a órbita circular a $p_\varphi$ dado e dê
sua energia --- a ser quantizada no
\cref{pb:b3:hamiltonian-mechanics:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:9}
(a) Verifique $\{L_x, L_y\} = L_z$ a partir das relações canônicas. (b)
Deduza os outros dois colchetes por permutação cíclica. (c) Mostre que
$\{L^2, L_z\} = 0$. (d) Um corpo rígido gira livremente: tomando $H =
L^2/2J$ (inércia de simetria esférica), mostre que os três $L_i$ se
conservam; e para um corpo com momentos de inércia desiguais (responda
qualitativamente a partir dos colchetes)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:10}
Níveis da velha teoria quântica. (a) Mostre que $\oint p\,\dd q = nh$
aplicada ao oscilador harmônico dá $E_n = n\hbar\omega$, usando a
\cref{prop:b3:hamiltonian-mechanics:sho-action}. (b) Aplique-a à
partícula numa caixa de comprimento $a$ e recupere exatamente $E_n =
n^2h^2/8ma^2$. (c) Para uma molécula diatômica modelada como oscilador
de constante elástica $k = \qty{1.9e3}{N/m}$ e massa reduzida $\qty{1.14e-26}
{kg}$ (monóxido de carbono), calcule $\hbar\omega$ em eV e o
comprimento de onda da emissão $n = 1 \to 0$; em que faixa espectral
ela cai? (d) Os níveis verdadeiros são $(n + \tfrac12)\hbar\omega$: a
metade que falta muda os comprimentos de onda \emph{emitidos}? Que
experimento detecta essa metade de ponto zero?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:11}
Invariância adiabática de $E/\omega$, deduzida. Uma massa numa mola
cuja constante elástica $k(t)$ cresce lentamente. (a) Mostre que $\dot E = \tfrac12\dot k\,
x^2$ ao longo do movimento exato. (b) Faça a média sobre um período a
$k$ fixo: usando $\langle\tfrac12 kx^2\rangle = E/2$, mostre que $\langle\dot E\rangle
= \dot k\,E/2k$. (c) Conclua que $\dd\ln E = \tfrac12\dd\ln k = \dd\ln
\omega$, e portanto que $E/\omega$ é constante. (d) Um pêndulo que
oscila com amplitude $\theta_0 = 5^\circ$ tem o fio lentamente
encurtado de \qty{1.0}{m} a \qty{0.5}{m}: encontre a nova amplitude e o
fator pelo qual sua energia cresceu, e diga de onde veio essa energia.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:12}
A área dentro da \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz} do pêndulo é um número de estados
quânticos. (a) Mostre que $\oint p\,\dd\theta$ em torno da \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz}
vale $16m\ell^2\omega_0$. (b) Pela regra de Bohr--Sommerfeld, o número
de estados quânticos com energia abaixo da \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriz} é $N \approx
16m\ell^2\omega_0/h$: avalie-o para uma massa de um grama num fio de
\qty{10}{cm}. (c) Avalie-o para um oscilador molecular do tipo amônia:
$m \sim \qty{1e-26}{kg}$, $\ell \sim \qty{1e-10}{m}$,
$\omega_0 \sim \qty{1e14}{rad/s}$. (d) Conclua: onde está a fronteira
entre a mecânica que precisa de $\hbar$ e a que não precisa?
\end{exercise}

\section{Problema: A velha teoria quântica e o átomo de hidrogênio}

\begin{problem}[{Problema de fim de semana --- quantizar o espaço de
fases, pesar a constante de Rydberg}]\label{pb:b3:hamiltonian-mechanics:1}
Em 1913 Bohr calculou o espectro do hidrogênio a partir da mecânica
planetária mais uma regra quântica; Sommerfeld reconheceu na regra um
enunciado sobre a área no \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{espaço de fases}. Este problema refaz o
cálculo deles com as ferramentas deste capítulo. Dados: $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $1/4\pi\varepsilon_0 =
\qty{8.99e9}{N.m^2/C^2}$, $h = \qty{6.63e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$. Escreva $k = e^2/4\pi\varepsilon_0$.

\medskip\noindent\textbf{Parte I --- A \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} de Kepler.}
Um elétron se move no plano em torno de um próton fixo.
\begin{enumerate}
  \item Justifique tratar o próton como fixo (razão das massas) e
        escreva a \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} $H = \dfrac{p_r^2}{2m_{\text{e}}} +
        \dfrac{p_\varphi^2}{2m_{\text{e}}r^2} - \dfrac{k}{r}$.
  \item Escreva as quatro \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{equações canônicas}.
  \item Mostre que $p_\varphi$ se conserva e dê-lhe um nome.
  \item Reduza o movimento radial ao potencial efetivo
        $U_{\text{eff}}(r) = p_\varphi^2/2m_{\text{e}}r^2 - k/r$ e
        esboce-o.
  \item Localize a órbita circular: $r_{\text{c}} =
        p_\varphi^2/m_{\text{e}}k$.
  \item Mostre que sua energia vale $E = -k/2r_{\text{c}} =
        -m_{\text{e}}k^2/2p_\varphi^2$ e verifique que é metade da
        energia potencial (a razão do virial das órbitas
        gravitacionais do volume do primeiro ano).
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A órbita clássica.}
\begin{enumerate}[resume]
  \item Para uma órbita circular de raio $r$, encontre a velocidade $v$
        e a frequência orbital $f_{\text{orb}}$ em função de $r$.
  \item Uma órbita de tamanho atômico, $r = \qty{0.05}{nm}$: calcule
        $v$ (e $v/c$), $f_{\text{orb}}$ e a energia em eV.
  \item Calcule o momento angular $p_\varphi$ dessa órbita e compare-o
        com $\hbar = h/2\pi$: o que essa proximidade sugere?
  \item Classicamente, um elétron em órbita é um dipolo oscilante e
        irradia (volume do segundo ano) em $f_{\text{orb}}$; sua
        energia decai em cerca de \qty{1e-11}{s}. Enuncie as duas
        previsões fatais que isso acarreta para os átomos, e o que se
        observa em lugar delas.
  \item Que característica dos espectros observados (linhas discretas,
        regra de combinação $1/\lambda = R_H(1/n^2 - 1/n'^2)$) sugeriu
        \emph{níveis} de energia discretos?
  \item Explique por que um \emph{\omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invariante adiabático}} é o candidato
        natural a grandeza que assume valores universais fixos
        (lembre-se da \cref{prop:b3:hamiltonian-mechanics:adiabatic}).
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A quantização.}
Imponha a regra de Sommerfeld ao movimento angular da órbita circular:
$\oint p_\varphi\,\dd\varphi = nh$, $n = 1, 2, 3, \dots$
\begin{enumerate}[resume]
  \item Mostre que a regra se lê $p_\varphi = n\hbar$.
  \item Deduza os raios permitidos $r_n = n^2a_0$ com $a_0 =
        \hbar^2/m_{\text{e}}k$, e calcule $a_0$.
  \item Deduza as energias permitidas $E_n = -E_{\text{I}}/n^2$ com
        $E_{\text{I}} = m_{\text{e}}k^2/2\hbar^2$, e calcule
        $E_{\text{I}}$ em joules e em eV.
  \item Calcule a velocidade na primeira órbita e verifique que $v_1/c =
        k/\hbar c \approx 1/137$ (a constante de estrutura fina).
  \item Um fóton carrega a energia de uma transição: deduza a fórmula
        de Rydberg e o valor de $R_H = E_{\text{I}}/hc$; compare com o
        valor medido \qty{1.097e7}{m^{-1}}.
  \item Calcule os comprimentos de onda das transições $2 \to 1$,
        $3 \to 2$ e $\infty \to 2$; qual delas é visível, e de que cor?
  \item O íon de hélio ionizado He$^+$ é um hidrogênio de carga nuclear
        $2e$: como se escalam $a_0$ e $E_{\text{I}}$, e onde cai a sua
        linha $2 \to 1$?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- O confronto.}
\begin{enumerate}[resume]
  \item Calcule a frequência orbital $f_{\text{orb}}(n)$ e a
        frequência de transição $\nu_{n \to n-1}$ para $n = 2$, $10$,
        $100$, e mostre que a razão entre elas tende a $1$ quando $n$
        cresce.
  \item Este é o \emph{princípio da correspondência} de Bohr:
        enuncie-o numa frase.
  \item A regra $p_\varphi = n\hbar$ começa em $n = 1$: que absurdo
        significaria $n = 0$ para uma órbita circular?
  \item A mecânica quântica manterá $E_n = -E_{\text{I}}/n^2$
        exatamente, mas descartará as órbitas: nomeie dois fatos
        mensuráveis que a imagem orbital erra (o valor do momento
        angular do estado fundamental; a existência de estados com o
        mesmo $n$ e formas diferentes).
  \item O múon é um elétron $207$ vezes mais pesado: para o hidrogênio
        muônico, calcule $a_0^\mu$ e $E_{\text{I}}^\mu$ e explique por
        que os átomos muônicos sondam o \emph{núcleo}.
  \item Resuma o resultado central: um \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invariante adiabático}, igualado
        a números inteiros de $h$, fornece $a_0 = \qty{52.9}{pm}$,
        $E_{\text{I}} = \qty{13.6}{eV}$ e o espectro do hidrogênio com
        quatro algarismos --- e entrega à teoria verdadeira suas duas
        constantes centrais.
\end{enumerate}
\end{problem}
