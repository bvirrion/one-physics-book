\chapter{Estatística quântica}\label{ch:b3:quantum-statistics}

Os elétrons de condução do cobre formam um gás mil vezes mais denso que
o ar --- e ainda assim quase nada acrescentam à capacidade térmica do
cobre, escândalo que assombrou a física clássica por cinquenta anos. Uma
nuvem de átomos de rubídio, resfriada a cem bilionésimos de kelvin,
desaba de repente num único estado quântico com dez mil átomos
de profundidade. Uma estrela morta com a massa do Sol e o tamanho da
Terra se recusa a encolher mais, sustentada por nada além do princípio
de exclusão. Os três casos são o mesmo capítulo da física: o que
acontece com um gás quando $n\lambda_T^3 \sim 1$ e as fórmulas de
ocupação do \cref{ch:b3:grand-canonical} se afastam de Boltzmann. Os
férmions constroem mares --- frios, rígidos, pressurizados; os bósons
constroem condensados --- gregários, coerentes, superfluidos. Entre a
maioria silenciosa do mar de Fermi e a debandada do condensado está a
maior parte da física moderna da matéria condensada e das baixas
temperaturas, e o destino das estrelas.

\section{Gases de ocupações}

\begin{proposition}[A maquinaria do gás quântico]\label{prop:b3:quantum-statistics:machinery}
\index{densidade de estados}
Para partículas sem interação numa caixa, as somas sobre níveis viram
integrais sobre a \omterm{prop:b3:schrodinger-three-dimensions:counting}{densidade de estados} da
\cref{prop:b3:schrodinger-three-dimensions:counting} (por estado de
spin, $g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$):
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
com $\bar n$ a ocupação de Fermi--Dirac ou de Bose--Einstein. A
primeira equação fixa $\mu(T, n)$; a segunda entrega a
termodinâmica. No limite diluído $n\lambda_T^3 \ll 1$, as duas
estatísticas desabam sobre Boltzmann e o gás clássico volta;
os regimes quânticos começam onde os pacotes de onda se tocam.
\end{proposition}
\admitted

\section{O mar de Fermi}

\begin{theorem}[O gás de Fermi degenerado]\label{thm:b3:quantum-statistics:fermi}
\index{energia de Fermi}\index{pressão de degenerescência}
Em $T = 0$, os férmions preenchem todos os níveis até a \emph{\omterm{thm:b3:quantum-statistics:fermi}{energia
de Fermi}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{spin } \tfrac12) ,
\]
com energia média $\tfrac35 E_{\text{F}}$ por partícula e a
\emph{\omterm{thm:b3:quantum-statistics:fermi}{pressão de degenerescência}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
uma pressão no zero absoluto, de origem puramente quântica --- a
rigidez do gás de elétrons de todo metal e o sustento das estrelas
mortas. Em $0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$,
só os elétrons a menos de $\sim k_{\text{B}}T$ da superfície deste
\emph{mar de Fermi} podem responder a qualquer coisa: a fração
$\sim T/T_{\text{F}}$ é a chave-mestra do comportamento metálico.
\end{theorem}

\begin{proof}[Demonstração parcial]
Preencha $g(\epsilon)$ até $E_{\text{F}}$: inverter $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ dá o $E_{\text{F}}$ enunciado
(\cref{exo:b3:schrodinger-three-dimensions:8}); $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ sai de mais uma integral. Pressão:
comprimir a caixa eleva todos os níveis como $V^{-2/3}$, e portanto $E
\propto V^{-2/3}$ e $P = -\partial E/\partial V = \tfrac23 E/V$
--- com $E = \tfrac35 NE_{\text{F}}$, a lei enunciada.
\end{proof}

\begin{example}[O escândalo da capacidade térmica, resolvido]\label{ex:b3:quantum-statistics:heat}
\index{capacidade térmica eletrônica}
Classicamente, o elétron livre por átomo do cobre acrescentaria $\tfrac32
k_{\text{B}}$ à capacidade térmica --- $50\%$ acima de
Dulong--Petit. Medido: cerca de $1\%$ à temperatura ambiente. O mar
explica: com $T_{\text{F}} \approx \qty{8e4}{K}$, só a
fração $\sim T/T_{\text{F}} \approx 0.4\%$ dos elétrons pode ser
excitada termicamente, cada um por $\sim k_{\text{B}}T$:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(o coeficiente de Sommerfeld exato;
\cref{exo:b3:quantum-statistics:12}). Linear em $T$, ela fica enterrada
sob o $T^3$ da rede às temperaturas ordinárias, mas
\emph{domina} abaixo de alguns kelvins --- exatamente como acham os
calorímetros, resolvendo de um golpe o enigma dos elétrons que
carregam corrente, mas não capacidade térmica.
\end{example}

\begin{example}[Estrelas sustentadas pela exclusão]\label{ex:b3:quantum-statistics:dwarf}
\index{anã branca}
Uma \omterm{ex:b3:quantum-statistics:dwarf}{anã branca} (\cref{exo:b3:identical-particles:11}) equilibra a
gravidade contra a \omterm{thm:b3:quantum-statistics:fermi}{pressão de degenerescência} eletrônica. Como $P \propto
n^{5/3}$ enrijece mais depressa do que crescem as exigências da
gravidade quando a estrela encolhe, existe um raio de equilíbrio para
qualquer massa modesta --- com a escala inquietante $R \propto M^{-1/3}$:
\emph{as anãs mais pesadas são menores}. Espremidos mais, os elétrons
ficam relativísticos, a lei de pressão amolece para $n^{4/3}$, e o
equilíbrio falha numa massa única, o limite de Chandrasekhar
$M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) --- passado ele,
colapso para uma estrela de nêutrons: um mar de Fermi de nêutrons à
densidade nuclear, a mesma física a $10^{14}$ vezes a densidade.
\end{example}

\section{A debandada de Bose}

\begin{theorem}[Condensação de Bose--Einstein]\label{thm:b3:quantum-statistics:bec}
\index{condensação de Bose--Einstein}
Para um gás de $N$ bósons conservados, os estados excitados podem
guardar no máximo $N_{\max}(T) = 2.612\,V/\lambda_T^3$ partículas (a
integral de $g\bar n_{\text{BE}}$ no teto $\mu \to 0$ dela). Abaixo da
temperatura crítica
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(equivalentemente } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
o excedente não tem para onde ir senão o único estado fundamental: uma
população \emph{macroscópica}
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
se condensa num só estado quântico --- matéria comportando-se como uma
única função de onda gigante. Prevista por Einstein em 1925 a partir da
estatística de fótons de Bose; realizada em rubídio diluído em 1995 a
$T_{\text{c}} \approx \qty{100}{nK}$ (Nobel de 2001), anunciada pela
reveladora distribuição de velocidades \emph{bimodal}: uma nuvem térmica
com uma agulha de condensado de velocidade nula subindo do centro dela.
\end{theorem}
\admitted

\begin{example}[O hélio, superfluido veterano]\label{ex:b3:quantum-statistics:helium}
\index{superfluidez}
O $^4$He líquido (bosônico) se transforma em $T_\lambda = \qty{2.17}{K}$
num \emph{superfluido}: viscosidade nula através dos capilares mais
finos, escoamento em filme pelas paredes do recipiente, efeitos de fonte
e circulação quantizada em unidades de $h/m$ --- a fase macroscópica de
um condensado em ação, deslocada da previsão do gás ideal
($\qty{3.1}{K}$ para a densidade do hélio) pelas interações, mas
inconfundivelmente o mesmo fenômeno. O $^3$He fermiônico não consegue
condensar sozinho: os átomos dele têm primeiro de \emph{emparelhar}
(como os elétrons de um supercondutor, \cref{ch:b3:electrons-in-solids})
em bósons compostos --- e a superfluidez de fato chega, mas a
\qty{2.6}{mK}, mil vezes mais frio: dois isótopos, uma lição --- a
estatística é destino.
\end{example}

\begin{method}[Que estatística, que regime]\label{met:b3:quantum-statistics:method}
(1) Calcule $n\lambda_T^3$ primeiro: $\ll1$ significa Boltzmann, e o
capítulo 17 basta. (2) Férmions degenerados: pense em
$E_{\text{F}}$ e $T/T_{\text{F}}$; as respostas (capacidade térmica,
suscetibilidade, espalhamento) carregam o fator $T/T_{\text{F}}$ ou
amostram apenas a superfície de Fermi. (3) Bósons degenerados: compare $T$
com $T_{\text{c}}$; abaixo dele, separe o gás em condensado mais
nuvem térmica. (4) Pressões: a rigidez fermiônica $n^{5/3}$ (ou
$n^{4/3}$ relativística) contra o que quer que aperte. (5) Partículas
compostas: conte os férmions constituintes para escolher a estatística
--- e lembre que o emparelhamento pode converter uma estatística na outra.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:1}
Calcule $E_{\text{F}}$ e $T_{\text{F}}$ para (a) o cobre
($n = \qty{8.5e28}{m^{-3}}$); (b) o sódio ($\qty{2.5e28}{m^{-3}}$);
(c) o $^3$He líquido ($\qty{1.6e28}{m^{-3}}$, $m = 3u$); (d) ordene
contra a temperatura ambiente e comente quem é degenerado e quando.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:2}
(a) Mostre que $\langle\epsilon\rangle = \tfrac35 E_{\text{F}}$ em $T =
0$. (b) Calcule a velocidade quadrática média dos elétrons no cobre. (c)
Essa velocidade corresponde classicamente a que temperatura? (d) Por que
um metal frio contém elétrons movendo-se a um centésimo da velocidade da
luz sem irradiar essa energia embora?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:3}
\omterm{thm:b3:quantum-statistics:fermi}{Pressão de degenerescência} dos elétrons do cobre: (a) avalie $P =
\tfrac25 nE_{\text{F}}$ em pascals e em atmosferas. (b) Por que
o metal não explode (o que puxa de volta)? (c) Essa pressão resiste à
compressão: relacione com a baixa compressibilidade dos metais. (d)
Estime o módulo de compressão $K \sim \tfrac53 P$ e compare com os
\qty{140}{GPa} medidos do cobre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:4}
Limiares de BEC. (a) Calcule $T_{\text{c}}$ para o rubídio-87 ($m =
\qty{1.44e-25}{kg}$) a $n = \qty{e20}{m^{-3}}$. (b) Para o mesmo
$n$, que $T_{\text{c}}$ teriam \omterm{thm:b3:hydrogen-atom:levels}{átomos de hidrogênio}? (c) Por que os
experimentos usam gases diluídos a nanokelvins em vez de gases
densos a milikelvins (o que acontece quimicamente a alta densidade e
baixa temperatura)? (d) Verifique a regra de consistência $n\lambda_{T_c}^3
= 2.612$ nos seus números do rubídio.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:5}
Duas capacidades térmicas. Num metal, $C = \gamma T +
\beta T^3$ (elétrons; rede). (a) Usando o
\cref{ex:b3:quantum-statistics:heat} e o termo de rede à moda de Debye
$\beta T^3 = 234\,Nk_{\text{B}}(T/\theta_{\text{D}})^3$,
estime a temperatura de cruzamento para o cobre
($\theta_{\text{D}} = \qty{343}{K}$, $T_{\text{F}} =
\qty{8.2e4}{K}$). (b) A \qty{300}{K}, a parcela eletrônica em
porcentagem. (c) E a \qty{0.1}{K}? (d) Por que os experimentadores
traçam $C/T$ contra $T^2$, e o que entregam o intercepto e a
inclinação dessa reta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}
Uma \omterm{ex:b3:quantum-statistics:dwarf}{anã branca} de uma massa solar ($M = \qty{2e30}{kg}$) e raio
terrestre ($R = \qty{6.4e6}{m}$), com um elétron por dois núcleons. (a)
Calcule $n_{\text{e}}$ e $E_{\text{F}}$ (fórmula não
relativística). (b) Compare $E_{\text{F}}$ com $m_{\text{e}}c^2$ e
com $k_{\text{B}}T$ aos \qty{e7}{K} do interior: justifique
``frio'' e ``quase relativístico'' ao mesmo tempo. (c) Compare
a \omterm{thm:b3:quantum-statistics:fermi}{pressão de degenerescência} $\tfrac25 nE_{\text{F}}$ com a
estimativa de pressão gravitacional $\sim GM^2/R^4$: mesma ordem? (d) O que acontece qualitativamente com
cada lado do equilíbrio se $M$ cresce?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:7}
Matéria de nêutrons. Uma estrela de nêutrons empacota $M = 1.4\,M_\odot$ em $R
\approx \qty{11}{km}$. (a) Calcule a densidade de nêutrons e
compare com a densidade nuclear ($\qty{2.3e17}{kg/m^3}$). (b) Calcule
$E_{\text{F}}$ para os nêutrons (não relativístico; $m_{\text{n}}c^2
= \qty{939}{MeV}$) e verifique quão relativísticos eles são. (c) Por que
os elétrons da estrela desapareceram (decaimento beta inverso:
$\text{e} + \text{p} \to \text{n} + \nu$ --- o que fez o
$E_{\text{F}}$ dos elétrons pagar por isso)? (d) Uma colher de chá dessa
matéria pesa quanto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:8}
Bósons sem número. (a) Os fótons têm sempre $\mu = 0$: por que eles
não podem sofrer a condensação do
\cref{thm:b3:quantum-statistics:bec} (o que o gás faz em vez disso
quando resfriado)? (b) Que lei de conservação os átomos de rubídio têm
e falta aos fótons? (c) Em 2010 um BEC de fótons \emph{foi} feito, numa
cavidade cheia de corante onde os fótons termalizam e o número deles
fica efetivamente fixo: que ingrediente a cavidade de corante restituiu?
(d) Fônons num cristal: condensação ou não, e por quê?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
Os dois isótopos do hélio. (a) Calcule o $T_{\text{c}}$ de gás ideal
do $^4$He líquido ($n = \qty{2.2e28}{m^{-3}}$, $m = 4u$) e
compare com $T_\lambda = \qty{2.17}{K}$. (b) Por que a concordância é
apenas grosseira (o que ``ideal'' ignora num líquido)? (c) O $^3$He, a
quase a mesma densidade, permanece normal até milikelvins: enuncie
o obstáculo e a saída que a natureza encontrou. (d) Verificação da lógica composta: um \emph{par} de $^3$He contém um número
par de férmions --- bóson; o $^6$Li guarda $3\text{p} + 3\text{n}
+ 3\text{e} = 9$ férmions: classifique-o e nomeie os célebres
experimentos de gases frios que exploram exatamente esse contraste Li--Li.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ com honestidade. (a) Escreva $N_{\text{exc}} = \int_0^\infty
g(\epsilon)\dd\epsilon/(\eu^{\beta\epsilon} - 1)$ em $\mu = 0$ e
reduza-a, com $x = \beta\epsilon$, a $N_{\text{exc}} =
(V/\lambda_T^3)\cdot\frac{2}{\sqrt\pi}\int_0^\infty\frac{\sqrt
x\,\dd x}{\eu^x - 1}$. (b) Dado que $\int_0^\infty\sqrt x\,\dd x/(
\eu^x - 1) = 2.612\,\sqrt\pi/2$, obtenha o teto e o
$T_{\text{c}}$. (c) Deduza a fração condensada $1 -
(T/T_{\text{c}})^{3/2}$. (d) Por que o mesmo argumento não produz
condensação \emph{alguma} numa caixa bidimensional (como muda $g(
\epsilon)$, e o que acontece com a convergência da
integral)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
Por que os metais se mantêm coesos. Modele um metal como elétrons de
densidade $n$ (custo de \omterm{thm:b3:quantum-statistics:fermi}{energia de Fermi} $\propto n^{2/3}$ por elétron)
atraídos por um fundo iônico neutralizante (ganho de Coulomb por elétron
$\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (a) Escreva a energia por
elétron $\epsilon(n) = An^{2/3} - Bn^{1/3}$ e mostre que ela tem
mínimo. (b) Mostre que o espaçamento eletrônico de equilíbrio é da ordem
de $a_0$ --- as densidades metálicas não são acidentes. (c) Estime
a escala de energia coesiva no mínimo, em eV. (d) Que papel a
\omterm{thm:b3:quantum-statistics:fermi}{pressão de degenerescência} desempenha neste estado ligado (o que
detém o colapso de Coulomb)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
A casca de Sommerfeld, quantificada. (a) Argumente: de $N$ elétrons, só
$\sim g(E_{\text{F}})k_{\text{B}}T$ ficam na casca ativa, cada um
guardando energia extra $\sim k_{\text{B}}T$: deduza $C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ a menos de uma constante. (b) Com
$g(E_{\text{F}}) = 3N/2E_{\text{F}}$, recupere $C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ (a constante exata é $\pi^2/2$).
(c) A mesma lógica de casca dá a suscetibilidade de spin de Pauli,
independente da temperatura: só os elétrons da casca podem virar --- mas
a casca cresce como $T$ enquanto cada contribuição cai como $1/T$: mostre
o cancelamento. (d) Enuncie a moral geral dos férmions degenerados:
\emph{toda} resposta é a clássica vezes
$\sim T/T_{\text{F}}$, ou o análogo dela restrito à superfície.
\end{exercise}

\section{Problema: a anã branca mais pesada possível}

\begin{problem}[{Problema de fim de semana --- o limite de
Chandrasekhar e as velas que ele acendeu}]\label{pb:b3:quantum-statistics:1}
Em 1930, aos dezenove anos, Subrahmanyan Chandrasekhar calculou no
navio para a Inglaterra que a degenerescência eletrônica só sustenta uma
estrela morta até uma massa definida. O resultado --- resistido com
amargor por Eddington --- é a base das estrelas de nêutrons, dos buracos
negros e das ``velas-padrão'' de supernova que revelaram o universo em
aceleração. Este problema o constrói por escalas. Dados: $M_\odot =
\qty{2e30}{kg}$; um elétron por $\mu_e = 2$ núcleons; $m_{\text{
n}} = \qty{1.67e-27}{kg}$; $\hbar c = \qty{3.16e-26}{J.m}$; $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{Parte I --- O equilíbrio de duas energias.}
Para uma estrela de massa $M$ e raio $R$, trate as energias por unidade
de massa por escalas.
\begin{enumerate}
  \item Energia gravitacional: $E_{\text{g}} \sim -GM^2/R$ ---
        lembre a origem dela.
  \item Contagem de elétrons $N_{\text{e}} = M/\mu_e m_{\text{n}}$ e
        densidade $n \sim N_{\text{e}}/R^3$: escreva a escala da
        \omterm{thm:b3:quantum-statistics:fermi}{energia de Fermi} não relativística em função de $M$
        e $R$.
  \item Mostre que a energia cinética (de degenerescência) total escala
        como $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item Minimize $E_{\text{k}} + E_{\text{g}}$ em $R$ e mostre que
        o raio de equilíbrio é
        $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$.
  \item Leia a escala $R \propto M^{-1/3}$: as anãs mais pesadas
        são \emph{menores}. Por que isso já cheira a
        encrenca?
  \item Avalie $R$ para $M = 1\,M_\odot$ e compare com os
        $\approx\qty{5800}{km}$ medidos de Sirius B.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A relatividade puxa o chão
embora.}
\begin{enumerate}[resume]
  \item Quando $R$ encolhe, $E_{\text{F}}$ cresce: estime a densidade
        em que $E_{\text{F}} \sim m_{\text{e}}c^2$ e a
        região de massa de anã correspondente.
  \item No regime ultrarrelativístico, a energia de cada elétron é
        $\sim\hbar c\,n^{1/3}$: mostre que a energia cinética vira
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$.
  \item As duas energias escalam agora como $1/R$: conclua que o
        equilíbrio deixa de selecionar um raio --- compare os dois
        coeficientes em vez disso.
  \item Impondo $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$, deduza o
        número crítico de elétrons e a massa
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item Avalie a razão adimensional $\hbar c/Gm_{\text{n}}^2$
        --- um dos grandes números puros da física --- e depois
        $M_{\text{Ch}}$ em massas solares (o tratamento exato dá
        $1.4\,M_\odot$).
  \item Interprete a fórmula: que três constantes conspiram, e
        por que um limite \emph{quântico} para uma \emph{estrela}
        carrega $\hbar$?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Além do limite.}
\begin{enumerate}[resume]
  \item Uma anã empurrada para além de $M_{\text{Ch}}$ (acreção de
        uma companheira): o que a sustenta em seguida, e a que escala
        de raio (troque $m_{\text{e}}$ por $m_{\text{n}}$ e
        $\mu_e$ por 1 no raio da Parte I)?
  \item Mostre que o limite relativístico da própria estrela de nêutrons
        é da mesma ordem de $M_{\text{Ch}}$ --- a natureza concede só um
        adiamento de fator dois (máximo observado $\approx
        2.2\,M_\odot$). O que espera adiante?
  \item Durante o colapso para estrela de nêutrons, libera-se
        $\sim GM^2/R_{
        \text{ns}}$ de energia gravitacional: avalie
        isso para $M = 1.4\,M_\odot$, $R = \qty{11}{km}$, e
        compare com toda a produção do Sol em \qty{10}{Gyr}
        ($\sim\qty{1.2e44}{J}$).
  \item Quase tudo isso sai como neutrinos em segundos: que
        observação de 1987 confirmou este quadro de modo espetacular?
  \item O ricochete do caroço em colapso e o empurrão dos neutrinos
        arrancam o envelope: uma supernova por colapso de caroço.
        Distinga-a, em uma frase cada, da supernova
        \emph{termonuclear} (tipo Ia) de uma \omterm{ex:b3:quantum-statistics:dwarf}{anã branca} que se acende no
        limiar de Chandrasekhar.
  \item Por que a detonação \emph{a uma massa universal} torna
        as supernovas de tipo Ia quase idênticas --- e, portanto, úteis
        como velas-padrão?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- As velas e o
cosmo.}
\begin{enumerate}[resume]
  \item O brilho aparente de uma vela-padrão dá a
        distância dela: enuncie a lógica do inverso do quadrado e por que
        \emph{padrão} é a palavra decisiva.
  \item Em 1998, duas equipes acharam supernovas de tipo Ia distantes
        \emph{mais fracas} do que o esperado: enuncie a inferência
        (expansão acelerando --- energia escura) e por que
        a confiabilidade da vela era o ponto crucial.
  \item Liste a cadeia deste problema em uma linha: Pauli $\to$
        \omterm{thm:b3:quantum-statistics:fermi}{pressão de degenerescência} $\to$ $M_{\text{Ch}}$ $\to$ explosões
        uniformes $\to$ aceleração cósmica.
  \item Eddington escarneceu do resultado (``uma estrela cometendo
        absurdos''); Chandrasekhar recebeu o Nobel
        cinquenta e três anos depois: o que o episódio ensina sobre
        seguir as equações para além das conclusões confortáveis?
  \item O espectro de Sirius B em 1915 implicava uma tonelada por
        centímetro cúbico --- ``absurdo'' até que Fowler, poucos meses
        depois da estatística de Fermi de 1926, explicou a estrela como
        um gás degenerado: enuncie por que o enigma era insolúvel num
        ano e rotineiro no seguinte.
  \item Levantamentos de milhares de \omterm{ex:b3:quantum-statistics:dwarf}{anãs brancas} acham massas
        agrupadas perto de $0.6\,M_\odot$ e nenhuma acima de
        $\approx1.4\,M_\odot$: que duas previsões deste
        problema esse único histograma confirma?
  \item Resuma o resultado nomeado: $(\hbar c/Gm_{\text{n}}^2)^{
        3/2}m_{\text{n}} \approx 1.4\,M_\odot$ --- três constantes
        da natureza fixando a \omterm{ex:b3:quantum-statistics:dwarf}{anã branca} mais pesada possível, a
        massa de gatilho das explosões calibradas do universo.
\end{enumerate}
\end{problem}
