\chapter{Teoria de perturbações}\label{ch:b3:perturbation-theory}

Apenas um punhado de problemas quânticos se dissolve exatamente: a
caixa, o oscilador, o hidrogênio, o sistema de dois níveis. Todo o
resto --- um átomo num campo, a ligação enrijecida de uma molécula, um
nível empurrado por um vizinho --- é um desses esqueletos solúveis
vestindo um pequeno termo a mais. A \omterm{thm:b3:perturbation-theory:corrections}{teoria de perturbações} é a arte de
tratar o termo extra pelo que ele é: uma correção, calculada ordem a
ordem na sua pequenez. Sua regra de primeira ordem é uma média de uma
linha; sua regra de segunda ordem explica por que os níveis
\emph{se repelem}, por que todo átomo é polarizável e por que a atração
de van der Waals é universal; sua variante degenerada trata os casos
delicados em que níveis quase iguais se reorganizam por completo; e sua
forma dependente do tempo --- a regra de ouro de Fermi --- é a equação
por trás de toda linha de absorção, taxa de decaimento e seção de
choque medidas num laboratório. Como recompensa de percurso, o capítulo
termina calculando a cor do céu.

\section{Pequenos deslocamentos: a teoria não degenerada}

\begin{theorem}[Correções perturbativas]\label{thm:b3:perturbation-theory:corrections}
\index{teoria de perturbações}
Seja $\hat H = \hat H_0 + \hat V$ com $\hat H_0$ resolvido
($\hat H_0\ket n = E_n\ket n$, espectro não degenerado) e $\hat V$
pequeno. Então, ordem a ordem em $\hat V$,
\[
E_n' = E_n + \bra n\hat V\ket n
 + \sum_{m \neq n}\frac{|\bra m\hat V\ket n|^2}{E_n - E_m}
 + \cdots
\]
e o estado adquire as misturas $\ket{n'} = \ket n + \sum_{m\neq
n}\dfrac{\bra m\hat V\ket n}{E_n - E_m}\,\ket m + \cdots$
\emph{Primeira ordem}: o deslocamento é apenas a média da perturbação
no estado não perturbado. \emph{Segunda ordem}: os acoplamentos com os
outros níveis empurram $E_n$ para \emph{longe} deles --- cada $m$ acima
empurra para baixo, e cada um abaixo empurra para cima --- de modo que o
deslocamento de segunda ordem do estado fundamental é sempre negativo.
Validade: as frações de mistura $|\bra m\hat V\ket n/(E_n - E_m)|$
precisam ser pequenas.
\end{theorem}

\begin{proof}[Demonstração parcial]
Expanda $E' = E + \epsilon_1 + \epsilon_2 + \cdots$ e $\ket{\psi} =
\ket n + \ket{\psi_1} + \cdots$ em potências de $\hat V$, insira em
$(\hat H_0 + \hat V)\ket\psi = E'\ket\psi$ e iguale as ordens.
Projetar a equação de primeira ordem sobre $\bra n$ dá $\epsilon_1 =
\bra n\hat V\ket n$; projetar sobre $\bra m$ ($m \neq n$) dá os
coeficientes de mistura; levar esses coeficientes à equação de segunda
ordem e projetar de novo sobre $\bra n$ dá $\epsilon_2$. Quanto ao
enunciado sobre o sinal para o estado fundamental: todo denominador
$E_0 - E_m$ é negativo, e os numeradores são quadrados.
\end{proof}

\begin{proposition}[Uma perturbação com resposta exata]\label{prop:b3:perturbation-theory:oscillator-field}
Um oscilador de carga $q$ num campo uniforme $\mathcal E$ tem
$\hat V = -q\mathcal E\hat x$. Completar o quadrado o resolve
exatamente: o potencial é a mesma parábola, deslocada, com todos os
níveis rebaixados de $q^2\mathcal E^2/2m\omega^2$. A \omterm{thm:b3:perturbation-theory:corrections}{teoria de
perturbações} tem de concordar --- e concorda: a primeira ordem se anula
($\langle\hat
x\rangle_n = 0$) e a soma de segunda ordem, na qual $\hat x$ acopla $n$
apenas a $n \pm 1$, dá exatamente $-q^2\mathcal E^2/2m\omega^2$ para
todo nível (\cref{exo:b3:perturbation-theory:2}). Um caso solúvel,
conferido dos dois modos, é a melhor calibração que um método pode ter.
\end{proposition}

\begin{proof}
Os dois elementos de matriz e seus denominadores dão
\[
\frac{|\bra{n+1}\hat x\ket{n}|^2}{-\hbar\omega}
 + \frac{|\bra{n-1}\hat x\ket n|^2}{+\hbar\omega}
 = \frac{\hbar}{2m\omega}\,\frac{-(n+1) + n}{\hbar\omega}
 = -\frac{1}{2m\omega^2} ;
\]
multiplique por $q^2\mathcal E^2$.
\end{proof}

\section{Quando os níveis são degenerados}

\begin{theorem}[Teoria de perturbações degenerada]\label{thm:b3:perturbation-theory:degenerate}
\index{teoria de perturbações degenerada}
Se $E_n$ é degenerado, a fórmula acima divide por zero --- a
perturbação pode reorganizar por completo os estados degenerados, e o
remédio é deixá-la fazer isso: \emph{diagonalize $\hat V$ restrito ao
subespaço degenerado}. Seus \omterm{def:b3:quantum-formalism:observable}{autovalores} são os deslocamentos de
primeira ordem; seus autovetores são os estados corretos de ordem zero,
as combinações que a própria perturbação seleciona.
\end{theorem}
\admitted

\begin{example}[O efeito Stark linear]\label{ex:b3:perturbation-theory:stark}
\index{efeito Stark}
O nível $n = 2$ do hidrogênio contém os estados degenerados $2s$ e $2p$.
Num campo $\mathcal E\vect e_z$, $\hat V = e\mathcal E\hat z$
acopla $2s$ a $2p_0$ com $\bra{2s}\hat z\ket{2p_0} = -3a_0$:
diagonalizar o bloco $2\times2$ dá os deslocamentos
\[
\Delta E = \pm\,3e a_0\mathcal E ,
\]
\emph{lineares} no campo --- ao passo que o estado fundamental, não
degenerado, se desloca apenas quadraticamente. Em
$\mathcal E = \qty{e7}{V/m}$: $\pm
\qty{1.6}{meV}$, um desdobramento facilmente resolvido. Os estados que
fazem o desdobramento são $(\ket{2s} \mp \ket{2p_0})/\sqrt2$ ---
híbridos com dipolo elétrico permanente, possíveis apenas porque a
degenerescência deixou o átomo livre para se polarizar em ordem zero.
(A mesma hibridização $s$--$p$, movida por vizinhos em vez de por um
campo, é a química do carbono de toda molécula orgânica.)
\end{example}

\section{Transições: a regra de ouro}

\begin{theorem}[Perturbações dependentes do tempo]\label{thm:b3:perturbation-theory:golden}
\index{regra de ouro de Fermi}
Ligue $\hat V(t) = \hat V\cos\omega t$ em $t = 0$. Em primeira ordem, a
probabilidade de encontrar o sistema em $\ket f$ no instante $t$, tendo
começado em $\ket i$, vale
\[
\mathcal P_{i\to f}(t) = \frac{|\bra f\hat V\ket i|^2}{4\hbar^2}\;
\frac{\sin^2\!\big[(\omega_{fi} - \omega)\,t/2\big]}
{\big[(\omega_{fi} - \omega)/2\big]^2} ,
\qquad \omega_{fi} = \frac{E_f - E_i}{\hbar} :
\]
uma ressonância agudamente centrada em $\hbar\omega = E_f - E_i$ ---
absorção quando $E_f > E_i$, \emph{emissão estimulada} para o termo
companheiro com $E_f < E_i$. Quando os estados finais formam um
contínuo de densidade $\rho(E)$, o crescimento do pico se torna uma
\emph{taxa} constante:
\[
\Gamma_{i\to f} = \frac{2\pi}{\hbar}\,
|\bra f\hat V\ket i|^2\,\rho(E_f)
\]
--- a \emph{regra de ouro de Fermi}, a fórmula mestra das taxas de
decaimento, dos coeficientes de absorção e das seções de choque.
\end{theorem}

\begin{proof}[Demonstração parcial]
Amplitude de primeira ordem: $c_f(t) = -\tfrac{\iu}{\hbar}\int_0^t\bra
f\hat V(t')\ket i\,\eu^{\iu\omega_{fi}t'}\dd t'$; guardar a
exponencial quase ressonante e integrar dá a forma em $\sin^2$
enunciada. Para o contínuo: como função da dessintonia $x$, o fator de
ressonância tem altura $t^2$ e largura $\sim 2\pi/t$ --- área $2\pi t$
--- de modo que somar sobre um contínuo suave o substitui por
$2\pi t\,\rho$: probabilidade linear em $t$, isto é, uma taxa.
\end{proof}

\begin{example}[O que a regra de ouro governa]\label{ex:b3:perturbation-theory:applications}
Regras de seleção: uma transição só acontece se o elemento de matriz
$\bra f\hat V\ket i$ sobrevive --- para a luz, $\hat V \propto \hat
z$, cuja paridade e cujo momento angular matam tudo, exceto
$\Delta l = \pm1$ e $\Delta m = 0, \pm1$: as regras invocadas desde o
\cref{ch:b3:quantum-angular-momentum}, agora teoremas. Taxas: o
decaimento espontâneo do estado $2p$ do hidrogênio dá
\qty{1.6}{ns} (\cref{exo:b3:perturbation-theory:8}); a transição de
maser da amônia, com seu $\omega^3$ de micro-ondas, leva meses --- o
$\omega^3$ das taxas de radiação é a razão de as linhas de rádio
(\cref{ex:b3:spin-two-level:hyperfine}) viverem megaanos, enquanto as
linhas ultravioletas relampejam em nanossegundos. Espectros de
absorção, fotoionização, decaimento beta nuclear
(\cref{ch:b3:nuclear-physics}), seções de choque de espalhamento
(\cref{ch:b3:scattering-theory}): tudo é esta única fórmula, com
elementos de matriz e contagens de estados diferentes.
\end{example}

\begin{method}[Escolher a ferramenta perturbativa certa]\label{met:b3:perturbation-theory:method}
(1) Questão estática, nível não degenerado: tome a média de $\hat V$
(primeira ordem); se ela se anular por simetria, vá à segunda ordem ---
espere repulsão de níveis. (2) Nível degenerado: diagonalize primeiro
$\hat V$ no bloco degenerado; são as combinações adaptadas à simetria
que fazem o desdobramento. (3) Taxas de transição: regra de ouro --- um
elemento de matriz, uma \omterm{prop:b3:schrodinger-three-dimensions:counting}{densidade de estados}; confira as regras de
seleção antes de calcular qualquer coisa. (4) Teste sempre a expansão:
a fração de mistura $|V_{mn}/(E_n - E_m)| \ll 1$; do contrário,
diagonalize exatamente (fórmula de dois níveis). (5) Os limites
solúveis exatos (oscilador num campo) são calibrações gratuitas: use-os.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:1}
Uma partícula numa caixa $[0, a]$ sente o potencial adicional $V(x) =
\lambda x/a$. (a) Calcule o deslocamento de primeira ordem de todo
nível (use $\langle x\rangle = a/2$ em qualquer estado da caixa ---
justifique). (b) Por que o deslocamento é aqui o mesmo para todos os
níveis? (c) Qual é a variação de primeira ordem dos \emph{espaçamentos},
e por que uma medida de frequências de transição poderia perder
inteiramente esta perturbação? (d) Quando a primeira ordem deixa de ser
confiável?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:2}
Faça por inteiro o cálculo de segunda ordem da
\cref{prop:b3:perturbation-theory:oscillator-field}: os elementos de
matriz de $\hat x$, os dois termos, o cancelamento de $n$ e o acordo
exato com a técnica de completar o quadrado. Por que a correção de
primeira ordem do \emph{estado} não se anula, embora a da energia se
anule?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:3}
A \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} de dois níveis $\begin{pmatrix}E_1 & v\\ v &
E_2\end{pmatrix}$, com $E_1 < E_2$ e $v$ real. (a) Encontre os
\omterm{def:b3:quantum-formalism:observable}{autovalores} exatos. (b) Expanda para $|v| \ll E_2 - E_1$ e identifique
a fórmula de segunda ordem. (c) Mostre que os níveis se repelem: a
separação sempre excede $E_2 - E_1$. (d) Em $E_1 = E_2$: o que a \omterm{thm:b3:perturbation-theory:corrections}{teoria
de perturbações} dá, o que a resposta exata dá, e qual teorema do
capítulo os reconcilia?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:4}
Para o \cref{thm:b3:perturbation-theory:corrections}: (a) mostre que a
correção de primeira ordem do estado é ortogonal a $\ket n$; (b) mostre
que o coeficiente de mistura de $\ket m$ vale $V_{mn}/(E_n - E_m)$ e
enuncie o critério de validade; (c) um nível a $\qty{1}{eV}$ do vizinho
mais próximo está acoplado a ele por \qty{0.1}{eV}: estime a
probabilidade de mistura; (d) o mesmo com uma separação de
\qty{10}{meV} --- que ferramenta deve substituir a expansão?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:5}
Ligações anarmônicas. Acrescente $\hat V = \beta\hat x^3 + \gamma\hat x^4$
ao oscilador. (a) Mostre que o termo em $x^3$ não desloca nível algum
em primeira ordem (paridade). (b) Mostre que o termo em $x^4$ dá $\Delta E_n =
3\gamma(\hbar/2m\omega)^2(2n^2 + 2n + 1)$ (expanda
$\hat x^4$ em \omterm{def:b3:harmonic-oscillator:ladder}{operadores de escada}; só os termos ``equilibrados''
sobrevivem). (c) Para uma ligação real, o $\gamma$ efetivo é negativo
(o poço amolece para fora): mostre que o espaçamento entre níveis então
\emph{encolhe} com $n$.
(d) Que característica observada dos espectros moleculares
(\cref{exo:b3:harmonic-oscillator:4}) isso reproduz?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:6}
Na caixa quadrada bidimensional, o par degenerado $\ket{1,2},
\ket{2,1}$ é perturbado por $\hat V = \lambda\,xy$. (a) Explique por
que a teoria não degenerada falha. (b) Calcule a matriz $2\times2$ de
$\hat V$ no par (as integrais se fatoram; $\int_0^a
x\sin(\pi x/a)\sin(2\pi x/a)\dd x = -8a^2/9\pi^2$). (c) Encontre os
desdobramentos e os estados corretos de ordem zero. (d) Compare com a
análise de simetria do \cref{exo:b3:quantum-formalism:9}: que resposta
a simetria deu de graça?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:7}
\omterm{ex:b3:perturbation-theory:stark}{Efeitos Stark}, linear e quadrático. (a) Usando o
\cref{ex:b3:perturbation-theory:stark}, calcule o desdobramento de
$n = 2$ em $\mathcal E = \qty{2.5e6}{V/m}$. (b) Por que o estado
fundamental \emph{não} mostra deslocamento linear (duas razões: a
paridade e a ausência de parceiro degenerado)? (c) Seu deslocamento
quadrático define a polarizabilidade,
$\Delta E = -\tfrac12\alpha\mathcal E^2$: estime
$\alpha \sim e^2a_0^2/E_{\text{I}}$ e avalie em unidades de
$a_0^3$ (a resposta exata é $4.5\,a_0^3 \times
4\pi\varepsilon_0$). (d) De que propriedade cotidiana da matéria --- a
resposta dielétrica dela --- você acabou de calcular a semente atômica?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:8}
Emissão espontânea (com um dado admitido): um estado excitado decai a
$A = \omega^3|d_{fi}|^2/3\pi\varepsilon_0\hbar c^3$, em que elementos
de matriz do tipo $d_{fi} = e\bra f\hat z\ket i$ fixam o dipolo. (a)
Para o $2p \to 1s$ do hidrogênio: com $|d| \approx 0.74\,
ea_0$ e $\lambda = \qty{121.6}{nm}$, calcule $A$ e a vida média. (b)
Reescale para a linha D do sódio. (c) Reescale para a transição
hiperfina de 21 cm ($|d| \to$ um momento magnético, com a taxa baixando
mais um fator da ordem de $\alpha^2$; aceite $A \approx
\qty{2.9e-15}{s^{-1}}$): confira os ``dez milhões de anos''. (d) A
partir do $\omega^3$: por que as linhas ultravioletas são rápidas, as
de rádio são eternas, e por que isso tornou a \omterm{ex:b3:spin-two-level:hyperfine}{linha de 21 cm}
\emph{previsível}, mas quase indetectável num laboratório?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:9}
Regras de seleção como integrais. (a) Mostre que $\bra{100}\hat z\ket{200} =
0$ por paridade. (b) Mostre que $\bra{100}\hat z\ket{21m} = 0$ a menos que $m =
0$ (a integral em $\varphi$). (c) Enuncie as regras resultantes $\Delta
l = \pm1$, $\Delta m = 0, \pm1$ e sua leitura
física (o spin do fóton). (d) O estado $2s$ não alcança o $1s$ por rota
dipolar alguma: o que isso prevê para a vida média dele, e que caminho
``proibido'' de dois fótons a natureza de fato toma?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:10}
Polarizabilidade, honestamente. O deslocamento de segunda ordem do
estado fundamental do hidrogênio num campo vale
$\Delta E = -e^2\mathcal E^2\sum_{m\neq
0}|\bra m\hat z\ket 0|^2/(E_m - E_0)$. (a) Limite a soma substituindo
todo denominador pela menor separação $E_2 - E_1 =
\tfrac34 E_{\text{I}}$ e usando a relação de fechamento $\sum_m|\bra
m\hat z\ket0|^2 = \langle z^2\rangle_0 = a_0^2$: obtenha $\alpha
\le \tfrac{16}3\,a_0^3$ (em unidades ao estilo gaussiano de
$4\pi\varepsilon_0$). (b) Compare com o valor exato $\tfrac92 a_0^3$.
(c) A partir de $\alpha$, estime o índice de refração do gás
hidrogênio à densidade atmosférica ($n - 1 \approx N\alpha/2\varepsilon_0
\cdot e^2\dots$ --- use $n - 1 = N\alpha_{\text{SI}}/2
\varepsilon_0$) e compare com o valor medido $\num{1.3e-4}$.
(d) Enuncie numa frase a cadeia átomo $\to$ polarizabilidade $\to$
refração que a Parte II do problema de fim de semana vai percorrer.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:11}
Cruzamentos evitados e varreduras lentas. Um sistema de dois níveis tem
energias nuas $\pm\lambda t$ que se cruzam em $t = 0$, acopladas por
$v$. (a) Esboce os níveis exatos contra $t$: um cruzamento evitado de
separação $2|v|$. (b) Se a varredura for \emph{lenta}, o sistema segue a
curva inferior (teorema adiabático, admitido): em que estado ele
termina? (c) O critério de Landau--Zener compara o tempo de varredura
na região da separação, $\sim v/\lambda$, com o relógio interno
$\hbar/v$: dê a condição para o seguimento adiabático. (d) Dois usos: o
estado de um qubit transferido por uma varredura lenta de frequência;
uma colisão atômica que salta carga entre núcleos --- explique um deles
em duas frases.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:12}
A segunda ordem é pessimista, e outros teoremas. (a) Demonstre que a
correção de segunda ordem ao estado \emph{fundamental} é sempre
negativa. (b) Conclua (com o
\cref{exo:b3:harmonic-oscillator:10}) por que as forças de van der
Waals entre átomos no estado fundamental são sempre atrativas. (c)
Mostre que a teoria de primeira ordem \emph{superestima} toda energia
de estado fundamental: ela é um limite variacional (avalie $\bra{\psi_0}\hat
H\ket{\psi_0}$ com o estado não perturbado como
função de teste). (d) Uma verificação de coerência sobre o
\cref{ex:b3:perturbation-theory:stark}: por que o \omterm{ex:b3:perturbation-theory:stark}{efeito Stark}
\emph{linear} não viola (a)?
\end{exercise}

\section{Problema: A cor do céu}

\begin{problem}[{Problema de fim de semana --- espalhamento Rayleigh
por átomos perturbados}]\label{pb:b3:perturbation-theory:1}
Por que o céu do dia é azul, o pôr do sol é vermelho e a nuvem é
branca? A resposta completa é uma cadeia que atravessa este capítulo:
uma onda de luz perturba cada molécula do ar; o dipolo induzido
reirradia (a radiação de dipolo do volume do segundo ano); e a
interferência decide o que sobrevive. Dados: polarizabilidade molecular
do ar $\alpha/4\pi\varepsilon_0 \approx \qty{2.2e-30}{m^3}$; densidade
numérica $N = \qty{2.5e25}{m^{-3}}$; faixa visível de
$450$ a \qty{700}{nm}.

\medskip\noindent\textbf{Parte I --- A molécula perturbada.}
\begin{enumerate}
  \item Um campo estático $\mathcal E$ desloca o estado fundamental de
        uma molécula de $-\tfrac12\alpha\mathcal E^2$: relacione esse
        enunciado à \omterm{thm:b3:perturbation-theory:corrections}{teoria de perturbações} de segunda ordem (que
        fórmula, e por que negativa?).
  \item O mesmo $\alpha$ dá o dipolo induzido $p =
        \alpha\mathcal E$: para o campo da luz solar, $\mathcal E \sim
        \qty{800}{V/m}$, calcule o dipolo induzido de uma molécula de
        nitrogênio e compare-o com $ea_0$.
  \item Por que o campo óptico (de frequência $\num{5e14}$ Hz) pode ser
        tratado com a polarizabilidade \emph{estática} no caso de
        moléculas cujas transições eletrônicas ficam no ultravioleta?
        (Compare $\hbar\omega$ com a separação; este é o limite de
        forte fora de ressonância do denominador da regra de ouro.)
  \item Um dipolo oscilante $p_0\cos\omega t$ irradia a potência média
        $P = p_0^2\omega^4/12\pi\varepsilon_0c^3$ (volume do segundo
        ano). Combinando com $p_0 = \alpha\mathcal
        E_0$: como a potência espalhada depende de $\omega$ a
        iluminação fixa?
  \item Calcule a razão entre as potências espalhadas em \qty{450}{nm}
        e em \qty{700}{nm}.
  \item Numa frase: por que o céu é azul e não violeta (dois
        ingredientes: o espectro do Sol, que cai rumo ao violeta, e a
        sensibilidade do olho)?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- De uma molécula ao céu
inteiro.}
\begin{enumerate}[resume]
  \item A seção de choque de espalhamento resulta em $\sigma =
        \dfrac{8\pi}{3}\Big(\dfrac{\alpha}{4\pi\varepsilon_0}
        \Big)^2\dfrac{\omega^4}{c^4}$: verifique as dimensões dela e
        avalie-a em \qty{550}{nm}.
  \item O comprimento de atenuação é $\ell = 1/N\sigma$: avalie-o em
        \qty{550}{nm} e compare com a espessura da atmosfera
        ($\sim\qty{8}{km}$ equivalentes à densidade do nível do mar).
  \item Em \qty{450}{nm} contra \qty{700}{nm}, que fração de um raio de
        sol vertical é espalhada para fora? (Use $1 -
        \eu^{-L/\ell}$ com $L = \qty{8}{km}$.)
  \item No pôr do sol o caminho é trinta vezes mais longo: refaça o
        cálculo da sobrevivência do azul e explique a cor do Sol baixo
        --- e a da luz que avermelha a Lua num eclipse lunar.
  \item Por que a luz do céu é \emph{polarizada} a $90^\circ$ do Sol
        (lembre o padrão de radiação de um dipolo: não há emissão ao
        longo do eixo do dipolo)?
  \item Diz-se que abelhas e vikings navegavam por essa polarização: o
        que um dia encoberto faz com ela, e por quê?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Por que o céu não é mais
brilhante.}
\begin{enumerate}[resume]
  \item Um paradoxo: num meio perfeitamente uniforme, as ondículas de
        elementos de volume vizinhos interferem destrutivamente para os
        lados, e \emph{nenhuma} luz seria espalhada (só a refração
        sobreviveria). O que de fato quebra a uniformidade do ar --- o
        que está distribuído ao acaso?
  \item As flutuações de densidade de um gás ideal fazem as
        intensidades espalhadas por moléculas independentes \emph{se
        somarem}: diga por que a independência (posições ao acaso,
        separações da escala do comprimento de onda) mata a
        interferência destrutiva.
  \item A partir do seu $\ell$ em \qty{550}{nm}: que fração da luz
        solar uma atmosfera de dia claro espalha --- o número bate com
        o brilho cotidiano do céu comparado ao do Sol?
  \item Uma gotícula de nuvem (\qty{10}{\micro m}) contém
        $\sim10^{12}$ moléculas dentro de um comprimento de onda: as
        ondículas delas se somam \emph{coerentemente}. Como a potência
        espalhada escala então com o número de moléculas, e por que a
        seleção de cor pelo $\omega^4$ desaparece (todos os
        comprimentos de onda se espalham fortemente): de que cor é a
        nuvem?
  \item O leite, a névoa e a tinta branca são brancos pela mesma razão:
        enuncie a regra geral --- espalhadores pequenos em relação a
        $\lambda$ colorem a luz; espalhadores grandes em relação a
        $\lambda$ a embranquecem.
  \item Opalescência crítica: perto do ponto crítico de um líquido
        (os capítulos de mudança de fase do volume do primeiro ano e o
        \cref{ch:b3:phase-transitions} adiante), as flutuações de densidade crescem até escalas ópticas
        e o fluido fica leitoso: relacione com o item 13.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- A cadeia, completada.}
\begin{enumerate}[resume]
  \item As ondículas espalhadas para a frente \emph{não} se cancelam:
        elas interferem com o feixe e atrasam sua fase --- o índice de
        refração. Usando $n - 1 = N\alpha/2\varepsilon_0$
        (com $\alpha$ no SI), avalie $n - 1$ para o ar e compare com o
        valor medido \num{2.9e-4}.
  \item Uma única constante $\alpha$, calculada por \omterm{thm:b3:perturbation-theory:corrections}{teoria de
        perturbações} de segunda ordem, fixa assim três fenômenos de uma
        vez: nomeie-os (o deslocamento num campo estático; o azul do
        céu; o encurvamento da luz no ar).
  \item A absorção pelo ozônio, os aerossóis e o espalhamento múltiplo
        complicam os céus reais: nomeie um observável de cada (as cores
        do crepúsculo; o horizonte leitoso; o brilho residual do céu no
        zênite depois do pôr do sol).
  \item Por que a Lua, sem ar, tem um céu diurno \emph{preto} --- e o
        que cada fotografia tirada de sua superfície confirmou, com
        isso, sobre a origem do nosso?
  \item O céu diurno de Marte tem cor de caramelo e seus poentes são
        \emph{azuis}: seu ar rarefeito espalha pouco, e são os grãos de
        poeira micrométricos em suspensão que fazem o trabalho. Usando
        a regra do item 18, explique como a poeira inverte a lógica das
        cores.
  \item Para os raios X, o $\hbar\omega$ excede em muito toda transição
        molecular: os elétrons respondem como se estivessem livres, e o
        espalhamento (Thomson) perde o seu $\omega^4$. O que acontece
        com o mecanismo do ``azul do céu'' nesse regime --- e por que
        essa resposta plana é exatamente o que faz dos raios X sondas
        limpas da densidade eletrônica?
  \item Resuma o resultado central: uma polarizabilidade de
        $\qty{2.2e-30}{m^3}$, elevada ao quadrado e multiplicada por
        $\omega^4$, dá um comprimento de espalhamento da escala de
        \qty{60}{km} no verde --- longo o bastante para que o céu do
        meio-dia seja gentil, curto o bastante para que os poentes
        avermelhem e os céus sejam azuis, com o padrão de polarização
        de uma floresta de dipolos excitados.
\end{enumerate}
\end{problem}
