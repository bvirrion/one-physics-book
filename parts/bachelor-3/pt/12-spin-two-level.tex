\chapter{Spin e sistemas de dois níveis}\label{ch:b3:spin-two-level}

Em 1922, Stern e Gerlach enviaram um feixe de átomos de prata através
de um ímã assimétrico, esperando ou nenhum desvio ou um borrão
contínuo. O feixe se dividiu limpamente em \emph{dois}. Nenhum momento
angular orbital consegue isso: $2l + 1$ é sempre ímpar. Os átomos
anunciavam uma posse nova e puramente quântica --- o \emph{\omterm{def:b3:spin-two-level:spin}{spin}}, um
momento angular intrínseco de $j = \tfrac12$, o degrau semi-inteiro que
a álgebra do \cref{ch:b3:quantum-angular-momentum} mantinha pronto e
que as órbitas jamais poderiam usar. O \omterm{def:b3:spin-two-level:spin}{spin} duplica todo estado
eletrônico (completando a contagem por trás da tabela periódica),
carrega o magnetismo do ferro e do próton e é o sistema de dois níveis
mais limpo que a natureza oferece: a física deste capítulo faz
funcionar o aparelho de \omterm{prop:b3:spin-two-level:rabi}{ressonância magnética} de todo hospital, o
relógio de césio que define o segundo, o sussurro de vinte e um
centímetros com que se mapeiam galáxias, e o qubit.

\section{O experimento que o encontrou}

\begin{example}[Stern--Gerlach]\label{ex:b3:spin-two-level:stern-gerlach}
Um momento magnético $\vect\mu$ num campo \emph{não homogêneo} sente a
força $F_z = \mu_z\,\partial B_z/\partial z$: o desvio mede $\mu_z$.
Expectativa clássica: momentos orientados ao acaso, um leque contínuo.
Expectativa quântica para momentos orbitais: $2l + 1$ manchas --- uma,
três, cinco, sempre um número ímpar. Observado para a prata (e para o
hidrogênio): \emph{duas} manchas, simétricas, e nada entre elas. A
componente medida assume exatamente dois valores --- a assinatura de $j
= \tfrac12$, proibida às órbitas, e a exibição direta da quantização: o
aparelho é um dispositivo de medida de uma componente do \omterm{def:b3:spin-two-level:spin}{spin}, e o
feixe se divide nos dois \omterm{def:b3:quantum-formalism:observable}{autovalores} dela.
\end{example}

\begin{definition}[Spin um meio]\label{def:b3:spin-two-level:spin}
\index{spin}\index{matrizes de Pauli}\index{espinor}
O elétron (e o próton, o nêutron, os quarks) carrega um momento angular
intrínseco com $j = \tfrac12$: um espaço de estados bidimensional
gerado por $\ket\uparrow, \ket\downarrow$ (autoestados de $\hat S_z$
com $\pm\hbar/2$), sobre o qual
\[
\hat{\vect S} = \frac\hbar2\,\hat{\vect\sigma} , \qquad
\sigma_x = \begin{pmatrix}0&1\\1&0\end{pmatrix} ,\;
\sigma_y = \begin{pmatrix}0&-\iu\\\iu&0\end{pmatrix} ,\;
\sigma_z = \begin{pmatrix}1&0\\0&-1\end{pmatrix}
\]
--- as \emph{matrizes de Pauli}, que realizam a álgebra do momento
angular na menor casa possível. Um estado geral
$\alpha\ket\uparrow + \beta\ket\downarrow$ é um \emph{espinor}. O
momento magnético do elétron é
\[
\hat{\vect\mu} = -g\,\frac{\mu_{\text{B}}}{\hbar}\,\hat{\vect S} ,
\qquad g \approx 2 :
\]
o dobro da taxa orbital por unidade de momento angular --- a anomalia
que Einstein e de Haas haviam medido
(\cref{exo:b3:quantum-angular-momentum:12}) e que a equação de Dirac
mais tarde preveria. O spin não é a rotação de coisa alguma: nenhum
raio, nenhuma ``bolinha girando'' sobrevive ao exame --- ele é
intrínseco, como a carga.
\end{definition}

\begin{example}[Filtros de Stern--Gerlach em cadeia]\label{ex:b3:spin-two-level:chained}
Selecione o feixe de $S_z = +\hbar/2$ e meça $S_z$ de novo: todos os
átomos respondem $+$. Meça $S_x$ em vez disso: metade e metade --- o
estado $\ket\uparrow$ é a superposição $(\ket{+x} + \ket{-x})/\sqrt2$.
Agora guarde o feixe de $S_x = +$ e meça $S_z$ mais uma vez: metade e
metade \emph{de novo} --- a medida de $S_x$ apagou o $S_z$ antes bem
definido. Três ímãs bastam para exibir a incompatibilidade, o colapso e
a regra de Born; esta cadeia é o
\cref{ex:b3:quantum-formalism:threepolarizers} realizado com átomos.
\end{example}

\section{A esfera de Bloch}

\begin{proposition}[Spin ao longo de um eixo qualquer]\label{prop:b3:spin-two-level:bloch}
\index{esfera de Bloch}
Para a direção unitária $\vect n = (\sin\theta\cos\varphi,
\sin\theta\sin\varphi, \cos\theta)$, o operador $\vect
n\cdot\hat{\vect\sigma}$ tem \omterm{def:b3:quantum-formalism:observable}{autovalores} $\pm1$, com
\[
\ket{+\vect n} = \cos\tfrac\theta2\,\ket\uparrow +
\eu^{\iu\varphi}\sin\tfrac\theta2\,\ket\downarrow .
\]
Todo \omterm{def:b3:spin-two-level:spin}{espinor} puro é $\ket{+\vect n}$ para exatamente uma direção: o
espaço de estados se aplica sobre uma esfera --- a \emph{\omterm{prop:b3:spin-two-level:bloch}{esfera de
Bloch}} --- com os estados ortogonais em \emph{antípodas} (note os meios
ângulos: são as direções opostas, e não as perpendiculares, que dão
estados ortogonais). As rotações da esfera são exatamente as evoluções
que um campo magnético gera: a geometria de toda manipulação de qubit
já realizada.
\end{proposition}

\begin{proof}[Demonstração parcial]
Diagonalize a matriz $2\times2$ $\vect n\cdot\vect\sigma =
\begin{pmatrix}\cos\theta & \sin\theta\,\eu^{-\iu\varphi}\\
\sin\theta\,\eu^{\iu\varphi} & -\cos\theta\end{pmatrix}$: traço $0$,
determinante $-1$, \omterm{def:b3:quantum-formalism:observable}{autovalores} $\pm1$; o \omterm{def:b3:spin-two-level:spin}{espinor} dado se verifica por
substituição (identidades de meio ângulo). Antípodas: $\theta \to \pi -
\theta$, $\varphi \to \varphi + \pi$ dá $\braket{+\vect
n'}{+\vect n} = 0$.
\end{proof}

\section{Spin num campo: precessão e ressonância}

\begin{proposition}[Precessão de Larmor]\label{prop:b3:spin-two-level:larmor}
\index{precessão de Larmor}\index{razão giromagnética}
Num campo estático $\vect B_0 = B_0\vect e_z$, a \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} $\hat
H = -\hat{\vect\mu}\cdot\vect B_0$ separa os dois níveis por
$\hbar\omega_0$ e faz o \omterm{def:b3:spin-two-level:spin}{spin} médio \emph{precessar} em torno do
campo:
\[
\frac{\dd\langle\hat{\vect S}\rangle}{\dd t} =
\gamma\,\langle\hat{\vect S}\rangle\wedge\vect B_0 , \qquad
\omega_0 = |\gamma|B_0 ,
\]
com a \omterm{prop:b3:spin-two-level:larmor}{razão giromagnética} $\gamma$ ($\mu = \gamma S$). O vetor de Bloch
gira em torno de $z$ a $\omega_0$, com o ângulo em relação ao campo
congelado --- exatamente a imagem clássica do giroscópio, aqui exata
porque os \omterm{def:b3:quantum-formalism:commutator}{comutadores} são lineares. Para o elétron: $\qty{28}{GHz/T}$;
para o próton (momento de $2.79$ magnétons nucleares): $\gamma_p/2\pi =
\qty{42.6}{MHz/T}$ --- a placa de identificação de
todo aparelho de \omterm{prop:b3:spin-two-level:rabi}{ressonância magnética}.
\end{proposition}

\begin{proof}
Aplique a \cref{prop:b3:quantum-formalism:evolution} com $\hat H =
-\gamma B_0\hat S_z$: os \omterm{def:b3:quantum-formalism:commutator}{comutadores} dão $\dd\langle\hat
S_x\rangle/\dd t = -\gamma B_0\langle\hat S_y\rangle$,
$\dd\langle\hat S_y\rangle/\dd t = +\gamma B_0\langle\hat
S_x\rangle$ e $\dd\langle\hat S_z\rangle/\dd t = 0$, que se
reúnem no produto vetorial enunciado.
\end{proof}

\begin{proposition}[Ressonância magnética]\label{prop:b3:spin-two-level:rabi}
\index{ressonância magnética}\index{oscilação de Rabi}
Acrescente um pequeno campo $B_1$ que gira no plano transversal com
frequência $\omega$. No referencial que gira com ele, $B_0$ fica
efetivamente reduzido a $(\omega_0 - \omega)/|\gamma|$; \emph{na
ressonância} ($\omega = \omega_0$) só $B_1$ sobrevive, e o \omterm{def:b3:spin-two-level:spin}{spin}
precessa em torno dele à \emph{frequência de Rabi} $\Omega_1 =
|\gamma|B_1$: partindo de $\ket\uparrow$, a probabilidade de ter virado
é
\[
\mathcal P_\downarrow(t) = \sin^2\Big(\frac{\Omega_1t}{2}\Big) .
\]
Um pulso de duração $\pi/\Omega_1$ (um ``pulso $\pi$'') inverte o \omterm{def:b3:spin-two-level:spin}{spin};
metade dele (um ``pulso $\pi/2$'') o deita no plano equatorial, onde
ele precessa a $\omega_0$ e, por indução de Faraday, \emph{transmite}
sua frequência a qualquer bobina próxima. Fora da ressonância, a
amplitude do vaivém colapsa como $\Omega_1^2/[\Omega_1^2 +
(\omega - \omega_0)^2]$: a resposta é agudamente seletiva --- uma
ressonância --- e é isso que torna os \omterm{def:b3:spin-two-level:spin}{spins} endereçáveis, espécie por
espécie e, com um gradiente de campo, lugar por lugar
(\cref{pb:b3:spin-two-level:1}).
\end{proposition}
\admitted

\begin{example}[Estrutura hiperfina e a linha de 21 cm]\label{ex:b3:spin-two-level:hyperfine}
\index{estrutura hiperfina}\index{linha de 21 cm}
No estado fundamental do hidrogênio, os \omterm{def:b3:spin-two-level:spin}{spins} do elétron e do próton
interagem pelos seus momentos magnéticos: os quatro estados de \omterm{def:b3:spin-two-level:spin}{spin} se
separam num tripleto e num singleto afastados por apenas $\Delta E =
\qty{5.9}{\micro eV}$ --- o desdobramento \emph{hiperfino}, com $\nu =
\qty{1420}{MHz}$ e $\lambda = \qty{21}{cm}$. A transição é
absurdamente lenta (uma virada a cada dez milhões de anos), mas a
Galáxia contém $10^{66}$ \omterm{thm:b3:hydrogen-atom:levels}{átomos de hidrogênio}: a \omterm{ex:b3:spin-two-level:hyperfine}{linha de 21 cm} é
brilhante o bastante para ter mapeado os braços espirais, o empenamento
do disco e --- por seus desvios Doppler --- as curvas de rotação planas
que argumentam a favor da matéria escura. A mesma física, no estado
fundamental do césio, separa níveis por exatamente
$\qty{9192631770}{Hz}$: desde 1967, a \emph{definição do segundo} é uma
virada hiperfina de \omterm{def:b3:spin-two-level:spin}{spin}, contada uma a uma
(\cref{exo:b3:spin-two-level:12}).
\end{example}

\begin{method}[Ofício dos dois níveis]\label{met:b3:spin-two-level:method}
(1) Escreva qualquer \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{hamiltoniana} de dois níveis em \omterm{def:b3:spin-two-level:spin}{matrizes de
Pauli}: $\hat H =
\epsilon_0\mathbb 1 + \vect h\cdot\hat{\vect\sigma}$ --- então o vetor de
Bloch precessa em torno de $\vect h$ a
$2|\vect h|/\hbar$; todo o resto é geometria. (2) Autoestados ao longo
de $\vect n$: use os meios ângulos. (3) Problemas de ressonância: passe
ao referencial girante; na ressonância, só $B_1$ resta. (4) Pulsos:
$\pi/2$ para iniciar um sinal de precessão, $\pi$ para inverter (e para
refocalizar --- o eco de \omterm{def:b3:spin-two-level:spin}{spin}). (5) Escalas de energia:
$\mu_{\text{B}}B$ para elétrons ($\qty{58}{\micro eV/T}$), três ordens
menos para núcleos --- radiofrequências, espectroscopia sem kelvins.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:1}
(a) Verifique $\sigma_x^2 = \sigma_y^2 = \sigma_z^2 = \mathbb 1$ e
$\sigma_x\sigma_y = \iu\sigma_z$. (b) Deduza $[\hat S_x, \hat S_y]
= \iu\hbar\hat S_z$: a álgebra do momento angular em dimensão dois.
(c) Mostre que $\hat S^2 = \tfrac34\hbar^2\,\mathbb 1$ e verifique
$j(j+1)$ para $j = \tfrac12$. (d) Mostre que toda matriz hermitiana
$2\times2$ é uma combinação real de $\mathbb 1$ e das três
$\sigma_i$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:2}
(a) Encontre os autoestados normalizados de $\sigma_x$ e de $\sigma_y$.
(b) Um \omterm{def:b3:spin-two-level:spin}{spin} preparado em $\ket{+x}$ é medido ao longo de $z$: quais as
probabilidades dos resultados? (c) E ao longo de $y$? (d) Confira as
duas com a imagem de Bloch (que ângulos $\theta, \varphi$?).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:3}
Três ímãs de Stern--Gerlach em cadeia, com eixos $z$, depois $\vect n$
no ângulo $\theta$ do plano $xz$, e $z$ de novo; o feixe $+$ é guardado
a cada vez. (a) Probabilidade de sobreviver ao segundo ímã. (b) De
sobreviver aos três. (c) Avalie em $\theta = 90^\circ$ e compare com o
\cref{ex:b3:spin-two-level:chained}. (d) Para que $\theta$ a
sobrevivência aos três ímãs é máxima, e que resposta do
\cref{exo:b3:quantum-formalism:5} isso ecoa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:4}
Calcule a separação dos níveis $2|\mu|B$ e a frequência de ressonância
num campo de \qty{1}{T} para (a) o elétron ($g = 2$); (b) o
próton ($\mu_p = 2.79\,\mu_{\text{N}}$, $\mu_{\text{N}} =
e\hbar/2m_{\text{p}}$); (c) compare os dois com o $k_{\text{B}}T$ a
\qty{300}{K}: quão polarizados estão os \omterm{def:b3:spin-two-level:spin}{spins} eletrônicos e nucleares à
temperatura ambiente? (d) Que faixas do espectro a RPE e a RMN
habitam, portanto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:5}
Deduza a \omterm{prop:b3:spin-two-level:larmor}{precessão de Larmor}: com $\hat H = -\gamma B_0\hat S_z$, (a)
calcule $\dd\langle\hat S_x\rangle/\dd t$, $\dd\langle\hat
S_y\rangle/\dd t$ e $\dd\langle\hat S_z\rangle/\dd t$ a partir dos
\omterm{def:b3:quantum-formalism:commutator}{comutadores}; (b) resolva para $\langle\hat{\vect S}\rangle(t)$ com o
\omterm{def:b3:spin-two-level:spin}{spin} inicial ao longo de $x$; (c) mostre que
$\|\langle\hat{\vect S}\rangle\|$ e o ângulo com $\vect B_0$ são
constantes; (d) por que a imagem dos \emph{autoestados de energia}
(dois níveis estacionários) e a imagem da \emph{precessão} (um vetor
que gira) descrevem a mesma física --- que estado é que precessa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:6}
(a) Verifique que o $\ket{+\vect n}$ da
\cref{prop:b3:spin-two-level:bloch} é o autovetor alegado. (b) Mostre
que estados antípodas são ortogonais. (c) Onde ficam, na esfera, os
autoestados de $\sigma_x$ e de $\sigma_y$? (d) Uma porta ``NÃO'' de
qubit leva $\ket\uparrow \leftrightarrow \ket\downarrow$: que rotação
da esfera é essa, e que pulso da
\cref{prop:b3:spin-two-level:rabi} a executa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:7}
O referencial girante, honestamente. Com $\hat H(t) = -\gamma(B_0\hat S_z
+ B_1[\hat S_x\cos\omega t - \hat S_y\sin\omega t])$: (a) explique por
que passar ao referencial que gira a $\omega$ em torno de $z$ substitui
$B_0$ por $B_0 - \omega/\gamma$ e congela $B_1$; (b) na ressonância,
descreva o movimento do vetor de Bloch no referencial girante; (c)
deduza a duração do pulso $\pi$ para um próton com $B_1 =
\qty{10}{\micro T}$; (d) esboce o que um observador no referencial do
laboratório vê durante esse pulso (duas rotações encaixadas).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:8}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:10}
Acoplamento spin--órbita, estimado. No referencial de repouso do

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:11}
Dois \omterm{def:b3:spin-two-level:spin}{spins} juntos. Para duas partículas de \omterm{def:b3:spin-two-level:spin}{spin} $\tfrac12$, o \omterm{def:b3:spin-two-level:spin}{spin}
total é $\hat{\vect S} = \hat{\vect S}_1 + \hat{\vect S}_2$. (a) Mostre
que os quatro estados produto se reorganizam num \emph{tripleto} ($S = 1$:
$\ket{\uparrow\uparrow}$, $(\ket{\uparrow\downarrow} +
\ket{\downarrow\uparrow})/\sqrt2$, $\ket{\downarrow\downarrow}$)
e num \emph{singleto} ($S = 0$: $(\ket{\uparrow\downarrow} -
\ket{\downarrow\uparrow})/\sqrt2$) --- verifique as contagens de $S_z$
e, para os dois estados do meio, o efeito de $\hat S^2$ (use $\hat
S^2 = \hat S_1^2 + \hat S_2^2 + 2\hat{\vect S}_1\cdot\hat{\vect
S}_2$ e $\hat{\vect S}_1\cdot\hat{\vect S}_2 = \tfrac12(\hat
S_{1+}\hat S_{2-} + \hat S_{1-}\hat S_{2+}) + \hat S_{1z}\hat
S_{2z}$). (b) Qual dos quatro é emaranhado --- não escrevível como
produto? (c) O par hiperfino do hidrogênio
(\cref{ex:b3:spin-two-level:hyperfine}) é exatamente esse
tripleto/singleto: qual está mais alto em energia, dado que a linha é
\emph{emitida} em 21 cm? (d) Os \omterm{def:b3:spin-two-level:spin}{spins} do singleto são perfeitamente
anticorrelacionados ao longo de \emph{todo} eixo: meça um deles em
qualquer $\vect n$ e o outro responde o oposto. Por que essa
correlação, por mais impressionante que seja, não transmite sinal
algum?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:12}
O relógio que define o segundo. O desdobramento hiperfino do estado

\end{exercise}

\section{Problema: Ver por dentro do corpo}

\begin{problem}[{Problema de fim de semana --- imagem por ressonância
magnética, do spin ao exame}]\label{pb:b3:spin-two-level:1}
Dois terços de um ser humano são água; toda molécula de água carrega
dois prótons, cada um deles um ímã de \omterm{def:b3:spin-two-level:spin}{spin} $\tfrac12$. Ponha uma pessoa
num campo forte, faça cócegas nos prótons na frequência de Larmor deles
e escute: isso é a imagem por \omterm{prop:b3:spin-two-level:rabi}{ressonância magnética}, física de \omterm{def:b3:spin-two-level:spin}{spin}
como medicina. Dados:
$\gamma_{\text{p}}/2\pi = \qty{42.58}{MHz/T}$; $B_0 = \qty{3.0}{T}$;
densidade de prótons do tecido $n \approx \qty{6.6e28}{m^{-3}}$;
$k_{\text{B}}T$ a \qty{310}{K} vale \qty{26.7}{meV}; $h =
\qty{4.14e-15}{eV.s}$.

\medskip\noindent\textbf{Parte I --- Os \omterm{def:b3:spin-two-level:spin}{spins} no ímã.}
\begin{enumerate}
  \item Calcule a frequência de Larmor a \qty{3.0}{T} e a energia do
        fóton $h\nu$ em eV.
  \item Calcule o desequilíbrio de população entre os dois níveis do
        próton, $\Delta n/n \approx h\nu/2k_{\text{B}}T$.
  \item Só esse excesso --- partes por milhão --- contribui para o
        sinal: quantos prótons ``úteis'' há por centímetro cúbico de
        tecido?
  \item Por que a polarização térmica é tão débil aqui, enquanto o
        feixe de Stern--Gerlach se dividia por completo? (O que cada
        experimento mede: \omterm{def:b3:quantum-formalism:observable}{autovalores} de um átomo isolado ou uma média
        térmica?)
  \item Dobrar $B_0$ faz o quê ao sinal (dois efeitos: a polarização e
        a fem induzida em frequência mais alta)? Por que os hospitais
        pagam por ímãs maiores?
  \item O ímã é supercondutor e fica sempre ligado: por que um cilindro
        de oxigênio de aço solto na sala é um projétil letal (que força
        do \cref{ex:b3:spin-two-level:stern-gerlach} age sobre ele)?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Pulsos e ecos.}
\begin{enumerate}[resume]
  \item Uma bobina transversal aplica $B_1 = \qty{10}{\micro T}$ em
        ressonância: calcule a frequência de Rabi e as durações dos
        pulsos $\pi/2$ e $\pi$.
  \item Depois do pulso $\pi/2$, o que a magnetização faz, e o que a
        lei de Faraday induz na bobina receptora (em que frequência)?
  \item O sinal transversal decai com a constante de tempo $T_2$ (os
        \omterm{def:b3:spin-two-level:spin}{spins} se desfasam nos campos uns dos outros e nas
        inomogeneidades locais); a magnetização longitudinal se
        recompõe com $T_1$ (a energia escoa para o tecido). Valores
        típicos: $T_1 \approx \qty{1}{s}$, $T_2 \approx \qty{0.1}{s}$.
        Por que o tempo de coerência nunca pode exceder muito o tempo
        de relaxação de energia (o que toda virada que troca energia
        faz com a fase)?
  \item O eco de \omterm{def:b3:spin-two-level:spin}{spin}: depois do pulso $\pi/2$ e de um atraso $\tau$,
        aplica-se um pulso $\pi$ e, em $2\tau$, os \omterm{def:b3:spin-two-level:spin}{spins} desfasados se
        realinham e o sinal volta. Explique o truque com corredores de
        pista mandados dar meia-volta.
  \item Que decaimento o eco desfaz --- o desfasamento por
        inomogeneidades \emph{estáticas} do campo ou o devido a campos
        moleculares flutuantes --- e por que só esse?
  \item Tecidos diferentes têm $T_1$ e $T_2$ diferentes (gordura:
        $T_1$ curto; água e líquidos: longo; muitos tumores: $T_2$
        mais longo que o do tecido hospedeiro). Numa frase: como o
        cronometrar da sequência de pulsos transforma tempos de
        relaxação em \emph{contraste} de imagem?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Do sinal à imagem.}
\begin{enumerate}[resume]
  \item Superponha um gradiente $G = \qty{40}{mT/m}$ ao longo de $z$: a
        frequência de Larmor passa a depender da posição. Calcule
        $\dd\nu/\dd z$ em kHz por milímetro.
  \item Um pulso de radiofrequência conformado que contenha só a banda $\nu_0 \pm
        \qty{1}{kHz}$ excita que fatia do corpo?
        (Seleção de fatia.)
  \item Durante a leitura, um gradiente ao longo de $x$ faz cada coluna
        da fatia transmitir a sua própria frequência: que operação
        matemática transforma o sinal temporal recebido num perfil
        espacial (lembre o ferramental de Fourier do volume do segundo
        ano)?
  \item Estime a informação total do exame: uma imagem de
        $256 \times 256$ com 12 bits, e por que adquiri-la linha a
        linha (um eco por linha, com tempo de repetição $\sim T_1$)
        faz um exame durar minutos.
  \item O paciente ouve batidas altas: o que está batendo mecanicamente
        (pense nas bobinas de gradiente que chaveiam dentro do campo de
        \qty{3}{T} --- que força)?
  \item A imagem por raios X contrasta a densidade eletrônica; a
        \omterm{prop:b3:spin-two-level:rabi}{ressonância magnética} contrasta a densidade de prótons \emph{e}
        a relaxação: por que a ressonância é a ferramenta de escolha
        para tecidos moles e para o cérebro, e que dose ionizante ela
        entrega?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- Os outros empregos do \omterm{def:b3:spin-two-level:spin}{spin}.}
\begin{enumerate}[resume]
  \item Os químicos fazem o mesmo experimento com resolução de
        \qty{10}{ppm}: as nuvens eletrônicas blindam ligeiramente cada
        núcleo, deslocando sua ressonância (o ``deslocamento
        químico''). Por que isso transforma a RMN numa impressão
        digital molecular?
  \item A \omterm{prop:b3:spin-two-level:rabi}{ressonância magnética} funcional mapeia o pensamento: a
        hemoglobina desoxigenada é paramagnética e encurta o $T_2^*$
        local. Trace a cadeia que vai da atividade neural ao brilho da
        imagem.
  \item A mesma física de Larmor com o momento do \emph{elétron} opera
        a \qty{84}{GHz} em \qty{3}{T}: por que a ressonância eletrônica
        é inútil dentro de um ser humano, mas preciosa para estudar
        radicais e defeitos?
  \item Compare a energia do fóton do item 1 com as energias típicas
        das ligações químicas: justifique a expressão ``não ionizante''
        e contraste com um fóton de raios X de \qty{60}{keV}.
  \item A polarização de dez partes por milhão pode ser aumentada: o
        gás xenônio polarizado por laser alcança polarização da ordem
        de um e é inalado para produzir imagens dos espaços aéreos do
        pulmão. Por qual fator, grosso modo, sobe o sinal por núcleo, e
        por que a baixa densidade do gás ainda torna o truque
        necessário?
  \item Um único \omterm{def:b3:spin-two-level:spin}{spin} de elétron numa armadilha de silício parecida com
        um transistor já é lido e acionado como qubit exatamente com os
        pulsos deste capítulo: nomeie as duas propriedades do \omterm{def:b3:spin-two-level:spin}{spin} (o
        tamanho do seu \omterm{def:b3:quantum-formalism:state}{espaço de Hilbert}; a fraqueza do seu acoplamento
        ao ruído) que o recomendam.
  \item Resuma o resultado central: uma precessão de \omterm{def:b3:spin-two-level:spin}{spin} a
        \qty{128}{MHz}, uma polarização térmica de dez partes por
        milhão, dois relógios de relaxação e três gradientes de campo
        bastam para fotografar em fatias um cérebro vivo --- as duas
        manchas de Stern e Gerlach, crescidas até virar um hospital.
\end{enumerate}
\end{problem}
