\chapter{O ensemble grande canônico}\label{ch:b3:grand-canonical}

Uma gota de água no ar úmido não cresce nem encolhe; o oxigênio se
liga à hemoglobina nos pulmões e se desliga nos músculos; uma bateria
empurra elétrons através de um telefone porque eles são ``mais caros''
em um eletrodo do que no outro. Em cada caso, a moeda que se equilibra
não é a temperatura, mas o \emph{\omterm{thm:b3:grand-canonical:distribution}{potencial químico}} $\mu$: o custo em
\omterm{prop:b3:canonical-ensemble:machine}{energia livre} de mais uma partícula. Este capítulo abre a segunda
válvula do reservatório --- partículas além de energia --- e constrói o
\omterm{thm:b3:grand-canonical:distribution}{ensemble grande canônico}, cuja recompensa é uma eficiência espantosa: a
ocupação de um único nível quântico, fermiônico ou bosônico, sai em
duas linhas, e com ela as fórmulas mestras do próximo capítulo. As
aplicações vão das isotermas de adsorção das máscaras de gás e dos
catalisadores automotivos à \omterm{prop:b3:grand-canonical:saha}{equação de Saha}, com a qual uma doutoranda
de Harvard leu, em 1925, as temperaturas das estrelas e descobriu,
contra toda expectativa, que o universo é feito de hidrogênio.

\section{As partículas entram na negociação}

\begin{theorem}[Distribuição grande canônica]\label{thm:b3:grand-canonical:distribution}
\index{ensemble grande canônico}\index{potencial químico}
Um sistema que troca energia \emph{e partículas} com um reservatório a
temperatura $T$ e \omterm{thm:b3:grand-canonical:distribution}{potencial químico} $\mu$ ocupa o seu estado $s$
(energia $E_s$, número de partículas $N_s$) com probabilidade
\[
\mathcal P_s = \frac{\eu^{-\beta(E_s - \mu N_s)}}{\Xi} , \qquad
\Xi = \sum_s\eu^{-\beta(E_s - \mu N_s)} ,
\]
a \emph{grande \omterm{thm:b3:canonical-ensemble:boltzmann}{função de partição}}. A partir de $\Xi$: $\langle N\rangle
= k_{\text{B}}T\,\partial\ln\Xi/\partial\mu$, e o \emph{grande
potencial} $\Phi = -k_{\text{B}}T\ln\Xi$ obedece a $\Phi = -PV$.
\end{theorem}

\begin{proof}[Demonstração parcial]
Como no \cref{thm:b3:canonical-ensemble:boltzmann}, expanda a entropia
do reservatório nas duas retiradas:
$S_{\text{res}}(E - E_s, N - N_s) \approx S_{\text{res}} - E_s/T +
\mu N_s/T$, usando $\partial S/\partial N = -\mu/T$
(\cref{thm:b3:microcanonical-ensemble:temperature}). As médias saem por
derivação; $\Phi = -PV$ fica admitido (segue da extensividade).
\end{proof}

\begin{definition}[O que $\mu$ significa]\label{def:b3:grand-canonical:mu}
\index{potencial químico!significado}
$\mu$ é o custo, em \omterm{prop:b3:canonical-ensemble:machine}{energia livre}, de acrescentar uma partícula a $T$ e
$V$ fixos. As partículas fluem espontaneamente de $\mu$ alto para $\mu$
baixo --- difusão, osmose, evaporação, reações químicas e correntes
elétricas são todos gradientes de $\mu$ relaxando. No equilíbrio entre
duas fases ou dois lugares quaisquer, $\mu$ é igual dos dois lados ---
a terceira igualdade (com $T$ e $P$) que rege a coexistência. Para
a classical ideal gas,
\[
\mu = k_{\text{B}}T\,\ln\!\big(n\lambda_T^3\big) < 0
\quad\text{(gás diluído)} :
\]
acrescentar uma partícula a um gás diluído \emph{abaixa} a \omterm{prop:b3:canonical-ensemble:machine}{energia
livre}, porque o ganho de entropia supera qualquer custo em energia ---
o sinal que move a evaporação para o ar seco. E uma bateria é uma
máquina de $\mu$: a tensão dela é a queda de \omterm{thm:b3:grand-canonical:distribution}{potencial químico} por
elétron entre os eletrodos, $U = \Delta\mu/e$ --- os volts são a taxa
de câmbio da química do elétron, e por isso as pilhas dão volts de um
só dígito.
\end{definition}

\section{Um nível de cada vez: o truque mestre}

\begin{theorem}[Ocupação de um único nível quântico]\label{thm:b3:grand-canonical:occupation}
\index{distribuição de Fermi--Dirac}\index{distribuição de Bose--Einstein}
Tome como ``o sistema'' um único nível de partícula única, de energia
$\epsilon$, em contato grande canônico com o gás de todos os outros.
\emph{Férmions} (ocupação $0$ ou $1$):
\[
\Xi = 1 + \eu^{-\beta(\epsilon - \mu)}
\;\Longrightarrow\;
\bar n_{\text{FD}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} + 1} .
\]
\emph{Bósons} (ocupação $0, 1, 2, \dots$):
\[
\Xi = \frac{1}{1 - \eu^{-\beta(\epsilon - \mu)}}
\;\Longrightarrow\;
\bar n_{\text{BE}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} - 1} ,
\]
exigindo $\mu < \epsilon_{\min}$. As duas se reduzem ao fator de
Boltzmann $\eu^{-\beta(\epsilon - \mu)}$ quando as ocupações são raras.
Estas duas fórmulas, obtidas em duas linhas cada, são toda a entrada da
estatística quântica: o próximo capítulo é a colheita delas.
\end{theorem}

\begin{proof}
Férmions: dois termos. Bósons: uma série geométrica, convergente apenas
para $\mu < \epsilon$. Nos dois casos, $\bar n = k_{\text{B}}T\,
\partial\ln\Xi/\partial\mu$ dá as formas enunciadas.
\end{proof}

\section{Adsorção: sítios para alugar}

\begin{proposition}[A isoterma de Langmuir]\label{prop:b3:grand-canonical:langmuir}
\index{isoterma de Langmuir}\index{adsorção}
Uma superfície oferece sítios independentes, cada um vazio ou com uma
molécula de gás de energia de ligação $-\epsilon_0$; o gás acima, à
pressão $P$, é o reservatório. Cada sítio é um sistema grande canônico
de dois estados, e a ocupação dele é
\[
\theta = \frac{1}{\eu^{\beta(-\epsilon_0 - \mu)} + 1}
 = \frac{P}{P + P_0(T)} , \qquad
P_0(T) = \frac{k_{\text{B}}T}{\lambda_T^3}\,
\eu^{-\epsilon_0/k_{\text{B}}T} ,
\]
usando o $\mu$ do gás ideal. A cobertura sobe linearmente a baixa
pressão e satura em uma monocamada --- a \emph{\omterm{prop:b3:grand-canonical:langmuir}{isoterma de Langmuir}},
a curva de trabalho dos catalisadores automotivos, do carvão ativado,
das máscaras de gás e da ligação do oxigênio ao sangue
(\cref{exo:b3:grand-canonical:6}). A pressão de cruzamento $P_0$ cai
\emph{exponencialmente} com a energia de ligação e sobe abruptamente com
a temperatura: aquecer uma superfície sacode os hóspedes dela para fora
--- e é por isso que assar vidraria sob vácuo é o que de fato a limpa.
\end{proposition}

\begin{proof}
A ocupação do sítio é a
\cref{thm:b3:grand-canonical:occupation} (na forma fermiônica: um sítio
guarda no máximo um) com $\epsilon = -\epsilon_0$; substitua
$\eu^{\beta\mu} = n\lambda_T^3 = P\lambda_T^3/k_{\text{B}}T$.
\end{proof}

\section{Equilíbrios de ionização: a equação de Saha}

\begin{proposition}[A equação de Saha]\label{prop:b3:grand-canonical:saha}
\index{equação de Saha}
Num gás quente, a reação $\text{H} \rightleftharpoons \text{p} +
\text{e}$ se equilibra quando $\mu_{\text{H}} = \mu_{\text{p}} +
\mu_{\text{e}}$. Escrever cada $\mu$ na forma do gás ideal (com o átomo
creditado pela ligação dele, $-E_{\text{I}}$) dá
\[
\frac{n_{\text{e}}\,n_{\text{p}}}{n_{\text{H}}}
 = \frac{1}{\lambda_{T,\text{e}}^3}\,
\eu^{-E_{\text{I}}/k_{\text{B}}T}
\qquad\Big(\lambda_{T,\text{e}} =
h/\sqrt{2\pi m_{\text{e}}k_{\text{B}}T}\Big)
\]
(a menos de fatores de degenerescência de spin da ordem da unidade).
De novo duas moedas em disputa: o fator de Boltzmann taxa a ionização em
$E_{\text{I}} = \qty{13.6}{eV}$, enquanto o prefator --- a
\emph{entropia do elétron libertado}, com o seu $\lambda_T^{-3}$ de
estados acessíveis --- a subsidia. O subsídio é tão grande às densidades
estelares que o hidrogênio se ioniza de modo apreciável já perto de
\qty{10000}{K}, onde $k_{\text{B}}T$ é apenas de
\qty{0.9}{eV}: os equilíbrios são disputados em \omterm{prop:b3:canonical-ensemble:machine}{energia livre}, não em
energia --- e os espectros das estrelas são o livro-caixa
(\cref{pb:b3:grand-canonical:1}).
\end{proposition}

\begin{proof}[Demonstração parcial]
Imponha $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$ com $\mu_i
= k_{\text{B}}T\ln(n_i\lambda_{T,i}^3) + \varepsilon_i$
($\varepsilon_{\text{H}} = -E_{\text{I}}$, os outros nulos) e
exponencie. Os comprimentos de onda térmicos do próton e do átomo quase
se cancelam (massas iguais); os fatores de degenerescência ficam admitidos.
\end{proof}

\begin{method}[Ofício grande canônico]\label{met:b3:grand-canonical:method}
(1) Qualquer coisa que troque partículas com um parceiro grande: dê ao
parceiro um $\mu$ e pese os estados por $\eu^{-\beta(E - \mu N)}$.
(2) Sítios ou níveis independentes fatorizam: resolva \emph{um} e
multiplique. (3) Para um reservatório de gás clássico, $\eu^{\beta\mu} =
n\lambda_T^3$ converte $\mu$ em pressão mensurável. (4) Reações e
mudanças de fase: equilibre os $\mu$, um por espécie, energias de
ligação incluídas --- espere que os prefatores de entropia desloquem os
equilíbrios para longe dos palpites ingênuos de Boltzmann. (5) Elétrons
em sólidos: o $\mu$ deles é o nível de Fermi, e diferenças de $\mu$ são
tensões --- a ponte para o
\cref{ch:b3:electrons-in-solids}.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:1}
(a) Duas caixas do mesmo gás à mesma $T$, mas com densidades
diferentes, são conectadas: para que lado fluem as partículas, na
linguagem de $\mu$? (b) Mostre, a partir do $\mu$ do gás ideal, que o
fluxo para em densidades iguais. (c) Água sob ar úmido: enuncie a
condição de equilíbrio em termos dos $\mu$. (d) Por que a roupa seca
mesmo abaixo de $100\,^\circ$C quando o ar está seco (compare os $\mu$,
não as temperaturas)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:2}
O \omterm{thm:b3:grand-canonical:distribution}{potencial químico} do ar. (a) Calcule $\lambda_T$ para o N$_2$
a \qty{300}{K}. (b) Calcule $n\lambda_T^3$ a \qty{1}{bar} e o
$\mu$ resultante, em eV. (c) Por que $\mu < 0$ não é paradoxo algum
(que termo de $F = E - TS$ domina quando se acrescenta uma molécula)? (d)
A que densidade o $\mu$ chegaria a zero, e o que anuncia
$n\lambda_T^3 \sim 1$
(\cref{exo:b3:grand-canonical:11})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:3}
Deduza a ocupação de Fermi--Dirac a partir do $\Xi$ de dois termos,
passo a passo; verifique os limites $\epsilon \ll \mu$, $\epsilon \gg
\mu$, e o valor em $\epsilon = \mu$. Onde esta função já apareceu
neste livro
(\cref{exo:b3:microcanonical-ensemble:8})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:4}
Volts são química. (a) Uma célula de lítio dá \qty{3.0}{V}:
exprima a diferença de \omterm{thm:b3:grand-canonical:distribution}{potencial químico} por elétron em eV e
por mol em kJ. (b) Por que todas as baterias do dia a dia têm um só
dígito de volts (o que fixa a escala das diferenças químicas de $\mu$)?
(c) Uma bateria de carro de chumbo-ácido empilha seis células: por que
empilhar em vez de procurar uma reação de \qty{12}{eV}? (d) Na
linguagem de $\mu$, o que faz uma bateria quando está ``descarregada''?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:5}
Bósons num nível. (a) Some a série geométrica de $\Xi$ e deduza
$\bar n_{\text{BE}}$. (b) Por que o $\mu$ tem de ficar abaixo do
nível mais baixo --- o que diverge em caso contrário, e o que a
divergência prenuncia (\cref{ch:b3:quantum-statistics})? (c)
Calcule $\bar n$ para $(\epsilon - \mu)/k_{\text{B}}T = 0.1$, $1$,
$3$ e compare com Boltzmann. (d) Os fótons têm $\mu = 0$: dê a
razão física (o número deles não se conserva --- as paredes
absorvem e emitem à vontade).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:6}
Langmuir na prática. (a) Pela isoterma, que pressão alcança
$\theta = 0.5$? E $0.9$? (b) A mioglobina liga O$_2$ num único sítio:
a saturação segue Langmuir na pressão parcial do oxigênio ---
por que se observa uma curva \emph{hiperbólica} (não sigmoidal) para a
mioglobina, mas um S cooperativo para a hemoglobina de quatro sítios?
(c) O monóxido de carbono se liga aos mesmos sítios com $\epsilon_0$
maior por $\approx\qty{0.13}{eV}$: a pressões parciais iguais e
\qty{310}{K}, calcule a razão de ocupação CO:O$_2$ e explique o
envenenamento por CO. (d) Por que funciona a oxigenoterapia
hiperbárica, na linguagem das isotermas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:7}
Flutuações de número. (a) Mostre que $\operatorname{Var}N =
k_{\text{B}}T\,\partial\langle N\rangle/\partial\mu$. (b) Para o
gás ideal clássico, deduza $\operatorname{Var}N = \langle
N\rangle$: Poisson. (c) Concilie com as flutuações de densidade do
\cref{exo:b3:microcanonical-ensemble:12}. (d) Para um único
nível fermiônico, calcule $\operatorname{Var}n = \bar n(1 - \bar
n)$: onde os números de férmions são mais silenciosos, e por quê?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:8}
Osmose a partir de $\mu$. Uma membrana deixa passar a água, mas não o
soluto (concentração $c$ por volume, diluído). (a) Igualando o $\mu$ da
água dos dois lados da membrana, mostre que o lado de água pura tem de
ficar sobrepressurizado em $\Pi = ck_{\text{B}}T$ (van 't Hoff) ---
aceite que o soluto dissolvido baixa o $\mu$ da água em
$-k_{\text{B}}T\,c/n_{\text{w}}$ por molécula de água. (b) Calcule
$\Pi$ para o soro fisiológico ($c \approx \qty{3e26}{m^{-3}}$
de partículas totais). (c) Hemácias em água pura: para que lado a água
flui, e com que resultado? (d) Por que as árvores não precisam de bombas
nem de milagres para erguer a seiva por dezenas de metros (gradientes
osmóticos e capilares de $\mu$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:9}
Equilíbrio num campo. Na gravidade, o equilíbrio de uma coluna de gás
exige que o \omterm{thm:b3:grand-canonical:distribution}{potencial químico} \emph{total} $\mu(h) =
k_{\text{B}}T\ln(n\lambda_T^3) + mgh$ seja uniforme. (a) Deduza daí
a lei barométrica em uma linha. (b) Generalize para uma centrífuga
de velocidade angular $\omega$ (potencial
$-\tfrac12 m\omega^2r^2$) e obtenha o perfil radial. (c)
As centrífugas de enriquecimento de urânio giram a $\omega r \sim
\qty{600}{m/s}$: calcule o enriquecimento de equilíbrio de $^{238}$UF$_6$
sobre $^{235}$UF$_6$ por estágio, $\exp[\Delta m\,\omega^2
r^2/2k_{\text{B}}T]$ com $\Delta m = \qty{5e-27}{kg}$ e $T =
\qty{320}{K}$. (d) Por que se encadeiam muitos estágios?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:10}
Saha, à mão. (a) Faça a demonstração pelo balanço de $\mu$ da
\cref{prop:b3:grand-canonical:saha}. (b) Defina a fração ionizada
$x$ à densidade total de hidrogênio $n$ e mostre que $x^2/(1 - x)
= \eu^{-E_{\text{I}}/k_{\text{B}}T}/n\lambda_{T,\text{e}}^3$. (c)
Fotosfera solar: $T = \qty{5800}{K}$, $n \approx
\qty{e23}{m^{-3}}$: mostre que a fração ionizada do hidrogênio é de
apenas $\sim\num{e-4}$ --- a superfície do Sol é neutra. (d) Em $T =
\qty{10000}{K}$ (mesmo $n$): recalcule e comente a inclinação.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:11}
Quando os gases ficam quânticos. A degenerescência começa em $n\lambda_T^3 \sim
1$. (a) Para o ar a \qty{300}{K}: quão longe do
limiar (\cref{exo:b3:grand-canonical:2})? (b) Para os elétrons de
condução do cobre ($n = \qty{8.5e28}{m^{-3}}$): ache a temperatura
abaixo da qual $n\lambda_T^3 > 1$ --- e conclua que os elétrons de um
metal são degenerados a \emph{qualquer} temperatura de laboratório.
(c) Para átomos de rubídio a $n = \qty{e20}{m^{-3}}$: a temperatura
limiar quântica (a escala do BEC do
\cref{ch:b3:quantum-statistics}). (d) Ordene os três e enuncie a
regra: partículas leves e densidades altas ficam quânticas primeiro.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:12}
O universo perde os pósitrons. No universo primitivo quente,
$\eu^+ + \eu^- \rightleftharpoons 2\gamma$ mantinha os pares em
equilíbrio com $\mu_{e^+} + \mu_{e^-} = 0$ (fótons: $\mu = 0$).
(a) Para um plasma simétrico, isso dá $\mu_{e^\pm} \approx 0$:
mostre que a densidade de pares é então $n_\pm \approx
\lambda_T^{-3}\eu^{-m_{\text{e}}c^2/k_{\text{B}}T}$ assim que
$k_{\text{B}}T \ll m_{\text{e}}c^2$. (b) Calcule a temperatura
em que $m_{\text{e}}c^2 = k_{\text{B}}T$ e a escala de tempo cósmica
correspondente (aceite $t \sim \qty{1}{s}$ a
$\qty{e10}{K}$). (c) Mostre que um resfriamento adicional modesto (um
fator de poucas unidades) faz a densidade de pares desabar por muitas
ordens de grandeza: os pósitrons se aniquilam e resta apenas o
minúsculo excesso primordial de elétrons. (d) Os fótons liberados
reaquecem o banho de radiação: que radiação relíquia carrega o recibo
(\cref{ch:b3:astrophysics})?
\end{exercise}

\section{Problema: lendo as temperaturas das estrelas}

\begin{problem}[{Problema de fim de semana --- Saha, Payne e do que
o universo é feito}]\label{pb:b3:grand-canonical:1}
Os espectros estelares foram catalogados ao longo da década de 1890 nas
classes O, B, A, F, G, K, M pela intensidade das linhas de absorção ---
com as linhas de Balmer do hidrogênio mais fortes na classe A e
misteriosamente fracas tanto nas estrelas mais quentes quanto nas mais
frias. Em 1925, Cecilia Payne aplicou a novíssima \omterm{prop:b3:grand-canonical:saha}{equação de Saha} e
mostrou que todo o zoológico era uma só substância a temperaturas
diferentes --- e que as estrelas são esmagadoramente hidrogênio, uma
conclusão tão herética que a obrigaram a suavizá-la na tese. Este
problema refaz o raciocínio dela.
Dados: a absorção de Balmer parte de $n = 2$, a $E_2 - E_1 =
\qty{10.2}{eV}$ acima do estado fundamental; $E_{\text{I}} =
\qty{13.6}{eV}$; densidades eletrônicas fotosféricas $n_{\text{e}}
\sim \qty{e20}{m^{-3}}$.

\medskip\noindent\textbf{Parte I --- Dois fatores em disputa.}
\begin{enumerate}
  \item A absorção de Balmer precisa de hidrogênio neutro \emph{em} $n =
        2$: escreva o fator de Boltzmann da população de $n = 2$
        (degenerescências: $g_2/g_1 = 4$).
  \item Avalie-o a $5000$, $10000$ e \qty{20000}{K}: para que
        lado esse fator empurra a intensidade da linha quando $T$ sobe?
  \item Agora o fator de Saha: quando $T$ sobe, o que acontece com a
        fração \emph{neutra} e, portanto, com as linhas de Balmer?
  \item Explique qualitativamente por que o produto --- excitado
        \emph{e} ainda neutro --- tem de ter um máximo a alguma
        temperatura intermediária.
  \item Usando a \omterm{prop:b3:grand-canonical:saha}{equação de Saha} com $n_{\text{e}} =
        \qty{e20}{m^{-3}}$, estime a temperatura de
        meia ionização do hidrogênio.
  \item Junte o máximo: perto de que temperatura as linhas de Balmer
        devem ser mais fortes, e que classe espectral
        (estrelas A, $\sim\qty{10000}{K}$) isso seleciona?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A sequência espectral
decifrada.}
\begin{enumerate}[resume]
  \item As estrelas M (\qty{3000}{K}) mostram bandas moleculares fortes
        (TiO) e hidrogênio fraco: explique as duas coisas com os dois fatores.
  \item As estrelas O (\qty{40000}{K}) mostram linhas de hélio e de novo
        hidrogênio fraco: explique (o que aconteceu com o hidrogênio, e
        por que o hélio, $E_{\text{I}} = \qty{24.6}{eV}$, tem agora a
        vez dele?).
  \item O Sol (G, \qty{5800}{K}) mostra linhas fortes de
        \emph{cálcio ionizado} ($E_{\text{I,Ca}} = \qty{6.1}{eV}$)
        e ainda assim hidrogênio fraco: mostre, com a lógica de Saha,
        por que o cálcio já está ionizado enquanto o hidrogênio ainda é
        neutro e está em sua maioria em $n = 1$.
  \item Antes de Payne, os astrônomos liam a intensidade da linha como
        abundância: o que as poderosas linhas de cálcio e as débeis
        linhas de Balmer do Sol os levaram a concluir sobre a composição dele?
  \item A ideia de Payne: corrigir a intensidade de cada linha pelo
        ``fator de visibilidade'' de Boltzmann--Saha dela. Quando ela o
        fez, que hierarquia de abundâncias verdadeiras emergiu?
  \item O resultado dela --- hidrogênio um milhão de vezes mais
        abundante que o cálcio, estrelas feitas sobretudo de hidrogênio
        e hélio --- por que foi resistido (o que sugeria a química
        centrada na Terra), e quando ela foi reabilitada?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Números sobre o máximo.}
\begin{enumerate}[resume]
  \item Calcule o fator de Boltzmann de $n = 2$ exatamente a
        \qty{9600}{K}.
  \item Com a estimativa de Saha do item 5, tome a fração neutra
        a $9600$, $15000$ e \qty{25000}{K} como sendo
        grosso modo $0.5$, $\num{e-3}$ e $\num{e-5}$: forme o produto
        (fração excitada $\times$ fração neutra) nas suas três
        temperaturas e localize o máximo dele.
  \item Por que o pico de Balmer é \emph{abrupto} do lado quente
        (qual exponencial vence) e \emph{gradual} do lado
        frio?
  \item A fotosfera de uma anã branca tem $n_{\text{e}}$ mil
        vezes maior: para que lado se move a temperatura de meia
        ionização, e por quê (Le Chatelier na linguagem de $\mu$)?
  \item Duas estrelas de temperatura idêntica, mas de pressões
        diferentes, mostram assim intensidades de linha diferentes: que
        propriedade estelar (gigante contra anã --- a gravidade
        superficial) isso permite aos espectroscopistas ler?
  \item Os levantamentos modernos fixam as temperaturas estelares em
        $\pm1\%$ a partir de razões de linhas: enuncie a cadeia
        postulado $\to$ Boltzmann $\to$ Saha $\to$ termômetro.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- O que isso resolveu.}
\begin{enumerate}[resume]
  \item A sequência OBAFGKM era um caos alfabético reordenado
        empiricamente: depois de Saha, que variável física
        isolada a ordena?
  \item Enuncie por que o \emph{mesmo} elemento pode dominar um
        espectro a uma temperatura e sumir a outra sem
        mudança alguma de abundância --- a lição central que Payne
        ensinou à astronomia.
  \item Hidrogênio em três quartos de toda a massa bariônica: cite dois
        outros pilares deste livro que repousam sobre essa abundância
        (a fusão estelar, \cref{ch:b3:astrophysics}; o céu de 21 cm,
        \cref{ch:b3:spin-two-level}).
  \item Otto Struve chamou a tese de Payne de ``a mais brilhante
        tese de doutorado já escrita em astronomia'': a partir deste
        problema, justifique o elogio em uma frase.
  \item A mesma física de Saha, rodada ao contrário em escala cósmica,
        prevê a temperatura em que o hidrogênio de todo o universo se
        combinou pela primeira vez ($\sim\qty{3000}{K}$, à densidade
        cósmica bem mais baixa): nomeie o evento e a relíquia dele
        (\cref{ch:b3:astrophysics}).
  \item Os equilíbrios de ionização também fazem funcionar as lâmpadas
        fluorescentes e o processamento por plasma: enuncie em uma frase
        por que um estado ionizado ao estilo de \qty{10000}{K} pode
        existir num tubo cujo vidro fica frio (que dois subsistemas
        deixam de se equilibrar?).
  \item Resuma o resultado nomeado: uma única condição de equilíbrio,
        $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$, transforma
        o alfabeto espectral num termômetro que vai de
        $3000$ a \qty{40000}{K}, põe o pico de Balmer nas estrelas
        A e revela um universo feito de hidrogênio.
\end{enumerate}
\end{problem}
