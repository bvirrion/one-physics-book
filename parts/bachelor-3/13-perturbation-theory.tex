\chapter{Perturbation Theory}\label{ch:b3:perturbation-theory}

Only a handful of quantum problems dissolve exactly: the box, the
oscillator, hydrogen, the two-level system. Everything else --- an
atom in a field, a molecule's stiffening bond, a level nudged by a
neighbour --- is one of those solvable skeletons wearing a small
extra term. \omterm{thm:b3:perturbation-theory:corrections}{Perturbation theory} is the art of treating the extra
term as what it is: a correction, computed order by order in its
smallness. Its first-order rule is a one-line average; its
second-order rule explains why levels \emph{repel}, why every atom is
polarisable, and why van der Waals attraction is universal; its
degenerate variant handles the delicate cases where near-equal
levels reorganise entirely; and its time-dependent form --- Fermi's
golden rule --- is the equation behind every absorption line, decay
rate, and scattering cross-section measured in a laboratory. As a
running reward, the chapter ends by computing the colour of the sky.

\section{Small shifts: the non-degenerate theory}

\begin{theorem}[Perturbative corrections]\label{thm:b3:perturbation-theory:corrections}
\index{perturbation theory}
Let $\hat H = \hat H_0 + \hat V$ with $\hat H_0$ solved
($\hat H_0\ket n = E_n\ket n$, spectrum non-degenerate) and $\hat V$
small. Then, order by order in $\hat V$,
\[
E_n' = E_n + \bra n\hat V\ket n
 + \sum_{m \neq n}\frac{|\bra m\hat V\ket n|^2}{E_n - E_m}
 + \cdots
\]
and the state acquires admixtures $\ket{n'} = \ket n + \sum_{m\neq
n}\dfrac{\bra m\hat V\ket n}{E_n - E_m}\,\ket m + \cdots$
\emph{First order}: the shift is just the average of the
perturbation in the unperturbed state. \emph{Second order}:
couplings to other levels push $E_n$ \emph{away} from them ---
each $m$ above pushes down, each below pushes up --- so the ground
state's second-order shift is always negative. Validity: the mixing
fractions $|\bra m\hat V\ket n/(E_n - E_m)|$ must be small.
\end{theorem}

\begin{proof}[Partial proof]
Expand $E' = E + \epsilon_1 + \epsilon_2 + \cdots$ and $\ket{\psi} =
\ket n + \ket{\psi_1} + \cdots$ in powers of $\hat V$, insert into
$(\hat H_0 + \hat V)\ket\psi = E'\ket\psi$, and match orders.
Projecting the first-order equation on $\bra n$ gives $\epsilon_1 =
\bra n\hat V\ket n$; projecting on $\bra m$ ($m \neq n$) gives the
admixture coefficients; carrying those into the second-order
equation and projecting on $\bra n$ again gives $\epsilon_2$. The
sign statement for the ground state: every denominator $E_0 - E_m$
is negative while the numerators are squares.
\end{proof}

\begin{proposition}[A perturbation with an exact answer]\label{prop:b3:perturbation-theory:oscillator-field}
An oscillator of charge $q$ in a uniform field $\mathcal E$ has
$\hat V = -q\mathcal E\hat x$. Completing the square solves it
exactly: the potential is the same parabola, shifted, with all
levels lowered by $q^2\mathcal E^2/2m\omega^2$. \omterm{thm:b3:perturbation-theory:corrections}{Perturbation theory}
must agree --- and does: first order vanishes ($\langle\hat
x\rangle_n = 0$), and the second-order sum, in which $\hat x$
couples $n$ only to $n \pm 1$, gives exactly
$-q^2\mathcal E^2/2m\omega^2$ for every level
(\cref{exo:b3:perturbation-theory:2}). One solvable case, checked
both ways, is the best calibration a method can have.
\end{proposition}

\begin{proof}
The two matrix elements and their denominators give
\[
\frac{|\bra{n+1}\hat x\ket{n}|^2}{-\hbar\omega}
 + \frac{|\bra{n-1}\hat x\ket n|^2}{+\hbar\omega}
 = \frac{\hbar}{2m\omega}\,\frac{-(n+1) + n}{\hbar\omega}
 = -\frac{1}{2m\omega^2} ;
\]
multiply by $q^2\mathcal E^2$.
\end{proof}

\section{When levels are degenerate}

\begin{theorem}[Degenerate perturbation theory]\label{thm:b3:perturbation-theory:degenerate}
\index{degenerate perturbation theory}
If $E_n$ is degenerate, the formula above divides by zero --- the
perturbation may reorganise the degenerate states completely, and
the cure is to let it: \emph{diagonalise $\hat V$ restricted to the
degenerate subspace}. Its \omterm{def:b3:quantum-formalism:observable}{eigenvalues} are the first-order shifts;
its eigenvectors are the correct zeroth-order states, the
combinations the perturbation itself selects.
\end{theorem}
\admitted

\begin{example}[The linear Stark effect]\label{ex:b3:perturbation-theory:stark}
\index{Stark effect}
Hydrogen's $n = 2$ level holds the degenerate $2s$ and $2p$ states.
In a field $\mathcal E\vect e_z$, $\hat V = e\mathcal E\hat z$
couples $2s$ to $2p_0$ with $\bra{2s}\hat z\ket{2p_0} = -3a_0$:
diagonalising the $2\times2$ block gives shifts
\[
\Delta E = \pm\,3e a_0\mathcal E ,
\]
\emph{linear} in the field --- while the non-degenerate ground state
shifts only quadratically. At $\mathcal E = \qty{e7}{V/m}$: $\pm
\qty{1.6}{meV}$, an easily resolved splitting. The states doing the
splitting are $(\ket{2s} \mp \ket{2p_0})/\sqrt2$ --- hybrids with a
permanent electric dipole, possible only because degeneracy left
the atom free to polarise at zeroth order. (The same $s$--$p$
hybridisation, driven by neighbours instead of a field, is the
carbon chemistry of every organic molecule.)
\end{example}

\section{Transitions: the golden rule}

\begin{theorem}[Time-dependent perturbations]\label{thm:b3:perturbation-theory:golden}
\index{Fermi's golden rule}
Switch on $\hat V(t) = \hat V\cos\omega t$ at $t = 0$. To first
order, the probability of finding the system in $\ket f$ at time
$t$, having started in $\ket i$, is
\[
\mathcal P_{i\to f}(t) = \frac{|\bra f\hat V\ket i|^2}{4\hbar^2}\;
\frac{\sin^2\!\big[(\omega_{fi} - \omega)\,t/2\big]}
{\big[(\omega_{fi} - \omega)/2\big]^2} ,
\qquad \omega_{fi} = \frac{E_f - E_i}{\hbar} :
\]
a resonance sharply peaked at $\hbar\omega = E_f - E_i$ ---
absorption when $E_f > E_i$, \emph{stimulated emission} for the
companion term with $E_f < E_i$. When the final states form a
continuum of density $\rho(E)$, the peak's growth becomes a constant
\emph{rate}:
\[
\Gamma_{i\to f} = \frac{2\pi}{\hbar}\,
|\bra f\hat V\ket i|^2\,\rho(E_f)
\]
--- \emph{Fermi's golden rule}, the master formula of decay rates,
absorption coefficients and cross-sections.
\end{theorem}

\begin{proof}[Partial proof]
First-order amplitude: $c_f(t) = -\tfrac{\iu}{\hbar}\int_0^t\bra
f\hat V(t')\ket i\,\eu^{\iu\omega_{fi}t'}\dd t'$; keeping the
near-resonant exponential and integrating gives the stated
$\sin^2$ form. For the continuum: as a function of the detuning
$x$, the resonance factor has height $t^2$ and width $\sim 2\pi/t$
--- area $2\pi t$ --- so summing over a smooth continuum replaces
it by $2\pi t\,\rho$: probability linear in $t$, i.e.\ a rate.
\end{proof}

\begin{example}[What the golden rule runs]\label{ex:b3:perturbation-theory:applications}
Selection rules: a transition happens only if the matrix element
$\bra f\hat V\ket i$ survives --- for light, $\hat V \propto \hat
z$, whose parity and angular momentum kill everything except
$\Delta l = \pm1$, $\Delta m = 0, \pm1$: the rules invoked since
\cref{ch:b3:quantum-angular-momentum}, now theorems. Rates: the
hydrogen $2p$ state's spontaneous decay computes to
\qty{1.6}{ns} (\cref{exo:b3:perturbation-theory:8}); ammonia's
maser transition, with its microwave $\omega^3$, takes months ---
the $\omega^3$ in radiated rates is why radio lines
(\cref{ex:b3:spin-two-level:hyperfine}) live for megayears while
ultraviolet lines flash in nanoseconds. Absorption spectra,
photoionisation, nuclear beta decay
(\cref{ch:b3:nuclear-physics}), scattering cross-sections
(\cref{ch:b3:scattering-theory}): all are this one formula with
different matrix elements and state counts.
\end{example}

\begin{method}[Choosing the right perturbative tool]\label{met:b3:perturbation-theory:method}
(1) Static question, non-degenerate level: average $\hat V$ (first
order); if that vanishes by symmetry, second order --- expect
level repulsion. (2) Degenerate level: diagonalise $\hat V$ in the
degenerate block first; the symmetry-adapted combinations do the
splitting. (3) Transition rates: golden rule --- one matrix
element, one \omterm{prop:b3:schrodinger-three-dimensions:counting}{density of states}; check selection rules before
computing anything. (4) Always test the expansion: mixing fraction
$|V_{mn}/(E_n - E_m)| \ll 1$, else diagonalise exactly (two-level
formula). (5) Exact solvable limits (oscillator in a field) are
free calibrations: use them.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:1}
A particle in a box $[0, a]$ feels the extra potential $V(x) =
\lambda x/a$. (a) Compute the first-order shift of every level
(use $\langle x\rangle = a/2$ in any box state --- justify it).
(b) Why is the shift the same for all levels here? (c) What is the
first-order change of the \emph{spacings}, and why could a
measurement of transition frequencies miss this perturbation
entirely? (d) When does first order stop being trustworthy?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:2}
Carry out the second-order computation of
\cref{prop:b3:perturbation-theory:oscillator-field} in full: matrix
elements of $\hat x$, the two terms, the cancellation of $n$, and
the exact agreement with completing the square. Why does the
\emph{state}'s first-order correction not vanish although the
energy's does?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:3}
The two-level \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} $\begin{pmatrix}E_1 & v\\ v &
E_2\end{pmatrix}$, $E_1 < E_2$, $v$ real. (a) Find the exact
\omterm{def:b3:quantum-formalism:observable}{eigenvalues}. (b) Expand for $|v| \ll E_2 - E_1$ and identify the
second-order formula. (c) Show the levels repel: the gap always
exceeds $E_2 - E_1$. (d) At $E_1 = E_2$: what does \omterm{thm:b3:perturbation-theory:corrections}{perturbation
theory} give, what does the exact answer give, and which theorem of
the chapter reconciles them?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:4}
For \cref{thm:b3:perturbation-theory:corrections}: (a) show the
first-order state correction is orthogonal to $\ket n$; (b) show
the mixing coefficient of $\ket m$ is $V_{mn}/(E_n - E_m)$ and
state the validity criterion; (c) a level $\qty{1}{eV}$ from its
nearest neighbour is coupled to it by \qty{0.1}{eV}: estimate the
admixture probability; (d) same with a \qty{10}{meV} gap ---
which tool should replace the expansion?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:5}
Anharmonic bonds. Add $\hat V = \beta\hat x^3 + \gamma\hat x^4$ to
the oscillator. (a) Show the $x^3$ term shifts no level at first
order (parity). (b) Show the $x^4$ term gives $\Delta E_n =
3\gamma(\hbar/2m\omega)^2(2n^2 + 2n + 1)$ (expand $\hat x^4$ in
\omterm{def:b3:harmonic-oscillator:ladder}{ladder operators}; only the ``balanced'' terms survive). (c) For a real bond the effective $\gamma$ is negative (the well
softens outward): show the level spacing then \emph{shrinks} with
$n$.
(d) Which observed feature of molecular spectra
(\cref{exo:b3:harmonic-oscillator:4}) does this reproduce?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:6}
In the square two-dimensional box, the degenerate pair $\ket{1,2},
\ket{2,1}$ is perturbed by $\hat V = \lambda\,xy$. (a) Explain why
non-degenerate theory fails. (b) Compute the $2\times2$ matrix of
$\hat V$ in the pair (the integrals factorise; $\int_0^a
x\sin(\pi x/a)\sin(2\pi x/a)\dd x = -8a^2/9\pi^2$). (c) Find the
splittings and the correct zeroth-order states. (d) Compare with
the symmetry analysis of \cref{exo:b3:quantum-formalism:9}: which
answer did symmetry give for free?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:7}
\omterm{ex:b3:perturbation-theory:stark}{Stark effects}, linear and quadratic. (a) Using
\cref{ex:b3:perturbation-theory:stark}, compute the $n = 2$
splitting at $\mathcal E = \qty{2.5e6}{V/m}$. (b) Why does the
ground state show \emph{no} linear shift (two reasons: parity, and
no degenerate partner)? (c) Its quadratic shift defines the
polarisability, $\Delta E = -\tfrac12\alpha\mathcal E^2$: estimate
$\alpha \sim e^2a_0^2/E_{\text{I}}$ and evaluate in units of
$a_0^3$ (the exact answer is $4.5\,a_0^3 \times
4\pi\varepsilon_0$). (d) Which everyday property of matter ---
its dielectric response --- did you just compute the atomic seed
of?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:8}
Spontaneous emission (with one admitted input): an excited state
decays at $A = \omega^3|d_{fi}|^2/3\pi\varepsilon_0\hbar c^3$,
where $d_{fi} = e\bra f\hat z\ket i$-type matrix elements set the
dipole. (a) For hydrogen $2p \to 1s$: with $|d| \approx 0.74\,
ea_0$ and $\lambda = \qty{121.6}{nm}$, compute $A$ and the
lifetime. (b) Scale to the sodium D line. (c) Scale to the
hyperfine 21 cm transition ($|d| \to$ a magnetic moment, rate
down by another $\alpha^2$-ish factor; accept $A \approx
\qty{2.9e-15}{s^{-1}}$): check the ``ten million years''. (d)
From the $\omega^3$: why are ultraviolet lines fast, radio lines
eternal, and why did that make the \omterm{ex:b3:spin-two-level:hyperfine}{21 cm line}
\emph{predictable} but almost undetectable in a laboratory?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:9}
Selection rules as integrals. (a) Show $\bra{100}\hat z\ket{200} =
0$ by parity. (b) Show $\bra{100}\hat z\ket{21m} = 0$ unless $m =
0$ (the $\varphi$ integral). (c) State the resulting rules $\Delta
l = \pm1$, $\Delta m = 0, \pm1$ and their physical reading (the
photon's spin). (d) The $2s$ state can reach $1s$ by no dipole
route: what does that predict for its lifetime, and what
``forbidden'' two-photon path nature actually takes?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:10}
Polarisability, honestly. The second-order shift of hydrogen's
ground state in a field is $\Delta E = -e^2\mathcal E^2\sum_{m\neq
0}|\bra m\hat z\ket 0|^2/(E_m - E_0)$. (a) Bound the sum by
replacing every denominator by the smallest gap $E_2 - E_1 =
\tfrac34 E_{\text{I}}$ and using the closure relation $\sum_m|\bra

\le \tfrac{16}3\,a_0^3$ (in Gaussian-style units of
$4\pi\varepsilon_0$). (b) Compare with the exact $\tfrac92 a_0^3$.
(c) From $\alpha$, estimate the refractive index of hydrogen gas
at atmospheric density ($n - 1 \approx N\alpha/2\varepsilon_0
\cdot e^2\dots$ --- use $n - 1 = N\alpha_{\text{SI}}/2
\varepsilon_0$) and compare with the measured $\num{1.3e-4}$.
(d) State in one sentence the chain atom $\to$ polarisability
$\to$ refraction that Part II of the weekend problem will ride.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:11}
Avoided crossings and slow sweeps. A two-level system has bare
energies $\pm\lambda t$ crossing at $t = 0$, coupled by $v$. (a)
Sketch the exact levels against $t$: an avoided crossing of gap

in? (c) The Landau--Zener criterion compares the sweep time
across the gap region, $\sim v/\lambda$, with the internal clock
$\hbar/v$: give the condition for adiabatic following. (d) Two
uses: a qubit's state transferred by a slow chirp; an atomic
collision hopping charge between nuclei --- explain one of them in
two sentences.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:12}
Second order is a pessimist, and other theorems. (a) Prove that

\cref{exo:b3:harmonic-oscillator:10}) why van der Waals forces are

\end{exercise}

\section{Problem: The colour of the sky}

\begin{problem}[{Weekend problem --- Rayleigh scattering from
perturbed atoms}]\label{pb:b3:perturbation-theory:1}
Why is the day sky blue, the sunset red, and the cloud white? The
complete answer is a chain through this chapter: a light wave
perturbs each air molecule; the induced dipole re-radiates (the
dipole radiation of the Year 2 volume); and interference decides
what survives. Data: molecular polarisability of air
$\alpha/4\pi\varepsilon_0 \approx \qty{2.2e-30}{m^3}$; number
density $N = \qty{2.5e25}{m^{-3}}$; visible range
$450$--\qty{700}{nm}.

\medskip\noindent\textbf{Part I --- The perturbed molecule.}
\begin{enumerate}
  \item A static field $\mathcal E$ shifts a molecule's ground
        state by $-\tfrac12\alpha\mathcal E^2$: connect this
        statement to second-order \omterm{thm:b3:perturbation-theory:corrections}{perturbation theory} (which
        formula, and why negative?).
  \item The same $\alpha$ gives the induced dipole $p =
        \alpha\mathcal E$: for sunlight's field $\mathcal E \sim
        \qty{800}{V/m}$, compute the induced dipole of one
        nitrogen molecule and compare it with $ea_0$.
  \item Why may the optical field (frequency $\num{5e14}$ Hz) be
        treated with the \emph{static} polarisability for
        molecules whose electronic transitions lie in the
        ultraviolet? (Compare $\hbar\omega$ with the gap; this is
        the far-off-resonance limit of the golden-rule
        denominator.)
  \item An oscillating dipole $p_0\cos\omega t$ radiates the
        average power $P = p_0^2\omega^4/12\pi\varepsilon_0c^3$
        (Year 2 volume). Combining with $p_0 = \alpha\mathcal
        E_0$: how does the scattered power depend on $\omega$ at
        fixed illumination?
  \item Compute the ratio of scattered powers at \qty{450}{nm}
        and \qty{700}{nm}.
  \item In one sentence: why is the sky blue and not violet
        (two ingredients: the Sun's spectrum falling toward the
        violet, and the eye's sensitivity)?
\end{enumerate}

\medskip\noindent\textbf{Part II --- From one molecule to the
sky.}
\begin{enumerate}[resume]
  \item The scattering cross-section works out to $\sigma =
        \dfrac{8\pi}{3}\Big(\dfrac{\alpha}{4\pi\varepsilon_0}
        \Big)^2\dfrac{\omega^4}{c^4}$: verify its dimensions and
        evaluate it at \qty{550}{nm}.
  \item The attenuation length is $\ell = 1/N\sigma$: evaluate it
        at \qty{550}{nm} and compare with the thickness of the
        atmosphere ($\sim\qty{8}{km}$ equivalent at sea-level
        density).
  \item At \qty{450}{nm} versus \qty{700}{nm}, what fraction of a
        vertical sunbeam is scattered out? (Use $1 -
        \eu^{-L/\ell}$ with $L = \qty{8}{km}$.)
  \item At sunset the path is thirty times longer: recompute the
        blue's survival and explain the colour of the low Sun ---
        and of the light that reddens the Moon in a lunar
        eclipse.
  \item Why is the sky's light \emph{polarised} at $90^\circ$
        from the Sun (recall the dipole radiation pattern: no
        emission along the dipole's axis)?
  \item Bees and Vikings allegedly navigated by this
        polarisation: what does an overcast day do to it, and
        why?
\end{enumerate}

\medskip\noindent\textbf{Part III --- Why the sky is not
brighter.}
\begin{enumerate}[resume]
  \item A paradox: in a perfectly uniform medium, the wavelets
        from neighbouring volume elements interfere destructively
        sideways, and \emph{no} light would scatter at all (only
        refraction survives). What actually breaks the uniformity
        of air --- what is randomly distributed?
  \item Density fluctuations of an ideal gas make the scattered
        intensities of independent molecules \emph{add}: state
        why independence (random positions, wavelength-scale
        separations) kills the destructive interference.
  \item From your $\ell$ at \qty{550}{nm}: what fraction of
        sunlight does one clear-day atmosphere scatter --- does
        the number match the everyday brightness of the sky
        against the Sun?
  \item A cloud droplet (\qty{10}{\micro m}) holds $\sim10^{12}$
        molecules within a wavelength: their wavelets add
        \emph{coherently}. How does the scattered power then
        scale with molecule number, and why does the $\omega^4$
        colour selection disappear (all wavelengths scatter
        strongly): what colour is the cloud?
  \item Milk, fog, and white paint are white for the same
        reason: state the general rule --- scatterers small
        against $\lambda$ colour the light, scatterers large
        against $\lambda$ whiten it.
  \item Critical opalescence: near a liquid's critical point
        (the phase-change chapters of the Year 1 volume, and
        \cref{ch:b3:phase-transitions} ahead), density fluctuations grow to optical scales and
        the fluid turns milky: connect to item 13.
\end{enumerate}

\medskip\noindent\textbf{Part IV --- The chain, completed.}
\begin{enumerate}[resume]
  \item The forward-scattered wavelets do \emph{not} cancel: they
        interfere with the beam and slow its phase --- the
        refractive index. Using $n - 1 = N\alpha/2\varepsilon_0$
        (with $\alpha$ in SI), evaluate $n - 1$ for air and
        compare with the measured \num{2.9e-4}.
  \item One constant $\alpha$, computed by second-order
        \omterm{thm:b3:perturbation-theory:corrections}{perturbation theory}, thus fixes three phenomena at once:
        name them (shift in a static field; the sky's blue; the
        bending of light in air).
  \item Ozone absorption, aerosols and multiple scattering
        complicate real skies: name one observable each
        (twilight's colours; milky horizon; the sky's residual
        brightness at the zenith after sunset).
  \item Why does the Moon, airless, carry a \emph{black} daytime
        sky --- and what did every photograph from its surface
        thereby confirm about the origin of ours?
  \item Mars's daytime sky is butterscotch and its sunsets are
        \emph{blue}: its thin air scatters little, and suspended
        micrometre dust grains do the work instead. Using the rule
        of item 18, explain how dust reverses the colour logic.
  \item For X-rays, $\hbar\omega$ far exceeds every molecular
        transition: the electrons respond as if free and the
        scattering (Thomson) loses its $\omega^4$. What happens to
        the ``sky-blue'' mechanism in that regime --- and why is
        that flat response exactly what makes X-rays clean probes
        of electron density?
  \item Summarise the named result: a $\qty{2.2e-30}{m^3}$
        polarisability, squared and multiplied by $\omega^4$,
        gives a \qty{60}{km}-scale scattering length in the green
        --- long enough that the noon sky is gentle, short enough
        that sunsets redden and the heavens are blue, with the
        polarisation pattern of a forest of driven dipoles.
\end{enumerate}
\end{problem}
