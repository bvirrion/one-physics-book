\chapter{The Hydrogen Atom}\label{ch:b3:hydrogen-atom}

Nine-tenths of the atoms in the universe are hydrogen: one proton
holding one electron, the only atom physics can solve exactly --- and
the key that opened every other. The old quantum theory of
\cref{ch:b3:hamiltonian-mechanics} guessed its energies; the
machinery is now in place to \emph{derive} them: the Coulomb
potential enters the \omterm{prop:b3:schrodinger-three-dimensions:swave}{radial equation} of the last chapter, and out
come the levels $-E_{\text{I}}/n^2$, the \omterm{thm:b3:hydrogen-atom:levels}{Bohr radius}, the shells and
subshells of chemistry's periodic table, and the spectral series
astronomers read in every nebula. The same solution, rescaled,
describes ionised helium, muonic atoms, positronium, and the
electron--hole ``atoms'' inside semiconductors; stretched to $n
\approx 100$ it describes Rydberg atoms half a micrometre across,
whose radio whispers map the Galaxy's ionised clouds and whose
interactions now drive quantum computers.

% exactly; emphasis on scaling laws and applications.

\section{The Coulomb problem solved}

\begin{theorem}[Levels of hydrogen]\label{thm:b3:hydrogen-atom:levels}
\index{hydrogen atom}\index{principal quantum number}\index{Bohr radius}
For $V(r) = -e^2/4\pi\varepsilon_0 r$ (write $k = e^2/4\pi
\varepsilon_0$), the bound states are labelled $(n, l, m)$ with
\[
E_n = -\frac{E_{\text{I}}}{n^2} , \qquad
E_{\text{I}} = \frac{m_{\text{e}}k^2}{2\hbar^2} = \qty{13.6}{eV} ,
\qquad n = 1, 2, 3, \dots
\]
and, for each $n$, the orbital numbers $l = 0, 1, \dots, n-1$ and
$m = -l, \dots, l$. The natural length is the \omterm{thm:b3:hydrogen-atom:levels}{Bohr radius} $a_0 =
\hbar^2/m_{\text{e}}k = \qty{52.9}{pm}$; the ground state is
\[
\psi_{100} = \frac{1}{\sqrt{\pi a_0^3}}\;\eu^{-r/a_0} .
\]
The energy depends on $n$ \emph{alone}: counting the $m$'s and
$l$'s, level $n$ is $n^2$-fold degenerate (doubled by spin,
\cref{ch:b3:spin-two-level}) --- far beyond the $(2l+1)$-fold
degeneracy that isotropy explains.
\end{theorem}

\begin{proof}[Partial proof]
For the ground state, try $u = Cr\,\eu^{-r/a}$ in the \omterm{prop:b3:schrodinger-three-dimensions:swave}{radial
equation} with $l = 0$: $u'' = (1/a^2 - 2/ar)u$, so
$-\tfrac{\hbar^2}{2m_{\text{e}}}u'' - \tfrac kr u = Eu$ requires
$\hbar^2/m_{\text{e}}a = k$ (matching the $1/r$ terms) --- which is
$a = a_0$ --- and $E = -\hbar^2/2m_{\text{e}}a_0^2 = -E_{\text{I}}$.
The general solution (Laguerre polynomials, $n - l - 1$ radial
nodes) is admitted; each step of the construction is elementary but
long. The ``accidental'' $l$-degeneracy mirrors a classical secret
of the pure $1/r$ force --- Kepler ellipses do not precess, and an
extra conserved vector (Laplace--Runge--Lenz) points along the fixed
major axis; its quantum version is what ties different $l$ to one
energy (\cref{exo:b3:hydrogen-atom:12}).
\end{proof}

\section{Where the electron is}

\begin{proposition}[Radial distributions]\label{prop:b3:hydrogen-atom:radial}
\index{radial probability}
The probability of finding the electron between $r$ and $r + \dd r$
is $P(r)\,\dd r$ with $P(r) = |u(r)|^2$, the square of the
normalised radial function of \cref{thm:b3:quantum-angular-momentum:radial}. For the lowest states:
$P_{1s} \propto r^2\eu^{-2r/a_0}$, peaking at exactly $r = a_0$ with
$\langle r\rangle = \tfrac32 a_0$; $P_{2s}$ shows two humps
separated by a spherical \emph{node}; $P_{2p} \propto
r^4\eu^{-r/a_0}$, peaking at $4a_0$. Generally $\langle r\rangle
\approx n^2a_0$: atoms grow \emph{quadratically} with excitation,
while their binding shrinks as $1/n^2$ --- the leverage behind the
Rydberg giants of \cref{pb:b3:hydrogen-atom:1}.
\end{proposition}
\admitted

\begin{example}[Reading the sizes]\label{ex:b3:hydrogen-atom:sizes}
Ground state: \qty{0.1}{nm} across, \qty{13.6}{eV} deep --- the
scales of all chemistry. At $n = 10$: radius $\sim\qty{5}{nm}$,
binding \qty{0.136}{eV} --- loosely held. At $n = 100$: a
\emph{micrometre}-scale atom bound by \qty{1.4}{meV}, wrecked by the
feeblest field --- yet interstellar space is empty enough for such
atoms to live and broadcast
(\cref{pb:b3:hydrogen-atom:1}).
\end{example}

\section{The spectrum}

\begin{proposition}[Series and selection rules]\label{prop:b3:hydrogen-atom:series}
\index{Balmer series}\index{Lyman series}
A jump $n' \to n$ emits the wavelength
\[
\frac{1}{\lambda} = R_{\text{H}}\Big(\frac{1}{n^2} -
\frac{1}{n'^2}\Big) , \qquad
R_{\text{H}} = \frac{E_{\text{I}}}{hc} = \qty{1.097e7}{m^{-1}} ,
\]
subject to the dipole selection rule $\Delta l = \pm1$ (the photon
carries $\hbar$). The \omterm{prop:b3:hydrogen-atom:series}{Lyman series} ($\to n = 1$) lies in the far
ultraviolet from \qty{121.6}{nm}; the \omterm{prop:b3:hydrogen-atom:series}{Balmer series} ($\to n = 2$)
begins at the red $H\alpha$ line, \qty{656.3}{nm} --- the colour of
emission nebulae and solar prominences; the Paschen and later series
recede into the infrared, and between $n = 110$ and $109$ the same
formula lands at \qty{5.0}{GHz}: hydrogen speaks from the
ultraviolet to the radio dial.
\end{proposition}

\begin{proof}
Energy conservation with \cref{thm:b3:hydrogen-atom:levels};
$R_{\text{H}}$ as in \cref{pb:b3:hamiltonian-mechanics:1}, now
derived rather than postulated.
\end{proof}

\begin{example}[Hydrogen-like atoms: one solution, many atoms]\label{ex:b3:hydrogen-atom:scaling}
Replace the proton's charge by $Ze$ and the electron by any orbiting
mass $\mu$: every formula rescales as
\[
a \to a_0\,\frac{m_{\text{e}}}{\mu}\,\frac1Z , \qquad
E_{\text{I}} \to E_{\text{I}}\,\frac{\mu}{m_{\text{e}}}\,Z^2 .
\]
He$^+$ ($Z = 2$): \qty{54.4}{eV}. Inner electrons of heavy atoms
($Z \sim 30$): K-shell energies in the keV --- the characteristic
X-rays by which Moseley ordered the elements
(\cref{exo:b3:hydrogen-atom:5}). Muonic hydrogen ($\mu = 207
m_{\text{e}}$): a femtometre-scale atom probing the proton itself.
Positronium ($e^+e^-$, $\mu = m_{\text{e}}/2$): half the binding,
twice the size, and a short life ending in annihilation photons.
And in a semiconductor, an electron and a hole orbit each other
with small effective masses in a screening dielectric: an
\emph{exciton}, the same atom grown to ten nanometres and
millielectronvolts (\cref{exo:b3:hydrogen-atom:7}) --- hydrogen is
less an atom than a template.
\end{example}

\begin{method}[Working with hydrogenic systems]\label{met:b3:hydrogen-atom:method}
(1) Scale first: $a \propto 1/\mu Z$, $E \propto \mu Z^2$ turn any
hydrogen answer into any hydrogen-like answer. (2) Sizes: $\langle
r\rangle \sim n^2a$; spacings near level $n$: $\dd E/\dd n =
2E_{\text{I}}/n^3$. (3) Spectra: Rydberg formula plus $\Delta l =
\pm1$. (4) Penetration: $s$ states feel the nucleus, high-$l$
states orbit outside --- the lever of multi-electron chemistry. (5)
Sanity anchors: $\qty{13.6}{eV}$, $\qty{52.9}{pm}$,
$\qty{121.6}{nm}$, $\qty{656.3}{nm}$ --- four numbers worth
memorising for life.
\end{method}

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:10}
Averages and the virial. For the ground state, compute (a)
$\langle 1/r\rangle$ and check $\langle E_p\rangle = -2E_{\text{I}}
= 2E_1$; (b) $\langle E_k\rangle = +E_{\text{I}}$, verifying the

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:11}
How fine is fine structure. (a) Show the ground-state speed scale
is $\langle v^2\rangle^{1/2}/c = \alpha \approx 1/137$, and for
hydrogen-like $Z$: $Z\alpha/n$. (b) Relativistic corrections enter
at relative order $(v/c)^2$: estimate the fine-structure scale
$\alpha^2E_{\text{I}}$ in meV, and the splitting's order for
$n = 2$ in GHz. (c) For hydrogen-like uranium ($Z = 92$): $v/c$ at
$n = 1$ --- is the non-relativistic treatment tenable? (d) What
does the formal divergence at $Z\alpha \to 1$ ($Z \approx 137$)
announce physically?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hydrogen-atom:12}
The unreasonable degeneracy. (a) In classical mechanics, show that
for $V = -k/r$ the orbit closes (no precession) by citing the
conserved Laplace--Runge--Lenz vector $\vect A = \vect p\wedge\vect
L - mk\,\vect e_r$ --- verify $\dd\vect A/\dd t = 0$ using Newton's
law. (b) Argue: a conserved vector fixing the ellipse's axis is an
extra symmetry beyond rotations --- and extra symmetry means extra
degeneracy (recall \cref{exo:b3:quantum-formalism:9}). (c) Add a
small $1/r^2$ correction to the potential and show the ellipse
precesses: the axis turns, $\vect A$ is no longer conserved. (d)
Connect: in multi-electron atoms the effective potential is not
pure $1/r$, and the $l$-degeneracy breaks (sodium!); in hydrogen
itself, relativity supplies the small correction --- which
observed feature of \cref{exo:b3:hydrogen-atom:11} is that?
\end{exercise}

\section{Problem: Giant atoms}

\begin{problem}[{Weekend problem --- Rydberg atoms, from the
Galaxy's radio glow to quantum computers}]\label{pb:b3:hydrogen-atom:1}
Stretch hydrogen to $n \approx 100$ and it becomes a different kind
of object: micrometres across, bound by millielectronvolts,
absurdly sensitive --- and absurdly useful. This problem scales the
hydrogen solution up, first to the interstellar clouds that
broadcast at centimetre wavelengths, then to the laboratory arrays
where such giants entangle each other into processors. Data:
$E_{\text{I}} = \qty{13.6}{eV}$, $a_0 = \qty{52.9}{pm}$,
$R_{\text{H}}c = \qty{3.29e15}{Hz}$, $k_{\text{B}}T$ at
\qty{300}{K} is \qty{25.9}{meV}.

\medskip\noindent\textbf{Part I --- The scaling laws.}
\begin{enumerate}
  \item Give the four basic scalings with $n$: size $\langle
        r\rangle$, binding $|E_n|$, spacing $E_{n+1} - E_n$ (for
        large $n$), and the classical orbital frequency of the
        corresponding Bohr orbit.
  \item Show that for large $n$ the transition frequency $n+1 \to
        n$ approaches $2R_{\text{H}}c/n^3$, and verify it equals
        the classical orbital frequency (the correspondence
        principle of \cref{pb:b3:hamiltonian-mechanics:1},
        completed).
  \item Evaluate size and binding at $n = 50$ and $n = 100$.
  \item The electric dipole of a Rydberg atom scales as $n^2ea_0$:
        evaluate at $n = 50$ in debye ($1\,\text{D} =
        \qty{3.34e-30}{C.m}$) and compare with a water molecule's
        \qty{1.85}{D}.
  \item Estimate the electric field that rips the $n = 50$ atom
        apart, as $F \sim |E_n|/(e\,\langle r\rangle)$, in V/cm.
  \item Why can such atoms not survive ordinary laboratory vacuum
        chemistry --- yet survive fine in interstellar space
        (compare collision rates)?
\end{enumerate}

\medskip\noindent\textbf{Part II --- The Galaxy's recombination
lines.}
In ionised nebulae, protons capture electrons into high $n$; the
cascade radiates a comb of ``radio recombination lines''.
\begin{enumerate}[resume]
  \item Compute the frequency of the $n = 110 \to 109$ transition
        (called H109$\alpha$).
  \item In what band does it fall, and what kind of telescope
        receives it?
  \item Compute the wavelength of H109$\alpha$ and compare it with
        the size of the emitting atom: how many atoms fit in one
        wavelength?
  \item These lines measure the nebula's temperature by their
        Doppler width: at $T = \qty{e4}{K}$, the hydrogen thermal
        speed is $\sim\qty{12.8}{km/s}$ --- compute the fractional
        linewidth $\Delta\nu/\nu$ and the width of H109$\alpha$ in
        kHz.
  \item Radio waves cross the dust that blackens the optical sky:
        what does that let recombination-line astronomers map that
        Balmer-line astronomers cannot?
  \item Above roughly $n \sim 1000$ (atoms $0.05$ millimetres
        across!), the levels blur into a continuum even in space:
        name two effects that broaden or destroy such states.
\end{enumerate}

\medskip\noindent\textbf{Part III --- Giants in the laboratory.}
\begin{enumerate}[resume]
  \item Lasers drive rubidium atoms to $n \approx 70$ in
        micrometre-spaced optical-tweezer arrays. Two such atoms a
        distance $R$ apart interact by their induced dipoles
        (recall \cref{exo:b3:harmonic-oscillator:10}): with dipole
        $\propto n^2$, show the van der Waals strength scales as a
        colossal $n^{11}$ (use $C_6 \propto d^4/\Delta E$ with
        level spacing $\Delta E \propto n^{-3}$).
  \item The ``blockade'': within a radius $R_{\text{b}}$, the
        interaction shifts the doubly excited state out of laser
        resonance, so \emph{two} atoms cannot both be excited.
        Explain in one sentence why this realises a two-qubit
        gate.
  \item Blockade radii reach several micrometres --- thousands of
        times the atoms' ground-state size: why is this
        long reach (compare chemistry's nanometre range)
        exactly what a scalable processor wants?
  \item A Rydberg state at $n = 70$ lives $\sim\qty{100}{\micro s}$
        while a gate takes $\sim\qty{0.5}{\micro s}$: roughly how
        many operations fit in one lifetime, and why does that
        ratio, not the lifetime alone, matter?
  \item Room-temperature blackbody radiation, peaking near
        \qty{100}{meV} but rich in a low-energy tail, drives
        transitions between neighbouring Rydberg levels spaced by
        only $\sim\qty{0.1}{meV}$: why must precision experiments
        enclose the atoms in cold shields?
  \item Estimate how many antenna-like dipole transitions
        ($\propto n^2$) a \qty{300}{K} photon bath drives per
        second compared with an $n = 1$ atom: which scaling makes
        Rydberg atoms exquisite \emph{sensors} of microwave and
        terahertz fields?
\end{enumerate}

\medskip\noindent\textbf{Part IV --- The big picture.}
\begin{enumerate}[resume]
  \item One formula, $E_n = -E_{\text{I}}\mu Z^2/m_{\text{e}}n^2$,
        spans how many orders of magnitude of binding energy from
        hydrogen-like uranium ($Z = 92$, $n = 1$) down to an
        $n = 1000$ Rydberg state? Compute both ends.
  \item Radiative lifetimes of low-$l$ states scale as $n^3$: from
        the $2p$ lifetime of \qty{1.6}{ns}, estimate the lifetime
        of a $100p$ state --- and compare with the room-temperature
        blackbody problem of Part III.
  \item At CERN, anti-atoms of antihydrogen are now spectroscopied
        on the $1s \to 2s$ interval to fifteen digits: what
        fundamental symmetry does agreement with ordinary hydrogen
        test, and why is hydrogen the right atom for the
        comparison?
  \item At $n \sim 100$ the electron's de Broglie wave wraps a
        nearly classical orbit; at $n = 1$ no orbit exists at all:
        in one sentence, where does the classical picture switch
        on?
  \item Rydberg constants are measured to fifteen digits: why is
        hydrogen, of all systems, the natural precision anchor of
        atomic physics?
  \item The 2012 Nobel Prize (Haroche) used Rydberg atoms as
        photon-counters that do not destroy the photon: which two
        properties from Part I and III make them ideal
        non-demolition probes of microwave fields?
  \item Summarise the named result: the $n^2$, $n^{-2}$, $n^{-3}$
        and $n^{11}$ scalings of one exactly solved atom stretch
        hydrogen from \qty{52.9}{pm} to half a micrometre, tune its
        voice from \qty{121.6}{nm} to \qty{6}{cm}, and turn it into
        both the Galaxy's radio beacon and the two-qubit gate of
        neutral-atom quantum computers.
\end{enumerate}
\end{problem}
