\chapter{Quantum Angular Momentum}\label{ch:b3:quantum-angular-momentum}

Point a radio telescope at a dark patch of sky and tune it to
\qty{115}{GHz}: a bright, needle-sharp line appears --- carbon
monoxide molecules, ten kelvin above absolute zero, stepping down one
rung of a ladder of \emph{rotational} states. Their tumbling, like
everything that turns in quantum mechanics, is quantised twice over:
the magnitude of the angular momentum can only be
$\sqrt{l(l+1)}\,\hbar$, and its component along any chosen axis only
$m\hbar$. This chapter derives that double quantisation --- once
algebraically, from the \omterm{def:b3:quantum-formalism:commutator}{commutators} inherited from the \omterm{def:b3:hamiltonian-mechanics:poisson}{Poisson
brackets} of \cref{ch:b3:hamiltonian-mechanics}, and once concretely,
as the \omterm{prop:b3:quantum-angular-momentum:harmonics}{spherical harmonics} that shape every atomic orbital. The
algebra will hand us more than we ask: it permits half-integer values
that no orbital motion can realise, a vacancy nature fills two
chapters from now with spin. On the way we meet the molecules'
rotational ladders --- the millimetre-wave lines by which
astronomers weigh the galaxies' cold gas --- and the magnetic moments
by which angular momentum first showed itself split.

\section{The algebra of rotation}

\begin{definition}[Angular momentum operators]\label{def:b3:quantum-angular-momentum:operators}
\index{angular momentum!quantum}
Orbital angular momentum is the operator $\hat{\vect L} =
\hat{\vect r}\wedge\hat{\vect p}$, componentwise $\hat L_z = \hat
x\hat p_y - \hat y\hat p_x$ and cyclic. From $[\hat x, \hat p_x] =
\iu\hbar$:
\[
[\hat L_x, \hat L_y] = \iu\hbar\,\hat L_z
\quad\text{(and cyclic)} , \qquad
[\hat L^2, \hat L_z] = 0 :
\]
the components are mutually incompatible --- no state has two of
them sharp --- but the total square is compatible with any one of
them: the pair $(\hat L^2, \hat L_z)$ is the standard choice of
labels. These \omterm{def:b3:quantum-formalism:commutator}{commutators} are exactly $\iu\hbar$ times the \omterm{def:b3:hamiltonian-mechanics:poisson}{Poisson
brackets} computed in \cref{ex:b3:hamiltonian-mechanics:brackets}:
Dirac's dictionary at work.
\end{definition}

\begin{theorem}[The spectrum, from the algebra alone]\label{thm:b3:quantum-angular-momentum:spectrum}
\index{magnetic quantum number}
Let $\hat J_x, \hat J_y, \hat J_z$ be \emph{any} three \omterm{def:b3:quantum-formalism:observable}{Hermitian
operators} obeying the commutation relations above. Then the joint
\omterm{def:b3:quantum-formalism:observable}{eigenvalues} of $(\hat J^2, \hat J_z)$ are
\[
\hat J^2:\ j(j+1)\hbar^2 , \qquad
\hat J_z:\ m\hbar , \quad m = -j, -j+1, \dots, +j ,
\]
where $j$ is a non-negative \emph{integer or half-integer}: $2j + 1$
values of $m$ for each $j$. The \omterm{def:b3:harmonic-oscillator:ladder}{ladder operators} $\hat J_\pm = \hat
J_x \pm \iu\hat J_y$ step $m$ by $\pm1$ at fixed $j$:
\[
\hat J_\pm\ket{j,m} = \hbar\sqrt{j(j+1) - m(m\pm1)}\,\ket{j,m\pm1} .
\]
\end{theorem}

\begin{proof}[Partial proof]
$[\hat J_z, \hat J_\pm] = \pm\hbar\hat J_\pm$: acting with $\hat
J_\pm$ shifts the $\hat J_z$ \omterm{def:b3:quantum-formalism:observable}{eigenvalue} by $\pm\hbar$, at fixed
$\hat J^2$ (which commutes with everything built from the $\hat
J_i$). The norm $\|\hat J_\pm\ket{j,m}\|^2 = \hbar^2[j(j+1) -
m(m\pm1)]$ (computed from $\hat J_\mp\hat J_\pm = \hat J^2 - \hat
J_z^2 \mp \hbar\hat J_z$) must stay non-negative: the ladder must
terminate above at some $m_{\max}$ with $m_{\max}(m_{\max}+1) =
j(j+1)$, i.e.\ $m_{\max} = j$, and below at $m_{\min} = -j$.
Climbing from $-j$ to $+j$ in unit steps forces $2j$ to be a
non-negative integer. The \omterm{def:b3:quantum-formalism:observable}{eigenvalue} notation $j(j+1)\hbar^2$ is
thereby justified after the fact.
\end{proof}

\begin{remark}[A vacancy in the catalogue]\label{rem:b3:quantum-angular-momentum:halfinteger}
The algebra allows $j = \tfrac12, \tfrac32, \dots$ --- ladders with
an even number of rungs. Orbital motion, we show next, uses only
integers. Nature, however, wastes nothing: the electron itself
carries $j = \tfrac12$, with no orbit behind it
(\cref{ch:b3:spin-two-level}); the algebra derived here, unchanged,
will run atomic magnetism, nuclear spins and the qubit.
\end{remark}

\section{Orbital angular momentum: spherical harmonics}

\begin{proposition}[Spherical harmonics]\label{prop:b3:quantum-angular-momentum:harmonics}
\index{spherical harmonics}\index{orbital quantum number}
In spherical coordinates $\hat L_z = -\iu\hbar\,\partial_\varphi$,
and the joint eigenfunctions of $(\hat L^2, \hat L_z)$ on the sphere
are the \emph{\omterm{prop:b3:quantum-angular-momentum:harmonics}{spherical harmonics}} $Y_l^m(\theta, \varphi)$:
\[
\hat L^2\,Y_l^m = l(l+1)\hbar^2\,Y_l^m , \qquad
\hat L_z\,Y_l^m = m\hbar\,Y_l^m ,
\]
with $l = 0, 1, 2, \dots$ \emph{integer} --- single-valuedness of
$\eu^{\iu m\varphi}$ forces integer $m$, hence integer $l$. The
first few, up to normalisation: $Y_0^0 = \text{const}$ (the $s$
shape, a sphere); $Y_1^0 \propto \cos\theta$ and $Y_1^{\pm1}
\propto \sin\theta\,\eu^{\pm\iu\varphi}$ (the $p$ shapes, two lobes);
$Y_2^0 \propto 3\cos^2\theta - 1$ and its partners (the $d$ family).
Parity: $Y_l^m(-\vect r$ direction$) = (-1)^l\,Y_l^m$.
\end{proposition}
\admitted

\begin{example}[The rigid rotor and its ladder]\label{ex:b3:quantum-angular-momentum:rotor}
A diatomic molecule tumbling with moment of inertia $I$ has $\hat H
= \hat L^2/2I$: energies
\[
E_J = \frac{\hbar^2}{2I}\,J(J+1) = B\,J(J+1) ,
\qquad J = 0, 1, 2, \dots
\]
each $(2J+1)$-fold degenerate. Photon absorption obeys $\Delta J =
\pm1$, so the absorption frequencies are $\nu_{J+1\leftarrow J} =
2B(J+1)/h$: a comb of \emph{equally spaced} lines --- the signature
by which a rotational spectrum is recognised at a glance. For carbon
monoxide, $B/h = \qty{57.6}{GHz}$: the fundamental line falls at
\qty{115}{GHz} ($\lambda = \qty{2.6}{mm}$), the workhorse line of
millimetre radio astronomy
(\cref{pb:b3:quantum-angular-momentum:1}).
\end{example}

\section{Central potentials, completed}

\begin{theorem}[Separation in any central potential]\label{thm:b3:quantum-angular-momentum:radial}
\index{radial equation}\index{centrifugal barrier}
For $V(r)$, the stationary states can be taken as
\[
\psi(\vect r) = \frac{u(r)}{r}\,Y_l^m(\theta, \varphi) ,
\]
where $u$ solves the one-dimensional \omterm{prop:b3:schrodinger-three-dimensions:swave}{radial equation}
\[
-\frac{\hbar^2}{2m}\,u'' + \Big[V(r) +
\frac{\hbar^2\,l(l+1)}{2mr^2}\Big]u = E\,u , \qquad u(0) = 0 .
\]
The angular problem is solved once and for all by the $Y_l^m$; each
$l$ adds the repulsive \emph{\omterm{thm:b3:quantum-angular-momentum:radial}{centrifugal barrier}}
$\hbar^2l(l+1)/2mr^2$ --- the quantum version of the effective
potential of \cref{ex:b3:lagrangian-mechanics:central} --- and every
level of given $l$ is $(2l+1)$-fold degenerate, because no central
force can care about the orientation of the $z$ axis. The $s$-wave
trick of \cref{prop:b3:schrodinger-three-dimensions:swave} was the
$l = 0$ row of this theorem.
\end{theorem}

\begin{proof}[Partial proof]
The Laplacian splits as $\Delta = \tfrac1r\partial_r^2(r\,\cdot) -
\hat L^2/\hbar^2r^2$ (admitted; it is the statement that $\hat L^2$
is the angular part of $-\hbar^2\Delta$). Insert $\psi = (u/r)Y_l^m$
and use the \omterm{def:b3:quantum-formalism:observable}{eigenvalue} of $\hat L^2$: the stated equation. The
degeneracy follows because $\hat H$ commutes with all three $\hat
L_i$, whose \omterm{def:b3:harmonic-oscillator:ladder}{ladder operators} move $m$ without changing the energy.
\end{proof}

\section{Magnetic moments: angular momentum made visible}

\begin{proposition}[Orbital magnetic moment and the Zeeman effect]\label{prop:b3:quantum-angular-momentum:magnetic}
\index{Bohr magneton}\index{Zeeman effect}
A particle of charge $q$ and mass $m$ with orbital angular momentum
$\hat{\vect L}$ carries the magnetic moment
\[
\hat{\vect\mu} = \frac{q}{2m}\,\hat{\vect L} ;
\]
for the electron, the natural unit is the \emph{\omterm{prop:b3:quantum-angular-momentum:magnetic}{Bohr magneton}}
$\mu_{\text{B}} = e\hbar/2m_{\text{e}} = \qty{5.79e-5}{eV/T}$
(compare \cref{exo:b3:schrodinger-three-dimensions:5}). In a field
$B\vect e_z$ the energy $-\hat{\vect\mu}\cdot\vect B$ shifts each
level by $+m\,\mu_{\text{B}}B$ (electron charge negative): a level
of given $l$ splits into its $2l + 1$ components --- the
\emph{\omterm{prop:b3:quantum-angular-momentum:magnetic}{Zeeman effect}}, the first direct display of the quantisation
of $m$, and the reason $m$ is called the \omterm{thm:b3:quantum-angular-momentum:spectrum}{magnetic quantum number}.
At $B = \qty{1}{T}$ the splitting is $\qty{58}{\micro eV}$: small
beside optical energies, easily resolved as a shift of spectral
lines --- and, read in reverse, a magnetometer: the Zeeman splitting
of sunlight's lines maps the magnetic fields of sunspots.
\end{proposition}

\begin{proof}[Partial proof]
Classically a charge on an orbit is a current loop: $\mu = IA =
(qv/2\pi r)(\pi r^2) = qL/2m$; the operator statement inherits it
(and follows from the $\hat{\vect A}$ coupling of
\cref{prop:b3:lagrangian-mechanics:charge}). The energy shift is
first-order perturbation theory, anticipated here and justified in
\cref{ch:b3:perturbation-theory}.
\end{proof}

\begin{method}[Angular momentum in practice]\label{met:b3:quantum-angular-momentum:method}
(1) Label states by $(l, m)$ --- or $(j, m)$ for the abstract
algebra --- and never ask for two components at once. (2) Matrix
elements: use the ladder formulas; $\hat L_x = (\hat L_+ + \hat
L_-)/2$. (3) Central potential: quote the $Y_l^m$, solve only the
\omterm{prop:b3:schrodinger-three-dimensions:swave}{radial equation} with the \omterm{thm:b3:quantum-angular-momentum:radial}{centrifugal barrier}. (4) Rotational
energies: $BJ(J+1)$, lines at $2B(J+1)$ --- extract $B$, hence bond
lengths, from equal spacings. (5) Magnetic questions: convert
angular momenta to moments at $\mu_{\text{B}}$ per $\hbar$, energies
at $\mu_{\text{B}}B$ per unit $m$.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b3:quantum-angular-momentum:1}
(a) Derive $[\hat L_x, \hat L_y] = \iu\hbar\hat L_z$ from the
canonical \omterm{def:b3:quantum-formalism:commutator}{commutators}. (b) Show $[\hat L^2, \hat L_z] = 0$. (c) Show
$[\hat L_z, \hat x] = \iu\hbar\hat y$: what does $\hat L_z$
generate? (d) Why do the three components admit no common eigenbasis
--- except for one particular state (which)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-angular-momentum:2}
For $l = 1$, in the basis $\{\ket{1,1}, \ket{1,0}, \ket{1,-1}\}$:
(a) write the matrix of $\hat L_z$; (b) use the ladder formula to
write $\hat L_+$ and $\hat L_-$; (c) assemble $\hat L_x$ and check
its \omterm{def:b3:quantum-formalism:observable}{eigenvalues} are $\hbar, 0, -\hbar$; (d) a state with $L_x =
+\hbar$ is measured along $z$: give the probabilities of the three
outcomes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-angular-momentum:3}
Carbon monoxide: bond length \qty{0.113}{nm}, reduced mass
$\qty{1.14e-26}{kg}$. (a) Compute $I$ and $B = \hbar^2/2I$ in meV.
(b) The frequency and wavelength of the $J = 1 \leftarrow 0$ line.
(c) The next two lines. (d) An astronomer measures the comb spacing
to five digits: what molecular quantity does she obtain, and to what
precision?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-angular-momentum:4}
A particle is in a state of $l = 2$. (a) List the possible outcomes
of measuring $L_z$. (b) The smallest angle between $\vect L$ and the
$z$ axis, using $\cos\theta = m/\sqrt{l(l+1)}$: evaluate it. (c) Why
can the angle never be zero? (d) Show the minimal angle tends to
zero as $l \to \infty$: classical vectors recovered.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:5}
The angular part of $-\hbar^2\Delta$ is $\hat L^2$, with
\[
\hat L^2 = -\hbar^2\Big[\frac{1}{\sin\theta}\,\partial_\theta
(\sin\theta\,\partial_\theta) +
\frac{1}{\sin^2\theta}\,\partial_\varphi^2\Big] .
\]
(a) Verify that $Y_1^0 \propto \cos\theta$ has $\hat L^2$-eigenvalue
$2\hbar^2$. (b) Verify $Y_1^{\pm1} \propto \sin\theta\,
\eu^{\pm\iu\varphi}$ likewise, and their $\hat L_z$ \omterm{def:b3:quantum-formalism:observable}{eigenvalues}. (c)
Check the parities. (d) Why must $\braket{Y_1^0}{Y_1^1} = 0$ without
computing any integral?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:6}
Which rotational line is brightest? The population of level $J$ is
$\propto (2J+1)\,\eu^{-BJ(J+1)/k_{\text{B}}T}$. (a) Explain the two
factors. (b) Show the maximum sits near $J_{\max} \approx
\sqrt{k_{\text{B}}T/2B} - \tfrac12$. (c) For CO at \qty{10}{K}
(a dark cloud) and at \qty{300}{K}: which lines dominate? (d)
Inverting: an observed CO ladder peaking at $J = 7$ betrays what
temperature?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:7}
Normal \omterm{prop:b3:quantum-angular-momentum:magnetic}{Zeeman effect}. A spectral line at \qty{500}{nm} comes from a
transition $l = 2 \to l = 1$ in a field $B = \qty{2}{T}$; ignore
spin (valid for special ``singlet'' states). (a) Sketch the split
levels. (b) With the selection rule $\Delta m = 0, \pm1$, show only
\emph{three} line positions appear, at $0, \pm\mu_{\text{B}}B/h$.
(c) Compute the splitting in GHz and in picometres of wavelength.
(d) Most real lines split into more than three components
(``anomalous'' Zeeman): what missing ingredient, carried by the
electron itself, was historical evidence for?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:8}
The centrifugal wall. For hydrogen ($V = -e^2/4\pi\varepsilon_0r$),
write $U_{\text{eff}}$ for $l = 1$ in units of $E_{\text{I}}$ and
$a_0$. (a) Locate its minimum and value. (b) Show $U_{\text{eff}} >
0$ for $r < \dots$ --- find the radius inside which the barrier
dominates. (c) Compare $|\psi|^2$ near $r = 0$ for $s$ and $p$
states ($u \sim r^{l+1}$, admitted): which states ``touch'' the
nucleus? (d) Electron capture --- a nucleus swallowing one of its
atom's electrons --- proceeds overwhelmingly from $s$ shells:
explain in one sentence.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-angular-momentum:9}
On $\ket{l, m}$: (a) show $\langle\hat L_x\rangle = \langle\hat
L_y\rangle = 0$; (b) show $\langle\hat L_x^2\rangle = \langle\hat
L_y^2\rangle = \tfrac12[l(l+1) - m^2]\hbar^2$; (c) verify the
\omterm{thm:b3:quantum-formalism:uncertainty}{uncertainty relation} $\Delta L_x\,\Delta L_y \ge \tfrac\hbar2
|\langle\hat L_z\rangle|$ on the stretched state $m = l$; (d)
interpret the ``cone'' picture of the figure in the light of (a) and
(b).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-angular-momentum:10}
Vibration--rotation bands. An infrared vibrational transition
($\hbar\omega_0$) of a diatomic changes $J$ by $\pm1$
simultaneously. (a) Show the absorption lines fall at $h\nu =
\hbar\omega_0 + 2B(J+1)$ (the $R$ branch, from $J \to J+1$) and
$h\nu = \hbar\omega_0 - 2BJ$ (the $P$ branch, $J \to J-1$), $J \ge
1$. (b) Why is there a \emph{gap} at $\hbar\omega_0$ itself? (c)
Sketch the band: two combs flanking a missing central line, each
line's strength following
\cref{exo:b3:quantum-angular-momentum:6}. (d) The
\qty{15}{\micro m} band of \cref{pb:b3:harmonic-oscillator:1} has
exactly this anatomy: what sets its overall width, and hence the
``wings'' that make added carbon dioxide effective?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-angular-momentum:11}
Finish the algebra. (a) From $\hat J_\mp\hat J_\pm = \hat J^2 -
\hat J_z^2 \mp \hbar\hat J_z$, compute $\|\hat J_\pm\ket{j,m}\|^2$
and the ladder coefficients. (b) Show that if the ladder failed to
terminate, some norm would go negative: exhibit the offending state.
(c) Conclude that $m_{\max} - m_{\min} = 2j$ must be a non-negative
integer and enumerate the allowed $j$. (d) Where exactly does the
argument \emph{fail} to exclude half-integers --- and which
additional requirement (single-valuedness on the sphere) excludes
them for orbital motion only?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-angular-momentum:12}
Einstein--de Haas: angular momentum is mechanical. An iron cylinder
(radius $a = \qty{1.0}{cm}$, density $\rho = \qty{7.9e3}{kg/m^3}$,
$n = \qty{8.5e28}{atoms/m^3}$) hangs from a torsion fibre; reversing
its magnetisation flips about one $\hbar$ of angular momentum per
atom. (a) By conservation, the rod must recoil: compute the angular
momentum per unit volume transferred. (b) With the rod's moment of inertia $\tfrac12 M
a^2$, show the rod acquires $\omega = 2n\hbar/\rho a^2$ and evaluate
it. (c) The 1915 experiment drove the reversal at the fibre's
resonance to accumulate the tiny kicks: estimate the oscillation
amplitude after $Q \sim 100$ resonant reversals, taking your
$\omega$ per kick. (d) The measured ratio moment/angular-momentum
came out twice the orbital prediction $e/2m_{\text{e}}$: what was
that factor of two announcing?
\end{exercise}

\section{Problem: Weighing galaxies at 2.6 millimetres}

\begin{problem}[{Weekend problem --- carbon monoxide, radio
astronomy and the cold universe}]\label{pb:b3:quantum-angular-momentum:1}
Most of a galaxy's star-making matter is cold molecular hydrogen ---
and it is invisible: H$_2$, symmetric, has no dipole and no
rotational lines at cold-cloud temperatures. Astronomy's solution is
its faithful companion. One CO molecule per $\num{e4}$ H$_2$, with
its \qty{115}{GHz} ladder, lights up every molecular cloud in the
sky. Data: for CO, $B/h = \qty{57.6}{GHz}$ ($B =
\qty{0.238}{meV}$); bond length $r_0 = \qty{0.113}{nm}$; reduced
mass $\mu = \qty{1.14e-26}{kg}$; $k_{\text{B}}T$ at \qty{10}{K} is
\qty{0.862}{meV}.

\medskip\noindent\textbf{Part I --- The quantum rotor.}
\begin{enumerate}
  \item From $\hat H = \hat L^2/2I$, give the energies $E_J$ and
        their degeneracies.
  \item Verify $B = \hbar^2/2I$ for CO from $r_0$ and $\mu$.
  \item With the selection rule $\Delta J = \pm1$, derive the
        absorption frequencies $\nu_{J+1\leftarrow J} = 2B(J+1)/h$
        and list the first three.
  \item Why are the lines \emph{equally spaced} --- and what does a
        measured spacing hand the astronomer (through $I$)?
  \item Compute the wavelength of the fundamental line, and explain
        why observing it needs high, dry sites (what absorbs
        millimetre waves in our own atmosphere --- recall the
        molecule of \cref{pb:b3:harmonic-oscillator:1}'s
        neighbour, water)?
  \item The photon of the $1 \to 0$ line carries $\hbar$ of angular
        momentum: why is $\Delta J = \pm1$ not merely a rule of
        thumb but a conservation law?
\end{enumerate}

\medskip\noindent\textbf{Part II --- Why CO and not H$_2$.}
\begin{enumerate}[resume]
  \item H$_2$ is homonuclear: state why it emits no rotational
        dipole radiation at all.
  \item H$_2$ is also \emph{light}: compute the rotational constant
        of H$_2$ ($\mu = m_{\text{p}}/2$, $r_0 = \qty{0.074}{nm}$)
        and its first accessible transition energy; compare with
        $k_{\text{B}}T$ at \qty{10}{K}: even by quadrupole
        back-doors, could cold H$_2$ radiate?
  \item CO's first rung costs $2B = \qty{0.48}{meV}$ against
        $k_{\text{B}}T = \qty{0.86}{meV}$: is the ladder alive at
        \qty{10}{K}?
  \item Compute the population ratio $n_1/n_0$ at \qty{10}{K}
        (degeneracy included).
  \item Near which $J$ does the population peak at \qty{10}{K}
        (use $J_{\max} \approx \sqrt{k_{\text{B}}T/2B} - \tfrac12$)?
  \item Summarise in one sentence the pact of the trade: what CO
        provides, what must be assumed about the CO-to-H$_2$
        ratio.
\end{enumerate}

\medskip\noindent\textbf{Part III --- Reading the sky.}
\begin{enumerate}[resume]
  \item A cloud's line arrives at \qty{115.156}{GHz} instead of the
        laboratory \qty{115.271}{GHz}: compute the cloud's velocity
        along the line of sight, and its direction.
  \item The line is Doppler-\emph{broadened} by internal motions of
        $\pm\qty{2}{km/s}$: compute the linewidth in MHz, and
        compare with the thermal width expected at \qty{10}{K}
        ($v_{\text{th}} \sim \sqrt{k_{\text{B}}T/m_{\text{CO}}}
        \approx \qty{55}{m/s}$): what dominates the broadening?
  \item Mapping the Doppler shift of CO across a spiral galaxy's
        disc yields its rotation curve $v(r)$. The curves stay
        \emph{flat} far beyond the visible disc: recall (Year 1
        volume, gravitation) what $v(r)$ should do around a
        centrally concentrated mass, and state what the flatness
        implies.
  \item The ratio of the $2\to1$ line's brightness to the
        $1\to0$'s measures level populations: which single physical
        quantity of the cloud does it deliver?
  \item ALMA resolves CO in galaxies whose light left when the
        universe was young; the observed frequency of the
        $1 \to 0$ line from a galaxy at ``redshift $z = 1$''
        arrives at half its rest value --- at what frequency, and
        (recalling \cref{ex:b3:relativistic-kinematics:redshift})
        what stretched it?
  \item From a measured CO line luminosity astronomers quote cloud
        masses in solar masses: list the chain of conversions (line
        photons $\to$ CO count $\to$ H$_2$ mass) and the weakest
        link.
\end{enumerate}

\medskip\noindent\textbf{Part IV --- The cold nurseries.}
\begin{enumerate}[resume]
  \item Star-forming cores sit at \qty{10}{K}: from Wien's law
        their thermal glow peaks near \qty{290}{\micro m} ---
        invisible to every optical telescope. In one sentence: why
        is the rotational ladder the \emph{only} thermometer and
        scale such a core offers?
  \item The $J = 1$ level lies \qty{5.5}{K} (in temperature units)
        above the ground state: why does that make the $1\to0$ line
        an exquisite thermometer precisely in the \qty{10}{K}
        regime (rather than, say, an optical line at
        \qty{2}{eV} $\sim \qty{23000}{K}$)?
  \item Why does a \emph{collapsing} core eventually stop being
        visible in CO ($1\to0$) --- consider what high density and
        dust do to the line and its escape.
  \item Molecular clouds also show lines of NH$_3$, HCN, and dozens
        more, each with its own $B$ and dipole: what does the
        richness of this ``chemical radio dial'' let astronomers
        disentangle?
  \item The whole sky glows at \qty{2.7}{K} (the cosmic microwave
        background): what happens to the contrast of a cloud's CO
        lines as the cloud's own temperature approaches
        \qty{2.7}{K}, and why is that a hard floor for this
        thermometer?
  \item A millimetre dish must hold its shape to about
        $\lambda/20$: compute that tolerance at \qty{115}{GHz},
        and compare with the tolerance an optical mirror
        (\qty{500}{nm}) must meet --- why could ALMA's
        twelve-metre dishes be built in the open air?
  \item Summarise the named result: a ladder $BJ(J+1)$ with $B/h =
        \qty{57.6}{GHz}$, populated at ten kelvin and read at
        \qty{2.6}{mm}, is how the mass, temperature, velocity and
        rotation of the cold universe --- and the evidence for dark
        matter around spiral galaxies --- are measured.
\end{enumerate}
\end{problem}
