\chapter{El electromagnetismo en la materia}\label{ch:b3:electromagnetism-in-matter}

% Provenance (not for print): completes the "électromagnétisme dans la
% matière" module (dielectrics, magnetism, ferromagnetism) common to the
% surveyed third-year curricula; the Year 2 volume covered only wave
% propagation in media.

Deslícese una lámina de plástico entre las placas de un condensador cargado y la
tensión cae, como si hubiera aparecido carga de la nada. Enróllese una bobina
alrededor de un clavo de hierro y su campo magnético crece mil veces. Despéguese
un imán decorativo de la nevera y \emph{recuerda} en qué
sentido se lo magnetizó --- durante décadas, sin fuente de alimentación. El volumen
del Año 2 trató los campos en la materia al por mayor, escondiendo el material dentro
de un índice de refracción; este capítulo abre el material en canal. El plan
es el mismo para la electricidad y para el magnetismo: la materia es una multitud de
dipolos microscópicos; súmense en una densidad ($\vect P$ o $\vect M$);
invéntese un campo auxiliar ($\vect D$ o $\vect H$) que solo vea las
cargas y corrientes que controlamos; después bájese a la escala microscópica
--- donde toman el relevo los espines del \cref{ch:b3:spin-two-level}, el factor de
Boltzmann del \cref{ch:b3:canonical-ensemble} y la transición de campo medio del
\cref{ch:b3:phase-transitions} --- para calcular qué hace el
material. El capítulo termina en el desguace, diseñando el
electroimán que levanta coches.

\section{La polarización y el campo de desplazamiento}

\begin{definition}[Polarización y cargas ligadas]\label{def:b3:electromagnetism-in-matter:polarisation}
\index{polarización}\index{carga ligada}\index{dieléctrico}
Un \emph{dieléctrico} es un aislante cuyas moléculas adquieren (o
ya poseen) momentos dipolares eléctricos. La respuesta del material se
resume en la \emph{polarización} $\vect P$: el momento dipolar
por unidad de volumen, $\vect P = n\langle\vect p\rangle$ para $n$ moléculas
por unidad de volumen. Un bloque polarizado lleva \emph{cargas ligadas} ---
que no son libres de marcharse, pero son perfectamente reales: una densidad superficial
$\sigma_{\text{b}} = \vect P\cdot\vect n$ allí donde la polarización
alcanza la superficie, y una densidad de volumen $\rho_{\text{b}} =
-\nabla\cdot\vect P$ dondequiera que no sea uniforme. Una lámina uniformemente
polarizada equivale exactamente a dos hojas de carga $\pm P$
en sus caras: en el interior, las cabezas y las colas de sus dipolos se cancelan.
\end{definition}

\begin{proposition}[El campo de desplazamiento]\label{prop:b3:electromagnetism-in-matter:displacement}
\index{desplazamiento eléctrico}\index{susceptibilidad eléctrica}\index{permitividad relativa}
La carga total es la libre más la ligada; la ley de Gauss con $\rho =
\rho_{\text{f}} + \rho_{\text{b}}$ se reordena en una ley para el
\emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{desplazamiento eléctrico}} $\vect D = \varepsilon_0\vect E +
\vect P$:
\[
\nabla\cdot\vect D = \rho_{\text{f}} :
\]
las fuentes de $\vect D$ son solo las cargas libres --- las que están en nuestras
placas y cables. En un \omterm{def:b3:electromagnetism-in-matter:polarisation}{dieléctrico} \emph{lineal}, la respuesta es
proporcional, $\vect P = \varepsilon_0\chi_{\text{e}}\vect E$, de modo que
$\vect D = \varepsilon_0\varepsilon_{\text{r}}\vect E$ con
$\varepsilon_{\text{r}} = 1 + \chi_{\text{e}}$ la \emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{permitividad
relativa}}: el aire $1.0006$, el polietileno $2.3$, el vidrio $\sim 5$ y
el agua unos enormes $80$. Entre las placas de un condensador mantenidas a
carga fija, el \omterm{def:b3:electromagnetism-in-matter:polarisation}{dieléctrico} no cambia $\vect D$, de modo que $\vect E =
\vect D/\varepsilon_0\varepsilon_{\text{r}}$ \emph{cae} en un factor
$\varepsilon_{\text{r}}$: las cargas superficiales ligadas encaran a las
libres y cancelan casi todo su campo. La capacidad queda multiplicada por
$\varepsilon_{\text{r}}$ --- toda la industria del condensador en una
letra griega.
\end{proposition}

\begin{proof}
$\nabla\cdot(\varepsilon_0\vect E) = \rho_{\text{f}} +
\rho_{\text{b}} = \rho_{\text{f}} - \nabla\cdot\vect P$; pásese el
término de \omterm{def:b3:electromagnetism-in-matter:polarisation}{polarización} al miembro izquierdo.
\end{proof}

\section{De dónde viene la permitividad}

\begin{proposition}[Polarización inducida y Clausius--Mossotti]\label{prop:b3:electromagnetism-in-matter:mossotti}
\index{polarizabilidad}\index{relación de Clausius--Mossotti}
Un átomo en un campo $\vect E_{\text{loc}}$ se estira hasta ser un dipolo
$\vect p = \alpha\vect E_{\text{loc}}$, con $\alpha$ la
\emph{polarizabilidad} (de dimensión $\varepsilon_0\times$volumen ---
aproximadamente $\varepsilon_0$ por el volumen atómico). En un gas diluido,
$E_{\text{loc}} \approx E$ y $\chi_{\text{e}} =
n\alpha/\varepsilon_0$ es pequeña. En un medio denso, cada molécula siente además
a sus vecinas polarizadas: recortar una pequeña cavidad esférica
a su alrededor da $\vect E_{\text{loc}} = \vect E + \vect
P/3\varepsilon_0$, y la autoconsistencia lleva a la
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{relación de Clausius--Mossotti}}
\[
\frac{\varepsilon_{\text{r}} - 1}{\varepsilon_{\text{r}} + 2}
= \frac{n\alpha}{3\varepsilon_0} :
\]
mídase la susceptibilidad minúscula de un vapor y predígase la del líquido ---
funciona con un error de un pequeño tanto por ciento para los líquidos no polares.
\end{proposition}
\admitted

\begin{remark}
El campo de la cavidad $\vect P/3\varepsilon_0$ es el resultado de la esfera
uniformemente polarizada, citado aquí sin demostración; la deducción honesta
con condiciones de contorno pertenece a un curso dedicado de electrodinámica. Todo
lo demás de arriba es contabilidad.
\end{remark}

\begin{proposition}[Moléculas polares: la función de Langevin]\label{prop:b3:electromagnetism-in-matter:langevin}
\index{función de Langevin}\index{polarización de orientación}
Una molécula con un dipolo \emph{permanente} $p$ (el agua:
$p = \qty{6.2e-30}{C.m}$) no necesita estirarse --- solo alinearse,
contra la agitación térmica. El promedio canónico del
\cref{ch:b3:canonical-ensemble} sobre las orientaciones, con energía
$-pE\cos\theta$, da
\[
\langle\cos\theta\rangle = L(x) = \coth x - \frac1x ,
\qquad x = \frac{pE}{k_{\text{B}}T} ,
\]
la \emph{\omterm{prop:b3:electromagnetism-in-matter:langevin}{función de Langevin}}: lineal ($L \approx x/3$) con $x$ pequeña, y
saturada en $1$ cuando el campo gana de calle. En el régimen lineal,
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ --- una ley de Curie
eléctrica, que cae como $1/T$: caliéntese un líquido polar y su permitividad
\emph{baja}, la firma que distingue los dipolos permanentes de los
inducidos. Para el agua a temperatura ambiente, esto predice
$\chi_{\text{e}} \sim 12$ --- el orden correcto, con la cooperación de los enlaces
de hidrógeno empujando el valor real hasta $80$.
\end{proposition}

\begin{proof}[Demostración parcial]
$Z = \int_0^\pi \eu^{x\cos\theta}\,2\pi\sin\theta\,\dd\theta \propto
\sinh x/x$, y $\langle\cos\theta\rangle = \dd\ln Z/\dd x = \coth x
- 1/x$. Desarrollar $\coth x = 1/x + x/3 - \dots$ para $x$ pequeña da
$L \approx x/3$, luego $P = np\langle\cos\theta\rangle =
np^2E/3k_{\text{B}}T$.
\end{proof}

\section{La magnetización y el campo \texorpdfstring{$H$}{H}}

\begin{definition}[Magnetización, corrientes ligadas y H]\label{def:b3:electromagnetism-in-matter:magnetisation}
\index{magnetización}\index{campo magnético H}\index{susceptibilidad magnética}
La historia magnética discurre en paralelo. La materia está llena de espiras
de corriente microscópicas --- electrones que orbitan y giran, cada uno un dipolo
magnético $\vect\mu$; su densidad es la \emph{magnetización}
$\vect M = n\langle\vect\mu\rangle$. Un bloque uniformemente magnetizado equivale
a una \emph{corriente superficial ligada} que circula por sus
costados (las espiras interiores se cancelan por parejas; un imán de barra \emph{es} un
solenoide de corriente ligada). Al separar las corrientes en libres y ligadas,
la ley de Ampère se reordena en torno al campo auxiliar
\[
\vect H = \frac{\vect B}{\mu_0} - \vect M , \qquad
\oint \vect H\cdot\dd\vect\ell = I_{\text{libre}} :
\]
$\vect H$ lo generan las corrientes de nuestros cables (unidades
\unit{A/m}), mientras que $\vect B$ conserva la física --- fuerzas, flujo,
inducción. Los medios lineales cumplen $\vect M = \chi_{\text{m}}\vect H$ y
$\vect B = \mu_0(1 + \chi_{\text{m}})\vect H =
\mu_0\mu_{\text{r}}\vect H$.
\end{definition}

\begin{remark}[Tres personalidades magnéticas]\label{rem:b3:electromagnetism-in-matter:zoo}
\index{diamagnetismo}\index{paramagnetismo}\index{ferromagnetismo}
Los materiales responden a $\vect H$ de tres maneras.
Los \emph{diamagnéticos} ($\chi_{\text{m}} \sim -10^{-5}$: agua, cobre,
grafito) responden como la ley de Lenz hecha permanente: el campo aplicado
perturba todas las órbitas electrónicas de modo que se le oponen --- respuesta débil, universal,
independiente de la temperatura, y a la que los imanes repelen (así se ha levitado
una rana en un imán de \qty{16}{T}).
Los \emph{paramagnéticos} ($\chi_{\text{m}} \sim +10^{-5}$ a $10^{-3}$:
aluminio, O$_2$) poseen momentos permanentes --- espines desapareados del
\cref{ch:b3:identical-particles} --- que se alinean como los dipolos de
Langevin: $\chi_{\text{m}} = \mu_0 n\mu^2/3k_{\text{B}}T$, una ley de Curie
en $1/T$ (la versión cuántica de dos niveles es el
\cref{exo:b3:canonical-ensemble:11}).
Los \emph{ferromagnéticos} (hierro, cobalto, níquel: con $\mu_{\text{r}}$ de
miles) son paramagnéticos cuyos espines además hablan entre sí ---
el acoplamiento de intercambio del \cref{exo:b3:identical-particles:12} ---
y por debajo de la \omterm{thm:b3:phase-transitions:meanfield}{temperatura de Curie} se alinean \emph{espontáneamente}:
el \cref{thm:b3:phase-transitions:meanfield}, ahora puesto a trabajar.
\end{remark}

\section{El ferromagnetismo en acción: dominios e histéresis}

\begin{definition}[Dominios e histéresis]\label{def:b3:electromagnetism-in-matter:hysteresis}
\index{dominio magnético}\index{histéresis}\index{remanencia}\index{coercitividad}
Un pedazo bruto de hierro \emph{no} está magnetizado: se rompe en
\emph{dominios}, regiones de escala micrométrica magnetizadas por completo cada una pero
apuntando en direcciones distintas, de modo que el campo exterior (y su coste energético)
casi se cancela. Un $\vect H$ aplicado mueve las paredes de dominio ---
los dominios favorables crecen --- y luego hace girar dominios enteros hasta
alinearlos. Las paredes se enganchan en los defectos, así que el proceso es irreversible: barra
$H$ arriba y abajo y $B(H)$ traza un lazo, el \emph{ciclo de
histéresis}. Córtese la corriente y sobrevive un campo \emph{remanente}
$B_{\text{r}}$; cancelarlo exige el campo \emph{coercitivo} inverso
$H_{\text{c}}$. El área encerrada por el lazo es la
energía disipada por ciclo y por unidad de volumen. Los materiales \emph{blandos}
(acero al silicio, ferritas: lazo estrecho, $H_{\text{c}}$ pequeño) sirven para los
núcleos de transformadores y motores; los \emph{duros}
(alnico, Nd$_2$Fe$_{14}$B: lazo gordo, $H_{\text{c}}$ enorme) sirven para los
imanes permanentes --- y para la memoria magnética: el imán de nevera
y el disco duro son lazos de histéresis que se niegan a olvidar.
\end{definition}

\begin{example}[El electroimán de núcleo de hierro]\label{ex:b3:electromagnetism-in-matter:electromagnet}
Enróllense $N$ vueltas que llevan $I$ alrededor de un anillo de hierro ($\mu_{\text{r}}
\sim 5000$) con un pequeño entrehierro $e$. La ley de Ampère para $\vect H$
a lo largo del lazo: $H_{\text{hierro}}\ell + H_{\text{ent}}e = NI$, y
la continuidad del flujo hace común a $B$, de modo que
\[
B\Big(\frac{\ell}{\mu_0\mu_{\text{r}}} + \frac{e}{\mu_0}\Big) = NI .
\]
Con $\ell = \qty{1}{m}$ y $e = \qty{1}{cm}$, el término del entrehierro es
cincuenta veces el del hierro: casi todo el esfuerzo de la bobina se gasta
en empujar campo a través de un centímetro de aire. Ese es el circuito
magnético en una línea --- el hierro es un conductor de flujo casi perfecto
y el aire es la resistencia --- y por eso los motores, los relés y los
elevadores de desguace mantienen sus entrehierros implacablemente finos
(\cref{pb:b3:electromagnetism-in-matter:1}).
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:1}
Contabilidad de \omterm{def:b3:electromagnetism-in-matter:polarisation}{cargas ligadas}. Una lámina de espesor $d$ está uniformemente
polarizada según su normal, con magnitud $P$. (a) Dense las densidades de
\omterm{def:b3:electromagnetism-in-matter:polarisation}{carga ligada} de cada cara. (b) Muéstrese que el campo \emph{interior} debido
a esas hojas es $E = P/\varepsilon_0$, opuesto a $\vect P$. (c)
¿Cuánto vale $\vect D$ dentro, sin carga libre en ninguna parte? (d) Un
electreto en barra ($\vect P$ congelada), ¿de qué objeto magnético es
el análogo eléctrico?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:2}
Condensador con \omterm{def:b3:electromagnetism-in-matter:polarisation}{dieléctrico}. Un condensador de placas paralelas
($C_0 = \qty{100}{pF}$) se carga a \qty{12}{V} y se desconecta.
Una lámina con $\varepsilon_{\text{r}} = 4$ lo llena. Hállese (a) la nueva
capacidad; (b) la nueva tensión; (c) la energía antes y después
--- ¿adónde fue la diferencia? (d) Repítase (b)–(c) con la
batería conectada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:3}
El zoo de las susceptibilidades. Clasifíquese (dia-, para- o ferromagnético)
y justifíquese a partir de la estructura electrónica cuando pueda: el agua
($\chi_{\text{m}} = -9\times10^{-6}$), el aluminio
($+2.2\times10^{-5}$), el oxígeno líquido ($+3.5\times10^{-3}$ ---
¿por qué es magnético siquiera el O$_2$?) y el níquel ($\mu_{\text{r}} \sim 600$
a campo pequeño). ¿A cuál de los cuatro atraería visiblemente un imán
potente, y a cuál repelería levísimamente?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:4}
$B$, $H$ y $M$. Un solenoide largo ($n = \num{2000}$ vueltas por metro,
$I = \qty{1.5}{A}$) se llena de hierro, y se mide $B = \qty{1.6}{T}$.
Calcúlese (a) $H$; (b) $M$; (c) la amplificación
$B/\mu_0H$; (d) la corriente superficial ligada por metro equivalente a
$M$ --- compárese con el propio $nI$ de la bobina.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:5}
Cable coaxial, aislado. Una línea coaxial (de radio interior $a$ y exterior
$b$) lleva carga libre $\lambda$ por unidad de longitud en el núcleo,
con \omterm{def:b3:electromagnetism-in-matter:polarisation}{dieléctrico} $\varepsilon_{\text{r}}$ entre medias. (a) Hállese
$\vect D(r)$ por simetría. (b) Dedúzcanse $\vect E$ y $\vect P$.
(c) Calcúlense las densidades de \omterm{def:b3:electromagnetism-in-matter:polarisation}{carga ligada} en $r = a$ y $r = b$ y
compruébese que suman cero por unidad de longitud. (d) Muéstrese que la capacidad
por unidad de longitud queda multiplicada por $\varepsilon_{\text{r}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:6}
Clausius--Mossotti en acción. El argón gaseoso a \qty{1}{bar} y
\qty{300}{K} tiene $\varepsilon_{\text{r}} = 1.00052$. (a) Extráigase
$\alpha/\varepsilon_0$ (un volumen) y compárese con el volumen
atómico. (b) El argón líquido tiene $n = \qty{2.1e28}{m^{-3}}$: predígase
$\varepsilon_{\text{r}}$ con Clausius--Mossotti (medido: 1.53).
(c) ¿Por qué se pasa menos aquí la ingenua $\chi = n\alpha/\varepsilon_0$
que para el agua? (d) ¿Para qué moléculas debe fallar
el esquema entero, y qué lo sustituye?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:7}
La permitividad gigante del agua. (a) A partir de la
\cref{prop:b3:electromagnetism-in-matter:langevin}, calcúlese
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ para el agua
($n = \qty{3.3e28}{m^{-3}}$, $p = \qty{6.2e-30}{C.m}$,
$T = \qty{300}{K}$). (b) Estímese el campo necesario para $x = 1$
(saturación seria) y compárese con la rigidez dieléctrica del agua,
$\sim\qty{7e7}{V/m}$. (c) Los hornos microondas funcionan a \qty{2.45}{GHz}:
¿por qué calienta la comida una respuesta \emph{con pérdidas} de dipolos que rotan, y
por qué se calienta el hielo mucho más despacio que el agua líquida? (d) Predígase
el signo de $\dd\varepsilon_{\text{r}}/\dd T$ para el agua y para el
argón.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:8}
El diamagnetismo, estimado. Trátese un electrón atómico como una carga en un
anillo de radio $r$; un $B$ aplicado cambia su frecuencia angular en
el desplazamiento de Larmor $\Delta\omega = eB/2m_{\text{e}}$. (a) Muéstrese que el
momento inducido es $\Delta\mu = -e^2r^2B/4m_{\text{e}}$ (opuesto a
$B$). (b) Promediando orientaciones y sumando $Z$ electrones por
átomo, se halla $\chi_{\text{m}} = -\mu_0ne^2Z\langle
r^2\rangle/6m_{\text{e}}$: evalúese para el agua ($n =
\qty{3.3e28}{m^{-3}}$ moléculas, $Z = 10$, $\langle r^2\rangle
\approx (\qty{0.7}{\angstrom})^2$) y compárese con el
$-9\times10^{-6}$ medido. (c) ¿Por qué no depende el diamagnetismo de la temperatura
mientras que el paramagnetismo sí? (d) ¿Por qué levita incluso una rana en
\qty{16}{T}, y por qué debe ser \emph{no uniforme} el campo del imán?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:9}
Paramagnetismo de Curie y enfriamiento. Una sal lleva $n =
\qty{2e27}{m^{-3}}$ iones de momento $\mu \approx \mu_{\text{B}}$.
(a) Calcúlese $\chi_{\text{m}} = \mu_0n\mu^2/3k_{\text{B}}T$ a
\qty{300}{K} y a \qty{1}{K}. (b) ¿A qué temperatura alcanzaría
$\chi_{\text{m}}$ el valor $1$ --- y qué física (aquí desatendida)
se interpone antes en casi todas las sales? (c) Desmagnetización adiabática:
magnetícese a \qty{1}{K}, aíslese y retírese después el campo despacio ---
explíquese con la entropía de espín del
\cref{ch:b3:microcanonical-ensemble} por qué se enfría la muestra. (d) ¿Por qué
necesita el método un \emph{para}magnético y no un ferromagnético?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:10}
El circuito magnético. Un toro de hierro ($\ell = \qty{60}{cm}$,
$\mu_{\text{r}} = 4000$, sección $A = \qty{16}{cm^2}$) lleva
$N = 500$ vueltas y un entrehierro $e = \qty{4}{mm}$. (a) Muéstrese que $NI =
B(\ell/\mu_0\mu_{\text{r}} + e/\mu_0)$ y calcúlese la corriente para
$B = \qty{1.2}{T}$. (b) ¿Qué fracción de $NI$ se gasta en el entrehierro?
(c) Defínase la reluctancia $\mathcal R = \ell/\mu A$ de cada
tramo y reformúlese (a) como una ``ley de Ohm'' en serie para el flujo. (d)
El hierro satura cerca de \qty{1.8}{T}: explíquese qué le ocurre al
modelo de circuito --- y a su motor --- más allá de ese valor.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:11}
Pérdidas por \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histéresis}. El núcleo de un transformador (de volumen \qty{4e-3}{m^3})
funciona a \qty{50}{Hz}. Su lazo de acero al silicio encierra
$\sim\qty{40}{J/m^3}$ por ciclo. (a) Calcúlese la potencia perdida por
\omterm{def:b3:electromagnetism-in-matter:hysteresis}{histéresis}. (b) El mismo núcleo en acero duro
($\sim\qty{6e3}{J/m^3}$): pérdida y veredicto. (c) Las corrientes de Foucault añaden
una pérdida $\propto f^2$: explíquese por qué los núcleos son laminados y por qué
las ferritas toman el relevo a radiofrecuencias. (d) Un bit de disco duro debe
\emph{conservar} su \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetización} frente a las sacudidas térmicas durante diez años:
relaciónense las exigencias sobre $H_{\text{c}}$ para la memoria con las de un
transformador, y conclúyase que ningún material puede hacer los dos trabajos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:12}
El campo desmagnetizante. Dentro de un cuerpo uniformemente magnetizado sin
corriente libre, $\oint\vect H\cdot\dd\vect\ell = 0$ obliga a que
$\vect H$ dentro \emph{se oponga} a $\vect M$ ($\vect H =
-N_{\text{d}}\vect M$, con $N_{\text{d}}$ un factor de forma: $1/3$
para una esfera, $\approx 0$ a lo largo de una aguja larga, $\approx 1$ a través de una
placa fina). (a) Justifíquense los límites de la aguja y de la placa con la
imagen de las corrientes ligadas (el solenoide). (b) ¿Por qué se desmagnetiza en parte
un imán rechoncho mientras que una aguja conserva su
\omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetización}? (c) Conéctese con las agujas de brújula y con los granos
alargados de la cinta magnética. (d) Una esfera de hierro dulce está en un
$B_0$ externo uniforme: explíquese por qué su respuesta satura en $\vect H
\approx 0$ dentro, es decir, en $M \approx 3B_0/\mu_0$ como mucho, por
grande que sea $\mu_{\text{r}}$.
\end{exercise}

\section{Problema: el elevador de desguace}

\begin{problem}\label{pb:b3:electromagnetism-in-matter:1}
\emph{El elevador de desguace.} Un centro de reciclaje encarga un
electroimán de grúa: una olla de hierro de cara plana, de un metro de ancho, que debe
levantar balas de coches prensados de dos toneladas y después --- igual de importante ---
\emph{soltarlas} a la orden. Usted es el ingeniero de diseño.

\medskip\noindent\textbf{Parte I --- Por qué hierro.}

\begin{enumerate}
  \item La bobina desnuda: 400 vueltas que llevan \qty{25}{A} repartidas sobre
        un camino magnético de $\sim\qty{1}{m}$. Estímese $B \sim
        \mu_0NI/\ell$ sin hierro.
  \item Explíquese en dos frases, con dominios, por qué llenar la
        bobina de hierro multiplica esto por $\sim\mu_{\text{r}}$.
  \item La \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetización} del hierro satura en $M_{\text{s}} \approx
        \qty{1.7e6}{A/m}$. Calcúlese el techo $\mu_0M_{\text{s}}$
        que esto impone al campo en los polos, haga lo que haga la bobina.
  \item Cada átomo de hierro aporta $\approx 2.2$ magnetones de Bohr.
        Compruébese: con $n = \qty{8.5e28}{m^{-3}}$, recupérese
        $M_{\text{s}}$.
  \item ¿Por qué deben mecanizarse planas las caras polares y mantenerse libres de
        óxido y suciedad? (Piénsese en el
        \cref{ex:b3:electromagnetism-in-matter:electromagnet}.)
  \item La propia carga pasa a formar parte del circuito magnético.
        Esbócese el camino del flujo: núcleo de la olla, cara norte, bala de acero,
        cara sur y de vuelta por la culata.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- El circuito magnético y la fuerza.}

\begin{enumerate}[resume]
  \item Modélese el circuito: camino de hierro $\ell = \qty{1.2}{m}$,
        $\mu_{\text{r}} = 2000$, y dos entrehierros efectivos
        (el contacto cara--bala) de $e = \qty{1.5}{mm}$ cada uno. Escríbase
        la ley de Ampère para $H$ a lo largo del lazo.
  \item Calcúlense los tres términos de reluctancia por unidad de área
        ($\ell/\mu_{\text{r}}$ frente a $2e$) y muéstrese que los
        entrehierros milimétricos siguen consumiendo casi todo el esfuerzo de la bobina.
  \item Con $NI = \num{10000}$ amperios-vuelta, calcúlese $B$ en los
        entrehierros.
  \item La presión de elevación sobre cada cara polar es $B^2/2\mu_0$
        (la densidad de energía magnética liberada por metro de acercamiento).
        Evalúese en \unit{kPa} para su $B$.
  \item Área polar total $A = \qty{0.12}{m^2}$: calcúlese la
        fuerza de elevación y conviértase a toneladas.
  \item ¿Cumple la especificación de dos toneladas con un margen de factor
        dos? Si no, ajuste $NI$ y dígase la nueva corriente.
  \item Las balas prensadas tocan las caras quizá en un tercio de su
        área, con entrehierros efectivos más anchos. Recalcúlese la fuerza para
        $e = \qty{4}{mm}$ y $A_{\text{ef}} = \qty{0.04}{m^2}$, y
        coméntese por qué las cargas nominales se dan en capacidad ``sobre placa plana''.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Recordar y olvidar.}

\begin{enumerate}[resume]
  \item En el desguace accionan el interruptor para soltar una bala --- y esta se queda
        colgada. Nómbrese al culpable, con el lazo de \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histéresis}.
  \item ¿Por qué debe ser la olla de hierro \emph{dulce} ($H_{\text{c}}$
        pequeño, $B_{\text{r}}$ pequeña) y no de acero
        magnético duro?
  \item Incluso el hierro dulce conserva algo de \omterm{def:b3:electromagnetism-in-matter:hysteresis}{remanencia}. Propóngase el
        remedio habitual: un breve pulso de corriente \emph{invertida} ---
        ¿a qué punto del lazo apunta?
  \item Algunos controladores aplican en su lugar una corriente alterna
        que decae. Esbócese lo que hace la trayectoria $B$--$H$ y por qué
        acaba desmagnetizada.
  \item La grúa maneja además planchas calientes recién salidas de un horno,
        a \qty{800}{\celsius}. El punto de Curie del hierro está en
        \qty{1043}{K}: ¿qué le ocurre a la fuerza de elevación, y por qué?
        (\cref{thm:b3:phase-transitions:meanfield}.)
  \item Un aprendiz de verano propone ahorrar cobre doblando
        el entrehierro y doblando $NI$. Úsense sus fórmulas de la Parte II para
        explicar la asimetría: ¿qué se reduce a la mitad y qué solo
        se mantiene?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- Potencia, calor y la factura.}

\begin{enumerate}[resume]
  \item La bobina: 400 vueltas de cobre, longitud media de vuelta
        \qty{2.5}{m}, sección del hilo \qty{16}{mm^2}, resistividad
        $\rho = \qty{1.7e-8}{\Omega.m}$. Calcúlese su resistencia.
  \item A $I = \qty{25}{A}$: la potencia disipada y la energía
        diaria en un turno de ocho horas con un \qty{60}{\percent} de ciclo de trabajo.
  \item La energía magnética almacenada en los dos entrehierros
        ($B^2/2\mu_0 \times$ volumen): calcúlese y compárese con
        un \emph{segundo} de disipación de la bobina. ¿Adónde va todo
        el resto de la energía eléctrica?
  \item Soltar la carga no devuelve a la red casi nada de la energía
        almacenada. Explíquese, con el área del lazo y el
        pico inductivo, por qué el controlador necesita un camino de
        rueda libre (flyback) en paralelo con la bobina.
  \item Falla la alimentación con una bala en el aire. ¿Qué hace la bala
        y qué sostiene la \omterm{def:b3:electromagnetism-in-matter:hysteresis}{remanencia} \emph{sola}? Justifíquense
        la norma del desguace de que nadie pasa bajo un imán energizado ---
        y el argumento de venta de los elevadores con batería de respaldo.
  \item Resúmase el diseño en cuatro líneas: el techo del campo lo fija
        $M_{\text{s}}$, la fuerza $B^2A/2\mu_0$, la controlabilidad la
        \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histéresis} del hierro dulce y el coste de operación $RI^2$ --- todo
        el capítulo colgando de una grúa.
\end{enumerate}
\end{problem}
