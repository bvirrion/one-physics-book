\chapter{Electrones en los sólidos}\label{ch:b3:electrons-in-solids}

% Provenance (not for print): second half of the "physique du solide"
% module (Drude, Sommerfeld, bands, semiconductors) common to the
% surveyed third-year curricula.

El cobre conduce la electricidad un millón de billones de billones de veces mejor
que el cuarzo --- el rango más ancho de cualquier propiedad física, y lo mandan
materiales que, en ambos casos, parecen pedruscos grises. El andamiaje del
capítulo anterior no explica nada de eso; los \emph{electrones} vertidos en
él lo explican todo. Este capítulo recorre los tres asaltos del siglo al
problema, cada uno heredando los escombros del anterior: el pinball clásico de
Drude (1900), que acierta con la ley de Ohm y falla escandalosamente con la
capacidad calorífica; el gas de Fermi de Sommerfeld, que es
el \cref{ch:b3:quantum-statistics} cobrando su cheque; y la intuición de Bloch de que
una onda electrónica en una \omterm{def:b3:crystalline-solids:lattice}{red} periódica se refleja a lo Bragg
como cualquier otra onda --- lo que abre \emph{\omterm{thm:b3:electrons-in-solids:bloch}{bandas prohibidas}} que clasifican todos los sólidos
en \omterm{def:b3:electrons-in-solids:classes}{metales}, \omterm{def:b3:electrons-in-solids:classes}{aislantes} y los hijos medianos de banda estrecha, los
\omterm{def:b3:electrons-in-solids:classes}{semiconductores}, sobre los que se construye toda la era electrónica. El
capítulo termina dentro de un panel solar.

\section{El metal de pinball de Drude}

\begin{proposition}[La conductividad de Drude]\label{prop:b3:electrons-in-solids:drude}
\index{modelo de Drude}\index{conductividad!microscópica}\index{velocidad de arrastre}
Trátense los electrones de valencia de un \omterm{def:b3:electrons-in-solids:classes}{metal} como un gas clásico, de densidad $n$,
que rebota en obstáculos cada $\tau$ segundos. Un campo $\vect E$
acelera a cada uno entre colisiones; la \emph{\omterm{prop:b3:electrons-in-solids:drude}{velocidad de arrastre}}
media es $\vect v_{\text{d}} = -e\tau\vect E/m_{\text{e}}$,
minúscula al lado del movimiento térmico. La densidad de corriente $\vect j =
-ne\vect v_{\text{d}}$ obedece entonces localmente a la ley de Ohm,
$\vect j = \sigma\vect E$, con
\[
\sigma = \frac{ne^2\tau}{m_{\text{e}}} .
\]
La $\sigma$ medida del cobre exige $\tau \approx \qty{2.5e-14}{s}$
--- y se siguen las dos glorias del modelo: explica por qué se calientan las
resistencias (el trabajo del campo se termaliza en cada colisión) y
predice la ley de Wiedemann--Franz, que $\kappa/\sigma T$ es la
misma constante para todos los \omterm{def:b3:electrons-in-solids:classes}{metales} --- los electrones llevan tanto la carga como el
calor (\cref{exo:b3:electrons-in-solids:5}).
\end{proposition}

\begin{proof}
Entre colisiones, $\dot{\vect v} = -e\vect E/m_{\text{e}}$;
como se empieza de cero en cada colisión, la velocidad media adquirida en un
tiempo medio $\tau$ es $-e\tau\vect E/m_{\text{e}}$. Entonces $\vect j =
-ne\vect v_{\text{d}} = (ne^2\tau/m_{\text{e}})\vect E$.
\end{proof}

\begin{remark}[El rescate de Sommerfeld]\label{rem:b3:electrons-in-solids:sommerfeld}
\index{modelo de Sommerfeld}\index{energía de Fermi!en los metales}
El gas de Drude debería añadir $\tfrac32k_{\text{B}}$ por electrón a la
capacidad calorífica de un \omterm{def:b3:electrons-in-solids:classes}{metal}; el experimento encuentra cien veces menos.
El \cref{ch:b3:quantum-statistics} ya absolvió a los electrones:
son un \emph{gas de Fermi degenerado}, con $T_{\text{F}} \sim
\qty{8e4}{K}$, y solo la capa térmica de anchura
$k_{\text{B}}T$ en torno a la superficie de Fermi puede responder --- al calor y
a la dispersión. La reejecución de Drude por Sommerfeld con estadística de
Fermi--Dirac conserva la fórmula de la conductividad (con $\tau$ ahora el
tiempo de vida de los electrones de la superficie de Fermi, cuyos \qty{e6}{m/s} son
$v_{\text{F}}$ y no una velocidad térmica), corrige la capacidad calorífica hasta la
astilla $T/T_{\text{F}}$ medida e incluso repara la constante numérica de
Wiedemann--Franz. Un misterio sobrevive a todo
modelo de electrones libres: por qué el cuarzo no conduce en absoluto.
\end{remark}

\section{Bloch: la red toma la palabra}

\begin{theorem}[Estados de Bloch y bandas prohibidas]\label{thm:b3:electrons-in-solids:bloch}
\index{teorema de Bloch}\index{banda prohibida}\index{estructura de bandas}
En un potencial periódico perfecto, los estados estacionarios son
\emph{ondas de Bloch} $\psi_k(x) = u_k(x)\eu^{\iu kx}$ ---
ondas planas vestidas con la periodicidad propia de la \omterm{def:b3:crystalline-solids:lattice}{red}, $u_k$ ---
que se propagan \emph{sin dispersarse}: un \omterm{def:b3:crystalline-solids:lattice}{cristal} perfecto tiene
resistencia nula, y la resistencia real viene de las imperfecciones
(vibraciones, impurezas). Pero no todas las energías sobreviven. En
$k = \pi/a$, la longitud de onda del electrón es $2a$: la condición de Bragg
sobre la propia \omterm{def:b3:crystalline-solids:lattice}{red} (el \cref{thm:b3:crystalline-solids:dispersion}
hizo el mismo descubrimiento para el sonido). Las ondas reflejada e incidente
forman dos patrones estacionarios --- con la carga amontonada sobre los iones
(menor energía) o entre ellos (mayor) ---, de modo que la parábola de
electrón libre $E = \hbar^2k^2/2m_{\text{e}}$ se desgarra: un intervalo de
energías prohibidas, la \emph{\omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida}} $E_{\text{g}}$, separa
una \emph{banda} completa de estados permitidos de la siguiente. La
identidad electrónica de un \omterm{def:b3:crystalline-solids:lattice}{cristal} es su escalera de bandas y \omterm{thm:b3:electrons-in-solids:bloch}{bandas prohibidas}.
\end{theorem}
\admitted

\begin{definition}[Metales, aislantes y semiconductores]\label{def:b3:electrons-in-solids:classes}
\index{metal}\index{aislante}\index{semiconductor}
Llénense las bandas con los electrones del \omterm{def:b3:crystalline-solids:lattice}{cristal}, dos por estado
(\cref{ch:b3:identical-particles}), empezando por los más fríos. Si la banda ocupada
más alta está \emph{parcialmente} llena, una energía infinitesimal puede
poner electrones en movimiento neto: un \emph{metal} (el sodio: un
electrón de valencia, media banda; los metales divalentes conducen por
bandas que se solapan). Si está \emph{exactamente} llena, con una \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida} ancha
encima, ningún empujón pequeño puede reordenar nada: un \emph{aislante}
(el diamante, $E_{\text{g}} = \qty{5.5}{eV}$). Si la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida} es pequeña ---
$\qty{1.12}{eV}$ en el silicio ---, la agitación térmica iza unos pocos
electrones al otro lado, y cada uno deja debajo un \emph{\omterm{def:b3:electrons-in-solids:doping}{hueco}} móvil: un
\emph{semiconductor}, cuyo recuento de portadores,
$\propto\eu^{-E_{\text{g}}/2k_{\text{B}}T}$, se duplica cada pocos
grados. De ahí la gran divisoria: los metales conducen peor al calentarse
(más vibraciones en que dispersarse) y los semiconductores mejor
(exponencialmente más portadores) --- un cambio de signo que identifica la
clase de un material en una sola medida.
\end{definition}

\section{Ingeniería de la banda prohibida: dopado y unión}

\begin{definition}[Dopado]\label{def:b3:electrons-in-solids:doping}
\index{dopado}\index{donante}\index{aceptor}\index{hueco}
Sustitúyase un átomo de silicio de cada millón por fósforo (cinco electrones
de valencia): cuatro enlaces quedan satisfechos y el quinto electrón,
ligado por unos meros \qty{45}{meV}
(\cref{exo:b3:electrons-in-solids:10}), está libre a temperatura
ambiente. Esos \emph{donantes} hacen un \omterm{def:b3:electrons-in-solids:classes}{semiconductor} de \emph{tipo n},
con conducción por electrones; el boro (tres electrones)
es un \emph{aceptor}, que atrapa un electrón de enlace y libera un
\emph{hueco} móvil --- un electrón ausente que se mueve, responde a los
campos y lleva carga positiva tan de verdad como una burbuja
lleva flotabilidad: el \emph{tipo p}. El dopado columpia la densidad de portadores
--- y por tanto la conductividad --- a lo largo de seis órdenes de magnitud a
voluntad: la perilla que convierte la arena en circuitos.
\end{definition}

\begin{proposition}[La unión p--n]\label{prop:b3:electrons-in-solids:junction}
\index{unión p--n}\index{diodo}\index{zona de deplexión}
Únase tipo p con tipo n. Los electrones se derraman hacia los \omterm{def:b3:electrons-in-solids:doping}{huecos} y
los aniquilan cerca de la interfaz, y dejan expuesta una \emph{\omterm{prop:b3:electrons-in-solids:junction}{zona de
deplexión}} de dopantes ionizados fijos cuya doble capa levanta un
campo interno --- equilibrio en la tensión de contacto
$V_{\text{bi}} = (k_{\text{B}}T/e)
\ln(N_{\text{a}}N_{\text{d}}/n_{\text{i}}^2) \approx
\qty{0.7}{V}$ en el silicio. La unión rectifica entonces: la polarización
directa baja la barrera y la corriente crece como
$\eu^{eV/k_{\text{B}}T}$; la inversa la sube y solo pasa una
fuga $I_0$:
\[
I = I_0\big(\eu^{eV/k_{\text{B}}T} - 1\big) .
\]
Esta asimetría es el \emph{diodo} --- y, en sus variantes,
el LED (los pares que se recombinan emiten fotones de la energía de la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida}), la célula
solar (los fotones de esa energía crean pares que el campo de contacto
barre hacia fuera: \cref{pb:b3:electrons-in-solids:1}) y, doblada
en sándwiches, el transistor: el interruptor del que un chip
moderno imprime cientos de miles de millones.
\end{proposition}
\admitted

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:1}
El cobre de Drude. $n = \qty{8.5e28}{m^{-3}}$, resistividad
$\rho = \qty{1.7e-8}{\ohm.m}$. Calcúlese (a) $\tau =
m_{\text{e}}/ne^2\rho$; (b) el recorrido libre medio usando la velocidad
de Fermi $v_{\text{F}} = \qty{1.6e6}{m/s}$; (c) ese recorrido en
\omterm{def:b3:crystalline-solids:lattice}{constantes de red} ($a = \qty{3.61}{\angstrom}$); (d) explíquese por qué
un vuelo tan largo ya insinúa que no son los iones los que
dispersan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:2}
El caracol en el hilo. Un hilo de cobre de \qty{1}{mm^2} lleva
\qty{10}{A}. (a) Calcúlese la \omterm{prop:b3:electrons-in-solids:drude}{velocidad de arrastre}. (b) ¿Cuánto tardaría
un electrón en recorrer un cable de lámpara de \qty{1}{m}? (c) ¿Por qué se enciende
la lámpara al instante, aun así? (d) Compárese
$v_{\text{d}}$ con $v_{\text{F}}$ y coméntese la imagen de pinball
de Drude.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:3}
Clasificar por bandas. Clasifíquese, con la regla del llenado de bandas: (a)
el sodio (un electrón de valencia); (b) el magnesio (dos --- y aun así un
\omterm{def:b3:electrons-in-solids:classes}{metal}: ¿qué deben hacer sus bandas?); (c) el diamante
($E_{\text{g}} = \qty{5.5}{eV}$); (d) el silicio
($\qty{1.12}{eV}$) --- y dígase para cada uno el signo de
$\dd\rho/\dd T$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electrons-in-solids:4}
\omterm{thm:b3:electrons-in-solids:bloch}{Bandas prohibidas} y fotones. (a) ¿Qué longitud de onda de fotón corresponde a la \omterm{thm:b3:electrons-in-solids:bloch}{banda
prohibida} del silicio, y en qué región espectral se vuelve transparente el
silicio? (b) Lo mismo para el diamante: ¿por qué es transparente en el
visible? (c) Un LED de nitruro de galio tiene $E_{\text{g}} =
\qty{3.4}{eV}$: ¿qué borde de color fija eso, y por qué necesitaron los LED
azules este material? (d) ¿Por qué muere apagado un LED rojo en vez
de brillar con un blanco débil cuando se le sobreexcita?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:5}
Wiedemann--Franz. (a) A partir de la $\kappa =
\qty{400}{W/(m.K)}$ y la $\sigma = \qty{5.9e7}{S/m}$ del cobre a
\qty{300}{K}, calcúlese $\kappa/\sigma T$. (b) Compárese con el
valor de Sommerfeld $L = \pi^2k_{\text{B}}^2/3e^2 =
\qty{2.44e-8}{W.\ohm/K^2}$. (c) Explíquese en una frase por qué un
solo tipo de portador ata las dos conductividades. (d) Las sartenes
y sus mangos: úsese la ley para explicar las elecciones de material de una
cocina.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:6}
La absolución de la capacidad calorífica. (a) Drude predice
$\tfrac32nk_{\text{B}}$ de los electrones: con la $n$ del cobre,
¿qué fracción añadiría eso al $3nk_{\text{B}}$ de la
\omterm{def:b3:crystalline-solids:lattice}{red} (un electrón de conducción por átomo)? (b)
El \cref{ch:b3:quantum-statistics} da en cambio $C_{\text{el}}
\sim \tfrac{\pi^2}{2}nk_{\text{B}}(T/T_{\text{F}})$: evalúese la
supresión $T/T_{\text{F}}$ a \qty{300}{K}
($T_{\text{F}} = \qty{8.1e4}{K}$). (c) ¿Por qué \emph{gana}, pese a todo, el término
electrónico a temperaturas de helio líquido
(recuérdese el $T^3$ de la \omterm{def:b3:crystalline-solids:lattice}{red})? (d) ¿Qué medida, representada como
$C/T$ frente a $T^2$, desenreda las dos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:7}
La superficie de Fermi se topa con la zona. (a) Para el cobre, calcúlese la
longitud de onda de Fermi $\lambda_{\text{F}} = 2\pi/k_{\text{F}}$ con
$k_{\text{F}} = (3\pi^2n)^{1/3}$. (b) Compárese con $2a$: ¿cuán
cerca está la superficie de Fermi de la condición de Bragg? (c) Explíquese
físicamente por qué las dos ondas estacionarias en $k = \pi/a$ (con la carga sobre
los iones o entre ellos) han de diferir en energía. (d) ¿Qué
hecho experimental del \cref{exo:b3:electrons-in-solids:1} explica
por fin el \omterm{thm:b3:electrons-in-solids:bloch}{teorema de Bloch} de la no dispersión?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:8}
El termómetro exponencial. El silicio intrínseco tiene $n_{\text{i}}
\propto \eu^{-E_{\text{g}}/2k_{\text{B}}T}$. (a) Calcúlese la
razón de densidades de portadores entre \qty{400}{K} y \qty{300}{K}. (b)
Esbócese así la resistencia de un termistor frente a la temperatura y
contrástela con la de un hilo de platino. (c) ¿Por qué el factor 2 en el
exponente (qué se crea \emph{por parejas})? (d) Estímese la
temperatura a la que la electrónica de silicio falla porque los portadores
intrínsecos ahogan un \omterm{def:b3:electrons-in-solids:doping}{dopado} de \qty{5e21}{m^{-3}}
($n_{\text{i}}(300\,\text{K}) \approx \qty{1e16}{m^{-3}}$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electrons-in-solids:9}
Aritmética del \omterm{def:b3:electrons-in-solids:doping}{dopado}. El silicio tiene $\qty{5e28}{m^{-3}}$ átomos. (a) Un
fósforo por cada millón de átomos de silicio: ¿qué densidad de portadores da, y
qué razón respecto de $n_{\text{i}} \approx \qty{1e16}{m^{-3}}$? (b) ¿En
qué factor ha cambiado la conductividad una parte por millón de suciedad? (c)
Explíquese por qué la fabricación de \omterm{def:b3:electrons-in-solids:classes}{semiconductores} purifica primero hasta partes
por \emph{mil millones} antes de dopar a propósito. (d) Estímese la
distancia media entre \omterm{def:b3:electrons-in-solids:doping}{donantes} y compárese con la
órbita de \qty{2.4}{nm} del \cref{exo:b3:electrons-in-solids:10}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electrons-in-solids:10}
El \omterm{def:b3:electrons-in-solids:doping}{donante} como \omterm{thm:b3:hydrogen-atom:levels}{átomo de hidrógeno}. El quinto electrón del fósforo
orbita a su ion $+e$ \emph{dentro} del silicio: apantállese el Coulomb con
$\varepsilon_{\text{r}} = 11.7$
(\cref{ch:b3:electromagnetism-in-matter}) y aligérese el
electrón hasta su masa efectiva de banda $m^* = 0.26\,m_{\text{e}}$.
(a) Escálense los resultados del hidrógeno del \cref{ch:b3:hydrogen-atom}:
muéstrese que $E = \qty{13.6}{eV}\times(m^*/m_{\text{e}})/
\varepsilon_{\text{r}}^2$ y evalúese. (b) Compárese con
$k_{\text{B}}T$ a \qty{300}{K}: ¿están ionizados los \omterm{def:b3:electrons-in-solids:doping}{donantes}? (c) Escálese
el \omterm{thm:b3:hydrogen-atom:levels}{radio de Bohr} del mismo modo. (d) La órbita abarca decenas de
celdas de la \omterm{def:b3:crystalline-solids:lattice}{red}: explíquese por qué eso justifica de forma autoconsistente
usar $\varepsilon_{\text{r}}$ y $m^*$ siquiera.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electrons-in-solids:11}
Números de la unión. Un diodo de silicio tiene $N_{\text{a}} =
N_{\text{d}} = \qty{1e22}{m^{-3}}$, $n_{\text{i}} =
\qty{1e16}{m^{-3}}$ y $T = \qty{300}{K}$. (a) Calcúlese
$V_{\text{bi}} = (k_{\text{B}}T/e)\ln(N_{\text{a}}N_{\text{d}}/
n_{\text{i}}^2)$. (b) A partir de la ley del diodo, calcúlese la razón de
corrientes a $+\qty{0.5}{V}$ y a $-\qty{0.5}{V}$. (c) ¿Por qué se
duplica $I_0$ --- y con ella la fuga inversa --- aproximadamente cada
\qty{10}{K}? (d) Un puente de cuatro diodos así convierte alterna en continua:
esbócese con palabras la idea del circuito.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electrons-in-solids:12}
Diseñar la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida} solar. (a) Los fotones por debajo de $E_{\text{g}}$ pasan
de largo; la energía del fotón por encima de $E_{\text{g}}$ se pierde como calor
en picosegundos. Explíquese el compromiso resultante al elegir
$E_{\text{g}}$. (b) El óptimo, cerca de \qty{1.3}{eV}, topa
la eficiencia de una sola unión en alrededor de un tercio
(Shockley--Queisser): ¿dónde queda el silicio
(\qty{1.12}{eV})? (c) ¿Por qué baten las pilas en tándem (una célula de banda
ancha sobre otra de banda estrecha) ese tope? (d) ¿Por qué es peor un panel
solar \emph{caliente}? (Dos razones honradas con el capítulo:
la $I_0$ de la ley del diodo y el ligero encogimiento de la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida} con $T$.)
\end{exercise}

\section{Problema: el referéndum del tejado}

\begin{problem}\label{pb:b3:electrons-in-solids:1}
\emph{El referéndum del tejado.} Un vecino está decidiendo si
techar su casa con paneles fotovoltaicos y ha nombrado árbitro
al físico del barrio. El panel candidato: de silicio,
\qty{2}{m^2}, sesenta células en serie, con \qty{400}{W} nominales bajo
los \qty{1000}{W/m^2} estándar del sol de mediodía.

\medskip\noindent\textbf{Parte I --- La luz solar se topa con la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida}.}

\begin{enumerate}
  \item La luz solar es aproximadamente un espectro térmico de \qty{5800}{K}
        (\cref{ch:b3:photons-phonons}): calcúlese en eV la energía de sus
        fotones del máximo (Wien).
  \item La \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida} del silicio es de \qty{1.12}{eV}: ¿cuál es la mayor
        longitud de onda que puede cosechar una célula de silicio?
  \item Aproximadamente una quinta parte de la potencia del sol llega en fotones
        por debajo de la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida}. ¿Qué le ocurre en el panel?
  \item Se absorbe un fotón verde de \qty{2.5}{eV}: ¿cuánta de su
        energía sobrevive como energía del par electrón--hueco, y adónde
        va el resto, y con qué rapidez?
  \item Combínense los apartados 3 y 4 en el veredicto del espectro: alrededor de
        la mitad de la potencia incidente se ha ido antes de que empiece la electrónica.
        Enúnciense los dos canales de pérdida en una frase cada uno.
  \item ¿Por qué necesita un fotón $\ge E_{\text{g}}$ siquiera --- qué
        prohíbe absorber dos fotones de media \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida} en rápida
        sucesión en el silicio corriente?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- La unión como motor.}

\begin{enumerate}[resume]
  \item Cada célula es una \omterm{prop:b3:electrons-in-solids:junction}{unión p--n}. Explíquese en tres frases
        cómo el campo de contacto de la \omterm{prop:b3:electrons-in-solids:junction}{zona de deplexión} convierte un par creado
        en corriente externa --- qué portador va a qué lado y por qué no
        se limitan a recombinarse.
  \item Con $N_{\text{a}} = N_{\text{d}} = \qty{1e22}{m^{-3}}$
        y $n_{\text{i}} = \qty{1e16}{m^{-3}}$, calcúlese
        $V_{\text{bi}}$ a \qty{300}{K}.
  \item Una célula en funcionamiento entrega unos \qty{0.6}{V}. Sesenta en
        serie: ¿cuál es la tensión de trabajo del panel?
  \item A partir de los \qty{400}{W} nominales, dedúzcase la corriente de
        trabajo.
  \item Estímese el flujo de fotones (fotones por segundo) que el panel
        absorbe de forma útil, tomando $\sim\qty{2}{eV}$ por fotón
        absorbido, y compárese con el flujo de electrones que implica su corriente:
        ¿qué fracción de los fotones absorbidos rinde un
        electrón recogido?
  \item La eficiencia nominal: \qty{400}{W} de
        \qty{2000}{W} incidentes. Concíliese ese 20\,\% con la
        ``mitad perdida antes de la electrónica'' de la Parte I: ¿adónde van los
        treinta puntos restantes?
  \item ¿Por qué entrega la célula \qty{0.6}{V} y no los
        \qty{1.12}{V} enteros de la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida}? (Nómbrese al culpable en la ley
        del diodo.)
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Tejados reales.}

\begin{enumerate}[resume]
  \item Un mediodía de invierno sin nubes a latitud $50^\circ$ entrega
        luz con $30^\circ$ de elevación. Calcúlese el factor
        geométrico respecto de la incidencia normal sobre un panel
        horizontal.
  \item La hoja de datos del panel indica un $-0.4\,\%/\text{K}$ de salida:
        un panel sobre un tejado negro alcanza \qty{65}{\celsius} en verano.
        ¿Cuánto sobrevive de lo nominal?
  \item Explíquese la pérdida por temperatura con el
        \cref{exo:b3:electrons-in-solids:12}(d).
  \item Una chimenea da sombra a una de las sesenta células. ¿Por qué puede eso
        estrangular toda la cadena en serie --- y qué le hace a la pobre célula
        sombreada su propia naturaleza de diodo?
  \item Los fabricantes cablean un \emph{diodo de paso} en paralelo con cada
        subcadena: explíquese su cometido en una frase.
  \item Promediado sobre los días y el tiempo atmosférico, el tejado rinde
        el \qty{15}{\percent} de la potencia nominal. Estímese la energía anual
        del panel de \qty{400}{W} en kilovatios-hora.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- El veredicto.}

\begin{enumerate}[resume]
  \item A \qty{0.25}{euros} por kWh, ¿cuánto gana el panel
        al año? Con un coste instalado de \qty{300}{euros},
        ¿cuál es el plazo de amortización?
  \item El panel costó aproximadamente \qty{500}{kWh} de fabricar
        (el silicio se purifica fundiéndolo): ¿cuánto tarda en
        devolver su propia energía?
  \item Su vecino pregunta por qué no se puede
        ``hacer el panel negro y ya'' para atrapar la quinta parte de sub-banda. Respóndase
        con teoría de bandas en dos frases.
  \item Pregunta después por qué no apilar debajo un segundo panel de banda
        más estrecha: respóndase con el
        \cref{exo:b3:electrons-in-solids:12}(c) --- y nómbrese la
        pega práctica.
  \item Treinta años después, el panel se ha desvanecido hasta el 85\,\% de lo
        nominal. ¿Qué villanos microscópicos de este capítulo
        (recuérdese qué limita $\tau$ y a qué temen las uniones)
        envejecen plausiblemente una célula?
  \item Entréguese el resumen del árbitro en cuatro líneas: la física de la \omterm{thm:b3:electrons-in-solids:bloch}{banda prohibida}
        fija la cosecha, la unión la convierte, el cableado en serie
        y la temperatura la gravan, y el balance --- de energía
        y de euros --- se cierra a favor del panel.
\end{enumerate}
\end{problem}
