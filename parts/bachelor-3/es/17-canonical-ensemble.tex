\chapter{El colectivo canónico}\label{ch:b3:canonical-ensemble}

Los sistemas aislados son una ficción de teórico: las muestras reales viven
en termostatos, en habitaciones, en océanos de aire --- en contacto con un
reservorio que no fija su energía, sino su \emph{temperatura}. El volumen
del Año 2 se topó empíricamente con la ley resultante: la probabilidad de un
estado de energía $E$ lleva el factor $\eu^{-E/k_{\text{B}}T}$. Este
capítulo deduce ese factor de Boltzmann en cuatro líneas a partir del
recuento del \cref{ch:b3:microcanonical-ensemble}, y construye después la
máquina que convierte la física estadística en una disciplina industrial: la
\emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{función de partición}} $Z$, una sola suma de la que caen por
derivación la energía, la entropía, la presión, la capacidad calorífica y
las fluctuaciones. Aquí se relatan las primeras victorias de esa máquina:
por qué mueren las capacidades caloríficas a baja temperatura (Einstein,
1907 --- la primera teoría cuántica de la materia), por qué la capacidad
calorífica del hidrógeno sube una escalera, por qué el cielo se adelgaza
exponencialmente --- y cómo Perrin, observando cómo sedimentaban unos granos
microscópicos en una gota de agua, contó el número de Avogadro y puso fin al
debate sobre la realidad de los átomos.

\section{La distribución de Boltzmann, deducida}

\begin{theorem}[Distribución canónica]\label{thm:b3:canonical-ensemble:boltzmann}
\index{distribución de Boltzmann}\index{colectivo canónico}\index{función de partición}
Un sistema que intercambia energía con un gran reservorio a temperatura $T$
ocupa su \omterm{def:b3:microcanonical-ensemble:states}{microestado} $s$ (de energía $E_s$) con probabilidad
\[
\mathcal P_s = \frac{\eu^{-E_s/k_{\text{B}}T}}{Z} , \qquad
Z = \sum_s \eu^{-E_s/k_{\text{B}}T}
\]
(en lo sucesivo, $\beta = 1/k_{\text{B}}T$). $Z$, la \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{función de
partición}}, normaliza las probabilidades --- y resulta contener
toda la termodinámica del sistema.
\end{theorem}

\begin{proof}
El sistema en el estado $s$ deja al reservorio la energía $E - E_s$:
por el \omterm{thm:b3:microcanonical-ensemble:postulate}{postulado fundamental} aplicado al conjunto, $\mathcal P_s
\propto \Omega_{\text{res}}(E - E_s)$. Desarróllese la entropía del
reservorio, enorme y suave, a primer orden: $S_{\text{res}}(E - E_s)
= S_{\text{res}}(E) - E_s\,\partial S_{\text{res}}/\partial E =
S_{\text{res}}(E) - E_s/T$ por la definición de temperatura. De ahí que
$\Omega_{\text{res}} \propto \eu^{-E_s/k_{\text{B}}T}$. (Los órdenes
superiores mueren con el tamaño del reservorio).
\end{proof}

\begin{proposition}[La máquina]\label{prop:b3:canonical-ensemble:machine}
\index{energía libre}\index{capacidad calorífica!y fluctuaciones}
A partir de $Z(T, V, N)$:
\[
\langle E\rangle = -\frac{\partial\ln Z}{\partial\beta} , \qquad
F = -k_{\text{B}}T\ln Z , \qquad
S = -\frac{\partial F}{\partial T} , \qquad
P = -\frac{\partial F}{\partial V} ,
\]
donde $F = \langle E\rangle - TS$ es la \emph{\omterm{prop:b3:canonical-ensemble:machine}{energía libre}}; y
la capacidad calorífica es una \emph{fluctuación}:
\[
C = \frac{\partial\langle E\rangle}{\partial T}
 = \frac{\langle E^2\rangle - \langle E\rangle^2}
{k_{\text{B}}T^2} .
\]
Para subsistemas independientes y distinguibles, $Z = z^N$; para $N$
\omterm{def:b3:identical-particles:exchange}{partículas idénticas} en una caja común, $Z = z^N/N!$. A $T$ y $V$
fijos, el equilibrio \emph{minimiza} $F$: la naturaleza cambia energía por
entropía al tipo de cambio $T$ --- el principio más útil, con diferencia, de
toda la física de la materia.
\end{proposition}

\begin{proof}[Demostración parcial]
$-\partial_\beta\ln Z = \sum E_s\eu^{-\beta E_s}/Z = \langle
E\rangle$; una derivada más da $\langle E^2\rangle - \langle
E\rangle^2$, y la regla de la cadena convierte $\partial_\beta$ en
$\partial_T$. Las identificaciones de $F$, $S$ y $P$ se siguen de
comparar $\dd(-k_{\text{B}}T\ln Z)$ con el
$\dd F = -S\dd T - P\dd V$ termodinámico; la factorización es la
exponencial de una suma. El principio de mínimo: que $F_{\text{sistema}}$
decrezca es que $S_{\text{total}}$ crezca, ya que $\Delta S_{\text{res}} =
-\Delta E_{\text{sist}}/T$.
\end{proof}

\section{Primeras victorias}

\begin{example}[Dos niveles, en dos líneas]\label{ex:b3:canonical-ensemble:twolevel}
Para niveles $0, \epsilon$: $z = 1 + \eu^{-\beta\epsilon}$, lo que da
$\langle E\rangle = N\epsilon/(\eu^{\beta\epsilon} + 1)$ --- el
resultado que a la ruta microcanónica le costó una página de Stirling
(\cref{exo:b3:microcanonical-ensemble:8}). El formalismo canónico
es el microcanónico con la combinatoria predigerida; la joroba de Schottky
en $C(T)$ se sigue con una sola derivación.
\end{example}

\begin{example}[El sólido de Einstein y la muerte de Dulong--Petit]\label{ex:b3:canonical-ensemble:einstein}
\index{sólido de Einstein}\index{ley de Dulong--Petit}
Modelícese un cristal como $3N$ osciladores cuánticos de frecuencia $\omega$
(\cref{ch:b3:harmonic-oscillator}). Por oscilador,
\[
z = \sum_n\eu^{-\beta\hbar\omega(n + 1/2)}
 = \frac{1}{2\sinh(\beta\hbar\omega/2)} , \qquad
\langle E\rangle = \frac{\hbar\omega}{2} +
\frac{\hbar\omega}{\eu^{\beta\hbar\omega} - 1} .
\]
A $T$ alta: $\langle E\rangle \to k_{\text{B}}T$ por oscilador y
$C \to 3Nk_{\text{B}}$ --- la \omterm{ex:b3:canonical-ensemble:einstein}{ley de Dulong--Petit} del volumen del Año 1,
explicada. A $T$ baja, el cuanto $\hbar\omega$ se vuelve
inasequible y $C$ se desploma exponencialmente --- tal como se mide, e
inexplicable clásicamente. La curva de Einstein de 1907, con un parámetro
por elemento, fue la primera aplicación de los cuantos a la materia
ordinaria; el diamante, con enlaces rígidos y átomos ligeros ($\hbar\omega/k_{\text{B}}
\approx \qty{1300}{K}$), sigue ``congelado'' a
temperatura ambiente --- la anomalía que había desconcertado a los químicos
durante ochenta años (\cref{exo:b3:canonical-ensemble:6}).
\end{example}

\begin{theorem}[Equipartición, con su licencia]\label{thm:b3:canonical-ensemble:equipartition}
\index{teorema de equipartición}
Toda coordenada o momento que entre \emph{cuadráticamente} en la energía
aporta, en el régimen clásico (de alta temperatura),
\[
\langle\epsilon\rangle = \tfrac12 k_{\text{B}}T
\]
a la energía media: $\tfrac32 k_{\text{B}}T$ para un átomo de gas
monoatómico, $\tfrac52$ para una diatómica que rota y $3k_{\text{B}}T$ para
un oscilador. La licencia caduca cuando $k_{\text{B}}T$ cae por debajo del
espaciado de niveles del modo: el modo se congela y su contribución
se desvanece --- la resolución de los escándalos de las capacidades
caloríficas del siglo XIX, dibujada como la escalera de arriba.
\end{theorem}

\begin{proof}[Demostración parcial]
Para $\epsilon = ax^2$,
\[
\langle\epsilon\rangle
 = \frac{\int ax^2\,\eu^{-\beta ax^2}\dd x}
        {\int\eu^{-\beta ax^2}\dd x}
 = -\partial_\beta\ln\!\int\eu^{-\beta ax^2}\dd x
 = -\partial_\beta\ln\beta^{-1/2} = \frac{1}{2\beta} .
\] La congelación es el cálculo del
\cref{ex:b3:canonical-ensemble:einstein}, modo a
modo.
\end{proof}

\begin{example}[El gas, canónicamente]\label{ex:b3:canonical-ensemble:gas}
Un átomo en una caja: $z = V/\lambda_T^3$ (el recuento de estados de la
\cref{prop:b3:schrodinger-three-dimensions:counting} pesado con
Boltzmann); $N$ átomos idénticos: $Z = z^N/N!$. Entonces $F =
-Nk_{\text{B}}T[\ln(V/N\lambda_T^3) + 1]$, y la derivación
entrega $PV = Nk_{\text{B}}T$, la entropía de Sackur--Tetrode y
$\langle E\rangle = \tfrac32 Nk_{\text{B}}T$ --- todo el gas ideal
a partir de una integral gaussiana. El factor de Boltzmann aplicado a la
energía cinética de una molécula da la distribución de velocidades de
Maxwell del volumen del Año 1, ahora deducida; aplicado a su energía
potencial $mgh$ da la atmósfera exponencial --- y, en una gota de
agua, la escalera de granos de Perrin
(\cref{pb:b3:canonical-ensemble:1}).
\end{example}

\begin{method}[El oficio canónico]\label{met:b3:canonical-ensemble:method}
(1) Enumérense los estados y las energías de \emph{una} unidad; calcúlese $z$.
(2) Factorícese: $Z = z^N$ (o $z^N/N!$ para \omterm{def:b3:identical-particles:exchange}{partículas idénticas}
que comparten espacio); tómese pronto el $\ln$. (3) Deríve: respecto de
$\beta$ para la energía, respecto de $T$ para la entropía vía $F$ y
respecto de $V$ para la presión; una segunda derivada para $C$ y las
fluctuaciones. (4) Compruébense los dos extremos: a $T$ alta debe
reproducirse la equipartición y a $T$ baja debe congelarse con una cola
$\eu^{-\Delta/k_{\text{B}}T}$. (5) Competiciones (plegamiento,
unión, alineamiento): escríbase $F = E - TS$ para cada alternativa y
gane la menor --- el cruce se sitúa en $T \approx \Delta
E/\Delta S$.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:1}
(a) Escríbase la razón de poblaciones de dos niveles separados por $\epsilon$
a temperatura $T$. (b) En una llama de \qty{2500}{K}, ¿qué fracción de los
átomos de sodio ocupa el estado excitado de \qty{2.1}{eV} de la línea D
(degeneración 2 del fundamental y 6 del excitado: razón de poblaciones
$3\eu^{-\beta\epsilon}$)?
(c) ¿Por qué arde la llama en amarillo pese a todo (cuántos átomos por
cm$^3$ bastan)? (d) ¿A qué temperatura alcanzaría el $10\%$ la fracción
excitada?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:2}
Un sistema de tres niveles: $0, \epsilon, 2\epsilon$. (a) Escríbase $z$. (b)
Calcúlese $\langle E\rangle$ y compruébense los dos límites de temperatura. (c)
¿A qué $T$ está el nivel intermedio máximamente poblado \emph{en términos
absolutos}? (d) Muéstrese que ninguna temperatura, por alta que sea, hace
que un nivel superior esté más poblado que uno inferior --- ¿qué concepto
del capítulo 16 haría falta para eso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:3}
La atmósfera isoterma. (a) Aplíquese el factor de Boltzmann a la
energía potencial $mgh$ y dedúzcase $n(h) = n_0\eu^{-mgh/k_{\text{B}}
T}$. (b) Calcúlese la altura de escala para el aire ($m = \qty{4.8e-26}{kg}$)
a \qty{288}{K}. (c) La presión en la cumbre del Everest como fracción de la
del nivel del mar. (d) ¿Por qué es la caída de la atmósfera real próxima a
una exponencial, pero no exactamente (qué mantuvimos constante que no lo es)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:4}
Contabilidad de la equipartición. Cuéntense los términos cuadráticos y predígase
el $C_V$ molar de (a) el argón; (b) el N$_2$ a temperatura ambiente
(con la rotación activa y la vibración congelada); (c) el N$_2$ a
\qty{3000}{K}; (d) un sólido cristalino (Dulong--Petit). ¿Dónde coinciden
los valores medidos $12.5$, $20.8$, $\approx26$ y
$\approx\qty{25}{J/(mol.K)}$, y qué enseña cada discrepancia?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:5}
El oscilador cuántico, canónicamente. (a) Súmese la serie geométrica
para $z$. (b) Dedúzcase $\langle E\rangle$ e identifíquese $\langle
n\rangle = 1/(\eu^{\beta\hbar\omega} - 1)$ --- el resultado del
\cref{exo:b3:harmonic-oscillator:6}, ahora sin esfuerzo. (c)
Dedúzcase para obtener $C(T)$ y verifíquense los dos límites. (d) ¿Por qué está a
punto de hacerse famosa la forma de $\langle n\rangle$
(\cref{ch:b3:photons-phonons})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:6}
Einstein frente a los datos. (a) A partir de la fórmula de la figura, ¿a qué
$T/\theta_{\text{E}}$ ha caído $C$ a la mitad del valor de Dulong--Petit? (b)
Diamante: $\theta_{\text{E}} \approx \qty{1300}{K}$ --- calcúlese
$C/3Nk_{\text{B}}$ a \qty{300}{K} y explíquese la ``anomalía del diamante''
del siglo XIX. (c) Plomo: $\theta_{\text{E}}
\approx \qty{90}{K}$ --- ¿por qué se portó siempre bien el plomo? (d)
Las medidas a muy baja $T$ muestran $C \propto T^3$, y no una
exponencial: ¿qué hipótesis de Einstein falla (que todos los
osciladores tengan \emph{una} sola frecuencia) y quién la reparó
(\cref{ch:b3:photons-phonons})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:7}
La cola de Maxwell y el hidrógeno que falta. (a) A partir del factor de
Boltzmann sobre la energía cinética, escríbase la distribución de velocidades
y localícese la velocidad más probable del N$_2$ y del H$_2$ a \qty{288}{K}. (b)
La velocidad de escape de la Tierra es \qty{11.2}{km/s}: calcúlese
$mv_{\text{esc}}^2/2k_{\text{B}}T$ para ambos gases. (c) La fracción que
escapa va como $\eu^{-mv_{\text{esc}}^2/2k_{\text{B}}T}$: compárense
los dos exponentes y conclúyase qué gas se fuga a lo largo del tiempo
geológico. (d) Conéctese con la composición observada de las atmósferas de
la Tierra (sin H$_2$ libre) y de Júpiter (sobre todo H$_2$) --- ¿qué dos
parámetros invierten el veredicto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:8}
Las fluctuaciones se encuentran con la respuesta. (a) Muéstrese $C = (\langle E^2\rangle -
\langle E\rangle^2)/k_{\text{B}}T^2$ a partir de dos derivadas de $\ln
Z$. (b) Verifíquese explícitamente en el sistema de dos niveles. (c) Para $N$
unidades independientes, muéstrese que la fluctuación \emph{relativa} de
energía cae como $1/\sqrt N$. (d) Enúnciese la moraleja: qué tiene que ver la
disposición de un sistema a absorber calor (una respuesta) con cuánto
tiembla su energía (una fluctuación) --- el pacto recurrente de la física
estadística.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:9}
Boltzmann en el laboratorio de química. Los ritmos de reacción llevan el
factor $\eu^{-E_a/k_{\text{B}}T}$ (cruzar una barrera de activación $E_a$
--- Arrhenius). (a) Muéstrese que la regla práctica ``el ritmo se duplica
cada \qty{10}{K} cerca de la temperatura ambiente'' corresponde a $E_a \approx
\qty{0.55}{eV}$. (b) ¿En qué factor se ralentiza esa reacción en un
frigorífico (\qty{5}{\celsius})? (c) Cocer un huevo en altura: el agua
hierve a \qty{93}{\celsius} en un puerto de \qty{2000}{m} ---
estímese el tiempo de cocción adicional. (d) ¿Por qué gobierna la química una
cola por encima de la barrera y no la energía media (qué moléculas
reaccionan)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:10}
Plegamiento de dos estados. Una biomolécula está plegada (energía $0$, una
configuración) o desplegada (energía $\Delta E > 0$, con $\Omega_u =
\eu^{\Delta S/k_{\text{B}}}$ configuraciones). (a) Escríbase la fracción
plegada frente a $T$. (b) Muéstrese que el punto medio de ``fusión'' es
$T_{\text{m}} = \Delta E/\Delta S$ e interprétese como un empate en $F = E -
TS$. (c) Con $\Delta E = \qty{3.0}{eV}$ y $\Delta S =
100\,k_{\text{B}}$ (la unidad cooperativa de una proteína pequeña): calcúlense
$T_{\text{m}}$ y la anchura de la transición. (d) ¿Por qué afila la fusión
la \emph{cooperatividad} (muchos contactos que se rompen a la vez, con
$\Delta E$ \emph{y} $\Delta S$ grandes), y cómo explotan exactamente esto
las máquinas de ciclado térmico de ADN (la PCR)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:11}
Paramagnetismo y enfriamiento magnético. $N$ espines $\tfrac12$ de momento
$\mu$ en un campo $B$. (a) Muéstrese que $M = N\mu\tanh(\mu B/k_{\text{B}}T)$
y desarróllese hasta la ley de Curie $M \approx N\mu^2B/k_{\text{B}}T$. (b)
Evalúese el alineamiento $\mu B/k_{\text{B}}T$ para momentos electrónicos
con $B = \qty{1}{T}$ y $T = \qty{300}{K}$, y luego a \qty{1}{K}. (c)
Desimanación adiabática: imánese a \qty{1}{K}, aíslese y
redúzcase $B$ diez veces --- muéstrese que entropía constante significa
$\mu B/k_{\text{B}}T$ constante, de modo que $T$ cae diez veces. (d) ¿Qué
fija el suelo de este refrigerador (las interacciones entre los espines ---
estímese la escala dipolar $\mu_0\mu^2/4\pi a^3$ para $a =
\qty{0.5}{nm}$, en unidades de temperatura)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:12}
La escalera rotacional, cuantitativamente. Los niveles rotacionales de una
diatómica son $E_J = BJ(J+1)$, con degeneración $2J + 1$
(\cref{ch:b3:quantum-angular-momentum}). (a) Escríbase $z_{\text{rot}}$
y muéstrese que, para $k_{\text{B}}T \gg B$, la suma se convierte en la
integral $k_{\text{B}}T/B$: el $k_{\text{B}}$ de la equipartición en
$C_V$, recuperado. (b) Defínanse $\theta_{\text{rot}} =
B/k_{\text{B}}$ y evalúese para el H$_2$ ($B = \qty{7.5}{meV}$) y el
N$_2$ ($B = \qty{0.25}{meV}$): ¿qué gas muestra el escalón rotacional
a temperaturas accesibles? (c) Esbócese la escalera completa $C_V(T)$ del
hidrógeno con sus tres mesetas y sus dos contrahuellas, poniendo números
en ambas. (d) El comportamiento medido a $T$ baja del H$_2$ se complica
además por la mezcla orto--para 3:1 del
\cref{exo:b3:identical-particles:8}: enúnciese en una frase cómo llega la
estadística de espines nucleares hasta la capacidad calorífica de un gas.
\end{exercise}

\section{Problema: contar el número de Avogadro en una gota de agua}

\begin{problem}[{Problema de fin de semana --- los granos de Perrin y la
realidad de los átomos}]\label{pb:b3:canonical-ensemble:1}
En 1908, Jean Perrin suspendió granos microscópicos de resina en agua,
dejó que sedimentaran y los contó capa por capa bajo el
microscopio. La escalera exponencial que halló era el factor de Boltzmann
hecho visible --- y de su altura de escala extrajo el
número de Avogadro, con lo que convenció a los últimos escépticos de que los
átomos existen. Premio Nobel de 1926. Datos: granos de gutagamba de radio $r =
\qty{0.212}{\micro m}$ y densidad $\rho_{\text{g}} =
\qty{1207}{kg/m^3}$; agua con $\rho_{\text{w}} = \qty{999}{kg/m^3}$;
$T = \qty{293}{K}$; $g = \qty{9.81}{m/s^2}$.

\medskip\noindent\textbf{Parte I --- El factor de Boltzmann visible.}
\begin{enumerate}
  \item Un grano en agua siente la gravedad menos el empuje: calcúlese su
        masa \emph{efectiva} $m' = \tfrac43\pi r^3(\rho_{\text{g}}
        - \rho_{\text{w}})$ y su peso.
  \item Escríbase el perfil de concentración en equilibrio $n(h)$ a partir
        del factor de Boltzmann.
  \item Calcúlese la altura de escala $h_0 = k_{\text{B}}T/m'g$ con
        el $k_{\text{B}}$ moderno.
  \item ¿Por qué deben ser los granos de tamaño tan preciso (cómo depende
        $h_0$ de $r$)?
  \item Compárese $h_0$ con la misma fórmula para las moléculas del aire:
        ¿por qué es la atmósfera de los granos de micrómetros de altura
        mientras que la del aire es de kilómetros?
  \item Perrin contó (en una tanda) concentraciones relativas
        $100 : 55 : 30 : 17$ a cuatro profundidades igualmente espaciadas
        \qty{30}{\micro m}: compruébese que es una escalera
        exponencial y extráigase su $h_0$.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Pesar lo invisible.}
\begin{enumerate}[resume]
  \item Inviértase: a partir del $h_0$ medido en el apartado 6 y del
        $m'g$ conocido, extráigase $k_{\text{B}}$.
  \item La constante de los gases $R = \qty{8.314}{J/(mol.K)}$ se conocía
        por la química macroscópica: combínese con su
        $k_{\text{B}}$ para obtener el número de Avogadro $N_{\text{A}} =
        R/k_{\text{B}}$.
  \item Las tandas de Perrin dieron $N_{\text{A}}$ entre $5.5$ y
        $\num{7.2e23}$: compárese con el moderno
        \num{6.022e23} y coméntese el logro dados su microscopio y su
        cronómetro.
  \item A partir de $N_{\text{A}}$, calcúlese la masa de un solo
        \omterm{thm:b3:hydrogen-atom:levels}{átomo de hidrógeno} --- el número que los atomistas habían deseado
        durante un siglo.
  \item Explíquese la estructura lógica: ¿qué medidas \emph{macroscópicas}
        ($R$; el tamaño y la densidad de los granos; la escalera) se
        combinan para pesar un átomo sin haber visto jamás ninguno?
  \item ¿Por qué obedecen los granos --- de $10^{10}$ masas atómicas cada
        uno --- la misma estadística de Boltzmann que las moléculas (qué
        hay en la deducción del
        \cref{thm:b3:canonical-ensemble:boltzmann} a lo que le importe el
        tamaño)?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El temblor que lo sella.}
\begin{enumerate}[resume]
  \item Esos mismos granos tiemblan: la fórmula de Einstein de 1905 para
        el movimiento browniano da $\langle x^2\rangle = 2Dt$ con $D =
        k_{\text{B}}T/6\pi\eta r$ (agua: $\eta =
        \qty{e-3}{Pa.s}$). Calcúlese $D$ para los granos de Perrin.
  \item ¿Cuánto vaga un grano en un minuto? ¿Podía Perrin
        medirlo con un ocular micrométrico y un cronómetro?
  \item Perrin verificó $\langle x^2\rangle \propto t$ y
        extrajo $k_{\text{B}}$ \emph{otra vez}, de forma independiente:
        ¿por qué tuvieron tanto peso probatorio dos rutas sin relación
        (una escalera estática y un temblor dinámico) hacia un mismo
        número?
  \item El temblor es la equipartición aplicada al grano: cada
        componente de la velocidad lleva $\tfrac12 k_{\text{B}}T$.
        Estímese la velocidad térmica cuadrática media del grano ($m \approx
        \qty{4.8e-17}{kg}$).
  \item ¿Por qué no se ve nunca esa velocidad directamente (qué
        interrumpe el vuelo libre tras unos nanómetros) y qué es lo que
        \emph{sí} se ve en su lugar?
  \item Dígase qué dos capítulos de este libro se encuentran en la
        observación: la mecánica del arrastre (volumen del Año 2, la
        viscosidad) y la estadística de este capítulo.
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- Qué quedó zanjado.}
\begin{enumerate}[resume]
  \item Ostwald y Mach sostenían que los átomos eran ficciones
        contables: dígase en una frase por qué acabó con esa postura un
        $N_{\text{A}}$ \emph{contado} a partir de escaleras de granos.
  \item Enumérense otras tres rutas de la década de 1900 que convergieron en
        el mismo $N_{\text{A}}$ (el azul del cielo,
        \cref{exo:b3:microcanonical-ensemble:12}; la electrólisis
        más la carga del electrón; la producción de helio por
        radiactividad) --- ¿por qué importó más la \emph{convergencia}
        que cualquier valor aislado?
  \item La escalera de Perrin es un equilibrio entre qué dos monedas de
        este capítulo (la energía que tira hacia abajo y la entropía que
        reparte hacia arriba), tasado ¿a qué tipo de cambio?
  \item Usos modernos de la misma física: las ultracentrifugadoras
        analíticas hacen girar proteínas a $10^5g$ para comprimir
        sus ``atmósferas'' en escaleras medibles --- muéstrese
        que multiplicar $g$ por $10^5$ divide $h_0$ por ese mismo
        factor, y estímese $h_0$ para una proteína de masa efectiva
        \qty{e-22}{kg} a $10^5g$ y \qty{293}{K}.
  \item Duplicar el radio del grano divide $h_0$ por ocho: acótese
        la ventana práctica de tamaños de grano entre ``escalera demasiado
        alta para ver un gradiente'' y ``escalera más fina que un
        grano'' para un campo de microscopio de \qty{100}{\micro m} de
        profundidad.
  \item Desde 2019, el SI \emph{define} exactamente $k_{\text{B}}$ y
        $N_{\text{A}}$: dígase qué mide \emph{hoy} un experimento exacto al
        estilo de Perrin (¿una comprobación de coherencia o una
        calibración del aparato y de los granos?) --- y por qué no cambia
        la física.
  \item Resúmase el resultado con nombre propio: la concentración de un
        grano de \qty{0.2}{\micro m} se reduce a la mitad cada
        $\sim\qty{35}{\micro m}$ de altura; leída con
        $n(h) = n_0\eu^{-m'gh/k_{\text{B}}T}$,
        esa escalera dio $N_{\text{A}} \approx \num{6e23}$ ---
        átomos contados, y no conjeturados, en una gota de agua sobre la
        platina de un microscopio.
\end{enumerate}
\end{problem}
