\chapter{Estadística cuántica}\label{ch:b3:quantum-statistics}

Los electrones de conducción del cobre forman un gas mil veces más denso que
el aire --- y, sin embargo, casi no añaden nada a la capacidad calorífica del
cobre, escándalo que atormentó a la física clásica durante cincuenta años. Una
nube de átomos de rubidio, enfriada hasta unas cien milmillonésimas de kelvin,
colapsa de pronto en un solo estado cuántico con diez mil átomos
dentro. Una estrella muerta con la masa del Sol y el tamaño de la Tierra
se niega a encogerse más, sostenida por nada más que el principio de
exclusión. Las tres cosas son un mismo capítulo de la física: lo que le ocurre
a un gas cuando $n\lambda_T^3 \sim 1$ y las fórmulas de ocupación del
\cref{ch:b3:grand-canonical} se apartan de Boltzmann. Los fermiones construyen
mares --- fríos, rígidos, a presión; los bosones construyen condensados ---
gregarios, coherentes, superfluidos. Entre la mayoría silenciosa del mar de
Fermi y la estampida del condensado cabe casi toda la física moderna de la
materia condensada y de las bajas temperaturas, y el destino de las
estrellas.

\section{Gases de ocupaciones}

\begin{proposition}[La maquinaria del gas cuántico]\label{prop:b3:quantum-statistics:machinery}
\index{densidad de estados}
Para partículas sin interacción en una caja, las sumas sobre niveles se
convierten en integrales sobre la \omterm{prop:b3:schrodinger-three-dimensions:counting}{densidad de estados} de la
\cref{prop:b3:schrodinger-three-dimensions:counting} (por estado
de espín, $g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$):
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
con $\bar n$ la ocupación de Fermi--Dirac o de Bose--Einstein. La
primera ecuación fija $\mu(T, n)$; la segunda entrega la
termodinámica. En el límite diluido $n\lambda_T^3 \ll 1$, ambas
estadísticas colapsan sobre Boltzmann y regresa el gas clásico;
los regímenes cuánticos empiezan donde los paquetes de onda se tocan.
\end{proposition}
\admitted

\section{El mar de Fermi}

\begin{theorem}[El gas de Fermi degenerado]\label{thm:b3:quantum-statistics:fermi}
\index{energía de Fermi}\index{presión de degeneración}
A $T = 0$, los fermiones llenan todos los niveles hasta la \emph{\omterm{thm:b3:quantum-statistics:fermi}{energía
de Fermi}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{espín } \tfrac12) ,
\]
con energía media $\tfrac35 E_{\text{F}}$ por partícula y la
\emph{\omterm{thm:b3:quantum-statistics:fermi}{presión de degeneración}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
una presión en el cero absoluto, de origen puramente cuántico --- la
rigidez del gas electrónico de todo metal y el sostén de las estrellas
muertas. Para $0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$,
solo los electrones que están a $\sim k_{\text{B}}T$ de la superficie de
este \emph{mar de Fermi} pueden responder a nada: la fracción
$\sim T/T_{\text{F}}$ es la llave maestra del comportamiento metálico.
\end{theorem}

\begin{proof}[Demostración parcial]
Llénese $g(\epsilon)$ hasta $E_{\text{F}}$: invertir $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ da la $E_{\text{F}}$ enunciada
(\cref{exo:b3:schrodinger-three-dimensions:8}); $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ con una integral más. La presión:
comprimir la caja sube todos los niveles como $V^{-2/3}$, luego $E
\propto V^{-2/3}$ y $P = -\partial E/\partial V = \tfrac23 E/V$
--- con $E = \tfrac35 NE_{\text{F}}$, la ley enunciada.
\end{proof}

\begin{example}[El escándalo de la capacidad calorífica, resuelto]\label{ex:b3:quantum-statistics:heat}
\index{capacidad calorífica electrónica}
Clásicamente, el electrón libre por átomo del cobre debería añadir $\tfrac32
k_{\text{B}}$ a la capacidad calorífica --- un $50\%$ por encima de
Dulong--Petit. Lo medido: alrededor del $1\%$ a temperatura ambiente. El mar
lo explica: con $T_{\text{F}} \approx \qty{8e4}{K}$, solo la
fracción $\sim T/T_{\text{F}} \approx 0.4\%$ de los electrones puede
excitarse térmicamente, cada uno en $\sim k_{\text{B}}T$:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(el coeficiente exacto de Sommerfeld;
\cref{exo:b3:quantum-statistics:12}). Lineal en $T$, queda sepultada
bajo el $T^3$ de la red a temperaturas ordinarias, pero
\emph{domina} por debajo de unos pocos kelvin --- exactamente lo que
encuentran los calorímetros, y resuelve de un golpe el enigma de los
electrones que transportan corriente pero no capacidad calorífica.
\end{example}

\begin{example}[Estrellas sostenidas por la exclusión]\label{ex:b3:quantum-statistics:dwarf}
\index{enana blanca}
Una \omterm{ex:b3:quantum-statistics:dwarf}{enana blanca} (\cref{exo:b3:identical-particles:11}) equilibra la
gravedad con la \omterm{thm:b3:quantum-statistics:fermi}{presión de degeneración} electrónica. Como $P \propto
n^{5/3}$ se endurece más deprisa que las exigencias de la gravedad a medida
que la estrella se encoge, existe un radio de equilibrio para cualquier masa
modesta --- con el inquietante escalado $R \propto M^{-1/3}$: \emph{las
enanas más pesadas son más pequeñas}. Al apretarlas más, los electrones se
vuelven relativistas, la ley de presión se ablanda hasta $n^{4/3}$ y el
equilibrio falla en una masa única, el límite de Chandrasekhar $M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) ---
más allá, el colapso a una estrella de neutrones: un mar de Fermi de
neutrones a densidad nuclear, la misma física a $10^{14}$ veces la densidad.
\end{example}

\section{La estampida de Bose}

\begin{theorem}[Condensación de Bose--Einstein]\label{thm:b3:quantum-statistics:bec}
\index{condensación de Bose--Einstein}
Para un gas de $N$ bosones conservados, los estados excitados pueden alojar
como mucho $N_{\max}(T) = 2.612\,V/\lambda_T^3$ partículas (la integral
de $g\bar n_{\text{BE}}$ en su techo $\mu \to 0$). Por debajo de la
temperatura crítica
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(equivalentemente } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
el excedente no tiene otro sitio adonde ir que el único estado fundamental:
una población \emph{macroscópica}
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
se condensa en un solo estado cuántico --- materia que se comporta como una
única función de onda gigante. Predicha por Einstein en 1925 a partir de la
estadística fotónica de Bose; realizada en rubidio diluido en 1995 a
$T_{\text{c}} \approx \qty{100}{nK}$ (Nobel de 2001), y anunciada por
la reveladora distribución de velocidades \emph{bimodal}: una nube térmica
con una aguja de condensado a velocidad nula que se alza en su centro.
\end{theorem}
\admitted

\begin{example}[El helio, superfluido veterano]\label{ex:b3:quantum-statistics:helium}
\index{superfluidez}
El $^4$He líquido (bosónico) se transforma a $T_\lambda = \qty{2.17}{K}$
en un \emph{superfluido}: viscosidad nula a través de los capilares más
finos, flujo en película por las paredes del recipiente, efectos de fuente y
circulación cuantizada en unidades de $h/m$ --- la fase macroscópica de un
condensado en acción, desplazada respecto de la predicción de gas ideal
($\qty{3.1}{K}$ para la densidad del helio) por las interacciones, pero
inconfundiblemente el mismo fenómeno. El $^3$He fermiónico no puede condensar
solo: sus átomos deben \emph{aparearse} antes (como los electrones de un
superconductor, \cref{ch:b3:electrons-in-solids}) en bosones
compuestos --- y la superfluidez llega en efecto, pero a \qty{2.6}{mK}, mil
veces más frío: dos isótopos, una lección --- la estadística es el destino.
\end{example}

\begin{method}[Qué estadística, qué régimen]\label{met:b3:quantum-statistics:method}
(1) Calcúlese primero $n\lambda_T^3$: $\ll1$ significa Boltzmann y basta
con el capítulo 17. (2) Fermiones degenerados: piénsese en
$E_{\text{F}}$ y $T/T_{\text{F}}$; las respuestas (capacidad calorífica,
susceptibilidad, dispersión) llevan el factor $T/T_{\text{F}}$ o
solo muestrean la superficie de Fermi. (3) Bosones degenerados: compárese $T$
con $T_{\text{c}}$; por debajo, sepárese el gas en condensado más
nube térmica. (4) Presiones: la rigidez fermiónica $n^{5/3}$ (o
$n^{4/3}$ relativista) frente a lo que sea que apriete. (5) Partículas
compuestas: cuéntense los fermiones constituyentes para elegir la
estadística --- y recuérdese que el apareamiento puede convertir una
estadística en la otra.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:1}
Calcúlense $E_{\text{F}}$ y $T_{\text{F}}$ para (a) el cobre
($n = \qty{8.5e28}{m^{-3}}$); (b) el sodio ($\qty{2.5e28}{m^{-3}}$);
(c) el $^3$He líquido ($\qty{1.6e28}{m^{-3}}$, $m = 3u$); (d) ordénense
frente a la temperatura ambiente y coméntese quién está degenerado y cuándo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:2}
(a) Muéstrese que $\langle\epsilon\rangle = \tfrac35 E_{\text{F}}$ a $T =
0$. (b) Calcúlese la velocidad cuadrática media de los electrones del cobre. (c) ¿A
qué temperatura corresponde clásicamente esa velocidad? (d) ¿Por qué contiene
un metal frío electrones que se mueven a una centésima de la velocidad de la
luz sin radiar esa energía?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:3}
\omterm{thm:b3:quantum-statistics:fermi}{Presión de degeneración} de los electrones del cobre: (a) evalúese $P =
\tfrac25 nE_{\text{F}}$ en pascales y en atmósferas. (b) ¿Por qué no
explota el metal (qué tira en sentido contrario)? (c) Esta presión resiste
la compresión: relaciónese con la baja compresibilidad de los metales. (d) Estímese
el módulo de compresibilidad $K \sim \tfrac53 P$ y compárese con los
\qty{140}{GPa} medidos del cobre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:4}
Umbrales de condensación. (a) Calcúlese $T_{\text{c}}$ para el rubidio-87 ($m =
\qty{1.44e-25}{kg}$) a $n = \qty{e20}{m^{-3}}$. (b) Con la misma
$n$, ¿qué $T_{\text{c}}$ tendrían los \omterm{thm:b3:hydrogen-atom:levels}{átomos de hidrógeno}? (c) ¿Por qué usan
los experimentos gases diluidos a nanokelvin en vez de gases
densos a milikelvin (qué ocurre químicamente a densidad alta y
temperatura baja)? (d) Compruébese la regla de consistencia $n\lambda_{T_c}^3
= 2.612$ con sus números del rubidio.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:5}
Dos capacidades caloríficas. En un metal, $C = \gamma T +
\beta T^3$ (electrones; red). (a) Usando el
\cref{ex:b3:quantum-statistics:heat} y el término de red de tipo Debye
$\beta T^3 = 234\,Nk_{\text{B}}(T/\theta_{\text{D}})^3$,
estímese la temperatura de cruce para el cobre
($\theta_{\text{D}} = \qty{343}{K}$, $T_{\text{F}} =
\qty{8.2e4}{K}$). (b) A \qty{300}{K}, la parte electrónica en
porcentaje. (c) ¿Y a \qty{0.1}{K}? (d) ¿Por qué representan los experimentadores $C/T$
frente a $T^2$, y qué entregan la ordenada en el origen y la pendiente de esa
recta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}
Una \omterm{ex:b3:quantum-statistics:dwarf}{enana blanca} de una masa solar ($M = \qty{2e30}{kg}$) y radio
terrestre ($R = \qty{6.4e6}{m}$), con un electrón por cada dos nucleones. (a)
Calcúlense $n_{\text{e}}$ y $E_{\text{F}}$ (con la fórmula no
relativista). (b) Compárese $E_{\text{F}}$ con $m_{\text{e}}c^2$ y
con $k_{\text{B}}T$ a los \qty{e7}{K} del interior: justifíquese
``fría'' y ``casi relativista'' a la vez. (c) Compárese
la \omterm{thm:b3:quantum-statistics:fermi}{presión de degeneración} $\tfrac25 nE_{\text{F}}$ con la
estimación de presión gravitatoria $\sim GM^2/R^4$: ¿del mismo orden? (d) ¿Qué le ocurre cualitativamente a
cada lado del equilibrio si $M$ crece?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:7}
Materia de neutrones. Una estrella de neutrones comprime $M = 1.4\,M_\odot$ en $R
\approx \qty{11}{km}$. (a) Calcúlese la densidad de neutrones y compárese
con la densidad nuclear ($\qty{2.3e17}{kg/m^3}$). (b) Calcúlese
$E_{\text{F}}$ para los neutrones (de forma no relativista; $m_{\text{n}}c^2
= \qty{939}{MeV}$) y compruébese cuán relativistas son. (c) ¿Por qué
desaparecieron los electrones de la estrella (desintegración beta inversa:
$\text{e} + \text{p} \to \text{n} + \nu$ --- qué hizo que la
$E_{\text{F}}$ de los electrones la pagara)? (d) Una cucharadita de esta
materia, ¿cuánto pesa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:8}
Bosones sin número. (a) Los fotones tienen siempre $\mu = 0$: ¿por qué no
pueden sufrir la condensación del
\cref{thm:b3:quantum-statistics:bec} (qué hace el gas en su lugar
al enfriarse)? (b) ¿Qué ley de conservación tienen los átomos de rubidio
de la que carecen los fotones? (c) En 2010 \emph{sí} se logró un condensado
de fotones, en una cavidad llena de colorante donde los fotones se termalizan
y su número queda de hecho fijado: ¿qué ingrediente restituyó esa cavidad?
(d) Los fonones de un cristal: ¿condensación o no, y por qué?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
Los dos isótopos del helio. (a) Calcúlese la $T_{\text{c}}$ de gas ideal
para el $^4$He líquido ($n = \qty{2.2e28}{m^{-3}}$, $m = 4u$) y
compárese con $T_\lambda = \qty{2.17}{K}$. (b) ¿Por qué el acuerdo es
solo aproximado (qué ignora ``ideal'' en un líquido)? (c) El $^3$He, a
densidad casi igual, sigue normal hasta los milikelvin: enúnciense
el obstáculo y el rodeo que halló la naturaleza. (d) Comprobación de la lógica de los compuestos: un \emph{par} de $^3$He contiene un número
par de fermiones --- bosón; el $^6$Li alberga $3\text{p} + 3\text{n}
+ 3\text{e} = 9$ fermiones: clasifíquese y nómbrense los célebres
experimentos de gases fríos que explotan justamente ese contraste Li--Li.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ con honradez. (a) Escríbase $N_{\text{exc}} = \int_0^\infty
g(\epsilon)\dd\epsilon/(\eu^{\beta\epsilon} - 1)$ en $\mu = 0$ y
redúzcase, con $x = \beta\epsilon$, a $N_{\text{exc}} =
(V/\lambda_T^3)\cdot\frac{2}{\sqrt\pi}\int_0^\infty\frac{\sqrt
x\,\dd x}{\eu^x - 1}$. (b) Dado que $\int_0^\infty\sqrt x\,\dd x/(
\eu^x - 1) = 2.612\,\sqrt\pi/2$, obténganse el techo y
$T_{\text{c}}$. (c) Dedúzcase la fracción condensada $1 -
(T/T_{\text{c}})^{3/2}$. (d) ¿Por qué produce el mismo argumento
\emph{ninguna} condensación en una caja bidimensional (cómo cambia $g(
\epsilon)$ y qué le ocurre a la convergencia de la
integral)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
Por qué se mantienen unidos los metales. Modélese un metal como electrones de densidad
$n$ (con coste de \omterm{thm:b3:quantum-statistics:fermi}{energía de Fermi} $\propto n^{2/3}$ por electrón) atraídos
por un fondo iónico neutralizador (con ganancia de Coulomb por electrón
$\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (a) Escríbase la energía por
electrón $\epsilon(n) = An^{2/3} - Bn^{1/3}$ y muéstrese que tiene un
mínimo. (b) Muéstrese que el espaciado electrónico de equilibrio es del orden de
$a_0$ --- las densidades metálicas no son accidentes. (c) Estímese
la escala de energía de cohesión en el mínimo, en eV. (d) ¿Qué papel juega la
\omterm{thm:b3:quantum-statistics:fermi}{presión de degeneración} en este estado ligado (qué frena el
colapso de Coulomb)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
La capa de Sommerfeld, cuantificada. (a) Arguméntese: de $N$ electrones, solo
$\sim g(E_{\text{F}})k_{\text{B}}T$ están en la capa activa, cada uno
con una energía extra $\sim k_{\text{B}}T$: dedúzcase $C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ salvo una constante. (b) Con
$g(E_{\text{F}}) = 3N/2E_{\text{F}}$, recupérese $C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ (la constante exacta es $\pi^2/2$).
(c) La misma lógica de la capa da la susceptibilidad de espín de Pauli,
independiente de la temperatura: solo los electrones de la capa pueden
voltear --- pero la capa crece como $T$ mientras cada aportación cae como $1/T$: muéstrese
la cancelación. (d) Enúnciese la moraleja general para los fermiones
degenerados: \emph{toda} respuesta es la clásica multiplicada por
$\sim T/T_{\text{F}}$ o por su análogo de solo superficie.
\end{exercise}

\section{Problema: la enana blanca más pesada posible}

\begin{problem}[{Problema de fin de semana --- el límite de Chandrasekhar y
las velas que encendió}]\label{pb:b3:quantum-statistics:1}
En 1930, con diecinueve años, Subrahmanyan Chandrasekhar calculó en el
barco rumbo a Inglaterra que la degeneración electrónica puede sostener una
estrella muerta solo hasta una masa determinada. El resultado --- al que
Eddington se opuso con encono --- subyace a las estrellas de neutrones, los
agujeros negros y las ``velas estándar'' de supernova que revelaron el
universo en aceleración. Este problema lo construye por escalado. Datos: $M_\odot =
\qty{2e30}{kg}$; un electrón por cada $\mu_e = 2$ nucleones; $m_{\text{
n}} = \qty{1.67e-27}{kg}$; $\hbar c = \qty{3.16e-26}{J.m}$; $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{Parte I --- Equilibrio de dos energías.}
Para una estrella de masa $M$ y radio $R$, trátense las energías por unidad
de masa por escalado.
\begin{enumerate}
  \item Energía gravitatoria: $E_{\text{g}} \sim -GM^2/R$ ---
        recuérdese su origen.
  \item Número de electrones $N_{\text{e}} = M/\mu_e m_{\text{n}}$ y
        densidad $n \sim N_{\text{e}}/R^3$: escríbase la
        escala de \omterm{thm:b3:quantum-statistics:fermi}{energía de Fermi} no relativista en función de $M$
        y $R$.
  \item Muéstrese que la energía cinética (de degeneración) total escala como
        $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item Minimícese $E_{\text{k}} + E_{\text{g}}$ respecto de $R$ y muéstrese
        que el radio de equilibrio es
        $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$.
  \item Léase el escalado $R \propto M^{-1/3}$: las enanas más pesadas
        son \emph{más pequeñas}. ¿Por qué huele eso ya a
        problemas?
  \item Evalúese $R$ para $M = 1\,M_\odot$ y compárese con los
        $\approx\qty{5800}{km}$ medidos de Sirio B.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- La relatividad quita el suelo de en
medio.}
\begin{enumerate}[resume]
  \item A medida que $R$ se encoge, $E_{\text{F}}$ crece: estímese la densidad
        a la que $E_{\text{F}} \sim m_{\text{e}}c^2$ y la
        región de masas de enana correspondiente.
  \item En el régimen ultrarrelativista, la energía de cada electrón es
        $\sim\hbar c\,n^{1/3}$: muéstrese que la energía cinética pasa a ser
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$.
  \item Ambas energías escalan ahora como $1/R$: conclúyase que el
        equilibrio ya no selecciona un radio --- compárese en su lugar los dos
        coeficientes.
  \item Imponiendo $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$, dedúzcanse el
        número crítico de electrones y la masa
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item Evalúese la razón adimensional $\hbar c/Gm_{\text{n}}^2$
        --- uno de los grandes números puros de la física --- y luego
        $M_{\text{Ch}}$ en masas solares (el tratamiento exacto da
        $1.4\,M_\odot$).
  \item Interprétese la fórmula: ¿qué tres constantes conspiran y
        por qué lleva un límite \emph{cuántico} sobre una \emph{estrella}
        una $\hbar$?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Más allá del límite.}
\begin{enumerate}[resume]
  \item Una enana empujada más allá de $M_{\text{Ch}}$ (por acreción de una
        compañera): ¿qué la sostiene a continuación y a qué escala
        de radio (sustitúyase $m_{\text{e}}$ por $m_{\text{n}}$ y
        $\mu_e$ por 1 en el radio de la Parte I)?
  \item Muéstrese que el límite relativista propio de la estrella de neutrones es del
        mismo orden que $M_{\text{Ch}}$ --- la naturaleza solo concede un
        indulto de factor dos (máximo observado $\approx
        2.2\,M_\odot$). ¿Qué espera más allá?
  \item Durante el colapso a estrella de neutrones se libera $\sim GM^2/R_{
        \text{ns}}$ de energía gravitatoria: evalúese
        para $M = 1.4\,M_\odot$ y $R = \qty{11}{km}$, y
        compárese con toda la producción del Sol en \qty{10}{Gyr}
        ($\sim\qty{1.2e44}{J}$).
  \item Casi toda ella se marcha como neutrinos en segundos: ¿qué
        observación de 1987 confirmó este cuadro de forma espectacular?
  \item El rebote del núcleo en colapso y el empuje de los neutrinos expulsan
        la envoltura: una supernova por colapso de núcleo. Distíngase,
        en una frase cada una, de la supernova \emph{termonuclear}
        (de tipo Ia) de una \omterm{ex:b3:quantum-statistics:dwarf}{enana blanca} que se enciende en el
        umbral de Chandrasekhar.
  \item ¿Por qué hace la detonación \emph{a una masa universal} que
        las supernovas de tipo Ia sean casi idénticas --- y por tanto
        utilizables como velas estándar?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- Las velas y el
cosmos.}
\begin{enumerate}[resume]
  \item El brillo aparente de una vela estándar da su
        distancia: enúnciese la lógica del inverso del cuadrado y por qué
        \emph{estándar} es la palabra decisiva.
  \item En 1998, dos equipos hallaron supernovas de tipo Ia lejanas
        \emph{más débiles} de lo esperado: enúnciese la inferencia
        (expansión en aceleración --- energía oscura) y por qué
        la fiabilidad de la vela era el punto crucial.
  \item Enumérese en una línea la cadena de este problema: Pauli $\to$
        \omterm{thm:b3:quantum-statistics:fermi}{presión de degeneración} $\to$ $M_{\text{Ch}}$ $\to$ explosiones
        uniformes $\to$ aceleración cósmica.
  \item Eddington se burló del resultado (``una estrella cometiendo
        absurdos''); Chandrasekhar recibió el Nobel
        cincuenta y tres años después: ¿qué enseña el episodio sobre
        seguir a las ecuaciones más allá de las conclusiones cómodas?
  \item El espectro de Sirio B en 1915 implicaba una tonelada por centímetro
        cúbico --- un ``disparate'' hasta que Fowler, a los pocos meses de
        la estadística de Fermi de 1926, explicó la estrella como un gas
        degenerado: dígase por qué el enigma era irresoluble un
        año y rutinario al siguiente.
  \item Los muestreos de miles de \omterm{ex:b3:quantum-statistics:dwarf}{enanas blancas} encuentran masas
        agrupadas cerca de $0.6\,M_\odot$ y ninguna por encima de
        $\approx1.4\,M_\odot$: ¿qué dos predicciones de este
        problema confirma ese único histograma?
  \item Resúmase el resultado con nombre propio: $(\hbar c/Gm_{\text{n}}^2)^{
        3/2}m_{\text{n}} \approx 1.4\,M_\odot$ --- tres constantes
        de la naturaleza fijando la \omterm{ex:b3:quantum-statistics:dwarf}{enana blanca} más pesada posible, la
        masa que dispara las explosiones calibradas del universo.
\end{enumerate}
\end{problem}
