\chapter{Ensembel Mikrokanonik}\label{ch:b3:microcanonical-ensemble}

Satu sentimeter kubik udara memuat $2.5\times10^{19}$ molekul. Tak akan
pernah ada komputer yang mengintegralkan persamaan geraknya, dan tak ada
percobaan yang dapat memasok syarat awalnya --- namun gasnya menaati
hukum yang sangat sederhana, dan termodinamika jilid Tahun 1 memerikannya
tanpa pernah menyebut satu molekul pun. Bab ini membangun jembatannya:
\emph{fisika statistik}, yang menurunkan hukum kalor dari mekanika
ditambah pencacahan yang jujur. Seluruh bangunannya berdiri di atas satu
postulat --- sistem terkucil sama mungkinnya berada dalam keadaan
mikroskopis mana pun yang selaras dengan energinya --- yang disahkan
teorema Liouville (\cref{ch:b3:hamiltonian-mechanics}) dan dijadikan
ampuh oleh besarnya bilangan yang terlibat: ketika susunannya dicacah
dalam satuan $10^{10^{20}}$, ``sangat mungkin'' dan ``pasti'' menjadi tak
terbedakan, dan peluangnya mengeras menjadi hukum. Dari pencacahannya
lahirlah entropi, suhu, tekanan dan hukum kedua --- dan, pada akhir
babnya, alasan sehelai karet gelang menarik kembali.

\section{Keadaan mikro, keadaan makro, dan postulatnya}

\begin{definition}[Keadaan mikro dan keadaan makro]\label{def:b3:microcanonical-ensemble:states}
\index{keadaan mikro}\index{keadaan makro}\index{kelipatgandaan}
Yang disebut \emph{keadaan mikro} adalah perincian mikroskopis lengkap
sebuah sistem --- tiap bilangan kuantum tiap partikelnya (atau, secara
klasik, tiap kedudukan dan momentumnya, yang dicacah dalam sel \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang
fase} sebesar $h^3$ tiap partikel,
\cref{prop:b3:schrodinger-three-dimensions:counting}). Yang disebut
\emph{keadaan makro} adalah apa yang dapat dilihat termodinamika: energi
$E$, volume $V$, cacah partikel $N$, kemagnetan\dots\ Yang disebut
\emph{kelipatgandaan} $\Omega(E, V, N)$ adalah banyaknya keadaan mikro
yang mengenakan keadaan makro yang sama.
\end{definition}

\begin{theorem}[Postulat dasarnya]\label{thm:b3:microcanonical-ensemble:postulate}
\index{ensembel mikrokanonik}\index{postulat dasar}
Sistem terkucil dalam kesetimbangan ditemukan dengan peluang yang sama
dalam tiap $\Omega(E, V, N)$ \omterm{def:b3:microcanonical-ensemble:states}{keadaan mikro} yang terjangkaunya. Seluruh
fisika statistik kesetimbangan menyusul dari satu kalimat ini.
Kredensialnya: teorema Liouville menunjukkan bahwa sebaran seragam pada
kulit energinyalah yang dipertahankan dinamikanya --- tak ada aliran yang
pernah menggerombolkan peluang \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fasenya} --- dan seabad setengah
akibatnya tak pernah berselisih dengan sebuah percobaan.
\end{theorem}
\admitted

\begin{example}[Sebuah magnet berisi $N$ koin]\label{ex:b3:microcanonical-ensemble:spins}
Sebanyak $N$ spin yang saling bebas, masing-masing ke atas atau ke bawah:
ada $2^N$ \omterm{def:b3:microcanonical-ensemble:states}{keadaan mikro}. \omterm{def:b3:microcanonical-ensemble:states}{Keadaan makro} ``$n$ ke atas'' berkelipatgandaan
$\Omega(n) = \binom Nn$, yang memuncak sangat tajam pada $n = N/2$ dengan
lebar nisbi $\sim1/\sqrt N$. Untuk $N = 100$: keadaan semuanya ke atas
ada satu; \omterm{def:b3:microcanonical-ensemble:states}{keadaan makro} berimbangnya memuat $\num{1e29}$. Untuk
$N = 10^{20}$, simpangan bahkan sebesar $10^{-8}$ dalam bagian ke atasnya
tertekan oleh faktor seperti $\eu^{-10^4}$: sistemnya \emph{memang}
berada pada puncaknya, dan yang kita sebut kesetimbangan adalah \omterm{def:b3:microcanonical-ensemble:states}{keadaan
makro} dengan \omterm{def:b3:microcanonical-ensemble:states}{keadaan mikro} terbanyak.
\end{example}

\section{Entropi adalah sebuah cacah}

\begin{definition}[Entropi Boltzmann]\label{def:b3:microcanonical-ensemble:entropy}
\index{entropi!statistik}
Yang disebut \emph{entropi statistik} sebuah \omterm{def:b3:microcanonical-ensemble:states}{keadaan makro} adalah
\[
S = k_{\text{B}}\ln\Omega , \qquad
k_{\text{B}} = \qty{1.38e-23}{J/K} .
\]
Logaritmanya membuat entropinya \emph{aditif} di tempat
\omterm{def:b3:microcanonical-ensemble:states}{kelipatgandaannya} \emph{mengalikan}: untuk sistem yang saling bebas
$\Omega = \Omega_1\Omega_2$ dan $S = S_1 + S_2$; tetapan Boltzmann
mengalibrasi cacahnya ke satuan kelvin-dan-joule yang sudah dipilih
termodinamika. $S$ ini, seperti akan kita periksa, adalah entropi hukum
kedua jilid Tahun 1 --- kini dengan makna mikroskopis: entropi mengukur,
dalam mata uang logaritmik, \emph{berapa banyak cara} \omterm{def:b3:microcanonical-ensemble:states}{keadaan makronya}
dapat diwujudkan.
\end{definition}

\begin{proposition}[Entropi gas ideal]\label{prop:b3:microcanonical-ensemble:gas}
\index{rumus Sackur--Tetrode}
Mencacah keadaan kuantum $N$ atom identik dalam sebuah kotak
(\cref{prop:b3:schrodinger-three-dimensions:counting}, satu sel sebesar
$h^3$ tiap atom dalam \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase}, dibagi $N!$ karena keidentikannya ---
\cref{exo:b3:identical-particles:10}) memberikan
\[
S = Nk_{\text{B}}\Big[\ln\Big(\frac VN\,\frac1{\lambda_T^3}\Big)
 + \frac52\Big] , \qquad
\lambda_T = \frac{h}{\sqrt{2\pi mk_{\text{B}}T}}
\]
(yakni \emph{rumus Sackur--Tetrode}; $\lambda_T$, yaitu panjang
gelombang de Broglie termal, adalah ukuran paket gelombang sebuah atom
pada suhu $T$). Dua audit yang luar biasa: rumusnya ekstensif
\emph{hanya} berkat $N!$-nya; dan nilai mutlaknya --- termasuk $h$-nya
--- cocok dengan entropi kalorimetrik gas nyata
(\cref{exo:b3:microcanonical-ensemble:11}): termodinamika klasik, yang
diukur pada abad kesembilan belas, sudah mengetahui tetapan Planck tanpa
menyadarinya.
\end{proposition}
\admitted

\section{Suhu adalah turunan sebuah pencacahan}

\begin{theorem}[Kesetimbangan dan suhu]\label{thm:b3:microcanonical-ensemble:temperature}
\index{suhu!definisi statistik}
Biarkanlah dua sistem bertukar energi, dengan totalnya $E = E_1 + E_2$
tetap. \omterm{def:b3:microcanonical-ensemble:states}{Kelipatgandaan} bersamanya $\Omega_1(E_1)\,\Omega_2(E - E_1)$
maksimum --- dan sangat tajam --- pada pembagian yang membuat
\[
\frac{\partial S_1}{\partial E_1} =
\frac{\partial S_2}{\partial E_2} .
\]
Dengan mendefinisikan
\[
\frac1T = \frac{\partial S}{\partial E}\bigg|_{V,N} ,
\]
kesetimbangan berarti \emph{kesamaan suhu}, dan energi mengalir secara
spontan dari $T$ tinggi ke $T$ rendah karena arah itulah yang menaikkan
cacah totalnya. Logika yang sama yang diterapkan pada pertukaran volume
dan partikel memberikan $P/T = \partial S/\partial V$ dan $-\mu/T
= \partial S/\partial N$: semua besaran intensif termodinamika
adalah turunan cacahnya.
\end{theorem}

\begin{proof}
Maksimumkanlah $\ln\Omega_1 + \ln\Omega_2$ terhadap $E_1$: syarat
stasionernya adalah kesamaan yang dinyatakan itu. Ketajamannya: lebar
puncaknya $\sim E/\sqrt N$ (\cref{exo:b3:microcanonical-ensemble:7}),
sehingga maksimumnya adalah keadaan yang teramati. Jika
$\partial S_1/\partial E_1 >
\partial S_2/\partial E_2$, memindahkan energi \emph{ke dalam} 1
menaikkan $S_{\text{tot}}$: benda yang lebih dingin (dengan
$\partial S/\partial E$ lebih besar) menyerapnya --- kalor mengalir dari
panas ke dingin sebagai pernyataan pencacahan. Periksalah pada gas ideal:
$S \propto \tfrac32 Nk_{\text{B}}\ln E + \cdots$ memberikan
$1/T = \tfrac32 Nk_{\text{B}}/E$, yakni $E = \tfrac32
Nk_{\text{B}}T$ --- hasil teori kinetik jilid Tahun 1, yang diperoleh
kembali dari pencacahan murni.
\end{proof}

\section{Hukum kedua, dan hal-hal yang lebih ganjil}

\begin{proposition}[Ketakberbalikan adalah aritmetika]\label{prop:b3:microcanonical-ensemble:second-law}
\index{hukum kedua!statistik}
\omterm{def:b3:lagrangian-mechanics:coordinates}{Kendala} yang dilepaskan (sebuah dinding yang dibuang, sebuah katup yang
dibuka) hanya dapat memperbesar cacah yang terjangkaunya: $\Omega$
tumbuh, $S$ tumbuh --- yakni hukum kedua jilid Tahun 1, yang kini menjadi
pernyataan tentang peluang. Pembalikannya tidak terlarang melainkan
\emph{terdiskon}: gas berisi $N$ molekul yang berkumpul kembali di
separuh kirinya berpeluang $2^{-N}$, dan untuk satu sentimeter kubik
udara waktu tunggunya melampaui umur semesta sebesar faktor berangka
$10^{19}$. Anak panah waktu, pada tingkat ini, adalah arah yang membuat
cacahnya bertambah.
\end{proposition}

\begin{proof}[Bukti sebagian]
Membuang sebuah \omterm{def:b3:lagrangian-mechanics:coordinates}{kendala} membuat tiap \omterm{def:b3:microcanonical-ensemble:states}{keadaan mikro} yang dahulu terjangkau
tetap terjangkau lalu menambahkan yang baru: $\Omega$ tak dapat
berkurang. Cacah pengembangan bebasnya: tiap molekulnya melipatduakan
volume terjangkaunya, $\Omega \to 2^N\Omega$, sehingga $\Delta S = Nk_{\text{B}}\ln2$
--- tepat hasil termodinamika bagi pelipatduaan bebas isotermalnya.
\end{proof}

\begin{example}[Suhu negatif]\label{ex:b3:microcanonical-ensemble:negative}
\index{suhu negatif}
Sistem yang energinya \emph{terbatas di atas} --- $N$ spin dalam sebuah
medan, dengan energi dari $-N\epsilon$ sampai $+N\epsilon$ --- mempunyai
$\Omega(E)$ yang naik lalu \emph{turun}: di luar titik berimbangnya,
menambahkan energi justru mengurangi cacahnya. Di sana
$\partial S/\partial E < 0$: suhunya \emph{negatif}. Keadaan semacam itu
memang ada (pembalikan populasi --- yakni keadaan kerja tiap medium
laser --- dan sistem spin yang disiapkan lewat pembalikan medan) dan
keadaan itu bukanlah dingin melainkan \emph{lebih panas daripada tiap
suhu positif}: energi mengalir dari benda ber-$T < 0$ mana pun ke benda
ber-$T > 0$ mana pun. Urutan suhu yang jujur berjalan
$+0, \dots, +\infty \equiv -\infty, \dots, -0$: apa yang ditata $1/T$
secara wajar, justru dikacaukan $T$
(\cref{exo:b3:microcanonical-ensemble:9}).
\end{example}

\begin{method}[Keterampilan mikrokanonik]\label{met:b3:microcanonical-ensemble:method}
(1) Daftarkanlah: apakah \omterm{def:b3:microcanonical-ensemble:states}{keadaan mikro} di sini, dan apa yang ditetapkan
\omterm{def:b3:microcanonical-ensemble:states}{keadaan makronya}? (2) Cacahlah $\Omega$ --- kombinatorika untuk satuan
diskretnya, dan volume \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase} dibagi $h^{3N}N!$ untuk gasnya. (3)
Ambillah logaritmanya sejak awal: Stirling ($\ln N! \approx N\ln N - N$)
mengubah hasil kalinya menjadi jumlahan yang dapat dikerjakan. (4)
Turunkanlah $S$: turunan terhadap energinya untuk $T$, terhadap volumenya
untuk $P$, terhadap cacahnya untuk $\mu$; turunan terhadap panjang atau
kemagnetannya untuk tegangan dan medan
(\cref{pb:b3:microcanonical-ensemble:1}). (5) Percayailah puncaknya:
fluktuasinya turun sebesar $1/\sqrt N$, sehingga gantilah ``paling
mungkin'' dengan ``sama dengan'' dan janganlah meminta maaf kepada
siapa pun.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:microcanonical-ensemble:1}
Empat spin. (a) Daftarkanlah \omterm{def:b3:microcanonical-ensemble:states}{kelipatgandaan} $\Omega(n)$ \omterm{def:b3:microcanonical-ensemble:states}{keadaan makro}
``$n$ ke atas''. (b) Berapa peluang tiap \omterm{def:b3:microcanonical-ensemble:states}{keadaan makronya} menurut
postulatnya? (c) Berapa entropinya (dalam satuan $k_{\text{B}}$) bagi
masing-masingnya? (d) \omterm{def:b3:microcanonical-ensemble:states}{Keadaan makro} yang manakah ``kesetimbangan'', dan
seburuk apa hampiran ``sistemnya pasti berada di sana'' pada $N = 4$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:microcanonical-ensemble:2}
Stirling dalam praktik. (a) Bandingkanlah $\ln N!$ dengan $N\ln N - N$
untuk $N = 10$ dan $N = 100$ (pakailah $\ln10! = 15.10$,
$\ln100! = 363.7$). (b) Tunjukkanlah bahwa galat nisbinya turun seperti
$\ln N/N$. (c) Pakailah Stirling untuk menunjukkan
$\ln\binom N{N/2} \approx N\ln2 - \tfrac12\ln N + \text{tetapan}$.
(d) Mengapa membuang $\tfrac12\ln N$ sama sekali aman pada satu mol?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:microcanonical-ensemble:3}
(a) Tunjukkanlah bahwa untuk dua sistem yang saling bebas
$S = S_1 + S_2$. (b) $\Omega$ sebuah sistem berlipat dua: berapa
kenaikan $S$-nya --- dan tindakan fisis apa
(\cref{prop:b3:microcanonical-ensemble:second-law}) yang bersesuaian
dengannya bagi satu partikel? (c) Hitunglah entropi, dalam $k_{\text{B}}$
dan dalam J/K, sebuah dek 52 kartu yang telah dikocok
($\ln52! \approx 156.4$). (d) Mengapa entropi kartunya dapat diabaikan
secara termodinamika sedangkan entropi molekulnya tidak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:microcanonical-ensemble:4}
Pengembangan bebas, diaudit. Satu mol melipatduakan volumenya ke dalam
vakum. (a) Berapa $\Delta S$-nya dari argumen pencacahannya? (b) Berapa
peluang mengamati gasnya kembali di separuh asalnya? (c)
Perkirakanlah skala waktu bagi fluktuasi semacam itu jika waktu penataan
ulang molekulnya \qty{e-10}{s} --- lalu bandingkanlah dengan umur
semesta (\qty{4e17}{s}). (d) Dalam pengertian apa tepatnya hukum
keduanya ``hanya'' bersifat peluang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:5}
Pencacahan gas ideal. Bermula dari
$\Omega \propto \dfrac{V^N}{N!\,h^{3N}}\times$ (volume kulit
momentumnya $\propto E^{3N/2}$): (a) tunjukkanlah $S = Nk_{\text{B}}[\ln V +
\tfrac32\ln E] - k_{\text{B}}\ln N! + \text{tetapan}$. (b) Tunjukkanlah bahwa
tanpa $N!$-nya, melipatduakan sistemnya (baik $V$ maupun $N$)
\emph{tidak} akan melipatduakan $S$ --- yakni kegagalan ekstensivitas
pada \cref{exo:b3:identical-particles:10}. (c) Dengan Stirling,
tuliskanlah hasilnya dalam bentuk $S = Nk_{\text{B}}[\ln(V/N) + \tfrac32\ln(E/N)]
+ \text{tetapan}\cdot N$. (d) Masukan fisis apa yang menetapkan kedua
``tetapan'' yang ditinggalkan sejarah tanpa penentuan (ukuran sel
kuantumnya; keidentikan partikelnya)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:6}
Termodinamika dari turunan. Dengan memakai bentuk pada
\cref{exo:b3:microcanonical-ensemble:5}(c): (a) hitunglah $1/T =
\partial S/\partial E$ lalu perolehlah kembali $E = \tfrac32 Nk_{\text{B}}T$;
(b) hitunglah $P/T = \partial S/\partial V$ lalu perolehlah kembali $PV =
Nk_{\text{B}}T$; (c) turunkanlah \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invarian adiabatiknya}: pada $S$ dan $N$
tetap, tunjukkanlah bahwa $VE^{3/2}$ tertetapkan, yakni $TV^{2/3}$ ---
lalu cocokkanlah dengan $PV^{5/3}$ jilid Tahun 1; (d) nyatakanlah apa
yang telah dicapai: hukum empiris mana yang baru saja menjadi teorema
pencacahan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:7}
Ketajaman kesetimbangannya. Dua bongkah gas ideal yang sama berbagi
energi total $E$. (a) Tunjukkanlah bahwa $\ln[\Omega_1\Omega_2]$
maksimum pada pembagian yang rata. (b) Jabarkanlah sampai orde kedua
dalam ketakseimbangannya $x =
(E_1 - E_2)/E$ lalu tunjukkanlah bahwa
peluangnya $\propto\eu^{-3Nx^2/4}$ (tiap bongkahnya berderajat kebebasan
$\tfrac32 N$ --- biarkanlah faktornya longgar). (c) Untuk
$N = 10^{22}$: berapa ketakseimbangan akar rata-rata kuadratnya? (d)
Berapa beda suhu yang diwakili ketakseimbangan itu, untuk gas pada
\qty{300}{K}: dapatkah termometer mana pun melihatnya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:8}
Zat padat dua aras (Schottky): $N$ tapak, masing-masing berenergi $0$
atau $\epsilon$; dengan $n$ tereksitasi. (a) Berapa $\Omega(n)$ dan
$S(n)$? (b) Dengan $E =
n\epsilon$ dan Stirling, turunkanlah
\[
\frac1T = \frac{k_{\text{B}}}{\epsilon}\,
\ln\frac{N - n}{n} ,
\qquad\text{sehingga}\qquad
\frac nN = \frac{1}{\eu^{\epsilon/k_{\text{B}}T} + 1} .
\]
(c) Sketsalah $E(T)$ beserta kapasitas kalornya: ada sebuah tonjolan
(yakni \emph{anomali Schottky}) di dekat $k_{\text{B}}T \sim \epsilon/2$
--- mengapa $C$ lenyap pada kedua ujungnya? (d) Di mana bentuk rumus
pendudukannya sudah pernah muncul, dan di mana ia akan kembali
(\cref{ch:b3:quantum-statistics})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:microcanonical-ensemble:9}
Di bawah nol. Untuk sistem spin pada
\cref{ex:b3:microcanonical-ensemble:negative} dengan pembelahan arasnya
$\epsilon$: (a) tunjukkanlah bahwa $T < 0$ tepat ketika lebih dari
separuh spinnya ke atas (populasinya terbalik). (b) Hitunglah $T$ untuk
pembalikan 60:40 dengan $\epsilon = \qty{e-4}{eV}$. (c) Tunjukkanlah
bahwa energi selalu mengalir dari benda ber-$T$ negatif ke benda ber-$T$
positif (bandingkanlah $\partial
S/\partial E$). (d) Mengapa sistem berenergi tak terbatas (sebuah gas)
tak pernah dapat mencapai $T < 0$?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:microcanonical-ensemble:10}
Pencampuran, secara kuantitatif. Dua gas yang berbeda (masing-masing $N$,
bervolume $V$) disatukan. (a) Hitunglah $\Delta S$ lewat pencacahan. (b)
Untuk gas yang sama: tunjukkanlah, dengan $N!$-nya disertakan, bahwa
$\Delta S = 0$. (c) Helium-3 dan helium-4 berbeda: menyatukannya
\emph{memang} menciptakan $2Nk_{\text{B}}\ln2$ --- namun tak ada kalor
yang mengalir dan tak ada usaha yang dilakukan: di manakah kenaikan
entropinya secara fisis (berapa ongkos memisahkannya kembali)? (d)
Berapa usaha minimum untuk memisahkannya kembali pada suhu $T$:
nyatakanlah, lalu sebutkanlah industri modern yang dibangun untuk
membayarnya (pemisahan isotop).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:microcanonical-ensemble:11}
Tabelnya sudah tahu tentang $h$. Sackur--Tetrode untuk argon ($m =
\qty{6.6e-26}{kg}$) pada \qty{300}{K}, \qty{1}{bar} ($V/N =
k_{\text{B}}T/P$): (a) hitunglah $\lambda_T$; (b) hitunglah $S/N
k_{\text{B}}$; (c) bandingkanlah dengan nilai kalorimetriknya
$S_{\text{molar}} = \qty{154.8}{J/(mol.K)}$, yakni $S/Nk_{\text{B}}
= 18.6$; (d) jelaskanlah apa yang sedang diuji: dua masukan kuantum yang
mana yang duduk di dalam sebuah bilangan yang diukur dengan kalorimeter
es sebelum 1912?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:microcanonical-ensemble:12}
Fluktuasi yang dapat Anda lihat. Cacah molekul dalam volume kecil $v$
sebuah gas berfluktuasi; peluang simpangan nisbi $\delta$ adalah
$\propto\eu^{-\bar n\delta^2/2}$ ($\bar n = Nv/V$ --- terimalah bentuk
Gauss ini, atau turunkanlah dari binomialnya). (a) Untuk $v$ berupa
kubus bersisi \qty{550}{nm} pada kerapatan atmosfer: berapa $\bar n$ dan
berapa $\delta$ akar rata-rata kuadratnya? (b) Mengapa riak kerapatan
beberapa per seribu yang tetap pada skala optis ini justru yang
diperlukan \cref{pb:b3:perturbation-theory:1} agar udara dapat menghambur
sama sekali? (c) Pada $\bar n$ berapa (sehingga pada skala berapa)
fluktuasinya akan mencapai $10\%$? (d) Tutuplah lingkarnya dalam satu
kalimat: birunya langit adalah sebuah pernyataan tentang statistik
$\sqrt N$.
\end{exercise}

\section{Soal: Pegas entropik}

\begin{problem}[{Soal akhir pekan --- mengapa karet menarik kembali}]\label{pb:b3:microcanonical-ensemble:1}
Regangkanlah karet gelang dan ia menyentak kembali --- padahal ikatan
molekulnya nyaris tidak terdeformasi dan energi dalamnya nyaris tidak
berubah. Gaya pemulih karet bukanlah energi yang mencari minimumnya
melainkan \emph{entropi yang mencari maksimumnya}: yakni kemenangan besar
pertama pencacahan murni atas mekanisme, dengan matematika yang sama yang
menjalankan percobaan DNA molekul tunggal hari ini. Model: sebuah rantai
berisi $N$ mata rantai tegar, masing-masing sepanjang $b$, masing-masing
menunjuk ke kiri atau ke kanan sepanjang sumbu tarikannya; pemuluran
ujung-ke-ujungnya $x = (n_+ - n_-)\,b$.

\medskip\noindent\textbf{Bagian I --- Mencacah susunannya.}
\begin{enumerate}
  \item Nyatakanlah $n_\pm$ dalam $N$ dan $x$, lalu tuliskanlah
        \omterm{def:b3:microcanonical-ensemble:states}{kelipatgandaannya} $\Omega(x)$.
  \item Di manakah $\Omega$ maksimum, dan apa yang dikatakannya tentang
        pemuluran yang disukai rantai yang bebas?
  \item Dengan Stirling, tunjukkanlah untuk $x \ll Nb$:
        \[
        S(x) \approx S(0) - \frac{k_{\text{B}}x^2}{2Nb^2} .
        \]
  \item Energi rantainya (dalam model ini) tidak bergantung pada $x$:
        berilah alasan menyebut gaya pemulih apa pun di sini
        ``entropik''.
  \item Pemuluran akar rata-rata kuadrat rantai bebasnya adalah
        $b\sqrt N$ (jalan acak): untuk $N = 10^4$ dan $b = \qty{0.5}{nm}$,
        bandingkanlah ukuran gulungannya dengan panjang teregangnya $Nb$.
  \item Karet sungguhan adalah jaringan rantai semacam itu di antara
        ikatan silangnya: parameter tunggal model yang mana yang diubah
        vulkanisasi (ikatan silang yang lebih banyak)?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Gaya dari pencacahan.}
\begin{enumerate}[resume]
  \item Untuk sistem yang ditahan pada suhu $T$, tegangan yang
        diperlukan untuk menahan pemuluran $x$ adalah
        $f = -T\,\partial S/\partial x$
        (terimalah ini dari $\dd E = T\dd S + f\dd x$ pada energi tetap):
        turunkanlah
        \[
        f = \frac{k_{\text{B}}T\,x}{Nb^2} .
        \]
  \item \omterm{thm:b3:continuum-elasticity:hooke}{Hukum Hooke} telah muncul dengan tetapan pegas $k =
        k_{\text{B}}T/Nb^2$: hitunglah untuk satu rantai dengan $N =
        \num{e4}$, $b = \qty{0.5}{nm}$ pada \qty{300}{K}.
  \item Faktor yang mengejutkan adalah $T$: apa yang seharusnya
        dilakukan karet gelang di bawah beban tetap ketika
        \emph{dipanaskan}? Pertentangkanlah dengan pegas baja.
  \item Perkirakanlah gaya untuk meregangkan satu rantai sampai separuh
        panjang penuhnya, beserta skala gaya $k_{\text{B}}T/b$ yang
        membuat model linearnya gagal.
  \item Karet gelang berpenampang \qty{1}{mm^2} memuat
        $\sim\num{e14}$ rantai efektif yang sejajar: perkirakanlah
        tetapan pegasnya lalu bandingkanlah dengan pengalaman.
  \item Mengapa karet mengeras (bukan melunak) seiring suhunya, gram per
        gram, sedangkan logam melunak?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Dapur Gough--Joule.}
\begin{enumerate}[resume]
  \item Regangkanlah karet gelang dengan cepat (secara adiabatik) pada
        bibir Anda: ia menghangat. Jelaskanlah dengan neraca entropinya:
        meregangkannya \emph{mengurangi} entropi susunannya, sehingga ke
        manakah entropinya (kalornya) harus pergi pada total yang tetap?
  \item Lepaskanlah dengan cepat: ia mendingin. Lengkapilah argumen
        setangkupnya.
  \item Karet gelang yang diberi beban dipanaskan perlahan (dengan
        pengering rambut): ke arah mana bebannya bergerak, dan mengapa
        inilah sidik bersih elastisitas entropik?
  \item Rancanglah percobaan tandingannya dengan pegas baja: apa yang
        terjadi sebagai gantinya, dan mengapa (elastisitas jenis apa)?
  \item Sebuah mesin: sebuah roda berjari-jari karet, yang dipanaskan
        pada satu sisinya, berputar mantap. Telusurilah logika satu
        daurnya (jari-jari yang dipanaskan menciut, sehingga rodanya tak
        seimbang).
  \item Hukum $f = k_{\text{B}}Tx/Nb^2$ yang sama, yang diukur dengan
        penjepit optis pada molekul DNA tunggal (dengan $b \approx
        \qty{100}{nm}$, $N \approx 500$ untuk sepotong genom bakteri),
        memberikan gaya dalam rentang berapa? (Hitunglah
        $k_{\text{B}}T/b$.) Mengapa hal ini membuat modelnya
        \emph{langsung} dapat diuji pada satu molekul?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Pelajarannya.}
\begin{enumerate}[resume]
  \item Model rantainya tidak mempunyai interaksi dan tidak mempunyai
        skala energi, namun menghasilkan sebuah hukum gaya: nyatakanlah
        dengan tepat dari mana gayanya ``datang'' dalam bahasa
        mikrokanonik bab ini.
  \item Mengapa gayanya lenyap pada $T = 0$ --- dan apa yang dikatakannya
        tentang hakikat elastisitas dalam dunia tanpa kegelisahan termal?
  \item Bandingkanlah kelinearan terhadap $T$ tetapan pegas entropiknya
        dengan kelinearan terhadap $T$ tekanan gas idealnya: tunjukkanlah
        bahwa keduanya gejala yang sama (mencacah susunan terhadap sebuah
        \omterm{def:b3:lagrangian-mechanics:coordinates}{kendala}) dalam pakaian yang berbeda.
  \item Karet yang dipanaskan mengangkat sebuah beban, sehingga melakukan
        usaha yang sungguhan: telusurilah sumber dan rute energinya
        (kalor masuk dari pengeringnya, pembukuan entropinya, usaha
        keluar) lalu pastikanlah bahwa hukum pertamanya tak tersentuh.
  \item Rantai ideal itu mengabaikan penghindaran diri dan energi
        pelenturan mata rantainya: sebutkanlah satu ciri terukur karet
        sungguhan yang disembunyikan tiap pengabaian itu (pengerasan tajam
        di dekat pemuluran penuhnya; pengkristalan akibat regangan beserta
        histeresisnya).
  \item Protein melipat, membran berkedip, polimer menggulung: mengapa
        $k_{\text{B}}T$ pada suhu tubuh merupakan mata uang gaya yang
        wajar ($k_{\text{B}}T/\unit{nm}$ dalam pikonewton --- hitunglah)
        bagi seluruh materi lunak dan materi hidup?
  \item Ringkaslah hasil bernama itu: sebuah rantai berisi $10^4$ mata
        rantai buta, yang dicacah dengan jujur, menghasilkan \omterm{thm:b3:continuum-elasticity:hooke}{hukum
        Hooke} dengan $k =
        k_{\text{B}}T/Nb^2$, meramalkan bahwa karet menghangat
        ketika diregangkan dan mengangkat beban ketika dipanaskan ---
        yakni entropi yang bekerja sebagai gaya, yang terbenarkan mulai
        dari ban dalam sepeda sampai molekul DNA tunggal pada skala
        $\sim\qty{0.04}{pN}$.
\end{enumerate}
\end{problem}
