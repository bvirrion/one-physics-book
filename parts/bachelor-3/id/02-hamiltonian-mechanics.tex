\chapter{Mekanika Hamilton}\label{ch:b3:hamiltonian-mechanics}

Potretlah sebuah bandul seribu kali lalu gambarlah, untuk tiap jepretan,
sudutnya terhadap momentum sudutnya: titik-titiknya jatuh pada sebuah
kurva tertutup, dan setiap gerak yang mungkin dari bandul itu adalah
satu kurva semacam itu --- ayunan kecil pada oval yang bersarang,
putaran penuh pada garis bergelombang di atas dan di bawahnya, dan di
antara keduanya satu kurva bersilang yang memisahkan kedua rezim itu.
Gambaran ini, yang disebut \emph{\omterm{def:b3:hamiltonian-mechanics:portrait}{potret fase}}, adalah inti perumusan
ulang yang diberikan Hamilton kepada mekanika pada 1833: keadaan sebuah
sistem adalah sebuah titik dalam ruang koordinat \emph{dan} momentum,
perkembangannya adalah aliran dalam ruang itu, dan aliran itu
dibangkitkan oleh satu fungsi tunggal, yaitu energinya, lewat dua
persamaan orde satu yang indah setangkupnya. Ganjarannya bukan
perhitungan yang lebih mudah --- Lagrange biasanya menang di situ ---
melainkan \emph{geometri} yang tepat: alirannya mempertahankan volume
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase}, yang kelak menjadi landasan fisika statistik; dan kerangka
aljabarnya, yaitu \omterm{def:b3:hamiltonian-mechanics:poisson}{kurung Poisson}, persis itulah yang kelak diangkat
mekanika kuantum menjadi komutator. Bab ini adalah engsel antara
mekanika benda dan fisika abad kedua puluh.

\section{Dari Lagrange ke Hamilton}

\begin{definition}[Hamiltonian dan persamaan kanonik]\label{def:b3:hamiltonian-mechanics:hamiltonian}
\index{Hamiltonian}\index{persamaan kanonik}\index{ruang fase}
Untuk sistem dengan \omterm{def:b3:lagrangian-mechanics:action}{Lagrangian} $L(q, \dot q, t)$, nyatakanlah
kecepatannya dalam momentum $p_i = \partial L/\partial\dot q_i$ lalu
definisikanlah \emph{Hamiltonian} sebagai transformasi Legendre
\[
H(q, p, t) = \sum_i p_i\dot q_i - L ,
\]
yaitu fungsi koordinat dan \emph{momentum}. Ruang berdimensi $2n$ yang
beranggotakan $(q_1, \dots, q_n, p_1, \dots, p_n)$ adalah \emph{ruang
fase}; satu titik di dalamnya --- satu \emph{keadaan} --- menentukan
seluruh masa depan dan masa lalu sistemnya lewat \emph{persamaan
kanonik} pada
\cref{thm:b3:hamiltonian-mechanics:canonical}.
\end{definition}

\begin{theorem}[Persamaan Hamilton]\label{thm:b3:hamiltonian-mechanics:canonical}
\omterm{thm:b3:lagrangian-mechanics:euler-lagrange}{Persamaan Euler--Lagrange} setara dengan $2n$ persamaan orde satu
berikut
\[
\dot q_i = \frac{\partial H}{\partial p_i} , \qquad
\dot p_i = -\frac{\partial H}{\partial q_i} .
\]
\end{theorem}

\begin{proof}
Turunkanlah $H = \sum p_i\dot q_i - L$ sebagai fungsi $(q, p, t)$:
$\dd H = \sum(\dot q_i\,\dd p_i + p_i\,\dd\dot q_i) - \sum(
\partial_{q_i}L\,\dd q_i + \partial_{\dot q_i}L\,\dd\dot q_i) -
\partial_tL\,\dd t$. Suku $\dd\dot q_i$ saling meniadakan karena
definisi $p_i$ --- justru itulah maksud transformasi Legendre ---
sehingga tersisa
$\dd H = \sum(\dot q_i\,\dd p_i - \partial_{q_i}L\,\dd q_i) -
\partial_tL\,\dd t$. Membaca turunan parsialnya:
$\partial H/\partial p_i = \dot q_i$ dan $\partial H/\partial q_i =
-\partial L/\partial q_i$, yang menurut Euler--Lagrange sama dengan $-\dot p_i$.
(Juga $\partial_tH = -\partial_tL$.)
\end{proof}

\begin{proposition}[Apakah $H$ itu]\label{prop:b3:hamiltonian-mechanics:energy}
$H$ berimpit dengan \omterm{prop:b3:lagrangian-mechanics:energy}{fungsi energi} $h$ pada bab sebelumnya: sepanjang
sebuah gerak, $\dd H/\dd t = \partial H/\partial t$, sehingga $H$ kekal
setiap kali ia tidak bergantung waktu secara eksplisit; dan bila energi
kinetiknya kuadratik dalam kecepatan dengan \omterm{def:b3:lagrangian-mechanics:coordinates}{kendala} yang bebas waktu,
$H = E_k + E_p$, yakni energi mekaniknya --- kini ditulis dalam
variabel $(q, p)$.
\end{proposition}

\begin{proof}
$\dd H/\dd t = \sum(\partial_{q_i}H\,\dot q_i + \partial_{p_i}H\,\dot
p_i) + \partial_tH = \sum(\partial_{q_i}H\,\partial_{p_i}H -
\partial_{p_i}H\,\partial_{q_i}H) + \partial_tH = \partial_tH$:
kesetangkupan \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya} membuat jumlahannya lenyap secara
identik. Penyamaan dengan $E_k + E_p$ adalah
\cref{prop:b3:lagrangian-mechanics:energy}.
\end{proof}

\begin{example}[Dua Hamiltonian]\label{ex:b3:hamiltonian-mechanics:examples}
Massa pada pegas: $p = m\dot x$, sehingga
\[
H = \frac{p^2}{2m} + \frac12 kx^2 ;
\]
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya} $\dot x = p/m$, $\dot p = -kx$ adalah pasangan yang
sudah dikenal. Bandul: $p = m\ell^2\dot\theta$ dan
\[
H = \frac{p^2}{2m\ell^2} - mg\ell\cos\theta .
\]
Pada kedua kasus $H$ adalah energinya, tetap pada tiap gerak: gerak itu
\emph{adalah} kurva aras $H$ pada bidang $(q,p)$.
\end{example}

\begin{method}[Resep Hamilton]\label{met:b3:hamiltonian-mechanics:recipe}
(1) Dari $L$, hitunglah momentum $p_i = \partial L/\partial\dot q_i$
lalu baliklah untuk memperoleh $\dot q_i$. (2) $H = \sum p_i\dot q_i - L$,
dinyatakan hanya dalam $(q, p)$ --- untuk sistem yang wajar cukuplah
$E_k + E_p$ dengan $E_k$ ditulis ulang dalam momentum. (3) Tuliskanlah
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya}. (4) Gambarlah kurva aras $H$: untuk satu \omterm{def:b3:lagrangian-mechanics:coordinates}{derajat
kebebasan} kurva itulah lintasannya, dan seluruh gerak kualitatifnya ---
osilasi, rotasi, kesetimbangan, \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriks} --- terbaca tanpa
menyelesaikan apa pun.
\end{method}

\section{Ruang fase}

\begin{definition}[Potret fase, titik tetap, separatriks]\label{def:b3:hamiltonian-mechanics:portrait}
\index{potret fase}\index{titik tetap}\index{separatriks}
Yang disebut \emph{potret fase} sebuah sistem adalah keluarga
lintasannya di \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase}. Yang disebut \emph{titik tetap} adalah
keadaan yang membuat kedua \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya} lenyap --- yakni sebuah
kesetimbangan: minimum potensialnya muncul sebagai \emph{pusat} yang
dikelilingi kurva tertutup, maksimumnya sebagai \emph{pelana} yang
dilalui sebuah \emph{separatriks}, yaitu lintasan yang membagi \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang
fase} menjadi daerah-daerah dengan gerak yang berbeda secara kualitatif.
\end{definition}

\begin{example}[Potret fase bandul]\label{ex:b3:hamiltonian-mechanics:pendulum-portrait}
Untuk $H = p^2/2m\ell^2 - mg\ell\cos\theta$: oval tertutup mengelilingi
$(0, 0)$ --- \emph{librasi}, yakni ayunan biasa; kurva bergelombang pada
$|p|$ besar --- \emph{rotasi}, yakni bandul yang berputar melewati
puncaknya; dan lewat pelana di $\theta = \pm\pi$ terdapat \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriks},
dengan energi $E = +mg\ell$, yang membuat bandulnya memerlukan waktu tak
hingga untuk mencapai puncak. Keadaan pada \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriks} adalah ayunan
yang dilontarkan \emph{tepat} cukup kuat untuk tiba di kedudukan
terbalik tanpa sisa sedikit pun.
\end{example}

\begin{theorem}[Teorema Liouville]\label{thm:b3:hamiltonian-mechanics:liouville}
\index{teorema Liouville}
Aliran Hamilton mempertahankan volume \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase}: jika sebuah daerah
keadaan awal terbawa oleh \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya}, volumenya (luasnya,
untuk satu \omterm{def:b3:lagrangian-mechanics:coordinates}{derajat kebebasan}) tak pernah berubah --- apa pun bentuk yang
kemudian dimilikinya.
\end{theorem}

\begin{proof}
Aliran di \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase} mempunyai medan kecepatan $(\dot q_i, \dot p_i) =
(\partial_{p_i}H, -\partial_{q_i}H)$. Divergensinya adalah
\[
\sum_i\Big(\frac{\partial\dot q_i}{\partial q_i}
 + \frac{\partial\dot p_i}{\partial p_i}\Big)
 = \sum_i\Big(\frac{\partial^2H}{\partial q_i\partial p_i}
 - \frac{\partial^2H}{\partial p_i\partial q_i}\Big) = 0 :
\]
alirannya tak termampatkan, dan aliran tak termampatkan mengangkut
volume tanpa mengubahnya, seperti pada fluida dalam jilid Tahun 2.
(Langkah formal dari divergensi nol ke volume yang kekal adalah teorema
pengangkutan yang dibuktikan di sana.)
\end{proof}

\begin{remark}[Mengapa Liouville penting]\label{rem:b3:hamiltonian-mechanics:liouville-matters}
Tak ada satu pun dalam perumusan Newton yang menyiratkan bahwa sesuatu
bersifat tak termampatkan. Dalam gambaran Hamilton hal itu otomatis ---
dan itulah izin bagi fisika statistik: ketika kita menggambarkan gas
berisi $10^{23}$ molekul dengan segumpal titik di \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase}, teorema
Liouville mengatakan bahwa gumpalan itu mengalir bagaikan fluida tak
termampatkan, sehingga ``banyaknya keadaan dalam suatu volume \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang
fase}'' menjadi besaran yang tak dapat diciptakan atau dimusnahkan oleh
dinamikanya sendiri. Postulat mikrokanonik fisika statistik, di bagian
selanjutnya jilid ini, berdiri tepat di atasnya.
\end{remark}

\section{Kurung Poisson}

\begin{definition}[Kurung Poisson]\label{def:b3:hamiltonian-mechanics:poisson}
\index{kurung Poisson}
Yang disebut \emph{kurung Poisson} dua fungsi $f(q, p, t)$ dan
$g(q, p, t)$ pada \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase} adalah
\[
\{f, g\} = \sum_i\Big(
 \frac{\partial f}{\partial q_i}\frac{\partial g}{\partial p_i}
 - \frac{\partial f}{\partial p_i}\frac{\partial g}{\partial q_i}\Big) .
\]
Kurung ini antisetangkup, linear dalam tiap argumennya, memenuhi aturan
hasil kali $\{f, gh\} = \{f, g\}h + g\{f, h\}$, dan bagi koordinatnya
sendiri memenuhi \emph{hubungan kanonik}
\[
\{q_i, p_j\} = \delta_{ij} , \qquad
\{q_i, q_j\} = \{p_i, p_j\} = 0 .
\]
\end{definition}

\begin{theorem}[Perkembangan sebagai kurung]\label{thm:b3:hamiltonian-mechanics:evolution}
Sepanjang gerak apa pun,
\[
\frac{\dd f}{\dd t} = \{f, H\} + \frac{\partial f}{\partial t} .
\]
Khususnya, besaran yang tidak bergantung waktu secara eksplisit kekal
jika dan hanya jika kurungnya dengan \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} lenyap; dan \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan
kanoniknya} sendiri berbunyi $\dot q_i = \{q_i, H\}$, $\dot p_i
= \{p_i, H\}$: \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} \emph{membangkitkan} perkembangan waktu.
\end{theorem}

\begin{proof}
Aturan rantai ditambah \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya}: $\dd f/\dd t = \sum(
\partial_{q_i}f\,\dot q_i + \partial_{p_i}f\,\dot p_i) + \partial_tf =
\sum(\partial_{q_i}f\,\partial_{p_i}H - \partial_{p_i}f\,
\partial_{q_i}H) + \partial_tf$.
\end{proof}

\begin{example}[Kurung momentum sudut]\label{ex:b3:hamiltonian-mechanics:brackets}
Untuk satu partikel, $L_z = xp_y - yp_x$ beserta pasangan sikliknya.
Perhitungan langsung dari hubungan kanoniknya memberikan
\[
\{L_x, L_y\} = L_z , \qquad
\{L_y, L_z\} = L_x , \qquad
\{L_z, L_x\} = L_y ,
\]
dan $\{L^2, L_z\} = 0$: komponen momentum sudut tidak saling
``berkomutasi'', tetapi masing-masingnya berkomutasi dengan kuadrat
totalnya. Ingatlah bentuk hubungan ini --- semuanya akan kembali, kata
demi kata, sebagai komutator dalam mekanika kuantum, tempat hubungan itu
menentukan segala hal tentang struktur atom.
\end{example}

\begin{remark}[Pintu menuju mekanika kuantum]\label{rem:b3:hamiltonian-mechanics:doorway}
Dirac mengamati pada 1925 bahwa seluruh mekanika kuantum diperoleh
dengan mempertahankan aljabar mekanika Hamilton dan menggantikan \omterm{def:b3:hamiltonian-mechanics:poisson}{kurung
Poisson} dengan komutator operator yang dibagi
$\iu\hbar$: $\{q, p\} = 1$ menjadi $[\hat x, \hat p] = \iu\hbar$,
kekekalan tetap berarti ``kurungnya dengan $H$ lenyap'', dan
perkembangan waktu tetap dibangkitkan oleh \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian}. Teori klasik
mengandung, jauh di dalam tulangnya, kerangka teori kuantum; bab tentang
mekanika kuantum akan membuat kesesuaian ini gamblang.
\end{remark}

\section{Aksi dan invarian adiabatik}

\begin{definition}[Variabel aksi]\label{def:b3:hamiltonian-mechanics:action-variable}
\index{variabel aksi}
Untuk sistem berderajat kebebasan satu yang berosilasi pada lintasan
fase tertutup, yang disebut \emph{variabel \omterm{def:b3:lagrangian-mechanics:action}{aksi}} adalah luas yang
dilingkupinya dibagi $2\pi$:
\[
I = \frac{1}{2\pi}\oint p\,\dd q .
\]
\end{definition}

\begin{proposition}[Aksi osilator harmonik]\label{prop:b3:hamiltonian-mechanics:sho-action}
Untuk $H = p^2/2m + \tfrac12 m\omega^2q^2$ pada energi $E$, lintasannya
adalah elips dengan setengah sumbu $\sqrt{2E/m\omega^2}$ dan $\sqrt{2mE}$,
yang melingkupi luas $2\pi E/\omega$: karena itu
\[
I = \frac{E}{\omega} , \qquad E = \omega I ,
\]
dan frekuensi osilasinya $\partial E/\partial I = \omega$, sebagaimana
seharusnya.
\end{proposition}

\begin{proof}
Luas sebuah elips, $\pi ab = \pi\sqrt{2E/m\omega^2}\sqrt{2mE} =
2\pi E/\omega$.
\end{proof}

\begin{proposition}[Keinvarianan adiabatik]\label{prop:b3:hamiltonian-mechanics:adiabatic}
\index{invarian adiabatik}
Jika sebuah parameter sistemnya (panjang, kekakuan, medan) diubah
secara \emph{perlahan} --- sepanjang banyak periode osilasi --- energinya
berubah, frekuensinya berubah, tetapi \omterm{def:b3:lagrangian-mechanics:action}{aksi} $I$ tetap dengan hampiran
yang sangat baik: $I$ adalah sebuah \emph{\omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invarian adiabatik}}. Untuk
osilator yang diubah perlahan, $E/\omega$ dengan demikian kekal:
keraskanlah pegasnya perlahan hingga $\omega$ berlipat dua, dan
energinya pun berlipat dua.
\end{proposition}

\begin{proof}[Bukti sebagian]
Untuk osilator dengan $\omega(t)$ yang berubah perlahan: sepanjang satu
periode, usaha yang dilakukan parameter yang berubah itu dapat dihitung
dengan perata-rataan; perhitungannya (dituntun dalam
\cref{exo:b3:hamiltonian-mechanics:11}) memberikan
$\dot E/E = \dot\omega/\omega$, yakni $\dd(E/\omega)/\dd t = 0$ pada
orde terdepan. Pernyataan umumnya, untuk sembarang sistem berosilasi
yang diubah perlahan, diterima tanpa bukti --- itulah sebabnya orbit
planet bertahan terhadap gangguan lambat, sekaligus titik tolak ``teori
kuantum lama'' di bawah ini.
\end{proof}

\begin{remark}[Teori kuantum lama]\label{rem:b3:hamiltonian-mechanics:old-quantum}
Mengapa atom mempunyai energi yang diskret? Jawaban kuantitatif yang
pertama (Bohr 1913, Sommerfeld 1915) ditulis dalam bahasa bab ini: di
antara semua gerak klasik, alam mempertahankan gerak yang integral
\omterm{def:b3:lagrangian-mechanics:action}{aksinya} merupakan kelipatan bulat tetapan Planck,
\[
\oint p\,\dd q = nh .
\]
Diterapkan pada osilator harmonik, aturan ini memberikan $E_n = n\hbar\omega$
(hanya kehilangan setengah dari $\big(n + \tfrac12\big)\hbar
\omega$ yang sebenarnya); diterapkan pada partikel dalam kotak ia
memberikan \emph{tepat} tingkat energi jilid Tahun 2; diterapkan pada
atom hidrogen (\cref{pb:b3:hamiltonian-mechanics:1}) ia memberikan
spektrum terukur sampai empat angka. Aturan itu benar secara kuantitatif
namun sementara secara konseptual --- dan karena \omterm{def:b3:lagrangian-mechanics:action}{aksi} merupakan \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invarian
adiabatik}, bilangan kuantum $n$ tidak berubah di bawah gangguan lambat,
dan itulah sebabnya aturan semacam itu dapat bekerja sama sekali. Teori
yang sebenarnya bermula lima bab dari sini.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:1}
Untuk massa pada pegas: (a) susunlah $H$ dari $L$ lalu periksalah $H =
E_k + E_p$; (b) tuliskanlah \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya} dan periksalah bahwa
persamaan itu menghasilkan kembali $\ddot x = -\omega^2x$; (c)
tunjukkanlah bahwa lintasannya berupa elips pada bidang $(x, p)$ lalu
berikanlah setengah sumbunya pada energi $E$; (d) dalam pengertian apa
titik wakilnya bergerak searah jarum jam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:2}
Manik pada gelang berputar dari bab sebelumnya mempunyai $L =
\tfrac12 mR^2\dot\theta^2 + \tfrac12 m\omega^2R^2\sin^2\theta +
mgR\cos\theta$. (a) Hitunglah $p$ dan $H$. (b) Apakah $H$ kekal? Apakah
ia energi mekaniknya? (c) Sketsalah kurva aras $H$ untuk
$\omega^2 < g/R$ dan $\omega^2 > g/R$, dengan memakai potensial efektif
pada bab tersebut. (d) Kenalilah pusat, pelana dan \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriks} pada
kasus cepatnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:3}
Sebuah bola memantul secara elastis di lantai, $H = p^2/2m + mgz$ ($z >
0$). (a) Gambarlah lintasan fase untuk satu penerbangan dan untuk
seluruh gerak pantulannya. (b) Apa yang dilakukan pantulan elastis itu
terhadap titik wakilnya? (c) Hitunglah luas yang dilingkupinya untuk
ketinggian maksimum $z_{\max}$, beserta \omterm{def:b3:lagrangian-mechanics:action}{aksi} $I$. (d) Aturan
Bohr--Sommerfeld $\oint p\,\dd z = nh$ yang diterapkan pada neutron yang
memantul ($m =
\qty{1.67e-27}{kg}$) memberikan ketinggian pantulan terkuantumkan:
perkirakanlah yang terendah, lalu bandingkanlah dengan \qty{15}{\micro m}
yang terukur dalam percobaan keadaan kuantum gravitasi pada 2002.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hamiltonian-mechanics:4}
Hanya dari hubungan kanoniknya, hitunglah (a) $\{x^2, p\}$; (b)
$\{xp, H\}$ untuk $H = p^2/2m + E_p(x)$, lalu tafsirkanlah kedua
sukunya (kurung inilah yang menggerakkan \emph{teorema virial}); (c)
$\{L_z, x\}$ dan $\{L_z, p_x\}$; (d) tunjukkanlah bahwa jika
$\{f, H\} = 0$ dan $\{g, H\} = 0$ maka $\{f, g\}$ juga kekal (pakailah
identitas Jacobi, yang diterima tanpa bukti:
$\{f,\{g,h\}\} + \{g,\{h,f\}\} + \{h,\{f,g\}\} = 0$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:5}
Bandul di dekat separatriksnya. (a) Berikanlah energi separatriksnya dan
nilai $|p|$ maksimum padanya. (b) Tunjukkanlah bahwa pada \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriks} $p =
\pm 2m\ell^2\omega_0\cos(\theta/2)$ dengan $\omega_0 = \sqrt{g/\ell}$.
(c) Dengan memakai $\dot\theta = p/m\ell^2$, tunjukkanlah bahwa waktu
untuk pergi dari $\theta$ ke puncaknya membesar secara logaritmik. (d)
Bandul nyata yang dilepaskan sedikit di bawah separatriksnya menggantung
lama di dekat kedudukan terbalik sebelum berayun kembali --- kaitkanlah
hal ini dengan (c), dan dengan perlambatan yang terlihat di dekat setiap
pelana.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:6}
Partikel bermuatan dalam medan magnet mempunyai $H = (\vect p -
q\vect A)^2/2m$ dengan $\vect p$ adalah momentum kanonik pada bab
sebelumnya. (a) Tuliskanlah \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya} untuk $\vect A = \tfrac12
B(-y, x, 0)$ lalu periksalah bahwa persamaan itu memberikan gerak
siklotron. (b) Tunjukkanlah $H =
\tfrac12 m\vect v^{\,2}$: medan magnetnya tidak melakukan usaha. (c)
Hitunglah kurung dua besaran kekal $\pi_x = p_x - \tfrac12
qBy$ dan $\pi_y = p_y + \tfrac12 qBx$ (periksalah lebih dahulu bahwa
masing-masingnya kekal), lalu tunjukkanlah $\{\pi_x, \pi_y\} = -qB$: dua
besaran kekal yang kurungnya sebuah tetapan. (d) Pakailah
\cref{exo:b3:hamiltonian-mechanics:4}(d) untuk menjelaskan mengapa tak
ada besaran kekal bebas ketiga yang dapat diharapkan dari keduanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:7}
Liouville, dengan tangan. (a) Untuk partikel bebas, alirannya $q \to q +
pt/m$, $p \to p$: tunjukkanlah bahwa sebuah persegi panjang menjadi
jajargenjang dengan luas yang sama. (b) Untuk osilator harmonik,
tunjukkanlah bahwa alirannya sebuah rotasi (dalam satuan yang sesuai)
lalu simpulkanlah. (c) Untuk osilator \emph{teredam}
$\ddot x = -\omega^2x - \gamma\dot x$, tuliskanlah divergensi alirannya
pada bidang $(x, v)$ lalu tunjukkanlah bahwa luasnya menyusut sebagai
$\eu^{-\gamma t}$: peredaman bukanlah gerak Hamilton. (d) Ke manakah
luas yang hilang itu ``pergi'' secara fisis?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:8}
Gerak bidang dalam potensial sentral, $H = (p_r^2 + p_\varphi^2/r^2)/2m
+ E_p(r)$. (a) Tuliskanlah keempat \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya}. (b)
Tunjukkanlah $\{
p_\varphi, H\} = 0$ lalu kenalilah hukum kekekalannya. (c) Reduksilah
menjadi \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} radial satu dimensi dengan sebuah potensial efektif.
(d) Untuk $E_p = -k/r$, tentukanlah orbit lingkarannya pada $p_\varphi$
tertentu lalu berikanlah energinya --- yang akan dikuantumkan dalam
\cref{pb:b3:hamiltonian-mechanics:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hamiltonian-mechanics:9}
(a) Periksalah $\{L_x, L_y\} = L_z$ dari hubungan kanoniknya. (b)
Simpulkanlah kedua kurung lainnya lewat permutasi siklik. (c)
Tunjukkanlah $\{L^2, L_z\} = 0$. (d) Sebuah benda tegar berputar bebas:
dengan mengambil $H =
L^2/2J$ (kelembaman bersimetri bola), tunjukkanlah
bahwa ketiga $L_i$ kekal; bagaimana halnya benda dengan momen kelembaman
yang tidak sama (jawablah secara kualitatif dari kurungnya)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:10}
Tingkat energi kuantum lama. (a) Tunjukkanlah bahwa $\oint p\,\dd q = nh$
yang diterapkan pada osilator harmonik memberikan $E_n = n\hbar\omega$,
dengan memakai
\cref{prop:b3:hamiltonian-mechanics:sho-action}. (b) Terapkanlah pada
partikel dalam kotak sepanjang $a$ lalu perolehlah kembali secara tepat $E_n =
n^2h^2/8ma^2$. (c) Untuk molekul dwiatom yang dimodelkan sebagai
osilator berkekakuan $k = \qty{1.9e3}{N/m}$ dan bermassa tereduksi $\qty{1.14e-26}
{kg}$ (karbon monoksida), hitunglah $\hbar\omega$ dalam eV serta panjang
gelombang pancaran $n = 1 \to 0$; ia jatuh pada daerah spektrum yang
mana? (d) Tingkat energi yang sebenarnya adalah $(n + \tfrac12)\hbar\omega$:
apakah setengah yang hilang itu mengubah panjang gelombang yang
\emph{dipancarkan}? Percobaan apa yang memang mendeteksi setengah titik
nol itu?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:11}
Keinvarianan adiabatik $E/\omega$, diturunkan. Sebuah massa pada pegas
yang kekakuannya $k(t)$ tumbuh perlahan. (a) Tunjukkanlah $\dot E = \tfrac12\dot k\,
x^2$ sepanjang gerak eksaknya. (b) Rata-ratakanlah atas satu periode
pada $k$ tetap: dengan memakai $\langle\tfrac12 kx^2\rangle = E/2$,
tunjukkanlah $\langle\dot E\rangle
= \dot k\,E/2k$. (c) Simpulkanlah $\dd\ln E = \tfrac12\dd\ln k = \dd\ln
\omega$, sehingga $E/\omega$ tetap. (d) Sebuah bandul yang berayun
dengan amplitudo $\theta_0 = 5^\circ$ talinya dipendekkan perlahan dari
\qty{1.0}{m} menjadi \qty{0.5}{m}: carilah amplitudo barunya dan faktor
pertumbuhan energinya, lalu katakanlah dari mana energi itu datang.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hamiltonian-mechanics:12}
Luas di dalam \omterm{def:b3:hamiltonian-mechanics:portrait}{separatriks} bandul adalah cacah keadaan kuantum. (a)
Tunjukkanlah bahwa $\oint p\,\dd\theta$ mengelilingi separatriksnya sama
dengan $16m\ell^2\omega_0$. (b) Menurut aturan Bohr--Sommerfeld,
banyaknya keadaan kuantum berenergi di bawah separatriksnya adalah $N \approx
16m\ell^2\omega_0/h$: hitunglah nilainya untuk massa satu gram pada tali
sepanjang \qty{10}{cm}. (c) Hitunglah nilainya untuk osilator molekul
mirip amonia: $m \sim \qty{1e-26}{kg}$, $\ell \sim \qty{1e-10}{m}$,
$\omega_0 \sim \qty{1e14}{rad/s}$. (d) Simpulkanlah: di manakah batas
antara mekanika yang memerlukan $\hbar$ dan mekanika yang tidak?
\end{exercise}

\section{Soal: Teori kuantum lama dan atom hidrogen}

\begin{problem}[{Soal akhir pekan --- mengkuantumkan ruang fase,
menimbang Rydberg}]\label{pb:b3:hamiltonian-mechanics:1}
Pada 1913 Bohr menghitung spektrum hidrogen dari mekanika planet
ditambah satu aturan kuantum; Sommerfeld mengenali aturan itu sebagai
pernyataan tentang luas \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{ruang fase}. Soal ini membangun kembali
perhitungan mereka dengan perkakas bab ini. Data: $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $1/4\pi\varepsilon_0 =
\qty{8.99e9}{N.m^2/C^2}$, $h = \qty{6.63e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$. Tuliskanlah $k = e^2/4\pi\varepsilon_0$.

\medskip\noindent\textbf{Bagian I --- \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} Kepler.}
Sebuah elektron bergerak pada bidang mengelilingi proton yang tetap.
\begin{enumerate}
  \item Berilah alasan mengapa protonnya boleh dianggap tetap
        (nisbah massa), lalu tuliskanlah
        \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} $H = \dfrac{p_r^2}{2m_{\text{e}}} +
        \dfrac{p_\varphi^2}{2m_{\text{e}}r^2} - \dfrac{k}{r}$.
  \item Tuliskanlah keempat \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{persamaan kanoniknya}.
  \item Tunjukkanlah bahwa $p_\varphi$ kekal lalu namailah.
  \item Reduksilah gerak radialnya ke potensial efektif
        $U_{\text{eff}}(r) = p_\varphi^2/2m_{\text{e}}r^2 - k/r$ lalu
        sketsalah.
  \item Tentukanlah orbit lingkarannya: $r_{\text{c}} =
        p_\varphi^2/m_{\text{e}}k$.
  \item Tunjukkanlah bahwa energinya $E = -k/2r_{\text{c}} =
        -m_{\text{e}}k^2/2p_\varphi^2$, lalu periksalah bahwa nilainya
        setengah energi potensialnya (nisbah virial pada orbit
        gravitasi jilid Tahun 1).
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Orbit klasiknya.}
\begin{enumerate}[resume]
  \item Untuk orbit lingkaran berjejari $r$, carilah kelajuan $v$ dan
        frekuensi orbit $f_{\text{orb}}$ sebagai fungsi $r$.
  \item Untuk orbit seukuran atom, $r = \qty{0.05}{nm}$: hitunglah $v$
        (beserta $v/c$), $f_{\text{orb}}$, dan energinya dalam eV.
  \item Hitunglah momentum sudut $p_\varphi$ orbit itu lalu
        bandingkanlah dengan $\hbar = h/2\pi$: apa yang disarankan oleh
        kedekatannya?
  \item Secara klasik, elektron yang mengorbit adalah dipol berosilasi
        dan memancarkan radiasi (jilid Tahun 2) pada $f_{\text{orb}}$;
        energinya meluruh dalam sekitar \qty{1e-11}{s}. Nyatakanlah dua
        ramalan fatal yang ditimbulkannya bagi atom, dan apa yang
        teramati sebagai gantinya.
  \item Ciri manakah dari spektrum teramati (garis diskret, aturan
        gabungan $1/\lambda = R_H(1/n^2 - 1/n'^2)$) yang menyarankan
        adanya \emph{tingkat} energi yang diskret?
  \item Jelaskanlah mengapa sebuah \emph{\omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invarian adiabatik}} adalah
        calon yang wajar bagi besaran yang mengambil nilai semesta yang
        tetap (ingatlah \cref{prop:b3:hamiltonian-mechanics:adiabatic}).
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Kuantisasi.}
Kenakanlah aturan Sommerfeld pada gerak sudut orbit lingkarannya:
$\oint p_\varphi\,\dd\varphi = nh$, $n = 1, 2, 3, \dots$
\begin{enumerate}[resume]
  \item Tunjukkanlah bahwa aturan itu berbunyi $p_\varphi = n\hbar$.
  \item Simpulkanlah jejari yang diizinkan $r_n = n^2a_0$ dengan $a_0 =
        \hbar^2/m_{\text{e}}k$, lalu hitunglah $a_0$.
  \item Simpulkanlah energi yang diizinkan $E_n = -E_{\text{I}}/n^2$ dengan
        $E_{\text{I}} = m_{\text{e}}k^2/2\hbar^2$, lalu hitunglah
        $E_{\text{I}}$ dalam joule dan dalam eV.
  \item Hitunglah kelajuan pada orbit pertamanya lalu periksalah $v_1/c =
        k/\hbar c \approx 1/137$ (tetapan struktur halus).
  \item Sebuah foton membawa energi sebuah transisi: turunkanlah rumus
        Rydberg dan nilai $R_H = E_{\text{I}}/hc$;
        bandingkanlah dengan \qty{1.097e7}{m^{-1}} yang terukur.
  \item Hitunglah panjang gelombang transisi $2 \to 1$,
        $3 \to 2$ dan $\infty \to 2$; yang manakah tampak, dan apa
        warnanya?
  \item Ion helium terionkan He$^+$ adalah hidrogen dengan muatan inti
        $2e$: bagaimana $a_0$ dan $E_{\text{I}}$ berskala, dan di
        manakah garis $2 \to 1$-nya jatuh?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Perhadapan.}
\begin{enumerate}[resume]
  \item Hitunglah frekuensi orbit $f_{\text{orb}}(n)$ dan frekuensi
        transisi $\nu_{n \to n-1}$ untuk $n = 2$, $10$,
        $100$, lalu tunjukkanlah bahwa nisbahnya menuju $1$ ketika $n$
        membesar.
  \item Inilah \emph{asas korespondensi} Bohr: nyatakanlah dalam satu
        kalimat.
  \item Aturan $p_\varphi = n\hbar$ bermula pada $n = 1$: kemustahilan
        apa yang akan berarti $n = 0$ bagi orbit lingkaran?
  \item Mekanika kuantum akan mempertahankan $E_n = -E_{\text{I}}/n^2$
        secara persis, namun membuang orbitnya: sebutkanlah dua fakta
        terukur yang salah diramalkan gambaran orbit itu (besar momentum
        sudut keadaan dasarnya; keberadaan keadaan dengan $n$ sama tetapi
        berbeda bentuk).
  \item Muon adalah elektron yang $207$ kali lebih berat: untuk hidrogen
        muonik, hitunglah $a_0^\mu$ dan $E_{\text{I}}^\mu$, lalu
        jelaskanlah mengapa atom muonik menyelidiki \emph{intinya}.
  \item Ringkaslah hasil bernama itu: satu \omterm{prop:b3:hamiltonian-mechanics:adiabatic}{invarian adiabatik}, yang
        disamakan dengan kelipatan bulat $h$, menghasilkan $a_0 = \qty{52.9}{pm}$,
        $E_{\text{I}} = \qty{13.6}{eV}$ dan spektrum hidrogen sampai
        empat angka --- sekaligus menyerahkan kepada teori yang
        sebenarnya dua tetapan pusatnya.
\end{enumerate}
\end{problem}
