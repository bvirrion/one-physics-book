\chapter{Spin dan Sistem Dua Aras}\label{ch:b3:spin-two-level}

Pada 1922 Stern dan Gerlach mengirimkan seberkas atom perak melalui
sebuah magnet yang timpang, dengan harapan tak ada pembelokan atau
sebuah lelehan yang sinambung. Berkasnya terbelah bersih menjadi
\emph{dua}. Tak ada momentum sudut orbital yang dapat berbuat demikian:
$2l + 1$ selalu ganjil. Atomnya sedang mengumumkan kepunyaan baru yang
murni kuantum --- yakni \emph{\omterm{def:b3:spin-two-level:spin}{spin}}, sebuah momentum sudut hakiki
sebesar $j = \tfrac12$, yaitu anak tangga setengah bulat yang disiapkan
aljabar \cref{ch:b3:quantum-angular-momentum} dan yang tak pernah dapat
dipakai orbit. \omterm{def:b3:spin-two-level:spin}{Spin} melipatduakan tiap keadaan elektron (sehingga
melengkapi pencacahan di balik tabel berkala), membawa kemagnetan besi
dan proton, serta merupakan sistem dua aras terbersih yang ditawarkan
alam: fisika bab ini menjalankan pemindai \omterm{prop:b3:spin-two-level:rabi}{resonansi magnetik} di tiap
rumah sakit, jam sesium yang mendefinisikan sekon, bisikan 21 sentimeter
yang dipakai untuk memetakan galaksi, dan kubit.

\section{Percobaan yang menemukannya}

\begin{example}[Stern--Gerlach]\label{ex:b3:spin-two-level:stern-gerlach}
Sebuah momen magnetik $\vect\mu$ dalam medan yang \emph{takhomogen}
merasakan gaya $F_z = \mu_z\,\partial B_z/\partial z$: pembelokannya
mengukur $\mu_z$. Harapan klasiknya: momennya terarah acak, sehingga
membentuk kipas yang sinambung. Harapan kuantum bagi momentum orbital:
$2l + 1$ bercak: satu, tiga, lima --- selalu ganjil. Yang teramati untuk
perak (dan untuk hidrogen): \emph{dua} bercak, setangkup, tanpa apa pun
di antaranya. Komponen terukurnya hanya mengambil dua nilai --- yakni ciri $j
= \tfrac12$, yang terlarang bagi orbit, sekaligus peragaan
langsung kuantisasinya: perkakasnya adalah alat ukur bagi satu komponen
\omterm{def:b3:spin-two-level:spin}{spin}, dan berkasnya terbelah menjadi kedua \omterm{def:b3:quantum-formalism:observable}{nilai eigennya}.
\end{example}

\begin{definition}[Spin setengah]\label{def:b3:spin-two-level:spin}
\index{spin}\index{matriks Pauli}\index{spinor}
Elektron (dan proton, neutron, kuark) membawa momentum sudut hakiki
dengan $j = \tfrac12$: yakni ruang keadaan berdimensi dua yang direntang
$\ket\uparrow, \ket\downarrow$ (keadaan eigen $\hat S_z$ dengan
$\pm\hbar/2$), yang di atasnya berlaku
\[
\hat{\vect S} = \frac\hbar2\,\hat{\vect\sigma} , \qquad
\sigma_x = \begin{pmatrix}0&1\\1&0\end{pmatrix} ,\;
\sigma_y = \begin{pmatrix}0&-\iu\\\iu&0\end{pmatrix} ,\;
\sigma_z = \begin{pmatrix}1&0\\0&-1\end{pmatrix}
\]
--- yakni \emph{matriks Pauli}, yang mewujudkan aljabar momentum
sudutnya dalam rumah sekecil mungkin. Keadaan umum
$\alpha\ket\uparrow + \beta\ket\downarrow$ adalah sebuah \emph{spinor}.
Momen magnetik elektronnya adalah
\[
\hat{\vect\mu} = -g\,\frac{\mu_{\text{B}}}{\hbar}\,\hat{\vect S} ,
\qquad g \approx 2 :
\]
dua kali laju orbitalnya tiap satuan momentum sudut --- yakni anomali
yang telah diukur Einstein--de Haas
(\cref{exo:b3:quantum-angular-momentum:12}) dan yang kelak diramalkan
persamaan Dirac. Spin bukanlah rotasi apa pun: tak ada jejari, tak ada
``bola yang berputar'' yang selamat dari pemeriksaan --- ia hakiki,
seperti muatan.
\end{definition}

\begin{example}[Penapis Stern--Gerlach yang dirantai]\label{ex:b3:spin-two-level:chained}
Pilihlah berkas $S_z = +\hbar/2$, lalu ukurlah $S_z$ lagi: semua
atomnya menjawab $+$. Ukurlah $S_x$ sebagai gantinya: separuh dan
separuh --- keadaan $\ket\uparrow$ adalah superposisi
$(\ket{+x} + \ket{-x})/\sqrt2$. Kini pertahankanlah berkas $S_x = +$
lalu ukurlah $S_z$ sekali lagi: separuh dan separuh \emph{lagi} ---
pengukuran $S_x$-nya menghapus $S_z$ yang tadinya tajam. Tiga magnet
sudah cukup untuk memperagakan ketakserasian, keruntuhan dan kaidah
Born; rantai ini adalah
\cref{ex:b3:quantum-formalism:threepolarizers} yang dilakukan dengan atom.
\end{example}

\section{Bola Bloch}

\begin{proposition}[Spin sepanjang sumbu mana pun]\label{prop:b3:spin-two-level:bloch}
\index{bola Bloch}
Untuk arah satuan $\vect n = (\sin\theta\cos\varphi,
\sin\theta\sin\varphi, \cos\theta)$, operator $\vect
n\cdot\hat{\vect\sigma}$ mempunyai \omterm{def:b3:quantum-formalism:observable}{nilai eigen} $\pm1$, dengan
\[
\ket{+\vect n} = \cos\tfrac\theta2\,\ket\uparrow +
\eu^{\iu\varphi}\sin\tfrac\theta2\,\ket\downarrow .
\]
Tiap \omterm{def:b3:spin-two-level:spin}{spinor} murni adalah $\ket{+\vect n}$ untuk tepat satu arah: ruang
keadaannya memetakan ke sebuah bola --- yakni \emph{\omterm{prop:b3:spin-two-level:bloch}{bola Bloch}} ---
dengan keadaan ortogonalnya berada di \emph{antipoda} (perhatikanlah
setengah sudutnya: arah yang berlawananlah, bukan yang tegak lurus,
yang ortogonal). Rotasi bolanya persis merupakan perkembangan yang
dibangkitkan sebuah medan magnet: itulah geometri tiap pengolahan kubit
yang pernah dilakukan.
\end{proposition}

\begin{proof}[Bukti sebagian]
Diagonalkanlah matriks $2\times2$-nya $\vect n\cdot\vect\sigma =
\begin{pmatrix}\cos\theta & \sin\theta\,\eu^{-\iu\varphi}\\
\sin\theta\,\eu^{\iu\varphi} & -\cos\theta\end{pmatrix}$: terasnya $0$,
determinannya $-1$, \omterm{def:b3:quantum-formalism:observable}{nilai eigennya} $\pm1$; \omterm{def:b3:spin-two-level:spin}{spinor} yang diberikan
diperiksa lewat penyulihan (identitas setengah sudut). Antipodanya:
$\theta \to \pi -
\theta$, $\varphi \to \varphi + \pi$ memberikan $\braket{+\vect
n'}{+\vect n} = 0$.
\end{proof}

\section{Spin dalam medan: presesi dan resonansi}

\begin{proposition}[Presesi Larmor]\label{prop:b3:spin-two-level:larmor}
\index{presesi Larmor}\index{nisbah giromagnetik}
Dalam medan statis $\vect B_0 = B_0\vect e_z$, \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} $\hat
H = -\hat{\vect\mu}\cdot\vect B_0$ membelah kedua arasnya sebesar
$\hbar\omega_0$ dan membuat \omterm{def:b3:spin-two-level:spin}{spin} rata-ratanya \emph{berpresesi}
terhadap medannya:
\[
\frac{\dd\langle\hat{\vect S}\rangle}{\dd t} =
\gamma\,\langle\hat{\vect S}\rangle\wedge\vect B_0 , \qquad
\omega_0 = |\gamma|B_0 ,
\]
dengan \omterm{prop:b3:spin-two-level:larmor}{nisbah giromagnetiknya} $\gamma$ ($\mu = \gamma S$). Vektor
Blochnya berputar terhadap $z$ pada $\omega_0$, dengan sudutnya terhadap
medan membeku --- persis gambaran gasing klasik, yang eksak di sini
karena \omterm{def:b3:quantum-formalism:commutator}{komutatornya} linear. Untuk elektron: $\qty{28}{GHz/T}$; untuk
proton (momennya $2.79$ magneton inti): $\gamma_p/2\pi =
\qty{42.6}{MHz/T}$ --- yakni pelat nomor tiap mesin MRI.
\end{proposition}

\begin{proof}
Kenakanlah \cref{prop:b3:quantum-formalism:evolution} dengan $\hat H =
-\gamma B_0\hat S_z$: \omterm{def:b3:quantum-formalism:commutator}{komutatornya} memberikan $\dd\langle\hat
S_x\rangle/\dd t = -\gamma B_0\langle\hat S_y\rangle$,
$\dd\langle\hat S_y\rangle/\dd t = +\gamma B_0\langle\hat
S_x\rangle$ dan $\dd\langle\hat S_z\rangle/\dd t = 0$, yang tersusun
menjadi hasil kali vektor yang dinyatakan itu.
\end{proof}

\begin{proposition}[Resonansi magnetik]\label{prop:b3:spin-two-level:rabi}
\index{resonansi magnetik}\index{osilasi Rabi}
Tambahkanlah medan kecil $B_1$ yang berputar pada bidang melintangnya
dengan frekuensi $\omega$. Dalam kerangka yang ikut berputar
bersamanya, $B_0$ secara efektif menyusut menjadi
$(\omega_0 - \omega)/|\gamma|$; \emph{pada resonansinya}
($\omega = \omega_0$) hanya $B_1$ yang bertahan, dan \omterm{def:b3:spin-two-level:spin}{spinnya} berpresesi
terhadapnya pada \emph{frekuensi Rabi} $\Omega_1 =
|\gamma|B_1$: bermula dari $\ket\uparrow$, peluang telah terbalik adalah
\[
\mathcal P_\downarrow(t) = \sin^2\Big(\frac{\Omega_1t}{2}\Big) .
\]
Sebuah pulsa berdurasi $\pi/\Omega_1$ (yakni ``pulsa $\pi$'') membalik
\omterm{def:b3:spin-two-level:spin}{spinnya}; separuhnya (yakni ``pulsa $\pi/2$'') membaringkannya pada
bidang khatulistiwa, tempat ia berpresesi pada $\omega_0$ dan, lewat
induksi Faraday, \emph{menyiarkan} frekuensinya ke tiap kumparan di
dekatnya. Di luar resonansinya amplitudo bolak-baliknya runtuh sebagai
$\Omega_1^2/[\Omega_1^2 +
(\omega - \omega_0)^2]$: tanggapannya sangat memilih --- yakni sebuah
resonansi --- dan itulah yang membuat \omterm{def:b3:spin-two-level:spin}{spin} dapat dialamati, jenis demi
jenis dan, dengan gradien medan, tempat demi tempat
(\cref{pb:b3:spin-two-level:1}).
\end{proposition}
\admitted

\begin{example}[Struktur hiperhalus dan garis 21 cm]\label{ex:b3:spin-two-level:hyperfine}
\index{struktur hiperhalus}\index{garis 21 cm}
Dalam keadaan dasar hidrogen, \omterm{def:b3:spin-two-level:spin}{spin} elektron dan \omterm{def:b3:spin-two-level:spin}{spin} protonnya
berinteraksi lewat momen magnetiknya: keempat keadaan \omterm{def:b3:spin-two-level:spin}{spinnya} terbelah
menjadi sebuah triplet dan sebuah singlet yang terpisah hanya $\Delta E =
\qty{5.9}{\micro eV}$ --- yakni pembelahan \emph{hiperhalus}, $\nu =
\qty{1420}{MHz}$, $\lambda = \qty{21}{cm}$. Transisinya sangat lambat
(satu pembalikan tiap sepuluh juta tahun), tetapi Galaksi memuat
$10^{66}$ \omterm{thm:b3:hydrogen-atom:levels}{atom hidrogen}: \omterm{ex:b3:spin-two-level:hyperfine}{garis 21 cm} cukup terang untuk telah memetakan
lengan spiralnya, lengkungan cakramnya, dan --- lewat pergeseran
Dopplernya --- kurva rotasi datar yang mengemukakan adanya materi gelap.
Fisika yang sama, dalam keadaan dasar sesium, membelah arasnya tepat
sebesar $\qty{9192631770}{Hz}$: sejak 1967, \emph{definisi sekon} adalah
sebuah pembalikan \omterm{def:b3:spin-two-level:spin}{spin} hiperhalus yang dicacah
(\cref{exo:b3:spin-two-level:12}).
\end{example}

\begin{method}[Keterampilan dua aras]\label{met:b3:spin-two-level:method}
(1) Tuliskanlah sembarang \omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} dua aras dalam \omterm{def:b3:spin-two-level:spin}{matriks Pauli}: $\hat H =
\epsilon_0\mathbb 1 + \vect h\cdot\hat{\vect\sigma}$ ---
maka vektor Blochnya berpresesi terhadap $\vect h$ pada
$2|\vect h|/\hbar$; selebihnya hanyalah geometri. (2) Keadaan eigen
sepanjang $\vect n$: pakailah setengah sudutnya. (3) Soal resonansi:
pindahlah ke kerangka yang berputar; pada resonansinya, hanya $B_1$
yang tersisa. (4) Pulsanya: $\pi/2$ untuk memulai isyarat presesinya,
$\pi$ untuk membalik (dan untuk memfokuskan kembali --- yakni gema
\omterm{def:b3:spin-two-level:spin}{spinnya}). (5) Skala energinya: $\mu_{\text{B}}B$ untuk elektron
($\qty{58}{\micro eV/T}$), tiga orde lebih kecil untuk inti --- yakni
frekuensi radio, spektroskopi tanpa kelvin.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:1}
(a) Periksalah $\sigma_x^2 = \sigma_y^2 = \sigma_z^2 = \mathbb 1$ dan
$\sigma_x\sigma_y = \iu\sigma_z$. (b) Simpulkanlah $[\hat S_x, \hat S_y]
= \iu\hbar\hat S_z$: yakni aljabar momentum sudut dalam dimensi dua.
(c) Tunjukkanlah $\hat S^2 = \tfrac34\hbar^2\,\mathbb 1$ lalu
periksalah $j(j+1)$ untuk $j = \tfrac12$. (d) Tunjukkanlah bahwa
sembarang matriks Hermitian $2\times2$ adalah gabungan real $\mathbb 1$
dan ketiga $\sigma_i$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:2}
(a) Carilah keadaan eigen ternormalkan $\sigma_x$ dan $\sigma_y$.
(b) Sebuah \omterm{def:b3:spin-two-level:spin}{spin} yang disiapkan dalam $\ket{+x}$ diukur sepanjang $z$:
berapa peluang hasilnya? (c) Sepanjang $y$? (d) Periksalah keduanya
terhadap gambaran Blochnya (sudut $\theta, \varphi$ yang mana?).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:3}
Tiga magnet Stern--Gerlach berantai, dengan sumbu $z$, lalu $\vect n$
bersudut $\theta$ pada bidang $xz$, lalu $z$ lagi; berkas $+$-nya
dipertahankan tiap kali. (a) Berapa peluang selamat melewati magnet
keduanya? (b) Berapa peluang selamat melewati ketiganya? (c) Hitunglah
pada $\theta = 90^\circ$ lalu bandingkanlah dengan
\cref{ex:b3:spin-two-level:chained}. (d) Untuk $\theta$ berapa
kebertahanan tiga magnetnya maksimum, dan jawabannya menggemakan apa
dari \cref{exo:b3:quantum-formalism:5}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spin-two-level:4}
Hitunglah pembelahan arasnya $2|\mu|B$ beserta frekuensi resonansinya
dalam medan \qty{1}{T} untuk (a) elektron ($g = 2$); (b)
proton ($\mu_p = 2.79\,\mu_{\text{N}}$, $\mu_{\text{N}} =
e\hbar/2m_{\text{p}}$); (c) bandingkanlah keduanya dengan
$k_{\text{B}}T$ pada \qty{300}{K}: seberapa terpolarisasikah \omterm{def:b3:spin-two-level:spin}{spin}
elektron dan \omterm{def:b3:spin-two-level:spin}{spin} inti pada suhu ruang? (d) Karena itu, pita spektrum
manakah yang dihuni ESR dan NMR?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:5}
Turunkanlah \omterm{prop:b3:spin-two-level:larmor}{presesi Larmor}: dengan $\hat H = -\gamma B_0\hat S_z$, (a)
hitunglah $\dd\langle\hat S_x\rangle/\dd t$, $\dd\langle\hat
S_y\rangle/\dd t$, $\dd\langle\hat S_z\rangle/\dd t$ dari
\omterm{def:b3:quantum-formalism:commutator}{komutatornya}; (b) selesaikanlah $\langle\hat{\vect S}\rangle(t)$ dengan
\omterm{def:b3:spin-two-level:spin}{spin} awalnya sepanjang $x$; (c) tunjukkanlah bahwa
$\|\langle\hat{\vect S}\rangle\|$ dan sudutnya terhadap $\vect B_0$
bersifat tetap; (d) mengapa gambaran \emph{keadaan eigen energi} (dua
aras stasioner) dan gambaran \emph{presesi} (sebuah vektor yang
berputar) memerikan fisika yang sama --- keadaan apa yang berpresesi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:6}
(a) Periksalah bahwa $\ket{+\vect n}$ pada
\cref{prop:b3:spin-two-level:bloch} memang vektor eigen yang diakukan.
(b) Tunjukkanlah bahwa keadaan antipodanya ortogonal. (c) Di manakah
pada bolanya keadaan eigen $\sigma_x$ dan $\sigma_y$ berada? (d) Gerbang
``NOT'' sebuah kubit memetakan $\ket\uparrow \leftrightarrow \ket\downarrow$:
rotasi bola yang manakah itu, dan pulsa mana pada
\cref{prop:b3:spin-two-level:rabi} yang melakukannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:7}
Kerangka berputar, secara jujur. Dengan $\hat H(t) = -\gamma(B_0\hat S_z
+ B_1[\hat S_x\cos\omega t - \hat S_y\sin\omega t])$: (a) jelaskanlah
mengapa berpindah ke kerangka yang berputar pada $\omega$ terhadap $z$
menggantikan $B_0$ dengan $B_0 - \omega/\gamma$ dan membekukan $B_1$;
(b) pada resonansinya, perikanlah gerak vektor Blochnya dalam kerangka
yang berputar; (c) turunkanlah durasi pulsa $\pi$ untuk proton dengan $B_1 =
\qty{10}{\micro T}$; (d) sketsalah apa yang dilihat
pengamat dalam kerangka laboratorium selama pulsa itu (dua rotasi yang
bersarang).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:8}
\omterm{ex:b3:spin-two-level:hyperfine}{Garis 21 cm}. (a) Periksalah bahwa $\Delta E = \qty{5.9}{\micro eV}$
memberikan $\nu = \qty{1420}{MHz}$ dan $\lambda = \qty{21.1}{cm}$. (b)
Keadaan tereksitasinya hidup $\qty{e7}{yr}$: lebar garis sebesar apakah
itu, dan mengapa lebar teramatinya ($\sim$ kHz ke atas) murni Doppler?
(c) Tepi cakram sebuah galaksi menjauh dengan \qty{220}{km/s} terhadap
pusatnya: hitunglah rentang frekuensi profil 21 cm-nya. (d) Mengapa
\omterm{ex:b3:spin-two-level:hyperfine}{garis 21 cm} menelusuri gas \emph{atomik netral}, sedangkan CO
(\cref{pb:b3:quantum-angular-momentum:1}) menelusuri gas
\emph{molekul} --- dan mengapa para astronom memerlukan keduanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spin-two-level:9}
Nitrogen molekul amonia menerowong menembus bidang H$_3$-nya:
kedua susunan ``payung''nya bercampur menjadi keadaan setangkup
dan taksetangkup yang terbelah sebesar $\Delta E = \qty{e-4}{eV}$ (yakni
dublet inversi jilid Tahun 2). (a) Berilah alasan bagi perlakuan amonia
sebagai sistem dua aras. (b) Berapa frekuensi transisi dan panjang
gelombangnya? (c) Dalam maser (1954), molekul terpilih keadaannya yang
melintasi rongga resonan menguatkan gelombang mikro itu: syarat populasi
apa yang harus dipenuhi berkas yang masuk, dan bagaimana bedanya dari
keadaan termal? (d) Apa yang diperagakan maser amonia tiga tahun sebelum
laser optis jilid Tahun 2?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:10}
Kopling spin--orbit, ditaksir. Dalam kerangka diam elektronnya, intinya
mengelilinginya: elektronnya duduk dalam medan magnet $B_{\text
{int}}$. (a) Dari \cref{exo:b3:hydrogen-atom:10}(d), ambillah
$B_{\text{int}} \sim \qty{10}{T}$ untuk hidrogen $n = 2$ lalu
perkirakanlah pergeseran arasnya $\mu_{\text{B}}B_{\text{int}}$:
bandingkanlah dengan skala struktur halus pada
\cref{exo:b3:hydrogen-atom:11}. (b) Garis D natrium terbelah sebesar
\qty{0.6}{nm} pada \qty{589}{nm}: ubahlah menjadi meV dan menjadi sebuah
medan internal. (c) Mengapa pembelahannya tumbuh curam terhadap $Z$
(medan dalamnya berskala kira-kira seperti $Z^4$ dibagi $n^3$)?
(d) Keadaannya dilabeli dengan $j = l \pm \tfrac12$ total: cacahlah
keadaan sebuah aras $p$ lalu periksalah bahwa tak ada yang hilang.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:11}
Dua \omterm{def:b3:spin-two-level:spin}{spin} bersama. Untuk dua partikel berspin $\tfrac12$, \omterm{def:b3:spin-two-level:spin}{spin} totalnya
$\hat{\vect S} = \hat{\vect S}_1 + \hat{\vect S}_2$. (a) Tunjukkanlah
bahwa keempat keadaan hasil kalinya tertata ulang menjadi sebuah
\emph{triplet} ($S = 1$:
$\ket{\uparrow\uparrow}$, $(\ket{\uparrow\downarrow} +
\ket{\downarrow\uparrow})/\sqrt2$, $\ket{\downarrow\downarrow}$)
dan sebuah \emph{singlet} ($S = 0$: $(\ket{\uparrow\downarrow} -
\ket{\downarrow\uparrow})/\sqrt2$) --- periksalah cacah $S_z$-nya dan,
untuk kedua keadaan tengahnya, pengaruh $\hat S^2$ (pakailah $\hat
S^2 = \hat S_1^2 + \hat S_2^2 + 2\hat{\vect S}_1\cdot\hat{\vect
S}_2$ dan $\hat{\vect S}_1\cdot\hat{\vect S}_2 = \tfrac12(\hat
S_{1+}\hat S_{2-} + \hat S_{1-}\hat S_{2+}) + \hat S_{1z}\hat
S_{2z}$). (b) Yang manakah di antara keempatnya yang terbelit --- yang
tak dapat dituliskan sebagai hasil kali? (c) Pasangan hiperhalus
hidrogen (\cref{ex:b3:spin-two-level:hyperfine}) persis merupakan
triplet/singlet ini: yang mana yang lebih tinggi energinya, mengingat
garisnya \emph{dipancarkan} pada 21 cm? (d) \omterm{def:b3:spin-two-level:spin}{Spin} singletnya
berantikorelasi sempurna sepanjang \emph{tiap} sumbu: ukurlah yang satu
sepanjang $\vect n$ mana pun dan yang lain menjawab sebaliknya. Mengapa
korelasi ini, sedahsyat apa pun, tidak mengirimkan isyarat apa pun?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spin-two-level:12}
Jam yang mendefinisikan sekon. Pembelahan hiperhalus keadaan dasar
sesium tepat $\qty{9192631770}{Hz}$ (menurut definisi sekonnya). Dalam
jam pancuran, atomnya memperoleh pulsa $\pi/2$, terbang bebas selama
$T = \qty{0.5}{s}$, lalu memperoleh pulsa $\pi/2$ kedua; bagian yang
terpindahkan berosilasi sebagai $\cos^2(\pi\,\delta\,T)$ dengan
pelencengan $\delta$ (yakni rumbai Ramsey --- terimalah ini). (a)
Jelaskanlah dalam bahasa Bloch apa yang dilakukan tiap pulsa $\pi/2$
beserta penerbangan bebasnya. (b) Berapa lebar rumbai pusatnya? (c)
Dengan nisbah isyarat terhadap derau yang memungkinkan pusat rumbainya
dibelah sebesar $10^4$, ketepatan frekuensi pecahan berapa yang
dihasilkannya? (d) Mengapa jam optis (transisi petahertz) mengalahkan
jam gelombang mikro pada kemampuan pembelahan rumbai yang tetap --- dan
kira-kira sebesar faktor berapa?
\end{exercise}

\section{Soal: Melihat ke dalam tubuh}

\begin{problem}[{Soal akhir pekan --- pencitraan resonansi magnetik,
dari spin ke pindaian}]\label{pb:b3:spin-two-level:1}
Dua pertiga tubuh manusia adalah air; tiap molekul air membawa dua
proton, yang masing-masingnya sebuah magnet berspin $\tfrac12$.
Tempatkanlah seseorang dalam medan yang kuat, gelitiklah protonnya pada
frekuensi Larmornya, lalu dengarkanlah: itulah pencitraan \omterm{prop:b3:spin-two-level:rabi}{resonansi
magnetik}, yakni fisika \omterm{def:b3:spin-two-level:spin}{spin} sebagai kedokteran. Data:
$\gamma_{\text{p}}/2\pi = \qty{42.58}{MHz/T}$; $B_0 = \qty{3.0}{T}$;
kerapatan proton jaringannya $n \approx \qty{6.6e28}{m^{-3}}$;
$k_{\text{B}}T$ pada \qty{310}{K} adalah \qty{26.7}{meV}; $h =
\qty{4.14e-15}{eV.s}$.

\medskip\noindent\textbf{Bagian I --- \omterm{def:b3:spin-two-level:spin}{Spin} di dalam magnetnya.}
\begin{enumerate}
  \item Hitunglah frekuensi Larmornya pada \qty{3.0}{T} beserta energi
        fotonnya $h\nu$ dalam eV.
  \item Hitunglah ketakseimbangan populasi antara kedua aras protonnya,
        $\Delta n/n \approx h\nu/2k_{\text{B}}T$.
  \item Hanya kelebihan ini --- beberapa per sejuta --- yang menyumbang
        isyarat: berapa proton yang ``berguna'' tiap sentimeter kubik
        jaringannya?
  \item Mengapa polarisasi termalnya begitu lemah di sini padahal berkas
        Stern--Gerlachnya terbelah sepenuhnya? (Apa yang diukur
        masing-masing percobaan: \omterm{def:b3:quantum-formalism:observable}{nilai eigen} atom tunggal, atau
        rata-rata termal?)
  \item Melipatduakan $B_0$ berbuat apa pada isyaratnya (dua efek:
        polarisasinya dan ggl imbasannya pada frekuensi yang lebih
        tinggi)? Mengapa rumah sakit membayar magnet yang lebih besar?
  \item Magnetnya adiwarsa dan selalu menyala: mengapa sebuah tabung
        oksigen baja yang lepas di dalam ruangan itu menjadi proyektil
        yang mematikan (gaya yang mana pada
        \cref{ex:b3:spin-two-level:stern-gerlach} yang bekerja padanya)?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pulsa dan gema.}
\begin{enumerate}[resume]
  \item Sebuah kumparan melintang mengenakan $B_1 = \qty{10}{\micro T}$
        pada resonansinya: hitunglah frekuensi Rabinya beserta durasi
        pulsa $\pi/2$ dan $\pi$-nya.
  \item Setelah pulsa $\pi/2$, apa yang dilakukan kemagnetannya, dan apa
        yang diimbaskan hukum Faraday dalam kumparan penerimanya (pada
        frekuensi berapa)?
  \item Isyarat melintangnya meluruh dengan tetapan waktu $T_2$
        (\omterm{def:b3:spin-two-level:spin}{spinnya} kehilangan fase dalam medan satu sama lain dan dalam
        ketakhomogenan setempatnya); kemagnetan membujurnya tumbuh
        kembali dengan $T_1$ (energinya mengalir ke jaringannya). Nilai
        khasnya: $T_1 \approx \qty{1}{s}$, $T_2 \approx \qty{0.1}{s}$.
        Mengapa waktu keterpaduannya tak pernah dapat jauh melampaui
        waktu pengenduran energinya (apa yang dilakukan tiap pembalikan
        yang menukar energi terhadap fasenya)?
  \item Gema \omterm{def:b3:spin-two-level:spin}{spin}: setelah pulsa $\pi/2$ dan selang $\tau$, sebuah pulsa
        $\pi$ dikenakan dan, pada $2\tau$, \omterm{def:b3:spin-two-level:spin}{spin} yang tadinya kehilangan
        fase menyearah kembali dan isyaratnya kembali. Jelaskanlah
        muslihatnya dengan pelari di lintasan yang disuruh berbalik arah.
  \item Peluruhan yang mana yang dibatalkan gemanya --- kehilangan fase
        akibat ketakhomogenan medan yang \emph{statis}, atau akibat
        medan molekul yang berfluktuasi --- dan mengapa hanya yang itu?
  \item Jaringan yang berbeda mempunyai $T_1$, $T_2$ yang berbeda
        (lemak: $T_1$ pendek; air atau cairan: panjang; banyak tumor:
        $T_2$ lebih panjang daripada jaringan tuan rumahnya). Dalam satu
        kalimat: bagaimana mengatur waktu urutan pulsanya mengubah waktu
        pengenduran menjadi \emph{ketegasan} citranya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Dari isyarat ke citra.}
\begin{enumerate}[resume]
  \item Tumpangkanlah gradien $G = \qty{40}{mT/m}$ sepanjang $z$:
        frekuensi Larmornya menjadi bergantung kedudukan. Hitunglah
        $\dd\nu/\dd z$ dalam kHz tiap milimeter.
  \item Sebuah pulsa RF berbentuk yang hanya memuat pita $\nu_0 \pm
        \qty{1}{kHz}$ mengeksitasi lempeng tubuh yang mana? (Yakni
        pemilihan irisannya.)
  \item Selama pembacaannya, gradien sepanjang $x$ membuat tiap kolom
        irisannya menyiarkan frekuensinya sendiri: operasi matematis apa
        yang mengubah isyarat waktu yang diterima menjadi profil ruang
        (ingatlah perkakas Fourier jilid Tahun 2)?
  \item Perkirakanlah informasi pindaian totalnya: sebuah citra
        $256 \times 256$ pada 12 bit, dan mengapa memerolehnya baris
        demi baris (satu gema tiap baris, dengan waktu pengulangan
        $\sim T_1$) membuat satu pindaian memakan waktu beberapa menit.
  \item Pasiennya mendengar ketukan keras: apa yang berdentam secara
        mekanis (pikirkanlah kumparan gradiennya yang beralih dalam
        medan \qty{3}{T} --- gaya yang mana)?
  \item Pencitraan sinar-X menegaskan kerapatan elektron; MRI menegaskan
        kerapatan proton \emph{dan} pengendurannya: mengapa MRI menjadi
        pilihan bagi jaringan lunak dan otak, dan dosis pengion sebesar
        apa yang diberikannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Pekerjaan lain sang \omterm{def:b3:spin-two-level:spin}{spin}.}
\begin{enumerate}[resume]
  \item Para kimiawan menjalankan percobaan yang sama pada resolusi
        \qty{10}{ppm}: awan elektron sedikit memerisai tiap intinya,
        sehingga menggeser resonansinya (yakni ``pergeseran kimia'').
        Mengapa hal ini menjadikan NMR sidik jari molekul?
  \item MRI fungsional memetakan pikiran: hemoglobin yang kehabisan
        oksigen bersifat paramagnetik dan memperpendek $T_2^*$
        setempatnya. Telusurilah rantainya dari kegiatan saraf sampai
        keterangan citranya.
  \item Fisika Larmor yang sama dengan momen \emph{elektron} berjalan
        pada \qty{84}{GHz} pada \qty{3}{T}: mengapa resonansi elektron
        tak berguna di dalam tubuh manusia namun berharga untuk
        mempelajari radikal dan cacat?
  \item Bandingkanlah energi foton pada butir 1 dengan energi ikatan
        kimia yang khas: berilah alasan bagi sebutan ``tak mengionkan''
        lalu pertentangkanlah dengan foton sinar-X \qty{60}{keV}.
  \item Polarisasi sepuluh per sejutanya dapat ditingkatkan: gas xenon
        yang terpolarisasi laser mencapai polarisasi berorde satu lalu
        dihirup untuk mencitrakan rongga udara paru-paru. Kira-kira
        sebesar faktor berapa isyarat tiap intinya naik, dan mengapa
        kerapatan gasnya yang rendah tetap membuat muslihat itu
        diperlukan?
  \item Sebuah \omterm{def:b3:spin-two-level:spin}{spin} elektron tunggal dalam perangkap mirip transistor
        silikon kini dibaca dan digerakkan sebagai kubit dengan persis
        pulsa bab ini: sebutkanlah dua sifat \omterm{def:b3:spin-two-level:spin}{spin} (ukuran \omterm{def:b3:quantum-formalism:state}{ruang
        Hilbertnya}; lemahnya koplingnya terhadap derau) yang
        merekomendasikannya.
  \item Ringkaslah hasil bernama itu: presesi \omterm{def:b3:spin-two-level:spin}{spin} \qty{128}{MHz},
        polarisasi termal sepuluh per sejuta, dua jam pengenduran dan
        tiga gradien medan sudah cukup untuk memotret otak yang hidup
        dalam irisan --- yakni dua bercak Stern dan Gerlach, yang tumbuh
        menjadi sebuah rumah sakit.
\end{enumerate}
\end{problem}
