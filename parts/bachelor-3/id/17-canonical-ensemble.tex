\chapter{Ensembel Kanonik}\label{ch:b3:canonical-ensemble}

Sistem terkucil adalah khayalan seorang teoretikus: cuplikan yang
sebenarnya duduk dalam termostat, ruangan, samudra udara --- yakni dalam
sentuhan dengan sebuah tandon yang menetapkan bukan energinya melainkan
\emph{suhunya}. Jilid Tahun 2 menemui hukum yang dihasilkannya secara
empiris: peluang sebuah keadaan berenergi $E$ membawa faktor
$\eu^{-E/k_{\text{B}}T}$. Bab ini menurunkan faktor Boltzmann itu dalam
empat baris dari pencacahan \cref{ch:b3:microcanonical-ensemble}, lalu
membangun mesin yang menjadikan fisika statistik sebuah disiplin
industri: yakni \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{fungsi partisi}} $Z$, satu jumlahan yang darinya
energi, entropi, tekanan, kapasitas kalor dan fluktuasi semuanya jatuh
lewat penurunan. Kemenangan pertama mesin itu diceritakan di sini:
mengapa kapasitas kalor mati pada suhu rendah (Einstein, 1907 --- yakni
teori kuantum pertama tentang materi), mengapa kapasitas kalor hidrogen
memanjat sebuah tangga, mengapa langit menipis secara eksponensial ---
dan bagaimana Perrin, dengan mengamati butiran mikroskopis mengendap
dalam setetes air, mencacah bilangan Avogadro lalu mengakhiri perdebatan
tentang kenyataan atom.

\section{Sebaran Boltzmann, diturunkan}

\begin{theorem}[Sebaran kanonik]\label{thm:b3:canonical-ensemble:boltzmann}
\index{sebaran Boltzmann}\index{ensembel kanonik}\index{fungsi partisi}
Sistem yang bertukar energi dengan tandon besar bersuhu $T$ menempati
\omterm{def:b3:microcanonical-ensemble:states}{keadaan mikronya} $s$ (berenergi $E_s$) dengan peluang
\[
\mathcal P_s = \frac{\eu^{-E_s/k_{\text{B}}T}}{Z} , \qquad
Z = \sum_s \eu^{-E_s/k_{\text{B}}T}
\]
($\beta = 1/k_{\text{B}}T$ untuk selanjutnya). $Z$, yakni \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{fungsi
partisi}}, menormalkan peluangnya --- dan ternyata memuat seluruh
termodinamika sistemnya.
\end{theorem}

\begin{proof}
Sistem yang berada dalam keadaan $s$ menyisakan energi $E - E_s$ bagi
tandonnya: menurut \omterm{thm:b3:microcanonical-ensemble:postulate}{postulat dasar} yang dikenakan pada keseluruhannya,
$\mathcal P_s
\propto \Omega_{\text{res}}(E - E_s)$. Jabarkanlah entropi tandonnya,
yang raksasa dan mulus, sampai orde pertama: $S_{\text{res}}(E - E_s)
= S_{\text{res}}(E) - E_s\,\partial S_{\text{res}}/\partial E =
S_{\text{res}}(E) - E_s/T$ menurut definisi suhunya. Sehingga
$\Omega_{\text{res}} \propto \eu^{-E_s/k_{\text{B}}T}$. (Orde yang lebih
tinggi mati bersama besarnya tandonnya.)
\end{proof}

\begin{proposition}[Mesinnya]\label{prop:b3:canonical-ensemble:machine}
\index{energi bebas}\index{kapasitas kalor!dan fluktuasi}
Dari $Z(T, V, N)$:
\[
\langle E\rangle = -\frac{\partial\ln Z}{\partial\beta} , \qquad
F = -k_{\text{B}}T\ln Z , \qquad
S = -\frac{\partial F}{\partial T} , \qquad
P = -\frac{\partial F}{\partial V} ,
\]
dengan $F = \langle E\rangle - TS$ adalah \emph{energi bebasnya}; dan
kapasitas kalornya merupakan sebuah \emph{fluktuasi}:
\[
C = \frac{\partial\langle E\rangle}{\partial T}
 = \frac{\langle E^2\rangle - \langle E\rangle^2}
{k_{\text{B}}T^2} .
\]
Untuk subsistem yang saling bebas dan dapat dibedakan $Z = z^N$; untuk
$N$ \omterm{def:b3:identical-particles:exchange}{partikel identik} dalam kotak bersama, $Z = z^N/N!$. Pada $T$ dan $V$
tetap, kesetimbangan \emph{meminimumkan} $F$: alam mempertukarkan energi
dengan entropi pada nilai tukar $T$ --- yakni asas paling berguna dalam
fisika materi.
\end{proposition}

\begin{proof}[Bukti sebagian]
$-\partial_\beta\ln Z = \sum E_s\eu^{-\beta E_s}/Z = \langle
E\rangle$; satu turunan lagi memberikan $\langle E^2\rangle - \langle
E\rangle^2$, dan aturan rantainya mengubah $\partial_\beta$ menjadi
$\partial_T$. Penyamaan $F$, $S$, $P$ menyusul dengan membandingkan
$\dd(-k_{\text{B}}T\ln Z)$ dengan bentuk termodinamikanya
$\dd F = -S\dd T - P\dd V$; pemfaktorannya adalah eksponensial sebuah
jumlahan. Asas minimumnya: $F_{\text{sistem}}$ yang menurun berarti
$S_{\text{total}}$ yang menaik, karena $\Delta S_{\text{res}} =
-\Delta E_{\text{sis}}/T$.
\end{proof}

\section{Kemenangan pertama}

\begin{example}[Dua aras, dalam dua baris]\label{ex:b3:canonical-ensemble:twolevel}
Untuk aras $0, \epsilon$: $z = 1 + \eu^{-\beta\epsilon}$, sehingga
$\langle E\rangle = N\epsilon/(\eu^{\beta\epsilon} + 1)$ --- yakni hasil
yang menghabiskan satu halaman Stirling pada rute mikrokanoniknya
(\cref{exo:b3:microcanonical-ensemble:8}). Formalisme kanoniknya adalah
formalisme mikrokanoniknya dengan kombinatorikanya sudah dicerna
terlebih dahulu; tonjolan Schottky pada $C(T)$ menyusul lewat satu
penurunan.
\end{example}

\begin{example}[Zat padat Einstein, dan matinya Dulong--Petit]\label{ex:b3:canonical-ensemble:einstein}
\index{zat padat Einstein}\index{hukum Dulong--Petit}
Modelkanlah sebuah kristal sebagai $3N$ osilator kuantum berfrekuensi
$\omega$ (\cref{ch:b3:harmonic-oscillator}). Tiap osilatornya,
\[
z = \sum_n\eu^{-\beta\hbar\omega(n + 1/2)}
 = \frac{1}{2\sinh(\beta\hbar\omega/2)} , \qquad
\langle E\rangle = \frac{\hbar\omega}{2} +
\frac{\hbar\omega}{\eu^{\beta\hbar\omega} - 1} .
\]
Pada $T$ tinggi: $\langle E\rangle \to k_{\text{B}}T$ tiap osilatornya,
$C \to 3Nk_{\text{B}}$ --- yakni \omterm{ex:b3:canonical-ensemble:einstein}{hukum Dulong--Petit} jilid Tahun 1, yang
kini terjelaskan. Pada $T$ rendah kuantum $\hbar\omega$-nya menjadi tak
terbeli dan $C$ runtuh secara eksponensial --- sebagaimana terukur, dan
tak terjelaskan secara klasik. Kurva Einstein 1907, dengan satu
parameter tiap unsurnya, adalah penerapan pertama kuanta pada materi
biasa; intan, dengan ikatannya yang kaku dan atomnya yang ringan
($\hbar\omega/k_{\text{B}}
\approx \qty{1300}{K}$), masih ``beku'' pada suhu ruang --- yakni
kejanggalan yang membingungkan para kimiawan selama delapan puluh tahun
(\cref{exo:b3:canonical-ensemble:6}).
\end{example}

\begin{theorem}[Ekipartisi, beserta izinnya]\label{thm:b3:canonical-ensemble:equipartition}
\index{teorema ekipartisi}
Tiap koordinat atau momentum yang masuk ke energinya secara
\emph{kuadratik} menyumbang, dalam rezim klasik (suhu tinggi),
\[
\langle\epsilon\rangle = \tfrac12 k_{\text{B}}T
\]
pada energi rata-ratanya: $\tfrac32 k_{\text{B}}T$ untuk sebuah atom gas
monatom, $\tfrac52$ untuk molekul dwiatom yang berputar,
$3k_{\text{B}}T$ untuk sebuah osilator. Izinnya kedaluwarsa ketika
$k_{\text{B}}T$ jatuh di bawah jarak aras ragamnya: ragamnya membeku dan
sumbangannya lenyap --- yakni penyelesaian bagi skandal kapasitas kalor
abad kesembilan belas, yang tergambar sebagai tangga di atas.
\end{theorem}

\begin{proof}[Bukti sebagian]
Untuk $\epsilon = ax^2$,
\[
\langle\epsilon\rangle
 = \frac{\int ax^2\,\eu^{-\beta ax^2}\dd x}
        {\int\eu^{-\beta ax^2}\dd x}
 = -\partial_\beta\ln\!\int\eu^{-\beta ax^2}\dd x
 = -\partial_\beta\ln\beta^{-1/2} = \frac{1}{2\beta} .
\] Pembekuannya adalah perhitungan
\cref{ex:b3:canonical-ensemble:einstein}, ragam demi
ragam.
\end{proof}

\begin{example}[Gasnya, secara kanonik]\label{ex:b3:canonical-ensemble:gas}
Satu atom dalam sebuah kotak: $z = V/\lambda_T^3$ (yakni cacah keadaan
\cref{prop:b3:schrodinger-three-dimensions:counting} yang dibobot
Boltzmann); untuk $N$ atom identik: $Z = z^N/N!$. Maka $F =
-Nk_{\text{B}}T[\ln(V/N\lambda_T^3) + 1]$, dan penurunannya menyerahkan
$PV = Nk_{\text{B}}T$, entropi Sackur--Tetrode, dan
$\langle E\rangle = \tfrac32 Nk_{\text{B}}T$ --- yakni seluruh gas ideal
dari satu integral Gauss. Faktor Boltzmann yang dikenakan pada energi
kinetik sebuah molekul memberikan sebaran laju Maxwell jilid Tahun 1,
yang kini diturunkan; dikenakan pada energi potensialnya $mgh$ ia
memberikan atmosfer eksponensialnya --- dan, dalam setetes air, tangga
butiran Perrin
(\cref{pb:b3:canonical-ensemble:1}).
\end{example}

\begin{method}[Keterampilan kanonik]\label{met:b3:canonical-ensemble:method}
(1) Daftarkanlah keadaan dan energi \emph{satu} satuannya; lalu
hitunglah $z$. (2) Faktorkanlah: $Z = z^N$ (atau $z^N/N!$ untuk \omterm{def:b3:identical-particles:exchange}{partikel
identik} yang berbagi ruang); lalu ambillah $\ln$-nya sejak awal. (3)
Turunkanlah: terhadap $\beta$ untuk energinya, terhadap $T$ untuk
entropinya lewat $F$, terhadap $V$ untuk tekanannya; turunan kedua untuk
$C$ dan fluktuasinya. (4) Periksalah kedua ujungnya: $T$ tinggi harus
menghasilkan kembali ekipartisinya, dan $T$ rendah harus membeku dengan
ekor $\eu^{-\Delta/k_{\text{B}}T}$. (5) Persaingan (pelipatan,
pengikatan, penyearahan): tuliskanlah $F = E - TS$ untuk tiap pilihannya
lalu biarkanlah yang lebih kecil menang --- perpindahannya terletak pada $T \approx \Delta
E/\Delta S$.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:1}
(a) Tuliskanlah nisbah populasi dua aras yang terbelah sebesar $\epsilon$
pada suhu $T$. (b) Dalam nyala \qty{2500}{K}, berapa bagian atom natrium
yang duduk dalam keadaan tereksitasi \qty{2.1}{eV} garis D-nya
(degenerasi dasarnya 2, tereksitasinya 6: nisbah populasinya
$3\eu^{-\beta\epsilon}$)?
(c) Mengapa nyalanya toh berpijar kuning (berapa atom tiap
cm$^3$ sudah cukup)? (d) Pada suhu berapa bagian tereksitasinya akan
mencapai $10\%$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:2}
Sebuah sistem tiga aras: $0, \epsilon, 2\epsilon$. (a) Tuliskanlah $z$.
(b) Hitunglah $\langle E\rangle$ lalu periksalah kedua limit suhunya. (c)
Pada $T$ berapa aras tengahnya terisi maksimum \emph{secara mutlak}? (d)
Tunjukkanlah bahwa tak ada suhu, setinggi apa pun, yang membuat aras yang
lebih tinggi lebih terisi daripada yang lebih rendah --- gagasan bab 16
yang mana yang diperlukan untuk itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:3}
Atmosfer isotermal. (a) Kenakanlah faktor Boltzmann pada energi
potensialnya $mgh$ lalu turunkanlah $n(h) = n_0\eu^{-mgh/k_{\text{B}}
T}$. (b) Hitunglah tinggi skalanya untuk udara ($m = \qty{4.8e-26}{kg}$)
pada \qty{288}{K}. (c) Berapa tekanan puncak Everest sebagai bagian dari
tekanan permukaan laut? (d) Mengapa peluruhan atmosfer yang sebenarnya
dekat tetapi tidak persis eksponensial (apa yang kita tahan tetap padahal
sebenarnya tidak)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:canonical-ensemble:4}
Pembukuan ekipartisi. Cacahlah suku kuadratiknya lalu ramalkanlah $C_V$
molar (a) argon; (b) N$_2$ pada suhu ruang (rotasinya menyala,
getarannya beku); (c) N$_2$ pada \qty{3000}{K}; (d) sebuah zat padat
kristal (Dulong--Petit). Di manakah nilai terukurnya
$12.5$, $20.8$, $\approx26$, $\approx\qty{25}{J/(mol.K)}$ sesuai,
dan apa yang diajarkan tiap ketaksesuaiannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:5}
Osilator kuantum, secara kanonik. (a) Jumlahkanlah deret geometrinya
untuk memperoleh $z$. (b) Turunkanlah $\langle E\rangle$ lalu kenalilah $\langle
n\rangle = 1/(\eu^{\beta\hbar\omega} - 1)$ --- yakni hasil
\cref{exo:b3:harmonic-oscillator:6}, yang kini tanpa susah payah. (c)
Turunkanlah untuk memperoleh $C(T)$ lalu periksalah kedua limitnya. (d)
Mengapa bentuk $\langle n\rangle$ segera menjadi termasyhur
(\cref{ch:b3:photons-phonons})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:6}
Einstein berhadapan dengan datanya. (a) Dari rumus pada gambarnya, pada
$T/\theta_{\text{E}}$ berapa $C$ telah jatuh menjadi separuh
Dulong--Petit? (b) Intan: $\theta_{\text{E}} \approx \qty{1300}{K}$ ---
hitunglah $C/3Nk_{\text{B}}$ pada \qty{300}{K} lalu jelaskanlah
``kejanggalan intan'' abad kesembilan belas. (c) Timbal: $\theta_{\text{E}}
\approx \qty{90}{K}$ --- mengapa timbal selalu ``berkelakuan baik''? (d)
Pengukuran pada $T$ yang sangat rendah menunjukkan $C \propto T^3$, bukan
eksponensial: anggapan Einstein yang mana yang gagal (semua osilatornya
berfrekuensi \emph{satu}), dan siapa yang memperbaikinya
(\cref{ch:b3:photons-phonons})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:7}
Ekor Maxwell dan hidrogen yang hilang. (a) Dari faktor Boltzmann pada
energi kinetiknya, tuliskanlah sebaran lajunya lalu tentukanlah laju
paling mungkin untuk N$_2$ dan H$_2$ pada \qty{288}{K}. (b)
Laju lepas Bumi \qty{11.2}{km/s}: hitunglah
$mv_{\text{esc}}^2/2k_{\text{B}}T$ untuk kedua gasnya. (c) Bagian yang
lolos berjalan sebagai $\eu^{-mv_{\text{esc}}^2/2k_{\text{B}}T}$:
bandingkanlah kedua eksponennya lalu simpulkanlah gas mana yang bocor
sepanjang waktu geologi. (d) Kaitkanlah dengan susunan teramati atmosfer
Bumi (tanpa H$_2$ bebas) berbanding atmosfer Jupiter (sebagian besar
H$_2$) --- dua parameter apa yang membalikkan putusannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:8}
Fluktuasi bertemu tanggapan. (a) Buktikanlah $C = (\langle E^2\rangle -
\langle E\rangle^2)/k_{\text{B}}T^2$ dari dua turunan $\ln
Z$. (b) Periksalah secara gamblang pada sistem dua arasnya. (c) Untuk
$N$ satuan yang saling bebas, tunjukkanlah bahwa fluktuasi energi
\emph{nisbi}nya turun sebagai $1/\sqrt N$. (d) Nyatakanlah pelajarannya:
apa hubungan kesediaan sebuah sistem menyerap kalor (sebuah tanggapan)
dengan seberapa besar energinya bergetar (sebuah fluktuasi) --- yakni
tawar-menawar yang berulang dalam fisika statistik.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:canonical-ensemble:9}
Boltzmann di laboratorium kimia. Laju reaksi membawa faktor
$\eu^{-E_a/k_{\text{B}}T}$ (melintasi penghalang pengaktifan $E_a$ ---
Arrhenius). (a) Tunjukkanlah bahwa patokan kasar ``lajunya berlipat dua
tiap \qty{10}{K} di dekat suhu ruang'' bersesuaian dengan $E_a \approx
\qty{0.55}{eV}$. (b) Sebesar faktor berapa reaksi itu melambat dalam
lemari es (\qty{5}{\celsius})? (c) Merebus telur di ketinggian: air
mendidih pada \qty{93}{\celsius} di sebuah celah \qty{2000}{m} ---
perkirakanlah tambahan waktu memasaknya. (d) Mengapa ekor penghalangnya,
bukan energi rata-ratanya, yang memerintah kimia (molekul mana yang
bereaksi)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:10}
Pelipatan dua keadaan. Sebuah biomolekul berada dalam keadaan terlipat
(berenergi $0$, satu susunan) atau terbuka (berenergi $\Delta E > 0$,
dengan $\Omega_u =
\eu^{\Delta S/k_{\text{B}}}$ susunan). (a) Tuliskanlah bagian yang
terlipat terhadap $T$. (b) Tunjukkanlah bahwa titik tengah
``peleburannya'' adalah $T_{\text{m}} = \Delta E/\Delta S$ lalu
tafsirkanlah sebagai pertarungan $F = E -
TS$. (c) Dengan $\Delta E = \qty{3.0}{eV}$ dan $\Delta S =
100\,k_{\text{B}}$ (yakni satuan kooperatif sebuah protein kecil):
hitunglah $T_{\text{m}}$ beserta lebar transisinya. (d) Mengapa
\emph{kekooperatifan} (banyak sentuhan yang putus bersamaan, dengan
$\Delta E$ \emph{dan} $\Delta S$ yang besar) menajamkan peleburannya ---
dan bagaimana mesin pendaur termal DNA (PCR) memanfaatkan persis hal
ini?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:11}
Paramagnetisme dan pendinginan magnetik. Sebanyak $N$ spin $\tfrac12$
bermomen $\mu$ dalam medan $B$. (a) Tunjukkanlah
$M = N\mu\tanh(\mu B/k_{\text{B}}T)$ lalu jabarkanlah menjadi hukum
Curie $M \approx N\mu^2B/k_{\text{B}}T$. (b) Hitunglah penyearahannya
$\mu B/k_{\text{B}}T$ untuk momen elektron pada $B = \qty{1}{T}$,
$T = \qty{300}{K}$ dan pada \qty{1}{K}. (c) Penyahmagnetan adiabatik:
magnetkanlah pada \qty{1}{K}, kucilkanlah, lalu kurangilah $B$ sepuluh
kali lipat --- tunjukkanlah bahwa entropi yang tetap berarti
$\mu B/k_{\text{B}}T$ yang tetap, sehingga $T$ turun sepuluh kali lipat.
(d) Apa yang menetapkan lantai lemari es ini (interaksi antarspinnya ---
perkirakanlah skala dipolarnya $\mu_0\mu^2/4\pi a^3$ untuk $a =
\qty{0.5}{nm}$, dalam satuan suhu)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:canonical-ensemble:12}
Tangga rotasi, secara kuantitatif. Aras rotasi sebuah molekul dwiatom
adalah $E_J = BJ(J+1)$, dengan degenerasi $2J + 1$
(\cref{ch:b3:quantum-angular-momentum}). (a) Tuliskanlah
$z_{\text{rot}}$ lalu tunjukkanlah bahwa untuk $k_{\text{B}}T \gg B$
jumlahannya menjadi integral $k_{\text{B}}T/B$: sehingga $k_{\text{B}}$
ekipartisi dalam $C_V$ diperoleh kembali. (b) Definisikanlah $\theta_{\text{rot}} =
B/k_{\text{B}}$ lalu hitunglah untuk H$_2$
($B = \qty{7.5}{meV}$) dan N$_2$ ($B = \qty{0.25}{meV}$): gas mana yang
menunjukkan anak tangga rotasinya pada suhu yang terjangkau? (c)
Sketsalah tangga $C_V(T)$ lengkap hidrogen dengan ketiga datarannya dan
kedua tanjakannya, lalu tempatkanlah bilangannya pada keduanya. (d)
Perilaku $T$ rendah H$_2$ yang terukur makin rumit karena campuran
orto--para 3:1 pada \cref{exo:b3:identical-particles:8}: nyatakanlah
dalam satu kalimat bagaimana statistik spin inti menjangkau sampai ke
kapasitas kalor sebuah gas.
\end{exercise}

\section{Soal: Mencacah Avogadro dalam setetes air}

\begin{problem}[{Soal akhir pekan --- butiran Perrin dan kenyataan
atom}]\label{pb:b3:canonical-ensemble:1}
Pada 1908 Jean Perrin melayangkan butiran damar mikroskopis dalam air,
membiarkannya mengendap, lalu mencacahnya lapis demi lapis di bawah
mikroskop. Tangga eksponensial yang ditemukannya adalah faktor Boltzmann
yang dijadikan kasat mata --- dan dari tinggi skalanya ia menarik
bilangan Avogadro, sehingga meyakinkan para peragu terakhir bahwa atom
memang ada. Hadiah Nobel 1926. Data: butiran gamboge berjejari $r =
\qty{0.212}{\micro m}$, bermassa jenis $\rho_{\text{g}} =
\qty{1207}{kg/m^3}$; air $\rho_{\text{w}} = \qty{999}{kg/m^3}$;
$T = \qty{293}{K}$; $g = \qty{9.81}{m/s^2}$.

\medskip\noindent\textbf{Bagian I --- Faktor Boltzmann yang kasat mata.}
\begin{enumerate}
  \item Sebuah butiran dalam air merasakan gravitasi dikurangi gaya
        apungnya: hitunglah massa \emph{efektif}nya $m' = \tfrac43\pi r^3(\rho_{\text{g}}
        - \rho_{\text{w}})$ beserta beratnya.
  \item Tuliskanlah profil kepekatan kesetimbangannya $n(h)$ dari
        faktor Boltzmannnya.
  \item Hitunglah tinggi skalanya $h_0 = k_{\text{B}}T/m'g$ dengan
        $k_{\text{B}}$ modern.
  \item Mengapa butirannya harus berukuran seragam dengan begitu tepat
        (bagaimana $h_0$ bergantung pada $r$)?
  \item Bandingkanlah $h_0$ dengan rumus yang sama untuk molekul udara:
        mengapa atmosfer butirannya setinggi beberapa mikrometer
        sedangkan atmosfer udaranya setinggi beberapa kilometer?
  \item Perrin mencacah (dalam satu percobaan) kepekatan nisbi
        $100 : 55 : 30 : 17$ pada empat kedalaman berjarak sama
        \qty{30}{\micro m}: periksalah bahwa ini sebuah tangga
        eksponensial lalu tariklah $h_0$-nya.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Menimbang yang tak terlihat.}
\begin{enumerate}[resume]
  \item Baliklah: dari $h_0$ terukur pada butir 6 dan $m'g$ yang
        diketahui, tariklah $k_{\text{B}}$.
  \item Tetapan gas $R = \qty{8.314}{J/(mol.K)}$ sudah dikenal dari
        kimia makroskopis: gabungkanlah dengan $k_{\text{B}}$ Anda untuk
        memperoleh bilangan Avogadro $N_{\text{A}} =
        R/k_{\text{B}}$.
  \item Percobaan Perrin memberikan $N_{\text{A}}$ antara $5.5$ dan
        $\num{7.2e23}$: bandingkanlah dengan nilai modern
        \num{6.022e23} lalu berilah ulasan tentang pencapaiannya
        mengingat mikroskop dan pencatat waktu yang dimilikinya.
  \item Dari $N_{\text{A}}$, hitunglah massa satu \omterm{thm:b3:hydrogen-atom:levels}{atom hidrogen} ---
        yakni bilangan yang diinginkan para atomis selama seabad.
  \item Jelaskanlah struktur logikanya: pengukuran \emph{makroskopis}
        mana ($R$; ukuran dan massa jenis butirannya; tangganya) yang
        bergabung untuk menimbang satu atom, tanpa satu atom pun pernah
        terlihat?
  \item Mengapa butirannya --- yang masing-masing bermassa $10^{10}$
        massa atom --- menaati statistik Boltzmann yang sama seperti
        molekul (apa dalam penurunan
        \cref{thm:b3:canonical-ensemble:boltzmann} yang peduli pada
        ukurannya)?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Getaran yang mengesahkannya.}
\begin{enumerate}[resume]
  \item Butiran yang sama bergetar: rumus Einstein 1905 bagi gerak
        Brown memberikan $\langle x^2\rangle = 2Dt$ dengan $D =
        k_{\text{B}}T/6\pi\eta r$ (air: $\eta =
        \qty{e-3}{Pa.s}$). Hitunglah $D$ untuk butiran Perrin.
  \item Seberapa jauh sebutir butiran mengembara dalam satu menit?
        Dapatkah Perrin mengukurnya dengan lensa mata bermikrometer dan
        sebuah pencatat waktu?
  \item Perrin memeriksa $\langle x^2\rangle \propto t$ lalu menarik
        $k_{\text{B}}$ \emph{lagi}, secara bebas: mengapa dua rute yang
        tak berkaitan (sebuah tangga statis; sebuah getaran dinamis)
        menuju satu bilangan membawa bobot pembuktian sedemikian besar?
  \item Getarannya adalah ekipartisi yang dikenakan pada butirannya:
        tiap komponen kecepatannya membawa $\tfrac12 k_{\text{B}}T$.
        Perkirakanlah laju termal akar rata-rata kuadrat butirannya ($m \approx
        \qty{4.8e-17}{kg}$).
  \item Mengapa laju itu tak pernah terlihat langsung (apa yang
        memutus penerbangan bebasnya setelah beberapa nanometer), dan
        apa yang \emph{terlihat} sebagai gantinya?
  \item Nyatakanlah dua bab buku ini yang bertemu dalam pengamatan itu:
        mekanika seretan (jilid Tahun 2, kekentalan) dan statistik bab
        ini.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Apa yang terselesaikan.}
\begin{enumerate}[resume]
  \item Ostwald dan Mach berpendapat bahwa atom hanyalah khayalan
        pembukuan: nyatakanlah dalam satu kalimat mengapa $N_{\text{A}}$
        yang \emph{tercacah} dari tangga butiran mengakhiri pendirian itu.
  \item Daftarkanlah tiga rute lain pada awal abad kedua puluh yang menuju
        $N_{\text{A}}$ yang sama (birunya langit,
        \cref{exo:b3:microcanonical-ensemble:12}; elektrolisis
        ditambah muatan elektron; produksi helium oleh keradioaktifan)
        --- mengapa \emph{kesepakatannya} lebih berarti daripada nilai
        tunggal mana pun?
  \item Tangga Perrin adalah kesetimbangan antara dua mata uang bab ini
        yang mana (energi yang menarik ke bawah, entropi yang menyebar
        ke atas), dan dihargai pada nilai berapa?
  \item Kegunaan modern fisika yang sama: ultrasentrifuga analitik
        memutar protein pada $10^5g$ untuk memampatkan
        ``atmosfer''nya menjadi tangga yang terukur --- tunjukkanlah
        bahwa mengalikan $g$ dengan $10^5$ membagi $h_0$ dengan faktor
        yang sama, lalu perkirakanlah $h_0$ untuk sebuah protein
        bermassa efektif \qty{e-22}{kg} pada $10^5g$, \qty{293}{K}.
  \item Melipatduakan jejari butirannya membagi $h_0$ dengan delapan:
        kurunglah jendela praktis ukuran butirannya di antara ``tangganya
        terlalu tinggi untuk melihat gradiennya'' dan ``tangganya lebih
        tipis daripada satu butiran'' bagi medan mikroskop sedalam
        \qty{100}{\micro m}.
  \item Sejak 2019 SI \emph{mendefinisikan} $k_{\text{B}}$ dan
        $N_{\text{A}}$ secara eksak: nyatakanlah apa yang diukur
        percobaan bergaya Perrin \emph{hari ini} (sebuah pemeriksaan
        ketaatasasan, atau kalibrasi perkakas dan butirannya) --- dan
        mengapa fisikanya tidak berubah.
  \item Ringkaslah hasil bernama itu: kepekatan butiran
        \qty{0.2}{\micro m} berkurang separuh tiap $\sim\qty{35}{\micro m}$
        ketinggian; dibaca dengan $n(h) = n_0\eu^{-m'gh/k_{\text{B}}T}$,
        tangga itu menghasilkan $N_{\text{A}} \approx \num{6e23}$ ---
        atom yang dicacah, bukan diterka, dalam setetes air di atas meja
        mikroskop.
\end{enumerate}
\end{problem}
