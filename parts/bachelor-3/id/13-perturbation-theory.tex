\chapter{Teori Gangguan}\label{ch:b3:perturbation-theory}

Hanya segelintir soal kuantum yang larut secara eksak: kotak, osilator,
hidrogen, dan sistem dua aras. Selebihnya --- sebuah atom dalam medan,
ikatan sebuah molekul yang mengeras, sebuah aras yang disenggol
tetangganya --- adalah salah satu kerangka terselesaikan itu yang
mengenakan satu suku tambahan yang kecil. \omterm{thm:b3:perturbation-theory:corrections}{Teori gangguan} adalah seni
memperlakukan suku tambahan itu sebagaimana adanya: sebuah koreksi, yang
dihitung orde demi orde dalam kekecilannya. Kaidah orde pertamanya
adalah sebuah rata-rata satu baris; kaidah orde keduanya menjelaskan
mengapa aras saling \emph{menolak}, mengapa tiap atom dapat
terpolarisasi, dan mengapa tarikan van der Waals bersifat semesta;
ragam berdegenerasinya menangani kasus peka tempat aras yang nyaris sama
tertata ulang seluruhnya; dan bentuk bergantung waktunya --- yakni
kaidah emas Fermi --- adalah persamaan di balik setiap garis serapan,
laju peluruhan, dan penampang lintang hamburan yang diukur di
laboratorium. Sebagai ganjaran yang mengalir, bab ini berakhir dengan
menghitung warna langit.

\section{Pergeseran kecil: teori tak berdegenerasi}

\begin{theorem}[Koreksi perturbatif]\label{thm:b3:perturbation-theory:corrections}
\index{teori gangguan}
Misalkan $\hat H = \hat H_0 + \hat V$ dengan $\hat H_0$ sudah
terselesaikan ($\hat H_0\ket n = E_n\ket n$, spektrumnya tak
berdegenerasi) dan $\hat V$ kecil. Maka, orde demi orde dalam $\hat V$,
\[
E_n' = E_n + \bra n\hat V\ket n
 + \sum_{m \neq n}\frac{|\bra m\hat V\ket n|^2}{E_n - E_m}
 + \cdots
\]
dan keadaannya memperoleh campuran $\ket{n'} = \ket n + \sum_{m\neq
n}\dfrac{\bra m\hat V\ket n}{E_n - E_m}\,\ket m + \cdots$
\emph{Orde pertama}: pergeserannya hanyalah rata-rata gangguannya dalam
keadaan yang belum terganggu. \emph{Orde kedua}: kopling ke aras lain
mendorong $E_n$ \emph{menjauhi} aras itu --- tiap $m$ di atasnya
mendorong ke bawah, tiap yang di bawahnya mendorong ke atas --- sehingga
pergeseran orde kedua keadaan dasarnya selalu negatif. Keabsahannya:
bagian percampurannya $|\bra m\hat V\ket n/(E_n - E_m)|$ haruslah kecil.
\end{theorem}

\begin{proof}[Bukti sebagian]
Jabarkanlah $E' = E + \epsilon_1 + \epsilon_2 + \cdots$ dan $\ket{\psi} =
\ket n + \ket{\psi_1} + \cdots$ dalam pangkat $\hat V$, masukkanlah ke
$(\hat H_0 + \hat V)\ket\psi = E'\ket\psi$, lalu cocokkanlah ordenya.
Memproyeksikan persamaan orde pertamanya pada $\bra n$ memberikan $\epsilon_1 =
\bra n\hat V\ket n$; memproyeksikannya pada $\bra m$
($m \neq n$) memberikan koefisien campurannya; membawa koefisien itu ke
persamaan orde keduanya lalu memproyeksikannya pada $\bra n$ lagi
memberikan $\epsilon_2$. Pernyataan tandanya bagi keadaan dasar: tiap
penyebut $E_0 - E_m$ negatif sedangkan pembilangnya kuadrat.
\end{proof}

\begin{proposition}[Sebuah gangguan yang berjawaban eksak]\label{prop:b3:perturbation-theory:oscillator-field}
Sebuah osilator bermuatan $q$ dalam medan seragam $\mathcal E$
mempunyai $\hat V = -q\mathcal E\hat x$. Melengkapkan kuadratnya
menyelesaikannya secara eksak: potensialnya tetap parabola yang sama,
hanya tergeser, dengan semua arasnya turun sebesar
$q^2\mathcal E^2/2m\omega^2$. \omterm{thm:b3:perturbation-theory:corrections}{Teori gangguan} harus sepakat --- dan
memang sepakat: orde pertamanya lenyap ($\langle\hat
x\rangle_n = 0$), dan jumlahan orde keduanya, yang di dalamnya $\hat x$
hanya mengopel $n$ ke $n \pm 1$, memberikan tepat
$-q^2\mathcal E^2/2m\omega^2$ untuk tiap arasnya
(\cref{exo:b3:perturbation-theory:2}). Satu kasus terselesaikan, yang
diperiksa lewat kedua jalan, adalah kalibrasi terbaik yang dapat
dimiliki sebuah metode.
\end{proposition}

\begin{proof}
Kedua unsur matriksnya beserta penyebutnya memberikan
\[
\frac{|\bra{n+1}\hat x\ket{n}|^2}{-\hbar\omega}
 + \frac{|\bra{n-1}\hat x\ket n|^2}{+\hbar\omega}
 = \frac{\hbar}{2m\omega}\,\frac{-(n+1) + n}{\hbar\omega}
 = -\frac{1}{2m\omega^2} ;
\]
kalikanlah dengan $q^2\mathcal E^2$.
\end{proof}

\section{Ketika arasnya berdegenerasi}

\begin{theorem}[Teori gangguan berdegenerasi]\label{thm:b3:perturbation-theory:degenerate}
\index{teori gangguan berdegenerasi}
Jika $E_n$ berdegenerasi, rumus di atas membagi dengan nol ---
gangguannya boleh jadi menata ulang keadaan berdegenerasinya
seluruhnya, dan penawarnya adalah membiarkannya berbuat demikian:
\emph{diagonalkanlah $\hat V$ yang dibatasi pada subruang
berdegenerasinya}. \omterm{def:b3:quantum-formalism:observable}{Nilai eigennya} adalah pergeseran orde pertamanya;
vektor eigennya adalah keadaan orde nol yang benar, yakni gabungan yang
dipilih gangguannya sendiri.
\end{theorem}
\admitted

\begin{example}[Efek Stark linear]\label{ex:b3:perturbation-theory:stark}
\index{efek Stark}
Aras $n = 2$ hidrogen memuat keadaan $2s$ dan $2p$ yang berdegenerasi.
Dalam medan $\mathcal E\vect e_z$, $\hat V = e\mathcal E\hat z$
mengopel $2s$ ke $2p_0$ dengan $\bra{2s}\hat z\ket{2p_0} = -3a_0$:
mendiagonalkan blok $2\times2$-nya memberikan pergeseran
\[
\Delta E = \pm\,3e a_0\mathcal E ,
\]
yang \emph{linear} dalam medannya --- sedangkan keadaan dasarnya yang
tak berdegenerasi hanya bergeser secara kuadratik. Pada
$\mathcal E = \qty{e7}{V/m}$: $\pm
\qty{1.6}{meV}$, yakni pembelahan yang mudah dipilah. Keadaan yang
melakukan pembelahannya adalah $(\ket{2s} \mp \ket{2p_0})/\sqrt2$ ---
yakni hibrida berdipol listrik tetap, yang hanya mungkin karena
degenerasinya membiarkan atomnya bebas terpolarisasi pada orde nol.
(Hibridisasi $s$--$p$ yang sama, yang digerakkan tetangga alih-alih
medan, adalah kimia karbon tiap molekul organik.)
\end{example}

\section{Transisi: kaidah emasnya}

\begin{theorem}[Gangguan bergantung waktu]\label{thm:b3:perturbation-theory:golden}
\index{kaidah emas Fermi}
Nyalakanlah $\hat V(t) = \hat V\cos\omega t$ pada $t = 0$. Sampai orde
pertama, peluang menemukan sistemnya dalam $\ket f$ pada waktu $t$,
setelah bermula dalam $\ket i$, adalah
\[
\mathcal P_{i\to f}(t) = \frac{|\bra f\hat V\ket i|^2}{4\hbar^2}\;
\frac{\sin^2\!\big[(\omega_{fi} - \omega)\,t/2\big]}
{\big[(\omega_{fi} - \omega)/2\big]^2} ,
\qquad \omega_{fi} = \frac{E_f - E_i}{\hbar} :
\]
yakni sebuah resonansi yang memuncak tajam pada
$\hbar\omega = E_f - E_i$ --- serapan ketika $E_f > E_i$, dan
\emph{pancaran terangsang} bagi suku pasangannya dengan $E_f < E_i$.
Ketika keadaan akhirnya membentuk kontinum berkerapatan $\rho(E)$,
pertumbuhan puncaknya menjadi sebuah \emph{laju} yang tetap:
\[
\Gamma_{i\to f} = \frac{2\pi}{\hbar}\,
|\bra f\hat V\ket i|^2\,\rho(E_f)
\]
--- yakni \emph{kaidah emas Fermi}, rumus induk laju peluruhan,
koefisien serapan dan penampang lintang.
\end{theorem}

\begin{proof}[Bukti sebagian]
Amplitudo orde pertamanya: $c_f(t) = -\tfrac{\iu}{\hbar}\int_0^t\bra
f\hat V(t')\ket i\,\eu^{\iu\omega_{fi}t'}\dd t'$; mempertahankan
eksponensial yang dekat resonansinya lalu mengintegralkannya memberikan
bentuk $\sin^2$ yang dinyatakan itu. Untuk kontinumnya: sebagai fungsi
pelencengannya $x$, faktor resonansinya bertinggi $t^2$ dan berlebar
$\sim 2\pi/t$ --- sehingga berluas $2\pi t$ --- maka menjumlahkannya
pada kontinum yang mulus menggantikannya dengan $2\pi t\,\rho$:
peluangnya linear dalam $t$, yakni sebuah laju.
\end{proof}

\begin{example}[Apa yang dijalankan kaidah emas]\label{ex:b3:perturbation-theory:applications}
Kaidah pemilihan: sebuah transisi terjadi hanya jika unsur matriksnya
$\bra f\hat V\ket i$ bertahan --- untuk cahaya, $\hat V \propto \hat
z$, yang paritas dan momentum sudutnya membunuh segalanya kecuali
$\Delta l = \pm1$, $\Delta m = 0, \pm1$: yakni kaidah yang dipanggil
sejak \cref{ch:b3:quantum-angular-momentum}, yang kini menjadi teorema.
Lajunya: peluruhan spontan keadaan $2p$ hidrogen terhitung
\qty{1.6}{ns} (\cref{exo:b3:perturbation-theory:8}); transisi maser
amonia, dengan $\omega^3$ gelombang mikronya, memakan waktu berbulan
--- $\omega^3$ dalam laju radiasinya itulah sebabnya garis radio
(\cref{ex:b3:spin-two-level:hyperfine}) hidup berjuta tahun sedangkan
garis ultraungu berkelebat dalam nanosekon. Spektrum serapan,
pengionan foto, peluruhan beta inti
(\cref{ch:b3:nuclear-physics}), penampang lintang hamburan
(\cref{ch:b3:scattering-theory}): semuanya adalah satu rumus ini dengan
unsur matriks dan cacah keadaan yang berbeda.
\end{example}

\begin{method}[Memilih perkakas perturbatif yang tepat]\label{met:b3:perturbation-theory:method}
(1) Pertanyaan statis, aras tak berdegenerasi: rata-ratakanlah $\hat V$
(orde pertama); jika hasilnya lenyap karena simetrinya, pakailah orde
kedua --- harapkanlah tolakan aras. (2) Aras berdegenerasi:
diagonalkanlah $\hat V$ dalam blok berdegenerasinya lebih dahulu;
gabungan yang selaras simetrinyalah yang melakukan pembelahannya. (3)
Laju transisi: kaidah emas --- satu unsur matriks, satu \omterm{prop:b3:schrodinger-three-dimensions:counting}{kerapatan
keadaan}; periksalah kaidah pemilihannya sebelum menghitung apa pun. (4)
Ujilah selalu penjabarannya: bagian percampurannya
$|V_{mn}/(E_n - E_m)| \ll 1$, kalau tidak diagonalkanlah secara eksak
(rumus dua arasnya). (5) Limit terselesaikan yang eksak (osilator dalam
medan) adalah kalibrasi cuma-cuma: pakailah.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:1}
Sebuah partikel dalam kotak $[0, a]$ merasakan potensial tambahan $V(x) =
\lambda x/a$. (a) Hitunglah pergeseran orde pertama tiap arasnya
(pakailah $\langle x\rangle = a/2$ dalam keadaan kotak mana pun ---
berilah alasannya). (b) Mengapa pergeserannya sama untuk semua arasnya
di sini? (c) Berapa perubahan orde pertama \emph{jarak arasnya}, dan
mengapa pengukuran frekuensi transisi dapat sama sekali melewatkan
gangguan ini? (d) Kapan orde pertamanya berhenti dapat dipercaya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:2}
Kerjakanlah perhitungan orde kedua pada
\cref{prop:b3:perturbation-theory:oscillator-field} selengkapnya: unsur
matriks $\hat x$, kedua sukunya, peniadaan $n$-nya, dan kesesuaian
eksaknya dengan melengkapkan kuadratnya. Mengapa koreksi orde pertama
\emph{keadaannya} tidak lenyap walaupun koreksi energinya lenyap?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:3}
\omterm{def:b3:hamiltonian-mechanics:hamiltonian}{Hamiltonian} dua aras $\begin{pmatrix}E_1 & v\\ v &
E_2\end{pmatrix}$, $E_1 < E_2$, dengan $v$ real. (a) Carilah \omterm{def:b3:quantum-formalism:observable}{nilai
eigen} eksaknya. (b) Jabarkanlah untuk $|v| \ll E_2 - E_1$ lalu kenalilah
rumus orde keduanya. (c) Tunjukkanlah bahwa arasnya saling menolak:
celahnya selalu melampaui $E_2 - E_1$. (d) Pada $E_1 = E_2$: apa yang
diberikan \omterm{thm:b3:perturbation-theory:corrections}{teori gangguan}, apa yang diberikan jawaban eksaknya, dan
teorema mana dalam bab ini yang mendamaikan keduanya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:perturbation-theory:4}
Untuk \cref{thm:b3:perturbation-theory:corrections}: (a) tunjukkanlah
bahwa koreksi keadaan orde pertamanya ortogonal terhadap $\ket n$; (b)
tunjukkanlah bahwa koefisien campuran $\ket m$ adalah
$V_{mn}/(E_n - E_m)$ lalu nyatakanlah patokan keabsahannya; (c) sebuah
aras yang berjarak $\qty{1}{eV}$ dari tetangga terdekatnya terkopel
dengannya sebesar \qty{0.1}{eV}: perkirakanlah peluang campurannya; (d)
hal yang sama dengan celah \qty{10}{meV} --- perkakas mana yang
seharusnya menggantikan penjabarannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:5}
Ikatan yang anharmonik. Tambahkanlah $\hat V = \beta\hat x^3 + \gamma\hat x^4$
pada osilatornya. (a) Tunjukkanlah bahwa suku $x^3$-nya tidak menggeser
aras mana pun pada orde pertama (paritas). (b) Tunjukkanlah bahwa suku
$x^4$-nya memberikan $\Delta E_n =
3\gamma(\hbar/2m\omega)^2(2n^2 + 2n + 1)$ (jabarkanlah $\hat x^4$ dalam
\omterm{def:b3:harmonic-oscillator:ladder}{operator tangga}; hanya suku yang ``berimbang'' yang bertahan). (c) Untuk ikatan sungguhan $\gamma$ efektifnya negatif (sumurnya
melunak ke arah luar): tunjukkanlah bahwa jarak arasnya lalu
\emph{menyusut} seiring $n$.
(d) Ciri teramati manakah dari spektrum molekul
(\cref{exo:b3:harmonic-oscillator:4}) yang dihasilkannya kembali?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:6}
Dalam kotak persegi berdimensi dua, pasangan berdegenerasi $\ket{1,2},
\ket{2,1}$ diganggu oleh $\hat V = \lambda\,xy$. (a) Jelaskanlah mengapa
teori tak berdegenerasinya gagal. (b) Hitunglah matriks $2\times2$
$\hat V$ dalam pasangan itu (integralnya terfaktorkan; $\int_0^a
x\sin(\pi x/a)\sin(2\pi x/a)\dd x = -8a^2/9\pi^2$). (c) Carilah
pembelahannya beserta keadaan orde nolnya yang benar. (d)
Bandingkanlah dengan analisis simetri pada
\cref{exo:b3:quantum-formalism:9}: jawaban mana yang diberikan
simetrinya secara cuma-cuma?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:7}
\omterm{ex:b3:perturbation-theory:stark}{Efek Stark}, linear dan kuadratik. (a) Dengan memakai
\cref{ex:b3:perturbation-theory:stark}, hitunglah pembelahan $n = 2$
pada $\mathcal E = \qty{2.5e6}{V/m}$. (b) Mengapa keadaan dasarnya
\emph{tidak} menunjukkan pergeseran linear (dua alasan: paritasnya, dan
tak adanya pasangan berdegenerasi)? (c) Pergeseran kuadratiknya
mendefinisikan keterpolarisasiannya, $\Delta E = -\tfrac12\alpha\mathcal E^2$:
perkirakanlah $\alpha \sim e^2a_0^2/E_{\text{I}}$ lalu hitunglah dalam
satuan $a_0^3$ (jawaban eksaknya adalah $4.5\,a_0^3 \times
4\pi\varepsilon_0$). (d) Sifat materi sehari-hari yang mana --- yakni
tanggapan dielektriknya --- yang benih atomnya baru saja Anda hitung?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:8}
Pancaran spontan (dengan satu masukan yang diterima tanpa bukti): sebuah
keadaan tereksitasi meluruh dengan $A = \omega^3|d_{fi}|^2/3\pi\varepsilon_0\hbar c^3$,
dengan unsur matriks jenis $d_{fi} = e\bra f\hat z\ket i$ menetapkan
dipolnya. (a) Untuk hidrogen $2p \to 1s$: dengan $|d| \approx 0.74\,
ea_0$ dan $\lambda = \qty{121.6}{nm}$, hitunglah $A$ beserta umurnya.
(b) Skalakanlah ke garis D natrium. (c) Skalakanlah ke transisi
hiperhalus 21 cm ($|d| \to$ sebuah momen magnetik, dengan lajunya turun
lagi sebesar faktor sekitar $\alpha^2$; terimalah $A \approx
\qty{2.9e-15}{s^{-1}}$): periksalah ``sepuluh juta tahun''-nya. (d)
Dari $\omega^3$-nya: mengapa garis ultraungu cepat, garis radio abadi,
dan mengapa hal itu membuat \omterm{ex:b3:spin-two-level:hyperfine}{garis 21 cm} \emph{dapat diramalkan} namun
nyaris tak terdeteksi di laboratorium?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:perturbation-theory:9}
Kaidah pemilihan sebagai integral. (a) Tunjukkanlah $\bra{100}\hat z\ket{200} =
0$ lewat paritasnya. (b) Tunjukkanlah $\bra{100}\hat z\ket{21m} = 0$
kecuali bila $m =
0$ (integral $\varphi$-nya). (c) Nyatakanlah kaidah
yang dihasilkannya $\Delta
l = \pm1$, $\Delta m = 0, \pm1$ beserta pembacaan fisisnya (spin
fotonnya). (d) Keadaan $2s$ tak dapat mencapai $1s$ lewat jalur dipol
mana pun: apa yang diramalkannya bagi umurnya, dan jalur ``terlarang''
dua foton apa yang sesungguhnya ditempuh alam?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:10}
Keterpolarisasian, secara jujur. Pergeseran orde kedua keadaan dasar
hidrogen dalam sebuah medan adalah $\Delta E = -e^2\mathcal E^2\sum_{m\neq
0}|\bra m\hat z\ket 0|^2/(E_m - E_0)$. (a) Batasilah jumlahannya dengan
mengganti tiap penyebutnya dengan celah terkecilnya $E_2 - E_1 =
\tfrac34 E_{\text{I}}$ dan memakai hubungan ketertutupannya $\sum_m|\bra
m\hat z\ket0|^2 = \langle z^2\rangle_0 = a_0^2$: perolehlah $\alpha
\le \tfrac{16}3\,a_0^3$ (dalam satuan bergaya Gauss
$4\pi\varepsilon_0$). (b) Bandingkanlah dengan nilai eksaknya $\tfrac92 a_0^3$.
(c) Dari $\alpha$, perkirakanlah indeks bias gas hidrogen
pada kerapatan atmosfer ($n - 1 \approx N\alpha/2\varepsilon_0
\cdot e^2\dots$ --- pakailah $n - 1 = N\alpha_{\text{SI}}/2
\varepsilon_0$) lalu bandingkanlah dengan nilai terukur $\num{1.3e-4}$.
(d) Nyatakanlah dalam satu kalimat rantai atom $\to$ keterpolarisasian
$\to$ pembiasan yang akan ditunggangi Bagian II soal akhir pekannya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:11}
Perpotongan terhindar dan sapuan lambat. Sebuah sistem dua aras
mempunyai energi telanjang $\pm\lambda t$ yang berpotongan pada
$t = 0$, terkopel oleh $v$. (a)
Sketsalah aras eksaknya terhadap $t$: sebuah perpotongan terhindar
bercelah $2|v|$. (b) Jika sapuannya \emph{lambat}, sistemnya mengikuti
kurva bawahnya (teorema adiabatik, diterima tanpa bukti): keadaan apa
yang akhirnya ditempatinya? (c) Patokan Landau--Zener membandingkan
waktu sapuan melintasi daerah celahnya, $\sim v/\lambda$, dengan jam
internalnya $\hbar/v$: berikanlah syarat pengikutan adiabatiknya. (d)
Dua kegunaannya: keadaan sebuah kubit yang dipindahkan lewat kicauan
lambat; sebuah tumbukan atom yang melompatkan muatan antarinti ---
jelaskanlah salah satunya dalam dua kalimat.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:perturbation-theory:12}
Orde kedua itu pesimis, beserta teorema lainnya. (a) Buktikanlah bahwa
koreksi orde kedua bagi keadaan \emph{dasar} selalu negatif. (b)
Simpulkanlah (dengan \cref{exo:b3:harmonic-oscillator:10}) mengapa gaya
van der Waals selalu bersifat menarik antara atom dalam keadaan dasar.
(c) Tunjukkanlah bahwa teori orde pertama \emph{menaksir terlalu besar}
tiap energi keadaan dasar: ia merupakan batas variasi (hitunglah $\bra{\psi_0}\hat
H\ket{\psi_0}$ dengan keadaan tak terganggunya sebagai
fungsi coba). (d) Sebuah pemeriksaan taat asas atas
\cref{ex:b3:perturbation-theory:stark}: mengapa \omterm{ex:b3:perturbation-theory:stark}{efek Stark} \emph{linear}
bukan pelanggaran atas (a)?
\end{exercise}

\section{Soal: Warna langit}

\begin{problem}[{Soal akhir pekan --- hamburan Rayleigh dari atom yang
terganggu}]\label{pb:b3:perturbation-theory:1}
Mengapa langit siang berwarna biru, senjanya merah, dan awannya putih?
Jawaban lengkapnya adalah sebuah rantai lewat bab ini: gelombang cahaya
mengganggu tiap molekul udara; dipol imbasannya memancar ulang (yakni
radiasi dipol jilid Tahun 2); dan interferensinya memutuskan apa yang
bertahan. Data: keterpolarisasian molekul udara
$\alpha/4\pi\varepsilon_0 \approx \qty{2.2e-30}{m^3}$; kerapatan
cacahnya $N = \qty{2.5e25}{m^{-3}}$; rentang tampaknya
$450$--\qty{700}{nm}.

\medskip\noindent\textbf{Bagian I --- Molekul yang terganggu.}
\begin{enumerate}
  \item Sebuah medan statis $\mathcal E$ menggeser keadaan dasar sebuah
        molekul sebesar $-\tfrac12\alpha\mathcal E^2$: kaitkanlah
        pernyataan ini dengan \omterm{thm:b3:perturbation-theory:corrections}{teori gangguan} orde kedua (rumus yang
        mana, dan mengapa negatif?).
  \item $\alpha$ yang sama memberikan dipol imbasannya $p =
        \alpha\mathcal E$: untuk medan sinar matahari $\mathcal E \sim
        \qty{800}{V/m}$, hitunglah dipol imbasan satu molekul nitrogen
        lalu bandingkanlah dengan $ea_0$.
  \item Mengapa medan optisnya (berfrekuensi $\num{5e14}$ Hz) boleh
        diperlakukan dengan keterpolarisasian \emph{statis} bagi
        molekul yang transisi elektroniknya terletak di ultraungu?
        (Bandingkanlah $\hbar\omega$ dengan celahnya; inilah limit jauh
        di luar resonansi bagi penyebut kaidah emasnya.)
  \item Sebuah dipol berosilasi $p_0\cos\omega t$ memancarkan daya
        rata-rata $P = p_0^2\omega^4/12\pi\varepsilon_0c^3$
        (jilid Tahun 2). Digabungkan dengan $p_0 = \alpha\mathcal
        E_0$: bagaimana daya hamburannya bergantung pada $\omega$ pada
        penyinaran yang tetap?
  \item Hitunglah nisbah daya hamburannya pada \qty{450}{nm}
        dan \qty{700}{nm}.
  \item Dalam satu kalimat: mengapa langitnya biru dan bukan ungu
        (dua bahan: spektrum Matahari yang menurun ke arah ungu, dan
        kepekaan mata)?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Dari satu molekul ke
langit.}
\begin{enumerate}[resume]
  \item Penampang lintang hamburannya diperoleh sebagai $\sigma =
        \dfrac{8\pi}{3}\Big(\dfrac{\alpha}{4\pi\varepsilon_0}
        \Big)^2\dfrac{\omega^4}{c^4}$: periksalah dimensinya lalu
        hitunglah pada \qty{550}{nm}.
  \item Panjang pelemahannya adalah $\ell = 1/N\sigma$: hitunglah
        pada \qty{550}{nm} lalu bandingkanlah dengan tebal
        atmosfernya ($\sim\qty{8}{km}$ setara pada kerapatan
        permukaan laut).
  \item Pada \qty{450}{nm} berbanding \qty{700}{nm}, berapa bagian
        seberkas sinar matahari tegak yang terhamburkan keluar?
        (Pakailah $1 -
        \eu^{-L/\ell}$ dengan $L = \qty{8}{km}$.)
  \item Pada senja lintasannya tiga puluh kali lebih panjang:
        hitunglah kembali kebertahanan birunya lalu jelaskanlah warna
        Matahari yang rendah --- dan warna cahaya yang memerahkan Bulan
        dalam gerhana bulan.
  \item Mengapa cahaya langit \emph{terpolarisasi} pada $90^\circ$
        dari Matahari (ingatlah pola radiasi dipolnya: tak ada
        pancaran sepanjang sumbu dipolnya)?
  \item Lebah dan orang Viking konon bernavigasi dengan polarisasi
        ini: apa yang dilakukan hari yang mendung terhadapnya, dan
        mengapa?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Mengapa langitnya tidak lebih
terang.}
\begin{enumerate}[resume]
  \item Sebuah paradoks: dalam medium yang benar-benar seragam, riak
        gelombang dari unsur volume yang bertetangga berinterferensi
        merusak ke arah samping, sehingga \emph{tak ada} cahaya yang
        akan terhambur sama sekali (hanya pembiasan yang bertahan). Apa
        yang sesungguhnya merusak keseragaman udara --- apa yang
        tersebar secara acak?
  \item Fluktuasi kerapatan sebuah gas ideal membuat keterangan
        hamburan molekul yang saling bebas \emph{dijumlahkan}:
        nyatakanlah mengapa kebebasannya (kedudukan acak, jarak
        berskala panjang gelombang) membunuh interferensi merusaknya.
  \item Dari $\ell$ Anda pada \qty{550}{nm}: berapa bagian sinar
        matahari yang dihamburkan satu atmosfer di hari cerah ---
        apakah bilangannya cocok dengan keterangan langit sehari-hari
        dibandingkan Matahari?
  \item Sebutir titik awan (\qty{10}{\micro m}) memuat
        $\sim10^{12}$ molekul dalam satu panjang gelombang: riak
        gelombangnya berjumlah secara \emph{terpadu}. Bagaimana daya
        hamburannya lalu berskala terhadap cacah molekulnya, dan
        mengapa pemilihan warna $\omega^4$-nya lenyap (semua panjang
        gelombangnya terhambur kuat): berwarna apakah awannya?
  \item Susu, kabut dan cat putih berwarna putih karena alasan yang
        sama: nyatakanlah kaidah umumnya --- penghambur yang kecil
        terhadap $\lambda$ mewarnai cahayanya, penghambur yang besar
        terhadap $\lambda$ memutihkannya.
  \item Opalesensi kritis: di dekat titik kritis sebuah zat cair
        (bab perubahan fase jilid Tahun 1, dan
        \cref{ch:b3:phase-transitions} di depan), fluktuasi kerapatannya tumbuh sampai skala optis dan
        fluidanya menjadi seperti susu: kaitkanlah dengan butir 13.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Rantainya, dilengkapi.}
\begin{enumerate}[resume]
  \item Riak gelombang yang terhambur ke depan \emph{tidak} saling
        meniadakan: ia berinterferensi dengan berkasnya lalu
        memperlambat fasenya --- yakni indeks biasnya. Dengan memakai
        $n - 1 = N\alpha/2\varepsilon_0$ (dengan $\alpha$ dalam SI),
        hitunglah $n - 1$ untuk udara lalu bandingkanlah dengan nilai
        terukur \num{2.9e-4}.
  \item Satu tetapan $\alpha$, yang dihitung dengan \omterm{thm:b3:perturbation-theory:corrections}{teori gangguan} orde
        kedua, dengan demikian menetapkan tiga gejala sekaligus:
        sebutkanlah ketiganya (pergeseran dalam medan statis; birunya
        langit; pembelokan cahaya dalam udara).
  \item Serapan ozon, aerosol dan hamburan berganda merumitkan langit
        yang sebenarnya: sebutkanlah satu hal teramati bagi
        masing-masingnya (warna senja; cakrawala yang seperti susu;
        keterangan sisa langit di zenit setelah matahari terbenam).
  \item Mengapa Bulan, yang tanpa udara, mempunyai langit siang yang
        \emph{hitam} --- dan apa yang karenanya dibenarkan tiap foto
        dari permukaannya tentang asal usul langit kita?
  \item Langit siang Mars berwarna karamel dan senjanya \emph{biru}:
        udaranya yang tipis menghamburkan sedikit, dan butiran debu
        mikrometer yang melayanglah yang melakukan pekerjaannya. Dengan
        memakai kaidah butir 18, jelaskanlah bagaimana debunya
        membalikkan logika warnanya.
  \item Untuk sinar-X, $\hbar\omega$ jauh melampaui tiap transisi
        molekul: elektronnya menanggapi seolah-olah bebas dan hamburannya
        (Thomson) kehilangan $\omega^4$-nya. Apa yang terjadi pada
        mekanisme ``biru langit'' dalam rezim itu --- dan mengapa
        tanggapan yang rata itu justru yang menjadikan sinar-X
        penyelidik kerapatan elektron yang bersih?
  \item Ringkaslah hasil bernama itu: keterpolarisasian
        $\qty{2.2e-30}{m^3}$, yang dikuadratkan lalu dikalikan
        $\omega^4$, memberikan panjang hamburan berskala \qty{60}{km}
        pada warna hijau --- cukup panjang sehingga langit siangnya
        lembut, cukup pendek sehingga senjanya memerah dan langitnya
        biru, dengan pola polarisasi sebuah hutan dipol yang digerakkan.
\end{enumerate}
\end{problem}
