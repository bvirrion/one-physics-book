\chapter{Ensembel Kanonik Besar}\label{ch:b3:grand-canonical}

Setetes air dalam udara lembap tidak tumbuh dan tidak pula menyusut;
oksigen terikat pada hemoglobin di paru-paru lalu terlepas di otot;
sebuah baterai menggiring elektron menembus telepon karena elektronnya
``lebih mahal'' di satu elektroda daripada di elektroda lainnya. Dalam
tiap perkara itu, mata uang yang disetimbangkan bukanlah suhu melainkan
\emph{\omterm{thm:b3:grand-canonical:distribution}{potensial kimia}} $\mu$: yakni ongkos \omterm{prop:b3:canonical-ensemble:machine}{energi bebas} untuk satu
partikel tambahan. Bab ini membuka katup tandon yang kedua --- partikel
di samping energi --- lalu membangun \omterm{thm:b3:grand-canonical:distribution}{ensembel kanonik besar}, yang
imbalannya adalah kehematan yang menakjubkan: keterisian satu aras
kuantum tunggal, fermion maupun boson, menyusul dalam dua baris,
demikian pula rumus induk bab berikutnya. Penerapannya terbentang dari
isoterm penjerapan pada masker gas dan pengubah katalitik sampai ke
\omterm{prop:b3:grand-canonical:saha}{persamaan Saha}, yang dengannya seorang mahasiswa doktoral Harvard pada
1925 membaca suhu bintang lalu menemukan, di luar segala dugaan, bahwa
alam semesta terbuat dari hidrogen.

\section{Partikel ikut berunding}

\begin{theorem}[Sebaran kanonik besar]\label{thm:b3:grand-canonical:distribution}
\index{ensembel kanonik besar}\index{potensial kimia}
Sistem yang bertukar energi \emph{dan partikel} dengan tandon bersuhu
$T$ dan berpotensial kimia $\mu$ menempati keadaannya $s$
(berenergi $E_s$, bercacah partikel $N_s$) dengan peluang
\[
\mathcal P_s = \frac{\eu^{-\beta(E_s - \mu N_s)}}{\Xi} , \qquad
\Xi = \sum_s\eu^{-\beta(E_s - \mu N_s)} ,
\]
yakni \emph{\omterm{thm:b3:canonical-ensemble:boltzmann}{fungsi partisi} besarnya}. Dari $\Xi$: $\langle N\rangle
= k_{\text{B}}T\,\partial\ln\Xi/\partial\mu$, dan \emph{potensial
besarnya} $\Phi = -k_{\text{B}}T\ln\Xi$ menaati $\Phi = -PV$.
\end{theorem}

\begin{proof}[Bukti sebagian]
Seperti dalam \cref{thm:b3:canonical-ensemble:boltzmann}, jabarkanlah
entropi tandonnya terhadap kedua penarikannya:
$S_{\text{res}}(E - E_s, N - N_s) \approx S_{\text{res}} - E_s/T +
\mu N_s/T$, dengan memakai $\partial S/\partial N = -\mu/T$
(\cref{thm:b3:microcanonical-ensemble:temperature}). Nilai rata-ratanya
menyusul lewat penurunan; $\Phi = -PV$ diterima tanpa bukti (ia
menyusul dari keekstensifannya).
\end{proof}

\begin{definition}[Apa arti $\mu$]\label{def:b3:grand-canonical:mu}
\index{potensial kimia!maknanya}
$\mu$ adalah ongkos, dalam \omterm{prop:b3:canonical-ensemble:machine}{energi bebas}, untuk menambahkan satu partikel
pada $T$ dan $V$ tetap. Partikel mengalir secara spontan dari $\mu$
tinggi ke $\mu$ rendah --- difusi, osmosis, penguapan, reaksi kimia dan
arus listrik semuanya merupakan gradien $\mu$ yang mengendur. Pada
kesetimbangan antara dua fase atau dua tempat mana pun, $\mu$ sama pada
kedua sisinya --- yakni kesamaan ketiga (di samping $T$ dan $P$) yang
memerintah kebersamaan fase. Untuk gas ideal klasik,
\[
\mu = k_{\text{B}}T\,\ln\!\big(n\lambda_T^3\big) < 0
\quad\text{(gas encer)} :
\]
menambahkan sebutir partikel pada gas encer \emph{menurunkan} energi
bebasnya, karena perolehan entropinya mengalahkan ongkos energi apa pun
--- yakni tanda yang menggerakkan penguapan ke dalam udara kering. Dan
sebuah baterai adalah mesin $\mu$: tegangannya adalah jatuhnya \omterm{thm:b3:grand-canonical:distribution}{potensial
kimia} tiap elektron di antara kedua elektrodanya,
$U = \Delta\mu/e$ --- volt adalah nilai tukar kimia elektron, dan
karena itulah sel baterai memberikan tegangan berangka satuan.
\end{definition}

\section{Satu aras pada satu waktu: kiat induknya}

\begin{theorem}[Keterisian satu aras kuantum]\label{thm:b3:grand-canonical:occupation}
\index{sebaran Fermi--Dirac}\index{sebaran Bose--Einstein}
Ambillah sebagai ``sistemnya'' satu aras satu partikel berenergi
$\epsilon$, dalam sentuhan besar dengan gas semua aras lainnya.
\emph{Fermion} (keterisian $0$ atau $1$):
\[
\Xi = 1 + \eu^{-\beta(\epsilon - \mu)}
\;\Longrightarrow\;
\bar n_{\text{Fermi}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} + 1} .
\]
\emph{Boson} (keterisian $0, 1, 2, \dots$):
\[
\Xi = \frac{1}{1 - \eu^{-\beta(\epsilon - \mu)}}
\;\Longrightarrow\;
\bar n_{\text{Bose}} = \frac{1}{\eu^{\beta(\epsilon - \mu)} - 1} ,
\]
yang menuntut $\mu < \epsilon_{\min}$. Keduanya menyusut menjadi faktor
Boltzmann $\eu^{-\beta(\epsilon - \mu)}$ ketika keterisiannya jarang.
Kedua rumus itu, yang masing-masing diperoleh dalam dua baris, adalah
seluruh masukan statistika kuantum: bab berikutnya adalah panennya.
\end{theorem}

\begin{proof}
Fermion: dua suku. Boson: sebuah deret geometri, yang hanya menyatu
untuk $\mu < \epsilon$. Pada kedua perkaranya $\bar n = k_{\text{B}}T\,
\partial\ln\Xi/\partial\mu$ memberikan bentuk yang dinyatakan itu.
\end{proof}

\section{Penjerapan: tapak yang disewakan}

\begin{proposition}[Isoterm Langmuir]\label{prop:b3:grand-canonical:langmuir}
\index{isoterm Langmuir}\index{penjerapan}
Sebuah permukaan menawarkan tapak yang saling bebas, masing-masing
kosong atau menampung satu molekul gas berenergi ikat $-\epsilon_0$;
gas di atasnya, pada tekanan $P$, adalah tandonnya. Tiap tapaknya
merupakan sistem besar dua keadaan, dan keterisiannya adalah
\[
\theta = \frac{1}{\eu^{\beta(-\epsilon_0 - \mu)} + 1}
 = \frac{P}{P + P_0(T)} , \qquad
P_0(T) = \frac{k_{\text{B}}T}{\lambda_T^3}\,
\eu^{-\epsilon_0/k_{\text{B}}T} ,
\]
dengan memakai $\mu$ gas idealnya. Liputannya menaik secara lurus pada
tekanan rendah lalu menjenuh pada satu lapis tunggal --- yakni
\emph{\omterm{prop:b3:grand-canonical:langmuir}{isoterm Langmuir}}, kurva kerja pengubah katalitik, arang aktif,
masker gas dan pengikatan oksigen oleh darah
(\cref{exo:b3:grand-canonical:6}). Tekanan perpindahannya $P_0$ turun
secara \emph{eksponensial} bersama energi ikatnya lalu menaik curam
bersama suhunya: memanaskan sebuah permukaan mengguncang lepas para
tamunya --- karena itulah memanggang kaca laboratorium dalam hampa
adalah cara membersihkannya yang sesungguhnya.
\end{proposition}

\begin{proof}
Keterisian tapaknya adalah
\cref{thm:b3:grand-canonical:occupation} (bentuk fermioniknya: sebuah
tapak menampung paling banyak satu) dengan $\epsilon = -\epsilon_0$;
gantikanlah $\eu^{\beta\mu} = n\lambda_T^3 = P\lambda_T^3/k_{\text{B}}T$.
\end{proof}

\section{Kesetimbangan pengionan: persamaan Saha}

\begin{proposition}[Persamaan Saha]\label{prop:b3:grand-canonical:saha}
\index{persamaan Saha}
Dalam gas yang panas, reaksi $\text{H} \rightleftharpoons \text{p} +
\text{e}$ menyetimbang ketika $\mu_{\text{H}} = \mu_{\text{p}} +
\mu_{\text{e}}$. Menuliskan tiap $\mu$ dengan bentuk gas idealnya (dengan
atomnya dikreditkan energi ikatnya $-E_{\text{I}}$) memberikan
\[
\frac{n_{\text{e}}\,n_{\text{p}}}{n_{\text{H}}}
 = \frac{1}{\lambda_{T,\text{e}}^3}\,
\eu^{-E_{\text{I}}/k_{\text{B}}T}
\qquad\Big(\lambda_{T,\text{e}} =
h/\sqrt{2\pi m_{\text{e}}k_{\text{B}}T}\Big)
\]
(sampai faktor degenerasi spin berorde satu). Sekali lagi dua mata uang
yang bersaing: faktor Boltzmann mengenakan pajak atas pengionannya
sebesar $E_{\text{I}} = \qty{13.6}{eV}$, sedangkan pemfaktor di depannya
--- yakni \emph{entropi elektron yang dibebaskan}, dengan $\lambda_T^{-3}$
keadaan terjangkau yang dimilikinya --- menyubsidinya. Subsidinya begitu
besar pada kerapatan bintang sehingga hidrogen sudah terion secara berarti
di dekat \qty{10000}{K}, tempat $k_{\text{B}}T$ baru sebesar
\qty{0.9}{eV}: kesetimbangan diperebutkan atas \omterm{prop:b3:canonical-ensemble:machine}{energi bebas}, bukan
energi --- dan spektrum bintang adalah buku besarnya
(\cref{pb:b3:grand-canonical:1}).
\end{proposition}

\begin{proof}[Bukti sebagian]
Tetapkanlah $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$ dengan $\mu_i
= k_{\text{B}}T\ln(n_i\lambda_{T,i}^3) + \varepsilon_i$
($\varepsilon_{\text{H}} = -E_{\text{I}}$, yang lainnya nol); lalu
eksponensialkanlah. Panjang gelombang termal proton dan atomnya nyaris
saling meniadakan (massanya sama); faktor degenerasinya diterima tanpa bukti.
\end{proof}

\begin{method}[Keterampilan kanonik besar]\label{met:b3:grand-canonical:method}
(1) Apa pun yang bertukar partikel dengan mitra yang besar: berikanlah
$\mu$ kepada mitranya lalu bobotkanlah keadaannya dengan
$\eu^{-\beta(E - \mu N)}$.
(2) Tapak atau aras yang saling bebas terfaktorkan: selesaikanlah
\emph{satu}, lalu kalikanlah. (3) Untuk tandon gas klasik, $\eu^{\beta\mu} =
n\lambda_T^3$ mengubah $\mu$ menjadi tekanan yang terukur. (4) Reaksi dan perubahan fase: setimbangkanlah $\mu$-nya, satu
tiap jenisnya, lengkap dengan energi ikatnya --- harapkanlah pemfaktor
entropi menggeser kesetimbangannya jauh dari terkaan Boltzmann yang
naif. (5) Elektron dalam zat padat: $\mu$-nya adalah aras Fermi, dan
selisih $\mu$ adalah tegangan --- yakni jembatan menuju
\cref{ch:b3:electrons-in-solids}.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:1}
(a) Dua kotak berisi gas yang sama pada $T$ yang sama tetapi berbeda
kerapatannya dihubungkan: ke arah mana partikelnya mengalir, dalam
bahasa $\mu$? (b) Tunjukkanlah dari $\mu$ gas idealnya bahwa alirannya
berhenti pada kerapatan yang sama. (c) Air berada di bawah udara lembap:
nyatakanlah syarat kesetimbangannya dalam $\mu$. (d) Mengapa cucian
mengering bahkan di bawah $100\,^\circ$C ketika udaranya kering
(bandingkanlah $\mu$-nya, bukan suhunya)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:2}
\omterm{thm:b3:grand-canonical:distribution}{Potensial kimia} udara. (a) Hitunglah $\lambda_T$ untuk N$_2$
pada \qty{300}{K}. (b) Hitunglah $n\lambda_T^3$ pada \qty{1}{bar} dan
$\mu$ yang dihasilkannya dalam eV. (c) Mengapa $\mu < 0$ bukanlah
paradoks (suku mana dari $F = E - TS$ yang menguasai ketika sebuah
molekul ditambahkan)? (d) Pada kerapatan berapa $\mu$ akan mencapai nol,
dan apa yang dipertandakan oleh
$n\lambda_T^3 \sim 1$
(\cref{exo:b3:grand-canonical:11})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:3}
Turunkanlah keterisian Fermi--Dirac dari $\Xi$ dua sukunya,
langkah demi langkah; periksalah limitnya $\epsilon \ll \mu$, $\epsilon \gg
\mu$, dan nilainya pada $\epsilon = \mu$. Di manakah fungsi ini sudah
pernah muncul dalam buku ini
(\cref{exo:b3:microcanonical-ensemble:8})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:grand-canonical:4}
Volt adalah kimia. (a) Sebuah sel litium menahan \qty{3.0}{V}:
nyatakanlah selisih \omterm{thm:b3:grand-canonical:distribution}{potensial kimia} tiap elektronnya dalam eV dan
tiap molnya dalam kJ. (b) Mengapa semua baterai sehari-hari bertegangan
berangka satuan (apa yang menetapkan skala selisih $\mu$ kimia)? (c)
Sebuah aki mobil timbal--asam menyusun enam sel: mengapa menyusun dan
bukan mencari reaksi ber-\qty{12}{eV}? (d) Dalam bahasa $\mu$, apa yang
sedang dilakukan sebuah baterai ketika ia ``habis''?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:5}
Boson pada sebuah aras. (a) Jumlahkanlah deret geometrinya untuk $\Xi$
lalu turunkanlah $\bar n_{\text{Bose}}$. (b) Mengapa $\mu$ harus tetap di
bawah aras terendahnya --- apa yang menyebar jika tidak, dan apa yang
dipertandakan oleh penyebaran itu (\cref{ch:b3:quantum-statistics})? (c)
Hitunglah $\bar n$ untuk $(\epsilon - \mu)/k_{\text{B}}T = 0.1$, $1$,
$3$ lalu bandingkanlah dengan Boltzmann. (d) Foton memiliki $\mu = 0$:
berikanlah alasan fisisnya (cacahnya tidak kekal --- dindingnya menyerap
dan memancar dengan bebas).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:6}
Langmuir dalam kerja. (a) Dari isotermnya, tekanan berapa yang mencapai
$\theta = 0.5$? $0.9$? (b) Mioglobin mengikat O$_2$ pada satu tapak:
penjenuhannya mengikuti Langmuir terhadap tekanan parsial oksigennya ---
mengapa kurva \emph{hiperbolik} (bukan sigmoid) yang teramati untuk
mioglobin sedangkan kurva S yang kooperatif untuk hemoglobin bertapak
empat? (c) Karbon monoksida terikat pada tapak yang sama dengan
$\epsilon_0$ yang lebih besar sebanyak $\approx\qty{0.13}{eV}$: pada
tekanan parsial yang sama dan \qty{310}{K}, hitunglah nisbah
keterisian CO:O$_2$ lalu jelaskanlah keracunan CO. (d) Mengapa terapi
oksigen hiperbarik berhasil, dalam bahasa isoterm?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:7}
Fluktuasi cacah. (a) Tunjukkanlah $\operatorname{Var}N =
k_{\text{B}}T\,\partial\langle N\rangle/\partial\mu$. (b) Untuk gas
ideal klasik, simpulkanlah $\operatorname{Var}N = \langle
N\rangle$: yakni Poisson. (c) Serasikanlah dengan fluktuasi kerapatan
\cref{exo:b3:microcanonical-ensemble:12}. (d) Untuk satu aras
fermionik, hitunglah $\operatorname{Var}n = \bar n(1 - \bar
n)$: di manakah cacah fermionnya paling tenang, dan mengapa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:8}
Osmosis dari $\mu$. Sebuah selaput meloloskan air tetapi tidak zat
terlarut (berkepekatan $c$ tiap volumenya, encer). (a) Dengan
menyamakan $\mu$ air pada kedua sisi selaputnya, tunjukkanlah bahwa sisi
air murninya harus ditekan berlebih sebesar $\Pi = ck_{\text{B}}T$ (van
't Hoff) --- terimalah bahwa zat terlarut menurunkan $\mu$ air sebesar
$-k_{\text{B}}T\,c/n_{\text{w}}$ tiap molekul airnya. (b) Hitunglah
$\Pi$ untuk larutan garam fisiologis ($c \approx \qty{3e26}{m^{-3}}$
partikel seluruhnya). (c) Sel darah merah dalam air murni: ke arah mana
airnya mengalir, dan dengan akibat apa? (d) Mengapa pohon tak memerlukan
pompa maupun mukjizat untuk menaikkan getahnya sampai puluhan meter
(gradien $\mu$ osmotik dan kapiler)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:grand-canonical:9}
Kesetimbangan dalam sebuah medan. Dalam gravitasi, kesetimbangan sebuah
kolom gas menuntut \omterm{thm:b3:grand-canonical:distribution}{potensial kimia} \emph{total} $\mu(h) =
k_{\text{B}}T\ln(n\lambda_T^3) + mgh$ agar seragam. (a) Turunkanlah
hukum barometriknya dari sini dalam satu baris. (b) Rampatkanlah ke
sebuah sentrifuga berkecepatan sudut $\omega$ (dengan potensial
$-\tfrac12 m\omega^2r^2$) lalu perolehlah profil radialnya. (c)
Sentrifuga pengayaan uranium berputar pada $\omega r \sim
\qty{600}{m/s}$: hitunglah nisbah kesetimbangan penguatan $^{238}$UF$_6$
terhadap $^{235}$UF$_6$ tiap tahapnya, $\exp[\Delta m\,\omega^2
r^2/2k_{\text{B}}T]$ dengan $\Delta m = \qty{5e-27}{kg}$, $T =
\qty{320}{K}$. (d) Mengapa banyak tahap dirangkai berjenjang?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:10}
Saha, dengan tangan. (a) Kerjakanlah bukti penyetimbangan $\mu$
\cref{prop:b3:grand-canonical:saha}. (b) Definisikanlah bagian yang
terion $x$ pada kerapatan hidrogen total $n$ lalu tunjukkanlah $x^2/(1 - x)
= \eu^{-E_{\text{I}}/k_{\text{B}}T}/n\lambda_{T,\text{e}}^3$. (c)
Fotosfer matahari: $T = \qty{5800}{K}$, $n \approx
\qty{e23}{m^{-3}}$: tunjukkanlah bahwa bagian hidrogen yang terion hanya
$\sim\num{e-4}$ --- permukaan Matahari bersifat netral. (d) Pada $T =
\qty{10000}{K}$ (dengan $n$ yang sama): hitunglah ulang lalu berilah
ulasan tentang kecuramannya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:11}
Ketika gas menjadi kuantum. Kemerosotan mulai berlaku pada $n\lambda_T^3 \sim
1$. (a) Untuk udara pada \qty{300}{K}: seberapa jauh di bawah ambangnya
(\cref{exo:b3:grand-canonical:2})? (b) Untuk elektron hantaran
dalam tembaga ($n = \qty{8.5e28}{m^{-3}}$): carilah suhu yang di
bawahnya $n\lambda_T^3 > 1$ --- lalu simpulkanlah bahwa elektron sebuah
logam merosot pada suhu laboratorium \emph{mana pun}.
(c) Untuk atom rubidium pada $n = \qty{e20}{m^{-3}}$: carilah suhu
ambang kuantumnya (yakni skala BEC
\cref{ch:b3:quantum-statistics}). (d) Urutkanlah ketiganya lalu
nyatakanlah aturannya: partikel ringan dan kerapatan tinggi menjadi
kuantum terlebih dahulu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:grand-canonical:12}
Alam semesta kehilangan positronnya. Dalam alam semesta dini yang panas,
$\eu^+ + \eu^- \rightleftharpoons 2\gamma$ menahan pasangannya dalam
kesetimbangan dengan $\mu_{e^+} + \mu_{e^-} = 0$ (foton: $\mu = 0$).
(a) Untuk plasma yang setangkup ini memberikan $\mu_{e^\pm} \approx 0$:
tunjukkanlah bahwa kerapatan pasangannya lalu menjadi $n_\pm \approx
\lambda_T^{-3}\eu^{-m_{\text{e}}c^2/k_{\text{B}}T}$ begitu
$k_{\text{B}}T \ll m_{\text{e}}c^2$. (b) Hitunglah suhu tempat
$m_{\text{e}}c^2 = k_{\text{B}}T$ beserta skala waktu kosmik yang
bersesuaian dengannya (terimalah $t \sim \qty{1}{s}$ pada
$\qty{e10}{K}$). (c) Tunjukkanlah bahwa pendinginan lanjutan yang
sederhana (sebesar beberapa kali lipat) meruntuhkan kerapatan
pasangannya sebanyak banyak orde besar: positronnya saling memusnah
habis, lalu menyisakan hanya kelebihan elektron purba yang sangat kecil.
(d) Foton yang terlepas memanaskan kembali rendaman radiasinya: radiasi
peninggalan mana yang membawa kuitansinya
(\cref{ch:b3:astrophysics})?
\end{exercise}

\section{Soal: Membaca suhu bintang}

\begin{problem}[{Soal akhir pekan --- Saha, Payne, dan terbuat dari apa
alam semesta}]\label{pb:b3:grand-canonical:1}
Spektrum bintang telah dikatalogkan sepanjang dasawarsa 1890 ke dalam
kelas O, B, A, F, G, K, M menurut kuatnya garis serapannya ---
dengan garis Balmer hidrogen paling kuat pada kelas A dan secara
membingungkan lemah baik pada bintang yang lebih panas maupun yang lebih
dingin. Pada 1925 Cecilia Payne menerapkan \omterm{prop:b3:grand-canonical:saha}{persamaan Saha} yang masih
baru lalu menunjukkan bahwa seluruh kebun binatang itu adalah satu zat
pada suhu yang berbeda --- dan bahwa bintang sebagian besarnya hidrogen,
yakni kesimpulan yang begitu sesat sehingga ia dipaksa melunakkannya
dalam disertasinya. Soal ini menelusuri kembali penalarannya.
Data: serapan Balmer berangkat dari $n = 2$, pada $E_2 - E_1 =
\qty{10.2}{eV}$ di atas keadaan dasarnya; $E_{\text{I}} =
\qty{13.6}{eV}$; kerapatan elektron fotosfer $n_{\text{e}}
\sim \qty{e20}{m^{-3}}$.

\medskip\noindent\textbf{Bagian I --- Dua faktor yang bersaing.}
\begin{enumerate}
  \item Serapan Balmer memerlukan hidrogen netral yang \emph{berada
        dalam} $n =
        2$: tuliskanlah faktor Boltzmann bagi populasi $n = 2$-nya
        (degenerasi: $g_2/g_1 = 4$).
  \item Hitunglah nilainya pada $5000$, $10000$ dan \qty{20000}{K}: ke
        arah mana faktor ini mendorong kuatnya garis ketika $T$ menaik?
  \item Kini faktor Sahanya: ketika $T$ menaik, apa yang terjadi pada
        bagian yang \emph{netral}, dan karenanya pada garis Balmernya?
  \item Jelaskanlah secara kualitatif mengapa hasil kalinya --- yang
        tereksitasi \emph{dan} masih netral --- harus memuncak pada
        suatu suhu pertengahan.
  \item Dengan memakai \omterm{prop:b3:grand-canonical:saha}{persamaan Saha} dengan $n_{\text{e}} =
        \qty{e20}{m^{-3}}$, perkirakanlah suhu pengionan separuh
        hidrogennya.
  \item Rangkailah puncaknya: di dekat suhu berapa garis Balmer
        seharusnya paling kuat, dan kelas spektrum yang mana
        (bintang A, $\sim\qty{10000}{K}$) yang terpilih olehnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Deret spektrum yang
terpecahkan.}
\begin{enumerate}[resume]
  \item Bintang M (\qty{3000}{K}) menunjukkan pita molekuler yang kuat
        (TiO) dan hidrogen yang lemah: jelaskanlah keduanya dengan kedua
        faktornya.
  \item Bintang O (\qty{40000}{K}) menunjukkan garis helium dan lagi-lagi
        hidrogen yang lemah: jelaskanlah (apa yang telah terjadi pada
        hidrogennya, dan mengapa helium, $E_{\text{I}} = \qty{24.6}{eV}$,
        kini mendapat gilirannya?).
  \item Matahari (G, \qty{5800}{K}) menunjukkan garis kuat
        \emph{kalsium terion} ($E_{\text{I,Ca}} = \qty{6.1}{eV}$)
        namun hidrogennya lemah: tunjukkanlah dengan penalaran Saha
        mengapa kalsiumnya sudah terion sedangkan hidrogennya masih
        netral dan sebagian besarnya berada dalam $n = 1$.
  \item Para astronom sebelum Payne membaca kuatnya garis sebagai
        kelimpahan: apa yang disimpulkan garis kalsium Matahari yang
        perkasa dan garis Balmernya yang lemah tentang susunannya?
  \item Wawasan Payne: perbaikilah kuatnya tiap garis menurut ``faktor
        keterlihatan'' Boltzmann--Sahanya. Ketika ia melakukannya,
        susunan kelimpahan sejati yang bagaimana yang muncul?
  \item Hasilnya --- hidrogen sejuta kali lebih melimpah daripada
        kalsium, bintang sebagian besarnya hidrogen dan helium ---
        mengapa hasil itu ditolak (apa yang disarankan kimia yang
        berpusat pada Bumi), dan kapan ia dibenarkan?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Bilangan pada puncaknya.}
\begin{enumerate}[resume]
  \item Hitunglah faktor Boltzmann $n = 2$-nya persis pada
        \qty{9600}{K}.
  \item Dengan perkiraan Saha butir 5, ambillah bagian yang netral pada
        $9600$, $15000$ dan \qty{25000}{K} kira-kira sebesar
        $0.5$, $\num{e-3}$, $\num{e-5}$: bentuklah hasil kalinya
        (bagian tereksitasi $\times$ bagian netral) pada ketiga suhu
        Anda lalu carilah maksimumnya.
  \item Mengapa puncak Balmernya \emph{tajam} di sisi panasnya
        (eksponensial mana yang menang) dan \emph{landai} di sisi
        dinginnya?
  \item Fotosfer katai putih memiliki $n_{\text{e}}$ seribu kali lebih
        besar: ke arah mana suhu pengionan separuhnya berpindah, dan
        mengapa (Le Chatelier dalam bahasa $\mu$)?
  \item Dua bintang bersuhu sama tetapi berbeda tekanannya karenanya
        menunjukkan kuat garis yang berbeda: sifat bintang yang mana
        (raksasa lawan katai --- gravitasi permukaan) yang karenanya
        dapat dibaca para spektroskopis?
  \item Survei modern memancang suhu bintang sampai $\pm1\%$ dari
        nisbah garisnya: nyatakanlah rantai postulat $\to$ Boltzmann
        $\to$ Saha $\to$ termometer.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Apa yang terselesaikan.}
\begin{enumerate}[resume]
  \item Deret OBAFGKM dahulu merupakan kekacauan abjad yang ditata ulang
        secara empiris: setelah Saha, peubah fisis tunggal apa yang
        menatanya?
  \item Nyatakanlah mengapa unsur yang \emph{sama} dapat menguasai
        sebuah spektrum pada satu suhu lalu lenyap pada suhu lain tanpa
        perubahan kelimpahan apa pun --- yakni pelajaran inti yang
        diajarkan Payne kepada astronomi.
  \item Hidrogen sebesar tiga perempat seluruh massa barionik:
        sebutkanlah dua tiang lain buku ini yang bertumpu pada
        kelimpahan itu (fusi bintang, \cref{ch:b3:astrophysics}; langit
        21 cm, \cref{ch:b3:spin-two-level}).
  \item Otto Struve menyebut disertasi Payne sebagai ``disertasi doktoral
        paling cemerlang yang pernah ditulis dalam astronomi'': dari soal ini,
        benarkanlah pujian itu dalam satu kalimat.
  \item Fisika Saha yang sama, yang dijalankan terbalik pada skala
        kosmik, meramalkan suhu tempat hidrogen seluruh alam semesta
        pertama kali bergabung ($\sim\qty{3000}{K}$ pada kerapatan
        kosmik yang jauh lebih rendah): sebutkanlah peristiwanya beserta
        peninggalannya (\cref{ch:b3:astrophysics}).
  \item Kesetimbangan pengionan juga menjalankan lampu neon dan
        pemrosesan plasma: nyatakanlah dalam satu kalimat mengapa
        keadaan terion bergaya \qty{10000}{K} dapat berada dalam sebuah
        tabung yang kacanya tetap dingin (dua subsistem mana yang gagal
        menyetimbang?).
  \item Ringkaslah hasil bernama itu: satu syarat kesetimbangan,
        $\mu_{\text{H}} = \mu_{\text{p}} + \mu_{\text{e}}$, mengubah
        abjad spektrum menjadi termometer yang berjalan dari
        $3000$ sampai \qty{40000}{K}, menempatkan puncak Balmer pada
        bintang A, lalu menyingkapkan alam semesta yang terbuat dari
        hidrogen.
\end{enumerate}
\end{problem}
