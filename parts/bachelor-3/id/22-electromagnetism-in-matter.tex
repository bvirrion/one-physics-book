\section{Pengutuban dan medan pergeseran}

\begin{definition}[Pengutuban dan muatan terikat]\label{def:b3:electromagnetism-in-matter:polarisation}
\index{pengutuban}\index{muatan terikat}\index{dielektrik}
Sebuah \emph{dielektrik} adalah penyekat yang molekulnya memperoleh (atau
sudah memiliki) momen dwikutub listrik. Tanggapan bahannya diringkas
oleh \emph{pengutuban} $\vect P$: yakni momen dwikutub tiap satuan
volumenya, $\vect P = n\langle\vect p\rangle$ untuk $n$ molekul tiap
satuan volumenya. Balok yang terkutub membawa \emph{muatan terikat} ---
yang tak bebas pergi, tetapi sepenuhnya nyata: yakni kerapatan permukaan
$\sigma_{\text{b}} = \vect P\cdot\vect n$ tempat pengutubannya
bertemu permukaannya, dan kerapatan volume $\rho_{\text{b}} =
-\nabla\cdot\vect P$ di mana pun ia tak seragam. Lempeng yang terkutub
seragam persis setara dengan dua lembar muatan $\pm P$
pada kedua mukanya: seluruh kepala dan ekor dwikutub di dalamnya saling meniadakan.
\end{definition}

\begin{proposition}[Medan pergeseran]\label{prop:b3:electromagnetism-in-matter:displacement}
\index{pergeseran listrik}\index{kerentanan listrik}\index{permitivitas nisbi}
Muatan totalnya adalah yang bebas ditambah yang terikat; hukum Gauss dengan $\rho =
\rho_{\text{f}} + \rho_{\text{b}}$ tertata ulang menjadi hukum bagi
\emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{pergeseran listrik}} $\vect D = \varepsilon_0\vect E +
\vect P$:
\[
\nabla\cdot\vect D = \rho_{\text{f}} :
\]
Sumber $\vect D$ hanyalah muatan bebasnya --- yakni muatan pada
pelat dan kawat kita. Dalam \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielektrik} \emph{lelurus} tanggapannya
sebanding, $\vect P = \varepsilon_0\chi_{\text{e}}\vect E$, sehingga
$\vect D = \varepsilon_0\varepsilon_{\text{r}}\vect E$ dengan
$\varepsilon_{\text{r}} = 1 + \chi_{\text{e}}$ sebagai \emph{\omterm{prop:b3:electromagnetism-in-matter:displacement}{permitivitas
nisbi}}: udara $1.0006$, polietilena $2.3$, kaca $\sim 5$,
air yang luar biasa besar $80$. Di antara pelat kapasitor yang ditahan pada
muatan tetap, $\vect D$ tak berubah oleh \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielektriknya}, sehingga $\vect E =
\vect D/\varepsilon_0\varepsilon_{\text{r}}$ \emph{turun} sebesar
$\varepsilon_{\text{r}}$: muatan permukaan terikatnya berhadapan dengan
muatan bebasnya lalu meniadakan sebagian besar medannya. Kapasitansinya
dikalikan
$\varepsilon_{\text{r}}$ --- yakni seluruh industri kapasitor dalam satu
huruf Yunani.
\end{proposition}

\begin{proof}
$\nabla\cdot(\varepsilon_0\vect E) = \rho_{\text{f}} +
\rho_{\text{b}} = \rho_{\text{f}} - \nabla\cdot\vect P$; pindahkanlah
suku \omterm{def:b3:electromagnetism-in-matter:polarisation}{pengutubannya} ke ruas kiri.
\end{proof}

\section{Dari mana permitivitas datang}

\begin{proposition}[Pengutuban terimbas dan Clausius--Mossotti]\label{prop:b3:electromagnetism-in-matter:mossotti}
\index{keterkutuban}\index{hubungan Clausius--Mossotti}
Sebuah atom dalam medan $\vect E_{\text{loc}}$ merentang menjadi dwikutub
$\vect p = \alpha\vect E_{\text{loc}}$, dengan $\alpha$ sebagai
\emph{keterkutuban} (berdimensi: $\varepsilon_0\times$volume ---
kira-kira $\varepsilon_0$ dikalikan volume atomnya). Dalam gas encer
$E_{\text{loc}} \approx E$ dan $\chi_{\text{e}} =
n\alpha/\varepsilon_0$ bernilai kecil. Dalam medium yang rapat tiap
molekulnya juga merasakan tetangganya yang terkutub: mengukir rongga bola
kecil di sekelilingnya memberikan $\vect E_{\text{loc}} = \vect E + \vect
P/3\varepsilon_0$, dan ketaatasasan dirinya menghasilkan
\emph{\omterm{prop:b3:electromagnetism-in-matter:mossotti}{hubungan Clausius--Mossotti}}
\[
\frac{\varepsilon_{\text{r}} - 1}{\varepsilon_{\text{r}} + 2}
= \frac{n\alpha}{3\varepsilon_0} :
\]
ukurlah kerentanan uapnya yang mungil lalu ramalkanlah kerentanan
cairannya --- ia berhasil sampai beberapa persen untuk cairan takkutub.
\end{proposition}
\admitted

\begin{remark}
Medan rongga $\vect P/3\varepsilon_0$ adalah hasil bola yang terkutub
seragam, yang dikutip di sini tanpa bukti; penurunan nilai batasnya yang
jujur termasuk dalam kuliah elektrodinamika tersendiri. Segala hal
lainnya di atas hanyalah pembukuan.
\end{remark}

\begin{proposition}[Molekul berkutub: fungsi Langevin]\label{prop:b3:electromagnetism-in-matter:langevin}
\index{fungsi Langevin}\index{pengutuban arah}
Molekul dengan dwikutub \emph{tetap} $p$ (air:
$p = \qty{6.2e-30}{C.m}$) tak perlu direntangkan --- hanya disearahkan,
melawan kegelisahan termalnya. Rata-rata kanonik
\cref{ch:b3:canonical-ensemble} atas arahnya, dengan energi
$-pE\cos\theta$, memberikan
\[
\langle\cos\theta\rangle = L(x) = \coth x - \frac1x ,
\qquad x = \frac{pE}{k_{\text{B}}T} ,
\]
yakni \emph{\omterm{prop:b3:electromagnetism-in-matter:langevin}{fungsi Langevin}}: lurus ($L \approx x/3$) pada $x$ kecil,
lalu menjenuh pada $1$ ketika medannya menang telak. Dalam rezim
lelurusnya $\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ --- yakni
hukum Curie listrik, yang turun sebagai $1/T$: panaskanlah cairan
berkutub dan permitivitasnya \emph{turun}, yakni tanda pengenal yang
membedakan dwikutub tetap dari dwikutub terimbas. Untuk air pada suhu
ruang ini meramalkan
$\chi_{\text{e}} \sim 12$ --- orde besarnya tepat, dengan kerja sama
ikatan hidrogen mendorong nilai sebenarnya menjadi $80$.
\end{proposition}

\begin{proof}[Bukti sebagian]
$Z = \int_0^\pi \eu^{x\cos\theta}\,2\pi\sin\theta\,\dd\theta \propto
\sinh x/x$, dan $\langle\cos\theta\rangle = \dd\ln Z/\dd x = \coth x
- 1/x$. Menjabarkan $\coth x = 1/x + x/3 - \dots$ untuk $x$ kecil memberikan
$L \approx x/3$, sehingga $P = np\langle\cos\theta\rangle =
np^2E/3k_{\text{B}}T$.
\end{proof}

\section{Magnetisasi dan medan \texorpdfstring{$H$}{H}}

\begin{definition}[Magnetisasi, arus terikat, dan H]\label{def:b3:electromagnetism-in-matter:magnetisation}
\index{magnetisasi}\index{medan magnet H}\index{kerentanan magnet}
Kisah magnetnya berjalan sejajar. Materi penuh dengan gelung arus
mikroskopis --- elektron yang mengorbit dan berputar, masing-masing sebuah
dwikutub magnet $\vect\mu$; kerapatannya adalah \emph{magnetisasi}
$\vect M = n\langle\vect\mu\rangle$. Balok yang termagnet seragam setara
dengan \emph{arus permukaan terikat} yang beredar mengelilingi
sisinya (gelung di dalamnya saling meniadakan berpasangan; sebatang
magnet \emph{adalah} solenoida arus terikat). Dengan membelah arusnya
menjadi yang bebas dan yang terikat, hukum Ampère tertata ulang di
sekitar medan bantu
\[
\vect H = \frac{\vect B}{\mu_0} - \vect M , \qquad
\oint \vect H\cdot\dd\vect\ell = I_{\text{bebas}} :
\]
$\vect H$ bersumber dari arus dalam kawat kita (bersatuan
\unit{A/m}), sedangkan $\vect B$ menyimpan fisikanya --- gaya, fluks,
induksi. Medium lelurus memiliki $\vect M = \chi_{\text{m}}\vect H$ dan
$\vect B = \mu_0(1 + \chi_{\text{m}})\vect H =
\mu_0\mu_{\text{r}}\vect H$.
\end{definition}

\begin{remark}[Tiga watak magnet]\label{rem:b3:electromagnetism-in-matter:zoo}
\index{diamagnetisme}\index{paramagnetisme}\index{feromagnetisme}
Bahan menjawab $\vect H$ dengan tiga cara.
\emph{Diamagnet} ($\chi_{\text{m}} \sim -10^{-5}$: air, tembaga,
grafit) menanggapi seperti hukum Lenz yang dibuat abadi: medan terapannya
mengusik tiap orbit elektronnya sedemikian rupa sehingga melawannya ---
lemah, semesta, tak bergantung suhu, dan ditolak magnet (seekor katak
pernah dilayangkan dengan cara ini dalam lubang \qty{16}{T}).
\emph{Paramagnet} ($\chi_{\text{m}} \sim +10^{-5}$ sampai $10^{-3}$:
aluminium, O$_2$) memiliki momen tetap --- yakni spin tak berpasangan dari
\cref{ch:b3:identical-particles} --- yang menyearah seperti dwikutub
Langevin: $\chi_{\text{m}} = \mu_0 n\mu^2/3k_{\text{B}}T$, yakni hukum
Curie $1/T$ (versi kuantum dua arasnya adalah
\cref{exo:b3:canonical-ensemble:11}).
\emph{Feromagnet} (besi, kobalt, nikel: $\mu_{\text{r}}$ dalam
ribuan) adalah paramagnet yang spinnya juga saling berbicara ---
yakni gandengan pertukaran \cref{exo:b3:identical-particles:12} ---
dan di bawah \omterm{thm:b3:phase-transitions:meanfield}{suhu Curie} mereka menyearah \emph{secara spontan}:
\cref{thm:b3:phase-transitions:meanfield}, yang kini dipekerjakan.
\end{remark}

\section{Feromagnetisme dalam kerja: ranah dan histeresis}

\begin{definition}[Ranah dan histeresis]\label{def:b3:electromagnetism-in-matter:hysteresis}
\index{ranah magnet}\index{histeresis}\index{keteringatan}\index{kekoersifan}
Bongkahan besi mentah \emph{tidak} termagnet: ia pecah menjadi
\emph{ranah}, yakni daerah berskala mikron yang masing-masing termagnet
penuh tetapi menunjuk ke arah berbeda, sehingga medan di luarnya (beserta
ongkos energinya) nyaris meniadakan diri. Medan terapan $\vect H$
menggerakkan dinding ranahnya --- ranah yang menguntungkan tumbuh ---
lalu memutar seluruh ranahnya sampai sejajar. Dindingnya tersangkut pada
cacat, sehingga prosesnya taktermampatbalikkan: sapulah
$H$ naik dan turun dan $B(H)$ menjejak sebuah gelung, yakni \emph{daur histeresis}nya. Matikanlah arusnya dan medan \emph{teringat}
$B_{\text{r}}$ bertahan; meniadakannya memerlukan medan \emph{koersif}
kebalikannya $H_{\text{c}}$. Luas yang dilingkupi gelungnya adalah
energi yang terlesap tiap daur dan tiap satuan volumenya. Bahan
\emph{lunak} (baja silikon, ferit: gelung tipis, $H_{\text{c}}$ kecil)
membuat teras trafo dan motor; bahan \emph{keras}
(alniko, Nd$_2$Fe$_{14}$B: gelung gemuk, $H_{\text{c}}$ raksasa) membuat
magnet tetap --- dan ingatan magnetik: magnet lemari es dan cakram keras
adalah gelung histeresis yang menolak lupa.
\end{definition}

\begin{example}[Elektromagnet berteras besi]\label{ex:b3:electromagnetism-in-matter:electromagnet}
Lilitkanlah $N$ lilitan berarus $I$ mengelilingi cincin besi ($\mu_{\text{r}}
\sim 5000$) dengan celah udara kecil $e$. Hukum Ampère untuk $\vect H$
mengelilingi gelungnya: $H_{\text{besi}}\ell + H_{\text{celah}}e = NI$, dan
kesinambungan fluksnya membuat $B$ sama, sehingga
\[
B\Big(\frac{\ell}{\mu_0\mu_{\text{r}}} + \frac{e}{\mu_0}\Big) = NI .
\]
Dengan $\ell = \qty{1}{m}$ dan $e = \qty{1}{cm}$, suku celahnya
lima puluh kali suku besinya: nyaris seluruh upaya kumparannya
dihabiskan untuk mendorong medan menyeberangi satu sentimeter udara.
Itulah rangkaian magnetnya dalam satu baris --- besi adalah penghantar
fluks yang nyaris sempurna, udara adalah hambatannya --- dan karena
itulah motor, relai dan pengangkat rongsokan menjaga celah udaranya
setipis mungkin
(\cref{pb:b3:electromagnetism-in-matter:1}).
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:1}
Pembukuan \omterm{def:b3:electromagnetism-in-matter:polarisation}{muatan terikat}. Sebuah lempeng setebal $d$ terkutub
seragam sepanjang normalnya, dengan besar $P$. (a) Berikanlah kerapatan
\omterm{def:b3:electromagnetism-in-matter:polarisation}{muatan terikat} pada tiap mukanya. (b) Tunjukkanlah bahwa medan \emph{di
dalamnya} akibat lembar itu adalah $E = P/\varepsilon_0$, yang melawan $\vect P$. (c)
Berapakah $\vect D$ di dalamnya, tanpa muatan bebas di mana pun? (d) Sebatang
elektret (dengan $\vect P$ yang dibekukan) adalah padanan listrik benda
magnet yang mana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:2}
Kapasitor berdielektrik. Sebuah kapasitor pelat sejajar
($C_0 = \qty{100}{pF}$) diisi sampai \qty{12}{V} lalu diputus hubungannya.
Sebuah lempeng ber-$\varepsilon_{\text{r}} = 4$ mengisinya. Carilah (a)
kapasitansi barunya; (b) tegangan barunya; (c) energinya sebelum dan
sesudah --- ke mana perginya selisihnya? (d) Ulangilah (b)–(c) dengan
baterainya tetap terhubung.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:3}
Kebun binatang kerentanan. Golongkanlah (dia-, para-, atau feromagnetik)
lalu benarkanlah dari struktur elektroniknya bila Anda bisa: air
($\chi_{\text{m}} = -9\times10^{-6}$), aluminium
($+2.2\times10^{-5}$), oksigen cair ($+3.5\times10^{-3}$ ---
mengapa O$_2$ bersifat magnet sama sekali?), nikel ($\mu_{\text{r}} \sim 600$
pada medan kecil). Yang mana dari keempatnya yang akan ditarik magnet kuat
secara kasat mata, dan yang mana yang ditolaknya sedikit sekali?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:electromagnetism-in-matter:4}
$B$, $H$, $M$. Sebuah solenoida panjang ($n = \num{2000}$ lilitan tiap meternya,
$I = \qty{1.5}{A}$) diisi besi, dan $B = \qty{1.6}{T}$
terukur. Hitunglah (a) $H$; (b) $M$; (c) penguatannya
$B/\mu_0H$; (d) arus permukaan terikat tiap meternya yang setara dengan
$M$ --- bandingkanlah dengan $nI$ kumparannya sendiri.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:5}
Kabel sesumbu, yang tersekat. Sebuah saluran sesumbu (berjejari dalam $a$, luar
$b$) mengangkut muatan bebas $\lambda$ tiap satuan panjang pada terasnya,
dengan \omterm{def:b3:electromagnetism-in-matter:polarisation}{dielektrik} $\varepsilon_{\text{r}}$ di antaranya. (a) Carilah
$\vect D(r)$ dari kesetangkupannya. (b) Simpulkanlah $\vect E$ dan $\vect P$.
(c) Hitunglah kerapatan \omterm{def:b3:electromagnetism-in-matter:polarisation}{muatan terikatnya} pada $r = a$ dan $r = b$ lalu
periksalah bahwa jumlahnya nol tiap satuan panjangnya. (d) Tunjukkanlah
bahwa kapasitansi tiap satuan panjangnya dikalikan $\varepsilon_{\text{r}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:6}
Clausius--Mossotti dalam kerja. Argon berwujud gas pada \qty{1}{bar},
\qty{300}{K} memiliki $\varepsilon_{\text{r}} = 1.00052$. (a) Tariklah
$\alpha/\varepsilon_0$ (yakni sebuah volume) lalu bandingkanlah dengan
volume atomnya. (b) Argon cair memiliki $n = \qty{2.1e28}{m^{-3}}$:
ramalkanlah $\varepsilon_{\text{r}}$ dari Clausius--Mossotti (terukur: 1.53).
(c) Mengapa nilai naifnya $\chi = n\alpha/\varepsilon_0$ melampaui
sasarannya dengan lebih ringan di sini daripada untuk air? (d) Untuk
molekul yang mana seluruh skemanya pasti gagal, dan apa penggantinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:7}
Permitivitas raksasa air. (a) Dari
\cref{prop:b3:electromagnetism-in-matter:langevin}, hitunglah
$\chi_{\text{e}} = np^2/3\varepsilon_0k_{\text{B}}T$ untuk air
($n = \qty{3.3e28}{m^{-3}}$, $p = \qty{6.2e-30}{C.m}$,
$T = \qty{300}{K}$). (b) Perkirakanlah medan yang diperlukan untuk $x = 1$
(yakni penjenuhan yang serius) lalu bandingkanlah dengan kekuatan
\omterm{def:b3:electromagnetism-in-matter:polarisation}{dielektrik} air $\sim\qty{7e7}{V/m}$. (c) Oven gelombang mikro berjalan pada \qty{2.45}{GHz}:
mengapa tanggapan dwikutub berputar yang \emph{merugi} memanaskan makanan, dan
mengapa es memanas jauh lebih lambat daripada air cair? (d) Ramalkanlah
tanda $\dd\varepsilon_{\text{r}}/\dd T$ untuk air dan untuk
argon.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:8}
Diamagnetisme yang diperkirakan. Perlakukanlah elektron atom sebagai muatan pada
cincin berjejari $r$; medan terapan $B$ mengubah frekuensi sudutnya sebesar
geseran Larmor $\Delta\omega = eB/2m_{\text{e}}$. (a) Tunjukkanlah bahwa
momen terimbasnya adalah $\Delta\mu = -e^2r^2B/4m_{\text{e}}$ (yang melawan
$B$). (b) Dengan merata-ratakan arahnya lalu menjumlahkan $Z$ elektron tiap
atomnya, orang memperoleh $\chi_{\text{m}} = -\mu_0ne^2Z\langle
r^2\rangle/6m_{\text{e}}$: hitunglah untuk air ($n =
\qty{3.3e28}{m^{-3}}$ molekul, $Z = 10$, $\langle r^2\rangle
\approx (\qty{0.7}{\angstrom})^2$) lalu bandingkanlah dengan nilai terukur
$-9\times10^{-6}$. (c) Mengapa diamagnetisme tak bergantung suhu
sedangkan paramagnetisme bergantung? (d) Mengapa bahkan seekor katak melayang dalam
\qty{16}{T}, dan mengapa medan magnetnya harus \emph{tak seragam}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:electromagnetism-in-matter:9}
Paramagnetisme Curie dan pendinginan. Sebuah garam membawa $n =
\qty{2e27}{m^{-3}}$ ion bermomen $\mu \approx \mu_{\text{B}}$.
(a) Hitunglah $\chi_{\text{m}} = \mu_0n\mu^2/3k_{\text{B}}T$ pada
\qty{300}{K} dan pada \qty{1}{K}. (b) Pada suhu berapa
$\chi_{\text{m}}$ akan mencapai $1$ --- dan fisika apa (yang diabaikan di
sini) yang menyela lebih dahulu pada sebagian besar garam? (c) Penyahmagnetan
adiabatik: magnetkanlah pada \qty{1}{K}, kucilkanlah, lalu singkirkanlah medannya perlahan ---
jelaskanlah dengan entropi spin
\cref{ch:b3:microcanonical-ensemble} mengapa cuplikannya mendingin. (d) Mengapa
metodenya memerlukan \emph{para}magnet dan bukan feromagnet?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:10}
Rangkaian magnetnya. Sebuah torus besi ($\ell = \qty{60}{cm}$,
$\mu_{\text{r}} = 4000$, bertampang $A = \qty{16}{cm^2}$) membawa
$N = 500$ lilitan dan celah $e = \qty{4}{mm}$. (a) Tunjukkanlah $NI =
B(\ell/\mu_0\mu_{\text{r}} + e/\mu_0)$ lalu hitunglah arusnya untuk
$B = \qty{1.2}{T}$. (b) Berapa bagian dari $NI$ yang dihabiskan pada celahnya?
(c) Definisikanlah keengganan $\mathcal R = \ell/\mu A$ tiap
ruasnya lalu nyatakanlah kembali (a) sebagai ``hukum Ohm'' seri bagi fluksnya. (d)
Besinya menjenuh di dekat \qty{1.8}{T}: jelaskanlah apa yang terjadi pada
model rangkaiannya --- dan pada motor Anda --- di baliknya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:11}
Rugi \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histeresis}. Sebuah teras trafo (bervolume \qty{4e-3}{m^3})
berjalan pada \qty{50}{Hz}. Gelung baja silikonnya melingkupi
$\sim\qty{40}{J/m^3}$ tiap daurnya. (a) Hitunglah rugi daya
histeresisnya. (b) Teras yang sama dari baja keras
($\sim\qty{6e3}{J/m^3}$): hitunglah ruginya, lalu berilah putusan. (c) Arus pusar menambahkan
rugi $\propto f^2$: jelaskanlah mengapa terasnya dilaminasi dan mengapa
ferit mengambil alih pada frekuensi radio. (d) Sebuah bit cakram keras harus
\emph{menyimpan} \omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetisasinya} melawan tendangan termal selama sepuluh tahun:
kaitkanlah tuntutan atas $H_{\text{c}}$ untuk ingatan dengan tuntutan untuk
sebuah trafo, lalu simpulkanlah bahwa tak ada satu bahan pun yang dapat melakukan kedua tugasnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:electromagnetism-in-matter:12}
Medan penyahmagnetnya. Di dalam benda yang termagnet seragam tanpa
arus bebas, $\oint\vect H\cdot\dd\vect\ell = 0$ memaksa
$\vect H$ di dalamnya untuk \emph{melawan} $\vect M$ ($\vect H =
-N_{\text{d}}\vect M$, dengan $N_{\text{d}}$ sebagai faktor bentuk: $1/3$
untuk bola, $\approx 0$ sepanjang jarum yang panjang, $\approx 1$ melintasi
pelat tipis). (a) Benarkanlah limit jarum dan pelatnya dengan
gambaran arus terikat (solenoida). (b) Mengapa magnet yang pendek gempal
menyahmagnetkan dirinya sebagian sedangkan sebuah jarum menyimpan
\omterm{def:b3:electromagnetism-in-matter:magnetisation}{magnetisasinya}? (c) Kaitkanlah dengan jarum kompas dan dengan butiran
memanjang pita magnetik. (d) Sebuah bola besi lunak duduk dalam
$B_0$ luar yang seragam: jelaskanlah mengapa tanggapannya menjenuh pada $\vect H
\approx 0$ di dalamnya, yakni $M \approx 3B_0/\mu_0$ paling banyak, sebesar apa pun
$\mu_{\text{r}}$-nya.
\end{exercise}

\section{Soal: Pengangkat rongsokan}

\begin{problem}\label{pb:b3:electromagnetism-in-matter:1}
\emph{Pengangkat rongsokan.} Sebuah tempat daur ulang memesan sebuah
elektromagnet derek: yakni periuk besi bermuka datar, bergaris tengah satu
meter, yang harus mengangkat bal mobil rongsok seberat dua ton, lalu ---
sama pentingnya --- \emph{menjatuhkannya} atas perintah. Anda perancangnya.

\medskip\noindent\textbf{Bagian I --- Mengapa besi.}

\begin{enumerate}
  \item Kumparan telanjangnya: 400 lilitan berarus \qty{25}{A} yang
        tersebar pada lintasan magnet sepanjang $\sim\qty{1}{m}$.
        Perkirakanlah $B \sim
        \mu_0NI/\ell$ tanpa besi.
  \item Jelaskanlah dalam dua kalimat, dengan ranah, mengapa mengisi
        kumparannya dengan besi mengalikannya dengan $\sim\mu_{\text{r}}$.
  \item \omterm{def:b3:electromagnetism-in-matter:magnetisation}{Magnetisasi} besi menjenuh pada $M_{\text{s}} \approx
        \qty{1.7e6}{A/m}$. Hitunglah langit-langit $\mu_0M_{\text{s}}$
        yang dikenakannya pada medan kutubnya, apa pun yang dilakukan kumparannya.
  \item Tiap atom besi menyumbang $\approx 2.2$ \omterm{prop:b3:quantum-angular-momentum:magnetic}{magneton Bohr}.
        Periksalah: dengan $n = \qty{8.5e28}{m^{-3}}$, perolehlah kembali
        $M_{\text{s}}$.
  \item Mengapa muka kutubnya harus dimesin rata dan dijaga bebas dari
        karat dan kerikil? (Ingatlah
        \cref{ex:b3:electromagnetism-in-matter:electromagnet}.)
  \item Bebannya sendiri menjadi bagian rangkaian magnetnya.
        Sketsalah lintasan fluksnya: teras periuknya, muka utaranya, bal
        bajanya, muka selatannya, lalu kembali menembus kuk.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Rangkaian magnet dan gayanya.}

\begin{enumerate}[resume]
  \item Modelkanlah rangkaiannya: lintasan besi $\ell = \qty{1.2}{m}$,
        $\mu_{\text{r}} = 2000$, dan dua celah udara efektif
        (sentuhan muka dengan bal) masing-masing $e = \qty{1.5}{mm}$. Tuliskanlah
        hukum Ampère untuk $H$ mengelilingi gelungnya.
  \item Hitunglah ketiga suku keengganannya tiap satuan luas
        ($\ell/\mu_{\text{r}}$ berbanding $2e$) lalu tunjukkanlah
        bahwa celah milimeternya tetap memakan sebagian besar upaya
        kumparannya.
  \item Dengan $NI = \num{10000}$ amper-lilitan, hitunglah $B$ dalam
        celahnya.
  \item Tekanan angkat pada tiap muka kutubnya adalah $B^2/2\mu_0$
        (yakni kerapatan energi magnet yang terlepas tiap meter pendekatannya).
        Hitunglah dalam \unit{kPa} untuk $B$ Anda.
  \item Luas muka kutub totalnya $A = \qty{0.12}{m^2}$: hitunglah gaya
        angkatnya lalu ubahlah ke ton.
  \item Apakah ia memenuhi spesifikasi dua ton dengan marjin dua kali
        lipat? Jika tidak, sesuaikanlah $NI$ lalu nyatakanlah arus barunya.
  \item Bal yang terhimpit menyentuh mukanya barangkali pada sepertiga
        luasnya, dengan celah efektif yang lebih lebar. Hitunglah ulang gayanya untuk
        $e = \qty{4}{mm}$, $A_{\text{eff}} = \qty{0.04}{m^2}$ lalu
        berilah ulasan mengapa daya angkat terpasangnya mengutip kapasitas ``pelat datar''.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Mengingat dan melupakan.}

\begin{enumerate}[resume]
  \item Tempat rongsokannya membalik sakelarnya untuk menjatuhkan sebuah
        bal --- dan balnya tetap menggantung. Sebutkanlah biang keladinya,
        dengan gelung histeresisnya.
  \item Mengapa periuknya harus dari besi \emph{lunak} (dengan
        $H_{\text{c}}$ kecil, $B_{\text{r}}$ kecil) dan bukan dari baja
        magnet yang keras?
  \item Bahkan besi lunak menyimpan sedikit \omterm{def:b3:electromagnetism-in-matter:hysteresis}{keteringatan}. Usulkanlah
        penawar bakunya: yakni denyut arus \emph{terbalik} yang singkat ---
        titik gelung yang mana yang dibidiknya?
  \item Sebagian pengendali justru mengenakan arus bolak-balik yang
        meluruh. Sketsalah apa yang dilakukan lintasan $B$--$H$-nya dan mengapa
        ia berakhir tersahmagnetkan.
  \item Dereknya juga menangani lempeng panas langsung dari tanur,
        pada \qty{800}{\celsius}. Titik Curie besi adalah
        \qty{1043}{K}: apa yang terjadi pada gaya angkatnya, dan mengapa?
        (\cref{thm:b3:phase-transitions:meanfield}.)
  \item Seorang magang musim panas mengusulkan penghematan tembaga dengan
        melipatduakan celahnya dan melipatduakan $NI$. Pakailah rumus Bagian II Anda untuk
        menjelaskan ketaksetangkupannya: yang mana yang terbagi dua dan yang mana
        yang sekadar bertahan?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Daya, kalor, dan tagihannya.}

\begin{enumerate}[resume]
  \item Kumparannya: 400 lilitan tembaga, panjang lilitan rata-rata
        \qty{2.5}{m}, tampang kawat \qty{16}{mm^2}, hambatan jenis
        $\rho = \qty{1.7e-8}{\Omega.m}$. Hitunglah hambatannya.
  \item Pada $I = \qty{25}{A}$: hitunglah daya yang terlesap, dan energi
        hariannya untuk giliran kerja delapan jam pada tugas \qty{60}{\percent}.
  \item Energi magnet yang tersimpan dalam kedua celahnya
        ($B^2/2\mu_0 \times$ volume): hitunglah lalu bandingkanlah dengan
        satu \emph{detik} pelesapan kumparannya. Ke mana perginya seluruh
        sisa energi listriknya?
  \item Menjatuhkan bebannya nyaris tak mengembalikan energi tersimpan
        apa pun ke jaringan. Jelaskanlah, dengan luas gelungnya dan
        lonjakan induktifnya, mengapa pengendalinya memerlukan lintasan
        roda bebas (flyback) melintasi kumparannya.
  \item Dayanya padam ketika sebuah bal melayang di udara. Apa yang
        dilakukan balnya, dan apa yang ditahan \omterm{def:b3:electromagnetism-in-matter:hysteresis}{keteringatannya}
        \emph{sendirian}? Benarkanlah aturan tempat rongsokan bahwa tak
        seorang pun berjalan di bawah magnet yang bertenaga ---
        beserta bujukan penjualan pengangkat bercadangan baterai.
  \item Ringkaslah rancangannya dalam empat baris: langit-langit medannya
        ditetapkan $M_{\text{s}}$, gayanya oleh $B^2A/2\mu_0$,
        keterkendaliannya oleh \omterm{def:b3:electromagnetism-in-matter:hysteresis}{histeresis} besi lunak, dan ongkos
        pengoperasiannya oleh $RI^2$ --- yakni seluruh bab yang
        tergantung pada satu derek.
\end{enumerate}
\end{problem}
