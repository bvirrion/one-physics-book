\chapter{Peralihan Fase}\label{ch:b3:phase-transitions}

Panaskanlah air satu derajat, dan tak banyak yang terjadi --- sampai,
pada satu suhu yang tertakrif tajam, cairannya mencabik dirinya menjadi
uap. Dinginkanlah besi menembus \qty{1043}{K} dan, tanpa perubahan
susunan apa pun, ia memagnet secara spontan. Tak ada apa pun dalam satu
molekul atau satu spin yang mengumumkan revolusi ini: semuanya bersifat
\emph{kolektif} --- $10^{23}$ pelaku, masing-masing berinteraksi dengan
beberapa tetangga, bersekongkol mengubah fase masyarakatnya sekaligus.
Jilid Tahun 1 mengkatalogkan peralihannya; bab ini menjelaskan
mekanismenya dengan perkakas lima bab terakhir: persaingan \omterm{prop:b3:canonical-ensemble:machine}{energi bebas},
\omterm{def:b3:phase-transitions:order}{parameter keteraturan}, \omterm{def:b3:phase-transitions:order}{pematahan kesetangkupan}, dan penemuan menakjubkan
bahwa di dekat \omterm{prop:b3:phase-transitions:vdw}{titik genting} rincian mikroskopisnya berhenti berarti ---
sebuah magnet dan sebuah fluida yang mendidih berbagi \omterm{prop:b3:phase-transitions:landau}{eksponen genting}
yang sama. Gagasan bab ini kini berjalan dari panci presto sampai ke
adikonduktor sampai ke medan Higgs
\cref{ch:b3:particle-physics}.

\section{Parameter keteraturan dan persaingan energi bebas}

\begin{definition}[Fase dan parameter keteraturan]\label{def:b3:phase-transitions:order}
\index{peralihan fase}\index{parameter keteraturan}\index{pematahan kesetangkupan}
Sebuah \emph{peralihan fase} adalah perubahan tak mulus pada keadaan
kesetimbangan sebuah sistem ketika parameter kendalinya ($T$, $P$, $B$)
melintasi sebuah batas. Pembukunya adalah \emph{parameter keteraturan}:
yakni besaran yang nol dalam fase yang tak teratur dan tak nol dalam fase
yang teratur --- yaitu magnetisasi $m$ sebuah feromagnet, selisih
kerapatan $\rho_{\text{cair}} - \rho_{\text{gas}}$ sebuah fluida, bagian
kondensat \cref{ch:b3:quantum-statistics}. Peralihan datang dalam
dua jenis: \emph{orde pertama} (parameter keteraturannya melompat; ada
kalor laten; fasenya berada bersama --- pendidihan, peleburan) dan
\emph{sinambung} (parameter keteraturannya tumbuh dari nol; tanpa kalor
laten; dengan fluktuasi yang liar --- titik Curie, \omterm{prop:b3:phase-transitions:vdw}{titik genting}, helium
adiluncur). Ketika keadaan yang teratur harus \emph{memilih} di antara
pilihan yang setara --- kutub utara sebuah magnet harus menunjuk ke suatu
arah --- peralihannya \emph{mematahkan sebuah kesetangkupan} yang dihormati
hukum yang mendasarinya: yakni tema yang akan kembali, dengan taruhan
tertinggi, bersama mekanisme Higgs.
Mesin di balik tiap perkaranya adalah \omterm{prop:b3:canonical-ensemble:machine}{energi bebas} $F = E - TS$
(\cref{prop:b3:canonical-ensemble:machine}): energi menyukai keteraturan,
entropi menyukai ketakteraturan, dan $T$ menetapkan nilai tukarnya ---
batas fase adalah tempat buku besarnya seri.
\end{definition}

\section{Peralihan cair--gas: van der Waals}

\begin{proposition}[Isoterm van der Waals]\label{prop:b3:phase-transitions:vdw}
\index{persamaan van der Waals}\index{titik genting}
Persamaan gas nyata jilid Tahun 1,
\[
\Big(P + \frac{aN^2}{V^2}\Big)(V - Nb) = Nk_{\text{B}}T ,
\]
menyandikan tarikan ($a$) dan teras kerasnya ($b$). Di atas suhu
genting $T_{\text{c}} = 8a/27bk_{\text{B}}$ isotermnya bersifat
monoton: yakni satu fase fluida. Di bawahnya isotermnya menumbuhkan
gelung yang memuat rentang yang \emph{tak mantap} secara mekanis
($\partial P/\partial V >
0$): fluidanya di sana membelah menjadi cairan dan gas yang berada
bersama, pada tekanan yang ditetapkan oleh kesamaan \omterm{thm:b3:grand-canonical:distribution}{potensial kimianya}
(yakni aturan luas sama Maxwell). Daerah kebersamaannya menutup pada \omterm{prop:b3:phase-transitions:vdw}{titik
gentingnya} ($P_{\text{c}}, V_{\text{c}}, T_{\text{c}}$), tempat cairan
dan gasnya menjadi \emph{tak terbedakan}: ketika didekati, selisih
kerapatannya lenyap, kemampatannya menyebar, dan fluktuasi kerapatannya
tumbuh sampai berukuran optik --- yakni keopalan genting
\cref{pb:b3:perturbation-theory:1}.
\end{proposition}
\admitted

\begin{proposition}[Clausius--Clapeyron]\label{prop:b3:phase-transitions:clapeyron}
\index{hubungan Clausius--Clapeyron}
Sepanjang garis kebersamaan orde pertama mana pun, kesamaan \omterm{thm:b3:grand-canonical:distribution}{potensial
kimia} pada kedua sisinya
(\cref{def:b3:grand-canonical:mu}) memaksa kemiringan garisnya menjadi
\[
\frac{\dd P}{\dd T} = \frac{L}{T\,\Delta v} ,
\]
dengan $L$ sebagai kalor latennya dan $\Delta v$ sebagai perubahan volume
tiap partikelnya (atau tiap kilogramnya, secara taat asas). Segalanya
tentang garis kebersamaan termuat dalam satu rumus ini: garis pendidihan
air memanjat pada $\approx\qty{3.6}{kPa/K}$ di dekat \qty{100}{\celsius}
(yakni panci presto, memasak di ketinggian ---
\cref{pb:b3:phase-transitions:1}); dan garis \emph{peleburan} air
memiring ke belakang ($\Delta v < 0$: es mengapung), sehingga tekanan
melelehkan es --- yakni kelangkaan di antara zat, dengan akibat dari
gletser sampai ke gelanggang seluncur.
\end{proposition}

\begin{proof}
Pada garisnya, $\mu_1(T, P) = \mu_2(T, P)$; bergeraklah sepanjangnya:
$-s_1\dd T + v_1\dd P = -s_2\dd T + v_2\dd P$ (tiap partikelnya, dengan
memakai $\dd\mu = -s\,\dd T + v\,\dd P$). Selesaikanlah untuk $\dd P/\dd T$
lalu pakailah $L = T(s_2 - s_1)$.
\end{proof}

\section{Feromagnetnya: teori medan rata-rata}

\begin{theorem}[Peralihan Ising medan rata-rata]\label{thm:b3:phase-transitions:meanfield}
\index{model Ising}\index{teori medan rata-rata}\index{suhu Curie}
Modelkanlah sebuah magnet sebagai spin $s_i = \pm1$ pada sebuah kisi,
dengan tetangganya digandeng oleh energi pertukaran $-J\,s_is_j$
(\cref{exo:b3:identical-particles:12}), tiap spinnya bertetangga
sebanyak $q$. Gantikanlah tetangga tiap spinnya dengan rata-ratanya $m =
\langle s\rangle$ (yakni hampiran \emph{medan rata-rata}): sebuah spin
lalu duduk dalam medan efektif $qJm$, dan
\cref{exo:b3:canonical-ensemble:11} memberikan syarat
ketaatasasan dirinya
\[
m = \tanh\!\Big(\frac{T_{\text{c}}}{T}\,m\Big) , \qquad
k_{\text{B}}T_{\text{c}} = qJ .
\]
Di atas $T_{\text{c}}$ satu-satunya penyelesaiannya adalah $m = 0$; di
bawahnya dua penyelesaian setangkup $\pm m(T)$ muncul dan menjadi yang
mantap: magnetnya \emph{harus} memagnet, dengan memilih arah secara acak
--- yakni \omterm{def:b3:phase-transitions:order}{pematahan kesetangkupan} spontan. Di dekat $T_{\text{c}}$, $m \propto
(T_{\text{c}} - T)^{1/2}$, dan di atasnya kerentanannya menaati
hukum Curie--Weiss $\chi \propto 1/(T - T_{\text{c}})$:
yang menyebar pada peralihannya --- yakni magnet yang tak terhingga baik hati.
\end{theorem}

\begin{proof}[Bukti sebagian]
Sebuah spin dalam medan $B_{\text{eff}} = qJm/\mu$ memiliki $\langle s\rangle =
\tanh(\beta qJm)$; menuntut $\langle s\rangle = m$ memberikan persamaan
yang dinyatakan itu. Secara grafis: garis $y = m$ memotong kurva $y =
\tanh(T_{\text{c}}m/T)$ hanya pada titik asalnya ketika kemiringan awal
kurvanya $T_{\text{c}}/T < 1$, dan juga pada $\pm m^* \neq 0$ ketika
$T < T_{\text{c}}$. Menjabarkan $\tanh$ untuk $m$ yang kecil
memberikan $m^2 = 3(1 - T/T_{\text{c}})$, yakni pertumbuhan akar
kuadratnya; menambahkan medan terapan yang kecil lalu menjabarkannya
serupa memberikan Curie--Weiss (\cref{exo:b3:phase-transitions:6}).
\end{proof}

\section{Sudut pandang Landau dan kesemestaan}

\begin{proposition}[Teori Landau]\label{prop:b3:phase-transitions:landau}
\index{teori Landau}\index{eksponen genting}\index{kesemestaan}
Di dekat peralihan yang sinambung, jabarkanlah energi bebasnya dalam
\omterm{def:b3:phase-transitions:order}{parameter keteraturan} yang kecil, dengan hanya mempertahankan apa yang
diizinkan kesetangkupannya ($m \to -m$ di sini):
\[
F(m) = a\,(T - T_{\text{c}})\,m^2 + b\,m^4 , \qquad a, b > 0 .
\]
Di atas $T_{\text{c}}$: satu sumur tunggal pada $m = 0$. Di bawahnya
titik asalnya menjadi puncak dan dua sumur setangkup muncul pada $m = \pm
\sqrt{a(T_{\text{c}} - T)/2b}$ --- yakni akar kuadrat yang sama seperti
medan rata-ratanya, kini dari kesetangkupannya semata. Karena itulah
sistem yang sangat berbeda berperilaku sama di dekat \omterm{prop:b3:phase-transitions:vdw}{titik gentingnya}:
yakni \emph{kesemestaan}. Eksponen terukurnya (misalnya $m \propto |T -
T_{\text{c}}|^\beta$ dengan $\beta \approx 0.33$ untuk fluida
\emph{maupun} magnet sesumbu) berbeda dari nilai medan rata-ratanya $1/2$
karena di dekat $T_{\text{c}}$ fluktuasinya mengamuk pada segala skala ---
penjinakannya, yakni grup renormalisasi (Wilson, Nobel 1982),
menunjukkan bahwa hanya dimensi dan kesetangkupannya, tak pernah rincian
mikroskopisnya, yang menentukan eksponennya. Gambaran sumur gandanya
sendiri berkarier melampaui magnet: berikanlah kepada \omterm{def:b3:phase-transitions:order}{parameter
keteraturannya} peran sebuah medan yang mengisi ruang, dan sumur pematah
kesetangkupannya adalah mekanisme Higgs
(\cref{ch:b3:particle-physics}).
\end{proposition}
\admitted

\begin{method}[Membaca sebuah peralihan]\label{met:b3:phase-transitions:method}
(1) Kenalilah \omterm{def:b3:phase-transitions:order}{parameter keteraturannya} dan apakah ia melompat
(orde pertama: kalor laten, kebersamaan, histeresis, pengintian) atau
tumbuh secara sinambung (\omterm{prop:b3:phase-transitions:vdw}{titik genting}: penyebaran, keopalan,
kesemestaan). (2) Garis orde pertama: Clausius--Clapeyron untuk
kemiringannya; kesamaan $\mu$ untuk kebersamaannya. (3) Yang sinambung:
medan rata-rata lebih dahulu ($\tanh$ atau Landau) untuk bentuk diagram
fasenya; percayailah \emph{topologinya}, bukan eksponennya. (4) Di bawah
$T_{\text{c}}$, ingatlah pilihannya: kesetangkupan yang patah berarti
ranah, cacat dan kebergantungan pada riwayat. (5) Perkiraan:
$k_{\text{B}}T_{\text{c}} \sim$
(gandengan tiap tetangganya) $\times$ (cacah tetangganya).
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:1}
Golongkanlah (orde pertama atau sinambung), sambil menyebutkan \omterm{def:b3:phase-transitions:order}{parameter
keteraturannya} dan kesetangkupan yang patah bila ada: (a) pendidihan pada
\qty{1}{atm}; (b) titik Curie; (c) titik $\lambda$ adiluncurnya; (d)
pendidihan \emph{persis pada} \omterm{prop:b3:phase-transitions:vdw}{titik gentingnya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:2}
Kalor laten sebagai entropi. (a) Air: $L_{\text{vap}} =
\qty{2.26e6}{J/kg}$ pada \qty{373}{K}: hitunglah $\Delta S$ tiap
molekulnya dalam satuan $k_{\text{B}}$. (b) Hal yang sama untuk peleburan
($L_{\text{fus}} = \qty{3.3e5}{J/kg}$ pada \qty{273}{K}). (c) Sebagian
besar cairan menguap dengan $\Delta S \approx 10$--$11\,k_{\text{B}}$
(yakni aturan Trouton): apa yang kira-kira dicacah bilangan itu (yakni
$\ln$ nisbah volume yang mana)? (d) Mengapa lompatan entropi peleburannya
jauh lebih kecil daripada lompatan pendidihannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:3}
Tetapan genting van der Waals ($V_{\text{c}} = 3Nb$,
$k_{\text{B}}T_{\text{c}} = 8a/27b$, $P_{\text{c}} = a/27b^2$ ---
terimalah). (a) Tunjukkanlah $P_{\text{c}}V_{\text{c}}/Nk_{\text{B}}T_{\text{
c}} = 3/8$, yakni bilangan semesta bagi modelnya; nilai terukurnya
bergerombol di dekat $0.29$ --- berilah ulasan. (b) Untuk CO$_2$
($T_{\text{c}} = \qty{304}{K}$, $P_{\text{c}} = \qty{73.8}{bar}$):
tariklah $a$ dan $b$. (c) Dari $b$, perkirakanlah garis tengah
molekulnya. (d) Mengapa CO$_2$ bersuhu ruang mencair di bawah
tekanan sedangkan N$_2$ bersuhu ruang ($T_{\text{c}} =
\qty{126}{K}$) tak pernah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:phase-transitions:4}
\omterm{def:b3:phase-transitions:order}{Parameter keteraturan}, secara cepat: berikanlah satu untuk (a) sebuah
feromagnet; (b) peralihan cair--gas; (c) \omterm{thm:b3:quantum-statistics:bec}{kondensasi Bose--Einstein}; (d)
sebuah antiferomagnet (dua subkisi --- besaran apa yang teratur sementara
magnetisasi totalnya tetap nol?).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:5}
Medan rata-rata dengan tangan. (a) Turunkanlah $m = \tanh(\beta qJm)$
dari sebuah spin dalam medan rata-rata $q$ tetangganya. (b) Tunjukkanlah
bahwa penyelesaian tak nol ada persis ketika $T < qJ/k_{\text{B}}$. (c)
Jabarkanlah sampai orde ketiga lalu turunkanlah $m(T) \approx \sqrt{3(1 -
T/T_{\text{c}})}$. (d) Besi: $T_{\text{c}} = \qty{1043}{K}$, $q =
8$: tariklah $J$ dalam eV lalu periksalah skala pertukaran
\cref{exo:b3:identical-particles:12}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:6}
Curie--Weiss. Tambahkanlah medan yang kecil: $m = \tanh[\beta(qJm + \mu B)]$.
(a) Lelurusanlah di atas $T_{\text{c}}$ lalu turunkanlah $\chi = \partial
m/\partial B \propto 1/(T - T_{\text{c}})$. (b) Bandingkanlah dengan
hukum Curie spin bebasnya: apa yang diberikan pemplotan $1/\chi$ terhadap $T$
pada tiap perkaranya, dan bagaimana titik potongnya mendiagnosis gandengan
feromagnetik? (c) Garis $1/\chi$ nikel mengenai nol pada \qty{631}{K}:
tafsirkanlah. (d) Untuk sebuah antiferomagnet titik potongnya bernilai
\emph{negatif}: tanda $J$ yang mana yang dikhianatinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:7}
Clausius--Clapeyron dalam kerja. (a) Air pada \qty{100}{\celsius}
($\Delta v = \qty{1.67}{m^3/kg}$): hitunglah $\dd P/\dd T$. (b)
Pakailah $\dd\ln P/\dd T = Lm_{\text{w}}/k_{\text{B}}T^2$ untuk
memperkirakan suhu didihnya pada $P = \qty{0.7}{atm}$ (yakni puncak
\qty{3000}{m}). (c) Peleburan: dengan $\Delta v = \qty{-9.1e-5}{m^3/kg}$ dan
$L = \qty{3.3e5}{J/kg}$, hitunglah $\dd T/\dd P$ beserta tekanan
untuk menurunkan titik lebur es sebesar \qty{5}{K}. (d) Seorang
peluncur \qty{70}{kg} di atas \qty{30}{mm^2} bilahnya: hitunglah tekanannya dan
geseran titik leburnya --- apakah peleburan oleh tekanan menjelaskan
seluncur? (Apa yang menjelaskannya: gesekan dan pralelehan permukaan.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:8}
Perhitungan Landau. (a) Minimumkanlah $F = a(T - T_{\text{c}})m^2 +
bm^4$ di bawah $T_{\text{c}}$. (b) Hitunglah $F$ pada minimumnya lalu
tunjukkanlah bahwa kapasitas kalornya memiliki \emph{lompatan} berhingga $\Delta C =
a^2T_{\text{c}}/2b$ pada peralihannya. (c) Sketsalah $C(T)$:
anak tangga medan rata-ratanya berbanding puncak nyaris logaritmik yang
terukur pada titik $\lambda$ helium --- apa yang ditambahkan fluktuasi
yang sebenarnya? (d) Tambahkanlah suku medan $-hm$: tunjukkanlah bahwa
peralihannya membulat --- kesetangkupannya harus eksak agar ia dapat patah.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:phase-transitions:9}
Pengintian. Sebuah tetesan berjejari $r$ dalam uap yang lewat jenuh
memperoleh \omterm{prop:b3:canonical-ensemble:machine}{energi bebas} limbak $-\tfrac43\pi r^3\,n\,\delta\mu$ tetapi
membayar tegangan permukaan $4\pi r^2\sigma$. (a) Tunjukkanlah bahwa
penghalangnya memuncak pada
$r^* = 2\sigma/n\,\delta\mu$. (b) Jelaskanlah: mengapa uap lewat jenuh
dan cairan lewat panas yang bersih bertahan
(kelengaian), dan apa yang disediakan debu, ion dan goresan?
(c) Bilik awan (yakni pendeteksi leluhur
\cref{ch:b3:particle-physics}) dan bilik gelembung berjalan \emph{atas}
fisika ini: peran mana yang dimainkan jejak ion partikel yang lewat? (d)
Mengapa ``kerikil pendidih'' dimasukkan ke dalam labu laboratorium?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:phase-transitions:10}
Tanpa peralihan dalam satu dimensi. Tinjaulah rantai Ising satu dimensi
berisi $N$ spin, semuanya ke atas, lalu sisipkanlah satu ``dinding ranah''
(yakni semua spin di balik suatu ikatan dibalik). (a) Ongkos energi
dindingnya: $2J$, tak bergantung kedudukannya. (b) Perolehan entropinya:
dindingnya dapat duduk pada salah satu dari $\sim N$ ikatannya ---
$\Delta S = k_{\text{B}}\ln N$. (c)
Tunjukkanlah $\Delta F = 2J - k_{\text{B}}T\ln N < 0$ untuk $T > 0$ mana pun pada
$N$ yang besar: dindingnya selalu berkembang biak, dan keteraturan
jangkauan panjangnya mustahil --- \omterm{thm:b3:phase-transitions:meanfield}{model Ising} satu dimensi memiliki
$T_{\text{c}} = 0$. (d) Dalam dua dimensi, sebuah dinding adalah garis
sepanjang $\ell$ yang berongkos $2J\ell$ tetapi dengan
$\sim3^\ell$ bentuk: kerjakanlah ulang perkiraannya lalu tunjukkanlah
bahwa keteraturannya \emph{selamat} di bawah $T_{\text{c}}$ yang
berhingga --- yakni penalaran Peierls di balik nilai eksak Onsager
$2.269\,J/k_{\text{B}}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:phase-transitions:11}
Kesemestaan yang diaudit. Medan rata-rata meramalkan $\beta = 1/2$, $\gamma
= 1$ ($\chi \sim |t|^{-\gamma}$), $\delta = 3$ ($m \sim h^{1/3}$
pada $T_{\text{c}}$). Terukur, untuk \omterm{prop:b3:phase-transitions:vdw}{titik genting} cair--gas
\emph{maupun} magnet sesumbu berdimensi tiga: $\beta
\approx 0.326$, $\gamma \approx 1.237$, $\delta \approx 4.79$.
(a) Apa yang membenarkan membandingkan sebuah fluida dengan sebuah magnet
sama sekali (apa yang dibagi bersama \omterm{def:b3:phase-transitions:order}{parameter keteraturannya} --- yakni
satu pilihan tanda)? (b)
Mengapa medan rata-ratanya gagal di dekat $T_{\text{c}}$ (apa yang
menyebar, sehingga membatalkan pengabaian fluktuasinya)? (c) Mengapa ia
gagal \emph{lebih sedikit} dalam dimensi yang lebih tinggi (cacah
tetangganya berbanding jangkauan fluktuasinya; medan rata-ratanya menjadi
eksak di atas empat
dimensi)? (d) Nyatakanlah pelajaran grup renormalisasinya dalam dua
kalimat: apa yang dirata-ratakan skala demi skala, dan mengapa hanya
kesetangkupan dan dimensi yang selamat dari perata-rataan itu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:phase-transitions:12}
Dari magnet ke Higgs. Naikkanlah \omterm{def:b3:phase-transitions:order}{parameter keteraturannya} menjadi sebuah
medan $\phi(\vect r)$ berkerapatan energi Landau $\alpha(T -
T_{\text{c}})\phi^2 + b\phi^4$ ditambah suku gradien. (a) Di bawah
$T_{\text{c}}$ medannya duduk pada $\pm\phi_0$: getaran kecil
\emph{sepanjang} sumurnya memiliki kelengkungan pemulih --- hitunglah
$F''(\phi_0)$ lalu tafsirkanlah, dalam bahasa medan, sebagai sebuah
\emph{massa}. (b) Untuk \omterm{def:b3:phase-transitions:order}{parameter keteraturan} yang \emph{kompleks} (yakni
lingkaran minimum --- ``topi Meksiko''), getaran \emph{mengelilingi}
tepinya tak berongkos energi potensial: yakni ragam tak bermassa ---
sebutkanlah padanan benda rapatnya (gelombang spin, fonon adiluncur). (c)
Dalam teori elektrolemah, vakumnya sendiri duduk dalam sumur pematah
kesetangkupan semacam itu; partikel memperoleh massa dari nilai tak nol
medannya, dan getaran radialnya \emph{adalah} boson
Higgs: petakanlah tiap unsurnya ke (a)--(b). (d) Dalam satu kalimat: apa
yang diajarkan fisika benda rapat kepada fisika partikel di sini (yakni
ekspor \omterm{def:b3:phase-transitions:order}{pematahan kesetangkupan} lewat Anderson)?
\end{exercise}

\section{Soal: Diagram fase dapur}

\begin{problem}[{Soal akhir pekan --- panci presto, teh gunung,
dan akhir pendidihan}]\label{pb:b3:phase-transitions:1}
Diagram fase air tergantung, tanpa disadari, di atas tiap kompor. Soal
ini menyusuri ketiga garisnya dengan Clausius--Clapeyron di tangan ---
dari mengapa panci presto memangkas separuh waktu memasak, lewat mengapa
teh mengecewakan di ketinggian, sampai ke dunia aneh di balik \omterm{prop:b3:phase-transitions:vdw}{titik
gentingnya} tempat pendidihan sendiri berhenti ada. Data: $L_{\text{vap}}
= \qty{2.26e6}{J/kg}$, $L_{\text{fus}} = \qty{3.34e5}{J/kg}$;
uap air pada \qty{373}{K}, \qty{1}{atm}: $\Delta v_{\text{vap}}
= \qty{1.67}{m^3/kg}$; peleburan: $\Delta v = \qty{-9.1e-5}{m^3/kg}$;
massa molekulnya $m_{\text{w}} = \qty{3.0e-26}{kg}$; \omterm{prop:b3:phase-transitions:vdw}{titik genting}:
$T_{\text{c}} = \qty{647}{K}$, $P_{\text{c}} = \qty{221}{bar}$.

\medskip\noindent\textbf{Bagian I --- Garis pendidihannya.}
\begin{enumerate}
  \item Apa yang mendefinisikan ``pendidihan'' pada tekanan tertentu (dua
        \omterm{thm:b3:grand-canonical:distribution}{potensial kimia} yang mana yang seri)?
  \item Hitunglah kemiringan $\dd P/\dd T$ garis penguapannya
        pada \qty{100}{\celsius}.
  \item Sebuah panci presto menahan \qty{2}{atm}: integralkanlah
        kemiringannya (atau pakailah bentuk eksponensialnya dengan
        $Lm_{\text{w}}/
        k_{\text{B}} \approx \qty{4900}{K}$) untuk mencari suhu
        didihnya.
  \item Dengan aturan dapur Arrhenius
        \cref{exo:b3:canonical-ensemble:9} (lajunya berlipat dua tiap
        \qty{10}{K}), perkirakanlah percepatannya terhadap memasak
        dengan panci terbuka.
  \item Di La Paz (\qty{3600}{m}, $P \approx \qty{0.65}{atm}$):
        carilah suhu didihnya, dan mengapa tehnya menyeduh dengan buruk.
  \item Mengapa menambahkan garam menaikkan titik didihnya ($\mu$ siapa
        yang diturunkan zat terlarutnya --- ingatlah kembali
        \cref{exo:b3:grand-canonical:8})?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Garis peleburannya.}
\begin{enumerate}[resume]
  \item Hitunglah $\dd T/\dd P$ untuk peleburannya lalu perhatikanlah tandanya.
  \item Pengamatan sehari-hari yang mana yang menyandikan $\Delta v < 0$
        (apa yang dilakukan es dalam gelas Anda), dan mengapa
        kejanggalan itu langka di antara zat?
  \item Sebuah gletser setebal \qty{300}{m}: carilah tekanan pada dasarnya
        beserta penurunan titik leburnya --- apakah peleburan oleh
        tekanan di dasarnya berarti bagi aliran gletsernya?
  \item Tinjaulah kembali peluncur
        \cref{exo:b3:phase-transitions:7}(d): apa yang sebenarnya
        melumasi bilahnya?
  \item Seandainya es tenggelam (dengan kemiringan positif, seperti
        hampir tiap zat padat lainnya), apa yang akan dilakukan musim
        dingin terhadap danau dan ikannya --- telusurilah rantai akibatnya.
  \item Kemiringan negatifnya berakhir pada tekanan tinggi tempat bentuk
        kristal es yang lain mengambil alih (es III, V, VI): apa yang
        dikatakan keberadaan selusin es tentang lanskap \omterm{prop:b3:canonical-ensemble:machine}{energi bebas} satu
        molekul yang sederhana?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Ujung garisnya.}
\begin{enumerate}[resume]
  \item Garis pendidihannya berakhir pada ($T_{\text{c}},
        P_{\text{c}}$): apa yang terjadi pada kalor laten dan selisih
        kerapatannya ketika ujungnya didekati?
  \item Di balik \omterm{prop:b3:phase-transitions:vdw}{titik gentingnya} orang dapat membawa air dari
        ``bergaya cairan'' ke ``bergaya gas'' \emph{tanpa} peralihan
        sama sekali: lukiskanlah lintasannya dalam bidang $(P, T)$.
  \item Ketika \omterm{prop:b3:phase-transitions:vdw}{titik gentingnya} didekati, fluidanya menjadi seperti susu:
        kaitkanlah keopalan ini dengan
        \cref{exo:b3:microcanonical-ensemble:12} dan
        \cref{pb:b3:perturbation-theory:1}.
  \item CO$_2$ superkritis ($T_{\text{c}} = \qty{304}{K}$,
        $P_{\text{c}} = \qty{74}{bar}$) melarutkan seperti cairan
        dan menembus seperti gas: sebutkanlah kegunaan industri yang
        menghilangkan kafeina dari kopi, dan mengapa pelarutnya tak
        meninggalkan sisa.
  \item Air superkritis (dalam ketel pembangkit listrik di atas
        \qty{221}{bar}): mengapa melenyapkan pendidihan --- gelembung,
        selaput, kehangusan --- menarik bagi para insinyur ketel?
  \item Lubang hidrotermal laut dalam memancarkan air bersuhu
        \qty{400}{\celsius} yang tidak mendidih: periksalah terhadap
        tekanan setempatnya ($\sim\qty{300}{bar}$ pada \qty{3}{km}) dan
        diagram fasenya.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Membaca seluruh petanya.}
\begin{enumerate}[resume]
  \item Sketsalah (dengan kata-kata) diagram $(P, T)$ air: ketiga
        garisnya, titik tripelnya ($\qty{0.01}{\celsius}$,
        \qty{611}{Pa}), dan \omterm{prop:b3:phase-transitions:vdw}{titik gentingnya}.
  \item Di bawah \qty{611}{Pa}, es menyublim tanpa melebur: perkakas
        yang mana (pengeringan beku) dan tudung kutub planet yang mana
        (Mars, CO$_2$ dan H$_2$O) yang hidup dari kenyataan ini?
  \item Mengapa titik tripel air --- yang tepat
        \qty{273.16}{K} menurut definisi sampai 2019 --- menjadi baku
        suhu yang begitu baik (berapa fase yang berada bersama memancang
        berapa \omterm{def:b3:lagrangian-mechanics:coordinates}{derajat kebebasan})?
  \item Tiap garis petanya adalah peralihan orde pertama; hanya titik
        ujungnya yang genting: nyatakanlah kembali pembedaannya dengan
        \omterm{def:b3:phase-transitions:order}{parameter keteraturan} dan kalor laten.
  \item Garis embun udara lembap adalah fisika yang sama yang dijalankan
        terbalik: jelaskanlah embun, kabut dan uap pagi pada jendela yang
        dingin dengan satu kalimat pelintasan $\mu$.
  \item Tiap tutup panci adalah sebuah percobaan: tetesan mengembun
        padanya lalu turun kembali sebagai hujan --- kenalilah kedua
        \omterm{def:b3:phase-transitions:order}{peralihan fase} daur air mini itu beserta ulang-alik kalor laten
        yang membuat panci bertutup mendidih lebih cepat.
  \item Ringkaslah hasil bernama itu: satu rumus kemiringan, $\dd P/
        \dd T = L/T\Delta v$, menjelaskan \qty{121}{\celsius} panci
        prestonya, teh \qty{88}{\celsius} pegunungannya, es yang
        mengapung mundur menaiki garis leburnya, dan ketel superkritis
        tempat pendidihan berakhir pada
        $\qty{647}{K}, \qty{221}{bar}$.
\end{enumerate}
\end{problem}
