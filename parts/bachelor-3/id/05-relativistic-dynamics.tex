\chapter{Dinamika Relativistik}\label{ch:b3:relativistic-dynamics}

Pada 1964 William Bertozzi merekam elektron yang melesat menuruni
pemercepat linear: tiap semburannya diberi energi terukur, dan lajunya
dicatat sepanjang \qty{8.4}{m} penerbangan. Newton meramalkan bahwa
elektron \qty{15}{MeV} seharusnya terbang delapan kali laju cahaya;
pencatat waktunya mengatakan $0.9999c$ dan menolak melampauinya,
sekeras apa pun elektronnya didorong. Kinematika mengatakan kepada kita
pada bab lalu bahwa $c$ adalah sebuah batas; dinamika kini harus
menjelaskan apa yang terjadi pada dorongannya --- ke mana usahanya pergi
ketika lajunya tidak lagi tumbuh. Jawabannya menata ulang mekanika di
sekitar momentum baru dan energi baru, yang disatukan dalam persamaan
paling termasyhur dalam fisika: energi yang tersimpan adalah massa, dan
massa adalah energi yang bermukim, $E = mc^2$. Bab ini membangun
dinamikanya, lalu mempekerjakannya di tempat ia hidup sehari-hari ---
peluruhan radioaktif, tumbukan partikel, dan pemercepat yang mengubah
gerak menjadi materi baru.

\section{Momentum dan energi, dibangun ulang}

\begin{remark}[Mengapa $m\vect v$ harus pergi]\label{rem:b3:relativistic-dynamics:why}
Momentum klasik $m\vect v$ kekal dalam setiap kerangka \emph{jika}
kecepatannya bertransformasi menurut Galileo. Kenyataannya tidak
(\cref{prop:b3:relativistic-kinematics:addition}): tumbukan yang
mengekalkan $\sum m\vect v$ dalam satu kerangka gagal mengekalkannya
dalam kerangka lain, sehingga momentum lama itu tidak dapat menjadi
hukum alam --- paling banter bayangannya pada laju rendah.
Perbaikannya adalah mengukur kecepatannya dengan jam partikel itu
\emph{sendiri}: $\dd\vect r/\dd\tau =
\gamma\,\vect v$ bertransformasi dengan rapi, dan $m\,\dd\vect r/\dd\tau$
ternyata kekal dalam semua kerangka sekaligus.
\end{remark}

\begin{definition}[Momentum dan energi relativistik]\label{def:b3:relativistic-dynamics:momentum-energy}
\index{momentum relativistik}\index{energi relativistik}\index{energi diam}
Partikel bermassa $m$ dan berkecepatan $\vect v$ ($\gamma = 1/\sqrt{1 -
v^2/c^2}$) membawa momentum dan energi
\[
\vect p = \gamma m\vect v , \qquad
E = \gamma mc^2 .
\]
Dalam keadaan diam, $E_0 = mc^2$: itulah \emph{energi diam}. Energi
kinetiknya adalah tambahan dari geraknya, $E_k = E - mc^2 = (\gamma - 1)mc^2$,
yang untuk $v
\ll c$ menyusut menjadi $\tfrac12 mv^2$ yang sudah dikenal (jabarkanlah
$\gamma$). Dalam setiap proses terkucil, $\vect p$ total dan $E$ total
--- termasuk energi diamnya --- kekal.
\end{definition}

\begin{theorem}[Hubungan energi--momentum]\label{thm:b3:relativistic-dynamics:relation}
\index{hubungan energi--momentum}
Untuk sembarang partikel,
\[
E^2 = (pc)^2 + (mc^2)^2 ,
\qquad
\vect v = \frac{\vect p\,c^2}{E} :
\]
gabungan $E^2 - p^2c^2 = m^2c^4$ bernilai sama dalam setiap kerangka
--- ia bagi energi dan momentum sebagaimana selang bagi waktu dan
ruang. Dua rezimnya: pada laju rendah, $E \approx mc^2 + p^2/2m$; pada
energi tinggi ($E \gg mc^2$), $E \approx pc$ dan $v \to c$: mendorong
lebih keras menambah energi dan momentum, hampir tak menambah laju.
\end{theorem}

\begin{proof}
$E^2 - p^2c^2 = \gamma^2m^2c^4(1 - v^2/c^2) = m^2c^4$; dan $pc^2/E =
\gamma mvc^2/\gamma mc^2 = v$. Keinvariannya terhadap kerangka menyusul
karena $m$ dan $c$ tidak bergantung kerangka.
\end{proof}

\begin{proposition}[Partikel tak bermassa]\label{prop:b3:relativistic-dynamics:massless}
\index{foton!momentum}
Partikel bermassa nol mempunyai $E = pc$ dan bergerak tepat pada $c$
dalam setiap kerangka; ia tidak dapat diperlambat, hanya digeser merah.
Foton adalah kasus bakunya: $E = h\nu$ dan $p = h\nu/c = h/\lambda$ ---
yakni hubungan Planck--Einstein pada jilid Tahun 1, yang kini terlihat
sebagai sudut $m = 0$ dari mekanika.
\end{proposition}

\begin{proof}
Ambillah $m = 0$ dalam \cref{thm:b3:relativistic-dynamics:relation}: $E = pc$
dan $v = pc^2/E = c$.
\end{proof}

\begin{example}[Orde besar]\label{ex:b3:relativistic-dynamics:orders}
Fisika partikel menghitung energi dalam elektronvolt dan massa dalam
$\unit{MeV}/c^2$: elektron $0.511$, proton $938.3$, muon $105.7$, pion
$139.6$. Elektron berenergi kinetik \qty{1}{MeV} mempunyai $\gamma =
2.96$ dan $v = 0.94c$ --- sudah sepenuhnya relativistik, dan itulah
sebabnya elektronika tetap klasik (\unit{eV}) sedangkan fisika inti
tidak (\unit{MeV}). Proton LHC \qty{6.8}{TeV} mempunyai $\gamma = 7250$
dan tertinggal dari foton hanya \qty{2.9}{m/s}.
\end{example}

\section{Massa adalah energi yang bermukim}

\begin{theorem}[Kesetaraan massa--energi]\label{thm:b3:relativistic-dynamics:emc2}
\index{kesetaraan massa--energi}\index{defek massa}
Massa sebuah benda adalah energi total isinya, dibagi $c^2$, yang
diukur dalam kerangka diamnya: panaskanlah sebuah benda, putarlah
pegasnya, eksitasikanlah atomnya, dan beratnya bertambah sebesar
$\Delta E/c^2$; ikatkanlah bagiannya menjadi satu dan beratnya
\emph{berkurang} sebesar energi ikat dibagi $c^2$ --- yakni \emph{\omterm{thm:b3:relativistic-dynamics:emc2}{defek
massa}}. Sebaliknya, \omterm{def:b3:relativistic-dynamics:momentum-energy}{energi diam} dapat dilepaskan: setiap kali jumlah
massa akhirnya kurang dari jumlah massa awalnya, selisih $\Delta m\,c^2$
muncul sebagai energi kinetik atau radiasi.
\end{theorem}
\admitted

\begin{example}[Pembukuan $c^2$]\label{ex:b3:relativistic-dynamics:ledger}
$c^2 = \qty{9e16}{J/kg}$: selisih massa satu gram setara dengan
\qty{9e13}{J}, yaitu energi sebuah kota kecil selama sehari. Ikatan
kimia (beberapa eV per molekul) menggeser massa sebesar satu per
$10^{10}$ --- selamanya tak tertimbang, dan itulah sebabnya kimia tak
pernah menyadarinya. Ikatan inti menggesernya hampir $1\%$: helium
berbobot $0.7\%$ lebih ringan daripada empat bahan hidrogennya, dan
bagian yang hilang itu, yang mengalir dari Matahari sebagai cahaya,
merugikannya $\Delta m = L_\odot/c^2 =
\qty{4.3e9}{kg}$ tiap sekon --- empat juta ton sinar matahari.
Pemusnahan melunasi seluruh rekeningnya: elektron yang bertemu
positron tak meninggalkan apa pun selain dua foton \qty{511}{keV},
yakni isyarat yang dipakai pemindai PET untuk mengamati otak yang hidup.
\end{example}

\begin{remark}[Apa arti ``perubahan'' itu]\label{rem:b3:relativistic-dynamics:conversion}
Tak ada materi yang ``berubah menjadi'' energi: energi totalnya kekal
sejak semula. Yang berubah adalah \emph{tempat mukimnya} --- dari \omterm{def:b3:relativistic-dynamics:momentum-energy}{energi
diam}, yang berbobot, menjadi energi kinetik dan radiasi, yang terbang.
Pernyataan mendalam $E = mc^2$ adalah bahwa kelembaman itu sendiri
merupakan kandungan energi: sekotak gas panas melawan percepatan lebih
kuat daripada kotak yang sama dalam keadaan dingin.
\end{remark}

\section{Tumbukan dan peluruhan}

\begin{definition}[Momentum-empat dan massa invarian]\label{def:b3:relativistic-dynamics:four-momentum}
\index{momentum-empat}\index{massa invarian}
Bungkuslah energi dan momentum sebuah partikel menjadi
\emph{momentum-empat}nya
$P = (E/c, \vect p)$. Untuk sistem partikel, jumlahkanlah komponen demi
komponen: $E_{\text{tot}} = \sum E_i$, $\vect p_{\text{tot}} = \sum\vect p_i$.
Yang disebut \emph{massa invarian} $M$ sistemnya didefinisikan oleh
\[
M^2c^4 = E_{\text{tot}}^2 - \|\vect p_{\text{tot}}\|^2c^2 :
\]
bilangan yang sama dalam setiap kerangka, dan sama dengan energi total
(dibagi $c^2$) dalam \emph{kerangka pusat momentum} tempat
$\vect p_{\text{tot}} = \vect 0$. Perhatikan bahwa $M$ melebihi jumlah
massa bagiannya setiap kali bagian itu bergerak satu terhadap yang lain
--- dua foton yang terbang saling menjauh mempunyai $M > 0$ walaupun
masing-masingnya tak bermassa.
\end{definition}

\begin{method}[Menyelesaikan tumbukan relativistik]\label{met:b3:relativistic-dynamics:method}
(1) Tuliskanlah kekekalan \omterm{def:b3:relativistic-dynamics:four-momentum}{momentum-empat}: $E$ dan $\vect p$ bersama,
tak pernah salah satunya saja. (2) Hitunglah invarian $E^2 - p^2c^2$
bungkusan mana pun yang paling nyaman --- ia dapat dihitung dalam
kerangka yang paling mudah lalu dipakai dalam kerangka lain mana pun.
(3) Ambang: sebuah reaksi mungkin terjadi bila \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invarian} keadaan
awalnya mencapai jumlah massa diam keadaan akhirnya. (4) Peluruhan
dalam keadaan diam: momentum hasilnya berlawanan dan sama besar;
energinya menyusul hanya dari massanya. (5) Barulah pada akhir sekali,
jika diminta, ubahlah menjadi laju lewat $v = pc^2/E$.
\end{method}

\begin{example}[Sidik jari pion]\label{ex:b3:relativistic-dynamics:pion}
Pion bermuatan yang diam meluruh, $\pi^+ \to \mu^+ + \nu$ (neutrinonya
praktis tak bermassa). Kekekalan momentum membuat hasilnya saling
membelakangi dengan $p$ yang sama; kekekalan energi berbunyi $m_\pi c^2 =
\sqrt{p^2c^2 + m_\mu^2c^4} + pc$. Penyelesaiannya:
\[
pc = \frac{(m_\pi^2 - m_\mu^2)c^2}{2m_\pi} = \qty{29.8}{MeV} ,
\]
sehingga setiap muon dari pion yang meluruh dalam keadaan diam lahir
dengan energi kinetik tepat \qty{4.1}{MeV} --- sebuah garis
berenergi tunggal, dan pengamatannya (Powell, 1947) adalah cara massa
pion pertama kali ditimbang. Peluruhan dua benda selalu menghasilkan
garis semacam itu; peluruhan tiga benda mengaburkannya menjadi
spektrum, dan justru itulah cara neutrino pertama kali dicurigai dalam
peluruhan $\beta$ inti.
\end{example}

\begin{proposition}[Ambang: kelaliman sasaran diam]\label{prop:b3:relativistic-dynamics:threshold}
\index{energi ambang}\index{penumbuk}
Untuk menciptakan partikel baru, hanya energi \emph{pusat momentum}
$Mc^2$ yang tersedia. Berkas berenergi $E$ yang menghantam sasaran
bermassa $m_{\text{t}}$ yang diam menghasilkan
\[
M^2c^4 = m_{\text{b}}^2c^4 + m_{\text{t}}^2c^4 +
2\,E\,m_{\text{t}}c^2 :
\]
energi yang berguna hanya tumbuh sebagai $\sqrt E$ --- sisanya
terbuang untuk mendorong serpihannya ke depan. Dua berkas serupa yang
bertumbukan berhadapan menghasilkan $Mc^2 = 2E$: seluruhnya berguna.
Satu rumus inilah sebabnya mesin di garis depan berupa penumbuk
(\cref{pb:b3:relativistic-dynamics:1}).
\end{proposition}

\begin{proof}
Hitunglah $E_{\text{tot}}^2 - p_{\text{tot}}^2c^2$ di laboratorium:
$(E + m_{\text{t}}c^2)^2 - p^2c^2$ dengan $p$ momentum berkasnya;
jabarkanlah lalu pakailah $E^2 - p^2c^2 = m_{\text{b}}^2c^4$. Untuk
penumbuknya, $\vect p_{\text{tot}} = \vect 0$ dan $Mc^2 = E_{\text{tot}}$.
\end{proof}

\section{Gaya dan mesinnya}

\begin{proposition}[Persamaan gerak relativistik]\label{prop:b3:relativistic-dynamics:force}
\index{sinkrotron}
Hukum Newton bertahan dalam bentuk
\[
\vect F = \frac{\dd\vect p}{\dd t} = \frac{\dd(\gamma m\vect v)}
{\dd t} , \qquad
\frac{\dd E}{\dd t} = \vect F\cdot\vect v .
\]
Medan magnet, yang selamanya tegak lurus terhadap $\vect v$, tidak
mengubah energi dan melenturkan lintasannya menjadi lingkaran berjejari
\[
r = \frac{p}{qB}
\]
--- rumus yang sama seperti dalam jilid Tahun 1, dengan $p$ yang
relativistik. Tetapi frekuensi peredarannya $qB/\gamma m$ kini
\emph{turun} ketika partikelnya memperoleh energi: dorongan berfrekuensi
tetap sebuah siklotron kehilangan langkah di sekitar skala MeV, dan
mesin berenergi tinggi --- yaitu \emph{sinkrotron} --- justru menaikkan
medan dan frekuensinya seirama dengan $\gamma$, sehingga berkasnya tetap
pada satu cincin.
\end{proposition}

\begin{proof}[Bukti sebagian]
Hukum gayanya adalah definisi dinamika yang taat asas dengan momentum
barunya ($\dd E/\dd t$ menyusul dari
$E^2 = p^2c^2 + m^2c^4$: $E\,\dot E = c^2\vect p\cdot\dot{\vect p}$).
Untuk lingkarannya: $|\dd\vect p/\dd t| = qvB$ dengan $|\vect p|$ tetap
berarti vektor momentumnya berputar dengan laju $qvB/p$, sehingga
jejarinya $v/(qvB/p) = p/qB$.
\end{proof}

\begin{example}[Membaca LHC seperti sebuah latihan]\label{ex:b3:relativistic-dynamics:lhc}
Proton berenergi $E = \qty{6.8}{TeV}$ dalam dipol $B = \qty{8.3}{T}$:
$p \approx E/c$ (ultrarelativistik), sehingga $r = p/qB =
\qty{2.7}{km}$ --- dan memang jejari lentur magnetik cincinnya
\qty{2.8}{km}, dengan keliling \qty{27}{km} yang sebagian besar berupa
magnet. Tumbukan berhadapannya menyediakan $Mc^2 = \qty{13.6}{TeV}$;
untuk mencapainya pada sasaran diam diperlukan berkas sebesar
$2E^2/m_{\text{p}}c^2 \approx
\qty{e17}{eV}$ --- seratus kali jangkauan mesin mana pun yang pernah
dibangun. Rumus penumbuk itulah, bukan magnet yang lebih kuat, yang
membeli garis depan energi modern.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:1}
Sebuah elektron ($mc^2 = \qty{0.511}{MeV}$) bergerak pada $0.99c$.
Hitunglah $\gamma$, momentumnya dalam $\unit{MeV}/c$, energi total dan
energi kinetiknya. Pertanyaan yang sama untuk proton ($mc^2 = \qty{938}{MeV}$)
berenergi kinetik \qty{1}{GeV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:2}
(a) Berapa energi yang tertidur dalam satu gram materi? (b) Ledakan
Hiroshima melepaskan sekitar \qty{6e13}{J}: selisih massa sebesar apakah
itu? (c) Matahari memancarkan $L = \qty{3.8e26}{W}$: berapa massa yang
dilepaskannya tiap sekon, dan berapa bagian dari $\qty{2e30}{kg}$-nya
dalam sepuluh miliar tahun? (d) Baterai telepon Anda menyimpan
\qty{5e4}{J}: seberapa lebih berat telepon yang terisi penuh?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:3}
(a) Berapa momentum yang dibawa berkas penunjuk laser \qty{5}{mW} tiap
sekon, dan berapa gaya yang dirasakan penunjuk itu? (b) Berapa gaya yang
dikerahkan sinar matahari (\qty{1.4}{kW/m^2}) pada layar surya
\qty{100}{m^2} yang memantul sempurna? (c) Dari keadaan diam, seberapa
cepat wahana berlayar \qty{10}{kg} bergerak setelah sebulan? (d) Mengapa
ekor debu komet menunjuk menjauhi Matahari?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:relativistic-dynamics:4}
Berlatih dengan satuan. (a) Tunjukkanlah bahwa $\unit{MeV}/c^2$ adalah
satuan massa lalu ubahlah massa elektron ke kilogram. (b) Berapa
\qty{1}{GeV/c} dalam $\unit{kg.m/s}$? (c) Sebuah kuadrat \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invarian}
diperoleh sebagai $s = \qty{16}{GeV^2}$ (dalam satuan $c = 1$): energi
pusat momentum sebesar apakah itu? (d) Mengapa fisikawan partikel
mengambil $c = 1$, dan apa yang harus dimasukkan kembali oleh pembaca
untuk memperoleh bilangan SI?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:5}
Elektron Bertozzi berenergi kinetik $0.5$, $1$, $4.5$ dan
\qty{15}{MeV}. (a) Hitunglah ramalan Newton $v^2 = 2E_k/m$ untuk
masing-masingnya, dalam satuan $c^2$. (b) Hitunglah $v^2/c^2$
relativistiknya. (c) Waktu terbangnya sepanjang \qty{8.4}{m} pada
\qty{15}{MeV}: berapa yang terbaca pada jamnya, dan berapa yang akan
diramalkan Newton? (d) Energi elektronnya juga diukur secara
kalorimetrik --- lewat kalor yang ditimbulkannya saat menumbuk: mengapa
langkah itu merupakan inti sebenarnya percobaan tersebut?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:6}
Pion bermuatan meluruh dalam keadaan diam: $\pi^+ \to \mu^+\nu$, $m_\pi c^2 =
\qty{139.6}{MeV}$, $m_\mu c^2 = \qty{105.7}{MeV}$, $m_\nu \approx 0$.
(a) Tunjukkanlah $pc = (m_\pi^2 - m_\mu^2)c^4/2m_\pi c^2$ lalu hitunglah.
(b) Berapa energi kinetik dan laju muonnya? (c) Berapa energi
neutrinonya? (d) Dalam keadaan terbang pada $\gamma_\pi = 50$, berapa
energi laboratorium maksimum dan minimum muon peluruhannya (peluruhan ke
depan dan ke belakang)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:7}
\omterm{def:b3:relativistic-dynamics:four-momentum}{Massa invarian} dalam kerja. (a) Dua foton \qty{200}{MeV} terbang
membentuk sudut $60^\circ$ satu terhadap yang lain: hitunglah \omterm{def:b3:relativistic-dynamics:four-momentum}{massa
invarian} pasangannya. (b) Pion netral ($m_{\pi^0}c^2 = \qty{135}{MeV}$)
meluruh menjadi dua foton: berapa \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invarian} pasangan fotonnya,
dalam setiap kerangka? (c) Jelaskanlah bagaimana sebuah percobaan
``menemukan'' sebuah partikel sebagai tonjolan pada sebaran \omterm{def:b3:relativistic-dynamics:four-momentum}{massa
invarian} hasil peluruhannya. (d) Dua foton berenergi sama terbang tepat
sejajar: berapa \omterm{def:b3:relativistic-dynamics:four-momentum}{massa invariannya}? Apa yang dikatakan hal ini tentang
kerangka diam sebuah berkas cahaya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:8}
Pembukuan LHC (proton $E = \qty{6.8}{TeV}$). (a) Berapa $\gamma$, dan
berapa kekurangan lajunya $c - v$ (pakailah $c - v \approx c/2\gamma^2$)?
(b) Berapa frekuensi peredarannya pada cincin \qty{27}{km} dan berapa
putaran tiap sekon? (c) Berapa energi tersimpan berkas berisi
\num{3e14} proton, dalam kilogram TNT (\qty{4.2e6}{J/kg})? (d) Berapa
setara massa energi kinetik kedua berkasnya, dalam mikrogram --- yaitu
materi yang segera dibuat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:relativistic-dynamics:9}
Menumbukkan foton. (a) Tunjukkanlah bahwa satu foton dalam vakum tidak
dapat meluruh menjadi pasangan elektron--positron, seberapa pun besar
energinya (pakailah invariannya). (b) Terhadap foton kedua berenergi
$\epsilon$ secara berhadapan, tunjukkanlah bahwa penciptaan pasangannya
menuntut $E\epsilon \ge (m_{\text{e}}c^2)^2$. (c) Foton \qty{511}{keV}
memerlukan pasangan seperti apa? Bagaimana halnya sinar-X \qty{50}{keV}?
(d) Semesta dipenuhi cahaya bintang dan latar gelombang mikro kosmik:
apa yang diramalkan (b) tentang jangkauan sinar $\gamma$ berenergi TeV
melintasi ruang antargalaksi?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:10}
Compton, lewat \omterm{def:b3:relativistic-dynamics:four-momentum}{momentum-empat}. Sebuah foton berpanjang gelombang
$\lambda$ menghantam elektron yang diam lalu terhambur pada sudut
$\theta$. (a) Tuliskanlah kekekalan \omterm{def:b3:relativistic-dynamics:four-momentum}{momentum-empatnya} lalu kucilkanlah
invarian elektron akhirnya. (b) Turunkanlah
\[
\lambda' - \lambda = \frac{h}{m_{\text{e}}c}\,(1 - \cos\theta) .
\]
(c) Hitunglah $h/m_{\text{e}}c$ dan pergeseran maksimumnya; mengapa
efeknya tak terlihat dengan cahaya tampak tetapi menentukan bagi
sinar-X? (d) Apa yang ditetapkan pengukuran Compton pada 1923 tentang
cahaya, yang belum ditetapkan efek fotolistrik?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:11}
Ujung spektrum sinar kosmik. Semesta mandi dalam foton gelombang mikro
berenergi khas $\epsilon = \qty{6e-4}{eV}$. Proton berenergi $E$ yang
menghantam satu foton secara berhadapan dapat tereksitasi ke resonansi
$\Delta$ ($m_\Delta c^2 = \qty{1232}{MeV}$), sambil kehilangan energi
tiap kali. (a) Tunjukkanlah bahwa syarat ambangnya kira-kira $4E\epsilon =
(m_\Delta^2 - m_{\text{p}}^2)c^4$. (b) Hitunglah ambang $E$-nya.
(c) Di atasnya, semesta menjadi legap bagi proton di luar sekitar
\qty{100}{Mly}: apa yang diramalkan hal ini bagi spektrum sinar kosmik
yang teramati (pemenggalan Greisen--Zatsepin--Kuzmin)? (d) Sinar kosmik
berenergi $\qty{3e20}{eV}$ pernah terekam --- partikel ``Oh-My-God'':
apa yang dituntut keberadaannya dari sumbernya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:relativistic-dynamics:12}
Roket foton. Sebuah roket bermassa awal $m_{\text{i}}$ memancarkan
buangannya sebagai berkas cahaya terkolimasi lalu mencapai laju $\beta c$.
(a) Dengan mengekalkan energi dan momentum antara awal dan akhirnya,
tunjukkanlah
\[
\frac{m_{\text{i}}}{m_{\text{f}}} =
\sqrt{\frac{1 + \beta}{1 - \beta}} = \eu^{\varphi} ,
\]
yakni eksponensial rapiditasnya --- persamaan Tsiolkovsky relativistik
dengan buangan sebaik-baiknya. (b) Berapa nisbah massanya untuk
mencapai $0.9c$? Dan untuk mencapai $0.9c$ lalu \emph{berhenti} di
tujuan? (c) Berkasnya harus dibuat dengan cara tertentu: dengan
pemusnahan materi--antimateri berefisiensi sempurna, berapa kilogram
antimateri per kilogram muatan untuk perjalanan sekali jalan $0.9c$?
(d) Produksi antiproton sedunia hanya beberapa nanogram tiap tahun:
simpulkanlah, dalam satu kalimat, tentang roket foton.
\end{exercise}

\section{Soal: Membuat materi --- antiproton dan Bevatron}

\begin{problem}[{Soal akhir pekan --- bagaimana kekekalan momentum
menetapkan harga antidunia}]\label{pb:b3:relativistic-dynamics:1}
Persamaan Dirac menuntut, sejak 1931, bahwa proton mempunyai saudara
kembar cermin bermuatan berlawanan. Membuat satu berarti membeli \omterm{def:b3:relativistic-dynamics:momentum-energy}{energi
diamnya} dengan energi berkas --- dan kekekalan momentum menetapkan
harganya. Soal ini menetapkan harga itu, merancang mesin yang
membayarnya, lalu mengikuti penemuan 1955. Data: $m_{\text{p}}c^2 = \qty{938.3}{MeV}$;
muatan dan \emph{bilangan barion} (proton dan neutron $+1$, antiproton $-1$)
kekal dalam setiap reaksi.

\medskip\noindent\textbf{Bagian I --- Perkakas \omterm{def:b3:relativistic-dynamics:four-momentum}{momentum-empat}.}
\begin{enumerate}
  \item Untuk satu partikel, tunjukkanlah $E^2 - p^2c^2 = m^2c^4$ dari
        definisi $E$ dan $\vect p$.
  \item Untuk sebuah sistem, mengapa $E_{\text{tot}}^2 -
        p_{\text{tot}}^2c^2$ sama dalam semua kerangka, dan sama dengan
        apakah ia dalam kerangka pusat momentum?
  \item Berkas proton berenergi $E$ menghantam proton yang diam:
        tunjukkanlah
        $M^2c^4 = 2m_{\text{p}}^2c^4 + 2E\,m_{\text{p}}c^2$.
  \item Periksalah kedua limitnya: pada energi rendah $Mc^2 \to
        2m_{\text{p}}c^2$; pada energi tinggi $Mc^2 \approx
        \sqrt{2E\,m_{\text{p}}c^2}$ --- yakni hukum akar kuadratnya.
  \item Sebuah reaksi hanya diizinkan jika $Mc^2$ mencapai jumlah massa
        diam hasilnya, dengan semua hasilnya diam dalam kerangka pusat
        momentum pada ambangnya: berilah alasan bagi anak kalimat
        terakhir ini.
  \item Mengapa energi kinetik hasilnya di laboratorium tidak pernah
        dapat diambil kembali untuk penciptaan partikel pada sasaran
        diam?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Menetapkan harga antiproton.}
\begin{enumerate}[resume]
  \item Dalam tumbukan $p + p$, reaksi termurah yang membuat sebuah
        antiproton $\bar p$ harus juga membuat sebuah proton
        \emph{tambahan}: tuliskanlah, lalu berilah alasan dengan muatan
        dan bilangan barionnya.
  \item Tunjukkanlah bahwa ambangnya menuntut $Mc^2 = 4m_{\text{p}}c^2$.
  \item Simpulkanlah energi berkas yang diperlukan $E = 7m_{\text{p}}c^2$
        dan energi kinetik $T = 6m_{\text{p}}c^2$; hitunglah $T$.
  \item Pada ambangnya, berapa bagian energi kinetik berkasnya yang
        sungguh-sungguh menjadi massa diam baru ($2m_{\text{p}}c^2$
        di antaranya)? Di manakah sisanya?
  \item Pada penumbuk $p$--$p$ berhadapan, berapa energi kinetik
        \emph{tiap berkas} yang dituntut reaksi yang sama?
        Bandingkanlah kedua rancangan itu.
  \item Pada 1954 belum ada penumbuk (berkasnya terlalu tipis untuk
        saling menghantam): fakta praktis apa yang memaksa pilihan
        sasaran diam, dan berapa harganya dalam energi berkas?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Merancang Bevatron.}
Mesin yang dibangun di Berkeley mencapai $T = \qty{6.2}{GeV}$ ---
cukup nyaman di atas ambangnya --- dengan medan dipol $B =
\qty{1.56}{T}$.
\begin{enumerate}[resume]
  \item Hitunglah energi total dan momentum berkasnya pada energi
        puncaknya.
  \item Hitunglah jejari lenturnya $r = p/qB$ lalu bandingkanlah dengan
        \qty{15.2}{m} milik mesin itu.
  \item Hitunglah laju protonnya dan frekuensi peredarannya pada cincin
        $2\pi r$.
  \item Pada penyuntikannya protonnya tiba dengan energi kinetik
        \qty{10}{MeV}: hitunglah lajunya, lalu jelaskanlah mengapa
        frekuensi pemercepatnya harus \emph{menyapu} sepanjang tiap
        daurnya (dua besaran mana yang berubah ketika $E$ tumbuh?).
  \item Dengan perolehan sekitar \qty{1.5}{kV} tiap putaran, berapa
        putaran dan berapa lama satu daur pemercepatannya berlangsung?
  \item Nama ``Bevatron'' berasal dari BeV, yakni miliaran
        elektronvolt: nyatakanlah dalam satu baris fisika apa yang
        menetapkan angka rancangan $6$ BeV lebih sedikit itu.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Penemuannya, 1955.}
\begin{enumerate}[resume]
  \item Sebuah magnet melenturkan tiap calonnya pada sebuah lingkaran:
        jelaskanlah mengapa hal ini memilih partikel menurut
        \emph{momentum}nya ($r = p/qB$), bukan menurut energi atau
        lajunya --- dan mengapa berkas terpilih menurut momentum tetap
        mencampur berbagai jenis.
  \item Berkas pada sasaran tembaga menghasilkan curahan pion negatif
        ($m_\pi c^2 = \qty{139.6}{MeV}$) yang di antaranya bersembunyi
        beberapa antiproton. Spektrometernya memilih partikel negatif
        bermomentum $p = \qty{1.19}{GeV}/c$: hitunglah laju sebuah
        antiproton dan laju sebuah pion pada momentum itu.
  \item Sepanjang lintasan terbang \qty{12}{m}, hitunglah kedua waktu
        terbangnya: resolusi waktu sebesar apa yang dituntut penemuan
        itu?
  \item Sidik kedua yang bebas memakai pencacah Cerenkov, yang hanya
        menyala di atas suatu ambang laju: partikel mana yang diatur
        agar menyalakannya, dan mengapa keberlebihan itu penting bagi
        sebuah penemuan?
  \item Chamberlain dan Segrè menghitung sekitar satu antiproton tiap
        $\num{44000}$ pion: mengapa argumen ambangnya menjamin lajunya
        akan sangat kecil bahkan di atas ambang?
  \item Sebuah antiproton akhirnya bertemu proton lalu saling
        memusnahkan: berapa energi yang dilepaskan tiap peristiwa, dan
        menjadi apa, lazimnya?
  \item Ringkaslah hasil bernama itu: kekekalan \omterm{def:b3:relativistic-dynamics:four-momentum}{momentum-empat}
        menetapkan harga antiproton pada $T = 6m_{\text{p}}c^2 =
        \qty{5.6}{GeV}$ di sasaran diam; sebuah sinkrotron
        \qty{6.2}{GeV} berjejari \qty{15}{m} membayarnya, dan antidunia
        menjadi fakta laboratorium --- Hadiah Nobel 1959.
\end{enumerate}
\end{problem}
