\chapter{Foton dan Fonon}\label{ch:b3:photons-phonons}

Teori kuantum lahir dari pokok bahasan bab ini. Pada 1900 spektrum
benda berpijar menolak menaati fisika klasik --- lebih buruk lagi,
fisika klasik meramalkan energi \emph{takhingga} pada panjang gelombang
pendek, yakni ``bencana ultraungu'' --- dan penawar Planck yang lahir
dari keputusasaan, yaitu energi dalam bungkusan $h\nu$, ternyata menjadi
kunci induk abad itu. Dengan permesinan yang kini sudah terangkai ---
pencacahan ragam dari \cref{ch:b3:schrodinger-three-dimensions},
termodinamika osilator dari \cref{ch:b3:canonical-ensemble} --- \omterm{thm:b3:photons-phonons:planck}{hukum
Planck} memakan setengah halaman: yakni gas foton, satu osilator
terisi Bose tiap ragam rongganya. Setengah halaman yang sama, dengan
bunyi menggantikan cahaya, adalah teori zat padat Debye beserta
kapasitas kalor $T^3$ yang terlewatkan model Einstein. Dan yang mengisi
tiap sentimeter kubik alam semesta adalah adikarya pokok bahasan ini:
latar gelombang mikro kosmik ber-2,725 kelvin, benda hitam paling
sempurna yang pernah terukur --- yakni radiasi termal alam semesta muda
itu sendiri, yang masih terus tiba.

\section{Hukum Planck, diturunkan}

\begin{theorem}[Spektrum benda hitam]\label{thm:b3:photons-phonons:planck}
\index{hukum Planck}\index{radiasi benda hitam}
Sebuah rongga bersuhu $T$ menyangga ragam elektromagnetik ---
yakni gelombang tegak yang dicacah persis seperti dalam
\cref{prop:b3:schrodinger-three-dimensions:counting}, dua kali untuk
pengutubannya:
\[
g(\nu)\,\dd\nu = \frac{8\pi V}{c^3}\,\nu^2\,\dd\nu .
\]
Tiap ragamnya adalah osilator kuantum yang menahan, pada kesetimbangan,
$\bar n = 1/(\eu^{h\nu/k_{\text{B}}T} - 1)$ foton
(\cref{exo:b3:canonical-ensemble:5}; setara dengan boson pada $\mu =
0$). Energi tiap volume dan tiap frekuensinya karena itu adalah
\[
u(\nu, T) = \frac{8\pi h\nu^3}{c^3}\,
\frac{1}{\eu^{h\nu/k_{\text{B}}T} - 1} :
\]
yakni \emph{\omterm{thm:b3:photons-phonons:planck}{hukum Planck}} (1900). Akibatnya, semuanya sebagaimana terukur:
puncaknya pada $\lambda_{\max}T = \qty{2898}{\micro m.K}$ (yakni hukum
pergeseran Wien jilid Tahun 2, yang kini diturunkan); fluks total
dari permukaan hitam $\Phi = \sigma T^4$ dengan
\[
\sigma = \frac{2\pi^5k_{\text{B}}^4}{15h^3c^2}
 = \qty{5.67e-8}{W.m^{-2}.K^{-4}}
\]
(yakni Stefan--Boltzmann, dengan tetapannya kini dibangun dari $h$, $c$,
$k_{\text{B}}$); dan kerapatan fotonnya $n_\gamma \approx
\num{2.03e7}\,(T/\unit{K})^3$ tiap meter kubik.
\end{theorem}

\begin{proof}[Bukti sebagian]
Kalikanlah cacah ragamnya dengan $h\nu\,\bar n$. Integral energi
totalnya, dengan $x = h\nu/k_{\text{B}}T$ dan $\int_0^\infty
x^3\dd x/(\eu^x - 1) = \pi^4/15$ (diterima tanpa bukti), memberikan $u_{\text{tot}}
= aT^4$ dan, dengan faktor fluks bakunya $c/4$, memberikan
$\sigma$ yang dinyatakan itu. Hukum Wien adalah pemaksimuman
\cref{exo:b3:photons-phonons:1}. Pada frekuensi rendah ($h\nu \ll
k_{\text{B}}T$) tiap ragamnya mengangkut $k_{\text{B}}T$ ---
yakni Rayleigh--Jeans, yang benar untuk radio, tetapi bencana bila
dirambatkan ke seluruh
$\nu$: pemenggalan Planck adalah izin ekipartisi
(\cref{thm:b3:canonical-ensemble:equipartition}) yang dicabut
ragam demi ragam.
\end{proof}

\begin{example}[Tiga termometer]\label{ex:b3:photons-phonons:thermometers}
Matahari ($T = \qty{5800}{K}$): puncaknya pada \qty{0.50}{\micro m} ---
mata kita berevolusi tepat pada maksimumnya. Sebuah tubuh pada \qty{310}{K}:
puncaknya di dekat \qty{9.4}{\micro m}, yang memancarkan $\sigma T^4 \approx
\qty{520}{W/m^2}$ dari tiap meter persegi kulitnya (untunglah,
ruangannya memancar balik;
\cref{exo:b3:photons-phonons:2}). Alam semesta ($T =
\qty{2.725}{K}$): puncaknya di dekat \qty{1.9}{mm} dalam bentuk
frekuensinya (\qty{160}{GHz}), $411$ foton tiap sentimeter kubik ---
yakni latar gelombang mikro kosmik, contoh agung bab ini
(\cref{pb:b3:photons-phonons:1}).
\end{example}

\begin{proposition}[Termodinamika cahaya]\label{prop:b3:photons-phonons:thermo}
\index{tekanan radiasi!termal}
Gas foton memiliki kerapatan energi $u = aT^4$ ($a =
4\sigma/c$), tekanan
\[
P = \frac u3 ,
\]
kerapatan entropi $\tfrac43 aT^3$, dan cacah foton $\propto T^3$.
Akibatnya: kotak berisi radiasi yang memuai secara adiabatik
mendingin sebagai $T \propto 1/L$ --- foton alam semesta yang memuai
mendingin dari $3000$ K menjadi $2.725$ K hari ini lewat faktor rentang
$\approx1100$ sambil \emph{mempertahankan} spektrum Planck yang
sempurna; dan dalam bintang masif pertumbuhan $T^4$ membuat tekanan
radiasi menyaingi tekanan gasnya, sehingga menetapkan batas atas massa
bintang (\cref{ch:b3:astrophysics}).
\end{proposition}

\begin{proof}[Bukti sebagian]
$P = u/3$: gas partikel yang isotrop pada laju $c$ menyerahkan
sepertiga kerapatan energinya sebagai tekanan (yakni faktor $\langle
\cos^2\theta\rangle = 1/3$, seperti dalam teori kinetik jilid Tahun 1).
Entropinya dari $\dd U = T\dd S - P\dd V$ yang dikenakan pada $U =
aT^4V$. Secara adiabatik: $S \propto T^3V$ yang tetap memberikan $T \propto
V^{-1/3}$.
\end{proof}

\section{Fonon: penuntasan Debye}

\begin{theorem}[Model Debye]\label{thm:b3:photons-phonons:debye}
\index{model Debye}\index{fonon}
Getaran sebuah kristal adalah gelombang lenting yang terkuantumkan
persis seperti cahaya: yakni \emph{fonon}, dengan tiga pengutuban pada
laju bunyi $c_{\text{s}}$, dicacah seperti foton tetapi dengan
totalnya dibatasi pada
$3N$ ragam --- yakni batas yang mendefinisikan \emph{suhu Debye}
\[
k_{\text{B}}\theta_{\text{D}} = \hbar c_{\text{s}}
(6\pi^2n)^{1/3} .
\]
Kapasitas kalornya menyisip di antara kedua hukum klasiknya:
\[
C \to 3Nk_{\text{B}} \;(T \gg \theta_{\text{D}})
\qquad\text{(Dulong--Petit)} ,
\qquad
C \to \frac{12\pi^4}{5}\,Nk_{\text{B}}
\Big(\frac{T}{\theta_{\text{D}}}\Big)^3
\;(T \ll \theta_{\text{D}}) :
\]
yakni hukum $T^3$ yang dituntut kalorimetri suhu rendah dan yang tak
dapat diberikan frekuensi tunggal Einstein
(\cref{exo:b3:canonical-ensemble:6}). Alasannya sama seperti
untuk cahaya: sedingin apa pun kristalnya, selalu ada kuanta bunyi
berpanjang gelombang besar yang semurah-murahnya, dan pencacahannya yang
mirip foton memberikan $T^3$ yang mirip foton pula.
\end{theorem}

\begin{proof}[Bukti sebagian]
Ulangilah perhitungan fotonnya dengan $c \to c_{\text{s}}$, tiga
pengutuban, dan total ragam $\int g\,\dd\nu = 3N$ yang menetapkan
pemenggalannya. Pada $T \ll \theta_{\text{D}}$ pemenggalannya tak
terlihat dan integral energi $T^4$-nya memberikan $C \propto T^3$; pada $T$
tinggi tiap ragam dari $3N$ ragamnya mengangkut $k_{\text{B}}T$. Integral
penyisipan lengkapnya diterima tanpa bukti.
\end{proof}

\begin{example}[Fonon adalah partikel sungguhan]\label{ex:b3:photons-phonons:phonon-real}
Fonon menghamburkan neutron dan sinar-X dengan kekekalan energi dan
momentum (kristal) --- kurva pencaran terukur
$\omega(\vect k)$ \cref{ch:b3:crystalline-solids} adalah
spektroskopinya. Mereka mengangkut kalor: hantaran termal sebuah
penyekat adalah $\kappa = \tfrac13 Cc_{\text{s}}\ell$ sebuah gas fonon, dan
intan --- kaku, ringan, murni --- menghantar lima kali lebih baik
daripada tembaga pada suhu ruang hanya lewat fonon
(\cref{exo:b3:photons-phonons:12}). Dan mereka mati pada suhu
rendah sebagai $T^3$, dan karena itulah para insinyur kriogenik
berbicara tentang kristal yang menjadi ``sunyi secara akustik''.
\end{example}

\begin{method}[Keterampilan gas radiasi]\label{met:b3:photons-phonons:method}
(1) Cacahlah ragamnya ($\propto\nu^2$ tiap pengutubannya dalam tiga
dimensi); isilah dengan $1/(\eu^{\beta h\nu} - 1)$; lalu
integralkanlah --- harapkanlah energi $T^4$ serta cacah dan kapasitas
$T^3$. (2) Puncaknya lewat Wien; totalnya lewat Stefan; nilai klasik
$k_{\text{B}}T$ tiap ragamnya hanya di bawah
$h\nu \sim k_{\text{B}}T$. (3) Untuk zat padat, batasilah ragamnya pada $3N$
($\theta_{\text{D}}$); bentuk Einstein untuk ragam cabang optiknya,
bentuk Debye untuk yang akustik. (4) Radiasi adiabatik: $T \propto
V^{-1/3}$. (5) Periksalah kedua tetapan andalannya: $\lambda_{
\max}T = \qty{2898}{\micro m.K}$ dan $\sigma =
\qty{5.67e-8}{W/m^2K^4}$.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:photons-phonons:1}
(a) Dari \omterm{thm:b3:photons-phonons:planck}{hukum Planck} dalam bentuk panjang gelombang, tunjukkanlah
bahwa puncaknya menaati $\lambda_{\max}T = \text{tetapan}$ (persamaan
melampaui aljabarnya boleh dikutip: $x = 4.965$). (b) Hitunglah panjang
gelombang puncak untuk Matahari, sebuah filamen \qty{2700}{K}, sebuah
tubuh \qty{310}{K} dan langit \qty{2.725}{K}. (c) Yang mana dari
keempatnya yang tercuplik dengan baik oleh mata manusia, dan berapa
bagian keluaran sebuah filamen yang kasat mata (perkirakan: kecil)? (d)
Mengapa ``kamera inframerah'' adalah perkakas yang tepat untuk manusia
dan bangunan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:photons-phonons:2}
Anggaran radiatif. (a) Seseorang ($\qty{1.7}{m^2}$, kulit
\qty{306}{K}) dalam ruangan \qty{293}{K}: daya radiasi bersihnya
$\sigma A(T_1^4 - T_2^4)$. (b) Bandingkanlah dengan metabolisme
\qty{100}{W} tubuhnya: mengapa kita menahannya (pakaian, dan
ruangan yang memancar balik)? (c) Bumi menyerap sinar matahari pada $\pi
R^2$ lalu memancar pada $4\pi R^2$: turunkanlah $T_{\text{eff}}
= [(1-A)\,\Phi_\odot/4\sigma]^{1/4}$ dengan $\Phi_\odot =
\qty{1361}{W/m^2}$, albedo $A = 0.30$, lalu hitunglah. (d) Rata-rata
permukaan terukurnya adalah \qty{288}{K}: sebutkanlah mekanisme
selisih \qty{33}{K} itu
(\cref{pb:b3:harmonic-oscillator:1}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:photons-phonons:3}
Mencacah foton. (a) Dengan $n_\gamma = \num{2.03e7}\,T^3$ tiap
meter kubik: hitunglah kerapatan foton latar \qty{2.725}{K}.
(b) Sinar matahari di Bumi: dari $\Phi_\odot$ dan energi foton rata-rata
$\approx 2.7\,k_{\text{B}}T_\odot \approx \qty{1.35}{eV}$,
hitunglah fluks fotonnya tiap meter persegi tiap detik. (c) Sebuah bola
lampu pijar \qty{60}{W} (5\% kasat mata): berapa foton kasat mata tiap
detiknya. (d) Mengapa ``pencacahan foton'' menjadi hal biasa bagi para
astronom tetapi mustahil bagi pemanas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:photons-phonons:4}
Bencana dan ekornya. (a) Tunjukkanlah bahwa ekipartisi klasik
($k_{\text{B}}T$ tiap ragamnya) memberikan $u(\nu) =
8\pi\nu^2k_{\text{B}}T/c^3$, yang menyebar bila diintegralkan. (b) Di
manakah \omterm{thm:b3:photons-phonons:planck}{hukum Planck} menyusut menjadi bentuk itu? (c) Para astronom
radio mengukur CMB pada \qty{1.4}{GHz}: periksalah $h\nu/k_{\text{B}}T \ll 1$ di sana lalu
benarkanlah kebiasaan ``suhu antena'' mereka yang mengutip keamatan
dalam kelvin. (d) Pada puncak CMB \qty{160}{GHz}, hitunglah
$h\nu/k_{\text{B}}T$: apakah puncaknya bersifat klasik?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:photons-phonons:5}
Tekanan radiasi. (a) Turunkanlah $P = u/3$ dari keisotropannya (atau
terimalah faktor kinetiknya) lalu hitunglah tekanan sinar matahari pada
orbit Bumi; bandingkanlah dengan tekanan atmosfer. (b) Di dalam
Matahari ($T \approx \qty{1.5e7}{K}$): hitunglah $P_{\text{rad}} =
aT^4/3$ lalu bandingkanlah dengan tekanan gasnya $\sim\qty{2e16}{Pa}$.
(c) Tekanan radiasi tumbuh sebagai $T^4$, tekanan gas sebagai $T$:
tunjukkanlah mengapa bintang yang cukup masif (sehingga lebih panas)
menjadi dikuasai radiasi dan tak mantap --- yakni langit-langit
Eddington di dekat
$\sim100\,M_\odot$. (d) Laser: sebuah berkas \qty{1}{kW} yang difokuskan
untuk mendorong --- gayanya berapa? Mengapa ``berkas penarik'' tetap
tinggal dalam film sedangkan \emph{penjepit} optik (gradien, pikonewton)
merupakan perkakas nyata?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:photons-phonons:6}
Rentangan alam semesta. CMB dipancarkan pada penggabungan kembali
($T \approx \qty{3000}{K}$, wilayah \cref{exo:b3:grand-canonical:12})
lalu tiba pada \qty{2.725}{K}. (a) Dari $T \propto
1/a$, carilah faktor rentang semua panjang sejak pemancarannya. (b)
Tunjukkanlah bahwa spektrum Planck tetap Planck di bawah perentangan
seragam tiap panjang gelombangnya (apa yang terjadi pada $T$?). (c)
Puncak yang dipancarkan (cahaya jingga yang kasat mata!) tiba di mana?
(d) Mengapa langitnya karena itu gelap bagi mata tetapi berkobar bagi
sebuah penerima gelombang mikro?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:photons-phonons:7}
Suhu Debye dari bunyi. (a) Hitunglah $\theta_{\text{D}} =
\hbar c_{\text{s}}(6\pi^2n)^{1/3}/k_{\text{B}}$ untuk tembaga
($c_{\text{s}} \approx \qty{3800}{m/s}$, $n =
\qty{8.5e28}{m^{-3}}$) lalu bandingkanlah dengan nilai kalorimetrinya
\qty{343}{K}. (b) Intan: $c_{\text{s}} \approx
\qty{13000}{m/s}$, $n = \qty{1.76e29}{m^{-3}}$: hitunglah lalu
bandingkanlah dengan $\approx\qty{2200}{K}$. (c) Timbal: lunak dan berat
--- jelaskanlah $\theta_{\text{D}} \approx \qty{90}{K}$-nya dalam satu
kalimat. (d) Nyatakanlah aturan yang menghubungkan kekakuan sebuah zat
padat, massa atomnya dan suhu pembekuan kuantumnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:photons-phonons:8}
Tembaga pada satu kelvin. (a) Dengan memakai $C = \gamma T + \beta T^3$ dengan
$\gamma$ dari \cref{ex:b3:quantum-statistics:heat} dan koefisien
$T^3$ Debye untuk $\theta_{\text{D}} = \qty{343}{K}$: hitunglah
kedua sukunya tiap molnya pada \qty{1}{K}. (b) Yang mana yang menang, dan
seberapa banyak? (c) Pada suhu berapa keduanya berpotongan? (d) Mengapa
rezim ini menjadi jendela ahli kalorimetri menuju \omterm{prop:b3:schrodinger-three-dimensions:counting}{kerapatan keadaan}
elektroniknya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:photons-phonons:9}
Rumah kaca satu lapis. Misalkanlah permukaannya (bersuhu $T_s$) duduk
di bawah satu lapis atmosfer tipis (bersuhu $T_a$) yang tembus
sinar matahari tetapi kedap inframerah termal. (a) Tuliskanlah
kesetimbangan energi bagi lapisnya (yang menyerap $\sigma T_s^4$, memancar
$2\sigma T_a^4$) dan bagi permukaannya (yang menyerap sinar matahari $S$ ditambah
$\sigma T_a^4$). (b) Tunjukkanlah $T_s = 2^{1/4}\,T_{\text{eff}}$ dengan
$\sigma T_{\text{eff}}^4 = S$. (c) Dengan $T_{\text{eff}} =
\qty{255}{K}$: hitunglah $T_s$ lalu bandingkanlah dengan \qty{288}{K}
(atmosfer yang sebenarnya adalah selimut sebagian dan berlapis banyak). (d)
Kaitkanlah dengan \cref{pb:b3:harmonic-oscillator:1}: fisika molekuler
mana yang membuat lapisnya kedap dalam inframerah namun tembus dalam
cahaya tampak?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:photons-phonons:10}
Tetapan Stefan dari awal. (a) Rangkailah $u_{\text{tot}} =
\int u(\nu)\dd\nu$ dengan $x = h\nu/k_{\text{B}}T$ dan
$\pi^4/15$ yang diberikan, sehingga diperoleh $u = aT^4$ dengan $a =
8\pi^5k_{\text{B}}^4/15h^3c^3$. (b) Fluks dari sebuah permukaan
hitam adalah $c\,u/4$ (terimalah seperempat geometrinya): bentuklah
$\sigma$ lalu hitunglah secara bilangan dari $h$, $c$,
$k_{\text{B}}$. (c) Bandingkanlah dengan \qty{5.67e-8}{W.m^{-2}.K^{-4}}.
(d) Pembalikan sejarah: Planck mencocokkan hukumnya pada spektrum lalu
menarik \emph{keduanya}, $h$ dan $k_{\text{B}}$ (sehingga juga bilangan
Avogadro) pada 1900 --- nyatakanlah mengapa kurva benda hitam memancang
dua tetapan sekaligus.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:photons-phonons:11}
Koefisien A dan B Einstein (1917). Atom dua aras (berenergi $0,
h\nu_0$; berpopulasi $N_1, N_2$) duduk dalam radiasi berkerapatan
spektral $u(\nu_0)$. Laju serapannya: $B_{12}uN_1$; pancaran
terangsang: $B_{21}uN_2$; pancaran spontan: $AN_2$. (a) Tuliskanlah
kesetimbangannya. (b) Tuntutlah agar nisbah Boltzmann
$N_2/N_1 = \eu^{-h\nu_0/k_{\text{B}}T}$ dan $u$ Planck berlaku
serentak untuk \emph{semua} $T$: turunkanlah $B_{12} = B_{21}$ dan
\[
\frac{A}{B} = \frac{8\pi h\nu_0^3}{c^3} .
\]
(c) Tafsirkanlah: mengapa kesetimbangan termal \emph{memaksa}
pancaran spontan untuk ada, sekaligus menetapkan lajunya dari laju
terangsangnya? (d) Faktor $\nu^3$-nya: kaitkanlah dengan
umur \cref{exo:b3:perturbation-theory:8} dan dengan mengapa pelaseran
sukar pada frekuensi sinar-X.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:photons-phonons:12}
Intan, jalan tol fonon. Hantaran termal sebuah
penyekat: $\kappa = \tfrac13 C_v c_{\text{s}}\ell$ tiap volumenya,
dengan $\ell$ sebagai lintasan bebas rata-rata fononnya. (a) Untuk intan pada
\qty{300}{K} ($C_v \approx \qty{1.8e6}{J.m^{-3}K^{-1}}$, yakni
sebagian beku; $c_{\text{s}} = \qty{13000}{m/s}$; $\ell
\approx \qty{0.3}{\micro m}$): hitunglah $\kappa$ lalu bandingkanlah dengan
nilai tembaga \qty{400}{W/(m.K)}. (b) Mengapa $\ell$ begitu panjang dalam intan
(kaku, ringan, bersih secara isotop --- apa yang menghamburkan fonon?).
(c) Intan yang dimurnikan isotopnya nyaris melipatduakan $\kappa$: apa
yang disingkapkannya tentang hamburan fonon oleh ketakteraturan massa
isotop? (d) Dua kegunaan: penyebar kalor dalam elektronika; dan mengapa
sebutir intan terasa dingin di bibir --- jelaskanlah rasa itu.
\end{exercise}

\section{Soal: Cahaya tertua}

\begin{problem}[{Soal akhir pekan --- membaca latar gelombang mikro
kosmik}]\label{pb:b3:photons-phonons:1}
Arahkanlah penerima gelombang milimeter mana pun ke petak langit mana
pun dan ia melaporkan isyarat yang sama: yakni spektrum Planck pada $T =
\qty{2.72548(57)}{K}$, yang lebih licin daripada tanur mana pun yang
pernah dibangun manusia. Inilah kilatan penggabungan kembali alam semesta
(\cref{exo:b3:grand-canonical:12}), yang terentang seribu seratus kali,
dan ia membawa sensus jagat raya. Data: $T_0 =
\qty{2.725}{K}$; $n_\gamma = \num{2.03e7}\,T^3\,\unit{m^{-3}}$;
$u = aT^4$, $a = \qty{7.56e-16}{J.m^{-3}.K^{-4}}$; kerapatan barion
hari ini $n_{\text{b}} \approx \qty{0.25}{m^{-3}}$; kerapatan genting
$\rho_{\text{c}} \approx \qty{8.6e-27}{kg/m^3}$.

\medskip\noindent\textbf{Bagian I --- Spektrumnya.}
\begin{enumerate}
  \item Hitunglah panjang gelombang puncak (Wien) CMB beserta kerapatan
        fotonnya tiap sentimeter kubik.
  \item Hitunglah kerapatan energinya, dalam $\unit{J/m^3}$ dan dalam
        $\unit{eV/cm^3}$, beserta bagian setara massanya terhadap
        kerapatan gentingnya.
  \item Spektrum COBE (1990) cocok dengan Planck sampai beberapa bagian
        dalam $10^4$ --- hadirinnya bertepuk tangan bagi grafiknya.
        Mengapa spektrum \emph{termal yang sempurna} dari langit kosong
        begitu sukar dijelaskan oleh apa pun kecuali masa lalu yang
        panas dan rapat?
  \item Pada pemancarannya foton yang sama membentuk benda hitam
        \qty{3000}{K}: periksalah faktor rentangnya $1 + z =
        T_{\text{dahulu}}/T_0$.
  \item Mengapa alam semesta menjadi tembus cahaya persis pada suhu itu
        (ingatlah kembali kesetimbangan mana dari
        \cref{ch:b3:grand-canonical} yang padam)?
  \item Langit malam itu gelap (yakni paradoks Olbers dalam astronomi
        lama): dalam arti persis apa ia sebenarnya \emph{terang} --- pada
        panjang gelombang dan suhu berapa?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Sensusnya.}
\begin{enumerate}[resume]
  \item Hitunglah nisbah foton terhadap barionnya $\eta^{-1} =
        n_\gamma/n_{\text{b}}$.
  \item Bilangan itu --- sekitar dua miliar foton tiap protonnya ---
        adalah salah satu data mendasar kosmologi: nyatakanlah apa yang
        diukurnya tentang ketaksetangkupan materi--antimateri (ingatlah
        kembali \cref{exo:b3:grand-canonical:12}: fotonnya sebagian besar
        adalah hasil pemusnahan).
  \item Bandingkanlah kerapatan energi CMB dengan kerapatan cahaya
        bintang ($\sim\qty{2e-15}{J/m^3}$ yang dirata-ratakan atas alam
        semesta): cahaya mana yang menguasai jagat raya?
  \item CMB melampaui, dalam cacah foton, segala yang pernah dipancarkan
        bintang: mengapa hal ini membuat alam semesta dini ``dikuasai
        radiasi'', dan apa yang mengambil alih kemudian (penskalaan
        $T^4$ lawan $T^3$ terhadap $a^{-3}$ materinya)?
  \item Perkirakanlah zaman kesetaraan materi--radiasinya: dengan
        kerapatan radiasi berskala sebagai $(1+z)^4$ dan materi sebagai
        $(1+z)^3$, dan dengan nisbah hari ini
        $u_{\text{rad}}/\rho_{\text{m}}c^2 \approx 1/3400$, carilah
        $z_{\text{eq}}$.
  \item Neutrino tersapih lebih awal lalu mendingin sedikit lebih banyak
        (mereka melewatkan pemanasan ulang pemusnahan $e^+e^-$
        \cref{exo:b3:grand-canonical:12}): ramalannya adalah
        $T_\nu = (4/11)^{1/3}T_\gamma$ --- hitunglah, lalu
        kagumilah ramalan tentang laut neutrino \qty{1.95}{K}
        yang belum pernah terdeteksi langsung oleh siapa pun.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Ketakisotropannya.}
\begin{enumerate}[resume]
  \item CMB lebih panas \qty{3.4}{mK} pada satu arah dan lebih dingin
        pada arah berlawanannya: tafsirkanlah lewat fisika Doppler
        \cref{ch:b3:relativistic-kinematics} lalu tariklah
        kecepatan kita ($\Delta T/T = v/c$).
  \item Setelah dwikutub itu disingkirkan, riak sebesar $\Delta T/T \sim
        \num{e-5}$ tersisa (COBE 1992, lalu WMAP, Planck): apa yang
        dipetakannya pada $z = 1100$?
  \item Benih kerapatan $10^{-5}$ itu tumbuh menjadi galaksi: mengapa
        gravitasi menguatkan kelebihan kerapatan (ketakmantapan yang
        mana), dan mengapa pertumbuhannya memerlukan kesetaraan
        materi--radiasi untuk mulai sungguh-sungguh?
  \item Ukuran riak terkuatnya (kira-kira satu derajat di langit)
        adalah panjang gelombang akustik --- yakni bunyi dalam plasma
        foton--barion (gas bab ini ditambah gelombang
        \cref{ch:b3:continuum-elasticity}!): apa yang menetapkan
        skalanya (yakni jarak tempuh bunyi sebelum penggabungan
        kembalinya)?
  \item Mengukur sudut itu terhadap skala fisis yang diketahui
        menriangulasi geometri alam semesta: nyatakanlah hasilnya
        (datar, sampai ketelitian persen).
  \item Ekakutub \qty{2.7}{K}, dwikutub \qty{3}{mK}, riak
        \qty{e-5}: urutkanlah apa yang diajarkan masing-masing (masa lalu
        yang panas; gerak kita; benih dan geometri segalanya).
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Pusakanya.}
\begin{enumerate}[resume]
  \item Penzias dan Wilson menemukan isyaratnya pada 1965 sebagai derau
        antena yang tak tersingkirkan (setelah mengusir merpatinya):
        mengapa kelebihan yang \emph{isotrop} dan tak bergantung musim
        pada \qty{3}{K} tak terjelaskan sebagai sumber lokal atau galaksi
        mana pun?
  \item Kira-kira $1\%$ semut televisi analog yang tak tersetel adalah
        isyarat ini: renungkanlah dalam satu kalimat.
  \item CMB melewati gugus galaksi lalu sedikit tertendang ke frekuensi
        yang lebih tinggi oleh elektron panas (yakni gejala
        Sunyaev--Zel'dovich): mengapa hal ini membuat gugusnya terlihat
        sebagai \emph{bayangan} pada frekuensi rendah, dan apa keutamaan
        khusus bayangan semacam itu (tak bergantung pada
        jaraknya)?
  \item Perbatasan hari ini memburu pengutuban CMB untuk mencari cap
        gelombang gravitasi purba: bab mana saja dalam buku ini yang
        akan dinikahkan oleh penemuan itu
        (pengutuban \cref{ch:b3:covariant-electromagnetism},
        gelombang gravitasi)?
  \item Hitunglah fluks foton CMB yang menembus tubuh Anda
        ($n_\gamma c/4$ tiap satuan luasnya, dengan tampang lintang
        tubuh $\sim\qty{0.5}{m^2}$): berapa banyak peninggalan Dentuman
        Besar yang melintasi Anda tiap detik?
  \item CMB memilih satu kerangka diam (yakni kerangka tempat
        dwikutubnya lenyap): jelaskanlah mengapa hal ini \emph{tidak}
        bertentangan dengan kenisbian --- kerangka macam apa yang
        diistimewakan oleh kandungan materinya dan bukan oleh hukumnya?
  \item Ringkaslah hasil bernama itu: sebuah kurva Planck ber-$2.725$
        kelvin yang mengisi langit --- $411$ foton tiap sentimeter
        kubik, dua miliar tiap barionnya, dengan puncak pada
        \qty{160}{GHz} --- adalah kuitansi termal permulaan yang
        panas: yakni fisika statistik sebagai batu fondasi
        kosmologi.
\end{enumerate}
\end{problem}
