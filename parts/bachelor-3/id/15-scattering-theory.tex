\chapter{Teori Hamburan}\label{ch:b3:scattering-theory}

Hampir semua yang kita ketahui tentang yang kecil dipelajari dengan
melemparkan sesuatu kepadanya lalu menyaksikan apa yang memantul.
Rutherford menemukan inti atom dengan mencacah kilatan partikel alfa
pada layar seng-sulfida; lima puluh tahun kemudian percobaan yang sama,
yang dijalankan dengan elektron berenergi lebih tinggi, menemukan kuark
di dalam proton; di antara keduanya dan sejak itu, tiap \omterm{def:b3:scattering-theory:cross-section}{penampang
lintang} dalam fisika inti, partikel, atom dan zat mampat dibaca lewat
teori bab ini. Bahasanya adalah \emph{\omterm{def:b3:scattering-theory:cross-section}{penampang lintang}} --- yakni luas
sasaran yang efektif; benda kuantumnya adalah \emph{\omterm{thm:b3:scattering-theory:amplitude}{amplitudo hamburan}},
yang \omterm{thm:b3:scattering-theory:born}{hampiran Bornnya} mengatakan sesuatu yang tak terlupakan: sasaran
yang meruah diselidiki dalam ruang Fourier, sudut demi sudut. Pada
energi rendah \omterm{thm:b3:scattering-theory:partial}{gelombang parsialnya} mengambil alih, meringkas tiap
potensial yang rumit menjadi satu bilangan --- yakni \omterm{thm:b3:scattering-theory:partial}{panjang hamburan}
--- yang hari ini menala gas kuantum ultradingin; dan di tempat sebuah
proyektil dapat berlama-lama dalam keadaan kuasiterikat, \omterm{def:b3:scattering-theory:cross-section}{penampang
lintangnya} melonjak menjadi resonansi, yakni spektroskopi bagi yang
selain itu tak terlihat.

\section{Penampang lintang}

\begin{definition}[Penampang lintang]\label{def:b3:scattering-theory:cross-section}
\index{penampang lintang}\index{penampang lintang diferensial}
Kirimkanlah berkas seragam berfluks $\Phi$ (partikel tiap satuan luas
tiap sekon) ke satu partikel sasaran. Laju proyektil yang terhambur ke
dalam sudut ruang $\dd\Omega$ di sekitar arah $(\theta,
\varphi)$ mendefinisikan \emph{penampang lintang diferensial},
\[
\dd\dot N = \Phi\,\frac{\dd\sigma}{\dd\Omega}\,\dd\Omega ,
\]
dan integralnya $\sigma = \int(\dd\sigma/\dd\Omega)\,\dd\Omega$ adalah
\emph{penampang lintang total}: yakni luas yang secara efektif
disajikan sasarannya. Untuk lempeng tipis dengan $n$ sasaran tiap
satuan volume, sebuah berkas melemah sebagai $\eu^{-n\sigma x}$:
penampang lintang diukur dengan mencacah apa yang keluar. Satuannya:
fisika inti memakai \emph{barn}, $\qty{1}{b} = \qty{e-28}{m^2}$ ---
kira-kira seukuran geometris sebuah inti uranium, ``sebesar lumbung''
bagi sebuah neutron.
\end{definition}

\section{Amplitudo dan hampiran Born}

\begin{theorem}[Keadaan hamburan dan amplitudonya]\label{thm:b3:scattering-theory:amplitude}
\index{amplitudo hamburan}
Untuk partikel berenergi $E = \hbar^2k^2/2m$ yang bertemu potensial
terlokalkan $V(\vect r)$, keadaan hamburan stasionernya berperilaku pada
jarak jauh sebagai
\[
\psi(\vect r) \;\longrightarrow\;
\eu^{\iu kz} + f(\theta)\,\frac{\eu^{\iu kr}}{r} ,
\]
yakni gelombang bidang datang ditambah gelombang bola keluar yang
dimodulasi \emph{\omterm{thm:b3:scattering-theory:amplitude}{amplitudo hamburan}} $f(\theta)$ (sebuah panjang), dan
\[
\frac{\dd\sigma}{\dd\Omega} = |f(\theta)|^2 .
\]
\end{theorem}

\begin{proof}[Bukti sebagian]
Peluruhan $1/r$ membuat fluks yang keluar menembus bola besar mana pun
tetap berhingga; menghitung \omterm{prop:b3:schrodinger-three-dimensions:current}{arus peluangnya}
(\cref{prop:b3:schrodinger-three-dimensions:current}) bagi kedua
sukunya, arus radial keluar tiap sudut ruangnya adalah
$(\hbar k/m)|f|^2$ berbanding fluks datangnya $\hbar k/m$: nisbahnya
adalah definisi $\dd\sigma/\dd\Omega$. Bahwa penyelesaian berbentuk
asimtotik ini ada diterima tanpa bukti.
\end{proof}

\begin{theorem}[Hampiran Born]\label{thm:b3:scattering-theory:born}
\index{hampiran Born}\index{pemindahan momentum}
Untuk potensial yang lemah (atau proyektil yang cepat), \omterm{thm:b3:perturbation-theory:corrections}{teori gangguan}
orde pertama dalam $V$ memberikan
\[
f(\theta) = -\frac{m}{2\pi\hbar^2}
\int V(\vect r)\,\eu^{\iu\vect q\cdot\vect r}\,\dd^3r ,
\qquad
\vect q = \vect k - \vect k' , \quad
q = 2k\sin\frac\theta2 :
\]
amplitudonya adalah, kecuali tetapannya, \emph{transformasi Fourier
potensialnya} yang dihitung pada \omterm{thm:b3:scattering-theory:born}{pemindahan momentum} $\hbar\vect q$.
Percobaan hamburan adalah penganalisis Fourier bagi materi: sudut kecil
membaca panjang gelombang panjang (struktur kasar), sudut besar membaca
rinciannya yang halus --- yakni alasan terdalam mengapa ``energi lebih
tinggi'' berarti ``jarak lebih kecil''.
\end{theorem}

\begin{proof}[Bukti sebagian]
Inilah kaidah emas Fermi
(\cref{thm:b3:perturbation-theory:golden}) yang dijalankan antara
gelombang bidang $\ket{\vect k} \to \ket{\vect k'}$: unsur matriksnya
adalah $\int V\eu^{\iu(\vect k - \vect k')\cdot\vect r}\dd^3r$,
\omterm{prop:b3:schrodinger-three-dimensions:counting}{kerapatan keadaan} akhirnya memasok faktor kinematiknya, dan
pembukuannya (yang diterima tanpa bukti) menyusun keduanya menjadi $f$.
Keelastikannya memberikan $|\vect k'| = |\vect k|$, sehingga
$q = 2k\sin(\theta/2)$ menurut segitiga sama kaki $\vect k, \vect k'$.
\end{proof}

\begin{example}[Dari Yukawa ke Rutherford]\label{ex:b3:scattering-theory:rutherford}
\index{hamburan Rutherford}
Untuk potensial Coulomb tersaring $V = \dfrac{Q_1Q_2}{4\pi
\varepsilon_0 r}\,\eu^{-r/a}$ integral Bornnya bersifat dasar
(\cref{exo:b3:scattering-theory:5}) dan, ketika jejari penyaringnya
$a \to \infty$,
\[
\frac{\dd\sigma}{\dd\Omega} =
\Big(\frac{Q_1Q_2}{16\pi\varepsilon_0 E}\Big)^2
\frac{1}{\sin^4(\theta/2)} :
\]
yakni \emph{\omterm{def:b3:scattering-theory:cross-section}{penampang lintang} Rutherford}. Karena kebetulan yang hanya
dimiliki $1/r$, perhitungan klasiknya, \omterm{thm:b3:scattering-theory:born}{hampiran Bornnya} dan jawaban
kuantum eksaknya semuanya sepakat --- dan itulah sebabnya Rutherford,
yang menghitung secara klasik pada 1911, memperoleh rumus yang benar
dan, dari pembenarannya kilatan demi kilatan, memperoleh inti atomnya
(\cref{pb:b3:scattering-theory:1}).
\end{example}

\section{Gelombang parsial dan panjang hamburan}

\begin{theorem}[Pergeseran fase]\label{thm:b3:scattering-theory:partial}
\index{gelombang parsial}\index{pergeseran fase}\index{panjang hamburan}
Untuk potensial sentral, uraikanlah keadaan hamburannya atas momentum
sudutnya: tiap \omterm{thm:b3:scattering-theory:partial}{gelombang parsial} $l$ meninggalkan daerah potensialnya
dengan osilasi radialnya \emph{tergeser} sebesar fase $\delta_l(k)$ ---
tarikan memajukannya, tolakan menundanya --- dan amplitudo teramatinya
tersusun kembali sebagai
\[
f(\theta) = \frac1k\sum_{l=0}^\infty(2l+1)\,\eu^{\iu\delta_l}
\sin\delta_l\,P_l(\cos\theta) , \qquad
\sigma = \frac{4\pi}{k^2}\sum_l(2l+1)\sin^2\delta_l .
\]
Pada energi rendah, gerak hati klasiknya ($l \sim kb$ untuk parameter
dampak $b$) mengatakan bahwa hanya $l = 0$ yang menembus: hamburannya
menjadi isotropik dan satu bilangan yang memerintah,
\[
f \to -a , \qquad \sigma \to 4\pi a^2 ,
\]
dengan $a = -\lim_{k\to0}\delta_0/k$ adalah \emph{\omterm{thm:b3:scattering-theory:partial}{panjang hamburannya}}.
Apa pun kerumitan dalamnya, sebuah sasaran pada energi rendah
\emph{adalah} \omterm{thm:b3:scattering-theory:partial}{panjang hamburannya} --- yakni penyederhanaan besar yang
menjadi tumpuan fisika atom dingin.
\end{theorem}
\admitted

\begin{example}[Neutron bertemu proton]\label{ex:b3:scattering-theory:np}
Neutron lambat terhambur oleh proton dengan $\sigma \approx
\qty{20.4}{b}$ --- yakni luas empat puluh kali ukuran geometris
deuteronnya. Penjelasannya: tumbukan energi rendahnya murni gelombang
$s$ dengan dua saluran spin, yang \omterm{thm:b3:scattering-theory:partial}{panjang hamburannya} terukur sebesar
$a_{\text{t}} = \qty{5.4}{fm}$ (triplet) dan $a_{\text{s}} =
\qty{-23.7}{fm}$ (singlet), keduanya jauh lebih besar daripada
jangkauan gayanya yang \qty{2}{fm}. $|a|$ yang besar berarti adanya
keadaan yang \emph{nyaris} pada energi nol: deuteron triplet yang
nyaris tak terikat
(\cref{ex:b3:schrodinger-three-dimensions:deuteron}), dan dalam
singletnya sebuah pasangan ``maya'' yang tepat gagal mengikat ---
kegagalannya tertulis dalam $a_{\text{s}} < 0$. Sebuah teka-teki dua
puluh barn, yang diselesaikan oleh dua keadaan terikat yang nyaris jadi
(\cref{exo:b3:scattering-theory:7}).
\end{example}

\section{Resonansi}

\begin{proposition}[Resonansi Breit--Wigner]\label{prop:b3:scattering-theory:resonance}
\index{resonansi!hamburan}\index{rumus Breit--Wigner}
Jika proyektilnya dapat tertangkap sementara ke dalam keadaan
kuasiterikat berenergi $E_{\text{R}}$ dan berumur $\tau = \hbar/\Gamma$,
\omterm{thm:b3:scattering-theory:partial}{pergeseran fase} yang bersesuaian menyapu melewati $\pi/2$ dan \omterm{def:b3:scattering-theory:cross-section}{penampang
lintang} parsialnya melonjak:
\[
\sigma_l(E) \approx \frac{4\pi}{k^2}\,(2l+1)\,
\frac{\Gamma^2/4}{(E - E_{\text{R}})^2 + \Gamma^2/4} ,
\]
yakni sebuah puncak Lorentz selebar $\Gamma$ pada langit-langit yang
dapat dicapai gelombangnya (yakni ``batas keuniteran''
$4\pi(2l+1)/k^2$). Dibaca terbalik, sebuah resonansi dalam \omterm{def:b3:scattering-theory:cross-section}{penampang
lintang} \emph{adalah} penemuan sebuah keadaan: letaknya adalah sebuah
aras energi, lebarnya sebuah umur. Sebagian besar kebun binatang
partikel pada \cref{ch:b3:particle-physics} ditemukan tepat sebagai
tonjolan semacam itu.
\end{proposition}
\admitted

\begin{method}[Membaca sebuah soal hamburan]\label{met:b3:scattering-theory:method}
(1) Kinematikanya lebih dahulu: $k$, beserta rentang $q$ terjangkaunya $0 \le q
\le 2k$ --- skala struktur mana yang terlihat? (2) Cepat dan
lemah: Born --- transformasikanlah potensialnya secara Fourier; lalu
periksalah kekecilannya. (3)
Lambat: \omterm{thm:b3:scattering-theory:partial}{gelombang parsial}, biasanya $s$ saja; pikirkanlah \omterm{thm:b3:scattering-theory:partial}{panjang
hamburannya}, dan harapkanlah $\sigma = 4\pi a^2$ (untuk boson identik:
$8\pi a^2$). (4) Puncak pada $\sigma(E)$: cocokkanlah Breit--Wigner,
lalu laporkanlah sebuah aras dan sebuah umur; lekukan: sebuah fase yang
melewati $\pi$. (5) Sasaran yang meruah: kalikanlah amplitudo titiknya
dengan faktor bentuknya --- yakni transformasi Fourier kerapatannya
(\cref{exo:b3:scattering-theory:12}).
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:1}
Sebuah berkas proton \qty{1}{\micro A} ($\num{6.2e12}$ proton/s)
berpenampang \qty{1}{cm^2} melintasi keping emas \qty{1}{\micro m} ($n =
\qty{5.9e28}{m^{-3}}$). (a) Berapa kerapatan sasaran per luasnya? (b)
Dengan $\sigma = \qty{10}{b}$ tiap intinya, berapa bagian berkasnya yang
terhambur? (c) Berapa lajunya ke dalam detektor bersudut ruang
\qty{e-3}{sr} seandainya hamburannya isotropik? (d) Mengapa
``\omterm{def:b3:scattering-theory:cross-section}{penampang lintang}'' merupakan sifat yang terdefinisi baik bagi
\emph{pasangan} (proyektil, sasaran, energi) alih-alih bagi sasarannya
sendiri?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:2}
Bola keras klasik berjejari $R_1, R_2$: (a) tunjukkanlah $\sigma =
\pi(R_1 + R_2)^2$. (b) Untuk molekul udara ($R \approx
\qty{0.15}{nm}$, $n = \qty{2.5e25}{m^{-3}}$), hitunglah jalan bebas
rata-ratanya $1/n\sigma$, lalu bandingkanlah dengan nilai teori kinetik
jilid Tahun 1. (c) Mengapa bola keras klasik tidak menunjukkan struktur
sudut ($\dd\sigma/\dd\Omega$ tetap --- terimalah atau turunkanlah)? (d)
Ciri manakah dari hamburan bola keras \emph{kuantum}
(\cref{exo:b3:scattering-theory:6}) yang tidak mempunyai padanan klasik?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:3}
\omterm{thm:b3:scattering-theory:born}{Pemindahan momentum}. (a) Gambarlah segitiga $\vect k$, $\vect k'$ lalu
turunkanlah $q = 2k\sin(\theta/2)$. (b) Untuk alfa \qty{5.5}{MeV}
($k = \qty{1.0e15}{m^{-1}}$): berapa rentang $q$ sepanjang $\theta \in
[0, \pi]$, dan berapa struktur terkecil yang terpilah $\sim 1/q_{\max}$?
(c) Untuk elektron \qty{500}{MeV} ($k \approx E/\hbar c$): hal yang sama.
(d) Bacalah pelajarannya yang mengaitkan energi berkas dengan daya
mikroskopnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:scattering-theory:4}
Dari \cref{thm:b3:scattering-theory:amplitude}: (a) periksalah bahwa
$|f|^2$ berdimensi luas tiap sudut ruang; (b) tunjukkanlah bahwa fluks
terhambur menembus sebuah bola adalah $\Phi\sigma$; (c) mengapa $f$
umumnya harus kompleks (hukum kekekalan mana yang dijaga fasenya)? (d)
Nyatakanlah teorema optis $\sigma_{\text{tot}} =
(4\pi/k)\operatorname{Im}f(0)$ beserta maknanya (gelombang ke depannya
harus terkuras tepat sebanyak yang terhambur pergi).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:5}
Born untuk Yukawa. Dengan $V(r) = \dfrac{g\,\eu^{-\mu r}}{r}$: (a)
kerjakanlah integral Bornnya (bagian sudutnya lebih dahulu, lalu $\int_0^
\infty\sin(qr)\eu^{-\mu r}\dd r$) untuk memperoleh $f = -\dfrac{2mg}
{\hbar^2(q^2 + \mu^2)}$. (b) Biarkanlah $\mu \to 0$ dengan $g =
Q_1Q_2/4\pi\varepsilon_0$ lalu perolehlah kembali Rutherford. (c) Mengapa
\omterm{def:b3:scattering-theory:cross-section}{penampang lintang} \emph{total} Rutherfordnya membesar tanpa batas, dan
bahan fisis yang mana (penyaringan oleh elektron atomnya) yang
memulihkan jawaban berhingga? (d) Dalam fisika partikel bentuk Yukawa dengan $\mu =
m_\pi c/\hbar$ memodelkan gaya intinya: hitunglah
jangkauannya $1/\mu$ untuk $m_\pi c^2 = \qty{140}{MeV}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:6}
Bola keras kuantum (jejari $a$), gelombang $s$: di luarnya, $u_0 =
\sin(kr + \delta_0)$ dengan $u_0(a) = 0$. (a) Tunjukkanlah $\delta_0 = -ka$.
(b) Tunjukkanlah $\sigma \to 4\pi a^2$ ketika $k \to 0$: yakni
\emph{empat} kali bayangan geometrisnya. (c) Berikanlah penjelasan
gelombang bagi faktornya (difraksi tidak mempunyai bayangan yang tajam
pada panjang gelombang panjang). (d) Pada energi tinggi jawabannya
menuju $2\pi a^2$ --- masih dua kali geometrisnya (difraksi bayangan):
mengapa tak pernah sekadar $\pi a^2$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:7}
Dua puluh barn milik hidrogen. Neutron lambat tanpa polarisasi yang
mengenai proton mencuplik saluran tripletnya dengan bobot $3/4$ dan
singletnya dengan $1/4$: $\sigma = \pi(3a_{\text{t}}^2 +
a_{\text{s}}^2)$. (a) Hitunglah dengan $a_{\text{t}} = \qty{5.4}{fm}$,
$a_{\text{s}} = \qty{-23.7}{fm}$ lalu bandingkanlah dengan nilai terukur
\qty{20.4}{b}. (b) Saluran mana yang menguasainya, walaupun bobot
statistiknya lebih kecil? (c) Kaitkanlah besar dan tanda $a_{\text{t}}$
dengan ikatan deuteron yang nyaris nol; apa yang dikatakan
$a_{\text{s}} < 0$ tentang sistem singletnya? (d) Mengapa air merupakan
moderator neutron reaktor yang begitu berdaya guna --- dan mengapa
\omterm{def:b3:scattering-theory:cross-section}{penampang lintang} ini berperan baginya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:8}
Ramsauer--Townsend. Modelkanlah atom argon, bagi elektron yang datang,
sebagai sumur persegi menarik berjangkauan $R =
\qty{0.2}{nm}$. (a) Di dalamnya, bilangan gelombangnya $K =
\sqrt{2m(E + V_0)}/\hbar$:
penyesuaiannya dapat membuat gelombang luarnya muncul dengan $\delta_0 = \pi$
tepat --- berapa \omterm{def:b3:scattering-theory:cross-section}{penampang lintang} gelombang $s$-nya lalu? (b) Mengapa
atomnya menjadi nyaris tak terlihat pada energi itu padahal sumurnya
kuat? (c) Perkirakanlah $V_0$ yang menempatkan ketembusannya di dekat
$E = \qty{1}{eV}$ (buatlah sumurnya memuat setengah panjang gelombang:
$KR \approx \pi$). (d) Mengapa helium dan neon \emph{tidak} menunjukkan
efek itu pada energi yang setara (sumurnya terlalu dangkal untuk membuat
$\delta_0$ mencapai $\pi$) --- dan apa yang dibuktikan keberadaan efek
itu saja tentang gelombang materi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:scattering-theory:9}
Untuk amplitudo gelombang $s$ murni $f = \eu^{\iu\delta_0}\sin\delta_0/
k$: (a) hitunglah $\sigma$; (b) hitunglah $(4\pi/k)\operatorname{Im}
f(0)$ lalu periksalah teorema optisnya; (c) tunjukkanlah bahwa \omterm{def:b3:scattering-theory:cross-section}{penampang
lintang} gelombang $s$ maksimumnya adalah $4\pi/k^2$ (yakni batas
keuniterannya) lalu hitunglah untuk neutron termal ($k = \qty{3.1e10}{m^{-1}}$)
dalam barn; (d) mengapa sebuah resonansi tak pernah dapat mendorong
$\sigma_0$ melampaui langit-langit itu, sekuat apa pun interaksinya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:10}
Sebuah resonansi neutron. Uranium-238 menunjukkan resonansi termasyhur
bagi neutron pada $E_{\text{R}} = \qty{6.67}{eV}$ dengan lebar total
$\Gamma = \qty{27}{meV}$. (a) Hitunglah umur resonansinya.
(b) Hitunglah $4\pi/k^2$ pada energi itu, dalam barn, lalu bandingkanlah
dengan puncak terukurnya $\sim\qty{2e4}{b}$ (selisihnya adalah faktor
percabangan antara saluran elastik dan saluran penangkapannya ---
berilah ulasan). (c) Berapa lebarnya dalam satuan suhu: mengapa
pelebaran Doppler resonansi ini seiring suhu bahan bakarnya berperan
bagi kemantapan reaktornya (umpan balik bertanda apa)? (d) Dalam dua
kalimat: bagaimana resonansi semacam itu menjadikan $^{238}$U penyerap
neutron dalam rentang epitermalnya, dan mengapa hal itu pusat bagi
rancangan reaktornya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:11}
Atom dingin hidup dari satu bilangan. (a) Untuk boson identik \omterm{def:b3:scattering-theory:cross-section}{penampang
lintang} energi rendahnya adalah $8\pi a^2$ (yakni interferensi
penukaran yang membangun --- terimalah): hitunglah untuk rubidium dengan $a =
100\,a_0$. (b) Pada $T = \qty{100}{nK}$, periksalah bahwa
$ka \ll 1$ (hitunglah $k$ dari $k_{\text{B}}T = \hbar^2k^2/2m$, $m =
\qty{1.4e-25}{kg}$): gasnya sungguh melupakan segalanya kecuali $a$.
(c) Di dekat resonansi ``Feshbach'' sebuah medan magnet menyeret aras
molekulnya melewati energi nol sehingga $a$ membesar tanpa batas lalu
berganti tanda, persis seperti dalam \cref{ex:b3:scattering-theory:np}:
apa yang terjadi pada interaksi gasnya sesuka kita? (d) Mengapa
kedapattalaan ini --- kekuatan interaksi pada sebuah tombol putar ---
merupakan alat impian seorang fisikawan (sebutkanlah satu kegunaannya:
membuat molekul, mensimulasikan materi berkopling kuat, meruntuhkan
kondensat)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:scattering-theory:12}
Faktor bentuk: menimbang awan muatan. Untuk kerapatan muatan meruah
$\rho(\vect r)$, amplitudo Bornnya mengalikan jawaban titiknya dengan
$F(q) = \int\rho(\vect r)\eu^{\iu\vect q\cdot\vect r}
\dd^3r/Q$. (a) Tunjukkanlah $F(0) = 1$. (b) Jabarkanlah untuk $q$ kecil: $F
\approx 1 - q^2\langle r^2\rangle/6$: mengukur peluruhannya pada $q$
rendah menimbang $\langle r^2\rangle$. (c) Hamburan elektron--proton
mencocokkan $\sqrt{\langle r^2\rangle} \approx \qty{0.84}{fm}$:
pengukuran bebas yang mana dari
\cref{pb:b3:hydrogen-atom:1} (hidrogen muonik) yang memeriksanya silang,
dan mengapa ketaksesuaiannya dalam sejarah (``teka-teki jejari proton'')
ditanggapi begitu serius? (d) Pada $q \gg 1/r_{\text{p}}$ faktor bentuk
elastiknya runtuh namun hamburan kerasnya bertahan dari
\emph{penyusun mirip titik}: nyatakanlah dalam satu kalimat apa yang
ditemukan versi taklenting dalam percobaan Rutherford pada 1968 di dalam
protonnya.
\end{exercise}

\section{Soal: Para pencacah kilatan yang menemukan inti atom}

\begin{problem}[{Soal akhir pekan --- hamburan Rutherford, dari
geometri ke kuark}]\label{pb:b3:scattering-theory:1}
Pada 1909 Geiger dan Marsden duduk dalam gelap mencacah kilatan
sintilasi: partikel alfa yang ditembakkan menembus keping emas, dengan
beberapa memantul nyaris ke belakang --- ``seolah-olah sebutir peluru
meriam memantul dari kertas tisu''. Analisis Rutherford atas cacah itu
pada 1911 menciptakan atom berinti. Soal ini membangunnya kembali. Data:
energi alfanya $E =
\qty{5.5}{MeV}$, muatannya $z = 2$; emas $Z = 79$, tebal kepingnya $t
= \qty{0.4}{\micro m}$, $n = \qty{5.9e28}{atom/m^3}$; $k_C =
1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$.

\medskip\noindent\textbf{Bagian I --- Satu alfa, satu inti.}
\begin{enumerate}
  \item Dalam gambaran puding kismis (muatannya tersebar pada atomnya,
        \qty{e-10}{m}), perkirakanlah pembelokan maksimum yang dapat
        diberikan satu atom kepada alfa \qty{5.5}{MeV} (medan sebuah
        bola yang terlumat: $\theta \sim$ gaya $\times$ waktu / momentum
        $\sim 2Zze^2k_C/(E\cdot\qty{e-10}{m})$ dalam radian). Dapatkah
        penumpukan sentakan semacam itu mengirimkan alfa ke belakang?
  \item Kini biarkanlah muatannya berupa sebuah titik. Tuliskanlah
        jarak pendekatan terdekat $d_0$ untuk tumbukan berhadapan
        (seluruh energi kinetiknya menjadi energi Coulomb) lalu
        hitunglah.
  \item Karena itu, apa yang \emph{langsung} disiratkan laju hamburan
        nyaris ke belakang yang terukur tentang seberapa terpusat muatan
        positif atomnya?
  \item Untuk parameter dampak $b$ yang umum, orbit klasiknya
        memberikan $\tan(\theta/2) = d_0/2b$ (diterima tanpa bukti ---
        yakni hiperbola pada bab gravitasi jilid Tahun 1, dengan
        tolakannya): periksalah limitnya pada $b \to 0$ dan $b \to
        \infty$.
  \item Hitunglah parameter dampak yang menghamburkan sebuah alfa lebih
        dari $90^\circ$, beserta bagian alfa yang melewati kepingnya
        yang mendekati sebuah inti dalam jarak itu ($n t\,\pi
        b^2$).
  \item Geiger dan Marsden mendapati sekitar 1 dari 8000 alfa terbelokkan
        melampaui $90^\circ$: bandingkanlah.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- \omterm{def:b3:scattering-theory:cross-section}{Penampang lintangnya}.}
\begin{enumerate}[resume]
  \item Alfa dengan parameter dampak dalam $[b, b + \dd b]$ terhambur
        ke dalam $[\theta, \theta + \dd\theta]$: dari $\dd\sigma =
        2\pi b\,|\dd b|$ dan hubungan orbitnya, turunkanlah
        \[
        \frac{\dd\sigma}{\dd\Omega} =
        \Big(\frac{d_0}{4}\Big)^2\frac{1}{\sin^4(\theta/2)} .
        \]
  \item Periksalah bahwa ini cocok dengan jawaban Born atau kuantum pada
        \cref{ex:b3:scattering-theory:rutherford} (tuliskanlah ulang
        $d_0$ dalam $E$).
  \item Hitunglah $\dd\sigma/\dd\Omega$ pada $\theta = 30^\circ$,
        $60^\circ$, $120^\circ$ (dalam barn tiap steradian).
  \item Cacah Geiger dan Marsden pada 1913 dengan geometri tetap
        berskala sebagai $1{:}\,\sin^{-4}(\theta/2)$: hitunglah nisbah
        cacah yang diramalkan antara $30^\circ$ dan $120^\circ$ lalu
        perhatikanlah bahwa datanya mengikuti nisbah semacam itu
        sepanjang lima orde besar laju.
  \item Mengapa $Z^2$ dalam rumus yang sama membiarkan percobaannya
        \emph{menimbang} muatan intinya --- dan bagaimana pencocokan
        semacam itu membantu memancangkan $Z$ (emas: 79) sebagai nomor
        atomnya?
  \item Perhatikan $1/E^2$: apa yang terjadi pada seluruh pola sudutnya
        ketika energi alfanya dilipatduakan?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Ukuran intinya.}
\begin{enumerate}[resume]
  \item Pendekatan terdekat pada sudut $\theta$: $d(\theta) =
        \tfrac{d_0}2[1 + 1/\sin(\theta/2)]$ (diterima tanpa bukti).
        Hitunglah untuk $\theta = 180^\circ$ dan \qty{60}{\degree}.
  \item Selama $d(\theta)$ melampaui jejari intinya, rumus muatan
        titiknya berlaku: batas atas ukuran inti emas yang mana yang
        ditegakkan titik $180^\circ$ Rutherford yang terbenarkan itu?
  \item Dengan proyektil berenergi lebih tinggi, cacah pada sudut besar
        \emph{jatuh di bawah} nilai Rutherford: apa yang ditandakan
        penyimpangan itu (gaya baru yang mana, pada jarak berapa)?
  \item Hamburan elektron modern memberikan jejari inti $R
        \approx r_0A^{1/3}$ dengan $r_0 = \qty{1.2}{fm}$: hitunglah
        untuk emas ($A = 197$) lalu bandingkanlah dengan batas Anda.
  \item Perhatikan $A^{1/3}$: apa yang dikatakannya tentang kerapatan
        materi inti (tetap? tumbuh?) --- yakni fakta yang akan menjadi
        pijakan \cref{ch:b3:nuclear-physics}.
  \item Mengapa model puding kismis (yang buta terhadap muatan) mati
        justru oleh percobaan \emph{ini} alih-alih oleh spektroskopi?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Cucu-cucu Rutherford.}
\begin{enumerate}[resume]
  \item Daftarkanlah tabel terjemahan yang tepat antara 1911 dan 1968:
        alfa $\to$ berkas elektron, atom emas $\to$ proton,
        inti $\to$ kuark --- apa yang berperan sebagai ``peristiwa
        bersudut besar yang tak terduga'' di SLAC?
  \item Mengapa \emph{elektron} merupakan penyelidik yang lebih bersih
        (kerumitan hamburan alfa--inti yang mana yang tidak dimilikinya)?
  \item Dalam bahasa Born: hamburan elastik pada $q$ tinggi mengukur
        runtuhnya faktor bentuknya (yakni awan yang lunak), sedangkan
        laju taklentingnya tetap besar dan nyaris mirip titik:
        nyatakanlah kesimpulan yang ditarik.
  \item Busur satu abad dalam tiga baris: berikanlah (energi
        penyelidiknya, jarak yang terpilah) untuk alfa 1911
        ($\sim\qty{e-14}{m}$), SLAC 1968 (\qty{20}{GeV},
        $\sim\qty{e-16}{m}$), dan LHC (skala TeV, $\sim\qty{e-19}{m}$)
        --- dengan memakai kaidah ``resolusinya $\sim \hbar c/(qc)$''.
  \item Neutron, yang tak bermuatan, hanya terhambur oleh inti (dan oleh
        momen magnetik): sebutkanlah dua hal yang karenanya dipetakan
        hamburan neutron dalam bahan yang kurang terlihat sinar-X (atom
        ringan seperti hidrogen; penataan magnetik).
  \item \omterm{def:b3:scattering-theory:cross-section}{Penampang lintang} penemuan di LHC diukur dalam
        \emph{femtobarn} ($\qty{e-43}{m^2}$): dari barn ke femtobarn
        terbentang lima belas orde besar --- apa yang dikatakannya
        tentang betapa langkanya tumbukan yang menarik, dan mengapa
        luminositas (fluks yang diantarkan) sama berharganya dengan
        energinya?
  \item Ringkaslah hasil bernama itu: sebuah hukum
        $\sin^{-4}(\theta/2)$, yang dibenarkan kilatan demi kilatan,
        menempatkan $99.98\%$ massa atomnya dalam volume $10^{-14}$
        kali ukurannya --- dan percobaan yang sama, yang diulang makin
        keras, tak pernah berhenti menemukan lapisan berikutnya.
\end{enumerate}
\end{problem}
