\chapter{Statistika Kuantum}\label{ch:b3:quantum-statistics}

Elektron hantaran tembaga adalah gas yang seribu kali lebih rapat
daripada udara --- namun elektronnya nyaris tak menambah apa pun pada
kapasitas kalor tembaga, yakni skandal yang menghantui fisika klasik
selama lima puluh tahun. Awan atom rubidium, yang didinginkan sampai
seperseratus miliar kelvin, tiba-tiba runtuh menjadi satu keadaan
kuantum tunggal setebal sepuluh ribu atom. Sebuah bintang mati bermassa
Matahari dan berukuran Bumi menolak menyusut lebih jauh, yang ditahan
oleh tak lain daripada asas pengecualian. Ketiganya adalah bab fisika
yang sama: apa yang terjadi pada sebuah gas ketika $n\lambda_T^3 \sim 1$
dan rumus keterisian \cref{ch:b3:grand-canonical} berpisah dari
Boltzmann. Fermion membangun lautan --- dingin, kaku, bertekanan; boson
membangun kondensat --- suka berkumpul, koheren, adiluncur. Di antara
mayoritas diam laut Fermi dan gempuran kondensatnya terbentang sebagian
besar fisika benda rapat dan suhu rendah masa kini, beserta nasib
bintang.

\section{Gas keterisian}

\begin{proposition}[Permesinan gas kuantum]\label{prop:b3:quantum-statistics:machinery}
\index{kerapatan keadaan}
Untuk partikel yang tak berinteraksi dalam sebuah kotak, jumlahan atas
arasnya menjadi integral atas \omterm{prop:b3:schrodinger-three-dimensions:counting}{kerapatan keadaan}
\cref{prop:b3:schrodinger-three-dimensions:counting} (tiap keadaan
spinnya, $g(\epsilon) = (V/4\pi^2)(2m/\hbar^2)^{3/2}\sqrt\epsilon$):
\[
N = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\dd\epsilon ,
\qquad
E = \int_0^\infty g(\epsilon)\,\bar n(\epsilon)\,\epsilon\,
\dd\epsilon ,
\]
dengan $\bar n$ sebagai keterisian Fermi--Dirac atau Bose--Einstein.
Persamaan pertamanya menetapkan $\mu(T, n)$; yang kedua menyerahkan
termodinamikanya. Dalam limit encer $n\lambda_T^3 \ll 1$ kedua
statistikanya runtuh menjadi Boltzmann dan gas klasiknya kembali;
rezim kuantumnya mulai ketika paket gelombangnya bersentuhan.
\end{proposition}
\admitted

\section{Laut Fermi}

\begin{theorem}[Gas Fermi yang merosot]\label{thm:b3:quantum-statistics:fermi}
\index{energi Fermi}\index{tekanan kemerosotan}
Pada $T = 0$, fermion mengisi tiap arasnya sampai ke \emph{\omterm{thm:b3:quantum-statistics:fermi}{energi
Fermi}}
\[
E_{\text{F}} = \frac{\hbar^2}{2m}\,(3\pi^2n)^{2/3}
\qquad (\text{spin } \tfrac12) ,
\]
dengan energi rata-rata $\tfrac35 E_{\text{F}}$ tiap partikelnya dan
\emph{\omterm{thm:b3:quantum-statistics:fermi}{tekanan kemerosotan}}
\[
P = \frac25\,n\,E_{\text{F}} \;\propto\; n^{5/3} :
\]
yakni sebuah tekanan pada nol mutlak, yang murni berasal kuantum ---
kekakuan gas elektron tiap logam dan penopang bintang mati. Pada
$0 < T \ll T_{\text{F}} = E_{\text{F}}/k_{\text{B}}$,
hanya elektron dalam jarak $\sim k_{\text{B}}T$ dari permukaan
\emph{laut Fermi} ini yang dapat menanggapi apa pun: bagian
$\sim T/T_{\text{F}}$ adalah kunci induk perilaku logam.
\end{theorem}

\begin{proof}[Bukti sebagian]
Isilah $g(\epsilon)$ sampai ke $E_{\text{F}}$: membalikkan $N = \int_0^{
E_{\text{F}}}g\,\dd\epsilon$ memberikan $E_{\text{F}}$ yang dinyatakan
itu (\cref{exo:b3:schrodinger-three-dimensions:8}); $\langle\epsilon
\rangle = \tfrac35 E_{\text{F}}$ lewat satu integral lagi. Tekanannya:
memampatkan kotaknya menaikkan tiap arasnya sebagai $V^{-2/3}$, sehingga $E
\propto V^{-2/3}$ dan $P = -\partial E/\partial V = \tfrac23 E/V$
--- dengan $E = \tfrac35 NE_{\text{F}}$, diperoleh hukum yang dinyatakan itu.
\end{proof}

\begin{example}[Skandal kapasitas kalor, terselesaikan]\label{ex:b3:quantum-statistics:heat}
\index{kapasitas kalor elektronik}
Secara klasik, satu elektron bebas tiap atom tembaga seharusnya
menambahkan $\tfrac32
k_{\text{B}}$ pada kapasitas kalornya --- yakni $50\%$ di atas
Dulong--Petit. Terukur: kira-kira $1\%$ pada suhu ruang. Lautnya
menjelaskan: dengan $T_{\text{F}} \approx \qty{8e4}{K}$, hanya
bagian $\sim T/T_{\text{F}} \approx 0.4\%$ elektronnya yang dapat
tereksitasi secara termal, masing-masing sebesar $\sim k_{\text{B}}T$:
\[
C_{\text{el}} \approx \frac{\pi^2}{2}\,
Nk_{\text{B}}\,\frac{T}{T_{\text{F}}}
\]
(yakni koefisien Sommerfeld yang eksak;
\cref{exo:b3:quantum-statistics:12}). Karena lurus dalam $T$, ia terkubur
di bawah $T^3$ kisinya pada suhu biasa tetapi
\emph{menguasai} di bawah beberapa kelvin --- persis seperti yang
ditemukan kalorimeter, sehingga terselesaikanlah dalam satu tebasan
teka-teki elektron yang mengangkut arus tetapi bukan kapasitas kalor.
\end{example}

\begin{example}[Bintang yang ditopang pengecualian]\label{ex:b3:quantum-statistics:dwarf}
\index{katai putih}
Sebuah \omterm{ex:b3:quantum-statistics:dwarf}{katai putih} (\cref{exo:b3:identical-particles:11})
menyetimbangkan gravitasi terhadap \omterm{thm:b3:quantum-statistics:fermi}{tekanan kemerosotan} elektron. Karena $P \propto
n^{5/3}$ mengaku lebih cepat daripada tuntutan gravitasi ketika
bintangnya menyusut, jejari kesetimbangan ada pada tiap massa yang
sederhana --- dengan penskalaan yang mengerikan
$R \propto M^{-1/3}$: \emph{katai yang lebih berat justru lebih
kecil}. Bila diperas lebih keras, elektronnya menjadi relativistik,
hukum tekanannya melunak menjadi $n^{4/3}$, dan kesetimbangannya gagal
pada satu massa tunggal, yakni batas Chandrasekhar $M_{\text{Ch}} \approx
1.4\,M_\odot$ (\cref{pb:b3:quantum-statistics:1}) --- di luar itu,
keruntuhan menjadi bintang neutron: yakni laut Fermi neutron pada
kerapatan inti, fisika yang sama pada kerapatan $10^{14}$ kali lipat.
\end{example}

\section{Gempuran Bose}

\begin{theorem}[Kondensasi Bose--Einstein]\label{thm:b3:quantum-statistics:bec}
\index{kondensasi Bose--Einstein}
Untuk gas berisi $N$ boson yang kekal, keadaan tereksitasinya dapat
menampung paling banyak $N_{\max}(T) = 2.612\,V/\lambda_T^3$ partikel
(yakni integral $g\bar n_{\text{Bose}}$ pada langit-langitnya $\mu \to 0$). Di bawah
suhu gentingnya
\[
k_{\text{B}}T_{\text{c}} = \frac{2\pi\hbar^2}{m}
\Big(\frac{n}{2.612}\Big)^{2/3}
\quad\text{(setara dengan } n\lambda_{T_{\text{c}}}^3 = 2.612
\text{)} ,
\]
kelebihannya tak punya tempat lain selain keadaan dasar tunggalnya:
yakni populasi \emph{makroskopis}
\[
\frac{N_0}{N} = 1 - \Big(\frac{T}{T_{\text{c}}}\Big)^{3/2}
\]
memampat menjadi satu keadaan kuantum --- yakni materi yang berperilaku
sebagai satu fungsi gelombang raksasa. Diramalkan Einstein pada 1925 dari
statistika foton Bose; diwujudkan dalam rubidium encer pada 1995 pada
$T_{\text{c}} \approx \qty{100}{nK}$ (Nobel 2001), yang diumumkan oleh
sebaran kecepatan \emph{dwipuncak} yang khas: yakni awan termal
dengan jarum kondensat berkecepatan nol yang menyembul dari pusatnya.
\end{theorem}
\admitted

\begin{example}[Helium, sang adiluncur veteran]\label{ex:b3:quantum-statistics:helium}
\index{keadiluncuran}
Helium $^4$He cair (bersifat boson) berubah pada $T_\lambda = \qty{2.17}{K}$
menjadi \emph{adiluncur}: tanpa kekentalan menembus pipa kapiler
terhalus, aliran selaput menaiki dinding wadahnya, gejala pancuran, dan
peredaran yang terkuantumkan dalam satuan $h/m$ --- yakni fase
makroskopis sebuah kondensat yang sedang bekerja, yang tergeser dari
ramalan gas idealnya ($\qty{3.1}{K}$ untuk kerapatan helium) oleh
interaksi tetapi tak salah lagi merupakan gejala yang sama. Helium $^3$He
yang bersifat fermion tak dapat memampat sendirian: atomnya harus
\emph{berpasangan} terlebih dahulu (seperti elektron sebuah
adikonduktor, \cref{ch:b3:electrons-in-solids}) menjadi boson
gabungan --- dan keadiluncurannya pun tiba, tetapi pada \qty{2.6}{mK},
seribu kali lebih dingin: dua isotop, satu pelajaran --- statistika
adalah takdir.
\end{example}

\begin{method}[Statistika mana, rezim mana]\label{met:b3:quantum-statistics:method}
(1) Hitunglah $n\lambda_T^3$ lebih dahulu: $\ll1$ berarti Boltzmann dan
bab 17 sudah memadai. (2) Fermion yang merosot: berpikirlah dalam
$E_{\text{F}}$ dan $T/T_{\text{F}}$; tanggapannya (kapasitas kalor,
kerentanan, hamburan) membawa faktor $T/T_{\text{F}}$ atau hanya
mencuplik permukaan Ferminya. (3) Boson yang merosot: bandingkanlah $T$
dengan $T_{\text{c}}$; di bawahnya, belahlah gasnya menjadi kondensat
ditambah awan termal. (4) Tekanan: kekakuan fermionik $n^{5/3}$ (atau
$n^{4/3}$ relativistik) melawan apa pun yang memerasnya. (5) Partikel
gabungan: cacahlah fermion penyusunnya untuk memilih statistikanya ---
dan ingatlah bahwa pemasangan dapat mengubah statistika yang satu menjadi yang lain.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:1}
Hitunglah $E_{\text{F}}$ dan $T_{\text{F}}$ untuk (a) tembaga
($n = \qty{8.5e28}{m^{-3}}$); (b) natrium ($\qty{2.5e28}{m^{-3}}$);
(c) $^3$He cair ($\qty{1.6e28}{m^{-3}}$, $m = 3u$); (d) urutkanlah
terhadap suhu ruang lalu berilah ulasan tentang siapa merosot kapan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:2}
(a) Tunjukkanlah $\langle\epsilon\rangle = \tfrac35 E_{\text{F}}$ pada $T =
0$. (b) Hitunglah laju akar rata-rata kuadrat elektron dalam tembaga.
(c) Laju itu secara klasik bersesuaian dengan suhu berapa? (d) Mengapa
logam yang dingin memuat elektron yang bergerak seperseratus laju cahaya
tanpa memancarkan energi itu keluar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:3}
\omterm{thm:b3:quantum-statistics:fermi}{Tekanan kemerosotan} elektron tembaga: (a) hitunglah $P =
\tfrac25 nE_{\text{F}}$ dalam pascal dan atmosfer. (b) Mengapa
logamnya tidak meledak (apa yang menarik balik)? (c) Tekanan ini
melawan pemampatan: kaitkanlah dengan rendahnya kemampatan logam. (d)
Perkirakanlah modulus limbaknya $K \sim \tfrac53 P$ lalu bandingkanlah
dengan nilai terukur tembaga \qty{140}{GPa}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:quantum-statistics:4}
Ambang BEC. (a) Hitunglah $T_{\text{c}}$ untuk rubidium-87 ($m =
\qty{1.44e-25}{kg}$) pada $n = \qty{e20}{m^{-3}}$. (b) Untuk $n$ yang
sama, $T_{\text{c}}$ berapa yang akan dimiliki \omterm{thm:b3:hydrogen-atom:levels}{atom hidrogen}? (c)
Mengapa percobaan memakai gas encer pada nanokelvin dan bukan gas rapat
pada milikelvin (apa yang terjadi secara kimiawi pada kerapatan tinggi
dan suhu rendah)? (d) Periksalah aturan ketaatasasan $n\lambda_{T_c}^3
= 2.612$ pada bilangan rubidium Anda.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:5}
Dua kapasitas kalor. Dalam sebuah logam, $C = \gamma T +
\beta T^3$ (elektronnya; kisinya). (a) Dengan memakai
\cref{ex:b3:quantum-statistics:heat} dan suku kisi bergaya Debye
$\beta T^3 = 234\,Nk_{\text{B}}(T/\theta_{\text{D}})^3$,
perkirakanlah suhu perpindahannya untuk tembaga
($\theta_{\text{D}} = \qty{343}{K}$, $T_{\text{F}} =
\qty{8.2e4}{K}$). (b) Pada \qty{300}{K}, berapa bagian elektronnya dalam
persen. (c) Pada \qty{0.1}{K}? (d) Mengapa para pelaku percobaan
memplot $C/T$ terhadap $T^2$, dan apa yang diserahkan oleh titik potong
serta kemiringan garis lurus itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:6}
Sebuah \omterm{ex:b3:quantum-statistics:dwarf}{katai putih} bermassa satu matahari ($M = \qty{2e30}{kg}$) dan
berjejari Bumi ($R = \qty{6.4e6}{m}$), dengan satu elektron tiap dua
nukleonnya. (a) Hitunglah $n_{\text{e}}$ dan $E_{\text{F}}$ (dengan
rumus taknisbian). (b) Bandingkanlah $E_{\text{F}}$ dengan $m_{\text{e}}c^2$ dan
dengan $k_{\text{B}}T$ pada \qty{e7}{K} bagian dalamnya: benarkanlah
sebutan ``dingin'' dan ``nyaris relativistik'' sekaligus. (c)
Bandingkanlah \omterm{thm:b3:quantum-statistics:fermi}{tekanan kemerosotannya} $\tfrac25 nE_{\text{F}}$ dengan
perkiraan tekanan gravitasinya $\sim GM^2/R^4$: seorde? (d) Apa yang
terjadi secara kualitatif pada tiap sisi kesetimbangannya jika $M$ tumbuh?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:7}
Materi neutron. Sebuah bintang neutron memampatkan $M = 1.4\,M_\odot$ ke dalam $R
\approx \qty{11}{km}$. (a) Hitunglah kerapatan neutronnya lalu
bandingkanlah dengan kerapatan inti ($\qty{2.3e17}{kg/m^3}$). (b)
Hitunglah $E_{\text{F}}$ untuk neutronnya (secara taknisbian; $m_{\text{n}}c^2
= \qty{939}{MeV}$) lalu periksalah seberapa relativistik mereka. (c)
Mengapa elektron bintangnya lenyap (peluruhan beta terbalik:
$\text{e} + \text{p} \to \text{n} + \nu$ --- apa yang membuat
$E_{\text{F}}$ elektronnya membayarnya)? (d) Satu sendok teh materi ini
seberat apa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:8}
Boson tanpa cacah. (a) Foton selalu memiliki $\mu = 0$: mengapa mereka
tak dapat mengalami pemampatan
\cref{thm:b3:quantum-statistics:bec} (apa yang dilakukan gasnya sebagai
gantinya ketika didinginkan)? (b) Hukum kekekalan mana yang dimiliki
atom rubidium tetapi tak dimiliki foton? (c) Pada 2010 sebuah BEC foton
\emph{berhasil} dibuat, dalam rongga berisi zat warna tempat fotonnya
menermalkan diri dan cacahnya praktis tetap: bahan mana yang dipulihkan
oleh rongga zat warna itu?
(d) Fonon dalam sebuah kristal: memampat atau tidak, dan mengapa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:quantum-statistics:9}
Dua isotop helium. (a) Hitunglah $T_{\text{c}}$ gas idealnya
untuk $^4$He cair ($n = \qty{2.2e28}{m^{-3}}$, $m = 4u$) lalu
bandingkanlah dengan $T_\lambda = \qty{2.17}{K}$. (b) Mengapa
kesesuaiannya hanya kasar (apa yang diabaikan ``ideal'' dalam sebuah
cairan)? (c) $^3$He pada kerapatan yang nyaris sama tetap normal sampai
ke milikelvin: nyatakanlah rintangannya beserta jalan keluar yang
ditemukan alam. (d) Pemeriksaan penalaran gabungan: sebuah \emph{pasangan} $^3$He memuat cacah
fermion yang genap --- boson; $^6$Li menahan $3\text{p} + 3\text{n}
+ 3\text{e} = 9$ fermion: golongkanlah ia, lalu sebutkanlah percobaan
gas dingin termasyhur yang memanfaatkan persis pertentangan Li--Li ini.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:10}
$T_{\text{c}}$ secara jujur. (a) Tuliskanlah $N_{\text{exc}} = \int_0^\infty
g(\epsilon)\dd\epsilon/(\eu^{\beta\epsilon} - 1)$ pada $\mu = 0$ lalu
susutkanlah, dengan $x = \beta\epsilon$, menjadi $N_{\text{exc}} =
(V/\lambda_T^3)\cdot\frac{2}{\sqrt\pi}\int_0^\infty\frac{\sqrt
x\,\dd x}{\eu^x - 1}$. (b) Dengan diberikan $\int_0^\infty\sqrt x\,\dd x/(
\eu^x - 1) = 2.612\,\sqrt\pi/2$, perolehlah langit-langitnya dan
$T_{\text{c}}$. (c) Turunkanlah bagian kondensatnya $1 -
(T/T_{\text{c}})^{3/2}$. (d) Mengapa penalaran yang sama menghasilkan
\emph{tanpa} pemampatan dalam kotak berdimensi dua (bagaimana $g(
\epsilon)$ berubah, dan apa yang terjadi pada penyatuan
integralnya)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:11}
Mengapa logam bertahan bersama. Modelkanlah sebuah logam sebagai
elektron berkerapatan
$n$ (dengan ongkos \omterm{thm:b3:quantum-statistics:fermi}{energi Fermi} $\propto n^{2/3}$ tiap elektronnya) yang
tertarik ke latar ion penetral (dengan perolehan Coulomb tiap elektronnya
$\sim -e^2n^{1/3}/4\pi\varepsilon_0$). (a) Tuliskanlah energi tiap
elektronnya $\epsilon(n) = An^{2/3} - Bn^{1/3}$ lalu tunjukkanlah bahwa
ia memiliki minimum. (b) Tunjukkanlah bahwa jarak elektron
kesetimbangannya berorde $a_0$ --- kerapatan logam bukanlah kebetulan.
(c) Perkirakanlah skala energi kelekatannya pada minimumnya, dalam eV.
(d) Peran apa yang dimainkan \omterm{thm:b3:quantum-statistics:fermi}{tekanan kemerosotan} dalam keadaan terikat
ini (apa yang menghentikan keruntuhan Coulombnya)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:quantum-statistics:12}
Kulit Sommerfeld, secara kuantitatif. (a) Berargumenlah: dari $N$
elektron, hanya $\sim g(E_{\text{F}})k_{\text{B}}T$ yang duduk dalam
kulit aktifnya, masing-masing menahan energi tambahan
$\sim k_{\text{B}}T$: turunkanlah $C_{\text{el}}
\sim g(E_{\text{F}})k_{\text{B}}^2T$ sampai sebuah tetapan. (b) Dengan
$g(E_{\text{F}}) = 3N/2E_{\text{F}}$, perolehlah kembali $C_{\text{el}} \sim
Nk_{\text{B}}T/T_{\text{F}}$ (tetapan eksaknya adalah $\pi^2/2$).
(c) Penalaran kulit yang sama memberikan kerentanan spin Pauli yang tak
bergantung suhu: hanya elektron kulitnya yang dapat berbalik --- tetapi
kulitnya tumbuh sebagai $T$ sedangkan tiap sumbangannya turun sebagai $1/T$:
tunjukkanlah peniadaannya. (d) Nyatakanlah pelajaran umumnya bagi
fermion yang merosot: \emph{tiap} tanggapannya adalah tanggapan
klasiknya dikalikan $\sim T/T_{\text{F}}$ atau padanan hanya-permukaannya.
\end{exercise}

\section{Soal: Katai putih terberat yang mungkin}

\begin{problem}[{Soal akhir pekan --- batas Chandrasekhar, dan lilin
yang dinyalakannya}]\label{pb:b3:quantum-statistics:1}
Pada 1930, dalam usia sembilan belas tahun, Subrahmanyan Chandrasekhar
menghitung di atas kapal menuju Inggris bahwa kemerosotan elektron hanya
dapat menopang bintang mati sampai suatu massa tertentu. Hasilnya ---
yang ditolak keras oleh Eddington --- mendasari bintang neutron, lubang
hitam dan ``lilin baku'' supernova yang menyingkapkan alam semesta yang
memecut. Soal ini membangunnya lewat penskalaan. Data: $M_\odot =
\qty{2e30}{kg}$; satu elektron tiap $\mu_e = 2$ nukleon; $m_{\text{
n}} = \qty{1.67e-27}{kg}$; $\hbar c = \qty{3.16e-26}{J.m}$; $G =
\qty{6.67e-11}{N.m^2/kg^2}$.

\medskip\noindent\textbf{Bagian I --- Kesetimbangan dua energi.}
Untuk bintang bermassa $M$ dan berjejari $R$, tanganilah energi tiap
satuan massanya lewat penskalaan.
\begin{enumerate}
  \item Energi gravitasinya: $E_{\text{g}} \sim -GM^2/R$ ---
        ingatlah kembali asal-usulnya.
  \item Cacah elektronnya $N_{\text{e}} = M/\mu_e m_{\text{n}}$ dan
        kerapatannya $n \sim N_{\text{e}}/R^3$: tuliskanlah skala
        \omterm{thm:b3:quantum-statistics:fermi}{energi Fermi} taknisbiannya sebagai fungsi $M$
        dan $R$.
  \item Tunjukkanlah bahwa energi kinetik (kemerosotan) totalnya
        berskala sebagai
        $E_{\text{k}} \sim \hbar^2N_{\text{e}}^{5/3}/m_{\text{e}}
        R^2$.
  \item Minimumkanlah $E_{\text{k}} + E_{\text{g}}$ terhadap $R$ lalu
        tunjukkanlah bahwa jejari kesetimbangannya adalah
        $R \sim \hbar^2 N_{\text{e}}^{-1/3}/(G\,m_{\text{e}}\,
        m_{\text{n}}^2\mu_e^2)$.
  \item Bacalah penskalaannya $R \propto M^{-1/3}$: katai yang lebih
        berat justru lebih \emph{kecil}. Mengapa hal itu sudah berbau
        masalah?
  \item Hitunglah $R$ untuk $M = 1\,M_\odot$ lalu bandingkanlah dengan
        nilai terukur Sirius B $\approx\qty{5800}{km}$.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Kenisbian menarik lantainya
pergi.}
\begin{enumerate}[resume]
  \item Ketika $R$ menyusut, $E_{\text{F}}$ tumbuh: perkirakanlah
        kerapatan tempat $E_{\text{F}} \sim m_{\text{e}}c^2$, beserta
        daerah massa katai yang bersesuaian.
  \item Dalam rezim ultrarelativistiknya, energi tiap elektronnya
        adalah $\sim\hbar c\,n^{1/3}$: tunjukkanlah bahwa energi
        kinetiknya menjadi
        $E_{\text{k}} \sim \hbar c\,N_{\text{e}}^{4/3}/R$.
  \item Kedua energinya kini berskala sebagai $1/R$: simpulkanlah
        bahwa kesetimbangannya tak lagi memilih sebuah jejari ---
        bandingkanlah kedua koefisiennya sebagai gantinya.
  \item Dengan menetapkan $\hbar c\,N_{\text{e}}^{4/3} \sim
        GM^2 = G(\mu_em_{\text{n}}N_{\text{e}})^2$, turunkanlah
        cacah elektron gentingnya beserta massanya
        \[
        M_{\text{Ch}} \sim
        \Big(\frac{\hbar c}{G\,m_{\text{n}}^2}\Big)^{3/2}
        \frac{m_{\text{n}}}{\mu_e^2} .
        \]
  \item Hitunglah nisbah tak berdimensi $\hbar c/Gm_{\text{n}}^2$
        --- salah satu bilangan murni terbesar dalam fisika --- lalu
        $M_{\text{Ch}}$ dalam massa matahari (penanganan eksaknya
        memberikan $1.4\,M_\odot$).
  \item Tafsirkanlah rumusnya: tiga tetapan mana yang bersekongkol, dan
        mengapa sebuah batas \emph{kuantum} pada sebuah \emph{bintang}
        membawa $\hbar$?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Melampaui batasnya.}
\begin{enumerate}[resume]
  \item Sebuah katai yang didorong melewati $M_{\text{Ch}}$ (lewat
        pengumpulan massa dari pasangannya): apa yang menopangnya
        kemudian, dan pada skala jejari berapa (gantilah $m_{\text{e}}$
        dengan $m_{\text{n}}$ dan $\mu_e$ dengan 1 dalam jejari Bagian I)?
  \item Tunjukkanlah bahwa batas relativistik bintang neutronnya sendiri
        seorde dengan $M_{\text{Ch}}$ --- alam hanya mengizinkan
        keringanan sebesar dua kali lipat (maksimum teramati $\approx
        2.2\,M_\odot$). Apa yang menanti di baliknya?
  \item Selama keruntuhan menjadi bintang neutron, energi gravitasi
        sebesar $\sim GM^2/R_{
        \text{ns}}$ terlepas: hitunglah
        untuk $M = 1.4\,M_\odot$, $R = \qty{11}{km}$, lalu
        bandingkanlah dengan seluruh keluaran Matahari selama \qty{10}{Gyr}
        ($\sim\qty{1.2e44}{J}$).
  \item Hampir seluruhnya pergi sebagai neutrino dalam hitungan detik:
        pengamatan tahun 1987 mana yang membenarkan gambaran ini secara
        menakjubkan?
  \item Pantulan teras yang runtuh beserta dorongan neutrinonya meniup
        pergi selubungnya: yakni supernova keruntuhan teras. Bedakanlah
        ia, masing-masing dalam satu kalimat, dari supernova
        \emph{termonuklir} (jenis Ia) sebuah \omterm{ex:b3:quantum-statistics:dwarf}{katai putih} yang menyala
        pada ambang Chandrasekhar.
  \item Mengapa peledakan \emph{pada satu massa yang semesta} membuat
        supernova jenis Ia nyaris seragam --- sehingga dapat dipakai
        sebagai lilin baku?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Lilin dan
kosmosnya.}
\begin{enumerate}[resume]
  \item Kecerlangan tampak sebuah lilin baku memberikan jaraknya:
        nyatakanlah penalaran kuadrat terbaliknya dan mengapa
        \emph{baku} adalah kata kuncinya.
  \item Pada 1998 dua tim menemukan supernova jenis Ia yang jauh
        \emph{lebih redup} daripada yang diharapkan: nyatakanlah
        simpulannya (pemuaian yang memecut --- energi gelap) dan mengapa
        keterandalan lilinnya menjadi pokok perkaranya.
  \item Daftarkanlah rantai soal ini dalam satu baris: Pauli $\to$
        \omterm{thm:b3:quantum-statistics:fermi}{tekanan kemerosotan} $\to$ $M_{\text{Ch}}$ $\to$ ledakan yang
        seragam $\to$ pemecutan kosmik.
  \item Eddington mengejek hasilnya (``sebuah bintang yang melakukan
        kemustahilan''); Chandrasekhar menerima Hadiah Nobel lima puluh
        tiga tahun kemudian: apa yang diajarkan babak itu tentang
        mengikuti persamaan melewati kesimpulan yang nyaman?
  \item Spektrum Sirius B pada 1915 mengisyaratkan satu ton tiap
        sentimeter kubik --- ``omong kosong'' sampai Fowler, dalam
        hitungan bulan setelah statistika Fermi tahun 1926, menjelaskan
        bintangnya sebagai gas yang merosot: nyatakanlah mengapa
        teka-tekinya tak terpecahkan pada satu tahun lalu menjadi biasa
        pada tahun berikutnya.
  \item Survei atas ribuan \omterm{ex:b3:quantum-statistics:dwarf}{katai putih} menemukan massa yang
        bergerombol di dekat $0.6\,M_\odot$ tanpa satu pun di atas
        $\approx1.4\,M_\odot$: dua ramalan soal ini yang mana yang
        dibenarkan oleh satu histogram itu?
  \item Ringkaslah hasil bernama itu: $(\hbar c/Gm_{\text{n}}^2)^{
        3/2}m_{\text{n}} \approx 1.4\,M_\odot$ --- yakni tiga tetapan
        alam yang menetapkan \omterm{ex:b3:quantum-statistics:dwarf}{katai putih} terberat yang mungkin, yaitu
        massa pemicu ledakan alam semesta yang terkalibrasi.
\end{enumerate}
\end{problem}
