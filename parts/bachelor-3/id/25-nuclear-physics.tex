Jilid sekolah menengah menceritakan garis besar kisahnya: atom memiliki
inti; sebagian intinya runtuh, menurut jadwal dari mikrodetik sampai
miliaran tahun; dan beberapa gram materi menyembunyikan megaton energi.
Bab ini membuka kembali intinya dengan perkakas Tahun 3. Sebuah model
tetesan berisi lima suku meramalkan ikatan ratusan \omterm{def:b3:nuclear-physics:nuclide}{nuklida} sampai
ketelitian persen; penerowongan kuantum --- yakni masalah penghalang
jilid Tahun 2, yang dinaikkan pangkatnya --- menjelaskan bagaimana satu
rumus membentang dua puluh lima orde besar dalam umur peluruhan
$\alpha$; dan maksimum yang tenang pada kurva \omterm{prop:b3:nuclear-physics:binding}{energi ikatnya} di besi
membagi seluruh teknologi inti menjadi dua kubu: membelah yang berat
(reaktor) atau menggabungkan yang ringan (bintang, dan suatu hari nanti
pembangkit listrik). Bab ini berakhir di samping kolam reaktor, yang
berpijar biru Cherenkov.

\section{Intinya: setetes yang jenuh}

\begin{definition}[Nuklida, ukuran, dan kerapatan]\label{def:b3:nuclear-physics:nuclide}
\index{nuklida}\index{isotop}\index{jejari inti}
Sebuah \emph{nuklida} $^{A}_{Z}\text{X}$ menahan $Z$ proton dan $N = A -
Z$ neutron, yang terikat oleh \emph{interaksi kuat}: sengit
(${\sim}\qty{100}{}\times$ Coulomb pada sentuhannya), berjangkauan pendek
(${\sim}\qty{1}{fm}$, hanya terasa oleh tetangga yang bersentuhan), dan buta
terhadap muatan (ia mencengkeram proton dan neutron sama saja). Percobaan
hamburan memberi tiap intinya resep yang sama: yakni jejari
\[
R = r_0A^{1/3} , \qquad r_0 \approx \qty{1.2}{fm} ,
\]
sehingga volumenya tumbuh seperti $A$ dan kerapatannya menjadi semesta
$\rho \approx \qty{2.3e17}{kg/m^3}$ --- setetes hujan darinya akan
seberat armada kapal tangki, dan
\cref{ch:b3:astrophysics} akan menemui bintang utuh pada kerapatan ini.
\emph{Isotop} berbagi $Z$ tetapi berbeda $N$-nya: kimia yang sama,
nasib inti yang berbeda.
\end{definition}

\begin{proposition}[Cacat massa dan energi ikat]\label{prop:b3:nuclear-physics:binding}
\index{cacat massa}\index{energi ikat}\index{puncak besi}
Inti yang terikat berbobot \emph{lebih ringan} daripada bagiannya; menurut
\cref{ch:b3:relativistic-dynamics}, massa yang hilang itu adalah
\omterm{prop:b3:nuclear-physics:binding}{energi ikatnya},
\[
B = \big(Zm_{\text{p}} + Nm_{\text{n}} - m_{\text{inti}}\big)c^2 ,
\]
biasanya ${\sim}\qty{8}{MeV}$ \emph{tiap nukleonnya} --- yakni sejuta
kali kimia. Diplot tiap nukleonnya, $B/A$ memanjat curam menembus inti ringannya
(karena gejala permukaannya memudar), memuncak pada $\approx\qty{8.8}{MeV}$ di dekat
besi-56, lalu menghanyut turun ketika tolakan Coulomb, yang berjangkauan
panjang dan tak menjenuh, memajaki kelas beratnya. Kedua lerengnya adalah
tambang energi: \emph{fusi} memanjat lereng kirinya (yakni usaha Matahari),
\emph{fisi} menggelinding menuruni lereng kanannya (yakni usaha reaktor) --- dan besi,
di puncaknya, adalah abu inti bagi keduanya.
\end{proposition}

\begin{proposition}[Rumus massa semiempiris]\label{prop:b3:nuclear-physics:semf}
\index{rumus massa semiempiris}\index{lembah kemantapan}
Modelkanlah intinya sebagai setetes cairan bermuatan dan seluruh kurvanya
menyusul dari lima suku:
\[
B = a_{\text{V}}A - a_{\text{S}}A^{2/3}
- a_{\text{C}}\frac{Z^2}{A^{1/3}}
- a_{\text{A}}\frac{(A - 2Z)^2}{A} + \delta(A) ,
\]
dengan (dalam \unit{MeV}) $a_{\text{V}} = 15.8$ (tiap nukleonnya mengikat
tetangganya: yakni volume), $a_{\text{S}} = 17.8$ (nukleon permukaannya mengikat
lebih sedikit: yakni ``tegangan permukaan'' tetesannya), $a_{\text{C}} = 0.71$
(yakni Coulomb, sang penyabot berjangkauan panjang), $a_{\text{A}} = 23.7$
(ketaksetangkupan: menurut \cref{ch:b3:quantum-statistics}, proton dan
neutron mengisi dua tangga Fermi yang terpisah, dan termurah ketika sama
penuhnya), dan $\delta$ sebagai bonus pemasangan kecil untuk bilangan genap
(\cref{ch:b3:identical-particles}). Meminimumkan pada $A$ tetap menjejak
\emph{\omterm{prop:b3:nuclear-physics:semf}{lembah kemantapan}}: $N \approx Z$ untuk inti ringan, lalu
membengkok kaya neutron ($N/Z \to 1.5$) ketika Coulomb menghukum protonnya
--- dan \omterm{def:b3:nuclear-physics:nuclide}{nuklida} di luar lembahnya menggelinding kembali lewat peluruhan $\beta$.
\end{proposition}
\admitted

\section{Keradioaktifan, menurut jamnya}

\begin{theorem}[Hukum peluruhan]\label{thm:b3:nuclear-physics:decay}
\index{hukum peluruhan radioaktif}\index{umur paruh}\index{keaktifan}
Inti yang tak mantap tak punya ingatan dan tak menua: dalam tiap detik
yang dijalaninya, ia meluruh dengan peluang $\lambda$ yang sama. Untuk $N$
inti, $\dd N = -\lambda N\,\dd t$, sehingga
\[
N(t) = N_0\,\eu^{-\lambda t} = N_0\,2^{-t/T_{1/2}} ,
\qquad T_{1/2} = \frac{\ln2}{\lambda} ,
\]
dan \emph{keaktifannya} $\mathcal A = \lambda N$ (yakni peluruhan tiap
detiknya, \unit{Bq}) memudar pada eksponensial yang sama. \omterm{thm:b3:nuclear-physics:decay}{Umur paruhnya} berjalan
dari mikrodetik sampai miliaran tahun; statistika
\cref{ch:b3:microcanonical-ensemble} menjamin bahwa sementara satu
intinya sama sekali tak dapat diramalkan, semol darinya menjaga waktu
eksponensialnya lebih baik daripada jam mana pun --- yakni fondasi
penanggalan radiometri.
\end{theorem}

\begin{proof}
Peluang tiap detiknya yang tetap berarti $N(t + \dd t) = N(t)(1 -
\lambda\dd t)$; integralkanlah. Menetapkan $N/N_0 = \tfrac12$ memberikan
$T_{1/2}$.
\end{proof}

\begin{remark}[Tiga pintu keluar dari sebuah inti]\label{rem:b3:nuclear-physics:modes}
\index{peluruhan alfa}\index{peluruhan beta}%
\index{peluruhan gama}\index{model Gamow}
$\boldsymbol\alpha$: inti berat memancarkan gugus $^4$He. Secara klasik
mustahil --- penghalang Coulombnya mencapai \qty{25}{MeV} sedangkan
$\alpha$-nya pergi dengan \qty{5}{}--\qty{9}{MeV} --- ia terjadi lewat
\emph{penerowongan} (yakni penghalang jilid Tahun 2, yang dilengkungkan):
kepekaan eksponensial Gamow mengubah faktor dua dalam energi menjadi dua
puluh orde besar dalam umurnya, persis pola Geiger--Nuttall
(\cref{exo:b3:nuclear-physics:8}).
$\boldsymbol\beta$: sebutir neutron menjadi proton (atau sebaliknya), sambil memancarkan
sebuah elektron (atau positron) dan sebuah neutrino --- yakni \emph{interaksi
lemah} yang sedang bekerja, sang gaya pemulih lembahnya; kisah neutrinonya
menunggu \cref{ch:b3:particle-physics}.
$\boldsymbol\gamma$: inti yang tereksitasi melepaskan foton \unit{MeV}, yakni
padanan inti bagi garis spektrum \cref{ch:b3:hydrogen-atom}.
Peluruhannya berantai sampai dasar lembahnya tercapai: tangga uranium
berakhir, empat belas anak tangga kemudian, pada timbal yang mantap.
\end{remark}

\section{Menuruni lerengnya: fisi dan fusi}

\begin{proposition}[Fisi dan reaksi berantai]\label{prop:b3:nuclear-physics:fission}
\index{fisi}\index{reaksi berantai}\index{massa genting}
Neutron lambat yang diserap $^{235}$U membuat tetesan yang bergoyang lalu
membelah: dua pecahan bermassa menengah, dua sampai tiga neutron baru, dan
$\approx\qty{200}{MeV}$ --- yakni celah $B/A$ antara \qty{7.6}{MeV}
uranium dan \qty{8.5}{MeV} pecahannya, dikalikan 235.
Neutron yang terbebaskan dapat memfisikan inti berikutnya: dengan faktor
penggandaan $k$ (yakni neutron tiap neutronnya, satu keturunan
kemudian), populasinya tumbuh sebagai $k^n$ --- di bawah genting ($k < 1$)
padam, genting ($k = 1$) bertahan tunak, di atas genting meledak.
Sebuah reaktor memperlambat neutronnya sampai laju rendah (yakni makanan
kesukaan fisinya), bersandar pada 0.7\,\% neutronnya yang tiba
\emph{beberapa detik} terlambat dari peluruhan pecahannya --- yakni jeda yang membuat
$k \approx 1$ dapat disetel manusia --- lalu membiarkan batang kendalinya memakan
kelebihannya (\cref{pb:b3:nuclear-physics:1}).
\end{proposition}
\admitted

\begin{remark}[Fusi: lereng yang lebih sukar dan lebih baik]\label{rem:b3:nuclear-physics:fusion}
\index{fusi}
Menggabungkan inti ringan membayar $\sim4\times$ lebih banyak tiap
kilogramnya daripada membelah yang berat, tanpa pecahan berumur panjang ---
tetapi pereaksinya, yang keduanya positif, harus menerowongi penghalang
Coulomb bersamanya, yang menuntut suhu sebesar $10^7$--$10^8\,\unit{K}$
(\cref{exo:b3:nuclear-physics:11}). Bintang melakukannya dengan menjadi raksasa
(\cref{ch:b3:astrophysics}: Matahari mengubah \qty{600}{} juta
ton hidrogen menjadi helium tiap detiknya); laboratorium melakukannya
dengan plasma yang dibotolkan secara magnetis dan, sejauh ini, dengan buku
besar energi yang masih malu-malu terhadap cita-cita titik impasnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:1}
Pembukuan. (a) Berikanlah $Z$, $N$, $A$ untuk $^{12}$C, $^{14}$C,
$^{235}$U, $^{238}$U. (b) Pasangan yang mana yang merupakan \omterm{def:b3:nuclear-physics:nuclide}{isotop}? (c) Mengapa
\omterm{def:b3:nuclear-physics:nuclide}{isotop} berbagi kimia tetapi tidak kemantapan intinya? (d) $^{14}$C
dan $^{14}$N berbagi $A = 14$: \omterm{def:b3:nuclear-physics:nuclide}{nuklida} semacam itu disebut apa, dan
peluruhan apa yang menghubungkannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:2}
Kerapatan yang semesta. (a) Dari $R = r_0A^{1/3}$, tunjukkanlah bahwa
kerapatan intinya tak bergantung pada $A$ lalu hitunglah. (b)
Hitunglah massa satu sendok teh (\qty{5}{mL}) materi inti.
(c) Berapa bagian volume sebuah atom yang diisi intinya (ambillah
$R_{\text{atom}} = \qty{1e-10}{m}$, $A = 60$)? (d) Apa yang dikatakan
hukum $A^{1/3}$ itu sendiri tentang jangkauan gaya kuatnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:3}
Menimbang lemnya. Massa intinya: $m_{\text{p}} =
\qty{1.007276}{u}$, $m_{\text{n}} = \qty{1.008665}{u}$,
$m(^4\text{He}) = \qty{4.001506}{u}$;
$\qty{1}{u}\,c^2 = \qty{931.5}{MeV}$. (a) Hitunglah \omterm{prop:b3:nuclear-physics:binding}{cacat massa} dan
\omterm{prop:b3:nuclear-physics:binding}{energi ikat} $^4$He. (b) Berikanlah $B/A$. (c) Bandingkanlah $B$ dengan
\qty{13.6}{eV} \cref{ch:b3:hydrogen-atom}: berapa orde
besar yang memisahkan lem inti dari lem atom? (d) Mengapa
ikatan $^4$He yang luar biasa (yakni paku pada gambarnya) menjadikannya
mata uang peluruhan $\alpha$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:nuclear-physics:4}
Curie yang pertama. Radium-226: $T_{1/2} = 1600$ tahun, dengan massa molar
\qty{226}{g/mol}. Untuk satu gram: (a) cacahlah intinya; (b) hitunglah
$\lambda$; (c) hitunglah keaktifannya lalu bandingkanlah dengan satuan
bersejarah $1\,\text{Ci} = \qty{3.7e10}{Bq}$ (yang didefinisikan tepat sebagai
gram ini); (d) berapa banyak gramnya yang selamat setelah 4800 tahun?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:5}
Rumusnya dalam kerja. Dengan $a_{\text{V}} = 15.75$,
$a_{\text{S}} = 17.8$, $a_{\text{C}} = 0.711$,
$a_{\text{A}} = 23.7$ (\unit{MeV}) dan pemasangan
$+12/\sqrt A$ untuk inti genap--genap: (a) hitunglah keempat suku
utama $B$ untuk $^{56}$Fe ($Z = 26$). (b) Jumlahkanlah (dengan pemasangannya) lalu
bandingkanlah $B/A$ dengan nilai terukurnya \qty{8.79}{MeV}. (c) Kedua
suku mana yang bertarung paling keras, dan apa yang ditetapkan keseimbangannya? (d) Mengapa
suku Coulombnya, sendirian, harus tumbuh \emph{lebih cepat} daripada $A$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:6}
Dasar lembahnya. (a) Dengan hanya mempertahankan suku yang bergantung $Z$,
minimumkanlah massanya pada $A$ tetap lalu turunkanlah
$Z^* = \dfrac{A/2}{1 + a_{\text{C}}A^{2/3}/4a_{\text{A}}}$. (b)
Hitunglah untuk $A = 101$ lalu bandingkanlah dengan rutenium ($Z = 44$).
(c) Tunjukkanlah bahwa limit $A$ ringannya adalah $Z^* = A/2$ lalu tafsirkanlah dengan
kedua tangga Ferminya. (d) Sebuah pecahan fisi memiliki $A = 95$,
$Z = 36$: seberapa jauh ia dari lembahnya, dan urutan peluruhan apa yang menyusul?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:7}
Radiokarbon. Materi hidup menjaga $^{14}$C
($T_{1/2} = 5730$ tahun) tetap terisi pada 1 atom tiap $10^{12}$
karbonnya; kematian menghentikan pengisiannya. (a) Sebuah cuplikan arang menunjukkan
30\,\% keaktifan hidupnya: berapa umur apinya? (b) Satu gram
karbon hidup: hitunglah keaktifan $^{14}$C-nya
(diharapkan $\approx\qty{0.23}{Bq}$). (c) Mengapa metodenya tak berguna
di balik $\sim\qty{50000}{}$ tahun? (d) Dan mengapa ia tak berguna untuk
menanggali batuan (jam mana yang mengambil alih)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:8}
Tebing Gamow. Polonium-212 memancarkan $\alpha$ ber-\qty{8.78}{MeV};
penghalangnya mencapai $\approx\qty{26}{MeV}$. (a) Tunjukkanlah bahwa pelolosan
klasiknya mustahil, dan masuknya secara klasik pun mustahil --- namun ia meluruh
dalam \qty{0.3}{\micro s}. (b) $\alpha$ uranium-238 membawa
\qty{4.2}{MeV} dan \omterm{thm:b3:nuclear-physics:decay}{umur paruhnya} adalah $4.5\times10^9$ tahun:
hitunglah nisbah kedua tetapan peluruhannya. (c) Jelaskanlah, dengan
eksponensial penerowongannya, bagaimana faktor dua dalam energi membeli
$\sim10^{24}$ dalam lajunya. (d) Mengapa fisika yang sama menetapkan
tuntutan suhu teras Matahari?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:nuclear-physics:9}
Dua ratus juta elektronvolt. (a) Benarkanlah
$\qty{200}{MeV}$ tiap fisinya dari kurva $B/A$. (b) Hitunglah
energi dalam satu kilogram $^{235}$U yang terfisi penuh, dalam
joule dan dalam ton setara minyak
(\qty{42}{GJ/t}). (c) Sebuah pembangkit \qty{1}{GW_e} berkehematan 33\,\%:
berapa kilogram $^{235}$U tiap tahunnya? (d) Mengapa
pecahannya, dan bukan neutronnya, yang mengangkut sebagian besar
\qty{200}{MeV}-nya, dan ke bentuk apa ia segera berubah?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:nuclear-physics:10}
Menjinakkan $k$. Dalam sebuah reaktor, neutron seketikanya berkembang biak dalam $\ell
\approx \qty{e-4}{s}$. (a) Dengan $k = 1.001$ pada neutron seketikanya
saja, hitunglah penggandaan dayanya dalam satu detik --- lalu berikanlah
putusan tentang kendali mekanisnya. (b) Sebagian sebesar $\beta =
0.7\,\%$ neutronnya \emph{tertunda} sebesar $\sim\qty{10}{s}$:
jelaskanlah mengapa, untuk $k - 1 < \beta$, jam efektif rantainya
menjadi hitungan detik. (c) Hitunglah penggandaan satu detik yang sama
dengan waktu keturunan efektif $\sim\qty{0.1}{s}$. (d) Nyatakanlah
dalam satu kalimat apa yang hilang dari operator Chernobyl ketika
reaktornya menjadi di atas genting secara seketika.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:nuclear-physics:11}
Hitungan Matahari. Secara keseluruhan, $4\,^1\text{H} \to {}^4\text{He}$
mengubah \qty{0.71}{\percent} massanya menjadi energi
(\qty{26.7}{MeV}, termasuk neutrinonya). (a) Dari $L_\odot =
\qty{3.8e26}{W}$, hitunglah massa yang terubah tiap detiknya, dan
hidrogen yang terpakai tiap detiknya. (b) Dengan $\qty{2e30}{kg}$ matahari,
sepersepuluhnya berupa hidrogen teras yang dapat terbakar (ambillah bagian
hidrogennya $\approx0.75$): perkirakanlah umurnya. (c) Dua proton pada
\qty{1.5e7}{K}: bandingkanlah $k_{\text{B}}T$ dengan penghalang Coulombnya
pada \qty{1}{fm} lalu simpulkanlah siapa yang melakukan penyeberangannya
(\cref{exo:b3:nuclear-physics:8}). (d) Mengapa fusi, tak seperti
fisi, pada dasarnya tak meninggalkan abu radioaktif?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:nuclear-physics:12}
Kedokteran nuklir. (a) Teknesium-99m ($T_{1/2} = \qty{6}{h}$,
dengan $\gamma$ sebesar \qty{140}{keV}) adalah kuda beban kedokteran: hitunglah
cacah intinya, dan massanya, di balik suntikan
\qty{500}{MBq}. (b) Jelaskanlah mengapa \omterm{thm:b3:nuclear-physics:decay}{umur paruh} enam \emph{jam} dan
$\gamma$ yang murni masing-masing persis yang diinginkan sebuah diagnosis. (c) PET:
sebutir positron dari $^{18}$F saling memusnah dengan sebutir elektron ---
pakailah \cref{ch:b3:relativistic-dynamics} untuk memberikan energi dan
geometri kedua fotonnya, dan karenanya asas pendeteksinya.
(d) Radioterapi menyerahkan \qty{2}{Gy} (\unit{J/kg}) kepada sebuah tumor:
bandingkanlah energinya dengan seteguk air hangat, lalu serasikanlah
ketakberbahayaan joulenya dengan kemematian pengionannya.
\end{exercise}

\section{Soal: Hari terbuka di reaktor riset}

\begin{problem}\label{pb:b3:nuclear-physics:1}
\emph{Hari terbuka di reaktor riset.} Laboratorium nasional
membuka reaktor riset jenis kolamnya yang \qty{20}{MW} bagi pengunjung, dan
Anda telah berhasil membujuk untuk naik ke balkonnya: di bawah, menembus delapan
meter air yang jernih, terasnya berpijar biru yang bukan dari dunia ini.
Sang pemandu --- seorang fisikawan reaktor yang akan pensiun --- setuju membiarkan
Anda melakukan hitungannya.

\medskip\noindent\textbf{Bagian I --- Buku besar energinya.}

\begin{enumerate}
  \item Tiap fisi $^{235}$U melepaskan kira-kira \qty{200}{MeV}.
        Ubahlah ke joule, lalu hitunglah cacah fisi tiap detiknya
        yang menopang \qty{20}{MW}.
  \item Ubahlah ke gram $^{235}$U yang terpakai tiap harinya.
  \item Nilai \qty{20}{MW} yang sama dari batu bara (\qty{30}{MJ/kg}): berapa ton
        tiap harinya? Bentuklah nisbah massanya lalu kaitkanlah dengan
        skala \qty{}{MeV} berbanding \qty{}{eV} bab ini dan
        kimianya.
  \item Pecahannya lahir pada ikatan $\sim\qty{8.5}{MeV}$ tiap
        nukleonnya, uraniumnya pada \qty{7.6}{MeV}: periksalah bahwa
        $235\times0.9 \approx \qty{200}{MeV}$ taat asas.
  \item Pecahannya berhenti dalam jarak mikrometer dari tempat lahirnya.
        Menjadi apa \qty{200}{MeV}-nya berubah, dan pada skala waktu apa
        ia mencapai air pendinginnya?
  \item Mengapa pecahannya (misalnya $A = 95$, $Z = 36$)
        pasti bersifat radioaktif? Pakailah \omterm{prop:b3:nuclear-physics:semf}{lembah kemantapannya}.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Menjaga $k = 1$.}

\begin{enumerate}[resume]
  \item Definisikanlah faktor penggandaan $k$ lalu nyatakanlah apa arti
        $k = 0.999$, $1.000$, $1.001$ bagi populasi
        neutronnya.
  \item Air kolamnya memperlambat. Jelaskanlah dalam dua kalimat mengapa
        memperlambat neutronnya sampai laju termal \emph{membantu} fisi
        $^{235}$U, dan mengapa hidrogen menjadi pemerlambat terbaik
        (ingatlah tumbukan lenting jilid Tahun 1).
  \item Batang kendalinya terbuat dari baja boron. Apa yang dilakukan boron, dan
        bagaimana menaikkan atau menurunkan batangnya menyetir $k$?
  \item Dengan neutron seketikanya saja ($\ell = \qty{e-4}{s}$) dan
        $k = 1.0005$, hitunglah pertumbuhan dayanya selama satu detik
        lalu simpulkanlah.
  \item Neutron tertunda sebesar 0.7\,\% merentangkan waktu keturunan
        efektifnya menjadi $\sim\qty{0.1}{s}$: hitunglah ulang, lalu
        nyatakanlah kaidah rancangan yang menjaga $k - 1$ tetap aman di bawah
        bagian tertundanya.
  \item Airnya juga menjadi pendinginnya. Jelaskanlah ciri keselamatan
        hakikinya: apa yang terjadi pada perlambatannya --- dan karenanya
        pada $k$ --- jika airnya mendidih habis?
  \item Mengapa reaktornya masih memerlukan pendinginan darurat setelah
        dimatikan? (Sebutkanlah sumber kalor yang tak dapat disentuh batang
        kendalinya.)
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Cahaya birunya.}

\begin{enumerate}[resume]
  \item Pijarnya adalah radiasi Cherenkov: laju cahaya dalam air
        adalah $c/n$ dengan $n = 1.33$. Apa yang harus dilakukan sebuah partikel
        bermuatan untuk memancarkannya?
  \item Hitunglah laju ambangnya, dan dengan
        \cref{ch:b3:relativistic-dynamics} hitunglah energi kinetik
        ambang bagi sebuah elektron.
  \item Partikel reaktor yang mana yang memenuhi syarat? Telusurilah rantainya dari
        pecahan fisi sampai ke elektron $\beta$ cepat dalam airnya.
  \item Keamatan spektrumnya tumbuh menuju panjang gelombang pendek:
        mengapa kolamnya berpijar biru dan bukan merah?
  \item Sang pemandu berkata: ``pijarnya \emph{adalah} perisai yang
        bekerja.'' Jelaskanlah --- apa yang dilakukan delapan meter air
        terhadap radiasi terasnya, dan kira-kira mengapa balkonnya aman?
  \item Setelah dimatikan birunya meredup selama beberapa menit tetapi tak
        lenyap selama berhari-hari. Kaitkanlah dengan butir 6 Bagian I.
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Untuk apa reaktornya.}

\begin{enumerate}[resume]
  \item Pekerjaan harian reaktor ini adalah membuat molibdenum-99
        ($T_{1/2} = \qty{66}{h}$), yakni induk teknesium-99m
        kedokteran. Mengapa rumah sakit di seluruh dunia harus
        dipasok ulang tiap pekan, dan mengapa tak ada gudang yang dapat menimbunnya?
  \item Sebuah kiriman Mo-99 sebesar \qty{100}{GBq} berangkat pada Senin;
        rumah sakitnya melarutkan Tc-99m pada Jumat (96 jam kemudian). Berapa
        keaktifan induknya yang tersisa?
  \item Berkas neutron dari terasnya juga memberi makan instrumen difraksi
        \cref{ch:b3:crystalline-solids}: mengapa neutron reaktor,
        begitu ditermalkan, lahir dengan panjang gelombang yang tepat?
  \item Soal perpisahan sang pemandu: sebuah elemen bahan bakar menghabiskan lima
        tahun dalam kolamnya sebelum dikirim. Dengan memakai campuran \omterm{thm:b3:nuclear-physics:decay}{umur
        paruh} pecahannya, jelaskanlah penalarannya --- apa yang dibeli lima
        tahun waktu kolam itu?
  \item Perkirakanlah kalor peluruhannya satu jam setelah dimatikan
        ($\approx1\,\%$ dari \qty{20}{MW}) lalu bandingkanlah dengan
        pemanas listrik sebuah rumah tangga: apakah pendinginan kolam
        secara pasif masuk akal?
  \item Tutuplah buku besarnya dalam empat baris: $B/A$ membayar
        \qty{200}{MeV} tiap belahannya; neutron tertundanya meminjamkan
        detik yang membuat $k$ dapat disetir; airnya memperlambat,
        mendinginkan, memerisai, dan berpijar; dan hasil sejati harinya
        pergi dalam vial obat, bukan dalam megawatt.
\end{enumerate}
\end{problem}
