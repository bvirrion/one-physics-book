\chapter{Fisika Partikel}\label{ch:b3:particle-physics}

% Provenance (not for print): second half of the "physique subatomique"
% module common to the surveyed third-year curricula: standard model
% inventory, interactions, conservation laws, detectors.

Kupaslah materi dari strukturnya, lapis demi lapis --- molekul, atom,
inti, nukleon --- dan pengupasannya berhenti, secara tiba-tiba, pada
sebuah daftar pendek: enam \emph{\omterm{def:b3:particle-physics:zoo}{kuark}}, enam \emph{\omterm{def:b3:particle-physics:zoo}{lepton}}, segenggam
pembawa gaya, dan satu skalar pemalu yang ditemukan pada 2012. Segala
yang pernah Anda sentuh adalah tiga entri daftar itu; sisanya hidup pada
mikrodetik pertama alam semesta lalu hidup lagi, sebentar, di mana pun
$E = mc^2$ dibayar penuh --- yakni tumbukan sinar kosmik dan cincin
pemercepat. Bab ini melakukan pendataan: partikelnya, keempat
interaksi yang menggerakkannya, hukum kekekalannya
(yakni arus Noether \cref{ch:b3:lagrangian-mechanics}, yang kini menjalankan
pembukuan subatomnya), dan pendeteksi yang memotret semuanya. Ia
berakhir di loteng Anda sendiri, sambil mencacah muon yang dibuat sepuluh kilometer di atas kepala.

\section{Pendataannya}

\begin{definition}[Lepton, kuark, hadron]\label{def:b3:particle-physics:zoo}
\index{lepton}\index{kuark}\index{hadron}\index{barion}\index{meson}\index{antipartikel}
Balok pembangun materi adalah fermion berspin $\tfrac12$
(\cref{ch:b3:spin-two-level}, \cref{ch:b3:identical-particles}) dalam
dua keluarga berisi enam. \emph{Lepton} tak merasakan gaya kuat: yakni
elektron, muon dan tau (bermuatan $-e$, masing-masing lebih berat daripada sebelumnya)
beserta ketiga neutrinonya --- yang nyaris tak bermassa dan nyaris
tak terdeteksi. \emph{Kuark} datang dalam enam rasa --- atas, bawah,
aneh, pesona, dasar, puncak --- bermuatan $+\tfrac23e$ (u, c,
t) dan $-\tfrac13e$ (d, s, b), dan mereka tak pernah terlihat sendirian:
\emph{pengurungan} mengikatnya menjadi \emph{hadron} --- yakni
\emph{barion} bertiga kuark (proton $=$ uud, neutron $=$ udd) dan
\emph{meson} berpasangan kuark--antikuark (pion, kaon). Tiap partikelnya memiliki
\emph{antipartikel} bermassa serupa dan bermuatan berlawanan
(\cref{exo:b3:particle-physics:8}). Materi biasa hanya memakai
keturunan pertamanya --- u, d, e, $\nu_{\text{e}}$; kedua salinan yang lebih berat
berbeda, sejauh yang diketahui siapa pun, hanya dalam massanya. ``Siapa
yang memesan itu?'' masih ditanyakan fisika.
\end{definition}

\begin{remark}[Empat interaksi]\label{rem:b3:particle-physics:forces}
\index{interaksi kuat}\index{interaksi lemah}\index{elektrolemah}
Tiap prosesnya adalah salah satu dari empat gaya yang sedang bekerja,
masing-masing dibawa bosonnya. Gaya \emph{kuat} (gluon) mencengkeram \omterm{def:b3:particle-physics:zoo}{kuark} dengan
potensial yang \emph{tumbuh} bersama jaraknya --- tariklah dua \omterm{def:b3:particle-physics:zoo}{kuark}
saling menjauh dan medan yang terentang itu patah menjadi pasangan
kuark--antikuark yang baru (\cref{exo:b3:particle-physics:1}): sehingga muncullah
pengurungannya, dan sisanya melem inti
(\cref{ch:b3:nuclear-physics}). \emph{Elektromagnetisme} (foton,
tak bermassa: berjangkauan takhingga) menjalankan kimia dan cahaya. Gaya
\emph{lemah} (W dan Z, yang bermassa
$\approx80$--$\qty{91}{GeV}$: berjangkauan $10^{-3}\,\unit{fm}$,
sehingga ``lemah'') adalah satu-satunya pengubah rasa --- peluruhan $\beta$ adalah
sebutir \omterm{def:b3:particle-physics:zoo}{kuark} d yang menjadi u
(\cref{prop:b3:particle-physics:conservation}) --- dan satu-satunya
gaya yang dijawab neutrino. \emph{Gravitasi} $10^{36}$ kali
lebih lemah daripada elektromagnetisme antara dua proton dan tak berperan
di laboratorium --- namun, karena tak tertapis dan berjangkauan panjang, ia akan menjalankan
seluruh \cref{ch:b3:astrophysics}. Elektromagnetisme dan gaya
lemahnya menyatu pada energi tinggi menjadi satu interaksi
\emph{elektrolemah} --- yang terbelah, pada energi kita yang dingin, oleh
\omterm{def:b3:phase-transitions:order}{pematahan kesetangkupan} medan Higgs (\cref{exo:b3:particle-physics:9}).
\end{remark}

\section{Pembukuannya: hukum kekekalan}

\begin{proposition}[Hukum kekekalan reaksi partikel]\label{prop:b3:particle-physics:conservation}
\index{bilangan barion}\index{bilangan lepton}\index{hukum kekekalan!partikel}
Sebuah reaksi terjadi hanya jika bukunya seimbang. Yang selalu
kekal: energi--momentum
(\cref{ch:b3:relativistic-dynamics}), momentum sudut, dan
muatan listrik --- dengan teorema Noether
(\cref{thm:b3:lagrangian-mechanics:noether}) menjamin masing-masingnya.
Yang kekal dalam tiap reaksi yang pernah teramati: \emph{\omterm{prop:b3:particle-physics:conservation}{bilangan barion}}
$B$ (\omterm{def:b3:particle-physics:zoo}{barion} $+1$, antibarion $-1$) dan \emph{\omterm{prop:b3:particle-physics:conservation}{bilangan lepton}}
$L$ (tiap keluarganya, dengan ketelitian yang sangat baik). Maka proton, yakni
\omterm{def:b3:particle-physics:zoo}{barion} teringan, tak punya tempat untuk meluruh --- kemantapan materi adalah
sebuah teorema pembukuan --- dan peluruhan $\beta$ harus mengirimkan sebuah
antineutrino:
\[
\text{n} \;\to\; \text{p} + \text{e}^- + \bar\nu_{\text{e}}
\qquad (\text{tingkat kuark: } \text{d} \to \text{u} +
\text{e}^- + \bar\nu_{\text{e}}) ,
\]
yang menyeimbangkan $L$ (sekaligus menjelaskan spektrum energi elektronnya
yang sinambung, yang nyaris membuat fisika kehilangan imannya pada kekekalan
energi sebelum Pauli menciptakan neutrino untuk menyelamatkannya).
Rasanya (keanehan, pesona) kekal pada proses kuat dan
elektromagnetiknya tetapi \emph{dipatahkan oleh yang lemah}: partikel aneh
lahir berlimpah secara berpasangan, lalu mati perlahan dan sendirian
--- celah $10^{13}$ antara umur kuat $\qty{e-23}{s}$
dan umur lemah $\qty{e-10}{s}$ adalah sidik jari gaya mana yang
menandatangani peluruhannya.
\end{proposition}
\admitted

\section{Membuat dan melihat partikel}

\begin{remark}[Pemercepat dan pendeteksi]\label{rem:b3:particle-physics:machines}
\index{penumbuk}\index{pendeteksi partikel}
Dua alasan untuk mempercepat: \emph{daya pisah} --- sebuah pemeriksa tak melihat
rincian yang lebih halus daripada panjang gelombangnya $\lambda = h/p$
(\cref{ch:b3:crystalline-solids} memakai ångström; femtometer
berongkos \unit{GeV}) --- dan \emph{penciptaan}: energi tumbukannya
menjadi massa baru, dan menumbukkan dua berkas secara berhadapan
membelanjakan seluruhnya (sasaran tetap memboroskan sebagian besarnya untuk
membawa puingnya maju: \cref{exo:b3:particle-physics:4}). Penumbuk
melingkar mendaur ulang partikel yang sama menembus celah pemercepat yang sama
sementara magnetnya menyetirnya --- sambil membayar radiasi sinkrotron
$\propto\gamma^4$, dan karena itulah cincin yang menemukan Higgs
menumbukkan proton yang berat, bukan elektron yang seringan bulu. Di sekeliling
titik silangnya duduk lapisan bawang, masing-masing menanyakan satu pertanyaan: sebuah
\emph{penjejak} membengkokkan lintasan bermuatan dalam sebuah magnet (yakni momentumnya); sebuah
\emph{kalorimeter} elektromagnetik menelan elektron dan
foton (yakni energinya); yang hadronik menelan pion dan proton;
dan hanya muon yang menembus sampai ke bilik terluarnya ---
yakni jati diri menurut kedalaman penguburannya. Neutrino pergi sebagai momentum
yang hilang: pembukuan mendeteksi apa yang tak dapat dihentikan apa pun.
\end{remark}

\begin{remark}[Higgs, dan buku yang masih terbuka]\label{rem:b3:particle-physics:higgs}
\index{boson Higgs}\index{materi gelap}
Kesetangkupan Model Baku melarang massa telanjang bagi
partikelnya; \emph{medan Higgs}, yang mengisi seluruh ruang dengan
kondensat tak nol --- yakni pilihan sumur ganda
\cref{ch:b3:phase-transitions}, yang dinaikkan ke vakumnya sendiri
--- mematahkan kesetangkupannya lalu menganugerahkan massa lewat gandengannya:
makin berat partikelnya, makin kuat cengkeramannya. Kuantumnya sendiri, yakni \omterm{rem:b3:particle-physics:higgs}{boson
Higgs} (\qty{125}{GeV}, ditemukan 2012), melengkapi tabelnya. Buku
yang masih terbuka lebih panjang: neutrino berayun antarrasa sehingga
pasti bermassa, yang tak pernah dituliskan modelnya
(\cref{exo:b3:particle-physics:10}); galaksi berputar seolah ditahan
oleh materi lima kali lebih banyak daripada yang bersinar
(\cref{ch:b3:astrophysics}); dan alam semesta memuat materi
tetapi nyaris tanpa antimateri, yakni ketaksetangkupan yang belum dapat dibayar
oleh pelanggaran CP model yang mungil itu. Daftarnya adalah silabus
apa pun yang datang berikutnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b3:particle-physics:1}
Hitungan \omterm{def:b3:particle-physics:zoo}{kuark}. (a) Periksalah muatan p $=$ uud dan
n $=$ udd. (b) Pion $\pi^+$ adalah $\text{u}\bar{\text{d}}$:
periksalah muatannya. (c) Usulkanlah isi \omterm{def:b3:particle-physics:zoo}{kuark} untuk $\pi^-$ dan untuk
$\Delta^{++}$ (bermuatan $+2e$). (d) Mengapa menarik sebutir \omterm{def:b3:particle-physics:zoo}{kuark}
keluar dari sebuah proton menghasilkan hujan \omterm{def:b3:particle-physics:zoo}{meson} dan bukan \omterm{def:b3:particle-physics:zoo}{kuark} bebas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:particle-physics:2}
Memilah. Untuk masing-masing e$^-$, $\nu_\mu$, $\pi^+$, p, n,
$\mu^-$: (a) \omterm{def:b3:particle-physics:zoo}{lepton}, \omterm{def:b3:particle-physics:zoo}{barion} atau \omterm{def:b3:particle-physics:zoo}{meson}? (b) yang mana yang merasakan gaya
kuatnya? (c) yang mana yang mantap dalam keterpencilan? (d) yang mana yang tak dapat
dibelokkan medan magnet?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:particle-physics:3}
Diizinkan atau dilarang? Berilah putusan, sambil menyebutkan hukum yang dilanggar bila ada:
(a) $\text{n} \to \text{p} + \text{e}^- + \bar\nu_{\text{e}}$;
(b) $\text{p} \to \text{n} + \text{e}^+ + \nu_{\text{e}}$ (untuk
proton \emph{bebas});
(c) $\text{p} \to \text{e}^+ + \gamma$;
(d) $\mu^- \to \text{e}^- + \gamma$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:particle-physics:4}
Harga penciptaannya. Menghasilkan pasangan proton--antiproton dari
berkas proton pada sasaran hidrogen menuntut (menurut
penalaran massa tak berubah \cref{ch:b3:relativistic-dynamics})
energi kinetik berkas sebesar $6m_{\text{p}}c^2$. (a) Hitunglah dalam GeV.
(b) Dalam sebuah penumbuk, dua berkas berenergi kinetik berapa yang memadai untuk
pasangan yang sama? (c) Hitunglah nisbah ``pemborosannya'' lalu jelaskanlah ke mana
energi sasaran tetapnya pergi. (d) Mengapa mesin penemu antiproton pada 1955
memerlukan $\sim\qty{6}{GeV}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:particle-physics:5}
Muon, dua kali. Muon ($m = \qty{106}{MeV}$,
$\tau = \qty{2.2}{\micro s}$) adalah kembaran berat elektron. (a)
Jarak berapakah $c\tau$? (b) Muon kosmik yang lahir pada
\qty{15}{km} mencapai tanah: $\gamma$ berapa yang dituntut
kelangsungan hidupnya, dan berapa energinya? (c) Lukiskanlah kelangsungan hidup
yang sama dari kerangka muonnya sendiri. (d) Mengapa muonnya meluruh menjadi
elektron tetapi tak pernah menjadi proton, dan mengapa peluruhannya
\emph{lambat} menurut ukuran partikel?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:particle-physics:6}
Di dalam peluruhan $\beta$. (a) Tuliskanlah peluruhan neutron pada tingkat
\omterm{def:b3:particle-physics:zoo}{kuarknya} lalu periksalah semua besaran kekalnya. (b) Spektrum energi
elektron yang terpancar bersifat sinambung sampai sebuah titik ujung:
jelaskanlah bagaimana hal ini dahulu mengancam kekekalan energi dan bagaimana
neutrino Pauli menyelamatkannya. (c) Boson W berbobot
\qty{80.4}{GeV}: perkirakanlah jangkauan gaya lemahnya
$\hbar/m_{\text{W}}c$ ($\hbar c = \qty{197}{MeV.fm}$). (d) Pakailah
jangkauan itu untuk menjelaskan dalam satu kalimat mengapa gaya ``lemah'' itu
lemah pada energi rendah namun sekuat elektrolemah pada energi tinggi.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:particle-physics:7}
Pembukuan yang aneh. Kaon dan hiperon $\Lambda$ membawa sebuah
rasa, yakni \emph{keanehan}, yang kekal menurut gaya kuat tetapi
tidak menurut yang lemah. (a) Jelaskanlah mengapa tumbukan sinar kosmik menghasilkan
partikel aneh hanya berpasangan. (b) $\Lambda$ hidup
$\qty{2.6e-10}{s}$ --- bandingkanlah dengan $\qty{e-23}{s}$ peluruhan
kuatnya lalu simpulkanlah gaya mana yang membunuhnya. (c) Nisbah
skala waktu berapa yang memisahkan keduanya, dan dengan nisbah sehari-hari apa
hal itu sebanding (satu detik berbanding apa)? (d) Bagaimana perilaku
``aneh'' ini --- yakni kelahiran berlimpah, kematian yang enggan ---
memaksa penciptaan sebuah bilangan kuantum baru?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:particle-physics:8}
Antimateri, yang dicairkan. (a) Mengapa
$\text{e}^+ + \text{e}^- \to \gamma$ (satu foton) pasti dilarang,
sedangkan dua foton tak apa-apa? (b) Hitunglah energi yang dilepaskan lewat
pemusnahan satu gram antimateri dengan satu gram materi, lalu
bandingkanlah dengan satu kilogram uranium terfisi
\cref{exo:b3:nuclear-physics:9}. (c) Pemindai PET
(\cref{exo:b3:nuclear-physics:12}) berjalan atas positron: dari mana
positron kedokteran datang? (d) Membuat antiproton berongkos
$\sim10^{6}$ kali \omterm{def:b3:relativistic-dynamics:momentum-energy}{energi diamnya} dalam daya dari stopkontak: berilah ulasan
tentang antimateri sebagai bahan bakar.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:particle-physics:9}
Higgs sebagai \omterm{def:b3:phase-transitions:order}{peralihan fase}. (a) Nyatakanlah kembali, dalam bahasa
\cref{exo:b3:phase-transitions:12}, apa artinya bahwa
vakumnya duduk dalam sumur berkesetangkupan patah. (b) Mengapa W dan Z
memperoleh massa sedangkan fotonnya tetap tak bermassa (bagian kesetangkupan
yang mana yang selamat)? (c) \omterm{rem:b3:particle-physics:higgs}{Boson Higgs} adalah getaran radial medannya:
apa yang memainkan peran ini dalam magnet
\cref{thm:b3:phase-transitions:meanfield}? (d) Di atas
$\sim10^{15}\,\unit{K}$ kesetangkupannya pulih: di mana dan kapan
alam semesta terakhir kali sepanas itu (\cref{ch:b3:astrophysics})?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:particle-physics:10}
Neutrino bermiliar-miliar. (a) Matahari membuat dua neutrino tiap
daur $4\text{p} \to {}^4\text{He}$-nya (\qty{26.7}{MeV}): dari
tetapan surya \qty{1360}{W/m^2}, hitunglah fluks neutrinonya di
Bumi. (b) Berapa banyak yang menyeberangi kuku ibu jari Anda
(\qty{1}{cm^2}) tiap detiknya, dan kira-kira berapa banyak yang akan berinteraksi
dengan tubuh Anda seumur hidup (ambillah peluang interaksi
berorde $10^{-22}$ tiap meter dagingnya tiap neutrinonya sebagai
diberikan)? (c) Pendeteksi awal
hanya menangkap sepertiga $\nu_{\text{e}}$ yang diramalkan:
nyatakanlah penyelesaiannya (yakni ayunannya) beserta akibat
strukturalnya bagi Model Baku. (d) Mengapa sebuah pendeteksi
neutrino tinggal di bawah gunung?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:particle-physics:11}
Bilangan mesinnya. Proton LHC: $E = \qty{6.8}{TeV}$ masing-masing. (a)
Hitunglah $\gamma$ dan $1 - v/c \approx 1/2\gamma^2$. (b) Panjang
gelombang $\lambda \approx hc/E$ berapa yang memeriksa strukturnya, lalu bandingkanlah
dengan jejari proton \qty{0.8}{fm}. (c) Energi tumbukan
berhadapannya: \qty{13.6}{TeV} --- berapa massa diam proton yang dapat
diciptakannya? (d) Rugi sinkrotronnya berskala sebagai $\gamma^4$:
hitunglah nisbah $\gamma$ elektron terhadap proton pada energi tetap
lalu benarkanlah proton untuk cincinnya, elektron untuk mesin ketelitiannya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:particle-physics:12}
Utusan dari tepi. Sinar kosmik berenergi tertinggi yang pernah
terekam membawa $\sim\qty{3e20}{eV}$ --- yakni sebutir proton tunggal. (a)
Ubahlah ke joule lalu sebutkanlah benda makroskopis yang berenergi
kinetik demikian. (b) Hitunglah $\gamma$-nya, dan lebar
Galaksi ($\sim10^{5}$ tahun cahaya) dalam kerangkanya sendiri. (c) Di atas
$\sim\qty{5e19}{eV}$, protonnya menghambur pada latar gelombang mikro
kosmik (\cref{ch:b3:photons-phonons}) --- jelaskanlah
secara kualitatif mengapa alam semesta kedap terhadapnya. (d) Mengapa
energi semacam itu, yang jauh melampaui pemercepat mana pun, toh mengajari kita
lebih sedikit daripada LHC? (Pikirkanlah fluks dan kendalinya.)
\end{exercise}

\section{Soal: Teleskop muon loteng}

\begin{problem}\label{pb:b3:particle-physics:1}
\emph{Teleskop muon loteng.} Untuk pameran sains Anda membangun sebuah
teleskop sinar kosmik: dua dayung sintilator $20\times20$\,\unit{cm},
yang satu \qty{30}{cm} di atas yang lain, masing-masing dibaca sebuah pengganda foton,
dikabeli secara berkebetulan --- yakni satu cacahan hanya ketika \emph{keduanya} menyala
dalam \qty{100}{ns}. Fluks muon aras laut menembus permukaan
mendatar: kira-kira 1 tiap \unit{cm^2} tiap menitnya.

\medskip\noindent\textbf{Bagian I --- Mencacah langitnya.}

\begin{enumerate}
  \item Di mana muon Anda bermula? Telusurilah rantainya: proton
        primernya, atmosfer atasnya, pionnya, muonnya --- sambil menyebutkan gaya
        mana yang memerintah tiap langkahnya.
  \item Mengapa pionnya sendiri nyaris tak pernah mencapai
        tanah ($\tau_\pi = \qty{26}{ns}$), sedangkan muon anaknya
        mencapainya?
  \item Ramalkanlah laju satu dayung Anda dari fluks yang dikutip itu.
  \item Dayungnya sendirian sebenarnya berdetak jauh lebih sering ---
        karena derau elektronika dan keradioaktifan sekelilingnya
        (\cref{ch:b3:nuclear-physics}). Jelaskanlah bagaimana
        kebetulan dua dayungnya membunuh nyaris semuanya.
  \item Laju kebetulan tak sengaja berapa yang dihasilkan dua sumber derau
        \qty{100}{Hz} yang saling bebas dalam jendela \qty{100}{ns}
        ($R_{\text{acc}} = R_1R_2\Delta t$)? Bandingkanlah dengan
        laju muon yang sebenarnya.
  \item Laju kebetulan Anda menetap di dekat 100 tiap menitnya ---
        jauh di bawah 400 satu dayungnya. Jelaskanlah: muon mana
        yang diterima geometri dua dayungnya?
  \item Miringkanlah teleskopnya: lajunya turun kira-kira sebagai
        $\cos^2\theta$ sudut puncaknya. Berikanlah alasan fisisnya
        (apa yang tumbuh bersama $\theta$?).
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Kenisbian, yang dibuktikan di loteng.}

\begin{enumerate}[resume]
  \item Muon: $\tau = \qty{2.2}{\micro s}$. Hitunglah $c\tau$ lalu
        simpulkanlah apa yang dapat dilakukan muon \emph{klasik} yang lahir pada
        \qty{15}{km}.
  \item Muon Anda rata-rata $E \approx \qty{4}{GeV}$
        ($m_\mu c^2 = \qty{106}{MeV}$): hitunglah $\gamma$.
  \item Hitunglah umur yang terulur, beserta jangkauannya: apakah
        laju loteng Anda kini masuk akal?
  \item Ceritakanlah kembali kelangsungan hidupnya dari kerangka muonnya
        (\cref{ch:b3:relativistic-kinematics}).
  \item Untuk $\gamma = 38$, hitunglah bagian yang selamat dari
        \qty{15}{km} ($t_{\text{diri}} = L/\gamma v$, dengan kelangsungan hidup
        $\eu^{-t/\tau}$).
  \item Teleskop Anda karenanya adalah percobaan klasik yang
        \emph{mana}, yang dibangun ulang dengan suku cadang sekolah? Nyatakanlah apa yang
        akan diramalkan hasil nihilnya (yakni peluruhan klasiknya) bagi
        pencacah Anda.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Apa itu sebutir muon.}

\begin{enumerate}[resume]
  \item Tempatkanlah muonnya dalam tabel Model Baku: keluarganya,
        keturunannya, muatannya, spinnya.
  \item Ia meluruh: $\mu^- \to \text{e}^- + \bar\nu_{\text{e}} +
        \nu_\mu$. Periksalah muatannya dan kedua bilangan keluarga \omterm{def:b3:particle-physics:zoo}{leptonnya}.
  \item Mengapa spektrum energi elektron peluruhannya bersifat sinambung,
        dan berapa titik ujungnya ($\approx m_\mu c^2/2$)?
  \item Mengapa $\mu^- \to \text{e}^- + \gamma$ --- yang tak pernah
        teramati --- begitu menarik bagi percobaan masa kini?
        (Apa yang akan dipatahkan penemuannya?)
  \item Sebuah muon yang berhenti dalam dayung Anda meluruh dalam keadaan diam: lukiskanlah
        kilatan kedua yang tertunda itu dan bagaimana ia menyerahkan $\tau$
        kepada Anda secara langsung.
  \item Muon adalah ``elektron berat'': berikanlah satu akibatnya bagi
        atom --- apa yang terjadi pada \omterm{thm:b3:hydrogen-atom:levels}{jejari Bohr} hidrogen muonik
        (\cref{ch:b3:hydrogen-atom}), yang diskalakan dengan
        $m_\mu/m_{\text{e}} \approx 207$?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Menaiki rantainya.}

\begin{enumerate}[resume]
  \item Sebutir proton primer $\qty{e15}{eV}$ melahirkan hujan berisi
        jutaan partikel. Perkirakanlah energi rata-rata tiap partikel hujannya
        jika $10^{6}$ partikel berbagi energinya.
  \item Observatorium besar menggantikan dayung Anda dengan ratusan
        tangki air yang berjarak berkilo-kilometer, sambil mengawasi
        cahaya \emph{Cherenkov} (\cref{pb:b3:nuclear-physics:1}):
        apa yang dilakukan muon dalam airnya, dan mengapa tangkinya
        \emph{berjauhan}?
  \item Sintilator dan pengganda foton Anda memakai asas yang sama
        seperti kalorimeter di penumbuk besar:
        nyatakanlah asas bersama itu dalam satu kalimat
        (partikel $\to$ cahaya $\to$ elektron $\to$ cacahan).
  \item Foto bilik gelembung dahulu yang melakukan pelihatannya: yakni jejak
        bermuatan yang melingkar dalam medan magnet. Kedua besaran apa
        yang diserahkan lingkarannya kepada Anda, dan partikel mana yang sama sekali tak
        meninggalkan jejak?
  \item Sinar kosmik pemecah rekor
        \cref{exo:b3:particle-physics:12} menghantam dengan energi
        sebuah bola tenis yang diservis dalam satu proton. Apa yang dikatakan
        keberadaannya tentang pemercepat alam berbanding pemercepat kita?
  \item Ringkaslah poster pamerannya dalam empat baris: lahir dari sebutir
        proton sepuluh kilometer di atas, diantarkan oleh \omterm{prop:b3:relativistic-kinematics:dilation}{pemuluran waktu},
        dikenali oleh dua kilatan yang berkebetulan, dan dijelaskan oleh sebuah
        tabel berisi dua belas kursi.
\end{enumerate}
\end{problem}
