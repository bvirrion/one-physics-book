\section*{Chapter \ref{ch:b3:spin-two-level} --- Spin and Two-Level Systems}

\begin{solution}{exo:b3:spin-two-level:1}
(a) Direct multiplication. (b) $[\hat S_x, \hat S_y] =
(\hbar/2)^2\,2\iu\sigma_z = \iu\hbar\hat S_z$. (c) $\hat S^2 =
(\hbar/2)^2 \times 3\,\mathbb 1 = \tfrac34\hbar^2$: indeed
$\tfrac12\cdot\tfrac32\hbar^2$. (d) $\{\mathbb 1, \sigma_x,
\sigma_y, \sigma_z\}$ span the four real dimensions of Hermitian
$2\times2$ matrices ($a + \vect b\cdot\vect\sigma$ with real $a,
\vect b$).
\end{solution}

\begin{solution}{exo:b3:spin-two-level:2}
(a) $\ket{\pm x} = (\ket\uparrow \pm \ket\downarrow)/\sqrt2$;
$\ket{\pm y} = (\ket\uparrow \pm \iu\ket\downarrow)/\sqrt2$. (b)
Half--half. (c) Half--half again: $|\braket{\pm y}{+x}|^2 =
|1 \mp \iu|^2/4 = \tfrac12$. (d) $\ket{+x}$: equator at $\varphi =
0$; measuring along $z$ or $y$ projects onto poles $90^\circ$ away
--- always $\cos^2(45^\circ) = \tfrac12$.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:3}
(a) $\cos^2(\theta/2)$. (b) $\cos^2(\theta/2) \times
\cos^2(\theta/2) = \cos^4(\theta/2)$. (c) $\theta = 90^\circ$:
$(\tfrac12)^2 = \tfrac14$, the polarizer chain's number. (d)
Trivially $\theta = 0$; the real lesson is that \emph{gentle}
intermediate measurements cost least --- the small-step limit of
\cref{exo:b3:quantum-formalism:5}, where many small rotations pass
the state with vanishing loss.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:4}
(a) $2\mu_{\text{B}}B = \qty{116}{\micro eV}$: $\nu =
\qty{28}{GHz}$. (b) $2 \times 2.79\,\mu_{\text{N}}B =
\qty{0.18}{\micro eV}$: $\nu = \qty{42.6}{MHz}$. (c) Against
$k_{\text{B}}T = \qty{25.9}{meV}$: electron polarisation
$\sim\num{2e-3}$, proton $\sim\num{3e-6}$ --- thermal \omterm{def:b3:spin-two-level:spin}{spin}
ensembles are almost perfectly scrambled. (d) ESR: microwaves;
NMR: radio --- three orders of magnitude apart, one physics.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:5}
(a) $\dd\langle\hat S_x\rangle/\dd t = -\gamma B_0\langle\hat
S_y\rangle$, $\dd\langle\hat S_y\rangle/\dd t = \gamma
B_0\langle\hat S_x\rangle$, $\dd\langle\hat S_z\rangle/\dd t = 0$.
(b) $\langle\hat S_x\rangle = \tfrac\hbar2\cos\omega_0t$,
$\langle\hat S_y\rangle = \tfrac\hbar2\sin\omega_0t$ (sign per
$\gamma$'s sign). (c) Both are manifest from (b). (d) The state is
a \emph{superposition} of the two energy eigenstates; its relative
phase advances at $\omega_0$, and that rotating phase \emph{is} the
precession --- stationary levels and turning vector are the two
readings of one two-level evolution.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:6}
(a) Substitute and use half-angle identities. (b)
$\cos\tfrac\theta2\cos\tfrac{\pi-\theta}2 +
\eu^{\iu\pi}\sin\tfrac\theta2\sin\tfrac{\pi-\theta}2 =
\cos\tfrac\theta2\sin\tfrac\theta2 -
\sin\tfrac\theta2\cos\tfrac\theta2 = 0$. (c) On the equator, at
$\varphi = 0, \pi$ ($\sigma_x$) and $\varphi = \pm\pi/2$
($\sigma_y$). (d) A $180^\circ$ rotation about any equatorial axis
--- the $\pi$ pulse of \omterm{prop:b3:spin-two-level:rabi}{magnetic resonance} is the qubit's NOT gate.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:7}
(a) In the rotating frame the drive stands still, while the frame's
rotation subtracts $\omega/\gamma$ from the effective static field
(the same bookkeeping as a rotating-frame inertial force). (b) Only
$B_1$ survives: the Bloch vector precesses about the (fixed)
transverse $B_1$ axis at $\Omega_1 = |\gamma|B_1$ --- steady
flipping. (c) $t_\pi = \pi/\gamma B_1 = 1/(2 \times 42.58\,\text{
MHz/T} \times 10\,\unit{\micro T}) = \qty{1.2}{ms}$. (d) A fast
cone-tightening spiral: precession at \qty{128}{MHz} about $z$,
slowly nutating down at \qty{426}{Hz} --- a thousand-turn
corkscrew from pole to pole.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:8}
(a) $\nu = \Delta E/h = \num{5.9e-6}/\num{4.14e-15} =
\qty{1.42e9}{Hz}$; $\lambda = c/\nu = \qty{21.1}{cm}$. (b) $\Delta
\nu \sim 1/2\pi\tau \sim \qty{e-15}{Hz}$: utterly negligible ---
every observed width is motion. (c) $\Delta\nu =
\nu\,(2v/c) = 1420\,\text{MHz} \times \num{1.5e-3} \approx
\qty{2}{MHz}$: the double-horned profile of a rotating disc. (d)
21 cm speaks wherever hydrogen is atomic (warm, diffuse); CO
speaks where hydrogen has paired into invisible H$_2$ (cold,
dense): together they inventory a galaxy's whole interstellar
medium.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:9}
(a) Two configurations, tunnel-coupled: at low energy the space is
two-dimensional --- ammonia \emph{is} a spin-$\tfrac12$ in
disguise. (b) $\nu = \Delta E/h = \qty{24}{GHz}$, $\lambda =
\qty{1.25}{cm}$. (c) The beam must arrive with the \emph{upper}
state overpopulated (selected by electrostatic deflection) ---
population inversion, unobtainable thermally, since Boltzmann
always favours the lower level. (d) Stimulated amplification by
population inversion --- the working principle of the laser,
demonstrated at microwave frequencies first.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:10}
(a) $\mu_{\text{B}} \times \qty{10}{T} = \qty{0.58}{meV}$ --- the
$\alpha^2E_{\text{I}} \approx \qty{0.7}{meV}$ scale of
\cref{exo:b3:hydrogen-atom:11}: fine structure \emph{is} \omterm{def:b3:spin-two-level:spin}{spin}
meeting the motional field. (b) $\Delta E = hc\,\Delta\lambda/
\lambda^2 = \qty{2.1}{meV}$: an internal field $\Delta E/2\mu_{
\text{B}} \approx \qty{18}{T}$. (c) The inner field grows like
$Z^4$ (nuclear charge cubed in the field, once more in the orbit
radius): heavy atoms have fine structure you can see with a pocket
spectroscope. (d) $j = \tfrac32$: four states; $j = \tfrac12$: two
--- six in all, exactly $2 \times (2l + 1)$ for $l = 1$.
\end{solution}

\begin{solution}{exo:b3:spin-two-level:11}
(a) $S_z$ counts: $+\hbar, 0, 0, -\hbar$; acting with $\hat S^2$
(via the ladder identity) on the symmetric combination gives
$2\hbar^2$ ($S = 1$), on the antisymmetric one $0$ ($S = 0$). (b)
Only the singlet (and the middle triplet state) cannot be written
as a product --- the singlet is \emph{the} maximally entangled
pair. (c) Emission means the triplet lies above: the \omterm{def:b3:spin-two-level:spin}{parallel-spin}
configuration is the more energetic by \qty{5.9}{\micro eV}. (d)
Each observer alone sees perfectly random outcomes; the
correlation appears only when the two lists are \emph{brought
together} --- no local statistics change, so nothing propagates
(the argument of \cref{rem:b3:quantum-formalism:nosignal}).
\end{solution}

\begin{solution}{exo:b3:spin-two-level:12}
(a) The first $\pi/2$ lays the Bloch vector on the equator; during
$T$ it precesses at the atom's own $\nu_0$ while the local
oscillator turns at $\nu$: the accumulated angle difference is
$2\pi\delta T$; the second $\pi/2$ converts that phase into a
population --- an interferometer in time. (b) $\Delta\delta \sim
1/2T = \qty{1}{Hz}$. (c) $\qty{e-4}{Hz}/\qty{9.19e9}{Hz} \approx
\num{e-14}$ --- a second per three million years. (d) The same
absolute fringe-splitting on a carrier $10^5$ times higher wins
$10^5$ in fractional accuracy: hence strontium and ytterbium
optical clocks at $10^{-18}$, and a pending redefinition of the
second.
\end{solution}

\begin{solution}{pb:b3:spin-two-level:1}
\textbf{1.} $\nu = 42.58 \times 3.0 = \qty{127.7}{MHz}$; $h\nu =
\qty{5.3e-7}{eV}$.
\textbf{2.} $\Delta n/n = h\nu/2k_{\text{B}}T = \num{5.3e-7}/(2
\times 0.0267) \approx \num{e-5}$: ten parts per million.
\textbf{3.} $\num{6.6e22}\,\unit{cm^{-3}} \times \num{e-5} \approx
\num{7e17}$ signal-bearing protons per cubic centimetre --- feeble
per \omterm{def:b3:spin-two-level:spin}{spin}, mighty in numbers.
\textbf{4.} Stern--Gerlach measured each atom and split it by
\omterm{def:b3:quantum-formalism:observable}{eigenvalue}; MRI listens to a \emph{thermal average} of $10^{23}$
\omterm{def:b3:spin-two-level:spin}{spins}, and Boltzmann keeps that average within microvolts of zero.
\textbf{5.} Polarisation $\propto B_0$ and the induced EMF
$\propto \omega \propto B_0$: signal roughly $\propto B_0^2$ ---
the case for \qty{3}{T} over \qty{1.5}{T}, and for the
\qty{7}{T} research machines.
\textbf{6.} A ferromagnet in the stray gradient feels
$F = \mu\,\partial B/\partial z$ scaled to kilograms of iron:
hundreds of newtons appearing in a doorway --- the reason for the
screening and the checklists.
\textbf{7.} $\nu_1 = \gamma B_1/2\pi = \qty{426}{Hz}$: $\pi/2$
pulse \qty{0.59}{ms}, $\pi$ pulse \qty{1.2}{ms}.
\textbf{8.} It precesses in the transverse plane at
\qty{127.7}{MHz}; the rotating magnetisation's flux through the
coil induces (Faraday) a radio EMF at exactly that frequency ---
the raw MR signal.
\textbf{9.} Every energy-exchanging flip also randomises the
flipped \omterm{def:b3:spin-two-level:spin}{spin}'s phase: whatever causes $T_1$ contributes to $T_2$,
and dephasing has extra channels of its own --- coherence dies
first.
\textbf{10.} At $\tau$ every runner turns around: the fast ones,
farthest ahead, now have farthest to run back; at $2\tau$ all
cross the start line abreast --- the dephasing rewinds and the
echo rings.
\textbf{11.} Only the \emph{static} part: a \omterm{def:b3:spin-two-level:spin}{spin} that kept a
constant (if wrong) frequency retraces its phase exactly;
fluctuating molecular fields change between the two halves and do
not rewind --- their decay is the true, tissue-specific $T_2$.
\textbf{12.} Sample early and often (short TR, short TE) and
fat's short $T_1$ shines; wait long and echo late and long-$T_2$
fluids glow: the sequence's clock settings choose which relaxation
constant paints the picture.
\textbf{13.} $\dd\nu/\dd z = 42.58\,\unit{MHz/T} \times
0.04\,\unit{T/m} = \qty{1.7}{kHz/mm}$.
\textbf{14.} The slab where the local Larmor frequency falls in
the band: thickness $\qty{2}{kHz}/\qty{1.7}{kHz/mm} \approx
\qty{1.2}{mm}$ --- a selected slice.
\textbf{15.} A Fourier transform: frequency labels position, so
the spectrum of the echo \emph{is} the projection of the slice.
\textbf{16.} $256^2 \times 12 \approx \qty{100}{kB}$; at one
encoded line per repetition time of order $T_1$, $256$ lines cost
minutes --- why patients are asked to hold still.
\textbf{17.} The gradient coils carry kiloampere-scale switched
currents inside \qty{3}{T}: the Laplace force hammers them against
their mounts at every switch --- the machine-gun soundtrack of
every scan.
\textbf{18.} X-rays shadow electron density --- bone versus air ---
and deposit ionising dose; MRI reads proton density and two
relaxation clocks, which differ richly among soft tissues: brain,
cartilage and tumours, all alike to X-rays, are distinct to \omterm{def:b3:spin-two-level:spin}{spins},
at zero ionising dose.
\textbf{19.} Each chemical environment shields its proton by a
few parts per million: a molecule's protons report as a resolved
comb of shifted lines --- structure determination in a tube.
\textbf{20.} Activity raises local blood flow and oxygenation;
paramagnetic deoxyhaemoglobin is diluted; local $T_2^*$ lengthens;
the voxel brightens seconds after the thought.
\textbf{21.} At \qty{84}{GHz} tissue is opaque (millimetre
microwaves barely penetrate skin) and electron relaxation is
microseconds --- hopeless in vivo, but perfect for counting
radicals and defects in materials and dosimetry.
\textbf{22.} $\num{5e-7}\,\unit{eV}$ against eV bonds: ten million
times too weak to break anything --- non-ionising; one
\qty{60}{keV} X-ray photon carries $10^{11}$ times more.
\textbf{23.} From $10^{-5}$ to order one: a $\sim10^5$ gain per
nucleus --- necessary because the inhaled gas is thousands of
times more dilute than water's protons.
\textbf{24.} A spin-$\tfrac12$ is a \emph{perfect} two-level
system (no leakage levels), and it couples to charge noise only
weakly through magnetic moments: long coherence in an industrial
material.
\textbf{25.} A \qty{128}{MHz} precession, a $10^{-5}$
polarisation, two relaxation clocks and three gradients: \omterm{def:b3:spin-two-level:spin}{spin}
physics photographing a living brain, slice by slice, with zero
ionising dose --- the two silver spots of 1922 grown into a
hospital department.
\end{solution}
