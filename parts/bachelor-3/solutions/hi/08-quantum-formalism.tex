\section*{अध्याय \ref{ch:b3:quantum-formalism} --- क्वांटम यांत्रिकी की औपचारिक रचना}

\begin{solution}{exo:b3:quantum-formalism:1}
(क) $(4 + 1)/5 = 1$ और $(1 + 1)/2 = 1$। (ख) $\braket\varphi\psi =
(2 - \iu)/\sqrt{10}$; $\braket\psi\varphi = (2 + \iu)/\sqrt{10}$:
अर्थात् संयुग्मी। (ग) $|(2 - \iu)|^2/10 = 1/2$। (घ)

\end{solution}

\begin{solution}{exo:b3:quantum-formalism:2}

\end{solution}

\begin{solution}{exo:b3:quantum-formalism:3}
(क) $\braket\beta\alpha = \cos\beta\cos\alpha + \sin\beta\sin\alpha
= \cos(\beta - \alpha)$: प्रायिकता $\cos^2(\beta - \alpha)$। (ख)
तीव्रता $\propto$ फ़ोटॉनों की संख्या: अर्थात् मैलस। (ग) हर फ़ोटॉन एक

अक्ष को नहीं चुनता।
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:4}

\hat p^2] = 2\iu\hbar\hat p$। (ग) $\hat B\hat A
\hat C$ जोड़िए और घटाइए। (घ) (ग) के साथ आगमन: $\iu\hbar\,n\hat p^{n-1}$ ---

जो $x$ के साथ \omterm{def:b3:hamiltonian-mechanics:poisson}{प्वासों कोष्ठक} की क्वांटम छाया है।
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:5}

\begin{solution}{exo:b3:quantum-formalism:6}
(क) $[\hat A, \hat B] = \begin{pmatrix}0 & -2\\ 2 & 0\end{pmatrix}
\neq 0$: असंगत। (ख) $\hat B$, $\ket1$ पर: निश्चय ही $+1$, और कोई
संकुचन नहीं चाहिए; फिर $\hat A$, $\pm1$ देता है, प्रायिकता
$\tfrac12$ से। (ग) $\hat A \to +1$ के बाद अवस्था
$(\ket1 + \ket2)/\sqrt2$ है; $\hat B$ उसे $\ket1$ या
$\ket2$ में संकुचित कर देता है (हर एक $\tfrac12$); और दोनों ही स्थितियों में
अंतिम $\hat A$, $\pm1$ देता है प्रायिकता $\tfrac12$ से: अतः खंडन की
प्रायिकता $\tfrac12$। (घ) क्रमविनिमेय \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} अभिलक्षणिक अवस्थाएँ
बाँटते हैं: तब बीच की माप विक्षोभ न करती, और पुनरावृत्ति निश्चय ही
सहमत होती।
\end{solution}

$\hat\beta\ket\psi \propto \hat\alpha\ket\psi$, और अनुपात विशुद्ध
काल्पनिक --- $x, p$ के लिए इस अवकल समीकरण के हल केवल गाउसीय हैं।
(ग) $\Delta p \ge \qty{5.3e-25}{kg.m/s}$: $E \sim
\Delta p^2/2m_{\text{e}} \approx \qty{1}{eV}$ --- परमाणु
इलेक्ट्रॉनवोल्ट की मशीनें इसलिए हैं कि वे ऐंग्स्ट्रॉम के संदूक हैं। (घ)
$\Delta v \ge \qty{5e-27}{m/s}$: कुछ भी नहीं, कभी नहीं।
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:8}
(क) $[\hat H, \hat x] = -\iu\hbar\hat p/m$ और $[\hat H, \hat p] =
\iu\hbar V'(\hat x)$ से $\dd\langle x\rangle/\dd t = \langle
p\rangle/m$, $\dd\langle p\rangle/\dd t = -\langle V'\rangle$। (ख)
आंतरिक गुणनफल में चर-परिवर्तन हर्मिटीयता दिखा देता है;
$\hat\Pi^2 = \mathbb 1$ \omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm1$ पर विवश कर देता है। (ग) $V(-x) =
V(x)$ होने से $\hat H$ सम-विषमता के प्रति अंधा हो जाता है; तब किसी
अ-अपह्रासी अभिलक्षणिक अवस्था को $\hat\Pi$ की भी अभिलक्षणिक अवस्था होना
ही पड़ता है: सम या विषम। (घ) सममित द्विकूप का लगभग-अपह्रासी द्विक:
एक सम, एक विषम अवस्था --- अर्थात् अमोनिया का उत्क्रमण-युग्म, जो सुरंगन
से विपाटित है और \qty{24}{GHz} पर विकिरित करता है।
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:9}
(क) दोनों अवस्थाएँ $E$ बाँटती हैं: ऊर्जा बता देने पर भी दो
संभावनाएँ बचती हैं। (ख) $\hat S$ नाम आपस में बदल देता है: हर्मिटीय, और
उसका वर्ग तत्समक; सममित $\hat H$ के साथ क्रमविनिमेय; और उस स्तर के भीतर उसकी
अभिलक्षणिक अवस्थाएँ $(\ket{1,2} \pm \ket{2,1})/\sqrt2$ हैं, जिनके
\omterm{def:b3:quantum-formalism:observable}{अभिलक्षणिक मान} $\pm1$ हैं। (ग) $(E, +)$ और $(E, -)$: अद्वितीय नाम। (घ)
कोई भी अपह्रासिता अधूरा पता है; और जो सममिति उसे बनाती है, वही छूटा हुआ

\end{solution}

\begin{solution}{exo:b3:quantum-formalism:10}
(क) $\Delta E\,\Delta A \ge \tfrac12|\langle[\hat H, \hat
A]\rangle| = \tfrac\hbar2|\dd\langle A\rangle/\dd t|$; अब भाग दीजिए। (ख)
काल इस सिद्धांत का एक प्राचल है, कोई संकारक नहीं: यह संबंध बाँधता है कि
दी गई ऊर्जा-फैलाव के साथ कोई भी मापनीय वस्तु कितनी \emph{तेज़ी} से
विकसित हो सकती है। (ग) $\Delta E \sim \hbar/2\tau = \qty{3.3e-8}{eV}$:
अर्थात् कुछ मेगाहर्ट्ज़ की प्राकृतिक रेखा-चौड़ाई। (घ) प्रति सेकंड कोई
दिखाई देने लायक परिवर्तन केवल $\Delta E \gtrsim \qty{5e-35}{J}$ माँगता
है --- स्थूल ऊर्जाओं के लिए कोई \omterm{def:b3:lagrangian-mechanics:coordinates}{प्रतिबंध} ही नहीं: घड़ियाँ टिक-टिक कर सकती हैं।
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:11}
(क) $\|\hat\alpha\psi\|^2 = \braket{\psi}{\hat\alpha^2\psi} =
\Delta A^2$, क्योंकि $\hat\alpha$ हर्मिटीय है। (ख) $\braket{\hat\alpha
\psi}{\hat\beta\psi} = \tfrac12\langle\{\hat\alpha,
\hat\beta\}\rangle + \tfrac12\langle[\hat A, \hat B]\rangle$: पहला
पद वास्तविक है (हर्मिटीय भाग), दूसरा विशुद्ध काल्पनिक।
(ग) $|z|^2 \ge (\operatorname{Im}z)^2$ प्रमेय दे देता है; और समता के
लिए सदिशों का अनुक्रमानुपाती होना \emph{तथा} प्रति-क्रमविनिमेयक का औसत
शून्य होना, दोनों चाहिए। (घ) $\hat A$ की किसी अभिलक्षणिक अवस्था का
$\Delta A = 0$ होता है, जिससे
$0 \ge \hbar/2$ बन जाता --- प्रसामान्यनीय अवस्थाओं के लिए असंभव। समतल
तरंगें उसे ``पा'' लेती हैं, पर केवल इसलिए कि वे अ-प्रसामान्यनीय
आदर्शीकरण हैं, जो \omterm{def:b3:quantum-formalism:state}{हिल्बर्ट समष्टि} के बाहर हैं।
\end{solution}

\begin{solution}{pb:b3:quantum-formalism:1}
\textbf{1.} कोई घूर्णन किसी लांबिक प्रसामान्य युग्म को लांबिक
प्रसामान्य युग्म ही बनाता है: $\braket{\nu_e}{\nu_\mu} = -\cos\theta\sin\theta +
\sin\theta\cos\theta = 0$।
\textbf{2.} अ6: मुक्त उड़ान $\hat H$ से उत्पन्न होती है, और उसकी
अभिलक्षणिक अवस्थाएँ कलाओं से स्वतंत्र रूप से विकसित होती हैं ---
उत्पादन ने चाहे कोई भी आधार चुना हो।
\textbf{3.} $\ket{\psi(0)} = \ket{\nu_\mu} = -\sin\theta\ket{\nu_1}
+ \cos\theta\ket{\nu_2}$।
\textbf{4.} $\ket{\psi(t)} = -\sin\theta\,\eu^{-\iu E_1t/\hbar}
\ket{\nu_1} + \cos\theta\,\eu^{-\iu E_2t/\hbar}\ket{\nu_2}$।
\textbf{5.} वैश्विक कलाएँ हर $|\braket\cdot\cdot|^2$ से गिर जाती हैं:
अतः $\Delta\phi = (E_2 - E_1)t/\hbar$ रख लीजिए।
\textbf{6.} $E_2 - E_1 = (m_2^2 - m_1^2)c^4/2E = \Delta m^2c^4/2E$।
\textbf{7.} $\braket{\nu_e}{\psi(t)} = \eu^{-\iu E_1t/\hbar}
\sin\theta\cos\theta\,(\eu^{-\iu\Delta\phi} - 1)$।
\textbf{8.} $|\cdots|^2 = \sin^2\theta\cos^2\theta\,|{\eu^{-\iu
\Delta\phi} - 1}|^2 = \sin^22\theta\,\sin^2(\Delta\phi/2)$।
\textbf{9.} न कोई मिश्रण, या न कोई द्रव्यमान-विपाटन: तो कोई दोलन
नहीं। अतः कोई भी प्रेक्षित दोलन सिद्ध करता है कि $\theta \neq 0$
\emph{और} $m_1 \neq
m_2$ --- अर्थात् न्यूट्रिनो का तौल है।
\textbf{10.} $\Delta\phi = \Delta m^2c^4L/2\hbar cE$ रखिए;
$L_{\text{दो}}$ उस कोणांक को $\pi$ बना देती है।
\textbf{11.} दोनों स्वाद-प्रायिकताएँ किसी प्रसामान्यित अवस्था के
लांबिक प्रसामान्य आधार पर घटकों के वर्ग हैं: अतः उनका योग $1$ है ---
विकास की एकात्मकता ने प्रसामान्य बचा रखा।
\textbf{12.} केवल \emph{आपेक्षिक} कला \omterm{def:b3:quantum-formalism:observable}{प्रेक्ष्य} है, और उसमें
$E_2 - E_1 \propto m_2^2 - m_1^2$ है: निरपेक्ष द्रव्यमान कट जाते हैं।
\textbf{13.} कथित मात्रकों में $\dfrac{L}{4\hbar cE}$:
$\num{e3}\,\unit{m}/(4 \times \num{1.973e-7}\,\unit{eV.m} \times
\num{e9}) = 1.27$ प्रति $\unit{eV^2}$।
\textbf{14.} नीचे भली-भाँति विकसित: $1.27\,\Delta m^2 \times 12800
\gtrsim 1$, अर्थात् $\Delta m^2c^4 \gtrsim \num{6e-5}\,\unit{eV^2}$;
ऊपर नगण्य: $1.27\,\Delta m^2 \times 15 \ll 1$, अर्थात् $\ll
\num{5e-2}$: अतः कहीं $10^{-4}$--$10^{-2}\,\unit{eV^2}$ के आसपास।
\textbf{15.} \qty{1}{GeV} पर $L_{\text{दो}} = \pi E/(1.27\,\Delta m^2c^4) \approx
\qty{990}{km}$: \qty{15}{km} अछूता रह जाता है, जबकि
\qty{12800}{km} पूरी तेरह लंबाइयाँ है --- ठीक वही प्रेक्षित ढाँचा।
\textbf{16.} अनेक लंबाइयों और ऊर्जाओं के फैलाव पर
$\langle\sin^2\rangle = \tfrac12$: अतः दमन $\tfrac12\sin^22
\theta$; और मापा गया आधा $\sin^22\theta \approx 1$ पर विवश करता है,
$\theta \approx 45^\circ$ --- प्रकृति ने अधिकतम मिश्रण चुना।
\textbf{17.} $m_2c^2 \approx \sqrt{\num{2.5e-3}} = \qty{0.05}{eV}$:
अर्थात् इलेक्ट्रॉन से एक करोड़ गुना हलका --- ज्ञात सबसे हलका पदार्थ,
और अब तक कोई नहीं जानता कि क्यों।
\textbf{18.} दाया बे: $1.27 \times \num{2.5e-3} \times 1.5/0.004
\approx 1.2$ --- कोटि एक, अर्थात् सुर में; कामलैंड: $1.27 \times
\num{7.5e-5} \times 180/0.004 \approx 4.3$ --- सौर विपाटन के लिए
कोटि एक: हर आधार-रेखा किसी एक $\Delta m^2$ पर सधा व्यतिकरणमापी है।
\textbf{19.} समस्या: सूर्य के भविष्यवाणी किए गए $\nu_e$ का केवल एक
तिहाई पहुँचता था। समाधान: छूटे हुए दो तिहाई दूसरे
स्वादों के रूप में पहुँचते हैं, जिनमें $\nu_e$ घूम चुके होते हैं (SNO ने
कुल गिनकर सूर्य को निर्दोष पाया)।
\textbf{20.} दोलन के लिए $\Delta m^2 \neq 0$ चाहिए: अतः कम से कम एक
न्यूट्रिनो द्रव्यमान वाला है --- मूल मानक निदर्श से परे की पहली
प्रयोगशाला-भौतिकी।
\textbf{21.} $10^7$ मीटर तक कला-संसक्ति का अर्थ है कि रास्ते में
\emph{कुछ भी} अन्योन्यक्रिया नहीं कर पाया: अनुप्रस्थ काट इतनी छोटी कि
मुट्ठी भर पकड़ने के लिए किलोटन जल चाहिए --- इसीलिए
सूपर-कामिओकांदे के पचास हज़ार टन।
\textbf{22.} स्वाद के संकारक स्वाद-आधार में विकर्ण होते हैं,
जो ऊर्जा-आधार नहीं है: $[\hat H, \text{स्वाद}] \neq 0$ ---
अर्थात् स्वाद मुक्त उड़ान की कोई संरक्षित राशि है ही नहीं, जबकि
ऊर्जा और संवेग हैं।
\textbf{23.} दोनों विपाटन तीस गुना भिन्न हैं: किसी भी दिए गए
$L/E$ पर एक दोलन सक्रिय रहता है और दूसरा या तो जमा रहता है या पूरी तरह
औसत हो जाता है --- अतः हर प्रयोग एक प्रभावी द्वि-स्तरीय तंत्र देखता है।
\textbf{24.} उत्पादन-आधार $\neq$ संचरण-आधार (वही
घूर्णन $\theta$); अ6 दोनों कलाएँ देता है; और \omterm{thm:b3:quantum-formalism:postulates}{बोर्न का नियम}
आपेक्षिक कला को प्रायिकता में बदल देता है।
\textbf{25.} $\theta \approx 45^\circ$ और $\Delta m^2c^4 =
\qty{2.5e-3}{eV^2}$ किसी GeV म्युऑन-न्यूट्रिनो को
\qty{1000}{km} के पास की दोलन-लंबाई दे देते हैं: द्वि-स्तरीय रैखिक
बीजगणित, जिसकी पुष्टि ग्रह के भीतर से हुई, और आधे अभिवाह के ग़ायब हो
जाने पर एक नोबेल पुरस्कार।
\end{solution}
