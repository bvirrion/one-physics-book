\begin{solution}{exo:b3:hydrogen-atom:12}
(क) अवकलन कीजिए: $\dot{\vect p}\wedge\vect L = -(mk/r^3)\vect
r\wedge(\vect r\wedge\dot{\vect r}\,m)$; और दुहरे वज्र-गुणनफल का
प्रसार ठीक $mk\,\dd\vect e_r/\dd t$ देता है: अतः $\dot{\vect A} =
\vect 0$। (ख) कोई संरक्षित अक्ष समदैशिकता से परे की सममिति है;
और अपह्रासिता अधूरा नामकरण है, तथा वह अतिरिक्त नाम ($\vect
A$ का क्वांटम चचेरा भाई) $l$ को एक ही $n$ के भीतर जोड़ देता है। (ग)
$V = -k/r + \epsilon/r^2$ के साथ प्रभावी कोणीय संवेग खिसक जाता है,
कोणीय आवर्तकाल त्रिज्य आवर्तकाल से मेल नहीं खाता, और
दीर्घवृत्त की अक्ष $\propto\epsilon$ की दर से घूम जाती है। (घ) स्वयं
हाइड्रोजन में $\epsilon$ की भूमिका आपेक्षिकता के संशोधन निभाते हैं:
\cref{exo:b3:hydrogen-atom:11} की सूक्ष्म संरचना ठीक वही
$l$-अपह्रासिता का टूटना है।
\end{solution}

\begin{solution}{pb:b3:hydrogen-atom:1}
\textbf{1.} $\langle r\rangle \sim n^2a_0$; $|E_n| =
E_{\text{I}}/n^2$; अंतराल $\approx 2E_{\text{I}}/n^3$; कक्षीय
आवृत्ति $\propto 1/n^3$ (बोर कक्षा पर केप्लर)।
\textbf{2.} $E_{n+1} - E_n \to 2E_{\text{I}}/n^3$, और $h$ गुना
चिरसम्मत आवृत्ति $v_n/2\pi r_n$ वही व्यंजक देता है
(\cref{pb:b3:hamiltonian-mechanics:1} की गणना, मद 20)।
\textbf{3.} $n = 50$: $\qty{132}{nm}$, \qty{5.4}{meV}; $n = 100$:
$\qty{0.53}{\micro m}$, \qty{1.4}{meV}।
\textbf{4.} $n^2ea_0 = \qty{2.1e-26}{C.m} \approx 6300\,\text{D}$:
अर्थात् एक ही परमाणु पर तीन हज़ार जल-अणुओं जितना द्विध्रुव।
\textbf{5.} $F \sim \num{5.4e-3}/\num{1.32e-7} \approx
\qty{4e4}{V/m} = \qty{400}{V/cm}$; और यथार्थ चिरसम्मत-आयनन की
देहली उससे लगभग आठ गुना छोटी है ($\sim\qty{50}{V/cm}$):
दोनों ही तरह, किसी क्षेत्र की फुसफुसाहट भर।
\textbf{6.} $\sim\pi n^4a_0^2$ की ज्यामितीय अनुप्रस्थ काटों के साथ,
प्रयोगशाला का अति-उच्च निर्वात भी ऐसे परमाणु को
माइक्रोसेकंडों में टकरा देता है; जबकि अंतरतारकीय घनत्व
($\sim\num{e6}$ कण प्रति $\unit{m^3}$) उसे दिनों तक छोड़ देते हैं ---
दिक् बेहतर निर्वात-कक्ष है।
\textbf{7.} $\nu = R_{\text{H}}c\,(1/109^2 - 1/110^2) =
\qty{5.01}{GHz}$।
\textbf{8.} सेंटीमीटर रेडियो: कोई रेडियो दूरदर्शी --- एक बड़ी थाली और
एक शांत ग्राही।
\textbf{9.} $\lambda = \qty{6.0}{cm}$; और $n = 110$ का परमाणु
$\sim\qty{0.6}{\micro m}$ का है: अर्थात् प्रति तरंगदैर्घ्य एक लाख
परमाणु --- आराम से ``ऐंटेना'' परिसर।
\textbf{10.} $\Delta\nu/\nu \sim v/c = \num{4.3e-5}$: अर्थात्
\qty{5}{GHz} पर लगभग \qty{200}{kHz} --- और मापी गई चौड़ाई
नीहारिका का ताप लौटा देती है।
\textbf{11.} आयनित भीतरी मंदाकिनी: वे HII क्षेत्र, जो उस धूल के पीछे
छिपे हैं जो हर बामर फ़ोटॉन बुझा देती है --- रेडियो पुनर्संयोजन
रेखाओं ने वह सर्पिल संरचना नक्शे पर लाई जो प्रकाशिक खगोलिकी नहीं देख सकी।
\textbf{12.} पड़ोसी आयनों के सूक्ष्म विद्युत् क्षेत्र (स्टार्क
चौड़ीकरण) स्तरों को धुँधला कर देते हैं, और कृष्णिका/ब्रह्मांडीय विकिरण
तथा संघट्ट इन नाज़ुक अवस्थाओं को आयनित या $l$-मिश्रित कर देते हैं।
\textbf{13.} $C_6 \sim d^4/\Delta E \propto (n^2)^4/n^{-3} =
n^{11}$: $n$ को $1$ से $70$ कीजिए और अन्योन्यक्रिया
बीस कोटि बढ़ जाती है।
\textbf{14.} $R_{\text{b}}$ के भीतर युग्म-अवस्था अनुनाद से बाहर
खिसक जाती है, अतः एक परमाणु का उत्तेजन उसके पड़ोसी की अनुक्रिया को
\emph{प्रतिबंधित} कर देता है: ठीक वही नियंत्रित तर्क, जो किसी
द्वि-क्यूबिट फाटक को चाहिए।
\textbf{15.} माइक्रोमीटर-परास की अन्योन्यक्रियाएँ हर परमाणु को अपनी
ही संबोधनीय चिमटी में बैठने देती हैं --- परमाणु के मानकों से बहुत दूर,
फिर भी प्रबल युग्मित: अर्थात् अन्योन्यक्रिया की परास, प्रकाशिक
विभेदन से सुर में।
\textbf{16.} प्रति जीवनकाल $\sim 200$ फाटक: यही अनुपात प्राप्य
निष्ठा की सीमा बाँधता है, और किसी संसाधक की जान कच्चे
माइक्रोसेकंडों में नहीं, इसी अनुपात में बसती है।
\textbf{17.} \qty{300}{K} पर फ़ोटॉन-कुंड
$\sim\qty{0.1}{meV}$ पर सघन है: वह रिडबर्ग स्तरों को उनके विकिरणी
जीवनकालों के भीतर ही फेंट देता और आयनित कर देता है; जबकि ठंडे
(\qty{4}{K}) कवच उन संक्रमणों को भूखा रख देते हैं।
\textbf{18.} संक्रमण-दरें $d^2 \propto n^4$ की तरह मापित होती हैं: $n =
50$ पर किसी साधारण परमाणु के युग्मन से साठ लाख गुना --- और इसीलिए
रिडबर्ग परमाणुओं की कोई वाष्प-कोठरी अब अंशांकित सूक्ष्मतरंग और
टेराहर्ट्ज़ क्षेत्र-संवेदक है।
\textbf{19.} $92^2 \times \qty{13.6}{eV} = \qty{115}{keV}$ से नीचे
$13.6/10^6\,\unit{eV} = \qty{13.6}{\micro eV}$ तक: एक ही सूत्र से
दस कोटियाँ।
\textbf{20.} चिरसम्मतता वहाँ उगती है जहाँ अंतराल ऊर्जा का लुप्तप्राय
अंश बन जाए, $n \gg 1$ --- अर्थात् परमाणु की अपनी
संगतता-सीमा।
\textbf{21.} यही अकेला परमाणु है जिसे प्रयोग की यथार्थता तक मूल
सिद्धांतों से निकाला जा सकता है: अतः कोई भी बेमेल खोज है, और इसीलिए
हाइड्रोजन मूल नियतांकों का लंगर है।
\textbf{22.} $\tau \approx 1.6\,\unit{ns} \times 50^3 =
\qty{0.2}{ms}$ --- उदार, पर कृष्णिका द्वारा पुनर्वितरण (भाग
III) उसमें से खाता रहता है, जो ठंडे कवचों का एक और कारण है।
\textbf{23.} CPT सममिति --- पदार्थ और प्रतिपदार्थ के परमाणुओं की
रेखा-दर-रेखा मेल खानी चाहिए; और हाइड्रोजन इसलिए लंगर है कि केवल वहीं
सिद्धांत प्रयोग के साथ-साथ पंद्रहवें अंक तक पहुँचता है।
\textbf{24.} $n^2$ वाला द्विध्रुव परमाणु को किसी एक
सूक्ष्मतरंग फ़ोटॉन का क्षेत्र अनुभव करा देता है, जबकि उसका लंबा
जीवनकाल और स्तर-संरचना उसे फ़ोटॉन सोखे \emph{बिना} एक मापनीय कला पा
लेने देते हैं: अर्थात् क्वांटम अ-विध्वंसक संवेदन।
\textbf{25.} मापन $n^2$, $n^{-2}$, $n^{-3}$, $n^{11}$ उस एक हल किए
गए परमाणु को \qty{52.9}{pm} से आधे माइक्रोमीटर तक और
\qty{121.6}{nm} से \qty{6}{cm} तक खींच देते हैं --- वही श्रोडिंगर
हल, जो मंदाकिनी के HII क्षेत्रों से प्रसारण करता है और चिमटी-सरणियों
में द्वि-क्यूबिट फाटक चलाता है।
