\section*{Chapter \ref{ch:b3:canonical-ensemble} --- The Canonical Ensemble}

\begin{solution}{exo:b3:canonical-ensemble:1}
(a) $\mathcal P_2/\mathcal P_1 = (g_2/g_1)\eu^{-\beta\epsilon}$.
(b) $\beta\epsilon = 2.1/(8.62\times10^{-5} \times 2500) = 9.7$:
fraction $3\eu^{-9.7} \approx \num{2e-4}$. (c) A flame carries
$\sim\num{e15}$ sodium atoms per cm$^3$: even $10^{-4}$ of them,
cycling every few nanoseconds, pour out $10^{19}$ yellow photons
per second --- blinding. (d) $3\eu^{-\beta\epsilon} = 0.1$:
$T = \epsilon/(k_{\text{B}}\ln30) \approx \qty{7200}{K}$ --- a
stellar photosphere, not a flame.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:2}
(a) $z = 1 + \eu^{-\beta\epsilon} + \eu^{-2\beta\epsilon}$. (b)
$\langle E\rangle = \epsilon(\eu^{-\beta\epsilon} +
2\eu^{-2\beta\epsilon})/z$: $\to 0$ at low $T$, $\to \epsilon$
(the mean level) at high. (c) $\mathcal P_1 = 1/(\eu^{\beta
\epsilon} + 1 + \eu^{-\beta\epsilon})$ grows monotonically with
$T$, approaching its supremum $1/3$ only as $T \to \infty$. (d)
Boltzmann weights only decrease with energy at $T > 0$; a
population inversion needs the \omterm{ex:b3:microcanonical-ensemble:negative}{negative temperatures} of
\cref{ex:b3:microcanonical-ensemble:negative}, unreachable by any
reservoir.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:3}
(a) The Boltzmann factor on $E_p = mgh$ at uniform $T$. (b) $h_0 =
k_{\text{B}}T/mg = \qty{8.4}{km}$. (c) $\eu^{-8848/8440} \approx
0.35$: one-third of an atmosphere --- why summiteers carry oxygen.
(d) The real atmosphere is not isothermal: temperature falls with
height, so the profile bends away from a single exponential.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:4}
(a) Three translations: $\tfrac32 R = \qty{12.5}{J/(mol.K)}$,
agreed exactly. (b) Add two rotations: $\tfrac52 R =
\qty{20.8}{J/(mol.K)}$, as measured: vibration is frozen. (c)
$\tfrac72 R = \qty{29.1}{J/(mol.K)}$ predicted; the measured
$\approx26$ shows vibration only partly thawed ($\theta_{\text{vib}}
\approx \qty{3400}{K}$). (d) $3R = \qty{24.9}{J/(mol.K)}$:
Dulong--Petit, obeyed at room temperature by most metals and
failed by diamond --- the freezing story of
\cref{exo:b3:canonical-ensemble:6}.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:5}
(a) $z = \eu^{-\beta\hbar\omega/2}/(1 - \eu^{-\beta\hbar\omega})$.
(b) $\langle E\rangle = -\partial_\beta\ln z$ gives the stated
form with $\langle n\rangle = 1/(\eu^{\beta\hbar\omega} - 1)$. (c)
$C \to k_{\text{B}}$ at high $T$; $C \approx k_{\text{B}}(\beta
\hbar\omega)^2\eu^{-\beta\hbar\omega} \to 0$ at low. (d) With
$\hbar\omega$ the energy of a light quantum, $\langle n\rangle$ is
the thermal photon number per mode: Planck's law is one chapter
away.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:6}
(a) Numerically, $C = \tfrac12 \times 3Nk_{\text{B}}$ near
$T/\theta_{\text{E}} \approx 0.34$. (b) $\theta_{\text{E}}/T =
4.3$: $C/3Nk_{\text{B}} = 4.3^2\eu^{-4.3}/(1 - \eu^{-4.3})^2
\approx 0.26$ --- diamond at room temperature has barely a quarter
of the classical heat capacity: the ``anomaly'' is quantum
freezing in plain sight. (c) $\theta_{\text{E}} = \qty{90}{K}$
puts lead deep in the classical regime at \qty{300}{K}. (d) The
single-frequency assumption: real solids have a spectrum of modes
down to long-wavelength sound, whose cheap quanta give the $T^3$
law --- Debye's repair, in \cref{ch:b3:photons-phonons}.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:7}
(a) $f(v) \propto v^2\eu^{-mv^2/2k_{\text{B}}T}$: $v_{\text{p}} =
\sqrt{2k_{\text{B}}T/m}$: \qty{413}{m/s} for N$_2$,
\qty{1540}{m/s} for H$_2$. (b) $mv_{\text{esc}}^2/2k_{\text{B}}T
\approx 730$ for N$_2$, $52$ for H$_2$. (c) $\eu^{-730}$ is
never; $\eu^{-52} \sim \num{e-23}$ per residence time is slow ---
but the hot upper atmosphere ($\sim\qty{1000}{K}$) softens the
hydrogen exponent to $\sim15$: hydrogen bleeds away over
geological time, nitrogen stays. (d) Escape speed and exospheric
temperature: Jupiter's \qty{60}{km/s} well makes even hydrogen's
exponent astronomical --- gas giants keep what small warm worlds
lose.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:8}
(a) $\partial_\beta^2\ln Z = \langle E^2\rangle - \langle
E\rangle^2$, and $C = \partial_T\langle E\rangle =
-k_{\text{B}}\beta^2\partial_\beta\langle E\rangle$. (b) Both
sides give $Nk_{\text{B}}(\beta\epsilon)^2\eu^{\beta\epsilon}/(
\eu^{\beta\epsilon} + 1)^2$. (c) $\langle E\rangle \propto N$,
$\Delta E \propto \sqrt N$. (d) How strongly a system responds to
heating equals how much its energy spontaneously jitters ---
response and fluctuation are two readings of the same second
derivative, a pattern (fluctuation--dissipation) that recurs
throughout physics.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:9}
(a) $\ln2 = E_a\,\Delta T/k_{\text{B}}T^2$ with $\Delta T =
\qty{10}{K}$, $T = \qty{298}{K}$: $E_a =
0.693\,k_{\text{B}}T^2/10 \approx \qty{0.55}{eV}$. (b) From
$298$ to \qty{278}{K}: factor $\eu^{E_a(1/278 - 1/298)/k_{\text{B}}}
\approx 4.4$ slower --- why refrigerators preserve food. (c)
$373 \to \qty{366}{K}$: rate falls by $\approx1.4$: the
eleven-minute egg needs a quarter of an hour. (d) Reactions are
won by the exponential tail of molecules above the barrier: shift
the temperature slightly and the tail's population shifts
enormously --- the mean hardly matters.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:10}
(a) $f_{\text{folded}} = 1/(1 + \eu^{-\beta(\Delta E -
T\Delta S)})$ with the unfolded state's entropy folded into its
\omterm{prop:b3:canonical-ensemble:machine}{free energy}. (b) At $T_{\text{m}} = \Delta E/\Delta S$ the two
free energies tie: half and half. (c) $T_{\text{m}} =
\num{4.8e-19}/\num{1.38e-21} = \qty{348}{K}$ (\qty{75}{\celsius});
width $\delta T \sim k_{\text{B}}T_{\text{m}}^2/\Delta E \approx
\qty{3.5}{K}$. (d) Cooperativity multiplies both $\Delta E$ and
$\Delta S$ by the number of contacts breaking together, keeping
$T_{\text{m}}$ but shrinking the width $\propto 1/\Delta E$: DNA
strands separate over a couple of kelvin, which is what lets a
PCR machine cycle cleanly between ``melted'' and ``annealed''.
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:11}
(a) $z = 2\cosh(\beta\mu B)$; $M = N\mu\tanh(\beta\mu B) \approx
N\mu^2B/k_{\text{B}}T$ for small argument: Curie's $1/T$. (b)
$\mu_{\text{B}}B/k_{\text{B}}T$: $\num{2.2e-3}$ at \qty{300}{K};
$0.67$ at \qty{1}{K} --- from indifferent to strongly aligned. (c)
$S$ depends on $\mu B/k_{\text{B}}T$ alone; lowering $B$ at fixed
entropy drags $T$ down proportionally: $\qty{1}{K} \to
\qty{0.1}{K}$. (d) When $k_{\text{B}}T$ reaches the spin--spin
energy the entropy is no longer field-controlled:
$\mu_0\mu_{\text{B}}^2/4\pi a^3 \approx \qty{7e-26}{J} \sim
\qty{5}{mK}$ --- the classic floor (nuclear moments, a thousand
times weaker, push it to microkelvins).
\end{solution}

\begin{solution}{exo:b3:canonical-ensemble:12}
(a) $z_{\text{rot}} = \sum_J(2J+1)\eu^{-\beta BJ(J+1)}
\to \int(2J+1)\eu^{-\beta BJ(J+1)}\dd J = k_{\text{B}}T/B$: then
$\langle E\rangle = k_{\text{B}}T$ and $C_{\text{rot}} =
k_{\text{B}}$. (b) H$_2$: $\theta_{\text{rot}} = \qty{87}{K}$ ---
the step sits in the laboratory range; N$_2$: \qty{2.9}{K},
frozen out only near liquid helium, so nitrogen always shows
$\tfrac52 R$. (c) Plateaus $\tfrac32 R$ (below $\sim\qty{50}{K}$),
$\tfrac52 R$ (from $\sim200$ to $\sim\qty{1000}{K}$), rising
toward $\tfrac72 R$ near $\theta_{\text{vib}} \approx
\qty{6000}{K}$. (d) Odd and even $J$ belong to different nuclear
spin species that interconvert slowly, so cold hydrogen's $C_V$
depends on its ortho--para history --- nuclear statistics audited
by a calorimeter.
\end{solution}

\begin{solution}{pb:b3:canonical-ensemble:1}
\textbf{1.} $V = \tfrac43\pi r^3 = \qty{4.0e-20}{m^3}$: $m' = V
\Delta\rho = \qty{8.3e-18}{kg}$, weight $m'g = \qty{8.1e-17}{N}$.
\textbf{2.} $n(h) = n_0\,\eu^{-m'gh/k_{\text{B}}T}$: the
barometric law, shrunk to a microscope slide.
\textbf{3.} $h_0 = k_{\text{B}}T/m'g = \num{4.04e-21}/\num{8.1e-17}
\approx \qty{50}{\micro m}$.
\textbf{4.} $h_0 \propto 1/r^3$: a $10\%$ spread in radius is a
$30\%$ spread in scale height --- polydisperse grains smear the
ladder into mush. Perrin fractionated for months by repeated
centrifugation.
\textbf{5.} Same formula, masses $10^{10}$ apart: the grain
``atmosphere'' is $10^{10}$ times shallower --- kilometres shrink
to tens of micrometres, which is precisely what makes it
observable whole under a microscope.
\textbf{6.} Successive ratios $0.55$, $0.55$, $0.57$: constant
within counting error --- exponential. $h_0 =
\qty{30}{\micro m}/\ln(100/55) \approx \qty{50}{\micro m}$.
\textbf{7.} $k_{\text{B}} = m'g\,h_0/T = \num{8.1e-17} \times
\num{5.0e-5}/293 \approx \qty{1.4e-23}{J/K}$.
\textbf{8.} $N_{\text{A}} = R/k_{\text{B}} \approx
\num{6.0e23}\,\unit{mol^{-1}}$.
\textbf{9.} Within a few per cent here (with idealised data);
Perrin's real runs scattered by $\pm15\%$ around the modern value
--- astonishing for hand-counted grains, and utterly decisive for
the order of magnitude.
\textbf{10.} $m_{\text{H}} = \qty{e-3}{kg/mol}/N_{\text{A}} =
\qty{1.7e-27}{kg}$.
\textbf{11.} Macroscopic chemistry supplies $R$; light microscopy
and weighing supply $m'$; counting supplies the ladder: three
tabletop measurements triangulate the mass of an atom no one can
see.
\textbf{12.} Nothing: the derivation used only ``system
exchanging energy with a reservoir'' --- Boltzmann's factor is
size-blind, which is exactly what Perrin verified.
\textbf{13.} $D = k_{\text{B}}T/6\pi\eta r =
\num{4.04e-21}/\num{4.0e-9} \approx \qty{e-12}{m^2/s}$.
\textbf{14.} $\sqrt{2Dt} = \sqrt{\num{1.2e-10}} \approx
\qty{11}{\micro m}$ per minute: comfortably measurable with an
eyepiece graticule and patience.
\textbf{15.} Two independent phenomena, two independent formulas,
one number: agreement of the static ladder and the dynamic jitter
left no niche for coincidence --- the molecular hypothesis
predicted both.
\textbf{16.} $\sqrt{k_{\text{B}}T/m} = \sqrt{\num{4.04e-21}/
\num{4.8e-17}} \approx \qty{9}{mm/s}$ per component.
\textbf{17.} The grain is struck $10^{19}$ times per second and
forgets its velocity within nanometres: the ballistic flight is
unobservable, and what the eye sees is its integral --- the
diffusive random walk.
\textbf{18.} Stokes drag (the viscosity of the Year 2 volume's
fluids) supplies the $6\pi\eta r$; the \omterm{thm:b3:canonical-ensemble:boltzmann}{canonical ensemble} supplies
the $k_{\text{B}}T$: Einstein's $D$ is their quotient, mechanics
and statistics in one fraction.
\textbf{19.} A fiction cannot be counted: once $N_{\text{A}}$ is
the ratio of two measured numbers, with error bars, atoms are
objects of experiment --- Ostwald conceded in print in 1909.
\textbf{20.} Sky-blue scattering, electrolysis with the measured
electron charge, helium accumulated from radium: four unrelated
physical channels converging on one $6\times10^{23}$ made the
number a property of nature rather than of any theory.
\textbf{21.} Gravitational energy pulling the grains down,
configurational entropy spreading them up, traded at the rate $T$:
the ladder \emph{is} the minimum of $F = E - TS$.
\textbf{22.} $h_0 \propto 1/g$: at $10^5g$,
$h_0 = k_{\text{B}}T/(m'\times10^5g) = \num{4.04e-21}/
\num{9.8e-17} \approx \qty{40}{\micro m}$ for the protein ---
sedimentation equilibrium, Perrin's experiment run daily in
biochemistry departments.
\textbf{23.} From ``too tall'' ($h_0 \gg$ field depth: no visible
gradient) to ``too thin'' ($h_0 \lesssim r$): usable radii span
roughly $0.1$--$\qty{0.5}{\micro m}$ --- Perrin's choice was not
luck but design.
\textbf{24.} With $k_{\text{B}}$ now exact by definition, the
same experiment calibrates the grains (their size or density) or
audits the setup: the physics --- Boltzmann's ladder --- is
untouched; only which quantity counts as unknown has moved.
\textbf{25.} A \qty{0.2}{\micro m} grain's population halves every
$\sim\qty{35}{\micro m}$; read through $n_0\eu^{-m'gh/k_{\text{B}}
T}$, the ladder returned $N_{\text{A}} \approx \num{6e23}$:
Avogadro's number counted grain by grain --- and the atomic
debate closed on a microscope stage.
\end{solution}
