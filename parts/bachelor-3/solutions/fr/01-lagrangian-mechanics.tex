\section*{Chapitre \ref{ch:b3:lagrangian-mechanics} --- Mécanique lagrangienne}

\begin{solution}{exo:b3:lagrangian-mechanics:1}
(a) 2 (deux angles sur la surface du bol). (b) 1~: l'abscisse $x$ le
long de la pente, l'angle de rotation lui étant lié par le roulement, $x =
R\phi$. (c) 3~: $X$, $\theta_1$, $\theta_2$. (d) 5~: trois pour le
centre, deux pour la direction de la tige. (e) 2~: un angle chacune. La
condition de roulement est une relation entre \emph{vitesses}, $\dot x =
R\dot\phi$~; dans ce problème plan elle s'intègre aussitôt en $x = R\phi +
\text{cte}$, donc holonome. (Pour une bille roulant sur un plan elle ne
s'intègre pas, et la mécanique lagrangienne demande une extension.)
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:2}
(a) $[L] = \unit{J}$~; $[p_\theta] = \unit{kg.m^2/s}$ --- un moment
cinétique, car $\theta$ est sans dimension. (b) $p_\theta =
m\ell^2\dot\theta$ est le moment cinétique de la masse autour du pivot.
(c) $\ddot\theta = -(g/\ell)\theta$ donne $T = 2\pi\sqrt{\ell/g}$. (d)
Avec $E_p$ comptée depuis l'équilibre, une oscillation harmonique a des
énergies cinétique et potentielle de mêmes moyennes temporelles, donc $\langle
L\rangle = 0$ et $S = 0$ sur une période entière --- le calcul le
confirme~: $S = \int_0^T\tfrac12 m\ell^2\theta_0^2[\omega^2\sin^2
\omega t - (g/\ell)\cos^2\omega t]\,\dd t = 0$ puisque $\omega^2 =
g/\ell$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:3}
(a) Soit $x$ la descente de $m_1$ (donc $m_2$ monte de $x$)~: $L =
\tfrac12(m_1 + m_2)\dot x^2 + (m_1 - m_2)gx$. (b) $(m_1 + m_2)\ddot x =
(m_1 - m_2)g$~: $a = (m_1 - m_2)g/(m_1 + m_2)$. (c) La tension agit aux
deux bouts d'un fil inextensible~: dans tout déplacement autorisé ses
travaux se compensent, elle n'entre donc jamais dans $L$. (d) Newton sur $m_1$ seule~: $T =
m_1(g - a) = 2m_1m_2g/(m_1 + m_2)$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:4}
(a) $\varphi$ et $z$~: $p_\varphi = mr^2\dot\varphi$ (moment cinétique
autour de l'axe) et $p_z = m\dot z$ sont conservés. (b) $r$ figure dans
$L$ via $r^2\dot\varphi^2$, il n'est donc pas cyclique~: $\dot p_r =
mr\dot\varphi^2 \neq 0$. Rien d'anormal --- $p_r = m\dot r$ est la
composante radiale d'un vecteur constant $\vect p$, et une composante sur
une direction \emph{tournante} n'a pas à être constante. (c) $m\ddot r =
mr\dot\varphi^2$~: le terme centrifuge, prix de coordonnées attachées à
des directions qui tournent. (d) Pour une droite à la distance $b$
parcourue à la vitesse $v$~: $r = \sqrt{b^2 + v^2t^2}$, $r^2\dot\varphi = bv$~;
alors $\ddot r = b^2v^2/r^3 = r(bv/r^2)^2 = r\dot\varphi^2$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:5}
(a) Position du bloc $(X + s\cos\alpha,\ -s\sin\alpha)$, donc
\[
L = \tfrac12 M\dot X^2 + \tfrac12 m\big(\dot X^2 +
2\dot X\dot s\cos\alpha + \dot s^2\big) + mgs\sin\alpha .
\]
(b) $X$ est cyclique~: $P = (M + m)\dot X + m\dot s\cos\alpha$ est
conservée --- la quantité de mouvement horizontale totale, aucune force
extérieure horizontale n'agissant. (c) L'équation en $s$ est $\ddot s + \ddot
X\cos\alpha = g\sin\alpha$~; avec $\ddot X = -m\ddot s\cos\alpha/(M +
m)$ tiré de (b),
\[
\ddot s = \frac{(M + m)g\sin\alpha}{M + m\sin^2\alpha} , \qquad
\ddot X = -\frac{mg\sin\alpha\cos\alpha}{M + m\sin^2\alpha} .
\]
(d) $M \to \infty$~: $\ddot s \to g\sin\alpha$, le plan incliné fixe~;
$\alpha \to 90^\circ$~: $\ddot s \to g$, $\ddot X \to 0$ --- chute libre
le long d'une face verticale, le coin restant immobile.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:6}
(a) $L = \tfrac12 m\ell^2(\dot\theta^2 + \sin^2\theta\,\dot\varphi^2) +
mg\ell\cos\theta$. (b) $\varphi$ cyclique~: $p_\varphi =
m\ell^2\sin^2\theta\,\dot\varphi$, le moment cinétique vertical. (c)
En éliminant $\dot\varphi$, $\tfrac12 m\ell^2\dot\theta^2 +
U_{\text{eff}}(\theta)$ est conservée avec
\[
U_{\text{eff}}(\theta) = \frac{p_\varphi^2}{2m\ell^2\sin^2\theta}
 - mg\ell\cos\theta :
\]
un mur en $\theta = 0$ et $\theta = \pi$ (pour $p_\varphi \neq 0$) avec
un minimum entre les deux --- la masse nute entre deux cercles. (d) Au
minimum, $\dot\theta = 0$ avec $\theta = \theta_0$ constant~:
$p_\varphi^2\cos\theta_0/m\ell^2\sin^3\theta_0 = mg\ell\sin\theta_0$~;
en y portant $p_\varphi = m\ell^2\sin^2\theta_0\,\omega$, il vient
$\ell\omega^2\cos\theta_0 = g$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:7}
(a) Masse en $(X + \ell\sin\theta,\ -\ell\cos\theta)$~:
\[
L = \tfrac12(M + m)\dot X^2 + m\ell\cos\theta\,\dot X\dot\theta
 + \tfrac12 m\ell^2\dot\theta^2 + mg\ell\cos\theta .
\]
(b) $X$ cyclique~: $(M + m)\dot X + m\ell\cos\theta\,\dot\theta$ est
conservée --- la quantité de mouvement horizontale du système entier
(le rail ne pousse que verticalement). (c) Petits angles~: $(M + m)\ddot X +
m\ell\ddot\theta = 0$ et $\ell\ddot\theta + \ddot X + g\theta = 0$~;
en éliminant $\ddot X$, $\ell\ddot\theta\,[1 - m/(M + m)] = -g\theta$,
donc $\Omega^2 = (M + m)g/M\ell = (1 + m/M)\,g/\ell$. (d) $M \to \infty$~:
le pivot fixe, $\Omega^2 = g/\ell$. $M$ petite~: le chariot recule à
l'opposé de la masse, l'oscillation se fait autour d'un point situé
entre eux (le centre de masse fixe), ce qui raccourcit le pendule
effectif --- d'où la fréquence plus élevée, divergente quand $M \to 0$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:8}
(a) $\vect A = \tfrac12 B(-y, x, 0)$~:
$L = \tfrac12 m(\dot x^2 + \dot y^2 + \dot z^2) + \tfrac12 qB(x\dot y -
y\dot x)$. (b) L'équation en $x$~: $\dd(m\dot x - \tfrac12 qBy)/\dd t =
\tfrac12 qB\dot y$, c'est-à-dire $m\ddot x = qB\dot y$~; de même $m\ddot y =
-qB\dot x$~: mouvement circulaire à $\omega_{\text{c}} = qB/m$, $\dot z$
restant constante. (c) $p_x = m\dot x - \tfrac12 qBy \neq m\dot x$~; ni $x$
ni $y$ n'est cyclique ($\partial L/\partial x = \tfrac12 qB\dot y$), donc
aucun des deux moments n'est conservé --- seules des combinaisons comme $m\dot x -
qBy$ le sont (vérifier~: sa dérivée est nulle). (d) En coordonnées
cylindriques $A_\varphi = \tfrac12 Br$~:
$L = \tfrac12 m(\dot r^2 + r^2\dot\varphi^2 + \dot z^2) + \tfrac12
qBr^2\dot\varphi$~; $\varphi$ est cyclique, $p_\varphi = mr^2\dot\varphi +
\tfrac12 qBr^2$. Sur un cercle de rayon $R$ centré sur l'axe,
$\dot\varphi = -\omega_{\text{c}}$, donc $p_\varphi = -qBR^2 + \tfrac12
qBR^2 = -\tfrac12 qBR^2$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:9}
(a) $h = \dot\theta\,\partial L/\partial\dot\theta - L = \tfrac12
mR^2\dot\theta^2 - \tfrac12 m\omega^2R^2\sin^2\theta - mgR\cos\theta =
\tfrac12 mR^2\dot\theta^2 + U_{\text{eff}}$~; $\dd h/\dd t =
\dot\theta\,[mR^2\ddot\theta + U_{\text{eff}}'(\theta)] = 0$ par
l'équation du mouvement. (b) $E = \tfrac12 mR^2\dot\theta^2 + \tfrac12
m\omega^2R^2\sin^2\theta - mgR\cos\theta = h +
m\omega^2R^2\sin^2\theta$. (c) $P = \dd E/\dd t =
m\omega^2R^2\sin2\theta\,\dot\theta$~: positive tant que la perle
s'éloigne de l'axe (le moteur travaille contre son inertie),
négative au retour. (d) $U_{\text{eff}}'' = mgR\cos\theta -
m\omega^2R^2\cos2\theta$~; en $\cos\theta_{\text{eq}} = g/\omega^2R$
cela vaut $m\omega^2R^2\sin^2\theta_{\text{eq}}$, donc
$\omega_{\text{osc}} = \omega\sin\theta_{\text{eq}} = \omega\sqrt{1 -
(g/\omega^2R)^2} \to 0$ quand $\omega^2 \to g/R$~: le rappel
s'aplatit exactement lorsque les puits fusionnent.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:10}
(a) Positions $x_2 = \ell(\sin\theta_1 + \sin\theta_2)$, etc.~; en
gardant les termes quadratiques, le terme croisé de vitesse est
$m\ell^2\dot\theta_1\dot\theta_2$, ce qui donne le $L$ annoncé. (b)
$2\ddot\theta_1 + \ddot\theta_2 = -2\omega_0^2\theta_1$ et
$\ddot\theta_1 + \ddot\theta_2 = -\omega_0^2\theta_2$, avec $\omega_0^2
= g/\ell$. (c) En posant $\theta_i = a_i\cos\Omega t$~:
$\det\begin{pmatrix} 2\omega_0^2 - 2\Omega^2 & -\Omega^2\\ -\Omega^2 &
\omega_0^2 - \Omega^2\end{pmatrix} = \Omega^4 - 4\omega_0^2\Omega^2 +
2\omega_0^4 = 0$, donc $\Omega_\pm^2 = (2 \mp \sqrt2)\omega_0^2$~; la
seconde ligne donne $a_2/a_1 = \Omega^2/(\omega_0^2 - \Omega^2) =
\pm\sqrt2$~: masses ensemble (mode lent), masses opposées (mode rapide).
(d) À grande amplitude, les couplages en $\sin$ et $\cos$ rendent les
équations non linéaires~; les solutions ne se superposent plus et des
conditions initiales voisines s'écartent exponentiellement (chaos).
L'énergie reste exactement conservée --- $L$ ne contient pas le temps
explicitement --- le chaos porte sur la prévisibilité, non la conservation.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:11}
(a) $v = \sqrt{2gy}$ et $\dd s = \sqrt{1 + y'^2}\,\dd x$ donnent la
fonctionnelle. (b) Le calcul de
\cref{prop:b3:lagrangian-mechanics:energy} avec $x$ pour temps~: $\dd
h/\dd x = -\partial F/\partial x = 0$. (c) $h = -1/\sqrt{2gy(1 +
y'^2)}$, donc $y(1 + y'^2) = 2r$. Pour la cycloïde, $y' =
\sin\phi/(1 - \cos\phi)$ et $1 + y'^2 = 2/(1 - \cos\phi)$, d'où
$y(1 + y'^2) = 2r$. (d) $\dd t = \dd s/v = \sqrt{r/g}\,\dd\phi$ (toute
dépendance en $\phi$ disparaît), donc le bas ($\phi = \pi$) est atteint en
$\pi\sqrt{r/g}$ --- quel que soit le point de départ~: la cycloïde est
aussi la tautochrone. La glissière rectiligne vers $(\pi r, 2r)$ prend $\sqrt{\pi^2
+ 4}\,\sqrt{r/g} \approx 3.72\sqrt{r/g}$, soit environ $18\%$ de plus que
$\pi\sqrt{r/g}$.
\end{solution}

\begin{solution}{exo:b3:lagrangian-mechanics:12}
(a) $\dd t = \dd s/(c/n)$. (b) $h = -n(y)/\sqrt{1 + y'^2}$ est
conservé~; $1/\sqrt{1 + y'^2} = \cos\theta$ ($\theta$ étant l'angle de pente)
$= \sin i$ pour $i$ compté depuis la verticale~: $n\sin i = \text{cte}$ ---
la loi de Descartes, appliquée continûment. (c) Pour un rayon presque horizontal,
$n\cos\theta \approx \text{cte}$ avec $\theta$ petit donne
$\theta\,\dd\theta = \dd n/n \approx \beta\,\dd y$~; comme $\dd y =
\theta\,\dd x$, la courbure vaut $\dd\theta/\dd x = \beta$, soit $R =
1/\beta \approx \qty{83}{km}$, la courbure se faisant \emph{vers le haut} (vers les grands
$n$). (d) Un rayon parti de l'œil, tangent à la route après s'être incurvé du
rayon $R$, la touche à $d = \sqrt{2hR} = \sqrt{2 \times 1.2 \times
8.3e4} \approx \qty{450}{m}$~: au-delà de cette distance la route elle-même
n'est plus vue --- on voit le ciel réfracté vers le haut, l'«~eau~» miroitante.
\end{solution}

\begin{solution}{pb:b3:lagrangian-mechanics:1}
\textbf{1.} $L = \tfrac12 m\ell^2\dot\theta^2 + mg\ell\cos\theta$~;
$\ddot\theta = -(g/\ell)\sin\theta$.
\textbf{2.} $f_0 = \sqrt{g/\ell}/2\pi = \qty{0.79}{Hz}$.
\textbf{3.} La suspension est fixe, donc $h = E = \tfrac12
m\ell^2\dot\theta^2 - mg\ell\cos\theta$~; du bas au sommet, $\tfrac12
mv^2 = 2mg\ell$~: $v = 2\sqrt{g\ell} = \qty{4.0}{m/s}$.
\textbf{4.} $\ddot\epsilon = +(g/\ell)\epsilon$~: $\epsilon \propto
\eu^{t/\tau}$ avec $\tau = \sqrt{\ell/g} = \qty{0.20}{s}$.
\textbf{5.} L'équilibre inversé existe mais est exponentiellement
instable~: toute inclinaison, si petite soit-elle, est multipliée par $e$ toutes les
\qty{0.2}{s}.
\textbf{6.} Masse en $(\ell\sin\theta,\ a\cos\Omega t - \ell\cos\theta)$~;
dériver et élever au carré~: le $\vect v^{\,2}$ annoncé, avec le terme
croisé $-2a\Omega\ell\sin(\Omega t)\sin\theta\,\dot\theta$.
\textbf{7.} Si $L' = L + \dd F(q,t)/\dd t$, l'\omterm{def:b3:lagrangian-mechanics:action}{action} change de
$F(q_2, t_2) - F(q_1, t_1)$, constante pour des variations à extrémités
fixées~: mêmes chemins stationnaires, mêmes équations.
\textbf{8.} Le terme croisé vaut $-ma\Omega\ell\sin(\Omega t)\sin\theta\,
\dot\theta = \dd[ma\Omega\ell\sin(\Omega t)\cos\theta]/\dd t -
ma\Omega^2\ell\cos(\Omega t)\cos\theta$~; en abandonnant la dérivée totale
et les termes ne dépendant que de $t$ ($\tfrac12 ma^2\Omega^2\sin^2\Omega t$ et
$-mga\cos\Omega t$), il reste le $L$ annoncé.
\textbf{9.} $m\ell^2\ddot\theta = -mg\ell\sin\theta +
ma\Omega^2\ell\cos(\Omega t)\sin\theta$. Comme $\ddot y_{\text{s}} =
-a\Omega^2\cos\Omega t$, cela s'écrit $\ddot\theta =
-[(g + \ddot y_{\text{s}})/\ell]\sin\theta$~: dans le référentiel du
pivot, la pesanteur apparente oscille.
\textbf{10.} $L$ dépend maintenant explicitement de $t$~: $h$ n'est pas
conservée, $E$ non plus --- le vibreur injecte et retire de l'énergie par le pivot.
\textbf{11.} $a/\ell = 0.05$~; $\Omega = \qty{251}{rad/s}$ contre
$\omega_0 = \qty{4.9}{rad/s}$~: rapport $51$. En une période d'excitation
(\qty{25}{ms}) la pesanteur modifie à peine $\dot\theta$~: la masse est trop
inerte pour suivre, et ne fait que frémir.
\textbf{12.} $\ddot\xi$ est la plus grande dérivée ($\propto\Omega^2$)
et l'excitation le plus grand terme de force~: $\ddot\xi =
(a\Omega^2/\ell)\cos(\Omega t)\sin\Theta$.
\textbf{13.} En intégrant deux fois à $\Theta$ fixé~: $\xi =
-(a/\ell)\cos(\Omega t)\sin\Theta$, d'amplitude au plus $a/\ell =
0.05$~: un frémissement de deux degrés.
\textbf{14.} $\sin\theta \approx \sin\Theta + \xi\cos\Theta$, donc
\[
\ddot\Theta + \ddot\xi = -\frac{g}{\ell}(\sin\Theta + \xi\cos\Theta)
 + \frac{a\Omega^2}{\ell}\cos\Omega t\,(\sin\Theta + \xi\cos\Theta) .
\]
\textbf{15.} La moyenne tue $\ddot\xi$, $\langle\cos\Omega t\rangle$
et $\langle\xi\rangle$~; le terme croisé survivant est
$(a\Omega^2/\ell)\cos\Theta\,\langle\xi\cos\Omega t\rangle =
-(a^2\Omega^2/2\ell^2)\sin\Theta\cos\Theta$, ce qui donne l'équation
annoncée pour $\Theta$.
\textbf{16.} $\ddot\Theta = -(1/m\ell^2)\,U_{\text{eff}}'(\Theta)$ avec
$U_{\text{eff}} = mg\ell[-\cos\Theta + (a^2\Omega^2/4g\ell)
\sin^2\Theta]$ --- dériver pour le vérifier.
\textbf{17.} Même terme en $\sin^2$ que pour le cerceau, mais de signe
\emph{opposé}~: la rotation creusait des puits sur les flancs et ne
pouvait qu'aplatir le bas~; la secousse verticale raidit le puits du bas
et creuse un nouveau puits au \emph{sommet}.
\textbf{18.} Excitation lente ($a^2\Omega^2 < 2g\ell$)~: minimum en $\Theta =
0$, maximum en $\pi$ --- rien de neuf. Excitation rapide~: minima en $0$
\emph{et} $\pi$, séparés par des maxima en $\cos\Theta^* =
-2g\ell/a^2\Omega^2$~; pendule pendant et pendule dressé oscillent tous
deux de façon stable.
\textbf{19.} Près de $\pi$, avec $\Theta = \pi + \epsilon$~:
$\ddot\epsilon = [g/\ell - a^2\Omega^2/2\ell^2]\,\epsilon$~; la stabilité
exige la parenthèse négative~: $a^2\Omega^2 > 2g\ell$.
\textbf{20.} $\Omega_{\text{c}} = \sqrt{2g\ell}/a = \qty{140}{rad/s}$~:
$f_{\text{c}} = \qty{22}{Hz}$~; vitesse maximale $a\Omega_{\text{c}} =
\qty{2.8}{m/s}$, accélération maximale $a\Omega_{\text{c}}^2 =
\qty{392}{m/s^2} \approx 40g$.
\textbf{21.} $\omega_{\text{lent}} = \sqrt{a^2\Omega^2/2\ell^2 -
g/\ell}$~; à $\Omega = 2\Omega_{\text{c}}$, $a^2\Omega^2 = 8g\ell$ donc
$\omega_{\text{lent}} = \sqrt{3g/\ell} = \qty{8.6}{rad/s}$~:
\qty{1.4}{Hz}, trente fois plus lent que l'excitation à \qty{45}{Hz}.
\textbf{22.} $\xi_{\max} = (a/\ell)\sin\Theta$~: penché de $10^\circ$ sur
la verticale, $\xi_{\max} = 0.05\sin170^\circ = \qty{8.7}{mrad}
\approx 0.5^\circ$ --- une photographie ordinaire montre un balai
immobile.
\textbf{23.} Le puits du pendule dressé atteint les maxima qui l'encadrent~:
$|\Theta - \pi| < \arccos(2g\ell/a^2\Omega^2)$~; à $\Omega =
2\Omega_{\text{c}}$, $\arccos\tfrac14 = 75^\circ$ --- un puits
remarquablement tolérant.
\textbf{24.} Le jongleur utilise une rétroaction~: les yeux mesurent
l'inclinaison, la main accélère latéralement pour l'annuler. La
stabilisation de Kapitza est en boucle ouverte --- l'excitation ne sait
jamais où est le pendule.
\textbf{25.} Secoué à \qty{45}{Hz} avec une amplitude de \qty{2}{cm}, le
pendule de \qty{40}{cm} tient à l'envers dans un puits de $75^\circ$, en
oscillant à \qty{1.4}{Hz}. Le même potentiel effectif moyenné, obtenu
avec un champ électrique quadrupolaire oscillant au lieu d'un pivot
secoué, confine des ions uniques dans le piège de Paul --- l'outil de
base des horloges atomiques et du calcul quantique à ions piégés.
\end{solution}
