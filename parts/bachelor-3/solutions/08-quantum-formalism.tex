\section*{Chapter \ref{ch:b3:quantum-formalism} --- The Formalism of Quantum Mechanics}

\begin{solution}{exo:b3:quantum-formalism:1}
(a) $(4 + 1)/5 = 1$ and $(1 + 1)/2 = 1$. (b) $\braket\varphi\psi =
(2 - \iu)/\sqrt{10}$; $\braket\psi\varphi = (2 + \iu)/\sqrt{10}$:
conjugates. (c) $|(2 - \iu)|^2/10 = 1/2$. (d) Solve
$\braket\psi\chi = 0$: $\ket\chi = (\ket1 - 2\iu\ket2)/\sqrt5$.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:2}
First: Hermitian; $\pm1$ with $(\ket1 \pm \ket2)/\sqrt2$. Second:
Hermitian; $\pm1$ with $(\ket1 \pm \iu\ket2)/\sqrt2$. Third: not
Hermitian (the transpose-conjugate differs). Fourth: Hermitian;
$\lambda^2 - 5\lambda + 4 = 0$ gives $1$ and $4$, with eigenvectors
$\big({-(1 - \iu)}\ket1 + \ket2\big)/\sqrt3$ and $\big(\ket1 +
(1 + \iu)\ket2\big)/\sqrt3$.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:3}
(a) $\braket\beta\alpha = \cos\beta\cos\alpha + \sin\beta\sin\alpha
= \cos(\beta - \alpha)$: probability $\cos^2(\beta - \alpha)$. (b)
Intensity $\propto$ photon number: Malus. (c) Each photon is an
independent trial: variance $N\mathcal P(1 - \mathcal P)$ --- the
noise itself certifies photons. (d) $|\cos\beta + \iu\sin\beta|^2/2
= 1/2$ for every $\beta$: circular light singles out no transverse
axis.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:4}
(a) $(\hat x\hat p - \hat p\hat x)f = -\iu\hbar(xf' - (xf)') =
\iu\hbar f$. (b) $[\hat x^2, \hat p] = 2\iu\hbar\hat x$; $[\hat x,
\hat p^2] = 2\iu\hbar\hat p$. (c) Add and subtract $\hat B\hat A
\hat C$. (d) Induction with (c): $\iu\hbar\,n\hat p^{n-1}$ ---
$[\hat x, \cdot]$ acts as $\iu\hbar\,\partial/\partial p$, the
quantum shadow of the \omterm{def:b3:hamiltonian-mechanics:poisson}{Poisson bracket} with $x$.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:5}
(a) $|\braket{45^\circ}{V}|^2 = \tfrac12$, collapse, then
$|\braket{H}{45^\circ}|^2 = \tfrac12$: total $\tfrac14$. (b) Taking
them in the order $60^\circ$, $30^\circ$ (three steps of $30^\circ$):
$(\cos^230^\circ)^3 = 27/64 \approx 0.42$. (c) Each step of
$\pi/2N$ passes with $\cos^2(\pi/2N)$: survival $[\cos^2(\pi/2N)]^N
= 0.25$, $0.60$, $0.88$ for $N = 2, 5, 20$. (d) As $N \to \infty$
the survival tends to $1$: many gentle measurements steer the state
through $90^\circ$ without loss --- measurement used as a steering
wheel.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:6}
(a) $[\hat A, \hat B] = \begin{pmatrix}0 & -2\\ 2 & 0\end{pmatrix}
\neq 0$: incompatible. (b) $\hat B$ on $\ket1$: $+1$ with
certainty, no collapse needed; $\hat A$ then gives $\pm1$ with
probability $\tfrac12$ each. (c) After $\hat A \to +1$ the state is
$(\ket1 + \ket2)/\sqrt2$; $\hat B$ collapses it to $\ket1$ or
$\ket2$ (each $\tfrac12$); either way the final $\hat A$ gives
$\pm1$ with probability $\tfrac12$: contradiction with probability
$\tfrac12$. (d) Commuting \omterm{def:b3:quantum-formalism:observable}{observables} share eigenstates: the middle
measurement would not disturb, and the repetition would agree with
certainty.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:7}
(a) $\Delta x = \sigma$ and $\Delta p = \hbar/2\sigma$ (the Fourier
transform of a Gaussian of width $\sigma$ has width $1/2\sigma$ in
$k$): product exactly $\hbar/2$. (b) Equality in Cauchy--Schwarz:
$\hat\beta\ket\psi \propto \hat\alpha\ket\psi$ with purely imaginary
ratio --- for $x, p$ this differential equation has Gaussian
solutions only. (c) $\Delta p \ge \qty{5.3e-25}{kg.m/s}$: $E \sim
\Delta p^2/2m_{\text{e}} \approx \qty{1}{eV}$ --- atoms are
electron-volt machines because they are ångström boxes. (d)
$\Delta v \ge \qty{5e-27}{m/s}$: nothing, ever.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:8}
(a) $[\hat H, \hat x] = -\iu\hbar\hat p/m$ and $[\hat H, \hat p] =
\iu\hbar V'(\hat x)$ give $\dd\langle x\rangle/\dd t = \langle
p\rangle/m$, $\dd\langle p\rangle/\dd t = -\langle V'\rangle$. (b)
Change of variable in the inner product shows Hermiticity;
$\hat\Pi^2 = \mathbb 1$ forces \omterm{def:b3:quantum-formalism:observable}{eigenvalues} $\pm1$. (c) $V(-x) =
V(x)$ makes $\hat H$ parity-blind; a non-degenerate eigenstate must
then be an eigenstate of $\hat\Pi$ too: even or odd. (d) The
symmetric double well's near-degenerate doublet: one even, one odd
state --- the ammonia inversion pair, split by tunnelling, radiating
at \qty{24}{GHz}.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:9}
(a) Both states share $E$: announcing the energy leaves two
possibilities. (b) $\hat S$ swaps the labels: Hermitian, squares to
identity, commutes with the symmetric $\hat H$; within the level its
eigenstates are $(\ket{1,2} \pm \ket{2,1})/\sqrt2$ with \omterm{def:b3:quantum-formalism:observable}{eigenvalues}
$\pm1$. (c) $(E, +)$ and $(E, -)$: unique labels. (d) A degeneracy
is an incomplete address; the symmetry that causes it also provides
the missing digit.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:10}
(a) $\Delta E\,\Delta A \ge \tfrac12|\langle[\hat H, \hat
A]\rangle| = \tfrac\hbar2|\dd\langle A\rangle/\dd t|$; divide. (b)
Time is a parameter of the theory, not an operator: the relation
bounds how \emph{fast} anything measurable can evolve, given the
energy spread. (c) $\Delta E \sim \hbar/2\tau = \qty{3.3e-8}{eV}$:
a natural linewidth of a few megahertz. (d) A visible change per
second demands only $\Delta E \gtrsim \qty{5e-35}{J}$ --- for
macroscopic energies, no \omterm{def:b3:lagrangian-mechanics:coordinates}{constraint} at all: watches may tick.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:11}
(a) $\|\hat\alpha\psi\|^2 = \braket{\psi}{\hat\alpha^2\psi} =
\Delta A^2$ by Hermiticity of $\hat\alpha$. (b) $\braket{\hat\alpha
\psi}{\hat\beta\psi} = \tfrac12\langle\{\hat\alpha,
\hat\beta\}\rangle + \tfrac12\langle[\hat A, \hat B]\rangle$: the
first term is real (Hermitian part), the second purely imaginary.
(c) $|z|^2 \ge (\operatorname{Im}z)^2$ gives the theorem; equality
needs proportional vectors \emph{and} vanishing anticommutator
average. (d) An eigenstate of $\hat A$ has $\Delta A = 0$, forcing
$0 \ge \hbar/2$: impossible for normalisable states. Plane waves
``achieve'' it only by being non-normalisable idealisations outside
the \omterm{def:b3:quantum-formalism:state}{Hilbert space}.
\end{solution}

\begin{solution}{exo:b3:quantum-formalism:12}
(a) At $\omega t = \pi/2$, i.e.\ $t = T$. (b) At each check the
state has rotated by $\pi/2N$; it is found in $\ket1$ with
$\cos^2(\pi/2N)$ and \emph{collapses back} to $\ket1$; the checks
are independent, so the product. (c) $0$, $0.53$, $0.88$, $0.98$:
watched closely enough, the pot never boils. (d) The projection
postulate (P5): each observation resets the evolution to its
starting line.
\end{solution}

\begin{solution}{pb:b3:quantum-formalism:1}
\textbf{1.} A rotation sends an orthonormal pair to an orthonormal
pair: $\braket{\nu_e}{\nu_\mu} = -\cos\theta\sin\theta +
\sin\theta\cos\theta = 0$.
\textbf{2.} P6: free flight is generated by $\hat H$, whose
eigenstates evolve autonomously by phases --- whatever basis
production chose.
\textbf{3.} $\ket{\psi(0)} = \ket{\nu_\mu} = -\sin\theta\ket{\nu_1}
+ \cos\theta\ket{\nu_2}$.
\textbf{4.} $\ket{\psi(t)} = -\sin\theta\,\eu^{-\iu E_1t/\hbar}
\ket{\nu_1} + \cos\theta\,\eu^{-\iu E_2t/\hbar}\ket{\nu_2}$.
\textbf{5.} Global phases drop from every $|\braket\cdot\cdot|^2$:
keep $\Delta\phi = (E_2 - E_1)t/\hbar$.
\textbf{6.} $E_2 - E_1 = (m_2^2 - m_1^2)c^4/2E = \Delta m^2c^4/2E$.
\textbf{7.} $\braket{\nu_e}{\psi(t)} = \eu^{-\iu E_1t/\hbar}
\sin\theta\cos\theta\,(\eu^{-\iu\Delta\phi} - 1)$.
\textbf{8.} $|\cdots|^2 = \sin^2\theta\cos^2\theta\,|{\eu^{-\iu
\Delta\phi} - 1}|^2 = \sin^22\theta\,\sin^2(\Delta\phi/2)$.
\textbf{9.} No mixing, or no mass splitting: no oscillation. Any
observed oscillation proves $\theta \neq 0$ \emph{and} $m_1 \neq
m_2$ --- neutrinos weigh.
\textbf{10.} Substitute $\Delta\phi = \Delta m^2c^4L/2\hbar cE$;
$L_{\text{osc}}$ makes the argument $\pi$.
\textbf{11.} The two flavour probabilities are squared components in
an orthonormal basis of a normalised state: they sum to $1$ ---
unitarity of the evolution preserved the norm.
\textbf{12.} Only the \emph{relative} phase is \omterm{def:b3:quantum-formalism:observable}{observable}, and it
contains $E_2 - E_1 \propto m_2^2 - m_1^2$: absolute masses cancel.
\textbf{13.} $\dfrac{L}{4\hbar cE}$ in the stated units:
$\num{e3}\,\unit{m}/(4 \times \num{1.973e-7}\,\unit{eV.m} \times
\num{e9}) = 1.27$ per $\unit{eV^2}$.
\textbf{14.} Well developed below: $1.27\,\Delta m^2 \times 12800
\gtrsim 1$, i.e.\ $\Delta m^2c^4 \gtrsim \num{6e-5}\,\unit{eV^2}$;
negligible above: $1.27\,\Delta m^2 \times 15 \ll 1$, i.e.\ $\ll
\num{5e-2}$: somewhere around $10^{-4}$--$10^{-2}\,\unit{eV^2}$.
\textbf{15.} $L_{\text{osc}} = \pi E/(1.27\,\Delta m^2c^4) \approx
\qty{990}{km}$ at \qty{1}{GeV}: \qty{15}{km} is untouched,
\qty{12800}{km} is thirteen full lengths --- exactly the observed
pattern.
\textbf{16.} Over many lengths and a spread of energies
$\langle\sin^2\rangle = \tfrac12$: suppression $\tfrac12\sin^22
\theta$; the measured one-half forces $\sin^22\theta \approx 1$,
$\theta \approx 45^\circ$ --- nature chose maximal mixing.
\textbf{17.} $m_2c^2 \approx \sqrt{\num{2.5e-3}} = \qty{0.05}{eV}$:
ten million times lighter than the electron --- the lightest matter
known, and nobody yet knows why.
\textbf{18.} Daya Bay: $1.27 \times \num{2.5e-3} \times 1.5/0.004
\approx 1.2$ --- order one, tuned; KamLAND: $1.27 \times
\num{7.5e-5} \times 180/0.004 \approx 4.3$ --- order one for the
solar splitting: each baseline is an interferometer set to one
$\Delta m^2$.
\textbf{19.} Problem: only a third of the Sun's predicted $\nu_e$
arrived. Resolution: the missing two-thirds arrive as other
flavours, into which the $\nu_e$ have rotated (SNO counted the
total and found the Sun innocent).
\textbf{20.} Oscillation requires $\Delta m^2 \neq 0$: at least one
neutrino is massive --- the first laboratory physics beyond the
original Standard Model.
\textbf{21.} Phase coherence over $10^7$ metres means essentially
\emph{nothing} interacted en route: cross-sections so small that
kilotonnes of water are needed to catch a handful --- hence
Super-Kamiokande's fifty thousand tonnes.
\textbf{22.} Flavour operators are diagonal in the flavour basis,
which is not the energy basis: $[\hat H, \text{flavour}] \neq 0$ ---
flavour is simply not a conserved quantity of free flight, while
energy and momentum are.
\textbf{23.} The two splittings differ thirtyfold: at any given
$L/E$ one oscillation is active and the other either frozen or fully
averaged --- each experiment sees an effective two-level system.
\textbf{24.} Production basis $\neq$ propagation basis (the
rotation $\theta$); P6 supplies the two phases; Born's rule turns
the relative phase into a probability.
\textbf{25.} $\theta \approx 45^\circ$ and $\Delta m^2c^4 =
\qty{2.5e-3}{eV^2}$ give a GeV muon-neutrino an oscillation length
near \qty{1000}{km}: two-level linear algebra, verified through the
planet, and a Nobel Prize for the disappearance of half a flux.
\end{solution}
