\chapter{Pousser et tirer~: les forces}\label{ch:g2:pushes-and-pulls}

Ouvrir une porte, remonter une fermeture éclair, taper dans un
ballon, tirer une luge~: ta journée entière se passe à pousser et à
tirer. La physique rassemble tout cela sous un nom fier --- la
\omterm{def:g2:pushes-and-pulls:force}{force} --- et pose deux questions simples~: que peut faire une \omterm{def:g2:pushes-and-pulls:force}{force},
et dans quel sens agit-elle~?

\section{Des poussées et des tractions partout}

\begin{definition}[Force]\label{def:g2:pushes-and-pulls:force}
Une \emph{force}\index{force} est une poussée ou une traction.
Pousser une balançoire, tirer un chariot, appuyer sur une sonnette,
étirer un élastique, presser de la pâte~: chacun de ces gestes est
une force au travail. Une force a toujours une intensité --- douce ou
puissante --- et un \emph{sens}~: elle pousse ou tire \emph{dans une
certaine direction}.
\end{definition}

\begin{example}[Pousser ou tirer~?]\label{ex:g2:pushes-and-pulls:sort}
Trie ta journée~: taper dans un ballon --- une poussée du pied.
Ouvrir un tiroir --- une traction. Le fermer --- une poussée. Remonter
la fermeture d'un blouson --- une traction. Pétrir de la pâte ---
beaucoup de petites poussées. Le tir à la corde --- deux équipes qui
tirent la même corde en sens opposés.
\end{example}

\begin{method}[Dessiner une force par une flèche]\label{met:g2:pushes-and-pulls:arrow}
Les physiciens dessinent une \omterm{def:g2:pushes-and-pulls:force}{force} par une \emph{flèche}~:
\begin{enumerate}
  \item la flèche part de l'endroit où la poussée ou la traction
        s'exerce sur l'objet~;
  \item elle pointe dans le sens où la \omterm{def:g2:pushes-and-pulls:force}{force} agit~;
  \item une \omterm{def:g2:pushes-and-pulls:force}{force} plus intense reçoit une flèche plus longue.
\end{enumerate}
Une seule petite flèche raconte toute l'histoire~: où, dans quel
sens, avec quelle intensité. Nous dessinerons ces flèches pendant des
années.
\end{method}

\section{Ce que font les forces}

\begin{proposition}[Les quatre métiers d'une force]\label{prop:g2:pushes-and-pulls:jobs}
Une \omterm{def:g2:pushes-and-pulls:force}{force} peut faire quatre choses à un objet~:
\begin{enumerate}
  \item le \emph{mettre en mouvement} --- le coup de pied envoie au
        loin le ballon immobile~;
  \item l'\emph{arrêter} --- les mains du gardien arrêtent le tir~;
  \item le \emph{faire tourner} --- une pichenette de côté courbe la
        trajectoire du ballon qui roule~;
  \item \emph{changer sa forme} --- presser écrase l'éponge, tirer
        étire l'élastique.
\end{enumerate}
Pas de \omterm{def:g2:pushes-and-pulls:force}{force}, aucun des quatre~: un objet laissé complètement
tranquille continue exactement ce qu'il faisait.
\end{proposition}

\begin{example}[Force plus intense, effet plus
grand]\label{ex:g2:pushes-and-pulls:stronger}
Tapote une bille~: elle roule mollement jusqu'au bord de la table.
Envoie-la d'une pichenette énergique~: elle traverse la pièce en
volant. Même bille, même doigt --- seule l'intensité de la poussée a
changé, et le vol de la bille a changé avec elle. \omterm{def:g2:pushes-and-pulls:force}{Force} douce, effet
doux~; \omterm{def:g2:pushes-and-pulls:force}{force} puissante, effet puissant.
\end{example}

\begin{example}[À toi de jouer~: les changeurs de
forme]\label{ex:g2:pushes-and-pulls:shape}
Rassemble une éponge, un élastique, un morceau de pâte à modeler et
un bloc de bois. Presse et étire chacun d'eux. L'éponge reprend sa
forme dès que tu lâches~; l'élastique aussi. La pâte à modeler garde
sa nouvelle forme --- ta \omterm{def:g2:pushes-and-pulls:force}{force} y a laissé un creux pour toujours. Et
le bloc~? Ta plus forte pression ne le change en rien (du moins à
l'œil~!). Chaque objet répond à la \omterm{def:g2:pushes-and-pulls:force}{force} à sa manière.
\end{example}

\section{Des forces sans contact}

\begin{example}[L'attraction de la Terre]\label{ex:g2:pushes-and-pulls:earthpull}
Lâche une cuillère~: elle tombe. Lance un ballon aussi haut que tu
veux~: il redescend. Les feuilles, les gouttes de pluie, les
miettes --- tout ce qu'on lâche tombe \emph{vers le bas}. Quelque
chose les tire tous, sans arrêt, sans les toucher~: la Terre
elle-même. L'attraction de la Terre est la \omterm{def:g2:pushes-and-pulls:force}{force} la plus fidèle de ta
vie --- elle ne lâche jamais, et c'est pour cela que «~le bas~»
signifie la même chose partout dans la cour.
\end{example}

\begin{remark}[De vieilles connaissances]\label{rem:g2:pushes-and-pulls:magnets}
Tu connais déjà une autre \omterm{def:g2:pushes-and-pulls:force}{force} sans contact~: l'\omterm{def:g1:magnets:magnet}{aimant}, qui tire son
trombone par-dessus un espace d'\omterm{def:g2:air-around-us:air}{air}. Les \omterm{def:g1:magnets:magnet}{aimants} et l'attraction de
la Terre sont les deux \omterm{def:g2:pushes-and-pulls:force}{forces} sans contact de ta vie de tous les
jours --- des exceptions rares et précieuses, car toutes les autres
poussées et tractions que tu rencontres exigent un contact. Toutes
deux deviendront de grands chapitres plus loin dans ce cours.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:1}
Poussée ou traction~? Sonner à une porte~; \ ouvrir un tiroir~; \ faire
une tête au football~; \ tirer une luge dans la montée~; \ presser un
tube de dentifrice.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:2}
Quelles trois choses la flèche d'une \omterm{def:g2:pushes-and-pulls:force}{force} nous apprend-elle sur
cette \omterm{def:g2:pushes-and-pulls:force}{force}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:3}
Nomme les quatre métiers d'une \omterm{def:g2:pushes-and-pulls:force}{force}, et donne ton propre exemple
pour chacun --- différent de ceux du chapitre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:4}
Une bille roule droit vers toi. Avec un seul doigt, comment la faire~:
s'arrêter~? \ aller plus vite~? \ dévier sur le côté~? Dis dans quel
sens ton doigt pousse à chaque fois.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:5}
Qu'est-ce qui reprend sa forme et qu'est-ce qui garde le creux~: une
éponge~; \ de la pâte à modeler~; \ un élastique~; \ de la neige
molle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:6}
Dessine une caisse poussée doucement vers la droite et la même caisse
poussée fortement vers la gauche. En quoi tes deux flèches
diffèrent-elles~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:7}
Quelle \omterm{def:g2:pushes-and-pulls:force}{force} fait tomber tout ce qu'on lâche~? A-t-elle besoin de
toucher la cuillère qui tombe~? Dans quel sens tire-t-elle toujours~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:8}
Donne les deux \omterm{def:g2:pushes-and-pulls:force}{forces} sans contact que tu as rencontrées, chacune
avec un exemple vu à la maison.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:pushes-and-pulls:9}
Au tir à la corde, les deux équipes tirent de toutes leurs \omterm{def:g2:pushes-and-pulls:force}{forces}, et
pourtant le ruban noué sur la corde ne bouge parfois pas du tout.
Dessine la corde avec les flèches des deux équipes. Que doit-il être
vrai des deux tractions quand le ruban reste immobile~? Que se
passe-t-il à l'instant où une équipe tire plus fort~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:pushes-and-pulls:10}
Un ballon lancé droit vers le haut ralentit, s'arrête le temps d'un
clin d'œil, et retombe. Quelle \omterm{def:g2:pushes-and-pulls:force}{force} agit à la montée, au sommet, et
à la descente~? Cesse-t-elle jamais de tirer~? (Dessine la flèche à
chacun des trois instants.)
\end{exercise}
