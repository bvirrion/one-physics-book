\chapter{Mesurer les longueurs}\label{ch:g2:measuring-length}

Deux portes, dans deux maisons différentes~: laquelle est la plus
haute~? Tu ne peux pas porter une porte à travers la ville pour la
dresser contre l'autre. L'an dernier, nous comparions les objets côte
à côte~; cette année, nous apprenons le plus grand tour de la
science --- transformer une longueur en un \emph{nombre} capable de
voyager.

\section{Comparer sans nombres}

\begin{example}[Côte à côte]\label{ex:g2:measuring-length:sidebyside}
Pour trouver le plus long de deux crayons, pose-les l'un contre
l'autre sur la table~: celui qui dépasse gagne. Pour comparer ta
taille à celle d'un camarade, mettez-vous dos à dos. Cela marche à
merveille --- tant que les deux objets peuvent être réunis.
\end{example}

\begin{example}[Le tour de la ficelle]\label{ex:g2:measuring-length:string}
Les deux portes ne peuvent pas se rencontrer --- mais une ficelle,
elle, peut leur rendre visite à toutes les deux. Tends une ficelle en
travers de la première porte et coupe-la exactement à cette hauteur.
Porte la ficelle jusqu'à la seconde porte~: si la ficelle est trop
courte, la seconde porte est la plus haute. La ficelle transporte la
hauteur de la première porte à travers la ville. Un pas de plus~: et
si nous pouvions envoyer la hauteur dans une \emph{lettre}~? Pour
cela, il nous faut des nombres.
\end{example}

\section{Mesurer avec des pas et des empans}

\begin{definition}[Mesurer une longueur]\label{def:g2:measuring-length:measure}
\emph{Mesurer}\index{mesurer} une longueur, c'est compter combien de
fois une \emph{unité}\index{unité} choisie --- un pas, un empan de la
main, un petit bâton --- tient le long de cette longueur, bout à
bout, sans trou et sans chevauchement. La réponse est un nombre
d'unités~: «~la pièce mesure $12$ pas de long~».
\end{definition}

\begin{example}[À toi de jouer~: arpente la
pièce]\label{ex:g2:measuring-length:pace}
Traverse ta chambre en posant un pied juste devant l'autre, en
comptant les longueurs de pied. Demande à un adulte de faire pareil.
Tu comptes peut-être $18$ pieds et l'adulte seulement $12$ --- pour
la \emph{même pièce}~! Personne ne s'est trompé~: tes pieds sont plus
courts, il en tient donc davantage. Une mesure faite avec \emph{mon}
pied ne peut pas être vérifiée avec \emph{ton} pied.
\end{example}

\begin{remark}[La dispute au marché]\label{rem:g2:measuring-length:argument}
Pendant très longtemps, les gens ont vraiment mesuré en pieds, en
pouces et en brasses, et les marchés résonnaient de disputes~: le
pied de qui~? Celui du grand marchand ou celui du petit client~? La
solution, acceptée dans le monde entier, a été de choisir \emph{une
seule} \omterm{def:g2:measuring-length:measure}{unité} pour tous --- une \omterm{def:g2:measuring-length:measure}{unité} qui n'appartient au corps de
personne.
\end{remark}

\section{Le centimètre et le mètre}

\begin{definition}[Centimètre et mètre]\label{def:g2:measuring-length:units}
Le \emph{centimètre}\index{centimètre} (noté \unit{cm}) est une
petite \omterm{def:g2:measuring-length:measure}{unité} de longueur, à peu près la largeur d'un ongle~; il est
imprimé sur toutes les règles. Le \emph{mètre}\index{mètre} (noté
\unit{m}) est une grande \omterm{def:g2:measuring-length:measure}{unité}, exactement $100$ centimètres, la
longueur d'un très grand pas. Ces \omterm{def:g2:measuring-length:measure}{unités} sont les mêmes pour tout le
monde sur Terre~: tes \qty{20}{cm} et mes \qty{20}{cm} sont
exactement égaux.
\end{definition}

\begin{method}[Mesurer avec une règle]\label{met:g2:measuring-length:ruler}
\begin{enumerate}
  \item Poser la règle le long de l'objet, en la faisant toucher~;
  \item placer le \emph{trait du zéro} de la règle exactement à une
        extrémité de l'objet --- pas le bord de la règle, le $0$~;
  \item lire le nombre à l'autre extrémité~: c'est la longueur en
        \omterm{def:g2:measuring-length:units}{centimètres}.
\end{enumerate}
Si l'extrémité tombe entre deux traits, dis «~entre \qty{7}{cm} et
\qty{8}{cm}, plus près de $8$~» --- des mots honnêtes valent mieux
qu'un nombre inventé.
\end{method}

\begin{example}[Quelle unité pour quel
travail~?]\label{ex:g2:measuring-length:whichunit}
Les petits objets, comme un scarabée, un timbre ou ta main, se
mesurent en \omterm{def:g2:measuring-length:units}{centimètres}. Les grands, comme un couloir, un autobus ou
une piscine, se mesurent en \omterm{def:g2:measuring-length:units}{mètres}~: la piscine fait \qty{25}{m} de
long --- imagine un peu, en largeurs d'ongle~! Choisir une \omterm{def:g2:measuring-length:measure}{unité}
raisonnable, c'est la moitié du travail de mesure.
\end{example}

\begin{example}[Estimer, puis vérifier]\label{ex:g2:measuring-length:estimate}
Avant de \omterm{def:g2:measuring-length:measure}{mesurer}, chacun devine~: quelle est la longueur de la table
de la cuisine~? Papa dit \qty{2}{m}, Lina dit \qty{1}{m} et un peu.
Puis le \omterm{def:g2:measuring-length:units}{mètre} tranche~: \qty{1}{m} et presque la moitié --- un point
pour Lina. Deviner d'abord entraîne l'œil~; \omterm{def:g2:measuring-length:measure}{mesurer} garde tout le
monde honnête.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:1}
Comment comparer ta taille à celle d'un camarade sans aucune règle~?
Et à celle d'un camarade qui habite une autre ville~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:2}
Mara mesure la salle de classe en pas et trouve $20$~; sa maîtresse
en trouve $14$. Qui s'est trompé en comptant~? Explique.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:3}
Qu'est-ce que le \omterm{def:g2:measuring-length:units}{centimètre} a de particulier, comparé à une longueur
de pied ou à un empan de main~? Pourquoi les gens se sont-ils mis
d'accord dessus~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:4}
Quelle \omterm{def:g2:measuring-length:measure}{unité} convient le mieux, \unit{cm} ou \unit{m}~: la longueur
d'une fourmi~; \ la hauteur d'une porte~; \ la longueur de la cour de
récréation~; \ la largeur de ce livre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:5}
Tim place le bord de sa règle --- et non le trait du zéro --- contre
l'extrémité de son crayon, et lit \qty{13}{cm}. Son crayon mesure-t-il
vraiment \qty{13}{cm}~? Qu'a-t-il oublié de
\cref{met:g2:measuring-length:ruler}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:6}
Mesure avec une règle~: la largeur de ta main~; \ la longueur d'une
cuillère~; \ la hauteur d'une tasse. Écris chaque réponse avec son
\omterm{def:g2:measuring-length:measure}{unité}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:7}
Une porte mesure \qty{2}{m} de haut. Cela fait combien de
\omterm{def:g2:measuring-length:units}{centimètres}~? (Rappel~: \qty{1}{m} vaut $100$ \unit{cm}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:8}
Devine d'abord, mesure ensuite~: combien de \omterm{def:g2:measuring-length:units}{mètres} mesure ton lit~?
Ta réponse était-elle trop grande, trop petite, ou presque juste~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:9}
Un ruban mesure \qty{45}{cm} de long, un autre \qty{38}{cm}. Lequel
est le plus long, et de combien de \omterm{def:g2:measuring-length:units}{centimètres}~? Aurais-tu pu décider
\emph{sans} nombres~? Quelle façon de faire est la plus facile à
écrire dans une lettre~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:10}
Papi dit~: «~Mon jardin fait $30$ enjambées de long.~» Explique
gentiment à Papi pourquoi «~$30$ enjambées~» ne peut pas être
vérifié par quelqu'un d'autre, et ce qu'il devrait utiliser à la
place. Que se passerait-il si un petit enfant arpentait le même
jardin~?
\end{exercise}
