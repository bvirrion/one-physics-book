\chapter{Glace, eau, vapeur}\label{ch:g2:ice-water-steam}

La même eau tinte dans ton verre sous forme de glaçon, se verse comme
boisson et monte de la casserole de soupe en panache blanc. Une seule
matière, trois \emph{états}. L'an dernier, nous avons regardé la
chaleur faire \omterm{def:g2:ice-water-steam:meltfreeze}{fondre} un glaçon~; cette année, nous rencontrons les
trois états de la matière que l'eau peut prendre.

\section{Les trois états de l'eau}

\begin{example}[Où vit chaque état]\label{ex:g2:ice-water-steam:where}
L'eau peut se trouver dans trois \emph{états de la matière}.
\emph{Solide}~: la glace --- glaçons, stalactites de glace sous le
toit, givre sur l'herbe, neige. Un solide est dur et garde sa forme.
\emph{Liquide}~: l'eau qui se verse --- la pluie, les rivières, la
mer, ton verre. Un liquide \omterm{def:g1:floating-and-sinking:words}{coule} et prend la forme de son récipient.
\emph{Gaz}~: la vapeur d'eau, de l'eau qui vole dans l'air sans se
voir. Solide, liquide, gaz --- mais toujours la même matière en
dessous.
\end{example}

\begin{remark}[La vapeur d'eau est invisible]\label{rem:g2:ice-water-steam:invisible}
Le panache blanc que tu vois au-dessus de la casserole est fait de
minuscules gouttelettes d'eau en suspension --- de petites gouttes
\emph{liquides}. Le vrai gaz --- la vapeur d'eau mêlée à l'air --- ne
se voit pas du tout~! Regarde de près le bec de la casserole~: juste
au-dessus, il y a un court espace \emph{transparent} avant que le
panache blanc ne commence. Dans cet espace, l'eau est un gaz,
invisible~; un peu plus haut, elle se rassemble en gouttelettes
visibles. De la vapeur d'eau invisible se cache dans l'air de chaque
pièce, et nous la prendrons la main dans le sac avant la fin du
chapitre.
\end{remark}

\section{Changer d'état}

\begin{definition}[Fondre et geler]\label{def:g2:ice-water-steam:meltfreeze}
Quand la chaleur transforme la glace solide en eau liquide, la glace
\emph{fond}\index{fondre}. Quand un grand froid transforme l'eau
liquide en solide, l'eau \emph{gèle}\index{geler}. Fondre et geler
sont deux portes opposées entre les deux mêmes états~: la chaleur
pousse dans un sens, le froid pousse dans l'autre.
\end{definition}

\begin{definition}[S'évaporer et se condenser]\label{def:g2:ice-water-steam:evapcond}
Quand la chaleur fait partir l'eau liquide dans l'air sous forme de
gaz, l'eau \emph{s'évapore}\index{évaporer}~--- la flaque rétrécit,
le linge sèche. Quand la vapeur d'eau touche quelque chose de froid
et redevient des gouttelettes liquides, elle \emph{se
condense}\index{condenser} --- la buée sur la vitre froide, le voile
sur le miroir de la salle de bain. De nouveau deux portes opposées~:
la chaleur envoie l'eau à l'état gazeux, le froid la rappelle à
l'état liquide.
\end{definition}

\begin{example}[À toi de jouer~: attrape l'eau
invisible]\label{ex:g2:ice-water-steam:catch}
Prends une canette ou un verre à la sortie du réfrigérateur, sèche
bien l'extérieur avec un torchon, et pose-le sur la table. Attends en
comptant jusqu'à $100$. L'extérieur se couvre de buée, puis devient
mouillé --- des gouttes apparaissent de nulle part~! Elles ne
traversent pas le verre~: la vapeur d'eau invisible de l'air de la
pièce a touché la paroi froide et s'est condensée. Tu viens
d'attraper le gaz invisible.
\end{example}

\begin{example}[Les nuages de la respiration]\label{ex:g2:ice-water-steam:breath}
Ton souffle transporte toujours de la vapeur d'eau invisible. Un jour
de chaleur, elle reste invisible. Un matin glacé, l'air froid du
dehors la condense aussitôt en un petit nuage devant ta bouche ---
et la même chose se produit sur une vitre froide quand tu souffles
dessus pour dessiner du doigt.
\end{example}

\begin{remark}[Vapeur et cuisinière~: prudence]\label{rem:g2:ice-water-steam:safety}
La vapeur d'eau au-dessus d'une casserole qui bout est brûlante ---
plus brûlante que l'eau elle-même ne le paraît. Ne mets jamais la
main dans le panache, et soulève les couvercles en les écartant de
toi, avec un adulte à côté. Les changements d'état se regardent, ils
ne se touchent pas, dès qu'une cuisinière est de la partie.
\end{remark}

\section{La même eau, encore et encore}

\begin{example}[La flaque qui rétrécit]\label{ex:g2:ice-water-steam:puddle}
Après la pluie, une flaque brille dans la cour. L'après-midi, elle a
rétréci~; demain, elle aura disparu. Personne ne l'a bue et elle n'a
pas filé à travers le goudron~: la chaleur du Soleil l'a fait
s'\omterm{def:g2:ice-water-steam:evapcond}{évaporer}, goutte après goutte, dans l'air. Trace un trait de craie
autour d'une flaque le matin et regarde après le déjeuner~: l'eau
s'est repliée à l'intérieur de ton trait.
\end{example}

\begin{example}[De la mer à la pluie]\label{ex:g2:ice-water-steam:cycle}
Ce qui \omterm{def:g2:ice-water-steam:evapcond}{s'évapore} doit bien se \omterm{def:g2:ice-water-steam:evapcond}{condenser} quelque part. L'eau s'envole
des mers et des flaques, invisible~; haut dans le ciel, où l'air est
froid, elle \omterm{def:g2:ice-water-steam:evapcond}{se condense} en gouttelettes qui forment les nuages~; les
gouttelettes se rassemblent, deviennent lourdes, et tombent --- la
pluie~! La pluie remplit les flaques et les mers, et le
voyage recommence. L'eau de ton verre a fait ce trajet plus de fois
qu'on ne peut le compter.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:1}
Quel est l'état de la matière de chacun de ces exemples --- solide,
liquide ou gaz~: une stalactite de glace~; \ la mer~; \ l'eau
invisible juste au-dessus du bec de la casserole de soupe~; \ le
givre sur l'herbe~; \ la pluie~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:2}
Nomme le changement d'état~: un bonhomme de neige qui rétrécit au
soleil~; \ une flaque qui disparaît~; \ la buée qui apparaît sur le
miroir de la salle de bain~; \ un étang qui devient dur en hiver.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:3}
Sur le schéma des changements d'état, dans quel sens la chaleur
pousse-t-elle toujours~: vers le solide ou vers le gaz~? Et le
froid~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:4}
Les gouttes sur la canette froide de
\cref{ex:g2:ice-water-steam:catch} --- ont-elles filtré à travers le
verre~? D'où viennent-elles~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:5}
Pourquoi vois-tu ton souffle un matin glacé, et pas un jour de
chaleur~? Sers-toi des mots de
\cref{def:g2:ice-water-steam:evapcond}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:6}
Le linge mouillé sèche plus vite au soleil qu'à l'\omterm{def:g1:light-and-shadows:shadow}{ombre}. Quel
changement d'état est à l'œuvre, et qu'apporte le Soleil pour
l'accélérer~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:7}
Pourquoi le panache au-dessus d'une casserole qui bout est-il
dangereux, et quelle est la règle de sécurité de
\cref{rem:g2:ice-water-steam:safety}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:8}
Raconte le voyage circulaire d'une goutte de pluie~: pars de la mer,
reviens à la mer, et utilise en chemin les mots
\emph{s'\omterm{def:g2:ice-water-steam:evapcond}{évaporer}} et \emph{se \omterm{def:g2:ice-water-steam:evapcond}{condenser}}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:ice-water-steam:9}
Au bec de la casserole, il y a un espace transparent, et le panache
blanc ne commence qu'au-dessus. Qu'y a-t-il dans l'espace
transparent~? Que montre cela sur l'eau à l'état gazeux~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:ice-water-steam:10}
Nino laisse un verre d'eau sur le rebord de la fenêtre pendant une
semaine~; le niveau baisse jour après jour, alors que personne ne le
boit. On accuse son chat. Défends le chat~: explique ce qui s'est
vraiment passé, et invente un moyen de montrer que l'eau part
\emph{dans l'air}, et non dans le chat.
\end{exercise}
