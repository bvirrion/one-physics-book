\chapter{الدفع والسحب: القوى}\label{ch:g2:pushes-and-pulls}

تفتح بابًا، وتغلق سحّابًا، وتركل كرة، وتجرّ زلّاقة: يومك كله
دفعٌ وسحب. وتجمع الفيزياء هذا كله تحت اسم واحد شريف --- \omterm{def:g2:pushes-and-pulls:force}{القوة}
--- وتسأل سؤالين بسيطين: ما الذي تستطيع \omterm{def:g2:pushes-and-pulls:force}{القوة} أن تفعله، وفي أيّ
اتجاه تعمل؟

\section{دفع وسحب في كل مكان}

\begin{definition}[القوة]\label{def:g2:pushes-and-pulls:force}
\emph{القوة}\index{قوة} دفع أو سحب. دفع أرجوحة، وسحب عربة، وضغط
جرس الباب، ومطّ شريط مطّاطي، وعصر عجين: كل واحد من هذه قوة في
عملها. وللقوة دائمًا شدّة --- رفيقة أو جبّارة --- و\emph{اتجاه}:
فهي تدفع أو تسحب في \emph{جهة ما}.
\end{definition}

\begin{example}[دفع أم سحب؟]\label{ex:g2:pushes-and-pulls:sort}
صنّف يومك: ركل كرة --- دفع بالقدم. وفتح درج --- سحب. وإغلاقه ---
دفع. وإغلاق سحّاب سترة --- سحب. وعجن العجين --- دفعات صغيرة
كثيرة. وشدّ الحبل --- فريقان يسحبان الحبل نفسه في جهتين
متعاكستين.
\end{example}

\begin{method}[رسم القوة سهمًا]\label{met:g2:pushes-and-pulls:arrow}
يرسم الفيزيائيون \omterm{def:g2:pushes-and-pulls:force}{القوة} \emph{سهمًا}:
\begin{enumerate}
  \item يبدأ السهم حيث يمسك الدفع أو السحب بالجسم؛
  \item ويشير في الجهة التي تعمل فيها \omterm{def:g2:pushes-and-pulls:force}{القوة}؛
  \item \omterm{def:g2:pushes-and-pulls:force}{والقوة} الأشدّ تنال سهمًا أطول.
\end{enumerate}
سهم صغير واحد يحكي القصة كلها: أين، وفي أيّ جهة، وبأيّ شدّة.
وسنرسم هذه الأسهم سنوات مقبلة.
\end{method}

\section{ماذا تفعل القوى}

\begin{proposition}[أعمال القوة الأربعة]\label{prop:g2:pushes-and-pulls:jobs}
تستطيع \omterm{def:g2:pushes-and-pulls:force}{القوة} أن تفعل بالجسم أربعة أشياء:
\begin{enumerate}
  \item أن \emph{تبدأ} حركته --- فالركلة تُطيّر الكرة الساكنة؛
  \item أن \emph{توقفه} --- فيدا حارس المرمى توقفان التسديدة؛
  \item أن \emph{تحرف} مساره --- فنقرة جانبية تثني مسار الكرة
        المتدحرجة؛
  \item أن \emph{تغيّر شكله} --- فالعصر يهرس الإسفنجة، والمطّ
        يمدّ الشريط المطّاطي.
\end{enumerate}
فإن لم تكن \omterm{def:g2:pushes-and-pulls:force}{قوة} فلا شيء من الأربعة: والجسم المتروك وحده تمامًا
يواصل ما كان يفعله بالضبط.
\end{proposition}

\begin{example}[قوة أشدّ وأثر أكبر]\label{ex:g2:pushes-and-pulls:stronger}
انقر كرة زجاجية: تتدحرج إلى حافة الطاولة. واقذفها بشدّة: تطير
عبر الغرفة. الكرة نفسها والإصبع نفسه --- ولم تتغيّر إلا شدّة
الدفع، فتغيّر معها طيران الكرة. \omterm{def:g2:pushes-and-pulls:force}{قوة} رفيقة وأثر رفيق؛ \omterm{def:g2:pushes-and-pulls:force}{وقوة}
جبّارة وأثر جبّار.
\end{example}

\begin{example}[جرّب: مغيّرات الشكل]\label{ex:g2:pushes-and-pulls:shape}
اجمع إسفنجة وشريطًا مطّاطيًّا وقطعة معجونة ومكعّبًا خشبيًّا.
واعصر كلًّا منها ومُدَّه. تعود الإسفنجة إلى شكلها لحظة تفلتها؛
والشريط المطّاطي كذلك. أما المعجونة فتحفظ شكلها الجديد ---
فقوّتك صنعت فيها غمّازة إلى الأبد. والمكعّب؟ أشدّ عصرة تعصرها
لا تغيّره البتة (ممّا يمكن أن تراه!). فالأشياء تجيب \omterm{def:g2:pushes-and-pulls:force}{القوة} كلٌّ
على طريقته.
\end{example}

\section{قوى بلا لمس}

\begin{example}[جذب الأرض]\label{ex:g2:pushes-and-pulls:earthpull}
أفلت ملعقة: تنزل. واقذف كرة عاليًا كما تشاء: تعود نازلة. الأوراق،
وقطرات المطر، والفتات --- كل ما يُترك يسقط \emph{إلى الأسفل}. فثمّة
شيء يسحبها كلها في كل حين من غير أن يلمسها: الأرض نفسها. وجذب
الأرض أثبت \omterm{def:g2:pushes-and-pulls:force}{قوة} في حياتك --- فهو لا يفلت أبدًا، وهو السبب في أن
``الأسفل'' يعني الشيء نفسه في كل موضع من ساحة اللعب.
\end{example}

\begin{remark}[أصدقاء قدامى]\label{rem:g2:pushes-and-pulls:magnets}
أنت تعرف \omterm{def:g2:pushes-and-pulls:force}{قوة} أخرى بلا لمس: \omterm{def:g1:magnets:magnet}{المغناطيس}، وهو يسحب مشبكه عبر فجوة من
\omterm{def:g2:air-around-us:air}{هواء}. والمغانط وجذب الأرض هما القوّتان بلا لمس في حياتك اليومية
--- وهما استثناءان نادران ثمينان، إذ يحتاج كل دفع أو سحب آخر
تلقاه إلى تماسّ. وستنمو الاثنتان فصولًا عظيمة لاحقًا في هذا
المنهاج.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:1}
دفع أم سحب؟ قرع جرس الباب؛ \ فتح درج؛ \ ضرب كرة القدم بالرأس؛
\ جرّ زلّاقة صعودًا؛ \ عصر معجون الأسنان.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:2}
ما الأشياء الثلاثة التي يخبرك بها سهم \omterm{def:g2:pushes-and-pulls:force}{القوة} عن \omterm{def:g2:pushes-and-pulls:force}{القوة}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:3}
سمِّ أعمال \omterm{def:g2:pushes-and-pulls:force}{القوة} الأربعة، وأعطِ مثالك أنت لكل واحد منها ---
مختلفًا عن أمثلة الفصل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:4}
كرة زجاجية تتدحرج نحوك مستقيمة. كيف تجعلها بإصبع واحد فقط:
تقف؟ \ تسرع؟ \ تنحرف إلى الجانب؟ قل في أيّ جهة يدفع إصبعك في
كل مرة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:5}
أيّها يعود إلى شكله وأيّها يحفظ الغمّازة: إسفنجة؛ \ معجونة؛
\ شريط مطّاطي؛ \ ثلج ليّن؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:6}
ارسم صندوقًا مدفوعًا برفق إلى اليمين، والصندوق نفسه مدفوعًا
بشدّة إلى اليسار. بمَ يختلف السهمان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:7}
أيّ \omterm{def:g2:pushes-and-pulls:force}{قوة} تجعل كل شيء مفلَت يسقط؟ أتحتاج إلى لمس الملعقة الساقطة؟
وفي أيّ جهة تسحب دائمًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:pushes-and-pulls:8}
اذكر القوّتين بلا لمس اللتين لقيتهما، مع مثال لكل واحدة رأيته في
البيت.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:pushes-and-pulls:9}
في شدّ الحبل يسحب الفريقان بكل ما لديهما من \omterm{def:g2:pushes-and-pulls:force}{قوة}، ومع ذلك لا
يتحرّك الشريط المربوط في الحبل أحيانًا البتة. ارسم الحبل وسهمَي
الفريقين. ما الذي يجب أن يصحّ في السحبتين حين يقف الشريط ساكنًا؟
وماذا يحدث لحظة يسحب أحد الفريقين بشدّة أكبر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:pushes-and-pulls:10}
كرة تُقذف إلى الأعلى مستقيمة تبطئ، وتقف قدر رمشة عين، ثم تسقط
راجعة. أيّ \omterm{def:g2:pushes-and-pulls:force}{قوة} تعمل في الصعود، وعند القمة، وفي النزول؟ وهل
تكفّ عن السحب في لحظة؟ (ارسم السهم في كل لحظة من الثلاث.)
\end{exercise}
