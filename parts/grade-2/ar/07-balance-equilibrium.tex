\chapter{الاتزان والتوازن}\label{ch:g2:balance-equilibrium}

أرجوحة متأرجحة تستوي أفقيًّا وعلى كل طرف منها طفل. وبرج من
المكعّبات يقف --- ومكعّب واحد زيادة، فينهار. وبهلوانة تمشي على
حبل أرفع من حذائها. والثلاثة يلعبون اللعبة العميقة نفسها: لعبة
\omterm{def:g2:balance-equilibrium:equilibrium}{التوازن}، حيث تتلاشى القوى فلا يتحرّك شيء.

\section{السكون توازن}

\begin{definition}[التوازن]\label{def:g2:balance-equilibrium:equilibrium}
يكون الشيء في \emph{توازن}\index{توازن} --- ونقول أيضًا إنه
\emph{متوازن}\index{متوازن} --- حين تلغي الدفعات والسحبات الواقعة
عليه إحداها الأخرى، فيبقى على حاله بالضبط. فشريط شدّ الحبل واقفًا
ساكنًا بين فريقين جبّارين: توازن. وليس هذا عالمًا بلا قوى ---
بل عالمًا تتعادل فيه القوى تعادلًا تامًّا.
\end{definition}

\begin{example}[الكتاب على الطاولة]\label{ex:g2:balance-equilibrium:book}
كتاب على الطاولة، ساكن تمامًا. أمملّ؟ انظر عن قرب بعينَي الفصل
الماضي. الأرض تسحب الكتاب إلى الأسفل --- فلماذا لا يسقط؟ لأن
الطاولة تدفعه \emph{إلى الأعلى}، بالشدّة نفسها التي تسحب بها
الأرض إلى الأسفل. سهمان متساويان متعاكسان: تعادل. أزِل الطاولة
ينكسر التعادل: فينزل الكتاب.
\end{example}

\begin{remark}[السكون لا يعني انعدام القوى]\label{rem:g2:balance-equilibrium:notfree}
قبل هذا الفصل كان قولنا ``لا شيء يحدث للكتاب'' يبدو بديهيًّا.
أما الآن فقد صرت تعلم خيرًا من ذلك: قوّتان تعملان عليه في كل
لحظة. ومن حولك تُمسَك الكراسي والمصابيح والبيوت والجبال في تعادلات
هادئة من السحب والدفع. فالسكون ليس خلاءً --- إنما هو \omterm{def:g2:balance-equilibrium:equilibrium}{توازن}.
\end{remark}

\section{الأرجوحة المتأرجحة}

\begin{example}[تعادل في ساحة اللعب]\label{ex:g2:balance-equilibrium:seesaw}
يجلس صديقان متقاربان في الحجم على طرفَي أرجوحة متأرجحة: فتستوي
أفقيًّا --- تعادل. والآن يجلس بابا على أحد الطرفين: فينزل، وتطير
الطفلة إلى الأعلى. ولكي يعدل في اللعب ينزلق بابا \emph{نحو
الوسط}: فكلما جلس أقرب إلى المحور ضغط جانبُه أضعف --- حتى تستوي
الأرجوحة أفقيًّا من جديد عند الموضع المناسب تمامًا. فالوزن الكبير
قرب الوسط يستطيع أن يتعادل مع وزن صغير بعيد.
\end{example}

{\small تعادل تامّ في ساحة اللعب: راكبان متساويان في الحجم، وعلى المسافة نفسها --- فتستوي الأرجوحة أفقيًّا.}
\end{omfigure}

\begin{proposition}[قاعدة الأرجوحة]\label{prop:g2:balance-equilibrium:rule}
الأوزان المتساوية على مسافات متساوية من المحور تتوازن. والراكب
الأثقل يميل بالأرجوحة --- إلا أن يقترب الراكب الأثقل من المحور،
أو يبتعد الأخفّ. فالوزن \emph{والمسافة} كلاهما يحسب.
\end{proposition}

\begin{remark}[وعد]\label{rem:g2:balance-equilibrium:promise}
كم ينبغي أن ينزلق بابا بالضبط؟ في الأرجوحة قاعدة جميلة دقيقة
مختبئة --- وفيها أعداد حقيقية. وهي تنتظرك في السنة القادمة، في
فصل الروافع والموازين؛ وألعاب الأرجوحة اليوم تدريبها الأمثل.
\end{remark}

\section{الانقلاب ونقطة التوازن}

\begin{example}[البرج الذي مال أكثر مما ينبغي]\label{ex:g2:balance-equilibrium:tower}
ابنِ برجًا من المكعّبات، كل مكعّب بارز عن الذي تحته قليلًا. يقف،
ويقف\dots\ ثم ينهار فجأة. فالشيء المائل ينقلب حين يميل \emph{أبعد
مما ينبغي} خارج قاعدته. ولهذا يصعب جدًّا إسقاط هرم من مكعّبات ذي
قاعدة عريضة، بينما يسقط عمود واحد طويل من نفخة: فالأقدام العريضة
تصنع أبراجًا راسخة.
\end{example}

\begin{definition}[نقطة التوازن]\label{def:g2:balance-equilibrium:point}
\emph{نقطة التوازن}\index{نقطة توازن} في جسم هي الموضع الخاصّ
الذي يمكن أن يُحمل عنده مستويًا --- كأن وزنه كله تجمّع هناك.
والمسطرة تحمل نقطة توازنها في الوسط؛ والمطرقة تخبّئها قرب رأسها
الثقيل. وأيّ جسم يُمسك تحت نقطة توازنه تمامًا يستقرّ مستويًا على
إصبع واحدة.
\end{definition}

\begin{method}[كيف تجد نقطة التوازن]\label{met:g2:balance-equilibrium:find}
لتجد \omterm{def:g2:balance-equilibrium:point}{نقطة التوازن} في مسطرة أو مكنسة أو ملعقة خشبية:
\begin{enumerate}
  \item ضعها عرضًا على سبّابتين ممدودتين، واحدة قرب كل طرف؛
  \item وحرّك إصبعيك على مهل إحداهما نحو الأخرى؛
  \item فستلتقيان --- والجسم \omterm{def:g2:balance-equilibrium:equilibrium}{متوازن} دائمًا --- عند موضع واحد:
        هو \omterm{def:g2:balance-equilibrium:point}{نقطة التوازن}.
\end{enumerate}
جرّب المكنسة: تلتقي إصبعاك بعيدًا عن الوسط، في جهة الفرشاة
الثقيلة. \omterm{def:g2:balance-equilibrium:point}{فنقطة التوازن} تحبّ الجهة الثقيلة.
\end{method}

\begin{example}[عصا البهلوانة]\label{ex:g2:balance-equilibrium:acrobat}
لماذا تحمل ماشية الحبل تلك العصا الطويلة المتهدّلة؟ إنها معينتها
على \omterm{def:g2:balance-equilibrium:equilibrium}{التوازن}: فحين تبدأ الميل إلى جهة تجعل العصا الطويلة الميل
بطيئًا متثاقلًا بدل أن يكون سريعًا، وتمنحها وقتًا لتعود فوق
الحبل. وأنت تستعمل الحيلة نفسها بذراعين عاريتين حين تمشي على حافة
رصيف --- ذراعان ممدودتان، وترنّح مُبطَّأ، \omterm{def:g2:balance-equilibrium:equilibrium}{وتوازن} منقذ.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:1}
مصباح واقف ساكن على المكتب. سمِّ القوّتين العاملتين عليه وقل
لماذا لا يتحرّك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:2}
في شدّ الحبل، متى يكون الشريط في \omterm{def:g2:balance-equilibrium:equilibrium}{توازن}؟ وما الذي يكسر التعادل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:3}
يجلس توأمان متساويان على أرجوحة متأرجحة على مسافتين متساويتين.
ماذا تفعل الأرجوحة؟ ثم ينزلق أحد التوأمين إلى الخلف، فيبتعد عن
المحور --- فأيّ الجانبين ينزل الآن؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:4}
بابا أثقل من لينا بكثير، ومع ذلك تستوي أرجوحتهما أفقيًّا. فأين
لا بدّ أن يكون بابا جالسًا؟ واذكر القاعدة التي تنقذ اللعبة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:5}
أين \omterm{def:g2:balance-equilibrium:point}{نقطة توازن} المسطرة؟ وأين \omterm{def:g2:balance-equilibrium:point}{نقطة توازن} المطرقة --- نحو طرف
المقبض أم نحو الرأس الثقيل؟ وكيف تتحقّق بإصبع واحدة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:6}
اصنع حيلة الإصبعين في \cref{met:g2:balance-equilibrium:find}
بمكنسة. في أيّ جهة تلتقي إصبعاك، وماذا يخبرك ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:7}
أيّهما أرسخ وقوفًا ولماذا: هرم من مكعّبات ذو قاعدة عريضة، أم برج
عرضه مكعّب واحد بالارتفاع نفسه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:8}
لماذا تحمل ماشية الحبل عصا طويلة؟ وماذا تفعل بذراعيك حين تمشي
على حافة رصيف، ولماذا يفيد ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:9}
قِف وكتفك الأيمن وقدمك اليمنى ملتصقان بجدار، وحاول أن ترفع قدمك
اليسرى. لا تستطيع --- إلا أن تسقط! اشرح ذلك ببرج
\cref{ex:g2:balance-equilibrium:tower}: ماذا كان ينبغي أن يفعل
جسمك ليتوازن على القدم اليمنى، وما الذي يمنعه الجدار؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:balance-equilibrium:10}
رافعة في ورشة بناء ترفع أحمالًا ثقيلة على ذراعها الأمامية
الطويلة --- ومع ذلك لا تنقلب. وعلى ذراعها الخلفية القصيرة كتل
خرسانية ضخمة. اشرح الرافعة بقاعدة الأرجوحة في
\cref{prop:g2:balance-equilibrium:rule}: مَن الراكبان، ولماذا
كانت الذراع الخلفية قصيرة؟
\end{exercise}
