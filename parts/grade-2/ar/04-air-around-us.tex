\chapter{الهواء من حولنا}\label{ch:g2:air-around-us}

لوّح بيدك بسرعة أمام وجهك: يمسّ خدَّك شيء ليّن. لا تستطيع أن تراه
ولا أن تمسكه ولا أن تضعه في درج --- ومع ذلك يملأ كل غرفة، وكل
قنّينة، وكل ساحة لعب على الأرض. هذا الفصل عن \omterm{def:g2:air-around-us:air}{الهواء}: تلك المادة
الخفيّة التي نعيش في قاعها.

\section{أهو شيء حقًّا؟}

\begin{definition}[الهواء]\label{def:g2:air-around-us:air}
\emph{الهواء}\index{هواء} هو المادة الخفيّة التي حولنا في كل مكان.
لا لون له، ولا رائحة خاصة به، \omterm{def:g1:light-and-shadows:source}{والضوء} ينفذ منه نفاذًا --- ولذلك
ننسى وجوده. لكنه مادة حقيقية، كما ستثبت تجارب هذا الفصل، وهو
ثمين: فكل نَفَس تأخذه جرعة منه.
\end{definition}

\begin{example}[جرّب: أمسك شيئًا من الهواء]\label{ex:g2:air-around-us:catch}
افتح كيسًا بلاستيكيًّا على سعته، ومرّره في الغرفة، ثم اِفتله
مغلقًا. يقف الكيس منتفخًا كالوسادة --- اعصره: يدفع في وجهك! فمن
الواضح أن شيئًا محبوس في داخله. افتح ركنًا منه واعصره ثانية: تحسّ
بالسجين الخفيّ ينسكب على أصابعك. وكيس ``فارغ'' مملوء بلا شيء ما
كان ليدفعك.
\end{example}

\begin{example}[أين يُظهر الهواء نفسه]\label{ex:g2:air-around-us:shows}
\omterm{def:g2:air-around-us:air}{الهواء} خفيّ، لكن حيله ليست خفيّة: فهو يتموّج مرتعشًا فوق طريق
ساخن، ويرفع الطائرات الورقية، ويحفّ في الأوراق، ويصنع فقاعات
حيثما هرب تحت الماء. انفخ في كأسك من خلال ماصّة: كل فقاعة فضّية
طرد صغير من \omterm{def:g2:air-around-us:air}{الهواء} في طريقه إلى الأعلى.
\end{example}

\section{الهواء يشغل حيّزًا}

\begin{method}[غطسة المنديل الجافّ]\label{met:g2:air-around-us:drytowel}
حيلة سحرية هي فيزياء خالصة --- اصنعها فوق حوض.
\begin{enumerate}
  \item احشُ منديلًا ورقيًّا جافًّا في قاع كأس حشوًا محكمًا حتى
        لا يسقط؛
  \item اقلب الكأس رأسًا على عقب بالضبط؛
  \item ادفعها مستقيمة إلى الأسفل في حوض ماء، أعمق من الكأس
        كلها، من غير إمالة؛
  \item ثم اسحبها مستقيمة إلى الخارج والمِس المنديل.
\end{enumerate}
المنديل جافّ تمامًا! فالكأس لم تكن فارغة قطّ: كانت مملوءة هواءً،
\omterm{def:g2:air-around-us:air}{والهواء} المحبوس احتفظ بحيّزه، فلم يستطع الماء أن يدخل.
\end{method}

\begin{example}[القنّينة المقرقرة]\label{ex:g2:air-around-us:glug}
ادفع قنّينة ``فارغة'' فوهتها إلى الأسفل في الحوض وأملها قليلًا:
قرقرة، فقرقرة، فقرقرة --- فقاعات فضّية كبيرة تسابق إلى السطح،
ويأخذ الماء مكانها في الداخل. كانت القنّينة مملوءة هواءً؛ ولا
يستطيع الماء أن يدخل إلا بقدر ما يخرج \omterm{def:g2:air-around-us:air}{الهواء}. وتلك القرقرة عند
حوض المطبخ هي صوت \omterm{def:g2:air-around-us:air}{الهواء} والماء يتبادلان الحيّز.
\end{example}

{\small القنّينة المقرقرة: بينما ينسكب الماء خارجًا تشقّ فقاعات \omterm{def:g2:air-around-us:air}{هواء} كبيرة طريقها إلى الداخل --- فالحيّز ``الفارغ'' لا بدّ أن يُملأ.}
\end{omfigure}

\begin{remark}[لا وجود لقنّينة فارغة]\label{rem:g2:air-around-us:noempty}
من الآن فصاعدًا، ارتب في كلمة ``فارغ''. القنّينة الفارغة، والكأس
الفارغة، والعلبة الفارغة: كلها مملوءة هواءً حتى الحافة. أما
الفراغ الحقيقي --- ألّا يكون في الداخل شيء البتة، ولا حتى \omterm{def:g2:air-around-us:air}{هواء}
--- فتدبيره صعب جدًّا، وستلقى يومًا الاسم الغريب الذي يعطيه
العلماء له.
\end{remark}

\section{الهواء يدفع}

\begin{definition}[الريح]\label{def:g2:air-around-us:wind}
\emph{الريح}\index{ريح} \omterm{def:g2:air-around-us:air}{هواء} متحرّك. \omterm{def:g2:air-around-us:air}{فالهواء} الساكن لا يمسّ
أحدًا؛ أما \omterm{def:g2:air-around-us:air}{الهواء} المتحرّك فيدفع كل ما يقف في طريقه --- برفق في
يوم نسيم، وبقوة تحني الأشجار في العاصفة.
\end{definition}

\begin{example}[الهواء في عمله]\label{ex:g2:air-around-us:work}
\omterm{def:g2:air-around-us:air}{الهواء} المتحرّك يؤدّي أعمالًا حقيقية: يملأ أشرعة القوارب، ويدير
لُعَب \omterm{def:g2:air-around-us:wind}{الريح}، ويجفّف الغسيل، ويحمل الطائرات الورقية. وأنت تصنع
ريحًا بنفسك كلما أطفأت شمعة عيد ميلاد --- فنَفَسك هبّة صغيرة
خاصة بك، تكفي قوّتها لدفع اللهب عن فتيلته.
\end{example}

\begin{example}[جرّب: بالون نفّاث]\label{ex:g2:air-around-us:balloon}
انفخ بالونًا وأمسك عنقه مغلقًا. في الداخل \omterm{def:g2:air-around-us:air}{هواء} مضغوط ينتظر.
أفلته: يندفع \omterm{def:g2:air-around-us:air}{الهواء} خارجًا إلى الخلف، وينطلق البالون إلى الأمام
عبر الغرفة يترنّح ترنّحًا شديدًا. \omterm{def:g2:air-around-us:air}{الهواء} الهارب يدفع البالون ---
وهو ابن عمّ صغير للطريقة التي يدفع بها الغاز المضغوط الصواريخ
الحقيقية.
\end{example}

\begin{remark}[الهواء ثمين]\label{rem:g2:air-around-us:breathe}
أهمّ دفعة على الإطلاق تحدث في صدرك بهدوء: فمع كل نَفَس يدخل
\omterm{def:g2:air-around-us:air}{الهواء}، ويحتفظ جسمك بقليل ممّا يحتاج إليه، ويخرج الباقي. والناس
والقطط والطيور، بل حتى السمك (الذي يرتشف \omterm{def:g2:air-around-us:air}{الهواء} المختبئ في
الماء)، كلها تحتاج إلى نصيبها. فالخفيّ لا يعني غير المهمّ ---
بل يعني في \omterm{def:g2:air-around-us:air}{الهواء} عكس ذلك.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:1}
اذكر ثلاث قرائن رأيتها هذا الأسبوع على أن \omterm{def:g2:air-around-us:air}{الهواء} حقيقي --- من
غير أن تستعمل أي تجربة من كتاب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:2}
لماذا يدفع الكيسُ المفتول في \cref{ex:g2:air-around-us:catch} في
وجهك حين تعصره؟ وما الذي في داخله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:3}
في غطسة المنديل الجافّ، لماذا لا يصل الماء إلى المنديل؟ قل ذلك
بكلمات الفصل: ما الذي يحتفظ بحيّزه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:4}
ما الذي يجب \emph{ألّا} تفعله بالكأس أثناء غطسة المنديل الجافّ،
وماذا يحدث للمنديل لو فعلته؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:5}
ما قرقرة قنّينة تمتلئ تحت الماء؟ ما الذي يخرج، وما الذي يدخل،
ولماذا يجب أن يتناوبا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:6}
``ناولني القنّينة الفارغة.'' ما الذي في القنّينة الفارغة حقًّا؟
(انظر \cref{rem:g2:air-around-us:noempty}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:7}
ما \omterm{def:g2:air-around-us:wind}{الريح}، بكلمات \cref{def:g2:air-around-us:wind}؟ واذكر عملين
يؤدّيهما \omterm{def:g2:air-around-us:air}{الهواء} المتحرّك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:air-around-us:8}
حين يطير البالون النفّاث، في أيّ جهة يندفع \omterm{def:g2:air-around-us:air}{الهواء} خارجًا، وفي
أيّ جهة يذهب البالون؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:air-around-us:9}
قُمع محكم في عنق قنّينة، مسدود من كل جانب بمعجون لاصق. تصبّ آنا
الماء في القُمع --- وبعد جرعة صغيرة يستقرّ الماء في القُمع
رافضًا أن ينزل إلى القنّينة. اشرح ما الذي يسدّ عليه الطريق،
واقترح الحلّ الذي لا يستغرق إلا ثانية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:air-around-us:10}
تحت الماء في حوض الحمّام، أمِل كأسًا مملوءة هواءً حتى تصير فوهتها
مقابلة لفوهة كأس مملوءة ماءً، وكلتاهما تحت الماء. أتستطيع أن
تصبّ \omterm{def:g2:air-around-us:air}{الهواء} \emph{إلى الأعلى} من إحدى الكأسين إلى الأخرى؟ صِف
ما تراه، ولماذا تصعد الفقاعات بينما ينزل الماء المصبوب.
\end{exercise}
