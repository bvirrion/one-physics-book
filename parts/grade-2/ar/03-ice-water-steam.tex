\chapter{ثلج وماء وبخار}\label{ch:g2:ice-water-steam}

الماء نفسه الذي يقرقع في كأسك مكعّبَ ثلج يُسكب شرابًا ويرتفع من
قِدر الحساء عمودًا أبيض. مادة واحدة وثلاث \emph{حالات}. في السنة
الماضية رأينا الدفء يصهر مكعّب ثلج؛ وهذه السنة نلتقي حالات المادة
الثلاث كلها التي يستطيع الماء أن يتّخذها.

\section{حالات الماء الثلاث}

\begin{example}[أين تسكن كل حالة]\label{ex:g2:ice-water-steam:where}
يستطيع الماء أن يكون في ثلاث \emph{حالات للمادة}. \emph{الصلبة}:
الثلج --- مكعّبات الثلج، والدلايات تحت السقف، والصقيع على العشب،
والثلج المتساقط. والجسم الصلب قاسٍ يحفظ شكله. \emph{السائلة}:
الماء الذي يُسكب --- المطر، والأنهار، والبحر، وكأسك. والسائل يجري
ويأخذ شكل إنائه. \emph{الغازية}: بخار الماء، وهو ماء يطير في
الهواء من غير أن يُرى. صلبة، فسائلة، فغازية --- والمادة تحتها
واحدة دائمًا.
\end{example}

\begin{remark}[بخار الماء لا يُرى]\label{rem:g2:ice-water-steam:invisible}
العمود الأبيض الذي تراه فوق القِدر مؤلَّف من قطيرات ماء دقيقة
عائمة --- أي قطرات صغيرة \emph{سائلة}. أما الغاز الحقيقي --- بخار
الماء الممتزج بالهواء --- فلا يُرى البتة! انظر عن قرب إلى فوهة
القِدر: فوقها مباشرة فجوة \emph{صافية} قصيرة قبل أن يبدأ العمود
الأبيض. في تلك الفجوة يكون الماء غازًا لا يُرى؛ وأعلى منها بقليل
يتجمّع قطيرات تستطيع أن تراها. وبخار الماء الخفيّ مختبئ في هواء
كل غرفة، وسنمسكه بالجرم المشهود قبل نهاية الفصل.
\end{remark}

\section{تغيّر الحالة}

\begin{definition}[الانصهار والتجمّد]\label{def:g2:ice-water-steam:meltfreeze}
حين يحوّل الدفء الثلج الصلب ماءً سائلًا يقال إن الثلج
\emph{ينصهر}\index{انصهار}. وحين يحوّل البردُ الشديد الماءَ
السائل صلبًا يقال إن الماء \emph{يتجمّد}\index{تجمّد}. والانصهار
والتجمّد بابان متقابلان بين الحالتين نفسيهما: الدفء يدفع في جهة،
والبرد يدفع في الجهة الأخرى.
\end{definition}

\begin{definition}[التبخّر والتكاثف]\label{def:g2:ice-water-steam:evapcond}
حين يجعل الدفءُ الماءَ السائل يطير في الهواء غازًا يقال إن الماء
\emph{يتبخّر}\index{تبخّر} --- فتصغر البركة، ويجفّ الغسيل. وحين
يلمس بخار الماء شيئًا باردًا فيعود قطيرات سائلة يقال إنه
\emph{يتكاثف}\index{تكاثف} --- ضباب على النافذة الباردة، وغَبَش
على مرآة الحمّام. وهما بابان متقابلان مرة أخرى: الدفء يرسل الماء
إلى الحالة الغازية، والبرد يردّه إلى السائلة.
\end{definition}

\begin{example}[جرّب: أمسك الماء الخفيّ]\label{ex:g2:ice-water-steam:catch}
خذ علبة أو كأسًا من الثلاجة توًّا، وجفّف ظاهرها بعناية بمنشفة،
وضعها على الطاولة. وانتظر وأنت تعدّ حتى $100$. يغبشّ الظاهر ثم
يبتلّ --- وتظهر قطرات من العدم! وهي لا ترشح من خلال الزجاج: بل
بخار الماء الخفيّ في هواء الغرفة لمس الجدار البارد \omterm{def:g2:ice-water-steam:evapcond}{فتكاثف}. لقد
أمسكت الغاز الخفيّ توًّا.
\end{example}

\begin{example}[سحب النَّفَس]\label{ex:g2:ice-water-steam:breath}
نَفَسك يحمل دائمًا بخار ماء لا يُرى. وفي يوم دافئ يبقى غير مرئي.
أما في صباح متجمّد فهواء الخارج البارد يكاثفه في الحال سحابة
صغيرة أمام فمك --- ومثل ذلك يحدث على نافذة باردة حين تنفخ عليها
لترسم بإصبعك.
\end{example}

\begin{remark}[البخار والمواقد: احذر]\label{rem:g2:ice-water-steam:safety}
بخار الماء فوق القِدر الغالي حارق --- أشدّ حرارة مما يبدو عليه
الماء نفسه. فلا تضع يدك في العمود أبدًا، وارفع الأغطية بعيدًا عن
وجهك، وليكن شخص كبير بجوارك. تغيّرات الحالة تُشاهَد ولا تُلمَس
كلما كان الموقد في الأمر.
\end{remark}

\section{الماء نفسه، دورةً بعد دورة}

\begin{example}[البركة المتقلّصة]\label{ex:g2:ice-water-steam:puddle}
بعد المطر تلمع بركة في ساحة اللعب. وبعد الظهر تكون قد صغرت؛ وفي
الغد تختفي. لم يشربها أحد ولم ترشح من الإسفلت: بل جعلها دفء الشمس
تتبخّر قطرةً قطرة في الهواء. ارسم بالطباشير خطًّا حول بركة صباحًا
وتحقّق بعد الغداء: يكون الماء قد انسحب إلى داخل خطّك.
\end{example}

\begin{example}[من البحر إلى المطر]\label{ex:g2:ice-water-steam:cycle}
ما \omterm{def:g2:ice-water-steam:evapcond}{تبخّر} لا بدّ أن \omterm{def:g2:ice-water-steam:evapcond}{يتكاثف} في مكان ما. يطير الماء صاعدًا من البحار
والبِرَك من غير أن يُرى؛ وعاليًا في السماء، حيث الهواء بارد،
\omterm{def:g2:ice-water-steam:evapcond}{يتكاثف} قطيرات تصنع الغيوم؛ ثم تتجمّع القطيرات وتثقل وتسقط ---
مطرًا! والمطر يملأ البِرَك والبحار، فتبدأ الرحلة الدائرية من
جديد. والماء الذي في كأسك قطع هذه الرحلة مرات لا يحصيها أحد.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:1}
ما حالة المادة في كل ممّا يلي --- صلبة أم سائلة أم غازية:
دلاية ثلج؛ \ البحر؛ \ الماء الخفيّ فوق فوهة قِدر الحساء مباشرة؛
\ الصقيع على العشب؛ \ المطر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:2}
سمِّ تغيّر الحالة: رجل ثلج يتقلّص في الشمس؛ \ بركة تختفي؛
\ غبش يظهر على مرآة الحمّام؛ \ غدير يقسو في الشتاء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:3}
في صورة تغيّرات الحالة، إلى أيّ جهة يدفع الدفء دائمًا: نحو
الحالة الصلبة أم نحو الغازية؟ وإلى أين يدفع البرد؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:4}
القطرات على العلبة الباردة في \cref{ex:g2:ice-water-steam:catch}
--- أرشحت من خلال الزجاج؟ ومن أين جاءت؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:5}
لماذا ترى نَفَسك في صباح متجمّد ولا تراه في يوم دافئ؟ استعمل
كلمات \cref{def:g2:ice-water-steam:evapcond}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:6}
يجفّ الغسيل المبلول في الشمس أسرع منه في \omterm{def:g1:light-and-shadows:shadow}{الظلّ}. أيّ تغيّر حالة
يعمل هنا، وما الذي تضيفه الشمس ليسرّعه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:7}
لماذا يكون العمود فوق قِدر غالٍ خطِرًا، وما القاعدة الآمنة في
\cref{rem:g2:ice-water-steam:safety}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:ice-water-steam:8}
احكِ الرحلة الدائرية لقطرة مطر: ابدأ من البحر وانتهِ إلى البحر،
واستعمل كلمتَي \omterm{def:g2:ice-water-steam:evapcond}{التبخّر} \omterm{def:g2:ice-water-steam:evapcond}{والتكاثف} في الطريق.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:ice-water-steam:9}
عند فوهة القِدر فجوة صافية، ولا يبدأ العمود الأبيض إلا فوقها.
ما الذي في الفجوة الصافية؟ وماذا يبيّن هذا عن الماء في حالته
الغازية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:ice-water-steam:10}
يترك نينو كأس ماء على حافة النافذة أسبوعًا؛ فينزل المستوى يومًا
بعد يوم، مع أن أحدًا لا يشربه. فيُتَّهم قطّه. دافع عن القطّ:
اشرح ما حدث حقًّا، وابتكر طريقة تبيّن أن الماء يذهب \emph{إلى
الهواء}، لا إلى القطّ.
\end{exercise}
