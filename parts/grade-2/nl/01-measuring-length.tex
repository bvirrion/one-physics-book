\chapter{Lengte meten}\label{ch:g2:measuring-length}

Twee deuren, in twee verschillende huizen: welke is hoger? Je kunt een
deur niet door de stad dragen om hem naast de andere te zetten. Vorig
jaar vergeleken we dingen naast elkaar; dit jaar leren we de grootste
truc van de wetenschap --- een lengte veranderen in een \emph{getal}
dat kan reizen.

\section{Vergelijken zonder getallen}

\begin{example}[Naast elkaar]\label{ex:g2:measuring-length:sidebyside}
Om de langste van twee potloden te vinden, zet je ze samen rechtop op
tafel: degene die uitsteekt wint. Om je lengte met die van een vriend
te vergelijken, ga je rug aan rug staan. Dat werkt prachtig --- zolang
je de twee dingen bij elkaar kunt brengen.
\end{example}

\begin{example}[De touwtruc]\label{ex:g2:measuring-length:string}
De twee deuren kunnen elkaar niet ontmoeten --- maar een touwtje kan
ze allebei bezoeken. Span een touwtje langs de eerste deur en knip het
af op precies die hoogte. Neem het touwtje mee naar de tweede deur: is
het touwtje te kort, dan is de tweede deur hoger. Het touwtje draagt de
hoogte van de eerste deur door de stad. Nog één stap beter: en als we
de hoogte nu eens in een \emph{brief} konden versturen? Daarvoor hebben
we getallen nodig.
\end{example}

\section{Meten met stappen en handbreedtes}

\begin{definition}[Een lengte meten]\label{def:g2:measuring-length:measure}
Een lengte \emph{meten}\index{meten} is tellen hoe vaak een gekozen
\emph{eenheid}\index{eenheid} --- een stap, een handspan, een stokje
--- erlangs past, kop aan staart, zonder gaten en zonder overlap. Het
antwoord is een aantal eenheden: ``de kamer is $12$ stappen lang.''
\end{definition}

\begin{example}[Probeer het: stap de kamer af]\label{ex:g2:measuring-length:pace}
Steek je slaapkamer over met je hakken tegen je tenen en tel je
voetlengtes. Vraag een volwassene hetzelfde te doen. Jij telt
misschien $18$ voeten en de volwassene maar $12$ --- voor
\emph{dezelfde kamer}! Niemand heeft verkeerd geteld: jouw voeten zijn
korter, dus passen er meer. Een meting met \emph{mijn} voet valt niet
na te kijken met \emph{jouw} voet.
\end{example}

\begin{remark}[De ruzie op de markt]\label{rem:g2:measuring-length:argument}
Lange tijd maten mensen echt in voeten, duimen en armlengtes, en op de
markten klonken de ruzies: wiens voet? die van de lange koopman of die
van de kleine klant? De oplossing, waar de hele wereld het over eens
werd, was om \emph{één} \omterm{def:g2:measuring-length:measure}{eenheid} voor iedereen te kiezen --- een
\omterm{def:g2:measuring-length:measure}{eenheid} die van niemands lichaam is.
\end{remark}

\section{De centimeter en de meter}

\begin{definition}[Centimeter en meter]\label{def:g2:measuring-length:units}
De \emph{centimeter}\index{centimeter} (geschreven \unit{cm}) is een
kleine \omterm{def:g2:measuring-length:measure}{eenheid} van lengte, ongeveer de breedte van een nagel; hij
staat op elke liniaal gedrukt. De \emph{meter}\index{meter}
(geschreven \unit{m}) is een grote \omterm{def:g2:measuring-length:measure}{eenheid}, precies $100$ centimeter,
de lengte van een heel grote stap. Die eenheden zijn voor iedereen op
Aarde hetzelfde: jouw \qty{20}{cm} en mijn \qty{20}{cm} zijn precies
gelijk.
\end{definition}

\begin{method}[Meten met een liniaal]\label{met:g2:measuring-length:ruler}
\begin{enumerate}
  \item Leg de liniaal tegen het voorwerp aan;
  \item zet het \emph{nulstreepje} van de liniaal precies bij het ene
        uiteinde van het voorwerp --- niet de rand van de liniaal,
        maar de $0$;
  \item lees het getal af bij het andere uiteinde: dat is de lengte in
        \omterm{def:g2:measuring-length:units}{centimeters}.
\end{enumerate}
Valt het uiteinde tussen twee streepjes, zeg dan ``tussen \qty{7}{cm}
en \qty{8}{cm}, dichter bij $8$'' --- eerlijke woorden zijn beter dan
een verzonnen getal.
\end{method}

\begin{example}[Welke eenheid voor welk werk?]\label{ex:g2:measuring-length:whichunit}
Kleine dingen als een kever, een postzegel of je hand meet je in
\omterm{def:g2:measuring-length:units}{centimeters}. Grote dingen als een gang, een bus of een zwembad meet
je in \omterm{def:g2:measuring-length:units}{meters}: het bad is \qty{25}{m} lang --- stel je voor dat je dat
in nagelbreedtes moest tellen! Een verstandige \omterm{def:g2:measuring-length:measure}{eenheid} kiezen is het
halve werk van het \omterm{def:g2:measuring-length:measure}{meten}.
\end{example}

\begin{example}[Eerst schatten, dan nakijken]\label{ex:g2:measuring-length:estimate}
Vóór het \omterm{def:g2:measuring-length:measure}{meten} gokken de spelers: hoe lang is de keukentafel? Papa
zegt \qty{2}{m}, Lina zegt \qty{1}{m} en een beetje. Dan beslist de
meterlat: \qty{1}{m} en bijna een half --- punt voor Lina. Eerst
gokken traint je oog; \omterm{def:g2:measuring-length:measure}{meten} houdt iedereen eerlijk.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:1}
Hoe vergelijk je je lengte met die van een vriend zonder liniaal? En
met de lengte van een vriend die in een andere stad woont?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:2}
Mara meet het klaslokaal in stappen en komt op $20$; haar juf komt op
$14$. Wie heeft verkeerd geteld? Leg uit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:3}
Wat is er bijzonder aan de \omterm{def:g2:measuring-length:units}{centimeter}, vergeleken met een voetlengte of
een handspan? Waarom werden mensen het erover eens?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:4}
Welke \omterm{def:g2:measuring-length:measure}{eenheid} past beter, \unit{cm} of \unit{m}: de lengte van een
mier; \ de hoogte van een deur; \ de lengte van het schoolplein; \ de
breedte van dit boek?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:5}
Tim zet de rand van zijn liniaal --- niet het nulstreepje --- tegen
het uiteinde van zijn potlood en leest \qty{13}{cm} af. Is zijn
potlood echt \qty{13}{cm} lang? Wat is hij vergeten uit
\cref{met:g2:measuring-length:ruler}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:6}
Meet met een liniaal: de breedte van je hand; \ de lengte van een
lepel; \ de hoogte van een beker. Schrijf elk antwoord met zijn
\omterm{def:g2:measuring-length:measure}{eenheid}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:7}
Een deur is \qty{2}{m} hoog. Hoeveel \omterm{def:g2:measuring-length:units}{centimeter} is dat? (Denk eraan:
\qty{1}{m} is $100$ \unit{cm}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:8}
Eerst gokken, dan \omterm{def:g2:measuring-length:measure}{meten}: hoeveel \omterm{def:g2:measuring-length:units}{meter} lang is je bed? Was je gok te
groot, te klein, of bijna goed?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:9}
Een lint is \qty{45}{cm} lang, een ander \qty{38}{cm}. Welk is langer,
en hoeveel \omterm{def:g2:measuring-length:units}{centimeter} scheelt het? Had je dat ook \emph{zonder}
getallen kunnen beslissen? Welke manier is makkelijker om in een brief
te zetten?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:10}
Opa zegt: ``Mijn tuin is $30$ passen lang.'' Leg opa vriendelijk uit
waarom ``$30$ passen'' door iemand anders niet valt na te kijken, en
wat hij in plaats daarvan moet gebruiken. Wat zou er gebeuren als een
klein kind dezelfde tuin afstapte?
\end{exercise}
