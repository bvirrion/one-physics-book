\chapter{De Maan aan de hemel}\label{ch:g2:moon-in-the-sky}

Sommige avonden hangt er een dunne zilveren glimlach boven de daken;
twee weken later overstroomt een volle ronde lantaarn de tuin met bleek
\omterm{def:g1:light-and-shadows:source}{licht}. Het is dezelfde \omterm{def:g2:moon-in-the-sky:moon}{Maan} --- onze naaste buur aan de hemel, en de
enige andere wereld waar mensen ooit hebben gelopen. Tijd om echt
kennis te maken.

\section{Onze buur aan de hemel}

\begin{definition}[De Maan]\label{def:g2:moon-in-the-sky:moon}
De \emph{Maan}\index{Maan} is een enorme bal van grijs gesteente die
om de Aarde reist, onze trouwe metgezel in de ruimte. Ze is veel
kleiner dan de Aarde en heel ver weg --- verder dan welk vliegtuig ook
vliegt, zo ver dat de astronauten die haar bezochten er drie dagen
over deden om er te komen.
\end{definition}

\begin{example}[Goed kijken]\label{ex:g2:moon-in-the-sky:closely}
Met het blote oog laat de \omterm{def:g2:moon-in-the-sky:moon}{Maan} bleke vlekken en heldere gebieden zien
--- sommige mensen zien er een gezicht in. Door een verrekijker (een
heerlijk proefje op een avond met \omterm{ex:g2:moon-in-the-sky:shapes}{volle maan}) worden de vlekken
scherper en zie je vlaktes en ronde \emph{kraters}, komvormige littekens
die lang geleden zijn achtergelaten door rotsen uit de ruimte. De
\omterm{def:g2:moon-in-the-sky:moon}{Maan} houdt haar littekens voor altijd: ze heeft geen \omterm{def:g2:air-around-us:wind}{wind} en geen
regen om ze uit te wissen --- ze heeft namelijk helemaal geen
\omterm{def:g2:air-around-us:air}{lucht}. Geen \omterm{def:g2:air-around-us:air}{lucht} betekent ook een stille wereld: op de \omterm{def:g2:moon-in-the-sky:moon}{Maan} maakt
zelfs een schreeuw geen geluid.
\end{example}

\begin{example}[De Maan die je volgt]\label{ex:g2:moon-in-the-sky:follows}
Loop op een maanverlichte avond door de straat: huizen en bomen
glijden voorbij, maar de \omterm{def:g2:moon-in-the-sky:moon}{Maan} lijkt je te volgen, altijd naast je
schouder. Ze zit je niet echt achterna --- ze is zo ver weg dat een
hele straat lopen jouw uitzicht op haar helemaal niet verandert,
terwijl de huizen vlakbij voortdurend veranderen. Heel verre dingen
lijken elke reiziger te volgen; de \omterm{def:g2:moon-in-the-sky:moon}{Maan} doet het het best, want zij is
kampioen ver weg.
\end{example}

\section{Weerkaatst licht}

\begin{example}[Maanlicht is weerkaatst zonlicht]\label{ex:g2:moon-in-the-sky:borrowed}
Is de \omterm{def:g2:moon-in-the-sky:moon}{Maan} een \omterm{def:g1:light-and-shadows:source}{lichtbron}? Denk aan de proef van vorig jaar: maakt ze
haar eigen \omterm{def:g1:light-and-shadows:source}{licht}? Nee --- de \omterm{def:g2:moon-in-the-sky:moon}{Maan} is een bal donker gesteente, als
een reusachtige kiezelsteen aan de hemel. Wat wij maanlicht noemen is
zonlicht dat door dat gesteente wordt \emph{weerkaatst} --- naar ons
teruggekaatst, zoals een witte muur een kamer verlicht zonder een lamp
te zijn. De Zon verlicht de \omterm{def:g2:moon-in-the-sky:moon}{Maan} helder aan één kant; wij zien het deel
van die heldere kant dat naar ons toe gekeerd is.
\end{example}

\begin{remark}[Een Maan bij daglicht]\label{rem:g2:moon-in-the-sky:daytime}
Veel mensen denken dat de \omterm{def:g2:moon-in-the-sky:moon}{Maan} een wezen van de nacht is --- maar kijk
op sommige middagen omhoog en daar is ze, een bleek spookje in de
blauwe \omterm{def:g2:air-around-us:air}{lucht}! De \omterm{def:g2:moon-in-the-sky:moon}{Maan} reist volgens haar eigen rooster om de Aarde en
trekt zich er niets van aan of het bij ons dag of nacht is. Ze laat
zich overdag alleen minder trots zien, wanneer de felheid van de Zon de
hele hemel vult.
\end{remark}

\section{De Maan verandert van vorm}

\begin{example}[De wisselende vormen van de Maan]\label{ex:g2:moon-in-the-sky:shapes}
Kijk vele avonden naar de \omterm{def:g2:moon-in-the-sky:moon}{Maan} en ze verandert van vorm: een dunne
\emph{maansikkel}\index{maansikkel} als een afgeknipt nageltje, dan
een halve cirkel, dan een dikke bijna ronde vorm, dan de perfecte
schijf van de \emph{volle maan}\index{volle maan} --- en dan weer
terug, naar een \omterm{ex:g2:moon-in-the-sky:shapes}{maansikkel} die de andere kant op wijst, tot de
\omterm{def:g2:moon-in-the-sky:moon}{Maan} een nacht of twee helemaal lijkt te verdwijnen. De hele
voorstelling duurt ongeveer een maand, en dan herhaalt ze zich, net zo
betrouwbaar als dag en nacht.
\end{example}

\begin{method}[Een maandagboek bijhouden]\label{met:g2:moon-in-the-sky:diary}
\begin{enumerate}
  \item Kijk elke heldere avond, rond hetzelfde uur, waar de
        \omterm{def:g2:moon-in-the-sky:moon}{Maan} staat;
  \item teken de vorm die je ziet in een klein kalenderhokje --- een
        cirkel, en arceer het donkere deel;
  \item verstoppen wolken haar, schrijf dan ``bewolkt'' en ga door;
  \item leg je tekeningen na twee weken op een rij.
\end{enumerate}
Je eigen tekeningen laten zien dat de vorm verandert, elke nacht een
beetje meer. \emph{Waarom} de \omterm{def:g2:moon-in-the-sky:moon}{Maan} dat doet is een heerlijk raadsel ---
het wordt in een later jaar van deze cursus netjes opgelost, en jouw
dagboek is de eerste stap van het bewijs.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:1}
Is de \omterm{def:g2:moon-in-the-sky:moon}{Maan} groter of kleiner dan de Aarde? Dichterbij of verder weg
dan de wolken? Waaraan weten we dat mensen haar kunnen bezoeken?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:2}
Hoe heten de ronde komvormige plekken op de \omterm{def:g2:moon-in-the-sky:moon}{Maan}, en waardoor zijn ze
ontstaan? Waarom zijn ze er na al die tijd nog steeds?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:3}
Is de \omterm{def:g2:moon-in-the-sky:moon}{Maan} een \omterm{def:g1:light-and-shadows:source}{lichtbron}, zoals een lamp, of een lichtvanger, zoals
een witte muur? Waar komt maanlicht werkelijk vandaan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:4}
Welke helft van de \omterm{def:g2:moon-in-the-sky:moon}{Maan} verlicht de Zon: altijd de helft die naar de
Zon toe gekeerd is, altijd dezelfde helft, of elke keer een andere
helft?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:5}
Kan de \omterm{def:g2:moon-in-the-sky:moon}{Maan} overdag te zien zijn? Als een vriend nee zegt, waarmee
overtuig je hem dan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:6}
Noem de vormen van de \omterm{def:g2:moon-in-the-sky:moon}{Maan} op volgorde, te beginnen bij het afgeknipte
nageltje en eindigend bij de ronde lantaarn. Hoe lang duurt de hele
voorstelling ongeveer voordat ze zich herhaalt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:7}
Op de \omterm{def:g2:moon-in-the-sky:moon}{Maan} is geen \omterm{def:g2:air-around-us:air}{lucht}. Wat zou daar met een schreeuw gebeuren? En
met een vlieger?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:8}
Begin vanavond met het maandagboek van
\cref{met:g2:moon-in-the-sky:diary}. Welke twee dingen moeten elke
avond hetzelfde blijven om het dagboek eerlijk te maken?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:moon-in-the-sky:9}
``De \omterm{def:g2:moon-in-the-sky:moon}{Maan} is onze auto helemaal naar huis gevolgd!'' zegt kleine Rosa.
Leg haar met de straatwandeling van
\cref{ex:g2:moon-in-the-sky:follows} uit waarom heel verre dingen
lijken mee te gaan, terwijl dingen dichtbij voorbijglijden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:moon-in-the-sky:10}
Kijk op een avond met een \omterm{ex:g2:moon-in-the-sky:shapes}{maansikkel} goed: soms gloeit het
\emph{donkere} deel van de \omterm{def:g2:moon-in-the-sky:moon}{Maan} heel zwakjes, zodat je de hele cirkel
net kunt raden. De Zon schijnt niet op dat deel --- maar iets anders
schijnt wel in de buurt. Welke grote, heldere, door de zon beschenen
bal zou de donkere kant van de \omterm{def:g2:moon-in-the-sky:moon}{Maan} kunnen verlichten, zoals de
\omterm{def:g2:moon-in-the-sky:moon}{Maan} onze nachten verlicht? (Tip: je staat erop.)
\end{exercise}
