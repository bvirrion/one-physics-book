\chapter{Medir el tiempo}\label{ch:g2:measuring-time}

``¡Solo cinco minutos más!'' --- pero ¿cuánto es eso en realidad? El
tiempo no se ve, no se coge y no se puede tumbar a lo largo de una
regla. Y aun así lo medimos todos los días: las carreras tienen
ganador, los bizcochos salen del horno ni crudos ni quemados y el
colegio acaba en el momento justo. Vamos a aprender cómo se domestica
el tiempo.

\section{Instantes y duraciones}

\begin{example}[El orden del día]\label{ex:g2:measuring-time:order}
Un día es un desfile de acontecimientos, siempre en orden:
levantarse, desayunar, colegio, comida, tarde, cena, cama. ``El
desayuno viene \emph{antes} del colegio'' y ``la cena viene
\emph{después} de la comida'' --- antes y después son las primeras
palabras del tiempo, igual que más largo y más corto fueron las
primeras palabras de la longitud.
\end{example}

\begin{definition}[La duración]\label{def:g2:measuring-time:duration}
Un \emph{instante}\index{instante} es un punto del día: el instante en
que suena el timbre. Una \emph{duración}\index{duración} es el trecho
de tiempo que hay entre dos instantes: desde el timbre hasta el final
de la clase. Los instantes responden a \emph{¿cuándo?}; las
duraciones, a \emph{¿cuánto dura?}
\end{definition}

\begin{example}[¿Cuándo o cuánto dura?]\label{ex:g2:measuring-time:whenhowlong}
``La película empieza después de la cena'' habla de un instante. ``La
película dura lo mismo que dos recreos de la comida'' habla de una
\omterm{def:g2:measuring-time:duration}{duración}. ``Mi cumpleaños es en primavera'': instante. ``Las
vacaciones se han hecho cortísimas'': \omterm{def:g2:measuring-time:duration}{duración} --- y este capítulo
trata sobre todo de duraciones.
\end{example}

\section{Comparar duraciones}

\begin{method}[Una carrera justa de duraciones]\label{met:g2:measuring-time:race}
Para saber cuál de dos cosas dura más:
\begin{enumerate}
  \item empieza las dos en el \emph{mismo instante} --- ``¡preparados,
        listos, ya!'';
  \item mira cuál acaba antes;
  \item la que sigue en marcha cuando la otra ya ha terminado es la
        que dura más.
\end{enumerate}
Si no pueden correr a la vez (el baño de hoy y la cena de ayer), haz
correr cada una contra la \emph{misma} tercera cosa: un reloj de
arena.
\end{method}

\begin{example}[El reloj de arena]\label{ex:g2:measuring-time:sandtimer}
Un reloj de arena es un vidrio pequeño con la cintura estrecha: la
arena tarda siempre la \emph{misma} \omterm{def:g2:measuring-time:duration}{duración} en caer del todo. Eso
lo convierte en árbitro. Lavarse los dientes contra el reloj: la arena
se acabó antes --- ¡a cepillarse más rato! Una canción contra el
reloj: ganó la canción. Así que la canción dura más que el cepillado
de dientes, aunque nunca hayan corrido la una contra el otro.
\end{example}

\begin{remark}[Las sensaciones son malos relojes]\label{rem:g2:measuring-time:feelings}
Una tarde divertida vuela; una cola aburrida se arrastra --- y sin
embargo el reloj puede contar lo mismo para las dos. Nuestra sensación
del tiempo se engaña con facilidad, igual que se engañaba el tacto con
lo caliente y lo frío. Justo por eso importan árbitros como el reloj
de arena: a la arena no le entra el aburrimiento.
\end{remark}

\section{Contar golpes regulares}

\begin{definition}[El segundo]\label{def:g2:measuring-time:second}
El \emph{segundo}\index{segundo} es la \omterm{def:g2:measuring-length:measure}{unidad} de tiempo que se usa en
el mundo entero y se escribe \unit{s}. Un segundo dura más o menos lo
que un latido tranquilo del corazón, o el tictac regular de un reloj
grande. Decir ``un cocodrilo, dos cocodrilos'' a un ritmo tranquilo
cuenta aproximadamente un segundo por cocodrilo.
\end{definition}

\begin{example}[Pruébalo: ¿cuántos segundos?]\label{ex:g2:measuring-time:count}
Cuenta segundos a ritmo regular mientras: aguantas la respiración sin
agobiarte; el grifo llena un vaso; un amigo corre hasta la valla y
vuelve. El vaso puede tardar $6$ segundos y la carrera $20$. Ahora el
cepillado de dientes y la canción de antes pueden recibir
\emph{números} --- y los números, como con las longitudes, pueden
viajar y se pueden comprobar.
\end{example}

\begin{example}[Las horas del reloj]\label{ex:g2:measuring-time:clock}
Los segundos sirven para las cosas cortas. Los trechos largos se
cuentan en \emph{horas}\index{hora}: las mañanas de colegio duran
\omterm{ex:g2:measuring-time:clock}{horas}, y una noche de sueño también. La aguja corta del reloj
señala la hora: justo en el $3$, son las tres; entre el $3$ y el $4$,
con la aguja larga señalando hacia abajo, son las tres y media. Un día
y una noche juntos suman $24$ \omterm{ex:g2:measuring-time:clock}{horas} --- mientras la Tierra, ya te
acuerdas, da una vuelta entera.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:1}
Ordena: la cena; \ levantarse; \ el rato de juego de la tarde; \ el
desayuno; \ la cama.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:2}
¿Instante o \omterm{def:g2:measuring-time:duration}{duración}? ``El timbre suena a mediodía''; \ ``el
bizcocho se hornea durante media hora''; \ ``nos fuimos después de
comer''; \ ``el viaje se hizo interminable''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:3}
Se encienden dos velas en el mismo instante; la roja se apaga mientras
la azul sigue ardiendo. ¿Qué vela ha durado más? ¿Qué regla del
\cref{met:g2:measuring-time:race} han seguido las velas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:4}
¿Por qué es un reloj de arena un árbitro justo para las duraciones?
¿Qué es siempre igual en él, cada vez que se le da la vuelta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:5}
Ben dice que su aburrida espera en el dentista duró ``una eternidad''.
Su reloj dice que duró menos que sus dibujos animados favoritos. ¿Cómo
es posible? (Mira la \cref{rem:g2:measuring-time:feelings}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:6}
Cuenta segundos a ritmo regular y mide: ¿cuántos segundos tardas en
escribir tu nombre completo? ¿Y en ponerte los zapatos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:7}
¿Qué \omterm{def:g2:measuring-length:measure}{unidad} viene mejor, segundos u \omterm{ex:g2:measuring-time:clock}{horas}: un estornudo; \ una
noche de sueño; \ atarse un cordón; \ una mañana de colegio?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:8}
Dibuja un reloj que marque las cinco en punto y otro que marque las
cinco y media. ¿Adónde señala la aguja corta a las cinco y media?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-time:9}
El baño de ayer y la ducha de hoy no pueden correr el uno contra la
otra. Explica, paso a paso, cómo averiguar cuál dura más usando un
solo reloj de arena --- y qué hay que anotar para que tu amigo lo
pueda comprobar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-time:10}
Lila cuenta ``uno, dos, tres\dots'' muy deprisa y dice que el vaso se
llenó en $12$ segundos; Tom cuenta con calma y le salen $6$. El grifo
corrió igual las dos veces. ¿Quién está más cerca de la verdad y qué
hace honrado un recuento de segundos?
\end{exercise}
