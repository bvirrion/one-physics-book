\chapter{La Luna en el cielo}\label{ch:g2:moon-in-the-sky}

Algunas tardes cuelga sobre los tejados una fina sonrisa plateada; dos
semanas más tarde, un farol redondo y lleno inunda el jardín de una
\omterm{def:g1:light-and-shadows:source}{luz} pálida. Es la misma \omterm{def:g2:moon-in-the-sky:moon}{Luna} --- nuestra vecina más cercana en el
cielo y el único otro mundo que han pisado las personas. Va siendo
hora de presentarse como es debido.

\section{Nuestra vecina del cielo}

\begin{definition}[La Luna]\label{def:g2:moon-in-the-sky:moon}
La \emph{Luna}\index{Luna} es una bola enorme de roca gris que viaja
alrededor de la Tierra, nuestra fiel compañera en el espacio. Es
mucho más pequeña que la Tierra y está muy lejos --- más lejos de lo
que vuela ningún avión, tan lejos que los astronautas que la
visitaron viajaron tres días para llegar.
\end{definition}

\begin{example}[Mirar de cerca]\label{ex:g2:moon-in-the-sky:closely}
A simple vista, la \omterm{def:g2:moon-in-the-sky:moon}{Luna} enseña manchas pálidas y zonas brillantes ---
hay quien ve una cara en ellas. Con unos prismáticos (una cosa
preciosa que probar una noche de \omterm{ex:g2:moon-in-the-sky:shapes}{luna llena}) las manchas se afilan
hasta convertirse en llanuras y \emph{cráteres} redondos, cicatrices
en forma de cuenco que dejaron rocas del espacio al estrellarse allí
hace muchísimo tiempo. La \omterm{def:g2:moon-in-the-sky:moon}{Luna} conserva sus cicatrices para siempre:
no tiene \omterm{def:g2:air-around-us:wind}{viento} ni lluvia que las borren --- de hecho, no tiene
\omterm{def:g2:air-around-us:air}{aire} en absoluto. No tener \omterm{def:g2:air-around-us:air}{aire} significa además un mundo
silencioso: en la \omterm{def:g2:moon-in-the-sky:moon}{Luna}, ni siquiera un grito hace ruido.
\end{example}

\begin{example}[La Luna que te sigue]\label{ex:g2:moon-in-the-sky:follows}
Camina por la calle una tarde de luna: las casas y los árboles se
quedan atrás, pero la \omterm{def:g2:moon-in-the-sky:moon}{Luna} parece seguirte, siempre a tu lado. En
realidad no te persigue --- está tan lejos que recorrer una calle
entera no cambia nada tu manera de verla, mientras que las casas
cercanas cambian todo el rato. Las cosas muy lejanas parecen seguir a
todos los viajeros; la \omterm{def:g2:moon-in-the-sky:moon}{Luna} lo hace mejor que ninguna porque es la
campeona de la lejanía.
\end{example}

\section{Luz reflejada}

\begin{example}[La luz de la Luna es luz del Sol reflejada]\label{ex:g2:moon-in-the-sky:borrowed}
¿Es la \omterm{def:g2:moon-in-the-sky:moon}{Luna} una \omterm{def:g1:light-and-shadows:source}{fuente de luz}? Acuérdate de la prueba del año
pasado: ¿fabrica su propia \omterm{def:g1:light-and-shadows:source}{luz}? No la fabrica --- la \omterm{def:g2:moon-in-the-sky:moon}{Luna} es una
bola de roca oscura, como un guijarro enorme en el cielo. Lo que
llamamos \omterm{def:g1:light-and-shadows:source}{luz} de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} es \omterm{def:g1:light-and-shadows:source}{luz} del Sol \emph{reflejada} por esa roca
--- devuelta hacia nosotros, igual que una pared blanca ilumina una
habitación sin ser una lámpara. El Sol ilumina la \omterm{def:g2:moon-in-the-sky:moon}{Luna} por un lado;
nosotros vemos la parte de ese lado iluminado que nos mira.
\end{example}

\begin{remark}[Una Luna de día]\label{rem:g2:moon-in-the-sky:daytime}
Mucha gente cree que la \omterm{def:g2:moon-in-the-sky:moon}{Luna} es una criatura de la noche --- ¡pero
mira hacia arriba algunas tardes y ahí está, un fantasma pálido en el
cielo azul! La \omterm{def:g2:moon-in-the-sky:moon}{Luna} viaja alrededor de la Tierra con su propio
horario y le da igual que para nosotros sea de día o de noche.
Sencillamente se luce menos de día, cuando el resplandor del Sol llena
el cielo.
\end{remark}

\section{La Luna cambia de forma}

\begin{example}[Las formas cambiantes de la Luna]\label{ex:g2:moon-in-the-sky:shapes}
Mira la \omterm{def:g2:moon-in-the-sky:moon}{Luna} muchas tardes seguidas y verás que cambia de forma: un
fino \emph{creciente}\index{creciente} como el recorte de una uña,
después medio círculo, después una forma gorda y casi redonda, después
el disco perfecto de la \emph{luna llena}\index{luna llena} --- y
después de vuelta hacia abajo, hasta un creciente vuelto del otro
lado, hasta que durante una noche o dos la \omterm{def:g2:moon-in-the-sky:moon}{Luna} parece desaparecer
del todo. El espectáculo entero dura más o menos un mes y luego se
repite, tan puntual como el día y la noche.
\end{example}

\begin{method}[Llevar un diario de la Luna]\label{met:g2:moon-in-the-sky:diary}
\begin{enumerate}
  \item cada tarde despejada, más o menos a la misma hora, busca la
        \omterm{def:g2:moon-in-the-sky:moon}{Luna};
  \item dibuja la forma que ves en una casilla pequeña de un
        calendario --- un círculo, y sombrea la parte oscura;
  \item si la tapan las nubes, escribe ``nubes'' y sigue adelante;
  \item al cabo de dos semanas, pon tus dibujos en fila.
\end{enumerate}
Tus propios dibujos enseñarán la forma cambiando un poco más cada
noche. \emph{Por qué} hace esto la \omterm{def:g2:moon-in-the-sky:moon}{Luna} es un enigma delicioso ---
quedará bien resuelto en un año posterior de este curso, y tu diario
es el primer paso de la demostración.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:1}
¿Es la \omterm{def:g2:moon-in-the-sky:moon}{Luna} más grande o más pequeña que la Tierra? ¿Está más cerca o
más lejos que las nubes? ¿Cómo sabemos que las personas la pueden
visitar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:2}
¿Cómo se llaman las marcas redondas en forma de cuenco de la
\omterm{def:g2:moon-in-the-sky:moon}{Luna} y qué las hizo? ¿Por qué siguen ahí después de tantísimo
tiempo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:3}
¿Es la \omterm{def:g2:moon-in-the-sky:moon}{Luna} una \omterm{def:g1:light-and-shadows:source}{fuente de luz}, como una lámpara, o una
recogedora de \omterm{def:g1:light-and-shadows:source}{luz}, como una pared blanca? ¿De dónde viene de verdad la
\omterm{def:g1:light-and-shadows:source}{luz} de la \omterm{def:g2:moon-in-the-sky:moon}{Luna}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:4}
¿Qué mitad de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} ilumina el Sol: siempre la mitad que lo
mira, siempre la misma mitad o una mitad distinta al azar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:5}
¿Se puede ver alguna vez la \omterm{def:g2:moon-in-the-sky:moon}{Luna} de día? Si un amigo dice que no,
¿qué lo convencería?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:6}
Nombra las formas de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} en orden, empezando por el fino recorte
de uña y acabando en el farol redondo. ¿Cuánto dura más o menos el
espectáculo entero antes de repetirse?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:7}
En la \omterm{def:g2:moon-in-the-sky:moon}{Luna} no hay \omterm{def:g2:air-around-us:air}{aire}. ¿Qué le pasaría allí a un grito? ¿Y a una
cometa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:8}
Empieza esta noche el diario de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} del
\cref{met:g2:moon-in-the-sky:diary}. ¿Qué dos cosas tienen que ser
iguales cada tarde para que el diario sea justo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:moon-in-the-sky:9}
``¡La \omterm{def:g2:moon-in-the-sky:moon}{Luna} siguió nuestro coche hasta casa!'', dice la pequeña Rosa.
Explícale, con el paseo por la calle del
\cref{ex:g2:moon-in-the-sky:follows}, por qué las cosas muy lejanas
parecen seguirnos mientras las cercanas se quedan atrás.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:moon-in-the-sky:10}
Una noche de creciente, fíjate bien: a veces la parte \emph{oscura}
de la \omterm{def:g2:moon-in-the-sky:moon}{Luna} brilla muy débilmente, de modo que se adivina el círculo
entero. El Sol no alumbra esa parte --- pero cerca hay algo más que sí
brilla. ¿Qué bola grande, brillante e iluminada por el Sol podría
estar alumbrando el lado oscuro de la \omterm{def:g2:moon-in-the-sky:moon}{Luna}, igual que la \omterm{def:g2:moon-in-the-sky:moon}{Luna}
ilumina nuestras noches? (Pista: estás de pie encima de ella.)
\end{exercise}
