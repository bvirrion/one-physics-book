\chapter{Medir longitudes}\label{ch:g2:measuring-length}

Dos puertas, en dos casas distintas: ¿cuál es más alta? No puedes
cruzar la ciudad con una puerta a cuestas para ponerla al lado de la
otra. El año pasado comparábamos las cosas poniéndolas juntas; este
año aprendemos el mayor truco de la ciencia --- convertir una longitud
en un \emph{número} que puede viajar.

\section{Comparar sin números}

\begin{example}[Uno al lado del otro]\label{ex:g2:measuring-length:sidebyside}
Para saber cuál de dos lápices es más largo, ponlos juntos de pie
sobre la mesa: gana el que sobresale. Para comparar tu altura con la
de un amigo, poneos espalda contra espalda. Esto funciona de maravilla
--- mientras se puedan juntar las dos cosas.
\end{example}

\begin{example}[El truco de la cuerda]\label{ex:g2:measuring-length:string}
Las dos puertas no se pueden juntar --- pero una cuerda puede visitar
las dos. Estira una cuerda a lo alto de la primera puerta y córtala
exactamente a esa altura. Lleva la cuerda a la segunda puerta: si la
cuerda se queda corta, la segunda puerta es más alta. La cuerda lleva
la altura de la primera puerta al otro lado de la ciudad. Un paso más
allá: ¿y si pudiéramos mandar la altura en una \emph{carta}? Para eso
hacen falta números.
\end{example}

\section{Medir con pasos y palmos}

\begin{definition}[Medir una longitud]\label{def:g2:measuring-length:measure}
\emph{Medir}\index{medir} una longitud es contar cuántas veces cabe en
ella una \emph{unidad}\index{unidad} elegida --- un paso, un palmo, un
palito ---, colocada de punta a punta, sin huecos y sin montarse. La
respuesta es un número de unidades: ``la habitación mide $12$ pasos de
largo''.
\end{definition}

\begin{example}[Pruébalo: recorre la habitación a pasos]\label{ex:g2:measuring-length:pace}
Cruza tu habitación poniendo un pie justo delante del otro y contando
los pies. Pide a una persona mayor que haga lo mismo. Puede que tú
cuentes $18$ pies y ella solo $12$ --- ¡para la \emph{misma
habitación}! Nadie ha contado mal: tus pies son más cortos, así que
caben más. Una medida hecha con \emph{mi} pie no se puede comprobar
con \emph{tu} pie.
\end{example}

\begin{remark}[La discusión del mercado]\label{rem:g2:measuring-length:argument}
Durante mucho tiempo la gente midió de verdad en pies, pulgares y
brazadas, y en los mercados se armaban unas discusiones tremendas:
¿el pie de quién? ¿El del comerciante alto o el del cliente bajito? La
solución, acordada en el mundo entero, fue elegir \emph{una} \omterm{def:g2:measuring-length:measure}{unidad}
para todos --- una \omterm{def:g2:measuring-length:measure}{unidad} que no sea el cuerpo de nadie.
\end{remark}

\section{El centímetro y el metro}

\begin{definition}[El centímetro y el metro]\label{def:g2:measuring-length:units}
El \emph{centímetro}\index{centímetro} (que se escribe \unit{cm}) es
una \omterm{def:g2:measuring-length:measure}{unidad} de longitud pequeña, más o menos el ancho de una uña;
viene impreso en todas las reglas. El \emph{metro}\index{metro} (que
se escribe \unit{m}) es una \omterm{def:g2:measuring-length:measure}{unidad} grande, exactamente $100$
centímetros, lo que mide un paso muy largo. Estas \omterm{def:g2:measuring-length:measure}{unidades} son las
mismas para todo el mundo en la Tierra: tus \qty{20}{cm} y mis
\qty{20}{cm} son exactamente iguales.
\end{definition}

\begin{method}[Medir con una regla]\label{met:g2:measuring-length:ruler}
\begin{enumerate}
  \item coloca la regla a lo largo del objeto, pegada a él;
  \item pon la \emph{marca del cero} de la regla justo en un extremo
        del objeto --- no el borde de la regla, sino el $0$;
  \item lee el número que hay en el otro extremo: esa es la longitud
        en \omterm{def:g2:measuring-length:units}{centímetros}.
\end{enumerate}
Si el extremo cae entre dos marcas, di ``entre \qty{7}{cm} y
\qty{8}{cm}, más cerca del $8$'' --- unas palabras honradas valen más
que un número inventado.
\end{method}

\begin{example}[¿Qué unidad para cada trabajo?]\label{ex:g2:measuring-length:whichunit}
Las cosas pequeñas, como un escarabajo, un sello o tu mano, se miden
en \omterm{def:g2:measuring-length:units}{centímetros}. Las cosas grandes, como un pasillo, un autobús o
una piscina, se miden en \omterm{def:g2:measuring-length:units}{metros}: la piscina mide \qty{25}{m} de
largo --- ¡imagínate contar eso en anchos de uña! Elegir una \omterm{def:g2:measuring-length:measure}{unidad}
sensata es la mitad del trabajo de \omterm{def:g2:measuring-length:measure}{medir}.
\end{example}

\begin{example}[Calcula a ojo y comprueba]\label{ex:g2:measuring-length:estimate}
Antes de \omterm{def:g2:measuring-length:measure}{medir}, cada jugador dice lo que cree: ¿cuánto mide la mesa de
la cocina? Papá dice \qty{2}{m}; Lina dice \qty{1}{m} y pico. Después
decide la regla de un \omterm{def:g2:measuring-length:units}{metro}: \qty{1}{m} y casi medio --- punto para
Lina. Calcular a ojo primero entrena la vista; \omterm{def:g2:measuring-length:measure}{medir} mantiene a todo
el mundo honrado.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:1}
¿Cómo puedes comparar tu altura con la de un amigo sin ninguna regla?
¿Y con la de un amigo que vive en otra ciudad?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:2}
Mara mide la clase a pasos y le salen $20$; a su maestra le salen
$14$. ¿Quién ha contado mal? Explícalo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:3}
¿Qué tiene de especial el \omterm{def:g2:measuring-length:units}{centímetro} frente a un pie o un palmo?
¿Por qué se puso la gente de acuerdo en él?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:4}
¿Qué \omterm{def:g2:measuring-length:measure}{unidad} viene mejor, \unit{cm} o \unit{m}: el largo de una
hormiga; \ la altura de una puerta; \ el largo del patio del colegio;
\ el ancho de este libro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:5}
Tim apoya el borde de la regla --- no la marca del cero --- en la
punta de su lápiz, y lee \qty{13}{cm}. ¿Mide de verdad su lápiz
\qty{13}{cm}? ¿Qué se le ha olvidado del
\cref{met:g2:measuring-length:ruler}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:6}
Mide con una regla: el ancho de tu mano; \ el largo de una cuchara; \
la altura de una taza. Escribe cada respuesta con su \omterm{def:g2:measuring-length:measure}{unidad}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:7}
Una puerta mide \qty{2}{m} de alto. ¿Cuántos \omterm{def:g2:measuring-length:units}{centímetros} son?
(Recuerda: \qty{1}{m} son $100$ \unit{cm}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:8}
Primero calcula a ojo y después mide: ¿cuántos \omterm{def:g2:measuring-length:units}{metros} mide de
largo tu cama? ¿Te habías pasado, te habías quedado corto o casi
acertaste?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:9}
Una cinta mide \qty{45}{cm} de largo y otra \qty{38}{cm}. ¿Cuál es más
larga y por cuántos \omterm{def:g2:measuring-length:units}{centímetros}? ¿Podrías haberlo decidido
\emph{sin} números? ¿Qué manera es más fácil de escribir en una carta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:10}
El abuelo dice: ``Mi huerto mide $30$ zancadas de largo''. Explícale
al abuelo, con cariño, por qué ``$30$ zancadas'' no lo puede comprobar
otra persona y qué debería usar en su lugar. ¿Qué pasaría si midiera
el mismo huerto a zancadas un niño pequeño?
\end{exercise}
