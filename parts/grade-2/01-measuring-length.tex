\chapter{Measuring Length}\label{ch:g2:measuring-length}

Two doors, in two different houses: which one is taller? You cannot
carry a door across town to stand it against the other. Last year we
compared things side by side; this year we learn the greatest trick
of science --- turning a length into a \emph{number} that can travel.

\section{Comparing without numbers}

\begin{example}[Side by side]\label{ex:g2:measuring-length:sidebyside}
To find the longer of two pencils, stand them together on the table:
the one that sticks out wins. To compare your height with a friend's,
stand back to back. This works beautifully --- as long as the two
things can be brought together.
\end{example}

\begin{example}[The string trick]\label{ex:g2:measuring-length:string}
The two doors cannot meet --- but a string can visit both. Stretch a
string across the first door and cut it to exactly that height. Carry
the string to the second door: if the string is too short, the second
door is taller. The string carries the first door's height across
town. One step better: what if we could send the height in a
\emph{letter}? For that, we need numbers.
\end{example}

\section{Measuring with steps and spans}

\begin{definition}[Measuring a length]\label{def:g2:measuring-length:measure}
To \emph{measure}\index{measure} a length is to count how many times
a chosen \emph{unit}\index{unit} --- a step, a hand span, a small
stick --- fits along it, end to end with no gaps and no overlaps. The
answer is a number of units: ``the room is $12$ steps long.''
\end{definition}

\begin{example}[Try it: pace the room]\label{ex:g2:measuring-length:pace}
Cross your bedroom heel-to-toe, counting your foot-lengths. Ask a
grown-up to do the same. You might count $18$ feet and the grown-up
only $12$ --- for the \emph{same room}! Nobody counted wrong: your
feet are shorter, so more of them fit. A measurement made with
\emph{my} foot cannot be checked with \emph{your} foot.
\end{example}

\begin{remark}[The argument in the marketplace]\label{rem:g2:measuring-length:argument}
For a long time, people really did \omterm{def:g2:measuring-length:measure}{measure} in feet, thumbs and arm
spans, and markets rang with arguments: whose foot? the tall
merchant's or the short customer's? The fix, agreed upon all over the
world, was to choose \emph{one} \omterm{def:g2:measuring-length:measure}{unit} for everyone --- a \omterm{def:g2:measuring-length:measure}{unit} that
belongs to nobody's body.
\end{remark}

\section{The centimetre and the metre}

\begin{definition}[Centimetre and metre]\label{def:g2:measuring-length:units}
The \emph{centimetre}\index{centimetre} (written \unit{cm}) is a
small \omterm{def:g2:measuring-length:measure}{unit} of length, about the width of a fingernail; it is printed
on every ruler. The \emph{metre}\index{metre} (written \unit{m}) is a
big \omterm{def:g2:measuring-length:measure}{unit}, exactly $100$ centimetres, the length of a very big step.
These \omterm{def:g2:measuring-length:measure}{units} are the same for everyone on Earth: your
\qty{20}{cm} and my \qty{20}{cm} are exactly equal.
\end{definition}

\begin{method}[Measuring with a ruler]\label{met:g2:measuring-length:ruler}
\begin{enumerate}
  \item Lay the ruler along the object, touching it;
  \item put the ruler's \emph{zero mark} exactly at one end of the
        object --- not the ruler's edge, the $0$;
  \item read the number at the other end: that is the length in
        \omterm{def:g2:measuring-length:units}{centimetres}.
\end{enumerate}
If the end lands between two marks, say ``between \qty{7}{cm} and
\qty{8}{cm}, closer to $8$'' --- honest words beat a made-up number.
\end{method}

\begin{example}[Which unit for which job?]\label{ex:g2:measuring-length:whichunit}
Small things like a beetle, a stamp or your hand are measured in
\omterm{def:g2:measuring-length:units}{centimetres}. Big things like a corridor, a bus or a swimming pool are
measured in \omterm{def:g2:measuring-length:units}{metres}: the pool is \qty{25}{m} long --- imagine counting
that in fingernail-widths! Choosing a sensible \omterm{def:g2:measuring-length:measure}{unit} is half the work
of measuring.
\end{example}

\begin{example}[Estimate, then check]\label{ex:g2:measuring-length:estimate}
Before measuring, players guess: how long is the kitchen table?
Papa says \qty{2}{m}, Lina says \qty{1}{m} and a bit. Then the \omterm{def:g2:measuring-length:units}{metre}
stick decides: \qty{1}{m} and almost a half --- point for Lina.
Guessing first trains your eye; measuring keeps everyone honest.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:1}
How can you compare your height with a friend's without any ruler?
And with the height of a friend who lives in another town?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:2}
Mara \omterm{def:g2:measuring-length:measure}{measures} the classroom in steps and finds $20$; her teacher
finds $14$. Who counted wrong? Explain.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:3}
What is special about the \omterm{def:g2:measuring-length:units}{centimetre}, compared to a foot-length or a
hand span? Why did people agree on it?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:4}
Which \omterm{def:g2:measuring-length:measure}{unit} fits better, \unit{cm} or \unit{m}: the length of an ant;
\ the height of a door; \ the length of the schoolyard; \ the width
of this book?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:5}
Tim puts the edge of his ruler --- not the zero mark --- against the
end of his pencil, and reads \qty{13}{cm}. Is his pencil really
\qty{13}{cm} long? What did he forget from
\cref{met:g2:measuring-length:ruler}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:6}
Measure with a ruler: the width of your hand; \ the length of a
spoon; \ the height of a mug. Write each answer with its \omterm{def:g2:measuring-length:measure}{unit}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:7}
A door is \qty{2}{m} tall. How many \omterm{def:g2:measuring-length:units}{centimetres} is that? (Remember:
\qty{1}{m} is $100$ \unit{cm}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:8}
First guess, then \omterm{def:g2:measuring-length:measure}{measure}: how many \omterm{def:g2:measuring-length:units}{metres} long is your bed? Was
your guess too big, too small, or nearly right?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:9}
A ribbon is \qty{45}{cm} long, another is \qty{38}{cm} long. Which
is longer, and by how many \omterm{def:g2:measuring-length:units}{centimetres}? Could you have decided
\emph{without} numbers? Which way is easier to write in a letter?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:10}
Grandpa says: ``My garden is $30$ paces long.'' Explain to Grandpa,
kindly, why ``$30$ paces'' cannot be checked by someone else, and
what he should use instead. What would happen if a small child paced
the same garden?
\end{exercise}
