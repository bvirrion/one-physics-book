\chapter{Medir comprimentos}\label{ch:g2:measuring-length}

Duas portas, em duas casas diferentes: qual é a mais alta? Você não
pode carregar uma porta pela cidade para encostá-la na outra. No ano
passado comparávamos as coisas lado a lado; neste ano aprendemos o
maior truque da ciência --- transformar um comprimento num
\emph{número} que pode viajar.

\section{Comparar sem números}

\begin{example}[Lado a lado]\label{ex:g2:measuring-length:sidebyside}
Para descobrir qual de dois lápis é o mais comprido, encoste os dois
em pé na mesa: ganha aquele que sobra por cima. Para comparar a sua
altura com a de um colega, fiquem de costas um para o outro. Isso
funciona lindamente --- desde que as duas coisas possam ficar juntas.
\end{example}

\begin{example}[O truque do barbante]\label{ex:g2:measuring-length:string}
As duas portas não podem se encontrar --- mas um barbante pode
visitar as duas. Estique um barbante na primeira porta e corte-o
exatamente naquela altura. Leve o barbante até a segunda porta: se o
barbante ficar curto, a segunda porta é mais alta. O barbante carrega
a altura da primeira porta pela cidade. Um passo além: e se
pudéssemos mandar a altura numa \emph{carta}? Para isso, precisamos
de números.
\end{example}

\section{Medir com passos e palmos}

\begin{definition}[Medir um comprimento]\label{def:g2:measuring-length:measure}
\emph{Medir}\index{medir} um comprimento é contar quantas vezes uma
\emph{unidade}\index{unidade} escolhida --- um passo, um palmo, um
palitinho --- cabe ao longo dele, uma depois da outra, sem buracos e
sem sobreposições. A resposta é um número de unidades: ``a sala tem
$12$ passos de comprimento''.
\end{definition}

\begin{example}[Experimente: meça o quarto a pé]\label{ex:g2:measuring-length:pace}
Atravesse o seu quarto pisando calcanhar com dedão, contando os seus
pés. Peça a um adulto que faça o mesmo. Você pode contar $18$ pés e o
adulto só $12$ --- para o \emph{mesmo quarto}! Ninguém contou errado:
os seus pés são menores, então cabem mais deles. Uma medida feita com
o \emph{meu} pé não pode ser conferida com o \emph{seu} pé.
\end{example}

\begin{remark}[A discussão na feira]\label{rem:g2:measuring-length:argument}
Durante muito tempo as pessoas realmente mediam em pés, polegadas e
braçadas, e as feiras viviam cheias de discussão: o pé de quem? o do
vendedor alto ou o do freguês baixinho? A solução, combinada no mundo
inteiro, foi escolher \emph{uma} \omterm{def:g2:measuring-length:measure}{unidade} para todos --- uma \omterm{def:g2:measuring-length:measure}{unidade}
que não é o corpo de ninguém.
\end{remark}

\section{O centímetro e o metro}

\begin{definition}[Centímetro e metro]\label{def:g2:measuring-length:units}
O \emph{centímetro}\index{centímetro} (escrito \unit{cm}) é uma
\omterm{def:g2:measuring-length:measure}{unidade} pequena de comprimento, mais ou menos a largura de uma unha;
ele vem impresso em toda régua. O \emph{metro}\index{metro} (escrito
\unit{m}) é uma \omterm{def:g2:measuring-length:measure}{unidade} grande, exatamente $100$ centímetros, o
comprimento de um passo bem largo. Essas \omterm{def:g2:measuring-length:measure}{unidades} são as mesmas para
todo mundo na Terra: os seus \qty{20}{cm} e os meus \qty{20}{cm} são
exatamente iguais.
\end{definition}

\begin{method}[Medir com uma régua]\label{met:g2:measuring-length:ruler}
\begin{enumerate}
  \item Encoste a régua no objeto, ao longo dele;
  \item ponha a \emph{marca do zero} da régua exatamente numa das
        pontas do objeto --- não a borda da régua, e sim o $0$;
  \item leia o número na outra ponta: esse é o comprimento em
        \omterm{def:g2:measuring-length:units}{centímetros}.
\end{enumerate}
Se a ponta cair entre duas marcas, diga ``entre \qty{7}{cm} e
\qty{8}{cm}, mais perto de $8$'' --- palavras honestas valem mais que
um número inventado.
\end{method}

\begin{example}[Qual unidade para qual serviço?]\label{ex:g2:measuring-length:whichunit}
Coisas pequenas como um besouro, um selo ou a sua mão são medidas em
\omterm{def:g2:measuring-length:units}{centímetros}. Coisas grandes como um corredor, um ônibus ou uma
piscina são medidas em \omterm{def:g2:measuring-length:units}{metros}: a piscina tem \qty{25}{m} de
comprimento --- imagine contar isso em larguras de unha! Escolher uma
\omterm{def:g2:measuring-length:measure}{unidade} sensata já é metade do trabalho de \omterm{def:g2:measuring-length:measure}{medir}.
\end{example}

\begin{example}[Estimar e depois conferir]\label{ex:g2:measuring-length:estimate}
Antes de \omterm{def:g2:measuring-length:measure}{medir}, os jogadores chutam: qual é o comprimento da mesa da
cozinha? O pai diz \qty{2}{m}, Lina diz \qty{1}{m} e um pouquinho.
Aí a régua de um \omterm{def:g2:measuring-length:units}{metro} decide: \qty{1}{m} e quase meio --- ponto para
Lina. Chutar primeiro treina o olho; \omterm{def:g2:measuring-length:measure}{medir} mantém todo mundo
honesto.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:1}
Como comparar a sua altura com a de um colega sem régua nenhuma? E
com a altura de um colega que mora em outra cidade?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:2}
Mara mede a sala de aula em passos e encontra $20$; a professora dela
encontra $14$. Quem contou errado? Explique.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:3}
O que o \omterm{def:g2:measuring-length:units}{centímetro} tem de especial, comparado a um pé ou a um palmo?
Por que as pessoas combinaram usá-lo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:4}
Qual \omterm{def:g2:measuring-length:measure}{unidade} combina melhor, \unit{cm} ou \unit{m}: o comprimento de
uma formiga; \ a altura de uma porta; \ o comprimento do pátio da
escola; \ a largura deste livro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:5}
Tim encosta a borda da régua --- e não a marca do zero --- na ponta
do lápis e lê \qty{13}{cm}. O lápis dele tem mesmo \qty{13}{cm}? O
que ele esqueceu do \cref{met:g2:measuring-length:ruler}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:6}
Meça com uma régua: a largura da sua mão; \ o comprimento de uma
colher; \ a altura de uma caneca. Escreva cada resposta com a sua
\omterm{def:g2:measuring-length:measure}{unidade}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:7}
Uma porta tem \qty{2}{m} de altura. Quantos \omterm{def:g2:measuring-length:units}{centímetros} são?
(Lembre: \qty{1}{m} são $100$ \unit{cm}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-length:8}
Primeiro chute e depois meça: quantos \omterm{def:g2:measuring-length:units}{metros} tem a sua cama? O seu
chute ficou grande demais, pequeno demais ou quase certo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:9}
Uma fita tem \qty{45}{cm} de comprimento e outra tem \qty{38}{cm}.
Qual é a mais comprida, e por quantos \omterm{def:g2:measuring-length:units}{centímetros}? Você poderia ter
decidido \emph{sem} números? De que jeito é mais fácil escrever numa
carta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-length:10}
O avô diz: ``A minha horta tem $30$ passos de comprimento.'' Explique
ao avô, com jeitinho, por que ``$30$ passos'' não pode ser conferido
por outra pessoa, e o que ele deveria usar no lugar. O que
aconteceria se uma criança pequena medisse a mesma horta em passos?
\end{exercise}
