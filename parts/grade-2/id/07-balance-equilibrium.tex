\chapter{Keseimbangan dan Kesetimbangan}\label{ch:g2:balance-equilibrium}

Sebuah jungkat-jungkit tergantung mendatar dengan seorang anak di tiap
ujungnya. Menara balok berdiri --- tambah satu balok, dan ia runtuh.
Seorang akrobat berjalan di atas tali yang lebih tipis daripada
sepatunya. Ketiganya sedang memainkan permainan yang sama dan dalam:
permainan keseimbangan, tempat \omterm{def:g2:pushes-and-pulls:force}{gaya} saling meniadakan dan tidak ada
yang bergerak.

\section{Diam pun sebuah keseimbangan}

\begin{definition}[Kesetimbangan]\label{def:g2:balance-equilibrium:equilibrium}
Sebuah benda berada dalam \emph{kesetimbangan}\index{kesetimbangan}
--- kita juga mengatakan benda itu \emph{setimbang}\index{setimbang}
--- ketika dorongan dan tarikan padanya saling meniadakan, sehingga ia
tetap persis seperti keadaannya. Pita tarik tambang yang diam di
antara dua regu yang perkasa: itulah kesetimbangan. Bukan dunia tanpa
\omterm{def:g2:pushes-and-pulls:force}{gaya} --- melainkan dunia tempat \omterm{def:g2:pushes-and-pulls:force}{gaya} sedang bermain seri sempurna.
\end{definition}

\begin{example}[Buku di atas meja]\label{ex:g2:balance-equilibrium:book}
Sebuah buku tergeletak di atas meja, diam sempurna. Membosankan?
Amatilah lebih dekat dengan mata bab yang lalu. Bumi menarik buku itu
ke bawah --- jadi mengapa ia tidak jatuh? Karena meja mendorongnya
\emph{ke atas}, tepat sekuat tarikan Bumi ke bawah. Dua anak panah,
sama besar dan berlawanan arah: seri. Singkirkan mejanya dan serinya
pecah: buku itu pun jatuh.
\end{example}

\begin{remark}[Diam bukan berarti tanpa gaya]\label{rem:g2:balance-equilibrium:notfree}
Sebelum bab ini, ``tidak ada yang terjadi pada buku itu'' tampak sudah
jelas. Sekarang kamu tahu lebih baik: dua \omterm{def:g2:pushes-and-pulls:force}{gaya} bekerja padanya setiap
saat. Di sekelilingmu, kursi, lampu, rumah, dan gunung ditahan dalam
seri yang tenang antara tarikan dan dorongan. Kediaman bukanlah
kekosongan --- kediaman adalah keseimbangan.
\end{remark}

\section{Jungkat-jungkit}

\begin{example}[Seri di halaman bermain]\label{ex:g2:balance-equilibrium:seesaw}
Dua teman yang kira-kira sama besar duduk di kedua ujung
jungkat-jungkit: papannya tergantung mendatar --- seri. Sekarang Ayah
duduk di satu ujung: ia turun, dan si anak terlempar naik. Supaya
permainannya adil, Ayah bergeser \emph{ke arah tengah}: makin dekat ia
duduk ke titik tumpu, makin lemah sisinya menekan --- sampai, pada
tempat yang tepat, papannya tergantung mendatar lagi. Berat yang besar
di dekat tengah dapat berseri dengan berat yang kecil jauh di ujung.
\end{example}

{\small Seri yang sempurna di halaman bermain: dua penunggang yang sama besar, pada jarak yang sama --- dan papannya tergantung mendatar.}
\end{omfigure}

\begin{proposition}[Aturan jungkat-jungkit]\label{prop:g2:balance-equilibrium:rule}
Berat yang sama pada jarak yang sama dari titik tumpu akan seimbang.
Penunggang yang lebih berat memiringkan jungkat-jungkit --- kecuali
kalau penunggang yang lebih berat itu bergeser mendekat ke titik
tumpu, atau yang lebih ringan bergeser menjauh. Berat \emph{dan} jarak
sama-sama menentukan.
\end{proposition}

\begin{remark}[Sebuah janji]\label{rem:g2:balance-equilibrium:promise}
Seberapa dekat tepatnya Ayah harus bergeser? Ada aturan yang indah dan
persis bersembunyi di dalam jungkat-jungkit --- dengan bilangan
sungguhan di dalamnya. Aturan itu menantimu tahun depan, pada bab
tentang tuas dan neraca; permainan jungkat-jungkit hari ini adalah
latihan yang sempurna untuknya.
\end{remark}

\section{Tumbang dan titik seimbang}

\begin{example}[Menara yang miring terlalu jauh]\label{ex:g2:balance-equilibrium:tower}
Bangunlah menara balok, setiap balok menjorok sedikit lebih jauh
daripada balok di bawahnya. Menara itu berdiri, berdiri\dots\ dan
tiba-tiba runtuh. Benda yang miring akan tumbang ketika ia miring
\emph{terlalu jauh} melewati alasnya sendiri. Itulah sebabnya piramida
balok beralas lebar begitu sukar dirobohkan, sedangkan satu tiang
tinggi jatuh oleh embusan napas: kaki yang lebar membuat menara yang
kukuh.
\end{example}

\begin{definition}[Titik seimbang]\label{def:g2:balance-equilibrium:point}
Setiap benda punya \emph{titik seimbang}\index{titik seimbang}: tempat
istimewa yang membuatnya dapat ditahan mendatar --- seolah-olah
seluruh beratnya berkumpul di sana. Penggaris membawa titik
seimbangnya di tengah; palu menyembunyikannya di dekat kepala yang
berat. Ditahan tepat di bawah titik seimbangnya, benda apa pun akan
bertumpu mendatar di atas satu jari.
\end{definition}

\begin{method}[Menemukan titik seimbang]\label{met:g2:balance-equilibrium:find}
Untuk menemukan \omterm{def:g2:balance-equilibrium:point}{titik seimbang} penggaris, sapu, atau sendok kayu:
\begin{enumerate}
  \item letakkan bendanya melintang di atas dua jari telunjuk yang
        terentang, satu di dekat tiap ujung;
  \item geserkan kedua jarimu pelan-pelan saling mendekat;
  \item keduanya akan bertemu --- selalu dengan bendanya tetap
        seimbang --- pada satu tempat: \omterm{def:g2:balance-equilibrium:point}{titik seimbangnya}.
\end{enumerate}
Cobalah dengan sapu: jarimu bertemu jauh dari tengah, di sisi sapu
yang berat. \omterm{def:g2:balance-equilibrium:point}{Titik seimbang} menyukai sisi yang berat.
\end{method}

\begin{example}[Galah sang akrobat]\label{ex:g2:balance-equilibrium:acrobat}
Mengapa pejalan tali membawa galah panjang yang melengkung itu? Galah
itu pembantu keseimbangannya: ketika ia mulai miring ke satu sisi,
galah yang panjang membuat kemiringan itu pelan dan malas, bukan
cepat, dan memberinya waktu untuk bergeser kembali ke atas talinya.
Kamu memakai tipuan yang sama dengan lengan telanjang ketika berjalan
di atas tepi trotoar --- lengan terentang, goyangan melambat,
keseimbangan terselamatkan.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:1}
Sebuah lampu berdiri diam di atas meja belajar. Sebutkan dua \omterm{def:g2:pushes-and-pulls:force}{gaya} yang
bekerja padanya dan katakan mengapa lampu itu tidak bergerak.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:2}
Dalam tarik tambang, kapan pita berada dalam \omterm{def:g2:balance-equilibrium:equilibrium}{kesetimbangan}? Apa yang
memecahkan serinya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:3}
Dua anak kembar yang sama duduk di jungkat-jungkit pada jarak yang
sama. Apa yang dilakukan papannya? Salah satu kembar bergeser mundur,
lebih jauh dari titik tumpu --- sisi mana yang turun sekarang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:4}
Ayah jauh lebih berat daripada Lina, namun jungkat-jungkit mereka
tergantung mendatar. Di mana Ayah pasti duduk? Sebutkan aturan yang
menyelamatkan permainan itu.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:5}
Di mana \omterm{def:g2:balance-equilibrium:point}{titik seimbang} sebuah penggaris? Dan sebuah palu --- ke arah
ujung gagangnya atau ke arah kepala yang berat? Bagaimana kamu dapat
memeriksanya dengan satu jari?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:6}
Lakukan tipuan dua jari pada \cref{met:g2:balance-equilibrium:find}
dengan sebuah sapu. Di sisi mana jarimu bertemu, dan apa yang
diberitahukannya kepadamu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:7}
Mana yang berdiri lebih kukuh, dan mengapa: piramida balok beralas
lebar, atau menara selebar satu balok dengan tinggi yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:8}
Mengapa pejalan tali membawa galah yang panjang? Apa yang kamu lakukan
dengan lenganmu ketika berjalan di tepi trotoar, dan mengapa itu
menolong?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:balance-equilibrium:9}
Berdirilah dengan bahu kanan dan kaki kananmu menempel pada dinding,
lalu cobalah mengangkat kaki kirimu. Kamu tidak bisa --- tanpa jatuh!
Jelaskan dengan menara pada \cref{ex:g2:balance-equilibrium:tower}:
apa yang harus dilakukan tubuhmu untuk seimbang di atas kaki kanan,
dan apa yang dilarang oleh dinding itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:balance-equilibrium:10}
Sebuah derek di lokasi pembangunan mengangkat beban berat pada lengan
depannya yang panjang --- namun ia tidak terguling. Di lengan
belakangnya yang pendek duduk balok beton yang besar. Jelaskan derek
itu dengan aturan jungkat-jungkit pada
\cref{prop:g2:balance-equilibrium:rule}: siapa kedua penunggangnya,
dan mengapa lengan belakangnya pendek?
\end{exercise}
