\chapter{Mengukur Waktu}\label{ch:g2:measuring-time}

``Lima menit lagi saja!'' --- tetapi berapa lama itu sebenarnya? Waktu
tidak dapat dilihat, dipegang, atau dibentangkan di sepanjang
penggaris. Namun kita \omterm{def:g2:measuring-length:measure}{mengukurnya} setiap hari: lomba lari punya
pemenang, kue keluar dari oven tidak mentah dan tidak gosong, dan
sekolah usai pada saat yang tepat. Mari kita belajar bagaimana waktu
dijinakkan.

\section{Saat dan selang waktu}

\begin{example}[Urutan sebuah hari]\label{ex:g2:measuring-time:order}
Satu hari adalah barisan peristiwa, selalu berurutan: bangun tidur,
sarapan, sekolah, makan siang, sore, makan malam, dan tidur.
``Sarapan
datang \emph{sebelum} sekolah'' dan ``makan malam datang
\emph{sesudah} makan siang'' --- sebelum dan sesudah adalah kata
pertama tentang waktu, seperti lebih panjang dan lebih pendek adalah
kata pertama tentang panjang.
\end{example}

\begin{definition}[Selang waktu]\label{def:g2:measuring-time:duration}
Satu titik dalam hari disebut sebuah \emph{saat}\index{saat}: saat bel
berbunyi. Rentang waktu di antara dua saat disebut \emph{selang
waktu}\index{selang waktu}: dari bel sampai pelajaran berakhir. Saat
menjawab \emph{kapan?}; selang waktu menjawab \emph{berapa lama?}
\end{definition}

\begin{example}[Kapan atau berapa lama?]\label{ex:g2:measuring-time:whenhowlong}
``Filmnya mulai sesudah makan malam'' berbicara tentang sebuah saat.
``Filmnya berlangsung selama dua kali istirahat makan siang''
berbicara tentang \omterm{def:g2:measuring-time:duration}{selang waktu}. ``Ulang tahunku pada musim semi'':
saat. ``Liburan terasa begitu singkat'': \omterm{def:g2:measuring-time:duration}{selang waktu} --- dan bab ini
sebagian besar tentang \omterm{def:g2:measuring-time:duration}{selang waktu}.
\end{example}

\section{Membandingkan selang waktu}

\begin{method}[Lomba yang adil antara dua selang waktu]\label{met:g2:measuring-time:race}
Untuk mencari mana dari dua peristiwa yang berlangsung lebih lama:
\begin{enumerate}
  \item mulailah keduanya pada \emph{saat yang sama} --- ``siap, ya,
        mulai!'';
  \item perhatikan mana yang selesai lebih dahulu;
  \item yang masih berjalan ketika yang lain sudah selesai berlangsung
        lebih lama.
\end{enumerate}
Kalau keduanya tidak dapat berjalan bersama (mandi hari ini dan makan
malam kemarin), lombakan masing-masing melawan benda ketiga yang
\emph{sama} --- sebuah \omterm{ex:g2:measuring-time:clock}{jam} pasir.
\end{method}

\begin{example}[Jam pasir]\label{ex:g2:measuring-time:sandtimer}
\omterm{ex:g2:measuring-time:clock}{Jam} pasir adalah kaca kecil berpinggang sempit: pasirnya selalu
memerlukan \omterm{def:g2:measuring-time:duration}{selang waktu} yang \emph{sama} untuk mengalir habis. Itulah
yang membuatnya menjadi wasit. Menyikat gigi dilombakan dengan \omterm{ex:g2:measuring-time:clock}{jam}
pasir: pasirnya habis lebih dahulu --- sikatlah lebih lama! Sebuah
lagu dilombakan dengan \omterm{ex:g2:measuring-time:clock}{jam} pasir: lagunya menang. Jadi lagu itu
berlangsung lebih lama daripada menyikat gigi, walaupun keduanya
tidak pernah berlomba langsung.
\end{example}

\begin{remark}[Perasaan adalah jam yang buruk]\label{rem:g2:measuring-time:feelings}
Sore yang menyenangkan berlalu cepat; antrean yang membosankan
merayap --- padahal \omterm{ex:g2:measuring-time:clock}{jam} bisa saja menghitung sama untuk keduanya.
Rasa waktu kita mudah tertipu, sama seperti perabaan tertipu tentang
panas dan dingin. Justru itulah gunanya wasit seperti \omterm{ex:g2:measuring-time:clock}{jam} pasir:
pasir tidak pernah bosan.
\end{remark}

\section{Menghitung ketukan yang ajek}

\begin{definition}[Sekon]\label{def:g2:measuring-time:second}
\omterm{def:g2:measuring-length:measure}{Satuan} waktu yang dipakai di seluruh dunia adalah
\emph{sekon}\index{sekon}, ditulis \unit{s}. Satu sekon kira-kira
selama satu denyut jantung yang tenang, atau satu ketukan mantap \omterm{ex:g2:measuring-time:clock}{jam}
besar. Mengucapkan ``satu buaya kecil, dua buaya kecil'' dengan tempo
tenang menghitung kira-kira satu sekon untuk tiap buaya.
\end{definition}

\begin{example}[Cobalah: berapa sekon?]\label{ex:g2:measuring-time:count}
Hitunglah sekon dengan tempo ajek selagi: kamu menahan napas dengan
nyaman; keran mengisi sebuah gelas; seorang teman berlari ke pagar
lalu kembali. Gelas itu mungkin memerlukan $6$ sekon, larinya $20$.
Sekarang menyikat gigi dan lagu tadi bisa mendapat \emph{bilangan}
--- dan bilangan, seperti pada panjang, dapat bepergian dan diperiksa.
\end{example}

\begin{example}[Jam pada arloji]\label{ex:g2:measuring-time:clock}
Sekon dipakai untuk hal yang singkat. Rentang yang panjang dihitung
dalam \emph{jam}\index{jam}: pagi di sekolah berlangsung berjam-jam,
tidur semalam pun begitu. Jarum pendek arloji menunjuk \omterm{ex:g2:measuring-time:clock}{jamnya}: tepat
pada angka $3$, berarti pukul tiga; di antara $3$ dan $4$ dengan jarum
panjang menunjuk lurus ke bawah, berarti pukul setengah empat. Satu
siang dan satu malam bersama berisi $24$ \omterm{ex:g2:measuring-time:clock}{jam} --- selagi Bumi, ingat,
menyelesaikan satu putaran penuh.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:1}
Urutkan: makan malam; \ bangun tidur; \ bermain sore; \ sarapan;
\ tidur.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:2}
Saat atau \omterm{def:g2:measuring-time:duration}{selang waktu}? ``Bel berbunyi pada tengah hari''; \ ``kue
dipanggang selama setengah \omterm{ex:g2:measuring-time:clock}{jam}''; \ ``kami berangkat sesudah makan
siang''; \ ``perjalanan itu terasa tanpa akhir''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:3}
Dua lilin dinyalakan pada saat yang sama; yang merah padam sementara
yang biru masih menyala. Lilin mana yang bertahan lebih lama? Aturan
mana pada \cref{met:g2:measuring-time:race} yang diikuti kedua lilin?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:4}
Mengapa \omterm{ex:g2:measuring-time:clock}{jam} pasir menjadi wasit yang adil bagi \omterm{def:g2:measuring-time:duration}{selang waktu}? Apa yang
selalu sama padanya, setiap kali ia dibalik?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:5}
Ben berkata penantiannya yang membosankan di dokter gigi berlangsung
``selamanya''. Arlojinya berkata penantian itu lebih singkat daripada
film kartun kesukaannya. Bagaimana ini mungkin? (Lihat
\cref{rem:g2:measuring-time:feelings}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:6}
Hitunglah sekon dengan tempo ajek dan ukurlah: berapa sekon yang kamu
perlukan untuk menulis nama lengkapmu? Untuk memakai sepatumu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:7}
\omterm{def:g2:measuring-length:measure}{Satuan} mana yang lebih cocok, sekon atau \omterm{ex:g2:measuring-time:clock}{jam}: satu kali bersin;
\ tidur semalam; \ mengikat tali sepatu; \ satu pagi di sekolah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measuring-time:8}
Gambarlah arloji yang menunjukkan pukul lima, dan satu lagi yang
menunjukkan pukul setengah enam. Ke mana jarum pendek menunjuk pada
pukul setengah enam?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-time:9}
Mandi kemarin dan mandi pancuran hari ini tidak dapat saling
berlomba. Jelaskan, langkah demi langkah, cara mengetahui mana yang
berlangsung lebih lama dengan satu \omterm{ex:g2:measuring-time:clock}{jam} pasir --- dan apa yang harus
dicatat supaya temanmu dapat memeriksanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measuring-time:10}
Lila menghitung ``satu, dua, tiga\dots'' dengan sangat cepat dan
berkata gelas itu terisi dalam $12$ sekon; Tom menghitung dengan
tenang dan mendapat $6$. Keran mengalir sama pada kedua kali itu.
Siapa yang lebih dekat kepada kebenaran, dan apa yang membuat
hitungan sekon menjadi jujur?
\end{exercise}
