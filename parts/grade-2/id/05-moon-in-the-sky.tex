\chapter{Bulan di Langit}\label{ch:g2:moon-in-the-sky}

Pada beberapa petang, senyum perak yang tipis tergantung di atas atap
rumah; dua minggu kemudian, lentera bulat penuh membanjiri kebun
dengan \omterm{def:g1:light-and-shadows:source}{cahaya} pucat. Itu \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang sama --- tetangga terdekat kita di
langit, dan satu-satunya dunia lain yang pernah diinjak manusia.
Saatnya berkenalan dengan sungguh-sungguh.

\section{Tetangga kita di langit}

\begin{definition}[Bulan]\label{def:g2:moon-in-the-sky:moon}
\emph{Bulan}\index{Bulan} adalah bola raksasa dari batuan kelabu yang
mengelilingi Bumi, teman setia kita di ruang angkasa. Ia jauh lebih
kecil daripada Bumi, dan sangat jauh --- lebih jauh daripada
penerbangan pesawat mana pun, begitu jauh sehingga para antariksawan
yang mengunjunginya menempuh perjalanan tiga hari untuk sampai ke
sana.
\end{definition}

\begin{example}[Mengamati dari dekat]\label{ex:g2:moon-in-the-sky:closely}
Dengan mata telanjang, \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menampakkan bercak pucat dan daerah
terang --- sebagian orang melihat wajah di sana. Lewat teropong (hal
yang menyenangkan untuk dicoba pada malam \omterm{ex:g2:moon-in-the-sky:shapes}{bulan purnama}), bercak itu
menajam menjadi dataran dan \emph{kawah} bundar, bekas luka berbentuk
mangkuk yang ditinggalkan batuan dari ruang angkasa yang menghantam di
sana sejak lama. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menyimpan bekas lukanya selamanya: di sana
tidak ada \omterm{def:g2:air-around-us:wind}{angin} dan tidak ada hujan yang menghapusnya --- memang di
sana sama sekali tidak ada \omterm{def:g2:air-around-us:air}{udara}. Tidak ada \omterm{def:g2:air-around-us:air}{udara} juga berarti dunia
yang sunyi: di \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, teriakan pun tidak menimbulkan bunyi.
\end{example}

\begin{example}[Bulan yang mengikutimu]\label{ex:g2:moon-in-the-sky:follows}
Berjalanlah menyusuri jalan pada petang yang bermandi \omterm{def:g1:light-and-shadows:source}{cahaya} bulan:
rumah dan pohon meluncur lewat, tetapi \omterm{def:g2:moon-in-the-sky:moon}{Bulan} seakan mengikutimu,
selalu di sebelah bahumu. Sebenarnya ia tidak sedang mengejarmu --- ia
begitu jauh sehingga berjalan sepanjang satu jalan sama sekali tidak
mengubah pandanganmu terhadapnya, sementara rumah di dekatmu berubah
terus-menerus. Benda yang sangat jauh selalu tampak mengikuti setiap
pejalan; \omterm{def:g2:moon-in-the-sky:moon}{Bulan} melakukannya paling baik karena ia juaranya jauh.
\end{example}

\section{Cahaya pantulan}

\begin{example}[Cahaya bulan adalah cahaya matahari yang dipantulkan]\label{ex:g2:moon-in-the-sky:borrowed}
Apakah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} sebuah \omterm{def:g1:light-and-shadows:source}{sumber cahaya}? Ingat ujinya dari tahun lalu:
apakah ia membuat \omterm{def:g1:light-and-shadows:source}{cahayanya} sendiri? Tidak --- \omterm{def:g2:moon-in-the-sky:moon}{Bulan} adalah bola
batuan gelap, seperti kerikil raksasa di langit. Yang kita sebut
\omterm{def:g1:light-and-shadows:source}{cahaya} bulan adalah \omterm{def:g1:light-and-shadows:source}{cahaya} matahari yang \emph{dipantulkan} oleh
batuan itu --- dilemparkan kembali ke arah kita, seperti dinding putih
menerangi sebuah ruangan tanpa menjadi lampu. Matahari menerangi \omterm{def:g2:moon-in-the-sky:moon}{Bulan}
dengan terang pada satu sisi; kita melihat bagian mana pun dari sisi
terang itu yang menghadap kita.
\end{example}

\begin{remark}[Bulan pada siang hari]\label{rem:g2:moon-in-the-sky:daytime}
Banyak orang percaya \omterm{def:g2:moon-in-the-sky:moon}{Bulan} adalah makhluk malam --- tetapi tengadahlah
pada beberapa sore, dan ia ada di sana, hantu pucat di langit biru!
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} mengelilingi Bumi menurut jadwalnya sendiri dan tidak peduli
apakah saat itu siang atau malam bagi kita. Ia hanya tampil kurang
megah pada siang hari, ketika silau Matahari memenuhi langit.
\end{remark}

\section{Bulan berubah bentuk}

\begin{example}[Bentuk Bulan yang berubah-ubah]\label{ex:g2:moon-in-the-sky:shapes}
Amati \omterm{def:g2:moon-in-the-sky:moon}{Bulan} selama banyak petang dan bentuknya berubah:
\emph{sabit}\index{sabit} yang tipis seperti potongan kuku, lalu
setengah lingkaran, lalu bentuk gemuk yang hampir bundar, lalu cakram
sempurna \emph{bulan purnama}\index{bulan purnama} --- dan kemudian
menyusut lagi, menjadi \omterm{ex:g2:moon-in-the-sky:shapes}{sabit} yang menghadap ke arah sebaliknya, sampai
selama satu dua malam \omterm{def:g2:moon-in-the-sky:moon}{Bulan} seakan lenyap sama sekali. Seluruh
pertunjukan itu memakan waktu kira-kira sebulan, lalu berulang, seajek
siang dan malam.
\end{example}

\begin{method}[Membuat buku harian Bulan]\label{met:g2:moon-in-the-sky:diary}
\begin{enumerate}
  \item Setiap petang yang cerah, kira-kira pada \omterm{ex:g2:measuring-time:clock}{jam} yang sama,
        carilah \omterm{def:g2:moon-in-the-sky:moon}{Bulan};
  \item gambarlah bentuk yang kamu lihat pada kotak kecil sebuah
        kalender --- sebuah lingkaran, lalu arsirlah bagian gelapnya;
  \item kalau awan menutupinya, tulislah ``berawan'' dan teruskan;
  \item sesudah dua minggu, jajarkan gambar-gambarmu.
\end{enumerate}
Gambarmu sendiri akan memperlihatkan bentuknya berubah, sedikit demi
sedikit setiap malam. \emph{Mengapa} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berbuat begitu adalah
teka-teki yang lezat --- teka-teki itu akan dipecahkan dengan tuntas
pada tahun yang lebih tinggi dalam pelajaran ini, dan buku harianmu
adalah langkah pertama pembuktiannya.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:1}
Apakah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} lebih besar atau lebih kecil daripada Bumi? Lebih dekat
atau lebih jauh daripada awan? Dari mana kita tahu bahwa orang dapat
mengunjunginya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:2}
Bekas bundar berbentuk mangkuk di \omterm{def:g2:moon-in-the-sky:moon}{Bulan} itu disebut apa, dan apa yang
membuatnya? Mengapa bekas itu masih ada sesudah waktu selama itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:3}
Apakah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} sebuah \omterm{def:g1:light-and-shadows:source}{sumber cahaya}, seperti lampu, atau penangkap
\omterm{def:g1:light-and-shadows:source}{cahaya}, seperti dinding putih? Dari mana sebenarnya \omterm{def:g1:light-and-shadows:source}{cahaya} bulan
datang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:4}
Separuh \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang mana yang diterangi Matahari: selalu separuh yang
menghadap Matahari, selalu separuh yang sama, atau separuh yang
berganti secara acak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:5}
Dapatkah \omterm{def:g2:moon-in-the-sky:moon}{Bulan} terlihat pada siang hari? Kalau seorang teman berkata
tidak, apa yang akan meyakinkannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:6}
Sebutkan bentuk-bentuk \omterm{def:g2:moon-in-the-sky:moon}{Bulan} secara berurutan, mulai dari potongan
kuku yang tipis dan berakhir pada lentera bundar. Kira-kira berapa
lama seluruh pertunjukan itu berlangsung sebelum berulang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:7}
Di \omterm{def:g2:moon-in-the-sky:moon}{Bulan} tidak ada \omterm{def:g2:air-around-us:air}{udara}. Apa yang akan terjadi pada teriakan di sana?
Dan pada layang-layang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:moon-in-the-sky:8}
Mulailah buku harian \omterm{def:g2:moon-in-the-sky:moon}{Bulan} pada \cref{met:g2:moon-in-the-sky:diary}
malam ini. Dua hal apa yang harus tetap sama setiap petang supaya buku
hariannya adil?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:moon-in-the-sky:9}
``\omterm{def:g2:moon-in-the-sky:moon}{Bulan} mengikuti mobil kami sepanjang jalan pulang!'' kata Rosa
kecil. Jelaskan kepadanya, dengan jalan kaki menyusuri jalan pada
\cref{ex:g2:moon-in-the-sky:follows}, mengapa benda yang sangat jauh
tampak mengikuti, sementara benda yang dekat meluncur lewat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:moon-in-the-sky:10}
Pada malam berbulan \omterm{ex:g2:moon-in-the-sky:shapes}{sabit}, amatilah baik-baik: kadang bagian
\emph{gelap} \omterm{def:g2:moon-in-the-sky:moon}{Bulan} berpendar sangat samar, sehingga seluruh
lingkarannya masih dapat ditebak. Matahari tidak menyinari bagian itu
--- tetapi ada benda lain yang bersinar di dekatnya. Bola besar,
terang, dan tersinari matahari apa yang mungkin menerangi sisi gelap
\omterm{def:g2:moon-in-the-sky:moon}{Bulan}, seperti \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menerangi malam kita? (Petunjuk: kamu sedang
berdiri di atasnya.)
\end{exercise}
