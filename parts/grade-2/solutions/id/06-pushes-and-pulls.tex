\section*{Bab \ref{ch:g2:pushes-and-pulls} --- Dorongan dan Tarikan: Gaya}

\begin{solution}{exo:g2:pushes-and-pulls:1}
Dorongan; tarikan; dorongan; tarikan; dorongan (remasan adalah dorongan dari dua sisi).
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:2}
Di mana \omterm{def:g2:pushes-and-pulls:force}{gaya} itu mencengkeram benda, ke arah mana ia bekerja, dan
seberapa kuat ia (lewat panjang anak panahnya).
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:3}
Menggerakkan, menghentikan, membelokkan, mengubah bentuk. Contohnya
beragam --- misalnya: menyentil kelereng (menggerakkan), menangkap
piring terbang (menghentikan), memantulkan bola pada dinding
(membelokkan), memipihkan bola salju (mengubah bentuk).
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:4}
Menghentikannya: doronglah melawan geraknya, jari ke arah kelereng.
Melajukannya: doronglah searah gulirannya, dari belakang.
Membelokkannya: doronglah dari samping, melintang terhadap
lintasannya.
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:5}
Melenting kembali: spons dan karet gelang. Mempertahankan lekukan:
plastisin dan salju yang lembut.
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:6}
Dorongan yang lembut: anak panah pendek menunjuk ke kanan. Dorongan
yang kuat: anak panah panjang menunjuk ke kiri. Panjangnya berbeda,
arahnya berlawanan.
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:7}
Tarikan Bumi. Ia tidak perlu menyentuh --- ia menarik sendok itu
menembus \omterm{def:g2:air-around-us:air}{udara}, dan selalu ke arah bawah.
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:8}
Tarikan Bumi (kaus kaki yang dijatuhkan akan turun) dan tarikan \omterm{def:g1:magnets:magnet}{magnet}
(\omterm{def:g1:magnets:magnet}{magnet} lemari es menahan gambar menembus kertas).
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:9}
Dua anak panah pada tali, sama panjang, menunjuk ke arah berlawanan.
Pita itu diam ketika kedua tarikan \emph{sama kuat} --- keduanya
saling meniadakan. Begitu satu regu menarik lebih kuat, anak panahnya
menang dan pita bergerak ke arah itu.
\end{solution}

\begin{solution}{exo:g2:pushes-and-pulls:10}
\omterm{def:g2:pushes-and-pulls:force}{Gaya} yang sama pada ketiga saat itu: tarikan Bumi, dengan anak panah
menunjuk lurus ke bawah --- ketika naik (memperlambat bola), di puncak
(bersiap membawanya kembali), dan ketika turun (mempercepatnya).
Tarikan itu tidak pernah berhenti walau sesaat.
\end{solution}
