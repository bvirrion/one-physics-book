\section*{Bab \ref{ch:g2:moon-in-the-sky} --- Bulan di Langit}

\begin{solution}{exo:g2:moon-in-the-sky:1}
Jauh lebih kecil daripada Bumi, dan jauh lebih jauh daripada awan ---
lebih jauh daripada penerbangan pesawat mana pun. Kita tahu orang
dapat mengunjunginya karena para antariksawan sudah melakukannya:
perjalanannya memakan waktu tiga hari untuk tiap arah.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:2}
Kawah --- bekas luka berbentuk mangkuk yang dibuat oleh batuan dari
ruang angkasa yang menghantam \omterm{def:g2:moon-in-the-sky:moon}{Bulan}. Bekas itu bertahan karena di
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} tidak ada \omterm{def:g2:air-around-us:wind}{angin} dan tidak ada hujan yang mengikisnya.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:3}
Penangkap \omterm{def:g1:light-and-shadows:source}{cahaya}. \omterm{def:g1:light-and-shadows:source}{Cahaya} bulan adalah \omterm{def:g1:light-and-shadows:source}{cahaya} matahari yang memantul
pada batuan kelabu \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, seperti dinding putih menerangi ruangan
tanpa menjadi lampu.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:4}
Selalu separuh yang menghadap Matahari --- sisi yang kena matahari
terang, sisi jauhnya sedang mengalami malamnya sendiri.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:5}
Bisa --- pada banyak sore, \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang pucat tergantung di langit biru.
Bukti terbaik bagi teman yang ragu: keluarlah pada sore seperti itu
lalu tunjuklah \omterm{def:g2:moon-in-the-sky:moon}{Bulannya}.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:6}
\omterm{ex:g2:moon-in-the-sky:shapes}{Sabit}, setengah, hampir purnama, \omterm{ex:g2:moon-in-the-sky:shapes}{bulan purnama} --- lalu turun lagi
lewat hampir purnama, setengah, \omterm{ex:g2:moon-in-the-sky:shapes}{sabit}, sampai ke \omterm{def:g2:moon-in-the-sky:moon}{Bulan} ``baru'' yang
bersembunyi. Seluruh pertunjukannya memakan waktu kira-kira sebulan.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:7}
Teriakan tidak akan menimbulkan bunyi --- bunyi memerlukan \omterm{def:g2:air-around-us:air}{udara}.
Layang-layang tidak dapat terbang: tanpa \omterm{def:g2:air-around-us:air}{udara} tidak ada \omterm{def:g2:air-around-us:wind}{angin}, dan
tidak ada apa pun untuk disandari layang-layang.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:8}
\omterm{ex:g2:measuring-time:clock}{Jam} yang sama dan gambar yang sama jujurnya setiap petang (tempat yang
sama juga menolong): buku harian yang adil hanya mengubah satu hal
setiap kali --- yaitu malamnya.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:9}
Rumah-rumah itu dekat, sehingga melewatinya dengan mobil mengubah cara
Rosa melihatnya --- rumah itu meluncur lewat lalu hilang. \omterm{def:g2:moon-in-the-sky:moon}{Bulan} begitu
jauh sehingga seluruh perjalanan itu sama sekali tidak mengubah
pandangannya: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} tetap di sebelah bahunya, seakan mengikuti. Sang
juara jauh selalu tampak ``mengikuti''.
\end{solution}

\begin{solution}{exo:g2:moon-in-the-sky:10}
Bumi itu sendiri. \omterm{def:g1:light-and-shadows:source}{Cahaya} matahari memantul dari Bumi yang terang ke
sisi gelap \omterm{def:g2:moon-in-the-sky:moon}{Bulan} dan menerangi sisi itu dengan samar --- \omterm{def:g1:light-and-shadows:source}{cahaya} bumi:
planet kita membalas kebaikan \omterm{def:g2:moon-in-the-sky:moon}{Bulan}.
\end{solution}
