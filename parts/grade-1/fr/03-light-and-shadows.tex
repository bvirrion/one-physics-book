\chapter{La lumière et les ombres}\label{ch:g1:light-and-shadows}

Éteins la lampe dans une pièce sans fenêtre~: tout disparaît.
Rallume-la~: la pièce revient. Pour voir, il faut de la \omterm{def:g1:light-and-shadows:source}{lumière}. Et
partout où il y a de la \omterm{def:g1:light-and-shadows:source}{lumière}, il y a aussi des \omterm{def:g1:light-and-shadows:shadow}{ombres} --- ton
fidèle jumeau sombre te suit sur tous les trottoirs ensoleillés.
Cherchons d'où vient la \omterm{def:g1:light-and-shadows:source}{lumière}, et comment naissent les \omterm{def:g1:light-and-shadows:shadow}{ombres}.

\section{D'où vient la lumière}

\begin{definition}[Sources de lumière]\label{def:g1:light-and-shadows:source}
Une \emph{source de lumière}\index{source de lumière} est un objet
qui fabrique sa propre \emph{lumière}\index{lumière}~: le Soleil, une
lampe, la flamme d'une bougie, une luciole. Tout le reste --- un
livre, un ballon, les murs, toi --- ne fabrique aucune lumière~: nous
ne voyons ces objets que lorsque la lumière d'une source tombe sur
eux.
\end{definition}

\begin{example}[Sources ou non~?]\label{ex:g1:light-and-shadows:sources}
Le Soleil, une lampe de poche, une allumette qui brûle et un écran de
télévision fabriquent tous de la \omterm{def:g1:light-and-shadows:source}{lumière}~: ce sont des sources. La
Lune, un miroir et un mur blanc peuvent paraître très brillants, mais
ils ne fabriquent aucune \omterm{def:g1:light-and-shadows:source}{lumière}~; ils ne font que renvoyer la
\omterm{def:g1:light-and-shadows:source}{lumière} qui les atteint. Éteins la lampe, et le miroir le plus
brillant devient tout noir.
\end{example}

\begin{example}[À toi de jouer~: la pièce
noire]\label{ex:g1:light-and-shadows:darkroom}
Le soir, ferme les volets ou les rideaux d'une pièce, éteins toutes
les lampes, et attends en comptant lentement jusqu'à $20$. Que
vois-tu~? Au début, presque rien --- sans source, il n'y a pas de
\omterm{def:g1:light-and-shadows:source}{lumière} pour voir. Entrouvre ensuite la porte~: une mince tranche de
\omterm{def:g1:light-and-shadows:source}{lumière} suffit à faire revenir les tables, les chaises et les jouets.
\end{example}

\begin{remark}[Ne jamais regarder le Soleil]\label{rem:g1:light-and-shadows:sun}
Le Soleil est de très loin la \omterm{def:g1:light-and-shadows:source}{source de lumière} la plus puissante que
nous rencontrions jamais. Ne le regarde jamais en face, pas même un
instant, et jamais à travers des jumelles ou une loupe~: sa \omterm{def:g1:light-and-shadows:source}{lumière}
est si forte qu'elle peut abîmer tes yeux pour toujours. Le Soleil se
savoure de côté --- par la \omterm{def:g1:light-and-shadows:source}{lumière} qu'il jette sur tout le reste.
\end{remark}

\section{Comment naissent les ombres}

\begin{definition}[Ombre]\label{def:g1:light-and-shadows:shadow}
Une \emph{ombre}\index{ombre} est la tache sombre qui apparaît
derrière un objet qui arrête la \omterm{def:g1:light-and-shadows:source}{lumière}. La \omterm{def:g1:light-and-shadows:source}{lumière} ne peut pas
traverser ton corps~: derrière toi, le sol ne reçoit donc rien du
Soleil, et cette tache sombre à ta forme est ton ombre.
\end{definition}

\begin{method}[Faire une ombre bien nette]\label{met:g1:light-and-shadows:make}
\begin{enumerate}
  \item Faire le noir dans la pièce --- les \omterm{def:g1:light-and-shadows:shadow}{ombres} se cachent quand
        la \omterm{def:g1:light-and-shadows:source}{lumière} vient de partout~;
  \item allumer \emph{une seule} lampe ou lampe de poche~;
  \item tenir un jouet entre la lampe et le mur~;
  \item regarder le mur~: voilà l'\omterm{def:g1:light-and-shadows:shadow}{ombre}, du côté du jouet opposé à la
        lampe.
\end{enumerate}
Rapproche le jouet de la lampe~: l'\omterm{def:g1:light-and-shadows:shadow}{ombre} devient énorme.
Rapproche-le du mur~: l'\omterm{def:g1:light-and-shadows:shadow}{ombre} rétrécit jusqu'à la vraie taille du
jouet.
\end{method}

\begin{example}[À toi de jouer~: les animaux
d'ombre]\label{ex:g1:light-and-shadows:animals}
Avec la lampe de \cref{met:g1:light-and-shadows:make}, mets tes mains
dans la \omterm{def:g1:light-and-shadows:source}{lumière} à la place d'un jouet~: deux doigts dressés font des
oreilles de lapin, les deux pouces croisés font un oiseau qui vole.
Le mur montre la forme que tes mains découpent dans la \omterm{def:g1:light-and-shadows:source}{lumière} ---
une \omterm{def:g1:light-and-shadows:shadow}{ombre} garde toujours la forme de son obstacle, vu depuis la
lampe.
\end{example}

\begin{remark}[Une ombre n'est pas un objet]\label{rem:g1:light-and-shadows:notathing}
On ne peut pas ramasser une \omterm{def:g1:light-and-shadows:shadow}{ombre}, la plier, ni la garder dans une
boîte. Une \omterm{def:g1:light-and-shadows:shadow}{ombre} n'est faite de rien du tout~: c'est un \emph{endroit
où la \omterm{def:g1:light-and-shadows:source}{lumière} manque}. Voilà pourquoi elle disparaît à la seconde où
la lampe s'éteint --- il n'y a rien, là, qui puisse rester.
\end{remark}

\section{Les ombres suivent la lumière}

\begin{example}[De quel côté est l'ombre~?]\label{ex:g1:light-and-shadows:opposite}
Le matin, le Soleil est d'un côté de toi et ton \omterm{def:g1:light-and-shadows:shadow}{ombre} s'allonge de
l'\emph{autre} côté. Fais face au Soleil~: ton \omterm{def:g1:light-and-shadows:shadow}{ombre} se cache
derrière toi. Tourne-lui le dos~: ton \omterm{def:g1:light-and-shadows:shadow}{ombre} s'étire devant toi.
L'\omterm{def:g1:light-and-shadows:shadow}{ombre} se place toujours exactement à l'opposé de la source.
\end{example}

\begin{example}[À toi de jouer~: la craie autour d'une
ombre]\label{ex:g1:light-and-shadows:chalk}
Un matin ensoleillé, tiens-toi immobile dans la cour pendant qu'un
camarade trace à la craie le contour de ton \omterm{def:g1:light-and-shadows:shadow}{ombre}. Reviens après le
déjeuner et place-toi au même endroit~: ton \omterm{def:g1:light-and-shadows:shadow}{ombre} est sortie de son
trait de craie et a changé de longueur~! Le Soleil a changé de place
dans le ciel --- et l'\omterm{def:g1:light-and-shadows:shadow}{ombre} l'a suivi. Le chapitre suivant accompagne
le Soleil pendant toute sa journée.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:1}
Range en \omterm{def:g1:light-and-shadows:source}{sources de lumière} et non-sources~: la flamme d'une
bougie~; \ la Lune~; \ une luciole~; \ un miroir~; \ une lampe de
poche~; \ un mur blanc.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:2}
Pourquoi ne vois-tu absolument rien dans une pièce fermée, sans
fenêtre et sans lampe~? Qu'est-ce qui manque~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:3}
Nomme les trois choses nécessaires pour faire une \omterm{def:g1:light-and-shadows:shadow}{ombre} sur le mur,
et range-les dans le bon ordre le long d'une ligne.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:4}
Le Soleil est à ta gauche. De quel côté est ton \omterm{def:g1:light-and-shadows:shadow}{ombre}~? Tu fais
demi-tour --- où est-elle maintenant~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:5}
À l'aide de \cref{met:g1:light-and-shadows:make}, rends l'\omterm{def:g1:light-and-shadows:shadow}{ombre} du
même jouet d'abord énorme, puis toute petite. Qu'as-tu déplacé, et
dans quel sens~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:6}
Peut-on ramasser une \omterm{def:g1:light-and-shadows:shadow}{ombre} et la mettre dans sa poche~? Explique ce
qu'est vraiment une \omterm{def:g1:light-and-shadows:shadow}{ombre}, à l'aide de
\cref{rem:g1:light-and-shadows:notathing}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:7}
Pourquoi ne faut-il jamais regarder le Soleil en face, alors que
regarder une lampe ne fait qu'un peu éblouir~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:light-and-shadows:8}
À midi, le Soleil est haut~; en fin d'après-midi, il est bas. Quand
ton \omterm{def:g1:light-and-shadows:shadow}{ombre} est-elle la plus longue~? Dessine les deux \omterm{def:g1:light-and-shadows:shadow}{ombres} d'un
arbre, une pour chaque moment.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:light-and-shadows:9}
Mia dit~: «~Mon \omterm{def:g1:light-and-shadows:shadow}{ombre} était devant moi, et après avoir fait demi-tour
elle était derrière moi --- l'\omterm{def:g1:light-and-shadows:shadow}{ombre} a donc tourné avec moi.~» Sam dit
que l'\omterm{def:g1:light-and-shadows:shadow}{ombre} n'a pas bougé du tout. Pense à
\cref{ex:g1:light-and-shadows:opposite}~: qui a raison, et pourquoi
a-t-on eu l'\emph{impression} que l'\omterm{def:g1:light-and-shadows:shadow}{ombre} tournait~?
\end{exercise}
