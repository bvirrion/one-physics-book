\chapter{Le chaud et le froid}\label{ch:g1:hot-and-cold}

La soupe fume, la glace fait mal aux dents, et le siège de la voiture
brûle les jambes en été. Le chaud et le froid sont partout. Dans ce
chapitre, nous apprenons à les comparer, à voir ce qui réchauffe les
choses --- et à regarder ce qui arrive à un glaçon quand la chaleur
gagne.

\section{Plus chaud, plus froid}

\begin{definition}[Chaud et froid]\label{def:g1:hot-and-cold:words}
\emph{Chaud}\index{chaud} et \emph{froid}\index{froid} sont des mots
de comparaison, comme grand et petit. La soupe est chaude \emph{par
rapport à} ta main~; la glace est froide \emph{par rapport à} ta
main. Entre les deux, il y a toute une échelle de mots~: glacé,
froid, frais, tiède, chaud, brûlant.
\end{definition}

\begin{example}[Ranger trois boissons]\label{ex:g1:hot-and-cold:drinks}
Sur la table~: un chocolat chaud, un verre d'eau du robinet, un verre
de jus d'orange sorti du réfrigérateur. Du plus froid au plus chaud~:
le jus, puis l'eau du robinet, puis le chocolat chaud. L'eau du
robinet est froide à côté du chocolat, mais chaude à côté du jus ---
les mots de comparaison marchent toujours par deux.
\end{example}

\begin{remark}[Le toucher se laisse tromper]\label{rem:g1:hot-and-cold:fooled}
Souviens-toi des trois bols du chapitre précédent~: la même eau tiède
paraissait chaude à une main et fraîche à l'autre. Et touche une
cuillère en métal et une cuillère en bois qui ont passé la nuit dans
la même cuisine~: celle en métal \emph{paraît} plus froide, alors que
rien, dans la cuisine, n'est plus froid que le reste. Le toucher est
un gardien utile, mais ce n'est pas un juge équitable --- il nous
faudra bientôt un instrument qui, lui, ne se trompe pas~: le
thermomètre, qui arrive dans un chapitre plus loin.
\end{remark}

\begin{method}[Comparer honnêtement le chaud et le
froid]\label{met:g1:hot-and-cold:compare}
Pour décider lequel de deux bols d'eau est le plus chaud~:
\begin{enumerate}
  \item se servir de la \emph{même} main pour les deux --- deux mains
        peuvent ne pas être d'accord~;
  \item toucher l'un, puis l'autre, tout de suite --- attendre laisse
        les choses changer~;
  \item pour en être encore plus sûr, demander à un camarade
        d'essayer aussi, sans lui dire ta réponse avant.
\end{enumerate}
Si ton camarade et toi êtes d'accord, l'observation est solide.
\end{method}

\section{Ce qui réchauffe}

\begin{example}[Les réchauffeurs]\label{ex:g1:hot-and-cold:warmers}
Certaines choses réchauffent tout ce qui se trouve près d'elles~: le
Soleil réchauffe le banc du parc, la flamme réchauffe la casserole,
le radiateur réchauffe la pièce, et ton propre corps réchauffe la
couverture. Frotter réchauffe aussi~: presse tes paumes l'une contre
l'autre et frotte vite en comptant jusqu'à $10$ --- sens-les
chauffer.
\end{example}

\begin{example}[À toi de jouer~: soleil et
ombre]\label{ex:g1:hot-and-cold:sunshade}
Un jour de soleil, pose une main sur un mur au soleil et l'autre sur
le même mur à l'ombre. Même mur, même jour --- et pourtant le côté
ensoleillé est nettement plus chaud. Le Soleil est le grand
réchauffeur de tout ce qui est dehors~; il aura bientôt un chapitre
rien qu'à lui.
\end{example}

\begin{remark}[Un manteau ne fabrique pas de chaleur]\label{rem:g1:hot-and-cold:coat}
Un manteau ne te réchauffe pas comme le fait un radiateur. C'est ton
\emph{corps} qui fabrique la chaleur~; le manteau empêche seulement
cette chaleur de s'échapper, comme un couvercle sur la soupe. C'est
pour cela qu'un manteau posé tout seul sur une chaise reste froid à
l'intérieur --- rien, dedans, ne fabrique de chaleur à garder.
\end{remark}

\section{Quand la chaleur gagne~: la glace fond}

\begin{example}[La vie d'un glaçon]\label{ex:g1:hot-and-cold:icecube}
Pose un glaçon sur une assiette, dans la cuisine. Il est dur et
froid. L'air chaud de la pièce le réchauffe, petit à petit~: le
glaçon devient de plus en plus petit, une flaque grandit autour de
lui, et au bout d'un moment il ne reste que de l'eau. La chaleur de
la pièce a changé la glace dure en eau liquide --- on dit que la
glace a \emph{fondu}.
\end{example}

\begin{example}[Le froid aussi peut gagner]\label{ex:g1:hot-and-cold:freezer}
Le congélateur fait le travail inverse~: mets-y un petit gobelet
d'eau le soir, et au matin tu le trouveras dur comme la pierre ---
de la glace. La chaleur fait fondre la glace en eau~; un grand froid
change l'eau en glace. L'eau n'arrêtera pas de faire ces
allers-retours tout au long du livre.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:1}
Range du plus froid au plus chaud~: l'eau du bain~; \ un glaçon~; \ la
soupe qui sort de la casserole~; \ l'air de ta chambre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:2}
Trouve le mot de comparaison~: l'eau du robinet est \dots\ à côté du
chocolat chaud, mais \dots\ à côté du jus d'orange sorti du
réfrigérateur.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:3}
Nomme quatre choses qui réchauffent ce qui les entoure. Laquelle
réchauffe le parc tout entier~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:4}
Frotte vite tes paumes l'une contre l'autre en comptant jusqu'à $10$,
puis touche ta joue. Que sens-tu~? Qu'a fait le frottement~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:5}
Deux glaçons sortent ensemble du congélateur. L'un est posé sur une
assiette au soleil, l'autre sur une assiette à l'ombre. Lequel fond
le premier, et pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:6}
Qu'y a-t-il dans la flaque laissée sur l'assiette après qu'un glaçon
a fondu~? Où est passée la glace dure~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:7}
Un toboggan en métal et un banc en bois sont restés à l'ombre toute
la matinée. Tom touche les deux et dit que le toboggan est plus
froid. Sers-toi de \cref{rem:g1:hot-and-cold:fooled} pour expliquer
ce qui se passe vraiment.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:8}
À l'aide de \cref{met:g1:hot-and-cold:compare}, explique comment
savoir lequel de deux bols d'eau est le plus chaud. Pourquoi
faut-il utiliser la même main pour les deux~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:hot-and-cold:9}
Léa fabrique un bonhomme de neige et, prise de pitié, elle
l'enveloppe dans son manteau bien chaud. Son frère dit que le manteau
le fera fondre plus vite. Léa dit que le manteau le gardera froid
plus longtemps. Qui a raison, et pourquoi~? (Pense à
\cref{rem:g1:hot-and-cold:coat}~: que fait vraiment un manteau~?)
\end{exercise}
