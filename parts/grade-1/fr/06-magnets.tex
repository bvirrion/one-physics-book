\chapter{Les aimants}\label{ch:g1:magnets}

Sur la porte du réfrigérateur, de petites figurines tiennent des
dessins sans colle, sans ruban adhésif, sans crochet. Décolles-en
une~: rien de collant au dos --- et pourtant elle saute en arrière et
tient bon. Ces petites figurines cachent un \omterm{def:g1:magnets:magnet}{aimant}, et les \omterm{def:g1:magnets:magnet}{aimants}
sont pleins de surprises qui valent tout un chapitre d'expériences.

\section{Ce qu'un aimant attrape}

\begin{definition}[Aimants]\label{def:g1:magnets:magnet}
Un \emph{aimant}\index{aimant} est un morceau de pierre ou de métal
un peu spécial, qui tire certains objets vers lui sans les toucher.
On dit qu'il les \emph{attire}\index{attirer}. Les objets qu'il
attire se collent à lui~; les autres ne s'occupent pas de lui du
tout.
\end{definition}

\begin{example}[À toi de jouer~: l'essai
d'attraction]\label{ex:g1:magnets:test}
Promène-toi dans la maison avec un \omterm{def:g1:magnets:magnet}{aimant} et essaie-le sur~: un
trombone, un clou, la porte du réfrigérateur, une cuillère en bois,
une brique de plastique, une feuille de papier, un verre, un bouchon
de liège. L'\omterm{def:g1:magnets:magnet}{aimant} attrape le trombone, le clou et la porte du
réfrigérateur --- et ignore tous les autres. Seuls certains objets
obéissent aux \omterm{def:g1:magnets:magnet}{aimants}.
\end{example}

\begin{example}[Le métal préféré de l'aimant]\label{ex:g1:magnets:iron}
Regarde ce qui s'est fait attraper~: le trombone, le clou, la porte
du réfrigérateur. Tous sont en \emph{fer} (ou en acier, qui est
surtout du fer). Le bois, le plastique, le papier, le verre, la
pierre~: jamais. Et voici la belle surprise~: même certains
\emph{métaux} sont ignorés --- essaie une pièce de monnaie ou une
feuille de papier d'aluminium. Un \omterm{def:g1:magnets:magnet}{aimant} n'aime pas les métaux~; il
aime le fer.
\end{example}

\section{La portée de l'aimant}

\begin{example}[À toi de jouer~: attraper à
travers]\label{ex:g1:magnets:through}
Pose un trombone sur la table et tiens ton \omterm{def:g1:magnets:magnet}{aimant} \emph{sous} le
plateau, juste au-dessous du trombone. Déplace l'\omterm{def:g1:magnets:magnet}{aimant}~: le trombone
glisse sur la table comme par magie --- l'attraction traverse le
bois. Cela marche aussi à travers le papier, le carton, un verre
d'eau~: laisse tomber un trombone dans le verre et fais-le remonter
le long de la paroi avec l'\omterm{def:g1:magnets:magnet}{aimant}, sans te mouiller les doigts.
\end{example}

\begin{remark}[Tirer sans toucher]\label{rem:g1:magnets:notouch}
Pense à quel point c'est étrange. Pour déplacer un jouet, il faut le
toucher~: le pousser, le tirer, lui donner un coup de pied. L'\omterm{def:g1:magnets:magnet}{aimant},
lui, déplace le trombone \emph{à travers l'air vide}, à travers le
bois, à travers l'eau, sans rien toucher. Bien peu de choses, dans la
nature, agissent sans toucher --- garde cet émerveillement dans un
coin de ta tête, car il reviendra, bien plus tard, comme l'une des
idées les plus profondes de ce cours.
\end{remark}

\begin{method}[La pêche à l'aimant]\label{met:g1:magnets:fishing}
Un jeu à deux, et un vrai test de la portée~:
\begin{enumerate}
  \item Attacher un \omterm{def:g1:magnets:magnet}{aimant} à une ficelle --- c'est la canne à pêche~;
  \item éparpiller des trombones sur le sol --- ce sont les
        poissons~;
  \item descendre l'\omterm{def:g1:magnets:magnet}{aimant} lentement, de très haut, au-dessus d'un
        trombone~;
  \item arrêter de descendre à l'instant où le trombone saute vers
        l'\omterm{def:g1:magnets:magnet}{aimant}.
\end{enumerate}
Le saut n'a lieu que lorsque l'\omterm{def:g1:magnets:magnet}{aimant} est proche~: l'attraction est
forte près de l'\omterm{def:g1:magnets:magnet}{aimant} et faiblit vite quand on s'éloigne.
\end{method}

\section{Deux aimants ensemble}

\begin{example}[À toi de jouer~: tirer ou
pousser~?]\label{ex:g1:magnets:twomagnets}
Prends deux \omterm{def:g1:magnets:magnet}{aimants}, un dans chaque main, et rapproche lentement
leurs extrémités. Parfois ils se jettent l'un sur l'autre et se
collent d'un coup. Retourne maintenant un des \omterm{def:g1:magnets:magnet}{aimants}, l'autre bout
en avant~: rapproche-les de nouveau, et cette fois tu sens une
\emph{poussée} douce et élastique --- les \omterm{def:g1:magnets:magnet}{aimants} refusent de se
toucher, comme si un coussin invisible se trouvait entre eux. Les
deux mêmes \omterm{def:g1:magnets:magnet}{aimants}~: dans un sens ils s'attirent, retournés ils se
repoussent.
\end{example}

\begin{remark}[Les deux extrémités]\label{rem:g1:magnets:ends}
Les deux extrémités d'un \omterm{def:g1:magnets:magnet}{aimant} ne sont donc pas pareilles, même
quand elles se ressemblent~! Chaque \omterm{def:g1:magnets:magnet}{aimant} a deux extrémités
différentes, et le jeu de l'attraction et de la répulsion dépend de
celles qui se rencontrent. Fais aussi l'essai des trombones~: ils
s'agglutinent aux deux bouts d'un \omterm{def:g1:magnets:magnet}{aimant} droit et ignorent à peu près
son milieu. Ces deux extrémités particulières ont un nom et un métier
célèbre --- guider les voyageurs à travers le monde --- mais cette
histoire attend une année plus avancée de ce livre.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:magnets:1}
Range en «~l'\omterm{def:g1:magnets:magnet}{aimant} l'attrape~» et «~l'\omterm{def:g1:magnets:magnet}{aimant} l'ignore~»~: un clou~;
\ un bouchon de liège~; \ un trombone~; \ un canard en plastique~;
\ une cuillère en acier~; \ une feuille de papier.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:2}
Quelle matière un \omterm{def:g1:magnets:magnet}{aimant} aime-t-il~? Nomme deux métaux ou matériaux
que l'\omterm{def:g1:magnets:magnet}{aimant} ignore malgré tout.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:3}
Un dessin tient sur la porte du réfrigérateur sous une figurine
aimantée. Qu'y a-t-il entre l'\omterm{def:g1:magnets:magnet}{aimant} et la porte~? Que montre cela
sur l'attraction de l'\omterm{def:g1:magnets:magnet}{aimant}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:4}
Un trombone tombe dans un verre plein d'eau. Comment le récupérer
avec un \omterm{def:g1:magnets:magnet}{aimant}, sans te mouiller les doigts~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:5}
Dans le jeu de la pêche à l'\omterm{def:g1:magnets:magnet}{aimant} de \cref{met:g1:magnets:fishing},
le trombone ne saute que lorsque l'\omterm{def:g1:magnets:magnet}{aimant} s'approche. Que t'apprend
cela sur l'attraction loin de l'\omterm{def:g1:magnets:magnet}{aimant}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:6}
Approche deux \omterm{def:g1:magnets:magnet}{aimants} bout à bout~: ils se collent d'un coup. Quel
seul geste fait que ces deux mêmes \omterm{def:g1:magnets:magnet}{aimants} se repoussent~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:7}
Où les trombones s'agglutinent-ils sur un \omterm{def:g1:magnets:magnet}{aimant} droit~: au milieu ou
aux extrémités~? Que dit cela de l'endroit où un \omterm{def:g1:magnets:magnet}{aimant} est le plus
fort~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:8}
Donne un point commun entre un \omterm{def:g1:magnets:magnet}{aimant} et ta main qui déplace un
jouet, et une différence totale. (Pense à
\cref{rem:g1:magnets:notouch}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:magnets:9}
Tom dit~: «~Les \omterm{def:g1:magnets:magnet}{aimants} attirent tout ce qui est en métal.~» Imagine
une petite expérience avec une pièce de monnaie, un clou et du papier
d'aluminium pour tester la phrase de Tom, dis ce qui va se passer ---
et corrige sa phrase.
\end{exercise}
