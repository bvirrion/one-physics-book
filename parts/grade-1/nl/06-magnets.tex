\chapter{Magneten}\label{ch:g1:magnets}

Op de koelkastdeur houden kleine figuurtjes tekeningen omhoog zonder
lijm, zonder plakband, zonder haakjes. Trek er een af: niets plakkerigs
aan de achterkant --- en toch springt het terug en houdt het vast. In
die figuurtjes zit een \omterm{def:g1:magnets:magnet}{magneet}, en magneten zitten vol verrassingen
die een heel hoofdstuk aan proefjes waard zijn.

\section{Wat een magneet pakt}

\begin{definition}[Magneten]\label{def:g1:magnets:magnet}
Een \emph{magneet}\index{magneet} is een bijzonder stuk steen of
metaal dat bepaalde dingen naar zich toe trekt zonder ze aan te raken.
We zeggen dat de magneet ze \emph{aantrekt}\index{aantrekken}. De
dingen die hij aantrekt blijven eraan hangen; de rest laat hem
volkomen koud.
\end{definition}

\begin{example}[Probeer het: de pakproef]\label{ex:g1:magnets:test}
Loop met een \omterm{def:g1:magnets:magnet}{magneet} door je huis en probeer hem uit op: een paperclip,
een spijker, de koelkastdeur, een houten lepel, een plastic
bouwsteentje, een vel papier, een glas, een kurk. De \omterm{def:g1:magnets:magnet}{magneet} pakt de
paperclip, de spijker en de koelkastdeur --- en negeert alle andere.
Alleen sommige dingen gehoorzamen aan magneten.
\end{example}

\begin{example}[Het lievelingsmetaal van de magneet]\label{ex:g1:magnets:iron}
Kijk naar wat er gepakt werd: paperclip, spijker, koelkastdeur. Ze
zijn allemaal van \emph{ijzer} gemaakt (of van staal, dat vooral uit
ijzer bestaat). Hout, plastic, papier, glas, steen: nooit. En hier
komt de mooie verrassing: zelfs sommige \emph{metalen} worden genegeerd
--- probeer maar eens een muntje of een velletje aluminiumfolie. Een
\omterm{def:g1:magnets:magnet}{magneet} houdt niet van metaal; hij houdt van ijzer.
\end{example}

\section{Hoe ver een magneet reikt}

\begin{example}[Probeer het: door dingen heen pakken]\label{ex:g1:magnets:through}
Leg een paperclip op tafel en houd je \omterm{def:g1:magnets:magnet}{magneet} \emph{onder} het
tafelblad, vlak onder de clip. Beweeg de \omterm{def:g1:magnets:magnet}{magneet}: de clip glijdt als
bij toverslag over de tafel --- de aantrekking gaat dwars door het
hout heen. Het lukt ook door papier, door karton en door een glas
water: laat een clip in het glas vallen en laat hem met de
\omterm{def:g1:magnets:magnet}{magneet} langs de wand omhoog lopen, zonder je vingers nat te maken.
\end{example}

\begin{remark}[Trekken zonder aanraken]\label{rem:g1:magnets:notouch}
Bedenk hoe vreemd dit is. Om speelgoed te verplaatsen moet je het
aanraken: duwen, trekken, schoppen. De \omterm{def:g1:magnets:magnet}{magneet} verplaatst de clip
\emph{dwars door lege lucht}, door hout, door water, zonder iets aan
te raken. Maar heel weinig dingen in de natuur kunnen werken zonder
aanraking --- bewaar dit wonder in een hoekje van je hoofd, want het
komt veel later terug als een van de diepste ideeën van deze cursus.
\end{remark}

\begin{method}[Magneetvissen]\label{met:g1:magnets:fishing}
Een spel voor twee, en een eerlijke proef op het bereik:
\begin{enumerate}
  \item Bind een \omterm{def:g1:magnets:magnet}{magneet} aan een touwtje --- dat is de hengel;
  \item strooi paperclips over de vloer --- dat zijn de vissen;
  \item laat de \omterm{def:g1:magnets:magnet}{magneet} van hoog boven een clip langzaam zakken;
  \item stop met zakken op het moment dat de clip omhoogspringt naar
        de \omterm{def:g1:magnets:magnet}{magneet}.
\end{enumerate}
De sprong komt pas als de \omterm{def:g1:magnets:magnet}{magneet} dichtbij is: de aantrekking is sterk
vlak bij de \omterm{def:g1:magnets:magnet}{magneet} en zwakt verderop snel af.
\end{method}

\section{Twee magneten samen}

\begin{example}[Probeer het: trekken of duwen?]\label{ex:g1:magnets:twomagnets}
Neem twee magneten, in elke hand één, en breng hun uiteinden langzaam
bij elkaar. Soms springen ze op elkaar af en klikken ze aan elkaar
vast. Draai nu één \omterm{def:g1:magnets:magnet}{magneet} om, met het andere uiteinde vooraan: breng
ze weer bij elkaar, en nu voel je een zachte, veerkrachtige
\emph{duw} --- de magneten weigeren elkaar aan te raken, alsof er een
onzichtbaar kussen tussen zit. Dezelfde twee magneten: zo trekken ze,
omgedraaid duwen ze.
\end{example}

\begin{remark}[De twee uiteinden]\label{rem:g1:magnets:ends}
De twee uiteinden van een \omterm{def:g1:magnets:magnet}{magneet} zijn dus niet hetzelfde, ook al zien
ze er eender uit! Elke \omterm{def:g1:magnets:magnet}{magneet} heeft twee verschillende uiteinden, en
het spel van trekken en duwen hangt af van \emph{welke} uiteinden
elkaar ontmoeten. Doe ook de paperclipproef: clips klitten samen aan
de twee uiteinden van een staafmagneet en negeren het midden
grotendeels. Die twee bijzondere uiteinden hebben namen en een beroemde
taak --- reizigers de weg wijzen over de hele wereld --- maar dat
verhaal wacht op een later jaar van dit boek.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:magnets:1}
Sorteer in ``de \omterm{def:g1:magnets:magnet}{magneet} pakt het'' en ``de \omterm{def:g1:magnets:magnet}{magneet} negeert het'': een
spijker; \ een kurk; \ een paperclip; \ een plastic eendje; \ een
stalen lepel; \ een vel papier.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:2}
Van welk materiaal houdt een \omterm{def:g1:magnets:magnet}{magneet}? Noem twee metalen of materialen
die de \omterm{def:g1:magnets:magnet}{magneet} toch negeert.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:3}
Een tekening hangt aan de koelkastdeur onder een magneetfiguurtje. Wat
zit er tussen de \omterm{def:g1:magnets:magnet}{magneet} en de deur? Wat laat dat zien over de
aantrekking van de \omterm{def:g1:magnets:magnet}{magneet}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:4}
Een paperclip valt in een glas vol water. Hoe krijg je hem er met een
\omterm{def:g1:magnets:magnet}{magneet} uit zonder je vingers nat te maken?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:5}
Bij het magneetvissen van \cref{met:g1:magnets:fishing} springt de clip
pas op als de \omterm{def:g1:magnets:magnet}{magneet} dichtbij komt. Wat vertelt dat je over de
aantrekking ver van de \omterm{def:g1:magnets:magnet}{magneet}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:6}
Breng twee magneten met de uiteinden naar elkaar toe: ze klikken aan
elkaar. Met welke ene beweging laat je diezelfde twee magneten juist
uit elkaar duwen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:7}
Waar klitten de paperclips op een staafmagneet samen: in het midden of
aan de uiteinden? Wat zegt dat over waar een \omterm{def:g1:magnets:magnet}{magneet} het sterkst is?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:8}
Noem één manier waarop een \omterm{def:g1:magnets:magnet}{magneet} lijkt op jouw hand die speelgoed
verschuift, en één manier waarop hij daar volkomen van verschilt.
(Denk aan \cref{rem:g1:magnets:notouch}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:magnets:9}
Tom zegt: ``Magneten trekken alles aan wat van metaal is.'' Bedenk
een proefje met een muntje, een spijker en aluminiumfolie om de zin
van Tom te toetsen, zeg wat er zal gebeuren --- en verbeter zijn zin.
\end{exercise}
