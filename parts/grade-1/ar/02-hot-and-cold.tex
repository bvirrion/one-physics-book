\chapter{الساخن والبارد}\label{ch:g1:hot-and-cold}

الحساء يتصاعد منه البخار، والمثلّجات تُوجع أسنانك، ومقعد السيارة في
الصيف يلسع ساقيك. الساخن والبارد في كل مكان. في هذا الفصل نتعلّم
كيف نقارن بينهما، وما الذي يُدفئ الأشياء --- ونراقب ما يحدث لمكعّب
الثلج حين ينتصر الدفء.

\section{أدفأ وأبرد}

\begin{definition}[الساخن والبارد]\label{def:g1:hot-and-cold:words}
\emph{ساخن}\index{ساخن} و\emph{بارد}\index{بارد} كلمتا مقارنة،
مثل كبير وصغير. فالحساء ساخن \emph{بالقياس إلى} يدك؛ والثلج بارد
\emph{بالقياس إلى} يدك. وبين الاثنين سلّم كامل من الكلمات: متجمّد،
بارد، منعش، فاتر، دافئ، ساخن، حارق.
\end{definition}

\begin{example}[ترتيب ثلاثة مشروبات]\label{ex:g1:hot-and-cold:drinks}
على الطاولة: شوكولاتة ساخنة، وكأس ماء من الصنبور، وكأس عصير برتقال
خارج من الثلاجة توًّا. من الأبرد إلى الأسخن: العصير، ثم ماء الصنبور،
ثم الشوكولاتة الساخنة. فماء الصنبور بارد إلى جانب الشوكولاتة ودافئ
إلى جانب العصير --- كلمات المقارنة تعمل دائمًا مثنى مثنى.
\end{example}

\begin{remark}[اللمس يُخدع]\label{rem:g1:hot-and-cold:fooled}
تذكّر أوعية الماء الثلاثة في الفصل الماضي: الماء الفاتر نفسه بدا
دافئًا ليد وباردًا للأخرى. والمِس ملعقة معدنية وملعقة خشبية باتتا
الليلة في المطبخ نفسه: المعدنية \emph{تبدو} أبرد، مع أنه ليس في
المطبخ شيء أبرد من شيء. اللمس حارس نافع، لكنه ليس قاضيًا عادلًا
--- وسنحتاج قريبًا إلى جهاز لا يُخدع: ميزان الحرارة، ويأتي في فصل
لاحق.
\end{remark}

\begin{method}[مقارنة عادلة بين الساخن والبارد]\label{met:g1:hot-and-cold:compare}
لتقرّر أيّ وعاءي ماء أدفأ:
\begin{enumerate}
  \item استعمل اليد \emph{نفسها} للاثنين --- فاليدان قد تختلفان؛
  \item المِس أحدهما ثم الآخر في الحال --- فالانتظار يغيّر
        الأشياء؛
  \item ولمزيد من اليقين، اطلب من صديق أن يجرّب أيضًا، من غير أن
        تخبره بجوابك قبل ذلك.
\end{enumerate}
إذا اتفقت أنت وصديقك كانت الملاحظة متينة.
\end{method}

\section{ما الذي يُدفئ الأشياء}

\begin{example}[أشياء تُدفئ]\label{ex:g1:hot-and-cold:warmers}
بعض الأشياء تُدفئ ما حولها: الشمس تُدفئ المقعد في الحديقة، واللهب
يُدفئ المقلاة، والمدفأة تُدفئ الغرفة، وجسمك أنت يُدفئ البطانية. بل
الدلك يُدفئ أيضًا: اضغط راحتيك إحداهما على الأخرى وادلكهما بسرعة
وأنت تعدّ حتى $10$ --- ستحسّ بهما تسخنان.
\end{example}

\begin{example}[جرّب: الشمس والظلّ]\label{ex:g1:hot-and-cold:sunshade}
في يوم مشمس، ضع إحدى يديك على جدار في الشمس والأخرى على الجدار
نفسه في الظلّ. الجدار نفسه واليوم نفسه --- ومع ذلك فالجهة المشمسة
أدفأ بوضوح. الشمس هي المدفئة الكبرى لكل ما في الخارج؛ وسيكون لها
فصل كامل قريبًا.
\end{example}

\begin{remark}[المعطف لا يصنع الحرارة]\label{rem:g1:hot-and-cold:coat}
المعطف لا يُدفئك كما تفعل المدفأة. \emph{جسمك} هو الذي يصنع
الدفء؛ والمعطف يمنع ذلك الدفء من الهرب فحسب، مثل غطاء على قِدر
الحساء. ولهذا يبقى المعطف الملقى وحده على كرسيّ باردًا من الداخل
--- فليس فيه ما يصنع دفئًا ليحفظه.
\end{remark}

\section{حين ينتصر الدفء: انصهار الثلج}

\begin{example}[حياة مكعّب ثلج]\label{ex:g1:hot-and-cold:icecube}
ضع مكعّب ثلج في صحن في المطبخ. إنه صلب بارد. فيدفئه هواء الغرفة
الدافئ شيئًا فشيئًا: يصغر المكعّب ويصغر، وتكبر بركة حوله، وبعد
حين لا يبقى إلا ماء. لقد حوّل دفء الغرفة الثلج الصلب إلى ماء
سائل --- ونقول إن الثلج \emph{انصهر}.
\end{example}

\begin{example}[البرد ينتصر أيضًا]\label{ex:g1:hot-and-cold:freezer}
المجمّدة تقوم بالعمل المعاكس: ضع فيها كوبًا صغيرًا من الماء في
المساء، وستجده في الصباح صلبًا كالحجر --- ثلجًا. الدفء يصهر
الثلج ماءً؛ والبرد الشديد يعيد الماء ثلجًا. وسيظلّ الماء يتقلّب
هكذا ذهابًا وإيابًا في الكتاب كله.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:1}
رتّب من الأبرد إلى الأسخن: ماء الحمّام؛ \ مكعّب ثلج؛ \ حساء
خارج من القِدر توًّا؛ \ هواء غرفة نومك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:2}
جِد كلمة المقارنة: ماء الصنبور \dots\ بالقياس إلى الشوكولاتة
الساخنة، لكنه \dots\ بالقياس إلى عصير البرتقال الخارج من
الثلاجة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:3}
سمِّ أربعة أشياء تُدفئ ما حولها. وأيّها يُدفئ الحديقة كلها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:4}
ادلك راحتيك بسرعة وأنت تعدّ حتى $10$، ثم المِس بهما خدّك. بمَ
تحسّ؟ وما الذي فعله الدلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:5}
خرج مكعّبا ثلج من المجمّدة معًا. فوُضع أحدهما في صحن في الشمس،
والآخر في صحن في الظلّ. أيّهما ينصهر أولًا، ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:6}
ما الذي في البركة الباقية في الصحن بعد انصهار مكعّب الثلج؟ وأين
ذهب الثلج الصلب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:7}
بقي مِزلج معدني ومقعد خشبي في الظلّ طوال الصباح. يلمس توم
الاثنين ويقول إن المِزلج أبرد. استعمل
\cref{rem:g1:hot-and-cold:fooled} لتشرح ما يجري حقًّا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:8}
باستعمال \cref{met:g1:hot-and-cold:compare}، اشرح كيف تعرف أيّ
وعاءي ماء أدفأ. ولماذا ينبغي أن تستعمل اليد نفسها للاثنين؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:hot-and-cold:9}
تبني ليا رجل ثلج، وإشفاقًا عليه تلفّه في معطفها الدافئ. يقول
أخوها إن المعطف سيصهره أسرع. وتقول ليا إن المعطف سيبقيه باردًا
مدة أطول. مَن المصيب، ولماذا؟ (فكّر في
\cref{rem:g1:hot-and-cold:coat}: ما الذي يفعله المعطف حقًّا؟)
\end{exercise}
