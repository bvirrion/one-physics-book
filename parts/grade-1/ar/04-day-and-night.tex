\chapter{النهار والليل: الشمس}\label{ch:g1:day-and-night}

في كل صباح تعود الشمس، كأنها لم تغب قطّ. تصعد، ثم تنزلق في الجهة
الأخرى من السماء، ثم تختفي --- فتتولّى مصابيح الشارع الأمر. نهار،
فليل، فنهار، فليل: أقدم إيقاع في العالم. فأين تذهب الشمس ليلًا؟
هيا نتبعها.

\section{رحلة الشمس}

\begin{example}[يوم واحد من أيام الشمس]\label{ex:g1:day-and-night:journey}
راقب موضع الشمس في السماء يومًا كاملًا (من الجانب، ولا تنظر إليها
مباشرة أبدًا!). في الصباح تكون منخفضة قرب سطوح البيوت، في جهة من
السماء. وتصعد على مدى الصباح. وعند الظهر تقف في أعلى نقطة. وبعد
الظهر تنزل في الجهة \emph{الأخرى}، وفي المساء تنزلق تحت السطوح
هناك. أما الجهة التي تشرق منها صباحًا فهي الجهة نفسها دائمًا ---
وغدًا ستشرق منها من جديد.
\end{example}

\begin{definition}[النهار والليل]\label{def:g1:day-and-night:daynight}
\emph{النهار}\index{نهار} هو الجزء من اليوم الذي تكون فيه الشمس
مرتفعة في سمائنا تمنحنا ضوءها. و\emph{الليل}\index{ليل} هو الجزء
الذي تغيب فيه الشمس عن سمائنا فيُظلم العالم من حولنا. واللحظة التي
تظهر فيها الشمس هي \emph{الشروق}\index{شروق}؛ واللحظة التي تختفي
فيها هي \emph{الغروب}\index{غروب}.
\end{definition}

\begin{remark}[مصباح النهار الأكبر]\label{rem:g1:day-and-night:lamp}
تذكّر منابع \omterm{def:g1:light-and-shadows:source}{الضوء} في الفصل الماضي: الشمس نهارًا هي المنبع الذي
يضيء العالم كله دفعة واحدة --- كل شارع، وكل ساحة لعب، وكل بحر. أما
ليلًا فنكتفي بمنابع صغيرة: مصابيح، ومصابيح يدوية، ونجوم. ولهذا
يبقى الليل مظلمًا وإن أُشعلت المصابيح: فلا مصباح صغير يحلّ محلّ
المصباح الأكبر.
\end{remark}

\begin{example}[الإيقاع يتكرّر]\label{ex:g1:day-and-night:rhythm}
صباح، فظهر، فمساء، فليل --- ثم صباح من جديد. ولم يتخلّف هذا
الإيقاع مرة واحدة: بعد كل ليل نهار، وبعد كل نهار ليل. ودورة كاملة
من هذه العجلة، نهار \emph{وليل} معًا، هي ما يعدّه التقويم:
الاثنين، والثلاثاء، والأربعاء، كل واحد منها دورة واحدة للعجلة.
\end{example}

\section{أين تذهب الشمس ليلًا؟}

الشمس ليلًا ما زالت مشرقة --- لكن لا علينا نحن. فبينما ننام يقع
ضوءها على أطفال في الجهة الأخرى من العالم، وهم في ظهيرتهم. الشمس
لا تنطفئ؛ بل \emph{نحن} من يستدير عنها.

\begin{example}[الأرض الدائرة]\label{ex:g1:day-and-night:spinning}
الأرض --- تلك الكرة الضخمة التي نعيش عليها جميعًا --- تدور على مهل
مثل دوّامة الملاهي، ونحن ندور معها. فحين تواجه جهتنا من الكرة
الشمسَ يكون عندنا نهار. وبينما تواصل الأرض دورانها تنزلق جهتنا
خارج \omterm{def:g1:light-and-shadows:source}{ضوء} الشمس إلى العتمة: فيأتي المساء ثم الليل. وواصِلِ الدوران،
فتعود جهتنا إلى \omterm{def:g1:light-and-shadows:source}{الضوء}: \omterm{def:g1:day-and-night:daynight}{الشروق}! والشمس لا \emph{تبدو} مسافرة عبر
سمائنا إلا ظاهرًا --- أما المسافر الحقيقي فالأرض تحت أقدامنا.
\end{example}

\begin{method}[جرّب: كن أنت الأرض]\label{met:g1:day-and-night:beearth}
تحتاج إلى مصباح في غرفة مظلمة، وإلى نفسك.
\begin{enumerate}
  \item المصباح هو الشمس؛ ورأسك هو الأرض؛ وأنفك هي مدينتك.
  \item واجه المصباح: \omterm{def:g1:light-and-shadows:source}{الضوء} على وجهك --- عند مدينتك نهار.
  \item استدر على مهل في مكانك. ينزلق \omterm{def:g1:light-and-shadows:source}{الضوء} عن خدّك: ذلك هو
        المساء.
  \item مؤخّرة رأسك نحو المصباح: وجهك في العتمة --- ليل في
        مدينتك.
  \item وواصِلِ الدوران حتى يعود \omterm{def:g1:light-and-shadows:source}{الضوء} إلى خدّك: \omterm{def:g1:day-and-night:daynight}{الشروق}!
\end{enumerate}
دورة كاملة منك هي نهار وليل كاملان لأنفك.
\end{method}

\begin{remark}[الشمس لا تنطفئ أبدًا]\label{rem:g1:day-and-night:neveroff}
في لعبة \cref{met:g1:day-and-night:beearth} بقي المصباح مشتعلًا
طوال الوقت --- ودورانك أنت وحده هو الذي جعل \omterm{def:g1:light-and-shadows:source}{الضوء} والعتمة يتناوبان
على وجهك. وكذلك الشمس الحقيقية: إنها تشرق طوال الليل، في الجهة
الأخرى من العالم. فالليل ليس راحةً للشمس؛ إنما هو جهتنا من الكرة
وقد نظرت إلى الجهة الأخرى.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:1}
رتّب ابتداءً من الصباح: الظهر؛ \ \omterm{def:g1:day-and-night:daynight}{الشروق}؛ \ الليل؛
\ \omterm{def:g1:day-and-night:daynight}{الغروب}؛ \ بعد الظهر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:2}
أين تكون الشمس عند الظهر: منخفضة قرب السطوح، أم في أعلى نقطة لها؟
وظلّك في تلك اللحظة --- طويل أم قصير؟ (فكّر في الفصل الماضي.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:3}
أتشرق الشمس كل صباح من الجهة نفسها من بيتك، أم من جهة مختلفة كل
يوم؟ كيف تتحقّق من ذلك غدًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:4}
صحيح أم غير صحيح: ``تنطفئ الشمس ليلًا.'' اشرح جوابك مستعينًا
بما في \cref{rem:g1:day-and-night:neveroff}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:5}
بينما تتناول فطورك في \omterm{def:g1:light-and-shadows:source}{ضوء} الصباح، ما الذي يفعله الأطفال في الجهة
المقابلة من الأرض على الأرجح؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:6}
العب \cref{met:g1:day-and-night:beearth} بمصباح. حين يقع \omterm{def:g1:light-and-shadows:source}{الضوء}
على خدّك الأيمن وأنت تواصل الدوران في الجهة نفسها، ماذا يأتي بعد
ذلك لأنفك: \omterm{def:g1:day-and-night:daynight}{الشروق} أم \omterm{def:g1:day-and-night:daynight}{الغروب}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:7}
أيّهما المسافر الحقيقي، الشمس عبر سمائنا أم الأرض تحت أقدامنا؟
قل ذلك بكلماتك أنت.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:day-and-night:8}
عُدّ بالتقويم: كم شروقًا في الأسبوع الواحد؟ وكم ليلة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:day-and-night:9}
تقول زوي: ``لو توقّفت الأرض عن الدوران لبقي في مدينتنا نهار إلى
الأبد.'' أيمكن أن تكون مصيبة --- وماذا سيقول الأطفال في الجهة
الأخرى من الأرض عن فكرتها؟
\end{exercise}
