\chapter{المغانط}\label{ch:g1:magnets}

على باب الثلاجة تحمل تماثيل صغيرة رسومًا بلا غراء ولا شريط لاصق
ولا خطّاف. انزع أحدها: لا شيء لزج على ظهره --- ومع ذلك يقفز راجعًا
ويتشبّث. تخبّئ هذه التماثيل الصغيرة مغناطيسًا، والمغانط مليئة
بالمفاجآت التي تستحقّ فصلًا كاملًا من التجارب.

\section{ما الذي يمسك به المغناطيس}

\begin{definition}[المغانط]\label{def:g1:magnets:magnet}
\emph{المغناطيس}\index{مغناطيس} قطعة خاصة من الحجر أو المعدن تسحب
أشياء بعينها نحوها من غير أن تلمسها. ونقول إن المغناطيس
\emph{يجذبها}\index{جذب}. فالأشياء التي يجذبها تلتصق به؛ وما عداها
لا يعبأ به البتة.
\end{definition}

\begin{example}[جرّب: اختبار الإمساك]\label{ex:g1:magnets:test}
تجوّل في بيتك ومعك \omterm{def:g1:magnets:magnet}{مغناطيس}، وجرّبه على: مشبك ورق، ومسمار، وباب
الثلاجة، وملعقة خشبية، وقالب بلاستيكي، وورقة، وكأس، وفلّينة. يمسك
\omterm{def:g1:magnets:magnet}{المغناطيس} بمشبك الورق والمسمار وباب الثلاجة --- ويهمل كل ما عداها.
فليست تطيع المغانطَ إلا بعض الأشياء.
\end{example}

\begin{example}[المعدن المفضّل عند المغناطيس]\label{ex:g1:magnets:iron}
انظر إلى ما أُمسك به: مشبك الورق، والمسمار، وباب الثلاجة. كلها
مصنوعة من \emph{الحديد} (أو من الفولاذ، وأكثره حديد). أما الخشب
والبلاستيك والورق والزجاج والحجر فلا، أبدًا. وهذه هي المفاجأة
اللطيفة: حتى بعض \emph{المعادن} يُهمَل --- جرّب قطعة نقود أو رقاقة
ألمنيوم من المطبخ. \omterm{def:g1:magnets:magnet}{فالمغناطيس} ليس محبًّا للمعادن؛ إنما هو محبّ
للحديد.
\end{example}

\section{مدى المغناطيس}

\begin{example}[جرّب: الجذب من خلال الأشياء]\label{ex:g1:magnets:through}
ضع مشبك ورق على الطاولة وامسك مغناطيسك \emph{تحت} سطح الطاولة،
أسفل المشبك مباشرة. حرّك \omterm{def:g1:magnets:magnet}{المغناطيس}: ينزلق المشبك على الطاولة كأنه
سحر --- فالجذب ينفذ من الخشب نفاذًا. وينفذ كذلك من الورق، ومن
الورق المقوّى، ومن كأس ماء: ألقِ مشبكًا في الكأس وامشِ به صاعدًا
على جدارها \omterm{def:g1:magnets:magnet}{بالمغناطيس}، وأصابعك جافة.
\end{example}

\begin{remark}[جذب بلا لمس]\label{rem:g1:magnets:notouch}
فكّر كم هذا غريب. فلكي تحرّك لعبة لا بدّ أن تلمسها: تدفعها أو
تسحبها أو تركلها. أما \omterm{def:g1:magnets:magnet}{المغناطيس} فيحرّك المشبك \emph{عبر الهواء
الخالي}، ومن خلال الخشب، ومن خلال الماء، ولا يلمس شيئًا. وقليل
جدًّا في الطبيعة ما يستطيع أن يفعل بلا لمس --- فاحتفظ بهذه
الأعجوبة في ركن من رأسك، فإنها ستعود بعد حين طويل بوصفها إحدى
أعمق الأفكار في هذا المنهاج.
\end{remark}

\begin{method}[الصيد بالمغناطيس]\label{met:g1:magnets:fishing}
لعبة لاثنين، واختبار عادل للمدى:
\begin{enumerate}
  \item اربط مغناطيسًا بخيط --- هذه هي صنّارة الصيد؛
  \item انثر مشابك ورق على الأرض --- هذه هي الأسماك؛
  \item أنزل \omterm{def:g1:magnets:magnet}{المغناطيس} على مهل من علوٍّ فوق أحد المشابك؛
  \item وقف الإنزال لحظة يقفز المشبك إلى \omterm{def:g1:magnets:magnet}{المغناطيس}.
\end{enumerate}
لا تقع القفزة إلا حين يقترب \omterm{def:g1:magnets:magnet}{المغناطيس}: فالجذب قوي قربه ويخبو
سريعًا كلما بَعُد.
\end{method}

\section{مغناطيسان معًا}

\begin{example}[جرّب: جذب أم دفع؟]\label{ex:g1:magnets:twomagnets}
خذ مغناطيسين، واحدًا في كل يد، وقرّب طرفيهما على مهل. تجدهما
أحيانًا يثبان أحدهما نحو الآخر ويلتصقان. والآن أدر أحد
المغناطيسين، وقدّم طرفه الآخر: قرّبهما من جديد، فتحسّ هذه المرة
\emph{دفعًا} ليّنًا مرنًا --- يرفض المغناطيسان أن يتلامسا، كأن
وسادة خفيّة بينهما. المغناطيسان نفسهما: في وضع يتجاذبان، وإذا
أُديرَ أحدهما تدافعا.
\end{example}

\begin{remark}[الطرفان]\label{rem:g1:magnets:ends}
إذن فطرفا \omterm{def:g1:magnets:magnet}{المغناطيس} ليسا سواءً، ولو تشابها في المنظر! فلكل
\omterm{def:g1:magnets:magnet}{مغناطيس} طرفان مختلفان، ولعبة الجذب والدفع تتوقّف على \emph{أيّ}
الطرفين يلتقيان. وجرّب اختبار مشابك الورق أيضًا: تتجمّع المشابك
عند طرفَي \omterm{def:g1:magnets:magnet}{المغناطيس} المستقيم وتهمل وسطه في الغالب. ولهذين الطرفين
الخاصّين اسمان وعمل مشهور --- هداية المسافرين عبر العالم --- لكن
تلك القصة تنتظر سنة لاحقة من هذا الكتاب.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:magnets:1}
صنّف إلى ``يمسك به \omterm{def:g1:magnets:magnet}{المغناطيس}'' و``يهمله \omterm{def:g1:magnets:magnet}{المغناطيس}'': مسمار؛
\ فلّينة؛ \ مشبك ورق؛ \ بطّة بلاستيكية؛ \ ملعقة فولاذية؛ \ ورقة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:2}
أيّ مادة يحبّها \omterm{def:g1:magnets:magnet}{المغناطيس}؟ وسمِّ معدنين أو مادتين يهملهما
\omterm{def:g1:magnets:magnet}{المغناطيس} مع ذلك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:3}
رسم متشبّث بباب الثلاجة تحت تمثال مغناطيسي. ما الذي بين \omterm{def:g1:magnets:magnet}{المغناطيس}
والباب؟ وماذا يبيّن هذا عن جذب \omterm{def:g1:magnets:magnet}{المغناطيس}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:4}
سقط مشبك ورق في كأس مملوءة بالماء. كيف تخرجه \omterm{def:g1:magnets:magnet}{بمغناطيس} من غير أن
تبلّل أصابعك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:5}
في لعبة الصيد \omterm{def:g1:magnets:magnet}{بالمغناطيس} في \cref{met:g1:magnets:fishing} لا يقفز
المشبك إلا حين يقترب \omterm{def:g1:magnets:magnet}{المغناطيس}. فماذا يخبرك هذا عن الجذب بعيدًا
عن \omterm{def:g1:magnets:magnet}{المغناطيس}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:6}
قرّب مغناطيسين طرفًا إلى طرف: يلتصقان. فأيّ حركة واحدة تجعل
المغناطيسين نفسيهما يتدافعان بدل ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:7}
أين تتجمّع مشابك الورق على \omterm{def:g1:magnets:magnet}{مغناطيس} مستقيم: في الوسط أم عند
الطرفين؟ وماذا يقول هذا عن الموضع الذي يكون فيه \omterm{def:g1:magnets:magnet}{المغناطيس} أقوى؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:magnets:8}
اذكر وجهًا يشبه فيه المغناطيسُ يدَك وهي تحرّك لعبة، ووجهًا يختلف
فيه عنها اختلافًا تامًّا. (فكّر في
\cref{rem:g1:magnets:notouch}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:magnets:9}
يقول توم: ``تجذب المغانط كل ما صُنع من معدن.'' صمّم تجربة صغيرة
بقطعة نقود ومسمار ورقاقة ألمنيوم لاختبار جملة توم، وقل ماذا
سيحدث --- ثم صحّح جملته.
\end{exercise}
