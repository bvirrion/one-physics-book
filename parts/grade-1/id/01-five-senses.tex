\chapter{Mengamati Dunia: Lima Indra}\label{ch:g1:five-senses}

Seekor burung berkicau di suatu tempat di atas kepalamu. Roti berbau
hangat. Lantai terasa dingin menembus kaus kakimu. Sebelum melakukan
apa pun, fisika melihat, mendengarkan, dan meraba: bab ini tentang
\omterm{def:g1:five-senses:observation}{mengamati} dunia dengan lima indra --- dan tentang tipuan yang kadang
mereka mainkan pada kita.

\section{Lima indra}

\begin{definition}[Lima indra]\label{def:g1:five-senses:senses}
Suatu \emph{indra}\index{indra} adalah cara tubuh mencari tahu tentang
dunia di sekitarnya. Kita punya lima:
\begin{enumerate}
  \item \emph{penglihatan}\index{penglihatan} --- mata melihat cahaya,
        warna, dan bentuk;
  \item \emph{pendengaran}\index{pendengaran} --- telinga mendengar
        bunyi;
  \item \emph{perabaan}\index{perabaan} --- kulit merasakan kasar atau
        halus, hangat atau dingin;
  \item \emph{penciuman}\index{penciuman} --- hidung mencium bau roti
        hangat;
  \item \emph{pengecapan}\index{pengecapan} --- lidah mengecap manis,
        asin, masam.
\end{enumerate}
\end{definition}

\begin{example}[Satu apel, lima indra]\label{ex:g1:five-senses:apple}
Memakan sebuah apel memakai semua indra: kamu \emph{melihat} kulitnya
yang merah, kamu \emph{merasakan} apel itu halus dan bulat di
tanganmu, kamu \emph{mendengar} bunyi kriuknya, kamu \emph{mencium}
harumnya, kamu \emph{mengecap} manis airnya. Lima indra, satu apel.
\end{example}

\begin{example}[Cobalah: jalan-jalan bunyi]\label{ex:g1:five-senses:sounds}
Duduklah diam, pejamkan matamu, dan hitunglah bunyi di sekitarmu
selama beberapa saat: sebuah mobil, sebuah suara, angin, napasmu
sendiri. Dengan mata terpejam, telinga menangkap bunyi yang tadi
terlewat karena mata sedang sibuk. Kebanyakan orang menemukan $4$ atau
$5$ bunyi yang sebelumnya tidak mereka sadari.
\end{example}

\begin{remark}[Indra yang menjaga kita]\label{rem:g1:five-senses:danger}
Indra juga penjaga. Hidung mencium asap sebelum mata melihat api;
kulit menarik tangan menjauh dari wajan panas; rasa pahit
memperingatkan ``jangan telan ini''. Jangan pernah sengaja mencicipi
atau mencium sesuatu yang tidak kamu kenal: tanyalah kepada orang
dewasa lebih dahulu. Penjaga bekerja paling baik kalau kita
memperlakukannya dengan hati-hati.
\end{remark}

\section{Mengamati dengan cermat}

Melihat belum tentu \omterm{def:g1:five-senses:observation}{mengamati}. Semua orang pernah \emph{melihat}
kepik; siapa yang bisa mengatakan berapa jumlah kakinya?

\begin{definition}[Pengamatan]\label{def:g1:five-senses:observation}
Kata \emph{mengamati}\index{mengamati} berarti memakai indra dengan
cermat dan sengaja untuk menjawab sebuah pertanyaan --- lalu
mengatakan atau menggambarkan tepat apa yang terlihat, bukan apa yang
diharapkan. Sebuah \emph{pengamatan}\index{pengamatan} adalah apa yang
berhasil ditemukan dengan melihat secara cermat.
\end{definition}

\begin{method}[Cara mengamati]\label{met:g1:five-senses:how}
\begin{enumerate}
  \item Pilihlah sebuah pertanyaan: \emph{berapa jumlah kaki
        kepik?}
  \item Amatilah (atau dengarkan, atau rabalah) pelan-pelan, tanpa
        tergesa-gesa;
  \item katakan apa yang kamu perhatikan dengan kata yang tepat: bukan
        ``benda itu kecil'', melainkan ``benda itu lebih kecil daripada
        kukuku'';
  \item gambarlah atau ceritakan, supaya orang lain dapat memeriksanya.
\end{enumerate}
\omterm{def:g1:five-senses:observation}{Pengamatan} yang baik dapat diperiksa oleh seorang teman: kalau kalian
berdua melihat, kalian berdua semestinya menemukan $6$ kaki.
\end{method}

\begin{example}[Kata yang tepat]\label{ex:g1:five-senses:words}
Bandingkan ``batu itu bagus'' dengan ``batu itu abu-abu, halus,
bulat, dan lebih dingin daripada tanganku''. Kalimat kedua adalah
sebuah \omterm{def:g1:five-senses:observation}{pengamatan}: anak lain dapat memilih batu itu dari satu
keranjang penuh.
\end{example}

\section{Ketika indra tertipu}

Indra memang hebat, tetapi bisa ditipu. Justru karena itulah, pada
bab-bab berikutnya kita akan belajar \emph{mengukur}.

\begin{example}[Dua garis]\label{ex:g1:five-senses:lines}
Pada gambar di bawah, garis mana yang lebih panjang? Hampir semua
orang menjawab ``yang atas''. Sekarang periksalah dengan seutas tali:
kedua garis itu sama panjang. Mata tertipu oleh ujung-ujung
panahnya.
\end{example}

\begin{example}[Cobalah: tiga mangkuk]\label{ex:g1:five-senses:bowls}
Mintalah orang dewasa menyiapkan tiga mangkuk air: satu sejuk, satu
suam-suam kuku, satu hangat yang nyaman. Rendam satu tangan di
mangkuk sejuk dan tangan yang lain di mangkuk hangat sambil kamu
menghitung pelan sampai $20$. Lalu masukkan kedua tangan bersama-sama
ke mangkuk tengah: air yang sama terasa hangat bagi tangan yang satu
dan sejuk bagi tangan yang lain! Perabaan saja tidak bisa dipercaya
soal panas dan dingin --- bab berikutnya membahas persoalan ini.
\end{example}

\begin{remark}[Pembantu indra]\label{rem:g1:five-senses:tools}
Orang membuat alat yang membantu indra: kacamata dan kaca pembesar
membantu mata, stetoskop dokter membantu telinga. Nanti dalam buku
ini, alat yang disebut \emph{instrumen} akan bekerja lebih baik lagi:
alat itu mengubah melihat dan meraba menjadi menghitung dan mengukur,
dan hitungan jauh lebih sukar ditipu.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:1}
Sebutkan lima indra, dan untuk masing-masing sebutkan bagian tubuh
yang bekerja.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:2}
Indra mana yang memberitahumu: (a) bahwa radio menyala? (b) bahwa sup
terlalu asin? (c) bahwa bajumu terasa gatal? (d) bahwa lampu padam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:3}
Sebuah kue sedang dipanggang di oven, pintunya tertutup, di ruangan
lain. Indra mana yang menemukannya lebih dahulu? Indra mana saja yang
kamu pakai ketika akhirnya kamu mendapat sepotong?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:4}
Lakukan jalan-jalan bunyi pada \cref{ex:g1:five-senses:sounds}: mata
terpejam, hitunglah bunyi di sekitarmu. Berapa banyak yang kamu
dengar? Bunyi mana yang mengejutkanmu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:5}
Berikan sebuah \omterm{def:g1:five-senses:observation}{pengamatan} tentang sepatumu dengan tiga kata yang
tepat, seperti pada \cref{ex:g1:five-senses:words}. ``Bagus'' dan
``cantik'' tidak boleh!
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:6}
Ana berkata: ``Kucing itu lembut.'' Bob berkata: ``Kucing itu kucing
terbaik.'' Kalimat mana yang merupakan \omterm{def:g1:five-senses:observation}{pengamatan}, dan indra mana yang
membuatnya? Kalimat yang satu lagi itu apa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:7}
Indra mana yang memperingatkanmu pada tiap bahaya: (a) asap di suatu
tempat di dalam rumah; (b) wajan yang terlalu panas untuk dipegang;
(c) mobil yang datang dari belakangmu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:five-senses:8}
Seorang teman memejamkan mata. Kamu meletakkan sebuah sendok, sebuah
spons, dan sebuah kerikil di depannya, satu per satu. Dapatkah dia
menyebut ketiganya hanya dengan meraba? Mana yang paling mudah
dikenali?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:five-senses:9}
Lihatlah dua garis pada \cref{ex:g1:five-senses:lines}. Jelaskan
kepada seorang teman cara memastikan garis mana yang lebih panjang,
tanpa memercayai mata. Dapatkah caramu juga membandingkan sebuah garis
dengan seutas tali yang berkelok?
\end{exercise}
