\chapter{Terapung dan Tenggelam}\label{ch:g1:floating-and-sinking}

Jatuhkan sebutir kerikil ke genangan air: ia lenyap sampai ke dasar.
Jatuhkan selembar daun: ia menumpang di permukaan seperti perahu
kecil. Waktu mandi, genangan air, dan kolam sama-sama mengajukan satu
pertanyaan --- apa yang \omterm{def:g1:floating-and-sinking:words}{terapung}, apa yang \omterm{def:g1:floating-and-sinking:words}{tenggelam}, dan dapatkah
kita menebaknya sebelum menjatuhkannya?

\section{Terapung atau tenggelam?}

\begin{definition}[Terapung dan tenggelam]\label{def:g1:floating-and-sinking:words}
Sebuah benda \emph{terapung}\index{terapung} apabila air menahannya di
permukaan; benda itu \emph{tenggelam}\index{tenggelam} apabila ia
turun sampai ke dasar. Gabus terapung; kerikil tenggelam.
\end{definition}

\begin{method}[Uji apung yang adil]\label{met:g1:floating-and-sinking:test}
\begin{enumerate}
  \item Isilah baskom besar (atau bak cuci) dengan air;
  \item letakkan bendanya di atas air dengan \emph{pelan} --- jangan
        dilempar: cipratan dapat menenggelamkan benda \omterm{def:g1:floating-and-sinking:words}{terapung} sesaat;
  \item lepaskan sepenuhnya, dan tunggulah sambil menghitung sampai $5$;
  \item lihatlah: yang di permukaan \omterm{def:g1:floating-and-sinking:words}{terapung}, yang di dasar \omterm{def:g1:floating-and-sinking:words}{tenggelam}.
\end{enumerate}
Ujilah setiap benda dengan cara yang sama, supaya perbandingannya adil.
\end{method}

\begin{example}[Cobalah: permainan memilah]\label{ex:g1:floating-and-sinking:sorting}
Kumpulkan gabus, koin, sendok plastik, sendok logam,
kerikil, pensil, dan daun kering, lalu ujilah satu per satu dengan
\cref{met:g1:floating-and-sinking:test}. Yang \omterm{def:g1:floating-and-sinking:words}{terapung}: gabus, sendok
plastik, pensil, daun. Yang \omterm{def:g1:floating-and-sinking:words}{tenggelam}: koin, sendok logam, kerikil.
Sebelum setiap benda dijatuhkan, ucapkan tebakanmu dengan lantang ---
itulah yang disebut para ilmuwan sebagai \emph{ramalan}, dan salah
tebak adalah separuh dari keseruannya.
\end{example}

\section{Berat bukan seluruh ceritanya}

Banyak orang menebak ``benda berat \omterm{def:g1:floating-and-sinking:words}{tenggelam}, benda ringan \omterm{def:g1:floating-and-sinking:words}{terapung}''.
Air menyimpan kejutan bagi mereka.

\begin{example}[Batang kayu dan koin]\label{ex:g1:floating-and-sinking:log}
Batang pohon yang tumbang jauh terlalu berat untuk kamu angkat ---
tetapi ia hanyut \omterm{def:g1:floating-and-sinking:words}{terapung} di sungai. Sekeping koin kecil lebih ringan
daripada sepotong biskuit --- tetapi ia langsung \omterm{def:g1:floating-and-sinking:words}{tenggelam}. Jadi
berat-atau-ringan tidak mungkin menjadi seluruh aturannya: batang
besar \omterm{def:g1:floating-and-sinking:words}{terapung} sementara koin mungil \omterm{def:g1:floating-and-sinking:words}{tenggelam}.
\end{example}

\begin{example}[Bahannyalah yang menentukan]\label{ex:g1:floating-and-sinking:stuff}
Lihatlah kembali permainan memilah: sendok plastik \omterm{def:g1:floating-and-sinking:words}{terapung} dan sendok
logam \omterm{def:g1:floating-and-sinking:words}{tenggelam}, padahal bentuk dan tugasnya sama. Yang membuat
berbeda adalah \emph{bahan} pembuatnya --- kayu, gabus, dan kebanyakan
plastik adalah bahan yang \omterm{def:g1:floating-and-sinking:words}{terapung}; logam, batu, dan kaca adalah bahan
yang \omterm{def:g1:floating-and-sinking:words}{tenggelam}. Sebuah benda biasanya berperilaku seperti bahannya:
kayu \omterm{def:g1:floating-and-sinking:words}{terapung}, entah berupa ranting kecil atau batang utuh.
\end{example}

\begin{remark}[Sedikit teka-teki]\label{rem:g1:floating-and-sinking:why}
\emph{Mengapa} kayu \omterm{def:g1:floating-and-sinking:words}{terapung} dan logam \omterm{def:g1:floating-and-sinking:words}{tenggelam}? Itu pertanyaan yang
sungguh-sungguh, dan jawaban lengkapnya adalah salah satu harta
pelajaran ini --- jawabannya memerlukan gagasan yang baru datang
beberapa tahun lagi. Untuk sekarang kita mengumpulkan pengamatan yang
baik; ``mengapa''-nya akan menemukan kita dalam keadaan siap.
\end{remark}

\section{Air mendorong balik}

\begin{example}[Cobalah: tekan bolanya ke bawah]\label{ex:g1:floating-and-sinking:push}
Ambillah bola yang \omterm{def:g1:floating-and-sinking:words}{terapung} di bak mandi dan tekanlah pelan-pelan ke
dalam air dengan telapak tanganmu. Terasa? Air mendorongnya balik ke
atas, makin kuat makin dalam kamu menekan --- dan kalau tanganmu
terpeleset, bola itu melompat keluar dari air. Air bukan sekadar
\emph{membiarkan} benda \omterm{def:g1:floating-and-sinking:words}{terapung} duduk di permukaan: air benar-benar
menahannya, seperti tanganmu menggendong bayi.
\end{example}

\begin{example}[Perahu dari plastisin]\label{ex:g1:floating-and-sinking:boat}
Bulatkan segumpal plastisin menjadi bola dan jatuhkan ke dalam air: ia
\omterm{def:g1:floating-and-sinking:words}{tenggelam} --- plastisin adalah bahan yang \omterm{def:g1:floating-and-sinking:words}{tenggelam}. Sekarang ambillah
gumpalan yang \emph{sama} dan bentuklah menjadi mangkuk kecil bertepi
tinggi, lalu letakkan pelan-pelan di atas air: ia \omterm{def:g1:floating-and-sinking:words}{terapung}! Bahan yang
sama, gumpalan yang sama --- hanya bentuknya yang berubah. Bentuk yang
lebar dan berongga memberi air lebih banyak tempat untuk mendorong ke
atas. Itulah rahasia kapal-kapal besar: logam adalah bahan yang
\omterm{def:g1:floating-and-sinking:words}{tenggelam}, tetapi lambung logam dibentuk seperti mangkuk raksasa yang
berongga.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:1}
Kelompokkan menjadi yang \omterm{def:g1:floating-and-sinking:words}{terapung} dan yang \omterm{def:g1:floating-and-sinking:words}{tenggelam}: gabus; \ kunci;
\ ranting kering; \ kelereng; \ bebek plastik; \ batu.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:2}
Pada \cref{met:g1:floating-and-sinking:test}, mengapa kamu harus
meletakkan bendanya pelan-pelan di atas air dan bukan melemparnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:3}
Sebelum menguji, Nora menebak: ``Sendok logam akan \omterm{def:g1:floating-and-sinking:words}{terapung} karena
sendok memang \omterm{def:g1:floating-and-sinking:words}{terapung}.'' Apa yang terjadi dalam permainan memilah?
Bukan tugas bentuknya --- apa yang harus dilihat Nora: bahannya atau
namanya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:4}
Sebutkan satu benda yang berat dan \omterm{def:g1:floating-and-sinking:words}{terapung}, dan satu benda yang
ringan dan \omterm{def:g1:floating-and-sinking:words}{tenggelam}. Apa yang ditunjukkan ini tentang tebakan ``benda
berat \omterm{def:g1:floating-and-sinking:words}{tenggelam}''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:5}
Sebuah ranting \omterm{def:g1:floating-and-sinking:words}{terapung}. Apakah satu batang pohon utuh juga akan
\omterm{def:g1:floating-and-sinking:words}{terapung}? Contoh mana dalam bab ini yang membantumu menjawab?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:6}
Tekanlah bola yang \omterm{def:g1:floating-and-sinking:words}{terapung} ke dalam air di bak mandi lalu lepaskan.
Apa yang kamu rasakan selagi menekan, dan apa yang dilakukan bola itu
ketika kamu lepaskan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:7}
Kapal dibuat dari logam, dan logam adalah bahan yang \omterm{def:g1:floating-and-sinking:words}{tenggelam}. Bentuk
apa yang menyelamatkan kapal? Percobaan mana dalam bab ini yang
menunjukkannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:floating-and-sinking:8}
Ujilah waktu mandi: apakah botol air yang penuh dan tertutup \omterm{def:g1:floating-and-sinking:words}{terapung}
atau \omterm{def:g1:floating-and-sinking:words}{tenggelam}? Dan botol yang sama dalam keadaan kosong, dengan tutup
terpasang? Ucapkan dulu tebakanmu, baru cobalah.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:floating-and-sinking:9}
Perahu plastisin Leo \omterm{def:g1:floating-and-sinking:words}{terapung} dengan bagus --- sampai ia memencetnya
kembali menjadi bola, dan bola itu \omterm{def:g1:floating-and-sinking:words}{tenggelam}. ``Airnya lelah
menahannya,'' kata Leo. Berikan penjelasan yang lebih baik, dengan
\cref{ex:g1:floating-and-sinking:boat}.
\end{exercise}
