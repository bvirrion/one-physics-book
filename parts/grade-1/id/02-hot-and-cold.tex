\chapter{Panas dan Dingin}\label{ch:g1:hot-and-cold}

Sup mengepulkan uap, es krim membuat gigimu ngilu, dan jok mobil pada
musim panas menyengat kakimu. Panas dan dingin ada di mana-mana. Pada
bab ini kita belajar membandingkan keduanya, melihat apa yang membuat
benda menjadi hangat --- dan mengamati apa yang terjadi pada sebongkah
es ketika kehangatan itu menang.

\section{Lebih hangat dan lebih dingin}

\begin{definition}[Panas dan dingin]\label{def:g1:hot-and-cold:words}
Kata \emph{panas}\index{panas} dan \emph{dingin}\index{dingin} adalah
kata pembanding, seperti besar dan kecil. Sup terasa panas
\emph{dibandingkan dengan} tanganmu; es terasa dingin
\emph{dibandingkan dengan} tanganmu. Di antara keduanya ada seluruh
tangga kata: beku, dingin, sejuk, suam-suam kuku, hangat, panas,
membakar.
\end{definition}

\begin{example}[Mengurutkan tiga minuman]\label{ex:g1:hot-and-cold:drinks}
Di atas meja: secangkir cokelat panas, segelas air keran, dan segelas
jus jeruk langsung dari lemari es. Dari yang paling dingin ke yang
paling panas: jus, lalu air keran, lalu cokelat panas. Air keran
terasa dingin di sebelah cokelat, tetapi hangat di sebelah jus ---
kata pembanding selalu bekerja berpasangan.
\end{example}

\begin{remark}[Perabaan bisa tertipu]\label{rem:g1:hot-and-cold:fooled}
Ingat tiga mangkuk pada bab lalu: air suam-suam kuku yang sama terasa
hangat bagi satu tangan dan sejuk bagi tangan yang lain. Cobalah juga
memegang sendok logam dan sendok kayu yang semalaman berada di dapur
yang sama: sendok logam \emph{terasa} lebih dingin, padahal tidak ada
benda di dapur itu yang lebih dingin daripada benda lain. Perabaan
memang penjaga yang berguna, tetapi bukan hakim yang adil --- suatu
saat nanti kita memerlukan alat yang tidak bisa ditipu: termometer,
yang akan kita temui pada bab kemudian.
\end{remark}

\begin{method}[Membandingkan panas dan dingin secara adil]\label{met:g1:hot-and-cold:compare}
Untuk memutuskan mangkuk air mana yang lebih hangat:
\begin{enumerate}
  \item pakailah tangan yang \emph{sama} untuk keduanya --- dua tangan
        bisa berbeda pendapat;
  \item sentuh yang satu, lalu yang lain, segera --- menunggu membuat
        keadaannya berubah;
  \item agar lebih yakin, mintalah seorang teman mencoba juga, tanpa
        memberitahukan jawabanmu lebih dahulu.
\end{enumerate}
Kalau kamu dan temanmu sependapat, pengamatan itu kukuh.
\end{method}

\section{Apa yang membuat benda menjadi hangat}

\begin{example}[Para penghangat]\label{ex:g1:hot-and-cold:warmers}
Ada benda yang menghangatkan apa saja di dekatnya: Matahari
menghangatkan bangku di taman, nyala api menghangatkan wajan, radiator
menghangatkan ruangan, dan tubuhmu sendiri menghangatkan selimut.
Menggosok pun menghangatkan: tempelkan kedua telapak tanganmu lalu
gosokkan cepat-cepat sambil menghitung sampai $10$ --- rasakan
keduanya memanas.
\end{example}

\begin{example}[Cobalah: matahari dan tempat teduh]\label{ex:g1:hot-and-cold:sunshade}
Pada hari yang cerah, letakkan satu tangan pada dinding yang kena
sinar matahari dan tangan yang lain pada dinding yang sama di tempat
teduh. Dinding yang sama, hari yang sama --- tetapi sisi yang kena
matahari jelas lebih hangat. Matahari adalah penghangat besar bagi
segala sesuatu di luar rumah; ia akan mendapat satu bab tersendiri tak
lama lagi.
\end{example}

\begin{remark}[Mantel tidak membuat panas]\label{rem:g1:hot-and-cold:coat}
Sebuah mantel tidak menghangatkanmu seperti radiator. \emph{Tubuhmu}
sendirilah yang membuat kehangatan itu; mantel hanya menahan
kehangatan itu agar tidak lolos, seperti tutup pada panci sup. Itulah
sebabnya mantel yang tergeletak sendirian di kursi tetap dingin di
dalamnya --- tidak ada apa pun di dalamnya yang membuat kehangatan
untuk ditahan.
\end{remark}

\section{Ketika kehangatan menang: es yang mencair}

\begin{example}[Riwayat sebongkah es]\label{ex:g1:hot-and-cold:icecube}
Letakkan sebongkah es di atas piring di dapur. Es itu keras dan
dingin. Udara hangat di ruangan menghangatkannya sedikit demi sedikit:
bongkahan itu makin kecil, genangan tumbuh di sekelilingnya, dan
setelah beberapa waktu hanya air yang tersisa. Kehangatan ruangan
mengubah es yang keras menjadi air --- kita katakan es itu
\emph{mencair}.
\end{example}

\begin{example}[Dingin pun bisa menang]\label{ex:g1:hot-and-cold:freezer}
Lemari beku melakukan pekerjaan sebaliknya: masukkan secangkir kecil
air ke dalamnya pada malam hari, dan pada pagi hari kamu akan
menemukannya keras seperti batu --- es. Kehangatan mencairkan es
menjadi air; dingin yang kuat mengubah air kembali menjadi es. Air
akan terus berubah bolak-balik seperti ini sepanjang buku ini.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:1}
Urutkan dari yang paling dingin ke yang paling panas: air mandi; \ sebongkah
es; \ sup yang baru diangkat dari panci; \ udara di kamar tidurmu.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:2}
Carilah kata pembandingnya: air keran itu \dots\ dibandingkan dengan
cokelat panas, tetapi \dots\ dibandingkan dengan jus jeruk dari lemari
es.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:3}
Sebutkan empat benda yang menghangatkan apa yang ada di dekatnya. Yang
mana yang menghangatkan seluruh taman?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:4}
Gosokkan kedua telapak tanganmu cepat-cepat sambil menghitung sampai
$10$, lalu sentuhkan ke pipimu. Apa yang kamu rasakan? Apa yang
dilakukan gosokan tadi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:5}
Dua bongkah es dikeluarkan dari lemari beku bersama-sama. Yang satu
diletakkan di piring di bawah matahari, yang lain di piring di tempat
teduh. Yang mana yang mencair lebih dahulu, dan mengapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:6}
Apa isi genangan yang tertinggal di piring setelah sebongkah es
mencair? Ke mana perginya es yang keras itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:7}
Perosotan logam dan bangku kayu berada di tempat teduh sepanjang pagi.
Tom memegang keduanya dan berkata perosotan itu lebih dingin. Pakailah
\cref{rem:g1:hot-and-cold:fooled} untuk menjelaskan apa yang sebenarnya
terjadi.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:hot-and-cold:8}
Dengan \cref{met:g1:hot-and-cold:compare}, jelaskan cara mengetahui
mangkuk air mana dari dua mangkuk yang lebih hangat. Mengapa kamu
harus memakai tangan yang sama untuk keduanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:hot-and-cold:9}
Lea membuat manusia salju dan, karena kasihan, membungkusnya dengan
mantel hangatnya. Kakaknya berkata mantel itu akan membuatnya mencair
lebih cepat. Lea berkata mantel itu akan membuatnya tetap dingin lebih
lama. Siapa yang benar, dan mengapa? (Ingat
\cref{rem:g1:hot-and-cold:coat}: apa sebenarnya kerja sebuah mantel?)
\end{exercise}
