\section*{Bab \ref{ch:g1:hot-and-cold} --- Panas dan Dingin}

\begin{solution}{exo:g1:hot-and-cold:1}
Bongkahan es, lalu udara kamar tidur, lalu air mandi, lalu sup yang
baru diangkat dari panci.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:2}
Air keran itu \emph{dingin} (atau lebih sejuk) dibandingkan dengan
cokelat panas, tetapi \emph{hangat} (atau lebih hangat) dibandingkan
dengan jus.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:3}
Misalnya: Matahari, nyala api, radiator, tubuhmu sendiri (atau
menggosokkan tangan). Matahari menghangatkan seluruh taman.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:4}
Telapak tangan terasa hangat di pipi: gosokan tadi menghangatkannya.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:5}
Bongkahan yang di bawah matahari mencair lebih dahulu: Matahari
menghangatkannya, dan kehangatanlah yang mencairkan es. Yang di tempat
teduh menerima lebih sedikit kehangatan dan bertahan lebih lama.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:6}
Genangan itu air --- air yang sama yang tadinya menjadi bongkahan es.
Es yang keras tidak lenyap; kehangatan ruangan mencairkannya menjadi
air.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:7}
Setelah sepagi penuh di tempat teduh yang sama, perosotan dan bangku
sama-sama dingin. Logam hanya \emph{terasa} lebih dingin di tangan ---
perabaan bukan hakim yang adil, seperti ditunjukkan oleh sendok logam
dan sendok kayu.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:8}
Sentuh mangkuk yang satu, lalu yang lain, segera, dengan tangan yang
\emph{sama}, dan mintalah seorang teman memeriksanya tanpa mendengar
jawabanmu. Tangan yang sama diperlukan karena tipuan tiga mangkuk
menunjukkan bahwa dua tangan bisa berbeda pendapat tentang air yang
persis sama.
\end{solution}

\begin{solution}{exo:g1:hot-and-cold:9}
Lea benar. Mantel tidak membuat panas: ia hanya menahan kehangatan
agar tidak lewat. Manusia salju tidak punya kehangatan untuk ditahan
di dalam --- sebaliknya mantel itu menahan \emph{udara hangat siang
hari di luar}, sehingga manusia salju yang terbungkus mencair lebih
lambat dan bertahan lebih lama.
\end{solution}
