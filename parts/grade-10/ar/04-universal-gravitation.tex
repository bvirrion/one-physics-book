\chapter{الجاذبية الكونية والوزن}\label{ch:g10:universal-gravitation}

أسقِط تفاحة: تقع. أما القمر، على ارتفاع ستين نصف قطر أرضي، فلا يبدو أنه يقع أبدًا.
وبصيرة نيوتن الكبرى أن القمر \emph{يقع} فعلًا --- نحو
1.4 مِلّيمتر في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}، وهو بالضبط ما يتنبأ به تجاذب
يضعف كمربع المسافة. صيغة واحدة، بثابت
كوني واحد، تحكم التفاحة والقمر والمدّ والجزر
والكواكب. وقد وصف المجلد الأول \omterm{def:g10:universal-gravitation:force}{الجاذبية} بالكلام؛ أما هذا
الفصل فيقيسها.

\section{قانون الجاذبية الكونية}

\begin{definition}[قوة الجاذبية]\label{def:g10:universal-gravitation:force}
يتجاذب كل جسمين، مهما تكن مادتهما ومهما
تباعدا. وهذا التجاذب هو \emph{قوة
الجاذبية}\index{قوة الجاذبية}، والظاهرة هي
\emph{الجاذبية}\index{الجاذبية}. وهي لا تحتاج تماسًّا ولا حبلًا ولا
وسطًا، وهي \emph{متبادلة}: فكل جسم يجذب الآخر.
\end{definition}

\begin{theorem}[قانون الجاذبية الكونية]\label{thm:g10:universal-gravitation:law}
جسمان كتلتاهما $m_1$ و$m_2$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{kg})، والمسافة بين مركزيهما
$d$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{m})، يتجاذبان بقوتين
موجَّهتين على المستقيم الواصل بين المركزين، كلٌّ نحو
الجسم الآخر، وشدتهما متساوية
\[
F = G\,\frac{m_1 m_2}{d^2}
\qquad \text{(بوحدة \unit{N})},
\]
حيث $G$ هو الثابت نفسه لكل زوج من الأجسام في الكون.
\end{theorem}
\admitted

\begin{remark}[من أين جاء القانون]
استخلص نيوتن هذا القانون من حركة الكواكب المرصودة
(قوانين كبلر)؛ ويُجرى الاستنتاج الأمين في مجلد السنة
الأولى. وسيضع \Cref{ch:g12:satellites-kepler} القانون في العمل
على التوابع والكواكب.
\end{remark}

\begin{definition}[ثابت الجاذبية]\label{def:g10:universal-gravitation:constant}
الثابت
\[
G = \qty{6.67e-11}{N.m^2/kg^2}
\]
هو \emph{ثابت الجاذبية}\index{ثابت الجاذبية}. وضآلة
قيمته هي سبب كون \omterm{def:g10:universal-gravitation:force}{الجاذبية} بين الأجسام اليومية
غير محسوسة: فلا يبلغ $G m_1 m_2 / d^2$ شيئًا يُذكر إلا حين تكون إحدى الكتلتين
على الأقل فلكية.
\end{definition}

\begin{remark}[قراءة الصيغة]\label{rem:g10:universal-gravitation:reading}
أربع سمات تستحق الانتباه.
\begin{itemize}
  \item \emph{متبادلتان ومتساويتان.} تجذبك الأرض بوزنك
        --- وأنت تجذب الأرض بالقوة نفسها بالضبط. (لكن
        الأرض تستجيب أقلّ: فهي أثقل نحو $\num{e23}$
        مرة.)
  \item \emph{مربع المسافة في المقام.} مضاعفة $d$ تقسم القوة على $4$؛
        والابتعاد عشر مرات أكثر يقسمها على $100$.
  \item \emph{المسافة بين المركزين.} في الأجسام الكروية --- النجوم
        والكواكب وكرات المدفع --- تُقاس $d$ من مركز إلى مركز، لا
        من سطح إلى سطح.
  \item \emph{الكتلتان كلتاهما تهمّان.} مضاعفة أي من الكتلتين تضاعف
        القوة؛ فالصيغة متناظرة في $m_1$ و$m_2$.
\end{itemize}
\end{remark}

\begin{example}[صديقان]\label{ex:g10:universal-gravitation:friends}
يقف صديقان كتلة كل منهما \qty{70}{kg} والمسافة بين مركزيهما \qty{1.0}{m}:
\[
F = \num{6.67e-11} \times \frac{70 \times 70}{1.0^2}
\approx \qty{3.3e-7}{N}.
\]
\omterm{def:g10:universal-gravitation:weight}{ووزن} كل صديق نحو \qty{6.9e2}{N} --- أي أكبر \emph{مليارَي}
مرة. فتجاذبهما المتبادل حقيقي، لكن الإحساس به ميؤوس منه.
\end{example}

\begin{example}[الأرض تمسك القمر]\label{ex:g10:universal-gravitation:earthmoon}
بأخذ $m_1 = \qty{5.97e24}{kg}$ (الأرض) و$m_2 = \qty{7.35e22}{kg}$ (القمر)
و$d = \qty{3.84e8}{m}$:
\[
F = \num{6.67e-11} \times
\frac{\num{5.97e24} \times \num{7.35e22}}{(\num{3.84e8})^2}
\approx \qty{2.0e20}{N}.
\]
ويجذب القمر الأرض بالمقدار \qty{2.0e20}{N} نفسه --- وهو
جذب تجيب عنه المحيطات مرتين في اليوم، مدًّا وجزرًا.
\end{example}

\begin{method}[حساب قوة جاذبية]\label{met:g10:universal-gravitation:compute}
\begin{enumerate}
  \item حوّل الكتل إلى \omterm{def:g10:orders-of-magnitude:unit}{الكيلوغرام} والمسافة بين المركزين
        إلى \omterm{def:g10:orders-of-magnitude:unit}{المتر}.
  \item طبّق $F = G m_1 m_2 / d^2$.
  \item تحقّق من قوة العشرة على حدة عن الأرقام، كما في
        \cref{ch:g10:orders-of-magnitude}: فقوى \omterm{def:g10:universal-gravitation:force}{الجاذبية} تمتد
        من $\num{e-7}$ نيوتن (شخصان على بعد \omterm{def:g10:orders-of-magnitude:unit}{متر}) إلى $\num{e22}$ نيوتن
        (الشمس على الأرض)، ومن ثم يسهل التقاط أسٍّ في غير موضعه.
\end{enumerate}
\end{method}

\section{الوزن وشدة الجاذبية}

\begin{definition}[الوزن]\label{def:g10:universal-gravitation:weight}
\emph{وزن}\index{وزن} $\vect P$ جسم عند سطح
نجم (أو كوكب أو قمر، \dots) هو \omterm{def:g10:universal-gravitation:force}{قوة الجاذبية} التي يمارسها النجم
عليه. وهو موجَّه نحو مركز النجم --- أي في الاتجاه الذي نسميه
\emph{العمودي}\index{العمودي} --- ويقاس بالنيوتن.
\end{definition}

\begin{proposition}[الوزن عند سطح نجم]\label{prop:g10:universal-gravitation:surfaceg}
عند سطح نجم كروي كتلته $M$ ونصف قطره $R$، يكون \omterm{def:g10:universal-gravitation:weight}{وزن} جسم
كتلته $m$
\[
P = m\,g,
\qquad\text{حيث}\qquad
g = \frac{G M}{R^2}
\]
لا يتوقف إلا على النجم، لا على الجسم.
\end{proposition}

\begin{proof}
المسافة بين مركز الجسم ومركز النجم هي $R$، ومن ثم يعطي
قانون \omterm{def:g10:universal-gravitation:force}{الجاذبية} الكونية
(\cref{thm:g10:universal-gravitation:law})
$P = G M m / R^2 = m \times \left(G M / R^2\right)$. والمقدار بين القوسين هو
نفسه لكل جسم عند السطح: فلنسمّه $g$.
\end{proof}

\begin{definition}[شدة مجال الجاذبية]\label{def:g10:universal-gravitation:fieldstrength}
المقدار $g = G M / R^2$، بالنيوتن لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام}، هو
\emph{شدة مجال الجاذبية}\index{شدة مجال الجاذبية}
عند سطح النجم: فالنجم يجذب كل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} بقوة $g$ نيوتن.
وعلى الأرض $g = \qty{9.81}{N/kg}$.
\end{definition}

\begin{example}[حساب $g$]\label{ex:g10:universal-gravitation:gvalues}
في الأرض ($M = \qty{5.97e24}{kg}$، $R = \qty{6.37e6}{m}$):
\[
g = \frac{\num{6.67e-11} \times \num{5.97e24}}{(\num{6.37e6})^2}
= \frac{\num{3.98e14}}{\num{4.06e13}}
\approx \qty{9.81}{N/kg}.
\]
وفي القمر ($M = \qty{7.35e22}{kg}$، $R = \qty{1.74e6}{m}$) يعطي الحساب
نفسه $g_{\text{القمر}} \approx \qty{1.62}{N/kg}$ --- أي أقلّ ست
مرات: فالقمر أخفّ بكثير، وصغره لا يعوّض
ذلك. وفي المريخ، $g_{\text{المريخ}} \approx \qty{3.73}{N/kg}$.
\end{example}

\begin{remark}[لماذا يقفز روّاد الفضاء]\label{rem:g10:universal-gravitation:bounce}
رائد فضاء كتلته \qty{120}{kg} (الجسم مع البدلة) يزن
$120 \times 9.81 \approx \qty{1180}{N}$ على الأرض ولا يزن إلا
$120 \times 1.62 \approx \qty{194}{N}$ على القمر. فالعضلات المدرَّبة
تحت \qty{1180}{N} تحمل فجأة سُدس الحمل: ومن هنا
المشية القمرية الوثّابة البطيئة الشهيرة. أما الكتلة --- ومعها الجهد
اللازم \emph{للتوقف} أو \emph{للانعطاف} --- فلم تتغير، ولذلك
كان روّاد القمر يسقطون كثيرًا.
\end{remark}

\section{كتلة أم وزن؟ الميزان ذو الكفتين والميزان الزنبركي}

تقول لغة الحياة اليومية إن كيس الطحين ``يزن ثلاثة كيلوات''. أما الفيزياء
فتشقّ تلك الجملة شقّين.

\begin{remark}[الكتلة ليست الوزن]\label{rem:g10:universal-gravitation:massweight}
\begin{itemize}
  \item \emph{الكتلة} $m$ تقيس كمية المادة،
        \omterm{def:g10:orders-of-magnitude:unit}{بالكيلوغرام}. وهي خاصية للجسم وحده: فهي نفسها على
        الأرض وعلى القمر وفي أعماق الفضاء.
  \item \emph{\omterm{def:g10:universal-gravitation:weight}{الوزن}} $P = mg$ قوة، بالنيوتن، يمارسها
        \emph{نجمٌ} على الجسم. وهو يتغير من نجم إلى نجم
        بتغير $g$ --- ويتلاشى بعيدًا عن كل نجم.
\end{itemize}
والخلط بينهما غير ضارّ في السوق وقاتل في مسألة
فيزياء: \omterm{def:g10:orders-of-magnitude:unit}{فالكيلوغرام} ليس \omterm{def:g10:orders-of-magnitude:unit}{وحدة} قوة.
\end{remark}

\begin{method}[الميزان ذو الكفتين والميزان الزنبركي]\label{met:g10:universal-gravitation:instruments}
\begin{itemize}
  \item \emph{الميزان ذو الكفتين} يقارن الجسم بكتل معيارية:
        \omterm{def:g10:universal-gravitation:weight}{فوزن} الكفتين يتغير بالعامل $g$ نفسه، ومن ثم تقيس المقارنة
        \emph{الكتلة} --- بالحكم نفسه على أي نجم.
  \item \emph{الميزان الزنبركي} يقيس القوة التي تمطّ زنبركه:
        فيقرأ \emph{\omterm{def:g10:universal-gravitation:weight}{الوزن}}، بالنيوتن، وتتغير قراءته
        من نجم إلى نجم. ولاسترجاع الكتلة، اقسم على شدة
        المجال المحلية: $m = P/g$.
\end{itemize}
\end{method}

\begin{example}[الطحين على القمر]\label{ex:g10:universal-gravitation:flour}
كيس طحين كتلته \qty{3.0}{kg} يوازن \qty{3.0}{kg} من الكتل
المعيارية على الأرض \emph{وعلى} القمر. أما الميزان الزنبركي فيقرأ
$3.0 \times 9.81 \approx \qty{29}{N}$ على الأرض و
$3.0 \times 1.62 \approx \qty{4.9}{N}$ على القمر. فالتاجر الذي يبيع
``بالنيوتن'' على القمر سيسلّم ستة أضعاف الطحين ---
فالتجارة، كالعلم، تقوم على الكتلة.
\end{example}

\section{السقوط حول الأرض: مدفع نيوتن}

اقذف حجرًا أفقيًّا: تحني \omterm{def:g10:universal-gravitation:force}{الجاذبية} مساره قوسًا فيهبط
على بعد أمتار قليلة. اقذفه أسرع: يهبط أبعد. وتدفع تجربة نيوتن
الذهنية الفكرة إلى حدّها. فالأرض كروية، وينخفض
سطحها نحو \qty{5}{m} عن الأفقي في كل
\qty{8}{km} مقطوع؛ والجسم الساقط سقوطًا حرًّا ينخفض \qty{4.9}{m} في
ثانيته الأولى (وسيثبت ذلك \cref{ch:g12:projectile-motion}). ومن ثم فالمقذوف
الذي يلامس الأرض بسرعة \qty{8}{km/s} يسقط بالسرعة نفسها التي
تنحني بها الأرض مبتعدة: فيسقط أبدًا دون أن يهبط قط.

\begin{definition}[التابع والمدار]\label{def:g10:universal-gravitation:satellite}
\emph{تابع}\index{تابع} نجمٍ هو جسم يسقط
حول النجم دون أن يبلغ سطحه قط؛ والمسار الذي يكرّره
هو \emph{مداره}\index{مدار}.
\end{definition}

\begin{remark}[القمر حجر ساقط]\label{rem:g10:universal-gravitation:fallingmoon}
القمر \omterm{def:g10:universal-gravitation:satellite}{تابع} الأرض الطبيعي: فهو يسقط نحونا نحو
\qty{1.4}{mm} في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}، وتحمله حركته الجانبية حولنا
قبل أن يقترب أكثر
(ويتيح لك \cref{exo:g10:universal-gravitation:15} إعادة تحقّق نيوتن نفسه
من هذا العدد). أما روّاد محطة مدارية فهم في السقوط
الحر نفسه الذي فيه سفينتهم --- وذلك، لا انعدام \omterm{def:g10:universal-gravitation:force}{الجاذبية}،
هو سبب طفوّهم. \omterm{def:g10:universal-gravitation:force}{فالجاذبية} تشكّل كل مسار، من
قوس الحجر إلى إهليلج الكوكب؛ أما القصة الكمية للمدارات
فتُروى في \cref{ch:g12:satellites-kepler}.
\end{remark}

\end{center}
المسافة بين الأرض والقمر: \qty{3.84e8}{m}؛ والمسافة بين الأرض والشمس:
\qty{1.496e11}{m} (من مركز إلى مركز).
\end{notation}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:1}
شريكان في الرقص كتلة كل منهما \qty{60}{kg} يمسك أحدهما بالآخر والمسافة بين مركزيهما
\qty{0.80}{m}.
\begin{enumerate}
  \item احسب تجاذبهما الثقالي المتبادل.
  \item قارنه \omterm{def:g10:universal-gravitation:weight}{بوزن} بعوضة كتلتها \qty{2.5}{mg}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:2}
كتلة تلميذة \qty{55}{kg}.
\begin{enumerate}
  \item احسب وزنها على الأرض.
  \item سافرت إلى القمر. فما كتلتها ووزنها
        هناك؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:3}
\begin{enumerate}
  \item انطلاقًا من بطاقة المعطيات، أعِد حساب \omterm{def:g10:universal-gravitation:fieldstrength}{شدة مجال الجاذبية}
        عند سطح المريخ.
  \item تُرسَل إليه مركبة جوّالة كتلتها \qty{900}{kg}. احسب وزنها على المريخ
        ووزنها على الأرض.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:4}
باستعمال بطاقة المعطيات، احسب القوة التي تمارسها الأرض على
القمر. وما القوة التي يمارسها القمر على الأرض؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:5}
يتجاذب جسمان بقوة $F_0$. فماذا تصير القوة إذا:
\begin{enumerate}
  \item ضوعفت المسافة بين المركزين؛
  \item نُصّفت المسافة؛
  \item ضوعفت الكتلتان كلتاهما (والمسافة نفسها)؛
  \item ثُلّثت إحدى الكتلتين \emph{و} ثُلّثت المسافة؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:6}
ناقلتا نفط عملاقتان محمّلتان كتلة كل منهما \qty{5.0e8}{kg} راسيتان والمسافة بين
مركزيهما \qty{100}{m}.
\begin{enumerate}
  \item احسب تجاذبهما الثقالي المتبادل.
  \item أيّ كتلة يساوي وزنها هذه القوة على الأرض؟
  \item القوة كبيرة على نحو مفاجئ --- ومع ذلك لا تُبدي الناقلتان أي
        ميل إلى الانجراف إحداهما نحو الأخرى. احسب التسارع
        $a = F/m$ الذي تعطيه لناقلة واحدة وعلّق.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:7}
يبيع متجر قمري كيس أرزّ كتلته \qty{3.0}{kg}.
\begin{enumerate}
  \item ماذا يشير إليه ميزان ذو كفتين وميزان زنبركي لهذا الكيس
        على الأرض؟ وعلى القمر؟
  \item يدفع زبون ثمن ``\qty{3.0}{kg}''. فأيّ الأداتين
        تضمن صفقة عادلة على أي عالم، ولماذا؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:8}
تحلّق محطة الفضاء الدولية على ارتفاع \qty{400}{km}.
\begin{enumerate}
  \item احسب شدة مجال جاذبية الأرض عند ذلك
        الارتفاع. (انتبه: ما المسافة إلى \emph{مركز}
        الأرض؟)
  \item عبّر عنها بنسبة مئوية من قيمتها السطحية.
  \item يطفو روّاد الفضاء على متنها. وفّق بين ذلك وجوابك.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:9}
عند سطح الزهرة $g = \qty{8.87}{N/kg}$، ونصف قطر الكوكب
\qty{6.05e6}{m}. استنتج كتلة الزهرة وقارنها بكتلة
الأرض.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:10}
\begin{enumerate}
  \item احسب القوة التي تمارسها الشمس على \qty{1.0}{kg} من ماء
        البحر، ثم القوة التي يمارسها القمر على \omterm{def:g10:orders-of-magnitude:unit}{الكيلوغرام}
        نفسه.
  \item الشمس تجذب أقوى --- بأي عامل؟
  \item ومع ذلك يهيمن القمر على المدّ والجزر. فالمدّ لا يقوده
        الجذب نفسه بل \emph{الفرق} بين الجذبين على
        الجانبين القريب والبعيد من الأرض. اشرح، دون
        حساب، لماذا يقدر القمر القريب على التغلب على الشمس البعيدة
        بهذا المعيار.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:11}
\begin{enumerate}
  \item كوكب كتلته ضعف كتلة الأرض ونصف قطره ضعف نصف قطرها.
        عبّر عن شدة مجاله السطحية بدلالة $g$
        الأرضية، ثم \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{N/kg}.
  \item أيّ نصف قطر يحتاجه كوكب كتلته كتلة الأرض لتكون شدة
        مجاله السطحية $2g$؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:12}
في مكان ما على المستقيم الواصل بين الأرض والقمر، يتلاشى الجذبان على مسبار
فضائي. ليكن $x$ المسافة من مركز الأرض إلى تلك النقطة
و$D$ المسافة بين الأرض والقمر.
\begin{enumerate}
  \item بيّن أنه عند تلك النقطة
        $\left(\dfrac{x}{D - x}\right)^{2}
        = \dfrac{M_{\text{الأرض}}}{M_{\text{القمر}}}$.
  \item استنتج $x$. وأيّ جزء من الرحلة إلى القمر هو ذاك؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:13}
مدفع نيوتن بالأرقام. مقذوف يلامس الأرض
أفقيًّا بسرعة $v$.
\begin{enumerate}
  \item بعد $x = \qty{8.0}{km}$ من الانتقال الأفقي، تكون الأرض
        الكروية قد ``انخفضت'' بارتفاع $h$ يحقق
        $(R + h)^2 = R^2 + x^2$. باستعمال
        $(R + h)^2 \approx R^2 + 2Rh$ من أجل $h$ الصغيرة، بيّن أن
        $h \approx x^2 / (2R)$ واحسب $h$.
  \item ينخفض جسم يسقط سقوطًا حرًّا \qty{4.9}{m} في ثانيته الأولى.
        قارن ذلك بالمقدار $h$ واستنتج السرعة التي عندها لا يهبط
        المقذوف أبدًا.
  \item احسب مدة دورة مدارية كاملة بتلك السرعة، ملامسًا
        السطح.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:14}
نجم نيوتروني يحشر $M = \qty{2.8e30}{kg}$ (أي 1.4 ضعف كتلة الشمس)
في نصف قطر \qty{12}{km}.
\begin{enumerate}
  \item احسب \omterm{def:g10:universal-gravitation:fieldstrength}{شدة مجال الجاذبية} عند سطحه.
  \item كم يزن هناك إنسان كتلته \qty{70}{kg}؟ قارن ذلك بوزنه
        على الأرض.
  \item وما \omterm{def:g10:universal-gravitation:weight}{وزن} حبة رمل كتلتها \qty{1.0}{mg} هناك؟
        وأيّ كتلة يكون لها هذا \omterm{def:g10:universal-gravitation:weight}{الوزن} على الأرض؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:15}
اختبار نيوتن للقمر (1666). نصف قطر \omterm{def:g10:universal-gravitation:satellite}{مدار} القمر نحو $60$ نصف قطر
أرضي.
\begin{enumerate}
  \item بأي عامل يكون جذب الأرض لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} أضعف عند
        القمر منه عند سطح الأرض، إذا كانت \omterm{def:g10:universal-gravitation:force}{الجاذبية} تتبع
        مربع المسافة في المقام؟
  \item قرب السطح، ينخفض جسم ساقط \qty{4.9}{m} في
        ثانيته الأولى. فكم ينبغي أن يسقط القمر نحو الأرض
        في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}؟
  \item تحقّق من ذلك \omterm{def:g10:universal-gravitation:satellite}{بالمدار}: يقطع القمر دائرته ذات نصف القطر
        $d = \qty{3.84e8}{m}$ في $T = 27.3$ يومًا. احسب سرعته
        $v$، والمسافة $x$ التي يقطعها في \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة، ثم السقوط
        $h \approx x^2/(2d)$ (كما في
        \cref{exo:g10:universal-gravitation:13}).
  \item قارن جوابي السؤالين 2 و3، وقل ما الذي استنتجه
        نيوتن.
\end{enumerate}
\end{exercise}

\section{مسألة: وزن الأرض}

\begin{problem}[{مسألة نهاية الأسبوع --- من كرتي رصاص في سقيفة إلى
كتلة الأرض ثم كتلة الشمس: ما قيمة ثابت صغير واحد
قيس مرة واحدة}]
\label{pb:g10:universal-gravitation:1}
سنة 1798 علّق هنري كافنديش قضيبًا خفيفًا يحمل كرتي رصاص
صغيرتين بسلك رفيع، وقرّب منه كرتي رصاص كبيرتين،
وقاس فتل السلك الذي يكاد لا يوجد. وقالت الصحف
إنه \emph{\omterm{def:g10:universal-gravitation:weight}{وزن} الأرض} --- وكانت محقّة: فمتى عُرف $G$،
سلّمت $g$ الأرض ونصف قطرها كتلتَها، وسلّم
\omterm{def:g10:universal-gravitation:satellite}{مدار} الأرض السنوي كتلة الشمس. وتعيد هذه المسألة السير في
السلسلة كلها. استعمل بطاقة المعطيات في كل مكان.

\medskip
\noindent\textbf{الجزء I --- أضعف قوة.}
\begin{enumerate}
  \item كتلة حديد برج إيفل نحو \qty{7.3e6}{kg}. ويقف
        سائح كتلته \qty{55}{kg} على بعد \qty{100}{m} من مركز
        كتلته. احسب جذب البرج للسائح.
  \item قارنه \omterm{def:g10:universal-gravitation:weight}{بوزن} حبة رمل كتلتها \qty{1.0}{mg}.
  \item تمرين على التناسب: ماذا يحدث لقوة جاذبية
        إذا (أ) ضُربت المسافة في $10$؛ (ب) ضُربت الكتلتان
        في $1000$؛ (ج) ضُربت الكتلتان \emph{والمسافة}
        في $1000$؟
  \item احسب جذب الأرض لجسم كتلته \qty{1.00}{kg} عند
        سطحها، انطلاقًا من $M$ و$R$ و$G$. فأيّ اسم ورمز يوميين
        يحملهما هذا العدد؟
  \item \omterm{def:g10:universal-gravitation:force}{الجاذبية} أضعف التفاعلات التي لقيتها بفارق كبير ---
        ومع ذلك تدير الكون. اشرح في جملة أو جملتين
        لماذا يهيمن جذب الأرض لك على جذب جارك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- ميزان كافنديش.}
في ميزان الفتل، تجذب كرة كبيرة كتلتها $m_1 = \qty{158}{kg}$
كرة صغيرة كتلتها $m_2 = \qty{0.73}{kg}$؛ والمسافة بين مركزيهما
$d = \qty{0.23}{m}$.
\begin{enumerate}[resume]
  \item بالقيمة الحديثة للثابت $G$، احسب القوة بين
        الكرتين.
  \item احسب \omterm{def:g10:universal-gravitation:weight}{وزن} الكرة الصغيرة ونسبة القوتين.
        ولماذا احتاج كافنديش سلك فتل بدل أي
        ميزان عادي؟
  \item يوافق الفتل الذي قاسه كافنديش (بالوحدات الحديثة)
        $F = \qty{1.47e-7}{N}$. استنتج قيمته للثابت $G$ وقارنها
        بقيمة اليوم.
  \item بأخذ $G = \qty{6.67e-11}{N.m^2/kg^2}$ و$g = \qty{9.81}{N/kg}$
        و$R = \qty{6.37e6}{m}$، استنتج كتلة الأرض. (وهذه
        هي الخطوة التي احتفت بها الصحف.)
  \item استنتج الكتلة الحجمية المتوسطة للأرض $\rho = M / V$، بأخذ
        $V = \frac43 \pi R^3$. أما صخور السطح فكتلتها الحجمية نحو
        \qty{2.7e3}{kg/m^3}: فماذا تكشف المقارنة عن
        الباطن العميق؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- عوالم أخرى.}
\begin{enumerate}[resume]
  \item انطلاقًا من بطاقة المعطيات، أعِد حساب شدة مجال القمر
        السطحية.
  \item على الأرض، ترفع دفعة وثب جيدة مركز كتلتك بمقدار
        \qty{0.50}{m}. وللدفعة نفسها، يتناسب الارتفاع المبلوغ عكسًا
        مع $g$ (كما سيبرّره فصل الطاقة،
        \cref{ch:g10:energy-conservation}). فإلى أي ارتفاع تحملك
        القفزة نفسها على القمر؟
  \item عطارد: $M = \qty{3.30e23}{kg}$، $R = \qty{2.44e6}{m}$.
        احسب شدة مجاله السطحية وقارنها بالمريخ.
        وعطارد أصغر من المريخ بكثير --- فكيف تكون القيمتان
        بهذا التقارب؟
  \item على أي ارتفاع فوق سطح الأرض تكون شدة
        المجال قد هبطت إلى نصف قيمتها السطحية؟
  \item وعلى أي مسافة من مركز الأرض يهبط جذب الأرض
        لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} إلى قيمته عند سطح القمر،
        \qty{1.62}{N/kg}؟ عبّر عنها بأنصاف الأقطار الأرضية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- \omterm{def:g10:universal-gravitation:weight}{وزن} الشمس.}
تقطع الأرض مدارًا شبه دائري نصف قطره
$d = \qty{1.496e11}{m}$ في سنة واحدة، $T = \qty{3.156e7}{s}$. وخذ
مسلَّمًا (وهو مبرهن عليه في \cref{ch:g12:satellites-kepler}) أن جسمًا على
مسار دائري نصف قطره $d$ بسرعة $v$ يحتاج جذبًا قدره $v^2/d$
نيوتن لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} نحو المركز.
\begin{enumerate}[resume]
  \item احسب السرعة المدارية للأرض $v$.
  \item استنتج الجذب لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} الذي تمارسه الشمس على الأرض.
  \item وذلك الجذب هو أيضًا $G M_{\text{الشمس}} / d^2$. استنتج كتلة
        الشمس.
  \item كم أرضًا يساوي ذلك؟ قارن $M_{\text{الشمس}}$ بالكتلة
        $M_{\text{الأرض}}$.
  \item الخاتمة. كرتا كافنديش، والأرض، والشمس: ثابت واحد
        $G$ خدم الثلاثة جميعًا. قل في جملة واحدة ما الذي اشترته لنا
        كلمة \emph{الكونية} في ``\omterm{def:g10:universal-gravitation:force}{الجاذبية} الكونية'' في
        هذه المسألة.
\end{enumerate}
\end{problem}
