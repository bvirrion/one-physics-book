\chapter{القوى ومبدأ العطالة}\label{ch:g10:inertia}

كفّ عن الدوس فتتباطأ الدراجة حتى تقف؛ وكفّ عن السحب فتقف
الزحّافة. وتهمس التجربة بأن الحركة لا بد أن تُغذّى
بجهد --- ووافقتها الفيزياء ألفي سنة. ويبني هذا الفصل
\omterm{def:g10:inertia:force}{القوة}، وهي سهم يقاس \omterm{def:g10:inertia:force}{بالنيوتن}، ويذكر أول
قوانين الميكانيك الكبرى: إن الجسم المتحرك، إذا تُرك وشأنه حقًّا، حفظ
حركته، مستقيمة منتظمة، إلى الأبد.

\section{نمذجة تأثير بقوة}

\begin{definition}[القوة]\label{def:g10:inertia:force}
\emph{القوة}\index{قوة} تنمذج تأثيرًا ميكانيكيًّا (دفعًا أو سحبًا
أو تجاذبًا) يُمارَس على جسم. وتتميز بما يلي:
\begin{itemize}
  \item \emph{نقطة تأثيرها}\index{نقطة التأثير}؛
  \item واتجاهها --- أي المستقيم الذي تؤثر عليه والجهة
        على ذلك المستقيم؛
  \item وشدتها، ووحدتها \emph{النيوتن}\index{نيوتن}
        (رمزه \unit{N}).
\end{itemize}
وتُرسم القوة سهمًا $\vect F$ (أي متجهة، كما في مجلد
الرياضيات) من نقطة تأثيرها، وطوله يتناسب مع
الشدة وفق مقياس مذكور.
\end{definition}

\begin{example}[وزن حقيبة مدرسية]\label{ex:g10:inertia:schoolbag}
تجذب الأرض حقيبة مدرسية كتلتها $m = \qty{4.0}{kg}$ إلى أسفل
بوزنها (\cref{ch:g10:universal-gravitation}):
$P = mg = 4.0 \times 9.81 \approx \qty{39}{N}$ --- \omterm{def:g10:inertia:force}{والنيوتن} الواحد نحو
\omterm{def:g10:universal-gravitation:weight}{وزن} تفاحة. وبالمقياس $\qty{1}{cm} \leftrightarrow
\qty{10}{N}$: سهم عمودي طوله \qty{3.9}{cm}، مؤثر في
مركز ثقل الحقيبة، متجه إلى أسفل.
\end{example}

\section{جرد القوى}

\begin{definition}[قوى التماس وقوى البعد]\label{def:g10:inertia:contact}
\emph{قوة التماس}\index{قوة تماس} تُمارَس عند النقاط
التي يتلامس فيها جسمان. أما \emph{القوة عن بعد}\index{قوة عن
بعد} فلا تحتاج تماسًّا: \omterm{def:g10:universal-gravitation:weight}{والوزن} --- أي جذب \omterm{def:g10:universal-gravitation:force}{الجاذبية} في
\cref{ch:g10:universal-gravitation} --- هو \omterm{def:g10:inertia:force}{القوة} الوحيدة من هذا النوع المستعملة في هذا
الفصل.
\end{definition}

\begin{definition}[قوى التماس الشائعة]\label{def:g10:inertia:inventory}
أربع قوى تماس تغطي معظم المواقف اليومية:
\begin{itemize}
  \item \emph{رد الفعل الناظمي}\index{رد الفعل الناظمي} $\vect R$ لسند،
        عمودي على سطحه، يدفع مبتعدًا عنه؛
  \item و\emph{الشدّ}\index{شدّ} $\vect T$ لسلك أو حبل أو
        سلسلة، موجَّه عليه، يسحب نحوه؛
  \item و\emph{الاحتكاك}\index{احتكاك} $\vect f$ من سطح أو
        من الهواء أو الماء، يعاكس الانزلاق أو الحركة؛
  \item و\emph{قوة الدفع}\index{قوة الدفع} لمحرك أو نفّاثة أو يد،
        تدفع في جهة الجهد.
\end{itemize}
\end{definition}

\begin{example}[جرد على زحّافة مقطورة]\label{ex:g10:inertia:sled}
زحّافة مقطورة على الثلج: \omterm{def:g10:universal-gravitation:weight}{الوزن} $\vect P$ (عن بعد)، \omterm{def:g10:inertia:inventory}{ورد
الفعل الناظمي} $\vect R$ للثلج، وشدّ الحبل $\vect T$، \omterm{def:g10:inertia:inventory}{والاحتكاك}
$\vect f$ المعاكس للانزلاق. ولا شيء آخر يلامس الزحّافة:
فالجرد كامل --- \emph{\omterm{def:g10:universal-gravitation:weight}{الوزن} دائمًا، ثم \omterm{def:g10:inertia:contact}{قوة تماس} واحدة
لكل جسم ملامس}.
\end{example}

\section{مبدأ العطالة}

\begin{definition}[الحركة المستقيمة المنتظمة]\label{def:g10:inertia:uniform}
يكون الجسم في \emph{حركة مستقيمة منتظمة}\index{حركة مستقيمة
منتظمة} في مرجعٍ ما إذا كان مساره مستقيمًا يُقطع
بسرعة ثابتة: أي أن متجهة سرعته $\vect v$ هي نفسها في كل
لحظة. والسكون هو الحالة الخاصة $\vect v = \vect 0$.
\end{definition}

\begin{definition}[القوى المتعادلة]\label{def:g10:inertia:compensate}
تكون القوى المؤثرة في جسم \emph{متعادلة}\index{قوى متعادلة}
إذا كان أثرها المجموع معدومًا: أي إذا عادت سهامها، رأسًا إلى ذيل، إلى
نقطة البداية. وتتعادل قوتان تعادلًا تامًّا حين تشتركان في مستقيم
التأثير نفسه بشدتين متساويتين وجهتين متعاكستين.
\end{definition}

\begin{theorem}[مبدأ العطالة]\label{thm:g10:inertia:principle}
\index{مبدأ العطالة}\index{عطالة}
في \omterm{def:g10:relative-motion:frames}{المرجع الأرضي}، الجسم الذي لا تؤثر فيه \omterm{def:g10:inertia:force}{قوة}، أو تؤثر فيه \omterm{def:g10:inertia:compensate}{قوى
متعادلة}، يبقى ساكنًا أو يحفظ \omterm{def:g10:inertia:uniform}{حركة مستقيمة منتظمة}؛
وعكسًا، الجسم الساكن أو ذو \omterm{def:g10:inertia:uniform}{الحركة المستقيمة المنتظمة} تؤثر فيه
\omterm{def:g10:inertia:compensate}{قوى متعادلة} (أو لا تؤثر فيه \omterm{def:g10:inertia:force}{قوة}).
\end{theorem}
\admitted

\begin{remark}
لا تجربة تقدر أن تعزل جسمًا عزلًا تامًّا، ولذلك فالمبدأ مصادرة
--- لمحها غاليليو، وذكرها نيوتن. وسيرقّيها فصل لاحق إلى
قانون يربط القوى بما يطرأ من \emph{تغيرات} على السرعة؛ ويسأل مجلد السنة الأولى
في أي مراجع يصحّ فعلًا.
\end{remark}

\begin{proposition}[قراءة الحركة من القوى، والعكس]\label{prop:g10:inertia:converse}
في \omterm{def:g10:relative-motion:frames}{المرجع الأرضي}:
\begin{enumerate}
  \item إذا تغيرت سرعة جسم --- في الشدة \emph{أو} في
        الاتجاه --- فالقوى المؤثرة فيه \emph{غير} \omterm{def:g10:inertia:compensate}{متعادلة}؛
  \item وإذا كان الجسم ساكنًا أو في \omterm{def:g10:inertia:uniform}{حركة مستقيمة منتظمة}، وجب أن
        تتلاشى قواه المجرودة --- وهو ما يحدّد المجاهيل.
\end{enumerate}
\end{proposition}

\begin{proof}
الجملة الأولى هي نقيض عكس
\cref{thm:g10:inertia:principle}، \omterm{def:g10:orders-of-magnitude:unit}{والثانية} جزؤها العكسي مطبَّقًا
على جرد كامل.
\end{proof}

\begin{example}[قرص الهوكي]\label{ex:g10:inertia:puck}
إذا أطلقته العصا، عبر القرص \qty{30}{m} من الجليد الأملس في
\qty{2.5}{s}، مستقيمًا بالسرعة الثابتة $v = 30/2.5 = \qty{12}{m/s}$.
والجرد: \omterm{def:g10:universal-gravitation:weight}{الوزن} \omterm{def:g10:inertia:inventory}{ورد الفعل الناظمي} (\omterm{def:g10:inertia:inventory}{والاحتكاك} مهمَل). وهما
يتعادلان، ويتنبأ المبدأ بالحركة المرصودة بالضبط:
فلا شيء يدفع القرص إلى الأمام --- ولا حاجة إلى شيء يدفعه.
\end{example}

\begin{remark}[غاليليو في وجه أرسطو]
علّم أرسطو أن كل حركة تحتاج محرّكًا، ويبدو أن الحياة اليومية ---
\omterm{def:g10:inertia:inventory}{بالاحتكاك} في كل تماس --- توافقه. أما غاليليو فمثّل \omterm{def:g10:inertia:inventory}{الاحتكاك}
إلى العدم: فالسطوح كلما ازدادت ملاسة احتاجت دفعًا أقل
\emph{لحفظ} الحركة. فالدوام، لا السكون، هو الحالة الطبيعية.
\end{remark}

\begin{remark}[المسافر المندفع]
حين يكبح حافلة كبحًا حادًّا، يميل المسافرون الواقفون إلى الأمام. ولا \omterm{def:g10:inertia:force}{قوة}
تقذفهم: إنما تحفظ أجسامهم سرعتها بالعطالة بينما تفقد الحافلة
سرعتها. وحزام المقعد يوفّر \omterm{def:g10:inertia:force}{القوة} الخلفية التي ينقصها الجرد.
\end{remark}

\section{القوى المتوازنة: السكون من رسم}

\begin{definition}[مخطط القوى]\label{def:g10:inertia:diagram}
\emph{مخطط القوى}\index{مخطط القوى} لجسمٍ ما يعرض الجسم
وحده، وقد رُسمت بالمقياس كل \omterm{def:g10:inertia:force}{قوة} تؤثر \emph{فيه} من
\omterm{def:g10:inertia:force}{نقطة تأثيرها}؛ أما القوى التي يمارسها \emph{الجسم} فلا تظهر أبدًا.
\end{definition}

\begin{method}[تحليل وضع ساكن]\label{met:g10:inertia:method}
\begin{enumerate}
  \item اختر الجسم المدروس؛ واشتغل في \omterm{def:g10:relative-motion:frames}{المرجع الأرضي}.
  \item اجرد: \omterm{def:g10:universal-gravitation:weight}{الوزن} دائمًا، \omterm{def:g10:inertia:contact}{وقوة تماس} واحدة لكل تماس.
  \item ارسم \omterm{def:g10:inertia:diagram}{مخطط القوى}.
  \item الجسم ساكن، \omterm{def:g10:inertia:compensate}{فالقوى متعادلة}
        (\cref{thm:g10:inertia:principle}): اقرأ الشدات المجهولة من
        الرسم (أطوال متساوية في بُعد واحد، ومركّبات في بُعدين).
\end{enumerate}
\end{method}

\begin{example}[الكتاب على الطاولة]\label{ex:g10:inertia:book}
كتاب كتلته \qty{0.50}{kg} يستقر على طاولة: \omterm{def:g10:universal-gravitation:weight}{الوزن}
$P = 0.50 \times 9.81 \approx \qty{4.9}{N}$ \omterm{def:g10:inertia:inventory}{ورد الفعل الناظمي}
$\vect R$. وبما أنه ساكن، تتعادل $\vect R$ مع $\vect P$: فهي عمودية، إلى أعلى،
$R = \qty{4.9}{N}$. وأضِف كتابًا ثانيًا فتتكيف $R$ --- فالسند
يدفع بالقدر اللازم بالضبط.
\end{example}

\begin{example}[المصباح على سلكين]\label{ex:g10:inertia:lamp}
مصباح كتلته \qty{3.0}{kg} ($P \approx \qty{29.4}{N}$) معلّق بسلكين،
كلٌّ منهما على $45^\circ$ من العمودي. وبالتناظر يتساوى الشدّان
($T_1 = T_2 = T$) وتتلاشى مركّبتاهما الأفقيتان؛ أما المركّبتان
العموديتان، وكلٌّ منهما $T \cos 45^\circ$، فتحملان \omterm{def:g10:universal-gravitation:weight}{الوزن}:
\[
2\,T \cos 45^\circ = P
\qquad\text{ومنه}\qquad
T = \frac{29.4}{2 \times 0.707} \approx \qty{21}{N}
\]
--- أي أكثر من نصف \omterm{def:g10:universal-gravitation:weight}{الوزن} لكل سلك: فالسحب بزاوية يكلّف
شدًّا.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g10:inertia:1}
اجرد وصنّف (تماس أو بعد) القوى المؤثرة في: (أ) كتاب
ساكن على طاولة؛ (ب) مصباح معلّق بسلكه؛ (ج) قرص
ينزلق على جليد أملس؛ (د) تفاحة ساقطة (ومقاومة الهواء
مهمَلة).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:2}
احسب \omterm{def:g10:universal-gravitation:weight}{وزن} كتاب كتلته \qty{250}{g} \omterm{def:g10:universal-gravitation:weight}{ووزن} شخص كتلته \qty{60}{kg}.
وبالمقياس $\qty{1}{cm} \leftrightarrow \qty{1}{N}$، كم يبلغ
طول سهم الكتاب، ولماذا يكون المقياس ميؤوسًا منه في حالة
الشخص؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:3}
في مخطط بالمقياس $\qty{1}{cm} \leftrightarrow \qty{2}{N}$، تكون
\omterm{def:g10:inertia:force}{قوة} سهمًا أفقيًّا طوله \qty{3.5}{cm} متجهًا إلى اليسار، مؤثرًا
في نقطة $A$. أعطِ خواصها الثلاث.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:4}
يستقر قاموس كتلته \qty{1.2}{kg} على رفّ. ارسم مخطط
قواه وأعطِ شدة كل \omterm{def:g10:inertia:force}{قوة} واتجاهها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:5}
صحيح أم خطأ --- صحّح الخاطئ: (أ) ينتهي كل جسم إلى
التوقف ما لم تستمر \omterm{def:g10:inertia:force}{قوة} في دفعه؛ (ب) \omterm{def:g10:inertia:uniform}{الحركة المستقيمة المنتظمة}
لا تقتضي \omterm{def:g10:inertia:force}{قوة} غير متعادَلة؛ (ج) الجسم الساكن لا تؤثر فيه
\omterm{def:g10:inertia:force}{قوة}؛ (د) إذا تغيرت سرعة جسم فالقوى المؤثرة فيه غير
\omterm{def:g10:inertia:compensate}{متعادلة}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:6}
بعد الإطلاق، يقطع قرص \qty{24}{m} من الجليد في \qty{3.0}{s}،
مستقيمًا بسرعة ثابتة. احسب سرعته؛ فهل تتعادل القوى المؤثرة
فيه؟ وعلى الإسمنت يتوقف القرص نفسه في حدود أمتار: فما الذي تغير
في الجرد، وما الذي يستنتجه \cref{prop:g10:inertia:converse}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:7}
مسافر كتلته \qty{70}{kg} واقف في مصعد يصعد بالسرعة الثابتة
\qty{1.5}{m/s}. أحصِ القوى المؤثرة في المسافر؛ فهل تتعادل؟
أعطِ شدة \omterm{def:g10:inertia:force}{قوة} الأرضية. وهل يتغير شيء لو كان
المصعد ساكنًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:8}
مصباح كتلته \qty{1.8}{kg} معلّق ساكنًا بسلك عمودي واحد.
احسب \omterm{def:g10:inertia:inventory}{شدّ} السلك. ثم يُستبدل بالسلك سلكان
\emph{عموديان} يتقاسمان الحمل بالتساوي: فما \omterm{def:g10:inertia:inventory}{شدّ}
كلٍّ منهما؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:9}
تكبح حافلة كبحًا حادًّا فيندفع مسافر واقف إلى الأمام؛ وفي
منعطف بسرعة ثابتة، يميل المسافر نفسه إلى الخارج. اشرح
الأمرين دون اختراع \omterm{def:g10:inertia:force}{قوة} ``أمامية'' أو ``خارجية''.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:10}
ينزل مظلّي كتلته الكلية \qty{85}{kg} عموديًّا بالسرعة
الثابتة \qty{5.0}{m/s}. ارسم \omterm{def:g10:inertia:diagram}{مخطط القوى} واحسب مقاومة
الهواء على المظلة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:11}
يُقطر متزلج على الماء مستقيمًا بسرعة ثابتة؛ ويسحب الحبل الأفقي
\omterm{def:g10:inertia:force}{بقوة} \qty{300}{N}. فما مقاومة الماء الأفقية على
الزلاجتين (برّر)؟ وأيّ قوتين عموديتين تتعادلان؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:12}
مصباح كتلته \qty{2.4}{kg} معلّق بسلكين على $40^\circ$ من
العمودي ($\cos 40^\circ \approx 0.766$). احسب كل \omterm{def:g10:inertia:inventory}{شدّ}.
ولماذا يكبر \omterm{def:g10:inertia:inventory}{الشدّ} كلما اقترب السلكان من الوضع الأفقي،
ولماذا لا يقدر سلكان أفقيان تمامًا على حمل المصباح أبدًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:13}
يُدفع حجر كيرلنغ، ثم يُطلق، فينزلق متباطئًا تباطؤًا
تدريجيًّا جدًّا، ثم يستقر. لكل طور (الدفع، الانزلاق، السكون)، ارسم
\omterm{def:g10:inertia:diagram}{مخطط القوى} وقل أتتعادل القوى، مستشهدًا
بما جاء في \cref{thm:g10:inertia:principle} أو في \cref{prop:g10:inertia:converse}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:14}
ينساب المسبار \emph{فوياجر 1} في الفضاء بين النجوم بسرعة
\qty{17}{km/s}، ومحركاته مطفأة. فلماذا لا يحتاج وقودًا لحفظ
سرعته، وبماذا كان أرسطو يتنبأ؟ وكم يقطع في
سنة، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{km} ثم بالوحدات الفلكية ($\qty{1}{au} \approx
\qty{1.496e8}{km}$)؟ ولماذا يجب أن يشغّل دافعة لمجرد تغيير
الاتجاه \emph{دون} تغيير السرعة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:15}
تسحب ثلاثة حبال حلقة ساكنة. وعلى شبكة مربعها الواحد
\qty{1}{N}، تُقرأ قوتان $\vect F_1\,(3, 2)$ و
$\vect F_2\,(-1, 2)$. جِد مركّبتي \omterm{def:g10:inertia:force}{القوة} الثالثة وشدتها
واتجاهها.
\end{exercise}

\section{مسألة: العطالة في ثلاثة فصول}

\begin{problem}[{مسألة نهاية الأسبوع --- سير المطار وحلبة الجليد
والمظلة: السرعة الثابتة بصمةً للقوى المتعادلة،
ولماذا لا تخفض المظلة المقاومة بل السرعة}]
\label{pb:g10:inertia:1}
ثلاثة مشاهد --- حقيبة على سير متحرك، وقرص على حلبة جليد،
ومظلّي في هواء المساء --- وقانون واحد يتكلم في الثلاثة.

\medskip
\noindent\textbf{الجزء I --- سير المطار.}
حقيبة كتلتها \qty{23}{kg} واقفة على سير يتحرك بالسرعة
الثابتة \qty{0.75}{m/s}.
\begin{enumerate}
  \item ما حركة الحقيبة في \omterm{def:g10:relative-motion:frames}{المرجع الأرضي}؟ واجرد القوى المؤثرة فيها.
  \item فهل تتعادل هذه القوى؟ برّر؛ واحسب الشدتين.
  \item قارن مخطط قواها بمخطط الحقيبة نفسها على
        الأرضية. فأيّ جملة عميقة توضّحها المقارنة؟
  \item يتوقف السير فجأة. صِف ما تفعله الحقيبة،
        دون اختراع \omterm{def:g10:inertia:force}{قوة} أمامية.
  \item تمشي صاحبتها بسرعة \qty{1.2}{m/s} بالنسبة إلى السير. فما
        سرعتها في \omterm{def:g10:relative-motion:frames}{المرجع الأرضي}؟ وهل تتعادل القوى المؤثرة \emph{فيها}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- حلبة الجليد.}
\begin{enumerate}[resume]
  \item ينزلق قرص كتلته \qty{160}{g} مستقيمًا بالسرعة الثابتة
        \qty{8.0}{m/s}. ارسم مخطط قواه بالشدات.
  \item يزعم أرسطو أن القرص يحتاج محرّكًا. فبماذا تردّ الحلبة،
        ولماذا تبدو السطوح اليومية منحازة إليه؟
  \item يعبر القرص رقعة خشنة فيتباطأ. فما الاستنتاج عن
        القوى هناك؟ وأيّ \omterm{def:g10:inertia:force}{قوة} هي الجانية؟
  \item ينعطف متزلج انعطافًا سلسًا بسرعة ثابتة. فهل تتعادل القوى المؤثرة
        فيه؟ انتبه: أيّ نوع من المقادير هي السرعة؟
  \item اربط المخطط بالحركة: (أ) \omterm{def:g10:inertia:compensate}{قوى متعادلة}؛ (ب) \omterm{def:g10:inertia:force}{قوة} زائدة
        معاكسة للمتجهة $\vect v$؛ (ج) \omterm{def:g10:inertia:force}{قوة} زائدة عمودية على
        $\vect v$ --- (1) تباطؤ مستقيم؛ (2) انعطاف بسرعة
        ثابتة؛ (3) \omterm{def:g10:inertia:uniform}{حركة مستقيمة منتظمة}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- المظلة.}
كتلة قافز كلية، مع العتاد، \qty{80}{kg}.
\begin{enumerate}[resume]
  \item لحظة مغادرة الطائرة، تكون السرعة العمودية والمقاومة معدومتين.
        احسب القوى؛ فهل تتعادل؟ وما الذي يجب أن يحدث تاليًا؟
  \item تكبر المقاومة مع السرعة. صِف السقوط ما دامت المقاومة $<$ \omterm{def:g10:universal-gravitation:weight}{الوزن}.
  \item قبل فتح المظلة، يبلغ القافز سرعة ثابتة قدرها
        \qty{50}{m/s} (``السرعة الحدية''). احسب المقاومة. وما
        الحركة، ولماذا هي ``عطالة متنكّرة''؟
  \item تُفتح المظلة: فتتجاوز المقاومة الآن \omterm{def:g10:universal-gravitation:weight}{الوزن} بكثير. فماذا
        يحدث للسرعة؟ (والقافز يتحرك \emph{إلى أسفل}
        طول الوقت.)
  \item تُبلَغ سرعة ثابتة جديدة قدرها \qty{5.0}{m/s}. احسب المقاومة؛
        وقارنها بالسؤال 13.
  \item بيت القصيد: ما الذي غيّرته المظلة --- المقاومة
        الحدية، أم السرعة التي عندها تبلغ المقاومة \omterm{def:g10:universal-gravitation:weight}{الوزن}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الأحكام.}
\begin{enumerate}[resume]
  \item أكمل وبرّر: ``ساكن أو في \omterm{def:g10:inertia:uniform}{حركة مستقيمة منتظمة}
        $\iff$ \dots''. ولماذا لا يستحق السكون قانونًا منفصلًا؟
  \item في أي فصل من الفصول كانت مقولة أرسطو ``الحركة تحتاج محرّكًا''
        ستخفق أوضح إخفاق، وإلى أيّ شيء كان غاليليو سيشير؟
  \item تزعم ورقة امتحان: ``يسقط المظلّي بسرعة ثابتة،
        فلا \omterm{def:g10:inertia:force}{قوة} تؤثر فيه''. صحّحها في سطر واحد.
  \item الخاتمة: لكل فصل، سمِّ زوج القوى المتعادلتين،
        واذكر القانون الواحد المستعمل عشرين مرة في هذه المسألة.
\end{enumerate}
\end{problem}
