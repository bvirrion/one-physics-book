\chapter{رتب القدر: قياس الكون}\label{ch:g10:orders-of-magnitude}

تبدأ الفيزياء حيث ينتهي الرأي: عند القياس. غير أن ما تقيسه الفيزياء
يمتد على مدى مذهل --- فقطر نواة الذرة نحو
\qty{e-15}{m}، وقطر الكون المرصود نحو \qty{e27}{m}: أي بعامل
قدره $10^{42}$، وهو أبعد من أن تسعه صورة ذهنية.
يركّب هذا الفصل العُدّة التي تروّض هذا المدى --- الوحدات
وقوى العشرة \omterm{def:g10:orders-of-magnitude:sigfig}{والأرقام المعنوية} --- وعادتين ترافقان كل
صفحة بعده: اكتب \omterm{def:g10:orders-of-magnitude:unit}{الوحدة}، وتحقّق من \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر}.

\section{الوحدات: قواعد لغة القياس}

\begin{definition}[المقادير الفيزيائية ووحدات النظام الدولي]\label{def:g10:orders-of-magnitude:unit}
\emph{المقدار الفيزيائي}\index{مقدار فيزيائي} خاصية من خواص
العالم يمكن قياسها: طول أو مدة أو كتلة.
وقياسه يعني عدّ المرات التي ينطبق فيها مقدار مرجعي ---
هو \emph{الوحدة}\index{وحدة} --- على المقدار المقيس.
و\emph{النظام الدولي للوحدات}\index{النظام الدولي للوحدات (SI)} (SI) يثبّت سبع وحدات أساسية،
ثلاث منها تحكم هذا المجلد كله: \emph{المتر}\index{متر} (\unit{m}) للطول،
و\emph{الكيلوغرام}\index{كيلوغرام} (\unit{kg}) للكتلة،
و\emph{الثانية}\index{ثانية} (\unit{s}) للزمن. (أما الأربع الباقية ---
الأمبير والكلفن والمول والشمعة --- فتلتحق بنا حين تدخل الكهرباء والحرارة
والكيمياء في القصة.)
\end{definition}

\begin{remark}[عدد بلا وحدة ليس نتيجة]\label{rem:g10:orders-of-magnitude:units}
قولنا ``المسافة $384$'' لا يعني شيئًا: إذ إنّ \qty{384}{mm}
و\qty{384}{km} و\qty{384}{Mm} ثلاثة عوالم مختلفة (وآخرها
يصادف أنه المسافة إلى القمر). في 1999 أضاعت NASA مسبار \emph{مارس كلايمت أوربيتر}،
وقد كلّف \$327 مليون دولار، لأن برنامجًا حسب
قوى الدفع بالرطل-قوة بينما قرأها برنامج آخر بالنيوتن.
قاعدة البيت، من هذا السطر إلى آخر الكتاب: كل عدد مقيس أو
محسوب يحمل وحدته، في كل سطر من الحساب.
\end{remark}

\begin{definition}[سوابق النظام الدولي]\label{def:g10:orders-of-magnitude:prefixes}
\emph{السابقة}\index{سابقة (SI)} الملصقة \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} تضربها في
قوة ثابتة من قوى العشرة:
\begin{center}
\begin{tabular}{llc@{\qquad\qquad}llc}
فمتو & f & $10^{-15}$ & سنتي & c & $10^{-2}$ \\
بيكو  & p & $10^{-12}$ & كيلو  & k & $10^{3}$  \\
نانو  & n & $10^{-9}$  & ميغا  & M & $10^{6}$  \\
ميكرو & $\mu$ & $10^{-6}$ & جيغا & G & $10^{9}$ \\
مِلّي & m & $10^{-3}$  & تيرا  & T & $10^{12}$ \\
\end{tabular}
\end{center}
وهكذا $\qty{1}{nm} = \qty{e-9}{m}$ و$\qty{1}{km} = \qty{e3}{m}$.
\end{definition}

\begin{example}[قراءة السوابق]\label{ex:g10:orders-of-magnitude:prefixes}
طول موجة الضوء الأخضر نحو $\qty{500}{nm} = \qty{5.00e-7}{m}$؛
وسُمك شعرة الإنسان نحو $\qty{70}{\micro m} = \qty{7.0e-5}{m}$؛
ويدور القمر على بعد $\qty{384}{Mm} = \qty{3.84e8}{m}$. وكتلة قطرة مطر
واحدة نحو $\qty{30}{mg} = \qty{3.0e-5}{kg}$ --- ولاحظ أن \omterm{def:g10:orders-of-magnitude:unit}{وحدة} الكتلة
الأساسية في النظام الدولي تحمل سابقة أصلًا، وحدها من بين السبع.
\end{example}

\section{الكتابة العلمية والأرقام المعنوية}

\begin{definition}[الكتابة العلمية]\label{def:g10:orders-of-magnitude:scinot}
يكون العدد مكتوبًا \emph{بالكتابة العلمية}\index{كتابة علمية}
إذا كُتب على الصورة
\[
a \times 10^{n}, \qquad 1 \leq a < 10, \quad n \text{ عدد صحيح.}
\]
ويسمّى العامل $a$ \emph{المُعامِل}\index{معامل}، ويسمّى العدد الصحيح
$n$ \emph{الأُسّ}\index{أس}.
وهكذا $\num{149600000} = \num{1.496e8}$ و
$\num{0.00052} = \num{5.2e-4}$.
\end{definition}

\begin{proposition}[الحساب بقوى العشرة]\label{prop:g10:orders-of-magnitude:powers}
مهما يكن العددان الصحيحان $m$ و$n$:
\[
10^{m} \times 10^{n} = 10^{m+n}, \qquad
\frac{1}{10^{n}} = 10^{-n}, \qquad
\left(10^{m}\right)^{n} = 10^{mn}.
\]
\end{proposition}

\begin{proof}[برهان جزئي]
في حالة الأسس الموجبة، عُدّ العوامل: إنّ $10^{m} \times 10^{n}$ جداء
$m + n$ عشرة، و$(10^m)^n$ يسلسل $n$ مجموعة من $m$ عشرة.
وكتابة $10^{-n}$ بدل $1/10^{n}$ هي بالضبط الاصطلاح الذي
يبقي القاعدة الأولى صحيحة مهما يكن \omterm{def:g10:orders-of-magnitude:scinot}{الأس} صحيحًا: فهو يفرض
$10^{n} \times 10^{-n} = 10^{0} = 1$. ويدرس مجلد الرياضيات
المرافق قواعد الأسس دراسة كاملة.
\end{proof}

\begin{example}[الضرب بالكتابة العلمية]\label{ex:g10:orders-of-magnitude:scinot}
كم أرضًا تسع الشمس، كتلةً؟ بأخذ
$m_{\text{الشمس}} = \qty{1.99e30}{kg}$ و
$m_{\text{الأرض}} = \qty{5.97e24}{kg}$:
\[
\frac{\num{1.99e30}}{\num{5.97e24}}
= \frac{1.99}{5.97} \times 10^{30-24}
= 0.333 \times 10^{6}
= \num{3.33e5}.
\]
تُعالَج المعاملات والأسس على حدة؛ ثم تُعاد كتابة الجواب النهائي
حتى يعود \omterm{def:g10:orders-of-magnitude:scinot}{المعامل} إلى المجال $[1, 10)$.
\end{example}

\begin{definition}[الأرقام المعنوية]\label{def:g10:orders-of-magnitude:sigfig}
\emph{الأرقام المعنوية}\index{أرقام معنوية} لعدد مقيس هي
الأرقام التي يضمنها القياس فعلًا: كل الأرقام
المكتوبة عدا الأصفار البادئة. والأصفار اللاحقة تُحسب ---
إذ إنّ $\qty{5.20e-3}{s}$ (ثلاثة أرقام معنوية)
يَعِد بأكثر مما يعد به $\qty{5.2e-3}{s}$ (رقمان). \omterm{def:g10:orders-of-magnitude:scinot}{والكتابة العلمية} تُظهر
عددها بنظرة واحدة: \omterm{def:g10:orders-of-magnitude:scinot}{فالمعامل} يحمل الأرقام المعنوية
بالضبط.
\end{definition}

\begin{method}[كم رقمًا نحتفظ به]\label{met:g10:orders-of-magnitude:sigfig}
لا يمكن لنتيجة محسوبة أن تكون أدقّ من مكوّناتها.
\begin{enumerate}
  \item في الجداء أو القسمة، احتفظ من \omterm{def:g10:orders-of-magnitude:sigfig}{الأرقام المعنوية} بعدد
        العامل \emph{الأقل} دقة.
  \item قرّب الجواب النهائي وحده، ولا تقرّب الخطوات الوسيطة أبدًا.
  \item عند الشك، ثلاثة \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية} هي دقة العمل
        في هذا الكتاب.
\end{enumerate}
\end{method}

\begin{example}[محيط الأرض]\label{ex:g10:orders-of-magnitude:sigfig}
قطر الأرض $d = \qty{1.27e7}{m}$ (ثلاثة \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية})،
فمحيطها إذن
$\pi d = \num{3.14159} \times \qty{1.27e7}{m} = \qty{3.99e7}{m}$ ---
ثلاثة أرقام، لا الثمانية التي تعرضها الآلة الحاسبة. وكتابة
$\qty{39898227}{m}$ ادّعاء بمعرفة الكوكب حتى \omterm{def:g10:orders-of-magnitude:unit}{المتر}.
\end{example}

\section{سُلّم الكون}

\begin{definition}[رتبة القدر]\label{def:g10:orders-of-magnitude:oom}
\emph{رتبة القدر}\index{رتبة القدر} لمقدار ما هي
أقرب قوة من قوى العشرة إليه: تكتب المقدار على الصورة
$a \times 10^{n}$ حيث $1 \leq a < 10$، ثم تأخذ $10^{n}$ إذا كان $a < 5$،
و$10^{n+1}$ إذا كان $a \geq 5$. (ولا يهمّ اصطلاح الحدّ الفاصل
بدقة أبدًا: فالقول برتبة القدر لا يُقصد به قط إلا ما هو مضبوط
في حدود عامل قدره عشرة.)
\end{definition}

\begin{example}[رتب القدر]\label{ex:g10:orders-of-magnitude:oom}
يعيش الإنسان نحو $80$ سنة، أي\
$80 \times \qty{3.16e7}{s} \approx \qty{2.5e9}{s}$: \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر}
$\qty{e9}{s}$ --- بضعة مليارات من الثواني، فأحسِن إنفاقها. وطول الإنسان
$\qty{1.7}{m}$: \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر} $\qty{e0}{m}$. وقمة إفرست،
$\qty{8848}{m}$: \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر} $\qty{e4}{m}$، لأن $8.848 \geq 5$.
\end{example}

ورصّ هذه التقديرات حسب \omterm{def:g10:orders-of-magnitude:scinot}{الأس} ينتج الصورة الجامعة
لهذا الفصل: سُلّم كل درجة فيه عالم أوسع عشر مرات
من العالم الذي تحته.

وحداته الخاصة، وكلتاهما معرَّفة من السُّلّم نفسه.

\begin{definition}[الوحدة الفلكية والسنة الضوئية]\label{def:g10:orders-of-magnitude:lightyear}
\emph{الوحدة الفلكية}\index{وحدة فلكية}
($\qty{1}{au} = \qty{1.496e11}{m}$) هي متوسط المسافة بين الأرض والشمس.
و\emph{السنة الضوئية}\index{سنة ضوئية} (\unit{ly}) هي
المسافة التي يقطعها الضوء في سنة واحدة. وينتقل الضوء في الخلاء بسرعة
\[
c = \qty{3.00e8}{m/s},
\]
وهي أكبر سرعة تسمح بها الطبيعة --- ولذلك فالسنة الضوئية مسافة
لا زمن.
\end{definition}

\begin{example}[السنة الضوئية بالمتر]\label{ex:g10:orders-of-magnitude:ly}
تحوي السنة $365.25 \times 24 \times 3600 = \qty{3.156e7}{s}$
(وهو رقم قريب على نحو لطيف من $\pi \times 10^{7}\,\unit{s}$). ومنه
\[
\qty{1}{ly} = c \times \qty{3.156e7}{s}
= \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}.
\]
وتقع بروكسيما قنطورس، أقرب نجم بعد الشمس، على بعد
$\qty{4.2}{ly} \approx \qty{4.0e16}{m}$: فالضوء الذي يعبر
المسافة بين الأرض والشمس في ثماني دقائق يحتاج إلى أكثر من أربع سنوات ليصل
من هناك.
\end{example}

\begin{method}[التحقق برتبة القدر]\label{met:g10:orders-of-magnitude:sanity}
قبل الوثوق بأي نتيجة محسوبة:
\begin{enumerate}
  \item قرّب كل معطًى إلى رتبة قدره؛
  \item أعِد الحساب بقوى العشرة وحدها، مستعملًا
        \cref{prop:g10:orders-of-magnitude:powers} --- ولا يستغرق ذلك
        إلا ثوانيَ؛
  \item قارن: يجب أن يوافق الجواب الدقيق التقديرَ
        في حدود عامل قدره عشرة أو نحو ذلك، وأن تكون وحدته هي
        الصحيحة. فإن أخفق أحد التحققين، فتعقّب الخطأ قبل أن
        تمضي.
\end{enumerate}
\end{method}

\begin{remark}[لِمَ كل هذا العناء]\label{rem:g10:orders-of-magnitude:sanity}
سوف تخبرك الآلة الحاسبة راضيةً بأن كتلة تفاحة
$\qty{4e8}{kg}$ --- فهي لا تملك أن تعرف أكثر. ولا صيغة تكشف
السخيف؛ إنما تكشفه العادة. فالفيزيائيون يقدّرون أولًا ويحسبون ثانيًا:
والتقدير يمسك بالفواصل العشرية في غير موضعها، وبالوحدات المتبادلة،
وبالكسور المقلوبة، وهي الأخطاء الثلاثة وراء أكثر الأجوبة الخاطئة في
كل مستوى.
\end{remark}

\section{القياس والارتياب}

لا تقرأ أداة القيمة الحقيقية. والقياس لا يكون أمينًا إلا إذا
أخبر كذلك عن مقدار ما قد يبتعد عنها.

\begin{definition}[الارتياب المطلق والارتياب النسبي]\label{def:g10:orders-of-magnitude:uncertainty}
يُكتب قياس مقدار $x$ على الصورة
\[
x = x_{\text{m}} \pm \Delta x ,
\]
حيث $x_{\text{m}}$ هي القيمة المقيسة، و\emph{الارتياب
المطلق}\index{ارتياب!مطلق} $\Delta x > 0$، \omterm{def:g10:orders-of-magnitude:unit}{بالوحدة}
نفسها، يحدّ الخطأ الممكن: إذ يُدّعى أن القيمة الحقيقية تقع
بين $x_{\text{m}} - \Delta x$ و$x_{\text{m}} + \Delta x$. أما
\emph{الارتياب النسبي}\index{ارتياب!نسبي} فهو النسبة
$\Delta x / x_{\text{m}}$، وتُذكر عادة بالمئة؛ وهو ما تقيسه
كلمة ``دقيق''.
\end{definition}

\begin{method}[قراءة أداة مدرّجة]\label{met:g10:orders-of-magnitude:reading}
في المسطرة والمخبار المدرّج والمؤشر:
\begin{enumerate}
  \item اقرأ التدريجة الأقرب إلى العلامة، والعين عمودية على
        السلّم (فالعين المائلة تزيح القراءة)؛
  \item خذ نصف أصغر تدريجة على الأقل ارتيابًا
        مطلقًا --- وأكثر إن كانت حافة الجسم أو سطح
        السائل غير حادّة؛
  \item اكتب النتيجة على الصورة $x_{\text{m}} \pm \Delta x$ مع وحدتها،
        ولا تحتفظ بأي رقم بعد الرقم المرتاب فيه.
\end{enumerate}
\end{method}

\begin{example}[الدقة النسبية]\label{ex:g10:orders-of-magnitude:reading}
للقراءة \omterm{def:g10:orders-of-magnitude:prefixes}{السابقة} ارتياب نسبي
$\frac{0.5}{127} \approx 0.4\%$. وحين تقيس المسطرة نفسها
برغيًّا طوله $\qty{12}{mm}$ تعطي $\frac{0.5}{12} \approx 4\%$: فالأداة عينها
أقلّ دقة عشر مرات على الجسم الصغير. فالدقة
خاصية \emph{للقياس} لا للأداة.
\end{example}

\begin{remark}[مطلق صغير ونسبي هائل --- والعكس]\label{rem:g10:orders-of-magnitude:precision}
تقيس محطات الليزر المسافة بين الأرض والقمر، $\qty{3.84e8}{m}$، إلى
نحو $\pm\qty{3}{cm}$: أي بارتياب نسبي قدره $10^{-10}$، وهو من
أحدّ ما قيس على الإطلاق --- بخطأ مطلق لا تتباهى به
مسطرة. فاسأل دائمًا \emph{نسبةً إلى ماذا}: فالزوج
$(\Delta x,\ \Delta x / x_{\text{m}})$ يروي القصة كاملة، وكل
من العددين وحده قد يضلّل.
\end{remark}

\section{مسألة: من الذرة إلى أندروميدا}

\begin{problem}[{مسألة نهاية الأسبوع --- حبة رمل واحدة تقيس
الذرة والمجموعة الشمسية والمجرّة والكون المرصود، وتحدّد
موقعك على السُّلّم بينها}]
\label{pb:g10:orders-of-magnitude:1}
حبة الرمل أوضع جسم في الفيزياء، وهي في نهاية هذا الأسبوع
تعمل عملًا إضافيًّا: نعدّ أولًا ذراتها، ثم نقلّص الشمس إلى
حجمها ونقيس الكون بالخطى، ثم ندع الضوء يتولّى المساحة،
وأخيرًا نعدّ الذرات في \emph{جسمك}. المعطيات، وتُستعمل في الكل:
قطر الحبة $\qty{0.5}{mm}$؛ قطر الذرة $\qty{2.5e-10}{m}$؛
كتلة الرمل الحجمية $\qty{2500}{kg/m^3}$؛ الأقطار: الشمس
$\qty{1.39e9}{m}$، الأرض $\qty{1.27e7}{m}$، درب التبانة
$\qty{9.5e20}{m}$، الكون المرصود $\qty{8.8e26}{m}$؛ المسافات
عنا: القمر $\qty{3.84e8}{m}$، الشمس $\qty{1.496e11}{m}$، نبتون
$\qty{4.5e12}{m}$، بروكسيما قنطورس $\qty{4.0e16}{m}$، مجرّة
أندروميدا $\qty{2.4e22}{m}$؛ $c = \qty{3.00e8}{m/s}$؛ وكتلة جزيء الماء
$\qty{3.0e-26}{kg}$ ويحوي ثلاث ذرات.

\medskip
\noindent\textbf{الجزء I --- تشريح الحبة.}
\begin{enumerate}
  \item اكتب قطر الحبة وقطر الذرة \omterm{def:g10:orders-of-magnitude:unit}{بالمتر}، \omterm{def:g10:orders-of-magnitude:scinot}{بالكتابة
        العلمية}، وسمِّ سابقة النظام الدولي الأقرب إلى كل
        منهما.
  \item كم ذرة تصطفّ جنبًا إلى جنب على قطر حبة واحدة؟
  \item نمذِج الحبة مكعبًا طول ضلعه $\qty{0.5}{mm}$ مرصوصًا
        بالذرات. قدّر عدد ذرات الحبة برتبة
        قدر.
  \item احسب حجم هذا المكعب، ثم كتلة الحبة.
        عبّر عن الكتلة بالمِلّيغرام.
  \item تحقّق من سلامة النموذج: اقسم كتلة الحبة على عدد
        ذراتها وقارن \omterm{def:g10:orders-of-magnitude:oom}{برتبة القدر} المعروفة
        للكتل الذرية (بضعة $10^{-26}\,\unit{kg}$). فما حكمك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الشمس حبة رمل.}
نبني مجسّمًا مصغَّرًا للكون تكون فيه الشمس هي الحبة:
فيساوي كل بُعد في المجسّم البُعد الحقيقي مضروبًا في عامل $s$.
\begin{enumerate}[resume]
  \item احسب عامل المقياس $s$.
  \item جِد في المجسّم قطر الأرض وبعدها عن
        الشمس-الحبة. وأيّ أجسام عالم الجزء I يقارب
        حجم الأرض المجسّمة؟
  \item كم يبعد مدار نبتون المجسّم عن الحبة؟ وأيّ قطعة
        أثاث تسع المنظومة الكوكبية كلها؟
  \item وكم تبعد بروكسيما قنطورس المجسّمة؟
  \item حبتا رمل تفصل بينهما $\qty{14}{km}$ ولا شيء تقريبًا
        بينهما: بيّن في جملة أو جملتين ما يكشفه المجسّم
        عن خواء المجرّة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الضوء يتولّى المساحة.}
\begin{enumerate}[resume]
  \item كم يستغرق الضوء ليعبر حبة الرمل الحقيقية؟
  \item وكم يستغرق ليصل إلينا من القمر، ومن
        الشمس؟
  \item أعِد حساب \omterm{def:g10:orders-of-magnitude:lightyear}{السنة الضوئية} \omterm{def:g10:orders-of-magnitude:unit}{بالمتر} انطلاقًا من $c$ وطول
        السنة، وتحقّق من أن بُعد بروكسيما في المعطيات
        يوافق نحو $\qty{4.2}{ly}$.
  \item حوّل بُعد أندروميدا إلى سنوات ضوئية. ومتى تقريبًا
        غادر الضوءُ الداخلُ الآن إلى عينك أندروميدا، وما الذي
        كان يمشي على الأرض حينها؟
  \item كم يقطع الضوء في نانوثانية واحدة؟ ومعالج
        تواتره $\qty{3}{GHz}$ يبدأ عملية كل ثلث
        نانوثانية: اشرح في جملة واحدة لماذا يجب أن يكون الحاسوب
        السريع حاسوبًا صغيرًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- أنت، بين الذرة وأندروميدا.}
\begin{enumerate}[resume]
  \item الإنسان تقريبًا $\qty{70}{kg}$ من الماء. باستعمال المعطيات،
        قدّر متوسط كتلة ذرة واحدة في الماء، ثم
        عدد الذرات في إنسان.
  \item قارن هذا العدد بعدد ذرات الحبة (السؤال 3)
        وبعدد نجوم الكون المرصود البالغ نحو
        $2\times10^{23}$. أعطِ العاملين معًا.
  \item على السُّلّم، أأنت أقرب إلى الذرة أم إلى الكون
        المرصود؟ قارن النسبتين $\text{(أنت)}/\text{(الذرة)}$ و
        $\text{(الكون)}/\text{(أنت)}$، بأخذ $\qty{1.7}{m}$ طولًا
        لك.
  \item جِد الحجم $x$ الواقع في منتصف السُّلّم تمامًا
        بين النواة ($\qty{e-15}{m}$) والكون
        المرصود: يجب أن يحقق $x$ العلاقة $x / 10^{-15} = 10^{26.9} / x$
        (بأخذ $\qty{8.8e26}{m} \approx 10^{26.9}\,\unit{m}$). وأيّ أجرام
        المجموعة الشمسية يصادف أن له هذا الحجم بالذات؟
  \item بيت القصيد. قلّص الكون المرصود بعامل الجزء II
        $s$، حتى تصير الشمس حبة رمل. أعطِ قطر الكون
        المجسّم \omterm{def:g10:orders-of-magnitude:unit}{بالمتر}، ثم بالوحدات الفلكية، ثم
        اختم بجملة واحدة: حتى لو صار كل نجم حبة
        رمل، يفيض الكون عن كل مقياس بشري --- ولا يسعه
        إلا قوى العشرة.
\end{enumerate}
\end{problem}
