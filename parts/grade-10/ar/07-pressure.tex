\chapter{الضغط: من الرياضة إلى الغوص}\label{ch:g10:pressure}

ينزلق متزلج على ثلج يغوص فيه الماشي؛ ودفعة قدرها \qty{5}{N}
تدفع إبرة عبر جلد لا تقدر قبضة على ثنيه. القوى نفسها،
وآثار متباينة تباينًا هائلًا: والمقدار الناقص هو \emph{\omterm{def:g10:pressure:pressure}{الضغط}}،
أي \omterm{def:g10:inertia:force}{القوة} مقسومة على المساحة المتلقّية لها. ويعرّفه هذا الفصل،
ويتبعه نازلًا في مسبح، وصاعدًا في بارومتر، وداخلًا في محقنة مضغوطة ---
وينتهي على عمق أربعين مترًا تحت البحر، حيث يصير حيويًّا.

\section{الضغط: قوة موزّعة على مساحة}

\begin{definition}[الضغط]\label{def:g10:pressure:pressure}
حين تضغط \omterm{def:g10:inertia:force}{قوة} شدتها $F$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{N}) عموديًّا على
سطح مساحته $S$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{m^2})، يكون \emph{الضغط}\index{ضغط}
الممارَس على ذلك السطح
\[
P = \frac{F}{S}.
\]
ووحدته، أي \unit{N/m^2}، هي \emph{الباسكال}\index{باسكال}
(\unit{Pa}) --- وهي \omterm{def:g10:orders-of-magnitude:unit}{وحدة} ضئيلة، أي \omterm{def:g10:universal-gravitation:weight}{وزن} تفاحة موزّع على \omterm{def:g10:orders-of-magnitude:unit}{متر}
مربع، ولذلك تكون \unit{kPa} و\unit{MPa} وحدتي العمل.
\end{definition}

\begin{example}[الكعب في مقابل حذاء الثلج]\label{ex:g10:pressure:snowshoe}
شخص كتلته \qty{60}{kg} يزن $F = 60 \times 9.81 \approx \qty{590}{N}$.
وعلى كعب رفيع واحد مساحته \qty{1.0}{cm^2} = \qty{1.0e-4}{m^2}:
$P = 590/\num{1.0e-4} \approx \qty{5.9e6}{Pa} = \qty{5.9}{MPa}$. وعلى حذاءي
ثلج مساحة كلٍّ منهما \qty{0.20}{m^2}:
$P = 590/0.40 \approx \qty{1.5}{kPa}$. \omterm{def:g10:universal-gravitation:weight}{فالوزن} \emph{نفسه} يضغط
أشدّ $4000$ مرة تحت الكعب: ولذلك تنقّر الكعوب الأرضيات الخشبية،
وتطفو أحذية الثلج على الثلج الناعم.
\end{example}

\begin{remark}[ركّز أو وزّع]\label{rem:g10:pressure:tools}
إبرة خياطة مدفوعة \omterm{def:g10:inertia:force}{بقوة} \qty{5}{N} على رأس مساحته \qty{0.01}{mm^2} =
\qty{1e-8}{m^2} تمارس $\qty{5e8}{Pa} = \qty{500}{MPa}$ --- وهو ما يكفي
لفصل ألياف الجلد. فأدوات الثقب تركّز \omterm{def:g10:inertia:force}{القوة}؛ أما التصاميم
الحاملة (الزلاجات، والجنازير، والأساسات) فتوزّعها. والرياضة
إدارة تطبيقية \omterm{def:g10:pressure:pressure}{للضغط}: فالمزالج تعضّ الجليد، والزلاجات تنزلق على الثلج.
\end{remark}

\section{الضغط في سائل ساكن}

لماذا يضغط الماء أشدّ في الأسفل؟ اعزل عمودًا عموديًّا متخيَّلًا
من الماء مقطعه $S$، من السطح إلى العمق $h$:
حجمه $Sh$، وكتلته $\rho S h$ (حيث $\rho$ الكتلة الحجمية للسائل، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\unit{kg/m^3})، ووزنه $\rho S h g$. ويحمل أسفل العمود
ذلك \omterm{def:g10:universal-gravitation:weight}{الوزن} \emph{زائدًا} دفع الغلاف الجوي $P_0 S$ على أعلاه.
وبقسمة \omterm{def:g10:inertia:force}{القوة} الكلية على $S$: يجب أن يكون \omterm{def:g10:pressure:pressure}{الضغط} على العمق $h$
$P_0 + \rho g h$ --- وهو كذلك فعلًا، في كل موضع من السائل.

\begin{theorem}[الضغط على عمق]\label{thm:g10:pressure:depth}
في سائل كتلته الحجمية $\rho$ ساكن، مفتوح على الهواء، يكون \omterm{def:g10:pressure:pressure}{الضغط} على
العمق $h$ تحت السطح
\[
P = P_0 + \rho g h,
\]
حيث $P_0$ \omterm{def:g10:pressure:atmospheric}{الضغط الجوي} عند السطح. \omterm{def:g10:pressure:pressure}{والضغط}
لا يتوقف إلا على العمق، لا على شكل الوعاء، وعند
نقطة معطاة يضغط السائل \omterm{def:g10:inertia:force}{بالقوة} نفسها في كل الاتجاهات.
\end{theorem}
\admitted

\begin{remark}
حجّة العمود لا تجعل القانون إلا معقولًا؛ أما الاشتقاق الأمين
(لأي شكل، وفي كل الاتجاهات) ففي مجلد السنة الأولى. والحدّ العائد إلى الماء
وحده $\rho g h$ هو \omterm{def:g10:pressure:pressure}{الضغط} \emph{النسبي}\index{ضغط نسبي} --- أي ما يعرضه مقياس
\omterm{def:g10:pressure:pressure}{ضغط} الغطّاس.
\end{remark}

\begin{example}[عشرة أمتار من الماء]\label{ex:g10:pressure:tenmetres}
في الماء العذب ($\rho = \qty{1000}{kg/m^3}$)، عند $h = \qty{10}{m}$:
$\rho g h = 1000 \times 9.81 \times 10 \approx \qty{98}{kPa}$، ومنه
$P = 101 + 98 \approx \qty{199}{kPa}$. فعشرة أمتار من الماء تضيف جوًّا واحدًا
تقريبًا بالضبط: وهي قاعدة الغطّاسين التجريبية، \emph{جوّ إضافي
واحد لكل عشرة أمتار}.
\end{example}

\section{محيط الهواء}

\begin{definition}[الضغط الجوي]\label{def:g10:pressure:atmospheric}
نحن نعيش في قاع محيط من الهواء، ينشئ وزنه
\emph{الضغط الجوي}\index{ضغط جوي}، وهو عند مستوى البحر
\[
P_0 \approx \qty{101}{kPa} = \qty{1.01e5}{Pa}.
\]
وعلى كل سنتيمتر مربع يكون ذلك $F = \num{1.01e5} \times \num{1e-4}
\approx \qty{10}{N}$: أي \omterm{def:g10:universal-gravitation:weight}{وزن} كتلة قدرها \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} واحد على كل
\unit{cm^2} من جلدك ومن مكتبك ومن كل شيء.
\end{definition}

\begin{example}[الحمل غير المرئي]\label{ex:g10:pressure:sheet}
ورقة A4 (\qty{21.0}{cm} $\times$ \qty{29.7}{cm}، ومنه
$S \approx \qty{0.0624}{m^2}$) تتلقى من الهواء فوقها
$F = \num{1.01e5} \times 0.0624 \approx \qty{6.3e3}{N}$ --- أي \omterm{def:g10:universal-gravitation:weight}{وزن}
سيارة صغيرة. ولا شيء يتمزّق: لأن الهواء تحتها يدفع
\emph{إلى أعلى} \omterm{def:g10:inertia:force}{بالقوة} نفسها. ولا ننتبه إلى الغلاف الجوي إلا حين يفقده أحد
الجانبين --- \omterm{def:g10:orders-of-magnitude:scinot}{كأس} شفط، أو مصّاصة، أو مقصورة طائرة.
\end{example}

\begin{remark}[بارومتر توريتشيلي]\label{rem:g10:pressure:torricelli}
سنة 1643 قلب توريتشيلي أنبوبًا مغلقًا من الزئبق فوق حوض
زئبق. فهبط العمود إلى الارتفاع الذي عنده $\rho g h = P_0$:
$h = \num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$، تاركًا
فوقه أول خلاء صُنع في التاريخ. والارتفاع يتعقّب الطقس
ويتقلص مع الارتفاع عن سطح البحر: فالبارومتر يزن الهواء الذي فوقك (أما الماء
فكان سيحتاج \qty{10.3}{m}: \cref{exo:g10:pressure:9}).
\end{remark}

\section{الغازات: القصف وقانون بويل}

يضغط الغاز أيضًا على وعائه: فهو سِرب من الجزيئات في
حركة عشوائية لا تنقطع، وكل تصادم يعطي الجدار دفعة صغيرة إلى الخارج،
وتتشوّش مليارات المليارات من الاصطدامات في \omterm{def:g10:orders-of-magnitude:unit}{الثانية} لتصير \omterm{def:g10:inertia:force}{القوة}
الثابتة التي نقيسها ضغطًا للغاز. اضغط الغاز إلى نصف
الحجم: فتصطدم الجزيئات مرتين أكثر، فيجب أن يتضاعف \omterm{def:g10:pressure:pressure}{الضغط}.
وهو ما يحدث.

\begin{proposition}[قانون بويل]\label{prop:g10:pressure:boyle}
\index{قانون بويل}
لكمية غاز ثابتة محفوظة عند درجة حرارة ثابتة، يتناسب \omterm{def:g10:pressure:pressure}{الضغط}
والحجم تناسبًا عكسيًّا:
\[
P \, V = \text{ثابت},
\qquad\text{أي}\qquad
P_1 V_1 = P_2 V_2 .
\]
\end{proposition}
\admitted

\begin{remark}
هو واقعة تجريبية في هذا المستوى --- ويشتقّه مجلد السنة الأولى
من صورة التصادمات. أما التجربة (محقنة هواء مغلقة،
وحسّاس \omterm{def:g10:pressure:pressure}{ضغط}، \qty{20}{\degreeCelsius}) فهي:
\begin{center}
\begin{tabular}{lccccc}
$V$ (\unit{mL}) & 60 & 50 & 40 & 30 & 20 \\
$P$ (\unit{kPa}) & 100 & 120 & 150 & 200 & 300 \\
$P \times V$ (\unit{kPa.mL}) & 6000 & 6000 & 6000 & 6000 & 6000
\end{tabular}
\end{center}
\end{remark}

\begin{example}[محقنة مضغوطة]\label{ex:g10:pressure:boyle}
يشغل هواء \qty{4.0}{L} عند \qty{100}{kPa}؛ وإذا ضُغط ببطء (حتى تبقى
درجة الحرارة ثابتة) إلى \qty{250}{kPa}، شغل
$V_2 = P_1 V_1 / P_2 = 400/250 = \qty{1.6}{L}$.
\end{example}

\section{الغوص: الضغط في العمل}

\begin{method}[الضغط المطلق على عمق]\label{met:g10:pressure:dive}
لغطّاس على العمق $h$ في البحر ($\rho \approx \qty{1025}{kg/m^3}$،
ومنه $\rho g h \approx \qty{101}{kPa}$ لكل \qty{10}{m})، احسب
$P = P_0 + \rho g h$ --- واستعمل في \omterm{prop:g10:pressure:boyle}{قانون بويل} دائمًا هذا \omterm{def:g10:pressure:pressure}{الضغط}
\emph{المطلق}\index{ضغط مطلق}، لا الحدّ النسبي وحده. فالغطّاس يقع
تحت $2$ جوّ على \qty{10}{m}، و$3$ على \qty{20}{m}، و$5$ على
\qty{40}{m}.
\end{method}

\begin{remark}[لا تحبس نفَسك أبدًا]\label{rem:g10:pressure:breath}
يوصّل المنظّم الهواء \omterm{def:g10:pressure:pressure}{بالضغط} المحيط: فتمتلئ الرئتان امتلاءً طبيعيًّا على
أي عمق. لكن املأهما \qty{6.0}{L} على \qty{20}{m} ($\qty{302}{kPa}$)،
واحبس ذلك النفَس واصعد: فيقتضي بويل
$6.0 \times 302/101 \approx \qty{18}{L}$ --- أي ثلاثة أضعاف ما يسعه
الصدر، والرئتان تتمزقان قبل ذلك بكثير، ولو في صعود بضعة أمتار.
ومن هنا القاعدة الأولى في الغوص بالأسطوانة: \emph{لا تحبس نفَسك أبدًا أثناء الصعود}
--- واصِل التنفس فيخرج الهواء المتمدّد ببساطة. أما الغطّاسة الحرّة
فآمنة: لأن هواءها أُخذ عند السطح، فلا يفعل في طريق الصعود إلا
أن يعود إلى حجمه الأصلي.
\end{remark}

\begin{remark}[تخفيف الضغط]\label{rem:g10:pressure:deco}
تنفّس هواء عالي \omterm{def:g10:pressure:pressure}{الضغط} يذيب أيضًا نيتروجينًا زائدًا في الدم،
كالغاز في زجاجة مشروب غازي مغلقة؛ فإذا صعدت سريعًا جدًّا فار
فقاعات داخل الجسم --- وهو داء تخفيف \omterm{def:g10:pressure:pressure}{الضغط}. ومن هنا
الصعود البطيء (نحو \qty{10}{m} في الدقيقة) والوقفات الضحلة، التي تدع
الغاز يغادر بهدوء عبر الرئتين. أما النظرية فتنتظر مجلد السنة
الأولى؛ وأما قاعدة السلوك فتخصّ كل غطّاس.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:10}
باستعمال معطيات المحقنة في \cref{prop:g10:pressure:boyle}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:12}
\omterm{def:g10:orders-of-magnitude:scinot}{كأس} شفط مساحته \qty{12}{cm^2} يُضغط مسطَّحًا على سقف.
\begin{enumerate}
  \item خلاء تامّ في الداخل: فما \omterm{def:g10:inertia:force}{القوة} التي يمارسها الغلاف الجوي على

  \item ويبقي \omterm{def:g10:orders-of-magnitude:scinot}{كأس} حقيقي \qty{20}{kPa} من الهواء في الداخل. فأيّ كتلة الآن؟
\end{enumerate}
\end{exercise}

وكم تدوم الأسطوانة على \qty{30}{m} (\omterm{met:g10:pressure:dive}{والضغط المطلق}
$\approx \qty{4}{bar}$)؟ وعلى \qty{10}{m}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:14}
تسع رئتا غطّاس حرّ \qty{6.0}{L} عند السطح، ولا تقدران على
الانكماش دون الحجم المتبقي \qty{1.5}{L}. فعلى أي عمق بحري
($\rho = \qty{1025}{kg/m^3}$) يُبلغ ذلك الحدّ؟ (وقد أُعلن مرةً
حدَّ الغوص الحرّ؛ أما الأرقام القياسية فتتجاوز الآن \qty{100}{m}، إذ ينتقل

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:15}
في رافعة هيدروليكية، ينقل سائل \omterm{def:g10:pressure:pressure}{الضغط} دون تغيير من
مكبس صغير ($s = \qty{2.0}{cm^2}$) إلى مكبس كبير
($S = \qty{400}{cm^2}$) يحمل سيارة كتلتها \qty{1200}{kg}.
\begin{enumerate}
  \item أيّ \omterm{def:g10:inertia:force}{قوة} على المكبس الصغير تبقي السيارة مرفوعة؟ وأيّ \omterm{def:g10:pressure:pressure}{ضغط}
        يحمله السائل؟
  \item ولرفع السيارة \qty{2.0}{cm}، كم يجب أن ينتقل المكبس
        الصغير؟ قارن الشغل المبذول على كل مكبس.
\end{enumerate}
\end{exercise}

\section{مسألة: نفَس واحد نزولًا، ونفَس واحد صعودًا}

\begin{problem}[{مسألة نهاية الأسبوع --- نفَس واحد نزولًا ونفَس واحد صعودًا:
فيزياء غوصة حرّة وصعود بأسطوانة، من عصر القناع إلى
القاعدة القائلة إن الأمتار العشرة الأخيرة أخطرها}]
\label{pb:g10:pressure:1}
تغوص لينا غوصًا حرًّا: نفَس واحد عند السطح، نزولًا إلى \qty{30}{m} وعودةً.
ويغوص ماركو بجانبها بأسطوانة، يتنفس منها. البحر نفسه
($\rho = \qty{1025}{kg/m^3}$، $g = \qty{9.81}{N/kg}$،
$P_0 = \qty{101}{kPa}$)، والرئتان نفساهما \qty{6.0}{L}، والخطران متعاكسان ---
وكل ذلك يحكمه $P = P_0 + \rho g h$ و$PV$ = ثابت (ودرجة الحرارة
ثابتة طول الوقت).

\medskip
\noindent\textbf{الجزء I --- \omterm{def:g10:universal-gravitation:weight}{وزن} الماء.}
\begin{enumerate}
  \item احسب $\rho g$ لماء البحر \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{kPa} لكل \omterm{def:g10:orders-of-magnitude:unit}{متر}،
        وتحقّق من القاعدة التجريبية: جوّ واحد لكل \qty{10}{m}.
  \item احسب \omterm{met:g10:pressure:dive}{الضغط المطلق} على \qty{10}{m} و\qty{20}{m}
        و\qty{30}{m} و\qty{40}{m}.
  \item عبّر عن كلٍّ منها مضاعفًا \omterm{def:g10:pressure:pressure}{لضغط} السطح.
  \item يغطي قناع لينا \qty{150}{cm^2}. فعلى \qty{30}{m}، لو كان الهواء
        داخله ما زال عند \omterm{def:g10:pressure:pressure}{ضغط} السطح، فأيّ \omterm{def:g10:inertia:force}{قوة} محصّلة
        كان الماء سيمارسها؟ (والغطّاسون يزفرون في القناع عبر الأنف
        منعًا لهذا \emph{العصر للقناع}.)
  \item مساحة طبلة الأذن نحو \qty{0.60}{cm^2}. احسب \omterm{def:g10:inertia:force}{القوة} المحصّلة
        عليها على \qty{3.0}{m} لو بقيت الأذن الوسطى عند \omterm{def:g10:pressure:pressure}{ضغط}
        السطح؛ ولماذا تؤلم الآذان في مجرد مسبح، وما الذي يجب أن
        تحققه ``الموازنة''؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- نفَس واحد نزولًا.}
تغادر لينا السطح و\qty{6.0}{L} من الهواء في رئتيها.
\begin{enumerate}[resume]
  \item باستعمال \omterm{prop:g10:pressure:boyle}{قانون بويل} بالضغوط المطلقة، احسب حجم رئتيها
        على \qty{10}{m} و\qty{20}{m} و\qty{30}{m}.
  \item على أي عمق ينتصف حجم رئتيها؟
  \item هل يتناسب حجم الرئتين مع العمق؟ صِف منحني
        $V$ بدلالة \omterm{met:g10:pressure:dive}{الضغط المطلق} $P$.
  \item حجمها المتبقي \qty{1.5}{L}: فلا تقدر الرئتان على الانكماش
        أكثر. بيّن أن هذا الحدّ يُبلغ قرب \qty{30}{m}.
  \item ومع ذلك تتجاوز أرقام الغوص الحرّ القياسية \qty{100}{m}: فما الذي
        يملأ الحجم الناقص؟ (وأيّ شيء آخر يقدر أن يتدفق إلى الصدر؟)
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- نفَس واحد صعودًا.}
على \qty{30}{m}، يملأ منظّم ماركو رئتيه \qty{6.0}{L}.
\begin{enumerate}[resume]
  \item بأي \omterm{def:g10:pressure:pressure}{ضغط} يوصّل المنظّم ذلك الهواء؟ وكم مرة يكون
        أكثف من هواء السطح؟
  \item يُذعر ماركو فيحبس نفَسه ويصعد إلى \qty{10}{m}:
        فأيّ حجم يطلبه هواؤه المحبوس هناك؟
  \item وأيّ حجم عند السطح --- أي كم مرة سعة رئتيه؟
  \item اذكر القاعدة الأولى للغطّاس بالأسطوانة، ولماذا يزيل التنفس
        الطبيعي الخطر.
  \item وتحبس لينا نفَسها هي أيضًا من \qty{30}{m} إلى السطح،
        ومع ذلك تبقى رئتاها آمنتين تمامًا. اشرح عدم التناظر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الفقاعات، والأمتار العشرة الأخيرة.}
\begin{enumerate}[resume]
  \item يطلق ماركو فقاعة حجمها \qty{1.0}{cm^3} على \qty{40}{m}.
        احسب حجمها على \qty{30}{m} و\qty{20}{m} و\qty{10}{m}
        وتحت السطح مباشرة.
  \item ولكل مرحلة من \qty{10}{m} في الصعود، احسب العامل الذي
        تكبر به الفقاعة. فأين يكون النمو أسرع؟
  \item يصعد الغطّاسون \qty{10}{m} في الدقيقة على الأكثر: فكم يستغرق الصعود من
        \qty{40}{m}، وفي أي دقيقة يتمدد الغاز المحبوس أو المذاب
        أكثر ما يتمدد؟
  \item في العمق يذيب دم ماركو نيتروجينًا زائدًا، كالغاز في
        زجاجة مشروب غازي مغلقة. فماذا يفعل صعود سريع جدًّا، وكيف
        يمنعه الصعود البطيء ووقفة أمان مدتها \qty{5}{m}؟
  \item بيت القصيد: أعطِ، في جملة لكلٍّ، حكم الغطّاسة الحرّة،
        وحكم الغطّاس بالأسطوانة، والسبب الكمّي في كون
        الأمتار العشرة الأخيرة من أي صعود أجدرها بالاحترام.
\end{enumerate}
\end{problem}
