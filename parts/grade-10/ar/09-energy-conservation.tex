\chapter{الطاقة: أشكالها وانحفاظها}\label{ch:g10:energy-conservation}

سيارة متحركة، وهاتف مشحون، وفرن ساخن، ومطرقة مرفوعة، تشترك في
خاصية واحدة: فكلٌّ منها يقدر أن يُحدث شيئًا. وتقيس الفيزياء تلك
القدرة بعدد واحد --- هو \omterm{def:g10:energy-conservation:energy}{الطاقة} --- وتمسك دفاترها
بقاعدة حديدية واحدة: المجموع لا يتغير أبدًا، إنما ينتقل ويغيّر
لباسه. ويفتح هذا الفصل الدفتر: أشكال \omterm{def:g10:energy-conservation:energy}{الطاقة}،
والسلاسل التي تسلكها، وما يشتريه \omterm{def:g10:energy-conservation:kwh}{الكيلوواط ساعة}.

\section{الطاقة: العملة المشتركة}

\begin{definition}[الطاقة والجول]\label{def:g10:energy-conservation:energy}
\emph{الطاقة}\index{طاقة} تقيس قدرة جملة على إحداث
تغيرات: تحريك المادة، أو رفعها، أو تسخينها، أو إضاءتها. ووحدتها
هي \emph{الجول}\index{جول} (\unit{J}). ومخزونةً أو منقولةً
أو محوَّلةً --- يعدّ كل فرع من الفيزياء بهذه العملة الواحدة.
\end{definition}

\begin{definition}[الطاقة الحركية]\label{def:g10:energy-conservation:kinetic}
جسم كتلته $m$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{kg}) يتحرك بسرعة $v$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\unit{m/s}) يحمل \emph{الطاقة الحركية}\index{طاقة حركية}
$E_k = \tfrac12\, m v^2$: فمضاعفة الكتلة تضاعفها، ومضاعفة
السرعة تربّعها.
\end{definition}

\begin{example}[السيارة في مقابل الماشي]\label{ex:g10:energy-conservation:kinetic}
سيارة كتلتها \qty{1300}{kg} بسرعة \qty{100}{km/h} $= \qty{27.8}{m/s}$ تحمل
$E_k = \tfrac12 \times 1300 \times 27.8^2 \approx \qty{5.0e5}{J}$؛
وماشٍ كتلته \qty{70}{kg} بسرعة \qty{1.5}{m/s} يحمل
$\tfrac12 \times 70 \times 1.5^2 \approx \qty{79}{J}$ --- أي أقلّ ستة
آلاف مرة. فالسرعة غالية: والتربيع هو الذي يتكفّل بذلك.
\end{example}

\begin{definition}[الطاقة الكامنة الثقالية]\label{def:g10:energy-conservation:potential}
جسم كتلته $m$ على ارتفاع $h$ (\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{m}) فوق مستوى
مرجعي مختار يخزّن \emph{الطاقة الكامنة
الثقالية}\index{طاقة كامنة!ثقالية} $E_p = m g h$، حيث
$g = \qty{9.81}{N/kg}$ --- وهي صالحة قرب الأرض، حيث تكون $g$
ثابتة عمليًّا. ولا يهمّ إلا \emph{فروق} الارتفاع: أما
المستوى الذي عنده $E_p = 0$ فيُختار بحرية.
\end{definition}

\begin{example}[خزّان الماء]\label{ex:g10:energy-conservation:potential}
كل \omterm{def:g10:orders-of-magnitude:unit}{متر} مكعب (\qty{1000}{kg}) من الماء المخزون على ارتفاع \qty{30}{m} في
خزّان ماء يحوي $1000 \times 9.81 \times 30 \approx \qty{2.9e5}{J}$
--- أي نحو ستة أعشار \omterm{def:g10:energy-conservation:kinetic}{الطاقة الحركية} للسيارة أعلاه. فالمدن
تكدّس \omterm{def:g10:energy-conservation:energy}{الطاقة} فوق رؤوسها.
\end{example}

\begin{definition}[الأشكال الأخرى]\label{def:g10:energy-conservation:forms}
تختبئ \omterm{def:g10:energy-conservation:energy}{الطاقة} أيضًا في أثواب أقلّ ظهورًا: \emph{الطاقة
الحرارية}\index{طاقة حرارية}، وهي الاهتياج غير المنتظم لجزيئات
جسمٍ ما (فكلما سخن ازداد)؛ و\emph{الطاقة
الكيميائية}\index{طاقة كيميائية}، المخزونة في الروابط الجزيئية --- الغذاء
والخشب والبنزين والبطاريات؛ و\emph{الطاقة
الكهربائية}\index{طاقة كهربائية}، التي تحملها التيارات من المحطة إلى
المأخذ؛ و\emph{الطاقة الإشعاعية}\index{طاقة إشعاعية}، التي يحملها
الضوء حتى عبر الفضاء الخالي --- وبها تصلنا \omterm{def:g10:energy-conservation:energy}{طاقة} الشمس؛
و\emph{الطاقة النووية}\index{طاقة نووية}، المخزونة داخل نوى
الذرات، والمحرَّرة في النجوم (\cref{ch:g12:nuclear-energy}).
\end{definition}

\section{النقل وسلاسل التحويل}

\begin{definition}[النقل والتحويل والسلسلة]\label{def:g10:energy-conservation:transfer}
\emph{نقل الطاقة}\index{نقل الطاقة} ينقل \omterm{def:g10:energy-conservation:energy}{الطاقة}
بين الجمل --- أي بين \emph{الخزّانات}\index{خزّان (طاقة)}؛
أما \emph{تحويل الطاقة}\index{تحويل الطاقة} فيغيّر
شكلها. والجهاز الذي يفعل أحدهما (عنفة أو محرك أو مصباح أو عضلة) هو
\emph{محوّل}\index{محوّل}. و\emph{سلسلة
التحويل}\index{سلسلة التحويل} تصوّر الرحلة: صناديق
للخزّانات والأشكال، وسهام معنوَنة للنقل.
\end{definition}

\begin{method}[رسم سلسلة تحويل]\label{met:g10:energy-conservation:chain}
\begin{enumerate}
  \item عيّن الخزّان الابتدائي والشكل الابتدائي (ما الذي يُستهلك؟).
  \item تتبّع \omterm{def:g10:energy-conservation:energy}{الطاقة} محوّلًا محوّلًا: سهم واحد لكل
        نقل، واسم الشكل في كل صندوق.
  \item أضِف سهم حرارة عند كل \omterm{def:g10:energy-conservation:transfer}{محوّل} حقيقي --- فبعض \omterm{def:g10:energy-conservation:energy}{الطاقة}
        يتسرّب دائمًا حرارةً (\omterm{def:g10:inertia:inventory}{احتكاك}، أسلاك، عادم).
\end{enumerate}
\end{method}

ويجري النوّاس أبسط سلسلة على الإطلاق --- كامنة كلها عند
أعلى التأرجح، وحركية كلها عند أخفض نقطة (بأخذ
$E_p = 0$ هناك):

\section{انحفاظ الطاقة}

\begin{theorem}[انحفاظ الطاقة]\label{thm:g10:energy-conservation:conservation}
\index{انحفاظ الطاقة}
\omterm{def:g10:energy-conservation:energy}{الطاقة} الكلية لجملة معزولة --- أي جملة لا تتبادل شيئًا
مع الخارج --- ثابتة. \omterm{def:g10:energy-conservation:energy}{فالطاقة} لا تُخلق ولا
تُفنى: إنما تغيّر شكلها ومكانها، وما يفقده خزّان
تكسبه \omterm{def:g10:energy-conservation:transfer}{الخزّانات} الأخرى، جولًا \omterm{def:g10:energy-conservation:energy}{بجول}.
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:energy-conservation:admitted}
لم تُمسك تجربة قط هذا المبدأ مخفقًا؛ ونحن هنا نأخذه
مسلّمًا ونستعمله عدديًّا. ويشتقّ \cref{ch:g12:work-and-energy}
جزأه الميكانيكي ($E_k + E_p$) من قوانين نيوتن؛ أما
المحاسبة الكاملة فتنتظر المجلدات الجامعية.
\end{remark}

\begin{example}[السقوط الحر، مسعَّرًا بالجول]\label{ex:g10:energy-conservation:fall}
كرة كتلتها \qty{2.0}{kg} تسقط من \qty{10}{m} (ومقاومة الهواء
مهمَلة): فتبدأ \omterm{def:g10:energy-conservation:energy}{بطاقة}
$E_p = 2.0 \times 9.81 \times 10 \approx \qty{196}{J}$، وتهبط \omterm{def:g10:energy-conservation:energy}{بطاقة}
$E_k = \qty{196}{J}$، ومنه $v = \sqrt{2 \times 196/2.0} =
\qty{14}{m/s} \approx \qty{50}{km/h}$. فلا قوى ولا ساعة إيقاف:
إنما يسعّر الانحفاظ وحده الارتطام.
\end{example}

\section{المردود}

\begin{definition}[المردود]\label{def:g10:energy-conservation:efficiency}
يستهلك \omterm{def:g10:energy-conservation:transfer}{محوّل} $E_{\text{المستهلكة}}$ ويوصّل الشكل المطلوب
$E_{\text{المفيدة}}$؛ و\emph{مردوده}\index{مردود} هو
\[
\eta = \frac{E_{\text{المفيدة}}}{E_{\text{المستهلكة}}} < 1 .
\]
ويمنع الانحفاظ $\eta > 1$، ولا تبلغ المحوّلات الحقيقية $1$ أبدًا:
\omterm{def:g10:energy-conservation:energy}{والطاقة} الناقصة لم تُفنَ --- إنما غادرت حرارةً.
\end{definition}

\begin{example}[أين يذهب البنزين]\label{ex:g10:energy-conservation:engine}
يخزّن \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من البنزين \qty{45}{MJ} من \omterm{def:g10:energy-conservation:forms}{الطاقة الكيميائية}؛ ويحوّله
محرك سيارة \omterm{def:g10:energy-conservation:efficiency}{بمردود} $\eta \approx 0.30$: أي \qty{13.5}{MJ} من
الحركة، و\qty{31.5}{MJ} تسخّن العادم والمشعّ
والشارع. فثلثا كل خزّان يدفّئان الغلاف الجوي.
\end{example}

\begin{remark}[المردودات تتضارب]\label{rem:g10:energy-conservation:chain-eta}
المحوّلات المتسلسلة تضرب مردوداتها بعضها في بعض: فمحطة توليد
($0.38$) تغذّي الشبكة ($0.94$) التي تغذّي شاحنًا ($0.85$)
توصّل $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.30$. فالسلاسل
الطويلة أنابيب مثقوبة: كل سهم يفرض ضريبة على التدفق.
\end{remark}

\section{الاستطاعة والواط والكيلوواط ساعة}

\begin{definition}[الاستطاعة]\label{def:g10:energy-conservation:power}
\emph{الاستطاعة}\index{استطاعة} هي معدل تدفق \omterm{def:g10:energy-conservation:energy}{الطاقة}،
$P = E/t$، ووحدتها \emph{الواط}\index{واط} (\unit{W})، مع
$\qty{1}{W} = \qty{1}{J/s}$. \omterm{def:g10:energy-conservation:energy}{فالطاقة} كمية، والاستطاعة تدفق ---
كحوض استحمام وحنفيته.

\end{definition}

\begin{example}[الغلاية في مقابل الإنسان]\label{ex:g10:energy-conservation:power}
غلاية استطاعتها \qty{2200}{W} تشتغل \qty{150}{s} تستعمل
$E = 2200 \times 150 = \qty{3.3e5}{J}$. ويشتغل الإنسان على نحو
\qty{10}{MJ} من الغذاء يوميًّا، أي بمتوسط
$\num{1.0e7}/\num{86400} \approx \qty{116}{W}$: فأنت تخمل مثل
مصباح قديم ساطع.
\end{example}

\begin{definition}[الكيلوواط ساعة]\label{def:g10:energy-conservation:kwh}
\emph{الكيلوواط ساعة}\index{كيلوواط ساعة} (\unit{kW.h}) هو
\omterm{def:g10:energy-conservation:energy}{طاقة} تدفق قدره \qty{1}{kW} يشتغل ساعة واحدة:
$\qty{1}{kW.h} = 1000 \times 3600\,\unit{J} = \qty{3.6}{MJ}$ ---
وهي \omterm{def:g10:orders-of-magnitude:unit}{الوحدة} التي تعدّها عدّادات الكهرباء: أي \omterm{def:g10:energy-conservation:power}{الاستطاعة} \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{kW} في الساعات.
\end{definition}

\begin{example}[قراءة الفاتورة]\label{ex:g10:energy-conservation:bill}
الفاتورة طرح وضرب: قراءة العدّاد الجديدة
ناقص القديمة تعطي \omterm{def:g10:energy-conservation:kwh}{الكيلوواط ساعة}؛ وضربُها في سعر \omterm{def:g10:orders-of-magnitude:unit}{الوحدة} (نحو
$0.25$ يورو لكل \unit{kW.h})، زائدًا اشتراكًا ثابتًا، يعطي
المجموع. فمدفأة استطاعتها \qty{2}{kW} تشتغل \qty{2.5}{h} كل ليلة تستعمل
\qty{5}{kW.h} في اليوم: أي $38$ يورو في الشهر وحدها.
\end{example}

\begin{example}[رتب القدر]\label{ex:g10:energy-conservation:orders}
جدير بأن يُحفظ عن ظهر قلب: رفع تفاحة مترًا واحدًا، نحو \qty{1}{J}؛
وبطارية هاتف، $\qty{10}{W.h} \approx \qty{4e4}{J}$؛ وصحن حبوب،
\qty{1.5}{MJ}؛ وغذاؤك ليوم، \qty{10}{MJ}؛ \omterm{def:g10:orders-of-magnitude:unit}{وكيلوغرام}
من البنزين، \qty{45}{MJ} --- أي بالضبط تقريبًا كهرباء أسرة في يوم
($\approx \qty{12}{kW.h}$)؛ وصاعقة، \qty{5}{GJ}:
أي مئة يوم أسرة، تُسلَّم في ومضة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:1}
تسير سيارة كتلتها \qty{1200}{kg} بسرعة \qty{90}{km/h}، أي\ \qty{25}{m/s}:
احسب طاقتها الحركية. وأعِد الحساب عند \qty{130}{km/h}
(\qty{36.1}{m/s}): فماذا فعلت $44\%$ من السرعة الزائدة \omterm{def:g10:energy-conservation:energy}{بالطاقة} $E_k$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:2}
يصعد متنزّه كتلته \qty{72}{kg} من الارتفاع \qty{1200}{m} إلى
\qty{1650}{m}. احسب \omterm{def:g10:energy-conservation:energy}{الطاقة} الكامنة المكتسبة، وحوّلها إلى
\omterm{def:g10:energy-conservation:kwh}{الكيلوواط ساعة}، وسعّرها بسعر $0.25$ يورو لكل \unit{kW.h}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:3}

\section{مسألة: تغذية بيت ليوم واحد}

\begin{problem}[{مسألة نهاية الأسبوع --- من صحن الحبوب إلى ألواح
السطح: تدقيق طاقي كامل ليوم عادي في البيت، وكم يكلّف
الجول تبعًا لمن يبيعه}]
\label{pb:g10:energy-conservation:1}
في ثلاثاء عادي، تنطلق أسرة لتعرف أين تذهب
كهرباؤها، مسلَّحة بهذا الفصل وبعدّاد المدخل
وبظهر الفاتورة. وحكم هذه الليلة: يومهم يعادل
كيلوغرامًا واحدًا من البنزين، وستة عشر مترًا مربعًا من الشمس --- أو خمسة
راكبي دراجات يدوسون على \omterm{def:g10:universal-gravitation:satellite}{مدار} الساعة.

\medskip
\noindent\textbf{الجزء I --- التدقيق.}
جرد اليوم: ثلّاجة \qty{100}{W} مدة \qty{24}{h}؛ وسخّان
ماء \qty{2400}{W} مدة \qty{2.5}{h}؛ وفرن \qty{2000}{W} مدة
\qty{1.0}{h}؛ وغسّالة \qty{500}{W} مدة \qty{2.0}{h}؛ وإضاءة
\qty{60}{W} مدة \qty{5.0}{h}؛ وشاشات \qty{150}{W} مدة \qty{6.0}{h}.
\begin{enumerate}
  \item احسب \omterm{def:g10:energy-conservation:energy}{طاقة} كل جهاز في اليوم، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{kW.h}.
  \item اجمع اليوم \omterm{def:g10:energy-conservation:kwh}{بالكيلوواط ساعة}، ثم بالميغاجول؛
        وقارنه \omterm{def:g10:orders-of-magnitude:unit}{بكيلوغرام} واحد من البنزين (\qty{45}{MJ}).
  \item احسب \omterm{def:g10:energy-conservation:power}{الاستطاعة} المتوسطة للبيت على مدى \qty{24}{h}؛
        وأيّ جزء من غلاية شغّالة هي؟
  \item أيّ جهاز واحد يهيمن؟ أعِد حساب المجموع اليومي
        لو لم يشتغل السخّان إلا \qty{1.25}{h}.
  \item تسحب أضواء وضع الاستعداد \qty{15}{W} دائمًا: فما \omterm{def:g10:energy-conservation:energy}{الطاقة} اليومية،
        وحصتها من المجموع، والكلفة السنوية بسعر $0.25$ يورو لكل \unit{kW.h}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الفاتورة.}
\begin{enumerate}[resume]
  \item في 30 يومًا انتقل العدّاد من \qty{56214}{kW.h} إلى
        \qty{56604}{kW.h}: فما الاستهلاك، وما المتوسط اليومي مقارَنًا
        بالتدقيق؟
  \item احسب الفاتورة: $0.25$ يورو لكل \unit{kW.h} زائدًا
        اشتراكًا ثابتًا قدره $14$ يورو.
  \item ما سعر الميغاجول من الشبكة؟ ومن البنزين، بسعر
        $1.80$ يورو لكل لتر يخزّن \qty{34}{MJ}؟
  \item صحن حبوب \qty{1.5}{MJ} يكلّف نحو $0.50$ يورو: فما
        سعر ميغاجوله؟ ورتّب بائعي \omterm{def:g10:energy-conservation:energy}{الجول} الثلاثة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- السطح.}
\begin{enumerate}[resume]
  \item توصّل الشمس الكاملة نحو \qty{1000}{W} لكل \omterm{def:g10:orders-of-magnitude:unit}{متر}
        مربع؛ وتحوّلها الألواح \omterm{def:g10:energy-conservation:efficiency}{بمردود} $\eta = 0.20$: فما \omterm{def:g10:energy-conservation:power}{الاستطاعة}
        الكهربائية لمساحة \qty{1}{m^2} في شمس كاملة؟
  \item ويبلغ متوسط المنطقة ما يعادل \qty{4.0}{h} من الشمس
        الكاملة في اليوم: فما \omterm{def:g10:energy-conservation:efficiency}{المردود} اليومي لمساحة \qty{1}{m^2}؟
  \item أيّ مساحة ألواح تغطي يوم الأسرة البالغ \qty{12.6}{kW.h}؟
        ويعرض السطح الجنوبي \qty{20}{m^2}: فهل تسع؟
  \item وفي كانون الأول يهبط المعادل إلى \qty{1.5}{h}: فما إنتاج
        الألواح، وما حصته من الحاجة، وما الذي يسدّ
        الفجوة؟
  \item كلّف تصنيع الألواح نحو \qty{700}{kW.h} لكل \omterm{def:g10:orders-of-magnitude:unit}{متر}
        مربع: فما زمن استرداد \omterm{def:g10:energy-conservation:energy}{الطاقة}، باستعمال السؤال 11؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- صحن الحبوب.}
\begin{enumerate}[resume]
  \item تعطي دراجة مولّدة \qty{100}{W} من الكهرباء:
        فكم ساعة دوس تصنع \qty{12.6}{kW.h} --- وكم شخصًا
        يدوسون \qty{24}{h} بلا توقف؟
  \item وتشتغل العضلات \omterm{def:g10:energy-conservation:efficiency}{بمردود} $\eta \approx 0.25$: فما \omterm{def:g10:energy-conservation:energy}{الطاقة} الغذائية وراء
        تلك \qty{12.6}{kW.h}، بالميغاجول وبصحون الحبوب؟
  \item وما كلفة اليوم المدفوع بالدوس (والصحن بسعر $0.50$ يورو)، في مقابل
        ما تتقاضاه الشبكة عن اليوم نفسه؟
  \item صاعقة قدرها \qty{5}{GJ} تعادل كم يوم بيت؟
        وبالنظر إلى مدتها \qty{100}{\micro s}، لماذا تبقى عاجزة
        عن تغذية البيت؟
  \item \omterm{def:g10:energy-conservation:energy}{الجول} لا يضيع أبدًا، ومع ذلك تُباع جولات الشبكة والبنزين
        والغذاء بأسعار شديدة الاختلاف: فما الذي تدفع ثمنه فعلًا؟
  \item الخاتمة: اذكر حكم اليوم في جملة واحدة --- \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام}
        واحد من البنزين، أو ستة عشر مترًا مربعًا من سطح مشمس، أو
        خمسة راكبي دراجات على \omterm{def:g10:universal-gravitation:satellite}{مدار} الساعة --- وسمِّ المقدارين
        (أحدهما مخزون والآخر تدفق) اللذين لا يجوز الخلط بينهما بعد اليوم.
\end{enumerate}
\end{problem}
