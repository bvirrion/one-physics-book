\chapter{La energía: formas y conservación}\label{ch:g10:energy-conservation}

Un coche en marcha, un móvil cargado, un horno caliente y un martillo
levantado comparten una propiedad: cada uno puede hacer que ocurra algo.
La física mide esa capacidad con un solo número --- la \omterm{def:g10:energy-conservation:energy}{energía} --- y
lleva sus cuentas bajo una regla de hierro: el total no cambia nunca,
solo se desplaza y se cambia de disfraz. Este capítulo abre el libro de
cuentas: las formas que toma la \omterm{def:g10:energy-conservation:energy}{energía}, las cadenas que recorre y lo
que compra un \omterm{def:g10:energy-conservation:kwh}{kilovatio-hora}.

\section{La energía: la moneda común}

\begin{definition}[La energía y el julio]\label{def:g10:energy-conservation:energy}
La \emph{energía}\index{energía} mide la capacidad de un sistema para
producir cambios: poner materia en movimiento, levantarla, calentarla,
iluminarla. Su \omterm{def:g10:orders-of-magnitude:unit}{unidad} es el \emph{julio}\index{julio} (\unit{J}).
Almacenada, transferida, convertida --- todas las ramas de la física
cuentan en esta única moneda.
\end{definition}

\begin{definition}[Energía cinética]\label{def:g10:energy-conservation:kinetic}
Un cuerpo de masa $m$ (en \unit{kg}) que se mueve a velocidad $v$ (en
\unit{m/s}) lleva una \emph{energía cinética}\index{energía cinética}
$E_k = \tfrac12\, m v^2$: duplicar la masa la duplica, duplicar la
velocidad la cuadruplica.
\end{definition}

\begin{example}[El coche frente al peatón]\label{ex:g10:energy-conservation:kinetic}
Un coche de \qty{1300}{kg} a \qty{100}{km/h} $= \qty{27.8}{m/s}$ lleva
$E_k = \tfrac12 \times 1300 \times 27.8^2 \approx \qty{5.0e5}{J}$; un
peatón de \qty{70}{kg} a \qty{1.5}{m/s},
$\tfrac12 \times 70 \times 1.5^2 \approx \qty{79}{J}$ --- seis mil veces
menos. La velocidad sale cara: de eso se encarga el cuadrado.
\end{example}

\begin{definition}[Energía potencial gravitatoria]\label{def:g10:energy-conservation:potential}
Un cuerpo de masa $m$ situado a una altura $h$ (en \unit{m}) por encima
de un nivel de referencia elegido almacena la \emph{energía potencial
gravitatoria}\index{energía potencial!gravitatoria} $E_p = m g h$, con
$g = \qty{9.81}{N/kg}$ --- válida cerca del suelo, donde $g$ es
prácticamente constante. Solo cuentan las \emph{diferencias} de altura:
el nivel en el que $E_p = 0$ se elige libremente.
\end{definition}

\begin{example}[Un depósito elevado]\label{ex:g10:energy-conservation:potential}
Cada \omterm{def:g10:orders-of-magnitude:unit}{metro} cúbico (\qty{1000}{kg}) de agua almacenado \qty{30}{m} arriba
en un depósito guarda
$1000 \times 9.81 \times 30 \approx \qty{2.9e5}{J}$ --- unas seis
décimas de la \omterm{def:g10:energy-conservation:kinetic}{energía cinética} del coche anterior. Las ciudades
almacenan \omterm{def:g10:energy-conservation:energy}{energía} por encima de sus cabezas.
\end{example}

\begin{definition}[Las demás formas]\label{def:g10:energy-conservation:forms}
La \omterm{def:g10:energy-conservation:energy}{energía} también se esconde bajo disfraces menos visibles: la
\emph{energía térmica}\index{energía térmica}, la agitación desordenada
de las moléculas de un cuerpo (cuanto más caliente, mayor); la
\emph{energía química}\index{energía química}, almacenada en los enlaces
moleculares --- alimentos, madera, gasolina, pilas; la \emph{energía
eléctrica}\index{energía eléctrica}, que las corrientes llevan de la
central al enchufe; la \emph{energía radiante}\index{energía radiante},
que la luz transporta incluso a través del espacio vacío --- así nos
llega la \omterm{def:g10:energy-conservation:energy}{energía} del Sol; y la \emph{energía
nuclear}\index{energía nuclear}, almacenada en el interior de los
\omterm{def:g10:refraction:fiber}{núcleos} atómicos y liberada en las estrellas
(\cref{ch:g12:nuclear-energy}).
\end{definition}

\section{Transferencias y cadenas de conversión}

\begin{definition}[Transferencia, conversión, cadena]\label{def:g10:energy-conservation:transfer}
Una \emph{transferencia de energía}\index{transferencia de energía}
desplaza \omterm{def:g10:energy-conservation:energy}{energía} entre sistemas --- los
\emph{depósitos}\index{depósito (de energía)}; una \emph{conversión de
energía}\index{conversión de energía} cambia su forma. Un dispositivo
que hace una cosa u otra (turbina, motor, lámpara, músculo) es un
\emph{convertidor}\index{convertidor}. Una \emph{cadena de
conversión}\index{cadena de conversión} dibuja el viaje: cajas para los
depósitos y las formas, flechas rotuladas para las transferencias.
\end{definition}

\begin{method}[Dibujar una cadena de conversión]\label{met:g10:energy-conservation:chain}
\begin{enumerate}
  \item Identificar el depósito y la forma iniciales (¿qué se
        consume?).
  \item Seguir la \omterm{def:g10:energy-conservation:energy}{energía} \omterm{def:g10:energy-conservation:transfer}{convertidor} a \omterm{def:g10:energy-conservation:transfer}{convertidor}: una flecha por
        transferencia y la forma nombrada en cada caja.
  \item Añadir una flecha de calor en cada \omterm{def:g10:energy-conservation:transfer}{convertidor} real --- siempre
        se escapa algo de \omterm{def:g10:energy-conservation:energy}{energía} como calor (\omterm{def:g10:inertia:inventory}{rozamiento}, cables,
        escape).
\end{enumerate}
\end{method}

lo alto de la oscilación, todo cinético al pasar por el punto más bajo
(tomando allí $E_p = 0$):

\section{La conservación de la energía}

\begin{theorem}[Conservación de la energía]\label{thm:g10:energy-conservation:conservation}
\index{conservación de la energía}
La \omterm{def:g10:energy-conservation:energy}{energía} total de un sistema aislado --- uno que no intercambia nada
con el exterior --- es constante. La \omterm{def:g10:energy-conservation:energy}{energía} no se crea nunca ni se
destruye nunca: solo cambia de forma y de lugar, y lo que un depósito
pierde lo ganan los demás, \omterm{def:g10:energy-conservation:energy}{julio} a \omterm{def:g10:energy-conservation:energy}{julio}.
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:energy-conservation:admitted}
Ningún experimento ha pillado nunca en falta a este principio; aquí lo
damos por bueno y lo usamos con números. El
\cref{ch:g12:work-and-energy} deduce su parte mecánica ($E_k + E_p$) de
las leyes de Newton; la contabilidad completa espera a los volúmenes
universitarios.
\end{remark}

\begin{example}[Caída libre, tasada en julios]\label{ex:g10:energy-conservation:fall}
Una bola de \qty{2.0}{kg} cae desde \qty{10}{m} (resistencia del aire
despreciable): parte con
$E_p = 2.0 \times 9.81 \times 10 \approx \qty{196}{J}$ y llega con
$E_k = \qty{196}{J}$, luego
$v = \sqrt{2 \times 196/2.0} = \qty{14}{m/s} \approx \qty{50}{km/h}$.
Sin \omterm{def:g10:inertia:force}{fuerzas} y sin cronómetro: la conservación sola tasa el impacto.
\end{example}

\section{El rendimiento}

\begin{definition}[Rendimiento]\label{def:g10:energy-conservation:efficiency}
Un \omterm{def:g10:energy-conservation:transfer}{convertidor} consume $E_{\text{consumed}}$ y entrega la forma deseada
$E_{\text{useful}}$; su \emph{rendimiento}\index{rendimiento} es
\[
\eta = \frac{E_{\text{useful}}}{E_{\text{consumed}}} < 1 .
\]
La conservación prohíbe $\eta > 1$, y ningún \omterm{def:g10:energy-conservation:transfer}{convertidor} real llega a
$1$: la \omterm{def:g10:energy-conservation:energy}{energía} que falta no se ha destruido --- se va en forma de
calor.
\end{definition}

\begin{example}[Adónde va la gasolina]\label{ex:g10:energy-conservation:engine}
Un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina almacena \qty{45}{MJ} de \omterm{def:g10:energy-conservation:forms}{energía química}; el
motor de un coche la convierte con $\eta \approx 0.30$: \qty{13.5}{MJ}
de movimiento y \qty{31.5}{MJ} que calientan el escape, el radiador y la
calle. Dos tercios de cada depósito calientan la atmósfera.
\end{example}

\begin{remark}[Los rendimientos se multiplican]\label{rem:g10:energy-conservation:chain-eta}
Los \omterm{def:g10:energy-conservation:transfer}{convertidores} encadenados multiplican sus \omterm{def:g10:energy-conservation:efficiency}{rendimientos}: una central
($0.38$) que alimenta la red ($0.94$) que alimenta un cargador ($0.85$)
entrega $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.30$. Las cadenas
largas son tuberías con fugas: cada flecha cobra su peaje.
\end{remark}

\section{La potencia, el vatio y el kilovatio-hora}

\begin{definition}[Potencia]\label{def:g10:energy-conservation:power}
La \emph{potencia}\index{potencia} es el ritmo al que fluye la \omterm{def:g10:energy-conservation:energy}{energía},
$P = E/t$, medida en \emph{vatios}\index{vatio} (\unit{W}), con
$\qty{1}{W} = \qty{1}{J/s}$. La \omterm{def:g10:energy-conservation:energy}{energía} es una cantidad y la potencia,
un caudal --- la bañera frente al grifo.
\end{definition}

\begin{example}[El hervidor frente a la persona]\label{ex:g10:energy-conservation:power}
Un hervidor de \qty{2200}{W} funcionando \qty{150}{s} gasta
$E = 2200 \times 150 = \qty{3.3e5}{J}$. Una persona funciona con unos
\qty{10}{MJ} de comida al día, una media de
$\num{1.0e7}/\num{86400} \approx \qty{116}{W}$: en reposo consumes lo
que una bombilla vieja bien encendida.
\end{example}

\begin{definition}[El kilovatio-hora]\label{def:g10:energy-conservation:kwh}
El \emph{kilovatio-hora}\index{kilovatio-hora} (\unit{kW.h}) es la
\omterm{def:g10:energy-conservation:energy}{energía} de un caudal de \qty{1}{kW} que dura una hora:
$\qty{1}{kW.h} = 1000 \times 3600\,\unit{J} = \qty{3.6}{MJ}$ --- la
\omterm{def:g10:orders-of-magnitude:unit}{unidad} que cuentan los contadores eléctricos: \omterm{def:g10:energy-conservation:power}{potencia} en \unit{kW} por
horas.
\end{definition}

\begin{example}[Leer la factura]\label{ex:g10:energy-conservation:bill}
Una factura es una resta y una multiplicación: la lectura nueva del
contador menos la vieja da los kilovatios-hora; multiplicados por el
precio unitario (unos $0.25$ euros por \unit{kW.h}) y sumada la cuota
fija, dan el total. Una estufa de \qty{2}{kW} encendida \qty{2.5}{h}
cada noche gasta \qty{5}{kW.h} al día: $38$ euros al mes ella sola.
\end{example}

\begin{example}[Órdenes de magnitud]\label{ex:g10:energy-conservation:orders}
Conviene saberlos de memoria: subir una manzana un \omterm{def:g10:orders-of-magnitude:unit}{metro}, aproximadamente
\qty{1}{J}; la batería de un móvil,
$\qty{10}{W.h} \approx \qty{4e4}{J}$; un tazón de cereales,
\qty{1.5}{MJ}; tu comida de un día, \qty{10}{MJ}; un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de
gasolina, \qty{45}{MJ} --- casi exactamente la electricidad diaria de un
hogar ($\approx \qty{12}{kW.h}$); un rayo, \qty{5}{GJ}: cien días de
hogar, entregados en un destello.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:1}
Un coche de \qty{1200}{kg} circula a \qty{90}{km/h}, es decir
\qty{25}{m/s}: calcula su \omterm{def:g10:energy-conservation:kinetic}{energía cinética}. Repítelo a \qty{130}{km/h}
(\qty{36.1}{m/s}): ¿qué le ha hecho a $E_k$ un $44\,\%$ más de velocidad?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:2}
Un excursionista de \qty{72}{kg} sube de la altitud \qty{1200}{m} a la
de \qty{1650}{m}. Calcula la \omterm{def:g10:energy-conservation:energy}{energía} potencial ganada, conviértela a
kilovatios-hora y ponle precio a $0.25$ euros el \unit{kW.h}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:3}
Convierte: \qty{1}{kW.h} a \omterm{def:g10:energy-conservation:energy}{julios}; la batería de móvil de \qty{12}{W.h}
a kilojulios; \qty{45}{MJ} (un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina) a
kilovatios-hora; y tu día de \qty{10}{MJ} de comida a kilovatios-hora.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:4}
Un hervidor de \qty{2200}{W} funciona \qty{150}{s}. Calcula la \omterm{def:g10:energy-conservation:energy}{energía}
gastada, en \omterm{def:g10:energy-conservation:energy}{julios} y en kilovatios-hora. Calcula después la \omterm{def:g10:energy-conservation:power}{potencia}
media de una persona que funciona con \qty{10}{MJ} al día.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:5}
Escribe la \omterm{def:g10:energy-conservation:transfer}{cadena de conversión}
(\cref{met:g10:energy-conservation:chain}) de: una dinamo de bicicleta
que alimenta una lámpara; un calentador de agua de gas; un panel solar
que carga un móvil. Señala las fugas de calor.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:6}
Un motor quema \qty{2.0}{kg} de gasolina (\qty{45}{MJ/kg}) y entrega
\qty{27}{MJ} de movimiento. Calcula su \omterm{def:g10:energy-conservation:efficiency}{rendimiento} y la \omterm{def:g10:energy-conservation:energy}{energía} liberada

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:7}
Un saltador de \qty{70}{kg} se lanza desde un acantilado de \qty{20}{m}
(resistencia del aire despreciable). Con la conservación, calcula la
\omterm{def:g10:energy-conservation:kinetic}{energía cinética} al llegar al agua y la velocidad de entrada, en
\unit{m/s} y en \unit{km/h}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:8}
Un contador marcaba \qty{41236}{kW.h} el 1 de marzo y \qty{41562}{kW.h}
30 días después: ¿\omterm{def:g10:energy-conservation:energy}{energía} gastada, media diaria, factura a $0.26$ euros
el \unit{kW.h} más $12$ euros de cuota fija, y \omterm{def:g10:energy-conservation:power}{potencia} media en \omterm{def:g10:energy-conservation:power}{vatios}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:9}
Una bombilla vieja de \qty{60}{W} convierte en luz solo el $5\,\%$ de lo
que consume; un led da la misma luz con \qty{9}{W}. Calcula la \omterm{def:g10:energy-conservation:power}{potencia}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:10}
Una presa deja pasar \qty{150}{m^3} de agua (\qty{1.5e5}{kg}) por
segundo por un salto de \qty{40}{m}. Calcula la \omterm{def:g10:energy-conservation:energy}{energía} potencial

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:11}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:12}

\section{Problema: dar energía a una casa durante un día}

\begin{problem}[{Problema de fin de semana --- del tazón de cereales a
los paneles del tejado: una auditoría energética completa de un día
corriente en casa y lo que cuesta un julio según quién lo venda}]
\label{pb:g10:energy-conservation:1}
Un martes cualquiera, una familia se propone averiguar adónde va su
electricidad, armada con este capítulo, el contador del pasillo y el
dorso de la factura. Veredicto de esta noche: su día equivale a un
\omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina, dieciséis \omterm{def:g10:orders-of-magnitude:unit}{metros} cuadrados de sol --- o cinco
ciclistas pedaleando sin parar.

\medskip
\noindent\textbf{Parte I --- La auditoría.}
Inventario del día: frigorífico \qty{100}{W} durante \qty{24}{h};
calentador de agua \qty{2400}{W} durante \qty{2.5}{h}; horno
\qty{2000}{W} durante \qty{1.0}{h}; lavadora \qty{500}{W} durante
\qty{2.0}{h}; iluminación \qty{60}{W} durante \qty{5.0}{h}; pantallas
\qty{150}{W} durante \qty{6.0}{h}.
\begin{enumerate}
  \item Calcula la \omterm{def:g10:energy-conservation:energy}{energía} diaria de cada aparato, en \unit{kW.h}.
  \item Suma el día en kilovatios-hora y después en megajulios;
        compáralo con un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina (\qty{45}{MJ}).
  \item Calcula la \omterm{def:g10:energy-conservation:power}{potencia} media de la casa a lo largo de las
        \qty{24}{h}; ¿qué fracción de un hervidor en marcha es?
  \item ¿Qué aparato domina él solo? Recalcula el total diario si el
        calentador hubiera funcionado únicamente \qty{1.25}{h}.
  \item Los pilotos en espera consumen \qty{15}{W} de forma permanente:
        ¿\omterm{def:g10:energy-conservation:energy}{energía} diaria, porcentaje del total y coste anual a $0.25$
        euros el \unit{kW.h}?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La factura.}
\begin{enumerate}[resume]
  \item En 30 días el contador pasó de \qty{56214}{kW.h} a
        \qty{56604}{kW.h}: ¿consumo y media diaria comparada con la
        auditoría?
  \item Calcula la factura: $0.25$ euros el \unit{kW.h} más una cuota
        fija de $14$ euros.
  \item ¿Precio de un megajulio de la red? ¿Y de gasolina, a $1.80$
        euros el litro, que almacena \qty{34}{MJ}?
  \item Un tazón de cereales de \qty{1.5}{MJ} cuesta unos $0.50$ euros:
        ¿su precio por megajulio? Ordena a los tres vendedores de
        \omterm{def:g10:energy-conservation:energy}{julios}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El tejado.}
\begin{enumerate}[resume]
  \item El sol a pleno \omterm{def:g10:energy-conservation:efficiency}{rendimiento} entrega unos \qty{1000}{W} por \omterm{def:g10:orders-of-magnitude:unit}{metro}
        cuadrado; los paneles lo convierten con $\eta = 0.20$: ¿\omterm{def:g10:energy-conservation:power}{potencia}
        eléctrica de \qty{1}{m^2} a pleno sol?
  \item La región tiene una media equivalente a \qty{4.0}{h} de sol
        pleno al día: ¿producción diaria de \qty{1}{m^2}?
  \item ¿Qué superficie de paneles cubre el día de \qty{12.6}{kW.h} de
        la familia? El tejado sur ofrece \qty{20}{m^2}: ¿cabe?
  \item En diciembre el equivalente baja a \qty{1.5}{h}: ¿la producción
        de los paneles, su porcentaje de la necesidad y qué cubre la
        diferencia?
  \item Fabricar los paneles costó unos \qty{700}{kW.h} por \omterm{def:g10:orders-of-magnitude:unit}{metro}
        cuadrado: ¿tiempo de retorno energético, usando la pregunta 11?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El tazón de cereales.}
\begin{enumerate}[resume]
  \item Una bicicleta generadora produce \qty{100}{W} de electricidad:
        ¿cuántas horas de pedaleo hacen \qty{12.6}{kW.h} --- cuántas
        personas pedaleando \qty{24}{h} sin parar?
  \item Los músculos trabajan con $\eta \approx 0.25$: ¿la \omterm{def:g10:energy-conservation:energy}{energía}
        alimentaria detrás de esos \qty{12.6}{kW.h}, en megajulios y en
        tazones de cereales?
  \item ¿Coste del día pedaleado (tazones a $0.50$ euros) frente a lo
        que cobra la red por ese mismo día?
  \item Un rayo de \qty{5}{GJ} equivale a cuántos días de casa? Dada su
        duración de \qty{100}{\micro s}, ¿por qué sigue sin poder
        alimentar la casa?
  \item Los \omterm{def:g10:energy-conservation:energy}{julios} no se pierden nunca y, sin embargo, los de la red,
        los de la gasolina y los de la comida se venden a precios muy
        distintos: ¿qué estás pagando en realidad?
  \item Final: enuncia el veredicto del día en una frase --- un
        \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina, dieciséis \omterm{def:g10:orders-of-magnitude:unit}{metros} cuadrados de tejado
        soleado o cinco ciclistas sin descanso --- y nombra las dos
        magnitudes (una de existencias y otra de caudal) que no hay que
        volver a confundir.
\end{enumerate}
\end{problem}
