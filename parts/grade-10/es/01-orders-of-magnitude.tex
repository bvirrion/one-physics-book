\chapter{Órdenes de magnitud: medir el universo}\label{ch:g10:orders-of-magnitude}

La física empieza donde acaba la opinión: en una medida. Pero las cosas
que la física mide abarcan un intervalo absurdo --- un núcleo atómico
mide unos \qty{e-15}{m} y el universo observable unos \qty{e27}{m}: un
factor $10^{42}$, mucho más de lo que cabe en una imagen mental. Este
capítulo instala las herramientas que domestican ese intervalo ---
\omterm{def:g10:orders-of-magnitude:unit}{unidades}, potencias de diez, \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas} --- y dos costumbres
que se usarán en todas las páginas siguientes: escribir la \omterm{def:g10:orders-of-magnitude:unit}{unidad} y
comprobar el \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud}.

\section{Las unidades: la gramática de la medida}

\begin{definition}[Magnitudes físicas y unidades del SI]\label{def:g10:orders-of-magnitude:unit}
Una \emph{magnitud física}\index{magnitud física} es una propiedad del
mundo que se puede medir: una longitud, una duración, una masa. Medirla
consiste en contar cuántas veces cabe en ella una magnitud de
referencia: la \emph{unidad}\index{unidad}. El \emph{Sistema
Internacional de Unidades}\index{Sistema Internacional de Unidades
(SI)} (SI) fija siete unidades básicas, de las cuales tres gobiernan
todo este volumen: el \emph{metro}\index{metro} (\unit{m}) para la
longitud, el \emph{kilogramo}\index{kilogramo} (\unit{kg}) para la masa
y el \emph{segundo}\index{segundo} (\unit{s}) para el tiempo. (Las otras
cuatro --- amperio, kelvin, mol, candela --- entrarán en escena con la
electricidad, el calor y la química.)
\end{definition}

\begin{remark}[Un número sin unidad no es un resultado]\label{rem:g10:orders-of-magnitude:units}
``La distancia es $384$'' no dice nada: \qty{384}{mm},
\qty{384}{km} y \qty{384}{Mm} son tres mundos distintos (el último
resulta ser la distancia a la Luna). En 1999 la NASA perdió el
\emph{Mars Climate Orbiter}, de $327$ millones de dólares, porque un
programa calculaba las fuerzas de los propulsores en libras-fuerza y
otro las leía como newtons. Norma de la casa, desde esta línea hasta el
final del libro: todo número medido o calculado lleva su \omterm{def:g10:orders-of-magnitude:unit}{unidad}, en cada
renglón del cálculo.
\end{remark}

\begin{definition}[Prefijos del SI]\label{def:g10:orders-of-magnitude:prefixes}
Un \emph{prefijo}\index{prefijo (SI)} pegado a una \omterm{def:g10:orders-of-magnitude:unit}{unidad} la multiplica
por una potencia de diez fija:
\begin{center}
\begin{tabular}{llc@{\qquad\qquad}llc}
femto & f & $10^{-15}$ & centi & c & $10^{-2}$ \\
pico  & p & $10^{-12}$ & kilo  & k & $10^{3}$  \\
nano  & n & $10^{-9}$  & mega  & M & $10^{6}$  \\
micro & $\mu$ & $10^{-6}$ & giga & G & $10^{9}$ \\
mili  & m & $10^{-3}$  & tera  & T & $10^{12}$ \\
\end{tabular}
\end{center}
Así, $\qty{1}{nm} = \qty{e-9}{m}$ y $\qty{1}{km} = \qty{e3}{m}$.
\end{definition}

\begin{example}[Leer los prefijos]\label{ex:g10:orders-of-magnitude:prefixes}
La luz verde tiene una longitud de onda de unos
$\qty{500}{nm} = \qty{5.00e-7}{m}$; un cabello humano mide unos
$\qty{70}{\micro m} = \qty{7.0e-5}{m}$ de grosor; la Luna orbita a
$\qty{384}{Mm} = \qty{3.84e8}{m}$. Una gota de lluvia tiene una masa de
unos $\qty{30}{mg} = \qty{3.0e-5}{kg}$ --- obsérvese que la \omterm{def:g10:orders-of-magnitude:unit}{unidad}
básica de masa del SI es la única de las siete que ya lleva \omterm{def:g10:orders-of-magnitude:prefixes}{prefijo}.
\end{example}

\section{Notación científica y cifras significativas}

\begin{definition}[Notación científica]\label{def:g10:orders-of-magnitude:scinot}
Un número está en \emph{notación científica}\index{notación científica}
cuando se escribe
\[
a \times 10^{n}, \qquad 1 \leq a < 10, \quad n \text{ entero.}
\]
El factor $a$ es la \emph{mantisa}\index{mantisa} y el entero $n$ el
\emph{exponente}\index{exponente}. Así,
$\num{149600000} = \num{1.496e8}$ y $\num{0.00052} = \num{5.2e-4}$.
\end{definition}

\begin{proposition}[Calcular con potencias de diez]\label{prop:g10:orders-of-magnitude:powers}
Para todos los enteros $m$ y $n$:
\[
10^{m} \times 10^{n} = 10^{m+n}, \qquad
\frac{1}{10^{n}} = 10^{-n}, \qquad
\left(10^{m}\right)^{n} = 10^{mn}.
\]
\end{proposition}

\begin{proof}[Demostración parcial]
Para \omterm{def:g10:orders-of-magnitude:scinot}{exponentes} positivos basta contar los factores: $10^{m} \times
10^{n}$ es un producto de $m + n$ dieces, y $(10^m)^n$ encadena $n$
grupos de $m$ dieces. Escribir $10^{-n}$ en lugar de $1/10^{n}$ es
precisamente el convenio que mantiene válida la primera regla para todo
entero: obliga a que $10^{n} \times 10^{-n} = 10^{0} = 1$. El volumen de
matemáticas correspondiente estudia a fondo las reglas de los
\omterm{def:g10:orders-of-magnitude:scinot}{exponentes}.
\end{proof}

\begin{example}[Multiplicar en notación científica]\label{ex:g10:orders-of-magnitude:scinot}
¿Cuántas Tierras caben en el Sol, en masa? Con
$m_{\text{Sun}} = \qty{1.99e30}{kg}$ y
$m_{\text{Earth}} = \qty{5.97e24}{kg}$:
\[
\frac{\num{1.99e30}}{\num{5.97e24}}
= \frac{1.99}{5.97} \times 10^{30-24}
= 0.333 \times 10^{6}
= \num{3.33e5}.
\]
Las \omterm{def:g10:orders-of-magnitude:scinot}{mantisas} y los \omterm{def:g10:orders-of-magnitude:scinot}{exponentes} se tratan por separado; el resultado final
se vuelve a normalizar para que la \omterm{def:g10:orders-of-magnitude:scinot}{mantisa} caiga de nuevo en $[1, 10)$.
\end{example}

\begin{definition}[Cifras significativas]\label{def:g10:orders-of-magnitude:sigfig}
Las \emph{cifras significativas}\index{cifras significativas} de un
número medido son los dígitos que la medida garantiza realmente: todos
los escritos salvo los ceros iniciales. Los ceros finales cuentan ---
$\qty{5.20e-3}{s}$ (tres cifras significativas) promete más que
$\qty{5.2e-3}{s}$ (dos). La \omterm{def:g10:orders-of-magnitude:scinot}{notación científica} muestra el recuento de
un vistazo: la \omterm{def:g10:orders-of-magnitude:scinot}{mantisa} lleva exactamente las cifras significativas.
\end{definition}

\begin{method}[Cuántos dígitos conservar]\label{met:g10:orders-of-magnitude:sigfig}
Un resultado calculado no puede ser más preciso que sus ingredientes.
\begin{enumerate}
  \item En un producto o un cociente, se conservan tantas \omterm{def:g10:orders-of-magnitude:sigfig}{cifras
        significativas} como tenga el factor \emph{menos} preciso.
  \item Se redondea solo el resultado final, nunca los pasos
        intermedios.
  \item En caso de duda, tres \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas}: es la precisión de
        trabajo de este libro.
\end{enumerate}
\end{method}

\begin{example}[La circunferencia de la Tierra]\label{ex:g10:orders-of-magnitude:sigfig}
El diámetro de la Tierra es $d = \qty{1.27e7}{m}$ (tres \omterm{def:g10:orders-of-magnitude:sigfig}{cifras
significativas}), de modo que su circunferencia vale
$\pi d = \num{3.14159} \times \qty{1.27e7}{m} = \qty{3.99e7}{m}$ ---
tres cifras, no las ocho que muestra la calculadora. Escribir
$\qty{39898227}{m}$ sería afirmar que conocemos el planeta con precisión
de un \omterm{def:g10:orders-of-magnitude:unit}{metro}.
\end{example}

\section{La escalera del universo}

\begin{definition}[Orden de magnitud]\label{def:g10:orders-of-magnitude:oom}
El \emph{orden de magnitud}\index{orden de magnitud} de una cantidad es
la potencia de diez más próxima a ella: se escribe la cantidad como
$a \times 10^{n}$ con $1 \leq a < 10$ y se toma $10^{n}$ si $a < 5$ y
$10^{n+1}$ si $a \geq 5$. (El convenio exacto en la frontera nunca
importa: una afirmación de orden de magnitud solo pretende ser válida
salvo un factor diez.)
\end{definition}

\begin{example}[Órdenes de magnitud]\label{ex:g10:orders-of-magnitude:oom}
Una vida humana dura unos $80$ años, es decir
$80 \times \qty{3.16e7}{s} \approx \qty{2.5e9}{s}$: \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud}
$\qty{e9}{s}$ --- unos pocos miles de millones de segundos, conviene
gastarlos bien. Una persona mide $\qty{1.7}{m}$: orden $\qty{e0}{m}$. El
Everest, $\qty{8848}{m}$: orden $\qty{e4}{m}$, porque $8.848 \geq 5$.
\end{example}

Apilar estas estimaciones por \omterm{def:g10:orders-of-magnitude:scinot}{exponente} produce la imagen maestra del
capítulo: una escalera en la que cada peldaño es un mundo diez veces más
ancho que el de abajo.

Por encima del peldaño de la Tierra, los \omterm{def:g10:orders-of-magnitude:unit}{metros} dejan de ser cómodos; la
astronomía tiene sus propias \omterm{def:g10:orders-of-magnitude:unit}{unidades}, definidas ambas a partir de la
escalera misma.

\begin{definition}[Unidad astronómica y año luz]\label{def:g10:orders-of-magnitude:lightyear}
La \emph{unidad astronómica}\index{unidad astronómica}
($\qty{1}{au} = \qty{1.496e11}{m}$) es la distancia media entre la
Tierra y el Sol. El \emph{año luz}\index{año luz} (\unit{ly}) es la
distancia que recorre la luz en un año. La luz se propaga en el vacío a
\[
c = \qty{3.00e8}{m/s},
\]
la mayor velocidad que la naturaleza permite --- de modo que el año luz
es una distancia, no un tiempo.
\end{definition}

\begin{example}[El año luz en metros]\label{ex:g10:orders-of-magnitude:ly}
Un año contiene $365.25 \times 24 \times 3600 = \qty{3.156e7}{s}$
(agradablemente cercano a $\pi \times 10^{7}\,\unit{s}$). Por tanto
\[
\qty{1}{ly} = c \times \qty{3.156e7}{s}
= \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}.
\]
Próxima Centauri, la estrella más cercana después del Sol, está a
$\qty{4.2}{ly} \approx \qty{4.0e16}{m}$: la luz, que cruza la distancia
Tierra--Sol en ocho minutos, tarda más de cuatro años en llegar desde
allí.
\end{example}

\begin{method}[Comprobar un resultado por su orden de magnitud]\label{met:g10:orders-of-magnitude:sanity}
Antes de fiarse de cualquier resultado calculado:
\begin{enumerate}
  \item redondear cada dato a su \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud};
  \item rehacer el cálculo solo con potencias de diez, usando la
        \cref{prop:g10:orders-of-magnitude:powers} --- cuesta unos
        segundos;
  \item comparar: el resultado preciso debe coincidir con la estimación
        salvo un factor diez, y su \omterm{def:g10:orders-of-magnitude:unit}{unidad} debe ser la correcta. Si falla
        cualquiera de las dos comprobaciones, hay que cazar el error
        antes de seguir.
\end{enumerate}
\end{method}

\begin{remark}[Por qué molestarse]\label{rem:g10:orders-of-magnitude:sanity}
Una calculadora afirmará tan tranquila que una manzana tiene una masa de
$\qty{4e8}{kg}$ --- no puede saber más. Ninguna fórmula avisa de lo
absurdo; la costumbre sí. Los físicos estiman primero y calculan
después: la estimación caza las comas decimales mal puestas, las
\omterm{def:g10:orders-of-magnitude:unit}{unidades} intercambiadas y las fracciones invertidas, los tres errores
que hay detrás de la mayoría de las respuestas falsas a cualquier nivel.
\end{remark}

\section{Medida e incertidumbre}

Ningún instrumento da el valor verdadero. Una medida solo es honrada si
indica además cuánto puede desviarse.

\begin{definition}[Incertidumbre absoluta y relativa]\label{def:g10:orders-of-magnitude:uncertainty}
La medida de una magnitud $x$ se escribe
\[
x = x_{\text{m}} \pm \Delta x ,
\]
donde $x_{\text{m}}$ es el valor medido y la \emph{incertidumbre
absoluta}\index{incertidumbre!absoluta} $\Delta x > 0$, en la misma
\omterm{def:g10:orders-of-magnitude:unit}{unidad}, acota el error plausible: se afirma que el valor verdadero está
comprendido entre $x_{\text{m}} - \Delta x$ y $x_{\text{m}} + \Delta x$.
La \emph{incertidumbre relativa}\index{incertidumbre!relativa} es el
cociente $\Delta x / x_{\text{m}}$, que suele darse en porcentaje; es lo
que mide la palabra ``preciso''.
\end{definition}

\begin{method}[Leer un instrumento graduado]\label{met:g10:orders-of-magnitude:reading}
Para una regla, una probeta o un dial:
\begin{enumerate}
  \item leer la graduación más próxima a la marca, con el ojo
        perpendicular a la escala (mirar de lado desplaza la lectura);
  \item tomar como \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre absoluta} al menos la mitad de la
        graduación más pequeña --- más si el borde del objeto o la
        superficie del líquido están difusos;
  \item escribir el resultado como $x_{\text{m}} \pm \Delta x$ con su
        \omterm{def:g10:orders-of-magnitude:unit}{unidad}, sin conservar ningún dígito más allá del dígito
        incierto.
\end{enumerate}
\end{method}

\begin{example}[Precisión relativa]\label{ex:g10:orders-of-magnitude:reading}
La lectura anterior tiene una \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre relativa}
$\frac{0.5}{127} \approx 0.4\,\%$. La misma regla, midiendo un perno de
$\qty{12}{mm}$, da $\frac{0.5}{12} \approx 4\,\%$: el mismo instrumento es
diez veces menos preciso sobre el objeto pequeño. La precisión es una
propiedad de la \emph{medida}, no de la herramienta.
\end{example}

\begin{remark}[Poco en absoluto, mucho en relativo --- y al revés]\label{rem:g10:orders-of-magnitude:precision}
Las estaciones láser miden la distancia Tierra--Luna,
$\qty{3.84e8}{m}$, con unos $\pm\qty{3}{cm}$: una \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre relativa}
de $10^{-10}$, de las medidas más finas jamás realizadas --- y con un
error absoluto del que ninguna regla presumiría. Hay que preguntar
siempre \emph{relativo a qué}: la pareja
$(\Delta x,\ \Delta x / x_{\text{m}})$ cuenta la historia completa, y
cualquiera de los dos números por separado puede engañar.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:1}
Convierte a \omterm{def:g10:orders-of-magnitude:unit}{metros}, en \omterm{def:g10:orders-of-magnitude:scinot}{notación científica}:
\begin{enumerate}
  \item $\qty{750}{nm}$ (longitud de onda de la luz roja);
  \item $\qty{2.5}{km}$;
  \item $\qty{3.2}{\micro m}$ (una bacteria del suelo);
  \item $\qty{0.15}{Tm}$ (la distancia Tierra--Sol, con un disfraz
        insólito).
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:2}
Escribe en \omterm{def:g10:orders-of-magnitude:scinot}{notación científica} e indica el número de \omterm{def:g10:orders-of-magnitude:sigfig}{cifras
significativas}:
\begin{enumerate}
  \item la distancia Tierra--Sol, $\qty{149600000}{km}$, en \omterm{def:g10:orders-of-magnitude:unit}{metros};
  \item el diámetro de un átomo de hidrógeno, $\qty{0.000000000106}{m}$;
  \item la población mundial, unos $\num{8100000000}$;
  \item una duración de $\qty{0.00520}{s}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:3}
Calcula sin calculadora y da cada resultado en \omterm{def:g10:orders-of-magnitude:scinot}{notación científica}:
\[
(3\times10^{8})\times(2\times10^{-5}), \qquad
\frac{8\times10^{4}}{4\times10^{-3}}, \qquad
(5\times10^{6})^{2}, \qquad
\frac{(4\times10^{-3})\times(3\times10^{-9})}{6\times10^{-7}} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:4}
Da el \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud}, con su \omterm{def:g10:orders-of-magnitude:unit}{unidad}, de: una estatura humana de
$\qty{1.7}{m}$; el Everest, $\qty{8848}{m}$; una bacteria
\emph{E.~coli}, $\qty{2e-6}{m}$; la masa de la Tierra,
$\qty{5.97e24}{kg}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:5}
Una regla está graduada en milímetros. Mides un lápiz y obtienes
$\qty{127}{mm}$.
\begin{enumerate}
  \item Escribe el resultado con su \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre absoluta}.
  \item Calcula la \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre relativa}.
  \item La misma regla mide una goma de $\qty{12}{mm}$. ¿Cuál de las dos
        medidas es más precisa, y en qué factor?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:orders-of-magnitude:6}
Con $c = \qty{3.00e8}{m/s}$ y $\qty{1}{ly} = \qty{9.47e15}{m}$:

  \item La sonda \emph{Voyager 1} está a unos $\qty{2.4e13}{m}$ de la

        llegar hasta nosotros? Expresa la respuesta en horas.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:orders-of-magnitude:7}

($\qty{3.84e8}{m}$), desde el Sol ($\qty{1.496e11}{m}$) y desde Neptuno

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:orders-of-magnitude:8}

  \item ¿Cuál es la \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre relativa} de ese grosor?

\section{Problema: del átomo a Andrómeda}

\begin{problem}[{Problema de fin de semana --- un solo grano de arena
mide el átomo, el Sistema Solar, la galaxia y el universo observable, y
te sitúa a ti en la escalera que los separa}]
\label{pb:g10:orders-of-magnitude:1}
Un grano de arena es el objeto más humilde de la física, y este fin de
semana hará horas extra: primero contamos sus átomos, luego encogemos el
Sol hasta su tamaño y recorremos el cosmos a pasos, después dejamos que
la luz haga de topógrafo y, por último, contamos los átomos que hay en
\emph{ti}. Datos, usados a lo largo de todo el problema: diámetro del
grano $\qty{0.5}{mm}$; diámetro de un átomo $\qty{2.5e-10}{m}$; densidad
de la arena $\qty{2500}{kg/m^3}$; diámetros: Sol $\qty{1.39e9}{m}$,
Tierra $\qty{1.27e7}{m}$, Vía Láctea $\qty{9.5e20}{m}$, universo
observable $\qty{8.8e26}{m}$; distancias desde nosotros: Luna
$\qty{3.84e8}{m}$, Sol $\qty{1.496e11}{m}$, Neptuno $\qty{4.5e12}{m}$,
Próxima Centauri $\qty{4.0e16}{m}$, galaxia de Andrómeda
$\qty{2.4e22}{m}$; $c = \qty{3.00e8}{m/s}$; una molécula de agua tiene
una masa de $\qty{3.0e-26}{kg}$ y contiene tres átomos.

\medskip
\noindent\textbf{Parte I --- Anatomía del grano.}
\begin{enumerate}
  \item Escribe el diámetro del grano y el del átomo en \omterm{def:g10:orders-of-magnitude:unit}{metros}, en
        \omterm{def:g10:orders-of-magnitude:scinot}{notación científica}, y nombra el \omterm{def:g10:orders-of-magnitude:prefixes}{prefijo} del SI al que más se
        acerca cada uno.
  \item ¿Cuántos átomos caben uno al lado de otro a lo ancho de un
        grano?
  \item Modeliza el grano como un cubo de $\qty{0.5}{mm}$ de arista
        relleno de átomos. Estima el número de átomos del grano, como
        \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud}.
  \item Calcula el volumen de ese cubo y después la masa del grano.
        Expresa la masa en miligramos.
  \item Comprueba el modelo: divide la masa del grano entre su número de
        átomos y compara con el \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud} conocido de las masas
        atómicas (unos pocos $10^{-26}\,\unit{kg}$). ¿Veredicto?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El Sol como grano de arena.}
Construimos una maqueta del cosmos en la que el Sol es el grano: cada
tamaño de la maqueta es el tamaño real multiplicado por un factor $s$.
\begin{enumerate}[resume]
  \item Calcula el factor de escala $s$.
  \item En la maqueta, halla el diámetro de la Tierra y su distancia al
        Sol-grano. ¿Qué objeto del mundo de la Parte I tiene
        aproximadamente el tamaño de esa Tierra?
  \item ¿A qué distancia del grano orbita el Neptuno de la maqueta? ¿Qué
        mueble albergaría el sistema planetario entero?
  \item ¿A qué distancia queda la Próxima Centauri de la maqueta?
  \item Dos granos de arena a $\qty{14}{km}$ uno de otro, y casi nada
        entre medias: di en una o dos frases qué revela la maqueta sobre
        lo vacía que está la galaxia.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La luz hace de topógrafo.}
\begin{enumerate}[resume]
  \item ¿Cuánto tarda la luz en cruzar el grano de arena real?
  \item ¿Cuánto tarda la luz en llegar hasta nosotros desde la Luna y
        desde el Sol?
  \item Recalcula el \omterm{def:g10:orders-of-magnitude:lightyear}{año luz} en \omterm{def:g10:orders-of-magnitude:unit}{metros} a partir de $c$ y de la duración
        del año, y comprueba que la distancia a Próxima que figura en
        los datos corresponde a unos $\qty{4.2}{ly}$.
  \item Convierte la distancia a Andrómeda en \omterm{def:g10:orders-of-magnitude:lightyear}{años luz}. ¿Cuándo, más o
        menos, salió de Andrómeda la luz que ahora entra en tu ojo, y
        qué caminaba por la Tierra en aquel momento?
  \item ¿Qué distancia recorre la luz en un nanosegundo? Un procesador a
        $\qty{3}{GHz}$ inicia una operación cada tercio de nanosegundo:
        explica en una frase por qué un ordenador rápido tiene que ser
        un ordenador pequeño.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Tú, entre el átomo y Andrómeda.}
\begin{enumerate}[resume]
  \item Una persona es, a grandes rasgos, $\qty{70}{kg}$ de agua. Con
        los datos, estima la masa media de un átomo en el agua y después
        el número de átomos de una persona.
  \item Compara ese recuento con los átomos del grano (pregunta 3) y con
        las aproximadamente $2\times10^{23}$ estrellas del universo
        observable. Da los dos factores.
  \item En la escalera, ¿estás más cerca del átomo o del universo
        observable? Compara los cocientes $\text{(tú)}/\text{(átomo)}$ y
        $\text{(universo)}/\text{(tú)}$, tomando $\qty{1.7}{m}$ para ti.
  \item Halla el tamaño $x$ que se sitúa exactamente a media escalera
        entre el núcleo ($\qty{e-15}{m}$) y el universo observable: $x$
        debe cumplir $x / 10^{-15} = 10^{26.9} / x$ (tomando
        $\qty{8.8e26}{m} \approx 10^{26.9}\,\unit{m}$). ¿Qué objeto del
        Sistema Solar tiene justamente ese tamaño?
  \item El remate. Encoge el universo observable con el factor $s$ de la
        Parte II, de modo que el Sol sea un grano de arena. Da el
        diámetro del universo de la maqueta en \omterm{def:g10:orders-of-magnitude:unit}{metros} y después en
        \omterm{def:g10:orders-of-magnitude:lightyear}{unidades astronómicas}, y concluye en una frase: aun con cada
        estrella convertida en grano de arena, el universo desborda toda
        escala humana --- solo las potencias de diez lo abarcan.
\end{enumerate}
\end{problem}
