\chapter{Las fuerzas y el principio de inercia}\label{ch:g10:inertia}

Deja de pedalear y la bicicleta se va parando; deja de tirar y el trineo
se queda quieto. La experiencia susurra que el movimiento hay que
alimentarlo con un esfuerzo --- y durante dos mil años la física le dio
la razón. Este capítulo construye la \omterm{def:g10:inertia:force}{fuerza}, una flecha medida en
\omterm{def:g10:inertia:force}{newtons}, y enuncia la primera gran ley de la mecánica: si de verdad se
la deja en paz, un cuerpo en movimiento conserva su movimiento, recto y
constante, para siempre.

\section{Modelizar una acción mediante una fuerza}

\begin{definition}[Fuerza]\label{def:g10:inertia:force}
Una \emph{fuerza}\index{fuerza} modeliza una acción mecánica (un empujón,
un tirón, una atracción) ejercida sobre un cuerpo. Queda caracterizada
por:
\begin{itemize}
  \item su \emph{punto de aplicación}\index{punto de aplicación};
  \item su dirección --- la recta a lo largo de la cual actúa --- y su
        sentido sobre esa recta;
  \item su módulo, medido en \emph{newtons}\index{newton} (símbolo
        \unit{N}).
\end{itemize}
Una fuerza se dibuja como una flecha $\vect F$ (un vector, como en el
volumen de matemáticas) desde su punto de aplicación, con una longitud
proporcional al módulo según una escala declarada.
\end{definition}

\begin{example}[El peso de una mochila]\label{ex:g10:inertia:schoolbag}
La Tierra tira hacia abajo de una mochila de masa $m = \qty{4.0}{kg}$
con su \omterm{def:g10:universal-gravitation:weight}{peso} (\cref{ch:g10:universal-gravitation}):
$P = mg = 4.0 \times 9.81 \approx \qty{39}{N}$ --- un \omterm{def:g10:inertia:force}{newton} es
aproximadamente el \omterm{def:g10:universal-gravitation:weight}{peso} de una manzana. A la escala
$\qty{1}{cm} \leftrightarrow \qty{10}{N}$: una flecha vertical de
\qty{3.9}{cm}, aplicada en el centro de gravedad de la mochila y
apuntando hacia abajo.
\end{example}

\section{Un inventario de fuerzas}

\begin{definition}[Fuerzas de contacto y a distancia]\label{def:g10:inertia:contact}
Una \emph{fuerza de contacto}\index{fuerza de contacto} se ejerce en los
puntos donde dos cuerpos se tocan. Una \emph{fuerza a
distancia}\index{fuerza a distancia} no necesita contacto: el \omterm{def:g10:universal-gravitation:weight}{peso} ---
el tirón gravitatorio del \cref{ch:g10:universal-gravitation} --- es la
única que se usa en este capítulo.
\end{definition}

\begin{definition}[Las fuerzas de contacto habituales]\label{def:g10:inertia:inventory}
Cuatro \omterm{def:g10:inertia:contact}{fuerzas de contacto} cubren casi todas las situaciones
cotidianas:
\begin{itemize}
  \item la \emph{reacción normal}\index{reacción normal} $\vect R$ de un
        soporte, perpendicular a su superficie y dirigida hacia fuera de
        él;
  \item la \emph{tensión}\index{tensión} $\vect T$ de un hilo, una
        cuerda o una cadena, dirigida a lo largo de ella y tirando hacia
        ella;
  \item el \emph{rozamiento}\index{rozamiento} $\vect f$ de una
        superficie, del aire o del agua, que se opone al deslizamiento o
        al movimiento;
  \item el \emph{empuje}\index{empuje} de un motor, de un chorro o de
        una mano, que empuja en el sentido del esfuerzo.
\end{itemize}
\end{definition}

\begin{example}[Inventario de un trineo remolcado]\label{ex:g10:inertia:sled}
Un trineo remolcado por la nieve: \omterm{def:g10:universal-gravitation:weight}{peso} $\vect P$ (a distancia), \omterm{def:g10:inertia:inventory}{reacción
normal} $\vect R$ de la nieve, \omterm{def:g10:inertia:inventory}{tensión} $\vect T$ de la cuerda y
\omterm{def:g10:inertia:inventory}{rozamiento} $\vect f$ que se opone al deslizamiento. Nada más toca el
trineo: el inventario está completo --- \emph{el \omterm{def:g10:universal-gravitation:weight}{peso} siempre, y
después una \omterm{def:g10:inertia:contact}{fuerza de contacto} por cada objeto que toque}.
\end{example}

\section{El principio de inercia}

\begin{definition}[Movimiento rectilíneo uniforme]\label{def:g10:inertia:uniform}
Un cuerpo tiene un \emph{movimiento rectilíneo
uniforme}\index{movimiento rectilíneo uniforme} en un sistema cuando su
\omterm{def:g10:relative-motion:trajectory}{trayectoria} es una recta recorrida a velocidad constante: su vector
velocidad $\vect v$ es el mismo en todo instante. El reposo es el caso
particular $\vect v = \vect 0$.
\end{definition}

\begin{definition}[Fuerzas que se compensan]\label{def:g10:inertia:compensate}
Las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre un cuerpo \emph{se
compensan}\index{fuerzas que se compensan} cuando su efecto conjunto es
nulo: puestas punta con cola, sus flechas vuelven al punto de partida.
Dos \omterm{def:g10:inertia:force}{fuerzas} se compensan exactamente cuando comparten la recta de acción
con módulos iguales y sentidos opuestos.
\end{definition}

\begin{theorem}[Principio de inercia]\label{thm:g10:inertia:principle}
\index{principio de inercia}\index{inercia}
En el \omterm{def:g10:relative-motion:frames}{sistema terrestre}, un cuerpo no sometido a ninguna \omterm{def:g10:inertia:force}{fuerza}, o
sometido a \omterm{def:g10:inertia:compensate}{fuerzas que se compensan}, permanece en reposo o conserva un
\omterm{def:g10:inertia:uniform}{movimiento rectilíneo uniforme}; recíprocamente, un cuerpo en reposo o
con \omterm{def:g10:inertia:uniform}{movimiento rectilíneo uniforme} está sometido a \omterm{def:g10:inertia:compensate}{fuerzas que se
compensan} (o a ninguna).
\end{theorem}
\admitted

\begin{remark}
Ningún experimento puede aislar por completo un cuerpo, de modo que el
principio es un postulado --- entrevisto por Galileo y enunciado por
Newton. Un capítulo posterior lo asciende a una ley que relaciona las
\omterm{def:g10:inertia:force}{fuerzas} con los \emph{cambios} de velocidad; el volumen del primer año
pregunta en qué sistemas se cumple realmente.
\end{remark}

\begin{proposition}[Leer el movimiento a partir de las fuerzas, y al revés]\label{prop:g10:inertia:converse}
En el \omterm{def:g10:relative-motion:frames}{sistema terrestre}:
\begin{enumerate}
  \item si la velocidad de un cuerpo cambia --- en módulo \emph{o} en
        dirección --- las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre él \emph{no} \omterm{def:g10:inertia:compensate}{se
        compensan};
  \item si un cuerpo está en reposo o tiene \omterm{def:g10:inertia:uniform}{movimiento rectilíneo
        uniforme}, las \omterm{def:g10:inertia:force}{fuerzas} de su inventario deben anularse --- lo que
        determina las incógnitas.
\end{enumerate}
\end{proposition}

\begin{proof}
La primera afirmación es el contrarrecíproco del
\cref{thm:g10:inertia:principle}, y la segunda su parte recíproca
aplicada a un inventario completo.
\end{proof}

\begin{example}[El disco de hockey]\label{ex:g10:inertia:puck}
Soltado por el stick, un disco recorre \qty{30}{m} de hielo liso en
\qty{2.5}{s}, en línea recta y a la velocidad constante
$v = 30/2.5 = \qty{12}{m/s}$. Inventario: \omterm{def:g10:universal-gravitation:weight}{peso} y \omterm{def:g10:inertia:inventory}{reacción normal} (el
\omterm{def:g10:inertia:inventory}{rozamiento} es despreciable). \omterm{def:g10:inertia:compensate}{Se compensan}, y el principio predice
exactamente el movimiento observado: nada empuja el disco hacia
adelante --- y nada necesita hacerlo.
\end{example}

\begin{remark}[Galileo contra Aristóteles]
Aristóteles enseñaba que todo movimiento necesita un motor, y la vida
diaria --- con \omterm{def:g10:inertia:inventory}{rozamiento} en cada contacto --- parece darle la razón.
Galileo idealizó el \omterm{def:g10:inertia:inventory}{rozamiento} hasta hacerlo desaparecer: cuanto más
lisas son las superficies, menos empujón hace falta para
\emph{mantener} el movimiento. El estado natural es la persistencia, no
el reposo.
\end{remark}

\begin{remark}[El viajero que se va hacia adelante]
Cuando un autobús frena de golpe, los viajeros de pie se van hacia
adelante. No los lanza ninguna \omterm{def:g10:inertia:force}{fuerza}: por inercia, sus cuerpos
conservan su velocidad mientras el autobús pierde la suya. El cinturón
de seguridad aporta la \omterm{def:g10:inertia:force}{fuerza} hacia atrás que le falta al inventario.
\end{remark}

\section{Fuerzas equilibradas: estática sobre un dibujo}

\begin{definition}[Diagrama de fuerzas]\label{def:g10:inertia:diagram}
El \emph{diagrama de fuerzas}\index{diagrama de fuerzas} de un cuerpo
muestra el cuerpo solo, con todas las \omterm{def:g10:inertia:force}{fuerzas} que actúan \emph{sobre} él
dibujadas a escala desde su \omterm{def:g10:inertia:force}{punto de aplicación}; las \omterm{def:g10:inertia:force}{fuerzas} ejercidas
\emph{por} el cuerpo no aparecen nunca.
\end{definition}

\begin{method}[Analizar una situación estática]\label{met:g10:inertia:method}
\begin{enumerate}
  \item Elegir el cuerpo que se va a estudiar; trabajar en el \omterm{def:g10:relative-motion:frames}{sistema
        terrestre}.
  \item Inventario: el \omterm{def:g10:universal-gravitation:weight}{peso} siempre, y una \omterm{def:g10:inertia:contact}{fuerza de contacto} por cada
        contacto.
  \item Dibujar el \omterm{def:g10:inertia:diagram}{diagrama de fuerzas}.
  \item El cuerpo está en reposo, luego las \omterm{def:g10:inertia:force}{fuerzas} \omterm{def:g10:inertia:compensate}{se compensan}
        (\cref{thm:g10:inertia:principle}): leer del dibujo los módulos
        desconocidos (longitudes iguales en una dimensión, componentes
        en dos).
\end{enumerate}
\end{method}

\begin{example}[El libro sobre la mesa]\label{ex:g10:inertia:book}
Un libro de masa \qty{0.50}{kg} descansa sobre una mesa: \omterm{def:g10:universal-gravitation:weight}{peso}
$P = 0.50 \times 9.81 \approx \qty{4.9}{N}$ y \omterm{def:g10:inertia:inventory}{reacción normal}
$\vect R$. En reposo, $\vect R$ compensa a $\vect P$: vertical, hacia
arriba, $R = \qty{4.9}{N}$. Si se añade un segundo libro, $R$ se ajusta
--- un soporte empuja exactamente con la \omterm{def:g10:inertia:force}{fuerza} que hace falta.
\end{example}

\begin{example}[La lámpara colgada de dos cables]\label{ex:g10:inertia:lamp}
Una lámpara de masa \qty{3.0}{kg} ($P \approx \qty{29.4}{N}$) cuelga de
dos cables, cada uno a $45^\circ$ de la vertical. Por simetría las
tensiones son iguales ($T_1 = T_2 = T$) y sus componentes horizontales
se anulan; las verticales, cada una $T \cos 45^\circ$, cargan con el
\omterm{def:g10:universal-gravitation:weight}{peso}:
\[
2\,T \cos 45^\circ = P
\qquad\text{luego}\qquad
T = \frac{29.4}{2 \times 0.707} \approx \qty{21}{N}
\]
--- más de medio \omterm{def:g10:universal-gravitation:weight}{peso} por cable: tirar en ángulo cuesta \omterm{def:g10:inertia:inventory}{tensión}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:inertia:1}
Inventaría y clasifica (de contacto o a distancia) las \omterm{def:g10:inertia:force}{fuerzas} que
actúan sobre: (a) un libro en reposo sobre una mesa; (b) una lámpara
colgada de su cable; (c) un disco que se desliza por hielo liso; (d) una
manzana que cae (resistencia del aire despreciable).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:2}
Calcula el \omterm{def:g10:universal-gravitation:weight}{peso} de un libro de \qty{250}{g} y el de una persona de
\qty{60}{kg}. A la escala $\qty{1}{cm} \leftrightarrow \qty{1}{N}$,
¿cuánto mide la flecha del libro, y por qué esa escala es inservible
para la persona?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:3}
En un diagrama a la escala $\qty{1}{cm} \leftrightarrow \qty{2}{N}$, una
\omterm{def:g10:inertia:force}{fuerza} es una flecha horizontal de \qty{3.5}{cm} que apunta hacia la
izquierda, aplicada en un punto $A$. Da sus tres características.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:4}
Un diccionario de masa \qty{1.2}{kg} descansa sobre un estante. Dibuja
su \omterm{def:g10:inertia:diagram}{diagrama de fuerzas} y da el módulo y la dirección de cada \omterm{def:g10:inertia:force}{fuerza}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:5}
Verdadero o falso --- corrige las falsas: (a) un cuerpo siempre acaba
parándose si no hay una \omterm{def:g10:inertia:force}{fuerza} que lo siga empujando; (b) un \omterm{def:g10:inertia:uniform}{movimiento
rectilíneo uniforme} no exige ninguna \omterm{def:g10:inertia:force}{fuerza} sin compensar; (c) un cuerpo
en reposo no está sometido a ninguna \omterm{def:g10:inertia:force}{fuerza}; (d) si la velocidad de un
cuerpo cambia, las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre él no \omterm{def:g10:inertia:compensate}{se compensan}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:6}
Tras el lanzamiento, un disco recorre \qty{24}{m} de hielo en
\qty{3.0}{s}, en línea recta y a velocidad constante. Calcula su
velocidad; ¿\omterm{def:g10:inertia:compensate}{se compensan} las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre él? Sobre hormigón
ese mismo disco se para en unos \omterm{def:g10:orders-of-magnitude:unit}{metros}: ¿qué ha cambiado en el
inventario, y qué concluye la
\cref{prop:g10:inertia:converse}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:7}
Un pasajero de \qty{70}{kg} va de pie en un ascensor que sube a la
velocidad constante de \qty{1.5}{m/s}. Enumera las \omterm{def:g10:inertia:force}{fuerzas} que actúan
sobre él; ¿\omterm{def:g10:inertia:compensate}{se compensan}? Da el módulo de la \omterm{def:g10:inertia:force}{fuerza} del suelo. ¿Cambiaría
algo con el ascensor parado?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:8}
Una lámpara de masa \qty{1.8}{kg} cuelga en reposo de un único cable
vertical. Calcula la \omterm{def:g10:inertia:inventory}{tensión} del cable. Se sustituye el cable por dos
cables \emph{verticales} que se reparten la carga por igual: ¿cuál es la
\omterm{def:g10:inertia:inventory}{tensión} de cada uno?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:9}
Un autobús frena de golpe y un viajero de pie se va hacia adelante; en
una curva a velocidad constante, ese mismo viajero se inclina hacia
fuera. Explica las dos cosas sin inventar ninguna \omterm{def:g10:inertia:force}{fuerza} ``hacia
adelante'' ni ``hacia fuera''.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:10}
Un paracaidista de masa total \qty{85}{kg} desciende verticalmente a la
velocidad constante de \qty{5.0}{m/s}. Dibuja el \omterm{def:g10:inertia:diagram}{diagrama de fuerzas} y
calcula la resistencia del aire sobre la vela.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:11}
A un esquiador acuático lo remolcan en línea recta a velocidad
constante; la cuerda, horizontal, tira con \qty{300}{N}. ¿Cuál es la
resistencia horizontal del agua sobre los esquís (justifícalo)? ¿Qué dos
\omterm{def:g10:inertia:force}{fuerzas} verticales \omterm{def:g10:inertia:compensate}{se compensan} entre sí?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:12}
Una lámpara de masa \qty{2.4}{kg} cuelga de dos cables a $40^\circ$ de
la vertical ($\cos 40^\circ \approx 0.766$). Calcula cada \omterm{def:g10:inertia:inventory}{tensión}. ¿Por
qué crece la \omterm{def:g10:inertia:inventory}{tensión} a medida que los cables se acercan a la horizontal,
y por qué dos cables exactamente horizontales no podrían sostener nunca
la lámpara?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:13}
A una piedra de curling se la empuja, se la suelta, se desliza mientras
frena muy poco a poco y por fin se detiene. Para cada fase (\omterm{def:g10:inertia:inventory}{empuje},
deslizamiento, reposo), dibuja el \omterm{def:g10:inertia:diagram}{diagrama de fuerzas} y di si las
\omterm{def:g10:inertia:force}{fuerzas} \omterm{def:g10:inertia:compensate}{se compensan}, citando el \cref{thm:g10:inertia:principle} o la
\cref{prop:g10:inertia:converse}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:14}
La sonda \emph{Voyager 1} avanza por el espacio interestelar a
\qty{17}{km/s} con los motores apagados. ¿Por qué no necesita
combustible para mantener su velocidad, y qué predeciría Aristóteles?
¿Qué distancia recorre en un año, en \unit{km} y después en \omterm{def:g10:orders-of-magnitude:lightyear}{unidades
astronómicas} ($\qty{1}{au} \approx \qty{1.496e8}{km}$)? ¿Por qué tiene
que encender un propulsor aunque solo quiera cambiar de dirección
\emph{sin} cambiar de velocidad?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:15}
Tres cuerdas tiran de una anilla en reposo. En una cuadrícula en la que
un cuadro es \qty{1}{N}, dos de las \omterm{def:g10:inertia:force}{fuerzas} son
$\vect F_1\,(3, 2)$ y $\vect F_2\,(-1, 2)$. Halla las componentes, el
módulo y la dirección de la tercera \omterm{def:g10:inertia:force}{fuerza}.
\end{exercise}

\section{Problema: la inercia en tres actos}

\begin{problem}[{Problema de fin de semana --- la cinta del aeropuerto,
la pista de hielo y el paracaídas: la velocidad constante como firma de
unas fuerzas compensadas, y por qué un paracaídas no reduce la
resistencia sino la velocidad}]
\label{pb:g10:inertia:1}
Tres escenas --- una maleta sobre una cinta transportadora, un disco
sobre una pista de hielo y un paracaidista en el aire del atardecer ---
y una sola ley hablando en las tres.

\medskip
\noindent\textbf{Parte I --- La cinta del aeropuerto.}
Una maleta de \qty{23}{kg} está de pie sobre una cinta que se mueve a la
velocidad constante de \qty{0.75}{m/s}.
\begin{enumerate}
  \item ¿Cuál es el movimiento de la maleta en el \omterm{def:g10:relative-motion:frames}{sistema terrestre}?
        Inventaría las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre ella.
  \item ¿\omterm{def:g10:inertia:compensate}{Se compensan} esas \omterm{def:g10:inertia:force}{fuerzas}? Justifícalo; calcula los dos
        módulos.
  \item Compara su \omterm{def:g10:inertia:diagram}{diagrama de fuerzas} con el de esa misma maleta
        apoyada en el suelo. ¿Qué afirmación profunda ilustra la
        comparación?
  \item La cinta se para de golpe. Describe lo que hace la maleta, sin
        inventar ninguna \omterm{def:g10:inertia:force}{fuerza} hacia adelante.
  \item Su dueña camina a \qty{1.2}{m/s} respecto de la cinta. ¿Cuál es
        su velocidad en el \omterm{def:g10:relative-motion:frames}{sistema terrestre}? ¿\omterm{def:g10:inertia:compensate}{Se compensan} las \omterm{def:g10:inertia:force}{fuerzas}
        que actúan sobre \emph{ella}?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La pista de hielo.}
\begin{enumerate}[resume]
  \item Un disco de \qty{160}{g} se desliza en línea recta a la
        velocidad constante de \qty{8.0}{m/s}. Dibuja su \omterm{def:g10:inertia:diagram}{diagrama de
        fuerzas} con los módulos.
  \item Aristóteles sostiene que el disco necesita un motor. ¿Qué le
        responde la pista, y por qué parecen darle la razón las
        superficies cotidianas?
  \item El disco atraviesa una zona rugosa y frena. ¿Qué se concluye
        sobre las \omterm{def:g10:inertia:force}{fuerzas} allí? ¿Cuál es la culpable?
  \item Una patinadora describe una curva suave a velocidad constante.
        ¿\omterm{def:g10:inertia:compensate}{Se compensan} las \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre ella? Cuidado: ¿qué
        clase de magnitud es la velocidad?
  \item Empareja diagrama y movimiento: (a) \omterm{def:g10:inertia:force}{fuerzas} compensadas; (b) una
        \omterm{def:g10:inertia:force}{fuerza} extra opuesta a $\vect v$; (c) una \omterm{def:g10:inertia:force}{fuerza} extra
        perpendicular a $\vect v$ --- (i) frenar en línea recta; (ii)
        girar a velocidad constante; (iii) \omterm{def:g10:inertia:uniform}{movimiento rectilíneo
        uniforme}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El paracaídas.}
La masa total de un saltador, equipo incluido, es \qty{80}{kg}.
\begin{enumerate}[resume]
  \item Al salir del avión, la velocidad vertical y la resistencia son
        todavía nulas. Calcula las \omterm{def:g10:inertia:force}{fuerzas}; ¿\omterm{def:g10:inertia:compensate}{se compensan}? ¿Qué tiene
        que ocurrir a continuación?
  \item La resistencia crece con la velocidad. Describe la caída
        mientras la resistencia sea menor que el \omterm{def:g10:universal-gravitation:weight}{peso}.
  \item Antes de abrir la vela, el saltador alcanza unos estables
        \qty{50}{m/s} (``velocidad límite''). Calcula la resistencia.
        ¿Cuál es el movimiento, y por qué es ``inercia disfrazada''?
  \item Se abre la vela: ahora la resistencia supera con mucho al \omterm{def:g10:universal-gravitation:weight}{peso}.
        ¿Qué le ocurre a la velocidad? (El saltador se mueve
        \emph{hacia abajo} todo el tiempo.)
  \item Se alcanzan unos nuevos estables \qty{5.0}{m/s}. Calcula la
        resistencia; compárala con la de la pregunta 13.
  \item El remate: ¿qué ha cambiado el paracaídas --- la resistencia
        límite o la velocidad a la que la resistencia alcanza al \omterm{def:g10:universal-gravitation:weight}{peso}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Veredictos.}
\begin{enumerate}[resume]
  \item Completa y justifica: ``en reposo o con \omterm{def:g10:inertia:uniform}{movimiento rectilíneo
        uniforme} $\iff$ \dots''. ¿Por qué el reposo no merece una ley
        aparte?
  \item ¿En qué acto habría fracasado de forma más visible el
        ``el movimiento necesita un motor'' de Aristóteles, y qué
        señalaría Galileo?
  \item Un examen afirma: ``el paracaidista cae a velocidad constante,
        luego no actúa ninguna \omterm{def:g10:inertia:force}{fuerza} sobre él''. Corrígelo en una
        línea.
  \item Final: nombra, para cada acto, la pareja de \omterm{def:g10:inertia:compensate}{fuerzas que se
        compensan}, y enuncia la única ley usada veinte veces en este
        problema.
\end{enumerate}
\end{problem}
