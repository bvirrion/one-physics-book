\chapter{La presión: del deporte al buceo}\label{ch:g10:pressure}

Un esquiador se desliza sobre la nieve en la que un caminante se hunde;
un empujón de \qty{5}{N} clava una aguja en un cuero que un puño no
consigue abollar. Las mismas \omterm{def:g10:inertia:force}{fuerzas}, efectos disparatadamente
distintos: la magnitud que falta es la \emph{\omterm{def:g10:pressure:pressure}{presión}}, la \omterm{def:g10:inertia:force}{fuerza}
dividida entre el área que la recibe. Este capítulo la define, la sigue
piscina abajo, barómetro arriba y jeringa adentro --- y acaba a cuarenta
\omterm{def:g10:orders-of-magnitude:unit}{metros} bajo el mar, donde se vuelve cuestión de vida.

\section{La presión: fuerza repartida sobre un área}

\begin{definition}[Presión]\label{def:g10:pressure:pressure}
Cuando una \omterm{def:g10:inertia:force}{fuerza} de módulo $F$ (en \unit{N}) aprieta
perpendicularmente sobre una superficie de área $S$ (en \unit{m^2}), la
\emph{presión}\index{presión} ejercida sobre esa superficie vale
\[
P = \frac{F}{S}.
\]
Su \omterm{def:g10:orders-of-magnitude:unit}{unidad}, el \unit{N/m^2}, es el \emph{pascal}\index{pascal}
(\unit{Pa}) --- una \omterm{def:g10:orders-of-magnitude:unit}{unidad} diminuta, el \omterm{def:g10:universal-gravitation:weight}{peso} de una manzana repartido
sobre un \omterm{def:g10:orders-of-magnitude:unit}{metro} cuadrado, de modo que en la práctica se usan el
\unit{kPa} y el \unit{MPa}.
\end{definition}

\begin{example}[Tacón contra raqueta de nieve]\label{ex:g10:pressure:snowshoe}
Una persona de \qty{60}{kg} pesa
$F = 60 \times 9.81 \approx \qty{590}{N}$. Sobre un tacón de aguja de
\qty{1.0}{cm^2} = \qty{1.0e-4}{m^2}:
$P = 590/\num{1.0e-4} \approx \qty{5.9e6}{Pa} = \qty{5.9}{MPa}$. Sobre
dos raquetas de nieve de \qty{0.20}{m^2} cada una:
$P = 590/0.40 \approx \qty{1.5}{kPa}$. El \emph{mismo} \omterm{def:g10:universal-gravitation:weight}{peso} aprieta
$4000$ veces más fuerte bajo el tacón: los tacones marcan los suelos de
madera y las raquetas flotan sobre la nieve polvo.
\end{example}

\begin{remark}[Concentrar o repartir]\label{rem:g10:pressure:tools}
Una aguja de coser empujada con \qty{5}{N} sobre una punta de
\qty{0.01}{mm^2} = \qty{1e-8}{m^2} ejerce
$\qty{5e8}{Pa} = \qty{500}{MPa}$ --- suficiente para separar las fibras
del cuero. Las herramientas que perforan concentran la \omterm{def:g10:inertia:force}{fuerza}; los
diseños que soportan cargas (esquís, orugas, cimientos) la reparten. El
deporte es gestión aplicada de la \omterm{def:g10:pressure:pressure}{presión}: los patines muerden el hielo
y los esquís se deslizan sobre la nieve.
\end{remark}

\section{La presión en un líquido en reposo}

¿Por qué aprieta más el agua a mayor profundidad? Aislemos una columna
vertical imaginaria de agua de sección $S$, desde la superficie hasta la
profundidad $h$: volumen $Sh$, masa $\rho S h$ ($\rho$ es la densidad
del líquido, en \unit{kg/m^3}) y \omterm{def:g10:universal-gravitation:weight}{peso} $\rho S h g$. El fondo de la
columna soporta ese \omterm{def:g10:universal-gravitation:weight}{peso} \emph{más} el \omterm{def:g10:inertia:inventory}{empuje} $P_0 S$ de la atmósfera
sobre su parte superior. Dividiendo la \omterm{def:g10:inertia:force}{fuerza} total entre $S$: la
\omterm{def:g10:pressure:pressure}{presión} a la profundidad $h$ debería ser $P_0 + \rho g h$ --- y lo es,
en todos los puntos del líquido.

\begin{theorem}[La presión con la profundidad]\label{thm:g10:pressure:depth}
En un líquido de densidad $\rho$ en reposo y abierto al aire, la \omterm{def:g10:pressure:pressure}{presión}
a la profundidad $h$ por debajo de la superficie vale
\[
P = P_0 + \rho g h,
\]
donde $P_0$ es la \omterm{def:g10:pressure:atmospheric}{presión atmosférica} en la superficie. La \omterm{def:g10:pressure:pressure}{presión}
depende solo de la profundidad, no de la forma del recipiente, y en un
punto dado el líquido aprieta con la misma \omterm{def:g10:inertia:force}{fuerza} en todas las
direcciones.
\end{theorem}
\admitted

\begin{remark}
El razonamiento de la columna solo hace verosímil la ley; la deducción
honrada (para cualquier forma y en todas las direcciones) está en el
volumen del primer año. El término propio del agua, $\rho g h$, es la
\emph{presión manométrica}\index{presión manométrica} --- lo que muestra
el manómetro de un buceador.
\end{remark}

\begin{example}[Diez metros de agua]\label{ex:g10:pressure:tenmetres}
En agua dulce ($\rho = \qty{1000}{kg/m^3}$), a $h = \qty{10}{m}$:
$\rho g h = 1000 \times 9.81 \times 10 \approx \qty{98}{kPa}$, de modo
que $P = 101 + 98 \approx \qty{199}{kPa}$. Diez \omterm{def:g10:orders-of-magnitude:unit}{metros} de agua añaden
casi exactamente una atmósfera: la regla práctica del buceador, \emph{una
atmósfera más por cada diez \omterm{def:g10:orders-of-magnitude:unit}{metros}}.
\end{example}

\section{El océano de aire}

\begin{definition}[Presión atmosférica]\label{def:g10:pressure:atmospheric}
Vivimos en el fondo de un océano de aire cuyo \omterm{def:g10:universal-gravitation:weight}{peso} crea la
\emph{presión atmosférica}\index{presión atmosférica}, que a nivel del
mar vale
\[
P_0 \approx \qty{101}{kPa} = \qty{1.01e5}{Pa}.
\]
Sobre cada centímetro cuadrado son
$F = \num{1.01e5} \times \num{1e-4} \approx \qty{10}{N}$: el \omterm{def:g10:universal-gravitation:weight}{peso} de una
masa de un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} sobre cada \unit{cm^2} de tu piel, de tu mesa, de
todo.
\end{definition}

\begin{example}[La carga invisible]\label{ex:g10:pressure:sheet}
Una hoja A4 (\qty{21.0}{cm} $\times$ \qty{29.7}{cm}, luego
$S \approx \qty{0.0624}{m^2}$) recibe del aire que tiene encima
$F = \num{1.01e5} \times 0.0624 \approx \qty{6.3e3}{N}$ --- el \omterm{def:g10:universal-gravitation:weight}{peso} de
un coche pequeño. Y no se rasga nada: el aire de debajo empuja
\emph{hacia arriba} con la misma \omterm{def:g10:inertia:force}{fuerza}. Solo notamos la atmósfera
cuando uno de los dos lados la pierde --- una ventosa, una pajita, la
cabina de un avión.
\end{example}

\begin{remark}[El barómetro de Torricelli]\label{rem:g10:pressure:torricelli}
En 1643, Torricelli invirtió un tubo sellado lleno de mercurio sobre una
cubeta de mercurio. La columna bajó hasta la altura a la que
$\rho g h = P_0$:
$h = \num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$, dejando
por encima el primer vacío jamás fabricado. La altura sigue al tiempo
atmosférico y mengua con la altitud: un barómetro pesa el aire que
tienes encima (con agua harían falta \qty{10.3}{m}:
\cref{exo:g10:pressure:9}).
\end{remark}

\section{Los gases: bombardeo y ley de Boyle}

Un gas también aprieta sobre su recipiente: es un enjambre de moléculas
en movimiento desordenado incesante, cada choque da a la pared un
diminuto empujón hacia fuera, y miles de millones de miles de millones
de impactos por segundo se difuminan en la \omterm{def:g10:inertia:force}{fuerza} constante que medimos
como \omterm{def:g10:pressure:pressure}{presión} del gas. Si el gas se comprime hasta la mitad de volumen,
las moléculas chocan el doble de a menudo, así que la \omterm{def:g10:pressure:pressure}{presión} debería
duplicarse. Y se duplica.

\begin{proposition}[Ley de Boyle]\label{prop:g10:pressure:boyle}
\index{ley de Boyle}
Para una cantidad fija de gas mantenida a temperatura constante, la
\omterm{def:g10:pressure:pressure}{presión} y el volumen son inversamente proporcionales:
\[
P \, V = \text{constante},
\qquad\text{es decir}\qquad
P_1 V_1 = P_2 V_2 .
\]
\end{proposition}
\admitted

\begin{remark}
A este nivel es un hecho experimental --- el volumen del primer año la
deduce a partir de la imagen de los choques. El experimento (una jeringa
de aire sellada, un sensor de \omterm{def:g10:pressure:pressure}{presión}, \qty{20}{\degreeCelsius}):
\begin{center}
\begin{tabular}{lccccc}
$V$ (\unit{mL}) & 60 & 50 & 40 & 30 & 20 \\
$P$ (\unit{kPa}) & 100 & 120 & 150 & 200 & 300 \\
$P \times V$ (\unit{kPa.mL}) & 6000 & 6000 & 6000 & 6000 & 6000
\end{tabular}
\end{center}
\end{remark}

\begin{example}[Una jeringa comprimida]\label{ex:g10:pressure:boyle}
Un aire ocupa \qty{4.0}{L} a \qty{100}{kPa}; comprimido despacio (para
que la temperatura no cambie) hasta \qty{250}{kPa}, ocupa
$V_2 = P_1 V_1 / P_2 = 400/250 = \qty{1.6}{L}$.
\end{example}

\section{El buceo: la presión puesta a trabajar}

\begin{method}[Presión absoluta a una profundidad]\label{met:g10:pressure:dive}
Para un buceador a la profundidad $h$ en el mar
($\rho \approx \qty{1025}{kg/m^3}$, de modo que
$\rho g h \approx \qty{101}{kPa}$ por cada \qty{10}{m}), se calcula
$P = P_0 + \rho g h$ --- y en la \omterm{prop:g10:pressure:boyle}{ley de Boyle} hay que usar siempre esta
\omterm{def:g10:pressure:pressure}{presión} \emph{absoluta}\index{presión absoluta}, nunca el término
manométrico solo. Un buceador está bajo $2$ atmósferas a \qty{10}{m},
$3$ a \qty{20}{m} y $5$ a \qty{40}{m}.
\end{method}

\begin{remark}[Nunca contengas la respiración]\label{rem:g10:pressure:breath}
Un regulador entrega aire a la \omterm{def:g10:pressure:pressure}{presión} ambiente: los pulmones se llenan
con normalidad a cualquier profundidad. Pero si se llenan \qty{6.0}{L} a
\qty{20}{m} ($\qty{302}{kPa}$), se retiene esa respiración y se sube,
Boyle exige $6.0 \times 302/101 \approx \qty{18}{L}$ --- tres veces lo
que cabe en un tórax, y los pulmones se desgarran mucho antes de eso,
incluso con unos pocos \omterm{def:g10:orders-of-magnitude:unit}{metros} de ascenso. De ahí la primera regla del
buceo con botella: \emph{nunca contener la respiración mientras se
asciende} --- basta con seguir respirando y el aire que se expande sale
solo. Una apneísta está a salvo: su aire lo tomó en la superficie, así
que al subir solo vuelve a expandirse hasta su volumen original.
\end{remark}

\begin{remark}[La descompresión]\label{rem:g10:pressure:deco}
Respirar aire a alta \omterm{def:g10:pressure:pressure}{presión} disuelve además nitrógeno de más en la
sangre, como el gas de una botella de refresco cerrada; si se sube
demasiado deprisa, burbujea dentro del cuerpo --- es la enfermedad por
descompresión. De ahí el ascenso lento (unos \qty{10}{m} por minuto) y
las paradas a poca profundidad, que dejan salir el gas en silencio por
los pulmones. La teoría espera al volumen del primer año; la norma de
conducta es de todo buceador.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:pressure:1}
Una caja que pesa \qty{600}{N} se apoya sobre una cara de
$\qty{0.50}{m} \times \qty{0.30}{m}$ y después sobre una cara de
$\qty{0.30}{m} \times \qty{0.20}{m}$. Calcula las dos presiones sobre el
suelo. ¿Qué orientación conviene más sobre arena blanda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:2}
Expresa \qty{0.25}{MPa} en \unit{kPa} y en \unit{Pa}. ¿Qué aprieta más:
\qty{3.0}{N} sobre \qty{2.0}{cm^2} o \qty{60}{N} sobre
\qty{500}{cm^2}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:3}
Calcula la \omterm{thm:g10:pressure:depth}{presión manométrica} $\rho g h$ y la \omterm{met:g10:pressure:dive}{presión absoluta} en el
fondo de una piscina de \qty{3.0}{m} de profundidad
($\rho = \qty{1000}{kg/m^3}$, $P_0 = \qty{101}{kPa}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:4}
Un aire ocupa \qty{4.0}{L} a \qty{100}{kPa}, a temperatura constante.
¿Qué volumen ocupa a \qty{250}{kPa}? ¿Y a \qty{50}{kPa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:5}
Una patinadora de \qty{65}{kg} se desliza sobre una sola cuchilla de
área de contacto \qty{12}{cm^2} y después se queda de pie sobre unas
suelas que suman \qty{360}{cm^2}. Calcula las dos presiones y su
cociente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:6}
Un pulgar aprieta una chincheta con \qty{20}{N}. La cabeza tiene un área
de \qty{1.2}{cm^2} y la punta, \qty{0.010}{mm^2}. Calcula las presiones
sobre el pulgar y sobre la pared; ¿por qué solo se perfora la pared?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:7}
Una escotilla de submarino de \qty{0.50}{m^2} está a \qty{30}{m} de
profundidad en agua de mar ($\rho = \qty{1025}{kg/m^3}$); por dentro, el
aire está a \omterm{def:g10:pressure:atmospheric}{presión atmosférica}. Calcula la \omterm{def:g10:inertia:force}{fuerza} neta que la mantiene
cerrada y la masa cuyo \omterm{def:g10:universal-gravitation:weight}{peso} es igual a esa \omterm{def:g10:inertia:force}{fuerza}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:8}
Un sensor situado a \qty{5.00}{m} de profundidad en un líquido
desconocido marca una \omterm{thm:g10:pressure:depth}{presión manométrica} de \qty{66.2}{kPa}. Halla la
densidad. ¿Es agua dulce?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:9}
¿Qué altura tendría el barómetro de Torricelli con agua
($\rho = \qty{1000}{kg/m^3}$) en lugar de mercurio
($\rho = \qty{13600}{kg/m^3}$)? Recupera las dos alturas a partir de
$\rho g h = P_0$; ¿por qué ganó el mercurio?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:10}
Con los datos de la jeringa de la \cref{prop:g10:pressure:boyle}
($PV = \qty{6000}{kPa.mL}$), predice el volumen a \qty{240}{kPa} y la
\omterm{def:g10:pressure:pressure}{presión} a \qty{75}{mL}. ¿Qué curva es $P$ frente a $V$? ¿Y $P$ frente a
$1/V$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:11}
Un buceador a \qty{20}{m} suelta una burbuja de \qty{0.50}{cm^3}. ¿Qué
volumen tiene justo por debajo de la superficie? (Temperatura constante;
el agua de mar añade \qty{101}{kPa} por cada \qty{10}{m}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:12}
Una ventosa de \qty{12}{cm^2} se aprieta contra un techo.
\begin{enumerate}
  \item Con vacío perfecto dentro: ¿qué \omterm{def:g10:inertia:force}{fuerza} ejerce la atmósfera sobre
        la ventosa, y qué masa colgada puede sostener?
  \item Una ventosa real conserva dentro aire a \qty{20}{kPa}. ¿Qué masa
        sostiene ahora?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:13}
Una botella de buceo contiene \qty{12}{L} de aire a \qty{200}{bar}
(\qty{1}{bar} = \qty{100}{kPa} $\approx 1$ atmósfera); hay que dejar
\qty{50}{bar} de reserva. Un buceador respira \qty{15}{L} por minuto a
la \omterm{def:g10:pressure:pressure}{presión} ambiente. ¿Cuántos litros de aire a \omterm{def:g10:pressure:pressure}{presión} de superficie son
utilizables? ¿Cuánto dura la botella a \qty{30}{m} (\omterm{met:g10:pressure:dive}{presión absoluta}
$\approx \qty{4}{bar}$)? ¿Y a \qty{10}{m}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:14}
Los pulmones de una apneísta contienen \qty{6.0}{L} en la superficie y
no pueden encogerse por debajo del volumen residual, \qty{1.5}{L}. ¿A
qué profundidad marina ($\rho = \qty{1025}{kg/m^3}$) se alcanza ese
límite? (En su día se declaró el límite de la apnea; hoy los récords
pasan de \qty{100}{m}, porque la sangre se desplaza al tórax y ocupa el
volumen que falta.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:15}
En un elevador hidráulico, un líquido transmite la \omterm{def:g10:pressure:pressure}{presión} sin
alteración desde un pistón pequeño ($s = \qty{2.0}{cm^2}$) a uno grande
($S = \qty{400}{cm^2}$) que sostiene un coche de \qty{1200}{kg}.
\begin{enumerate}
  \item ¿Qué \omterm{def:g10:inertia:force}{fuerza} sobre el pistón pequeño mantiene el coche en alto?
        ¿Qué \omterm{def:g10:pressure:pressure}{presión} soporta el líquido?
  \item Para subir el coche \qty{2.0}{cm}, ¿cuánto tiene que recorrer el

\end{enumerate}
\end{exercise}

\section{Problema: una respiración hacia abajo, una hacia arriba}

\begin{problem}[{Problema de fin de semana --- una respiración hacia
abajo y otra hacia arriba: la física de una apnea y de un ascenso con
botella, del apretón de la máscara a la regla de que los últimos diez
metros son los más peligrosos}]
\label{pb:g10:pressure:1}
Lena bucea en apnea: una respiración en la superficie, hasta
\qty{30}{m} y vuelta. Marco bucea a su lado con botella. El mismo mar
($\rho = \qty{1025}{kg/m^3}$, $g = \qty{9.81}{N/kg}$,
$P_0 = \qty{101}{kPa}$), los mismos pulmones de \qty{6.0}{L}, peligros
opuestos --- todo ello gobernado por $P = P_0 + \rho g h$ y por $PV$
constante (la temperatura no cambia en ningún momento).

\medskip
\noindent\textbf{Parte I --- El \omterm{def:g10:universal-gravitation:weight}{peso} del agua.}
\begin{enumerate}
  \item Calcula $\rho g$ para el agua de mar en \unit{kPa} por \omterm{def:g10:orders-of-magnitude:unit}{metro} y
        comprueba la regla práctica: una atmósfera por cada \qty{10}{m}.
  \item Calcula la \omterm{met:g10:pressure:dive}{presión absoluta} a \qty{10}{m}, \qty{20}{m},
        \qty{30}{m} y \qty{40}{m}.
  \item Expresa cada una como múltiplo de la \omterm{def:g10:pressure:pressure}{presión} en la superficie.
  \item La máscara de Lena cubre \qty{150}{cm^2}. A \qty{30}{m}, si el
        aire de su interior siguiera a la \omterm{def:g10:pressure:pressure}{presión} de la superficie, ¿qué
        \omterm{def:g10:inertia:force}{fuerza} neta ejercería el agua? (Los buceadores exhalan por la
        nariz dentro de la máscara para evitar este \emph{apretón de
        máscara}.)
  \item Un tímpano tiene un área de unos \qty{0.60}{cm^2}. Calcula la
        \omterm{def:g10:inertia:force}{fuerza} neta sobre él a \qty{3.0}{m} si el oído medio se queda a
        la \omterm{def:g10:pressure:pressure}{presión} de la superficie; ¿por qué duelen los oídos en una
        simple piscina, y qué tiene que conseguir la ``compensación''?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Una respiración hacia abajo.}
Lena deja la superficie con \qty{6.0}{L} de aire en los pulmones.
\begin{enumerate}[resume]
  \item Con la \omterm{prop:g10:pressure:boyle}{ley de Boyle} y presiones absolutas, calcula su volumen
        pulmonar a \qty{10}{m}, \qty{20}{m} y \qty{30}{m}.
  \item ¿A qué profundidad se reduce a la mitad su volumen pulmonar?
  \item ¿Es el volumen pulmonar proporcional a la profundidad? Describe
        la curva de $V$ frente a la \omterm{met:g10:pressure:dive}{presión absoluta} $P$.
  \item Su volumen residual es \qty{1.5}{L}: los pulmones no pueden
        encogerse más. Demuestra que ese límite se alcanza cerca de los
        \qty{30}{m}.
  \item Y sin embargo los récords de apnea superan los \qty{100}{m}:
        ¿qué llena el volumen que falta? (¿Qué otra cosa puede fluir
        hacia el tórax?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Una respiración hacia arriba.}
A \qty{30}{m}, el regulador de Marco le llena los pulmones de
\qty{6.0}{L}.
\begin{enumerate}[resume]
  \item ¿A qué \omterm{def:g10:pressure:pressure}{presión} entrega el regulador ese aire? ¿Cuántas veces más
        denso es que el aire de la superficie?
  \item Marco entra en pánico, contiene la respiración y sube hasta
        \qty{10}{m}: ¿qué volumen exige allí su aire atrapado?
  \item ¿Y qué volumen en la superficie --- cuántas veces su capacidad?
  \item Enuncia la primera regla del buceador con botella y explica por
        qué respirar con normalidad elimina el peligro.
  \item Lena también contiene la respiración desde \qty{30}{m} hasta la
        superficie y sus pulmones están perfectamente a salvo. Explica
        la asimetría.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Las burbujas y los últimos diez \omterm{def:g10:orders-of-magnitude:unit}{metros}.}
\begin{enumerate}[resume]
  \item Marco suelta una burbuja de \qty{1.0}{cm^3} a \qty{40}{m}.
        Calcula su volumen a \qty{30}{m}, \qty{20}{m}, \qty{10}{m} y
        justo por debajo de la superficie.
  \item Para cada tramo de \qty{10}{m} de la subida, calcula el factor
        en que crece la burbuja. ¿Dónde es más rápido el crecimiento?
  \item Los buceadores ascienden como mucho a \qty{10}{m} por minuto:
        ¿cuánto se tarda desde \qty{40}{m}, y en qué minuto se expande
        más el gas atrapado o disuelto?
  \item En profundidad, la sangre de Marco disuelve nitrógeno de más,
        como el gas de una botella de refresco cerrada. ¿Qué provoca un
        ascenso demasiado rápido, y cómo lo evitan el ascenso lento y
        una parada de seguridad a \qty{5}{m}?
  \item El remate: en una frase cada uno, da el veredicto de la
        apneísta, el veredicto del buceador con botella y la razón
        cuantificada de que los últimos diez \omterm{def:g10:orders-of-magnitude:unit}{metros} de cualquier ascenso
        merezcan el máximo respeto.
\end{enumerate}
\end{problem}
