\chapter{Señales y ondas}\label{ch:g10:signals-and-waves}

Todos los mensajes que has recibido en tu vida llegaron como una \omterm{def:g10:signals-and-waves:signal}{señal}:
una voz apretada contra tu tímpano, esta página que te alcanzó como un
parpadeo de radio. Detrás de la variedad hay una sola idea --- una
\omterm{def:g10:orders-of-magnitude:unit}{magnitud física} que varía en el tiempo --- y una sola caja de
herramientas: \omterm{def:g10:signals-and-waves:period}{periodo}, \omterm{def:g10:signals-and-waves:frequency}{frecuencia}, \omterm{def:g10:signals-and-waves:amplitude}{amplitud} y el arte de cronometrar un
retraso. Con ella, un barco mide el fondo del mar y un médico cuenta
latidos.

\section{Las señales transportan información}

\begin{definition}[Señal]\label{def:g10:signals-and-waves:signal}
Una \emph{señal}\index{señal} es una \omterm{def:g10:orders-of-magnitude:unit}{magnitud física} que varía en el
tiempo para transportar información: la \omterm{def:g10:pressure:pressure}{presión} del aire sobre tu
tímpano, la tensión en los bornes de un micrófono, el brillo de la luz
en una \omterm{def:g10:refraction:fiber}{fibra óptica}. Para estudiar una señal se registra en función del
tiempo y se lee la información en la curva.
\end{definition}

En una llamada de teléfono, esa misma información cambia de portador
varias veces: \omterm{def:g10:pressure:pressure}{presión} en el aire, después tensión en el micrófono,
después radio en la antena --- y al revés en el otro extremo. Un solo
vocabulario sirve para todos; este capítulo lo construye.

\section{Las señales periódicas}

\begin{definition}[Señal periódica, periodo]\label{def:g10:signals-and-waves:period}
Una \omterm{def:g10:signals-and-waves:signal}{señal} es \emph{periódica}\index{señal periódica} cuando se repite
idéntica a intervalos regulares. El \emph{periodo}\index{periodo} $T$ es
la duración de un ciclo completo, en segundos.
\end{definition}

\begin{definition}[Frecuencia]\label{def:g10:signals-and-waves:frequency}
La \emph{frecuencia}\index{frecuencia} de una \omterm{def:g10:signals-and-waves:period}{señal periódica} es el
número de ciclos por segundo:
\[
f = \frac{1}{T} .
\]
Su \omterm{def:g10:orders-of-magnitude:unit}{unidad}, un ciclo por segundo, es el \emph{hercio}\index{hercio}
(\unit{Hz}), con los múltiplos habituales \unit{kHz}, \unit{MHz} y
\unit{GHz}.
\end{definition}

\begin{definition}[Amplitud]\label{def:g10:signals-and-waves:amplitude}
La \emph{amplitud}\index{amplitud} de una \omterm{def:g10:signals-and-waves:period}{señal periódica} es su
desviación máxima respecto del valor de reposo. Para una tensión que
oscila simétricamente en torno a cero, la amplitud es el valor de pico;
de cresta a valle se mide el \emph{doble} de la amplitud (el valor de
pico a pico).
\end{definition}

\begin{example}[La red eléctrica y el la del diapasón]\label{ex:g10:signals-and-waves:mains}
La tensión de la red eléctrica oscila a $f = \qty{50}{Hz}$: su \omterm{def:g10:signals-and-waves:period}{periodo}
vale $T = 1/f = \qty{0.020}{s} = \qty{20}{ms}$. El la de afinación de una
orquesta es una \omterm{def:g10:signals-and-waves:signal}{señal} de \omterm{def:g10:pressure:pressure}{presión} a \qty{440}{Hz}: un ciclo dura
$T = 1/440 = \qty{2.3}{ms}$. A mayor \omterm{def:g10:signals-and-waves:frequency}{frecuencia}, ciclo más corto: $f$ y
$T$ son inversos el uno del otro.
\end{example}

\section{Leer un oscilograma}

\begin{definition}[Osciloscopio]\label{def:g10:signals-and-waves:oscilloscope}
Un \emph{osciloscopio}\index{osciloscopio} representa una tensión frente
al tiempo; la curva sobre su pantalla cuadriculada es un
\emph{oscilograma}\index{oscilograma}. Dos ajustes convierten las
divisiones de la pantalla en valores físicos: la \emph{base de
tiempos}\index{base de tiempos} (segundos por división horizontal) y la
\emph{ganancia}\index{ganancia} vertical (voltios por división
vertical).
\end{definition}

\begin{method}[De la pantalla a los números]\label{met:g10:signals-and-waves:oscillogram}
\begin{enumerate}
  \item Contar las divisiones horizontales que abarca un ciclo completo
        (de cresta a cresta es lo más seguro); multiplicar por la \omterm{def:g10:signals-and-waves:oscilloscope}{base
        de tiempos} para obtener $T$, y después $f = 1/T$.
  \item Contar las divisiones verticales desde la línea central hasta
        una cresta y multiplicar por la \omterm{def:g10:signals-and-waves:oscilloscope}{ganancia} para obtener la
        \omterm{def:g10:signals-and-waves:amplitude}{amplitud} (de cresta a valle se obtiene el valor de pico a pico:
        hay que dividirlo entre dos).
\end{enumerate}
\end{method}

\begin{example}[Una lectura completa]\label{ex:g10:signals-and-waves:reading}
En el \omterm{def:g10:signals-and-waves:oscilloscope}{oscilograma} siguiente, la \omterm{def:g10:signals-and-waves:oscilloscope}{base de tiempos} es de \qty{5}{ms} por
división y la \omterm{def:g10:signals-and-waves:oscilloscope}{ganancia}, de \qty{2}{V} por división. Un ciclo abarca
$4.0$ divisiones: $T = 4.0 \times \qty{5}{ms} = \qty{20}{ms}$ y
$f = 1/\qty{0.020}{s} = \qty{50}{Hz}$ --- otra vez la red eléctrica. Una
cresta queda $3.0$ divisiones por encima del eje: la \omterm{def:g10:signals-and-waves:amplitude}{amplitud} vale
$3.0 \times \qty{2}{V} = \qty{6.0}{V}$.
\end{example}

\section{Señales sonoras y electromagnéticas}

\begin{definition}[Sonido]\label{def:g10:signals-and-waves:sound}
El \emph{sonido}\index{sonido} es una \omterm{def:g10:signals-and-waves:signal}{señal} de \omterm{def:g10:pressure:pressure}{presión}: un objeto que
vibra --- una cuerda, una membrana, las cuerdas vocales --- empuja
rítmicamente sobre el aire, y las compresiones se propagan hacia fuera a
unos \qty{340}{m/s}. Un micrófono convierte la \omterm{def:g10:pressure:pressure}{presión} que llega en una
tensión; el tímpano, en impulsos nerviosos. El sonido necesita un medio:
una campana que suena bajo una campana de vacío enmudece a medida que se
extrae el aire.
\end{definition}

\begin{definition}[Onda]\label{def:g10:signals-and-waves:wave}
Una perturbación que se propaga --- el patrón de compresiones de un
\omterm{def:g10:signals-and-waves:sound}{sonido}, la ondulación de un estanque, un pulso de radio --- es una
\emph{onda}\index{onda}. El medio, cuando lo hay, se queda donde está:
cada porción de agua cabecea en su sitio mientras la ondulación cruza el
estanque. La geometría de las ondas --- \omterm{def:g10:light-spectra:wavelength}{longitud de onda},
interferencias, difracción --- se retoma en un capítulo posterior; a
este solo le hacen falta sus velocidades.
\end{definition}

\begin{definition}[Ultrasonidos e infrasonidos]\label{def:g10:signals-and-waves:ultrasound}
El oído humano escucha \omterm{def:g10:signals-and-waves:sound}{sonidos} entre unos \qty{20}{Hz} y \qty{20}{kHz}
(el techo baja con la edad). El \omterm{def:g10:signals-and-waves:sound}{sonido} por encima de \qty{20}{kHz} son
los \emph{ultrasonidos}\index{ultrasonidos} --- los murciélagos cazan a
\qty{50}{kHz} y los ecógrafos médicos exploran el cuerpo con unos pocos
megahercios. El \omterm{def:g10:signals-and-waves:sound}{sonido} por debajo de \qty{20}{Hz} son los
\emph{infrasonidos}\index{infrasonidos}, que perciben los elefantes y
los sismómetros.
\end{definition}

\begin{proposition}[Señales electromagnéticas]\label{prop:g10:signals-and-waves:em}
La radio, el wifi y la luz son \emph{señales
electromagnéticas}\index{señal electromagnética}: una sola familia, que
se propaga por el vacío --- sin necesidad de medio --- y toda ella a una
misma velocidad, la de la luz del \cref{ch:g10:refraction},
\[
c = \qty{3.00e8}{m/s} \quad \text{(casi la misma en el aire).}
\]
\end{proposition}
\admitted

\begin{remark}\label{rem:g10:signals-and-waves:honest}
Aquí esto es un hecho experimental --- la luz del Sol cruza $150$
millones de kilómetros de espacio vacío; se deduce honradamente en el
volumen del segundo año a partir de las leyes de la electricidad y el
magnetismo. Conviene retener el contraste: el \omterm{def:g10:signals-and-waves:sound}{sonido} es una vibración
\emph{del} medio y muere sin él; una \omterm{prop:g10:signals-and-waves:em}{señal electromagnética} lleva
consigo su propio campo a través del vacío, un millón de veces más
deprisa.
\end{remark}

\section{Los ecos: distancias a partir de retrasos}

Una \omterm{def:g10:signals-and-waves:signal}{señal} que se mueve a la velocidad $v$ recorre $d = v\,t$ en un
tiempo $t$: todo retraso es una distancia disfrazada. Mejor aún: se
envía un pulso corto contra un obstáculo y se cronometra su
\emph{eco}\index{eco} --- la ida y vuelta cubre $2d$.

\begin{method}[Medir distancias con ecos]\label{met:g10:signals-and-waves:echo}
\begin{enumerate}
  \item Enviar un pulso corto y medir el tiempo $t$ de ida y vuelta del
        \omterm{rem:g10:signals-and-waves:honest}{eco}.
  \item Tomar la velocidad $v$ de la \omterm{def:g10:signals-and-waves:signal}{señal} \emph{en el medio
        atravesado}.
  \item El pulso ha recorrido $2d = v\,t$, luego
        $d = \dfrac{v\,t}{2}$.
\end{enumerate}
Los pulsos de \omterm{def:g10:signals-and-waves:sound}{sonido} en el agua dan el \emph{sonar}\index{sonar}
($v \approx \qty{1500}{m/s}$); los pulsos de radio en el aire dan el
\emph{radar}\index{radar} ($v = c$); y los pulsos de \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonidos} en el
cuerpo dan el ecógrafo ($v \approx \qty{1540}{m/s}$ en tejido blando).
\end{method}

\begin{example}[El sonar]\label{ex:g10:signals-and-waves:sonar}
El \omterm{met:g10:signals-and-waves:echo}{sonar} de un barco emite un pulso y el \omterm{rem:g10:signals-and-waves:honest}{eco} del fondo vuelve al cabo de
$t = \qty{0.80}{s}$: la profundidad vale
$d = \frac{1500 \times 0.80}{2} = \qty{600}{m}$. Olvidarse del factor
$2$ duplica el océano.
\end{example}

\begin{example}[El radar]\label{ex:g10:signals-and-waves:radar}
Un \omterm{met:g10:signals-and-waves:echo}{radar} de aeropuerto recibe el \omterm{rem:g10:signals-and-waves:honest}{eco} de un avión $\qty{0.30}{ms}$
después del pulso:
$d = \frac{\qty{3.00e8}{m/s} \times \qty{3.0e-4}{s}}{2} = \qty{45}{km}$.
Como $c$ es tan grande, el \omterm{met:g10:signals-and-waves:echo}{radar} vive con relojes de microsegundos:
cronometrar la luz es trabajo de precisión.
\end{example}

\begin{example}[El corazón como señal]\label{ex:g10:signals-and-waves:ecg}
Cada latido envía un pequeño pulso de tensión a través del tórax; unos
electrodos lo registran como el
\emph{electrocardiograma}\index{electrocardiograma} (ECG), una \omterm{def:g10:signals-and-waves:signal}{señal}
casi \omterm{def:g10:signals-and-waves:period}{periódica}. Si las espigas altas distan $T = \qty{0.75}{s}$, el
corazón late a $f = 1/0.75 = \qty{1.33}{Hz}$, es decir
$1.33 \times 60 = 80$ latidos por minuto: leer $T$ en un trazado es como
te toma el pulso un cardiólogo.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:1}
Calcula la \omterm{def:g10:signals-and-waves:frequency}{frecuencia} de la \omterm{def:g10:signals-and-waves:period}{señal periódica} de \omterm{def:g10:signals-and-waves:period}{periodo}
(a) $T = \qty{20}{ms}$; (b) $T = \qty{4.0}{ms}$;
(c) $T = \qty{50}{\micro s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:2}
Calcula el \omterm{def:g10:signals-and-waves:period}{periodo} de (a) la red eléctrica de \qty{50}{Hz}; (b) un la de
\qty{440}{Hz}; (c) una \omterm{def:g10:signals-and-waves:signal}{señal} de wifi de \qty{2.4}{GHz}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:3}
Clasifica como infrasonido, \omterm{def:g10:signals-and-waves:sound}{sonido} audible o ultrasonido:
\qty{12}{Hz}; \qty{440}{Hz}; \qty{18}{kHz}; \qty{40}{kHz} (un sensor de
aparcamiento); \qty{5}{MHz} (una sonda médica).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:4}
En un \omterm{def:g10:signals-and-waves:oscilloscope}{osciloscopio} ajustado a \qty{2}{ms} por división y \qty{0.5}{V}
por división, un ciclo abarca $5.0$ divisiones y una cresta queda $4.0$
divisiones por encima del eje. Halla $T$, $f$ y la \omterm{def:g10:signals-and-waves:amplitude}{amplitud}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:5}
Ves el relámpago y oyes el trueno \qty{6.0}{s} después. ¿A qué distancia
cayó ($v_{\text{sound}} = \qty{340}{m/s}$)? ¿Por qué se puede despreciar
el tiempo de viaje de la luz?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:6}
Un \omterm{rem:g10:signals-and-waves:honest}{eco} de \omterm{met:g10:signals-and-waves:echo}{sonar} vuelve del fondo marino al cabo de \qty{0.60}{s}
($v = \qty{1500}{m/s}$ en agua de mar). ¿Por qué hay que dividir entre
dos el producto $v\,t$? Calcula la profundidad. ¿Cuánto tardaría el \omterm{rem:g10:signals-and-waves:honest}{eco}
sobre una fosa de \qty{1200}{m}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:7}
Un \omterm{met:g10:signals-and-waves:echo}{radar} de aeropuerto recibe el \omterm{rem:g10:signals-and-waves:honest}{eco} de un avión al cabo de
\qty{0.24}{ms} y, exactamente \qty{1.0}{s} después, un segundo \omterm{rem:g10:signals-and-waves:honest}{eco} al
cabo de \qty{0.238}{ms}. Calcula las dos distancias y después la
velocidad del avión. ¿Se está acercando?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:8}
Un ECG se imprime a \qty{25}{mm/s}; las espigas altas distan
\qty{20}{mm}. Halla el \omterm{def:g10:signals-and-waves:period}{periodo} y la \omterm{def:g10:signals-and-waves:frequency}{frecuencia} cardíaca en latidos por
minuto. Hay taquicardia por encima de $100$ latidos por minuto: ¿por
debajo de qué separación entre espigas aparece en este papel?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:9}
Tienes que mostrar una \omterm{def:g10:signals-and-waves:signal}{señal} de \qty{200}{Hz} y \omterm{def:g10:signals-and-waves:amplitude}{amplitud} \qty{3.0}{V} en
una pantalla de $10$ divisiones horizontales y $8$ verticales ($4$ por
encima de la línea central). Elige una \omterm{def:g10:signals-and-waves:oscilloscope}{base de tiempos} que muestre unos
dos ciclos completos y una \omterm{def:g10:signals-and-waves:oscilloscope}{ganancia} que aproveche casi toda la pantalla
sin recortar la \omterm{def:g10:signals-and-waves:signal}{señal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:10}
Una sonda médica envía un pulso de \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonidos} al abdomen
($v = \qty{1540}{m/s}$ en tejido blando) y oye \omterm{rem:g10:signals-and-waves:honest}{ecos} al cabo de

\section{Problema: el sonar, la tormenta y el cardiograma}

\begin{problem}[{Problema de fin de semana --- leer las señales del
mundo: una sola ley, $d = v\,t$, mide una tormenta, cartografía un fondo
marino, cuenta un latido y deja entrever cómo funciona el GPS}]
\label{pb:g10:signals-and-waves:1}
Un barco de pesca trabaja en una noche de tormenta: relámpagos en el
horizonte, el \omterm{met:g10:signals-and-waves:echo}{sonar} barriendo el fondo, el corazón del patrón sobre la
tira de papel del médico y un receptor GPS escuchando a los \omterm{def:g10:universal-gravitation:satellite}{satélites}.
Cuatro instrumentos, una ley. Toma
$v_{\text{sound}} = \qty{340}{m/s}$ en el aire, \qty{1500}{m/s} en el
agua de mar y $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Parte I --- La tormenta.}
\begin{enumerate}
  \item Un destello; el trueno llega \qty{9.0}{s} después. ¿A qué
        distancia está?
  \item Calcula el tiempo de viaje de la luz en esa distancia y
        justifica que se trate el destello como instantáneo.
  \item Los marineros cuentan los segundos entre el destello y el trueno
        y los dividen entre tres para obtener kilómetros. Justifica la
        regla.
  \item Cinco minutos después, un destello da \qty{6.0}{s}. ¿A qué
        distancia está ahora? Halla la \omterm{def:g10:relative-motion:speed}{velocidad media} de acercamiento
        de la tormenta en \unit{m/s} y en \unit{km/h}.
  \item El trueno retumba en lugar de restallar: el canal del rayo mide
        kilómetros, así que sus partes están a distancias distintas. Si
        el canal se extiende de \qty{2.0}{km} a \qty{5.0}{km} del barco,
        ¿cuánto dura el retumbo?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El \omterm{met:g10:signals-and-waves:echo}{sonar}.}
\begin{enumerate}[resume]
  \item El \omterm{met:g10:signals-and-waves:echo}{sonar} emite un pulso; el fondo responde en \qty{0.90}{s}.
        ¿Profundidad?
  \item Sobre la plataforma continental el \omterm{rem:g10:signals-and-waves:honest}{eco} se acorta a
        \qty{0.20}{s}. ¿Profundidad?
  \item Un pulso devuelve \emph{dos} \omterm{rem:g10:signals-and-waves:honest}{ecos}, a \qty{0.16}{s} y
        \qty{0.20}{s}. Interprétalos; ¿a qué altura sobre el fondo nada
        el banco de peces?
  \item El \omterm{met:g10:signals-and-waves:echo}{sonar} emite un pulso cada \qty{0.50}{s}. ¿Cuál es la mayor
        profundidad que puede medir sin confusión, y qué falla más
        allá?
  \item En el aire, ese mismo retraso de \qty{0.60}{s} significaría qué
        distancia? Moraleja: ¿qué hay que saber antes de convertir un
        retraso en una distancia?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El cardiograma.}
\begin{enumerate}[resume]
  \item Las espigas del ECG del patrón distan \qty{0.86}{s}.
        ¿\omterm{def:g10:signals-and-waves:frequency}{Frecuencia}? ¿Latidos por minuto?
  \item La tira avanza a \qty{25}{mm/s}. ¿Qué separación entre espigas
        midió el médico?
  \item Después de izar las redes, la separación es de \qty{15}{mm}.
        ¿Nueva \omterm{def:g10:signals-and-waves:frequency}{frecuencia} cardíaca?
  \item El joven marinero de cubierta, remero entrenado, descansa a $50$
        latidos por minuto. ¿\omterm{def:g10:signals-and-waves:period}{Periodo}? ¿Separación en la tira?
  \item ¿Es un ECG estrictamente periódico? Entonces, ¿qué lee el médico
        en el trazado?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- De vuelta a casa con la luz.}
\begin{enumerate}[resume]
  \item El barco llama por radio al puerto, a \qty{50}{km}. ¿Cuánto
        tarda el mensaje? ¿Cuánto tardaría el \omterm{def:g10:signals-and-waves:sound}{sonido}?
  \item Un \omterm{def:g10:universal-gravitation:satellite}{satélite} GPS orbita a unos \qty{20200}{km} de altura.
        ¿Cuánto tarda su \omterm{def:g10:signals-and-waves:signal}{señal} en llegar al barco?
  \item El GPS convierte tiempo en posición. ¿Qué error de distancia
        provoca un error de reloj de \qty{1.0}{\micro s}? ¿Qué precisión
        temporal exige una posición con \qty{3.0}{m} de error?
  \item ¿Por qué no podría existir un GPS basado en el \omterm{def:g10:signals-and-waves:sound}{sonido}, ni
        siquiera en principio?
  \item Final: enumera las tres velocidades usadas esta noche y la única
        ley que hay detrás de los cuatro instrumentos, y enuncia después
        en una frase la costumbre para medir distancias con retrasos.
\end{enumerate}
\end{problem}
