\chapter{Gravitación universal y peso}\label{ch:g10:universal-gravitation}

Suelta una manzana: cae. La Luna, a sesenta radios terrestres de altura,
parece no caer nunca. La gran intuición de Newton es que la Luna
\emph{sí} está cayendo --- unos 1.4 milímetros cada segundo, exactamente
lo que predice una atracción que se debilita con el cuadrado de la
distancia. Una sola fórmula, con una sola constante universal, gobierna
la manzana, la Luna, las mareas y los planetas. El primer volumen
describió la \omterm{def:g10:universal-gravitation:force}{gravitación} con palabras; este capítulo la mide.

\section{La ley de gravitación universal}

\begin{definition}[Fuerza gravitatoria]\label{def:g10:universal-gravitation:force}
Dos cuerpos cualesquiera se atraen, sea cual sea la materia de la que
estén hechos y por lejos que se encuentren. Esa atracción es la
\emph{fuerza gravitatoria}\index{fuerza gravitatoria}, y el fenómeno es
la \emph{gravitación}\index{gravitación}. No necesita contacto, ni
cuerda, ni medio alguno, y es \emph{mutua}: cada cuerpo tira del otro.
\end{definition}

\begin{theorem}[Ley de gravitación universal]\label{thm:g10:universal-gravitation:law}
Dos cuerpos de masas $m_1$ y $m_2$ (en \unit{kg}), cuyos centros distan
$d$ (en \unit{m}), se atraen con dos fuerzas dirigidas a lo largo de la
recta que une los centros, cada una apuntando hacia el otro cuerpo, de
igual módulo
\[
F = G\,\frac{m_1 m_2}{d^2}
\qquad \text{(en \unit{N})},
\]
donde $G$ es la misma constante para toda pareja de cuerpos del
universo.
\end{theorem}
\admitted

\begin{remark}[De dónde sale la ley]
Newton destiló esta ley del movimiento observado de los planetas (las
leyes de Kepler); la deducción honrada se hace en el volumen del primer
año. El \cref{ch:g12:satellites-kepler} pondrá ya la ley a trabajar
sobre \omterm{def:g10:universal-gravitation:satellite}{satélites} y planetas.
\end{remark}

\begin{definition}[La constante de gravitación]\label{def:g10:universal-gravitation:constant}
La constante
\[
G = \qty{6.67e-11}{N.m^2/kg^2}
\]
es la \emph{constante de gravitación}\index{constante de gravitación}.
Su valor minúsculo es la razón de que la gravedad entre objetos
cotidianos sea imperceptible: solo cuando al menos una de las masas es
astronómica llega $G m_1 m_2 / d^2$ a valer algo.
\end{definition}

\begin{remark}[Cómo leer la fórmula]\label{rem:g10:universal-gravitation:reading}
Cuatro rasgos merecen atención.
\begin{itemize}
  \item \emph{Mutua e igual.} La Tierra tira de ti con tu \omterm{def:g10:universal-gravitation:weight}{peso} --- y tú
        tiras de la Tierra con exactamente la misma fuerza. (La Tierra
        se limita a responder menos: es unas $\num{e23}$ veces más
        pesada.)
  \item \emph{Inversa del cuadrado.} Duplicar $d$ divide la fuerza entre
        $4$; alejarse diez veces más la divide entre $100$.
  \item \emph{Distancia entre centros.} Para los cuerpos esféricos ---
        estrellas, planetas, balas de cañón --- $d$ se mide de centro a
        centro, no de superficie a superficie.
  \item \emph{Las dos masas cuentan.} Duplicar cualquiera de las masas
        duplica la fuerza; la fórmula es simétrica en $m_1$ y $m_2$.
\end{itemize}
\end{remark}

\begin{example}[Dos amigos]\label{ex:g10:universal-gravitation:friends}
Dos amigos de \qty{70}{kg} están de pie con sus centros a \qty{1.0}{m}:
\[
F = \num{6.67e-11} \times \frac{70 \times 70}{1.0^2}
\approx \qty{3.3e-7}{N}.
\]
El \omterm{def:g10:universal-gravitation:weight}{peso} de cada uno es de unos \qty{6.9e2}{N} --- dos \emph{mil
millones} de veces mayor. Su atracción mutua es real, pero imposible de
notar.
\end{example}

\begin{example}[La Tierra sujeta a la Luna]\label{ex:g10:universal-gravitation:earthmoon}
Con $m_1 = \qty{5.97e24}{kg}$ (Tierra), $m_2 = \qty{7.35e22}{kg}$ (Luna)
y $d = \qty{3.84e8}{m}$:
\[
F = \num{6.67e-11} \times
\frac{\num{5.97e24} \times \num{7.35e22}}{(\num{3.84e8})^2}
\approx \qty{2.0e20}{N}.
\]
Y la Luna tira de la Tierra con los mismos \qty{2.0e20}{N} --- un tirón
al que los océanos responden dos veces al día: las mareas.
\end{example}

\begin{method}[Calcular una fuerza gravitatoria]\label{met:g10:universal-gravitation:compute}
\begin{enumerate}
  \item Pasar las masas a \omterm{def:g10:orders-of-magnitude:unit}{kilogramos} y la distancia de centro a centro a
        \omterm{def:g10:orders-of-magnitude:unit}{metros}.
  \item Aplicar $F = G m_1 m_2 / d^2$.
  \item Comprobar la potencia de diez por separado de las cifras, como
        en el \cref{ch:g10:orders-of-magnitude}: las \omterm{def:g10:universal-gravitation:force}{fuerzas
        gravitatorias} van de $\num{e-7}$ newtons (dos personas a un
        \omterm{def:g10:orders-of-magnitude:unit}{metro}) a $\num{e22}$ newtons (el Sol sobre la Tierra), de modo
        que un \omterm{def:g10:orders-of-magnitude:scinot}{exponente} mal puesto es fácil de cazar.
\end{enumerate}
\end{method}

\section{El peso y la intensidad de la gravedad}

\begin{definition}[Peso]\label{def:g10:universal-gravitation:weight}
El \emph{peso}\index{peso} $\vect P$ de un objeto en la superficie de un
astro (un planeta, una luna, \dots) es la \omterm{def:g10:universal-gravitation:force}{fuerza gravitatoria} que el
astro ejerce sobre él. Apunta hacia el centro del astro --- la dirección
que llamamos \emph{vertical}\index{vertical} --- y se mide en newtons.
\end{definition}

\begin{proposition}[El peso en la superficie de un astro]\label{prop:g10:universal-gravitation:surfaceg}
En la superficie de un astro esférico de masa $M$ y radio $R$, un objeto
de masa $m$ tiene un \omterm{def:g10:universal-gravitation:weight}{peso}
\[
P = m\,g,
\qquad\text{donde}\qquad
g = \frac{G M}{R^2}
\]
depende solo del astro, no del objeto.
\end{proposition}

\begin{proof}
El centro del objeto y el centro del astro distan $R$, de modo que la
ley de \omterm{def:g10:universal-gravitation:force}{gravitación} universal
(\cref{thm:g10:universal-gravitation:law}) da
$P = G M m / R^2 = m \times \left(G M / R^2\right)$. El paréntesis es el
mismo para cualquier objeto de la superficie: lo llamamos $g$.
\end{proof}

\begin{definition}[Intensidad del campo gravitatorio]\label{def:g10:universal-gravitation:fieldstrength}
La magnitud $g = G M / R^2$, en newtons por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo}, es la
\emph{intensidad del campo gravitatorio}\index{intensidad del campo
gravitatorio} en la superficie del astro: el astro tira de cada
\omterm{def:g10:orders-of-magnitude:unit}{kilogramo} con $g$ newtons. En la Tierra, $g = \qty{9.81}{N/kg}$.
\end{definition}

\begin{example}[Calcular $g$]\label{ex:g10:universal-gravitation:gvalues}
Para la Tierra ($M = \qty{5.97e24}{kg}$, $R = \qty{6.37e6}{m}$):
\[
g = \frac{\num{6.67e-11} \times \num{5.97e24}}{(\num{6.37e6})^2}
= \frac{\num{3.98e14}}{\num{4.06e13}}
\approx \qty{9.81}{N/kg}.
\]
Para la Luna ($M = \qty{7.35e22}{kg}$, $R = \qty{1.74e6}{m}$) el mismo
cálculo da $g_{\text{Moon}} \approx \qty{1.62}{N/kg}$ --- seis veces
menos: la Luna es mucho más ligera, y ser más pequeña no lo compensa.
Para Marte, $g_{\text{Mars}} \approx \qty{3.73}{N/kg}$.
\end{example}

\begin{remark}[Por qué los astronautas dan saltitos]\label{rem:g10:universal-gravitation:bounce}
Un astronauta de \qty{120}{kg} (cuerpo más traje) pesa
$120 \times 9.81 \approx \qty{1180}{N}$ en la Tierra, pero solo
$120 \times 1.62 \approx \qty{194}{N}$ en la Luna. Unos músculos
entrenados con \qty{1180}{N} cargan de pronto con la sexta parte: de ahí
el famoso andar lunar, lento y a brincos. La masa --- y con ella el
esfuerzo necesario para \emph{frenar} o \emph{girar} --- no ha
cambiado, y por eso los astronautas lunares se caían tan a menudo.
\end{remark}

\section{¿Masa o peso? La balanza y el dinamómetro}

El habla corriente dice que un saco de harina ``pesa tres kilos''. La
física parte esa frase en dos.

\begin{remark}[La masa no es el peso]\label{rem:g10:universal-gravitation:massweight}
\begin{itemize}
  \item La \emph{masa} $m$ mide la cantidad de materia, en \omterm{def:g10:orders-of-magnitude:unit}{kilogramos}.
        Es una propiedad del objeto solo: la misma en la Tierra, en la
        Luna o a la deriva en el espacio profundo.
  \item El \emph{\omterm{def:g10:universal-gravitation:weight}{peso}} $P = mg$ es una fuerza, en newtons, ejercida
        \emph{por un astro} sobre el objeto. Cambia de un astro a otro
        con $g$ --- y se desvanece lejos de todos ellos.
\end{itemize}
Confundirlos es inofensivo en el mercado y fatal en un problema de
física: el \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} no es una \omterm{def:g10:orders-of-magnitude:unit}{unidad} de fuerza.
\end{remark}

\begin{method}[La balanza y el dinamómetro]\label{met:g10:universal-gravitation:instruments}
\begin{itemize}
  \item Una \emph{balanza de brazos} compara el objeto con masas patrón:
        los \omterm{def:g10:universal-gravitation:weight}{pesos} de los dos platillos se multiplican por el mismo $g$,
        de modo que la comparación mide la \emph{masa} --- el mismo
        veredicto en cualquier astro.
  \item Un \emph{dinamómetro} mide la fuerza que estira su muelle: marca
        el \emph{\omterm{def:g10:universal-gravitation:weight}{peso}}, en newtons, y su lectura cambia de un astro a
        otro. Para recuperar la masa se divide entre la intensidad local
        del campo: $m = P/g$.
\end{itemize}
\end{method}

\begin{example}[Harina en la Luna]\label{ex:g10:universal-gravitation:flour}
Un saco de harina de \qty{3.0}{kg} equilibra a \qty{3.0}{kg} de masas
patrón en la Tierra \emph{y} en la Luna. Un dinamómetro, en cambio,
marca $3.0 \times 9.81 \approx \qty{29}{N}$ en la Tierra y
$3.0 \times 1.62 \approx \qty{4.9}{N}$ en la Luna. Un comerciante que
vendiera ``por newton'' en la Luna entregaría seis veces más harina ---
el comercio, como la ciencia, funciona con masas.
\end{example}

\section{Caer alrededor de la Tierra: el cañón de Newton}

Lanza una piedra en horizontal: la gravedad curva su camino en un arco y
cae unos \omterm{def:g10:orders-of-magnitude:unit}{metros} más allá. Lánzala más deprisa: cae más lejos. El
experimento mental de Newton lleva la idea al límite. La Tierra es
redonda y su superficie baja unos \qty{5}{m} por debajo de la horizontal
cada \qty{8}{km} recorridos; un cuerpo en caída libre baja \qty{4.9}{m}
en su primer segundo (el \cref{ch:g12:projectile-motion} lo
establecerá). Así que un proyectil que rasara la Tierra a \qty{8}{km/s}
caería exactamente tan deprisa como el suelo se curva: cae para siempre
sin llegar a aterrizar nunca.

\begin{definition}[Satélite y órbita]\label{def:g10:universal-gravitation:satellite}
Un \emph{satélite}\index{satélite} de un astro es un cuerpo que cae
alrededor del astro sin alcanzar nunca su superficie; el camino que
repite es su \emph{órbita}\index{órbita}.
\end{definition}

\begin{remark}[La Luna es una piedra que cae]\label{rem:g10:universal-gravitation:fallingmoon}
La Luna es el \omterm{def:g10:universal-gravitation:satellite}{satélite} natural de la Tierra: cae hacia nosotros unos
\qty{1.4}{mm} cada segundo, y su movimiento lateral la lleva de largo
antes de que pueda acercarse más
(el \cref{exo:g10:universal-gravitation:15} te permite rehacer la
comprobación que el propio Newton hizo de ese número). Los astronautas
de una estación en \omterm{def:g10:universal-gravitation:satellite}{órbita} están en la misma caída libre que su nave ---
eso, y no una supuesta ausencia de gravedad, es la razón de que floten.
La \omterm{def:g10:universal-gravitation:force}{gravitación} moldea todas las trayectorias, del arco de la piedra a la
elipse del planeta; la historia cuantitativa de las \omterm{def:g10:universal-gravitation:satellite}{órbitas} se cuenta en
el \cref{ch:g12:satellites-kepler}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:1}
Dos bailarines de \qty{60}{kg} se sujetan con sus centros a
\qty{0.80}{m}.
\begin{enumerate}
  \item Calcula su atracción gravitatoria mutua.
  \item Compárala con el \omterm{def:g10:universal-gravitation:weight}{peso} de un mosquito de \qty{2.5}{mg}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:2}
Una estudiante tiene una masa de \qty{55}{kg}.
\begin{enumerate}
  \item Calcula su \omterm{def:g10:universal-gravitation:weight}{peso} en la Tierra.
  \item Viaja a la Luna. ¿Cuáles son allí su masa y su \omterm{def:g10:universal-gravitation:weight}{peso}?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:3}
\begin{enumerate}
  \item Con la ficha de datos, recalcula la \omterm{def:g10:universal-gravitation:fieldstrength}{intensidad del campo
        gravitatorio} en la superficie de Marte.
  \item Se envía allí un vehículo de \qty{900}{kg}. Calcula su \omterm{def:g10:universal-gravitation:weight}{peso} en
        Marte y su \omterm{def:g10:universal-gravitation:weight}{peso} en la Tierra.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:4}
Con la ficha de datos, calcula la fuerza que la Tierra ejerce sobre la
Luna. ¿Qué fuerza ejerce la Luna sobre la Tierra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:5}
Dos cuerpos se atraen con una fuerza $F_0$. ¿En qué se convierte la
fuerza si:
\begin{enumerate}
  \item se duplica la distancia entre los centros;
  \item se reduce la distancia a la mitad;
  \item se duplican las dos masas (misma distancia);
  \item se triplica una masa \emph{y} se triplica la distancia?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:6}
Dos superpetroleros cargados de \qty{5.0e8}{kg} cada uno están amarrados
con sus centros a \qty{100}{m}.
\begin{enumerate}
  \item Calcula su atracción gravitatoria mutua.
  \item ¿Qué masa tiene en la Tierra un \omterm{def:g10:universal-gravitation:weight}{peso} igual a esa fuerza?
  \item La fuerza es sorprendentemente grande --- y sin embargo los
        petroleros no muestran ninguna tendencia a juntarse. Calcula la
        aceleración $a = F/m$ que le comunica a uno de ellos y coméntalo.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:7}
Una tienda lunar vende una bolsa de arroz de \qty{3.0}{kg}.
\begin{enumerate}
  \item ¿Qué indican para esa bolsa una balanza de brazos y un
        dinamómetro en la Tierra? ¿Y en la Luna?
  \item Un cliente paga por ``\qty{3.0}{kg}''. ¿Qué instrumento
        garantiza un trato justo en cualquier mundo, y por qué?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:8}
La Estación Espacial Internacional vuela a una altitud de \qty{400}{km}.
\begin{enumerate}
  \item Calcula la \omterm{def:g10:universal-gravitation:fieldstrength}{intensidad del campo gravitatorio} terrestre a esa
        altitud. (Cuidado: ¿cuál es la distancia al \emph{centro} de la
        Tierra?)
  \item Exprésala como porcentaje del valor en la superficie.
  \item Los astronautas de a bordo flotan. Concilia este hecho con tu
        respuesta.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:9}
En la superficie de Venus, $g = \qty{8.87}{N/kg}$, y el radio del
planeta es \qty{6.05e6}{m}. Deduce la masa de Venus y compárala con la
de la Tierra.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:10}
\begin{enumerate}
  \item Calcula la fuerza que ejerce el Sol sobre \qty{1.0}{kg} de agua
        de mar y después la fuerza que ejerce la Luna sobre ese mismo
        \omterm{def:g10:orders-of-magnitude:unit}{kilogramo}.
  \item El Sol tira con más fuerza --- ¿en qué factor?
  \item Y sin embargo la Luna domina las mareas. La marea no la produce
        el tirón en sí, sino la \emph{diferencia} entre los tirones
        sobre la cara próxima y la cara lejana de la Tierra. Explica,
        sin calcular, por qué la Luna cercana puede ganarle al Sol
        lejano con ese criterio.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:11}
\begin{enumerate}
  \item Un planeta tiene el doble de la masa de la Tierra y el doble de
        su radio. Expresa la intensidad de su campo en la superficie en
        función de la $g$ terrestre y después en \unit{N/kg}.
  \item ¿Qué radio necesitaría un planeta de la masa de la Tierra para
        que la intensidad de su campo en la superficie fuera $2g$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:12}
En algún punto de la recta Tierra--Luna, los dos tirones sobre una sonda
espacial se anulan. Sea $x$ la distancia del centro de la Tierra a ese
punto y $D$ la distancia Tierra--Luna.
\begin{enumerate}
  \item Demuestra que en ese punto
        $\left(\dfrac{x}{D - x}\right)^{2}
        = \dfrac{M_{\text{Earth}}}{M_{\text{Moon}}}$.
  \item Deduce $x$. ¿Qué fracción del viaje a la Luna es?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:13}
El cañón de Newton, con números. Un proyectil rasa la Tierra en
horizontal con velocidad $v$.
\begin{enumerate}
  \item Tras $x = \qty{8.0}{km}$ de recorrido horizontal, la Tierra
        esférica se ha ``caído'' una altura $h$ que cumple
        $(R + h)^2 = R^2 + x^2$. Usando
        $(R + h)^2 \approx R^2 + 2Rh$ para $h$ pequeña, demuestra que
        $h \approx x^2 / (2R)$ y calcula $h$.
  \item Un cuerpo en caída libre baja \qty{4.9}{m} en su primer segundo.
        Compara con $h$ y deduce la velocidad a la que el proyectil no
        aterriza nunca.
  \item Calcula la duración de una \omterm{def:g10:universal-gravitation:satellite}{órbita} completa a esa velocidad,
        rasando la superficie.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:14}
Una estrella de neutrones aprieta $M = \qty{2.8e30}{kg}$ (1.4 veces la
masa del Sol) en un radio de \qty{12}{km}.
\begin{enumerate}
  \item Calcula la \omterm{def:g10:universal-gravitation:fieldstrength}{intensidad del campo gravitatorio} en su superficie.
  \item ¿Cuánto pesaría allí una persona de \qty{70}{kg}? Compáralo con
        su \omterm{def:g10:universal-gravitation:weight}{peso} en la Tierra.
  \item ¿Cuál es allí el \omterm{def:g10:universal-gravitation:weight}{peso} de un grano de arena de \qty{1.0}{mg}?
        ¿Qué masa tiene ese \omterm{def:g10:universal-gravitation:weight}{peso} en la Tierra?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:15}
La prueba lunar de Newton (1666). El radio de la \omterm{def:g10:universal-gravitation:satellite}{órbita} de la Luna es de
unos $60$ radios terrestres.
\begin{enumerate}
  \item ¿En qué factor es más débil el tirón terrestre por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} en
        la Luna que en la superficie de la Tierra, si la gravedad sigue
        la inversa del cuadrado?
  \item Cerca de la superficie, un cuerpo que cae baja \qty{4.9}{m} en
        su primer segundo. ¿Cuánto debería caer la Luna hacia la Tierra
        cada segundo?
  \item Compruébalo con la \omterm{def:g10:universal-gravitation:satellite}{órbita}: la Luna recorre su circunferencia de
        radio $d = \qty{3.84e8}{m}$ en $T = 27.3$ días. Calcula su
        velocidad $v$, la distancia $x$ que recorre en un segundo y
        después la caída $h \approx x^2/(2d)$ (como en el
        \cref{exo:g10:universal-gravitation:13}).
  \item Compara las respuestas de las preguntas 2 y 3, y di qué concluyó
        Newton.
\end{enumerate}
\end{exercise}

\section{Problema: pesar la Tierra}

\begin{problem}[{Problema de fin de semana --- de dos esferas de plomo
en un cobertizo a la masa de la Tierra y después a la del Sol: lo que
vale una constante pequeña medida una sola vez}]
\label{pb:g10:universal-gravitation:1}
En 1798 Henry Cavendish colgó de un hilo fino una varilla ligera con dos
esferas pequeñas de plomo, acercó dos esferas grandes de plomo y midió
el retorcimiento casi inexistente del hilo. Los periódicos dijeron que
había \emph{pesado la Tierra} --- y tenían razón: una vez que se conoce
$G$, la propia $g$ terrestre y el radio entregan su masa, y la \omterm{def:g10:universal-gravitation:satellite}{órbita}
anual de la Tierra entrega la del Sol. Este problema recorre la cadena
entera. Usa la ficha de datos en todo momento.

\medskip
\noindent\textbf{Parte I --- La fuerza más endeble.}
\begin{enumerate}
  \item El hierro de la torre Eiffel tiene una masa de unos
        \qty{7.3e6}{kg}. Un turista de \qty{55}{kg} está a \qty{100}{m}
        de su centro de masas. Calcula el tirón de la torre sobre el
        turista.
  \item Compáralo con el \omterm{def:g10:universal-gravitation:weight}{peso} de un grano de arena de \qty{1.0}{mg}.
  \item Ejercicio de proporcionalidad: ¿qué le ocurre a una \omterm{def:g10:universal-gravitation:force}{fuerza
        gravitatoria} si (a) la distancia se multiplica por $10$; (b) las
        dos masas se multiplican por $1000$; (c) las dos masas \emph{y}
        la distancia se multiplican por $1000$?
  \item Calcula el tirón de la Tierra sobre un objeto de \qty{1.00}{kg}
        situado en su superficie, a partir de $M$, $R$ y $G$. ¿Qué
        nombre y qué símbolo cotidianos lleva ese número?
  \item La gravedad es con mucho la interacción más endeble que has
        conocido --- y sin embargo gobierna el universo. Explica en una
        o dos frases por qué el tirón de la Tierra sobre ti domina al de
        tu vecino.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La balanza de Cavendish.}
En la balanza de torsión, una esfera grande de masa $m_1 = \qty{158}{kg}$
atrae a una pequeña de masa $m_2 = \qty{0.73}{kg}$; sus centros distan
$d = \qty{0.23}{m}$.
\begin{enumerate}[resume]
  \item Con el valor moderno de $G$, calcula la fuerza entre las dos
        esferas.
  \item Calcula el \omterm{def:g10:universal-gravitation:weight}{peso} de la esfera pequeña y el cociente de las dos
        fuerzas. ¿Por qué necesitaba Cavendish un hilo de torsión y no
        una báscula corriente?
  \item El retorcimiento que Cavendish midió corresponde (en \omterm{def:g10:orders-of-magnitude:unit}{unidades}
        modernas) a $F = \qty{1.47e-7}{N}$. Deduce su valor de $G$ y
        compáralo con el de hoy.
  \item Con $G = \qty{6.67e-11}{N.m^2/kg^2}$, $g = \qty{9.81}{N/kg}$ y
        $R = \qty{6.37e6}{m}$, deduce la masa de la Tierra. (Este es el
        paso que celebraron los periódicos.)
  \item Deduce la densidad media de la Tierra $\rho = M / V$, con
        $V = \frac43 \pi R^3$. Las rocas de la superficie tienen
        densidades de unos \qty{2.7e3}{kg/m^3}: ¿qué revela la
        comparación sobre el interior profundo?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Otros mundos.}
\begin{enumerate}[resume]
  \item Con la ficha de datos, recalcula la intensidad del campo en la
        superficie de la Luna.
  \item En la Tierra, un buen impulso eleva tu centro de masas
        \qty{0.50}{m}. Para el mismo impulso, la altura alcanzada es
        inversamente proporcional a $g$ (como justificará el capítulo de
        energía, \cref{ch:g10:energy-conservation}). ¿Hasta qué altura
        te lleva el mismo salto en la Luna?
  \item Mercurio: $M = \qty{3.30e23}{kg}$, $R = \qty{2.44e6}{m}$.
        Calcula la intensidad de su campo en la superficie y compárala
        con la de Marte. Mercurio es mucho más pequeño que Marte:
        ¿cómo pueden ser los dos valores tan parecidos?
  \item ¿A qué altitud sobre la superficie de la Tierra ha caído la
        intensidad del campo a la mitad de su valor en la superficie?
  \item ¿A qué distancia del centro de la Tierra baja el tirón terrestre
        por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} hasta el valor de la superficie lunar,
        \qty{1.62}{N/kg}? Exprésala en radios terrestres.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Pesar el Sol.}
La Tierra recorre una \omterm{def:g10:universal-gravitation:satellite}{órbita} casi circular de radio
$d = \qty{1.496e11}{m}$ en un año, $T = \qty{3.156e7}{s}$. Admite (se
demuestra en el \cref{ch:g12:satellites-kepler}) que un cuerpo que
recorre una trayectoria circular de radio $d$ con velocidad $v$ necesita
un tirón de $v^2/d$ newtons por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} hacia el centro.
\begin{enumerate}[resume]
  \item Calcula la velocidad orbital $v$ de la Tierra.
  \item Deduce el tirón por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} que el Sol ejerce sobre la Tierra.
  \item Ese tirón vale también $G M_{\text{Sun}} / d^2$. Deduce la masa
        del Sol.
  \item ¿Cuántas Tierras son? Compara $M_{\text{Sun}}$ con
        $M_{\text{Earth}}$.
  \item Final. Las esferas de Cavendish, la Tierra, el Sol: una sola
        constante $G$ ha servido para las tres. Di en una frase qué nos
        ha comprado en este problema la palabra \emph{universal} de
        ``\omterm{def:g10:universal-gravitation:force}{gravitación} universal''.
\end{enumerate}
\end{problem}
