\chapter{ऊर्जा: रूप और संरक्षण}\label{ch:g10:energy-conservation}

चलती कार, चार्ज किया हुआ फ़ोन, गरम तंदूर और उठाया हुआ हथौड़ा --- चारों
में एक गुण साझा है: हर एक कुछ करा सकता है। भौतिकी इस क्षमता को एक ही
संख्या से नापती है --- ऊर्जा --- और उसका हिसाब एक लोहे जैसे नियम
के नीचे रखती है: कुल कभी नहीं बदलता, वह केवल जगह बदलता है और भेस।
यह अध्याय वही बही खोलता है: ऊर्जा जो रूप लेती है, जिन शृंखलाओं से
गुज़रती है, और एक \omterm{def:g10:energy-conservation:kwh}{किलोवाट-घंटा} क्या ख़रीदता है।

\section{ऊर्जा: साझी मुद्रा}

\begin{definition}[ऊर्जा और जूल]\label{def:g10:energy-conservation:energy}
\emph{ऊर्जा}\index{ऊर्जा} किसी निकाय की वह क्षमता नापती है जिससे वह
बदलाव पैदा कर सके: पदार्थ को गति दे, उठाए, गरम करे, रोशन करे। उसका
मात्रक \emph{जूल}\index{जूल} (\unit{J}) है। संचित, स्थानांतरित,
रूपांतरित --- भौतिकी की हर शाखा इसी एक मुद्रा में हिसाब रखती है।
\end{definition}

\begin{definition}[गतिज ऊर्जा]\label{def:g10:energy-conservation:kinetic}
$m$ द्रव्यमान (\unit{kg} में) का कोई पिंड $v$ चाल (\unit{m/s} में) से चलते
हुए \emph{गतिज ऊर्जा}\index{गतिज ऊर्जा} $E_k = \tfrac12\, m v^2$ ढोता है: द्रव्यमान दुगुना
करने पर वह दुगुनी हो जाती है, और चाल दुगुनी करने पर चौगुनी।
\end{definition}

\begin{example}[कार बनाम राहगीर]\label{ex:g10:energy-conservation:kinetic}
\qty{100}{km/h} $= \qty{27.8}{m/s}$ पर चलती \qty{1300}{kg} की कार $E_k = \tfrac12 \times 1300 \times 27.8^2 \approx \qty{5.0e5}{J}$ ढोती है; \qty{1.5}{m/s} पर चलता
\qty{70}{kg} का राहगीर $\tfrac12 \times 70 \times 1.5^2 \approx \qty{79}{J}$ --- यानी छह हज़ार गुना कम। चाल महँगी पड़ती है:
वर्ग इसका ध्यान रखता है।
\end{example}

\begin{definition}[गुरुत्वीय स्थितिज ऊर्जा]\label{def:g10:energy-conservation:potential}
किसी चुने गए संदर्भ-तल से $h$ ऊँचाई (\unit{m} में) पर रखा $m$
द्रव्यमान का पिंड \emph{गुरुत्वीय स्थितिज ऊर्जा}\index{स्थितिज
ऊर्जा!गुरुत्वीय} $E_p = m g h$ संचित रखता है, जहाँ $g = \qty{9.81}{N/kg}$ --- यह ज़मीन के पास
ही सही है, जहाँ $g$ व्यावहारिक रूप से अचर है। मायने केवल ऊँचाई के
\emph{अंतर} रखते हैं: जिस तल पर $E_p = 0$ लिया जाए वह स्वतंत्र रूप से चुना
जा सकता है।
\end{definition}

\begin{example}[पानी की टंकी]\label{ex:g10:energy-conservation:potential}
पानी की टंकी में \qty{30}{m} ऊपर संचित हर घन \omterm{def:g10:orders-of-magnitude:unit}{मीटर} (\qty{1000}{kg}) पानी
$1000 \times 9.81 \times 30 \approx \qty{2.9e5}{J}$ रखता है --- यानी ऊपर वाली कार की \omterm{def:g10:energy-conservation:kinetic}{गतिज ऊर्जा} का लगभग
छह-दसवाँ हिस्सा। शहर सिर के ऊपर ऊर्जा जमा करते हैं।
\end{example}

\begin{definition}[बाक़ी रूप]\label{def:g10:energy-conservation:forms}
ऊर्जा कम दिखने वाले भेसों में भी छिपी रहती है: \emph{तापीय
ऊर्जा}\index{तापीय ऊर्जा}, यानी किसी पिंड के अणुओं की अव्यवस्थित
हलचल (जितना गरम, उतनी अधिक); \emph{रासायनिक ऊर्जा}\index{रासायनिक
ऊर्जा}, जो आणविक बंधों में संचित रहती है --- भोजन, लकड़ी, पेट्रोल,
बैटरियाँ; \emph{विद्युत ऊर्जा}\index{विद्युत ऊर्जा}, जिसे धाराएँ संयंत्र
से सॉकेट तक ढोती हैं; \emph{विकिरण ऊर्जा}\index{विकिरण ऊर्जा}, जिसे
प्रकाश खाली अंतरिक्ष के पार भी ले जाता है --- सूर्य की ऊर्जा
इसी तरह हम तक आती है; और \emph{नाभिकीय ऊर्जा}\index{नाभिकीय ऊर्जा}, जो
परमाणु नाभिकों के भीतर संचित है और तारों में मुक्त होती है (\cref{ch:g12:nuclear-energy})।
\end{definition}

\section{स्थानांतरण और रूपांतरण शृंखलाएँ}

\begin{definition}[स्थानांतरण, रूपांतरण, शृंखला]\label{def:g10:energy-conservation:transfer}
\emph{ऊर्जा स्थानांतरण}\index{ऊर्जा स्थानांतरण} ऊर्जा को एक
निकाय से दूसरे --- यानी \emph{भंडारों}\index{भंडार (ऊर्जा का)} के बीच ---
ले जाता है; \emph{ऊर्जा रूपांतरण}\index{ऊर्जा रूपांतरण} उसका रूप बदल
देता है। जो युक्ति इनमें से कुछ भी करे (टरबाइन, मोटर, लैंप, माँसपेशी) वह
\emph{रूपांतरक}\index{रूपांतरक} है। \emph{रूपांतरण
शृंखला}\index{रूपांतरण शृंखला} पूरी यात्रा का चित्र खींचती है: भंडारों
और रूपों के लिए डिब्बे, और स्थानांतरणों के लिए नामांकित तीर।
\end{definition}

\begin{method}[रूपांतरण शृंखला खींचना]\label{met:g10:energy-conservation:chain}
\begin{enumerate}
  \item आरंभिक भंडार और रूप पहचानिए (खपत किसकी हो रही है?)।
  \item ऊर्जा का पीछा रूपांतरक-दर-रूपांतरक
        कीजिए: हर स्थानांतरण के लिए एक तीर, और हर डिब्बे में रूप का नाम।
  \item हर असली \omterm{def:g10:energy-conservation:transfer}{रूपांतरक} पर एक ऊष्मा-तीर भी जोड़िए --- कुछ
        ऊर्जा हमेशा ऊष्मा बनकर रिसती है (\omterm{def:g10:inertia:inventory}{घर्षण}, तार,
        निकास)।
\end{enumerate}
\end{method}

निचले बिंदु पर पूरी गतिज (वहाँ $E_p = 0$ लेकर):

\section{ऊर्जा संरक्षण}

\begin{theorem}[ऊर्जा संरक्षण]\label{thm:g10:energy-conservation:conservation}
\index{ऊर्जा संरक्षण}
किसी विलगित निकाय --- यानी ऐसे निकाय का जो बाहर से कुछ भी आदान-प्रदान न
करे --- की कुल ऊर्जा अचर रहती है। ऊर्जा न कभी बनती है, न
कभी नष्ट होती है: वह केवल रूप और जगह बदलती है, और जो एक भंडार खोता है वही
बाक़ी पाते हैं, जूल-दर-जूल।
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:energy-conservation:admitted}
किसी प्रयोग ने इस सिद्धांत को आज तक चूकते नहीं पकड़ा; यहाँ हम इसे दिया
हुआ मानकर संख्याओं में काम में लेते हैं। \cref{ch:g12:work-and-energy} इसका यांत्रिक हिस्सा
($E_k + E_p$) न्यूटन के नियमों से निकालता है; पूरा हिसाब स्नातक स्तर के खंडों
की प्रतीक्षा करता है।
\end{remark}

\begin{example}[मुक्त गिरावट, जूलों में]\label{ex:g10:energy-conservation:fall}
\qty{2.0}{kg} की एक गेंद \qty{10}{m} से गिराई जाती है (वायु का प्रतिरोध नगण्य): वह
$E_p = 2.0 \times 9.81 \times 10 \approx \qty{196}{J}$ से चलती है और $E_k = \qty{196}{J}$ पर उतरती है, इसलिए $v = \sqrt{2 \times 196/2.0} =
\qty{14}{m/s} \approx \qty{50}{km/h}$। न बल, न
घड़ी: अकेला संरक्षण ही टक्कर का दाम बता देता है।
\end{example}

\section{दक्षता}

\begin{definition}[दक्षता]\label{def:g10:energy-conservation:efficiency}
कोई \omterm{def:g10:energy-conservation:transfer}{रूपांतरक} $E_{\text{खपत}}$ खपाता है और चाहा गया रूप $E_{\text{उपयोगी}}$ देता है;
उसकी \emph{दक्षता}\index{दक्षता}
\[
\eta = \frac{E_{\text{उपयोगी}}}{E_{\text{खपत}}} < 1 .
\]
होती है। संरक्षण $\eta > 1$ को मना करता है, और असली \omterm{def:g10:energy-conservation:transfer}{रूपांतरक} $1$
तक कभी नहीं पहुँचते: लापता ऊर्जा नष्ट नहीं होती --- वह ऊष्मा
बनकर निकल जाती है।
\end{definition}

\begin{example}[पेट्रोल कहाँ जाता है]\label{ex:g10:energy-conservation:engine}
एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पेट्रोल \qty{45}{MJ} \omterm{def:g10:energy-conservation:forms}{रासायनिक ऊर्जा} संचित रखता
है; कार का इंजन उसे $\eta \approx 0.30$ से रूपांतरित करता है: गति के \qty{13.5}{MJ}, और
\qty{31.5}{MJ} निकास, रेडिएटर और सड़क को गरम करते हुए। हर टंकी का दो-तिहाई
वायुमंडल गरम करता है।
\end{example}

\begin{remark}[दक्षताएँ गुणा होती हैं]\label{rem:g10:energy-conservation:chain-eta}
एक के बाद एक जुड़े \omterm{def:g10:energy-conservation:transfer}{रूपांतरक} अपनी दक्षताएँ गुणा कर देते हैं: कोई
शक्ति-संयंत्र ($0.38$) ग्रिड को ($0.94$) और ग्रिड किसी चार्जर को
($0.85$) खिलाए तो कुल $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.30$ मिलता है। लंबी शृंखलाएँ रिसती पाइप हैं: हर
तीर बहाव पर कर लगा देता है।
\end{remark}

\section{शक्ति, वाट और किलोवाट-घंटा}

\begin{definition}[शक्ति]\label{def:g10:energy-conservation:power}
\emph{शक्ति}\index{शक्ति} वह दर है जिससे ऊर्जा बहती है, $P = E/t$,
और उसे \emph{वाट}\index{वाट} (\unit{W}) में नापा जाता है, साथ में
$\qty{1}{W} = \qty{1}{J/s}$। ऊर्जा एक मात्रा है और शक्ति एक बहाव --- जैसे बाथटब और
उसकी टोंटी।
\end{definition}

\begin{example}[केतली बनाम मनुष्य]\label{ex:g10:energy-conservation:power}
\qty{150}{s} चलती \qty{2200}{W} की केतली $E = 2200 \times 150 = \qty{3.3e5}{J}$ खपाती है। मनुष्य दिन भर में लगभग
\qty{10}{MJ} भोजन पर चलता है, यानी औसतन $\num{1.0e7}/\num{86400} \approx \qty{116}{W}$: आप किसी पुराने चमकीले बल्ब
जितनी ही धीमी आँच पर चलते हैं।
\end{example}

\begin{definition}[किलोवाट-घंटा]\label{def:g10:energy-conservation:kwh}
\emph{किलोवाट-घंटा}\index{किलोवाट-घंटा} (\unit{kW.h}) वह ऊर्जा है जो
\qty{1}{kW} का बहाव एक घंटे तक चलने पर देता है:
$\qty{1}{kW.h} = 1000 \times 3600\,\unit{J} = \qty{3.6}{MJ}$ --- और बिजली के \omterm{def:g10:orders-of-magnitude:unit}{मीटर} इसी मात्रक में गिनती करते हैं:
\unit{kW} में शक्ति गुणा घंटे।
\end{definition}

\begin{example}[बिल पढ़ना]\label{ex:g10:energy-conservation:bill}
बिल एक घटाव और एक गुणा है: \omterm{def:g10:orders-of-magnitude:unit}{मीटर} का नया पाठ घटा पुराना पाठ
किलोवाट-घंटे देता है; उन्हें प्रति मात्रक दाम (लगभग
\unit{kW.h} के $0.25$ यूरो) से गुणा कीजिए, ऊपर से नियत सदस्यता जोड़िए, और
कुल मिल गया। हर रात \qty{2.5}{h} चलने वाला \qty{2}{kW} का हीटर रोज़ \qty{5}{kW.h}
खपाता है: यानी अकेले उसी के महीने में $38$ यूरो।
\end{example}

\begin{example}[परिमाण की कोटियाँ]\label{ex:g10:energy-conservation:orders}
ये कंठस्थ रखने लायक हैं: एक सेब को एक \omterm{def:g10:orders-of-magnitude:unit}{मीटर} उठाना, लगभग \qty{1}{J};

दिन भर की बिजली है ($\approx \qty{12}{kW.h}$); और बिजली की एक कड़क, \qty{5}{GJ}: यानी सौ
घर-दिन, एक चमक में।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:1}
\qty{1200}{kg} की एक कार \qty{90}{km/h}, यानी \qty{25}{m/s}, से चलती है: उसकी
\omterm{def:g10:energy-conservation:kinetic}{गतिज ऊर्जा} निकालिए। अब \qty{130}{km/h} (\qty{36.1}{m/s}) पर दोहराइए: $44\%$
अधिक चाल ने $E_k$ के साथ क्या किया?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:2}
\qty{72}{kg} द्रव्यमान का एक पदयात्री \qty{1200}{m} की ऊँचाई से \qty{1650}{m} तक चढ़ता
है। अर्जित स्थितिज ऊर्जा निकालिए, उसे
किलोवाट-घंटों में बदलिए, और \unit{kW.h} के $0.25$ यूरो के भाव पर
उसका दाम लगाइए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:3}
बदलिए: \qty{1}{kW.h} को जूलों में; \qty{12}{W.h} की फ़ोन बैटरी को
किलोजूलों में; \qty{45}{MJ} (एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पेट्रोल) को
किलोवाट-घंटों में; और अपने \qty{10}{MJ} वाले भोजन-दिन को
किलोवाट-घंटों में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:4}
\qty{2200}{W} की एक केतली \qty{150}{s} तक चलती है। खपाई गई ऊर्जा
जूलों और किलोवाट-घंटों में निकालिए। फिर प्रति दिन
\qty{10}{MJ} पर चलते मनुष्य की औसत शक्ति निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:5}
इनकी \omterm{def:g10:energy-conservation:transfer}{रूपांतरण शृंखला} (\cref{met:g10:energy-conservation:chain}) लिखिए: लैंप जलाता साइकिल का
डायनमो; गैस का पानी गरम करने वाला यंत्र; फ़ोन चार्ज करता सौर पैनल।
ऊष्मा के रिसाव भी अंकित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:6}
कोई इंजन \qty{2.0}{kg} पेट्रोल (\qty{45}{MJ/kg}) जलाता है और \qty{27}{MJ} गति देता है।
उसकी \omterm{def:g10:energy-conservation:efficiency}{दक्षता} और ऊष्मा के रूप में मुक्त हुई ऊर्जा
निकालिए। वह ऊष्मा कहाँ जाती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:7}
\qty{70}{kg} का एक गोताख़ोर \qty{20}{m} ऊँची चट्टान से कूदता है (वायु का
प्रतिरोध नगण्य)। संरक्षण से पानी पर उसकी \omterm{def:g10:energy-conservation:kinetic}{गतिज ऊर्जा} और प्रवेश
की चाल \unit{m/s} तथा \unit{km/h} में निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:8}
1 मार्च को \omterm{def:g10:orders-of-magnitude:unit}{मीटर} \qty{41236}{kW.h} पढ़ता था और 30 दिन बाद \qty{41562}{kW.h}: खपी हुई
ऊर्जा, दैनिक औसत, \unit{kW.h} के $0.26$ यूरो के भाव पर और $12$
यूरो नियत शुल्क जोड़कर बिल, और \omterm{def:g10:energy-conservation:power}{वाट} में औसत शक्ति?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:9}
कोई पुराना \qty{60}{W} का बल्ब अपनी खपत का केवल $5\%$ प्रकाश में बदलता
है; एक LED उतना ही प्रकाश \qty{9}{W} से दे देता है। बल्ब की प्रकाश
शक्ति और LED की \omterm{def:g10:energy-conservation:efficiency}{दक्षता} निकालिए; फिर प्रति दिन \qty{4}{h}
के हिसाब से हर एक की सालाना खपत, और \unit{kW.h} के $0.25$ यूरो के भाव पर
बचत।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:10}
कोई बाँध प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} \qty{150}{m^3} पानी (\qty{1.5e5}{kg}) को \qty{40}{m} की गिरावट से
नीचे भेजता है। प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} मुक्त होती स्थितिज ऊर्जा --- यानी
उपलब्ध शक्ति --- निकालिए, फिर $\eta = 0.90$ पर विद्युत
शक्ति, और यह भी कि वह \qty{525}{W} औसत वाले कितने घरों को आपूर्ति
देता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:11}
\qty{0.20}{kg} का एक लोलक अपने सबसे निचले बिंदु से \qty{5.0}{cm} ऊपर से छोड़ा जाता
है: वहाँ उसकी चाल निकालिए। कुछ मिनट बाद वह चुपचाप लटका रह जाता है: उसकी
ऊर्जा कहाँ गई, और यह \cref{thm:g10:energy-conservation:conservation} का खंडन क्यों नहीं करता?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:12}
कोई कोयला संयंत्र ($\eta_1 = 0.38$), ग्रिड ($\eta_2 = 0.94$) और एक चार्जर ($\eta_3 = 0.85$)
मिलकर \qty{12}{W.h} की फ़ोन बैटरी फिर से भरते हैं। कुल \omterm{def:g10:energy-conservation:efficiency}{दक्षता}
निकालिए; संयंत्र में खपी \omterm{def:g10:energy-conservation:forms}{रासायनिक ऊर्जा} \unit{W.h} और किलोजूलों
में; और प्रति चार्ज कोयला (\qty{25}{MJ/kg}), फिर रोज़ चार्ज करने के एक वर्ष के
लिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:13}
बिजली की एक कड़क लगभग \qty{5}{GJ} ढोती है। उसे किलोवाट-घंटों में
और प्रति दिन \qty{12.6}{kW.h} वाले घर के दिनों में बदलिए। वह लगभग \qty{100}{\micro s} में दी
जाती है: शक्ति निकालिए, उसकी तुलना मानवता की औसत विद्युत
शक्ति (\qty{3}{TW}) से कीजिए, और समझाइए कि बिजली को कोई क्यों नहीं
समेटता।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:14}
\qty{1300}{kg} की एक कार \qty{110}{km/h} (\qty{30.6}{m/s}) से ब्रेक लगाकर रुक जाती है। नष्ट
हुई \omterm{def:g10:energy-conservation:kinetic}{गतिज ऊर्जा} निकालिए; वह किस रूप में और कहाँ जाती है? कोई
विद्युत कार उसका $60\%$ अपनी बैटरी में वापस पा लेती है: प्रति ठहराव
ऊर्जा, और एक \omterm{def:g10:energy-conservation:kwh}{किलोवाट-घंटा} भरने के लिए कितने ठहराव
चाहिए?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:15}
एक साइकिल सवार और साइकिल (कुल \qty{85}{kg}) किसी पहाड़ी दर्रे पर चढ़ते हैं और
\qty{1200}{m} ऊँचाई पाते हैं; माँसपेशियाँ भोजन को $\eta \approx 0.25$ से रूपांतरित करती
हैं। चढ़ाई की यांत्रिक ऊर्जा, उसके लिए ज़रूरी भोजन-ऊर्जा,
उसके बराबर \qty{1.5}{MJ} वाले अनाज-कटोरों की संख्या, और उतनी ही ऊर्जा
संचित रखने वाले पेट्रोल का द्रव्यमान (\qty{45}{MJ/kg}) निकालिए। टिप्पणी कीजिए।
\end{exercise}

\section{समस्या: एक दिन के लिए घर चलाना}

\begin{problem}[{सप्ताहांत समस्या --- अनाज के कटोरे से छत के पैनलों तक:
घर के एक साधारण दिन का पूरा ऊर्जा-लेखा, और यह कि एक जूल का दाम इस पर
निर्भर करता है कि उसे बेच कौन रहा है}]
\label{pb:g10:energy-conservation:1}
किसी साधारण मंगलवार को एक परिवार यह खोजने निकलता है कि उसकी बिजली जाती
कहाँ है, और उसके पास हैं यह अध्याय, गलियारे का \omterm{def:g10:orders-of-magnitude:unit}{मीटर} और बिल का पिछला
पन्ना। आज रात का फ़ैसला: उनका एक दिन एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पेट्रोल के
बराबर है, या सोलह वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} धूप के --- या चौबीसों घंटे पैडल
मारते पाँच साइकिल सवारों के।

\medskip
\noindent\textbf{भाग I --- लेखा।}
दिन भर की सूची: फ़्रिज \qty{100}{W}, \qty{24}{h} तक; पानी का हीटर \qty{2400}{W},
\qty{2.5}{h} तक; तंदूर \qty{2000}{W}, \qty{1.0}{h} तक; कपड़े धोने की मशीन \qty{500}{W},
\qty{2.0}{h} तक; रोशनी \qty{60}{W}, \qty{5.0}{h} तक; परदे \qty{150}{W}, \qty{6.0}{h} तक।
\begin{enumerate}
  \item हर उपकरण की दिन भर की ऊर्जा \unit{kW.h} में निकालिए।
  \item दिन का कुल किलोवाट-घंटों में, फिर मेगाजूलों में जोड़िए;
        और उसकी तुलना एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पेट्रोल (\qty{45}{MJ}) से कीजिए।
  \item \qty{24}{h} पर घर की औसत शक्ति निकालिए; वह चलती केतली का
        कौन-सा अंश है?
  \item अकेला कौन-सा उपकरण सब पर भारी है? यदि हीटर केवल \qty{1.25}{h} चलता तो
        दिन का कुल फिर से निकालिए।
  \item प्रतीक्षा अवस्था की बत्तियाँ लगातार \qty{15}{W} खींचती हैं: दैनिक
        ऊर्जा, कुल में हिस्सा, और \unit{kW.h} के $0.25$ यूरो के
        भाव पर सालाना ख़र्च?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- बिल।}
\begin{enumerate}[resume]
  \item 30 दिनों में \omterm{def:g10:orders-of-magnitude:unit}{मीटर} \qty{56214}{kW.h} से \qty{56604}{kW.h} पर पहुँचा: खपत कितनी, और
        लेखे के सामने जाँचा गया दैनिक औसत कितना?
  \item बिल निकालिए: \unit{kW.h} के $0.25$ यूरो, ऊपर से $14$ यूरो की
        नियत सदस्यता।
  \item ग्रिड से एक मेगाजूल का दाम? और पेट्रोल से, जिसका एक लीटर \qty{34}{MJ}
        संचित रखता है और $1.80$ यूरो का आता है?
  \item अनाज का \qty{1.5}{MJ} वाला एक कटोरा लगभग $0.50$ यूरो का पड़ता है:
        उसका मेगाजूल-दाम? \omterm{def:g10:energy-conservation:energy}{जूल} बेचने वाले तीनों को क्रम में
        रखिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- छत।}
\begin{enumerate}[resume]
  \item भरी धूप प्रति वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} लगभग \qty{1000}{W} देती है; पैनल उसे
        $\eta = 0.20$ से रूपांतरित करते हैं: भरी धूप में \qty{1}{m^2} की विद्युत
        शक्ति?
  \item इस इलाक़े में औसतन प्रति दिन \qty{4.0}{h} भरी धूप के बराबर मिलता है:
        \qty{1}{m^2} की दैनिक उपज?
  \item परिवार के \qty{12.6}{kW.h} वाले दिन को ढकने के लिए कितना पैनल-क्षेत्रफल
        चाहिए? दक्षिण की छत \qty{20}{m^2} देती है: क्या वह पूरा पड़ेगा?
  \item दिसंबर में यह बराबरी घटकर \qty{1.5}{h} रह जाती है: पैनलों की उपज,
        ज़रूरत में उसका हिस्सा, और बची हुई कमी कौन भरेगा?
  \item पैनल बनाने में प्रति वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} लगभग \qty{700}{kW.h} लगे थे:
        प्रश्न 11 की सहायता से ऊर्जा की वापसी का समय?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- अनाज का कटोरा।}
\begin{enumerate}[resume]
  \item जनित्र वाली एक साइकिल \qty{100}{W} बिजली देती है: \qty{12.6}{kW.h} बनाने के
        लिए कितने घंटे पैडल मारने पड़ेंगे --- और \qty{24}{h} लगातार पैडल
        मारने वाले कितने लोग चाहिए?
  \item माँसपेशियाँ $\eta \approx 0.25$ पर चलती हैं: उन \qty{12.6}{kW.h} के पीछे की भोजन
        ऊर्जा मेगाजूलों में और अनाज के कटोरों में कितनी बैठती
        है?
  \item पैडल वाले दिन का ख़र्च (कटोरे $0.50$ यूरो के), और उसी दिन के
        लिए ग्रिड जो वसूलता है, उसकी तुलना?
  \item \qty{5}{GJ} की बिजली की एक कड़क कितने घर-दिनों के बराबर है? उसकी
        \qty{100}{\micro s} की अवधि देखते हुए वह घर को शक्ति फिर भी क्यों

  \item \omterm{def:g10:energy-conservation:energy}{जूल} कभी नष्ट नहीं होते, फिर भी ग्रिड, पेट्रोल और भोजन के
        \omterm{def:g10:energy-conservation:energy}{जूल} बहुत अलग-अलग दामों पर बिकते हैं: आप असल में किसका

  \item समापन: दिन का फ़ैसला एक वाक्य में बताइए --- एक
        \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पेट्रोल, धूप वाली छत के सोलह वर्ग

        दो राशियाँ बताइए (एक भंडार, एक बहाव) जिन्हें फिर कभी गड्डमड्ड
        नहीं करना है।
