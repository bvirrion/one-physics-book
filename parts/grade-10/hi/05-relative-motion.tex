\chapter{सापेक्ष गति}\label{ch:g10:relative-motion}

एक यात्री हाथ में कॉफ़ी लिए तेज़ रफ़्तार रेलगाड़ी के अगले सिरे की ओर
इत्मीनान से टहल रहा है। पटरी के किनारे खड़ा प्रेक्षक उसी यात्री को
\qty{300}{km/h} से भी अधिक पर आँकता है। दोनों मापें सही हैं। इनमें से कोई भी
यात्री की ``असली'' चाल नहीं है, क्योंकि ऐसी कोई चीज़ होती ही नहीं: गति
अकेले पिंड का गुण नहीं, बल्कि पिंड \emph{और} उसे बताने के लिए चुने गए
दृष्टिकोण का गुण है। यह अध्याय उस दृष्टिकोण को ठीक-ठीक परिभाषित करता है
--- और आपको अच्छा दृष्टिकोण चुनना सिखाता है।

\section{गति सापेक्ष होती है}

\begin{definition}[निर्देश तंत्र]\label{def:g10:relative-motion:frame}
\emph{निर्देश तंत्र}\index{निर्देश तंत्र} कोई दृढ़ पिंड है --- ज़मीन,
रेलगाड़ी का डिब्बा, या समूची पृथ्वी --- जिसके सापेक्ष स्थितियाँ दर्ज की
जाती हैं, और उनके साथ उन्हें समय देने के लिए एक घड़ी। कोई पिंड किसी तंत्र
में \emph{विराम}\index{विराम} में है जब उस तंत्र में उसकी स्थिति समय के
साथ नहीं बदलती; वरना वह उस तंत्र में गति में है।
\end{definition}

\begin{definition}[गतिपथ]\label{def:g10:relative-motion:trajectory}
किसी दिए गए \omterm{def:g10:relative-motion:frame}{निर्देश तंत्र} में किसी पिंड का
\emph{गतिपथ}\index{गतिपथ} उस तंत्र में उसकी लगातार स्थितियों से बना वक्र
है: कोई सरल रेखा, कोई वृत्त, कोई चाप\dots{} गतिपथ का अर्थ तभी बनता है जब
तंत्र का नाम ले लिया गया हो।
\end{definition}

\begin{example}[रेलगाड़ी में टहलता यात्री]\label{ex:g10:relative-motion:walker}
एक रेलगाड़ी ज़मीन के सापेक्ष \qty{300}{km/h} से चलती है, और उसी समय एक यात्री
रेलगाड़ी के सापेक्ष \qty{4}{km/h} से अगले सिरे की ओर चलता है।
\begin{itemize}
  \item \emph{रेलगाड़ी} के तंत्र में: यात्री गलियारे में \qty{4}{km/h} से चलता
        है; सीटें \omterm{def:g10:relative-motion:frame}{विराम} में हैं; बाहर के पेड़ पीछे की ओर \qty{300}{km/h}
        से भागते हैं।
  \item \emph{ज़मीन} के तंत्र में: सीटें \qty{300}{km/h} से चलती हैं, यात्री
        \qty{304}{km/h} से, और पेड़ \omterm{def:g10:relative-motion:frame}{विराम} में हैं।
\end{itemize}
वही दृश्य, दो तंत्र, दो बिलकुल अलग वर्णन --- और रेलगाड़ी के भीतर किया
गया कोई भी प्रयोग इनमें से किसी एक को ``असली'' घोषित नहीं कर सकता।
\end{example}

\begin{proposition}[गति की सापेक्षता]\label{prop:g10:relative-motion:relativity}
किसी पिंड की \omterm{def:g10:relative-motion:frame}{विराम} या गति की अवस्था, उसका \omterm{def:g10:relative-motion:trajectory}{गतिपथ} और उसकी
चाल --- ये सब उस \omterm{def:g10:relative-motion:frame}{निर्देश तंत्र} पर निर्भर करते हैं जो उन्हें
बताने के लिए चुना गया है।
\end{proposition}

\begin{proof}
हर दावे को एक-एक उदाहरण तय कर देता है। \cref{ex:g10:relative-motion:walker} का बैठा हुआ यात्री
रेलगाड़ी के तंत्र में \omterm{def:g10:relative-motion:frame}{विराम} में है और \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}
में \qty{300}{km/h} से चलता है (\omterm{def:g10:relative-motion:frame}{विराम} और चाल तंत्र पर निर्भर हैं)।
\omterm{def:g10:relative-motion:trajectory}{गतिपथ} के लिए साइकिल के पहिये का वाल्व देखिए: साइकिल पर सवार
प्रेक्षक उसे धुरी के चारों ओर वृत्त बनाते देखता है, जबकि सड़क पर खड़ा
प्रेक्षक उसे मेहराबों की एक कतार बनाते देखता है (इस वक्र को
\emph{चक्रज}\index{चक्रज} कहते हैं) --- नीचे का चित्र देखिए।
\end{proof}

\begin{remark}
ध्यान दीजिए कि दोनों चित्रों में क्या \emph{नहीं} बदलता: वाल्व,
पहिया, भौतिकी। केवल वर्णन बदलता है। ``क्या वाल्व सचमुच वृत्त में घूम रहा
है?'' पूछना वैसा ही है जैसे पूछना कि कोई पहाड़ ``सचमुच'' बाईं ओर है या
नहीं: यह इस पर निर्भर है कि आप कहाँ खड़े हैं।
\end{remark}

\section{औसत चाल}

\begin{definition}[औसत चाल]\label{def:g10:relative-motion:speed}
किसी दिए गए \omterm{def:g10:relative-motion:frame}{निर्देश तंत्र} में किसी पिंड की \emph{औसत
चाल}\index{चाल!औसत} उस तंत्र में तय की गई दूरी $d$ को यात्रा-समय
$t$ से भाग देकर मिलती है:
\[
v = \frac{d}{t} .
\]
$d$ को मीटरों में और $t$ को सेकंडों में लेने पर $v$
\omterm{def:g10:orders-of-magnitude:unit}{मीटर} प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} (\unit{m/s}) में आता है; रोज़मर्रा की ज़िंदगी
किलोमीटर प्रति घंटा (\unit{km/h}) पसंद करती है।
\end{definition}

\begin{method}[m/s और km/h के बीच बदलना]\label{met:g10:relative-motion:convert}
एक किलोमीटर \qty{1000}{m} होता है और एक घंटा \qty{3600}{s}, इसलिए $\qty{1}{m/s} = \qty{3.6}{km/h}$:
\begin{itemize}
  \item \unit{m/s} से \unit{km/h} में: $3.6$ से \emph{गुणा} कीजिए;
  \item \unit{km/h} से \unit{m/s} में: $3.6$ से \emph{भाग} दीजिए।
\end{itemize}
जाँच: \unit{km/h} वाली संख्या हमेशा दोनों में \emph{बड़ी} होती है।
\end{method}

\begin{example}[जानने लायक कुछ चालें]\label{ex:g10:relative-motion:convert}
तेज़ चाल से चलना: $\qty{1.5}{m/s} \times 3.6 = \qty{5.4}{km/h}$। शहर में \qty{54}{km/h} पर चलती कार: $54 / 3.6 = \qty{15}{m/s}$ --- यानी
हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} लगभग \qty{15}{m}, जिसे सड़क पर पैर रखने से पहले याद रखना चाहिए। वायु
में ध्वनि: $\qty{340}{m/s} \times 3.6 = \qty{1224}{km/h}$।
\end{example}

\begin{example}[बीच में विश्राम वाली एक पदयात्रा]\label{ex:g10:relative-motion:hike}
एक पदयात्री \qty{2}{h} चलकर \qty{12}{km} तय करता है, फिर \qty{1}{h} तक
विश्राम करता है, और फिर आख़िरी \qty{1.5}{h} में \qty{6}{km} तय करता है।
पूरी सैर पर \omterm{def:g10:relative-motion:speed}{औसत चाल}:
\[
v = \frac{d}{t} = \frac{12 + 6}{2 + 1 + 1.5}
= \frac{\qty{18}{km}}{\qty{4.5}{h}} = \qty{4.0}{km/h}.
\]
औसत बीता हुआ \emph{सारा} समय गिनता है, विश्राम भी --- और वह प्रायः
अलग-अलग चरणों की चालों का औसत नहीं होता।
\end{example}

\section{तंत्र चुनना: भू, भूकेंद्रीय, सूर्यकेंद्रीय}

चूँकि हर तंत्र वैध है, इसलिए हम वही चुनते हैं जिसमें हमारी दिलचस्पी वाली
गति सबसे सरल दिखे। इस पुस्तक की अधिकांश भौतिकी तीन चुनावों से चल जाती है।

\begin{definition}[तीन मानक तंत्र]\label{def:g10:relative-motion:frames}
\begin{itemize}
  \item \emph{भू-निर्देश तंत्र}\index{भू-निर्देश तंत्र} (या
        \emph{प्रयोगशाला तंत्र}\index{प्रयोगशाला तंत्र}) पृथ्वी की उसी
        सतह से जुड़ा है जहाँ प्रयोग हो रहा है। रोज़मर्रा की गति के लिए
        यही सही तंत्र है: गिरती वस्तुएँ, वाहन, खेल, प्रयोगशाला की मेज़ें।
  \item \emph{भूकेंद्रीय तंत्र}\index{भूकेंद्रीय तंत्र} पृथ्वी पर
        केंद्रित है पर उसके साथ घूमता \emph{नहीं} (उसकी दिशाएँ दूर के
        तारों की ओर सधी रहती हैं)। चंद्रमा और कृत्रिम उपग्रहों
        के लिए यही सही तंत्र है, क्योंकि वे समूची पृथ्वी की
        कक्षा में हैं और उसके दैनिक घूर्णन की परवाह नहीं करते।
  \item \emph{सूर्यकेंद्रीय तंत्र}\index{सूर्यकेंद्रीय तंत्र} सूर्य पर
        केंद्रित है, और उसकी दिशाएँ भी दूर के तारों पर सधी हैं। ग्रहों
        के लिए यही सही तंत्र है: तब हर ग्रह एक सरल, लगभग वृत्तीय
        कक्षा पर चलता है।
\end{itemize}
\end{definition}

\begin{method}[तंत्र चुनना]\label{met:g10:relative-motion:choose}
पहले उन पिंडों के नाम लीजिए जिनकी गति आपको बतानी है, फिर वह तंत्र चुनिए
जिसमें वह गति सबसे सरल हो:
\begin{enumerate}
  \item पृथ्वी की सतह के पास की, मिनटों भर चलने वाली गति ---
        \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र};
  \item पृथ्वी \emph{के चारों ओर} की कक्षाएँ (चंद्रमा,
        \omterm{def:g10:universal-gravitation:satellite}{उपग्रह}) --- \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र};
  \item सूर्य के चारों ओर की कक्षाएँ (ग्रह, धूमकेतु) ---
        \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र}।
\end{enumerate}
किसी भी गति को बताने से पहले अपना चुनाव बता दीजिए: तंत्र बताए बिना कहा
गया \omterm{def:g10:relative-motion:trajectory}{गतिपथ} या कही गई चाल निरर्थक है (\cref{prop:g10:relative-motion:relativity})।
\end{method}

\begin{remark}[कौन सही है, टॉलेमी या कोपरनिकस?]\label{rem:g10:relative-motion:history}
``सूर्य पूरब में उगता है'' और ``पृथ्वी पूरब की ओर घूमती है'' एक ही
प्रेक्षण की दो तंत्रों में दी गई रपटें हैं --- पहली \omterm{def:g10:relative-motion:frames}{भू-निर्देश
तंत्र} में, जहाँ सूर्य आकाश पार करता है, और दूसरी \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय
तंत्र} में, जहाँ प्रेक्षक पृथ्वी की धुरी के चारों ओर ले जाया जाता है।
सदियों तक ये तंत्र ही युद्धभूमि बने रहे: टॉलेमी का पृथ्वी-केंद्रित वर्णन
आकाश का ठीक-ठीक हिसाब रखता था, पर ग्रहों का पीछा करने के लिए उसे फंदों
पर फंदे चाहिए थे, जबकि 1543 में कोपरनिकस द्वारा फिर से खड़ा किया गया
सूर्य-केंद्रित वर्णन हर ग्रह को एक सरल अर्ध-वृत्ताकार पथ दे देता था और
मंगल के अजीब उलटे फंदों को महज़ तंत्र का असर बता देता था --- यानी पृथ्वी
का किसी धीमे ग्रह को पीछे छोड़ जाना। आधुनिक फ़ैसला यह नहीं है कि एक
तंत्र सच है और दूसरा झूठ, बल्कि यह कि दोनों बराबर \emph{सुविधाजनक} नहीं
हैं; कुछ तंत्रों (जड़त्वीय तंत्रों) को वरीयता देने की गहरी कसौटी \cref{ch:g10:inertia}
में जड़त्व के सिद्धांत के साथ आती है।
\end{remark}

\section{एक रेखा पर गतियों का संयोजन}

टहलते यात्री की \qty{4}{km/h} (रेलगाड़ी का तंत्र) को \qty{304}{km/h} (\omterm{def:g10:relative-motion:frames}{भू-निर्देश
तंत्र}) से क्या जोड़ता है? एक ही रेखा पर उत्तर सीधा जोड़ है।

\begin{proposition}[एक विमा में चालों का संयोजन]\label{prop:g10:relative-motion:compose}
मान लीजिए कोई पिंड किसी चलते आधार (चलपथ, रेलगाड़ी, बहता पानी) पर एक रेखा
के अनुदिश चलता है, और आधार स्वयं ज़मीन के सापेक्ष उसी रेखा के अनुदिश
सरकता है। रेखा पर एक धनात्मक दिशा चुनकर और चालों को चिह्न के साथ गिनकर:
\[
v_{\text{पिंड/ज़मीन}} \;=\; v_{\text{पिंड/आधार}} \;+\;
v_{\text{आधार/ज़मीन}} .
\]
शब्दों में: एक ही दिशा की चालें जुड़ती हैं; उलटी दिशाओं की चालें घटती
हैं।
\end{proposition}

\begin{proof}
एक \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में आधार पिंड को रेखा के अनुदिश $v_{\text{आधार/ज़मीन}}$ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} ले जाता है,
और पिंड आधार पर $v_{\text{पिंड/आधार}}$ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} आगे बढ़ता है। दोनों विस्थापन उसी एक
रेखा के अनुदिश होते हैं, इसलिए उस \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में ज़मीन के सापेक्ष पिंड का कुल
विस्थापन उनका बीजीय योग है --- और वही ज़मीन के सापेक्ष उसकी चाल है।
\end{proof}

\begin{example}[चलती पट्टी]\label{ex:g10:relative-motion:walkway}
हवाई अड्डे का चलपथ \qty{1.0}{m/s} से सरकता है। एक यात्री उस पर पट्टी के सापेक्ष
\qty{1.5}{m/s} से, चलने की दिशा में ही चलता है: ज़मीन के सापेक्ष चाल $1.5 + 1.0 = \qty{2.5}{m/s}$। उसी
रफ़्तार से उलटी दिशा में चलने पर: $1.5 - 1.0 = \qty{0.5}{m/s}$, यानी पट्टी के विरुद्ध भी वह
(धीरे-धीरे) आगे बढ़ता ही रहता है। पट्टी पर चुपचाप खड़े रहने पर: ज़मीन के
सापेक्ष \qty{1.0}{m/s}, पट्टी के सापेक्ष \qty{0}{m/s} --- यानी दो अलग तंत्रों में एक
ही समय \omterm{def:g10:relative-motion:frame}{विराम} में और गति में।
\end{example}

\begin{remark}[एक साथ दो दिशाएँ]\label{rem:g10:relative-motion:twodim}
जब दोनों गतियाँ एक ही रेखा के अनुदिश \emph{न} हों --- जैसे कोई नाव सीधे
नदी पार करने की कोशिश करे और धारा उसे बहाव की ओर धकेलती रहे --- तो हर
गति अपना काम स्वतंत्र रूप से करती है, और वेग तीरों की तरह नोक-से-पूँछ
जुड़ते हैं, ठीक वैसे ही जैसे गणित के खंड के सदिश। नाव फिर भी दूसरे किनारे
पर उतने ही समय में पहुँचती है जितना अकेली उसकी पार-चाल तय करती है, पर वह
अपने लक्ष्य-बिंदु से बहाव की ओर उतरती है, और भू-तंत्र में उसका
\omterm{def:g10:relative-motion:trajectory}{गतिपथ} एक तिरछी सरल रेखा होता है। पूरा सदिश उपचार आगे के किसी
अध्याय में आता है; यहाँ हम केवल चित्र पढ़ रहे हैं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:1}
एक यात्री \qty{100}{km/h} से चलती रेलगाड़ी में सो रहा है। बताइए कि वह इनके
सापेक्ष \omterm{def:g10:relative-motion:frame}{विराम} में है या गति में, और किस चाल से: डिब्बे के; ज़मीन
के; पटरी के किनारे खड़े पेड़ के; और दूसरी ओर से \qty{100}{km/h} पर आती रेलगाड़ी
के। फिर यात्री की रेलगाड़ी के तंत्र में पेड़ की गति बताइए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:2}
बदलिए: \qty{72}{km/h} और \qty{108}{km/h} को \unit{m/s} में; \qty{25}{m/s} और \qty{340}{m/s} को \unit{km/h}
में। हर उत्तर को \cref{met:g10:relative-motion:convert} के जाँच-नियम से परखिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:3}
एक साइकिल सवार \qty{1}{h}~\qty{30}{min} में \qty{36}{km} तय करता है; एक धावक \qty{10.0}{s}
में \qty{100}{m} तय करता है। दोनों की \omterm{def:g10:relative-motion:speed}{औसत चाल} \unit{m/s} और \unit{km/h} में
निकालिए। कौन तेज़ है, और क्या उत्तर आपको चौंकाता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:4}
एक चलपथ \qty{0.9}{m/s} से सरकता है। एक यात्री पट्टी के सापेक्ष \qty{1.4}{m/s} से चलता
है। पट्टी की दिशा में चलते हुए, उसके विरुद्ध चलते हुए, और पट्टी पर चुपचाप
खड़े रहते हुए ज़मीन के सापेक्ष यात्री की चाल बताइए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:5}
हर गति के लिए भू, भूकेंद्रीय और सूर्यकेंद्रीय में से सबसे सुविधाजनक
तंत्र चुनिए और एक वाक्य में उसे उचित ठहराइए: चलती रेलगाड़ी के भीतर गिराई
गई कलम (बोनस: इससे भी बेहतर एक तंत्र उपलब्ध है); अंतर्राष्ट्रीय अंतरिक्ष
स्टेशन की कक्षा; मंगल का एक वर्ष; \qty{100}{m} की दौड़।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:6}
एक रेलगाड़ी \qty{108}{km/h} से चलती है। एक यात्री रेलगाड़ी के सापेक्ष \qty{1.5}{m/s}
से अगले सिरे की ओर चलता है।
\begin{enumerate}
  \item रेलगाड़ी की चाल \unit{m/s} में बदलिए, फिर ज़मीन के सापेक्ष यात्री
        की चाल बताइए --- अगले सिरे की ओर चलते हुए और पिछले सिरे की ओर
        चलते हुए।
  \item डिब्बा \qty{60}{m} लंबा है। अगले सिरे तक की चहलक़दमी में कितना समय
        लगता है, और उस दौरान यात्री ज़मीन के सापेक्ष कितनी दूरी तय करता
        है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:7}
शांत पानी में एक नाव की चाल \qty{5.0}{m/s} है; नदी \qty{2.0}{m/s} से बहती है। नाव
बहाव की ओर \qty{420}{m} जाती है, फिर मुड़कर वापस आती है।
\begin{enumerate}
  \item हर चरण में किनारे के सापेक्ष नाव की चाल और हर चरण की अवधि
        निकालिए।
  \item पूरे चक्कर की \omterm{def:g10:relative-motion:speed}{औसत चाल} निकालिए। वह \qty{5.0}{m/s} से कम क्यों
        है, जबकि धारा एक चरण में ठीक उतनी ही मदद करती है जितनी दूसरे में
        अड़चन डालती है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:8}
दो रेलगाड़ियाँ समांतर पटरियों पर एक-दूसरे के पास से गुज़रती हैं:
रेलगाड़ी A (\qty{180}{m} लंबी) \qty{25}{m/s} से, रेलगाड़ी B (\qty{90}{m} लंबी) \qty{20}{m/s}
से।
\begin{enumerate}
  \item वे उलटी दिशाओं में चल रही हैं। A के तंत्र में B की चाल क्या है,
        और पूरा पार होना (नाकों के मिलने से पूँछों के अलग होने तक)
        कितनी देर चलता है?
  \item वही प्रश्न, यदि वे एक ही दिशा में चल रही हों।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:9}
एक साइकिल सीधी सड़क पर चल रही है। टायर के वाल्व का \omterm{def:g10:relative-motion:trajectory}{गतिपथ} इनमें
बताइए: साइकिल सवार का तंत्र; \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}; और स्वयं वाल्व
का तंत्र। \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में पहिये की धुरी का \omterm{def:g10:relative-motion:trajectory}{गतिपथ}
भी बताइए। कोई गणना नहीं --- बस हर वक्र का नाम दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:10}
पृथ्वी की भूमध्य रेखा लगभग \qty{40000}{km} लंबी है, और पृथ्वी \qty{24}{h} में एक बार
घूम जाती है।
\begin{enumerate}
  \item \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} में भूमध्य रेखा पर किसी बिंदु की चाल
        \unit{km/h} और \unit{m/s} में निकालिए।
  \item एक-एक वाक्य में समझाइए: हमें यह चाल महसूस क्यों नहीं होती, और
        ``सूर्य उगता है'' तथा ``पृथ्वी घूमती है'' एक ही प्रेक्षण की
        रपट क्यों हैं।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:11}
चलती सीढ़ी पर चुपचाप खड़े रहने पर ऊपर पहुँचने में \qty{60}{s} लगते हैं। उसी
सीढ़ी के बंद होने पर उस पर चलकर चढ़ने में \qty{90}{s} लगते हैं। चलती सीढ़ी
पर चलकर चढ़ने में कितना समय लगेगा? (समय के साथ नहीं, चालों के साथ काम
कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:12}
एक चालक वही दूरी दो बार तय करता है: जाते समय \qty{60}{km/h} से, लौटते समय
\qty{90}{km/h} से।
\begin{enumerate}
  \item दिखाइए कि पूरे चक्कर की \omterm{def:g10:relative-motion:speed}{औसत चाल} \qty{72}{km/h} है, \qty{75}{km/h}
        नहीं।
  \item समझाइए कि भोला उत्तर कहाँ चूकता है, और बताइए कि औसत पर कौन-सा
        चरण हावी रहता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:13}
दो साइकिल सवार \qty{35}{km} की दूरी से एक-दूसरे की ओर चलते हैं, एक \qty{15}{km/h}
से और दूसरा \qty{20}{km/h} से।
\begin{enumerate}
  \item पहले के तंत्र में दूसरे साइकिल सवार की गति बताइए, और इससे
        निकालिए कि वे कब मिलेंगे।
  \item मिलने तक हर एक ने कितनी दूरी तय की?
  \item यदि वे एक ही दिशा में चलें (तेज़ वाला पीछे), तो पीछा कितनी देर
        चलेगा?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:14}
एक तैराक \qty{60}{m} चौड़ी नदी पार करता है, पानी के सापेक्ष \qty{1.2}{m/s} से सीधे
पार की ओर तैरते हुए; धारा \qty{0.9}{m/s} से बहती है।
\begin{enumerate}
  \item समझाइए कि पार करने का समय अकेली तैराक की पार-चाल से क्यों तय
        होता है, और उसे निकालिए।
  \item तैराक बहाव की ओर कितनी दूर उतरता है?
  \item समकोण त्रिभुज की सहायता से \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में तैराक
        की चाल और ज़मीन के सापेक्ष सचमुच तैरे गए पथ की लंबाई निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:15}
पृथ्वी सूर्य की कक्षा में \qty{1.50e11}{m} त्रिज्या के लगभग वृत्त पर एक
वर्ष ($\num{3.16e7}$ \omterm{def:g10:orders-of-magnitude:unit}{सेकंड}) में चक्कर लगाती है।
\begin{enumerate}
  \item \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} में पृथ्वी की चाल \unit{km/s} में
        निकालिए।
  \item उसकी तुलना \cref{exo:g10:relative-motion:10} में निकाली गई भूमध्य रेखा की घूर्णन चाल से
        कीजिए।
  \item पृथ्वी से देखने पर मंगल कभी-कभी रुककर कुछ हफ़्तों तक तारों के
        बीच पीछे की ओर सरकता दिखता है। इस अध्याय की शब्दावली से दो
        वाक्यों में समझाइए कि \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} यह रहस्य कैसे
        घोल देता है।
\end{enumerate}
\end{exercise}

\section{समस्या: चलपथ, नदी और उपग्रह}

\begin{problem}[{सप्ताहांत समस्या --- चलता पथ, नदी और उपग्रह: कौन चल रहा
है, किसके सापेक्ष, और आरामकुर्सी पर बैठे यात्री की अंतिम चाल-रपट}]
\label{pb:g10:relative-motion:1}
एक हवाई अड्डा, एक नौका-यात्रा, और साँझ के आकाश में सरकता एक चमकीला बिंदु:
तीन रोज़मर्रा के दृश्य, और तीन बार पूछा गया एक ही सवाल --- कौन चल रहा है,
किसके सापेक्ष? हर भाग अपना तंत्र चुनता है, गणना करता है, और देखता है कि
दृष्टिकोण के साथ उत्तर कैसे बदल जाता है; और समापन जोड़ लगाता है कि
\emph{आप} इस समय, बिलकुल स्थिर बैठे हुए, कितनी तेज़ चल रहे हैं।

\medskip
\noindent\textbf{भाग I --- चलपथ की दौड़।}
हवाई अड्डे का एक चलपथ \qty{90}{m} लंबा है और \qty{1.0}{m/s} से सरकता है। एक यात्री
\qty{1.5}{m/s} से चलता है (पैरों तले जो भी हो, उसके सापेक्ष)।
\begin{enumerate}
  \item चुपचाप खड़े यात्री को एक सिरे से दूसरे सिरे तक ले जाने में चलपथ
        को कितना समय लगता है?
  \item चलती पट्टी पर, उसी दिशा में चलते हुए, यात्री को कितना समय लगता
        है?
  \item चलपथ के बगल में, स्थिर फ़र्श पर चलते हुए कितना समय लगता है?
  \item एक शरारती व्यक्ति पट्टी पर उलटी दिशा में, उसके सापेक्ष \qty{1.5}{m/s}
        से चलता है। ज़मीन के सापेक्ष उसकी चाल, और दूसरे सिरे से वापस
        प्रवेश तक पहुँचने का समय?
  \item एक बच्चा पट्टी के सापेक्ष ठीक \qty{1.0}{m/s} से उलटी दिशा में दौड़ता
        है। \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में बच्चे की गति बताइए, और उस
        व्यायामशाला की मशीन का नाम लीजिए जिसका यह पुनराविष्कार है।
  \item प्रश्न 1--5 को दो स्तंभों की एक सारणी में समेटिए: हर चलने वाले
        की चाल पट्टी के तंत्र में और \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में।
        यात्रियों की टाँगें कौन-सा स्तंभ महसूस करती हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- नदी।}
एक नौका \qty{120}{m} चौड़ी नदी पार करती है। पानी के सापेक्ष उसकी चाल \qty{2.0}{m/s}
है; धारा \qty{1.5}{m/s} से बहती है। चालक पहले सीधे सामने वाले किनारे की ओर
निशाना साधता है।
\begin{enumerate}[resume]
  \item पार करने में कितना समय लगता है?
  \item नौका बहाव की ओर कितनी दूर उतरती है?
  \item समकोण त्रिभुज की सहायता से \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में नौका की
        चाल और उसके भू-तंत्र वाले \omterm{def:g10:relative-motion:trajectory}{गतिपथ} की लंबाई निकालिए।
        (संख्याओं की एक जानी-पहचानी तिकड़ी उभर आती है।)
  \item पानी के तंत्र में और \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में नौका का
        \omterm{def:g10:relative-motion:trajectory}{गतिपथ} बताइए। दोनों में क्या एक जैसा है और क्या अलग?
  \item पानी के सापेक्ष \qty{1.0}{m/s} चाल वाला एक तैराक उसी किनारे से,
        नौका की तरह सीधे पार की ओर निशाना साधकर चल पड़ता है। पार करने
        में कितना समय लगता है, और तैराक बहाव की ओर कितनी दूर उतरता है?
        तैराक की भू-तंत्र चाल निकालिए (फिर वही समकोण त्रिभुज)। धीमा
        तैराक अधिक दूर क्यों बह जाता है?
  \item नौका के पास से एक लट्ठा बहता हुआ निकलता है। \omterm{def:g10:relative-motion:frames}{भू-निर्देश
        तंत्र} में और पानी के तंत्र में उसकी चाल बताइए। किस तंत्र में
        लट्ठा ``चल रहा'' है? टिप्पणी कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} का गुज़रना।}
साँझ के समय अंतर्राष्ट्रीय अंतरिक्ष स्टेशन कुछ ही मिनटों में आकाश पार कर
जाता है। वह \qty{400}{km} की ऊँचाई पर कक्षा में है; पृथ्वी की
त्रिज्या \qty{6.37e6}{m} है, और एक कक्षा में \qty{5.56e3}{s} (लगभग \qty{93}{min})
लगते हैं।
\begin{enumerate}[resume]
  \item किस मानक तंत्र में स्टेशन की गति सबसे सरल है? उसी तंत्र में
        उसकी चाल \unit{km/s} में निकालिए।
  \item दो वाक्यों में समझाइए कि \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} स्टेशन का
        वर्णन बेढंगे ढंग से क्यों करता है --- एक कक्षा से अगली
        कक्षा तक ज़मीन पर उसकी पटरी का क्या होता है?
  \item स्टेशन की कक्षीय चाल की तुलना भूमध्य रेखा की घूर्णन चाल
        ($\approx \qty{0.46}{km/s}$) से कीजिए: स्टेशन कितने गुना तेज़ है?
  \item एक भूस्थिर \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} \qty{4.22e7}{m} त्रिज्या पर ठीक एक दिन
        ($\qty{8.64e4}{s}$) में कक्षा पूरी करता है। \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय
        तंत्र} में उसकी चाल निकालिए, फिर \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में
        उसकी गति बताइए। एक वाक्य में दोनों उत्तरों में मेल बिठाइए।
  \item \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} में चंद्रमा लगभग $27$ दिनों में
        पृथ्वी के चारों ओर एक लगभग वृत्त बनाता है। कुछ महीनों में
        \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} में उसका \omterm{def:g10:relative-motion:trajectory}{गतिपथ} शब्दों में
        खींचिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- चल कौन रहा है?}
\begin{enumerate}[resume]
  \item ``सूर्य पूरब में उगता है।'' ``पृथ्वी पूरब की ओर घूमती है।''
        बताइए कि हर वाक्य किस तंत्र में प्रेक्षण का वर्णन करता है, और
        समझाइए कि कोई भी ग़लत क्यों नहीं है।
  \item एक संयत अनुच्छेद में: टॉलेमी के तंत्र ने क्या अच्छा किया,
        कोपरनिकस के तंत्र ने क्या सरल किया (मंगल के उलटे फंदों का
        ज़िक्र कीजिए), और आधुनिक फ़ैसला किसी तंत्र के सच होने के बजाय
        \emph{सुविधा} के बारे में क्यों है?
  \item आरामकुर्सी वाले यात्री की चाल-रपट। आप भूमध्य रेखा के पास एक
        आरामकुर्सी पर स्थिर बैठे हैं। \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में,
        \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} में ($\approx \qty{0.46}{km/s}$) और
        \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} में ($\approx \qty{30}{km/s}$) अपनी चाल बताइए। फिर
        \qty{90}{min} की एक फ़िल्म के दौरान \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} में
        आपके द्वारा तय की गई दूरी निकालिए, और अध्याय की एक वाक्य वाली
        सीख के साथ समस्या का समापन कीजिए।
\end{enumerate}
\end{problem}
