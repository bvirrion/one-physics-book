\chapter{सार्वत्रिक गुरुत्वाकर्षण और भार}\label{ch:g10:universal-gravitation}

एक सेब गिराइए: वह गिरता है। चंद्रमा, पृथ्वी की साठ त्रिज्या ऊपर, कभी
गिरता नहीं लगता। न्यूटन की बड़ी सूझ यही थी कि चंद्रमा गिर \emph{रहा} है
--- हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} लगभग 1.4 मिलीमीटर, ठीक उतना जितना दूरी के वर्ग के साथ
कमज़ोर पड़ता आकर्षण बताता है। एक ही सूत्र, एक ही सार्वत्रिक नियतांक के
साथ, सेब, चंद्रमा, ज्वार-भाटे और ग्रहों --- सब पर राज करता है। पहले खंड
ने \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण} का वर्णन शब्दों में किया था; यह अध्याय उसे नापता
है।

\section{सार्वत्रिक गुरुत्वाकर्षण का नियम}

\begin{definition}[गुरुत्वाकर्षण बल]\label{def:g10:universal-gravitation:force}
कोई भी दो पिंड एक-दूसरे को आकर्षित करते हैं, चाहे वे किसी भी चीज़ के बने
हों और कितनी भी दूर बैठे हों। यही आकर्षण \emph{गुरुत्वाकर्षण
बल}\index{गुरुत्वाकर्षण बल} है, और यह परिघटना
\emph{गुरुत्वाकर्षण}\index{गुरुत्वाकर्षण} कहलाती है। इसे न स्पर्श
चाहिए, न रस्सी, न कोई माध्यम, और यह \emph{परस्पर} है: हर पिंड दूसरे को
खींचता है।
\end{definition}

\begin{theorem}[सार्वत्रिक गुरुत्वाकर्षण का नियम]\label{thm:g10:universal-gravitation:law}
$m_1$ और $m_2$ द्रव्यमान (\unit{kg} में) वाले दो पिंड, जिनके केंद्र $d$
दूरी (\unit{m} में) पर हैं, एक-दूसरे को दो बलों से आकर्षित करते हैं जो
केंद्रों को मिलाने वाली रेखा के अनुदिश, हर एक दूसरे पिंड की ओर लगते हैं,
और जिनका परिमाण बराबर है:
\[
F = G\,\frac{m_1 m_2}{d^2}
\qquad \text{(\unit{N} में)},
\]
जहाँ $G$ ब्रह्मांड के हर युग्म के लिए एक ही नियतांक है।
\end{theorem}
\admitted

\begin{remark}[यह नियम कहाँ से आता है]
न्यूटन ने यह नियम ग्रहों की प्रेक्षित गति (केप्लर के नियमों) से निचोड़ा
था; ईमानदार व्युत्पत्ति स्नातक वर्ष 1 के खंड में की गई है। \Cref{ch:g12:satellites-kepler} इस
नियम को उपग्रहों और ग्रहों पर काम में लगा ही देगा।
\end{remark}

\begin{definition}[गुरुत्वाकर्षण नियतांक]\label{def:g10:universal-gravitation:constant}
नियतांक
\[
G = \qty{6.67e-11}{N.m^2/kg^2}
\]
\emph{गुरुत्वाकर्षण नियतांक}\index{गुरुत्वाकर्षण नियतांक} कहलाता है।
उसका नन्हा मान ही वह कारण है जिससे रोज़मर्रा की वस्तुओं के बीच का
गुरुत्व पकड़ में नहीं आता: $G m_1 m_2 / d^2$ तभी कुछ बनता है जब कम से कम एक
द्रव्यमान खगोलीय हो।
\end{definition}

\begin{remark}[सूत्र को पढ़ना]\label{rem:g10:universal-gravitation:reading}
चार बातें ध्यान माँगती हैं।
\begin{itemize}
  \item \emph{परस्पर और बराबर।} पृथ्वी आपको आपके भार से खींचती
        है --- और आप पृथ्वी को ठीक उतने ही बल से। (पृथ्वी बस कम
        प्रतिक्रिया करती है: वह लगभग $\num{e23}$ गुना भारी जो है।)
  \item \emph{व्युत्क्रम वर्ग।} $d$ दुगुना करने पर बल $4$ से भाग
        हो जाता है; दस गुना दूर जाने पर $100$ से।
  \item \emph{केंद्रों के बीच की दूरी।} गोलाकार पिंडों के लिए --- तारे,
        ग्रह, तोप के गोले --- $d$ केंद्र से केंद्र तक नापी जाती है,
        सतह से सतह तक नहीं।
  \item \emph{दोनों द्रव्यमान मायने रखते हैं।} किसी भी एक द्रव्यमान को
        दुगुना करने पर बल दुगुना हो जाता है; सूत्र $m_1$ और $m_2$ में
        सममित है।
\end{itemize}
\end{remark}

\begin{example}[दो मित्र]\label{ex:g10:universal-gravitation:friends}
\qty{70}{kg} के दो मित्र अपने केंद्रों को \qty{1.0}{m} की दूरी पर रखकर खड़े हैं:
\[
F = \num{6.67e-11} \times \frac{70 \times 70}{1.0^2}
\approx \qty{3.3e-7}{N}.
\]
हर मित्र का भार लगभग \qty{6.9e2}{N} है --- यानी दो \emph{अरब} गुना
बड़ा। उनका पारस्परिक आकर्षण असली है, पर महसूस होने की कोई उम्मीद नहीं।
\end{example}

\begin{example}[पृथ्वी चंद्रमा को थामे है]\label{ex:g10:universal-gravitation:earthmoon}
$m_1 = \qty{5.97e24}{kg}$ (पृथ्वी), $m_2 = \qty{7.35e22}{kg}$ (चंद्रमा) और $d = \qty{3.84e8}{m}$ लेकर:
\[
F = \num{6.67e-11} \times
\frac{\num{5.97e24} \times \num{7.35e22}}{(\num{3.84e8})^2}
\approx \qty{2.0e20}{N}.
\]
और चंद्रमा भी पृथ्वी को उसी \qty{2.0e20}{N} से वापस खींचता है --- वही खिंचाव
जिसका उत्तर महासागर दिन में दो बार ज्वार-भाटे से देते हैं।
\end{example}

\begin{method}[कोई गुरुत्वाकर्षण बल निकालना]\label{met:g10:universal-gravitation:compute}
\begin{enumerate}
  \item द्रव्यमानों को \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} में और केंद्र-से-केंद्र दूरी को
        \omterm{def:g10:orders-of-magnitude:unit}{मीटर} में बदलिए।
  \item $F = G m_1 m_2 / d^2$ लगाइए।
  \item दस की घात को अंकों से अलग जाँचिए, जैसा \cref{ch:g10:orders-of-magnitude} में किया था:
        \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण बल} $\num{e-7}$ न्यूटन (एक \omterm{def:g10:orders-of-magnitude:unit}{मीटर} की दूरी
        पर दो लोग) से $\num{e22}$ न्यूटन (पृथ्वी पर सूर्य) तक जाते हैं,
        इसलिए खिसका हुआ \omterm{def:g10:orders-of-magnitude:scinot}{घातांक} आसानी से पकड़ में आ जाता है।
\end{enumerate}
\end{method}

\section{भार और गुरुत्व की प्रबलता}

\begin{definition}[भार]\label{def:g10:universal-gravitation:weight}
किसी तारे (ग्रह, चंद्रमा, \dots) की सतह पर रखी वस्तु का
\emph{भार}\index{भार} $\vect P$ वह \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण बल} है जो तारा उस
पर लगाता है। वह तारे के केंद्र की ओर लगता है --- और इसी दिशा को हम
\emph{ऊर्ध्वाधर}\index{ऊर्ध्वाधर} कहते हैं --- और उसे न्यूटन में नापा
जाता है।
\end{definition}

\begin{proposition}[किसी तारे की सतह पर भार]\label{prop:g10:universal-gravitation:surfaceg}
$M$ द्रव्यमान और $R$ त्रिज्या वाले गोलाकार तारे की सतह पर $m$
द्रव्यमान की वस्तु का भार
\[
P = m\,g,
\qquad\text{जहाँ}\qquad
g = \frac{G M}{R^2}
\]
है, जो केवल तारे पर निर्भर करता है, वस्तु पर नहीं।
\end{proposition}

\begin{proof}
वस्तु का केंद्र और तारे का केंद्र $R$ दूरी पर हैं, इसलिए सार्वत्रिक
\omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण} का नियम (\cref{thm:g10:universal-gravitation:law}) $P = G M m / R^2 = m \times \left(G M / R^2\right)$ देता है। कोष्ठक में रखी
राशि सतह की हर वस्तु के लिए वही है: उसे $g$ कहिए।
\end{proof}

\begin{definition}[गुरुत्वीय क्षेत्र प्रबलता]\label{def:g10:universal-gravitation:fieldstrength}
राशि $g = G M / R^2$, जो न्यूटन प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} में मापी जाती है, तारे
की सतह पर \emph{गुरुत्वीय क्षेत्र प्रबलता}\index{गुरुत्वीय क्षेत्र
प्रबलता} है: तारा हर \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} को $g$ न्यूटन से खींचता है।
पृथ्वी पर $g = \qty{9.81}{N/kg}$।
\end{definition}

\begin{example}[$g$ निकालना]\label{ex:g10:universal-gravitation:gvalues}
पृथ्वी के लिए ($M = \qty{5.97e24}{kg}$, $R = \qty{6.37e6}{m}$):
\[
g = \frac{\num{6.67e-11} \times \num{5.97e24}}{(\num{6.37e6})^2}
= \frac{\num{3.98e14}}{\num{4.06e13}}
\approx \qty{9.81}{N/kg}.
\]
चंद्रमा के लिए ($M = \qty{7.35e22}{kg}$, $R = \qty{1.74e6}{m}$) वही गणना $g_{\text{चंद्रमा}} \approx \qty{1.62}{N/kg}$ देती है --- यानी छह
गुना कम: चंद्रमा कहीं हल्का है, और छोटा होना इसकी भरपाई नहीं करता। मंगल
के लिए $g_{\text{मंगल}} \approx \qty{3.73}{N/kg}$।
\end{example}

\begin{remark}[अंतरिक्ष यात्री उछलते क्यों हैं]\label{rem:g10:universal-gravitation:bounce}
\qty{120}{kg} द्रव्यमान (शरीर और परिधान) वाला अंतरिक्ष यात्री पृथ्वी पर $120 \times 9.81 \approx \qty{1180}{N}$
भारी है, पर चंद्रमा पर केवल $120 \times 1.62 \approx \qty{194}{N}$। \qty{1180}{N} के नीचे सधी माँसपेशियाँ
अचानक छठा हिस्सा भर ढोने लगती हैं: इसीलिए चंद्रमा पर वह मशहूर धीमी,
उछलती चाल दिखती है। द्रव्यमान --- और उसके साथ \emph{रुकने} या
\emph{मुड़ने} में लगने वाला ज़ोर --- वही रहता है, और इसीलिए चंद्रमा पर
अंतरिक्ष यात्री इतनी बार गिरते थे।
\end{remark}

\section{द्रव्यमान या भार? तुला और कमानीदार तराज़ू}

रोज़मर्रा की भाषा कहती है कि आटे की बोरी ``तीन किलो भारी है''। भौतिकी इस
वाक्य को दो हिस्सों में काट देती है।

\begin{remark}[द्रव्यमान भार नहीं है]\label{rem:g10:universal-gravitation:massweight}
\begin{itemize}
  \item \emph{द्रव्यमान} $m$ पदार्थ की मात्रा नापता है,
        \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} में। यह अकेली वस्तु का गुण है: पृथ्वी पर,
        चंद्रमा पर, या गहरे अंतरिक्ष में बहते हुए --- सब जगह वही।
  \item \emph{भार} $P = mg$ एक बल है, न्यूटन में, जो वस्तु पर
        \emph{किसी तारे द्वारा} लगाया जाता है। वह $g$ के साथ तारे-दर
        तारे बदलता है --- और हर तारे से दूर जाकर मिट जाता है।
\end{itemize}
इन्हें गड्डमड्ड करना बाज़ार में हानिरहित है और भौतिकी की समस्या में
घातक: \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} बल का मात्रक नहीं है।
\end{remark}

\begin{method}[तुला और कमानीदार तराज़ू]\label{met:g10:universal-gravitation:instruments}
\begin{itemize}
  \item \emph{दंड तुला} वस्तु की तुलना मानक द्रव्यमानों से करती है:
        दोनों पलड़ों के भार उसी $g$ से बढ़ते-घटते हैं,
        इसलिए यह तुलना \emph{द्रव्यमान} नापती है --- किसी भी तारे पर
        वही फैसला।
  \item \emph{कमानीदार तराज़ू} वह बल नापता है जो उसकी कमानी खींचता है:
        वह \emph{भार} पढ़ता है, न्यूटन में, और उसका पाठ
        तारे-दर-तारे बदलता है। द्रव्यमान वापस पाने के लिए स्थानीय
        क्षेत्र प्रबलता से भाग दीजिए: $m = P/g$।
\end{itemize}
\end{method}

\begin{example}[चंद्रमा पर आटा]\label{ex:g10:universal-gravitation:flour}
आटे की \qty{3.0}{kg} की बोरी पृथ्वी \emph{और} चंद्रमा, दोनों पर \qty{3.0}{kg} के
मानक द्रव्यमानों से संतुलित होती है। कमानीदार तराज़ू, इसके उलट, पृथ्वी
पर $3.0 \times 9.81 \approx \qty{29}{N}$ और चंद्रमा पर $3.0 \times 1.62 \approx \qty{4.9}{N}$ पढ़ता है। जो व्यापारी चंद्रमा पर
``न्यूटन के हिसाब से'' बेचे, वह छह गुना आटा थमा देगा --- व्यापार भी,
विज्ञान की तरह, द्रव्यमान पर चलता है।
\end{example}

\section{पृथ्वी के चारों ओर गिरना: न्यूटन की तोप}

एक पत्थर क्षैतिज फेंकिए: गुरुत्व उसका पथ एक चाप में मोड़ देता है और वह
कुछ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} दूर गिरता है। और तेज़ फेंकिए: वह और दूर गिरता है। न्यूटन का
विचार-प्रयोग इस विचार को उसकी हद तक ले जाता है। पृथ्वी गोल है, और हर
\qty{8}{km} चलने पर उसकी सतह क्षैतिज से लगभग \qty{5}{m} नीचे उतर जाती है;
स्वतंत्र रूप से गिरता पिंड अपने पहले \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में \qty{4.9}{m} गिरता है (यह
\cref{ch:g12:projectile-motion} स्थापित करेगा)। इसलिए पृथ्वी को \qty{8}{km/s} पर छूता हुआ जाता
प्रक्षेप्य ठीक उतनी ही तेज़ी से गिरता है जितनी तेज़ी से ज़मीन नीचे मुड़
जाती है: वह सदा गिरता रहता है और कभी उतरता नहीं।

\begin{definition}[उपग्रह और कक्षा]\label{def:g10:universal-gravitation:satellite}
किसी तारे का \emph{उपग्रह}\index{उपग्रह} वह पिंड है जो तारे की सतह तक
कभी पहुँचे बिना उसके चारों ओर गिरता रहता है; और जिस पथ को वह दोहराता है
वह उसकी \emph{कक्षा}\index{कक्षा} है।
\end{definition}

\begin{remark}[चंद्रमा एक गिरता पत्थर है]\label{rem:g10:universal-gravitation:fallingmoon}
चंद्रमा पृथ्वी का प्राकृतिक \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} है: वह हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} हमारी ओर
लगभग \qty{1.4}{mm} गिरता है, और उसकी बग़ल की गति उसे और पास आने से पहले ही
चारों ओर ले जाती है (\cref{exo:g10:universal-gravitation:15} से आप न्यूटन की अपनी जाँच दोहरा सकते हैं)।
किसी कक्षा में घूमते स्टेशन के अंतरिक्ष यात्री अपने यान जैसी ही मुक्त
गिरावट में हैं --- और यही, न कि गुरुत्व का अभाव, उनके तैरने का कारण है।
\omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण} हर प्रक्षेप-पथ को गढ़ता है, पत्थर के चाप से लेकर
ग्रह के दीर्घवृत्त तक; कक्षाओं की संख्यात्मक कहानी \cref{ch:g12:satellites-kepler} में
कही गई है।
\end{remark}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:5}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:6}
\qty{5.0e8}{kg} के दो लदे हुए महाजहाज़ अपने केंद्रों को \qty{100}{m} की दूरी पर रखकर
लंगर डाले हैं।
\begin{enumerate}
  \item उनका पारस्परिक \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण} निकालिए।
  \item पृथ्वी पर किस द्रव्यमान का भार इस बल के बराबर होगा?
  \item बल आश्चर्यजनक रूप से बड़ा है --- फिर भी जहाज़ों में एक-दूसरे की
        ओर सरकने का कोई रुझान नहीं दिखता। यह एक जहाज़ को जो त्वरण
        $a = F/m$ देता है उसे निकालिए और टिप्पणी कीजिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:7}

  \item एक ग्राहक ``\qty{3.0}{kg}'' का दाम चुकाता है। कौन-सा उपकरण किसी भी
        लोक पर सौदे को न्यायपूर्ण रखता है, और क्यों?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:8}
अंतर्राष्ट्रीय अंतरिक्ष स्टेशन \qty{400}{km} की ऊँचाई पर उड़ता है।
\begin{enumerate}
  \item उस ऊँचाई पर पृथ्वी की \omterm{def:g10:universal-gravitation:fieldstrength}{गुरुत्वीय क्षेत्र प्रबलता}

  \item उसे सतह वाले मान के प्रतिशत में बताइए।
  \item यान पर सवार अंतरिक्ष यात्री तैरते हैं। इसे अपने उत्तर से मेल
        बिठाइए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:9}
शुक्र की सतह पर $g = \qty{8.87}{N/kg}$ है, और ग्रह की त्रिज्या \qty{6.05e6}{m} है। इससे शुक्र
का द्रव्यमान निकालिए और उसकी तुलना पृथ्वी के द्रव्यमान से कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:10}
\begin{enumerate}
  \item समुद्र के \qty{1.0}{kg} पानी पर सूर्य द्वारा लगाया गया बल निकालिए,
        फिर उसी \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पर चंद्रमा द्वारा लगाया गया बल।
  \item सूर्य अधिक ज़ोर से खींचता है --- कितने गुना?
  \item फिर भी ज्वार-भाटे पर चंद्रमा का बोलबाला है। ज्वार खिंचाव से
        नहीं, बल्कि पृथ्वी के पास वाले और दूर वाले पहलुओं पर लगे
        खिंचावों के \emph{अंतर} से चलता है। बिना गणना किए समझाइए कि इस
        कसौटी पर पास का चंद्रमा दूर के सूर्य को कैसे हरा देता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:11}
\begin{enumerate}
  \item किसी ग्रह का द्रव्यमान पृथ्वी से दुगुना और त्रिज्या भी दुगुनी
        है। उसकी सतह की क्षेत्र प्रबलता पहले पृथ्वी के $g$ के पदों
        में बताइए, फिर \unit{N/kg} में।
  \item पृथ्वी जितने द्रव्यमान वाले ग्रह की त्रिज्या कितनी होनी चाहिए
        कि उसकी सतह की क्षेत्र प्रबलता $2g$ हो जाए?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:12}
पृथ्वी--चंद्रमा रेखा पर कहीं एक ऐसा बिंदु है जहाँ किसी अंतरिक्ष यान पर
दोनों खिंचाव कट जाते हैं। मान लीजिए पृथ्वी के केंद्र से उस बिंदु तक की
दूरी $x$ है और पृथ्वी--चंद्रमा दूरी $D$।
\begin{enumerate}
  \item दिखाइए कि उस बिंदु पर $\left(\dfrac{x}{D - x}\right)^{2}
        = \dfrac{M_{\text{पृथ्वी}}}{M_{\text{चंद्रमा}}}$।
  \item इससे $x$ निकालिए। यह चंद्रमा तक की यात्रा का कितना भाग है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:13}
न्यूटन की तोप, संख्याओं के साथ। एक प्रक्षेप्य पृथ्वी को क्षैतिज रूप से
$v$ चाल से छूता हुआ जाता है।
\begin{enumerate}
  \item $x = \qty{8.0}{km}$ क्षैतिज चलने के बाद गोलाकार पृथ्वी $h$ ऊँचाई भर
        ``नीचे हट'' जाती है, जो $(R + h)^2 = R^2 + x^2$ संतुष्ट करती है। छोटे $h$
        के लिए $(R + h)^2 \approx R^2 + 2Rh$ का उपयोग करते हुए दिखाइए कि $h \approx x^2 / (2R)$, और $h$
        निकालिए।
  \item स्वतंत्र रूप से गिरता पिंड अपने पहले \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में \qty{4.9}{m} गिरता
        है। उसकी $h$ से तुलना कीजिए और वह चाल निकालिए जिस पर
        प्रक्षेप्य कभी उतरता ही नहीं।
  \item सतह को छूते हुए उसी चाल पर एक पूरी कक्षा की अवधि
        निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:14}
एक न्यूट्रॉन तारा $M = \qty{2.8e30}{kg}$ (सूर्य के द्रव्यमान का 1.4 गुना) को \qty{12}{km}

\begin{enumerate}
  \item उसकी सतह पर \omterm{def:g10:universal-gravitation:fieldstrength}{गुरुत्वीय क्षेत्र प्रबलता} निकालिए।
  \item वहाँ \qty{70}{kg} का मनुष्य कितना भारी पड़ेगा? पृथ्वी पर उसके
        भार से तुलना कीजिए।
  \item वहाँ \qty{1.0}{mg} के रेत के एक कण का भार कितना होगा? पृथ्वी
        पर किस द्रव्यमान का भार इतना होता है?
\end{enumerate}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:15}
न्यूटन की चंद्रमा-परख (1666)। चंद्रमा की कक्षा की त्रिज्या
पृथ्वी की लगभग $60$ त्रिज्याएँ है।
\begin{enumerate}
  \item यदि गुरुत्व व्युत्क्रम वर्ग मानता है, तो प्रति
        \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पृथ्वी का खिंचाव चंद्रमा पर पृथ्वी की सतह की
        तुलना में कितने गुना कमज़ोर है?
  \item सतह के पास गिरता पिंड अपने पहले \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में \qty{4.9}{m} गिरता है।
        चंद्रमा को पृथ्वी की ओर हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कितना गिरना चाहिए?
  \item कक्षा से जाँचिए: चंद्रमा $d = \qty{3.84e8}{m}$ त्रिज्या का अपना
        वृत्त $T = 27.3$ दिनों में पूरा करता है। उसकी चाल $v$, एक
        \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में तय की गई दूरी $x$, और फिर गिरावट $h \approx x^2/(2d)$
        निकालिए (जैसा \cref{exo:g10:universal-gravitation:13} में किया था)।
  \item प्रश्न 2 और 3 के उत्तरों की तुलना कीजिए, और बताइए कि न्यूटन ने
        क्या निष्कर्ष निकाला।
\end{enumerate}
\end{exercise}

\section{समस्या: पृथ्वी को तौलना}

\begin{problem}[{सप्ताहांत समस्या --- एक छप्पर में रखे सीसे के दो गोलों
से पृथ्वी के, और फिर सूर्य के द्रव्यमान तक: एक छोटा-सा नियतांक, एक बार
नाप लेने पर, किस काम आता है}]
\label{pb:g10:universal-gravitation:1}
1798 में हेनरी कैवेंडिश ने एक पतले तार से एक हल्की छड़ लटकाई जिस पर सीसे
के दो छोटे गोले लगे थे, पास में सीसे के दो बड़े गोले लाए, और तार की
लगभग न के बराबर ऐंठन नाप ली। अख़बारों ने कहा कि उन्होंने \emph{पृथ्वी
तौल ली} --- और वे सही थे: $G$ पता चल जाने पर पृथ्वी का अपना $g$
और त्रिज्या उसका द्रव्यमान सौंप देते हैं, और पृथ्वी की सालाना
कक्षा सूर्य का। यह समस्या पूरी कड़ी दोहराती है। पूरे समय
आँकड़ा-पत्रक का उपयोग कीजिए।

\medskip
\noindent\textbf{भाग I --- सबसे कमज़ोर बल।}
\begin{enumerate}
  \item एफ़िल मीनार के लोहे का द्रव्यमान लगभग \qty{7.3e6}{kg} है। \qty{55}{kg} का एक
        पर्यटक उसके द्रव्यमान-केंद्र से \qty{100}{m} दूर खड़ा है। पर्यटक पर
        मीनार का खिंचाव निकालिए।
  \item उसकी तुलना \qty{1.0}{mg} के रेत के एक कण के भार से कीजिए।
  \item समानुपात का अभ्यास: \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण बल} का क्या होगा यदि
        (क) दूरी $10$ से गुणा कर दी जाए; (ख) दोनों द्रव्यमान $1000$
        से गुणा कर दिए जाएँ; (ग) दोनों द्रव्यमान \emph{और} दूरी, तीनों
        $1000$ से गुणा कर दिए जाएँ?
  \item $M$, $R$ और $G$ से, सतह पर रखी \qty{1.00}{kg} की वस्तु पर
        पृथ्वी का खिंचाव निकालिए। इस संख्या का रोज़मर्रा का नाम और
        संकेत क्या है?
  \item गुरुत्व आपकी जानी-पहचानी सभी अन्योन्यक्रियाओं में कहीं सबसे
        कमज़ोर है --- और फिर भी ब्रह्मांड इसी से चलता है। एक-दो वाक्यों
        में समझाइए कि पृथ्वी का खिंचाव आपके पड़ोसी के खिंचाव पर क्यों
        भारी पड़ता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- कैवेंडिश की तुला।}
मरोड़ तुला में $m_1 = \qty{158}{kg}$ द्रव्यमान का बड़ा गोला $m_2 = \qty{0.73}{kg}$ द्रव्यमान के छोटे
गोले को आकर्षित करता है; उनके केंद्र $d = \qty{0.23}{m}$ की दूरी पर हैं।
\begin{enumerate}[resume]
  \item $G$ के आधुनिक मान से दोनों गोलों के बीच का बल निकालिए।
  \item छोटे गोले का भार और दोनों बलों का अनुपात निकालिए।
        कैवेंडिश को किसी साधारण तराज़ू के बजाय मरोड़ तार की ज़रूरत
        क्यों पड़ी?
  \item कैवेंडिश की नापी हुई ऐंठन (आधुनिक मात्रकों में)
        $F = \qty{1.47e-7}{N}$ के बराबर बैठती है। इससे $G$ का उनका मान निकालिए और
        आज के मान से तुलना कीजिए।
  \item $G = \qty{6.67e-11}{N.m^2/kg^2}$, $g = \qty{9.81}{N/kg}$ और $R = \qty{6.37e6}{m}$ लेकर पृथ्वी का द्रव्यमान निकालिए।
        (यही वह चरण है जिस पर अख़बारों ने जश्न मनाया था।)
  \item $V = \frac43 \pi R^3$ लेकर पृथ्वी का औसत घनत्व $\rho = M / V$ निकालिए। सतह की
        चट्टानों का घनत्व लगभग \qty{2.7e3}{kg/m^3} होता है: यह तुलना गहरे भीतरी भाग
        के बारे में क्या खोलती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- और लोक।}
\begin{enumerate}[resume]
  \item आँकड़ा-पत्रक से चंद्रमा की सतह की क्षेत्र प्रबलता फिर से
        निकालिए।
  \item पृथ्वी पर एक अच्छी उछाल आपके द्रव्यमान-केंद्र को \qty{0.50}{m} ऊपर
        उठा देती है। उसी उछाल के लिए पहुँची गई ऊँचाई $g$ के
        व्युत्क्रमानुपाती होती है (जैसा ऊर्जा वाला अध्याय, \cref{ch:g10:energy-conservation},
        उचित ठहराएगा)। वही छलाँग चंद्रमा पर आपको कितना ऊपर ले जाएगी?
  \item बुध: $M = \qty{3.30e23}{kg}$, $R = \qty{2.44e6}{m}$। उसकी सतह की क्षेत्र प्रबलता निकालिए और
        मंगल से तुलना कीजिए। बुध मंगल से कहीं छोटा है --- फिर दोनों मान
        इतने पास कैसे हो सकते हैं?
  \item पृथ्वी की सतह से कितनी ऊँचाई पर क्षेत्र प्रबलता घटकर सतह वाले
        मान की आधी रह जाती है?
  \item पृथ्वी के केंद्र से कितनी दूरी पर पृथ्वी का प्रति
        \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} खिंचाव घटकर चंद्रमा की सतह वाले मान \qty{1.62}{N/kg}
        तक आ जाता है? इसे पृथ्वी की त्रिज्याओं में बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सूर्य को तौलना।}
पृथ्वी $d = \qty{1.496e11}{m}$ त्रिज्या की लगभग वृत्तीय कक्षा एक वर्ष, यानी
$T = \qty{3.156e7}{s}$, में पूरी करती है। दिया हुआ मानिए (इसकी उपपत्ति \cref{ch:g12:satellites-kepler} में है)
कि $d$ त्रिज्या के वृत्तीय पथ पर $v$ चाल से चलते पिंड को केंद्र
की ओर $v^2/d$ न्यूटन प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} का खिंचाव चाहिए।
\begin{enumerate}[resume]
  \item पृथ्वी की कक्षीय चाल $v$ निकालिए।
  \item इससे वह प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} खिंचाव निकालिए जो सूर्य पृथ्वी
        पर लगाता है।
  \item वह खिंचाव $G M_{\text{सूर्य}} / d^2$ भी है। इससे सूर्य का द्रव्यमान निकालिए।
  \item यह कितनी पृथ्वियों के बराबर है? $M_{\text{सूर्य}}$ की $M_{\text{पृथ्वी}}$ से तुलना
        कीजिए।
  \item समापन। कैवेंडिश के गोले, पृथ्वी, सूर्य: तीनों के लिए एक ही
        नियतांक $G$ काम आया। एक वाक्य में बताइए कि ``सार्वत्रिक
        \omterm{def:g10:universal-gravitation:force}{गुरुत्वाकर्षण}'' में पड़े शब्द \emph{सार्वत्रिक} ने इस
        समस्या में हमें क्या दिलाया।
\end{enumerate}
\end{problem}
