\chapter{प्रकाश का अपवर्तन}\label{ch:g10:refraction}

पानी के गिलास में एक तिनका डुबोइए: वह पानी की सतह पर टूटा हुआ दिखता है।
तरणताल हमेशा दिखने से गहरे होते हैं, हीरे रंगहीन पत्थर से रंगीन आग
कौंधाते हैं, और आप जो वाक्य पढ़ रहे हैं वह शायद बाल से भी पतले काँच के
धागे के भीतर एक महासागर पार करके आया है। चारों की व्याख्या एक ही
नियम करता है --- दो ज्याओं के बीच की एक बराबरी, जिसे स्नेल ने 1621 में
खोजा, तब जब पंद्रह सदियों से भौतिक विज्ञानी कोणों की सारणियाँ ताकते रहे
थे और पैटर्न देख नहीं पाए थे।

\section{परावर्तन: एक स्मरण}

समांगी पारदर्शी माध्यम में प्रकाश सीधी रेखाओं में चलता है --- \cref{ch:g10:light-spectra}
वाली किरणें। दो माध्यमों की सीमा पर प्रकाश का एक हिस्सा वापस उछल जाता
है (\emph{परावर्तन}) और एक हिस्सा पार चला जाता है
(\emph{\omterm{def:g10:refraction:refraction}{अपवर्तन}})। दोनों ठीक-ठीक ज्यामितीय नियम मानते हैं, जो एक
ही परिपाटी के साथ कहे जाते हैं: कोण हमेशा \emph{\omterm{def:g10:refraction:angles}{अभिलंब}} से नापे
जाते हैं, सतह से कभी नहीं।

\begin{definition}[अभिलंब और आपतन कोण]\label{def:g10:refraction:angles}
जिस बिंदु पर कोई किरण सतह से मिलती है, वहाँ सतह पर खींची गई लंब रेखा
\emph{अभिलंब}\index{अभिलंब} कहलाती है। आती हुई किरण और अभिलंब के बीच का
कोण \emph{आपतन कोण}\index{आपतन कोण} $i_1$ है; परावर्तन और
\omterm{def:g10:refraction:refraction}{अपवर्तन} के कोण भी इसी तरह अभिलंब से नापे जाते हैं।
\end{definition}

\begin{proposition}[परावर्तन का नियम]\label{prop:g10:refraction:reflection}
परावर्तित किरण उसी तल में होती है जिसमें आपतित किरण और \omterm{def:g10:refraction:angles}{अभिलंब}
हैं, \omterm{def:g10:refraction:angles}{अभिलंब} के दूसरी ओर, और परावर्तन कोण \omterm{def:g10:refraction:angles}{आपतन कोण} के
बराबर होता है:
\[
i_1' = i_1 .
\]
\end{proposition}
\admitted

\begin{example}[कोण सही ढंग से पढ़ना]\label{ex:g10:refraction:mirror}
लेज़र की एक किरण दर्पण पर ``$35^\circ$ पर'' पड़ती है --- यानी दर्पण की सतह
से नापकर, जैसे किरणों का वर्णन अक्सर किया जाता है। \omterm{def:g10:refraction:angles}{आपतन कोण}
$90^\circ - 35^\circ = 55^\circ$ है, इसलिए परावर्तित किरण भी \omterm{def:g10:refraction:angles}{अभिलंब} से $55^\circ$ पर निकलती
है, और किरण $180^\circ - 2 \times 55^\circ = 70^\circ$ मुड़ जाती है। \omterm{def:g10:refraction:angles}{अभिलंब} वाली परिपाटी भूल जाना
इस अध्याय की सबसे आम गलती है; हर बार उसकी क़ीमत $90^\circ$ पड़ती है।
\end{example}

\section{अपवर्तन और अपवर्तनांक}

आधे भरे गिलास में पेंसिल पकड़िए: वह सतह पर टूटी दिखती है। पानी के भीतर
वाले हिस्से से आता प्रकाश पानी छोड़ते समय दिशा बदल देता है, इसलिए आँख
--- जो किरणों को सीधा मानती है --- नोक को वहाँ गढ़ देती है जहाँ वह है
ही नहीं।

\begin{definition}[अपवर्तन]\label{def:g10:refraction:refraction}
\emph{अपवर्तन}\index{अपवर्तन} प्रकाश का वह दिशा-परिवर्तन है जो दो
पारदर्शी माध्यमों की सीमा पार करते समय होता है। दूसरे माध्यम की किरण
\emph{अपवर्तित किरण}\index{अपवर्तित किरण} है, और \omterm{def:g10:refraction:angles}{अभिलंब} से उसका
कोण अपवर्तन कोण $i_2$ है।
\end{definition}

\omterm{def:g10:refraction:refraction}{अपवर्तन} इसलिए होता है कि प्रकाश की चाल अलग-अलग माध्यमों में
अलग-अलग होती है; हर पदार्थ इसी से पहचाना जाता है कि वह प्रकाश को कितना
धीमा करता है।

\begin{definition}[अपवर्तनांक]\label{def:g10:refraction:index}
किसी पारदर्शी माध्यम का \emph{अपवर्तनांक}\index{अपवर्तनांक} यह अनुपात
है:
\[
n = \frac{c}{v},
\]
जहाँ $c = \qty{3.00e8}{m/s}$ निर्वात में प्रकाश की चाल है और $v$ माध्यम में उसकी
चाल। चूँकि $v \leq c$, इसलिए अपवर्तनांक $n \geq 1$ मानता है; इसका कोई
मात्रक नहीं होता। कुछ उपयोगी मान:
\begin{center}
\begin{tabular}{lccccc}
माध्यम & वायु & पानी & प्लेक्सिग्लास & काँच & हीरा \\
$n$ & $1.0003$ & $1.33$ & $1.49$ & $1.50$ & $2.42$ \\
\end{tabular}
\end{center}
\end{definition}

\begin{example}[पानी में प्रकाश की चाल]\label{ex:g10:refraction:speed}
पानी में $v = \dfrac{c}{n} = \dfrac{\qty{3.00e8}{m/s}}{1.33}
= \qty{2.26e8}{m/s}$। हीरे में प्रकाश $\qty{3.00e8}{m/s}/2.42 = \qty{1.24e8}{m/s}$ की चाल से रेंगता है --- यानी
निर्वात वाली चाल के आधे से भी कम। वायु के लिए $n = 1.0003$ चाल को केवल $0.03\%$
बदलता है: इस अध्याय में हम वायु को निर्वात मानेंगे और $n_{\text{वायु}} = 1.00$ लेंगे।
\end{example}

\begin{theorem}[स्नेल का अपवर्तन नियम]\label{thm:g10:refraction:snell}
\index{स्नेल का नियम}
\omterm{def:g10:refraction:refraction}{अपवर्तित किरण} आपतित किरण और \omterm{def:g10:refraction:angles}{अभिलंब} के तल में होती है,
\omterm{def:g10:refraction:angles}{अभिलंब} के दूसरी ओर, और आपतन तथा \omterm{def:g10:refraction:refraction}{अपवर्तन} के कोण यह संबंध
मानते हैं:
\[
n_1 \sin i_1 = n_2 \sin i_2 ,
\]
जहाँ $n_1$ और $n_2$ दोनों माध्यमों के \omterm{def:g10:refraction:index}{अपवर्तनांक} हैं। पथ प्रतिवर्ती
है: \omterm{def:g10:refraction:refraction}{अपवर्तित किरण} के साथ उलटी दिशा में चलता प्रकाश आपतित किरण
के रास्ते ही बाहर निकलता है।
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:refraction:honest}
परावर्तन और \omterm{def:g10:refraction:refraction}{अपवर्तन} के नियम यहाँ प्रायोगिक तथ्यों के रूप में
रखे गए हैं। उनकी ईमानदार व्युत्पत्ति स्नातक वर्ष 1 के खंड में है, जहाँ
प्रकाश को तरंग की तरह लिया जाता है: तब दोनों नियम --- और स्वयं सूत्र
$n = c/v$ --- सघन माध्यम में तरंग के धीमे पड़ने से निकल आते हैं।
\end{remark}

\begin{example}[वायु से पानी]\label{ex:g10:refraction:airwater}
एक किरण शांत तालाब पर $i_1 = 30^\circ$ पर पड़ती है। $n_1 = 1.00$ और $n_2 = 1.33$ लेकर:
\[
\sin i_2 = \frac{n_1 \sin i_1}{n_2} = \frac{\sin 30^\circ}{1.33}
= \frac{0.500}{1.33} = 0.376,
\qquad
i_2 = 22.1^\circ .
\]
किरण \omterm{def:g10:refraction:angles}{अभिलंब} की ओर मुड़ती है, जैसा धीमे माध्यम में घुसते समय वह
हमेशा करती है ($n_2 > n_1$)। पानी छोड़ते समय वही गणना उलटी चलती है और किरण
\omterm{def:g10:refraction:angles}{अभिलंब} से \emph{दूर} मुड़ती है।
\end{example}

\begin{method}[अपवर्तन की कोई समस्या हल करना]\label{met:g10:refraction:method}
\begin{enumerate}
  \item सतह, \omterm{def:g10:refraction:angles}{अभिलंब} और किरण खींचिए; $n_1$ (आपतित किरण का
        माध्यम) और $n_2$ पहचानिए।
  \item जाँचिए कि दिए गए कोण \omterm{def:g10:refraction:angles}{अभिलंब} से नापे गए हैं; सतह से नापे
        गए हों तो बदल लीजिए।
  \item $n_1 \sin i_1 = n_2 \sin i_2$ लिखिए और अज्ञात ज्या अलग कीजिए; कोण कैलकुलेटर की
        प्रतिलोम ज्या से निकालिए, जैसा गणित के खंड में है।
  \item जाँच: $n_2 > n_1$ हो तो किरण \omterm{def:g10:refraction:angles}{अभिलंब} की ओर मुड़ती है, $n_2 < n_1$
        हो तो उससे दूर, और $\sin i_2$ $0$ तथा $1$ के बीच आना चाहिए।
\end{enumerate}
\end{method}

नियम के पीछे का प्रयोग एक मेज़ पर समा जाता है: प्लेक्सिग्लास का आधा
चक्र, एक चाँदा, एक लेज़र। कच्चे कोण-युग्म $(i_1, i_2)$ बेतरतीब दिखते हैं ---
टॉलेमी ने लगभग सन 150 में उन्हें सारणीबद्ध किया और गलत निष्कर्ष निकाला
कि $i_2$, $i_1$ के समानुपाती है। इसके बजाय ज्या के सामने ज्या का
आलेख खींचिए, और आँकड़े मूल बिंदु से जाती $n_1/n_2$ ढाल वाली एक सीधी रेखा
में सीध जाते हैं: वही रेखा \omterm{thm:g10:refraction:snell}{स्नेल का नियम} \emph{है}।

\begin{proposition}[आभासी गहराई]\label{prop:g10:refraction:depth}
\index{आभासी गहराई}
पानी के नीचे $d$ गहराई पर रखी कोई वस्तु, ऊपर से छोटे कोणों पर देखने
पर, \emph{\omterm{prop:g10:refraction:depth}{आभासी गहराई}}
\[
d' = \frac{d}{n} ,
\]
पर दिखती है, जहाँ $n$ पानी का \omterm{def:g10:refraction:index}{अपवर्तनांक} है। \qty{1.20}{m} गहरा तरणताल
केवल $1.20/1.33 = \qty{0.90}{m}$ गहरा दिखता है --- और यही बेपरवाह छलाँगों का एक जाना-माना
कारण है।
\end{proposition}
\admitted

\begin{remark}\label{rem:g10:refraction:depthhonest}
यह सूत्र आँख तक पहुँचती लगभग ऊर्ध्वाधर किरणों पर \omterm{thm:g10:refraction:snell}{स्नेल का नियम}
लगाने से निकलता है; व्युत्पत्ति स्नातक वर्ष 1 के खंड में की गई है।
प्रभाव की दिशा के लिए कोई सूत्र चाहिए ही नहीं: पानी से निकलती किरणें
\omterm{def:g10:refraction:angles}{अभिलंब} से दूर मुड़ती हैं, इसलिए आँख उन्हें पीछे की ओर वस्तु से
ऊँचे किसी बिंदु तक ले जाती है।
\end{remark}

\section{वर्ण-विक्षेपण: हर रंग का अपना नियम}

प्रिज़्म श्वेत धूप को इंद्रधनुष बना देता है, जैसा \cref{ch:g10:light-spectra} के वर्णक्रमों
ने दिखाया था। \omterm{def:g10:refraction:refraction}{अपवर्तन} बताता है कि ऐसा \emph{क्यों} होता है:
किसी माध्यम का \omterm{def:g10:refraction:index}{अपवर्तनांक} हर रंग के लिए बिलकुल एक जैसा नहीं
होता।

\begin{definition}[वर्ण-विक्षेपण]\label{def:g10:refraction:dispersion}
कोई माध्यम \emph{विक्षेपी}\index{वर्ण-विक्षेपण} तब होता है जब उसका
\omterm{def:g10:refraction:index}{अपवर्तनांक} प्रकाश की \omterm{def:g10:light-spectra:wavelength}{तरंगदैर्घ्य} पर निर्भर करता है।
काँच और पानी में $n$ नीले प्रकाश (छोटी \omterm{def:g10:light-spectra:wavelength}{तरंगदैर्घ्य}) के लिए
लाल प्रकाश (लंबी \omterm{def:g10:light-spectra:wavelength}{तरंगदैर्घ्य}) की तुलना में थोड़ा बड़ा होता है
--- इसलिए नीला अधिक मुड़ता है।
\end{definition}

\begin{example}[दो रंग, एक सतह]\label{ex:g10:refraction:twocolors}
\omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} की किरण क्राउन काँच पर $i_1 = 60^\circ$ पर पड़ती है। इस
काँच के लिए $n_{\text{लाल}} = 1.51$ और $n_{\text{नीला}} = 1.53$। हर रंग के लिए एक बार \omterm{thm:g10:refraction:snell}{स्नेल का
नियम}:
\[
\sin i_{\text{लाल}} = \frac{\sin 60^\circ}{1.51} = 0.574,
\quad i_{\text{लाल}} = 35.0^\circ ;
\qquad
\sin i_{\text{नीला}} = \frac{\sin 60^\circ}{1.53} = 0.566,
\quad i_{\text{नीला}} = 34.5^\circ .
\]
एक ही सतह \omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} को आधे अंश भर बाँटती है --- खिड़की के
शीशे पर अदृश्य, पर प्रिज़्म एक ही घूर्णन दिशा में दो बार \omterm{def:g10:refraction:refraction}{अपवर्तन} करता
है, पंखा चौड़ा कर देता है, और कुछ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} दूर रंग सेंटीमीटरों की दूरी पर
आ जाते हैं।
\end{example}

\begin{remark}\label{rem:g10:refraction:rainbow}
इंद्रधनुष वही गणना है, बस बारिश की बूँद के भीतर चलती हुई: धूप बूँद में
घुसते समय अपवर्तित होती है, उसकी पिछली दीवार पर परावर्तित होती है, और
बाहर निकलते समय फिर अपवर्तित होती है। पानी का \omterm{def:g10:refraction:index}{अपवर्तनांक} $1.331$ (लाल)
से $1.343$ (नीला) तक जाता है, इसलिए हर रंग अपने ही कोण पर निकलता है
--- और कोई दो प्रेक्षक एक ही इंद्रधनुष नहीं देखते।
\end{remark}

\section{पूर्ण आंतरिक परावर्तन}

\omterm{thm:g10:refraction:snell}{स्नेल के नियम} में एक जाल छिपा है। पानी से वायु की ओर जाते समय
\omterm{def:g10:refraction:refraction}{अपवर्तित किरण} \omterm{def:g10:refraction:angles}{अभिलंब} से दूर मुड़ती है: $\sin i_2 = 1.33 \sin i_1 > \sin i_1$।
$i_1$ बढ़ाइए, और सूत्र जल्दी ही $\sin i_2 > 1$ माँगने लगता है, जो कोई कोण दे
ही नहीं सकता। तब \omterm{def:g10:refraction:refraction}{अपवर्तन} बस होना बंद हो जाता है।

\begin{definition}[क्रांतिक कोण और पूर्ण आंतरिक परावर्तन]\label{def:g10:refraction:critical}
मान लीजिए प्रकाश $n_1$ \omterm{def:g10:refraction:index}{अपवर्तनांक} वाले माध्यम में चलकर $n_2 < n_1$ छोटे
\omterm{def:g10:refraction:index}{अपवर्तनांक} वाले माध्यम की ओर जाता है। \emph{क्रांतिक
कोण}\index{क्रांतिक कोण} $i_c$ वह \omterm{def:g10:refraction:angles}{आपतन कोण} है जिस पर
\omterm{def:g10:refraction:refraction}{अपवर्तित किरण} सतह को छूती हुई निकलती है ($i_2 = 90^\circ$)। $i_1 > i_c$ के
लिए कोई \omterm{def:g10:refraction:refraction}{अपवर्तित किरण} होती ही नहीं: सतह सारा प्रकाश परावर्तन
के नियम के अनुसार पहले माध्यम में लौटा देती है। यही \emph{पूर्ण आंतरिक
परावर्तन}\index{पूर्ण आंतरिक परावर्तन} है।
\end{definition}

\begin{proposition}[क्रांतिक कोण का सूत्र]\label{prop:g10:refraction:criticalformula}
$n_1 > n_2$ के लिए,
\[
\sin i_c = \frac{n_2}{n_1} .
\]
\end{proposition}

\begin{proof}
\omterm{thm:g10:refraction:snell}{स्नेल के नियम} में $i_2 = 90^\circ$ रखिए: $n_1 \sin i_c = n_2 \sin 90^\circ
= n_2$। $i_1 > i_c$ के लिए
\omterm{thm:g10:refraction:snell}{स्नेल का नियम} $\sin i_2 = \frac{n_1}{n_2}\sin i_1 > 1$ माँगेगा: कोई अपवर्तित दिशा है ही नहीं।
(और तब प्रकाश \emph{पूरा का पूरा} परावर्तित होता है, अवशोषित नहीं ---
यह एक प्रायोगिक तथ्य है, जिसकी पुष्टि स्नातक वर्ष 1 के खंड का तरंग
सिद्धांत करता है।)
\end{proof}

वायु की ओर: पानी के लिए $\sin i_c = 1/1.33 = 0.752$, इसलिए $i_c = 48.8^\circ$; काँच के लिए $\sin i_c = 1/1.50$,
इसलिए $i_c = 41.8^\circ$; हीरे के लिए $\sin i_c = 1/2.42 = 0.413$, इसलिए $i_c = 24.4^\circ$। \omterm{def:g10:refraction:index}{अपवर्तनांक} जितना
बड़ा, निकास-शंकु उतना सँकरा: हीरे में किसी फलक के \omterm{def:g10:refraction:angles}{अभिलंब} से
$24.4^\circ$ से अधिक हटी हर किरण फँस जाती है --- यही उसकी झिलमिल का रहस्य
है, जिसकी चीर-फाड़ \cref{pb:g10:refraction:1} में की गई है।
\end{example}

\section{प्रकाशिक तंतु}

\omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन} काँच के धागे को प्रकाश की नली बना देता
है: अक्ष के लगभग समांतर घुसती किरण दीवार से \omterm{def:g10:refraction:critical}{क्रांतिक कोण} से
कहीं आगे टकराती है, बिना हानि के परावर्तित होती है, और प्रति किलोमीटर
लाखों बार ऐसा करती है, बिना रिसे।

\begin{definition}[प्रकाशिक तंतु]\label{def:g10:refraction:fiber}
\emph{प्रकाशिक तंतु}\index{प्रकाशिक तंतु} काँच का एक पतला धागा है, जो
$n_1$ \omterm{def:g10:refraction:index}{अपवर्तनांक} वाले \emph{क्रोड}\index{क्रोड} से बना होता है और उस
पर थोड़े छोटे \omterm{def:g10:refraction:index}{अपवर्तनांक} $n_2 < n_1$ का \emph{आवरण}\index{आवरण} लिपटा होता
है। एक सिरे से डाला गया प्रकाश क्रोड--आवरण सीमा पर बार-बार
\omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन} करता हुआ क्रोड के साथ-साथ चलता है।
\end{definition}

साथ-साथ ले जाया जाता प्रकाश (उछाल के कोण बहुत बढ़ाकर दिखाए गए हैं: असली
किरणें दीवार को \omterm{def:g10:refraction:angles}{अभिलंब} से $80^\circ$ से अधिक पर छूती हुई निकलती
हैं)।}
\end{omfigure}

\begin{example}[संख्याओं में एक तंतु]\label{ex:g10:refraction:fibernumbers}
$n_1 = 1.48$ और $n_2 = 1.46$ लेकर,
\[
\sin i_c = \frac{1.46}{1.48} = 0.986,
\qquad i_c = 80.6^\circ :
\]
यानी केवल अक्ष से लगभग $9^\circ$ के भीतर की किरणें ही ले जाई जाती हैं ---
जो ठीक भी है, क्योंकि लेज़र प्रकाश ठीक इसी तरह डाला जाता है।
\omterm{def:g10:refraction:fiber}{क्रोड} के भीतर प्रकाश $v = c/1.48 = \qty{2.03e8}{m/s}$ की चाल से चलता है: एक स्पंद
\qty{100}{km} लंबे तंतु को लगभग आधे मिलीसेकंड में पार कर लेता है।
\end{example}

\begin{remark}[आँकड़ा-कड़ियों की परिमाण की कोटि]\label{rem:g10:refraction:datarates}
समुद्री केबल --- समुद्र-तल पर बिछे कुछ दर्जन तंतुओं के गट्ठर --- लगभग
सारा अंतरमहाद्वीपीय यातायात ढोते हैं। एक अकेला आधुनिक तंतु एक साथ कई
\omterm{def:g10:light-spectra:wavelength}{तरंगदैर्घ्य} भेजकर लगभग $10^{13}$ बिट प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} ले जाता है; पूरी
केबल $10^{14}$ बिट प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} तक पहुँचती है, और हर \qty{80}{km} या उसके आसपास
एक प्रवर्धक लगता है। तुलना में उपग्रह इस यातायात का एक प्रतिशत का छोटा
सा अंश ही सँभालते हैं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:refraction:1}
लेज़र की एक किरण समतल दर्पण पर \emph{दर्पण की सतह से} $25^\circ$ पर पड़ती
है। \omterm{def:g10:refraction:angles}{आपतन कोण}, परावर्तन कोण, और आपतित तथा परावर्तित किरणों के
बीच का कोण बताइए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:2}
\begin{enumerate}
  \item पानी ($n = 1.33$) और हीरे ($n = 2.42$) में प्रकाश की चाल निकालिए।
  \item किसी ठोस में प्रकाश \qty{1.97e8}{m/s} की चाल से चलता है। उसका
        \omterm{def:g10:refraction:index}{अपवर्तनांक} निकालिए और पदार्थ पहचानिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:3}
वायु में चलती एक किरण काँच के गुटके ($n = 1.50$) पर $i_1 = 40^\circ$ पर पड़ती है।
\omterm{def:g10:refraction:refraction}{अपवर्तन} कोण निकालिए। काँच के भीतर चलती किरण को $40^\circ$ पर वायु
में निकलने के लिए सतह से किस कोण पर टकराना होगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:4}
धूप झील की सतह पर ऊर्ध्वाधर से $55^\circ$ पर पहुँचती है।
\omterm{def:g10:refraction:refraction}{अपवर्तित किरण} का कोण निकालिए, और बताइए कि किरण \omterm{def:g10:refraction:angles}{अभिलंब}
की ओर मुड़ती है या उससे दूर --- एक ही शब्द के कारण के साथ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:5}
किसी रंगहीन द्रव को पहचानने के लिए उसकी सतह पर $i_1 = 45^\circ$ पर एक किरण भेजी
जाती है; \omterm{def:g10:refraction:refraction}{अपवर्तित किरण} $i_2 = 32^\circ$ पर नापी जाती है।
\omterm{def:g10:refraction:index}{अपवर्तनांक} निकालिए और \cref{def:g10:refraction:index} की सारणी से द्रव पहचानिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:6}
पारदर्शी प्लास्टिक के एक अर्धचक्र से ये मापें मिलती हैं:
\begin{center}
\begin{tabular}{lccccc}
$i_1$ & $20^\circ$ & $30^\circ$ & $40^\circ$ & $50^\circ$ & $60^\circ$ \\
$i_2$ & $13.5^\circ$ & $19.5^\circ$ & $25.5^\circ$ & $31.0^\circ$
      & $35.5^\circ$ \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item हर युग्म के लिए $\dfrac{\sin i_1}{\sin i_2}$ निकालिए। आप क्या देखते हैं?
  \item इससे प्लास्टिक का \omterm{def:g10:refraction:index}{अपवर्तनांक} निकालिए और उसे पहचानिए।
  \item $i_2$ के सामने $i_1$ का आलेख खींचने की तुलना में $\sin i_1$
        के सामने $\sin i_2$ का आलेख खींचना \omterm{thm:g10:refraction:snell}{स्नेल के नियम} की बेहतर
        परख क्यों है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:7}
पानी, काँच ($n = 1.50$) और हीरे के लिए वायु की ओर \omterm{def:g10:refraction:critical}{क्रांतिक कोण}
निकालिए। फिर \omterm{thm:g10:refraction:snell}{स्नेल के नियम} से समझाइए कि वायु \emph{से} पानी
\emph{में} जाते प्रकाश के लिए कोई \omterm{def:g10:refraction:critical}{क्रांतिक कोण} क्यों नहीं होता।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:8}
\begin{enumerate}
  \item एक तरणताल \qty{2.0}{m} गहरा है। किनारे पर खड़े तैराक को कितनी गहराई
        दिखती है?
  \item एक बगुला सतह से \qty{40}{cm} नीचे तैरती मछली पर नज़र गड़ाए है। मछली
        किस गहराई पर दिखती है, और बगुले को अपने देखे हुए प्रतिबिंब से
        ऊपर, उसी पर, या नीचे चोंच मारनी चाहिए?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:9}
एक किरण खिड़की के शीशे ($n = 1.50$, समांतर फलक) पर $50^\circ$ पर पड़ती है।
\begin{enumerate}
  \item पहले फलक पर \omterm{def:g10:refraction:refraction}{अपवर्तन} का कोण निकालिए।
  \item \omterm{def:g10:refraction:refraction}{अपवर्तित किरण} दूसरे फलक से भीतर से, उसी कोण पर टकराती
        है। निर्गत कोण निकालिए और उसकी $50^\circ$ से तुलना कीजिए।
  \item तो फिर मोटी खिड़की किसी पार करती किरण की दिशा और स्थिति के साथ
        क्या करती है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:10}
\omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} फ्लिंट काँच के गुटके पर $55^\circ$ पर पड़ता है। इस
काँच के लिए $n_{\text{लाल}} = 1.61$ और $n_{\text{नीला}} = 1.66$। लाल और नीली किरणों के \omterm{def:g10:refraction:refraction}{अपवर्तन}
कोण तथा उनके बीच का कोण निकालिए। कौन-सा रंग \omterm{def:g10:refraction:angles}{अभिलंब} के अधिक पास
रह जाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:11}
किसी \omterm{def:g10:refraction:fiber}{प्रकाशिक तंतु} में $1.48$ \omterm{def:g10:refraction:index}{अपवर्तनांक} का \omterm{def:g10:refraction:fiber}{क्रोड} और
$1.46$ \omterm{def:g10:refraction:index}{अपवर्तनांक} का \omterm{def:g10:refraction:fiber}{आवरण} है।
\begin{enumerate}
  \item क्रोड--आवरण सीमा पर \omterm{def:g10:refraction:critical}{क्रांतिक कोण} निकालिए।
  \item कोई निर्देशित किरण ठीक \omterm{def:g10:refraction:critical}{क्रांतिक कोण} पर उछलती हुई चलती
        है। दिखाइए कि \qty{1.0}{km} सीधे तंतु में वह अक्ष के सीधे नीचे जाती
        किरण से लगभग \qty{14}{m} अधिक चलती है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:12}
एक गोताखोर की आँख बिलकुल शांत सतह से \qty{2.0}{m} नीचे है।
\omterm{def:g10:refraction:refraction}{अपवर्तन} के कारण गोताखोर को \emph{पूरा} आकाश ठीक सिर के ऊपर एक
चमकीली चकती में सिमटा दिखता है --- यही स्नेल की खिड़की है।
\begin{enumerate}
  \item समझाइए कि क्षितिज से आता प्रकाश गोताखोर की आँख तक पानी के
        \omterm{def:g10:refraction:critical}{क्रांतिक कोण} पर, यानी ऊर्ध्वाधर से $48.8^\circ$ पर, क्यों
        पहुँचता है।
  \item आँख--ऊर्ध्वाधर--चकती-किनारा वाले समकोण त्रिभुज की स्पर्शज्या का
        उपयोग करते हुए सतह पर चमकीली चकती की त्रिज्या निकालिए।
  \item चकती के \emph{बाहर} सतह पर गोताखोर को क्या दिखता है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:13}
एक प्रिज़्म के कोण $30^\circ$, $60^\circ$, $90^\circ$ हैं।
\omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} की एक पतली किरण समकोण से लगी किसी एक भुजा पर लंबवत
घुसती है, बिना विचलित हुए पार करती है, और लंबे फलक से भीतर से $i_1 = 30^\circ$
पर टकराती है। इस काँच के लिए $n_{\text{लाल}} = 1.51$ और $n_{\text{नीला}} = 1.53$।
\begin{enumerate}
  \item लाल और नीली किरणों के निर्गत कोण और उनके बीच पंखे की कोणीय
        चौड़ाई निकालिए।
  \item प्रिज़्म की जगह $45^\circ$--$45^\circ$--$90^\circ$ वाला प्रिज़्म रख दिया
        जाता है, जिससे भीतर का \omterm{def:g10:refraction:angles}{आपतन कोण} $45^\circ$ हो जाता है।
        दिखाइए कि लंबे फलक से कोई प्रकाश बाहर निकलता ही नहीं।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:14}
पेरिस्कोप प्रकाश को $45^\circ$--$45^\circ$--$90^\circ$ वाले काँच के प्रिज़्मों
($n = 1.50$) से मोड़ते हैं: किरण एक छोटे फलक में लंबवत घुसती है और लंबे
फलक से भीतर से $45^\circ$ पर टकराती है।
\begin{enumerate}
  \item दिखाइए कि लंबा फलक एक संपूर्ण दर्पण की तरह काम करता है। यह उस
        धात्विक दर्पण से बेहतर क्यों है जो हर परावर्तन पर कुछ प्रतिशत
        अवशोषित कर लेता है?
  \item पेरिस्कोप में पानी भर जाता है और लंबा फलक अब पानी के संपर्क में
        है। काँच--पानी सीमा का नया \omterm{def:g10:refraction:critical}{क्रांतिक कोण} निकालिए और
        दिखाइए कि दर्पण काम करना बंद कर देता है।
  \item प्रश्न 2 का प्रकाश प्रिज़्म से पानी में किस कोण पर निकलता है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:15}
एक अंतर-अटलांटिक केबल लंदन और न्यूयॉर्क के बीच \qty{6200}{km} तंतु
(\omterm{def:g10:refraction:fiber}{क्रोड} का \omterm{def:g10:refraction:index}{अपवर्तनांक} $1.47$) बिछाती है।
\begin{enumerate}
  \item \omterm{def:g10:refraction:fiber}{क्रोड} में प्रकाश की चाल और एक स्पंद का एकतरफ़ा
        यात्रा-समय निकालिए।
  \item कोई रेडियो तरंग $c$ की चाल से चलती है। उसी दूरी पर उसे
        कितना समय लगता, और रेडियो की तुलना में तंतु का काँच आने-जाने के
        कुल समय में कितना जोड़ देता है?
  \item कुछ वित्तीय कंपनियाँ इस कड़ी से कुछ मिलीसेकंड कम कराने के लिए
        भारी रकम देती हैं। प्रकाश को जल्दी पहुँचाने के दो अलग-अलग
        भौतिक उपाय सुझाइए।
\end{enumerate}
\end{exercise}

\section{समस्या: हीरा, तरणताल और तंतु}

\begin{problem}[{सप्ताहांत समस्या --- स्नेल के नियम के तीन जीवन: एक
तरणताल जो अपनी गहराई के बारे में झूठ बोलता है, एक पत्थर जो ऐसे तराशा
गया है कि प्रकाश उससे निकल ही न सके, और इंटरनेट के नीचे बहता साठ
मिलीसेकंड का महासागर}]
\label{pb:g10:refraction:1}
दो ज्याओं की एक बराबरी, और तीन पेशे। तरणताल में वह आपकी आँखों को गहराई
के बारे में धोखा देती है और पूरे आकाश को एक चकती में निचोड़ देती है।
हीरे में, अपनी हद तक धकेली जाकर, वह प्रकाश को बाहर निकलने से मना कर देती
है --- और जौहरी सदियों से इसी मनाही के इर्द-गिर्द पत्थर तराशते आए हैं।
अटलांटिक के नीचे वह काँच की एक लट के भीतर लेज़र स्पंदों को राह दिखाती है
और वैश्विक इंटरनेट की गति-सीमा तय करती है। यह समस्या तीनों जीवन क्रम से
देखती है, हर बार उसी एक नियम के साथ।

\medskip
\noindent\textbf{भाग I --- वार्म-अप।}
\begin{enumerate}
  \item पानी ($n = 1.33$) और हीरे ($n = 2.42$) में प्रकाश की चाल निकालिए।
        हीरा प्रकाश को कितने गुना धीमा करता है?
  \item वायु में चलती एक किरण तालाब पर $i_1 = 40^\circ$ पर पड़ती है।
        \omterm{def:g10:refraction:refraction}{अपवर्तन} कोण निकालिए।
  \item पानी में चलती एक किरण सतह से नीचे से $28.9^\circ$ पर टकराती है। वह
        वायु में किस कोण पर निकलती है? यह किस सिद्धांत को दिखाता है?
  \item पानी--वायु सीमा का \omterm{def:g10:refraction:critical}{क्रांतिक कोण} निकालिए, और बताइए कि
        सतह से $60^\circ$ पर टकराती पानी के भीतर की किरण का क्या होता है।
  \item वायु का \omterm{def:g10:refraction:index}{अपवर्तनांक} $1.0003$ है। वायु में प्रकाश की चाल निकालिए
        और पूरी पुस्तक में इस्तेमाल हुई सन्निकटन $n_{\text{वायु}} = 1.00$ को उचित
        ठहराइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तरणताल।}
\begin{enumerate}[resume]
  \item एक तरणताल \qty{3.0}{m} गहरा है। \omterm{prop:g10:refraction:depth}{आभासी गहराई} के सूत्र (\cref{prop:g10:refraction:depth}) से
        निकालिए कि ऊपर खड़े व्यक्ति को कितनी गहराई दिखती है।
  \item शब्दों में बताई गई दो-किरण वाली रेखाचित्र-कल्पना से समझाइए कि
        तल पर पड़ा सिक्का ऊपर उठा हुआ क्यों लगता है: पानी से निकलती
        किरणें क्या करती हैं, और आँख उनका कटान कहाँ रखती है?
  \item एक गोताखोर की आँख \qty{3.0}{m} पर है। स्नेल की खिड़की --- वह चमकीली
        चकती जिससे गोताखोर पूरा आकाश देखता है --- की त्रिज्या निकालिए
        (\cref{exo:g10:refraction:12})।
  \item तल पर रखा एक लैंप ऊर्ध्वाधर से $30^\circ$, $45^\circ$ और $50^\circ$ पर
        ऊपर की ओर किरणें भेजता है। हर किरण के लिए बताइए कि वह निकल
        पाती है या नहीं और किस कोण पर, या उसके बदले क्या होता है।
  \item किनारे पर खड़ा एक भालेवाला मछुआरा मछली पर निशाना साधता है। वार
        मछली के प्रतिबिंब से ऊपर करना चाहिए, उसी पर, या नीचे --- और
        क्या मछली को भी, पलटकर देखने पर, मछुआरा खिसका हुआ दिखता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- हीरा।}
\begin{enumerate}[resume]
  \item हीरे ($n = 2.42$) का वायु की ओर \omterm{def:g10:refraction:critical}{क्रांतिक कोण} निकालिए और
        उसकी तुलना काँच ($n = 1.52$) के $41.1^\circ$ से कीजिए।
  \item पत्थर के भीतर की एक किरण किसी फलक से उसके \omterm{def:g10:refraction:angles}{अभिलंब} से
        $30^\circ$ पर टकराती है। हीरे में क्या होता है? काँच की नकल में?
        एक वाक्य में समझाइए कि हीरे की तराश, उसके नन्हे
        \omterm{def:g10:refraction:critical}{क्रांतिक कोण} के साथ मिलकर, झिलमिल क्यों पैदा करती है।
  \item हीरा प्रबलता से \omterm{def:g10:refraction:dispersion}{विक्षेपी} भी है: $n_{\text{लाल}} = 2.41$, $n_{\text{नीला}} = 2.45$।
        \omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} किसी फलक में $45^\circ$ पर घुसता है; हर रंग
        का \omterm{def:g10:refraction:refraction}{अपवर्तन} कोण और कोणीय बँटवारा निकालिए।
  \item आगे चलकर लाल और नीली, दोनों किरणें भीतर से किसी निर्गत फलक पर
        $17.0^\circ$ पर पहुँचती हैं। दोनों निर्गत कोण और बाहर निकलते पंखे का
        बँटवारा निकालिए। प्रवेश वाले बँटवारे से तुलना कीजिए:
        \omterm{def:g10:refraction:critical}{क्रांतिक कोण} के पास \omterm{def:g10:refraction:dispersion}{वर्ण-विक्षेपण} का क्या
        होता है?
  \item पानी की एक बूँद ($n = 1.33$) किसी फलक पर चढ़ जाती है। हीरा--पानी
        सीमा का नया \omterm{def:g10:refraction:critical}{क्रांतिक कोण} निकालिए, प्रश्न 12 की $30^\circ$
        वाली किरण का भाग्य तय कीजिए, और समझाइए कि जौहरी हीरों को इतनी
        कड़ाई से साफ़ क्यों रखते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- तंतु।}
\begin{enumerate}[resume]
  \item किसी तंतु में $1.48$ \omterm{def:g10:refraction:index}{अपवर्तनांक} का \omterm{def:g10:refraction:fiber}{क्रोड} और $1.46$
        \omterm{def:g10:refraction:index}{अपवर्तनांक} का \omterm{def:g10:refraction:fiber}{आवरण} है। क्रोड--आवरण सीमा पर
        \omterm{def:g10:refraction:critical}{क्रांतिक कोण} निकालिए।
  \item वायु में रखे काँच के नंगे धागे का \omterm{def:g10:refraction:critical}{क्रांतिक कोण} $42.5^\circ$
        होता --- देखने में कहीं बेहतर। फिर भी \omterm{def:g10:refraction:fiber}{आवरण} क्यों लगाया
        जाता है, इसके दो कारण दीजिए। (धूल, खरोंच, और सतह को छूने से
        \omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन} पर क्या असर पड़ेगा --- इस पर
        सोचिए।)
  \item \omterm{def:g10:refraction:fiber}{क्रोड} में प्रकाश की चाल निकालिए, फिर \qty{6000}{km} लंबे तंतु
        में एक स्पंद का एकतरफ़ा यात्रा-समय।
  \item ठीक \omterm{def:g10:refraction:critical}{क्रांतिक कोण} पर उछलती किरण अक्ष की लंबाई का $1/\sin i_c$
        गुना चलती है। \qty{6000}{km} पर अतिरिक्त दूरी और उसमें लगने वाला
        अतिरिक्त समय निकालिए।
  \item और अंतिम चोट। आधुनिक तंतुओं का एक युग्म लगभग $10^{13}$ बिट प्रति
        \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} ढोता है, और यह पुस्तक लगभग $2.4 \times 10^8$ बिट की है। महासागर के
        आरपार आने-जाने का (``पिंग'') समय निकालिए और यह भी कि तंतु प्रति
        \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} इस पुस्तक की कितनी प्रतियाँ हिला देता है --- फिर एक वाक्य
        में बताइए कि \omterm{thm:g10:refraction:snell}{स्नेल का नियम} इंटरनेट के लिए क्या करता है।
\end{enumerate}
\end{problem}
