\chapter{Druk: van sport tot duiken}\label{ch:g10:pressure}

Een skiër glijdt over sneeuw waarin een wandelaar wegzakt; een duw van
\qty{5}{N} drijft een naald door leer waar een vuist geen deuk in maakt.
Dezelfde \omterm{def:g10:inertia:force}{krachten}, wild uiteenlopende gevolgen: de ontbrekende grootheid
is de \emph{\omterm{def:g10:pressure:pressure}{druk}}, de \omterm{def:g10:inertia:force}{kracht} gedeeld door het oppervlak dat haar
ontvangt. Dit hoofdstuk definieert die grootheid, volgt haar naar de
bodem van een zwembad, omhoog in een barometer, een samengeknepen
injectiespuit in --- en eindigt veertig \omterm{def:g10:orders-of-magnitude:unit}{meter} onder zee, waar ze van
levensbelang wordt.

\section{Druk: kracht verdeeld over een oppervlak}

\begin{definition}[Druk]\label{def:g10:pressure:pressure}
Wanneer een \omterm{def:g10:inertia:force}{kracht} met grootte $F$ (in \unit{N}) loodrecht op een
oppervlak met grootte $S$ (in \unit{m^2}) drukt, is de
\emph{druk}\index{druk} op dat oppervlak
\[
P = \frac{F}{S}.
\]
Haar \omterm{def:g10:orders-of-magnitude:unit}{eenheid}, de \unit{N/m^2}, heet de \emph{pascal}\index{pascal}
(\unit{Pa}) --- een minieme \omterm{def:g10:orders-of-magnitude:unit}{eenheid}, het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een appel
uitgesmeerd over een vierkante \omterm{def:g10:orders-of-magnitude:unit}{meter}, zodat \unit{kPa} en \unit{MPa} de
werkeenheden zijn.
\end{definition}

\begin{example}[Naaldhak tegen sneeuwschoen]\label{ex:g10:pressure:snowshoe}
Een persoon van \qty{60}{kg} weegt $F = 60 \times 9.81 \approx
\qty{590}{N}$. Op één naaldhak van \qty{1.0}{cm^2} = \qty{1.0e-4}{m^2}:
$P = 590/\num{1.0e-4} \approx \qty{5.9e6}{Pa} = \qty{5.9}{MPa}$. Op twee
sneeuwschoenen van elk \qty{0.20}{m^2}:
$P = 590/0.40 \approx \qty{1.5}{kPa}$. \emph{Hetzelfde} \omterm{def:g10:universal-gravitation:weight}{gewicht} drukt
onder de hak $4000$ keer harder: hakken laten putjes na in houten
vloeren, sneeuwschoenen drijven op poedersneeuw.
\end{example}

\begin{remark}[Bundelen of spreiden]\label{rem:g10:pressure:tools}
Een naainaald die met \qty{5}{N} op een punt van \qty{0.01}{mm^2} =
\qty{1e-8}{m^2} wordt geduwd, oefent $\qty{5e8}{Pa} = \qty{500}{MPa}$
uit --- genoeg om leervezels te scheiden. Doorborend gereedschap bundelt
de \omterm{def:g10:inertia:force}{kracht}; dragende constructies (ski's, rupsbanden, funderingen)
spreiden haar. Sport is toegepast drukbeheer: schaatsen bijten in het
ijs, ski's glijden over sneeuw.
\end{remark}

\section{Druk in een vloeistof in rust}

Waarom drukt water dieper harder? Zonder in gedachten een verticale
waterkolom af met dwarsdoorsnede $S$, van het oppervlak tot diepte $h$:
volume $Sh$, massa $\rho S h$ ($\rho$ de dichtheid van de vloeistof, in
\unit{kg/m^3}), \omterm{def:g10:universal-gravitation:weight}{gewicht} $\rho S h g$. De onderkant van de kolom draagt
dat \omterm{def:g10:universal-gravitation:weight}{gewicht} \emph{plus} de duw $P_0 S$ van de atmosfeer op de bovenkant.
Deel de totale \omterm{def:g10:inertia:force}{kracht} door $S$: de \omterm{def:g10:pressure:pressure}{druk} op diepte $h$ zou
$P_0 + \rho g h$ moeten zijn --- en dat is ze, overal in de vloeistof.

\begin{theorem}[Druk op diepte]\label{thm:g10:pressure:depth}
In een vloeistof met dichtheid $\rho$ die in rust is en aan de lucht
grenst, is de \omterm{def:g10:pressure:pressure}{druk} op diepte $h$ onder het oppervlak
\[
P = P_0 + \rho g h,
\]
waarin $P_0$ de \omterm{def:g10:pressure:atmospheric}{atmosferische druk} aan het oppervlak is. De \omterm{def:g10:pressure:pressure}{druk} hangt
alleen van de diepte af, niet van de vorm van het vat, en in een gegeven
punt drukt de vloeistof even hard in alle richtingen.
\end{theorem}
\admitted

\begin{remark}
Het kolomargument maakt de wet alleen aannemelijk; de eerlijke afleiding
(elke vorm, alle richtingen) staat in het volume van bachelorjaar 1. De
eigen term $\rho g h$ van het water is de \emph{overdruk}\index{overdruk}
--- wat de drukmeter van een duiker aanwijst.
\end{remark}

\begin{example}[Tien meter water]\label{ex:g10:pressure:tenmetres}
In zoet water ($\rho = \qty{1000}{kg/m^3}$), op $h = \qty{10}{m}$:
$\rho g h = 1000 \times 9.81 \times 10 \approx \qty{98}{kPa}$, dus
$P = 101 + 98 \approx \qty{199}{kPa}$. Tien \omterm{def:g10:orders-of-magnitude:unit}{meter} water voegt bijna
precies één atmosfeer toe: de vuistregel van de duiker, \emph{één
atmosfeer extra per tien \omterm{def:g10:orders-of-magnitude:unit}{meter}}.
\end{example}

\section{De oceaan van lucht}

\begin{definition}[Atmosferische druk]\label{def:g10:pressure:atmospheric}
We leven op de bodem van een oceaan van lucht, waarvan het \omterm{def:g10:universal-gravitation:weight}{gewicht} de
\emph{atmosferische druk}\index{atmosferische druk} veroorzaakt, op
zeeniveau
\[
P_0 \approx \qty{101}{kPa} = \qty{1.01e5}{Pa}.
\]
Op elke vierkante centimeter is dat
$F = \num{1.01e5} \times \num{1e-4} \approx \qty{10}{N}$: het \omterm{def:g10:universal-gravitation:weight}{gewicht} van
één \omterm{def:g10:orders-of-magnitude:unit}{kilogram} op elke \unit{cm^2} van je huid, je bureau, alles.
\end{definition}

\begin{example}[De onzichtbare last]\label{ex:g10:pressure:sheet}
Een A4-blad (\qty{21.0}{cm} $\times$ \qty{29.7}{cm}, dus
$S \approx \qty{0.0624}{m^2}$) ontvangt van de lucht erboven
$F = \num{1.01e5} \times 0.0624 \approx \qty{6.3e3}{N}$ --- het \omterm{def:g10:universal-gravitation:weight}{gewicht}
van een kleine auto. Er scheurt niets: de lucht eronder duwt precies zo
hard \emph{omhoog}. We merken de atmosfeer pas op wanneer één kant haar
kwijtraakt --- een zuignap, een rietje, een vliegtuigcabine.
\end{example}

\begin{remark}[De barometer van Torricelli]\label{rem:g10:pressure:torricelli}
In 1643 keerde Torricelli een aan één kant gesloten buis kwik om boven
een kwikbad. De kolom zakte tot de hoogte waarop $\rho g h = P_0$:
$h = \num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$, met
erboven het eerste vacuüm dat ooit gemaakt is. De hoogte volgt het weer
en krimpt met de hoogte boven zeeniveau: een barometer weegt de lucht
boven je (met water zou er \qty{10.3}{m} nodig zijn:
\cref{exo:g10:pressure:9}).
\end{remark}

\section{Gassen: bombardement en de wet van Boyle}

Ook een gas drukt op zijn vat: het is een zwerm moleculen in onophoudelijk
willekeurige beweging, elke botsing geeft de wand een minuscule duw naar
buiten, en miljarden miljarden inslagen per seconde vervagen tot de
gestage \omterm{def:g10:inertia:force}{kracht} die we als gasdruk meten. Pers het gas in het halve
volume: de moleculen slaan twee keer zo vaak toe, dus zou de \omterm{def:g10:pressure:pressure}{druk} moeten
verdubbelen. En dat doet ze.

\begin{proposition}[Wet van Boyle]\label{prop:g10:pressure:boyle}
\index{wet van Boyle}
Voor een vaste hoeveelheid gas op constante temperatuur zijn \omterm{def:g10:pressure:pressure}{druk} en
volume omgekeerd evenredig:
\[
P \, V = \text{constant},
\qquad\text{dus}\qquad
P_1 V_1 = P_2 V_2 .
\]
\end{proposition}
\admitted

\begin{remark}
Op dit niveau een experimenteel feit --- het volume van bachelorjaar 1
leidt de wet af uit het botsingsbeeld. De proef (een afgesloten
injectiespuit met lucht, een druksensor,
\qty{20}{\degreeCelsius}):
\begin{center}
\begin{tabular}{lccccc}
$V$ (\unit{mL}) & 60 & 50 & 40 & 30 & 20 \\
$P$ (\unit{kPa}) & 100 & 120 & 150 & 200 & 300 \\
$P \times V$ (\unit{kPa.mL}) & 6000 & 6000 & 6000 & 6000 & 6000
\end{tabular}
\end{center}
\end{remark}

\begin{example}[Een samengeknepen injectiespuit]\label{ex:g10:pressure:boyle}
Lucht neemt \qty{4.0}{L} in bij \qty{100}{kPa}; langzaam samengeperst
(zodat de temperatuur constant blijft) tot \qty{250}{kPa} neemt ze
$V_2 = P_1 V_1 / P_2 = 400/250 = \qty{1.6}{L}$ in.
\end{example}

\section{Duiken: druk aan het werk}

\begin{method}[Absolute druk op diepte]\label{met:g10:pressure:dive}
Voor een duiker op diepte $h$ in zee ($\rho \approx \qty{1025}{kg/m^3}$,
dus $\rho g h \approx \qty{101}{kPa}$ per \qty{10}{m}) bereken je
$P = P_0 + \rho g h$ --- en gebruik in de wet van Boyle altijd deze
\emph{absolute druk}\index{absolute druk}, nooit de \omterm{thm:g10:pressure:depth}{overdruk} alleen. Een
duiker zit onder $2$ atmosfeer op \qty{10}{m}, $3$ op \qty{20}{m} en $5$
op \qty{40}{m}.
\end{method}

\begin{remark}[Nooit je adem inhouden]\label{rem:g10:pressure:breath}
Een ademautomaat levert lucht op de omgevingsdruk: de longen vullen zich
op elke diepte normaal. Maar vul \qty{6.0}{L} op \qty{20}{m}
($\qty{302}{kPa}$), houd die adem in en kom boven: Boyle eist
$6.0 \times 302/101 \approx \qty{18}{L}$ --- drie keer wat een borstkas
bevat, en longen scheuren ruim daarvoor al, zelfs bij een stijging van
enkele \omterm{def:g10:orders-of-magnitude:unit}{meters}. Vandaar de eerste regel van het persluchtduiken:
\emph{houd bij het stijgen nooit je adem in} --- blijf ademen en de
uitzettende lucht stroomt gewoon naar buiten. Een vrijduiker loopt geen
gevaar: haar lucht is aan het oppervlak ingenomen, dus zet die op de weg
omhoog alleen weer uit tot haar oorspronkelijke volume.
\end{remark}

\begin{remark}[Decompressie]\label{rem:g10:pressure:deco}
Lucht onder hoge \omterm{def:g10:pressure:pressure}{druk} inademen lost bovendien extra stikstof op in het
bloed, zoals gas in een gesloten fles frisdrank; kom je te snel boven,
dan bruist die stikstof tot bellen in het lichaam --- decompressieziekte.
Vandaar het trage stijgen (ongeveer \qty{10}{m} per minuut) en de
ondiepe stops, die het gas rustig via de longen laten ontsnappen. De
theorie wacht op het volume van bachelorjaar 1; de gedragsregel hoort
bij elke duiker.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g10:pressure:1}
Een krat met een \omterm{def:g10:universal-gravitation:weight}{gewicht} van \qty{600}{N} staat eerst op een vlak van
$\qty{0.50}{m} \times \qty{0.30}{m}$ en daarna op een vlak van
$\qty{0.30}{m} \times \qty{0.20}{m}$. Bereken beide drukken op de grond.
Welke stand is de beste op zacht zand?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:2}
\omterm{def:g10:pressure:pressure}{Druk} \qty{0.25}{MPa} uit in \unit{kPa} en in \unit{Pa}. Wat drukt
harder: \qty{3.0}{N} op \qty{2.0}{cm^2}, of \qty{60}{N} op
\qty{500}{cm^2}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:3}
Bereken de \omterm{thm:g10:pressure:depth}{overdruk} $\rho g h$ en de \omterm{met:g10:pressure:dive}{absolute druk} op de bodem van een
zwembad van \qty{3.0}{m} diep ($\rho = \qty{1000}{kg/m^3}$,
$P_0 = \qty{101}{kPa}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:4}
Lucht neemt \qty{4.0}{L} in bij \qty{100}{kPa}, bij constante
temperatuur. Welk volume neemt ze in bij \qty{250}{kPa}? En bij
\qty{50}{kPa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:5}
Een schaatser van \qty{65}{kg} glijdt op één ijzer met een
contactoppervlak van \qty{12}{cm^2}, en gaat daarna op zolen van in
totaal \qty{360}{cm^2} staan. Bereken beide drukken en hun verhouding.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:6}
Een duim duwt met \qty{20}{N} op een punaise. De kop heeft een
oppervlakte van \qty{1.2}{cm^2}, de punt \qty{0.010}{mm^2}. Bereken de
drukken op de duim en op de muur; waarom wordt alleen de muur
doorboord?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:7}
Een luik van een onderzeeër met een oppervlakte van \qty{0.50}{m^2} zit
op \qty{30}{m} diepte in zeewater ($\rho = \qty{1025}{kg/m^3}$);
binnenin staat de lucht op \omterm{def:g10:pressure:atmospheric}{atmosferische druk}. Bereken de resulterende
\omterm{def:g10:inertia:force}{kracht} die het luik dicht houdt, en de massa waarvan het \omterm{def:g10:universal-gravitation:weight}{gewicht} daaraan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:8}
Een sensor op \qty{5.00}{m} diepte in een onbekende vloeistof wijst een
\omterm{thm:g10:pressure:depth}{overdruk} van \qty{66.2}{kPa} aan. Bepaal de dichtheid. Is het zoet
water?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:9}
Hoe hoog zou de barometer van Torricelli zijn met water
($\rho = \qty{1000}{kg/m^3}$) in plaats van kwik
($\rho = \qty{13600}{kg/m^3}$)? Haal beide hoogten uit

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:11}

(Constante temperatuur; zeewater voegt \qty{101}{kPa} per \qty{10}{m}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:12}
Een zuignap met een oppervlakte van \qty{12}{cm^2} wordt plat tegen een
plafond gedrukt.
\begin{enumerate}
  \item Bij een volmaakt vacuüm binnenin: welke \omterm{def:g10:inertia:force}{kracht} oefent de

        dragen?
  \item Een echte zuignap houdt \qty{20}{kPa} lucht binnenin. Welke

\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:13}
Een duikfles bevat \qty{12}{L} lucht bij \qty{200}{bar}
(\qty{1}{bar} = \qty{100}{kPa} $\approx 1$ atmosfeer); \qty{50}{bar}
moet als reserve overblijven. Een duiker ademt \qty{15}{L} per minuut op
de omgevingsdruk. Hoeveel liter lucht op oppervlaktedruk is bruikbaar?
Hoe lang gaat de fles mee op \qty{30}{m} (\omterm{met:g10:pressure:dive}{absolute druk}
$\approx \qty{4}{bar}$)? En op \qty{10}{m}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:14}
De longen van een vrijduiker bevatten \qty{6.0}{L} aan het oppervlak en
kunnen niet krimpen onder het residuvolume van \qty{1.5}{L}. Op welke
diepte in zee ($\rho = \qty{1025}{kg/m^3}$) wordt die grens bereikt?
(Ooit tot grens van het vrijduiken uitgeroepen; de records gaan
inmiddels voorbij \qty{100}{m}, doordat bloed naar de borstkas schuift
en het ontbrekende volume opvult.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:15}
In een hydraulische lift geeft een vloeistof de \omterm{def:g10:pressure:pressure}{druk} onveranderd door
van een kleine zuiger ($s = \qty{2.0}{cm^2}$) aan een grote
($S = \qty{400}{cm^2}$) die een auto van \qty{1200}{kg} draagt.
\begin{enumerate}
  \item Welke \omterm{def:g10:inertia:force}{kracht} op de kleine zuiger houdt de auto omhoog? Welke \omterm{def:g10:pressure:pressure}{druk}
        draagt de vloeistof?
  \item Hoever moet de kleine zuiger bewegen om de auto \qty{2.0}{cm} op
        te tillen? Vergelijk de arbeid aan elke zuiger.
\end{enumerate}
\end{exercise}

\section{Probleem: Eén adem omlaag, één adem omhoog}

\begin{problem}[{Weekendopgave --- één adem omlaag, één adem omhoog: de
natuurkunde van een vrijduik en van een stijging met perslucht, van
maskerknijp tot de regel dat de laatste tien meter de gevaarlijkste
zijn}]
\label{pb:g10:pressure:1}
Lena vrijduikt: één ademteug aan het oppervlak, naar \qty{30}{m} en
terug. Marco duikt naast haar met perslucht uit een fles. Dezelfde zee
($\rho = \qty{1025}{kg/m^3}$, $g = \qty{9.81}{N/kg}$,
$P_0 = \qty{101}{kPa}$), dezelfde longen van \qty{6.0}{L},
tegengestelde gevaren --- alles geregeerd door $P = P_0 + \rho g h$ en
$PV$ = constant (de temperatuur blijft overal constant).

\medskip
\noindent\textbf{Deel I --- Het \omterm{def:g10:universal-gravitation:weight}{gewicht} van water.}
\begin{enumerate}
  \item Bereken $\rho g$ voor zeewater in \unit{kPa} per \omterm{def:g10:orders-of-magnitude:unit}{meter}, en toets
        de vuistregel: één atmosfeer per \qty{10}{m}.
  \item Bereken de \omterm{met:g10:pressure:dive}{absolute druk} op \qty{10}{m}, \qty{20}{m},
        \qty{30}{m} en \qty{40}{m}.
  \item \omterm{def:g10:pressure:pressure}{Druk} elke waarde uit als veelvoud van de \omterm{def:g10:pressure:pressure}{druk} aan het oppervlak.
  \item Het masker van Lena bedekt \qty{150}{cm^2}. Welke resulterende
        \omterm{def:g10:inertia:force}{kracht} zou het water op \qty{30}{m} uitoefenen als de lucht
        erbinnen nog op oppervlaktedruk stond? (Duikers blazen door hun
        neus lucht in het masker om die \emph{maskerknijp} te
        voorkomen.)
  \item Een trommelvlies heeft een oppervlakte van ongeveer
        \qty{0.60}{cm^2}. Bereken de resulterende \omterm{def:g10:inertia:force}{kracht} erop op
        \qty{3.0}{m} als het middenoor op oppervlaktedruk blijft; waarom
        doen oren al in een gewoon zwembad pijn, en wat moet
        ``klaren'' bereiken?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Eén adem omlaag.}
Lena verlaat het oppervlak met \qty{6.0}{L} lucht in haar longen.
\begin{enumerate}[resume]
  \item Bereken met de wet van Boyle en absolute drukken haar longvolume
        op \qty{10}{m}, \qty{20}{m} en \qty{30}{m}.
  \item Op welke diepte is haar longvolume gehalveerd?
  \item Is het longvolume evenredig met de diepte? Beschrijf de kromme
        van $V$ tegen de \omterm{met:g10:pressure:dive}{absolute druk} $P$.
  \item Haar residuvolume is \qty{1.5}{L}: verder kunnen de longen niet
        krimpen. Toon aan dat die grens rond \qty{30}{m} wordt bereikt.
  \item Toch gaan de records vrijduiken voorbij \qty{100}{m}: wat vult
        het ontbrekende volume op? (Wat kan er verder nog naar de
        borstkas stromen?)
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Eén adem omhoog.}
Op \qty{30}{m} vult de ademautomaat de longen van Marco met
\qty{6.0}{L}.
\begin{enumerate}[resume]
  \item Op welke \omterm{def:g10:pressure:pressure}{druk} levert de automaat die lucht? Hoeveel keer dichter
        is ze dan lucht aan het oppervlak?
  \item Marco raakt in paniek, houdt zijn adem in en stijgt naar
        \qty{10}{m}: welk volume eist zijn opgesloten lucht daar op?
  \item Welk volume aan het oppervlak --- hoeveel keer zijn
        longinhoud?
  \item Formuleer de eerste regel van de persluchtduiker, en waarom
        normaal doorademen het gevaar wegneemt.
  \item Ook Lena houdt haar adem in van \qty{30}{m} tot het oppervlak,
        en toch lopen haar longen geen gevaar. Verklaar die asymmetrie.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Bellen, en de laatste tien \omterm{def:g10:orders-of-magnitude:unit}{meter}.}
\begin{enumerate}[resume]
  \item Marco laat op \qty{40}{m} een bel van \qty{1.0}{cm^3} los.
        Bereken haar volume op \qty{30}{m}, \qty{20}{m}, \qty{10}{m} en
        net onder het oppervlak.
  \item Bereken voor elk traject van \qty{10}{m} de factor waarmee de
        bel groeit. Waar groeit ze het snelst?
  \item Duikers stijgen hoogstens \qty{10}{m} per minuut: hoe lang duurt
        het vanaf \qty{40}{m}, en in welke minuut zet opgesloten of
        opgelost gas het sterkst uit?
  \item Op diepte lost het bloed van Marco extra stikstof op, zoals gas
        in een gesloten fles frisdrank. Wat richt een te snelle stijging
        aan, en hoe voorkomen het trage stijgen en een veiligheidsstop
        op \qty{5}{m} dat?
  \item De clou: geef in telkens één zin het oordeel over de vrijduiker,
        het oordeel over de persluchtduiker, en de becijferde reden
        waarom de laatste tien \omterm{def:g10:orders-of-magnitude:unit}{meter} van elke stijging het meeste
        respect verdienen.
\end{enumerate}
\end{problem}
