\chapter{Signalen en golven}\label{ch:g10:signals-and-waves}

Elke boodschap die je ooit hebt ontvangen kwam binnen als een \omterm{def:g10:signals-and-waves:signal}{signaal}:
een stem die op je trommelvlies drukte, deze bladzijde die je als een
flikkering radiogolf bereikte. Achter al die verscheidenheid zit één
idee --- een \omterm{def:g10:orders-of-magnitude:unit}{natuurkundige grootheid} die met de tijd verandert --- en
één gereedschapskist: \omterm{def:g10:signals-and-waves:period}{periode}, \omterm{def:g10:signals-and-waves:frequency}{frequentie}, \omterm{def:g10:signals-and-waves:amplitude}{amplitude} en de kunst van het
klokken van een vertraging. Daarmee meet een boot de zeebodem en telt een
arts hartslagen.

\section{Signalen dragen informatie}

\begin{definition}[Signaal]\label{def:g10:signals-and-waves:signal}
Een \emph{signaal}\index{signaal} is een \omterm{def:g10:orders-of-magnitude:unit}{natuurkundige grootheid} die met
de tijd verandert om informatie te dragen: de luchtdruk op je
trommelvlies, de spanning op de klemmen van een microfoon, de helderheid
van het licht in een \omterm{def:g10:refraction:fiber}{glasvezel}. Om een signaal te bestuderen neem je het
op als functie van de tijd en lees je de informatie van de kromme af.
\end{definition}

In een telefoongesprek wisselt dezelfde informatie herhaaldelijk van
drager: \omterm{def:g10:pressure:pressure}{druk} in de lucht, dan spanning in de microfoon, dan radiogolven
bij de antenne --- en aan de andere kant weer terug. Eén woordenschat
bedient ze allemaal; dit hoofdstuk bouwt die op.

\section{Periodieke signalen}

\begin{definition}[Periodiek signaal, periode]\label{def:g10:signals-and-waves:period}
Een \omterm{def:g10:signals-and-waves:signal}{signaal} is \emph{periodiek}\index{periodiek signaal} wanneer het zich
op regelmatige tijdsafstanden identiek herhaalt. De
\emph{periode}\index{periode} $T$ is de duur van één volledige cyclus, in
seconde.
\end{definition}

\begin{definition}[Frequentie]\label{def:g10:signals-and-waves:frequency}
De \emph{frequentie}\index{frequentie} van een \omterm{def:g10:signals-and-waves:period}{periodiek signaal} is het
aantal cycli per seconde:
\[
f = \frac{1}{T} .
\]
De \omterm{def:g10:orders-of-magnitude:unit}{eenheid}, één cyclus per seconde, is de \emph{hertz}\index{hertz}
(\unit{Hz}), met de gebruikelijke veelvouden \unit{kHz}, \unit{MHz} en
\unit{GHz}.
\end{definition}

\begin{definition}[Amplitude]\label{def:g10:signals-and-waves:amplitude}
De \emph{amplitude}\index{amplitude} van een \omterm{def:g10:signals-and-waves:period}{periodiek signaal} is de
maximale afwijking van de rustwaarde. Voor een spanning die symmetrisch
om nul slingert, is de amplitude de topwaarde; van top tot dal meet je
\emph{tweemaal} de amplitude (de top-tot-topwaarde).
\end{definition}

\begin{example}[Netspanning en kamertoon]\label{ex:g10:signals-and-waves:mains}
De netspanning slingert met $f = \qty{50}{Hz}$: haar \omterm{def:g10:signals-and-waves:period}{periode} is
$T = 1/f = \qty{0.020}{s} = \qty{20}{ms}$. De kamertoon a van een
orkest is een druksignaal van \qty{440}{Hz}: één cyclus duurt
$T = 1/440 = \qty{2.3}{ms}$. Hogere \omterm{def:g10:signals-and-waves:frequency}{frequentie}, kortere cyclus: $f$ en
$T$ zijn elkaars omgekeerde.
\end{example}

\section{Een oscillogram lezen}

\begin{definition}[Oscilloscoop]\label{def:g10:signals-and-waves:oscilloscope}
Een \emph{oscilloscoop}\index{oscilloscoop} zet een spanning uit tegen de
tijd; de kromme op het geruite scherm is een
\emph{oscillogram}\index{oscillogram}. Twee instellingen zetten hokjes op
het scherm om in natuurkundige waarden: de
\emph{tijdbasis}\index{tijdbasis} (seconden per horizontaal hokje) en de
verticale \emph{gevoeligheid}\index{gevoeligheid} (volt per verticaal
hokje).
\end{definition}

\begin{method}[Van scherm naar getallen]\label{met:g10:signals-and-waves:oscillogram}
\begin{enumerate}
  \item Tel de horizontale hokjes die één volledige cyclus beslaat (van
        top tot top is het veiligst); vermenigvuldig met de \omterm{def:g10:signals-and-waves:oscilloscope}{tijdbasis} om
        $T$ te krijgen, en daarna $f = 1/T$.
  \item Tel de verticale hokjes van de middellijn tot een top en
        vermenigvuldig met de \omterm{def:g10:signals-and-waves:oscilloscope}{gevoeligheid} om de \omterm{def:g10:signals-and-waves:amplitude}{amplitude} te krijgen
        (van top tot dal krijg je de top-tot-topwaarde: halveer die).
\end{enumerate}
\end{method}

\begin{example}[Een volledige aflezing]\label{ex:g10:signals-and-waves:reading}
Op het \omterm{def:g10:signals-and-waves:oscilloscope}{oscillogram} hieronder staat de \omterm{def:g10:signals-and-waves:oscilloscope}{tijdbasis} op \qty{5}{ms} per hokje
en de \omterm{def:g10:signals-and-waves:oscilloscope}{gevoeligheid} op \qty{2}{V} per hokje. Eén cyclus beslaat $4.0$
hokjes: $T = 4.0 \times \qty{5}{ms} = \qty{20}{ms}$ en
$f = 1/\qty{0.020}{s} = \qty{50}{Hz}$ --- opnieuw het lichtnet. Een top
ligt $3.0$ hokjes boven de as: de \omterm{def:g10:signals-and-waves:amplitude}{amplitude} is
$3.0 \times \qty{2}{V} = \qty{6.0}{V}$.
\end{example}

\section{Geluid en elektromagnetische signalen}

\begin{definition}[Geluid]\label{def:g10:signals-and-waves:sound}
\emph{Geluid}\index{geluid} is een druksignaal: een trillend voorwerp ---
een snaar, een vlies, stembanden --- duwt ritmisch tegen de lucht, en de
verdichtingen reizen naar buiten met ongeveer \qty{340}{m/s}. Een
microfoon zet de aankomende \omterm{def:g10:pressure:pressure}{druk} om in een spanning; het trommelvlies zet
haar om in zenuwprikkels. Geluid heeft een tussenstof nodig: een bel die
onder een vacuümstolp wordt geluid, verstomt naarmate de lucht wordt
weggepompt.
\end{definition}

\begin{definition}[Golf]\label{def:g10:signals-and-waves:wave}
Een reizende verstoring --- het verdichtingspatroon van \omterm{def:g10:signals-and-waves:sound}{geluid}, de
rimpeling op een vijver, een radiopuls --- is een \emph{golf}\index{golf}.
De tussenstof, als die er is, blijft ter plaatse: elk plekje water dobbert
op zijn plaats terwijl de rimpeling de vijver oversteekt. De meetkunde van
golven --- \omterm{def:g10:light-spectra:wavelength}{golflengte}, interferentie, buiging --- komt in een later
hoofdstuk aan bod; dit hoofdstuk heeft alleen hun snelheden nodig.
\end{definition}

\begin{definition}[Ultrageluid en infrageluid]\label{def:g10:signals-and-waves:ultrasound}
Het menselijk oor hoort \omterm{def:g10:signals-and-waves:sound}{geluid} tussen ongeveer \qty{20}{Hz} en
\qty{20}{kHz} (de bovengrens zakt met de jaren). \omterm{def:g10:signals-and-waves:sound}{Geluid} boven
\qty{20}{kHz} heet \emph{ultrageluid}\index{ultrageluid} --- vleermuizen
jagen op \qty{50}{kHz}, medische scanners brengen het lichaam in beeld op
enkele megahertz. \omterm{def:g10:signals-and-waves:sound}{Geluid} onder \qty{20}{Hz} heet
\emph{infrageluid}\index{infrageluid}, gevoeld door olifanten en
seismometers.
\end{definition}

\begin{proposition}[Elektromagnetische signalen]\label{prop:g10:signals-and-waves:em}
Radio, wifi en licht zijn \emph{elektromagnetische
signalen}\index{elektromagnetisch signaal}: één familie, die alle door
vacuüm reizen --- er is geen tussenstof nodig --- en alle met één
snelheid, de lichtsnelheid uit \cref{ch:g10:refraction},
\[
c = \qty{3.00e8}{m/s} \quad \text{(in lucht vrijwel dezelfde).}
\]
\end{proposition}
\admitted

\begin{remark}\label{rem:g10:signals-and-waves:honest}
Hier is dat een experimenteel feit --- zonlicht doorkruist $150$ miljoen
kilometer lege ruimte; het volume van bachelorjaar 2 leidt het eerlijk af
uit de wetten van elektriciteit en magnetisme. Houd het contrast vast:
\omterm{def:g10:signals-and-waves:sound}{geluid} is een trilling \emph{van} de tussenstof en sterft zonder haar;
een \omterm{prop:g10:signals-and-waves:em}{elektromagnetisch signaal} draagt zijn eigen veld door het vacuüm, een
miljoen keer sneller.
\end{remark}

\section{Echo's: afstanden uit vertragingen}

Een \omterm{def:g10:signals-and-waves:signal}{signaal} dat met snelheid $v$ beweegt, legt in een tijd $t$ een
afstand $d = v\,t$ af: elke vertraging is een vermomde afstand. Beter
nog: stuur een korte puls naar een obstakel en klok zijn
\emph{echo}\index{echo} --- de heen-en-terugweg beslaat $2d$.

\begin{method}[Afstandsmeting met een echo]\label{met:g10:signals-and-waves:echo}
\begin{enumerate}
  \item Stuur een korte puls; meet de heen-en-terugtijd $t$ van de \omterm{rem:g10:signals-and-waves:honest}{echo}.
  \item Neem de snelheid $v$ van het \omterm{def:g10:signals-and-waves:signal}{signaal} \emph{in de doorkruiste
        tussenstof}.
  \item De puls legde $2d = v\,t$ af, dus $d = \dfrac{v\,t}{2}$.
\end{enumerate}
Geluidspulsen in water maken \emph{sonar}\index{sonar}
($v \approx \qty{1500}{m/s}$); radiopulsen in lucht maken
\emph{radar}\index{radar} ($v = c$); pulsen \omterm{def:g10:signals-and-waves:ultrasound}{ultrageluid} in het lichaam
maken de medische scanner ($v \approx \qty{1540}{m/s}$ in zacht weefsel).
\end{method}

\begin{example}[Sonar]\label{ex:g10:signals-and-waves:sonar}
De \omterm{met:g10:signals-and-waves:echo}{sonar} van een schip zendt een ping uit en de \omterm{rem:g10:signals-and-waves:honest}{echo} van de zeebodem
komt terug na $t = \qty{0.80}{s}$: de diepte is
$d = \frac{1500 \times 0.80}{2} = \qty{600}{m}$. Wie de factor $2$
vergeet, verdubbelt de oceaan.
\end{example}

\begin{example}[Radar]\label{ex:g10:signals-and-waves:radar}
Een luchthavenradar ontvangt de \omterm{rem:g10:signals-and-waves:honest}{echo} van een vliegtuig \qty{0.30}{ms} na
de puls:
$d = \frac{\qty{3.00e8}{m/s} \times \qty{3.0e-4}{s}}{2} = \qty{45}{km}$.
Omdat $c$ zo groot is, leeft \omterm{met:g10:signals-and-waves:echo}{radar} op klokken van microseconden: licht
klokken is precisiewerk.
\end{example}

\begin{example}[Het hart als signaal]\label{ex:g10:signals-and-waves:ecg}
Elke hartslag stuurt een kleine spanningspuls door de borstkas;
elektroden nemen die op als het
\emph{elektrocardiogram}\index{elektrocardiogram} (ecg), een vrijwel
\omterm{def:g10:signals-and-waves:period}{periodiek signaal}. Liggen de hoge pieken $T = \qty{0.75}{s}$ uit elkaar,
dan klopt het hart met $f = 1/0.75 = \qty{1.33}{Hz}$, dus
$1.33 \times 60 = 80$ slagen per minuut: $T$ van een strook aflezen is
hoe een cardioloog je pols opneemt.
\end{example}

\section{Probleem: De sonar, de storm en het cardiogram}

\begin{problem}[{Weekendopgave --- de signalen van de wereld lezen: één
wet, $d = v\,t$, meet een storm, brengt een zeebodem in kaart, telt een
hartslag en ontrafelt hoe gps werkt}]
\label{pb:g10:signals-and-waves:1}
Een vissersboot werkt een stormachtige nacht door: bliksem aan de
horizon, de \omterm{met:g10:signals-and-waves:echo}{sonar} die de bodem aftast, het hart van de schipper op de
papierstrook van de dokter, een gps-ontvanger die naar \omterm{def:g10:universal-gravitation:satellite}{satellieten}
luistert. Vier instrumenten, één wet. Neem
$v_{\text{geluid}} = \qty{340}{m/s}$ in lucht, \qty{1500}{m/s} in
zeewater, en $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Deel I --- De storm.}
\begin{enumerate}
  \item Een flits; \qty{9.0}{s} later volgt de donder. Hoever weg is
        hij?
  \item Bereken de reistijd van het licht over die afstand, en
        verantwoord dat je de flits als ogenblikkelijk behandelt.
  \item Zeelieden tellen de seconden tussen flits en donder en delen door
        drie om kilometers te krijgen. Verantwoord die regel.
  \item Vijf minuten later geeft een flits \qty{6.0}{s}. Hoever nu?
        Bepaal de gemiddelde naderingssnelheid van de storm in
        \unit{m/s} en in \unit{km/h}.
  \item De donder rommelt in plaats van te knallen: het bliksemkanaal is
        kilometers lang, dus liggen de delen ervan op verschillende
        afstanden. Als het kanaal zich van \qty{2.0}{km} tot
        \qty{5.0}{km} van de boot uitstrekt, hoe lang duurt het gerommel
        dan?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De \omterm{met:g10:signals-and-waves:echo}{sonar}.}
\begin{enumerate}[resume]
  \item De \omterm{met:g10:signals-and-waves:echo}{sonar} zendt een ping uit; de zeebodem antwoordt in
        \qty{0.90}{s}. Diepte?
  \item Boven het plat verkort de \omterm{rem:g10:signals-and-waves:honest}{echo} tot \qty{0.20}{s}. Diepte?
  \item Eén ping levert \emph{twee} \omterm{rem:g10:signals-and-waves:honest}{echo}'s op, na \qty{0.16}{s} en
        \qty{0.20}{s}. Verklaar ze; hoe hoog boven de bodem zwemt de
        vissenschool?
  \item De \omterm{met:g10:signals-and-waves:echo}{sonar} zendt om de \qty{0.50}{s} een ping uit. Wat is de
        grootste diepte die hij zonder verwarring kan meten, en wat gaat
        er daarvoorbij mis?
  \item Welke afstand zou dezelfde vertraging van \qty{0.60}{s} in lucht
        betekenen? Moraal: wat moet je weten voordat je een vertraging in
        een afstand omzet?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het cardiogram.}
\begin{enumerate}[resume]
  \item De ecg-pieken van de schipper liggen \qty{0.86}{s} uit elkaar.
        \omterm{def:g10:signals-and-waves:frequency}{Frequentie}? Slagen per minuut?
  \item De strook loopt met \qty{25}{mm/s}. Welke piekafstand heeft de
        dokter gemeten?
  \item Na het binnenhalen van de netten is de afstand \qty{15}{mm}.
        Nieuwe hartslag?
  \item De jonge dekknecht, een getrainde roeier, heeft in rust $50$
        slagen per minuut. \omterm{def:g10:signals-and-waves:period}{Periode}? Afstand op de strook?
  \item Is een ecg strikt \omterm{def:g10:signals-and-waves:period}{periodiek}? Wat leest de dokter dan van de
        strook af?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Huiswaarts met het licht.}
\begin{enumerate}[resume]
  \item De boot roept de haven op, \qty{50}{km} verderop. Hoe lang doet
        het bericht erover? Hoe lang zou \omterm{def:g10:signals-and-waves:sound}{geluid} erover doen?
  \item Een \omterm{def:g10:universal-gravitation:satellite}{gps-satelliet} draait ongeveer \qty{20200}{km} boven ons.
        Hoe lang doet haar \omterm{def:g10:signals-and-waves:signal}{signaal} erover om de boot te bereiken?
  \item Gps zet tijd om in positie. Welke afstandsfout veroorzaakt een
        klokfout van \qty{1.0}{\micro s}? Welke tijdnauwkeurigheid vergt
        een plaatsbepaling op \qty{3.0}{m}?
  \item Waarom kan er, zelfs in beginsel, geen gps op \omterm{def:g10:signals-and-waves:sound}{geluid} bestaan?
  \item Slot: noem de drie snelheden die vannacht zijn gebruikt en de
        ene wet achter alle vier de instrumenten, en formuleer daarna in
        één zin de gewoonte van het afstandmeten.
\end{enumerate}
\end{problem}
