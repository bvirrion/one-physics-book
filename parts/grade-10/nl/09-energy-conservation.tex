\chapter{Energie: vormen en behoud}\label{ch:g10:energy-conservation}

Een rijdende auto, een opgeladen telefoon, een hete oven en een geheven
hamer delen één eigenschap: elk kan iets laten gebeuren. De natuurkunde
meet dat \omterm{def:g10:energy-conservation:power}{vermogen} met één getal --- de \omterm{def:g10:energy-conservation:energy}{energie} --- en voert de
boekhouding onder één ijzeren regel: het totaal verandert nooit, het
verplaatst zich alleen en verkleedt zich. Dit hoofdstuk opent het
grootboek: de vormen die \omterm{def:g10:energy-conservation:energy}{energie} aanneemt, de ketens waarlangs ze reist,
en wat je voor een \omterm{def:g10:energy-conservation:kwh}{kilowattuur} koopt.

\section{Energie: de gemeenschappelijke munt}

\begin{definition}[Energie en de joule]\label{def:g10:energy-conservation:energy}
\emph{Energie}\index{energie} meet het \omterm{def:g10:energy-conservation:power}{vermogen} van een systeem om
veranderingen teweeg te brengen: materie in beweging zetten, optillen,
verwarmen, laten oplichten. De \omterm{def:g10:orders-of-magnitude:unit}{eenheid} is de \emph{joule}\index{joule}
(\unit{J}). Opgeslagen, overgedragen, omgezet --- elke tak van de
natuurkunde rekent in deze ene munt.
\end{definition}

\begin{definition}[Kinetische energie]\label{def:g10:energy-conservation:kinetic}
Een lichaam met massa $m$ (in \unit{kg}) dat met snelheid $v$ (in
\unit{m/s}) beweegt, draagt de \emph{kinetische
energie}\index{kinetische energie} $E_k = \tfrac12\, m v^2$: de massa
verdubbelen verdubbelt haar, de snelheid verdubbelen verviervoudigt
haar.
\end{definition}

\begin{example}[Auto tegen voetganger]\label{ex:g10:energy-conservation:kinetic}
Een auto van \qty{1300}{kg} met \qty{100}{km/h} $= \qty{27.8}{m/s}$
draagt $E_k = \tfrac12 \times 1300 \times 27.8^2 \approx
\qty{5.0e5}{J}$; een voetganger van \qty{70}{kg} met \qty{1.5}{m/s}
draagt $\tfrac12 \times 70 \times 1.5^2 \approx \qty{79}{J}$ --- zesduizend
keer minder. Snelheid is duur: het kwadraat zorgt daarvoor.
\end{example}

\begin{definition}[Potentiële energie in het zwaarteveld]\label{def:g10:energy-conservation:potential}
Een lichaam met massa $m$ op hoogte $h$ (in \unit{m}) boven een gekozen
referentieniveau slaat de \emph{potentiële energie}\index{potentiële
energie} $E_p = m g h$ op, met $g = \qty{9.81}{N/kg}$ --- geldig vlak bij
de grond, waar $g$ in de praktijk constant is. Alleen
\emph{verschillen} in hoogte tellen: het niveau waarop $E_p = 0$ kies je
vrij.
\end{definition}

\begin{example}[Watertoren]\label{ex:g10:energy-conservation:potential}
Elke kubieke \omterm{def:g10:orders-of-magnitude:unit}{meter} water (\qty{1000}{kg}) die \qty{30}{m} hoog in een
watertoren is opgeslagen, bevat
$1000 \times 9.81 \times 30 \approx \qty{2.9e5}{J}$ --- ongeveer zes
tiende van de \omterm{def:g10:energy-conservation:kinetic}{kinetische energie} van de auto hierboven. Steden leggen
\omterm{def:g10:energy-conservation:energy}{energie} boven hun hoofd aan.
\end{example}

\begin{definition}[De andere vormen]\label{def:g10:energy-conservation:forms}
\omterm{def:g10:energy-conservation:energy}{Energie} schuilt ook in minder zichtbare vermommingen: \emph{thermische
energie}\index{thermische energie}, de wanordelijke beweging van de
moleculen van een lichaam (hoe heter, hoe meer); \emph{chemische
energie}\index{chemische energie}, opgeslagen in moleculaire bindingen
--- voedsel, hout, benzine, batterijen; \emph{elektrische
energie}\index{elektrische energie}, door stromen van centrale naar
stopcontact gedragen; \emph{stralingsenergie}\index{stralingsenergie},
door licht gedragen, zelfs door lege ruimte --- zo bereikt de \omterm{def:g10:energy-conservation:energy}{energie} van
de Zon ons; \emph{kernenergie}\index{kernenergie}, opgeslagen binnen in
atoomkernen en vrijgemaakt in sterren
(\cref{ch:g12:nuclear-energy}).
\end{definition}

\section{Overdrachten en omzettingsketens}

\begin{definition}[Overdracht, omzetting, keten]\label{def:g10:energy-conservation:transfer}
Een \emph{energieoverdracht}\index{energieoverdracht} verplaatst \omterm{def:g10:energy-conservation:energy}{energie}
tussen systemen --- \emph{reservoirs}\index{reservoir (van energie)}; een
\emph{energieomzetting}\index{energieomzetting} verandert haar vorm. Een
apparaat dat een van beide doet (turbine, motor, lamp, spier) is een
\emph{omzetter}\index{omzetter}. Een
\emph{omzettingsketen}\index{omzettingsketen} brengt de reis in beeld:
vakjes voor reservoirs en vormen, pijlen met opschrift voor de
overdrachten.
\end{definition}

\begin{method}[Een omzettingsketen tekenen]\label{met:g10:energy-conservation:chain}
\begin{enumerate}
  \item Bepaal het beginreservoir en de beginvorm (wat wordt verbruikt?).
  \item Volg de \omterm{def:g10:energy-conservation:energy}{energie} van \omterm{def:g10:energy-conservation:transfer}{omzetter} naar \omterm{def:g10:energy-conservation:transfer}{omzetter}: één pijl per
        overdracht, met de vorm benoemd in elk vakje.
  \item Zet bij elke echte \omterm{def:g10:energy-conservation:transfer}{omzetter} een warmtepijl --- er lekt altijd
        \omterm{def:g10:energy-conservation:energy}{energie} weg als warmte (\omterm{def:g10:inertia:inventory}{wrijving}, draden, uitlaat).
\end{enumerate}
\end{method}

potentieel op het hoogste punt van de uitslag, alles kinetisch door het
laagste punt (met $E_p = 0$ daar):

\section{Behoud van energie}

\begin{theorem}[Behoud van energie]\label{thm:g10:energy-conservation:conservation}
\index{behoud van energie}
De totale \omterm{def:g10:energy-conservation:energy}{energie} van een geïsoleerd systeem --- een systeem dat niets
met de buitenwereld uitwisselt --- is constant. \omterm{def:g10:energy-conservation:energy}{Energie} wordt nooit
geschapen en nooit vernietigd: ze verandert alleen van vorm en van
plaats, en wat het ene reservoir verliest, winnen de andere, \omterm{def:g10:energy-conservation:energy}{joule} voor
\omterm{def:g10:energy-conservation:energy}{joule}.
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:energy-conservation:admitted}
Geen enkele proef heeft dit beginsel ooit op een fout betrapt; hier nemen
we het als gegeven aan en gebruiken we het om te rekenen.
\cref{ch:g12:work-and-energy} leidt het mechanische deel ervan
($E_k + E_p$) af uit de wetten van Newton; de volledige boekhouding wacht
op de universitaire volumes.
\end{remark}

\begin{example}[Vrije val, in joules geprijsd]\label{ex:g10:energy-conservation:fall}
Een bal van \qty{2.0}{kg} valt van \qty{10}{m} (de luchtweerstand is
verwaarloosbaar): hij begint met
$E_p = 2.0 \times 9.81 \times 10 \approx \qty{196}{J}$ en landt met
$E_k = \qty{196}{J}$, dus $v = \sqrt{2 \times 196/2.0} = \qty{14}{m/s}
\approx \qty{50}{km/h}$. Geen \omterm{def:g10:inertia:force}{krachten}, geen stopwatch: het behoud
alleen prijst de klap.
\end{example}

\section{Rendement}

\begin{definition}[Rendement]\label{def:g10:energy-conservation:efficiency}
Een \omterm{def:g10:energy-conservation:transfer}{omzetter} verbruikt $E_{\text{verbruikt}}$ en levert de gewenste vorm
$E_{\text{nuttig}}$; zijn \emph{rendement}\index{rendement} is
\[
\eta = \frac{E_{\text{nuttig}}}{E_{\text{verbruikt}}} < 1 .
\]
Het behoud verbiedt $\eta > 1$, en echte \omterm{def:g10:energy-conservation:transfer}{omzetters} halen de $1$ nooit:
de ontbrekende \omterm{def:g10:energy-conservation:energy}{energie} is niet vernietigd --- ze vertrekt als warmte.
\end{definition}

\begin{example}[Waar de benzine blijft]\label{ex:g10:energy-conservation:engine}
Een \omterm{def:g10:orders-of-magnitude:unit}{kilogram} benzine bevat \qty{45}{MJ} \omterm{def:g10:energy-conservation:forms}{chemische energie}; een automotor
zet die om met $\eta \approx 0.30$: \qty{13.5}{MJ} beweging,
\qty{31.5}{MJ} die de uitlaat, de radiator en de straat verwarmt.
Tweederde van elke tank verwarmt de atmosfeer.
\end{example}

\begin{remark}[Rendementen vermenigvuldigen]\label{rem:g10:energy-conservation:chain-eta}
Geketende \omterm{def:g10:energy-conservation:transfer}{omzetters} vermenigvuldigen hun \omterm{def:g10:energy-conservation:efficiency}{rendementen}: een centrale
($0.38$) die het net voedt ($0.94$) dat een lader voedt ($0.85$) levert
$\eta = 0.38 \times 0.94 \times 0.85 \approx 0.30$. Lange ketens zijn
lekkende leidingen: elke pijl heft belasting op de stroom.
\end{remark}

\section{Vermogen, de watt en het kilowattuur}

\begin{definition}[Vermogen]\label{def:g10:energy-conservation:power}
Het \emph{vermogen}\index{vermogen} is de snelheid waarmee \omterm{def:g10:energy-conservation:energy}{energie}
stroomt, $P = E/t$, gemeten in \emph{watt}\index{watt} (\unit{W}), met
$\qty{1}{W} = \qty{1}{J/s}$. \omterm{def:g10:energy-conservation:energy}{Energie} is een hoeveelheid, vermogen een
stroom --- een badkuip tegenover haar kraan.
\end{definition}

\begin{example}[Waterkoker tegen mens]\label{ex:g10:energy-conservation:power}
Een waterkoker van \qty{2200}{W} die \qty{150}{s} draait, verbruikt
$E = 2200 \times 150 = \qty{3.3e5}{J}$. Een mens draait op ongeveer
\qty{10}{MJ} voedsel per dag, gemiddeld
$\num{1.0e7}/\num{86400} \approx \qty{116}{W}$: je staat stationair te
draaien als een felle ouderwetse gloeilamp.
\end{example}

\begin{definition}[Het kilowattuur]\label{def:g10:energy-conservation:kwh}
Het \emph{kilowattuur}\index{kilowattuur} (\unit{kW.h}) is de \omterm{def:g10:energy-conservation:energy}{energie} van
een stroom van \qty{1}{kW} die één uur loopt:
$\qty{1}{kW.h} = 1000 \times 3600\,\unit{J} = \qty{3.6}{MJ}$ --- de
\omterm{def:g10:orders-of-magnitude:unit}{eenheid} die elektriciteitsmeters tellen: \omterm{def:g10:energy-conservation:power}{vermogen} in \unit{kW} maal
uren.
\end{definition}

\begin{example}[De rekening lezen]\label{ex:g10:energy-conservation:bill}
Een rekening is een aftrekking en een vermenigvuldiging: de nieuwe
meterstand min de oude geeft de \omterm{def:g10:energy-conservation:kwh}{kilowattuur}; maal de eenheidsprijs
(ongeveer $0.25$ euro per \unit{kW.h}), plus een vast abonnement, geeft
het totaal. Een kachel van \qty{2}{kW} die elke nacht \qty{2.5}{h}
draait, verbruikt \qty{5}{kW.h} per dag: in zijn eentje $38$ euro per
maand.
\end{example}

\begin{example}[Ordegrootten]\label{ex:g10:energy-conservation:orders}
Uit het hoofd te kennen: een appel één \omterm{def:g10:orders-of-magnitude:unit}{meter} optillen, ongeveer
\qty{1}{J}; een telefoonaccu, $\qty{10}{W.h} \approx \qty{4e4}{J}$; een
kom cornflakes, \qty{1.5}{MJ}; je voedsel voor een dag, \qty{10}{MJ}; een
\omterm{def:g10:orders-of-magnitude:unit}{kilogram} benzine, \qty{45}{MJ} --- vrijwel precies het dagelijkse
stroomverbruik van een huishouden ($\approx \qty{12}{kW.h}$); een
bliksemschicht, \qty{5}{GJ}: honderd huishouddagen, in een flits
geleverd.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:1}
Een auto van \qty{1200}{kg} rijdt \qty{90}{km/h}, dus \qty{25}{m/s}:
bereken zijn \omterm{def:g10:energy-conservation:kinetic}{kinetische energie}. Doe het opnieuw bij \qty{130}{km/h}
(\qty{36.1}{m/s}): wat heeft $44\%$ meer snelheid met $E_k$ gedaan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:2}
Een wandelaar met massa \qty{72}{kg} klimt van hoogte \qty{1200}{m} naar
\qty{1650}{m}. Bereken de gewonnen \omterm{def:g10:energy-conservation:potential}{potentiële energie}, zet die om naar
\omterm{def:g10:energy-conservation:kwh}{kilowattuur} en prijs haar tegen $0.25$ euro per \unit{kW.h}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:3}
Zet om: \qty{1}{kW.h} naar \omterm{def:g10:energy-conservation:energy}{joule}; een telefoonaccu van \qty{12}{W.h}
naar kilojoule; \qty{45}{MJ} (één \omterm{def:g10:orders-of-magnitude:unit}{kilogram} benzine) naar \omterm{def:g10:energy-conservation:kwh}{kilowattuur}; je
voedseldag van \qty{10}{MJ} naar \omterm{def:g10:energy-conservation:kwh}{kilowattuur}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:4}
Een waterkoker van \qty{2200}{W} draait \qty{150}{s}. Bereken de
verbruikte \omterm{def:g10:energy-conservation:energy}{energie}, in \omterm{def:g10:energy-conservation:energy}{joule} en in \omterm{def:g10:energy-conservation:kwh}{kilowattuur}. Bereken daarna het
gemiddelde \omterm{def:g10:energy-conservation:power}{vermogen} van een mens die op \qty{10}{MJ} per dag draait.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:5}
Schrijf de \omterm{def:g10:energy-conservation:transfer}{omzettingsketen}
(\cref{met:g10:energy-conservation:chain}) op van: een fietsdynamo die
een lamp voedt; een gasboiler; een zonnepaneel dat een telefoon oplaadt.
Geef de warmtelekken aan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:6}
Een motor verbrandt \qty{2.0}{kg} benzine (\qty{45}{MJ/kg}) en levert
\qty{27}{MJ} beweging. Bereken zijn \omterm{def:g10:energy-conservation:efficiency}{rendement} en de \omterm{def:g10:energy-conservation:energy}{energie} die als
warmte vrijkomt. Waar gaat die warmte heen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:7}
Een klifduiker van \qty{70}{kg} stapt van een klif van \qty{20}{m} (de
luchtweerstand is verwaarloosbaar). Bereken met het \omterm{thm:g10:energy-conservation:conservation}{behoud van energie} de
\omterm{def:g10:energy-conservation:kinetic}{kinetische energie} bij het water en de snelheid waarmee hij erin gaat, in
\unit{m/s} en in \unit{km/h}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:8}
Een \omterm{def:g10:orders-of-magnitude:unit}{meter} stond op \qty{41236}{kW.h} op 1 maart en op \qty{41562}{kW.h}
30 dagen later: hoeveel \omterm{def:g10:energy-conservation:energy}{energie} is verbruikt, wat is het daggemiddelde,
hoe hoog is de rekening bij $0.26$ euro per \unit{kW.h} plus $12$ euro
vast, en hoe groot is het gemiddelde \omterm{def:g10:energy-conservation:power}{vermogen} in \omterm{def:g10:energy-conservation:power}{watt}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:9}
Een oude gloeilamp van \qty{60}{W} zet slechts $5\%$ van haar verbruik om
in licht; een led geeft hetzelfde licht met \qty{9}{W}. Bereken het
lichtvermogen van de gloeilamp en het \omterm{def:g10:energy-conservation:efficiency}{rendement} van de led; bereken
daarna, bij \qty{4}{h} per dag, het jaarverbruik van elk en de besparing
tegen $0.25$ euro per \unit{kW.h}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:10}
Een stuwdam laat \qty{150}{m^3} water (\qty{1.5e5}{kg}) per seconde over
een val van \qty{40}{m} zakken. Bereken de \omterm{def:g10:energy-conservation:potential}{potentiële energie} die elke
seconde vrijkomt --- het beschikbare \omterm{def:g10:energy-conservation:power}{vermogen} --- en daarna het
elektrische \omterm{def:g10:energy-conservation:power}{vermogen} bij $\eta = 0.90$, en hoeveel huishoudens met een
gemiddelde van \qty{525}{W} daarmee bediend worden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:11}
Een slinger van \qty{0.20}{kg} wordt \qty{5.0}{cm} boven zijn laagste
punt losgelaten: bereken zijn snelheid daar. Na een paar minuten hangt
hij stil: waar is zijn \omterm{def:g10:energy-conservation:energy}{energie} gebleven, en waarom is dat niet in strijd
met \cref{thm:g10:energy-conservation:conservation}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:12}
Een kolencentrale ($\eta_1 = 0.38$), het net ($\eta_2 = 0.94$) en een
lader ($\eta_3 = 0.85$) laden een telefoonaccu van \qty{12}{W.h} op.
Bereken het totale \omterm{def:g10:energy-conservation:efficiency}{rendement}; de \omterm{def:g10:energy-conservation:forms}{chemische energie} die in de centrale
wordt verbruikt, in \unit{W.h} en in kilojoule; en de kolen
(\qty{25}{MJ/kg}) per oplaadbeurt, en daarna per jaar bij dagelijks
opladen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:13}
Een bliksemschicht draagt ongeveer \qty{5}{GJ}. Zet dat om naar
\omterm{def:g10:energy-conservation:kwh}{kilowattuur} en naar dagen van een huishouden dat \qty{12.6}{kW.h} per dag
gebruikt. De schicht wordt in ongeveer \qty{100}{\micro s} geleverd:
bereken het \omterm{def:g10:energy-conservation:power}{vermogen}, vergelijk het met het gemiddelde elektrische
\omterm{def:g10:energy-conservation:power}{vermogen} van de mensheid (\qty{3}{TW}), en leg uit waarom niemand
bliksem oogst.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:14}

vorm neemt ze aan, en waar? Een elektrische auto wint $60\%$ ervan terug

\section{Probleem: Een huis een dag lang van energie voorzien}

\begin{problem}[{Weekendopgave --- van kom cornflakes tot panelen op het
dak: een volledige energiedoorlichting van één gewone dag thuis, en wat
een joule kost naargelang wie hem verkoopt}]
\label{pb:g10:energy-conservation:1}
Op een gewone dinsdag gaat een gezin op zoek naar waar zijn elektriciteit
blijft, gewapend met dit hoofdstuk, de \omterm{def:g10:orders-of-magnitude:unit}{meter} in de gang en de achterkant
van de rekening. Het oordeel van vanavond: hun dag komt overeen met één
\omterm{def:g10:orders-of-magnitude:unit}{kilogram} benzine, zestien vierkante \omterm{def:g10:orders-of-magnitude:unit}{meter} zonneschijn --- of vijf
fietsers die dag en nacht doortrappen.

\medskip
\noindent\textbf{Deel I --- De doorlichting.}
De inventaris van de dag: koelkast \qty{100}{W} gedurende \qty{24}{h};
boiler \qty{2400}{W} gedurende \qty{2.5}{h}; oven \qty{2000}{W}
gedurende \qty{1.0}{h}; wasmachine \qty{500}{W} gedurende \qty{2.0}{h};
verlichting \qty{60}{W} gedurende \qty{5.0}{h}; schermen \qty{150}{W}
gedurende \qty{6.0}{h}.
\begin{enumerate}
  \item Bereken de \omterm{def:g10:energy-conservation:energy}{energie} van elk apparaat voor die dag, in
        \unit{kW.h}.
  \item Tel de dag op in \omterm{def:g10:energy-conservation:kwh}{kilowattuur} en daarna in megajoule; vergelijk
        met één \omterm{def:g10:orders-of-magnitude:unit}{kilogram} benzine (\qty{45}{MJ}).
  \item Bereken het gemiddelde \omterm{def:g10:energy-conservation:power}{vermogen} van het huis over de
        \qty{24}{h}; welk deel van een draaiende waterkoker is dat?
  \item Welk apparaat is in zijn eentje het grootst? Bereken het
        dagtotaal opnieuw als de boiler maar \qty{1.25}{h} had gedraaid.
  \item Sluimerlampjes trekken permanent \qty{15}{W}: hoeveel \omterm{def:g10:energy-conservation:energy}{energie} per
        dag, welk aandeel van het totaal, en welke jaarkosten bij $0.25$
        euro per \unit{kW.h}?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De rekening.}
\begin{enumerate}[resume]
  \item In 30 dagen liep de \omterm{def:g10:orders-of-magnitude:unit}{meter} van \qty{56214}{kW.h} naar
        \qty{56604}{kW.h}: hoeveel verbruik, en klopt het daggemiddelde
        met de doorlichting?
  \item Bereken de rekening: $0.25$ euro per \unit{kW.h} plus een vast
        abonnement van $14$ euro.
  \item Wat kost een megajoule van het net? En uit benzine, tegen $1.80$
        euro per liter die \qty{34}{MJ} bevat?
  \item Een kom cornflakes van \qty{1.5}{MJ} kost ongeveer $0.50$ euro:
        wat is de prijs per megajoule? Rangschik de drie verkopers van
        \omterm{def:g10:energy-conservation:energy}{joules}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het dak.}
\begin{enumerate}[resume]
  \item Vol zonlicht levert ongeveer \qty{1000}{W} per vierkante \omterm{def:g10:orders-of-magnitude:unit}{meter};
        panelen zetten dat om met $\eta = 0.20$: welk elektrisch \omterm{def:g10:energy-conservation:power}{vermogen}
        geeft \qty{1}{m^2} in de volle zon?
  \item De streek haalt gemiddeld het equivalent van \qty{4.0}{h} volle
        zon per dag: wat is de dagopbrengst van \qty{1}{m^2}?
  \item Welke paneeloppervlakte dekt de dag van \qty{12.6}{kW.h} van het
        gezin? Het zuiddak biedt \qty{20}{m^2}: past dat?
  \item In december zakt het equivalent tot \qty{1.5}{h}: hoeveel
        produceren de panelen dan, welk aandeel van de behoefte is dat,
        en wat vult het gat?
  \item Het maken van de panelen kostte ongeveer \qty{700}{kW.h} per
        vierkante \omterm{def:g10:orders-of-magnitude:unit}{meter}: hoe lang duurt de energetische terugverdientijd,
        uitgaande van vraag 11?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De kom cornflakes.}
\begin{enumerate}[resume]
  \item Een fiets met generator levert \qty{100}{W} elektriciteit:
        hoeveel uur trappen levert \qty{12.6}{kW.h} op --- hoeveel
        mensen die \qty{24}{h} onafgebroken trappen?
  \item Spieren draaien op $\eta \approx 0.25$: hoeveel voedselenergie
        zit er achter die \qty{12.6}{kW.h}, in megajoule en in kommen
        cornflakes?
  \item Wat kost de dag op de pedalen (kommen tegen $0.50$ euro),
        vergeleken met wat het net voor dezelfde dag rekent?
  \item Met hoeveel huisdagen komt een bliksemschicht van \qty{5}{GJ}
        overeen? Waarom kan hij, gegeven zijn duur van
        \qty{100}{\micro s}, het huis toch niet voeden?
  \item Joules gaan nooit verloren, en toch worden \omterm{def:g10:energy-conservation:energy}{joules} van het net,
        uit benzine en uit voedsel voor heel verschillende prijzen
        verkocht: waar betaal je eigenlijk voor?
  \item Slot: geef het oordeel over de dag in één zin --- één \omterm{def:g10:orders-of-magnitude:unit}{kilogram}
        benzine, zestien vierkante \omterm{def:g10:orders-of-magnitude:unit}{meter} zonnig dak of vijf fietsers rond
        de klok --- en noem de twee grootheden (één voorraad, één stroom)
        die je nooit meer zult verwarren.
\end{enumerate}
\end{problem}
