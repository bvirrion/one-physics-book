\chapter{Krachten en het traagheidsbeginsel}\label{ch:g10:inertia}

Stop met trappen en de fiets rolt uit tot stilstand; stop met trekken en
de slee blijft staan. De ervaring fluistert dat beweging met inspanning
gevoed moet worden --- en tweeduizend jaar lang gaf de natuurkunde haar
gelijk. Dit hoofdstuk bouwt de \omterm{def:g10:inertia:force}{kracht} op, een pijl gemeten in \omterm{def:g10:inertia:force}{newton}, en
formuleert de eerste grote wet van de mechanica: werkelijk met rust
gelaten houdt een bewegend lichaam zijn beweging, recht en gelijkmatig,
voor altijd.

\section{Een werking als kracht modelleren}

\begin{definition}[Kracht]\label{def:g10:inertia:force}
Een \emph{kracht}\index{kracht} modelleert een mechanische werking (een
duw, een trek, een aantrekking) die op een lichaam wordt uitgeoefend. Ze
wordt gekenmerkt door:
\begin{itemize}
  \item haar \emph{aangrijpingspunt}\index{aangrijpingspunt};
  \item haar richting --- de lijn waarlangs ze werkt en de zin op die
        lijn;
  \item haar grootte, gemeten in \emph{newton}\index{newton} (symbool
        \unit{N}).
\end{itemize}
Een kracht wordt getekend als een pijl $\vect F$ (een vector, zoals in
het wiskundevolume) vanuit haar aangrijpingspunt, met een lengte die bij
een aangegeven schaal evenredig is met de grootte.
\end{definition}

\begin{example}[Het gewicht van een schooltas]\label{ex:g10:inertia:schoolbag}
Een schooltas met massa $m = \qty{4.0}{kg}$ wordt door de Aarde omlaag
getrokken met haar \omterm{def:g10:universal-gravitation:weight}{gewicht} (\cref{ch:g10:universal-gravitation}):
$P = mg = 4.0 \times 9.81 \approx \qty{39}{N}$ --- één \omterm{def:g10:inertia:force}{newton} is
ongeveer het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een appel. Op schaal
$\qty{1}{cm} \leftrightarrow \qty{10}{N}$: een verticale pijl van
\qty{3.9}{cm}, aangrijpend in het zwaartepunt van de tas, omlaag
gericht.
\end{example}

\section{Een inventaris van krachten}

\begin{definition}[Contactkrachten en krachten op afstand]\label{def:g10:inertia:contact}
Een \emph{contactkracht}\index{contactkracht} wordt uitgeoefend op de
punten waar twee lichamen elkaar raken. Een \emph{kracht op
afstand}\index{kracht op afstand} heeft geen contact nodig: het \omterm{def:g10:universal-gravitation:weight}{gewicht}
--- de gravitatietrek uit \cref{ch:g10:universal-gravitation} --- is de
enige die in dit hoofdstuk voorkomt.
\end{definition}

\begin{definition}[De gebruikelijke contactkrachten]\label{def:g10:inertia:inventory}
Vier \omterm{def:g10:inertia:contact}{contactkrachten} dekken de meeste alledaagse situaties:
\begin{itemize}
  \item de \emph{normaalkracht}\index{normaalkracht} $\vect R$ van een
        steunvlak, loodrecht op dat vlak en ervan af gericht;
  \item de \emph{spankracht}\index{spankracht} $\vect T$ van een draad,
        touw of ketting, gericht langs het touw en er naartoe trekkend;
  \item de \emph{wrijving}\index{wrijving} $\vect f$ van een oppervlak,
        de lucht of het water, die het schuiven of de beweging
        tegenwerkt;
  \item de \emph{stuwkracht}\index{stuwkracht} van een motor, een
        straalpijp of een hand, die in de richting van de inspanning
        duwt.
\end{itemize}
\end{definition}

\begin{example}[Inventaris bij een getrokken slee]\label{ex:g10:inertia:sled}
Een slee die over de sneeuw wordt getrokken: \omterm{def:g10:universal-gravitation:weight}{gewicht} $\vect P$ (op
afstand), \omterm{def:g10:inertia:inventory}{normaalkracht} $\vect R$ van de sneeuw, \omterm{def:g10:inertia:inventory}{spankracht} $\vect T$ van
het touw, \omterm{def:g10:inertia:inventory}{wrijving} $\vect f$ die het schuiven tegenwerkt. Niets anders
raakt de slee: de inventaris is compleet --- \emph{altijd eerst het
\omterm{def:g10:universal-gravitation:weight}{gewicht}, daarna één \omterm{def:g10:inertia:contact}{contactkracht} per aanrakend voorwerp}.
\end{example}

\section{Het traagheidsbeginsel}

\begin{definition}[Eenparige rechtlijnige beweging]\label{def:g10:inertia:uniform}
Een lichaam voert in een stelsel een \emph{eenparige rechtlijnige
beweging}\index{eenparige rechtlijnige beweging} uit wanneer zijn \omterm{def:g10:relative-motion:trajectory}{baan}
een rechte lijn is die met constante snelheid wordt afgelegd: zijn
snelheidsvector $\vect v$ is op elk ogenblik dezelfde. Rust is het
bijzondere geval $\vect v = \vect 0$.
\end{definition}

\begin{definition}[Krachten in evenwicht]\label{def:g10:inertia:compensate}
De \omterm{def:g10:inertia:force}{krachten} op een lichaam zijn in \emph{evenwicht}\index{evenwicht}
wanneer hun gezamenlijke werking nul is: kop aan staart gelegd keren hun
pijlen naar het beginpunt terug. Twee \omterm{def:g10:inertia:force}{krachten} zijn precies dan in
evenwicht wanneer ze dezelfde werklijn hebben, even groot zijn en
tegengesteld gericht.
\end{definition}

\begin{theorem}[Traagheidsbeginsel]\label{thm:g10:inertia:principle}
\index{traagheidsbeginsel}\index{traagheid}
In het \omterm{def:g10:relative-motion:frames}{aardse referentiestelsel} blijft een lichaam waarop geen \omterm{def:g10:inertia:force}{kracht}
werkt, of waarop de \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht} zijn, in rust of houdt het een
\omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging} aan; omgekeerd werken op een lichaam dat
in rust is of een \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging} uitvoert, \omterm{def:g10:inertia:force}{krachten} in
\omterm{def:g10:inertia:compensate}{evenwicht} (of helemaal geen).
\end{theorem}
\admitted

\begin{remark}
Geen enkele proef kan een lichaam volledig isoleren, dus het beginsel is
een postulaat --- vermoed door Galilei, geformuleerd door Newton. Een
later hoofdstuk maakt er een wet van die \omterm{def:g10:inertia:force}{krachten} verbindt met
\emph{veranderingen} van de snelheid; het volume van bachelorjaar 1
vraagt in welke stelsels het beginsel werkelijk geldt.
\end{remark}

\begin{proposition}[Beweging uit krachten lezen, en omgekeerd]\label{prop:g10:inertia:converse}
In het \omterm{def:g10:relative-motion:frames}{aardse referentiestelsel} geldt:
\begin{enumerate}
  \item verandert de snelheid van een lichaam --- in grootte \emph{of} in
        richting --- dan zijn de \omterm{def:g10:inertia:force}{krachten} erop \emph{niet} in \omterm{def:g10:inertia:compensate}{evenwicht};
  \item is een lichaam in rust of in \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging}, dan
        moeten de geïnventariseerde \omterm{def:g10:inertia:force}{krachten} elkaar opheffen --- en dat
        legt onbekenden vast.
\end{enumerate}
\end{proposition}

\begin{proof}
De eerste uitspraak is de contrapositie van
\cref{thm:g10:inertia:principle}, de tweede het omgekeerde deel ervan,
toegepast op een volledige inventaris.
\end{proof}

\begin{example}[De ijshockeypuck]\label{ex:g10:inertia:puck}
Losgelaten door de stick legt een puck \qty{30}{m} glad ijs af in
\qty{2.5}{s}, rechtdoor met de constante snelheid
$v = 30/2.5 = \qty{12}{m/s}$. Inventaris: \omterm{def:g10:universal-gravitation:weight}{gewicht} en \omterm{def:g10:inertia:inventory}{normaalkracht} (de
\omterm{def:g10:inertia:inventory}{wrijving} is verwaarloosbaar). Ze zijn in \omterm{def:g10:inertia:compensate}{evenwicht}, en het beginsel
voorspelt precies de waargenomen beweging: niets duwt de puck vooruit
--- en dat hoeft ook niet.
\end{example}

\begin{remark}[Galilei tegen Aristoteles]
Aristoteles leerde dat elke beweging een beweger nodig heeft, en het
dagelijks leven --- \omterm{def:g10:inertia:inventory}{wrijving} bij elk contact --- lijkt hem gelijk te
geven. Galilei dacht de \omterm{def:g10:inertia:inventory}{wrijving} weg: hoe gladder het oppervlak, hoe
minder duw er nodig is om de beweging te \emph{onderhouden}. Volhouden,
niet stilstaan, is de natuurlijke toestand.
\end{remark}

\begin{remark}[De struikelende passagier]
Wanneer een bus hard remt, schieten staande passagiers naar voren. Geen
enkele \omterm{def:g10:inertia:force}{kracht} werpt hen vooruit: door hun traagheid houden hun lichamen
hun snelheid terwijl de bus de zijne verliest. De veiligheidsgordel
levert de achterwaartse \omterm{def:g10:inertia:force}{kracht} die in de inventaris ontbreekt.
\end{remark}

\section{Krachten in evenwicht: statica van een tekening}

\begin{definition}[Krachtenschema]\label{def:g10:inertia:diagram}
Het \emph{krachtenschema}\index{krachtenschema} van een lichaam toont het
lichaam alleen, met elke \omterm{def:g10:inertia:force}{kracht} die \emph{op} het lichaam werkt, op
schaal getekend vanuit haar \omterm{def:g10:inertia:force}{aangrijpingspunt}; \omterm{def:g10:inertia:force}{krachten} die \emph{door}
het lichaam worden uitgeoefend komen er nooit in voor.
\end{definition}

\begin{method}[Een statische situatie ontleden]\label{met:g10:inertia:method}
\begin{enumerate}
  \item Kies het lichaam dat je bestudeert; werk in het \omterm{def:g10:relative-motion:frames}{aardse
        referentiestelsel}.
  \item Inventariseer: altijd het \omterm{def:g10:universal-gravitation:weight}{gewicht}, en één \omterm{def:g10:inertia:contact}{contactkracht} per
        contact.
  \item Teken het \omterm{def:g10:inertia:diagram}{krachtenschema}.
  \item Het lichaam is in rust, dus zijn de \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht}
        (\cref{thm:g10:inertia:principle}): lees de onbekende groottes af
        van de tekening (gelijke lengten in één dimensie, componenten in
        twee).
\end{enumerate}
\end{method}

\begin{example}[Het boek op de tafel]\label{ex:g10:inertia:book}
Een boek met massa \qty{0.50}{kg} ligt op een tafel: \omterm{def:g10:universal-gravitation:weight}{gewicht}
$P = 0.50 \times 9.81 \approx \qty{4.9}{N}$ en \omterm{def:g10:inertia:inventory}{normaalkracht} $\vect R$.
Omdat het in rust is, houdt $\vect R$ het \omterm{def:g10:universal-gravitation:weight}{gewicht} $\vect P$ in \omterm{def:g10:inertia:compensate}{evenwicht}:
verticaal, omhoog, $R = \qty{4.9}{N}$. Leg er een tweede boek op en $R$
past zich aan --- een steunvlak duwt precies zo hard als nodig is.
\end{example}

\begin{example}[De lamp aan twee draden]\label{ex:g10:inertia:lamp}
Een lamp met massa \qty{3.0}{kg} ($P \approx \qty{29.4}{N}$) hangt aan
twee draden, elk onder $45^\circ$ met de verticaal. Door de symmetrie
zijn de \omterm{def:g10:inertia:inventory}{spankrachten} gelijk ($T_1 = T_2 = T$) en heffen hun horizontale
componenten elkaar op; de verticale componenten, elk $T \cos 45^\circ$,
dragen het \omterm{def:g10:universal-gravitation:weight}{gewicht}:
\[
2\,T \cos 45^\circ = P
\qquad\text{dus}\qquad
T = \frac{29.4}{2 \times 0.707} \approx \qty{21}{N}
\]
--- meer dan de helft van het \omterm{def:g10:universal-gravitation:weight}{gewicht} per draad: onder een hoek trekken
kost \omterm{def:g10:inertia:inventory}{spankracht}.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g10:inertia:1}
Inventariseer de \omterm{def:g10:inertia:force}{krachten} en zeg of het \omterm{def:g10:inertia:contact}{contactkrachten} of \omterm{def:g10:inertia:force}{krachten} op
afstand zijn, voor: (a) een boek dat op een tafel ligt; (b) een lamp die
aan haar draad hangt; (c) een puck die over glad ijs glijdt; (d) een
vallende appel (luchtweerstand verwaarloosbaar).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:2}
Bereken het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een boek van \qty{250}{g} en van een persoon van
\qty{60}{kg}. Hoe lang is bij schaal
$\qty{1}{cm} \leftrightarrow \qty{1}{N}$ de pijl van het boek, en waarom
is die schaal hopeloos voor de persoon?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:3}
Op een tekening met schaal $\qty{1}{cm} \leftrightarrow \qty{2}{N}$ is
een \omterm{def:g10:inertia:force}{kracht} een horizontale pijl van \qty{3.5}{cm} die naar links wijst en
in een punt $A$ aangrijpt. Geef haar drie kenmerken.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:4}
Een woordenboek met massa \qty{1.2}{kg} ligt op een plank. Teken het
\omterm{def:g10:inertia:diagram}{krachtenschema} en geef van elke \omterm{def:g10:inertia:force}{kracht} de grootte en de richting.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:5}
Juist of onjuist --- verbeter de onjuiste: (a) een lichaam komt altijd
uiteindelijk tot stilstand tenzij een \omterm{def:g10:inertia:force}{kracht} blijft duwen; (b) een
\omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging} vergt geen enkele niet-opgeheven \omterm{def:g10:inertia:force}{kracht};
(c) op een lichaam in rust werkt geen enkele \omterm{def:g10:inertia:force}{kracht}; (d) verandert de
snelheid van een lichaam, dan zijn de \omterm{def:g10:inertia:force}{krachten} erop niet in \omterm{def:g10:inertia:compensate}{evenwicht}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:6}
Na het loslaten legt een puck \qty{24}{m} ijs af in \qty{3.0}{s},
rechtdoor met constante snelheid. Bereken die snelheid; zijn de \omterm{def:g10:inertia:force}{krachten}
erop in \omterm{def:g10:inertia:compensate}{evenwicht}? Op beton komt dezelfde puck binnen enkele \omterm{def:g10:orders-of-magnitude:unit}{meters} tot
stilstand: wat is er in de inventaris veranderd, en wat besluit
\cref{prop:g10:inertia:converse}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:7}
Een passagier van \qty{70}{kg} staat in een lift die met de constante
snelheid \qty{1.5}{m/s} omhooggaat. Noem de \omterm{def:g10:inertia:force}{krachten} op de passagier;
zijn ze in \omterm{def:g10:inertia:compensate}{evenwicht}? Geef de grootte van de \omterm{def:g10:inertia:force}{kracht} van de vloer. Zou er
iets veranderen als de lift stilstond?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:8}
Een lamp met massa \qty{1.8}{kg} hangt in rust aan één enkele verticale
draad. Bereken de \omterm{def:g10:inertia:inventory}{spankracht} in de draad. De draad wordt vervangen door
twee \emph{verticale} draden die de last gelijk verdelen: hoe groot is
dan de \omterm{def:g10:inertia:inventory}{spankracht} in elk?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:9}
Een bus remt hard en een staande passagier schiet naar voren; in een
bocht met constante snelheid helt diezelfde passagier naar buiten. Leg
beide uit zonder een \omterm{def:g10:inertia:force}{kracht} ``naar voren'' of ``naar buiten'' te
verzinnen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:10}
Een parachutist met een totale massa van \qty{85}{kg} daalt verticaal
met de constante snelheid \qty{5.0}{m/s}. Teken het \omterm{def:g10:inertia:diagram}{krachtenschema} en
bereken de luchtweerstand op het scherm.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:11}
Een waterskiër wordt rechtdoor met constante snelheid getrokken; het
horizontale touw trekt met \qty{300}{N}. Hoe groot is de horizontale
weerstand van het water op de ski's (verantwoord dat)? Welke twee
verticale \omterm{def:g10:inertia:force}{krachten} houden elkaar in \omterm{def:g10:inertia:compensate}{evenwicht}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:12}
Een lamp met massa \qty{2.4}{kg} hangt aan twee draden onder $40^\circ$
met de verticaal ($\cos 40^\circ \approx 0.766$). Bereken elke
\omterm{def:g10:inertia:inventory}{spankracht}. Waarom groeit de \omterm{def:g10:inertia:inventory}{spankracht} naarmate de draden de
horizontaal naderen, en waarom zouden twee precies horizontale draden de
lamp nooit kunnen dragen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:13}
Een curlingsteen wordt geduwd, losgelaten, glijdt terwijl hij heel
geleidelijk vertraagt, en komt tot stilstand. Teken voor elke fase (duw,
glijden, rust) het \omterm{def:g10:inertia:diagram}{krachtenschema} en zeg of de \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht}
zijn, met een verwijzing naar \cref{thm:g10:inertia:principle} of
\cref{prop:g10:inertia:converse}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:14}
De sonde \emph{Voyager 1} zweeft met \qty{17}{km/s} door de
interstellaire ruimte, met uitgeschakelde motoren. Waarom heeft ze geen
brandstof nodig om die snelheid te houden, en wat zou Aristoteles
voorspellen? Hoever komt ze in een jaar, in \unit{km} en daarna in
astronomische eenheden ($\qty{1}{au} \approx \qty{1.496e8}{km}$)? Waarom
moet ze een stuwmotor gebruiken, zelfs om alleen van richting te
veranderen \emph{zonder} van snelheid te veranderen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:15}
Drie touwen trekken aan een ring die in rust is. Op een rooster waarvan
één vakje \qty{1}{N} voorstelt, leest men twee \omterm{def:g10:inertia:force}{krachten} af:
$\vect F_1\,(3, 2)$ en $\vect F_2\,(-1, 2)$. Bepaal de componenten, de
grootte en de richting van de derde \omterm{def:g10:inertia:force}{kracht}.
\end{exercise}

\section{Probleem: Traagheid in drie bedrijven}

\begin{problem}[{Weekendopgave --- het rolpad, de ijsbaan en de
parachute: constante snelheid als handtekening van krachten in
evenwicht, en waarom een parachute niet de weerstand maar de snelheid
verlaagt}]
\label{pb:g10:inertia:1}
Drie taferelen --- een koffer op een rolpad, een puck op een ijsbaan,
een parachutist in de avondlucht --- en één wet die in alle drie het
woord voert.

\medskip
\noindent\textbf{Deel I --- Het rolpad op de luchthaven.}
Een koffer van \qty{23}{kg} staat op een rolpad dat met de constante
snelheid \qty{0.75}{m/s} loopt.
\begin{enumerate}
  \item Welke beweging maakt de koffer in het aardse stelsel?
        Inventariseer de \omterm{def:g10:inertia:force}{krachten} erop.
  \item Zijn die \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht}? Verantwoord dat; bereken beide
        groottes.
  \item Vergelijk het \omterm{def:g10:inertia:diagram}{krachtenschema} met dat van dezelfde koffer op de
        vloer. Welke diepe uitspraak illustreert die vergelijking?
  \item Het rolpad komt met een schok tot stilstand. Beschrijf wat de
        koffer doet, zonder een \omterm{def:g10:inertia:force}{kracht} naar voren te verzinnen.
  \item De eigenares loopt met \qty{1.2}{m/s} ten opzichte van het
        rolpad. Wat is haar snelheid in het aardse stelsel? Zijn de
        \omterm{def:g10:inertia:force}{krachten} op \emph{haar} in \omterm{def:g10:inertia:compensate}{evenwicht}?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De ijsbaan.}
\begin{enumerate}[resume]
  \item Een puck van \qty{160}{g} glijdt rechtdoor met de constante
        snelheid \qty{8.0}{m/s}. Teken het \omterm{def:g10:inertia:diagram}{krachtenschema} met de
        groottes erbij.
  \item Aristoteles beweert dat de puck een beweger nodig heeft. Wat
        antwoordt de ijsbaan, en waarom lijken alledaagse oppervlakken
        hem gelijk te geven?
  \item De puck komt over een ruwe plek en vertraagt. Welke conclusie
        volgt over de \omterm{def:g10:inertia:force}{krachten} daar? Welke \omterm{def:g10:inertia:force}{kracht} is de schuldige?
  \item Een schaatsster maakt bij constante snelheid een vloeiende
        bocht. Zijn de \omterm{def:g10:inertia:force}{krachten} op haar in \omterm{def:g10:inertia:compensate}{evenwicht}? Pas op: wat voor
        soort grootheid is snelheid?
  \item Koppel schema aan beweging: (a) \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht}; (b) een
        extra \omterm{def:g10:inertia:force}{kracht} tegengesteld aan $\vect v$; (c) een extra \omterm{def:g10:inertia:force}{kracht}
        loodrecht op $\vect v$ --- (i) rechtdoor en vertragend; (ii)
        draaiend bij constante snelheid; (iii) \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige
        beweging}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De parachute.}
De totale massa van een springer, uitrusting inbegrepen, is
\qty{80}{kg}.
\begin{enumerate}[resume]
  \item Bij het verlaten van het vliegtuig zijn de verticale snelheid en
        de luchtweerstand nog nul. Bereken de \omterm{def:g10:inertia:force}{krachten}; zijn ze in
        \omterm{def:g10:inertia:compensate}{evenwicht}? Wat moet er nu gebeuren?
  \item De luchtweerstand groeit met de snelheid. Beschrijf de val
        zolang de weerstand kleiner is dan het \omterm{def:g10:universal-gravitation:weight}{gewicht}.
  \item Voordat het scherm opengaat, bereikt de springer een constante
        \qty{50}{m/s} (``eindsnelheid''). Bereken de luchtweerstand. Wat
        voor beweging is dit, en waarom is het ``traagheid in
        vermomming''?
  \item Het scherm gaat open: de weerstand overtreft nu het \omterm{def:g10:universal-gravitation:weight}{gewicht}
        ruimschoots. Wat gebeurt er met de snelheid? (De springer
        beweegt de hele tijd \emph{omlaag}.)
  \item Er wordt een nieuwe constante \qty{5.0}{m/s} bereikt. Bereken de
        luchtweerstand en vergelijk met vraag 13.
  \item De clou: wat heeft de parachute veranderd --- de uiteindelijke
        weerstand, of de snelheid waarbij de weerstand het \omterm{def:g10:universal-gravitation:weight}{gewicht}
        bereikt?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Vonnissen.}
\begin{enumerate}[resume]
  \item Vul aan en verantwoord: ``in rust of in \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige
        beweging} $\iff$ \dots''. Waarom verdient rust geen aparte wet?
  \item In welk bedrijf zou ``beweging heeft een beweger nodig'' van
        Aristoteles het zichtbaarst hebben gefaald, en waar zou Galilei
        naar wijzen?
  \item Een proefwerk beweert: ``de parachutist valt met constante
        snelheid, dus werkt er geen \omterm{def:g10:inertia:force}{kracht} op hem''. Verbeter dat in één
        regel.
  \item Slot: noem voor elk bedrijf het paar \omterm{def:g10:inertia:force}{krachten} dat in \omterm{def:g10:inertia:compensate}{evenwicht}
        is, en noem de ene wet die in deze opgave twintig keer is
        gebruikt.
\end{enumerate}
\end{problem}
