\chapter{Ordegrootten: het heelal opmeten}\label{ch:g10:orders-of-magnitude}

Natuurkunde begint waar de mening ophoudt: bij een meting. Maar wat de
natuurkunde meet, beslaat een absurd bereik --- een atoomkern meet
ongeveer \qty{e-15}{m}, het waarneembare heelal ongeveer \qty{e27}{m}:
een factor $10^{42}$, veel meer dan een voorstellingsvermogen aankan.
Dit hoofdstuk zet het gereedschap klaar dat dat bereik temt ---
eenheden, machten van tien, \omterm{def:g10:orders-of-magnitude:sigfig}{significante cijfers} --- en twee gewoonten
die op elke volgende bladzijde terugkeren: schrijf de \omterm{def:g10:orders-of-magnitude:unit}{eenheid} op, en
controleer de \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte}.

\section{Eenheden: de grammatica van het meten}

\begin{definition}[Natuurkundige grootheden en SI-eenheden]\label{def:g10:orders-of-magnitude:unit}
Een \emph{natuurkundige grootheid}\index{natuurkundige grootheid} is
een eigenschap van de wereld die je kunt meten: een lengte, een duur,
een massa. Meten is tellen hoe vaak een referentiegrootheid --- de
\emph{eenheid}\index{eenheid} --- erin past. Het
\emph{Internationaal Stelsel van Eenheden}\index{Internationaal Stelsel
van Eenheden (SI)} (SI) legt zeven basiseenheden vast; drie ervan
lopen door dit hele boek heen: de \emph{meter}\index{meter}
(\unit{m}) voor lengte, de \emph{kilogram}\index{kilogram} (\unit{kg})
voor massa en de \emph{seconde}\index{seconde} (\unit{s}) voor tijd.
(De vier andere --- ampère, kelvin, mol, candela --- komen erbij zodra
elektriciteit, warmte en stofhoeveelheid in het verhaal stappen.)
\end{definition}

\begin{remark}[Een getal zonder eenheid is geen resultaat]\label{rem:g10:orders-of-magnitude:units}
``De afstand is $384$'' zegt niets: \qty{384}{mm}, \qty{384}{km} en
\qty{384}{Mm} zijn drie verschillende werelden (de laatste is toevallig
de afstand tot de Maan). In 1999 verloor NASA de \emph{Mars Climate
Orbiter} van 327 miljoen dollar doordat het ene programma stuwkrachten
in pound-force berekende en het andere ze als newton las. Huisregel,
vanaf deze regel tot het einde van het boek: elk gemeten of berekend
getal draagt zijn \omterm{def:g10:orders-of-magnitude:unit}{eenheid}, op elke regel van de berekening.
\end{remark}

\begin{definition}[SI-voorvoegsels]\label{def:g10:orders-of-magnitude:prefixes}
Een \emph{voorvoegsel}\index{voorvoegsel (SI)} vóór een \omterm{def:g10:orders-of-magnitude:unit}{eenheid}
vermenigvuldigt die \omterm{def:g10:orders-of-magnitude:unit}{eenheid} met een vaste macht van tien:
\begin{center}
\begin{tabular}{llc@{\qquad\qquad}llc}
femto & f & $10^{-15}$ & centi & c & $10^{-2}$ \\
pico  & p & $10^{-12}$ & kilo  & k & $10^{3}$  \\
nano  & n & $10^{-9}$  & mega  & M & $10^{6}$  \\
micro & $\mu$ & $10^{-6}$ & giga & G & $10^{9}$ \\
milli & m & $10^{-3}$  & tera  & T & $10^{12}$ \\
\end{tabular}
\end{center}
Zo is $\qty{1}{nm} = \qty{e-9}{m}$ en $\qty{1}{km} = \qty{e3}{m}$.
\end{definition}

\begin{example}[Voorvoegsels lezen]\label{ex:g10:orders-of-magnitude:prefixes}
Groen licht heeft een golflengte van ongeveer
$\qty{500}{nm} = \qty{5.00e-7}{m}$; een mensenhaar is ongeveer
$\qty{70}{\micro m} = \qty{7.0e-5}{m}$ dik; de Maan draait op
$\qty{384}{Mm} = \qty{3.84e8}{m}$ om de Aarde. Een regendruppel heeft
een massa van ongeveer $\qty{30}{mg} = \qty{3.0e-5}{kg}$ --- merk op
dat de basiseenheid van massa als enige van de zeven zelf al een
\omterm{def:g10:orders-of-magnitude:prefixes}{voorvoegsel} draagt.
\end{example}

\section{Wetenschappelijke notatie en significante cijfers}

\begin{definition}[Wetenschappelijke notatie]\label{def:g10:orders-of-magnitude:scinot}
Een getal staat in \emph{wetenschappelijke notatie}\index{wetenschappelijke
notatie} wanneer het geschreven is als
\[
a \times 10^{n}, \qquad 1 \leq a < 10, \quad n \text{ geheel.}
\]
De factor $a$ heet de \emph{mantisse}\index{mantisse}, het gehele
getal $n$ de \emph{exponent}\index{exponent}.
Zo is $\num{149600000} = \num{1.496e8}$ en
$\num{0.00052} = \num{5.2e-4}$.
\end{definition}

\begin{proposition}[Rekenen met machten van tien]\label{prop:g10:orders-of-magnitude:powers}
Voor alle gehele getallen $m$ en $n$:
\[
10^{m} \times 10^{n} = 10^{m+n}, \qquad
\frac{1}{10^{n}} = 10^{-n}, \qquad
\left(10^{m}\right)^{n} = 10^{mn}.
\]
\end{proposition}

\begin{proof}[Gedeeltelijk bewijs]
Voor positieve \omterm{def:g10:orders-of-magnitude:scinot}{exponenten} volstaat het de factoren te tellen:
$10^{m} \times 10^{n}$ is een product van $m + n$ tientallen, en
$(10^m)^n$ rijgt $n$ groepen van $m$ tientallen aaneen. Dat we
$10^{-n}$ schrijven voor $1/10^{n}$ is precies de afspraak die de
eerste regel voor alle gehele getallen waar houdt: ze dwingt
$10^{n} \times 10^{-n} = 10^{0} = 1$ af. Het bijbehorende
wiskundevolume behandelt de rekenregels voor \omterm{def:g10:orders-of-magnitude:scinot}{exponenten} volledig.
\end{proof}

\begin{example}[Vermenigvuldigen in wetenschappelijke notatie]\label{ex:g10:orders-of-magnitude:scinot}
Hoeveel Aardes passen er, naar massa, in de Zon? Met
$m_{\text{Zon}} = \qty{1.99e30}{kg}$ en
$m_{\text{Aarde}} = \qty{5.97e24}{kg}$:
\[
\frac{\num{1.99e30}}{\num{5.97e24}}
= \frac{1.99}{5.97} \times 10^{30-24}
= 0.333 \times 10^{6}
= \num{3.33e5}.
\]
\omterm{def:g10:orders-of-magnitude:scinot}{Mantisses} en \omterm{def:g10:orders-of-magnitude:scinot}{exponenten} worden apart behandeld; het eindantwoord wordt
opnieuw genormeerd zodat de \omterm{def:g10:orders-of-magnitude:scinot}{mantisse} weer in $[1, 10)$ terechtkomt.
\end{example}

\begin{definition}[Significante cijfers]\label{def:g10:orders-of-magnitude:sigfig}
De \emph{significante cijfers}\index{significante cijfers} van een
gemeten getal zijn de cijfers waarvoor de meting werkelijk instaat:
alle opgeschreven cijfers behalve de nullen vooraan. Nullen achteraan
tellen wel mee --- $\qty{5.20e-3}{s}$ (drie significante cijfers)
belooft meer dan $\qty{5.2e-3}{s}$ (twee). De \omterm{def:g10:orders-of-magnitude:scinot}{wetenschappelijke
notatie} laat het aantal in één oogopslag zien: de \omterm{def:g10:orders-of-magnitude:scinot}{mantisse} draagt
precies de significante cijfers.
\end{definition}

\begin{method}[Hoeveel cijfers je overhoudt]\label{met:g10:orders-of-magnitude:sigfig}
Een berekend resultaat kan niet nauwkeuriger zijn dan zijn
ingrediënten.
\begin{enumerate}
  \item Houd bij een product of quotiënt evenveel \omterm{def:g10:orders-of-magnitude:sigfig}{significante cijfers}
        over als de \emph{minst} nauwkeurige factor er heeft.
  \item Rond alleen het eindantwoord af, nooit de tussenstappen.
  \item Bij twijfel: drie \omterm{def:g10:orders-of-magnitude:sigfig}{significante cijfers} is de werkprecisie van
        dit boek.
\end{enumerate}
\end{method}

\begin{example}[De omtrek van de Aarde]\label{ex:g10:orders-of-magnitude:sigfig}
De diameter van de Aarde is $d = \qty{1.27e7}{m}$ (drie \omterm{def:g10:orders-of-magnitude:sigfig}{significante
cijfers}), dus haar omtrek is
$\pi d = \num{3.14159} \times \qty{1.27e7}{m} = \qty{3.99e7}{m}$ ---
drie cijfers, niet de acht die de rekenmachine toont. Wie
$\qty{39898227}{m}$ opschrijft, beweert de planeet op de \omterm{def:g10:orders-of-magnitude:unit}{meter}
nauwkeurig te kennen.
\end{example}

\section{De ladder van het heelal}

\begin{definition}[Ordegrootte]\label{def:g10:orders-of-magnitude:oom}
De \emph{ordegrootte}\index{ordegrootte} van een grootheid is de macht
van tien die er het dichtst bij ligt: schrijf de grootheid als
$a \times 10^{n}$ met $1 \leq a < 10$ en neem $10^{n}$ als $a < 5$,
en $10^{n+1}$ als $a \geq 5$. (De precieze grensafspraak doet er nooit
toe: een uitspraak over ordegrootte geldt sowieso maar op een factor
tien na.)
\end{definition}

\begin{example}[Ordegrootten]\label{ex:g10:orders-of-magnitude:oom}
Een mensenleven duurt ongeveer $80$ jaar, dus
$80 \times \qty{3.16e7}{s} \approx \qty{2.5e9}{s}$: \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte}
$\qty{e9}{s}$ --- een paar miljard seconden, besteed ze goed. Een mens
is $\qty{1.7}{m}$ lang: orde $\qty{e0}{m}$. De Mount Everest,
$\qty{8848}{m}$: orde $\qty{e4}{m}$, want $8.848 \geq 5$.
\end{example}

Zulke schattingen op \omterm{def:g10:orders-of-magnitude:scinot}{exponent} gestapeld leveren het hoofdbeeld van dit
hoofdstuk op: een ladder waarvan elke sport een wereld is die tien keer
wijder is dan de sport eronder.

\begin{definition}[Astronomische eenheid en lichtjaar]\label{def:g10:orders-of-magnitude:lightyear}
De \emph{astronomische eenheid}\index{astronomische eenheid}
($\qty{1}{au} = \qty{1.496e11}{m}$) is de gemiddelde afstand
Aarde--Zon. Het \emph{lichtjaar}\index{lichtjaar} (\unit{ly}) is de
afstand die het licht in één jaar aflegt. Licht beweegt in vacuüm met
\[
c = \qty{3.00e8}{m/s},
\]
de grootste snelheid die de natuur toelaat --- het lichtjaar is dus een
afstand, geen tijd.
\end{definition}

\begin{example}[Het lichtjaar in meters]\label{ex:g10:orders-of-magnitude:ly}
Een jaar telt $365.25 \times 24 \times 3600 = \qty{3.156e7}{s}$
(prettig dicht bij $\pi \times 10^{7}\,\unit{s}$). Dus
\[
\qty{1}{ly} = c \times \qty{3.156e7}{s}
= \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}.
\]
Proxima Centauri, de dichtstbijzijnde ster na de Zon, staat op
$\qty{4.2}{ly} \approx \qty{4.0e16}{m}$: licht dat de afstand
Aarde--Zon in acht minuten oversteekt, heeft van daar ruim vier jaar
nodig.
\end{example}

\begin{method}[Controle op ordegrootte]\label{met:g10:orders-of-magnitude:sanity}
Voordat je een berekend resultaat vertrouwt:
\begin{enumerate}
  \item rond elk gegeven af op zijn \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte};
  \item doe de berekening opnieuw met alleen machten van tien, met
        \cref{prop:g10:orders-of-magnitude:powers} --- dat kost
        seconden;
  \item vergelijk: het nauwkeurige antwoord moet tot op ongeveer een
        factor tien met de schatting kloppen, en de \omterm{def:g10:orders-of-magnitude:unit}{eenheid} moet de
        juiste zijn. Faalt een van beide controles, spoor dan eerst de
        fout op voordat je verdergaat.
\end{enumerate}
\end{method}

\begin{remark}[Waarom die moeite]\label{rem:g10:orders-of-magnitude:sanity}
Een rekenmachine meldt zonder blozen dat een appel een massa van
$\qty{4e8}{kg}$ heeft --- ze kan niet beter weten. Geen enkele formule
signaleert het absurde; de gewoonte doet dat wel. Natuurkundigen
schatten eerst en rekenen daarna: de schatting vangt verschoven
decimale punten, verwisselde eenheden en omgekeerde breuken op, de drie
fouten achter de meeste verkeerde antwoorden, op elk niveau.
\end{remark}

\section{Meten en onzekerheid}

Geen enkel instrument leest de ware waarde af. Een meting is pas
eerlijk als ze er ook bij vermeldt hoever ze ernaast kan zitten.

\begin{definition}[Absolute en relatieve onzekerheid]\label{def:g10:orders-of-magnitude:uncertainty}
Een meting van een grootheid $x$ wordt genoteerd als
\[
x = x_{\text{m}} \pm \Delta x ,
\]
waarbij $x_{\text{m}}$ de gemeten waarde is en de \emph{absolute
onzekerheid}\index{onzekerheid!absolute} $\Delta x > 0$, in dezelfde
\omterm{def:g10:orders-of-magnitude:unit}{eenheid}, de aannemelijke fout begrenst: de ware waarde ligt naar
verluidt tussen $x_{\text{m}} - \Delta x$ en $x_{\text{m}} + \Delta x$.
De \emph{relatieve onzekerheid}\index{onzekerheid!relatieve} is de
verhouding $\Delta x / x_{\text{m}}$, meestal in procent gegeven; zij
is wat het woord ``nauwkeurig'' meet.
\end{definition}

\begin{method}[Een schaalverdeling aflezen]\label{met:g10:orders-of-magnitude:reading}
Voor een liniaal, een maatcilinder, een wijzerplaat:
\begin{enumerate}
  \item lees de streep af die het dichtst bij het merkteken ligt, met
        het oog loodrecht op de schaal (een scheve blik verschuift de
        aflezing);
  \item neem minstens de helft van de kleinste schaaldeling als
        \omterm{def:g10:orders-of-magnitude:uncertainty}{absolute onzekerheid} --- meer als de rand van het voorwerp of
        het vloeistofoppervlak vaag is;
  \item schrijf het resultaat als $x_{\text{m}} \pm \Delta x$ met zijn
        \omterm{def:g10:orders-of-magnitude:unit}{eenheid}, en houd geen cijfer over voorbij het onzekere cijfer.
\end{enumerate}
\end{method}

\begin{example}[Relatieve nauwkeurigheid]\label{ex:g10:orders-of-magnitude:reading}
De aflezing hierboven heeft \omterm{def:g10:orders-of-magnitude:uncertainty}{relatieve onzekerheid}
$\frac{0.5}{127} \approx 0.4\%$. Dezelfde liniaal geeft bij een bout
van $\qty{12}{mm}$ juist $\frac{0.5}{12} \approx 4\%$: hetzelfde
instrument is tien keer minder nauwkeurig op het kleine voorwerp.
Nauwkeurigheid is een eigenschap van de \emph{meting}, niet van het
gereedschap.
\end{example}

\begin{remark}[Klein absoluut, enorm relatief --- en omgekeerd]\label{rem:g10:orders-of-magnitude:precision}
Laserstations meten de afstand Aarde--Maan, $\qty{3.84e8}{m}$, tot op
ongeveer $\pm\qty{3}{cm}$: een \omterm{def:g10:orders-of-magnitude:uncertainty}{relatieve onzekerheid} van $10^{-10}$,
een van de scherpste metingen ooit gedaan --- met een absolute fout
waar geen liniaal over zou opscheppen. Vraag je altijd af:
\emph{relatief ten opzichte waarvan}? Het paar
$(\Delta x,\ \Delta x / x_{\text{m}})$ vertelt het hele verhaal, en
elk van beide getallen alleen kan misleiden.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:5}

\begin{enumerate}

  \item Bereken de \omterm{def:g10:orders-of-magnitude:uncertainty}{relatieve onzekerheid}.

\begin{exercise}[$\star\star$]\label{exo:g10:orders-of-magnitude:6}

\begin{exercise}[$\star\star\star$]\label{exo:g10:orders-of-magnitude:12}

\section{Probleem: Van het atoom tot Andromeda}

\begin{problem}[{Weekendopgave --- één zandkorrel bepaalt de maat van
het atoom, het zonnestelsel, de melkweg en het waarneembare heelal, en
plaatst jou op de ladder daartussen}]
\label{pb:g10:orders-of-magnitude:1}
Een zandkorrel is het nederigste voorwerp uit de natuurkunde, en dit
weekend maakt hij overuren: eerst tellen we zijn atomen, dan krimpen we
de Zon tot zijn afmeting en stappen we de kosmos af, dan laten we het
licht het landmeten doen, en ten slotte tellen we de atomen in
\emph{jou}. Gegevens, overal te gebruiken: korreldiameter
$\qty{0.5}{mm}$; atoomdiameter $\qty{2.5e-10}{m}$; dichtheid van zand
$\qty{2500}{kg/m^3}$; diameters: Zon $\qty{1.39e9}{m}$, Aarde
$\qty{1.27e7}{m}$, Melkweg $\qty{9.5e20}{m}$, waarneembaar heelal
$\qty{8.8e26}{m}$; afstanden vanaf ons: Maan $\qty{3.84e8}{m}$, Zon
$\qty{1.496e11}{m}$, Neptunus $\qty{4.5e12}{m}$, Proxima Centauri
$\qty{4.0e16}{m}$, Andromedastelsel $\qty{2.4e22}{m}$;
$c = \qty{3.00e8}{m/s}$; een watermolecuul heeft massa
$\qty{3.0e-26}{kg}$ en bevat drie atomen.

\medskip
\noindent\textbf{Deel I --- Anatomie van de korrel.}
\begin{enumerate}
  \item Schrijf de korreldiameter en de atoomdiameter in \omterm{def:g10:orders-of-magnitude:unit}{meters}, in
        \omterm{def:g10:orders-of-magnitude:scinot}{wetenschappelijke notatie}, en noem het \omterm{def:g10:orders-of-magnitude:prefixes}{SI-voorvoegsel} waar elk
        het dichtst bij ligt.
  \item Hoeveel atomen liggen er zij aan zij over één korreldiameter?
  \item Modelleer de korrel als een kubus met ribbe $\qty{0.5}{mm}$,
        volgepakt met atomen. Schat het aantal atomen in de korrel, als
        \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte}.
  \item Bereken het volume van die kubus en daarna de massa van de
        korrel. Druk de massa in milligram uit.
  \item Toets het model: deel de massa van de korrel door het aantal
        atomen en vergelijk met de bekende \omterm{def:g10:orders-of-magnitude:oom}{ordegrootte} van
        atoommassa's (een paar $10^{-26}\,\unit{kg}$). Oordeel?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De Zon als zandkorrel.}
We bouwen een schaalmodel van de kosmos waarin de Zon de korrel is:
elke modelafmeting is de werkelijke afmeting maal een factor $s$.
\begin{enumerate}[resume]
  \item Bereken de schaalfactor $s$.
  \item Bepaal in het model de diameter van de Aarde en haar afstand
        tot de korrelzon. Welk voorwerp uit de wereld van deel I heeft
        ongeveer de afmeting van de modelaarde?
  \item Op welke afstand van de korrel draait de modelneptunus? Welk
        meubelstuk zou het hele planetenstelsel herbergen?
  \item Hoe ver weg staat de modelster Proxima Centauri?
  \item Twee zandkorrels, $\qty{14}{km}$ uit elkaar, en er bijna niets
        tussenin: zeg in een of twee zinnen wat het model onthult over
        hoe leeg de melkweg is.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het licht doet het landmeten.}
\begin{enumerate}[resume]
  \item Hoe lang doet licht erover om de echte zandkorrel over te
        steken?
  \item Hoe lang doet licht erover om ons te bereiken vanaf de Maan, en
        vanaf de Zon?
  \item Bereken het \omterm{def:g10:orders-of-magnitude:lightyear}{lichtjaar} opnieuw in \omterm{def:g10:orders-of-magnitude:unit}{meters} uit $c$ en de lengte
        van het jaar, en ga na dat de afstand tot Proxima uit de
        gegevens overeenkomt met ongeveer $\qty{4.2}{ly}$.
  \item Zet de afstand tot Andromeda om in lichtjaren. Wanneer
        vertrok, ruwweg, het licht dat nu je oog binnenkomt uit
        Andromeda, en wat liep er in die tijd op Aarde rond?
  \item Hoe ver reist licht in één nanoseconde? Een processor met een
        klok van $\qty{3}{GHz}$ start elke derde nanoseconde een
        bewerking: leg in één zin uit waarom een snelle computer een
        kleine computer moet zijn.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Jij, tussen atoom en Andromeda.}
\begin{enumerate}[resume]
  \item Een mens is ruwweg $\qty{70}{kg}$ water. Schat met de gegevens
        de gemiddelde massa van één atoom in water, en daarna het
        aantal atomen in een mens.
  \item Vergelijk dat aantal met de atomen in de korrel (vraag 3) en
        met de ruwweg $2\times10^{23}$ sterren van het waarneembare
        heelal. Geef beide factoren.
  \item Sta je op de ladder dichter bij het atoom of bij het
        waarneembare heelal? Vergelijk de verhoudingen
        $\text{(jij)}/\text{(atoom)}$ en
        $\text{(heelal)}/\text{(jij)}$, met $\qty{1.7}{m}$ voor jou.
  \item Bepaal de afmeting $x$ die precies halverwege de ladder ligt
        tussen de atoomkern ($\qty{e-15}{m}$) en het waarneembare
        heelal: $x$ moet voldoen aan $x / 10^{-15} = 10^{26.9} / x$
        (met $\qty{8.8e26}{m} \approx 10^{26.9}\,\unit{m}$). Welk
        voorwerp uit het zonnestelsel heeft toevallig precies die
        afmeting?
  \item De clou. Krimp het waarneembare heelal met de factor $s$ uit
        deel II, zodat de Zon een zandkorrel wordt. Geef de diameter
        van het modelheelal in \omterm{def:g10:orders-of-magnitude:unit}{meters} en daarna in astronomische
        eenheden, en besluit in één zin: zelfs met elke ster tot
        zandkorrel gekrompen loopt het heelal over elke menselijke
        schaal heen --- alleen machten van tien houden het bijeen.
\end{enumerate}
\end{problem}
