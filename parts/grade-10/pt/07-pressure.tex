\chapter{Pressão: do esporte ao mergulho}\label{ch:g10:pressure}

Um esquiador desliza sobre a neve em que um caminhante afunda; um
empurrão de \qty{5}{N} enfia uma agulha num couro que um soco não
consegue amassar. Mesmas \omterm{def:g10:inertia:force}{forças}, efeitos completamente diferentes: a
grandeza que falta é a \emph{\omterm{def:g10:pressure:pressure}{pressão}}, a \omterm{def:g10:inertia:force}{força} dividida pela área que a
recebe. Este capítulo a define, segue-a piscina abaixo, barômetro acima,
para dentro de uma seringa comprimida --- e termina a quarenta \omterm{def:g10:orders-of-magnitude:unit}{metros}
sob o mar, onde ela se torna vital.

\section{Pressão: força espalhada por uma área}

\begin{definition}[Pressão]\label{def:g10:pressure:pressure}
Quando uma \omterm{def:g10:inertia:force}{força} de módulo $F$ (em \unit{N}) pressiona
perpendicularmente uma superfície de área $S$ (em \unit{m^2}), a
\emph{pressão}\index{pressão} exercida sobre essa superfície é
\[
P = \frac{F}{S}.
\]
A sua \omterm{def:g10:orders-of-magnitude:unit}{unidade}, o \unit{N/m^2}, é o \emph{pascal}\index{pascal}
(\unit{Pa}) --- uma \omterm{def:g10:orders-of-magnitude:unit}{unidade} minúscula, o \omterm{def:g10:universal-gravitation:weight}{peso} de uma maçã espalhado por
um \omterm{def:g10:orders-of-magnitude:unit}{metro} quadrado, de modo que na prática se usam o \unit{kPa} e o
\unit{MPa}.
\end{definition}

\begin{example}[Salto agulha contra raquete de neve]\label{ex:g10:pressure:snowshoe}
Uma pessoa de \qty{60}{kg} pesa $F = 60 \times 9.81 \approx \qty{590}{N}$.
Sobre um único salto agulha de \qty{1.0}{cm^2} = \qty{1.0e-4}{m^2}:
$P = 590/\num{1.0e-4} \approx \qty{5.9e6}{Pa} = \qty{5.9}{MPa}$. Sobre
duas raquetes de neve de \qty{0.20}{m^2} cada:
$P = 590/0.40 \approx \qty{1.5}{kPa}$. O \emph{mesmo} \omterm{def:g10:universal-gravitation:weight}{peso} pressiona
$4000$ vezes mais forte sob o salto: saltos marcam pisos de madeira,
raquetes flutuam sobre a neve fofa.
\end{example}

\begin{remark}[Concentrar ou espalhar]\label{rem:g10:pressure:tools}
Uma agulha de costura empurrada com \qty{5}{N} sobre uma ponta de
\qty{0.01}{mm^2} = \qty{1e-8}{m^2} exerce
$\qty{5e8}{Pa} = \qty{500}{MPa}$ --- o bastante para separar as fibras do
couro. As ferramentas de furar concentram \omterm{def:g10:inertia:force}{força}; os projetos que
sustentam carga (esquis, esteiras de trator, fundações) a espalham. O
esporte é gestão aplicada de \omterm{def:g10:pressure:pressure}{pressão}: os patins mordem o gelo, os esquis
deslizam sobre a neve.
\end{remark}

\section{Pressão num líquido em repouso}

Por que a água pressiona mais forte lá embaixo? Isole uma coluna vertical
imaginária de água, de seção $S$, da superfície até a profundidade $h$:
volume $Sh$, massa $\rho S h$ ($\rho$ é a densidade do líquido, em
\unit{kg/m^3}), \omterm{def:g10:universal-gravitation:weight}{peso} $\rho S h g$. A base da coluna sustenta esse \omterm{def:g10:universal-gravitation:weight}{peso}
\emph{mais} o empurrão $P_0 S$ da atmosfera sobre o topo. Dividindo a
\omterm{def:g10:inertia:force}{força} total por $S$: a \omterm{def:g10:pressure:pressure}{pressão} à profundidade $h$ deveria ser
$P_0 + \rho g h$ --- e é, em toda parte do líquido.

\begin{theorem}[Pressão em função da profundidade]\label{thm:g10:pressure:depth}
Num líquido de densidade $\rho$ em repouso, aberto ao ar, a \omterm{def:g10:pressure:pressure}{pressão} à
profundidade $h$ abaixo da superfície é
\[
P = P_0 + \rho g h,
\]
em que $P_0$ é a \omterm{def:g10:pressure:atmospheric}{pressão atmosférica} na superfície. A \omterm{def:g10:pressure:pressure}{pressão} depende só
da profundidade, e não da forma do recipiente, e num dado ponto o líquido
pressiona com a mesma intensidade em todas as direções.
\end{theorem}
\admitted

\begin{remark}
O argumento da coluna só torna a lei plausível; a dedução honesta
(qualquer forma, todas as direções) está no volume do primeiro ano de
graduação. O termo próprio da água, $\rho g h$, é a \omterm{def:g10:pressure:pressure}{pressão}
\emph{manométrica}\index{pressão manométrica} --- o que o manômetro de um
mergulhador exibe.
\end{remark}

\begin{example}[Dez metros de água]\label{ex:g10:pressure:tenmetres}
Em água doce ($\rho = \qty{1000}{kg/m^3}$), a $h = \qty{10}{m}$:
$\rho g h = 1000 \times 9.81 \times 10 \approx \qty{98}{kPa}$, logo
$P = 101 + 98 \approx \qty{199}{kPa}$. Dez \omterm{def:g10:orders-of-magnitude:unit}{metros} de água somam quase
exatamente uma atmosfera: é a regra prática do mergulhador, \emph{uma
atmosfera a mais a cada dez \omterm{def:g10:orders-of-magnitude:unit}{metros}}.
\end{example}

\section{O oceano de ar}

\begin{definition}[Pressão atmosférica]\label{def:g10:pressure:atmospheric}
Vivemos no fundo de um oceano de ar, cujo \omterm{def:g10:universal-gravitation:weight}{peso} cria a \emph{pressão
atmosférica}\index{pressão atmosférica}, ao nível do mar
\[
P_0 \approx \qty{101}{kPa} = \qty{1.01e5}{Pa}.
\]
Sobre cada centímetro quadrado isso vale
$F = \num{1.01e5} \times \num{1e-4} \approx \qty{10}{N}$: o \omterm{def:g10:universal-gravitation:weight}{peso} de uma
massa de um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} sobre cada \unit{cm^2} da sua pele, da sua
escrivaninha, de tudo.
\end{definition}

\begin{example}[A carga invisível]\label{ex:g10:pressure:sheet}
Uma folha A4 (\qty{21.0}{cm} $\times$ \qty{29.7}{cm}, logo
$S \approx \qty{0.0624}{m^2}$) recebe do ar acima dela
$F = \num{1.01e5} \times 0.0624 \approx \qty{6.3e3}{N}$ --- o \omterm{def:g10:universal-gravitation:weight}{peso} de um
carro pequeno. Nada se rasga: o ar de baixo empurra \emph{para cima} com
a mesma intensidade. Só percebemos a atmosfera quando um dos lados a
perde --- uma ventosa, um canudo, a cabine de um avião.
\end{example}

\begin{remark}[O barômetro de Torricelli]\label{rem:g10:pressure:torricelli}
Em 1643, Torricelli inverteu um tubo selado de mercúrio sobre uma cuba de
mercúrio. A coluna desceu até a altura em que $\rho g h = P_0$:
$h = \num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$, deixando
acima dela o primeiro vácuo já produzido. A altura acompanha o tempo e
diminui com a altitude: um barômetro pesa o ar que está acima de você
(com água seriam necessários \qty{10.3}{m}: \cref{exo:g10:pressure:9}).
\end{remark}

\section{Gases: bombardeio e lei de Boyle}

Um gás também pressiona o seu recipiente: ele é um enxame de moléculas em
movimento desordenado incessante, cada colisão dá à parede um empurrãozinho
para fora, e bilhões de bilhões de impactos por segundo se fundem na
\omterm{def:g10:inertia:force}{força} constante que medimos como \omterm{def:g10:pressure:pressure}{pressão} do gás. Comprima o gás até
metade do volume: as moléculas batem com o dobro da frequência, de modo
que a \omterm{def:g10:pressure:pressure}{pressão} deveria dobrar. E dobra.

\begin{proposition}[Lei de Boyle]\label{prop:g10:pressure:boyle}
\index{lei de Boyle}
Para uma quantidade fixa de gás mantida a temperatura constante, \omterm{def:g10:pressure:pressure}{pressão}
e volume são inversamente proporcionais:
\[
P \, V = \text{constante},
\qquad\text{ou seja,}\qquad
P_1 V_1 = P_2 V_2 .
\]
\end{proposition}
\admitted

\begin{remark}
Neste nível trata-se de um fato experimental --- o volume do primeiro ano
de graduação a deduz a partir da imagem das colisões. A experiência (uma
seringa de ar selada, um sensor de \omterm{def:g10:pressure:pressure}{pressão}, \qty{20}{\degreeCelsius}):
\begin{center}
\begin{tabular}{lccccc}
$V$ (\unit{mL}) & 60 & 50 & 40 & 30 & 20 \\
$P$ (\unit{kPa}) & 100 & 120 & 150 & 200 & 300 \\
$P \times V$ (\unit{kPa.mL}) & 6000 & 6000 & 6000 & 6000 & 6000
\end{tabular}
\end{center}
\end{remark}

\begin{example}[Uma seringa comprimida]\label{ex:g10:pressure:boyle}
O ar ocupa \qty{4.0}{L} a \qty{100}{kPa}; comprimido devagar (para que a
temperatura fique constante) até \qty{250}{kPa}, ele ocupa
$V_2 = P_1 V_1 / P_2 = 400/250 = \qty{1.6}{L}$.
\end{example}

\section{Mergulho: a pressão posta a trabalhar}

\begin{method}[Pressão absoluta em profundidade]\label{met:g10:pressure:dive}
Para um mergulhador à profundidade $h$ no mar
($\rho \approx \qty{1025}{kg/m^3}$, de modo que
$\rho g h \approx \qty{101}{kPa}$ a cada \qty{10}{m}), calcule
$P = P_0 + \rho g h$ --- e, na \omterm{prop:g10:pressure:boyle}{lei de Boyle}, use sempre essa \omterm{def:g10:pressure:pressure}{pressão}
\emph{absoluta}\index{pressão absoluta}, nunca só o termo manométrico. Um
mergulhador está sob $2$ atmosferas a \qty{10}{m}, $3$ a \qty{20}{m} e
$5$ a \qty{40}{m}.
\end{method}

\begin{remark}[Nunca prenda a respiração]\label{rem:g10:pressure:breath}
Um regulador entrega ar à \omterm{def:g10:pressure:pressure}{pressão} ambiente: os pulmões se enchem
normalmente em qualquer profundidade. Mas encha \qty{6.0}{L} a
\qty{20}{m} ($\qty{302}{kPa}$), prenda essa respiração e suba: Boyle
exige $6.0 \times 302/101 \approx \qty{18}{L}$ --- três vezes o que um
tórax comporta, e os pulmões se rasgam bem antes disso, mesmo com poucos
\omterm{def:g10:orders-of-magnitude:unit}{metros} de subida. Daí a primeira regra do mergulho autônomo: \emph{nunca
prenda a respiração na subida} --- continue respirando e o ar em expansão
simplesmente sai. Uma apneísta está segura: o ar dela foi tomado na
superfície, de modo que, na subida, ele apenas volta a se expandir até o
volume original.
\end{remark}

\begin{remark}[Descompressão]\label{rem:g10:pressure:deco}
Respirar ar a alta \omterm{def:g10:pressure:pressure}{pressão} também dissolve nitrogênio extra no sangue,
como o gás numa garrafa de refrigerante fechada; suba rápido demais e ele
efervesce em bolhas dentro do corpo --- é a doença descompressiva. Daí a
subida lenta (cerca de \qty{10}{m} por minuto) e as paradas rasas, que
deixam o gás sair sem alarde pelos pulmões. A teoria espera o volume do
primeiro ano de graduação; a regra de conduta é de todo mergulhador.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:pressure:1}
Um engradado de \qty{600}{N} apoia-se sobre uma face de
$\qty{0.50}{m} \times \qty{0.30}{m}$ e depois sobre uma face de
$\qty{0.30}{m} \times \qty{0.20}{m}$. Calcule as duas pressões sobre o
chão. Que orientação é melhor sobre areia fofa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:2}
Exprima \qty{0.25}{MPa} em \unit{kPa} e em \unit{Pa}. Qual pressiona mais
forte: \qty{3.0}{N} sobre \qty{2.0}{cm^2} ou \qty{60}{N} sobre
\qty{500}{cm^2}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:3}
Calcule a \omterm{thm:g10:pressure:depth}{pressão manométrica} $\rho g h$ e a \omterm{met:g10:pressure:dive}{pressão absoluta} no fundo de
uma piscina de \qty{3.0}{m} de profundidade
($\rho = \qty{1000}{kg/m^3}$, $P_0 = \qty{101}{kPa}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:4}
O ar ocupa \qty{4.0}{L} a \qty{100}{kPa}, com temperatura constante. Qual
é o seu volume a \qty{250}{kPa}? E a \qty{50}{kPa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:5}
Uma patinadora de \qty{65}{kg} desliza sobre uma única lâmina de área de
contato \qty{12}{cm^2} e depois fica de pé sobre solas que somam
\qty{360}{cm^2}. Calcule as duas pressões e a razão entre elas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:6}
Um polegar empurra uma tachinha com \qty{20}{N}. A cabeça tem área
\qty{1.2}{cm^2} e a ponta, \qty{0.010}{mm^2}. Calcule as pressões sobre o
polegar e sobre a parede; por que só a parede é perfurada?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:7}
Uma escotilha de submarino de área \qty{0.50}{m^2} está a \qty{30}{m} de
profundidade em água do mar ($\rho = \qty{1025}{kg/m^3}$); por dentro, o
ar está à \omterm{def:g10:pressure:atmospheric}{pressão atmosférica}. Calcule a \omterm{def:g10:inertia:force}{força} resultante que a mantém
fechada e a massa cujo \omterm{def:g10:universal-gravitation:weight}{peso} é igual a ela.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:8}
Um sensor a \qty{5.00}{m} de profundidade num líquido desconhecido marca
uma \omterm{thm:g10:pressure:depth}{pressão manométrica} de \qty{66.2}{kPa}. Encontre a densidade. É água
doce?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:9}
Que altura teria o barômetro de Torricelli com água
($\rho = \qty{1000}{kg/m^3}$) em vez de mercúrio
($\rho = \qty{13600}{kg/m^3}$)? Obtenha as duas alturas de
$\rho g h = P_0$; por que o mercúrio venceu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:10}
Usando os dados da seringa da \cref{prop:g10:pressure:boyle}
($PV = \qty{6000}{kPa.mL}$), preveja o volume a \qty{240}{kPa} e a
\omterm{def:g10:pressure:pressure}{pressão} a \qty{75}{mL}. Que curva é $P$ em função de $V$? E $P$ em função
de $1/V$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:11}
Um mergulhador a \qty{20}{m} solta uma bolha de volume \qty{0.50}{cm^3}.
Qual é o volume dela logo abaixo da superfície? (Temperatura constante; a
água do mar acrescenta \qty{101}{kPa} a cada \qty{10}{m}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:12}
Uma ventosa de área \qty{12}{cm^2} é pressionada contra um teto.
\begin{enumerate}
  \item Vácuo perfeito por dentro: que \omterm{def:g10:inertia:force}{força} a atmosfera exerce sobre a
        ventosa, e que massa pendurada ela consegue sustentar?
  \item Uma ventosa real mantém \qty{20}{kPa} de ar por dentro. Que massa
        agora?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:13}
Um cilindro de mergulho contém \qty{12}{L} de ar a \qty{200}{bar}
(\qty{1}{bar} = \qty{100}{kPa} $\approx 1$ atmosfera); \qty{50}{bar}
precisam ficar de reserva. Um mergulhador respira \qty{15}{L} por minuto
à \omterm{def:g10:pressure:pressure}{pressão} ambiente. Quantos litros de ar à \omterm{def:g10:pressure:pressure}{pressão} da superfície são
utilizáveis? Quanto tempo o cilindro dura a \qty{30}{m} (\omterm{met:g10:pressure:dive}{pressão absoluta}
$\approx \qty{4}{bar}$)? E a \qty{10}{m}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:14}
Os pulmões de uma apneísta comportam \qty{6.0}{L} na superfície e não
podem encolher abaixo do volume residual de \qty{1.5}{L}. A que
profundidade no mar ($\rho = \qty{1025}{kg/m^3}$) esse limite é
alcançado? (Já se declarou que esse era o limite do mergulho em apneia;
hoje os recordes passam de \qty{100}{m}, com o sangue deslocando-se para
o tórax e ocupando o volume que falta.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:15}
Num elevador hidráulico, um líquido transmite a \omterm{def:g10:pressure:pressure}{pressão} sem alteração de
um pistão pequeno ($s = \qty{2.0}{cm^2}$) a um pistão grande
($S = \qty{400}{cm^2}$) que sustenta um carro de \qty{1200}{kg}.
\begin{enumerate}
  \item Que \omterm{def:g10:inertia:force}{força} sobre o pistão pequeno mantém o carro suspenso? Que
        \omterm{def:g10:pressure:pressure}{pressão} o líquido carrega?
  \item Para levantar o carro \qty{2.0}{cm}, que distância o pistão
        pequeno precisa percorrer? Compare o trabalho realizado em cada
        pistão.
\end{enumerate}
\end{exercise}

\section{Problema: uma respiração para descer, uma para subir}

\begin{problem}[{Problema de fim de semana --- uma respiração para descer
e uma para subir: a física de um mergulho em apneia e de uma subida com
cilindro, do aperto da máscara à regra de que os últimos dez metros são
os mais perigosos}]
\label{pb:g10:pressure:1}
Lena mergulha em apneia: uma respiração na superfície, descida até
\qty{30}{m} e volta. Marco mergulha ao lado dela com cilindro,
respirando do equipamento. Mesmo mar ($\rho = \qty{1025}{kg/m^3}$,
$g = \qty{9.81}{N/kg}$, $P_0 = \qty{101}{kPa}$), os mesmos pulmões de
\qty{6.0}{L}, perigos opostos --- tudo governado por
$P = P_0 + \rho g h$ e $PV$ = constante (temperatura constante o tempo
todo).

\medskip
\noindent\textbf{Parte I --- O \omterm{def:g10:universal-gravitation:weight}{peso} da água.}
\begin{enumerate}
  \item Calcule $\rho g$ para a água do mar em \unit{kPa} por \omterm{def:g10:orders-of-magnitude:unit}{metro} e
        confira a regra prática: uma atmosfera a cada \qty{10}{m}.
  \item Calcule a \omterm{met:g10:pressure:dive}{pressão absoluta} a \qty{10}{m}, \qty{20}{m},
        \qty{30}{m} e \qty{40}{m}.
  \item Exprima cada uma como múltiplo da \omterm{def:g10:pressure:pressure}{pressão} na superfície.
  \item A máscara de Lena cobre \qty{150}{cm^2}. A \qty{30}{m}, se o ar
        dentro dela ainda estivesse à \omterm{def:g10:pressure:pressure}{pressão} da superfície, que \omterm{def:g10:inertia:force}{força}
        resultante a água exerceria? (Os mergulhadores expiram pelo nariz
        para dentro da máscara justamente para evitar esse \emph{aperto
        de máscara}.)
  \item Um tímpano tem área de cerca de \qty{0.60}{cm^2}. Calcule a \omterm{def:g10:inertia:force}{força}
        resultante sobre ele a \qty{3.0}{m} se o ouvido médio ficar à
        \omterm{def:g10:pressure:pressure}{pressão} da superfície; por que os ouvidos doem já numa simples
        piscina, e o que a ``compensação'' precisa alcançar?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Uma respiração para descer.}
Lena deixa a superfície com \qty{6.0}{L} de ar nos pulmões.
\begin{enumerate}[resume]
  \item Usando a \omterm{prop:g10:pressure:boyle}{lei de Boyle} com \omterm{met:g10:pressure:dive}{pressões absolutas}, calcule o volume
        pulmonar dela a \qty{10}{m}, \qty{20}{m} e \qty{30}{m}.
  \item A que profundidade o volume pulmonar dela cai à metade?
  \item O volume pulmonar é proporcional à profundidade? Descreva a curva
        de $V$ em função da \omterm{met:g10:pressure:dive}{pressão absoluta} $P$.
  \item O volume residual dela é \qty{1.5}{L}: os pulmões não encolhem
        mais do que isso. Mostre que esse limite é alcançado perto de
        \qty{30}{m}.
  \item Ainda assim, os recordes de apneia passam de \qty{100}{m}: o que
        preenche o volume que falta? (Que outra coisa pode fluir para o
        tórax?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Uma respiração para subir.}
A \qty{30}{m}, o regulador de Marco enche os seus pulmões de
\qty{6.0}{L}.
\begin{enumerate}[resume]
  \item A que \omterm{def:g10:pressure:pressure}{pressão} o regulador entrega esse ar? Quantas vezes mais
        denso ele é do que o ar da superfície?
  \item Marco entra em pânico, prende a respiração e sobe até
        \qty{10}{m}: que volume o ar preso nele exige ali?
  \item E que volume na superfície --- quantas vezes a capacidade dele?
  \item Enuncie a primeira regra do mergulhador autônomo e diga por que
        respirar normalmente elimina o perigo.
  \item Lena também prende a respiração de \qty{30}{m} até a superfície e,
        ainda assim, os pulmões dela estão perfeitamente seguros. Explique
        a assimetria.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Bolhas e os últimos dez \omterm{def:g10:orders-of-magnitude:unit}{metros}.}
\begin{enumerate}[resume]
  \item Marco solta uma bolha de \qty{1.0}{cm^3} a \qty{40}{m}. Calcule o
        volume dela a \qty{30}{m}, \qty{20}{m}, \qty{10}{m} e logo abaixo
        da superfície.
  \item Para cada etapa de \qty{10}{m} da subida, calcule o fator de
        crescimento da bolha. Onde o crescimento é mais rápido?
  \item Os mergulhadores sobem no máximo \qty{10}{m} por minuto: quanto
        tempo desde \qty{40}{m}, e em que minuto qualquer gás preso ou
        dissolvido se expande mais?
  \item Em profundidade, o sangue de Marco dissolve nitrogênio extra,
        como o gás numa garrafa de refrigerante fechada. O que uma subida
        rápida demais provoca, e como a subida lenta e uma parada de
        segurança a \qty{5}{m} evitam isso?
  \item Desfecho: em uma frase cada, dê o veredito da apneísta, o
        veredito do mergulhador autônomo e a razão quantificada pela qual
        os últimos dez \omterm{def:g10:orders-of-magnitude:unit}{metros} de qualquer subida merecem o maior
        respeito.
\end{enumerate}
\end{problem}
