\chapter{Forças e o princípio da inércia}\label{ch:g10:inertia}

Pare de pedalar e a bicicleta vai perdendo velocidade até parar; pare de
puxar e o trenó fica imóvel. A experiência sussurra que o movimento
precisa ser alimentado por um esforço --- e por dois mil anos a física
concordou. Este capítulo constrói a \omterm{def:g10:inertia:force}{força}, uma flecha medida em \omterm{def:g10:inertia:force}{newtons},
e enuncia a primeira grande lei da mecânica: deixado realmente em paz, um
corpo em movimento conserva o seu movimento, retilíneo e regular, para
sempre.

\section{Modelar uma ação como força}

\begin{definition}[Força]\label{def:g10:inertia:force}
Uma \emph{força}\index{força} modela uma ação mecânica (um empurrão, um
puxão, uma atração) exercida sobre um corpo. Ela é caracterizada por:
\begin{itemize}
  \item o seu \emph{ponto de aplicação}\index{ponto de aplicação};
  \item a sua direção --- a reta ao longo da qual ela age --- e o seu
        sentido sobre essa reta;
  \item o seu módulo, medido em \emph{newtons}\index{newton} (símbolo
        \unit{N}).
\end{itemize}
Uma força se desenha como uma flecha $\vect F$ (um vetor, como no volume
de matemática) a partir do seu ponto de aplicação, com comprimento
proporcional ao módulo numa escala declarada.
\end{definition}

\begin{example}[O peso de uma mochila]\label{ex:g10:inertia:schoolbag}
Uma mochila de massa $m = \qty{4.0}{kg}$ é puxada para baixo pela Terra
com o seu \omterm{def:g10:universal-gravitation:weight}{peso} (\cref{ch:g10:universal-gravitation}):
$P = mg = 4.0 \times 9.81 \approx \qty{39}{N}$ --- um \omterm{def:g10:inertia:force}{newton} é mais ou
menos o \omterm{def:g10:universal-gravitation:weight}{peso} de uma maçã. Na escala $\qty{1}{cm} \leftrightarrow
\qty{10}{N}$: uma flecha vertical de \qty{3.9}{cm}, aplicada no centro de
gravidade da mochila, apontando para baixo.
\end{example}

\section{Um inventário de forças}

\begin{definition}[Forças de contato e forças a distância]\label{def:g10:inertia:contact}
Uma \emph{força de contato}\index{força de contato} é exercida nos pontos
em que dois corpos se tocam. Uma \emph{força a distância}\index{força a
distância} dispensa contato: o \omterm{def:g10:universal-gravitation:weight}{peso} --- a atração gravitacional do
\cref{ch:g10:universal-gravitation} --- é a única usada neste capítulo.
\end{definition}

\begin{definition}[As forças de contato mais comuns]\label{def:g10:inertia:inventory}
Quatro \omterm{def:g10:inertia:contact}{forças de contato} cobrem a maioria das situações do dia a dia:
\begin{itemize}
  \item a \emph{reação normal}\index{reação normal} $\vect R$ de um
        apoio, perpendicular à sua superfície, empurrando para fora dela;
  \item a \emph{tensão}\index{tensão} $\vect T$ de um fio, de uma corda
        ou de uma corrente, dirigida ao longo dele e puxando na sua
        direção;
  \item o \emph{atrito}\index{atrito} $\vect f$ de uma superfície, do ar
        ou da água, opondo-se ao deslizamento ou ao movimento;
  \item a \emph{propulsão}\index{propulsão} de um motor, de um jato ou de
        uma mão, empurrando no sentido do esforço.
\end{itemize}
\end{definition}

\begin{example}[Inventário de um trenó rebocado]\label{ex:g10:inertia:sled}
Um trenó rebocado sobre a neve: \omterm{def:g10:universal-gravitation:weight}{peso} $\vect P$ (a distância), \omterm{def:g10:inertia:inventory}{reação
normal} $\vect R$ da neve, \omterm{def:g10:inertia:inventory}{tensão} $\vect T$ da corda, \omterm{def:g10:inertia:inventory}{atrito} $\vect f$ que
se opõe ao deslizamento. Nada mais toca o trenó: o inventário está
completo --- \emph{sempre o \omterm{def:g10:universal-gravitation:weight}{peso}, e depois uma \omterm{def:g10:inertia:contact}{força de contato} por
objeto que toca}.
\end{example}

\section{O princípio da inércia}

\begin{definition}[Movimento retilíneo uniforme]\label{def:g10:inertia:uniform}
Um corpo está em \emph{movimento retilíneo uniforme}\index{movimento
retilíneo uniforme} num \omterm{def:g10:relative-motion:frame}{referencial} quando a sua \omterm{def:g10:relative-motion:trajectory}{trajetória} é uma reta
percorrida com velocidade constante: o seu vetor velocidade $\vect v$ é o
mesmo em todos os instantes. O repouso é o caso particular
$\vect v = \vect 0$.
\end{definition}

\begin{definition}[Forças que se compensam]\label{def:g10:inertia:compensate}
As \omterm{def:g10:inertia:force}{forças} que agem sobre um corpo \emph{se compensam}\index{forças que se
compensam} quando o efeito conjunto delas é nulo: postas ponta com cauda,
as suas flechas voltam ao ponto de partida. Duas \omterm{def:g10:inertia:force}{forças} se compensam
exatamente quando compartilham a mesma reta de ação com módulos iguais e
sentidos opostos.
\end{definition}

\begin{theorem}[Princípio da inércia]\label{thm:g10:inertia:principle}
\index{princípio da inércia}\index{inércia}
No \omterm{def:g10:relative-motion:frames}{referencial do solo}, um corpo sujeito a nenhuma \omterm{def:g10:inertia:force}{força}, ou a \omterm{def:g10:inertia:compensate}{forças que
se compensam}, permanece em repouso ou mantém um \omterm{def:g10:inertia:uniform}{movimento retilíneo
uniforme}; reciprocamente, um corpo em repouso ou em \omterm{def:g10:inertia:uniform}{movimento retilíneo
uniforme} está sujeito a \omterm{def:g10:inertia:compensate}{forças que se compensam} (ou a nenhuma \omterm{def:g10:inertia:force}{força}).
\end{theorem}
\admitted

\begin{remark}
Nenhuma experiência consegue isolar um corpo por completo, de modo que o
princípio é um postulado --- entrevisto por Galileu, enunciado por
Newton. Um capítulo posterior o promove a uma lei que relaciona \omterm{def:g10:inertia:force}{forças} a
\emph{variações} de velocidade; o volume do primeiro ano de graduação
pergunta em que referenciais ele de fato vale.
\end{remark}

\begin{proposition}[Ler o movimento a partir das forças, e vice-versa]\label{prop:g10:inertia:converse}
No \omterm{def:g10:relative-motion:frames}{referencial do solo}:
\begin{enumerate}
  \item se a velocidade de um corpo muda --- em módulo \emph{ou} em
        direção ---, as \omterm{def:g10:inertia:force}{forças} sobre ele \emph{não} \omterm{def:g10:inertia:compensate}{se compensam};
  \item se um corpo está em repouso ou em \omterm{def:g10:inertia:uniform}{movimento retilíneo uniforme},
        as \omterm{def:g10:inertia:force}{forças} do seu inventário têm de se cancelar --- o que
        determina as incógnitas.
\end{enumerate}
\end{proposition}

\begin{proof}
A primeira afirmação é a contrapositiva do
\cref{thm:g10:inertia:principle}, e a segunda é a sua parte recíproca
aplicada a um inventário completo.
\end{proof}

\begin{example}[O disco de hóquei]\label{ex:g10:inertia:puck}
Solto pelo taco, um disco atravessa \qty{30}{m} de gelo liso em
\qty{2.5}{s}, em linha reta, com a velocidade constante
$v = 30/2.5 = \qty{12}{m/s}$. Inventário: \omterm{def:g10:universal-gravitation:weight}{peso} e \omterm{def:g10:inertia:inventory}{reação normal} (\omterm{def:g10:inertia:inventory}{atrito}
desprezível). Eles \omterm{def:g10:inertia:compensate}{se compensam}, e o princípio prevê exatamente o
movimento observado: nada empurra o disco para a frente --- e nada
precisa empurrar.
\end{example}

\begin{remark}[Galileu contra Aristóteles]
Aristóteles ensinava que todo movimento precisa de um motor, e a vida
cotidiana --- com \omterm{def:g10:inertia:inventory}{atrito} em cada contato --- parece concordar. Galileu
idealizou o \omterm{def:g10:inertia:inventory}{atrito} para fora: superfícies cada vez mais lisas exigem cada
vez menos empurrão para \emph{manter} o movimento. A persistência, e não
o repouso, é o estado natural.
\end{remark}

\begin{remark}[O passageiro que é jogado para a frente]
Quando um ônibus freia bruscamente, os passageiros em pé são projetados
para a frente. \omterm{def:g10:inertia:force}{Força} nenhuma os arremessa: por inércia, os corpos deles
conservam a sua velocidade enquanto o ônibus perde a sua. O cinto de
segurança fornece a \omterm{def:g10:inertia:force}{força} para trás que falta no inventário.
\end{remark}

\section{Forças equilibradas: estática a partir de um desenho}

\begin{definition}[Diagrama de forças]\label{def:g10:inertia:diagram}
O \emph{diagrama de forças}\index{diagrama de forças} de um corpo mostra
o corpo sozinho, com todas as \omterm{def:g10:inertia:force}{forças} que agem \emph{sobre} ele desenhadas
em escala a partir do seu \omterm{def:g10:inertia:force}{ponto de aplicação}; as \omterm{def:g10:inertia:force}{forças} exercidas
\emph{pelo} corpo nunca aparecem.
\end{definition}

\begin{method}[Analisar uma situação estática]\label{met:g10:inertia:method}
\begin{enumerate}
  \item Escolha o corpo a estudar; trabalhe no \omterm{def:g10:relative-motion:frames}{referencial do solo}.
  \item Inventário: sempre o \omterm{def:g10:universal-gravitation:weight}{peso}, e uma \omterm{def:g10:inertia:contact}{força de contato} por contato.
  \item Desenhe o \omterm{def:g10:inertia:diagram}{diagrama de forças}.
  \item O corpo está em repouso, logo as \omterm{def:g10:inertia:force}{forças} \omterm{def:g10:inertia:compensate}{se compensam}
        (\cref{thm:g10:inertia:principle}): leia os módulos desconhecidos
        no desenho (comprimentos iguais em 1D, componentes em 2D).
\end{enumerate}
\end{method}

\begin{example}[O livro sobre a mesa]\label{ex:g10:inertia:book}
Um livro de massa \qty{0.50}{kg} repousa sobre uma mesa: \omterm{def:g10:universal-gravitation:weight}{peso}
$P = 0.50 \times 9.81 \approx \qty{4.9}{N}$ e \omterm{def:g10:inertia:inventory}{reação normal} $\vect R$. Em
repouso, $\vect R$ compensa $\vect P$: vertical, para cima,
$R = \qty{4.9}{N}$. Coloque um segundo livro e $R$ se ajusta --- um apoio
empurra exatamente com a \omterm{def:g10:inertia:force}{força} necessária.
\end{example}

\begin{example}[A luminária em dois fios]\label{ex:g10:inertia:lamp}
Uma luminária de massa \qty{3.0}{kg} ($P \approx \qty{29.4}{N}$) pende de
dois fios, cada um a $45^\circ$ da vertical. Por simetria as tensões são
iguais ($T_1 = T_2 = T$) e as suas componentes horizontais se cancelam;
as componentes verticais, cada uma $T \cos 45^\circ$, sustentam o \omterm{def:g10:universal-gravitation:weight}{peso}:
\[
2\,T \cos 45^\circ = P
\qquad\text{logo}\qquad
T = \frac{29.4}{2 \times 0.707} \approx \qty{21}{N}
\]
--- mais da metade do \omterm{def:g10:universal-gravitation:weight}{peso} por fio: puxar em ângulo custa \omterm{def:g10:inertia:inventory}{tensão}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:inertia:1}
Faça o inventário e classifique (contato ou distância) as \omterm{def:g10:inertia:force}{forças} sobre:
(a) um livro em repouso sobre uma mesa; (b) uma luminária pendurada no
seu fio; (c) um disco deslizando sobre gelo liso; (d) uma maçã caindo
(resistência do ar desprezível).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:2}
Calcule o \omterm{def:g10:universal-gravitation:weight}{peso} de um livro de \qty{250}{g} e o de uma pessoa de
\qty{60}{kg}. Na escala $\qty{1}{cm} \leftrightarrow \qty{1}{N}$, qual é
o comprimento da flecha do livro, e por que a escala é inviável para a
pessoa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:3}
Num diagrama na escala $\qty{1}{cm} \leftrightarrow \qty{2}{N}$, uma
\omterm{def:g10:inertia:force}{força} é uma flecha horizontal de \qty{3.5}{cm} apontando para a esquerda,
aplicada num ponto $A$. Dê as suas três características.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:4}
Um dicionário de massa \qty{1.2}{kg} está sobre uma prateleira. Desenhe o
seu \omterm{def:g10:inertia:diagram}{diagrama de forças} e dê o módulo e a direção de cada \omterm{def:g10:inertia:force}{força}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:inertia:5}
Verdadeiro ou falso --- corrija as falsas: (a) um corpo sempre acaba
parando, a menos que uma \omterm{def:g10:inertia:force}{força} continue a empurrá-lo; (b) o \omterm{def:g10:inertia:uniform}{movimento
retilíneo uniforme} não exige \omterm{def:g10:inertia:force}{força} não compensada alguma; (c) um corpo em
repouso não está sujeito a \omterm{def:g10:inertia:force}{força} alguma; (d) se a velocidade de um corpo
muda, as \omterm{def:g10:inertia:force}{forças} sobre ele não \omterm{def:g10:inertia:compensate}{se compensam}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:6}
Depois de solto, um disco percorre \qty{24}{m} de gelo em \qty{3.0}{s},
em linha reta e com velocidade constante. Calcule a sua velocidade; as
\omterm{def:g10:inertia:force}{forças} sobre ele \omterm{def:g10:inertia:compensate}{se compensam}? No concreto, o mesmo disco para em poucos
\omterm{def:g10:orders-of-magnitude:unit}{metros}: o que mudou no inventário, e o que conclui a
\cref{prop:g10:inertia:converse}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:7}
Um passageiro de \qty{70}{kg} está de pé num elevador que sobe com a
velocidade constante de \qty{1.5}{m/s}. Liste as \omterm{def:g10:inertia:force}{forças} sobre o
passageiro; elas \omterm{def:g10:inertia:compensate}{se compensam}? Dê o módulo da \omterm{def:g10:inertia:force}{força} do piso. Alguma coisa
mudaria com o elevador em repouso?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:8}
Uma luminária de massa \qty{1.8}{kg} pende em repouso de um único fio
vertical. Calcule a \omterm{def:g10:inertia:inventory}{tensão} do fio. O fio é substituído por dois fios
\emph{verticais} que repartem a carga igualmente: qual é a \omterm{def:g10:inertia:inventory}{tensão} de cada
um?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:9}
Um ônibus freia bruscamente e um passageiro em pé é projetado para a
frente; numa curva a velocidade constante, o mesmo passageiro se inclina
para fora. Explique as duas coisas sem inventar uma \omterm{def:g10:inertia:force}{força} ``para a
frente'' ou ``para fora''.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:10}
Um paraquedista de massa total \qty{85}{kg} desce verticalmente com a
velocidade constante de \qty{5.0}{m/s}. Desenhe o \omterm{def:g10:inertia:diagram}{diagrama de forças} e
calcule a resistência do ar sobre o velame.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:inertia:11}
Um esquiador aquático é rebocado em linha reta com velocidade constante;
a corda horizontal puxa com \qty{300}{N}. Qual é o arrasto horizontal da
água sobre os esquis (justifique)? Que duas \omterm{def:g10:inertia:force}{forças} verticais \omterm{def:g10:inertia:compensate}{se compensam}
uma à outra?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:12}
Uma luminária de massa \qty{2.4}{kg} pende de dois fios a $40^\circ$ da
vertical ($\cos 40^\circ \approx 0.766$). Calcule cada \omterm{def:g10:inertia:inventory}{tensão}. Por que a
\omterm{def:g10:inertia:inventory}{tensão} cresce à medida que os fios se aproximam da horizontal, e por que
dois fios exatamente horizontais nunca conseguiriam sustentar a
luminária?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:13}
Uma pedra de curling é empurrada, solta, desliza desacelerando bem aos
poucos e por fim para. Para cada fase (empurrão, deslizamento, repouso),
desenhe o \omterm{def:g10:inertia:diagram}{diagrama de forças} e diga se as \omterm{def:g10:inertia:force}{forças} \omterm{def:g10:inertia:compensate}{se compensam}, citando o
\cref{thm:g10:inertia:principle} ou a
\cref{prop:g10:inertia:converse}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:14}
A sonda \emph{Voyager 1} desliza pelo espaço interestelar a
\qty{17}{km/s}, com os motores desligados. Por que ela não precisa de
combustível para manter a velocidade, e o que Aristóteles preveria? Que
distância ela percorre em um ano, em \unit{km} e depois em \omterm{def:g10:orders-of-magnitude:lightyear}{unidades
astronômicas} ($\qty{1}{au} \approx \qty{1.496e8}{km}$)? Por que ela
precisa acionar um propulsor mesmo para mudar de direção \emph{sem} mudar
de velocidade escalar?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:inertia:15}
Três cordas puxam um anel em repouso. Numa malha em que um quadradinho
vale \qty{1}{N}, duas \omterm{def:g10:inertia:force}{forças} valem $\vect F_1\,(3, 2)$ e
$\vect F_2\,(-1, 2)$. Encontre as componentes, o módulo e a direção da
terceira \omterm{def:g10:inertia:force}{força}.
\end{exercise}

\section{Problema: inércia em três atos}

\begin{problem}[{Problema de fim de semana --- a esteira do aeroporto, o
rinque de gelo e o paraquedas: a velocidade constante como assinatura de
forças compensadas, e por que um paraquedas não reduz o arrasto, mas a
velocidade}]
\label{pb:g10:inertia:1}
Três cenas --- uma mala numa esteira rolante, um disco num rinque de
gelo, um paraquedista no ar da tarde --- e uma única lei falando nas
três.

\medskip
\noindent\textbf{Parte I --- A esteira do aeroporto.}
Uma mala de \qty{23}{kg} está parada sobre uma esteira que se move com a
velocidade constante de \qty{0.75}{m/s}.
\begin{enumerate}
  \item Qual é o movimento da mala no \omterm{def:g10:relative-motion:frames}{referencial do solo}? Faça o
        inventário das \omterm{def:g10:inertia:force}{forças} sobre ela.
  \item Essas \omterm{def:g10:inertia:force}{forças} \omterm{def:g10:inertia:compensate}{se compensam}? Justifique; calcule os dois módulos.
  \item Compare o \omterm{def:g10:inertia:diagram}{diagrama de forças} dela com o da mesma mala parada no
        piso. Que afirmação profunda a comparação ilustra?
  \item A esteira para de repente. Descreva o que a mala faz, sem
        inventar uma \omterm{def:g10:inertia:force}{força} para a frente.
  \item A dona dela caminha a \qty{1.2}{m/s} em relação à esteira. Qual é
        a velocidade dela em relação ao solo? As \omterm{def:g10:inertia:force}{forças} sobre \emph{ela}
        \omterm{def:g10:inertia:compensate}{se compensam}?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O rinque de gelo.}
\begin{enumerate}[resume]
  \item Um disco de \qty{160}{g} desliza em linha reta com a velocidade
        constante de \qty{8.0}{m/s}. Desenhe o seu \omterm{def:g10:inertia:diagram}{diagrama de forças} com
        os módulos.
  \item Aristóteles afirma que o disco precisa de um motor. O que o
        rinque responde, e por que as superfícies do dia a dia parecem
        dar razão a ele?
  \item O disco atravessa um trecho áspero e desacelera. Que conclusão se
        tira sobre as \omterm{def:g10:inertia:force}{forças} ali? Qual \omterm{def:g10:inertia:force}{força} é a culpada?
  \item Uma patinadora faz uma curva suave a velocidade constante. As
        \omterm{def:g10:inertia:force}{forças} sobre ela \omterm{def:g10:inertia:compensate}{se compensam}? Cuidado: que tipo de grandeza é a
        velocidade?
  \item Associe diagrama e movimento: (a) \omterm{def:g10:inertia:force}{forças} compensadas; (b) uma
        \omterm{def:g10:inertia:force}{força} extra oposta a $\vect v$; (c) uma \omterm{def:g10:inertia:force}{força} extra perpendicular
        a $\vect v$ --- (i) desacelerando em linha reta; (ii) fazendo
        curva a velocidade constante; (iii) \omterm{def:g10:inertia:uniform}{movimento retilíneo uniforme}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O paraquedas.}
A massa total de um saltador, equipamento incluído, é \qty{80}{kg}.
\begin{enumerate}[resume]
  \item Ao deixar o avião, a velocidade vertical e o arrasto ainda são
        nulos. Calcule as \omterm{def:g10:inertia:force}{forças}; elas \omterm{def:g10:inertia:compensate}{se compensam}? O que tem de
        acontecer em seguida?
  \item O arrasto cresce com a velocidade. Descreva a queda enquanto o
        arrasto $<$ \omterm{def:g10:universal-gravitation:weight}{peso}.
  \item Antes de o velame abrir, o saltador atinge um regime estável de
        \qty{50}{m/s} (``velocidade limite''). Calcule o arrasto. Qual é
        o movimento, e por que ele é ``inércia disfarçada''?
  \item O velame abre: o arrasto passa a superar de longe o \omterm{def:g10:universal-gravitation:weight}{peso}. O que
        acontece com a velocidade? (O saltador se move \emph{para baixo}
        o tempo todo.)
  \item Uma nova velocidade estável de \qty{5.0}{m/s} é atingida. Calcule
        o arrasto; compare com a questão 13.
  \item O desfecho: o que o paraquedas mudou --- o arrasto final ou a
        velocidade em que o arrasto alcança o \omterm{def:g10:universal-gravitation:weight}{peso}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Vereditos.}
\begin{enumerate}[resume]
  \item Complete e justifique: ``em repouso ou em \omterm{def:g10:inertia:uniform}{movimento retilíneo
        uniforme} $\iff$ \dots''. Por que o repouso não merece uma lei
        própria?
  \item Em qual dos atos o ``o movimento precisa de um motor'' de
        Aristóteles teria falhado de modo mais visível, e para o que
        Galileu apontaria?
  \item Uma prova afirma: ``o paraquedista cai com velocidade constante,
        logo nenhuma \omterm{def:g10:inertia:force}{força} age sobre ele''. Corrija a frase em uma linha.
  \item Final: para cada ato, nomeie o par de \omterm{def:g10:inertia:compensate}{forças que se compensam} e
        enuncie a única lei usada vinte vezes neste problema.
\end{enumerate}
\end{problem}
