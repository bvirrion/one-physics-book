\chapter{Energia: formas e conservação}\label{ch:g10:energy-conservation}

Um carro em movimento, um celular carregado, um forno quente e um martelo
erguido compartilham uma propriedade: cada um deles pode fazer algo
acontecer. A física mede essa capacidade com um único número --- a
\omterm{def:g10:energy-conservation:energy}{energia} --- e mantém a contabilidade sob uma regra de ferro: o total
nunca muda, ele apenas se desloca e troca de roupa. Este capítulo abre o
livro-caixa: as formas que a \omterm{def:g10:energy-conservation:energy}{energia} assume, as cadeias que ela percorre
e o que um \omterm{def:g10:energy-conservation:kwh}{quilowatt-hora} compra.

\section{Energia: a moeda comum}

\begin{definition}[Energia e o joule]\label{def:g10:energy-conservation:energy}
A \emph{energia}\index{energia} mede a capacidade de um sistema de
produzir mudanças: pôr matéria em movimento, erguê-la, aquecê-la,
iluminá-la. A sua \omterm{def:g10:orders-of-magnitude:unit}{unidade} é o \emph{joule}\index{joule} (\unit{J}).
Armazenada, transferida, convertida --- todos os ramos da física contam
nessa moeda única.
\end{definition}

\begin{definition}[Energia cinética]\label{def:g10:energy-conservation:kinetic}
Um corpo de massa $m$ (em \unit{kg}) que se move com velocidade $v$ (em
\unit{m/s}) carrega a \emph{energia cinética}\index{energia cinética}
$E_k = \tfrac12\, m v^2$: dobrar a massa a dobra, dobrar a velocidade a
quadruplica.
\end{definition}

\begin{example}[Carro contra pedestre]\label{ex:g10:energy-conservation:kinetic}
Um carro de \qty{1300}{kg} a \qty{100}{km/h} $= \qty{27.8}{m/s}$ carrega
$E_k = \tfrac12 \times 1300 \times 27.8^2 \approx \qty{5.0e5}{J}$; um
pedestre de \qty{70}{kg} a \qty{1.5}{m/s},
$\tfrac12 \times 70 \times 1.5^2 \approx \qty{79}{J}$ --- seis mil vezes
menos. Velocidade é cara: o quadrado se encarrega disso.
\end{example}

\begin{definition}[Energia potencial gravitacional]\label{def:g10:energy-conservation:potential}
Um corpo de massa $m$ a uma altura $h$ (em \unit{m}) acima de um nível de
referência escolhido armazena a \emph{energia potencial
gravitacional}\index{energia potencial!gravitacional} $E_p = m g h$, com
$g = \qty{9.81}{N/kg}$ --- válida perto do solo, onde $g$ é
efetivamente constante. Só as \emph{diferenças} de altura importam: o
nível em que $E_p = 0$ é escolhido livremente.
\end{definition}

\begin{example}[Caixa-d'água elevada]\label{ex:g10:energy-conservation:potential}
Cada \omterm{def:g10:orders-of-magnitude:unit}{metro} cúbico (\qty{1000}{kg}) de água armazenado \qty{30}{m} acima,
numa caixa-d'água elevada, guarda
$1000 \times 9.81 \times 30 \approx \qty{2.9e5}{J}$ --- cerca de seis
décimos da \omterm{def:g10:energy-conservation:kinetic}{energia cinética} do carro acima. As cidades estocam \omterm{def:g10:energy-conservation:energy}{energia} lá
em cima.
\end{example}

\begin{definition}[As outras formas]\label{def:g10:energy-conservation:forms}
A \omterm{def:g10:energy-conservation:energy}{energia} também se esconde em trajes menos visíveis: a \emph{energia
térmica}\index{energia térmica}, a agitação desordenada das moléculas de
um corpo (quanto mais quente, maior); a \emph{energia
química}\index{energia química}, guardada nas ligações moleculares ---
alimento, madeira, gasolina, pilhas; a \emph{energia
elétrica}\index{energia elétrica}, transportada por correntes da usina à
tomada; a \emph{energia radiante}\index{energia radiante}, carregada pela
luz mesmo através do espaço vazio --- é assim que a \omterm{def:g10:energy-conservation:energy}{energia} do Sol chega
até nós; a \emph{energia nuclear}\index{energia nuclear}, guardada dentro
dos núcleos atômicos e liberada nas estrelas
(\cref{ch:g12:nuclear-energy}).
\end{definition}

\section{Transferências e cadeias de conversão}

\begin{definition}[Transferência, conversão, cadeia]\label{def:g10:energy-conservation:transfer}
Uma \emph{transferência de energia}\index{transferência de energia} move
\omterm{def:g10:energy-conservation:energy}{energia} entre sistemas --- os \emph{reservatórios}\index{reservatório (de
energia)}; uma \emph{conversão de energia}\index{conversão de energia}
muda a forma dela. Um dispositivo que faz uma coisa ou outra (turbina,
motor, lâmpada, músculo) é um \emph{conversor}\index{conversor}. Uma
\emph{cadeia de conversão}\index{cadeia de conversão} desenha o percurso:
caixas para os reservatórios e as formas, flechas rotuladas para as
transferências.
\end{definition}

\begin{method}[Desenhar uma cadeia de conversão]\label{met:g10:energy-conservation:chain}
\begin{enumerate}
  \item Identifique o reservatório e a forma iniciais (o que é
        consumido?).
  \item Siga a \omterm{def:g10:energy-conservation:energy}{energia} \omterm{def:g10:energy-conservation:transfer}{conversor} por \omterm{def:g10:energy-conservation:transfer}{conversor}: uma flecha por
        transferência, com a forma nomeada em cada caixa.
  \item Acrescente uma flecha de calor em cada \omterm{def:g10:energy-conservation:transfer}{conversor} real --- alguma
        \omterm{def:g10:energy-conservation:energy}{energia} sempre vaza como calor (\omterm{def:g10:inertia:inventory}{atrito}, fios, escapamento).
\end{enumerate}
\end{method}

Um pêndulo percorre a cadeia mais simples de todas --- toda potencial no
alto da oscilação, toda cinética na passagem pelo ponto mais baixo
(tomando $E_p = 0$ ali):

\section{Conservação da energia}

\begin{theorem}[Conservação da energia]\label{thm:g10:energy-conservation:conservation}
\index{conservação da energia}
A \omterm{def:g10:energy-conservation:energy}{energia} total de um sistema isolado --- um sistema que nada troca com o
exterior --- é constante. A \omterm{def:g10:energy-conservation:energy}{energia} nunca é criada nem destruída: ela só
muda de forma e de lugar, e o que um reservatório perde os outros ganham,
\omterm{def:g10:energy-conservation:energy}{joule} por \omterm{def:g10:energy-conservation:energy}{joule}.
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:energy-conservation:admitted}
Nenhuma experiência jamais flagrou esse princípio falhando; aqui o
tomamos como dado e o usamos numericamente. O
\cref{ch:g12:work-and-energy} deduz a sua parte mecânica ($E_k + E_p$)
das leis de Newton; a contabilidade completa espera os volumes de
graduação.
\end{remark}

\begin{example}[Queda livre, cotada em joules]\label{ex:g10:energy-conservation:fall}
Uma bola de \qty{2.0}{kg} cai de \qty{10}{m} (resistência do ar
desprezível): ela parte com
$E_p = 2.0 \times 9.81 \times 10 \approx \qty{196}{J}$ e chega ao chão
com $E_k = \qty{196}{J}$, de modo que
$v = \sqrt{2 \times 196/2.0} = \qty{14}{m/s} \approx \qty{50}{km/h}$. Sem
\omterm{def:g10:inertia:force}{forças}, sem cronômetro: só a conservação já cota o impacto.
\end{example}

\section{Rendimento}

\begin{definition}[Rendimento]\label{def:g10:energy-conservation:efficiency}
Um \omterm{def:g10:energy-conservation:transfer}{conversor} consome $E_{\text{consumida}}$ e entrega a forma desejada
$E_{\text{útil}}$; o seu \emph{rendimento}\index{rendimento} é
\[
\eta = \frac{E_{\text{útil}}}{E_{\text{consumida}}} < 1 .
\]
A conservação proíbe $\eta > 1$, e \omterm{def:g10:energy-conservation:transfer}{conversores} reais nunca chegam a $1$:
a \omterm{def:g10:energy-conservation:energy}{energia} que falta não foi destruída --- ela saiu como calor.
\end{definition}

\begin{example}[Para onde vai a gasolina]\label{ex:g10:energy-conservation:engine}
Um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de gasolina armazena \qty{45}{MJ} de \omterm{def:g10:energy-conservation:forms}{energia química}; um
motor de carro a converte com $\eta \approx 0.30$: \qty{13.5}{MJ} de
movimento e \qty{31.5}{MJ} aquecendo o escapamento, o radiador e a rua.
Dois terços de cada tanque aquecem a atmosfera.
\end{example}

\begin{remark}[Os rendimentos se multiplicam]\label{rem:g10:energy-conservation:chain-eta}
\omterm{def:g10:energy-conservation:transfer}{Conversores} encadeados multiplicam os seus \omterm{def:g10:energy-conservation:efficiency}{rendimentos}: uma usina
($0.38$) que alimenta a rede ($0.94$) que alimenta um carregador
($0.85$) entrega $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.30$.
Cadeias longas são canos furados: cada flecha cobra um imposto sobre o
fluxo.
\end{remark}

\section{Potência, o watt e o quilowatt-hora}

\begin{definition}[Potência]\label{def:g10:energy-conservation:power}
A \emph{potência}\index{potência} é a taxa com que a \omterm{def:g10:energy-conservation:energy}{energia} flui,
$P = E/t$, medida em \emph{watts}\index{watt} (\unit{W}), com
$\qty{1}{W} = \qty{1}{J/s}$. A \omterm{def:g10:energy-conservation:energy}{energia} é uma quantidade; a potência, um
fluxo --- uma banheira contra a torneira dela.
\end{definition}

\begin{example}[Chaleira contra ser humano]\label{ex:g10:energy-conservation:power}
Uma chaleira elétrica de \qty{2200}{W} ligada por \qty{150}{s} usa
$E = 2200 \times 150 = \qty{3.3e5}{J}$. Um ser humano funciona com cerca
de \qty{10}{MJ} de alimento por dia, uma média de
$\num{1.0e7}/\num{86400} \approx \qty{116}{W}$: você fica em marcha lenta
como uma lâmpada incandescente forte.
\end{example}

\begin{definition}[O quilowatt-hora]\label{def:g10:energy-conservation:kwh}
O \emph{quilowatt-hora}\index{quilowatt-hora} (\unit{kW.h}) é a \omterm{def:g10:energy-conservation:energy}{energia}
de um fluxo de \qty{1}{kW} durante uma hora:
$\qty{1}{kW.h} = 1000 \times 3600\,\unit{J} = \qty{3.6}{MJ}$ --- a
\omterm{def:g10:orders-of-magnitude:unit}{unidade} que os medidores de eletricidade contam: \omterm{def:g10:energy-conservation:power}{potência} em \unit{kW}
vezes horas.
\end{definition}

\begin{example}[Ler a conta]\label{ex:g10:energy-conservation:bill}
Uma conta é uma subtração e uma multiplicação: a leitura nova do medidor
menos a antiga dá os \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}; vezes o preço unitário (cerca de
$0.25$ euros por \unit{kW.h}), mais uma taxa fixa de assinatura, dá o
total. Um aquecedor de \qty{2}{kW} ligado \qty{2.5}{h} toda noite usa
\qty{5}{kW.h} por dia: $38$ euros por mês só ele.
\end{example}

\begin{example}[Ordens de grandeza]\label{ex:g10:energy-conservation:orders}
Vale a pena saber de cor: erguer uma maçã um \omterm{def:g10:orders-of-magnitude:unit}{metro}, cerca de \qty{1}{J};
a bateria de um celular, $\qty{10}{W.h} \approx \qty{4e4}{J}$; uma tigela
de cereal, \qty{1.5}{MJ}; a sua comida de um dia, \qty{10}{MJ}; um
\omterm{def:g10:orders-of-magnitude:unit}{quilograma} de gasolina, \qty{45}{MJ} --- quase exatamente a eletricidade
diária de uma casa ($\approx \qty{12}{kW.h}$); um raio, \qty{5}{GJ}: cem
dias de casa, entregues num relâmpago.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:1}
Um carro de \qty{1200}{kg} anda a \qty{90}{km/h}, ou seja,
\qty{25}{m/s}: calcule a sua \omterm{def:g10:energy-conservation:kinetic}{energia cinética}. Refaça a conta a
\qty{130}{km/h} (\qty{36.1}{m/s}): o que $44\%$ a mais de velocidade
fizeram com $E_k$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:2}
Um caminhante de massa \qty{72}{kg} sobe da altitude \qty{1200}{m} até
\qty{1650}{m}. Calcule a \omterm{def:g10:energy-conservation:energy}{energia} potencial ganha, converta-a em
\omterm{def:g10:energy-conservation:kwh}{quilowatt-horas} e coloque um preço nela a $0.25$ euros por \unit{kW.h}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:3}
Converta: \qty{1}{kW.h} para \omterm{def:g10:energy-conservation:energy}{joules}; uma bateria de celular de
\qty{12}{W.h} para quilojoules; \qty{45}{MJ} (um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de gasolina)
para \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}; o seu dia de \qty{10}{MJ} de comida para
\omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:4}
Uma chaleira elétrica de \qty{2200}{W} fica ligada por \qty{150}{s}.
Calcule a \omterm{def:g10:energy-conservation:energy}{energia} usada, em \omterm{def:g10:energy-conservation:energy}{joules} e em \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}. Depois calcule a
\omterm{def:g10:energy-conservation:power}{potência} média de um ser humano que funciona com \qty{10}{MJ} por dia.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:energy-conservation:5}
Escreva a \omterm{def:g10:energy-conservation:transfer}{cadeia de conversão} (\cref{met:g10:energy-conservation:chain})
de: um dínamo de bicicleta alimentando uma lâmpada; um aquecedor de água
a gás; um painel solar carregando um celular. Marque os vazamentos de
calor.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:6}
Um motor queima \qty{2.0}{kg} de gasolina (\qty{45}{MJ/kg}) e entrega
\qty{27}{MJ} de movimento. Calcule o seu \omterm{def:g10:energy-conservation:efficiency}{rendimento} e a \omterm{def:g10:energy-conservation:energy}{energia} liberada
como calor. Para onde vai esse calor?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:7}
Um saltador de \qty{70}{kg} se lança de um penhasco de \qty{20}{m}
(resistência do ar desprezível). Usando a conservação, calcule a \omterm{def:g10:energy-conservation:kinetic}{energia
cinética} na chegada à água e a velocidade de entrada, em \unit{m/s} e em
\unit{km/h}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:8}
Um medidor marcava \qty{41236}{kW.h} em 1\textsuperscript{o} de março e
\qty{41562}{kW.h} 30 dias depois: qual é a \omterm{def:g10:energy-conservation:energy}{energia} usada, a média diária,
a conta a $0.26$ euros por \unit{kW.h} mais $12$ euros fixos, e a
\omterm{def:g10:energy-conservation:power}{potência} média em \omterm{def:g10:energy-conservation:power}{watts}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:9}
Uma lâmpada incandescente antiga de \qty{60}{W} transforma apenas $5\%$
do que consome em luz; um LED dá a mesma luz com \qty{9}{W}. Calcule a

\qty{4}{h} por dia, o consumo anual de cada um e a economia a $0.25$
euros por \unit{kW.h}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:10}
Uma barragem faz passar \qty{150}{m^3} de água (\qty{1.5e5}{kg}) por

liberada a cada segundo --- a \omterm{def:g10:energy-conservation:power}{potência} disponível ---, depois a \omterm{def:g10:energy-conservation:power}{potência}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:energy-conservation:11}
Um pêndulo de \qty{0.20}{kg} é solto \qty{5.0}{cm} acima do seu ponto
mais baixo: calcule a sua velocidade ali. Depois de alguns minutos ele
fica pendurado, imóvel: para onde foi a sua \omterm{def:g10:energy-conservation:energy}{energia}, e por que isso não

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:energy-conservation:12}

\section{Problema: alimentar uma casa por um dia}

\begin{problem}[{Problema de fim de semana --- da tigela de cereal aos
painéis no telhado: uma auditoria energética completa de um dia comum em
casa, e quanto custa um joule dependendo de quem o vende}]
\label{pb:g10:energy-conservation:1}
Numa terça-feira comum, uma família resolve descobrir para onde vai a sua
eletricidade, armada com este capítulo, o medidor do corredor e o verso
da conta. O veredito da noite: o dia deles equivale a um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de
gasolina, dezesseis \omterm{def:g10:orders-of-magnitude:unit}{metros} quadrados de sol --- ou cinco ciclistas
pedalando sem parar.

\medskip
\noindent\textbf{Parte I --- A auditoria.}
O inventário do dia: geladeira \qty{100}{W} por \qty{24}{h}; aquecedor de
água \qty{2400}{W} por \qty{2.5}{h}; forno \qty{2000}{W} por
\qty{1.0}{h}; máquina de lavar \qty{500}{W} por \qty{2.0}{h}; iluminação
\qty{60}{W} por \qty{5.0}{h}; telas \qty{150}{W} por \qty{6.0}{h}.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:energy-conservation:energy}{energia} de cada aparelho no dia, em \unit{kW.h}.
  \item Some o dia em \omterm{def:g10:energy-conservation:kwh}{quilowatt-horas} e depois em megajoules; compare com
        um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de gasolina (\qty{45}{MJ}).
  \item Calcule a \omterm{def:g10:energy-conservation:power}{potência} média da casa nas \qty{24}{h}; que fração de
        uma chaleira ligada isso representa?
  \item Que aparelho, sozinho, domina o consumo? Recalcule o total diário
        se o aquecedor funcionasse apenas \qty{1.25}{h}.
  \item As luzinhas de espera consomem \qty{15}{W} permanentes: qual é a
        \omterm{def:g10:energy-conservation:energy}{energia} diária, a fatia do total e o custo anual a $0.25$ euros
        por \unit{kW.h}?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A conta.}
\begin{enumerate}[resume]
  \item Em 30 dias o medidor foi de \qty{56214}{kW.h} a
        \qty{56604}{kW.h}: qual é o consumo e a média diária, conferida
        contra a auditoria?
  \item Calcule a conta: $0.25$ euros por \unit{kW.h} mais uma
        assinatura fixa de $14$ euros.
  \item Qual é o preço de um megajoule vindo da rede? E vindo da
        gasolina, a $1.80$ euros o litro, que armazena \qty{34}{MJ}?
  \item Uma tigela de cereal de \qty{1.5}{MJ} custa cerca de $0.50$
        euros: qual é o preço do megajoule dela? Ordene os três
        vendedores de \omterm{def:g10:energy-conservation:energy}{joules}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O telhado.}
\begin{enumerate}[resume]
  \item O sol a pino entrega cerca de \qty{1000}{W} por \omterm{def:g10:orders-of-magnitude:unit}{metro} quadrado;
        os painéis o convertem com $\eta = 0.20$: qual é a \omterm{def:g10:energy-conservation:power}{potência}
        elétrica de \qty{1}{m^2} sob sol pleno?
  \item A região tem, em média, o equivalente a \qty{4.0}{h} de sol pleno
        por dia: qual é a produção diária de \qty{1}{m^2}?
  \item Que área de painéis cobre o dia de \qty{12.6}{kW.h} da família? O
        telhado voltado ao sul oferece \qty{20}{m^2}: cabe?
  \item Em dezembro o equivalente cai para \qty{1.5}{h}: qual é a
        produção dos painéis, a sua fatia da necessidade e o que preenche
        o buraco?
  \item Fabricar os painéis custou cerca de \qty{700}{kW.h} por \omterm{def:g10:orders-of-magnitude:unit}{metro}
        quadrado: qual é o tempo de retorno energético, usando a questão
        11?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A tigela de cereal.}
\begin{enumerate}[resume]
  \item Uma bicicleta geradora rende \qty{100}{W} de eletricidade:
        quantas horas de pedalada fazem \qty{12.6}{kW.h} --- quantas
        pessoas pedalando \qty{24}{h} sem parar?
  \item Os músculos trabalham com $\eta \approx 0.25$: qual é a \omterm{def:g10:energy-conservation:energy}{energia}
        de alimento por trás desses \qty{12.6}{kW.h}, em megajoules e em
        tigelas de cereal?
  \item Qual é o custo do dia pedalado (tigelas a $0.50$ euros) contra o
        que a rede cobra pelo mesmo dia?
  \item Um raio de \qty{5}{GJ} equivale a quantos dias de casa? Dada a
        sua duração de \qty{100}{\micro s}, por que ele ainda assim não
        consegue alimentar a casa?
  \item Os \omterm{def:g10:energy-conservation:energy}{joules} nunca se perdem e, mesmo assim, os \omterm{def:g10:energy-conservation:energy}{joules} da rede, da
        gasolina e do alimento são vendidos a preços muito diferentes: o
        que você está de fato pagando?
  \item Final: enuncie o veredito do dia em uma frase --- um \omterm{def:g10:orders-of-magnitude:unit}{quilograma}
        de gasolina, dezesseis \omterm{def:g10:orders-of-magnitude:unit}{metros} quadrados de telhado ensolarado ou
        cinco ciclistas sem parar --- e nomeie as duas grandezas (um
        estoque e um fluxo) que nunca mais devem ser confundidas.
\end{enumerate}
\end{problem}
