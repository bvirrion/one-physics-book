\chapter{Gravitação universal e peso}\label{ch:g10:universal-gravitation}

Solte uma maçã: ela cai. A Lua, a sessenta raios terrestres de altura,
nunca parece cair. A grande intuição de Newton é que a Lua \emph{está}
caindo --- cerca de 1.4 milímetro por segundo, exatamente o que prevê uma
atração que enfraquece com o quadrado da distância. Uma fórmula, com uma
constante universal, governa a maçã, a Lua, as marés e os planetas. O
primeiro volume descreveu a \omterm{def:g10:universal-gravitation:force}{gravitação} com palavras; este capítulo a
mede.

\section{A lei da gravitação universal}

\begin{definition}[Força gravitacional]\label{def:g10:universal-gravitation:force}
Dois corpos quaisquer se atraem, sejam feitos do que forem e estejam à
distância que estiverem. Essa atração é a \emph{força
gravitacional}\index{força gravitacional}, e o fenômeno é a
\emph{gravitação}\index{gravitação}. Ela não precisa de contato, nem de
corda, nem de meio, e é \emph{mútua}: cada corpo puxa o outro.
\end{definition}

\begin{theorem}[Lei da gravitação universal]\label{thm:g10:universal-gravitation:law}
Dois corpos de massas $m_1$ e $m_2$ (em \unit{kg}), cujos centros estão a
uma distância $d$ (em \unit{m}), atraem-se com duas forças dirigidas ao
longo da reta que une os centros, cada uma apontando para o outro corpo,
de mesmo módulo
\[
F = G\,\frac{m_1 m_2}{d^2}
\qquad \text{(em \unit{N})},
\]
em que $G$ é a mesma constante para todo par de corpos do universo.
\end{theorem}
\admitted

\begin{remark}[De onde vem a lei]
Newton destilou essa lei do movimento observado dos planetas (as leis de
Kepler); a dedução honesta é feita no volume do primeiro ano de
graduação. O \cref{ch:g12:satellites-kepler} já porá a lei para trabalhar
com \omterm{def:g10:universal-gravitation:satellite}{satélites} e planetas.
\end{remark}

\begin{definition}[A constante gravitacional]\label{def:g10:universal-gravitation:constant}
A constante
\[
G = \qty{6.67e-11}{N.m^2/kg^2}
\]
é a \emph{constante gravitacional}\index{constante gravitacional}. O seu
valor minúsculo é a razão pela qual a gravidade entre objetos do dia a
dia é imperceptível: só quando pelo menos uma das massas é astronômica é
que $G m_1 m_2 / d^2$ chega a alguma coisa.
\end{definition}

\begin{remark}[Ler a fórmula]\label{rem:g10:universal-gravitation:reading}
Quatro traços merecem atenção.
\begin{itemize}
  \item \emph{Mútuas e iguais.} A Terra puxa você com o seu \omterm{def:g10:universal-gravitation:weight}{peso} --- e
        você puxa a Terra exatamente com a mesma força. (A Terra apenas
        responde menos: ela é cerca de $\num{e23}$ vezes mais pesada.)
  \item \emph{Inverso do quadrado.} Dobrar $d$ divide a força por $4$;
        afastar-se dez vezes mais divide-a por $100$.
  \item \emph{Distância entre centros.} Para corpos esféricos ---
        estrelas, planetas, balas de canhão --- $d$ se mede de centro a
        centro, não de superfície a superfície.
  \item \emph{As duas massas contam.} Dobrar qualquer uma das massas
        dobra a força; a fórmula é simétrica em $m_1$ e $m_2$.
\end{itemize}
\end{remark}

\begin{example}[Dois amigos]\label{ex:g10:universal-gravitation:friends}
Dois amigos de \qty{70}{kg} estão de pé com os centros a \qty{1.0}{m} um
do outro:
\[
F = \num{6.67e-11} \times \frac{70 \times 70}{1.0^2}
\approx \qty{3.3e-7}{N}.
\]
O \omterm{def:g10:universal-gravitation:weight}{peso} de cada amigo é de cerca de \qty{6.9e2}{N} --- dois \emph{bilhões}
de vezes maior. A atração mútua deles é real, mas impossível de sentir.
\end{example}

\begin{example}[A Terra segura a Lua]\label{ex:g10:universal-gravitation:earthmoon}
Com $m_1 = \qty{5.97e24}{kg}$ (Terra), $m_2 = \qty{7.35e22}{kg}$ (Lua) e
$d = \qty{3.84e8}{m}$:
\[
F = \num{6.67e-11} \times
\frac{\num{5.97e24} \times \num{7.35e22}}{(\num{3.84e8})^2}
\approx \qty{2.0e20}{N}.
\]
E a Lua puxa a Terra de volta com os mesmos \qty{2.0e20}{N} --- um puxão
a que os oceanos respondem duas vezes por dia, na forma das marés.
\end{example}

\begin{method}[Calcular uma força gravitacional]\label{met:g10:universal-gravitation:compute}
\begin{enumerate}
  \item Converta as massas para \omterm{def:g10:orders-of-magnitude:unit}{quilogramas} e a distância de centro a
        centro para \omterm{def:g10:orders-of-magnitude:unit}{metros}.
  \item Aplique $F = G m_1 m_2 / d^2$.
  \item Confira a potência de dez separadamente dos algarismos, como no
        \cref{ch:g10:orders-of-magnitude}: as \omterm{def:g10:universal-gravitation:force}{forças gravitacionais} vão
        de $\num{e-7}$ newtons (duas pessoas a um \omterm{def:g10:orders-of-magnitude:unit}{metro}) a $\num{e22}$
        newtons (Sol sobre a Terra), de modo que um \omterm{def:g10:orders-of-magnitude:scinot}{expoente} fora do
        lugar é fácil de pegar.
\end{enumerate}
\end{method}

\section{Peso e intensidade da gravidade}

\begin{definition}[Peso]\label{def:g10:universal-gravitation:weight}
O \emph{peso}\index{peso} $\vect P$ de um objeto na superfície de um
astro (um planeta, uma lua, \dots) é a \omterm{def:g10:universal-gravitation:force}{força gravitacional} que o astro
exerce sobre ele. Ele aponta para o centro do astro --- a direção que
chamamos de \emph{vertical}\index{vertical} --- e se mede em newtons.
\end{definition}

\begin{proposition}[Peso na superfície de um astro]\label{prop:g10:universal-gravitation:surfaceg}
Na superfície de um astro esférico de massa $M$ e raio $R$, um objeto de
massa $m$ tem \omterm{def:g10:universal-gravitation:weight}{peso}
\[
P = m\,g,
\qquad\text{em que}\qquad
g = \frac{G M}{R^2}
\]
depende só do astro, e não do objeto.
\end{proposition}

\begin{proof}
O centro do objeto e o centro do astro estão a uma distância $R$, de modo
que a lei da \omterm{def:g10:universal-gravitation:force}{gravitação} universal
(\cref{thm:g10:universal-gravitation:law}) dá
$P = G M m / R^2 = m \times \left(G M / R^2\right)$. O parêntese é o
mesmo para todo objeto na superfície: chame-o de $g$.
\end{proof}

\begin{definition}[Intensidade do campo gravitacional]\label{def:g10:universal-gravitation:fieldstrength}
A grandeza $g = G M / R^2$, em newtons por \omterm{def:g10:orders-of-magnitude:unit}{quilograma}, é a
\emph{intensidade do campo gravitacional}\index{intensidade do campo
gravitacional} na superfície do astro: o astro puxa cada \omterm{def:g10:orders-of-magnitude:unit}{quilograma} com
$g$ newtons. Na Terra, $g = \qty{9.81}{N/kg}$.
\end{definition}

\begin{example}[Calcular $g$]\label{ex:g10:universal-gravitation:gvalues}
Para a Terra ($M = \qty{5.97e24}{kg}$, $R = \qty{6.37e6}{m}$):
\[
g = \frac{\num{6.67e-11} \times \num{5.97e24}}{(\num{6.37e6})^2}
= \frac{\num{3.98e14}}{\num{4.06e13}}
\approx \qty{9.81}{N/kg}.
\]
Para a Lua ($M = \qty{7.35e22}{kg}$, $R = \qty{1.74e6}{m}$) a mesma conta
dá $g_{\text{Lua}} \approx \qty{1.62}{N/kg}$ --- seis vezes menos: a Lua
é muito mais leve, e ser menor não compensa. Para Marte,
$g_{\text{Marte}} \approx \qty{3.73}{N/kg}$.
\end{example}

\begin{remark}[Por que os astronautas saltitam]\label{rem:g10:universal-gravitation:bounce}
Um astronauta de massa \qty{120}{kg} (corpo mais traje) pesa
$120 \times 9.81 \approx \qty{1180}{N}$ na Terra, mas apenas
$120 \times 1.62 \approx \qty{194}{N}$ na Lua. Músculos treinados sob
\qty{1180}{N} passam a carregar de repente um sexto da carga: daí o
famoso andar lunar lento e aos saltos. A massa --- e com ela o esforço
necessário para \emph{parar} ou \emph{virar} --- não muda, e é por isso
que os astronautas lunares caíam com tanta frequência.
\end{remark}

\section{Massa ou peso? A balança de pratos e o dinamômetro}

A linguagem cotidiana diz que um saco de farinha ``pesa três quilos''. A
física parte essa frase em duas.

\begin{remark}[Massa não é peso]\label{rem:g10:universal-gravitation:massweight}
\begin{itemize}
  \item A \emph{massa} $m$ mede a quantidade de matéria, em \omterm{def:g10:orders-of-magnitude:unit}{quilogramas}.
        É uma propriedade só do objeto: a mesma na Terra, na Lua ou à
        deriva no espaço profundo.
  \item O \emph{\omterm{def:g10:universal-gravitation:weight}{peso}} $P = mg$ é uma força, em newtons, exercida
        \emph{por um astro} sobre o objeto. Ele muda de astro para astro
        com $g$ --- e se apaga longe de todo astro.
\end{itemize}
Confundi-los é inofensivo na feira e fatal num problema de física: o
\omterm{def:g10:orders-of-magnitude:unit}{quilograma} não é \omterm{def:g10:orders-of-magnitude:unit}{unidade} de força.
\end{remark}

\begin{method}[A balança de pratos e o dinamômetro]\label{met:g10:universal-gravitation:instruments}
\begin{itemize}
  \item Uma \emph{balança de pratos} compara o objeto com massas padrão:
        os \omterm{def:g10:universal-gravitation:weight}{pesos} dos dois pratos são multiplicados pelo mesmo $g$, de
        modo que a comparação mede a \emph{massa} --- mesmo veredito em
        qualquer astro.
  \item Um \emph{dinamômetro} mede a força que estica a sua mola: ele lê
        o \emph{\omterm{def:g10:universal-gravitation:weight}{peso}}, em newtons, e a sua leitura muda de astro para
        astro. Para recuperar a massa, divida pela intensidade local do
        campo: $m = P/g$.
\end{itemize}
\end{method}

\begin{example}[Farinha na Lua]\label{ex:g10:universal-gravitation:flour}
Um saco de \qty{3.0}{kg} de farinha equilibra \qty{3.0}{kg} de massas
padrão na Terra \emph{e} na Lua. Um dinamômetro, porém, marca
$3.0 \times 9.81 \approx \qty{29}{N}$ na Terra e
$3.0 \times 1.62 \approx \qty{4.9}{N}$ na Lua. Um comerciante que
vendesse ``por newton'' na Lua entregaria seis vezes mais farinha --- o
comércio, como a ciência, funciona com massa.
\end{example}

\section{Cair em volta da Terra: o canhão de Newton}

Lance uma pedra horizontalmente: a gravidade curva a sua trajetória num
arco e ela cai a alguns \omterm{def:g10:orders-of-magnitude:unit}{metros} de distância. Lance-a mais rápido: ela cai
mais longe. A experiência de pensamento de Newton leva a ideia ao limite.
A Terra é redonda, e a sua superfície desce cerca de \qty{5}{m} abaixo da
horizontal a cada \qty{8}{km} percorridos; um corpo em queda livre desce
\qty{4.9}{m} no primeiro segundo (o
\cref{ch:g12:projectile-motion} estabelecerá isso). Logo, um projétil que
rase a Terra a \qty{8}{km/s} cai exatamente tão depressa quanto o chão se
curva embaixo dele: ele cai para sempre sem nunca aterrissar.

\begin{definition}[Satélite e órbita]\label{def:g10:universal-gravitation:satellite}
Um \emph{satélite}\index{satélite} de um astro é um corpo que cai em
volta do astro sem jamais alcançar a sua superfície; a trajetória que ele
repete é a sua \emph{órbita}\index{órbita}.
\end{definition}

\begin{remark}[A Lua é uma pedra caindo]\label{rem:g10:universal-gravitation:fallingmoon}
A Lua é o \omterm{def:g10:universal-gravitation:satellite}{satélite} natural da Terra: ela cai em nossa direção cerca de
\qty{1.4}{mm} por segundo, e o seu movimento lateral a leva em volta
antes que ela consiga se aproximar mais
(o \cref{exo:g10:universal-gravitation:15} deixa você refazer a própria
verificação de Newton desse número). Os astronautas de uma estação em
\omterm{def:g10:universal-gravitation:satellite}{órbita} estão na mesma queda livre da sua nave --- é isso, e não uma
ausência de gravidade, que os faz flutuar. A \omterm{def:g10:universal-gravitation:force}{gravitação} molda toda
trajetória, do arco da pedra à elipse do planeta; a história quantitativa
das \omterm{def:g10:universal-gravitation:satellite}{órbitas} é contada no \cref{ch:g12:satellites-kepler}.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:1}
Dois dançarinos de \qty{60}{kg} se abraçam com os centros a
\qty{0.80}{m} um do outro.
\begin{enumerate}
  \item Calcule a atração gravitacional mútua deles.
  \item Compare-a com o \omterm{def:g10:universal-gravitation:weight}{peso} de um mosquito de \qty{2.5}{mg}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:2}
Uma estudante tem massa de \qty{55}{kg}.
\begin{enumerate}
  \item Calcule o \omterm{def:g10:universal-gravitation:weight}{peso} dela na Terra.
  \item Ela viaja até a Lua. Quais são a massa e o \omterm{def:g10:universal-gravitation:weight}{peso} dela lá?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:3}
\begin{enumerate}
  \item A partir da ficha de dados, recalcule a \omterm{def:g10:universal-gravitation:fieldstrength}{intensidade do campo
        gravitacional} na superfície de Marte.
  \item Um jipe-robô de \qty{900}{kg} é enviado para lá. Calcule o \omterm{def:g10:universal-gravitation:weight}{peso}
        dele em Marte e o \omterm{def:g10:universal-gravitation:weight}{peso} dele na Terra.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:4}
Usando a ficha de dados, calcule a força exercida pela Terra sobre a Lua.
Que força a Lua exerce sobre a Terra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-gravitation:5}
Dois corpos se atraem com uma força $F_0$. Em que a força se transforma
se:
\begin{enumerate}
  \item a distância entre os centros é dobrada;
  \item a distância é reduzida à metade;
  \item as duas massas são dobradas (mesma distância);
  \item uma das massas é triplicada \emph{e} a distância é triplicada?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:6}
Dois superpetroleiros carregados, de \qty{5.0e8}{kg} cada, estão
atracados com os centros a \qty{100}{m} um do outro.
\begin{enumerate}
  \item Calcule a atração gravitacional mútua deles.
  \item Que massa tem, na Terra, \omterm{def:g10:universal-gravitation:weight}{peso} igual a essa força?
  \item A força é surpreendentemente grande --- e ainda assim os
        petroleiros não mostram inclinação alguma a se aproximar. Calcule
        a aceleração $a = F/m$ que ela dá a um dos navios e comente.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:7}
Uma loja lunar vende um saco de \qty{3.0}{kg} de arroz.
\begin{enumerate}
  \item O que indicam, para esse saco, uma balança de pratos e um
        dinamômetro na Terra? E na Lua?
  \item Um cliente paga por ``\qty{3.0}{kg}''. Que instrumento garante um
        negócio justo em qualquer mundo, e por quê?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:8}
A Estação Espacial Internacional voa a uma altitude de \qty{400}{km}.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:universal-gravitation:fieldstrength}{intensidade do campo gravitacional} da Terra nessa
        altitude. (Cuidado: qual é a distância até o \emph{centro} da
        Terra?)
  \item Exprima-a como porcentagem do valor na superfície.
  \item Os astronautas a bordo flutuam. Concilie isso com a sua resposta.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:9}
Na superfície de Vênus, $g = \qty{8.87}{N/kg}$, e o raio do planeta é
\qty{6.05e6}{m}. Deduza a massa de Vênus e compare-a com a da Terra.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:10}
\begin{enumerate}
  \item Calcule a força exercida pelo Sol sobre \qty{1.0}{kg} de água do
        mar e, em seguida, a força exercida pela Lua sobre esse mesmo
        \omterm{def:g10:orders-of-magnitude:unit}{quilograma}.
  \item O Sol puxa mais forte --- por que fator?
  \item E ainda assim a Lua domina as marés. A maré é comandada não pelo
        puxão em si, mas pela \emph{diferença} entre os puxões nos lados
        próximo e distante da Terra. Explique, sem calcular, por que a
        Lua, mais próxima, consegue vencer o Sol, mais distante, nesse
        critério.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-gravitation:11}
\begin{enumerate}
  \item Um planeta tem o dobro da massa da Terra e o dobro do seu raio.
        Exprima a intensidade do campo na sua superfície em função do $g$
        terrestre e depois em \unit{N/kg}.
  \item Que raio precisaria ter um planeta de massa igual à da Terra para
        que a intensidade do campo na sua superfície fosse $2g$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:12}
Em algum ponto da reta Terra--Lua os dois puxões sobre uma sonda espacial
se cancelam. Seja $x$ a distância do centro da Terra a esse ponto e $D$ a
distância Terra--Lua.
\begin{enumerate}
  \item Mostre que, nesse ponto,
        $\left(\dfrac{x}{D - x}\right)^{2}
        = \dfrac{M_{\text{Terra}}}{M_{\text{Lua}}}$.
  \item Deduza $x$. Que fração da viagem até a Lua isso representa?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:13}
O canhão de Newton, com números. Um projétil rasa a Terra
horizontalmente com velocidade $v$.
\begin{enumerate}
  \item Depois de $x = \qty{8.0}{km}$ de percurso horizontal, a Terra
        esférica ``desceu'' de uma altura $h$ que satisfaz
        $(R + h)^2 = R^2 + x^2$. Usando
        $(R + h)^2 \approx R^2 + 2Rh$ para $h$ pequeno, mostre que
        $h \approx x^2 / (2R)$ e calcule $h$.
  \item Um corpo em queda livre desce \qty{4.9}{m} no primeiro segundo.
        Compare com $h$ e deduza a velocidade com que o projétil nunca
        aterrissa.
  \item Calcule a duração de uma \omterm{def:g10:universal-gravitation:satellite}{órbita} completa a essa velocidade,
        rasando a superfície.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:14}
Uma estrela de nêutrons comprime $M = \qty{2.8e30}{kg}$ (1.4 vez a massa
do Sol) num raio de \qty{12}{km}.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:universal-gravitation:fieldstrength}{intensidade do campo gravitacional} na sua superfície.
  \item Quanto pesaria lá um ser humano de \qty{70}{kg}? Compare com o
        \omterm{def:g10:universal-gravitation:weight}{peso} dele na Terra.
  \item Qual é, lá, o \omterm{def:g10:universal-gravitation:weight}{peso} de um grão de areia de \qty{1.0}{mg}? Que
        massa tem esse \omterm{def:g10:universal-gravitation:weight}{peso} na Terra?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-gravitation:15}
O teste da Lua, de Newton (1666). O raio da \omterm{def:g10:universal-gravitation:satellite}{órbita} da Lua é de cerca de
$60$ raios terrestres.
\begin{enumerate}
  \item Por que fator o puxão da Terra por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} é mais fraco na Lua
        do que na superfície da Terra, se a gravidade segue o inverso do
        quadrado?
  \item Perto da superfície, um corpo em queda desce \qty{4.9}{m} no
        primeiro segundo. Quanto a Lua deveria cair em direção à Terra a
        cada segundo?
  \item Confira contra a \omterm{def:g10:universal-gravitation:satellite}{órbita}: a Lua percorre o seu círculo de raio
        $d = \qty{3.84e8}{m}$ em $T = 27.3$ dias. Calcule a sua
        velocidade $v$, a distância $x$ que ela cobre em um segundo e
        depois a queda $h \approx x^2/(2d)$ (como no
        \cref{exo:g10:universal-gravitation:13}).
  \item Compare as respostas das questões 2 e 3 e diga o que Newton
        concluiu.
\end{enumerate}
\end{exercise}

\section{Problema: pesar a Terra}

\begin{problem}[{Problema de fim de semana --- de duas esferas de chumbo
num galpão até a massa da Terra e depois a do Sol: quanto vale uma
constantezinha medida uma única vez}]
\label{pb:g10:universal-gravitation:1}
Em 1798, Henry Cavendish suspendeu de um fio fino uma haste leve que
carregava duas esferinhas de chumbo, aproximou duas esferas grandes de
chumbo e mediu a torção quase inexistente do fio. Os jornais disseram que
ele havia \emph{pesado a Terra} --- e tinham razão: conhecido $G$, o
próprio $g$ da Terra e o seu raio entregam a massa dela, e a \omterm{def:g10:universal-gravitation:satellite}{órbita} anual
da Terra entrega a do Sol. Este problema refaz a cadeia inteira. Use a
ficha de dados o tempo todo.

\medskip
\noindent\textbf{Parte I --- A força mais fraca de todas.}
\begin{enumerate}
  \item O ferro da torre Eiffel tem massa de cerca de \qty{7.3e6}{kg}. Um
        turista de \qty{55}{kg} está a \qty{100}{m} do seu centro de
        massa. Calcule o puxão da torre sobre o turista.
  \item Compare-o com o \omterm{def:g10:universal-gravitation:weight}{peso} de um grão de areia de \qty{1.0}{mg}.
  \item Exercício de proporcionalidade: o que acontece com uma \omterm{def:g10:universal-gravitation:force}{força
        gravitacional} se (a) a distância é multiplicada por $10$; (b) as
        duas massas são multiplicadas por $1000$; (c) as duas massas
        \emph{e} a distância são multiplicadas por $1000$?
  \item Calcule o puxão da Terra sobre um objeto de \qty{1.00}{kg} na sua
        superfície, a partir de $M$, $R$ e $G$. Que nome e que símbolo do
        dia a dia esse número carrega?
  \item A gravidade é de longe a mais fraca interação que você já
        encontrou --- e, mesmo assim, governa o universo. Explique em uma
        ou duas frases por que o puxão da Terra sobre você domina o do
        seu vizinho.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A balança de Cavendish.}
Na balança de torção, uma esfera grande de massa $m_1 = \qty{158}{kg}$
atrai uma pequena de massa $m_2 = \qty{0.73}{kg}$; os centros delas estão
a $d = \qty{0.23}{m}$.
\begin{enumerate}[resume]
  \item Com o valor moderno de $G$, calcule a força entre as duas
        esferas.
  \item Calcule o \omterm{def:g10:universal-gravitation:weight}{peso} da esfera pequena e a razão entre as duas forças.
        Por que Cavendish precisou de um fio de torção em vez de uma
        balança comum?
  \item A torção medida por Cavendish corresponde (em \omterm{def:g10:orders-of-magnitude:unit}{unidades} modernas)
        a $F = \qty{1.47e-7}{N}$. Deduza o valor de $G$ que ele obteve e
        compare-o com o de hoje.
  \item Com $G = \qty{6.67e-11}{N.m^2/kg^2}$, $g = \qty{9.81}{N/kg}$ e
        $R = \qty{6.37e6}{m}$, deduza a massa da Terra. (É este o passo
        que os jornais celebraram.)
  \item Deduza a densidade média da Terra $\rho = M / V$, com
        $V = \frac43 \pi R^3$. As rochas de superfície têm densidades em
        torno de \qty{2.7e3}{kg/m^3}: o que a comparação revela sobre o
        interior profundo?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Outros mundos.}
\begin{enumerate}[resume]
  \item A partir da ficha de dados, recalcule a intensidade do campo na
        superfície da Lua.
  \item Na Terra, um bom impulso eleva o seu centro de massa em
        \qty{0.50}{m}. Para o mesmo impulso, a altura alcançada é
        inversamente proporcional a $g$ (como o capítulo de energia,
        \cref{ch:g10:energy-conservation}, justificará). A que altura o
        mesmo salto leva você na Lua?
  \item Mercúrio: $M = \qty{3.30e23}{kg}$, $R = \qty{2.44e6}{m}$. Calcule
        a intensidade do campo na sua superfície e compare com Marte.
        Mercúrio é bem menor do que Marte --- como os dois valores podem
        estar tão próximos?
  \item A que altitude acima da superfície da Terra a intensidade do
        campo caiu à metade do seu valor na superfície?
  \item A que distância do centro da Terra o puxão terrestre por
        \omterm{def:g10:orders-of-magnitude:unit}{quilograma} cai ao valor da superfície da Lua, \qty{1.62}{N/kg}?
        Exprima-a em raios terrestres.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Pesar o Sol.}
A Terra percorre uma \omterm{def:g10:universal-gravitation:satellite}{órbita} quase circular de raio
$d = \qty{1.496e11}{m}$ em um ano, $T = \qty{3.156e7}{s}$. Admita (isso é
demonstrado no \cref{ch:g12:satellites-kepler}) que um corpo numa
trajetória circular de raio $d$ com velocidade $v$ precisa de um puxão de
$v^2/d$ newtons por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} em direção ao centro.
\begin{enumerate}[resume]
  \item Calcule a velocidade orbital $v$ da Terra.
  \item Deduza o puxão por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} que o Sol exerce sobre a Terra.
  \item Esse puxão vale também $G M_{\text{Sol}} / d^2$. Deduza a massa
        do Sol.
  \item Quantas Terras isso representa? Compare $M_{\text{Sol}}$ com
        $M_{\text{Terra}}$.
  \item Final. As esferas de Cavendish, a Terra, o Sol: uma única
        constante $G$ serviu para os três. Diga em uma frase o que a
        palavra \emph{universal} em ``\omterm{def:g10:universal-gravitation:force}{gravitação} universal'' nos comprou
        neste problema.
\end{enumerate}
\end{problem}
