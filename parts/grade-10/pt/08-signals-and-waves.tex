\chapter{Sinais e ondas}\label{ch:g10:signals-and-waves}

Toda mensagem que você já recebeu chegou como um \omterm{def:g10:signals-and-waves:signal}{sinal}: uma voz
pressionada contra o seu tímpano, esta página que lhe chegou como um
tremular de rádio. Por trás da variedade está uma única ideia --- uma
\omterm{def:g10:orders-of-magnitude:unit}{grandeza física} que varia no tempo --- e um único conjunto de
ferramentas: \omterm{def:g10:signals-and-waves:period}{período}, \omterm{def:g10:signals-and-waves:frequency}{frequência}, \omterm{def:g10:signals-and-waves:amplitude}{amplitude} e a arte de cronometrar um
atraso. Com ele, um barco mede o fundo do mar e um médico conta
batimentos.

\section{Os sinais carregam informação}

\begin{definition}[Sinal]\label{def:g10:signals-and-waves:signal}
Um \emph{sinal}\index{sinal} é uma \omterm{def:g10:orders-of-magnitude:unit}{grandeza física} que varia no tempo
para carregar informação: a \omterm{def:g10:pressure:pressure}{pressão} do ar no seu tímpano, a tensão nos
terminais de um microfone, o brilho da luz numa \omterm{def:g10:refraction:fiber}{fibra óptica}. Para
estudar um sinal, registre-o em função do tempo e leia a informação na
curva.
\end{definition}

Numa ligação telefônica a mesma informação troca de portador várias
vezes: \omterm{def:g10:pressure:pressure}{pressão} no ar, depois tensão no microfone, depois rádio na antena
--- e o caminho inverso na outra ponta. Um único vocabulário dá conta de
todos eles; este capítulo o constrói.

\section{Sinais periódicos}

\begin{definition}[Sinal periódico, período]\label{def:g10:signals-and-waves:period}
Um \omterm{def:g10:signals-and-waves:signal}{sinal} é \emph{periódico}\index{sinal periódico} quando se repete de
forma idêntica em intervalos regulares. O \emph{período}\index{período}
$T$ é a duração de um ciclo completo, em segundos.
\end{definition}

\begin{definition}[Frequência]\label{def:g10:signals-and-waves:frequency}
A \emph{frequência}\index{frequência} de um \omterm{def:g10:signals-and-waves:period}{sinal periódico} é o número de
ciclos por segundo:
\[
f = \frac{1}{T} .
\]
A sua \omterm{def:g10:orders-of-magnitude:unit}{unidade}, um ciclo por segundo, é o \emph{hertz}\index{hertz}
(\unit{Hz}), com os múltiplos habituais \unit{kHz}, \unit{MHz},
\unit{GHz}.
\end{definition}

\begin{definition}[Amplitude]\label{def:g10:signals-and-waves:amplitude}
A \emph{amplitude}\index{amplitude} de um \omterm{def:g10:signals-and-waves:period}{sinal periódico} é o seu
afastamento máximo em relação ao valor de repouso. Para uma tensão que
oscila simetricamente em torno de zero, a amplitude é o valor de pico; de
crista a vale mede-se o \emph{dobro} da amplitude (o valor de pico a
pico).
\end{definition}

\begin{example}[Rede elétrica e lá de concerto]\label{ex:g10:signals-and-waves:mains}
A tensão da rede elétrica oscila a $f = \qty{50}{Hz}$: o seu \omterm{def:g10:signals-and-waves:period}{período} é
$T = 1/f = \qty{0.020}{s} = \qty{20}{ms}$. O lá de concerto de uma
orquestra é um \omterm{def:g10:signals-and-waves:signal}{sinal} de \omterm{def:g10:pressure:pressure}{pressão} a \qty{440}{Hz}: um ciclo dura
$T = 1/440 = \qty{2.3}{ms}$. \omterm{def:g10:signals-and-waves:frequency}{Frequência} maior, ciclo mais curto: $f$ e
$T$ são inversos um do outro.
\end{example}

\section{Ler um oscilograma}

\begin{definition}[Osciloscópio]\label{def:g10:signals-and-waves:oscilloscope}
Um \emph{osciloscópio}\index{osciloscópio} traça uma tensão em função do
tempo; a curva na sua tela quadriculada é um
\emph{oscilograma}\index{oscilograma}. Dois ajustes convertem divisões da
tela em valores físicos: a \emph{base de tempo}\index{base de tempo}
(segundos por divisão horizontal) e o \emph{ganho}\index{ganho} vertical
(volts por divisão vertical).
\end{definition}

\begin{method}[Da tela para os números]\label{met:g10:signals-and-waves:oscillogram}
\begin{enumerate}
  \item Conte as divisões horizontais ocupadas por um ciclo completo (de
        crista a crista é o mais seguro); multiplique pela \omterm{def:g10:signals-and-waves:oscilloscope}{base de tempo}
        para obter $T$, e depois $f = 1/T$.
  \item Conte as divisões verticais da linha central até uma crista e
        multiplique pelo \omterm{def:g10:signals-and-waves:oscilloscope}{ganho} para obter a \omterm{def:g10:signals-and-waves:amplitude}{amplitude} (de crista a vale
        obtém-se o valor de pico a pico: divida pela metade).
\end{enumerate}
\end{method}

\begin{example}[Uma leitura completa]\label{ex:g10:signals-and-waves:reading}
No \omterm{def:g10:signals-and-waves:oscilloscope}{oscilograma} abaixo, a \omterm{def:g10:signals-and-waves:oscilloscope}{base de tempo} é \qty{5}{ms} por divisão e o
\omterm{def:g10:signals-and-waves:oscilloscope}{ganho}, \qty{2}{V} por divisão. Um ciclo ocupa $4.0$ divisões:
$T = 4.0 \times \qty{5}{ms} = \qty{20}{ms}$ e
$f = 1/\qty{0.020}{s} = \qty{50}{Hz}$ --- a rede elétrica de novo. Uma
crista fica $3.0$ divisões acima do eixo: a \omterm{def:g10:signals-and-waves:amplitude}{amplitude} é
$3.0 \times \qty{2}{V} = \qty{6.0}{V}$.
\end{example}

\section{Sinais sonoros e eletromagnéticos}

\begin{definition}[Som]\label{def:g10:signals-and-waves:sound}
O \emph{som}\index{som} é um \omterm{def:g10:signals-and-waves:signal}{sinal} de \omterm{def:g10:pressure:pressure}{pressão}: um objeto que vibra ---
corda, membrana, cordas vocais --- empurra o ar ritmicamente, e as
compressões viajam para longe a cerca de \qty{340}{m/s}. Um microfone
converte a \omterm{def:g10:pressure:pressure}{pressão} que chega numa tensão; o tímpano, em impulsos
nervosos. O som precisa de um meio: um sino tocado dentro de uma campânula
de vácuo emudece à medida que o ar é bombeado para fora.
\end{definition}

\begin{definition}[Onda]\label{def:g10:signals-and-waves:wave}
Uma perturbação que se propaga --- o padrão de compressões de um \omterm{def:g10:signals-and-waves:sound}{som}, a
ondulação num lago, um pulso de rádio --- é uma \emph{onda}\index{onda}.
O meio, quando existe, fica no lugar: cada porção de água balança onde
está enquanto a ondulação atravessa o lago. A geometria das ondas ---
\omterm{def:g10:light-spectra:wavelength}{comprimento de onda}, interferência, difração --- é retomada num capítulo
posterior; este capítulo só precisa das velocidades delas.
\end{definition}

\begin{definition}[Ultrassom e infrassom]\label{def:g10:signals-and-waves:ultrasound}
O ouvido humano escuta o \omterm{def:g10:signals-and-waves:sound}{som} entre cerca de \qty{20}{Hz} e \qty{20}{kHz}
(o teto baixa com a idade). O \omterm{def:g10:signals-and-waves:sound}{som} acima de \qty{20}{kHz} é
\emph{ultrassom}\index{ultrassom} --- morcegos caçam a \qty{50}{kHz} e os
aparelhos médicos formam imagens do corpo a alguns megahertz. O \omterm{def:g10:signals-and-waves:sound}{som}
abaixo de \qty{20}{Hz} é \emph{infrassom}\index{infrassom}, sentido por
elefantes e sismógrafos.
\end{definition}

\begin{proposition}[Sinais eletromagnéticos]\label{prop:g10:signals-and-waves:em}
O rádio, o Wi-Fi e a luz são \emph{sinais
eletromagnéticos}\index{sinal eletromagnético}: uma única família, todos
viajando pelo vácuo --- sem precisar de meio --- e todos com uma única
velocidade, a velocidade da luz do \cref{ch:g10:refraction},
\[
c = \qty{3.00e8}{m/s} \quad \text{(praticamente a mesma no ar).}
\]
\end{proposition}
\admitted

\begin{remark}\label{rem:g10:signals-and-waves:honest}
Aqui isso é um fato experimental --- a luz do Sol atravessa $150$ milhões
de quilômetros de espaço vazio; ela é honestamente deduzida no volume do
segundo ano de graduação a partir das leis da eletricidade e do
magnetismo. Guarde o contraste: o \omterm{def:g10:signals-and-waves:sound}{som} é uma vibração \emph{do} meio e
morre sem ele; um \omterm{prop:g10:signals-and-waves:em}{sinal eletromagnético} carrega o seu próprio campo
através do vácuo, um milhão de vezes mais rápido.
\end{remark}

\section{Ecos: distâncias a partir de atrasos}

Um \omterm{def:g10:signals-and-waves:signal}{sinal} que se propaga com velocidade $v$ percorre $d = v\,t$ num tempo
$t$: todo atraso é uma distância disfarçada. Melhor ainda, envie um pulso
curto contra um obstáculo e cronometre o seu \emph{eco}\index{eco} --- a
ida e volta cobre $2d$.

\begin{method}[Medir distâncias por eco]\label{met:g10:signals-and-waves:echo}
\begin{enumerate}
  \item Envie um pulso curto; meça o tempo $t$ de ida e volta do \omterm{rem:g10:signals-and-waves:honest}{eco}.
  \item Tome a velocidade $v$ do \omterm{def:g10:signals-and-waves:signal}{sinal} \emph{no meio atravessado}.
  \item O pulso cobriu $2d = v\,t$, logo $d = \dfrac{v\,t}{2}$.
\end{enumerate}
Pulsos de \omterm{def:g10:signals-and-waves:sound}{som} na água fazem o \emph{sonar}\index{sonar}
($v \approx \qty{1500}{m/s}$); pulsos de rádio no ar fazem o
\emph{radar}\index{radar} ($v = c$); pulsos de \omterm{def:g10:signals-and-waves:ultrasound}{ultrassom} no corpo fazem o
aparelho de ultrassonografia ($v \approx \qty{1540}{m/s}$ nos tecidos
moles).
\end{method}

\begin{example}[Sonar]\label{ex:g10:signals-and-waves:sonar}
O \omterm{met:g10:signals-and-waves:echo}{sonar} de um navio emite um pulso e o \omterm{rem:g10:signals-and-waves:honest}{eco} do fundo do mar volta depois
de $t = \qty{0.80}{s}$: a profundidade é
$d = \frac{1500 \times 0.80}{2} = \qty{600}{m}$. Esquecer o fator $2$
dobra o oceano.
\end{example}

\begin{example}[Radar]\label{ex:g10:signals-and-waves:radar}
Um \omterm{met:g10:signals-and-waves:echo}{radar} de aeroporto recebe o \omterm{rem:g10:signals-and-waves:honest}{eco} de um avião $\qty{0.30}{ms}$ depois do
pulso:
$d = \frac{\qty{3.00e8}{m/s} \times \qty{3.0e-4}{s}}{2} = \qty{45}{km}$.
Como $c$ é tão grande, o \omterm{met:g10:signals-and-waves:echo}{radar} vive de relógios de microssegundo:
cronometrar a luz é trabalho de precisão.
\end{example}

\begin{example}[O coração como sinal]\label{ex:g10:signals-and-waves:ecg}
Cada batimento cardíaco envia um pequeno pulso de tensão pelo tórax;
eletrodos o registram como o
\emph{eletrocardiograma}\index{eletrocardiograma} (ECG), um \omterm{def:g10:signals-and-waves:signal}{sinal} quase
\omterm{def:g10:signals-and-waves:period}{periódico}. Se os picos altos estão a $T = \qty{0.75}{s}$ um do outro, o
coração bate a $f = 1/0.75 = \qty{1.33}{Hz}$, ou seja,
$1.33 \times 60 = 80$ batimentos por minuto: ler $T$ num traçado é o modo
como o cardiologista toma o seu pulso.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:1}
Calcule a \omterm{def:g10:signals-and-waves:frequency}{frequência} do \omterm{def:g10:signals-and-waves:period}{sinal periódico} de \omterm{def:g10:signals-and-waves:period}{período} (a)
$T = \qty{20}{ms}$; (b) $T = \qty{4.0}{ms}$; (c)
$T = \qty{50}{\micro s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:2}
Calcule o \omterm{def:g10:signals-and-waves:period}{período} (a) da rede elétrica de \qty{50}{Hz}; (b) de um lá de
concerto de \qty{440}{Hz}; (c) de um \omterm{def:g10:signals-and-waves:signal}{sinal} de Wi-Fi de \qty{2.4}{GHz}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:3}
Classifique como \omterm{def:g10:signals-and-waves:ultrasound}{infrassom}, \omterm{def:g10:signals-and-waves:sound}{som} audível ou \omterm{def:g10:signals-and-waves:ultrasound}{ultrassom}: \qty{12}{Hz};
\qty{440}{Hz}; \qty{18}{kHz}; \qty{40}{kHz} (um sensor de
estacionamento); \qty{5}{MHz} (uma sonda médica).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:4}
Num \omterm{def:g10:signals-and-waves:oscilloscope}{osciloscópio} ajustado em \qty{2}{ms} por divisão e \qty{0.5}{V} por
divisão, um ciclo ocupa $5.0$ divisões e uma crista fica $4.0$ divisões
acima do eixo. Encontre $T$, $f$ e a \omterm{def:g10:signals-and-waves:amplitude}{amplitude}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:5}
Você vê o relâmpago e, \qty{6.0}{s} depois, ouve o trovão. A que
distância ele caiu ($v_{\text{som}} = \qty{340}{m/s}$)? Por que se pode
desprezar o tempo de percurso da luz?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:6}
Um \omterm{rem:g10:signals-and-waves:honest}{eco} de \omterm{met:g10:signals-and-waves:echo}{sonar} volta do fundo do mar depois de \qty{0.60}{s}
($v = \qty{1500}{m/s}$ na água do mar). Por que o produto $v\,t$ precisa
ser dividido pela metade? Calcule a profundidade. Quanto tempo o \omterm{rem:g10:signals-and-waves:honest}{eco}
levaria sobre uma fossa de \qty{1200}{m} de profundidade?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:7}
Um \omterm{met:g10:signals-and-waves:echo}{radar} de aeroporto recebe o \omterm{rem:g10:signals-and-waves:honest}{eco} de um avião depois de \qty{0.24}{ms}
e, exatamente \qty{1.0}{s} mais tarde, um segundo \omterm{rem:g10:signals-and-waves:honest}{eco} depois de
\qty{0.238}{ms}. Calcule as duas distâncias e depois a velocidade do

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:10}

\section{Problema: o sonar, a tempestade e o cardiograma}

\begin{problem}[{Problema de fim de semana --- ler os sinais do mundo:
uma única lei, $d = v\,t$, mede a distância de uma tempestade, mapeia um
fundo de mar, conta um batimento cardíaco e revela como funciona o GPS}]
\label{pb:g10:signals-and-waves:1}
Um barco de pesca trabalha ao longo de uma noite de tempestade:
relâmpagos no horizonte, o \omterm{met:g10:signals-and-waves:echo}{sonar} varrendo o fundo, o coração do mestre na
tira de papel do médico, um receptor de GPS escutando \omterm{def:g10:universal-gravitation:satellite}{satélites}. Quatro
instrumentos, uma lei. Tome $v_{\text{som}} = \qty{340}{m/s}$ no ar,
\qty{1500}{m/s} na água do mar e $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Parte I --- A tempestade.}
\begin{enumerate}
  \item Um clarão; o trovão vem \qty{9.0}{s} depois. A que distância ele
        está?
  \item Calcule o tempo de percurso da luz nessa distância e justifique
        tratar o clarão como instantâneo.
  \item Os marinheiros contam os segundos entre o clarão e o trovão e
        dividem por três para obter quilômetros. Justifique a regra.
  \item Cinco minutos depois, um clarão dá \qty{6.0}{s}. A que distância
        agora? Encontre a \omterm{def:g10:relative-motion:speed}{velocidade média} de aproximação da tempestade
        em \unit{m/s} e em \unit{km/h}.
  \item O trovão ronca em vez de estalar: o canal do relâmpago tem
        quilômetros de comprimento, de modo que as suas partes estão a
        distâncias diferentes. Se o canal se estende de \qty{2.0}{km} a
        \qty{5.0}{km} do barco, quanto tempo dura o ronco?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O \omterm{met:g10:signals-and-waves:echo}{sonar}.}
\begin{enumerate}[resume]
  \item O \omterm{met:g10:signals-and-waves:echo}{sonar} emite um pulso; o fundo responde em \qty{0.90}{s}. Qual é
        a profundidade?
  \item Sobre a plataforma continental o \omterm{rem:g10:signals-and-waves:honest}{eco} encurta para \qty{0.20}{s}.
        Qual é a profundidade?
  \item Um pulso devolve \emph{dois} \omterm{rem:g10:signals-and-waves:honest}{ecos}, em \qty{0.16}{s} e
        \qty{0.20}{s}. Interprete-os; a que altura do fundo nada o cardume?
  \item O \omterm{met:g10:signals-and-waves:echo}{sonar} emite um pulso a cada \qty{0.50}{s}. Qual é a maior
        profundidade que ele pode medir sem confusão, e o que dá errado
        além dela?
  \item No ar, o mesmo atraso de \qty{0.60}{s} significaria que
        distância? Moral: o que é preciso saber antes de transformar um
        atraso em distância?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O cardiograma.}
\begin{enumerate}[resume]
  \item Os picos do ECG do mestre estão a \qty{0.86}{s} um do outro. Qual
        é a \omterm{def:g10:signals-and-waves:frequency}{frequência}? E os batimentos por minuto?
  \item A tira avança a \qty{25}{mm/s}. Que espaçamento entre picos o
        médico mediu?
  \item Depois de recolher as redes, o espaçamento é de \qty{15}{mm}.
        Qual é a nova \omterm{def:g10:signals-and-waves:frequency}{frequência} cardíaca?
  \item O jovem marinheiro, remador treinado, fica em repouso a $50$
        batimentos por minuto. Qual é o \omterm{def:g10:signals-and-waves:period}{período}? E o espaçamento na tira?
  \item Um ECG é rigorosamente \omterm{def:g10:signals-and-waves:period}{periódico}? O que, então, o médico lê no
        traçado?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- De volta para casa pela luz.}
\begin{enumerate}[resume]
  \item O barco chama o porto pelo rádio, a \qty{50}{km}. Quanto tempo a
        mensagem leva? Quanto tempo o \omterm{def:g10:signals-and-waves:sound}{som} levaria?
  \item Um \omterm{def:g10:universal-gravitation:satellite}{satélite} de GPS orbita a cerca de \qty{20200}{km} de altitude.
        Quanto tempo o seu \omterm{def:g10:signals-and-waves:signal}{sinal} leva para chegar ao barco?
  \item O GPS transforma tempo em posição. Que erro de distância um erro
        de relógio de \qty{1.0}{\micro s} provoca? Que precisão de
        cronometragem uma localização com \qty{3.0}{m} de erro exige?
  \item Por que não poderia existir um GPS baseado em \omterm{def:g10:signals-and-waves:sound}{som}, nem mesmo em
        princípio?
  \item Final: liste as três velocidades usadas nesta noite e a única lei
        por trás dos quatro instrumentos, e enuncie em uma frase o hábito
        de medir distâncias por atraso.
\end{enumerate}
\end{problem}
