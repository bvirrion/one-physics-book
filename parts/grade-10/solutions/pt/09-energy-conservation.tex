\section*{Capítulo \ref{ch:g10:energy-conservation} --- Energia: formas e conservação}

\begin{solution}{exo:g10:energy-conservation:1}
$E_k = \tfrac12 \times 1200 \times 25^2 = \qty{3.75e5}{J}$. A
\qty{36.1}{m/s}:
$\tfrac12 \times 1200 \times 36.1^2 \approx \qty{7.8e5}{J}$ --- $44\%$ a
mais de velocidade, $(130/90)^2 \approx 2.1$ vezes a \omterm{def:g10:energy-conservation:kinetic}{energia cinética}: é
o quadrado em ação.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:2}
$\Delta h = \qty{450}{m}$:
$E_p = 72 \times 9.81 \times 450 \approx \qty{3.2e5}{J}
= \num{3.2e5}/\num{3.6e6} \approx \qty{0.088}{kW.h}$, que vale cerca de
$0.022$ euros. Um dia de subida são dois centavos de eletricidade.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:3}
$\qty{1}{kW.h} = \qty{3.6e6}{J}$;
$\qty{12}{W.h} = 12 \times \qty{3600}{J} \approx \qty{43}{kJ}$;
$\qty{45}{MJ} = 45/3.6 = \qty{12.5}{kW.h}$;
$\qty{10}{MJ} = 10/3.6 \approx \qty{2.8}{kW.h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:4}
$E = 2200 \times 150 = \qty{3.3e5}{J}
= \num{3.3e5}/\num{3.6e6} \approx \qty{0.092}{kW.h}$. Ser humano:
$P = \num{1.0e7}/\num{86400} \approx \qty{116}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:5}
Dínamo: química (comida do ciclista) $\to$ cinética (roda) $\to$ elétrica
(dínamo) $\to$ radiante $+$ térmica (lâmpada); há vazamento de calor nos
músculos, no contato do pneu, no dínamo e na lâmpada. Aquecedor a gás:
química (gás) $\to$ térmica (água), com vazamento nos gases da chaminé.
Painel: radiante (Sol) $\to$ elétrica (painel) $\to$ química (bateria),
com vazamento de calor no painel e no carregador.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:6}
Consumida: $2.0 \times 45 = \qty{90}{MJ}$; $\eta = 27/90 = 0.30$. Calor:
$90 - 27 = \qty{63}{MJ}$, levados pelos gases de escapamento e pelo
radiador para o ar em volta.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:7}
$E_k = E_p = 70 \times 9.81 \times 20 \approx \qty{1.37e4}{J}$;
$v = \sqrt{2 \times 13734/70} = \sqrt{392} \approx \qty{19.8}{m/s}
\approx \qty{71}{km/h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:8}
$41562 - 41236 = \qty{326}{kW.h}$; média diária
$326/30 \approx \qty{10.9}{kW.h}$. Conta:
$326 \times 0.26 + 12 = 96.76$ euros. \omterm{def:g10:energy-conservation:power}{Potência} média:
$\qty{326}{kW.h}/\qty{720}{h} \approx \qty{453}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:9}
\emph{1.} Luz: $0.05 \times 60 = \qty{3.0}{W}$; LED:
$\eta = 3/9 \approx 0.33$.
\emph{2.} Por ano ($\qty{1460}{h}$): lâmpada
$60 \times 1460 = \qty{87.6}{kW.h}$, LED \qty{13.1}{kW.h}; economia de
$\qty{74.5}{kW.h} \approx 19$ euros por lâmpada por ano.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:10}
\emph{1.} $P = \num{1.5e5} \times 9.81 \times 40 \approx

\emph{2.} $0.90 \times 59 \approx \qty{53}{MW}$;

\begin{solution}{exo:g10:energy-conservation:11}
$v = \sqrt{2 \times 9.81 \times 0.050} \approx \qty{0.99}{m/s}$ (a massa
se cancela: $\tfrac12 m v^2 = mgh$). A \omterm{def:g10:energy-conservation:energy}{energia} potencial inicial
$E_p = 0.20 \times 9.81 \times 0.050 \approx \qty{0.098}{J}$ termina como
\omterm{def:g10:energy-conservation:forms}{energia térmica} no ar e no eixo: o pêndulo não é isolado, e o total ---
oscilação mais calor --- se conserva exatamente.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:12}
\emph{1.} $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.304$; consumida
$12/0.304 \approx \qty{39.5}{W.h} = 39.5 \times 3600 \approx
\qty{142}{kJ}$.
\emph{2.} $\num{1.42e5}/\num{2.5e7} \approx \qty{5.7}{g}$ de carvão por
carga; $\times 365 \approx \qty{2.1}{kg}$ por ano.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:13}
\emph{1.} $\num{5e9}/\num{3.6e6} \approx \qty{1400}{kW.h}$;
$1400/12.6 \approx 110$ dias.
\emph{2.} $P = \num{5e9}/\num{1e-4} = \qty{5e13}{W} = \qty{50}{TW}$ ---
dezessete vezes a \omterm{def:g10:energy-conservation:power}{potência} elétrica média da humanidade, por
\qty{100}{\micro s} e num lugar imprevisível. Um raio é muita \omterm{def:g10:energy-conservation:power}{potência} e
pouca \omterm{def:g10:energy-conservation:energy}{energia}, entregue depressa demais para servir: \omterm{def:g10:energy-conservation:transfer}{conversor} nenhum
aceita um pico desses.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:14}
\emph{1.} $E_k = \tfrac12 \times 1300 \times 30.6^2 \approx
\qty{6.1e5}{J}$, convertida em \omterm{def:g10:energy-conservation:forms}{energia térmica} nos discos e nas pastilhas
de freio (e depois no ar).
\emph{2.} $0.60 \times \qty{6.1e5}{J} \approx \qty{3.7e5}{J}$ por parada;
$\num{3.6e6}/\num{3.7e5} \approx 10$ paradas por \omterm{def:g10:energy-conservation:kwh}{quilowatt-hora}.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:15}
$E_p = 85 \times 9.81 \times 1200 \approx \qty{1.0}{MJ}$; alimento:
$1.0/0.25 = \qty{4.0}{MJ}$, ou seja, $4.0/1.5 \approx 2.7$ tigelas de
cereal --- ou $4.0/45 \approx \qty{0.09}{kg}$ de gasolina. Uma manhã
inteira de esforço honesto cabe numa xícara de café de combustível: a
\omterm{def:g10:energy-conservation:forms}{energia química} é absurdamente concentrada.
\end{solution}

\begin{solution}{pb:g10:energy-conservation:1}
\textbf{1.} Geladeira $0.1 \times 24 = \qty{2.4}{kW.h}$; aquecedor
$2.4 \times 2.5 = \qty{6.0}{kW.h}$; forno \qty{2.0}{kW.h}; máquina de
lavar \qty{1.0}{kW.h}; iluminação \qty{0.3}{kW.h}; telas
\qty{0.9}{kW.h}.
\textbf{2.} Total \qty{12.6}{kW.h} $= 12.6 \times 3.6 \approx
\qty{45}{MJ}$: um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de gasolina, \omterm{def:g10:energy-conservation:energy}{joule} por \omterm{def:g10:energy-conservation:energy}{joule}.
\textbf{3.} $\qty{12600}{W.h}/\qty{24}{h} = \qty{525}{W}$ --- a casa fica
em marcha lenta a um quarto de chaleira.
\textbf{4.} O aquecedor de água (\qty{6.0}{kW.h}, quase metade). Com
\qty{1.25}{h}: aquecedor \qty{3.0}{kW.h}, total \qty{9.6}{kW.h}.
\textbf{5.} $15 \times 24 = \qty{0.36}{kW.h}$ por dia, cerca de $2.8\%$
do total medido; por ano,
$0.36 \times 365 \approx \qty{131}{kW.h} \approx 33$ euros --- por luzes
que ninguém olha.
\textbf{6.} $56604 - 56214 = \qty{390}{kW.h}$;
$390/30 = \qty{13.0}{kW.h}$ por dia: os \qty{12.6}{kW.h} da auditoria
mais os \qty{0.36}{kW.h} do modo de espera --- confirmado.
\textbf{7.} $390 \times 0.25 + 14 = 111.50$ euros.
\textbf{8.} Rede: $0.25/3.6 \approx 0.07$ euros por megajoule. Gasolina:
$1.80/34 \approx 0.05$ euros por megajoule.
\textbf{9.} Tigela: $0.50/1.5 \approx 0.33$ euros por megajoule ---
cinco vezes a rede. Do mais barato ao mais caro: gasolina, rede e, bem
atrás, alimento.
\textbf{10.} $0.20 \times 1000 = \qty{200}{W}$ por \omterm{def:g10:orders-of-magnitude:unit}{metro} quadrado.
\textbf{11.} $200 \times 4.0 = \qty{800}{W.h} = \qty{0.80}{kW.h}$ por
\omterm{def:g10:orders-of-magnitude:unit}{metro} quadrado por dia.
\textbf{12.} $12.6/0.80 = 15.75 \approx \qty{16}{m^2}$: cabe no telhado
de \qty{20}{m^2}.
\textbf{13.} $16 \times 0.2 \times 1.5 = \qty{4.8}{kW.h}$: cerca de
$38\%$ da necessidade; a rede (ou uma bateria carregada em meses
melhores) tem de fornecer o resto.
\textbf{14.} Um \omterm{def:g10:orders-of-magnitude:unit}{metro} quadrado rende
$0.80 \times 365 \approx \qty{292}{kW.h}$ por ano; retorno
$700/292 \approx 2.4$ anos --- pouco diante de uma vida útil de 25 anos
do painel.
\textbf{15.} $\qty{12.6}{kW.h}/\qty{0.1}{kW} = \qty{126}{h}$: mais de
cinco pessoas ($126/24 = 5.25$) pedalando dia e noite.
\textbf{16.} Alimento $= 12.6/0.25 = \qty{50.4}{kW.h} \approx
\qty{181}{MJ}$, ou seja, $181/1.5 \approx 121$ tigelas de cereal.
\textbf{17.} $121 \times 0.50 \approx 60$ euros, contra
$12.6 \times 0.25 \approx 3.15$ euros vindos da rede: a \omterm{def:g10:inertia:force}{força} da pedalada
custa cerca de $19$ vezes mais --- e isso antes de pagar os ciclistas.
\textbf{18.} $5000/45 \approx 110$ dias de casa. Mas
$\num{5e9}/\num{1e-4} = \qty{5e13}{W}$ por \qty{100}{\micro s}, num lugar
qualquer: \omterm{def:g10:energy-conservation:transfer}{conversor} ou bateria nenhuma bebe dessa mangueira. Uma casa
precisa de um fluxo constante de \qty{525}{W}, não de um pico.
\textbf{19.} Não são os \omterm{def:g10:energy-conservation:energy}{joules} --- eles se conservam e são idênticos.
Você paga pela \emph{forma e pela disponibilidade} deles: \omterm{def:g10:energy-conservation:energy}{energia}
concentrada, armazenável e disponível na hora custa toda a \omterm{def:g10:energy-conservation:transfer}{cadeia de
conversão} que a deixou assim.
\textbf{20.} Um dia da família $= \qty{12.6}{kW.h} \approx \qty{45}{MJ}$:
um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de gasolina, dezesseis \omterm{def:g10:orders-of-magnitude:unit}{metros} quadrados de telhado
ensolarado ou cinco ciclistas sem parar. O estoque é a \omterm{def:g10:energy-conservation:energy}{energia} (\omterm{def:g10:energy-conservation:energy}{joules},
\omterm{def:g10:energy-conservation:kwh}{quilowatt-horas}); o fluxo é a \omterm{def:g10:energy-conservation:power}{potência} (\omterm{def:g10:energy-conservation:power}{watts}) --- a banheira e a
torneira, nunca mais confundidas.
\end{solution}
