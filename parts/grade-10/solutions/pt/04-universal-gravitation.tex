\section*{Capítulo \ref{ch:g10:universal-gravitation} --- Gravitação universal e peso}

\begin{solution}{exo:g10:universal-gravitation:1}
\emph{1.} $F = \num{6.67e-11} \times \dfrac{60 \times 60}{0.80^2}
= \dfrac{\num{2.40e-7}}{0.64} \approx \qty{3.8e-7}{N}$.

\emph{2.} O \omterm{def:g10:universal-gravitation:weight}{peso} do mosquito é
$P = \num{2.5e-6} \times 9.81 \approx \qty{2.5e-5}{N}$ --- cerca de $65$
vezes \emph{maior} do que a atração mútua dos dançarinos. A gravidade
entre pessoas é mais fraca do que o \omterm{def:g10:universal-gravitation:weight}{peso} de um mosquito.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:2}
\emph{1.} $P = mg = 55 \times 9.81 \approx \qty{5.4e2}{N}$.

\emph{2.} A massa dela não muda: \qty{55}{kg} (mesma quantidade de
matéria). O \omterm{def:g10:universal-gravitation:weight}{peso} passa a
$P = 55 \times 1.62 \approx \qty{89}{N}$ --- seis vezes menos.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:3}
\emph{1.} $g_{\text{Marte}} = \dfrac{G M}{R^2}
= \dfrac{\num{6.67e-11} \times \num{6.42e23}}{(\num{3.39e6})^2}
= \dfrac{\num{4.28e13}}{\num{1.15e13}} \approx \qty{3.73}{N/kg}$.

\emph{2.} Em Marte: $P = 900 \times 3.73 \approx \qty{3.4e3}{N}$. Na
Terra: $P = 900 \times 9.81 \approx \qty{8.8e3}{N}$ --- o jipe-robô pesa
cerca de $2.6$ vezes menos em Marte, com a mesma massa.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:4}
\[
F = \num{6.67e-11} \times
\frac{\num{5.97e24} \times \num{7.35e22}}{(\num{3.84e8})^2}
= \frac{\num{2.93e37}}{\num{1.47e17}}

A Lua exerce sobre a Terra uma força de \emph{mesmo} módulo,

\end{solution}

\begin{solution}{exo:g10:universal-gravitation:5}
\emph{1.} Distância dobrada: $d^2$ fica multiplicado por $4$, logo
$F = F_0/4$.

\emph{2.} Distância pela metade: $F = 4F_0$.

\emph{3.} As duas massas dobradas: $m_1 m_2$ fica multiplicado por $4$,
logo $F = 4F_0$.

\emph{4.} Numerador $\times 3$, denominador $\times 9$: $F = F_0/3$.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:6}
\emph{1.} $F = \num{6.67e-11} \times
\dfrac{(\num{5.0e8})^2}{100^2}
= \num{6.67e-11} \times \dfrac{\num{2.5e17}}{\num{1.0e4}}
\approx \qty{1.7e3}{N}$.

\emph{2.} $m = F/g = 1667/9.81 \approx \qty{1.7e2}{kg}$: os dois navios
se atraem com o \omterm{def:g10:universal-gravitation:weight}{peso} de uma motocicleta grande.

\emph{3.} $a = F/m = \num{1.7e3}/\num{5.0e8} \approx
\qty{3.3e-6}{m/s^2}$. Mesmo sem oposição, os petroleiros levariam cerca
de cinco dias para atingir a velocidade de uma caminhada; na prática, a
resistência da água e as amarras engolem a atração por completo.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:7}
\emph{1.} A balança de pratos indica \qty{3.0}{kg} nos dois mundos: o
saco e as massas padrão têm os seus \omterm{def:g10:universal-gravitation:weight}{pesos} divididos pelo mesmo fator, de
modo que a comparação não muda. O dinamômetro lê o \omterm{def:g10:universal-gravitation:weight}{peso}:
$3.0 \times 9.81 \approx \qty{29}{N}$ na Terra e
$3.0 \times 1.62 \approx \qty{4.9}{N}$ na Lua.

\emph{2.} A balança: ela mede a massa, que é o que o cliente está
comprando. Um dinamômetro calibrado em ``\omterm{def:g10:orders-of-magnitude:unit}{quilogramas}'' na Terra marcaria
$4.9/9.81 \approx \qty{0.50}{kg}$ para o mesmo saco na Lua --- o lojista
entregaria seis sacos pelo preço de um.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:8}
\emph{1.} Distância até o centro:
$d = R + h = \num{6.37e6} + \num{4.0e5} = \qty{6.77e6}{m}$. Então
$g = \dfrac{\num{6.67e-11} \times \num{5.97e24}}{(\num{6.77e6})^2}
= \dfrac{\num{3.98e14}}{\num{4.58e13}} \approx \qty{8.69}{N/kg}$.

\emph{2.} $8.69/9.81 \approx 89\%$ do valor na superfície.

\emph{3.} A gravidade está quase com força total lá em cima: os
astronautas flutuam porque a estação e tudo o que há dentro dela estão
\emph{caindo juntos} em volta da Terra (queda livre), e não porque a
gravidade tenha desaparecido.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:9}
De $g = GM/R^2$,
\[
M = \frac{g R^2}{G}
= \frac{8.87 \times (\num{6.05e6})^2}{\num{6.67e-11}}
= \frac{\num{3.25e14}}{\num{6.67e-11}}
\approx \qty{4.87e24}{kg},
\]
cerca de $0.8$ vez a massa da Terra --- Vênus é quase a gêmea da Terra em
tamanho e em massa.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:10}
\emph{1.} Sol:
$F = \dfrac{\num{6.67e-11} \times \num{1.99e30}}{(\num{1.496e11})^2}
\approx \qty{5.9e-3}{N}$. Lua:
$F = \dfrac{\num{6.67e-11} \times \num{7.35e22}}{(\num{3.84e8})^2}
\approx \qty{3.3e-5}{N}$.

\emph{2.} O Sol puxa cerca de $\num{5.9e-3}/\num{3.3e-5} \approx 180$
vezes mais forte.

\emph{3.} O diâmetro da Terra é uma \emph{fração} muito maior da
distância Terra--Lua ($\approx 3\%$) do que da distância Terra--Sol
($\approx 0.01\%$), de modo que o puxão da Lua varia muito mais do lado
próximo ao lado distante do globo. As marés se alimentam dessa diferença
--- e aí a Lua ganha.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:11}
\emph{1.} $g' = \dfrac{G (2M)}{(2R)^2} = \dfrac{2}{4}\,\dfrac{GM}{R^2}
= \dfrac{g}{2} \approx \qty{4.9}{N/kg}$: o raio dobrado vence a massa
dobrada.

\emph{2.} Precisamos de $\dfrac{GM}{R'^2} = 2\,\dfrac{GM}{R^2}$, logo
$R'^2 = R^2/2$ e $R' = R/\sqrt{2} \approx \qty{4.5e6}{m}$ --- cerca de
\qty{4500}{km}.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:12}
\emph{1.} Puxões por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} iguais:
$\dfrac{G M_{\text{Terra}}}{x^2} = \dfrac{G M_{\text{Lua}}}{(D-x)^2}$.
Multiplicando cruzado e dividindo por $G$:
$\left(\dfrac{x}{D-x}\right)^2 = \dfrac{M_{\text{Terra}}}{M_{\text{Lua}}}$.

\emph{2.} $\dfrac{x}{D-x}
= \sqrt{\dfrac{\num{5.97e24}}{\num{7.35e22}}} = \sqrt{81.2} \approx
9.0$, logo $x = 9.0\,(D - x)$, ou seja, $x = \dfrac{9.0}{10.0}\,D
\approx 0.90 \times \num{3.84e8} \approx \qty{3.5e8}{m}$. O ponto de
equilíbrio fica a $90\%$ do caminho até a Lua: o puxão da Terra domina
quase a viagem inteira.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:13}
\emph{1.} Desenvolvendo: $R^2 + 2Rh + h^2 = R^2 + x^2$; como $h$ é
minúsculo comparado com $R$, descartamos $h^2$: $2Rh \approx x^2$, logo
$h \approx \dfrac{x^2}{2R}
= \dfrac{(\num{8.0e3})^2}{2 \times \num{6.37e6}}
= \dfrac{\num{6.4e7}}{\num{1.27e7}} \approx \qty{5.0}{m}$.

\emph{2.} Em um segundo o projétil cai \qty{4.9}{m} --- quase exatamente
os \qty{5.0}{m} que o chão desce ao longo de \qty{8.0}{km}. Logo, a
$v \approx \qty{8.0}{km/s}$ a queda nunca alcança o chão: o projétil
entra em \omterm{def:g10:universal-gravitation:satellite}{órbita}.

\emph{3.} $T = \dfrac{2\pi R}{v}
= \dfrac{2\pi \times \num{6.37e6}}{\num{8.0e3}}
\approx \qty{5.0e3}{s} \approx \qty{83}{min}$ --- perto dos cerca de
$\qty{90}{min}$ dos \omterm{def:g10:universal-gravitation:satellite}{satélites} reais de \omterm{def:g10:universal-gravitation:satellite}{órbita} baixa, que voam um pouco
mais alto.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:14}
\emph{1.} $g = \dfrac{\num{6.67e-11} \times \num{2.8e30}}
{(\num{1.2e4})^2} = \dfrac{\num{1.87e20}}{\num{1.44e8}}
\approx \qty{1.3e12}{N/kg}.$

\emph{2.} $P = 70 \times \num{1.3e12} \approx \qty{9.1e13}{N}$, contra
$\qty{6.9e2}{N}$ na Terra: cerca de $\num{1.3e11}$ vezes mais. Estrutura
alguma feita de átomos sobrevive de pé ali.

\emph{3.} $P = \num{1.0e-6} \times \num{1.3e12} \approx
\qty{1.3e6}{N}$. Na Terra isso é o \omterm{def:g10:universal-gravitation:weight}{peso} de
$m = \num{1.3e6}/9.81 \approx \qty{1.3e5}{kg}$ --- um grão de areia
pesando tanto quanto uma locomotiva de cem toneladas.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:15}
\emph{1.} A distância é $60$ vezes maior, de modo que o puxão por
\omterm{def:g10:orders-of-magnitude:unit}{quilograma} é $60^2 = 3600$ vezes mais fraco.

\emph{2.} A queda em um segundo escala do mesmo jeito:
$4.9/3600 \approx \qty{1.4e-3}{m}$ --- cerca de \qty{1.4}{mm}.

\emph{3.} $T = 27.3 \times \qty{86400}{s} = \qty{2.36e6}{s}$, logo
$v = \dfrac{2\pi \times \num{3.84e8}}{\num{2.36e6}} \approx
\qty{1.02e3}{m/s}$, e $x = \qty{1.02e3}{m}$ em um segundo. A queda vale
$h \approx \dfrac{x^2}{2d}
= \dfrac{(\num{1.02e3})^2}{2 \times \num{3.84e8}}
\approx \qty{1.4e-3}{m}$.

\emph{4.} Os dois números concordam: a Lua cai em direção à Terra, a cada
segundo, exatamente o que prevê a gravidade em inverso do quadrado,
calibrada numa maçã em queda. Newton concluiu que a força que segura a
Lua \emph{é} a força que derruba a maçã --- a gravidade é universal.
\end{solution}

\begin{solution}{pb:g10:universal-gravitation:1}
\textbf{1.} $F = \num{6.67e-11} \times
\dfrac{\num{7.3e6} \times 55}{100^2} \approx \qty{2.7e-6}{N}$.

\textbf{2.} O grão de areia pesa
$\num{1.0e-6} \times 9.81 \approx \qty{9.8e-6}{N}$: a torre Eiffel
inteira puxa o turista cerca de $3.7$ vezes \emph{menos} do que o \omterm{def:g10:universal-gravitation:weight}{peso} de
um grão de areia.

\textbf{3.} (a) Distância $\times 10$: força dividida por $100$.
(b) As duas massas $\times 1000$: força multiplicada por $10^6$.
(c) Massas $\times 1000$ e distância $\times 1000$: $10^6 / 10^6$ --- a
força não muda.

\textbf{4.} $F = \dfrac{\num{6.67e-11} \times \num{5.97e24} \times
1.00}{(\num{6.37e6})^2} \approx \qty{9.81}{N}$: o \omterm{def:g10:universal-gravitation:weight}{peso} de um \omterm{def:g10:orders-of-magnitude:unit}{quilograma}
--- é exatamente a intensidade do campo $g = \qty{9.81}{N/kg}$.

\textbf{5.} A gravidade é fraca por \omterm{def:g10:orders-of-magnitude:unit}{quilograma}, mas só sabe
\emph{atrair}, de modo que os puxões de todos os $\num{5.97e24}$
\omterm{def:g10:orders-of-magnitude:unit}{quilogramas} da Terra se somam na mesma direção --- e uma massa
astronômica ganha de uma massa de vizinho por vinte e duas potências de
dez.

\textbf{6.} $F = \num{6.67e-11} \times \dfrac{158 \times 0.73}{0.23^2}
= \num{6.67e-11} \times \dfrac{115.3}{0.0529}
\approx \qty{1.5e-7}{N}$.

\textbf{7.} A esfera pequena pesa
$0.73 \times 9.81 \approx \qty{7.2}{N}$ --- cerca de $\num{5e7}$ vezes a
atração a ser detectada. Balança de prato nenhuma resolve uma parte em
cinquenta milhões; um fio de torção fino, que se torce visivelmente sob
torques da ordem do nanonewton, resolve.

\textbf{8.} $G = \dfrac{F d^2}{m_1 m_2}
= \dfrac{\num{1.47e-7} \times 0.0529}{115.3}
\approx \qty{6.74e-11}{N.m^2/kg^2}$ --- a cerca de $1\%$ do valor
moderno \num{6.67e-11}, obtido num galpão de madeira em 1798.

\textbf{9.} De $g = G M / R^2$:
\[
M = \frac{g R^2}{G}
= \frac{9.81 \times (\num{6.37e6})^2}{\num{6.67e-11}}
\approx \qty{5.97e24}{kg}.
\]
Seis milhões de bilhões de bilhões de \omterm{def:g10:orders-of-magnitude:unit}{quilogramas}: a Terra, pesada.

\textbf{10.} $V = \frac43 \pi R^3 = \frac43 \pi
(\num{6.37e6})^3 \approx \qty{1.08e21}{m^3}$, logo
$\rho = \num{5.97e24}/\num{1.08e21} \approx \qty{5.5e3}{kg/m^3}$ --- o
dobro da densidade da rocha de superfície. O interior tem de esconder
algo muito mais denso do que granito: a Terra tem um \omterm{def:g10:refraction:fiber}{núcleo} pesado, de
ferro.

\textbf{11.} $g_{\text{Lua}} = \dfrac{\num{6.67e-11} \times
\num{7.35e22}}{(\num{1.74e6})^2} = \dfrac{\num{4.90e12}}
{\num{3.03e12}} \approx \qty{1.62}{N/kg}$.

\textbf{12.} Altura $\propto 1/g$:
$h = 0.50 \times \dfrac{9.81}{1.62} \approx \qty{3.0}{m}$ --- um salto
parado por cima de uma cesta de basquete, em câmera lenta.

\textbf{13.} $g_{\text{Mercúrio}} = \dfrac{\num{6.67e-11} \times
\num{3.30e23}}{(\num{2.44e6})^2} = \dfrac{\num{2.20e13}}
{\num{5.95e12}} \approx \qty{3.70}{N/kg}$ --- quase exatamente os
\qty{3.73}{N/kg} de Marte. Mercúrio tem menos massa, mas o seu raio menor
(um $R^2$ menor embaixo) compensa: é um mundo muito mais denso.

\textbf{14.} Metade da intensidade do campo exige
$(R + h)^2 = 2 R^2$, logo $R + h = R\sqrt{2}$ e
$h = (\sqrt{2} - 1) R \approx 0.414 \times \num{6.37e6}
\approx \qty{2.6e6}{m}$ --- cerca de \qty{2600}{km} de altitude.

\textbf{15.} $\dfrac{G M_{\text{Terra}}}{d^2} = 1.62$ dá
$d = \sqrt{\dfrac{\num{6.67e-11} \times \num{5.97e24}}{1.62}}
= \sqrt{\num{2.46e14}} \approx \qty{1.57e7}{m} \approx 2.5$ raios
terrestres. A apenas um raio e meio acima do solo, a Terra já não puxa
mais forte do que a superfície da Lua.

\textbf{16.} $v = \dfrac{2\pi d}{T}
= \dfrac{2\pi \times \num{1.496e11}}{\num{3.156e7}}
\approx \qty{2.98e4}{m/s}$ --- trinta quilômetros a cada segundo.

\textbf{17.} $\dfrac{v^2}{d} = \dfrac{(\num{2.98e4})^2}{\num{1.496e11}}
\approx \qty{5.9e-3}{N/kg}$: o Sol puxa cada \omterm{def:g10:orders-of-magnitude:unit}{quilograma} da Terra com
cerca de seis milinewtons.

\textbf{18.} $\dfrac{G M_{\text{Sol}}}{d^2} = \dfrac{v^2}{d}$ dá
\[
M_{\text{Sol}} = \frac{v^2 d}{G}
= \frac{\num{8.87e8} \times \num{1.496e11}}{\num{6.67e-11}}
\approx \qty{1.99e30}{kg}.
\]

\textbf{19.} $\dfrac{\num{1.99e30}}{\num{5.97e24}} \approx
\num{3.3e5}$: o Sol é um terço de milhão de Terras.

\textbf{20.} Como o mesmo $G$ governa todo par de massas, medi-lo uma só
vez entre duas esferas de chumbo num galpão nos permitiu colocar na
balança primeiro o planeta sob os nossos pés e depois a estrela que
orbitamos --- é isso que a palavra \emph{universal} vale.
\end{solution}
