\end{solution}

\begin{solution}{exo:g10:signals-and-waves:5}

\end{solution}

\begin{solution}{exo:g10:signals-and-waves:6}
O \omterm{def:g10:signals-and-waves:sound}{som} desce \emph{e volta}, cobrindo $2d$:
$d = \frac{1500 \times 0.60}{2} = \qty{450}{m}$. Sobre a fossa:
$t = 2d/v = 2 \times 1200/1500 = \qty{1.6}{s}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:7}

\begin{solution}{exo:g10:signals-and-waves:8}
$T = \qty{20}{mm}/\qty{25}{mm/s} = \qty{0.80}{s}$, logo
$f = \qty{1.25}{Hz}$: $75$ batimentos por minuto. Para $100$ batimentos
por minuto, $T = \qty{0.60}{s}$, ou seja,
$25 \times 0.60 = \qty{15}{mm}$: a taquicardia aparece abaixo de

\end{solution}

\begin{solution}{exo:g10:signals-and-waves:9}
$T = 1/200 = \qty{5.0}{ms}$; dois ciclos duram \qty{10}{ms}, espalhados
por $10$ divisões: \omterm{def:g10:signals-and-waves:oscilloscope}{base de tempo} de \qty{1}{ms} por divisão. A crista
tem de caber em $4$ divisões: \omterm{def:g10:signals-and-waves:oscilloscope}{ganho} de pelo menos
$3.0/4 = \qty{0.75}{V}$ por divisão --- tome \qty{1}{V} por divisão
(crista a $3.0$ divisões de altura).
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:11}

$3.0\times10^{6}/(3.00\times10^{8}) = \qty{0.010}{s}$. O ouvinte de

\begin{solution}{exo:g10:signals-and-waves:12}
Enquanto emite por \qty{3.0}{ms}, o morcego fica surdo a \omterm{rem:g10:signals-and-waves:honest}{ecos}, ou seja, a
tudo que esteja a menos de
$d = \frac{340 \times 3.0\times10^{-3}}{2} = \qty{0.51}{m}$. Mariposa a
\qty{1.7}{m}: $t = 2 \times 1.7/340 = \qty{10}{ms}$. À medida que se
aproxima, o \omterm{rem:g10:signals-and-waves:honest}{eco} volta cada vez mais cedo; o pulso tem de terminar antes
que ele chegue, e por isso precisa encurtar.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:13}
$d = \frac{340 \times 1.5}{2} = \qty{255}{m}$. A \qty{155}{m}:
$t = 2 \times 155/340 = \qty{0.91}{s}$. \omterm{rem:g10:signals-and-waves:honest}{Eco} e palma se fundem quando
$t < \qty{0.1}{s}$, ou seja, dentro de
$d < \frac{340 \times 0.1}{2} = \qty{17}{m}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:14}

$f = 1/0.015 \approx \qty{67}{Hz}$.
\emph{2.} $\qty{15}{ms}/\qty{2}{ms} = 7.5$ divisões.
\emph{3.} A tela mostra \qty{50}{ms} e depois \qty{20}{ms}: $3$ ciclos

\end{solution}

\begin{solution}{exo:g10:signals-and-waves:15}

\begin{solution}{pb:g10:signals-and-waves:1}
\textbf{1.} $d = 340 \times 9.0 = \qty{3060}{m} \approx \qty{3.1}{km}$.

\textbf{2.} $3060/(3.00\times10^{8}) \approx \qty{10}{\micro s}$, um
milhão de vezes mais curto do que \qty{9.0}{s}: o clarão marca o instante
da descarga.

\textbf{3.} Em \qty{3}{s} o \omterm{def:g10:signals-and-waves:sound}{som} cobre
$340 \times 3 = \qty{1020}{m} \approx \qty{1}{km}$: segundos divididos
por três dão quilômetros.

\textbf{4.} $d = 340 \times 6.0 = \qty{2040}{m}$. A tempestade encurtou
\qty{1020}{m} em \qty{300}{s}:
$v = \qty{3.4}{m/s} \approx \qty{12}{km/h}$, rumo ao barco.

\textbf{5.} A ponta mais próxima é ouvida depois de
$2000/340 = \qty{5.9}{s}$ e a mais distante depois de
$5000/340 = \qty{14.7}{s}$: o ronco dura cerca de \qty{8.8}{s}.

\textbf{6.} $d = 1500 \times 0.90/2 = \qty{675}{m}$.

\textbf{7.} $d = 1500 \times 0.20/2 = \qty{150}{m}$.

\textbf{8.} Dois obstáculos: um cardume a
$1500 \times 0.16/2 = \qty{120}{m}$ e o fundo a \qty{150}{m}. O cardume
nada \qty{30}{m} acima do fundo.

\textbf{9.} O \omterm{rem:g10:signals-and-waves:honest}{eco} tem de voltar antes do pulso seguinte:
$d_{\max} = 1500 \times 0.50/2 = \qty{375}{m}$. De água mais funda o \omterm{rem:g10:signals-and-waves:honest}{eco}
chega \emph{depois} do pulso seguinte e é atribuído a ele: a tela mostra
um fundo falso, raso demais.

\textbf{10.} No ar, $340 \times 0.60/2 = \qty{102}{m}$ em vez de
\qty{450}{m}. Um atraso só vira distância depois que se conhece o
portador e a velocidade dele no meio atravessado.

\textbf{11.} $f = 1/0.86 = \qty{1.16}{Hz}$, ou seja,
$1.16 \times 60 \approx 70$ batimentos por minuto.

\textbf{12.} $25 \times 0.86 = \qty{21.5}{mm}$.

\textbf{13.} $T = 15/25 = \qty{0.60}{s}$: $100$ batimentos por minuto.

\textbf{14.} $T = 60/50 = \qty{1.2}{s}$; espaçamento
$25 \times 1.2 = \qty{30}{mm}$.

\textbf{15.} Não: o \omterm{def:g10:signals-and-waves:period}{período} varia de batimento para batimento com o
esforço, a respiração e o estresse. O médico lê o \omterm{def:g10:signals-and-waves:period}{período} (a \omterm{def:g10:signals-and-waves:frequency}{frequência}),
a sua regularidade e a forma de cada ciclo --- o diagnóstico está no
\omterm{def:g10:signals-and-waves:signal}{sinal}.

\textbf{16.} Rádio: $5.0\times10^{4}/(3.00\times10^{8}) =
\qty{0.17}{ms}$ --- instantâneo para os sentidos humanos. \omterm{def:g10:signals-and-waves:sound}{Som}:
$50\,000/340 \approx \qty{147}{s}$, dois minutos e meio.

\textbf{17.} $t = 2.02\times10^{7}/(3.00\times10^{8}) = \qty{67}{ms}$.

\textbf{18.} $c \times \qty{1.0}{\micro s} = \qty{300}{m}$. Para uma
localização com \qty{3.0}{m}: $3.0/(3.00\times10^{8}) = \qty{10}{ns}$ ---
e é por isso que os \omterm{def:g10:universal-gravitation:satellite}{satélites} de GPS carregam relógios atômicos.

\textbf{19.} Os \omterm{def:g10:universal-gravitation:satellite}{satélites} estão no vácuo, e o \omterm{def:g10:signals-and-waves:sound}{som} precisa de um meio:
\omterm{def:g10:signals-and-waves:signal}{sinal} nenhum sairia do \omterm{def:g10:universal-gravitation:satellite}{satélite}. (Mesmo concedendo ar em todo o caminho,
\qty{20200}{km} a \qty{340}{m/s} dão mais de \qty{16}{h} --- uma posição
com horas de atraso.)

\textbf{20.} \qty{340}{m/s} (\omterm{def:g10:signals-and-waves:sound}{som} no ar), \qty{1500}{m/s} (\omterm{def:g10:signals-and-waves:sound}{som} na água do
mar) e $c = \qty{3.00e8}{m/s}$ (rádio e luz); a lei é $d = v\,t$,
dividida pela metade quando se trata de \omterm{rem:g10:signals-and-waves:honest}{eco}. O hábito: conheça o seu
portador, conheça a velocidade dele no meio, cronometre o atraso --- um
atraso é uma distância disfarçada.
\end{solution}
