\section*{Capítulo \ref{ch:g10:refraction} --- Refração da luz}

\begin{solution}{exo:g10:refraction:1}
Os ângulos se medem a partir da \omterm{def:g10:refraction:angles}{normal}: o \omterm{def:g10:refraction:angles}{ângulo de incidência} é
$90^\circ - 25^\circ = 65^\circ$, de modo que o ângulo de reflexão também
é $65^\circ$. Os feixes incidente e refletido formam entre si um ângulo
de $2 \times 65^\circ = 130^\circ$.
\end{solution}

\begin{solution}{exo:g10:refraction:2}
\emph{1.} $v = c/n$: na água,
$v = \qty{3.00e8}{m/s}/1.33 = \qty{2.26e8}{m/s}$; no diamante,
$v = \qty{3.00e8}{m/s}/2.42 = \qty{1.24e8}{m/s}$.

\emph{2.} $n = \dfrac{c}{v} = \dfrac{\qty{3.00e8}{m/s}}
{\qty{1.97e8}{m/s}} = 1.52$: vidro.
\end{solution}

\begin{solution}{exo:g10:refraction:3}
$\sin i_2 = \dfrac{\sin 40^\circ}{1.50} = \dfrac{0.643}{1.50} = 0.429$,
logo $i_2 = 25.4^\circ$. Pela reversibilidade dos percursos luminosos, um
raio dentro do vidro que atinja a superfície a $25.4^\circ$ sai para o ar
exatamente a $40^\circ$.
\end{solution}

\begin{solution}{exo:g10:refraction:4}
$\sin i_2 = \dfrac{\sin 55^\circ}{1.33} = \dfrac{0.819}{1.33} = 0.616$,
logo $i_2 = 38.0^\circ$. O raio se aproxima \emph{da} \omterm{def:g10:refraction:angles}{normal}, porque a
água é o meio mais lento ($n_2 > n_1$).
\end{solution}

\begin{solution}{exo:g10:refraction:5}
$n = \dfrac{\sin 45^\circ}{\sin 32^\circ} = \dfrac{0.707}{0.530}
= 1.33$: o líquido é água.
\end{solution}

\begin{solution}{exo:g10:refraction:6}
\emph{1.} As cinco razões são
\[
\frac{0.342}{0.233} = 1.47,\quad
\frac{0.500}{0.334} = 1.50,\quad
\frac{0.643}{0.431} = 1.49,\quad
\frac{0.766}{0.515} = 1.49,\quad
\frac{0.866}{0.581} = 1.49 :
\]
constantes dentro do erro de medida --- a \omterm{thm:g10:refraction:snell}{lei de Snell}.

\emph{2.} $n \approx 1.49$: acrílico.

\emph{3.} A \omterm{thm:g10:refraction:snell}{lei de Snell} prevê $\sin i_2 = \sin i_1 / n$: seno contra
seno, os dados têm de se alinhar numa reta pela origem, o que uma régua
confere de relance. Traçando ângulo contra ângulo, os dados caem sobre
uma curva; em ângulos pequenos a curva fica enganosamente perto de uma
reta, e foi exatamente isso que enganou Ptolomeu.
\end{solution}

\begin{solution}{exo:g10:refraction:7}
$\sin i_c = n_2/n_1$ com $n_2 = 1.00$:
água $\sin i_c = 1/1.33 = 0.752$, $i_c = 48.8^\circ$;
vidro $\sin i_c = 1/1.50 = 0.667$, $i_c = 41.8^\circ$;
diamante $\sin i_c = 1/2.42 = 0.413$, $i_c = 24.4^\circ$.

Do ar para a água, $\sin i_2 = \sin i_1/1.33 \leq 1/1.33 < 1$ para toda
incidência: sempre existe ângulo refratado, de modo que nada de especial
acontece --- a \omterm{def:g10:refraction:critical}{reflexão total} precisa de $n_1 > n_2$.
\end{solution}

\begin{solution}{exo:g10:refraction:8}
\emph{1.} $d' = \dfrac{d}{n} = \dfrac{\qty{2.0}{m}}{1.33}
= \qty{1.5}{m}$.

\emph{2.} O peixe aparece a $\qty{40}{cm}/1.33 = \qty{30}{cm}$: a imagem
está \emph{acima} do peixe, e por isso a garça tem de atacar
\emph{abaixo} da imagem que vê. As garças, sem o peso da \omterm{thm:g10:refraction:snell}{lei de Snell},
aprendem isso por tentativa e erro.
\end{solution}

\begin{solution}{exo:g10:refraction:9}
\emph{1.} $\sin r = \dfrac{\sin 50^\circ}{1.50} = \dfrac{0.766}{1.50}
= 0.511$, logo $r = 30.7^\circ$.

\emph{2.} Do vidro para o ar a $30.7^\circ$:
$\sin i_2 = 1.50 \times 0.511 = 0.766$, logo $i_2 = 50^\circ$ --- o
ângulo de saída é igual ao de entrada.

\emph{3.} As duas refrações se desfazem mutuamente: o raio sai
\emph{paralelo} à sua direção original, apenas deslocado lateralmente. É
por isso que uma vidraça não distorce a cena, apenas a translada
(imperceptivelmente).
\end{solution}

\begin{solution}{exo:g10:refraction:10}
$\sin 55^\circ = 0.819$. Vermelho:
$\sin i_2 = 0.819/1.61 = 0.509$, $i_2 = 30.6^\circ$. Azul:
$\sin i_2 = 0.819/1.66 = 0.493$, $i_2 = 29.6^\circ$. As duas cores ficam
separadas por $1.0^\circ$; o azul, com índice maior, desvia mais e acaba
mais perto da \omterm{def:g10:refraction:angles}{normal}.
\end{solution}

\begin{solution}{exo:g10:refraction:11}
\emph{1.} $\sin i_c = \dfrac{1.46}{1.48} = 0.986$, logo
$i_c = 80.6^\circ$.

\emph{2.} Um raio a $i_c$ da \omterm{def:g10:refraction:angles}{normal} faz $90^\circ - i_c$ com o eixo, de
modo que o seu percurso ao longo de um comprimento axial $L$ vale
$\dfrac{L}{\sin i_c}$:
\[
\qty{1000}{m} \times \frac{1.48}{1.46} = \qty{1013.7}{m},
\]
cerca de \qty{14}{m} a mais do que o raio axial.
\end{solution}

\begin{solution}{exo:g10:refraction:12}
\emph{1.} A luz que vem rasante do horizonte chega com $i_1$ próximo de
$90^\circ$; ela refrata para $\sin i_2 = \sin 90^\circ / 1.33 = 0.752$,
ou seja, $i_2 = 48.8^\circ$ --- o \omterm{def:g10:refraction:critical}{ângulo limite}, como a reversibilidade
exige. Os raios de todo o céu chegam, portanto, ao olho dentro de um cone
de semiabertura $48.8^\circ$.

\emph{2.} A borda do disco é onde esse cone encontra a superfície:
\[
r = d \tan i_c = \qty{2.0}{m} \times \tan 48.8^\circ
= \qty{2.0}{m} \times 1.14 = \qty{2.3}{m}.
\]

\emph{3.} Fora do disco, a luz vinda de baixo atinge a superfície além do
\omterm{def:g10:refraction:critical}{ângulo limite} e é totalmente refletida: o mergulhador vê um espelho
prateado que mostra o fundo da piscina.
\end{solution}

\begin{solution}{exo:g10:refraction:13}
\emph{1.} Vermelho: $\sin i_2 = 1.51 \times \sin 30^\circ = 0.755$, logo
$i_2 = 49.0^\circ$. Azul: $\sin i_2 = 1.53 \times 0.500 = 0.765$, logo
$i_2 = 49.9^\circ$. O feixe branco se abre em leque por
$49.9^\circ - 49.0^\circ = 0.9^\circ$ --- um \omterm{def:g10:light-spectra:white}{espectro} de prisma a partir
de uma única \omterm{def:g10:refraction:refraction}{refração}.

\emph{2.} A $45^\circ$: $\sin i_2 = 1.51 \times 0.707 = 1.07 > 1$ (e
$1.53 \times 0.707 = 1.08$ para o azul). Não existe ângulo refratado para
nenhuma das cores: o \omterm{def:g10:refraction:critical}{ângulo limite} vale
$\arcsin(1/1.51) = 41.5^\circ < 45^\circ$, de modo que a face longa
reflete totalmente o feixe inteiro.
\end{solution}

\begin{solution}{exo:g10:refraction:14}
\emph{1.} \omterm{def:g10:refraction:critical}{Ângulo limite} vidro--ar: $\sin i_c = 1/1.50$, logo
$i_c = 41.8^\circ < 45^\circ$: o feixe é totalmente refletido ---
$100\%$ da luz, sem revestimento algum para se deteriorar. Um espelho
metálico perde alguns por cento a cada reflexão, e um periscópio precisa
de duas.

\emph{2.} Vidro--água: $\sin i_c = \dfrac{1.33}{1.50} = 0.887$, logo
$i_c = 62.5^\circ$. Agora $45^\circ < i_c$: a incidência está abaixo da
limite, a maior parte da luz refrata para fora, na água, e o espelho ---
logo, a imagem --- falha.

\emph{3.} $\sin i_2 = \dfrac{1.50}{1.33} \times \sin 45^\circ
= 1.128 \times 0.707 = 0.798$, logo $i_2 = 52.9^\circ$ dentro da água.
\end{solution}

\begin{solution}{exo:g10:refraction:15}
\emph{1.} $v = \dfrac{\qty{3.00e8}{m/s}}{1.47} = \qty{2.04e8}{m/s}$, logo
\[
t = \frac{\qty{6.2e6}{m}}{\qty{2.04e8}{m/s}} = \qty{30.4}{ms} \approx
\qty{30}{ms}.
\]

\emph{2.} Rádio: $t = \qty{6.2e6}{m}/\qty{3.00e8}{m/s} =
\qty{20.7}{ms}$. O vidro custa $30.4 - 20.7 = \qty{9.7}{ms}$ em cada
sentido, cerca de \qty{19}{ms} por ida e volta.

\emph{3.} Baixar o índice --- fibras de \omterm{def:g10:refraction:fiber}{núcleo} oco guiam a luz pelo ar
($n \approx 1.00$), e os enlaces de micro-ondas pela atmosfera fazem o
mesmo --- ou encurtar o caminho: um cabo lançado mais perto da rota de
círculo máximo entre as duas cidades. As duas coisas são efetivamente
vendidas, a preços por milissegundo.
\end{solution}

\begin{solution}{pb:g10:refraction:1}
\textbf{1.} $v_{\text{água}} = \qty{3.00e8}{m/s}/1.33 =
\qty{2.26e8}{m/s}$; $v_{\text{diamante}} = \qty{3.00e8}{m/s}/2.42 =
\qty{1.24e8}{m/s}$ --- o diamante retarda a luz pelo fator $n = 2.42$.

\textbf{2.} $\sin i_2 = \sin 40^\circ / 1.33 = 0.643/1.33 = 0.483$, logo
$i_2 = 28.9^\circ$.

\textbf{3.} $\sin i_2 = 1.33 \times \sin 28.9^\circ = 1.33 \times
0.483 = 0.643$, logo $i_2 = 40^\circ$: a luz refaz o seu próprio caminho
--- reversibilidade dos raios luminosos.

\textbf{4.} $\sin i_c = 1/1.33 = 0.752$, logo $i_c = 48.8^\circ$. A
$60^\circ > i_c$, a \omterm{thm:g10:refraction:snell}{lei de Snell} exigiria $\sin i_2 = 1.33 \sin
60^\circ = 1.15 > 1$: não há \omterm{def:g10:refraction:refraction}{raio refratado}; a superfície reflete
totalmente a luz de volta para baixo.

\textbf{5.} $v = \qty{3.00e8}{m/s}/1.0003 = \qty{2.999e8}{m/s}$: a luz no
ar é mais lenta do que no vácuo em $0.03\%$, muito abaixo da precisão das
nossas medidas de ângulo, de modo que $n_{\text{ar}} = 1.00$ é inofensivo.

\textbf{6.} $d' = \qty{3.0}{m}/1.33 = \qty{2.3}{m}$: a piscina esconde
\qty{70}{cm} de si mesma.

\textbf{7.} Dois raios que saem da moeda refratam na superfície,
afastando-se \emph{da} \omterm{def:g10:refraction:angles}{normal}, e portanto divergem mais acentuadamente
acima da água do que abaixo. O olho prolonga os dois raios emergentes
para trás em linha reta; eles se cruzam \emph{acima} da moeda, e é nesse
cruzamento que o olho coloca a imagem.

\textbf{8.} $r = d \tan i_c = \qty{3.0}{m} \times \tan 48.8^\circ =
\qty{3.0}{m} \times 1.14 = \qty{3.4}{m}$: um disco de cerca de
\qty{6.9}{m} de diâmetro abriga o céu inteiro.

\textbf{9.} A $30^\circ$: $\sin i_2 = 1.33 \times 0.500 = 0.665$, o raio
sai a $41.7^\circ$. A $45^\circ$: $\sin i_2 = 1.33 \times 0.707 = 0.940$,
ele sai a $70.1^\circ$, quase rasante. A $50^\circ$:
$1.33 \times 0.766 = 1.02 > 1$ --- \omterm{def:g10:refraction:critical}{reflexão total}, o raio mergulha de
volta na piscina.

\textbf{10.} A imagem está acima do peixe real
(\cref{prop:g10:refraction:depth}), então mire \emph{abaixo} dela. E sim:
o peixe também vê o pescador deslocado, espremido em direção à vertical
dentro da janela de Snell --- cada um vê o outro onde o outro não está.

\textbf{11.} Diamante: $\sin i_c = 1/2.42 = 0.413$, logo
$i_c = 24.4^\circ$, contra $41.1^\circ$ do vidro: o cone de escape do
diamante não tem nem metade da largura.

\textbf{12.} No diamante, $30^\circ > 24.4^\circ$: \omterm{def:g10:refraction:critical}{reflexão total}, o raio
fica na pedra. No vidro, $30^\circ < 41.1^\circ$: o raio vaza pelo fundo.
As facetas de uma lapidação brilhante são anguladas de modo que a luz que
entra quase sempre as atinge além de $24.4^\circ$ e ricocheteia até sair
para cima, em direção ao olho --- já uma imitação de vidro, com o seu
cone de escape largo, simplesmente deixa a luz cair para fora.

\textbf{13.} $\sin 45^\circ = 0.707$. Vermelho: $0.707/2.41 = 0.293$,
logo $i_2 = 17.1^\circ$. Azul: $0.707/2.45 = 0.289$, logo
$i_2 = 16.8^\circ$. Separação na entrada: cerca de $0.3^\circ$.

\textbf{14.} $\sin 17.0^\circ = 0.292$. Vermelho: $\sin i_2 = 2.41 \times
0.292 = 0.705$, logo $i_2 = 44.8^\circ$. Azul: $\sin i_2 = 2.45 \times
0.292 = 0.716$, logo $i_2 = 45.8^\circ$. O leque tem agora $1.0^\circ$ de
abertura: a \omterm{def:g10:refraction:refraction}{refração} de saída, trabalhando perto do \omterm{def:g10:refraction:critical}{ângulo limite},
esticou em três vezes a separação de $0.3^\circ$ da entrada. Esse jato
esticado de cores é o \emph{fogo} do diamante.

\textbf{15.} Diamante--água: $\sin i_c = 1.33/2.42 = 0.550$, logo
$i_c = 33.3^\circ$. O raio de $30^\circ$, preso quando a faceta dava para
o ar, agora escapa ($30^\circ < 33.3^\circ$) para o filme de água: toda
gota, impressão digital ou película de gordura eleva o índice externo,
alarga o cone de escape e drena o brilho --- daí o pano do joalheiro.

\textbf{16.} $\sin i_c = \dfrac{1.46}{1.48} = 0.986$, logo
$i_c = 80.6^\circ$.

\textbf{17.} A \omterm{def:g10:refraction:critical}{reflexão total} acontece \emph{na superfície}, de modo que
a superfície tem de permanecer perfeita: num fio nu, cada grão de poeira,
arranhão, gota de água ou dedo encostado eleva o índice externo naquele
ponto e sangra luz para fora. A \omterm{def:g10:refraction:fiber}{casca} enterra o espelho dentro de vidro
limpo, onde nada o alcança --- e, de brinde, protege mecanicamente o
\omterm{def:g10:refraction:fiber}{núcleo} fino como um fio de cabelo.

\textbf{18.} $v = \qty{3.00e8}{m/s}/1.48 = \qty{2.03e8}{m/s}$, logo
\[
t = \frac{\qty{6.0e6}{m}}{\qty{2.03e8}{m/s}} = \qty{29.6}{ms} \approx
\qty{30}{ms}.
\]

\textbf{19.} Fator de percurso $1/\sin i_c = 1.48/1.46 = 1.0137$: no pior
caso, $\qty{6000}{km} \times 0.0137 \approx \qty{82}{km}$ de vidro extra,
custando $\qty{8.2e4}{m} / \qty{2.03e8}{m/s} \approx \qty{0.4}{ms}$ ---
o ziguezague sai praticamente de graça.

\textbf{20.} Ping: $2 \times (29.6 + 0.4) \approx \qty{60}{ms}$. Vazão:
$\dfrac{10^{13}}{2.4 \times 10^8} \approx 40\,000$ cópias deste livro por
segundo. Em uma frase: ao prender a luz dentro do vidro por \omterm{def:g10:refraction:critical}{reflexão
total}, a \omterm{thm:g10:refraction:snell}{lei de Snell} carrega uma biblioteca através do oceano a cada
segundo, sessenta milissegundos na ida e na volta.
\end{solution}
