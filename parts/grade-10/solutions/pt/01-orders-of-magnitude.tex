\section*{Capítulo \ref{ch:g10:orders-of-magnitude} --- Ordens de grandeza: medir o universo}

\begin{solution}{exo:g10:orders-of-magnitude:1}
\emph{1.} $\qty{750}{nm} = 750 \times \qty{e-9}{m} = \qty{7.50e-7}{m}$.
\emph{2.} $\qty{2.5}{km} = \qty{2.5e3}{m}$.
\emph{3.} $\qty{3.2}{\micro m} = \qty{3.2e-6}{m}$.
\emph{4.} $\qty{0.15}{Tm} = 0.15 \times \qty{e12}{m} = \qty{1.5e11}{m}$
--- uma \omterm{def:g10:orders-of-magnitude:lightyear}{unidade astronômica}.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:2}
\emph{1.} $\qty{149600000}{km} = \qty{1.496e11}{m}$: $4$ \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos
significativos}.
\emph{2.} $\qty{1.06e-10}{m}$: $3$ \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos significativos} (os zeros à
esquerda nunca contam).
\emph{3.} $\num{8.1e9}$: $2$ \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos significativos}.
\emph{4.} $\qty{5.20e-3}{s}$: $3$ \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos significativos} (o zero à
direita é uma promessa).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:3}
\omterm{def:g10:orders-of-magnitude:scinot}{Mantissas} e \omterm{def:g10:orders-of-magnitude:scinot}{expoentes} separadamente
(\cref{prop:g10:orders-of-magnitude:powers}):
\[
3 \times 2 \times 10^{8-5} = \num{6e3}, \qquad
\tfrac{8}{4} \times 10^{4-(-3)} = \num{2e7},
\]
\[
5^{2} \times 10^{12} = \num{2.5e13}, \qquad
\tfrac{4 \times 3}{6} \times 10^{-3-9+7} = \num{2e-5} .
\]
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:4}
Humano: $1.7 < 5$, ordem $\qty{e0}{m}$. Everest:
$\qty{8848}{m} = \qty{8.848e3}{m}$ e $8.848 \geq 5$: ordem
$\qty{e4}{m}$. Bactéria: ordem $\qty{e-6}{m}$. Terra: $5.97 \geq 5$,
ordem $\qty{e25}{kg}$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:5}
\emph{1.} Meia graduação: $L = (127.0 \pm 0.5)\,\unit{mm}$.
\emph{2.} $\Delta L / L = 0.5/127 \approx 0.4\%$.
\emph{3.} Borracha: $0.5/12 \approx 4\%$. A medida do lápis é cerca de
dez vezes mais precisa --- mesma régua, mesma \omterm{def:g10:orders-of-magnitude:uncertainty}{incerteza absoluta}, objeto
dez vezes mais comprido.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:6}
\emph{1.} Próxima: $4.24 \times \qty{9.47e15}{m} = \qty{4.01e16}{m}$.
Sirius: $8.6 \times \qty{9.47e15}{m} = \qty{8.1e16}{m}$ (dois \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos
significativos}, como o dado de entrada).
\emph{2.} $t = d/c = \qty{2.4e13}{m} / \qty{3.00e8}{m/s}
= \qty{8.0e4}{s}$, ou seja, $\num{8.0e4}/3600 \approx \qty{22}{h}$:
quase um dia inteiro para cada mensagem da sonda.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:7}
$t = d/c$ em cada caso. Lua:
$\qty{3.84e8}{m}/\qty{3.00e8}{m/s} = \qty{1.28}{s}$. Sol:
$\qty{1.496e11}{m}/\qty{3.00e8}{m/s} = \qty{499}{s} \approx
\qty{8.3}{min}$. Netuno:
$\qty{4.50e12}{m}/\qty{3.00e8}{m/s} = \qty{1.50e4}{s} \approx
\qty{4.2}{h}$. Quando você olha para Netuno, vê-o como ele era há cerca
de quatro horas --- todo telescópio é uma máquina do tempo modesta.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:8}
\emph{1.} Uma folha: $\qty{52.0}{mm}/500 = \qty{0.104}{mm}
= \qty{1.04e-4}{m}$. A incerteza também se divide:
$\qty{0.5}{mm}/500 = \qty{0.001}{mm}$, de modo que
$t = (104 \pm 1)\,\unit{\micro m}$.
\emph{2.} $1/104 \approx 1\%$.
\emph{3.} Medir uma folha diretamente daria
$(0.1 \pm 0.5)\,\unit{mm}$: uma \omterm{def:g10:orders-of-magnitude:uncertainty}{incerteza relativa} de cerca de $500\%$
--- a régua nem sequer consegue provar que a folha existe. Empilhar
$500$ folhas multiplica o comprimento medido por $500$ enquanto a
\omterm{def:g10:orders-of-magnitude:uncertainty}{incerteza absoluta} da régua fica fixa: a \omterm{def:g10:orders-of-magnitude:uncertainty}{incerteza relativa} encolhe
nesse mesmo fator $500$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:9}
Divida cada comprimento por $\qty{2e-10}{m}$. Bactéria:
$\qty{2.0e-6}{m}/\qty{2e-10}{m} = \num{1.0e4}$ átomos. Cabelo:
$\qty{7.0e-5}{m}/\qty{2e-10}{m} = \num{3.5e5}$. Humano:
$1.7/\num{2e-10} = \num{8.5e9}$, \omterm{def:g10:orders-of-magnitude:oom}{ordem de grandeza} $10^{10}$: dez
bilhões de átomos da cabeça aos pés.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:10}
\emph{1.} $s = \qty{0.22}{m}/\qty{1.39e9}{m} = \num{1.58e-10}$.
\emph{2.} Terra:
$\qty{1.27e7}{m} \times \num{1.58e-10} = \qty{2.0e-3}{m} =
\qty{2.0}{mm}$ --- um grão de pimenta --- orbitando a
$\qty{1.496e11}{m} \times \num{1.58e-10} = \qty{23.7}{m} \approx
\qty{24}{m}$ da bola.
\emph{3.} Lua: $\qty{3.84e8}{m} \times \num{1.58e-10} = \qty{6.1}{cm}$
do grão de pimenta. Netuno:
$\qty{4.50e12}{m} \times \num{1.58e-10} \approx \qty{710}{m}$: a maquete
não cabe no pátio da escola.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:11}
\emph{1.} $21.0 \times 29.7 = \qty{623.7}{cm^2}$, arredondado para os
três \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos significativos} dos dados: $\qty{624}{cm^2}$.
\emph{2.} $v = \qty{100.0}{m}/\qty{12.3}{s} = \qty{8.13}{m/s}$ (três
algarismos, fixados pelo tempo).
\emph{3.} Cinco \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos significativos} alegariam conhecer a área com
precisão de $\qty{0.01}{cm^2}$, precisão que dados de nível milimétrico
jamais entregaram --- uma casa decimal forjada continua sendo uma
falsificação.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:12}
\emph{1.} Raio $r = \qty{0.55}{m}$, área
$A = \pi r^{2} = \pi \times 0.55^{2} \approx \qty{0.95}{m^2}$. Volume
$V = \qty{1.0}{mm^3} = \qty{1.0e-9}{m^3}$, e um filme é (área)
$\times$ (espessura), logo
\[
e = \frac{V}{A} = \frac{\qty{1.0e-9}{m^3}}{\qty{0.95}{m^2}}
= \qty{1.1e-9}{m} .
\]
\emph{2.} Uma molécula é da ordem de $\qty{e-9}{m}$ --- um degrau acima
do $\qty{e-10}{m}$ do átomo, coerente com a molécula ser um pequeno
agrupamento de átomos. A colher de óleo de Franklin acalmou meio hectare
de lago e, sem que ele soubesse, mediu o nanômetro.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:13}
Hipóteses: cerca de $70$ batimentos por minuto em repouso, uma vida de
$80$ anos. Por dia: $70 \times 60 \times 24 \approx \num{1e5}$
batimentos. Por ano: $\num{1e5} \times 365 \approx \num{4e7}$. Por vida:
$\num{4e7} \times 80 \approx \num{3e9}$ --- \omterm{def:g10:orders-of-magnitude:oom}{ordem de grandeza} $10^{9}$,
alguns bilhões de batimentos. Quatro \omterm{def:g10:orders-of-magnitude:sigfig}{algarismos significativos} fingiriam
conhecer uma frequência cardíaca, minuto a minuto, durante oitenta anos:
os dados de entrada são palpites de \omterm{def:g10:orders-of-magnitude:oom}{ordem de grandeza}, e nenhuma saída
supera as suas entradas (\cref{met:g10:orders-of-magnitude:sigfig}).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:14}
Depois de $n$ dobras a espessura vale $2^{n} \times \qty{1.0e-4}{m}$.
\emph{1.} É preciso $2^{n} \geq 8848/\num{1.0e-4} = \num{8.8e7}$. Como
$2^{26} \approx \num{6.7e7}$ não basta e $2^{27} \approx \num{1.3e8}$
basta: $27$ dobras (pilha $\approx \qty{13}{km}$).
\emph{2.} É preciso $2^{n} \geq \num{3.84e12}$: $2^{41} \approx
\num{2.2e12}$ fica aquém, $2^{42} \approx \num{4.4e12}$ alcança ---
$42$ dobras, pilha de $\qty{4.4e8}{m}$, folgadamente além da Lua.
\emph{3.} Só a experiência: cada dobra reduz a área pela metade e,
depois de sete dobras, a pilha é mais grossa do que larga. A aritmética
da duplicação fica intacta --- o crescimento exponencial simplesmente
ultrapassa o que o papel (e a intuição) conseguem acompanhar.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:15}
\emph{1.} Lua: $\qty{0.03}{m}/\qty{3.84e8}{m} = \num{7.8e-11}$. Mesa:
$\qty{0.001}{m}/\qty{1.000}{m} = \num{1.0e-3}$. A medida da Lua é mais
precisa por um fator $\num{1.0e-3}/\num{7.8e-11} \approx \num{1.3e7}$
--- mais de dez milhões.
\emph{2.} Igualar $\num{7.8e-11}$ numa mesa de um \omterm{def:g10:orders-of-magnitude:unit}{metro} exige
$\Delta x = \qty{7.8e-11}{m}$, menor do que um átomo
($\approx \qty{e-10}{m}$): a borda da mesa nem sequer está
\emph{definida} com essa nitidez. Em termos relativos, a distância
Terra--Lua medida a laser é uma das medidas mais finas que a nossa
espécie possui.
\end{solution}

\begin{solution}{pb:g10:orders-of-magnitude:1}
\textbf{1.} Grão: $\qty{0.5}{mm} = \qty{5e-4}{m}$, na escala do mili;
átomo: $\qty{2.5e-10}{m} = \qty{0.25}{nm}$, na escala do nano.

\textbf{2.} $\qty{5e-4}{m}/\qty{2.5e-10}{m} = \num{2.0e6}$: dois milhões
de átomos de ponta a ponta.

\textbf{3.} Um cubo de $2 \times 10^{6}$ átomos por aresta contém
$(2\times10^{6})^{3} = 8 \times 10^{18}$ átomos: \omterm{def:g10:orders-of-magnitude:oom}{ordem de grandeza}
$10^{19}$.

\textbf{4.} $V = (\qty{5e-4}{m})^{3} = \qty{1.25e-10}{m^3}$, logo
$m = \rho V = \qty{2500}{kg/m^3} \times \qty{1.25e-10}{m^3}
= \qty{3.1e-7}{kg} = \qty{0.31}{mg}$.

\textbf{5.} Massa por átomo:
$\qty{3.1e-7}{kg}/(8\times10^{18}) = \qty{3.9e-26}{kg}$ --- bem dentro
da faixa dos ``alguns $10^{-26}\,\unit{kg}$''. O cubo ingênuo de átomos
reproduz massas atômicas reais: o modelo passa na conferência
(\cref{met:g10:orders-of-magnitude:sanity}) com louvor.

\textbf{6.} $s = \qty{5e-4}{m}/\qty{1.39e9}{m} = \num{3.6e-13}$.

\textbf{7.} Terra da maquete: diâmetro
$\qty{1.27e7}{m} \times \num{3.6e-13} = \qty{4.6e-6}{m}$ --- o tamanho
de uma bactéria --- à distância
$\qty{1.496e11}{m} \times \num{3.6e-13} = \qty{5.4}{cm}$ do grão: a
Terra é um micróbio a cinco centímetros de um grão de areia.

\textbf{8.} $\qty{4.5e12}{m} \times \num{3.6e-13} = \qty{1.6}{m}$: o
sistema planetário inteiro orbita dentro de uma mesa de jantar.

\textbf{9.} $\qty{4.0e16}{m} \times \num{3.6e-13} = \qty{1.4e4}{m}
= \qty{14}{km}$.

\textbf{10.} A estrela mais próxima do nosso Sistema Solar de mesa é
outro grão de areia a quatorze quilômetros de distância, com
essencialmente nada no meio: a galáxia é esmagadoramente espaço vazio, e
é por isso que colisões entre estrelas são quase inauditas.

\textbf{11.} $t = \qty{5e-4}{m}/\qty{3.00e8}{m/s} = \qty{1.7e-12}{s}$
--- menos de dois picossegundos.

\textbf{12.} Lua: $\qty{3.84e8}{m}/\qty{3.00e8}{m/s} = \qty{1.28}{s}$.
Sol: $\qty{1.496e11}{m}/\qty{3.00e8}{m/s} = \qty{499}{s} \approx
\qty{8.3}{min}$: a luz do Sol tem sempre oito minutos de idade.

\textbf{13.} Um ano vale $365.25 \times 86400 = \qty{3.156e7}{s}$, logo
$\qty{1}{ly} = \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}$; então
$\qty{4.0e16}{m}/\qty{9.47e15}{m} = 4.2$: a distância dos dados
corresponde de fato a cerca de $\qty{4.2}{ly}$.

\textbf{14.} $\qty{2.4e22}{m}/\qty{9.47e15}{m} = \num{2.5e6}$: dois
milhões e meio de anos-luz. Essa luz deixou Andrômeda há cerca de $2.5$
milhões de anos, quando os primeiros membros do gênero \emph{Homo}
aprendiam a lascar pedras --- e nenhum \emph{Homo sapiens} existia ainda
em lugar algum.

\textbf{15.} $\qty{3.00e8}{m/s} \times \qty{1e-9}{s} = \qty{0.30}{m}$.
Em um terço de nanossegundo um sinal cobre no máximo uns
$\qty{10}{cm}$, de modo que quaisquer duas partes da máquina que
precisem conversar dentro de um mesmo ciclo de relógio têm de ficar a
menos de dez centímetros uma da outra: a velocidade limita o tamanho.

\textbf{16.} Átomo médio na água:
$\qty{3.0e-26}{kg}/3 = \qty{1.0e-26}{kg}$. Número de átomos:
$\qty{70}{kg}/\qty{1.0e-26}{kg} = \num{7e27}$.

\textbf{17.} Contra o grão: $\num{7e27}/\num{8e18} \approx 10^{9}$ ---
você contém átomos equivalentes a um bilhão de grãos. Contra as
estrelas: $\num{7e27}/\num{2e23} = \num{3.5e4}$ --- os seus átomos são
mais numerosos do que todas as estrelas do universo observável na
proporção de trinta e cinco mil para um.

\textbf{18.} $\text{(você)}/\text{(átomo)} =
\qty{1.7}{m}/\qty{2.5e-10}{m} \approx \num{7e9} \approx 10^{10}$,
enquanto $\text{(universo)}/\text{(você)} =
\qty{8.8e26}{m}/\qty{1.7}{m} \approx \num{5e26}$. Na escada, você está
cerca de dez degraus acima do átomo e quase vinte e sete abaixo do
universo: multiplicativamente, o ser humano vive bem perto dos seus
átomos.

\textbf{19.} A condição dá
$x^{2} = 10^{-15} \times 10^{26.9} = 10^{11.9}$, logo
$x = 10^{5.95} \approx \qty{9e5}{m} \approx \qty{900}{km}$ --- quase
exatamente o diâmetro de Ceres, o maior asteroide
($\approx \qty{940}{km}$). Bem no meio da escada, entre um núcleo e o
cosmos, fica um modesto planeta anão.

\textbf{20.} $\qty{8.8e26}{m} \times \num{3.6e-13} = \qty{3.2e14}{m}$,
ou seja, $\qty{3.2e14}{m}/\qty{1.496e11}{m} \approx \num{2.1e3}$
\omterm{def:g10:orders-of-magnitude:lightyear}{unidades astronômicas}. O desfecho: encolha o cosmos até que o Sol seja
um grão de areia, e o universo observável ainda cobre duas mil vezes a
distância Terra--Sol real --- maquete nenhuma traz o universo à escala
humana; só as potências de dez o carregam.
\end{solution}
