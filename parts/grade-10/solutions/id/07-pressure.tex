\section*{Bab \ref{ch:g10:pressure} --- Tekanan: Dari Olahraga sampai Penyelaman}

\begin{solution}{exo:g10:pressure:1}
Sisi besar: $S = \qty{0.15}{m^2}$, $P = 600/0.15 = \qty{4.0}{kPa}$.
Sisi kecil: $S = \qty{0.060}{m^2}$, $P = \qty{10}{kPa}$. Di atas pasir,
sisi yang besar: \omterm{def:g10:pressure:pressure}{tekanannya} lebih rendah, benamnya lebih sedikit.
\end{solution}

\begin{solution}{exo:g10:pressure:2}
$\qty{0.25}{MPa} = \qty{250}{kPa} = \qty{2.5e5}{Pa}$.
$3.0/\num{2.0e-4} = \qty{15}{kPa}$ mengalahkan
$60/0.050 = \qty{1.2}{kPa}$: \omterm{def:g10:inertia:force}{gaya} yang kecil itulah yang menekan lebih
kuat.
\end{solution}

\begin{solution}{exo:g10:pressure:3}
$\rho g h = 1000 \times 9.81 \times 3.0 \approx \qty{29}{kPa}$;
\omterm{met:g10:pressure:dive}{tekanan mutlaknya} $P = 101 + 29 = \qty{130}{kPa}$.
\end{solution}

\begin{solution}{exo:g10:pressure:4}
$V_2 = 100 \times 4.0 / 250 = \qty{1.6}{L}$;
pada \qty{50}{kPa}, $V_2 = 400/50 = \qty{8.0}{L}$.
\end{solution}

\begin{solution}{exo:g10:pressure:5}
$F = 65 \times 9.81 \approx \qty{638}{N}$. Pisaunya:
$638/\num{1.2e-3} \approx \qty{5.3e5}{Pa} = \qty{530}{kPa}$. Solnya:
$638/0.036 \approx \qty{18}{kPa}$. Nisbahnya $\approx 30$: pisaunya
menggigit esnya, solnya tidak.
\end{solution}

\begin{solution}{exo:g10:pressure:6}
Ibu jari: $20/\num{1.2e-4} \approx \qty{1.7e5}{Pa} = \qty{170}{kPa}$.
Dinding: $20/\num{1.0e-8} = \qty{2.0e9}{Pa}$. \omterm{def:g10:inertia:force}{Gaya} \qty{20}{N} yang sama
bekerja pada luas yang $12\,000$ kali lebih kecil di ujungnya, sehingga
hanya di sanalah \omterm{def:g10:pressure:pressure}{tekanannya} melampaui apa yang sanggup ditahan bahannya.
\end{solution}

\begin{solution}{exo:g10:pressure:7}
$\Delta P = \rho g h = 1025 \times 9.81 \times 30 \approx
\qty{3.0e5}{Pa}$, sehingga
$F = \num{3.0e5} \times 0.50 \approx \qty{1.5e5}{N}$ --- setara \omterm{def:g10:universal-gravitation:weight}{berat}
$F/g \approx \qty{1.5e4}{kg}$, yaitu lima belas ton. Tidak ada awak
kapal yang sanggup membukanya ke luar melawan laut.
\end{solution}

\begin{solution}{exo:g10:pressure:8}
$\rho = \dfrac{P}{g h} = \dfrac{\num{66.2e3}}{9.81 \times 5.00}
\approx \qty{1350}{kg/m^3}$. Bukan air tawar ($\qty{1000}{kg/m^3}$):
sebuah zat cair yang jauh lebih rapat, misalnya air garam pekat.
\end{solution}

\begin{solution}{exo:g10:pressure:9}
$h = P_0/(\rho g)$: untuk air
$\num{1.01e5}/9810 \approx \qty{10.3}{m}$; untuk raksa
$\num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$. Tabung
sepanjang \qty{76}{cm} muat di atas meja; barometer air setinggi
\qty{10}{m} memerlukan ruang tangga --- massa jenis raksalah yang menang.
\end{solution}

\begin{solution}{exo:g10:pressure:10}
$V = 6000/240 = \qty{25}{mL}$; $P = 6000/75 = \qty{80}{kPa}$.
$P$ terhadap $V$ berupa hiperbola; $P$ terhadap $1/V$ berupa garis lurus
melalui titik asal, dengan kemiringan \qty{6000}{kPa.mL}.
\end{solution}

\begin{solution}{exo:g10:pressure:11}
Pada \qty{20}{m}: $P = 101 + 2 \times 101 = \qty{303}{kPa}$, yaitu tiga
atmosfer. Di permukaan,
$V = 0.50 \times 303/101 = \qty{1.5}{cm^3}$: gelembungnya menjadi tiga
kali lipat.
\end{solution}

\begin{solution}{exo:g10:pressure:12}
\emph{1.} $F = \num{1.01e5} \times \num{1.2e-3} \approx \qty{121}{N}$,
yang menahan $m = 121/9.81 \approx \qty{12}{kg}$.

\emph{2.} $\Delta P = 101 - 20 = \qty{81}{kPa}$:
$F = \num{8.1e4} \times \num{1.2e-3} \approx \qty{97}{N}$, yaitu sekitar
\qty{9.9}{kg}.
\end{solution}

\begin{solution}{exo:g10:pressure:13}
Menurut Boyle: $12 \times (200 - 50) = \qty{1800}{L}$ udara bertekanan
permukaan yang dapat dipakai. Pada \qty{30}{m} tiap tarikan napas
diambil pada \qty{4}{bar}, sehingga \qty{15}{L/min} pada \omterm{def:g10:pressure:pressure}{tekanan}
sekitarnya memakan $15 \times 4 = \qty{60}{L/min}$ udara permukaan:
$1800/60 =
\qty{30}{min}$. Pada \qty{10}{m} ($\qty{2}{bar}$):
\qty{30}{L/min}, sehingga \qty{60}{min}. Kedalaman dibayar dengan udara.
\end{solution}

\begin{solution}{exo:g10:pressure:14}
$P = P_0 \dfrac{V_0}{V} = 101 \times \dfrac{6.0}{1.5} =
\qty{404}{kPa}$. Dengan $\rho g \approx \qty{10.1}{kPa/m}$,
$h = (404 - 101)/10.1 \approx \qty{30}{m}$.
\end{solution}

\begin{solution}{exo:g10:pressure:15}
\emph{1.} \omterm{def:g10:universal-gravitation:weight}{Beratnya} $F = 1200 \times 9.81 \approx \qty{1.18e4}{N}$;
\omterm{def:g10:pressure:pressure}{tekanannya} $P = F/S = \num{1.18e4}/0.040 \approx \qty{2.9e5}{Pa}$; pada
torak kecil: $f = P s = \num{2.9e5} \times \num{2.0e-4} \approx \qty{59}{N}$ --- \omterm{def:g10:inertia:force}{gayanya} dibagi $S/s = 200$.

\emph{2.} Volume zat cairnya kekal:
$400 \times 2.0 = \qty{800}{cm^3}$ harus datang dari tabung kecilnya,
yang bergerak $800/2.0 = \qty{400}{cm} = \qty{4.0}{m}$. Usahanya:
$59 \times 4.0 \approx \qty{235}{J}$ pada satu sisi,
$\num{1.18e4} \times 0.020 \approx \qty{235}{J}$ pada sisi yang lain ---
dongkrak itu menukar jarak dengan \omterm{def:g10:inertia:force}{gaya}, tidak pernah dengan usaha.
\end{solution}

\begin{solution}{pb:g10:pressure:1}
\textbf{1.} $\rho g = 1025 \times 9.81 \approx \qty{10.1}{kPa}$ tiap
\omterm{def:g10:orders-of-magnitude:unit}{meter}, sehingga \qty{10}{m} air laut menambahkan
$\approx \qty{101}{kPa}$: satu atmosfer tiap sepuluh \omterm{def:g10:orders-of-magnitude:unit}{meter}, hampir
persis.

\textbf{2.} $P = 101 + 10.1\,h$: \qty{202}{kPa} pada \qty{10}{m},
\qty{302}{kPa} pada \qty{20}{m}, \qty{403}{kPa} pada \qty{30}{m},
\qty{503}{kPa} pada \qty{40}{m}.

\textbf{3.} $2.0$, $3.0$, $4.0$ dan $5.0$ kali $P_0$.

\textbf{4.} $\Delta P = 403 - 101 = \qty{302}{kPa}$ pada
$\qty{0.015}{m^2}$: $F = \num{3.02e5} \times 0.015 \approx
\qty{4.5e3}{N}$ --- setara \omterm{def:g10:universal-gravitation:weight}{berat} hampir setengah ton pada wajahnya.

\textbf{5.} $\Delta P = 10.1 \times 3.0 \approx \qty{30}{kPa}$;
$F = \num{3.0e4} \times \num{6.0e-5} \approx \qty{1.8}{N}$ --- seperti
sebuah jari yang menekan gendang telinga, sudah pada kedalaman kolam
renang. Menyamakan \omterm{def:g10:pressure:pressure}{tekanan} berarti mendorong udara ke telinga tengah
sampai \omterm{def:g10:pressure:pressure}{tekanan} di dalamnya menyamai \omterm{def:g10:pressure:pressure}{tekanan} airnya.

\textbf{6.} $V = 6.0 \, P_0/P$: $6.0 \times 101/202 = \qty{3.0}{L}$;
$606/302 = \qty{2.0}{L}$; $606/403 = \qty{1.5}{L}$.

\textbf{7.} Menjadi separuh ketika $P = 2P_0$, yaitu pada \qty{10}{m}
--- pada penggal pertama penyelamannya.

\textbf{8.} Tidak: $V = 606/P$ berupa hiperbola dalam \omterm{met:g10:pressure:dive}{tekanan mutlaknya}
(dan $P$, bukan $h$, itulah yang dilihat \omterm{prop:g10:pressure:boyle}{hukum Boyle}); tiap tambahan
\qty{10}{m} mengurangi volume lebih sedikit daripada tambahan
sebelumnya.

\textbf{9.} $V = \qty{1.5}{L}$ menuntut
$P = 101 \times 6.0/1.5 = \qty{404}{kPa}$, yaitu
$h = 303/10.1 \approx \qty{30}{m}$ (\cref{exo:g10:pressure:14}).

\textbf{10.} Darah: plasmanya berpindah ke pembuluh di rongga dada,
tidak dapat dimampatkan, lalu menempati volume yang tidak lagi diisi
udara.

\textbf{11.} Pada \omterm{def:g10:pressure:pressure}{tekanan} sekitarnya, yaitu \qty{403}{kPa}. Menurut
Boyle, udara yang sama pada \qty{101}{kPa} akan mengisi $4$ kali
volumenya: udara itu $4.0$ kali lebih rapat daripada udara permukaan.

\textbf{12.} $V = 6.0 \times 403/202 \approx \qty{12}{L}$ --- dua kali
lipat, hanya setelah naik \qty{20}{m}.

\textbf{13.} $V = 6.0 \times 403/101 \approx \qty{24}{L}$: empat kali
daya tampung paru-parunya.

\textbf{14.} \emph{Jangan pernah menahan napas ketika naik.} Dengan
saluran napas yang terbuka, udara yang memuai mengalir keluar melalui
regulatornya, dan volume paru-parunya tidak pernah melampaui
\qty{6.0}{L}.

\textbf{15.} Volume \qty{1.5}{L} milik Lena pada \qty{30}{m}
\emph{adalah} \qty{6.0}{L} permukaannya yang termampatkan: dalam
perjalanan naik volume itu memuai kembali tepat menjadi \qty{6.0}{L},
tidak pernah lebih. Sedangkan \qty{6.0}{L} milik Marco \emph{diambil}
pada \qty{403}{kPa}: di dalamnya terkandung \qty{24}{L} udara
permukaan.

\textbf{16.} $V = 1.0 \times 503/P$: \qty{1.25}{cm^3} pada \qty{30}{m},
\qty{1.67}{cm^3} pada \qty{20}{m}, \qty{2.49}{cm^3} pada \qty{10}{m},
\qty{4.98}{cm^3} tepat di bawah permukaan.

\textbf{17.} Faktor tiap tahapnya: $503/403 = 1.25$, lalu $1.33$,
$1.49$, dan $202/101 = 2.0$ untuk sepuluh \omterm{def:g10:orders-of-magnitude:unit}{meter} terakhir ---
pertumbuhannya makin cepat ketika permukaan makin dekat.

\textbf{18.} $40/10 = \qty{4.0}{min}$. Menit terakhirnya: dari
\qty{10}{m} sampai permukaan, \omterm{met:g10:pressure:dive}{tekanan mutlaknya} menjadi separuh, yaitu
penurunan relatif terbesar sepanjang kenaikan itu.

\textbf{19.} Kenaikan yang cepat membuat nitrogen yang terlarut
berbuih menjadi gelembung di dalam darah dan persendian (penyakit
dekompresi); naik perlahan-lahan dan berhenti sejenak di dekat
\qty{5}{m} menjaga gasnya tetap terlarut cukup lama untuk keluar
dengan tenang melalui paru-paru.

\textbf{20.} Penyelam bebas: udaranya hanya kembali ke volume semula
--- Boyle melindunginya. Penyelam skuba: udara yang diambil di
kedalaman berlipat ganda dalam perjalanan naik --- bernapaslah, jangan
pernah menahannya. Dan sepuluh \omterm{def:g10:orders-of-magnitude:unit}{meter} terakhir menggandakan setiap
volume yang terperangkap ($\times 2.0$, berbanding $\times 1.25$ di
bawah sana pada \qty{40}{m}): makin dekat permukaannya, makin
perlahan-lahan kamu harus mendekatinya.
\end{solution}
