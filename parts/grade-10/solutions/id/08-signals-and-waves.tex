\section*{Bab \ref{ch:g10:signals-and-waves} --- Sinyal dan Gelombang}

\begin{solution}{exo:g10:signals-and-waves:1}
$f = 1/T$: (a) $1/\qty{0.020}{s} = \qty{50}{Hz}$;
(b) $1/\qty{4.0e-3}{s} = \qty{250}{Hz}$;
(c) $1/\qty{5.0e-5}{s} = \qty{20}{kHz}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:2}
$T = 1/f$: (a) \qty{20}{ms}; (b) $1/440 = \qty{2.3}{ms}$;
(c) $1/\qty{2.4e9}{Hz} = \qty{0.42}{ns}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:3}
\qty{12}{Hz}: \omterm{def:g10:signals-and-waves:ultrasound}{infrasonik}. \qty{440}{Hz} dan \qty{18}{kHz}: terdengar
(yang terakhir hanya oleh telinga muda). \qty{40}{kHz} dan \qty{5}{MHz}:
\omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik}.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:4}
$T = 5.0 \times \qty{2}{ms} = \qty{10}{ms}$, sehingga
$f = \qty{100}{Hz}$; \omterm{def:g10:signals-and-waves:amplitude}{amplitudonya}
$4.0 \times \qty{0.5}{V} = \qty{2.0}{V}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:5}
$d = 340 \times 6.0 = \qty{2040}{m} \approx \qty{2.0}{km}$. Cahaya
menempuhnya dalam $2040/(3.00\times10^{8}) \approx \qty{7}{\micro s}$,
sejuta kali lebih singkat daripada \qty{6.0}{s}: kilatnya menandai saat
sambaran itu terjadi.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:6}
\omterm{def:g10:signals-and-waves:sound}{Bunyinya} turun \emph{lalu kembali}, menempuh $2d$:
$d = \frac{1500 \times 0.60}{2} = \qty{450}{m}$. Di atas palungnya:
$t = 2d/v = 2 \times 1200/1500 = \qty{1.6}{s}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:7}
$d_1 = \frac{3.00\times10^{8} \times 2.4\times10^{-4}}{2} =
\qty{36.0}{km}$ dan $d_2 = \qty{35.7}{km}$: pesawat itu menempuh
\qty{300}{m} dalam \qty{1.0}{s}, sehingga $v = \qty{300}{m/s} \approx
\qty{1100}{km/h}$, dan ia mendekat.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:8}
$T = \qty{20}{mm}/\qty{25}{mm/s} = \qty{0.80}{s}$, sehingga
$f = \qty{1.25}{Hz}$: $75$ detak tiap menit. Untuk $100$ detak tiap
menit, $T = \qty{0.60}{s}$, yaitu $25 \times 0.60 = \qty{15}{mm}$:
takikardia tampak di bawah \qty{15}{mm}.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:9}
$T = 1/200 = \qty{5.0}{ms}$; dua daur berlangsung \qty{10}{ms}, tersebar
pada $10$ petak: \omterm{def:g10:signals-and-waves:oscilloscope}{basis waktunya} \qty{1}{ms} tiap petak. Puncaknya harus
muat pada $4$ petak: \omterm{def:g10:signals-and-waves:oscilloscope}{penguatannya} sekurang-kurangnya
$3.0/4 = \qty{0.75}{V}$ tiap petak --- ambillah \qty{1}{V} tiap petak
(puncaknya $3.0$ petak ke atas).
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:10}
$d = v\,t/2$: dinding depannya
$\frac{1540 \times 4.0\times10^{-5}}{2} = \qty{3.1}{cm}$; dinding
belakangnya $\frac{1540 \times 6.0\times10^{-5}}{2} = \qty{4.6}{cm}$;
tebalnya sekitar \qty{1.5}{cm}.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:11}
\omterm{def:g10:signals-and-waves:sound}{Bunyi}: $30/340 = \qty{0.088}{s}$. Radio:
$3.0\times10^{6}/(3.00\times10^{8}) = \qty{0.010}{s}$. Pendengar radio
yang jauh itulah yang lebih dahulu mendengar gendangnya, sekitar
\qty{78}{ms} lebih awal.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:12}
Selama memancar \qty{3.0}{ms}, kelelawar itu tuli terhadap \omterm{rem:g10:signals-and-waves:honest}{gema}, yaitu
terhadap apa pun yang berada di dalam
$d = \frac{340 \times 3.0\times10^{-3}}{2} =
\qty{0.51}{m}$. Ngengat
pada \qty{1.7}{m}: $t = 2 \times 1.7/340 = \qty{10}{ms}$. Ketika
mendekat, \omterm{rem:g10:signals-and-waves:honest}{gemanya} kembali makin cepat; denyutnya harus berakhir sebelum
\omterm{rem:g10:signals-and-waves:honest}{gemanya} tiba, sehingga denyut itu harus memendek.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:13}
$d = \frac{340 \times 1.5}{2} = \qty{255}{m}$. Pada \qty{155}{m}:
$t = 2 \times 155/340 = \qty{0.91}{s}$. \omterm{rem:g10:signals-and-waves:honest}{Gema} dan tepukannya melebur
ketika $t < \qty{0.1}{s}$, yaitu di dalam
$d < \frac{340 \times 0.1}{2} = \qty{17}{m}$.
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:14}
\emph{1.} $T = 3.0 \times \qty{5}{ms} = \qty{15}{ms}$,
$f = 1/0.015 \approx \qty{67}{Hz}$.
\emph{2.} $\qty{15}{ms}/\qty{2}{ms} = 7.5$ petak.
\emph{3.} Layarnya menampilkan \qty{50}{ms}, lalu \qty{20}{ms}: $3$ daur
lengkap ($50/15 = 3.3$), lalu $1$ daur ($20/15 = 1.3$).
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:15}
\emph{1.} $t = 2 \times 1.2\times10^{6}/(3.00\times10^{8}) =
\qty{8.0}{ms}$.
\emph{2.} $2 \times 1.2\times10^{6}/(2.0\times10^{8}) = \qty{12}{ms}$.
\emph{3.} Seratnya tidak terbentang lurus antara kedua mesin itu, dan
tiap perute serta peladen di sepanjang jalan menambahkan waktu tunda
pemrosesan.
\end{solution}

\begin{solution}{pb:g10:signals-and-waves:1}
\textbf{1.} $d = 340 \times 9.0 = \qty{3060}{m} \approx \qty{3.1}{km}$.

\textbf{2.} $3060/(3.00\times10^{8}) \approx \qty{10}{\micro s}$,
sejuta kali lebih singkat daripada \qty{9.0}{s}: kilatnya menandai saat
sambaran itu terjadi.

\textbf{3.} Dalam \qty{3}{s} \omterm{def:g10:signals-and-waves:sound}{bunyi} menempuh
$340 \times 3 = \qty{1020}{m}
\approx \qty{1}{km}$: sekon yang dibagi
tiga memberikan kilometer.

\textbf{4.} $d = 340 \times 6.0 = \qty{2040}{m}$. Badai itu mendekat
\qty{1020}{m} dalam \qty{300}{s}: $v = \qty{3.4}{m/s} \approx
\qty{12}{km/h}$, menuju kapalnya.

\textbf{5.} Ujung yang dekat terdengar setelah
$2000/340 = \qty{5.9}{s}$, ujung yang jauh setelah
$5000/340 = \qty{14.7}{s}$: gemuruhnya berlangsung sekitar
\qty{8.8}{s}.

\textbf{6.} $d = 1500 \times 0.90/2 = \qty{675}{m}$.

\textbf{7.} $d = 1500 \times 0.20/2 = \qty{150}{m}$.

\textbf{8.} Dua penghalang: gerombolan ikan pada
$1500 \times 0.16/2 =
\qty{120}{m}$ dan dasar laut pada \qty{150}{m}.
Gerombolan itu berenang \qty{30}{m} di atas dasarnya.

\textbf{9.} \omterm{rem:g10:signals-and-waves:honest}{Gemanya} harus kembali sebelum denyut berikutnya:
$d_{\max} = 1500 \times 0.50/2 = \qty{375}{m}$. Dari air yang lebih
dalam, \omterm{rem:g10:signals-and-waves:honest}{gemanya} tiba \emph{sesudah} denyut berikutnya lalu dianggap
sebagai \omterm{rem:g10:signals-and-waves:honest}{gema} denyut itu: layarnya menampilkan dasar palsu yang jauh
terlalu dangkal.

\textbf{10.} Di udara, $340 \times 0.60/2 = \qty{102}{m}$ dan bukan
\qty{450}{m}. Waktu tunda baru menjadi jarak setelah kamu mengetahui
pembawanya dan lajunya di dalam medium yang dilintasinya.

\textbf{11.} $f = 1/0.86 = \qty{1.16}{Hz}$, yaitu $1.16 \times 60
\approx 70$ detak tiap menit.

\textbf{12.} $25 \times 0.86 = \qty{21.5}{mm}$.

\textbf{13.} $T = 15/25 = \qty{0.60}{s}$: $100$ detak tiap menit.

\textbf{14.} $T = 60/50 = \qty{1.2}{s}$; jaraknya
$25 \times 1.2 = \qty{30}{mm}$.

\textbf{15.} Tidak: \omterm{def:g10:signals-and-waves:period}{periodenya} bergeser dari detak ke detak mengikuti
usaha, pernapasan dan \omterm{def:g10:pressure:pressure}{tekanan} batin. Dokternya membaca \omterm{def:g10:signals-and-waves:period}{periodenya}
(lajunya), keteraturannya, dan bentuk tiap daurnya --- diagnosisnya ada
di dalam \omterm{def:g10:signals-and-waves:signal}{sinyal} itu.

\textbf{16.} Radio: $5.0\times10^{4}/(3.00\times10^{8}) =
\qty{0.17}{ms}$ --- seketika bagi indra manusia. \omterm{def:g10:signals-and-waves:sound}{Bunyi}:
$50\,000/340 \approx \qty{147}{s}$, dua setengah menit.

\textbf{17.} $t = 2.02\times10^{7}/(3.00\times10^{8}) = \qty{67}{ms}$.

\textbf{18.} $c \times \qty{1.0}{\micro s} = \qty{300}{m}$. Untuk
penentuan kedudukan \qty{3.0}{m}:
$3.0/(3.00\times10^{8}) = \qty{10}{ns}$ --- dan itulah sebabnya \omterm{def:g10:universal-gravitation:satellite}{satelit}
GPS membawa jam atom.

\textbf{19.} \omterm{def:g10:universal-gravitation:satellite}{Satelit} berada di ruang hampa, sedangkan \omterm{def:g10:signals-and-waves:sound}{bunyi} memerlukan
medium: tidak akan ada \omterm{def:g10:signals-and-waves:signal}{sinyal} yang meninggalkan \omterm{def:g10:universal-gravitation:satellite}{satelit} itu sama
sekali. (Sekalipun seluruh jalannya diberi udara, \qty{20200}{km} pada
\qty{340}{m/s} berarti lebih dari \qty{16}{h} --- penentuan kedudukan
yang sudah usang berjam-jam.)

\textbf{20.} \qty{340}{m/s} (\omterm{def:g10:signals-and-waves:sound}{bunyi} di udara), \qty{1500}{m/s} (\omterm{def:g10:signals-and-waves:sound}{bunyi} di
air laut) dan $c = \qty{3.00e8}{m/s}$ (\omterm{def:g10:signals-and-waves:wave}{gelombang} radio dan cahaya);
hukumnya $d = v\,t$, yang diseparuhkan untuk \omterm{rem:g10:signals-and-waves:honest}{gema}. Kebiasaannya:
ketahuilah pembawanya, ketahuilah lajunya di dalam mediumnya, lalu
ukurlah waktu tundanya --- sebuah waktu tunda adalah jarak yang
menyamar.
\end{solution}
