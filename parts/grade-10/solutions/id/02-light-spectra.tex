\section*{Bab \ref{ch:g10:light-spectra} --- Spektrum Cahaya dan Pesan yang Dibawa Cahaya}

\begin{solution}{exo:g10:light-spectra:1}
$\qty{254}{nm} < \qty{400}{nm}$: \omterm{def:g10:light-spectra:wavelength}{ultraviolet}. $\qty{480}{nm}$: tampak,
biru. $\qty{589}{nm}$: tampak, kuning-jingga. $\qty{656}{nm}$: tampak,
merah. $\qty{1064}{nm} > \qty{800}{nm}$: \omterm{def:g10:light-spectra:wavelength}{inframerah} --- dan itulah yang
membuat laser pemotong berbahaya dua kali: kuat sekaligus tak terlihat.
\end{solution}

\begin{solution}{exo:g10:light-spectra:2}
\emph{1.} Pita warna yang kontinu, dari merah sampai ungu: \omterm{def:g10:light-spectra:white}{spektrum}
cahaya matahari.

\emph{2.} Ujung ungunya --- prisma membelokkan \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang
pendek paling kuat.

\emph{3.} Satu bercak hijau saja: dibelokkan, tetapi tidak terurai
menjadi pita. Cahaya laser bersifat \omterm{def:g10:light-spectra:white}{monokromatik} --- hanya satu \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang}, sehingga tidak ada yang perlu disusun.
\end{solution}

\begin{solution}{exo:g10:light-spectra:3}
(a) Kontinu (kawat pijar yang rapat dan panas). (b) Garis emisi (gas
tereksitasi bertekanan rendah). (c) Kontinu (logam cair yang rapat).
(d) Garis emisi (pada dasarnya satu garis pada $\qty{589}{nm}$).
(e) Serapan: \omterm{def:g10:light-spectra:continuous}{spektrum kontinu} permukaan yang panas, tergores garis
gelap atmosfer Matahari yang lebih dingin.
\end{solution}

\begin{solution}{exo:g10:light-spectra:4}
\emph{1.} Tetes hujannya: masing-masing menerima cahaya matahari yang
putih, menguraikannya, lalu mengirimkannya kembali tersusun menurut
warna.

\emph{2.} Matahari menyediakan \omterm{def:g10:light-spectra:white}{cahaya putihnya}; hujan menyediakan
jutaan prisma mungil; susunan ruangnya hanya cocok bagi cahaya yang
dikembalikan tetes air ke arah pengamat, dan itulah sebabnya Matahari
harus berada di punggung pengamat sedangkan hujan berada di depannya.

\emph{3.} Tetes air tidak menambahkan cahaya miliknya sendiri --- ia
hanya menyusun apa yang diterimanya. Jika cahaya matahari yang tersusun
menampilkan setiap warna, maka setiap warna itu memang sudah terkandung
di dalam cahaya matahari: cahaya matahari adalah \omterm{def:g10:light-spectra:white}{cahaya putih}.
\end{solution}

\begin{solution}{exo:g10:light-spectra:5}
\emph{1.} Merah suram $<$ jingga terang $<$ hampir putih.

\emph{2.} Keduanya: pijarnya menjadi \emph{lebih terang} (akibat 1), dan
warnanya bergeser dari merah ke sisi biru (akibat 2, dengan puncak
$\lambda_{\max}$ yang mengecil), sehingga makin banyak bagian pita
tampak yang tersinari kuat --- pita yang penuh terbaca sebagai putih.

\emph{3.} Benda yang berpijar putih memancar kuat pada seluruh pita
tampak; benda yang berpijar merah hanya sanggup pada ujung \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang} yang panjang. Menurut hukum Wien, yang pertamalah yang lebih
panas.
\end{solution}

\begin{solution}{exo:g10:light-spectra:6}
\emph{1.}
\[
T = \frac{\qty{2.90e-3}{m.K}}{\qty{5.00e-7}{m}} = \qty{5800}{K}.
\]

\emph{2.} $\theta = 5800 - 273 = \qty{5527}{\degreeCelsius} \approx
\qty{5500}{\degreeCelsius}$.

\emph{3.} $\qty{400}{nm} < \qty{500}{nm} < \qty{800}{nm}$: puncaknya
tampak, pada warna hijau.
\end{solution}

\begin{solution}{exo:g10:light-spectra:7}
\emph{1.} Lilin:
$\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{1800}{K} = \qty{1.61e-6}{m}
= \qty{1.6}{\micro m}$, di daerah \omterm{def:g10:light-spectra:wavelength}{inframerah}. Pemanas:
$\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{500}{K} = \qty{5.8e-6}{m}
= \qty{5.8}{\micro m}$, \omterm{def:g10:light-spectra:wavelength}{inframerah} jauh.

\emph{2.} Pada $\qty{500}{K}$ hampir seluruh radiasinya \omterm{def:g10:light-spectra:wavelength}{inframerah}:
kulit menyerapnya dan merasakan panas, tetapi mata tidak menerima
apa-apa --- panas, namun hitam.

\emph{3.} \omterm{def:g10:light-spectra:continuous}{Spektrum kontinu} itu lebar: sekalipun memuncak pada
$\qty{1.6}{\micro m}$, \omterm{def:g10:light-spectra:white}{spektrum} lilin masih menyisakan ekor tampak di
ujung merah-kuning --- lemah, tetapi cukup bagi mata yang telah
menyesuaikan diri dengan gelap.
\end{solution}

\begin{solution}{exo:g10:light-spectra:8}
Lampu A: garis $434$, $486$, $\qty{656}{nm}$ cocok dengan hidrogen, dan
garis ungu yang samar melengkapi katalognya pada $\qty{410}{nm}$:
hidrogen.

Lampu B: garisnya berada pada $\qty{589}{nm}$, satu-satunya garis
tampak natrium yang kuat. Helium disingkirkan bukan oleh garis itu
sendiri (helium memang memiliki garis di dekatnya, pada
$\qty{588}{nm}$) melainkan oleh garis yang \emph{tidak ada}: helium
seharusnya juga menunjukkan $447$, $502$ dan $\qty{668}{nm}$, dan tidak
satu pun muncul. Sebuah unsur dikenali oleh katalognya yang lengkap:
lampu B memuat natrium.
\end{solution}

\begin{solution}{exo:g10:light-spectra:9}
\emph{1.} Pelangi kontinu yang tergores satu garis gelap yang tipis:
\omterm{def:g10:light-spectra:absorption}{spektrum serapan} natrium.

\emph{2.} Pada $\qty{589}{nm}$: sebuah gas menyerap tepat \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang} yang sanggup dipancarkannya
(\cref{prop:g10:light-spectra:kirchhoff}), dan garis tampak natrium
yang kuat berada pada $\qty{589}{nm}$.

\emph{3.} \omterm{def:g10:light-spectra:emission}{Spektrum emisi} natrium: satu garis kuning-jingga yang terang
pada $\qty{589}{nm}$ di atas latar hitam --- \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang
sama, dengan kedua \omterm{def:g10:light-spectra:white}{spektrum} bertukar terang dan gelap.
\end{solution}

\begin{solution}{exo:g10:light-spectra:10}
\emph{1.} $410$, $434$, $486$, $\qty{656}{nm}$: katalog hidrogen yang
lengkap. $517$ dan $\qty{518}{nm}$: pasangan magnesium. Kedua unsur itu
hadir.

\emph{2.} Di atmosfer Matahari, yaitu gas yang lebih dingin dan lebih
tipis di atas permukaan yang berpijar. Permukaannya mengirimkan
\omterm{def:g10:light-spectra:continuous}{spektrum kontinu} ke atas; atmosfernya mengambil \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}
miliknya sendiri dari cahaya yang lewat, sehingga garisnya muncul
sebagai cahaya yang hilang --- gelap di depan latar yang terang.
\end{solution}

\begin{solution}{exo:g10:light-spectra:11}
\emph{1.} $T = 37 + 273 = \qty{310}{K}$, sehingga
\[
\lambda_{\max} = \frac{\qty{2.90e-3}{m.K}}{\qty{310}{K}}
= \qty{9.4e-6}{m} = \qty{9.4}{\micro m}.
\]

\emph{2.} \omterm{def:g10:light-spectra:wavelength}{Inframerah} jauh --- lebih dari sepuluh kali \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}
cahaya merah.

\emph{3.} Pada $\qty{310}{K}$ bagian tampak radiasinya benar-benar
dapat diabaikan: tidak ada pijar yang terlihat, segelap apa pun
ruangannya. Kamera termal membawa pengindra yang dilaraskan pada
radiasi yang sungguh dipancarkan, di sekitar $\qty{9}{\micro m}$, dan
melihat tubuh yang hangat bersinar di depan sekelilingnya yang lebih
dingin.
\end{solution}

\begin{solution}{exo:g10:light-spectra:12}
\emph{1.} $\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{2700}{K}
= \qty{1.07e-6}{m} \approx \qty{1.1}{\micro m}$: \omterm{def:g10:light-spectra:wavelength}{inframerah} dekat.

\emph{2.} \omterm{def:g10:light-spectra:white}{Spektrumnya} memuncak di luar daerah tampak: sebagian besar
daya listriknya kembali sebagai \omterm{def:g10:light-spectra:wavelength}{inframerah} yang tak terlihat --- yaitu
kalor --- dan hanya ekor tampak yang kecil menerangi ruangan. Sebagai
lampu, bohlam itu memboroskan hampir seluruh dayanya; sebagai pemanas,
ia nyaris sempurna.

\emph{3.} $T = \qty{2.90e-3}{m.K} / \qty{5.50e-7}{m} \approx
\qty{5300}{K}$ --- jauh di atas titik lebur wolfram, yaitu
$\qty{3700}{K}$. Tidak ada kawat pijar padat yang dapat didorong sampai
suhu cahaya siang, sehingga lampu yang menyerupai cahaya siang terpaksa
meninggalkan pemijaran sama sekali (tabung neon, LED), yakni
menghasilkan cahaya tampak tanpa memanaskan benda sampai
$\qty{5300}{K}$.
\end{solution}

\begin{solution}{exo:g10:light-spectra:13}
\emph{1.} $T = \qty{2.90e-3}{m.K} / \qty{4.20e-7}{m} \approx
\qty{6900}{K}$: lebih panas daripada Matahari yang $\qty{5800}{K}$.

\emph{2.} Hidrogen (rangkaian lengkap $410$, $434$, $486$,
$\qty{656}{nm}$ hadir) dan kalsium (kedua garisnya, $393$ dan
$\qty{397}{nm}$).

\emph{3.} Tidak. Garis natrium pada $\qty{589}{nm}$ tidak ada, padahal
natrium yang melimpah di atmosfernya pasti akan mencetakkannya pada
latar kontinu. Kalaupun natrium ada di sana, ia hanya berupa jejak yang
terlalu samar untuk terdeteksi.
\end{solution}

\begin{solution}{exo:g10:light-spectra:14}
\emph{1.} Hidrogen menjelaskan $410$, $434$, $486$ dan $\qty{656}{nm}$;
helium menjelaskan $447$, $502$, $588$ dan $\qty{668}{nm}$. Setiap
garis yang teramati sudah terjelaskan: campurannya adalah hidrogen dan
helium.

\emph{2.} Helium sudah merupakan penjelasan yang \emph{memadai} karena
ketiga garisnya yang lain ($447$, $502$, $\qty{668}{nm}$) semuanya
hadir --- garis $\qty{588}{nm}$ kemudian memang diharapkan dari helium
saja, dan tidak ada apa pun di dalam \omterm{def:g10:light-spectra:white}{spektrum} itu yang
\emph{mengharuskan} natrium. Untuk menuntaskan perkara ini yang
diperlukan adalah daya pisah, bukan perdebatan: \omterm{def:g10:light-spectra:white}{spektroskop} yang lebih
halus memisahkan garis tunggal helium pada $\qty{587.6}{nm}$ dari garis
natrium, yang sebenarnya berupa pasangan rapat pada $589.0$ dan
$\qty{589.6}{nm}$. Melihat satu garis di sana membebaskan natrium;
melihat tiga garis memberatkannya.
\end{solution}

\begin{solution}{exo:g10:light-spectra:15}
\emph{1.} Betelgeuse:
$T = \qty{2.90e-3}{m.K} / \qty{8.5e-7}{m} \approx \qty{3400}{K}$.
Rigel: $T = \qty{2.90e-3}{m.K} / \qty{2.4e-7}{m} \approx \qty{12000}{K}$.

\emph{2.} Karena $\lambda_{\max} \times T$ bernilai tetapan yang sama
bagi keduanya,
$\dfrac{T_{\text{Rigel}}}{T_{\text{Betelgeuse}}}
= \dfrac{\qty{850}{nm}}{\qty{240}{nm}} \approx 3.5$.

\emph{3.} Betelgeuse memuncak di \omterm{def:g10:light-spectra:wavelength}{inframerah}: di dalam pita tampak
\omterm{def:g10:light-spectra:white}{spektrumnya} paling kuat pada ujung merah, sehingga bintang itu tampak
merah. Rigel memuncak di \omterm{def:g10:light-spectra:wavelength}{ultraviolet}: pita tampaknya paling kuat pada
ujung biru, sehingga ia biru-putih.

\emph{4.} \omterm{def:g10:light-spectra:continuous}{Spektrum kontinu} meliputi rentang \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang
sangat lebar. Sekalipun puncaknya berada di luar pita tampak, sebuah
bintang tetap mencurahkan banyak cahaya \emph{ke dalam} pita itu ---
puncaknya menentukan rona warnanya, bukan keterlihatannya.
\end{solution}

\begin{solution}{pb:g10:light-spectra:1}
\textbf{1.} Dari sekitar $\qty{400}{nm}$ (ujung ungu) sampai sekitar
$\qty{800}{nm}$ (ujung merah); $\qty{550}{nm}$ berada di dekat
tengahnya, pada warna hijau.

\textbf{2.} $\qty{410}{nm}$: ungu; $\qty{434}{nm}$: ungu-biru;
$\qty{486}{nm}$: biru-hijau; $\qty{656}{nm}$: merah.

\textbf{3.} Pijar kuning-jingga yang \omterm{def:g10:light-spectra:white}{monokromatik}. Warna sebuah mobil
adalah campuran \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang dipantulkan catnya; di bawah
lampu yang menawarkan \emph{hanya} $\qty{589}{nm}$, setiap cat hanya
dapat memantulkan lebih banyak atau lebih sedikit \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}
tunggal itu --- semua mobil tampak dalam rona jingga dan kelabu.

\textbf{4.} Setiap unsur memancarkan katalog garis yang tetap, dan
tidak ada dua unsur yang memiliki katalog yang sama
(\cref{prop:g10:light-spectra:fingerprint}): rangkaian garis yang
lengkap karena itu mengenali unsurnya seperti sidik jari. Namun
kecocokan yang \emph{sebagian} membuktikan sedikit sekali --- unsur
yang berbeda dapat memiliki garis tunggal yang nyaris berimpit (helium
pada $\qty{588}{nm}$, natrium pada $\qty{589}{nm}$) --- sehingga
pernyataan bahwa sebuah unsur hadir menuntut seluruh garis kuat pada
katalognya.

\textbf{5.} (a) Pelangi kontinu dengan satu garis gelap pada
$\qty{589}{nm}$: \omterm{def:g10:light-spectra:absorption}{spektrum serapan}. (b) Satu garis terang pada
$\qty{589}{nm}$ di atas latar hitam: \omterm{def:g10:light-spectra:emission}{spektrum emisi} --- \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang} yang sama, dengan terang dan gelap yang bertukar
(\cref{prop:g10:light-spectra:kirchhoff}).

\textbf{6.} \omterm{def:g10:light-spectra:white}{Spektrumnya} menjadi lebih terang pada setiap \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang}, dan puncaknya $\lambda_{\max}$ bergerak ke \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}
yang lebih pendek, mengikuti
$\lambda_{\max} \times T = \qty{2.90e-3}{m.K}$
(\cref{prop:g10:light-spectra:wien}).

\textbf{7.} $\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{3400}{K}
= \qty{8.5e-7}{m} = \qty{850}{nm}$: \omterm{def:g10:light-spectra:wavelength}{inframerah} dekat, tepat di luar
tepi merah --- dan itulah sebabnya cahaya halogen berona lebih hangat
daripada cahaya siang.

\textbf{8.} $T = \qty{2.90e-3}{m.K} / \qty{5.00e-7}{m} = \qty{5800}{K}$.

\textbf{9.} Puncaknya landai, bukan lonjakan tajam: Matahari memancarkan
seluruh pita tampak dengan kekuatan yang sebanding. Pita warna yang
penuh dan bercampur terbaca sebagai putih --- Matahari tampak putih
(kekuningan melalui atmosfer kita), tidak pernah hijau.

\textbf{10.} Besi yang masih dingin: puncaknya jauh di \omterm{def:g10:light-spectra:wavelength}{inframerah},
tanpa ekor tampak --- radiasinya terasa, tidak ada yang terlihat. Lebih
panas: \omterm{def:g10:light-spectra:white}{spektrum} yang menguat mendorong ekor tampak yang pertama
melewati $\qty{800}{nm}$ --- ujung merahnya menyala sendirian: merah
suram. Lebih panas lagi: seluruh pita tampak terisi sementara semuanya
menguat: jingga, lalu menuju putih.

\textbf{11.} Latar kontinunya datang dari permukaan berpijar yang rapat
dan panas; garis gelapnya dicetak oleh atmosfer yang lebih dingin dan
lebih tipis di atasnya, yang menyerap \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} miliknya
sendiri dari cahaya yang lewat.

\textbf{12.} Hidrogen ($410$, $434$, $486$, $\qty{656}{nm}$: lengkap),
natrium ($\qty{589}{nm}$) dan kalsium ($393$ dan $\qty{397}{nm}$:
keduanya) --- ketiganya hadir di atmosfer Matahari.

\textbf{13.} Bahwa atmosfernya dikuasai hidrogen. Pengukuran modern
sependapat dan memberikan angkanya: kira-kira tiga perempat massa
Matahari adalah hidrogen, dan sebagian besar sisanya helium.

\textbf{14.} Garis itu tidak cocok dengan unsur mana pun yang dikenal;
karena katalog setiap unsur bersifat tetap dan khas, garis yang tidak
termasuk katalog mana pun tentu termasuk unsur yang belum pernah
terlihat di laboratorium. Sebuah unsur baru diumumkan dan dinamai
helium, menurut \emph{helios}, yaitu Matahari. Unsur itu dipisahkan di
Bumi pada tahun 1895 --- ramalannya terbukti, dua puluh tujuh tahun
kemudian.

\textbf{15.} Dipandang di depan latar kontinu yang cemerlang,
atmosfernya \emph{mengambil} cahaya pada \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} miliknya:
garis gelap. Pada saat gerhana, Bulan menutup latarnya; atmosfer yang
tipis itu lalu berdiri sendirian di depan langit yang gelap dan cahaya
yang dipancarkannya kembali adalah satu-satunya yang terlihat: \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang} yang sama, kini terang. Satu gas, satu katalog --- tandanya
bergantung pada apa yang berdiri di belakangnya.

\textbf{16.} $T = \qty{2.90e-3}{m.K} / \qty{2.90e-7}{m}
= \qty{10000}{K}$.

\textbf{17.} Biru-putih: di dalam pita tampak, \omterm{def:g10:light-spectra:white}{spektrumnya} paling kuat
pada ujung biru. Dan tidak ada ketakterlihatan: \omterm{def:g10:light-spectra:continuous}{spektrum kontinunya}
membanjiri seluruh pita tampak dalam perjalanannya menuju puncak
\omterm{def:g10:light-spectra:wavelength}{ultraviolet} --- puncaknya menetapkan rona warnanya, bukan
keterangannya.

\textbf{18.} Garis gelap $410$, $434$, $486$, $\qty{656}{nm}$ adalah
sidik jari hidrogen yang lengkap: hidrogen menguasai atmosfer Vega.

\textbf{19.} $T_{\text{Vega}} / T_{\text{Matahari}} = 10000 / 5800 \approx
1.7$: Vega lebih panas, karena itu lebih biru (puncaknya pada
$\qty{290}{nm}$ berbanding $\qty{500}{nm}$) dan, menurut bagian pertama
\cref{prop:g10:light-spectra:wien}, lebih terang per \omterm{def:g10:orders-of-magnitude:unit}{satuan} luas
permukaan berpijar pada setiap \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}.

\textbf{20.} Formulir jati diri Vega --- suhu permukaan: sekitar
$\qty{10000}{K}$; unsur yang menguasai atmosfernya: hidrogen; warna
tampaknya: biru-putih. Semuanya terbaca dari satu berkas cahaya, dengan
sebuah prisma, sebuah katalog garis laboratorium dan satu pembagian.
Kata ``tidak akan pernah'' milik Comte dikeluarkan pada tahun 1835 dan
berakhir pada tahun 1859: kata itu bertahan dua puluh empat tahun.
\end{solution}
