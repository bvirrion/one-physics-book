\section*{Bab \ref{ch:g10:orders-of-magnitude} --- Orde Besar: Mengukur Alam Semesta}

\begin{solution}{exo:g10:orders-of-magnitude:1}
\emph{1.} $\qty{750}{nm} = 750 \times \qty{e-9}{m} = \qty{7.50e-7}{m}$.
\emph{2.} $\qty{2.5}{km} = \qty{2.5e3}{m}$.
\emph{3.} $\qty{3.2}{\micro m} = \qty{3.2e-6}{m}$.
\emph{4.} $\qty{0.15}{Tm} = 0.15 \times \qty{e12}{m} = \qty{1.5e11}{m}$
--- satu \omterm{def:g10:orders-of-magnitude:lightyear}{satuan astronomi}.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:2}
\emph{1.} $\qty{149600000}{km} = \qty{1.496e11}{m}$: $4$ \omterm{def:g10:orders-of-magnitude:sigfig}{angka
penting}.
\emph{2.} $\qty{1.06e-10}{m}$: $3$ \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} (nol di depan tidak
pernah dihitung).
\emph{3.} $\num{8.1e9}$: $2$ \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting}.
\emph{4.} $\qty{5.20e-3}{s}$: $3$ \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} (nol di belakangnya
adalah sebuah janji).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:3}
\omterm{def:g10:orders-of-magnitude:scinot}{Mantisa} dan \omterm{def:g10:orders-of-magnitude:scinot}{eksponennya} ditangani terpisah
(\cref{prop:g10:orders-of-magnitude:powers}):
\[
3 \times 2 \times 10^{8-5} = \num{6e3}, \qquad
\tfrac{8}{4} \times 10^{4-(-3)} = \num{2e7},
\]
\[
5^{2} \times 10^{12} = \num{2.5e13}, \qquad
\tfrac{4 \times 3}{6} \times 10^{-3-9+7} = \num{2e-5} .
\]
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:4}
Manusia: $1.7 < 5$, ordenya $\qty{e0}{m}$. Everest:
$\qty{8848}{m} = \qty{8.848e3}{m}$ dan $8.848 \geq 5$: ordenya
$\qty{e4}{m}$. Bakteri: ordenya $\qty{e-6}{m}$. Bumi:
$5.97 \geq 5$, ordenya $\qty{e25}{kg}$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:5}
\emph{1.} Setengah goresan: $L = (127.0 \pm 0.5)\,\unit{mm}$.
\emph{2.} $\Delta L / L = 0.5/127 \approx 0.4\%$.
\emph{3.} Penghapus: $0.5/12 \approx 4\%$. Pengukuran pensilnya
sekitar sepuluh kali lebih teliti --- penggaris yang sama,
\omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian mutlak} yang sama, benda sepuluh kali lebih panjang.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:6}
\emph{1.} Proxima:
$4.24 \times \qty{9.47e15}{m} = \qty{4.01e16}{m}$. Sirius:
$8.6 \times \qty{9.47e15}{m} = \qty{8.1e16}{m}$ (dua \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting},
seperti masukannya).
\emph{2.} $t = d/c = \qty{2.4e13}{m} / \qty{3.00e8}{m/s}
= \qty{8.0e4}{s}$, yaitu $\num{8.0e4}/3600 \approx \qty{22}{h}$:
hampir satu hari penuh untuk setiap pesan dari wahana itu.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:7}
$t = d/c$ pada setiap kasus. Bulan:
$\qty{3.84e8}{m}/\qty{3.00e8}{m/s} = \qty{1.28}{s}$. Matahari:
$\qty{1.496e11}{m}/\qty{3.00e8}{m/s} = \qty{499}{s} \approx
\qty{8.3}{min}$. Neptunus:
$\qty{4.50e12}{m}/\qty{3.00e8}{m/s} = \qty{1.50e4}{s} \approx
\qty{4.2}{h}$. Ketika kita memandang Neptunus, kita melihatnya
sebagaimana ia sekitar empat jam yang lalu --- setiap teleskop adalah
mesin waktu yang sederhana.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:8}
\emph{1.} Satu lembar: $\qty{52.0}{mm}/500 = \qty{0.104}{mm}
= \qty{1.04e-4}{m}$. Ketidakpastiannya ikut terbagi:
$\qty{0.5}{mm}/500 = \qty{0.001}{mm}$, sehingga
$t = (104 \pm 1)\,\unit{\micro m}$.
\emph{2.} $1/104 \approx 1\%$.
\emph{3.} Mengukur satu lembar langsung akan memberikan
$(0.1 \pm 0.5)\,\unit{mm}$: \omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian relatif} sekitar $500\%$
--- penggarisnya bahkan tidak sanggup membuktikan lembar itu ada.
Menumpuk $500$ lembar mengalikan panjang terukurnya dengan $500$,
sementara \omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian mutlak} penggarisnya tetap: \omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian
relatifnya} menyusut dengan faktor $500$ yang sama.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:9}
Bagilah setiap panjangnya dengan $\qty{2e-10}{m}$. Bakteri:
$\qty{2.0e-6}{m}/\qty{2e-10}{m} = \num{1.0e4}$ atom. Rambut:
$\qty{7.0e-5}{m}/\qty{2e-10}{m} = \num{3.5e5}$. Manusia:
$1.7/\num{2e-10} = \num{8.5e9}$, \omterm{def:g10:orders-of-magnitude:oom}{orde besarnya} $10^{10}$: sepuluh
miliar atom dari ujung kepala sampai ujung kaki.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:10}
\emph{1.} $s = \qty{0.22}{m}/\qty{1.39e9}{m} = \num{1.58e-10}$.
\emph{2.} Bumi:
$\qty{1.27e7}{m} \times \num{1.58e-10} = \qty{2.0e-3}{m} =
\qty{2.0}{mm}$ --- sebutir merica --- yang mengorbit pada jarak
$\qty{1.496e11}{m} \times \num{1.58e-10} = \qty{23.7}{m} \approx
\qty{24}{m}$ dari bola sepak itu.
\emph{3.} Bulan:
$\qty{3.84e8}{m} \times \num{1.58e-10} = \qty{6.1}{cm}$ dari butir
merica itu. Neptunus:
$\qty{4.50e12}{m} \times \num{1.58e-10} \approx \qty{710}{m}$:
modelnya tidak muat di halaman sekolah.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:11}
\emph{1.} $21.0 \times 29.7 = \qty{623.7}{cm^2}$, dibulatkan ke tiga
\omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} masukannya: $\qty{624}{cm^2}$.
\emph{2.} $v = \qty{100.0}{m}/\qty{12.3}{s} = \qty{8.13}{m/s}$ (tiga
angka, ditentukan oleh waktunya).
\emph{3.} Lima \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} akan mengklaim luas itu sampai
$\qty{0.01}{cm^2}$, ketelitian yang tidak pernah diberikan masukan
setingkat milimeter --- desimal palsu tetaplah pemalsuan.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:12}
\emph{1.} Jari-jari $r = \qty{0.55}{m}$, luas
$A = \pi r^{2} = \pi \times 0.55^{2} \approx \qty{0.95}{m^2}$. Volume
$V = \qty{1.0}{mm^3} = \qty{1.0e-9}{m^3}$, dan selaput adalah (luas)
$\times$ (tebal), sehingga
\[
e = \frac{V}{A} = \frac{\qty{1.0e-9}{m^3}}{\qty{0.95}{m^2}}
= \qty{1.1e-9}{m} .
\]
\emph{2.} Sebuah molekul berorde $\qty{e-9}{m}$ --- satu anak tangga
di atas $\qty{e-10}{m}$ milik atom, sesuai dengan gambaran molekul
sebagai gugus kecil atom. Satu sendok teh minyak Franklin menenangkan
separuh hektar kolam dan, tanpa ia sadari, mengukur nanometer.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:13}
Anggapan: sekitar $70$ denyut per menit saat istirahat, usia hidup
$80$ tahun. Per hari: $70 \times 60 \times 24 \approx \num{1e5}$
denyut. Per tahun: $\num{1e5} \times 365 \approx \num{4e7}$. Sepanjang
hidup: $\num{4e7} \times 80 \approx \num{3e9}$ --- \omterm{def:g10:orders-of-magnitude:oom}{orde besarnya}
$10^{9}$, beberapa miliar detak. Empat \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} akan berlagak
mengetahui denyut jantung, menit demi menit, selama delapan puluh
tahun: masukannya hanya terkaan \omterm{def:g10:orders-of-magnitude:oom}{orde besar}, dan tidak ada keluaran
yang melampaui masukannya (\cref{met:g10:orders-of-magnitude:sigfig}).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:14}
Setelah $n$ lipatan tebalnya menjadi $2^{n} \times \qty{1.0e-4}{m}$.
\emph{1.} Diperlukan $2^{n} \geq 8848/\num{1.0e-4} = \num{8.8e7}$.
Karena $2^{26} \approx \num{6.7e7}$ masih kurang dan
$2^{27} \approx \num{1.3e8}$ sudah cukup: $27$ lipatan (tumpukannya
$\approx \qty{13}{km}$).
\emph{2.} Diperlukan $2^{n} \geq \num{3.84e12}$: $2^{41} \approx
\num{2.2e12}$ masih kurang, $2^{42} \approx \num{4.4e12}$ sudah
sampai --- $42$ lipatan, tumpukannya $\qty{4.4e8}{m}$, melewati Bulan
dengan leluasa.
\emph{3.} Hanya percobaannya: setiap lipatan memangkas luasnya menjadi
setengah, dan setelah tujuh lipatan tumpukannya lebih tebal daripada
lebarnya. Aritmetika penggandaannya tidak tersentuh --- pertumbuhan
eksponensial memang melampaui apa yang sanggup diikuti kertas (dan
naluri).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:15}
\emph{1.} Bulan: $\qty{0.03}{m}/\qty{3.84e8}{m} = \num{7.8e-11}$.
Meja: $\qty{0.001}{m}/\qty{1.000}{m} = \num{1.0e-3}$. Pengukuran
Bulannya lebih teliti dengan faktor
$\num{1.0e-3}/\num{7.8e-11} \approx \num{1.3e7}$ --- lebih dari
sepuluh juta kali.
\emph{2.} Menyamai $\num{7.8e-11}$ pada meja sepanjang satu \omterm{def:g10:orders-of-magnitude:unit}{meter}
berarti $\Delta x = \qty{7.8e-11}{m}$, lebih kecil daripada satu atom
($\approx \qty{e-10}{m}$): tepi mejanya bahkan tidak
\emph{terdefinisi} setajam itu. Secara relatif, pengukuran laser jarak
Bumi--Bulan termasuk pengukuran terhalus yang dimiliki spesies kita.
\end{solution}

\begin{solution}{pb:g10:orders-of-magnitude:1}
\textbf{1.} Butir: $\qty{0.5}{mm} = \qty{5e-4}{m}$, pada tingkat mili;
atom: $\qty{2.5e-10}{m} = \qty{0.25}{nm}$, pada tingkat nano.

\textbf{2.} $\qty{5e-4}{m}/\qty{2.5e-10}{m} = \num{2.0e6}$: dua juta
atom melintang.

\textbf{3.} Sebuah kubus berisi $2 \times 10^{6}$ atom per rusuk
memuat $(2\times10^{6})^{3} = 8 \times 10^{18}$ atom: \omterm{def:g10:orders-of-magnitude:oom}{orde besarnya}
$10^{19}$.

\textbf{4.} $V = (\qty{5e-4}{m})^{3} = \qty{1.25e-10}{m^3}$, sehingga
$m = \rho V = \qty{2500}{kg/m^3} \times \qty{1.25e-10}{m^3}
= \qty{3.1e-7}{kg} = \qty{0.31}{mg}$.

\textbf{5.} Massa per atom:
$\qty{3.1e-7}{kg}/(8\times10^{18}) = \qty{3.9e-26}{kg}$ --- tepat di
dalam rentang ``beberapa $10^{-26}\,\unit{kg}$'' itu. Kubus atom yang
sederhana ini menghasilkan kembali massa atom yang sebenarnya:
modelnya lulus pemeriksaan kewajaran
(\cref{met:g10:orders-of-magnitude:sanity}) dengan pujian.

\textbf{6.} $s = \qty{5e-4}{m}/\qty{1.39e9}{m} = \num{3.6e-13}$.

\textbf{7.} Model Bumi: diameternya
$\qty{1.27e7}{m} \times \num{3.6e-13} = \qty{4.6e-6}{m}$ --- seukuran
bakteri --- pada jarak
$\qty{1.496e11}{m} \times \num{3.6e-13} = \qty{5.4}{cm}$ dari butir
itu: Bumi adalah sebuah mikroba lima sentimeter dari sebutir pasir.

\textbf{8.} $\qty{4.5e12}{m} \times \num{3.6e-13} = \qty{1.6}{m}$:
seluruh sistem keplanetan mengorbit di dalam ruang selebar meja makan.

\textbf{9.} $\qty{4.0e16}{m} \times \num{3.6e-13} = \qty{1.4e4}{m}
= \qty{14}{km}$.

\textbf{10.} Bintang terdekat dari Tata Surya di atas meja itu adalah
sebutir pasir lain sejauh empat belas kilometer, dengan hampir tidak
ada apa-apa di antaranya: galaksi kita luar biasa kosong, dan itulah
sebabnya tumbukan antarbintang nyaris tidak pernah terdengar.

\textbf{11.} $t = \qty{5e-4}{m}/\qty{3.00e8}{m/s} = \qty{1.7e-12}{s}$
--- kurang dari dua pikosekon.

\textbf{12.} Bulan: $\qty{3.84e8}{m}/\qty{3.00e8}{m/s} =
\qty{1.28}{s}$. Matahari: $\qty{1.496e11}{m}/\qty{3.00e8}{m/s} =
\qty{499}{s} \approx \qty{8.3}{min}$: cahaya matahari selalu berumur
delapan menit.

\textbf{13.} Satu tahun adalah
$365.25 \times 86400 = \qty{3.156e7}{s}$, sehingga
$\qty{1}{ly} = \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}$; lalu
$\qty{4.0e16}{m}/\qty{9.47e15}{m} = 4.2$: jarak pada data itu memang
sekitar $\qty{4.2}{ly}$.

\textbf{14.} $\qty{2.4e22}{m}/\qty{9.47e15}{m} = \num{2.5e6}$: dua
setengah juta \omterm{def:g10:orders-of-magnitude:lightyear}{tahun cahaya}. Cahaya itu meninggalkan Andromeda sekitar
$2.5$ juta tahun yang lalu, ketika anggota pertama marga \emph{Homo}
belajar menyerpih batu --- dan belum ada satu pun \emph{Homo sapiens}
di mana pun.

\textbf{15.} $\qty{3.00e8}{m/s} \times \qty{1e-9}{s} = \qty{0.30}{m}$.
Dalam sepertiga nanosekon sebuah sinyal menempuh paling jauh sekitar
$\qty{10}{cm}$, sehingga dua bagian mesin yang harus saling berbicara
dalam satu detak jam wajib duduk dalam jarak sepuluh sentimeter satu
sama lain: laju membatasi ukuran.

\textbf{16.} Atom rata-rata di dalam air:
$\qty{3.0e-26}{kg}/3 = \qty{1.0e-26}{kg}$. Banyaknya atom:
$\qty{70}{kg}/\qty{1.0e-26}{kg} = \num{7e27}$.

\textbf{17.} Terhadap butir pasir:
$\num{7e27}/\num{8e18} \approx 10^{9}$ --- tubuhmu memuat atom
sebanyak semiliar butir pasir. Terhadap bintang:
$\num{7e27}/\num{2e23} = \num{3.5e4}$ --- atommu melebihi seluruh
bintang alam semesta teramati tiga puluh lima ribu banding satu.

\textbf{18.} Nisbah $\text{(kamu)}/\text{(atom)} =
\qty{1.7}{m}/\qty{2.5e-10}{m} \approx \num{7e9} \approx 10^{10}$,
sedangkan $\text{(alam semesta)}/\text{(kamu)} =
\qty{8.8e26}{m}/\qty{1.7}{m} \approx \num{5e26}$. Pada tangga itu kamu
berdiri sekitar sepuluh anak tangga di atas atom dan hampir dua puluh
tujuh di bawah alam semesta: secara perkalian, manusia hidup sangat
dekat dengan atomnya.

\textbf{19.} Syaratnya memberikan
$x^{2} = 10^{-15} \times 10^{26.9} = 10^{11.9}$, sehingga
$x = 10^{5.95} \approx \qty{9e5}{m} \approx \qty{900}{km}$ --- hampir
persis diameter Ceres, asteroid terbesar
($\approx \qty{940}{km}$). Tepat di tengah tangga antara inti atom dan
kosmos duduk sebuah planet kerdil yang bersahaja.

\textbf{20.} $\qty{8.8e26}{m} \times \num{3.6e-13} = \qty{3.2e14}{m}$,
yaitu $\qty{3.2e14}{m}/\qty{1.496e11}{m} \approx \num{2.1e3}$
\omterm{def:g10:orders-of-magnitude:lightyear}{satuan astronomi}. Puncak ceritanya: susutkan kosmos sampai Matahari
menjadi sebutir pasir, dan alam semesta teramati masih terbentang dua
ribu kali jarak Bumi--Matahari yang sebenarnya --- tidak ada model
berskala yang membawa alam semesta turun ke ukuran manusia; hanya
pangkat sepuluh yang sanggup memikulnya.
\end{solution}
