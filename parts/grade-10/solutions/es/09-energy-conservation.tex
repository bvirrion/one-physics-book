\section*{Capítulo \ref{ch:g10:energy-conservation} --- La energía: formas y conservación}

\begin{solution}{exo:g10:energy-conservation:1}
$E_k = \tfrac12 \times 1200 \times 25^2 = \qty{3.75e5}{J}$. A
\qty{36.1}{m/s}:
$\tfrac12 \times 1200 \times 36.1^2 \approx \qty{7.8e5}{J}$ --- un
$44\,\%$ más de velocidad, $(130/90)^2 \approx 2.1$ veces la \omterm{def:g10:energy-conservation:kinetic}{energía
cinética}: el cuadrado en acción.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:2}
$\Delta h = \qty{450}{m}$:
$E_p = 72 \times 9.81 \times 450 \approx \qty{3.2e5}{J}
= \num{3.2e5}/\num{3.6e6} \approx \qty{0.088}{kW.h}$, que valen unos
$0.022$ euros. Un día de subida son dos céntimos de electricidad.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:3}
$\qty{1}{kW.h} = \qty{3.6e6}{J}$;
$\qty{12}{W.h} = 12 \times \qty{3600}{J} \approx \qty{43}{kJ}$;
$\qty{45}{MJ} = 45/3.6 = \qty{12.5}{kW.h}$;
$\qty{10}{MJ} = 10/3.6 \approx \qty{2.8}{kW.h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:4}
$E = 2200 \times 150 = \qty{3.3e5}{J}
= \num{3.3e5}/\num{3.6e6} \approx \qty{0.092}{kW.h}$. Persona:
$P = \num{1.0e7}/\num{86400} \approx \qty{116}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:5}
Dinamo: química (comida del ciclista) $\to$ cinética (rueda) $\to$
eléctrica (dinamo) $\to$ radiante $+$ térmica (lámpara); hay fugas de
calor en los músculos, en el contacto del neumático, en la dinamo y en
la lámpara. Calentador de gas: química (gas) $\to$ térmica (agua), con
fuga en los gases de la chimenea. Panel: radiante (Sol) $\to$ eléctrica
(panel) $\to$ química (batería), con fugas de calor en el panel y en el
cargador.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:6}
Consumidos $2.0 \times 45 = \qty{90}{MJ}$; $\eta = 27/90 = 0.30$.
Calor: $90 - 27 = \qty{63}{MJ}$, que se llevan al aire de alrededor los
gases de escape y el radiador.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:7}
$E_k = E_p = 70 \times 9.81 \times 20 \approx \qty{1.37e4}{J}$;
$v = \sqrt{2 \times 13734/70} = \sqrt{392} \approx \qty{19.8}{m/s}
\approx \qty{71}{km/h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:8}
$41562 - 41236 = \qty{326}{kW.h}$; al día,
$326/30 \approx \qty{10.9}{kW.h}$. Factura:
$326 \times 0.26 + 12 = 96.76$ euros. \omterm{def:g10:energy-conservation:power}{Potencia} media:
$\qty{326}{kW.h}/\qty{720}{h} \approx \qty{453}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:9}
\emph{1.} Luz: $0.05 \times 60 = \qty{3.0}{W}$; led:
$\eta = 3/9 \approx 0.33$.
\emph{2.} Al año ($\qty{1460}{h}$): bombilla
$60 \times 1460 = \qty{87.6}{kW.h}$, led \qty{13.1}{kW.h}; ahorro
$\qty{74.5}{kW.h} \approx 19$ euros por bombilla y año.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:10}
\emph{1.} $P = \num{1.5e5} \times 9.81 \times 40 \approx
\qty{5.9e7}{W} = \qty{59}{MW}$.
\emph{2.} $0.90 \times 59 \approx \qty{53}{MW}$;

\begin{solution}{exo:g10:energy-conservation:11}
$v = \sqrt{2 \times 9.81 \times 0.050} \approx \qty{0.99}{m/s}$ (la masa
se cancela: $\tfrac12 m v^2 = mgh$). La
$E_p = 0.20 \times 9.81 \times 0.050 \approx \qty{0.098}{J}$ inicial
acaba como \omterm{def:g10:energy-conservation:forms}{energía térmica} en el aire y en el eje: el péndulo no está
aislado, y el total --- oscilación más calor --- se conserva
exactamente.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:12}
\emph{1.} $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.304$;
consumidos $12/0.304 \approx \qty{39.5}{W.h}
= 39.5 \times 3600 \approx \qty{142}{kJ}$.
\emph{2.} $\num{1.42e5}/\num{2.5e7} \approx \qty{5.7}{g}$ de carbón por
carga; $\times 365 \approx \qty{2.1}{kg}$ al año.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:13}
\emph{1.} $\num{5e9}/\num{3.6e6} \approx \qty{1400}{kW.h}$;
$1400/12.6 \approx 110$ días.
\emph{2.} $P = \num{5e9}/\num{1e-4} = \qty{5e13}{W} = \qty{50}{TW}$ ---
diecisiete veces la \omterm{def:g10:energy-conservation:power}{potencia} eléctrica media de la humanidad, durante
\qty{100}{\micro s} y en un lugar impredecible. Un rayo es mucha
\omterm{def:g10:energy-conservation:power}{potencia} pero poca \omterm{def:g10:energy-conservation:energy}{energía}, entregada a una velocidad inútil: ningún
\omterm{def:g10:energy-conservation:transfer}{convertidor} acepta semejante pico.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:14}
\emph{1.} $E_k = \tfrac12 \times 1300 \times 30.6^2 \approx
\qty{6.1e5}{J}$, convertida en \omterm{def:g10:energy-conservation:forms}{energía térmica} en los discos y las
pastillas de freno (y después en el aire).
\emph{2.} $0.60 \times \qty{6.1e5}{J} \approx \qty{3.7e5}{J}$ por
frenada; $\num{3.6e6}/\num{3.7e5} \approx 10$ frenadas por
\omterm{def:g10:energy-conservation:kwh}{kilovatio-hora}.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:15}
$E_p = 85 \times 9.81 \times 1200 \approx \qty{1.0}{MJ}$; comida:
$1.0/0.25 = \qty{4.0}{MJ}$, es decir $4.0/1.5 \approx 2.7$ tazones de
cereales --- o $4.0/45 \approx \qty{0.09}{kg}$ de gasolina. Una mañana
entera de esfuerzo honrado cabe en una taza de café de combustible: la
\omterm{def:g10:energy-conservation:forms}{energía química} está absurdamente concentrada.
\end{solution}

\begin{solution}{pb:g10:energy-conservation:1}
\textbf{1.} Frigorífico $0.1 \times 24 = \qty{2.4}{kW.h}$; calentador
$2.4 \times 2.5 = \qty{6.0}{kW.h}$; horno \qty{2.0}{kW.h}; lavadora
\qty{1.0}{kW.h}; iluminación \qty{0.3}{kW.h}; pantallas
\qty{0.9}{kW.h}.
\textbf{2.} Total \qty{12.6}{kW.h} $= 12.6 \times 3.6 \approx
\qty{45}{MJ}$: un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina, \omterm{def:g10:energy-conservation:energy}{julio} a \omterm{def:g10:energy-conservation:energy}{julio}.
\textbf{3.} $\qty{12600}{W.h}/\qty{24}{h} = \qty{525}{W}$ --- la casa
consume de fondo un cuarto de hervidor.
\textbf{4.} El calentador de agua (\qty{6.0}{kW.h}, casi la mitad). Con
\qty{1.25}{h}: calentador \qty{3.0}{kW.h}, total \qty{9.6}{kW.h}.
\textbf{5.} $15 \times 24 = \qty{0.36}{kW.h}$ al día, un $2.8\,\%$ del
total del contador; al año,
$0.36 \times 365 \approx \qty{131}{kW.h} \approx 33$ euros --- por unos
pilotos que nadie mira.
\textbf{6.} $56604 - 56214 = \qty{390}{kW.h}$;
$390/30 = \qty{13.0}{kW.h}$ al día: los \qty{12.6}{kW.h} de la
auditoría más los \qty{0.36}{kW.h} de los pilotos en espera ---
confirmado.
\textbf{7.} $390 \times 0.25 + 14 = 111.50$ euros.
\textbf{8.} Red: $0.25/3.6 \approx 0.07$ euros por megajulio.
Gasolina: $1.80/34 \approx 0.05$ euros por megajulio.
\textbf{9.} Tazón: $0.50/1.5 \approx 0.33$ euros por megajulio ---
cinco veces la red. De más barato a más caro: gasolina, red y, muy por
detrás, comida.
\textbf{10.} $0.20 \times 1000 = \qty{200}{W}$ por \omterm{def:g10:orders-of-magnitude:unit}{metro} cuadrado.
\textbf{11.} $200 \times 4.0 = \qty{800}{W.h} = \qty{0.80}{kW.h}$ por
\omterm{def:g10:orders-of-magnitude:unit}{metro} cuadrado y día.
\textbf{12.} $12.6/0.80 = 15.75 \approx \qty{16}{m^2}$: cabe en el
tejado de \qty{20}{m^2}.
\textbf{13.} $16 \times 0.2 \times 1.5 = \qty{4.8}{kW.h}$: un $38\,\%$ de
la necesidad; el resto tiene que ponerlo la red (o una batería cargada
en meses mejores).
\textbf{14.} Un \omterm{def:g10:orders-of-magnitude:unit}{metro} cuadrado produce
$0.80 \times 365 \approx \qty{292}{kW.h}$ al año; retorno
$700/292 \approx 2.4$ años --- poco frente a los 25 años de vida de un
panel.
\textbf{15.} $\qty{12.6}{kW.h}/\qty{0.1}{kW} = \qty{126}{h}$: más de
cinco personas ($126/24 = 5.25$) pedaleando día y noche.
\textbf{16.} Comida $= 12.6/0.25 = \qty{50.4}{kW.h} \approx
\qty{181}{MJ}$, es decir $181/1.5 \approx 121$ tazones de cereales.
\textbf{17.} $121 \times 0.50 \approx 60$ euros, frente a los
$12.6 \times 0.25 \approx 3.15$ euros de la red: la \omterm{def:g10:energy-conservation:energy}{energía} a pedales
cuesta unas $19$ veces más --- y sin pagar a los ciclistas.
\textbf{18.} $5000/45 \approx 110$ días de casa. Pero
$\num{5e9}/\num{1e-4} = \qty{5e13}{W}$ durante \qty{100}{\micro s} y en
un lugar al azar: ningún \omterm{def:g10:energy-conservation:transfer}{convertidor} ni batería puede beber de esa
manguera. Una casa necesita un caudal constante de \qty{525}{W}, no un
pico.
\textbf{19.} Los \omterm{def:g10:energy-conservation:energy}{julios} no --- se conservan y son idénticos. Lo que se
paga es su \emph{forma y su disponibilidad}: una \omterm{def:g10:energy-conservation:energy}{energía} concentrada,
almacenable y a demanda cuesta toda la \omterm{def:g10:energy-conservation:transfer}{cadena de conversión} que la
volvió así.
\textbf{20.} Un día de la familia $= \qty{12.6}{kW.h} \approx
\qty{45}{MJ}$: un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de gasolina, dieciséis \omterm{def:g10:orders-of-magnitude:unit}{metros} cuadrados de
tejado soleado o cinco ciclistas sin descanso. Las existencias son la
\omterm{def:g10:energy-conservation:energy}{energía} (\omterm{def:g10:energy-conservation:energy}{julios}, kilovatios-hora); el caudal es la \omterm{def:g10:energy-conservation:power}{potencia} (\omterm{def:g10:energy-conservation:power}{vatios}) ---
la bañera y el grifo, nunca más confundidos.
\end{solution}
