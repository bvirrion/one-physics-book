\section*{Capítulo \ref{ch:g10:universal-gravitation} --- Gravitación universal y peso}

\begin{solution}{exo:g10:universal-gravitation:1}
\emph{1.} $F = \num{6.67e-11} \times \dfrac{60 \times 60}{0.80^2}
= \dfrac{\num{2.40e-7}}{0.64} \approx \qty{3.8e-7}{N}$.

\emph{2.} El \omterm{def:g10:universal-gravitation:weight}{peso} del mosquito vale
$P = \num{2.5e-6} \times 9.81 \approx \qty{2.5e-5}{N}$ --- unas $65$
veces \emph{mayor} que la atracción mutua de los bailarines. La gravedad
entre personas es más débil que el \omterm{def:g10:universal-gravitation:weight}{peso} de un mosquito.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:2}
\emph{1.} $P = mg = 55 \times 9.81 \approx \qty{5.4e2}{N}$.

\emph{2.} Su masa no cambia: \qty{55}{kg} (la misma cantidad de
materia). Su \omterm{def:g10:universal-gravitation:weight}{peso} pasa a ser $P = 55 \times 1.62 \approx \qty{89}{N}$
--- seis veces menos.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:3}
\emph{1.} $g_{\text{Mars}} = \dfrac{G M}{R^2}
= \dfrac{\num{6.67e-11} \times \num{6.42e23}}{(\num{3.39e6})^2}
= \dfrac{\num{4.28e13}}{\num{1.15e13}} \approx \qty{3.73}{N/kg}$.

\emph{2.} En Marte: $P = 900 \times 3.73 \approx \qty{3.4e3}{N}$. En la
Tierra: $P = 900 \times 9.81 \approx \qty{8.8e3}{N}$ --- el vehículo
pesa unas $2.6$ veces menos en Marte, con la misma masa.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:4}
\[
F = \num{6.67e-11} \times
\frac{\num{5.97e24} \times \num{7.35e22}}{(\num{3.84e8})^2}
= \frac{\num{2.93e37}}{\num{1.47e17}}

La Luna ejerce sobre la Tierra una fuerza del \emph{mismo} módulo,

\end{solution}

\begin{solution}{exo:g10:universal-gravitation:5}
\emph{1.} Distancia duplicada: $d^2$ queda multiplicado por $4$, así que
$F = F_0/4$.

\emph{2.} Distancia a la mitad: $F = 4F_0$.

\emph{3.} Las dos masas duplicadas: $m_1 m_2$ queda multiplicado por
$4$, así que $F = 4F_0$.

\emph{4.} Numerador $\times 3$, denominador $\times 9$: $F = F_0/3$.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:6}
\emph{1.} $F = \num{6.67e-11} \times
\dfrac{(\num{5.0e8})^2}{100^2}
= \num{6.67e-11} \times \dfrac{\num{2.5e17}}{\num{1.0e4}}
\approx \qty{1.7e3}{N}$.

\emph{2.} $m = F/g = 1667/9.81 \approx \qty{1.7e2}{kg}$: los dos barcos
se atraen con el \omterm{def:g10:universal-gravitation:weight}{peso} de una motocicleta grande.

\emph{3.} $a = F/m = \num{1.7e3}/\num{5.0e8} \approx
\qty{3.3e-6}{m/s^2}$. Aun sin nada que se opusiera, los petroleros
tardarían unos cinco días en alcanzar la velocidad de un paseo; en la
práctica, la resistencia del agua y las amarras ahogan la atracción por
completo.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:7}
\emph{1.} La balanza de brazos indica \qty{3.0}{kg} en los dos mundos:
la bolsa y las masas patrón ven sus \omterm{def:g10:universal-gravitation:weight}{pesos} divididos por el mismo factor,
así que la comparación no cambia. El dinamómetro marca el \omterm{def:g10:universal-gravitation:weight}{peso}:
$3.0 \times 9.81 \approx \qty{29}{N}$ en la Tierra y
$3.0 \times 1.62 \approx \qty{4.9}{N}$ en la Luna.

\emph{2.} La balanza: mide la masa, que es lo que el cliente compra. Un
dinamómetro graduado en ``\omterm{def:g10:orders-of-magnitude:unit}{kilogramos}'' en la Tierra marcaría
$4.9/9.81 \approx \qty{0.50}{kg}$ para esa misma bolsa en la Luna --- el
tendero regalaría seis bolsas al precio de una.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:8}
\emph{1.} Distancia al centro:
$d = R + h = \num{6.37e6} + \num{4.0e5} = \qty{6.77e6}{m}$. Entonces
$g = \dfrac{\num{6.67e-11} \times \num{5.97e24}}{(\num{6.77e6})^2}
= \dfrac{\num{3.98e14}}{\num{4.58e13}} \approx \qty{8.69}{N/kg}$.

\emph{2.} $8.69/9.81 \approx 89\,\%$ del valor en la superficie.

\emph{3.} Ahí arriba la gravedad conserva casi toda su fuerza: los
astronautas flotan porque la estación y todo lo que hay dentro
\emph{caen juntos} alrededor de la Tierra (caída libre), no porque la
gravedad haya desaparecido.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:9}
De $g = GM/R^2$,
\[
M = \frac{g R^2}{G}
= \frac{8.87 \times (\num{6.05e6})^2}{\num{6.67e-11}}
= \frac{\num{3.25e14}}{\num{6.67e-11}}
\approx \qty{4.87e24}{kg},
\]
unas $0.8$ veces la masa de la Tierra --- Venus es casi la gemela de la
Tierra en tamaño y masa.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:10}
\emph{1.} Sol:
$F = \dfrac{\num{6.67e-11} \times \num{1.99e30}}{(\num{1.496e11})^2}
\approx \qty{5.9e-3}{N}$. Luna:
$F = \dfrac{\num{6.67e-11} \times \num{7.35e22}}{(\num{3.84e8})^2}
\approx \qty{3.3e-5}{N}$.

\emph{2.} El Sol tira unas $\num{5.9e-3}/\num{3.3e-5} \approx 180$ veces
más fuerte.

\emph{3.} El diámetro de la Tierra es una \emph{fracción} mucho mayor de
la distancia Tierra--Luna ($\approx 3\,\%$) que de la distancia
Tierra--Sol ($\approx 0.01\,\%$), de modo que el tirón de la Luna cambia
mucho más de la cara próxima a la cara lejana del globo. Las mareas se
alimentan de esa diferencia --- y ahí gana la Luna.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:11}
\emph{1.} $g' = \dfrac{G (2M)}{(2R)^2} = \dfrac{2}{4}\,\dfrac{GM}{R^2}
= \dfrac{g}{2} \approx \qty{4.9}{N/kg}$: el radio duplicado le gana a la
masa duplicada.

\emph{2.} Hace falta $\dfrac{GM}{R'^2} = 2\,\dfrac{GM}{R^2}$, de donde
$R'^2 = R^2/2$ y $R' = R/\sqrt{2} \approx \qty{4.5e6}{m}$ --- unos
\qty{4500}{km}.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:12}
\emph{1.} Tirones por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} iguales:
$\dfrac{G M_{\text{Earth}}}{x^2} = \dfrac{G M_{\text{Moon}}}{(D-x)^2}$.
Multiplicando en cruz y dividiendo entre $G$:
$\left(\dfrac{x}{D-x}\right)^2 = \dfrac{M_{\text{Earth}}}{M_{\text{Moon}}}$.

\emph{2.} $\dfrac{x}{D-x}
= \sqrt{\dfrac{\num{5.97e24}}{\num{7.35e22}}} = \sqrt{81.2} \approx
9.0$, luego $x = 9.0\,(D - x)$, es decir $x = \dfrac{9.0}{10.0}\,D
\approx 0.90 \times \num{3.84e8} \approx \qty{3.5e8}{m}$. El punto de
equilibrio está al $90\,\%$ del camino hacia la Luna: el tirón de la
Tierra domina casi todo el viaje.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:13}
\emph{1.} Desarrollando: $R^2 + 2Rh + h^2 = R^2 + x^2$; como $h$ es
diminuta frente a $R$, se desprecia $h^2$: $2Rh \approx x^2$, luego
$h \approx \dfrac{x^2}{2R}
= \dfrac{(\num{8.0e3})^2}{2 \times \num{6.37e6}}
= \dfrac{\num{6.4e7}}{\num{1.27e7}} \approx \qty{5.0}{m}$.

\emph{2.} En un segundo el proyectil cae \qty{4.9}{m} --- casi
exactamente los \qty{5.0}{m} que el suelo se aparta en \qty{8.0}{km}.
Así que a $v \approx \qty{8.0}{km/s}$ la caída no alcanza nunca el
suelo: el proyectil entra en \omterm{def:g10:universal-gravitation:satellite}{órbita}.

\emph{3.} $T = \dfrac{2\pi R}{v}
= \dfrac{2\pi \times \num{6.37e6}}{\num{8.0e3}}
\approx \qty{5.0e3}{s} \approx \qty{83}{min}$ --- cerca de los
$\qty{90}{min}$ aproximados de los \omterm{def:g10:universal-gravitation:satellite}{satélites} reales de \omterm{def:g10:universal-gravitation:satellite}{órbita} baja, que
vuelan un poco más alto.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:14}
\emph{1.} $g = \dfrac{\num{6.67e-11} \times \num{2.8e30}}
{(\num{1.2e4})^2} = \dfrac{\num{1.87e20}}{\num{1.44e8}}
\approx \qty{1.3e12}{N/kg}.$

\emph{2.} $P = 70 \times \num{1.3e12} \approx \qty{9.1e13}{N}$, frente a

\emph{3.} $P = \num{1.0e-6} \times \num{1.3e12} \approx

\end{solution}

\begin{solution}{exo:g10:universal-gravitation:15}
\emph{1.} La distancia es $60$ veces mayor, así que el tirón por
\omterm{def:g10:orders-of-magnitude:unit}{kilogramo} es $60^2 = 3600$ veces más débil.

\emph{2.} La caída en un segundo se reduce en el mismo factor:
$4.9/3600 \approx \qty{1.4e-3}{m}$ --- unos \qty{1.4}{mm}.

\emph{3.} $T = 27.3 \times \qty{86400}{s} = \qty{2.36e6}{s}$, de modo
que $v = \dfrac{2\pi \times \num{3.84e8}}{\num{2.36e6}} \approx
\qty{1.02e3}{m/s}$, y $x = \qty{1.02e3}{m}$ en un segundo. La caída vale
$h \approx \dfrac{x^2}{2d}
= \dfrac{(\num{1.02e3})^2}{2 \times \num{3.84e8}}
\approx \qty{1.4e-3}{m}$.

\emph{4.} Los dos números coinciden: la Luna cae hacia la Tierra cada
segundo exactamente lo que predice la gravedad de la inversa del
cuadrado, calibrada con una manzana que cae. Newton concluyó que la
fuerza que sujeta a la Luna \emph{es} la que deja caer la manzana: la
gravedad es universal.
\end{solution}

\begin{solution}{pb:g10:universal-gravitation:1}
\textbf{1.} $F = \num{6.67e-11} \times
\dfrac{\num{7.3e6} \times 55}{100^2} \approx \qty{2.7e-6}{N}$.

\textbf{2.} El grano de arena pesa
$\num{1.0e-6} \times 9.81 \approx \qty{9.8e-6}{N}$: la torre Eiffel
entera tira del turista unas $3.7$ veces \emph{menos} que el \omterm{def:g10:universal-gravitation:weight}{peso} de un
grano de arena.

\textbf{3.} (a) Distancia $\times 10$: fuerza dividida entre $100$.
(b) Las dos masas $\times 1000$: fuerza multiplicada por $10^6$.
(c) Masas $\times 1000$ y distancia $\times 1000$: $10^6 / 10^6$ --- la
fuerza no cambia.

\textbf{4.} $F = \dfrac{\num{6.67e-11} \times \num{5.97e24} \times
1.00}{(\num{6.37e6})^2} \approx \qty{9.81}{N}$: el \omterm{def:g10:universal-gravitation:weight}{peso} de un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo}
--- que es exactamente la intensidad del campo,
$g = \qty{9.81}{N/kg}$.

\textbf{5.} La gravedad es endeble por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo}, pero solo
\emph{atrae}, de modo que los tirones de los $\num{5.97e24}$ \omterm{def:g10:orders-of-magnitude:unit}{kilogramos}
de la Tierra se suman en la misma dirección --- y una masa astronómica
le gana a una masa vecina por veintidós potencias de diez.

\textbf{6.} $F = \num{6.67e-11} \times \dfrac{158 \times 0.73}{0.23^2}
= \num{6.67e-11} \times \dfrac{115.3}{0.0529}
\approx \qty{1.5e-7}{N}$.

\textbf{7.} La esfera pequeña pesa
$0.73 \times 9.81 \approx \qty{7.2}{N}$ --- unas $\num{5e7}$ veces la
atracción que hay que detectar. Ninguna balanza de platillos resuelve
una parte en cincuenta millones; un hilo fino de torsión, que se
retuerce de forma visible con pares del orden del nanonewton, sí.

\textbf{8.} $G = \dfrac{F d^2}{m_1 m_2}
= \dfrac{\num{1.47e-7} \times 0.0529}{115.3}
\approx \qty{6.74e-11}{N.m^2/kg^2}$ --- a un $1\,\%$ escaso del
\num{6.67e-11} moderno, desde un cobertizo de madera en 1798.

\textbf{9.} De $g = G M / R^2$:
\[
M = \frac{g R^2}{G}
= \frac{9.81 \times (\num{6.37e6})^2}{\num{6.67e-11}}
\approx \qty{5.97e24}{kg}.
\]
Seis cuatrillones de \omterm{def:g10:orders-of-magnitude:unit}{kilogramos}: la Tierra, pesada.

\textbf{10.} $V = \frac43 \pi R^3 = \frac43 \pi
(\num{6.37e6})^3 \approx \qty{1.08e21}{m^3}$, así que
$\rho = \num{5.97e24}/\num{1.08e21} \approx \qty{5.5e3}{kg/m^3}$ --- el
doble de la densidad de la roca superficial. El interior debe esconder
algo mucho más denso que el granito: la Tierra tiene un \omterm{def:g10:refraction:fiber}{núcleo} pesado,
de hierro.

\textbf{11.} $g_{\text{Moon}} = \dfrac{\num{6.67e-11} \times
\num{7.35e22}}{(\num{1.74e6})^2} = \dfrac{\num{4.90e12}}
{\num{3.03e12}} \approx \qty{1.62}{N/kg}$.

\textbf{12.} Altura $\propto 1/g$:
$h = 0.50 \times \dfrac{9.81}{1.62} \approx \qty{3.0}{m}$ --- un salto
sin carrerilla por encima de una canasta de baloncesto, a cámara lenta.

\textbf{13.} $g_{\text{Mercury}} = \dfrac{\num{6.67e-11} \times
\num{3.30e23}}{(\num{2.44e6})^2} = \dfrac{\num{2.20e13}}
{\num{5.95e12}} \approx \qty{3.70}{N/kg}$ --- casi exactamente los
\qty{3.73}{N/kg} de Marte. Mercurio tiene menos masa, pero su radio
menor (un $R^2$ menor abajo) lo compensa: es un mundo mucho más denso.

\textbf{14.} La mitad de la intensidad del campo exige
$(R + h)^2 = 2 R^2$, luego $R + h = R\sqrt{2}$ y
$h = (\sqrt{2} - 1) R \approx 0.414 \times \num{6.37e6}
\approx \qty{2.6e6}{m}$ --- unos \qty{2600}{km} de altitud.

\textbf{15.} $\dfrac{G M_{\text{Earth}}}{d^2} = 1.62$ da
$d = \sqrt{\dfrac{\num{6.67e-11} \times \num{5.97e24}}{1.62}}
= \sqrt{\num{2.46e14}} \approx \qty{1.57e7}{m} \approx 2.5$ radios
terrestres. Desde solo un radio y medio por encima del suelo, la Tierra
ya no tira más fuerte que la superficie de la Luna.

\textbf{16.} $v = \dfrac{2\pi d}{T}
= \dfrac{2\pi \times \num{1.496e11}}{\num{3.156e7}}
\approx \qty{2.98e4}{m/s}$ --- treinta kilómetros cada segundo.

\textbf{17.} $\dfrac{v^2}{d} = \dfrac{(\num{2.98e4})^2}{\num{1.496e11}}
\approx \qty{5.9e-3}{N/kg}$: el Sol tira de cada \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de la Tierra
con unos seis milinewtons.

\textbf{18.} $\dfrac{G M_{\text{Sun}}}{d^2} = \dfrac{v^2}{d}$ da
\[
M_{\text{Sun}} = \frac{v^2 d}{G}
= \frac{\num{8.87e8} \times \num{1.496e11}}{\num{6.67e-11}}
\approx \qty{1.99e30}{kg}.
\]

\textbf{19.} $\dfrac{\num{1.99e30}}{\num{5.97e24}} \approx
\num{3.3e5}$: el Sol es un tercio de millón de Tierras.

\textbf{20.} Como la misma $G$ gobierna toda pareja de masas, medirla
una sola vez entre dos esferas de plomo en un cobertizo nos permitió
poner en la balanza primero el planeta que pisamos y después la estrella
que orbitamos --- eso es lo que vale la palabra \emph{universal}.
\end{solution}
