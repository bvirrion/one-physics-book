\section*{Capítulo \ref{ch:g10:orders-of-magnitude} --- Órdenes de magnitud: medir el universo}

\begin{solution}{exo:g10:orders-of-magnitude:1}
\emph{1.} $\qty{750}{nm} = 750 \times \qty{e-9}{m} = \qty{7.50e-7}{m}$.
\emph{2.} $\qty{2.5}{km} = \qty{2.5e3}{m}$.
\emph{3.} $\qty{3.2}{\micro m} = \qty{3.2e-6}{m}$.
\emph{4.} $\qty{0.15}{Tm} = 0.15 \times \qty{e12}{m} = \qty{1.5e11}{m}$
--- una \omterm{def:g10:orders-of-magnitude:lightyear}{unidad astronómica}.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:2}
\emph{1.} $\qty{149600000}{km} = \qty{1.496e11}{m}$: $4$ \omterm{def:g10:orders-of-magnitude:sigfig}{cifras
significativas}.
\emph{2.} $\qty{1.06e-10}{m}$: $3$ \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas} (los ceros
iniciales nunca cuentan).
\emph{3.} $\num{8.1e9}$: $2$ \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas}.
\emph{4.} $\qty{5.20e-3}{s}$: $3$ \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas} (el cero final
es una promesa).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:3}
\omterm{def:g10:orders-of-magnitude:scinot}{Mantisas} y \omterm{def:g10:orders-of-magnitude:scinot}{exponentes} por separado
(\cref{prop:g10:orders-of-magnitude:powers}):
\[
3 \times 2 \times 10^{8-5} = \num{6e3}, \qquad
\tfrac{8}{4} \times 10^{4-(-3)} = \num{2e7},
\]
\[
5^{2} \times 10^{12} = \num{2.5e13}, \qquad
\tfrac{4 \times 3}{6} \times 10^{-3-9+7} = \num{2e-5} .
\]
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:4}
Persona: $1.7 < 5$, orden $\qty{e0}{m}$. Everest:
$\qty{8848}{m} = \qty{8.848e3}{m}$ y $8.848 \geq 5$: orden
$\qty{e4}{m}$. Bacteria: orden $\qty{e-6}{m}$. Tierra: $5.97 \geq 5$,
orden $\qty{e25}{kg}$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:5}
\emph{1.} Media graduación: $L = (127.0 \pm 0.5)\,\unit{mm}$.
\emph{2.} $\Delta L / L = 0.5/127 \approx 0.4\,\%$.
\emph{3.} Goma: $0.5/12 \approx 4\,\%$. La medida del lápiz es unas diez
veces más precisa --- la misma regla, la misma \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre absoluta},
un objeto diez veces más largo.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:6}
\emph{1.} Próxima:
$4.24 \times \qty{9.47e15}{m} = \qty{4.01e16}{m}$. Sirio:
$8.6 \times \qty{9.47e15}{m} = \qty{8.1e16}{m}$ (dos \omterm{def:g10:orders-of-magnitude:sigfig}{cifras
significativas}, como el dato).
\emph{2.} $t = d/c = \qty{2.4e13}{m} / \qty{3.00e8}{m/s}
= \qty{8.0e4}{s}$, es decir $\num{8.0e4}/3600 \approx \qty{22}{h}$:
casi un día entero por cada mensaje de la sonda.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:7}
$t = d/c$ en cada caso. Luna:
$\qty{3.84e8}{m}/\qty{3.00e8}{m/s} = \qty{1.28}{s}$. Sol:
$\qty{1.496e11}{m}/\qty{3.00e8}{m/s} = \qty{499}{s} \approx
\qty{8.3}{min}$. Neptuno:
$\qty{4.50e12}{m}/\qty{3.00e8}{m/s} = \qty{1.50e4}{s} \approx
\qty{4.2}{h}$. Cuando miras Neptuno, lo ves tal como era hace unas
cuatro horas --- todo telescopio es una máquina del tiempo modesta.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:8}
\emph{1.} Una hoja: $\qty{52.0}{mm}/500 = \qty{0.104}{mm}
= \qty{1.04e-4}{m}$. La incertidumbre se divide igual:
$\qty{0.5}{mm}/500 = \qty{0.001}{mm}$, así que
$t = (104 \pm 1)\,\unit{\micro m}$.
\emph{2.} $1/104 \approx 1\,\%$.
\emph{3.} Medir una sola hoja daría $(0.1 \pm 0.5)\,\unit{mm}$: una
\omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre relativa} de unos $500\,\%$ --- la regla ni siquiera puede
demostrar que la hoja existe. Apilar $500$ hojas multiplica por $500$ la
longitud medida mientras la \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre absoluta} de la regla sigue
fija: la \omterm{def:g10:orders-of-magnitude:uncertainty}{incertidumbre relativa} se reduce en ese mismo factor $500$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:9}
Se divide cada longitud entre $\qty{2e-10}{m}$. Bacteria:
$\qty{2.0e-6}{m}/\qty{2e-10}{m} = \num{1.0e4}$ átomos. Cabello:
$\qty{7.0e-5}{m}/\qty{2e-10}{m} = \num{3.5e5}$. Persona:
$1.7/\num{2e-10} = \num{8.5e9}$, \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud} $10^{10}$: diez mil
millones de átomos de la cabeza a los pies.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:10}
\emph{1.} $s = \qty{0.22}{m}/\qty{1.39e9}{m} = \num{1.58e-10}$.
\emph{2.} Tierra:
$\qty{1.27e7}{m} \times \num{1.58e-10} = \qty{2.0e-3}{m} =
\qty{2.0}{mm}$ --- un grano de pimienta --- que orbita a
$\qty{1.496e11}{m} \times \num{1.58e-10} = \qty{23.7}{m} \approx
\qty{24}{m}$ del balón.
\emph{3.} Luna:
$\qty{3.84e8}{m} \times \num{1.58e-10} = \qty{6.1}{cm}$ del grano de
pimienta. Neptuno:
$\qty{4.50e12}{m} \times \num{1.58e-10} \approx \qty{710}{m}$: la
maqueta no cabe en el patio del colegio.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:11}
\emph{1.} $21.0 \times 29.7 = \qty{623.7}{cm^2}$, redondeado a las tres
\omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas} de los datos: $\qty{624}{cm^2}$.
\emph{2.} $v = \qty{100.0}{m}/\qty{12.3}{s} = \qty{8.13}{m/s}$ (tres
cifras, fijadas por el tiempo).
\emph{3.} Cinco \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas} afirmarían conocer el área con
precisión de $\qty{0.01}{cm^2}$, una precisión que unos datos al
milímetro nunca dieron --- un decimal falsificado sigue siendo una
falsificación.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:12}
\emph{1.} Radio $r = \qty{0.55}{m}$, área
$A = \pi r^{2} = \pi \times 0.55^{2} \approx \qty{0.95}{m^2}$. Volumen
$V = \qty{1.0}{mm^3} = \qty{1.0e-9}{m^3}$, y una película es (área)
$\times$ (grosor), de modo que
\[
e = \frac{V}{A} = \frac{\qty{1.0e-9}{m^3}}{\qty{0.95}{m^2}}
= \qty{1.1e-9}{m} .
\]
\emph{2.} Una molécula es del orden de $\qty{e-9}{m}$ --- un peldaño por
encima de los $\qty{e-10}{m}$ del átomo, lo que concuerda con que una
molécula sea un pequeño racimo de átomos. La cucharada de aceite de
Franklin calmó media hectárea de estanque y, sin que él lo supiera,
midió el nanómetro.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:13}
Hipótesis: unos $70$ latidos por minuto en reposo y una vida de $80$
años. Por día: $70 \times 60 \times 24 \approx \num{1e5}$ latidos. Por
año: $\num{1e5} \times 365 \approx \num{4e7}$. Por vida:
$\num{4e7} \times 80 \approx \num{3e9}$ --- \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud} $10^{9}$,
unos pocos miles de millones de latidos. Cuatro \omterm{def:g10:orders-of-magnitude:sigfig}{cifras significativas}
fingirían conocer la frecuencia cardíaca minuto a minuto durante ochenta
años: los datos de partida son estimaciones a un \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud}, y
ningún resultado puede superar a sus datos
(\cref{met:g10:orders-of-magnitude:sigfig}).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:14}
Tras $n$ pliegues el grosor es $2^{n} \times \qty{1.0e-4}{m}$.
\emph{1.} Hace falta $2^{n} \geq 8848/\num{1.0e-4} = \num{8.8e7}$. Como
$2^{26} \approx \num{6.7e7}$ se queda corto y
$2^{27} \approx \num{1.3e8}$ basta: $27$ pliegues (montón
$\approx \qty{13}{km}$).
\emph{2.} Hace falta $2^{n} \geq \num{3.84e12}$: $2^{41} \approx
\num{2.2e12}$ no llega y $2^{42} \approx \num{4.4e12}$ sí ---
$42$ pliegues, montón de $\qty{4.4e8}{m}$, holgadamente más allá de la
Luna.
\emph{3.} Solo el experimento: cada pliegue reduce el área a la mitad y,
tras siete, el montón es más grueso que ancho. La aritmética de la
duplicación queda intacta --- el crecimiento exponencial simplemente
adelanta a lo que el papel (y la intuición) pueden seguir.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:15}
\emph{1.} Luna: $\qty{0.03}{m}/\qty{3.84e8}{m} = \num{7.8e-11}$. Mesa:
$\qty{0.001}{m}/\qty{1.000}{m} = \num{1.0e-3}$. La medida lunar es más
precisa en un factor
$\num{1.0e-3}/\num{7.8e-11} \approx \num{1.3e7}$ --- más de diez
millones.
\emph{2.} Igualar $\num{7.8e-11}$ en una mesa de un \omterm{def:g10:orders-of-magnitude:unit}{metro} exige
$\Delta x = \qty{7.8e-11}{m}$, menos que un átomo
($\approx \qty{e-10}{m}$): el borde de la mesa ni siquiera está
\emph{definido} con esa nitidez. En términos relativos, la distancia
Tierra--Luna por láser es una de las medidas más finas que posee nuestra
especie.
\end{solution}

\begin{solution}{pb:g10:orders-of-magnitude:1}
\textbf{1.} Grano: $\qty{0.5}{mm} = \qty{5e-4}{m}$, en la escala mili;
átomo: $\qty{2.5e-10}{m} = \qty{0.25}{nm}$, en la escala nano.

\textbf{2.} $\qty{5e-4}{m}/\qty{2.5e-10}{m} = \num{2.0e6}$: dos millones
de átomos de lado a lado.

\textbf{3.} Un cubo de $2 \times 10^{6}$ átomos de arista contiene
$(2\times10^{6})^{3} = 8 \times 10^{18}$ átomos: \omterm{def:g10:orders-of-magnitude:oom}{orden de magnitud}
$10^{19}$.

\textbf{4.} $V = (\qty{5e-4}{m})^{3} = \qty{1.25e-10}{m^3}$, de modo que
$m = \rho V = \qty{2500}{kg/m^3} \times \qty{1.25e-10}{m^3}
= \qty{3.1e-7}{kg} = \qty{0.31}{mg}$.

\textbf{5.} Masa por átomo:
$\qty{3.1e-7}{kg}/(8\times10^{18}) = \qty{3.9e-26}{kg}$ --- de lleno en
el intervalo de ``unos pocos $10^{-26}\,\unit{kg}$''. El ingenuo cubo de
átomos reproduce las masas atómicas reales: el modelo supera la
comprobación (\cref{met:g10:orders-of-magnitude:sanity}) con nota.

\textbf{6.} $s = \qty{5e-4}{m}/\qty{1.39e9}{m} = \num{3.6e-13}$.

\textbf{7.} Tierra de la maqueta: diámetro
$\qty{1.27e7}{m} \times \num{3.6e-13} = \qty{4.6e-6}{m}$ --- el tamaño
de una bacteria --- a una distancia
$\qty{1.496e11}{m} \times \num{3.6e-13} = \qty{5.4}{cm}$ del grano: la
Tierra es un microbio a cinco centímetros de un grano de arena.

\textbf{8.} $\qty{4.5e12}{m} \times \num{3.6e-13} = \qty{1.6}{m}$: el
sistema planetario entero orbita dentro de una mesa de comedor.

\textbf{9.} $\qty{4.0e16}{m} \times \num{3.6e-13} = \qty{1.4e4}{m}
= \qty{14}{km}$.

\textbf{10.} La estrella más cercana a nuestro Sistema Solar de sobremesa
es otro grano de arena a catorce kilómetros, y entre medias
prácticamente nada: la galaxia es abrumadoramente espacio vacío, y por
eso las colisiones entre estrellas son casi inauditas.

\textbf{11.} $t = \qty{5e-4}{m}/\qty{3.00e8}{m/s} = \qty{1.7e-12}{s}$
--- menos de dos picosegundos.

\textbf{12.} Luna: $\qty{3.84e8}{m}/\qty{3.00e8}{m/s} =
\qty{1.28}{s}$. Sol: $\qty{1.496e11}{m}/\qty{3.00e8}{m/s} =
\qty{499}{s} \approx \qty{8.3}{min}$: la luz del Sol tiene siempre ocho
minutos de antigüedad.

\textbf{13.} Un año son $365.25 \times 86400 = \qty{3.156e7}{s}$, así
que $\qty{1}{ly} = \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}$; entonces
$\qty{4.0e16}{m}/\qty{9.47e15}{m} = 4.2$: la distancia de los datos es,
en efecto, de unos $\qty{4.2}{ly}$.

\textbf{14.} $\qty{2.4e22}{m}/\qty{9.47e15}{m} = \num{2.5e6}$: dos
millones y medio de \omterm{def:g10:orders-of-magnitude:lightyear}{años luz}. Esa luz salió de Andrómeda hace unos $2.5$
millones de años, cuando los primeros miembros del género \emph{Homo}
aprendían a tallar piedras --- y ningún \emph{Homo sapiens} existía aún
en ninguna parte.

\textbf{15.} $\qty{3.00e8}{m/s} \times \qty{1e-9}{s} = \qty{0.30}{m}$.
En un tercio de nanosegundo una señal recorre a lo sumo unos
$\qty{10}{cm}$, de modo que dos partes de la máquina que deban hablarse
dentro de un mismo ciclo de reloj tienen que estar a menos de diez
centímetros una de otra: la velocidad pone techo al tamaño.

\textbf{16.} Átomo medio en el agua:
$\qty{3.0e-26}{kg}/3 = \qty{1.0e-26}{kg}$. Número de átomos:
$\qty{70}{kg}/\qty{1.0e-26}{kg} = \num{7e27}$.

\textbf{17.} Frente al grano:
$\num{7e27}/\num{8e18} \approx 10^{9}$ --- contienes los átomos de mil
millones de granos. Frente a las estrellas:
$\num{7e27}/\num{2e23} = \num{3.5e4}$ --- tus átomos superan a todas las
estrellas del universo observable en treinta y cinco mil a uno.

\textbf{18.} $\text{(tú)}/\text{(átomo)} =
\qty{1.7}{m}/\qty{2.5e-10}{m} \approx \num{7e9} \approx 10^{10}$,
mientras que $\text{(universo)}/\text{(tú)} =
\qty{8.8e26}{m}/\qty{1.7}{m} \approx \num{5e26}$. En la escalera estás
unos diez peldaños por encima del átomo y casi veintisiete por debajo
del universo: en términos multiplicativos, el ser humano vive muy cerca
de sus átomos.

\textbf{19.} La condición da
$x^{2} = 10^{-15} \times 10^{26.9} = 10^{11.9}$, de donde
$x = 10^{5.95} \approx \qty{9e5}{m} \approx \qty{900}{km}$ --- casi
exactamente el diámetro de Ceres, el mayor asteroide
($\approx \qty{940}{km}$). A media escalera entre un núcleo y el cosmos
se sienta un modesto planeta enano.

\textbf{20.} $\qty{8.8e26}{m} \times \num{3.6e-13} = \qty{3.2e14}{m}$,
es decir $\qty{3.2e14}{m}/\qty{1.496e11}{m} \approx \num{2.1e3}$
\omterm{def:g10:orders-of-magnitude:lightyear}{unidades astronómicas}. El remate: encoge el cosmos hasta que el Sol sea
un grano de arena y el universo observable seguirá abarcando dos mil
veces la distancia real Tierra--Sol --- ninguna maqueta reduce el
universo a tamaño humano; solo las potencias de diez pueden con él.
\end{solution}
