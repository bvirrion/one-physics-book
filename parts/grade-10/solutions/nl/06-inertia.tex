\section*{Hoofdstuk \ref{ch:g10:inertia} --- Krachten en het traagheidsbeginsel}

\begin{solution}{exo:g10:inertia:1}
(a) \omterm{def:g10:universal-gravitation:weight}{Gewicht} (op afstand), \omterm{def:g10:inertia:inventory}{normaalkracht} van de tafel (contact).
(b) \omterm{def:g10:universal-gravitation:weight}{Gewicht} (op afstand), \omterm{def:g10:inertia:inventory}{spankracht} van de draad (contact).
(c) \omterm{def:g10:universal-gravitation:weight}{Gewicht} (op afstand), \omterm{def:g10:inertia:inventory}{normaalkracht} van het ijs (contact); de
\omterm{def:g10:inertia:inventory}{wrijving} is verwaarloosbaar. (d) Alleen het \omterm{def:g10:universal-gravitation:weight}{gewicht} --- niets raakt de
appel aan.
\end{solution}

\begin{solution}{exo:g10:inertia:2}
Boek: $P = 0.250 \times 9.81 \approx \qty{2.45}{N}$, een pijl van
ongeveer \qty{2.5}{cm}. Persoon: $P = 60 \times 9.81 \approx
\qty{589}{N}$ --- een pijl van \qty{5.89}{m}. Kies een schaal waarop de
grootste \omterm{def:g10:inertia:force}{kracht} op de bladzijde past.
\end{solution}

\begin{solution}{exo:g10:inertia:3}
\omterm{def:g10:inertia:force}{Aangrijpingspunt} $A$; richting horizontaal, naar links wijzend; grootte
$3.5 \times 2 = \qty{7.0}{N}$.
\end{solution}

\begin{solution}{exo:g10:inertia:4}
\omterm{def:g10:universal-gravitation:weight}{Gewicht} $\vect P$: verticaal, omlaag,
$P = 1.2 \times 9.81 \approx \qty{11.8}{N}$. \omterm{def:g10:inertia:inventory}{Normaalkracht} $\vect R$ van
de plank: dezelfde lijn, omhoog, $R = \qty{11.8}{N}$ (het woordenboek is
in rust, dus zijn de twee in \omterm{def:g10:inertia:compensate}{evenwicht}).
\end{solution}

\begin{solution}{exo:g10:inertia:5}
(a) Onjuist: het is de \omterm{def:g10:inertia:inventory}{wrijving} die dingen stopt; zonder enige \omterm{def:g10:inertia:force}{kracht}
houdt een lichaam zijn \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging}. (b) Juist (het
traagheidsbeginsel). (c) Onjuist: er werken \omterm{def:g10:inertia:force}{krachten} die \emph{in
\omterm{def:g10:inertia:compensate}{evenwicht}} zijn (bijvoorbeeld \omterm{def:g10:universal-gravitation:weight}{gewicht} en \omterm{def:g10:inertia:inventory}{normaalkracht}). (d) Juist ---
de contrapositie van het beginsel.
\end{solution}

\begin{solution}{exo:g10:inertia:6}
\emph{1.} $v = 24/3.0 = \qty{8.0}{m/s}$. \omterm{def:g10:inertia:force}{Krachten}: \omterm{def:g10:universal-gravitation:weight}{gewicht} en
\omterm{def:g10:inertia:inventory}{normaalkracht}; ze zijn in \omterm{def:g10:inertia:compensate}{evenwicht} (\omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging},
verwaarloosbare \omterm{def:g10:inertia:inventory}{wrijving}). \emph{2.} De \omterm{def:g10:inertia:inventory}{wrijving} van het beton is niet
langer verwaarloosbaar: de inventaris krijgt er een niet-opgeheven
achterwaartse \omterm{def:g10:inertia:force}{kracht} bij, en inderdaad verandert de snelheid --- de puck
vertraagt en stopt.
\end{solution}

\begin{solution}{exo:g10:inertia:7}
\omterm{def:g10:universal-gravitation:weight}{Gewicht} $P = 70 \times 9.81 \approx \qty{687}{N}$ omlaag; \omterm{def:g10:inertia:force}{kracht} van de
vloer omhoog. \omterm{def:g10:inertia:uniform}{Eenparige rechtlijnige beweging}, dus zijn ze in \omterm{def:g10:inertia:compensate}{evenwicht}:
de vloer duwt met \qty{687}{N}. Bij stilstand verandert er niets ---
rust en eenparige beweging geven hetzelfde \omterm{def:g10:inertia:diagram}{krachtenschema}.
\end{solution}

\begin{solution}{exo:g10:inertia:8}
\emph{1.} In rust houdt de \omterm{def:g10:inertia:inventory}{spankracht} het \omterm{def:g10:universal-gravitation:weight}{gewicht} in \omterm{def:g10:inertia:compensate}{evenwicht}:
$T = 1.8 \times 9.81 \approx \qty{17.7}{N}$. \emph{2.} Elke verticale
draad draagt de helft: $T = \qty{8.8}{N}$.
\end{solution}

\begin{solution}{exo:g10:inertia:9}
Bij het remmen houdt het lichaam van de passagier zijn snelheid
(traagheid) terwijl de bus de zijne verliest, dus beweegt de passagier
naar voren \emph{ten opzichte van de bus}. In de bocht neigt het lichaam
ernaar rechtdoor te gaan terwijl de bus eronder wegdraait, zodat het naar
de buitenkant van de bocht schuift. In beide gevallen is de ``\omterm{def:g10:inertia:force}{kracht}''
verzonnen; wat ontbreekt is een echte \omterm{def:g10:inertia:force}{kracht} (grip, handgreep) om de
snelheid van de passagier met die van de bus mee te veranderen.
\end{solution}

\begin{solution}{exo:g10:inertia:10}
\omterm{def:g10:inertia:uniform}{Eenparige rechtlijnige beweging}, dus zijn \omterm{def:g10:universal-gravitation:weight}{gewicht} en luchtweerstand in
\omterm{def:g10:inertia:compensate}{evenwicht}: de weerstand is $P = 85 \times 9.81 \approx \qty{834}{N}$,
verticaal en omhoog gericht.
\end{solution}

\begin{solution}{exo:g10:inertia:11}
\emph{1.} \omterm{def:g10:inertia:uniform}{Eenparige rechtlijnige beweging}: de horizontale \omterm{def:g10:inertia:force}{krachten} zijn
in \omterm{def:g10:inertia:compensate}{evenwicht}, dus is de weerstand \qty{300}{N}, tegengesteld aan het
touw. \emph{2.} Het \omterm{def:g10:universal-gravitation:weight}{gewicht} van de skiër en de opwaartse duw van het
water op de ski's.
\end{solution}

\begin{solution}{exo:g10:inertia:12}
\emph{1.} $P = 2.4 \times 9.81 \approx \qty{23.5}{N}$; de verticale
componenten dragen het \omterm{def:g10:universal-gravitation:weight}{gewicht}: $2T\cos 40^\circ = P$, dus
$T = 23.5/(2 \times 0.766) \approx \qty{15.4}{N}$. \emph{2.}
$T = P/(2\cos\theta)$ groeit onbegrensd wanneer $\theta \to 90^\circ$,
want $\cos\theta \to 0$; precies horizontale draden hebben
\emph{helemaal} geen verticale component, dus zou niets het \omterm{def:g10:universal-gravitation:weight}{gewicht}
dragen.
\end{solution}

\begin{solution}{exo:g10:inertia:13}
Duw: \omterm{def:g10:universal-gravitation:weight}{gewicht}, \omterm{def:g10:inertia:inventory}{normaalkracht}, \omterm{def:g10:inertia:inventory}{stuwkracht} van de hand, kleine \omterm{def:g10:inertia:inventory}{wrijving}; de
snelheid neemt toe, dus zijn de \omterm{def:g10:inertia:force}{krachten} niet in \omterm{def:g10:inertia:compensate}{evenwicht}
(\cref{prop:g10:inertia:converse}). Glijden: \omterm{def:g10:universal-gravitation:weight}{gewicht}, \omterm{def:g10:inertia:inventory}{normaalkracht},
kleine \omterm{def:g10:inertia:inventory}{wrijving}; de steen vertraagt, dus opnieuw geen \omterm{def:g10:inertia:compensate}{evenwicht} --- de
\omterm{def:g10:inertia:inventory}{wrijving} is wat overblijft (op ideaal ijs zou hij eeuwig doorglijden).
Rust: alleen \omterm{def:g10:universal-gravitation:weight}{gewicht} en \omterm{def:g10:inertia:inventory}{normaalkracht}, in \omterm{def:g10:inertia:compensate}{evenwicht}
(\cref{thm:g10:inertia:principle}).
\end{solution}

\begin{solution}{exo:g10:inertia:14}
\emph{1.} Volgens het traagheidsbeginsel vergt het behouden van een
constante snelheid helemaal geen \omterm{def:g10:inertia:force}{kracht} --- en in de interstellaire
ruimte werkt er vrijwel geen. Aristoteles zou voorspellen dat de sonde
stopt zodra de ``beweger'' ermee ophoudt. \emph{2.} Eén jaar is
$\approx \qty{3.156e7}{s}$, dus
$d = 17 \times \num{3.156e7} \approx \qty{5.4e8}{km} \approx
\qty{3.6}{au}$. \emph{3.} Snelheid is een vector: haar richting
veranderen verandert de snelheid, en dat vergt een niet-opgeheven
\omterm{def:g10:inertia:force}{kracht}.
\end{solution}

\begin{solution}{exo:g10:inertia:15}
In rust zijn de drie \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht}:
$\vect F_3 = -(\vect F_1 + \vect F_2)$, dus $\vect F_3\,(-2, -4)$, met
grootte $\sqrt{4 + 16} = \sqrt{20} \approx \qty{4.5}{N}$, gericht
tegengesteld aan de resultante van de eerste twee (linksonder op het
rooster).
\end{solution}

\begin{solution}{pb:g10:inertia:1}
\textbf{1.} Een \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging} met \qty{0.75}{m/s}.
\omterm{def:g10:inertia:force}{Krachten}: het \omterm{def:g10:universal-gravitation:weight}{gewicht} en de \omterm{def:g10:inertia:inventory}{normaalkracht} van de band (er is geen
\omterm{def:g10:inertia:inventory}{wrijving} nodig: de koffer schuift niet over de band).
\textbf{2.} Ja --- \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging}
(\cref{thm:g10:inertia:principle}). $P = 23 \times 9.81 \approx
\qty{226}{N}$, dus $R = \qty{226}{N}$.
\textbf{3.} De schema's zijn identiek. Rust en \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige
beweging} zijn een en dezelfde mechanische toestand: de traagheid maakt
er geen onderscheid tussen.
\textbf{4.} De koffer houdt zijn \qty{0.75}{m/s} terwijl de band stopt:
hij schuift door of kantelt naar voren tot de \omterm{def:g10:inertia:inventory}{wrijving} zijn beweging
opslokt. Niets heeft hem geduwd --- hij hield alleen de snelheid die hij
had.
\textbf{5.} Snelheden in dezelfde richting tellen op:
$0.75 + 1.2 = \qty{1.95}{m/s}$. Haar snelheid is constant, dus zijn de
\omterm{def:g10:inertia:force}{krachten} op haar in \omterm{def:g10:inertia:compensate}{evenwicht}.
\textbf{6.} $P = 0.160 \times 9.81 \approx \qty{1.57}{N}$ omlaag,
$R = \qty{1.57}{N}$ omhoog, en \emph{geen} horizontale \omterm{def:g10:inertia:force}{kracht}.
\textbf{7.} De puck houdt \qty{8.0}{m/s} aan zonder ook maar één
beweger. Alledaagse oppervlakken verbergen \omterm{def:g10:inertia:inventory}{wrijving} --- een
niet-opgeheven achterwaartse \omterm{def:g10:inertia:force}{kracht} --- zodat beweging vanzelf lijkt te
sterven.
\textbf{8.} Haar snelheid verandert, dus zijn de \omterm{def:g10:inertia:force}{krachten} niet langer in
\omterm{def:g10:inertia:compensate}{evenwicht} (\cref{prop:g10:inertia:converse}); de schuldige is de
\omterm{def:g10:inertia:inventory}{wrijving} van het ruwe ijs.
\textbf{9.} Nee: snelheid is een \emph{vector}, en haar richting
verandert, dus zijn de \omterm{def:g10:inertia:force}{krachten} niet in \omterm{def:g10:inertia:compensate}{evenwicht} --- het ijs duwt
zijwaarts tegen haar schaatsen.
\textbf{10.} (a)--(iii), (b)--(i), (c)--(ii).
\textbf{11.} \omterm{def:g10:universal-gravitation:weight}{Gewicht} $P = 80 \times 9.81 \approx \qty{785}{N}$,
luchtweerstand nul: niets houdt het \omterm{def:g10:universal-gravitation:weight}{gewicht} in \omterm{def:g10:inertia:compensate}{evenwicht}, dus moet de
snelheid veranderen --- de springer versnelt omlaag.
\textbf{12.} Zolang de weerstand kleiner is dan het \omterm{def:g10:universal-gravitation:weight}{gewicht}, zijn de
\omterm{def:g10:inertia:force}{krachten} nog steeds niet in \omterm{def:g10:inertia:compensate}{evenwicht}: de snelheid blijft groeien, en de
weerstand groeit mee.
\textbf{13.} Constante snelheid: \omterm{def:g10:universal-gravitation:weight}{gewicht} en weerstand zijn in \omterm{def:g10:inertia:compensate}{evenwicht},
dus is de weerstand \qty{785}{N}. De beweging is recht en eenparig ---
een vallend lichaam in precies dezelfde mechanische toestand als de
koffer op het rolpad: traagheid in vermomming.
\textbf{14.} De weerstand overtreft nu het \omterm{def:g10:universal-gravitation:weight}{gewicht}: de \omterm{def:g10:inertia:force}{krachten} zijn niet
in \omterm{def:g10:inertia:compensate}{evenwicht}, dus verandert de snelheid --- de springer daalt nog steeds,
maar vertraagt; naarmate de snelheid zakt, krimpt de weerstand terug naar
het \omterm{def:g10:universal-gravitation:weight}{gewicht}.
\textbf{15.} Opnieuw constant: weerstand $= P = \qty{785}{N}$ --- exact
dezelfde waarde als bij \qty{50}{m/s}.
\textbf{16.} De uiteindelijke weerstand is altijd gelijk aan het
\omterm{def:g10:universal-gravitation:weight}{gewicht}, met of zonder scherm. De parachute verandert de \emph{snelheid}
waarbij de weerstand \qty{785}{N} bereikt: \qty{5}{m/s} in plaats van
\qty{50}{m/s}.
\textbf{17.} ``\dots de \omterm{def:g10:inertia:force}{krachten} op het lichaam zijn in \omterm{def:g10:inertia:compensate}{evenwicht} (of er
werkt er geen enkele)'' --- de twee lezingen van
\cref{thm:g10:inertia:principle}. Rust is gewoon $\vect v = \vect 0$,
een \omterm{def:g10:inertia:uniform}{eenparige rechtlijnige beweging} als alle andere.
\textbf{18.} De ijsbaan: de puck glijdt door zonder beweger in zicht.
Galilei wijst naar het steeds gladdere oppervlak --- de \omterm{def:g10:inertia:inventory}{wrijving}, niet
de natuur, is wat de dingen stopt.
\textbf{19.} Er werken wél \omterm{def:g10:inertia:force}{krachten} --- een \omterm{def:g10:universal-gravitation:weight}{gewicht} van \qty{785}{N}
omlaag en een weerstand van \qty{785}{N} omhoog; ze zijn \emph{in
\omterm{def:g10:inertia:compensate}{evenwicht}}, en juist daarom is de snelheid constant.
\textbf{20.} Rolpad: \omterm{def:g10:universal-gravitation:weight}{gewicht} en \omterm{def:g10:inertia:inventory}{normaalkracht}. IJsbaan: \omterm{def:g10:universal-gravitation:weight}{gewicht} en
\omterm{def:g10:inertia:inventory}{normaalkracht}. Parachute: \omterm{def:g10:universal-gravitation:weight}{gewicht} en luchtweerstand. De ene wet: het
traagheidsbeginsel, in zijn directe en zijn omgekeerde lezing.
\end{solution}
