\section*{Hoofdstuk \ref{ch:g10:refraction} --- Breking van licht}

\begin{solution}{exo:g10:refraction:1}
Hoeken worden vanaf de \omterm{def:g10:refraction:angles}{normaal} gemeten: de \omterm{def:g10:refraction:angles}{invalshoek} is
$90^\circ - 25^\circ = 65^\circ$, dus de terugkaatsingshoek is ook
$65^\circ$. De invallende en de teruggekaatste bundel maken onderling een
hoek van $2 \times 65^\circ = 130^\circ$.
\end{solution}

\begin{solution}{exo:g10:refraction:2}
\emph{1.} $v = c/n$: in water
$v = \qty{3.00e8}{m/s}/1.33 = \qty{2.26e8}{m/s}$; in diamant
$v = \qty{3.00e8}{m/s}/2.42 = \qty{1.24e8}{m/s}$.

\emph{2.} $n = \dfrac{c}{v} = \dfrac{\qty{3.00e8}{m/s}}
{\qty{1.97e8}{m/s}} = 1.52$: glas.
\end{solution}

\begin{solution}{exo:g10:refraction:3}
$\sin i_2 = \dfrac{\sin 40^\circ}{1.50} = \dfrac{0.643}{1.50} = 0.429$,
dus $i_2 = 25.4^\circ$. Door de omkeerbaarheid van lichtwegen verlaat een
straal die binnen in het glas het oppervlak onder $25.4^\circ$ treft, het
glas onder precies $40^\circ$.
\end{solution}

\begin{solution}{exo:g10:refraction:4}
$\sin i_2 = \dfrac{\sin 55^\circ}{1.33} = \dfrac{0.819}{1.33} = 0.616$,
dus $i_2 = 38.0^\circ$. De straal buigt \emph{naar} de \omterm{def:g10:refraction:angles}{normaal} toe, want
water is het tragere midden ($n_2 > n_1$).
\end{solution}

\begin{solution}{exo:g10:refraction:5}
$n = \dfrac{\sin 45^\circ}{\sin 32^\circ} = \dfrac{0.707}{0.530}
= 1.33$: de vloeistof is water.
\end{solution}

\begin{solution}{exo:g10:refraction:6}
\emph{1.} De vijf verhoudingen zijn
\[
\frac{0.342}{0.233} = 1.47,\quad
\frac{0.500}{0.334} = 1.50,\quad
\frac{0.643}{0.431} = 1.49,\quad
\frac{0.766}{0.515} = 1.49,\quad
\frac{0.866}{0.581} = 1.49 :
\]
constant binnen de meetfout --- de wet van Snellius.

\emph{2.} $n \approx 1.49$: plexiglas.

\emph{3.} De wet van Snellius voorspelt $\sin i_2 = \sin i_1 / n$: sinus
tegen sinus moeten de punten op een rechte door de oorsprong liggen, en
dat controleer je met een liniaal in één oogopslag. Zet je hoek tegen
hoek uit, dan liggen de punten op een kromme; bij kleine hoeken lijkt die
kromme bedrieglijk veel op een rechte, en precies dat heeft Ptolemaeus om
de tuin geleid.
\end{solution}

\begin{solution}{exo:g10:refraction:7}
$\sin i_c = n_2/n_1$ met $n_2 = 1.00$:
water $\sin i_c = 1/1.33 = 0.752$, $i_c = 48.8^\circ$;
glas $\sin i_c = 1/1.50 = 0.667$, $i_c = 41.8^\circ$;
diamant $\sin i_c = 1/2.42 = 0.413$, $i_c = 24.4^\circ$.

Van lucht naar water is $\sin i_2 = \sin i_1/1.33 \leq 1/1.33 < 1$ bij
elke inval: er bestaat altijd een brekingshoek, dus gebeurt er nooit iets
bijzonders --- \omterm{def:g10:refraction:critical}{totale terugkaatsing} vergt $n_1 > n_2$.
\end{solution}

\begin{solution}{exo:g10:refraction:8}
\emph{1.} $d' = \dfrac{d}{n} = \dfrac{\qty{2.0}{m}}{1.33}
= \qty{1.5}{m}$.

\emph{2.} De vis lijkt op $\qty{40}{cm}/1.33 = \qty{30}{cm}$ te zitten:
het beeld ligt \emph{boven} de vis, dus moet de reiger \emph{onder} het
beeld toeslaan. Reigers, niet gehinderd door de wet van Snellius, leren
dat met vallen en opstaan.
\end{solution}

\begin{solution}{exo:g10:refraction:9}
\emph{1.} $\sin r = \dfrac{\sin 50^\circ}{1.50} = \dfrac{0.766}{1.50}
= 0.511$, dus $r = 30.7^\circ$.

\emph{2.} Van glas naar lucht onder $30.7^\circ$:
$\sin i_2 = 1.50 \times 0.511 = 0.766$, dus $i_2 = 50^\circ$ --- de
uittredehoek is gelijk aan de intredehoek.

\emph{3.} De twee \omterm{def:g10:refraction:refraction}{brekingen} heffen elkaar op: de straal vertrekt
\emph{evenwijdig} aan haar oorspronkelijke richting, alleen zijwaarts
verschoven. Daarom vervormt een ruit het tafereel niet, maar verschuift
ze het slechts (onmerkbaar).
\end{solution}

\begin{solution}{exo:g10:refraction:10}
$\sin 55^\circ = 0.819$. Rood:
$\sin i_2 = 0.819/1.61 = 0.509$, $i_2 = 30.6^\circ$. Blauw:
$\sin i_2 = 0.819/1.66 = 0.493$, $i_2 = 29.6^\circ$. De twee kleuren
liggen $1.0^\circ$ uit elkaar; blauw, met de grotere index, buigt sterker
af en komt het dichtst bij de \omterm{def:g10:refraction:angles}{normaal} terecht.
\end{solution}

\begin{solution}{exo:g10:refraction:11}
\emph{1.} $\sin i_c = \dfrac{1.46}{1.48} = 0.986$, dus
$i_c = 80.6^\circ$.

\emph{2.} Een straal onder $i_c$ met de \omterm{def:g10:refraction:angles}{normaal} maakt $90^\circ - i_c$
met de as, dus is zijn weg over een aslengte $L$ gelijk aan
$\dfrac{L}{\sin i_c}$:
\[
\qty{1000}{m} \times \frac{1.48}{1.46} = \qty{1013.7}{m},
\]
ongeveer \qty{14}{m} meer dan de axiale straal.
\end{solution}

\begin{solution}{exo:g10:refraction:12}
\emph{1.} Licht dat scherend van de horizon komt, arriveert onder een
$i_1$ dicht bij $90^\circ$; het breekt naar
$\sin i_2 = \sin 90^\circ / 1.33 = 0.752$, dus $i_2 = 48.8^\circ$ --- de
\omterm{def:g10:refraction:critical}{grenshoek}, zoals de omkeerbaarheid eist. Stralen uit de hele hemel
bereiken het oog dus binnen een kegel met halve tophoek $48.8^\circ$.

\emph{2.} De rand van de schijf ligt daar waar die kegel het oppervlak
snijdt:
\[
r = d \tan i_c = \qty{2.0}{m} \times \tan 48.8^\circ
= \qty{2.0}{m} \times 1.14 = \qty{2.3}{m}.
\]

\emph{3.} Buiten de schijf treft licht van onderaf het oppervlak voorbij
de \omterm{def:g10:refraction:critical}{grenshoek} en wordt het volledig teruggekaatst: de duiker ziet een
zilverachtige spiegel waarin de bodem van het bad verschijnt.
\end{solution}

\begin{solution}{exo:g10:refraction:13}
\emph{1.} Rood: $\sin i_2 = 1.51 \times \sin 30^\circ = 0.755$, dus
$i_2 = 49.0^\circ$. Blauw: $\sin i_2 = 1.53 \times 0.500 = 0.765$, dus
$i_2 = 49.9^\circ$. De witte bundel waaiert uit over
$49.9^\circ - 49.0^\circ = 0.9^\circ$ --- een prismaspectrum uit één
enkele \omterm{def:g10:refraction:refraction}{breking}.

\emph{2.} Onder $45^\circ$: $\sin i_2 = 1.51 \times 0.707 = 1.07 > 1$
(en $1.53 \times 0.707 = 1.08$ voor blauw). Voor geen van beide kleuren
bestaat er een brekingshoek: de \omterm{def:g10:refraction:critical}{grenshoek} is
$\arcsin(1/1.51) = 41.5^\circ < 45^\circ$, dus kaatst het lange vlak de
hele bundel volledig terug.
\end{solution}

\begin{solution}{exo:g10:refraction:14}
\emph{1.} \omterm{def:g10:refraction:critical}{Grenshoek} glas--lucht: $\sin i_c = 1/1.50$, dus
$i_c = 41.8^\circ < 45^\circ$: de bundel wordt volledig teruggekaatst ---
$100\%$ van het licht, zonder laag die kan aanslaan. Een metalen spiegel
verliest bij elke kaats een paar procent, en een periscoop heeft er twee
nodig.

\emph{2.} Glas--water: $\sin i_c = \dfrac{1.33}{1.50} = 0.887$, dus
$i_c = 62.5^\circ$. Nu geldt $45^\circ < i_c$: de inval blijft onder de
\omterm{def:g10:refraction:critical}{grenshoek}, het meeste licht breekt het water in, en de spiegel --- en dus
het beeld --- begeeft het.

\emph{3.} $\sin i_2 = \dfrac{1.50}{1.33} \times \sin 45^\circ
= 1.128 \times 0.707 = 0.798$, dus $i_2 = 52.9^\circ$ het water in.
\end{solution}

\begin{solution}{exo:g10:refraction:15}
\emph{1.} $v = \dfrac{\qty{3.00e8}{m/s}}{1.47} = \qty{2.04e8}{m/s}$, dus
\[
t = \frac{\qty{6.2e6}{m}}{\qty{2.04e8}{m/s}} = \qty{30.4}{ms} \approx
\qty{30}{ms}.
\]

\emph{2.} Radio: $t = \qty{6.2e6}{m}/\qty{3.00e8}{m/s} =
\qty{20.7}{ms}$. Het glas kost $30.4 - 20.7 = \qty{9.7}{ms}$ per
richting, ongeveer \qty{19}{ms} heen en terug.

\emph{3.} De index verlagen --- vezels met een holle \omterm{def:g10:refraction:fiber}{kern} geleiden licht
door lucht ($n \approx 1.00$), en straalverbindingen door de atmosfeer
doen hetzelfde --- of de weg verkorten: een kabel die dichter bij de
grootcirkelroute tussen de twee steden ligt. Beide worden werkelijk
verkocht, tegen prijzen per milliseconde.
\end{solution}

\begin{solution}{pb:g10:refraction:1}
\textbf{1.} $v_{\text{water}} = \qty{3.00e8}{m/s}/1.33 =
\qty{2.26e8}{m/s}$; $v_{\text{diamant}} = \qty{3.00e8}{m/s}/2.42 =
\qty{1.24e8}{m/s}$ --- diamant remt het licht af met de factor
$n = 2.42$.

\textbf{2.} $\sin i_2 = \sin 40^\circ / 1.33 = 0.643/1.33 = 0.483$, dus
$i_2 = 28.9^\circ$.

\textbf{3.} $\sin i_2 = 1.33 \times \sin 28.9^\circ = 1.33 \times
0.483 = 0.643$, dus $i_2 = 40^\circ$: het licht loopt zijn weg terug ---
de omkeerbaarheid van lichtstralen.

\textbf{4.} $\sin i_c = 1/1.33 = 0.752$, dus $i_c = 48.8^\circ$. Onder
$60^\circ > i_c$ zou de wet van Snellius $\sin i_2 = 1.33 \sin 60^\circ
= 1.15 > 1$ vergen: er is geen \omterm{def:g10:refraction:refraction}{gebroken straal}; het oppervlak kaatst het
licht volledig terug naar beneden.

\textbf{5.} $v = \qty{3.00e8}{m/s}/1.0003 = \qty{2.999e8}{m/s}$: licht
is in lucht $0.03\%$ trager dan in vacuüm, ver onder de nauwkeurigheid
van onze hoekmetingen, dus is $n_{\text{lucht}} = 1.00$ onschadelijk.

\textbf{6.} $d' = \qty{3.0}{m}/1.33 = \qty{2.3}{m}$: het bad verbergt
\qty{70}{cm} van zichzelf.

\textbf{7.} Twee stralen die het muntstuk verlaten, breken aan het
oppervlak en buigen \emph{van} de \omterm{def:g10:refraction:angles}{normaal} \emph{af}, zodat ze boven het
water sterker uiteenlopen dan eronder. Het oog verlengt de twee
uittredende stralen rechtlijnig naar achteren; die snijden elkaar
\emph{boven} het muntstuk, en op dat snijpunt plaatst het oog het beeld.

\textbf{8.} $r = d \tan i_c = \qty{3.0}{m} \times \tan 48.8^\circ =
\qty{3.0}{m} \times 1.14 = \qty{3.4}{m}$: een schijf van ongeveer
\qty{6.9}{m} breed bevat de hele hemel.

\textbf{9.} Onder $30^\circ$: $\sin i_2 = 1.33 \times 0.500 = 0.665$, de
straal treedt uit onder $41.7^\circ$. Onder $45^\circ$: $\sin i_2 =
1.33 \times 0.707 = 0.940$, hij treedt uit onder $70.1^\circ$, bijna
scherend. Onder $50^\circ$: $1.33 \times 0.766 = 1.02 > 1$ --- \omterm{def:g10:refraction:critical}{totale
terugkaatsing}, de straal duikt het bad weer in.

\textbf{10.} Het beeld ligt boven de echte vis
(\cref{prop:g10:refraction:depth}), dus moet hij \emph{eronder} mikken.
En inderdaad: de vis ziet de visser ook verplaatst, samengeperst naar de
verticaal binnen het venster van Snellius --- elk ziet de ander waar de
ander niet is.

\textbf{11.} Diamant: $\sin i_c = 1/2.42 = 0.413$, dus
$i_c = 24.4^\circ$, tegenover $41.1^\circ$ voor glas: de
ontsnappingskegel van diamant is nog geen half zo wijd.

\textbf{12.} In diamant geldt $30^\circ > 24.4^\circ$: \omterm{def:g10:refraction:critical}{totale
terugkaatsing}, de straal blijft in de steen. In glas geldt
$30^\circ < 41.1^\circ$: de straal lekt er aan de achterkant uit. De
facetten van een briljantslijpsel staan zo, dat binnenkomend licht ze
vrijwel altijd voorbij $24.4^\circ$ treft en blijft kaatsen tot het naar
boven, naar het oog, ontsnapt --- een glazen imitatie, met haar wijde
ontsnappingskegel, laat het licht er gewoon doorheen vallen.

\textbf{13.} $\sin 45^\circ = 0.707$. Rood: $0.707/2.41 = 0.293$, dus
$i_2 = 17.1^\circ$. Blauw: $0.707/2.45 = 0.289$, dus $i_2 = 16.8^\circ$.
Splitsing bij binnenkomst: ongeveer $0.3^\circ$.

\textbf{14.} $\sin 17.0^\circ = 0.292$. Rood: $\sin i_2 = 2.41 \times
0.292 = 0.705$, dus $i_2 = 44.8^\circ$. Blauw: $\sin i_2 = 2.45 \times
0.292 = 0.716$, dus $i_2 = 45.8^\circ$. De waaier is nu $1.0^\circ$
breed: de uittredende \omterm{def:g10:refraction:refraction}{breking}, die vlak bij de \omterm{def:g10:refraction:critical}{grenshoek} werkt, heeft de
splitsing van $0.3^\circ$ bij binnenkomst verdrievoudigd. Die uitgerekte
kleurenspreiding is het \emph{vuur} van de diamant.

\textbf{15.} Diamant--water: $\sin i_c = 1.33/2.42 = 0.550$, dus
$i_c = 33.3^\circ$. De straal van $30^\circ$, gevangen toen het facet aan
lucht grensde, ontsnapt nu ($30^\circ < 33.3^\circ$) in het waterlaagje:
elke druppel, vingerafdruk of vetlaag verhoogt de index buiten de steen,
verwijdt de ontsnappingskegel en laat de schittering weglopen --- vandaar
de doek van de juwelier.

\textbf{16.} $\sin i_c = \dfrac{1.46}{1.48} = 0.986$, dus
$i_c = 80.6^\circ$.

\textbf{17.} \omterm{def:g10:refraction:critical}{Totale terugkaatsing} gebeurt \emph{aan het oppervlak}, dus
moet dat oppervlak volmaakt blijven: op een kale draad verhoogt elk
stofje, elke kras, elke waterdruppel en elke vinger die het aanraakt
plaatselijk de index buiten het glas en laat er licht uit weglopen. De
\omterm{def:g10:refraction:fiber}{mantel} begraaft de spiegel binnen in schoon glas, waar niets erbij kan
--- en beschermt bovendien de haardunne \omterm{def:g10:refraction:fiber}{kern} mechanisch.

\textbf{18.} $v = \qty{3.00e8}{m/s}/1.48 = \qty{2.03e8}{m/s}$, dus
\[
t = \frac{\qty{6.0e6}{m}}{\qty{2.03e8}{m/s}} = \qty{29.6}{ms} \approx
\qty{30}{ms}.
\]

\textbf{19.} Wegfactor $1/\sin i_c = 1.48/1.46 = 1.0137$: in het ergste
geval $\qty{6000}{km} \times 0.0137 \approx \qty{82}{km}$ extra glas, wat
$\qty{8.2e4}{m} / \qty{2.03e8}{m/s} \approx \qty{0.4}{ms}$ kost --- de
zigzag is bijna gratis.

\textbf{20.} Ping: $2 \times (29.6 + 0.4) \approx \qty{60}{ms}$.
Doorvoer: $\dfrac{10^{13}}{2.4 \times 10^8} \approx 40\,000$ exemplaren
van dit boek per seconde. In één zin: door licht met \omterm{def:g10:refraction:critical}{totale
terugkaatsing} binnen in glas gevangen te houden, draagt de wet van
Snellius elke seconde een bibliotheek over de oceaan, zestig
milliseconden heen en terug.
\end{solution}
