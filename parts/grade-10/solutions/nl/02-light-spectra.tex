\section*{Hoofdstuk \ref{ch:g10:light-spectra} --- Lichtspectra en de boodschap van het licht}

\begin{solution}{exo:g10:light-spectra:1}
$\qty{254}{nm} < \qty{400}{nm}$: \omterm{def:g10:light-spectra:wavelength}{ultraviolet}. $\qty{480}{nm}$:
zichtbaar, blauw. $\qty{589}{nm}$: zichtbaar, geeloranje.
$\qty{656}{nm}$: zichtbaar, rood. $\qty{1064}{nm} > \qty{800}{nm}$:
\omterm{def:g10:light-spectra:wavelength}{infrarood} --- en juist dat maakt een snijlaser dubbel gevaarlijk:
krachtig én onzichtbaar.
\end{solution}

\begin{solution}{exo:g10:light-spectra:2}
\emph{1.} Een doorlopende kleurenband, van rood tot violet: het \omterm{def:g10:light-spectra:white}{spectrum}
van zonlicht.

\emph{2.} Het violette uiteinde --- het prisma buigt korte \omterm{def:g10:light-spectra:wavelength}{golflengten}
het sterkst af.

\emph{3.} Eén groene vlek: wel afgebogen, maar niet uitgespreid tot een
band. Laserlicht is \omterm{def:g10:light-spectra:white}{monochromatisch} --- één enkele \omterm{def:g10:light-spectra:wavelength}{golflengte}, dus valt
er niets te sorteren.
\end{solution}

\begin{solution}{exo:g10:light-spectra:3}
(a) Continu (dichte hete gloeidraad). (b) Lijnenemissie (aangeslagen gas
onder lage druk). (c) Continu (dicht gesmolten metaal). (d)
Lijnenemissie (in wezen de ene lijn op $\qty{589}{nm}$). (e) Absorptie:
het continue \omterm{def:g10:light-spectra:white}{spectrum} van het hete oppervlak, doorkrast met de donkere
lijnen van de koelere zonneatmosfeer.
\end{solution}

\begin{solution}{exo:g10:light-spectra:4}
\emph{1.} De regendruppels: elke druppel ontvangt wit zonlicht, ontleedt
het en stuurt het op kleur gesorteerd terug.

\emph{2.} De Zon levert het witte licht; de regen levert miljoenen
piepkleine prisma's; de meetkunde klopt alleen voor licht dat de
druppels naar de waarnemer terugsturen, en daarom moet de Zon in de rug
van de waarnemer staan en de regen ervoor.

\emph{3.} De druppels voegen geen eigen licht toe --- ze sorteren alleen
wat ze ontvangen. Als gesorteerd zonlicht elke kleur laat zien, dan zat
elke kleur al in het zonlicht: zonlicht is \omterm{def:g10:light-spectra:white}{wit licht}.
\end{solution}

\begin{solution}{exo:g10:light-spectra:5}
\emph{1.} Dofrood $<$ feloranje $<$ bijna wit.

\emph{2.} Allebei: de gloed wordt \emph{feller} (effect 1) en de kleur
schuift van rood naar de blauwe kant (effect 2, de krimpende piek
$\lambda_{\max}$), zodat een steeds groter deel van de zichtbare band
sterk verlicht raakt --- en een volle band leest als wit.

\emph{3.} Een wit gloeiend lichaam straalt sterk over de hele zichtbare
band; een rood gloeiend lichaam haalt alleen het uiteinde met lange
\omterm{def:g10:light-spectra:wavelength}{golflengten}. Volgens de wet van Wien is het eerste het hetere.
\end{solution}

\begin{solution}{exo:g10:light-spectra:6}
\emph{1.}
\[
T = \frac{\qty{2.90e-3}{m.K}}{\qty{5.00e-7}{m}} = \qty{5800}{K}.
\]

\emph{2.} $\theta = 5800 - 273 = \qty{5527}{\degreeCelsius} \approx
\qty{5500}{\degreeCelsius}$.

\emph{3.} $\qty{400}{nm} < \qty{500}{nm} < \qty{800}{nm}$: de piek is
zichtbaar, in het groen.
\end{solution}

\begin{solution}{exo:g10:light-spectra:7}
\emph{1.} Kaars:
$\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{1800}{K} = \qty{1.61e-6}{m}
= \qty{1.6}{\micro m}$, in het \omterm{def:g10:light-spectra:wavelength}{infrarood}. Kookplaat:
$\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{500}{K} = \qty{5.8e-6}{m}
= \qty{5.8}{\micro m}$, ver \omterm{def:g10:light-spectra:wavelength}{infrarood}.

\emph{2.} Bij $\qty{500}{K}$ is vrijwel alle straling \omterm{def:g10:light-spectra:wavelength}{infrarood}: de huid
absorbeert die en voelt warmte, maar het oog ontvangt niets --- heet en
toch zwart.

\emph{3.} Een \omterm{def:g10:light-spectra:continuous}{continu spectrum} is breed: zelfs met een piek op
$\qty{1.6}{\micro m}$ houdt het \omterm{def:g10:light-spectra:white}{spectrum} van de kaars een zichtbare
staart aan de rood-gele kant --- zwak, maar genoeg voor het aan het
donker gewende oog.
\end{solution}

\begin{solution}{exo:g10:light-spectra:8}
Lamp A: de lijnen $434$, $486$ en $\qty{656}{nm}$ passen bij waterstof,
en de zwakke violette lijn maakt de catalogus compleet op
$\qty{410}{nm}$: waterstof.

Lamp B: de lijn ligt op $\qty{589}{nm}$, de enige sterke zichtbare lijn
van natrium. Helium wordt niet uitgesloten door die lijn zelf (helium
bezit inderdaad een lijn op het nabije $\qty{588}{nm}$) maar door de
\emph{ontbrekende} lijnen: helium zou ook $447$, $502$ en
$\qty{668}{nm}$ tonen, en geen ervan verschijnt. Een element wordt
geïdentificeerd door zijn volledige catalogus: lamp B bevat natrium.
\end{solution}

\begin{solution}{exo:g10:light-spectra:9}
\emph{1.} Een doorlopende regenboog met één dunne donkere lijn erin: het
\omterm{def:g10:light-spectra:absorption}{absorptiespectrum} van natrium.

\emph{2.} Op $\qty{589}{nm}$: een gas absorbeert precies de \omterm{def:g10:light-spectra:wavelength}{golflengten}
die het kan uitzenden (\cref{prop:g10:light-spectra:kirchhoff}), en de
sterke zichtbare lijn van natrium ligt op $\qty{589}{nm}$.

\emph{3.} Het \omterm{def:g10:light-spectra:emission}{emissiespectrum} van natrium: één heldere geeloranje lijn
op $\qty{589}{nm}$ op een zwarte achtergrond --- dezelfde \omterm{def:g10:light-spectra:wavelength}{golflengte},
met licht en donker verwisseld tussen de twee spectra.
\end{solution}

\begin{solution}{exo:g10:light-spectra:10}
\emph{1.} $410$, $434$, $486$, $\qty{656}{nm}$: de volledige catalogus
van waterstof. $517$ en $\qty{518}{nm}$: het magnesiumpaar. Beide
elementen zijn aanwezig.

\emph{2.} In de atmosfeer van de Zon, het koelere en ijlere gas boven het
gloeiende oppervlak. Het oppervlak stuurt een \omterm{def:g10:light-spectra:continuous}{continu spectrum} omhoog;
de atmosfeer haalt haar eigen \omterm{def:g10:light-spectra:wavelength}{golflengten} uit het passerende licht, zodat
de lijnen als ontbrekend licht verschijnen --- donker tegen de heldere
achtergrond.
\end{solution}

\begin{solution}{exo:g10:light-spectra:11}
\emph{1.} $T = 37 + 273 = \qty{310}{K}$, dus
\[
\lambda_{\max} = \frac{\qty{2.90e-3}{m.K}}{\qty{310}{K}}
= \qty{9.4e-6}{m} = \qty{9.4}{\micro m}.
\]

\emph{2.} Ver \omterm{def:g10:light-spectra:wavelength}{infrarood} --- meer dan tien keer de \omterm{def:g10:light-spectra:wavelength}{golflengte} van rood
licht.

\emph{3.} Bij $\qty{310}{K}$ is het zichtbare deel van de straling
volstrekt verwaarloosbaar: geen zichtbare gloed, hoe donker de kamer ook
is. Een warmtebeeldcamera draagt een sensor die is afgestemd op de
straling die werkelijk wordt uitgezonden, rond $\qty{9}{\micro m}$, en
ziet warme lichamen oplichten tegen koelere omgeving.
\end{solution}

\begin{solution}{exo:g10:light-spectra:12}
\emph{1.} $\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{2700}{K}
= \qty{1.07e-6}{m} \approx \qty{1.1}{\micro m}$: nabij \omterm{def:g10:light-spectra:wavelength}{infrarood}.

\emph{2.} Het \omterm{def:g10:light-spectra:white}{spectrum} piekt voorbij het zichtbare: het grootste deel van
het elektrisch vermogen keert terug als onzichtbaar \omterm{def:g10:light-spectra:wavelength}{infrarood} --- dus als
warmte --- en slechts een kleine zichtbare staart verlicht de kamer. Als
lamp verspilt de gloeilamp het meeste van haar vermogen; als kacheltje is
ze bijna volmaakt.

\emph{3.} $T = \qty{2.90e-3}{m.K} / \qty{5.50e-7}{m} \approx
\qty{5300}{K}$ --- ver boven het smeltpunt van wolfraam,
$\qty{3700}{K}$. Geen enkele vaste gloeidraad laat zich tot
daglichttemperaturen opdrijven, dus moesten daglichtachtige lampen het
gloeien helemaal opgeven (tl-buizen, leds) en zichtbaar licht maken
zonder een lichaam tot $\qty{5300}{K}$ te verhitten.
\end{solution}

\begin{solution}{exo:g10:light-spectra:13}
\emph{1.} $T = \qty{2.90e-3}{m.K} / \qty{4.20e-7}{m} \approx
\qty{6900}{K}$: heter dan de $\qty{5800}{K}$ van de Zon.

\emph{2.} Waterstof (de volledige reeks $410$, $434$, $486$,
$\qty{656}{nm}$ is aanwezig) en calcium (beide lijnen, $393$ en
$\qty{397}{nm}$).

\emph{3.} Nee. De natriumlijn op $\qty{589}{nm}$ ontbreekt, terwijl
overvloedig natrium in de atmosfeer die lijn noodzakelijk op de continue
achtergrond zou afdrukken. Als er al natrium is, dan alleen in sporen die
te zwak zijn om te meten.
\end{solution}

\begin{solution}{exo:g10:light-spectra:14}
\emph{1.} Waterstof verklaart $410$, $434$, $486$ en $\qty{656}{nm}$;
helium verklaart $447$, $502$, $588$ en $\qty{668}{nm}$. Elke
waargenomen lijn is verantwoord: het mengsel is waterstof en helium.

\emph{2.} Helium is een \emph{afdoende} verklaring omdat zijn drie andere
lijnen ($447$, $502$, $\qty{668}{nm}$) alle aanwezig zijn --- de lijn op
$\qty{588}{nm}$ is dan uit helium alleen al te verwachten, en niets in
het \omterm{def:g10:light-spectra:white}{spectrum} \emph{vereist} natrium. Om de kwestie te beslechten is geen
discussie nodig maar oplossend vermogen: een fijnere \omterm{def:g10:light-spectra:white}{spectroscoop}
scheidt de enkele heliumlijn op $\qty{587.6}{nm}$ van die van natrium,
dat er in werkelijkheid een dicht paar heeft op $589.0$ en
$\qty{589.6}{nm}$. Zie je daar één lijn, dan gaat natrium vrijuit; zie je
er drie, dan is het veroordeeld.
\end{solution}

\begin{solution}{exo:g10:light-spectra:15}
\emph{1.} Betelgeuze:
$T = \qty{2.90e-3}{m.K} / \qty{8.5e-7}{m} \approx \qty{3400}{K}$. Rigel:
$T = \qty{2.90e-3}{m.K} / \qty{2.4e-7}{m} \approx \qty{12000}{K}$.

\emph{2.} Omdat $\lambda_{\max} \times T$ voor allebei dezelfde constante
is, geldt $\dfrac{T_{\text{Rigel}}}{T_{\text{Betelgeuze}}}
= \dfrac{\qty{850}{nm}}{\qty{240}{nm}} \approx 3.5$.

\emph{3.} Betelgeuze piekt in het \omterm{def:g10:light-spectra:wavelength}{infrarood}: binnen de zichtbare band is
haar \omterm{def:g10:light-spectra:white}{spectrum} het sterkst aan de rode kant, dus de ster oogt rood. Rigel
piekt in het \omterm{def:g10:light-spectra:wavelength}{ultraviolet}: haar zichtbare band is het sterkst aan de
blauwe kant, vandaar blauwwit.

\emph{4.} Een \omterm{def:g10:light-spectra:continuous}{continu spectrum} bestrijkt een reusachtig
golflengtegebied. Zelfs met haar piek buiten de zichtbare band giet een
ster volop licht \emph{in} die band --- de piek bepaalt de tint, niet de
zichtbaarheid.
\end{solution}

\begin{solution}{pb:g10:light-spectra:1}
\textbf{1.} Van ongeveer $\qty{400}{nm}$ (violette uiteinde) tot ongeveer
$\qty{800}{nm}$ (rode uiteinde); $\qty{550}{nm}$ ligt ongeveer in het
midden, in het groen.

\textbf{2.} $\qty{410}{nm}$: violet; $\qty{434}{nm}$: violetblauw;
$\qty{486}{nm}$: blauwgroen; $\qty{656}{nm}$: rood.

\textbf{3.} Een monochromatische geeloranje gloed. De kleur van een auto
is het mengsel van \omterm{def:g10:light-spectra:wavelength}{golflengten} dat haar lak weerkaatst; onder een lamp
die \emph{alleen} $\qty{589}{nm}$ aanbiedt, kan elke lak niet anders dan
meer of minder van die ene \omterm{def:g10:light-spectra:wavelength}{golflengte} terugkaatsen --- alle auto's
verschijnen in tinten oranje en grijs.

\textbf{4.} Elk element zendt een vaste catalogus lijnen uit, en geen
twee elementen delen dezelfde catalogus
(\cref{prop:g10:light-spectra:fingerprint}): een volledige reeks lijnen
identificeert het element dus als een vingerafdruk. Maar een
\emph{gedeeltelijke} overeenkomst bewijst weinig --- verschillende
elementen kunnen bijna samenvallende losse lijnen bezitten (helium op
$\qty{588}{nm}$, natrium op $\qty{589}{nm}$) --- en daarom vraagt de
bewering ``aanwezig'' om alle sterke lijnen van de catalogus.

\textbf{5.} (a) Een doorlopende regenboog met een donkere lijn op
$\qty{589}{nm}$: het \omterm{def:g10:light-spectra:absorption}{absorptiespectrum}. (b) Eén heldere lijn op
$\qty{589}{nm}$ op zwart: het \omterm{def:g10:light-spectra:emission}{emissiespectrum} --- dezelfde \omterm{def:g10:light-spectra:wavelength}{golflengte},
met helder en donker verwisseld
(\cref{prop:g10:light-spectra:kirchhoff}).

\textbf{6.} Het \omterm{def:g10:light-spectra:white}{spectrum} wordt bij elke \omterm{def:g10:light-spectra:wavelength}{golflengte} feller, en zijn piek
$\lambda_{\max}$ schuift naar kortere \omterm{def:g10:light-spectra:wavelength}{golflengten}, volgens
$\lambda_{\max} \times T = \qty{2.90e-3}{m.K}$
(\cref{prop:g10:light-spectra:wien}).

\textbf{7.} $\lambda_{\max} = \qty{2.90e-3}{m.K} / \qty{3400}{K}
= \qty{8.5e-7}{m} = \qty{850}{nm}$: nabij \omterm{def:g10:light-spectra:wavelength}{infrarood}, net voorbij de rode
rand --- en daarom oogt halogeenlicht warmer van toon dan daglicht.

\textbf{8.} $T = \qty{2.90e-3}{m.K} / \qty{5.00e-7}{m} = \qty{5800}{K}$.

\textbf{9.} De piek is flauw, geen naald: de Zon zendt de hele zichtbare
band met vergelijkbare sterkte uit. Een volle band kleuren door elkaar
leest als wit --- de Zon oogt wit (geelachtig door onze atmosfeer heen),
nooit groen.

\textbf{10.} Koude pook: piek diep in het \omterm{def:g10:light-spectra:wavelength}{infrarood}, geen zichtbare
staart --- de straling wordt gevoeld, er is niets te zien. Heter: het
feller wordende \omterm{def:g10:light-spectra:white}{spectrum} duwt een eerste zichtbare staart voorbij
$\qty{800}{nm}$ --- alleen de rode kant licht op: dofrood. Nog heter: de
hele zichtbare band vult zich terwijl alles feller wordt: oranje, en dan
richting wit.

\textbf{11.} De continue achtergrond komt van het dichte, hete gloeiende
oppervlak; de donkere lijnen worden erop afgedrukt door de koelere,
ijlere atmosfeer erboven, die haar eigen \omterm{def:g10:light-spectra:wavelength}{golflengten} uit het passerende
licht wegneemt.

\textbf{12.} Waterstof ($410$, $434$, $486$, $\qty{656}{nm}$: volledig),
natrium ($\qty{589}{nm}$) en calcium ($393$ en $\qty{397}{nm}$: allebei)
--- alle drie zijn aanwezig in de atmosfeer van de Zon.

\textbf{13.} Dat de atmosfeer door waterstof wordt beheerst. Moderne
metingen bevestigen dat en zetten er een getal op: de Zon bestaat naar
massa voor ruwweg driekwart uit waterstof, en de rest is grotendeels
helium.

\textbf{14.} De lijn paste bij geen enkel bekend element; omdat de
catalogus van elk element vast en uniek is, moet een lijn die bij geen
enkele catalogus hoort, bij een element horen dat nog nooit in een
laboratorium is gezien. Er werd een nieuw element aangekondigd, helium
genoemd naar \emph{helios}, de Zon. Het werd in 1895 op Aarde
geïsoleerd --- de voorspelling bevestigd, zevenentwintig jaar later.

\textbf{15.} Tegen de schitterende continue achtergrond \emph{haalt} de
atmosfeer licht weg bij haar eigen \omterm{def:g10:light-spectra:wavelength}{golflengten}: donkere lijnen. Tijdens
de verduistering blokkeert de Maan de achtergrond; de ijle atmosfeer
staat dan alleen tegen een donkere hemel en het licht dat zij weer
uitzendt is alles wat er te zien valt: dezelfde \omterm{def:g10:light-spectra:wavelength}{golflengten}, nu helder.
Eén gas, één catalogus --- het teken hangt af van wat erachter staat.

\textbf{16.} $T = \qty{2.90e-3}{m.K} / \qty{2.90e-7}{m}
= \qty{10000}{K}$.

\textbf{17.} Blauwwit: binnen de zichtbare band is het \omterm{def:g10:light-spectra:white}{spectrum} het
sterkst aan de blauwe kant. En van onzichtbaarheid is geen sprake: het
continue \omterm{def:g10:light-spectra:white}{spectrum} overspoelt de hele zichtbare band op weg naar de
ultraviolette piek --- de piek legt de tint vast, niet de helderheid.

\textbf{18.} De donkere lijnen $410$, $434$, $486$, $\qty{656}{nm}$
vormen de volledige vingerafdruk van waterstof: waterstof beheerst de
atmosfeer van Wega.

\textbf{19.} $T_{\text{Wega}} / T_{\text{Zon}} = 10000 / 5800 \approx
1.7$: Wega is heter, dus blauwer (piek op $\qty{290}{nm}$ tegen
$\qty{500}{nm}$) en, volgens het eerste deel van
\cref{prop:g10:light-spectra:wien}, bij elke \omterm{def:g10:light-spectra:wavelength}{golflengte} feller per
\omterm{def:g10:orders-of-magnitude:unit}{eenheid} gloeiend oppervlak.

\textbf{20.} Identiteitsformulier van Wega ---
oppervlaktetemperatuur: ongeveer $\qty{10000}{K}$; overheersend element
van de atmosfeer: waterstof; schijnbare kleur: blauwwit. Alles afgelezen
uit één bundel licht, met een prisma, een catalogus van
laboratoriumlijnen en één deling. Comtes ``nooit'' werd in 1835
uitgesproken en verliep in 1859: het heeft vierentwintig jaar geduurd.
\end{solution}
