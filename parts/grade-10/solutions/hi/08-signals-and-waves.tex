\begin{solution}{exo:g10:signals-and-waves:1}

\end{solution}

\end{solution}

कानों को)। \qty{40}{kHz} और \qty{5}{MHz}: \omterm{def:g10:signals-and-waves:ultrasound}{पराश्रव्य}।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:4}

\end{solution}

\begin{solution}{exo:g10:signals-and-waves:5}
$d = 340 \times 6.0 = \qty{2040}{m} \approx \qty{2.0}{km}$। प्रकाश यह दूरी $2040/(3.00\times10^{8}) \approx \qty{7}{\micro s}$ में तय कर लेता है, जो \qty{6.0}{s} से दस लाख
गुना कम है: इसलिए चमक ही गिरने का क्षण बता देती है।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:6}
\omterm{def:g10:signals-and-waves:sound}{ध्वनि} नीचे \emph{और वापस} जाती है, यानी $2d$ तय करती है:
$d = \frac{1500 \times 0.60}{2} = \qty{450}{m}$। खाई पर: $t = 2d/v = 2 \times 1200/1500 = \qty{1.6}{s}$।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:7}
$d_1 = \frac{3.00\times10^{8} \times 2.4\times10^{-4}}{2} =
\qty{36.0}{km}$ और $d_2 = \qty{35.7}{km}$: विमान ने \qty{1.0}{s} में \qty{300}{m} तय किए, इसलिए $v = \qty{300}{m/s} \approx
\qty{1100}{km/h}$, और वह
पास आ रहा है।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:8}
$T = \qty{20}{mm}/\qty{25}{mm/s} = \qty{0.80}{s}$, इसलिए $f = \qty{1.25}{Hz}$: यानी $75$ धड़कन प्रति मिनट। $100$ धड़कन प्रति
मिनट के लिए $T = \qty{0.60}{s}$, यानी $25 \times 0.60 = \qty{15}{mm}$: क्षिप्रहृदयता \qty{15}{mm} से नीचे दिखने
लगती है।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:9}
$T = 1/200 = \qty{5.0}{ms}$; दो चक्र \qty{10}{ms} चलते हैं और $10$ खानों में फैलते हैं:
\omterm{def:g10:signals-and-waves:oscilloscope}{कालाधार} \qty{1}{ms} प्रति खाना। शिखर को $4$ खानों में समाना
चाहिए: \omterm{def:g10:signals-and-waves:oscilloscope}{लब्धि} कम से कम $3.0/4 = \qty{0.75}{V}$ प्रति खाना --- \qty{1}{V} प्रति खाना
लीजिए (शिखर $3.0$ खाने ऊपर)।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:10}
$d = v\,t/2$: अगली दीवार $\frac{1540 \times 4.0\times10^{-5}}{2} = \qty{3.1}{cm}$; पिछली दीवार $\frac{1540 \times 6.0\times10^{-5}}{2} = \qty{4.6}{cm}$; मोटाई लगभग \qty{1.5}{cm}।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:11}
\omterm{def:g10:signals-and-waves:sound}{ध्वनि}: $30/340 = \qty{0.088}{s}$। रेडियो: $3.0\times10^{6}/(3.00\times10^{8}) = \qty{0.010}{s}$। दूर बैठा रेडियो श्रोता ढोल पहले
सुनता है, लगभग \qty{78}{ms} पहले।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:12}
\qty{3.0}{ms} तक उत्सर्जन करते समय चमगादड़ प्रतिध्वनियों के प्रति बहरा
रहता है, यानी $d = \frac{340 \times 3.0\times10^{-3}}{2} =
\qty{0.51}{m}$ के भीतर की हर चीज़ के प्रति। \qty{1.7}{m} पर पतंगा:
$t = 2 \times 1.7/340 = \qty{10}{ms}$। पास आते-आते \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} और जल्दी लौटने लगती है; स्पंद को
उसके पहुँचने से पहले ख़त्म होना चाहिए, इसलिए उसे छोटा होना ही पड़ता है।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:13}
$d = \frac{340 \times 1.5}{2} = \qty{255}{m}$। \qty{155}{m} पर: $t = 2 \times 155/340 = \qty{0.91}{s}$। \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} और ताली तब घुल जाती हैं
जब $t < \qty{0.1}{s}$, यानी $d < \frac{340 \times 0.1}{2} = \qty{17}{m}$ के भीतर।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:14}
\emph{1.} $T = 3.0 \times \qty{5}{ms} = \qty{15}{ms}$, $f = 1/0.015 \approx \qty{67}{Hz}$।
\emph{2.} $\qty{15}{ms}/\qty{2}{ms} = 7.5$ खाने।
\emph{3.} परदा पहले \qty{50}{ms} दिखाता है, फिर \qty{20}{ms}: यानी $3$ पूरे चक्र
($50/15 = 3.3$), फिर $1$ ($20/15 = 1.3$)।
\end{solution}

\begin{solution}{exo:g10:signals-and-waves:15}
\emph{1.} $t = 2 \times 1.2\times10^{6}/(3.00\times10^{8}) =
\qty{8.0}{ms}$।
\emph{2.} $2 \times 1.2\times10^{6}/(2.0\times10^{8}) = \qty{12}{ms}$।
\emph{3.} तंतु दोनों मशीनों के बीच सीधा नहीं जाता, और रास्ते का हर

\end{solution}

\begin{solution}{pb:g10:signals-and-waves:1}
\textbf{1.} $d = 340 \times 9.0 = \qty{3060}{m} \approx \qty{3.1}{km}$।

\textbf{2.} $3060/(3.00\times10^{8}) \approx \qty{10}{\micro s}$, जो \qty{9.0}{s} से दस लाख गुना छोटा है: इसलिए चमक ही
गिरने का क्षण बता देती है।

\textbf{3.} \qty{3}{s} में \omterm{def:g10:signals-and-waves:sound}{ध्वनि} $340 \times 3 = \qty{1020}{m}
\approx \qty{1}{km}$ तय करती है: सेकंडों को तीन
से भाग देने पर किलोमीटर मिल जाते हैं।

\textbf{4.} $d = 340 \times 6.0 = \qty{2040}{m}$। तूफ़ान ने \qty{300}{s} में \qty{1020}{m} पाट दिए: $v = \qty{3.4}{m/s} \approx
\qty{12}{km/h}$, और
वह जहाज़ की ओर बढ़ रहा है।

\textbf{5.} पास वाला सिरा $2000/340 = \qty{5.9}{s}$ बाद सुनाई देता है और दूर वाला $5000/340 = \qty{14.7}{s}$
बाद: इसलिए गड़गड़ाहट लगभग \qty{8.8}{s} चलती है।

\textbf{6.} $d = 1500 \times 0.90/2 = \qty{675}{m}$।

\textbf{7.} $d = 1500 \times 0.20/2 = \qty{150}{m}$।

\textbf{8.} दो रुकावटें: $1500 \times 0.16/2 =
\qty{120}{m}$ पर मछलियों का झुंड और \qty{150}{m} पर
समुद्र-तल। झुंड तल से \qty{30}{m} ऊपर तैर रहा है।

\textbf{9.} \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} को अगले पिंग से पहले लौट आना चाहिए:
$d_{\max} = 1500 \times 0.50/2 = \qty{375}{m}$। इससे गहरे पानी से \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} अगले पिंग के \emph{बाद}
पहुँचती है और उसी की मान ली जाती है: परदा एक झूठा, कहीं अधिक उथला तल
दिखाने लगता है।

\textbf{10.} वायु में \qty{450}{m} के बजाय $340 \times 0.60/2 = \qty{102}{m}$। कोई देरी दूरी तभी बनती
है जब आपको वाहक और पार किए गए माध्यम में उसकी चाल, दोनों पता हों।

\textbf{11.} $f = 1/0.86 = \qty{1.16}{Hz}$, यानी $1.16 \times 60
\approx 70$ धड़कन प्रति मिनट।

\textbf{12.} $25 \times 0.86 = \qty{21.5}{mm}$।

\textbf{13.} $T = 15/25 = \qty{0.60}{s}$: $100$ धड़कन प्रति मिनट।

\textbf{14.} $T = 60/50 = \qty{1.2}{s}$; दूरी $25 \times 1.2 = \qty{30}{mm}$।

\textbf{15.} नहीं: मेहनत, साँस और \omterm{def:g10:inertia:inventory}{तनाव} के साथ \omterm{def:g10:signals-and-waves:period}{आवर्तकाल}
धड़कन-दर-धड़कन खिसकता रहता है। चिकित्सक \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} (यानी गति),
उसकी नियमितता, और हर चक्र का आकार पढ़ता है --- निदान संकेत में
ही छिपा होता है।

\textbf{16.} रेडियो: $5.0\times10^{4}/(3.00\times10^{8}) =
\qty{0.17}{ms}$ --- मानवीय इंद्रियों के लिए तात्क्षणिक।
\omterm{def:g10:signals-and-waves:sound}{ध्वनि}: $50\,000/340 \approx \qty{147}{s}$, यानी ढाई मिनट।

\textbf{17.} $t = 2.02\times10^{7}/(3.00\times10^{8}) = \qty{67}{ms}$।

\textbf{18.} $c \times \qty{1.0}{\micro s} = \qty{300}{m}$। \qty{3.0}{m} की सटीकता के लिए: $3.0/(3.00\times10^{8}) = \qty{10}{ns}$ --- और इसीलिए
GPS \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} परमाणु घड़ियाँ लेकर चलते हैं।

\textbf{19.} \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} निर्वात में बैठे हैं, और \omterm{def:g10:signals-and-waves:sound}{ध्वनि} को
माध्यम चाहिए: तब \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} से कोई संकेत निकलता ही नहीं।
(पूरे रास्ते वायु मान भी लें, तो \qty{340}{m/s} पर \qty{20200}{km} \qty{16}{h} से अधिक बैठता
है --- यानी घंटों पुरानी स्थिति।)

\textbf{20.} \qty{340}{m/s} (वायु में \omterm{def:g10:signals-and-waves:sound}{ध्वनि}), \qty{1500}{m/s} (समुद्री पानी
में \omterm{def:g10:signals-and-waves:sound}{ध्वनि}) और $c = \qty{3.00e8}{m/s}$ (रेडियो और प्रकाश); और नियम है $d = v\,t$,
जिसे \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} के लिए आधा कर दीजिए। आदत यह: अपना वाहक जानिए,
माध्यम में उसकी चाल जानिए, और देरी नापिए --- हर देरी भेस बदली हुई दूरी
है।
\end{solution}
