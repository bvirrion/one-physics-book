\section*{अध्याय \ref{ch:g10:relative-motion} --- सापेक्ष गति}

\begin{solution}{exo:g10:relative-motion:1}
डिब्बे के सापेक्ष: \omterm{def:g10:relative-motion:frame}{विराम} में। ज़मीन और पेड़ के सापेक्ष: \qty{100}{km/h} से
गति में। सामने से आती रेलगाड़ी के सापेक्ष: उलटी दिशाओं की चालें जुड़ती
हैं, $100 + 100 = \qty{200}{km/h}$। यात्री की रेलगाड़ी के तंत्र में पेड़ एक सरल रेखा पर पीछे की
ओर \qty{100}{km/h} से चलता है।
\end{solution}

\begin{solution}{exo:g10:relative-motion:2}
$72 / 3.6 = \qty{20}{m/s}$; $108 / 3.6 = \qty{30}{m/s}$; $25 \times 3.6 = \qty{90}{km/h}$; $340 \times 3.6 = \qty{1224}{km/h}$। हर युग्म में \unit{km/h} वाली संख्या ही बड़ी है,
जैसा जाँच-नियम माँगता है।
\end{solution}

\begin{solution}{exo:g10:relative-motion:3}
साइकिल सवार: $v = \frac{\qty{36}{km}}{\qty{1.5}{h}} = \qty{24}{km/h}
= 24/3.6 \approx \qty{6.7}{m/s}$। धावक: $v = \frac{\qty{100}{m}}{\qty{10.0}{s}} = \qty{10.0}{m/s}
= \qty{36}{km/h}$। धावक डेढ़ गुना तेज़ है --- पर केवल
दस \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} के लिए, जबकि साइकिल सवार यही रफ़्तार डेढ़ घंटे तक बनाए रखता है।
औसत चालों की न्यायसंगत तुलना तभी होती है जब अवधियाँ तुलनीय हों।
\end{solution}

\begin{solution}{exo:g10:relative-motion:4}

\end{solution}

\begin{solution}{exo:g10:relative-motion:5}
गिराई गई कलम: \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} काम कर जाता है, पर
\emph{रेलगाड़ी} का तंत्र और भी बेहतर है --- वहाँ कलम बस सीधी नीचे गिरती
है। अंतरिक्ष स्टेशन: \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} (वह समूची पृथ्वी की
कक्षा में है और उसके घूर्णन की परवाह नहीं करता)। मंगल का वर्ष:
\omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} (मंगल सूर्य के चारों ओर लगभग वृत्त बनाता
है)। दौड़: \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} (आरंभ रेखा, अंत रेखा और घड़ी, तीनों
ज़मीन से जुड़े हैं)।
\end{solution}

\begin{solution}{exo:g10:relative-motion:6}
\emph{1.} $108/3.6 = \qty{30}{m/s}$। अगले सिरे की ओर: $30 + 1.5 = \qty{31.5}{m/s}$; पिछले सिरे की ओर: $30 - 1.5 = \qty{28.5}{m/s}$।

\emph{2.} रेलगाड़ी के सापेक्ष यात्री \qty{1.5}{m/s} से \qty{60}{m} तय करता है:
$t = 60/1.5 = \qty{40}{s}$। ज़मीन के सापेक्ष यात्री $31.5 \times 40 = \qty{1260}{m}$ तय करता है --- यानी रेलगाड़ी
द्वारा ले जाए गए \qty{1200}{m} और चलकर तय किए गए \qty{60}{m} का जोड़, ठीक जैसा
संयोजन-नियम बताता है।
\end{solution}

\begin{solution}{exo:g10:relative-motion:7}
\emph{1.} बहाव के साथ: $5.0 + 2.0 = \qty{7.0}{m/s}$, इसलिए $t_1 = 420/7.0 = \qty{60}{s}$। बहाव के विरुद्ध: $5.0 - 2.0 = \qty{3.0}{m/s}$,
इसलिए $t_2 = 420/3.0 = \qty{140}{s}$।

\emph{2.} \omterm{def:g10:relative-motion:speed}{औसत चाल}: $v = \frac{840}{60 + 140}
= \frac{840}{200} = \qty{4.2}{m/s} < \qty{5.0}{m/s}$। धारा की मदद और अड़चन \emph{चाल} में
बराबर हैं, \emph{समय} में नहीं: नाव \qty{140}{s} तक धीमी पड़ी रहती है और केवल
\qty{60}{s} तक मदद पाती है, इसलिए धीमा चरण औसत पर हावी हो जाता है।
\end{solution}

\begin{solution}{exo:g10:relative-motion:8}
\emph{1.} उलटी दिशाएँ: A के तंत्र में B $25 + 20 = \qty{45}{m/s}$ से पास आती है। पार होना
तब पूरा होता है जब B कुल लंबाई $180 + 90 = \qty{270}{m}$ भर सरक जाए: $t = 270/45 = \qty{6.0}{s}$।

\emph{2.} एक ही दिशा: सापेक्ष चाल $25 - 20 = \qty{5.0}{m/s}$, और $t = 270/5.0 = \qty{54}{s}$ --- यानी नौ गुना
अधिक, और इसीलिए किसी ट्रक को पीछे छोड़ना कभी न ख़त्म होने वाला लगता है।
\end{solution}

\begin{solution}{exo:g10:relative-motion:9}
साइकिल सवार का तंत्र: एक वृत्त (रिम का वृत्त, धुरी पर केंद्रित)।
\omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}: एक \omterm{prop:g10:relative-motion:relativity}{चक्रज} --- मेहराबों की एक कड़ी, हर
चक्कर पर एक। स्वयं वाल्व का तंत्र: वाल्व \omterm{def:g10:relative-motion:frame}{विराम} में है, एक अकेला
बिंदु। \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में धुरी: पहिये की एक त्रिज्या की ऊँचाई
पर एक क्षैतिज सरल रेखा।
\end{solution}

\begin{solution}{exo:g10:relative-motion:10}
\emph{1.} $v = \frac{\qty{40000}{km}}{\qty{24}{h}}
\approx \qty{1.7e3}{km/h} = 1667/3.6 \approx \qty{463}{m/s}$ --- यानी ध्वनि से भी तेज़।

\emph{2.} हमें यह महसूस नहीं होता क्योंकि हमारे चारों ओर की हर चीज़ ---
ज़मीन, वायु, इमारतें --- ठीक वही गति साझा करती है, इसलिए हमारे सापेक्ष
कुछ भी नहीं चलता (और जड़त्व का सिद्धांत, \cref{ch:g10:inertia}, समझाएगा कि साझा एकसमान
गति भीतर से पकड़ में क्यों नहीं आती)। ``सूर्य उगता है'' भू-तंत्र की रपट
है और ``पृथ्वी घूमती है'' \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} की, और दोनों एक ही प्रेक्षण
की।
\end{solution}

\begin{solution}{exo:g10:relative-motion:11}
मान लीजिए सीढ़ी की लंबाई $L$ है। सीढ़ी की चाल: $L/60$; चलने की चाल:
$L/90$। चलती सीढ़ी पर चलने से चालें जुड़ जाती हैं:
\[
v = \frac{L}{60} + \frac{L}{90} = L\left(\frac{3 + 2}{180}\right)
= \frac{L}{36},
\]
इसलिए चढ़ाई में \qty{36}{s} लगते हैं। (यह $60$ और $90$ का औसत नहीं है:
चालें जुड़ती हैं, समय नहीं।)
\end{solution}

\begin{solution}{exo:g10:relative-motion:12}
\emph{1.} हर ओर $d$ दूरी पर:
$t = \frac{d}{60} + \frac{d}{90} = d\,\frac{3 + 2}{180} = \frac{d}{36}$,
इसलिए $v = \frac{2d}{d/36} = \qty{72}{km/h}$।

\emph{2.} भोला उत्तर $75$ दोनों चालों का औसत ऐसे लेता है मानो चालक ने
हर चरण पर बराबर \emph{समय} बिताया हो; असल में दूरियाँ बराबर हैं, इसलिए
धीमा चरण अधिक देर चलता है ($\frac{d}{60} > \frac{d}{90}$) और औसत को \qty{60}{km/h} की ओर खींच ले जाता
है। \omterm{def:g10:relative-motion:speed}{औसत चाल} कुल दूरी बटा कुल समय है --- और कुछ नहीं।
\end{solution}

\begin{solution}{exo:g10:relative-motion:13}
\emph{1.} पहले साइकिल सवार के तंत्र में दूसरा \qty{35}{km} दूर से एक सरल रेखा
पर $15 + 20 = \qty{35}{km/h}$ से पास आता है: वे ठीक \qty{1.0}{h} बाद मिलते हैं।

\emph{2.} पहले ने $15 \times 1 = \qty{15}{km}$ तय किए, दूसरे ने $20 \times 1 = \qty{20}{km}$ --- कुल \qty{35}{km}, जैसा
होना ही चाहिए।

\emph{3.} एक ही दिशा: पास आने की चाल $20 - 15 = \qty{5}{km/h}$, इसलिए पीछा $35/5 = \qty{7}{h}$ चलता है।
\end{solution}

\begin{solution}{exo:g10:relative-motion:14}
\emph{1.} धारा तैराक को नदी के अनुदिश धकेलती है, पार की ओर नहीं: दोनों
गतियाँ स्वतंत्र रूप से जुड़ती हैं, और बहाव चाहे जितना हो, दूसरे किनारे की
ओर बढ़त \qty{1.2}{m/s} से ही होती है। पार करने का समय: $t = 60/1.2 = \qty{50}{s}$।

\emph{2.} बहाव: बहाव की ओर $0.9 \times 50 = \qty{45}{m}$।

\emph{3.} भू-तंत्र का वेग $1.2$ और $0.9$ भुजाओं वाले समकोण त्रिभुज का
कर्ण है: $v = \sqrt{1.2^2 + 0.9^2} = \sqrt{2.25} = \qty{1.5}{m/s}$ (यानी $0.3$ से गुणा किया $3$--$4$--$5$
त्रिभुज)। पथ की लंबाई: $1.5 \times 50 = \qty{75}{m}$ --- या सीधे $\sqrt{60^2 + 45^2} = \qty{75}{m}$।
\end{solution}

\begin{solution}{exo:g10:relative-motion:15}
\emph{1.} $v = \frac{2\pi \times \num{1.50e11}}{\num{3.16e7}}
= \frac{\num{9.42e11}}{\num{3.16e7}} \approx \qty{2.98e4}{m/s}
\approx \qty{30}{km/s}$।

\emph{2.} $30 / 0.46 \approx 65$: पृथ्वी हमें अपनी धुरी के चारों ओर घुमाने से लगभग
पैंसठ गुना तेज़ी से सूर्य के चारों ओर ले जाती है।

\emph{3.} \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} में मंगल का आभासी पथ उसकी अपनी कक्षीय
गति और पृथ्वी की गति, दोनों को मिलाता है, और जब तेज़ पृथ्वी मंगल को पीछे

चारों ओर घूमते हैं --- ``उलटा फंदा'' मंगल की गति नहीं, बल्कि तंत्र की
गढ़ी हुई चीज़ है।
\end{solution}

\begin{solution}{pb:g10:relative-motion:1}
\textbf{1.} $t = 90/1.0 = \qty{90}{s}$।

\textbf{2.} ज़मीन के सापेक्ष चाल $1.5 + 1.0 = \qty{2.5}{m/s}$: $t = 90/2.5 = \qty{36}{s}$।

\textbf{3.} $t = 90/1.5 = \qty{60}{s}$।

\textbf{4.} ज़मीन के सापेक्ष $1.5 - 1.0 = \qty{0.5}{m/s}$, पट्टी की दिशा के विरुद्ध: $t = 90/0.5 = \qty{180}{s}$
--- यानी पूरे तीन मिनट तेज़ चहलक़दमी।

\textbf{5.} ज़मीन के सापेक्ष चाल $1.0 - 1.0 = \qty{0}{m/s}$: बच्चा जी-जान से दौड़ता है और
कहीं नहीं पहुँचता, \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में \omterm{def:g10:relative-motion:frame}{विराम} में और
पट्टी के तंत्र में पूरी रफ़्तार पर। यही तो ट्रेडमिल है।

\textbf{6.} पट्टी का तंत्र / \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}: खड़ा यात्री
$0$ / \qty{1.0}{m/s}; पट्टी के साथ चलने वाला $1.5$ / \qty{2.5}{m/s}; फ़र्श पर
चलने वाला $0.5$ (पट्टी के तंत्र में फ़र्श पीछे की ओर सरकता है) /
\qty{1.5}{m/s}; शरारती $1.5$ / \qty{0.5}{m/s}; बच्चा $1.0$ / \qty{0}{m/s}। हर चलने वाले
की टाँगें उसी तंत्र का स्तंभ महसूस करती हैं जो उसके पैरों तले है ---
पट्टी पर सवार लोगों के लिए पट्टी वाला स्तंभ, फ़र्श पर चलने वाले के लिए
भू-तंत्र वाला: मेहनत पैरों तले की चीज़ के सापेक्ष चाल से तय होती है।

\textbf{7.} धारा चाहे जो करे, नदी पार करने की बढ़त \qty{2.0}{m/s} से ही होती है:
$t = 120/2.0 = \qty{60}{s}$।

\textbf{8.} बहाव: बहाव की ओर $1.5 \times 60 = \qty{90}{m}$।

\textbf{9.} ज़मीन के सापेक्ष चाल: $2.0$ और $1.5$ भुजाओं वाले समकोण
त्रिभुज का कर्ण, $v = \sqrt{2.0^2 + 1.5^2} = \sqrt{6.25}
= \qty{2.5}{m/s}$। \omterm{def:g10:relative-motion:trajectory}{गतिपथ} की लंबाई: $2.5 \times 60 = \qty{150}{m}$ --- यानी फिर
वही $3$--$4$--$5$ तिकड़ी, $90$--$120$--$150$ के पैमाने
पर।

\textbf{10.} पानी का तंत्र: एक सरल रेखा, \qty{120}{m}, किनारों के लंबवत।
\omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}: भी एक सरल रेखा, पर तिरछी और \qty{150}{m} लंबी। आकार
वही (सीधा), दिशा और लंबाई अलग: \omterm{def:g10:relative-motion:trajectory}{गतिपथ} तंत्र का होता है, नौका का
नहीं।

\textbf{11.} पार की बढ़त \qty{1.0}{m/s} से होती है, इसलिए $t = 120/1.0 = \qty{120}{s}$। बहाव:
$1.5 \times 120 = \qty{180}{m}$ --- यानी नौका के \qty{90}{m} से दुगुना। ज़मीन के सापेक्ष चाल:
$1.0$ और $1.5$ भुजाओं वाला कर्ण, $v = \sqrt{1.0^2 + 1.5^2} = \sqrt{3.25}
\approx \qty{1.8}{m/s}$। धीमे तैराक को पार करने में
दुगुना समय लगता है, इसलिए उसी धारा को उसे बहाकर ले जाने के लिए दुगुना
समय मिल जाता है।

\textbf{12.} \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}: लट्ठा धारा की \qty{1.5}{m/s} से बहता
है। पानी का तंत्र: वह \omterm{def:g10:relative-motion:frame}{विराम} में है, अपने चारों ओर के पानी के
साथ तैरता हुआ। लट्ठा ``चल रहा है'' या नहीं, इसका कोई तंत्र-मुक्त उत्तर
है ही नहीं --- और यही पूरे अध्याय की बात है।

\textbf{13.} \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र}। कक्षा की त्रिज्या
$r = \num{6.37e6} + \num{4.0e5} = \qty{6.77e6}{m}$, इसलिए
\[
v = \frac{2\pi r}{T} = \frac{2\pi \times \num{6.77e6}}{\num{5.56e3}}
\approx \qty{7.65e3}{m/s} \approx \qty{7.7}{km/s}.
\]

\textbf{14.} जब तक स्टेशन एक कक्षा पूरी करता है, पृथ्वी उसके
नीचे लगभग $23^\circ$ घूम जाती है, इसलिए ज़मीन पर उसकी पटरी हर चक्कर में
पश्चिम की ओर खिसक जाती है। \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में स्टेशन का पथ एक
उलझा हुआ लहरदार वक्र है जो कभी ठीक-ठीक दोहराया नहीं जाता;
\omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} में वह एक ही स्थिर, लगभग वृत्ताकार पथ है।

\textbf{15.} $7.65 / 0.46 \approx 17$: स्टेशन घूमती भूमध्य रेखा से मोटे तौर पर सत्रह गुना
तेज़ दौड़ता है, और इसीलिए वह पूरा आकाश मिनटों में पार कर जाता है।

\textbf{16.} \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} में $v = \frac{2\pi \times \num{4.22e7}}{\num{8.64e4}}
\approx \qty{3.07e3}{m/s} \approx \qty{3.1}{km/s}$। \omterm{def:g10:relative-motion:frames}{भू-निर्देश
तंत्र} में वह \emph{\omterm{def:g10:relative-motion:frame}{विराम} में} है, भूमध्य रेखा के एक ही स्थिर
बिंदु के ऊपर टँगा हुआ --- वह पृथ्वी की धुरी का चक्कर ठीक उतने ही समय में
लगाता है जितने में ज़मीन स्वयं, इसलिए \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} में
दोनों गतियाँ कट जाती हैं। \omterm{def:g10:relative-motion:frame}{विराम} में और \qty{3.1}{km/s} पर: दोनों सच,
एक-एक तंत्र में।

\textbf{17.} \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} में चंद्रमा लगभग पृथ्वी की ही
कक्षा पर चलता है --- सूर्य के चारों ओर एक विशाल, लगभग वृत्ताकार
पथ --- जिस पर \qty{27}{day} के चक्र का एक हल्का डगमगाना चढ़ा होता है। सूर्य का
प्रभाव इतना हावी है कि पथ हमेशा सूर्य की ओर ही मुड़ता है; वह कभी पीछे की
ओर फंदा नहीं बनाता।

\textbf{18.} ``सूर्य उगता है'' \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र} की रपट है, जहाँ
प्रेक्षक \omterm{def:g10:relative-motion:frame}{विराम} में है और सूर्य आकाश पार करता है। ``पृथ्वी घूमती
है'' \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र} की रपट है, जहाँ सूर्य की दिशा स्थिर है और
प्रेक्षक धुरी के चारों ओर पूरब की ओर ले जाया जाता है। दोनों एक ही
प्रेक्षण बताते हैं; हर एक अपने तंत्र में सही है, और \cref{prop:g10:relative-motion:relativity} कहता है कि
अकेली गति का कोई प्रेक्षण इनके बीच फ़ैसला नहीं कर सकता।

\textbf{19.} टॉलेमी का पृथ्वी-केंद्रित तंत्र आकाश की दैनिक गति से
स्वाभाविक रूप से मेल खाता था और ग्रहों की स्थितियाँ काम भर ठीक बताता था,
पर इसकी क़ीमत यह थी कि हर ग्रह की भटकन दोहराने के लिए फंदों पर फंदे
(उपवृत्त) चाहिए थे। कोपरनिकस के सूर्य-केंद्रित तंत्र ने हर ग्रह को एक
स्थिर अर्ध-वृत्ताकार पथ दे दिया और मंगल के उलटे फंदों को तंत्र का असर
बना दिया --- यानी पृथ्वी का भीतरी गलियारे से किसी धीमे धावक को पीछे छोड़
जाना। आधुनिक फ़ैसला यह नहीं है कि सूर्य ``सचमुच'' स्थिर खड़ा है (वह भी
आकाशगंगा के केंद्र के चारों ओर चलता है), बल्कि यह कि तंत्र औज़ार होते
हैं: \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र} ग्रहों की गति को बेहिसाब सरल कर देता है, और
तंत्र सरलता के लिए होता है, सच्चाई के लिए नहीं।

\textbf{20.} \omterm{def:g10:relative-motion:frames}{भू-निर्देश तंत्र}: \qty{0}{km/s} --- आरामकुर्सी पर
\omterm{def:g10:relative-motion:frame}{विराम} में। \omterm{def:g10:relative-motion:frames}{भूकेंद्रीय तंत्र}: लगभग \qty{0.46}{km/s}, पृथ्वी की
धुरी के चारों ओर ले जाए जाते हुए। \omterm{def:g10:relative-motion:frames}{सूर्यकेंद्रीय तंत्र}: लगभग
\qty{30}{km/s}, सूर्य के चारों ओर ले जाए जाते हुए। \qty{90}{min} की फ़िल्म के दौरान:
$d = \num{3.0e4} \times 5400
\approx \qty{1.6e8}{m} = \qty{160000}{km}$ --- यानी आरामकुर्सी छोड़े बिना ही चंद्रमा तक के लगभग आधे रास्ते
तक। सीख: गति पिंड का गुण नहीं, पिंड और तंत्र का गुण है; वह तंत्र चुनिए
जो कहानी को सरल बना दे, और हमेशा बताइए कि आपने कौन-सा चुना।
\end{solution}
