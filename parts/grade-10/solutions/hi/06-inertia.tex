\begin{solution}{exo:g10:inertia:5}

\emph{\omterm{def:g10:inertia:compensate}{संतुलनकारी}} हैं (जैसे भार और प्रतिक्रिया)। (घ) सही
--- यह सिद्धांत का प्रतिधनात्मक पाठ है।

\begin{solution}{exo:g10:inertia:6}
\emph{1.} $v = 24/3.0 = \qty{8.0}{m/s}$। बल: भार और \omterm{def:g10:inertia:inventory}{अभिलंब प्रतिक्रिया};
वे \omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं (\omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान गति}, \omterm{def:g10:inertia:inventory}{घर्षण}
नगण्य)। \emph{2.} कंक्रीट का \omterm{def:g10:inertia:inventory}{घर्षण} अब नगण्य नहीं रहा: सूची में
एक बिना कटा पीछे की ओर लगता बल जुड़ जाता है, और सचमुच वेग बदलता
है --- चकती धीमी पड़कर रुक जाती है।
\end{solution}

\end{solution}

\begin{solution}{exo:g10:inertia:8}

$T = 1.8 \times 9.81 \approx \qty{17.7}{N}$। \emph{2.} हर \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} तार आधा बोझ उठाता है: $T = \qty{8.8}{N}$।
\end{solution}

\begin{solution}{exo:g10:inertia:9}
ब्रेक लगने पर: यात्री का शरीर अपना वेग बनाए रखता है (जड़त्व) जबकि बस
अपना वेग खो देती है, इसलिए यात्री \emph{बस के सापेक्ष} आगे बढ़ जाता है।
मोड़ पर: शरीर सीधा चलते रहना चाहता है जबकि बस उसके नीचे मुड़ जाती है,
इसलिए वह मोड़ के बाहर की ओर खिसक जाता है। दोनों हालात में वह
``बल'' काल्पनिक है; जिसकी कमी है वह कोई असली बल (पकड़,
रेलिंग) है जो यात्री का वेग भी बस के साथ बदल दे।
\end{solution}

\begin{solution}{exo:g10:inertia:10}
\omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान गति} है, इसलिए भार और वायु का प्रतिरोध
\omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं: कर्षण $= P = 85 \times 9.81 \approx \qty{834}{N}$, \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} और ऊपर की ओर।
\end{solution}

\begin{solution}{exo:g10:inertia:11}
\emph{1.} \omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान गति}: क्षैतिज बल
\omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं, इसलिए कर्षण \qty{300}{N} है, रस्सी के उलटी दिशा में।
\emph{2.} स्कीबाज़ का भार और स्कीयों पर पानी का ऊपर की ओर धक्का।
\end{solution}

\begin{solution}{exo:g10:inertia:12}
\emph{1.} $P = 2.4 \times 9.81 \approx \qty{23.5}{N}$; \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} घटक भार उठाते हैं: $2T\cos 40^\circ = P$, इसलिए
$T = 23.5/(2 \times 0.766) \approx \qty{15.4}{N}$। \emph{2.} $\theta \to 90^\circ$ होते-होते $T = P/(2\cos\theta)$ बिना किसी सीमा के बढ़ता जाता
है, क्योंकि $\cos\theta \to 0$; और ठीक-ठीक क्षैतिज तारों का कोई \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} घटक होता
ही नहीं, इसलिए भार उठाने वाला कुछ बचता ही नहीं।
\end{solution}

\begin{solution}{exo:g10:inertia:13}
धक्का: भार, प्रतिक्रिया, हाथ का \omterm{def:g10:inertia:inventory}{प्रणोद}, थोड़ा
\omterm{def:g10:inertia:inventory}{घर्षण}; चाल बढ़ती है, इसलिए बल \omterm{def:g10:inertia:compensate}{संतुलनकारी} नहीं हैं
(\cref{prop:g10:inertia:converse})। सरकना: भार, प्रतिक्रिया, थोड़ा \omterm{def:g10:inertia:inventory}{घर्षण}; पत्थर
धीमा पड़ता है, इसलिए फिर कोई संतुलन नहीं --- बचा हुआ बिना कटा हिस्सा
\omterm{def:g10:inertia:inventory}{घर्षण} ही है (आदर्श बर्फ़ पर वह सदा सरकता रहता)। \omterm{def:g10:relative-motion:frame}{विराम}: केवल
भार और प्रतिक्रिया, और वे \omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं (\cref{thm:g10:inertia:principle})।
\end{solution}

\begin{solution}{exo:g10:inertia:14}
\emph{1.} जड़त्व के सिद्धांत से अचर वेग बनाए रखने के लिए कोई बल
चाहिए ही नहीं --- और तारों के बीच के अंतरिक्ष में लगभग कोई बल लगता भी
नहीं। अरस्तू भविष्यवाणी करते कि ``चलाने वाले'' के हटते ही यान रुक जाएगा।
\emph{2.} एक वर्ष $\approx \qty{3.156e7}{s}$, इसलिए $d = 17 \times \num{3.156e7} \approx \qty{5.4e8}{km} \approx
\qty{3.6}{au}$। \emph{3.} वेग एक सदिश है: उसकी
दिशा बदलना वेग को बदलना है, और इसके लिए एक बिना कटा बल चाहिए।
\end{solution}

\begin{solution}{exo:g10:inertia:15}
\omterm{def:g10:relative-motion:frame}{विराम} में तीनों बल \omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं: $\vect F_3 = -(\vect F_1 + \vect F_2)$, इसलिए
$\vect F_3\,(-2, -4)$, परिमाण $\sqrt{4 + 16} = \sqrt{20} \approx \qty{4.5}{N}$, और दिशा पहले दोनों के परिणामी की उलटी (जाली पर
नीचे-बाएँ)।
\end{solution}

\begin{solution}{pb:g10:inertia:1}
\textbf{1.} \qty{0.75}{m/s} पर \omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान गति}। बल:
भार और पट्टी की \omterm{def:g10:inertia:inventory}{अभिलंब प्रतिक्रिया} (\omterm{def:g10:inertia:inventory}{घर्षण} की
ज़रूरत नहीं: सूटकेस पट्टी पर सरकता नहीं)।
\textbf{2.} हाँ --- \omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान गति} (\cref{thm:g10:inertia:principle})। $P = 23 \times 9.81 \approx
\qty{226}{N}$,
इसलिए $R = \qty{226}{N}$।
\textbf{3.} दोनों आरेख एक जैसे हैं। \omterm{def:g10:relative-motion:frame}{विराम} और \omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान
गति} एक ही यांत्रिक अवस्था हैं: जड़त्व दोनों में भेद नहीं करता।
\textbf{4.} पट्टी रुक जाती है पर सूटकेस अपना \qty{0.75}{m/s} बनाए रखता है: वह
सरकता या आगे झुकता है, जब तक \omterm{def:g10:inertia:inventory}{घर्षण} उसकी गति सोख न ले। उसे किसी
ने धकेला नहीं --- उसने बस अपना वेग बनाए रखा।
\textbf{5.} एक ही दिशा की चालें जुड़ती हैं: $0.75 + 1.2 =
\qty{1.95}{m/s}$। उसका वेग अचर है,
इसलिए उस पर लगते बल \omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं।
\textbf{6.} $P = 0.160 \times 9.81 \approx \qty{1.57}{N}$ नीचे, $R = \qty{1.57}{N}$ ऊपर, और कोई क्षैतिज बल
\emph{नहीं}।
\textbf{7.} चकती बिना किसी चलाने वाले के \qty{8.0}{m/s} बनाए रखती है।
रोज़मर्रा की सतहें \omterm{def:g10:inertia:inventory}{घर्षण} छिपा लेती हैं --- यानी एक बिना कटा
पीछे की ओर लगता बल --- इसलिए गति अपने आप मरती हुई लगती है।
\textbf{8.} उसका वेग बदलता है, इसलिए बल अब \omterm{def:g10:inertia:compensate}{संतुलनकारी}
नहीं रहे (\cref{prop:g10:inertia:converse}); दोषी खुरदरी बर्फ़ का \omterm{def:g10:inertia:inventory}{घर्षण} है।
\textbf{9.} नहीं: वेग एक \emph{सदिश} है, और उसकी दिशा बदल रही है, इसलिए
बल \omterm{def:g10:inertia:compensate}{संतुलनकारी} नहीं हैं --- बर्फ़ उसके ब्लेडों पर
बग़ल से धक्का देती है।
\textbf{10.} (क)--(3), (ख)--(1), (ग)--(2)।
\textbf{11.} भार $P = 80 \times 9.81 \approx \qty{785}{N}$, कर्षण शून्य: भार को कुछ भी
संतुलित नहीं करता, इसलिए वेग बदलना ही चाहिए --- छलाँग लगाने
वाला नीचे की ओर तेज़ होता जाता है।
\textbf{12.} जब तक कर्षण भार से $<$ है तब तक बल
\omterm{def:g10:inertia:compensate}{संतुलनकारी} नहीं होते: चाल बढ़ती जाती है, और उसके साथ कर्षण भी।
\textbf{13.} स्थिर चाल: भार और कर्षण \omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं,
इसलिए कर्षण \qty{785}{N} है। गति सीधी और एकसमान है --- यानी गिरता पिंड ठीक
उसी यांत्रिक अवस्था में है जिसमें चलपथ पर रखा सूटकेस था: भेस बदले
जड़त्व।
\textbf{14.} अब कर्षण भार से अधिक है: बल
\omterm{def:g10:inertia:compensate}{संतुलनकारी} नहीं हैं, इसलिए वेग बदलता है --- छलाँग लगाने वाला
नीचे उतरता तो रहता है पर धीमा पड़ जाता है; और चाल गिरने के साथ कर्षण भी
घटकर भार की ओर लौटने लगता है।
\textbf{15.} फिर स्थिर: कर्षण $= P = \qty{785}{N}$ --- यानी ठीक वही जो \qty{50}{m/s} पर
था।
\textbf{16.} सीमांत कर्षण हमेशा भार के बराबर होता है, छतरी हो
या न हो। पैराशूट वह \emph{चाल} बदलता है जिस पर कर्षण \qty{785}{N} तक पहुँचता
है: \qty{50}{m/s} के बजाय \qty{5}{m/s}।
\textbf{17.} ``\dots पिंड पर लगते बल \omterm{def:g10:inertia:compensate}{संतुलनकारी} हैं
(या कोई बल लगता ही नहीं)'' --- यानी \cref{thm:g10:inertia:principle} के दोनों पाठ। \omterm{def:g10:relative-motion:frame}{विराम} तो बस
$\vect v = \vect 0$ है, यानी बाक़ी सबकी तरह एक \omterm{def:g10:inertia:uniform}{सरल रेखीय एकसमान गति}।
\textbf{18.} बर्फ़ का मैदान: चकती बिना किसी दिखते चलाने वाले के सरकती
चली जाती है। गैलीलियो लगातार चिकनी होती सतह की ओर इशारा करते ---
चीज़ों को \omterm{def:g10:inertia:inventory}{घर्षण} रोकता है, प्रकृति नहीं।
\textbf{19.} बल लगते तो हैं --- भार \qty{785}{N} नीचे और
कर्षण \qty{785}{N} ऊपर; वे \emph{\omterm{def:g10:inertia:compensate}{संतुलनकारी}} हैं, और ठीक इसीलिए चाल
अचर है।
\textbf{20.} चलपथ: भार और \omterm{def:g10:inertia:inventory}{अभिलंब प्रतिक्रिया}। मैदान:
भार और \omterm{def:g10:inertia:inventory}{अभिलंब प्रतिक्रिया}। पैराशूट: भार और
कर्षण। इकलौता नियम: \omterm{thm:g10:inertia:principle}{जड़त्व का सिद्धांत}, अपने सीधे और विलोम, दोनों पाठों
में।
\end{solution}
