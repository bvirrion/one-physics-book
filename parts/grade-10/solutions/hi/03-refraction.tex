\begin{solution}{exo:g10:refraction:6}
\emph{1.} पाँचों अनुपात
\[
\frac{0.342}{0.233} = 1.47,\quad
\frac{0.500}{0.334} = 1.50,\quad
\frac{0.643}{0.431} = 1.49,\quad
\frac{0.766}{0.515} = 1.49,\quad
\frac{0.866}{0.581} = 1.49 :
\]
हैं --- माप की त्रुटि के भीतर अचर, यानी \omterm{thm:g10:refraction:snell}{स्नेल का नियम}।

\emph{2.} $n \approx 1.49$: प्लेक्सिग्लास।

\emph{3.} \omterm{thm:g10:refraction:snell}{स्नेल का नियम} $\sin i_2 = \sin i_1 / n$ बताता है: ज्या के सामने ज्या
रखने पर आँकड़ों को मूल बिंदु से जाती एक सीधी रेखा पर आना ही चाहिए, और
इसे एक रूलर पलक झपकते जाँच लेता है। कोण के सामने कोण रखने पर आँकड़े एक
वक्र पर पड़ते हैं; छोटे कोणों पर वह वक्र धोखा देकर सीधी रेखा जैसा लगता
है --- और ठीक इसी ने टॉलेमी को छला था।
\end{solution}

\begin{solution}{exo:g10:refraction:7}
$n_2 = 1.00$ के साथ $\sin i_c = n_2/n_1$: पानी $\sin i_c = 1/1.33 = 0.752$, $i_c = 48.8^\circ$; काँच $\sin i_c = 1/1.50 = 0.667$, $i_c = 41.8^\circ$; हीरा
$\sin i_c = 1/2.42 = 0.413$, $i_c = 24.4^\circ$।

वायु से पानी में जाते समय हर आपतन के लिए $\sin i_2 = \sin i_1/1.33 \leq 1/1.33 < 1$: अपवर्तित कोण हमेशा
मौजूद रहता है, इसलिए कुछ विशेष कभी होता ही नहीं --- \omterm{def:g10:refraction:critical}{पूर्ण आंतरिक
परावर्तन} के लिए $n_1 > n_2$ चाहिए।
\end{solution}

\begin{solution}{exo:g10:refraction:9}

\emph{3.} दोनों \omterm{def:g10:refraction:refraction}{अपवर्तन} एक-दूसरे को काट देते हैं: किरण अपनी
मूल दिशा के \emph{समांतर} ही निकलती है, बस बगल में खिसक जाती है। इसीलिए
खिड़की दृश्य को बिगाड़ती नहीं, केवल (अगोचर रूप से) सरका देती है।
\end{solution}

\begin{solution}{exo:g10:refraction:11}

\emph{2.} \omterm{def:g10:refraction:angles}{अभिलंब} से $i_c$ पर चलती किरण अक्ष के साथ $90^\circ - i_c$

\begin{solution}{exo:g10:refraction:12}
\emph{1.} क्षितिज से छूता हुआ आता प्रकाश $90^\circ$ के पास $i_1$ पर
पहुँचता है; वह अपवर्तित होकर $\sin i_2 = \sin 90^\circ / 1.33 = 0.752$ पर, यानी $i_2 = 48.8^\circ$ पर आता है ---
यानी ठीक \omterm{def:g10:refraction:critical}{क्रांतिक कोण} पर, जैसा प्रतिवर्तिता माँगती है। इसलिए
पूरे आकाश की किरणें आँख तक $48.8^\circ$ अर्ध-कोण वाले एक शंकु के भीतर ही
पहुँचती हैं।

\emph{2.} चकती का किनारा वहीं है जहाँ वह शंकु सतह से मिलता है:
\[
r = d \tan i_c = \qty{2.0}{m} \times \tan 48.8^\circ
= \qty{2.0}{m} \times 1.14 = \qty{2.3}{m}.
\]

\emph{3.} चकती के बाहर नीचे से आता प्रकाश सतह से \omterm{def:g10:refraction:critical}{क्रांतिक कोण}
के पार टकराता है और पूरी तरह परावर्तित हो जाता है: गोताखोर को एक चाँदी
जैसा दर्पण दिखता है जिसमें तरणताल का तल दिखाई देता है।
\end{solution}

\begin{solution}{exo:g10:refraction:13}
\emph{1.} लाल: $\sin i_2 = 1.51 \times \sin 30^\circ = 0.755$, इसलिए $i_2 = 49.0^\circ$। नीला: $\sin i_2 = 1.53 \times 0.500 = 0.765$, इसलिए $i_2 = 49.9^\circ$। श्वेत
किरण $49.9^\circ - 49.0^\circ = 0.9^\circ$ पर पंखे की तरह फैल जाती है --- एक ही \omterm{def:g10:refraction:refraction}{अपवर्तन} से
मिला प्रिज़्म का \omterm{def:g10:light-spectra:white}{वर्णक्रम}।

\emph{2.} $45^\circ$ पर: $\sin i_2 = 1.51 \times 0.707 = 1.07 > 1$ (और नीले के लिए $1.53 \times 0.707 = 1.08$)। किसी भी रंग के
लिए कोई अपवर्तित कोण नहीं बचता: \omterm{def:g10:refraction:critical}{क्रांतिक कोण} $\arcsin(1/1.51) = 41.5^\circ < 45^\circ$ है, इसलिए
लंबा फलक पूरी किरण को पूर्णतः परावर्तित कर देता है।
\end{solution}

किरण पूरी तरह परावर्तित होती है --- प्रकाश का $100\%$, और कोई परत नहीं

\emph{2.} काँच--पानी: $\sin i_c = \dfrac{1.33}{1.50} = 0.887$, इसलिए $i_c = 62.5^\circ$। अब $45^\circ < i_c$: आपतन क्रांतिक

\end{solution}

भी प्रति मिलीसेकंड के भाव पर।
\end{solution}

\begin{solution}{pb:g10:refraction:1}
\textbf{1.} $v_{\text{पानी}} = \qty{3.00e8}{m/s}/1.33 =
\qty{2.26e8}{m/s}$; $v_{\text{हीरा}} = \qty{3.00e8}{m/s}/2.42 =
\qty{1.24e8}{m/s}$ --- हीरा प्रकाश को $n = 2.42$ गुना धीमा कर
देता है।

\textbf{2.} $\sin i_2 = \sin 40^\circ / 1.33 = 0.643/1.33 = 0.483$, इसलिए $i_2 = 28.9^\circ$।

\textbf{3.} $\sin i_2 = 1.33 \times \sin 28.9^\circ = 1.33 \times
0.483 = 0.643$, इसलिए $i_2 = 40^\circ$: प्रकाश अपने ही पथ पर लौटता है ---
यानी प्रकाश-किरणों की प्रतिवर्तिता।

\textbf{4.} $\sin i_c = 1/1.33 = 0.752$, इसलिए $i_c = 48.8^\circ$। $60^\circ > i_c$ पर \omterm{thm:g10:refraction:snell}{स्नेल के नियम}
को $\sin i_2 = 1.33 \sin
60^\circ = 1.15 > 1$ चाहिए होता: कोई \omterm{def:g10:refraction:refraction}{अपवर्तित किरण} नहीं; सतह सारा
प्रकाश पूरी तरह नीचे लौटा देती है।

\textbf{5.} $v = \qty{3.00e8}{m/s}/1.0003 = \qty{2.999e8}{m/s}$: वायु में प्रकाश निर्वात से $0.03\%$ भर धीमा है, जो
हमारे कोण-मापों की परिशुद्धि से कहीं नीचे है, इसलिए $n_{\text{वायु}} = 1.00$ लेना
निरापद है।

\textbf{6.} $d' = \qty{3.0}{m}/1.33 = \qty{2.3}{m}$: तरणताल अपनी \qty{70}{cm} गहराई छिपा लेता है।

\textbf{7.} सिक्के से निकलती दो किरणें सतह पर अपवर्तित होकर
\omterm{def:g10:refraction:angles}{अभिलंब} से \emph{दूर} मुड़ती हैं, इसलिए वे पानी के ऊपर नीचे की
तुलना में अधिक तेज़ी से फैलती हैं। आँख दोनों निकलती किरणों को सीधी
रेखाओं में पीछे की ओर बढ़ाती है; वे सिक्के के \emph{ऊपर} कटती हैं, और
आँख प्रतिबिंब वहीं रख देती है।

\textbf{8.} $r = d \tan i_c = \qty{3.0}{m} \times \tan 48.8^\circ =
\qty{3.0}{m} \times 1.14 = \qty{3.4}{m}$: लगभग \qty{6.9}{m} चौड़ी एक चकती पूरा आकाश समेट लेती है।

\textbf{9.} $30^\circ$ पर: $\sin i_2 = 1.33 \times 0.500 = 0.665$, किरण $41.7^\circ$ पर निकलती है। $45^\circ$
पर: $\sin i_2 = 1.33 \times
0.707 = 0.940$, वह $70.1^\circ$ पर, लगभग छूती हुई निकलती है। $50^\circ$ पर:
$1.33 \times 0.766 = 1.02 > 1$ --- \omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन}, किरण वापस तरणताल में गोता
लगा जाती है।

\textbf{10.} प्रतिबिंब असली मछली के ऊपर है (\cref{prop:g10:refraction:depth}), इसलिए निशाना उससे
\emph{नीचे} लगाइए। और हाँ: मछली को भी, पलटकर देखने पर, मछुआरा खिसका हुआ
दिखता है, स्नेल की खिड़की के भीतर ऊर्ध्वाधर की ओर सिमटा हुआ --- दोनों
एक-दूसरे को वहाँ देखते हैं जहाँ दूसरा है ही नहीं।

\textbf{11.} हीरा: $\sin i_c = 1/2.42 = 0.413$, इसलिए $i_c = 24.4^\circ$, जबकि काँच के लिए $41.1^\circ$:
हीरे का निकास-शंकु आधा भी चौड़ा नहीं है।

\textbf{12.} हीरे में $30^\circ > 24.4^\circ$: \omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन}, किरण
पत्थर में ही रह जाती है। काँच में $30^\circ < 41.1^\circ$: किरण पीछे से रिस जाती है।
ब्रिलियंट तराश के फलक ऐसे कोणों पर होते हैं कि भीतर आता प्रकाश उनसे
लगभग हमेशा $24.4^\circ$ के पार टकराए और तब तक उछलता रहे जब तक ऊपर की ओर,
आँख की ओर, न निकल जाए --- जबकि काँच की नकल, अपने चौड़े निकास-शंकु के
साथ, प्रकाश को बस निकल जाने देती है।

\textbf{13.} $\sin 45^\circ = 0.707$। लाल: $0.707/2.41 = 0.293$, इसलिए $i_2 = 17.1^\circ$। नीला: $0.707/2.45 = 0.289$,
इसलिए $i_2 = 16.8^\circ$। प्रवेश पर बँटवारा: लगभग $0.3^\circ$।

\textbf{14.} $\sin 17.0^\circ = 0.292$। लाल: $\sin i_2 = 2.41 \times
0.292 = 0.705$, इसलिए $i_2 = 44.8^\circ$। नीला: $\sin i_2 = 2.45 \times
0.292 = 0.716$,
इसलिए $i_2 = 45.8^\circ$। अब पंखा $1.0^\circ$ चौड़ा है: \omterm{def:g10:refraction:critical}{क्रांतिक कोण} के पास
काम करता निर्गत \omterm{def:g10:refraction:refraction}{अपवर्तन} प्रवेश वाले $0.3^\circ$ के बँटवारे को तीन
गुना खींच देता है। रंगों की यही खिंची हुई फुहार हीरे की \emph{आग} है।

\textbf{15.} हीरा--पानी: $\sin i_c = 1.33/2.42 = 0.550$, इसलिए $i_c = 33.3^\circ$। जो $30^\circ$ वाली किरण
फलक के वायु के सामने होने पर फँस जाती थी, अब पानी की परत में निकल जाती
है ($30^\circ < 33.3^\circ$): हर बूँद, अँगुली की छाप या चिकनाई की परत बाहर का
\omterm{def:g10:refraction:index}{अपवर्तनांक} बढ़ा देती है, निकास-शंकु चौड़ा कर देती है और झिलमिल बहा ले
जाती है --- इसीलिए जौहरी का कपड़ा हमेशा साथ रहता है।

\textbf{16.} $\sin i_c = \dfrac{1.46}{1.48} = 0.986$, इसलिए $i_c = 80.6^\circ$।

\textbf{17.} \omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन} \emph{सतह पर} होता है,
इसलिए सतह का निर्दोष बना रहना ज़रूरी है: नंगे धागे पर धूल का हर कण,
खरोंच, पानी की बूँद या छूती हुई अँगुली उस बिंदु पर बाहर का \omterm{def:g10:refraction:index}{अपवर्तनांक}
बढ़ा देती है और प्रकाश रिसने लगता है। \omterm{def:g10:refraction:fiber}{आवरण} इस दर्पण को साफ़
काँच के भीतर दफ़्ना देता है, जहाँ उस तक कुछ पहुँच ही नहीं सकता --- और
बोनस में वह बाल जैसे पतले \omterm{def:g10:refraction:fiber}{क्रोड} की यांत्रिक रक्षा भी करता है।

\textbf{18.} $v = \qty{3.00e8}{m/s}/1.48 = \qty{2.03e8}{m/s}$, इसलिए
\[
t = \frac{\qty{6.0e6}{m}}{\qty{2.03e8}{m/s}} = \qty{29.6}{ms} \approx
\qty{30}{ms}.
\]

\textbf{19.} पथ-गुणक $1/\sin i_c = 1.48/1.46 = 1.0137$: बुरे से बुरे हाल में $\qty{6000}{km} \times 0.0137 \approx \qty{82}{km}$ अतिरिक्त
काँच, जिसकी क़ीमत $\qty{8.2e4}{m} / \qty{2.03e8}{m/s} \approx \qty{0.4}{ms}$ --- यानी टेढ़ी-मेढ़ी चाल लगभग मुफ़्त है।

\textbf{20.} पिंग: $2 \times (29.6 + 0.4) \approx \qty{60}{ms}$। क्षमता: इस पुस्तक की $\dfrac{10^{13}}{2.4 \times 10^8} \approx 40\,000$ प्रतियाँ
प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड}। एक वाक्य में: \omterm{def:g10:refraction:critical}{पूर्ण आंतरिक परावर्तन} से प्रकाश को
काँच के भीतर बाँधकर \omterm{thm:g10:refraction:snell}{स्नेल का नियम} हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} एक पुस्तकालय
महासागर के पार पहुँचा देता है --- आना-जाना मिलाकर साठ मिलीसेकंड में।
\end{solution}
