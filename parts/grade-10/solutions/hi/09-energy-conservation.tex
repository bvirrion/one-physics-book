\section*{अध्याय \ref{ch:g10:energy-conservation} --- ऊर्जा: रूप और संरक्षण}

\begin{solution}{exo:g10:energy-conservation:1}
$E_k = \tfrac12 \times 1200 \times 25^2 = \qty{3.75e5}{J}$। \qty{36.1}{m/s} पर: $\tfrac12 \times 1200 \times 36.1^2 \approx
\qty{7.8e5}{J}$ --- यानी $44\%$ अधिक चाल, और \omterm{def:g10:energy-conservation:kinetic}{गतिज ऊर्जा}
$(130/90)^2 \approx 2.1$ गुना: वर्ग अपना काम कर रहा है।
\end{solution}

\begin{solution}{exo:g10:energy-conservation:2}

बिजली के बराबर है।
\end{solution}

\end{solution}

= \num{3.3e5}/\num{3.6e6} \approx \qty{0.092}{kW.h}$। मनुष्य: $P = \num{1.0e7}/\num{86400} \approx \qty{116}{W}$।

\begin{solution}{exo:g10:energy-conservation:5}
डायनमो: रासायनिक (साइकिल सवार का भोजन) $\to$ गतिज (पहिया) $\to$
विद्युत (डायनमो) $\to$ विकिरण $+$ तापीय (लैंप); ऊष्मा माँसपेशियों,
टायर के संपर्क, डायनमो और लैंप पर रिसती है। गैस का हीटर: रासायनिक (गैस)
$\to$ तापीय (पानी), और रिसाव धुएँ की गैसों में। पैनल: विकिरण (सूर्य)
$\to$ विद्युत (पैनल) $\to$ रासायनिक (बैटरी), और ऊष्मा पैनल तथा चार्जर
पर रिसती है।
\end{solution}

\begin{solution}{exo:g10:energy-conservation:6}

आसपास की वायु में ले जाते हैं।
\end{solution}

\end{solution}

\begin{solution}{exo:g10:energy-conservation:11}
$v = \sqrt{2 \times 9.81 \times 0.050} \approx \qty{0.99}{m/s}$ (द्रव्यमान कट जाता है: $\tfrac12 m v^2 = mgh$)। आरंभिक $E_p = 0.20 \times 9.81 \times 0.050 \approx \qty{0.098}{J}$ अंततः वायु और धुरी
में \omterm{def:g10:energy-conservation:forms}{तापीय ऊर्जा} बन जाती है: लोलक विलगित नहीं है, और कुल --- झूला
और ऊष्मा मिलाकर --- ठीक-ठीक संरक्षित रहता है।
\end{solution}

\begin{solution}{exo:g10:energy-conservation:13}

\emph{2.} $P = \num{5e9}/\num{1e-4} = \qty{5e13}{W} = \qty{50}{TW}$ --- यानी मानवता की औसत विद्युत शक्ति का सत्रह

\end{solution}

\begin{solution}{exo:g10:energy-conservation:15}

\begin{solution}{pb:g10:energy-conservation:1}
\textbf{1.} फ़्रिज $0.1 \times 24 = \qty{2.4}{kW.h}$; हीटर $2.4 \times 2.5 = \qty{6.0}{kW.h}$; तंदूर \qty{2.0}{kW.h}; कपड़े धोने की मशीन
\qty{1.0}{kW.h}; रोशनी \qty{0.3}{kW.h}; परदे \qty{0.9}{kW.h}।
\textbf{2.} कुल \qty{12.6}{kW.h} $= 12.6 \times 3.6 \approx
\qty{45}{MJ}$: यानी ठीक एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} पेट्रोल,
\omterm{def:g10:energy-conservation:energy}{जूल} तक।
\textbf{3.} $\qty{12600}{W.h}/\qty{24}{h} = \qty{525}{W}$ --- घर एक चौथाई केतली भर की आँच पर चलता है।
\textbf{4.} पानी का हीटर (\qty{6.0}{kW.h}, यानी लगभग आधा)। \qty{1.25}{h} पर: हीटर
\qty{3.0}{kW.h}, कुल \qty{9.6}{kW.h}।
\textbf{5.} प्रति दिन $15 \times 24 = \qty{0.36}{kW.h}$, यानी \omterm{def:g10:orders-of-magnitude:unit}{मीटर} वाले कुल का लगभग $2.8\%$;
सालाना $0.36 \times 365 \approx
\qty{131}{kW.h} \approx 33$ यूरो --- और वह भी उन बत्तियों के लिए जिन्हें कोई देखता
तक नहीं।
\textbf{6.} $56604 - 56214 = \qty{390}{kW.h}$; प्रति दिन $390/30 = \qty{13.0}{kW.h}$: यानी लेखे के \qty{12.6}{kW.h} और प्रतीक्षा
अवस्था के \qty{0.36}{kW.h} --- पुष्टि हो गई।
\textbf{7.} $390 \times 0.25 + 14 = 111.50$ यूरो।
\textbf{8.} ग्रिड: $0.25/3.6 \approx 0.07$ यूरो प्रति मेगाजूल। पेट्रोल: $1.80/34 \approx 0.05$ यूरो
प्रति मेगाजूल।
\textbf{9.} कटोरा: $0.50/1.5 \approx 0.33$ यूरो प्रति मेगाजूल --- यानी ग्रिड से पाँच
गुना। सबसे सस्ते से क्रम में: पेट्रोल, ग्रिड, और फिर बहुत पीछे भोजन।
\textbf{10.} प्रति वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} $0.20 \times 1000 = \qty{200}{W}$।
\textbf{11.} प्रति वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} प्रति दिन $200 \times 4.0 = \qty{800}{W.h} = \qty{0.80}{kW.h}$।
\textbf{12.} $12.6/0.80 = 15.75 \approx \qty{16}{m^2}$: यह \qty{20}{m^2} वाली छत में समा जाता है।
\textbf{13.} $16 \times 0.2 \times 1.5 = \qty{4.8}{kW.h}$: यानी ज़रूरत का लगभग $38\%$; बाक़ी ग्रिड को (या
बेहतर महीनों में चार्ज की गई बैटरी को) देना पड़ेगा।
\textbf{14.} एक वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} प्रति वर्ष $0.80 \times 365 \approx
\qty{292}{kW.h}$ देता है; वापसी का
समय $700/292 \approx 2.4$ वर्ष --- जो पैनल के 25 वर्ष के जीवन के सामने छोटा है।
\textbf{15.} $\qty{12.6}{kW.h}/\qty{0.1}{kW} = \qty{126}{h}$: यानी दिन-रात पैडल मारते पाँच से अधिक लोग
($126/24 = 5.25$)।
\textbf{16.} भोजन $= 12.6/0.25 = \qty{50.4}{kW.h} \approx
\qty{181}{MJ}$, यानी $181/1.5 \approx 121$ अनाज के कटोरे।
\textbf{17.} $121 \times 0.50 \approx 60$ यूरो, जबकि ग्रिड से वही दिन $12.6 \times 0.25 \approx 3.15$ यूरो का पड़ता
है: पैडल की शक्ति लगभग $19$ गुना महँगी है --- और वह भी
साइकिल सवारों की मज़दूरी दिए बिना।
\textbf{18.} $5000/45 \approx 110$ घर-दिन। पर \qty{100}{\micro s} तक $\num{5e9}/\num{1e-4} = \qty{5e13}{W}$, वह भी किसी अनजान
जगह: कोई \omterm{def:g10:energy-conservation:transfer}{रूपांतरक} या बैटरी इस फ़व्वारे से पानी नहीं पी सकती।
घर को एक स्थिर \qty{525}{W} बहाव चाहिए, झटका नहीं।
\textbf{19.} \omterm{def:g10:energy-conservation:energy}{जूल} नहीं --- वे तो संरक्षित और एक जैसे होते हैं।
आप उनके \emph{रूप और उपलब्धता} का दाम चुकाते हैं: जो ऊर्जा सघन,
संचित करने योग्य और माँगते ही हाज़िर हो, उसकी क़ीमत में वह पूरी
\omterm{def:g10:energy-conservation:transfer}{रूपांतरण शृंखला} शामिल है जिसने उसे ऐसा बनाया।
\textbf{20.} परिवार का एक दिन $= \qty{12.6}{kW.h} \approx
\qty{45}{MJ}$: यानी एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम}
पेट्रोल, धूप वाली छत के सोलह वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर}, या चौबीसों घंटे चलते
पाँच साइकिल सवार। भंडार ऊर्जा है (\omterm{def:g10:energy-conservation:energy}{जूल},
किलोवाट-घंटे); बहाव शक्ति है (\omterm{def:g10:energy-conservation:power}{वाट}) --- बाथटब
और टोंटी, जिन्हें अब कभी गड्डमड्ड नहीं करना।
\end{solution}
