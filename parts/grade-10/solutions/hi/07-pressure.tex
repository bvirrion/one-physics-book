\section*{अध्याय \ref{ch:g10:pressure} --- दाब: खेल से गोताख़ोरी तक}

\begin{solution}{exo:g10:pressure:1}
बड़ा फलक: $S = \qty{0.15}{m^2}$, $P = 600/0.15 = \qty{4.0}{kPa}$। छोटा फलक: $S = \qty{0.060}{m^2}$, $P = \qty{10}{kPa}$। रेत पर बड़ा फलक बेहतर
है: कम दाब, कम धँसना।
\end{solution}

\begin{solution}{exo:g10:pressure:2}
$\qty{0.25}{MPa} = \qty{250}{kPa} = \qty{2.5e5}{Pa}$। $3.0/\num{2.0e-4} = \qty{15}{kPa}$ $60/0.050 = \qty{1.2}{kPa}$ पर भारी पड़ता है: छोटा बल अधिक ज़ोर से दबता
है।
\end{solution}

\begin{solution}{exo:g10:pressure:3}
$\rho g h = 1000 \times 9.81 \times 3.0 \approx \qty{29}{kPa}$; \omterm{met:g10:pressure:dive}{निरपेक्ष} $P = 101 + 29 = \qty{130}{kPa}$।
\end{solution}

\begin{solution}{exo:g10:pressure:4}
$V_2 = 100 \times 4.0 / 250 = \qty{1.6}{L}$; \qty{50}{kPa} पर $V_2 = 400/50 = \qty{8.0}{L}$।
\end{solution}

\begin{solution}{exo:g10:pressure:5}
$F = 65 \times 9.81 \approx \qty{638}{N}$। ब्लेड: $638/\num{1.2e-3} \approx \qty{5.3e5}{Pa} = \qty{530}{kPa}$। तलवे: $638/0.036 \approx \qty{18}{kPa}$। अनुपात $\approx 30$: ब्लेड बर्फ़ में गड़
जाता है, तलवे नहीं।
\end{solution}

\begin{solution}{exo:g10:pressure:6}
अँगूठा: $20/\num{1.2e-4} \approx \qty{1.7e5}{Pa} = \qty{170}{kPa}$। दीवार: $20/\num{1.0e-8} = \qty{2.0e9}{Pa}$। वही \qty{20}{N} नोक पर $12\,000$ गुना छोटे
क्षेत्रफल पर लगता है, इसलिए दाब केवल वहीं उससे आगे निकलता है
जितना पदार्थ सह सकता है।
\end{solution}

\begin{solution}{exo:g10:pressure:7}
$\Delta P = \rho g h = 1025 \times 9.81 \times 30 \approx
\qty{3.0e5}{Pa}$, इसलिए $F = \num{3.0e5} \times 0.50 \approx \qty{1.5e5}{N}$ --- यानी $F/g \approx \qty{1.5e4}{kg}$ का भार, पंद्रह टन। कोई भी
चालक दल उसे समुद्र के विरुद्ध बाहर की ओर नहीं खोल सकता।
\end{solution}

\begin{solution}{exo:g10:pressure:8}
$\rho = \dfrac{P}{g h} = \dfrac{\num{66.2e3}}{9.81 \times 5.00}
\approx \qty{1350}{kg/m^3}$। मीठा पानी नहीं ($\qty{1000}{kg/m^3}$): कहीं अधिक सघन कोई द्रव, जैसे गाढ़ा
नमकीन पानी।
\end{solution}

\begin{solution}{exo:g10:pressure:9}
$h = P_0/(\rho g)$: पानी $\num{1.01e5}/9810 \approx \qty{10.3}{m}$; पारा $\num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$। \qty{76}{cm} की नली मेज़ पर समा जाती है;
\qty{10}{m} ऊँचे पानी वाले बैरोमीटर को सीढ़ी-घर चाहिए --- पारे का घनत्व
जीत गया।
\end{solution}

\begin{solution}{exo:g10:pressure:10}
$V = 6000/240 = \qty{25}{mL}$; $P = 6000/75 = \qty{80}{kPa}$। $V$ के सामने $P$ एक अतिपरवलय है; $1/V$ के सामने
$P$ मूल बिंदु से जाती एक सरल रेखा है, जिसकी ढाल \qty{6000}{kPa.mL} है।
\end{solution}

\begin{solution}{exo:g10:pressure:11}
\qty{20}{m} पर: $P = 101 + 2 \times 101 = \qty{303}{kPa}$, यानी तीन वायुमंडल। सतह पर $V = 0.50 \times 303/101 = \qty{1.5}{cm^3}$: बुलबुला तिगुना हो
जाता है।
\end{solution}

\begin{solution}{exo:g10:pressure:12}
\emph{1.} $F = \num{1.01e5} \times \num{1.2e-3} \approx \qty{121}{N}$, जो $m = 121/9.81 \approx \qty{12}{kg}$ थामे रख सकता है।

\emph{2.} $\Delta P = 101 - 20 = \qty{81}{kPa}$: $F = \num{8.1e4} \times \num{1.2e-3} \approx \qty{97}{N}$, यानी लगभग \qty{9.9}{kg}।
\end{solution}

\begin{solution}{exo:g10:pressure:13}
बॉयल: सतह के दाब वाली $12 \times (200 - 50) = \qty{1800}{L}$ वायु काम में आ सकती है। \qty{30}{m} पर
हर साँस \qty{4}{bar} पर ली जाती है, इसलिए आसपास के दाब पर \qty{15}{L/min} की
क़ीमत सतह वाली $15 \times 4 = \qty{60}{L/min}$ वायु पड़ती है: $1800/60 =
\qty{30}{min}$। \qty{10}{m} पर ($\qty{2}{bar}$): \qty{30}{L/min},
इसलिए \qty{60}{min}। गहराई की क़ीमत वायु में चुकानी पड़ती है।
\end{solution}

\begin{solution}{exo:g10:pressure:14}
$P = P_0 \dfrac{V_0}{V} = 101 \times \dfrac{6.0}{1.5} =
\qty{404}{kPa}$। $\rho g \approx \qty{10.1}{kPa/m}$ लेने पर $h = (404 - 101)/10.1 \approx \qty{30}{m}$।
\end{solution}

\begin{solution}{exo:g10:pressure:15}
\emph{1.} भार $F = 1200 \times 9.81 \approx \qty{1.18e4}{N}$; दाब $P = F/S = \num{1.18e4}/0.040 \approx \qty{2.9e5}{Pa}$; छोटा पिस्टन: $f = P s = \num{2.9e5} \times \num{2.0e-4} \approx \qty{59}{N}$
--- यानी बल $S/s = 200$ से भाग हो जाता है।

\emph{2.} द्रव का आयतन संरक्षित रहता है: $400 \times 2.0 = \qty{800}{cm^3}$ छोटे सिलिंडर से ही आना
चाहिए, जो $800/2.0 = \qty{400}{cm} = \qty{4.0}{m}$ चलता है। कार्य: एक ओर $59 \times 4.0 \approx \qty{235}{J}$, दूसरी ओर $\num{1.18e4} \times 0.020 \approx \qty{235}{J}$ --- लिफ़्ट
दूरी के बदले बल देती है, कार्य कभी नहीं।
\end{solution}

\begin{solution}{pb:g10:pressure:1}
\textbf{1.} प्रति \omterm{def:g10:orders-of-magnitude:unit}{मीटर} $\rho g = 1025 \times 9.81 \approx \qty{10.1}{kPa}$, इसलिए समुद्री पानी के \qty{10}{m}
$\approx \qty{101}{kPa}$ जोड़ते हैं: हर दस \omterm{def:g10:orders-of-magnitude:unit}{मीटर} पर एक वायुमंडल, लगभग ठीक-ठीक।

\textbf{2.} $P = 101 + 10.1\,h$: \qty{10}{m} पर \qty{202}{kPa}, \qty{20}{m} पर \qty{302}{kPa}, \qty{30}{m} पर
\qty{403}{kPa}, \qty{40}{m} पर \qty{503}{kPa}।

\textbf{3.} $P_0$ के $2.0$, $3.0$, $4.0$ और $5.0$ गुना।

\textbf{4.} $\qty{0.015}{m^2}$ पर $\Delta P = 403 - 101 = \qty{302}{kPa}$: $F = \num{3.02e5} \times 0.015 \approx
\qty{4.5e3}{N}$ --- यानी चेहरे पर लगभग आधे टन
का भार।

\textbf{5.} $\Delta P = 10.1 \times 3.0 \approx \qty{30}{kPa}$; $F = \num{3.0e4} \times \num{6.0e-5} \approx \qty{1.8}{N}$ --- यानी तरणताल भर की गहराई पर ही कर्णपटह
पर दबी हुई एक अँगुली। दाब बराबर करने से मध्य कान में वायु भरती है, जब
तक भीतर का दाब पानी के दाब से मेल न खा ले।

\textbf{6.} $V = 6.0 \, P_0/P$: $6.0 \times 101/202 = \qty{3.0}{L}$; $606/302 = \qty{2.0}{L}$; $606/403 = \qty{1.5}{L}$।

\textbf{7.} आधा तब होता है जब $P = 2P_0$, यानी \qty{10}{m} पर --- यानी गोते के
बिलकुल पहले हिस्से में ही।

\textbf{8.} नहीं: $V = 606/P$ \omterm{met:g10:pressure:dive}{निरपेक्ष दाब} में एक अतिपरवलय है (और
\omterm{prop:g10:pressure:boyle}{बॉयल का नियम} $P$ देखता है, $h$ नहीं); हर अगला \qty{10}{m}
पिछले से कम आयतन घटाता है।

\textbf{9.} $V = \qty{1.5}{L}$ के लिए $P = 101 \times 6.0/1.5 = \qty{404}{kPa}$ चाहिए, यानी $h = 303/10.1 \approx \qty{30}{m}$ (\cref{exo:g10:pressure:14})।

\textbf{10.} रक्त: प्लाज़्मा छाती की वाहिकाओं में आ जाता है, जो असंपीड्य
है, और वही आयतन घेर लेता है जो वायु अब नहीं भरती।

\textbf{11.} आसपास के दाब पर \qty{403}{kPa}। बॉयल से वही वायु \qty{101}{kPa}
पर $4$ गुना आयतन घेरती, इसलिए वह सतह की वायु से $4.0$ गुना सघन
है।

\textbf{12.} $V = 6.0 \times 403/202 \approx \qty{12}{L}$ --- यानी केवल \qty{20}{m} की चढ़ाई के बाद ही दुगुना।

\textbf{13.} $V = 6.0 \times 403/101 \approx \qty{24}{L}$: यानी उसके फेफड़ों की क्षमता का चार गुना।

\textbf{14.} \emph{ऊपर आते समय साँस कभी मत रोकिए।} साँस-मार्ग खुला हो तो
फैलती वायु रेगुलेटर से बाहर बह जाती है, और फेफड़ों का आयतन \qty{6.0}{L} से
कभी आगे नहीं जाता।

\textbf{15.} \qty{30}{m} पर लीना के \qty{1.5}{L} \emph{वही} सतह वाले \qty{6.0}{L} हैं,
बस दबे हुए: ऊपर आते समय वे ठीक \qty{6.0}{L} तक ही फिर फैलते हैं, उससे आगे
कभी नहीं। मार्को के \qty{6.0}{L} तो \qty{403}{kPa} पर \emph{लिए} गए थे: उनमें सतह
वाली \qty{24}{L} वायु भरी है।

\textbf{16.} $V = 1.0 \times 503/P$: \qty{30}{m} पर \qty{1.25}{cm^3}, \qty{20}{m} पर \qty{1.67}{cm^3}, \qty{10}{m} पर
\qty{2.49}{cm^3}, और सतह के ठीक नीचे \qty{4.98}{cm^3}।

\textbf{17.} चरणवार गुणक: $503/403 = 1.25$, फिर $1.33$, $1.49$, और आख़िरी दस
\omterm{def:g10:orders-of-magnitude:unit}{मीटर} के लिए $202/101 = 2.0$ --- यानी सतह पास आते-आते बढ़त तेज़ होती
जाती है।

\textbf{18.} $40/10 = \qty{4.0}{min}$। आख़िरी मिनट: \qty{10}{m} से सतह तक \omterm{met:g10:pressure:dive}{निरपेक्ष
दाब} आधा रह जाता है, जो पूरी चढ़ाई की सबसे बड़ी सापेक्ष गिरावट है।

\textbf{19.} तेज़ चढ़ाई घुली हुई नाइट्रोजन को रक्त और जोड़ों के भीतर ही
बुलबुलों में फूट पड़ने देती है (विसंपीडन रोग); धीरे-धीरे ऊपर आने और
\qty{5}{m} के आसपास ठहरने से गैस इतनी देर घुली रहती है कि चुपचाप फेफड़ों से
निकल जाए।

\textbf{20.} मुक्त गोताख़ोर: उसकी वायु केवल अपने मूल आयतन तक लौटती है ---
बॉयल उसकी रक्षा करता है। स्कूबा गोताख़ोर: गहराई पर ली गई वायु ऊपर आते
समय कई गुना हो जाती है --- साँस लेते रहिए, रोकिए मत। और आख़िरी दस
\omterm{def:g10:orders-of-magnitude:unit}{मीटर} हर बंद आयतन को दुगुना कर देते हैं ($\times 2.0$, जबकि नीचे
\qty{40}{m} पर $\times 1.25$): सतह जितनी पास हो, उसके उतने ही धीरे पास जाइए।
\end{solution}
