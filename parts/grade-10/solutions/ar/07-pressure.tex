\section*{الفصل \ref{ch:g10:pressure} --- الضغط: من الرياضة إلى الغوص}

\begin{solution}{exo:g10:pressure:1}
الوجه الكبير: $S = \qty{0.15}{m^2}$، $P = 600/0.15 = \qty{4.0}{kPa}$.
والوجه الصغير: $S = \qty{0.060}{m^2}$، $P = \qty{10}{kPa}$. وعلى الرمل، الوجه
الكبير: \omterm{def:g10:pressure:pressure}{ضغط} أقلّ، وغوص أقلّ.
\end{solution}

\begin{solution}{exo:g10:pressure:2}
$\qty{0.25}{MPa} = \qty{250}{kPa} = \qty{2.5e5}{Pa}$.
و$3.0/\num{2.0e-4} = \qty{15}{kPa}$ يغلب
$60/0.050 = \qty{1.2}{kPa}$: \omterm{def:g10:inertia:force}{فالقوة} الصغيرة تضغط أشدّ.
\end{solution}

\begin{solution}{exo:g10:pressure:3}
$\rho g h = 1000 \times 9.81 \times 3.0 \approx \qty{29}{kPa}$؛
\omterm{met:g10:pressure:dive}{والمطلق} $P = 101 + 29 = \qty{130}{kPa}$.
\end{solution}

\begin{solution}{exo:g10:pressure:4}
$V_2 = 100 \times 4.0 / 250 = \qty{1.6}{L}$؛
وعند \qty{50}{kPa}، $V_2 = 400/50 = \qty{8.0}{L}$.
\end{solution}

\begin{solution}{exo:g10:pressure:5}
$F = 65 \times 9.81 \approx \qty{638}{N}$. النصل:
$638/\num{1.2e-3} \approx \qty{5.3e5}{Pa} = \qty{530}{kPa}$. والنعلان:
$638/0.036 \approx \qty{18}{kPa}$. والنسبة $\approx 30$: فالنصل يعضّ
الجليد، والنعلان لا يفعلان.
\end{solution}

\begin{solution}{exo:g10:pressure:6}
الإبهام: $20/\num{1.2e-4} \approx \qty{1.7e5}{Pa} = \qty{170}{kPa}$.
والجدار: $20/\num{1.0e-8} = \qty{2.0e9}{Pa}$. \omterm{def:g10:inertia:force}{فالقوة} \qty{20}{N} نفسها تؤثر على
مساحة أصغر $12\,000$ مرة عند السنّ، ولذلك لا يتجاوز \omterm{def:g10:pressure:pressure}{الضغط}
ما تحتمله المادة إلا هناك.
\end{solution}

\begin{solution}{exo:g10:pressure:7}
$\Delta P = \rho g h = 1025 \times 9.81 \times 30 \approx
\qty{3.0e5}{Pa}$، ومنه
$F = \num{3.0e5} \times 0.50 \approx \qty{1.5e5}{N}$ --- أي \omterm{def:g10:universal-gravitation:weight}{وزن}
$F/g \approx \qty{1.5e4}{kg}$، خمسة عشر طنًّا. فلا طاقم يفتح تلك الكوّة
إلى الخارج في وجه البحر.
\end{solution}

\begin{solution}{exo:g10:pressure:8}
$\rho = \dfrac{P}{g h} = \dfrac{\num{66.2e3}}{9.81 \times 5.00}
\approx \qty{1350}{kg/m^3}$. وليس ماءً عذبًا ($\qty{1000}{kg/m^3}$):
بل سائل أكثف بوضوح، كمحلول ملحي مركّز\ مثلًا.
\end{solution}

\begin{solution}{exo:g10:pressure:9}
$h = P_0/(\rho g)$: الماء
$\num{1.01e5}/9810 \approx \qty{10.3}{m}$؛ والزئبق
$\num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$. فأنبوب طوله
\qty{76}{cm} يسع طاولة مكتب؛ أما بارومتر مائي طوله \qty{10}{m} فيحتاج
بئر سلّم --- فقد فازت الكتلة الحجمية للزئبق.
\end{solution}

\begin{solution}{exo:g10:pressure:10}
$V = 6000/240 = \qty{25}{mL}$؛ $P = 6000/75 = \qty{80}{kPa}$.
و$P$ بدلالة $V$ قطع زائد؛ أما $P$ بدلالة $1/V$ فمستقيم
يمرّ بالمبدأ ميله \qty{6000}{kPa.mL}.
\end{solution}

\begin{solution}{exo:g10:pressure:11}
عند \qty{20}{m}: $P = 101 + 2 \times 101 = \qty{303}{kPa}$، أي ثلاثة
جووّ. وعند السطح،
$V = 0.50 \times 303/101 = \qty{1.5}{cm^3}$: فتتضاعف الفقاعة ثلاث مرات.
\end{solution}

\begin{solution}{exo:g10:pressure:12}
\emph{1.} $F = \num{1.01e5} \times \num{1.2e-3} \approx \qty{121}{N}$،
فيحمل $m = 121/9.81 \approx \qty{12}{kg}$.

\emph{2.} $\Delta P = 101 - 20 = \qty{81}{kPa}$:
$F = \num{8.1e4} \times \num{1.2e-3} \approx \qty{97}{N}$، أي نحو
\qty{9.9}{kg}.
\end{solution}

\begin{solution}{exo:g10:pressure:13}
بويل: $12 \times (200 - 50) = \qty{1800}{L}$ من هواء \omterm{def:g10:pressure:pressure}{ضغط} السطح
قابلة للاستعمال. وعلى \qty{30}{m} يُسحب كل نفَس عند \qty{4}{bar}، ومن ثم تكلّف
\qty{15}{L/min} عند \omterm{def:g10:pressure:pressure}{الضغط} المحيط
$15 \times 4 = \qty{60}{L/min}$ من هواء السطح: أي $1800/60 =
\qty{30}{min}$. وعلى \qty{10}{m} ($\qty{2}{bar}$): \qty{30}{L/min}،
ومنه \qty{60}{min}. فالعمق يُدفع ثمنه هواءً.
\end{solution}

\begin{solution}{exo:g10:pressure:14}
$P = P_0 \dfrac{V_0}{V} = 101 \times \dfrac{6.0}{1.5} =
\qty{404}{kPa}$. وبأخذ $\rho g \approx \qty{10.1}{kPa/m}$،
$h = (404 - 101)/10.1 \approx \qty{30}{m}$.
\end{solution}

\begin{solution}{exo:g10:pressure:15}
\emph{1.} \omterm{def:g10:universal-gravitation:weight}{الوزن} $F = 1200 \times 9.81 \approx \qty{1.18e4}{N}$؛
\omterm{def:g10:pressure:pressure}{والضغط} $P = F/S = \num{1.18e4}/0.040 \approx \qty{2.9e5}{Pa}$؛ والمكبس
الصغير: $f = P s = \num{2.9e5} \times \num{2.0e-4} \approx \qty{59}{N}$
--- أي أن \omterm{def:g10:inertia:force}{القوة} تُقسم على $S/s = 200$.

\emph{2.} حجم السائل محفوظ:
فلا بد أن يأتي $400 \times 2.0 = \qty{800}{cm^3}$ من الأسطوانة الصغيرة،
التي تنتقل $800/2.0 = \qty{400}{cm} = \qty{4.0}{m}$. والشغل:
$59 \times 4.0 \approx \qty{235}{J}$ على جانب،
و$\num{1.18e4} \times 0.020 \approx \qty{235}{J}$ على الآخر --- فالرافعة
تقايض المسافة \omterm{def:g10:inertia:force}{بالقوة}، ولا تقايض الشغل أبدًا.
\end{solution}

\begin{solution}{pb:g10:pressure:1}
\textbf{1.} $\rho g = 1025 \times 9.81 \approx \qty{10.1}{kPa}$ لكل
\omterm{def:g10:orders-of-magnitude:unit}{متر}، ومنه تضيف \qty{10}{m} من ماء البحر $\approx \qty{101}{kPa}$: أي جوًّا
واحدًا لكل عشرة أمتار، تقريبًا بالضبط.

\textbf{2.} $P = 101 + 10.1\,h$: \qty{202}{kPa} على \qty{10}{m}،
و\qty{302}{kPa} على \qty{20}{m}، و\qty{403}{kPa} على \qty{30}{m}،
و\qty{503}{kPa} على \qty{40}{m}.

\textbf{3.} $2.0$ و$3.0$ و$4.0$ و$5.0$ أضعاف $P_0$.

\textbf{4.} $\Delta P = 403 - 101 = \qty{302}{kPa}$ على
$\qty{0.015}{m^2}$: $F = \num{3.02e5} \times 0.015 \approx
\qty{4.5e3}{N}$ --- أي \omterm{def:g10:universal-gravitation:weight}{وزن} قرابة نصف طنّ على الوجه.

\textbf{5.} $\Delta P = 10.1 \times 3.0 \approx \qty{30}{kPa}$؛
$F = \num{3.0e4} \times \num{6.0e-5} \approx \qty{1.8}{N}$ --- أي إصبع
مضغوط على طبلة الأذن، وعلى عمق مسبح لا غير. أما الموازنة فتدفع
هواءً إلى الأذن الوسطى حتى يوافق \omterm{def:g10:pressure:pressure}{الضغط} الداخلي \omterm{def:g10:pressure:pressure}{ضغط} الماء.

\textbf{6.} $V = 6.0 \, P_0/P$: $6.0 \times 101/202 = \qty{3.0}{L}$؛
$606/302 = \qty{2.0}{L}$؛ $606/403 = \qty{1.5}{L}$.

\textbf{7.} ينتصف حين $P = 2P_0$، أي\ عند \qty{10}{m} --- أي في
أول مقطع من الغوصة.

\textbf{8.} لا: إذ إنّ $V = 606/P$ قطع زائد في \omterm{met:g10:pressure:dive}{الضغط المطلق}
(و$P$، لا $h$، هو ما يراه \omterm{prop:g10:pressure:boyle}{قانون بويل})؛ فكل
\qty{10}{m} إضافية تنزع حجمًا أقلّ من سابقتها.

\textbf{9.} يقتضي $V = \qty{1.5}{L}$
$P = 101 \times 6.0/1.5 = \qty{404}{kPa}$، أي\
$h = 303/10.1 \approx \qty{30}{m}$ (\cref{exo:g10:pressure:14}).

\textbf{10.} الدم: إذ تنتقل البلازما إلى أوعية الصدر،
وهي غير قابلة للانضغاط، فتشغل الحجم الذي لم يعد الهواء يملؤه.

\textbf{11.} عند \omterm{def:g10:pressure:pressure}{الضغط} المحيط، \qty{403}{kPa}. وبحسب بويل فإن الهواء نفسه
عند \qty{101}{kPa} كان سيملأ $4$ أضعاف الحجم: أي أنه أكثف $4.0$
مرة من هواء السطح.

\textbf{12.} $V = 6.0 \times 403/202 \approx \qty{12}{L}$ --- أي الضعف،
بعد \qty{20}{m} من الصعود لا غير.

\textbf{13.} $V = 6.0 \times 403/101 \approx \qty{24}{L}$: أي أربعة أضعاف
سعة رئتيه.

\textbf{14.} \emph{لا تحبس نفَسك أبدًا أثناء الصعود.} فمع بقاء
المجرى الهوائي مفتوحًا، يخرج الهواء المتمدّد عبر المنظّم، ولا
يتجاوز حجم الرئتين \qty{6.0}{L} أبدًا.

\textbf{15.} \qty{1.5}{L} لينا على \qty{30}{m} \emph{هي} \qty{6.0}{L}
عند السطح مضغوطة: فتتمدد في طريق الصعود لتعود إلى \qty{6.0}{L}
بالضبط، لا أكثر. أما \qty{6.0}{L} ماركو فقد \emph{أُخذت} على
\qty{403}{kPa}: ففيها \qty{24}{L} من هواء السطح.

\textbf{16.} $V = 1.0 \times 503/P$: \qty{1.25}{cm^3} على \qty{30}{m}،
و\qty{1.67}{cm^3} على \qty{20}{m}، و\qty{2.49}{cm^3} على \qty{10}{m}،
و\qty{4.98}{cm^3} تحت السطح.

\textbf{17.} عوامل المراحل: $503/403 = 1.25$، ثم $1.33$ و$1.49$،
و$202/101 = 2.0$ للأمتار العشرة الأخيرة --- فالنمو يتسارع كلما
اقترب السطح.

\textbf{18.} $40/10 = \qty{4.0}{min}$. أما الدقيقة الأخيرة: فمن
\qty{10}{m} إلى السطح ينتصف \omterm{met:g10:pressure:dive}{الضغط المطلق}، وهو أكبر
هبوط نسبي في الصعود كله.

\textbf{19.} الصعود السريع يدع النيتروجين المذاب يفور
فقاعات داخل الدم والمفاصل (داء تخفيف \omterm{def:g10:pressure:pressure}{الضغط})؛ أما الصعود
البطيء والتوقف قرب \qty{5}{m} فيبقيان الغاز مذابًا مدة تكفي
ليغادر بهدوء عبر الرئتين.

\textbf{20.} الغطّاسة الحرّة: هواؤها لا يعود إلا إلى حجمه الأصلي
--- فبويل يحميها. والغطّاس بالأسطوانة: الهواء المأخوذ في العمق يتضاعف
في طريق الصعود --- فتنفّس، ولا تحبس أبدًا. والأمتار العشرة الأخيرة تضاعف
كل حجم محبوس ($\times 2.0$، مقابل $\times 1.25$ في العمق على
\qty{40}{m}): فكلما اقترب السطح وجب أن تقترب منه أبطأ.
\end{solution}
