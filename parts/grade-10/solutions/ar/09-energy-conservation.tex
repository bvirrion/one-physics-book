\section*{الفصل \ref{ch:g10:energy-conservation} --- الطاقة: أشكالها وانحفاظها}

\begin{solution}{exo:g10:energy-conservation:1}
$E_k = \tfrac12 \times 1200 \times 25^2 = \qty{3.75e5}{J}$. وعند
\qty{36.1}{m/s}: $\tfrac12 \times 1200 \times 36.1^2 \approx
\qty{7.8e5}{J}$ --- أي $44\%$ من السرعة الزائدة، و$(130/90)^2 \approx 2.1$
أضعاف \omterm{def:g10:energy-conservation:kinetic}{الطاقة الحركية}: فالتربيع في العمل.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:2}
$\Delta h = \qty{450}{m}$:
$E_p = 72 \times 9.81 \times 450 \approx \qty{3.2e5}{J}
= \num{3.2e5}/\num{3.6e6} \approx \qty{0.088}{kW.h}$، وتساوي نحو
$0.022$ يورو. فيوم من التسلق سنتان من الكهرباء.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:3}
$\qty{1}{kW.h} = \qty{3.6e6}{J}$؛
$\qty{12}{W.h} = 12 \times \qty{3600}{J} \approx \qty{43}{kJ}$؛
$\qty{45}{MJ} = 45/3.6 = \qty{12.5}{kW.h}$؛
$\qty{10}{MJ} = 10/3.6 \approx \qty{2.8}{kW.h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:4}
$E = 2200 \times 150 = \qty{3.3e5}{J}
= \num{3.3e5}/\num{3.6e6} \approx \qty{0.092}{kW.h}$. والإنسان:
$P = \num{1.0e7}/\num{86400} \approx \qty{116}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:5}
المولّد: كيميائية (غذاء راكب الدراجة) $\to$ حركية (العجلة) $\to$
كهربائية (المولّد) $\to$ إشعاعية $+$ حرارية (المصباح)؛ وتتسرّب الحرارة
عند العضلات وتماس الإطار والمولّد والمصباح. وسخّان الغاز: كيميائية (الغاز)
$\to$ حرارية (الماء)، والتسرّب في غازات المدخنة. واللوح: إشعاعية
(الشمس) $\to$ كهربائية (اللوح) $\to$ كيميائية (البطارية)، وتتسرّب الحرارة
عند اللوح والشاحن.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:6}
المستهلكة $2.0 \times 45 = \qty{90}{MJ}$؛ $\eta = 27/90 = 0.30$.
والحرارة: $90 - 27 = \qty{63}{MJ}$، تحملها غازات العادم
والمشعّ إلى الهواء المحيط.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:7}
$E_k = E_p = 70 \times 9.81 \times 20 \approx \qty{1.37e4}{J}$؛
$v = \sqrt{2 \times 13734/70} = \sqrt{392} \approx \qty{19.8}{m/s}
\approx \qty{71}{km/h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:8}
$41562 - 41236 = \qty{326}{kW.h}$؛ واليومي $326/30 \approx
\qty{10.9}{kW.h}$. والفاتورة: $326 \times 0.26 + 12 = 96.76$ يورو.
\omterm{def:g10:energy-conservation:power}{والاستطاعة} المتوسطة: $\qty{326}{kW.h}/\qty{720}{h} \approx \qty{453}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:9}
\emph{1.} الضوء: $0.05 \times 60 = \qty{3.0}{W}$؛ وLED:
$\eta = 3/9 \approx 0.33$.
\emph{2.} في السنة ($\qty{1460}{h}$): المصباح
$60 \times 1460 = \qty{87.6}{kW.h}$، وLED $\qty{13.1}{kW.h}$؛
والوفر $\qty{74.5}{kW.h} \approx 19$ يورو لكل مصباح في السنة.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:10}

\emph{2.} $0.90 \times 59 \approx \qty{53}{MW}$؛

\begin{solution}{exo:g10:energy-conservation:11}
$v = \sqrt{2 \times 9.81 \times 0.050} \approx \qty{0.99}{m/s}$
(فتتلاشى الكتلة: $\tfrac12 m v^2 = mgh$). أما \omterm{def:g10:energy-conservation:energy}{الطاقة} الابتدائية
$E_p = 0.20 \times 9.81 \times 0.050 \approx \qty{0.098}{J}$ فتنتهي
طاقةً حرارية في الهواء وفي المحور: فالنوّاس ليس معزولًا،
والمجموع --- التأرجح مع الحرارة --- محفوظ
بالضبط.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:12}
\emph{1.} $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.304$؛
والمستهلكة $12/0.304 \approx \qty{39.5}{W.h}
= 39.5 \times 3600 \approx \qty{142}{kJ}$.
\emph{2.} $\num{1.42e5}/\num{2.5e7} \approx \qty{5.7}{g}$ من الفحم
لكل شحنة؛ و$\times 365 \approx \qty{2.1}{kg}$ في السنة.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:13}
\emph{1.} $\num{5e9}/\num{3.6e6} \approx \qty{1400}{kW.h}$؛
أي $1400/12.6 \approx 110$ يومًا.
\emph{2.} $P = \num{5e9}/\num{1e-4} = \qty{5e13}{W} = \qty{50}{TW}$
--- أي سبعة عشر ضعف \omterm{def:g10:energy-conservation:power}{الاستطاعة} الكهربائية المتوسطة للبشرية، مدة
\qty{100}{\micro s}، وفي مكان لا يُتنبأ به. فالبرق \omterm{def:g10:energy-conservation:power}{استطاعة} كبيرة
\omterm{def:g10:energy-conservation:energy}{وطاقة} صغيرة، تُسلَّم بسرعة عديمة الفائدة: فلا \omterm{def:g10:energy-conservation:transfer}{محوّل}
يقبل نتوءًا كهذا.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:14}
\emph{1.} $E_k = \tfrac12 \times 1300 \times 30.6^2 \approx
\qty{6.1e5}{J}$، محوَّلة إلى \omterm{def:g10:energy-conservation:forms}{طاقة حرارية} في أقراص الكبح
وقماشاته (ثم في الهواء).
\emph{2.} $0.60 \times \qty{6.1e5}{J} \approx \qty{3.7e5}{J}$ لكل
توقف؛ أي $\num{3.6e6}/\num{3.7e5} \approx 10$ توقفًا لكل \omterm{def:g10:energy-conservation:kwh}{كيلوواط ساعة}.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:15}
$E_p = 85 \times 9.81 \times 1200 \approx \qty{1.0}{MJ}$؛ والغذاء:
$1.0/0.25 = \qty{4.0}{MJ}$، أي\ $4.0/1.5 \approx 2.7$ صحن
حبوب --- أو $4.0/45 \approx \qty{0.09}{kg}$ من البنزين. فصباح كامل
من الجهد الصادق يسع في فنجان قهوة من الوقود: \omterm{def:g10:energy-conservation:forms}{فالطاقة

\end{solution}

\begin{solution}{pb:g10:energy-conservation:1}
\textbf{1.} الثلّاجة $0.1 \times 24 = \qty{2.4}{kW.h}$؛ والسخّان
$2.4 \times 2.5 = \qty{6.0}{kW.h}$؛ والفرن \qty{2.0}{kW.h}؛ والغسّالة
\qty{1.0}{kW.h}؛ والإضاءة \qty{0.3}{kW.h}؛ والشاشات
\qty{0.9}{kW.h}.
\textbf{2.} المجموع \qty{12.6}{kW.h} $= 12.6 \times 3.6 \approx
\qty{45}{MJ}$: أي \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} واحد من البنزين، حتى \omterm{def:g10:energy-conservation:energy}{الجول}.
\textbf{3.} $\qty{12600}{W.h}/\qty{24}{h} = \qty{525}{W}$ --- فالبيت
يخمل عند ربع غلاية.
\textbf{4.} سخّان الماء (\qty{6.0}{kW.h}، أي قرابة النصف). وعند
\qty{1.25}{h}: السخّان \qty{3.0}{kW.h}، والمجموع \qty{9.6}{kW.h}.
\textbf{5.} $15 \times 24 = \qty{0.36}{kW.h}$ في اليوم، أي نحو
$2.8\%$ من المجموع المقيس؛ وفي السنة $0.36 \times 365 \approx
\qty{131}{kW.h} \approx 33$ يورو --- من أجل أضواء لا يشاهدها أحد.
\textbf{6.} $56604 - 56214 = \qty{390}{kW.h}$؛
و$390/30 = \qty{13.0}{kW.h}$ في اليوم: أي \qty{12.6}{kW.h} في التدقيق
زائدًا \qty{0.36}{kW.h} لوضع الاستعداد --- وقد تأكد.
\textbf{7.} $390 \times 0.25 + 14 = 111.50$ يورو.
\textbf{8.} الشبكة: $0.25/3.6 \approx 0.07$ يورو للميغاجول.
والبنزين: $1.80/34 \approx 0.05$ يورو للميغاجول.
\textbf{9.} الصحن: $0.50/1.5 \approx 0.33$ يورو للميغاجول ---
أي خمسة أضعاف الشبكة. والأرخص أولًا: البنزين، ثم الشبكة، ثم الغذاء
متخلفًا بفارق كبير.
\textbf{10.} $0.20 \times 1000 = \qty{200}{W}$ لكل \omterm{def:g10:orders-of-magnitude:unit}{متر} مربع.
\textbf{11.} $200 \times 4.0 = \qty{800}{W.h} = \qty{0.80}{kW.h}$
لكل \omterm{def:g10:orders-of-magnitude:unit}{متر} مربع في اليوم.
\textbf{12.} $12.6/0.80 = 15.75 \approx \qty{16}{m^2}$: فتسع
سطح \qty{20}{m^2}.
\textbf{13.} $16 \times 0.2 \times 1.5 = \qty{4.8}{kW.h}$: أي نحو
$38\%$ من الحاجة؛ وعلى الشبكة (أو بطارية شُحنت في أشهر
أفضل) أن تمدّ الباقي.
\textbf{14.} يعطي \omterm{def:g10:orders-of-magnitude:unit}{المتر} المربع الواحد $0.80 \times 365 \approx
\qty{292}{kW.h}$ في السنة؛ وزمن الاسترداد $700/292 \approx 2.4$ سنة ---
وهو قصير أمام عمر لوح يبلغ 25 سنة.
\textbf{15.} $\qty{12.6}{kW.h}/\qty{0.1}{kW} = \qty{126}{h}$:
أي أكثر من خمسة أشخاص ($126/24 = 5.25$) يدوسون ليل نهار.
\textbf{16.} الغذاء $= 12.6/0.25 = \qty{50.4}{kW.h} \approx
\qty{181}{MJ}$، أي\ $181/1.5 \approx 121$ صحن حبوب.
\textbf{17.} $121 \times 0.50 \approx 60$ يورو، مقابل
$12.6 \times 0.25 \approx 3.15$ يورو من الشبكة: \omterm{def:g10:energy-conservation:power}{فاستطاعة} الدوس
تكلّف نحو $19$ مرة أكثر --- قبل دفع أجور راكبي الدراجات.
\textbf{18.} $5000/45 \approx 110$ يوم بيت. لكنها
$\num{5e9}/\num{1e-4} = \qty{5e13}{W}$ مدة \qty{100}{\micro s}، وفي
مكان عشوائي: فلا \omterm{def:g10:energy-conservation:transfer}{محوّل} ولا بطارية تقدر أن تشرب من ذلك الخرطوم.
فالبيت يحتاج تدفقًا ثابتًا قدره \qty{525}{W}، لا نتوءًا.
\textbf{19.} ليس الجولات --- فهي محفوظة ومتطابقة. إنما تدفع
ثمن \emph{شكلها وتوافرها}: \omterm{def:g10:energy-conservation:energy}{فالطاقة}
المركَّزة القابلة للتخزين والمتاحة عند الطلب تكلّف \omterm{def:g10:energy-conservation:transfer}{سلسلة التحويل}
كلها التي جعلتها كذلك.
\textbf{20.} يوم أسرة واحد $= \qty{12.6}{kW.h} \approx
\qty{45}{MJ}$: أي \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} واحد من البنزين، أو ستة عشر مترًا مربعًا من
سطح مشمس، أو خمسة راكبي دراجات على \omterm{def:g10:universal-gravitation:satellite}{مدار} الساعة. فالمخزون هو \omterm{def:g10:energy-conservation:energy}{الطاقة}
(\omterm{def:g10:energy-conservation:energy}{بالجول} \omterm{def:g10:energy-conservation:kwh}{والكيلوواط ساعة})؛ والتدفق هو \omterm{def:g10:energy-conservation:power}{الاستطاعة} (\omterm{def:g10:energy-conservation:power}{بالواط}) --- أي حوض الاستحمام
والحنفية، ولا خلط بينهما بعد اليوم.
\end{solution}
