\begin{solution}{exo:g10:universal-gravitation:7}

الكيس \omterm{def:g10:universal-gravitation:weight}{ووزن} الكتل المعيارية ينقسمان على العامل نفسه،
ومن ثم لا تتغير المقارنة. أما الميزان الزنبركي فيقرأ
\omterm{def:g10:universal-gravitation:weight}{الوزن}: $3.0 \times 9.81 \approx \qty{29}{N}$ على الأرض،
و$3.0 \times 1.62 \approx \qty{4.9}{N}$ على القمر.

\emph{2.} الميزان ذو الكفتين: لأنه يقيس الكتلة، وهي ما يشتريه
الزبون. أما ميزان زنبركي معايَر على ``\omterm{def:g10:orders-of-magnitude:unit}{الكيلوغرام}'' في الأرض فسيقرأ
$4.9/9.81 \approx \qty{0.50}{kg}$ للكيس نفسه على القمر ---
فيهب صاحب المتجر ستة أكياس بثمن كيس واحد.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:10}

$F = \dfrac{\num{6.67e-11} \times \num{7.35e22}}{(\num{3.84e8})^2}

\emph{2.} تجذب الشمس أقوى نحو $\num{5.9e-3}/\num{3.3e-5} \approx 180$

القريب إلى الجانب البعيد من الكرة الأرضية. والمدّ والجزر يتغذيان على ذلك الفرق ---

\end{solution}

\begin{solution}{exo:g10:universal-gravitation:11}
\emph{1.} $g' = \dfrac{G (2M)}{(2R)^2} = \dfrac{2}{4}\,\dfrac{GM}{R^2}
= \dfrac{g}{2} \approx \qty{4.9}{N/kg}$: فمضاعفة نصف القطر تغلب
مضاعفة الكتلة.

\emph{2.} نحتاج $\dfrac{GM}{R'^2} = 2\,\dfrac{GM}{R^2}$، ومنه
$R'^2 = R^2/2$ و$R' = R/\sqrt{2} \approx \qty{4.5e6}{m}$ --- أي نحو
\qty{4500}{km}.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:12}
\emph{1.} تساوي الجذبين لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام}:
$\dfrac{G M_{\text{الأرض}}}{x^2} = \dfrac{G M_{\text{القمر}}}{(D-x)^2}$.
وبالضرب التبادلي والقسمة على $G$:
$\left(\dfrac{x}{D-x}\right)^2 = \dfrac{M_{\text{الأرض}}}{M_{\text{القمر}}}$.

\emph{2.} $\dfrac{x}{D-x}
= \sqrt{\dfrac{\num{5.97e24}}{\num{7.35e22}}} = \sqrt{81.2} \approx
9.0$، ومنه $x = 9.0\,(D - x)$، أي\ $x = \dfrac{9.0}{10.0}\,D
\approx 0.90 \times \num{3.84e8} \approx \qty{3.5e8}{m}$. فتقع
نقطة التوازن على $90\%$ من الطريق إلى القمر: أي أن جذب الأرض
يهيمن على الرحلة كلها تقريبًا.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:13}
\emph{1.} بالنشر: $R^2 + 2Rh + h^2 = R^2 + x^2$؛ وبما أن $h$ ضئيل
أمام $R$، نهمل $h^2$: فيكون $2Rh \approx x^2$، ومنه
$h \approx \dfrac{x^2}{2R}
= \dfrac{(\num{8.0e3})^2}{2 \times \num{6.37e6}}
= \dfrac{\num{6.4e7}}{\num{1.27e7}} \approx \qty{5.0}{m}$.

\emph{2.} في \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة يسقط المقذوف \qty{4.9}{m} --- وهو تقريبًا
\qty{5.0}{m} بالضبط التي تنخفضها الأرض على مدى \qty{8.0}{km}. ومن ثم
عند $v \approx \qty{8.0}{km/s}$ لا يلحق السقوط بالأرض أبدًا: فيدور
المقذوف في \omterm{def:g10:universal-gravitation:satellite}{مدار}.

\emph{3.} $T = \dfrac{2\pi R}{v}
= \dfrac{2\pi \times \num{6.37e6}}{\num{8.0e3}}
\approx \qty{5.0e3}{s} \approx \qty{83}{min}$ --- وهو قريب من نحو
$\qty{90}{min}$ للتوابع الحقيقية المنخفضة \omterm{def:g10:universal-gravitation:satellite}{المدار}، وهي تحلّق أعلى
قليلًا.
\end{solution}

\begin{solution}{exo:g10:universal-gravitation:15}

تفاحة ساقطة، بالضبط. فاستنتج نيوتن أن القوة التي تمسك

\end{solution}

\begin{solution}{pb:g10:universal-gravitation:1}
\textbf{1.} $F = \num{6.67e-11} \times
\dfrac{\num{7.3e6} \times 55}{100^2} \approx \qty{2.7e-6}{N}$.

\textbf{2.} تزن حبة الرمل
$\num{1.0e-6} \times 9.81 \approx \qty{9.8e-6}{N}$: أي أن برج إيفل كله
يجذب السائح $3.7$ مرة \emph{أقلّ} من \omterm{def:g10:universal-gravitation:weight}{وزن}
حبة رمل.

\textbf{3.} (أ) المسافة $\times 10$: تُقسم القوة على $100$.
(ب) الكتلتان $\times 1000$: تُضرب القوة في $10^6$.
(ج) الكتلتان $\times 1000$ والمسافة $\times 1000$:
$10^6 / 10^6$ --- فالقوة لا تتغير.

\textbf{4.} $F = \dfrac{\num{6.67e-11} \times \num{5.97e24} \times
1.00}{(\num{6.37e6})^2} \approx \qty{9.81}{N}$: أي \omterm{def:g10:universal-gravitation:weight}{وزن} \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام}
واحد --- وهذا بالضبط هو شدة المجال
$g = \qty{9.81}{N/kg}$.

\textbf{5.} \omterm{def:g10:universal-gravitation:force}{الجاذبية} ضعيفة لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} لكنها لا \emph{تجذب} إلا
جذبًا، ومن ثم تتجمع جذوب كيلوغرامات الأرض البالغة $\num{5.97e24}$
كلها في الاتجاه نفسه --- والكتلة الفلكية تغلب
الكتلة الجارة باثنتين وعشرين قوة من قوى العشرة.

\textbf{6.} $F = \num{6.67e-11} \times \dfrac{158 \times 0.73}{0.23^2}
= \num{6.67e-11} \times \dfrac{115.3}{0.0529}
\approx \qty{1.5e-7}{N}$.

\textbf{7.} تزن الكرة الصغيرة
$0.73 \times 9.81 \approx \qty{7.2}{N}$ --- أي نحو $\num{5e7}$ مرة
التجاذب المطلوب كشفه. فلا ميزان ذو كفتين يفرّق جزءًا واحدًا من
خمسين مليونًا؛ أما سلك فتل دقيق، يلتوي التواءً مرئيًّا تحت
عزوم من رتبة النانونيوتن، فيقدر على ذلك.

\textbf{8.} $G = \dfrac{F d^2}{m_1 m_2}
= \dfrac{\num{1.47e-7} \times 0.0529}{115.3}
\approx \qty{6.74e-11}{N.m^2/kg^2}$ --- أي في حدود نحو $1\%$ من
القيمة الحديثة \num{6.67e-11}، من سقيفة خشبية سنة 1798.

\textbf{9.} انطلاقًا من $g = G M / R^2$:
\[
M = \frac{g R^2}{G}
= \frac{9.81 \times (\num{6.37e6})^2}{\num{6.67e-11}}
\approx \qty{5.97e24}{kg}.
\]
ستة ملايين مليار مليار \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام}: أي الأرض، موزونةً.

\textbf{10.} $V = \frac43 \pi R^3 = \frac43 \pi
(\num{6.37e6})^3 \approx \qty{1.08e21}{m^3}$، ومنه
$\rho = \num{5.97e24}/\num{1.08e21} \approx \qty{5.5e3}{kg/m^3}$ ---
أي ضعف الكتلة الحجمية لصخور السطح. فلا بد أن يخفي الباطن شيئًا
أكثف بكثير من الغرانيت: للأرض نواة ثقيلة (حديدية).

\textbf{11.} $g_{\text{القمر}} = \dfrac{\num{6.67e-11} \times
\num{7.35e22}}{(\num{1.74e6})^2} = \dfrac{\num{4.90e12}}
{\num{3.03e12}} \approx \qty{1.62}{N/kg}$.

\textbf{12.} الارتفاع $\propto 1/g$:
$h = 0.50 \times \dfrac{9.81}{1.62} \approx \qty{3.0}{m}$ --- أي
قفزة من الوقوف فوق سلة كرة السلة، بالحركة البطيئة.

\textbf{13.} $g_{\text{عطارد}} = \dfrac{\num{6.67e-11} \times
\num{3.30e23}}{(\num{2.44e6})^2} = \dfrac{\num{2.20e13}}
{\num{5.95e12}} \approx \qty{3.70}{N/kg}$ --- أي تقريبًا قيمة المريخ
\qty{3.73}{N/kg} بالضبط. فعطارد أقلّ كتلة، لكن نصف قطره الأصغر (أي
$R^2$ أصغر في المقام) يعوّض ذلك: فهو عالم أكثف بكثير.

\textbf{14.} يقتضي نصف شدة المجال
$(R + h)^2 = 2 R^2$، ومنه $R + h = R\sqrt{2}$ و
$h = (\sqrt{2} - 1) R \approx 0.414 \times \num{6.37e6}
\approx \qty{2.6e6}{m}$ --- أي نحو \qty{2600}{km} إلى أعلى.

\textbf{15.} $\dfrac{G M_{\text{الأرض}}}{d^2} = 1.62$ يعطي
$d = \sqrt{\dfrac{\num{6.67e-11} \times \num{5.97e24}}{1.62}}
= \sqrt{\num{2.46e14}} \approx \qty{1.57e7}{m} \approx 2.5$ نصف قطر
أرضي. فمن ارتفاع نصف قطر ونصف فوق الأرض لا غير، لا تجذب الأرض
أقوى مما يجذب سطح القمر.

\textbf{16.} $v = \dfrac{2\pi d}{T}
= \dfrac{2\pi \times \num{1.496e11}}{\num{3.156e7}}
\approx \qty{2.98e4}{m/s}$ --- أي ثلاثين كيلومترًا في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}.

\textbf{17.} $\dfrac{v^2}{d} = \dfrac{(\num{2.98e4})^2}{\num{1.496e11}}
\approx \qty{5.9e-3}{N/kg}$: أي أن الشمس تجذب كل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من الأرض
بنحو ستة أجزاء من الألف من النيوتن.

\textbf{18.} $\dfrac{G M_{\text{الشمس}}}{d^2} = \dfrac{v^2}{d}$ يعطي
\[
M_{\text{الشمس}} = \frac{v^2 d}{G}
= \frac{\num{8.87e8} \times \num{1.496e11}}{\num{6.67e-11}}
\approx \qty{1.99e30}{kg}.
\]

\textbf{19.} $\dfrac{\num{1.99e30}}{\num{5.97e24}} \approx
\num{3.3e5}$: أي أن الشمس ثلث مليون أرض.

\textbf{20.} لأن $G$ نفسه يحكم كل زوج من الكتل،
فقد أتاح لنا قياسه مرة واحدة بين كرتي رصاص في سقيفة أن نضع في
الميزان أولًا الكوكب الذي تحت أقدامنا، ثم النجم الذي ندور حوله ---
وهذا ما تساويه كلمة \emph{كونية}.
\end{solution}
