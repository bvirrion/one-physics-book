\section*{الفصل \ref{ch:g10:orders-of-magnitude} --- رتب القدر: قياس الكون}

\begin{solution}{exo:g10:orders-of-magnitude:1}
\emph{1.} $\qty{750}{nm} = 750 \times \qty{e-9}{m} = \qty{7.50e-7}{m}$.
\emph{2.} $\qty{2.5}{km} = \qty{2.5e3}{m}$.
\emph{3.} $\qty{3.2}{\micro m} = \qty{3.2e-6}{m}$.
\emph{4.} $\qty{0.15}{Tm} = 0.15 \times \qty{e12}{m} = \qty{1.5e11}{m}$
--- أي \omterm{def:g10:orders-of-magnitude:lightyear}{وحدة فلكية} واحدة.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:4}
الإنسان: $1.7 < 5$، \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر} $\qty{e0}{m}$. إفرست:
$\qty{8848}{m} = \qty{8.848e3}{m}$ و$8.848 \geq 5$: \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر}
$\qty{e4}{m}$. الجرثومة: \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر} $\qty{e-6}{m}$. الأرض:
$5.97 \geq 5$، \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر} $\qty{e25}{kg}$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:5}
\emph{1.} نصف تدريجة: $L = (127.0 \pm 0.5)\,\unit{mm}$.
\emph{2.} $\Delta L / L = 0.5/127 \approx 0.4\%$.
\emph{3.} الممحاة: $0.5/12 \approx 4\%$. قياس القلم أدقّ
نحو عشر مرات --- المسطرة نفسها، \omterm{def:g10:orders-of-magnitude:uncertainty}{والارتياب المطلق} نفسه،
وجسم أطول عشر مرات.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:6}
\emph{1.} بروكسيما:
$4.24 \times \qty{9.47e15}{m} = \qty{4.01e16}{m}$. الشعرى اليمانية:
$8.6 \times \qty{9.47e15}{m} = \qty{8.1e16}{m}$ (رقمان معنويان،
كما في المعطيات).
\emph{2.} $t = d/c = \qty{2.4e13}{m} / \qty{3.00e8}{m/s}
= \qty{8.0e4}{s}$، أي\ $\num{8.0e4}/3600 \approx \qty{22}{h}$:
أي يومًا كاملًا تقريبًا لكل رسالة من المسبار.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:7}
$t = d/c$ في كل مرة. القمر:
$\qty{3.84e8}{m}/\qty{3.00e8}{m/s} = \qty{1.28}{s}$. الشمس:
$\qty{1.496e11}{m}/\qty{3.00e8}{m/s} = \qty{499}{s} \approx
\qty{8.3}{min}$. نبتون:
$\qty{4.50e12}{m}/\qty{3.00e8}{m/s} = \qty{1.50e4}{s} \approx
\qty{4.2}{h}$. فحين تنظر إلى نبتون تراه كما كان قبل
أربع ساعات تقريبًا --- فكل مقراب آلة زمن متواضعة.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:8}
\emph{1.} الورقة الواحدة: $\qty{52.0}{mm}/500 = \qty{0.104}{mm}
= \qty{1.04e-4}{m}$. والارتياب ينقسم كذلك:
$\qty{0.5}{mm}/500 = \qty{0.001}{mm}$، ومنه
$t = (104 \pm 1)\,\unit{\micro m}$.
\emph{2.} $1/104 \approx 1\%$.
\emph{3.} قياس ورقة واحدة مباشرة يعطي
$(0.1 \pm 0.5)\,\unit{mm}$: أي ارتيابًا نسبيًّا قدره نحو $500\%$
--- فالمسطرة لا تكاد تثبت وجود الورقة أصلًا. أما رصّ $500$
ورقة فيضرب الطول المقيس في $500$ بينما يبقى \omterm{def:g10:orders-of-magnitude:uncertainty}{الارتياب
المطلق} للمسطرة ثابتًا: فينكمش \omterm{def:g10:orders-of-magnitude:uncertainty}{الارتياب النسبي}
بهذا العامل نفسه $500$.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:9}
اقسم كل طول على $\qty{2e-10}{m}$. الجرثومة:
$\qty{2.0e-6}{m}/\qty{2e-10}{m} = \num{1.0e4}$ ذرة. الشعرة:
$\qty{7.0e-5}{m}/\qty{2e-10}{m} = \num{3.5e5}$. الإنسان:
$1.7/\num{2e-10} = \num{8.5e9}$، \omterm{def:g10:orders-of-magnitude:oom}{رتبة القدر} $10^{10}$: أي عشرة
مليارات ذرة من الرأس إلى القدم.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:10}
\emph{1.} $s = \qty{0.22}{m}/\qty{1.39e9}{m} = \num{1.58e-10}$.
\emph{2.} الأرض:
$\qty{1.27e7}{m} \times \num{1.58e-10} = \qty{2.0e-3}{m} =
\qty{2.0}{mm}$ --- حبة فلفل --- تدور على بعد
$\qty{1.496e11}{m} \times \num{1.58e-10} = \qty{23.7}{m} \approx
\qty{24}{m}$ من كرة القدم.
\emph{3.} القمر:
$\qty{3.84e8}{m} \times \num{1.58e-10} = \qty{6.1}{cm}$ من حبة
الفلفل. نبتون:
$\qty{4.50e12}{m} \times \num{1.58e-10} \approx \qty{710}{m}$: فالمجسّم
لا يسع ساحة المدرسة.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:11}
\emph{1.} $21.0 \times 29.7 = \qty{623.7}{cm^2}$، مقرَّبة إلى \omterm{def:g10:orders-of-magnitude:sigfig}{الأرقام المعنوية}
الثلاثة للمعطيات: $\qty{624}{cm^2}$.
\emph{2.} $v = \qty{100.0}{m}/\qty{12.3}{s} = \qty{8.13}{m/s}$ (ثلاثة
أرقام، يحدّدها الزمن).
\emph{3.} خمسة \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية} تدّعي معرفة المساحة حتى
$\qty{0.01}{cm^2}$، وهي دقة لم تعطها قط معطيات على مستوى
المِلّيمتر --- والرقم العشري المزوَّر يبقى تزويرًا.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:12}
\emph{1.} نصف القطر $r = \qty{0.55}{m}$، والمساحة
$A = \pi r^{2} = \pi \times 0.55^{2} \approx \qty{0.95}{m^2}$. الحجم
$V = \qty{1.0}{mm^3} = \qty{1.0e-9}{m^3}$، والغشاء (مساحة)
$\times$ (سُمك)، ومنه
\[
e = \frac{V}{A} = \frac{\qty{1.0e-9}{m^3}}{\qty{0.95}{m^2}}
= \qty{1.1e-9}{m} .
\]
\emph{2.} رتبة قدر الجزيء $\qty{e-9}{m}$ --- أي درجة واحدة فوق
درجة الذرة $\qty{e-10}{m}$، وهذا متوافق مع كون الجزيء عنقودًا
صغيرًا من الذرات. لقد هدّأت ملعقة فرانكلين من الزيت نصف فدّان من
البركة، وقاست، من حيث لا يدري، النانومتر.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:13}
الفرضيات: نحو $70$ نبضة في الدقيقة في الراحة، وعمر قدره $80$
سنة. في اليوم: $70 \times 60 \times 24 \approx \num{1e5}$ نبضة. وفي
السنة: $\num{1e5} \times 365 \approx \num{4e7}$. وفي العمر:
$\num{4e7} \times 80 \approx \num{3e9}$ --- أي رتبة قدر
$10^{9}$، بضعة مليارات من النبضات. أما أربعة \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية} فتدّعي
معرفة تواتر القلب، دقيقة بدقيقة، على مدى ثمانين سنة:
فالمعطيات تخمينات \omterm{def:g10:orders-of-magnitude:oom}{برتبة القدر}، ولا مخرج يعلو على
مدخلاته (\cref{met:g10:orders-of-magnitude:sigfig}).
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:14}
بعد $n$ طيّة يصير السُّمك $2^{n} \times \qty{1.0e-4}{m}$.
\emph{1.} المطلوب $2^{n} \geq 8848/\num{1.0e-4} = \num{8.8e7}$. وبما أن
$2^{26} \approx \num{6.7e7}$ لا يكفي بينما
$2^{27} \approx \num{1.3e8}$ يكفي: فالجواب $27$ طيّة (والرزمة
$\approx \qty{13}{km}$).
\emph{2.} المطلوب $2^{n} \geq \num{3.84e12}$: إنّ $2^{41} \approx
\num{2.2e12}$ لا يكفي، و$2^{42} \approx \num{4.4e12}$ يبلغ ---
أي $42$ طيّة، والرزمة $\qty{4.4e8}{m}$، وهو ما يتجاوز القمر بأريحية.
\emph{3.} التجربة وحدها: فكل طيّة تنصّف المساحة، وبعد
سبع طيّات تصير الرزمة أسمك منها عرضًا. أما حساب
المضاعفة فلا يمسّه شيء --- إنما النمو الأسّي يسبق ببساطة ما يقدر
الورق (والحدس) على متابعته.
\end{solution}

\begin{solution}{exo:g10:orders-of-magnitude:15}
\emph{1.} القمر: $\qty{0.03}{m}/\qty{3.84e8}{m} = \num{7.8e-11}$.
الطاولة: $\qty{0.001}{m}/\qty{1.000}{m} = \num{1.0e-3}$. فقياس القمر
أدقّ بعامل قدره
$\num{1.0e-3}/\num{7.8e-11} \approx \num{1.3e7}$ --- أي أكثر من عشرة ملايين.
\emph{2.} بلوغ $\num{7.8e-11}$ على طاولة طولها \omterm{def:g10:orders-of-magnitude:unit}{متر} واحد يعني
$\Delta x = \qty{7.8e-11}{m}$، وهو أصغر من ذرة واحدة
($\approx \qty{e-10}{m}$): بل إن حافة الطاولة ليست \emph{معرَّفة}
بهذه الحدّة أصلًا. فنسبيًّا، يُعدّ قياس المسافة بين الأرض والقمر
بالليزر من أرقى ما تملكه بنو البشر من قياسات.
\end{solution}

\begin{solution}{pb:g10:orders-of-magnitude:1}
\textbf{1.} الحبة: $\qty{0.5}{mm} = \qty{5e-4}{m}$، على سُلّم المِلّي؛
والذرة: $\qty{2.5e-10}{m} = \qty{0.25}{nm}$، على سُلّم النانو.

\textbf{2.} $\qty{5e-4}{m}/\qty{2.5e-10}{m} = \num{2.0e6}$: أي
مليونا ذرة عرضًا.

\textbf{3.} مكعب فيه $2 \times 10^{6}$ ذرة على كل ضلع يحوي
$(2\times10^{6})^{3} = 8 \times 10^{18}$ ذرة: أي رتبة قدر
$10^{19}$.

\textbf{4.} $V = (\qty{5e-4}{m})^{3} = \qty{1.25e-10}{m^3}$، ومنه
$m = \rho V = \qty{2500}{kg/m^3} \times \qty{1.25e-10}{m^3}
= \qty{3.1e-7}{kg} = \qty{0.31}{mg}$.

\textbf{5.} الكتلة لكل ذرة:
$\qty{3.1e-7}{kg}/(8\times10^{18}) = \qty{3.9e-26}{kg}$ --- وهي تقع تمامًا
في مجال ``بضعة $10^{-26}\,\unit{kg}$''. فالمكعب الساذج من الذرات
يعيد إنتاج الكتل الذرية الحقيقية: أي أن النموذج ينجح في اختبار السلامة
(\cref{met:g10:orders-of-magnitude:sanity}) بامتياز.

\textbf{6.} $s = \qty{5e-4}{m}/\qty{1.39e9}{m} = \num{3.6e-13}$.

\textbf{7.} الأرض المجسّمة: قطرها
$\qty{1.27e7}{m} \times \num{3.6e-13} = \qty{4.6e-6}{m}$ --- أي بحجم
جرثومة --- على بعد
$\qty{1.496e11}{m} \times \num{3.6e-13} = \qty{5.4}{cm}$ من
الحبة: فالأرض ميكروب على بعد خمسة سنتيمترات من حبة رمل.

\textbf{8.} $\qty{4.5e12}{m} \times \num{3.6e-13} = \qty{1.6}{m}$: أي أن
المنظومة الكوكبية كلها تدور داخل طاولة طعام.

\textbf{9.} $\qty{4.0e16}{m} \times \num{3.6e-13} = \qty{1.4e4}{m}
= \qty{14}{km}$.

\textbf{10.} أقرب نجم إلى منظومتنا الشمسية الموضوعة على الطاولة حبة
رمل أخرى على بعد أربعة عشر كيلومترًا، ولا شيء تقريبًا
بينهما: فالمجرّة خواء غالب، ولذلك
تكاد تصادمات النجوم لا تُسمع بها أصلًا.

\textbf{11.} $t = \qty{5e-4}{m}/\qty{3.00e8}{m/s} = \qty{1.7e-12}{s}$
--- أي أقل من بيكوثانيتين.

\textbf{12.} القمر: $\qty{3.84e8}{m}/\qty{3.00e8}{m/s} =
\qty{1.28}{s}$. الشمس: $\qty{1.496e11}{m}/\qty{3.00e8}{m/s} =
\qty{499}{s} \approx \qty{8.3}{min}$: فضوء الشمس عمره ثماني دقائق
دائمًا.

\textbf{13.} السنة $365.25 \times 86400 = \qty{3.156e7}{s}$، ومنه
$\qty{1}{ly} = \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}$؛ ثم
$\qty{4.0e16}{m}/\qty{9.47e15}{m} = 4.2$: فبُعد المعطيات هو فعلًا
نحو $\qty{4.2}{ly}$.

\textbf{14.} $\qty{2.4e22}{m}/\qty{9.47e15}{m} = \num{2.5e6}$: أي
مليونان ونصف مليون \omterm{def:g10:orders-of-magnitude:lightyear}{سنة ضوئية}. وقد غادر ذلك الضوء أندروميدا قبل نحو $2.5$
مليون سنة، حين كان أوائل أفراد جنس \emph{الإنسان}
يتعلمون نحت الحجارة --- ولم يكن \emph{الإنسان العاقل} قد
وُجد بعد في أي مكان.

\textbf{15.} $\qty{3.00e8}{m/s} \times \qty{1e-9}{s} = \qty{0.30}{m}$.
وفي ثلث نانوثانية تقطع الإشارة على الأكثر نحو
$\qty{10}{cm}$، فأي جزأين من الآلة يجب أن يتحاورا خلال
نبضة ساعة واحدة عليهما أن يقعا على بعد عشرة سنتيمترات من بعضهما:
فالسرعة تحدّ الحجم.

\textbf{16.} متوسط كتلة الذرة في الماء:
$\qty{3.0e-26}{kg}/3 = \qty{1.0e-26}{kg}$. وعدد الذرات:
$\qty{70}{kg}/\qty{1.0e-26}{kg} = \num{7e27}$.

\textbf{17.} في مقابل الحبة:
$\num{7e27}/\num{8e18} \approx 10^{9}$ --- أي أنك تحوي من الذرات ما
في مليار حبة. وفي مقابل النجوم:
$\num{7e27}/\num{2e23} = \num{3.5e4}$ --- أي أن ذراتك تفوق عدد كل
نجوم الكون المرصود بخمسة وثلاثين ألف مرة.

\textbf{18.} $\text{(أنت)}/\text{(الذرة)} =
\qty{1.7}{m}/\qty{2.5e-10}{m} \approx \num{7e9} \approx 10^{10}$،
بينما $\text{(الكون)}/\text{(أنت)} =
\qty{8.8e26}{m}/\qty{1.7}{m} \approx \num{5e26}$. فأنت على السُّلّم
تقف على نحو عشر درجات فوق الذرة وقرابة سبع وعشرين تحت
الكون: أي أن البشر، ضربيًّا، يعيشون قريبين جدًّا من ذراتهم.

\textbf{19.} يعطي الشرط
$x^{2} = 10^{-15} \times 10^{26.9} = 10^{11.9}$، ومنه
$x = 10^{5.95} \approx \qty{9e5}{m} \approx \qty{900}{km}$ --- وهو تقريبًا
قطر سيريس بالضبط، أكبر الكويكبات
($\approx \qty{940}{km}$). ففي منتصف السُّلّم بين النواة
والكون يجلس كوكب قزم متواضع.

\textbf{20.} $\qty{8.8e26}{m} \times \num{3.6e-13} = \qty{3.2e14}{m}$،
أي\ $\qty{3.2e14}{m}/\qty{1.496e11}{m} \approx \num{2.1e3}$
\omterm{def:g10:orders-of-magnitude:lightyear}{وحدة فلكية}. وبيت القصيد: قلّص الكون حتى تصير الشمس
حبة رمل، فيبقى الكون المرصود ممتدًّا على ألفي ضعف
المسافة الحقيقية بين الأرض والشمس --- فلا مجسّم يهبط
بالكون إلى القياس البشري؛ ولا يحمله إلا قوى العشرة.
\end{solution}
