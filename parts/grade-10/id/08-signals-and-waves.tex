\chapter{Sinyal dan Gelombang}\label{ch:g10:signals-and-waves}

Setiap pesan yang pernah kamu terima tiba sebagai sebuah \omterm{def:g10:signals-and-waves:signal}{sinyal}: suara
yang menekan gendang telingamu, dan halaman ini sampai kepadamu sebagai
kelip \omterm{def:g10:signals-and-waves:wave}{gelombang} radio. Di balik keanekaragamannya duduk satu gagasan ---
sebuah \omterm{def:g10:orders-of-magnitude:unit}{besaran fisis} yang berubah terhadap waktu --- dan satu perkakas:
\omterm{def:g10:signals-and-waves:period}{periode}, \omterm{def:g10:signals-and-waves:frequency}{frekuensi}, \omterm{def:g10:signals-and-waves:amplitude}{amplitudo}, serta seni mengukur waktu tunda. Dengan
perkakas itu sebuah kapal mengukur dasar laut dan seorang dokter
menghitung detak jantung.

\section{Sinyal membawa keterangan}

\begin{definition}[Sinyal]\label{def:g10:signals-and-waves:signal}
Sebuah \emph{sinyal}\index{sinyal} adalah \omterm{def:g10:orders-of-magnitude:unit}{besaran fisis} yang berubah
terhadap waktu untuk membawa keterangan: \omterm{def:g10:pressure:pressure}{tekanan} udara pada gendang
telingamu, tegangan pada kutub sebuah mikrofon, keterangan cahaya di
dalam \omterm{def:g10:refraction:fiber}{serat optik}. Untuk menelaah sebuah sinyal, rekamlah ia sebagai
fungsi waktu lalu bacalah keterangannya dari kurvanya.
\end{definition}

Pada sebuah panggilan telepon, keterangan yang sama berganti pembawa
berkali-kali: \omterm{def:g10:pressure:pressure}{tekanan} di udara, lalu tegangan di dalam mikrofon, lalu
\omterm{def:g10:signals-and-waves:wave}{gelombang} radio di antenanya --- dan mundur kembali di ujung yang lain.
Satu kosakata menangani semuanya; bab ini membangunnya.

\section{Sinyal periodik}

\begin{definition}[Sinyal periodik, periode]\label{def:g10:signals-and-waves:period}
Sebuah \omterm{def:g10:signals-and-waves:signal}{sinyal} disebut \emph{periodik}\index{sinyal periodik} apabila ia
berulang secara serupa pada selang yang teratur.
\emph{Periode}\index{periode} $T$ adalah lama satu daur yang lengkap,
dalam sekon.
\end{definition}

\begin{definition}[Frekuensi]\label{def:g10:signals-and-waves:frequency}
\emph{Frekuensi}\index{frekuensi} sebuah \omterm{def:g10:signals-and-waves:period}{sinyal periodik} adalah
banyaknya daur tiap sekon:
\[
f = \frac{1}{T} .
\]
\omterm{def:g10:orders-of-magnitude:unit}{Satuannya}, satu daur tiap sekon, disebut \emph{hertz}\index{hertz}
(\unit{Hz}), dengan kelipatan yang lazim \unit{kHz}, \unit{MHz},
\unit{GHz}.
\end{definition}

\begin{definition}[Amplitudo]\label{def:g10:signals-and-waves:amplitude}
\emph{Amplitudo}\index{amplitudo} sebuah \omterm{def:g10:signals-and-waves:period}{sinyal periodik} adalah
simpangan terbesarnya dari nilai diamnya. Untuk tegangan yang berayun
setangkup di sekitar nol, amplitudonya adalah nilai puncaknya; dari
puncak ke lembah terukur \emph{dua kali} amplitudonya (nilai puncak ke
puncak).
\end{definition}

\begin{example}[Jala-jala listrik dan nada A]\label{ex:g10:signals-and-waves:mains}
Tegangan jala-jala listrik berayun pada $f = \qty{50}{Hz}$: \omterm{def:g10:signals-and-waves:period}{periodenya}
$T = 1/f = \qty{0.020}{s} = \qty{20}{ms}$. Nada A sebuah orkestra adalah
\omterm{def:g10:signals-and-waves:signal}{sinyal} \omterm{def:g10:pressure:pressure}{tekanan} pada \qty{440}{Hz}: satu daurnya berlangsung
$T = 1/440 = \qty{2.3}{ms}$. Makin tinggi \omterm{def:g10:signals-and-waves:frequency}{frekuensinya}, makin pendek
daurnya: $f$ dan $T$ berkebalikan satu sama lain.
\end{example}

\section{Membaca sebuah osilogram}

\begin{definition}[Osiloskop]\label{def:g10:signals-and-waves:oscilloscope}
Sebuah \emph{osiloskop}\index{osiloskop} menggambarkan tegangan terhadap
waktu; kurva pada layar berkisinya disebut \emph{osilogram}%
\index{osilogram}. Dua setelan mengubah petak layar menjadi nilai fisis:
\emph{basis waktu}\index{basis waktu} (sekon tiap petak mendatar) dan
\emph{penguatan}\index{penguatan} tegaknya (volt tiap petak tegak).
\end{definition}

\begin{method}[Dari layar ke angka]\label{met:g10:signals-and-waves:oscillogram}
\begin{enumerate}
  \item Hitunglah petak mendatar yang dicakup satu daur lengkap (puncak
        ke puncak paling aman); kalikan dengan \omterm{def:g10:signals-and-waves:oscilloscope}{basis waktunya} untuk
        memperoleh $T$, lalu $f = 1/T$.
  \item Hitunglah petak tegak dari garis tengah sampai sebuah puncak
        lalu kalikan dengan \omterm{def:g10:signals-and-waves:oscilloscope}{penguatannya} untuk memperoleh \omterm{def:g10:signals-and-waves:amplitude}{amplitudonya}
        (puncak ke lembah memberikan nilai puncak ke puncak:
        separuhkanlah).
\end{enumerate}
\end{method}

\begin{example}[Satu pembacaan yang lengkap]\label{ex:g10:signals-and-waves:reading}
Pada \omterm{def:g10:signals-and-waves:oscilloscope}{osilogram} di bawah ini, \omterm{def:g10:signals-and-waves:oscilloscope}{basis waktunya} \qty{5}{ms} tiap petak dan
\omterm{def:g10:signals-and-waves:oscilloscope}{penguatannya} \qty{2}{V} tiap petak. Satu daur mencakup $4.0$ petak:
$T = 4.0 \times \qty{5}{ms} = \qty{20}{ms}$ dan
$f = 1/\qty{0.020}{s} = \qty{50}{Hz}$ --- jala-jala listrik lagi. Sebuah
puncak berada $3.0$ petak di atas sumbunya: \omterm{def:g10:signals-and-waves:amplitude}{amplitudonya}
$3.0 \times \qty{2}{V} = \qty{6.0}{V}$.
\end{example}

\section{Sinyal bunyi dan sinyal elektromagnetik}

\begin{definition}[Bunyi]\label{def:g10:signals-and-waves:sound}
\emph{Bunyi}\index{bunyi} adalah \omterm{def:g10:signals-and-waves:signal}{sinyal} \omterm{def:g10:pressure:pressure}{tekanan}: sebuah benda yang
bergetar --- dawai, selaput, pita suara --- mendorong udara secara
berirama, dan rapatannya merambat ke luar dengan sekitar \qty{340}{m/s}.
Mikrofon mengubah \omterm{def:g10:pressure:pressure}{tekanan} yang tiba menjadi tegangan; gendang telinga
mengubahnya menjadi denyut saraf. Bunyi memerlukan medium: bel yang
dibunyikan di dalam bejana hampa menjadi bisu ketika udaranya dipompa
keluar.
\end{definition}

\begin{definition}[Gelombang]\label{def:g10:signals-and-waves:wave}
Gangguan yang merambat --- pola rapatan \omterm{def:g10:signals-and-waves:sound}{bunyi}, riak di kolam, denyut
gelombang radio --- disebut \emph{gelombang}\index{gelombang}.
Mediumnya, kalau memang ada, tetap tinggal di tempat: tiap petak air
naik turun di tempatnya sementara riaknya menyeberangi kolam. Ilmu ukur
gelombang --- \omterm{def:g10:light-spectra:wavelength}{panjang gelombang}, interferensi, difraksi --- dibahas pada
bab berikutnya; bab ini hanya memerlukan lajunya.
\end{definition}

\begin{definition}[Ultrasonik dan infrasonik]\label{def:g10:signals-and-waves:ultrasound}
Telinga manusia mendengar \omterm{def:g10:signals-and-waves:sound}{bunyi} antara sekitar \qty{20}{Hz} dan
\qty{20}{kHz} (batas atasnya menurun seiring usia). \omterm{def:g10:signals-and-waves:sound}{Bunyi} di atas
\qty{20}{kHz} disebut \emph{ultrasonik}\index{ultrasonik} --- kelelawar
berburu pada \qty{50}{kHz}, pemindai kedokteran mencitrakan tubuh pada
beberapa megahertz. \omterm{def:g10:signals-and-waves:sound}{Bunyi} di bawah \qty{20}{Hz} disebut
\emph{infrasonik}\index{infrasonik}, yang dirasakan gajah dan
seismometer.
\end{definition}

\begin{proposition}[Sinyal elektromagnetik]\label{prop:g10:signals-and-waves:em}
\omterm{def:g10:signals-and-waves:wave}{Gelombang} radio, Wi-Fi dan cahaya adalah \emph{sinyal
elektromagnetik}\index{sinyal elektromagnetik}: satu keluarga, semuanya
merambat menembus ruang hampa --- tanpa memerlukan medium --- dan
semuanya dengan satu laju, yaitu laju cahaya pada
\cref{ch:g10:refraction},
\[
c = \qty{3.00e8}{m/s} \quad \text{(nyaris sama di udara).}
\]
\end{proposition}
\admitted

\begin{remark}\label{rem:g10:signals-and-waves:honest}
Di sini hal itu merupakan fakta percobaan --- cahaya matahari
menyeberangi $150$ juta kilometer ruang kosong; hukum itu diturunkan
secara jujur pada jilid Tahun ke-2 dari hukum kelistrikan dan
kemagnetan. Peganglah perbedaannya: \omterm{def:g10:signals-and-waves:sound}{bunyi} adalah getaran \emph{dari}
mediumnya dan mati tanpa medium itu; \omterm{prop:g10:signals-and-waves:em}{sinyal elektromagnetik} membawa
medannya sendiri menyeberangi ruang hampa, sejuta kali lebih cepat.
\end{remark}

\section{Gema: jarak dari waktu tunda}

\omterm{def:g10:signals-and-waves:signal}{Sinyal} yang bergerak dengan laju $v$ menempuh $d = v\,t$ dalam waktu
$t$: setiap waktu tunda adalah jarak yang menyamar. Lebih baik lagi,
kirimkanlah denyut singkat ke sebuah penghalang lalu ukurlah waktu
\emph{gema}\index{gema}nya --- perjalanan pulang perginya menempuh $2d$.

\begin{method}[Pengukuran jarak dengan gema]\label{met:g10:signals-and-waves:echo}
\begin{enumerate}
  \item Kirimkan denyut singkat; ukurlah waktu pulang pergi $t$ \omterm{rem:g10:signals-and-waves:honest}{gemanya}.
  \item Ambillah laju $v$ \omterm{def:g10:signals-and-waves:signal}{sinyal} itu \emph{di dalam medium yang
        dilintasinya}.
  \item Denyut itu menempuh $2d = v\,t$, sehingga
        $d = \dfrac{v\,t}{2}$.
\end{enumerate}
Denyut \omterm{def:g10:signals-and-waves:sound}{bunyi} di dalam air menghasilkan \emph{sonar}\index{sonar}
($v \approx \qty{1500}{m/s}$); denyut \omterm{def:g10:signals-and-waves:wave}{gelombang} radio di udara
menghasilkan \emph{radar}\index{radar} ($v = c$); denyut \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik} di
dalam tubuh menghasilkan pemindai kedokteran
($v \approx \qty{1540}{m/s}$ di dalam jaringan lunak).
\end{method}

\begin{example}[Sonar]\label{ex:g10:signals-and-waves:sonar}
\omterm{met:g10:signals-and-waves:echo}{Sonar} sebuah kapal memancarkan denyut dan \omterm{rem:g10:signals-and-waves:honest}{gema} dasar lautnya kembali
setelah $t = \qty{0.80}{s}$: kedalamannya
$d = \frac{1500 \times 0.80}{2} = \qty{600}{m}$. Melupakan faktor
$2$ menggandakan samudranya.
\end{example}

\begin{example}[Radar]\label{ex:g10:signals-and-waves:radar}
\omterm{met:g10:signals-and-waves:echo}{Radar} sebuah bandara menerima \omterm{rem:g10:signals-and-waves:honest}{gema} sebuah pesawat $\qty{0.30}{ms}$
sesudah denyutnya:
$d = \frac{\qty{3.00e8}{m/s} \times \qty{3.0e-4}{s}}{2} = \qty{45}{km}$.
Karena $c$ begitu besar, \omterm{met:g10:signals-and-waves:echo}{radar} hidup pada jam berskala mikrosekon:
mengukur waktu tempuh cahaya adalah pekerjaan yang menuntut ketelitian.
\end{example}

\begin{example}[Jantung sebagai sinyal]\label{ex:g10:signals-and-waves:ecg}
Tiap detak jantung mengirimkan denyut tegangan yang kecil melintasi
dada; elektrode merekamnya sebagai
\emph{elektrokardiogram}\index{elektrokardiogram} (EKG), sebuah \omterm{def:g10:signals-and-waves:signal}{sinyal}
yang hampir \omterm{def:g10:signals-and-waves:period}{periodik}. Jika puncak-puncaknya yang tinggi berjarak
$T = \qty{0.75}{s}$, jantung itu berdetak pada
$f = 1/0.75 = \qty{1.33}{Hz}$, yaitu $1.33 \times 60 = 80$ detak tiap
menit: membaca $T$ dari sebuah rekaman, begitulah seorang dokter jantung
mengukur denyut nadimu.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:1}
Hitunglah \omterm{def:g10:signals-and-waves:frequency}{frekuensi} \omterm{def:g10:signals-and-waves:period}{sinyal periodik} yang \omterm{def:g10:signals-and-waves:period}{periodenya}
(a) $T = \qty{20}{ms}$; (b) $T = \qty{4.0}{ms}$;
(c) $T = \qty{50}{\micro s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:2}
Hitunglah \omterm{def:g10:signals-and-waves:period}{periode} (a) jala-jala listrik \qty{50}{Hz}; (b) nada A
\qty{440}{Hz}; (c) \omterm{def:g10:signals-and-waves:signal}{sinyal} Wi-Fi \qty{2.4}{GHz}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:3}
Golongkan sebagai \omterm{def:g10:signals-and-waves:ultrasound}{infrasonik}, \omterm{def:g10:signals-and-waves:sound}{bunyi} yang terdengar atau \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik}:
\qty{12}{Hz}; \qty{440}{Hz}; \qty{18}{kHz}; \qty{40}{kHz} (pengindra

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:4}
Pada \omterm{def:g10:signals-and-waves:oscilloscope}{osiloskop} yang disetel \qty{2}{ms} tiap petak dan \qty{0.5}{V} tiap
petak, satu daur mencakup $5.0$ petak dan sebuah puncak berada $4.0$
petak di atas sumbunya. Tentukan $T$, $f$ dan \omterm{def:g10:signals-and-waves:amplitude}{amplitudonya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:signals-and-waves:5}
Kamu melihat kilat, lalu mendengar guruhnya \qty{6.0}{s} kemudian.
Berapa jauh kilat itu menyambar ($v_{\text{bunyi}} = \qty{340}{m/s}$)?
Mengapa waktu tempuh cahayanya boleh diabaikan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:6}

($v = \qty{1500}{m/s}$ di air laut). Mengapa hasil kali $v\,t$ harus

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:signals-and-waves:7}
\omterm{met:g10:signals-and-waves:echo}{Radar} sebuah bandara menerima \omterm{rem:g10:signals-and-waves:honest}{gema} pesawat setelah \qty{0.24}{ms} dan,

Hitunglah kedua jaraknya, lalu laju pesawat itu. Apakah pesawat itu

\end{exercise}

\section{Soal: Sonar, badai, dan kardiogram}

\begin{problem}[{Soal akhir pekan --- membaca sinyal dunia: satu hukum,
$d = v\,t$, mengukur jarak badai, memetakan dasar laut, menghitung detak
jantung dan menyingkap cara kerja GPS}]
\label{pb:g10:signals-and-waves:1}
Sebuah kapal nelayan bekerja sepanjang malam yang berbadai: kilat di
kaki langit, \omterm{met:g10:signals-and-waves:echo}{sonar} yang menyapu dasar laut, jantung sang nakhoda di atas
kertas rekaman dokter, sebuah penerima GPS yang menyimak \omterm{def:g10:universal-gravitation:satellite}{satelit}. Empat
alat, satu hukum. Ambillah $v_{\text{bunyi}} = \qty{340}{m/s}$ di udara,
\qty{1500}{m/s} di air laut, dan $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Bagian I --- Badai.}
\begin{enumerate}
  \item Sebuah kilat menyambar; guruhnya menyusul \qty{9.0}{s}
        kemudian. Berapa jauh kilat itu?
  \item Hitunglah waktu tempuh cahaya pada jarak itu, lalu benarkan
        perlakuan kilatnya sebagai seketika.
  \item Pelaut menghitung sekon antara kilat dan guruh lalu membaginya
        dengan tiga untuk memperoleh kilometer. Benarkan aturan itu.
  \item Lima menit kemudian, sebuah kilat memberikan \qty{6.0}{s}.
        Berapa jauh sekarang? Tentukan laju rata-rata pendekatan
        badainya dalam \unit{m/s} dan \unit{km/h}.
  \item Guruhnya bergemuruh dan tidak menggelegar sekali: saluran
        kilatnya panjangnya berkilo-kilometer, sehingga bagiannya
        terletak pada jarak yang berbeda-beda. Jika salurannya
        terbentang dari \qty{2.0}{km} sampai \qty{5.0}{km} dari
        kapalnya, berapa lama gemuruh itu berlangsung?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{met:g10:signals-and-waves:echo}{Sonar}.}
\begin{enumerate}[resume]
  \item \omterm{met:g10:signals-and-waves:echo}{Sonarnya} memancarkan denyut; dasar lautnya menjawab dalam
        \qty{0.90}{s}. Berapa kedalamannya?
  \item Di atas paparan benua \omterm{rem:g10:signals-and-waves:honest}{gemanya} memendek menjadi \qty{0.20}{s}.
        Berapa kedalamannya?
  \item Satu denyut mengembalikan \emph{dua} \omterm{rem:g10:signals-and-waves:honest}{gema}, pada \qty{0.16}{s}
        dan \qty{0.20}{s}. Tafsirkanlah keduanya; berapa tinggi
        gerombolan ikan itu berenang di atas dasarnya?
  \item \omterm{met:g10:signals-and-waves:echo}{Sonar} itu memancarkan denyut tiap \qty{0.50}{s}. Berapa
        kedalaman terbesar yang dapat diukurnya tanpa kekeliruan, dan
        apa yang menjadi kacau di luar kedalaman itu?
  \item Di udara, waktu tunda \qty{0.60}{s} yang sama akan berarti
        jarak berapa? Pesan moralnya: apa yang harus kamu ketahui
        sebelum mengubah waktu tunda menjadi jarak?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Kardiogram.}
\begin{enumerate}[resume]
  \item Puncak EKG sang nakhoda berjarak \qty{0.86}{s}. Berapa
        \omterm{def:g10:signals-and-waves:frequency}{frekuensinya}? Berapa detak tiap menit?
  \item Kertasnya bergerak maju \qty{25}{mm/s}. Jarak puncak berapa
        yang diukur dokternya?
  \item Setelah menarik jala, jaraknya menjadi \qty{15}{mm}. Berapa
        denyut jantungnya sekarang?
  \item Awak kapal yang muda, seorang pendayung terlatih, beristirahat
        pada $50$ detak tiap menit. Berapa \omterm{def:g10:signals-and-waves:period}{periodenya}? Berapa jaraknya
        pada kertas rekaman?
  \item Apakah sebuah EKG benar-benar \omterm{def:g10:signals-and-waves:period}{periodik}? Lalu, apa yang dibaca
        dokternya dari rekaman itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pulang dengan cahaya.}
\begin{enumerate}[resume]
  \item Kapal itu memanggil pelabuhan dengan radio, yang berjarak
        \qty{50}{km}. Berapa lama pesannya sampai? Berapa lama jika
        dengan \omterm{def:g10:signals-and-waves:sound}{bunyi}?
  \item Sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} GPS mengorbit sekitar \qty{20200}{km} di atas
        kepala. Berapa lama \omterm{def:g10:signals-and-waves:signal}{sinyalnya} sampai ke kapal itu?
  \item GPS mengubah waktu menjadi kedudukan. Galat jarak berapa yang
        ditimbulkan galat jam sebesar \qty{1.0}{\micro s}? Ketelitian
        waktu berapa yang dituntut oleh penentuan kedudukan
        \qty{3.0}{m}?
  \item Mengapa GPS berbasis \omterm{def:g10:signals-and-waves:sound}{bunyi} tidak mungkin ada, bahkan pada
        asasnya sekalipun?
  \item Penutup: sebutkan ketiga laju yang dipakai malam ini dan satu
        hukum yang berdiri di balik keempat alatnya, lalu nyatakan
        kebiasaan mengukur jaraknya dalam satu kalimat.
\end{enumerate}
\end{problem}
