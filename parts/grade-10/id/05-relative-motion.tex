\chapter{Gerak Relatif}\label{ch:g10:relative-motion}

Seorang penumpang berjalan santai ke arah depan kereta cepat sambil
membawa kopi, dengan langkah orang berjalan biasa. Seorang pengamat di
tepi rel mencatat penumpang yang sama pada lebih dari \qty{300}{km/h}.
Kedua pengukuran itu benar. Tidak satu pun merupakan ``sang'' kelajuan
penumpang itu, sebab hal semacam itu tidak ada: gerak bukanlah sifat
sebuah benda semata, melainkan sifat sebuah benda \emph{beserta} sudut
pandang yang dipilih untuk melukiskannya. Bab ini menepatkan sudut
pandang itu --- dan mengajarimu memilih sudut pandang yang baik.

\section{Gerak bersifat relatif}

\begin{definition}[Kerangka acuan]\label{def:g10:relative-motion:frame}
\emph{Kerangka acuan}\index{kerangka acuan} adalah sebuah benda tegar
--- tanah, sebuah gerbong kereta, Bumi seutuhnya --- yang menjadi acuan
pencatatan kedudukan, beserta sebuah jam untuk menandai waktunya. Sebuah
benda dikatakan \emph{diam}\index{diam} di dalam sebuah kerangka apabila
kedudukannya di dalam kerangka itu tidak berubah terhadap waktu; jika
tidak, benda itu bergerak di dalam kerangka tersebut.
\end{definition}

\begin{definition}[Lintasan]\label{def:g10:relative-motion:trajectory}
\emph{Lintasan}\index{lintasan} sebuah benda di dalam \omterm{def:g10:relative-motion:frame}{kerangka acuan}
tertentu adalah kurva yang dibentuk oleh kedudukannya yang
berturut-turut di dalam kerangka itu: garis lurus, lingkaran, sebuah
busur\dots{} Lintasan baru bermakna setelah kerangkanya disebutkan.
\end{definition}

\begin{example}[Pejalan di dalam kereta]\label{ex:g10:relative-motion:walker}
Sebuah kereta melaju \qty{300}{km/h} terhadap tanah, sementara seorang
penumpang berjalan ke depan dengan \qty{4}{km/h} terhadap keretanya.
\begin{itemize}
  \item Di dalam kerangka \emph{kereta}: penumpang itu bergerak
        \qty{4}{km/h} menyusuri lorongnya; kursinya \omterm{def:g10:relative-motion:frame}{diam}; pepohonan di
        luar melesat mundur dengan \qty{300}{km/h}.
  \item Di dalam kerangka \emph{tanah}: kursinya bergerak
        \qty{300}{km/h}, penumpangnya \qty{304}{km/h}, dan pepohonannya
        \omterm{def:g10:relative-motion:frame}{diam}.
\end{itemize}
Pemandangan yang sama, dua kerangka, dua pemerian yang sama sekali
berbeda --- dan tidak ada percobaan yang dikerjakan di dalam kereta itu
yang dapat menyatakan salah satunya ``yang sungguhan''.
\end{example}

\begin{proposition}[Kerelatifan gerak]\label{prop:g10:relative-motion:relativity}
Keadaan \omterm{def:g10:relative-motion:frame}{diam} atau bergeraknya sebuah benda, \omterm{def:g10:relative-motion:trajectory}{lintasannya}, dan lajunya
semuanya bergantung pada \omterm{def:g10:relative-motion:frame}{kerangka acuan} yang dipilih untuk
melukiskannya.
\end{proposition}

\begin{proof}
Satu contoh sudah menuntaskan tiap pernyataan itu. Penumpang yang duduk
pada \cref{ex:g10:relative-motion:walker} \omterm{def:g10:relative-motion:frame}{diam} di dalam kerangka kereta
dan bergerak \qty{300}{km/h} di dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah} (keadaan \omterm{def:g10:relative-motion:frame}{diam} dan
lajunya bergantung pada kerangkanya). Untuk \omterm{def:g10:relative-motion:trajectory}{lintasannya}, amatilah pentil
pada roda sepeda: pengamat yang mengendarai sepeda itu melihat pentilnya
melukis lingkaran mengelilingi porosnya, sedangkan pengamat yang berdiri
di jalan melihatnya melukis rangkaian lengkungan (kurva yang disebut
\emph{sikloid}\index{sikloid}) --- lihat gambar di bawah.
\end{proof}

\begin{remark}
Perhatikanlah apa yang \emph{tidak} berubah pada kedua gambar itu:
pentilnya, rodanya, fisikanya. Hanya pemeriannya yang berubah.
Bertanya ``apakah pentil itu sungguh bergerak melingkar?'' sama saja
dengan bertanya apakah sebuah gunung ``sungguh'' berada di sebelah
kiri: jawabannya bergantung pada tempatmu berdiri.
\end{remark}

\section{Kelajuan rata-rata}

\begin{definition}[Kelajuan rata-rata]\label{def:g10:relative-motion:speed}
\emph{Kelajuan rata-rata}\index{kelajuan!rata-rata} sebuah benda, di
dalam \omterm{def:g10:relative-motion:frame}{kerangka acuan} tertentu, adalah jarak $d$ yang ditempuhnya di
dalam kerangka itu dibagi waktu tempuhnya $t$:
\[
v = \frac{d}{t} .
\]
Dengan $d$ dalam \omterm{def:g10:orders-of-magnitude:unit}{meter} dan $t$ dalam sekon, $v$ dinyatakan dalam \omterm{def:g10:orders-of-magnitude:unit}{meter}
per sekon (\unit{m/s}); kehidupan sehari-hari lebih menyukai kilometer
per jam (\unit{km/h}).
\end{definition}

\begin{method}[Mengubah m/s menjadi km/h dan sebaliknya]\label{met:g10:relative-motion:convert}
Satu kilometer adalah \qty{1000}{m} dan satu jam adalah \qty{3600}{s},
sehingga $\qty{1}{m/s} = \qty{3.6}{km/h}$:
\begin{itemize}
  \item dari \unit{m/s} ke \unit{km/h}: \emph{kalikan} dengan $3.6$;
  \item dari \unit{km/h} ke \unit{m/s}: \emph{bagilah} dengan $3.6$.
\end{itemize}
Pemeriksaan kewajaran: angka dalam \unit{km/h} selalu \emph{lebih
besar} di antara keduanya.
\end{method}

\begin{example}[Beberapa kelajuan yang patut diingat]\label{ex:g10:relative-motion:convert}
Jalan cepat: $\qty{1.5}{m/s} \times 3.6 = \qty{5.4}{km/h}$. Mobil kota
pada \qty{54}{km/h}: $54 / 3.6 = \qty{15}{m/s}$ --- sekitar \qty{15}{m}
ditempuh tiap sekon, patut diingat sebelum melangkah turun dari trotoar.
Bunyi di udara: $\qty{340}{m/s} \times 3.6 = \qty{1224}{km/h}$.
\end{example}

\begin{example}[Pendakian dengan istirahat]\label{ex:g10:relative-motion:hike}
Seorang pendaki menempuh \qty{12}{km} dalam \qty{2}{h} perjalanan, lalu
beristirahat \qty{1}{h}, lalu menempuh \qty{6}{km} dalam \qty{1.5}{h}
terakhir. \omterm{def:g10:relative-motion:speed}{Kelajuan rata-rata} sepanjang perjalanan itu:
\[
v = \frac{d}{t} = \frac{12 + 6}{2 + 1 + 1.5}
= \frac{\qty{18}{km}}{\qty{4.5}{h}} = \qty{4.0}{km/h}.
\]
Rata-ratanya memperhitungkan \emph{seluruh} waktu yang berlalu,
termasuk istirahatnya --- dan umumnya ia bukan rata-rata dari kelajuan
tiap penggalnya.
\end{example}

\section{Memilih kerangka: tanah, geosentrik, heliosentrik}

Karena setiap kerangka sah, kita memilih kerangka yang membuat gerak
yang kita perhatikan tampak paling sederhana. Tiga pilihan sudah
mencakup hampir seluruh fisika di dalam buku ini.

\begin{definition}[Tiga kerangka baku]\label{def:g10:relative-motion:frames}
\begin{itemize}
  \item \emph{Kerangka tanah}\index{kerangka tanah} (atau \emph{kerangka
        laboratorium}\index{kerangka laboratorium}) terpaut pada
        permukaan Bumi di tempat percobaannya berlangsung. Inilah
        kerangka yang tepat untuk gerak sehari-hari: benda jatuh,
        kendaraan, olahraga, meja laboratorium.
  \item \emph{Kerangka geosentrik}\index{kerangka geosentrik} berpusat
        di Bumi tetapi \emph{tidak} ikut berputar bersamanya (arahnya
        menunjuk bintang-bintang yang jauh). Inilah kerangka yang tepat
        untuk Bulan dan untuk \omterm{def:g10:universal-gravitation:satellite}{satelit} buatan, yang mengorbit Bumi
        seutuhnya dan tidak menghiraukan perputaran hariannya.
  \item \emph{Kerangka heliosentrik}\index{kerangka heliosentrik}
        berpusat di Matahari, dengan arah yang lagi-lagi tertambat pada
        bintang-bintang yang jauh. Inilah kerangka yang tepat untuk
        planet: masing-masing lalu menempuh \omterm{def:g10:universal-gravitation:satellite}{orbit} yang sederhana dan
        hampir berbentuk lingkaran.
\end{itemize}
\end{definition}

\begin{method}[Memilih kerangka]\label{met:g10:relative-motion:choose}
Sebutkanlah benda yang geraknya hendak kamu lukiskan, lalu pilihlah
kerangka yang membuat gerak itu paling sederhana:
\begin{enumerate}
  \item gerak di dekat permukaan Bumi, yang berlangsung beberapa menit
        --- \omterm{def:g10:relative-motion:frames}{kerangka tanah};
  \item \omterm{def:g10:universal-gravitation:satellite}{orbit} \emph{mengelilingi} Bumi (Bulan, \omterm{def:g10:universal-gravitation:satellite}{satelit}) --- \omterm{def:g10:relative-motion:frames}{kerangka
        geosentrik};
  \item \omterm{def:g10:universal-gravitation:satellite}{orbit} mengelilingi Matahari (planet, komet) --- \omterm{def:g10:relative-motion:frames}{kerangka
        heliosentrik}.
\end{enumerate}
Sebutkanlah selalu pilihanmu sebelum melukiskan sebuah gerak: \omterm{def:g10:relative-motion:trajectory}{lintasan}
atau kelajuan yang disebutkan tanpa kerangkanya tidak bermakna
(\cref{prop:g10:relative-motion:relativity}).
\end{method}

\begin{remark}[Siapa yang benar, Ptolemaios atau Copernicus?]\label{rem:g10:relative-motion:history}
``Matahari terbit di timur'' dan ``Bumi berputar ke arah timur'' adalah
pengamatan yang sama yang dilaporkan dalam dua kerangka --- yang pertama
dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah}, tempat Matahari menyapu langit, yang kedua dalam
\omterm{def:g10:relative-motion:frames}{kerangka geosentrik}, tempat pengamatnya terbawa mengelilingi sumbu Bumi.
Berabad-abad lamanya kerangkanya sendirilah yang menjadi medan
pertempuran: pemerian Ptolemaios yang berpusat di Bumi mengikuti langit
dengan tepat tetapi memerlukan lingkaran di atas lingkaran untuk
menyusuri planet, sedangkan pemerian yang berpusat di Matahari, yang
dihidupkan kembali Copernicus pada tahun 1543, menjadikan tiap planet
hampir sebuah lingkaran sederhana dan menjelaskan lengkung mundur Mars
yang aneh itu semata-mata sebagai akibat pemilihan kerangka --- Bumi yang
menyalip planet yang lebih lambat. Putusan modernnya bukanlah bahwa satu
kerangka benar dan yang lain salah, melainkan bahwa keduanya tidak
sama-sama \emph{praktis}; ukuran yang lebih dalam untuk lebih menyukai
kerangka tertentu (kerangka lembam) tiba bersama asas kelembaman pada
\cref{ch:g10:inertia}.
\end{remark}

\section{Memadukan gerak sepanjang satu garis}

Apa yang menghubungkan \qty{4}{km/h} milik si pejalan (kerangka kereta)
dengan \qty{304}{km/h} (\omterm{def:g10:relative-motion:frames}{kerangka tanah})? Sepanjang satu garis,
jawabannya adalah penjumlahan biasa.

\begin{proposition}[Perpaduan kelajuan pada satu dimensi]\label{prop:g10:relative-motion:compose}
Misalkan sebuah benda bergerak sepanjang sebuah garis di atas tumpuan
yang bergerak (jalur berjalan, kereta, air yang mengalir), dan tumpuan
itu sendiri meluncur sepanjang garis yang sama terhadap tanah. Dengan
memilih arah positif pada garis itu dan menghitung kelajuan beserta
tandanya,
\[
v_{\text{benda/tanah}} \;=\; v_{\text{benda/tumpuan}} \;+\;
v_{\text{tumpuan/tanah}} .
\]
Dengan kata-kata: kelajuan yang searah dijumlahkan; kelajuan yang
berlawanan arah dikurangkan.
\end{proposition}

\begin{proof}
Selama satu sekon, tumpuannya membawa bendanya sejauh
$v_{\text{tumpuan/tanah}}$ \omterm{def:g10:orders-of-magnitude:unit}{meter} sepanjang garis itu, dan bendanya maju
sejauh $v_{\text{benda/tumpuan}}$ \omterm{def:g10:orders-of-magnitude:unit}{meter} sepanjang tumpuannya. Kedua
perpindahan itu terjadi sepanjang garis yang sama, sehingga perpindahan
seluruhnya terhadap tanah selama sekon itu adalah jumlah aljabar
keduanya --- dan itulah lajunya terhadap tanah.
\end{proof}

\begin{example}[Jalur berjalan]\label{ex:g10:relative-motion:walkway}
Jalur berjalan di bandara bergerak \qty{1.0}{m/s}. Seorang pelancong
berjalan di atasnya dengan \qty{1.5}{m/s} terhadap sabuknya, searah
gerak sabuk: lajunya terhadap tanah $1.5 + 1.0 = \qty{2.5}{m/s}$.
Berjalan berlawanan arah dengan langkah yang sama:
$1.5 - 1.0 = \qty{0.5}{m/s}$, masih (pelan-pelan) maju melawan sabuknya.
Berdiri \omterm{def:g10:relative-motion:frame}{diam} di atas sabuknya: \qty{1.0}{m/s} terhadap tanah,
\qty{0}{m/s} terhadap sabuknya --- \omterm{def:g10:relative-motion:frame}{diam} sekaligus bergerak, di dalam dua
kerangka yang berbeda.
\end{example}

\begin{remark}[Dua arah sekaligus]\label{rem:g10:relative-motion:twodim}
Apabila kedua geraknya \emph{tidak} sepanjang garis yang sama --- sebuah
perahu yang membidik lurus menyeberangi sungai sementara arusnya
mendorongnya ke hilir --- masing-masing gerak berjalan sendiri-sendiri,
dan kecepatannya berpadu sebagai anak panah yang disusun ujung ke
pangkal, persis seperti vektor pada jilid matematika. Perahu itu tetap
mencapai seberang dalam waktu yang ditentukan kelajuan menyeberangnya
saja, tetapi ia mendarat di hilir titik bidikannya, dan \omterm{def:g10:relative-motion:trajectory}{lintasannya}
dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah} adalah garis lurus yang miring. Perlakuan vektor
yang lengkap datang pada bab berikutnya; di sini kita hanya membaca
gambarnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:1}
Seorang penumpang tertidur di dalam kereta yang melaju \qty{100}{km/h}.
Nyatakan apakah penumpang itu \omterm{def:g10:relative-motion:frame}{diam} atau bergerak, dan dengan laju
berapa, terhadap: gerbongnya; tanah; sebatang pohon di tepi rel; kereta
dari arah berlawanan yang melaju \qty{100}{km/h}. Lalu lukiskan gerak
pohon itu di dalam kerangka kereta si penumpang.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:2}
Ubahlah: \qty{72}{km/h} dan \qty{108}{km/h} menjadi \unit{m/s};
\qty{25}{m/s} dan \qty{340}{m/s} menjadi \unit{km/h}. Periksalah tiap
jawaban dengan aturan kewajaran pada
\cref{met:g10:relative-motion:convert}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:3}
Seorang pengendara sepeda menempuh \qty{36}{km} dalam
\qty{1}{h}~\qty{30}{min}; seorang pelari cepat menempuh \qty{100}{m}
dalam \qty{10.0}{s}. Hitunglah kedua kelajuan rata-ratanya dalam
\unit{m/s} dan dalam \unit{km/h}. Siapa yang lebih cepat, dan apakah
jawabannya mengejutkanmu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:4}
Sebuah jalur berjalan bergerak \qty{0.9}{m/s}. Seorang pelancong
berjalan \qty{1.4}{m/s} terhadap sabuknya. Sebutkan laju pelancong itu
terhadap tanah ketika berjalan searah sabuknya, ketika melawannya, dan
ketika berdiri \omterm{def:g10:relative-motion:frame}{diam} di atas sabuknya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:relative-motion:5}
Untuk setiap gerak berikut, pilihlah kerangka yang paling praktis di
antara \omterm{def:g10:relative-motion:frames}{kerangka tanah}, geosentrik dan heliosentrik, lalu berikan
alasannya dalam satu kalimat: sebuah pena yang dijatuhkan di dalam
kereta yang berjalan (tambahan: ada kerangka yang bahkan lebih baik);
\omterm{def:g10:universal-gravitation:satellite}{orbit} stasiun antariksa internasional; satu tahun Mars; lari cepat
\qty{100}{m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:6}
Sebuah kereta melaju \qty{108}{km/h}. Seorang penumpang berjalan ke
depan dengan \qty{1.5}{m/s} terhadap keretanya.
\begin{enumerate}
  \item Ubahlah laju keretanya ke \unit{m/s}, lalu sebutkan laju
        penumpang itu terhadap tanah, ketika berjalan ke depan dan
        ketika berjalan ke belakang.
  \item Gerbongnya sepanjang \qty{60}{m}. Berapa lama perjalanan ke
        depan itu, dan sejauh apa penumpang itu berpindah terhadap
        tanah selama perjalanan tersebut?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:7}
Laju sebuah perahu di air yang tenang adalah \qty{5.0}{m/s}; sebuah
sungai mengalir \qty{2.0}{m/s}. Perahu itu menempuh \qty{420}{m} ke
hilir, berputar arah, lalu kembali.
\begin{enumerate}
  \item Hitunglah laju perahu itu terhadap tepian pada tiap penggalnya,
        dan lama tiap penggalnya.
  \item Hitunglah kelajuan rata-rata sepanjang perjalanan pulang pergi
        itu. Mengapa nilainya kurang dari \qty{5.0}{m/s}, padahal
        arusnya membantu pada satu penggal persis sebanyak ia
        menghambat pada penggal yang lain?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:8}
Dua kereta berpapasan pada rel yang sejajar: kereta A (panjang
\qty{180}{m}) pada \qty{25}{m/s}, kereta B (panjang \qty{90}{m}) pada
\qty{20}{m/s}.
\begin{enumerate}
  \item Keduanya berjalan berlawanan arah. Berapa laju B di dalam
        kerangka A, dan berapa lama seluruh papasannya berlangsung
        (dari saat kedua moncongnya bertemu sampai kedua ekornya
        berpisah)?
  \item Pertanyaan yang sama jika keduanya berjalan searah.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:9}
Sebuah sepeda melaju di jalan yang lurus. Lukiskan \omterm{def:g10:relative-motion:trajectory}{lintasan} pentil
bannya di dalam: kerangka pengendaranya; \omterm{def:g10:relative-motion:frames}{kerangka tanah}; dan kerangka
pentil itu sendiri. Lukiskan pula \omterm{def:g10:relative-motion:trajectory}{lintasan} poros rodanya di dalam
\omterm{def:g10:relative-motion:frames}{kerangka tanah}. Tanpa perhitungan --- sebutkan saja nama tiap kurvanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:10}
Panjang khatulistiwa Bumi kira-kira \qty{40000}{km}, dan Bumi berputar
satu kali dalam \qty{24}{h}.
\begin{enumerate}
  \item Hitunglah laju sebuah titik di khatulistiwa di dalam \omterm{def:g10:relative-motion:frames}{kerangka
        geosentrik}, dalam \unit{km/h} dan dalam \unit{m/s}.
  \item Jelaskan dalam satu kalimat untuk masing-masing: mengapa kita
        tidak merasakan laju ini, dan mengapa ``Matahari terbit'' dan
        ``Bumi berputar'' melaporkan pengamatan yang sama.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:relative-motion:11}
Dengan berdiri \omterm{def:g10:relative-motion:frame}{diam} di atas eskalator yang bergerak, perjalanan naiknya
memakan \qty{60}{s}. Berjalan naik di eskalator yang sama ketika ia
dimatikan memakan \qty{90}{s}. Berapa lama berjalan naik di eskalator
yang bergerak? (Bekerjalah dengan kelajuannya, bukan dengan waktunya.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:12}
Seorang pengemudi menempuh jarak yang sama dua kali: berangkat pada
\qty{60}{km/h}, kembali pada \qty{90}{km/h}.
\begin{enumerate}
  \item Tunjukkan bahwa kelajuan rata-rata perjalanan pulang perginya
        adalah \qty{72}{km/h} --- bukan \qty{75}{km/h}.
  \item Jelaskan di mana letak kesalahan jawaban yang serampangan, lalu
        nyatakan penggal mana yang menguasai rata-ratanya.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:13}
Dua pengendara sepeda mulai dari jarak \qty{35}{km} dan mengayuh saling
mendekat, yang satu \qty{15}{km/h}, yang lain \qty{20}{km/h}.
\begin{enumerate}
  \item Lukiskan gerak pengendara kedua di dalam kerangka pengendara
        pertama, lalu simpulkan kapan keduanya bertemu.
  \item Sejauh apa masing-masing telah mengayuh pada saat bertemu?
  \item Jika sebaliknya keduanya mengayuh searah (yang lebih cepat
        berada di belakang), berapa lama pengejarannya berlangsung?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:14}
Seorang perenang menyeberangi sungai selebar \qty{60}{m}, menuju lurus
ke seberang dengan \qty{1.2}{m/s} terhadap airnya; arusnya mengalir
\qty{0.9}{m/s}.
\begin{enumerate}
  \item Jelaskan mengapa waktu menyeberangnya ditentukan oleh kelajuan
        menyeberang si perenang saja, lalu hitunglah waktunya.
  \item Sejauh apa ke hilir perenang itu mendarat?
  \item Dengan segitiga siku-siku, tentukan laju perenang itu di dalam
        \omterm{def:g10:relative-motion:frames}{kerangka tanah} dan panjang \omterm{def:g10:relative-motion:trajectory}{lintasan} yang benar-benar direnangi
        terhadap tanah.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:relative-motion:15}
Bumi mengorbit Matahari pada lingkaran hampir sempurna berjari-jari
\qty{1.50e11}{m}, dalam satu tahun ($\num{3.16e7}$ sekon).
\begin{enumerate}
  \item Hitunglah laju Bumi di dalam \omterm{def:g10:relative-motion:frames}{kerangka heliosentrik}, dalam
        \unit{km/s}.
  \item Bandingkan dengan laju perputaran khatulistiwa yang diperoleh
        pada \cref{exo:g10:relative-motion:10}.
  \item Dilihat dari Bumi, Mars kadang-kadang tampak berhenti lalu
        bergeser mundur di antara bintang selama beberapa pekan.
        Jelaskan dalam dua kalimat, dengan kosakata bab ini, mengapa
        \omterm{def:g10:relative-motion:frames}{kerangka heliosentrik} melarutkan kejanggalan itu.
\end{enumerate}
\end{exercise}

\section{Soal: Jalur berjalan, sungai, dan satelit}

\begin{problem}[{Soal akhir pekan --- jalur berjalan, sungai dan
satelit: siapa yang bergerak, terhadap apa, dan laporan kelajuan
terakhir bagi si pelancong yang duduk di kursi malas}]
\label{pb:g10:relative-motion:1}
Sebuah bandara, sebuah penyeberangan feri, dan sebuah titik terang yang
meluncur melintasi langit senja: tiga pemandangan sehari-hari, satu
pertanyaan yang diajukan tiga kali --- siapa yang bergerak, terhadap
apa? Tiap bagian memilih kerangkanya, menghitung, lalu menyaksikan
jawabannya berubah bersama sudut pandangnya; penutupnya menjumlahkan
seberapa cepat \emph{kamu} bergerak saat ini juga, sambil duduk
benar-benar \omterm{def:g10:relative-motion:frame}{diam}.

\medskip
\noindent\textbf{Bagian I --- Lomba di jalur berjalan.}
Sebuah jalur berjalan di bandara panjangnya \qty{90}{m} dan bergerak
\qty{1.0}{m/s}. Seorang pelancong berjalan \qty{1.5}{m/s} (terhadap apa
pun yang ada di bawah kakinya).
\begin{enumerate}
  \item Berapa lama jalur itu membawa pelancong yang berdiri \omterm{def:g10:relative-motion:frame}{diam} dari
        ujung ke ujung?
  \item Berapa lama pelancong itu berjalan di atas sabuk yang bergerak,
        searah dengan sabuknya?
  \item Berapa lama jika ia berjalan di samping jalur itu, di atas
        lantai yang \omterm{def:g10:relative-motion:frame}{diam}?
  \item Seorang pengusil berjalan melawan arah di atas sabuknya, dengan
        \qty{1.5}{m/s} terhadap sabuk itu. Berapa lajunya terhadap
        tanah, dan berapa lama ia sampai dari ujung jauh kembali ke
        pintu masuk?
  \item Seorang anak berlari melawan arah tepat \qty{1.0}{m/s} terhadap
        sabuknya. Lukiskan gerak anak itu di dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah}, dan
        sebutkan nama alat kebugaran yang ditemukan ulang di sini.
  \item Rangkumlah pertanyaan 1--5 dalam tabel dua lajur: laju tiap
        pelaku di dalam kerangka sabuk dan di dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah}.
        Lajur yang mana yang dirasakan kaki para pelancong itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Sungai.}
Sebuah feri menyeberangi sungai selebar \qty{120}{m}. Lajunya terhadap
air adalah \qty{2.0}{m/s}; arusnya mengalir \qty{1.5}{m/s}. Mula-mula
juru mudinya membidik lurus ke tepian seberang.
\begin{enumerate}[resume]
  \item Berapa lama penyeberangannya?
  \item Sejauh apa ke hilir feri itu mendarat?
  \item Dengan segitiga siku-siku, hitunglah laju feri itu di dalam
        \omterm{def:g10:relative-motion:frames}{kerangka tanah} dan panjang \omterm{def:g10:relative-motion:trajectory}{lintasannya} dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah}.
        (Sebuah tripel bilangan yang sudah dikenal akan muncul.)
  \item Lukiskan \omterm{def:g10:relative-motion:trajectory}{lintasan} feri itu di dalam kerangka air dan di dalam
        \omterm{def:g10:relative-motion:frames}{kerangka tanah}. Apa yang sama pada keduanya, dan apa yang
        berbeda?
  \item Seorang perenang yang lajunya terhadap air \qty{1.0}{m/s}
        berangkat dari tepian yang sama, membidik lurus ke seberang
        seperti ferinya. Berapa lama penyeberangannya, dan sejauh apa
        ke hilir perenang itu mendarat? Hitunglah laju perenang itu
        dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah} (lagi-lagi dengan segitiga siku-siku).
        Mengapa perenang yang lebih lambat justru hanyut lebih jauh
        daripada ferinya?
  \item Sebatang kayu hanyut melewati feri itu. Sebutkan lajunya di
        dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah} dan di dalam kerangka air. Di dalam
        kerangka yang mana kayu itu ``bergerak''? Berikan ulasanmu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{def:g10:relative-motion:trajectory}{Lintasan} \omterm{def:g10:universal-gravitation:satellite}{satelit}.}
Pada waktu senja, stasiun antariksa internasional melintasi langit
dalam beberapa menit. Ia mengorbit pada ketinggian \qty{400}{km};
jari-jari Bumi \qty{6.37e6}{m}, dan satu \omterm{def:g10:universal-gravitation:satellite}{orbit} memakan \qty{5.56e3}{s}
(sekitar \qty{93}{min}).
\begin{enumerate}[resume]
  \item Di dalam kerangka baku yang mana gerak stasiun itu paling
        sederhana? Hitunglah lajunya di dalam kerangka tersebut, dalam
        \unit{km/s}.
  \item Jelaskan dalam dua kalimat mengapa \omterm{def:g10:relative-motion:frames}{kerangka tanah} melukiskan
        stasiun itu dengan janggal --- apa yang terjadi pada jejaknya di
        atas permukaan Bumi antara satu \omterm{def:g10:universal-gravitation:satellite}{orbit} dan \omterm{def:g10:universal-gravitation:satellite}{orbit} berikutnya?
  \item Bandingkan laju \omterm{def:g10:universal-gravitation:satellite}{orbit} stasiun itu dengan laju perputaran
        khatulistiwa ($\approx \qty{0.46}{km/s}$): dengan faktor berapa
        stasiun itu lebih cepat?
  \item Sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} geostasioner mengorbit pada jari-jari
        \qty{4.22e7}{m} tepat dalam satu hari ($\qty{8.64e4}{s}$).
        Hitunglah lajunya di dalam \omterm{def:g10:relative-motion:frames}{kerangka geosentrik}, lalu lukiskan
        geraknya di dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah}. Damaikan kedua jawaban itu
        dalam satu kalimat.
  \item Di dalam \omterm{def:g10:relative-motion:frames}{kerangka geosentrik} Bulan melukis lingkaran hampir
        sempurna mengelilingi Bumi dalam sekitar $27$ hari. Sketsalah
        dengan kata-kata \omterm{def:g10:relative-motion:trajectory}{lintasannya} di dalam \omterm{def:g10:relative-motion:frames}{kerangka heliosentrik}
        sepanjang beberapa bulan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Siapa yang bergerak?}
\begin{enumerate}[resume]
  \item ``Matahari terbit di timur.'' ``Bumi berputar ke arah timur.''
        Sebutkan kerangka yang membuat tiap kalimat itu melukiskan
        pengamatannya, dan jelaskan mengapa tidak satu pun yang salah.
  \item Dalam satu paragraf yang tenang: apa yang dikerjakan dengan
        baik oleh kerangka Ptolemaios, apa yang disederhanakan kerangka
        Copernicus (sebutkan lengkung mundur Mars), dan mengapa putusan
        modernnya berkenaan dengan \emph{kepraktisan} dan bukan dengan
        satu kerangka yang benar?
  \item Laporan kelajuan si pelancong berkursi malas. Kamu duduk \omterm{def:g10:relative-motion:frame}{diam}
        di sebuah kursi malas di dekat khatulistiwa. Sebutkan lajumu di
        dalam \omterm{def:g10:relative-motion:frames}{kerangka tanah}, di dalam \omterm{def:g10:relative-motion:frames}{kerangka geosentrik}
        ($\approx \qty{0.46}{km/s}$), dan di dalam \omterm{def:g10:relative-motion:frames}{kerangka
        heliosentrik} ($\approx \qty{30}{km/s}$). Lalu hitunglah jarak
        yang kamu tempuh di dalam \omterm{def:g10:relative-motion:frames}{kerangka heliosentrik} selama film
        \qty{90}{min}, dan tutuplah soal ini dengan pesan moral bab ini
        dalam satu kalimat.
\end{enumerate}
\end{problem}
