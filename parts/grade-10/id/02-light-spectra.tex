\chapter{Spektrum Cahaya dan Pesan yang Dibawa Cahaya}\label{ch:g10:light-spectra}

Seberkas cahaya bintang adalah satu-satunya cuplikan bintang yang akan
pernah kita pegang, dan ternyata cuplikan itu sangat murah hati.
Uraikanlah berkas itu menjadi \omterm{def:g10:light-spectra:white}{spektrumnya} --- sebuah prisma kaca
sanggup melakukannya, tirai tetes hujan pun sanggup --- dan cahaya itu
berubah menjadi pesan: latar pelangi yang kontinu mengumumkan suhu
bintangnya, dan sederet garis halus mengejakan, unsur demi unsur, dari
apa bintang itu tersusun. Bab ini belajar membaca kedua bagian pesan
tersebut --- mula-mula pada lampu laboratorium, lalu pada Matahari.

\section{Cahaya putih dan spektrumnya}

\begin{definition}[Penguraian cahaya putih]\label{def:g10:light-spectra:white}
Ketika seberkas sempit cahaya matahari melewati prisma kaca, berkas itu
keluar terurai menjadi pita warna yang kontinu, dari merah sampai ungu.
Pita itulah \emph{spektrum}\index{spektrum} cahaya tersebut. Cahaya
yang memuat semua warna tampak, seperti cahaya matahari atau cahaya
bohlam biasa, disebut \emph{cahaya putih}\index{cahaya putih}. Cahaya
berwarna tunggal yang murni, yang tidak dapat diuraikan lagi oleh
prisma mana pun, bersifat
\emph{monokromatik}\index{cahaya monokromatik} (berkas laser,
misalnya); campuran beberapa warna bersifat
\emph{polikromatik}\index{cahaya polikromatik}. Alat yang melakukan
penguraian itu dan membaca susunan sebuah cahaya disebut
\emph{spektroskop}\index{spektroskop}.
\end{definition}

\begin{remark}[Prisma dan tetes hujan]\label{rem:g10:light-spectra:rainbow}
Prisma bekerja karena kaca membelokkan cahaya ungu sedikit lebih kuat
daripada cahaya merah --- \emph{mengapa} demikian termasuk pembahasan
pembiasan pada \cref{ch:g10:refraction}; di sini kita hanya memakai
akibatnya. Air melakukan pekerjaan yang sama: setiap tetes hujan yang
lewat berlaku sebagai prisma mungil, menerima cahaya matahari yang
putih dan mengembalikannya tersusun menurut warna. Pelangi adalah
\omterm{def:g10:light-spectra:white}{spektrum} Matahari, terlukis melintasi langit.
\end{remark}

\begin{definition}[Panjang gelombang, kartu pengenal sebuah radiasi]\label{def:g10:light-spectra:wavelength}
Setiap radiasi \omterm{def:g10:light-spectra:white}{monokromatik} dicirikan oleh satu bilangan, yaitu
\emph{panjang gelombang}\index{panjang gelombang} $\lambda$, yang
diukur dalam nanometer ($\qty{1}{nm} = \qty{e-9}{m}$). Panjang
gelombang adalah kartu pengenal radiasi itu: sebutkan $\lambda$ dan
segala sesuatu tentang warna murninya sudah terkatakan. Mata menanggapi
panjang gelombang antara kira-kira $\qty{400}{nm}$ (ungu) dan
$\qty{800}{nm}$ (merah) --- itulah \emph{spektrum tampak}\index{spektrum
tampak}. Di luar kedua ujungnya radiasi itu berlanjut, tak terlihat oleh
kita: \emph{ultraviolet}\index{ultraviolet} di bawah $\qty{400}{nm}$,
\emph{inframerah}\index{inframerah} di atas $\qty{800}{nm}$. Apa yang
sebenarnya diukur oleh panjang gelombang --- panjang satu riak
gelombang cahaya --- dijelaskan pada \cref{ch:g10:signals-and-waves};
pada bab ini, bilangannya sendirilah yang penting.
\end{definition}

\begin{example}[Membaca panjang gelombang]\label{ex:g10:light-spectra:classify}
Penunjuk laser hijau memancar pada $\qty{532}{nm}$: di dalam rentang
tampak, pada warna hijau. Lampu raksa pembasmi kuman memancar pada
$\qty{254}{nm}$: \omterm{def:g10:light-spectra:wavelength}{ultraviolet}, tak terlihat (dan berbahaya bagi mata
justru karena tak terlihat). Dioda pengendali jauh televisi memancar
pada $\qty{940}{nm}$: \omterm{def:g10:light-spectra:wavelength}{inframerah} --- arahkanlah ia ke kamera telepon,
yang pengindranya melihat sedikit di luar $\qty{800}{nm}$, dan kilatan
yang tak terlihat itu muncul pada layarnya.
\end{example}

\section{Spektrum kontinu dan suhu}

\begin{definition}[Spektrum kontinu]\label{def:g10:light-spectra:continuous}
Zat yang rapat dan panas --- kawat pijar bohlam, baja cair, permukaan
sebuah bintang --- berpijar, dan cahayanya memuat \emph{setiap} \omterm{def:g10:light-spectra:wavelength}{panjang
gelombang} pada rentang yang lebar: pita warna yang utuh tanpa satu pun
yang hilang. \omterm{def:g10:light-spectra:white}{Spektrum} semacam itu disebut \emph{spektrum
kontinu}\index{spektrum kontinu}. \omterm{def:g10:light-spectra:white}{Spektrum} itu tidak membawa jejak
sifat kimia bendanya: batang besi dan batang tembaga yang sama-sama
berpijar putih tampak serupa. Yang justru disandikannya --- seluruhnya
--- adalah suhu.
\end{definition}

\begin{definition}[Suhu mutlak]\label{def:g10:light-spectra:kelvin}
Untuk hukum-hukum radiasi, \emph{suhu mutlak}\index{suhu mutlak}
dihitung bukan dari titik beku air melainkan dari keadaan terdingin
yang mungkin, yaitu $\qty{-273}{\degreeCelsius}$. \omterm{def:g10:orders-of-magnitude:unit}{Satuannya} adalah
\emph{kelvin}\index{kelvin} (lambang $\unit{K}$), dengan besar derajat
yang sama:
\[
T \text{ (dalam } \unit{K}) = \theta \text{ (dalam } \unit{\degreeCelsius}) + 273 .
\]
Jadi ruangan bersuhu $\qty{20}{\degreeCelsius}$ berada pada
$\qty{293}{K}$, dan permukaan Matahari, yang mendekati
$\qty{5500}{\degreeCelsius}$, berada dekat $\qty{5800}{K}$.
\end{definition}

\begin{proposition}[Makin panas berarti makin terang dan makin biru (hukum Wien)]\label{prop:g10:light-spectra:wien}
Ketika suhu sebuah benda pijar yang rapat naik,
\begin{enumerate}
  \item benda itu memancar lebih kuat pada \emph{setiap} \omterm{def:g10:light-spectra:wavelength}{panjang
        gelombang}: seluruh \omterm{def:g10:light-spectra:white}{spektrumnya} menjadi lebih terang;
  \item \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} $\lambda_{\max}$ tempat benda itu memancar
        paling kuat bergeser ke sisi \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang pendek
        (biru), menurut
        \[
        \lambda_{\max} \times T = \qty{2.90e-3}{m.K},
        \]
        dengan $T$ adalah \omterm{def:g10:light-spectra:kelvin}{suhu mutlak} dalam \omterm{def:g10:light-spectra:kelvin}{kelvin}.
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[Dari mana hukum ini datang]\label{rem:g10:light-spectra:wienorigin}
Hukum Wien merangkum pengukuran cermat pada tanur dan kawat pijar yang
dilakukan pada akhir abad kesembilan belas. Menurunkannya memerlukan
teori kuantum tentang radiasi --- \omterm{def:g10:light-spectra:white}{spektrum} inilah yang justru
\emph{tidak} dapat dijelaskan fisika klasik, dan teka-teki inilah yang
memaksa Planck menciptakan kuantum. Penurunan yang jujur dikerjakan
pada jilid Tahun ke-3.
\end{remark}

\section{Spektrum garis gas yang tereksitasi}

\begin{definition}[Spektrum garis emisi]\label{def:g10:light-spectra:emission}
Gas bertekanan rendah yang dieksitasi oleh lucutan listrik (seperti di
dalam tabung neon atau lampu jalan natrium) juga berpijar --- tetapi
melalui \omterm{def:g10:light-spectra:white}{spektroskop} cahayanya sama sekali tidak menyerupai pelangi.
Hanya beberapa \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} terpencil yang hadir: garis-garis
terang yang tipis di atas latar hitam, yang disebut \emph{garis
spektrum}\index{garis spektrum}. \omterm{def:g10:light-spectra:white}{Spektrum} semacam itu adalah
\emph{spektrum garis emisi}\index{spektrum emisi}\index{spektrum garis}.
\end{definition}

\begin{example}[Hidrogen dan natrium]\label{ex:g10:light-spectra:hydrogen}
Hidrogen tereksitasi memancarkan tepat empat garis di daerah tampak:
garis merah pada $\qty{656}{nm}$, garis biru-hijau pada
$\qty{486}{nm}$, serta dua garis ungu pada $\qty{434}{nm}$ dan
$\qty{410}{nm}$. Natrium tereksitasi pada dasarnya memancarkan satu
garis kuning-jingga yang kuat pada $\qty{589}{nm}$ --- dan itulah
sebabnya lampu jalan model lama memandikan seluruh permukiman dengan
warna tunggal itu: cahayanya hampir tidak memuat apa pun yang lain.
\end{example}

\begin{proposition}[Sidik jari unsur]\label{prop:g10:light-spectra:fingerprint}
Setiap unsur kimia memancarkan daftar \omterm{def:g10:light-spectra:emission}{garis spektrumnya} sendiri yang
tetap, sama di setiap laboratorium dan pada setiap zaman, dan tidak ada
dua unsur yang memiliki daftar yang sama. Karena itu \omterm{def:g10:light-spectra:emission}{spektrum garis}
mengenali unsur yang memancarkannya, seperti sidik jari mengenali
seseorang.
\end{proposition}
\admitted

\begin{remark}[Mengapa berupa garis?]\label{rem:g10:light-spectra:whylines}
Mengapa sebuah atom hanya memancarkan \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} tertentu ---
dan mengapa \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} itu membocorkan unsurnya --- dijelaskan
oleh teori kuantum tentang atom, pada tahun terakhir jilid ini: setiap
unsur memiliki tangga tingkat energinya sendiri, dan setiap garis
menandai satu lompatan antara dua anak tangga. Untuk sementara kita
memakai sidik jari itu sebagaimana detektif memakainya: tanpa bertanya
kepada jari bagaimana ia tumbuh.
\end{remark}

\section{Spektrum serapan}

\begin{definition}[Spektrum serapan]\label{def:g10:light-spectra:absorption}
Lewatkanlah \omterm{def:g10:light-spectra:white}{cahaya putih} \emph{menembus} gas dingin bertekanan rendah,
lalu uraikanlah apa yang keluar: pelangi kontinu itu muncul kembali,
tetapi tergores garis-garis gelap yang tipis. Gas itu telah mengambil
\omterm{def:g10:light-spectra:wavelength}{panjang gelombang} tertentu dari cahaya yang lewat. \omterm{def:g10:light-spectra:white}{Spektrum} semacam itu
--- latar kontinu dikurangi sekumpulan garis --- adalah \emph{spektrum
serapan}\index{spektrum serapan}.
\end{definition}

\begin{proposition}[Garis emisi dan garis serapan berimpit]\label{prop:g10:light-spectra:kirchhoff}
Sebuah gas menyerap tepat \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang sanggup
dipancarkannya: garis gelap pada \omterm{def:g10:light-spectra:absorption}{spektrum serapan} sebuah unsur berada
pada \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang sama dengan garis terang pada \omterm{def:g10:light-spectra:emission}{spektrum
emisinya}. Karena itu sidik jari dari
\cref{prop:g10:light-spectra:fingerprint} dapat dibaca baik dalam gelap
maupun dalam terang.
\end{proposition}
\admitted

\begin{remark}[Satu mekanisme, dua tanda]\label{rem:g10:light-spectra:mechanism}
Keberimpitan itu bukan kebetulan: menyerap pada $\lambda$ menaiki anak
tangga energi yang sama dengan yang dituruni ketika memancar pada
$\lambda$ --- bab-bab kuantum di akhir jilid ini membuatnya tepat.
Ditetapkan oleh Kirchhoff dan Bunsen pada tahun 1859, hubungan itu
adalah kunci yang membuka seluruh bagian berikutnya.
\end{remark}

\begin{example}[Sebuah labu berisi uap natrium]\label{ex:g10:light-spectra:sodiumflask}
\omterm{def:g10:light-spectra:white}{Cahaya putih} melewati labu berisi uap natrium yang dingin. \omterm{def:g10:light-spectra:white}{Spektrum}
cahaya yang diteruskan berupa pelangi utuh dengan satu garis gelap pada

samar pada \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang sama.
\end{example}

\section{Analisis spektrum: membaca sebuah bintang}

\begin{definition}[Analisis spektrum]\label{def:g10:light-spectra:analysis}
Yang dimaksud \emph{analisis spektrum}\index{analisis spektrum} adalah
pengenalan unsur kimia yang terkandung di dalam sebuah sumber (nyala
api, lampu, bintang) dengan mencocokkan garis pada \omterm{def:g10:light-spectra:white}{spektrumnya} terhadap
daftar laboratorium untuk setiap unsur.
\end{definition}

\begin{method}[Membaca spektrum sebuah bintang]\label{met:g10:light-spectra:reading}
Sebuah bintang adalah bola zat panas yang rapat (permukaannya)
terbungkus atmosfer yang lebih tipis dan lebih dingin. Karena itu
cahayanya tiba sebagai \omterm{def:g10:light-spectra:continuous}{spektrum kontinu} yang tergores garis serapan ---
dan tiap bagiannya dibaca terpisah:
\begin{enumerate}
  \item Tentukan letak puncak $\lambda_{\max}$ latar kontinunya; hukum
        Wien (\cref{prop:g10:light-spectra:wien}) memberikan suhu
        permukaan $T = \qty{2.90e-3}{m.K} / \lambda_{\max}$.
  \item Daftarkan \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} garis-garis gelapnya.
  \item Cocokkan daftar itu dengan katalog: sebuah unsur baru boleh
        dinyatakan hadir apabila \emph{semua} garis kuatnya muncul.
  \item Simpulkan: penyerapannya terjadi di atmosfer bintang itu,
        sehingga unsur yang dikenali adalah unsur atmosfernya.
\end{enumerate}
\end{method}

\begin{example}[Resep Matahari]\label{ex:g10:light-spectra:sunrecipe}
\omterm{def:g10:light-spectra:white}{Spektrum} matahari membawa ribuan garis gelap, yang dipetakan Fraunhofer
pada tahun 1814. Di antara yang terkuat: $410$, $434$, $486$ dan
$\qty{656}{nm}$ --- sidik jari hidrogen yang lengkap --- bersama garis
natrium pada $\qty{589}{nm}$ dan sepasang garis kalsium pada $393$ dan
$\qty{397}{nm}$. Atmosfer Matahari terutama memuat hidrogen, ditambah
jejak unsur yang dikenal dari bebatuan Bumi. Pengukuran modern
mempertajam kesimpulannya: kira-kira tiga perempat massa Matahari
adalah hidrogen.
\end{example}

\begin{remark}[Unsur yang ditemukan di langit lebih dahulu]\label{rem:g10:light-spectra:helium}
Pada gerhana tahun 1868, sebuah garis terang pada $\qty{587.6}{nm}$
muncul di dalam \omterm{def:g10:light-spectra:white}{spektrum} atmosfer Matahari --- dan tidak cocok dengan
unsur mana pun yang dikenal. Menurut
\cref{prop:g10:light-spectra:fingerprint}, sidik jari yang tak dikenal
berarti unsur yang tak dikenal, sehingga sebuah unsur diumumkan dan
dinamai menurut matahari dalam bahasa Yunani, \emph{helios}: helium.
Unsur itu baru berhasil dipisahkan di Bumi pada tahun 1895 --- unsur
kimia satu-satunya yang ditemukan di ruang angkasa sebelum ditemukan di
rumah sendiri.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:light-spectra:1}
Untuk setiap radiasi, nyatakan apakah ia \omterm{def:g10:light-spectra:wavelength}{ultraviolet}, tampak atau
\omterm{def:g10:light-spectra:wavelength}{inframerah}, dan sebutkan warnanya jika tampak: $\qty{254}{nm}$ (lampu
raksa), $\qty{480}{nm}$, $\qty{589}{nm}$, $\qty{656}{nm}$,
$\qty{1064}{nm}$ (laser pemotong).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:light-spectra:2}
Seberkas sempit cahaya matahari jatuh pada sebuah prisma kaca.
\begin{enumerate}
  \item Uraikan apa yang tampak pada layar yang dipasang di belakang
        prisma itu.
  \item Ujung pita yang manakah yang paling banyak dibelokkan?
  \item Cahaya matahari itu diganti dengan berkas penunjuk laser hijau.
        Apa yang kini tampak pada layarnya, dan apa yang ditunjukkan
        hal itu tentang cahaya laser?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:light-spectra:3}
Pasangkan setiap sumber dengan jenis \omterm{def:g10:light-spectra:white}{spektrum} yang dihasilkannya
(kontinu, garis emisi, atau serapan): (a) kawat pijar sebuah bohlam;
(b) tabung reklame neon; (c) besi cair di sebuah pengecoran; (d) lampu
jalan natrium; (e) cahaya matahari yang diterima di Bumi.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:light-spectra:4}
Tentang pelangi:
\begin{enumerate}
  \item Apa yang berperan sebagai prisma?
  \item Pelangi hanya tampak apabila Matahari berada di belakang
        pengamat dan hujan berada di depannya. Apa yang disumbangkan
        masing-masing bahan itu?
  \item Jelaskan mengapa pelangi membuktikan bahwa cahaya matahari
        adalah \omterm{def:g10:light-spectra:white}{cahaya putih}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:light-spectra:5}
Seorang pandai besi memanaskan sebatang paku besi: paku itu berpijar
merah suram, lalu jingga terang, lalu hampir putih.
\begin{enumerate}
  \item Urutkan ketiga tahap itu menurut suhunya.
  \item Dua akibat yang mana dari \cref{prop:g10:light-spectra:wien}
        yang diperagakan oleh urutan itu?
  \item Benarkan ungkapan ``pijar putih lebih panas daripada pijar
        merah''.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:light-spectra:6}
\omterm{def:g10:light-spectra:continuous}{Spektrum kontinu} Matahari memuncak pada $\lambda_{\max} = \qty{500}{nm}$.
\begin{enumerate}
  \item Hitunglah suhu permukaan Matahari dalam \omterm{def:g10:light-spectra:kelvin}{kelvin}.
  \item Ubahlah ke derajat Celsius.
  \item Periksalah bahwa puncaknya terletak di dalam rentang tampak,
        lalu sebutkan warnanya.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:light-spectra:7}
Nyala lilin berpijar pada sekitar $\qty{1800}{K}$; sebuah pemanas
listrik yang disetel ``hangat'' berada pada sekitar $\qty{500}{K}$.
\begin{enumerate}
  \item Hitunglah $\lambda_{\max}$ untuk masing-masing sumber, dan
        sebutkan daerahnya.
  \item Pemanas itu jelas panas --- tangan yang ditahan di atasnya
        merasakan radiasinya --- namun ia tampak hitam. Jelaskan.
  \item Puncak nyala lilin pun berada di luar rentang tampak. Lalu
        mengapa kita tetap dapat melihat nyalanya?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:light-spectra:8}
Katalog laboratorium memberikan, di daerah tampak: hidrogen
$\{410, 434, 486, 656\}\,\unit{nm}$; natrium $\{589\}\,\unit{nm}$;
helium $\{447, 502, 588, 668\}\,\unit{nm}$. Dua lampu lucutan
dianalisis. Lampu A menunjukkan garis pada $434$, $486$ dan
$\qty{656}{nm}$, ditambah garis ungu samar di dekat tepi daerah tampak;
lampu B menunjukkan satu garis kuat pada $\qty{589}{nm}$. Kenalilah gas
di dalam masing-masing lampu, dan berikan alasan yang cermat mengapa
lampu B tidak memuat helium.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:light-spectra:9}
\omterm{def:g10:light-spectra:white}{Cahaya putih} dilewatkan menembus labu berisi uap natrium yang dingin.
\begin{enumerate}
  \item Uraikan \omterm{def:g10:light-spectra:white}{spektrum} cahaya yang diteruskan.
  \item Pada \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} berapa garis gelapnya berada, dan
        mengapa persis di sana?
  \item Lampunya dimatikan, ruangannya digelapkan, lalu uapnya
        dieksitasi oleh lucutan listrik. Uraikan \omterm{def:g10:light-spectra:white}{spektrum} yang baru.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:light-spectra:10}
\omterm{def:g10:light-spectra:white}{Spektrum} Matahari beresolusi tinggi menunjukkan garis gelap pada $410$,
$434$, $486$, $517$, $518$ dan $\qty{656}{nm}$. Katalog: hidrogen
$\{410, 434, 486, 656\}\,\unit{nm}$; magnesium $\{517, 518\}\,\unit{nm}$.
\begin{enumerate}
  \item Unsur apa saja yang diungkapkan garis-garis itu?
  \item Di bagian Matahari yang mana garis itu terbentuk, dan mengapa
        garisnya gelap dan bukan terang?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:light-spectra:11}
Tubuhmu memiliki suhu permukaan sekitar $\qty{37}{\degreeCelsius}$.
\begin{enumerate}
  \item Ubahlah ke \omterm{def:g10:light-spectra:kelvin}{kelvin} dan hitunglah $\lambda_{\max}$ radiasi termal
        tubuhmu sendiri.
  \item Pada daerah yang mana \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} itu jatuh?
  \item Simpulkan mengapa tidak ada orang yang terlihat berpijar di
        ruangan gelap, dan bagaimana kamera termal tetap menemukan
        orang.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:light-spectra:12}
Kawat pijar wolfram sebuah bohlam bekerja pada $\qty{2700}{K}$.
\begin{enumerate}
  \item Hitunglah $\lambda_{\max}$ dan sebutkan daerahnya.
  \item Jelaskan mengapa bohlam semacam itu menjadi lampu yang buruk
        tetapi pemanas kecil yang sangat baik.
  \item Suhu kawat pijar berapa yang akan menempatkan $\lambda_{\max}$
        pada $\qty{550}{nm}$, di tengah daerah tampak? Wolfram melebur
        pada $\qty{3700}{K}$; simpulkan mengapa tidak ada bohlam
        berkawat pijar yang sanggup menirukan cahaya siang, dan mengapa
        penerangan beralih ke teknologi lain.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:light-spectra:13}
\omterm{def:g10:light-spectra:white}{Spektrum} sebuah bintang menunjukkan latar kontinu yang memuncak pada
$\qty{420}{nm}$, tergores garis gelap pada $393$, $397$, $410$, $434$,
$486$ dan $\qty{656}{nm}$. Katalog: hidrogen
$\{410, 434, 486, 656\}\,\unit{nm}$; kalsium $\{393, 397\}\,\unit{nm}$;
natrium $\{589\}\,\unit{nm}$.
\begin{enumerate}
  \item Hitunglah suhu permukaan bintang itu. Lebih panas ataukah lebih
        dingin daripada Matahari?
  \item Unsur apa saja yang pasti terkandung di dalam atmosfernya?
  \item Apakah natrium melimpah di atmosfer itu? Berikan alasannya.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:light-spectra:14}
Sebuah lampu lucutan memuat campuran dua gas. \omterm{def:g10:light-spectra:white}{Spektrumnya} menunjukkan
garis pada $410$, $434$, $447$, $486$, $502$, $588$, $656$ dan
$\qty{668}{nm}$. Dengan memakai katalog pada
\cref{exo:g10:light-spectra:8}:
\begin{enumerate}
  \item Kenalilah kedua gas itu.
  \item Seorang teman sekelas mengaku bahwa garis $\qty{588}{nm}$
        membuktikan lampu itu juga memuat natrium (garisnya pada
        $\qty{589}{nm}$), sebab \omterm{def:g10:light-spectra:white}{spektroskop} murah tidak sanggup
        memisahkan \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} yang terpaut $\qty{1}{nm}$. Tanpa
        alat yang lebih baik, mengapa helium sudah merupakan penjelasan
        yang memadai bagi garis itu --- dan pengukuran apa yang akan
        menuntaskan pertanyaan tentang natrium untuk selamanya?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:light-spectra:15}
Di rasi Orion, Betelgeuse berpijar merah dengan jelas dan Rigel
biru-putih. \omterm{def:g10:light-spectra:continuous}{Spektrum kontinu} keduanya memuncak berturut-turut pada
sekitar $\qty{850}{nm}$ dan $\qty{240}{nm}$.
\begin{enumerate}
  \item Hitunglah kedua suhu permukaannya.
  \item Peroleh kembali nisbah kedua suhu itu langsung dari kedua
        \omterm{def:g10:light-spectra:wavelength}{panjang gelombangnya}, tanpa menghitung ulang suhunya.
  \item Jelaskan kedua warnanya, mengingat \emph{tidak satu pun}
        puncaknya terletak di daerah tampak.
  \item Kedua bintang itu terang bagi mata telanjang. Mengapa puncak di
        luar daerah tampak tidak membuat sebuah bintang menjadi tak
        terlihat?
\end{enumerate}
\end{exercise}

\section{Soal: Membaca cahaya bintang}

\begin{problem}[{Soal akhir pekan --- membaca cahaya bintang: filsuf
yang berkata ``tidak akan pernah'', dan bagaimana sebuah prisma
mengukur suhu bintang serta membaca resep kimianya}]
\label{pb:g10:light-spectra:1}
Pada tahun 1835 filsuf Auguste Comte memerlukan contoh pengetahuan yang
selamanya berada di luar jangkauan manusia, dan ia memilihnya dengan
cermat: susunan kimia bintang. Tidak akan pernah ada orang yang
membotolkan sebuah bintang. Pada tahun 1859 Kirchhoff dan Bunsen
mencocokkan garis gelap Matahari dengan garis terang nyala api
laboratorium, dan pengetahuan yang tak terjangkau itu tiba bersama
cahaya yang kembali. Soal ini menelusuri kembali seluruh jalannya:
sidik jari laboratorium, benda berpijar, garis gelap Matahari --- dan
akhirnya sebuah bintang yang suhu serta susunannya akan kamu tentukan
dari meja belajarmu.

\medskip
\noindent\textbf{Bagian I --- Sidik jari yang terarsip.}
\begin{enumerate}
  \item Sebutkan rentang \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} tampak dalam nanometer dan
        warna pada tiap ujungnya. Di mana letak $\qty{550}{nm}$?
  \item Lampu lucutan hidrogen menunjukkan garis pada $410$, $434$,
        $486$ dan $\qty{656}{nm}$. Sebutkan warna tiap garisnya.
  \item Lampu natrium pada dasarnya memancarkan satu garis pada
        $\qty{589}{nm}$. Bagaimana lampu itu tampak oleh mata, dan
        mengapa sia-sia menilai warna mobil yang terparkir di bawah
        lampu jalan semacam itu?
  \item Jelaskan mengapa sekumpulan \omterm{def:g10:light-spectra:emission}{garis spektrum} mengenali sebuah
        unsur --- dan mengapa \emph{semua} garis kuat sebuah unsur
        harus ditemukan lebih dahulu sebelum unsur itu dinyatakan
        hadir.
  \item Ramalkan \omterm{def:g10:light-spectra:white}{spektrum} yang tampak (a) ketika \omterm{def:g10:light-spectra:white}{cahaya putih} melewati
        labu berisi uap natrium yang dingin, dan (b) ketika uap yang
        sama, sendirian di ruangan gelap, dieksitasi oleh lucutan
        listrik.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Benda panas dan warnanya.}
\begin{enumerate}[resume]
  \item Sebutkan dua akibat kenaikan suhu sebuah benda pijar yang rapat
        terhadap \omterm{def:g10:light-spectra:continuous}{spektrum kontinunya}.
  \item Kawat pijar halogen bekerja pada $\qty{3400}{K}$. Hitunglah
        $\lambda_{\max}$ dan sebutkan daerahnya.
  \item \omterm{def:g10:light-spectra:continuous}{Spektrum kontinu} Matahari memuncak pada $\qty{500}{nm}$.
        Hitunglah suhu permukaannya.
  \item Puncak itu terletak di daerah hijau --- namun Matahari tidak
        tampak hijau. Jelaskan.
  \item Sebatang besi yang ditinggalkan di dalam tungku mula-mula
        terasa panas dari jauh tanpa terlihat berpijar, lalu berpijar
        merah suram, lalu jingga keputihan. Jelaskan seluruh urutan itu
        dengan pertanyaan 6.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Garis gelap Matahari.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:light-spectra:white}{Spektrum} matahari berupa pita kontinu yang dilintasi ribuan
        garis gelap yang halus. Bagian Matahari yang mana yang
        menghasilkan latar kontinunya, dan bagian mana yang
        menghasilkan garisnya?
  \item Garis matahari yang kuat berada pada $393$, $397$, $410$,
        $434$, $486$, $589$ dan $\qty{656}{nm}$. Katalog: hidrogen
        $\{410, 434, 486, 656\}\,\unit{nm}$; natrium
        $\{589\}\,\unit{nm}$; kalsium $\{393, 397\}\,\unit{nm}$. Unsur
        apa saja yang hadir di atmosfer Matahari?
  \item Garis hidrogennyalah yang jauh paling kuat. Apa yang
        disarankan hal itu tentang susunan Matahari --- dan apa yang
        dikatakan pengukuran modern?
  \item Pada tahun 1868, sebuah garis pada $\qty{587.6}{nm}$ yang
        teramati di atmosfer Matahari tidak cocok dengan katalog mana
        pun. Susunlah kembali penalaran yang membenarkan pengumuman
        sebuah unsur baru, dan ceritakan bagaimana kisahnya berakhir.
  \item Mengapa garis matahari itu gelap, padahal atom yang sama di
        dalam lucutan laboratorium memberikan garis terang? Pada
        gerhana total, selama beberapa sekon, hanya atmosfer tipis
        Matahari yang tetap terlihat di depan langit yang gelap --- dan
        garis yang sama berkelebat \emph{terang}. Jelaskan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Sebuah bintang di atas meja.}
\omterm{def:g10:light-spectra:white}{Spektrum} bintang terang Vega dihamparkan di hadapanmu: latar kontinu
yang memuncak pada $\qty{290}{nm}$, tergores garis gelap yang kuat pada
$410$, $434$, $486$ dan $\qty{656}{nm}$.
\begin{enumerate}[resume]
  \item Hitunglah suhu permukaan Vega.
  \item Puncaknya terletak di daerah \omterm{def:g10:light-spectra:wavelength}{ultraviolet}. Ramalkan warna tampak
        bintang itu, lalu damaikan ``puncaknya di luar daerah tampak''
        dengan ``bintang yang terang''.
  \item Kenalilah unsur yang menguasai atmosfer Vega.
  \item Bandingkan Vega dengan Matahari: nisbah suhunya, warnanya, dan
        keterangannya per \omterm{def:g10:orders-of-magnitude:unit}{satuan} luas permukaan yang berpijar.
  \item Penutup --- isilah formulir jati diri yang oleh Comte
        dinyatakan mustahil: suhu, unsur yang menguasai, dan warna
        tampak Vega, semuanya terbaca dari satu berkas cahaya. Berapa
        tahun ``tidak akan pernah'' itu bertahan?
\end{enumerate}
\end{problem}
