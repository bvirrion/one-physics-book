\chapter{Orde Besar: Mengukur Alam Semesta}\label{ch:g10:orders-of-magnitude}

Fisika bermula di tempat pendapat berakhir: pada sebuah pengukuran.
Namun benda yang diukur fisika terbentang pada rentang yang ganjil ---
sebuah inti atom berukuran sekitar \qty{e-15}{m}, alam semesta teramati
sekitar \qty{e27}{m}: selisih faktor $10^{42}$, jauh melampaui apa pun
yang sanggup dibayangkan. Bab ini memasang perkakas yang menjinakkan
rentang itu --- \omterm{def:g10:orders-of-magnitude:unit}{satuan}, pangkat sepuluh, \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} --- serta dua
kebiasaan yang dipakai pada setiap halaman berikutnya: tulislah
\omterm{def:g10:orders-of-magnitude:unit}{satuannya}, dan periksalah \omterm{def:g10:orders-of-magnitude:oom}{orde besarnya}.

\section{Satuan: tata bahasa pengukuran}

\begin{definition}[Besaran fisis dan satuan SI]\label{def:g10:orders-of-magnitude:unit}
Sebuah \emph{besaran fisis}\index{besaran fisis} adalah sifat dunia yang
dapat diukur: panjang, selang waktu, massa. Mengukurnya berarti
menghitung berapa kali sebuah besaran acuan --- yaitu
\emph{satuan}\index{satuan} --- termuat di dalamnya.
\emph{Sistem Satuan Internasional}\index{Sistem Satuan Internasional
(SI)} (SI) menetapkan tujuh satuan dasar, dan tiga di antaranya
menjalankan seluruh jilid ini: \emph{meter}\index{meter} (\unit{m})
untuk panjang, \emph{kilogram}\index{kilogram} (\unit{kg}) untuk massa,
dan \emph{sekon}\index{sekon} (\unit{s}) untuk waktu. (Empat sisanya
--- ampere, kelvin, mol, kandela --- menyusul ketika listrik, kalor dan
kimia memasuki cerita.)
\end{definition}

\begin{remark}[Bilangan tanpa satuan bukanlah hasil]\label{rem:g10:orders-of-magnitude:units}
``Jaraknya $384$'' tidak berarti apa-apa: \qty{384}{mm},
\qty{384}{km} dan \qty{384}{Mm} adalah tiga dunia yang berbeda (yang
terakhir kebetulan jarak ke Bulan). Pada tahun 1999, NASA kehilangan
\emph{Mars Climate Orbiter} seharga \$327 juta karena satu program
menghitung gaya pendorong dalam pound sementara program lain
membacanya dalam newton. Aturan rumah, dari baris ini sampai akhir
buku: setiap bilangan yang diukur atau dihitung membawa \omterm{def:g10:orders-of-magnitude:unit}{satuannya}, pada
setiap baris perhitungan.
\end{remark}

\begin{definition}[Awalan SI]\label{def:g10:orders-of-magnitude:prefixes}
Sebuah \emph{awalan}\index{awalan (SI)} yang dilekatkan pada \omterm{def:g10:orders-of-magnitude:unit}{satuan}
mengalikan \omterm{def:g10:orders-of-magnitude:unit}{satuan} itu dengan pangkat sepuluh yang tetap:
\begin{center}
\begin{tabular}{llc@{\qquad\qquad}llc}
femto & f & $10^{-15}$ & centi & c & $10^{-2}$ \\
pico  & p & $10^{-12}$ & kilo  & k & $10^{3}$  \\
nano  & n & $10^{-9}$  & mega  & M & $10^{6}$  \\
micro & $\mu$ & $10^{-6}$ & giga & G & $10^{9}$ \\
milli & m & $10^{-3}$  & tera  & T & $10^{12}$ \\
\end{tabular}
\end{center}
Jadi $\qty{1}{nm} = \qty{e-9}{m}$ dan $\qty{1}{km} = \qty{e3}{m}$.
\end{definition}

\begin{example}[Membaca awalan]\label{ex:g10:orders-of-magnitude:prefixes}
Cahaya hijau memiliki panjang gelombang sekitar
$\qty{500}{nm} = \qty{5.00e-7}{m}$; sehelai rambut manusia tebalnya
sekitar $\qty{70}{\micro m} = \qty{7.0e-5}{m}$; Bulan mengorbit pada
$\qty{384}{Mm} = \qty{3.84e8}{m}$. Satu tetes hujan bermassa sekitar
$\qty{30}{mg} = \qty{3.0e-5}{kg}$ --- perhatikan bahwa \omterm{def:g10:orders-of-magnitude:unit}{satuan} dasar SI
untuk massa, satu-satunya di antara ketujuhnya, sudah membawa \omterm{def:g10:orders-of-magnitude:prefixes}{awalan}.
\end{example}

\section{Notasi ilmiah dan angka penting}

\begin{definition}[Notasi ilmiah]\label{def:g10:orders-of-magnitude:scinot}
Sebuah bilangan ditulis dalam \emph{notasi ilmiah}\index{notasi ilmiah}
apabila berbentuk
\[
a \times 10^{n}, \qquad 1 \leq a < 10, \quad n \text{ bilangan bulat.}
\]
Faktor $a$ disebut \emph{mantisa}\index{mantisa}, dan bilangan bulat
$n$ disebut \emph{eksponen}\index{eksponen}.
Jadi $\num{149600000} = \num{1.496e8}$ dan
$\num{0.00052} = \num{5.2e-4}$.
\end{definition}

\begin{proposition}[Berhitung dengan pangkat sepuluh]\label{prop:g10:orders-of-magnitude:powers}
Untuk setiap bilangan bulat $m$ dan $n$ berlaku:
\[
10^{m} \times 10^{n} = 10^{m+n}, \qquad
\frac{1}{10^{n}} = 10^{-n}, \qquad
\left(10^{m}\right)^{n} = 10^{mn}.
\]
\end{proposition}

\begin{proof}[Bukti sebagian]
Untuk \omterm{def:g10:orders-of-magnitude:scinot}{eksponen} positif, hitunglah faktornya: $10^{m} \times 10^{n}$
adalah hasil kali $m + n$ buah sepuluh, sedangkan $(10^m)^n$ merangkai
$n$ kelompok berisi $m$ sepuluh. Menulis $10^{-n}$ untuk $1/10^{n}$
justru merupakan kesepakatan yang menjaga aturan pertama tetap benar
untuk semua bilangan bulat: kesepakatan itu memaksa
$10^{n} \times 10^{-n} = 10^{0} = 1$. Jilid matematika pendamping
membahas aturan \omterm{def:g10:orders-of-magnitude:scinot}{eksponen} secara lengkap.
\end{proof}

\begin{example}[Perkalian dalam notasi ilmiah]\label{ex:g10:orders-of-magnitude:scinot}
Berapa banyak Bumi yang muat di dalam Matahari, menurut massa? Dengan
$m_{\text{Matahari}} = \qty{1.99e30}{kg}$ dan
$m_{\text{Bumi}} = \qty{5.97e24}{kg}$:
\[
\frac{\num{1.99e30}}{\num{5.97e24}}
= \frac{1.99}{5.97} \times 10^{30-24}
= 0.333 \times 10^{6}
= \num{3.33e5}.
\]
\omterm{def:g10:orders-of-magnitude:scinot}{Mantisa} dan \omterm{def:g10:orders-of-magnitude:scinot}{eksponen} ditangani secara terpisah; jawaban akhirnya
dinormalkan kembali agar \omterm{def:g10:orders-of-magnitude:scinot}{mantisanya} jatuh kembali di dalam $[1, 10)$.
\end{example}

\begin{definition}[Angka penting]\label{def:g10:orders-of-magnitude:sigfig}
Yang disebut \emph{angka penting}\index{angka penting} sebuah bilangan
hasil pengukuran adalah angka yang benar-benar dijamin oleh pengukuran
itu: semua angka yang tertulis kecuali nol di depan. Nol di belakang
ikut dihitung --- $\qty{5.20e-3}{s}$ (tiga angka penting) menjanjikan
lebih banyak daripada $\qty{5.2e-3}{s}$ (dua). \omterm{def:g10:orders-of-magnitude:scinot}{Notasi ilmiah}
memperlihatkan cacahnya sekali pandang: \omterm{def:g10:orders-of-magnitude:scinot}{mantisanya} membawa tepat angka
pentingnya.
\end{definition}

\begin{method}[Berapa angka yang harus dipertahankan]\label{met:g10:orders-of-magnitude:sigfig}
Sebuah hasil perhitungan tidak mungkin lebih teliti daripada bahannya.
\begin{enumerate}
  \item Pada perkalian atau pembagian, pertahankan \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting}
        sebanyak \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} faktor yang \emph{paling} tidak teliti.
  \item Bulatkan hanya jawaban akhirnya, jangan pernah langkah
        antaranya.
  \item Jika ragu, tiga \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting} adalah ketelitian kerja buku ini.
\end{enumerate}
\end{method}

\begin{example}[Keliling Bumi]\label{ex:g10:orders-of-magnitude:sigfig}
Diameter Bumi adalah $d = \qty{1.27e7}{m}$ (tiga \omterm{def:g10:orders-of-magnitude:sigfig}{angka penting}),
sehingga kelilingnya
$\pi d = \num{3.14159} \times \qty{1.27e7}{m} = \qty{3.99e7}{m}$ ---
tiga angka, bukan delapan angka yang ditampilkan kalkulator. Menulis
$\qty{39898227}{m}$ berarti mengaku mengenal planet itu sampai ke
\omterm{def:g10:orders-of-magnitude:unit}{meternya}.
\end{example}

\section{Tangga alam semesta}

\begin{definition}[Orde besar]\label{def:g10:orders-of-magnitude:oom}
Yang disebut \emph{orde besar}\index{orde besar} sebuah besaran adalah
pangkat sepuluh yang paling dekat dengannya: tulislah besaran itu
sebagai $a \times 10^{n}$ dengan $1 \leq a < 10$, lalu ambil $10^{n}$
jika $a < 5$ dan $10^{n+1}$ jika $a \geq 5$. (Kesepakatan batas yang
persis tidak pernah penting: pernyataan orde besar hanya dimaksudkan
sampai ketelitian faktor sepuluh.)
\end{definition}

\begin{example}[Orde besar]\label{ex:g10:orders-of-magnitude:oom}
Hidup manusia berlangsung sekitar $80$ tahun, yaitu
$80 \times \qty{3.16e7}{s} \approx \qty{2.5e9}{s}$: \omterm{def:g10:orders-of-magnitude:oom}{orde besarnya}
$\qty{e9}{s}$ --- beberapa miliar \omterm{def:g10:orders-of-magnitude:unit}{sekon}, gunakanlah dengan baik. Tinggi
manusia $\qty{1.7}{m}$: ordenya $\qty{e0}{m}$. Gunung Everest,
$\qty{8848}{m}$: ordenya $\qty{e4}{m}$, karena $8.848 \geq 5$.
\end{example}

Menumpuk taksiran semacam itu menurut \omterm{def:g10:orders-of-magnitude:scinot}{eksponennya} menghasilkan gambar
utama bab ini: sebuah tangga yang setiap anak tangganya adalah dunia
sepuluh kali lebih lebar daripada dunia di bawahnya.

\begin{definition}[Satuan astronomi dan tahun cahaya]\label{def:g10:orders-of-magnitude:lightyear}
Yang disebut \emph{satuan astronomi}\index{satuan astronomi}
($\qty{1}{au} = \qty{1.496e11}{m}$) adalah jarak rata-rata
Bumi--Matahari, sedangkan \emph{tahun cahaya}\index{tahun cahaya}
(\unit{ly}) adalah jarak yang ditempuh cahaya dalam satu tahun. Di
ruang hampa cahaya bergerak dengan laju
\[
c = \qty{3.00e8}{m/s},
\]
laju tercepat yang diizinkan alam --- sehingga tahun cahaya adalah
sebuah jarak, bukan sebuah waktu.
\end{definition}

\begin{example}[Tahun cahaya dalam meter]\label{ex:g10:orders-of-magnitude:ly}
Satu tahun berisi $365.25 \times 24 \times 3600 = \qty{3.156e7}{s}$
(cukup dekat dengan $\pi \times 10^{7}\,\unit{s}$). Maka
\[
\qty{1}{ly} = c \times \qty{3.156e7}{s}
= \qty{3.00e8}{m/s} \times \qty{3.156e7}{s}
= \qty{9.47e15}{m}.
\]
Proxima Centauri, bintang terdekat sesudah Matahari, berjarak
$\qty{4.2}{ly} \approx \qty{4.0e16}{m}$: cahaya, yang menyeberangi
jarak Bumi--Matahari dalam delapan menit, memerlukan lebih dari empat
tahun untuk tiba dari sana.
\end{example}

\begin{method}[Memeriksa kewajaran dengan orde besar]\label{met:g10:orders-of-magnitude:sanity}
Sebelum mempercayai hasil perhitungan apa pun:
\begin{enumerate}
  \item bulatkan setiap masukan ke \omterm{def:g10:orders-of-magnitude:oom}{orde besarnya};
  \item ulangi perhitungannya dengan pangkat sepuluh saja, memakai
        \cref{prop:g10:orders-of-magnitude:powers} --- ini hanya
        memakan beberapa \omterm{def:g10:orders-of-magnitude:unit}{sekon};
  \item bandingkan: jawaban yang tepat harus sesuai dengan taksiran itu
        sampai kira-kira faktor sepuluh, dan \omterm{def:g10:orders-of-magnitude:unit}{satuannya} harus \omterm{def:g10:orders-of-magnitude:unit}{satuan}
        yang benar. Jika salah satu pemeriksaan gagal, telusurilah
        kesalahannya sebelum melangkah lebih jauh.
\end{enumerate}
\end{method}

\begin{remark}[Mengapa harus repot]\label{rem:g10:orders-of-magnitude:sanity}
Kalkulator akan dengan senang hati melaporkan bahwa sebuah apel
bermassa $\qty{4e8}{kg}$ --- ia memang tidak dapat tahu lebih baik.
Tidak ada rumus yang menandai hal yang mustahil; kebiasaanlah yang
menandainya. Fisikawan menaksir dahulu dan menghitung kemudian:
taksiran itu menangkap koma desimal yang salah tempat, \omterm{def:g10:orders-of-magnitude:unit}{satuan} yang
tertukar dan pecahan yang terbalik, yaitu tiga kesalahan yang menjadi
sebab sebagian besar jawaban salah pada setiap tingkat.
\end{remark}

\section{Pengukuran dan ketidakpastian}

Tidak ada alat yang membaca nilai sebenarnya. Sebuah pengukuran baru
jujur apabila ia juga melaporkan seberapa jauh ia mungkin meleset.

\begin{definition}[Ketidakpastian mutlak dan relatif]\label{def:g10:orders-of-magnitude:uncertainty}
Hasil pengukuran sebuah besaran $x$ dinyatakan sebagai
\[
x = x_{\text{m}} \pm \Delta x ,
\]
dengan $x_{\text{m}}$ adalah nilai terukurnya, sedangkan
\emph{ketidakpastian mutlak}\index{ketidakpastian!mutlak}
$\Delta x > 0$, dalam \omterm{def:g10:orders-of-magnitude:unit}{satuan} yang sama, membatasi kesalahan yang masuk
akal: nilai sebenarnya diklaim terletak antara
$x_{\text{m}} - \Delta x$ dan $x_{\text{m}} + \Delta x$. Adapun
\emph{ketidakpastian relatif}\index{ketidakpastian!relatif} adalah
nisbah $\Delta x / x_{\text{m}}$, biasanya dinyatakan dalam persen;
itulah yang diukur oleh kata ``teliti''.
\end{definition}

\begin{method}[Membaca alat berskala]\label{met:g10:orders-of-magnitude:reading}
Untuk penggaris, gelas ukur, atau jarum penunjuk:
\begin{enumerate}
  \item bacalah goresan skala yang terdekat dengan tanda, dengan mata
        tegak lurus terhadap skalanya (mata yang miring menggeser
        bacaannya);
  \item ambillah paling sedikit setengah goresan terkecil sebagai
        \omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian mutlak} --- lebih besar lagi jika tepi bendanya
        atau permukaan zat cairnya kabur;
  \item tulislah hasilnya sebagai $x_{\text{m}} \pm \Delta x$ beserta
        \omterm{def:g10:orders-of-magnitude:unit}{satuannya}, dan jangan pertahankan satu angka pun di belakang
        angka yang sudah tidak pasti.
\end{enumerate}
\end{method}

\begin{example}[Ketelitian relatif]\label{ex:g10:orders-of-magnitude:reading}
Bacaan di atas memiliki \omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian relatif}
$\frac{0.5}{127} \approx 0.4\%$. Penggaris yang sama, ketika mengukur
baut $\qty{12}{mm}$, memberikan $\frac{0.5}{12} \approx 4\%$: alat yang
persis sama sepuluh kali lebih tidak teliti pada benda yang kecil.
Ketelitian adalah sifat \emph{pengukurannya}, bukan sifat alatnya.
\end{example}

\begin{remark}[Mutlak kecil, relatif raksasa --- dan sebaliknya]\label{rem:g10:orders-of-magnitude:precision}
Stasiun laser mengukur jarak Bumi--Bulan, $\qty{3.84e8}{m}$, sampai
sekitar $\pm\qty{3}{cm}$: \omterm{def:g10:orders-of-magnitude:uncertainty}{ketidakpastian relatif} sebesar $10^{-10}$,
termasuk pengukuran paling tajam yang pernah dilakukan --- dengan
kesalahan mutlak yang tidak akan dibanggakan penggaris mana pun.
Bertanyalah selalu \emph{relatif terhadap apa}: pasangan
$(\Delta x,\ \Delta x / x_{\text{m}})$ menceritakan seluruh kisahnya,
dan masing-masing bilangan itu sendirian dapat menyesatkan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:1}
Ubahlah ke \omterm{def:g10:orders-of-magnitude:unit}{meter}, dalam \omterm{def:g10:orders-of-magnitude:scinot}{notasi ilmiah}:
\begin{enumerate}
  \item $\qty{750}{nm}$ (panjang gelombang cahaya merah);
  \item $\qty{2.5}{km}$;
  \item $\qty{3.2}{\micro m}$ (sebuah bakteri tanah);
  \item $\qty{0.15}{Tm}$ (jarak Bumi--Matahari, dalam busana yang tidak
        lazim).
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:orders-of-magnitude:2}

\section{Soal: Dari atom sampai Andromeda}

\begin{problem}[{Soal akhir pekan --- sebutir pasir menakar atom, Tata
Surya, galaksi dan alam semesta teramati, sekaligus menempatkan kita
pada tangga di antara keduanya}]
\label{pb:g10:orders-of-magnitude:1}
Sebutir pasir adalah benda paling sederhana dalam fisika, dan pada
akhir pekan ini ia bekerja lembur: mula-mula kita menghitung atomnya,
lalu kita menyusutkan Matahari sampai seukuran butir itu dan melangkahi
kosmos, kemudian kita membiarkan cahaya yang mengukur, dan akhirnya
kita menghitung atom di dalam \emph{tubuhmu}. Data, dipakai di seluruh
soal: diameter butir $\qty{0.5}{mm}$; diameter atom
$\qty{2.5e-10}{m}$; massa jenis pasir $\qty{2500}{kg/m^3}$; diameter:
Matahari $\qty{1.39e9}{m}$, Bumi $\qty{1.27e7}{m}$, Bima Sakti
$\qty{9.5e20}{m}$, alam semesta teramati $\qty{8.8e26}{m}$; jarak dari
kita: Bulan $\qty{3.84e8}{m}$, Matahari $\qty{1.496e11}{m}$, Neptunus
$\qty{4.5e12}{m}$, Proxima Centauri $\qty{4.0e16}{m}$, galaksi
Andromeda $\qty{2.4e22}{m}$; $c = \qty{3.00e8}{m/s}$; sebuah molekul
air bermassa $\qty{3.0e-26}{kg}$ dan berisi tiga atom.

\medskip
\noindent\textbf{Bagian I --- Anatomi butir pasir.}
\begin{enumerate}
  \item Tulislah diameter butir dan diameter atom dalam \omterm{def:g10:orders-of-magnitude:unit}{meter}, dalam
        \omterm{def:g10:orders-of-magnitude:scinot}{notasi ilmiah}, lalu sebutkan \omterm{def:g10:orders-of-magnitude:prefixes}{awalan} SI yang paling dekat
        dengan masing-masing.
  \item Berapa banyak atom yang berjajar melintasi satu diameter butir?
  \item Modelkan butir itu sebagai kubus bersisi $\qty{0.5}{mm}$ yang
        terisi penuh atom. Taksirlah banyaknya atom di dalam butir itu,
        sebagai \omterm{def:g10:orders-of-magnitude:oom}{orde besar}.
  \item Hitunglah volume kubus itu, lalu massa butirnya. Nyatakan
        massanya dalam miligram.
  \item Periksalah kewajaran modelnya: bagilah massa butir dengan
        banyaknya atom, lalu bandingkan dengan \omterm{def:g10:orders-of-magnitude:oom}{orde besar} massa atom
        yang sudah dikenal (beberapa $10^{-26}\,\unit{kg}$). Bagaimana
        putusannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Matahari sebagai sebutir pasir.}
Kita membangun model berskala dari kosmos yang di dalamnya Matahari
menjadi butir pasir itu: setiap ukuran model sama dengan ukuran
sebenarnya dikali sebuah faktor $s$.
\begin{enumerate}[resume]
  \item Hitunglah faktor skala $s$.
  \item Pada model itu, tentukan diameter Bumi dan jaraknya dari
        butir-Matahari. Benda apa dari dunia Bagian I yang kira-kira
        seukuran model Bumi?
  \item Berapa jauh dari butir itu model Neptunus mengorbit? Perabot
        apa yang sanggup memuat seluruh sistem keplanetan itu?
  \item Berapa jauh model Proxima Centauri?
  \item Dua butir pasir, terpisah $\qty{14}{km}$, dan hampir tidak ada
        apa-apa di antaranya: nyatakan dalam satu atau dua kalimat apa
        yang diungkapkan model itu tentang betapa kosongnya galaksi.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Cahaya yang mengukur.}
\begin{enumerate}[resume]
  \item Berapa lama cahaya menyeberangi butir pasir yang sebenarnya?
  \item Berapa lama cahaya sampai kepada kita dari Bulan, dan dari
        Matahari?
  \item Hitunglah kembali \omterm{def:g10:orders-of-magnitude:lightyear}{tahun cahaya} dalam \omterm{def:g10:orders-of-magnitude:unit}{meter} dari $c$ dan panjang
        satu tahun, lalu periksalah bahwa jarak Proxima pada data
        bersesuaian dengan sekitar $\qty{4.2}{ly}$.
  \item Ubahlah jarak Andromeda ke \omterm{def:g10:orders-of-magnitude:lightyear}{tahun cahaya}. Kira-kira kapan cahaya
        yang kini memasuki matamu meninggalkan Andromeda, dan makhluk
        apa yang berjalan di Bumi pada waktu itu?
  \item Berapa jauh cahaya merambat dalam satu nanosekon? Sebuah
        pemroses yang berdetak $\qty{3}{GHz}$ memulai satu operasi
        setiap sepertiga nanosekon: jelaskan dalam satu kalimat
        mengapa komputer yang cepat harus berupa komputer yang kecil.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kamu, di antara atom dan Andromeda.}
\begin{enumerate}[resume]
  \item Seorang manusia kira-kira $\qty{70}{kg}$ air. Dengan memakai
        data tadi, taksirlah massa rata-rata satu atom di dalam air,
        lalu banyaknya atom di dalam seorang manusia.
  \item Bandingkan cacah itu dengan banyaknya atom di dalam butir pasir
        (pertanyaan 3) dan dengan kira-kira $2\times10^{23}$ bintang di
        alam semesta teramati. Berikan kedua faktornya.
  \item Pada tangga itu, kamu lebih dekat ke atom ataukah ke alam
        semesta teramati? Bandingkan nisbah
        $\text{(kamu)}/\text{(atom)}$ dan
        $\text{(alam semesta)}/\text{(kamu)}$, dengan mengambil
        $\qty{1.7}{m}$ untuk dirimu.
  \item Tentukan ukuran $x$ yang duduk tepat di tengah tangga antara
        inti atom ($\qty{e-15}{m}$) dan alam semesta teramati: $x$
        harus memenuhi $x / 10^{-15} = 10^{26.9} / x$ (dengan memakai
        $\qty{8.8e26}{m} \approx 10^{26.9}\,\unit{m}$). Benda Tata
        Surya apa yang kebetulan berukuran persis seperti itu?
  \item Puncak ceritanya. Susutkan alam semesta teramati dengan faktor
        $s$ dari Bagian II, sehingga Matahari menjadi sebutir pasir.
        Berikan diameter alam semesta model itu dalam \omterm{def:g10:orders-of-magnitude:unit}{meter}, lalu dalam
        \omterm{def:g10:orders-of-magnitude:lightyear}{satuan astronomi}, dan simpulkan dalam satu kalimat: bahkan
        ketika setiap bintang diubah menjadi sebutir pasir, alam
        semesta tetap meluap dari setiap skala manusia --- hanya
        pangkat sepuluhlah yang sanggup memuatnya.
\end{enumerate}
\end{problem}
