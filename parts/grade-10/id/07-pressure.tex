\chapter{Tekanan: Dari Olahraga sampai Penyelaman}\label{ch:g10:pressure}

Seorang pemain ski meluncur di atas salju yang membenamkan orang yang
berjalan kaki; dorongan sebesar \qty{5}{N} menembuskan jarum ke dalam
kulit yang tidak tergores oleh kepalan tangan. \omterm{def:g10:inertia:force}{Gaya} yang sama, akibat
yang jauh berbeda: besaran yang hilang di situ adalah \emph{\omterm{def:g10:pressure:pressure}{tekanan}},
yaitu \omterm{def:g10:inertia:force}{gaya} dibagi luas yang menerimanya. Bab ini mendefinisikannya,
mengikutinya turun ke dasar kolam, naik ke sebuah barometer, masuk ke
alat suntik yang ditekan --- lalu berakhir empat puluh \omterm{def:g10:orders-of-magnitude:unit}{meter} di bawah
permukaan laut, tempat \omterm{def:g10:pressure:pressure}{tekanan} menjadi soal hidup dan mati.

\section{Tekanan: gaya yang tersebar pada suatu luas}

\begin{definition}[Tekanan]\label{def:g10:pressure:pressure}
Apabila sebuah \omterm{def:g10:inertia:force}{gaya} sebesar $F$ (dalam \unit{N}) menekan tegak lurus
pada permukaan seluas $S$ (dalam \unit{m^2}), \emph{tekanan}\index{tekanan}
yang dikerjakan pada permukaan itu adalah
\[
P = \frac{F}{S}.
\]
\omterm{def:g10:orders-of-magnitude:unit}{Satuannya}, \unit{N/m^2}, disebut \emph{pascal}\index{pascal}
(\unit{Pa}) --- \omterm{def:g10:orders-of-magnitude:unit}{satuan} yang mungil, yaitu \omterm{def:g10:universal-gravitation:weight}{berat} sebuah apel yang
tersebar pada satu \omterm{def:g10:orders-of-magnitude:unit}{meter} persegi, sehingga \unit{kPa} dan \unit{MPa}
menjadi \omterm{def:g10:orders-of-magnitude:unit}{satuan} kerjanya.
\end{definition}

\begin{example}[Hak sepatu lawan sepatu salju]\label{ex:g10:pressure:snowshoe}
Seseorang bermassa \qty{60}{kg} \omterm{def:g10:universal-gravitation:weight}{beratnya}
$F = 60 \times 9.81 \approx \qty{590}{N}$. Pada satu hak sepatu runcing
seluas \qty{1.0}{cm^2} = \qty{1.0e-4}{m^2}:
$P = 590/\num{1.0e-4} \approx \qty{5.9e6}{Pa} = \qty{5.9}{MPa}$. Pada
dua sepatu salju yang masing-masing seluas \qty{0.20}{m^2}:
$P = 590/0.40 \approx \qty{1.5}{kPa}$. \omterm{def:g10:universal-gravitation:weight}{Berat} yang \emph{sama} menekan
$4000$ kali lebih kuat di bawah haknya: hak sepatu melubangi lantai
kayu, sepatu salju mengapung di atas salju yang gembur.
\end{example}

\begin{remark}[Memusatkan atau menyebarkan]\label{rem:g10:pressure:tools}
Sebuah jarum jahit yang didorong dengan \qty{5}{N} pada ujung seluas
\qty{0.01}{mm^2} = \qty{1e-8}{m^2} mengerjakan
$\qty{5e8}{Pa} = \qty{500}{MPa}$ --- cukup untuk memisahkan serat kulit.
Alat penusuk memusatkan \omterm{def:g10:inertia:force}{gaya}; rancangan penahan beban (papan ski, rantai
roda, pondasi) menyebarkannya. Olahraga adalah pengelolaan \omterm{def:g10:pressure:pressure}{tekanan} yang
diterapkan: sepatu es menggigit esnya, papan ski meluncur di atas salju.
\end{remark}

\section{Tekanan di dalam zat cair yang diam}

Mengapa air menekan lebih kuat di bagian yang lebih dalam? Asingkanlah
sebuah kolom air tegak khayalan yang penampangnya $S$, dari permukaan
sampai kedalaman $h$: volumenya $Sh$, massanya $\rho S h$ ($\rho$ adalah
massa jenis zat cairnya, dalam \unit{kg/m^3}), \omterm{def:g10:universal-gravitation:weight}{beratnya} $\rho S h g$.
Dasar kolom itu memikul \omterm{def:g10:universal-gravitation:weight}{beratnya} \emph{ditambah} dorongan atmosfer
$P_0 S$ pada bagian atasnya. Dengan membagi \omterm{def:g10:inertia:force}{gaya} seluruhnya dengan $S$:
\omterm{def:g10:pressure:pressure}{tekanan} pada kedalaman $h$ semestinya $P_0 + \rho g h$ --- dan memang
demikianlah, di mana pun di dalam zat cair itu.

\begin{theorem}[Tekanan pada suatu kedalaman]\label{thm:g10:pressure:depth}
Di dalam zat cair bermassa jenis $\rho$ yang diam dan terbuka ke udara,
\omterm{def:g10:pressure:pressure}{tekanan} pada kedalaman $h$ di bawah permukaannya adalah
\[
P = P_0 + \rho g h,
\]
dengan $P_0$ adalah \omterm{def:g10:pressure:atmospheric}{tekanan atmosfer} di permukaannya. \omterm{def:g10:pressure:pressure}{Tekanan} itu hanya
bergantung pada kedalamannya, bukan pada bentuk wadahnya, dan pada satu
titik tertentu zat cair itu menekan sama kuat ke segala arah.
\end{theorem}
\admitted

\begin{remark}
Alasan kolom itu hanya membuat hukumnya masuk akal; penurunan yang jujur
(untuk bentuk apa pun, ke segala arah) ada pada jilid Tahun ke-1. Suku
milik airnya sendiri, $\rho g h$, disebut \emph{tekanan
ukur}\index{tekanan ukur} --- itulah yang ditampilkan alat ukur \omterm{def:g10:pressure:pressure}{tekanan}
seorang penyelam.
\end{remark}

\begin{example}[Sepuluh meter air]\label{ex:g10:pressure:tenmetres}
Di dalam air tawar ($\rho = \qty{1000}{kg/m^3}$), pada
$h = \qty{10}{m}$: $\rho g h = 1000 \times 9.81 \times 10 \approx \qty{98}{kPa}$, sehingga $P = 101 + 98 \approx \qty{199}{kPa}$. Sepuluh
\omterm{def:g10:orders-of-magnitude:unit}{meter} air menambahkan hampir persis satu atmosfer: itulah aturan praktis
para penyelam, \emph{satu atmosfer tambahan tiap sepuluh \omterm{def:g10:orders-of-magnitude:unit}{meter}}.
\end{example}

\section{Samudra udara}

\begin{definition}[Tekanan atmosfer]\label{def:g10:pressure:atmospheric}
Kita hidup di dasar sebuah samudra udara, yang beratnya menimbulkan
\emph{tekanan atmosfer}\index{tekanan atmosfer}, di permukaan laut
\[
P_0 \approx \qty{101}{kPa} = \qty{1.01e5}{Pa}.
\]
Pada tiap sentimeter persegi hal itu berarti
$F = \num{1.01e5} \times \num{1e-4}
\approx \qty{10}{N}$: \omterm{def:g10:universal-gravitation:weight}{berat} massa
satu \omterm{def:g10:orders-of-magnitude:unit}{kilogram} pada tiap \unit{cm^2} kulitmu, mejamu, segala sesuatu.
\end{definition}

\begin{example}[Beban yang tak terlihat]\label{ex:g10:pressure:sheet}
Selembar kertas A4 (\qty{21.0}{cm} $\times$ \qty{29.7}{cm}, sehingga
$S \approx \qty{0.0624}{m^2}$) menerima dari udara di atasnya
$F = \num{1.01e5} \times 0.0624 \approx \qty{6.3e3}{N}$ --- seberat
sebuah mobil kecil. Tidak ada yang robek: udara di bawahnya mendorong
\emph{ke atas} sama kuatnya. Kita baru menyadari atmosfer ketika salah
satu sisinya kehilangan atmosfer itu --- mangkuk isap, sedotan, kabin
pesawat.
\end{example}

\begin{remark}[Barometer Torricelli]\label{rem:g10:pressure:torricelli}
Pada tahun 1643 Torricelli membalikkan tabung tertutup berisi raksa di
atas bejana raksa. Kolomnya turun sampai ketinggian yang membuat
$\rho g h = P_0$: $h = \num{1.01e5}/(13600 \times 9.81) \approx \qty{0.76}{m}$, dan meninggalkan di atasnya ruang hampa pertama yang
pernah dibuat. Tingginya mengikuti cuaca dan menyusut bersama
ketinggian tempat: sebuah barometer menimbang udara di atasmu (air akan
memerlukan \qty{10.3}{m}: \cref{exo:g10:pressure:9}).
\end{remark}

\section{Gas: tumbukan dan hukum Boyle}

Gas pun menekan wadahnya: gas adalah gerombolan molekul yang bergerak
acak tanpa henti, tiap tumbukan memberi dinding sebuah dorongan mungil
ke luar, dan miliaran demi miliaran tumbukan tiap sekon melebur menjadi
\omterm{def:g10:inertia:force}{gaya} tetap yang kita ukur sebagai \omterm{def:g10:pressure:pressure}{tekanan} gas. Perasilah gas itu ke
dalam setengah volumenya: molekulnya menumbuk dua kali lebih sering,
sehingga \omterm{def:g10:pressure:pressure}{tekanannya} semestinya menjadi dua kali lipat. Dan memang
demikian.

\begin{proposition}[Hukum Boyle]\label{prop:g10:pressure:boyle}
\index{hukum Boyle}
Untuk gas yang jumlahnya tetap dan dijaga pada suhu tetap, \omterm{def:g10:pressure:pressure}{tekanan} dan
volumenya berbanding terbalik:
\[
P \, V = \text{constant},
\qquad\text{yaitu}\qquad
P_1 V_1 = P_2 V_2 .
\]
\end{proposition}
\admitted

\begin{remark}
Pada tingkat ini hukum itu merupakan fakta percobaan --- jilid Tahun
ke-1 menurunkannya dari gambaran tumbukan. Percobaannya (alat suntik
tertutup berisi udara, sebuah pengindra \omterm{def:g10:pressure:pressure}{tekanan},
\qty{20}{\degreeCelsius}):
\begin{center}
\begin{tabular}{lccccc}
$V$ (\unit{mL}) & 60 & 50 & 40 & 30 & 20 \\
$P$ (\unit{kPa}) & 100 & 120 & 150 & 200 & 300 \\
$P \times V$ (\unit{kPa.mL}) & 6000 & 6000 & 6000 & 6000 & 6000
\end{tabular}
\end{center}
\end{remark}

\begin{example}[Alat suntik yang ditekan]\label{ex:g10:pressure:boyle}
Udara menempati \qty{4.0}{L} pada \qty{100}{kPa}; dimampatkan
perlahan-lahan (agar suhunya tetap) sampai \qty{250}{kPa}, udara itu
menempati $V_2 = P_1 V_1 / P_2 = 400/250 = \qty{1.6}{L}$.
\end{example}

\section{Menyelam: tekanan yang dipekerjakan}

\begin{method}[Tekanan mutlak pada suatu kedalaman]\label{met:g10:pressure:dive}
Untuk penyelam pada kedalaman $h$ di laut ($\rho \approx \qty{1025}{kg/m^3}$, sehingga $\rho g h \approx \qty{101}{kPa}$ tiap
\qty{10}{m}), hitunglah $P = P_0 + \rho g h$ --- dan pada \omterm{prop:g10:pressure:boyle}{hukum Boyle}
pakailah selalu \omterm{def:g10:pressure:pressure}{tekanan} \emph{mutlak}\index{tekanan mutlak} ini, jangan
pernah suku ukurnya saja. Seorang penyelam berada di bawah $2$ atmosfer
pada \qty{10}{m}, $3$ atmosfer pada \qty{20}{m}, dan $5$ atmosfer pada
\qty{40}{m}.
\end{method}

\begin{remark}[Jangan pernah menahan napas]\label{rem:g10:pressure:breath}
Regulator memasok udara pada \omterm{def:g10:pressure:pressure}{tekanan} sekitarnya: paru-paru terisi secara
wajar pada kedalaman berapa pun. Tetapi isilah \qty{6.0}{L} pada
\qty{20}{m} ($\qty{302}{kPa}$), tahanlah napas itu lalu naiklah ke
permukaan: Boyle menuntut $6.0 \times 302/101 \approx \qty{18}{L}$ ---
tiga kali daya tampung sebuah dada, dan paru-paru sudah robek jauh
sebelum itu, bahkan pada kenaikan beberapa \omterm{def:g10:orders-of-magnitude:unit}{meter} saja. Karena itulah
aturan pertama selam skuba: \emph{jangan pernah menahan napas ketika
naik} --- teruslah bernapas dan udara yang memuai itu akan mengalir
keluar begitu saja. Penyelam bebas justru aman: udaranya diambil di
permukaan, sehingga dalam perjalanan naik udara itu hanya memuai
kembali ke volume semula.
\end{remark}

\begin{remark}[Dekompresi]\label{rem:g10:pressure:deco}
Menghirup udara bertekanan tinggi juga melarutkan nitrogen tambahan di
dalam darah, seperti gas di dalam botol soda yang tertutup; naiklah
terlalu cepat dan gas itu berbuih menjadi gelembung di dalam tubuh ---
itulah penyakit dekompresi. Karena itulah kenaikannya lambat (sekitar
\qty{10}{m} tiap menit) beserta perhentian di kedalaman dangkal, yang
membiarkan gasnya keluar dengan tenang melalui paru-paru. Teorinya
menunggu jilid Tahun ke-1; aturan perilakunya milik setiap penyelam.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:pressure:1}
Sebuah peti seberat \qty{600}{N} terletak pada sisi seluas
$\qty{0.50}{m} \times \qty{0.30}{m}$, lalu pada sisi seluas
$\qty{0.30}{m} \times \qty{0.20}{m}$. Hitunglah kedua \omterm{def:g10:pressure:pressure}{tekanannya} pada
tanah. Kedudukan yang mana yang lebih baik di atas pasir yang lembut?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:2}
Nyatakan \qty{0.25}{MPa} dalam \unit{kPa} dan dalam \unit{Pa}. Manakah
yang menekan lebih kuat: \qty{3.0}{N} pada \qty{2.0}{cm^2}, atau
\qty{60}{N} pada \qty{500}{cm^2}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:3}
Hitunglah \omterm{thm:g10:pressure:depth}{tekanan ukur} $\rho g h$ dan \omterm{met:g10:pressure:dive}{tekanan mutlak} di dasar kolam
renang sedalam \qty{3.0}{m} ($\rho = \qty{1000}{kg/m^3}$,
$P_0 = \qty{101}{kPa}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:4}
Udara menempati \qty{4.0}{L} pada \qty{100}{kPa}, dengan suhu tetap.
Berapa volumenya pada \qty{250}{kPa}? Dan pada \qty{50}{kPa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:pressure:5}
Seorang peluncur es bermassa \qty{65}{kg} meluncur pada satu pisau yang
luas sentuhnya \qty{12}{cm^2}, lalu berdiri pada sol sepatu yang
seluruhnya \qty{360}{cm^2}. Hitunglah kedua \omterm{def:g10:pressure:pressure}{tekanannya} dan nisbahnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:6}
Sebuah ibu jari mendorong paku payung dengan \qty{20}{N}. Kepalanya
seluas \qty{1.2}{cm^2}, ujungnya \qty{0.010}{mm^2}. Hitunglah \omterm{def:g10:pressure:pressure}{tekanan}
pada ibu jari dan pada dindingnya; mengapa hanya dindingnya yang
tertembus?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:7}
Sebuah palka kapal selam seluas \qty{0.50}{m^2} berada pada kedalaman
\qty{30}{m} di dalam air laut ($\rho = \qty{1025}{kg/m^3}$); di
dalamnya, udaranya berada pada \omterm{def:g10:pressure:atmospheric}{tekanan atmosfer}. Hitunglah \omterm{def:g10:inertia:force}{gaya} total
yang menahannya tetap tertutup, dan massa yang beratnya setara dengan
\omterm{def:g10:inertia:force}{gaya} itu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:8}
Sebuah pengindra pada kedalaman \qty{5.00}{m} di dalam zat cair yang
tidak dikenal membaca \omterm{thm:g10:pressure:depth}{tekanan ukur} \qty{66.2}{kPa}. Tentukan massa
jenisnya. Apakah zat cair itu air tawar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:9}
Setinggi apa barometer Torricelli jika memakai air
($\rho = \qty{1000}{kg/m^3}$) dan bukan raksa
($\rho = \qty{13600}{kg/m^3}$)? Perolehlah kembali kedua tingginya dari
$\rho g h = P_0$; mengapa raksa yang menang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:10}
Dengan data alat suntik pada \cref{prop:g10:pressure:boyle}
($PV = \qty{6000}{kPa.mL}$), ramalkan volumenya pada \qty{240}{kPa} dan
\omterm{def:g10:pressure:pressure}{tekanannya} pada \qty{75}{mL}. Kurva apakah $P$ terhadap $V$? Dan $P$
terhadap $1/V$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:pressure:11}
Seorang penyelam pada \qty{20}{m} melepaskan gelembung bervolume
\qty{0.50}{cm^3}. Berapa volumenya tepat di bawah permukaan? (Suhunya
tetap; air laut menambahkan \qty{101}{kPa} tiap \qty{10}{m}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:12}
Sebuah mangkuk isap seluas \qty{12}{cm^2} ditekan rata pada langit-langit.
\begin{enumerate}
  \item Jika di dalamnya hampa sempurna: berapa \omterm{def:g10:inertia:force}{gaya} yang dikerjakan
        atmosfer pada mangkuk itu, dan massa gantung berapa yang dapat
        ditahannya?
  \item Mangkuk yang sebenarnya masih menyisakan udara \qty{20}{kPa} di
        dalamnya. Berapa massanya sekarang?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:13}
Sebuah tabung selam memuat \qty{12}{L} udara pada \qty{200}{bar}
(\qty{1}{bar} = \qty{100}{kPa} $\approx 1$ atmosfer); \qty{50}{bar}
harus tersisa sebagai cadangan. Seorang penyelam menghirup \qty{15}{L}
tiap menit pada \omterm{def:g10:pressure:pressure}{tekanan} sekitarnya. Berapa liter udara bertekanan
permukaan yang dapat dipakai? Berapa lama tabung itu bertahan pada
\qty{30}{m} (\omterm{met:g10:pressure:dive}{tekanan mutlaknya} $\approx \qty{4}{bar}$)? Dan pada
\qty{10}{m}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:14}
Paru-paru seorang penyelam bebas memuat \qty{6.0}{L} di permukaan dan
tidak dapat menyusut di bawah volume sisa \qty{1.5}{L}. Pada kedalaman
laut berapa ($\rho = \qty{1025}{kg/m^3}$) batas itu tercapai? (Batas
itu dahulu dinyatakan sebagai batas penyelaman bebas; rekornya kini
melampaui \qty{100}{m}, karena darah yang berpindah ke rongga dada
mengisi volume yang hilang itu.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:pressure:15}
Pada sebuah dongkrak hidrolik, zat cair meneruskan \omterm{def:g10:pressure:pressure}{tekanan} tanpa
berubah dari torak kecil ($s = \qty{2.0}{cm^2}$) ke torak besar
($S = \qty{400}{cm^2}$) yang memikul mobil bermassa \qty{1200}{kg}.
\begin{enumerate}
  \item \omterm{def:g10:inertia:force}{Gaya} berapa pada torak kecilnya yang menahan mobil itu tetap
        terangkat? \omterm{def:g10:pressure:pressure}{Tekanan} berapa yang dibawa zat cairnya?
  \item Untuk menaikkan mobil itu \qty{2.0}{cm}, sejauh apa torak
        kecilnya harus bergerak? Bandingkan usaha yang dikerjakan pada
        masing-masing torak.
\end{enumerate}
\end{exercise}

\section{Soal: Satu napas turun, satu napas naik}

\begin{problem}[{Soal akhir pekan --- satu napas turun, satu napas
naik: fisika sebuah penyelaman bebas dan sebuah kenaikan selam skuba,
dari perasan masker sampai aturan bahwa sepuluh meter terakhir adalah
yang paling berbahaya}]
\label{pb:g10:pressure:1}
Lena menyelam bebas: satu tarikan napas di permukaan, turun ke
\qty{30}{m} lalu kembali. Marco menyelam skuba di sampingnya, bernapas
dari sebuah tabung. Laut yang sama ($\rho = \qty{1025}{kg/m^3}$,
$g = \qty{9.81}{N/kg}$, $P_0 = \qty{101}{kPa}$), paru-paru
\qty{6.0}{L} yang sama, bahaya yang berlawanan --- semuanya diatur oleh
$P = P_0 + \rho g h$ dan $PV$ = tetap (suhunya tetap di sepanjang soal).

\medskip
\noindent\textbf{Bagian I --- \omterm{def:g10:universal-gravitation:weight}{Berat} air.}
\begin{enumerate}
  \item Hitunglah $\rho g$ untuk air laut dalam \unit{kPa} per \omterm{def:g10:orders-of-magnitude:unit}{meter},
        lalu periksalah aturan praktisnya: satu atmosfer tiap
        \qty{10}{m}.
  \item Hitunglah \omterm{met:g10:pressure:dive}{tekanan mutlak} pada \qty{10}{m}, \qty{20}{m},
        \qty{30}{m} dan \qty{40}{m}.
  \item Nyatakan masing-masing sebagai kelipatan \omterm{def:g10:pressure:pressure}{tekanan} permukaan.
  \item Masker Lena menutupi \qty{150}{cm^2}. Pada \qty{30}{m}, jika
        udara di dalamnya masih bertekanan permukaan, \omterm{def:g10:inertia:force}{gaya} total berapa
        yang dikerjakan airnya? (Penyelam mengembuskan napas ke dalam
        maskernya melalui hidung untuk mencegah \emph{perasan masker}
        ini.)
  \item Sebuah gendang telinga luasnya sekitar \qty{0.60}{cm^2}.
        Hitunglah \omterm{def:g10:inertia:force}{gaya} total padanya pada \qty{3.0}{m} jika telinga
        tengahnya tetap bertekanan permukaan; mengapa telinga terasa
        sakit bahkan hanya di kolam renang, dan apa yang harus dicapai
        tindakan ``menyamakan \omterm{def:g10:pressure:pressure}{tekanan}''?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Satu napas turun.}
Lena meninggalkan permukaan dengan \qty{6.0}{L} udara di dalam
paru-parunya.
\begin{enumerate}[resume]
  \item Dengan \omterm{prop:g10:pressure:boyle}{hukum Boyle} memakai \omterm{met:g10:pressure:dive}{tekanan mutlak}, hitunglah volume
        paru-parunya pada \qty{10}{m}, \qty{20}{m} dan \qty{30}{m}.
  \item Pada kedalaman berapa volume paru-parunya tinggal separuh?
  \item Apakah volume paru-paru sebanding dengan kedalaman? Lukiskan
        kurva $V$ terhadap \omterm{met:g10:pressure:dive}{tekanan mutlak} $P$.
  \item Volume sisanya \qty{1.5}{L}: paru-parunya tidak dapat menyusut
        lebih jauh. Tunjukkan bahwa batas ini tercapai di dekat
        \qty{30}{m}.
  \item Rekor penyelaman bebas toh melampaui \qty{100}{m}: apa yang
        mengisi volume yang hilang itu? (Apa lagi yang dapat mengalir
        masuk ke rongga dada?)
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Satu napas naik.}
Pada \qty{30}{m}, regulator Marco mengisi paru-parunya yang
\qty{6.0}{L}.
\begin{enumerate}[resume]
  \item Pada \omterm{def:g10:pressure:pressure}{tekanan} berapa regulator itu memasok udaranya? Berapa kali
        lebih rapat udara itu daripada udara permukaan?
  \item Marco panik, menahan napas, lalu naik ke \qty{10}{m}: volume
        berapa yang dituntut udara yang terperangkap itu di sana?
  \item Volume berapa di permukaan --- berapa kali daya tampungnya?
  \item Nyatakan aturan pertama seorang penyelam skuba, dan mengapa
        bernapas secara wajar melenyapkan bahayanya.
  \item Lena pun menahan napas dari \qty{30}{m} sampai ke permukaan,
        namun paru-parunya benar-benar aman. Jelaskan ketaksetangkupan
        itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Gelembung, dan sepuluh \omterm{def:g10:orders-of-magnitude:unit}{meter} terakhir.}
\begin{enumerate}[resume]
  \item Marco melepaskan gelembung \qty{1.0}{cm^3} pada \qty{40}{m}.
        Hitunglah volumenya pada \qty{30}{m}, \qty{20}{m}, \qty{10}{m}
        dan tepat di bawah permukaan.
  \item Untuk tiap tahap \qty{10}{m} kenaikannya, hitunglah faktor
        pertumbuhan gelembung itu. Di mana pertumbuhannya paling cepat?
  \item Penyelam naik paling cepat \qty{10}{m} tiap menit: berapa lama
        dari \qty{40}{m}, dan pada menit yang mana gas yang terperangkap
        atau terlarut memuai paling banyak?
  \item Di kedalaman, darah Marco melarutkan nitrogen tambahan, seperti
        gas di dalam botol soda yang tertutup. Apa yang dilakukan
        kenaikan yang terlalu cepat, dan bagaimana kenaikan yang lambat
        beserta perhentian keselamatan \qty{5}{m} mencegahnya?
  \item Puncak ceritanya: dalam satu kalimat untuk masing-masing,
        berikan putusan bagi penyelam bebas, putusan bagi penyelam
        skuba, dan alasan berangka mengapa sepuluh \omterm{def:g10:orders-of-magnitude:unit}{meter} terakhir
        setiap kenaikan pantas mendapat penghormatan paling besar.
\end{enumerate}
\end{problem}
