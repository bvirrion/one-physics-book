\chapter{Pembiasan Cahaya}\label{ch:g10:refraction}

Celupkan sedotan ke dalam segelas air: sedotan itu tampak patah pada
batas airnya. Kolam renang selalu lebih dalam daripada penampakannya,
intan memancarkan kilat warna dari batu yang tidak berwarna, dan
kalimat yang sedang kamu baca ini besar kemungkinan menyeberangi
samudra di dalam benang kaca yang lebih tipis daripada sehelai rambut.
Satu hukum menjelaskan keempatnya --- sebuah kesamaan antara dua sinus,
yang ditemukan Snell pada tahun 1621 setelah lima belas abad para
fisikawan menatap tabel sudut tanpa melihat polanya.

\section{Pemantulan: sebuah pengingat}

Di dalam medium bening yang homogen, cahaya merambat lurus --- itulah
sinar pada \cref{ch:g10:light-spectra}. Pada batas antara dua medium,
sebagian cahaya memantul kembali (\emph{pemantulan}) dan sebagian
menyeberang (\emph{\omterm{def:g10:refraction:refraction}{pembiasan}}). Keduanya menaati hukum geometri yang
tepat, dengan satu kesepakatan: sudut selalu diukur dari \emph{\omterm{def:g10:refraction:angles}{garis
normal}}, tidak pernah dari permukaannya.

\begin{definition}[Garis normal dan sudut datang]\label{def:g10:refraction:angles}
Pada titik tempat sebuah sinar menemui permukaan, yang disebut
\emph{garis normal}\index{garis normal} adalah garis yang tegak lurus
permukaan itu di titik tersebut. \emph{Sudut datang}\index{sudut datang}
$i_1$ adalah sudut antara sinar yang datang dan garis normal; sudut
pantul dan sudut bias diukur dari garis normal dengan cara yang sama.
\end{definition}

\begin{proposition}[Hukum pemantulan]\label{prop:g10:refraction:reflection}
Sinar pantul terletak pada bidang yang memuat sinar datang dan \omterm{def:g10:refraction:angles}{garis
normal}, di seberang \omterm{def:g10:refraction:angles}{garis normal}, dan sudut pantulnya sama dengan \omterm{def:g10:refraction:angles}{sudut
datang}:
\[
i_1' = i_1 .
\]
\end{proposition}
\admitted

\begin{example}[Membaca sudut dengan benar]\label{ex:g10:refraction:mirror}
Seberkas laser mengenai cermin ``pada $35^\circ$'' --- diukur dari
permukaan cermin, sebagaimana berkas sering digambarkan. \omterm{def:g10:refraction:angles}{Sudut datangnya}
adalah $90^\circ - 35^\circ = 55^\circ$, sehingga sinar pantulnya juga
pergi pada $55^\circ$ dari \omterm{def:g10:refraction:angles}{garis normal}, dan berkas itu berbelok sebesar
$180^\circ - 2 \times 55^\circ = 70^\circ$. Melupakan kesepakatan \omterm{def:g10:refraction:angles}{garis
normal} adalah kesalahan paling umum pada bab ini; setiap kali kesalahan
itu berharga $90^\circ$.
\end{example}

\section{Pembiasan dan indeks bias}

Peganglah sebatang pensil di dalam gelas yang terisi separuh: pensil itu
tampak patah pada permukaan airnya. Cahaya dari bagian yang terendam
berubah arah ketika meninggalkan air, sehingga mata --- yang menganggap
sinar merambat lurus --- menyusun kembali ujung pensil di tempat yang
bukan tempatnya.

\begin{definition}[Pembiasan]\label{def:g10:refraction:refraction}
\emph{Pembiasan}\index{pembiasan} adalah perubahan arah cahaya ketika
cahaya menyeberangi batas antara dua medium yang bening. Sinar di dalam
medium kedua disebut \emph{sinar bias}\index{sinar bias}, dan sudutnya
terhadap \omterm{def:g10:refraction:angles}{garis normal} adalah sudut bias $i_2$.
\end{definition}

\omterm{def:g10:refraction:refraction}{Pembiasan} terjadi karena cahaya memiliki laju yang berbeda di dalam
medium yang berbeda; setiap bahan dicirikan oleh seberapa besar ia
memperlambat cahaya.

\begin{definition}[Indeks bias]\label{def:g10:refraction:index}
\emph{Indeks bias}\index{indeks bias} sebuah medium bening adalah
nisbah
\[
n = \frac{c}{v},
\]
dengan $c = \qty{3.00e8}{m/s}$ adalah laju cahaya di ruang hampa dan $v$
lajunya di dalam medium itu. Karena $v \leq c$, indeksnya memenuhi
$n \geq 1$; indeks tidak bersatuan. Beberapa nilai yang berguna:
\begin{center}
\begin{tabular}{lccccc}
medium & udara & air & pleksiglas & kaca & intan \\
$n$ & $1.0003$ & $1.33$ & $1.49$ & $1.50$ & $2.42$ \\
\end{tabular}
\end{center}
\end{definition}

\begin{example}[Laju cahaya di dalam air]\label{ex:g10:refraction:speed}
Di dalam air, $v = \dfrac{c}{n} = \dfrac{\qty{3.00e8}{m/s}}{1.33}
= \qty{2.26e8}{m/s}$. Di dalam intan, cahaya merayap pada
$\qty{3.00e8}{m/s}/2.42 = \qty{1.24e8}{m/s}$ --- kurang dari separuh
lajunya di ruang hampa. Untuk udara, $n = 1.0003$ mengubah lajunya
sebesar $0.03\%$: pada bab ini udara diperlakukan sebagai ruang hampa
dan kita ambil $n_{\text{udara}} = 1.00$.
\end{example}

\begin{theorem}[Hukum pembiasan Snell]\label{thm:g10:refraction:snell}
\index{hukum Snell}
\omterm{def:g10:refraction:refraction}{Sinar bias} terletak pada bidang yang memuat sinar datang dan \omterm{def:g10:refraction:angles}{garis
normal}, di seberang \omterm{def:g10:refraction:angles}{garis normal}, dan \omterm{def:g10:refraction:angles}{sudut datang} serta sudut biasnya
memenuhi
\[
n_1 \sin i_1 = n_2 \sin i_2 ,
\]
dengan $n_1$ dan $n_2$ adalah \omterm{def:g10:refraction:index}{indeks bias} kedua medium itu. Lintasannya
dapat dibalik: cahaya yang merambat mundur menyusuri \omterm{def:g10:refraction:refraction}{sinar bias} akan
keluar menyusuri sinar datang.
\end{theorem}
\admitted

\begin{remark}\label{rem:g10:refraction:honest}
Hukum pemantulan dan hukum \omterm{def:g10:refraction:refraction}{pembiasan} di sini dinyatakan sebagai fakta
percobaan. Keduanya diturunkan secara jujur pada jilid Tahun ke-1, saat
cahaya diperlakukan sebagai gelombang: kedua hukum itu --- dan rumus
$n = c/v$ itu sendiri --- lalu mengikuti dari melambatnya gelombang di
dalam medium yang lebih rapat.
\end{remark}

\begin{example}[Dari udara ke air]\label{ex:g10:refraction:airwater}
Sebuah sinar mengenai kolam yang tenang pada $i_1 = 30^\circ$. Dengan
$n_1 = 1.00$ dan $n_2 = 1.33$:
\[
\sin i_2 = \frac{n_1 \sin i_1}{n_2} = \frac{\sin 30^\circ}{1.33}
= \frac{0.500}{1.33} = 0.376,
\qquad
i_2 = 22.1^\circ .
\]
Sinar itu membelok mendekati \omterm{def:g10:refraction:angles}{garis normal}, sebagaimana selalu terjadi
ketika memasuki medium yang lebih lambat ($n_2 > n_1$). Ketika
meninggalkan air, perhitungan yang sama berjalan mundur dan sinarnya
membelok \emph{menjauhi} \omterm{def:g10:refraction:angles}{garis normal}.
\end{example}

\begin{method}[Menyelesaikan soal pembiasan]\label{met:g10:refraction:method}
\begin{enumerate}
  \item Gambarlah permukaannya, \omterm{def:g10:refraction:angles}{garis normalnya}, dan sinarnya; kenali
        $n_1$ (medium tempat sinar datang) dan $n_2$.
  \item Periksalah bahwa sudut yang diketahui memang diukur dari \omterm{def:g10:refraction:angles}{garis
        normal}; ubahlah jika sudut itu diukur dari permukaan.
  \item Tulislah $n_1 \sin i_1 = n_2 \sin i_2$ lalu pisahkan sinus yang
        belum diketahui; peroleh sudutnya dengan sinus invers pada
        kalkulator, seperti pada jilid matematika.
  \item Periksalah kewajarannya: sinarnya membelok mendekati \omterm{def:g10:refraction:angles}{garis
        normal} jika $n_2 > n_1$, menjauhinya jika $n_2 < n_1$, dan
        $\sin i_2$ harus keluar di antara $0$ dan $1$.
\end{enumerate}
\end{method}

Percobaan di balik hukum ini muat di atas satu meja: setengah cakram
pleksiglas, sebuah busur derajat, sebuah laser. Pasangan sudut mentah
$(i_1, i_2)$ tampak tidak beraturan --- Ptolemaios menabulasikannya
sekitar tahun 150 dan keliru menyimpulkan bahwa $i_2$ sebanding dengan
$i_1$. Gambarkanlah sinus terhadap sinus, dan datanya meluruskan diri
menjadi garis melalui titik asal dengan kemiringan $n_1/n_2$: garis
itu \emph{adalah} \omterm{thm:g10:refraction:snell}{hukum Snell}.

\begin{proposition}[Kedalaman semu]\label{prop:g10:refraction:depth}
\index{kedalaman semu}
Sebuah benda pada kedalaman $d$ di bawah air, dipandang dari atas pada
sudut yang kecil, tampak berada pada \emph{\omterm{prop:g10:refraction:depth}{kedalaman semu}}
\[
d' = \frac{d}{n} ,
\]
dengan $n$ adalah \omterm{def:g10:refraction:index}{indeks bias} air. Kolam sedalam \qty{1.20}{m} tampak
hanya sedalam $1.20/1.33 = \qty{0.90}{m}$ --- penyebab klasik lompatan
yang gegabah.
\end{proposition}
\admitted

\begin{remark}\label{rem:g10:refraction:depthhonest}
Rumus itu mengikuti dari \omterm{thm:g10:refraction:snell}{hukum Snell} yang diterapkan pada sinar yang
hampir tegak menuju mata; penurunannya dikerjakan pada jilid Tahun ke-1.
Arah gejalanya sendiri tidak memerlukan rumus: sinar yang meninggalkan
air membelok menjauhi \omterm{def:g10:refraction:angles}{garis normal}, sehingga mata menelusurinya kembali
ke titik yang lebih tinggi daripada bendanya.
\end{remark}

\section{Dispersi: satu hukum untuk tiap warna}

Prisma mengubah cahaya matahari yang putih menjadi pelangi, seperti
diperlihatkan \omterm{def:g10:light-spectra:white}{spektrum} pada \cref{ch:g10:light-spectra}. \omterm{def:g10:refraction:refraction}{Pembiasan}
menjelaskan \emph{mengapa} demikian: \omterm{def:g10:refraction:index}{indeks bias} sebuah medium tidak
persis sama untuk setiap warna.

\begin{definition}[Dispersi]\label{def:g10:refraction:dispersion}
Sebuah medium bersifat \emph{dispersif}\index{dispersi} apabila indeks
biasnya bergantung pada \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} cahayanya. Di dalam kaca dan
air, $n$ sedikit lebih besar untuk cahaya biru (\omterm{def:g10:light-spectra:wavelength}{panjang gelombang}
pendek) daripada untuk cahaya merah (\omterm{def:g10:light-spectra:wavelength}{panjang gelombang} panjang) ---
sehingga cahaya birulah yang membelok lebih kuat.
\end{definition}

\begin{example}[Dua warna, satu permukaan]\label{ex:g10:refraction:twocolors}
Seberkas \omterm{def:g10:light-spectra:white}{cahaya putih} mengenai kaca kron pada $i_1 = 60^\circ$. Untuk
kaca ini $n_{\text{merah}} = 1.51$ dan $n_{\text{biru}} = 1.53$. \omterm{thm:g10:refraction:snell}{Hukum
Snell}, sekali untuk tiap warna:
\[
\sin i_{\text{merah}} = \frac{\sin 60^\circ}{1.51} = 0.574,
\quad i_{\text{merah}} = 35.0^\circ ;
\qquad
\sin i_{\text{biru}} = \frac{\sin 60^\circ}{1.53} = 0.566,
\quad i_{\text{biru}} = 34.5^\circ .
\]
Satu permukaan saja sudah memecah \omterm{def:g10:light-spectra:white}{cahaya putih} sebesar setengah derajat
--- tak terlihat pada kaca jendela, tetapi prisma membiaskan dua kali
dengan arah putar yang sama, melebarkan kipasnya, dan beberapa \omterm{def:g10:orders-of-magnitude:unit}{meter}
kemudian warnanya sudah terpisah beberapa sentimeter.
\end{example}

\begin{remark}\label{rem:g10:refraction:rainbow}
Pelangi adalah perhitungan yang sama yang dijalankan di dalam tetes
hujan: cahaya matahari dibiaskan ketika masuk ke tetes itu, dipantulkan
pada dinding belakangnya, lalu dibiaskan lagi ketika keluar. Indeks air
berkisar dari $1.331$ (merah) sampai $1.343$ (biru), sehingga setiap
warna keluar pada sudutnya sendiri --- dan tidak ada dua pengamat yang
melihat pelangi yang sama.
\end{remark}

\section{Pemantulan sempurna}

\omterm{thm:g10:refraction:snell}{Hukum Snell} menyimpan sebuah jebakan. Dari air menuju udara, sinar
biasnya membelok menjauhi \omterm{def:g10:refraction:angles}{garis normal}:
$\sin i_2 = 1.33 \sin i_1 > \sin i_1$. Perbesarlah $i_1$, dan rumus itu
segera menuntut $\sin i_2 > 1$, yang tidak dapat dipenuhi sudut mana
pun. \omterm{def:g10:refraction:refraction}{Pembiasannya} lalu berhenti terjadi sama sekali.

\begin{definition}[Sudut kritis dan pemantulan sempurna]\label{def:g10:refraction:critical}
Misalkan cahaya merambat di dalam medium berindeks $n_1$ menuju medium
berindeks lebih kecil $n_2 < n_1$. Yang disebut \emph{sudut
kritis}\index{sudut kritis} $i_c$ adalah \omterm{def:g10:refraction:angles}{sudut datang} yang membuat sinar
biasnya menyusur permukaan ($i_2 = 90^\circ$). Untuk $i_1 > i_c$ tidak
ada \omterm{def:g10:refraction:refraction}{sinar bias} sama sekali: permukaannya memantulkan seluruh cahaya
kembali ke medium pertama, mengikuti hukum pemantulan. Peristiwa ini
disebut \emph{pemantulan sempurna}\index{pemantulan sempurna}.
\end{definition}

\begin{proposition}[Rumus sudut kritis]\label{prop:g10:refraction:criticalformula}
Untuk $n_1 > n_2$,
\[
\sin i_c = \frac{n_2}{n_1} .
\]
\end{proposition}

\begin{proof}
Ambillah $i_2 = 90^\circ$ pada \omterm{thm:g10:refraction:snell}{hukum Snell}: $n_1 \sin i_c = n_2 \sin 90^\circ
= n_2$. Untuk $i_1 > i_c$, \omterm{thm:g10:refraction:snell}{hukum Snell} akan menuntut
$\sin i_2 = \frac{n_1}{n_2}\sin i_1 > 1$: tidak ada arah bias yang
mungkin. (Bahwa cahayanya lalu dipantulkan \emph{seluruhnya} --- bukan
diserap --- adalah fakta percobaan, yang dibenarkan oleh teori gelombang
pada jilid Tahun ke-1.)
\end{proof}

\begin{example}[Tiga sudut kritis]\label{ex:g10:refraction:criticalvalues}
Menuju udara: untuk air, $\sin i_c = 1/1.33 = 0.752$, sehingga
$i_c = 48.8^\circ$; untuk kaca, $\sin i_c = 1/1.50$, sehingga
$i_c = 41.8^\circ$; untuk intan, $\sin i_c = 1/2.42 = 0.413$, sehingga
$i_c = 24.4^\circ$. Makin besar indeksnya, makin sempit kerucut
keluarnya: di dalam intan, setiap sinar yang lebih dari $24.4^\circ$
dari \omterm{def:g10:refraction:angles}{garis normal} sebuah faset terperangkap --- itulah rahasia
kilaunya, yang dibedah pada \cref{pb:g10:refraction:1}.
\end{example}

\section{Serat optik}

\omterm{def:g10:refraction:critical}{Pemantulan sempurna} mengubah benang kaca menjadi pipa bagi cahaya:
sinar yang masuk hampir sejajar sumbunya mengenai dindingnya jauh di
luar \omterm{def:g10:refraction:critical}{sudut kritis}, memantul tanpa kehilangan, dan melakukannya jutaan
kali per kilometer tanpa bocor.

\begin{definition}[Serat optik]\label{def:g10:refraction:fiber}
Sebuah \emph{serat optik}\index{serat optik} adalah benang kaca tipis
yang terbuat dari \emph{inti}\index{inti serat} berindeks $n_1$ yang
terbungkus \emph{selubung}\index{selubung serat} berindeks sedikit lebih
kecil, $n_2 < n_1$. Cahaya yang disuntikkan pada salah satu ujungnya
merambat sepanjang intinya melalui \omterm{def:g10:refraction:critical}{pemantulan sempurna} berulang pada
batas inti--selubung.
\end{definition}

\begin{example}[Sebuah serat dalam angka]\label{ex:g10:refraction:fibernumbers}
Dengan $n_1 = 1.48$ dan $n_2 = 1.46$,
\[
\sin i_c = \frac{1.46}{1.48} = 0.986,
\qquad i_c = 80.6^\circ :
\]
hanya sinar yang menyimpang sekitar $9^\circ$ dari sumbunya yang
dipandu --- dan itu tidak menjadi masalah, sebab persis seperti itulah
cahaya laser disuntikkan. Di dalam \omterm{def:g10:refraction:fiber}{intinya} cahaya merambat pada
$v = c/1.48 = \qty{2.03e8}{m/s}$: sebuah pulsa menyeberangi serat
sepanjang \qty{100}{km} dalam waktu sekitar setengah milisekon.
\end{example}

\begin{remark}[Orde besar sambungan data]\label{rem:g10:refraction:datarates}
Kabel bawah laut --- berkas berisi beberapa lusin serat di dasar
samudra --- mengangkut hampir seluruh lalu lintas antarbenua. Satu serat
modern mengangkut sekitar $10^{13}$ bit per sekon dengan mengirimkan
banyak \omterm{def:g10:light-spectra:wavelength}{panjang gelombang} sekaligus; satu kabel utuh mencapai $10^{14}$
bit per sekon, dengan satu penguat setiap kira-kira \qty{80}{km}.
Sebagai perbandingan, satelit menangani sebagian kecil dari satu persen
lalu lintasnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:refraction:1}
Seberkas laser mengenai cermin datar pada $25^\circ$ \emph{dari
permukaan cermin}. Sebutkan \omterm{def:g10:refraction:angles}{sudut datangnya}, sudut pantulnya, dan sudut
antara berkas datang dan berkas pantulnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:2}
\begin{enumerate}
  \item Hitunglah laju cahaya di dalam air ($n = 1.33$) dan di dalam
        intan ($n = 2.42$).
  \item Di dalam sebuah zat padat tertentu, cahaya merambat pada
        \qty{1.97e8}{m/s}. Hitunglah indeks biasnya dan kenali
        bahannya.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:3}
Sebuah sinar di udara mengenai balok kaca ($n = 1.50$) pada
$i_1 = 40^\circ$. Hitunglah sudut biasnya. Pada sudut berapa sebuah
sinar yang merambat di dalam kaca harus mengenai permukaannya agar
keluar ke udara pada $40^\circ$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:4}
Cahaya matahari mencapai permukaan sebuah danau pada $55^\circ$ dari
arah tegak. Hitunglah sudut sinar biasnya, dan nyatakan apakah sinarnya
membelok mendekati atau menjauhi \omterm{def:g10:refraction:angles}{garis normal} --- beserta alasannya
dalam satu kata.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:refraction:5}
Untuk mengenali sebuah zat cair yang bening, sebuah sinar dikirimkan ke
permukaannya pada $i_1 = 45^\circ$; sinar biasnya terukur pada
$i_2 = 32^\circ$. Hitunglah indeks biasnya dan kenali zat cair itu dari
tabel pada \cref{def:g10:refraction:index}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:6}
Setengah cakram dari sebuah plastik bening memberikan pengukuran berikut:
\begin{center}
\begin{tabular}{lccccc}
$i_1$ & $20^\circ$ & $30^\circ$ & $40^\circ$ & $50^\circ$ & $60^\circ$ \\
$i_2$ & $13.5^\circ$ & $19.5^\circ$ & $25.5^\circ$ & $31.0^\circ$
      & $35.5^\circ$ \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item Hitunglah $\dfrac{\sin i_1}{\sin i_2}$ untuk setiap pasangan.
        Apa yang kamu amati?
  \item Simpulkan \omterm{def:g10:refraction:index}{indeks bias} plastik itu dan kenali bahannya.
  \item Mengapa menggambarkan $\sin i_2$ terhadap $\sin i_1$ merupakan
        uji \omterm{thm:g10:refraction:snell}{hukum Snell} yang lebih baik daripada menggambarkan $i_2$
        terhadap $i_1$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:7}
Hitunglah \omterm{def:g10:refraction:critical}{sudut kritis} menuju udara untuk air, untuk kaca ($n = 1.50$)
dan untuk intan. Lalu jelaskan, dengan \omterm{thm:g10:refraction:snell}{hukum Snell}, mengapa tidak ada
\omterm{def:g10:refraction:critical}{sudut kritis} bagi cahaya yang berjalan \emph{dari} udara \emph{ke
dalam} air.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:8}
\begin{enumerate}
  \item Sebuah kolam dalamnya \qty{2.0}{m}. Kedalaman berapa yang
        ditangkap seorang perenang yang berdiri di tepinya?
  \item Seekor bangau mengincar ikan yang berenang \qty{40}{cm} di
        bawah permukaan. Pada kedalaman berapa ikan itu tampak, dan
        haruskah bangau itu mematuk di atas, tepat pada, atau di bawah
        bayangan yang dilihatnya?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:9}
Sebuah sinar mengenai kaca jendela ($n = 1.50$, kedua sisinya sejajar)
pada $50^\circ$.
\begin{enumerate}
  \item Hitunglah sudut biasnya pada sisi pertama.
  \item Sinar biasnya mengenai sisi kedua dari dalam, pada sudut yang
        sama. Hitunglah sudut keluarnya dan bandingkan dengan
        $50^\circ$.
  \item Jadi, apa yang dilakukan kaca jendela yang tebal terhadap arah
        dan terhadap kedudukan sinar yang menyeberanginya?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:10}
\omterm{def:g10:light-spectra:white}{Cahaya putih} mengenai balok kaca flinta pada $55^\circ$. Untuk kaca ini,
$n_{\text{merah}} = 1.61$ dan $n_{\text{biru}} = 1.66$. Hitunglah sudut
bias sinar merah dan sinar biru serta sudut antara keduanya. Warna
manakah yang berakhir lebih dekat dengan \omterm{def:g10:refraction:angles}{garis normal}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:refraction:11}
Sebuah \omterm{def:g10:refraction:fiber}{serat optik} memiliki \omterm{def:g10:refraction:fiber}{inti} berindeks $1.48$ dan \omterm{def:g10:refraction:fiber}{selubung}
berindeks $1.46$.
\begin{enumerate}
  \item Hitunglah \omterm{def:g10:refraction:critical}{sudut kritis} pada batas inti--selubung.
  \item Sebuah sinar terpandu memantul-mantul tepat pada \omterm{def:g10:refraction:critical}{sudut kritis}.
        Tunjukkan bahwa sepanjang \qty{1.0}{km} serat yang lurus, sinar
        itu menempuh sekitar \qty{14}{m} lebih jauh daripada sinar yang
        berjalan lurus menyusuri sumbunya.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:12}
Mata seorang penyelam berada \qty{2.0}{m} di bawah permukaan yang
benar-benar tenang. Karena \omterm{def:g10:refraction:refraction}{pembiasan}, penyelam itu melihat
\emph{seluruh} langit termampatkan menjadi satu cakram terang tepat di
atas kepalanya --- jendela Snell.
\begin{enumerate}
  \item Jelaskan mengapa cahaya dari kaki langit mencapai mata penyelam
        pada \omterm{def:g10:refraction:critical}{sudut kritis} air, yaitu $48.8^\circ$ dari arah tegak.
  \item Hitunglah jari-jari cakram terang itu pada permukaannya, dengan
        memakai tangen pada segitiga siku-siku mata--garis tegak--tepi
        cakram.
  \item Apa yang dilihat penyelam itu pada permukaan \emph{di luar}
        cakram tersebut?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:13}
Sebuah prisma bersudut $30^\circ$, $60^\circ$, $90^\circ$. Seberkas
tipis \omterm{def:g10:light-spectra:white}{cahaya putih} masuk tegak lurus pada salah satu sisi yang mengapit
sudut siku-sikunya, menyeberang tanpa dibelokkan, lalu mengenai sisi
panjangnya dari dalam pada $i_1 = 30^\circ$. Untuk kaca ini,
$n_{\text{merah}} = 1.51$ dan $n_{\text{biru}} = 1.53$.
\begin{enumerate}
  \item Hitunglah sudut keluar sinar merah dan sinar biru serta lebar
        sudut kipas di antara keduanya.
  \item Prisma itu diganti dengan prisma
        $45^\circ$--$45^\circ$--$90^\circ$, sehingga \omterm{def:g10:refraction:angles}{sudut datang} di
        dalamnya menjadi $45^\circ$. Tunjukkan bahwa tidak ada cahaya
        sama sekali yang keluar melalui sisi panjangnya.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:14}
Periskop melipat cahaya dengan prisma kaca
$45^\circ$--$45^\circ$--$90^\circ$ ($n = 1.50$): berkasnya masuk tegak
lurus pada salah satu sisi pendek dan mengenai sisi panjangnya dari
dalam pada $45^\circ$.
\begin{enumerate}
  \item Tunjukkan bahwa sisi panjang itu berlaku sebagai cermin yang
        sempurna. Mengapa hal ini lebih baik daripada cermin logam,
        yang menyerap beberapa persen pada setiap pemantulan?
  \item Periskop itu kemasukan air dan sisi panjangnya kini bersentuhan
        dengan air. Hitunglah \omterm{def:g10:refraction:critical}{sudut kritis} baru pada batas kaca--air
        dan tunjukkan bahwa cerminnya gagal.
  \item Pada sudut berapa cahaya pada pertanyaan 2 meninggalkan prisma
        itu menuju air?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:refraction:15}
Sebuah kabel lintas Atlantik membentang \qty{6200}{km} serat (indeks
\omterm{def:g10:refraction:fiber}{intinya} $1.47$) antara London dan New York.
\begin{enumerate}
  \item Hitunglah laju cahaya di dalam \omterm{def:g10:refraction:fiber}{intinya} dan waktu tempuh satu
        arah sebuah pulsa.
  \item Sebuah gelombang radio merambat dengan laju $c$. Berapa lama
        ia memerlukan waktu pada jarak yang sama, dan berapa banyak
        waktu pulang pergi yang dikorbankan kaca serat itu dibandingkan
        dengan radio?
  \item Beberapa perusahaan keuangan membayar mahal demi memangkas
        beberapa milisekon dari sambungan ini. Usulkan dua cara fisis
        yang berbeda agar cahayanya tiba lebih awal.
\end{enumerate}
\end{exercise}

\section{Soal: Intan, kolam, dan serat}

\begin{problem}[{Soal akhir pekan --- tiga kehidupan hukum Snell: kolam
yang berbohong tentang kedalamannya, batu yang dipotong sedemikian rupa
sehingga cahaya tidak dapat keluar, dan samudra enam puluh milisekon di
bawah internet}]
\label{pb:g10:refraction:1}
Satu kesamaan antara dua sinus, tiga karier. Di kolam renang ia menipu
matamu tentang kedalaman dan memerasi seluruh langit menjadi sebuah
lingkaran. Di dalam intan, ketika didorong sampai batasnya, ia menolak
melepaskan cahaya --- dan para pengasah permata telah memotong batu di
sekitar penolakan itu selama berabad-abad. Di bawah Atlantik ia memandu
pulsa laser di dalam seuntai kaca, dan menetapkan batas laju internet
sedunia. Soal ini menelusuri ketiga kehidupan itu berurutan, dengan
hukum yang sama setiap kali.

\medskip
\noindent\textbf{Bagian I --- Pemanasan.}
\begin{enumerate}
  \item Hitunglah laju cahaya di dalam air ($n = 1.33$) dan di dalam
        intan ($n = 2.42$). Dengan faktor berapa intan memperlambat
        cahaya?
  \item Sebuah sinar di udara mengenai kolam pada $i_1 = 40^\circ$.
        Hitunglah sudut biasnya.
  \item Sebuah sinar di dalam air mengenai permukaan dari bawah pada
        $28.9^\circ$. Pada sudut berapa ia keluar ke udara? Sebutkan
        asas yang diperagakan hal ini.
  \item Hitunglah \omterm{def:g10:refraction:critical}{sudut kritis} batas air--udara, lalu uraikan apa yang
        terjadi pada sinar di dalam air yang mengenai permukaan pada
        $60^\circ$.
  \item Indeks udara adalah $1.0003$. Hitunglah laju cahaya di udara dan
        benarkan hampiran $n_{\text{udara}} = 1.00$ yang dipakai di
        sepanjang bab ini.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Kolam.}
\begin{enumerate}[resume]
  \item Sebuah kolam dalamnya \qty{3.0}{m}. Dengan rumus \omterm{prop:g10:refraction:depth}{kedalaman semu}
        (\cref{prop:g10:refraction:depth}), hitunglah kedalaman yang
        ditangkap orang yang berdiri di atasnya.
  \item Jelaskan dengan sketsa dua sinar (diuraikan dengan kata-kata)
        mengapa sekeping uang logam di dasar tampak terangkat: apa yang
        dilakukan sinar yang meninggalkan air, dan di mana mata
        menempatkan titik potongnya?
  \item Mata seorang penyelam berada pada \qty{3.0}{m}. Hitunglah
        jari-jari jendela Snell --- cakram terang tempat penyelam itu
        melihat seluruh langit (\cref{exo:g10:refraction:12}).
  \item Sebuah lampu di dasar mengirimkan sinar ke atas pada $30^\circ$,
        $45^\circ$ dan $50^\circ$ dari arah tegak. Untuk tiap sinar,
        nyatakan apakah ia lolos dan pada sudut berapa, atau apa yang
        terjadi sebagai gantinya.
  \item Seorang penombak ikan di tepi membidik seekor ikan. Haruskah
        tombaknya diarahkan di atas, tepat pada, atau di bawah bayangan
        ikan itu --- dan apakah ikan itu, ketika menoleh, juga melihat
        si penombak tergeser?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Intan.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g10:refraction:critical}{sudut kritis} intan ($n = 2.42$) menuju udara, lalu
        bandingkan dengan \omterm{def:g10:refraction:critical}{sudut kritis} kaca ($n = 1.52$), yang bernilai
        $41.1^\circ$.
  \item Sebuah sinar di dalam batu itu mengenai sebuah faset pada
        $30^\circ$ dari \omterm{def:g10:refraction:angles}{garis normalnya}. Apa yang terjadi di dalam
        intan? Dan di dalam tiruan dari kaca? Jelaskan dalam satu
        kalimat mengapa cara pemotongan intan, dipadu dengan sudut
        kritisnya yang mungil, menghasilkan kilau itu.
  \item Intan juga sangat \omterm{def:g10:refraction:dispersion}{dispersif}: $n_{\text{merah}} = 2.41$,
        $n_{\text{biru}} = 2.45$. \omterm{def:g10:light-spectra:white}{Cahaya putih} memasuki sebuah faset
        pada $45^\circ$; hitunglah sudut bias untuk tiap warna dan
        pemisahan sudutnya.
  \item Kemudian, sinar merah dan sinar biru sama-sama mencapai faset
        keluar dari dalam pada $17.0^\circ$. Hitunglah kedua sudut
        keluarnya dan pemisahan kipas yang muncul. Bandingkan dengan
        pemisahan pada saat masuk: apa yang terjadi pada \omterm{def:g10:refraction:dispersion}{dispersi} di
        dekat \omterm{def:g10:refraction:critical}{sudut kritis}?
  \item Setetes air ($n = 1.33$) melapisi sebuah faset. Hitunglah \omterm{def:g10:refraction:critical}{sudut
        kritis} baru batas intan--air, putuskan nasib sinar $30^\circ$
        pada pertanyaan 12, dan jelaskan mengapa para pengasah permata
        menjaga intan tetap bersih benar.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Serat.}
\begin{enumerate}[resume]
  \item Sebuah serat memiliki \omterm{def:g10:refraction:fiber}{inti} berindeks $1.48$ dan \omterm{def:g10:refraction:fiber}{selubung}
        berindeks $1.46$. Hitunglah \omterm{def:g10:refraction:critical}{sudut kritis} pada batas
        inti--selubung.
  \item Benang kaca telanjang di udara akan bersudut kritis
        $42.5^\circ$ --- sepintas jauh lebih baik. Berikan dua alasan
        mengapa \omterm{def:g10:refraction:fiber}{selubung} itu tetap dipakai. (Pikirkanlah debu, goresan,
        dan apa yang akan terjadi pada \omterm{def:g10:refraction:critical}{pemantulan sempurna} jika
        permukaannya disentuh.)
  \item Hitunglah laju cahaya di dalam \omterm{def:g10:refraction:fiber}{intinya}, lalu waktu tempuh satu
        arah sebuah pulsa sepanjang \qty{6000}{km} serat.
  \item Sinar yang memantul tepat pada \omterm{def:g10:refraction:critical}{sudut kritis} menempuh
        $1/\sin i_c$ kali panjang sumbunya. Hitunglah jarak tambahannya
        sepanjang \qty{6000}{km} itu dan tambahan waktu yang
        dikorbankannya.
  \item Puncak ceritanya. Sepasang serat modern mengangkut sekitar
        $10^{13}$ bit per sekon, sedangkan buku ini kira-kira
        $2.4 \times 10^8$ bit. Hitunglah waktu pulang pergi (``ping'')
        melintasi samudra itu dan banyaknya salinan buku ini yang
        dipindahkan serat itu tiap sekon --- lalu nyatakan dalam satu
        kalimat apa yang dikerjakan \omterm{thm:g10:refraction:snell}{hukum Snell} bagi internet.
\end{enumerate}
\end{problem}
