\chapter{كيف ينتقل الصوت}\label{ch:g4:how-sound-travels}

ومضة البرق --- ثم ثانية، فثانيتان، فثلاث من الصمت --- ثم الآن
فقط زمجرة الرعد. مع أن الومضة والزمجرة وُلدتا معًا في الغيمة
نفسها! \omterm{def:g3:sound-around-us:sound}{والصوت}، كما تعلّمنا، ارتعاش ينتقل. وهذا الفصل يتبعه على
الطريق: في أيّ شيء ينتقل، وبأيّ سرعة يجري، وماذا يحدث حين يصطدم
بجدار.

\section{الصوت يحتاج إلى حامل}

\begin{proposition}[لا صوت بلا حامل]\label{prop:g4:how-sound-travels:carrier}
لا ينتقل \omterm{def:g3:sound-around-us:sound}{الصوت} إلا \emph{عبر} الأشياء: عبر \omterm{def:g2:air-around-us:air}{الهواء}، وعبر الماء،
وعبر الخشب والآجرّ والفولاذ. وحيث لا شيء البتة --- لا \omterm{def:g2:air-around-us:air}{هواء} ولا
مادة من أي نوع --- لا يستطيع \omterm{def:g3:sound-around-us:sound}{الصوت} أن ينتقل: فليس للارتعاش ما
يرتعش فيه.
\end{proposition}

\begin{example}[الجرس في الوعاء]\label{ex:g4:how-sound-travels:jar}
تجربة مشهورة عمرها ثلاثة قرون: جرس كهربائي يرنّ داخل وعاء زجاجي
--- ويُسمع بوضوح. ثم تسحب مضخّة \omterm{def:g2:air-around-us:air}{الهواء} من الوعاء المحكم بينما
تواصل مطرقة الجرس ضربها على مرأى العين. فيخبو الرنين\dots\
ويموت. ثم دع \omterm{def:g2:air-around-us:air}{الهواء} يندفع راجعًا: فيعود الرنين. ولم يتوقّف
الجرس قطّ --- إنما نُزع \emph{حامله} ثم أُعيد. وتستطيع الآن أن
تحسم سؤالًا قديمًا: \omterm{def:g2:moon-in-the-sky:moon}{فالقمر} الصامت، حيث تموت الصرخة بلا سامع،
صامت لأنه ليس فيه \omterm{def:g2:air-around-us:air}{هواء} البتة.
\end{example}

\begin{example}[حوامل خير من الهواء]\label{ex:g4:how-sound-travels:solids}
الأجسام الصلبة والماء تحمل \omterm{def:g3:sound-around-us:sound}{الصوت} حملًا بديعًا --- خيرًا من
\omterm{def:g2:air-around-us:air}{الهواء} في الغالب. ألصِق أذنك بالطاولة واطلب من صديق أن يخدش
طرفها البعيد خدشًا \emph{خفيفًا} جدًّا: فعبر \omterm{def:g2:air-around-us:air}{الهواء} لا تسمع
شيئًا، وعبر الخشب يأتيك الخدش عاليًا نقيًّا. والسبّاحون يسمعون
طرقات المسبح تحت الماء من طرف إلى طرف؛ والحيتان تتغنّى بعضها
لبعض عبر بحار كاملة. والجدران السميكة تنقل ثقب الجار إلى كل
غرفة --- فالأجسام الصلبة تكاد تكون بارعة في هذا أكثر ممّا
ينبغي.
\end{example}

\begin{method}[هاتف الخيط]\label{met:g4:how-sound-travels:telephone}
\begin{enumerate}
  \item اثقب ثقبًا صغيرًا في قاع كوبين ورقيّين؛
  \item وأدخِل طرفَي خيط طويل فيهما، واعقد كل طرف حتى لا ينزلق
        خارجًا؛
  \item وابتعدا حتى يصير الخيط \emph{مشدودًا} ولا يلمس شيئًا في
        الطريق؛
  \item ثم يتكلّم أحدكما همسًا في كوبه؛ ويضع الآخر كوبه على
        أذنه.
\end{enumerate}
تصل الهمسة واضحة عبر الساحة: فصوتك يجعل الكوب يرتعش، ويجري
الارتعاش على الخيط المشدود، فيعيده الكوب البعيد. وأرخِ الخيط أو
اقرصه بإصبعين، فيموت الهاتف --- فقد قُطع الطريق.
\end{method}

\section{الصوت يستغرق وقتًا}

\begin{proposition}[سريع لكن لا فوري]\label{prop:g4:how-sound-travels:speed}
\omterm{def:g3:sound-around-us:sound}{الصوت} عدّاء لا ساحر: فهو يقطع في \omterm{def:g2:air-around-us:air}{الهواء} نحو \qty{340}{m} في كل
ثانية --- أي ثلاث ثوانٍ للكيلومتر تقريبًا. أما \omterm{def:g1:light-and-shadows:source}{الضوء} فيصل من
السرعة بحيث تُعدّ رحلته فورية في كل مشهد تستطيع أن تراه. وذلك
الفرق هو قصة العاصفة الرعدية كلها.
\end{proposition}

\begin{method}[كم تبعد العاصفة؟]\label{met:g4:how-sound-travels:storm}
\begin{enumerate}
  \item حين ترى الومضة --- ابدأ عدّ الثواني بانتظام: تمساح صغير
        واحد، تمساحان صغيران، \dots؛
  \item وتوقّف عند أول زمجرة رعد؛
  \item واقسم عدّتك على $3$: فذلك بعد العاصفة بالكيلومترات
        تقريبًا.
\end{enumerate}
ستّ ثوانٍ: نحو \qty{2}{km}. وقِصَر العدّات بين صاعقة وأخرى يعني
أن العاصفة تقترب --- فقد آن أن تدخل.
\end{method}

\begin{example}[الأصداء]\label{ex:g4:how-sound-travels:echo}
اصرخ نحو جرف كبير أو مبنى منفرد بعيد عبر حقل: ``مرحبًا!'' ---
لحظة صمت --- ``مرحبًا!'' يجيبك صوتك أنت. \omterm{def:g3:sound-around-us:sound}{فالصوت} يرتدّ عن
الجدران القاسية الكبيرة، \omterm{def:g3:sound-around-us:sound}{والصوت} المرتدّ الذي يعود إليك متأخّرًا
بما يكفي لتسمعه منفصلًا هو \emph{الصدى}\index{صدى}. وكلما بَعُد
الجدار طالت رحلة صوتك ذهابًا وإيابًا وتأخّر \omterm{ex:g4:how-sound-travels:echo}{الصدى}. أما في غرفة
صغيرة فيعود الارتداد أسرع من أن يُسمع منفصلًا --- ولا يفعل إلا
أن يجعل صوتك أملأ.
\end{example}

\begin{remark}[السونار: الصدى وقد وُضع في العمل]\label{rem:g4:how-sound-travels:sonar}
\omterm{def:g2:measuring-length:measure}{تقيس} السفن عمق البحر بالصياح إلى الأسفل --- نقرة في الماء ---
\omterm{def:g2:measuring-length:measure}{وقياس} زمن \omterm{ex:g4:how-sound-travels:echo}{الصدى} العائد من القاع. فالماء العميق صداه متأخّر؛
والضحل صداه سريع. والدلافين والخفافيش وُلدت عارفةً بالحيلة:
تنقر وتصغي، وتقرأ الأصداء لتصطاد في الماء المظلم والكهوف
الحالكة. وأنت باستعمالك ثواني الرعد قد استعملت طريقتها أصلًا:
قِس زمن \omterm{def:g3:sound-around-us:sound}{الصوت} تعرف المسافة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:1}
سمِّ ثلاثة حوامل \omterm{def:g3:sound-around-us:sound}{للصوت} من هذا الفصل، وموضعًا واحدًا لا يستطيع
\omterm{def:g3:sound-around-us:sound}{الصوت} أن ينتقل فيه البتة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:2}
في تجربة الوعاء تواصل مطرقة الجرس ضربها بينما يموت الرنين. فما
الذي نُزع بالضبط، وماذا يثبت ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:3}
لماذا يكون \omterm{def:g2:moon-in-the-sky:moon}{القمر} عالَمًا صامتًا؟ أجب بقضية الفصل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:4}
في هاتف الخيط، ما الذي يحمل الهمسة؟ واذكر قاعدتَي إبقاء الهاتف
عاملًا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:5}
كم يقطع \omterm{def:g3:sound-around-us:sound}{الصوت} في \omterm{def:g2:air-around-us:air}{الهواء} في ثانية واحدة تقريبًا؟ وكم يقطع في
ثلاث ثوانٍ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:6}
تعدّ تسع ثوانٍ بين الومضة والرعد. فكم تبعد العاصفة؟ ثم يعدّ
صديقك ثلاث ثوانٍ فقط عند الصاعقة التالية --- فما الذي تغيّر،
وماذا ينبغي أن تفعلا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:7}
لماذا يتيح لك إلصاق أذنك بالطاولة أن تسمع خدشًا لم يوصله \omterm{def:g2:air-around-us:air}{الهواء}
قطّ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:8}
ما \omterm{ex:g4:how-sound-travels:echo}{الصدى}؟ ولماذا يجيب جرف أكبر وأبعد متأخّرًا عن جرف قريب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:9}
كيف يميّز سونار السفينة الماء العميق من الضحل؟ وأيّ حيوانين
يلعبان اللعبة نفسها، وأين؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:how-sound-travels:10}
تُظهر أفلام الغرب القديمة كشّافةً يُلصقون آذانهم بقضيب السكّة
الفولاذي ليلتقطوا قطارًا ما زال بعيدًا. فلماذا يفضح القضيبُ
القطارَ قبل \omterm{def:g2:air-around-us:air}{الهواء} بزمن طويل؟ اذكر الفائدتين اللتين يقدّمهما
القضيب \omterm{def:g3:sound-around-us:sound}{للصوت}. (إحداهما عن حسن الحمل؛ وأما الأخرى فتذكّر أن
القضيب يمتدّ مستقيمًا إلى القطار بينما ينتشر \omterm{def:g3:sound-around-us:sound}{الصوت} في \omterm{def:g2:air-around-us:air}{الهواء} في
كل اتجاه مثل حلقات البركة، فيرقّ كلما مضى.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:how-sound-travels:11}
تقف في فناء كبير خالٍ فتصفّق مرة فتسمع صفعة تصفيقتك عائدة من
الجدار البعيد بعد ثانية كاملة. فقد جرى \omterm{def:g3:sound-around-us:sound}{الصوت} إلى الجدار وعاد في
تلك الثانية. فكم يبعد الجدار --- وانتبه، وفكّر في الرحلة ذهابًا
وإيابًا!
\end{exercise}
