\chapter{المغانط والبوصلة}\label{ch:g4:magnets-and-compass}

اكتشفت منذ سنوات أن طرفَي \omterm{def:g1:magnets:magnet}{المغناطيس} مختلفان: ففي وضع يلتصق
مغناطيسان، وإذا قُلبا تدافعا. وقد وعدنا بأن لذينك الطرفين اسمين
وعملًا مشهورًا --- هداية المسافرين عبر العالم. واليوم يُوفى
الوعد: أهلًا بالأقطاب المغناطيسية، وبالجهاز الصغير الذي اهتدى به
كل مستكشف قبل وجود الأقمار الاصطناعية.

\section{القطبان}

\begin{definition}[الأقطاب المغناطيسية]\label{def:g4:magnets-and-compass:poles}
طرفا \omterm{def:g1:magnets:magnet}{المغناطيس} الخاصّان --- حيث يكون جذبه أشدّ، وحيث تتجمّع
مشابك الورق --- هما \emph{قطباه
المغناطيسيان}\index{قطب مغناطيسي}. ويُسمّى أحدهما \emph{القطب
الشمالي}\index{قطب شمالي} (وكثيرًا ما يُطلى بالأحمر ويُعلَّم
بحرف N) والآخر \emph{القطب الجنوبي}\index{قطب جنوبي} (ويُعلَّم
بحرف S). ولكل \omterm{def:g1:magnets:magnet}{مغناطيس} القطبان معًا: واحد من كل نوع، دائمًا.
\end{definition}

\begin{proposition}[قاعدة القطبين]\label{prop:g4:magnets-and-compass:rule}
القطبان المختلفان يتجاذبان: فيلتصق القطب N بالقطب S. والقطبان المتماثلان
يتنافران: فيدفع N قطبًا مثله بعيدًا، وكذلك S. وهذه القاعدة
الواحدة تفسّر كل التصاق وكل رفض مرن أحسست به يومًا بين
مغناطيسين.
\end{proposition}

\begin{example}[قراءة تجارب قديمة]\label{ex:g4:magnets-and-compass:reading}
حين التصق مغناطيساك كان قطب N يواجه قطب S. وحين تمانعا في مرونة
كان N يواجه N (أو S يواجه S). وأما مشابك الورق فكانت تتجمّع عند
الطرفين دائمًا لأن القطبين هما موضع \omterm{def:g2:pushes-and-pulls:force}{قوة} \omterm{def:g1:magnets:magnet}{المغناطيس} --- ووسطه
سكون بينهما.
\end{example}

\section{الأرض مغناطيس}

\begin{example}[أعظم مغناطيس على الإطلاق]\label{ex:g4:magnets-and-compass:earth}
علّق مغناطيسًا مستقيمًا بخيط حتى يستطيع أن يدور بحرّية، وانتظر
حتى يستقرّ --- فيحدث شيء بديع: إذ يستقرّ دائمًا مشيرًا في
\emph{الجهة نفسها}، أحد قطبيه نحو أقصى شمال العالم والآخر نحو
أقصى جنوبه. فلا بدّ أن مغناطيسًا عملاقًا خفيًّا يسحب قطبيه إلى
الاصطفاف. وذلك العملاق هو الأرض نفسها: فكوكبنا كله \omterm{def:g1:magnets:magnet}{مغناطيس}،
\omterm{def:g4:magnets-and-compass:poles}{وقطباه المغناطيسيان} قرب أعلى الكرة الأرضية وأسفلها.
\end{example}

\begin{definition}[البوصلة]\label{def:g4:magnets-and-compass:compass}
\emph{البوصلة}\index{بوصلة} إبرة مغناطيسية صغيرة خفيفة موازَنة
على \omterm{def:g3:levers-and-scales:lever}{محور ارتكاز} دقيق حتى تستطيع أن تدور بحرّية. وإذا تُركت في
سلام أطاعت الإبرةُ مغناطيسَ الأرض: فيستقرّ طرفها المعلَّم مشيرًا
إلى الشمال. وتحت الإبرة قرص يبيّن \emph{الجهات
الأصلية}\index{جهات أصلية} الأربع: الشمال والشرق والجنوب
والغرب.
\end{definition}

\begin{method}[كيف تستعمل البوصلة]\label{met:g4:magnets-and-compass:use}
\begin{enumerate}
  \item أمسك \omterm{def:g4:magnets-and-compass:compass}{البوصلة} مستوية ثابتة، بعيدًا عن المغانط والأشياء
        الحديدية الكبيرة (الأسيجة والسيارات --- فهي تشوّش
        الإبرة)؛
  \item وانتظر حتى تكفّ الإبرة عن التأرجح؛
  \item وأدِر قرص \omterm{def:g4:magnets-and-compass:compass}{البوصلة} برفق حتى تقع علامة الشمال فيه تحت طرف
        الإبرة المعلَّم؛
  \item واقرأ جهاتك: الشرق عن يمينك وأنت مواجه الشمال، والغرب عن
        يسارك، والجنوب خلف ظهرك.
\end{enumerate}
وتحقّق بالسماء: \omterm{def:g1:light-and-shadows:shadow}{فظلّ} الظهر في هذه البلاد، كما في فصل \omterm{def:g3:sun-days-seasons:sundial}{المزولة}،
يشير إلى الشمال --- ويجب أن يتّفق \omterm{def:g1:light-and-shadows:shadow}{الظلّ} والإبرة.
\end{method}

\begin{example}[اصنع واحدة بنفسك]\label{ex:g4:magnets-and-compass:make}
دلّك إبرة خياطة فولاذية بأحد قطبَي \omterm{def:g1:magnets:magnet}{مغناطيس} عشرين مرة، في الجهة
نفسها دائمًا، مبعدًا \omterm{def:g1:magnets:magnet}{المغناطيس} بين الدلكة والأخرى: فتصير الإبرة
مغناطيسًا صغيرًا بنفسها. ثم ضعها برفق على شريحة فلّين تطفو في
وعاء ماء --- وراقب الفلّينة تدور على مهل حتى تشير الإبرة شمالًا
وجنوبًا. لقد بنيت \omterm{def:g4:magnets-and-compass:compass}{بوصلة} البحّار قبل ألف سنة.
\end{example}

\begin{remark}[الجهاز الذي فتح المحيطات]\label{rem:g4:magnets-and-compass:history}
قبل \omterm{def:g4:magnets-and-compass:compass}{البوصلة} كان البحّارة يلازمون السواحل ويرجون ليالي صافية
ليهتدوا بالنجوم. ومع إبرة عائمة صار في وسعهم أن يحفظوا مسارًا
في الضباب والعاصفة والظلمة بلا \omterm{def:g2:moon-in-the-sky:moon}{قمر} --- عبر محيطات كاملة. وقليل
من الأجهزة أعاد رسم خريطة العالم إعادة بهذا التمام. وبقيت أحجية
واحدة مفتوحة قرونًا: \emph{لماذا} تتصرّف الأرض مغناطيسًا؟
والقصة كاملة تنتظر في السنوات الجامعية من هذا المنهاج ---
وتستحقّ الانتظار.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:1}
ما قطبا \omterm{def:g1:magnets:magnet}{المغناطيس}، وما اسماهما؟ وأين تتجمّع مشابك الورق على
\omterm{def:g1:magnets:magnet}{مغناطيس} مستقيم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:2}
اذكر قاعدة القطبين. التصق مغناطيسان: فأيّ قطبين كانا متقابلين؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:3}
يقترب القطب N في \omterm{def:g1:magnets:magnet}{مغناطيس} من القطب N في آخر. ماذا يحدث؟ وماذا
يحدث حين يقترب N من S؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:4}
لماذا يستقرّ \omterm{def:g1:magnets:magnet}{المغناطيس} المستقيم المعلّق بحرّية مشيرًا في الجهة
نفسها دائمًا؟ وماذا يكشف هذا عن الأرض؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:5}
سمِّ \omterm{def:g4:magnets-and-compass:compass}{الجهات الأصلية} الأربع بالترتيب، ابتداءً من الشمال ودورانًا
إلى اليمين. وأنت مواجه الشمال، أيّ جهة عن يمينك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:6}
لماذا يجب إبعاد \omterm{def:g4:magnets-and-compass:compass}{البوصلة} عن المغانط والأسيجة الحديدية والسيارات
وأنت تقرؤها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:7}
صِف كيف تحوّل إبرة خياطة إلى إبرة \omterm{def:g4:magnets-and-compass:compass}{بوصلة}، وكيف تدعها تدور بحرّية
من غير \omterm{def:g3:levers-and-scales:lever}{محور ارتكاز}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:8}
من فصل \omterm{def:g3:sun-days-seasons:sundial}{المزولة}: إلى أيّ جهة يشير \omterm{def:g1:light-and-shadows:shadow}{ظلّ} الظهر في هذه البلاد؟ وكيف
يتحقّق ذلك من \omterm{def:g4:magnets-and-compass:compass}{بوصلة} --- أو يحلّ محلّ \omterm{def:g4:magnets-and-compass:compass}{بوصلة} ضائعة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:9}
لماذا كانت \omterm{def:g4:magnets-and-compass:compass}{البوصلة} كنزًا للبحّارة، \omterm{def:g2:measuring-length:measure}{بالقياس} إلى الاهتداء بالساحل
أو بالنجوم؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:10}
تمشي متنزّهة نحو كوخ جبلي يقع في الشمال تمامًا. وإبرة بوصلتها،
وهي ممسوكة على أصولها، تشير إلى إبزيم حزام صديقها بدل الشمال
كلما وقف قريبًا عن يسارها. فماذا يجري، وإلى أيّ جهة تُخشى أن
تنحرف، وما الحلّ في خطوة واحدة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:11}
نُشر \omterm{def:g1:magnets:magnet}{مغناطيس} نصفين بين قطبيه، أملًا في الحصول على قطب N وحده في
يد وقطب S وحده في الأخرى. ويقول تراث البحّارة --- وكل تجربة
جُرّبت يومًا --- إن كل نصف يخرج مغناطيسًا صغيرًا كاملًا له
القطبان \emph{معًا}. اختبر الدعوى في ذهنك بوصفة إبرة \omterm{def:g4:magnets-and-compass:compass}{البوصلة}:
فالإبرة المدلوكة قصيرة --- أفلها قطب واحد أم قطبان؟ وبمَ يوحي
هذا في حقّ اصطياد قطب وحده بالقطع؟
\end{exercise}
