\chapter{الوزن والكتلة}\label{ch:g4:weight-and-mass}

كان روّاد الفضاء على \omterm{def:g2:moon-in-the-sky:moon}{القمر} يثبون مثل أطفال على ترامبولين، وإن
حملوا حقائب ثقيلة. أفكانت حقائبهم ثقيلة هناك أم خفيفة؟ ولكي نجيب
على أصوله لا بدّ أن نشقّ كلمة يومية شقّين: ما الشيء \emph{مصنوع
منه} --- وهو كتلته --- وكم تسحبه الأرض \emph{بشدّة} --- وهو
وزنه. وهما ابنا عمّ قريبان، لا توأمان.

\section{فكرتان مختبئتان في كلمة ``ثقيل''}

أنت تعرف \omterm{def:g3:measuring-mass:mass}{الكتلة}: كم فيه من مادة، بالغرامات والكيلوغرامات، يحكم
بها \omterm{def:g3:levers-and-scales:scale}{ميزان الكفّتين}. وتعرف جذب الأرض: تلك \omterm{def:g2:pushes-and-pulls:force}{القوة} بلا لمس التي تُنزل
كل شيء مفلَت. \omterm{def:g4:weight-and-mass:weight}{والوزن} هو ملتقى الاثنين.

\begin{definition}[الوزن]\label{def:g4:weight-and-mass:weight}
\emph{وزن}\index{وزن} الشيء هو \omterm{def:g2:pushes-and-pulls:force}{القوة} التي تسحبه بها الأرض إلى
الأسفل. والوزن هو ما تصارعه ذراعك حين تحمل دلوًا، وما يمدّ
النابض حين يتعلّق الدلو به. وللوزن، مثل كل \omterm{def:g2:pushes-and-pulls:force}{قوة}، اتجاه: إلى
الأسفل مستقيمًا، دائمًا.
\end{definition}

\begin{proposition}[مادة أكثر وجذب أشدّ]\label{prop:g4:weight-and-mass:morestuff}
تسحب الأرض بشدّة أكبر الأشياءَ التي فيها مادة أكثر: ففي الموضع
نفسه تعني \omterm{def:g3:measuring-mass:mass}{الكتلة} الأكبر وزنًا أكبر دائمًا --- وقالبان متماثلان
يزنان ضعف ما يزنه واحد. ولهذا يفلح \omterm{def:g3:levers-and-scales:scale}{ميزان الكفّتين}، الذي يقارن
\emph{الجذبين} على كفّتيه، في مقارنة \emph{الكتلتين}: ففي موضع
واحد يعني تساوي الجذبين تساوي المادة.
\end{proposition}

\begin{example}[ابنا عمّ لا توأمان]\label{ex:g4:weight-and-mass:cousins}
\omterm{def:g3:measuring-mass:mass}{الكتلة} تجيب عن: \emph{كم فيه من مادة؟} وهي خاصّية للشيء وحده ---
واحدة في يدك، وتحت الماء، وعلى \omterm{def:g2:moon-in-the-sky:moon}{القمر}. \omterm{def:g4:weight-and-mass:weight}{والوزن} يجيب عن: \emph{كم
يُسحب بشدّة هنا؟} وهو يتوقّف على الشيء \emph{وعلى} موضعه. غيّر
الموضع وحده، فتبقى \omterm{def:g3:measuring-mass:mass}{الكتلة}؛ أما \omterm{def:g4:weight-and-mass:weight}{الوزن} فقد يتغيّر.
\end{example}

\section{الميزان النابضي}

\begin{definition}[الميزان النابضي]\label{def:g4:weight-and-mass:springscale}
\emph{الميزان النابضي}\index{ميزان نابضي} يقيس \omterm{def:g4:weight-and-mass:weight}{الوزن} بأن يدعه
يمدّ نابضًا: فكلما اشتدّ الجذب طال المدّ. ويبيّن مؤشّر إلى جانب
النابض الممدود القراءةَ. وموازين الصيّادين وموازين الحقائب
وداخل ميزان المطبخ كلها تعمل هكذا --- فهي تحسّ \emph{بالجذب}
لا بالمادة.
\end{definition}

\begin{method}[ميزان من شريط مطّاطي]\label{met:g4:weight-and-mass:rubberband}
ابنِ \omterm{def:g1:five-senses:senses}{حاسّة} \omterm{def:g4:weight-and-mass:weight}{وزن} خاصة بك:
\begin{enumerate}
  \item علّق شريطًا مطّاطيًّا بخطّاف، وألصق بأسفله كوبًا ورقيًّا
        خفيفًا؛
  \item والكوب فارغ، علّم موضع حافّته ولا تقس شيئًا بعد: هذه هي
        علامة الصفر؛
  \item ثم حمّل الكوب --- عشر كرات زجاجية مثلًا --- وقِس
        بمسطرتك كم نزلت الحافة تحت علامة الصفر؛
  \item ثم حمّله عشرين كرة وقِس من جديد.
\end{enumerate}
ضِعف الكرات، وضِعف المدّ تقريبًا: فشريطك المطّاطي يؤكّد القضية
--- مادة أكثر وجذب أشدّ.
\end{method}

\section{رحلة إلى القمر}

\begin{example}[القمر جاذب أضعف]\label{ex:g4:weight-and-mass:moon}
\omterm{def:g2:moon-in-the-sky:moon}{القمر} أصغر من الأرض بكثير، وهو يجذب الأشياء جذبًا أضعف بكثير ---
أضعف نحو ست مرات. وحقيبة رائد الفضاء تحفظ كل \omterm{def:g3:measuring-mass:units}{غرام} من
\emph{كتلتها} في الرحلة، لكن \emph{وزنها} هناك يتقلّص إلى السدس.
وهذا هو سرّ الوثب كله: فسيقان مدرَّبة على مصارعة جذب الأرض تصارع
فجأة جذب \omterm{def:g2:moon-in-the-sky:moon}{قمر} رفيقًا، فتصير كل خطوة وثبة.
\end{example}

\begin{example}[ميزانان على القمر]\label{ex:g4:weight-and-mass:twoscales}
احمل الميزانين إلى \omterm{def:g2:moon-in-the-sky:moon}{القمر} وزِن الحقيبة نفسها. فالميزان
\emph{النابضي} يصدق في وصف الموضع: يقرأ أقلّ بست مرات ---
فالجذب أضعف حقًّا. أما ميزان \emph{الكفّتين} فلا ينتبه إلى شيء:
إذ يجذب \omterm{def:g2:moon-in-the-sky:moon}{القمر} الحقيبة والقطع المرجعية جذبًا ضعيفًا \emph{على
السواء}، فتظلّ الكفّتان متعادلتين، ويُبلّغ \omterm{def:g3:measuring-mass:mass}{الكتلة} نفسها التي
بلّغها على الأرض. فالنوابض تحسّ بالموضع؛ وموازين الكفّتين تقارن
المادة.
\end{example}

\begin{remark}[كلام الناس وكلام الفيزياء]\label{rem:g4:weight-and-mass:words}
في الدكّان يكون قولنا ``تزن البطّيخة كيلوغرامًا'' كلامًا لا ضير
فيه --- فعلى الأرض تسافر \omterm{def:g3:measuring-mass:mass}{الكتلة} \omterm{def:g4:weight-and-mass:weight}{والوزن} معًا في كل مكان، فلا
يلتبس على أحد شيء. وإنما تشقّ الفيزياء الكلمتين لأنها تخطّط
أبعد من الدكّان: لروّاد الفضاء، وللأشياء الساقطة، و--- في سنة
لاحقة --- \omterm{def:g2:measuring-length:measure}{لقياس} القوى بوحدتها الخاصة، وهي \omterm{def:g2:measuring-length:measure}{وحدة} ستُسمّى باسم
العالِم الذي كتب أول مرة قانون جذب الأرض.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:1}
عن أيّ سؤال تجيب \omterm{def:g3:measuring-mass:mass}{الكتلة}؟ وعن أيّ سؤال يجيب \omterm{def:g4:weight-and-mass:weight}{الوزن}؟ وأيّهما \omterm{def:g2:pushes-and-pulls:force}{قوة}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:2}
في أيّ جهة يسحب \omterm{def:g4:weight-and-mass:weight}{الوزن} دائمًا؟ وأيّ جهاز يحسّ \omterm{def:g4:weight-and-mass:weight}{بالوزن} بالمدّ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:3}
قالب واحد يمدّ نابض \omterm{def:g4:weight-and-mass:springscale}{الميزان النابضي} بمقدار عرض إصبعين. فكم
يمدّه قالبان متماثلان؟ وأيّ قضية تقول ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:4}
في ميزان الشريط المطّاطي في
\cref{met:g4:weight-and-mass:rubberband}، لماذا يجب أن تعلّم
موضع الكوب الفارغ قبل تحميله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:5}
يحمل رائد فضاء حقيبة فيها \qty{20}{kg} من المادة إلى \omterm{def:g2:moon-in-the-sky:moon}{القمر}.
أتبقى كتلتها هناك \qty{20}{kg}؟ وهل يصعب حملها صعوبتها على
الأرض؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:6}
لماذا استطاع روّاد الفضاء أن يثبوا عاليًا على \omterm{def:g2:moon-in-the-sky:moon}{القمر}، رغم حقائبهم
الثقيلة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:7}
على \omterm{def:g2:moon-in-the-sky:moon}{القمر}، أيّ ميزان يغيّر تبليغه --- النابضي أم \omterm{def:g3:levers-and-scales:scale}{ميزان
الكفّتين}؟ واشرح ما الذي يقارنه كل واحد منهما أو يحسّ به حقًّا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:8}
صحيح أم لا: ``شيء لا مادة فيه البتة لا \omterm{def:g4:weight-and-mass:weight}{وزن} له في أي مكان.''
دافع عن جوابك بما في
\cref{prop:g4:weight-and-mass:morestuff}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:9}
ميزان الخضريّة المعلّق ينزل بينما تصبّ التفّاح في كفّته. فبأيّ
ابنَي العمّ يحسّ ميزانها؟ ولماذا تظلّ حيلتها تبيع كيلوغرامات
عادلة من التفّاح --- على الأرض؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:weight-and-mass:10}
على وكالة فضاء أن تتحقّق من أن كتل أجزاء مسبار صحيحة تمامًا ---
لكن المسبار المنتهي سيعمل قرب كوكب صغير بعيد جذبه واهن. فيقلق
مهندس: ``كل موازيننا الأرضية ستكون خاطئة هناك!'' طمئنه: أيّ
القياسات يسافر بلا تغيّر، وبأيّ جهاز أُجري؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:weight-and-mass:11}
في عمق الفضاء، بعيدًا عن الأرض \omterm{def:g2:moon-in-the-sky:moon}{والقمر} والشمس جميعًا، تفلت رائدة
فضاء مفتاح ربط: \omterm{def:g1:floating-and-sinking:words}{فيطفو} إلى جانب يدها عديم \omterm{def:g4:weight-and-mass:weight}{الوزن}. أفقد المفتاح
كتلته؟ اذكر قرينتين من نوع الحياة اليومية تستطيع أن تنتبه
إليهما وتبيّنان أن المادة كلها ما زالت هناك. (تلميح: حاول أن
تهزّه، أو أن تقذفه.)
\end{exercise}
