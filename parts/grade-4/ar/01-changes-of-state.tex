\chapter{الانصهار والغليان: تغيّرات الحالة}\label{ch:g4:changes-of-state}

تعرف منذ سنوات أن الدفء يصهر الثلج ويغلي الماء. لكنك تملك الآن
\omterm{def:g3:temperature-thermometers:temperature}{ميزان حرارة} --- ولميزان الحرارة خبر مذهل يبلّغه من داخل دلو الثلج
المنصهر: فبينما \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} الثلج \emph{تأبى} \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} أن تتحرّك.
فلنشاهد تغيّرات الحالة القديمة بعينين جديدتين.

\section{الأسماء على أصولها}

تنال تغيّرات حالة الماء الآن أسماءها الراشدة، وهي صالحة لكل مادة
لا للماء وحده. فمن الحالة الصلبة إلى السائلة:
\emph{الانصهار}\index{انصهار}. ومن السائلة إلى الصلبة:
\emph{التجمّد}\index{تجمّد} (ويُسمّى أيضًا التصلّب). ومن السائلة
إلى الغازية: \emph{الغليان}\index{غليان} حين يحدث بسرعة وبفقاعات
كبيرة، \omterm{def:g2:ice-water-steam:evapcond}{والتبخّر} حين يحدث في هدوء من السطح. ومن الغازية إلى
السائلة: \omterm{def:g2:ice-water-steam:evapcond}{التكاثف}. وكل مادة في العالم --- الماء والشوكولاتة
والحديد والذهب --- تستطيع أن تلعب هذه اللعبة، كل واحدة عند درجات
حرارتها هي.

\begin{definition}[درجة الانصهار]\label{def:g4:changes-of-state:melting-point}
\emph{درجة انصهار}\index{درجة انصهار} مادة هي \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} التي
تنصهر عندها --- وهي أيضًا \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} التي \omterm{def:g2:ice-water-steam:meltfreeze}{يتجمّد} عندها سائلها
راجعًا. وهي للماء \qty{0}{\celsius}. ولكل مادة درجتها الخاصة:
فالشوكولاتة تنصهر قرب \qty{34}{\celsius} (وهي \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة} الفم ---
وليست مصادفة، فاسأل أي صانع شوكولاتة)، وشمع الشموع قرب
\qty{60}{\celsius}، والحديد لا \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} إلا عند \qty{1538}{\celsius}.
\end{definition}

\begin{definition}[درجة الغليان]\label{def:g4:changes-of-state:boiling-point}
\emph{درجة غليان}\index{درجة غليان} مادة هي \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} التي
تغلي عندها. وهي للماء \qty{100}{\celsius}. ولكل مادة درجتها
الخاصة مرة أخرى --- ولهذا يستطيع طاهٍ متأنٍّ أن يُذهب الماء
غليانًا بينما يبقى الزيت في المقلاة نفسها سائلًا هادئًا.
\end{definition}

\begin{example}[قراءة العالم بعددين]\label{ex:g4:changes-of-state:reading}
لماذا يكون لوح الشوكولاتة صلبًا على الرفّ (\qty{20}{\celsius})
وسائلًا في جيبك في نزهة صيفية (\qty{35}{\celsius})؟ لأن درجة
انصهاره، قرب \qty{34}{\celsius}، تقع \emph{بين} العددين. فالمادة
صلبة دون درجة انصهارها، سائلة فوقها --- حتى تبلغ درجة غليانها،
فتغادر غازًا. عددان يحكيان قصة أي مادة كلها عند أي \omterm{def:g3:temperature-thermometers:temperature}{درجة حرارة}.
\end{example}

\section{ميزان الحرارة العنيد}

\begin{method}[كيف تراقب انصهار الثلج على أصوله]\label{met:g4:changes-of-state:watch}
\begin{enumerate}
  \item املأ وعاءً بثلج مجروش أو ثلج متساقط واغرز فيه \omterm{def:g3:temperature-thermometers:temperature}{ميزان
        الحرارة}؛ وانتظر ثم اقرأ: \qty{0}{\celsius}؛
  \item وأدخِل الوعاء إلى المطبخ الدافئ واقرأ \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} كل
        بضع دقائق بينما \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} الثلج؛
  \item وواصِل القراءة حتى بعد اختفاء آخر شظيّة ثلج بوقت طويل.
\end{enumerate}
والخبر: أن \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} يقف عنيدًا عند \qty{0}{\celsius} طوال
\omterm{def:g2:ice-water-steam:meltfreeze}{الانصهار} --- والماء والثلج معًا في الوعاء. ولا تبدأ \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة}
بالصعود إلا حين تنصهر \emph{آخر} قطعة ثلج.
\end{method}

\begin{proposition}[قانون المسطبة]\label{prop:g4:changes-of-state:plateau}
بينما تغيّر مادةٌ حالتَها --- انصهارًا أو غليانًا --- تقف درجة
حرارتها ساكنة عند المعلم، مهما اجتهد الموقد. فالدفء الوارد كله
يُنفَق على تغيّر الحالة نفسه، ولا شيء منه على الصعود. فالصلب
\omterm{def:g3:solids-liquids-gases:liquid}{والسائل} يتعايشان عند \omterm{def:g4:changes-of-state:melting-point}{درجة الانصهار}؛ \omterm{def:g3:solids-liquids-gases:liquid}{والسائل} \omterm{def:g3:solids-liquids-gases:gas}{والغاز} عند \omterm{def:g4:changes-of-state:boiling-point}{درجة
الغليان}.
\end{proposition}

\begin{example}[المسطبة في المطبخ]\label{ex:g4:changes-of-state:kitchen}
ولهذا يطبخ الماء الغالي برفق والماء الغالي بعنف المعكرونةَ في
الزمن نفسه: فكلا القِدرين عند \qty{100}{\celsius} بالضبط ---
والعنيف لا يفعل بدفئه الزائد إلا أن يبدّده بخارًا أسرع. ولهذا
يبقى المشروب مع مكعّبات الثلج عند \qty{0}{\celsius} حتى آخر
شظيّة: فالثلج المنصهر يثبّت \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} عند المعلم.
\end{example}

\begin{remark}[عدد لا تراه العين]\label{rem:g4:changes-of-state:invisible}
لولا \omterm{def:g3:temperature-thermometers:temperature}{ميزان الحرارة} ما حزر أحد المسطبة: فالقِدر يبدو أشدّ نشاطًا
فأشدّ، ودلو الثلج يبدو أهدأ فأهدأ، \omterm{def:g3:temperature-thermometers:temperature}{ودرجة الحرارة} في كليهما واقفة
كالصخر. وهذا انتصار آخر \omterm{def:g2:measuring-length:measure}{للقياس} على الظاهر --- وأحد أوائل
المفاجآت العظيمة التي سلّمتها الأجهزة إلى العلم.
\end{remark}

\section{كل مادة تحفظ معالمها}

\begin{proposition}[المعالم علامات هوية]\label{prop:g4:changes-of-state:identity}
\omterm{def:g4:changes-of-state:melting-point}{درجة انصهار} مادة نقية ودرجة غليانها لا تتغيّران أبدًا: فالماء
\omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} عند \qty{0}{\celsius} اليوم وغدًا، في قصر أو في خيمة.
والمعالم جزء من هوية المادة --- محفوظة حفظًا يمكّن عالمًا من
التعرّف على مادة مجهولة \omterm{def:g2:measuring-length:measure}{بقياس} أين تنصهر وأين تغلي.
\end{proposition}

\begin{example}[الشمعة والمسبك]\label{ex:g4:changes-of-state:foundry}
شمعة إلى جانب المدفأة تنهار وتقطر: فقد تجاوز الشمع درجة انصهاره
قرب \qty{60}{\celsius}. وفي مسبك يسكب العمّال حديدًا سائلًا برتقالي
التوهّج مثل الحساء --- وقد تجاوز فرنُهم \qty{1538}{\celsius}.
اللعبة نفسها، ومعالم شديدة الاختلاف: فما تفعله المدفأة بالشمع لا
يفعله بالحديد إلا فرن. والذهب \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} عند \qty{1064}{\celsius}:
وقد عرف الصاغة هذا العدد، في أيديهم إن لم يكن بالدرجات، منذ
آلاف السنين.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:1}
أعطِ الاسم على أصوله: من الصلبة إلى السائلة؛ \ من السائلة إلى
الصلبة؛ \ من السائلة إلى الغازية بفقاعات كبيرة؛ \ من الغازية إلى
السائلة على نافذة باردة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:2}
ما معلما الماء، \omterm{def:g3:temperature-thermometers:temperature}{ودرجة حرارة} كل منهما؟ وما \omterm{def:g4:changes-of-state:melting-point}{درجة انصهار} الشوكولاتة
تقريبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:3}
وعاء فيه ثلج وماء معًا، وقد حُرّك جيدًا. فماذا يقرأ \omterm{def:g3:temperature-thermometers:temperature}{ميزان
الحرارة}؟ وماذا سيقرأ بعد خمس دقائق إن بقي بعض الثلج؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:4}
اذكر قانون المسطبة. وإلى أين يذهب دفء الموقد بينما يغلي الماء،
ما دامت \omterm{def:g3:temperature-thermometers:temperature}{درجة الحرارة} لم تعد تصعد؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:5}
الشمع \omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} قرب \qty{60}{\celsius}. أالشمعة صلبة أم سائلة: في
غرفة جلوس عند \qty{20}{\celsius}؟ \ إلى جانب لهب يدفئها إلى
\qty{80}{\celsius}؟ \ في عليّة صيفية عند \qty{35}{\celsius}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:6}
تُطبخ المعكرونة في ماء يغلي برفق. فيرفع بابا اللهب إلى أقصاه
فيغلي القِدر بعنف. أصار الماء أسخن الآن؟ وماذا ينتج الدفء الزائد
بدل ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:7}
في بيان المسطبة في الفصل، كيف تكتشف تغيّرَي الحالة من غير أن
تقرأ كلمة واحدة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:8}
\omterm{def:g2:ice-water-steam:meltfreeze}{ينصهر} الحديد عند \qty{1538}{\celsius} والذهب عند
\qty{1064}{\celsius}. وفرن يعمل عند \qty{1200}{\celsius}: فأيّ
المعدنين يُسكب فيه مثل الحساء، وأيّهما يبقى صلبًا؟ وبكم درجة
يقصّر الفرن عن الآخر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:9}
لماذا يبقى المشروب مع مكعّبات الثلج بارد الجودة نفسها حتى آخر
شظيّة ثلج --- ولا يبدأ بالدفء إلا حينئذ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:changes-of-state:10}
تتلقّى عالِمة مادة صلبة بيضاء مجهولة. فتجد أنها تنصهر عند
\qty{0}{\celsius} بالضبط وأن سائلها يغلي عند \qty{100}{\celsius}
بالضبط. فما أرجح ظنّها في المادة --- وأيّ قانون يجعل المعالم
تصلح بطاقات هوية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:changes-of-state:11}
يبلّغ متسلّقو الجبال أن الماء عاليًا على القمم العظيمة يغلي وهو
\emph{بالكاد} أسخن من أن يُلمس --- وأن شايهم لا يُنقَع كما ينبغي
أبدًا. فأيّ يقين يومي في حقّ الماء الغالي ينهار هناك؟ (والسبب
مختبئ في \omterm{def:g2:air-around-us:air}{الهواء} نفسه، وسيمسكه فصل لاحق عن ضغط \omterm{def:g2:air-around-us:air}{الهواء}.)
\end{exercise}
