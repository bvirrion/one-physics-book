\chapter{Les aimants et la boussole}\label{ch:g4:magnets-and-compass}

Il y a des années, tu as découvert que les deux extrémités d'un
\omterm{def:g1:magnets:magnet}{aimant} sont différentes~: dans un sens, deux \omterm{def:g1:magnets:magnet}{aimants} se collent d'un
coup~; retournés, ils se repoussent. Nous avions promis que ces
extrémités avaient un nom et un métier célèbre --- guider les
voyageurs à travers le monde. Aujourd'hui, la promesse est tenue~:
voici les \omterm{def:g4:magnets-and-compass:poles}{pôles}, et le petit instrument qui a trouvé le chemin de
tous les explorateurs avant l'existence des satellites.

\section{Les deux pôles}

\begin{definition}[Pôles]\label{def:g4:magnets-and-compass:poles}
Les deux extrémités particulières d'un \omterm{def:g1:magnets:magnet}{aimant} --- là où son
attraction est la plus forte, là où les trombones s'agglutinent ---
sont ses \emph{pôles}\index{pôle}. On les appelle le \emph{pôle
nord}\index{pôle nord} (souvent peint en rouge, marqué N) et le
\emph{pôle sud}\index{pôle sud} (marqué S). Tout \omterm{def:g1:magnets:magnet}{aimant} possède les
deux~: un de chaque, toujours.
\end{definition}

\begin{proposition}[La règle des pôles]\label{prop:g4:magnets-and-compass:rule}
Des \omterm{def:g4:magnets-and-compass:poles}{pôles} opposés s'attirent~: N se colle à S. Des \omterm{def:g4:magnets-and-compass:poles}{pôles} de même nom
se repoussent~: N repousse N, S repousse S. Cette unique règle
explique tous les claquements et tous les refus élastiques que tu as
jamais sentis entre deux \omterm{def:g1:magnets:magnet}{aimants}.
\end{proposition}

\begin{example}[Relire les vieilles
expériences]\label{ex:g4:magnets-and-compass:reading}
Quand tes deux \omterm{def:g1:magnets:magnet}{aimants} se collaient d'un coup, un N faisait face à un
S. Quand ils refusaient élastiquement, N faisait face à N (ou S à S).
Et les trombones s'agglutinaient toujours aux extrémités parce que
les \omterm{def:g4:magnets-and-compass:poles}{pôles} sont l'endroit où vit la puissance d'un \omterm{def:g1:magnets:magnet}{aimant} --- son
milieu est le calme qui les sépare.
\end{example}

\section{La Terre est un aimant}

\begin{example}[Le plus grand aimant de tous]\label{ex:g4:magnets-and-compass:earth}
Suspends un \omterm{def:g1:magnets:magnet}{aimant} droit à un fil de façon qu'il puisse tourner
librement, attends qu'il se stabilise --- et il se produit quelque
chose de merveilleux~: il s'immobilise toujours dans la \emph{même}
direction, un \omterm{def:g4:magnets-and-compass:poles}{pôle} vers le grand nord du monde, l'autre vers le grand
sud. Un \omterm{def:g1:magnets:magnet}{aimant} géant, invisible, doit bien aligner ses \omterm{def:g4:magnets-and-compass:poles}{pôles}. Ce
géant, c'est la Terre elle-même~: notre planète tout entière est un
\omterm{def:g1:magnets:magnet}{aimant}, avec ses \omterm{def:g4:magnets-and-compass:poles}{pôles} magnétiques près du haut et du bas du globe.
\end{example}

\begin{definition}[Boussole]\label{def:g4:magnets-and-compass:compass}
Une \emph{boussole}\index{boussole} est une petite aiguille aimantée
et légère, posée en équilibre sur un \omterm{def:g3:levers-and-scales:lever}{pivot} fin de façon à pouvoir
tourner librement. Laissée tranquille, l'aiguille obéit à
l'aimant-Terre~: son extrémité marquée s'immobilise en direction du
nord. Sous l'aiguille, un cadran indique les quatre \emph{points
cardinaux}\index{points cardinaux}~: Nord, Est, Sud, Ouest.
\end{definition}

\begin{method}[Se servir d'une boussole]\label{met:g4:magnets-and-compass:use}
\begin{enumerate}
  \item Tenir la \omterm{def:g4:magnets-and-compass:compass}{boussole} bien à plat et immobile, loin des \omterm{def:g1:magnets:magnet}{aimants}
        et des grosses \omterm{def:g3:measuring-mass:mass}{masses} de fer (grillages, voitures --- elles
        distraient l'aiguille)~;
  \item attendre que l'aiguille cesse d'osciller~;
  \item tourner doucement le cadran de la \omterm{def:g4:magnets-and-compass:compass}{boussole} jusqu'à ce que
        son N se place sous l'extrémité marquée de l'aiguille~;
  \item lire les directions~: l'est à ta droite quand tu fais face
        au nord, l'ouest à ta gauche, le sud dans ton dos.
\end{enumerate}
Vérifie avec le ciel~: par chez nous, l'\omterm{def:g1:light-and-shadows:shadow}{ombre} de midi du chapitre sur
le \omterm{def:g3:sun-days-seasons:sundial}{cadran solaire} pointe vers le nord --- l'\omterm{def:g1:light-and-shadows:shadow}{ombre} et l'aiguille
doivent être d'accord.
\end{method}

\begin{example}[En fabriquer une soi-même]\label{ex:g4:magnets-and-compass:make}
Frotte une aiguille à coudre en acier avec un \omterm{def:g4:magnets-and-compass:poles}{pôle} d'un \omterm{def:g1:magnets:magnet}{aimant}, vingt
fois, toujours dans le même sens, en écartant l'\omterm{def:g1:magnets:magnet}{aimant} entre chaque
passage~: l'aiguille devient elle-même un petit \omterm{def:g1:magnets:magnet}{aimant}. Pose-la
doucement sur une rondelle de liège qui \omterm{def:g1:floating-and-sinking:words}{flotte} dans un bol d'eau ---
et regarde le liège tourner lentement jusqu'à ce que l'aiguille
pointe nord--sud. Tu viens de construire la \omterm{def:g4:magnets-and-compass:compass}{boussole} des marins d'il
y a mille ans.
\end{example}

\begin{remark}[L'instrument qui a ouvert les
océans]\label{rem:g4:magnets-and-compass:history}
Avant la \omterm{def:g4:magnets-and-compass:compass}{boussole}, les marins longeaient les côtes et priaient pour
des nuits claires afin de se guider aux étoiles. Avec une aiguille
flottante, ils pouvaient tenir un cap dans le brouillard, la tempête
et les nuits sans lune --- à travers des océans entiers. Peu
d'instruments ont redessiné la carte du monde aussi complètement. Et
une énigme est restée ouverte pendant des siècles~: \emph{pourquoi}
la Terre se comporte-t-elle comme un \omterm{def:g1:magnets:magnet}{aimant}~? L'histoire complète
attend dans les années de licence de ce cours --- elle vaut
l'attente.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:1}
Que sont les \omterm{def:g4:magnets-and-compass:poles}{pôles} d'un \omterm{def:g1:magnets:magnet}{aimant}, et comment s'appellent-ils~? À quel
endroit d'un \omterm{def:g1:magnets:magnet}{aimant} droit les trombones s'agglutinent-ils~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:2}
Énonce la règle des \omterm{def:g4:magnets-and-compass:poles}{pôles}. Deux \omterm{def:g1:magnets:magnet}{aimants} se collent d'un coup~: quels
\omterm{def:g4:magnets-and-compass:poles}{pôles} se faisaient face~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:3}
Le \omterm{def:g4:magnets-and-compass:poles}{pôle} N d'un \omterm{def:g1:magnets:magnet}{aimant} s'approche du \omterm{def:g4:magnets-and-compass:poles}{pôle} N d'un autre. Que se
passe-t-il~? Et N vers S~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:4}
Pourquoi un \omterm{def:g1:magnets:magnet}{aimant} droit suspendu librement s'immobilise-t-il
toujours dans la même direction~? Que cela révèle-t-il sur la Terre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:5}
Nomme les quatre \omterm{def:g4:magnets-and-compass:compass}{points cardinaux} dans l'ordre, en partant du nord et
en tournant vers la droite. Face au nord, quelle direction est à ta
droite~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:6}
Pourquoi faut-il tenir une \omterm{def:g4:magnets-and-compass:compass}{boussole} loin des \omterm{def:g1:magnets:magnet}{aimants}, des grillages
en fer et des voitures pendant qu'on la lit~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:7}
Décris comment transformer une aiguille à coudre en aiguille de
\omterm{def:g4:magnets-and-compass:compass}{boussole}, et comment la laisser tourner librement sans \omterm{def:g3:levers-and-scales:lever}{pivot}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:8}
D'après le chapitre sur le \omterm{def:g3:sun-days-seasons:sundial}{cadran solaire}~: dans quelle direction
pointe l'\omterm{def:g1:light-and-shadows:shadow}{ombre} de midi par chez nous~? Comment cela permet-il de
vérifier une \omterm{def:g4:magnets-and-compass:compass}{boussole} --- ou d'en remplacer une perdue~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:9}
Pourquoi la \omterm{def:g4:magnets-and-compass:compass}{boussole} a-t-elle été un tel trésor pour les marins,
comparée au guidage le long de la côte ou aux étoiles~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:10}
Une randonneuse marche vers un refuge de montagne situé plein nord.
Son aiguille de \omterm{def:g4:magnets-and-compass:compass}{boussole}, tenue correctement, pointe vers la boucle
de ceinture de son ami au lieu du nord chaque fois qu'il se tient
tout près, sur sa gauche. Que se passe-t-il, dans quelle direction
risque-t-elle de dériver, et quelle est la solution en une étape~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:11}
On scie un \omterm{def:g1:magnets:magnet}{aimant} en deux entre ses deux \omterm{def:g4:magnets-and-compass:poles}{pôles}, dans l'espoir
d'obtenir un N tout seul dans une main et un S tout seul dans
l'autre. Le savoir des marins --- et toutes les expériences jamais
faites --- disent que chaque moitié se révèle être un petit \omterm{def:g1:magnets:magnet}{aimant}
complet, avec \emph{les deux} \omterm{def:g4:magnets-and-compass:poles}{pôles}. Mets l'affirmation à l'épreuve
par la pensée avec la recette de l'aiguille de \omterm{def:g4:magnets-and-compass:compass}{boussole}~: l'aiguille
à coudre frottée est courte --- a-t-elle un \omterm{def:g4:magnets-and-compass:poles}{pôle} ou deux~? Que cela
suggère-t-il sur la chasse à un \omterm{def:g4:magnets-and-compass:poles}{pôle} isolé par découpage~?
\end{exercise}
