\chapter{Les circuits en série}\label{ch:g4:series-circuits}

L'an dernier, une guirlande de fête s'est éteinte parce qu'une seule
lampe manquait --- et nous avons promis que cette disposition avait
un nom. Le voici~: le \omterm{def:g4:series-circuits:series}{circuit en série}, la façon la plus simple de
placer plusieurs joueurs sur une même boucle. Il a un règlement
strict, et la lampe de poche de ton tiroir lui obéit depuis toujours.

\section{Une boucle, plusieurs joueurs}

\begin{definition}[En série]\label{def:g4:series-circuits:series}
Des joueurs d'un circuit sont \emph{en série}\index{circuit en série}
quand ils se trouvent sur une seule boucle, les uns après les autres,
comme des perles sur un même collier~: la route du courant traverse
le premier, puis le second, puis le troisième, sans jamais se
séparer en deux. Un circuit construit ainsi est un \emph{circuit en
série}.
\end{definition}

\begin{method}[Le construire]\label{met:g4:series-circuits:build}
Avec une \omterm{def:g3:first-electric-circuit:battery}{pile}, deux lampes, un \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} et des fils~:
\begin{enumerate}
  \item partir de la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} avec un fil vers
        l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur}~;
  \item un fil de l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} à la première lampe, puis de la
        première lampe à la seconde~;
  \item fermer la boucle~: de la seconde lampe à la \omterm{def:g3:first-electric-circuit:battery}{borne} $-$ de la
        \omterm{def:g3:first-electric-circuit:battery}{pile}~;
  \item actionner l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} et observer les deux lampes
        ensemble.
\end{enumerate}
Suis la boucle du doigt avant d'allumer~: si ton doigt doit un jour
choisir entre deux routes, ce n'est pas un \omterm{def:g4:series-circuits:series}{circuit en série}.
\end{method}

\section{Le règlement}

\begin{proposition}[Les règles de la boucle unique]\label{prop:g4:series-circuits:rules}
Dans un \omterm{def:g4:series-circuits:series}{circuit en série}~:
\begin{enumerate}
  \item \emph{tout ou rien}~: une seule coupure quelque part --- un
        \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} ouvert, une lampe grillée, un fil mal serré ---
        et tous les joueurs s'arrêtent d'un coup~;
  \item \emph{l'ordre est sans importance}~: échange les lampes,
        déplace l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} de l'autre côté de la boucle --- tout
        se comporte exactement comme avant~;
  \item \emph{le partage}~: plus il y a de lampes sur la boucle,
        plus chacune brille faiblement~; la poussée de la \omterm{def:g3:first-electric-circuit:battery}{pile} se
        partage sur tout le collier.
\end{enumerate}
\end{proposition}

\begin{example}[Voir les règles à l'œuvre]\label{ex:g4:series-circuits:seeing}
Construis le circuit à deux lampes et éprouve chaque règle. Dévisse
\emph{l'une ou l'autre} lampe~: les deux s'éteignent --- règle
première (la douille vide est une coupure). Échange les deux lampes,
ou place l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} entre elles~: aucun changement --- règle
deuxième. Reconstruis maintenant avec une seule lampe~: elle brille
nettement plus fort, seule, qu'avec sa partenaire --- règle
troisième, à l'envers.
\end{example}

\begin{example}[La guirlande de fête, élucidée]\label{ex:g4:series-circuits:partylights}
Le vieux mystère n'en est plus un~: les lampes de la guirlande sont
\omterm{def:g4:series-circuits:series}{en série}, des dizaines de perles sur un même collier. Une lampe
manquante est une coupure, et la règle première éteint toute la
guirlande. Chercher la lampe coupable en les essayant une par une a
appris la règle première, à la dure, à des générations de familles
--- les guirlandes modernes sont construites autrement, comme le
montrera l'an prochain.
\end{example}

\begin{remark}[L'interrupteur commande tout le
monde]\label{rem:g4:series-circuits:switch}
La règle deuxième a une conséquence bien pratique~: un seul
\omterm{ex:g3:first-electric-circuit:switch}{interrupteur}, placé n'importe où sur la boucle, commande tous les
joueurs qui s'y trouvent. Voilà pourquoi un unique \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} mural
peut commander toute une rangée de plafonniers --- et pourquoi, pour
la même raison, tu ne peux pas éteindre \emph{une seule} lampe d'une
guirlande \omterm{def:g4:series-circuits:series}{en série} sans les tuer toutes.
\end{remark}

\section{Plusieurs piles sur le collier}

\begin{example}[Tête à queue]\label{ex:g4:series-circuits:batteries}
Les \omterm{def:g3:first-electric-circuit:battery}{piles} aussi peuvent faire la queue \omterm{def:g4:series-circuits:series}{en série} --- mais elles sont
plus difficiles que les lampes~: elles doivent se placer \emph{tête à
queue}, le $+$ de chaque \omterm{def:g3:first-electric-circuit:battery}{pile} touchant le $-$ de la suivante. Deux
\omterm{def:g3:first-electric-circuit:battery}{piles} ainsi alignées poussent ensemble, et la lampe brille nettement
plus fort qu'avec une seule. Retourne une \omterm{def:g3:first-electric-circuit:battery}{pile} --- tête contre
tête --- et les deux poussées se combattent~: la lampe ne donne rien.
\end{example}

    to[battery1, l=deux piles] (0,3.6)

      to[short] (0,3.6)
      to[short] (3.6,3.6)
      to[lamp, l=moins brillante] (3.6,0)
      to[short] (0,0);
  \end{scope}
\end{tikzpicture}

{\small Deux \omterm{def:g3:first-electric-circuit:battery}{piles} alignées tête à queue poussent ensemble~: la même
lampe brille plus fort qu'avec une seule \omterm{def:g3:first-electric-circuit:battery}{pile}.}
\end{omfigure}

\begin{example}[Dans la lampe de poche]\label{ex:g4:series-circuits:flashlight}
Ouvre une lampe de poche~: les \omterm{def:g3:first-electric-circuit:battery}{piles} se glissent l'une derrière
l'autre dans un tube --- une file tête à queue toute prête --- puis
la lampe et le \omterm{ex:g3:first-electric-circuit:switch}{bouton-interrupteur} complètent une seule boucle à
travers le boîtier. Une lampe de poche \emph{est} un \omterm{def:g4:series-circuits:series}{circuit en série}
dans une poche. Et tu sais maintenant pourquoi elle reste éteinte si
une seule \omterm{def:g3:first-electric-circuit:battery}{pile} est mise à l'envers~: tête contre tête, les poussées
s'annulent.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:1}
Que veut dire «~\omterm{def:g4:series-circuits:series}{en série}~»~? Quelle image de la vie de tous les jours
le chapitre utilise-t-il pour cela~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:2}
Énonce les trois règles du règlement du \omterm{def:g4:series-circuits:series}{circuit en série}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:3}
Dans le circuit à deux lampes, la lampe 2 est dévissée. Qu'arrive-t-il
à la lampe 1, et pourquoi~? Quelle règle est à l'œuvre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:4}
Zoé déplace l'\omterm{ex:g3:first-electric-circuit:switch}{interrupteur} d'avant la lampe 1 à après la lampe 2. Que
change-t-il dans le comportement du circuit~? Quelle règle le dit~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:5}
Une \omterm{def:g3:first-electric-circuit:battery}{pile} allume joliment une lampe. Prédis l'éclat avec~: deux lampes
\omterm{def:g4:series-circuits:series}{en série}~; \ trois lampes \omterm{def:g4:series-circuits:series}{en série}. Qu'est-ce qui se partage~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:6}
Comment deux \omterm{def:g3:first-electric-circuit:battery}{piles} doivent-elles être placées pour pousser ensemble~?
Dessine la file avec les $+$ et les $-$ marqués sur chaque \omterm{def:g3:first-electric-circuit:battery}{pile}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:7}
Suis la boucle d'une lampe de poche~: nomme les joueurs que le
courant visite depuis la \omterm{def:g3:first-electric-circuit:battery}{borne} $+$ de la \omterm{def:g3:first-electric-circuit:battery}{pile} jusqu'à son $-$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:8}
Le circuit d'une sonnette comporte une \omterm{def:g3:first-electric-circuit:battery}{pile}, un bouton et la sonnerie
\omterm{def:g4:series-circuits:series}{en série}. Où pourrais-tu ajouter un second bouton pour que la
sonnerie ne retentisse que si l'on appuie sur les \emph{deux}
boutons~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:9}
Une guirlande de dix lampes \omterm{def:g4:series-circuits:series}{en série} est éteinte. Les lampes ont
toutes l'\omterm{def:g2:air-around-us:air}{air} en bon état. Décris un plan patient, avec une lampe dont
on sait qu'elle marche, pour trouver celle qui est grillée.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:series-circuits:10}
La lampe de poche de Tom prend deux \omterm{def:g3:first-electric-circuit:battery}{piles} et brille fort. Après avoir
changé les \omterm{def:g3:first-electric-circuit:battery}{piles} dans le noir, elle ne donne plus rien --- alors que
les deux \omterm{def:g3:first-electric-circuit:battery}{piles} neuves sont pleines, la lampe est en bon état et aucun
fil n'est desserré. Qu'a fait Tom, et pourquoi la lampe de poche
reste-t-elle éteinte~? (Voir \cref{ex:g4:series-circuits:batteries}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:series-circuits:11}
Une ingénieure doit placer un unique \omterm{ex:g3:first-electric-circuit:switch}{interrupteur} d'arrêt d'urgence
pour une machine dont le moteur se trouve sur une seule boucle avec
sa \omterm{def:g3:first-electric-circuit:battery}{pile}. Elle dit~: «~N'importe où sur la boucle fera l'affaire.~»
A-t-elle raison~? Et pourquoi son plan échouerait-il si le circuit de
la machine comportait des routes séparées~? (Le chapitre de l'an
prochain a un nom pour celles-là.)
\end{exercise}
