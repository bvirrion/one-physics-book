\chapter{Geleiders en isolatoren}\label{ch:g4:conductors-insulators}

Waarom zijn elektriciteitsdraden van metaal gemaakt --- maar in
plastic gekleed? Waarom geleidt het bruggetje van de \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} de
stroom, terwijl zijn omhulsel je vingers veilig houdt? Vorig jaar
sloten we lussen; dit jaar vragen we door welke materialen
elektriciteit überhaupt kan reizen. Het antwoord sorteert elk
materiaal in twee soorten: \omterm{def:g4:conductors-insulators:conductor}{geleiders} en \omterm{def:g4:conductors-insulators:insulator}{isolatoren}.

\section{De tester}

\begin{method}[Een tester bouwen]\label{met:g4:conductors-insulators:tester}
Bouw de lus van vorig jaar --- \omterm{def:g3:first-electric-circuit:battery}{batterij}, draadjes, \omterm{def:g3:first-electric-circuit:bulb}{lampje} --- maar
laat er met opzet één onderbreking in: twee vrije draadeinden, een
vingerbreedte uit elkaar.
\begin{enumerate}
  \item raak de twee vrije einden aan elkaar: het \omterm{def:g3:first-electric-circuit:bulb}{lampje} brandt ---
        de tester werkt;
  \item overbrug de onderbreking met het voorwerp dat je onderzoekt,
        met een metalen uiteinde stevig tegen elk draadje;
  \item kijk naar het \omterm{def:g3:first-electric-circuit:bulb}{lampje}: \omterm{def:g1:light-and-shadows:source}{licht} betekent dat de elektriciteit
        het voorwerp is overgestoken; donker betekent dat het niet
        lukte.
\end{enumerate}
Het \omterm{def:g3:first-electric-circuit:bulb}{lampje} geeft het antwoord: het brandt precies dan als het
onderzochte voorwerp de lus sluit.
\end{method}

\section{Geleiders en isolatoren}

\begin{definition}[Geleider]\label{def:g4:conductors-insulators:conductor}
Een \emph{geleider}\index{geleider} is een materiaal dat
elektriciteit erdoorheen laat reizen. In de tester sluit een geleider
die de onderbreking overbrugt de lus, en brandt het \omterm{def:g3:first-electric-circuit:bulb}{lampje}.
\end{definition}

\begin{definition}[Isolator]\label{def:g4:conductors-insulators:insulator}
Een \emph{isolator}\index{isolator} is een materiaal dat
elektriciteit tegenhoudt. De onderbreking met een isolator overbruggen
laat de lus net zo open als daarvoor: het \omterm{def:g3:first-electric-circuit:bulb}{lampje} blijft donker.
\end{definition}

\begin{example}[Een ochtend proeven]\label{ex:g4:conductors-insulators:trials}
Uitslagen van een tester in de klas: een stalen paperclip ---
\omterm{def:g1:light-and-shadows:source}{licht}. Een muntje --- \omterm{def:g1:light-and-shadows:source}{licht}. Aluminiumfolie --- \omterm{def:g1:light-and-shadows:source}{licht}. Een sleutel
--- \omterm{def:g1:light-and-shadows:source}{licht}. Een plastic liniaal --- donker. Een houten potlood ---
donker. Een glazen knikker --- donker. Een rubberen gum --- donker.
Een strookje papier --- donker. Het patroon springt eruit: de
voorwerpen die het \omterm{def:g3:first-electric-circuit:bulb}{lampje} laten branden zijn allemaal van metaal.
\end{example}

\begin{proposition}[Metalen geleiden]\label{prop:g4:conductors-insulators:metals}
Alle metalen geleiden elektriciteit --- ijzer, koper, aluminium, goud,
stuk voor stuk, en daarom verraste de kieskeurigheid van de
\omterm{def:g1:magnets:magnet}{magneet} over ijzer ons wel, maar zal de tester dat nooit doen. De
meeste andere alledaagse vaste stoffen --- plastic, hout, glas,
rubber, papier, was --- zijn \omterm{def:g4:conductors-insulators:insulator}{isolatoren}.
\end{proposition}

\begin{remark}[Geleider is niet magnetisch]\label{rem:g4:conductors-insulators:notmagnet}
Verwar de twee talenten van metaal niet. De \omterm{def:g1:magnets:magnet}{magneet} kiest alleen ijzer
en zijn familie; de elektrische stroom is veel minder kieskeurig en
reist door \emph{elk} metaal. Het aluminiumfolie dat je \omterm{def:g1:magnets:magnet}{magneet}
negeerde, draagt de stroom prachtig --- twee verschillende spellen,
twee verschillende regels.
\end{remark}

\section{Geleiders en isolatoren werken samen}

\begin{example}[Anatomie van een draad]\label{ex:g4:conductors-insulators:wire}
Kijk naar het uiteinde van de kabel van een lamp waar hij de stekker
in gaat: binnenin loopt glanzend koper --- een \omterm{def:g4:conductors-insulators:conductor}{geleider}, de weg van de
stroom; eromheen zit gekleurd plastic gewikkeld --- een
\omterm{def:g4:conductors-insulators:insulator}{isolator}, het hek langs die weg. Het koper doet het werk; het plastic
houdt de stroom binnen en jouw vingers buiten. Elke kabel in je huis
combineert de twee: een \omterm{def:g4:conductors-insulators:conductor}{geleider} in een \omterm{def:g4:conductors-insulators:insulator}{isolator}.
\end{example}

\begin{example}[Alledaagse voorwerpen lezen]\label{ex:g4:conductors-insulators:objects}
Nu verklaren veel ontwerpen zichzelf. De \omterm{ex:g3:first-electric-circuit:switch}{schakelaar}: een metalen
bruggetje (dat moet stroom dragen) in een plastic behuizing (jouw
vinger niet). De schroevendraaier van de elektricien: stalen blad,
dikke isolerende greep. Het stopcontact: metaal diep binnenin, plastic
aan de buitenkant. Hoogspanningsmasten dragen blote metalen kabels
hoog buiten bereik, hangend aan lange isolerende schijven zodat de
stroom niet in de mast kan weglekken.
\end{example}

\begin{remark}[Water, mensen, en waarom er regels zijn]\label{rem:g4:conductors-insulators:safety}
Er moet nog één \omterm{def:g4:conductors-insulators:conductor}{geleider} genoemd worden: \emph{vochtige dingen} ---
waaronder vochtige mensenhuid. Wij geleiden slecht, maar het
stopcontact duwt zo machtig dat slecht ruimschoots genoeg is. Dat is de
diepe reden achter de veiligheidsregels die je kent: prik nooit in
stopcontacten, raak nooit iets elektrisch aan met natte handen, en
houd föhns ver van het bad. De batterijtester is veilig \emph{doordat}
de duw van een \omterm{def:g3:first-electric-circuit:battery}{batterij} minuscuul is; het stopcontact vergeeft niets.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:1}
Wat is een \omterm{def:g4:conductors-insulators:conductor}{geleider}? En een \omterm{def:g4:conductors-insulators:insulator}{isolator}? Welke van de twee laat het
\omterm{def:g3:first-electric-circuit:bulb}{lampje} van de tester branden?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:2}
Sorteer in \omterm{def:g4:conductors-insulators:conductor}{geleiders} en \omterm{def:g4:conductors-insulators:insulator}{isolatoren}: een stalen lepel; \ een kurk; \
aluminiumfolie; \ een glas; \ een koperen muntje; \ een plastic
bouwsteentje.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:3}
Hoe controleer je vóór elke proef of de tester zelf werkt? Waarom is
die controle belangrijk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:4}
Een onderzocht voorwerp laat het \omterm{def:g3:first-electric-circuit:bulb}{lampje} donker. Geef twee mogelijke
redenen --- één over het voorwerp, één over slordig testen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:5}
Waarom is een lampkabel vanbinnen koper en vanbuiten plastic? Welke
taak heeft elk materiaal?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:6}
Welk deel van een \omterm{ex:g3:first-electric-circuit:switch}{schakelaar} is een \omterm{def:g4:conductors-insulators:conductor}{geleider} en welk een
\omterm{def:g4:conductors-insulators:insulator}{isolator}? Waarom heeft hij allebei nodig?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:7}
Aluminiumfolie negeert een \omterm{def:g1:magnets:magnet}{magneet} volkomen. Laat het het \omterm{def:g3:first-electric-circuit:bulb}{lampje} van
de tester branden? Welke regel geeft het antwoord, en voor welke
verwarring waarschuwt \cref{rem:g4:conductors-insulators:notmagnet}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:8}
Een potlood is hout met een grafietstift. De tester van Nino blijft
donker over het gelakte hout, maar brandt zwakjes van punt tot punt
door het geslepen grafiet. Wat heeft Nino over de twee materialen van
het potlood ontdekt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:conductors-insulators:9}
Waarom hebben schroevendraaiers en tangen van elektriciens dikke
plastic grepen? Welk materiaal beschermt wie?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:conductors-insulators:10}
Een halsketting van schakels geeft ``\omterm{def:g1:light-and-shadows:source}{licht}'' in de tester; een ketting
van houten kralen geeft ``donker''; een ketting van metalen kralen die
op een dik katoenen koord geknoopt zitten geeft ook ``donker'' ---
terwijl elke kraal geleidt! Verklaar het derde resultaat door de weg
van de stroom van kraal tot kraal te volgen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:conductors-insulators:11}
Waarom is het veel gevaarlijker om elektrische dingen met natte handen
aan te raken dan met droge? En waarom is de batterijtester veilig voor
jouw proeven terwijl het stopcontact dat nooit is? Gebruik
\cref{rem:g4:conductors-insulators:safety}.
\end{exercise}
