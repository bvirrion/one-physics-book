\chapter{Smelten en koken: veranderingen van toestand}\label{ch:g4:changes-of-state}

Je weet al jaren dat warmte ijs \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} en water aan de kook brengt. Maar
nu heb je een \omterm{def:g3:temperature-thermometers:temperature}{thermometer} --- en de \omterm{def:g3:temperature-thermometers:temperature}{thermometer} heeft iets
verbazingwekkends te melden vanuit de emmer met smeltend ijs: terwijl
het ijs \omterm{def:g2:ice-water-steam:meltfreeze}{smelt}, \emph{weigert} de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} te bewegen. Laten we de
oude toestandsveranderingen met nieuwe ogen bekijken.

\section{De officiële namen}

De toestandsveranderingen van water krijgen nu hun volwassen namen, die
voor elke stof gelden en niet alleen voor water. Van vast naar
vloeibaar: \emph{smelten}\index{smelten}. Van vloeibaar terug naar
vast: \emph{stollen}\index{stollen} (bij water ook \omterm{def:g2:ice-water-steam:meltfreeze}{bevriezen}
genoemd). Van vloeibaar naar \omterm{def:g3:solids-liquids-gases:gas}{gas}: \emph{koken}\index{koken} als het
snel gaat met grote bellen, \omterm{def:g2:ice-water-steam:evapcond}{verdampen} als het rustig vanaf het
oppervlak gebeurt. Van \omterm{def:g3:solids-liquids-gases:gas}{gas} terug naar vloeibaar: \omterm{def:g2:ice-water-steam:evapcond}{condenseren}. Elke
stof ter wereld --- water, chocola, ijzer, goud --- kan dit spel
spelen, elk bij zijn eigen temperaturen.

\begin{definition}[Smeltpunt]\label{def:g4:changes-of-state:melting-point}
Het \emph{smeltpunt}\index{smeltpunt} van een stof is de
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur} waarbij ze \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} --- en tevens de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} waarbij de
\omterm{def:g3:solids-liquids-gases:liquid}{vloeistof} weer stolt. Voor water is dat \qty{0}{\celsius}. Elke stof
heeft haar eigen smeltpunt: chocola \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} rond \qty{34}{\celsius}
(mondtemperatuur --- geen toeval, vraag het maar aan een chocolatier),
kaarsvet rond \qty{60}{\celsius}, ijzer pas bij \qty{1538}{\celsius}.
\end{definition}

\begin{definition}[Kookpunt]\label{def:g4:changes-of-state:boiling-point}
Het \emph{kookpunt}\index{kookpunt} van een stof is de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur}
waarbij ze kookt. Voor water is dat \qty{100}{\celsius}. Ook hier
heeft elke stof haar eigen punt --- en daarom kan een zorgvuldige kok
water laten wegkoken terwijl de olie in dezelfde pan rustig
vloeibaar blijft.
\end{definition}

\begin{example}[De wereld lezen met twee getallen]\label{ex:g4:changes-of-state:reading}
Waarom is een reep chocola vast in het schap (\qty{20}{\celsius}) maar
vloeibaar in je zak tijdens een zomerwandeling (\qty{35}{\celsius})?
Omdat haar \omterm{def:g4:changes-of-state:melting-point}{smeltpunt}, rond \qty{34}{\celsius}, \emph{tussen} die twee
in ligt. Een stof is vast onder haar \omterm{def:g4:changes-of-state:melting-point}{smeltpunt} en vloeibaar
erboven --- tot aan haar \omterm{def:g4:changes-of-state:boiling-point}{kookpunt}, waar ze als \omterm{def:g3:solids-liquids-gases:gas}{gas} vertrekt. Twee
getallen vertellen het hele verhaal van elke stof bij elke
\omterm{def:g3:temperature-thermometers:temperature}{temperatuur}.
\end{example}

\section{De koppige thermometer}

\begin{method}[IJs zorgvuldig zien smelten]\label{met:g4:changes-of-state:watch}
\begin{enumerate}
  \item Vul een kom met gemalen ijs of sneeuw en plant de
        \omterm{def:g3:temperature-thermometers:temperature}{thermometer} erin; wacht en lees af: \qty{0}{\celsius};
  \item breng de kom naar de warme keuken en lees de \omterm{def:g3:temperature-thermometers:temperature}{thermometer} om
        de paar minuten af terwijl het ijs \omterm{def:g2:ice-water-steam:meltfreeze}{smelt};
  \item blijf aflezen tot lang nadat de laatste ijsschilfer is
        verdwenen.
\end{enumerate}
Het verslag: de hele smelttijd door --- water en ijs samen in de kom
--- blijft de \omterm{def:g3:temperature-thermometers:temperature}{thermometer} koppig op \qty{0}{\celsius} staan. Pas als het
\emph{laatste} stukje ijs gesmolten is, begint de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} te
klimmen.
\end{method}

\begin{proposition}[De plateauwet]\label{prop:g4:changes-of-state:plateau}
Terwijl een stof van toestand verandert --- \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} of kookt --- staat
haar \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} stil op de mijlpaal, hoe hard het fornuis ook werkt.
Alle binnenkomende warmte gaat op aan de toestandsverandering zelf, en
niets aan klimmen. Vast en vloeibaar wonen samen op het
\omterm{def:g4:changes-of-state:melting-point}{smeltpunt}; vloeibaar en \omterm{def:g3:solids-liquids-gases:gas}{gas} op het \omterm{def:g4:changes-of-state:boiling-point}{kookpunt}.
\end{proposition}

\begin{example}[Het plateau in de keuken]\label{ex:g4:changes-of-state:kitchen}
Daarom koken zachtjes pruttelend en woest bruisend water de pasta even
snel gaar: beide pannen staan op precies \qty{100}{\celsius} --- de
woeste verspilt haar extra warmte alleen aan sneller damp maken. En
daarom blijft een drankje met ijsblokjes tot de laatste schilfer op
\qty{0}{\celsius}: het smeltende ijs pint de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} vast op de
mijlpaal.
\end{example}

\begin{remark}[Een getal dat het oog niet ziet]\label{rem:g4:changes-of-state:invisible}
Zonder \omterm{def:g3:temperature-thermometers:temperature}{thermometer} zou niemand het plateau vermoeden: de pan ziet er
almaar drukker uit, de ijsemmer almaar rustiger, en in allebei staat
de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} muurvast. Het is opnieuw een overwinning van \omterm{def:g2:measuring-length:measure}{meten} op
schijn --- en een van de eerste grote verrassingen die instrumenten de
wetenschap ooit hebben aangereikt.
\end{remark}

\section{Elke stof houdt haar mijlpalen}

\begin{proposition}[Mijlpalen zijn kenmerken]\label{prop:g4:changes-of-state:identity}
Het \omterm{def:g4:changes-of-state:melting-point}{smeltpunt} en het \omterm{def:g4:changes-of-state:boiling-point}{kookpunt} van een zuivere stof veranderen
nooit: water \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} vandaag bij \qty{0}{\celsius}, morgen ook, in een
paleis of in een tent. De mijlpalen horen bij de identiteit van de
stof --- zo trouw dat een wetenschapper een onbekende stof kan
herkennen door te \omterm{def:g2:measuring-length:measure}{meten} waar ze \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} en kookt.
\end{proposition}

\begin{example}[De kaars en de gieterij]\label{ex:g4:changes-of-state:foundry}
Een kaars bij de open haard zakt in en druipt: kaarsvet gaat rond
\qty{60}{\celsius} voorbij zijn \omterm{def:g4:changes-of-state:melting-point}{smeltpunt}. In een gieterij schenken
arbeiders gloeiend oranje vloeibaar ijzer als soep --- hun oven ging
voorbij \qty{1538}{\celsius}. Hetzelfde spel, met wild verschillende
mijlpalen: wat de haard met kaarsvet doet, kan alleen een oven met
ijzer. En goud \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} bij \qty{1064}{\celsius}: goudsmeden kennen dat
getal, met hun handen zo niet in graden, al duizenden jaren.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:1}
Geef de officiële naam: vast naar vloeibaar; \ vloeibaar naar
vast; \ vloeibaar naar \omterm{def:g3:solids-liquids-gases:gas}{gas} met grote bellen; \ \omterm{def:g3:solids-liquids-gases:gas}{gas} naar
vloeibaar op een koud raam.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:2}
Wat zijn de twee mijlpalen van water, met hun temperaturen? En het
\omterm{def:g4:changes-of-state:melting-point}{smeltpunt} van chocola, ruwweg?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:3}
In een kom zitten ijs en water samen, goed geroerd. Wat leest de
\omterm{def:g3:temperature-thermometers:temperature}{thermometer} af? Wat leest hij vijf minuten later af, als er nog wat
ijs over is?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:4}
Formuleer de plateauwet. Waar gaat de warmte van het fornuis heen
terwijl het water kookt, nu de \omterm{def:g3:temperature-thermometers:temperature}{temperatuur} niet meer klimt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:5}
Kaarsvet \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} rond \qty{60}{\celsius}. Is een kaars vast of
vloeibaar: in een woonkamer van \qty{20}{\celsius}? \ naast een vlam
die hem tot \qty{80}{\celsius} verwarmt? \ op een zomerse zolder van
\qty{35}{\celsius}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:6}
Pasta kookt in zachtjes pruttelend water. Papa draait de vlam op de
hoogste stand en de pan bruist woest. Is het water nu heter? Wat maakt
de extra warmte in plaats daarvan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:7}
Hoe herken je op de plateaugrafiek uit het hoofdstuk de twee
toestandsveranderingen zonder ook maar één woord te lezen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:8}
IJzer \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} bij \qty{1538}{\celsius} en goud bij
\qty{1064}{\celsius}. Een oven draait op \qty{1200}{\celsius}: welk
van de twee metalen schenk je daarin als soep, en welk blijft
vast? Met hoeveel graden komt de oven bij het andere tekort?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:9}
Waarom blijft een drankje met ijsblokjes tot de laatste ijsschilfer
even koud --- en begint het pas daarna op te warmen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:changes-of-state:10}
Een wetenschapster krijgt een onbekende witte \omterm{def:g3:solids-liquids-gases:solid}{vaste stof}. Ze vindt
dat die precies bij \qty{0}{\celsius} \omterm{def:g2:ice-water-steam:meltfreeze}{smelt} en dat de \omterm{def:g3:solids-liquids-gases:liquid}{vloeistof}
precies bij \qty{100}{\celsius} kookt. Wat is haar beste gok over de
stof --- en welke wet maakt van mijlpalen identiteitskaarten?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:changes-of-state:11}
Bergbeklimmers melden dat water hoog op grote toppen kookt terwijl het
\emph{net} te heet is om aan te raken --- en dat hun thee nooit goed
trekt. Welke alledaagse zekerheid over kokend water gaat daarboven
onderuit? (De reden zit verstopt in de \omterm{def:g2:air-around-us:air}{lucht} zelf, en een later
hoofdstuk over luchtdruk zal haar te pakken krijgen.)
\end{exercise}
