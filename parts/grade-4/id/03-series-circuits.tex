\chapter{Rangkaian Seri}\label{ch:g4:series-circuits}

Tahun lalu, seuntai lampu pesta menjadi gelap karena satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} saja
hilang --- dan kita berjanji bahwa susunannya punya sebuah nama.
Inilah dia: \omterm{def:g4:series-circuits:series}{rangkaian seri}, cara paling sederhana menempatkan banyak
pemain pada satu lingkar. Rangkaian ini punya aturan rumah yang ketat,
dan senter di lacimu sudah menaatinya sejak dahulu.

\section{Satu lingkar, beberapa pemain}

\begin{definition}[Terangkai seri]\label{def:g4:series-circuits:series}
Para pemain dalam sebuah rangkaian dikatakan
\emph{terangkai seri}\index{rangkaian seri} ketika mereka duduk pada
satu lingkar tunggal, satu demi satu, seperti manik pada satu kalung:
jalan arusnya melewati yang pertama, lalu yang kedua, lalu yang
ketiga, tanpa percabangan di mana pun. Rangkaian yang dibangun begini
disebut \emph{rangkaian seri}.
\end{definition}

\begin{method}[Membangunnya]\label{met:g4:series-circuits:build}
Dengan satu \omterm{def:g3:first-electric-circuit:battery}{baterai}, dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, sebuah \omterm{ex:g3:first-electric-circuit:switch}{sakelar}, dan kawat:
\begin{enumerate}
  \item mulailah dari kutub $+$ \omterm{def:g3:first-electric-circuit:battery}{baterai} dengan sebuah kawat menuju
        \omterm{ex:g3:first-electric-circuit:switch}{sakelar};
  \item pasang kawat dari \omterm{ex:g3:first-electric-circuit:switch}{sakelar} ke \omterm{def:g3:first-electric-circuit:bulb}{bohlam} pertama, dan dari \omterm{def:g3:first-electric-circuit:bulb}{bohlam}
        pertama ke \omterm{def:g3:first-electric-circuit:bulb}{bohlam} kedua;
  \item tutuplah lingkarnya: dari \omterm{def:g3:first-electric-circuit:bulb}{bohlam} kedua kembali ke kutub $-$
        \omterm{def:g3:first-electric-circuit:battery}{baterai};
  \item klik \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} dan amati kedua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} bersama-sama.
\end{enumerate}
Telusurilah lingkarnya dengan jarimu sebelum menyalakan: kalau jarimu
sampai harus memilih di antara dua jalan, itu bukan \omterm{def:g4:series-circuits:series}{rangkaian seri}.
\end{method}

\section{Aturan rumahnya}

\begin{proposition}[Aturan lingkar tunggal]\label{prop:g4:series-circuits:rules}
Pada \omterm{def:g4:series-circuits:series}{rangkaian seri}:
\begin{enumerate}
  \item \emph{semua atau tidak sama sekali}: satu celah di mana pun
        --- \omterm{ex:g3:first-electric-circuit:switch}{sakelar} yang terbuka, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang putus, kawat yang
        longgar --- dan setiap pemain berhenti seketika;
  \item \emph{urutan tidak berpengaruh}: tukarlah kedua \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya},
        pindahkan \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} ke sisi lingkar yang lain --- semuanya
        berperilaku persis seperti sebelumnya;
  \item \emph{berbagi}: makin banyak \omterm{def:g3:first-electric-circuit:bulb}{bohlam} pada lingkarnya, makin
        redup nyala masing-masing; dorongan \omterm{def:g3:first-electric-circuit:battery}{baterai} dibagi di
        sepanjang kalungnya.
\end{enumerate}
\end{proposition}

\begin{example}[Melihat aturannya]\label{ex:g4:series-circuits:seeing}
Bangunlah rangkaian dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} itu lalu ujilah tiap aturannya.
Lepaskan \emph{salah satu} \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}: keduanya menjadi gelap ---
aturan pertama (soket yang kosong itu sebuah celah). Tukarlah kedua
\omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}, atau pindahkan \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} ke antara keduanya: sama sekali
tidak ada perubahan --- aturan kedua. Sekarang bangunlah kembali
dengan satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} saja: nyalanya jelas lebih terang sendirian
daripada ketika ditemani --- aturan ketiga, dibalik.
\end{example}

\begin{example}[Lampu pesta, terpecahkan]\label{ex:g4:series-circuits:partylights}
Misteri lama itu kini bukan misteri lagi: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} lampu pesta \omterm{def:g4:series-circuits:series}{terangkai
seri}, berpuluh manik pada satu kalung. Satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang hilang adalah
sebuah celah, dan aturan pertama menggelapkan seluruh untaiannya.
Mencari \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang bersalah dengan mencobanya satu per satu mengajari
banyak keluarga aturan pertama dengan cara yang berat --- untaian
zaman sekarang dibangun dengan cara lain, seperti akan ditunjukkan
tahun depan.
\end{example}

\begin{remark}[Sakelar memerintah semua pemain]\label{rem:g4:series-circuits:switch}
Aturan kedua punya akibat yang praktis: satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar}, diletakkan di
mana pun pada lingkarnya, memerintah setiap pemain di atasnya. Itulah
sebabnya satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} dinding dapat mengendalikan sebaris lampu
langit-langit --- dan mengapa, karena alasan yang sama, kamu tidak
dapat mematikan \emph{satu} \omterm{def:g3:first-electric-circuit:bulb}{bohlam} saja dari untaian seri tanpa
mematikan semuanya.
\end{remark}

\section{Lebih banyak baterai pada kalungnya}

\begin{example}[Kepala bertemu ekor]\label{ex:g4:series-circuits:batteries}
\omterm{def:g3:first-electric-circuit:battery}{Baterai} pun dapat berbaris seri --- tetapi mereka lebih cerewet
daripada \omterm{def:g3:first-electric-circuit:bulb}{bohlam}: mereka harus berdiri \emph{kepala bertemu ekor},
kutub $+$ tiap \omterm{def:g3:first-electric-circuit:battery}{baterai} menyentuh kutub $-$ \omterm{def:g3:first-electric-circuit:battery}{baterai} berikutnya. Dua
\omterm{def:g3:first-electric-circuit:battery}{baterai} yang berbaris begini mendorong bersama-sama, dan \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya}
menyala jelas lebih terang daripada dengan satu \omterm{def:g3:first-electric-circuit:battery}{baterai}. Balikkan satu
\omterm{def:g3:first-electric-circuit:battery}{baterainya} --- kepala bertemu kepala --- dan kedua dorongannya saling
melawan: \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} tidak memberi apa-apa.
\end{example}

  \end{scope}
\end{tikzpicture}

bersama-sama: \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang sama menyala lebih terang daripada dengan
satu \omterm{def:g3:first-electric-circuit:battery}{baterai} saja.}
\end{omfigure}

\begin{example}[Di dalam senter]\label{ex:g4:series-circuits:flashlight}
Bukalah sebuah senter: \omterm{def:g3:first-electric-circuit:battery}{baterainya} masuk satu di belakang yang lain di
dalam tabung --- barisan kepala-bertemu-ekor yang sudah jadi --- lalu
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} dan tombol \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} melengkapi satu lingkar tunggal lewat
selubungnya. Sebuah senter \emph{adalah} \omterm{def:g4:series-circuits:series}{rangkaian seri} di dalam saku.
Dan sekarang kamu tahu mengapa senter menjadi gelap kalau satu \omterm{def:g3:first-electric-circuit:battery}{baterai}
saja dimasukkan terbalik: kepala bertemu kepala, kedua dorongannya
saling meniadakan.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:1}
Apa arti ``\omterm{def:g4:series-circuits:series}{terangkai seri}''? Gambaran sehari-hari apa yang dipakai bab
ini untuk itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:2}
Sebutkan ketiga aturan rumah \omterm{def:g4:series-circuits:series}{rangkaian seri}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:3}
Pada rangkaian dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam}, \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2 dilepaskan. Apa yang terjadi pada
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1, dan mengapa? Aturan mana yang sedang bekerja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:4}
Zoe memindahkan \omterm{ex:g3:first-electric-circuit:switch}{sakelarnya} dari sebelum \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 1 ke sesudah \omterm{def:g3:first-electric-circuit:bulb}{bohlam} 2.
Apa yang berubah pada perilaku rangkaiannya? Aturan mana yang
mengatakannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:5}
Satu \omterm{def:g3:first-electric-circuit:battery}{baterai} menyalakan satu \omterm{def:g3:first-electric-circuit:bulb}{bohlam} dengan baik. Ramalkan kecerahannya
dengan: dua \omterm{def:g3:first-electric-circuit:bulb}{bohlam} seri; \ tiga \omterm{def:g3:first-electric-circuit:bulb}{bohlam} seri. Apa yang sedang dibagi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:6}
Bagaimana dua \omterm{def:g3:first-electric-circuit:battery}{baterai} harus berdiri supaya mendorong bersama-sama?
Gambarlah barisannya dengan tanda $+$ dan $-$ pada tiap \omterm{def:g3:first-electric-circuit:battery}{baterai}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:7}
Telusurilah lingkar sebuah senter: sebutkan pemain yang disinggahi
arusnya dari kutub $+$ \omterm{def:g3:first-electric-circuit:battery}{baterai} sampai kembali ke kutub $-$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:8}
Rangkaian bel pintu punya \omterm{def:g3:first-electric-circuit:battery}{baterai}, tombol, dan belnya \omterm{def:g4:series-circuits:series}{terangkai seri}.
Di mana kamu dapat menambahkan tombol kedua supaya belnya berbunyi
hanya ketika \emph{kedua} tombolnya ditekan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:9}
Seuntai sepuluh lampu pesta seri sedang gelap. Semua \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} tampak
baik-baik saja. Ceritakan rencana yang sabar, dengan memakai satu
\omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang pasti baik, untuk menemukan \omterm{def:g3:first-electric-circuit:bulb}{bohlam} yang rusak.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:series-circuits:10}
Senter Tom memuat dua \omterm{def:g3:first-electric-circuit:battery}{baterai} dan menyala terang. Sesudah mengganti
\omterm{def:g3:first-electric-circuit:battery}{baterainya} dalam gelap, senter itu tidak memberi apa-apa --- padahal
kedua \omterm{def:g3:first-electric-circuit:battery}{baterai} barunya penuh, \omterm{def:g3:first-electric-circuit:bulb}{bohlamnya} baik, dan tidak ada kawat yang
longgar. Apa yang telah dilakukan Tom, dan mengapa senternya tetap
gelap? (Lihat \cref{ex:g4:series-circuits:batteries}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:series-circuits:11}
Seorang insinyur harus memasang satu \omterm{ex:g3:first-electric-circuit:switch}{sakelar} penghenti darurat untuk
sebuah mesin yang motornya duduk pada satu lingkar bersama \omterm{def:g3:first-electric-circuit:battery}{baterainya}.
Ia berkata: ``Di mana pun pada lingkarnya boleh.'' Benarkah ia? Dan
mengapa rencananya akan gagal kalau rangkaian mesin itu punya jalan
bercabang? (Bab tahun depan punya namanya.)
\end{exercise}
