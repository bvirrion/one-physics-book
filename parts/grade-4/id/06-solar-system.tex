\chapter{Tata Surya}\label{ch:g4:solar-system}

Bumi, seperti kamu tahu, menempuh satu putaran setahun mengelilingi
Matahari. Tetapi apakah ia bepergian sendirian? Tengadahlah pada
petang yang cerah: di antara kerlip yang tetap, ada beberapa \omterm{def:g1:light-and-shadows:source}{cahaya}
terang yang perlahan mengembara dari minggu ke minggu. Orang zaman
dahulu menyebutnya pengembara; kita menyebutnya \omterm{def:g4:solar-system:planet}{planet} --- dunia yang
mengelilingi Matahari yang sama dengan kita.

\section{Satu bintang, delapan planet}

\begin{definition}[Bintang]\label{def:g4:solar-system:star}
Sebuah \emph{bintang}\index{bintang} adalah bola raksasa dari zat yang
berpijar panas dan membuat \omterm{def:g1:light-and-shadows:source}{cahayanya} sendiri --- \omterm{def:g1:light-and-shadows:source}{sumber cahaya} sejati.
Matahari adalah bintang: bintang kita, cukup dekat untuk menghangatkan
wajah kita. (Bintang lain pada malam hari mendapat bab tersendiri
tahun depan.)
\end{definition}

\begin{definition}[Planet]\label{def:g4:solar-system:planet}
Sebuah \emph{planet}\index{planet} adalah dunia besar yang bulat, yang
mengelilingi sebuah \omterm{def:g4:solar-system:star}{bintang} dan tidak membuat \omterm{def:g1:light-and-shadows:source}{cahaya} sendiri --- kita
melihat planet, seperti kita melihat \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, lewat \omterm{def:g1:light-and-shadows:source}{cahaya} matahari yang
dipantulkannya kembali. Bumi adalah sebuah planet.
\end{definition}

\begin{definition}[Tata Surya]\label{def:g4:solar-system:family}
\emph{Tata Surya}\index{Tata Surya} adalah Matahari bersama segala
sesuatu yang mengelilinginya: delapan \omterm{def:g4:solar-system:planet}{planet} --- berurutan: Merkurius, Venus, Bumi, Mars, Jupiter, Saturnus,
Uranus, Neptunus --- ditambah bulan-bulannya dan sekerumunan batuan
serta bola es yang lebih kecil.
\end{definition}

\begin{proposition}[Kebiasaan para planet]\label{prop:g4:solar-system:habits}
Para \omterm{def:g4:solar-system:planet}{planet} menjaga kebiasaan yang rapi: masing-masing mengelilingi
Matahari pada jalannya sendiri yang hampir berupa lingkaran, semuanya
berputar ke arah yang sama, dan semua jalannya terletak hampir pada
satu bidang datar --- seperti cincin yang digambar pada satu permukaan
meja. Dan makin jauh jalan sebuah \omterm{def:g4:solar-system:planet}{planet} dari Matahari, makin lama
waktu yang diperlukannya untuk satu putaran: Merkurius berputar dalam
$88$ hari, Bumi dalam satu tahun, Neptunus yang jauh dalam $165$
tahun.
\end{proposition}

\begin{example}[Planet batuan dan raksasa gas]\label{ex:g4:solar-system:kinds}
Empat \omterm{def:g4:solar-system:planet}{planet} yang di dalam --- Merkurius, Venus, Bumi, Mars --- adalah
bola batuan yang kecil dan padat: dunia yang dapat kamu pijaki. Empat
yang di luar --- Jupiter, Saturnus, Uranus, Neptunus --- adalah
\emph{raksasa}: bola yang sangat besar dan sebagian besar berupa \omterm{def:g3:solids-liquids-gases:gas}{gas}
tebal, tanpa tanah untuk dipijak sama sekali. Jupiter, yang terbesar,
dapat menelan lebih dari seribu Bumi; Saturnus mengenakan cincin
berkilaunya yang termasyhur --- keping es yang tak terhitung banyaknya
mengelilinginya seperti kerumunan bulan yang mungil.
\end{example}

\begin{example}[Bulan bukan planet]\label{ex:g4:solar-system:moons}
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} kita mengelilingi \emph{Bumi}, bukan langsung mengelilingi
Matahari --- jadi ia bukan \omterm{def:g4:solar-system:planet}{planet} melainkan sebuah \emph{bulan},
pendamping sebuah \omterm{def:g4:solar-system:planet}{planet}. \omterm{def:g4:solar-system:planet}{Planet} lain punya bulannya sendiri: Mars
punya dua yang kecil, Jupiter dan Saturnus punya berpuluh-puluh
masing-masing. Sebuah bulan milik \omterm{def:g4:solar-system:planet}{planetnya}; \omterm{def:g4:solar-system:planet}{planetnya} milik Matahari.
\end{example}

\section{Seberapa jauh itu jauh}

\begin{method}[Model berskala Tata Surya]\label{met:g4:solar-system:model}
Jarak antarplanet sangat besar --- jutaan kilometer. Untuk
merasakannya, bangunlah model berskala \omterm{def:g4:solar-system:family}{Tata Surya} di sebuah taman,
dengan menyusutkan setiap ukuran dan jarak menurut faktor yang sama:
\begin{enumerate}
  \item jadikan Matahari sebuah bola sepak yang diletakkan di gerbang;
  \item maka Bumi adalah sebutir merica --- dan untuk menempatkannya
        secara adil, berjalanlah kira-kira $30$ langkah besar dari
        bola sepaknya;
  \item Merkurius, sebutir kepala jarum, berdiri kira-kira $12$
        langkah dari bola sepaknya; Jupiter, sebutir kelereng,
        kira-kira $150$ langkah; Neptunus, sebutir manik kecil,
        kira-kira $900$ langkah --- hampir satu kilometer jauhnya!
\end{enumerate}
Berjalanlah menyusurinya, dan rasakan apa yang tidak ditunjukkan
gambar mana pun: \omterm{def:g4:solar-system:family}{Tata Surya} hampir seluruhnya ruang kosong, dengan
dunia mungil yang berjauhan. Pada model ini satu langkah besar
mewakili kira-kira lima juta kilometer ruang yang sesungguhnya.
\end{method}

\begin{example}[Para pengembara di langit kita]\label{ex:g4:solar-system:wanderers}
Venus sering menjadi ``\omterm{def:g4:solar-system:star}{bintang}'' pertama pada petang hari, menyala di
barat sesudah \omterm{def:g1:day-and-night:daynight}{matahari terbenam} --- begitu terang karena ia tetangga
terdekat kita, terbungkus awan putih yang memantulkan \omterm{def:g1:light-and-shadows:source}{cahaya} matahari
dengan murah hati. Mars bersinar jingga samar; Jupiter dan Saturnus
bersinar mantap dan terang. Amatilah salah satunya selama beberapa
minggu terhadap \omterm{def:g4:solar-system:star}{bintang} yang tetap: ia bergeser --- pengembaraan itu
sendirilah yang memberi \omterm{def:g4:solar-system:planet}{planet} namanya yang kuno.
\end{example}

\begin{remark}[Gambaran lama, digambar ulang]\label{rem:g4:solar-system:history}
Selama sebagian besar sejarah, orang menempatkan Bumi di pusat segala
sesuatu, dengan Matahari, \omterm{def:g2:moon-in-the-sky:moon}{Bulan}, dan \omterm{def:g4:solar-system:planet}{planet} mengelilingi kita. Para
pengamat pengembara yang cermat mendapati gambaran itu mulai berderit
--- \omterm{def:g4:solar-system:planet}{planet} kadang bergeser \emph{mundur} untuk sementara, sukar
dijelaskan dengan Bumi yang bertakhta di tengah. Gambaran berpusat
Matahari yang baru saja kamu pelajari menjelaskan pengembaraan itu
dengan sederhana, dan ia termasuk gagasan paling berani yang pernah
diterima manusia: rumah kita bukanlah pusatnya --- ia adalah \omterm{def:g4:solar-system:planet}{planet}
ketiga dari Matahari.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:1}
Apa bedanya \omterm{def:g4:solar-system:star}{bintang} dan \omterm{def:g4:solar-system:planet}{planet}? Yang mana Matahari itu? Dan Bumi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:2}
Sebutkan kedelapan \omterm{def:g4:solar-system:planet}{planet} berurutan dari Matahari. Yang mana \omterm{def:g4:solar-system:planet}{planet}
kita?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:3}
\omterm{def:g4:solar-system:planet}{Planet} mana saja yang kecil dan berbatu? Yang mana raksasa gasnya?
Sebutkan yang terbesar dari semuanya, dan yang bercincin termasyhur.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:4}
Mengapa \omterm{def:g2:moon-in-the-sky:moon}{Bulan} tidak terhitung sebagai \omterm{def:g4:solar-system:planet}{planet}? Apa yang dikelilinginya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:5}
Merkurius berputar mengelilingi Matahari dalam $88$ hari, Bumi dalam
$365$, Neptunus dalam $165$ tahun. Kebiasaan \omterm{def:g4:solar-system:planet}{planet} yang mana yang
ditunjukkan bilangan ini?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:6}
Pada model berskala di taman, apa yang berperan sebagai Matahari, dan
berapa langkah jauhnya Bumi berdiri? Hampir seluruh \omterm{def:g4:solar-system:family}{Tata Surya} terbuat
dari apa, dilihat dari ruangnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:7}
Venus bersinar cemerlang, namun ia bukan \omterm{def:g4:solar-system:star}{bintang}. Dari mana \omterm{def:g1:light-and-shadows:source}{cahayanya}
datang? (Seorang kawan lama menjawab: ingatlah \omterm{def:g2:moon-in-the-sky:moon}{Bulan}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:8}
Bagaimana kamu dapat mengetahui, dengan mengamati langit selama
beberapa minggu, bahwa sebuah \omterm{def:g1:light-and-shadows:source}{cahaya} terang adalah \omterm{def:g4:solar-system:planet}{planet} dan bukan
\omterm{def:g4:solar-system:star}{bintang}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:9}
Pada model di taman, Jupiter berdiri kira-kira $150$ langkah dan Bumi
kira-kira $30$. Kira-kira berapa kali lebih jauh Jupiter dari Matahari
dibandingkan Bumi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:solar-system:10}
Seorang teman menggambar \omterm{def:g4:solar-system:family}{Tata Surya} dengan kedelapan \omterm{def:g4:solar-system:planet}{planet} berjejal
rapat di sekeliling Matahari seperti gambar pertama dalam bab ini, dan
bertanya mengapa wahana antariksa memerlukan bertahun-tahun untuk
mencapai \omterm{def:g4:solar-system:planet}{planet} luar. Luruskanlah dengan model berskala di taman ---
tentang apa hampir semua gambar \omterm{def:g4:solar-system:family}{Tata Surya} berbohong, dan mengapa
mereka terpaksa begitu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:solar-system:11}
Ketika Bumi masih dipercaya duduk di pusat segala sesuatu, teka-teki
pengamatan langit apa yang berderit melawan gambaran itu? Bagaimana
menempatkan Matahari di pusatnya melarutkan teka-teki itu --- dan apa
harga yang dibayar potret baru itu dari kebanggaan kita?
\end{exercise}
