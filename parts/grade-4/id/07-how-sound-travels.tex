\chapter{Bagaimana Bunyi Merambat}\label{ch:g4:how-sound-travels}

Kilatan petir --- lalu satu, dua, tiga sekon kesunyian --- dan baru
sekarang guruh menggeram. Padahal kilat dan geram itu lahir bersama di
awan yang sama! \omterm{def:g3:sound-around-us:sound}{Bunyi}, seperti kita pelajari, adalah \omterm{def:g3:sound-around-us:vibration}{getaran} yang
merambat. Bab ini mengikutinya di jalan: lewat apa ia merambat,
seberapa cepat ia berlari, dan apa yang terjadi ketika ia menabrak
dinding.

\section{Bunyi memerlukan perantara}

\begin{proposition}[Tanpa perantara, tanpa bunyi]\label{prop:g4:how-sound-travels:carrier}
\omterm{def:g3:sound-around-us:sound}{Bunyi} merambat hanya \emph{melalui} benda: melalui \omterm{def:g2:air-around-us:air}{udara}, melalui air,
melalui kayu, batu bata, dan baja. Di tempat yang sama sekali tidak
berisi apa-apa --- tanpa \omterm{def:g2:air-around-us:air}{udara}, tanpa zat apa pun --- \omterm{def:g3:sound-around-us:sound}{bunyi} tidak
dapat merambat: \omterm{def:g3:sound-around-us:vibration}{getarannya} tidak punya apa pun untuk digetarkan.
\end{proposition}

\begin{example}[Lonceng di dalam toples]\label{ex:g4:how-sound-travels:jar}
Sebuah percobaan termasyhur, berumur tiga abad: sebuah lonceng listrik
berdering di dalam toples kaca --- terdengar jelas. Sebuah pompa lalu
mengisap \omterm{def:g2:air-around-us:air}{udara} keluar dari toples yang tertutup rapat itu sementara
pemukul loncengnya jelas terlihat terus memukul. Deringnya
memudar\dots\ lalu mati. Biarkan \omterm{def:g2:air-around-us:air}{udaranya} menyerbu masuk kembali:
deringnya kembali. Loncengnya tidak pernah berhenti ---
\emph{perantaranya} yang diambil lalu dikembalikan. Dan sekarang kamu
dapat menyelesaikan pertanyaan lama: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang sunyi, tempat teriakan
mati tanpa terdengar, sunyi karena di sana sama sekali tidak ada
\omterm{def:g2:air-around-us:air}{udara}.
\end{example}

\begin{example}[Perantara yang lebih baik daripada udara]\label{ex:g4:how-sound-travels:solids}
\omterm{def:g3:solids-liquids-gases:solid}{Zat padat} dan air membawa \omterm{def:g3:sound-around-us:sound}{bunyi} dengan sangat baik --- sering lebih
baik daripada \omterm{def:g2:air-around-us:air}{udara}. Tempelkan telingamu rata pada meja lalu mintalah
seorang teman menggaruk ujung meja yang jauh \emph{sangat} pelan:
lewat \omterm{def:g2:air-around-us:air}{udara} kamu tidak mendengar apa-apa, lewat kayunya garukan itu
terdengar keras dan tajam. Perenang mendengar dentingan di dalam kolam
dari ujung ke ujung; paus bernyanyi satu sama lain melintasi laut yang
utuh. Dan dinding yang tebal membawa \omterm{def:g3:sound-around-us:sound}{bunyi} bor tetangga ke setiap
ruangan --- \omterm{def:g3:solids-liquids-gases:solid}{zat padat} hampir terlalu pandai dalam hal ini.
\end{example}

\begin{method}[Telepon benang]\label{met:g4:how-sound-travels:telephone}
\begin{enumerate}
  \item Lubangi bagian bawah dua gelas kertas dengan lubang kecil;
  \item masukkan kedua ujung seutas benang panjang ke dalamnya, lalu
        simpulkan masing-masing supaya tidak terlepas;
  \item berjalanlah saling menjauh sampai benangnya teregang
        \emph{kencang} dan tidak menyentuh apa pun di sepanjang
        jalannya;
  \item yang satu berbicara pelan ke dalam gelasnya; yang lain
        menempelkan gelasnya ke telinga.
\end{enumerate}
Bisikan itu tiba dengan jelas menyeberangi halaman bermain: suaramu
membuat gelasnya bergetar, \omterm{def:g3:sound-around-us:vibration}{getarannya} berlari menyusuri benang yang
tegang, dan gelas yang jauh memutarnya kembali. Biarkan benangnya
kendur atau jepitlah dengan dua jari, dan teleponnya mati --- jalannya
terputus.
\end{method}

\section{Bunyi memerlukan waktu}

\begin{proposition}[Cepat, tetapi tidak seketika]\label{prop:g4:how-sound-travels:speed}
\omterm{def:g3:sound-around-us:sound}{Bunyi} adalah pelari, bukan pesulap: di \omterm{def:g2:air-around-us:air}{udara} ia menempuh kira-kira
\qty{340}{m} tiap sekon --- tiga sekon untuk kira-kira satu kilometer.
\omterm{def:g1:light-and-shadows:source}{Cahaya}, sebaliknya, tiba begitu cepat sehingga untuk pemandangan mana
pun yang dapat kamu amati, perjalanannya terhitung seketika. Perbedaan
itulah seluruh kisah badai guruh.
\end{proposition}

\begin{method}[Seberapa jauh badainya?]\label{met:g4:how-sound-travels:storm}
\begin{enumerate}
  \item Lihatlah kilatnya --- mulailah menghitung sekon dengan tempo
        ajek: satu buaya kecil, dua buaya kecil, \dots;
  \item berhentilah pada geram guruh yang pertama;
  \item bagilah hitunganmu dengan $3$: itulah kira-kira jarak badainya
        dalam kilometer.
\end{enumerate}
Enam sekon: kira-kira \qty{2}{km}. Hitungan yang makin pendek di
antara sambaran berarti badainya makin mendekat --- saatnya masuk ke
dalam rumah.
\end{method}

\begin{example}[Gema]\label{ex:g4:how-sound-travels:echo}
Berteriaklah ke arah tebing besar atau bangunan yang berdiri sendiri
jauh di seberang lapangan: ``HALO!'' --- sesaat sunyi --- ``Halo!''
suaramu sendiri menjawab. \omterm{def:g3:sound-around-us:sound}{Bunyi} memantul dari dinding keras yang
besar, dan \omterm{def:g3:sound-around-us:sound}{bunyi} pantulan yang kembali kepadamu cukup terlambat untuk
terdengar terpisah disebut \emph{gema}\index{gema}. Makin jauh
dindingnya, makin lama pula perjalanan bolak-balik suaramu, dan makin
terlambat \omterm{ex:g4:how-sound-travels:echo}{gemanya}. Di ruangan kecil pantulannya kembali terlalu cepat untuk
terdengar terpisah --- ia hanya membuat suaramu terdengar lebih penuh.
\end{example}

\begin{remark}[Sonar: gema yang dipekerjakan]\label{rem:g4:how-sound-travels:sonar}
Kapal \omterm{def:g2:measuring-length:measure}{mengukur} kedalaman laut dengan berteriak ke bawah --- sebuah
klik ke dalam air --- lalu menghitung waktu \omterm{ex:g4:how-sound-travels:echo}{gema} dari dasarnya. Air
yang dalam, \omterm{ex:g4:how-sound-travels:echo}{gema} yang terlambat; air yang dangkal, \omterm{ex:g4:how-sound-travels:echo}{gema} yang cepat.
Lumba-lumba dan kelelawar lahir sudah mengetahui tipuan itu: mereka
mengeklik lalu mendengarkan, membaca \omterm{ex:g4:how-sound-travels:echo}{gemanya} untuk berburu di air yang
gelap dan gua yang pekat. Dengan menghitung sekon guruh, kamu sudah
memakai cara mereka: hitunglah waktu \omterm{def:g3:sound-around-us:sound}{bunyinya}, ketahuilah jaraknya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:1}
Sebutkan tiga perantara \omterm{def:g3:sound-around-us:sound}{bunyi} dari bab ini, dan satu tempat yang sama
sekali tidak dapat dilewati \omterm{def:g3:sound-around-us:sound}{bunyi}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:2}
Pada percobaan toples, pemukul loncengnya terus memukul sementara
deringnya mati. Apa tepatnya yang diambil, dan apa yang dibuktikannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:3}
Mengapa \omterm{def:g2:moon-in-the-sky:moon}{Bulan} menjadi dunia yang sunyi? Jawablah dengan proposisi
dalam bab ini.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:4}
Pada telepon benang, apa yang membawa bisikannya? Berikan kedua aturan
untuk menjaga teleponnya tetap bekerja.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:5}
Kira-kira sejauh apa \omterm{def:g3:sound-around-us:sound}{bunyi} merambat melalui \omterm{def:g2:air-around-us:air}{udara} dalam satu sekon?
Dalam tiga sekon?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:6}
Kamu menghitung sembilan sekon antara kilat dan guruh. Seberapa jauh
badainya? Temanmu hanya menghitung tiga sekon pada sambaran berikutnya
--- apa yang berubah, dan apa yang sebaiknya kalian berdua lakukan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:7}
Mengapa menempelkan telinga pada meja membuatmu mendengar garukan yang
tidak pernah diantarkan \omterm{def:g2:air-around-us:air}{udara}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:8}
Apa itu \omterm{ex:g4:how-sound-travels:echo}{gema}? Mengapa tebing yang lebih besar dan lebih jauh menjawab
lebih terlambat daripada tebing yang dekat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:how-sound-travels:9}
Bagaimana sonar sebuah kapal membedakan air yang dalam dari yang
dangkal? Dua binatang mana yang memainkan permainan yang sama, dan di
mana?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:how-sound-travels:10}
Film koboi lama menunjukkan pengintai menempelkan telinga ke rel baja
untuk menangkap kereta yang masih jauh. Mengapa relnya membocorkan
kereta itu jauh sebelum \omterm{def:g2:air-around-us:air}{udara} melakukannya? Berikan kedua keuntungan
yang ditawarkan rel itu kepada \omterm{def:g3:sound-around-us:sound}{bunyinya}. (Yang satu tentang membawa
dengan baik; untuk yang lain, ingatlah bahwa relnya membentang lurus
sampai ke kereta sementara \omterm{def:g3:sound-around-us:sound}{bunyi} di \omterm{def:g2:air-around-us:air}{udara} menyebar ke segala arah
seperti riak kolam, dan menipis sambil berjalan.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:how-sound-travels:11}
Berdiri di halaman luas yang kosong, kamu bertepuk sekali dan
mendengar tepukanmu sendiri kembali dari dinding di seberang sesudah
satu sekon penuh. \omterm{def:g3:sound-around-us:sound}{Bunyinya} berlari ke dinding lalu kembali dalam sekon
itu. Seberapa jauh dindingnya --- hati-hati, pikirkan perjalanan
bolak-baliknya!
\end{exercise}
