\chapter{Magnet dan Kompas}\label{ch:g4:magnets-and-compass}

Bertahun-tahun lalu kamu menemukan bahwa kedua ujung \omterm{def:g1:magnets:magnet}{magnet} berbeda:
dengan satu cara dua \omterm{def:g1:magnets:magnet}{magnet} menempel rapat, dibalik keduanya saling
menolak. Kita berjanji bahwa kedua ujung itu punya nama dan tugas yang
termasyhur --- menuntun para pengembara melintasi dunia. Hari ini
janji itu ditepati: selamat datang pada \omterm{def:g4:magnets-and-compass:poles}{kutub}, dan pada alat kecil
yang menemukan jalan bagi setiap penjelajah sebelum ada satelit.

\section{Kedua kutub}

\begin{definition}[Kutub]\label{def:g4:magnets-and-compass:poles}
Kedua ujung istimewa sebuah \omterm{def:g1:magnets:magnet}{magnet} --- tempat tarikannya paling kuat,
tempat penjepit kertas berkerumun --- adalah \emph{kutub}\index{kutub}
\omterm{def:g1:magnets:magnet}{magnet} itu. Keduanya dinamai
\emph{kutub utara}\index{kutub utara} (yang sering dicat merah, bertanda U)
dan \emph{kutub selatan}\index{kutub selatan} (bertanda S). Setiap
\omterm{def:g1:magnets:magnet}{magnet} punya keduanya: satu dari masing-masing, selalu.
\end{definition}

\begin{proposition}[Aturan kutub]\label{prop:g4:magnets-and-compass:rule}
\omterm{def:g4:magnets-and-compass:poles}{Kutub} yang berlawanan saling menarik: U menempel pada S. \omterm{def:g4:magnets-and-compass:poles}{Kutub} yang
sejenis saling menolak: U mendorong U menjauh, S mendorong S menjauh.
Satu aturan ini menjelaskan setiap tempelan dan setiap penolakan
kenyal yang pernah kamu rasakan di antara dua \omterm{def:g1:magnets:magnet}{magnet}.
\end{proposition}

\begin{example}[Membaca percobaan lama]\label{ex:g4:magnets-and-compass:reading}
Ketika kedua magnetmu menempel rapat, U sedang berhadapan dengan S.
Ketika keduanya menolak dengan kenyal, U berhadapan dengan U (atau S
dengan S). Dan penjepit kertas selalu berkerumun di kedua ujungnya
karena di kutublah kekuatan sebuah \omterm{def:g1:magnets:magnet}{magnet} bersemayam --- bagian
tengahnya adalah ketenangan di antara keduanya.
\end{example}

\section{Bumi adalah sebuah magnet}

\begin{example}[Magnet terbesar dari semuanya]\label{ex:g4:magnets-and-compass:earth}
Gantungkan sebuah \omterm{def:g1:magnets:magnet}{magnet} batang pada seutas benang sehingga ia dapat
berputar bebas, tunggulah sampai ia tenang --- dan sesuatu yang
menakjubkan terjadi: ia selalu berhenti menunjuk ke \emph{arah yang
sama}, satu \omterm{def:g4:magnets-and-compass:poles}{kutubnya} ke ujung utara dunia, \omterm{def:g4:magnets-and-compass:poles}{kutub} yang lain ke ujung
selatan. Pasti ada \omterm{def:g1:magnets:magnet}{magnet} raksasa yang tak terlihat yang menarik
\omterm{def:g4:magnets-and-compass:poles}{kutubnya} sampai sejajar. Raksasa itu adalah Bumi sendiri: seluruh
planet kita adalah sebuah \omterm{def:g1:magnets:magnet}{magnet}, dengan \omterm{def:g4:magnets-and-compass:poles}{kutub} \omterm{def:g1:magnets:magnet}{magnetnya} di dekat
puncak dan dasar bola dunia.
\end{example}

\begin{definition}[Kompas]\label{def:g4:magnets-and-compass:compass}
Sebuah \emph{kompas}\index{kompas} adalah jarum \omterm{def:g1:magnets:magnet}{magnet} yang kecil dan
ringan, diseimbangkan pada poros yang halus sehingga dapat berayun
bebas. Kalau dibiarkan tenang, jarumnya menurut kepada \omterm{def:g1:magnets:magnet}{magnet} Bumi:
ujungnya yang bertanda berhenti menunjuk ke utara. Di bawah jarumnya,
sebuah piringan menunjukkan empat
\emph{arah mata angin}\index{arah mata angin}: Utara, Timur, Selatan,
Barat.
\end{definition}

\begin{method}[Memakai kompas]\label{met:g4:magnets-and-compass:use}
\begin{enumerate}
  \item Peganglah \omterm{def:g4:magnets-and-compass:compass}{kompas} mendatar dan mantap, jauh dari \omterm{def:g1:magnets:magnet}{magnet} dan
        benda besi yang besar (pagar, mobil --- semuanya mengganggu
        jarumnya);
  \item tunggulah sampai jarumnya berhenti berayun;
  \item putarlah piringan kompasnya pelan-pelan sampai huruf U-nya
        duduk di bawah ujung jarum yang bertanda;
  \item bacalah arahmu: Timur di sebelah kananmu ketika menghadap
        utara, Barat di sebelah kirimu, Selatan di punggungmu.
\end{enumerate}
Periksalah dengan langit: di daerah ini, \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari
pada bab \omterm{def:g3:sun-days-seasons:sundial}{jam matahari} menunjuk ke utara --- \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} dan jarumnya
seharusnya sependapat.
\end{method}

\begin{example}[Buatlah sendiri]\label{ex:g4:magnets-and-compass:make}
Gosokkan sebatang jarum jahit baja dengan satu \omterm{def:g4:magnets-and-compass:poles}{kutub} \omterm{def:g1:magnets:magnet}{magnet}, dua puluh
kali, selalu ke arah yang sama, dan angkatlah \omterm{def:g1:magnets:magnet}{magnetnya} di antara
gosokan: jarum itu menjadi \omterm{def:g1:magnets:magnet}{magnet} kecil. Letakkan pelan-pelan di atas
irisan gabus yang mengapung di baskom berisi air --- lalu amati
gabusnya berputar pelan sampai jarumnya menunjuk utara--selatan. Kamu
sudah membuat \omterm{def:g4:magnets-and-compass:compass}{kompas} pelaut seribu tahun yang lalu.
\end{example}

\begin{remark}[Alat yang membuka samudra]\label{rem:g4:magnets-and-compass:history}
Sebelum ada \omterm{def:g4:magnets-and-compass:compass}{kompas}, pelaut merapat ke pantai dan berdoa agar malam
cerah supaya dapat berkemudi dengan bintang. Dengan jarum yang
mengapung, mereka dapat menjaga haluan menembus kabut, badai, dan
gelap tanpa bulan --- menyeberangi samudra yang utuh. Sedikit alat
yang pernah menggambar ulang peta dunia sedemikian menyeluruh. Dan
satu teka-teki tetap terbuka selama berabad-abad: \emph{mengapa} Bumi
berlaku sebagai \omterm{def:g1:magnets:magnet}{magnet}? Kisah lengkapnya menunggu di tahun sarjana
pelajaran ini --- ia layak ditunggu.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:1}
Apa itu \omterm{def:g4:magnets-and-compass:poles}{kutub} sebuah \omterm{def:g1:magnets:magnet}{magnet}, dan apa namanya? Di bagian mana pada
\omterm{def:g1:magnets:magnet}{magnet} batang penjepit kertas berkerumun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:2}
Nyatakan aturan \omterm{def:g4:magnets-and-compass:poles}{kutub}. Dua \omterm{def:g1:magnets:magnet}{magnet} menempel rapat: \omterm{def:g4:magnets-and-compass:poles}{kutub} mana yang tadi
berhadapan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:3}
\omterm{def:g4:magnets-and-compass:poles}{Kutub} U sebuah \omterm{def:g1:magnets:magnet}{magnet} mendekati \omterm{def:g4:magnets-and-compass:poles}{kutub} U \omterm{def:g1:magnets:magnet}{magnet} yang lain. Apa yang
terjadi? Dan U yang mendekati S?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:4}
Mengapa \omterm{def:g1:magnets:magnet}{magnet} batang yang tergantung bebas selalu berhenti menunjuk
ke arah yang sama? Apa yang diungkapkan hal ini tentang Bumi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:5}
Sebutkan keempat \omterm{def:g4:magnets-and-compass:compass}{arah mata angin} berurutan, mulai dari Utara dan
berputar ke kanan. Menghadap utara, arah mana yang ada di sebelah
kananmu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:6}
Mengapa \omterm{def:g4:magnets-and-compass:compass}{kompas} harus dijauhkan dari \omterm{def:g1:magnets:magnet}{magnet}, pagar besi, dan mobil
selagi kamu membacanya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:7}
Ceritakan cara mengubah jarum jahit menjadi jarum \omterm{def:g4:magnets-and-compass:compass}{kompas}, dan cara
membiarkannya berayun bebas tanpa poros.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:8}
Dari bab \omterm{def:g3:sun-days-seasons:sundial}{jam matahari}: ke arah mana \omterm{def:g1:light-and-shadows:shadow}{bayang-bayang} tengah hari menunjuk
di daerah ini? Bagaimana hal itu dapat memeriksa sebuah \omterm{def:g4:magnets-and-compass:compass}{kompas} ---
atau menggantikan \omterm{def:g4:magnets-and-compass:compass}{kompas} yang hilang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:magnets-and-compass:9}
Mengapa \omterm{def:g4:magnets-and-compass:compass}{kompas} menjadi harta yang begitu berharga bagi pelaut,
dibandingkan dengan berkemudi mengikuti garis pantai atau bintang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:10}
Seorang pendaki berjalan menuju pondok gunung yang terletak tepat di
utara. Jarum kompasnya, yang dipegang dengan benar, menunjuk ke kepala
ikat pinggang temannya alih-alih ke utara setiap kali temannya berdiri
dekat di sebelah kirinya. Apa yang sedang terjadi, ke arah mana ia
terancam melenceng, dan apa perbaikan satu langkahnya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:11}
Sebuah \omterm{def:g1:magnets:magnet}{magnet} digergaji menjadi dua di antara kedua \omterm{def:g4:magnets-and-compass:poles}{kutubnya}, dengan
harapan mendapat U sendirian di satu tangan dan S sendirian di tangan
yang lain. Cerita para pelaut --- dan setiap percobaan yang pernah
dilakukan --- mengatakan bahwa tiap belahannya ternyata menjadi \omterm{def:g1:magnets:magnet}{magnet}
kecil yang lengkap dengan \emph{kedua} \omterm{def:g4:magnets-and-compass:poles}{kutubnya}. Ujilah pernyataan itu
dalam pikiran dengan resep jarum \omterm{def:g4:magnets-and-compass:compass}{kompas} tadi: jarum jahit yang digosok
itu pendek --- apakah ia punya satu \omterm{def:g4:magnets-and-compass:poles}{kutub} atau dua? Apa yang
disarankan hal ini tentang berburu \omterm{def:g4:magnets-and-compass:poles}{kutub} tunggal dengan cara memotong?
\end{exercise}
