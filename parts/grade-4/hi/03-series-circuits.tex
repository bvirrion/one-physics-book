\chapter{श्रेणी परिपथ}\label{ch:g4:series-circuits}

पिछले साल सजावटी बत्तियों की एक लड़ी सिर्फ़ एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} के ग़ायब होने से
अँधेरी हो गई थी --- और हमने वादा किया था कि उस तरतीब का एक नाम है।
वह नाम यह है: \omterm{def:g4:series-circuits:series}{श्रेणी परिपथ}, यानी एक ही फंदे पर कई खिलाड़ी बिठाने का
सबसे सीधा तरीक़ा। इसके घर के नियम सख़्त हैं, और तुम्हारी दराज़ में रखी
टॉर्च उन्हें हमेशा से मानती आई है।

\section{एक फंदा, कई खिलाड़ी}

\begin{definition}[श्रेणी में]\label{def:g4:series-circuits:series}
किसी परिपथ के खिलाड़ी \emph{श्रेणी में}\index{श्रेणी परिपथ} तब होते
हैं जब वे एक ही फंदे पर, एक के बाद एक बैठे हों, जैसे एक ही माला में
पिरोए मनके: धारा की सड़क पहले पहले खिलाड़ी से, फिर दूसरे से, फिर तीसरे
से गुज़रती है, और कहीं भी दो राहें नहीं फूटतीं। इस तरह बना परिपथ
\emph{श्रेणी परिपथ} कहलाता है।
\end{definition}

\begin{tikzpicture}[scale=0.9]
  \draw (0,0) to[battery1, l=बैटरी] (0,2.6)
    to[short] (1.6,2.6)

{\small \omterm{def:g4:series-circuits:series}{श्रेणी परिपथ}: \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, \omterm{ex:g3:first-electric-circuit:switch}{स्विच} और दो \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}, सब एक ही अकेले फंदे पर
--- हर खिलाड़ी उसी एक माला में पिरोया हुआ।}
\end{omfigure}

\begin{method}[इसे बनाना]\label{met:g4:series-circuits:build}
एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, दो \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}, एक \omterm{ex:g3:first-electric-circuit:switch}{स्विच} और तारों से:
\begin{enumerate}
  \item \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $+$ सिरे से एक तार \omterm{ex:g3:first-electric-circuit:switch}{स्विच} तक ले जाकर शुरू करो;
  \item \omterm{ex:g3:first-electric-circuit:switch}{स्विच} से पहले \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} तक, और पहले \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} से दूसरे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} तक तार
        जोड़ो;
  \item फंदा बंद करो: दूसरे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} से वापस \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $-$ सिरे तक;
  \item \omterm{ex:g3:first-electric-circuit:switch}{स्विच} दबाओ और दोनों बल्बों को एक साथ देखो।
\end{enumerate}
चालू करने से पहले उँगली से फंदे का पीछा करो: अगर तुम्हारी उँगली को
कहीं भी दो रास्तों में से चुनना पड़े, तो वह \omterm{def:g4:series-circuits:series}{श्रेणी परिपथ} नहीं है।
\end{method}

\section{घर के नियम}

\begin{proposition}[अकेले फंदे के नियम]\label{prop:g4:series-circuits:rules}
\omterm{def:g4:series-circuits:series}{श्रेणी परिपथ} में:
\begin{enumerate}
  \item \emph{सब या कुछ नहीं}: कहीं भी एक दरार --- खुला \omterm{ex:g3:first-electric-circuit:switch}{स्विच}, टूटा
        \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}, ढीला तार --- और हर खिलाड़ी उसी पल रुक जाता है;
  \item \emph{क्रम मायने नहीं रखता}: \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} आपस में बदल दो, \omterm{ex:g3:first-electric-circuit:switch}{स्विच} को
        फंदे की दूसरी ओर ले जाओ --- सब कुछ ठीक पहले जैसा ही चलता
        है;
  \item \emph{बँटवारा}: फंदे पर जितने ज़्यादा \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}, हर एक उतना ही
        मंद जलता है; \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का धक्का पूरी माला में बँट जाता है।
\end{enumerate}
\end{proposition}

\begin{example}[नियमों को देखना]\label{ex:g4:series-circuits:seeing}
दो बल्बों वाला परिपथ बनाओ और हर नियम जाँचो। \emph{कोई भी} एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}
खोल दो: दोनों बुझ जाते हैं --- नियम एक (खुला ख़ाली सॉकेट एक दरार
है)। दोनों \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} आपस में बदल दो, या \omterm{ex:g3:first-electric-circuit:switch}{स्विच} को उनके बीच ले जाओ: ज़रा भी
फ़र्क़ नहीं --- नियम दो। अब उसी परिपथ को एक ही \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} से बनाओ: वह अपने
साथी के साथ जितना जलता था, उससे साफ़ ज़्यादा चमकता है --- नियम तीन,
उलटी दिशा से।
\end{example}

\begin{example}[सजावटी बत्तियों की गुत्थी सुलझी]\label{ex:g4:series-circuits:partylights}
पुरानी पहेली अब पहेली रही ही नहीं: सजावटी \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} \omterm{def:g4:series-circuits:series}{श्रेणी में} लगे होते
हैं, यानी एक ही माला में दर्जनों मनके। एक ग़ायब \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} एक दरार है, और
नियम एक पूरी लड़ी को अँधेरा कर देता है। एक-एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} बदलकर दोषी \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}
ढूँढ़ने ने पीढ़ियों को नियम एक कठिन तरीक़े से सिखाया --- आजकल की
लड़ियाँ अलग ढंग से बनती हैं, जैसा अगला साल दिखाएगा।
\end{example}

\begin{remark}[स्विच सबको हुक्म देता है]\label{rem:g4:series-circuits:switch}
नियम दो का एक काम का नतीजा है: फंदे पर कहीं भी लगा अकेला \omterm{ex:g3:first-electric-circuit:switch}{स्विच} उस पर
बैठे हर खिलाड़ी को हुक्म देता है। इसीलिए दीवार का एक \omterm{ex:g3:first-electric-circuit:switch}{स्विच} छत की
बत्तियों की पूरी क़तार सँभाल सकता है --- और इसीलिए, उसी वजह से, तुम
श्रेणी की लड़ी का सिर्फ़ \emph{एक} \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} बुझा नहीं सकते, बाक़ी सबको
मारे बिना।
\end{remark}

\section{माला में और बैटरियाँ}

\begin{example}[नाक से पूँछ]\label{ex:g4:series-circuits:batteries}
बैटरियाँ भी \omterm{def:g4:series-circuits:series}{श्रेणी में} क़तार बना सकती हैं --- पर वे बल्बों से
ज़्यादा नखरैली हैं: उन्हें \emph{नाक से पूँछ} जोड़कर खड़ा होना पड़ता
है, यानी हर \omterm{def:g3:first-electric-circuit:battery}{बैटरी} का $+$ अगली के $-$ से लगा हुआ। इस तरह क़तार में लगी
दो बैटरियाँ मिलकर धकेलती हैं, और \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के मुक़ाबले साफ़
ज़्यादा चमकता है। एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी} को घुमा दो --- नाक से नाक --- और दोनों
धक्के आपस में लड़ पड़ते हैं: \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} कुछ भी नहीं देता।
\end{example}

\begin{tikzpicture}[scale=0.8]

      to[short] (3.6,3.6)
      to[lamp, l=ज़्यादा मंद] (3.6,0)
      to[short] (0,0);
  \end{scope}
\end{tikzpicture}

{\small नाक से पूँछ जोड़कर क़तार में लगी दो बैटरियाँ मिलकर धकेलती
हैं: वही \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} अकेली \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के मुक़ाबले ज़्यादा चमकता है।}
\end{omfigure}

\begin{example}[टॉर्च के भीतर]\label{ex:g4:series-circuits:flashlight}
टॉर्च खोलो: बैटरियाँ नली में एक के पीछे एक सरककर बैठती हैं ---
यानी पहले से तैयार नाक-से-पूँछ वाली क़तार --- और फिर \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} तथा बटन वाला
\omterm{ex:g3:first-electric-circuit:switch}{स्विच} खोल के रास्ते एक ही अकेला फंदा पूरा कर देते हैं। टॉर्च असल में
जेब में रखा \omterm{def:g4:series-circuits:series}{श्रेणी परिपथ} \emph{है}। और अब तुम यह भी जानते हो कि एक भी
\omterm{def:g3:first-electric-circuit:battery}{बैटरी} उलटी डालने पर वह अँधेरी क्यों रह जाती है: नाक से नाक होने पर
धक्के एक-दूसरे को काट देते हैं।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:1}
``\omterm{def:g4:series-circuits:series}{श्रेणी में}'' का क्या मतलब है? अध्याय इसके लिए रोज़मर्रा का कौन-सा
चित्र काम में लेता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:2}
\omterm{def:g4:series-circuits:series}{श्रेणी परिपथ} के तीनों घरेलू नियम बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:3}
दो बल्बों वाले परिपथ में \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} 2 खोल दिया जाता है। \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} 1 का क्या
होगा, और क्यों? यहाँ कौन-सा नियम काम कर रहा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:4}
ज़ो \omterm{ex:g3:first-electric-circuit:switch}{स्विच} को \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} 1 से पहले वाली जगह से हटाकर \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} 2 के बाद लगा देती
है। परिपथ के बरताव में क्या बदलता है? कौन-सा नियम ऐसा कहता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:5}
एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी} एक \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} को अच्छी तरह जलाती है। इनकी चमक का अंदाज़ा लगाओ:
\omterm{def:g4:series-circuits:series}{श्रेणी में} दो \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}; \ \omterm{def:g4:series-circuits:series}{श्रेणी में} तीन \omterm{def:g3:first-electric-circuit:bulb}{बल्ब}। बँट क्या रहा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:6}
मिलकर धकेलने के लिए दो बैटरियों को कैसे खड़ा होना चाहिए? हर \omterm{def:g3:first-electric-circuit:battery}{बैटरी} पर
$+$ और $-$ के निशान लगाकर क़तार बनाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:7}
टॉर्च के फंदे का पीछा करो: \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के $+$ से वापस उसके $-$ तक धारा जिन
खिलाड़ियों से मिलती है, उनके नाम बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:8}
दरवाज़े की घंटी वाले परिपथ में एक \omterm{def:g3:first-electric-circuit:battery}{बैटरी}, एक बटन और घंटी \omterm{def:g4:series-circuits:series}{श्रेणी में}
लगे हैं। दूसरा बटन कहाँ जोड़ा जाए ताकि घंटी सिर्फ़ तभी बजे जब
\emph{दोनों} बटन दबे हों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:series-circuits:9}
\omterm{def:g4:series-circuits:series}{श्रेणी में} लगी दस सजावटी बत्तियों की लड़ी अँधेरी है। सारे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} देखने
में ठीक लगते हैं। एक भले-चंगे \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} की मदद से टूटा हुआ \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} ढूँढ़ने की
धैर्य वाली योजना बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:series-circuits:10}
टॉम की टॉर्च में दो बैटरियाँ लगती हैं और वह ख़ूब चमकती है। अँधेरे में
बैटरियाँ बदलने के बाद वह कुछ भी नहीं देती --- जबकि दोनों नई बैटरियाँ
भरी हुई हैं, \omterm{def:g3:first-electric-circuit:bulb}{बल्ब} ठीक है, और कोई तार ढीला नहीं। टॉम ने क्या किया, और
टॉर्च अँधेरी क्यों रह जाती है?
(\cref{ex:g4:series-circuits:batteries} देखो।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:series-circuits:11}
एक अभियंता को किसी मशीन के लिए आपात स्थिति वाला एक रोक-स्विच लगाना
है; मशीन की मोटर अपनी \omterm{def:g3:first-electric-circuit:battery}{बैटरी} के साथ एक ही फंदे पर बैठी है। वह कहती है:
``फंदे पर कहीं भी लगा दो, चल जाएगा।'' क्या वह सही है? और अगर मशीन के
परिपथ में राहें फूटती होतीं तो उसकी योजना क्यों नाकाम होती? (अगले साल
के अध्याय में उन राहों का नाम है।)
\end{exercise}
