\chapter{भार और द्रव्यमान}\label{ch:g4:weight-and-mass}

\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर अंतरिक्ष-यात्री भारी थैले पीठ पर लादे होने के बावजूद
ट्रैंपोलिन पर कूदते बच्चों की तरह उछलते थे। वहाँ उनके थैले भारी थे या
हल्के? ठीक उत्तर देने के लिए हमें रोज़मर्रा के एक शब्द को दो हिस्सों
में बाँटना होगा: कोई चीज़ \emph{किससे बनी है} --- उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} ---
और पृथ्वी \emph{उसे कितने ज़ोर से खींचती है} --- उसका \omterm{def:g4:weight-and-mass:weight}{भार}। ये दोनों
क़रीबी भाई-बंद हैं, पर जुड़वाँ नहीं।

\section{``भारी'' में छिपे दो विचार}

\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} तुम जानते हो: कितना पदार्थ, \omterm{def:g3:measuring-mass:units}{ग्राम} और \omterm{def:g3:measuring-mass:units}{किलोग्राम} में, और
फ़ैसला \omterm{def:g3:levers-and-scales:scale}{तुला} का। और पृथ्वी का खिंचाव भी तुम जानते हो: वह छुए बिना लगने
वाला \omterm{def:g2:pushes-and-pulls:force}{बल} जो छोड़ी गई हर चीज़ को नीचे ले आता है। \omterm{def:g4:weight-and-mass:weight}{भार} वहीं है जहाँ ये
दोनों मिलते हैं।

\begin{definition}[भार]\label{def:g4:weight-and-mass:weight}
किसी चीज़ का \emph{भार}\index{भार} वह \omterm{def:g2:pushes-and-pulls:force}{बल} है जिससे पृथ्वी उसे नीचे की
ओर खींचती है। बालटी थामे रखते समय तुम्हारी बाँह जिससे लड़ती है वही
भार है, और बालटी लटकाने पर स्प्रिंग को जो तानता है वह भी। हर \omterm{def:g2:pushes-and-pulls:force}{बल} की
तरह भार की भी एक दिशा है: हमेशा सीधे नीचे।
\end{definition}

\begin{proposition}[ज़्यादा पदार्थ, ज़्यादा तेज़ खिंचाव]\label{prop:g4:weight-and-mass:morestuff}
जिन चीज़ों में ज़्यादा पदार्थ होता है, पृथ्वी उन्हें ज़्यादा ज़ोर से
खींचती है: एक ही जगह पर ज़्यादा \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} का मतलब हमेशा ज़्यादा \omterm{def:g4:weight-and-mass:weight}{भार}
होता है --- एक जैसी दो ईंटों का \omterm{def:g4:weight-and-mass:weight}{भार} एक ईंट से दोगुना होता है। इसीलिए
\omterm{def:g3:levers-and-scales:scale}{तुला}, जो अपने दोनों पलड़ों पर लगे \emph{खिंचावों} की तुलना करती है,
असल में \emph{द्रव्यमानों} की तुलना कर पाती है: एक ही जगह पर बराबर
खिंचाव का मतलब बराबर पदार्थ है।
\end{proposition}

\begin{example}[भाई-बंद, जुड़वाँ नहीं]\label{ex:g4:weight-and-mass:cousins}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} इसका उत्तर देता है: \emph{कितना पदार्थ?} वह अकेली चीज़ का
गुण है --- तुम्हारे हाथ में भी वही, पानी के भीतर भी वही, और \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}
पर भी वही। \omterm{def:g4:weight-and-mass:weight}{भार} इसका उत्तर देता है: \emph{यहाँ इसे कितने ज़ोर से नीचे
खींचा जा रहा है?} वह चीज़ पर \emph{और} इस पर निर्भर करता है कि वह
कहाँ है। सिर्फ़ जगह बदलो, और कुछ नहीं: \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} वही रहता है; \omterm{def:g4:weight-and-mass:weight}{भार} बदल
सकता है।
\end{example}

\section{स्प्रिंग तुला}

\begin{definition}[स्प्रिंग तुला]\label{def:g4:weight-and-mass:springscale}
\emph{स्प्रिंग तुला}\index{स्प्रिंग तुला} \omterm{def:g4:weight-and-mass:weight}{भार} को स्प्रिंग तानने
देकर मापती है: खिंचाव जितना ज़ोरदार, फैलाव उतना ही लंबा। तनी हुई
स्प्रिंग के बग़ल में लगी सुई पाठ दिखा देती है। मछुआरों की \omterm{def:g3:levers-and-scales:scale}{तुला},
सामान तौलने वाली \omterm{def:g3:levers-and-scales:scale}{तुला} और रसोई की तराज़ू का भीतरी हिस्सा --- सब इसी तरह
काम करते हैं: वे पदार्थ नहीं, \emph{खिंचाव} महसूस करते हैं।
\end{definition}

\begin{method}[रबर बैंड की तुला]\label{met:g4:weight-and-mass:rubberband}
अपना ख़ुद का भार-महसूस करने वाला यंत्र बनाओ:
\begin{enumerate}
  \item किसी हुक से एक रबर बैंड लटकाओ, और उसके नीचे टेप से काग़ज़ का
        हल्का कप चिपका दो;
  \item कप ख़ाली रहते हुए उसके किनारे की जगह पर निशान लगाओ, और अभी
        कुछ मत मापो: यही शून्य निशान है;
  \item कप में बोझ डालो --- मान लो $10$ काँच की गोलियाँ --- और रूलर
        से मापो कि किनारा शून्य निशान से कितना नीचे धँसा;
  \item अब $20$ गोलियाँ डालकर फिर मापो।
\end{enumerate}
गोलियाँ दोगुनी, फैलाव भी लगभग दोगुना: तुम्हारा रबर बैंड उस
प्रतिज्ञप्ति की पुष्टि कर देता है --- ज़्यादा पदार्थ, ज़्यादा तेज़
खिंचाव।
\end{method}

\section{चंद्रमा की यात्रा}

\begin{example}[चंद्रमा कमज़ोर खींचता है]\label{ex:g4:weight-and-mass:moon}
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पृथ्वी से बहुत छोटा है, और वह चीज़ों को कहीं ज़्यादा कमज़ोर
खींचता है --- लगभग छह गुना कमज़ोर। अंतरिक्ष-यात्री का थैला यात्रा में
अपने \emph{\omterm{def:g3:measuring-mass:mass}{द्रव्यमान}} का एक-एक \omterm{def:g3:measuring-mass:units}{ग्राम} साथ रखता है, पर वहाँ उसका
\emph{\omterm{def:g4:weight-and-mass:weight}{भार}} घटकर छठा हिस्सा रह जाता है। उछलने का पूरा राज़ यही है:
पृथ्वी के खिंचाव से लड़ने की सधी हुई टाँगें अचानक \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} के हल्के
खिंचाव से लड़ने लगती हैं, और हर क़दम छलाँग बन जाता है।
\end{example}

\begin{example}[चंद्रमा पर दो तुलाएँ]\label{ex:g4:weight-and-mass:twoscales}
दोनों तुलाएँ \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर ले जाओ और वही थैला तौलो। \emph{स्प्रिंग} वाली
\omterm{def:g3:levers-and-scales:scale}{तुला} जगह के बारे में सच बोलती है: वह छह गुना कम बताती है --- खिंचाव
सचमुच कमज़ोर है। दो-पलड़े वाली \omterm{def:g3:levers-and-scales:scale}{तुला} को कुछ पता ही नहीं चलता: \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}
थैले और संदर्भ बाटों, दोनों को \emph{बराबर} कमज़ोर खींचता है, पलड़े
फिर भी बराबरी पर आ जाते हैं, और वह वही \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बताती है जो पृथ्वी पर
था। स्प्रिंगें जगह महसूस करती हैं; तुलाएँ पदार्थ की तुलना करती हैं।
\end{example}

\begin{remark}[रोज़ के शब्द, भौतिकी के शब्द]\label{rem:g4:weight-and-mass:words}
दुकान पर ``ख़रबूज़े का वज़न एक \omterm{def:g3:measuring-mass:units}{किलोग्राम} है'' कहना बेकसूर बातचीत है
--- पृथ्वी पर \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} और \omterm{def:g4:weight-and-mass:weight}{भार} हर जगह साथ-साथ चलते हैं, इसलिए किसी को
भ्रम नहीं होता। भौतिकी इन शब्दों को इसलिए बाँटती है कि वह दुकान से
आगे की सोचती है: अंतरिक्ष-यात्रियों के लिए, गिरती चीज़ों के लिए, और
--- किसी आगे वाले \omterm{def:g3:sun-days-seasons:year}{वर्ष} में --- बलों को उनके अपने ठीक \omterm{def:g2:measuring-length:measure}{मात्रक} में मापने
के लिए, जिसका नाम उस वैज्ञानिक पर पड़ेगा जिसने पृथ्वी के खिंचाव का
नियम सबसे पहले लिखा।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:1}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} किस सवाल का उत्तर देता है? \omterm{def:g4:weight-and-mass:weight}{भार} किस सवाल का? इनमें से \omterm{def:g2:pushes-and-pulls:force}{बल}
कौन है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:2}
\omterm{def:g4:weight-and-mass:weight}{भार} हमेशा किस ओर खींचता है? कौन-सा उपकरण फैलकर \omterm{def:g4:weight-and-mass:weight}{भार} महसूस करता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:3}
एक ईंट \omterm{def:g4:weight-and-mass:springscale}{स्प्रिंग तुला} की स्प्रिंग को दो उँगली भर तान देती है। एक जैसी
दो ईंटें उसे कितना तानेंगी? कौन-सी प्रतिज्ञप्ति ऐसा कहती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:4}
\cref{met:g4:weight-and-mass:rubberband} वाली रबर बैंड की \omterm{def:g3:levers-and-scales:scale}{तुला} में
बोझ डालने से पहले ख़ाली कप की जगह पर निशान क्यों लगाना पड़ता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:5}
एक अंतरिक्ष-यात्री \qty{20}{kg} पदार्थ वाला थैला \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर ले जाता
है। वहाँ उसका \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} अब भी \qty{20}{kg} है? क्या उसे थामे रखना
पृथ्वी जितना ही कठिन है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:6}
भारी थैले लादे होने पर भी अंतरिक्ष-यात्री \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर इतनी ऊँची छलाँग
कैसे लगा सके?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:7}
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} पर कौन-सी \omterm{def:g3:levers-and-scales:scale}{तुला} अपना पाठ बदल देती है --- \omterm{def:g4:weight-and-mass:springscale}{स्प्रिंग तुला} या
दो-पलड़े वाली \omterm{def:g3:levers-and-scales:scale}{तुला}? बताओ कि हर एक असल में किसकी तुलना करती है या

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:8}
सही है या नहीं: ``जिस चीज़ में पदार्थ ही न हो, उसका \omterm{def:g4:weight-and-mass:weight}{भार} कहीं भी नहीं

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:weight-and-mass:9}
सब्ज़ी वाली जैसे-जैसे पलड़े में सेब डालती है, उसकी लटकी हुई \omterm{def:g3:levers-and-scales:scale}{तुला}
नीचे धँसती जाती है। उसकी \omterm{def:g3:levers-and-scales:scale}{तुला} दोनों भाई-बंदों में से किसे महसूस कर
रही है? फिर भी उसकी तरकीब सेबों के पूरे \omterm{def:g3:measuring-mass:units}{किलोग्राम} क्यों बेच देती है
--- पृथ्वी पर?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:weight-and-mass:10}
किसी अंतरिक्ष संस्था को जाँचना है कि उसके यान के पुरज़ों का \omterm{def:g3:measuring-mass:mass}{द्रव्यमान}
ठीक-ठीक सही है --- पर तैयार यान किसी छोटे और दूर के ग्रह के पास काम
करेगा, जिसका खिंचाव बहुत कमज़ोर है। एक अभियंता चिंता करता है:
``पृथ्वी पर की गई हमारी सारी तौलें वहाँ ग़लत निकलेंगी!'' उसे तसल्ली
दो: कौन-से मापन बिना बदले सफ़र करते हैं, और उन्हें किस उपकरण ने किया
था?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:weight-and-mass:11}
अंतरिक्ष की गहराई में, पृथ्वी, \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} और सूर्य --- सबसे बहुत दूर,
एक अंतरिक्ष-यात्री पाना छोड़ देती है: वह उसके हाथ के पास तैरता रहता
है, भार-रहित। क्या पाने ने अपना \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} खो दिया? रोज़मर्रा की तरह के
दो ऐसे सुराग़ बताओ जिन्हें वह अब भी देख सकती है और जो दिखा दें कि
पदार्थ पूरा का पूरा वहीं है। (सुराग़: उसे हिलाकर देखो, या फेंककर।)
\end{exercise}
