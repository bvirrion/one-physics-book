\chapter{प्रकाश सीधी रेखाओं में चलता है}\label{ch:g4:light-straight-lines}

सुबह की धुंध में पेड़ों के बीच से आती सूर्य की किरणें तिरछी उतरती हैं
--- और हर एक तनी हुई डोरी जितनी सीधी। धूल भरी अटारी में टॉर्च
\omterm{def:g1:light-and-shadows:source}{प्रकाश} का सीधे किनारों वाला फल फेंकती है। मोड़ के उस पार से पुकारते
साथी को तुम सुन तो लेते हो, पर वहाँ \emph{देख} नहीं पाते। ये सब एक ही
नियम के अलग-अलग लिबास हैं --- \omterm{def:g1:light-and-shadows:source}{प्रकाश} का सबसे पुराना नियम।

\section{सीधी रेखा का नियम}

\begin{proposition}[प्रकाश सीधी रेखाओं में चलता है]\label{prop:g4:light-straight-lines:law}
\omterm{def:g2:air-around-us:air}{हवा} में (और पानी, काँच या ख़ाली अंतरिक्ष में भी) \omterm{def:g1:light-and-shadows:source}{प्रकाश} सीधी रेखाओं
में चलता है। वह मोड़ों के चारों ओर मुड़ता नहीं, धुएँ की तरह बहता
नहीं, और दीवार के पीछे तुम्हें ढूँढ़ने के लिए झुकता नहीं। जहाँ स्रोत
से आती कोई सीधी रेखा पहुँच ही न सके, वहाँ उस स्रोत का कोई \omterm{def:g1:light-and-shadows:source}{प्रकाश} नहीं
पहुँचता।
\end{proposition}

\begin{definition}[प्रकाश किरण]\label{def:g4:light-straight-lines:ray}
\omterm{def:g1:light-and-shadows:source}{प्रकाश} बनाने के लिए भौतिकी वाले \emph{प्रकाश किरण}\index{प्रकाश
किरण} काम में लेते हैं: तीर लगी एक सीधी रेखा, जो \omterm{def:g1:light-and-shadows:source}{प्रकाश} का एक धागा
और उसके चलने की दिशा दिखाती है। टॉर्च के किरणपुंज को किरणों के पंखे
की तरह बनाया जाता है; लेज़र के पतले पुंज को एक अकेली किरण की तरह। \omterm{def:g2:pushes-and-pulls:force}{बल}
वाले तीर की तरह किरण भी एक नन्हा चित्र है जो अपने भीतर एक नियम लिए
चलता है: \omterm{def:g1:light-and-shadows:source}{प्रकाश} की सड़कें रूलर जितनी सीधी होती हैं।
\end{definition}

और जंगल भले मुड़ें, \omterm{def:g1:light-and-shadows:source}{प्रकाश} नहीं मुड़ेगा।
\end{example}

\section{तीन कार्डों से इसे साबित करना}

\begin{method}[तीन कार्ड वाला प्रयोग]\label{met:g4:light-straight-lines:cards}
\begin{enumerate}
  \item कड़े गत्ते के तीन कार्डों के बीचोंबीच, एक ही ऊँचाई पर,
        छोटा-सा छेद करो;
  \item तीनों कार्ड एक क़तार में, एक बालिश्त की दूरी पर खड़े करो, और
        सबसे आख़िरी के उस पार एक जलती मोमबत्ती (या टॉर्च) रखो;
  \item अपना सिर तब तक हिलाओ जब तक तीनों छेदों से होकर लौ न दिखने
        लगे --- यह तभी संभव है जब तीनों छेद और लौ एक ही सीधी रेखा
        पर हों (जाँचो: छेदों में से गुज़री तनी हुई डोरी लौ तक पहुँच
        जाती है);
  \item अब बीच वाले कार्ड को बग़ल की ओर सिर्फ़ एक उँगली भर सरका दो।
\end{enumerate}
लौ ग़ायब हो जाती है। जो \omterm{def:g1:light-and-shadows:source}{प्रकाश} टेढ़ा-मेढ़ा चल सकता था --- पहले छेद से
भीतर, ज़रा बग़ल में सरककर, सरकाए हुए छेद से पार, और फिर तुम्हारी आँख
तक --- वह ऐसा करता ही नहीं: सीधी रेखा नहीं, तो \omterm{def:g1:light-and-shadows:source}{प्रकाश} नहीं।
\end{method}

\begin{example}[छाया का राज़ हमेशा के लिए खुला]\label{ex:g4:light-straight-lines:shadows}
अब स्कूल के तुम्हारे पहले साल वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} अपना राज़ खोल देती है।
किरणें लैंप से सीधी दौड़ती हैं; गेंद उन किरणों को रोक लेती है जो उस
पर पड़ती हैं; और उसके किनारे से बाल-बाल बचकर निकलती किरणें सीधी आगे
चली जाती हैं। गेंद के पीछे ठीक वही जगह पड़ती है जहाँ कोई सीधी किरण
पहुँच ही नहीं सकती --- वही \omterm{def:g1:light-and-shadows:shadow}{छाया} है, और उसका तीखा किनारा उन आख़िरी
किरणों ने खींचा है जो किनारे से सरककर निकल गईं। सीधी किरणें, तीखी
छायाएँ: टेढ़ा \omterm{def:g1:light-and-shadows:source}{प्रकाश} हर \omterm{def:g1:light-and-shadows:shadow}{छाया} को धुंध में बदल देता।
\end{example}

\section{देखना यानी प्रकाश पाना}

\begin{proposition}[देखना काम कैसे करता है]\label{prop:g4:light-straight-lines:seeing}
तुम्हें कोई चीज़ तब दिखती है जब उससे आया \omterm{def:g1:light-and-shadows:source}{प्रकाश} सीधी रेखा में
तुम्हारी आँख के भीतर पहुँचता है --- स्रोत हो तो उसका अपना \omterm{def:g1:light-and-shadows:source}{प्रकाश},
वरना लौटाया हुआ सूर्य-प्रकाश या लैंप का \omterm{def:g1:light-and-shadows:source}{प्रकाश}। दुनिया को पकड़ने के
लिए तुम्हारी आँख से कुछ भी बाहर नहीं जाता: देखना \emph{पाना} है, यानी
किरणों को वैसे ही लपकना जैसे जाल गेंदें लपकता है।
\end{proposition}

\begin{example}[मोड़ के उस पार]\label{ex:g4:light-straight-lines:corner}
तुम्हारा साथी स्कूल के मोड़ के उस पार से पुकारता है: तुम उसे सुन लेते
हो, पर देख नहीं पाते। \omterm{def:g3:sound-around-us:sound}{ध्वनि} तालाब के छल्लों की तरह फैलती है और मोड़
के किनारे से लिपटकर आगे बह जाती है; \omterm{def:g1:light-and-shadows:source}{प्रकाश} अपनी सीधी सड़कें पकड़े
रहता है, और ईंट में से होकर तुम्हारे साथी के चेहरे को तुम्हारी आँख से
जोड़ने वाली कोई सीधी रेखा है ही नहीं। मोड़ के उस पार देखने के लिए
तुम्हें \emph{सड़क मोड़नी} पड़ेगी --- यानी किरण को टकराकर लौटने के
लिए कोई कठोर, चमकीली सतह देनी होगी। इस तरकीब का नाम दर्पण है, और वह
अगले साल के पहले प्रकाश-अध्याय की मालिक है।
\end{example}

\begin{remark}[दौड़ का चैंपियन]\label{rem:g4:light-straight-lines:speed}
\omterm{def:g3:sound-around-us:sound}{ध्वनि}, जैसा हमने मापा, हर सेकंड \qty{340}{m} की चाल से दौड़ती है।
अपनी सीधी सड़कों पर \omterm{def:g1:light-and-shadows:source}{प्रकाश} बिलकुल दूसरे दर्जे का खिलाड़ी है: वह पूरे
आकाश को इतनी तेज़ी से पार कर जाता है कि कोई गिनती उसे पकड़ ही नहीं
सकती --- इसीलिए कौंध ``उसी पल'' पहुँच जाती है और गड़गड़ाहट सेकंडों
पीछे घिसटती आती है। ठीक-ठीक कितनी तेज़? वह संख्या इतनी बेतुकी है कि
हम उसे किसी आगे वाले \omterm{def:g3:sun-days-seasons:year}{वर्ष} के लिए बचा रखते हैं --- उन चतुर तरकीबों की
कहानी के साथ, जिनसे उसे मापा जा सका।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:1}
इस अध्याय का नियम बताओ। बाहर दिखने वाले उसके दो नज़ारे बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:2}
\omterm{def:g4:light-straight-lines:ray}{प्रकाश किरण} क्या है, और उसका चित्र कौन-सी दो बातें दिखाता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:3}
तीन कार्ड वाले प्रयोग में लौ ठीक कब दिखती है? कौन-सा एक ही काम उसे
ग़ायब कर देता है, और क्यों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:4}
सीधी रेखा का नियम \omterm{def:g1:light-and-shadows:shadow}{छाया} के तीखे किनारे की व्याख्या कैसे करता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:5}
अभी इसी पल तुम्हें यह पुस्तक दिख रही है। इसे संभव बनाने वाले \omterm{def:g1:light-and-shadows:source}{प्रकाश}
की यात्रा बताओ, लैंप से तुम्हारी आँख तक।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:6}
मोड़ के उस पार खड़े साथी को तुम सुन क्यों लेते हो पर देख क्यों नहीं
पाते? मोड़ पर दोनों यात्री --- \omterm{def:g3:sound-around-us:sound}{ध्वनि} और \omterm{def:g1:light-and-shadows:source}{प्रकाश} --- क्या करते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:7}
एक चाबी का छेद, उसके उस पार जगमगाता कमरा, और तुम्हारी आँख: अँधेरे
गलियारे में किन-किन जगहों से तुम्हें छेद में से लैंप दिख सकता है?
सीधी रेखा से उत्तर दो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:8}
दूर छूटी आतिशबाज़ी से पहले क्या पहुँचता है, धमाका या कौंध? इस अंतर
की व्याख्या इस साल का कौन-सा नियम और कौन-सा मापन करते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:light-straight-lines:9}
``जब मैं किसी तारे को देखता हूँ, तो मेरी आँख से कुछ निकलकर उसे छूने
जाता है।'' \cref{prop:g4:light-straight-lines:seeing} से इस पुरानी
धारणा को सुधारो --- सफ़र क्या करता है, और किस ओर?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:light-straight-lines:10}
किरणों का चित्र बनाकर ऐसा तरीक़ा सोचो जिससे तुम अपने कमरे के भीतर से
गलियारा देख सको, और \cref{ex:g4:light-straight-lines:corner} में
जिस दर्पण वाली तरकीब की झलक दी गई है उसका उपयोग करो: चमकीली सतह
दरवाज़े पर कहाँ खड़ी करनी होगी, और वह किरण की सड़क का क्या करती है?
(पनडुब्बियाँ ठीक यही करती हैं, दो बार एक के ऊपर एक, ताकि लहरों के
ऊपर देख सकें।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:light-straight-lines:11}
धुंध भरी किसी सुबह पेड़ों के बीच सूर्य-किरणें साफ़ सीधी रेखाओं में
फैली दिखती हैं --- फिर भी तुम्हारा साथी कहता है: ``किरणें दिख रही
