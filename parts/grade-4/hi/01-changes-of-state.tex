\chapter{पिघलना और उबलना: अवस्था के बदलाव}\label{ch:g4:changes-of-state}

तुम वर्षों से जानते हो कि गरमी बर्फ़ पिघलाती है और पानी उबालती है। पर
अब तुम्हारे पास \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} है --- और \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} के पास \omterm{def:g2:ice-water-steam:meltfreeze}{पिघलती} बर्फ़ की
बालटी के भीतर से एक चौंकाने वाली ख़बर है: जब तक बर्फ़ \omterm{def:g2:ice-water-steam:meltfreeze}{पिघलती} रहती है,
\omterm{def:g3:temperature-thermometers:temperature}{ताप} \emph{हिलने से इनकार कर देता है}। आओ, अवस्था के उन्हीं पुराने
बदलावों को नई आँखों से देखें।

\section{सही-सही नाम}

पानी की अवस्था के बदलावों को अब उनके बड़े नाम मिलते हैं, जो सिर्फ़
पानी के लिए नहीं बल्कि हर पदार्थ के लिए चलते हैं। \omterm{def:g3:solids-liquids-gases:solid}{ठोस} से \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}:
\emph{गलन}\index{गलन}। \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} से वापस \omterm{def:g3:solids-liquids-gases:solid}{ठोस}:
\emph{हिमीकरण}\index{हिमीकरण} (जिसे ठोसीकरण भी कहते हैं)। \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} से
\omterm{def:g3:solids-liquids-gases:gas}{गैस}: बड़े-बड़े बुलबुलों के साथ तेज़ी से हो तो
\emph{क्वथन}\index{क्वथन}, और सतह से चुपचाप हो तो \omterm{def:g2:ice-water-steam:evapcond}{वाष्पन}। \omterm{def:g3:solids-liquids-gases:gas}{गैस} से
वापस \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}: \omterm{def:g2:ice-water-steam:evapcond}{संघनन}। दुनिया का हर पदार्थ --- पानी, चॉकलेट, लोहा, सोना
--- यह खेल खेल सकता है, हर एक अपने-अपने तापों पर।

\begin{definition}[गलनांक]\label{def:g4:changes-of-state:melting-point}
किसी पदार्थ का \emph{गलनांक}\index{गलनांक} वह \omterm{def:g3:temperature-thermometers:temperature}{ताप} है जिस पर वह
पिघलता है --- और वही \omterm{def:g3:temperature-thermometers:temperature}{ताप} है जिस पर उसका \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} वापस \omterm{def:g2:ice-water-steam:meltfreeze}{जमता} है। पानी के
लिए यह \qty{0}{\celsius} है। हर पदार्थ का अपना गलनांक होता है:
चॉकलेट लगभग \qty{34}{\celsius} पर \omterm{def:g2:ice-water-steam:meltfreeze}{पिघलती} है (यानी मुँह के \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर ---
यह संयोग नहीं, किसी भी चॉकलेट बनाने वाले से पूछ लो), मोमबत्ती का मोम
लगभग \qty{60}{\celsius} पर, और लोहा तो \qty{1538}{\celsius} पर ही।
\end{definition}

\begin{definition}[क्वथनांक]\label{def:g4:changes-of-state:boiling-point}
किसी पदार्थ का \emph{क्वथनांक}\index{क्वथनांक} वह \omterm{def:g3:temperature-thermometers:temperature}{ताप} है जिस पर वह
उबलता है। पानी के लिए यह \qty{100}{\celsius} है। यहाँ भी हर पदार्थ
का अपना क्वथनांक होता है --- इसीलिए ध्यान रखने वाली रसोइया उसी कड़ाही
में पानी उबालकर उड़ा सकती है जबकि तेल चुपचाप \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} बना रहता है।
\end{definition}

\begin{example}[दो संख्याओं से दुनिया पढ़ना]\label{ex:g4:changes-of-state:reading}
चॉकलेट की पट्टी दुकान की ताक़ पर (\qty{20}{\celsius}) \omterm{def:g3:solids-liquids-gases:solid}{ठोस} क्यों रहती
है, पर गरमियों की चढ़ाई में जेब के भीतर (\qty{35}{\celsius}) \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} हो
जाती है? क्योंकि उसका \omterm{def:g4:changes-of-state:melting-point}{गलनांक}, लगभग \qty{34}{\celsius}, इन दोनों के
\emph{बीच} पड़ता है। कोई पदार्थ अपने \omterm{def:g4:changes-of-state:melting-point}{गलनांक} से नीचे \omterm{def:g3:solids-liquids-gases:solid}{ठोस} रहता है, उससे
ऊपर \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} --- और अपने \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} तक, जहाँ वह \omterm{def:g3:solids-liquids-gases:gas}{गैस} बनकर निकल जाता है। किसी
भी पदार्थ की, किसी भी \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर, पूरी कहानी ये दो संख्याएँ बता देती हैं।
\end{example}

\section{ज़िद्दी तापमापी}

\begin{method}[बर्फ़ को ठीक से पिघलते देखना]\label{met:g4:changes-of-state:watch}
\begin{enumerate}
  \item एक कटोरा कुटी हुई बर्फ़ या हिम से भरो और उसमें \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} गाड़
        दो; ठहरकर पढ़ो: \qty{0}{\celsius};
  \item कटोरे को गरम रसोई में ले आओ और बर्फ़ पिघलते समय हर कुछ मिनट
        पर \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} पढ़ते रहो;
  \item बर्फ़ का आख़िरी टुकड़ा ख़त्म होने के काफ़ी देर बाद तक पढ़ते
        रहो।
\end{enumerate}
ख़बर यह है: पिघलने के पूरे समय --- जब तक कटोरे में पानी और बर्फ़ साथ
हैं --- \omterm{def:g3:temperature-thermometers:temperature}{तापमापी} ज़िद करके \qty{0}{\celsius} पर ही खड़ा रहता है। बर्फ़
का \emph{आख़िरी} टुकड़ा पिघल जाने के बाद ही \omterm{def:g3:temperature-thermometers:temperature}{ताप} चढ़ना शुरू करता है।
\end{method}

\begin{proposition}[पठार का नियम]\label{prop:g4:changes-of-state:plateau}
जब तक कोई पदार्थ अवस्था बदल रहा हो --- पिघल रहा हो या उबल रहा हो ---
तब तक उसका \omterm{def:g3:temperature-thermometers:temperature}{ताप} मील-पत्थर पर ठहरा रहता है, चूल्हा चाहे कितना भी ज़ोर
लगाए। आती हुई सारी गरमी अवस्था बदलने में ही ख़र्च होती है, चढ़ने में
ज़रा भी नहीं। \omterm{def:g4:changes-of-state:melting-point}{गलनांक} पर \omterm{def:g3:solids-liquids-gases:solid}{ठोस} और \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} साथ-साथ रहते हैं; \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} पर
\omterm{def:g3:solids-liquids-gases:liquid}{द्रव} और \omterm{def:g3:solids-liquids-gases:gas}{गैस}।
\end{proposition}

\begin{example}[रसोई में पठार]\label{ex:g4:changes-of-state:kitchen}
इसीलिए हल्के उबाल और तेज़ उबाल वाला पानी सिवइयाँ बराबर तेज़ी से पकाता
है: दोनों पतीले ठीक \qty{100}{\celsius} पर खड़े हैं --- तेज़ वाला बस
अपनी फ़ालतू गरमी ज़्यादा भाप बनाने में गँवा देता है। और इसीलिए बर्फ़
के टुकड़ों वाला पेय आख़िरी टुकड़े तक \qty{0}{\celsius} पर बना रहता
है: \omterm{def:g2:ice-water-steam:meltfreeze}{पिघलती} बर्फ़ \omterm{def:g3:temperature-thermometers:temperature}{ताप} को मील-पत्थर से बाँधे रखती है।
\end{example}

\begin{remark}[आँख जिसे देख ही नहीं सकती]\label{rem:g4:changes-of-state:invisible}
\omterm{def:g3:temperature-thermometers:temperature}{तापमापी} के बिना पठार का अंदाज़ा कोई नहीं लगाता: पतीला ज़्यादा से
ज़्यादा हलचल भरा दिखता है, बर्फ़ की बालटी ज़्यादा से ज़्यादा शांत,
और दोनों में \omterm{def:g3:temperature-thermometers:temperature}{ताप} पत्थर की तरह जमा रहता है। यह ``लगने'' पर ``मापने''
की एक और जीत है --- और उन पहली बड़ी हैरानियों में से एक, जो उपकरणों ने
विज्ञान के हाथ में थमाईं।
\end{remark}

\section{हर पदार्थ अपने मील-पत्थर सँभाले रखता है}

\begin{proposition}[मील-पत्थर पहचान के निशान हैं]\label{prop:g4:changes-of-state:identity}
किसी शुद्ध पदार्थ का \omterm{def:g4:changes-of-state:melting-point}{गलनांक} और \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} कभी नहीं बदलते: पानी आज भी
\qty{0}{\celsius} पर पिघलता है, कल भी, महल में भी और तंबू में भी।
मील-पत्थर उस पदार्थ की पहचान का हिस्सा हैं --- और इतने पक्के कि कोई
वैज्ञानिक किसी अनजाने पदार्थ को उसके पिघलने और उबलने के \omterm{def:g3:temperature-thermometers:temperature}{ताप} मापकर
पहचान सकता है।
\end{proposition}

\begin{example}[मोमबत्ती और ढलाईघर]\label{ex:g4:changes-of-state:foundry}
अँगीठी के पास रखी मोमबत्ती ढुलककर टपकने लगती है: मोम लगभग
\qty{60}{\celsius} पर अपना \omterm{def:g4:changes-of-state:melting-point}{गलनांक} पार कर लेता है। ढलाईघर में
कारीगर दहकते नारंगी \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} लोहे को सूप की तरह उँडेलते हैं --- उनकी
भट्ठी \qty{1538}{\celsius} पार कर चुकी है। खेल वही, पर मील-पत्थर
बिलकुल अलग: जो काम अँगीठी मोम के साथ कर देती है, लोहे के साथ वही काम
सिर्फ़ भट्ठी कर सकती है। और सोना \qty{1064}{\celsius} पर पिघलता है:
सुनारों को यह संख्या हज़ारों वर्षों से पता है, डिग्रियों में न सही,
अपने हाथों में तो सही।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:1}
सही नाम बताओ: \omterm{def:g3:solids-liquids-gases:solid}{ठोस} से \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}; \ \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} से \omterm{def:g3:solids-liquids-gases:solid}{ठोस}; \ बड़े बुलबुलों के साथ
\omterm{def:g3:solids-liquids-gases:liquid}{द्रव} से \omterm{def:g3:solids-liquids-gases:gas}{गैस}; \ ठंडी खिड़की पर \omterm{def:g3:solids-liquids-gases:gas}{गैस} से \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:2}
पानी के दोनों मील-पत्थर उनके तापों के साथ बताओ। और चॉकलेट का \omterm{def:g4:changes-of-state:melting-point}{गलनांक},
मोटे तौर पर?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:3}
एक कटोरे में बर्फ़ और पानी साथ-साथ हैं, अच्छी तरह हिलाए हुए। \omterm{def:g3:temperature-thermometers:temperature}{तापमापी}
क्या बताएगा? और पाँच मिनट बाद क्या बताएगा, अगर कुछ बर्फ़ अब भी बची
हो?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:4}
पठार का नियम बताओ। पानी उबलते समय \omterm{def:g3:temperature-thermometers:temperature}{ताप} चढ़ता ही नहीं, तो चूल्हे की
गरमी जाती कहाँ है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:5}
मोम लगभग \qty{60}{\celsius} पर पिघलता है। मोमबत्ती \omterm{def:g3:solids-liquids-gases:solid}{ठोस} है या \omterm{def:g3:solids-liquids-gases:liquid}{द्रव}:
\qty{20}{\celsius} वाली बैठक में? \ ऐसी लौ के पास जो उसे
\qty{80}{\celsius} तक गरम कर दे? \ \qty{35}{\celsius} वाली गरमियों
की अटारी में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:6}
सिवइयाँ हल्के उबलते पानी में पक रही हैं। पापा आँच पूरी तेज़ कर देते
हैं और पतीला ज़ोरों से खौलने लगता है। क्या अब पानी ज़्यादा गरम है?
फ़ालतू गरमी इसके बदले क्या बनाती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:7}
अध्याय के पठार वाले आलेख में एक भी शब्द पढ़े बिना अवस्था के दोनों
बदलाव कैसे पहचानोगे?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:8}
लोहा \qty{1538}{\celsius} पर और सोना \qty{1064}{\celsius} पर
पिघलता है। एक भट्ठी \qty{1200}{\celsius} पर चलती है: इनमें से
कौन-सी धातु उसमें सूप की तरह उँडेली जाती है, और कौन \omterm{def:g3:solids-liquids-gases:solid}{ठोस} बनी रहती है?
दूसरी वाली के मील-पत्थर से भट्ठी कितनी डिग्री पीछे रह जाती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:changes-of-state:9}
बर्फ़ के टुकड़ों वाला पेय बर्फ़ के आख़िरी टुकड़े तक उतना ही ठंडा
क्यों रहता है --- और उसके बाद ही गरम होना क्यों शुरू करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:changes-of-state:10}
एक वैज्ञानिक को कोई अनजाना सफ़ेद \omterm{def:g3:solids-liquids-gases:solid}{ठोस} मिलता है। वह पाती है कि वह ठीक
\qty{0}{\celsius} पर पिघलता है और उसका \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} ठीक \qty{100}{\celsius}
पर उबलता है। पदार्थ के बारे में उसका सबसे अच्छा अंदाज़ा क्या है ---
और कौन-सा नियम मील-पत्थरों को पहचान-पत्र बना देता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:changes-of-state:11}
पहाड़ चढ़ने वाले बताते हैं कि ऊँची चोटियों पर पानी तब उबल जाता है जब
वह \emph{बस} छूने लायक न रहा हो --- और उनकी चाय कभी ठीक से नहीं बनती।
उबलते पानी के बारे में तुम्हारा रोज़ का कौन-सा पक्का भरोसा वहाँ टूट
जाता है? (कारण ख़ुद \omterm{def:g2:air-around-us:air}{हवा} में छिपा है, और वायुदाब वाला कोई आगे का
अध्याय उसे पकड़ेगा।)
\end{exercise}
