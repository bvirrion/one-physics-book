\chapter{सौर मंडल}\label{ch:g4:solar-system}

तुम जानते हो कि पृथ्वी हर साल सूर्य के चारों ओर एक चक्कर लगाती है। पर
क्या वह अकेली सफ़र करती है? किसी साफ़ शाम ऊपर देखो: जमी हुई
चमकियों के बीच कुछ चमकीली रोशनियाँ हफ़्ते-दर-हफ़्ते धीरे-धीरे इधर-उधर
खिसकती दिखती हैं। पुराने ज़माने के लोग उन्हें भटकने वाले कहते थे; हम
उन्हें \omterm{def:g4:solar-system:planet}{ग्रह} कहते हैं --- यानी वे दुनियाएँ जो हमारी ही तरह उसी सूर्य
के चक्कर लगाती हैं।

\section{एक तारा, आठ ग्रह}

\begin{definition}[तारा]\label{def:g4:solar-system:star}
\emph{तारा}\index{तारा} दहकते-गरम पदार्थ की विशाल गेंद है जो अपना
\omterm{def:g1:light-and-shadows:source}{प्रकाश} ख़ुद बनाती है --- यानी सच्चा \omterm{def:g1:light-and-shadows:source}{प्रकाश स्रोत}। सूर्य एक तारा है:
हमारा तारा, इतना पास कि हमारे चेहरे गरम कर दे। (रात के बाक़ी तारों को
अगले साल अपना अलग अध्याय मिलेगा।)
\end{definition}

\begin{definition}[ग्रह]\label{def:g4:solar-system:planet}
\emph{ग्रह}\index{ग्रह} एक बड़ी, गोल दुनिया है जो किसी तारे के चारों
ओर चलती है और अपना कोई \omterm{def:g1:light-and-shadows:source}{प्रकाश} नहीं बनाती --- ग्रह हमें वैसे ही दिखते
हैं जैसे \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}, यानी उस सूर्य-प्रकाश से जो वे वापस भेजते हैं।
पृथ्वी एक ग्रह है।
\end{definition}

\begin{definition}[सौर मंडल]\label{def:g4:solar-system:family}
\emph{सौर मंडल}\index{सौर मंडल} सूर्य और उसके चारों ओर चलने वाली हर
चीज़ को मिलाकर बनता है: आठ \omterm{def:g4:solar-system:planet}{ग्रह} --- क्रम में: बुध, शुक्र, पृथ्वी,
मंगल, बृहस्पति, शनि, अरुण, वरुण --- और उनके \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} तथा छोटी चट्टानों
और बर्फ़ के गोलों की एक पूरी भीड़।
\end{definition}

\begin{proposition}[ग्रहों की आदतें]\label{prop:g4:solar-system:habits}
\omterm{def:g4:solar-system:planet}{ग्रह} सलीक़े की आदतें निभाते हैं: हर एक सूर्य के चारों ओर अपनी
लगभग-गोल राह पर चलता है, सब एक ही ओर चक्कर लगाते हैं, और सारी राहें
लगभग एक ही सपाट तल में पड़ती हैं --- मानो एक ही मेज़ पर खींचे गए
छल्ले। और किसी \omterm{def:g4:solar-system:planet}{ग्रह} की राह सूर्य से जितनी दूर हो, उसका एक चक्कर उतना
ही लंबा होता है: बुध $88$ दिन में, पृथ्वी एक \omterm{def:g3:sun-days-seasons:year}{वर्ष} में, और दूर बैठा
वरुण $165$ वर्षों में।
\end{proposition}

\begin{example}[चट्टानी ग्रह और गैस के दानव]\label{ex:g4:solar-system:kinds}
भीतर के चार \omterm{def:g4:solar-system:planet}{ग्रह} --- बुध, शुक्र, पृथ्वी, मंगल --- चट्टान के गोले
हैं, छोटे और \omterm{def:g3:solids-liquids-gases:solid}{ठोस}: ऐसी दुनियाएँ जिन पर तुम खड़े हो सकते हो। बाहर के
चार --- बृहस्पति, शनि, अरुण, वरुण --- \emph{दानव} हैं: विशाल गोले,
ज़्यादातर गाढ़ी \omterm{def:g3:solids-liquids-gases:gas}{गैस} के, जिन पर खड़े होने के लिए कोई ज़मीन है ही नहीं।
सबसे बड़ा बृहस्पति हज़ार से भी ज़्यादा पृथ्वियाँ निगल सकता है; और शनि
अपने मशहूर चमकते छल्ले पहने रहता है --- बर्फ़ के अनगिनत नन्हे टुकड़े,
जो नन्हे चंद्रमाओं की भीड़ की तरह उसके चक्कर लगाते हैं।
\end{example}

\begin{example}[चंद्रमा ग्रह नहीं होते]\label{ex:g4:solar-system:moons}
हमारा \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} \emph{पृथ्वी} के चक्कर लगाता है, सीधे सूर्य के नहीं ---
इसलिए वह \omterm{def:g4:solar-system:planet}{ग्रह} नहीं बल्कि \emph{\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा}} है, यानी किसी \omterm{def:g4:solar-system:planet}{ग्रह} का साथी।

बृहस्पति तथा शनि के दर्जनों। \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} अपने \omterm{def:g4:solar-system:planet}{ग्रह} का होता है; और \omterm{def:g4:solar-system:planet}{ग्रह}
सूर्य का।
\end{example}

\section{दूर कितना दूर होता है}

\begin{method}[सौर मंडल का मापनी वाला नमूना]\label{met:g4:solar-system:model}
ग्रहों के बीच की दूरियाँ बहुत बड़ी हैं --- लाखों किलोमीटर। उन्हें
महसूस करने के लिए किसी बाग़ में \omterm{def:g4:solar-system:family}{सौर मंडल} का मापनी वाला नमूना बनाओ, और
हर आकार तथा दूरी को उसी एक गुणक से छोटा करो:
\begin{enumerate}
  \item सूर्य को एक फ़ुटबॉल मान लो और उसे फाटक पर रख दो;
  \item तब पृथ्वी काली मिर्च का एक दाना है --- और उसे ईमानदारी से
        रखने के लिए फ़ुटबॉल से लगभग $30$ बड़े क़दम चलो;
  \item बुध, यानी पिन का सिरा, फ़ुटबॉल से लगभग $12$ क़दम पर खड़ा
        होता है; बृहस्पति, यानी काँच की गोली, लगभग $150$ क़दम पर; और
        वरुण, यानी छोटा मनका, लगभग $900$ क़दम पर --- यानी लगभग पूरा
        एक किलोमीटर दूर!
\end{enumerate}
इसे चलकर देखो, और वह महसूस करो जो कोई चित्र नहीं दिखाता: \omterm{def:g4:solar-system:family}{सौर मंडल}
लगभग पूरा का पूरा ख़ाली जगह है, जिसमें नन्ही दुनियाएँ एक-दूसरे से
बहुत दूर बिखरी हैं। इस नमूने में एक बड़ा क़दम असली अंतरिक्ष के लगभग
पचास लाख किलोमीटर के बराबर है।
\end{method}

\begin{example}[हमारे आकाश के भटकने वाले]\label{ex:g4:solar-system:wanderers}
शुक्र अकसर शाम का पहला ``\omterm{def:g4:solar-system:star}{तारा}'' होता है, जो \omterm{def:g1:day-and-night:daynight}{सूर्यास्त} के बाद पश्चिम
में दहकता है --- इतना चमकीला इसलिए कि वह हमारा सबसे पास का पड़ोसी है
और सफ़ेद बादलों में लिपटा रहकर सूर्य का \omterm{def:g1:light-and-shadows:source}{प्रकाश} खुले हाथों लौटाता है।
मंगल हल्का नारंगी दमकता है; बृहस्पति और शनि स्थिर और चमकीले चमकते
हैं। इनमें से किसी एक को कुछ हफ़्तों तक जमे हुए तारों की पृष्ठभूमि में
देखते रहो: वह खिसकता है --- यही भटकना है जिसने ग्रहों को उनका पुराना
नाम दिया।
\end{example}

\begin{remark}[एक पुराना चित्र, फिर से खींचा गया]\label{rem:g4:solar-system:history}
इतिहास के ज़्यादातर हिस्से में लोग पृथ्वी को हर चीज़ के केंद्र में
रखते थे, और मानते थे कि सूर्य, \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} और \omterm{def:g4:solar-system:planet}{ग्रह} हमारे चक्कर लगाते हैं।
भटकने वालों को ध्यान से देखने वालों ने पाया कि वह चित्र चरमराने लगा
है --- \omterm{def:g4:solar-system:planet}{ग्रह} कभी-कभी कुछ समय के लिए \emph{पीछे की ओर} खिसकते दिखते
हैं, और पृथ्वी को बीच में सिंहासन पर बिठाकर इसे समझाना कठिन था। जो
सूर्य-केंद्रित चित्र तुमने अभी सीखा, उसने इस भटकाव को सीधे-सीधे समझा
दिया, और वह इंसानों द्वारा स्वीकारे गए सबसे साहसी विचारों में गिना
जाता है: हमारा घर केंद्र नहीं है --- वह सूर्य से तीसरा \omterm{def:g4:solar-system:planet}{ग्रह} है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:1}
तारे और \omterm{def:g4:solar-system:planet}{ग्रह} में क्या फ़र्क़ है? सूर्य इनमें से क्या है? और पृथ्वी?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:2}
सूर्य से क्रम में आठों ग्रहों के नाम सुनाओ। इनमें हमारा कौन-सा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:3}
कौन-से \omterm{def:g4:solar-system:planet}{ग्रह} छोटे और चट्टानी हैं? \omterm{def:g3:solids-liquids-gases:gas}{गैस} के दानव कौन-से हैं? सबसे बड़े का
नाम बताओ, और मशहूर छल्लों वाले का भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:4}
\omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} को ग्रहों में क्यों नहीं गिना जाता? वह किसके चक्कर लगाता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:5}
बुध सूर्य का एक चक्कर $88$ दिन में लगाता है, पृथ्वी $365$ दिन में,
और वरुण $165$ वर्षों में। ये संख्याएँ ग्रहों की कौन-सी आदत दिखाती
हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:6}
बाग़ वाले मापनी नमूने में सूर्य की भूमिका कौन निभाता है, और पृथ्वी
कितने क़दम दूर खड़ी होती है? जगह के हिसाब से \omterm{def:g4:solar-system:family}{सौर मंडल} लगभग पूरा किससे
बना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:7}
शुक्र बहुत चमकीला चमकता है, फिर भी वह \omterm{def:g4:solar-system:star}{तारा} नहीं है। उसका \omterm{def:g1:light-and-shadows:source}{प्रकाश} आता
कहाँ से है? (एक पुराना दोस्त उत्तर देता है: \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} याद करो।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:solar-system:8}
कुछ हफ़्तों तक आकाश देखते रहकर तुम कैसे बता सकते हो कि कोई चमकीली
रोशनी \omterm{def:g4:solar-system:planet}{ग्रह} है, \omterm{def:g4:solar-system:star}{तारा} नहीं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:solar-system:10}
एक साथी \omterm{def:g4:solar-system:family}{सौर मंडल} का चित्र बनाता है जिसमें आठों \omterm{def:g4:solar-system:planet}{ग्रह} अध्याय के पहले
चित्र की तरह सूर्य के चारों ओर सटे हुए हैं, और पूछता है कि यानों को
बाहरी ग्रहों तक पहुँचने में वर्षों क्यों लगते हैं। बाग़ वाले मापनी
नमूने से उसकी ग़लतफ़हमी दूर करो --- \omterm{def:g4:solar-system:family}{सौर मंडल} के लगभग सारे चित्र किस
बात में झूठ बोलते हैं, और उन्हें ऐसा करना क्यों पड़ता है?
\end{exercise}

\end{exercise}
