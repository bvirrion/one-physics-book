\chapter{चुंबक और दिक्सूचक}\label{ch:g4:magnets-and-compass}

वर्षों पहले तुमने पाया था कि \omterm{def:g1:magnets:magnet}{चुंबक} के दोनों सिरे अलग हैं: एक तरह रखो
तो दो \omterm{def:g1:magnets:magnet}{चुंबक} चिपक जाते हैं, पलट दो तो दूर धकेल देते हैं। हमने वादा किया
था कि उन सिरों के नाम भी हैं और एक मशहूर काम भी --- दुनिया भर के
यात्रियों को राह दिखाना। आज वह वादा पूरा हो रहा है: स्वागत है ध्रुवों
का, और उस नन्हे उपकरण का जिसने उपग्रहों से पहले हर खोजी यात्री को राह
दिखाई।

\section{दो ध्रुव}

\begin{definition}[ध्रुव]\label{def:g4:magnets-and-compass:poles}
\omterm{def:g1:magnets:magnet}{चुंबक} के वे दो ख़ास सिरे --- जहाँ उसका खिंचाव सबसे तेज़ होता है और
जहाँ क्लिप गुच्छा बनाती हैं --- उसके \emph{ध्रुव}\index{ध्रुव} हैं।
इनके नाम हैं \emph{उत्तरी ध्रुव}\index{उत्तरी ध्रुव} (अकसर लाल रंगा
हुआ, N से चिह्नित) और \emph{दक्षिणी ध्रुव}\index{दक्षिणी ध्रुव} (S से
चिह्नित)। हर \omterm{def:g1:magnets:magnet}{चुंबक} के पास दोनों होते हैं: हमेशा एक-एक।
\end{definition}

\begin{proposition}[ध्रुवों का नियम]\label{prop:g4:magnets-and-compass:rule}
विपरीत \omterm{def:g4:magnets-and-compass:poles}{ध्रुव} एक-दूसरे को \omterm{def:g1:magnets:magnet}{आकर्षित} करते हैं: N जाकर S से चिपक जाता है।
एक जैसे \omterm{def:g4:magnets-and-compass:poles}{ध्रुव} एक-दूसरे को दूर धकेलते हैं: N, N को दूर हटाता है और S,
S को। यह अकेला नियम दो चुंबकों के बीच महसूस किए गए हर चिपकाव और हर
स्प्रिंग जैसे इनकार की व्याख्या कर देता है।
\end{proposition}

\begin{example}[पुराने प्रयोगों को फिर से पढ़ना]\label{ex:g4:magnets-and-compass:reading}
जब तुम्हारे दोनों \omterm{def:g1:magnets:magnet}{चुंबक} चिपक गए थे, तब एक N किसी S के सामने था। जब
उन्होंने स्प्रिंग जैसा इनकार किया था, तब N के सामने N था (या S के
सामने S)। और क्लिप हमेशा सिरों पर इसलिए गुच्छा बनाती थीं कि \omterm{def:g1:magnets:magnet}{चुंबक} की
ताक़त ध्रुवों में ही रहती है --- उसका बीच का हिस्सा उन दोनों के बीच
की शांति है।
\end{example}

\section{पृथ्वी ख़ुद एक चुंबक है}

\begin{example}[सबसे बड़ा चुंबक]\label{ex:g4:magnets-and-compass:earth}
छड़ \omterm{def:g1:magnets:magnet}{चुंबक} को धागे से ऐसे लटकाओ कि वह आज़ादी से घूम सके, और उसके ठहरने
का इंतज़ार करो --- तब कुछ कमाल होता है: वह हमेशा \emph{एक ही ओर}
इशारा करता हुआ ठहरता है, उसका एक \omterm{def:g4:magnets-and-compass:poles}{ध्रुव} दुनिया के दूर उत्तर की ओर और
दूसरा दूर दक्षिण की ओर। कोई विशाल, अनदेखा \omterm{def:g1:magnets:magnet}{चुंबक} ही उसके ध्रुवों को
क़तार में खींच रहा होगा। वह विशाल \omterm{def:g1:magnets:magnet}{चुंबक} ख़ुद पृथ्वी है: हमारा पूरा
ग्रह एक \omterm{def:g1:magnets:magnet}{चुंबक} है, जिसके चुंबकीय \omterm{def:g4:magnets-and-compass:poles}{ध्रुव} गोले के ऊपरी और निचले सिरों के
पास हैं।
\end{example}

\begin{definition}[दिक्सूचक]\label{def:g4:magnets-and-compass:compass}
\emph{दिक्सूचक}\index{दिक्सूचक} एक छोटी, हल्की चुंबक-सुई है जो किसी
महीन \omterm{def:g3:levers-and-scales:lever}{धुरी} पर इस तरह सधी होती है कि आज़ादी से घूम सके। उसे शांति से
छोड़ दो तो सुई पृथ्वी-चुंबक की बात मानती है: उसका चिह्नित सिरा उत्तर
की ओर इशारा करता हुआ ठहर जाता है। सुई के नीचे लगा कार्ड चार
\emph{मुख्य दिशाएँ}\index{मुख्य दिशाएँ} दिखाता है: उत्तर, पूर्व,
दक्षिण, पश्चिम।
\end{definition}

\begin{method}[दिक्सूचक काम में लाना]\label{met:g4:magnets-and-compass:use}
\begin{enumerate}
  \item \omterm{def:g4:magnets-and-compass:compass}{दिक्सूचक} को सपाट और स्थिर पकड़ो, चुंबकों और लोहे की बड़ी
        चीज़ों (बाड़, गाड़ियाँ --- ये सुई का ध्यान भटकाती हैं) से
        दूर;
  \item सुई का झूलना बंद होने का इंतज़ार करो;
  \item \omterm{def:g4:magnets-and-compass:compass}{दिक्सूचक} के कार्ड को धीरे-धीरे घुमाओ जब तक उसका N सुई के
        चिह्नित सिरे के नीचे न आ जाए;
  \item अपनी दिशाएँ पढ़ो: उत्तर की ओर मुँह हो तो पूर्व दाईं ओर,
        पश्चिम बाईं ओर, और दक्षिण पीठ पीछे।
\end{enumerate}
आकाश से जाँच लो: यहाँ \omterm{def:g3:sun-days-seasons:sundial}{धूपघड़ी} वाले अध्याय की दोपहर वाली \omterm{def:g1:light-and-shadows:shadow}{छाया} उत्तर की
ओर इशारा करती है --- \omterm{def:g1:light-and-shadows:shadow}{छाया} और सुई की बात मिलनी चाहिए।
\end{method}

\begin{example}[अपना दिक्सूचक ख़ुद बनाओ]\label{ex:g4:magnets-and-compass:make}
इस्पात की सिलाई वाली सुई पर \omterm{def:g1:magnets:magnet}{चुंबक} के एक \omterm{def:g4:magnets-and-compass:poles}{ध्रुव} को बीस बार रगड़ो, हमेशा
एक ही दिशा में, और हर रगड़ के बीच \omterm{def:g1:magnets:magnet}{चुंबक} को उठाकर अलग कर लो: सुई ख़ुद
एक छोटा \omterm{def:g1:magnets:magnet}{चुंबक} बन जाती है। उसे पानी के कटोरे में तैरते कॉर्क के टुकड़े
पर धीरे से रखो --- और देखो कि कॉर्क धीरे-धीरे घूमकर तब तक ठहरता है जब
तक सुई उत्तर--दक्षिण की ओर न हो जाए। तुमने हज़ार साल पुराना
नाविकों वाला \omterm{def:g4:magnets-and-compass:compass}{दिक्सूचक} बना लिया।
\end{example}

\begin{remark}[वह उपकरण जिसने महासागर खोल दिए]\label{rem:g4:magnets-and-compass:history}
\omterm{def:g4:magnets-and-compass:compass}{दिक्सूचक} से पहले नाविक तटों से चिपके रहते थे और तारों के सहारे राह
पाने के लिए साफ़ रातों की दुआ माँगते थे। \omterm{def:g1:floating-and-sinking:words}{तैरती} सुई मिल जाने पर वे
कोहरे, तूफ़ान और चाँद-रहित अँधेरे में भी अपनी राह पकड़े रह सके ---
पूरे महासागरों के पार। बहुत कम उपकरणों ने दुनिया का नक़्शा इतनी पूरी
तरह फिर से खींचा है। और एक पहेली सदियों तक खुली रही: पृथ्वी \omterm{def:g1:magnets:magnet}{चुंबक} की
तरह बरताव करती \emph{क्यों} है? पूरी कहानी इस पाठ्यक्रम के
विश्वविद्यालय वाले वर्षों में इंतज़ार कर रही है --- और वह इंतज़ार के
लायक़ है।
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g4:magnets-and-compass:11}
एक \omterm{def:g1:magnets:magnet}{चुंबक} को उसके दोनों ध्रुवों के बीच से आरे से काट दिया जाता है, इस
उम्मीद में कि एक हाथ में अकेला N और दूसरे में अकेला S आ जाएगा।
नाविकों की सीख --- और आज तक किया गया हर प्रयोग --- कहता है कि हर आधा
हिस्सा \emph{दोनों} ध्रुवों वाला एक पूरा नन्हा \omterm{def:g1:magnets:magnet}{चुंबक} निकलता है।
दिक्सूचक-सुई वाली विधि से इस दावे को मन ही मन जाँचो: रगड़ी हुई सिलाई
की सुई छोटी है --- उसके एक \omterm{def:g4:magnets-and-compass:poles}{ध्रुव} है या दो? इससे काटकर अकेला \omterm{def:g4:magnets-and-compass:poles}{ध्रुव}
ढूँढ़ने के बारे में क्या पता चलता है?
\end{exercise}
