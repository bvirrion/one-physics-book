\section*{Hoofdstuk \ref{ch:g4:how-sound-travels} --- Hoe geluid zich voortplant}

\begin{solution}{exo:g4:how-sound-travels:1}
Dragers: \omterm{def:g2:air-around-us:air}{lucht}, water, vaste stoffen (hout, baksteen, staal, een
strak touwtje). Helemaal geen drager: lege ruimte --- zoals de
luchtloze \omterm{def:g2:moon-in-the-sky:moon}{Maan}, of de leeggepompte stolp.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:2}
De \omterm{def:g2:air-around-us:air}{lucht} --- de drager van het \omterm{def:g3:sound-around-us:sound}{geluid} --- is weggepompt. De trillende
bel had niets meer om in te trillen, wat bewijst dat \omterm{def:g3:sound-around-us:sound}{geluid} alleen
door stof reist en nooit door niets.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:3}
Geen drager, geen \omterm{def:g3:sound-around-us:sound}{geluid}: de \omterm{def:g2:moon-in-the-sky:moon}{Maan} heeft geen \omterm{def:g2:air-around-us:air}{lucht}, dus heeft de
\omterm{def:g3:sound-around-us:vibration}{trilling} van een schreeuw niets om doorheen te reizen --- ze sterft
bij de lippen.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:4}
Het strakke touwtje draagt haar: de stem schudt het bekertje, het
bekertje schudt het touwtje, en het touwtje speelt haar opnieuw af in
het verre bekertje. Regels: houd het touwtje strak gespannen, en laat
het niets raken (niet doorhangen, geen knijpende vingers).
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:5}
Ongeveer \qty{340}{m} in één \omterm{def:g2:measuring-time:second}{seconde} --- ruwweg een kilometer in
drie.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:6}
$9 \div 3 = 3$: ongeveer \qty{3}{km} ver. Drie \omterm{def:g2:measuring-time:second}{seconden} later
betekent ongeveer \qty{1}{km}: het onweer is veel dichterbij gekomen
--- naar binnen gaan.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:7}
Het hout is een veel betere drager dan \omterm{def:g2:air-around-us:air}{lucht}: het zwakke krabbeltje
is te zwak voor de \omterm{def:g2:air-around-us:air}{lucht} om het over de tafel te bezorgen, maar het
vaste tafelblad draagt de \omterm{def:g3:sound-around-us:vibration}{trilling} luid en helder rechtstreeks je
oor in.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:8}
Een \omterm{ex:g4:how-sound-travels:echo}{echo} is je eigen \omterm{def:g3:sound-around-us:sound}{geluid} dat van een grote harde wand terugkaatst
en laat genoeg terugkomt om het apart te horen. Een verdere rotswand
betekent een langere rondreis voor het \omterm{def:g3:sound-around-us:sound}{geluid}, dus komt het antwoord
later.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:9}
Het schip klikt naar beneden en klokt de \omterm{ex:g4:how-sound-travels:echo}{echo} van de zeebodem:
snelle \omterm{ex:g4:how-sound-travels:echo}{echo}, ondiep water; late \omterm{ex:g4:how-sound-travels:echo}{echo}, diep water. Dolfijnen doen het
in donker water, vleermuizen in pikzwarte grotten.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:10}
Ten eerste draagt staal \omterm{def:g3:sound-around-us:sound}{geluid} veel beter dan \omterm{def:g2:air-around-us:air}{lucht}, zodat het
gerommel de lange reis overleeft. Ten tweede leidt de rail het
\omterm{def:g3:sound-around-us:sound}{geluid} recht langs één weg naar het oor van de verkenner, terwijl in
de open \omterm{def:g2:air-around-us:air}{lucht} het lawaai van de trein zich in alle richtingen
uitspreidt als kringen in een vijver en met elke \omterm{def:g2:measuring-length:units}{meter} verdunt --- dus
komt de boodschap van de rail zowel sterker als helderder aan, lang
vóór die van de \omterm{def:g2:air-around-us:air}{lucht}.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:11}
In één \omterm{def:g2:measuring-time:second}{seconde} legt \omterm{def:g3:sound-around-us:sound}{geluid} ongeveer \qty{340}{m} af --- maar die
\omterm{def:g2:measuring-time:second}{seconde} betaalde de heenweg \emph{en} de terugweg. De muur staat op de
helft: ongeveer \qty{170}{m} ver.
\end{solution}
