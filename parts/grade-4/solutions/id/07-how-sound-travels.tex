\section*{Bab \ref{ch:g4:how-sound-travels} --- Bagaimana Bunyi Merambat}

\begin{solution}{exo:g4:how-sound-travels:1}
Perantaranya: \omterm{def:g2:air-around-us:air}{udara}, air, \omterm{def:g3:solids-liquids-gases:solid}{zat padat} (kayu, batu bata, baja, benang
yang tegang). Tanpa perantara sama sekali: ruang hampa --- seperti
\omterm{def:g2:moon-in-the-sky:moon}{Bulan} yang tanpa \omterm{def:g2:air-around-us:air}{udara}, atau toples yang sudah dipompa kosong.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:2}
\omterm{def:g2:air-around-us:air}{Udaranya} --- perantara \omterm{def:g3:sound-around-us:sound}{bunyinya} --- dipompa keluar. Lonceng yang
bergetar tidak punya apa pun lagi untuk digetarkan, dan itu
membuktikan bahwa \omterm{def:g3:sound-around-us:sound}{bunyi} merambat hanya melalui zat, tidak pernah
melalui ketiadaan.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:3}
Tanpa perantara, tanpa \omterm{def:g3:sound-around-us:sound}{bunyi}: \omterm{def:g2:moon-in-the-sky:moon}{Bulan} tidak punya \omterm{def:g2:air-around-us:air}{udara}, sehingga
\omterm{def:g3:sound-around-us:vibration}{getaran} sebuah teriakan tidak punya apa pun untuk dilewati --- ia mati
di bibir.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:4}
Benang yang tegang yang membawanya: suara mengguncang gelasnya,
gelasnya mengguncang benangnya, benangnya memutarnya kembali di gelas
yang jauh. Aturannya: jagalah benangnya tetap teregang kencang, dan
biarkan ia tidak menyentuh apa pun (tanpa kendur, tanpa jari yang
menjepit).
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:5}
Kira-kira \qty{340}{m} dalam satu sekon --- kira-kira satu kilometer dalam tiga sekon.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:6}
$9 \div 3 = 3$: kira-kira \qty{3}{km} jauhnya. Tiga sekon kemudian
berarti kira-kira \qty{1}{km}: badainya sudah jauh lebih dekat ---
masuklah ke dalam rumah.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:7}
Kayunya perantara yang jauh lebih baik daripada \omterm{def:g2:air-around-us:air}{udara}: garukan yang
lemah itu terlalu lirih untuk diantarkan \omterm{def:g2:air-around-us:air}{udara} menyeberangi meja,
tetapi papan meja yang padat membawa \omterm{def:g3:sound-around-us:vibration}{getarannya} keras dan tajam lurus
ke telingamu.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:8}
\omterm{ex:g4:how-sound-travels:echo}{Gema} adalah bunyimu sendiri yang memantul dari dinding keras yang
besar lalu kembali cukup terlambat untuk terdengar terpisah. Tebing
yang lebih jauh berarti perjalanan bolak-balik yang lebih lama bagi
\omterm{def:g3:sound-around-us:sound}{bunyinya}, sehingga jawabannya datang lebih terlambat.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:9}
Kapalnya mengeklik ke bawah lalu menghitung waktu \omterm{ex:g4:how-sound-travels:echo}{gema} dari dasar
laut: \omterm{ex:g4:how-sound-travels:echo}{gema} cepat berarti air dangkal; \omterm{ex:g4:how-sound-travels:echo}{gema} terlambat berarti air
dalam. Lumba-lumba melakukannya di air yang gelap, kelelawar di gua
yang pekat.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:10}
Pertama, baja membawa \omterm{def:g3:sound-around-us:sound}{bunyi} jauh lebih baik daripada \omterm{def:g2:air-around-us:air}{udara}, sehingga
gemuruhnya bertahan sepanjang perjalanan yang panjang. Kedua, relnya
menuntun \omterm{def:g3:sound-around-us:sound}{bunyinya} lurus pada satu jalan sampai ke telinga pengintai,
sementara di \omterm{def:g2:air-around-us:air}{udara} terbuka \omterm{def:g3:sound-around-us:sound}{bunyi} keretanya menyebar ke segala arah
seperti riak kolam, dan menipis setiap \omterm{def:g2:measuring-length:units}{meter} --- sehingga pesan
relnya tiba lebih kuat sekaligus lebih jelas, jauh sebelum pesan
\omterm{def:g2:air-around-us:air}{udaranya}.
\end{solution}

\begin{solution}{exo:g4:how-sound-travels:11}
Dalam satu sekon \omterm{def:g3:sound-around-us:sound}{bunyi} menempuh kira-kira \qty{340}{m} --- tetapi
sekon itu dibayarkan untuk pergi \emph{dan} kembali. Dindingnya
berdiri di separuhnya: kira-kira \qty{170}{m} jauhnya.
\end{solution}
