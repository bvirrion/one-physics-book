\section*{Bab \ref{ch:g4:changes-of-state} --- Melebur dan Mendidih: Perubahan Wujud}

\begin{solution}{exo:g4:changes-of-state:1}
Peleburan; pembekuan (pemadatan); pendidihan; pengembunan.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:2}
\omterm{def:g4:changes-of-state:melting-point}{Titik lebur} \qty{0}{\celsius}, \omterm{def:g4:changes-of-state:boiling-point}{titik didih} \qty{100}{\celsius}.
Cokelat melebur di dekat \qty{34}{\celsius} --- \omterm{def:g3:temperature-thermometers:temperature}{suhu} mulut.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:3}
\qty{0}{\celsius} --- dan tetap \qty{0}{\celsius} lima menit kemudian:
selama es dan air hidup bersama, \omterm{def:g3:temperature-thermometers:temperature}{suhunya} terpaku pada \omterm{def:g4:changes-of-state:melting-point}{titik lebur}.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:4}
Selagi sebuah zat melebur atau mendidih, \omterm{def:g3:temperature-thermometers:temperature}{suhunya} berdiri diam pada
tonggaknya. Kehangatannya habis dipakai untuk perubahan wujud itu
sendiri --- mengubah \omterm{def:g3:solids-liquids-gases:liquid}{zat cair} menjadi uap --- bukan untuk memanjat.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:5}
Padat pada \qty{20}{\celsius} (di bawah $60$); cair pada
\qty{80}{\celsius} (di atas $60$); masih padat pada
\qty{35}{\celsius} --- walaupun sebatang cokelat di saku sudah akan
\omterm{def:g2:ice-water-steam:meltfreeze}{mencair} di sana.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:6}
Tidak lebih panas: kedua panci berdiri pada \qty{100}{\celsius},
dataran didihnya. Kehangatan tambahannya hanya membuat uap lebih
cepat --- api yang terbuang, pasta yang sama.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:7}
Lewat kedua bentangan yang mendatar --- dataran itu. Di mana pun
kurvanya berjalan mendatar sementara kompornya terus memanaskan, di
situ sedang berlangsung perubahan wujud.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:8}
Emas ($1064 < 1200$) mengalir; besi ($1538 > 1200$) tetap padat.
Tungkunya meleset dari tonggak besi sebanyak $1538 - 1200 = 338$
derajat.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:9}
Es yang \omterm{def:g2:ice-water-steam:meltfreeze}{mencair} memakukan minumannya pada \qty{0}{\celsius}: seluruh
kehangatan yang masuk habis dipakai mencairkan es, tidak ada yang
dipakai menghangatkan minumannya. Baru ketika serpihan terakhirnya
lenyap, \omterm{def:g3:temperature-thermometers:temperature}{suhunya} dapat mulai memanjat.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:10}
Tebakan terbaiknya: zat itu es --- air yang \omterm{def:g2:ice-water-steam:meltfreeze}{membeku}. Hukumnya adalah
bahwa tonggak merupakan tanda pengenal: \omterm{def:g4:changes-of-state:melting-point}{titik lebur} dan \omterm{def:g4:changes-of-state:boiling-point}{titik didih}
sebuah zat murni tidak pernah berubah, jadi mencocoki kedua bilangan
air dengan tepat sama saja dengan sebuah tanda tangan.
\end{solution}

\begin{solution}{exo:g4:changes-of-state:11}
Kepastian bahwa air selalu mendidih pada \qty{100}{\celsius}. Tinggi
di puncak gunung, air mendidih jauh di bawah itu --- panas tetapi
belum \qty{100}{\celsius} --- sehingga air yang ``mendidih'' itu
terlalu sejuk untuk menyeduh teh dengan baik. Biang keladinya adalah
\omterm{def:g2:air-around-us:air}{udara} yang lebih tipis dan menekan airnya lebih ringan, seperti akan
ditunjukkan bab tekanan \omterm{def:g2:air-around-us:air}{udara}.
\end{solution}
