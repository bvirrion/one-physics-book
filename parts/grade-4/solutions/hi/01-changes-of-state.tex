\section*{अध्याय \ref{ch:g4:changes-of-state} --- पिघलना और उबलना: अवस्था के बदलाव}

\begin{solution}{exo:g4:changes-of-state:1}
गलन; हिमीकरण (ठोसीकरण); क्वथन; \omterm{def:g2:ice-water-steam:evapcond}{संघनन}।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:2}
\omterm{def:g4:changes-of-state:melting-point}{गलनांक} \qty{0}{\celsius}, \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} \qty{100}{\celsius}। चॉकलेट
लगभग \qty{34}{\celsius} पर \omterm{def:g2:ice-water-steam:meltfreeze}{पिघलती} है --- यानी मुँह के \omterm{def:g3:temperature-thermometers:temperature}{ताप} पर।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:3}
\qty{0}{\celsius} --- और पाँच मिनट बाद भी \qty{0}{\celsius}: जब तक
बर्फ़ और पानी साथ-साथ हैं, \omterm{def:g3:temperature-thermometers:temperature}{ताप} \omterm{def:g4:changes-of-state:melting-point}{गलनांक} से बँधा रहता है।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:4}
जब तक कोई पदार्थ पिघलता या उबलता है, उसका \omterm{def:g3:temperature-thermometers:temperature}{ताप} मील-पत्थर पर ठहरा रहता
है। सारी गरमी अवस्था बदलने में ही ख़र्च होती है --- \omterm{def:g3:solids-liquids-gases:liquid}{द्रव} को भाप बनाने
में --- चढ़ने में नहीं।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:5}
\qty{20}{\celsius} पर \omterm{def:g3:solids-liquids-gases:solid}{ठोस} ($60$ से नीचे); \qty{80}{\celsius} पर
\omterm{def:g3:solids-liquids-gases:liquid}{द्रव} ($60$ से ऊपर); \qty{35}{\celsius} पर भी \omterm{def:g3:solids-liquids-gases:solid}{ठोस} --- हालाँकि जेब
में रखी चॉकलेट की पट्टी वहाँ तब तक पिघल चुकी होती।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:6}
ज़्यादा गरम नहीं: दोनों पतीले \qty{100}{\celsius} पर, यानी उबलने के
पठार पर खड़े हैं। फ़ालतू गरमी बस भाप ज़्यादा तेज़ी से बनाती है ---
आँच बरबाद, सिवइयाँ वही।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:7}
दोनों सपाट हिस्सों से --- यानी पठारों से। जहाँ चूल्हा जलता रहे और
वक्र फिर भी सीधा चले, वहाँ अवस्था का बदलाव चल रहा है।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:8}
सोना ($1064 < 1200$) उँडेला जा सकता है; लोहा ($1538 > 1200$) \omterm{def:g3:solids-liquids-gases:solid}{ठोस}
बना रहता है। भट्ठी लोहे के मील-पत्थर से $1538 - 1200 = 338$ डिग्री
पीछे रह जाती है।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:9}
\omterm{def:g2:ice-water-steam:meltfreeze}{पिघलती} बर्फ़ पेय को \qty{0}{\celsius} से बाँधे रखती है: आती हुई सारी
गरमी बर्फ़ पिघलाने में ख़र्च होती है, पेय को गरम करने में ज़रा भी
नहीं। आख़िरी टुकड़ा ख़त्म होने के बाद ही \omterm{def:g3:temperature-thermometers:temperature}{ताप} चढ़ना शुरू कर सकता है।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:10}
उसका सबसे अच्छा अंदाज़ा: यह पदार्थ बर्फ़ है --- यानी जमा हुआ पानी।
वही नियम काम आता है कि मील-पत्थर पहचान के निशान हैं: किसी शुद्ध
पदार्थ का \omterm{def:g4:changes-of-state:melting-point}{गलनांक} और \omterm{def:g4:changes-of-state:boiling-point}{क्वथनांक} कभी नहीं बदलते, इसलिए पानी की दोनों
संख्याओं से ठीक-ठीक मेल खाना दस्तख़त जितना पक्का है।
\end{solution}

\begin{solution}{exo:g4:changes-of-state:11}
यह पक्का भरोसा टूट जाता है कि पानी हमेशा \qty{100}{\celsius} पर ही
उबलता है। ऊँची चोटी पर वह उससे काफ़ी नीचे उबल जाता है --- गरम तो,
पर \qty{100}{\celsius} नहीं --- इसलिए वह ``उबला'' पानी चाय ठीक से
बनाने के लिए बहुत ठंडा रहता है। दोषी है पतली \omterm{def:g2:air-around-us:air}{हवा}, जो पानी पर कम दबाव
डालती है, जैसा वायुदाब वाला अध्याय दिखाएगा।
\end{solution}
