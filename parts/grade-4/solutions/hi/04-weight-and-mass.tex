\section*{अध्याय \ref{ch:g4:weight-and-mass} --- भार और द्रव्यमान}

\begin{solution}{exo:g4:weight-and-mass:1}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान}: कितना पदार्थ? \omterm{def:g4:weight-and-mass:weight}{भार}: यहाँ इसे कितने ज़ोर से नीचे खींचा जा
रहा है? \omterm{def:g2:pushes-and-pulls:force}{बल} \omterm{def:g4:weight-and-mass:weight}{भार} है --- पृथ्वी का नीचे की ओर खिंचाव।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:2}
हमेशा सीधे नीचे। \omterm{def:g4:weight-and-mass:springscale}{स्प्रिंग तुला} --- \omterm{def:g4:weight-and-mass:weight}{भार} उसकी स्प्रिंग को तान देता है,
और सुई उस फैलाव की ख़बर देती है।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:3}
लगभग चार उँगली भर --- दोगुना पदार्थ, दोगुना खिंचाव, दोगुना फैलाव:
यही ``ज़्यादा पदार्थ, ज़्यादा तेज़ खिंचाव'' वाली प्रतिज्ञप्ति है।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:4}
फैलाव \emph{ख़ाली वाली जगह से} ही मापा जाना चाहिए: बैंड ख़ुद कप के
\omterm{def:g4:weight-and-mass:weight}{भार} से पहले ही कुछ लटक चुका होता है, और बोझ का माप सिर्फ़ शून्य निशान
से नीचे धँसी अतिरिक्त दूरी ही देती है।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:5}
हाँ --- \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बिना बदले सफ़र करता है: वहाँ भी \qty{20}{kg}
पदार्थ। पर उसे थामे रखना कहीं आसान है: \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} उसे लगभग छह गुना
कमज़ोर खींचता है, इसलिए उसका \omterm{def:g4:weight-and-mass:weight}{भार} पृथ्वी वाले \omterm{def:g4:weight-and-mass:weight}{भार} का लगभग छठा हिस्सा
रह जाता है।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:6}
उनकी टाँगें उतने ही ज़ोर से धकेलती हैं जितने से पृथ्वी पर सधी हुई
टाँगें धकेलती हैं, पर \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} का वापस खींचना छह गुना कमज़ोर है ---
इसलिए हर आम क़दम छलाँग बन जाता है।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:7}
\omterm{def:g4:weight-and-mass:springscale}{स्प्रिंग तुला}: वह वहीं का खिंचाव महसूस करती है और छह गुना कम बताती
है। दो-पलड़े वाली \omterm{def:g3:levers-and-scales:scale}{तुला} थैले के खिंचाव की तुलना बाटों के खिंचाव से
करती है, और \omterm{def:g2:moon-in-the-sky:moon}{चंद्रमा} दोनों को बराबर कमज़ोर कर देता है --- इसलिए पलड़े
उसी \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} पर बराबरी कर लेते हैं।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:8}
सही है। \omterm{def:g4:weight-and-mass:weight}{भार} किसी चीज़ के पदार्थ पर पृथ्वी का खिंचाव है, और पदार्थ
जितना ज़्यादा खिंचाव उतना ही ज़्यादा --- पदार्थ ही नहीं, तो खींचने को
भी कुछ नहीं: \omterm{def:g4:weight-and-mass:weight}{भार} हर जगह शून्य।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:9}
भीतर की स्प्रिंग \emph{\omterm{def:g4:weight-and-mass:weight}{भार}} महसूस करती है --- यानी पलड़े पर सेबों का
खिंचाव। पृथ्वी पर खिंचाव और पदार्थ पक्के तौर पर साथ-साथ चलते हैं
(खिंचाव ज़्यादा ठीक तभी जब पदार्थ ज़्यादा), इसलिए पदार्थ के
किलोग्रामों में लिखी \omterm{def:g3:levers-and-scales:scale}{तुला} यहाँ भी सच ही बोलती है।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:10}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} के मापन बिना बदले सफ़र करते हैं: वे ख़ुद पदार्थ का वर्णन
करते हैं, जगह का नहीं। \emph{तुलना} से की गई हर तौल --- यानी संदर्भ
बाटों के सामने रखी दो-पलड़े वाली \omterm{def:g3:levers-and-scales:scale}{तुला} --- \omterm{def:g3:measuring-mass:mass}{द्रव्यमान} बताती है और उस
दूर के ग्रह के पास भी ठीक-ठीक सही रहती है। वहाँ सिर्फ़ स्प्रिंग वाले
भार-पाठ घटेंगे, और पुरज़ों की सूची में वे कभी दर्ज ही नहीं थे।
\end{solution}

\begin{solution}{exo:g4:weight-and-mass:11}
\omterm{def:g3:measuring-mass:mass}{द्रव्यमान} पूरा का पूरा वहीं है। सुराग़: पाने को हिलाने में अब भी
सचमुच ज़ोर लगता है --- उसका हाथ आगे-पीछे की हरकत का विरोध करते पदार्थ
को महसूस करता है; और फेंकने पर वह उतना ही तेज़ जाता है जितना उसकी
माँसपेशियाँ भेज सकें --- पदार्थ का हथौड़ा फेंकने में हथौड़ा ही रहता
है। \omterm{def:g4:weight-and-mass:weight}{भार} चला गया; पदार्थ कभी नहीं।
\end{solution}
