\chapter{Energia nuclear: fissão, fusão, $E=mc^2$}\label{ch:g12:nuclear-energy}

Uma pastilha de combustível do tamanho de uma borracha de lápis cobre a eletricidade de
uma casa por um ano; o mesmo serviço, a carvão, exige três toneladas. Lá em cima, o Sol
paga pela luz do dia sumindo --- quatro milhões de toneladas de si mesmo por segundo.
Este capítulo abre o livro-caixa por trás das duas pechinchas: a própria massa é \omterm{def:g10:energy-conservation:energy}{energia},
e os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} (\cref{ch:g11:nucleus-radioactivity}) são onde ela é sacada.

\section{Massa é energia}

\begin{theorem}[Equivalência massa--energia]\label{thm:g12:nuclear-energy:emc2}\index{equivalência massa--energia}
Um corpo de massa $m$, só por existir, guarda a \omterm{def:g10:energy-conservation:energy}{energia}
\[
E = m c^2, \qquad c = \qty{3.00e8}{m/s}.
\]
Sempre que um sistema perde uma massa $\Delta m$, ele libera a \omterm{def:g10:energy-conservation:energy}{energia}
$\Delta m\, c^2$ --- seja qual for o mecanismo.
\end{theorem}
\admitted

\begin{remark}[Einstein, 1905]\label{rem:g12:nuclear-energy:einstein}
Esta é a relação de Einstein, admitida aqui; de onde ela vem é assunto do capítulo de
relatividade restrita que fecha este ano (\cref{ch:g12:special-relativity}). Repare na
cotação: $c^2 = \qty{9.00e16}{J/kg}$ --- um erro de arredondamento de massa é uma
fortuna de \omterm{def:g10:energy-conservation:energy}{energia}.
\end{remark}

\begin{definition}[Elétron-volt e unidade de massa atômica]\label{def:g12:nuclear-energy:units}
Duas \omterm{def:g10:orders-of-magnitude:unit}{unidades} talhadas para o \omterm{def:g11:fundamental-interactions:ladder}{núcleo}. O \emph{elétron-volt}\index{elétron-volt}, a
\omterm{def:g10:energy-conservation:energy}{energia} que um elétron ganha ao atravessar \qty{1}{V}
(\cref{ch:g11:circuits-and-power}): $\qty{1}{eV} = \qty{1.60e-19}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$ --- as ligações químicas negociam alguns \unit{eV}, e
as reações nucleares, \omterm{def:g11:circuits-and-power:voltage}{megaelétron-volts}. E a \emph{unidade de massa atômica
unificada}\index{unidade de massa atômica unificada}, um doze avos da massa de um \omterm{def:g11:fundamental-interactions:ladder}{átomo}
de carbono-12: $\qty{1}{u} = \qty{1.66054e-27}{kg}$, de \omterm{def:g10:energy-conservation:energy}{energia} equivalente
\qty{931.5}{MeV} --- $\qty{1}{u} = \qty{931.5}{MeV}/c^2$.
\end{definition}

\begin{example}[A fortuna adormecida]\label{ex:g12:nuclear-energy:fortune}
Um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de qualquer coisa, integralmente convertido, são $\qty{9.00e16}{J}$ ---
três anos de produção de uma grande usina (\qty{1}{GW}). Ninguém percebeu antes de 1905
porque a física comum mal toca a massa: queimar um \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de carvão
(\qty{3.0e7}{J}) o alivia em um terço de micrograma. Os \omterm{def:g11:fundamental-interactions:ladder}{núcleos}, veremos, mexem uma parte
em mil: ainda pouco, mas um milhão de vezes a química.
\end{example}

\section{O defeito de massa}

\begin{definition}[Defeito de massa e energia de ligação]\label{def:g12:nuclear-energy:defect}
Pese um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ${}^{A}_{Z}\mathrm{X}$ (massa $m$) e depois pese separadamente os $Z$
prótons e os $A - Z$ nêutrons dele: as partes são \emph{mais pesadas} do que o todo. A
diferença
\[
\Delta m = Z\,m_p + (A - Z)\,m_n - m > 0,
\qquad m_p = \qty{1.00728}{u},\; m_n = \qty{1.00866}{u},
\]
é o \emph{defeito de massa}\index{defeito de massa}; a \emph{energia de
ligação}\index{energia de ligação} $E_b = \Delta m\,c^2$ é a \omterm{def:g10:energy-conservation:energy}{energia} necessária para
desmontar o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} em \omterm{def:g11:fundamental-interactions:ladder}{núcleons} livres --- e, igualmente, a \omterm{def:g10:energy-conservation:energy}{energia} liberada quando ele é
montado. Um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ligado fica \emph{abaixo} das partes dele.
\end{definition}

dele --- montá-lo libera $E_b$, desmontá-lo custa $E_b$.}
\end{omfigure}

\begin{example}[Hélio-4]\label{ex:g12:nuclear-energy:helium}
O \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ${}^{4}_{2}\mathrm{He}$ tem $m = \qty{4.00151}{u}$; as partes dele:
$2 \times 1.00728 + 2 \times 1.00866 = \qty{4.03188}{u}$. Logo
$\Delta m = \qty{0.03037}{u}$ e $E_b = 0.03037 \times 931.5 \approx \qty{28.3}{MeV}$: o
\omterm{def:g11:fundamental-interactions:ladder}{núcleo} pesa $0.75\%$ menos do que as partes dele --- a ``uma parte em mil'' do
\cref{ex:g12:nuclear-energy:fortune}.
\end{example}

\begin{method}[Balanço energético de uma reação nuclear]\label{met:g12:nuclear-energy:balance}
\begin{enumerate}
  \item Equilibre a equação: $A$ e $Z$ totais conservados
        (\cref{ch:g11:nucleus-radioactivity}).
  \item Some as massas antes e depois, em \unit{u} --- guarde as cinco casas decimais.
  \item $\Delta m = m_{\text{antes}} - m_{\text{depois}}$, e então
        $E = \Delta m\,(\text{em u}) \times \qty{931.5}{MeV}$, levada embora como \omterm{def:g11:mechanical-energy:kinetic}{energia
        cinética} dos produtos; negativa significa que a reação precisa ser paga.
\end{enumerate}
\end{method}

\section{A curva que governa o universo}

\begin{definition}[Energia de ligação por núcleon]\label{def:g12:nuclear-energy:pernucleon}
A coesão se compara com justiça pela \emph{energia de ligação por
núcleon}\index{energia de ligação por núcleon}, $E_b/A$, o custo médio de extrair um
\omterm{def:g11:fundamental-interactions:ladder}{núcleon}: $28.3/4 \approx \qty{7.1}{MeV}$ para o hélio-4.
\end{definition}

\begin{proposition}[O pico do ferro, e dois caminhos de descida]\label{prop:g12:nuclear-energy:ironpeak}\index{pico do ferro}
Traçada contra $A$, a \omterm{def:g12:nuclear-energy:pernucleon}{energia de ligação por núcleon} sobe abruptamente pelos \omterm{def:g11:fundamental-interactions:ladder}{núcleos}
leves, atinge o máximo perto do ferro, ${}^{56}_{26}\mathrm{Fe}$, em cerca de
\qty{8.8}{MeV} por \omterm{def:g11:fundamental-interactions:ladder}{núcleon}, e depois escorrega suavemente até uns \qty{7.6}{MeV} no
urânio: o ferro é o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} mais fortemente ligado da natureza. Uma reação libera \omterm{def:g10:energy-conservation:energy}{energia}
exatamente quando os produtos dela ficam \emph{mais acima} na curva; duas estratégias
opostas, portanto, pagam as duas, e as duas caminham em direção ao ferro ---
\emph{quebrar} os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} mais pesados e \emph{fundir} os mais leves.
\end{proposition}
\admitted

\begin{remark}[Por que um pico]\label{rem:g12:nuclear-energy:whypeak}
O cabo de guerra do \cref{ch:g11:fundamental-interactions}: a cola forte prende apenas os
vizinhos em contato, e a repulsão de Coulomb atravessa o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} inteiro --- os \omterm{def:g11:fundamental-interactions:ladder}{núcleos}
pequenos são quase só superfície, os grandes quase só repulsão, e o ferro é o
compromisso. A contabilidade honesta mora nos volumes universitários.
\end{remark}

\section{A fissão e a reação em cadeia}

\begin{definition}[Fissão]\label{def:g12:nuclear-energy:fission}
A \emph{fissão}\index{fissão} é a quebra de um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} pesado em dois fragmentos de
tamanho médio. O urânio-235 é \emph{físsil}\index{físsil}: um nêutron lento, absorvido,
deixa o \omterm{def:g11:fundamental-interactions:ladder}{núcleo} tão agitado que ele se rasga em dois, cuspindo dois ou três nêutrons
novos --- ${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to \text{dois fragmentos} +
2\text{--}3\;{}^{1}_{0}\mathrm{n}$, com o par variando de fissão para fissão.
\end{definition}

\begin{example}[Uma fissão, pesada]\label{ex:g12:nuclear-energy:fissionbalance}
Uma rota frequente, pelo \cref{met:g12:nuclear-energy:balance}:
\[
{}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \longrightarrow
{}^{94}_{38}\mathrm{Sr} + {}^{140}_{54}\mathrm{Xe} + 2\,{}^{1}_{0}\mathrm{n}.
\]
Antes: $1.00866 + 234.99350 = \qty{236.00216}{u}$. Depois:
$93.89450 + 139.89200 + 2 \times 1.00866 = \qty{235.80382}{u}$. Logo
$\Delta m = \qty{0.19834}{u}$ e $E \approx \qty{185}{MeV}$; contando os decaimentos
posteriores dos fragmentos radioativos (\cref{ch:g12:radioactive-decay}), cada \omterm{def:g12:nuclear-energy:fission}{fissão}
vale cerca de \qty{200}{MeV} --- alguns \unit{eV} compram um \omterm{def:g11:fundamental-interactions:ladder}{átomo} de carvão.
\end{example}

\begin{definition}[Reação em cadeia e massa crítica]\label{def:g12:nuclear-energy:chain}
Cada \omterm{def:g12:nuclear-energy:fission}{fissão} é acesa por um nêutron e liberta dois ou três: num bloco grande o bastante,
eles atingem outros \omterm{def:g11:fundamental-interactions:ladder}{núcleos} e a reação se alimenta sozinha --- uma \emph{reação em
cadeia}\index{reação em cadeia}. Num bloco pequeno, a maioria dos nêutrons escapa antes
pela superfície; a quantidade mínima que sustenta a cadeia é a \emph{massa
crítica}\index{massa crítica} --- abaixo dela a cadeia falha, acima dela cada geração se
multiplica: o princípio da bomba.
\end{definition}

quebram outros três --- um reator deixa exatamente um seguir adiante.}

\begin{definition}[O reator, domado]\label{def:g12:nuclear-energy:reactor}
Um reator de \omterm{def:g11:work-of-force:power}{potência} segura a cadeia em exatamente um nêutron por \omterm{def:g12:nuclear-energy:fission}{fissão}. O
combustível: pastilhas de urânio moderadamente enriquecidas em urânio-235. O
\emph{moderador}\index{moderador} (água comum, em geral) freia por colisões os nêutrons
rápidos da \omterm{def:g12:nuclear-energy:fission}{fissão} --- nêutrons lentos são muito melhores para disparar a \omterm{def:g12:nuclear-energy:fission}{fissão}
seguinte; as \emph{barras de controle}\index{barra de controle} (boro, cádmio) devoram
nêutrons --- enfiadas, a cadeia morre; retiradas, ela acelera. A agitação dos fragmentos
vira calor, depois vapor, depois eletricidade (\cref{ch:g10:energy-conservation}).
\end{definition}

\begin{example}[Um grão de pimenta contra um caminhão]\label{ex:g12:nuclear-energy:gram}
Um grama de urânio-235 guarda
$N = \num{1.0e-3} / (235 \times \num{1.66054e-27}) \approx \num{2.56e21}$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}; a
$\qty{200}{MeV} = \qty{3.2e-11}{J}$ cada um, fissioná-los todos rende cerca de
\qty{8.2e10}{J} --- a \omterm{def:g10:energy-conservation:forms}{energia química} de
$\num{8.2e10} / \num{3.0e7} \approx \qty{2.7e3}{kg}$ de carvão: um grão de pimenta
contra três toneladas.
\end{example}

\section{Fusão: o fogo do próprio Sol}

\begin{definition}[Fusão]\label{def:g12:nuclear-energy:fusion}
A \emph{fusão}\index{fusão} é a junção de dois \omterm{def:g11:fundamental-interactions:ladder}{núcleos} leves num \omterm{def:g11:fundamental-interactions:ladder}{núcleo} mais pesado e
mais fortemente ligado. A reação mais acessível casa o deutério com o trítio, os
\omterm{def:g11:nucleus-radioactivity:notation}{isótopos} pesados do hidrogênio (\cref{ch:g11:nucleus-radioactivity}):
\[
{}^{2}_{1}\mathrm{H} + {}^{3}_{1}\mathrm{H} \longrightarrow
{}^{4}_{2}\mathrm{He} + {}^{1}_{0}\mathrm{n}.
\]
\end{definition}

\begin{example}[D--T, pesada]\label{ex:g12:nuclear-energy:dt}
Antes: $2.01355 + 3.01550 = \qty{5.02905}{u}$; depois:
$4.00151 + 1.00866 = \qty{5.01017}{u}$; logo $\Delta m = \qty{0.01888}{u}$ e
$E \approx \qty{17.6}{MeV}$, quatro quintos dela no nêutron --- menos do que uma \omterm{def:g12:nuclear-energy:fission}{fissão},
mas a partir de cinco \omterm{def:g11:fundamental-interactions:ladder}{núcleons} em vez de $236$: cerca de \qty{3.4e14}{J} por \omterm{def:g10:orders-of-magnitude:unit}{quilograma}
de combustível, quatro vezes a \omterm{def:g12:nuclear-energy:fission}{fissão}. O hélio-4
(\cref{ex:g12:nuclear-energy:helium}) é, de novo, a cinza.
\end{example}

\begin{definition}[A barreira de Coulomb]\label{def:g12:nuclear-energy:barrier}
Dois \omterm{def:g11:fundamental-interactions:ladder}{núcleos} são ambos positivos: para se tocar, precisam primeiro escalar a colina da
repulsão elétrica deles (\cref{ch:g11:electric-gravitational-fields}) --- a
\emph{barreira de Coulomb}\index{barreira de Coulomb}. Só perto de \qty{1e8}{K} as
colisões carregam a \omterm{def:g10:energy-conservation:energy}{energia} para atravessá-la; o combustível é então um
\emph{plasma}\index{plasma} de \omterm{def:g11:fundamental-interactions:ladder}{núcleos} nus e elétrons. A \omterm{def:g12:nuclear-energy:fission}{fissão} não paga pedágio algum
desse tipo: o gatilho dela, o nêutron, é neutro, e funciona a frio.
\end{definition}

\begin{example}[O Sol funciona com ela]\label{ex:g12:nuclear-energy:sun}
O núcleo do Sol, a $\qty{1.5e7}{K}$, funde hidrogênio comum passo a passo --- a
\emph{cadeia próton--próton}, de efeito líquido
$4\,{}^{1}_{1}\mathrm{H} \to {}^{4}_{2}\mathrm{He} + 2\,{}^{0}_{+1}\mathrm{e}
+ \text{radiação}$, cerca de \qty{26}{MeV} por hélio: $0.7\%$ da massa do hidrogênio sai
como \omterm{def:g10:energy-conservation:energy}{energia}. A luz do Sol cujo \omterm{def:g12:sound-acoustics:timbre}{espectro} lemos no \cref{ch:g10:light-spectra} é essa
massa, chegando oito minutos atrasada; toda estrela brilha caminhando pela curva em
direção ao ferro.
\end{example}

\begin{remark}[ITER: engarrafar uma estrela]\label{rem:g12:nuclear-energy:iter}
Nenhum sólido segura um \omterm{def:g12:nuclear-energy:barrier}{plasma} a \qty{1.5e8}{K}; um \emph{tokamak} o enjaula num \omterm{def:g11:magnetic-fields:bfield}{campo
magnético} (\cref{ch:g11:magnetic-fields}), uma rosquinha de \omterm{def:g11:electric-gravitational-fields:lines}{linhas de campo} ao longo das
quais as partículas carregadas espiralam sem tocar a parede; o ITER, construído para
liberar dez vezes a \omterm{def:g11:work-of-force:power}{potência} de aquecimento dele, é o passo atual. Quebre o confinamento
e o \omterm{def:g12:nuclear-energy:barrier}{plasma} esfria e para em segundos: nada de \omterm{def:g12:nuclear-energy:chain}{reação em cadeia}, nada de \omterm{def:g12:nuclear-energy:chain}{massa crítica} ---
o difícil não é apagar o fogo, e sim mantê-lo aceso.
\end{remark}

\begin{example}[A escada de energia, por quilograma]\label{ex:g12:nuclear-energy:ladder}
\omterm{def:g10:energy-conservation:energy}{Energia} por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de combustível: carvão, \qty{3.0e7}{J}; urânio-235 por \omterm{def:g12:nuclear-energy:fission}{fissão},
\qty{8.2e13}{J}; deutério--trítio por \omterm{def:g12:nuclear-energy:fusion}{fusão}, \qty{3.4e14}{J}; conversão total
($E = mc^2$), \qty{9.0e16}{J}. Seis \omterm{def:g10:orders-of-magnitude:oom}{ordens de grandeza} separam a química do \omterm{def:g11:fundamental-interactions:ladder}{núcleo} --- a
razão pela qual um reator é reabastecido por caminhão e uma usina a carvão, por
composição ferroviária.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:1}
Converta: (a) \qty{1}{eV} e \qty{1}{MeV} em \omterm{def:g11:work-of-force:work}{joules}; (b) os \qty{200}{MeV} de uma \omterm{def:g12:nuclear-energy:fission}{fissão}
em \omterm{def:g11:work-of-force:work}{joules}; (c) confira, a partir de $\qty{1}{u} = \qty{1.66054e-27}{kg}$, que o
equivalente energético dela fica perto de \qty{931.5}{MeV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:2}
Um cubo de açúcar tem massa de \qty{6.0}{g}: calcule o valor completo $E = mc^2$ dele.
Uma casa consome $\qty{10}{kW.h} = \qty{3.6e7}{J}$ por dia --- por quantos anos o cubo,
integralmente convertido, a alimentaria?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:3}
O \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de carbono-12 tem massa \qty{11.99671}{u}: calcule o \omterm{def:g12:nuclear-energy:defect}{defeito de massa} dele, a
\omterm{def:g12:nuclear-energy:defect}{energia de ligação} em \unit{MeV} e a \omterm{def:g12:nuclear-energy:pernucleon}{energia de ligação por núcleon}
($m_p = \qty{1.00728}{u}$, $m_n = \qty{1.00866}{u}$).
\end{exercise}

\section{Problema: alimentar uma cidade}

\begin{problem}[{Problema de fim de semana --- alimentar uma cidade: um gigawatt
constante para meio milhão de pessoas, comprado de quatro maneiras --- pastilhas de
urânio, composições de carvão, uma piscina de água do mar, a própria substância do Sol
--- com o mesmo quilograma de massa sumida escondido sob todas as quatro}]
\label{pb:g12:nuclear-energy:1}
Uma cidade de $500\,000$ habitantes consome uma \omterm{prop:g11:circuits-and-power:power}{potência elétrica} constante
$P = \qty{1.0}{GW}$; as usinas dela, seja qual for o combustível, transformam calor em
eletricidade com \omterm{def:g10:energy-conservation:efficiency}{rendimento} de $33\%$ (\cref{ch:g10:energy-conservation}). Dados: uma
\omterm{def:g12:nuclear-energy:fission}{fissão} de urânio-235, \qty{200}{MeV} ao todo; carvão, \qty{3.0e7}{J/kg}; uma \omterm{def:g12:nuclear-energy:fusion}{fusão}
D--T, \qty{17.6}{MeV} a partir de \qty{5.029}{u} de combustível;
$\qty{1}{u} = \qty{1.66054e-27}{kg}$; $\qty{1}{eV} = \qty{1.60e-19}{J}$; um ano
$= \qty{3.16e7}{s}$; o Sol: \qty{3.9e26}{W}, \qty{2.0e30}{kg}; um \omterm{def:g10:orders-of-magnitude:unit}{metro} cúbico de água do
mar guarda cerca de \qty{33}{g} de deutério.

\medskip
\noindent\textbf{Parte I --- A usina de \omterm{def:g12:nuclear-energy:fission}{fissão}.}
\begin{enumerate}
  \item Que \omterm{def:g11:work-of-force:power}{potência} térmica o reator precisa entregar?
  \item Exprima a \omterm{def:g10:energy-conservation:energy}{energia} de uma \omterm{def:g12:nuclear-energy:fission}{fissão} em \omterm{def:g11:work-of-force:work}{joules}.
  \item Quantas fissões por segundo movem a cidade?
  \item Qual é a massa de um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de urânio-235 ($A = 235$) e, em seguida, o
        urânio-235 consumido por segundo?
  \item Deduza o urânio-235 consumido em um ano, em \omterm{def:g10:orders-of-magnitude:unit}{quilogramas}.
  \item O combustível é enriquecido a $4.0\%$ de urânio-235: que massa um ano exige, e
        ela cabe num único caminhão?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O balanço do carvão.}
\begin{enumerate}[resume]
  \item Calcule a \omterm{def:g10:energy-conservation:forms}{energia térmica} que a cidade consome em um ano.
  \item Que massa de carvão a entrega?
  \item Calcule a razão (massa de carvão)/(massa de urânio-235) para um ano e confira-a
        contra a escada do \cref{ex:g12:nuclear-energy:ladder}.
  \item Um trem de carga puxa \qty{3.0e6}{kg}: quantos trens por ano, e por dia? Em uma
        frase: imagine as duas linhas de abastecimento.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O sonho da \omterm{def:g12:nuclear-energy:fusion}{fusão}.}
\begin{enumerate}[resume]
  \item Exprima a \omterm{def:g10:energy-conservation:energy}{energia} de uma \omterm{def:g12:nuclear-energy:fusion}{fusão} D--T em \omterm{def:g11:work-of-force:work}{joules}.
  \item Calcule a \omterm{def:g10:energy-conservation:energy}{energia} liberada por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de combustível D--T; compare com a
        escada.
  \item Que massa de combustível D--T cobre o ano da cidade? E que massa disso é
        deutério ($2.014$ dos \qty{5.029}{u})?
  \item Que volume de água do mar contém esse deutério? Compare com uma piscina
        olímpica, cerca de \qty{2.5e3}{m^3}. (O trítio é gerado a partir do lítio.)
  \item Por que essa usina não tem \omterm{def:g12:nuclear-energy:chain}{massa crítica} nem \omterm{def:g12:nuclear-energy:chain}{reação em cadeia} a temer? O que é,
        então, a parte difícil?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A estrela que paga em massa.}
\begin{enumerate}[resume]
  \item A partir de $E = mc^2$, que massa o Sol converte a cada segundo?
  \item Que massa por segundo, e por ano, \emph{qualquer} fonte de \qty{3.0}{GW}
        térmicos precisa converter? Confira que os três combustíveis entregam essa mesma
        massa.
  \item Que massa total o Sol já irradiou nos \num{4.5e9} anos dele, e que fração do Sol
        é isso?
  \item A \omterm{def:g12:nuclear-energy:fusion}{fusão} consegue explorar cerca de $0.07\%$ da massa do Sol (a parcela queimável
        do \omterm{def:g11:fundamental-interactions:ladder}{núcleo} dele): estime a vida total dele como estrela, e o que resta.
  \item Feche a auditoria em três frases: um ano da cidade em toneladas de carvão, em
        \omterm{def:g10:orders-of-magnitude:unit}{quilogramas} de urânio-235 e em \omterm{def:g10:orders-of-magnitude:unit}{quilogramas} de combustível D--T --- e o
        \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de massa que, em todos os cenários, virou a luz da cidade.
\end{enumerate}
\end{problem}
