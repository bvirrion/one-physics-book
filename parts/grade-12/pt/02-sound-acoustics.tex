\chapter{Som e acústica}\label{ch:g12:sound-acoustics}

Toque a corda grave de um violão: a sala se enche de um lá grave e, ainda
assim, nada viajou até o seu ouvido além de ar comprimido e esticado.
Este capítulo acompanha esse padrão --- a velocidade dele, como a
\omterm{def:g10:signals-and-waves:frequency}{frequência} vira \omterm{def:g12:sound-acoustics:pitch-loudness}{altura} e a \omterm{def:g11:work-of-force:power}{potência} vira intensidade sonora, e por que
uma mesma nota soa diferente em cada instrumento.

\section{O som é uma onda de pressão}

\begin{definition}[Onda sonora]\label{def:g12:sound-acoustics:sound-wave}
Uma \emph{onda sonora}\index{onda sonora} é uma \omterm{def:g12:mechanical-waves:wave}{onda mecânica} de \omterm{def:g10:pressure:pressure}{pressão}
(\cref{ch:g12:mechanical-waves}): uma superfície que vibra cria
\emph{compressões}\index{compressão} (uma leve sobrepressão) e
\emph{rarefações}\index{rarefação} (subpressão) que viajam. As porções de
ar oscilam \emph{ao longo} da direção de propagação (o \omterm{def:g10:signals-and-waves:sound}{som} é
\emph{longitudinal}\index{onda longitudinal}), e um meio é indispensável:
um sino sob uma campânula de vácuo emudece
(\cref{ch:g10:signals-and-waves}).
\end{definition}

\begin{proposition}[Velocidade do som]\label{prop:g12:sound-acoustics:speed}
\index{velocidade do som}%
O \omterm{def:g10:signals-and-waves:sound}{som} viaja a cerca de \qty{340}{m/s} no ar a
\qty{20}{\degreeCelsius}, \qty{1500}{m/s} na água e \qty{5000}{m/s} no
aço: quanto mais rígido o meio, mais rápido. \admitted
\end{proposition}

\begin{remark}[Ar quente é mais rápido]\label{rem:g12:sound-acoustics:temperature}
Esses são valores medidos; prevê-los é assunto do volume do primeiro ano
de graduação. O \omterm{def:g10:signals-and-waves:sound}{som} fica mais rápido em ar mais quente (\qty{331}{m/s} a
\qty{0}{\degreeCelsius}) --- e é por isso que os sopros desafinam à
medida que a sala esquenta.
\end{remark}

\begin{example}[Os tamanhos dos sons]\label{ex:g12:sound-acoustics:wavelengths}
A relação $\lambda = v/f$ dimensiona qualquer \omterm{def:g10:signals-and-waves:sound}{som}: no ar, um ronco de
\qty{20}{Hz} ocupa $340/20 = \qty{17}{m}$ --- um prédio --- e um chiado
de \qty{20}{kHz} cabe em \qty{1.7}{cm}; os do tamanho de um \omterm{def:g10:orders-of-magnitude:unit}{metro}
contornam portas e esquinas com facilidade.
\end{example}

\section{Altura, intensidade sonora e o ouvido}

\begin{definition}[Altura e intensidade sonora]\label{def:g12:sound-acoustics:pitch-loudness}
A \emph{altura}\index{altura (do som)} é a \omterm{def:g11:color-light-sources:perception}{percepção} da \omterm{def:g10:signals-and-waves:frequency}{frequência}:
quanto maior $f$, mais aguda a nota; o lá de concerto vale \qty{440}{Hz},
e dobrar $f$ sobe a nota uma oitava. A \emph{intensidade
sonora}\index{intensidade sonora percebida} é a \omterm{def:g11:color-light-sources:perception}{percepção} da \omterm{def:g10:signals-and-waves:amplitude}{amplitude}:
quanto mais forte a oscilação de \omterm{def:g10:pressure:pressure}{pressão}, mais alto o \omterm{def:g10:signals-and-waves:sound}{som}; a amplificação
muda a intensidade, e não a altura.
\end{definition}

\begin{remark}[A janela do ouvido]\label{rem:g12:sound-acoustics:audible}
A audição cobre de cerca de \qty{20}{Hz} a \qty{20}{kHz}
(\cref{ch:g10:signals-and-waves}); abaixo está o \omterm{def:g10:signals-and-waves:ultrasound}{infrassom}, acima o
\omterm{def:g10:signals-and-waves:ultrasound}{ultrassom} --- morcegos a \qty{50}{kHz}, sondas médicas em megahertz. Um
piano amostra só o meio: \qtyrange{27.5}{4186}{Hz}.
\end{remark}

\section{Intensidade sonora e a escala de decibéis}

\begin{definition}[Intensidade sonora]\label{def:g12:sound-acoustics:intensity}
A \emph{intensidade sonora}\index{intensidade sonora} $I$ é a \omterm{def:g11:work-of-force:power}{potência}
acústica que atravessa um \omterm{def:g10:orders-of-magnitude:unit}{metro} quadrado voltado para a \omterm{def:g10:signals-and-waves:wave}{onda}: $I = P/S$,
em \unit{W/m^2}. O \omterm{def:g10:signals-and-waves:sound}{som} audível mais fraco tem cerca de
$I_0 = \qty{1.0e-12}{W/m^2}$; a dor começa perto de \qty{1}{W/m^2}.
\end{definition}

\begin{proposition}[Atenuação em inverso do quadrado]\label{prop:g12:sound-acoustics:inverse-square}
\index{lei do inverso do quadrado}%
Uma fonte pequena que irradia uma \omterm{def:g11:work-of-force:power}{potência} $P$ igualmente em todas as
direções dá, à distância $r$, $I = P/(4\pi r^2) \propto 1/r^2$: dobrar a
distância divide a intensidade por quatro.
\end{proposition}

\begin{proof}
Toda a \omterm{def:g11:work-of-force:power}{potência} $P$ atravessa a esfera de área $4\pi r^2$; divida.
\end{proof}

\begin{definition}[Nível sonoro]\label{def:g12:sound-acoustics:level}
O ouvido cobre doze potências de dez, e por isso as intensidades são
dadas em escala logarítmica. O \emph{nível sonoro}\index{nível sonoro} de
$I$ é
\[ L = 10 \log_{10}(I/I_0), \qquad I_0 = \qty{1.0e-12}{W/m^2}, \]
em \emph{decibéis}\index{decibel} (\unit{dB}): limiar de audição em
\qty{0}{dB}, dor perto de \qty{120}{dB}.
\end{definition}

\begin{proposition}[Aritmética dos decibéis]\label{prop:g12:sound-acoustics:db-arithmetic}
Multiplicar $I$ por $k$ acrescenta $10\log_{10}k$ \omterm{def:g12:sound-acoustics:level}{decibéis}. Em
particular, $\times 10$ acrescenta \qty{10}{dB}; $\times 2$ acrescenta
$10\log_{10}2 \approx \qty{3.0}{dB}$; $N$ fontes idênticas e incoerentes
somam \emph{intensidades}, nunca níveis, resultando em
$L_1 + 10\log_{10}N$; e dobrar a distância a uma fonte pequena ($I$
dividida por quatro) retira cerca de \qty{6.0}{dB}.
\end{proposition}

\begin{proof}
$10\log_{10}(kI/I_0) = 10\log_{10}(I/I_0) + 10\log_{10}k$; para a
distância, use a \cref{prop:g12:sound-acoustics:inverse-square} com
$k = 1/4$.
\end{proof}

\begin{method}[Trabalhar em decibéis]\label{met:g12:sound-acoustics:db}
\begin{enumerate}
  \item $L = 10\log_{10}(I/I_0)$; de volta: $I = I_0 \times 10^{L/10}$.
  \item Nunca some níveis: converta em intensidades, some e converta de
        volta --- ou use $+\qty{3}{dB}$ por duplicação e
        $+\qty{10}{dB}$ por fator dez.
  \item Cada duplicação da distância a uma fonte pequena custa
        \qty{6}{dB}.
\end{enumerate}
\end{method}

\begin{remark}[Danos à audição]\label{rem:g12:sound-acoustics:damage}
A perda auditiva é uma \omterm{rem:g11:nucleus-radioactivity:dose}{dose}: \qty{85}{dB} é seguro por oito horas, e cada
\qty{3}{dB} a mais --- uma duplicação de intensidade --- reduz o tempo
seguro à metade; as células ciliadas perdidas não voltam a crescer.
\end{remark}

\section{Timbre e harmônicos}

\begin{definition}[Harmônicos]\label{def:g12:sound-acoustics:harmonics}
Um \omterm{def:g10:signals-and-waves:sound}{som} \omterm{def:g10:signals-and-waves:period}{periódico} de \omterm{def:g10:signals-and-waves:frequency}{frequência} $f_1$ --- a
\emph{fundamental}\index{fundamental} --- é, em geral, uma soma de
senoides nas \omterm{def:g10:signals-and-waves:frequency}{frequências} $f_n = n f_1$, com $n = 1, 2, 3, \dots$, os seus
\emph{harmônicos}\index{harmônico}. Um \omterm{def:g10:signals-and-waves:sound}{som} reduzido à sua fundamental é
um \emph{som puro}\index{som puro}, como o de um diapasão.
\end{definition}

\begin{definition}[Espectro e timbre]\label{def:g12:sound-acoustics:timbre}
O \emph{espectro}\index{espectro} de um \omterm{def:g10:signals-and-waves:sound}{som} é o diagrama de barras dos
seus \omterm{def:g12:sound-acoustics:harmonics}{harmônicos}: uma barra por $f_n$, com \omterm{def:g12:sound-acoustics:pitch-loudness}{altura} igual à intensidade
daquele \omterm{def:g12:sound-acoustics:harmonics}{harmônico}. O ouvido escuta $f_1$ como a \omterm{def:g12:sound-acoustics:pitch-loudness}{altura} da nota e a
\emph{mistura} de intensidades como o \emph{timbre}\index{timbre} --- a
cor do \omterm{def:g10:signals-and-waves:sound}{som}: instrumentos que tocam a mesma nota compartilham $f_1$ e
diferem apenas nos espectros.
\end{definition}

\begin{example}[Por que você reconhece quem liga]\label{ex:g12:sound-acoustics:two-instruments}
O lá de uma flauta é quase um \omterm{def:g12:sound-acoustics:harmonics}{som puro}; um violino que toque com o arco
os mesmos \qty{220}{Hz} acrescenta \omterm{def:g12:sound-acoustics:harmonics}{harmônicos} fortes em
\qtylist{440;660;880}{Hz}. Mesma \omterm{def:g12:sound-acoustics:pitch-loudness}{altura} e mesma intensidade e, ainda
assim, ninguém os confunde: uma voz também é um \omterm{def:g12:sound-acoustics:timbre}{espectro} que você conhece
de cor.
\end{example}

\section{A corda que vibra}

\begin{definition}[Onda estacionária]\label{def:g12:sound-acoustics:standing-wave}
Uma \omterm{def:g10:signals-and-waves:wave}{onda} confinada entre as extremidades fixas de uma corda pode se
acomodar numa \emph{onda estacionária}\index{onda estacionária}: uma
oscilação no lugar, com pontos imóveis --- os \emph{nós}\index{nó} ---
alternando com pontos de oscilação máxima --- os
\emph{ventres}\index{ventre}. Nada mais viaja; a corda vibra num padrão
fixo.
\end{definition}

\begin{proposition}[Modos de uma corda]\label{prop:g12:sound-acoustics:string-modes}
Uma corda de comprimento $L$, fixa nas duas pontas e percorrida por \omterm{def:g10:signals-and-waves:wave}{ondas}
de velocidade $v$, vibra de modo estável apenas em padrões que encaixam
um número inteiro $n$ de meios \omterm{def:g12:mechanical-waves:wavelength}{comprimentos de onda} entre os \omterm{def:g12:sound-acoustics:standing-wave}{nós} impostos
nas extremidades:
\[ \lambda_n = \frac{2L}{n}, \quad f_n = n\,\frac{v}{2L} = n f_1, \quad n = 1, 2, 3, \dots \]
O modo mais baixo, $f_1 = v/(2L)$, é a \omterm{def:g12:sound-acoustics:harmonics}{fundamental}; os outros são
exatamente os \omterm{def:g12:sound-acoustics:harmonics}{harmônicos} dela. \admitted
\end{proposition}

\begin{remark}[Onde mora a demonstração honesta]\label{rem:g12:sound-acoustics:honest}
Que as \omterm{def:g10:signals-and-waves:wave}{ondas} confinadas escolhem exatamente esses modos decorre da
superposição da \omterm{def:g10:signals-and-waves:wave}{onda} com as suas reflexões, um cálculo feito no volume do
primeiro ano de graduação. A geometria convence: as duas pontas têm de
permanecer paradas, de modo que um número inteiro de meios \omterm{def:g12:mechanical-waves:wavelength}{comprimentos
de onda} precisa caber ali.
\end{remark}

\begin{example}[A corda lá de um violino]\label{ex:g12:sound-acoustics:violin}
A corda lá de um violino tem $L = \qty{32.8}{cm}$ e soa
$f_1 = \qty{440}{Hz}$: as \omterm{def:g10:signals-and-waves:wave}{ondas} correm por ela a
$v = 2Lf_1 = 2 \times 0.328 \times 440 \approx \qty{289}{m/s}$ ---
fixada pela tensão e pela massa, e afinada pela cravelha. Pressionar um
dedo encurta $L$ e eleva todos os $f_n$ de uma vez: o \omterm{def:g12:sound-acoustics:timbre}{espectro} inteiro
desliza para cima.
\end{example}

\begin{remark}[Por que os instrumentos diferem]\label{rem:g12:sound-acoustics:instruments}
O modo de excitar a corda --- dedilhada, com arco, percutida --- fixa
quanto de cada modo está presente, e o corpo do instrumento amplifica uns
\omterm{def:g12:sound-acoustics:harmonics}{harmônicos} mais do que outros; excitação e corpo fixam o \omterm{def:g12:sound-acoustics:timbre}{espectro}, e por
isso o mesmo $f_1$ sai de um violão, de um violino e de um piano soando
como cada um deles.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:1}
No ar ($v = \qty{340}{m/s}$), encontre o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} (a) de uma
nota grave de \qty{55}{Hz}; (b) de um bipe de \qty{1.0}{kHz}; (c) de um
zumbido de \qty{15}{kHz}. Compare o maior com o menor.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:2}
Classifique como \omterm{def:g10:signals-and-waves:ultrasound}{infrassom}, \omterm{def:g10:signals-and-waves:sound}{som} audível ou \omterm{def:g10:signals-and-waves:ultrasound}{ultrassom}: \qty{8}{Hz};
\qty{100}{Hz}; \qty{15}{kHz}; \qty{40}{kHz}; \qty{3}{MHz}. Entre os
audíveis, qual soa mais grave?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:3}
Um operário bate num trilho a \qty{1.0}{km} de distância. Calcule os
tempos de percurso no aço e no ar. O que escuta quem tem um ouvido
encostado no trilho?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:4}
(a) Qual é o \omterm{def:g12:sound-acoustics:level}{nível sonoro} para $I = \qty{1.0e-5}{W/m^2}$? (b) Qual é a
intensidade a \qty{60}{dB}? (c) Cite um \omterm{def:g10:signals-and-waves:sound}{som} do dia a dia perto de cada
nível.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:5}
A corda sol de um violino soa $f_1 = \qty{196}{Hz}$: dê os três
\omterm{def:g12:sound-acoustics:harmonics}{harmônicos} seguintes. Por que um apito de \qty{12}{kHz}, por mais rico
que seja na fonte, chega ao ouvido como um \omterm{def:g12:sound-acoustics:harmonics}{som puro}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:6}
Uma sirene irradia $P = \qty{0.50}{W}$ igualmente em todas as direções.
Calcule a intensidade e o \omterm{def:g12:sound-acoustics:level}{nível sonoro} a \qty{5.0}{m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:7}
Um violino dá \qty{68}{dB} na sua poltrona. Que nível dão dois? E
quatro? Quantos violinos seriam necessários para chegar a \qty{78}{dB}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:8}
O nível é de \qty{95}{dB} a \qty{2.0}{m} de um alto-falante pequeno.
Preveja o nível a \qty{8.0}{m}: fica abaixo da marca de \qty{85}{dB}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:9}
Uma nota de \qty{440}{Hz} passa do ar para a água. Calcule o \omterm{def:g12:mechanical-waves:wavelength}{comprimento
de onda} dela em cada meio. O mergulhador escuta a mesma \omterm{def:g12:sound-acoustics:pitch-loudness}{altura}: que
grandeza é fixada pela fonte, e qual se ajusta?
\end{exercise}

\section{Problema: a oficina do luthier}

\begin{problem}[{Problema de fim de semana --- a oficina do luthier: uma
corda de violão marcada traste a traste, os harmônicos dela afinados num
timbre, um recital mantido abaixo da linha de perigo e uma caça por
ultrassom a um defeito na madeira}]
\label{pb:g12:sound-acoustics:1}
Um luthier está terminando um violão. A corda lá dele tem comprimento
vibrante $L = \qty{65.0}{cm}$ e precisa soar $f_1 = \qty{110}{Hz}$. Tome
\qty{340}{m/s} para o \omterm{def:g10:signals-and-waves:sound}{som} no ar e admita $f_1 = v/(2L)$
(\cref{prop:g12:sound-acoustics:string-modes}), com $v$ sendo a
velocidade da \omterm{def:g10:signals-and-waves:wave}{onda} na corda.

\medskip
\noindent\textbf{Parte I --- Marcar o braço.}
\begin{enumerate}
  \item Calcule a velocidade $v$ da \omterm{def:g10:signals-and-waves:wave}{onda} na corda.
  \item Calcule o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} da \omterm{def:g12:sound-acoustics:harmonics}{fundamental} \emph{na corda} e
        depois o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} do \omterm{def:g10:signals-and-waves:sound}{som} de \qty{110}{Hz} \emph{no
        ar}. Por que eles diferem, com a mesma \omterm{def:g10:signals-and-waves:frequency}{frequência}?
  \item A oitava (\qty{220}{Hz}): que comprimento vibrante a dá? Onde o
        décimo segundo traste precisa ficar, como fração de $L$?
  \item A quinta (\qty{165}{Hz}): que comprimento vibrante a dá?
  \item Cada traste eleva a nota um semitom, um fator $2^{1/12}$.
        Verifique com logaritmos quantos semitons separam \qty{110}{Hz}
        de \qty{220}{Hz}; qual é o comprimento no primeiro traste?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- \omterm{def:g12:sound-acoustics:harmonics}{Harmônicos} e \omterm{def:g12:sound-acoustics:timbre}{timbre}.}
\begin{enumerate}[resume]
  \item Liste os cinco primeiros \omterm{def:g12:sound-acoustics:harmonics}{harmônicos} da corda lá solta.
  \item O lá de um violinista soa \qty{440}{Hz}. Que \omterm{def:g12:sound-acoustics:harmonics}{harmônico} da corda
        do violão é esse, e que intervalo musical o separa da
        \omterm{def:g12:sound-acoustics:harmonics}{fundamental}? E \qty{330}{Hz}?
  \item O luthier dedilha perto do cavalete e depois sobre a escala:
        mesma \omterm{def:g12:sound-acoustics:pitch-loudness}{altura}, cor diferente. Explique com os \omterm{def:g12:sound-acoustics:timbre}{espectros}.
  \item Um toque leve no ponto médio durante o dedilhado impõe um \omterm{def:g12:sound-acoustics:standing-wave}{nó}
        ali. Que \omterm{def:g12:sound-acoustics:harmonics}{harmônicos} sobrevivem, e que \omterm{def:g12:sound-acoustics:pitch-loudness}{altura} se escuta?
  \item Quantos \omterm{def:g12:sound-acoustics:harmonics}{harmônicos} da corda solta caem, em princípio, dentro da
        faixa audível?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O recital.}
No primeiro concerto do violão, cada instrumento irradia uma \omterm{def:g11:work-of-force:power}{potência}
acústica $P = \qty{6.0e-3}{W}$ como fonte pequena, igualmente em todas as
direções.
\begin{enumerate}[resume]
  \item Calcule a intensidade a $r = \qty{3.0}{m}$ de um instrumento.
  \item Deduza o \omterm{def:g12:sound-acoustics:level}{nível sonoro} ali.
  \item Um quarteto toca: que nível quatro instrumentos desses dão no
        mesmo ponto? Por que não é quatro vezes o nível?
  \item Doze instrumentos tocam: qual é o nível a \qty{3.0}{m}?
  \item A que distância um ouvinte precisa se sentar para que o conjunto
        caia a \qty{85}{dB}?
  \item Regra de segurança: \qty{85}{dB} é seguro por oito horas, e cada
        \qty{+3}{dB} reduz o tempo seguro à metade. Quanto tempo o
        ouvinte da primeira fila da questão 14 pode ficar em segurança?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O defeito na madeira.}
Antes de envernizar, o luthier examina um bloco de braço de \qty{45}{mm}
de espessura com um aparelho de \qty{2.0}{MHz}; nessa madeira, tome
$v = \qty{5.0e3}{m/s}$.
\begin{enumerate}[resume]
  \item Calcule o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} do \omterm{def:g10:signals-and-waves:ultrasound}{ultrassom} na madeira. Por que
        uma \omterm{def:g10:signals-and-waves:wave}{onda} que caça defeitos precisa ser ultrassônica em vez de
        audível?
  \item Calcule o \omterm{def:g12:mechanical-waves:delay}{atraso} do \omterm{rem:g10:signals-and-waves:honest}{eco} vindo da face de trás do bloco.
  \item Um \omterm{rem:g10:signals-and-waves:honest}{eco} chega depois de \qty{7.2}{\micro s}: a que profundidade
        está o defeito?
  \item Escreva a ficha da oficina: velocidade da \omterm{def:g10:signals-and-waves:wave}{onda} na corda,
        comprimentos no décimo segundo e no primeiro traste, distância
        segura de escuta e profundidade do defeito.
\end{enumerate}
\end{problem}
