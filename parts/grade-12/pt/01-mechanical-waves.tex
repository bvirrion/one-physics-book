\chapter{Ondas mecânicas}\label{ch:g12:mechanical-waves}

Sábado à noite, estádio cheio: uma ondulação de espectadores que se
levantam varre as arquibancadas em menos de um minuto e, ainda assim,
ninguém sai do lugar. O anel de uma pedrinha atravessa o lago enquanto a
rolha apenas balança. Este capítulo estuda o próprio viajante --- a \omterm{def:g10:signals-and-waves:wave}{onda}
--- e os quatro números que a domam: velocidade, \omterm{def:g12:mechanical-waves:delay}{atraso}, \omterm{def:g10:signals-and-waves:period}{período} e
\omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} --- o bastante para ler um terremoto numa tira de
papel.

\section{A perturbação viaja, o meio fica}

\begin{definition}[Onda mecânica]\label{def:g12:mechanical-waves:wave}
Uma \emph{onda mecânica}\index{onda mecânica} é uma perturbação que se
propaga num meio material --- corda, água, ar, rocha --- transportando
\omterm{def:g10:energy-conservation:energy}{energia} \emph{sem transportar matéria}: cada pedaço do meio oscila
brevemente no lugar, passa o movimento adiante e volta a se acomodar.
\end{definition}

\begin{example}[Sem transporte de matéria]\label{ex:g12:mechanical-waves:transport}
Na \omterm{def:g10:signals-and-waves:wave}{onda} do estádio o ``meio'' é a torcida: o que viaja é o ato de se
levantar, enquanto cada espectador continua no seu lugar. No lago, o anel
se expande, mas uma rolha flutuante apenas balança no lugar.
\end{example}

\begin{definition}[Ondas transversais e longitudinais]\label{def:g12:mechanical-waves:transverse}
Uma \omterm{def:g10:signals-and-waves:wave}{onda} é \emph{transversal}\index{onda transversal} quando o meio se
move \emph{perpendicularmente} à propagação: uma corcova corre por uma
corda sacudida enquanto cada ponto sobe e desce apenas. Ela é
\emph{longitudinal}\index{onda longitudinal} quando o meio se move
\emph{ao longo} da propagação: uma zona de espiras comprimidas viaja por
uma mola empurrada, com cada espira oscilando no lugar.
\end{definition}

\begin{remark}[O som é longitudinal]\label{rem:g12:mechanical-waves:sound}
O \omterm{def:g10:signals-and-waves:sound}{som} é a \omterm{def:g12:mechanical-waves:transverse}{onda longitudinal} por excelência: o ar comprimido --- de
\omterm{def:g10:pressure:pressure}{pressão} mais alta (\cref{ch:g10:pressure}) --- viaja da fonte ao tímpano;
o \cref{ch:g12:sound-acoustics} é dedicado a ele. Os terremotos enviam os
dois tipos: as \omterm{def:g10:signals-and-waves:wave}{ondas} P comprimem a rocha e as \omterm{def:g10:signals-and-waves:wave}{ondas} S a sacodem de lado.
\end{remark}

\section{Velocidade e atraso}

\begin{definition}[Velocidade de propagação]\label{def:g12:mechanical-waves:speed}
A \emph{velocidade de propagação}\index{velocidade de propagação} de uma
\omterm{def:g10:signals-and-waves:wave}{onda} é a distância percorrida pela perturbação dividida pelo tempo gasto:
$v = d/t$. Ela é a velocidade da \emph{forma} que viaja, e não a do meio.
\end{definition}

\end{omfigure}

\begin{proposition}[O meio fixa a velocidade]\label{prop:g12:mechanical-waves:medium}
Para perturbações pequenas, a \omterm{def:g12:mechanical-waves:speed}{velocidade de propagação} depende só do
\emph{meio}, e não da forma ou do tamanho da perturbação. Numa corda
esticada, $v$ cresce com a tensão e diminui com a massa por \omterm{def:g10:orders-of-magnitude:unit}{metro}: bem
esticada e leve é rápida; frouxa e pesada é lenta. \admitted
\end{proposition}

\begin{remark}[Onde mora a fórmula]\label{rem:g12:mechanical-waves:ropeformula}
A fórmula exata da velocidade na corda é deduzida no volume do primeiro
ano de graduação; neste ano bastam os fatos experimentais acima.
\end{remark}

\begin{example}[Velocidades do som]\label{ex:g12:mechanical-waves:speeds}
O \omterm{def:g10:signals-and-waves:sound}{som} viaja a \qty{340}{m/s} no ar, \qty{1500}{m/s} na água e
\qty{5900}{m/s} no aço --- quanto mais rígido, mais rápido. Daí o ouvido
colado ao trilho dos filmes de faroeste: o aço anuncia o trem primeiro.
\end{example}

\begin{definition}[Atraso]\label{def:g12:mechanical-waves:delay}
Uma \omterm{def:g10:signals-and-waves:wave}{onda} de velocidade $v$ alcança um ponto à distância $d$ depois do
\emph{atraso}\index{atraso} $\tau = d/v$: dois pontos separados por uma
distância $d$ repetem o mesmo movimento, com o mais distante atrasado de
$\tau$ --- a medida por \omterm{rem:g10:signals-and-waves:honest}{eco} do \cref{ch:g10:signals-and-waves}, lida ao
contrário.
\end{definition}

\begin{proposition}[Superposição sem interação]\label{prop:g12:mechanical-waves:superposition}
Onde duas \omterm{def:g10:signals-and-waves:wave}{ondas} se superpõem, o deslocamento do meio é, a cada instante,
a \emph{soma} dos deslocamentos que cada uma provocaria sozinha; depois
de se cruzarem, cada \omterm{def:g10:signals-and-waves:wave}{onda} segue inalterada --- o que é deduzido no
volume do segundo ano de graduação a partir da linearidade das equações
de movimento. \admitted
\end{proposition}

\begin{example}[Pulsos que se cruzam]\label{ex:g12:mechanical-waves:crossing}
Envie uma corcova para cima de cada ponta de uma corda: onde elas se
encontram, a corda sobe por um instante até a soma das duas alturas; e
depois emergem duas corcovas intactas --- como duas conversas que se
cruzam e chegam a quem as escuta.
\end{example}

\section{Ondas periódicas: a dupla periodicidade}

\begin{definition}[Onda periódica]\label{def:g12:mechanical-waves:periodic}
Quando a fonte repete o seu movimento com \omterm{def:g10:signals-and-waves:period}{período} $T$ e \omterm{def:g10:signals-and-waves:frequency}{frequência}
$f = 1/T$ (\cref{ch:g10:signals-and-waves}), ela alimenta o meio com uma
\emph{onda periódica}\index{onda periódica}: cada ponto repete o seu
próprio movimento com \omterm{def:g10:signals-and-waves:period}{período} $T$, atrasado em relação aos vizinhos pelo
\omterm{def:g12:mechanical-waves:delay}{atraso} de percurso.
\end{definition}

\begin{definition}[Comprimento de onda]\label{def:g12:mechanical-waves:wavelength}
Um instantâneo de uma \omterm{def:g12:mechanical-waves:periodic}{onda periódica} mostra um padrão que se repete no
espaço; o \omterm{def:g10:signals-and-waves:period}{período} espacial dele --- de crista a crista --- é o
\emph{comprimento de onda}\index{comprimento de onda} $\lambda$, em
\omterm{def:g10:orders-of-magnitude:unit}{metros}. Uma \omterm{def:g12:mechanical-waves:periodic}{onda periódica} é \emph{duplamente periódica}\index{dupla
periodicidade}: com \omterm{def:g10:signals-and-waves:period}{período} $T$ no tempo em cada ponto e \omterm{def:g10:signals-and-waves:period}{período}
$\lambda$ no espaço em cada instante.
\end{definition}

\begin{theorem}[A relação fundamental das ondas]\label{thm:g12:mechanical-waves:relation}
Uma \omterm{def:g12:mechanical-waves:periodic}{onda periódica} de velocidade $v$, \omterm{def:g10:signals-and-waves:period}{período} $T$ e \omterm{def:g10:signals-and-waves:frequency}{frequência} $f$ tem
\omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} $\lambda = v\,T = v/f$.
\end{theorem}

\begin{proof}
Durante um \omterm{def:g10:signals-and-waves:period}{período} o padrão desliza para a frente por $v\,T$; mas a fonte
--- e portanto cada ponto --- está então de volta ao seu estado inicial,
de modo que o padrão deslizado coincide com o original: a distância de
repetição é $v\,T$.
\end{proof}

\begin{remark}[A frequência pertence à fonte]\label{rem:g12:mechanical-waves:source}
Ao passar de um meio para outro --- o \omterm{def:g10:signals-and-waves:sound}{som} do ar para a água --- uma \omterm{def:g10:signals-and-waves:wave}{onda}
conserva a sua \omterm{def:g10:signals-and-waves:frequency}{frequência}, ditada pela fonte, enquanto a velocidade dela
muda; por $\lambda = v/f$, o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} muda junto.
\end{remark}

\begin{example}[O lá de concerto, no ar e debaixo d'água]\label{ex:g12:mechanical-waves:concertA}
O lá de concerto de \qty{440}{Hz} do \cref{ch:g10:signals-and-waves} tem
\omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} $340/440 = \qty{0.77}{m}$ no ar e
$1500/440 = \qty{3.4}{m}$ na água: a mesma nota, esticada pelo meio mais
rápido.
\end{example}

\begin{definition}[Em fase e em oposição de fase]\label{def:g12:mechanical-waves:phase}
Dois pontos separados por um número inteiro de \omterm{def:g12:mechanical-waves:wavelength}{comprimentos de onda},
$d = k\lambda$, se movem de forma idêntica a cada instante: eles estão
\emph{em fase}\index{em fase}. Pontos separados por
$d = (k + \tfrac12)\lambda$ se movem sempre de modo oposto --- crista
contra vale: eles estão em \emph{oposição de fase}\index{oposição de
fase}.
\end{definition}

\begin{method}[Ler a dupla periodicidade]\label{met:g12:mechanical-waves:reading}
\begin{enumerate}
  \item Num registro $y(t)$ feito num ponto fixo, leia $T$ de crista a
        crista; $f = 1/T$.
  \item Num instantâneo $y(x)$ tirado num instante fixo, leia $\lambda$
        de crista a crista.
  \item Deduza $v = \lambda/T = \lambda f$; confira contra uma
        cronometragem direta $v = d/\tau$, quando houver.
  \item Nunca confunda os dois gráficos: um eixo carrega segundos, o
        outro carrega \omterm{def:g10:orders-of-magnitude:unit}{metros}.
\end{enumerate}
\end{method}

\section{Frentes de onda e atenuação}

\begin{definition}[Frente de onda]\label{def:g12:mechanical-waves:wavefront}
Uma \emph{frente de onda}\index{frente de onda} é uma linha (ou
superfície) de pontos que a \omterm{def:g10:signals-and-waves:wave}{onda} alcança no mesmo instante --- uma linha
de crista no lago. A partir de uma fonte pontual as frentes de \omterm{def:g10:signals-and-waves:wave}{onda} são
círculos que se expandem; longe de qualquer fonte elas são quase retas. A
\omterm{def:g10:signals-and-waves:wave}{onda} se propaga perpendicularmente às suas frentes de \omterm{def:g10:signals-and-waves:wave}{onda}, com cristas
sucessivas separadas por $\lambda$.
\end{definition}

\begin{definition}[Atenuação]\label{def:g12:mechanical-waves:attenuation}
A diminuição da \omterm{def:g10:signals-and-waves:amplitude}{amplitude} de uma \omterm{def:g10:signals-and-waves:wave}{onda} à medida que ela se propaga é a sua
\emph{atenuação}\index{atenuação}. Duas causas se somam: a \omterm{def:g10:energy-conservation:energy}{energia} de uma
\omterm{def:g12:mechanical-waves:wavefront}{frente de onda} circular se reparte por uma crista cada vez mais longa, e
o meio absorve um pouco durante a passagem. As frentes retas escapam da
primeira causa --- e é por isso que a ondulação de uma tempestade
distante consegue atravessar um oceano.
\end{definition}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:1}
Uma \omterm{def:g10:signals-and-waves:wave}{onda} de estádio percorre o anel de \qty{320}{m} das arquibancadas em
\qty{40}{s}. (a) Calcule a velocidade dela. (b) Algum espectador dá a
volta no anel? O que dá? (c) Cada espectador se levanta \qty{1.5}{s}
depois do vizinho a \qty{12}{m}: confira a coerência.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:2}
\omterm{def:g12:mechanical-waves:transverse}{Transversal} ou \omterm{def:g12:mechanical-waves:transverse}{longitudinal}? (a) um pulso numa corda sacudida para cima e

ar; (d) uma ondulação num lago (a superfície balança na vertical, o anel
viaja na horizontal); (e) as \omterm{def:g10:signals-and-waves:wave}{ondas} sísmicas P (que comprimem na direção

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:3}

\section{Problema: ler um sismograma}

\begin{problem}[{Problema de fim de semana --- ler um sismograma: duas
ondas disparam de uma falha rompida, três círculos fixam o epicentro e
uma terceira onda, lenta, dá o aviso a um litoral}]
\label{pb:g12:mechanical-waves:1}
Turno da noite num observatório sismológico. Às 09:14:32 a agulha da
estação A salta: primeiro tremidas rápidas e agudas, depois oscilações
mais lentas e mais largas. Um terremoto rompeu rocha em algum lugar; duas
\omterm{def:g10:signals-and-waves:wave}{ondas} disparam pela crosta: as \omterm{def:g10:signals-and-waves:wave}{ondas} P, compressões a
$v_P = \qty{6.0}{km/s}$, e as \omterm{def:g10:signals-and-waves:wave}{ondas} S, um sacolejo lateral a
$v_S = \qty{3.5}{km/s}$.

\medskip
\noindent\textbf{Parte I --- Duas \omterm{def:g10:signals-and-waves:wave}{ondas}, um relógio.}
\begin{enumerate}
  \item A partir das descrições acima, classifique as \omterm{def:g10:signals-and-waves:wave}{ondas} P e S como
        transversais ou longitudinais.
  \item Que \omterm{def:g10:signals-and-waves:wave}{onda} chega primeiro a cada estação? Calcule os tempos de
        percurso das duas ao longo de $d = \qty{120}{km}$.
  \item Mostre que o intervalo entre as chegadas P e S vale
        $\Delta t = d\,(1/v_S - 1/v_P)$ e calcule-o para
        $d = \qty{120}{km}$.
  \item Deduza a regra de medida do sismólogo,
        $d = \dfrac{v_P\,v_S}{v_P - v_S}\,\Delta t$, e mostre que o
        coeficiente vale cerca de \qty{8.4}{km} por segundo de
        intervalo.
  \item Por que o intervalo cresce com a distância --- e por que isso é
        um presente para uma estação que nunca viu o terremoto começar?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Círculos no mapa.}
\begin{enumerate}[resume]
  \item A estação A mede $\Delta t_A = \qty{25}{s}$. A que distância de A
        está o epicentro?
  \item Explique por que essa leitura sozinha coloca o epicentro sobre um
        círculo, e não num ponto.
  \item As estações B e C medem $\Delta t_B = \qty{15}{s}$ e
        $\Delta t_C = \qty{32}{s}$: quais são as distâncias delas até o
        epicentro?
  \item Explique como os três círculos juntos fixam o epicentro.
  \item A \omterm{def:g10:signals-and-waves:wave}{onda} P chegou a A às 09:14:32. Calcule o tempo de percurso dela
        desde o epicentro e o instante de origem do terremoto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A \omterm{def:g10:signals-and-waves:wave}{onda} que atravessa o oceano.}
Os três círculos se encontram no mar: a falha rompeu sob o fundo oceânico
e pôs a coluna de água em movimento. As \omterm{def:g10:signals-and-waves:wave}{ondas} longas em água profunda
viajam a uma velocidade fixada pela profundidade --- cerca de
\qty{200}{m/s} em mar aberto e cerca de \qty{10}{m/s} perto de uma praia
(admitido aqui; a fórmula é deduzida no volume do primeiro ano de
graduação). Um porto fica a $D = \qty{600}{km}$ do epicentro.
\begin{enumerate}[resume]
  \item Converta \qty{200}{m/s} para \unit{km/h} e cite um veículo de
        velocidade de cruzeiro comparável.
  \item Quanto tempo o tsunami leva para chegar ao porto?
  \item O sismômetro do próprio porto sente a \omterm{def:g10:signals-and-waves:wave}{onda} P primeiro. Depois de
        quanto tempo? Se o alarme tocar nesse instante, quanto tempo de
        aviso o litoral ganha antes do tsunami?
  \item Em alto-mar o \omterm{def:g10:signals-and-waves:period}{período} do tsunami é de cerca de \qty{15}{min}.
        Calcule o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} dele ali e explique por que um
        navio não o percebe.
  \item Ao se aproximar da praia, a velocidade cai para \qty{10}{m/s}
        enquanto o \omterm{def:g10:signals-and-waves:period}{período} fica fixo. Calcule o novo \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda}.
        O que precisa acontecer com a \omterm{def:g10:signals-and-waves:wave}{onda} enquanto ela se comprime?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- \omterm{def:g10:energy-conservation:energy}{Energia} e o relatório.}
Um terremoto forte libera da ordem de \qty{1e15}{J}, levados embora pelas
\omterm{def:g10:signals-and-waves:wave}{ondas}.
\begin{enumerate}[resume]
  \item Calcule o comprimento de uma \omterm{def:g12:mechanical-waves:wavefront}{frente de onda} circular a
        \qty{10}{km} do epicentro e depois a \qty{210}{km}; por que fator
        ela cresceu?
  \item Deduza por que o tremor enfraquece com a distância mesmo numa
        crosta sem perdas --- e nomeie a segunda causa que uma crosta
        real acrescenta.
  \item Um líquido não resiste ao cisalhamento. As estações do lado
        oposto registram a \omterm{def:g10:signals-and-waves:wave}{onda} P, mas nenhuma \omterm{def:g10:signals-and-waves:wave}{onda} S: o que isso revela
        sobre o núcleo da Terra?
  \item O trem de alta velocidade do capítulo de \omterm{def:g10:energy-conservation:energy}{energia} do ano anterior
        (\cref{ch:g11:mechanical-energy}) carrega cerca de \qty{1.5e9}{J}
        de \omterm{def:g11:mechanical-energy:kinetic}{energia cinética}: a quantos trens desses correspondem
        \qty{1e15}{J}?
  \item O relatório, em duas frases: onde e quando foi o terremoto, e
        quanto tempo de aviso o porto teve? Dê os três números.
\end{enumerate}
\end{problem}
