\chapter{A luz como onda: difração e interferência}\label{ch:g12:light-as-wave}

Incline uma película de sabão em direção a uma janela: cores deslizam por
ela, embora nada no sabão ou na água seja colorido. Aponte um \omterm{def:g11:color-light-sources:lamps}{laser} para
um único fio de cabelo: a parede atrás não mostra uma sombra fina como o
fio, e sim um risco luminoso, barrado por linhas escuras, milhares de
vezes mais largo do que o cabelo. Raios que marcham em linha reta
(\cref{ch:g10:refraction}) não explicam nenhuma das duas visões. Este
capítulo promove a luz a \omterm{def:g10:signals-and-waves:wave}{onda} plena --- meio micrômetro de crista a
crista --- e depois converte a ideia em dinheiro: um \omterm{def:g11:color-light-sources:lamps}{laser} e uma parede
nua vão medir um fio de cabelo, um \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} e os sulcos de um
CD.

\section{A luz é uma onda}

\begin{definition}[Ondas luminosas]\label{def:g12:light-as-wave:lightwave}
A \omterm{def:g10:light-spectra:white}{luz monocromática} --- uma cor pura, que prisma nenhum consegue separar
mais (\cref{ch:g10:light-spectra}) --- é uma \emph{onda
luminosa}\index{onda luminosa} periódica: ela atravessa o vácuo a
$c = \qty{3.00e8}{m/s}$ e é caracterizada pela sua \omterm{def:g10:signals-and-waves:frequency}{frequência} $f$ ou pelo
seu \emph{comprimento de onda no vácuo}\index{comprimento de onda!da luz}
$\lambda = c/f$, a distância de crista a crista do
\cref{ch:g12:mechanical-waves}. O olho responde de cerca de \qty{400}{nm}
(violeta) a \qty{800}{nm} (vermelho); a \omterm{def:g10:light-spectra:white}{luz branca} mistura a faixa
inteira.
\end{definition}

\begin{example}[Ordens de grandeza]\label{ex:g12:light-as-wave:orders}
Um \omterm{def:g11:color-light-sources:lamps}{laser} de hélio-neônio emite luz vermelha de
$\lambda = \qty{633}{nm} = \qty{6.33e-7}{m}$, de modo que
$f = c/\lambda \approx \qty{4.7e14}{Hz}$: eletrônica alguma conta tão
depressa --- todo \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} deste capítulo será medido por
geometria. Para dar escala, um fio de cabelo (cerca de
\qty{70}{\micro m}) tem cem \omterm{def:g12:mechanical-waves:wavelength}{comprimentos de onda} de diâmetro. O que
ondula? Não é o ar: são um \omterm{def:g11:electric-gravitational-fields:efield}{campo elétrico} e um \omterm{def:g11:magnetic-fields:bfield}{campo magnético},
honestamente descritos no volume do segundo ano de graduação; abaixo, só
o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} importa.
\end{example}

\section{Difração}

\begin{definition}[Difração]\label{def:g12:light-as-wave:diffraction}
Quando uma \omterm{def:g10:signals-and-waves:wave}{onda} encontra uma abertura ou um obstáculo de tamanho $a$
comparável ao seu \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda}, ela se espalha para dentro da
região que os raios deixariam escura. Esse espalhamento é a
\emph{difração}\index{difração} --- desprezível enquanto $\lambda/a$ for
minúsculo, dominante à medida que $a$ encolhe rumo a $\lambda$.
\end{definition}

\begin{proposition}[Semiabertura angular do espalhamento]\label{prop:g12:light-as-wave:halfwidth}
Uma fenda de largura $a$, iluminada com \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda}
$\lambda < a$, espalha a luz em torno da direção incidente dentro da
semiabertura angular $\theta$ (em radianos); um anteparo à distância $D$
mostra uma faixa central luminosa de largura $L$, igual ao dobro de
$D \tan\theta \approx D\,\theta$ (ângulo pequeno):
\[
\theta \approx \frac{\lambda}{a}, \qquad L = \frac{2\lambda D}{a}.
\]
\end{proposition}
\admitted

\begin{remark}[Leia a fórmula ao contrário]\label{rem:g12:light-as-wave:honest}
Quanto \emph{mais estreita} a fenda, \emph{mais largo} o espalhamento ---
o oposto da previsão dos raios. A dedução honesta (somar as ondinhas
reemitidas ao longo da fenda) mora no volume do segundo ano de graduação;
aqui a fórmula é um instrumento de medida.
\end{remark}

\begin{example}[Um laser através de uma fenda]\label{ex:g12:light-as-wave:slit}
\omterm{def:g11:color-light-sources:lamps}{Laser} vermelho, $\lambda = \qty{633}{nm}$; fenda $a = \qty{0.10}{mm}$;
anteparo a $D = \qty{2.00}{m}$. Então
$\theta = \num{6.33e-7} / \num{1.0e-4} = \qty{6.3e-3}{rad}$ --- um terço
de grau, mas a faixa central na parede tem
$L = 2 \times \num{6.33e-7} \times 2.00 / \num{1.0e-4} \approx
\qty{2.5}{cm}$ de largura, ladeada por linhas escuras e por faixas
laterais mais fracas. Reduza a fenda à metade e a faixa dobra.
\end{example}

\begin{proposition}[Um obstáculo difrata como uma fenda]\label{prop:g12:light-as-wave:hair}
Um obstáculo opaco fino de largura $d$ --- um fio, um cabelo --- produz,
fora do feixe incidente, o mesmo padrão de uma fenda de largura $d$
(princípio de Babinet, deduzido no volume do segundo ano de graduação).
\end{proposition}
\admitted

\begin{method}[Medir uma largura minúscula]\label{met:g12:light-as-wave:hairwidth}
\begin{enumerate}
  \item Estique o fio de cabelo através do feixe de \omterm{def:g11:color-light-sources:lamps}{laser}; coloque um
        anteparo a uma distância medida $D$ (alguns \omterm{def:g10:orders-of-magnitude:unit}{metros}).
  \item Meça a largura $L$ da faixa central, de linha escura a linha
        escura.
  \item Como $L = 2\lambda D/d$: $d = 2\lambda D/L$ --- cabelo mais fino,
        padrão maior.
\end{enumerate}
\end{method}

\begin{remark}[A difração limita todo instrumento]\label{rem:g12:light-as-wave:limits}
A luz entra numa câmera, num telescópio ou num olho por uma abertura de
largura $a$, de modo que mesmo uma \omterm{def:g11:lenses-and-eye:lens}{lente} perfeita borra um ponto num
borrão de tamanho angular da ordem de $\lambda/a$: detalhes mais próximos
se fundem. Um microscópio de luz visível, portanto, não resolve nada
muito abaixo de $\lambda \approx \qty{0.5}{\micro m}$ (bactérias sim,
vírus não); os espelhos dos telescópios são construídos largos pela mesma
razão.
\end{remark}

\section{Interferência}

\begin{definition}[Fontes coerentes, diferença de caminho]\label{def:g12:light-as-wave:coherent}
Duas fontes de \omterm{def:g10:signals-and-waves:wave}{onda} são \emph{coerentes}\index{fontes coerentes} quando
têm a mesma \omterm{def:g10:signals-and-waves:frequency}{frequência} e vibram em sincronia --- na prática, quando são
duas cópias de uma mesma \omterm{def:g10:signals-and-waves:wave}{onda}, obtidas por divisão. Para um ponto $M$
alcançado pelas duas, a \emph{diferença de caminho}\index{diferença de
caminho} é $\delta = S_2M - S_1M$.
\end{definition}

\begin{proposition}[Condições de interferência]\label{prop:g12:light-as-wave:conditions}
Onde duas \omterm{def:g10:signals-and-waves:wave}{ondas} \omterm{def:g12:light-as-wave:coherent}{coerentes} de \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} $\lambda$ se superpõem
elas se somam --- elas \emph{interferem}\index{interferência}:
\begin{itemize}
  \item $\delta = k\lambda$ ($k$ inteiro): em sincronia, reforçando-se
        --- interferência \emph{construtiva}\index{interferência
        construtiva} (claro);
  \item $\delta = (k + \tfrac12)\lambda$: crista sobre vale,
        cancelando-se --- interferência
        \emph{destrutiva}\index{interferência destrutiva} (escuro).
\end{itemize}
\end{proposition}

\begin{proof}
Deslocar uma \omterm{def:g12:mechanical-waves:periodic}{onda periódica} por \omterm{def:g12:mechanical-waves:wavelength}{comprimentos de onda} inteiros não muda
nada: as cristas se empilham sobre cristas. Meio \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} a
mais põe crista sobre vale: \omterm{def:g10:signals-and-waves:wave}{ondas} iguais se cancelam.
\end{proof}

\begin{example}[A experiência das duas fendas de Young]\label{ex:g12:light-as-wave:young}
Perfure duas fendas finas $S_1$ e $S_2$, separadas por uma fração de
milímetro, numa placa opaca, e ilumine as duas com um único \omterm{def:g11:color-light-sources:lamps}{laser}: duas
\omterm{def:g12:light-as-wave:coherent}{fontes coerentes}. Num anteparo a uns dois \omterm{def:g10:orders-of-magnitude:unit}{metros}, a superposição fica
listrada: no centro $\delta = 0$, claro; subindo pelo anteparo, $\delta$
cresce passando por $\lambda/2$ (escuro), $\lambda$ (claro)\,\dots\
escuridão onde luz somada a luz se cancela.
\end{example}

\begin{definition}[Franjas e interfranja]\label{def:g12:light-as-wave:fringes}
As faixas claras e escuras que se alternam no padrão são as suas
\emph{franjas de interferência}\index{franjas de interferência}; a
distância $i$ entre centros claros consecutivos é a
\emph{interfranja}\index{interfranja}.
\end{definition}

\begin{proposition}[A interfranja]\label{prop:g12:light-as-wave:interfringe}
Para fendas separadas por uma distância $a$ e um anteparo à distância
$D \gg a$, o ponto do anteparo à distância $x$ do centro $O$ tem
\omterm{def:g12:light-as-wave:coherent}{diferença de caminho} $\delta \approx a\,x/D$ (geometria admitida); pela
\cref{prop:g12:light-as-wave:conditions}, as franjas claras ficam em
$x_k = k\,\lambda D/a$, igualmente espaçadas, com \omterm{def:g12:light-as-wave:fringes}{interfranja}
\[
i = \frac{\lambda D}{a}.
\]
Com $\lambda = \qty{633}{nm}$, $a = \qty{0.20}{mm}$ e
$D = \qty{2.0}{m}$: $i = \qty{6.3}{mm}$ --- a montagem amplia $\lambda$
por $D/a = 10^4$. A geometria por trás disso é trabalhada no volume do
segundo ano de graduação.
\end{proposition}
\admitted

\begin{method}[Medir um comprimento de onda]\label{met:g12:light-as-wave:lambda}
\begin{enumerate}
  \item Envie o \omterm{def:g11:color-light-sources:lamps}{laser} desconhecido por uma fenda dupla de espaçamento $a$
        conhecido; coloque o anteparo a uma distância medida $D$.
  \item Meça a extensão de $10$ \omterm{def:g12:light-as-wave:fringes}{interfranjas}; divida por $10$: obtém-se a
        \omterm{def:g12:light-as-wave:fringes}{interfranja} $i$, com o erro da régua reduzido dez vezes.
  \item Então $\lambda = i\,a/D$.
\end{enumerate}
\end{method}

\begin{remark}[Por que duas lâmpadas nunca interferem]\label{rem:g12:light-as-wave:lamps}
Duas luminárias de mesa iluminando uma mesma parede não produzem franja
alguma: cada uma emite trens de \omterm{def:g10:signals-and-waves:wave}{onda} curtos e não correlacionados, cuja
sincronia relativa salta ao acaso um bilhão de vezes por segundo, de modo
que as franjas se deslocam na mesma velocidade e o olho as promedia numa
luz uniforme. A interferência exige coerência --- uma \omterm{def:g10:signals-and-waves:wave}{onda} dividida em
duas, como fazem as fendas de Young; o \omterm{def:g11:color-light-sources:lamps}{laser} é o atalho moderno, uma
única \omterm{def:g10:signals-and-waves:wave}{onda} limpa e longa.
\end{remark}

\section{Cores pintadas por interferência}

\begin{definition}[Iridescência]\label{def:g12:light-as-wave:iridescence}
Cores que mudam com o ângulo de observação --- bolhas de sabão, manchas
de óleo, penas de pavão, o verso de um CD --- são
\emph{iridescência}\index{iridescência}: não há pigmento algum, e sim
interferência em \omterm{def:g10:light-spectra:white}{luz branca}, reforçando alguns \omterm{def:g12:mechanical-waves:wavelength}{comprimentos de onda} e
cancelando outros.
\end{definition}

\begin{example}[Películas de sabão e manchas de óleo]\label{ex:g12:light-as-wave:soap}
A luz que atinge uma película de sabão reflete em parte na face da frente
e em parte na de trás: duas cópias \omterm{def:g12:light-as-wave:coherent}{coerentes} cuja \omterm{def:g12:light-as-wave:coherent}{diferença de caminho} é
fixada pela espessura e pelo ângulo de observação. Cada espessura
devolve alguns \omterm{def:g12:mechanical-waves:wavelength}{comprimentos de onda} reforçados e outros cancelados: a
película reflete uma cor que desliza à medida que ela escorre e afina.
Uma mancha de óleo no asfalto molhado faz o mesmo truque.
\end{example}

\begin{remark}[Luz monocromática contra luz branca]\label{rem:g12:light-as-wave:white}
Em \omterm{def:g10:light-spectra:white}{luz branca} cada \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} desenha as suas próprias franjas
de Young com o seu próprio $i = \lambda D/a$: no centro todas as cores
ficam claras (uma franja branca); poucas franjas adiante os padrões se
descolam --- bordas iridescentes e depois borrão. A \omterm{def:g10:light-spectra:white}{luz monocromática} dá
franjas nítidas sem fim: daí o \omterm{def:g11:color-light-sources:lamps}{laser} em toda medida precisa aqui.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:1}
Calcule a \omterm{def:g10:signals-and-waves:frequency}{frequência} da luz vermelha ($\lambda = \qty{700}{nm}$) e da luz
violeta (\qty{400}{nm}). Qual é maior?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:2}
Um \omterm{def:g11:color-light-sources:lamps}{laser} verde ($\lambda = \qty{532}{nm}$) atravessa uma fenda de largura
$a = \qty{0.20}{mm}$: calcule $\theta$ e depois a largura da faixa
central num anteparo a \qty{1.5}{m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:3}
Uma porta tem \qty{0.80}{m} de largura: calcule $\lambda/a$ para uma voz
de \qty{425}{Hz} ($v = \qty{340}{m/s}$) e para luz de \qty{550}{nm}. Por
que você escuta, mas não enxerga, quem está virando a esquina?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:4}
Dois alto-falantes acionados por um mesmo gerador emitem, em sincronia,
\omterm{def:g10:signals-and-waves:sound}{som} de \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} \qty{2.0}{cm}. \omterm{def:g12:light-as-wave:coherent}{Diferença de caminho} em $M$:
\qty{6.0}{cm}; em $N$: \qty{5.0}{cm}. \omterm{def:g10:signals-and-waves:sound}{Som} forte ou silêncio em cada um?
Justifique.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:5}
Fendas de Young: $\lambda = \qty{633}{nm}$, $a = \qty{0.25}{mm}$,
$D = \qty{1.8}{m}$. Calcule a \omterm{def:g12:light-as-wave:fringes}{interfranja}. Quantas franjas claras cabem
nos \qty{2.0}{cm} centrais do anteparo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:6}
Um fio de cabelo num feixe de $\lambda = \qty{650}{nm}$, com anteparo a
\qty{1.6}{m}, dá uma faixa central de \qty{2.8}{cm} de largura. Calcule o

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:7}
Um \omterm{def:g11:color-light-sources:lamps}{laser} desconhecido atravessa fendas de Young ($a = \qty{0.20}{mm}$,
$D = \qty{1.50}{m}$): dez \omterm{def:g12:light-as-wave:fringes}{interfranjas} ocupam \qty{4.7}{cm}. Encontre

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:8}
Duas luminárias de mesa idênticas iluminam a mesma parede; os feixes se
superpõem e, ainda assim, franja alguma aparece, nunca. Que condição de
coerência (\cref{def:g12:light-as-wave:coherent}) falha, e como o arranjo
de Young a repara?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:9}
Num padrão de duas fendas ($a = \qty{0.30}{mm}$, $D = \qty{1.4}{m}$), os
centros da primeira e da sétima franjas claras estão a \qty{18.0}{mm} um

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:10}
As trilhas de um CD formam uma rede de espaçamento
$d = \qty{1.6}{\micro m}$. Admitindo $d \sin\theta = k\lambda$ ($k$

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:11}
Um telescópio de abertura $a = \qty{10}{cm}$ em
$\lambda = \qty{550}{nm}$: calcule o borrão de \omterm{def:g12:light-as-wave:diffraction}{difração} dele,

\qty{1.5}{m} um do outro, a \qty{100}{km} de distância?
\end{exercise}

\section{Problema: medir com luz}

\begin{problem}[{Problema de fim de semana --- medir com luz: um laser,
uma régua e uma parede nua viram uma oficina de micrômetros que afere uma
fenda, um fio de cabelo humano, um comprimento de onda desconhecido e os
sulcos de dois discos prateados}]
\label{pb:g12:light-as-wave:1}
O seu kit: um \omterm{def:g11:color-light-sources:lamps}{laser} vermelho ($\lambda = \qty{633}{nm}$, pelo rótulo), um
\omterm{def:g11:color-light-sources:lamps}{laser} verde de \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} desconhecido, uma fenda calibrada de
$a = \qty{100}{\micro m}$, uma fenda dupla de espaçamento
$a' = \qty{0.250}{mm}$, uma trena, dois fios de cabelo, um CD e um DVD.

\medskip
\noindent\textbf{Parte I --- Calibrar na fenda conhecida.}
\omterm{def:g11:color-light-sources:lamps}{Laser} vermelho, fenda, parede a $D = \qty{3.00}{m}$.
\begin{enumerate}
  \item Calcule a semiabertura angular $\theta$ do feixe difratado.
  \item Mostre que a faixa central tem largura $L = 2\lambda D/a$;
        calcule-a.
  \item Você mede $L = \qty{3.7}{cm}$: calcule o desvio relativo. O
        método está validado?
  \item Uma segunda fenda tem metade da largura, \qty{50}{\micro m}:
        preveja o $L$ dela. Enuncie a regra que liga a largura da fenda à
        largura do padrão.
  \item Justifique o passo $\tan\theta \approx \theta$ da questão 2:
        compare $\theta$ e $\tan\theta$ nesse ângulo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O fio de cabelo.} Mesmo \omterm{def:g11:color-light-sources:lamps}{laser} e mesma
parede; um fio de cabelo esticado substitui a fenda.
\begin{enumerate}[resume]
  \item Que resultado admitido permite manter a fórmula da fenda para um
        fio de cabelo, e o que ele diz?
  \item A faixa central tem $L = \qty{5.4}{cm}$ de largura: calcule o
        diâmetro $d$ do cabelo.
  \item A sua régua lê $L$ com $\pm\qty{0.2}{cm}$: dê o intervalo
        resultante para $d$, na forma $d \pm \Delta d$.
  \item O cabelo de um amigo dá $L = \qty{7.6}{cm}$: qual é o diâmetro
        dele? Qual dos dois cabelos é mais fino, e por que mais fino
        significa mais largo?
  \item Uma luminária de mesa poderia substituir o \omterm{def:g11:color-light-sources:lamps}{laser}? Dê duas razões.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} do \omterm{def:g11:color-light-sources:lamps}{laser} verde.}
\omterm{def:g11:color-light-sources:lamps}{Laser} verde, fenda dupla, parede a $D = \qty{2.00}{m}$.
\begin{enumerate}[resume]
  \item Por que as duas fendas precisam ser iluminadas pelo \emph{mesmo}
        \omterm{def:g11:color-light-sources:lamps}{laser}?
  \item Por que a franja central é clara qualquer que seja o \omterm{def:g12:mechanical-waves:wavelength}{comprimento
        de onda}?
  \item Dez \omterm{def:g12:light-as-wave:fringes}{interfranjas} ocupam \qty{4.26}{cm}. Dê $i$ e a razão de medir
        dez de uma vez.
  \item Deduza o \omterm{def:g12:mechanical-waves:wavelength}{comprimento de onda} do \omterm{def:g11:color-light-sources:lamps}{laser} verde.
  \item O rótulo do fabricante diz $\qty{532 \pm 10}{nm}$: é coerente?
        Que cor você espera?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Ler discos prateados.} O \omterm{def:g11:color-light-sources:lamps}{laser} vermelho
atinge um CD sob incidência normal; as trilhas dele agem como uma rede de
reflexão que obedece (admitido) a $d \sin\theta = k\lambda$. Na parede a
$D = \qty{1.00}{m}$, cada feixe de primeira ordem faz um ponto a $x$ do
ponto central.
\begin{enumerate}[resume]
  \item Para o CD, $x = \qty{43}{cm}$: calcule $\theta$. Por que o atalho
        de ângulo pequeno da Parte I agora é proibido?
  \item Deduza o espaçamento $d$ das trilhas do CD.
  \item O DVD manda o ponto de primeira ordem para $x = \qty{1.65}{m}$:
        calcule o $\theta$ dele e o espaçamento das trilhas.
  \item Compare os dois espaçamentos. Por que um DVD armazena mais do que
        um CD?
  \item Feche o caderno: liste os comprimentos medidos neste fim de
        semana e diga em uma frase o que fez da luz uma régua para todos
        eles.
\end{enumerate}
\end{problem}
