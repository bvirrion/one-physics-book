\chapter{Queda livre e lançamento de projéteis}\label{ch:g12:projectile-motion}

Todo jato de uma mangueira de jardim desenha a mesma curva no ar. E também uma bola de
basquete rumo à cesta, um saltador em distância, uma bala de canhão: qualquer corpo
abandonado ao seu \omterm{def:g10:universal-gravitation:weight}{peso}. Este capítulo conquista essa curva --- uma parábola --- a
partir da \omterm{thm:g12:newtons-laws:second}{segunda lei de Newton} em duas linhas de cálculo, lê nela o ponto mais alto,
o \omterm{def:g12:projectile-motion:range}{tempo de voo} e o \omterm{def:g12:projectile-motion:range}{alcance}, e depois deixa o ar voltar para ver o que ele estraga.

\section{Queda livre}

\begin{definition}[Queda livre]\label{def:g12:projectile-motion:freefall}
Um corpo está em \emph{queda livre}\index{queda livre} quando a única \omterm{def:g10:inertia:force}{força} que age
sobre ele é o seu \omterm{def:g10:universal-gravitation:weight}{peso} $\vect P = m\vect g$: sem apoio, sem fio, com resistência do ar
desprezível. Uma pedra solta, ou uma bola que já deixou a mão, está em queda livre com
boa precisão --- e ``queda'' inclui a subida.
\end{definition}

\begin{theorem}[Lei da queda livre de Galileu]\label{thm:g12:projectile-motion:galileo}
\index{lei da queda livre de Galileu}
Em \omterm{def:g12:projectile-motion:freefall}{queda livre}, todos os corpos compartilham a mesma aceleração, seja qual for a massa
deles:
\[
\vect a = \vect g, \qquad g = \qty{9.81}{m/s^2},
\]
dirigida verticalmente para baixo: a intensidade de \omterm{def:g11:electric-gravitational-fields:field}{campo} \qty{9.81}{N/kg} do
\cref{ch:g10:universal-gravitation} na outra roupa dela,
$\unit{N/kg} = \unit{m/s^2}$.
\end{theorem}

\begin{proof}
A \omterm{thm:g12:newtons-laws:second}{segunda lei de Newton} (\cref{ch:g12:newtons-laws}) com o \omterm{def:g10:universal-gravitation:weight}{peso} como única \omterm{def:g10:inertia:force}{força}:
$m\vect a = m\vect g$, e a massa se cancela. Martelo e pena diferem apenas por causa
do ar.
\end{proof}

\begin{proposition}[Queda vertical a partir do repouso]\label{prop:g12:projectile-motion:vertical}
Solto do repouso, com a profundidade de queda $h$ contada para baixo a partir do ponto
de soltura, um corpo em \omterm{def:g12:projectile-motion:freefall}{queda livre} obedece a
\[
a = g, \qquad v = g t, \qquad h = \tfrac12\, g t^2 .
\]
\end{proposition}

\begin{proof}
$v(t)$ é a primitiva da constante $a = g$ que se anula em $t = 0$
(\cref{ch:g12:kinematics-2d}); $h(t)$ é a de $v = gt$ que se anula em $t = 0$.
\end{proof}

\begin{example}[Ordens de grandeza]\label{ex:g12:projectile-motion:orders}
Após $1$, $2$, $3$ segundos de \omterm{def:g12:projectile-motion:freefall}{queda livre}: $v = 9.8$, $19.6$, \qty{29.4}{m/s}
($35$, $71$, \qty{106}{km/h}) e $h = 4.9$, $19.6$, \qty{44.1}{m}. De um telhado de
\qty{30}{m}: $t = \sqrt{2h/g} \approx \qty{2.5}{s}$, chegando a \qty{24}{m/s}
$\approx \qty{87}{km/h}$. Os tempos crescem como $\sqrt h$ --- e, passados alguns
segundos, o ar se recusa a continuar desprezível (última seção).
\end{example}

\begin{example}[Lançamento vertical]\label{ex:g12:projectile-motion:throw}
Lançado verticalmente para cima com $v_0$, eixo $Oy$ para cima a partir da mão: as
mesmas duas primitivas dão $v = v_0 - gt$ e $y = v_0 t - \tfrac12 gt^2$. O ponto mais
alto, onde $v = 0$, vem em $t = v_0/g$, na altura $h = v_0^2/2g$ --- a resposta
energética $v_0 = \sqrt{2gh}$ do \cref{ch:g11:mechanical-energy} lida ao contrário;
lançado \emph{para baixo}, $v_0$ troca de sinal. Para \qty{15}{m/s}: ponto mais alto a
\qty{11.5}{m} em \qty{1.53}{s}, de volta em \qty{3.06}{s}, a \qty{15}{m/s}.
\end{example}

\section{O projétil: uma lei, duas primitivas}

\begin{definition}[Projétil]\label{def:g12:projectile-motion:projectile}
Um \emph{projétil}\index{projétil} é um corpo lançado com uma velocidade $\vect v_0$ e
depois entregue à \omterm{def:g12:projectile-motion:freefall}{queda livre}. Eixos no ponto de lançamento: $Ox$ horizontal no plano
vertical de $\vect v_0$, $Oy$ vertical para cima; o \emph{ângulo de
lançamento}\index{ângulo de lançamento} $\alpha$ de $\vect v_0$ acima da horizontal
fixa as coordenadas dele, $(v_0\cos\alpha,\; v_0\sin\alpha)$.
\end{definition}

\begin{theorem}[Equações do movimento]\label{thm:g12:projectile-motion:equations}
Um \omterm{def:g12:projectile-motion:projectile}{projétil} lançado da origem com \omterm{def:g12:kinematics-2d:velocity}{velocidade escalar} $v_0$ e ângulo $\alpha$ tem
$\vect a = (0,\,-g)$ a cada instante, logo
\[
\vect v(t) = \bigl(v_0\cos\alpha,\; v_0\sin\alpha - g t\bigr),
\qquad
\begin{cases}
x(t) = v_0\cos\alpha\; t,\\[2pt]
y(t) = v_0\sin\alpha\; t - \tfrac12\, g t^2 .
\end{cases}
\]
\end{theorem}

\begin{proof}
A \omterm{thm:g12:newtons-laws:second}{segunda lei de Newton} dá $\vect a = \vect g = (0,-g)$. Primitivas eixo a eixo:
$\vect v(t)$ a partir de $\vect a$ com valor inicial
$(v_0\cos\alpha,\, v_0\sin\alpha)$, e depois a posição a partir de $\vect v$, saindo
de $(0,0)$.
\end{proof}

\begin{proposition}[Independência dos dois movimentos]\label{prop:g12:projectile-motion:independence}
O movimento horizontal é uniforme --- $v_x$ nunca muda --- e o movimento vertical é
exatamente o lançamento vertical da seção anterior: nenhuma das equações menciona a
coordenada da outra. Em particular, uma bola disparada horizontalmente e uma bola
solta no mesmo instante da mesma altura permanecem lado a lado --- elas chegam ao chão
juntas.
\end{proposition}

\begin{proof}
Na \cref{thm:g12:projectile-motion:equations}, $x(t)$ envolve apenas $v_0\cos\alpha$,
e $y(t)$ apenas $v_0\sin\alpha$ e $g$; as duas bolas têm $v_y(0) = 0$, de modo que as
duas obedecem a $h = \tfrac12 gt^2$.
\end{proof}

\begin{remark}[O macaco que se solta]\label{rem:g12:projectile-motion:monkey}
Um tratador mira o dardo \emph{exatamente no} macaco, que larga o galho no instante do
disparo. Má ideia: dardo e macaco caem cada um $\tfrac12 gt^2$ abaixo das posições que
teriam sem gravidade --- a linha de mira, o galho --- de modo que eles se encontram
exatamente, seja qual for a velocidade do dardo
(\cref{exo:g12:projectile-motion:15}).
\end{remark}

\section{A parábola, lida por inteiro}

\begin{theorem}[Trajetória]\label{thm:g12:projectile-motion:trajectory}
Eliminando $t$ entre $x(t)$ e $y(t)$: a \omterm{def:g10:relative-motion:trajectory}{trajetória} de um \omterm{def:g12:projectile-motion:projectile}{projétil}
($0 \leq \alpha < \ang{90}$) é a parábola de concavidade para baixo
\[
y = x\tan\alpha \;-\; \frac{g}{2 v_0^2 \cos^2\alpha}\, x^2 .
\]
\end{theorem}

\begin{proof}
$t = x/(v_0\cos\alpha)$, substituído em $y(t)$ da
\cref{thm:g12:projectile-motion:equations}.
\end{proof}

\begin{definition}[Alcance]\label{def:g12:projectile-motion:range}
Para um \omterm{def:g12:projectile-motion:projectile}{projétil} sobre terreno plano, o \emph{alcance}\index{alcance} $R$ é a
distância horizontal percorrida quando ele volta à altura de lançamento; o
\emph{tempo de voo}\index{tempo de voo} $t_f$ é a duração dessa viagem.
\end{definition}

\begin{proposition}[Ponto mais alto, tempo de voo, alcance]\label{prop:g12:projectile-motion:range}
Sobre terreno plano, com $0 < \alpha < \ang{90}$:
\[
t_f = \frac{2 v_0 \sin\alpha}{g}, \qquad
h_{\text{max}} = \frac{(v_0\sin\alpha)^2}{2g}
\ \ \text{atingido em } t_f/2, \qquad
R = \frac{v_0^2 \sin(2\alpha)}{g}.
\]
O \omterm{def:g12:projectile-motion:range}{alcance} é máximo em $\alpha = \ang{45}$, onde $R = v_0^2/g$; ângulos complementares
$\alpha$ e $\ang{90} - \alpha$ dão o mesmo \omterm{def:g12:projectile-motion:range}{alcance}.
\end{proposition}

\begin{proof}
$y = 0$ dá $t\,(v_0\sin\alpha - \tfrac12 gt) = 0$: o lançamento, e $t_f$. O ponto mais
alto, $v_y = 0$, fica em $t = v_0\sin\alpha/g = t_f/2$; substitua em $y(t)$. Então
$R = x(t_f) = 2v_0^2\sin\alpha\cos\alpha/g = v_0^2\sin(2\alpha)/g$. O seno é máximo em
$2\alpha = \ang{90}$, e $\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180} -
2\alpha) = \sin(2\alpha)$.
\end{proof}

\begin{example}[Tiro de meta]\label{ex:g12:projectile-motion:kick}
Um goleiro chuta a $v_0 = \qty{25}{m/s}$, $\alpha = \ang{45}$:
$R = 25^2/9.81 \approx \qty{64}{m}$; os chutes reais caem mais perto de \qty{55}{m}
--- é a mordida do ar.
\end{example}

\begin{method}[Contabilidade de um projétil]\label{met:g12:projectile-motion:method}
\begin{enumerate}
  \item Eixos no ponto de lançamento, $Ox$ horizontal, $Oy$ para cima; coordenadas de
        $\vect v_0$.
  \item Confirme a \omterm{def:g12:projectile-motion:freefall}{queda livre} (só o \omterm{def:g10:universal-gravitation:weight}{peso}): então $\vect a = (0,-g)$; duas primitivas,
        com as condições iniciais a cada vez.
  \item Pergunta sobre um \emph{lugar}: elimine $t$ (\omterm{def:g10:relative-motion:trajectory}{trajetória}). Sobre um
        \emph{instante}: mantenha $t$.
  \item Dicionário: ponto mais alto $\Leftrightarrow v_y = 0$; altura de lançamento
        $\Leftrightarrow y = 0$; muro a $d \Leftrightarrow x = d$; chão $\ell$ abaixo
        $\Leftrightarrow y = -\ell$.
  \item Bom senso: voo simétrico em torno do ponto mais alto; $R \leq v_0^2/g$; o ar
        só encurta.
\end{enumerate}
\end{method}

\section{O que o ar muda}

\begin{definition}[Velocidade limite]\label{def:g12:projectile-motion:terminal}
A resistência do ar se opõe à velocidade e cresce com ela. Um corpo em queda,
portanto, acelera apenas até que o arrasto equilibre o \omterm{def:g10:universal-gravitation:weight}{peso}; a partir daí
$\vect a = \vect 0$: ele cai com uma \emph{velocidade limite}\index{velocidade limite}
constante.
\end{definition}

\begin{example}[Ordens de grandeza da velocidade limite]\label{ex:g12:projectile-motion:airorders}
Um paraquedista de corpo aberto: cerca de \qty{55}{m/s} (\qty{200}{km/h}). Uma gota de
chuva: cerca de \qty{9}{m/s} --- no vácuo, \qty{2000}{m} de queda a entregariam a
$\sqrt{2gh} \approx \qty{2.0e2}{m/s}$. Uma pena: cerca de \qty{1}{m/s}, atingida em
alguns centímetros --- o segredo da lentidão dela: ela passa a queda inteira \emph{na}
\omterm{def:g12:projectile-motion:terminal}{velocidade limite}; o martelo nem chega perto da sua.
\end{example}

\begin{remark}[Balística de verdade]\label{rem:g12:projectile-motion:drag}
O arrasto empurra contra a velocidade ao longo de todo o voo: o \omterm{def:g12:projectile-motion:range}{alcance} encurta, o
ponto mais alto baixa e a simetria se quebra --- a descida mais íngreme do que a
subida, e os tiros longos visivelmente aquém da parábola. O tratamento honesto e
quantitativo é uma equação diferencial resolvida no volume do primeiro ano de
graduação; este ano, a parábola é o modelo certo para projéteis densos e de velocidade
moderada --- uma bola de basquete a \qty{10}{m/s}, e não uma bola de golfe a
\qty{70}{m/s}.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:1}
Uma pedra solta de uma ponte atinge a água \qty{2.0}{s} depois: qual é a altura da
ponte e a velocidade de impacto em \unit{m/s} e em \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:2}
Na Lua, um astronauta soltou juntos um martelo e uma pena: eles chegaram juntos. Por
quê --- e por que a pena perde tão feio na Terra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:3}
Desprezando o ar, calcule o tempo de queda e a velocidade de chegada de \qty{1.0}{m},
\qty{10}{m} e \qty{100}{m}. Como os tempos crescem com a altura?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:4}
Uma bola é lançada verticalmente para cima a \qty{15}{m/s}: qual é a altura e o
instante do ponto mais alto, o tempo total até voltar à mão e a velocidade de retorno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:5}
Uma bolinha de gude rola para fora de uma mesa de \qty{0.80}{m} de altura a
\qty{2.5}{m/s}. Calcule o tempo de queda e a distância do ponto de chegada até a
borda. O tempo de queda depende da velocidade da bolinha?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:6}
Na figura dos cinco ângulos do curso: que pares de \omterm{def:g12:projectile-motion:projectile}{ângulos de lançamento}
compartilham o \omterm{def:g12:projectile-motion:range}{alcance}, e que ângulo vence? Para um \omterm{def:g12:projectile-motion:range}{alcance} abaixo do máximo, qual dos
dois ângulos possíveis voa por mais tempo? Justifique com $t_f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:7}
De uma sacada a \qty{20}{m} uma bola é lançada verticalmente \emph{para baixo} a
\qty{5.0}{m/s}. Calcule o tempo até o chão e a velocidade de impacto; confira a última
com um balanço de \omterm{def:g10:energy-conservation:energy}{energia} (\cref{ch:g11:mechanical-energy}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:8}
Um \omterm{def:g12:projectile-motion:projectile}{projétil} parte do solo plano a \qty{40}{m/s}, $\alpha = \ang{30}$: \omterm{def:g12:projectile-motion:range}{tempo de voo},
altura do ponto mais alto e \omterm{def:g12:projectile-motion:range}{alcance}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:9}
Modele um saltador em distância como um \omterm{def:g12:projectile-motion:projectile}{projétil}: impulsão a \qty{10.5}{m/s},
$\alpha = \ang{20}$. Calcule o \omterm{def:g12:projectile-motion:range}{alcance}; os campeões saltam quase \qty{9}{m} ---
aponte dois ingredientes que o modelo pontual deixa de fora.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:10}
A \qty{1.2}{m} acima de um solo plano, uma bala é solta e outra idêntica é disparada
horizontalmente a \qty{300}{m/s}. Qual chega primeiro? Calcule o tempo de queda e a

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:11}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:12}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:13}
Uma bola é chutada do solo a \qty{18}{m/s}, $\alpha = \ang{35}$, na direção de uma

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:14}
Um paraquedista de \qty{80}{kg} cai de corpo aberto com \omterm{def:g12:projectile-motion:terminal}{velocidade limite} de
\qty{55}{m/s}. (a) Qual é a \omterm{def:g10:inertia:force}{força} de arrasto? (b) Esboce a forma de $v(t)$ desde o
salto. (c) Uma gota de chuva cai de \qty{2.0}{km}: qual seria a velocidade de chegada
no vácuo, comparada com os \qty{9}{m/s} reais dela?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:15}
O macaco que se solta, demonstrado: um dardo parte da origem mirando exatamente um
macaco à distância $D$ e à altura $H$ ($\tan\alpha = H/D$), que se solta em $t = 0$.
Mostre que, quando o dardo chega a $x = D$, a altura dele é igual à do macaco ---
seja qual for $v_0$.
\end{exercise}

\section{Problema: a cesta no estouro do cronômetro}

\begin{problem}[{Problema de fim de semana --- a cesta no estouro do cronômetro: um
lance livre dissecado flecha por flecha, a bola passando centímetros acima dos dedos
do marcador, um arremesso de quadra inteira pesado contra o que um braço consegue dar,
um balão por cima da tabela --- e um veredito sobre o arremesso mais teatral do
esporte}]
\label{pb:g12:projectile-motion:1}
O aro de uma cesta de basquete é um anel horizontal a \qty{3.05}{m} do piso; Nora
solta todo arremesso a \qty{2.00}{m}, de modo que, com os eixos no ponto de soltura
($Ox$ horizontal em direção à cesta, $Oy$ para cima), o centro do aro fica
\qty{1.05}{m} acima da origem. O ar é desprezado, salvo onde se disser o contrário.

\medskip\noindent\textbf{Parte I --- O lance livre.} Nora arremessa de
$d = \qty{4.20}{m}$ (na horizontal) com \omterm{def:g12:projectile-motion:projectile}{ângulo de lançamento} $\alpha = \ang{50}$.
\begin{enumerate}
  \item Que \omterm{def:g10:inertia:force}{forças} agem depois da soltura? Dê $\vect a$, citando a lei usada.
  \item Deduza as coordenadas de $\vect v(t)$.
  \item Deduza $x(t)$ e $y(t)$.
  \item Elimine $t$ para obter a \omterm{def:g10:relative-motion:trajectory}{trajetória} $y(x)$.
  \item A bola precisa passar pelo centro do aro, $y(d) = \qty{1.05}{m}$: resolva para
        $v_0$, em \unit{m/s} e em \unit{km/h}.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Por cima dos dedos, dentro do anel.}
\begin{enumerate}[resume]
  \item Quanto tempo a bola leva para chegar ao aro?
  \item Calcule o instante e a altura do ponto mais alto; verifique que a bola já está
        descendo no aro.
  \item Um marcador a \qty{0.90}{m} à frente de Nora estica os dedos até
        \qty{2.60}{m}. Por quanto a bola passa acima deles?
  \item Calcule $v_x$ e $v_y$ no aro, a \omterm{def:g12:kinematics-2d:velocity}{velocidade escalar} ali e o ângulo da
        velocidade abaixo da horizontal.
  \item Por que a bola precisa chegar \emph{descendo} --- e por que íngreme?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O arremesso de quadra inteira.} Com o
cronômetro prestes a soar, Nora está a \qty{26.0}{m} do aro do outro lado.
\begin{enumerate}[resume]
  \item A subida de \qty{1.05}{m} é pequena diante de \qty{26.0}{m}: justifique um
        modelo de terreno plano e deduza o \omterm{def:g12:projectile-motion:range}{tempo de voo} a partir de $y(t_f) = 0$.
  \item Deduza $R = v_0^2\sin(2\alpha)/g$ e o ângulo de maior \omterm{def:g12:projectile-motion:range}{alcance}.
  \item Calcule o $v_0$ necessário para cobrir \qty{26.0}{m} em $\alpha = \ang{45}$.
  \item Uma jogadora forte consegue arremessar uma bola de basquete a cerca de
        \qty{17.5}{m/s}. Qual é o \omterm{def:g12:projectile-motion:range}{alcance} máximo dela --- o arremesso é humanamente
        viável?
  \item A \qty{17.5}{m/s}, encontre os \emph{dois} ângulos que fazem a bola cair a
        \qty{26.0}{m} de distância, e os \omterm{def:g12:projectile-motion:range}{tempos de voo} deles: com \qty{2.0}{s} no
        relógio, qual bate o cronômetro?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- O balão por cima da tabela.} De trás da linha de
fundo Nora solta um balão: soltura a \qty{6.0}{m} do centro do aro,
$\alpha = \ang{75}$; o plano da tabela cruza o arremesso dela \qty{0.45}{m} antes do
centro do aro, com a borda superior a \qty{3.95}{m} do piso.
\begin{enumerate}[resume]
  \item Encontre o $v_0$ que faz a bola cair pelo centro do aro.
  \item Calcule a altura da bola no plano da tabela ($x = \qty{5.55}{m}$): ela passa
        por cima da borda superior, e por quanto?
  \item Calcule a altura do ponto mais alto e o \omterm{def:g12:projectile-motion:range}{tempo de voo} --- por que a torcida
        adora isso?
  \item Qualitativamente, o que a resistência do ar muda nesses três arremessos, e
        qual deles ela mais ameaça?
  \item \emph{O veredito da cesta no estouro do cronômetro}, em duas frases: compare a
        velocidade que o arremesso de quadra inteira exige com a que um braço fornece,
        dê a margem e julgue --- possível ou lenda?
\end{enumerate}
\end{problem}
