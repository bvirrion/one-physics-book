\chapter{Trabalho e energia mecânica}\label{ch:g12:work-and-energy}

Deixe uma bolinha de gude cair numa saladeira: ela mergulha, passa do fundo, sobe até a
altura de partida, hesita e volta --- e cada inversão pode ser prevista a partir de uma
curva (o perfil da tigela) e de uma reta horizontal (a \omterm{def:g10:energy-conservation:energy}{energia} da bolinha), sem
resolver nenhuma equação do movimento. O ano passado precificou \omterm{def:g10:inertia:force}{forças} em linhas retas
(\cref{ch:g11:work-of-force}) e equilibrou \omterm{def:g11:mechanical-energy:kinetic}{energia cinética} contra \omterm{def:g12:work-and-energy:ep}{energia potencial}
(\cref{ch:g11:mechanical-energy}); este ano a contabilidade vai a todos os lugares:
caminhos curvos, molas, paisagens lidas de relance.

\section{Trabalho ao longo de um caminho qualquer}

\begin{definition}[Trabalho ao longo de um caminho]\label{def:g12:work-and-energy:path}
Seja uma \omterm{def:g10:inertia:force}{força} $\vect F$ agindo sobre um corpo que se move por um caminho qualquer de
$A$ até $B$: corte o caminho em deslocamentos $\vect{\dd\ell}$ tão curtos que cada um
seja reto e $\vect F$ mal mude ao longo dele. Cada passo rende o pequeno \omterm{def:g11:work-of-force:work}{trabalho}
$\scal{F}{\dd\ell} = F\,\dd\ell\cos\theta$ (\cref{ch:g11:work-of-force}), e o
\emph{trabalho ao longo do caminho}\index{trabalho (de uma força)!ao longo de um
caminho} é a soma deles, $W_{AB}(\vect F) = \sum \scal{F}{\dd\ell}$ --- que, num
caminho reto com \omterm{def:g10:inertia:force}{força} constante, se reduz ao $F \times AB \times \cos\theta$ do ano
passado.
\end{definition}

\begin{proposition}[Trabalho do peso, por qualquer caminho]\label{prop:g12:work-and-energy:weight}
Ao longo de \emph{qualquer} caminho de $A$ (altitude $z_A$) até $B$ (altitude $z_B$), o
\omterm{def:g10:universal-gravitation:weight}{peso} de uma massa $m$ realiza o \omterm{def:g11:work-of-force:work}{trabalho} $W_{AB}(\vect P) = mg\,(z_A - z_B)$: entra o
desnível, nunca a rota.
\end{proposition}

\begin{proof}
Cada passo curto contribui com $mg \times (\text{pequeno desnível})$ (o cálculo em
segmentos retos do ano passado); os desníveis se telescopam em $z_A - z_B$. O caso
curvo, admitido então, está entregue.
\end{proof}

\begin{definition}[Forças conservativas e não conservativas]\label{def:g12:work-and-energy:conservative}
Uma \omterm{def:g10:inertia:force}{força} é \emph{conservativa}\index{força conservativa} se o \omterm{def:g11:work-of-force:work}{trabalho} dela entre dois
pontos é o mesmo por qualquer caminho (nulo em toda ida e volta), e
\emph{não conservativa}\index{força não conservativa} em caso contrário. O \omterm{def:g10:universal-gravitation:weight}{peso} e a
\omterm{def:g10:inertia:force}{força} de uma mola (próxima seção): conservativas. O \omterm{def:g10:inertia:inventory}{atrito} de deslizamento não é:
oposto ao movimento, ele realiza $W = -fL$ num caminho de comprimento $L$ --- um
pedágio por \omterm{def:g10:orders-of-magnitude:unit}{metro}, jamais reembolsado.
\end{definition}

\begin{example}[Duas trilhas até o refúgio]\label{ex:g12:work-and-energy:trails}
Uma caminhante de \qty{75}{kg} ganha \qty{300}{m} por uma trilha íngreme de
\qty{800}{m} ou por \qty{2.4}{km} de curvas em ziguezague. O \omterm{def:g10:universal-gravitation:weight}{peso} cobra
$-75 \times 9.81 \times 300 \approx \qty{-2.2e5}{J}$ nas duas; uma \omterm{def:g10:inertia:force}{força} de \omterm{def:g10:inertia:inventory}{atrito} de
\qty{30}{N}, \qty{-2.4e4}{J} numa e \qty{-7.2e4}{J} na outra. O \omterm{def:g10:universal-gravitation:weight}{peso} devolve na
descida; a conta do \omterm{def:g10:inertia:inventory}{atrito} virou calor, nos dois sentidos.
\end{example}

\section{Energia potencial}

\begin{definition}[Energia potencial]\label{def:g12:work-and-energy:ep}
A cada \omterm{def:g12:work-and-energy:conservative}{força conservativa} se associa uma \emph{energia potencial}\index{energia
potencial!a partir do trabalho conservativo} $E_p$, um número que depende apenas da
posição, definido por $W_{AB} = E_p(A) - E_p(B)$: \omterm{def:g11:work-of-force:work}{trabalho} entregue é \omterm{def:g10:energy-conservation:energy}{energia} potencial
gasta. Fixar $E_p = 0$ num ponto de referência escolhido a fixa em todo lugar; a
escolha desloca todos os valores por uma constante que se cancela em toda diferença.
Para o \omterm{def:g10:universal-gravitation:weight}{peso}, isso devolve o $E_p = mgz$ do ano passado, agora válido por qualquer
caminho. O \omterm{def:g10:inertia:inventory}{atrito} não pode ter $E_p$ algum: uma conta que depende da rota não é
diferença de números dos extremos --- só as \omterm{def:g12:work-and-energy:conservative}{forças conservativas} armazenam de modo
recuperável, exatamente o que o nome conserva.
\end{definition}

\begin{proposition}[Energia potencial elástica]\label{prop:g12:work-and-energy:elastic}
Uma mola de \omterm{def:g12:oscillators-and-time:spring}{constante elástica} $k$ (\unit{N/m}) esticada ou comprimida de $x$ puxa de
volta com a \omterm{def:g10:inertia:force}{força} $F = kx$ (\cref{ch:g12:oscillators-and-time}); o \omterm{def:g11:work-of-force:work}{trabalho} que ela
entrega ao voltar ao repouso --- a \emph{energia potencial elástica}\index{energia
potencial!elástica} dela --- é $E_p = \tfrac12\, k x^2$.
\end{proposition}

\begin{proof}
A \omterm{def:g12:oscillators-and-time:pendulum}{força restauradora} age ao longo do movimento, de modo que $\sum F\,\dd\ell$ é a
\emph{área sob o gráfico $F$--$x$}: um triângulo de base $x$ e altura $kx$, de área
$\tfrac12 x \times kx$. A \omterm{def:g10:inertia:force}{força} em cada alongamento é a mesma na ida e na volta:
\omterm{def:g11:work-of-force:work}{trabalho} fixado pelos extremos, \omterm{def:g10:inertia:force}{força} da mola \omterm{def:g12:work-and-energy:conservative}{conservativa}, $E_p$ bem definida.
\end{proof}

\section{Os dois teoremas da energia}

\begin{theorem}[Teorema da energia cinética]\label{thm:g12:work-and-energy:ketheorem}\index{teorema da energia cinética}
Para qualquer movimento de um corpo de massa $m$ de $A$ até $B$ --- curvo, em laço, em
três dimensões ---
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
com a soma percorrendo todas as \omterm{def:g10:inertia:force}{forças} que agem sobre o corpo.
\end{theorem}

\begin{proof}
Como $v^2 = \scal{v}{v}$, a regra do produto da matemática deste ano dá a derivada de
$v^2$ como $2\,\scal{a}{v}$, de modo que
$\dd E_k/\dd t = m\,\scal{a}{v} = \scal{F_{\text{tot}}}{v}$ pela \omterm{thm:g12:newtons-laws:second}{segunda lei de Newton}
(\cref{ch:g12:newtons-laws}). Num $\dd t$ curto o corpo se desloca
$\vect{\dd\ell} = \vect v\,\dd t$: $E_k$ ganha o pequeno \omterm{def:g11:work-of-force:work}{trabalho} da
\cref{def:g12:work-and-energy:path}; somar os passos soma os \omterm{def:g11:work-of-force:work}{trabalhos} --- o enunciado
admitido no ano passado, agora conquistado por inteiro.
\end{proof}

\begin{proposition}[Potência instantânea]\label{prop:g12:work-and-energy:power}
A cada instante uma \omterm{def:g10:inertia:force}{força} $\vect F$ sobre um corpo de velocidade $\vect v$ entrega a
\omterm{def:g11:work-of-force:power}{potência} $P = \scal{F}{v} = Fv\cos\theta$, e $\dd E_k/\dd t$ é a soma das \omterm{def:g11:work-of-force:power}{potências} de
todas as \omterm{def:g10:inertia:force}{forças}: a fórmula de velocidade constante do ano passado, exata a cada
instante de qualquer movimento.
\end{proposition}

\begin{proof}
Leia na demonstração anterior: cada \omterm{def:g10:inertia:force}{força} injeta
$\scal{F}{\dd\ell} = \scal{F}{v}\,\dd t$ em $E_k$ durante $\dd t$.
\end{proof}

\begin{theorem}[Teorema da energia mecânica]\label{thm:g12:work-and-energy:emtheorem}\index{teorema da energia mecânica}
Seja $E_m = E_k + E_p$, onde $E_p$ reúne as \omterm{def:g12:work-and-energy:ep}{energias potenciais} de todas as \omterm{def:g12:work-and-energy:conservative}{forças
conservativas} em ação. Então, por qualquer caminho,
\[
\Delta E_m = W_{AB}(\text{forças não conservativas}),
\]
e $E_m$ se conserva sempre que as \omterm{def:g12:work-and-energy:conservative}{forças não conservativas} não realizam \omterm{def:g11:work-of-force:work}{trabalho}.
\end{theorem}

\begin{proof}
Divida o \omterm{thm:g11:mechanical-energy:ketheorem}{teorema da energia cinética}:
$\Delta E_k = W_{\text{cons}} + W_{\text{nc}} = -\Delta E_p + W_{\text{nc}}$; passe
$\Delta E_p$ para o outro lado.
\end{proof}

\begin{example}[O pêndulo, desta vez com honestidade]\label{ex:g12:work-and-energy:pendulum}
Um corpo de pêndulo num fio de comprimento $L = \qty{2.5}{m}$ é solto em repouso a
$\qty{45}{\degree}$ da vertical. A tensão, perpendicular ao movimento, não tem
\omterm{def:g11:work-of-force:power}{potência}; o \omterm{def:g10:universal-gravitation:weight}{peso} é conservativo: $E_m$ se conserva \emph{sobre o arco curvo}, o que no
ano passado só podia ser conferido. O corpo parte $h = L(1 - \cos\qty{45}{\degree}) =
\qty{0.73}{m}$ acima, de modo que embaixo $v = \sqrt{2gh} \approx \qty{3.8}{m/s}$.
\end{example}

\begin{method}[Auditoria energética de um movimento qualquer]\label{met:g12:work-and-energy:audit}
\begin{enumerate}
  \item Escolha o sistema e liste as \omterm{def:g10:inertia:force}{forças} sobre ele.
  \item Classifique: perpendicular ao movimento $\to$ \omterm{def:g11:work-of-force:work}{trabalho} nulo; \omterm{def:g12:work-and-energy:conservative}{conservativa}
        $\to$ uma $E_p$ dentro de $E_m$; o resto $\to$ $W_{\text{nc}}$ (\omterm{def:g10:inertia:inventory}{atrito} de
        deslizamento: $-fL$, com $L$ o comprimento do caminho).
  \item Escreva $\Delta E_m = W_{\text{nc}}$ entre o ponto dos dados e o ponto da
        pergunta; resolva. A \omterm{def:g10:energy-conservation:energy}{energia} responde ``com que velocidade, a que altura'';
        só o tempo e a aceleração precisam das leis de Newton.
\end{enumerate}
\end{method}

\section{Diagramas de energia}

\begin{definition}[Diagrama de energia, pontos de retorno]\label{def:g12:work-and-energy:diagram}
Para um movimento ao longo de um eixo $x$ com $E_m$ conservada, o \emph{diagrama de
energia}\index{diagrama de energia} traça a curva $E_p(x)$ e a reta horizontal $E_m$.
Como $E_k = E_m - E_p \geq 0$, o movimento fica confinado ao trecho em que a curva está
abaixo da reta; a distância entre as duas é $E_k$, lida diretamente. Num \emph{ponto de
retorno}\index{ponto de retorno}, onde a curva encontra a reta, $v = 0$ e o movimento
se inverte.
\end{definition}

\begin{proposition}[Força a partir da paisagem; equilíbrios]\label{prop:g12:work-and-energy:equilibria}
A \omterm{def:g12:work-and-energy:conservative}{força conservativa} ao longo de $x$ é menos a inclinação de $E_p$:
$F = -\,\dd E_p/\dd x$ --- ela aponta \emph{ladeira abaixo} no diagrama e se anula onde
a tangente é horizontal: um equilíbrio. Um mínimo de $E_p$ é
\emph{estável}\index{equilíbrio estável} --- o corpo afastado é empurrado de volta, uma
bola num vale; um máximo é \emph{instável}\index{equilíbrio instável} --- ele é
empurrado para longe, uma bola no alto de uma colina.
\end{proposition}

\begin{proof}
Num passo pequeno $\dd x$ a \omterm{def:g10:inertia:force}{força} realiza $F\,\dd x = E_p(x) - E_p(x + \dd x) =
-\dd E_p$; divida por $\dd x$. A descida dos dois lados de um vale aponta para dentro;
em torno do alto de uma colina, para fora.
\end{proof}

\begin{remark}[Oscilações e escape]\label{rem:g12:work-and-energy:escape}
As pequenas oscilações do \cref{ch:g12:oscillators-and-time} vivem no fundo dos vales
de $E_p$. E, longe da Terra, a gravidade enfraquece: a curva $E_p(r)$ sobe cada vez
mais devagar em direção a um teto finito (a área sob o gráfico da \omterm{def:g10:inertia:force}{força} que encolhe é
finita). Um foguete cujo $E_m$ \omterm{def:g12:projectile-motion:range}{alcance} o teto nunca encontra um \omterm{def:g12:work-and-energy:diagram}{ponto de retorno}: ele
\emph{escapa}\index{velocidade de escape} --- cerca de \qty{11.2}{km/s} a partir do
solo (\cref{exo:g12:work-and-energy:13}); abaixo disso ele fica preso. O próprio teto é
calculado no volume do primeiro ano de graduação.
\end{remark}

\begin{example}[O looping]\label{ex:g12:work-and-energy:loop}
Uma bolinha solta da altura $H$ chega embalada a um looping vertical de raio $R$. A
dinâmica circular (\cref{ch:g12:newtons-laws}) exige $v_{\text{alto}}^2 \geq gR$ no
alto --- o dado emprestado pelo problema da montanha-russa do ano passado, agora nosso.
A conservação do repouso em $H$ até o alto (altura $2R$) dá
$v_{\text{alto}}^2 = 2g(H - 2R)$, logo $H \geq \tfrac52 R$: meio raio acima do topo do
looping, seja qual for a massa --- um pouco mais no mundo real, que raspa.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:1}
Uma mola de \omterm{def:g12:oscillators-and-time:spring}{constante elástica} $k = \qty{200}{N/m}$ é esticada de \qty{5.0}{cm}.
Calcule a \omterm{def:g10:energy-conservation:energy}{energia} armazenada; no que ela se transforma se o alongamento dobrar? Quanto
custam os \emph{segundos} \qty{5.0}{cm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:2}
Um drone de \qty{0.90}{kg} voa por uma rota sinuosa de um terraço a $z = \qty{12}{m}$
até uma sacada a $z = \qty{30}{m}$. Qual é o \omterm{def:g11:work-of-force:work}{trabalho} do \omterm{def:g10:universal-gravitation:weight}{peso} dele? E pela subida em
linha reta? E na ida e volta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:3}
\omterm{def:g12:work-and-energy:conservative}{Conservativa}, \omterm{def:g12:work-and-energy:conservative}{não conservativa} ou sem \omterm{def:g11:work-of-force:work}{trabalho}? Justifique a partir dos caminhos: (a)
o \omterm{def:g10:universal-gravitation:weight}{peso}; (b) a \omterm{def:g10:inertia:force}{força} de uma mola; (c) o \omterm{def:g10:inertia:inventory}{atrito} de deslizamento; (d) a \omterm{def:g12:newtons-laws:friction}{força normal}
sobre uma caixa que desliza; (e) o arrasto do ar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:4}
Um pêndulo de comprimento \qty{2.0}{m} é solto em repouso a $\qty{60}{\degree}$ da
vertical. Encontre a velocidade dele no ponto mais baixo. Por que a tensão do fio nunca
entra no balanço?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:5}
Uma ciclista anda a \qty{9.0}{m/s} no plano, entregando \qty{250}{W}: que \omterm{def:g10:inertia:force}{força}
resistiva total ela combate? Subindo a \qty{5.0}{m/s} com a mesma \omterm{def:g11:work-of-force:power}{potência}: que \omterm{def:g10:inertia:force}{força}
ela vence?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:work-and-energy:6}

\section{Problema: o salto de esqui}

\begin{problem}[{Problema de fim de semana --- o salto de esqui: uma rampa de impulso
auditada por um radar, um voo em queda livre, a geometria gentil de uma aterrissagem
íngreme e a torre que um projetista precisa comprar quando o atrito manda a conta}]
\label{pb:g12:work-and-energy:1}
Um saltador de esqui olímpico ($m = \qty{70}{kg}$, esquis incluídos) parte do repouso
de um portão $H = \qty{45}{m}$ acima da mesa de saída, desliza $L = \qty{90}{m}$ de
rampa de impulso curva e deixa a mesa horizontalmente; ali um \omterm{met:g10:signals-and-waves:echo}{radar} marca
$v_0 = \qty{26}{m/s}$. O ponto de aterrissagem fica $h = \qty{40}{m}$ mais baixo, numa
encosta de $\qty{35}{\degree}$. As Partes II e III desprezam a resistência do ar.

\medskip\noindent\textbf{Parte I --- A rampa de impulso.}
\begin{enumerate}
  \item Preveja a velocidade de saída sem \omterm{def:g10:inertia:inventory}{atrito}; por que a curvatura da rampa é
        irrelevante?
  \item Com a referência na mesa: calcule $E_m$ no portão e na saída.
  \item Calcule $\Delta E_m$; que teorema nomeia os culpados, e quem são eles?
  \item Deduza a \omterm{def:g10:inertia:force}{força} resistiva média (neve mais ar) ao longo da descida.
  \item Que fração de $E_m$ se perde? Por que encerar os esquis e se agachar?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O voo.}
\begin{enumerate}[resume]
  \item Só o \omterm{def:g10:universal-gravitation:weight}{peso} age durante o voo: que movimento do
        \cref{ch:g12:projectile-motion} é esse? Quais são as componentes da velocidade
        na aterrissagem?
  \item Calcule a velocidade de aterrissagem duas vezes: pelas componentes e pela
        conservação de $E_m$. E em \unit{km/h}?
  \item Calcule o \omterm{def:g12:projectile-motion:range}{tempo de voo} e a distância horizontal percorrida.
  \item Saltadores reais cavalgam o ar como uma asa e voam muito mais longe: o ar pode
        \emph{aumentar} a velocidade de aterrissagem? Argumente com o \omterm{thm:g12:work-and-energy:emtheorem}{teorema da
        energia mecânica}.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A aterrissagem.}
\begin{enumerate}[resume]
  \item Calcule o ângulo da velocidade de aterrissagem abaixo da horizontal.
  \item Decomponha-a em componentes paralela e perpendicular à encosta.
  \item As pernas absorvem apenas a parte perpendicular: calcule essa \omterm{def:g11:mechanical-energy:kinetic}{energia cinética};
        compare-a com a total.
  \item Um terreno plano absorveria tudo: a que queda vertical equivale cada
        aterrissagem ($h_{\text{eq}} = v_\perp^2/2g$ e, respectivamente, $v^2/2g$)?
  \item Em uma frase: por que as rampas de aterrissagem são íngremes e curvas?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- A torre do projetista.}
Uma rampa menor precisa de saída a apenas $v_0' = \qty{24}{m/s}$; a rampa de impulso
reta dela desce a $\qty{35}{\degree}$, de modo que uma altura de portão $H'$ significa
um comprimento de pista $H'/\sin\qty{35}{\degree}$; mesmo arrasto de \qty{80}{N}.
\begin{enumerate}[resume]
  \item Calcule a altura do portão que um projetista sem \omterm{def:g10:inertia:inventory}{atrito} construiria.
  \item Escreva o \omterm{thm:g12:work-and-energy:emtheorem}{teorema da energia mecânica}, do portão até a mesa, para $H'$.
  \item Resolva para $H'$.
  \item Em que porcentagem o \omterm{def:g10:inertia:inventory}{atrito} ergueu a torre?
  \item A massa não se cancela mais: recalcule $H'$ para uma saltadora de \qty{60}{kg}.
        Quem precisa da torre mais alta, e por quê?
  \item O arremate: a torre anunciada e a fatia dela que cabe ao \omterm{def:g10:inertia:inventory}{atrito}.
\end{enumerate}
\end{problem}
