\chapter{Decaimento radioativo}\label{ch:g12:radioactive-decay}

No porão de um museu, um contador Geiger estala sobre uma lasca de madeira do caixão de
um faraó. Cada estalo é sem lei: nada o anunciou, nada prevê o próximo. E, ainda assim,
na sexta-feira o laboratório vai imprimir uma data com precisão de um século. Este
capítulo resolve o paradoxo --- como eventos perfeitamente aleatórios um a um somam, aos
trilhões, os relógios mais estáveis que possuímos.

\section{Sem lei um a um, relojoaria por mol}

No ano passado (\cref{ch:g11:nucleus-radioactivity}) encontramos os \omterm{def:g11:fundamental-interactions:ladder}{núcleos} instáveis,
equilibramos as equações $\alpha$ e $\beta$ deles e vimos metade de qualquer amostra
grande sumir a cada \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida} $T_{1/2}$. O que não podíamos dizer era \emph{quando,
entre os degraus} --- nem por que um processo aleatório cumpre horário. As duas
perguntas têm uma só resposta.

\begin{definition}[Constante de decaimento]\label{def:g12:radioactive-decay:lambda}
Para cada \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo} instável existe uma constante $\lambda$, a \emph{constante de
decaimento}\index{constante de decaimento} dele (\omterm{def:g10:orders-of-magnitude:unit}{unidade} \unit{s^{-1}}), tal que,
durante qualquer intervalo curto $\dd t$, cada \omterm{def:g11:fundamental-interactions:ladder}{núcleo} tem probabilidade
$p = \lambda\,\dd t$ de decair --- a mesma para todo \omterm{def:g11:fundamental-interactions:ladder}{núcleo} da espécie, seja qual for a
idade dele, e intocada pela temperatura, pela \omterm{def:g10:pressure:pressure}{pressão} ou pela química.
\end{definition}

\begin{remark}[Sem lei um a um, exato em multidão]\label{rem:g12:radioactive-decay:crowds}
Um \omterm{def:g11:fundamental-interactions:ladder}{núcleo} que já esperou dez mil anos tem exatamente a mesma chance de decair neste
segundo que um feito hoje de manhã: sem desgaste, sem aviso --- qual vai a seguir é
genuinamente imprevisível. Multidões são outra história. Jogue $100$ moedas: espere $50$
caras, mas não se espante com $43$ --- flutuações da ordem de $\sqrt n$, aqui $10\%$.
Jogue $10^{20}$: a flutuação relativa $1/\sqrt n$ é $10^{-10}$, e o resultado é certo
até a décima casa. Um grama de matéria guarda uns $10^{22}$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}: a falta de lei se
promedia em relojoaria, uma lei exata para $N(t)$, o número de \omterm{def:g11:fundamental-interactions:ladder}{núcleos} ainda intactos no
instante $t$.
\end{remark}

\section{A lei do decaimento}

\begin{proposition}[A equação de evolução]\label{prop:g12:radioactive-decay:ode}
Numa amostra grande de $N$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}, durante $\dd t$ o número muda de
\[
\dd N = -\lambda N\,\dd t, \qquad\text{ou seja,}\qquad \frac{\dd N}{\dd t} = -\lambda N :
\]
a população encolhe a uma taxa proporcional a ela mesma.
\end{proposition}

\begin{proof}
Cada um dos $N$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos} decai durante $\dd t$ com probabilidade $\lambda\,\dd t$:
decaimentos esperados $N\lambda\,\dd t$, e para $N$ grande a contagem real gruda na
esperada (\cref{rem:g12:radioactive-decay:crowds}); cada decaimento remove um \omterm{def:g11:fundamental-interactions:ladder}{núcleo}.
\end{proof}

\begin{theorem}[Lei do decaimento radioativo]\label{thm:g12:radioactive-decay:law}
Uma amostra que guarda $N_0$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos} em $t = 0$ guarda, no instante $t$,
\[
N(t) = N_0 \, e^{-\lambda t}.
\]
\end{theorem}

\begin{proof}
Confira por substituição: $\dd N/\dd t = N_0(-\lambda)\,e^{-\lambda t} = -\lambda N(t)$
e $N(0) = N_0 e^0 = N_0$: equação e partida, as duas certas. (Que nenhuma \emph{outra}
função serve é demonstrado, com integrais, no volume universitário.)
\end{proof}

\begin{definition}[Decaimento exponencial]\label{def:g12:radioactive-decay:exponential}
Uma grandeza que segue $N_0\,e^{-\lambda t}$ sofre \emph{decaimento
exponencial}\index{decaimento exponencial}: em tempos iguais, perdem-se \emph{frações}
iguais. É exatamente o \omterm{def:g12:rc-rl-circuits:capacitor}{capacitor} que se descarrega do \cref{ch:g12:rc-rl-circuits}, com
$\lambda$ fazendo o papel de $1/(RC)$: uma equação diferencial, aprendida uma vez, roda
circuitos e \omterm{def:g11:fundamental-interactions:ladder}{núcleos} igualmente.
\end{definition}

\begin{proposition}[Meia-vida]\label{prop:g12:radioactive-decay:halflife}
A \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida} (\cref{def:g11:nucleus-radioactivity:halflife}) de um \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo} é fixada pela
\omterm{def:g12:radioactive-decay:lambda}{constante de decaimento} dele: $T_{1/2} = \ln 2/\lambda$, e depois de $n$ meias-vidas
$N = N_0/2^{\,n}$, com $n$ inteiro ou não.
\end{proposition}

\begin{proof}
$N(T_{1/2}) = N_0/2$ exige $e^{-\lambda T_{1/2}} = \tfrac12$, ou seja,
$\lambda T_{1/2} = \ln 2$; e $N(n\,T_{1/2}) = N_0\,e^{-n\ln 2} = N_0/2^{\,n}$ para
qualquer $n$: a exponencial enfia a escada num fio e preenche o que há entre os degraus.
\end{proof}

\section{Atividade: contar os estalos}

\begin{proposition}[Atividade]\label{prop:g12:radioactive-decay:activity}
A \omterm{def:g11:nucleus-radioactivity:activity}{atividade} de uma amostra --- os decaimentos por segundo dela, em \omterm{def:g11:nucleus-radioactivity:activity}{becquerels}
(\cref{def:g11:nucleus-radioactivity:activity}) --- é
\[
\mathcal A = \lambda N, \qquad\text{logo}\qquad \mathcal A(t) = \mathcal A_0 \, e^{-\lambda t}:
\]
proporcional à população sobrevivente, e igualmente exponencial.
\end{proposition}

\begin{proof}
$N$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}, cada um com probabilidade $\lambda$ por segundo: $\lambda N$ decaimentos
por segundo; multiplicar o \cref{thm:g12:radioactive-decay:law} por $\lambda$ dá
$\mathcal A(t) = \mathcal A_0 e^{-\lambda t}$.
\end{proof}

\begin{example}[$N$ enorme, $\lambda$ minúsculo]\label{ex:g12:radioactive-decay:body}
Uma banana tiquetaqueia a cerca de \qty{15}{Bq}, um corpo humano a cerca de
\qty{8000}{Bq}, um \omterm{def:g10:orders-of-magnitude:unit}{metro} cúbico de granito a uns \qty{e6}{Bq}
(\cref{ch:g11:nucleus-radioactivity}). Por trás dos estalos modestos há populações
astronômicas: o potássio-40 do corpo contribui com
$\mathcal A \approx \qty{4.4e3}{Bq}$ com $\lambda = \qty{1.8e-17}{s^{-1}}$:
$N = \mathcal A/\lambda \approx \num{2.5e20}$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}, dos quais mal alguns milhares
disparam a cada segundo.
\end{example}

\begin{method}[Medir uma meia-vida]\label{met:g12:radioactive-decay:semilog}
\begin{enumerate}
  \item Conte decaimentos com um contador Geiger em intervalos iguais e sucessivos: isso
        amostra $\mathcal A$ em instantes sucessivos.
  \item Trace $\ln \mathcal A$ contra $t$: como
        $\ln \mathcal A = \ln \mathcal A_0 - \lambda t$, o \omterm{def:g12:radioactive-decay:exponential}{decaimento exponencial} aparece
        como uma \emph{reta} --- dados tortos significam outra lei, ou uma mistura de
        \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeos}.
  \item Leia $\lambda$ como menos a inclinação; $T_{1/2} = \ln 2/\lambda$.
\end{enumerate}
\end{method}

\begin{remark}[Becquerels não são dano]\label{rem:g12:radioactive-decay:dose}
O \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} conta decaimentos, não estragos. O dano é acompanhado pela \omterm{rem:g11:nucleus-radioactivity:dose}{dose} em \omterm{rem:g11:nucleus-radioactivity:dose}{sieverts}
(\cref{ch:g11:nucleus-radioactivity}), que pesa a \omterm{def:g10:energy-conservation:energy}{energia} depositada por \omterm{def:g10:orders-of-magnitude:unit}{quilograma} de
tecido pela destrutividade de cada radiação: uma cintilografia de \qty{5e8}{Bq} de
tecnécio é rotina, e bem menos \omterm{def:g11:nucleus-radioactivity:activity}{becquerels} de um emissor $\alpha$ inalado não são;
converter \unit{Bq} em \unit{Sv} é matéria universitária.
\end{remark}

\section{Datação: ler o tempo numa razão}

A lei do decaimento lida ao contrário é um relógio: meça a fração sobrevivente, e a
exponencial nomeia o tempo decorrido.

\begin{method}[Datação por carbono-14]\label{met:g12:radioactive-decay:carbon}
\begin{enumerate}
  \item \emph{Vivo:} o tecido vivo troca carbono com a atmosfera e carrega a proporção
        de carbono-14 dela --- um \omterm{def:g11:fundamental-interactions:ladder}{átomo} em $10^{12}$, o que dá
        $\mathcal A_0 = 13.6$ decaimentos por minuto por grama de carbono.
  \item \emph{A morte dispara o relógio:} a entrada cessa e a proporção decai com
        $T_{1/2} = \num{5730}$ anos.
  \item Meça a \omterm{def:g11:nucleus-radioactivity:activity}{atividade} $\mathcal A$ da amostra por grama de carbono;
        $\mathcal A = \mathcal A_0 e^{-\lambda t}$ se inverte em
        $t = \frac{1}{\lambda}\ln(\mathcal A_0/\mathcal A)
        = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}
\end{method}

\begin{example}[Uma fogueira numa caverna]\label{ex:g12:radioactive-decay:hearth}
Carvão de uma fogueira soterrada dá $5.0$ decaimentos por minuto por grama:
$\mathcal A_0/\mathcal A = 13.6/5.0 = 2.72$, logo
$t = \num{5730} \times \ln(2.72)/\ln 2 \approx \num{8.3e3}$ anos. O fogo se apagou há
oito mil anos --- a própria madeira é o arquivo.
\end{example}

\begin{example}[Relógios para rochas]\label{ex:g12:radioactive-decay:rocks}
O carbono-14 emudece além de uns \num{50000} anos; a geologia mantém relógios mais
lentos. \emph{Potássio--argônio}: o potássio-40 ($T_{1/2} = \num{1.25e9}$ anos) decai em
argônio, um gás que escapa da lava derretida mas fica preso assim que a rocha se
cristaliza --- a solidificação zera o relógio. \emph{Urânio--chumbo}: o urânio-238
encabeça uma cadeia de decaimentos que termina no chumbo-206 estável
(\cref{ch:g11:nucleus-radioactivity}), e a razão chumbo/urânio num cristal o data. Os
dois relógios concordam sobre os minerais mais antigos ($\approx \num{4.4e9}$ anos) e,
via meteoritos, sobre a idade do Sistema Solar.
\end{example}

\begin{remark}[Cadeias, e o gás do porão]\label{rem:g12:radioactive-decay:radon}
Entre o urânio e o chumbo a cadeia passa pelo rádio e depois pelo radônio-222, um
\emph{gás} radioativo ($T_{1/2} = 3.8$ dias). Nascido no granito e no solo, ele se
infiltra em porões e se acumula; as filhas emissoras de $\alpha$ dele decaem nos pulmões
--- para a maioria das pessoas, a maior \omterm{rem:g11:nucleus-radioactivity:dose}{dose} natural, e a mais barata de reduzir: abra a
janela do porão.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:1}
O iodo-131 tem $\lambda = \qty{1.0e-6}{s^{-1}}$. Calcule a \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida} dele em segundos e
depois em dias.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:2}
O radônio-222 tem $T_{1/2} = 3.8$ dias. Calcule $\lambda$ em \unit{s^{-1}} e depois a
\omterm{def:g11:nucleus-radioactivity:activity}{atividade} de uma amostra de \num{1.0e10} \omterm{def:g11:fundamental-interactions:ladder}{núcleos}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:3}
Verifique por substituição que $N(t) = N_0\,e^{-\lambda t}$ satisfaz
$\dd N/\dd t = -\lambda N$ e $N(0) = N_0$; depois calcule $N(T_{1/2})/N_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:4}
Uma fonte de césio-137 ($T_{1/2} = 30$ anos) tem \omterm{def:g11:nucleus-radioactivity:activity}{atividade} de \qty{4.0e5}{Bq} hoje.
Calcule a \omterm{def:g11:nucleus-radioactivity:activity}{atividade} dela daqui a $90$ anos e depois daqui a $300$ anos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:5}
Um gráfico semilogarítmico de $\ln \mathcal A$ contra $t$ para uma amostra de tecnécio é
uma reta que passa por $(0, 20.00)$ e $(\qty{10}{horas}, 18.85)$. Encontre $\lambda$ (em
\unit{h^{-1}} e em \unit{s^{-1}}) e $T_{1/2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:6}
Um grama de rádio-226 ($T_{1/2} = 1600$ anos; massa molar \qty{226}{g/mol};

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:7}
Dois estudantes lacram amostras de $20$ e de \num{2.0e20} \omterm{def:g11:fundamental-interactions:ladder}{núcleos} de um mesmo \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo}.
O que cada um espera depois de uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}? De quem é a previsão confiável? (Estime a

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:11}
Um porão é lacrado com uma leva de radônio-222 dentro ($T_{1/2} = 3.8$ dias). Depois de

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:12}
Mostre, a partir de $N = N_0 e^{-\lambda t}$, que $t = T_{1/2}\ln(N_0/N)/\ln 2$. Um osso

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:13}
Num cristal que não aprisionou argônio algum na solidificação, cada \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de

$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$; uma rocha mostra uma razão de
$3.0$: encontre a idade dela ($T_{1/2} = \num{1.25e9}$ anos).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:14}
Um contador Geiger dá, em $t = 0$, $60$, $120$, $180$, $240$ segundos, taxas de $400$,
$225$, $127$, $71$, $40$ contagens por segundo. Calcule $\ln \mathcal A$ em cada
instante, verifique o alinhamento e deduza $\lambda$ e $T_{1/2}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:15}
Para um único \omterm{def:g11:fundamental-interactions:ladder}{núcleo}, a probabilidade de sobreviver a $n$ meias-vidas é $1/2^{\,n}$.
Calcule a probabilidade de que (a) um dado \omterm{def:g11:fundamental-interactions:ladder}{núcleo} sobreviva a $10$ meias-vidas; (b)
todos os $8$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos} observados sobrevivam a uma \omterm{def:g11:nucleus-radioactivity:halflife}{meia-vida}. (c) De \num{1000} \omterm{def:g11:fundamental-interactions:ladder}{núcleos},
quantos se esperam depois de $3$ meias-vidas --- e a lei consegue dizer \emph{quais}?
Conclua sobre o que a lei prevê.
\end{exercise}

\section{Problema: a idade das coisas}

\begin{problem}[{Problema de fim de semana --- a idade das coisas: um laboratório de
museu calibra o relógio do carbono, data uma estátua e um papiro, aprende por que nenhum
relógio marca além do mostrador dele e empurra uma rocha vulcânica um bilhão de anos
para trás}]
\label{pb:g12:radioactive-decay:1}
Uma semana no laboratório de datação de um museu. Dados: o tecido vivo mostra
$\mathcal A_0 = 13.6$ decaimentos de carbono-14 por minuto por grama de carbono;
$T_{1/2} = \num{5730}$ anos (carbono-14) e \num{1.25e9} anos (potássio-40); $1$ ano
$= \qty{3.156e7}{s}$; $N_A = \qty{6.02e23}{mol^{-1}}$; massa molar do carbono
\qty{12}{g/mol}; os contadores resolvem cerca de $0.2$ decaimento por minuto por grama.

\medskip
\noindent\textbf{Parte I --- Segunda-feira: calibrar o relógio.}
\begin{enumerate}
  \item Por que a proporção de carbono-14 é constante no tecido vivo, e o que muda na
        morte?
  \item Calcule $\lambda$ do carbono-14 em \unit{s^{-1}}.
  \item Verifique por substituição que $N_0 e^{-\lambda t}$ resolve
        $\dd N/\dd t = -\lambda N$.
  \item A partir de $\mathcal A_0$, calcule o número de \omterm{def:g11:fundamental-interactions:ladder}{átomos} de carbono-14 por grama
        de carbono no tecido vivo.
  \item Quantos \omterm{def:g11:fundamental-interactions:ladder}{átomos} de carbono um grama guarda? Deduza a proporção de carbono-14 ---
        compare com ``um em $10^{12}$''.
  \item Mostre que uma \omterm{def:g11:nucleus-radioactivity:activity}{atividade} medida $\mathcal A$ data uma amostra como
        $t = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Terça-feira: a estátua e o papiro.}
\begin{enumerate}[resume]
  \item Uma estátua de madeira dá $9.1$ decaimentos por minuto por grama. Date a
        madeira.
  \item Que evento, exatamente, essa data marca --- o entalhe ou outra coisa? Que
        cautela decorre disso?
  \item Um papiro egípcio dá $10.4$ decaimentos por minuto por grama. Date-o. Um escriba
        de vinte e dois séculos atrás é plausível?
  \item A \omterm{def:g11:nucleus-radioactivity:activity}{atividade} é medida com $\pm 0.2$ decaimento por minuto. Estime a incerteza da
        idade do papiro, usando $\Delta t \approx \Delta\mathcal A/(\lambda \mathcal A)$.
  \item O pergaminho ``antigo'' de um negociante dá $13.4$ decaimentos por minuto por
        grama. Veredito?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Quinta-feira: os limites do relógio.}
\begin{enumerate}[resume]
  \item Depois de quantas meias-vidas a \omterm{def:g11:nucleus-radioactivity:activity}{atividade} de uma amostra cai abaixo dos $0.2$
        decaimento por minuto por grama dos contadores? Que idade é essa?
  \item Os melhores laboratórios chegam a umas dez meias-vidas. Qual é o horizonte
        prático da datação por carbono, em anos?
  \item Um osso de dinossauro tem cerca de \num{6.6e7} anos: quantas meias-vidas do
        carbono-14 são essas? Usando a questão 4, depois de quantas meias-vidas não
        resta nem \emph{um} \omterm{def:g11:fundamental-interactions:ladder}{átomo} do carbono-14 daquele grama? Conclua sobre ``datar
        dinossauros por carbono''.
  \item Que tipo de \omterm{def:g11:nucleus-radioactivity:notation}{nuclídeo} poderia datar a rocha do dinossauro, em vez disso?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Sexta-feira: a rocha vulcânica.}
\begin{enumerate}[resume]
  \item O argônio é um gás e o potássio, um morador fiel do sólido: explique por que a
        solidificação da lava zera o relógio potássio--argônio.
  \item Supondo que cada \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de potássio-40 que decaiu deixe um \omterm{def:g11:fundamental-interactions:ladder}{átomo} de argônio
        preso, mostre que $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.
  \item A rocha sob o fóssil do museu mostra $N_{\mathrm{Ar}}/N_{\mathrm K} = 0.60$.
        Date a rocha.
  \item O carbono-14 poderia ter datado essa rocha, ou o potássio--argônio, a estátua?
        Dê a janela útil de cada relógio, em meias-vidas.
  \item Relatório de sexta, em uma frase: as três idades encontradas nesta semana e a
        única lei que leu todas elas.
\end{enumerate}
\end{problem}
