\chapter{Hukum-Hukum Newton}\label{ch:g12:newtons-laws}

Selundupkanlah sebuah timbangan badan ke dalam lift. Ketika kabinnya menarik ke atas,
jarumnya membaca tiga \omterm{def:g10:orders-of-magnitude:unit}{kilogram} lebih berat; menjelang puncaknya jarumnya membaca lebih
ringan; di antara keduanya, kebenaran yang polos --- tidak ada yang berubah di dalamnya
kecuali geraknya. Bab ini menyatakan ketiga hukum di balik jarum itu lalu menempanya
menjadi cara kerja yang menjalankan seluruh mekanika: pilihlah sistemnya, gambarlah
\omterm{def:g10:inertia:force}{gayanya}, proyeksikanlah $\sum \vect F = m\vect a$.

\section{Momentum, dan hukum pertama yang dibuat tepat}

\begin{definition}[Momentum]\label{def:g12:newtons-laws:momentum}
\emph{Momentum}\index{momentum} sebuah benda bermassa $m$ (\unit{kg}) yang bergerak
dengan kecepatan $\vect v$ adalah vektor
\[
\vect p = m \vect v \qquad (\unit{kg.m/s}):
\]
kecepatannya, yang ditimbang oleh massanya; untuk sebuah sistem, jumlah vektor atas
bagian-bagiannya.
\end{definition}

\begin{definition}[Kerangka acuan inersial]\label{def:g12:newtons-laws:inertial}
\emph{Kerangka acuan inersial}\index{kerangka acuan inersial} adalah kerangka yang di
dalamnya setiap benda terpencil --- tanpa \omterm{def:g10:inertia:force}{gaya}, atau dengan \omterm{def:g10:inertia:force}{gaya} yang \omterm{def:g10:inertia:compensate}{saling meniadakan}
--- bergerak lurus dengan kecepatan tetap (\cref{ch:g10:inertia}). Tanah adalah salah
satunya dengan ketelitian yang sangat baik, begitu pula kabin mana pun yang bergerak
\omterm{def:g12:kinematics-2d:uniform}{lurus beraturan} di atasnya; kereta yang mengerem atau komidi putar yang berbelok
bukanlah kerangka inersial: di sana, barang bawaan yang lepas berpercepatan tanpa
didorong.
\end{definition}

\begin{theorem}[Hukum pertama Newton]\label{thm:g12:newtons-laws:first}\index{hukum pertama Newton}
Di dalam kerangka inersial, \omterm{def:g12:newtons-laws:momentum}{momentum} sebuah sistem terpencil tidak berubah: $\vect p$
tetap, yaitu $d\vect p/dt = \vect 0$.
\end{theorem}
\admitted

\begin{remark}[Kelembaman, yang dipertajam dua kali]\label{rem:g12:newtons-laws:sharpened}
Asas kelembaman dari \cref{ch:g10:inertia}, yang ditingkatkan dua kali: ia menyebutkan
\emph{di mana} ia berlaku --- ia menakrifkan kerangka inersial sebagai kerangka tempat
ia berlaku --- dan ia menjaga sebuah vektor, $m\vect v$, yang pembukuannya berbuah di
penghujung bab ini.
\end{remark}

\section{Hukum kedua Newton}

\begin{theorem}[Hukum kedua Newton]\label{thm:g12:newtons-laws:second}\index{hukum kedua Newton}
Di dalam kerangka inersial, \omterm{def:g11:forces-and-motion:netforce}{gaya total} pada sebuah benda sama dengan laju perubahan
\omterm{def:g12:newtons-laws:momentum}{momentumnya}; untuk massa yang tetap, karena $\vect p = m\vect v$,
\[
\sum \vect F = \frac{d\vect p}{dt} = m\,\frac{d\vect v}{dt} = m \vect a .
\]
\end{theorem}
\admitted

\begin{remark}[Sebuah hukum alam, dan satuannya]\label{rem:g12:newtons-laws:nature}
Tidak ada buktinya: inilah hukum alam, yang ditimbang terhadap percobaan selama tiga
abad. Tahun lalu (\cref{ch:g11:forces-and-motion}) memberikan bayangannya,
$\Delta v \propto F\,\Delta t/m$; turunannya (\cref{ch:g12:kinematics-2d}) membuatnya
tepat pada setiap saat --- rinciannya yang halus menunggu pada jilid Tahun ke-1. \omterm{def:g10:inertia:force}{Gaya}
memperoleh \omterm{def:g10:orders-of-magnitude:unit}{satuan}, $\unit{N} = \unit{kg.m/s^2}$; dan sebuah benda yang hanya
menanggung \omterm{def:g10:universal-gravitation:weight}{beratnya} menaati $m\vect a = m\vect g$: setiap massa jatuh dengan
$\vect a = \vect g$, dan
$g = \qty{9.81}{N/kg} = \qty{9.81}{m/s^2}$ --- kedua bacaan itu satu.
\end{remark}

\begin{example}[Gaya rata-rata, tanpa perlu rinciannya]\label{ex:g12:newtons-laws:average}
Sebuah mobil \qty{1300}{kg} bergerak dari diam ke \qty{100}{km/h} (\qty{27.8}{m/s})
dalam \qty{8.0}{s}: apa pun ceritanya saat demi saat, \omterm{def:g11:forces-and-motion:netforce}{gaya total} rata-ratanya adalah
$\Delta p/\Delta t = 1300 \times 27.8/8.0 \approx \qty{4.5e3}{N}$ --- \omterm{def:g12:newtons-laws:momentum}{momentum} mengubah
pertanyaan tentang \omterm{def:g10:inertia:force}{gaya} menjadi pembukuan sebelum dan sesudahnya.
\end{example}

\section{Hukum ketiga Newton}

\begin{theorem}[Hukum ketiga Newton]\label{thm:g12:newtons-laws:third}\index{hukum ketiga Newton}
Jika sebuah benda $A$ mengerjakan \omterm{def:g10:inertia:force}{gaya} $\vect F_{A \to B}$ pada benda $B$, maka $B$
mengerjakan pada $A$ \omterm{def:g10:inertia:force}{gaya} $\vect F_{B \to A} = -\vect F_{A \to B}$: segaris kerja,
sama besar, berlawanan arah --- pada setiap saat, apa pun geraknya, secara bersentuhan
maupun dari kejauhan.
\end{theorem}
\admitted

\begin{example}[Buku, meja, dan Bumi]\label{ex:g12:newtons-laws:book}
Sebuah buku terletak di atas meja: dua interaksi, dua pasangan. Gravitasi: Bumi
menarik bukunya ke bawah ($\vect P$), bukunya menarik Bumi ke atas. Sentuhan: mejanya
mendorong bukunya ke atas ($\vect N$), bukunya menekan mejanya ke bawah. Berjalan pun
sama saja: doronglah tanahnya ke belakang; reaksinya adalah satu-satunya \omterm{def:g10:inertia:force}{gaya} yang
melajukanmu ke depan.
\end{example}

\section{Cara kerjanya: pilih, gambar, proyeksikan}

\begin{method}[Menyelesaikan soal dinamika]\label{met:g12:newtons-laws:method}
\begin{enumerate}
  \item \emph{Sistem}: sebutkan benda (atau himpunan benda) yang ditelaah.
  \item \emph{Kerangka}: pilihlah kerangka inersial dan sumbu yang cocok dengan
        geraknya --- sumbu sepanjang lerengnya, atau \omterm{def:g12:kinematics-2d:frenet}{kerangka Frenet} untuk kurva
        (\cref{ch:g12:kinematics-2d}).
  \item \emph{Daftar}: sebutkan setiap \omterm{def:g10:inertia:force}{gayanya} --- \omterm{def:g10:universal-gravitation:weight}{beratnya}, lalu satu \omterm{def:g10:inertia:force}{gaya} tiap
        sentuhan, tidak ada yang lain --- lalu gambarlah diagram benda bebasnya: satu
        titik, satu anak panah tiap \omterm{def:g10:inertia:force}{gaya}.
  \item \emph{Proyeksikan}: tulislah $\sum \vect F = m\vect a$, lalu proyeksikan pada
        tiap sumbunya.
  \item \emph{Selesaikan lalu periksalah}: \omterm{def:g10:orders-of-magnitude:unit}{satuannya}, tandanya, kasus batasnya
        ($\mu \to 0$, $\alpha \to 0$, \dots).
\end{enumerate}
\end{method}

\begin{definition}[Gaya normal dan gesekan]\label{def:g12:newtons-laws:friction}
Sebuah permukaan bekerja pada benda melalui dua komponen: \emph{gaya
normal}\index{gaya normal} $\vect N$, yang tegak lurus permukaannya, dan \omterm{def:g10:inertia:inventory}{gesekan}
$\vect f$, yang menyinggungnya. Percobaan memberikan hukum \omterm{def:g10:inertia:inventory}{gesekannya}: selama bendanya
belum tergelincir, \omterm{def:g10:inertia:inventory}{gesekan} statisnya menyesuaikan diri sampai sebuah langit-langit,
$f_s \leq \mu_s N$; sekali tergelincir, \omterm{def:g10:inertia:inventory}{gesekan} kinetisnya adalah $f_k = \mu_k N$,
melawan gelincirnya. \emph{Koefisien \omterm{def:g10:inertia:inventory}{gesekan}}%
\index{koefisien gesekan} $\mu_s$ dan $\mu_k$ ($\mu_k \leq \mu_s$) hanya bergantung
pada kedua bahannya.
\end{definition}

\begin{example}[Bidang miring, dua kali]\label{ex:g12:newtons-laws:incline}
Sebuah balok meluncur menuruni lereng bersudut $\alpha$; sumbunya: $x$ menuruni
lerengnya, $y$ tegak lurus. Pada $y$: $N = mg\cos\alpha$. Pada $x$, tanpa \omterm{def:g10:inertia:inventory}{gesekan}:
$ma = mg\sin\alpha$, sehingga $a = g\sin\alpha$ --- tanpa massa; pada
$\alpha = 30^\circ$, $a = \qty{4.9}{m/s^2}$. Dengan \omterm{def:g10:inertia:inventory}{gesekan} kinetis:
$a = g(\sin\alpha - \mu_k\cos\alpha)$; untuk $\mu_k = 0.20$,
$a = 9.81\,(0.500 - 0.173) = \qty{3.2}{m/s^2}$. Dan baloknya tetap diam pada mulanya
hanya selama \omterm{def:g10:inertia:inventory}{gesekan} statis yang dibutuhkannya muat di bawah langit-langitnya:
$\tan\alpha \leq \mu_s$.
\end{example}

\begin{definition}[Berat semu]\label{def:g12:newtons-laws:apparent}
\emph{Berat semu}\index{berat semu} sebuah benda adalah \omterm{def:g12:newtons-laws:friction}{gaya normal} antara benda itu
dan penopangnya --- yang dibaca timbangan di bawahnya.
\end{definition}

\begin{example}[Lift]\label{ex:g12:newtons-laws:elevator}
Seorang penumpang \qty{70}{kg} berdiri di atas timbangan di dalam lift yang percepatan
tegaknya $a_z$ (ke atas positif): $N - mg = ma_z$, sehingga $N = m(g + a_z)$. Menarik
ke atas pada $a_z = \qty{1.5}{m/s^2}$: \qty{792}{N}, berat; mengerem menjelang
puncaknya: \qty{582}{N}, ringan; melaju tetap: $mg = \qty{687}{N}$ --- hukum
pertamanya; jatuh bebas, $a_z = -g$:
$N = 0$, tanpa bobot di atas timbangan.
\end{example}

\begin{example}[Dua balok dan sebuah katrol]\label{ex:g12:newtons-laws:pulley}
Sebuah balok $m_1 = \qty{4.0}{kg}$ di atas meja tanpa \omterm{def:g10:inertia:inventory}{gesekan} diikat dengan tali
ringan, melewati katrol ideal, ke balok yang menggantung $m_2 = \qty{1.0}{kg}$.
Talinya meneruskan tegangan $T$ yang sama di kedua ujungnya; kedua baloknya berbagi
satu besar percepatan $a$. Satu hukum kedua untuk masing-masingnya, sepanjang tiap
geraknya: $m_1 a = T$ dan $m_2 a = m_2 g - T$, sehingga
$a = m_2\,g/(m_1 + m_2) = \qty{1.96}{m/s^2}$ dan $T = m_1 a = \qty{7.8}{N}$ ---
lebih kecil daripada $m_2 g = \qty{9.8}{N}$, sebagaimana seharusnya agar $m_2$
berpercepatan ke bawah.
\end{example}

\begin{example}[Tikungan yang datar]\label{ex:g12:newtons-laws:curve}
Sebuah mobil bermassa $m$ menempuh tikungan datar berjari-jari $R$ pada laju tetap
$v$. Di dalam \omterm{def:g12:kinematics-2d:frenet}{kerangka Frenet} (\cref{ch:g12:kinematics-2d}) percepatannya murni
normal, $a_N = v^2/R$, membidik pusatnya. Secara tegak, $N = mg$; secara mendatar,
satu-satunya \omterm{def:g10:inertia:force}{gaya} sentripetal yang tersedia adalah \omterm{def:g10:inertia:inventory}{gesekan} statis jalan pada bannya:
$f = mv^2/R \leq \mu_s N = \mu_s mg$, sehingga $v \leq \sqrt{\mu_s\, g R}$, berapa pun
massanya. Untuk $R = \qty{90}{m}$, jalan kering ($\mu_s = 0.70$):
$v_{\max} \approx \qty{25}{m/s}$ (\qty{89}{km/h}); di atas es langit-langitnya runtuh
--- \cref{exo:g12:newtons-laws:13} memiringkan jalannya;
\cref{exo:g12:newtons-laws:10} mengayunkan proyeksi yang sama pada seutas kawat.
\end{example}

\section{Sistem terpencil: momentumnya kekal}

\begin{proposition}[Kekekalan momentum]\label{prop:g12:newtons-laws:conservation}\index{kekekalan momentum}
Untuk dua benda yang \omterm{def:g10:inertia:force}{gaya} luarnya lenyap atau \omterm{def:g10:inertia:compensate}{saling meniadakan}, \omterm{def:g12:newtons-laws:momentum}{momentum} totalnya
$\vect p_1 + \vect p_2$ tetap, apa pun \omterm{def:g10:inertia:force}{gaya} yang keduanya kerjakan satu sama lain.
\end{proposition}

\begin{proof}
Jumlahkanlah kedua hukum keduanya: $\frac{d}{dt}(\vect p_1 + \vect p_2) = \vect F_{2 \to 1} +
\vect F_{1 \to 2} + \vect F_{\text{luar}}$: pasangan dalamnya \omterm{def:g10:inertia:compensate}{saling meniadakan}
menurut hukum ketiga, dan $\vect F_{\text{luar}} = \vect 0$ menurut andaiannya.
\end{proof}

\begin{example}[Hentakan balik]\label{ex:g12:newtons-laws:recoil}
Seorang peluncur es \qty{60}{kg} yang diam di atas es licin melemparkan bola medisin
\qty{4.0}{kg} secara mendatar pada \qty{6.0}{m/s}. Sebelumnya: $\vect p = \vect 0$;
sesudahnya: $0 = 60\,V + 4.0 \times 6.0$, sehingga $V = \qty{-0.40}{m/s}$ --- ia
melayang ke belakang. Tidak ada yang mendorongnya dari luar: bolanya yang mendorong
(hukum ketiga), dan pembukuannya tetap nol.
\end{example}

\begin{remark}[Bagaimana roket mendorong ketiadaan]\label{rem:g12:newtons-laws:rocket}
Roket adalah mesin pelempar massa. Tiap sekon mesinnya melontarkan segumpal gas ke
belakang; gasnya membawa pergi \omterm{def:g12:newtons-laws:momentum}{momentum} ke belakang, sehingga roketnya memperoleh
sebanyak itu ke depan --- \omterm{def:g10:inertia:inventory}{gaya dorong} yang mantap, kira-kira massa yang dilontarkan
tiap sekon dikalikan laju lontarannya. Roketnya mendorong gas buangannya sendiri,
bukan udara: ia bekerja paling baik di ruang hampa.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:1}
Hitunglah \omterm{def:g12:newtons-laws:momentum}{momentum} (a) seorang pelari cepat \qty{70}{kg} pada \qty{10}{m/s}; (b) mobil
\qty{1300}{kg} pada \qty{50}{km/h}; (c) peluru \qty{8.0}{g} pada \qty{800}{m/s}.
Urutkanlah ketiganya. Mengapa yang paling cepat menempati urutan terakhir?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:2}
Seorang penumpang \qty{70}{kg} berdiri di atas timbangan di dalam lift. Apa bacaannya
ketika kabinnya (a) berpercepatan ke atas \qty{1.2}{m/s^2}; (b) menanjak tetap
\qty{2.0}{m/s}; (c) berpercepatan ke bawah \qty{1.2}{m/s^2}? Hukum mana yang
menyelesaikan (b)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:4}

bolanya.

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:5}

\section{Soal: Kereta gantung}

\begin{problem}[{Soal akhir pekan --- mengesahkan kereta gantung: tegangan yang
dituntut lerengnya saat diam, tambahan saat bertolak, kelegaan di atas menaranya,
harga sebuah pemberhentian darurat --- dan satu bilangan yang boleh dijanjikan papan
di dalam kabinnya}]
\label{pb:g12:newtons-laws:1}
Sebuah kereta gantung gunung --- kabin beserta keretanya, bermassa kosong
$M_0 = \qty{2.0e3}{kg}$ --- berjalan pada kabel jalur yang miring $\alpha = 30^\circ$,
ditarik oleh kabel penghela yang sejajar lerengnya dengan tegangan $\vect T$. \omterm{def:g10:inertia:inventory}{Gesekan}
gelindingnya diabaikan; lantai kabinnya tetap mendatar sepanjang waktu. Kabinnya
memuat $25$ penumpang bermassa rata-rata \qty{80}{kg}; kamulah insinyur yang
mengesahkannya. Kecuali disebutkan lain, kabinnya penuh: $M = \qty{4.0e3}{kg}$.

\medskip
\noindent\textbf{Bagian I --- Diam di lerengnya.}
\begin{enumerate}
  \item Pilihlah sistemnya; sebutkan mengapa kerangka gunungnya memadai sebagai
        kerangka inersial; daftarlah ketiga \omterm{def:g10:inertia:force}{gayanya}, beserta arahnya.
  \item Hukum mana yang mengatur penantiannya? Proyeksikanlah hukum itu sepanjang dan
        tegak lurus lerengnya untuk menyatakan $T$ dan \omterm{def:g12:newtons-laws:friction}{gaya normal} $N$ kabel jalurnya.
  \item Hitunglah $T$ dan $N$.
  \item Hitunglah kembali $T$ untuk kabin yang kosong. Mengapa $T$ sebanding dengan
        massa totalnya?
  \item Kabel penghelanya putus, remnya lepas: tunjukkan bahwa percepatan meluncurnya
        adalah $g \sin\alpha$, bermuatan maupun kosong, lalu hitunglah.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Bertolak.} Derek penggulungnya melajukan kabin yang
penuh dengan $a = \qty{0.60}{m/s^2}$ menaiki lerengnya sampai melaju tetap pada
$v = \qty{6.0}{m/s}$.
\begin{enumerate}[resume]
  \item Nyatakan lalu hitunglah tegangan $T'$ selama tahap ini.
  \item Berapa bagian tambahan tegangan saat bertolak di atas nilainya saat diam?
  \item Berapa lama tahap itu berlangsung, dan sepanjang jarak berapa?
  \item Seorang penumpang \qty{80}{kg} berdiri di lantai yang mendatar. Proyeksikanlah
        hukum keduanya secara tegak untuk menyatakan lalu menghitung \omterm{def:g12:newtons-laws:apparent}{berat semunya};
        bandingkanlah dengan $mg$.
  \item \omterm{def:g10:inertia:force}{Gaya} mana yang melajukannya secara mendatar, dan seberapa besar? Mengapa
        penumpangnya condong?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Di atas menaranya.} Pada \qty{6.0}{m/s} yang tetap,
kabel jalurnya melengkung di atas menara penopang sepanjang busur lingkaran tegak
berjari-jari $R = \qty{40}{m}$; di puncaknya kecepatannya mendatar.
\begin{enumerate}[resume]
  \item Di dalam \omterm{def:g12:kinematics-2d:frenet}{kerangka Frenet} pada puncaknya (\cref{ch:g12:kinematics-2d}),
        sebutkan kedua komponen percepatannya, beserta nilainya.
  \item Proyeksikanlah hukum keduanya sepanjang normalnya untuk menyatakan lalu
        menghitung \omterm{def:g12:newtons-laws:friction}{gaya normal} $N'$ kabelnya pada keretanya.
  \item Pertanyaan yang sama untuk \omterm{def:g12:newtons-laws:apparent}{berat semu} penumpangnya di puncaknya: berapa bagian
        ia menjadi lebih ringan?
  \item Pada laju berapa $N'$ akan lenyap? Bandingkanlah dengan laju lajunya lalu
        berilah ulasan.
  \item \omterm{def:g10:inertia:force}{Gaya} apa yang dikerjakan keretanya pada kabelnya di atas menara itu --- hukum
        mana yang mengatakannya, dengan besar dan arah yang mana?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pengereman, sebuah lompatan, dan papannya.}
\begin{enumerate}[resume]
  \item Pada bentangan mendatar saat tiba, sebuah pemberhentian darurat membawa kabin
        yang penuh dari \qty{6.0}{m/s} sampai diam dalam \qty{1.5}{s}: bacalah
        $\sum \vect F = d\vect p/dt$ sebagai rata-rata sepanjang pemberhentian itu
        untuk menghitung \omterm{def:g10:inertia:force}{gaya} pengeremannya.
  \item \omterm{def:g10:inertia:force}{Gaya} mendatar berapa yang harus menghentikan penumpang \qty{80}{kg}, dan
        dapatkah \omterm{def:g10:inertia:inventory}{gesekan} statis sepatu--lantai ($\mu_s = 0.40$) memasoknya?
        Simpulkanlah.
  \item Di peronnya, remnya lepas, kabin yang kosong menggantung bebas dan diam;
        penumpang yang terakhir (\qty{80}{kg}) melompat turun secara mendatar pada
        \qty{2.5}{m/s} \emph{terhadap kabinnya}. Hitunglah laju hentakan balik
        kabinnya dan laju penumpangnya terhadap tanah.
  \item Dalam satu kalimat: alat \omterm{def:g11:work-of-force:sign}{pendorong} mana yang hidup dari asas hentakan balik
        ini, dan apa yang menetapkan \omterm{def:g10:inertia:inventory}{gaya dorongnya}?
  \item \emph{Pengesahan.} Kabel penghelanya putus pada \qty{100}{kN}; peraturannya
        membatasi tegangan kerjanya pada seperlima nilai itu. Tunjukkan bahwa keadaan
        terburuk yang ditemui di atas adalah saat bertolak, perolehlah massa total
        terbesarnya, lalu cetaklah papannya: berapa penumpang \qty{80}{kg} yang boleh
        menumpang? Periksalah faktor keselamatannya.
\end{enumerate}
\end{problem}
