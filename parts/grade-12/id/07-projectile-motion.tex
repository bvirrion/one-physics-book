\chapter{Jatuh Bebas dan Gerak Proyektil}\label{ch:g12:projectile-motion}

Setiap semburan selang taman melukis kurva yang sama di udara. Begitu pula
bola basket yang melengkung menuju ringnya, seorang pelompat jauh, sebutir
peluru meriam: benda apa pun yang diserahkan kepada \omterm{def:g10:universal-gravitation:weight}{beratnya}. Bab ini
memperoleh kurva itu --- sebuah parabola --- dari \omterm{thm:g12:newtons-laws:second}{hukum kedua Newton} dalam
dua baris kalkulus, membaca puncaknya, \omterm{def:g12:projectile-motion:range}{waktu terbangnya} dan \omterm{def:g12:projectile-motion:range}{jangkauannya}
dari persamaannya, lalu membiarkan udaranya kembali masuk untuk melihat apa
yang dirusaknya.

\section{Jatuh bebas}

\begin{definition}[Jatuh bebas]\label{def:g12:projectile-motion:freefall}
Sebuah benda berada dalam \emph{jatuh bebas}\index{jatuh bebas} apabila
satu-satunya \omterm{def:g10:inertia:force}{gaya} yang bekerja padanya adalah \omterm{def:g10:universal-gravitation:weight}{beratnya} $\vect P = m\vect g$:
tanpa penopang, tanpa tali, hambatan udaranya terabaikan. Batu yang
dijatuhkan atau bola yang sudah lepas dari tangan berada, dengan ketelitian
yang baik, dalam jatuh bebas --- ``jatuh'' mencakup naiknya juga.
\end{definition}

\begin{theorem}[Hukum jatuh bebas Galileo]\label{thm:g12:projectile-motion:galileo}
\index{hukum jatuh bebas Galileo}
Dalam \omterm{def:g12:projectile-motion:freefall}{jatuh bebas}, semua benda berbagi percepatan yang sama, berapa pun massanya:
\[
\vect a = \vect g, \qquad g = \qty{9.81}{m/s^2},
\]
yang terarah lurus ke bawah: kuat \omterm{def:g11:electric-gravitational-fields:field}{medan} \qty{9.81}{N/kg} pada
\cref{ch:g10:universal-gravitation} dalam busananya yang lain, $\unit{N/kg} = \unit{m/s^2}$.
\end{theorem}

\begin{proof}
\omterm{thm:g12:newtons-laws:second}{Hukum kedua Newton} (\cref{ch:g12:newtons-laws}) dengan \omterm{def:g10:universal-gravitation:weight}{beratnya} sebagai
satu-satunya \omterm{def:g10:inertia:force}{gaya}: $m\vect a = m\vect g$, dan massanya \omterm{def:g10:inertia:compensate}{saling meniadakan}.
Palu dan bulu berbeda hanya karena udaranya.
\end{proof}

\begin{proposition}[Jatuh tegak dari keadaan diam]\label{prop:g12:projectile-motion:vertical}
Dijatuhkan dari keadaan diam, dengan kedalaman jatuh $h$ dihitung ke bawah
dari titik lepasnya, sebuah benda dalam \omterm{def:g12:projectile-motion:freefall}{jatuh bebas} menaati
\[
a = g, \qquad v = g t, \qquad h = \tfrac12\, g t^2 .
\]
\end{proposition}

\begin{proof}
$v(t)$ adalah antiturunan dari tetapan $a = g$ yang lenyap pada $t = 0$
(\cref{ch:g12:kinematics-2d}); $h(t)$ antiturunan dari $v = gt$ yang lenyap pada $t = 0$.
\end{proof}

\begin{example}[Orde besaran]\label{ex:g12:projectile-motion:orders}
Setelah $1$, $2$, $3$ sekon \omterm{def:g12:projectile-motion:freefall}{jatuh bebas}: $v = 9.8$, $19.6$, \qty{29.4}{m/s}
($35$, $71$, \qty{106}{km/h}) dan $h = 4.9$, $19.6$, \qty{44.1}{m}. Dari
atap setinggi \qty{30}{m}: $t = \sqrt{2h/g} \approx \qty{2.5}{s}$, tiba pada
\qty{24}{m/s} $\approx \qty{87}{km/h}$. Waktunya tumbuh seperti $\sqrt h$ ---
dan lewat beberapa sekon udaranya menolak tetap terabaikan (bagian terakhir).
\end{example}

\begin{example}[Lemparan tegak]\label{ex:g12:projectile-motion:throw}
Dilempar lurus ke atas pada $v_0$, sumbu $Oy$ ke atas dari tangannya: kedua
antiturunan yang sama memberikan $v = v_0 - gt$ dan $y = v_0 t - \tfrac12 gt^2$.
Puncaknya, tempat $v = 0$, tercapai pada $t = v_0/g$, dengan tinggi
$h = v_0^2/2g$ --- jawaban \omterm{def:g10:energy-conservation:energy}{energi} $v_0 = \sqrt{2gh}$ pada
\cref{ch:g11:mechanical-energy} yang dijalankan mundur; dilempar \emph{ke
bawah}, $v_0$ berganti tanda. Untuk \qty{15}{m/s}: puncaknya
\qty{11.5}{m} pada \qty{1.53}{s}, kembali pada \qty{3.06}{s}, pada \qty{15}{m/s}.
\end{example}

\section{Proyektil: satu hukum, dua antiturunan}

\begin{definition}[Proyektil]\label{def:g12:projectile-motion:projectile}
\emph{Proyektil}\index{proyektil} adalah benda yang diluncurkan dengan
kecepatan $\vect v_0$ lalu ditinggalkan dalam \omterm{def:g12:projectile-motion:freefall}{jatuh bebas}. Sumbunya di titik
luncurnya: $Ox$ mendatar di dalam bidang tegak $\vect v_0$, $Oy$ tegak ke
atas; \emph{sudut lontar}\index{sudut lontar} $\alpha$ milik $\vect v_0$ di
atas garis mendatar menetapkan koordinatnya $(v_0\cos\alpha,\; v_0\sin\alpha)$.
\end{definition}

\begin{theorem}[Persamaan geraknya]\label{thm:g12:projectile-motion:equations}
Sebuah \omterm{def:g12:projectile-motion:projectile}{proyektil} yang diluncurkan dari titik asalnya dengan laju $v_0$ dan
sudut $\alpha$ memiliki $\vect a = (0,\,-g)$ pada setiap saat, sehingga
\[
\vect v(t) = \bigl(v_0\cos\alpha,\; v_0\sin\alpha - g t\bigr),
\qquad
\begin{cases}
x(t) = v_0\cos\alpha\; t,\\[2pt]
y(t) = v_0\sin\alpha\; t - \tfrac12\, g t^2 .
\end{cases}
\]
\end{theorem}

\begin{proof}
\omterm{thm:g12:newtons-laws:second}{Hukum kedua Newton} memberikan $\vect a = \vect g = (0,-g)$. Antiturunannya
sumbu demi sumbu: $\vect v(t)$ dari $\vect a$ dengan nilai awal
$(v_0\cos\alpha,\, v_0\sin\alpha)$, lalu kedudukannya dari $\vect v$, dari $(0,0)$.
\end{proof}

\begin{proposition}[Kebebasan kedua geraknya]\label{prop:g12:projectile-motion:independence}
Gerak mendatarnya beraturan --- $v_x$ tidak pernah berubah --- dan gerak
tegaknya persis lemparan tegak pada bagian sebelumnya: tidak ada persamaan
yang menyebutkan koordinat milik yang lain. Khususnya, bola yang ditembakkan
mendatar dan bola yang dijatuhkan pada saat yang sama dari ketinggian yang
sama tetap sejajar satu sama lain --- keduanya mendarat bersamaan.
\end{proposition}

\begin{proof}
Pada \cref{thm:g12:projectile-motion:equations}, $x(t)$ hanya melibatkan
$v_0\cos\alpha$, $y(t)$ hanya $v_0\sin\alpha$ dan $g$; kedua bolanya memiliki
$v_y(0) = 0$, sehingga keduanya menaati $h = \tfrac12 gt^2$.
\end{proof}

\begin{remark}[Monyet yang jatuh]\label{rem:g12:projectile-motion:monkey}
Seorang penjaga kebun binatang membidikkan panah bius \emph{tepat ke arah}
seekor monyet, yang melepaskan dahannya saat tembakannya. Langkah yang
keliru: panah dan monyetnya masing-masing turun $\tfrac12 gt^2$ di bawah
kedudukan tanpa-gravitasinya --- garis bidiknya, dahannya --- sehingga
keduanya bertemu persis, berapa pun laju panahnya
(\cref{exo:g12:projectile-motion:15}).
\end{remark}

\section{Parabolanya, dibaca selengkapnya}

\begin{theorem}[Lintasannya]\label{thm:g12:projectile-motion:trajectory}
Dengan melenyapkan $t$ antara $x(t)$ dan $y(t)$: \omterm{def:g10:relative-motion:trajectory}{lintasan} sebuah \omterm{def:g12:projectile-motion:projectile}{proyektil}
($0 \leq \alpha < \ang{90}$) adalah parabola yang terbuka ke bawah
\[
y = x\tan\alpha \;-\; \frac{g}{2 v_0^2 \cos^2\alpha}\, x^2 .
\]
\end{theorem}

\begin{proof}
$t = x/(v_0\cos\alpha)$, yang disulihkan ke dalam $y(t)$ pada
\cref{thm:g12:projectile-motion:equations}.
\end{proof}

\begin{definition}[Jangkauan]\label{def:g12:projectile-motion:range}
Untuk \omterm{def:g12:projectile-motion:projectile}{proyektil} di atas tanah yang rata, \emph{jangkauan}\index{jangkauan} $R$
adalah jarak mendatar yang ditempuhnya ketika ia kembali ke ketinggian
luncurnya; \emph{waktu terbang}\index{waktu terbang} $t_f$ adalah lama perjalanan itu.
\end{definition}

\begin{proposition}[Puncak, waktu terbang, jangkauan]\label{prop:g12:projectile-motion:range}
Di atas tanah yang rata, dengan $0 < \alpha < \ang{90}$:
\[
t_f = \frac{2 v_0 \sin\alpha}{g}, \qquad
h_{\text{puncak}} = \frac{(v_0\sin\alpha)^2}{2g}
\ \ \text{tercapai pada } t_f/2, \qquad
R = \frac{v_0^2 \sin(2\alpha)}{g}.
\]
\omterm{def:g12:projectile-motion:range}{Jangkauannya} terbesar pada $\alpha = \ang{45}$, tempat $R = v_0^2/g$;
sudut yang berpelurus $\alpha$ dan $\ang{90} - \alpha$ memberikan \omterm{def:g12:projectile-motion:range}{jangkauan} yang sama.
\end{proposition}

\begin{proof}
$y = 0$ memberikan $t\,(v_0\sin\alpha - \tfrac12 gt) = 0$: saat luncurnya, dan $t_f$.
Puncaknya, $v_y = 0$, terletak pada $t = v_0\sin\alpha/g = t_f/2$; sulihkanlah ke dalam
$y(t)$. Selanjutnya $R = x(t_f) = 2v_0^2\sin\alpha\cos\alpha/g =
v_0^2\sin(2\alpha)/g$. Sinusnya memuncak pada $2\alpha = \ang{90}$, dan
$\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180} - 2\alpha) =
\sin(2\alpha)$.
\end{proof}

\begin{example}[Tendangan gawang]\label{ex:g12:projectile-motion:kick}
Seorang penjaga gawang menendang pada $v_0 = \qty{25}{m/s}$, $\alpha = \ang{45}$:
$R = 25^2/9.81 \approx \qty{64}{m}$; tendangan yang sesungguhnya mendarat lebih
dekat, sekitar \qty{55}{m} --- potongan yang diambil udaranya.
\end{example}

\begin{method}[Pembukuan proyektil]\label{met:g12:projectile-motion:method}
\begin{enumerate}
  \item Sumbunya di titik luncurnya, $Ox$ mendatar, $Oy$ ke atas; koordinat
        $\vect v_0$.
  \item Periksalah jatuh bebasnya (hanya \omterm{def:g10:universal-gravitation:weight}{beratnya}): maka $\vect a = (0,-g)$;
        antiturunannya dua kali, syarat awalnya tiap kali.
  \item Pertanyaan tentang sebuah \emph{tempat}: lenyapkanlah $t$ (\omterm{def:g10:relative-motion:trajectory}{lintasannya}).
        Tentang sebuah \emph{saat}: pertahankanlah $t$.
  \item Kamusnya: puncaknya $\Leftrightarrow v_y = 0$; ketinggian luncurnya
        $\Leftrightarrow y = 0$; dinding di $d \Leftrightarrow x = d$; lantai
        sedalam $\ell$ di bawahnya $\Leftrightarrow y = -\ell$.
  \item Kewarasannya: penerbangannya setangkup terhadap puncaknya;
        $R \leq v_0^2/g$; udaranya selamanya hanya memendekkan.
\end{enumerate}
\end{method}

\section{Apa yang diubah udaranya}

\begin{definition}[Laju terminal]\label{def:g12:projectile-motion:terminal}
Hambatan udara melawan kecepatannya dan tumbuh bersama lajunya. Karena itu
benda yang jatuh berpercepatan hanya sampai hambatannya mengimbangi \omterm{def:g10:universal-gravitation:weight}{beratnya};
sejak saat itu $\vect a = \vect 0$: benda itu jatuh pada \emph{laju
terminal}\index{laju terminal} yang tetap.
\end{definition}

\begin{example}[Orde besaran terminalnya]\label{ex:g12:projectile-motion:airorders}
Penerjun payung yang terbentang: sekitar \qty{55}{m/s} (\qty{200}{km/h}).
Setetes hujan: sekitar \qty{9}{m/s} --- di ruang hampa, jatuh \qty{2000}{m}
akan mengantarkannya pada $\sqrt{2gh} \approx \qty{2.0e2}{m/s}$. Sehelai
bulu: sekitar \qty{1}{m/s}, yang tercapai dalam beberapa sentimeter ---
itulah rahasia kelambatannya: ia menghabiskan seluruh jatuhnya \emph{pada}
\omterm{def:g12:projectile-motion:terminal}{laju terminal}; palunya tidak pernah mendekati \omterm{def:g12:projectile-motion:terminal}{laju terminalnya} sendiri.
\end{example}

\begin{remark}[Balistik yang sesungguhnya]\label{rem:g12:projectile-motion:drag}
Hambatan udara mendorong melawan kecepatannya sepanjang penerbangannya:
\omterm{def:g12:projectile-motion:range}{jangkauannya} memendek, puncaknya merendah, dan ketangkupannya patah ---
turunnya lebih curam daripada naiknya, tembakan yang jauh terlihat kurang
daripada parabolanya. Perlakuan yang jujur dan berbilang adalah persamaan
diferensial yang diselesaikan pada jilid Tahun ke-1; tahun ini, parabolanya
adalah model yang tepat bagi \omterm{def:g12:projectile-motion:projectile}{proyektil} yang padat dan cukup cepat --- bola
basket pada \qty{10}{m/s}, bukan bola golf pada \qty{70}{m/s}.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:1}
Sebuah batu yang dijatuhkan dari jembatan mengenai airnya \qty{2.0}{s}
kemudian: berapa tinggi jembatannya, dan berapa laju tumbukannya dalam
\unit{m/s} dan \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:2}
Di Bulan, seorang antariksawan menjatuhkan palu dan bulu bersama-sama:
keduanya mendarat bersamaan. Mengapa --- dan mengapa bulunya kalah telak di Bumi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:3}
Dengan mengabaikan udaranya, hitunglah waktu jatuh dan laju tibanya dari
\qty{1.0}{m}, \qty{10}{m}, \qty{100}{m}. Bagaimana waktunya tumbuh bersama ketinggiannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:4}
Sebuah bola dilempar lurus ke atas pada \qty{15}{m/s}: berapa tinggi dan
waktu puncaknya, waktu seluruhnya sampai kembali ke tangan, dan laju kembalinya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:5}
Sebutir kelereng menggelinding lepas dari meja setinggi \qty{0.80}{m} pada
\qty{2.5}{m/s}. Carilah waktu jatuhnya dan jarak mendaratnya dari tepi meja.
Apakah waktu jatuhnya bergantung pada laju kelerengnya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:6}
Dari gambar lima sudut pada bab ini: pasangan \omterm{def:g12:projectile-motion:projectile}{sudut lontar} yang mana yang
berbagi \omterm{def:g12:projectile-motion:range}{jangkauannya}, dan sudut mana yang menang? Untuk \omterm{def:g12:projectile-motion:range}{jangkauan} di bawah
yang terbesar, mana di antara kedua sudut yang mungkin itu yang terbang
lebih lama? Berikan alasannya dengan $t_f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:7}
Dari balkon setinggi \qty{20}{m} sebuah bola dilempar lurus \emph{ke bawah}
pada \qty{5.0}{m/s}. Carilah waktunya sampai ke tanah dan laju tumbukannya;
periksalah yang terakhir itu dengan neraca \omterm{def:g10:energy-conservation:energy}{energi} (\cref{ch:g11:mechanical-energy}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:8}
Sebuah \omterm{def:g12:projectile-motion:projectile}{proyektil} lepas dari tanah yang rata pada \qty{40}{m/s},
$\alpha = \ang{30}$: berapa \omterm{def:g12:projectile-motion:range}{waktu terbang}, tinggi puncak, dan \omterm{def:g12:projectile-motion:range}{jangkauannya}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:9}
Modelkanlah seorang pelompat jauh sebagai \omterm{def:g12:projectile-motion:projectile}{proyektil}: lepas landas pada
\qty{10.5}{m/s}, $\alpha = \ang{20}$. Hitunglah \omterm{def:g12:projectile-motion:range}{jangkauannya}; juaranya
melompat hampir \qty{9}{m} --- sebutkan dua bahan yang ditinggalkan model titiknya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:10}
Pada \qty{1.2}{m} di atas tanah yang rata, satu peluru dijatuhkan dan satu

\omterm{def:g12:projectile-motion:range}{jangkauannya}, lalu carilah kedua sudut yang mendaratkan \omterm{def:g12:projectile-motion:projectile}{proyektil}
\qty{25}{m/s} sejauh \qty{50}{m}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:12}
Air lepas dari selang taman pada \qty{8.0}{m/s}: berapa capaian mendatar

\section{Soal: Tembakan bola basket di detik terakhir}

\begin{problem}[{Soal akhir pekan --- tembakan bola basket di detik terakhir:
sebuah lemparan bebas yang dibedah anak panah demi anak panah, ujung jari
yang terlewati beberapa sentimeter, lemparan sepanjang lapangan yang
ditimbang terhadap kemampuan sebuah lengan, sebuah umpan lambung melewati
papan pantul --- dan sebuah putusan tentang tembakan yang paling teatrikal
dalam olahraga}]
\label{pb:g12:projectile-motion:1}
Ring bola basket adalah cincin mendatar \qty{3.05}{m} di atas lantainya;
Nora melepaskan setiap tembakannya pada \qty{2.00}{m}, sehingga dengan sumbu
di titik lepasnya ($Ox$ mendatar ke arah ringnya, $Oy$ ke atas) pusat
cincinnya terletak \qty{1.05}{m} di atas titik asalnya. Udaranya diabaikan
kecuali bila disebutkan.

\medskip\noindent\textbf{Bagian I --- Lemparan bebasnya.} Nora menembak dari
$d = \qty{4.20}{m}$ (mendatar) dengan \omterm{def:g12:projectile-motion:projectile}{sudut lontar} $\alpha = \ang{50}$.
\begin{enumerate}
  \item \omterm{def:g10:inertia:force}{Gaya} apa yang bekerja setelah bolanya lepas? Sebutkan $\vect a$,
        beserta hukum yang dipakainya.
  \item Turunkanlah koordinat $\vect v(t)$.
  \item Turunkanlah $x(t)$ dan $y(t)$.
  \item Lenyapkanlah $t$ untuk memperoleh \omterm{def:g10:relative-motion:trajectory}{lintasannya} $y(x)$.
  \item Bolanya harus melewati pusat cincinnya, $y(d) = \qty{1.05}{m}$:
        selesaikanlah untuk $v_0$, dalam \unit{m/s} dan \unit{km/h}.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Melewati ujung jari, masuk ke cincinnya.}
\begin{enumerate}[resume]
  \item Berapa lama bolanya mencapai ringnya?
  \item Carilah waktu dan tinggi puncaknya; periksalah bahwa bolanya sudah
        menurun di ringnya.
  \item Seorang pemain bertahan \qty{0.90}{m} di depan Nora menjulurkan ujung
        jarinya sampai \qty{2.60}{m}. Seberapa jauh bolanya melewatinya?
  \item Hitunglah $v_x$ dan $v_y$ di ringnya, lajunya di sana, dan sudut
        kecepatannya di bawah garis mendatar.
  \item Mengapa bolanya harus tiba dalam keadaan \emph{menurun} --- dan mengapa curam?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Lemparan sepanjang lapangan.} Bel
hampir berbunyi, Nora berada \qty{26.0}{m} dari ring seberangnya.
\begin{enumerate}[resume]
  \item Kenaikan \qty{1.05}{m} itu kecil terhadap \qty{26.0}{m}: berikan
        alasan bagi model tanah yang rata, lalu turunkanlah \omterm{def:g12:projectile-motion:range}{waktu terbangnya}
        dari $y(t_f) = 0$.
  \item Perolehlah $R = v_0^2\sin(2\alpha)/g$ dan sudut \omterm{def:g12:projectile-motion:range}{jangkauan} terbesarnya.
  \item Hitunglah $v_0$ yang dibutuhkan untuk menempuh \qty{26.0}{m} pada
        $\alpha = \ang{45}$.
  \item Pemain yang kuat mampu melontarkan bola basket pada sekitar
        \qty{17.5}{m/s}. Berapa \omterm{def:g12:projectile-motion:range}{jangkauan} terbesarnya --- apakah tembakan itu
        terjangkau oleh manusia?
  \item Pada \qty{17.5}{m/s}, carilah \emph{kedua} sudut yang mendaratkan
        bolanya sejauh \qty{26.0}{m}, beserta \omterm{def:g12:projectile-motion:range}{waktu terbangnya}: dengan
        \qty{2.0}{s} tersisa di jamnya, yang mana yang mendahului belnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Umpan lambung melewati papan pantulnya.}
Dari belakang garis dasarnya Nora melambungkan bolanya: lepas \qty{6.0}{m}
dari pusat cincinnya, $\alpha = \ang{75}$; bidang papannya memotong
tembakannya \qty{0.45}{m} sebelum pusat cincinnya, dengan tepi atas
\qty{3.95}{m} di atas lantainya.
\begin{enumerate}[resume]
  \item Carilah $v_0$ yang menjatuhkan bolanya melewati pusat cincinnya.
  \item Hitunglah ketinggian bolanya pada bidang papannya
        ($x = \qty{5.55}{m}$): apakah bolanya melewati tepi atasnya, dan
        seberapa jauh?
  \item Carilah tinggi puncak dan \omterm{def:g12:projectile-motion:range}{waktu terbangnya} --- mengapa penontonnya
        menyukainya?
  \item Secara kualitatif, apa yang diubah hambatan udara pada ketiga
        tembakan ini, dan yang mana yang paling terancam olehnya?
  \item \emph{Putusan tembakan di detik terakhir}, dua kalimat: bandingkanlah
        laju yang dituntut lemparan sepanjang lapangan itu dengan yang
        dipasok sebuah lengan, sebutkan selisihnya, lalu putuskanlah ---
        mungkin atau mitos?
\end{enumerate}
\end{problem}
