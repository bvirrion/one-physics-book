\chapter{Bunyi dan Akustika}\label{ch:g12:sound-acoustics}

Petiklah senar rendah sebuah gitar: ruangan terisi nada A yang rendah,
namun tidak ada yang berjalan ke telingamu selain udara yang terperas dan
teregang. Bab ini mengikuti pola itu --- lajunya, bagaimana \omterm{def:g10:signals-and-waves:frequency}{frekuensi}
menjadi \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nada} dan \omterm{def:g11:work-of-force:power}{daya} menjadi \omterm{def:g12:sound-acoustics:pitch-loudness}{kenyaringan}, serta mengapa satu nada
terdengar berbeda pada setiap alat musik.

\section{Bunyi adalah gelombang tekanan}

\begin{definition}[Gelombang bunyi]\label{def:g12:sound-acoustics:sound-wave}
\emph{Gelombang bunyi}\index{gelombang bunyi} adalah \omterm{def:g10:signals-and-waves:wave}{gelombang} \omterm{def:g10:pressure:pressure}{tekanan}
yang mekanis (\cref{ch:g12:mechanical-waves}): permukaan yang bergetar
menciptakan \emph{rapatan}\index{rapatan} (\omterm{def:g10:pressure:pressure}{tekanan} yang sedikit berlebih)
dan \emph{renggangan}\index{renggangan} (\omterm{def:g10:pressure:pressure}{tekanan} yang kurang) yang
merambat. Petak udaranya berayun \emph{searah} arah rambatnya (\omterm{def:g10:signals-and-waves:sound}{bunyi}
bersifat \emph{longitudinal}\index{gelombang longitudinal}), dan sebuah
medium diperlukan: bel di dalam bejana hampa menjadi bisu
(\cref{ch:g10:signals-and-waves}).
\end{definition}

\begin{proposition}[Laju bunyi]\label{prop:g12:sound-acoustics:speed}
\index{laju bunyi}%
\omterm{def:g10:signals-and-waves:sound}{Bunyi} merambat sekitar \qty{340}{m/s} di udara pada
\qty{20}{\degreeCelsius}, \qty{1500}{m/s} di air dan \qty{5000}{m/s} di
baja: makin kaku mediumnya, makin cepat. \admitted
\end{proposition}

\begin{remark}[Udara yang hangat lebih cepat]\label{rem:g12:sound-acoustics:temperature}
Nilai-nilai itu terukur; meramalkannya dikerjakan pada jilid Tahun ke-1.
\omterm{def:g10:signals-and-waves:sound}{Bunyi} bertambah cepat di udara yang lebih hangat (\qty{331}{m/s} pada
\qty{0}{\degreeCelsius}) --- dan itulah sebabnya alat musik tiup
melenceng dari nadanya ketika aulanya menghangat.
\end{remark}

\begin{example}[Ukuran bunyi]\label{ex:g12:sound-acoustics:wavelengths}
Hubungan $\lambda = v/f$ menakar \omterm{def:g10:signals-and-waves:sound}{bunyi} apa pun: di udara, gemuruh
\qty{20}{Hz} terbentang $340/20 = \qty{17}{m}$ --- sebesar sebuah
bangunan --- sedangkan desis \qty{20}{kHz} muat di dalam \qty{1.7}{cm};
yang berukuran \omterm{def:g10:orders-of-magnitude:unit}{meter} mudah membelok mengitari pintu dan sudut ruangan.
\end{example}

\section{Tinggi nada, kenyaringan, dan telinga}

\begin{definition}[Tinggi nada dan kenyaringan]\label{def:g12:sound-acoustics:pitch-loudness}
\emph{Tinggi nada}\index{tinggi nada} adalah \omterm{def:g11:color-light-sources:perception}{tanggapan indra} terhadap
\omterm{def:g10:signals-and-waves:frequency}{frekuensi}: makin besar $f$, makin tinggi nadanya; nada A adalah
\qty{440}{Hz}, dan menggandakan $f$ menaikkan nadanya satu oktaf.
\emph{Kenyaringan}\index{kenyaringan} adalah \omterm{def:g11:color-light-sources:perception}{tanggapan indra} terhadap
\omterm{def:g10:signals-and-waves:amplitude}{amplitudo}: makin kuat ayunan \omterm{def:g10:pressure:pressure}{tekanannya}, makin nyaring; \omterm{def:g10:signals-and-waves:oscilloscope}{penguatan}
mengubah kenyaringannya, bukan tinggi nadanya.
\end{definition}

\begin{remark}[Jendela telinga]\label{rem:g12:sound-acoustics:audible}
Pendengaran terbentang dari sekitar \qty{20}{Hz} sampai \qty{20}{kHz}
(\cref{ch:g10:signals-and-waves}); di bawahnya terletak \omterm{def:g10:signals-and-waves:ultrasound}{infrasonik}, di
atasnya \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik} --- kelelawar pada \qty{50}{kHz}, penduga kedokteran
pada beberapa megahertz. Sebuah piano hanya mencuplik bagian tengahnya:
\qtyrange{27.5}{4186}{Hz}.
\end{remark}

\section{Intensitas bunyi dan skala desibel}

\begin{definition}[Intensitas bunyi]\label{def:g12:sound-acoustics:intensity}
\emph{Intensitas bunyi}\index{intensitas bunyi} $I$ adalah \omterm{def:g11:work-of-force:power}{daya} akustik
yang melewati satu \omterm{def:g10:orders-of-magnitude:unit}{meter} persegi yang menghadap \omterm{def:g10:signals-and-waves:wave}{gelombangnya}:
$I = P/S$, dalam \unit{W/m^2}. \omterm{def:g10:signals-and-waves:sound}{Bunyi} terpelan yang masih terdengar
sekitar $I_0 = \qty{1.0e-12}{W/m^2}$; rasa sakitnya bermula di dekat
\qty{1}{W/m^2}.
\end{definition}

\begin{proposition}[Pelemahan berkebalikan kuadrat]\label{prop:g12:sound-acoustics:inverse-square}
\index{hukum kebalikan kuadrat}%
Sebuah sumber kecil yang memancarkan \omterm{def:g11:work-of-force:power}{daya} $P$ sama rata ke segala arah
memberikan, pada jarak $r$, $I = P/(4\pi r^2) \propto 1/r^2$:
menggandakan jaraknya membagi intensitasnya dengan empat.
\end{proposition}

\begin{proof}
Seluruh \omterm{def:g11:work-of-force:power}{daya} $P$ melewati bola seluas $4\pi r^2$; bagilah.
\end{proof}

\begin{definition}[Taraf intensitas bunyi]\label{def:g12:sound-acoustics:level}
Telinga terbentang dua belas pangkat sepuluh, sehingga intensitasnya
disebutkan secara logaritma. \emph{Taraf intensitas}\index{taraf
intensitas} bagi $I$ adalah
\[ L = 10 \log_{10}(I/I_0), \qquad I_0 = \qty{1.0e-12}{W/m^2}, \]
dalam \emph{desibel}\index{desibel} (\unit{dB}): ambang pendengarannya
\qty{0}{dB}, rasa sakitnya di dekat \qty{120}{dB}.
\end{definition}

\begin{proposition}[Aritmetika desibel]\label{prop:g12:sound-acoustics:db-arithmetic}
Mengalikan $I$ dengan $k$ menambahkan $10\log_{10}k$ \omterm{def:g12:sound-acoustics:level}{desibel}. Khususnya
$\times 10$ menambahkan \qty{10}{dB}; $\times 2$ menambahkan
$10\log_{10}2 \approx \qty{3.0}{dB}$; $N$ sumber serupa yang tidak
\omterm{def:g12:mechanical-waves:phase}{sefase} menjumlahkan \emph{intensitasnya}, tidak pernah tarafnya,
sehingga menghasilkan $L_1 + 10\log_{10}N$; menggandakan jarak ke sumber
yang kecil ($I$ dibagi empat) membuang sekitar \qty{6.0}{dB}.
\end{proposition}

\begin{proof}
$10\log_{10}(kI/I_0) = 10\log_{10}(I/I_0) + 10\log_{10}k$; untuk
jaraknya: \cref{prop:g12:sound-acoustics:inverse-square}, dengan
$k = 1/4$.
\end{proof}

\begin{method}[Bekerja dalam desibel]\label{met:g12:sound-acoustics:db}
\begin{enumerate}
  \item $L = 10\log_{10}(I/I_0)$; sebaliknya:
        $I = I_0 \times 10^{L/10}$.
  \item Jangan pernah menjumlahkan tarafnya: ubahlah ke intensitas,
        jumlahkan, lalu ubah kembali --- atau pakailah
        $+\qty{3}{dB}$ tiap penggandaan dan $+\qty{10}{dB}$ tiap
        sepuluh kali lipat.
  \item Tiap penggandaan jarak ke sumber yang kecil berharga
        \qty{6}{dB}.
\end{enumerate}
\end{method}

\begin{remark}[Kerusakan pendengaran]\label{rem:g12:sound-acoustics:damage}
Kehilangan pendengaran adalah soal \omterm{rem:g11:nucleus-radioactivity:dose}{dosis}: \qty{85}{dB} aman selama
delapan jam, dan tiap \qty{3}{dB} tambahan --- yaitu penggandaan
intensitasnya --- menyeparuhkan waktu amannya; sel rambut yang hilang
tidak tumbuh kembali.
\end{remark}

\section{Warna bunyi dan harmonik}

\begin{definition}[Harmonik]\label{def:g12:sound-acoustics:harmonics}
Sebuah \omterm{def:g10:signals-and-waves:sound}{bunyi} \omterm{def:g10:signals-and-waves:period}{periodik} berfrekuensi $f_1$ --- yaitu \emph{nada
dasar}\index{nada dasar}nya --- pada umumnya merupakan jumlah sinus pada
\omterm{def:g10:signals-and-waves:frequency}{frekuensi} $f_n = n f_1$, $n = 1, 2, 3, \dots$, yang disebut
\emph{harmonik}\index{harmonik}nya. \omterm{def:g10:signals-and-waves:sound}{Bunyi} yang tinggal nada dasarnya
saja disebut \emph{nada murni}\index{nada murni}, seperti \omterm{def:g10:signals-and-waves:sound}{bunyi} garpu
tala.
\end{definition}

\begin{definition}[Spektrum dan warna bunyi]\label{def:g12:sound-acoustics:timbre}
\emph{Spektrum}\index{spektrum} sebuah \omterm{def:g10:signals-and-waves:sound}{bunyi} adalah diagram batang
\omterm{def:g12:sound-acoustics:harmonics}{harmoniknya}: satu batang tiap $f_n$, dengan tingginya menyatakan
kekuatan \omterm{def:g12:sound-acoustics:harmonics}{harmonik} itu. Telinga mendengar $f_1$ sebagai \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nadanya}
dan \emph{campuran} kekuatannya sebagai \emph{warna
bunyi}\index{warna bunyi} --- yaitu warna \omterm{def:g10:signals-and-waves:sound}{bunyinya}: alat musik yang
memainkan nada yang sama berbagi $f_1$, dan hanya berbeda pada
spektrumnya.
\end{definition}

Nada A sebuah suling nyaris merupakan \omterm{def:g12:sound-acoustics:harmonics}{nada murni}; sebuah biola yang

\omterm{def:g12:sound-acoustics:timbre}{spektrum} yang kamu hafal.
\end{example}

\section{Senar yang bergetar}

\begin{definition}[Gelombang berdiri]\label{def:g12:sound-acoustics:standing-wave}
\omterm{def:g10:signals-and-waves:wave}{Gelombang} yang terkurung di antara ujung tetap sebuah senar dapat
mengendap menjadi \emph{gelombang berdiri}\index{gelombang berdiri}:
sebuah ayunan di tempat, dengan titik yang tidak bergerak ---
\emph{simpul}\index{simpul} --- berselang-seling dengan titik yang
ayunannya terbesar --- \emph{perut}\index{perut}. Tidak ada lagi yang
merambat; senarnya bergetar dalam pola yang tetap.
\end{definition}

\begin{proposition}[Ragam getar sebuah senar]\label{prop:g12:sound-acoustics:string-modes}
Sebuah senar sepanjang $L$, yang kedua ujungnya tetap dan membawa
\omterm{def:g10:signals-and-waves:wave}{gelombang} berlaju $v$, hanya bergetar secara mantap dalam pola yang
memuat sejumlah bulat $n$ setengah \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} di antara \omterm{def:g12:sound-acoustics:standing-wave}{simpul}
ujungnya yang dipaksakan:
\[ \lambda_n = \frac{2L}{n}, \quad f_n = n\,\frac{v}{2L} = n f_1, \quad n = 1, 2, 3, \dots \]
Ragam yang terendah $f_1 = v/(2L)$ adalah \omterm{def:g12:sound-acoustics:harmonics}{nada dasarnya}; yang lain
tepat merupakan \omterm{def:g12:sound-acoustics:harmonics}{harmoniknya}. \admitted
\end{proposition}

\begin{remark}[Di mana buktinya yang jujur bersemayam]\label{rem:g12:sound-acoustics:honest}
Bahwa \omterm{def:g10:signals-and-waves:wave}{gelombang} yang terkurung memilih tepat ragam ini mengikuti dari
menumpangkan \omterm{def:g10:signals-and-waves:wave}{gelombangnya} dengan pantulannya, sebuah kalkulus yang
dikerjakan pada jilid Tahun ke-1. Ilmu ukurnya sudah meyakinkan: kedua
ujungnya harus tetap diam, sehingga sejumlah bulat setengah \omterm{def:g12:mechanical-waves:wavelength}{panjang
gelombang} harus muat di dalamnya.
\end{remark}

\begin{example}[Senar A sebuah biola]\label{ex:g12:sound-acoustics:violin}
Senar A sebuah biola memiliki $L = \qty{32.8}{cm}$ dan berbunyi
$f_1 = \qty{440}{Hz}$: \omterm{def:g10:signals-and-waves:wave}{gelombangnya} berjalan di sepanjangnya dengan
$v = 2Lf_1 = 2 \times 0.328 \times 440 \approx \qty{289}{m/s}$ --- yang
ditetapkan \omterm{def:g11:circuits-and-power:voltage}{tegangan} dan massanya, lalu ditala dengan pasaknya. Menekan
senarnya dengan jari memendekkan $L$ dan menaikkan setiap $f_n$
sekaligus: \omterm{def:g12:sound-acoustics:timbre}{spektrumnya} meluncur naik.
\end{example}

\begin{remark}[Mengapa alat musik berbeda-beda]\label{rem:g12:sound-acoustics:instruments}
Cara senarnya dibangkitkan --- dipetik, digesek, dipukul --- menetapkan
seberapa banyak tiap ragamnya hadir, dan badannya menguatkan sebagian
\omterm{def:g12:sound-acoustics:harmonics}{harmoniknya} lebih daripada yang lain; pembangkitan beserta badannya
menetapkan \omterm{def:g12:sound-acoustics:timbre}{spektrumnya}, sehingga $f_1$ yang sama meninggalkan gitar,
biola dan piano yang masing-masing berbunyi seperti dirinya sendiri.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:1}
Di udara ($v = \qty{340}{m/s}$), tentukan \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} (a) nada bas
\qty{55}{Hz}; (b) \omterm{def:g10:signals-and-waves:sound}{bunyi} bip \qty{1.0}{kHz}; (c) rengekan \qty{15}{kHz}.
Bandingkan yang terbesar dengan yang terkecil.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:2}
Golongkan sebagai \omterm{def:g10:signals-and-waves:ultrasound}{infrasonik}, \omterm{def:g10:signals-and-waves:sound}{bunyi} yang terdengar, atau \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik}:
\qty{8}{Hz}; \qty{100}{Hz}; \qty{15}{kHz}; \qty{40}{kHz}; \qty{3}{MHz}.
Di antara yang terdengar, yang mana yang nadanya lebih rendah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:3}
Seorang pekerja memukul rel yang berjarak \qty{1.0}{km}. Hitunglah waktu
tempuhnya di dalam bajanya dan di dalam udaranya. Apa yang didengar
seorang penyimak yang satu telinganya menempel pada relnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:4}
(a) Berapa taraf intensitasnya untuk $I = \qty{1.0e-5}{W/m^2}$?
(b) Berapa intensitasnya pada \qty{60}{dB}? (c) Sebutkan \omterm{def:g10:signals-and-waves:sound}{bunyi}
sehari-hari yang dekat dengan masing-masing tarafnya.
\end{exercise}

\section{Soal: Bengkel pembuat gitar}

\begin{problem}[{Soal akhir pekan --- bengkel pembuat gitar: sebuah
senar gitar ditata fret demi fret, harmoniknya ditala menjadi warna
bunyi, sebuah pergelaran dijaga di bawah garis bahaya, dan sebuah
perburuan cacat kayu dengan ultrasonik}]
\label{pb:g12:sound-acoustics:1}
Seorang pembuat gitar sedang menyelesaikan sebuah gitar. Senar A-nya
memiliki panjang bergetar $L = \qty{65.0}{cm}$ dan harus berbunyi
$f_1 = \qty{110}{Hz}$. Ambillah \qty{340}{m/s} untuk \omterm{def:g10:signals-and-waves:sound}{bunyi} di udara, dan
terimalah $f_1 = v/(2L)$
(\cref{prop:g12:sound-acoustics:string-modes}), dengan $v$ adalah laju
\omterm{def:g10:signals-and-waves:wave}{gelombang} pada senarnya.

\medskip
\noindent\textbf{Bagian I --- Menata papan fretnya.}
\begin{enumerate}
  \item Hitunglah laju \omterm{def:g10:signals-and-waves:wave}{gelombang} $v$ pada senarnya.
  \item Hitunglah \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} \omterm{def:g12:sound-acoustics:harmonics}{nada dasarnya} \emph{pada senarnya},
        lalu \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} \omterm{def:g10:signals-and-waves:sound}{bunyi} \qty{110}{Hz} itu \emph{di
        udaranya}. Mengapa keduanya berbeda pada \omterm{def:g10:signals-and-waves:frequency}{frekuensi} yang sama?
  \item Oktafnya (\qty{220}{Hz}): panjang bergetar berapa yang
        menghasilkannya? Di mana fret kedua belasnya harus berada,
        sebagai bagian dari $L$?
  \item Kuintnya (\qty{165}{Hz}): panjang bergetar berapa yang
        menghasilkannya?
  \item Tiap fret menaikkan nadanya satu seminada, dengan faktor
        $2^{1/12}$. Periksalah dengan logaritma berapa seminada yang
        memisahkan \qty{110}{Hz} dari \qty{220}{Hz}; berapa panjangnya
        pada fret pertama?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:g12:sound-acoustics:harmonics}{Harmonik} dan \omterm{def:g12:sound-acoustics:timbre}{warna bunyi}.}
\begin{enumerate}[resume]
  \item Daftarkan lima \omterm{def:g12:sound-acoustics:harmonics}{harmonik} pertama senar A yang terbuka.
  \item Nada A seorang pemain biola berbunyi \qty{440}{Hz}. \omterm{def:g12:sound-acoustics:harmonics}{Harmonik}
        yang keberapa dari senar gitarnya itu, dan selang musik apa yang
        memisahkannya dari \omterm{def:g12:sound-acoustics:harmonics}{nada dasarnya}? Dan bagaimana dengan
        \qty{330}{Hz}?
  \item Si pembuat gitar memetik di dekat jembatannya, lalu di atas
        papan jarinya: \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nadanya} sama, \omterm{def:g12:sound-acoustics:timbre}{warna bunyinya} berbeda.
        Jelaskan dengan \omterm{def:g12:sound-acoustics:timbre}{spektrumnya}.
  \item Sentuhan ringan di titik tengahnya selagi memetik memaksa
        adanya sebuah \omterm{def:g12:sound-acoustics:standing-wave}{simpul} di sana. \omterm{def:g12:sound-acoustics:harmonics}{Harmonik} yang mana yang bertahan,
        dan \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nada} apa yang terdengar?
  \item Berapa \omterm{def:g12:sound-acoustics:harmonics}{harmonik} senar terbuka itu yang, pada asasnya, terletak
        di dalam rentang yang terdengar?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Pergelarannya.}
Pada konser pertama gitar itu, tiap alat musiknya memancarkan \omterm{def:g11:work-of-force:power}{daya}
akustik $P = \qty{6.0e-3}{W}$ sebagai sumber kecil, sama rata ke segala
arah.
\begin{enumerate}[resume]
  \item Hitunglah intensitasnya pada $r = \qty{3.0}{m}$ dari satu alat
        musik.
  \item Simpulkan taraf intensitasnya di sana.
  \item Sebuah kuartet bermain: taraf berapa yang diberikan empat alat
        musik semacam itu di tempat yang sama? Mengapa nilainya bukan
        empat kali tarafnya?
  \item Dua belas alat musik bermain: berapa tarafnya pada
        \qty{3.0}{m}?
  \item Seberapa jauh ke belakang seorang penyimak harus duduk agar
        kelompoknya turun ke \qty{85}{dB}?
  \item Aturan keselamatannya: \qty{85}{dB} aman selama delapan jam,
        dan tiap \qty{+3}{dB} menyeparuhkan waktu amannya. Berapa lama
        penyimak di baris depan pada pertanyaan 14 boleh tinggal dengan
        aman?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Cacat di dalam kayunya.}
Sebelum dipernis, si pembuat gitar menduga sebuah balok leher setebal
\qty{45}{mm} dengan penguji \qty{2.0}{MHz}; pada kayu ini ambillah
$v = \qty{5.0e3}{m/s}$.
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} \omterm{def:g10:signals-and-waves:ultrasound}{ultrasoniknya} di dalam kayunya.
        Mengapa \omterm{def:g10:signals-and-waves:wave}{gelombang} pemburu cacat harus \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik} dan bukan
        \omterm{def:g10:signals-and-waves:wave}{gelombang} yang terdengar?
  \item Hitunglah \omterm{def:g12:mechanical-waves:delay}{waktu tunda} \omterm{rem:g10:signals-and-waves:honest}{gemanya} dari muka belakang baloknya.
  \item Sebuah \omterm{rem:g10:signals-and-waves:honest}{gema} tiba sesudah \qty{7.2}{\micro s}: seberapa dalam
        cacatnya?
  \item Tulislah kartu bengkelnya: laju \omterm{def:g10:signals-and-waves:wave}{gelombang} pada senarnya,
        panjang pada fret kedua belas dan fret pertama, jarak menyimak
        yang aman, dan kedalaman cacatnya.
\end{enumerate}
\end{problem}
