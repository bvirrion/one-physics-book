\chapter{Osilator Mekanik dan Pengukuran Waktu}\label{ch:g12:oscillators-and-time}

Sebuah jam bandul berdetak: tiap detaknya adalah bandul kuningan yang
melintasi garis tegak, terulang lima belas juta kali sejak Tahun Baru.
Sekeping kuarsa di dalam arlojimu melentur \num{32768} kali tiap sekon;
sebutir \omterm{def:g11:fundamental-interactions:ladder}{atom} sesium berdengung sembilan miliar kali lebih cepat. Segala
sesuatu yang berayun dapat menyimpan waktu: bab ini mengikuti ayunannya,
dari Huygens sampai ke \omterm{def:g10:universal-gravitation:satellite}{satelitnya}.

\section{Osilasi bebas}

\begin{definition}[Osilator, periode, frekuensi, amplitudo]\label{def:g12:oscillators-and-time:oscillator}
\emph{Osilator mekanik}\index{osilator} adalah sistem yang, ketika
disimpangkan dari \emph{kedudukan setimbang}\index{kedudukan
setimbang} yang mantap lalu dilepaskan, berayun maju mundur di sekitarnya:
inilah \emph{osilasi bebas}\index{osilasi bebas}. Geraknya berulang
setelah \emph{periode}\index{periode} $T_0$ (\unit{s});
\emph{frekuensinya}\index{frekuensi} $f_0 = 1/T_0$
(\unit{Hz}, \cref{ch:g12:mechanical-waves});
\emph{amplitudo}\index{amplitudo} adalah simpangan terbesarnya.
\end{definition}

\begin{example}[Sekebun osilator]\label{ex:g12:oscillators-and-time:zoo}
Ayunan taman bermain: $T_0 \approx \qty{3}{s}$. Senar A sebuah gitar:
$f_0 = \qty{110}{Hz}$. Sebuah hablur kuarsa: \qty{32768}{Hz}. Osilasi
sesium yang menakrifkan sekon: sekitar \qty{9.2e9}{Hz}. Sepuluh orde
besaran, satu tugas: menghitung.
\end{example}

\section{Bandul sederhana}

\begin{definition}[Bandul sederhana]\label{def:g12:oscillators-and-time:pendulum}
\emph{Bandul sederhana}\index{bandul sederhana}: sebutir beban kecil bermassa
$m$ pada kawat tak mulur sepanjang $L$ yang massanya terabaikan,
berayun di dalam bidang tegak. Dari kedua \omterm{def:g10:inertia:force}{gaya} pada bebannya
(\cref{ch:g12:newtons-laws}), tegangannya tegak lurus terhadap
geraknya, sedangkan komponen \omterm{def:g10:universal-gravitation:weight}{beratnya} sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasannya} menunjuk kembali
ke arah titik terendahnya: \omterm{def:g10:universal-gravitation:weight}{beratnya} adalah \emph{\omterm{def:g10:inertia:force}{gaya} pemulih}%
\index{gaya pemulih}, yang melawan simpangannya.
\end{definition}

\begin{proposition}[Periode bandul sederhana]\label{prop:g12:oscillators-and-time:pendulum-period}
\index{bandul sederhana!periode}
Untuk \omterm{def:g12:oscillators-and-time:oscillator}{amplitudo} yang kecil (di bawah sekitar $15^\circ$), \omterm{def:g12:oscillators-and-time:oscillator}{periodenya} adalah
\[
T_0 = 2\pi \sqrt{\frac{L}{g}}, \qquad g = \qty{9.81}{m/s^2}.
\]
\omterm{def:g10:signals-and-waves:period}{Periodenya} tidak bergantung pada massa bebannya maupun --- selama sudutnya
tetap kecil --- pada \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya}: ayunan yang kecil bersifat
\emph{isokron}\index{isokronisme}, semuanya menghabiskan waktu yang sama.
\end{proposition}
\admitted

\begin{remark}[Analisis dimensi, dan tempat $2\pi$ bersemayam]\label{rem:g12:oscillators-and-time:dimensional}
Rumusnya dapat diterka. Dari $m$ (\unit{kg}), $L$ (\unit{m}) dan
$g$ (\unit{m/s^2}), satu-satunya gabungan yang bersatuan waktu adalah
$\sqrt{L/g}$ --- dan massanya \emph{tidak mungkin} masuk, sebab tidak ada
data lain yang membawa \omterm{def:g10:orders-of-magnitude:unit}{kilogram} untuk meniadakannya. Dimensi tidak dapat
memberikan bilangan murni di depannya: $2\pi$ itu datang dari penyelesaian
persamaan geraknya, yang dikerjakan pada jilid Tahun ke-1. Syarat sudut
kecilnya pun bukan pilihan: pada $30^\circ$ \omterm{def:g12:oscillators-and-time:oscillator}{periode} sejatinya $2\%$ lebih
lama, dan keisokronannya gugur.
\end{remark}

\begin{example}[Bandul sesekon]\label{ex:g12:oscillators-and-time:seconds}
Panjang berapa yang berdetak sesekon, $T_0 = \qty{2.00}{s}$, satu detak tiap
arahnya? Dengan menyelesaikannya, $L = g\,T_0^2/4\pi^2 = 9.81 \times 2.00^2/4\pi^2 \approx
\qty{0.994}{m}$. Hampir semeter: jam bandul menjulang karena
sekon itu panjang.
\end{example}

\section{Osilator pegas--massa}

\begin{definition}[Gaya pemulih sebuah pegas]\label{def:g12:oscillators-and-time:spring}
Sebuah kereta luncur bermassa $m$ meluncur pada rel mendatar tanpa \omterm{def:g10:inertia:inventory}{gesekan},
terikat pada pegas ber\emph{kekakuan}\index{kekakuan} $k$ (\unit{N/m}). Misalkan
$x$ simpangannya dari kesetimbangannya. Ketika teregang atau termampat,
pegasnya menarik atau mendorong keretanya kembali dengan \omterm{def:g10:inertia:force}{gaya} $F = -kx$:
sebanding dengan simpangannya dan berlawanan dengannya --- sebuah \omterm{def:g12:oscillators-and-time:pendulum}{gaya
pemulih} lagi, kini yang dipasok kekenyalannya.
\end{definition}

\begin{proposition}[Persamaan gerak dan penyelesaiannya]\label{prop:g12:oscillators-and-time:spring-period}
\index{osilator pegas--massa}
\omterm{thm:g12:newtons-laws:second}{Hukum kedua Newton} sepanjang relnya (\cref{ch:g12:newtons-laws}) memberikan
$m \frac{d^2x}{dt^2} = -kx$, yaitu $\frac{d^2x}{dt^2} =
-\frac{k}{m}\,x$. Berapa pun tetapan $A$ dan $\varphi$,
\[
x(t) = A \cos\!\left(\frac{2\pi t}{T_0} + \varphi\right),
\qquad T_0 = 2\pi\sqrt{\frac{m}{k}},
\]
memenuhi persamaan ini: osilasi berperiode $T_0$, yang bebas
dari \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya}.
\end{proposition}

\begin{proof}
Dengan menurunkannya, $\frac{dx}{dt} = -\frac{2\pi}{T_0}A \sin(\frac{2\pi
t}{T_0} + \varphi)$; sekali lagi, $\frac{d^2x}{dt^2} = -(\frac{2\pi}{T_0})^2
A\cos(\frac{2\pi t}{T_0} + \varphi) = -(\frac{2\pi}{T_0})^2\,x(t)$,
yang sama dengan $-(k/m)\,x$ tepat ketika $(2\pi/T_0)^2 = k/m$.
\end{proof}

\begin{remark}[Segala gerak yang ada]\label{rem:g12:oscillators-and-time:uniqueness}
Bahwa \emph{setiap} gerak keretanya merupakan salah satu fungsi ini
diterima saja, dan dibuktikan pada jilid Tahun ke-1. Dan tidak ada $g$ pada $2\pi\sqrt{m/k}$:
yang lebih kaku berayun lebih cepat, yang lebih berat lebih lambat, sama saja di Bulan.
\end{remark}

\begin{definition}[Amplitudo dan fase]\label{def:g12:oscillators-and-time:phase}
Pada $x(t) = A\cos(2\pi t/T_0 + \varphi)$ tetapan positif $A$ adalah
\omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya} --- ujung-ujungnya $x = \pm A$ --- dan $2\pi t/T_0 +
\varphi$ adalah \emph{fasenya}\index{fase}, yang tumbuh sebesar $2\pi$ tiap
\omterm{def:g12:oscillators-and-time:oscillator}{periode}; \emph{fase awal}\index{fase awal} $\varphi$
mencatat awalnya: dilepaskan dari keadaan diam pada $x = A$, $\varphi = 0$. Sebagai
contoh $m = \qty{0.20}{kg}$ pada $k = \qty{20}{N/m}$ memberikan $T_0 =
2\pi\sqrt{0.20/20} = \qty{0.63}{s}$; ditarik sampai \qty{5.0}{cm} lalu
dilepaskan, $x(t) = 5.0\cos(2\pi t/0.63)$ dalam sentimeter.
\end{definition}

\section{Energi, peredaman, resonansi}

\begin{proposition}[Energi osilatornya]\label{prop:g12:oscillators-and-time:energy}
Pegas yang teregang sebesar $x$ menyimpan \omterm{def:g10:energy-conservation:energy}{energi} potensial elastis $E_p =
\tfrac12 kx^2$ (yang dibenarkan pada \cref{ch:g12:work-and-energy}). Selama
osilasi tanpa \omterm{def:g10:inertia:inventory}{gesekan} \omterm{def:g11:mechanical-energy:em}{energi mekaniknya}
\[
E_m = E_k + E_p = \tfrac12 m v^2 + \tfrac12 k x^2 = \tfrac12 k A^2
\]
bernilai tetap: seluruhnya elastis di ujung-ujungnya, seluruhnya kinetik di
kesetimbangannya. Bandulnya memainkan duet yang sama antara $E_k$ dan
$E_p$ gravitasi (\cref{ch:g11:mechanical-energy}).
\end{proposition}

\begin{proof}
Dengan $v = \frac{dx}{dt}$, turunan $E_m$ adalah $mv\frac{dv}{dt}
+ kx\frac{dx}{dt} = v\,(m\frac{d^2x}{dt^2} + kx) = 0$ menurut persamaan
geraknya; nilai $\tfrac12 kA^2$ dibaca di salah satu ujungnya.
\end{proof}

\begin{definition}[Osilasi teredam]\label{def:g12:oscillators-and-time:damping}
\omterm{def:g10:inertia:inventory}{Gesekan} menguras \omterm{def:g11:mechanical-energy:em}{energi mekanik} dari setiap \omterm{def:g12:oscillators-and-time:oscillator}{osilator} yang sesungguhnya:
osilasi bebasnya menjadi \emph{teredam}\index{peredaman}. Di bawah
peredaman yang ringan sistemnya masih berosilasi, dengan \omterm{def:g12:oscillators-and-time:oscillator}{amplitudo} yang
meredup tetapi waktu pengulangan yang hampir tidak berubah, yaitu
\emph{pseudoperiode}%
\index{pseudoperiode}, yang dekat dengan $T_0$; di bawah peredaman yang berat (madu)
sistemnya merayap kembali ke kesetimbangannya tanpa pernah melampauinya; \omterm{def:g10:energy-conservation:energy}{energinya},
yang sebanding dengan kuadrat \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya}, meluruh menjadi kalor.
\end{definition}

\begin{definition}[Osilasi paksa dan resonansi]\label{def:g12:oscillators-and-time:resonance}
Doronglah sebuah \omterm{def:g12:oscillators-and-time:oscillator}{osilator} secara berkala --- \emph{osilasi paksa}%
\index{osilasi paksa} --- dan setelah peralihan yang singkat ia
berosilasi pada \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} \emph{pemaksanya} $f$, bukan pada \omterm{def:g12:oscillators-and-time:oscillator}{frekuensinya}
sendiri. \omterm{def:g10:signals-and-waves:amplitude}{Amplitudonya} memuncak ketika $f$ mendekati \emph{\omterm{def:g12:oscillators-and-time:oscillator}{frekuensi}
alami}%
\index{frekuensi alami} $f_0$ \omterm{def:g12:oscillators-and-time:oscillator}{osilator} bebasnya: inilah
\emph{resonansi}\index{resonansi}, ketika tiap dorongan tiba selaras dengan
geraknya, memasok \omterm{def:g10:energy-conservation:energy}{energi} tiap daur; kian ringan \omterm{def:g12:oscillators-and-time:damping}{peredamannya},
kian tinggi dan kian tajam puncaknya.
\end{definition}

\begin{example}[Resonansi yang membangun dan yang merusak]\label{ex:g12:oscillators-and-time:resonance-zoo}
Orang tua yang mendorong ayunan mendorong pada \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} ayunan itu sendiri
--- \omterm{def:g12:oscillators-and-time:resonance}{resonansi} yang dipekerjakan. Pada \num{2000} Jembatan Milenium yang baru
di London berayun mencemaskan ketika langkah orang banyak terkunci pada
\omterm{def:g12:oscillators-and-time:resonance}{frekuensi alami} di dekat \qty{1}{Hz}; jembatan itu ditutup dua tahun untuk
dipasangi peredam. Setiap arloji kuarsa menjaga hablurnya tetap bernyanyi
dengan memaksanya pada \omterm{def:g12:oscillators-and-time:resonance}{resonansi}, \qty{32768}{Hz}.
\end{example}

\section{Mengukur waktu}

\begin{method}[Mengukur waktu sebuah periode]\label{met:g12:oscillators-and-time:timing}
Jangan pernah mengukur satu osilasi saja: jalankanlah pengukur waktunya saat
bebannya melintasi garis tegak (paling cepat, paling mudah ditimbang),
hitunglah $N = 50$ osilasi, lalu bagilah dengan $N$ --- galat tanggapan
\qty{0.2}{s} menjadi \qty{4}{ms} pada $T_0$. Ulangilah, lalu bandingkan
kedua jalannya.
\end{method}

\begin{example}[Tiga abad detak]\label{ex:g12:oscillators-and-time:clocks}
Huygens membangun jam bandul yang pertama pada \num{1657}: keisokronannya
membawa jam dari selisih seperempat jam tiap hari menjadi beberapa sekon, dan
bandul yang terimbangi sampai ke pecahan sekon tiap hari. Arloji tangan
kuarsa (\num{1969}) menukar batangnya dengan hablur yang melentur pada
\qty{32768}{Hz}: persepuluh sekon tiap bulan. Sejak \num{1967} sekon
itu sendiri \emph{ditakrifkan} oleh osilasi \omterm{def:g11:fundamental-interactions:ladder}{atom}: lamanya
\num{9192631770} \omterm{def:g12:oscillators-and-time:oscillator}{periode} radiasi \omterm{def:g10:signals-and-waves:wave}{gelombang} mikro dari sebuah
transisi dalam sesium-133 (\cref{ch:g12:quantum-world}). Jam
sesium yang terbaik menyimpang satu sekon dalam seratus juta tahun.
\end{example}

\begin{remark}[Mengapa mengejar nanosekon]\label{rem:g12:oscillators-and-time:gps}
Penerima GPS menemukan kedudukannya dengan mengukur waktu \omterm{def:g10:signals-and-waves:signal}{sinyal} radio dari
\omterm{def:g10:universal-gravitation:satellite}{satelitnya}; cahaya menempuh \qty{30}{cm} tiap nanosekon, sehingga galat jam
sebesar satu mikrosekon menyalahtempatkanmu sejauh \qty{300}{m}. Karena itu
\omterm{def:g10:universal-gravitation:satellite}{satelitnya} membawa jam \omterm{def:g11:fundamental-interactions:ladder}{atom} --- dan pada ketelitian itu laju sebuah
jam bergantung pada secepat apa ia bergerak dan setinggi apa ia berada,
pengaruh yang kita temui pada \cref{ch:g12:special-relativity}.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:oscillators-and-time:1}

(b) Panjangnya dilipatempatkan: apa yang terjadi pada \omterm{def:g12:oscillators-and-time:oscillator}{periodenya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:oscillators-and-time:2}
Jantung yang sedang beristirahat berdenyut $72$ kali tiap menit; sayap

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:oscillators-and-time:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:oscillators-and-time:4}
Puncak berturut-turut sebuah \omterm{def:g12:oscillators-and-time:oscillator}{osilator} \omterm{def:g12:oscillators-and-time:damping}{teredam}: $2.0$, $1.5$, $1.13$,
\qty{0.85}{cm} pada $t = 0$, $2.0$, $4.0$, \qty{6.0}{s}. Bacalah

tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode}; ramalkanlah puncak berikutnya.
\end{exercise}

\section{Soal: Meja kerja tukang jam}

\begin{problem}[{Soal akhir pekan --- meja kerja tukang jam: sebuah jam
dinding yang dikembalikan ke waktunya --- lima puluh ayunan yang diukur, sebatang
kuningan yang memuai saat kemarau, dorongan berbisik dari kaitannya, seorang
penantang kuarsa --- dengan putusan yang dijatuhkan dalam sekon tiap bulan}]
\label{pb:g12:oscillators-and-time:1}
Sebuah jam dinding berbandul tiba di meja kerjanya dalam keadaan ``berjalan
keliru''. Perlakukanlah batang kuningan dan beban \omterm{def:g10:universal-gravitation:weight}{beratnya} sebagai \omterm{def:g12:oscillators-and-time:pendulum}{bandul
sederhana} berpanjang efektif $L$, dengan massa bebannya $m = \qty{1.2}{kg}$;
rodagiginya menghitung tiap osilasi penuh sebagai tepat \qty{2.000}{s}.

\medskip
\noindent\textbf{Bagian I --- Mengukur denyut jamnya.}
\begin{enumerate}
  \item Mengapa mengukur waktu $50$ osilasi dan bukan satu? Taksirlah galat
        yang tersisa pada $T_0$ jika waktu tanggapanmu \qty{0.2}{s}.
  \item Pengukur waktunya membaca \qty{99.4}{s} untuk $50$ osilasi: berapa \omterm{def:g12:oscillators-and-time:oscillator}{periodenya}?
  \item Perolehlah panjang efektif bandulnya saat ini.
  \item Panjang berapa yang akan memberikan tepat \qty{2.000}{s}? Haruskah
        mur penyetelnya menaikkan atau menurunkan bebannya, dan sebanyak apa?
  \item Tanpa disetel, apakah jamnya cepat atau lambat, berapa sekon tiap hari?
  \item Kepekaannya terhadap $g$: berapa \omterm{def:g12:oscillators-and-time:oscillator}{periodenya} di Bulan
        ($g = \qty{1.62}{m/s^2}$)? Berapa galat hariannya di observatorium
        tempat $g = \qty{9.80}{m/s^2}$?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Kelesuan kemarau.} Kuningan memuai: pemanasan
$\Delta\theta$ meregangkan batangnya menjadi $L' = L\,(1 + \lambda\,
\Delta\theta)$, $\lambda = \num{1.9e-5}$ tiap \omterm{def:g10:light-spectra:kelvin}{kelvin}. Jamnya
disetel pada \qty{20}{\celsius}.
\begin{enumerate}[resume]
  \item Hitunglah $\Delta L$ untuk ruangan kemarau pada \qty{25}{\celsius}.
  \item Dengan $\sqrt{1+x} \approx 1 + x/2$, tunjukkan bahwa
        $\Delta T_0 / T_0 = \tfrac12 \lambda\, \Delta\theta$.
  \item Hitunglah $\Delta T_0/T_0$ dan galat harian kemaraunya: cepat atau lambat?
  \item Pertanyaan yang sama untuk lorong berhawa dingin pada \qty{10}{\celsius}.
  \item Jam presisi zaman dahulu memakai bandul ``panggangan'' yang memadukan
        batang kuningan dengan batang baja (yang memuai kira-kira separuhnya)
        dalam arah yang berselang-seling. Jelaskan bagaimana ini dapat menahan lajunya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Dorongan berbisik.} Dengan kaitannya
dilepaskan, ayunannya, yang dimulai pada \omterm{def:g12:oscillators-and-time:oscillator}{amplitudo} $\theta_m = 2.00^\circ$,
menyusut separuh \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya} dalam \qty{280}{s}; saat bertugas, dorongan
mungil dari kaitannya tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode} menjaga $\theta_m$ tetap.
\begin{enumerate}[resume]
  \item Berapa \omterm{def:g12:oscillators-and-time:oscillator}{periode} yang dihabiskan penyusutan separuhnya itu?
  \item Di ujungnya bebannya telah menanjak $h = L(1 - \cos\theta_m)
        \approx L\,\theta_m^2/2$ (dalam radian): perolehlah $E_m = \tfrac12
        m g L\,\theta_m^2$ lalu hitunglah
        (\cref{ch:g11:mechanical-energy}).
  \item \omterm{def:g10:energy-conservation:energy}{Energinya} sebanding dengan kuadrat \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya}: berapa bagian
        $E_m$ yang hilang tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode}?
  \item Perolehlah \omterm{def:g10:energy-conservation:energy}{energi} tiap dorongannya, dan \omterm{def:g11:work-of-force:power}{daya} rata-rata yang dibutuhkan.
  \item Beban penggeraknya $M = \qty{2.5}{kg}$ turun sejauh $h =
        \qty{1.0}{m}$ tiap pekan: cukupkah \omterm{def:g10:energy-conservation:energy}{energi} itu, pada dayaguna berapa?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Penantang kuarsanya.} Sebuah gerak
kuarsa berosilasi pada $f = \qty{32768}{Hz}$ dan menyimpang sekitar
\qty{0.3}{s} tiap bulan; bandul yang dipulihkan masih menyimpang
\qty{0.5}{s} tiap hari.
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g12:oscillators-and-time:oscillator}{periode} kuarsanya. Tunjukkan bahwa $\num{32768} = 2^{15}$: mengapa
        pangkat dua justru yang diinginkan sirkuit sebuah arloji?
  \item Hitunglah simpangan nisbi kuarsanya dan bandul yang dipulihkan itu,
        lalu masing-masing dalam sekon tiap bulan. Siapa yang menang, dengan faktor berapa?
  \item Baku sesium --- \num{9192631770} \omterm{def:g12:oscillators-and-time:oscillator}{periode} membentuk satu
        sekon --- menyimpang \qty{1}{s} tiap seratus juta tahun: berapa
        simpangan nisbinya? Berapa orde besaran di bawah kuarsanya,
        dan mengapa GPS menuntutnya?
  \item Putusannya, satu kalimat beserta angkanya: apa yang diperoleh jam
        yang dipulihkan itu di dindingnya, dan siapa yang kini memiliki sekon?
\end{enumerate}
\end{problem}
