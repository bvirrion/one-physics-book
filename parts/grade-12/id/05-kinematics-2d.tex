\chapter{Kinematika pada Dua Dimensi}\label{ch:g12:kinematics-2d}

Amatilah sebuah mobil menempuh bundaran dengan tetap \qty{30}{km/h}:
jarum penunjuknya tidak pernah bergerak, namun setiap penumpang merasa
tertarik ke samping --- geraknya sedang berubah, dan yang berubah
bukanlah lajunya. Satu bilangan tiap saat tidak lagi memadai: gerak di
dalam bidang memerlukan anak panah. Bab ini melengkapi sebuah titik yang
bergerak dengan tiga anak panah --- posisi, kecepatan, percepatan ---
yang dirangkai turunan, lalu membaca seluruh keluarga geraknya dari ilmu
ukurnya.

\section{Vektor posisi}

Gerak hanya ada terhadap \omterm{def:g10:relative-motion:frame}{kerangka acuan} yang disebutkan --- sebuah benda
tegar beserta jam (\cref{ch:g10:relative-motion}). Yang baru tahun ini
adalah ketepatannya: lekatkanlah pada kerangkanya sebuah titik asal $O$
beserta dua sumbu berskala yang saling tegak lurus, $Ox$ dan $Oy$, lalu
catatlah letak titiknya pada setiap saat, sebagai sepasang fungsi.

\begin{definition}[Vektor posisi]\label{def:g12:kinematics-2d:position}
Di dalam kerangka yang dilengkapi titik asal $O$ dan sumbu $Ox$, $Oy$,
\emph{vektor posisi}\index{vektor posisi} sebuah titik bergerak $M$ pada
saat $t$ adalah $\vect{OM}(t)$, yang berkoordinat
$\bigl(x(t),\, y(t)\bigr)$: dua fungsi waktu, satu untuk tiap sumbunya.
Kurva yang dilukis $M$ ketika $t$ berjalan adalah \omterm{def:g10:relative-motion:trajectory}{lintasannya}.
\end{definition}

\section{Vektor kecepatan}

\begin{definition}[Vektor kecepatan]\label{def:g12:kinematics-2d:velocity}
Sepanjang selang yang singkat, titiknya berpindah sebesar
$\vect{OM}(t+\Delta t) - \vect{OM}(t)$; bagilah dengan $\Delta t$ lalu
biarkan selangnya menyusut: inilah turunan pada pelajaran matematikamu,
yang diterapkan pada tiap koordinatnya. \emph{Vektor
kecepatan}\index{vektor kecepatan} $M$ adalah turunan \omterm{def:g12:kinematics-2d:position}{vektor posisinya},
$\vect v(t) = \bigl(x'(t),\, y'(t)\bigr)$, dalam \unit{m/s}. Normanya
$v = \sqrt{x'^2 + y'^2}$ disebut \emph{kelajuan}\index{kelajuan} ---
satu bilangan yang ditunjukkan alat pengukur laju.
\end{definition}

\begin{example}[Sebuah pesawat nirawak terbang mendatar]\label{ex:g12:kinematics-2d:drone}
Sebuah pesawat nirawak terbang dengan $x(t) = 4.0\,t$ dan
$y(t) = 3.0\,t$ (dalam \omterm{def:g10:orders-of-magnitude:unit}{meter} dan sekon): \omterm{def:g10:relative-motion:trajectory}{lintasannya} garis
$y = \tfrac34 x$, kecepatannya $\vect v = (4.0, 3.0)$ pada setiap saat,
dan lajunya $\sqrt{4.0^2 + 3.0^2} = \qty{5.0}{m/s}$, yang tetap.
\end{example}

\begin{proposition}[Menyinggung lintasannya]\label{prop:g12:kinematics-2d:tangent}
Pada setiap saat, $\vect v(t)$ menyinggung \omterm{def:g10:relative-motion:trajectory}{lintasannya} di kedudukan $M$
saat itu, dan menunjuk ke arah perjalanannya.
\end{proposition}

\begin{proof}
Perpindahannya merupakan tali busur \omterm{def:g10:relative-motion:trajectory}{lintasannya}; ketika $\Delta t$
menyusut, arah tali busurnya menjadi arah garis singgungnya.
\end{proof}

\section{Vektor percepatan}

\begin{definition}[Vektor percepatan]\label{def:g12:kinematics-2d:acceleration}
\emph{Vektor percepatan}\index{vektor percepatan} $M$ adalah turunan
\omterm{def:g12:kinematics-2d:velocity}{vektor kecepatannya},
$\vect a(t) = \bigl(v_x'(t),\, v_y'(t)\bigr) =
\bigl(x''(t),\, y''(t)\bigr)$, dalam \unit{m/s^2}: seberapa cepat, dan
ke arah mana, \omterm{def:g12:kinematics-2d:velocity}{vektor kecepatannya} sedang berubah.
\end{definition}

\begin{remark}[Berpercepatan tanpa bertambah cepat]\label{rem:g12:kinematics-2d:turning}
$\vect a \neq \vect 0$ setiap kali \emph{vektor} kecepatannya berubah
--- baik panjangnya \emph{maupun arahnya}: mobil yang mengerem
berpercepatan (ke belakang); mobil yang berbelok dengan laju tetap
berpercepatan (ke samping). Inilah besaran yang dikendalikan \omterm{def:g10:inertia:force}{gaya}
(\cref{ch:g11:forces-and-motion,ch:g12:newtons-laws}).
\end{remark}

\begin{method}[Membaca sebuah kronofotografi]\label{met:g12:kinematics-2d:chrono}
Sebuah \emph{kronofotografi}\index{kronofotografi} merekam kedudukan
$M_0, M_1, M_2, \dots$ sebuah titik yang bergerak pada selang waktu yang
sama $\Delta t$.
\begin{enumerate}
  \item Jarak antartitiknya membaca lajunya: jaraknya sama berarti
        lajunya tetap; melebar berarti bertambah cepat; merapat berarti
        melambat.
  \item Kecepatan di $M_i$ adalah tali busur setangkupnya,
        $\vect v_i \approx \vect{M_{i-1}M_{i+1}}/(2\,\Delta t)$, yang
        digambar di $M_i$ dan menyinggung \omterm{def:g10:relative-motion:trajectory}{lintasan} titik-titiknya.
  \item Untuk percepatan di $M_i$: salinlah $\vect v_{i-1}$ dan
        $\vect v_{i+1}$ ke $M_i$, lalu gambarlah
        $\Delta\vect v = \vect v_{i+1} - \vect v_{i-1}$ ujung ke ujung;
        maka $\vect a_i \approx \Delta\vect v/(2\,\Delta t)$.
\end{enumerate}
\end{method}

\section{Gerak lurus}

Sepanjang sebuah garis lurus, satu sumbu sudah memadai: $x(t)$,
$v(t) = x'(t)$, $a(t) = v'(t)$, yang semuanya bilangan bertanda.

\begin{definition}[Gerak lurus beraturan]\label{def:g12:kinematics-2d:uniform}
Sebuah gerak disebut \emph{lurus beraturan}\index{gerak lurus
beraturan} apabila $\vect v$ merupakan vektor yang tetap: \omterm{def:g10:relative-motion:trajectory}{lintasannya}
lurus, lajunya tetap, dan $\vect a = \vect 0$. Maka
$x(t) = v\,t + x_0$.
\end{definition}

\begin{proposition}[Gerak lurus berubah beraturan]\label{prop:g12:kinematics-2d:uarm}
\index{gerak lurus berubah beraturan}%
Jika sebuah titik bergerak sepanjang $Ox$ dengan percepatan $a$ yang
\emph{tetap}, dan bermula pada $t = 0$ dari kedudukan $x_0$ dengan
kecepatan $v_0$, maka
\[
v(t) = a\,t + v_0,
\qquad
x(t) = \tfrac12\,a\,t^2 + v_0\,t + x_0 .
\]
\end{proposition}

\begin{proof}
$v$ adalah antiturunan dari tetapan $a$: $v(t) = a t + C$, dan $t = 0$
memaksa $C = v_0$. Selanjutnya $x$ adalah antiturunan dari
$a t + v_0$: $x(t) = \tfrac12 a t^2 + v_0 t + C'$, dengan $C' = x_0$.
\end{proof}

\begin{example}[Jarak pengereman]\label{ex:g12:kinematics-2d:braking}
Sebuah mobil berlaju $v_0$ mengerem dengan perlambatan tetap $a$
(percepatannya $-a$). Mobil itu berhenti ketika
$v(t) = -a t + v_0 = 0$, yaitu pada $t_s = v_0/a$, setelah menempuh
$d = -\tfrac12 a t_s^2 + v_0 t_s =
v_0^2/(2a)$. Kuadratnya adalah tajuk
berita keselamatan jalan: menggandakan lajunya
\emph{melipatempatkan} jarak pengeremannya --- dengan
$a =
\qty{6.0}{m/s^2}$, \qty{16}{m} dari \qty{50}{km/h}, dan \qty{64}{m}
dari \qty{100}{km/h}.
\end{example}

\section{Gerak melingkar beraturan dan kerangka Frenet}

\begin{definition}[Gerak melingkar beraturan]\label{def:g12:kinematics-2d:ucm}
Sebuah gerak disebut \emph{melingkar beraturan}\index{gerak melingkar
beraturan} apabila \omterm{def:g10:relative-motion:trajectory}{lintasannya} berupa lingkaran berjari-jari $R$ dan
lajunya $v$ tetap. Lama satu putarannya disebut
\emph{periode}\index{periode!putaran} $T$, dan \emph{frekuensi}\index{frekuensi!putaran}
$f = 1/T$ (dalam \unit{Hz}) adalah banyaknya putaran tiap sekon:
$v = 2\pi R/T = 2\pi R f$.
\end{definition}

Lajunya tetap --- namun kecepatannya berputar: berarti \emph{ada}
percepatan. Ke arah mana, dan seberapa besar?

\begin{theorem}[Percepatan sentripetal]\label{thm:g12:kinematics-2d:centripetal}
\index{percepatan sentripetal}%
Pada \omterm{def:g12:kinematics-2d:ucm}{gerak melingkar beraturan} berjari-jari $R$ dan berlaju $v$,
percepatannya --- yang disebut \emph{sentripetal} --- menunjuk dari
titik yang bergerak menuju pusat lingkarannya, dengan norma yang tetap
\[
a = \frac{v^2}{R}.
\]
\end{theorem}

\begin{proof}
Pusatkanlah sumbunya di pusat lingkarannya lalu ambillah
$\omega = v/R$, sehingga busur $v t$ mengapit sudut $\omega t$:
kedudukannya $x(t) = R\cos(\omega t)$, $y(t) = R\sin(\omega t)$. Dengan
menurunkannya, $\vect v = (-R\omega\sin(\omega t),\, R\omega\cos(\omega t))$, yang normanya tetap $R\omega = v$; dengan
menurunkannya sekali lagi, $\vect a = -\omega^2\,\vect{OM}$: berlawanan
dengan \omterm{def:g12:kinematics-2d:position}{vektor posisinya} --- yaitu menuju pusatnya --- dengan norma
$\omega^2 R = v^2/R$.
\end{proof}

\begin{definition}[Kerangka Frenet]\label{def:g12:kinematics-2d:frenet}
\emph{Kerangka Frenet}\index{kerangka Frenet} pada kedudukan saat ini
sebuah titik pada \omterm{def:g10:relative-motion:trajectory}{lintasan} yang melengkung adalah pasangan vektor \omterm{def:g10:orders-of-magnitude:unit}{satuan}
yang saling tegak lurus $(\vect{u_t}, \vect{u_n})$: $\vect{u_t}$
menyinggung \omterm{def:g10:relative-motion:trajectory}{lintasannya}, searah geraknya; $\vect{u_n}$ normal, menuju
pusat lingkaran berjari-jari $R$ yang paling cocok dengan kurvanya di
sana.
\end{definition}

\begin{proposition}[Percepatan pada kerangka Frenet]\label{prop:g12:kinematics-2d:frenetacc}
Sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} bidang mana pun yang jari-jari setempatnya $R$ dan
ditempuh dengan laju $v(t)$,
\[
\vect a = a_t\,\vect{u_t} + a_n\,\vect{u_n},
\qquad a_t = \frac{\dd v}{\dd t},
\qquad a_n = \frac{v^2}{R}:
\]
\emph{percepatan tangensial}\index{percepatan tangensial} $a_t$
mengubah lajunya; \emph{percepatan normal}\index{percepatan normal}
$a_n$ memutar kecepatannya.
\end{proposition}
\admitted

\begin{remark}[Di mana buktinya bersemayam]\label{rem:g12:kinematics-2d:frenetproof}
Kita sudah membuktikan kedua ujungnya: $a_t$ murni pada sebuah garis
(\cref{prop:g12:kinematics-2d:uarm}), dan $a_n$ murni pada sebuah
lingkaran (\cref{thm:g12:kinematics-2d:centripetal}); penguraian yang
umum diturunkan pada jilid Tahun ke-1.
\end{remark}

\begin{example}[Mengerem memasuki tikungan]\label{ex:g12:kinematics-2d:bend}
Sebuah mobil memasuki tikungan berjari-jari $R = \qty{50}{m}$ pada
$v = \qty{15}{m/s}$ selagi mengerem dengan
$\dd v/\dd t = \qty{-3.0}{m/s^2}$: $a_t =
\qty{-3.0}{m/s^2}$,
$a_n = 15^2/50 = \qty{4.5}{m/s^2}$, sehingga
$a =
\sqrt{3.0^2 + 4.5^2} \approx \qty{5.4}{m/s^2}$, yang menunjuk ke
belakang \emph{sekaligus} ke dalam tikungannya. Bannya harus memasok
keduanya sekaligus --- dan itulah sebabnya pembalap mengerem
\emph{sebelum} tikungannya.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:1}
Untuk $x(t) = 3.0\,t$, $y(t) = 4.0\,t$ (dalam \omterm{def:g10:orders-of-magnitude:unit}{meter} dan sekon),
sebutkan $\vect v$, lajunya, \omterm{def:g10:relative-motion:trajectory}{lintasannya}, dan golongan geraknya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:2}
Pada sebuah \omterm{met:g12:kinematics-2d:chrono}{kronofotografi} yang diambil tiap \qty{0.10}{s}, titiknya
terletak pada satu garis, dengan jarak yang sama \qty{0.35}{m}. Kenali
geraknya lalu hitunglah lajunya. Apa yang akan ditunjukkan jarak yang
melebar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:3}
Untuk $x(t) = 2.0\,t^2$ sepanjang $Ox$ (dalam \omterm{def:g10:orders-of-magnitude:unit}{meter} dan sekon),
turunkanlah $v(t)$ dan $a(t)$, sebutkan golongan geraknya, lalu
hitunglah lajunya pada $t = \qty{3.0}{s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:4}
Sebuah kuda komidi putar menempuh lingkaran berjari-jari \qty{4.5}{m},
satu putaran tiap \qty{8.0}{s}: hitunglah \omterm{def:g12:kinematics-2d:ucm}{frekuensinya}, lajunya, dan
percepatannya (norma dan arahnya).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:5}
Sebuah grafik $v(t)$ menanjak lurus dari \qty{2.0}{m/s} pada $t = 0$ ke
\qty{10.0}{m/s} pada $t = \qty{4.0}{s}$. Bacalah percepatannya, lalu
jarak yang ditempuhnya (luas di bawah grafiknya).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:6}
Sebuah kereta cepat menempuh tikungan berjari-jari \qty{6.0}{km} pada
\qty{320}{km/h}. Hitunglah $a_n$ lalu bandingkan dengan $g$; mengapa
jalur kereta cepat harus menghindari tikungan yang tajam, sekalipun
tikungan itu dimiringkan dengan sempurna?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:7}
Sebuah troli pada rel yang lurus memiliki
$x(t) = -1.2\,t^2 + 8.0\,t + 3.0$ (dalam \omterm{def:g10:orders-of-magnitude:unit}{meter} dan sekon). Tentukan
$v(t)$ dan $a$, saat dan kedudukan berbaliknya, lalu lukiskan geraknya
sebelum dan sesudah saat itu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:8}
Sebuah \omterm{met:g12:kinematics-2d:chrono}{kronofotografi} sepanjang sebuah garis, tiap
$\Delta t = \qty{0.20}{s}$, memberikan $x = 0$, $0.10$, $0.28$,
$0.54$, $\qty{0.88}{m}$. Taksirlah kecepatannya pada ketiga titik di
dalamnya (\cref{met:g12:kinematics-2d:chrono}), lalu percepatannya;
simpulkan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:9}
Bulan mengorbit Bumi pada lingkaran yang hampir sempurna berjari-jari
\qty{3.84e8}{m} dalam \qty{27.3}{d}. Hitunglah lajunya dan \omterm{thm:g12:kinematics-2d:centripetal}{percepatan
sentripetalnya}; bandingkan percepatan itu dengan $g$, dengan mengetahui
bahwa Bulan berjarak sekitar $60$ jari-jari Bumi
(\cref{ch:g10:universal-gravitation}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:10}
Tabung mesin cuci berjari-jari \qty{25}{cm} berputar $1200$ putaran tiap
menit. Hitunglah $f$, $T$, laju tepinya, dan percepatan tepinya dalam
kelipatan $g$. Mengapa cuciannya tetap tertahan sementara airnya pergi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:11}
Sebuah mobil pada \qty{90}{km/h} mengerem dengan $a = \qty{7.5}{m/s^2}$
yang tetap: hitunglah waktu berhentinya dan jarak pengeremannya, lalu
sketsalah $v(t)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:12}
Untuk $x(t) = R\cos(\omega t)$, $y(t) = R\sin(\omega t)$: hitunglah
$\vect v$; periksalah bahwa lajunya tetap $R\omega$ dan
$\vect v \perp \vect{OM}$; hitunglah $\vect a$, tunjukkan bahwa nilainya
$-\omega^2\,\vect{OM}$, lalu perolehlah kembali $a = v^2/R$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:13}
Sebuah sepeda motor memasuki tikungan berjari-jari \qty{80}{m} pada
\qty{20}{m/s}, sambil mengerem \qty{2.5}{m/s^2}. Hitunglah $a_t$,
$a_n$, norma $\vect a$ dan sudutnya terhadap arah geraknya. Bannya
mencengkeram sampai \qty{7.0}{m/s^2} seluruhnya: apakah pengendaranya
masih di dalam batas itu?
\end{exercise}

\section{Soal: Laporan kecelakaan}

\begin{problem}[{Soal akhir pekan --- laporan kecelakaan: rekaman kamera
dasbor yang dibaca bingkai demi bingkai, hukum pengereman yang disusun
ulang dari bekas ban, sebuah bundaran yang bersaksi tentang cengkeraman,
dan sebuah putusan dalam kilometer per jam}]
\label{pb:g12:kinematics-2d:1}
Sebuah mobil menabrak van yang keluar dari jalan samping; tidak ada yang
terluka, tetapi perusahaan asuransinya berselisih. Penyidiknya memiliki
rekaman kamera dari atas (satu bingkai tiap \qty{0.50}{s}), bekas ban
yang terukur, dan kamera dasbor mobilnya. Batas kecepatannya
\qty{50}{km/h}.

\medskip
\noindent\textbf{Bagian I --- Membaca rekamannya.}
Bingkai sebelum ada tanggapan apa pun memberikan, sepanjang jalan yang
lurus:
\begin{center}\small
\begin{tabular}{lccccc}
$t$ (\unit{s}) & 0.00 & 0.50 & 1.00 & 1.50 & 2.00\\
$x$ (\unit{m}) & 0.0 & 9.0 & 18.0 & 27.0 & 36.0
\end{tabular}
\end{center}
\begin{enumerate}
  \item Mengapa laporannya harus lebih dahulu menyebutkan kerangka dan
        sumbunya? Yang mana yang dipakai di sini?
  \item Hitunglah kecepatan rata-rata pada masing-masing keempat
        selangnya.
  \item Golongan gerak apa ini? Berikan alasannya dari tabelnya.
  \item Sebutkan laju mobilnya $v_0$ dalam \unit{m/s} dan \unit{km/h}.
  \item Berapa percepatannya selama dua sekon itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Bekas bannya.}
Uji ban pada aspal kering ini memberikan perlambatan pengereman tetap
$a = \qty{6.0}{m/s^2}$; bekas bannya sepanjang $d_b = \qty{27}{m}$.
Ambillah $t = 0$, $x = 0$ pada saat pengeremannya bermula, dengan laju
$v_B$ yang belum diketahui.
\begin{enumerate}[resume]
  \item Dengan antiturunan, tulislah $v(t)$ dan $x(t)$ selama
        pengeremannya.
  \item Nyatakan waktu berhentinya $t_s$; tunjukkan bahwa
        $d_b = v_B^2/(2a)$.
  \item Simpulkan $v_B$ dari bekas bannya. Sesuaikah dengan Bagian~I?
  \item Waktu tanggapnya \qty{1.0}{s}: hitunglah jarak tanggapnya dan
        jarak berhentinya seluruhnya.
  \item Jelaskan, dari pertanyaan 7, mengapa menggandakan lajunya
        melipatempatkan jarak pengeremannya; hitunglah $d_b$ pada tepat
        \qty{50}{km/h}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Bundarannya bersaksi.}
Sebelumnya, kamera dasbornya menunjukkan mobil itu menempuh lajur
bundaran berjari-jari $R = \qty{12.5}{m}$ dengan laju tetap, seperempat
putaran dalam \qty{2.5}{s}. Jalannya dapat memasok paling banyak
$a_n = \mu g$, dengan $\mu = 0.70$.
\begin{enumerate}[resume]
  \item Lajunya tetap di sana --- apakah kecepatannya juga? Apakah
        mobilnya berpercepatan? Jelaskan.
  \item Hitunglah laju mobilnya di bundaran itu.
  \item Hitunglah \omterm{thm:g12:kinematics-2d:centripetal}{percepatan sentripetalnya} (norma dan arahnya).
  \item Hitunglah laju terbesar tanpa tergelincir. Apakah mobilnya masih
        di dalam batas itu?
  \item Sebutkan \omterm{def:g12:kinematics-2d:ucm}{periode} dan \omterm{def:g12:kinematics-2d:ucm}{frekuensi} satu putaran penuh pada laju itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Putusannya.}
Vannya muncul $D = \qty{41}{m}$ di depan mobilnya pada saat pengemudinya
mulai menanggapi.
\begin{enumerate}[resume]
  \item Sebanyak apa, dalam \unit{km/h} dan dalam persen, laju sebelum
        pengeremannya melampaui batasnya?
  \item Hitunglah laju mobilnya pada saat menabrak.
  \item Ulangi perhitungan berhentinya pada tepat \qty{50}{km/h} (dengan
        tanggapan dan pengereman yang sama): apakah mobilnya berhenti,
        dan dengan sisa jarak berapa?
  \item Berikan rumus umum jarak berhentinya $d(v)$; mengapa lajunya
        masuk dua kali --- sekali secara linear dan sekali secara
        kuadrat?
  \item Putusannya, satu kalimat beserta angkanya: laju berapa yang
        ditempuh pengemudinya, dan apakah kecelakaannya akan terjadi
        pada batas kecepatannya?
\end{enumerate}
\end{problem}
