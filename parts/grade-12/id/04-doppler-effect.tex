\chapter{Efek Doppler}\label{ch:g12:doppler-effect}

Sebuah ambulans mendekat dengan sirene meraung, lalu melintas --- dan pada saat itu juga
nadanya meluncur turun secara kasatindra, padahal pengemudinya tidak menyentuh apa pun.
Sirenenya jujur; gelombangnyalah yang tidak: geraknya memerasnya di depan dan
meregangkannya di belakang. Bab ini mengukur perasan itu, lalu menunggangi gagasan yang
sama dari sirene sampai pistol \omterm{met:g10:signals-and-waves:echo}{radar}, sampai darah di dalam pembuluh nadi, dan sampai ke
sebuah planet yang dibocorkan oleh goyangan pada cahaya bintang.

\section{Efeknya}

\begin{definition}[Efek Doppler]\label{def:g12:doppler-effect:doppler}
\emph{Efek Doppler}\index{efek Doppler} adalah perubahan \omterm{def:g10:signals-and-waves:frequency}{frekuensi} terima sebuah \omterm{def:g10:signals-and-waves:wave}{gelombang}
ketika sumber dan penerimanya bergerak relatif satu sama lain: \omterm{def:g10:signals-and-waves:frequency}{frekuensi} terima $f'$ lebih
besar daripada \omterm{def:g10:signals-and-waves:frequency}{frekuensi} pancar $f$ selama keduanya saling mendekat, lebih kecil selama
keduanya saling menjauh, dan sama dengan $f$ hanya selama jaraknya sejenak berhenti
berubah.
\end{definition}

\begin{remark}[Tinggi nada, bukan kenyaringan]\label{rem:g12:doppler-effect:pitch}
Ada dua hal yang berubah ketika ambulansnya melintas. \emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{Kenyaringan}}nya membesar lalu
memudar karena \omterm{def:g10:signals-and-waves:wave}{gelombangnya} menyebar bersama jaraknya (\cref{ch:g12:sound-acoustics});
\emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nada}}nya bergeser karena laju mendekat atau menjauhnya, dan turun mendadak
pada saat melintas. \omterm{def:g12:sound-acoustics:pitch-loudness}{Kenyaringan} berkata ``seberapa jauh''; \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nada} berkata ``seberapa
cepat''.
\end{remark}

\section{Sumber yang bergerak: muka gelombang yang berdesakan}

Gambaran kuncinya: begitu sebuah puncak meninggalkan sumbernya, mediumlah yang memilikinya.
Tiap puncaknya melebar sebagai lingkaran dengan laju \omterm{def:g10:signals-and-waves:wave}{gelombang} $v$
(\cref{ch:g12:mechanical-waves}), berpusat di titik tempat puncak itu \emph{dipancarkan}
--- mediumnya tidak tahu dan tidak peduli bahwa sumbernya sudah berpindah.

\begin{proposition}[Sumber yang bergerak]\label{prop:g12:doppler-effect:moving-source}
Sebuah sumber memancarkan puncak dengan \omterm{def:g10:signals-and-waves:period}{periode} $T$ (\omterm{def:g10:signals-and-waves:frequency}{frekuensinya} $f = 1/T$) selagi
bergerak pada garis lurus dengan laju $v_s < v$ menembus medium yang \omterm{def:g10:signals-and-waves:wave}{gelombangnya}
merambat dengan laju $v$. Seorang pengamat yang diam di dalam mediumnya menerima
\[
\lambda' = (v \mp v_s)\,T, \qquad f' = f\,\frac{v}{v \mp v_s},
\]
dengan tanda atas untuk sumber yang mendekat dan tanda bawah untuk sumber yang menjauh.
\end{proposition}

\begin{proof}
Di antara dua puncak, puncak yang pertama maju $vT$ sementara sumbernya maju $v_s T$. Di
depan, puncak yang kedua karena itu dipancarkan $v_s T$ \emph{lebih dekat} ke yang
pertama, sehingga $\lambda' = (v - v_s)T$; di belakang, $v_s T$ yang sama justru
ditambahkan. Puncaknya lalu menyapu melewati pengamat yang diam dengan laju mediumnya $v$,
satu tiap $\lambda'/v$ sekon, sehingga $f' = v/\lambda' = f v/(v \mp v_s)$.
\end{proof}

\begin{example}[Ambulansnya, dalam angka]\label{ex:g12:doppler-effect:siren}
Sebuah sirene pada $f = \qty{700}{Hz}$ mendekat dengan $v_s = \qty{25}{m/s}$; ambillah
$v = \qty{340}{m/s}$ di udara (\cref{ch:g12:sound-acoustics}). Ketika mendekat:
$f' = 700 \times 340/315 \approx \qty{756}{Hz}$; ketika menjauh:
$f' = 700 \times 340/365 \approx \qty{652}{Hz}$. \omterm{def:g10:relative-motion:trajectory}{Lintasannya} menurunkan nadanya sebesar
\qty{104}{Hz} --- nisbahnya $\approx 1.16$, sekitar dua setengah seminada, dan tidak
mungkin salah dengar oleh telinga mana pun.
\end{example}

\begin{remark}[Apa yang tidak diubah geraknya]\label{rem:g12:doppler-effect:speed-unchanged}
\omterm{def:g10:signals-and-waves:wave}{Gelombangnya} tetap merambat dengan $v$: mediumnyalah yang menetapkan lajunya
(\cref{ch:g12:mechanical-waves}). Gerak sumbernya memeras \emph{\omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang}}, tidak
pernah memeras laju. Dan perasannya memiliki batas: ketika $v_s \to v$, puncak di depannya
bertumpuk satu sama lain dan $f' \to \infty$ --- yaitu kemacetan \omterm{def:g12:mechanical-waves:wavefront}{muka gelombang} yang
diseret pesawat supersonik sebagai \omterm{def:g10:signals-and-waves:wave}{gelombang} kejut.
\end{remark}

\section{Pengamat yang bergerak, dan jalan pintas gerak lambat}

\begin{proposition}[Pengamat yang bergerak]\label{prop:g12:doppler-effect:moving-observer}
Seorang pengamat bergerak dengan laju $v_o$ lurus menuju (atau menjauhi) sebuah sumber
yang diam di dalam mediumnya. \omterm{def:g10:signals-and-waves:frequency}{Frekuensi} terimanya adalah
\[ f' = f\left(1 \pm \frac{v_o}{v}\right) \quad (+\ \text{mendekat},\ -\ \text{menjauh}). \]
\end{proposition}

\begin{proof}
Pola \omterm{def:g10:signals-and-waves:wave}{gelombangnya} sendiri tidak terganggu: puncak yang berjarak $\lambda = vT$ merambat
dengan $v$. Pengamat yang berlari ke arahnya menemui puncaknya dengan laju relatif
$v + v_o$, satu tiap $\lambda/(v + v_o)$ sekon: $f' = (v + v_o)/\lambda = f(1 + v_o/v)$.
Ketika berlari menjauh, lajunya $v - v_o$.
\end{proof}

\begin{definition}[Kecepatan radial]\label{def:g12:doppler-effect:radial}
\emph{Kecepatan radial}\index{kecepatan radial} $v_r$ sebuah sumber terhadap seorang
pengamat adalah komponen kecepatan relatifnya sepanjang garis penghubung keduanya ---
yaitu laju menyusut atau tumbuhnya jarak itu. Hanya komponen inilah yang menggeser
\omterm{def:g10:signals-and-waves:frequency}{frekuensinya}; gerak \emph{melintang} terhadap garis pandangnya tidak memberikan pergeseran
pada orde ini.
\end{definition}

\begin{proposition}[Gerak lambat: satu rumus untuk semuanya]\label{prop:g12:doppler-effect:slow}
Ketika laju radialnya kecil, $v_r \ll v$, tidak lagi berarti siapa yang bergerak:
\[
\frac{\Delta f}{f} = \frac{f' - f}{f} \approx +\frac{v_r}{v} \ \text{(mendekat)},
\qquad
\frac{\Delta f}{f} \approx -\frac{v_r}{v} \ \text{(menjauh)}.
\]
\end{proposition}

\begin{proof}
Rumus pengamatnya sudah tepat berbentuk $1 \pm v_o/v$. Untuk sumbernya, tulislah
$x = v_s/v$: kesamaan $\frac{1}{1 - x} = 1 + x + \frac{x^2}{1 - x}$ menunjukkan bahwa
$f'/f$ berbeda dari $1 + x$ oleh suku berorde $x^2$ --- untuk $x = 0.1$, koreksi $1\%$ pada
pergeseran $10\%$. Sampai orde pertama, keempat rumusnya melebur menjadi $1 \pm v_r/v$.
\end{proof}

\begin{method}[Membaca sebuah lintasan]\label{met:g12:doppler-effect:pass}
Rekaman sebuah sumber yang melintas menampilkan dua dataran: $f_1$ (saat mendekat) dan
$f_2$ (saat menjauh). Maka
\[
f = \frac{2 f_1 f_2}{f_1 + f_2}, \qquad
v_s = v\,\frac{f_1 - f_2}{f_1 + f_2}.
\]
Memang $f_1 + f_2 = \frac{2fv^2}{v^2 - v_s^2}$ dan
$f_1 - f_2 = \frac{2fv\,v_s}{v^2 - v_s^2}$: bagilah untuk memperoleh $v_s$, lalu hitunglah
$2f_1 f_2/(f_1 + f_2)$ untuk memperoleh $f$. Dua bacaan \omterm{def:g10:signals-and-waves:frequency}{frekuensi} menghasilkan \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nada}
yang sebenarnya sekaligus lajunya --- tanpa pencatat waktu, tanpa penggaris.
\end{method}

\section{Gema yang mengukur laju}

Pantulkanlah sebuah \omterm{def:g10:signals-and-waves:wave}{gelombang} dari sasaran yang bergerak dan \omterm{def:g12:doppler-effect:doppler}{efek Doppler} menyerang dua kali.

\begin{proposition}[Pemantulan menggandakan pergeserannya]\label{prop:g12:doppler-effect:double}
Sebuah \omterm{def:g10:signals-and-waves:wave}{gelombang} berfrekuensi $f$ dikirimkan ke pemantul yang mendekat lurus dengan laju
$u \ll v$. \omterm{rem:g10:signals-and-waves:honest}{Gemanya} kembali dengan
\[ f' = f\,\frac{v + u}{v - u}, \qquad \text{sehingga} \qquad \Delta f \approx \frac{2u}{v}\,f . \]
\end{proposition}

\begin{proof}
Ada dua pergeseran yang berturutan. Sebagai \emph{pengamat} yang bergerak, pemantulnya
menerima $f(1 + u/v)$; ketika memancarkan kembali apa yang diterimanya, pemantul itu kini
menjadi \emph{sumber} yang bergerak, sehingga \omterm{rem:g10:signals-and-waves:honest}{gemanya} tiba pada
$f' = f(1 + u/v)/(1 - u/v) \approx f(1 + 2u/v)$ untuk $u \ll v$.
\end{proof}

\begin{example}[Pistol radar]\label{ex:g12:doppler-effect:radar}
Sebuah \omterm{met:g10:signals-and-waves:echo}{radar} lalu lintas memancarkan \omterm{def:g10:signals-and-waves:wave}{gelombang} mikro pada $f = \qty{24.15}{GHz}$
($v = c = \qty{3.00e8}{m/s}$) lalu menumpangkan \omterm{rem:g10:signals-and-waves:honest}{gemanya} pada \omterm{def:g10:signals-and-waves:wave}{gelombang} yang keluar;
layangannya (\cref{ch:g12:sound-acoustics}) langsung memperdengarkan
$\Delta f = 2uf/c \approx \qty{161}{Hz}$ tiap \unit{m/s} laju mobilnya --- sebuah \omterm{def:g10:signals-and-waves:frequency}{frekuensi}
audio yang sepele diukur, yang dipahat dari pembawa \qty{24}{GHz}. Faktor $2$-nya tidak
boleh dilewatkan: melupakannya memuji setiap pengemudi dua kali lipat.
\end{example}

\begin{example}[Ultrasonografi Doppler]\label{ex:g12:doppler-effect:ultrasound}
Sebuah penduga kedokteran mengirimkan \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik} $f = \qty{5.0}{MHz}$ ke dalam pembuluh
nadi ($v \approx \qty{1540}{m/s}$ di dalam jaringan); sel darah merahnya memantulkannya.
Darah pada $u = \qty{0.50}{m/s}$ mengembalikan
$\Delta f = 2 \times 0.50 \times \num{5.0e6}/1540 \approx
\qty{3.2}{kHz}$ --- sebuah
layangan yang \emph{terdengar}: dokter jantungnya benar-benar mendengar darahnya
bertambah cepat pada tiap detak jantung, lalu mengukur $u$ dari $\Delta f$.
\end{example}

\section{Doppler pada cahaya bintang}

Cahaya adalah \omterm{def:g10:signals-and-waves:wave}{gelombang} (\cref{ch:g12:light-as-wave}), yang \omterm{def:g12:sound-acoustics:timbre}{spektrumnya} bergaris tajam pada
\omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} yang ditetapkan tiap unsur kimianya (\cref{ch:g10:light-spectra}) ---
sebuah penggaris tertera laboratorium yang tercetak pada setiap bintang.

\begin{definition}[Pergeseran merah dan pergeseran biru]\label{def:g12:doppler-effect:redshift}
Ketika \omterm{def:g10:light-spectra:emission}{garis spektrum} sebuah sumber muncul pada \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} yang \emph{lebih panjang}
daripada di laboratorium, \omterm{def:g12:sound-acoustics:timbre}{spektrumnya} dikatakan \emph{bergeser
merah}\index{pergeseran merah}; ketika garisnya muncul \emph{lebih pendek}, \omterm{def:g12:sound-acoustics:timbre}{spektrumnya}
\emph{bergeser biru}\index{pergeseran biru} --- yaitu bergeser ke ujung merah atau ke ujung
biru \omterm{def:g10:light-spectra:wavelength}{spektrum tampaknya}.
\end{definition}

\begin{proposition}[Pergeseran Doppler cahaya]\label{prop:g12:doppler-effect:light}
Untuk sumber yang menjauh dengan laju radial $v_r \ll c$, setiap \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombangnya}
teramati teregang sebesar
\[ \frac{\Delta \lambda}{\lambda} = \frac{\lambda' - \lambda}{\lambda} \approx \frac{v_r}{c} \quad \text{(pergeseran merah)}, \]
dan termampatkan sebanyak itu pula, $\Delta\lambda/\lambda \approx -v_r/c$, untuk sumber
yang mendekat (\omterm{def:g12:doppler-effect:redshift}{pergeseran biru}).
\end{proposition}
\admitted

\begin{remark}[Di mana rumusnya yang jujur bersemayam]\label{rem:g12:doppler-effect:honest}
Cahaya tidak memerlukan medium, sehingga penurunan untuk \omterm{def:g10:signals-and-waves:sound}{bunyi} tidak dapat disalin ---
tidak ada ``diam di dalam mediumnya''. Rumus yang tepat datang dari kerelatifan khusus,
nanti pada tahun ini juga (\cref{ch:g12:special-relativity}); koreksinya berorde
$(v_r/c)^2$, yang dapat diabaikan untuk pesawat, bintang dan galaksi yang dekat. Di bawah
ini kita memakai rumus gerak lambatnya dengan hati yang tenang.
\end{remark}

\begin{example}[Menakar larinya sebuah bintang]\label{ex:g12:doppler-effect:vega}
Pada \omterm{def:g12:sound-acoustics:timbre}{spektrum} bintang terang Vega, garis hidrogen yang \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} laboratoriumnya
\qty{656.30}{nm} terukur pada \qty{656.27}{nm}: $\Delta\lambda = \qty{-0.03}{nm}$, yaitu
\omterm{def:g12:doppler-effect:redshift}{pergeseran biru}, sehingga Vega mendekat dengan
$v_r = c\,|\Delta\lambda|/\lambda \approx \qty{1.4e4}{m/s}$ --- \qty{14}{km/s}, yang
terbaca dari pergeseran satu bagian dari dua puluh ribu.
\end{example}

\begin{example}[Bintang yang bergoyang: bintang ganda dan planet luar surya]\label{ex:g12:doppler-effect:wobble}
Ketika dua bintang saling mengorbit, garisnya berayun secara \omterm{def:g10:signals-and-waves:period}{periodik} antara biru dan merah
--- itulah \emph{bintang ganda spektroskopik}\index{bintang ganda spektroskopik}, yang
\omterm{def:g10:universal-gravitation:satellite}{orbitnya} terukur tanpa pernah memisahkan pasangannya. Sebuah planet melakukan hal yang
sama pada bintangnya, secara samar: keduanya mengitari \omterm{def:g11:forces-and-motion:com}{pusat massa} bersamanya, sehingga
\omterm{def:g12:doppler-effect:radial}{kecepatan radial} bintangnya berayun sebesar beberapa puluh \unit{m/s} --- yaitu pergeseran
$\Delta\lambda/\lambda \sim 10^{-7}$. Mendeteksinya adalah cara ditemukannya planet pertama
di sekeliling bintang serupa Matahari; soal akhir pekannya mengulang penemuan itu, lengkap
dengan angkanya.
\end{example}

\begin{remark}[Pergeseran merah galaksi]\label{rem:g12:doppler-effect:expansion}
\omterm{def:g12:sound-acoustics:timbre}{Spektrum} galaksi yang jauh semuanya bergeser \emph{merah}, dan makin jauh galaksinya, makin
besar pergeserannya: jarak di alam semesta sedang teregang. Untuk galaksi yang dekat,
$v_r = c\,\Delta\lambda/\lambda$ membaca laju menjauhnya; untuk yang terjauh,
pergeserannya melampaui apa yang sanggup ditangani rumus gerak lambat secara jujur, dan
pembukuannya diangkat kembali mulai dari \cref{ch:g12:special-relativity} dan pada jilid
perguruan tinggi.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:1}
Sirene mobil pemadam kebakaran memancar pada \qty{700}{Hz} dan melaju \qty{25}{m/s}
($v = \qty{340}{m/s}$). Hitunglah \omterm{def:g10:signals-and-waves:frequency}{frekuensi} yang didengar pejalan kaki (a) di depannya;
(b) di belakangnya. (c) Apa yang didengar pengemudinya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:2}
Seorang teman mengaku: ``nadanya turun ketika ambulansnya pergi karena \omterm{def:g10:signals-and-waves:sound}{bunyinya} melemah
bersama jaraknya.'' Uraikanlah kedua efek yang dicampuradukkannya, lalu nyatakan
masing-masing bergantung pada apa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:3}
Seorang pengendara sepeda melaju \qty{6.0}{m/s} lurus menuju sirene diam yang memancar pada
\qty{700}{Hz}. \omterm{def:g10:signals-and-waves:frequency}{Frekuensi} berapa yang didengarnya? Dan ketika mengayuh lurus menjauh? Rumus
yang mana yang berlaku, dan mengapa bukan rumus sumber yang bergerak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:4}
Klakson kereta memancar pada \qty{400}{Hz} selagi keretanya melaju \qty{30}{m/s}.
Hitunglah \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombangnya} di depan keretanya, di belakangnya, dan ketika diam.
Pengamat yang mana yang menerima yang mana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:5}
Sebuah telepon yang memainkan nada \qty{440}{Hz} dibawa seorang pelari dengan
\qty{5.0}{m/s}. Pakailah rumus gerak lambat untuk menaksir pergeseran yang didengar orang
yang didekati pelari itu. Bandingkan dengan satu seminada (sekitar $6\%$ pada
\omterm{def:g10:signals-and-waves:frequency}{frekuensinya}): apakah seorang pemusik akan menyadarinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:6}
Seekor kelelawar terbang \qty{6.0}{m/s} lurus ke sebuah dinding sambil memancar pada
\qty{50.0}{kHz}. Tunjukkan bahwa \omterm{rem:g10:signals-and-waves:honest}{gema} yang didengarnya kembali pada
$f' = f(v + v_b)/(v - v_b)$, lalu hitunglah $f'$ dan pergeserannya. Mengapa ini rumus
pistol \omterm{met:g10:signals-and-waves:echo}{radar} yang menyamar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:7}
Sebuah pistol \omterm{met:g10:signals-and-waves:echo}{radar} pada \qty{24.15}{GHz} mengukur layangan \qty{4.0}{kHz} dari mobil yang
mendekat. Tentukan laju mobilnya dalam \unit{m/s} dan \unit{km/h}. Layangan berapa yang
akan diberikan mobil yang lajunya tepat \qty{50}{km/h}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:8}
Sebuah penduga \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonik} Doppler \qty{5.0}{MHz} yang dibidikkan sepanjang pembuluh nadi
($v = \qty{1540}{m/s}$ di dalam jaringan) mendengar layangan \qty{2.6}{kHz}. Tentukan laju
darahnya. Mengapa penduganya harus dimiringkan \emph{sepanjang} alirannya dan bukan tegak
lurus terhadapnya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:9}
Ambillah $f = \qty{1000}{Hz}$, $v = \qty{340}{m/s}$ dan laju pendekatan \qty{34}{m/s}.
Hitunglah $f'$ secara tepat ketika (a) sumbernya yang bergerak, (b) pengamatnya yang
bergerak. Bandingkan keduanya dengan ramalan gerak lambat $f(1 + v_r/v)$ lalu jelaskan
besarnya ketidaksesuaiannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:10}
Rekaman sebuah skuter yang melintasi mikrofon menampilkan dataran pada \qty{465}{Hz} dan
\qty{415}{Hz}. Dengan memakai \cref{met:g12:doppler-effect:pass}, tentukan \omterm{def:g10:signals-and-waves:frequency}{frekuensi}
klakson yang sebenarnya dan laju skuternya dalam \unit{km/h}. Mengapa \omterm{def:g10:signals-and-waves:frequency}{frekuensi} yang
sebenarnya \emph{bukan} rata-rata keduanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:11}
Pada \omterm{def:g12:sound-acoustics:timbre}{spektrum} sebuah bintang, garis hidrogen yang \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} laboratoriumnya
\qty{656.3}{nm} teramati pada \qty{656.5}{nm}. \omterm{def:g12:doppler-effect:redshift}{Pergeseran merah} atau biru? Hitunglah
\omterm{def:g12:doppler-effect:radial}{kecepatan radial} bintang itu beserta arahnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:doppler-effect:12}
Sebuah bintang bergoyang dengan \qty{55}{m/s} karena sebuah planet yang tidak terlihat.
Hitunglah \omterm{def:g10:signals-and-waves:amplitude}{amplitudo} ayunan \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang} garis \qty{500}{nm}, sebagai sebuah panjang
dan sebagai bagian dari $\lambda$. Sebagai \omterm{def:g10:orders-of-magnitude:oom}{orde besar}: mengapa perburuan planet harus
menunggu spektrograf yang mantap sampai satu bagian dari $10^{7}$?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:doppler-effect:13}
Setiap garis pada \omterm{def:g12:sound-acoustics:timbre}{spektrum} sebuah galaksi ditemukan teregang sebesar $2.4\%$. Hitunglah
laju menjauhnya. Para astronom menemukan bahwa laju semacam itu tumbuh sebanding dengan
jaraknya, untuk setiap arah langitnya: gambaran apa yang disarankan hal ini? Apakah rumus
gerak lambatnya masih dapat dipercaya di sini?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:doppler-effect:14}
Matahari berputar: satu tepi cakramnya mendekati kita dengan sekitar \qty{2.0}{km/s}
sementara tepi yang lain menjauh sama cepatnya. Apa yang dilakukan hal itu terhadap garis
\qty{656.3}{nm} pada cahaya dari seluruh cakramnya --- menggesernya atau melebarkannya?
Hitunglah pengaruhnya dalam \unit{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:doppler-effect:15}
Sebuah mobil polisi dengan \qty{40}{m/s} dan sirene \qty{700}{Hz} mengejar truk yang melaju
\qty{30}{m/s} pada arah yang sama. (a) Benarkan $f' = f(v - v_o)/(v - v_s)$ bagi pengemudi
truknya. (b) Hitunglah $f'$. (c) Tunjukkan bahwa jika truknya menyamai laju mobil
polisinya, pergeserannya akan lenyap, lalu jelaskan mengapa memang seharusnya demikian.
\end{exercise}

\section{Soal: Perburuan sebuah planet luar surya}

\begin{problem}[{Soal akhir pekan --- perburuan sebuah planet luar surya: sebuah klakson
kereta menera telinganya, sebuah pistol radar melatih pergeseran gandanya, dan pada Minggu
malam spektrum sebuah bintang, yang bergoyang sebesar satu bagian dari lima juta, sudah
menimbang sebuah planet yang belum pernah dilihat siapa pun}]
\label{pb:g12:doppler-effect:1}
Sebuah klub astronomi menghabiskan akhir pekannya pada satu gagasan --- pergeseran Doppler
--- yang dilatih di Bumi, lalu dibidikkan ke bintang serupa Matahari. Datanya:
$v = \qty{340}{m/s}$ untuk \omterm{def:g10:signals-and-waves:sound}{bunyi} di udara, $c = \qty{3.00e8}{m/s}$,
$G = \qty{6.67e-11}{N.m^2/kg^2}$; bintangnya bermassa $M = \qty{2.0e30}{kg}$. Diterima,
dari mekanika \omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran (\cref{ch:g12:satellites-kepler}): sebuah planet bermassa $m$
pada \omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran berjari-jari $r$ memiliki \omterm{def:g10:signals-and-waves:period}{periode} dan laju yang diberikan oleh
$r^3 = \frac{G M P^2}{4\pi^2}$ dan $V = \frac{2\pi r}{P}$, dan bintangnya, yang mengitari
\omterm{def:g11:forces-and-motion:com}{pusat massa} bersamanya, bergoyang dengan laju $K = \frac{m}{M}\,V$.

\medskip
\noindent\textbf{Bagian I --- Jumat: perlintasan kereta.}
Sebuah kereta melintas dengan laju tetap sambil membunyikan klaksonnya; sebuah telepon
merekam $f_1 = \qty{471}{Hz}$ selagi kereta itu mendekat, dan $f_2 = \qty{411}{Hz}$
sesudah kereta itu melintas.
\begin{enumerate}
  \item Jelaskan, dengan gambaran \omterm{def:g12:mechanical-waves:wavefront}{muka gelombang} yang berdesakan, mengapa $f_1 > f_2$
        padahal klaksonnya tidak pernah berubah.
  \item Tulislah kedua persamaan yang mengaitkan $f_1$, $f_2$ dengan \omterm{def:g10:signals-and-waves:frequency}{frekuensi} yang
        sebenarnya $f$ dan laju keretanya $u$.
  \item Tunjukkan bahwa $u = v\,\dfrac{f_1 - f_2}{f_1 + f_2}$ lalu hitunglah nilainya dalam
        \unit{m/s} dan \unit{km/h}.
  \item Tunjukkan bahwa $f = \dfrac{2 f_1 f_2}{f_1 + f_2}$ lalu hitunglah nilainya. Mengapa
        $f$ sedikit \emph{di bawah} rata-rata $f_1$ dan $f_2$?
  \item Hitunglah \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombangnya} di depan dan di belakang keretanya, serta \omterm{def:g12:mechanical-waves:wavelength}{panjang
        gelombang} ketika diam.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Sabtu: pistol \omterm{met:g10:signals-and-waves:echo}{radar}.}
Sebuah patroli lalu lintas memperagakan \omterm{met:g10:signals-and-waves:echo}{radarnya}: $f = \qty{24.125}{GHz}$, dan pistolnya
membaca layangan antara \omterm{def:g10:signals-and-waves:wave}{gelombang} yang keluar dan \omterm{rem:g10:signals-and-waves:honest}{gemanya}.
\begin{enumerate}[resume]
  \item Jelaskan mengapa \omterm{rem:g10:signals-and-waves:honest}{gema} sebuah mobil yang bergerak tergeser \emph{dua kali} ---
        sebutkan peran yang dimainkan mobilnya pada tiap pergeserannya.
  \item Tunjukkan bahwa untuk mobil berlaju $u \ll c$ layangannya
        $\Delta f \approx \dfrac{2u}{c}\,f$.
  \item Sebuah mobil mengembalikan $\Delta f = \qty{3.55}{kHz}$: hitunglah $u$.
  \item Ubahlah ke \unit{km/h} lalu bandingkan dengan batas \qty{80}{km/h} yang terpasang
        di sana.
  \item Berapa \omterm{def:g10:signals-and-waves:frequency}{hertz} layangan yang sepadan dengan \qty{1}{km/h}? Berikan ulasanmu: apa yang
        membuat pergeseran relatif sekecil itu pada \omterm{def:g10:signals-and-waves:wave}{gelombang} \qty{24}{GHz} mudah diukur?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Minggu: bintang yang bergoyang.}
Klub itu mengunduh \omterm{def:g12:doppler-effect:radial}{kecepatan radial} terukur sebuah bintang serupa Matahari, malam demi
malam, yang tergambar di bawah ini.

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis, width=8.8cm, height=4.6cm,
    xlabel={$t$ (hari)}, ylabel={$v_r$ (\unit{m/s})},
    xmin=0, xmax=8.4, ymin=-72, ymax=72,
    xtick={0,2.1,4.2,6.3,8.4},
    xticklabels={$0$,$2.1$,$4.2$,$6.3$,$8.4$}, ytick={-55,0,55}]

  \addplot[dashed, omExo, domain=0:8.4, samples=2] {-55};

\end{tikzpicture}

{\small \omterm{def:g12:doppler-effect:radial}{Kecepatan radial} bintang itu sepanjang dua belas malam: sinusoid yang bersih ---
tanda tangan sebuah benda pada \omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran yang dilihat dari sisinya.}
\end{omfigure}

\begin{enumerate}[resume]
  \item Apa yang sebenarnya diukur, malam demi malam, pada \emph{\omterm{def:g12:sound-acoustics:timbre}{spektrum}} bintangnya?
        Nyatakan hubungan yang dipakai untuk mengubahnya menjadi $v_r$.
  \item Bacalah \omterm{def:g10:signals-and-waves:amplitude}{amplitudo} $K$ dan \omterm{def:g10:signals-and-waves:period}{periode} $P$ goyangannya dari kurvanya.
  \item Hitunglah ayunan $\Delta\lambda$ yang sepadan untuk garis \qty{550}{nm}, dan
        $\Delta\lambda/\lambda$. Bandingkan dengan Bagian I: seberapa jauh lebih sulit
        pengukuran ini daripada pengukuran keretanya?
  \item Mengapa sebuah planet yang tidak terlihat sanggup membuat bintangnya bergoyang sama
        sekali? Apa yang ditunjukkan bentuk sinusoidnya tentang \omterm{def:g10:universal-gravitation:satellite}{orbitnya}?
  \item Mengapa cara ini hanya mengukur bagian \emph{radial} gerak bintangnya, dan apa yang
        akan terlihat jika \omterm{def:g10:universal-gravitation:satellite}{orbitnya} menghadap tegak lurus kepada kita?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Minggu malam: menimbang yang tak terlihat.}
Ambillah \omterm{def:g10:universal-gravitation:satellite}{orbitnya} berbentuk lingkaran dan dilihat dari sisinya, sehingga $K$ adalah laju
\omterm{def:g10:universal-gravitation:satellite}{orbit} bintangnya seluruhnya.
\begin{enumerate}[resume]
  \item Dari $P$ dan hubungan yang diterima $r^3 = G M P^2/(4\pi^2)$, hitunglah jari-jari
        \omterm{def:g10:universal-gravitation:satellite}{orbit} planetnya $r$. Bandingkan dengan jarak Bumi--Matahari, \qty{1.5e11}{m}.
  \item Hitunglah laju \omterm{def:g10:universal-gravitation:satellite}{orbit} planetnya $V = 2\pi r/P$.
  \item Dari $K = (m/M)\,V$, hitunglah massa planetnya $m$.
  \item Bandingkan $m$ dengan Jupiter ($\qty{1.9e27}{kg}$) dan dengan Bumi
        ($\qty{6.0e24}{kg}$), serta $r$ dengan jarak Matahari--Merkurius
        ($\qty{5.8e10}{m}$). Dunia semacam apa ini?
  \item Jika \omterm{def:g10:universal-gravitation:satellite}{orbitnya} miring dan tidak dilihat dari sisinya, apakah massa yang sebenarnya
        lebih besar atau lebih kecil daripada nilaimu? Simpulkan dalam satu kalimat: apa
        yang baru saja diserahkan pergeseran berkala sebesar satu bagian dari lima juta?
\end{enumerate}
\end{problem}
