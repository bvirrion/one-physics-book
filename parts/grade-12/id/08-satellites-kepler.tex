\chapter{Satelit dan Gerak Planet}\label{ch:g12:satellites-kepler}

Antena parabola di balkon terpaku mati, namun sepanjang hari ia menatap satu
titik langit yang tetap, \qty{36000}{km} di atas; stasiun antariksa
melintasi langit petang dalam hitungan menit; serombongan jam terbang
memberi tahu telepon genggammu di mana kamu berdiri. Tidak satu pun mesin
ini menyalakan penggerak: semuanya sekadar \emph{jatuh}. Bab ini menghitung
secepat apa dan setinggi apa masing-masing harus jatuh --- lalu \omterm{met:g12:satellites-kepler:weigh}{menimbang
planet} dengan hukum yang sama.

\section{Jatuh mengelilingi Bumi}

Semuanya bertumpu pada hukum yang telah diukur sebelumnya
(\cref{ch:g10:universal-gravitation}): Bumi, bermassa $M$, menarik sebuah
\omterm{def:g10:universal-gravitation:satellite}{satelit} bermassa $m$ yang pusatnya berjarak $r$ dari pusat Bumi dengan \omterm{def:g10:inertia:force}{gaya}
yang terarah ke pusat Bumi, sebesar
\[ F = G\,\frac{m M}{r^2}, \qquad G = \qty{6.67e-11}{N.m^2/kg^2}. \]
Perhatikanlah apa itu $r$: jarak \emph{antarpusat}. \omterm{def:g10:universal-gravitation:satellite}{Satelit} pada ketinggian
$h$ memiliki $r = R + h$, dengan $R = \qty{6.37e6}{m}$ jari-jari Bumi ---
melupakan $R$ itu adalah kekeliruan klasik pada bab ini.

\begin{remark}[Meriam Newton, ditinjau kembali]\label{rem:g12:satellites-kepler:cannon}
Ditembakkan mendatar, sebutir peluru meriam mendarat; lebih cepat, ia
mendarat lebih jauh; pada laju yang tepat tanahnya melengkung menjauh persis
secepat pelurunya jatuh, dan jatuhnya menutup pada dirinya sendiri: sebuah
\omterm{def:g10:universal-gravitation:satellite}{orbit}. \omterm{def:g10:universal-gravitation:satellite}{Satelit} tidak memerlukan penggerak --- hanya laju \emph{ke samping}
yang tepat.
\end{remark}

\section{Orbit lingkaran: satu laju, satu periode}

\begin{theorem}[Laju orbit]\label{thm:g12:satellites-kepler:speed}
Sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} pada \omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran beraturan berjari-jari $r$ mengelilingi
benda bermassa $M$ bergerak dengan laju
\[ v = \sqrt{\frac{G M}{r}}, \]
yang tidak bergantung pada massa \omterm{def:g10:universal-gravitation:satellite}{satelitnya}.
\end{theorem}

\begin{proof}
Gravitasi adalah satu-satunya \omterm{def:g10:inertia:force}{gaya}, sehingga \omterm{thm:g12:newtons-laws:second}{hukum kedua Newton}
(\cref{thm:g12:newtons-laws:second}) berbunyi $m\vect a = \vect F$;
proyeksikanlah pada \omterm{def:g12:kinematics-2d:frenet}{kerangka Frenet} (\cref{def:g12:kinematics-2d:frenet}).
\omterm{def:g10:inertia:force}{Gayanya} membidik pusatnya: komponen tangensialnya nol, sehingga lajunya
tetap; komponen normalnya adalah seluruh $GmM/r^2$,
dan gerak melingkar memiliki $a_n = v^2/r$
(\cref{thm:g12:kinematics-2d:centripetal}). Karena itu $m v^2/r =
GmM/r^2$, dan --- dengan $m$ yang \omterm{def:g10:inertia:compensate}{saling meniadakan} --- $v^2 = GM/r$.
\end{proof}

\begin{proposition}[Periode orbit]\label{prop:g12:satellites-kepler:period}
\omterm{def:g10:universal-gravitation:satellite}{Satelitnya} merampungkan \omterm{def:g10:universal-gravitation:satellite}{orbitnya} dalam \omterm{def:g12:kinematics-2d:ucm}{periode}
\[ T = \frac{2\pi r}{v} = 2\pi \sqrt{\frac{r^3}{G M}}. \]
\end{proposition}

\begin{proof}
Satu putaran adalah keliling $2\pi r$ pada laju tetap; sisipkanlah
$v = \sqrt{GM/r}$: $T = 2\pi r\sqrt{r/(GM)} = 2\pi\sqrt{r^3/(GM)}$.
\end{proof}

Baik $v$ maupun $T$ tidak memuat $m$: sebuah sekrup yang terlepas dari
stasiun antariksa mengorbit persis di sampingnya. Dan keduanya menurun
bersama $r$: \emph{kian tinggi \omterm{def:g10:universal-gravitation:satellite}{orbitnya}, kian lambat \omterm{def:g10:universal-gravitation:satellite}{satelitnya}}, dalam laju
($v \propto 1/\sqrt r$) dan dalam sudut ($T \propto r^{3/2}$).

\begin{example}[Stasiun antariksa]\label{ex:g12:satellites-kepler:iss}
Stasiun Antariksa Internasional terbang pada ketinggian
$h \approx \qty{400}{km}$, sehingga $r = \num{6.37e6} + \num{4.0e5} =
\qty{6.77e6}{m}$; dengan $GM = \num{6.67e-11} \times \num{5.97e24} =
\qty{3.98e14}{m^3/s^2}$, $v = \sqrt{\num{3.98e14}/\num{6.77e6}}
\approx \qty{7.7e3}{m/s}$ dan $T = 2\pi r/v \approx \qty{5.5e3}{s}
\approx \qty{92}{min}$: satu putaran planetnya tiap satu setengah jam,
sekitar enam belas matahari terbit sehari bagi antariksawannya.
\end{example}

\begin{example}[Bulan jatuh seperti apelnya]\label{ex:g12:satellites-kepler:moon}
Bulan, pada $r = \qty{3.84e8}{m}$, mengorbit dalam $T = \qty{27.3}{d} =
\qty{2.36e6}{s}$, sehingga $v = 2\pi r/T \approx \qty{1.02e3}{m/s}$ dan
$a = v^2/r \approx \qty{2.7e-3}{m/s^2}$ --- persis $g/60^2$, dan
Bulan terletak $60$ jari-jari Bumi jauhnya: tarikan yang menjatuhkan apelnya
membelokkan Bulan, yang dilemahkan kebalikan kuadratnya. Tiap sekon Bulan
``jatuh'' sekitar \qty{1.4}{mm} ke arah kita, sedangkan kilometer ke
sampingnya membawanya berkeliling --- pemeriksaan yang dilakukan Newton pada 1666.
\end{example}

\section{Hukum-hukum Kepler}

Newton tidak menerka kebalikan kuadratnya: ia menyaringnya dari tiga
keteraturan yang disuling Johannes Kepler, antara 1609 dan 1619, dari dua
puluh tahun kedudukan planet yang diamati Tycho Brahe dengan mata telanjang
--- data terbaik dunia sebelum teleskop.

\begin{definition}[Elips]\label{def:g12:satellites-kepler:ellipse}
\emph{Elips}\index{elips} adalah himpunan titik yang jumlah jaraknya ke dua
titik tetap, yaitu \emph{fokus}\index{fokus (elips)}, bernilai tetap;
setengah garis tengah terpanjangnya adalah \emph{setengah sumbu
panjang}\index{setengah sumbu panjang} $a$, dan lingkaran adalah keadaan
ketika kedua fokusnya menyatu. Titik \omterm{def:g10:universal-gravitation:satellite}{orbit} planet yang terdekat dan terjauh
dari Matahari adalah \emph{perihelion}\index{perihelion}
dan \emph{aphelion}\index{aphelion}.
\end{definition}

\begin{theorem}[Hukum-hukum Kepler]\label{thm:g12:satellites-kepler:kepler}
\index{hukum-hukum Kepler}%
Untuk planet yang mengorbit Matahari (bermassa $M$):
\begin{enumerate}
  \item tiap \omterm{def:g10:universal-gravitation:satellite}{orbitnya} berupa \omterm{def:g12:satellites-kepler:ellipse}{elips} dengan Matahari pada salah satu fokusnya;
  \item ruas Matahari--planet menyapu luas yang sama dalam waktu yang sama;
  \item nisbah kuadrat \omterm{def:g12:kinematics-2d:ucm}{periode} terhadap pangkat tiga \omterm{def:g12:satellites-kepler:ellipse}{setengah sumbu
        panjangnya} sama bagi semua planet: $T^2/a^3 = 4\pi^2/(G M)$.
\end{enumerate}
Hukum yang sama mengatur keluarga \omterm{def:g10:universal-gravitation:satellite}{satelit} mana pun mengelilingi benda pusat
mana pun, dengan $M$ massa pusatnya. \admitted
\end{theorem}

\begin{remark}[Apa yang dapat kita buktikan tahun ini]\label{rem:g12:satellites-kepler:proof}
Untuk \omterm{def:g10:universal-gravitation:satellite}{orbit} \emph{lingkaran} hukum ketiganya milik kita: dengan
mengkuadratkan \cref{prop:g12:satellites-kepler:period} diperoleh
$T^2/r^3 = 4\pi^2/(GM)$ --- satu tetapan bagi tiap \omterm{def:g10:universal-gravitation:satellite}{satelit} dengan pusat yang
sama. Bahwa \omterm{def:g12:satellites-kepler:ellipse}{elips}, luas yang sama dan $a$ yang umum menyusul dari kebalikan
kuadratnya ditunjukkan, dengan kalkulus yang lebih banyak, pada jilid
Tahun~ke-1. Sementara itu hukum keduanya dapat dibaca: planetnya bergerak
paling cepat di \omterm{def:g12:satellites-kepler:ellipse}{perihelion}, tempat ruas sapuannya yang pendek harus bergegas.
\end{remark}

\section{Orbit yang bekerja}

\begin{definition}[Orbit geostasioner]\label{def:g12:satellites-kepler:geo}
Sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} disebut \emph{geostasioner}\index{orbit geostasioner} apabila
ia menggantung di atas satu titik tanah yang tetap. \omterm{def:g10:universal-gravitation:satellite}{Orbitnya} haruslah
lingkaran, khatulistiwa, dan berperiode satu \emph{hari
sideris}\index{hari sideris}, $T = \qty{86164}{s}$ (\qty{23}{h}
\qty{56}{min} \qty{4}{s}): waktu yang dibutuhkan Bumi untuk berputar sekali
\emph{terhadap bintang}. (Hari \qty{24}{h} itu terhadap Matahari, dan ke
arahnya Bumi juga maju satu derajat sehari.)
\end{definition}

\begin{example}[Ketinggian geostasionernya]\label{ex:g12:satellites-kepler:geoalt}
Hukum ketiga Kepler, dibaca mundur, menentukan jari-jarinya:
$r = \left(GMT^2/4\pi^2\right)^{1/3} =
\left(\num{3.98e14} \times (\num{86164})^2/4\pi^2\right)^{1/3}
\approx \qty{4.22e7}{m}$ --- suatu ketinggian $h = r - R \approx
\qty{3.58e7}{m} \approx \qty{35800}{km}$, yang ditempuh pada $v = 2\pi r/T
\approx \qty{3.1}{km/s}$. Setiap penerus \omterm{def:g10:signals-and-waves:signal}{sinyal} yang ``menggantung diam''
terletak pada satu lingkaran ini di atas khatulistiwa --- tidak ada yang lain.
\end{example}

\begin{example}[Rombongan satelit navigasi]\label{ex:g12:satellites-kepler:gps}
Sistem navigasi \omterm{def:g10:universal-gravitation:satellite}{satelit} menerbangkan sekitar tiga puluh \omterm{def:g10:universal-gravitation:satellite}{satelit} pada
$r \approx \qty{2.66e7}{m}$ (ketinggian $\approx \qty{20200}{km}$),
tempat \cref{prop:g12:satellites-kepler:period} memberikan $T \approx
\qty{4.32e4}{s}$: setengah \omterm{def:g12:satellites-kepler:geo}{hari sideris}, sehingga tiap \omterm{def:g10:universal-gravitation:satellite}{satelitnya} menyusuri
kembali jejak tanahnya tiap hari. Penerimamu membandingkan waktu tiba \omterm{def:g10:signals-and-waves:signal}{sinyal}
jamnya terhadap \omterm{def:g10:universal-gravitation:satellite}{orbit} yang dikenal sampai ke \omterm{def:g10:orders-of-magnitude:unit}{meter}.
\end{example}

\begin{method}[Menimbang sebuah dunia dari satelitnya]\label{met:g12:satellites-kepler:weigh}
\index{menimbang planet}%
Untuk mengukur massa sebuah planet atau bintang: carilah \omterm{def:g10:universal-gravitation:satellite}{satelit} mana pun
miliknya (sebuah bulan, sebuah wahana, sebuah planet), ukurlah jari-jari
\omterm{def:g10:universal-gravitation:satellite}{orbitnya} $r$ dan \omterm{def:g12:kinematics-2d:ucm}{periodenya}
$T$, lalu balikkanlah hukum ketiganya:
\[ M = \frac{4\pi^2 r^3}{G\,T^2}. \]
Ini hanya menimbang benda \emph{pusatnya} --- massa \omterm{def:g10:universal-gravitation:satellite}{satelitnya} sudah \omterm{def:g10:inertia:compensate}{saling
meniadakan} pada \cref{thm:g12:satellites-kepler:speed}. Setiap massa pada
kartu datanya diperoleh dengan cara ini: Bumi dari Bulan, Matahari dari \omterm{def:g10:universal-gravitation:satellite}{orbit}
Bumi sendiri.
\end{method}

\begin{remark}[Kian tinggi kian lambat --- tetapi kian mahal]\label{rem:g12:satellites-kepler:energy}
Karena $v = \sqrt{GM/r}$, Bulan merangkak pada \qty{1}{km/s} sementara
stasiunnya melesat pada \qty{7.7}{km/s}. Namun \omterm{def:g10:universal-gravitation:satellite}{orbit} yang lebih tinggi justru
lebih \emph{mahal}: menanjak melawan gravitasi menyimpan \omterm{def:g10:energy-conservation:energy}{energi} potensial
(perhitungannya dilunasi pada \cref{ch:g12:work-and-energy}), dan tambahan
$E_p$ mengalahkan susutnya $E_k$. Karena itu roket membakar \emph{ke depan}
untuk mencapai \omterm{def:g10:universal-gravitation:satellite}{orbit} yang lebih tinggi dan lebih lambat --- dan \omterm{def:g10:universal-gravitation:satellite}{satelit} yang
direm udara tipis membelit \emph{turun} lalu bertambah cepat.
\end{remark}

\begin{remark}[Kartu data]\label{rem:g12:satellites-kepler:data}
Kecuali disebutkan lain: $G = \qty{6.67e-11}{N.m^2/kg^2}$; Bumi:
$M = \qty{5.97e24}{kg}$, $R = \qty{6.37e6}{m}$, \omterm{def:g12:satellites-kepler:geo}{hari sideris}
\qty{86164}{s}; \omterm{def:g10:universal-gravitation:satellite}{orbit} Bulan: $r = \qty{3.84e8}{m}$, $T =
\qty{27.3}{d}$; Matahari: $M = \qty{1.99e30}{kg}$; jarak Bumi--Matahari
\qty{1.496e11}{m}; satu tahun $= \qty{3.156e7}{s}$; Mars: $M =
\qty{6.42e23}{kg}$, $R = \qty{3.39e6}{m}$.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:satellites-kepler:1}
Sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} \qty{1000}{kg} berada di landasan peluncurannya, lalu terbang
pada ketinggian \qty{400}{km}. Hitunglah tarikan Bumi di kedua tempatnya:
berapa persen yang bertahan di ketinggian --- dan mengapa penghuninya melayang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:satellites-kepler:2}
Untuk \omterm{def:g10:universal-gravitation:satellite}{satelit} (khayalan) yang menyerempet permukaan Bumi
($r = R$), hitunglah laju \omterm{def:g10:universal-gravitation:satellite}{orbit} dan \omterm{def:g12:kinematics-2d:ucm}{periodenya}. Bandingkanlah dengan peluru
meriam Newton (\cref{rem:g12:satellites-kepler:cannon}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:satellites-kepler:3}
Sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} dipindahkan dari \omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran berjari-jari $r$ ke \omterm{def:g10:universal-gravitation:satellite}{orbit}
berjari-jari $2r$. Dengan faktor berapa laju dan \omterm{def:g12:kinematics-2d:ucm}{periodenya} berubah? Apakah
jawabannya bergantung pada massanya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:satellites-kepler:4}
Ulangi angka stasiunnya: dari $r = \qty{6.77e6}{m}$, hitunglah
$v$, $T$, dan banyaknya \omterm{def:g10:universal-gravitation:satellite}{orbit} yang dirampungkan dalam \qty{24}{h}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:satellites-kepler:5}
Nyatakan ketiga hukum Kepler. Di sebelah mana pada elipsnya yang sangat
memanjang sebuah komet paling cepat, dan hukum mana yang mengatakannya?
Apakah \omterm{def:g12:kinematics-2d:ucm}{periode} sebuah \omterm{def:g10:universal-gravitation:satellite}{satelit} bergantung pada massanya sendiri?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:satellites-kepler:6}
Timbanglah Bumi: dari jari-jari \omterm{def:g10:universal-gravitation:satellite}{orbit} dan \omterm{def:g12:kinematics-2d:ucm}{periode} Bulan (kartu datanya),
hitunglah massa Bumi lalu bandingkan dengan nilai pada kartunya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:satellites-kepler:7}
Sebuah perusahaan mengusulkan \omterm{def:g10:universal-gravitation:satellite}{satelit} yang menggantung tetap di atas Paris
(lintang $49^\circ$ utara). Dari arah gravitasinya, jelaskan mengapa setiap
\omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran berpusat di pusat Bumi; simpulkan bahwa usulnya mustahil. Di
mana \omterm{def:g10:universal-gravitation:satellite}{satelit} \emph{dapat} menggantung?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:satellites-kepler:8}
Phobos mengelilingi Mars pada $r = \qty{9.38e6}{m}$ dalam $T = \qty{7}{h}
\qty{39}{min}$. Perolehlah massa Mars; bandingkan dengan kartu datanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:satellites-kepler:9}
Mars mengorbit Matahari pada $a = \qty{2.279e11}{m}$ dalam \qty{687}{d}.
Hitunglah $T^2/a^3$ untuk Mars dan untuk Bumi (kartu datanya), periksalah
hukum ketiga Kepler, lalu perolehlah massa Matahari.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:satellites-kepler:10}
Uji Bulan Newton: dari kartu datanya, hitunglah laju Bulan dan \omterm{thm:g12:kinematics-2d:centripetal}{percepatan
sentripetalnya} $a = v^2/r$. Tunjukkan bahwa nilainya $g/60^2$, lalu sebutkan
mengapa itu meyakinkan Newton bahwa satu hukum mengatur apel dan Bulan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:satellites-kepler:11}
Dalam \omterm{def:g10:orders-of-magnitude:unit}{satuan} Matahari (\omterm{def:g10:orders-of-magnitude:lightyear}{satuan astronomi}, tahun) hukum ketiga Kepler berbunyi
$T^2 = a^3$: garis lurus pada grafik log--log dalam bab ini. Tempatkanlah
padanya asteroid Ceres, $a = \qty{2.77}{au}$, dan komet Halley,
$a = \qty{17.8}{au}$: hitunglah kedua \omterm{def:g12:kinematics-2d:ucm}{periodenya}. Halley terakhir melewati
\omterm{def:g12:satellites-kepler:ellipse}{perihelion} pada 1986 --- kapan ia dinanti kembali?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:satellites-kepler:12}
Udara \omterm{def:g11:lenses-and-eye:lens}{tipis} menyapu stasiunnya, namun lajunya justru \emph{bertambah} sembari
ketinggiannya susut. Hitunglah $v$ pada ketinggian \qty{400}{km} dan
\qty{350}{km}, lalu selesaikanlah kemelutnya: \omterm{def:g10:inertia:inventory}{gesekan} memperlambat benda
--- dari mana datangnya tambahan laju itu?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:satellites-kepler:13}
Sebuah teleskop menemukan planet luar surya yang mengelilingi bintangnya pada
$r = \qty{7.48e9}{m}$ tiap \qty{3.50}{d}. Timbanglah bintangnya, dalam
\omterm{def:g10:orders-of-magnitude:unit}{kilogram} dan dalam \omterm{def:g10:orders-of-magnitude:unit}{satuan} Matahari. Massa yang mana --- bintangnya atau
planetnya --- yang \emph{tidak dapat} diperoleh dengan cara ini, dan mengapa?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:satellites-kepler:14}
Pemukim Mars menginginkan penerus \omterm{def:g10:signals-and-waves:signal}{sinyal} ``areostasioner'' di atas
pangkalannya. Mars berputar terhadap bintang dalam \qty{24}{h} \qty{37}{min}:
hitunglah jari-jari \omterm{def:g10:universal-gravitation:satellite}{orbit} dan ketinggiannya di atas permukaan Mars
(kartu datanya).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:satellites-kepler:15}
Komet Halley melewati \omterm{def:g12:satellites-kepler:ellipse}{perihelion} pada $r_p = \qty{8.77e10}{m}$ sambil
bergerak pada \qty{54.5}{km/s}; \omterm{def:g12:satellites-kepler:ellipse}{aphelionnya} pada $r_a = \qty{5.25e12}{m}$. Di
kedua ujungnya kecepatannya tegak lurus ruas Matahari--komet, dan hukum kedua
Kepler lalu memaksa $r_p v_p = r_a v_a$ (segitiga sapuan yang sama luas
$\tfrac12 r v\,\Delta t$). Perolehlah laju \omterm{def:g12:satellites-kepler:ellipse}{aphelionnya} lalu berilah ulasan
tentang nisbahnya.
\end{exercise}

\section{Soal: Sebuah tempat duduk di atas khatulistiwa}

\begin{problem}[{Soal akhir pekan --- menempatkan \omterm{def:g10:universal-gravitation:satellite}{satelit} di atas
khatulistiwa: mengapa satu-satunya tempat parkir di langit adalah lingkaran
\qty{35800}{km} di atas, bagaimana perbandingannya dengan stasiun yang
melesat di bawahnya, dan bagaimana hukum yang sama menimbang Yupiter sambil lalu}]
\label{pb:g12:satellites-kepler:1}
Sebuah perusahaan telekomunikasi menginginkan penerus \omterm{def:g10:signals-and-waves:signal}{sinyal} yang tidak
pernah harus dikejar antena tanahnya yang sekali dipaku: \omterm{def:g10:universal-gravitation:satellite}{satelit}
\omterm{def:g12:satellites-kepler:geo}{geostasioner}. Kamulah penganalisis misinya; pakailah kartu datanya. Untuk
Bagian~IV: Io, bulan Yupiter, mengorbit pada $r = \qty{4.22e8}{m}$ dalam
$T = \qty{1.77}{d}$, dan Yupiter berputar terhadap bintang dalam \qty{9}{h} \qty{56}{min}.

\medskip
\noindent\textbf{Bagian I --- Aturan menggantung diam.}
\begin{enumerate}
  \item \omterm{def:g10:universal-gravitation:satellite}{Satelitnya} harus tetap berada di atas satu titik tanah yang tetap.
        Berapa \omterm{def:g12:kinematics-2d:ucm}{periodenya} --- dan mengapa \omterm{def:g12:satellites-kepler:geo}{hari sideris}
        \qty{86164}{s} dan bukan \qty{24}{h}?
  \item Pada gerak melingkar \omterm{def:g11:forces-and-motion:netforce}{gaya totalnya} menunjuk ke pusat lingkarannya;
        di sini satu-satunya \omterm{def:g10:inertia:force}{gayanya} adalah gravitasi. Perolehlah bahwa
        pusat setiap \omterm{def:g10:universal-gravitation:satellite}{orbit} lingkaran adalah pusat Bumi.
  \item Sebuah lingkaran yang menggantung di atas lintang Paris akan berpusat
        pada sumbu Bumi, di sebelah utara pusatnya. Simpulkanlah: pada bidang
        mana \omterm{def:g12:satellites-kepler:geo}{orbit geostasioner} harus terletak?
  \item Tulislah \omterm{thm:g12:newtons-laws:second}{hukum kedua Newton} pada \omterm{def:g12:kinematics-2d:frenet}{kerangka Frenet} untuk \omterm{def:g10:universal-gravitation:satellite}{orbit}
        lingkaran berjari-jari $r$ lalu turunkanlah $v = \sqrt{GM/r}$.
  \item Perolehlah $T = 2\pi\sqrt{r^3/(GM)}$ lalu tulislah kembali sebagai
        hukum ketiga Kepler $T^2/r^3 = 4\pi^2/(GM)$.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Satu-satunya lingkaran yang berhasil.}
\begin{enumerate}[resume]
  \item Selesaikanlah hukum ketiganya untuk $r$ lalu hitunglah jari-jari
        \omterm{def:g12:satellites-kepler:geo}{geostasionernya}.
  \item Perolehlah ketinggiannya di atas tanah.
  \item Hitunglah laju \omterm{def:g10:universal-gravitation:satellite}{orbitnya} dari $v = 2\pi r/T$.
  \item Periksalah terhadap $v = \sqrt{GM/r}$.
  \item \omterm{def:g10:universal-gravitation:satellite}{Satelit} perusahaan itu bermassa \qty{4.0e3}{kg}; milik pesaingnya
        bermassa dua kali lipat. Bandingkanlah jari-jari \omterm{def:g12:satellites-kepler:geo}{geostasionernya}, lalu
        sebutkan apa yang diubah tambahan massa itu saat peluncurannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Stasiun di bawahnya.}
\begin{enumerate}[resume]
  \item Stasiun antariksa terbang pada $r = \qty{6.77e6}{m}$: hitunglah
        lajunya.
  \item Hitunglah \omterm{def:g12:kinematics-2d:ucm}{periodenya}, dalam sekon dan menit.
  \item Berapa \omterm{def:g10:universal-gravitation:satellite}{orbit} yang dirampungkannya dalam \qty{24}{h} --- kira-kira
        berapa matahari terbit yang disaksikan antariksawannya tiap hari?
  \item Hitunglah nisbah jari-jarinya $r_{\text{geo}}/r_{\text{sta}}$,
        lalu periksalah bahwa nisbah \omterm{def:g12:kinematics-2d:ucm}{periodenya} adalah bilangan itu
        berpangkat $3/2$, sebagaimana dituntut hukum ketiga Kepler.
  \item \omterm{def:g10:universal-gravitation:satellite}{Satelit} mana yang bergerak lebih cepat, dan mana yang menuntut lebih
        banyak \omterm{def:g10:energy-conservation:energy}{energi} tiap \omterm{def:g10:orders-of-magnitude:unit}{kilogram} untuk ditempatkan? Rukunkanlah keduanya
        dalam satu kalimat.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Menimbang Yupiter sambil lalu.}
\begin{enumerate}[resume]
  \item Dari hukum ketiga Kepler yang diterapkan pada Io, nyatakan massa
        Yupiter $M_J$ dalam $G$, $r$ dan $T$.
  \item Hitunglah $M_J$.
  \item Berapa kali massa Bumi nilainya? Kira-kira berapa bagian massa
        Matahari?
  \item Sebuah penerus \omterm{def:g10:signals-and-waves:signal}{sinyal} ``yovistasioner'' hendaknya menggantung di
        atas satu titik awan Yupiter: hitunglah jari-jari \omterm{def:g10:universal-gravitation:satellite}{orbitnya}; bandingkan
        dengan milik Io.
  \item Laporkanlah kepada direksinya, dua kalimat: ketinggian tempat
        perusahaannya harus memarkir penerus isyaratnya, dan massa Yupiter
        yang diserahkan hukum yang sama persis itu secara cuma-cuma.
\end{enumerate}
\end{problem}
