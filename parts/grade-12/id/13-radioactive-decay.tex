\chapter{Peluruhan Radioaktif}\label{ch:g12:radioactive-decay}

Di ruang bawah tanah sebuah museum, pencacah Geiger berdetik di atas sekeping kayu
dari peti mati seorang firaun. Tiap detiknya tanpa hukum: tidak ada yang
mengumumkannya, tidak ada yang meramalkan yang berikutnya. Namun pada hari Jumat
laboratoriumnya akan mencetak sebuah tarikh yang tepat sampai seabad. Bab ini
menyelesaikan kemelutnya --- bagaimana peristiwa yang sepenuhnya acak satu demi satu
berjumlah, dalam triliunan, menjadi jam yang paling mantap yang kita miliki.

\section{Tanpa hukum satu demi satu, serapi jam dalam hitungan mol}

Tahun lalu (\cref{ch:g11:nucleus-radioactivity}) kita menemui inti yang tidak
mantap, menyetimbangkan persamaan $\alpha$ dan $\beta$-nya, dan melihat separuh
contoh besar mana pun lenyap tiap \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} $T_{1/2}$. Yang tidak dapat kita katakan
adalah \emph{kapan di antara anak tangganya} --- maupun mengapa proses yang acak
sama sekali menepati jadwal. Kedua pertanyaan itu memiliki satu jawaban.

\begin{definition}[Tetapan peluruhan]\label{def:g12:radioactive-decay:lambda}
Bagi tiap \omterm{def:g11:nucleus-radioactivity:notation}{nuklida} yang tidak mantap ada sebuah tetapan $\lambda$, \emph{tetapan
peluruhan}\index{tetapan peluruhan}nya (bersatuan \unit{s^{-1}}), sedemikian sehingga selama
selang singkat $\dd t$ tiap intinya berpeluang $p = \lambda\,\dd t$ untuk
meluruh --- sama bagi setiap inti sejenisnya, berapa pun umurnya,
dan tak tersentuh oleh suhu, \omterm{def:g10:pressure:pressure}{tekanan} atau kimianya.
\end{definition}

\begin{remark}[Tanpa hukum satu-satu, tepat dalam kerumunan]\label{rem:g12:radioactive-decay:crowds}
Sebuah inti yang sudah menunggu sepuluh ribu tahun sama saja peluangnya untuk meluruh
pada sekon ini dengan inti yang dibuat pagi tadi: tanpa keausan, tanpa peringatan ---
mana yang meluruh berikutnya sungguh tak teramalkan. Kerumunan adalah perkara lain. Lemparlah
$100$ keping uang: harapkanlah $50$ kepala, tetapi janganlah heran menemukan $43$ ---
fluktuasinya berorde $\sqrt n$, di sini $10\%$. Lemparlah $10^{20}$: fluktuasi
nisbinya $1/\sqrt n$ bernilai $10^{-10}$, hasilnya pasti sampai
sepuluh desimal. Segram benda memuat sekitar $10^{22}$ inti: ketanpahukumannya
merata-rata menjadi kerapian jam, sebuah hukum yang tepat bagi $N(t)$, banyaknya inti
yang masih utuh pada waktu $t$.
\end{remark}

\section{Hukum peluruhannya}

\begin{proposition}[Persamaan evolusinya]\label{prop:g12:radioactive-decay:ode}
Pada contoh besar berisi $N$ inti, selama $\dd t$ banyaknya berubah sebesar
\[
\dd N = -\lambda N\,\dd t, \qquad\text{yaitu}\qquad \frac{\dd N}{\dd t} = -\lambda N :
\]
penduduknya menyusut pada laju yang sebanding dengan dirinya sendiri.
\end{proposition}

\begin{proof}
Masing-masing dari $N$ intinya meluruh selama $\dd t$ dengan peluang
$\lambda\,\dd t$: harapan peluruhannya $N\lambda\,\dd t$, dan untuk $N$ yang besar
cacah sesungguhnya melekat pada harapannya
(\cref{rem:g12:radioactive-decay:crowds}); tiap peluruhan membuang satu inti.
\end{proof}

\begin{theorem}[Hukum peluruhan radioaktif]\label{thm:g12:radioactive-decay:law}
Contoh yang memegang $N_0$ inti pada $t = 0$ memegang, pada waktu $t$,
\[
N(t) = N_0 \, e^{-\lambda t}.
\]
\end{theorem}

\begin{proof}
Periksalah dengan penyulihan:
$\dd N/\dd t = N_0(-\lambda)\,e^{-\lambda t} = -\lambda N(t)$, dan
$N(0) = N_0 e^0 = N_0$: persamaannya dan awalnya keduanya tepat. (Bahwa tidak ada
fungsi \emph{lain} yang cocok dibuktikan, dengan integral, pada jilid
universitasnya.)
\end{proof}

\begin{definition}[Peluruhan eksponensial]\label{def:g12:radioactive-decay:exponential}
Besaran yang menyusul $N_0\,e^{-\lambda t}$ mengalami \emph{peluruhan
eksponensial}\index{peluruhan eksponensial}: dalam waktu yang sama, \emph{bagian} yang
sama hilang. Inilah persis \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} yang mengosong pada
\cref{ch:g12:rc-rl-circuits}, dengan $\lambda$ memainkan $1/(RC)$: satu
persamaan diferensial, dipelajari sekali, menjalankan sirkuit dan inti sekaligus.
\end{definition}

\begin{proposition}[Waktu paruh]\label{prop:g12:radioactive-decay:halflife}
\omterm{def:g11:nucleus-radioactivity:halflife}{Waktu paruh} (\cref{def:g11:nucleus-radioactivity:halflife}) sebuah \omterm{def:g11:nucleus-radioactivity:notation}{nuklida}
ditetapkan oleh \omterm{def:g12:radioactive-decay:lambda}{tetapan peluruhannya}: $T_{1/2} = \ln 2/\lambda$, dan setelah $n$
\omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} $N = N_0/2^{\,n}$, baik $n$ bulat maupun bukan.
\end{proposition}

\begin{proof}
$N(T_{1/2}) = N_0/2$ menuntut $e^{-\lambda T_{1/2}} = \tfrac12$, yaitu
$\lambda T_{1/2} = \ln 2$; dan $N(n\,T_{1/2}) = N_0\,e^{-n\ln 2} =
N_0/2^{\,n}$ untuk $n$ berapa pun: eksponensialnya menjahit anak tangganya lalu mengisi
ruang di antaranya.
\end{proof}

\section{Aktivitas: menghitung detiknya}

\begin{proposition}[Aktivitas]\label{prop:g12:radioactive-decay:activity}
\omterm{def:g11:nucleus-radioactivity:activity}{Aktivitas} sebuah contoh --- peluruhannya tiap sekon, dalam \omterm{def:g11:nucleus-radioactivity:activity}{becquerel}
(\cref{def:g11:nucleus-radioactivity:activity}) --- adalah
\[
\mathcal A = \lambda N, \qquad\text{sehingga}\qquad \mathcal A(t) = \mathcal A_0 \, e^{-\lambda t}:
\]
sebanding dengan penduduk yang bertahan, dan sama-sama eksponensial.
\end{proposition}

\begin{proof}
$N$ inti, masing-masing berpeluang $\lambda$ tiap sekon: $\lambda N$
peluruhan tiap sekon; mengalikan \cref{thm:g12:radioactive-decay:law} dengan
$\lambda$ memberikan $\mathcal A(t) = \mathcal A_0 e^{-\lambda t}$.
\end{proof}

\begin{example}[$N$ raksasa, $\lambda$ mungil]\label{ex:g12:radioactive-decay:body}
Sebuah pisang berdetik pada sekitar \qty{15}{Bq}, tubuh manusia pada sekitar
\qty{8000}{Bq}, semeter kubik granit pada sekitar \qty{e6}{Bq}
(\cref{ch:g11:nucleus-radioactivity}). Di balik detik yang bersahaja itu duduk
penduduk yang mahabesar: kalium-40 di dalam tubuhnya menyumbang
$\mathcal A \approx \qty{4.4e3}{Bq}$ dengan $\lambda =
\qty{1.8e-17}{s^{-1}}$: $N = \mathcal A/\lambda \approx \num{2.5e20}$
inti, yang darinya hanya beberapa ribu meletup tiap sekon.
\end{example}

\begin{method}[Mengukur waktu paruh]\label{met:g12:radioactive-decay:semilog}
\begin{enumerate}
  \item Cacahlah peluruhannya dengan pencacah Geiger sepanjang selang yang sama
        berturut-turut: ini mencuplik $\mathcal A$ pada waktu yang berturut-turut.
  \item Lukiskanlah $\ln \mathcal A$ terhadap $t$: karena
        $\ln \mathcal A = \ln \mathcal A_0 - \lambda t$, \omterm{def:g12:radioactive-decay:exponential}{peluruhan
        eksponensialnya} tampil sebagai \emph{garis lurus} --- data yang bengkok berarti
        hukum yang lain, atau campuran \omterm{def:g11:nucleus-radioactivity:notation}{nuklida}.
  \item Bacalah $\lambda$ sebagai minus kemiringannya; $T_{1/2} = \ln 2/\lambda$.
\end{enumerate}
\end{method}

\begin{remark}[Becquerel bukanlah bahaya]\label{rem:g12:radioactive-decay:dose}
Becquerel menghitung peluruhan, bukan kerusakan. Bahayanya dilacak oleh \omterm{rem:g11:nucleus-radioactivity:dose}{dosis} dalam
\omterm{rem:g11:nucleus-radioactivity:dose}{sievert} (\cref{ch:g11:nucleus-radioactivity}), yang menimbang \omterm{def:g10:energy-conservation:energy}{energi} yang
diendapkan tiap \omterm{def:g10:orders-of-magnitude:unit}{kilogram} jaringan menurut kebengisan tiap radiasinya: sebuah
pindaian teknesium \qty{5e8}{Bq} adalah hal biasa, \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} yang jauh lebih sedikit dari
pemancar $\alpha$ yang terhirup tidaklah demikian; mengubah \unit{Bq} menjadi \unit{Sv} adalah
bahan universitas.
\end{remark}

\section{Penarikhan: membaca waktu dari sebuah nisbah}

Hukum peluruhannya yang dijalankan mundur adalah sebuah jam: ukurlah bagian yang bertahan,
dan eksponensialnya menyebutkan waktu yang berlalu.

\begin{method}[Penarikhan radiokarbon]\label{met:g12:radioactive-decay:carbon}
\begin{enumerate}
  \item \emph{Semasa hidup:} jaringan yang hidup bertukar karbon dengan atmosfernya
        lalu membawa perbandingan karbon-14-nya --- satu \omterm{def:g11:fundamental-interactions:ladder}{atom} dalam $10^{12}$,
        yang memberikan $\mathcal A_0 = 13.6$ peluruhan tiap menit tiap gram karbon.
  \item \emph{Kematian menjalankan jamnya:} asupannya berhenti, dan perbandingannya
        meluruh dengan $T_{1/2} = \num{5730}$ tahun.
  \item Ukurlah \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} contohnya $\mathcal A$ tiap gram karbon;
        $\mathcal A = \mathcal A_0 e^{-\lambda t}$ terbalik menjadi
        $t = \frac{1}{\lambda}\ln(\mathcal A_0/\mathcal A)
        = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}
\end{method}

\begin{example}[Sebuah tungku di dalam gua]\label{ex:g12:radioactive-decay:hearth}
Arang dari tungku yang terkubur memberikan $5.0$ peluruhan tiap menit tiap gram:
$\mathcal A_0/\mathcal A = 13.6/5.0 = 2.72$, sehingga
$t = \num{5730} \times \ln(2.72)/\ln 2 \approx \num{8.3e3}$ tahun. Apinya
padam delapan ribu tahun yang lalu --- kayunya sendirilah arsipnya.
\end{example}

\begin{example}[Jam bagi batuan]\label{ex:g12:radioactive-decay:rocks}
Karbon-14 membisu melewati sekitar \num{50000} tahun; geologi menyimpan jam yang
lebih lambat. \emph{Kalium--argon}: kalium-40 ($T_{1/2} = \num{1.25e9}$
tahun) meluruh menjadi argon, gas yang lolos dari lava yang meleleh tetapi terperangkap sekali
batuannya menghablur --- pemadatannya menolkan jamnya. \emph{Uranium--timbal}:
uranium-238 mengepalai rantai peluruhan yang berakhir pada timbal-206 yang mantap
(\cref{ch:g11:nucleus-radioactivity}), dan nisbah timbal terhadap uranium di dalam
sebuah hablur menarikhkannya. Kedua jamnya sepakat tentang mineral yang tertua
($\approx \num{4.4e9}$ tahun) dan, lewat meteorit, tentang umur Tata
Suryanya.
\end{example}

\begin{remark}[Rantai, dan gas di ruang bawah tanah]\label{rem:g12:radioactive-decay:radon}
Antara uranium dan timbal rantainya melewati radium, lalu radon-222,
sebuah \emph{gas} radioaktif ($T_{1/2} = 3.8$ hari). Lahir di dalam granit dan
tanah, ia merembes ke ruang bawah tanah lalu menumpuk; anak-anaknya yang memancarkan
$\alpha$ meluruh di dalam paru --- \omterm{rem:g11:nucleus-radioactivity:dose}{dosis} alami terbesar bagi kebanyakan orang, dan
yang paling murah untuk dikurangi: bukalah jendela ruang bawah tanahnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:1}
Iodium-131 memiliki $\lambda = \qty{1.0e-6}{s^{-1}}$. Hitunglah \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruhnya} dalam
sekon, lalu dalam hari.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:2}
Radon-222 memiliki $T_{1/2} = 3.8$ hari. Hitunglah $\lambda$ dalam \unit{s^{-1}},
lalu \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} sebuah contoh berisi \num{1.0e10} inti.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:3}
Periksalah dengan penyulihan bahwa $N(t) = N_0\,e^{-\lambda t}$ memenuhi
$\dd N/\dd t = -\lambda N$ dan $N(0) = N_0$; lalu hitunglah $N(T_{1/2})/N_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:4}
Sebuah sumber sesium-137 ($T_{1/2} = 30$ tahun) beraktivitas \qty{4.0e5}{Bq}
hari ini. Hitunglah aktivitasnya dalam $90$ tahun, lalu dalam $300$ tahun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:5}
Grafik semilog $\ln \mathcal A$ terhadap $t$ untuk sebuah contoh teknesium berupa
garis lurus melalui $(0, 20.00)$ dan $(\qty{10}{jam}, 18.85)$. Carilah
$\lambda$ (dalam \unit{h^{-1}} dan \unit{s^{-1}}) dan $T_{1/2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:6}
Segram radium-226 ($T_{1/2} = 1600$ tahun; massa molar
\qty{226}{g/mol}; $N_A = \qty{6.02e23}{mol^{-1}}$): hitunglah banyaknya
inti, lalu $\lambda$, lalu aktivitasnya --- \omterm{def:g10:orders-of-magnitude:unit}{satuan} \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} yang pertama dalam
sejarah, yaitu curie.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:7}
Dua siswa menyegel contoh berisi $20$ dan \num{2.0e20} inti dari satu
\omterm{def:g11:nucleus-radioactivity:notation}{nuklida} yang sama. Apa yang diharapkan masing-masingnya setelah satu \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}? Ramalan
siapa yang dapat diandalkan? (Taksirlah fluktuasi nisbi $1/\sqrt n$ untuk masing-masingnya.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:8}
Sebuah tubuh memegang sekitar \qty{140}{g} kalium, yang $0.012\%$-nya adalah
kalium-40 ($T_{1/2} = \num{1.25e9}$ tahun, massa molar \qty{40}{g/mol}).
Hitunglah banyaknya inti kalium-40, lalu aktivitasnya; bandingkanlah
dengan \qty{8000}{Bq} seluruh tubuhnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:9}
Seorang pasien menerima \qty{500}{MBq} teknesium-99m ($T_{1/2} = 6.0$
jam). Hitunglah aktivitasnya $15$ jam kemudian, lalu waktu baginya untuk
turun di bawah \qty{1}{MBq}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:10}
Arang dari sebuah lubang menunjukkan $3.4$ peluruhan tiap menit tiap gram karbon;
tulang dari lubang yang lain menunjukkan $30\%$ laju semasa hidupnya ($\mathcal A_0 = 13.6$
tiap menit tiap gram, $T_{1/2} = \num{5730}$ tahun). Tarikhkanlah kedua contohnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:11}
Sebuah ruang bawah tanah disegel dengan seangkatan radon-222 di dalamnya ($T_{1/2} = 3.8$
hari). Setelah berapa lama \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} angkatan itu turun sampai $1\%$? Ruang bawah tanah
yang sesungguhnya tetap radioaktif selama puluhan tahun: dari mana datangnya radon yang segar,
dan mengapa pengudaraannya harus tetap dan bukan sekali jalan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:12}
Tunjukkan dari $N = N_0 e^{-\lambda t}$ bahwa $t = T_{1/2}\ln(N_0/N)/\ln 2$. Sebuah
tulang menyisakan $5.0\%$ karbon-14 semasa hidupnya: tarikhkanlah.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:13}
Pada sebuah hablur yang tidak memerangkap argon saat pemadatannya, tiap inti
kalium-40 yang meluruh meninggalkan satu \omterm{def:g11:fundamental-interactions:ladder}{atom} argon di tempatnya. Tunjukkan bahwa
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$; sebuah batuan menunjukkan nisbah
$3.0$: carilah umurnya ($T_{1/2} = \num{1.25e9}$ tahun).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:14}
Sebuah pencacah Geiger memberikan, pada $t = 0$, $60$, $120$, $180$, $240$ sekon,
laju $400$, $225$, $127$, $71$, $40$ cacahan tiap sekon. Hitunglah
$\ln \mathcal A$ pada tiap waktunya, periksalah kesegarisannya, lalu perolehlah $\lambda$ dan
$T_{1/2}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:15}
Bagi sebuah inti tunggal, peluang bertahan selama $n$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} adalah
$1/2^{\,n}$. Hitunglah peluang bahwa (a) satu inti tertentu bertahan
$10$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}; (b) seluruh $8$ inti yang diamati bertahan satu \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}.
(c) Dari \num{1000} inti, berapa yang diharapkan setelah $3$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} ---
dan dapatkah hukumnya menyebutkan yang \emph{mana}? Simpulkanlah apa yang diramalkan hukumnya.
\end{exercise}

\section{Soal: Umur segala benda}

\begin{problem}[{Soal akhir pekan --- umur segala benda: sebuah laboratorium museum
menera jam karbonnya, menarikhkan sebuah arca dan sebuah papirus, mempelajari mengapa tidak ada
jam yang membaca melampaui pelatnya, lalu mendorong sebuah batuan gunung api mundur semiliar tahun}]
\label{pb:g12:radioactive-decay:1}
Sepekan di laboratorium penarikhan sebuah museum. Datanya: jaringan yang hidup menunjukkan
$\mathcal A_0 = 13.6$ peluruhan karbon-14 tiap menit tiap gram karbon;
$T_{1/2} = \num{5730}$ tahun (karbon-14), \num{1.25e9} tahun
(kalium-40); $1$ tahun $= \qty{3.156e7}{s}$;
$N_A = \qty{6.02e23}{mol^{-1}}$; massa molar karbon \qty{12}{g/mol};
pencacahnya memilah sampai sekitar $0.2$ peluruhan tiap menit tiap gram.

\medskip
\noindent\textbf{Bagian I --- Senin: menera jamnya.}
\begin{enumerate}
  \item Mengapa perbandingan karbon-14 tetap di dalam jaringan yang hidup, dan
        apa yang berubah saat kematiannya?
  \item Hitunglah $\lambda$ untuk karbon-14 dalam \unit{s^{-1}}.
  \item Periksalah dengan penyulihan bahwa $N_0 e^{-\lambda t}$ menyelesaikan
        $\dd N/\dd t = -\lambda N$.
  \item Dari $\mathcal A_0$, hitunglah banyaknya \omterm{def:g11:fundamental-interactions:ladder}{atom} karbon-14 tiap
        gram karbon di dalam jaringan yang hidup.
  \item Berapa \omterm{def:g11:fundamental-interactions:ladder}{atom} karbon yang dipegang segram? Perolehlah perbandingan
        karbon-14-nya --- bandingkan dengan ``satu dalam $10^{12}$''.
  \item Tunjukkan bahwa \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} terukur $\mathcal A$ menarikhkan sebuah contoh sebagai
        $t = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Selasa: arca dan papirusnya.}
\begin{enumerate}[resume]
  \item Sebuah arca kayu memberikan $9.1$ peluruhan tiap menit tiap gram. Tarikhkanlah
        kayunya.
  \item Peristiwa apa, tepatnya, yang ditandai tarikh itu --- pemahatannya atau
        sesuatu yang lain? Kehati-hatian apa yang menyusul?
  \item Sebuah papirus Mesir memberikan $10.4$ peluruhan tiap menit tiap gram. Tarikhkanlah.
        Masuk akalkah seorang juru tulis dua puluh dua abad yang lalu?
  \item Aktivitasnya terukur sampai $\pm 0.2$ peluruhan tiap menit. Taksirlah
        ketakpastian umur papirusnya, dengan memakai
        $\Delta t \approx \Delta\mathcal A/(\lambda \mathcal A)$.
  \item Perkamen ``kuno'' seorang pedagang memberikan $13.4$ peluruhan tiap menit
        tiap gram. Apa putusannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Kamis: batas jamnya.}
\begin{enumerate}[resume]
  \item Setelah berapa \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} \omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} sebuah contoh turun di bawah
        $0.2$ peluruhan tiap menit tiap gram milik pencacahnya? Umur berapa itu?
  \item Laboratorium yang terbaik mencapai sekitar sepuluh \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}. Berapa
        cakrawala praktis penarikhan radiokarbonnya, dalam tahun?
  \item Sebuah tulang dinosaurus berumur sekitar \num{6.6e7} tahun: berapa
        \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} karbon-14 itu? Dengan memakai pertanyaan 4, setelah berapa \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}
        bahkan tidak \emph{satu} \omterm{def:g11:fundamental-interactions:ladder}{atom} pun karbon-14 segram itu tersisa?
        Simpulkanlah tentang ``menarikhkan dinosaurus dengan karbon''.
  \item \omterm{def:g11:nucleus-radioactivity:notation}{Nuklida} macam apa yang justru dapat menarikhkan batuan dinosaurusnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Jumat: batuan gunung apinya.}
\begin{enumerate}[resume]
  \item Argon adalah gas, kalium penghuni setia sebuah padatan: jelaskan
        mengapa pemadatan lava menolkan jam kalium--argonnya.
  \item Dengan menganggap tiap inti kalium-40 yang meluruh meninggalkan satu \omterm{def:g11:fundamental-interactions:ladder}{atom}
        argon yang terperangkap, tunjukkan bahwa
        $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.
  \item Batuan di bawah fosil museumnya menunjukkan
        $N_{\mathrm{Ar}}/N_{\mathrm K} = 0.60$. Tarikhkanlah batuannya.
  \item Dapatkah karbon-14 menarikhkan batuan ini, atau kalium--argon menarikhkan
        arcanya? Sebutkan jendela guna tiap jamnya dalam \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}.
  \item Laporan Jumat, satu kalimat: ketiga umur yang ditemukan pekan ini, dan
        satu hukum yang membaca semuanya.
\end{enumerate}
\end{problem}
