\chapter{Osilasi Listrik Bebas: RLC}\label{ch:g12:rlc-oscillations}

Putarlah tombol \omterm{def:g10:universal-gravitation:weight}{berat} sebuah radio loteng dan stasiunnya berbaris melewati
jarumnya: sebuah suara, biola, desis, lalu sebuah suara lagi. Di balik papannya
tidak ada motor dan tidak ada komputer --- sebuah \omterm{def:g11:magnetic-fields:solenoid}{kumparan}, sebuah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} yang
kepingnya bersilang dan diputar tombolnya, dan seuntai kawat antena. Bab ini
menyambungkan kedua tangki \omterm{def:g10:energy-conservation:energy}{energi} bab lalu (\cref{ch:g12:rc-rl-circuits})
kepala ke kepala lalu menemukan bandul tercepat yang pernah dibangun: muatan yang
berayun maju mundur, sampai jutaan kali tiap sekon, pada irama yang dipilih
tombolnya.

\section{Bandul listrik: sirkuit LC}

Isilah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} $C$ sampai tegangan $U_0$ --- kepingnya lalu memegang
$\pm Q_0$ dengan $Q_0 = C U_0$ --- lalu pada $t = 0$ sambungkanlah ke
\omterm{def:g11:magnetic-fields:solenoid}{kumparan} \emph{ideal} $L$, yang \omterm{def:g11:circuits-and-power:resistance}{hambatannya} nol. Tidak ada sumber di mana pun pada
simpalnya: apa pun yang terjadi berikutnya, sirkuitnya melakukannya sendiri.

{\small \omterm{prop:g12:rlc-oscillations:equation}{Sirkuit LC}: sebuah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} yang terisi sampai $Q_0$, sebuah sakelar, sebuah
\omterm{def:g11:magnetic-fields:solenoid}{kumparan} ideal. Menutup $K$ membiarkan muatannya tumpah melalui \omterm{def:g11:magnetic-fields:solenoid}{kumparannya} --- dan
\omterm{def:g11:magnetic-fields:solenoid}{kumparannya}, yang membenci perubahan, tidak akan membiarkannya berhenti.}

\begin{proposition}[Persamaan LC]\label{prop:g12:rlc-oscillations:equation}
\index{sirkuit LC}
Pada simpal LC yang ideal muatan $q(t)$ keping atasnya menaati
\[
L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0,
\qquad\text{yaitu}\qquad
\frac{d^2q}{dt^2} = -\frac{1}{LC}\,q .
\]
\end{proposition}

\begin{proof}
Sepanjang simpalnya \omterm{def:g11:circuits-and-power:voltage}{tegangannya} berjumlah nol
(\cref{prop:g11:circuits-and-power:laws}): $u_L + u_C = 0$.
\omterm{def:g12:rc-rl-circuits:capacitor}{Kapasitornya} memberikan $u_C = q/C$ dan \omterm{def:g11:magnetic-fields:solenoid}{kumparannya} $u_L = L\,di/dt$
(\cref{def:g12:rc-rl-circuits:inductor}), dengan $i = dq/dt$ menghitung
muatan yang tiba pada keping atasnya
(\cref{prop:g12:rc-rl-circuits:cdudt}); sulihkanlah.
\end{proof}

\begin{theorem}[Osilasi bebas sirkuit LC]\label{thm:g12:rlc-oscillations:oscillation}
\index{osilasi listrik}
Dilepaskan pada $t = 0$ dengan $q(0) = Q_0$ dan $i(0) = 0$, sirkuitnya
berosilasi:
\[
q(t) = Q_0 \cos\!\left(\frac{2\pi t}{T_0}\right),
\qquad
i(t) = -I_0 \sin\!\left(\frac{2\pi t}{T_0}\right),
\qquad
T_0 = 2\pi\sqrt{LC},
\]
dengan arus puncak $I_0 = 2\pi Q_0/T_0 = Q_0/\sqrt{LC}$.
\end{theorem}

\begin{proof}
Periksalah dengan penyulihan. Dengan menurunkannya,
$i = dq/dt = -\frac{2\pi}{T_0} Q_0 \sin(2\pi t/T_0)$ --- yaitu
$i(t)$ yang disebutkan --- dan sekali lagi $\frac{d^2q}{dt^2} = -(\frac{2\pi}{T_0})^2 q$,
yang sama dengan $-q/(LC)$ tepat ketika $(2\pi/T_0)^2 = 1/(LC)$, yaitu
$T_0 = 2\pi\sqrt{LC}$. Dan $q(0) = Q_0$, $i(0) = 0$. Bahwa tidak ada
fungsi \emph{lain} yang cocok dengan persamaannya dan awalnya diterima saja,
persis seperti pada pegasnya
(\cref{rem:g12:oscillators-and-time:uniqueness}).
\end{proof}

\begin{definition}[Periode dan frekuensi alami]\label{def:g12:rlc-oscillations:period}
\emph{Periode alami}\index{periode alami!sirkuit LC} sebuah
\omterm{prop:g12:rlc-oscillations:equation}{sirkuit LC} adalah $T_0 = 2\pi\sqrt{LC}$;
\emph{frekuensi alaminya}\index{frekuensi alami!sirkuit LC} adalah
\[
f_0 = \frac{1}{T_0} = \frac{1}{2\pi\sqrt{LC}} .
\]
Pemeriksaan dimensinya: $\qty{1}{H} = \qty{1}{V.s/A}$ dan
$\qty{1}{F} = \qty{1}{C/V}$, sehingga $LC$ membawa
$\unit{s} \times \unit{C/A} = \unit{s^2}$ --- $\sqrt{LC}$ adalah waktu,
sebagaimana seharusnya.
\end{definition}

\begin{example}[Angka yang pertama]\label{ex:g12:rlc-oscillations:firstnumbers}
$L = \qty{10}{mH}$ dengan $C = \qty{1.0}{\micro F}$:
$\sqrt{LC} = \qty{1.0e-4}{s}$, sehingga $T_0 = \qty{0.63}{ms}$ dan
$f_0 \approx \qty{1.6}{kHz}$ --- sebuah nada yang terdengar. $L = \qty{1.0}{mH}$
dengan $C = \qty{100}{pF}$: $T_0 = \qty{2.0}{\micro s}$,
$f_0 \approx \qty{0.50}{MHz}$ --- sebuah \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} radio. Hanya
\emph{hasil kali} $LC$ yang menetapkan iramanya.
\end{example}

\begin{remark}[Terpaut seperempat periode]\label{rem:g12:rlc-oscillations:quarter}
Arusnya terpaut seperempat \omterm{def:g12:oscillators-and-time:oscillator}{periode} \omterm{def:g12:oscillators-and-time:phase}{fasenya} dari muatannya: $i = 0$
ketika $q = \pm Q_0$ (kepingnya penuh, alirannya berbalik) dan $i = \pm I_0$
ketika $q = 0$ (kepingnya kosong, alirannya deras penuh) --- persis seperti
bandul, yang diam di ujungnya dan tercepat di dasarnya.
\end{remark}

\section{Waltz energinya}

\begin{proposition}[Kekekalan energi yang tersimpan]\label{prop:g12:rlc-oscillations:energy}
\index{energi!sirkuit LC}
Pada \omterm{prop:g12:rlc-oscillations:equation}{sirkuit LC} yang ideal \omterm{def:g10:energy-conservation:energy}{energi} yang tersimpan seluruhnya
(\cref{prop:g12:rc-rl-circuits:energy})
\[
E \;=\; E_C + E_L \;=\; \frac{q^2}{2C} + \tfrac12\,L i^2
\;=\; \frac{Q_0^2}{2C} \;=\; \tfrac12\,L I_0^2
\]
bernilai tetap: seluruhnya di \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitornya} ketika $q = \pm Q_0$, seluruhnya di
\omterm{def:g11:magnetic-fields:solenoid}{kumparannya} ketika $i = \pm I_0$.
\end{proposition}

\begin{proof}
Turunkanlah: $\frac{dE}{dt} = \frac{q}{C}\frac{dq}{dt} +
Li\frac{di}{dt} = i\left(\frac{q}{C} + L\frac{d^2q}{dt^2}\right) = 0$
menurut \cref{prop:g12:rlc-oscillations:equation}. Nilainya dibaca pada
$t = 0$ (seluruhnya \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor}), dan sekali lagi pada saat seluruhnya arus
$q = 0$ --- dari situ $\tfrac12 L I_0^2 = Q_0^2/2C$, yang memulihkan
$I_0 = Q_0/\sqrt{LC}$ hanya dengan \omterm{def:g10:energy-conservation:energy}{energi}.
\end{proof}

lalu kembali dua kali tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode}, dengan jumlahnya terpaku pada $Q_0^2/2C$ --- persis
gambar \omterm{prop:g12:oscillators-and-time:spring-period}{osilator pegas--massanya}, yang dilabeli ulang.}
\end{omfigure}

\begin{example}[Waltznya, yang terukur]\label{ex:g12:rlc-oscillations:waltz}
$C = \qty{1.0}{\micro F}$ yang terisi sampai $U_0 = \qty{10}{V}$, disambungkan ke
$L = \qty{10}{mH}$: \omterm{def:g10:energy-conservation:energy}{energinya} $E = \tfrac12 C U_0^2 = \qty{5.0e-5}{J}$,
\omterm{def:g12:oscillators-and-time:oscillator}{periodenya} $T_0 = \qty{0.63}{ms}$, dan arus puncak
$I_0 = U_0\sqrt{C/L} = \qty{0.10}{A}$ yang tercapai seperempat \omterm{def:g12:oscillators-and-time:oscillator}{periode}
(\qty{0.16}{ms}) setelah sakelarnya menutup.
\end{example}

\section{Sirkuit yang sesungguhnya: osilasi teredam}

Setiap \omterm{def:g11:magnetic-fields:solenoid}{kumparan} yang sesungguhnya digulung dari kawat yang berhambatan: simpal yang jujur adalah
sirkuit \emph{RLC} seri, dan \omterm{def:g11:circuits-and-power:resistance}{hambatannya} memungut waltznya.

\begin{definition}[Rezim pseudoperiodik dan aperiodik]\label{def:g12:rlc-oscillations:regimes}
\index{sirkuit RLC}\index{rezim pseudoperiodik}\index{rezim aperiodik}
Pada sirkuit RLC seri resistornya melesapkan \omterm{def:g10:energy-conservation:energy}{energi} lewat pemanasan
Joule (\cref{ch:g11:circuits-and-power}): osilasi bebasnya menjadi
\emph{\omterm{def:g12:oscillators-and-time:damping}{teredam}}. Dua rezim, yang terbaca pada \omterm{def:g10:signals-and-waves:oscilloscope}{osiloskop}:
\begin{itemize}
  \item \emph{pseudoperiodik} ($R$ kecil): sirkuitnya masih
        berosilasi di dalam \omterm{def:g10:refraction:fiber}{selubung} yang menyusut, dengan waktu pengulangan ---
        yaitu \emph{pseudoperiode}\index{pseudoperiode!listrik} ---
        yang dekat dengan $T_0 = 2\pi\sqrt{LC}$ apabila \omterm{def:g12:oscillators-and-time:damping}{peredamannya} ringan;
  \item \emph{aperiodik} ($R$ besar): muatannya merayap kembali ke nol
        tanpa pernah melampauinya --- tanpa osilasi sama sekali.
\end{itemize}
Persis cerminan \omterm{def:g12:oscillators-and-time:oscillator}{osilator} mekanik yang \omterm{def:g12:oscillators-and-time:damping}{teredam}
(\cref{def:g12:oscillators-and-time:damping}).
\end{definition}

\begin{remark}[Memelihara osilasinya]\label{rem:g12:rlc-oscillations:maintain}
\index{osilasi terpelihara}
Untuk menjaga \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya} tetap hidup, suapilah sirkuitnya tepat sebanyak \omterm{def:g10:energy-conservation:energy}{energi} yang
dibakar \omterm{def:g11:circuits-and-power:resistance}{hambatannya}, sekali tiap daur dan \emph{selaras} dengan ayunannya: sebuah
penguat melakukan bagi simpal RLC apa yang dilakukan dorongan kaitan bagi
bandulnya (\cref{ch:g12:oscillators-and-time}). Setiap arloji kuarsa
dan setiap pemancar radio adalah \omterm{def:g12:oscillators-and-time:oscillator}{osilator} terpelihara semacam itu, yang berjalan
pada \omterm{def:g12:rlc-oscillations:period}{frekuensi alaminya} $f_0$.
\end{remark}

\section{Kembaran mekaniknya}

\begin{proposition}[Kamus listrik--mekanik]\label{prop:g12:rlc-oscillations:analogy}
\index{analogi listrik--mekanik}
Persamaan LC adalah persamaan pegas--massa
(\cref{prop:g12:oscillators-and-time:spring-period}) kata demi kata:
\begin{center}\small
\begin{tabular}{lcl}
\omterm{prop:g12:rlc-oscillations:equation}{sirkuit LC} & & kereta luncur pegas--massa\\[2pt]
muatan $q$ & $\longleftrightarrow$ & kedudukan $x$\\
arus $i = dq/dt$ & $\longleftrightarrow$ & kecepatan $v = dx/dt$\\
\omterm{def:g12:rc-rl-circuits:inductor}{induktansi} $L$ & $\longleftrightarrow$ & massa $m$\\
kebalikan \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitansi} $1/C$ & $\longleftrightarrow$ & \omterm{def:g12:oscillators-and-time:spring}{kekakuan} $k$\\
\omterm{def:g11:circuits-and-power:resistance}{hambatan} $R$ & $\longleftrightarrow$ & \omterm{def:g10:inertia:inventory}{gesekan}\\
$L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0$ & $\longleftrightarrow$ &
  $m\,\frac{d^2x}{dt^2} + kx = 0$\\[2pt]
$T_0 = 2\pi\sqrt{LC}$ & $\longleftrightarrow$ &
  $T_0 = 2\pi\sqrt{m/k}$\\[2pt]
$E_C = q^2/2C$ & $\longleftrightarrow$ & $E_p = \tfrac12 kx^2$\\
$E_L = \tfrac12 Li^2$ & $\longleftrightarrow$ & $E_k = \tfrac12 mv^2$
\end{tabular}
\end{center}
Persamaan yang sama, penyelesaian yang sama: setiap hasil tentang salah satu \omterm{def:g12:oscillators-and-time:oscillator}{osilatornya}
seketika beralih ke bahasa yang lain.
\end{proposition}

\begin{proof}
Bandingkanlah \cref{prop:g12:rlc-oscillations:equation} dengan
$m\,d^2x/dt^2 = -kx$: dengan menyulihkan $q \to x$, $L \to m$, $1/C \to k$
yang satu berubah menjadi yang lain, dan $LC \to m/k$ mengubah
$2\pi\sqrt{LC}$ menjadi $2\pi\sqrt{m/k}$.
\end{proof}

\begin{example}[Menala radio]\label{ex:g12:rlc-oscillations:tuning}
Antenanya menyuapkan bisikan \emph{setiap} stasiun ke dalam simpal LC, tetapi
hanya yang memancar di dekat $f_0$ yang mendorong selaras lalu menumpuk ---
\omterm{def:g12:oscillators-and-time:resonance}{resonansi} (\cref{def:g12:oscillators-and-time:resonance}). Tombolnya
memutar keping sebuah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} variabel: mengubah $C$ menggeser
$f_0 = 1/(2\pi\sqrt{LC})$, dan jarumnya menyerahkan satu stasiun kepada
telinganya pada satu waktu. Di sisi pemancarnya, sebuah \omterm{def:g12:oscillators-and-time:oscillator}{osilator} terpelihara
(\cref{rem:g12:rlc-oscillations:maintain}) menetapkan \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} siarannya;
di dalam arloji, sebutir hablur kuarsa --- kembaran mekanik yang
\omterm{def:g12:oscillators-and-time:spring}{kekakuannya} memesona --- memainkan peran yang sama.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:1}
Hitunglah $T_0$ dan $f_0$ untuk (a) $L = \qty{10}{mH}$,
$C = \qty{1.0}{\micro F}$; (b) $L = \qty{1.0}{mH}$,
$C = \qty{100}{pF}$. Dengan faktor berapa \omterm{def:g12:oscillators-and-time:oscillator}{frekuensinya} berbeda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:2}
Dengan $\qty{1}{H} = \qty{1}{V.s/A}$ dan $\qty{1}{F} = \qty{1}{C/V}$,
tunjukkan \omterm{def:g10:orders-of-magnitude:unit}{satuan} demi \omterm{def:g10:orders-of-magnitude:unit}{satuan} bahwa $\sqrt{LC}$ adalah waktu. Mengapa $2\pi$ itu
tidak mengubah apa pun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:3}
Sebuah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} \qty{470}{nF} diisi sampai \qty{6.0}{V}. Hitunglah $Q_0$
dan \omterm{def:g10:energy-conservation:energy}{energi} yang tersimpannya; lukiskanlah, tanpa perhitungan, apa yang menyusul
ketika \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} itu disambungkan ke \omterm{def:g11:magnetic-fields:solenoid}{kumparan} yang ideal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:4}
Sebuah \omterm{prop:g12:rlc-oscillations:equation}{sirkuit LC} menyusul $q(t) = 2.0\cos(2\pi t/T_0)$ dalam mikrocoulomb,
dengan $T_0 = \qty{4.0}{ms}$. Bacalah $Q_0$; hitunglah $f_0$, $q(T_0/2)$,
$i(0)$, dan arus puncaknya $I_0 = 2\pi Q_0/T_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:5}
Pada saat mana dalam daurnya \omterm{def:g10:energy-conservation:energy}{energinya} seluruhnya berada di
\omterm{def:g12:rc-rl-circuits:capacitor}{kapasitornya}? Seluruhnya di \omterm{def:g11:magnetic-fields:solenoid}{kumparannya}? Apa padanan mekanik tiap
saat itu bagi sebuah bandul?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:6}
Periksalah dengan penyulihan bahwa $q(t) = Q_0\cos(2\pi t/T_0)$ memenuhi
$L\,d^2q/dt^2 + q/C = 0$ tepat ketika $T_0 = 2\pi\sqrt{LC}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:7}
Sebuah stasiun \omterm{def:g10:signals-and-waves:wave}{gelombang} panjang memancar pada \qty{198}{kHz}. Dengan
$C = \qty{220}{pF}$, \omterm{def:g12:rc-rl-circuits:inductor}{induktansi} berapa yang menalanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:8}
$C = \qty{1.0}{\micro F}$ yang terisi sampai \qty{10}{V} mengosong ke dalam
$L = \qty{10}{mH}$. Hitunglah \omterm{def:g10:energy-conservation:energy}{energi} yang tersimpannya, arus puncaknya (dengan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:9}
Jelaskan mengapa $E_C$ dan $E_L$ berosilasi pada \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} $2f_0$, bukan $f_0$.
Tunjukkan bahwa pada $t = T_0/8$ (bermula dari $q = Q_0$) \omterm{def:g10:energy-conservation:energy}{energinya} terbagi
tepat separuh.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:10}
Pada \omterm{def:g10:signals-and-waves:oscilloscope}{osilogram} sebuah \omterm{def:g12:rlc-oscillations:regimes}{sirkuit RLC} yang \omterm{def:g12:oscillators-and-time:damping}{teredam} ringan, puncak berturut-turutnya
membaca \qty{4.0}{V} lalu \qty{3.0}{V}. Dengan menganggap nisbahnya tetap
tetap, ramalkanlah puncak keenam setelah yang \qty{4.0}{V} itu. Berapa
bagian \emph{\omterm{def:g10:energy-conservation:energy}{energinya}} yang bertahan tiap \omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode}? Apa yang dapat
kamu katakan tentang \omterm{def:g12:rlc-oscillations:regimes}{pseudoperiodenya} sendiri?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:11}
Sebuah \omterm{prop:g12:rlc-oscillations:equation}{sirkuit LC} memiliki $L = \qty{0.50}{H}$ dan $C = \qty{20}{\micro F}$.
Hitunglah $T_0$; lalu, dengan kamusnya
(\cref{prop:g12:rlc-oscillations:analogy}), carilah \omterm{def:g12:oscillators-and-time:spring}{kekakuan} $k$
yang memberi kereta luncur \qty{0.50}{kg} \omterm{def:g12:oscillators-and-time:oscillator}{periode} yang sama, lalu periksalah.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:12}
Sebuah penala memakai $L = \qty{0.20}{mH}$ dan \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} variabel yang menyapu
\qtyrange{50}{500}{pF}. Hitunglah kedua \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} ujungnya. Tunjukkan bahwa
$f_0 \propto 1/\sqrt{C}$, sehingga sepuluh kali lipat pada $C$ hanya memberikan
$\sqrt{10} \approx 3.2$ pada \omterm{def:g12:oscillators-and-time:oscillator}{frekuensinya}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:13}
Sebuah \omterm{def:g11:magnetic-fields:solenoid}{kumparan} yang sesungguhnya membuat sirkuitnya kehilangan $5.0\%$ \omterm{def:g10:energy-conservation:energy}{energinya} tiap
\omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode}. Setelah berapa \omterm{def:g12:oscillators-and-time:oscillator}{periode} \omterm{def:g10:energy-conservation:energy}{energinya} menyusut separuh? Berapa
\omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya} saat itu, sebagai bagian dari awalnya? Pada

\end{exercise}

Sebuah \omterm{def:g12:oscillators-and-time:oscillator}{osilator} arloji menyimpan $E = \qty{1.0e-9}{J}$ dan kehilangan $2.0\%$

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:15}

\section{Soal: Membangun radio kristal}

\begin{problem}[{Soal akhir pekan --- membangun radio kristal: satu
kumparan, satu kapasitor variabel, sebuah dioda dan sebuah telinga --- tanpa baterai,
tanpa penguat, dan warta petang tiba dengan energi gelombangnya
sendiri}]
\label{pb:g12:rlc-oscillations:1}
Sebuah perkakas dari loteng kakek: sebuah \omterm{def:g11:magnetic-fields:solenoid}{kumparan} $L = \qty{0.20}{mH}$ yang digulung pada
tabung kardus, sebuah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor} variabel yang kepingnya berputar dari
\qty{20}{pF} sampai \qty{480}{pF}, sebuah dioda kristal, sebuah telinga, dan
\qty{20}{m} kawat antena. Stasiun yang hendak ditangkap memancar antara
\qty{530}{kHz} dan \qty{1700}{kHz}.

\medskip
\noindent\textbf{Bagian I --- Simpal penalanya.}
\begin{enumerate}
  \item \omterm{def:g12:rc-rl-circuits:capacitor}{Kapasitornya}, yang sudah terisi, dibiarkan mengosong ke dalam \omterm{def:g11:magnetic-fields:solenoid}{kumparannya}:
        tulislah hukum simpalnya lalu ubahlah menjadi persamaan diferensial
        untuk $q(t)$.
  \item Periksalah dengan penyulihan bahwa $q(t) = Q_0\cos(2\pi t/T_0)$
        menyelesaikannya, asalkan $T_0 = 2\pi\sqrt{LC}$.
  \item Periksalah dengan analisis dimensi bahwa $\sqrt{LC}$ adalah waktu.
  \item Dengan tombolnya di $C = \qty{330}{pF}$: hitunglah $T_0$ dan $f_0$. Apakah
        simpalnya tertala di dalam pitanya?
  \item Carilah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitansi} yang menala tepat pada \qty{1.00}{MHz}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:g12:rc-rl-circuits:capacitor}{Kapasitor} variabelnya.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitansi} yang dibutuhkan untuk \qty{530}{kHz}.
  \item Hitunglah \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitansi} yang dibutuhkan untuk \qty{1700}{kHz}.
  \item Hitunglah kedua \omterm{def:g12:oscillators-and-time:oscillator}{frekuensi} yang sungguh dicapai \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitor}
        \qtyrange{20}{480}{pF} milik perkakas itu. Apakah ia meliputi seluruh pitanya?
  \item Tunjukkan bahwa $f_0 \propto 1/\sqrt{C}$ pada $L$ yang tetap; dengan faktor
        berapa $C$ harus berubah untuk menggandakan $f_0$?
  \item Kepingnya membuka secara linear, sehingga $C$ turun mantap sembari tombolnya
        berputar. Jelaskan mengapa stasiunnya lalu berdesakan di salah satu ujung
        tombolnya --- dan ujung yang mana.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{def:g10:energy-conservation:energy}{Energi} dan \omterm{def:g12:oscillators-and-time:damping}{peredamannya}.}
Tertala pada \qty{1.00}{MHz} ($C = \qty{127}{pF}$), antenanya sesaat
mengisi \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitornya} sampai $U_0 = \qty{20}{mV}$.
\begin{enumerate}[resume]
  \item Hitunglah muatan $Q_0$ dan \omterm{def:g10:energy-conservation:energy}{energi} yang tersimpannya $E$.
  \item Hitunglah arus puncaknya $I_0$, dengan \omterm{thm:g10:energy-conservation:conservation}{kekekalan energi}.
  \item \omterm{def:g11:circuits-and-power:resistance}{Hambatan} \omterm{def:g11:magnetic-fields:solenoid}{kumparannya} melesapkan $4.0\%$ \omterm{def:g10:energy-conservation:energy}{energi} yang tersimpannya
        tiap daur: setelah berapa daur \omterm{def:g10:energy-conservation:energy}{energinya} menyusut separuh?
  \item Berapa lama itu dalam mikrosekon? Mengapa ``dentingan'' ini
        sendirian tidak dapat memainkan warta petangnya?
  \item Jelaskan mengapa \omterm{def:g10:signals-and-waves:wave}{gelombang} yang tiba menjaga simpalnya tetap berayun hanya
        ketika tertala pada stasiun itu --- dan apa yang memelihara
        osilasinya di \emph{pemancarnya}
        (\cref{rem:g12:rlc-oscillations:maintain}).
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kembaran mekaniknya.}
\begin{enumerate}[resume]
  \item Dengan kamusnya (\cref{prop:g12:rlc-oscillations:analogy})
        beserta kereta luncur bermassa $m = \qty{0.20}{kg}$, hitunglah
        \omterm{def:g12:oscillators-and-time:spring}{kekakuan} $k$ pegas yang menyamai sirkuit yang tertala itu.
  \item Sebuah pegas mobil yang kaku memiliki $k \approx \qty{5e4}{N/m}$. Berilah ulasan:
        dapatkah sistem pegas--massa di atas meja berosilasi pada megahertz?
  \item Pertahankanlah $k = \qty{100}{N/m}$ yang wajar sebagai gantinya: massa berapa
        yang berosilasi pada \qty{1.00}{MHz}?
  \item Benda \omterm{def:g11:lenses-and-eye:images}{nyata} apa yang merupakan \omterm{def:g12:oscillators-and-time:oscillator}{osilator} mekanik seringan bulu dan
        sekaku itu, dan di mana kamu menemuinya lebih awal tahun
        ini (\cref{ch:g12:oscillators-and-time})?
  \item Lembar spesifikasinya, satu kalimat: pita yang disapu radiomu, \omterm{def:g12:rc-rl-circuits:capacitor}{kapasitansi}
        yang memilih \qty{1.00}{MHz}, berapa lama satu dentingan
        bertahan, dan apa yang tetap memainkan wartanya.
\end{enumerate}
\end{problem}
