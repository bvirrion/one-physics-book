\chapter{Usaha dan Energi Mekanik}\label{ch:g12:work-and-energy}

Jatuhkanlah sebutir kelereng ke dalam mangkuk salad: ia menukik, melampaui
dasarnya, menanjak sampai ketinggian awalnya, ragu sejenak, lalu berbalik ---
dan setiap pembalikannya dapat diramalkan dari satu kurva (tampang mangkuknya)
dan satu garis mendatar (\omterm{def:g10:energy-conservation:energy}{energi} kelerengnya), tanpa satu pun persamaan gerak
diselesaikan. Tahun lalu memberi harga pada \omterm{def:g10:inertia:force}{gaya} di garis lurus
(\cref{ch:g11:work-of-force}) lalu menimbang \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} terhadap \omterm{def:g12:work-and-energy:ep}{energi
potensial} (\cref{ch:g11:mechanical-energy}); tahun ini pembukuannya berlaku
di mana saja: \omterm{def:g10:relative-motion:trajectory}{lintasan} yang melengkung, pegas, bentang alam sekali pandang.

\section{Usaha sepanjang lintasan mana pun}

\begin{definition}[Usaha sepanjang lintasan]\label{def:g12:work-and-energy:path}
Misalkan \omterm{def:g10:inertia:force}{gaya} $\vect F$ bekerja pada benda yang bergerak sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan}
mana pun dari $A$ ke $B$: potonglah \omterm{def:g10:relative-motion:trajectory}{lintasannya} menjadi perpindahan
$\vect{\dd\ell}$ yang begitu pendek sehingga masing-masing lurus dan $\vect F$
hampir tidak berubah sepanjangnya. Tiap langkahnya memperoleh
\omterm{def:g11:work-of-force:work}{usaha} kecil $\scal{F}{\dd\ell} = F\,\dd\ell\cos\theta$
(\cref{ch:g11:work-of-force}), dan
\emph{usaha sepanjang lintasan}\index{usaha (sebuah gaya)!sepanjang lintasan} adalah
jumlahnya, $W_{AB}(\vect F) = \sum \scal{F}{\dd\ell}$ --- yang pada
\omterm{def:g10:relative-motion:trajectory}{lintasan} lurus dengan \omterm{def:g10:inertia:force}{gaya} tetap menyusut menjadi
$F \times AB \times \cos\theta$ tahun lalu.
\end{definition}

\begin{proposition}[Usaha berat, lintasan mana pun]\label{prop:g12:work-and-energy:weight}
Sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} \emph{mana pun} dari $A$ (berketinggian $z_A$) ke $B$
(berketinggian $z_B$), \omterm{def:g10:universal-gravitation:weight}{berat} sebuah massa $m$ mengerjakan \omterm{def:g11:work-of-force:work}{usaha}
$W_{AB}(\vect P) = mg\,(z_A - z_B)$: penurunannya yang masuk, tidak pernah rutenya.
\end{proposition}

\begin{proof}
Tiap langkah pendeknya menyumbang $mg \times (\text{penurunan kecil})$
(perhitungan ruas lurus tahun lalu); penurunannya berlipat teleskopik menjadi
$z_A - z_B$. Keadaan melengkungnya, yang saat itu diterima saja, kini terserahkan.
\end{proof}

\begin{definition}[Gaya konservatif dan tak konservatif]\label{def:g12:work-and-energy:conservative}
Sebuah \omterm{def:g10:inertia:force}{gaya} disebut \emph{konservatif}\index{gaya konservatif} apabila \omterm{def:g11:work-of-force:work}{usahanya}
antara dua titik sama sepanjang setiap \omterm{def:g10:relative-motion:trajectory}{lintasan} (nol pada setiap perjalanan pulang
pergi), dan \emph{tak konservatif}\index{gaya tak konservatif}
sebaliknya. \omterm{def:g10:universal-gravitation:weight}{Berat} dan \omterm{def:g10:inertia:force}{gaya} pegas (bagian berikutnya): konservatif.
\omterm{def:g10:inertia:inventory}{Gesekan} gelincir tidak: karena berlawanan dengan geraknya, ia mengerjakan $W = -fL$ pada
\omterm{def:g10:relative-motion:trajectory}{lintasan} sepanjang $L$ --- pungutan tiap \omterm{def:g10:orders-of-magnitude:unit}{meter}, yang tidak pernah dikembalikan.
\end{definition}

\begin{example}[Dua jalan setapak ke pondoknya]\label{ex:g12:work-and-energy:trails}
Seorang pendaki \qty{75}{kg} menanjak \qty{300}{m} melalui jalan curam sepanjang
\qty{800}{m} atau \qty{2.4}{km} jalan berkelok. \omterm{def:g10:universal-gravitation:weight}{Beratnya} memungut
$-75 \times 9.81 \times 300 \approx \qty{-2.2e5}{J}$ pada keduanya; \omterm{def:g10:inertia:force}{gaya}
\omterm{def:g10:inertia:inventory}{gesekan} \qty{30}{N} memungut \qty{-2.4e4}{J} pada yang satu, \qty{-7.2e4}{J}
pada yang lain. \omterm{def:g10:universal-gravitation:weight}{Beratnya} mengembalikan pungutannya saat menurun; tagihan
\omterm{def:g10:inertia:inventory}{gesekannya} lenyap menjadi kalor, pada kedua arahnya.
\end{example}

\section{Energi potensial}

\begin{definition}[Energi potensial]\label{def:g12:work-and-energy:ep}
Pada setiap \omterm{def:g12:work-and-energy:conservative}{gaya konservatif} dilekatkan sebuah
\emph{energi potensial}\index{energi potensial!dari usaha konservatif}
$E_p$, yaitu bilangan yang hanya bergantung pada kedudukannya, yang ditakrifkan oleh
$W_{AB} = E_p(A) - E_p(B)$: \omterm{def:g11:work-of-force:work}{usaha} yang diserahkan adalah \omterm{def:g10:energy-conservation:energy}{energi} potensial yang dibelanjakan.
Dengan menetapkan $E_p = 0$ pada titik acuan pilihan, nilainya tertetapkan di mana-mana;
pilihan itu menggeser semua nilainya sebesar tetapan yang \omterm{def:g10:inertia:compensate}{saling meniadakan} pada setiap
selisihnya. Untuk \omterm{def:g10:universal-gravitation:weight}{beratnya} ini memulihkan $E_p = mgz$ tahun lalu,
yang kini berlaku sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} mana pun. \omterm{def:g10:inertia:inventory}{Gesekan} tidak dapat memiliki $E_p$: tagihan
yang bergantung pada rutenya bukanlah selisih bilangan di kedua ujungnya --- hanya \omterm{def:g12:work-and-energy:conservative}{gaya
konservatif} yang menyimpan secara terambil kembali, tepat seperti yang dilestarikan namanya.
\end{definition}

\begin{proposition}[Energi potensial elastis]\label{prop:g12:work-and-energy:elastic}
Pegas berkekakuan $k$ (\unit{N/m}) yang teregang atau termampat sebesar $x$
menarik kembali dengan \omterm{def:g10:inertia:force}{gaya} $F = kx$
(\cref{ch:g12:oscillators-and-time}); \omterm{def:g11:work-of-force:work}{usaha} yang diserahkannya saat kembali
ke keadaan diamnya --- yaitu
\emph{energi potensial elastisnya}\index{energi potensial!elastis} --- adalah
$E_p = \tfrac12\, k x^2$.
\end{proposition}

\begin{proof}
\omterm{def:g12:oscillators-and-time:pendulum}{Gaya pemulihnya} bekerja sepanjang geraknya, sehingga $\sum F\,\dd\ell$ adalah
\emph{luas di bawah grafik $F$--$x$}: sebuah segitiga beralas $x$ dan
bertinggi $kx$, yang luasnya $\tfrac12 x \times kx$. \omterm{def:g10:inertia:force}{Gayanya} pada tiap reganganya
sama saat pergi dan saat kembali: \omterm{def:g11:work-of-force:work}{usahanya} tertetapkan oleh kedua ujungnya, \omterm{def:g10:inertia:force}{gaya}
pegasnya \omterm{def:g12:work-and-energy:conservative}{konservatif}, dan $E_p$ tertakrif dengan baik.
\end{proof}

\section{Kedua teorema energinya}

\begin{theorem}[Teorema energi kinetik]\label{thm:g12:work-and-energy:ketheorem}\index{teorema energi kinetik}
Untuk gerak apa pun sebuah benda bermassa $m$ dari $A$ ke $B$ --- melengkung,
melingkar penuh, tiga dimensi ---
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
dengan jumlahnya melingkupi semua \omterm{def:g10:inertia:force}{gaya} yang bekerja pada bendanya.
\end{theorem}

\begin{proof}
Karena $v^2 = \scal{v}{v}$, kaidah hasil kali pada matematika tahun ini
memberikan turunan $v^2$ sebagai $2\,\scal{a}{v}$, sehingga
$\dd E_k/\dd t = m\,\scal{a}{v} = \scal{F_{\text{tot}}}{v}$ menurut
\omterm{thm:g12:newtons-laws:second}{hukum kedua Newton} (\cref{ch:g12:newtons-laws}). Dalam $\dd t$ yang singkat
bendanya berpindah $\vect{\dd\ell} = \vect v\,\dd t$: $E_k$ memperoleh
\omterm{def:g11:work-of-force:work}{usaha} kecil pada \cref{def:g12:work-and-energy:path}; menjumlahkan langkahnya
berarti menjumlahkan \omterm{def:g11:work-of-force:work}{usahanya} --- pernyataan yang tahun lalu diterima saja, kini diperoleh penuh.
\end{proof}

\begin{proposition}[Daya sesaat]\label{prop:g12:work-and-energy:power}
Pada setiap saat sebuah \omterm{def:g10:inertia:force}{gaya} $\vect F$ pada benda berkecepatan $\vect v$
menyerahkan \omterm{def:g11:work-of-force:power}{daya} $P = \scal{F}{v} = Fv\cos\theta$, dan
$\dd E_k/\dd t$ adalah jumlah \omterm{def:g11:work-of-force:power}{daya} semua \omterm{def:g10:inertia:force}{gayanya}: rumus kecepatan tetap
tahun lalu, yang tepat pada tiap saat gerak apa pun.
\end{proposition}

\begin{proof}
Bacalah dari bukti sebelumnya: tiap \omterm{def:g10:inertia:force}{gayanya} memasok
$\scal{F}{\dd\ell} = \scal{F}{v}\,\dd t$ ke dalam $E_k$ selama $\dd t$.
\end{proof}

\begin{theorem}[Teorema energi mekanik]\label{thm:g12:work-and-energy:emtheorem}\index{teorema energi mekanik}
Misalkan $E_m = E_k + E_p$, dengan $E_p$ menghimpun \omterm{def:g12:work-and-energy:ep}{energi potensial}
semua \omterm{def:g12:work-and-energy:conservative}{gaya konservatif} yang bekerja. Maka, sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasan} mana pun,
\[
\Delta E_m = W_{AB}(\text{gaya tak konservatif}),
\]
dan $E_m$ kekal setiap kali \omterm{def:g12:work-and-energy:conservative}{gaya tak konservatifnya} tidak mengerjakan \omterm{def:g11:work-of-force:work}{usaha}.
\end{theorem}

\begin{proof}
Pecahlah teorema \omterm{def:g11:mechanical-energy:kinetic}{energi kinetiknya}: $\Delta E_k = W_{\text{kons}} +
W_{\text{nk}} = -\Delta E_p + W_{\text{nk}}$; pindahkanlah $\Delta E_p$ ke ruas lain.
\end{proof}

\begin{example}[Bandulnya, kali ini dengan jujur]\label{ex:g12:work-and-energy:pendulum}
Sebutir beban bandul pada kawat sepanjang $L = \qty{2.5}{m}$ dilepaskan dari
keadaan diam pada $\qty{45}{\degree}$ dari garis tegak. \omterm{def:g11:circuits-and-power:voltage}{Tegangannya}, yang
tegak lurus geraknya, tak berdaya; \omterm{def:g10:universal-gravitation:weight}{beratnya} \omterm{def:g12:work-and-energy:conservative}{konservatif}:
$E_m$ kekal \emph{pada busur yang melengkung itu}, yang tahun lalu hanya
dapat diperiksa. Bebannya bermula $h = L(1 - \cos\qty{45}{\degree}) =
\qty{0.73}{m}$ di atas, sehingga di dasarnya $v = \sqrt{2gh} \approx
\qty{3.8}{m/s}$.
\end{example}

\begin{method}[Memeriksa neraca energi gerak apa pun]\label{met:g12:work-and-energy:audit}
\begin{enumerate}
  \item Pilihlah sistemnya lalu daftarlah \omterm{def:g10:inertia:force}{gaya} yang bekerja padanya.
  \item Pilahlah: tegak lurus geraknya $\to$ tanpa \omterm{def:g11:work-of-force:work}{usaha}; \omterm{def:g12:work-and-energy:conservative}{konservatif}
        $\to$ sebuah $E_p$ di dalam $E_m$; sisanya $\to$ $W_{\text{nk}}$
        (\omterm{def:g10:inertia:inventory}{gesekan} gelincir: $-fL$, dengan $L$ panjang \omterm{def:g10:relative-motion:trajectory}{lintasannya}).
  \item Tulislah $\Delta E_m = W_{\text{nk}}$ antara titik datanya dan
        titik pertanyaannya; lalu selesaikan. \omterm{def:g10:energy-conservation:energy}{Energi} menjawab ``secepat apa,
        setinggi apa''; hanya waktu dan percepatan yang membutuhkan hukum Newton.
\end{enumerate}
\end{method}

\section{Diagram energi}

\begin{definition}[Diagram energi, titik balik]\label{def:g12:work-and-energy:diagram}
Untuk gerak sepanjang satu sumbu $x$ dengan $E_m$ yang kekal,
\emph{diagram energi}\index{diagram energi} melukiskan kurva $E_p(x)$
dan garis mendatar $E_m$. Karena $E_k = E_m - E_p \geq 0$,
geraknya terkurung pada tempat kurvanya terletak di bawah garisnya; celah
antara keduanya adalah $E_k$, yang terbaca langsung. Pada
\emph{titik balik}\index{titik balik}, tempat kurvanya bertemu garisnya,
$v = 0$ dan geraknya berbalik.
\end{definition}

\begin{proposition}[Gaya dari bentang alamnya; kesetimbangannya]\label{prop:g12:work-and-energy:equilibria}
\omterm{def:g12:work-and-energy:conservative}{Gaya konservatif} sepanjang $x$ adalah minus kemiringan $E_p$:
$F = -\,\dd E_p/\dd x$ --- ia menunjuk \emph{menuruni bukit} pada diagramnya,
dan lenyap di tempat garis singgungnya mendatar: sebuah kesetimbangan. Titik
minimum $E_p$ bersifat \emph{mantap}\index{kesetimbangan mantap} --- benda
yang disimpangkan didorong kembali, seperti bola di lembah; titik maksimum bersifat
\emph{goyah}\index{kesetimbangan goyah} --- bendanya didorong menjauh, seperti
bola di puncak bukit.
\end{proposition}

\begin{proof}
Pada langkah kecil $\dd x$ \omterm{def:g10:inertia:force}{gayanya} mengerjakan $F\,\dd x = E_p(x) -
E_p(x + \dd x) = -\dd E_p$; bagilah dengan $\dd x$. Arah menurun di kedua sisi
sebuah lembah menunjuk kembali ke dalam; di sekitar puncak bukit, ke luar.
\end{proof}

\begin{remark}[Osilasi dan pelolosan]\label{rem:g12:work-and-energy:escape}
Osilasi kecil pada \cref{ch:g12:oscillators-and-time} bersemayam di
dasar lembah $E_p$. Dan jauh dari Bumi gravitasinya melemah: kurva
$E_p(r)$ menanjak kian lambat menuju sebuah langit-langit yang berhingga (luas
di bawah grafik \omterm{def:g10:inertia:force}{gaya} yang menyusut itu berhingga). Roket yang $E_m$-nya
mencapai langit-langitnya tidak pernah menemui \omterm{def:g12:work-and-energy:diagram}{titik balik}: ia
\emph{lolos}\index{laju lepas} --- sekitar \qty{11.2}{km/s} dari
tanah (\cref{exo:g12:work-and-energy:13}); di bawah itu ia
terikat. Langit-langitnya sendiri dihitung pada jilid Tahun ke-1.
\end{remark}

\begin{example}[Lingkaran maut]\label{ex:g12:work-and-energy:loop}
Sebutir kelereng yang dilepaskan dari ketinggian $H$ meluncur masuk ke lingkaran
tegak berjari-jari $R$. Dinamika melingkarnya (\cref{ch:g12:newtons-laws}) menuntut
$v_{\text{atas}}^2 \geq gR$ di puncaknya --- data yang dipinjam soal
\omterm{def:g10:relative-motion:trajectory}{lintasan} luncur tahun lalu, kini milik kita. Kekekalan dari keadaan diam pada $H$
sampai ke puncaknya (setinggi $2R$) memberikan $v_{\text{atas}}^2 = 2g(H - 2R)$, sehingga
$H \geq \tfrac52 R$: setengah jari-jari di atas puncak lingkarannya, berapa pun
massanya --- sedikit lebih tinggi di dunia yang sesungguhnya, yang bergesekan.
\end{example}

\section{Latihan}

\section{Soal: Lompat ski}

\begin{problem}[{Soal akhir pekan --- lompat ski: sebuah landasan luncur yang
diperiksa neracanya oleh pengukur laju, sebuah penerbangan jatuh bebas, ilmu ukur
yang lembut dari pendaratan yang curam, dan menara yang harus dibeli perancangnya
sekali gesekannya menagih}]
\label{pb:g12:work-and-energy:1}
Seorang pelompat ski Olimpiade ($m = \qty{70}{kg}$, termasuk skinya) bermula dari
keadaan diam di gerbang $H = \qty{45}{m}$ di atas meja lepas landasnya, meluncur
$L = \qty{90}{m}$ di landasan luncur yang melengkung, lalu lepas dari mejanya secara mendatar;
pengukur laju di sana membaca $v_0 = \qty{26}{m/s}$. Titik pendaratannya terletak
$h = \qty{40}{m}$ lebih rendah, pada lereng $\qty{35}{\degree}$. Bagian II
dan III mengabaikan hambatan udara.

\medskip\noindent\textbf{Bagian I --- Landasan luncurnya.}
\begin{enumerate}
  \item Ramalkanlah \omterm{rem:g12:work-and-energy:escape}{laju lepas} landas tanpa \omterm{def:g10:inertia:inventory}{gesekannya}; mengapa lengkungan
        landasan luncurnya tidak penting?
  \item Dengan acuan di mejanya: hitunglah $E_m$ di gerbangnya dan saat lepas landasnya.
  \item Hitunglah $\Delta E_m$; teorema mana yang menyebutkan pelakunya, dan siapa?
  \item Perolehlah \omterm{def:g10:inertia:force}{gaya} \omterm{def:g11:work-of-force:sign}{penghambat} rata-ratanya (salju ditambah udara) sepanjang \omterm{def:g10:relative-motion:trajectory}{lintasannya}.
  \item Berapa bagian $E_m$ yang hilang? Mengapa skinya dilapisi lilin dan tubuhnya membungkuk?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Penerbangannya.}
\begin{enumerate}[resume]
  \item Hanya \omterm{def:g10:universal-gravitation:weight}{beratnya} yang bekerja saat terbang: gerak mana pada
        \cref{ch:g12:projectile-motion} ini? Berapa komponen kecepatannya
        saat mendarat?
  \item Berapa laju pendaratannya, dua kali: dari komponennya, dan dari kekekalan
        $E_m$. Berapa dalam \unit{km/h}?
  \item Hitunglah \omterm{def:g12:projectile-motion:range}{waktu terbangnya} dan jarak mendatar yang diterbanginya.
  \item Pelompat yang sesungguhnya menunggangi udara bagaikan sayap, terbang jauh
        lebih jauh: dapatkah udara \emph{menaikkan} laju pendaratannya? Berilah alasan dengan
        teorema \omterm{def:g11:mechanical-energy:em}{energi mekaniknya}.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pendaratannya.}
\begin{enumerate}[resume]
  \item Hitunglah sudut kecepatan pendaratannya di bawah garis mendatar.
  \item Uraikanlah kecepatan itu menjadi komponen sejajar dan tegak lurus
        lerengnya.
  \item Kakinya hanya menyerap bagian yang tegak lurusnya: hitunglah
        \omterm{def:g11:mechanical-energy:kinetic}{energi kinetik} itu; bandingkan dengan \omterm{def:g10:energy-conservation:energy}{energi} seluruhnya.
  \item Tanah yang datar akan menyerap semuanya: setara dengan penurunan tegak
        berapakah tiap pendaratannya ($h_{\text{ek}} = v_\perp^2/2g$,
        atau $v^2/2g$)?
  \item Satu kalimat: mengapa bukit pendaratannya curam dan melengkung?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Menara perancangnya.}
Bukit yang lebih kecil hanya membutuhkan lepas landas pada $v_0' = \qty{24}{m/s}$;
landasan luncurnya yang lurus menurun pada $\qty{35}{\degree}$, sehingga tinggi gerbang $H'$
berarti panjang \omterm{def:g10:relative-motion:trajectory}{lintasan} $H'/\sin\qty{35}{\degree}$; dengan \omterm{def:g11:circuits-and-power:resistance}{hambatan} \qty{80}{N} yang sama.
\begin{enumerate}[resume]
  \item Hitunglah tinggi gerbang yang akan dibangun perancang tanpa \omterm{def:g10:inertia:inventory}{gesekan}.
  \item Tulislah teorema \omterm{def:g11:mechanical-energy:em}{energi mekaniknya}, dari gerbang ke meja, untuk $H'$.
  \item Selesaikanlah untuk $H'$.
  \item Berapa persen \omterm{def:g10:inertia:inventory}{gesekannya} menaikkan menaranya?
  \item Massanya tidak lagi \omterm{def:g10:inertia:compensate}{saling meniadakan}: hitunglah kembali $H'$ untuk pelompat
        \qty{60}{kg}. Siapa yang membutuhkan menara lebih tinggi, dan mengapa?
  \item Kesimpulannya: menara yang diumumkan, dan bagian \omterm{def:g10:inertia:inventory}{gesekan} di dalamnya.
\end{enumerate}
\end{problem}
