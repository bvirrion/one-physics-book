\chapter{रेडियोसक्रिय क्षय}\label{ch:g12:radioactive-decay}

किसी संग्रहालय के तहख़ाने में गीगर गणित्र फ़िरौन के ताबूत से निकली लकड़ी की एक
फाँक पर खटकता रहता है। हर खटका बेक़ायदा है: न किसी ने उसकी घोषणा की, न कोई अगले
का पूर्वानुमान लगा सकता है। फिर भी शुक्रवार तक प्रयोगशाला सदी भर की परिशुद्धता
वाली तारीख़ छाप देगी। यह अध्याय इसी विरोधाभास को सुलझाता है --- कि एक-एक करके
बिलकुल यादृच्छिक घटनाएँ खरबों में जुड़कर हमारी सबसे स्थिर घड़ियाँ कैसे बन जाती
हैं।

\section{एक-एक करके बेक़ायदा, मोल भर में घड़ी की चाल}

पिछले वर्ष (\cref{ch:g11:nucleus-radioactivity}) हम अस्थिर नाभिकों से मिले थे, उनके $\alpha$ और $\beta$
समीकरण संतुलित किए थे, और देखा था कि हर \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} $T_{1/2}$ में किसी भी बड़े
प्रतिदर्श का आधा ग़ायब हो जाता है। जो हम नहीं कह सके वह यह था कि \emph{सीढ़ियों
के बीच कब} --- और यह कि कोई यादृच्छिक प्रक्रिया समय-सारणी रखती ही क्यों है।
दोनों प्रश्नों का उत्तर एक ही है।

\begin{definition}[क्षय नियतांक]\label{def:g12:radioactive-decay:lambda}
हर अस्थिर \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} के लिए एक अचर $\lambda$ होता है, उसका
\emph{क्षय नियतांक}\index{क्षय नियतांक} (मात्रक \unit{s^{-1}}), जिसके कारण
किसी भी छोटे अंतराल $\dd t$ में हर नाभिक के \omterm{def:g11:mechanical-energy:dissipation}{क्षय} करने की प्रायिकता
$p = \lambda\,\dd t$ रहती है --- और वह उस जाति के हर नाभिक के लिए एक ही है, उसकी उम्र
चाहे जो हो, और ताप, दाब या रसायन से अछूती।
\end{definition}

\begin{remark}[अकेले में बेक़ायदा, भीड़ में ठीक-ठीक]\label{rem:g12:radioactive-decay:crowds}
जो नाभिक दस हज़ार वर्ष प्रतीक्षा कर चुका है, उसके इसी \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} \omterm{def:g11:mechanical-energy:dissipation}{क्षय} करने
की संभावना ठीक उतनी ही है जितनी आज सुबह बने नाभिक की: न घिसाव, न कोई चेतावनी
--- अगला कौन जाएगा, यह सचमुच अपूर्वानुमेय है। भीड़ का मामला दूसरा है। $100$
सिक्के उछालिए: $50$ चित की उम्मीद कीजिए, पर $43$ पर भी चौंकिए मत ---
उतार-चढ़ाव $\sqrt n$ की कोटि के होते हैं, यहाँ $10\%$। अब $10^{20}$ उछालिए: सापेक्ष
उतार-चढ़ाव $1/\sqrt n$ केवल $10^{-10}$ है, और परिणाम दस दशमलव स्थानों तक पक्का। और एक
ग्राम पदार्थ में कोई $10^{22}$ नाभिक होते हैं: बेक़ायदगी औसत होकर घड़ी की चाल बन
जाती है, यानी समय $t$ पर अब भी साबुत बचे नाभिकों की संख्या $N(t)$ के लिए
एक ठीक-ठीक नियम।
\end{remark}

\section{क्षय का नियम}

\begin{proposition}[विकास-समीकरण]\label{prop:g12:radioactive-decay:ode}
$N$ नाभिकों के बड़े प्रतिदर्श में $\dd t$ के दौरान उनकी संख्या इतनी बदलती है:
\[
\dd N = -\lambda N\,\dd t, \qquad\text{अर्थात्}\qquad \frac{\dd N}{\dd t} = -\lambda N :
\]
यानी आबादी अपने ही समानुपाती दर से सिकुड़ती है।
\end{proposition}

\begin{proof}
$N$ नाभिकों में से हर एक $\dd t$ के दौरान प्रायिकता $\lambda\,\dd t$ से \omterm{def:g11:mechanical-energy:dissipation}{क्षय} करता है:
इसलिए अपेक्षित \omterm{def:g11:mechanical-energy:dissipation}{क्षय} $N\lambda\,\dd t$, और बड़े $N$ के लिए असली गिनती अपेक्षित गिनती से
चिपकी रहती है (\cref{rem:g12:radioactive-decay:crowds}); और हर \omterm{def:g11:mechanical-energy:dissipation}{क्षय} एक नाभिक हटा देता है।
\end{proof}

\begin{theorem}[रेडियोसक्रिय क्षय का नियम]\label{thm:g12:radioactive-decay:law}
जिस प्रतिदर्श में $t = 0$ पर $N_0$ नाभिक हों, उसमें समय $t$ पर इतने रह जाते
हैं:
\[
N(t) = N_0 \, e^{-\lambda t}.
\]
\end{theorem}

\begin{proof}
प्रतिस्थापन से जाँचिए: $\dd N/\dd t = N_0(-\lambda)\,e^{-\lambda t} = -\lambda N(t)$, और $N(0) = N_0 e^0 = N_0$: समीकरण भी सही और शुरुआत भी। (और यह
कि \emph{कोई दूसरा} फलन इस पर नहीं बैठता, यह समाकलों के साथ विश्वविद्यालय के
खंड में सिद्ध किया गया है।)
\end{proof}

\begin{definition}[चरघातांकी क्षय]\label{def:g12:radioactive-decay:exponential}
जो राशि $N_0\,e^{-\lambda t}$ मानती है वह \emph{चरघातांकी क्षय}\index{चरघातांकी क्षय} से
गुज़रती है: बराबर समयों में बराबर \emph{अंश} खोए जाते हैं। यह ठीक \cref{ch:g12:rc-rl-circuits} वाला
विसर्जित होता \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} है, जहाँ $\lambda$ $1/(RC)$ की भूमिका निभाता है:
एक ही अवकल समीकरण, एक बार सीखा हुआ, परिपथ भी चलाता है और नाभिक भी।
\end{definition}

\begin{proposition}[अर्धायु]\label{prop:g12:radioactive-decay:halflife}
किसी \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} की \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} (\cref{def:g11:nucleus-radioactivity:halflife}) उसका
\omterm{def:g12:radioactive-decay:lambda}{क्षय नियतांक} तय कर देता है: $T_{1/2} = \ln 2/\lambda$, और $n$ अर्धायुओं के बाद $N = N_0/2^{\,n}$,
चाहे $n$ पूरा हो या नहीं।
\end{proposition}

\begin{proof}
$N(T_{1/2}) = N_0/2$ के लिए $e^{-\lambda T_{1/2}} = \tfrac12$ चाहिए, यानी $\lambda T_{1/2} = \ln 2$; और किसी भी $n$ के लिए $N(n\,T_{1/2}) = N_0\,e^{-n\ln 2} =
N_0/2^{\,n}$:
चरघातांक सीढ़ी में धागा पिरो देता है और सीढ़ियों के बीच की जगह भर देता है।
\end{proof}

\section{सक्रियता: खटके गिनना}

\begin{proposition}[सक्रियता]\label{prop:g12:radioactive-decay:activity}
किसी प्रतिदर्श की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} --- यानी उसके प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} \omterm{def:g11:mechanical-energy:dissipation}{क्षय},
\omterm{def:g11:nucleus-radioactivity:activity}{बेक्रेल} (\cref{def:g11:nucleus-radioactivity:activity}) में --- है
\[
\mathcal A = \lambda N, \qquad\text{इसलिए}\qquad \mathcal A(t) = \mathcal A_0 \, e^{-\lambda t}:
\]
जो बचे हुए नाभिकों के समानुपाती है, और उतनी ही चरघातांकी।
\end{proposition}

\begin{proof}
$N$ नाभिक, हर एक की प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} प्रायिकता $\lambda$: यानी प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $\lambda N$
\omterm{def:g11:mechanical-energy:dissipation}{क्षय}; और \cref{thm:g12:radioactive-decay:law} को $\lambda$ से गुणा करने पर $\mathcal A(t) = \mathcal A_0 e^{-\lambda t}$ मिल जाता है।
\end{proof}

\begin{example}[विशाल $N$, नन्हा $\lambda$]\label{ex:g12:radioactive-decay:body}
केला लगभग \qty{15}{Bq} पर खटकता है, मनुष्य का शरीर लगभग \qty{8000}{Bq} पर, और ग्रेनाइट का
एक घन \omterm{def:g10:orders-of-magnitude:unit}{मीटर} कोई \qty{e6}{Bq} पर (\cref{ch:g11:nucleus-radioactivity})। इन मामूली खटकों के पीछे खगोलीय
आबादियाँ बैठी हैं: शरीर का पोटैशियम-40 $\lambda =
\qty{1.8e-17}{s^{-1}}$ के साथ $\mathcal A \approx \qty{4.4e3}{Bq}$ देता है: यानी
$N = \mathcal A/\lambda \approx \num{2.5e20}$ नाभिक, जिनमें से मुश्किल से कुछ हज़ार हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} चलते हैं।
\end{example}

\begin{method}[अर्धायु नापना]\label{met:g12:radioactive-decay:semilog}
\begin{enumerate}
  \item गीगर गणित्र से क्रमागत बराबर अंतरालों पर \omterm{def:g11:mechanical-energy:dissipation}{क्षय} गिनिए: इससे क्रमागत
        समयों पर $\mathcal A$ का प्रतिचयन हो जाता है।
  \item $t$ के सामने $\ln \mathcal A$ आलेखित कीजिए: चूँकि $\ln \mathcal A = \ln \mathcal A_0 - \lambda t$, इसलिए
        \omterm{def:g12:radioactive-decay:exponential}{चरघातांकी क्षय} एक \emph{सीधी रेखा} बनकर दिखता है --- और
        टेढ़े आँकड़े किसी और नियम, या न्यूक्लाइडों के मिश्रण का संकेत
        हैं।
  \item ढाल के ऋणात्मक के रूप में $\lambda$ पढ़ लीजिए; $T_{1/2} = \ln 2/\lambda$।
\end{enumerate}
\end{method}

\begin{remark}[बेक्रेल हानि नहीं हैं]\label{rem:g12:radioactive-decay:dose}
\omterm{def:g11:nucleus-radioactivity:activity}{बेक्रेल} \omterm{def:g11:mechanical-energy:dissipation}{क्षय} गिनता है, नुक़सान नहीं। नुक़सान का हिसाब
\omterm{rem:g11:nucleus-radioactivity:dose}{सीवर्ट} (\cref{ch:g11:nucleus-radioactivity}) में नापी जाती \omterm{rem:g11:nucleus-radioactivity:dose}{मात्रा} रखती है, जो ऊतक के
प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} जमा हुई ऊर्जा को हर विकिरण की विनाशकता से
तौलती है: \qty{5e8}{Bq} का टेक्नीशियम-स्कैन रोज़मर्रा की बात है, पर साँस में गए
$\alpha$ उत्सर्जक के उससे कहीं कम \omterm{def:g11:nucleus-radioactivity:activity}{बेक्रेल} भी नहीं;
और \unit{Bq} को \unit{Sv} में बदलना विश्वविद्यालय की सामग्री है।
\end{remark}

\section{तिथि-निर्धारण: अनुपात से समय पढ़ना}

\omterm{def:g11:mechanical-energy:dissipation}{क्षय} का नियम उलटा चलाइए और वह घड़ी बन जाता है: बचा हुआ अंश नापिए, और चरघातांक
बीता हुआ समय बता देता है।

\begin{method}[रेडियो-कार्बन तिथि-निर्धारण]\label{met:g12:radioactive-decay:carbon}
\begin{enumerate}
  \item \emph{जीवित:} जीवित ऊतक वायुमंडल से कार्बन का लेन-देन करता रहता है
        और उसका कार्बन-14 अनुपात ढोता है --- $10^{12}$ में एक \omterm{def:g11:fundamental-interactions:ladder}{परमाणु},
        यानी प्रति ग्राम कार्बन प्रति मिनट $\mathcal A_0 = 13.6$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय}।
  \item \emph{मृत्यु घड़ी चालू कर देती है:} आवक रुक जाती है, और अनुपात $T_{1/2} = \num{5730}$
        वर्ष के साथ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} होने लगता है।
  \item प्रतिदर्श की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} $\mathcal A$ प्रति ग्राम कार्बन नापिए;
        $\mathcal A = \mathcal A_0 e^{-\lambda t}$ उलटकर $t = \frac{1}{\lambda}\ln(\mathcal A_0/\mathcal A)
        = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$ बन जाता है।
\end{enumerate}
\end{method}

\begin{example}[गुफ़ा में एक चूल्हा]\label{ex:g12:radioactive-decay:hearth}
किसी दबे हुए चूल्हे का कोयला प्रति ग्राम प्रति मिनट $5.0$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} देता है:
$\mathcal A_0/\mathcal A = 13.6/5.0 = 2.72$, इसलिए $t = \num{5730} \times \ln(2.72)/\ln 2 \approx \num{8.3e3}$ वर्ष। वह आग आठ हज़ार वर्ष पहले बुझी थी --- और लकड़ी ही
उसका अभिलेख है।
\end{example}

\begin{example}[चट्टानों की घड़ियाँ]\label{ex:g12:radioactive-decay:rocks}
कार्बन-14 कोई \num{50000} वर्ष से आगे चुप हो जाता है; और भूविज्ञान उससे धीमी
घड़ियाँ रखता है। \emph{पोटैशियम--आर्गन}: पोटैशियम-40 ($T_{1/2} = \num{1.25e9}$ वर्ष) आर्गन में
\omterm{def:g11:mechanical-energy:dissipation}{क्षय} होता है, जो पिघले लावा से निकल भागती गैस है पर चट्टान के जमते ही
फँस जाती है --- यानी जमना घड़ी को शून्य कर देता है। \emph{यूरेनियम--सीसा}:
यूरेनियम-238 उस शृंखला का सिरा है जो स्थिर सीसा-206 (\cref{ch:g11:nucleus-radioactivity}) पर
ख़त्म होती है, और किसी क्रिस्टल में सीसे-और-यूरेनियम का अनुपात उसकी तिथि दे
देता है। दोनों घड़ियाँ सबसे पुराने खनिजों ($\approx \num{4.4e9}$ वर्ष) पर सहमत हैं, और
उल्कापिंडों के रास्ते सौर-मंडल की उम्र पर भी।
\end{example}

\begin{remark}[शृंखलाएँ, और तहख़ाने की गैस]\label{rem:g12:radioactive-decay:radon}
यूरेनियम और सीसे के बीच शृंखला रेडियम से होकर गुज़रती है, फिर रेडॉन-222 से, जो
एक रेडियोसक्रिय \emph{गैस} है ($T_{1/2} = 3.8$ दिन)। ग्रेनाइट और मिट्टी में जन्मी वह
तहख़ानों में रिसकर जमा होती जाती है; और उसकी $\alpha$-उत्सर्जक पुत्रियाँ
फेफड़ों के भीतर \omterm{def:g11:mechanical-energy:dissipation}{क्षय} करती हैं --- यानी अधिकांश लोगों की सबसे बड़ी प्राकृतिक
\omterm{rem:g11:nucleus-radioactivity:dose}{मात्रा}, और घटाने में सबसे सस्ती भी: तहख़ाने की खिड़की खोल दीजिए।
\end{remark}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:1}
आयोडीन-131 का $\lambda = \qty{1.0e-6}{s^{-1}}$ है। उसकी \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} सेकंडों में, फिर दिनों में
निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:2}
रेडॉन-222 का $T_{1/2} = 3.8$ दिन है। \unit{s^{-1}} में $\lambda$ निकालिए, फिर \num{1.0e10} नाभिकों
के प्रतिदर्श की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:3}
प्रतिस्थापन से जाँचिए कि $N(t) = N_0\,e^{-\lambda t}$ $\dd N/\dd t = -\lambda N$ और $N(0) = N_0$ को संतुष्ट करता है; फिर
$N(T_{1/2})/N_0$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:4}
सीज़ियम-137 के स्रोत ($T_{1/2} = 30$ वर्ष) की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} आज \qty{4.0e5}{Bq} है।
$90$ वर्ष बाद, फिर $300$ वर्ष बाद उसकी \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:5}
किसी टेक्नीशियम-प्रतिदर्श के लिए $t$ के सामने $\ln \mathcal A$ का अर्ध-लघुगणकीय
आलेख $(0, 20.00)$ और $(\qty{10}{hours}, 18.85)$ से होकर जाती सीधी रेखा है। $\lambda$ (\unit{h^{-1}} और \unit{s^{-1}}
में) तथा $T_{1/2}$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:6}
एक ग्राम रेडियम-226 ($T_{1/2} = 1600$ वर्ष; मोलर द्रव्यमान \qty{226}{g/mol}; $N_A = \qty{6.02e23}{mol^{-1}}$): नाभिकों
की संख्या निकालिए, फिर $\lambda$, फिर \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} --- यानी इतिहास का पहला
सक्रियता-मात्रक, क्यूरी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:7}
दो विद्यार्थी एक ही \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} के $20$ और \num{2.0e20} नाभिकों के
प्रतिदर्श बंद कर देते हैं। एक \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} बाद हर एक क्या अपेक्षा करता है?
किसका पूर्वानुमान भरोसे लायक़ है? (दोनों के लिए सापेक्ष उतार-चढ़ाव $1/\sqrt n$ का
अनुमान लगाइए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:8}
किसी शरीर में लगभग \qty{140}{g} पोटैशियम होता है, जिसका $0.012\%$ पोटैशियम-40
($T_{1/2} = \num{1.25e9}$ वर्ष, मोलर द्रव्यमान \qty{40}{g/mol}) है। पोटैशियम-40 नाभिकों की संख्या
निकालिए, फिर उनकी \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता}; और शरीर के कुल \qty{8000}{Bq} से तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:9}
किसी रोगी को टेक्नीशियम-99m ($T_{1/2} = 6.0$ घंटे) का \qty{500}{MBq} दिया जाता है। $15$
घंटे बाद \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} निकालिए, फिर उसके \qty{1}{MBq} से नीचे गिरने का समय।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:10}
एक गड्ढे का कोयला प्रति ग्राम कार्बन प्रति मिनट $3.4$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} दिखाता है; और
दूसरे की हड्डी जीवित दर का $30\%$ (प्रति ग्राम प्रति मिनट $\mathcal A_0 = 13.6$, $T_{1/2} = \num{5730}$
वर्ष)। दोनों प्रतिदर्शों की तिथि निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:radioactive-decay:11}
किसी तहख़ाने को रेडॉन-222 की एक खेप भीतर रखकर बंद कर दिया जाता है ($T_{1/2} = 3.8$
दिन)। कितने समय बाद उस खेप की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} गिरकर $1\%$ रह जाती है? असली
तहख़ाने दशकों तक रेडियोसक्रिय बने रहते हैं: ताज़ी रेडॉन कहाँ से आती है, और
हवादारी एक बार की नहीं, \omterm{prop:g12:work-and-energy:equilibria}{स्थायी} क्यों होनी चाहिए?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:12}
$N = N_0 e^{-\lambda t}$ से दिखाइए कि $t = T_{1/2}\ln(N_0/N)/\ln 2$। किसी हड्डी में उसके जीवित कार्बन-14 का $5.0\%$
बचा है: उसकी तिथि निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:13}
जिस क्रिस्टल ने जमते समय कोई आर्गन नहीं फँसाया था, उसमें हर \omterm{def:g11:mechanical-energy:dissipation}{क्षय} हुआ
पोटैशियम-40 नाभिक अपनी जगह एक आर्गन \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} छोड़ जाता है।
दिखाइए कि $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$; किसी चट्टान में अनुपात $3.0$ मिलता है: उसकी उम्र निकालिए
($T_{1/2} = \num{1.25e9}$ वर्ष)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:14}
गीगर गणित्र $t = 0$, $60$, $120$, $180$, $240$ \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} पर प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड}
$400$, $225$, $127$, $71$, $40$ गिनतियाँ देता है। हर समय पर $\ln \mathcal A$
निकालिए, रेखा में बैठना जाँचिए, और $\lambda$ तथा $T_{1/2}$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:radioactive-decay:15}
किसी एक नाभिक के $n$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} तक बचे रहने की प्रायिकता $1/2^{\,n}$
है। ये प्रायिकताएँ निकालिए: (क) कोई एक दिया हुआ नाभिक $10$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}
तक बचा रहे; (ख) देखे जा रहे सभी $8$ नाभिक एक \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} तक बचे रहें।
(ग) \num{1000} नाभिकों में से $3$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} बाद कितने अपेक्षित हैं --- और क्या
नियम बता सकता है कि \emph{कौन-से}? नियम क्या बताता है, इस पर निष्कर्ष लिखिए।
\end{exercise}

\section{समस्या: चीज़ों की उम्र}

\begin{problem}[{सप्ताहांत समस्या --- चीज़ों की उम्र: संग्रहालय की प्रयोगशाला
कार्बन-घड़ी अंशांकित करती है, एक मूर्ति और एक पपाइरस की तिथि निकालती है, सीखती
है कि कोई घड़ी अपने डायल से आगे क्यों नहीं पढ़ती, और एक ज्वालामुखीय चट्टान को
एक अरब वर्ष पीछे धकेल देती है}]
\label{pb:g12:radioactive-decay:1}
संग्रहालय की तिथि-प्रयोगशाला में एक सप्ताह। आँकड़े: जीवित ऊतक प्रति ग्राम
कार्बन प्रति मिनट $\mathcal A_0 = 13.6$ कार्बन-14 \omterm{def:g11:mechanical-energy:dissipation}{क्षय} दिखाता है; $T_{1/2} = \num{5730}$ वर्ष (कार्बन-14),
\num{1.25e9} वर्ष (पोटैशियम-40); $1$ वर्ष $= \qty{3.156e7}{s}$; $N_A = \qty{6.02e23}{mol^{-1}}$; कार्बन का मोलर
द्रव्यमान \qty{12}{g/mol}; और गणित्र प्रति ग्राम प्रति मिनट लगभग $0.2$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} तक ही
सुलझा पाते हैं।

\medskip
\noindent\textbf{भाग I --- सोमवार: घड़ी का अंशांकन।}
\begin{enumerate}
  \item जीवित ऊतक में कार्बन-14 का अनुपात अचर क्यों रहता है, और मृत्यु पर
        बदलता क्या है?
  \item कार्बन-14 के लिए \unit{s^{-1}} में $\lambda$ निकालिए।
  \item प्रतिस्थापन से जाँचिए कि $N_0 e^{-\lambda t}$ $\dd N/\dd t = -\lambda N$ को हल कर देता है।
  \item $\mathcal A_0$ से जीवित ऊतक में प्रति ग्राम कार्बन कार्बन-14 के
        परमाणुओं की संख्या निकालिए।
  \item एक ग्राम में कार्बन के कितने \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} होते हैं? इससे कार्बन-14
        का अनुपात निकालिए --- और उसकी तुलना ``$10^{12}$ में एक'' से कीजिए।
  \item दिखाइए कि नापी गई \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} $\mathcal A$ किसी प्रतिदर्श की तिथि
        $t = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$ बता देती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- मंगलवार: मूर्ति और पपाइरस।}
\begin{enumerate}[resume]
  \item लकड़ी की एक मूर्ति प्रति ग्राम प्रति मिनट $9.1$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} देती है। लकड़ी
        की तिथि निकालिए।
  \item वह तिथि ठीक-ठीक किस घटना पर निशान लगाती है --- तराशे जाने पर, या
        किसी और बात पर? इससे कौन-सी सावधानी निकलती है?
  \item मिस्र का एक पपाइरस प्रति ग्राम प्रति मिनट $10.4$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} देता है। उसकी
        तिथि निकालिए। क्या बाईस सदी पहले का कोई लेखक इसमें बैठता है?
  \item \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} प्रति मिनट $\pm 0.2$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} तक नापी जाती है। $\Delta t \approx \Delta\mathcal A/(\lambda \mathcal A)$ का
        उपयोग करके पपाइरस की उम्र की अनिश्चितता का अनुमान लगाइए।
  \item किसी व्यापारी का ``प्राचीन'' चर्म-पत्र प्रति ग्राम प्रति मिनट $13.4$
        \omterm{def:g11:mechanical-energy:dissipation}{क्षय} देता है। फ़ैसला?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- गुरुवार: घड़ी की सीमाएँ।}
\begin{enumerate}[resume]
  \item कितनी अर्धायुओं के बाद किसी प्रतिदर्श की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता} गणित्रों की
        प्रति मिनट प्रति ग्राम $0.2$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} वाली सीमा से नीचे गिर जाती है? वह
        कितनी उम्र है?
  \item सबसे अच्छी प्रयोगशालाएँ लगभग दस \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} तक पहुँचती हैं।
        रेडियो-कार्बन तिथि-निर्धारण का व्यावहारिक क्षितिज वर्षों में क्या है?
  \item डायनासोर की हड्डी लगभग \num{6.6e7} वर्ष पुरानी है: यह कार्बन-14 की कितनी
        \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} है? प्रश्न 4 का उपयोग करते हुए बताइए कि कितनी अर्धायुओं के बाद
        उस ग्राम भर कार्बन-14 का \emph{एक} \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} भी नहीं बचता।
        ``डायनासोर की कार्बन-तिथि'' पर निष्कर्ष लिखिए।
  \item इसके बजाय डायनासोर की चट्टान की तिथि कौन-सा \omterm{def:g11:nucleus-radioactivity:notation}{न्यूक्लाइड} दे
        सकता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- शुक्रवार: ज्वालामुखीय चट्टान।}
\begin{enumerate}[resume]
  \item आर्गन गैस है और पोटैशियम ठोस का वफ़ादार निवासी: समझाइए कि लावा के जमने
        से पोटैशियम--आर्गन घड़ी शून्य क्यों हो जाती है।
  \item यह मानते हुए कि हर \omterm{def:g11:mechanical-energy:dissipation}{क्षय} हुआ पोटैशियम-40 नाभिक एक फँसा हुआ
        आर्गन \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} छोड़ जाता है, दिखाइए कि $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$।
  \item संग्रहालय के जीवाश्म के नीचे की चट्टान $N_{\mathrm{Ar}}/N_{\mathrm K} = 0.60$ दिखाती है। चट्टान की
        तिथि निकालिए।
  \item क्या कार्बन-14 इस चट्टान की, या पोटैशियम--आर्गन उस मूर्ति की तिथि दे
        सकता था? हर घड़ी की काम की खिड़की अर्धायुओं में बताइए।
  \item शुक्रवार की रिपोर्ट, एक वाक्य में: इस सप्ताह मिली तीनों उम्रें, और वह
        एक नियम जिसने उन सबको पढ़ लिया।
\end{enumerate}
\end{problem}
