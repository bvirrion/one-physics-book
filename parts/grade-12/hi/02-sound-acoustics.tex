\chapter{ध्वनि और ध्वनिकी}\label{ch:g12:sound-acoustics}

गिटार का निचला तार छेड़िए: कमरा एक गहरे A से भर जाता है, फिर भी आपके कान तक
दबी और खिंची हवा के सिवा कुछ नहीं पहुँचा। यह अध्याय उसी ढर्रे के पीछे चलता है
--- उसकी चाल, आवृत्ति कैसे \omterm{def:g12:sound-acoustics:pitch-loudness}{तारत्व} बनती है और शक्ति कैसे
\omterm{def:g12:sound-acoustics:pitch-loudness}{प्रबलता}, और एक ही सुर हर वाद्य पर अलग क्यों लगता है।

\section{ध्वनि एक दाब-तरंग है}

\begin{definition}[ध्वनि तरंग]\label{def:g12:sound-acoustics:sound-wave}
\emph{ध्वनि तरंग}\index{ध्वनि तरंग} यांत्रिक दाब-तरंग है
(\cref{ch:g12:mechanical-waves}): कोई कंपित सतह चलते हुए \emph{संपीडन}\index{संपीडन} (हलका अधिदाब)
और \emph{विरलन}\index{विरलन} (न्यूनदाब) पैदा करती है। हवा के पिंड सफ़र की
दिशा के \emph{अनुदिश} दोलन करते हैं (\omterm{def:g10:signals-and-waves:sound}{ध्वनि}
\emph{अनुदैर्घ्य}\index{अनुदैर्घ्य तरंग} है), और माध्यम ज़रूरी है: निर्वात
की बरनी में रखी घंटी चुप हो जाती है (\cref{ch:g10:signals-and-waves})।
\end{definition}

\begin{proposition}[ध्वनि की चाल]\label{prop:g12:sound-acoustics:speed}
\index{ध्वनि की चाल}%
\omterm{def:g10:signals-and-waves:sound}{ध्वनि} \qty{20}{\degreeCelsius} पर हवा में लगभग \qty{340}{m/s}, पानी में \qty{1500}{m/s} और इस्पात में
\qty{5000}{m/s} चलती है: माध्यम जितना कड़ा, चाल उतनी तेज़। \admitted
\end{proposition}

\begin{remark}[गरम हवा तेज़ होती है]\label{rem:g12:sound-acoustics:temperature}
ये नापे हुए मान हैं; इनका पूर्वानुमान वर्ष 1 के खंड में लगाया गया है।
\omterm{def:g10:signals-and-waves:sound}{ध्वनि} गरम हवा में तेज़ हो जाती है (\qty{0}{\degreeCelsius} पर \qty{331}{m/s}) --- और इसीलिए
हॉल गरम होते ही फूँक वाले वाद्य सुर से उतर जाते हैं।
\end{remark}

\begin{example}[ध्वनियों के नाप]\label{ex:g12:sound-acoustics:wavelengths}
संबंध $\lambda = v/f$ हर \omterm{def:g10:signals-and-waves:sound}{ध्वनि} का नाप बता देता है: हवा में \qty{20}{Hz} की गड़गड़ाहट
$340/20 = \qty{17}{m}$ तक फैलती है --- यानी एक इमारत जितनी --- और \qty{20}{kHz} की सरसराहट \qty{1.7}{cm}
में समा जाती है; \omterm{def:g10:orders-of-magnitude:unit}{मीटर} भर वाली ध्वनियाँ दरवाज़ों और कोनों के चारों ओर आसानी से
मुड़ जाती हैं।
\end{example}

\section{तारत्व, प्रबलता और कान}

\begin{definition}[तारत्व और प्रबलता]\label{def:g12:sound-acoustics:pitch-loudness}
\emph{तारत्व}\index{तारत्व} आवृत्ति का \omterm{def:g11:color-light-sources:perception}{बोध} है: $f$ जितनी
अधिक, सुर उतना ऊँचा; सुर A \qty{440}{Hz} का होता है, और $f$ दुगुनी करने पर सुर
एक सप्तक ऊपर चढ़ जाता है। \emph{प्रबलता}\index{प्रबलता} आयाम का
\omterm{def:g11:color-light-sources:perception}{बोध} है: दाब का दोलन जितना प्रबल, \omterm{def:g10:signals-and-waves:sound}{ध्वनि} उतनी ऊँची; प्रवर्धन
प्रबलता बदलता है, तारत्व नहीं।
\end{definition}

\begin{remark}[कान की खिड़की]\label{rem:g12:sound-acoustics:audible}
सुनने की सीमा लगभग \qty{20}{Hz} से \qty{20}{kHz} तक है (\cref{ch:g10:signals-and-waves}); उससे नीचे
\omterm{def:g10:signals-and-waves:ultrasound}{अवश्रव्य} है और ऊपर \omterm{def:g10:signals-and-waves:ultrasound}{पराश्रव्य} --- चमगादड़ \qty{50}{kHz} पर, चिकित्सा
जाँच के यंत्र मेगाहर्ट्ज़ पर। पियानो केवल बीच का हिस्सा छूता है:
\qtyrange{27.5}{4186}{Hz}.
\end{remark}

\section{ध्वनि तीव्रता और डेसिबल पैमाना}

\begin{definition}[ध्वनि तीव्रता]\label{def:g12:sound-acoustics:intensity}
\emph{ध्वनि तीव्रता}\index{ध्वनि तीव्रता} $I$ वह ध्वनिक शक्ति है
जो तरंग के सामने रखे एक वर्ग \omterm{def:g10:orders-of-magnitude:unit}{मीटर} से गुज़रती है: $I = P/S$,
जिसे \unit{W/m^2} में नापते हैं। सबसे मंद सुनाई देती \omterm{def:g10:signals-and-waves:sound}{ध्वनि} लगभग $I_0 = \qty{1.0e-12}{W/m^2}$ की
होती है; और पीड़ा \qty{1}{W/m^2} के आसपास शुरू होती है।
\end{definition}

\begin{proposition}[व्युत्क्रम-वर्ग क्षीणन]\label{prop:g12:sound-acoustics:inverse-square}
\index{व्युत्क्रम-वर्ग नियम}%
हर दिशा में समान रूप से $P$ शक्ति विकिरित करता छोटा स्रोत दूरी
$r$ पर $I = P/(4\pi r^2) \propto 1/r^2$ देता है: दूरी दुगुनी करने पर तीव्रता चौथाई रह जाती है।
\end{proposition}

\begin{proof}
पूरी $P$ शक्ति क्षेत्रफल $4\pi r^2$ वाले गोले से गुज़रती है; भाग दे
दीजिए।
\end{proof}

\begin{definition}[ध्वनि स्तर]\label{def:g12:sound-acoustics:level}
कान दस की बारह घातों तक फैला है, इसलिए तीव्रताएँ लघुगणकीय रूप में
लिखी जाती हैं। $I$ का \emph{ध्वनि स्तर}\index{ध्वनि स्तर} है
\[ L = 10 \log_{10}(I/I_0), \qquad I_0 = \qty{1.0e-12}{W/m^2}, \]
जिसे \emph{डेसिबल}\index{डेसिबल} (\unit{dB}) में नापते हैं: श्रवण-देहली
\qty{0}{dB}, पीड़ा \qty{120}{dB} के आसपास।
\end{definition}

\begin{proposition}[डेसिबल का गणित]\label{prop:g12:sound-acoustics:db-arithmetic}
$I$ को $k$ से गुणा करने पर $10\log_{10}k$ \omterm{def:g12:sound-acoustics:level}{डेसिबल} जुड़ जाते हैं। विशेष
रूप से $\times 10$ \qty{10}{dB} जोड़ता है; $\times 2$ $10\log_{10}2 \approx \qty{3.0}{dB}$ जोड़ता है; $N$ असंबद्ध एक
जैसे स्रोत \emph{तीव्रताएँ} जोड़ते हैं, स्तर कभी नहीं, और $L_1 + 10\log_{10}N$ देते हैं;
और छोटे स्रोत से दूरी दुगुनी करने पर ($I$ चौथाई) लगभग \qty{6.0}{dB} घट जाते हैं।
\end{proposition}

\begin{proof}
$10\log_{10}(kI/I_0) = 10\log_{10}(I/I_0) + 10\log_{10}k$; दूरी के लिए: \cref{prop:g12:sound-acoustics:inverse-square}, $k = 1/4$।
\end{proof}

\begin{method}[डेसिबल में काम करना]\label{met:g12:sound-acoustics:db}
\begin{enumerate}
  \item $L = 10\log_{10}(I/I_0)$; और उलटा: $I = I_0 \times 10^{L/10}$।
  \item स्तर कभी मत जोड़िए: तीव्रताओं में बदलिए, जोड़िए, फिर वापस बदलिए ---
        या हर दुगुनेपन पर $+\qty{3}{dB}$, हर दस गुने पर $+\qty{10}{dB}$।
  \item छोटे स्रोत से दूरी हर बार दुगुनी करने का दाम \qty{6}{dB} है।
\end{enumerate}
\end{method}

\begin{remark}[श्रवण-हानि]\label{rem:g12:sound-acoustics:damage}
सुनने की शक्ति का नुक़सान एक \omterm{rem:g11:nucleus-radioactivity:dose}{मात्रा} है: \qty{85}{dB} आठ घंटे तक सुरक्षित
है, और उसके ऊपर हर \qty{3}{dB} --- यानी तीव्रता का दुगुनापन --- सुरक्षित समय आधा
कर देता है; और मरी हुई रोम-कोशिकाएँ फिर नहीं उगतीं।
\end{remark}

\section{गुणता और संनादी}

\begin{definition}[संनादी]\label{def:g12:sound-acoustics:harmonics}
आवृत्ति $f_1$ --- यानी \emph{मूल स्वर}\index{मूल स्वर} --- की
कोई \omterm{def:g10:signals-and-waves:period}{आवर्ती} \omterm{def:g10:signals-and-waves:sound}{ध्वनि} आम तौर पर आवृत्तियों $f_n = n f_1$, $n = 1, 2, 3, \dots$ पर
ज्याओं का योग होती है, जिन्हें उसके \emph{संनादी}\index{संनादी} कहते हैं।
जो \omterm{def:g10:signals-and-waves:sound}{ध्वनि} केवल अपने मूल स्वर तक सिमटी हो वह \emph{शुद्ध
स्वर}\index{शुद्ध स्वर} है, जैसा स्वरित्र द्विभुज का।
\end{definition}

\begin{definition}[वर्णक्रम और गुणता]\label{def:g12:sound-acoustics:timbre}
किसी \omterm{def:g10:signals-and-waves:sound}{ध्वनि} का \emph{वर्णक्रम}\index{वर्णक्रम} उसके
संनादियों का स्तंभ चित्र है: हर $f_n$ पर एक स्तंभ, और ऊँचाई उस
\omterm{def:g12:sound-acoustics:harmonics}{संनादी} की \omterm{def:g12:sound-acoustics:pitch-loudness}{प्रबलता}। कान $f_1$ को \omterm{def:g12:sound-acoustics:pitch-loudness}{तारत्व} के रूप में सुनता
है और प्रबलताओं के \emph{मिश्रण} को \emph{गुणता}\index{गुणता} के रूप में ---
यानी \omterm{def:g10:signals-and-waves:sound}{ध्वनि} के रंग के रूप में: एक ही सुर बजाते वाद्यों का $f_1$ एक
ही होता है, फ़र्क़ केवल उनके वर्णक्रमों में रहता है।
\end{definition}

\end{example}

\section{कंपित तार}

\begin{definition}[अप्रगामी तरंग]\label{def:g12:sound-acoustics:standing-wave}
तार के जकड़े सिरों के बीच बंद तरंग \emph{अप्रगामी
तरंग}\index{अप्रगामी तरंग} में बैठ सकती है: यानी अपनी जगह होता दोलन, जिसमें
निश्चल बिंदु --- \emph{निस्पंद}\index{निस्पंद} --- और अधिकतम झूल वाले बिंदु
--- \emph{प्रस्पंद}\index{प्रस्पंद} --- एक के बाद एक आते हैं। अब कुछ चलता
नहीं; तार एक जमे हुए ढर्रे में कंपन करता है।
\end{definition}

\begin{proposition}[तार के मोड]\label{prop:g12:sound-acoustics:string-modes}
दोनों सिरों पर जकड़ा लंबाई $L$ का तार, जिस पर चाल $v$ की
तरंगें चलती हैं, केवल उन्हीं ढर्रों में स्थिर कंपन करता है जिनमें
सिरों पर थोपे गए निस्पंदों के बीच पूरी संख्या $n$ भर
अर्ध-तरंगदैर्घ्य अँट जाएँ:
\[ \lambda_n = \frac{2L}{n}, \quad f_n = n\,\frac{v}{2L} = n f_1, \quad n = 1, 2, 3, \dots \]
सबसे निचला मोड $f_1 = v/(2L)$ \omterm{def:g12:sound-acoustics:harmonics}{मूल स्वर} है; बाक़ी ठीक उसके \omterm{def:g12:sound-acoustics:harmonics}{संनादी}
हैं। \admitted
\end{proposition}

\begin{remark}[ईमानदार उपपत्ति कहाँ रहती है]\label{rem:g12:sound-acoustics:honest}
बंद तरंगें ठीक यही मोड क्यों चुनती हैं, यह तरंग को उसके
परावर्तनों पर अध्यारोपित करने से निकलता है, और वह कलन वर्ष 1 के खंड में किया गया है।
ज्यामिति ही मन को मना देती है: दोनों सिरों को निश्चल रहना है, इसलिए पूरी
संख्या भर अर्ध-तरंगदैर्घ्य अँटने ही चाहिए।
\end{remark}

\begin{example}[वायलिन का A तार]\label{ex:g12:sound-acoustics:violin}
वायलिन के A तार पर $L = \qty{32.8}{cm}$ है और वह $f_1 = \qty{440}{Hz}$ बजाता है:
तरंगें उस पर $v = 2Lf_1 = 2 \times 0.328 \times 440 \approx \qty{289}{m/s}$ से दौड़ती हैं --- यह चाल \omterm{def:g10:inertia:inventory}{तनाव} और
द्रव्यमान से तय होती है, और खूँटी से सुर में बिठाई जाती है। उँगली दबाने पर
$L$ छोटा हो जाता है और हर $f_n$ एक साथ ऊपर चढ़ जाता है:
\omterm{def:g12:sound-acoustics:timbre}{वर्णक्रम} पूरा का पूरा सरक जाता है।
\end{example}

\begin{remark}[वाद्य अलग-अलग क्यों लगते हैं]\label{rem:g12:sound-acoustics:instruments}
तार को छेड़ा कैसे गया --- नोचकर, गज से, या चोट मारकर --- यह तय करता है कि हर
मोड कितना मौजूद है, और वाद्य का ढाँचा कुछ संनादियों को दूसरों से
अधिक प्रवर्धित करता है; छेड़ने का ढंग और ढाँचा मिलकर \omterm{def:g12:sound-acoustics:timbre}{वर्णक्रम} तय कर
देते हैं, इसलिए वही $f_1$ गिटार, वायलिन और पियानो से निकलकर हर बार अपने
वाद्य जैसा ही सुनाई देता है।
\end{remark}

\section{समस्या: वाद्य-कारीगर की कार्यशाला}

\begin{problem}[{सप्ताहांत समस्या --- वाद्य-कारीगर की कार्यशाला: गिटार का
एक तार पर्दा-दर-पर्दा बिछाया गया, उसके संनादी एक गुणता में सुर बिठाए गए, एक
संगीत-संध्या ख़तरे की रेखा से नीचे रखी गई, और लकड़ी में छिपे दोष की
पराश्रव्य खोज}]
\label{pb:g12:sound-acoustics:1}
एक वाद्य-कारीगर गिटार पूरा कर रहा है। उसके A तार की कंपित लंबाई $L = \qty{65.0}{cm}$ है
और उसे $f_1 = \qty{110}{Hz}$ बजाना है। हवा में \omterm{def:g10:signals-and-waves:sound}{ध्वनि} के लिए \qty{340}{m/s}
लीजिए, और $f_1 = v/(2L)$ (\cref{prop:g12:sound-acoustics:string-modes}) मान लीजिए, जहाँ $v$ तार पर तरंग
की चाल है।

\medskip
\noindent\textbf{भाग I --- पर्दे बिछाना।}
\begin{enumerate}
  \item तार पर तरंग की चाल $v$ निकालिए।
  \item \omterm{def:g12:sound-acoustics:harmonics}{मूल स्वर} का \emph{तार पर} \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए, फिर
        \emph{हवा में} \qty{110}{Hz} की \omterm{def:g10:signals-and-waves:sound}{ध्वनि} का \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य}।
        एक ही आवृत्ति पर वे अलग क्यों हैं?
  \item सप्तक (\qty{220}{Hz}): कौन-सी कंपित लंबाई उसे देती है? बारहवाँ पर्दा
        $L$ के किस अंश पर बैठना चाहिए?
  \item पंचम (\qty{165}{Hz}): कौन-सी कंपित लंबाई उसे देती है?
  \item हर पर्दा सुर को एक अर्धस्वर ऊपर चढ़ाता है, यानी गुणक $2^{1/12}$।
        लघुगणक से जाँचिए कि \qty{110}{Hz} और \qty{220}{Hz} के बीच कितने अर्धस्वर हैं;
        और पहले पर्दे पर लंबाई?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:g12:sound-acoustics:harmonics}{संनादी} और \omterm{def:g12:sound-acoustics:timbre}{गुणता}।}
\begin{enumerate}[resume]
  \item खुले A तार के पहले पाँच \omterm{def:g12:sound-acoustics:harmonics}{संनादी} गिनाइए।
  \item वायलिन-वादक का A \qty{440}{Hz} बजाता है। यह गिटार-तार का कौन-सा
        \omterm{def:g12:sound-acoustics:harmonics}{संनादी} है, और \omterm{def:g12:sound-acoustics:harmonics}{मूल स्वर} से उसका संगीत-अंतराल कितना
        है? और \qty{330}{Hz}?
  \item कारीगर पहले घोड़ी के पास तार छेड़ता है, फिर पर्दा-पट्टी के ऊपर: वही
        \omterm{def:g12:sound-acoustics:pitch-loudness}{तारत्व}, अलग रंग। वर्णक्रमों से समझाइए।
  \item छेड़ते समय बीचोंबीच हलका-सा छू देने से वहाँ एक \omterm{def:g12:sound-acoustics:standing-wave}{निस्पंद}
        बन जाता है। कौन-से \omterm{def:g12:sound-acoustics:harmonics}{संनादी} बचते हैं, और कौन-सा
        \omterm{def:g12:sound-acoustics:pitch-loudness}{तारत्व} सुनाई देता है?
  \item सिद्धांततः खुले तार के कितने \omterm{def:g12:sound-acoustics:harmonics}{संनादी} श्रव्य \omterm{def:g11:fundamental-interactions:strong}{परास} के
        भीतर पड़ते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- संगीत-संध्या।}
गिटार के पहले संगीत-कार्यक्रम में हर वाद्य छोटे स्रोत की तरह हर दिशा में
समान रूप से $P = \qty{6.0e-3}{W}$ ध्वनिक शक्ति विकिरित करता है।
\begin{enumerate}[resume]
  \item एक वाद्य से $r = \qty{3.0}{m}$ पर तीव्रता निकालिए।
  \item वहाँ का \omterm{def:g12:sound-acoustics:level}{ध्वनि स्तर} निकालिए।
  \item एक चौकड़ी बजती है: ऐसे चार वाद्य उसी जगह कितना स्तर देंगे? वह चार
        गुना स्तर क्यों नहीं है?
  \item बारह वाद्य बजते हैं: \qty{3.0}{m} पर स्तर क्या है?
  \item पूरी मंडली का स्तर \qty{85}{dB} तक गिरने के लिए श्रोता को कितना पीछे
        बैठना पड़ेगा?
  \item सुरक्षा-नियम: \qty{85}{dB} आठ घंटे तक सुरक्षित है, और हर \qty{+3}{dB} सुरक्षित
        समय आधा कर देता है। प्रश्न 14 वाला अगली पंक्ति का श्रोता कितनी देर
        सुरक्षित बैठ सकता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- लकड़ी में छिपा दोष।}
वार्निश करने से पहले कारीगर \qty{45}{mm} मोटे गरदन-गुटके को \qty{2.0}{MHz} के परीक्षक से
जाँचता है; इस लकड़ी में $v = \qty{5.0e3}{m/s}$ लीजिए।
\begin{enumerate}[resume]
  \item लकड़ी में \omterm{def:g10:signals-and-waves:ultrasound}{पराश्रव्य} का \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए। दोष
        खोजने वाली तरंग श्रव्य नहीं, \omterm{def:g10:signals-and-waves:ultrasound}{पराश्रव्य} ही क्यों होनी चाहिए?
  \item गुटके के पिछले फलक से \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} का \omterm{def:g12:mechanical-waves:delay}{विलंब} निकालिए।
  \item कोई \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} \qty{7.2}{\micro s} बाद लौटती है: दोष कितना गहरा है?
  \item कार्यशाला का पर्चा लिखिए: तार पर तरंग की चाल, बारहवें और
        पहले पर्दे की लंबाइयाँ, सुरक्षित श्रवण-दूरी, और दोष की गहराई।
\end{enumerate}
\end{problem}
