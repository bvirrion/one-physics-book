\chapter{तरंग के रूप में प्रकाश: विवर्तन और व्यतिकरण}\label{ch:g12:light-as-wave}

साबुन की झिल्ली को खिड़की की ओर तिरछा कीजिए: उस पर रंग सरकने लगते हैं, जबकि
साबुन या पानी में रंगीन कुछ भी नहीं है। किसी एक बाल पर \omterm{def:g11:color-light-sources:lamps}{लेज़र} डालिए:
पीछे की दीवार पर बाल भर की छाया नहीं, बल्कि एक चमकीली धारी दिखती है,
काली रेखाओं से कटी हुई और बाल से हज़ारों गुना चौड़ी। सीधी चलती किरणें
(\cref{ch:g10:refraction}) इनमें से कोई भी दृश्य नहीं समझातीं। यह अध्याय प्रकाश को पूरी
तरंग का दर्जा देता है --- शिखर से शिखर तक आधा माइक्रोमीटर --- और फिर
उस विचार को भुनाता है: एक \omterm{def:g11:color-light-sources:lamps}{लेज़र} और एक नंगी दीवार मिलकर बाल, \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य}
और CD की लीकें, सब नाप देंगे।

\section{प्रकाश एक तरंग है}

\begin{definition}[प्रकाश तरंगें]\label{def:g12:light-as-wave:lightwave}
\omterm{def:g10:light-spectra:white}{एकवर्णी प्रकाश} --- एक शुद्ध रंग, जिसे कोई प्रिज़्म आगे नहीं बाँट सकता
(\cref{ch:g10:light-spectra}) --- एक \omterm{def:g10:signals-and-waves:period}{आवर्ती} \emph{प्रकाश तरंग}\index{प्रकाश तरंग} है: वह
निर्वात को $c = \qty{3.00e8}{m/s}$ से पार करती है और उसकी पहचान उसकी आवृत्ति $f$ या
उसके \emph{निर्वात में तरंगदैर्घ्य}\index{तरंगदैर्घ्य!प्रकाश का} $\lambda = c/f$ से
होती है, जो \cref{ch:g12:mechanical-waves} की शिखर-से-शिखर दूरी है। आँख लगभग \qty{400}{nm} (बैंगनी) से
\qty{800}{nm} (लाल) तक प्रतिक्रिया करती है; \omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} पूरे \omterm{def:g11:fundamental-interactions:strong}{परास}
को मिला देता है।
\end{definition}

\begin{example}[परिमाण की कोटियाँ]\label{ex:g12:light-as-wave:orders}
हीलियम--नियॉन \omterm{def:g11:color-light-sources:lamps}{लेज़र} $\lambda = \qty{633}{nm} =
\qty{6.33e-7}{m}$ का लाल प्रकाश छोड़ता है, इसलिए $f = c/\lambda \approx \qty{4.7e14}{Hz}$: इतनी
तेज़ गिनती कोई इलेक्ट्रॉनिकी नहीं कर सकती --- इस अध्याय में हर
\omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} ज्यामिति से नापा जाएगा। नाप का अंदाज़ा: एक बाल (लगभग
\qty{70}{\micro m}) सौ \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} चौड़ा है। और तरंगित क्या होता है? हवा
नहीं: एक विद्युत और एक \omterm{def:g11:magnetic-fields:bfield}{चुंबकीय क्षेत्र}, जिनका ईमानदार वर्णन वर्ष 2 के
खंड में है; नीचे केवल \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} मायने रखता है।
\end{example}

\section{विवर्तन}

\begin{definition}[विवर्तन]\label{def:g12:light-as-wave:diffraction}
जब कोई तरंग अपने \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} जितने ही नाप $a$ के छिद्र या
अवरोध से मिलती है, तो वह उस क्षेत्र में भी फैल जाती है जिसे किरणें अँधेरा छोड़
देतीं। यही फैलाव \emph{विवर्तन}\index{विवर्तन} है --- $\lambda/a$ बहुत छोटा हो तो
नगण्य, और $a$ के $\lambda$ की ओर सिकुड़ते ही हावी।
\end{definition}

\begin{proposition}[फैलाव की कोणीय अर्ध-चौड़ाई]\label{prop:g12:light-as-wave:halfwidth}
चौड़ाई $a$ की झिरी, जिस पर \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} $\lambda < a$ का प्रकाश पड़ रहा
हो, प्रकाश को आपतन-दिशा के चारों ओर कोणीय अर्ध-चौड़ाई $\theta$ (रेडियन) के भीतर
फैला देती है; और दूरी $D$ पर रखा परदा $L$ चौड़ी केंद्रीय चमकीली पट्टी
दिखाता है, यानी $D \tan\theta \approx D\,\theta$ का दुगुना (छोटा कोण):
\[
\theta \approx \frac{\lambda}{a}, \qquad L = \frac{2\lambda D}{a}.
\]
\end{proposition}
\admitted

\begin{remark}[सूत्र को उलटा पढ़िए]\label{rem:g12:light-as-wave:honest}
झिरी जितनी \emph{सँकरी}, फैलाव उतना \emph{चौड़ा} --- यानी किरण-भविष्यवाणी का
ठीक उलटा। ईमानदार व्युत्पत्ति (झिरी भर से फिर उत्सर्जित तरंगिकाओं का योग)
वर्ष 2 के खंड में है; यहाँ यह सूत्र नापने का औज़ार भर है।
\end{remark}

\begin{example}[झिरी से गुज़रता लेज़र]\label{ex:g12:light-as-wave:slit}
लाल \omterm{def:g11:color-light-sources:lamps}{लेज़र}, $\lambda = \qty{633}{nm}$; झिरी $a = \qty{0.10}{mm}$; परदा $D = \qty{2.00}{m}$ पर। तब $\theta = \num{6.33e-7} /
\num{1.0e-4} = \qty{6.3e-3}{rad}$ --- यानी
डिग्री का तिहाई, पर दीवार पर केंद्रीय पट्टी $L = 2 \times \num{6.33e-7} \times 2.00 /
\num{1.0e-4} \approx \qty{2.5}{cm}$ चौड़ी है, और उसके दोनों ओर
काली रेखाएँ तथा मंद पार्श्व-पट्टियाँ हैं। झिरी आधी कीजिए और पट्टी दुगुनी हो
जाती है।
\end{example}

\begin{proposition}[अवरोध भी झिरी की तरह विवर्तित करता है]\label{prop:g12:light-as-wave:hair}
चौड़ाई $d$ का पतला अपारदर्शी अवरोध --- कोई तार, कोई बाल --- आपतित
पुंज के बाहर ठीक वही ढर्रा बनाता है जो चौड़ाई $d$ की झिरी बनाती (बाबिने का
सिद्धांत, जो वर्ष~2 के खंड में निकाला गया है)।
\end{proposition}
\admitted

\begin{method}[बहुत छोटी चौड़ाई नापना]\label{met:g12:light-as-wave:hairwidth}
\begin{enumerate}
  \item बाल को \omterm{def:g11:color-light-sources:lamps}{लेज़र} के पुंज के आगे तानिए; परदा नापी हुई दूरी
        $D$ (कुछ \omterm{def:g10:orders-of-magnitude:unit}{मीटर}) पर रखिए।
  \item केंद्रीय पट्टी की चौड़ाई $L$ नापिए, एक काली रेखा से दूसरी तक।
  \item चूँकि $L = 2\lambda D/d$: $d = 2\lambda D/L$ --- बाल जितना महीन, ढर्रा उतना बड़ा।
\end{enumerate}
\end{method}

\begin{remark}[विवर्तन हर यंत्र की सीमा है]\label{rem:g12:light-as-wave:limits}
प्रकाश कैमरे, दूरदर्शी या आँख में चौड़ाई $a$ के छिद्र से घुसता है, इसलिए
सबसे उत्तम लेंस भी किसी बिंदु को लगभग $\lambda/a$ कोणीय नाप के धब्बे में
धुँधला कर देता है: इससे पास के ब्योरे आपस में मिल जाते हैं। इसीलिए दृश्य
प्रकाश का सूक्ष्मदर्शी $\lambda \approx
\qty{0.5}{\micro m}$ से बहुत नीचे कुछ नहीं सुलझा पाता (जीवाणु हाँ,
विषाणु नहीं); और दूरदर्शियों के दर्पण इसी कारण चौड़े बनाए जाते हैं।
\end{remark}

\section{व्यतिकरण}

\begin{definition}[कलासंबद्ध स्रोत, पथांतर]\label{def:g12:light-as-wave:coherent}
दो तरंग-स्रोत \emph{कलासंबद्ध}\index{कलासंबद्ध स्रोत} तब हैं जब उनकी
आवृत्ति एक हो और वे क़दम-से-क़दम मिलाकर कंपन करें --- व्यवहार में, जब
वे एक ही तरंग को बाँटकर बनाई गई दो प्रतियाँ हों। जिस बिंदु $M$ तक
दोनों पहुँचती हों, उसके लिए \emph{पथांतर}% \index{पथांतर} $\delta = S_2M - S_1M$ होता
है।
\end{definition}

\begin{proposition}[व्यतिकरण की शर्तें]\label{prop:g12:light-as-wave:conditions}
जहाँ \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} $\lambda$ की दो \omterm{def:g12:light-as-wave:coherent}{कलासंबद्ध} तरंगें
एक-दूसरे पर चढ़ती हैं वहाँ वे जुड़ जाती हैं --- वे
\emph{व्यतिकरण}\index{व्यतिकरण} करती हैं:
\begin{itemize}
  \item $\delta = k\lambda$ ($k$ पूर्णांक): क़दम मिले हुए, एक-दूसरे को बढ़ाते हुए ---
        \emph{संपोषी}\index{संपोषी व्यतिकरण} \omterm{prop:g12:light-as-wave:conditions}{व्यतिकरण} (चमकीला);
  \item $\delta = (k + \tfrac12)\lambda$: शिखर पर गर्त, एक-दूसरे को काटते हुए ---
        \emph{विनाशी}\index{विनाशी व्यतिकरण} \omterm{prop:g12:light-as-wave:conditions}{व्यतिकरण} (अँधेरा)।
\end{itemize}
\end{proposition}

\begin{proof}
\omterm{def:g12:mechanical-waves:periodic}{आवर्ती तरंग} को पूरे तरंगदैर्घ्यों जितना खिसकाने से कुछ नहीं
बदलता: शिखर शिखरों पर ही चढ़ते हैं। और आधा \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} और खिसकाने पर
शिखर गर्त पर आ बैठता है: बराबर तरंगें एक-दूसरे को काट देती हैं।
\end{proof}

\begin{example}[यंग का दो-झिरी वाला प्रयोग]\label{ex:g12:light-as-wave:young}
किसी अपारदर्शी पट्टी में मिलीमीटर के अंश भर की दूरी पर दो महीन झिरियाँ
$S_1$, $S_2$ बनाइए और दोनों पर एक ही \omterm{def:g11:color-light-sources:lamps}{लेज़र} डालिए: यही दो
\omterm{def:g12:light-as-wave:coherent}{कलासंबद्ध स्रोत} हैं। दो-एक \omterm{def:g10:orders-of-magnitude:unit}{मीटर} दूर रखे परदे पर उनका
अध्यारोपण धारीदार दिखता है: बीच में $\delta = 0$, चमकीला; परदे पर ऊपर चढ़ते जाइए
और $\delta$ बढ़कर $\lambda/2$ (अँधेरा), $\lambda$ (चमकीला)\,\dots\ हो जाता है ---
यानी वहाँ अँधेरा, जहाँ प्रकाश में प्रकाश जोड़ने पर वह कट जाता है।
\end{example}

\begin{definition}[फ़्रिंज और फ़्रिंज-चौड़ाई]\label{def:g12:light-as-wave:fringes}
ढर्रे की बारी-बारी आती चमकीली और काली पट्टियाँ उसके \emph{व्यतिकरण
फ़्रिंज}\index{व्यतिकरण फ़्रिंज} हैं; और लगातार दो चमकीले केंद्रों के बीच की
दूरी $i$ \emph{फ़्रिंज-चौड़ाई}% \index{फ़्रिंज-चौड़ाई} कहलाती है।
\end{definition}

\begin{proposition}[फ़्रिंज-चौड़ाई]\label{prop:g12:light-as-wave:interfringe}
$a$ की दूरी पर रखी झिरियों और $D \gg a$ दूर रखे परदे के लिए, परदे के जिस
बिंदु की केंद्र $O$ से दूरी $x$ हो उसका \omterm{def:g12:light-as-wave:coherent}{पथांतर} $\delta \approx a\,x/D$ होता
है (ज्यामिति मान ली गई); और \cref{prop:g12:light-as-wave:conditions} से चमकीले फ़्रिंज $x_k = k\,\lambda D/a$ पर बैठते हैं,
बराबर दूरी पर, जिनकी \omterm{def:g12:light-as-wave:fringes}{फ़्रिंज-चौड़ाई} है
\[
i = \frac{\lambda D}{a}.
\]
$\lambda = \qty{633}{nm}$, $a = \qty{0.20}{mm}$, $D =
\qty{2.0}{m}$ के साथ: $i = \qty{6.3}{mm}$ --- यानी यह व्यवस्था $\lambda$ को
$D/a = 10^4$ गुना बड़ा कर देती है। इसके पीछे की ज्यामिति वर्ष~2 के खंड में निकाली
गई है।
\end{proposition}
\admitted

\begin{method}[तरंगदैर्घ्य नापना]\label{met:g12:light-as-wave:lambda}
\begin{enumerate}
  \item अनजान \omterm{def:g11:color-light-sources:lamps}{लेज़र} को ज्ञात दूरी $a$ वाली दो झिरियों से
        गुज़ारिए; परदा नापी हुई दूरी $D$ पर रखिए।
  \item $10$ फ़्रिंज-चौड़ाइयों का फैलाव नापिए; $10$ से भाग
        दीजिए: \omterm{def:g12:light-as-wave:fringes}{फ़्रिंज-चौड़ाई} $i$, और पैमाने की त्रुटि दस गुना
        सिकुड़ जाती है।
  \item फिर $\lambda = i\,a/D$।
\end{enumerate}
\end{method}

\begin{remark}[दो लैंप कभी व्यतिकरण क्यों नहीं करते]\label{rem:g12:light-as-wave:lamps}
एक ही दीवार पर पड़ते दो मेज़-लैंप कोई फ़्रिंज नहीं बनाते: हर लैंप छोटी,
असंबद्ध तरंग-रेलें छोड़ता है जिनका आपसी समय हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} अरबों बार
यादृच्छिक ढंग से उछलता है, इसलिए फ़्रिंज भी उतनी ही तेज़ी से खिसकते हैं और
आँख उन्हें औसत करके एकसार प्रकाश बना देती है। \omterm{prop:g12:light-as-wave:conditions}{व्यतिकरण} के लिए
कलासंबद्धता चाहिए --- यानी एक ही तरंग को दो में बाँटना,
जैसा यंग की झिरियाँ करती हैं; और \omterm{def:g11:color-light-sources:lamps}{लेज़र} इसी का आधुनिक शॉर्टकट है, एक
साफ़ लंबी तरंग।
\end{remark}

\section{व्यतिकरण से रँगे रंग}

\begin{definition}[रंगदीप्ति]\label{def:g12:light-as-wave:iridescence}
जो रंग देखने के कोण के साथ बदल जाएँ --- साबुन के बुलबुले, तेल की परत, मोर के
पंख, CD की पीठ --- वे \emph{रंगदीप्ति}% \index{रंगदीप्ति} हैं: वहाँ कोई
वर्णक नहीं, बल्कि \omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} में \omterm{prop:g12:light-as-wave:conditions}{व्यतिकरण} है, जो कुछ
तरंगदैर्घ्यों को बढ़ा देता है और बाक़ी को काट देता है।
\end{definition}

\begin{example}[साबुन की झिल्ली और तेल की परत]\label{ex:g12:light-as-wave:soap}
साबुन की झिल्ली पर पड़ता प्रकाश कुछ उसके अगले फलक से परावर्तित होता है और कुछ
पिछले से: यही दो \omterm{def:g12:light-as-wave:coherent}{कलासंबद्ध} प्रतियाँ हैं, जिनका \omterm{def:g12:light-as-wave:coherent}{पथांतर}
मोटाई और देखने के कोण से तय होता है। हर मोटाई कुछ तरंगदैर्घ्यों को
बढ़ाकर लौटाती है और बाक़ी को काट देती है: इसलिए झिल्ली जैसे-जैसे बहकर
पतली होती जाती है, उसका परावर्तित रंग सरकता जाता है। गीली सड़क पर तेल
की परत भी यही खेल खेलती है।
\end{example}

\begin{remark}[एकवर्णी बनाम श्वेत प्रकाश]\label{rem:g12:light-as-wave:white}
\omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} में हर \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} अपने ही $i = \lambda D/a$ के साथ अपने
यंग-फ़्रिंज खींचता है: बीच में सारे रंग चमकीले हैं (एक श्वेत फ़्रिंज); कुछ
फ़्रिंज बाहर जाते ही ढर्रे एक-दूसरे से हट जाते हैं --- पहले रंगदीप्त किनारे,
फिर धुँधलापन। \omterm{def:g10:light-spectra:white}{एकवर्णी प्रकाश} अंतहीन साफ़ फ़्रिंज देता है: यहाँ हर
सटीक माप में \omterm{def:g11:color-light-sources:lamps}{लेज़र} इसीलिए है।
\end{remark}

\section{समस्या: प्रकाश से नापना}

\begin{problem}[{सप्ताहांत समस्या --- प्रकाश से नापना: एक लेज़र, एक फ़ीता और
एक नंगी दीवार मिलकर माइक्रोमीटर की कार्यशाला बन जाते हैं, जो एक झिरी, एक
मानव बाल, एक अनजान तरंगदैर्घ्य और दो चाँदी-सी चकतियों की लीकें नाप देती है}]
\label{pb:g12:light-as-wave:1}
आपकी किट: एक लाल \omterm{def:g11:color-light-sources:lamps}{लेज़र} (लेबल के अनुसार $\lambda = \qty{633}{nm}$), अनजान
\omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} का एक हरा \omterm{def:g11:color-light-sources:lamps}{लेज़र}, एक अंशांकित झिरी $a = \qty{100}{\micro m}$,
$a' =
\qty{0.250}{mm}$ अंतराल वाली एक दो-झिरी, एक नापने का फ़ीता, दो बाल, एक CD और एक DVD।

\medskip
\noindent\textbf{भाग I --- ज्ञात झिरी पर अंशांकन।}
लाल \omterm{def:g11:color-light-sources:lamps}{लेज़र}, झिरी, दीवार $D = \qty{3.00}{m}$ पर।
\begin{enumerate}
  \item विवर्तित पुंज की कोणीय अर्ध-चौड़ाई $\theta$ निकालिए।
  \item दिखाइए कि केंद्रीय पट्टी की चौड़ाई $L = 2\lambda D/a$ है; उसे निकालिए।
  \item आप $L = \qty{3.7}{cm}$ नापते हैं: सापेक्ष विचलन निकालिए। क्या विधि प्रमाणित हो
        गई?
  \item दूसरी झिरी आधी चौड़ी है, \qty{50}{\micro m}: उसका $L$ बताइए। झिरी की चौड़ाई
        और ढर्रे की चौड़ाई को जोड़ने वाला नियम लिखिए।
  \item प्रश्न 2 के क़दम $\tan\theta \approx \theta$ का औचित्य बताइए: इस कोण पर $\theta$ और
        $\tan\theta$ की तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- बाल।} वही \omterm{def:g11:color-light-sources:lamps}{लेज़र} और वही दीवार; झिरी की
जगह अब तना हुआ एक बाल।
\begin{enumerate}[resume]
  \item कौन-सा माना हुआ परिणाम बाल के लिए भी झिरी वाला सूत्र चलने देता है,
        और वह कहता क्या है?
  \item केंद्रीय पट्टी $L = \qty{5.4}{cm}$ चौड़ी है: बाल का व्यास $d$ निकालिए।
  \item आपका पैमाना $L$ को $\pm\qty{0.2}{cm}$ तक ही पढ़ पाता है: इससे $d$ के लिए
        निकलता \omterm{def:g11:fundamental-interactions:strong}{परास} $d \pm \Delta d$ के रूप में दीजिए।
  \item आपके मित्र का बाल $L = \qty{7.6}{cm}$ देता है: उसका व्यास? किसका बाल महीन है, और
        महीन होने का अर्थ चौड़ा ढर्रा क्यों है?
  \item क्या मेज़-लैंप \omterm{def:g11:color-light-sources:lamps}{लेज़र} की जगह ले सकता था? दो कारण दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- हरे \omterm{def:g11:color-light-sources:lamps}{लेज़र} का \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य}।}
हरा \omterm{def:g11:color-light-sources:lamps}{लेज़र}, दो-झिरी, दीवार $D = \qty{2.00}{m}$ पर।
\begin{enumerate}[resume]
  \item दोनों झिरियों पर \emph{एक ही} \omterm{def:g11:color-light-sources:lamps}{लेज़र} डालना क्यों ज़रूरी है?
  \item \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} चाहे जो हो, बीच वाला फ़्रिंज चमकीला ही क्यों
        होता है?
  \item दस फ़्रिंज-चौड़ाइयाँ \qty{4.26}{cm} में फैलती हैं। $i$ दीजिए,
        और एक साथ दस नापने का कारण भी।
  \item हरे \omterm{def:g11:color-light-sources:lamps}{लेज़र} का \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए।
  \item निर्माता का लेबल कहता है $\qty{532 \pm 10}{nm}$: क्या यह संगत है? आप कौन-सा रंग
        अपेक्षित करते हैं?
\end{enumerate}
