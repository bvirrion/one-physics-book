\chapter{नाभिकीय ऊर्जा: विखंडन, संलयन, $E=mc^2$}\label{ch:g12:nuclear-energy}

पेंसिल की रबड़ जितनी ईंधन-गोली किसी घर की साल भर की बिजली चुका देती है; वही सेवा
कोयले से लेनी हो तो तीन टन लगते हैं। और ऊपर आसमान में सूर्य दिन के उजाले का दाम
ख़ुद ग़ायब होकर चुकाता है --- हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} अपने चालीस लाख टन। यह अध्याय दोनों सौदों
के पीछे का बहीखाता खोलता है: द्रव्यमान स्वयं ऊर्जा है, और नाभिक (\cref{ch:g11:nucleus-radioactivity})
वे जगहें हैं जहाँ वह भुनाया जाता है।

\section{द्रव्यमान ही ऊर्जा है}

\begin{theorem}[द्रव्यमान--ऊर्जा तुल्यता]\label{thm:g12:nuclear-energy:emc2}\index{द्रव्यमान--ऊर्जा तुल्यता}
द्रव्यमान $m$ का पिंड केवल अस्तित्व में रहने भर से ऊर्जा रखता है:
\[
E = m c^2, \qquad c = \qty{3.00e8}{m/s}.
\]
और जब भी कोई निकाय द्रव्यमान $\Delta m$ खोता है, वह ऊर्जा $\Delta m\, c^2$ छोड़ देता
है --- क्रियाविधि चाहे जो हो।
\end{theorem}
\admitted

\begin{remark}[आइंस्टाइन, 1905]\label{rem:g12:nuclear-energy:einstein}
यह आइंस्टाइन का संबंध है, जिसे यहाँ मान लिया गया है; वह आता कहाँ से है, यह इसी
वर्ष को समापित करते विशिष्ट-आपेक्षिकता वाले अध्याय का काम है (\cref{ch:g12:special-relativity})। दर पर
ध्यान दीजिए: $c^2 = \qty{9.00e16}{J/kg}$ --- यानी द्रव्यमान में गोल कर देने भर की चूक ऊर्जा
की दृष्टि से एक ख़ज़ाना है।
\end{remark}

\begin{definition}[इलेक्ट्रॉन-वोल्ट और परमाणु द्रव्यमान मात्रक]\label{def:g12:nuclear-energy:units}
नाभिक के नाप के दो मात्रक।
\emph{इलेक्ट्रॉन-वोल्ट}\index{इलेक्ट्रॉन-वोल्ट}, यानी वह ऊर्जा जो कोई
इलेक्ट्रॉन \qty{1}{V} पार करते हुए पाता है (\cref{ch:g11:circuits-and-power}): $\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{1}{MeV} = \qty{1.60e-13}{J}$ --- रासायनिक
बंध कुछ \unit{eV} का सौदा करते हैं, और नाभिकीय अभिक्रियाएँ
मेगाइलेक्ट्रॉन-वोल्ट का। और \emph{एकीकृत परमाणु द्रव्यमान
मात्रक}\index{एकीकृत परमाणु द्रव्यमान मात्रक}, यानी कार्बन-12 के \omterm{def:g11:fundamental-interactions:ladder}{परमाणु}
के द्रव्यमान का बारहवाँ भाग: $\qty{1}{u} = \qty{1.66054e-27}{kg}$, जिसका ऊर्जा-तुल्य \qty{931.5}{MeV} है ---
$\qty{1}{u} = \qty{931.5}{MeV}/c^2$।
\end{definition}

\begin{example}[सोया हुआ ख़ज़ाना]\label{ex:g12:nuclear-energy:fortune}
किसी भी चीज़ का एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम}, पूरा-का-पूरा बदल दिया जाए, तो $\qty{9.00e16}{J}$ है
--- यानी किसी बड़े (\qty{1}{GW}) शक्ति-संयंत्र का तीन साल का उत्पादन। 1905
से पहले इस पर किसी की नज़र इसलिए नहीं गई कि साधारण भौतिकी द्रव्यमान को मुश्किल
से छूती है: एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} कोयला जलाने (\qty{3.0e7}{J}) से वह माइक्रोग्राम के
तिहाई भर हलका होता है। हम देखेंगे कि नाभिक हज़ार में एक भाग तक हिलते हैं: तब भी
छोटा, पर रसायन से दस लाख गुना।
\end{example}

\section{द्रव्यमान क्षति}

\begin{definition}[द्रव्यमान क्षति और बंधन ऊर्जा]\label{def:g12:nuclear-energy:defect}
नाभिक ${}^{A}_{Z}\mathrm{X}$ (द्रव्यमान $m$) तौलिए, फिर उसके $Z$ प्रोटॉन और
$A - Z$ न्यूट्रॉन अलग-अलग तौलिए: भाग पूरे से \emph{भारी} निकलते हैं। यह अंतर
\[
\Delta m = Z\,m_p + (A - Z)\,m_n - m > 0,
\qquad m_p = \qty{1.00728}{u},\; m_n = \qty{1.00866}{u},
\]
\emph{द्रव्यमान क्षति}\index{द्रव्यमान क्षति} है; और \emph{बंधन
ऊर्जा}\index{बंधन ऊर्जा} $E_b = \Delta m\,c^2$ वह ऊर्जा है जो नाभिक को
तोड़कर मुक्त न्यूक्लियॉनों में बाँटने के लिए चाहिए --- और उतनी ही
ऊर्जा जोड़ते समय छूटती है। बँधा हुआ नाभिक अपने भागों से
\emph{नीचे} बैठता है।
\end{definition}

न्यूक्लियॉनों से $E_b$ नीचे पड़ता है --- उसे जोड़ने पर $E_b$ छूटता
है, तोड़ने पर उतना ही लगता है।}
\end{omfigure}

\begin{example}[हीलियम-4]\label{ex:g12:nuclear-energy:helium}
${}^{4}_{2}\mathrm{He}$ नाभिक का $m = \qty{4.00151}{u}$ है; और उसके भाग: $2 \times 1.00728 + 2 \times 1.00866 = \qty{4.03188}{u}$। इसलिए $\Delta m = \qty{0.03037}{u}$ और
$E_b = 0.03037 \times 931.5 \approx
\qty{28.3}{MeV}$: यानी नाभिक अपने भागों से $0.75\%$ कम भारी है --- \cref{ex:g12:nuclear-energy:fortune} वाला वही
``हज़ार में एक भाग''।
\end{example}

\begin{method}[नाभिकीय अभिक्रिया का ऊर्जा-लेखा]\label{met:g12:nuclear-energy:balance}
\begin{enumerate}
  \item समीकरण संतुलित कीजिए: कुल $A$ और $Z$ संरक्षित (\cref{ch:g11:nucleus-radioactivity})।
  \item पहले वाले द्रव्यमान जोड़िए, फिर बाद वाले, \unit{u} में --- और पाँचों
        दशमलव रखिए।
  \item $\Delta m = m_{\text{पहले}} - m_{\text{बाद}}$, फिर $E = \Delta m\,(\text{u में}) \times \qty{931.5}{MeV}$, जो उत्पादों की \omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} बनकर बह जाती
        है; और ऋणात्मक होने का अर्थ है कि अभिक्रिया का दाम चुकाना पड़ेगा।
\end{enumerate}
\end{method}

\section{वह वक्र जो ब्रह्मांड चलाता है}

\begin{definition}[प्रति न्यूक्लियॉन बंधन ऊर्जा]\label{def:g12:nuclear-energy:pernucleon}
संसंजन की न्यायपूर्ण तुलना \emph{प्रति न्यूक्लियॉन बंधन ऊर्जा}\index{प्रति
न्यूक्लियॉन बंधन ऊर्जा} $E_b/A$ से होती है, यानी एक \omterm{def:g11:fundamental-interactions:ladder}{न्यूक्लियॉन} खींच
निकालने का औसत दाम: हीलियम-4 के लिए $28.3/4 \approx \qty{7.1}{MeV}$।
\end{definition}

\begin{proposition}[लोहे का शिखर, और नीचे उतरती दो सड़कें]\label{prop:g12:nuclear-energy:ironpeak}\index{लोहे का शिखर}
$A$ के सामने आलेखित करने पर \omterm{def:g12:nuclear-energy:pernucleon}{प्रति न्यूक्लियॉन बंधन ऊर्जा} हलके
नाभिकों में तेज़ी से चढ़ती है, लोहे के पास ${}^{56}_{26}\mathrm{Fe}$ पर प्रति \omterm{def:g11:fundamental-interactions:ladder}{न्यूक्लियॉन}
लगभग \qty{8.8}{MeV} के शिखर पर पहुँचती है, और फिर धीरे-धीरे फिसलकर यूरेनियम पर लगभग
\qty{7.6}{MeV} रह जाती है: यानी प्रकृति में सबसे कसकर बँधा नाभिक लोहा है। कोई
अभिक्रिया ऊर्जा ठीक तभी छोड़ती है जब उसके उत्पाद वक्र पर \emph{ऊपर}
बैठें; इसलिए दो उलटी रणनीतियाँ, दोनों की दोनों, फ़ायदे का सौदा हैं और दोनों लोहे
की ओर ही चलती हैं --- सबसे भारी नाभिकों को \emph{तोड़ना}, और सबसे हलकों को
\emph{जोड़ना}।
\end{proposition}
\admitted

\begin{remark}[शिखर क्यों]\label{rem:g12:nuclear-energy:whypeak}
\cref{ch:g11:fundamental-interactions} वाली वही रस्साकशी: प्रबल गोंद केवल छूते पड़ोसियों को बाँधती है, जबकि
\omterm{def:g11:fundamental-interactions:charge}{कूलॉम} प्रतिकर्षण पूरे नाभिक पर फैला रहता है --- इसलिए छोटे नाभिक
ज़्यादातर सतह हैं, बड़े ज़्यादातर प्रतिकर्षण, और लोहा दोनों का समझौता। ईमानदार
हिसाब विश्वविद्यालय के खंडों में रहता है।
\end{remark}

\section{विखंडन और शृंखला अभिक्रिया}

\begin{definition}[विखंडन]\label{def:g12:nuclear-energy:fission}
\emph{विखंडन}\index{विखंडन} किसी भारी नाभिक का दो मँझोले टुकड़ों में
टूट जाना है। यूरेनियम-235 \emph{विखंडनीय}\index{विखंडनीय} है: कोई धीमा
न्यूट्रॉन सोख लेने पर नाभिक इतना उत्तेजित हो जाता है कि दो टुकड़ों में
फट पड़ता है और दो या तीन ताज़े न्यूट्रॉन उगल देता है --- ${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to \text{दो टुकड़े} +
2\text{--}3\;{}^{1}_{0}\mathrm{n}$, जहाँ हर विखंडन
में जोड़ा बदलता रहता है।
\end{definition}

\begin{example}[एक विखंडन, तौला हुआ]\label{ex:g12:nuclear-energy:fissionbalance}
\cref{met:g12:nuclear-energy:balance} वाला एक बहुधा होने वाला रास्ता:
\[
{}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \longrightarrow
{}^{94}_{38}\mathrm{Sr} + {}^{140}_{54}\mathrm{Xe} + 2\,{}^{1}_{0}\mathrm{n}.
\]
पहले: $1.00866 + 234.99350 = \qty{236.00216}{u}$। बाद में: $93.89450 + 139.89200 + 2 \times 1.00866 = \qty{235.80382}{u}$। इसलिए $\Delta m = \qty{0.19834}{u}$ और $E \approx \qty{185}{MeV}$; और रेडियोसक्रिय
टुकड़ों के बाद के \omterm{def:g11:mechanical-energy:dissipation}{क्षय} भी गिन लें (\cref{ch:g12:radioactive-decay}) तो हर \omterm{def:g12:nuclear-energy:fission}{विखंडन} लगभग \qty{200}{MeV}
का पड़ता है --- जबकि कोयले का एक \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} कुछ \unit{eV} में मिल जाता है।
\end{example}

\begin{definition}[शृंखला अभिक्रिया और क्रांतिक द्रव्यमान]\label{def:g12:nuclear-energy:chain}
हर \omterm{def:g12:nuclear-energy:fission}{विखंडन} एक न्यूट्रॉन से जलता है और दो-तीन छोड़ जाता है: पर्याप्त बड़े
ढेले में वे दूसरे नाभिकों से टकराते हैं और अभिक्रिया ख़ुद को खिलाने लगती है ---
यानी \emph{शृंखला अभिक्रिया}\index{शृंखला अभिक्रिया}। छोटे ढेले में अधिकांश
न्यूट्रॉन पहले ही सतह से निकल भागते हैं; और शृंखला को टिकाए रखने वाली न्यूनतम
\omterm{rem:g11:nucleus-radioactivity:dose}{मात्रा} \emph{क्रांतिक द्रव्यमान}\index{क्रांतिक द्रव्यमान} है --- उससे नीचे
शृंखला फुस्स हो जाती है, और उससे ऊपर हर पीढ़ी कई गुना हो जाती है: यही बम का
सिद्धांत है।
\end{definition}

\begin{definition}[साधा हुआ रिएक्टर]\label{def:g12:nuclear-energy:reactor}
शक्ति-रिएक्टर शृंखला को ठीक एक न्यूट्रॉन प्रति \omterm{def:g12:nuclear-energy:fission}{विखंडन} पर
थामे रखता है। ईंधन: यूरेनियम-235 में मामूली रूप से समृद्ध की गई यूरेनियम की
गोलियाँ। \emph{मंदक}\index{मंदक} (आम तौर पर साधारण पानी) टक्करों से तेज़
विखंडन-न्यूट्रॉनों को धीमा कर देता है --- और धीमे न्यूट्रॉन अगला
\omterm{def:g12:nuclear-energy:fission}{विखंडन} भड़काने में कहीं बेहतर हैं; जबकि \emph{नियंत्रक छड़ें}\index{नियंत्रक
छड़} (बोरॉन, कैडमियम) न्यूट्रॉन निगल जाती हैं --- भीतर धकेलिए और शृंखला मर जाती
है, बाहर खींचिए और वह तेज़ हो जाती है। और टुकड़ों की उत्तेजना ऊष्मा बनती है, फिर
भाप, फिर बिजली (\cref{ch:g10:energy-conservation})।
\end{definition}

\begin{example}[एक कालीमिर्च बनाम एक ट्रक]\label{ex:g12:nuclear-energy:gram}
एक ग्राम यूरेनियम-235 में
$N = \num{1.0e-3} / (235 \times \num{1.66054e-27}) \approx \num{2.56e21}$
नाभिक होते हैं; हर एक $\qty{200}{MeV} = \qty{3.2e-11}{J}$ पर, सबको विखंडित कर देने से लगभग \qty{8.2e10}{J} मिलता
है --- यानी $\num{8.2e10} /
\num{3.0e7} \approx \qty{2.7e3}{kg}$ कोयले की \omterm{def:g10:energy-conservation:forms}{रासायनिक ऊर्जा}: एक कालीमिर्च बनाम तीन
टन।
\end{example}

\section{संलयन: सूर्य की अपनी आग}

\begin{definition}[संलयन]\label{def:g12:nuclear-energy:fusion}
\emph{संलयन}\index{संलयन} दो हलके नाभिकों का जुड़कर एक भारी और अधिक कसकर बँधा
नाभिक बन जाना है। सबसे पहुँच में आती अभिक्रिया हाइड्रोजन के भारी
समस्थानिकों ड्यूटीरियम और ट्रिटियम का ब्याह कराती है (\cref{ch:g11:nucleus-radioactivity}):
\[
{}^{2}_{1}\mathrm{H} + {}^{3}_{1}\mathrm{H} \longrightarrow
{}^{4}_{2}\mathrm{He} + {}^{1}_{0}\mathrm{n}.
\]
\end{definition}

\begin{example}[D--T, तौला हुआ]\label{ex:g12:nuclear-energy:dt}
पहले: $2.01355 + 3.01550 = \qty{5.02905}{u}$; बाद में: $4.00151 + 1.00866 = \qty{5.01017}{u}$; इसलिए $\Delta m = \qty{0.01888}{u}$ और $E \approx \qty{17.6}{MeV}$, जिसका चार-पाँचवाँ
हिस्सा न्यूट्रॉन पर जाता है --- \omterm{def:g12:nuclear-energy:fission}{विखंडन} से कम, पर $236$ के बजाय
पाँच न्यूक्लियॉनों से: यानी प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} ईंधन लगभग
\qty{3.4e14}{J}, जो \omterm{def:g12:nuclear-energy:fission}{विखंडन} का चार गुना है। और राख फिर वही हीलियम-4 (\cref{ex:g12:nuclear-energy:helium})
है।
\end{example}

\begin{definition}[कूलॉम अवरोध]\label{def:g12:nuclear-energy:barrier}
दोनों नाभिक धनात्मक हैं: छूने के लिए उन्हें पहले अपने विद्युत प्रतिकर्षण की
पहाड़ी चढ़नी पड़ती है (\cref{ch:g11:electric-gravitational-fields}) --- यानी \emph{कूलॉम अवरोध}\index{कूलॉम अवरोध}।
केवल \qty{1e8}{K} के आसपास ही टक्करें उसे पार करने लायक़ ऊर्जा ढोती हैं;
और तब ईंधन नंगे नाभिकों और इलेक्ट्रॉनों का \emph{प्लाज़्मा}\index{प्लाज़्मा}
बन चुका होता है। \omterm{def:g12:nuclear-energy:fission}{विखंडन} को यह चुंगी नहीं चुकानी पड़ती: उसका पलीता,
न्यूट्रॉन, उदासीन है और ठंडे में ही काम कर देता है।
\end{definition}

\begin{example}[सूर्य इसी पर चलता है]\label{ex:g12:nuclear-energy:sun}
सूर्य का \omterm{def:g10:refraction:fiber}{क्रोड} $\qty{1.5e7}{K}$ पर साधारण हाइड्रोजन को क़दम-दर-क़दम संलयित
करता है --- यानी \emph{प्रोटॉन--प्रोटॉन शृंखला}, जिसका शुद्ध असर $4\,{}^{1}_{1}\mathrm{H} \to {}^{4}_{2}\mathrm{He} + 2\,{}^{0}_{+1}\mathrm{e}
+ \text{विकिरण}$ है और
प्रति हीलियम लगभग \qty{26}{MeV}: यानी हाइड्रोजन के द्रव्यमान का $0.7\%$
ऊर्जा बनकर निकल जाता है। जिस धूप का \omterm{def:g12:sound-acoustics:timbre}{वर्णक्रम} हम \cref{ch:g10:light-spectra} में
पढ़ते हैं वह यही द्रव्यमान है, आठ मिनट की देर से पहुँचा हुआ; और हर तारा वक्र पर
लोहे की ओर चलकर ही चमकता है।
\end{example}

\begin{remark}[ITER: तारे को बोतल में भरना]\label{rem:g12:nuclear-energy:iter}
कोई ठोस \qty{1.5e8}{K} पर \omterm{def:g12:nuclear-energy:barrier}{प्लाज़्मा} थाम नहीं सकता; \emph{टोकामाक} उसे
\omterm{def:g11:magnetic-fields:bfield}{चुंबकीय क्षेत्र} में क़ैद कर लेता है (\cref{ch:g11:magnetic-fields}), यानी क्षेत्र
रेखाओं का ऐसा गोल-गप्पा जिसके अनुदिश आवेशित कण दीवार छुए बिना सर्पिल में चलते
रहते हैं; और ITER, जो अपनी तापन-शक्ति का दस गुना छोड़ने के लिए बनाया
गया है, इस समय का क़दम है। परिरोध टूट जाए तो \omterm{def:g12:nuclear-energy:barrier}{प्लाज़्मा} सेकंडों में ठंडा
होकर रुक जाता है: न कोई \omterm{def:g12:nuclear-energy:chain}{शृंखला अभिक्रिया}, न कोई \omterm{def:g12:nuclear-energy:chain}{क्रांतिक
द्रव्यमान} --- यहाँ कठिन काम आग रोकना नहीं, उसे जलाए रखना है।
\end{remark}

\begin{example}[ऊर्जा की सीढ़ी, प्रति किलोग्राम]\label{ex:g12:nuclear-energy:ladder}
प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} ईंधन ऊर्जा: कोयला, \qty{3.0e7}{J}; \omterm{def:g12:nuclear-energy:fission}{विखंडन}
से यूरेनियम-235, \qty{8.2e13}{J}; \omterm{def:g12:nuclear-energy:fusion}{संलयन} से ड्यूटीरियम--ट्रिटियम, \qty{3.4e14}{J}; और
पूरा रूपांतरण ($E = mc^2$), \qty{9.0e16}{J}। रसायन और नाभिक के बीच छह कोटियाँ हैं
--- और इसीलिए रिएक्टर में ईंधन ट्रक से भरा जाता है और कोयला-संयंत्र में
रेलगाड़ियों से।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:1}
बदलिए: (क) \qty{1}{eV} और \qty{1}{MeV} को \omterm{def:g11:work-of-force:work}{जूल} में; (ख) एक \omterm{def:g12:nuclear-energy:fission}{विखंडन} के

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:2}

प्रतिदिन $\qty{10}{kW.h} = \qty{3.6e7}{J}$ खींचता है --- पूरी तरह बदल दी गई वह डली कितने वर्ष उसे

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:3}
कार्बन-12 के नाभिक का द्रव्यमान \qty{11.99671}{u} है: उसकी \omterm{def:g12:nuclear-energy:defect}{द्रव्यमान

क्रिप्टॉन, $Z = 52$: टेल्यूरियम):

{}^{97}_{40}\mathrm{Zr} + {}^{A}_{Z}\mathrm{Y} + 2\,{}^{1}_{0}\mathrm{n}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:5}
\cref{prop:g12:nuclear-energy:ironpeak} का वक्र पढ़िए: हीलियम-4, कार्बन-12, लोहा-56 और यूरेनियम-235 के लिए
$E_b/A$ का अनुमान लगाइए। सबसे कसकर बँधा कौन है? कौन \omterm{def:g12:nuclear-energy:fission}{विखंडन} से

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:nuclear-energy:6}
ड्यूटरॉन ${}^{2}_{1}\mathrm{H}$ का द्रव्यमान \qty{2.01355}{u} है: \unit{MeV} में उसकी \omterm{def:g12:nuclear-energy:defect}{बंधन ऊर्जा}
निकालिए। रासायनिक बंध कुछ \unit{eV} ही थामता है --- नाभिकीय ``बंध'' उसे कितने
गुना से हरा देता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:nuclear-energy:7}
\omterm{def:g12:nuclear-energy:fission}{विखंडन} ${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to
{}^{144}_{56}\mathrm{Ba} + {}^{89}_{36}\mathrm{Kr} + 3\,{}^{1}_{0}\mathrm{n}$ तौलिए: द्रव्यमान \qty{234.99350}{u} (U), \qty{143.89220}{u} (Ba), \qty{88.89790}{u}
(Kr), \qty{1.00866}{u} (न्यूट्रॉन)। छूटी ऊर्जा, \unit{MeV} और \omterm{def:g11:work-of-force:work}{जूल} में?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:nuclear-energy:8}
एक और \omterm{def:g12:nuclear-energy:fusion}{संलयन}: ${}^{2}_{1}\mathrm{H} + {}^{2}_{1}\mathrm{H} \to
{}^{3}_{2}\mathrm{He} + {}^{1}_{0}\mathrm{n}$, जहाँ $m({}^{3}_{2}\mathrm{He}) = \qty{3.01493}{u}$। छूटी ऊर्जा निकालिए।
D--T से इतनी कम क्यों? (सोचिए कि हर उत्पाद वक्र पर कहाँ बैठता है।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:nuclear-energy:9}
हर एक पर कुछ पंक्तियों में समझाइए: (क) \omterm{def:g12:nuclear-energy:fission}{विखंडन} को कोई तापन क्यों नहीं
चाहिए जबकि \omterm{def:g12:nuclear-energy:fusion}{संलयन} लगभग \qty{1e8}{K} माँगता है; (ख) सूर्य ``केवल'' \qty{1.5e7}{K} पर
भी कैसे निभा लेता है (भीड़ का नाप, उसके सबसे तेज़ सदस्य); (ग) कोई
संलयन-संयंत्र \omterm{def:g12:nuclear-energy:chain}{शृंखला अभिक्रिया} की तरह बेक़ाबू क्यों नहीं हो
सकता।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:nuclear-energy:10}
सूर्य की शुद्ध अभिक्रिया चार प्रोटॉन ($m_p = \qty{1.00728}{u}$) को एक हीलियम-4

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:nuclear-energy:11}
पानी से मंदित किसी रिएक्टर में: (क) \omterm{def:g12:nuclear-energy:reactor}{मंदक} क्या करता है, और शृंखला को
उसकी ज़रूरत क्यों है? (ख) \omterm{def:g12:nuclear-energy:reactor}{नियंत्रक छड़ें}? (ग) ठंडा करने वाला पानी ही
\omterm{def:g12:nuclear-energy:reactor}{मंदक} \emph{है}: उसके निकल जाने से शृंखला तेज़ होने के बजाय घुट क्यों
जाती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:nuclear-energy:12}
बेक़ाबू शृंखला हर पीढ़ी पर दुगुनी हो जाती है। एक \omterm{def:g12:nuclear-energy:fission}{विखंडन} से चलकर एक ग्राम
यूरेनियम-235 के $\num{2.56e21}$ नाभिकों को विखंडित होने में लगभग कितनी पीढ़ियाँ लगेंगी
($2^{10} \approx 10^{3}$)? प्रति पीढ़ी लगभग \qty{10}{ns} पर यह कितना समय है --- और तब क्रांतिकता की
इतनी सावधानी से रखवाली क्यों की जाती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:nuclear-energy:13}
क्या लोहे को विखंडित करके ऊर्जा खोदी जा सकती है? ${}^{56}_{26}\mathrm{Fe} \to 2\,{}^{28}_{14}\mathrm{Si}$ तौलिए, जहाँ
$m(\mathrm{Fe}) = \qty{55.92066}{u}$, $m(\mathrm{Si}) = \qty{27.96924}{u}$। $\Delta m$ निकालिए और वक्र के पदों में निष्कर्ष लिखिए कि लोहा
नाभिकीय राख क्यों है: किसी काम का ईंधन नहीं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:nuclear-energy:14}
$m({}^{56}_{26}\mathrm{Fe}) = \qty{55.92066}{u}$ और $m({}^{235}_{92}\mathrm{U}) = \qty{234.99350}{u}$ से दोनों के लिए $E_b/A$ निकालिए ($m_p = \qty{1.00728}{u}$, $m_n = \qty{1.00866}{u}$)।
\omterm{def:g12:nuclear-energy:fission}{विखंडन} के टुकड़े प्रति \omterm{def:g11:fundamental-interactions:ladder}{न्यूक्लियॉन} लगभग \qty{8.5}{MeV} पर बैठते
हैं: दोनों स्तरों से एक \omterm{def:g12:nuclear-energy:fission}{विखंडन} की ऊर्जा का \emph{अनुमान}
लगाइए, और \cref{ex:g12:nuclear-energy:fissionbalance} से तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:nuclear-energy:15}
ITER \qty{500}{MW} की संलयन-शक्ति का लक्ष्य रखता है। प्रति D--T
अभिक्रिया \qty{17.6}{MeV} (और खपा हुआ ईंधन \qty{5.029}{u}) पर प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} अभिक्रियाएँ
निकालिए, फिर प्रतिदिन जला ईंधन। \qty{500}{MW} की कोयला-भट्ठी प्रतिदिन लगभग \qty{1.4e6}{kg}
जलाती है: तुलना कीजिए।
\end{exercise}

\section{समस्या: किसी शहर को बिजली देना}

\begin{problem}[{सप्ताहांत समस्या --- किसी शहर को बिजली देना: पाँच लाख लोगों
के लिए एक स्थिर गीगावाट, चार तरह से ख़रीदा हुआ --- यूरेनियम की गोलियाँ, कोयले
की रेलगाड़ियाँ, समुद्र के पानी का एक ताल, और ख़ुद सूर्य का पदार्थ --- और चारों
के नीचे छिपा वही एक किलोग्राम ग़ायब हुआ द्रव्यमान}]
\label{pb:g12:nuclear-energy:1}
$500\,000$ निवासियों का एक शहर स्थिर \omterm{prop:g11:circuits-and-power:power}{विद्युत शक्ति} $P = \qty{1.0}{GW}$ खींचता है;
और उसके संयंत्र, ईंधन चाहे जो हो, ऊष्मा को \omterm{def:g10:energy-conservation:efficiency}{दक्षता} $33\%$ (\cref{ch:g10:energy-conservation}) से
बिजली में बदलते हैं। आँकड़े: यूरेनियम-235 का एक \omterm{def:g12:nuclear-energy:fission}{विखंडन}, कुल मिलाकर
\qty{200}{MeV}; कोयला, \qty{3.0e7}{J/kg}; एक D--T \omterm{def:g12:nuclear-energy:fusion}{संलयन}, \qty{5.029}{u} ईंधन से \qty{17.6}{MeV};
$\qty{1}{u} = \qty{1.66054e-27}{kg}$; $\qty{1}{eV} = \qty{1.60e-19}{J}$; एक वर्ष $= \qty{3.16e7}{s}$; सूर्य: \qty{3.9e26}{W}, \qty{2.0e30}{kg}; और समुद्र के पानी
के एक घन \omterm{def:g10:orders-of-magnitude:unit}{मीटर} में लगभग \qty{33}{g} ड्यूटीरियम होता है।

\medskip
\noindent\textbf{भाग I --- विखंडन-संयंत्र।}
\begin{enumerate}
  \item रिएक्टर को कितनी तापीय शक्ति देनी होगी?
  \item एक \omterm{def:g12:nuclear-energy:fission}{विखंडन} की ऊर्जा \omterm{def:g11:work-of-force:work}{जूल} में लिखिए।
  \item शहर को चलाने के लिए प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कितने \omterm{def:g12:nuclear-energy:fission}{विखंडन} चाहिए?
  \item यूरेनियम-235 के एक नाभिक का द्रव्यमान ($A = 235$), फिर प्रति
        \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} खपा यूरेनियम-235?
  \item इससे एक वर्ष में खपा यूरेनियम-235 \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} में निकालिए।
  \item ईंधन $4.0\%$ यूरेनियम-235 तक समृद्ध किया गया है: एक वर्ष में कितना
        द्रव्यमान लगेगा, और क्या वह एक ट्रक पर आ जाएगा?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- कोयले का बहीखाता।}
\begin{enumerate}[resume]
  \item शहर एक वर्ष में जितनी \omterm{def:g10:energy-conservation:forms}{तापीय ऊर्जा} खपाता है, वह निकालिए।
  \item उतनी ऊर्जा कितने द्रव्यमान का कोयला देगा?
  \item एक वर्ष के लिए (कोयले का द्रव्यमान)/(यूरेनियम-235 का द्रव्यमान)
        अनुपात निकालिए, और उसे \cref{ex:g12:nuclear-energy:ladder} की सीढ़ी से जाँचिए।
  \item मालगाड़ी \qty{3.0e6}{kg} ढोती है: प्रति वर्ष और प्रतिदिन कितनी गाड़ियाँ? एक
        वाक्य में दोनों आपूर्ति-रेखाओं का चित्र खींचिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g12:nuclear-energy:fusion}{संलयन} का सपना।}
\begin{enumerate}[resume]
  \item एक D--T \omterm{def:g12:nuclear-energy:fusion}{संलयन} की ऊर्जा \omterm{def:g11:work-of-force:work}{जूल} में लिखिए।
  \item प्रति \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} D--T ईंधन छूटी ऊर्जा निकालिए; सीढ़ी
        से तुलना कीजिए।
  \item शहर के एक वर्ष के लिए कितना D--T ईंधन चाहिए? उसमें से कितना
        ड्यूटीरियम है (\qty{5.029}{u} का $2.014$)?
  \item उतना ड्यूटीरियम समुद्र के पानी के कितने आयतन में है? लगभग \qty{2.5e3}{m^3}
        वाले ओलंपिक ताल से तुलना कीजिए। (ट्रिटियम लीथियम से बनाया जाता है।)
  \item इस संयंत्र को न कोई \omterm{def:g12:nuclear-energy:chain}{क्रांतिक द्रव्यमान} डराता है न कोई
        \omterm{def:g12:nuclear-energy:chain}{शृंखला अभिक्रिया} --- क्यों? तो फिर कठिन काम क्या है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह तारा जो द्रव्यमान में चुकाता है।}
\begin{enumerate}[resume]
  \item $E = mc^2$ से सूर्य हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कितना द्रव्यमान बदल देता है?
  \item \emph{किसी भी} \qty{3.0}{GW} तापीय स्रोत को प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} और प्रति वर्ष
        कितना द्रव्यमान बदलना पड़ेगा? जाँचिए कि तीनों ईंधन यही एक द्रव्यमान
        सौंपते हैं।
  \item सूर्य ने अपने $\num{4.5e9}$ वर्षों में कुल कितना द्रव्यमान विकिरित कर दिया
        है, और वह सूर्य का कितना अंश है?
  \item \omterm{def:g12:nuclear-energy:fusion}{संलयन} सूर्य के द्रव्यमान का लगभग $0.07\%$ ही छू सकता है (उसके
        \omterm{def:g10:refraction:fiber}{क्रोड} का जलने लायक़ हिस्सा): तारे के रूप में उसकी कुल आयु और बाद में
        बचने वाली चीज़ का अनुमान लगाइए।
  \item लेखा तीन वाक्यों में बंद कीजिए: शहर का एक वर्ष कितने टन कोयला, कितने
        \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} यूरेनियम-235 और कितने \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} D--T ईंधन
        माँगता है --- और वह \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} द्रव्यमान जो हर परिदृश्य में
        शहर की रोशनी बन गया।
\end{enumerate}
\end{problem}
