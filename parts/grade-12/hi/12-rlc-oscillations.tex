\chapter{मुक्त विद्युत दोलन: RLC}\label{ch:g12:rlc-oscillations}

अटारी में पड़े किसी रेडियो का भारी डायल घुमाइए और स्टेशन सुई के सामने से क़तार
में गुज़रने लगते हैं: एक आवाज़, वायलिन, खरखराहट, फिर एक आवाज़। पैनल के पीछे न कोई
मोटर है न कोई कंप्यूटर --- बस एक \omterm{def:g11:magnetic-fields:solenoid}{कुंडली}, एक \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र}
जिसकी गुँथी हुई पट्टियों को डायल घुमाता है, और ऐंटेना के तार का एक टुकड़ा। यह
अध्याय पिछले अध्याय की दोनों ऊर्जा-टंकियों (\cref{ch:g12:rc-rl-circuits}) को आमने-सामने जोड़
देता है और अब तक बना सबसे तेज़ लोलक खोज निकालता है: आगे-पीछे झूलता आवेश,
\omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में लाखों बार तक, और लय वही जो डायल चुन ले।

\section{एक विद्युत लोलक: LC परिपथ}

\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} $C$ को $U_0$ के \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} तक आवेशित कीजिए --- तब उसकी
पट्टियों पर $Q_0 = C U_0$ के साथ $\pm Q_0$ रहता है --- और $t = 0$ पर उसे शून्य
प्रतिरोध वाली \emph{आदर्श} \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} $L$ पर जोड़ दीजिए। फंदे
में कहीं कोई स्रोत नहीं: इसके बाद जो कुछ होगा, परिपथ अपने आप करेगा।

आदर्श \omterm{def:g11:magnetic-fields:solenoid}{कुंडली}। $K$ बंद करते ही आवेश \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} से होकर छलक पड़ता
है --- और बदलाव से चिढ़ती \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} उसे रुकने नहीं देगी।}
\end{omfigure}

\begin{proposition}[LC समीकरण]\label{prop:g12:rlc-oscillations:equation}
\index{LC परिपथ}
आदर्श LC फंदे में ऊपरी पट्टी का आवेश $q(t)$ यह मानता है:
\[
L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0,
\qquad\text{अर्थात्}\qquad
\frac{d^2q}{dt^2} = -\frac{1}{LC}\,q .
\]
\end{proposition}

\begin{proof}
फंदे भर \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} जुड़कर शून्य हो जाते हैं (\cref{prop:g11:circuits-and-power:laws}): $u_L + u_C = 0$।
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} $u_C = q/C$ देता है और \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} $u_L = L\,di/dt$ (\cref{def:g12:rc-rl-circuits:inductor}), जहाँ
$i = dq/dt$ ऊपरी पट्टी पर पहुँचता आवेश गिनता है (\cref{prop:g12:rc-rl-circuits:cdudt}); अब प्रतिस्थापित कीजिए।
\end{proof}

\begin{theorem}[LC परिपथ के मुक्त दोलन]\label{thm:g12:rlc-oscillations:oscillation}
\index{विद्युत दोलन}
$t = 0$ पर $q(0) = Q_0$ और $i(0) = 0$ के साथ छोड़ने पर परिपथ दोलन करने लगता है:
\[
q(t) = Q_0 \cos\!\left(\frac{2\pi t}{T_0}\right),
\qquad
i(t) = -I_0 \sin\!\left(\frac{2\pi t}{T_0}\right),
\qquad
T_0 = 2\pi\sqrt{LC},
\]
और शिखर धारा $I_0 = 2\pi Q_0/T_0 = Q_0/\sqrt{LC}$ होती है।
\end{theorem}

\begin{proof}
प्रतिस्थापन से जाँचिए। अवकलन करने पर $i = dq/dt = -\frac{2\pi}{T_0} Q_0 \sin(2\pi t/T_0)$ --- यानी वही $i(t)$ जो कहा गया
--- और फिर $\frac{d^2q}{dt^2} = -(\frac{2\pi}{T_0})^2 q$, जो $-q/(LC)$ के ठीक तभी बराबर होता है जब $(2\pi/T_0)^2 = 1/(LC)$, यानी
$T_0 = 2\pi\sqrt{LC}$। और $q(0) = Q_0$, $i(0) = 0$। और यह कि \emph{कोई दूसरा} फलन इस समीकरण तथा इस
शुरुआत पर नहीं बैठता, यह ठीक स्प्रिंग की तरह मान लिया गया है (\cref{rem:g12:oscillators-and-time:uniqueness})।
\end{proof}

\begin{definition}[स्वाभाविक आवर्तकाल और आवृत्ति]\label{def:g12:rlc-oscillations:period}
किसी \omterm{prop:g12:rlc-oscillations:equation}{LC परिपथ} का \emph{स्वाभाविक आवर्तकाल}\index{स्वाभाविक
आवर्तकाल!LC परिपथ का} $T_0 = 2\pi\sqrt{LC}$ है; और उसकी \emph{स्वाभाविक
आवृत्ति}\index{स्वाभाविक आवृत्ति!LC परिपथ की} है
\[
f_0 = \frac{1}{T_0} = \frac{1}{2\pi\sqrt{LC}} .
\]
विमीय जाँच: $\qty{1}{H} = \qty{1}{V.s/A}$ और $\qty{1}{F} = \qty{1}{C/V}$, इसलिए $LC$ $\unit{s} \times \unit{C/A} = \unit{s^2}$ ढोता है --- यानी $\sqrt{LC}$
एक समय है, जैसा होना ही चाहिए।
\end{definition}

\begin{example}[पहली संख्याएँ]\label{ex:g12:rlc-oscillations:firstnumbers}
$C = \qty{1.0}{\micro F}$ के साथ $L = \qty{10}{mH}$: $\sqrt{LC} = \qty{1.0e-4}{s}$, इसलिए $T_0 = \qty{0.63}{ms}$ और $f_0 \approx \qty{1.6}{kHz}$ --- यानी एक सुनाई देता
स्वर। और $C = \qty{100}{pF}$ के साथ $L = \qty{1.0}{mH}$: $T_0 = \qty{2.0}{\micro s}$, $f_0 \approx \qty{0.50}{MHz}$ --- यानी एक रेडियो
आवृत्ति। लय केवल \emph{गुणनफल} $LC$ तय करता है।
\end{example}

\begin{remark}[चौथाई आवर्तकाल की चूक]\label{rem:g12:rlc-oscillations:quarter}
धारा आवेश से चौथाई \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} की \omterm{def:g12:oscillators-and-time:phase}{कला} पीछे रहती है: $q = \pm Q_0$ होने
पर $i = 0$ (पट्टियाँ भरी, बहाव पलटता हुआ) और $q = 0$ होने पर $i = \pm I_0$ (पट्टियाँ
ख़ाली, बहाव पूरे ज़ोर पर) --- यानी ठीक वही लोलक, जो छोरों पर निश्चल और तल पर
सबसे तेज़ रहता है।
\end{remark}

\section{ऊर्जा का वाल्ट्ज़}

\begin{proposition}[संचित ऊर्जा का संरक्षण]\label{prop:g12:rlc-oscillations:energy}
\index{ऊर्जा!LC परिपथ की}
आदर्श \omterm{prop:g12:rlc-oscillations:equation}{LC परिपथ} में कुल संचित ऊर्जा (\cref{prop:g12:rc-rl-circuits:energy})
\[
E \;=\; E_C + E_L \;=\; \frac{q^2}{2C} + \tfrac12\,L i^2
\;=\; \frac{Q_0^2}{2C} \;=\; \tfrac12\,L I_0^2
\]
अचर रहती है: $q = \pm Q_0$ होने पर पूरी की पूरी \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} में, और $i = \pm I_0$
होने पर पूरी की पूरी \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} में।
\end{proposition}

\begin{proof}
अवकलन कीजिए: \cref{prop:g12:rlc-oscillations:equation} से $\frac{dE}{dt} = \frac{q}{C}\frac{dq}{dt} +
Li\frac{di}{dt} = i\left(\frac{q}{C} + L\frac{d^2q}{dt^2}\right) = 0$। उसका मान $t = 0$ पर पढ़ा जाता है (पूरा
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र}), और फिर पूरी-धारा वाले क्षण $q = 0$ पर --- जिससे $\tfrac12 L I_0^2 = Q_0^2/2C$
मिलता है और ऊर्जा भर से $I_0 = Q_0/\sqrt{LC}$ फिर मिल जाता है।
\end{proof}

में दो बार पट्टियों से \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} तक और वापस छलकती है, जबकि योग $Q_0^2/2C$
पर टँका रहता है --- यानी \omterm{prop:g12:oscillators-and-time:spring-period}{स्प्रिंग--द्रव्यमान दोलित्र} का हूबहू चित्र,
बस नए नामों के साथ।}
\end{omfigure}

\begin{example}[वाल्ट्ज़, नापा हुआ]\label{ex:g12:rlc-oscillations:waltz}
$U_0 = \qty{10}{V}$ तक आवेशित $C = \qty{1.0}{\micro F}$, $L = \qty{10}{mH}$ पर जोड़ा गया: ऊर्जा $E = \tfrac12 C U_0^2 = \qty{5.0e-5}{J}$,
\omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} $T_0 = \qty{0.63}{ms}$, और स्विच बंद होने के चौथाई \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल}
(\qty{0.16}{ms}) बाद पहुँची शिखर धारा $I_0 = U_0\sqrt{C/L} = \qty{0.10}{A}$।
\end{example}

\section{असली परिपथ: अवमंदित दोलन}

हर असली \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} \omterm{def:g11:work-of-force:sign}{प्रतिरोधी} तार से लपेटी जाती है: इसलिए ईमानदार
फंदा श्रेणीक्रम वाला \emph{RLC} परिपथ है, और प्रतिरोध इस वाल्ट्ज़ पर
कर लगा देता है।

\begin{definition}[छद्म-आवर्ती और अनावर्ती व्यवस्थाएँ]\label{def:g12:rlc-oscillations:regimes}
\index{RLC परिपथ}\index{छद्म-आवर्ती व्यवस्था}\index{अनावर्ती व्यवस्था}
श्रेणीक्रम वाले RLC परिपथ में प्रतिरोधक जूल-तापन से ऊर्जा \omterm{def:g11:mechanical-energy:dissipation}{क्षय} कर
देता है (\cref{ch:g11:circuits-and-power}): इसलिए \omterm{def:g12:oscillators-and-time:oscillator}{मुक्त दोलन} \emph{\omterm{def:g12:oscillators-and-time:damping}{अवमंदित}} होते हैं।
\omterm{def:g10:signals-and-waves:oscilloscope}{दोलनदर्शी} पर पढ़ी जा सकने वाली दो व्यवस्थाएँ:
\begin{itemize}
  \item \emph{छद्म-आवर्ती} (छोटा $R$): परिपथ अब भी सिकुड़ते \omterm{def:g10:refraction:fiber}{आवरण}
        के भीतर दोलन करता है, और \omterm{def:g12:oscillators-and-time:damping}{अवमंदन} हलका हो तो दोहराव का समय ---
        \emph{छद्म-आवर्तकाल}\index{छद्म-आवर्तकाल!विद्युत} --- $T_0 = 2\pi\sqrt{LC}$ के पास
        ही रहता है;
  \item \emph{अनावर्ती} (बड़ा $R$): आवेश बिना कभी छोर पार किए रेंगते हुए
        शून्य पर लौट आता है --- यानी दोलन बिलकुल नहीं।
\end{itemize}
यह \omterm{def:g12:oscillators-and-time:damping}{अवमंदित} \omterm{def:g12:oscillators-and-time:oscillator}{यांत्रिक दोलित्र} (\cref{def:g12:oscillators-and-time:damping}) का हूबहू दर्पण है।
\end{definition}

\begin{remark}[दोलन बनाए रखना]\label{rem:g12:rlc-oscillations:maintain}
\index{पोषित दोलन}
आयाम को जीवित रखने के लिए परिपथ को ठीक उतनी ऊर्जा खिलाइए
जितनी प्रतिरोध जलाता है, हर चक्र में एक बार और झूले के \emph{क़दम से
क़दम मिलाकर}: प्रवर्धक RLC फंदे के लिए वही करता है जो निर्गम-तंत्र का धक्का
लोलक के लिए (\cref{ch:g12:oscillators-and-time})। हर क्वार्ट्ज़ घड़ी और हर रेडियो प्रेषक ऐसा ही पोषित
\omterm{def:g12:oscillators-and-time:oscillator}{दोलित्र} है, जो अपनी \omterm{def:g12:rlc-oscillations:period}{स्वाभाविक आवृत्ति} $f_0$ पर चलता है।
\end{remark}

\section{यांत्रिक जुड़वाँ}

\begin{proposition}[विद्युत--यांत्रिक शब्दावली]\label{prop:g12:rlc-oscillations:analogy}
\index{विद्युत--यांत्रिक सादृश्य}
LC समीकरण शब्द-दर-शब्द स्प्रिंग--द्रव्यमान समीकरण (\cref{prop:g12:oscillators-and-time:spring-period}) ही है:
\begin{center}\small
\begin{tabular}{lcl}
\omterm{prop:g12:rlc-oscillations:equation}{LC परिपथ} & & स्प्रिंग--द्रव्यमान फिसलक\\[2pt]
आवेश $q$ & $\longleftrightarrow$ & स्थिति $x$\\
धारा $i = dq/dt$ & $\longleftrightarrow$ & वेग $v = dx/dt$\\
\omterm{def:g12:rc-rl-circuits:inductor}{प्रेरकत्व} $L$ & $\longleftrightarrow$ & द्रव्यमान $m$\\
व्युत्क्रम \omterm{def:g12:rc-rl-circuits:capacitor}{धारिता} $1/C$ & $\longleftrightarrow$ & \omterm{def:g12:oscillators-and-time:spring}{दृढ़ता} $k$\\
प्रतिरोध $R$ & $\longleftrightarrow$ & \omterm{def:g10:inertia:inventory}{घर्षण}\\
$L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0$ & $\longleftrightarrow$ & $m\,\frac{d^2x}{dt^2} + kx = 0$\\[2pt]
$T_0 = 2\pi\sqrt{LC}$ & $\longleftrightarrow$ & $T_0 = 2\pi\sqrt{m/k}$\\[2pt]
$E_C = q^2/2C$ & $\longleftrightarrow$ & $E_p = \tfrac12 kx^2$\\
$E_L = \tfrac12 Li^2$ & $\longleftrightarrow$ & $E_k = \tfrac12 mv^2$
\end{tabular}
\end{center}
वही समीकरण, वही हल: किसी भी \omterm{def:g12:oscillators-and-time:oscillator}{दोलित्र} के बारे में हर परिणाम तत्काल
दूसरी भाषा में अनूदित हो जाता है।
\end{proposition}

\begin{proof}
\cref{prop:g12:rlc-oscillations:equation} की तुलना $m\,d^2x/dt^2 = -kx$ से कीजिए: $q \to x$, $L \to m$, $1/C \to k$ रखने पर एक दूसरे
में बदल जाता है, और $LC \to m/k$ $2\pi\sqrt{LC}$ को $2\pi\sqrt{m/k}$ बना देता है।
\end{proof}

\begin{example}[रेडियो सुर में बिठाना]\label{ex:g12:rlc-oscillations:tuning}
ऐंटेना \emph{हर} स्टेशन की एक फुसफुसाहट LC फंदे में डाल देता है, पर क़दम मिलाकर
धक्का देकर बढ़ता केवल वही है जो $f_0$ के पास प्रसारित हो रहा हो --- यानी
\omterm{def:g12:oscillators-and-time:resonance}{अनुनाद} (\cref{def:g12:oscillators-and-time:resonance})। डायल एक परिवर्ती \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} की पट्टियाँ
घुमाता है: $C$ बदलने से $f_0 = 1/(2\pi\sqrt{LC})$ खिसक जाता है, और सुई कान को एक बार में एक
स्टेशन थमा देती है। और प्रेषण की ओर पोषित \omterm{def:g12:oscillators-and-time:oscillator}{दोलित्र} (\cref{rem:g12:rlc-oscillations:maintain}) प्रसारण
की आवृत्ति तय करता है; घड़ी में यही भूमिका क्वार्ट्ज़ का क्रिस्टल
निभाता है --- यानी अद्भुत \omterm{def:g12:oscillators-and-time:spring}{दृढ़ता} वाला एक यांत्रिक जुड़वाँ।
\end{example}

\section{समस्या: क्रिस्टल रेडियो बनाना}

\begin{problem}[{सप्ताहांत समस्या --- क्रिस्टल रेडियो बनाना: एक कुंडली, एक
परिवर्ती संधारित्र, एक डायोड और एक कान का यंत्र --- न बैटरी, न प्रवर्धक, और
शाम का समाचार ख़ुद तरंग की ऊर्जा पर चला आता है}]
\label{pb:g12:rlc-oscillations:1}
दादा-दादी की अटारी से मिली एक किट: गत्ते की नली पर लपेटी $L = \qty{0.20}{mH}$ की
\omterm{def:g11:magnetic-fields:solenoid}{कुंडली}, ऐसा परिवर्ती \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} जिसकी पट्टियाँ \qty{20}{pF} से
\qty{480}{pF} तक घूमती हैं, एक क्रिस्टल डायोड, कान का एक यंत्र, और \qty{20}{m} ऐंटेना का
तार। पकड़े जाने वाले स्टेशन \qty{530}{kHz} और \qty{1700}{kHz} के बीच प्रसारित करते हैं।

\medskip
\noindent\textbf{भाग I --- सुर बिठाने वाला फंदा।}
\begin{enumerate}
  \item आवेशित \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} को \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} में विसर्जित होने दिया
        जाता है: फंदा-नियम लिखिए और उसे $q(t)$ के अवकल समीकरण में बदलिए।
  \item प्रतिस्थापन से जाँचिए कि $q(t) = Q_0\cos(2\pi t/T_0)$ उसे हल कर देता है, बशर्ते $T_0 = 2\pi\sqrt{LC}$।
  \item विमीय विश्लेषण से जाँचिए कि $\sqrt{LC}$ एक समय है।
  \item डायल $C = \qty{330}{pF}$ पर: $T_0$ और $f_0$ निकालिए। क्या फंदा उस पट्टी के
        भीतर सुर में है?
  \item वह \omterm{def:g12:rc-rl-circuits:capacitor}{धारिता} निकालिए जो ठीक \qty{1.00}{MHz} पर सुर बिठा दे।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- परिवर्ती \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र}।}
\begin{enumerate}[resume]
  \item \qty{530}{kHz} के लिए ज़रूरी \omterm{def:g12:rc-rl-circuits:capacitor}{धारिता} निकालिए।
  \item \qty{1700}{kHz} के लिए ज़रूरी \omterm{def:g12:rc-rl-circuits:capacitor}{धारिता} निकालिए।
  \item किट के \qtyrange{20}{480}{pF} वाले \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} से असल में मिलती
        दोनों आवृत्तियाँ निकालिए। क्या वह पूरी पट्टी ढक लेता है?
  \item दिखाइए कि नियत $L$ पर $f_0 \propto 1/\sqrt{C}$; $f_0$ दुगुना करने के लिए $C$
        को कितने गुना बदलना पड़ेगा?
  \item पट्टियाँ रैखिक रूप से खुलती हैं, इसलिए डायल घूमने के साथ $C$
        एकसार गिरता है। समझाइए कि तब स्टेशन डायल के एक ही सिरे पर क्यों भीड़
        लगा लेते हैं --- और किस सिरे पर।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- ऊर्जा और \omterm{def:g12:oscillators-and-time:damping}{अवमंदन}।}
\qty{1.00}{MHz} ($C = \qty{127}{pF}$) पर सुर में बिठाने पर ऐंटेना पल भर के लिए
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} को $U_0 = \qty{20}{mV}$ तक आवेशित कर देता है।
\begin{enumerate}[resume]
  \item आवेश $Q_0$ और संचित ऊर्जा $E$ निकालिए।
  \item ऊर्जा के संरक्षण से शिखर धारा $I_0$ निकालिए।
  \item \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} का प्रतिरोध प्रति चक्र संचित ऊर्जा
        का $4.0\%$ \omterm{def:g11:mechanical-energy:dissipation}{क्षय} कर देता है: कितने चक्रों के बाद ऊर्जा आधी रह
        जाती है?
  \item यह माइक्रोसेकंडों में कितना है? अकेली यह ``झनझनाहट'' शाम का समाचार
        क्यों नहीं बजा सकती?
  \item समझाइए कि आती हुई तरंग फंदे को केवल तभी झुलाए रखती है जब वह
        उसी स्टेशन पर सुर में हो --- और \emph{प्रेषक} पर दोलन को कौन बनाए
        रखता है (\cref{rem:g12:rlc-oscillations:maintain})।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- यांत्रिक जुड़वाँ।}
\begin{enumerate}[resume]
  \item शब्दावली (\cref{prop:g12:rlc-oscillations:analogy}) और द्रव्यमान $m = \qty{0.20}{kg}$ के फिसलक के साथ वह
        \omterm{def:g12:oscillators-and-time:spring}{दृढ़ता} $k$ निकालिए जो सुर में बिठाए गए परिपथ से मेल खाए।
  \item कार की किसी कड़ी स्प्रिंग का $k \approx \qty{5e4}{N/m}$ होता है। टिप्पणी कीजिए: क्या
        मेज़ पर रखा कोई स्प्रिंग--द्रव्यमान निकाय मेगाहर्ट्ज़ पर दोलन कर सकता
        है?
  \item इसके बजाय एक तर्कसंगत $k = \qty{100}{N/m}$ रखिए: \qty{1.00}{MHz} पर कौन-सा द्रव्यमान दोलन
        करेगा?
  \item इतना हलका और इतना कड़ा \omterm{def:g12:oscillators-and-time:oscillator}{यांत्रिक दोलित्र} असल में कौन-सी चीज़ है,
        और इसी वर्ष आप उससे पहले कहाँ मिले थे (\cref{ch:g12:oscillators-and-time})?
  \item विवरण-पत्रक, एक वाक्य में: आपका रेडियो कौन-सी पट्टी बुहारता है,
        कौन-सी \omterm{def:g12:rc-rl-circuits:capacitor}{धारिता} \qty{1.00}{MHz} चुनती है, अकेली एक झनझनाहट कितनी देर
        टिकती है, और फिर भी समाचार बजता कैसे है।
