\chapter{विशिष्ट आपेक्षिकता: काल-विस्तारण}\label{ch:g12:special-relativity}

GPS का हर \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} ऐसी \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} घड़ी ढोता है जो तीस लाख वर्ष में एक \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} भर
चूकती है --- और प्रक्षेपण से पहले अभियंता जान-बूझकर उसे सुर से उतार देते हैं:
ईमानदार छोड़ दी जाए तो वह प्रतिदिन \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} के 38 लाखवें हिस्से आगे निकल जाएगी, और
कल शाम तक आपका फ़ोन आपको वहाँ से ग्यारह किलोमीटर दूर बिठा देगा जहाँ आप खड़े हैं।
दोष घड़ी का नहीं है --- समय ख़ुद चलते और ठहरे हुए के लिए अलग-अलग टिकता है: यह
अध्याय आइंस्टाइन के पीछे एक ज़िद्दी तथ्य से उसी निष्कर्ष तक चलता है।

\section{एक चाल जुड़ने से इनकार कर देती है}

चालें तंत्र पर निर्भर करती हैं और जोड़ से मिलती हैं (\cref{ch:g10:relative-motion}): \qty{300}{km/h} से चलती
रेलगाड़ी में आगे की ओर \qty{4}{km/h} से चलता यात्री पटरी के किनारे खड़े प्रेक्षक के
पास से \qty{304}{km/h} पर गुज़रता है।

\begin{definition}[जड़त्वीय तंत्र]\label{def:g12:special-relativity:inertial}
\emph{जड़त्वीय तंत्र}\index{जड़त्वीय तंत्र} वह \omterm{def:g10:relative-motion:frame}{निर्देश तंत्र} (\cref{ch:g10:relative-motion}) है
जिसमें बलहीन पिंड अचर वेग बनाए रखता है: ज़मीन, लगभग-लगभग, और उसके सापेक्ष सरल
रेखा में \omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान} चलता हर तंत्र।
\end{definition}

अब इस नियम को चरम तक खींचिए: रेलगाड़ी की हेडलाइट का पुंज ज़मीन पर खड़े प्रेक्षक के
पास से $c + \qty{300}{km/h}$ पर गुज़रना चाहिए। सन 1887 में माइकलसन और मॉर्ली ने पृथ्वी की
\qty{30}{km/s} वाली कक्षीय दौड़ के अनुदिश और उसके आड़े प्रकाश की चाल की तुलना करके ऐसे
ही अंतर खोजे; और अद्भुत परिशुद्धता तक उन्हें कोई नहीं मिला --- हर दिशा में, हर
ऋतु में वही $c$। सन 1905 में आइंस्टाइन ने उस प्रयोग पर भरोसा किया और समय तथा
अवकाश को उसी के चारों ओर फिर से खड़ा कर दिया, दो अभिगृहीतों पर, जिनका फ़ैसला
उनके परिणामों से होना था --- और अब सदी भर की परीक्षा में एक बार भी वे विफल नहीं
हुए।

\begin{theorem}[विशिष्ट आपेक्षिकता के अभिगृहीत]\label{thm:g12:special-relativity:postulates}
\index{विशिष्ट आपेक्षिकता}\index{आपेक्षिकता के अभिगृहीत}
\begin{enumerate}
  \item \emph{आपेक्षिकता का सिद्धांत}: भौतिकी के नियम हर \omterm{def:g12:special-relativity:inertial}{जड़त्वीय
        तंत्र} में एक ही रूप लेते हैं; कोई भी ``सचमुच \omterm{def:g10:relative-motion:frame}{विराम} में'' नहीं है।
  \item \emph{$c$ की अपरिवर्त्यता}: निर्वात में प्रकाश हर \omterm{def:g12:special-relativity:inertial}{जड़त्वीय
        तंत्र} में $c = \qty{3.00e8}{m/s}$ से चलता है, स्रोत या प्रेक्षक की गति चाहे जो हो।
\end{enumerate}
\end{theorem}
\admitted

\section{``एक ही क्षण'' को विदा}

\begin{proposition}[युगपतता की आपेक्षिकता]\label{prop:g12:special-relativity:simultaneity}
\index{युगपतता}
किसी एक \omterm{def:g12:special-relativity:inertial}{जड़त्वीय तंत्र} में युगपत दो घटनाएँ आम तौर पर दूसरे तंत्र
में युगपत नहीं होतीं: ``अभी, हर जगह'' तंत्र पर निर्भर है।
\end{proposition}

\begin{proof}
आइंस्टाइन की रेलगाड़ी। बिजली की दो कड़कें चलती रेलगाड़ी के दोनों सिरों पर गिरती
हैं और पटरी पर निशान झुलसा देती हैं। दोनों निशानों के ठीक बीच खड़ा प्लेटफ़ॉर्म का
प्रेक्षक $M$ दोनों चमकें एक साथ पाता है; हर चमक ने वही दूरी उसी चाल $c$
पर तय की थी, इसलिए दोनों कड़कें युगपत थीं। पर रेलगाड़ी के ठीक बीचोंबीच बैठा
यात्री $M'$ आगे वाली चमक की \emph{ओर} बढ़ रहा है, इसलिए वह उस तक पहले पहुँचती
है; और उसके तंत्र में भी दोनों चमकों ने बराबर दूरियाँ (आधी गाड़ी) उसी चाल $c$
पर तय कीं --- दूसरा अभिगृहीत! --- इसलिए आगे वाली कड़क पहले गिरी थी। दोनों फ़ैसले
अपने-अपने तंत्र में सही हैं; और यह असहमति, रेलगाड़ी के लिए नैनोसेकंडों की, इसीलिए
किसी की नज़र में नहीं आई।
\end{proof}

\section{प्रकाश-घड़ी और $\gamma$ गुणक}

\begin{definition}[घटना, निजी काल]\label{def:g12:special-relativity:proper-time}
\emph{घटना}\index{घटना} एक ही जगह और एक ही क्षण पर घटा हुआ कुछ है --- एक टिक,
एक चमक, एक \omterm{def:g11:mechanical-energy:dissipation}{क्षय}। जब दोनों घटनाओं पर एक ही घड़ी मौजूद हो, तो उनके बीच वह जो
अंतराल पढ़ती है वही \emph{निजी काल}\index{निजी काल} $\Delta t_0$ है। और जिस तंत्र में
वह घड़ी चलती है, वहाँ आम तौर पर कोई और अंतराल $\Delta t$ नापा जाता है।
\end{definition}

\begin{theorem}[काल-विस्तारण]\label{thm:g12:special-relativity:dilation}
\index{काल-विस्तारण}\index{गामा गुणक@$\gamma$ (गामा) गुणक}
किसी \omterm{def:g12:special-relativity:inertial}{जड़त्वीय तंत्र} में अचर चाल $v$ से चलती घड़ी वहाँ धीमी चलती हुई
नापी जाती है: घड़ी पर घटी दो घटनाओं के बीच
\[
\Delta t = \gamma\, \Delta t_0, \qquad
\gamma = \frac{1}{\sqrt{1 - v^2/c^2}} \geq 1 .
\]
\end{theorem}

\begin{proof}
सबसे सरल कल्पनीय घड़ी बनाइए: $L$ की दूरी पर आमने-सामने रखे दो दर्पण, और उनके
बीच उछलता एक \omterm{def:g12:quantum-world:photon}{फ़ोटॉन}; एक चक्कर एक टिक है, और \omterm{def:g10:relative-motion:frame}{विराम} में एक टिक $\Delta t_0 = 2L/c$
चलती है। अब उस घड़ी को चाल $v$ से जाते किसी जहाज़ पर कस दीजिए, दर्पण गति के
लंबवत, और ज़मीन से एक टिक का समय नापिए: जब तक \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} चढ़ता है, दर्पण
बग़ल में सरक जाते हैं, इसलिए \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} तिरछा पथ उड़ता है --- यानी एक साथ दो
गतियाँ, ठीक \cref{ch:g12:kinematics-2d} के \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} की तरह। यदि टिक $\Delta t$ चले, तो हर आधी
टिक में क्षैतिज दिशा में $v\,\Delta t/2$ और \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} दिशा में $L$ तय होता
है; और --- यही निर्णायक क़दम है --- दूसरा अभिगृहीत \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} की ज़मीनी चाल
$c$ पर जमा देता है, इसलिए तिरछा आधा-पथ $c\,\Delta t/2$ नापता है। पाइथागोरस:
\[
\Bigl(\frac{c\,\Delta t}{2}\Bigr)^{\!2} = L^2 + \Bigl(\frac{v\,\Delta t}{2}\Bigr)^{\!2}
\;\Longrightarrow\;
\Delta t = \frac{2L/c}{\sqrt{1 - v^2/c^2}} = \gamma\,\Delta t_0 .
\]
और उसके साथ सवार कोई भी घड़ी --- कलाई-घड़ी, दिल की धड़कन, \omterm{def:g11:mechanical-energy:dissipation}{क्षय} होता नाभिक
--- क़दम मिलाए रखनी ही होगी, वरना तुलना जहाज़ की गति खोल देती और पहला अभिगृहीत
टूट जाता।
\end{proof}

\begin{remark}[रोज़मर्रा की ज़िंदगी को इसकी भनक क्यों नहीं लगी]\label{rem:g12:special-relativity:everyday}
$v \ll c$ के लिए $x = v^2/c^2$ लिखिए: चूँकि $(1-x)(1+x) \approx 1$ और $(1 + x/2)^2 \approx 1 + x$, इसलिए $\gamma \approx 1 + v^2/(2c^2)$ मिलता है।
\qty{130}{km/h} से चलती कार का $\gamma - 1 \approx \num{7e-15}$ है (यानी चालीस लाख वर्ष चलाने पर एक \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} की
हानि); और जेट $\num{3.5e-13}$ तक पहुँचता है, यानी नब्बे हज़ार वर्ष उड़ने पर एक \omterm{def:g10:orders-of-magnitude:unit}{सेकंड}।
हमारी घड़ियाँ बस बहुत मोटी थीं।
\end{remark}

\section{प्रयोग का फ़ैसला}

\begin{example}[म्यूऑन ज़मीन तक पहुँच जाते हैं]\label{ex:g12:special-relativity:muons}
ऊपरी वायुमंडल से टकराती ब्रह्मांडीय किरणें \emph{म्यूऑन} बना देती हैं ---
यानी $\Delta t_0 = \qty{2.2}{\micro s}$ के निजी जीवनकाल वाले अल्पजीवी कण --- और वह भी लगभग \qty{15}{km} ऊपर, लगभग
$c$ पर चलते हुए। आपेक्षिकता के बिना कोई म्यूऑन एक जीवनकाल में $c\,\Delta t_0 \approx \qty{660}{m}$ तय करता
है: उतरने में कोई $23$ जीवनकाल लगते, और बचा हुआ अंश $\mathrm{e}^{-23} \approx \num{e-10}$ समुद्र-तल पर
म्यूऑन को लगभग नामुमकिन बना देता। फिर भी संसूचक प्रति वर्ग सेंटीमीटर प्रति मिनट
लगभग एक गिनते हैं। \omterm{thm:g12:special-relativity:dilation}{काल-विस्तारण} इसे सुलझा देता है: $\gamma \approx 30$ के साथ आते
म्यूऑन हमारी घड़ियों पर $30 \times \qty{2.2}{\micro s} = \qty{66}{\micro s}$ जीते हैं --- यानी \omterm{def:g12:projectile-motion:range}{परास} $\approx \qty{20}{km}$, पूरा
वायुमंडल।
\end{example}

पुष्टियाँ ढेर होती जाती हैं। सन 1971 में हैफ़ेल और कीटिंग सीज़ियम की \omterm{def:g11:fundamental-interactions:ladder}{परमाणु}
घड़ियाँ (\cref{ch:g12:oscillators-and-time}) यात्री विमानों पर दुनिया भर घुमा लाए; उतरने पर वे ज़मीनी घड़ियों
से ठीक उतने ही माइक्रोसेकंड के अंशों से भिन्न मिलीं जितना कहा गया था। त्वरक रोज़
इसी विस्तारण के भरोसे चलते हैं, और उनके कण $\gamma$ तक बढ़ाकर हज़ारों ऐसे चक्कर
जीते हैं जो चिरसम्मत भौतिकी में कभी पूरे न होते। और GPS तो क़ीमत की चिप्पी लगा
हुआ चलता-फिरता प्रयोग है।

\begin{remark}[GPS का आपेक्षिकता-बिल]\label{rem:g12:special-relativity:gps}
GPS का \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} लगभग \qty{3.9}{km/s} पर कक्षा में घूमता है, इसलिए
\omterm{thm:g12:special-relativity:postulates}{विशिष्ट आपेक्षिकता} उसकी घड़ी को प्रतिदिन कोई \qty{7}{\micro s} धीमा कर देती है।
पर ऊँचाई पर कमज़ोर गुरुत्व उलटी दिशा में काम करता है और जीत जाता है:
सामान्य आपेक्षिकता (यानी आइंस्टाइन का \omterm{def:g11:fundamental-interactions:four}{गुरुत्वाकर्षण} का सिद्धांत, जो
विश्वविद्यालय के खंडों में कहा गया है) उसे प्रतिदिन लगभग \qty{45}{\micro s} तेज़ कर देती
है; और शुद्ध हिसाब में प्रतिदिन \qty{38}{\micro s} तेज़। स्थिति-निर्धारण घड़ी के पाठ्यांकों
को चाल $c$ पर दूरियों में बदलता है: इसलिए बिना सुधार का एक दिन स्थितियों को
$c \times \qty{38}{\micro s} \approx \qty{11}{km}$ तक फैला देता --- और शुरुआती पैराग्राफ़ की वे सुर से उतारी गई घड़ियाँ
इसीलिए हैं।
\end{remark}

\begin{proposition}[लंबाई-संकुचन]\label{prop:g12:special-relativity:contraction}
\index{लंबाई-संकुचन}
अपने विराम-तंत्र में लंबाई $L_0$ वाला पिंड उस तंत्र में, जहाँ वह चाल $v$
से चलता है, अपनी गति की दिशा के अनुदिश छोटा नापा जाता है: $L = L_0/\gamma$।
\end{proposition}
\admitted

\begin{remark}[म्यूऑन की अपनी कहानी]\label{rem:g12:special-relativity:muonframe}
\omterm{prop:g12:special-relativity:contraction}{लंबाई-संकुचन} \omterm{thm:g12:special-relativity:dilation}{काल-विस्तारण} का ही दूसरा पहलू है (जिसे
विश्वविद्यालय के खंडों में ईमानदारी से निकाला गया है): म्यूऑन की घड़ी उसके अपने
तंत्र में सामान्य ही टिकती है, पर उसकी ओर दौड़ता वायुमंडल सिकुड़कर $\qty{15}{km}/30 =
\qty{500}{m}$ रह
जाता है, जिसे वह \qty{2.2}{\micro s} के भीतर ही पार कर लेता है --- और दोनों तंत्र सहमत हैं
कि म्यूऑन ज़मीन तक पहुँच जाता है।
\end{remark}

\section{द्रव्यमान, ऊर्जा, और लौटी हुई चिरसम्मत दुनिया}

\begin{theorem}[द्रव्यमान--ऊर्जा तुल्यता]\label{thm:g12:special-relativity:emc2}
\index{द्रव्यमान--ऊर्जा तुल्यता}
द्रव्यमान $m$ का पिंड केवल अपने द्रव्यमान भर से ऊर्जा रखता है:
\[ E = m c^2 . \]
द्रव्यमान ऊर्जा का सिमटा हुआ रूप है; और उलटे, ऊर्जा का
द्रव्यमान होता है।
\end{theorem}
\admitted

\begin{remark}[आप इससे कहाँ मिल चुके हैं]\label{rem:g12:special-relativity:emc2seen}
\omterm{def:g10:energy-conservation:forms}{नाभिकीय ऊर्जा} (\cref{ch:g12:nuclear-energy}) के पीछे यही बहीखाता है: किसी अभिक्रिया की
\omterm{def:g12:nuclear-energy:defect}{द्रव्यमान क्षति} गुणा $c^2$ ही छूटी ऊर्जा है। एक ग्राम
द्रव्यमान \qty{9.0e13}{J} है --- यानी किसी बड़े शक्ति-संयंत्र का पूरे दिन का
उत्पादन। ईमानदार व्युत्पत्ति विश्वविद्यालय के खंडों की है; पर रिएक्टर और तारे
उसकी परवाह किए बिना उसी पर चलते हैं।
\end{remark}

\begin{remark}[न्यूटन ग़लत नहीं थे]\label{rem:g12:special-relativity:newton}
$v \ll c$ रख दीजिए: $\gamma \to 1$, समय मेल खाने लगते हैं, लंबाइयाँ फिर तन जाती हैं,
युगपतता \omterm{met:g10:pressure:dive}{निरपेक्ष} हो जाती है --- और पिछले तीन वर्षों की यांत्रिकी ज्यों की त्यों,
धीमी चाल की सीमा के रूप में लौट आती है। आपेक्षिकता ने न्यूटन को ढहाया नहीं;
उसने उनके साम्राज्य की सरहद खींच दी। भौतिकी इसी तरह बढ़ती है: सिद्धांतों को
धार देकर, और हर नया सिद्धांत पुराने को सीमांत स्थिति के रूप में वापस लौटा देता
है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:1}
$v = 0.1c$, $0.6c$, $0.8c$, $0.995c$ के लिए $\gamma$ तीन सार्थक अंकों तक
निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:2}
कोई जहाज़ $0.6c$ से चलता है; उसका ताल-यंत्र जहाज़ की घड़ी से हर \qty{10.0}{s} पर
धड़कता है। ज़मीन कौन-सा अंतराल नापती है? \omterm{def:g12:special-relativity:proper-time}{निजी काल} कौन-सा है, और क्यों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:3}
किस चाल पर ($c$ के अंश के रूप में) $\gamma = 2$ होता है? और $\gamma = 10$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:4}
कोई कार \qty{130}{km/h} से चलती है। $v/c$ निकालिए, फिर $\gamma - 1$ (\cref{rem:g12:special-relativity:everyday}); उसकी घड़ी
को \qty{1}{s} पीछे पड़ने के लिए उसे कितनी देर चलना पड़ेगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:5}
आइंस्टाइन के रेलगाड़ी वाले प्रयोग में, प्रयुक्त अभिगृहीत का हवाला देते हुए
समझाइए कि यात्री \emph{आगे} वाली कड़क को पहला क्यों ठहराता है, और दोनों में से
कोई प्रेक्षक ग़लत क्यों नहीं है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:6}
आवेशित पायॉन $\Delta t_0 = \qty{2.6e-8}{s}$ जीते हैं। कोई पुंज त्वरक से $0.99c$ पर निकलता है।
$\gamma$, प्रयोगशाला में जीवनकाल, चिरसम्मत \omterm{def:g12:projectile-motion:range}{परास} $v\,\Delta t_0$, और असली माध्य
\omterm{def:g12:projectile-motion:range}{परास} निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:7}
$\gamma$ के वक्र से वे चालें पढ़िए जो $\gamma = 1.5$ और $\gamma = 3$ देती हैं (गणना से जाँच
लीजिए)। $v = c$ पर खड़ी अनंतस्पर्शी द्रव्यमान वाले किसी पिंड को त्वरित करने के
बारे में क्या कहती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:8}
किसी प्रकाश-घड़ी के दर्पण $L = \qty{1.5}{m}$ की दूरी पर हैं। \omterm{def:g10:relative-motion:frame}{विराम} में उसकी टिक निकालिए,
फिर $0.8c$ से चलते समय ज़मीन से नापी गई उसकी टिक।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:9}
इनके बीच \omterm{def:g12:special-relativity:proper-time}{निजी काल} कौन नापता है: (क) कलाई-घड़ी की दो टिक --- पहनने वाला,
या पास से गुज़रता कोई? (ख) किसी म्यूऑन का जन्म और \omterm{def:g11:mechanical-energy:dissipation}{क्षय} --- म्यूऑन, या
प्रयोगशाला? (ग) पृथ्वी से निकलना और मंगल पहुँचना --- चालक, या अभियान-नियंत्रण?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:10}
अंतरिक्ष स्थानक \qty{7.7}{km/s} पर कक्षा में घूमता है। $\gamma - 1 \approx v^2/2c^2$ निकालिए, फिर छह
महीने (\qty{1.58e7}{s}) बाद किसी अंतरिक्ष यात्री की घड़ी ज़मीनी घड़ी से कितना पीछे रह
जाती है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:11}
\qty{1.0}{g} पदार्थ की ऊर्जा-मात्रा $E = mc^2$ निकालिए। उसकी तुलना \qty{1.0}{GW} के
शक्ति-संयंत्र के दैनिक उत्पादन से कीजिए, और \qty{130}{km/h} पर चलती \qty{8.5e5}{J} की
कार की \omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} से भी।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:12}
$\gamma = 1/\sqrt{1 - v^2/c^2}$ से दिखाइए कि $v = c\sqrt{1 - 1/\gamma^2}$; और इससे $\gamma = 30$ वाले म्यूऑनों की चाल $c$ के
मात्रकों में पाँच सार्थक अंकों तक निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:13}
म्यूऑन की यात्रा उसके अपने तंत्र से फिर से सुनाइए: $\gamma = 30$ पर वायुमंडल कितना
मोटा है, $0.99944c$ पर वह कितनी देर में बग़ल से गुज़र जाता है, और क्या यह एक निजी
जीवनकाल में अँट जाता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:14}
कोई रैखिक त्वरक इलेक्ट्रॉनों को \qty{3.2}{km} की नली में $\gamma = \num{1.0e5}$ तक धकेलता है।
प्रयोगशाला में इस सफ़र में कितना समय लगता है? इलेक्ट्रॉन के तंत्र में कितना, और
वहाँ नली कितनी ``लंबी'' है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:15}
प्रॉक्सिमा सेंटौरी $4.2$ \omterm{def:g10:orders-of-magnitude:lightyear}{प्रकाश वर्ष} दूर है (\omterm{def:g10:orders-of-magnitude:lightyear}{प्रकाश वर्ष} वह
दूरी है जो प्रकाश एक वर्ष में तय करता है)। कोई यान वहाँ $0.9c$ से जाता है:
पृथ्वी के पंचांग पर और यान की घड़ी पर इस सफ़र में कितना समय लगता है?
\end{exercise}

\section{समस्या: म्यूऑन और जुड़वाँ बहन}

\begin{problem}[{सप्ताहांत समस्या --- म्यूऑन की यात्रा और अंतरिक्ष यात्री की
जुड़वाँ बहन: अपनी ही घड़ी से अधिक जीता एक कण, फिर से बनाई गई एक प्रकाश-घड़ी, अपनी
जुड़वाँ से छोटी लौटी एक यात्री, और वह बहीखाता जो GPS को ईमानदार रखता है}]
\label{pb:g12:special-relativity:1}
आँकड़े: म्यूऑन का निजी जीवनकाल $\Delta t_0 = \qty{2.2}{\micro s}$; म्यूऑन लगभग \qty{15}{km} ऊपर, लगभग $c$
पर जन्मते हैं; समुद्र-तल पर प्रवाह लगभग एक म्यूऑन प्रति \unit{cm^2} प्रति मिनट;
$c = \qty{3.00e8}{m/s}$।

\medskip
\noindent\textbf{भाग I --- म्यूऑन का असंभव आगमन।}
\begin{enumerate}
  \item आपेक्षिकता के बिना कोई म्यूऑन एक जीवनकाल में कितनी दूर जाता है?
  \item तब \qty{15}{km} के उतार में कितने जीवनकाल लगते हैं?
  \item $n$ जीवनकाल बाद अंश $\mathrm{e}^{-n}$ बचता है: उसका अनुमान लगाइए, नापे गए
        प्रवाह से भिड़ाइए, और एक वाक्य में चिरसम्मत भौतिकी पर फ़ैसला सुनाइए।
  \item ये म्यूऑन असल में $\gamma \approx 30$ के साथ पहुँचते हैं। ज़मीन से नापा गया उनका
        जीवनकाल और उससे निकलता \omterm{def:g12:projectile-motion:range}{परास} निकालिए।
  \item $v = c\sqrt{1 - 1/\gamma^2}$ का उपयोग करते हुए उनकी चाल $c$ के मात्रकों में
        पाँच सार्थक अंकों तक निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- प्रकाश-घड़ी, फिर से बनाई हुई।}
कोई प्रकाश-घड़ी ($L = \qty{1.5}{m}$) $v = 0.6c$ से जाते जहाज़ पर सवार है।
\begin{enumerate}[resume]
  \item जहाज़ पर पढ़ी जाती टिक $\Delta t_0$ निकालिए।
  \item ज़मीन से देखने पर एक टिक $\Delta t$ चलती है: प्रति आधी टिक क्षैतिज बढ़त,
        \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} की \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} चढ़ाई, और --- उस अभिगृहीत का नाम लेते हुए
        जो उसे जमा देता है --- तिरछे आधे-पथ की लंबाई दीजिए।
  \item आधी टिक वाले त्रिभुज पर पाइथागोरस लगाइए और $\Delta t$ के लिए हल कीजिए,
        यह दिखाते हुए कि $\Delta t = \gamma\,\Delta t_0$।
  \item $\gamma$ और $\Delta t$ संख्याओं में निकालिए।
  \item दो टिकों के बीच \omterm{def:g12:special-relativity:proper-time}{निजी काल} कौन-सा प्रेक्षक नापता है? परिभाषा से
        कारण दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- यात्री और उसकी जुड़वाँ।}
कोई अंतरिक्ष यात्री अपनी जुड़वाँ बहन को पृथ्वी पर छोड़कर $0.8c$ से आने-जाने की
यात्रा पर निकलती है; और पृथ्वी की घड़ियाँ पूरे सफ़र का समय \qty{10.0}{years} नापती हैं।
\begin{enumerate}[resume]
  \item $0.8c$ पर $\gamma$ निकालिए (पहले ठीक भिन्न के रूप में, फिर दशमलव में)।
  \item यात्री की घड़ी पर कितना समय बीतता है?
  \item मुड़ने का बिंदु (पृथ्वी के तंत्र में, \omterm{def:g10:orders-of-magnitude:lightyear}{प्रकाश वर्ष} में) कितनी
        दूर था?
  \item फिर मिलने पर उम्र का अंतर कितना है, और कौन-सी जुड़वाँ छोटी है?
  \item ``पर गति तो सापेक्ष है --- यात्री यह भी तो कह सकती है कि यात्रा
        पृथ्वी ने की!'' इस आपत्ति का समाधान एक वाक्य में कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- GPS का बिल।}
GPS का \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} त्रिज्या $r = \qty{2.66e7}{m}$ पर \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} $T = \qty{12}{h}$ के साथ
कक्षा में घूमता है।
\begin{enumerate}[resume]
  \item उसकी कक्षीय चाल $v = 2\pi r/T$ निकालिए।
  \item $\gamma - 1 \approx v^2/2c^2$ निकालिए।
  \item इससे निकालिए कि \omterm{def:g10:universal-gravitation:satellite}{उपग्रह} की घड़ी प्रतिदिन ज़मीनी घड़ियों से कितने
        माइक्रोसेकंड पीछे रह जाती है (\qty{86400}{s})।
  \item यदि अकेला यही बहकाव बिना सुधार छोड़ दिया जाए तो स्थितियों में $c \times$
        (बहकाव) की त्रुटि आती है: एक दिन बाद त्रुटि निकालिए।
  \item ऊँचाई पर कमज़ोर गुरुत्व घड़ी को प्रतिदिन \qty{45}{\micro s} \emph{तेज़} कर देता है
        (सामान्य आपेक्षिकता): शुद्ध बहकाव और किलोमीटरों में दैनिक स्थिति-त्रुटि
        निकालिए --- यानी वह संख्या जो अभियंताओं को हर GPS घड़ी सुर से उतारने पर
        मजबूर कर देती है।
\end{enumerate}
\end{problem}

\bigskip
\noindent
यहाँ स्कूल की भौतिकी समाप्त होती है --- भौतिकी नहीं। विश्वविद्यालय के खंडों में
यांत्रिकी कलन से लैस होकर लौटती है; बिजली और चुंबकत्व ख़ुद को एक ही ढाँचे के रूप
में खोल देते हैं, जिसमें प्रकाश, और स्वयं आपेक्षिकता, शुरू से छिपे बैठे थे; और
पिछले अध्याय में झलका क्वांटम संसार पदार्थ का काम करता सिद्धांत बन जाता है। हर
सिद्धांत को, आपेक्षिकता की तरह, वह लौटाना ही पड़ेगा जो आप अब जानते हैं, सीमांत
स्थिति के रूप में। उसे साथ लेकर आगे बढ़िए: वही हिस्सा है जो नहीं बदलेगा।
