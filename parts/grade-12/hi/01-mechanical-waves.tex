\chapter{यांत्रिक तरंगें}\label{ch:g12:mechanical-waves}

शनिवार की शाम, खचाखच भरा स्टेडियम: खड़े होते दर्शकों की एक लहर एक मिनट से भी
कम में पूरी दीर्घा घूम जाती है, फिर भी कोई अपनी सीट नहीं छोड़ता। कंकड़ का
बनाया छल्ला पूरा तालाब पार कर जाता है जबकि काग़ ऊपर-नीचे भर होता रहता है। यह
अध्याय ख़ुद उसी यात्री की पढ़ाई करता है --- तरंग की --- और उन चार
संख्याओं की जो उसे साध लेती हैं: चाल, \omterm{def:g12:mechanical-waves:delay}{विलंब}, \omterm{def:g10:signals-and-waves:period}{आवर्तकाल},
\omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} --- इतना कि काग़ज़ की एक पट्टी से भूकंप पढ़ा जा सके।

\section{विक्षोभ चलता है, माध्यम ठहरा रहता है}

\begin{definition}[यांत्रिक तरंग]\label{def:g12:mechanical-waves:wave}
\emph{यांत्रिक तरंग}\index{यांत्रिक तरंग} किसी भौतिक माध्यम --- रस्सी, पानी,
हवा, चट्टान --- में संचरित होता वह विक्षोभ है जो \emph{पदार्थ ढोए बिना}
ऊर्जा ले जाता है: माध्यम का हर टुकड़ा अपनी ही जगह पर पल भर झूलता है,
गति आगे सौंप देता है और फिर बैठ जाता है।
\end{definition}

\begin{example}[पदार्थ का कोई परिवहन नहीं]\label{ex:g12:mechanical-waves:transport}
स्टेडियम की तरंग में ``माध्यम'' भीड़ है: चलता वह कर्म है, खड़े हो
जाना, जबकि हर दर्शक अपनी सीट पर बना रहता है। तालाब पर छल्ला फैलता है, पर तैरता
काग़ अपनी ही जगह ऊपर-नीचे होता रहता है।
\end{example}

\begin{definition}[अनुप्रस्थ और अनुदैर्घ्य तरंगें]\label{def:g12:mechanical-waves:transverse}
कोई तरंग \emph{अनुप्रस्थ}\index{अनुप्रस्थ तरंग} तब है जब माध्यम
संचरण के \emph{लंबवत} हिलता हो: झटकी हुई रस्सी पर कूबड़ दौड़ता है जबकि हर
बिंदु केवल ऊपर-नीचे होता है। और वह \emph{अनुदैर्घ्य}\index{अनुदैर्घ्य तरंग}
तब है जब माध्यम संचरण के \emph{अनुदिश} हिले: धकेली गई स्प्रिंग में दबे हुए
फेरों का क्षेत्र आगे चलता जाता है, जबकि हर फेरा अपनी ही जगह
झूलता है।
\end{definition}

\begin{remark}[ध्वनि अनुदैर्घ्य है]\label{rem:g12:mechanical-waves:sound}
\omterm{def:g10:signals-and-waves:sound}{ध्वनि} सबसे बढ़कर \omterm{def:g12:mechanical-waves:transverse}{अनुदैर्घ्य तरंग} है: दबी हुई हवा --- यानी
अधिक दाब (\cref{ch:g10:pressure}) --- स्रोत से कर्णपटह तक चलती है; \cref{ch:g12:sound-acoustics} पूरा उसी
को समर्पित है। भूकंप दोनों तरह की तरंगें भेजते हैं: P~तरंगें चट्टान
को दबाती हैं, S~तरंगें उसे बग़ल में हिलाती हैं।
\end{remark}

\section{चाल और विलंब}

\begin{definition}[संचरण चाल]\label{def:g12:mechanical-waves:speed}
किसी तरंग की \emph{संचरण चाल}\index{संचरण चाल} वह दूरी है जो विक्षोभ
तय करता है, बटा उसमें लगा समय: $v = d/t$। यह चलती हुई \emph{आकृति} की चाल है,
माध्यम की नहीं।
\end{definition}

इसलिए $v = \qty{8.0}{m/s}$ --- आकृति चलती है, रस्सी ठहरी रहती है।}
\end{omfigure}

\begin{proposition}[चाल माध्यम तय करता है]\label{prop:g12:mechanical-waves:medium}
छोटे विक्षोभों के लिए \omterm{def:g12:mechanical-waves:speed}{संचरण चाल} केवल \emph{माध्यम} पर निर्भर करती
है, विक्षोभ के आकार या नाप पर नहीं। खिंची हुई रस्सी पर $v$ \omterm{def:g10:inertia:inventory}{तनाव}
के साथ बढ़ती है और प्रति \omterm{def:g10:orders-of-magnitude:unit}{मीटर} द्रव्यमान के साथ घटती है: कसी और हलकी
रस्सी तेज़, ढीली और भारी धीमी। \admitted
\end{proposition}

\begin{remark}[सूत्र कहाँ रहता है]\label{rem:g12:mechanical-waves:ropeformula}
रस्सी की चाल का ठीक सूत्र वर्ष 1 के खंड में निकाला गया है; इस वर्ष ऊपर दिए
प्रायोगिक तथ्य ही काफ़ी हैं।
\end{remark}

\begin{example}[ध्वनि की चालें]\label{ex:g12:mechanical-waves:speeds}
\omterm{def:g10:signals-and-waves:sound}{ध्वनि} हवा में \qty{340}{m/s}, पानी में \qty{1500}{m/s} और इस्पात में \qty{5900}{m/s} चलती है
--- माध्यम जितना कड़ा, चाल उतनी तेज़। इसीलिए पश्चिमी फ़िल्मों में कान पटरी पर
रखा जाता है: रेलगाड़ी की ख़बर इस्पात पहले देता है।
\end{example}

\begin{definition}[विलंब]\label{def:g12:mechanical-waves:delay}
चाल $v$ की तरंग दूरी $d$ पर बैठे बिंदु तक \emph{विलंब}\index{विलंब}
$\tau = d/v$ के बाद पहुँचती है: $d$ की दूरी पर बैठे दो बिंदु वही गति दोहराते
हैं, और दूर वाला $\tau$ पिछड़ जाता है --- यानी \cref{ch:g10:signals-and-waves} वाली \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि}
से दूरी नापने की विधि, उलटी दिशा में पढ़ी गई।
\end{definition}

\begin{proposition}[बिना अंतःक्रिया अध्यारोपण]\label{prop:g12:mechanical-waves:superposition}
जहाँ दो तरंगें एक-दूसरे पर चढ़ जाएँ, वहाँ माध्यम का विस्थापन हर क्षण
उन विस्थापनों का \emph{योग} होता है जो हर तरंग अकेले पैदा करती; और एक-दूसरे
को पार करने के बाद हर तरंग ज्यों की त्यों चलती रहती है --- यह वर्ष~2
के खंड में गति-समीकरणों की रैखिकता से निकाला गया है। \admitted
\end{proposition}

\begin{example}[एक-दूसरे को पार करती स्पंदें]\label{ex:g12:mechanical-waves:crossing}
रस्सी के दोनों सिरों से ऊपर की ओर एक-एक कूबड़ भेजिए: जहाँ वे मिलते हैं वहाँ
रस्सी पल भर के लिए दोनों ऊँचाइयों के योग तक उठ जाती है; फिर दोनों कूबड़ साबुत
निकल आते हैं --- ठीक जैसे एक-दूसरे को काटती दो बातचीतें अपने-अपने सुनने वालों
तक पहुँच जाती हैं।
\end{example}

\section{आवर्ती तरंगें: दुहरी आवर्तिता}

\begin{definition}[आवर्ती तरंग]\label{def:g12:mechanical-waves:periodic}
जब स्रोत अपनी गति \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} $T$ और आवृत्ति $f = 1/T$ (\cref{ch:g10:signals-and-waves})
के साथ दोहराता है, तो वह माध्यम को \emph{आवर्ती तरंग}\index{आवर्ती तरंग}
खिलाता है: हर बिंदु अपनी गति \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} $T$ के साथ दोहराता है, और हर
एक अपने पड़ोसियों से सफ़र के \omterm{def:g12:mechanical-waves:delay}{विलंब} जितना पिछड़ता है।
\end{definition}

\begin{definition}[तरंगदैर्घ्य]\label{def:g12:mechanical-waves:wavelength}
\omterm{def:g12:mechanical-waves:periodic}{आवर्ती तरंग} का एक क्षणचित्र अवकाश में दोहराता हुआ ढर्रा दिखाता है;
उसका स्थानिक आवर्त --- शिखर से शिखर तक --- \emph{तरंगदैर्घ्य}\index{तरंगदैर्घ्य}
$\lambda$ कहलाता है, और \omterm{def:g10:orders-of-magnitude:unit}{मीटर} में नापा जाता है। \omterm{def:g12:mechanical-waves:periodic}{आवर्ती तरंग}
\emph{दुहरी आवर्ती}\index{दुहरी आवर्तिता} होती है: हर बिंदु पर समय में
आवर्त $T$, और हर क्षण अवकाश में आवर्त $\lambda$।
\end{definition}

\begin{theorem}[तरंग-संबंध]\label{thm:g12:mechanical-waves:relation}
चाल $v$, \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} $T$ और आवृत्ति $f$ वाली
\omterm{def:g12:mechanical-waves:periodic}{आवर्ती तरंग} का \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} $\lambda = v\,T = v/f$ होता है।
\end{theorem}

\begin{proof}
एक \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} में ढर्रा $v\,T$ आगे सरक जाता है; पर तब तक स्रोत ---
और इसलिए हर बिंदु --- अपनी आरंभिक अवस्था में लौट चुका होता है, इसलिए सरका हुआ
ढर्रा मूल ढर्रे पर बैठ जाता है: यानी दोहराव की दूरी $v\,T$ है।
\end{proof}

\begin{remark}[आवृत्ति स्रोत की होती है]\label{rem:g12:mechanical-waves:source}
एक माध्यम से दूसरे में जाते समय --- जैसे \omterm{def:g10:signals-and-waves:sound}{ध्वनि} हवा से पानी में ---
तरंग अपनी आवृत्ति बनाए रखती है, जिसे स्रोत तय करता है, जबकि
उसकी चाल बदल जाती है; और $\lambda = v/f$ के कारण उसके साथ \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} भी बदल
जाता है।
\end{remark}

\begin{example}[सुर A, हवा में और पानी के भीतर]\label{ex:g12:mechanical-waves:concertA}
\cref{ch:g10:signals-and-waves} वाले \qty{440}{Hz} के सुर A का \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} हवा में $340/440 = \qty{0.77}{m}$ है और
पानी में $1500/440 =
\qty{3.4}{m}$: वही सुर, तेज़ माध्यम में खिंचा हुआ।
\end{example}

\begin{definition}[समकला और विपरीत कला में]\label{def:g12:mechanical-waves:phase}
पूरी संख्या भर तरंगदैर्घ्यों की दूरी पर बैठे दो बिंदु, $d = k\lambda$, हर
क्षण एक-सा हिलते हैं: वे \emph{समकला}\index{समकला} में हैं। और $d = (k + \tfrac12)\lambda$ की
दूरी पर बैठे बिंदु हमेशा उलटा हिलते हैं --- शिखर के सामने गर्त: वे
\emph{विपरीत कला}\index{विपरीत कला} में हैं।
\end{definition}

\begin{method}[दुहरी आवर्तिता पढ़ना]\label{met:g12:mechanical-waves:reading}
\begin{enumerate}
  \item किसी नियत बिंदु पर लिए गए अभिलेख $y(t)$ पर शिखर से शिखर तक $T$
        पढ़िए; $f = 1/T$।
  \item किसी नियत क्षण के क्षणचित्र $y(x)$ पर शिखर से शिखर तक $\lambda$
        पढ़िए।
  \item $v = \lambda/T = \lambda f$ निकालिए; और जहाँ संभव हो, सीधे नापे गए $v = d/\tau$ से जाँच
        लीजिए।
  \item दोनों आलेख कभी मत गड्डमड्ड कीजिए: एक का अक्ष \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} ढोता है, दूसरे का
        \omterm{def:g10:orders-of-magnitude:unit}{मीटर}।
\end{enumerate}
\end{method}

\section{तरंगाग्र और क्षीणन}

\begin{definition}[तरंगाग्र]\label{def:g12:mechanical-waves:wavefront}
\emph{तरंगाग्र}\index{तरंगाग्र} उन बिंदुओं की रेखा (या सतह) है जिन तक
तरंग एक ही क्षण पहुँचती है --- तालाब पर शिखरों की रेखा। बिंदु-स्रोत
से निकले तरंगाग्र फैलते हुए वृत्त होते हैं; और किसी भी स्रोत से बहुत दूर वे
लगभग सीधे हो जाते हैं। तरंग अपने तरंगाग्रों के लंबवत संचरित होती है,
और क्रमागत शिखर $\lambda$ की दूरी पर होते हैं।
\end{definition}

\begin{definition}[क्षीणन]\label{def:g12:mechanical-waves:attenuation}

\section{समस्या: भूकंपलेख पढ़ना}

\begin{problem}[{सप्ताहांत समस्या --- भूकंपलेख पढ़ना: टूटी हुई भ्रंश-रेखा से
दो तरंगें दौड़ पड़ती हैं, तीन वृत्त अधिकेंद्र को कील देते हैं, और एक धीमी
तीसरी तरंग किसी तट को उसकी चेतावनी दे जाती है}]
\label{pb:g12:mechanical-waves:1}
भूकंप-वेधशाला में रात की पाली। 09:14:32 पर स्टेशन A की सुई उछलती है: पहले
तीखी तेज़ लहरें, फिर धीमी, चौड़ी झूलें। कहीं भूकंप ने चट्टान तोड़ी है; और भूपर्पटी
में दो तरंगें दौड़ पड़ी हैं: P तरंगें, यानी $v_P = \qty{6.0}{km/s}$ पर चलते
संपीडन, और S तरंगें, यानी $v_S = \qty{3.5}{km/s}$ पर बग़ल का हिलाव।

\medskip
\noindent\textbf{भाग I --- दो तरंगें, एक घड़ी।}
\begin{enumerate}
  \item ऊपर के वर्णनों से P और S तरंगों को \omterm{def:g12:mechanical-waves:transverse}{अनुप्रस्थ} या
        \omterm{def:g12:mechanical-waves:transverse}{अनुदैर्घ्य} में बाँटिए।
  \item हर स्टेशन पर कौन-सी तरंग पहले पहुँचती है? $d = \qty{120}{km}$ पर दोनों
        के सफ़र-समय निकालिए।
  \item दिखाइए कि P--S आगमन-अंतराल $\Delta t = d\,(1/v_S - 1/v_P)$ है और उसे $d = \qty{120}{km}$ के लिए निकालिए।
  \item भूकंपविज्ञानी का दूरी-नियम $d = \dfrac{v_P\,v_S}{v_P - v_S}\,\Delta t$ निकालिए और दिखाइए कि गुणांक
        अंतराल के प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} लगभग \qty{8.4}{km} है।
  \item अंतराल दूरी के साथ बढ़ता क्यों है --- और उस स्टेशन के लिए, जिसने
        भूकंप का आरंभ देखा ही नहीं, यह वरदान क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- नक़्शे पर वृत्त।}
\begin{enumerate}[resume]
  \item स्टेशन A $\Delta t_A = \qty{25}{s}$ नापता है। अधिकेंद्र A से कितनी दूर है?
  \item समझाइए कि यह अकेला पाठ्यांक अधिकेंद्र को किसी बिंदु पर नहीं, एक वृत्त
        पर क्यों रखता है।
  \item स्टेशन B और C $\Delta t_B = \qty{15}{s}$ तथा $\Delta t_C = \qty{32}{s}$ नापते हैं: अधिकेंद्र से उनकी
        दूरियाँ?
  \item समझाइए कि तीनों वृत्त मिलकर अधिकेंद्र को कैसे कील देते हैं।
  \item P तरंग A पर 09:14:32 बजे पहुँची थी। अधिकेंद्र से उसका
        सफ़र-समय और भूकंप का आरंभ-समय निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- वह तरंग जो महासागर पार करती है।}
तीनों वृत्त समुद्र में जाकर मिलते हैं: भ्रंश समुद्र-तल के नीचे टूटा और उसने
पानी का पूरा स्तंभ हिला दिया। गहरे पानी पर लंबी तरंगें उस चाल से
चलती हैं जो गहराई तय करती है --- खुले महासागर में लगभग \qty{200}{m/s}, और तट के
पास लगभग \qty{10}{m/s} (यहाँ मान लिया गया; सूत्र वर्ष 1 के खंड में निकाला गया है)।
अधिकेंद्र से $D = \qty{600}{km}$ दूर एक बंदरगाह है।
\begin{enumerate}[resume]
  \item \qty{200}{m/s} को \unit{km/h} में बदलिए और इतनी ही यात्रा-चाल वाला कोई वाहन
        बताइए।
  \item सूनामी को बंदरगाह तक पहुँचने में कितना समय लगेगा?
  \item बंदरगाह का अपना भूकंपमापी P तरंग पहले महसूस करता है। कितनी
        देर बाद? यदि उसी क्षण चेतावनी बज उठे, तो सूनामी से पहले तट
        को कितनी चेतावनी-अवधि मिलती है?
  \item समुद्र में सूनामी का \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} लगभग \qty{15}{min} है। वहाँ उसका
        \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए, और समझाइए कि कोई जहाज़ उसे महसूस तक
        क्यों नहीं करता।
  \item तट के पास चाल गिरकर \qty{10}{m/s} रह जाती है जबकि \omterm{def:g10:signals-and-waves:period}{आवर्तकाल} वही
        रहता है। नया \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए। सिकुड़ती हुई तरंग
        के साथ क्या होना ही चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- ऊर्जा, और रिपोर्ट।}
कोई प्रबल भूकंप \qty{1e15}{J} की कोटि की ऊर्जा छोड़ता है, जिसे तरंगें ले
जाती हैं।
\begin{enumerate}[resume]
  \item अधिकेंद्र से \qty{10}{km} पर वृत्ताकार \omterm{def:g12:mechanical-waves:wavefront}{तरंगाग्र} की लंबाई
        निकालिए, फिर \qty{210}{km} पर; वह कितने गुना बढ़ गया?
  \item इससे निकालिए कि हानिरहित भूपर्पटी में भी कंपन दूरी के साथ क्यों
        कमज़ोर पड़ता है --- और असली भूपर्पटी जो दूसरा कारण जोड़ती है, उसका
        नाम लीजिए।
  \item द्रव अपरूपण नहीं झेल सकता। दूसरी ओर के स्टेशन P तरंग तो
        दर्ज करते हैं पर S तरंग नहीं: इससे पृथ्वी के \omterm{def:g10:refraction:fiber}{क्रोड} के बारे
        में क्या पता चलता है?
  \item पिछले वर्ष के ऊर्जा वाले अध्याय की तेज़ रफ़्तार रेलगाड़ी
        (\cref{ch:g11:mechanical-energy}) लगभग \qty{1.5e9}{J} \omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} ढोती है: \qty{1e15}{J} ऐसी कितनी
        रेलगाड़ियों के बराबर है?
  \item रिपोर्ट, दो वाक्यों में: भूकंप कहाँ और कब आया, और बंदरगाह को कितनी
        चेतावनी मिली? तीनों संख्याएँ दीजिए।
\end{enumerate}
\end{problem}
