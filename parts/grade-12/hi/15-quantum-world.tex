\section{प्रकाश दानों में आता है: फ़ोटॉन}

\begin{definition}[फ़ोटॉन]\label{def:g12:quantum-world:photon}
आवृत्ति $\nu$ का प्रकाश पदार्थ से ऊर्जा केवल पूरे दानों में
बदलता है, जिन्हें \emph{फ़ोटॉन}\index{फ़ोटॉन} कहते हैं और जिनमें से हर एक की
ऊर्जा है
\[
E = h\nu = \frac{hc}{\lambda}, \qquad h = \qty{6.63e-34}{J.s},
\]
जहाँ $h$ \emph{प्लांक नियतांक}\index{प्लांक नियतांक} है (प्लांक, 1900;
आइंस्टाइन, 1905) और $\lambda$ \omterm{def:g12:light-as-wave:lightwave}{निर्वात में तरंगदैर्घ्य}। किसी पुंज की
तीव्रता प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} फ़ोटॉनों की \emph{संख्या} नापती है; और हर दाने की
ऊर्जा अकेली आवृत्ति तय कर देती है।
\end{definition}

\begin{notation}[इलेक्ट्रॉनवोल्ट]\label{not:g12:quantum-world:ev}
\omterm{def:g11:fundamental-interactions:ladder}{परमाणु} भौतिकी \emph{इलेक्ट्रॉनवोल्ट}\index{इलेक्ट्रॉनवोल्ट} काम में लेती है,
$\qty{1}{eV} = \qty{1.60e-19}{J}$: यानी वह ऊर्जा जो एक \omterm{def:g11:fundamental-interactions:elementary}{मूल आवेश} \qty{1}{V} पार करते हुए पाता
है (\cref{ch:g11:electric-gravitational-fields})। काम का सूत्र: $E \,(\unit{eV}) \approx 1240/\lambda\,(\unit{nm})$।
\end{notation}

\section{प्रकाश-विद्युत प्रभाव}

निर्वात में रखी धातु की साफ़ पट्टी पर प्रकाश डालिए और इलेक्ट्रॉन उछलकर बाहर आने
लगते हैं --- यही \emph{प्रकाश-विद्युत प्रभाव}\index{प्रकाश-विद्युत प्रभाव} है।
और उसके नियमों ने चिरसम्मत भौतिकी तोड़ दी।

\begin{proposition}[प्रयोग क्या दिखाता है]\label{prop:g12:quantum-world:facts}
हर धातु की एक \emph{देहली आवृत्ति}\index{देहली आवृत्ति} $\nu_0$ होती है:
\begin{enumerate}
  \item $\nu_0$ से नीचे प्रकाश चाहे कितना ही तीव्र हो, कोई इलेक्ट्रॉन नहीं
        निकलता;
  \item $\nu_0$ से ऊपर प्रकाश चाहे कितना ही मंद हो, उत्सर्जन बिना
        \omterm{def:g12:mechanical-waves:delay}{विलंब} शुरू हो जाता है;
  \item तीव्रता प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} इलेक्ट्रॉनों की \emph{संख्या} बढ़ाती है, उनकी
        अधिकतम \omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} नहीं; $E_{k,\max}$ को आवृत्ति बढ़ाती है।
\end{enumerate}
इनमें से कुछ भी किसी सतत तरंग पर नहीं बैठता, जिस पर हर रंग को आवेश
जमा लेने देना चाहिए --- भले मंद प्रकाश में महीनों लगें (\cref{exo:g12:quantum-world:14}) --- और जिसकी
तीव्रता को इलेक्ट्रॉनों की ऊर्जा तय करनी चाहिए: तीनों पूर्वानुमान
विफल हो जाते हैं।
\end{proposition}
\admitted

\begin{theorem}[आइंस्टाइन का प्रकाश-विद्युत संबंध]\label{thm:g12:quantum-world:einstein}
\index{प्रकाश-विद्युत संबंध}किसी इलेक्ट्रॉन को धातु की सतह से भागने के लिए कम
से कम उस धातु का \emph{कार्य फलन}\index{कार्य फलन} $W_0$ चाहिए। एक
\omterm{def:g12:quantum-world:photon}{फ़ोटॉन} पूरा-का-पूरा एक ही इलेक्ट्रॉन सोख लेता है, इसलिए
\[
h\nu = W_0 + E_{k,\max}, \qquad \text{अतः } \nu_0 = \frac{W_0}{h}.
\]
\end{theorem}

\begin{proof}
ऊर्जा का बहीखाता: भीतर $h\nu$, भागने पर ख़र्च $W_0$, और बाक़ी गतिज ---
अधिक से अधिक $h\nu - W_0$, और भीतर से चले इलेक्ट्रॉनों के लिए उससे कम। $\nu_0$ से नीचे
एक दाना निकास की चुंगी चुका ही नहीं सकता, दाने चाहे जितने हों।
\end{proof}

\begin{definition}[निरोधी विभव]\label{def:g12:quantum-world:stopping}
$E_{k,\max}$ नापने के लिए उलटा \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} इलेक्ट्रॉनों पर ब्रेक लगाता है
(\cref{ch:g11:electric-gravitational-fields}); और \emph{निरोधी विभव}\index{निरोधी विभव} $U_s$ वह सबसे छोटा
\omterm{def:g11:circuits-and-power:voltage}{विभवांतर} है जो धारा को काट दे: सबसे तेज़ इलेक्ट्रॉन बस-बस पार नहीं कर
पाते, इसलिए $E_{k,\max} = e\,U_s$।
\end{definition}

\begin{method}[प्लांक नियतांक नापना]\label{met:g12:quantum-world:measure-h}
\begin{enumerate}
  \item ज्ञात तरंगदैर्घ्यों के फ़िल्टरों से होकर किसी
        प्रकाश-सेल पर प्रकाश डालिए; हर $\nu = c/\lambda$ निकालिए; और हर \omterm{def:g12:quantum-world:stopping}{निरोधी
        विभव} दर्ज कीजिए।
  \item $\nu$ के सामने $E_{k,\max} = eU_s$ आलेखित कीजिए: आइंस्टाइन सीधी रेखा $h\nu - W_0$ का
        वादा करते हैं।
  \item ढाल पढ़ लीजिए: वही $h$ \emph{है}; और अंतःखंड $\nu_0$ तथा $-W_0$
        दे देते हैं।
\end{enumerate}
\end{method}

\section{ऊर्जा स्तर: परमाणु रेखाएँ क्यों छोड़ते हैं}

\begin{definition}[ऊर्जा स्तर]\label{def:g12:quantum-world:levels}
किसी \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} में बँधे इलेक्ट्रॉन की ऊर्जा
\emph{क्वांटित}\index{क्वांटीकरण} होती है: यानी \emph{ऊर्जा स्तरों}\index{ऊर्जा
स्तर} $E_1 < E_2 < \dots < 0$ की एक विविक्त सीढ़ी तक सीमित, जिन्हें मुक्त इलेक्ट्रॉन से नीचे
ऋणात्मक गिना जाता है। सबसे नीचे का पायदान \emph{मूल अवस्था}\index{मूल अवस्था}
है और बाक़ी \emph{उत्तेजित अवस्थाएँ}\index{उत्तेजित अवस्था}; और $0$ तक
पहुँचना \emph{आयनन}\index{आयनन} है।
\end{definition}

\begin{proposition}[हाइड्रोजन की सीढ़ी]\label{prop:g12:quantum-world:hydrogen}
हाइड्रोजन \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} के \omterm{def:g12:quantum-world:levels}{ऊर्जा स्तर} हैं
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2}, \qquad n = 1, 2, 3, \dots
\]
इसलिए $E_1 = \qty{-13.6}{eV}$, $E_2 = \qty{-3.40}{eV}$, $E_3 =
\qty{-1.51}{eV}$, $E_4 = \qty{-0.85}{eV}$, और $0$ की ओर सटते जाते हैं।
\end{proposition}
\admitted

\begin{remark}[यह सीढ़ी आती कहाँ से है]\label{rem:g12:quantum-world:ladder-origin}
ये संख्याएँ क्यों हैं, यह क्वांटम यांत्रिकी है और विश्वविद्यालय के खंडों के लिए
बचाकर रखी गई है --- और वैसे ही दे ब्रॉय का सूत्र भी: इलेक्ट्रॉन की
\omterm{def:g12:quantum-world:debroglie}{द्रव्य तरंग} (नीचे \cref{def:g12:quantum-world:debroglie}) को नाभिक के चारों ओर ख़ुद पर बंद होना
पड़ता है, और उसमें केवल कुछ ही \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} अँटते हैं --- यानी गिटार के
तार की तरह \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} के पास भी सुरों का एक विविक्त समुच्चय होता है।
\end{remark}

\begin{theorem}[संक्रमण का फ़ोटॉन]\label{thm:g12:quantum-world:transition}
कोई \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} स्तर केवल तभी बदलता है जब वह ठीक उतने अंतर को ढोता एक
\omterm{def:g12:quantum-world:photon}{फ़ोटॉन} छोड़े या सोखे: स्तर $p$ से नीचे के स्तर $n$ पर गिरने पर
वह $h\nu = E_p - E_n$ छोड़ता है; और $n$ से $p$ पर चढ़ने पर वही ऊर्जा सोख
लेता है।
\end{theorem}

\begin{proof}
ऊर्जा का संरक्षण, एक बार में एक दाना: प्रकाश केवल \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} की
मुद्रा लेता है, और \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} का बहीखाता बराबर बैठना ही चाहिए।
\end{proof}

\begin{remark}[रेखा-वर्णक्रम की व्याख्या]\label{rem:g12:quantum-world:lines}
\cref{ch:g10:light-spectra} में किया गया वादा यही उत्तर है: गरम गैस केवल अपनी ही सीढ़ी की
आवृत्तियाँ $(E_p - E_n)/h$ छोड़ती है --- यानी चमकीली रेखाओं का बारकोड --- और ठंडी गैस
\omterm{def:g10:light-spectra:white}{श्वेत प्रकाश} में से ठीक वही आवृत्तियाँ चुरा लेती है और उनसे मेल खाती
काली अवशोषण-रेखाएँ छाप देती है। हर तत्व की अपनी सीढ़ी है, इसलिए अपना बारकोड ---
और इसी से हम तारों का प्रकाश पढ़ते हैं।
\end{remark}

\begin{example}[हाइड्रोजन की दृश्य रेखाएँ]\label{ex:g12:quantum-world:balmer}
$n = 2$ पर ख़त्म होते संक्रमण दृश्य क्षेत्र में पड़ते हैं। $E \,(\unit{eV}) =
1240/\lambda\,(\unit{nm})$ के साथ:
$3 \to 2$: \qty{1.89}{eV}, \qty{656}{nm} (लाल); $4 \to 2$: \qty{2.55}{eV}, \qty{486}{nm} (नीला-हरा);
$5 \to 2$: \qty{2.86}{eV}, \qty{434}{nm} (बैंगनी)। और $n = 1$ तक के संक्रमण \qty{10}{eV} से ऊपर
निकल जाते हैं: सब \omterm{def:g10:light-spectra:wavelength}{पराबैंगनी}।
\end{example}

\begin{remark}[लेज़र, एक साँस में]\label{rem:g12:quantum-world:laser}
ठीक $E_p - E_n$ का कोई \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} किसी \emph{उत्तेजित} \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} के पास
से गुज़रते हुए उसे बिलकुल वैसा ही \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} छोड़ने पर उकसा सकता है ---
वही ऊर्जा, वही दिशा, वही क़दम। नीचे वालों से अधिक \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} ऊपर

\emph{उद्दीपित उत्सर्जन}\index{उद्दीपित उत्सर्जन} --- और \omterm{def:g11:color-light-sources:lamps}{लेज़र} इससे
ज़्यादा कुछ है भी नहीं।
\end{remark}

\begin{proposition}[इलेक्ट्रॉन व्यतिकरण करते हैं]\label{prop:g12:quantum-world:interference}
इलेक्ट्रॉनों को एक-एक करके दो झिरियों से भेजिए (\cref{ch:g12:light-as-wave}): हर एक एक तीखे बिंदु

\omterm{def:g12:light-as-wave:fringes}{व्यतिकरण फ़्रिंज} खींच देते हैं: यानी हर इलेक्ट्रॉन, अकेला ही,
तरंग बनकर \emph{दोनों} झिरियों से गुज़रता है। क्रिस्टल भी
इलेक्ट्रॉन-पुंजों को विवर्तित करते हैं (डेविसन और जर्मर, 1927)।
\end{proposition}
\admitted

\begin{definition}[दे ब्रॉय तरंगदैर्घ्य]\label{def:g12:quantum-world:debroglie}
\omterm{def:g12:newtons-laws:momentum}{संवेग} $p = mv$ का हर कण \emph{दे ब्रॉय तरंगदैर्घ्य}\index{दे ब्रॉय
तरंगदैर्घ्य} वाली \emph{द्रव्य तरंग}\index{द्रव्य तरंग} बनकर चलता है:
\[
\lambda = \frac{h}{p} = \frac{h}{mv}.
\]
\end{definition}
\admitted

\begin{example}[आप कभी विवर्तित क्यों नहीं होते]\label{ex:g12:quantum-world:never}
\qty{100}{V} से त्वरित इलेक्ट्रॉन का $E_k = \qty{1.60e-17}{J}$, $p = \sqrt{2m_e E_k} = \qty{5.4e-24}{kg.m/s}$ होता है, इसलिए $\lambda \approx
\qty{0.12}{nm}$ ---
यानी किसी क्रिस्टल में परमाणुओं का अंतराल, जो इसीलिए उसकी दो-झिरी बन
जाता है। और \qty{25}{m/s} पर \qty{58}{g} की टेनिस गेंद: $\lambda =
6.63\times10^{-34}/1.45 \approx \qty{4.6e-34}{m}$, यानी किसी
नाभिक से भी $10^{19}$ गुना छोटा --- कोई झिरी उसे कभी फ़्रिंज बनाते नहीं
दिखाएगी। $h$ इतना छोटा है कि क्वांटम दानेदारी केवल \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} के
पैमाने पर ही सतह पर आती है।
\end{example}

\begin{example}[इलेक्ट्रॉन सूक्ष्मदर्शी]\label{ex:g12:quantum-world:microscope}
\omterm{def:g12:light-as-wave:diffraction}{विवर्तन} काम में लिए गए \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} से बहुत छोटे ब्योरे
सुलझाने से मना कर देता है (\cref{ch:g12:light-as-wave}): दृश्य प्रकाश \qty{0.5}{\micro m} के पास रुक जाता है।
पर \qty{10}{kV} पर इलेक्ट्रॉनों का $\lambda \approx \qty{12}{pm}$ होता है --- यानी दसियों हज़ार गुना छोटा:
इसीलिए इलेक्ट्रॉन सूक्ष्मदर्शी विषाणुओं, कोशिका की मशीनरी, बल्कि अकेले
परमाणुओं तक के चित्र ले लेते हैं। \omterm{def:g12:quantum-world:debroglie}{द्रव्य तरंग} ने भौतिकी को
एक नई आँख सौंप दी।
\end{example}

\begin{problem}[{सप्ताहांत समस्या --- सौर-पैनल की भौतिकी प्रयोगशाला: छत हर

\label{pb:g12:quantum-world:1}
आपकी कक्षा एक सप्ताहांत के लिए स्कूल की छत पर लगे सौर पैनल पर यंत्र लगा देती
है। आँकड़े: दोपहर का सौर विकिरण-प्रवाह \qty{1.0e3}{W/m^2}; माध्य सौर \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य}
\qty{550}{nm}; पैनल का क्षेत्रफल \qty{1.0}{m^2}, \omterm{def:g10:energy-conservation:efficiency}{दक्षता} $20\%$; और दिन भर में
पूरी धूप के \qty{5.0}{h} के बराबर मिलता है।

\medskip
\noindent\textbf{भाग I --- \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} का बहीखाता।}
\begin{enumerate}
  \item एक माध्य सौर \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} की ऊर्जा \omterm{def:g11:work-of-force:work}{जूल} और
        इलेक्ट्रॉनवोल्ट में निकालिए।
  \item इससे दोपहर में पैनल पर प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} टकराते फ़ोटॉनों की संख्या
        निकालिए।
  \item दोपहर में पैनल की \omterm{prop:g11:circuits-and-power:power}{विद्युत शक्ति} निकालिए, फिर दिन भर में बनी
        ऊर्जा \unit{kW.h} में।
  \item पैनल की धारा बिलकुल चिकनी दिखती है। प्रश्न 2 का उपयोग करके समझाइए कि
        दाना-दर-दाना पहुँचना अदृश्य क्यों है।
  \item धूप \omterm{def:g10:light-spectra:wavelength}{पराबैंगनी} से अधिक ऊर्जा \omterm{def:g10:light-spectra:wavelength}{अवरक्त} में ढोती
        है, फिर भी त्वचा को केवल \omterm{def:g10:light-spectra:wavelength}{पराबैंगनी} झुलसाती है (बंध $\sim \qtyrange{3}{4}{eV}$)।
        दानों की भाषा में समझाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- किट के प्रकाश-सेल से $h$ नापना।}
किट के प्रकाश-सेल का कैथोड अनजान है। \omterm{def:g11:color-light-sources:subtractive}{फ़िल्टर} ये देते हैं:
\begin{center}\small
\begin{tabular}{lcccc}
$\lambda$ (\unit{nm}) & 365 & 405 & 436 & 546\\
$U_s$ (\unit{V}) & 1.50 & 1.16 & 0.94 & 0.37
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item \omterm{def:g12:quantum-world:stopping}{निरोधी विभव} किस चीज़ की माप है? $U_s$, $\nu$, $W_0$,
        $h$ को आपस में जोड़िए।
  \item चारों तरंगदैर्घ्यों को आवृत्तियों में बदलिए ($10^{14}\,\unit{Hz}$ में)।
  \item $\nu$ के सामने $eU_s$ आलेखित करने पर सीधी रेखा क्यों मिलनी चाहिए?
        बिंदुओं का रेखा में बैठना जाँचिए।
  \item दोनों छोर के बिंदुओं से ढाल निकालिए: यही आपका नापा हुआ $h$ है।
  \item इससे कैथोड का \omterm{thm:g12:quantum-world:einstein}{कार्य फलन} (\unit{eV}), \omterm{prop:g12:quantum-world:facts}{देहली आवृत्ति}
        और देहली \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए।
  \item धूप का कौन-सा हिस्सा इस कैथोड से इलेक्ट्रॉन निकाल सकता है --- और
        \omterm{def:g12:sound-acoustics:timbre}{वर्णक्रम} का कौन-सा आधा उसके किसी काम का नहीं?
  \item अपने $h$ की तुलना \qty{6.63e-34}{J.s} से कीजिए: प्रतिशत में सापेक्ष त्रुटि?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- हाइड्रोजन का अंशांकन-लैंप।}
किट के तरंगदैर्घ्य-पैमाने की जाँच किसी हाइड्रोजन लैंप से की जाती है,
$E_n = -13.6/n^2$ \unit{eV}।
\begin{enumerate}[resume]
  \item $E_2$, $E_3$, $E_4$ और $E_5$ निकालिए।
  \item संक्रमणों $3 \to 2$, $4 \to 2$, $5 \to 2$ के फ़ोटॉन-ऊर्जा निकालिए।
  \item इससे तीनों \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} निकालिए और हर रेखा का रंग बताइए।
  \item लैंप सतत इंद्रधनुष के बजाय \emph{रेखाएँ} क्यों छोड़ता है? मुख्य विचार
        का नाम लेते हुए एक वाक्य में।
  \item क्या लैंप की हाइड्रोजन \qty{500}{nm} पर प्रकाश सोख सकती है? सीढ़ी से कारण
        दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कोशिकाओं की जाँच: इलेक्ट्रॉन की तरंग।}
किसी कोशिका में पड़ी सूक्ष्म दरारें प्रकाश-सूक्ष्मदर्शियों की \qty{0.5}{\micro m} वाली पहुँच
से कहीं नीचे हैं; और प्रयोगशाला का स्कैनिंग इलेक्ट्रॉन सूक्ष्मदर्शी \qty{10}{kV} पर
चलता है।
\begin{enumerate}[resume]
  \item इलेक्ट्रॉनों की चाल निकालिए; जाँचिए कि वह $0.25\,c$ से नीचे ही रहती है
        (चिरसम्मत सूत्र)।
  \item उनका \omterm{def:g12:newtons-laws:momentum}{संवेग} और \omterm{def:g12:quantum-world:debroglie}{दे ब्रॉय तरंगदैर्घ्य} निकालिए।
  \item विभेदन \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} के साथ चलता है: हरे प्रकाश (\qty{550}{nm}) की
        तुलना में लाभ निकालिए; और सप्ताहांत की दोनों संख्याओं --- आपका $h$
        और यह लाभ --- के साथ निष्कर्ष लिखिए।
\end{enumerate}
\end{problem}
