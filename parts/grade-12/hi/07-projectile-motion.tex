\chapter{मुक्त पतन और प्रक्षेप्य गति}\label{ch:g12:projectile-motion}

बग़ीचे की नली की हर धार हवा में वही वक्र खींचती है। वही वक्र टोकरी की ओर उठती
बास्केटबॉल खींचती है, लंबी कूद लगाता खिलाड़ी खींचता है, तोप का गोला खींचता है
--- यानी हर वह पिंड जो अपने भार के भरोसे छोड़ दिया जाए। यह अध्याय उस
वक्र को --- एक परवलय --- \omterm{thm:g12:newtons-laws:second}{न्यूटन के दूसरे नियम} से कलन की दो पंक्तियों में
कमा लेता है, उसके समीकरण से शिखर, \omterm{def:g12:projectile-motion:range}{उड़ान-काल} और \omterm{def:g12:projectile-motion:range}{परास} पढ़ लेता है, और फिर
हवा को भीतर आने देकर देखता है कि वह क्या बिगाड़ती है।

\section{मुक्त पतन}

\begin{definition}[मुक्त पतन]\label{def:g12:projectile-motion:freefall}
कोई पिंड \emph{मुक्त पतन}\index{मुक्त पतन} में तब है जब उस पर लगता एकमात्र
बल उसका भार $\vect P = m\vect g$ हो: न कोई आधार, न कोई डोरी, और वायु-प्रतिरोध
नगण्य। छोड़ा हुआ पत्थर या हाथ से निकल चुकी गेंद अच्छी परिशुद्धता तक मुक्त पतन
में है --- और ``पतन'' में ऊपर उठना भी शामिल है।
\end{definition}

\begin{theorem}[गैलीलियो का मुक्त पतन का नियम]\label{thm:g12:projectile-motion:galileo}
\index{गैलीलियो का मुक्त पतन का नियम}
\omterm{def:g12:projectile-motion:freefall}{मुक्त पतन} में सभी पिंडों का त्वरण एक ही होता है, उनका द्रव्यमान चाहे जो
हो:
\[
\vect a = \vect g, \qquad g = \qty{9.81}{m/s^2},
\]
और वह सीधे नीचे की ओर होता है: यानी \cref{ch:g10:universal-gravitation} की क्षेत्र-प्रबलता \qty{9.81}{N/kg}
अपनी दूसरी पोशाक में, $\unit{N/kg} = \unit{m/s^2}$।
\end{theorem}

\begin{proof}
\omterm{thm:g12:newtons-laws:second}{न्यूटन का दूसरा नियम} (\cref{ch:g12:newtons-laws}) एकमात्र बल के रूप में
भार के साथ: $m\vect a = m\vect g$, और द्रव्यमान कट जाता है। हथौड़ा और पंख केवल हवा के
कारण अलग-अलग गिरते हैं।
\end{proof}

\begin{proposition}[विराम से ऊर्ध्वाधर पतन]\label{prop:g12:projectile-motion:vertical}
\omterm{def:g10:relative-motion:frame}{विराम} से छोड़े जाने पर, जब गिरी हुई गहराई $h$ छोड़ने के बिंदु से नीचे की ओर
गिनी जाए, तो \omterm{def:g12:projectile-motion:freefall}{मुक्त पतन} में कोई पिंड यह मानता है:
\[
a = g, \qquad v = g t, \qquad h = \tfrac12\, g t^2 .
\]
\end{proposition}

\begin{proof}
$v(t)$ अचर $a = g$ का वह प्रतिअवकलज है जो $t = 0$ पर लुप्त हो जाता है
(\cref{ch:g12:kinematics-2d}); और $h(t)$ $v = gt$ का वह प्रतिअवकलज है जो $t = 0$ पर लुप्त होता है।
\end{proof}

\begin{example}[परिमाण की कोटियाँ]\label{ex:g12:projectile-motion:orders}
\omterm{def:g12:projectile-motion:freefall}{मुक्त पतन} के $1$, $2$, $3$ \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} बाद: $v = 9.8$, $19.6$, \qty{29.4}{m/s}
($35$, $71$, \qty{106}{km/h}) और $h = 4.9$, $19.6$, \qty{44.1}{m}। \qty{30}{m} ऊँची छत से:
$t = \sqrt{2h/g} \approx \qty{2.5}{s}$, और पहुँचते समय \qty{24}{m/s} $\approx \qty{87}{km/h}$। समय $\sqrt h$ की तरह बढ़ते हैं --- और कुछ
\omterm{def:g10:orders-of-magnitude:unit}{सेकंड} बाद हवा नगण्य बने रहने से इनकार कर देती है (अंतिम अनुभाग)।
\end{example}

\begin{example}[ऊर्ध्वाधर उछाल]\label{ex:g12:projectile-motion:throw}
$v_0$ से सीधे ऊपर फेंकिए, और अक्ष $Oy$ हाथ से ऊपर की ओर लीजिए: वही दो
प्रतिअवकलज $v = v_0 - gt$ और $y = v_0 t - \tfrac12 gt^2$ दे देते हैं। शिखर, जहाँ $v = 0$, $t = v_0/g$ पर आता है,
और ऊँचाई $h = v_0^2/2g$ --- यानी \cref{ch:g11:mechanical-energy} का ऊर्जा वाला उत्तर $v_0 = \sqrt{2gh}$ उलटी दिशा
में चलाया हुआ; और \emph{नीचे} फेंकने पर $v_0$ का चिह्न पलट जाता है। \qty{15}{m/s} के
लिए: शिखर \qty{1.53}{s} पर \qty{11.5}{m}, और वापस \qty{3.06}{s} पर, \qty{15}{m/s} से।
\end{example}

\section{प्रक्षेप्य: एक नियम, दो प्रतिअवकलज}

\begin{definition}[प्रक्षेप्य]\label{def:g12:projectile-motion:projectile}
\emph{प्रक्षेप्य}\index{प्रक्षेप्य} वह पिंड है जिसे वेग $\vect v_0$ से छोड़ दिया
जाए और फिर \omterm{def:g12:projectile-motion:freefall}{मुक्त पतन} पर छोड़ दिया जाए। छोड़ने के बिंदु पर अक्ष:
$Ox$ $\vect v_0$ के \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} तल में क्षैतिज, $Oy$ ऊपर की ओर \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर}; और
क्षैतिज से $\vect v_0$ का \emph{प्रक्षेप कोण}\index{प्रक्षेप कोण} $\alpha$ उसके
निर्देशांक $(v_0\cos\alpha,\; v_0\sin\alpha)$ तय कर देता है।
\end{definition}

\begin{theorem}[गति के समीकरण]\label{thm:g12:projectile-motion:equations}
मूल बिंदु से चाल $v_0$ और कोण $\alpha$ पर छोड़े गए \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} का हर
क्षण $\vect a = (0,\,-g)$ होता है, इसलिए
\[
\vect v(t) = \bigl(v_0\cos\alpha,\; v_0\sin\alpha - g t\bigr),
\qquad
\begin{cases}
x(t) = v_0\cos\alpha\; t,\\[2pt]
y(t) = v_0\sin\alpha\; t - \tfrac12\, g t^2 .
\end{cases}
\]
\end{theorem}

\begin{proof}
\omterm{thm:g12:newtons-laws:second}{न्यूटन का दूसरा नियम} $\vect a = \vect g = (0,-g)$ देता है। अक्ष-दर-अक्ष प्रतिअवकलज:
आरंभिक मान $(v_0\cos\alpha,\, v_0\sin\alpha)$ के साथ $\vect a$ से $\vect v(t)$, फिर $\vect v$ से और $(0,0)$ से
स्थिति।
\end{proof}

\begin{proposition}[दोनों गतियों की स्वतंत्रता]\label{prop:g12:projectile-motion:independence}
क्षैतिज गति \omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान} है --- $v_x$ कभी बदलता ही नहीं --- और \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} गति ठीक
पिछले अनुभाग का \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} उछाल है: किसी भी समीकरण में दूसरे का निर्देशांक आता
ही नहीं। विशेष रूप से, एक ही ऊँचाई से एक ही क्षण क्षैतिज दिशा में दागी गई गेंद
और छोड़ी गई गेंद हर समय एक-दूसरे के बराबर ऊँचाई पर रहती हैं --- और साथ-साथ
गिरती हैं।
\end{proposition}

\begin{proof}
\cref{thm:g12:projectile-motion:equations} में $x(t)$ केवल $v_0\cos\alpha$ को शामिल करता है और $y(t)$ केवल $v_0\sin\alpha$ तथा
$g$ को; और दोनों गेंदों का $v_y(0) = 0$ है, इसलिए दोनों $h = \tfrac12 gt^2$ मानती हैं।
\end{proof}

\begin{remark}[गिरता बंदर]\label{rem:g12:projectile-motion:monkey}
कोई चिड़ियाघर-रक्षक तीर सीधे बंदर पर \emph{ताककर} चलाता है, और बंदर गोली छूटते
ही डाल छोड़ देता है। बुरी चाल: तीर और बंदर, दोनों अपनी गुरुत्व-रहित स्थितियों
--- यानी ताक की रेखा और डाल --- से $\tfrac12 gt^2$ नीचे गिरते हैं, इसलिए वे ठीक मिल
जाते हैं, तीर की चाल चाहे जो हो (\cref{exo:g12:projectile-motion:15})।
\end{remark}

\section{परवलय, पूरा पढ़ा हुआ}

\begin{theorem}[गतिपथ]\label{thm:g12:projectile-motion:trajectory}
$x(t)$ और $y(t)$ के बीच से $t$ हटा दीजिए: \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} ($0 \leq \alpha < \ang{90}$)
का \omterm{def:g10:relative-motion:trajectory}{गतिपथ} नीचे की ओर खुलता वह परवलय है
\[
y = x\tan\alpha \;-\; \frac{g}{2 v_0^2 \cos^2\alpha}\, x^2 .
\]
\end{theorem}

\begin{proof}
$t = x/(v_0\cos\alpha)$ को \cref{thm:g12:projectile-motion:equations} के $y(t)$ में रख दीजिए।
\end{proof}

\begin{definition}[परास]\label{def:g12:projectile-motion:range}
समतल ज़मीन पर किसी \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} का \emph{परास}\index{परास} $R$ वह
क्षैतिज दूरी है जो वह अपनी छोड़ने वाली ऊँचाई पर लौटने तक तय करता है; और
\emph{उड़ान-काल}\index{उड़ान-काल} $t_f$ उसी सफ़र की अवधि।
\end{definition}

\begin{proposition}[शिखर, उड़ान-काल, परास]\label{prop:g12:projectile-motion:range}
समतल ज़मीन पर, $0 < \alpha < \ang{90}$ के साथ:
\[
t_f = \frac{2 v_0 \sin\alpha}{g}, \qquad
h_{\text{शीर्ष}} = \frac{(v_0\sin\alpha)^2}{2g}
\ \ \text{जो } t_f/2 \text{ पर मिलता है}, \qquad
R = \frac{v_0^2 \sin(2\alpha)}{g}.
\]
\omterm{def:g12:projectile-motion:range}{परास} $\alpha = \ang{45}$ पर सबसे बड़ा होता है, जहाँ $R = v_0^2/g$; और \omterm{prop:g11:color-light-sources:complementary}{पूरक} कोण $\alpha$
तथा $\ang{90} - \alpha$ वही \omterm{def:g12:projectile-motion:range}{परास} देते हैं।
\end{proposition}

\begin{proof}
$y = 0$ से $t\,(v_0\sin\alpha - \tfrac12 gt) = 0$ मिलता है: यानी छोड़ना, और $t_f$। शिखर, $v_y = 0$, $t = v_0\sin\alpha/g = t_f/2$
पर बैठता है; उसे $y(t)$ में रख दीजिए। फिर $R = x(t_f) = 2v_0^2\sin\alpha\cos\alpha/g =
v_0^2\sin(2\alpha)/g$। ज्या $2\alpha = \ang{90}$ पर चरम पर
होती है, और $\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180} - 2\alpha) =
\sin(2\alpha)$।
\end{proof}

\begin{example}[गोल-किक]\label{ex:g12:projectile-motion:kick}
गोलरक्षक $v_0 = \qty{25}{m/s}$, $\alpha = \ang{45}$ पर मारता है: $R = 25^2/9.81 \approx \qty{64}{m}$; असली किक \qty{55}{m} के आसपास
गिरती हैं --- यानी हवा की कटौती।
\end{example}

\begin{method}[प्रक्षेप्य का बहीखाता]\label{met:g12:projectile-motion:method}
\begin{enumerate}
  \item छोड़ने के बिंदु पर अक्ष, $Ox$ क्षैतिज और $Oy$ ऊपर; और $\vect v_0$
        के निर्देशांक।
  \item \omterm{def:g12:projectile-motion:freefall}{मुक्त पतन} (केवल भार) जाँच लीजिए: तब $\vect a = (0,-g)$;
        दो बार प्रतिअवकलज, और हर बार आरंभिक शर्तें।
  \item \emph{जगह} का प्रश्न हो: $t$ हटा दीजिए (\omterm{def:g10:relative-motion:trajectory}{गतिपथ})।
        \emph{क्षण} का हो: $t$ रखे रहिए।
  \item शब्दावली: शिखर $\Leftrightarrow v_y = 0$; छोड़ने की ऊँचाई $\Leftrightarrow y = 0$; $d \Leftrightarrow x = d$ पर दीवार;
        $\Leftrightarrow y = -\ell$ से $\ell$ नीचे फ़र्श।
  \item समझदारी की जाँच: उड़ान शिखर के दोनों ओर सममित; $R \leq v_0^2/g$; और हवा हमेशा
        घटाती ही है।
\end{enumerate}
\end{method}

\section{हवा क्या बदल देती है}

\begin{definition}[सीमांत चाल]\label{def:g12:projectile-motion:terminal}
वायु-प्रतिरोध वेग का विरोध करता है और चाल के साथ बढ़ता है। इसलिए गिरता
पिंड तभी तक त्वरित होता है जब तक कर्षण भार को नहीं काट देता; उसके बाद
$\vect a = \vect 0$: वह अचर \emph{सीमांत चाल}\index{सीमांत चाल} से गिरता है।
\end{definition}

\begin{example}[सीमांत चाल की कोटियाँ]\label{ex:g12:projectile-motion:airorders}
पूरे फैले हुए छत्रीधारी की: लगभग \qty{55}{m/s} (\qty{200}{km/h})। बारिश की बूँद की: लगभग
\qty{9}{m/s} --- निर्वात में \qty{2000}{m} गिरने पर वह $\sqrt{2gh} \approx \qty{2.0e2}{m/s}$ से पहुँचती। पंख की: लगभग
\qty{1}{m/s}, और वह भी कुछ सेंटीमीटर के भीतर ही पा ली जाती है --- उसकी सुस्ती का
यही रहस्य है: वह अपना पूरा पतन \omterm{def:g12:projectile-motion:terminal}{सीमांत चाल} \emph{पर} बिताता है, जबकि
हथौड़ा अपनी \omterm{def:g12:projectile-motion:terminal}{सीमांत चाल} के आसपास तक नहीं पहुँचता।
\end{example}

\begin{remark}[असली प्रक्षेप्य-विद्या]\label{rem:g12:projectile-motion:drag}
कर्षण पूरी उड़ान भर वेग के विरुद्ध धकेलता रहता है: \omterm{def:g12:projectile-motion:range}{परास} छोटा हो जाता
है, शिखर नीचे आ जाता है, और सममिति टूट जाती है --- उतार चढ़ाई से अधिक खड़ा, और
लंबे शॉट परवलय से साफ़ दिखते हुए पीछे। ईमानदार, मात्रात्मक बरताव एक अवकल समीकरण
है जो वर्ष 1 के खंड में हल किया गया है; इस वर्ष परवलय घने और मध्यम तेज़
प्रक्षेप्यों के लिए सही प्रतिरूप है --- \qty{10}{m/s} पर बास्केटबॉल के लिए
हाँ, \qty{70}{m/s} पर गोल्फ़ की गेंद के लिए नहीं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:1}
पुल से छोड़ा गया पत्थर \qty{2.0}{s} बाद पानी से टकराता है: पुल की ऊँचाई, और टकराने
की चाल \unit{m/s} तथा \unit{km/h} में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:2}
चंद्रमा पर किसी अंतरिक्ष यात्री ने हथौड़ा और पंख एक साथ छोड़े: वे साथ-साथ गिरे।
क्यों --- और पृथ्वी पर पंख इतनी बुरी तरह क्यों हारता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:3}
हवा छोड़कर \qty{1.0}{m}, \qty{10}{m}, \qty{100}{m} से गिरने का समय और पहुँचने की चाल
निकालिए। समय ऊँचाई के साथ किस तरह बढ़ते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:4}
गेंद सीधे ऊपर \qty{15}{m/s} से फेंकी जाती है: शिखर की ऊँचाई और समय, हाथ में लौटने
तक कुल समय, और लौटने की चाल?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:5}
कोई कंचा \qty{0.80}{m} ऊँची मेज़ से \qty{2.5}{m/s} पर लुढ़ककर गिर जाता है। गिरने का समय और
किनारे से गिरने की दूरी निकालिए। क्या गिरने का समय कंचे की चाल पर निर्भर करता
है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:6}
पाठ के पाँच-कोण वाले चित्र से: कौन-से \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप कोण} युग्म अपना
\omterm{def:g12:projectile-motion:range}{परास} साझा करते हैं, और कौन-सा कोण जीतता है? अधिकतम से कम
\omterm{def:g12:projectile-motion:range}{परास} के लिए दोनों संभव कोणों में से कौन-सा अधिक देर उड़ता है? $t_f$
से कारण दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:7}
\qty{20}{m} ऊँची बालकनी से गेंद सीधे \emph{नीचे} \qty{5.0}{m/s} से फेंकी जाती है। ज़मीन
तक का समय और टकराने की चाल निकालिए; और दूसरे को ऊर्जा-संतुलन से जाँच
लीजिए (\cref{ch:g11:mechanical-energy})।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:8}
कोई \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} समतल ज़मीन से \qty{40}{m/s}, $\alpha = \ang{30}$ पर निकलता है:
\omterm{def:g12:projectile-motion:range}{उड़ान-काल}, शिखर की ऊँचाई और \omterm{def:g12:projectile-motion:range}{परास}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:9}
लंबी कूद के खिलाड़ी को \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} मानिए: उछाल \qty{10.5}{m/s}, $\alpha = \ang{20}$ पर।
\omterm{def:g12:projectile-motion:range}{परास} निकालिए; चैंपियन लगभग \qty{9}{m} कूदते हैं --- बिंदु-प्रतिरूप जिन
दो बातों को छोड़ देता है, उनके नाम बताइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:10}
समतल ज़मीन से \qty{1.2}{m} ऊपर एक गोली छोड़ी जाती है और उसके जैसी दूसरी क्षैतिज
दिशा में \qty{300}{m/s} से दागी जाती है। कौन-सी पहले गिरेगी? गिरने का समय और दागी गई
गोली के गिरने की दूरी निकालिए। यहाँ किस सिद्धांत की परीक्षा हो रही है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:11}
बीजगणित से दिखाइए कि $\alpha$ और $\ang{90} - \alpha$ अपना \omterm{def:g12:projectile-motion:range}{परास} साझा करते हैं, फिर
वे दोनों कोण निकालिए जो \qty{25}{m/s} के \omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} को \qty{50}{m} दूर गिरा
देते हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:12}
बग़ीचे की नली से पानी \qty{8.0}{m/s} से निकलता है: सबसे बड़ी क्षैतिज पहुँच? सबसे बड़ी
\omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} ऊँचाई? और \qty{5.0}{m} दूर की क्यारी सींचते नली के दोनों कोण?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:13}
कोई गेंद ज़मीन से \qty{18}{m/s}, $\alpha = \ang{35}$ पर \qty{15}{m} दूर लगी \qty{4.0}{m} ऊँची बाड़ की ओर
मारी जाती है। \omterm{def:g10:relative-motion:trajectory}{गतिपथ} के समीकरण से तय कीजिए कि वह उसे पार करती है या
नहीं, और कितने से।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:14}
\qty{80}{kg} का छत्रीधारी चपटा फैलकर \omterm{def:g12:projectile-motion:terminal}{सीमांत चाल} \qty{55}{m/s} से गिरता है।
(क) कर्षण-बल? (ख) छलाँग से आगे $v(t)$ का आकार खींचिए। (ग) बारिश की
बूँद \qty{2.0}{km} से गिरती है: निर्वात में पहुँचने की चाल बनाम उसकी असली \qty{9}{m/s}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:15}
गिरता बंदर, सिद्ध करके: कोई तीर मूल बिंदु से ठीक उस बंदर पर ताककर छूटता है जो
दूरी $D$, ऊँचाई $H$ ($\tan\alpha = H/D$) पर है और $t = 0$ पर गिर पड़ता है।
दिखाइए कि तीर जब $x = D$ तक पहुँचता है तब उसकी ऊँचाई बंदर की ऊँचाई के बराबर
होती है --- $v_0$ चाहे जो हो।
\end{exercise}

\section{समस्या: बास्केटबॉल का आख़िरी घंटी वाला शॉट}

\begin{problem}[{सप्ताहांत समस्या --- आख़िरी घंटी वाला बास्केटबॉल शॉट: तीर-दर-तीर
चीरा गया एक फ़्री थ्रो, सेंटीमीटरों से बची उँगलियाँ, पूरे मैदान भर का वह उछाल
जो बाँह की ताक़त के तराज़ू पर तौला जाता है, तख़्ते के ऊपर से एक लॉब --- और खेल
के सबसे नाटकीय शॉट पर एक फ़ैसला}]
\label{pb:g12:projectile-motion:1}
बास्केटबॉल की टोकरी का घेरा फ़र्श से \qty{3.05}{m} ऊपर एक क्षैतिज छल्ला है; नोरा हर
शॉट \qty{2.00}{m} पर छोड़ती है, इसलिए छोड़ने के बिंदु पर अक्ष रखने पर ($Ox$
टोकरी की ओर क्षैतिज, $Oy$ ऊपर) घेरे का केंद्र मूल बिंदु से \qty{1.05}{m} ऊपर
बैठता है। जहाँ न कहा जाए, वहाँ हवा छोड़ दी गई है।

\medskip\noindent\textbf{भाग I --- फ़्री थ्रो।} नोरा $d = \qty{4.20}{m}$ (क्षैतिज) से
\omterm{def:g12:projectile-motion:projectile}{प्रक्षेप कोण} $\alpha = \ang{50}$ पर शॉट लगाती है।
\begin{enumerate}
  \item छोड़ने के बाद कौन-से बल लगते हैं? प्रयुक्त नियम का हवाला देते
        हुए $\vect a$ दीजिए।
  \item $\vect v(t)$ के निर्देशांक निकालिए।
  \item $x(t)$ और $y(t)$ निकालिए।
  \item $t$ हटाकर \omterm{def:g10:relative-motion:trajectory}{गतिपथ} $y(x)$ प्राप्त कीजिए।
  \item गेंद को घेरे के केंद्र $y(d) = \qty{1.05}{m}$ से गुज़रना है: $v_0$ के लिए हल कीजिए,
        \unit{m/s} और \unit{km/h} में।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- उँगलियों के ऊपर से, छल्ले के भीतर।}
\begin{enumerate}[resume]
  \item गेंद को घेरे तक पहुँचने में कितना समय लगता है?
  \item शिखर का समय और ऊँचाई निकालिए; जाँचिए कि गेंद घेरे पर पहुँचते समय पहले
        से उतर रही है।
  \item नोरा से \qty{0.90}{m} आगे खड़ा एक विरोधी उँगलियाँ \qty{2.60}{m} तक तान देता है।
        गेंद उनसे कितना ऊपर से निकलती है?
  \item घेरे पर $v_x$ और $v_y$, वहाँ की चाल, और वेग का क्षैतिज से नीचे
        का कोण निकालिए।
  \item गेंद को \emph{उतरते हुए} पहुँचना ही क्यों चाहिए --- और खड़ी ढाल पर
        क्यों?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- पूरे मैदान भर का उछाल।} घंटी बजने ही
वाली है, और नोरा दूर वाले घेरे से \qty{26.0}{m} पर है।
\begin{enumerate}[resume]
  \item \qty{26.0}{m} के मुक़ाबले \qty{1.05}{m} की चढ़ाई छोटी है: समतल-ज़मीन वाले प्रतिरूप
        का औचित्य बताइए, और $y(t_f) = 0$ से \omterm{def:g12:projectile-motion:range}{उड़ान-काल} निकालिए।
  \item $R = v_0^2\sin(2\alpha)/g$ और सबसे बड़े \omterm{def:g12:projectile-motion:range}{परास} का कोण निकालिए।
  \item $\alpha = \ang{45}$ पर \qty{26.0}{m} तय करने के लिए ज़रूरी $v_0$ निकालिए।
  \item कोई मज़बूत खिलाड़ी बास्केटबॉल को लगभग \qty{17.5}{m/s} से फेंक सकता है। उसका
        अधिकतम \omterm{def:g12:projectile-motion:range}{परास} क्या है --- क्या यह शॉट मनुष्य के बस का है?
  \item \qty{17.5}{m/s} पर वे \emph{दोनों} कोण निकालिए जो गेंद को \qty{26.0}{m} दूर गिराते
        हैं, और उनके \omterm{def:g12:projectile-motion:range}{उड़ान-काल} भी: घड़ी पर \qty{2.0}{s} बचे हों तो कौन-सा घंटी से
        पहले पहुँचता है?
\end{enumerate}

\medskip\noindent\textbf{भाग IV --- तख़्ते के ऊपर से लॉब।} पिछली रेखा के पीछे
से नोरा लॉब मारती है: छोड़ने का बिंदु घेरे के केंद्र से \qty{6.0}{m} दूर, $\alpha = \ang{75}$;
तख़्ते का तल उसके शॉट को घेरे के केंद्र से \qty{0.45}{m} पहले काटता है, और उसका ऊपरी
किनारा फ़र्श से \qty{3.95}{m} ऊपर है।
\begin{enumerate}[resume]
  \item वह $v_0$ निकालिए जो गेंद को घेरे के केंद्र से नीचे गिरा दे।
  \item तख़्ते के तल पर ($x = \qty{5.55}{m}$) गेंद की ऊँचाई निकालिए: क्या वह ऊपरी किनारे
        से पार निकलती है, और कितने से?
  \item शिखर की ऊँचाई और \omterm{def:g12:projectile-motion:range}{उड़ान-काल} निकालिए --- भीड़ को यह इतना प्यारा क्यों
        लगता है?
  \item गुणात्मक रूप से, वायु-प्रतिरोध इन तीनों शॉटों में क्या बदल देता है,
        और सबसे बड़ा ख़तरा किसे है?
  \item \emph{आख़िरी घंटी वाला फ़ैसला}, दो वाक्यों में: पूरे मैदान वाला शॉट
        जितनी चाल माँगता है उसकी तुलना बाँह जितनी दे सकती है उससे कीजिए,
        हाशिया बताइए, और फ़ैसला सुनाइए --- संभव, या मिथक?
\end{enumerate}
\end{problem}
