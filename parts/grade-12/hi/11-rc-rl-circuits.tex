\chapter{RC और RL परिपथ}\label{ch:g12:rc-rl-circuits}

कैमरे का शटर आधा दबाइए: एक पतली सीटी दो \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} तक चढ़ती है, एक बत्ती हरी
हो जाती है --- और तभी फ़्लैश चलेगा, जो दो \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में इकट्ठा किया हुआ सब कुछ एक
मिलीसेकंड में उड़ेल देगा। यह अध्याय उन दो अवयवों से मिलता है जो परिपथों को
याददाश्त और घड़ी दे देते हैं --- \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} और \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} --- और
भौतिकी के पहले अवकल समीकरणों से, जिन्हें इसी वर्ष के अवकलजों से जाँचा जा सकता
है।

\section{संधारित्र}

\begin{definition}[संधारित्र और धारिता]\label{def:g12:rc-rl-circuits:capacitor}
\emph{संधारित्र}\index{संधारित्र} \omterm{def:g11:lenses-and-eye:lens}{पतले} विद्युत्रोधी से अलग की गई धातु
की दो आमने-सामने रखी पट्टियों का युग्म है। किसी स्रोत से जोड़ने पर पट्टियाँ
विपरीत आवेश $+q$ और $-q$ ले लेती हैं --- और अंतराल में एक \omterm{def:g11:electric-gravitational-fields:efield}{विद्युत
क्षेत्र} भर जाता है (\cref{ch:g11:electric-gravitational-fields}) --- और पट्टियों के बीच का \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} $u$
संचित आवेश के समानुपाती होता है:
\[ q = C\,u . \]
अचर $C$ \emph{धारिता}\index{धारिता} है, जिसे \emph{फैराड}\index{फैराड}
(\unit{F}) में नापते हैं: $\qty{1}{F} = \qty{1}{C/V}$।
\end{definition}

\begin{example}[परिमाण की कोटियाँ]\label{ex:g12:rc-rl-circuits:orders}
एक \omterm{def:g12:rc-rl-circuits:capacitor}{फैराड} बहुत बड़ा है --- यानी प्रति \omterm{def:g11:circuits-and-power:voltage}{वोल्ट} एक \omterm{def:g11:fundamental-interactions:charge}{कूलॉम}
जमा। रेडियो परिपथों के सिरैमिक: \unit{pF} से \unit{nF}; शक्ति-आपूर्तियों और
फ़्लैश के इलेक्ट्रोलाइटिक: \unit{\micro F} से \unit{mF}; और सुपर-संधारित्र: हज़ारों
\omterm{def:g12:rc-rl-circuits:capacitor}{फैराड}।
\end{example}

\begin{proposition}[संधारित्र में जाती धारा]\label{prop:g12:rc-rl-circuits:cdudt}
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} की पट्टी पर पहुँचती धारा वह दर है जिससे उसका आवेश बढ़ता है:
\[
i = \frac{dq}{dt} = C\,\frac{du}{dt} ,
\]
जहाँ छोटे अक्षर समय के साथ बदलती राशियाँ दिखाते हैं ($U$, $I$ स्थिर
रहते हैं, \cref{ch:g11:circuits-and-power})।
\end{proposition}

\begin{proof}
धारा प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} पहुँचाया गया आवेश है (\cref{def:g11:circuits-and-power:current}): सिकुड़ते अंतराल पर $i$
$q$ का अवकलज है; और $C$ के अचर रहते $q = Cu$ से $C\,du/dt$ मिल जाता है।
\end{proof}

\begin{remark}[दो परिणाम]\label{rem:g12:rc-rl-circuits:continuity}
यदि $u$ अचर हो तो $i = 0$: यानी आवेशित \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} कोई स्थिर धारा नहीं
जाने देता। और $u$ कभी छलाँग नहीं लगा सकता --- छलाँग के लिए अनंत धारा
चाहिए: \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} का \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} ही परिपथ की याददाश्त है।
\end{remark}

\section{संधारित्र को आवेशित करना: RC परिपथ}

एक ही फंदा --- स्रोत $\mathcal{E}$ (\cref{def:g11:circuits-and-power:emf}), स्विच, प्रतिरोधक, ख़ाली
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र}: $t = 0$ पर $K$ बंद कीजिए।

\end{circuitikz}

{\small RC आवेशन-परिपथ: $K$ बंद करते ही स्रोत आवेश को $R$ से होकर
$C$ की पट्टियों पर धकेल देता है।}
\end{omfigure}

\begin{theorem}[RC परिपथ का आवेशन-नियम]\label{thm:g12:rc-rl-circuits:charging}
\index{RC परिपथ}
स्विच बंद होने के बाद \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} का \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} यह मानता है, और
फिर $u(0) = 0$ से यह निकलता है:
\[
RC\,\frac{du}{dt} + u = \mathcal{E},
\qquad
u(t) = \mathcal{E}\left(1 - e^{-t/\tau}\right),
\qquad \tau = RC .
\]
\end{theorem}

\begin{proof}
फंदे के अनुदिश \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} जुड़ते हैं (\cref{prop:g11:circuits-and-power:laws}): हर क्षण $\mathcal{E} = u_R + u = Ri + u$; उसमें
$i = C\,du/dt$ (\cref{prop:g12:rc-rl-circuits:cdudt}) रख दीजिए। अब उम्मीदवार को \emph{जाँचिए}: $du/dt = (\mathcal{E}/\tau)e^{-t/\tau}$, इसलिए
$RC\,(\mathcal{E}/RC)\,e^{-t/\tau} + \mathcal{E}(1 - e^{-t/\tau}) =
\mathcal{E}$, जो हर $t$ के लिए सही है; और $u(0) = \mathcal{E}(1-1) = 0$। और यह कि \emph{कोई दूसरा} वक्र
इस समीकरण तथा इस शुरुआत पर नहीं बैठता, यह इसी वर्ष के गणित के अवकल समीकरणों
वाले पाठ में सिद्ध किया गया है।
\end{proof}

\begin{definition}[कालांक]\label{def:g12:rc-rl-circuits:tau}
\omterm{thm:g12:rc-rl-circuits:charging}{RC परिपथ} का \emph{कालांक}\index{कालांक} $\tau = RC$ है। मात्रक:
$\unit{\ohm} \times \unit{F} =
(\unit{V/A})(\unit{C/V}) = \unit{C/A} = \unit{s}$ --- यानी परिपथ की अपनी लय।
\end{definition}

\begin{method}[वक्र पर $\tau$ पढ़ना]\label{met:g12:rc-rl-circuits:reading}
आलेख से सीधे तीन तुल्य पाठ:
\begin{enumerate}
  \item \emph{मूल बिंदु पर स्पर्शी:} $u'(0) = \mathcal{E}/\tau$, इसलिए आरंभिक स्पर्शी
        अनंतस्पर्शी तक $t = \tau$ पर पहुँच जाती है;
  \item \emph{63\%:} $u(\tau) = \mathcal{E}(1 - e^{-1}) \approx
        0.63\,\mathcal{E}$;
  \item \emph{95\%:} $u(3\tau) \approx 0.95\,\mathcal{E}$ --- यानी व्यवहार में आवेशित ($5\tau$: $99\%$)।
\end{enumerate}
\end{method}

\begin{proposition}[विसर्जन]\label{prop:g12:rc-rl-circuits:discharge}
$U_0$ तक आवेशित \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} को अकेले प्रतिरोधक $R$ पर बंद कर देने पर
वह $RC\,du/dt + u = 0$ मानता है और इस पर चलता है:
\[
u(t) = U_0\,e^{-t/\tau}, \qquad \tau = RC
\]
--- $\tau$ पर $37\%$ बचता है, और $3\tau$ पर $5\%$।
\end{proposition}

\begin{proof}
वही फंदा-नियम, बस कोई स्रोत नहीं; और फिर वही प्रतिस्थापन: $RC(-U_0/\tau)e^{-t/\tau} + U_0 e^{-t/\tau} = 0$, जहाँ $u(0) = U_0$।
\end{proof}

\begin{example}[फ़्लैश की संख्याएँ]\label{ex:g12:rc-rl-circuits:flashrc}
कोई फ़ोटो-फ़्लैश $C = \qty{800}{\micro F}$ को $R = \qty{1.0}{k\ohm}$ से होकर आवेशित करता है: $\tau = 10^3 \times 8.0\times10^{-4} =
\qty{0.80}{s}$, इसलिए
``तैयार'' बत्ती --- जो $95\%$ पर चलने के लिए जुड़ी है --- $3\tau \approx \qty{2.4}{s}$ प्रतीक्षा
करती है: यही वह सीटी है जो कान में खटकती है। और फ़्लैश-नली के महज़ \qty{0.50}{\ohm} से
होकर चलाने पर \emph{वही} \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} $\tau' = \qty{0.40}{ms}$ के साथ ख़ाली हो जाता है:
यानी भीतर आने से दो हज़ार गुना तेज़ बाहर।
\end{example}

\section{कुंडली: प्रेरकत्व और RL परिपथ}

\begin{definition}[प्रेरक और प्रेरकत्व]\label{def:g12:rc-rl-circuits:inductor}
\emph{प्रेरक}\index{प्रेरक} तार की एक \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} है: उसकी धारा उसके अपने
फेरों में \omterm{def:g11:magnetic-fields:bfield}{चुंबकीय क्षेत्र} पिरो देती है (\cref{ch:g11:magnetic-fields}); और जब भी वह
\emph{बदलता} है, \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} के सिरों पर एक \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} प्रकट हो
जाता है:
\[
u = L\,\frac{di}{dt} .
\]
अचर $L$ \emph{प्रेरकत्व}\index{प्रेरकत्व} है, जिसे \emph{हेनरी}\index{हेनरी}
(\unit{H}) में नापते हैं: $\qty{1}{H} = \qty{1}{V.s/A}$।
\end{definition}
\admitted

\begin{remark}[कुंडली बदलाव का विरोध करती है]\label{rem:g12:rc-rl-circuits:oppose}
बदलती धारा \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} क्यों बनाती है --- यानी प्रेरण --- यह
विश्वविद्यालय के खंडों में निकाला गया है; यहाँ हम नियम और उसका स्वभाव ले लेते
हैं: \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} अपनी धारा के \emph{बदलावों का विरोध करती} है। दर्पण-जुड़वाँ:
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} का $u$ छलाँग नहीं लगा सकता, \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} का $i$
नहीं लगा सकता।
\end{remark}

\begin{omfigure}
\begin{circuitikz}[font=\small]
  \draw (0,0) to[battery1, l=$\mathcal{E}$] (0,2.4) to[nos, l=$K$] (2.1,2.4)
    to[R=$R$] (4.2,2.4) -- (5.4,2.4) to[L=$L$, i=$i$, v=$u$] (5.4,0) -- (0,0);
\end{circuitikz}

{\small \omterm{prop:g12:rc-rl-circuits:rl}{RL परिपथ}: $K$ बंद होने के बाद \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} धारा को
केवल अपनी ही चाल $\tau = L/R$ से चढ़ने देती है।}
\end{omfigure}

\begin{proposition}[RL परिपथ में धारा का चढ़ाव]\label{prop:g12:rc-rl-circuits:rl}
\index{RL परिपथ}
स्रोत $\mathcal{E}$, प्रतिरोधक $R$ और \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} $L$ को श्रेणीक्रम में
जोड़कर स्विच बंद करने पर $\mathcal{E} = Ri + L\,di/dt$ मिलता है, यानी
\[
\frac{L}{R}\,\frac{di}{dt} + i = \frac{\mathcal{E}}{R},
\qquad\text{जिसका हल है}\quad
i(t) = \frac{\mathcal{E}}{R}\left(1 - e^{-t/\tau}\right),
\quad \tau = \frac{L}{R} .
\]
\end{proposition}

\begin{proof}
फिर वही फंदा-नियम; और यह \cref{thm:g12:rc-rl-circuits:charging} वाला \emph{वही समीकरण} है, जिसमें $u \to i$,
$\mathcal{E} \to \mathcal{E}/R$, $RC \to L/R$ --- वही प्रतिस्थापन हल को जाँच देता है, और \cref{met:g12:rc-rl-circuits:reading} का हर
पाठ यहाँ भी उतर आता है।
\end{proof}

\begin{example}[विद्युत्चुंबक जागता है]\label{ex:g12:rc-rl-circuits:rl}
रिले की \omterm{def:g11:magnetic-fields:solenoid}{कुंडली}, $L = \qty{0.30}{H}$ और $R = \qty{6.0}{\ohm}$, \qty{12}{V} की आपूर्ति पर: अंतिम धारा
$\mathcal{E}/R = \qty{2.0}{A}$, और चाल $\tau = 0.30/6.0 = \qty{50}{ms}$ --- यानी $3\tau \approx
\qty{0.15}{s}$ में जाग जाती है।
\end{example}

\begin{remark}[खोलने पर उठती चिंगारी]\label{rem:g12:rc-rl-circuits:spark}
जीवित \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} पर स्विच खोलिए और $i$ को मिलीसेकंड के अंश भर में
शून्य पर गिरना पड़ता है: $di/dt$ विशाल हो जाता है और $u = L\,di/dt$ सैकड़ों या हज़ारों
\omterm{def:g11:circuits-and-power:voltage}{वोल्ट} तक पहुँच जाता है --- और संपर्क पर चिंगारी कूद पड़ती है। कार की
इग्निशन \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} यही काम जान-बूझकर करती है, मिनट में हज़ारों बार।
\end{remark}

\section{संचित ऊर्जा}

\begin{proposition}[संधारित्र में ऊर्जा, कुंडली में ऊर्जा]\label{prop:g12:rc-rl-circuits:energy}
\index{ऊर्जा!संधारित्र की}\index{ऊर्जा!कुंडली की}
$u$ के \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} तक आवेशित \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} $C$ और $i$
धारा ले जाती \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} $L$ इतनी ऊर्जा संचित रखते हैं:
\[
E_C = \tfrac12\,C u^2 = \frac{q^2}{2C},
\qquad
E_L = \tfrac12\,L i^2 .
\]
\end{proposition}

\begin{proof}
पहले से $u$ \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} पर पड़ी पट्टियों पर छोटा आवेश $dq$ उतारने
का दाम $u\,dq$ है (\cref{def:g11:circuits-and-power:voltage}); और दाम एक पर एक चढ़ाते जाने पर कुल दाम रेखा
$u = q/C$ के नीचे का क्षेत्रफल --- यानी नीचे का त्रिभुज --- हो जाता है, इसलिए
$E_C = \tfrac12 qu = \tfrac12 Cu^2$। \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} के लिए वही त्रिभुज ($i$ को $di$ बढ़ाने का दाम
$Li\,di$) $E_L = \tfrac12 Li^2$ देता है।
\end{proof}

\begin{example}[ऊर्जा की परिमाण-कोटियाँ]\label{ex:g12:rc-rl-circuits:energies}
\cref{ex:g12:rc-rl-circuits:flashrc} का फ़्लैश-संधारित्र: $\tfrac12 \times 8.0\times10^{-4} \times 300^2 = \qty{36}{J}$। \qty{2.7}{V} पर \qty{3000}{F} का
सुपर-संधारित्र: \qty{1.1e4}{J} --- यानी एक AA सेल भर। और \qty{10}{A} पर \qty{10}{mH} की
\omterm{def:g11:magnetic-fields:solenoid}{कुंडली}: \qty{0.50}{J}। बैटरियों से कम --- पर एक मिलीसेकंड में छोड़ा जा सकने
वाला।
\end{example}

\begin{remark}[ये दोनों समय-रक्षक किस काम के हैं]\label{rem:g12:rc-rl-circuits:applications}
\index{समतलन}\index{कालनियंत्रण परिपथ}
धीरे भरना, तेज़ उड़ेलना --- यही कैमरे के फ़्लैश और डीफ़िब्रिलेटर को
शक्ति देता है। $\tau = RC$ का \omterm{def:g12:mechanical-waves:delay}{विलंब} \omterm{rem:g12:rc-rl-circuits:applications}{कालनियंत्रण परिपथ}
चलाता है: रुक-रुककर चलते वाइपर, टिमटिमाते संकेतक, सीढ़ियों की बत्तियाँ। आपूर्ति
के समांतर जुड़ा \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} को \emph{चिकना} कर देता है
($u$ छलाँग नहीं लगा सकता); और श्रेणीक्रम में जुड़ी \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} धारा को
($i$ छलाँग नहीं लगा सकता)। और \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} तथा \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} साथ
मिलकर एक \omterm{def:g12:oscillators-and-time:oscillator}{दोलित्र} बना देते हैं (\cref{ch:g12:rlc-oscillations})।
\end{remark}

उसका मान तथा बड़े $t$ की सीमा भौतिक रूप से क्या दर्ज करते हैं?

\begin{exercise}[$\star\star$]\label{exo:g12:rc-rl-circuits:9}

\omterm{def:g11:magnetic-fields:solenoid}{कुंडली} पर स्विच \emph{खोलने} से चिंगारी क्यों उठती है, बंद करने से

\section{समस्या: कैमरे का फ़्लैश}

\begin{problem}[{सप्ताहांत समस्या --- कैमरे का फ़्लैश: दो सेकंड की सीटी, एक
मिलीसेकंड का जलवा, और दीवार पर टँगा वह चचेरा भाई जो दिल फिर से चला देता है ---
और तीनों को एक ही चरघातांक चलाता है}]
\label{pb:g12:rc-rl-circuits:1}
कोई छोटा फ़्लैश-यंत्र $C = \qty{800}{\micro F}$ के \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} में ऊर्जा
संचित रखता है, जिसे $R = \qty{1.0}{k\ohm}$ से होकर उस परिवर्तित्र से आवेशित किया
जाता है जिसे आदर्श स्रोत $\mathcal{E} = \qty{300}{V}$ माना गया है; और ``तैयार'' बत्ती पूरे
\omterm{def:g11:circuits-and-power:voltage}{विभवांतर} के $95\%$ पर चल पड़ती है। चलने पर \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} को
केवल फ़्लैश-नली दिखती है, $R_f = \qty{0.50}{\ohm}$। आँकड़े: कोई AA सेल \qty{1.5}{V} और \qty{2.4}{A.h} देता
है; $e = \qty{1.60e-19}{C}$।

\medskip
\noindent\textbf{भाग I --- भंडार।}
\begin{enumerate}
  \item \qty{300}{V} पर \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} का आवेश निकालिए।
  \item वह कितने \omterm{def:g11:fundamental-interactions:elementary}{मूल आवेश} हैं?
  \item संचित ऊर्जा $E_C$ निकालिए।
  \item AA सेल $q = It \approx \qty{8.6e3}{C}$ रखता है, यानी लगभग \qty{1.3e4}{J}। यह \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} की
        ऊर्जा का कितना गुना है?
  \item तो फिर \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} की झंझट क्यों? ``शक्ति'' शब्द का
        उपयोग करते हुए एक वाक्य में।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हरी बत्ती की प्रतीक्षा।}
\begin{enumerate}[resume]
  \item फंदा-नियम लिखिए और उसे $u(t)$ के अवकल समीकरण में बदलिए।
  \item प्रतिस्थापन से जाँचिए कि $u(t) = \mathcal{E}(1 - e^{-t/RC})$ उसे हल कर देता है, और वह $u(0) = 0$ से
        शुरू होता है।
  \item \omterm{def:g12:rc-rl-circuits:tau}{कालांक} $\tau$ निकालिए।
  \item तैयार-बत्ती कितने समय बाद और किस \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} पर हरी होती है? दो
        \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} की उस सीटी को पहचानिए।
  \item आरंभिक आवेशन-धारा निकालिए; फिर वह मरती क्यों जाती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- एक मिलीसेकंड का जलवा।}
\begin{enumerate}[resume]
  \item नली से होकर विसर्जन का नियम $u(t)$ लिखिए और नया \omterm{def:g12:rc-rl-circuits:tau}{कालांक}
        $\tau'$ निकालिए।
  \item नली से गुज़रती शिखर धारा निकालिए।
  \item शिखर शक्ति निकालिए; \qty{2.2}{kW} की केतली से तुलना कीजिए।
  \item फ़्लैश को $3\tau'$ पर ख़त्म मानकर उसकी अवधि और औसत शक्ति दीजिए।
  \item आवेशन-काल और फ़्लैश-काल का अनुपात बनाइए: वही \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र}, वही
        आवेश --- तो बदला क्या, और इससे \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} क्या ठहरता है:
        ऊर्जा की टंकी, या शक्ति का लीवर?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दीवार पर टँगा चचेरा भाई।}
किसी डीफ़िब्रिलेटर को $U_0 = \qty{1500}{V}$ पर $E_C = \qty{200}{J}$ देना है; और छाती तथा पट्टियाँ मिलकर
लगभग \qty{50}{\ohm} बनाती हैं।
\begin{enumerate}[resume]
  \item ज़रूरी \omterm{def:g12:rc-rl-circuits:capacitor}{धारिता} निकालिए।
  \item संचित आवेश निकालिए।
  \item झटके का \omterm{def:g12:rc-rl-circuits:tau}{कालांक} और अवधि ($3\tau$) निकालिए; फ़्लैश से तुलना
        कीजिए।
  \item रोगी में शिखर धारा और शिखर शक्ति निकालिए।
  \item फिर से आवेशित करने वाले परिपथ को यंत्र को \qty{6.0}{s} में तैयार ($95\%$)
        कर देना है: ज़रूरी आवेशन-प्रतिरोध और आरंभिक आवेशन-धारा
        निकालिए --- यानी वे संख्याएँ जो कोई डिज़ाइनर मँगवाएगा।
\end{enumerate}
\end{problem}
