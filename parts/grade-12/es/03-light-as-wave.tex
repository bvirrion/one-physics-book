\chapter{La luz como onda: difracción e interferencias}\label{ch:g12:light-as-wave}

Inclina una lámina de jabón hacia una ventana: los colores se deslizan
por ella, aunque nada del jabón ni del agua sea de color. Apunta con un
\omterm{def:g11:color-light-sources:lamps}{láser} a un solo cabello: la pared de detrás no muestra una sombra fina
como un pelo, sino una raya brillante, cruzada por líneas oscuras y
miles de veces más ancha que el cabello. Los rayos que marchan rectos
(\cref{ch:g10:refraction}) no explican ninguna de las dos escenas. Este
capítulo asciende la luz a \omterm{def:g10:signals-and-waves:wave}{onda} de pleno derecho --- medio micrómetro de
cresta a cresta --- y después cobra la idea: un \omterm{def:g11:color-light-sources:lamps}{láser} y una pared
desnuda medirán un cabello, una \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} y los surcos de un
disco.

\section{La luz es una onda}

\begin{definition}[Ondas luminosas]\label{def:g12:light-as-wave:lightwave}
La \omterm{def:g10:light-spectra:white}{luz monocromática} --- un color puro, que ningún prisma puede
descomponer más (\cref{ch:g10:light-spectra}) --- es una \emph{onda
luminosa}\index{onda luminosa} \omterm{def:g10:signals-and-waves:period}{periódica}: cruza el vacío a
$c = \qty{3.00e8}{m/s}$ y queda caracterizada por su \omterm{def:g10:signals-and-waves:frequency}{frecuencia} $f$ o
por su \emph{longitud de onda en el vacío}\index{longitud de onda!de la
luz} $\lambda = c/f$, la distancia de cresta a cresta del
\cref{ch:g12:mechanical-waves}. El ojo responde desde unos
\qty{400}{nm} (violeta) hasta \qty{800}{nm} (rojo); la \omterm{def:g10:light-spectra:white}{luz blanca}
mezcla todo el intervalo.
\end{definition}

\begin{example}[Órdenes de magnitud]\label{ex:g12:light-as-wave:orders}
Un \omterm{def:g11:color-light-sources:lamps}{láser} de helio-neón emite luz roja de
$\lambda = \qty{633}{nm} = \qty{6.33e-7}{m}$, así que
$f = c/\lambda \approx \qty{4.7e14}{Hz}$: ninguna electrónica cuenta tan
deprisa --- todas las \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} de este capítulo se medirán con
geometría. Para hacerse una idea, un cabello (de unos
\qty{70}{\micro m}) mide cien \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda}. ¿Qué es lo que ondula?
El aire no: un \omterm{def:g11:electric-gravitational-fields:efield}{campo eléctrico} y otro magnético, descritos
honradamente en el volumen del segundo año; aquí abajo lo único que
importa es la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}.
\end{example}

\section{La difracción}

\begin{definition}[Difracción]\label{def:g12:light-as-wave:diffraction}
Cuando una \omterm{def:g10:signals-and-waves:wave}{onda} encuentra una abertura o un obstáculo de tamaño $a$
comparable a su \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}, se extiende por la región que los
rayos dejarían a oscuras. Esa extensión es la
\emph{difracción}\index{difracción} --- despreciable mientras
$\lambda/a$ sea diminuto y dominante a medida que $a$ se encoge hacia
$\lambda$.
\end{definition}

\begin{proposition}[Semianchura angular de la extensión]\label{prop:g12:light-as-wave:halfwidth}
Una rendija de anchura $a$, iluminada con una \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}
$\lambda < a$, extiende la luz en torno a la dirección incidente dentro
de la semianchura angular $\theta$ (en radianes); una pantalla situada
a la distancia $D$ muestra una banda central brillante de anchura $L$,
el doble de $D \tan\theta \approx D\,\theta$ (para ángulos pequeños):
\[
\theta \approx \frac{\lambda}{a}, \qquad L = \frac{2\lambda D}{a}.
\]
\end{proposition}
\admitted

\begin{remark}[Leer la fórmula al revés]\label{rem:g12:light-as-wave:honest}
Cuanto \emph{más estrecha} es la rendija, \emph{más ancha} es la
extensión --- lo contrario de lo que predicen los rayos. La deducción
honrada (sumar las \omterm{def:g10:signals-and-waves:wave}{ondas} elementales reemitidas a lo ancho de la
rendija) vive en el volumen del segundo año; aquí la fórmula es una
herramienta de medida.
\end{remark}

\begin{example}[Un láser a través de una rendija]\label{ex:g12:light-as-wave:slit}
\omterm{def:g11:color-light-sources:lamps}{Láser} rojo, $\lambda = \qty{633}{nm}$; rendija $a = \qty{0.10}{mm}$;
pantalla a $D = \qty{2.00}{m}$. Entonces
$\theta = \num{6.33e-7} / \num{1.0e-4} = \qty{6.3e-3}{rad}$ --- un
tercio de grado, pero la banda central de la pared mide
$L = 2 \times \num{6.33e-7} \times 2.00 / \num{1.0e-4} \approx
\qty{2.5}{cm}$ de ancho, flanqueada por líneas oscuras y bandas
laterales más tenues. Reduce la rendija a la mitad y la banda se
duplica.
\end{example}

\begin{proposition}[Un obstáculo difracta como una rendija]\label{prop:g12:light-as-wave:hair}
Un obstáculo opaco y fino de anchura $d$ --- un hilo, un cabello ---
produce, fuera del haz incidente, el mismo patrón que una rendija de
anchura $d$ (principio de Babinet, deducido en el volumen del segundo
año).
\end{proposition}
\admitted

\begin{method}[Medir una anchura diminuta]\label{met:g12:light-as-wave:hairwidth}
\begin{enumerate}
  \item Tensar el cabello atravesando el haz del \omterm{def:g11:color-light-sources:lamps}{láser}; colocar una
        pantalla a una distancia medida $D$ (de unos \omterm{def:g10:orders-of-magnitude:unit}{metros}).
  \item Medir la anchura $L$ de la banda central, de línea oscura a
        línea oscura.
  \item Como $L = 2\lambda D/d$: $d = 2\lambda D/L$ --- cuanto más fino
        el cabello, mayor el patrón.
\end{enumerate}
\end{method}

\begin{remark}[La difracción limita todos los instrumentos]\label{rem:g12:light-as-wave:limits}
La luz entra en una cámara, en un telescopio o en un ojo por una
abertura de anchura $a$, de modo que hasta una \omterm{def:g11:lenses-and-eye:lens}{lente} perfecta difumina
un punto en una mancha de tamaño angular en torno a $\lambda/a$: los
detalles más próximos se funden. Un microscopio de luz visible no
resuelve, por tanto, mucho menos de
$\lambda \approx \qty{0.5}{\micro m}$ (bacterias sí, virus no); y los
espejos de los telescopios se fabrican anchos por la misma razón.
\end{remark}

\section{Las interferencias}

\begin{definition}[Fuentes coherentes, diferencia de camino]\label{def:g12:light-as-wave:coherent}
Dos fuentes de \omterm{def:g10:signals-and-waves:wave}{ondas} son \emph{coherentes}\index{fuentes coherentes}
cuando tienen la misma \omterm{def:g10:signals-and-waves:frequency}{frecuencia} y vibran al unísono --- en la
práctica, cuando son dos copias de una misma \omterm{def:g10:signals-and-waves:wave}{onda}, obtenidas
dividiéndola. Para un punto $M$ al que llegan las dos, la
\emph{diferencia de camino}\index{diferencia de camino} es
$\delta = S_2M - S_1M$.
\end{definition}

\begin{proposition}[Condiciones de interferencia]\label{prop:g12:light-as-wave:conditions}
Donde se solapan dos \omterm{def:g10:signals-and-waves:wave}{ondas} \omterm{def:g12:light-as-wave:coherent}{coherentes} de \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} $\lambda$, se
suman --- \emph{interfieren}\index{interferencia}:
\begin{itemize}
  \item $\delta = k\lambda$ ($k$ entero): al unísono, reforzándose ---
        interferencia \emph{constructiva}\index{interferencia
        constructiva} (brillante);
  \item $\delta = (k + \tfrac12)\lambda$: cresta sobre valle,
        cancelándose --- interferencia
        \emph{destructiva}\index{interferencia destructiva} (oscura).
\end{itemize}
\end{proposition}

\begin{proof}
Desplazar una \omterm{def:g12:mechanical-waves:periodic}{onda periódica} un número entero de \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} no
cambia nada: las crestas se apilan sobre las crestas. Media \omterm{def:g12:mechanical-waves:wavelength}{longitud de
onda} más pone cresta sobre valle: dos \omterm{def:g10:signals-and-waves:wave}{ondas} iguales se cancelan.
\end{proof}

\begin{example}[El experimento de las dos rendijas de Young]\label{ex:g12:light-as-wave:young}
Perfora dos rendijas finas $S_1$ y $S_2$, separadas una fracción de
milímetro, en una placa opaca, e ilumina las dos con un mismo \omterm{def:g11:color-light-sources:lamps}{láser}: dos
\omterm{def:g12:light-as-wave:coherent}{fuentes coherentes}. En una pantalla situada un par de \omterm{def:g10:orders-of-magnitude:unit}{metros} más allá,
la zona de solapamiento aparece rayada: en el centro $\delta = 0$,
brillante; subiendo por la pantalla, $\delta$ crece pasando por
$\lambda/2$ (oscuro), $\lambda$ (brillante)\,\dots\ oscuridad allí donde
la luz sumada a la luz se cancela.
\end{example}

\begin{definition}[Franjas e interfranja]\label{def:g12:light-as-wave:fringes}
Las bandas alternadas brillantes y oscuras del patrón son sus
\emph{franjas de interferencia}\index{franjas de interferencia}; la
distancia $i$ entre los centros de dos franjas brillantes consecutivas
es la \emph{interfranja}\index{interfranja}.
\end{definition}

\begin{proposition}[La interfranja]\label{prop:g12:light-as-wave:interfringe}
Para unas rendijas separadas una distancia $a$ y una pantalla a la
distancia $D \gg a$, el punto de la pantalla situado a la distancia $x$
del centro $O$ tiene una \omterm{def:g12:light-as-wave:coherent}{diferencia de camino}
$\delta \approx a\,x/D$ (geometría admitida); por la
\cref{prop:g12:light-as-wave:conditions}, las franjas brillantes están
en $x_k = k\,\lambda D/a$, igualmente espaciadas con \omterm{def:g12:light-as-wave:fringes}{interfranja}
\[
i = \frac{\lambda D}{a}.
\]
Con $\lambda = \qty{633}{nm}$, $a = \qty{0.20}{mm}$ y
$D = \qty{2.0}{m}$: $i = \qty{6.3}{mm}$ --- el montaje amplifica
$\lambda$ en un factor $D/a = 10^4$. La geometría que hay detrás se
desarrolla en el volumen del segundo año.
\end{proposition}
\admitted

\begin{method}[Medir una longitud de onda]\label{met:g12:light-as-wave:lambda}
\begin{enumerate}
  \item Enviar el \omterm{def:g11:color-light-sources:lamps}{láser} desconocido a través de una doble rendija de
        separación $a$ conocida; colocar la pantalla a una distancia
        medida $D$.
  \item Medir lo que abarcan $10$ \omterm{def:g12:light-as-wave:fringes}{interfranjas} y dividir entre $10$: sale
        la \omterm{def:g12:light-as-wave:fringes}{interfranja} $i$, con el error de la regla reducido diez
        veces.
  \item Entonces $\lambda = i\,a/D$.
\end{enumerate}
\end{method}

\begin{remark}[Por qué dos lámparas no interfieren nunca]\label{rem:g12:light-as-wave:lamps}
Dos flexos que iluminan una misma pared no producen franjas: cada uno
emite trenes de \omterm{def:g10:signals-and-waves:wave}{onda} cortos y sin correlación, cuyo desfase relativo
salta al azar mil millones de veces por segundo, así que las franjas se
desplazan igual de rápido y el ojo las promedia en una luz uniforme. La
interferencia exige coherencia --- una sola \omterm{def:g10:signals-and-waves:wave}{onda} dividida en dos, como
hacen las rendijas de Young; el \omterm{def:g11:color-light-sources:lamps}{láser} es el atajo moderno, una \omterm{def:g10:signals-and-waves:wave}{onda}
limpia y larga.
\end{remark}

\section{Colores pintados por interferencia}

\begin{definition}[Irisación]\label{def:g12:light-as-wave:iridescence}
Los colores que cambian con el ángulo de observación --- las pompas de
jabón, las manchas de aceite, las plumas de un pavo real, el reverso de
un disco --- son \emph{irisación}\index{irisación}: no hay pigmento,
sino interferencia en \omterm{def:g10:light-spectra:white}{luz blanca}, que refuerza unas \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} y
cancela otras.
\end{definition}

\begin{example}[Láminas de jabón y manchas de aceite]\label{ex:g12:light-as-wave:soap}
La luz que llega a una lámina de jabón se refleja en parte en su cara
anterior y en parte en la posterior: dos copias \omterm{def:g12:light-as-wave:coherent}{coherentes} cuya
\omterm{def:g12:light-as-wave:coherent}{diferencia de camino} la fijan el grosor y el ángulo de observación. Cada
grosor devuelve unas \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} reforzadas y otras canceladas:
la lámina refleja un color que se desliza a medida que escurre y se
adelgaza. Una mancha de aceite sobre asfalto mojado hace el mismo truco.
\end{example}

\begin{remark}[Luz monocromática frente a luz blanca]\label{rem:g12:light-as-wave:white}
En \omterm{def:g10:light-spectra:white}{luz blanca}, cada \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} dibuja sus propias franjas de Young
con su propia $i = \lambda D/a$: en el centro todos los colores están
brillantes (una franja blanca); unas pocas franjas más allá los patrones
se separan --- bordes irisados y después difuminado. La \omterm{def:g10:light-spectra:white}{luz
monocromática} da franjas nítidas sin fin: de ahí el \omterm{def:g11:color-light-sources:lamps}{láser} en todas las
medidas precisas de aquí.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:1}
Calcula la \omterm{def:g10:signals-and-waves:frequency}{frecuencia} de la luz roja ($\lambda = \qty{700}{nm}$) y la de
la violeta (\qty{400}{nm}). ¿Cuál es mayor?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:2}
Un \omterm{def:g11:color-light-sources:lamps}{láser} verde ($\lambda = \qty{532}{nm}$) atraviesa una rendija de
anchura $a = \qty{0.20}{mm}$: calcula $\theta$ y después la anchura de
la banda central en una pantalla situada a \qty{1.5}{m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:3}
Una puerta tiene \qty{0.80}{m} de ancho: calcula $\lambda/a$ para una
voz de \qty{425}{Hz} ($v = \qty{340}{m/s}$) y para luz de
\qty{550}{nm}. ¿Por qué oyes, pero no ves, al doblar la esquina?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:4}
Dos altavoces alimentados por un mismo generador emiten al unísono un
\omterm{def:g10:signals-and-waves:sound}{sonido} de \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} \qty{2.0}{cm}. \omterm{def:g12:light-as-wave:coherent}{Diferencia de camino} en $M$:
\qty{6.0}{cm}; en $N$: \qty{5.0}{cm}. ¿Suena fuerte o hay silencio en
cada uno? Justifícalo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:light-as-wave:5}
Rendijas de Young: $\lambda = \qty{633}{nm}$, $a = \qty{0.25}{mm}$,
$D = \qty{1.8}{m}$. Calcula la \omterm{def:g12:light-as-wave:fringes}{interfranja}. ¿Cuántas franjas brillantes
caben en los \qty{2.0}{cm} centrales de la pantalla?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:6}
Un cabello en un haz de $\lambda = \qty{650}{nm}$, con la pantalla a
\qty{1.6}{m}, da una banda central de \qty{2.8}{cm} de ancho. Calcula el
diámetro del cabello.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:7}
Un \omterm{def:g11:color-light-sources:lamps}{láser} desconocido a través de unas rendijas de Young
($a = \qty{0.20}{mm}$, $D = \qty{1.50}{m}$): diez \omterm{def:g12:light-as-wave:fringes}{interfranjas} abarcan
\qty{4.7}{cm}. Halla $\lambda$ y el color. ¿Por qué medir diez y no
una?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:8}
Dos flexos idénticos iluminan la misma pared; los haces se solapan y,
sin embargo, no aparecen franjas nunca. ¿Qué condición de coherencia
(\cref{def:g12:light-as-wave:coherent}) falla, y cómo la repara el
montaje de Young?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:9}
En un patrón de dos rendijas ($a = \qty{0.30}{mm}$, $D = \qty{1.4}{m}$),
los centros de la primera y de la séptima franjas brillantes distan
\qty{18.0}{mm}: halla la \omterm{def:g12:light-as-wave:fringes}{interfranja} y después $\lambda$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:10}
Las pistas de un disco compacto forman una red de paso
$d = \qty{1.6}{\micro m}$. Admitiendo $d \sin\theta = k\lambda$ ($k$
entero), halla las direcciones de los haces para
$\lambda = \qty{633}{nm}$. ¿Cuál es el orden $k$ más alto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:light-as-wave:11}
Un telescopio de abertura $a = \qty{10}{cm}$ a
$\lambda = \qty{550}{nm}$: calcula su difuminado por \omterm{def:g12:light-as-wave:diffraction}{difracción}
$\theta \approx \lambda/a$. ¿Puede separar dos faros situados a
\qty{1.5}{m} el uno del otro, a \qty{100}{km}?
\end{exercise}

\section{Problema: medir con luz}

\begin{problem}[{Problema de fin de semana --- medir con luz: un láser,
una regla y una pared desnuda se convierten en un taller de micrómetros
que calibra una rendija, un cabello humano, una longitud de onda
desconocida y los surcos de dos discos plateados}]
\label{pb:g12:light-as-wave:1}
Tu equipo: un \omterm{def:g11:color-light-sources:lamps}{láser} rojo ($\lambda = \qty{633}{nm}$, según la etiqueta),
un \omterm{def:g11:color-light-sources:lamps}{láser} verde de \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} desconocida, una rendija calibrada
$a = \qty{100}{\micro m}$, una doble rendija de separación
$a' = \qty{0.250}{mm}$, una cinta métrica, dos cabellos, un CD y un
DVD.

\medskip
\noindent\textbf{Parte I --- Calibrar con la rendija conocida.}
\omterm{def:g11:color-light-sources:lamps}{Láser} rojo, rendija y pared a $D = \qty{3.00}{m}$.
\begin{enumerate}
  \item Calcula la semianchura angular $\theta$ del haz difractado.
  \item Demuestra que la banda central tiene una anchura
        $L = 2\lambda D/a$; calcúlala.
  \item Mides $L = \qty{3.7}{cm}$: calcula la desviación relativa. ¿Se
        valida el método?
  \item Una segunda rendija es la mitad de ancha, \qty{50}{\micro m}:
        predice su $L$. Enuncia la regla que liga la anchura de la
        rendija con la del patrón.
  \item Justifica el paso $\tan\theta \approx \theta$ de la pregunta 2:
        compara $\theta$ y $\tan\theta$ para este ángulo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El cabello.} El mismo \omterm{def:g11:color-light-sources:lamps}{láser} y la misma
pared; un cabello tensado sustituye a la rendija.
\begin{enumerate}[resume]
  \item ¿Qué resultado admitido te permite conservar la fórmula de la
        rendija para un cabello, y qué dice?
  \item La banda central mide $L = \qty{5.4}{cm}$: calcula el diámetro
        $d$ del cabello.
  \item Tu regla mide $L$ con $\pm\qty{0.2}{cm}$: da el intervalo
        resultante para $d$, como $d \pm \Delta d$.
  \item El cabello de tu amiga da $L = \qty{7.6}{cm}$: ¿su diámetro?
        ¿Qué cabello es más fino, y por qué más fino significa más
        ancho?
  \item ¿Podría un flexo sustituir al \omterm{def:g11:color-light-sources:lamps}{láser}? Da dos razones.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} del \omterm{def:g11:color-light-sources:lamps}{láser} verde.}
\omterm{def:g11:color-light-sources:lamps}{Láser} verde, doble rendija y pared a $D = \qty{2.00}{m}$.
\begin{enumerate}[resume]
  \item ¿Por qué tienen que iluminar las dos rendijas el \emph{mismo}
        \omterm{def:g11:color-light-sources:lamps}{láser}?
  \item ¿Por qué la franja central es brillante sea cual sea la \omterm{def:g12:mechanical-waves:wavelength}{longitud
        de onda}?
  \item Diez \omterm{def:g12:light-as-wave:fringes}{interfranjas} abarcan \qty{4.26}{cm}. Da $i$ y la razón de
        medir diez de golpe.
  \item Deduce la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} del \omterm{def:g11:color-light-sources:lamps}{láser} verde.
  \item La etiqueta del fabricante dice $\qty{532 \pm 10}{nm}$: ¿es
        coherente? ¿Qué color esperas?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Leer discos plateados.} El \omterm{def:g11:color-light-sources:lamps}{láser} rojo
incide perpendicularmente sobre un CD; sus pistas actúan como una red de
reflexión que cumple (se admite) $d \sin\theta = k\lambda$. Sobre la
pared, situada a $D = \qty{1.00}{m}$, cada haz de primer orden deja una
mancha a una distancia $x$ de la central.
\begin{enumerate}[resume]
  \item Para el CD, $x = \qty{43}{cm}$: calcula $\theta$. ¿Por qué está
        ahora prohibido el atajo de ángulos pequeños de la Parte I?
  \item Deduce el paso $d$ entre pistas del CD.
  \item El DVD manda su mancha de primer orden a $x = \qty{1.65}{m}$:
        calcula su $\theta$ y su paso entre pistas.
  \item Compara los dos pasos. ¿Por qué almacena un DVD más que un CD?
  \item Cierra el cuaderno: enumera las longitudes medidas este fin de
        semana y di en una frase qué convirtió a la luz en una regla
        para todas ellas.
\end{enumerate}
\end{problem}
