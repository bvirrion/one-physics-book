\chapter{Oscilaciones eléctricas libres: RLC}\label{ch:g12:rlc-oscillations}

Gira el pesado dial de una radio de desván y las emisoras desfilan ante la aguja: una
voz, unos violines, ruido, otra voz. Detrás del panel no hay ningún motor ni ningún
ordenador --- una \omterm{def:g12:rc-rl-circuits:inductor}{bobina}, un \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} cuyas placas intercaladas hace girar el dial
y un buen trozo de hilo de antena. Este capítulo conecta cara a cara los dos \omterm{def:g10:energy-conservation:transfer}{depósitos}
de \omterm{def:g10:energy-conservation:energy}{energía} del capítulo anterior (\cref{ch:g12:rc-rl-circuits}) y encuentra el péndulo
más rápido jamás construido: carga que va y viene, hasta millones de veces por
segundo, al tempo que elige el dial.

\section{Un péndulo eléctrico: el circuito LC}

Carga un \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} $C$ a una tensión $U_0$ --- sus placas albergan entonces
$\pm Q_0$ con $Q_0 = C U_0$ --- y en $t = 0$ conéctalo a una \omterm{def:g12:rc-rl-circuits:inductor}{bobina} \emph{ideal} $L$,
de \omterm{def:g11:circuits-and-power:resistance}{resistencia} nula. No hay ningún generador en la malla: pase lo que pase a partir de
ahora, el circuito lo hace por su cuenta.

\end{circuitikz}

{\small El \omterm{prop:g12:rlc-oscillations:equation}{circuito LC}: un \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} cargado a $Q_0$, un interruptor y una \omterm{def:g12:rc-rl-circuits:inductor}{bobina}
ideal. Al cerrar $K$, la carga se derrama a través de la \omterm{def:g12:rc-rl-circuits:inductor}{bobina} --- y la \omterm{def:g12:rc-rl-circuits:inductor}{bobina}, que
detesta los cambios, no la dejará detenerse.}
\end{omfigure}

\begin{proposition}[La ecuación del circuito LC]\label{prop:g12:rlc-oscillations:equation}
\index{circuito LC}
En la malla LC ideal, la carga $q(t)$ de la placa superior obedece
\[
L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0,
\qquad\text{es decir,}\qquad
\frac{d^2q}{dt^2} = -\frac{1}{LC}\,q .
\]
\end{proposition}

\begin{proof}
A lo largo de la malla las tensiones suman cero
(\cref{prop:g11:circuits-and-power:laws}): $u_L + u_C = 0$. El \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} da
$u_C = q/C$ y la \omterm{def:g12:rc-rl-circuits:inductor}{bobina}, $u_L = L\,di/dt$
(\cref{def:g12:rc-rl-circuits:inductor}), con $i = dq/dt$ contando la carga que llega
a la placa superior (\cref{prop:g12:rc-rl-circuits:cdudt}); sustitúyase.
\end{proof}

\begin{theorem}[Oscilaciones libres del circuito LC]\label{thm:g12:rlc-oscillations:oscillation}
\index{oscilaciones eléctricas}
Soltado en $t = 0$ con $q(0) = Q_0$ e $i(0) = 0$, el circuito oscila:
\[
q(t) = Q_0 \cos\!\left(\frac{2\pi t}{T_0}\right),
\qquad
i(t) = -I_0 \sin\!\left(\frac{2\pi t}{T_0}\right),
\qquad
T_0 = 2\pi\sqrt{LC},
\]
con una corriente máxima $I_0 = 2\pi Q_0/T_0 = Q_0/\sqrt{LC}$.
\end{theorem}

\begin{proof}
Verifiquémoslo por sustitución. Derivando,
$i = dq/dt = -\frac{2\pi}{T_0} Q_0 \sin(2\pi t/T_0)$ --- la $i(t)$ del enunciado ---
y, otra vez, $\frac{d^2q}{dt^2} = -(\frac{2\pi}{T_0})^2 q$, que es igual a $-q/(LC)$
exactamente cuando $(2\pi/T_0)^2 = 1/(LC)$, es decir, $T_0 = 2\pi\sqrt{LC}$. Y
$q(0) = Q_0$, $i(0) = 0$. Que ninguna \emph{otra} función encaja con la ecuación y con
el arranque se admite aquí, igual que en el caso del resorte
(\cref{rem:g12:oscillators-and-time:uniqueness}).
\end{proof}

\begin{definition}[Periodo y frecuencia propios]\label{def:g12:rlc-oscillations:period}
El \emph{periodo propio}\index{periodo propio!de un circuito LC} de un \omterm{prop:g12:rlc-oscillations:equation}{circuito LC} es
$T_0 = 2\pi\sqrt{LC}$; su \emph{frecuencia propia}\index{frecuencia propia!de un
circuito LC} es
\[
f_0 = \frac{1}{T_0} = \frac{1}{2\pi\sqrt{LC}} .
\]
Comprobación dimensional: $\qty{1}{H} = \qty{1}{V.s/A}$ y
$\qty{1}{F} = \qty{1}{C/V}$, así que $LC$ lleva
$\unit{s} \times \unit{C/A} = \unit{s^2}$ --- $\sqrt{LC}$ es un tiempo, como tiene que
ser.
\end{definition}

\begin{example}[Primeros números]\label{ex:g12:rlc-oscillations:firstnumbers}
$L = \qty{10}{mH}$ con $C = \qty{1.0}{\micro F}$: $\sqrt{LC} = \qty{1.0e-4}{s}$, luego
$T_0 = \qty{0.63}{ms}$ y $f_0 \approx \qty{1.6}{kHz}$ --- un \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} audible.
$L = \qty{1.0}{mH}$ con $C = \qty{100}{pF}$: $T_0 = \qty{2.0}{\micro s}$,
$f_0 \approx \qty{0.50}{MHz}$ --- una \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} de radio. Solo el \emph{producto}
$LC$ fija el tempo.
\end{example}

\begin{remark}[Un cuarto de periodo desfasados]\label{rem:g12:rlc-oscillations:quarter}
La corriente va desfasada un cuarto de \omterm{def:g12:oscillators-and-time:oscillator}{periodo} respecto de la carga: $i = 0$ cuando
$q = \pm Q_0$ (placas llenas, el flujo se invierte) e $i = \pm I_0$ cuando $q = 0$
(placas vacías, flujo a tope) --- exactamente el péndulo, inmóvil en los extremos y
rapidísimo abajo.
\end{remark}

\section{El vals de la energía}

\begin{proposition}[Conservación de la energía almacenada]\label{prop:g12:rlc-oscillations:energy}
\index{energía!de un circuito LC}
En el \omterm{prop:g12:rlc-oscillations:equation}{circuito LC} ideal, la \omterm{def:g10:energy-conservation:energy}{energía} total almacenada
(\cref{prop:g12:rc-rl-circuits:energy})
\[
E \;=\; E_C + E_L \;=\; \frac{q^2}{2C} + \tfrac12\,L i^2
\;=\; \frac{Q_0^2}{2C} \;=\; \tfrac12\,L I_0^2
\]
es constante: enteramente en el \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} cuando $q = \pm Q_0$ y enteramente en la
\omterm{def:g12:rc-rl-circuits:inductor}{bobina} cuando $i = \pm I_0$.
\end{proposition}

\begin{proof}
Derivemos: $\frac{dE}{dt} = \frac{q}{C}\frac{dq}{dt} + Li\frac{di}{dt} =
i\left(\frac{q}{C} + L\frac{d^2q}{dt^2}\right) = 0$ por la
\cref{prop:g12:rlc-oscillations:equation}. Su valor se lee en $t = 0$ (todo en el
\omterm{def:g12:rc-rl-circuits:capacitor}{condensador}) y también en el instante en que todo es corriente, $q = 0$ --- de donde
$\tfrac12 L I_0^2 = Q_0^2/2C$, recuperando $I_0 = Q_0/\sqrt{LC}$ solo con la \omterm{def:g10:energy-conservation:energy}{energía}.
\end{proof}

dos veces por \omterm{def:g12:oscillators-and-time:oscillator}{periodo}, con la suma clavada en $Q_0^2/2C$ --- la mismísima figura del
\omterm{prop:g12:oscillators-and-time:spring-period}{oscilador masa--resorte}, con otras etiquetas.}
\end{omfigure}

\begin{example}[El vals, medido]\label{ex:g12:rlc-oscillations:waltz}
$C = \qty{1.0}{\micro F}$ cargado a $U_0 = \qty{10}{V}$ y conectado a
$L = \qty{10}{mH}$: \omterm{def:g10:energy-conservation:energy}{energía} $E = \tfrac12 C U_0^2 = \qty{5.0e-5}{J}$, \omterm{def:g12:oscillators-and-time:oscillator}{periodo}
$T_0 = \qty{0.63}{ms}$ y una corriente máxima $I_0 = U_0\sqrt{C/L} = \qty{0.10}{A}$
alcanzada un cuarto de \omterm{def:g12:oscillators-and-time:oscillator}{periodo} (\qty{0.16}{ms}) después de cerrarse el interruptor.
\end{example}

\section{Circuitos reales: oscilaciones amortiguadas}

Toda \omterm{def:g12:rc-rl-circuits:inductor}{bobina} real está bobinada con hilo resistivo: la malla honrada es un circuito
\emph{RLC} serie, y la \omterm{def:g11:circuits-and-power:resistance}{resistencia} le pasa impuestos al vals.

\begin{definition}[Regímenes seudoperiódico y aperiódico]\label{def:g12:rlc-oscillations:regimes}
\index{circuito RLC}\index{régimen seudoperiódico}\index{régimen aperiódico}
En un circuito RLC serie, el resistor disipa \omterm{def:g10:energy-conservation:energy}{energía} por \omterm{def:g11:circuits-and-power:joule}{efecto Joule}
(\cref{ch:g11:circuits-and-power}): las \omterm{def:g12:oscillators-and-time:oscillator}{oscilaciones libres} son
\emph{\omterm{def:g12:oscillators-and-time:damping}{amortiguadas}}. Dos regímenes, legibles en un \omterm{def:g10:signals-and-waves:oscilloscope}{osciloscopio}:
\begin{itemize}
  \item \emph{seudoperiódico} ($R$ pequeña): el circuito sigue oscilando dentro de una
        envolvente que se encoge, con un tiempo de repetición --- el
        \emph{seudoperiodo}\index{seudoperiodo!eléctrico} --- próximo a
        $T_0 = 2\pi\sqrt{LC}$ cuando el \omterm{def:g12:oscillators-and-time:damping}{amortiguamiento} es débil;
  \item \emph{aperiódico} ($R$ grande): la carga vuelve reptando a cero sin pasarse
        nunca --- ninguna oscilación.
\end{itemize}
El espejo exacto del \omterm{def:g12:oscillators-and-time:oscillator}{oscilador mecánico} amortiguado
(\cref{def:g12:oscillators-and-time:damping}).
\end{definition}

una envolvente exponencial (a trazos), o una vuelta reptando a cero sin oscilar. Con
\omterm{def:g12:oscillators-and-time:damping}{amortiguamiento} débil, el \omterm{def:g12:rlc-oscillations:regimes}{seudoperiodo} se queda en $\approx T_0$.}
\end{omfigure}

\begin{remark}[Mantener las oscilaciones]\label{rem:g12:rlc-oscillations:maintain}
\index{oscilaciones mantenidas}
Para mantener viva la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}, hay que aportarle al circuito exactamente la \omterm{def:g10:energy-conservation:energy}{energía}
que quema la \omterm{def:g11:circuits-and-power:resistance}{resistencia}, una vez por ciclo y \emph{al paso} de la oscilación: un
amplificador hace por la malla RLC lo que el empujón del escape hace por el péndulo
(\cref{ch:g12:oscillators-and-time}). Todo reloj de cuarzo y todo emisor de radio son
\omterm{def:g12:oscillators-and-time:oscillator}{osciladores} mantenidos de este tipo, funcionando a su \omterm{def:g12:rlc-oscillations:period}{frecuencia propia} $f_0$.
\end{remark}

\section{El gemelo mecánico}

\begin{proposition}[El diccionario eléctrico--mecánico]\label{prop:g12:rlc-oscillations:analogy}
\index{analogía eléctrico--mecánica}
La ecuación LC es la ecuación masa--resorte
(\cref{prop:g12:oscillators-and-time:spring-period}) palabra por palabra:
\begin{center}\small
\begin{tabular}{lcl}
\omterm{prop:g12:rlc-oscillations:equation}{circuito LC} & & móvil sobre un resorte\\[2pt]
carga $q$ & $\longleftrightarrow$ & posición $x$\\
corriente $i = dq/dt$ & $\longleftrightarrow$ & velocidad $v = dx/dt$\\
\omterm{def:g12:rc-rl-circuits:inductor}{autoinducción} $L$ & $\longleftrightarrow$ & masa $m$\\
inversa de la \omterm{def:g12:rc-rl-circuits:capacitor}{capacidad} $1/C$ & $\longleftrightarrow$ & \omterm{def:g12:oscillators-and-time:spring}{constante elástica} $k$\\
\omterm{def:g11:circuits-and-power:resistance}{resistencia} $R$ & $\longleftrightarrow$ & \omterm{def:g10:inertia:inventory}{rozamiento}\\
$L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0$ & $\longleftrightarrow$ &
  $m\,\frac{d^2x}{dt^2} + kx = 0$\\[2pt]
$T_0 = 2\pi\sqrt{LC}$ & $\longleftrightarrow$ &
  $T_0 = 2\pi\sqrt{m/k}$\\[2pt]
$E_C = q^2/2C$ & $\longleftrightarrow$ & $E_p = \tfrac12 kx^2$\\
$E_L = \tfrac12 Li^2$ & $\longleftrightarrow$ & $E_k = \tfrac12 mv^2$
\end{tabular}
\end{center}
La misma ecuación, las mismas soluciones: todo resultado sobre cualquiera de los dos
\omterm{def:g12:oscillators-and-time:oscillator}{osciladores} se traduce al instante al otro idioma.
\end{proposition}

\begin{proof}
Compárese la \cref{prop:g12:rlc-oscillations:equation} con $m\,d^2x/dt^2 = -kx$:
sustituir $q \to x$, $L \to m$, $1/C \to k$ convierte una en la otra, y $LC \to m/k$
convierte $2\pi\sqrt{LC}$ en $2\pi\sqrt{m/k}$.
\end{proof}

\begin{example}[Sintonizar una radio]\label{ex:g12:rlc-oscillations:tuning}
Una antena introduce un susurro de \emph{todas} las emisoras en la malla LC, pero solo
la que emite cerca de $f_0$ empuja al paso y se amplifica --- la \omterm{def:g12:oscillators-and-time:resonance}{resonancia}
(\cref{def:g12:oscillators-and-time:resonance}). El dial hace girar las placas de un
\omterm{def:g12:rc-rl-circuits:capacitor}{condensador} variable: cambiar $C$ desplaza $f_0 = 1/(2\pi\sqrt{LC})$, y la aguja va
entregando al auricular una emisora cada vez. Del lado del emisor, un \omterm{def:g12:oscillators-and-time:oscillator}{oscilador}
mantenido (\cref{rem:g12:rlc-oscillations:maintain}) fija la \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} de emisión; en
un reloj de pulsera, un cristal de cuarzo --- un gemelo mecánico de una rigidez
asombrosa --- desempeña el mismo papel.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:1}
Calcula $T_0$ y $f_0$ para (a) $L = \qty{10}{mH}$, $C = \qty{1.0}{\micro F}$;
(b) $L = \qty{1.0}{mH}$, $C = \qty{100}{pF}$. ¿Por qué factor difieren las
\omterm{def:g12:oscillators-and-time:oscillator}{frecuencias}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:2}
Usando $\qty{1}{H} = \qty{1}{V.s/A}$ y $\qty{1}{F} = \qty{1}{C/V}$, demuestra \omterm{def:g10:orders-of-magnitude:unit}{unidad} a
\omterm{def:g10:orders-of-magnitude:unit}{unidad} que $\sqrt{LC}$ es un tiempo. ¿Por qué el $2\pi$ no cambia nada?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:3}
Un \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} de \qty{470}{nF} se carga a \qty{6.0}{V}. Calcula $Q_0$ y la \omterm{def:g10:energy-conservation:energy}{energía}
almacenada; describe, sin cálculos, qué ocurre a continuación cuando se conecta a una
\omterm{def:g12:rc-rl-circuits:inductor}{bobina} ideal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:4}
Un \omterm{prop:g12:rlc-oscillations:equation}{circuito LC} sigue $q(t) = 2.0\cos(2\pi t/T_0)$ en microculombios, con
$T_0 = \qty{4.0}{ms}$. Lee $Q_0$; calcula $f_0$, $q(T_0/2)$, $i(0)$ y la corriente
máxima $I_0 = 2\pi Q_0/T_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:5}
¿En qué instantes del ciclo está la \omterm{def:g10:energy-conservation:energy}{energía} enteramente en el \omterm{def:g12:rc-rl-circuits:capacitor}{condensador}?
¿Enteramente en la \omterm{def:g12:rc-rl-circuits:inductor}{bobina}? ¿Cuál es el momento mecánico correspondiente para un
péndulo en cada caso?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:6}
Verifica por sustitución que $q(t) = Q_0\cos(2\pi t/T_0)$ satisface
$L\,d^2q/dt^2 + q/C = 0$ exactamente cuando $T_0 = 2\pi\sqrt{LC}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:7}
Una emisora de \omterm{def:g10:signals-and-waves:wave}{onda} larga emite a \qty{198}{kHz}. Con $C = \qty{220}{pF}$, ¿qué
\omterm{def:g12:rc-rl-circuits:inductor}{autoinducción} la sintoniza?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:8}
$C = \qty{1.0}{\micro F}$ cargado a \qty{10}{V} se descarga sobre $L = \qty{10}{mH}$.
Calcula la \omterm{def:g10:energy-conservation:energy}{energía} almacenada, la corriente máxima (por \omterm{thm:g10:energy-conservation:conservation}{conservación de la energía}) y
el instante en que se alcanza por primera vez.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:9}
Explica por qué $E_C$ y $E_L$ oscilan a la \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} $2f_0$ y no a $f_0$. Demuestra
que en $t = T_0/8$ (partiendo de $q = Q_0$) la \omterm{def:g10:energy-conservation:energy}{energía} se reparte exactamente por la
mitad.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:10}
En el \omterm{def:g10:signals-and-waves:oscilloscope}{oscilograma} de un \omterm{def:g12:rlc-oscillations:regimes}{circuito RLC} débilmente amortiguado, dos máximos sucesivos
marcan \qty{4.0}{V} y después \qty{3.0}{V}. Suponiendo que el cociente se mantiene
constante, predice el sexto máximo después del de \qty{4.0}{V}. ¿Qué fracción de la
\emph{\omterm{def:g10:energy-conservation:energy}{energía}} sobrevive a cada \omterm{def:g12:rlc-oscillations:regimes}{seudoperiodo}? ¿Qué puedes decir del propio
\omterm{def:g12:rlc-oscillations:regimes}{seudoperiodo}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:11}
Un \omterm{prop:g12:rlc-oscillations:equation}{circuito LC} tiene $L = \qty{0.50}{H}$ y $C = \qty{20}{\micro F}$. Calcula $T_0$;
después, usando el diccionario (\cref{prop:g12:rlc-oscillations:analogy}), halla la
\omterm{def:g12:oscillators-and-time:spring}{constante elástica} $k$ que le da a un móvil de \qty{0.50}{kg} el mismo \omterm{def:g12:oscillators-and-time:oscillator}{periodo}, y
compruébalo.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:12}
Un sintonizador usa $L = \qty{0.20}{mH}$ y un \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} variable que barre
\qtyrange{50}{500}{pF}. Calcula las dos \omterm{def:g12:oscillators-and-time:oscillator}{frecuencias} extremas. Demuestra que
$f_0 \propto 1/\sqrt{C}$, de modo que multiplicar $C$ por diez solo divide la
\omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} entre $\sqrt{10} \approx 3.2$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:13}
Una \omterm{def:g12:rc-rl-circuits:inductor}{bobina} real hace que el circuito pierda un $5.0\,\%$ de su \omterm{def:g10:energy-conservation:energy}{energía} en cada
\omterm{def:g12:rlc-oscillations:regimes}{seudoperiodo}. ¿Al cabo de cuántos \omterm{def:g12:oscillators-and-time:oscillator}{periodos} se ha reducido la \omterm{def:g10:energy-conservation:energy}{energía} a la mitad? ¿Cuál
es entonces la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}, como fracción de la inicial? A $f_0 = \qty{1.0}{MHz}$,
¿cuánto es eso en microsegundos?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:14}
El \omterm{def:g12:oscillators-and-time:oscillator}{oscilador} de un reloj almacena $E = \qty{1.0e-9}{J}$ y pierde un $2.0\,\%$ por
ciclo a $f_0 = \qty{32768}{Hz}$. ¿Qué \omterm{def:g11:work-of-force:power}{potencia} media debe suministrar el circuito de
mantenimiento? ¿Por qué tiene que llegar el empujón al paso de la oscilación?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:15}
Tras cada flanco de una \omterm{def:g10:signals-and-waves:signal}{señal} cuadrada, el cableado de un experimentador ``repica'' a
\qty{25}{MHz}; la \omterm{def:g12:rc-rl-circuits:inductor}{autoinducción} parásita ronda \qty{1.0}{\micro H}. Estima la
\omterm{def:g12:rc-rl-circuits:capacitor}{capacidad} parásita responsable. ¿A qué régimen debería llevar el circuito un pequeño
resistor añadido en serie, y por qué cura eso el repique?
\end{exercise}

\section{Problema: Construir una radio de galena}

\begin{problem}[{Problema de fin de semana --- construir una radio de galena: una
bobina, un condensador variable, un diodo y un auricular --- sin pila, sin
amplificador, y el informativo de la noche llega con la energía de la propia onda}]
\label{pb:g12:rlc-oscillations:1}
Un kit del desván de los abuelos: una \omterm{def:g12:rc-rl-circuits:inductor}{bobina} $L = \qty{0.20}{mH}$ arrollada sobre un
tubo de cartón, un \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} variable cuyas placas giran de \qty{20}{pF} a
\qty{480}{pF}, un diodo de cristal, un auricular y \qty{20}{m} de hilo de antena. Las
emisoras que hay que captar emiten entre \qty{530}{kHz} y \qty{1700}{kHz}.

\medskip
\noindent\textbf{Parte I --- La malla de sintonía.}
\begin{enumerate}
  \item El \omterm{def:g12:rc-rl-circuits:capacitor}{condensador}, cargado, se deja descargar sobre la \omterm{def:g12:rc-rl-circuits:inductor}{bobina}: escribe la ley de
        mallas y conviértela en la ecuación diferencial de $q(t)$.
  \item Verifica por sustitución que $q(t) = Q_0\cos(2\pi t/T_0)$ la resuelve, siempre
        que $T_0 = 2\pi\sqrt{LC}$.
  \item Comprueba por análisis dimensional que $\sqrt{LC}$ es un tiempo.
  \item Con el dial en $C = \qty{330}{pF}$: calcula $T_0$ y $f_0$. ¿Queda la malla
        sintonizada dentro de la banda?
  \item Halla la \omterm{def:g12:rc-rl-circuits:capacitor}{capacidad} que sintoniza exactamente \qty{1.00}{MHz}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} variable.}
\begin{enumerate}[resume]
  \item Calcula la \omterm{def:g12:rc-rl-circuits:capacitor}{capacidad} necesaria para \qty{530}{kHz}.
  \item Calcula la \omterm{def:g12:rc-rl-circuits:capacitor}{capacidad} necesaria para \qty{1700}{kHz}.
  \item Calcula las dos \omterm{def:g12:oscillators-and-time:oscillator}{frecuencias} que alcanza realmente el \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} de
        \qtyrange{20}{480}{pF} del kit. ¿Cubre toda la banda?
  \item Demuestra que $f_0 \propto 1/\sqrt{C}$ con $L$ fija; ¿por qué factor debe
        cambiar $C$ para duplicar $f_0$?
  \item Las placas se abren linealmente, así que $C$ baja de forma uniforme al girar
        el dial. Explica por qué entonces las emisoras se apiñan en un extremo del
        dial --- y en cuál.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{def:g10:energy-conservation:energy}{Energía} y \omterm{def:g12:oscillators-and-time:damping}{amortiguamiento}.}
Sintonizada a \qty{1.00}{MHz} ($C = \qty{127}{pF}$), la antena carga momentáneamente
el \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} a $U_0 = \qty{20}{mV}$.
\begin{enumerate}[resume]
  \item Calcula la carga $Q_0$ y la \omterm{def:g10:energy-conservation:energy}{energía} almacenada $E$.
  \item Calcula la corriente máxima $I_0$ usando la \omterm{thm:g10:energy-conservation:conservation}{conservación de la energía}.
  \item La \omterm{def:g11:circuits-and-power:resistance}{resistencia} de la \omterm{def:g12:rc-rl-circuits:inductor}{bobina} disipa un $4.0\,\%$ de la \omterm{def:g10:energy-conservation:energy}{energía} almacenada por
        ciclo: ¿al cabo de cuántos ciclos se ha reducido la \omterm{def:g10:energy-conservation:energy}{energía} a la mitad?
  \item ¿Cuánto es eso en microsegundos? ¿Por qué este ``repique'' por sí solo no
        puede dar el informativo de la noche?
  \item Explica por qué la \omterm{def:g10:signals-and-waves:wave}{onda} que llega mantiene la malla oscilando solo cuando está
        sintonizada en esa emisora --- y qué mantiene la oscilación en el
        \emph{emisor} (\cref{rem:g12:rlc-oscillations:maintain}).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El gemelo mecánico.}
\begin{enumerate}[resume]
  \item Usando el diccionario (\cref{prop:g12:rlc-oscillations:analogy}) con un móvil
        de masa $m = \qty{0.20}{kg}$, calcula la \omterm{def:g12:oscillators-and-time:spring}{constante elástica} $k$ del resorte
        que iguala el circuito sintonizado.
  \item Un muelle rígido de coche tiene $k \approx \qty{5e4}{N/m}$. Comenta: ¿puede un
        sistema masa--resorte de sobremesa oscilar a megahercios?
  \item Quédate en cambio con un $k = \qty{100}{N/m}$ razonable: ¿qué masa oscila a
        \qty{1.00}{MHz}?
  \item ¿Qué objeto real es un \omterm{def:g12:oscillators-and-time:oscillator}{oscilador mecánico} así, ligerísimo y rígido hasta el
        extremo, y dónde te lo encontraste antes este curso
        (\cref{ch:g12:oscillators-and-time})?
  \item Hoja de características, en una frase: la banda que barre tu radio, la
        \omterm{def:g12:rc-rl-circuits:capacitor}{capacidad} que elige \qty{1.00}{MHz}, cuánto dura un repique solitario y qué es
        lo que, aun así, da el informativo.
\end{enumerate}
\end{problem}
