\chapter{Osciladores mecánicos y la medida del tiempo}\label{ch:g12:oscillators-and-time}

Un reloj de pie hace tic: cada tic es un péndulo de latón cruzando la vertical,
repetido quince millones de veces desde Año Nuevo. Una lámina de cuarzo en tu reloj de
pulsera se flexiona \num{32768} veces por segundo; un \omterm{def:g11:fundamental-interactions:ladder}{átomo} de cesio zumba nueve mil
millones de veces más deprisa. Todo lo que oscila puede medir el tiempo: este capítulo
sigue esas oscilaciones, de Huygens a los \omterm{def:g10:universal-gravitation:satellite}{satélites}.

\section{Oscilaciones libres}

\begin{definition}[Oscilador, periodo, frecuencia, amplitud]\label{def:g12:oscillators-and-time:oscillator}
Un \emph{oscilador mecánico}\index{oscilador} es un sistema que, apartado de una
\emph{posición de equilibrio}\index{posición de equilibrio} estable y soltado, va y
viene en torno a ella: son \emph{oscilaciones libres}\index{oscilaciones libres}. El
movimiento se repite al cabo del \emph{periodo}\index{periodo} $T_0$ (\unit{s}); la
\emph{frecuencia}\index{frecuencia} es $f_0 = 1/T_0$ (\unit{Hz},
\cref{ch:g12:mechanical-waves}); la \emph{amplitud}\index{amplitud} es el
desplazamiento máximo.
\end{definition}

\begin{example}[Un zoo de osciladores]\label{ex:g12:oscillators-and-time:zoo}
Un columpio de parque: $T_0 \approx \qty{3}{s}$. La cuerda la de una guitarra:
$f_0 = \qty{110}{Hz}$. Un cristal de cuarzo: \qty{32768}{Hz}. La oscilación del cesio
que define el segundo: unos \qty{9.2e9}{Hz}. Diez órdenes de magnitud y un solo
oficio: contar.
\end{example}

\section{El péndulo simple}

\begin{definition}[Péndulo simple]\label{def:g12:oscillators-and-time:pendulum}
Un \emph{péndulo simple}\index{péndulo simple}: una masa pequeña $m$ suspendida de un
hilo inextensible de longitud $L$ y masa despreciable, que oscila en un plano
vertical. De las dos \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre la masa (\cref{ch:g12:newtons-laws}), la
tensión es perpendicular al movimiento, mientras que la componente del \omterm{def:g10:universal-gravitation:weight}{peso} a lo largo
de la \omterm{def:g10:relative-motion:trajectory}{trayectoria} apunta de vuelta hacia el punto más bajo: el \omterm{def:g10:universal-gravitation:weight}{peso} es la \emph{fuerza
recuperadora}\index{fuerza recuperadora}, opuesta al desplazamiento.
\end{definition}

\begin{proposition}[Periodo del péndulo simple]\label{prop:g12:oscillators-and-time:pendulum-period}
\index{péndulo simple!periodo}
Para \omterm{def:g12:oscillators-and-time:oscillator}{amplitudes} pequeñas (por debajo de unos $15^\circ$), el \omterm{def:g12:oscillators-and-time:oscillator}{periodo} vale
\[
T_0 = 2\pi \sqrt{\frac{L}{g}}, \qquad g = \qty{9.81}{m/s^2}.
\]
No depende ni de la masa suspendida ni --- mientras el ángulo siga siendo pequeño ---
de la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}: las oscilaciones pequeñas son \emph{isócronas}\index{isocronismo} y
todas duran lo mismo.
\end{proposition}
\admitted

\begin{remark}[Análisis dimensional, y dónde vive el $2\pi$]\label{rem:g12:oscillators-and-time:dimensional}
La fórmula se puede adivinar. A partir de $m$ (\unit{kg}), $L$ (\unit{m}) y $g$
(\unit{m/s^2}), la única combinación con \omterm{def:g10:orders-of-magnitude:unit}{unidades} de tiempo es $\sqrt{L/g}$ --- y la
masa \emph{no puede} entrar, porque ningún otro dato lleva un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} que la
cancele. Las dimensiones no pueden dar el número puro de delante: el $2\pi$ sale de
resolver la ecuación del movimiento, cosa que se hace en el volumen del primer año de
universidad. La condición de ángulo pequeño tampoco es opcional: a $30^\circ$ el
\omterm{def:g12:oscillators-and-time:oscillator}{periodo} verdadero es un $2\,\%$ mayor y el isocronismo falla.
\end{remark}

\begin{example}[El péndulo que bate el segundo]\label{ex:g12:oscillators-and-time:seconds}
¿Qué longitud bate el segundo, $T_0 = \qty{2.00}{s}$, un tic en cada sentido?
Despejando, $L = g\,T_0^2/4\pi^2 = 9.81 \times 2.00^2/4\pi^2 \approx \qty{0.994}{m}$.
Casi un \omterm{def:g10:orders-of-magnitude:unit}{metro}: los relojes de péndulo son altos porque los segundos son largos.
\end{example}

\section{El oscilador masa--resorte}

\begin{definition}[Fuerza recuperadora de un resorte]\label{def:g12:oscillators-and-time:spring}
Un móvil de masa $m$ desliza sobre un raíl horizontal sin \omterm{def:g10:inertia:inventory}{rozamiento}, unido a un
resorte de \emph{constante elástica}\index{constante elástica} $k$ (\unit{N/m}). Sea
$x$ su desplazamiento respecto del equilibrio. Estirado o comprimido, el resorte tira
del móvil o lo empuja de vuelta con la \omterm{def:g10:inertia:force}{fuerza} $F = -kx$: proporcional al
desplazamiento y opuesta a él --- otra vez una \omterm{def:g12:oscillators-and-time:pendulum}{fuerza recuperadora}, suministrada ahora
por la elasticidad.
\end{definition}

\begin{proposition}[Ecuación del movimiento y su solución]\label{prop:g12:oscillators-and-time:spring-period}
\index{oscilador masa--resorte}
La \omterm{thm:g12:newtons-laws:second}{segunda ley de Newton} a lo largo del raíl (\cref{ch:g12:newtons-laws}) da
$m \frac{d^2x}{dt^2} = -kx$, es decir, $\frac{d^2x}{dt^2} = -\frac{k}{m}\,x$. Sean
cuales sean las constantes $A$ y $\varphi$,
\[
x(t) = A \cos\!\left(\frac{2\pi t}{T_0} + \varphi\right),
\qquad T_0 = 2\pi\sqrt{\frac{m}{k}},
\]
satisface esta ecuación: una oscilación de \omterm{def:g12:oscillators-and-time:oscillator}{periodo} $T_0$, independiente de la
\omterm{def:g12:oscillators-and-time:oscillator}{amplitud}.
\end{proposition}

\begin{proof}
Derivando, $\frac{dx}{dt} = -\frac{2\pi}{T_0}A \sin(\frac{2\pi t}{T_0} + \varphi)$; y
otra vez, $\frac{d^2x}{dt^2} = -(\frac{2\pi}{T_0})^2 A\cos(\frac{2\pi t}{T_0} +
\varphi) = -(\frac{2\pi}{T_0})^2\,x(t)$, que es $-(k/m)\,x$ exactamente cuando
$(2\pi/T_0)^2 = k/m$.
\end{proof}

\begin{remark}[Todos los movimientos que hay]\label{rem:g12:oscillators-and-time:uniqueness}
Que \emph{todo} movimiento del móvil es una de estas funciones se admite aquí y se
demuestra en el volumen del primer año de universidad. Y en $2\pi\sqrt{m/k}$ no
aparece $g$: cuanto más rígido, más rápido oscila; cuanto más pesado, más lento; y en
la Luna, igual.
\end{remark}

\begin{definition}[Amplitud y fase]\label{def:g12:oscillators-and-time:phase}
En $x(t) = A\cos(2\pi t/T_0 + \varphi)$, la constante positiva $A$ es la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud} ---
los extremos son $x = \pm A$ --- y $2\pi t/T_0 + \varphi$ es la
\emph{fase}\index{fase}, que crece $2\pi$ en cada \omterm{def:g12:oscillators-and-time:oscillator}{periodo}; la \emph{fase
inicial}\index{fase inicial} $\varphi$ registra el arranque: soltado en reposo desde
$x = A$, $\varphi = 0$. Por ejemplo, $m = \qty{0.20}{kg}$ sobre $k = \qty{20}{N/m}$ da
$T_0 = 2\pi\sqrt{0.20/20} = \qty{0.63}{s}$; llevado hasta \qty{5.0}{cm} y soltado,
$x(t) = 5.0\cos(2\pi t/0.63)$ en centímetros.
\end{definition}

\section{Energía, amortiguamiento, resonancia}

\begin{proposition}[Energía del oscilador]\label{prop:g12:oscillators-and-time:energy}
Un resorte estirado una longitud $x$ almacena la \omterm{def:g10:energy-conservation:energy}{energía} potencial elástica
$E_p = \tfrac12 kx^2$ (justificada en el \cref{ch:g12:work-and-energy}). Durante una
oscilación sin \omterm{def:g10:inertia:inventory}{rozamiento}, la \omterm{def:g11:mechanical-energy:em}{energía mecánica}
\[
E_m = E_k + E_p = \tfrac12 m v^2 + \tfrac12 k x^2 = \tfrac12 k A^2
\]
es constante: toda elástica en los extremos, toda cinética en el equilibrio. El
péndulo interpreta el mismo dúo entre $E_k$ y la $E_p$ gravitatoria
(\cref{ch:g11:mechanical-energy}).
\end{proposition}

\begin{proof}
Con $v = \frac{dx}{dt}$, la derivada de $E_m$ es
$mv\frac{dv}{dt} + kx\frac{dx}{dt} = v\,(m\frac{d^2x}{dt^2} + kx) = 0$ por la ecuación
del movimiento; el valor $\tfrac12 kA^2$ se lee en un extremo.
\end{proof}

\begin{definition}[Oscilaciones amortiguadas]\label{def:g12:oscillators-and-time:damping}
El \omterm{def:g10:inertia:inventory}{rozamiento} drena \omterm{def:g11:mechanical-energy:em}{energía mecánica} de todo \omterm{def:g12:oscillators-and-time:oscillator}{oscilador} real: sus \omterm{def:g12:oscillators-and-time:oscillator}{oscilaciones libres}
son \emph{amortiguadas}\index{amortiguamiento}. Con amortiguamiento débil el sistema
sigue oscilando, con una \omterm{def:g12:oscillators-and-time:oscillator}{amplitud} que se apaga pero con un tiempo de repetición casi
inalterado, el \emph{seudoperiodo}\index{seudoperiodo}, próximo a $T_0$; con
amortiguamiento fuerte (miel), vuelve reptando al equilibrio sin pasarse nunca; y la
\omterm{def:g10:energy-conservation:energy}{energía}, proporcional al cuadrado de la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}, se degrada en calor.
\end{definition}

\begin{definition}[Oscilaciones forzadas y resonancia]\label{def:g12:oscillators-and-time:resonance}
Empuja un \omterm{def:g12:oscillators-and-time:oscillator}{oscilador} periódicamente --- \emph{oscilaciones forzadas}\index{oscilaciones
forzadas} --- y, tras un breve transitorio, oscilará a la \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} $f$ \emph{del
excitador}, no a la suya. Su \omterm{def:g12:oscillators-and-time:oscillator}{amplitud} alcanza un máximo cuando $f$ se acerca a la
\emph{frecuencia propia}\index{frecuencia propia} $f_0$ del \omterm{def:g12:oscillators-and-time:oscillator}{oscilador} libre: es la
\emph{resonancia}\index{resonancia}, con cada empujón llegando al paso del movimiento
y aportando \omterm{def:g10:energy-conservation:energy}{energía} en cada ciclo; cuanto más débil es el \omterm{def:g12:oscillators-and-time:damping}{amortiguamiento}, más alto y
más agudo es el pico.
\end{definition}

\begin{example}[La resonancia, para bien y para mal]\label{ex:g12:oscillators-and-time:resonance-zoo}
Un padre que empuja un columpio empuja a la \omterm{def:g12:oscillators-and-time:resonance}{frecuencia propia} de este --- la
\omterm{def:g12:oscillators-and-time:resonance}{resonancia} puesta a trabajar. En \num{2000}, el nuevo puente del Milenio de Londres se
balanceó de forma alarmante cuando los pasos de la multitud se acoplaron a una
\omterm{def:g12:oscillators-and-time:resonance}{frecuencia propia} cercana a \qty{1}{Hz}; estuvo cerrado dos años para instalarle
amortiguadores. Y todo reloj de cuarzo mantiene su cristal cantando excitándolo en
\omterm{def:g12:oscillators-and-time:resonance}{resonancia}, a \qty{32768}{Hz}.
\end{example}

\section{Medir el tiempo}

\begin{method}[Cronometrar un periodo]\label{met:g12:oscillators-and-time:timing}
No cronometres nunca una sola oscilación: pon en marcha el cronómetro cuando la masa
cruce la vertical (donde va más rápido y resulta más fácil de apreciar), cuenta
$N = 50$ oscilaciones y divide entre $N$ --- un error de reacción de \qty{0.2}{s} se
convierte en \qty{4}{ms} sobre $T_0$. Repite y compara las series.
\end{method}

\begin{example}[Tres siglos de tic]\label{ex:g12:oscillators-and-time:clocks}
Huygens construyó el primer reloj de péndulo en \num{1657}: el isocronismo llevó a los
relojes de un cuarto de hora de deriva diaria a unos segundos, y a los péndulos
compensados, a fracciones de segundo al día. El reloj de pulsera de cuarzo
(\num{1969}) cambió la varilla por un cristal que se flexiona a \qty{32768}{Hz}:
décimas de segundo al mes. Desde \num{1967}, el propio segundo se \emph{define} por
una oscilación atómica: la duración de \num{9192631770} \omterm{def:g12:oscillators-and-time:oscillator}{periodos} de la radiación de
microondas de una transición interna del cesio-133
(\cref{ch:g12:quantum-world}). Los mejores relojes de cesio derivan un segundo en cien
millones de años.
\end{example}

\begin{remark}[Por qué perseguir el nanosegundo]\label{rem:g12:oscillators-and-time:gps}
Un receptor GPS halla su posición cronometrando \omterm{def:g10:signals-and-waves:signal}{señales} de radio procedentes de
\omterm{def:g10:universal-gravitation:satellite}{satélites}; la luz recorre \qty{30}{cm} por nanosegundo, así que un error de reloj de
un microsegundo te sitúa \qty{300}{m} más allá. Por eso los \omterm{def:g10:universal-gravitation:satellite}{satélites} llevan relojes
atómicos --- y a esa precisión el ritmo de un reloj depende de la rapidez a la que se
mueve y de la altura a la que está, un efecto que encontraremos en el
\cref{ch:g12:special-relativity}.
\end{remark}

\section{Ejercicios}

\section{Problema: El banco del relojero}

\begin{problem}[{Problema de fin de semana --- el banco del relojero: un reloj de pared
devuelto a la hora --- cincuenta oscilaciones cronometradas, una varilla de latón que
se dilata en verano, el empujón susurrado del escape, un aspirante de cuarzo --- y el
veredicto dictado en segundos al mes}]
\label{pb:g12:oscillators-and-time:1}
A tu banco de \omterm{def:g11:work-of-force:work}{trabajo} llega un reloj de pared de péndulo que ``va mal''. Trata su
varilla de latón y su pesada lenteja como un \omterm{def:g12:oscillators-and-time:pendulum}{péndulo simple} de longitud eficaz $L$,
con una masa $m = \qty{1.2}{kg}$; el engranaje cuenta cada oscilación completa como
exactamente \qty{2.000}{s}.

\medskip
\noindent\textbf{Parte I --- Tomarle el pulso al reloj.}
\begin{enumerate}
  \item ¿Por qué cronometrar $50$ oscilaciones y no una? Estima el error que queda
        sobre $T_0$ si tu tiempo de reacción es de \qty{0.2}{s}.
  \item El cronómetro marca \qty{99.4}{s} para $50$ oscilaciones: ¿el \omterm{def:g12:oscillators-and-time:oscillator}{periodo}?
  \item Deduce la longitud eficaz actual del péndulo.
  \item ¿Qué longitud daría exactamente \qty{2.000}{s}? ¿Debe la tuerca de reglaje
        subir o bajar la lenteja, y cuánto?
  \item Sin ajustar, ¿adelanta o atrasa el reloj, y cuántos segundos al día?
  \item Sensibilidad a $g$: ¿el \omterm{def:g12:oscillators-and-time:oscillator}{periodo} en la Luna ($g = \qty{1.62}{m/s^2}$)? ¿El
        error diario en un observatorio donde $g = \qty{9.80}{m/s^2}$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El bajón del verano.} El latón se dilata: un
calentamiento $\Delta\theta$ estira la varilla hasta
$L' = L\,(1 + \lambda\, \Delta\theta)$, con $\lambda = \num{1.9e-5}$ por \omterm{def:g10:light-spectra:kelvin}{kelvin}. El
reloj se ajusta a \qty{20}{\celsius}.
\begin{enumerate}[resume]
  \item Calcula $\Delta L$ para una habitación de verano a \qty{25}{\celsius}.
  \item Usando $\sqrt{1+x} \approx 1 + x/2$, demuestra que
        $\Delta T_0 / T_0 = \tfrac12 \lambda\, \Delta\theta$.
  \item Calcula $\Delta T_0/T_0$ y el error diario de verano: ¿adelanta o atrasa?
  \item Las mismas preguntas para un pasillo de invierno a \qty{10}{\celsius}.
  \item Los relojes antiguos de precisión usaban péndulos ``de parrilla'' que
        alternaban varillas de latón con otras de acero (que se dilata alrededor de la
        mitad) en sentidos opuestos. Explica cómo puede eso mantener la marcha.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El empujón susurrado.} Con el escape desembragado, la
oscilación, iniciada con \omterm{def:g12:oscillators-and-time:oscillator}{amplitud} $\theta_m = 2.00^\circ$, reduce su \omterm{def:g12:oscillators-and-time:oscillator}{amplitud} a la
mitad en \qty{280}{s}; en servicio, un empujoncito del escape en cada \omterm{def:g12:oscillators-and-time:oscillator}{periodo} mantiene
$\theta_m$ constante.
\begin{enumerate}[resume]
  \item ¿Cuántos \omterm{def:g12:oscillators-and-time:oscillator}{periodos} dura ese descenso a la mitad?
  \item En el extremo, la lenteja ha subido $h = L(1 - \cos\theta_m) \approx
        L\,\theta_m^2/2$ (en radianes): deduce $E_m = \tfrac12 m g L\,\theta_m^2$ y
        calcúlala (\cref{ch:g11:mechanical-energy}).
  \item La \omterm{def:g10:energy-conservation:energy}{energía} es proporcional al cuadrado de la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}: ¿qué fracción de $E_m$
        se pierde por \omterm{def:g12:oscillators-and-time:oscillator}{periodo}?
  \item Deduce la \omterm{def:g10:energy-conservation:energy}{energía} de cada empujón y la \omterm{def:g11:work-of-force:power}{potencia} media necesaria.
  \item La pesa motriz $M = \qty{2.5}{kg}$ desciende $h = \qty{1.0}{m}$ por semana:
        ¿basta esa \omterm{def:g10:energy-conservation:energy}{energía}, y con qué \omterm{def:g10:energy-conservation:efficiency}{rendimiento}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El aspirante de cuarzo.} Un movimiento de cuarzo oscila
a $f = \qty{32768}{Hz}$ y deriva unos \qty{0.3}{s} al mes; el péndulo restaurado sigue
derivando \qty{0.5}{s} al día.
\begin{enumerate}[resume]
  \item Calcula el \omterm{def:g12:oscillators-and-time:oscillator}{periodo} del cuarzo. Demuestra que $\num{32768} = 2^{15}$: ¿por qué
        una potencia de dos es justo lo que quiere el circuito de un reloj?
  \item Calcula las derivas relativas del cuarzo y del péndulo restaurado, y cada una
        en segundos al mes. ¿Quién gana, y por qué factor?
  \item El patrón de cesio --- \num{9192631770} \omterm{def:g12:oscillators-and-time:oscillator}{periodos} hacen un segundo --- deriva
        \qty{1}{s} en cien millones de años: ¿su deriva relativa? ¿Cuántos órdenes de
        magnitud por debajo del cuarzo, y por qué lo exige el GPS?
  \item El veredicto, en una frase con cifras: ¿qué se gana el reloj restaurado en la
        pared, y de quién es ahora el segundo?
\end{enumerate}
\end{problem}
