\chapter{Energía nuclear: fisión, fusión, $E=mc^2$}\label{ch:g12:nuclear-energy}

Una pastilla de combustible del tamaño de la goma de un lápiz cubre la electricidad de
un hogar durante un año; el mismo servicio con carbón exige tres toneladas. Sobre
nuestras cabezas, el Sol paga la luz del día desapareciendo --- cuatro millones de
toneladas de sí mismo por segundo. Este capítulo abre el libro de cuentas que hay
detrás de ambas gangas: la propia masa es \omterm{def:g10:energy-conservation:energy}{energía}, y los \omterm{def:g11:fundamental-interactions:ladder}{núcleos}
(\cref{ch:g11:nucleus-radioactivity}) son donde se cobra.

\section{La masa es energía}

\begin{theorem}[Equivalencia masa--energía]\label{thm:g12:nuclear-energy:emc2}\index{equivalencia masa--energía}
Un cuerpo de masa $m$, por el mero hecho de existir, alberga la \omterm{def:g10:energy-conservation:energy}{energía}
\[
E = m c^2, \qquad c = \qty{3.00e8}{m/s}.
\]
Siempre que un sistema pierde una masa $\Delta m$, libera la \omterm{def:g10:energy-conservation:energy}{energía}
$\Delta m\, c^2$ --- sea cual sea el mecanismo.
\end{theorem}
\admitted

\begin{remark}[Einstein, 1905]\label{rem:g12:nuclear-energy:einstein}
Esta es la relación de Einstein, admitida aquí; de dónde sale es asunto del capítulo
de relatividad especial que cierra el curso (\cref{ch:g12:special-relativity}).
Fíjate en la tarifa: $c^2 = \qty{9.00e16}{J/kg}$ --- un error de redondeo de masa es
una fortuna de \omterm{def:g10:energy-conservation:energy}{energía}.
\end{remark}

\begin{definition}[Electronvoltio y unidad de masa atómica]\label{def:g12:nuclear-energy:units}
Dos \omterm{def:g10:orders-of-magnitude:unit}{unidades} hechas a la medida del \omterm{def:g11:fundamental-interactions:ladder}{núcleo}. El
\emph{electronvoltio}\index{electronvoltio}, la \omterm{def:g10:energy-conservation:energy}{energía} que gana un electrón al
atravesar \qty{1}{V} (\cref{ch:g11:circuits-and-power}):
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{1}{MeV} = \qty{1.60e-13}{J}$ --- los enlaces
químicos manejan unos pocos \unit{eV}, y las reacciones nucleares,
megaelectronvoltios. Y la \emph{unidad de masa atómica unificada}\index{unidad de masa
atómica unificada}, la doceava parte de la masa de un \omterm{def:g11:fundamental-interactions:ladder}{átomo} de carbono-12:
$\qty{1}{u} = \qty{1.66054e-27}{kg}$, de \omterm{def:g10:energy-conservation:energy}{energía} equivalente \qty{931.5}{MeV} ---
$\qty{1}{u} = \qty{931.5}{MeV}/c^2$.
\end{definition}

\begin{example}[La fortuna dormida]\label{ex:g12:nuclear-energy:fortune}
Un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de cualquier cosa, convertido por completo, son $\qty{9.00e16}{J}$ ---
tres años de producción de una central grande (de \qty{1}{GW}). Nadie se dio cuenta
antes de 1905 porque la física ordinaria apenas toca la masa: quemar un \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de
carbón (\qty{3.0e7}{J}) lo aligera en un tercio de microgramo. Los \omterm{def:g11:fundamental-interactions:ladder}{núcleos}, ya lo
veremos, mueven una parte de cada mil: sigue siendo poco, pero es un millón de veces
la química.
\end{example}

\section{El defecto de masa}

\begin{definition}[Defecto de masa y energía de enlace]\label{def:g12:nuclear-energy:defect}
Pesa un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ${}^{A}_{Z}\mathrm{X}$ (de masa $m$) y pesa después por separado sus
$Z$ protones y sus $A - Z$ neutrones: las partes son \emph{más pesadas} que el todo.
La diferencia
\[
\Delta m = Z\,m_p + (A - Z)\,m_n - m > 0,
\qquad m_p = \qty{1.00728}{u},\; m_n = \qty{1.00866}{u},
\]
es el \emph{defecto de masa}\index{defecto de masa}; la \emph{energía de
enlace}\index{energía de enlace} $E_b = \Delta m\,c^2$ es la \omterm{def:g10:energy-conservation:energy}{energía} necesaria para
descomponer el \omterm{def:g11:fundamental-interactions:ladder}{núcleo} en nucleones libres --- e igualmente, la \omterm{def:g10:energy-conservation:energy}{energía} liberada cuando
se ensambla. Un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ligado se sitúa \emph{por debajo} de sus partes.
\end{definition}

\end{omfigure}

\begin{example}[Helio-4]\label{ex:g12:nuclear-energy:helium}
El \omterm{def:g11:fundamental-interactions:ladder}{núcleo} ${}^{4}_{2}\mathrm{He}$ tiene $m = \qty{4.00151}{u}$; sus partes:
$2 \times 1.00728 + 2 \times 1.00866 = \qty{4.03188}{u}$. Así,
$\Delta m = \qty{0.03037}{u}$ y $E_b = 0.03037 \times 931.5 \approx \qty{28.3}{MeV}$:
el \omterm{def:g11:fundamental-interactions:ladder}{núcleo} pesa un $0.75\,\%$ menos que sus partes --- la ``parte de cada mil'' del
\cref{ex:g12:nuclear-energy:fortune}.
\end{example}

\begin{method}[Balance energético de una reacción nuclear]\label{met:g12:nuclear-energy:balance}
\begin{enumerate}
  \item Ajusta la ecuación: $A$ y $Z$ totales se conservan
        (\cref{ch:g11:nucleus-radioactivity}).
  \item Suma las masas de antes y después, en \unit{u} --- conserva las cinco cifras
        decimales.
  \item $\Delta m = m_{\text{antes}} - m_{\text{después}}$, y después
        $E = \Delta m\,(\text{en u}) \times \qty{931.5}{MeV}$, que se llevan los
        productos como \omterm{def:g11:mechanical-energy:kinetic}{energía cinética}; si sale negativa, la reacción hay que
        pagarla.
\end{enumerate}
\end{method}

\section{La curva que gobierna el universo}

\begin{definition}[Energía de enlace por nucleón]\label{def:g12:nuclear-energy:pernucleon}
La cohesión se compara con equidad mediante la \emph{energía de enlace por
nucleón}\index{energía de enlace por nucleón}, $E_b/A$, el coste medio de extraer un
\omterm{def:g11:fundamental-interactions:ladder}{nucleón}: $28.3/4 \approx \qty{7.1}{MeV}$ para el helio-4.
\end{definition}

\begin{proposition}[El pico del hierro y dos caminos de bajada]\label{prop:g12:nuclear-energy:ironpeak}\index{pico del hierro}
Representada frente a $A$, la \omterm{def:g12:nuclear-energy:pernucleon}{energía de enlace por nucleón} trepa con \omterm{def:g10:inertia:force}{fuerza} a lo
largo de los \omterm{def:g11:fundamental-interactions:ladder}{núcleos} ligeros, alcanza su máximo cerca del hierro,
${}^{56}_{26}\mathrm{Fe}$, en unos \qty{8.8}{MeV} por \omterm{def:g11:fundamental-interactions:ladder}{nucleón}, y después desciende
suavemente hasta unos \qty{7.6}{MeV} en el uranio: el hierro es el \omterm{def:g11:fundamental-interactions:ladder}{núcleo} más
fuertemente ligado de la naturaleza. Una reacción libera \omterm{def:g10:energy-conservation:energy}{energía} exactamente cuando
sus productos quedan \emph{más arriba} en la curva; por eso dos estrategias opuestas
salen ambas rentables, y ambas caminan hacia el hierro --- \emph{partir} los \omterm{def:g11:fundamental-interactions:ladder}{núcleos}
más pesados y \emph{fusionar} los más ligeros.
\end{proposition}
\admitted

\begin{remark}[Por qué hay un pico]\label{rem:g12:nuclear-energy:whypeak}
El tira y afloja del \cref{ch:g11:fundamental-interactions}: el pegamento fuerte solo
liga a los vecinos que se tocan, mientras que la repulsión de Coulomb abarca el \omterm{def:g11:fundamental-interactions:ladder}{núcleo}
entero --- los \omterm{def:g11:fundamental-interactions:ladder}{núcleos} pequeños son casi todo superficie, los grandes casi toda
repulsión, y el hierro es el compromiso. La contabilidad honrada vive en los volúmenes
universitarios.
\end{remark}

\section{La fisión y la reacción en cadena}

\begin{definition}[Fisión]\label{def:g12:nuclear-energy:fission}
La \emph{fisión}\index{fisión} es la partición de un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} pesado en dos fragmentos
de tamaño medio. El uranio-235 es \emph{fisible}\index{fisible}: un neutrón lento,
absorbido, deja el \omterm{def:g11:fundamental-interactions:ladder}{núcleo} tan agitado que se parte en dos y escupe dos o tres neutrones
nuevos ---
${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to \text{dos fragmentos} +
2\text{--}3\;{}^{1}_{0}\mathrm{n}$, con la pareja variando de una fisión a otra.
\end{definition}

\begin{example}[Una fisión, pesada]\label{ex:g12:nuclear-energy:fissionbalance}
Una vía frecuente, siguiendo el \cref{met:g12:nuclear-energy:balance}:
\[
{}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \longrightarrow
{}^{94}_{38}\mathrm{Sr} + {}^{140}_{54}\mathrm{Xe} + 2\,{}^{1}_{0}\mathrm{n}.
\]
Antes: $1.00866 + 234.99350 = \qty{236.00216}{u}$. Después:
$93.89450 + 139.89200 + 2 \times 1.00866 = \qty{235.80382}{u}$. De ahí,
$\Delta m = \qty{0.19834}{u}$ y $E \approx \qty{185}{MeV}$; contando las
desintegraciones posteriores de los fragmentos radiactivos
(\cref{ch:g12:radioactive-decay}), cada \omterm{def:g12:nuclear-energy:fission}{fisión} vale unos \qty{200}{MeV} --- unos pocos
\unit{eV} compran un \omterm{def:g11:fundamental-interactions:ladder}{átomo} de carbón.
\end{example}

\begin{definition}[Reacción en cadena y masa crítica]\label{def:g12:nuclear-energy:chain}
Cada \omterm{def:g12:nuclear-energy:fission}{fisión} la enciende un neutrón y libera dos o tres: en un bloque suficientemente
grande golpean otros \omterm{def:g11:fundamental-interactions:ladder}{núcleos} y la reacción se alimenta sola --- una \emph{reacción en
cadena}\index{reacción en cadena}. En un bloque pequeño, la mayoría de los neutrones
se escapa antes por la superficie; la cantidad mínima que sostiene la cadena es la
\emph{masa crítica}\index{masa crítica} --- por debajo de ella la cadena se apaga y
por encima cada generación se multiplica: el principio de la bomba.
\end{definition}

\begin{definition}[El reactor, domado]\label{def:g12:nuclear-energy:reactor}
Un reactor de \omterm{def:g11:work-of-force:power}{potencia} mantiene la cadena en exactamente un neutrón por \omterm{def:g12:nuclear-energy:fission}{fisión}. El
combustible: pastillas de uranio modestamente enriquecidas en uranio-235. El
\emph{moderador}\index{moderador} (agua corriente, por lo general) frena por choques
los neutrones rápidos de \omterm{def:g12:nuclear-energy:fission}{fisión} --- los neutrones lentos son mucho mejores para
provocar la \omterm{def:g12:nuclear-energy:fission}{fisión} siguiente; las \emph{barras de control}\index{barra de control}
(boro, cadmio) devoran neutrones --- introducidas, la cadena muere; extraídas, se
aviva. La agitación de los fragmentos se convierte en calor, después en vapor y
después en electricidad (\cref{ch:g10:energy-conservation}).
\end{definition}

\begin{example}[Un grano de pimienta contra un camión]\label{ex:g12:nuclear-energy:gram}
Un gramo de uranio-235 alberga
$N = \num{1.0e-3} / (235 \times \num{1.66054e-27}) \approx \num{2.56e21}$ \omterm{def:g11:fundamental-interactions:ladder}{núcleos}; a
$\qty{200}{MeV} = \qty{3.2e-11}{J}$ cada uno, fisionarlos todos rinde unos
\qty{8.2e10}{J} --- la \omterm{def:g10:energy-conservation:forms}{energía química} de
$\num{8.2e10} / \num{3.0e7} \approx \qty{2.7e3}{kg}$ de carbón: un grano de pimienta
contra tres toneladas.
\end{example}

\section{La fusión: el fuego propio del Sol}

\begin{definition}[Fusión]\label{def:g12:nuclear-energy:fusion}
La \emph{fusión}\index{fusión} es la unión de dos \omterm{def:g11:fundamental-interactions:ladder}{núcleos} ligeros en otro más pesado y
más fuertemente ligado. La reacción más accesible casa el deuterio con el tritio, los
\omterm{def:g11:nucleus-radioactivity:notation}{isótopos} pesados del hidrógeno (\cref{ch:g11:nucleus-radioactivity}):
\[
{}^{2}_{1}\mathrm{H} + {}^{3}_{1}\mathrm{H} \longrightarrow
{}^{4}_{2}\mathrm{He} + {}^{1}_{0}\mathrm{n}.
\]
\end{definition}

\begin{example}[D--T, pesada]\label{ex:g12:nuclear-energy:dt}
Antes: $2.01355 + 3.01550 = \qty{5.02905}{u}$; después:
$4.00151 + 1.00866 = \qty{5.01017}{u}$; luego $\Delta m = \qty{0.01888}{u}$ y
$E \approx \qty{17.6}{MeV}$, cuatro quintas partes de ella sobre el neutrón --- menos
que una \omterm{def:g12:nuclear-energy:fission}{fisión}, pero a partir de cinco nucleones en lugar de $236$: unos
\qty{3.4e14}{J} por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de combustible, cuatro veces la \omterm{def:g12:nuclear-energy:fission}{fisión}. El helio-4
(\cref{ex:g12:nuclear-energy:helium}) vuelve a ser la ceniza.
\end{example}

\begin{definition}[La barrera de Coulomb]\label{def:g12:nuclear-energy:barrier}
Dos \omterm{def:g11:fundamental-interactions:ladder}{núcleos} son ambos positivos: para tocarse tienen que trepar antes la colina de su
repulsión eléctrica (\cref{ch:g11:electric-gravitational-fields}) --- la \emph{barrera
de Coulomb}\index{barrera de Coulomb}. Solo cerca de \qty{1e8}{K} llevan los choques
la \omterm{def:g10:energy-conservation:energy}{energía} necesaria para cruzarla; el combustible es entonces un
\emph{plasma}\index{plasma} de \omterm{def:g11:fundamental-interactions:ladder}{núcleos} desnudos y electrones. La \omterm{def:g12:nuclear-energy:fission}{fisión} no paga ese
peaje: su detonante, el neutrón, es neutro y funciona en frío.
\end{definition}

\begin{example}[El Sol funciona con ella]\label{ex:g12:nuclear-energy:sun}
El núcleo del Sol, a $\qty{1.5e7}{K}$, fusiona hidrógeno corriente paso a paso --- la
\emph{cadena protón--protón}, de efecto neto
$4\,{}^{1}_{1}\mathrm{H} \to {}^{4}_{2}\mathrm{He} + 2\,{}^{0}_{+1}\mathrm{e}
+ \text{radiación}$, unos \qty{26}{MeV} por cada helio: el $0.7\,\%$ de la masa del
hidrógeno se marcha como \omterm{def:g10:energy-conservation:energy}{energía}. La luz del Sol cuyo \omterm{def:g12:sound-acoustics:timbre}{espectro} leímos en el
\cref{ch:g10:light-spectra} es esa masa, llegando con ocho minutos de \omterm{def:g12:mechanical-waves:delay}{retraso}; toda
estrella brilla recorriendo la curva hacia el hierro.
\end{example}

\begin{remark}[ITER: embotellar una estrella]\label{rem:g12:nuclear-energy:iter}
Ningún sólido puede contener un \omterm{def:g12:nuclear-energy:barrier}{plasma} a \qty{1.5e8}{K}; un \emph{tokamak} lo enjaula
en un \omterm{def:g11:magnetic-fields:bfield}{campo magnético} (\cref{ch:g11:magnetic-fields}), un donut de \omterm{def:g11:electric-gravitational-fields:lines}{líneas de campo} a lo
largo de las cuales las partículas cargadas van en espiral sin tocar la pared; ITER,
construido para liberar diez veces la \omterm{def:g11:work-of-force:power}{potencia} con que se calienta, es el paso actual.
Rompe el confinamiento y el \omterm{def:g12:nuclear-energy:barrier}{plasma} se enfría y se para en segundos: sin \omterm{def:g12:nuclear-energy:chain}{reacción en
cadena} y sin \omterm{def:g12:nuclear-energy:chain}{masa crítica} --- lo difícil no es apagar el fuego, sino mantenerlo
encendido.
\end{remark}

\begin{example}[La escalera de energías, por kilogramo]\label{ex:g12:nuclear-energy:ladder}
\omterm{def:g10:energy-conservation:energy}{Energía} por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de combustible: carbón, \qty{3.0e7}{J}; uranio-235 por \omterm{def:g12:nuclear-energy:fission}{fisión},
\qty{8.2e13}{J}; deuterio--tritio por \omterm{def:g12:nuclear-energy:fusion}{fusión}, \qty{3.4e14}{J}; conversión total
($E = mc^2$), \qty{9.0e16}{J}. Seis órdenes de magnitud separan la química del \omterm{def:g11:fundamental-interactions:ladder}{núcleo}
--- la razón de que un reactor se recargue por camiones y una central de carbón, por
trenes enteros.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:1}
Convierte: (a) \qty{1}{eV} y \qty{1}{MeV} en \omterm{def:g11:work-of-force:work}{julios}; (b) los \qty{200}{MeV} de una
\omterm{def:g12:nuclear-energy:fission}{fisión} en \omterm{def:g11:work-of-force:work}{julios}; (c) comprueba, a partir de
$\qty{1}{u} = \qty{1.66054e-27}{kg}$, que su \omterm{def:g10:energy-conservation:energy}{energía} equivalente ronda los
\qty{931.5}{MeV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:2}
Un terrón de azúcar tiene una masa de \qty{6.0}{g}: calcula su valor íntegro
$E = mc^2$. Un hogar consume $\qty{10}{kW.h} = \qty{3.6e7}{J}$ al día --- ¿durante
cuántos años podría el terrón, convertido por completo, alimentarlo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:3}
El \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de carbono-12 tiene una masa de \qty{11.99671}{u}: calcula su \omterm{def:g12:nuclear-energy:defect}{defecto de
masa}, su \omterm{def:g12:nuclear-energy:defect}{energía de enlace} en \unit{MeV} y su \omterm{def:g12:nuclear-energy:pernucleon}{energía de enlace por nucleón}
($m_p = \qty{1.00728}{u}$, $m_n = \qty{1.00866}{u}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:nuclear-energy:4}
Completa cada \omterm{def:g12:nuclear-energy:fission}{fisión} dando $A$, $Z$ y el elemento ($Z = 36$: kriptón; $Z = 52$:
telurio):
(a) ${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to
{}^{144}_{56}\mathrm{Ba} + {}^{A}_{Z}\mathrm{X} + 3\,{}^{1}_{0}\mathrm{n}$;
(b) ${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to
{}^{97}_{40}\mathrm{Zr} + {}^{A}_{Z}\mathrm{Y} + 2\,{}^{1}_{0}\mathrm{n}$.
\end{exercise}

\section{Problema: Alimentar una ciudad}

\begin{problem}[{Problema de fin de semana --- alimentar una ciudad: un gigavatio
constante para medio millón de personas, comprado de cuatro maneras --- pastillas de
uranio, trenes de carbón, una piscina de agua de mar, la propia sustancia del Sol ---
con el mismo kilogramo de masa desvanecida escondido bajo las cuatro}]
\label{pb:g12:nuclear-energy:1}
Una ciudad de $500\,000$ habitantes consume una \omterm{prop:g11:circuits-and-power:power}{potencia eléctrica} constante
$P = \qty{1.0}{GW}$; sus centrales, sea cual sea el combustible, convierten el calor
en electricidad con un \omterm{def:g10:energy-conservation:efficiency}{rendimiento} del $33\,\%$
(\cref{ch:g10:energy-conservation}). Datos: una \omterm{def:g12:nuclear-energy:fission}{fisión} de uranio-235, \qty{200}{MeV}
en total; carbón, \qty{3.0e7}{J/kg}; una \omterm{def:g12:nuclear-energy:fusion}{fusión} D--T, \qty{17.6}{MeV} a partir de
\qty{5.029}{u} de combustible; $\qty{1}{u} = \qty{1.66054e-27}{kg}$;
$\qty{1}{eV} = \qty{1.60e-19}{J}$; un año $= \qty{3.16e7}{s}$; el Sol:
\qty{3.9e26}{W}, \qty{2.0e30}{kg}; un \omterm{def:g10:orders-of-magnitude:unit}{metro} cúbico de agua de mar alberga unos
\qty{33}{g} de deuterio.

\medskip
\noindent\textbf{Parte I --- La central de \omterm{def:g12:nuclear-energy:fission}{fisión}.}
\begin{enumerate}
  \item ¿Qué \omterm{def:g11:work-of-force:power}{potencia} térmica debe entregar el reactor?
  \item Expresa la \omterm{def:g10:energy-conservation:energy}{energía} de una \omterm{def:g12:nuclear-energy:fission}{fisión} en \omterm{def:g11:work-of-force:work}{julios}.
  \item ¿Cuántas fisiones por segundo hacen funcionar la ciudad?
  \item ¿La masa de un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de uranio-235 ($A = 235$) y, después, el uranio-235
        consumido por segundo?
  \item Deduce el uranio-235 consumido en un año, en \omterm{def:g10:orders-of-magnitude:unit}{kilogramos}.
  \item El combustible está enriquecido al $4.0\,\%$ en uranio-235: ¿qué masa exige un
        año, y cabe en un solo camión?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La cuenta del carbón.}
\begin{enumerate}[resume]
  \item Calcula la \omterm{def:g10:energy-conservation:forms}{energía térmica} que consume la ciudad en un año.
  \item ¿Qué masa de carbón la entrega?
  \item Calcula el cociente (masa de carbón)/(masa de uranio-235) para un año y
        contrástalo con la escalera del \cref{ex:g12:nuclear-energy:ladder}.
  \item Un tren de mercancías arrastra \qty{3.0e6}{kg}: ¿cuántos trenes al año y al
        día? En una frase: imagina las dos cadenas de suministro.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El sueño de la \omterm{def:g12:nuclear-energy:fusion}{fusión}.}
\begin{enumerate}[resume]
  \item Expresa la \omterm{def:g10:energy-conservation:energy}{energía} de una \omterm{def:g12:nuclear-energy:fusion}{fusión} D--T en \omterm{def:g11:work-of-force:work}{julios}.
  \item Calcula la \omterm{def:g10:energy-conservation:energy}{energía} liberada por \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de combustible D--T; compárala con
        la escalera.
  \item ¿Qué masa de combustible D--T cubre el año de la ciudad? ¿Qué parte de esa
        masa es deuterio ($2.014$ de las \qty{5.029}{u})?
  \item ¿Qué volumen de agua de mar contiene ese deuterio? Compáralo con una piscina
        olímpica, de unos \qty{2.5e3}{m^3}. (El tritio se fabrica a partir del litio.)
  \item ¿Por qué esta central no lleva consigo ninguna \omterm{def:g12:nuclear-energy:chain}{masa crítica} ni ninguna
        \omterm{def:g12:nuclear-energy:chain}{reacción en cadena} que temer? Entonces, ¿qué es lo difícil?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La estrella que paga en masa.}
\begin{enumerate}[resume]
  \item A partir de $E = mc^2$, ¿qué masa convierte el Sol cada segundo?
  \item ¿Qué masa por segundo, y por año, debe convertir \emph{cualquier} fuente de
        \qty{3.0}{GW} térmicos? Comprueba que los tres combustibles entregan esa misma
        masa.
  \item ¿Qué masa total ha radiado el Sol en sus \num{4.5e9} años, y qué fracción del
        Sol es eso?
  \item La \omterm{def:g12:nuclear-energy:fusion}{fusión} puede aprovechar alrededor del $0.07\,\%$ de la masa del Sol (la
        parte quemable de su \omterm{def:g11:fundamental-interactions:ladder}{núcleo}): estima su vida total como estrella y lo que le
        queda.
  \item Cierra la auditoría en tres frases: un año de la ciudad en toneladas de
        carbón, en \omterm{def:g10:orders-of-magnitude:unit}{kilogramos} de uranio-235 y en \omterm{def:g10:orders-of-magnitude:unit}{kilogramos} de combustible D--T --- y
        el \omterm{def:g10:orders-of-magnitude:unit}{kilogramo} de masa que, en todos los escenarios, se convirtió en la luz de
        la ciudad.
\end{enumerate}
\end{problem}
