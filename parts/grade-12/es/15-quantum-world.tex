\chapter{El mundo cuántico: fotones y niveles de energía}\label{ch:g12:quantum-world}

Una lámpara de \omterm{def:g10:light-spectra:wavelength}{infrarrojos} puede bañarte la piel toda la tarde sin dejarte marca; diez
minutos de sol de alta montaña dejan una quemadura. La lámpara entrega muchísima más
\omterm{def:g10:energy-conservation:energy}{energía} --- pero no es la \omterm{def:g10:energy-conservation:energy}{energía} lo que marca la piel. La luz, como muestra este
capítulo, llega en granos, y solo un grano que golpee con suficiente \omterm{def:g10:inertia:force}{fuerza} hace
química: un grano \omterm{def:g10:light-spectra:wavelength}{ultravioleta} rompe un enlace que mil millones de granos \omterm{def:g10:light-spectra:wavelength}{infrarrojos}
solo calientan. Sigue los granos y se abre un siglo entero: metales que sueltan
electrones, \omterm{def:g11:fundamental-interactions:ladder}{átomos} que cantan en \omterm{def:g10:light-spectra:emission}{líneas espectrales}, electrones que \omterm{prop:g12:light-as-wave:conditions}{interfieren} como
\omterm{def:g10:signals-and-waves:wave}{ondas}.

\section{La luz llega en granos: el fotón}

\begin{definition}[El fotón]\label{def:g12:quantum-world:photon}
La luz de \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} $\nu$ intercambia \omterm{def:g10:energy-conservation:energy}{energía} con la materia solo en granos enteros
llamados \emph{fotones}\index{fotón}, cada uno de \omterm{def:g10:energy-conservation:energy}{energía}
\[
E = h\nu = \frac{hc}{\lambda}, \qquad h = \qty{6.63e-34}{J.s},
\]
donde $h$ es la \emph{constante de Planck}\index{constante de Planck} (Planck, 1900;
Einstein, 1905) y $\lambda$ la \omterm{def:g12:light-as-wave:lightwave}{longitud de onda en el vacío}. La intensidad de un haz
mide el \emph{número} de fotones por segundo; la \omterm{def:g10:energy-conservation:energy}{energía} de cada grano la fija
únicamente la \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia}.
\end{definition}

\begin{notation}[El electronvoltio]\label{not:g12:quantum-world:ev}
La física atómica usa el \emph{electronvoltio}\index{electronvoltio},
$\qty{1}{eV} = \qty{1.60e-19}{J}$: la \omterm{def:g10:energy-conservation:energy}{energía} que gana una \omterm{def:g11:fundamental-interactions:elementary}{carga elemental} al
atravesar \qty{1}{V} (\cref{ch:g11:electric-gravitational-fields}). Muy cómodo:
$E \,(\unit{eV}) \approx 1240/\lambda\,(\unit{nm})$.
\end{notation}

\section{El efecto fotoeléctrico}

Ilumina una placa metálica limpia en el vacío y saltan electrones --- el \emph{efecto
fotoeléctrico}\index{efecto fotoeléctrico}. Sus reglas rompieron la física clásica.

\begin{proposition}[Lo que muestra el experimento]\label{prop:g12:quantum-world:facts}
Para cada metal existe una \emph{frecuencia umbral}\index{frecuencia umbral} $\nu_0$:
\begin{enumerate}
  \item por debajo de $\nu_0$ no sale ningún electrón, por intensa que sea la luz;
  \item por encima de $\nu_0$ la emisión empieza sin retardo, por débil que sea;
  \item la intensidad aumenta el \emph{número} de electrones por segundo, no su
        \omterm{def:g11:mechanical-energy:kinetic}{energía cinética} máxima; la \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} sí aumenta $E_{k,\max}$.
\end{enumerate}
Nada de esto encaja con una \omterm{def:g10:signals-and-waves:wave}{onda} continua, que debería permitir la acumulación de
carga con cualquier color --- durante meses, con luz débil
(\cref{exo:g12:quantum-world:14}) --- y cuya intensidad debería fijar la \omterm{def:g10:energy-conservation:energy}{energía} de
los electrones: las tres predicciones fallan.
\end{proposition}
\admitted

\begin{theorem}[Relación fotoeléctrica de Einstein]\label{thm:g12:quantum-world:einstein}
\index{relación fotoeléctrica}Un electrón necesita al menos el \emph{trabajo de
extracción}\index{trabajo de extracción} $W_0$ del metal para escapar de su
superficie. Un \omterm{def:g12:quantum-world:photon}{fotón} es absorbido entero por un electrón, así que
\[
h\nu = W_0 + E_{k,\max}, \qquad \text{de donde } \nu_0 = \frac{W_0}{h}.
\]
\end{theorem}

\begin{proof}
Contabilidad energética: entra $h\nu$, se gasta $W_0$ en escapar y el resto es
cinético --- como mucho $h\nu - W_0$, y menos para los electrones que parten de más
adentro. Por debajo de $\nu_0$ ningún grano puede pagar el peaje de salida, por
muchos que sean.
\end{proof}

\begin{definition}[Potencial de frenado]\label{def:g12:quantum-world:stopping}
Para medir $E_{k,\max}$ se frena a los electrones con una tensión inversa
(\cref{ch:g11:electric-gravitational-fields}); el \emph{potencial de
frenado}\index{potencial de frenado} $U_s$ es la menor tensión que anula la corriente:
los electrones más rápidos se quedan justo sin cruzar, luego $E_{k,\max} = e\,U_s$.
\end{definition}

\begin{method}[Medir la constante de Planck]\label{met:g12:quantum-world:measure-h}
\begin{enumerate}
  \item Ilumina una célula fotoeléctrica a través de \omterm{def:g11:color-light-sources:subtractive}{filtros} de \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}
        conocida; calcula cada $\nu = c/\lambda$; anota cada \omterm{def:g12:quantum-world:stopping}{potencial de frenado}.
  \item Representa $E_{k,\max} = eU_s$ frente a $\nu$: Einstein promete la recta
        $h\nu - W_0$.
  \item Lee la pendiente: \emph{es} $h$; las ordenadas en el origen dan $\nu_0$ y
        $-W_0$.
\end{enumerate}
\end{method}

\section{Niveles de energía: por qué los átomos emiten líneas}

\begin{definition}[Niveles de energía]\label{def:g12:quantum-world:levels}
La \omterm{def:g10:energy-conservation:energy}{energía} de un electrón ligado en un \omterm{def:g11:fundamental-interactions:ladder}{átomo} está
\emph{cuantizada}\index{cuantización}: restringida a una escalera discreta de
\emph{niveles de energía}\index{nivel de energía} $E_1 < E_2 < \dots < 0$, contados
como negativos a partir del electrón libre. El peldaño más bajo es el \emph{estado
fundamental}\index{estado fundamental} y los demás son \emph{estados
excitados}\index{estado excitado}; llegar a $0$ es la
\emph{ionización}\index{ionización}.
\end{definition}

\begin{proposition}[La escalera del hidrógeno]\label{prop:g12:quantum-world:hydrogen}
Los \omterm{def:g12:quantum-world:levels}{niveles de energía} del \omterm{def:g11:fundamental-interactions:ladder}{átomo} de hidrógeno son
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2}, \qquad n = 1, 2, 3, \dots
\]
de modo que $E_1 = \qty{-13.6}{eV}$, $E_2 = \qty{-3.40}{eV}$,
$E_3 = \qty{-1.51}{eV}$, $E_4 = \qty{-0.85}{eV}$, apiñándose hacia $0$.
\end{proposition}
\admitted

\begin{remark}[De dónde sale la escalera]\label{rem:g12:quantum-world:ladder-origin}
Por qué esos números es mecánica cuántica, reservada a los volúmenes universitarios
--- como también lo es la fórmula de De Broglie: la \omterm{def:g12:quantum-world:debroglie}{onda de materia} del electrón
(\cref{def:g12:quantum-world:debroglie}, más abajo) debe cerrarse sobre sí misma
alrededor del \omterm{def:g11:fundamental-interactions:ladder}{núcleo}, y solo caben ciertas \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} --- igual que una cuerda
de guitarra, un \omterm{def:g11:fundamental-interactions:ladder}{átomo} tiene un conjunto discreto de notas.
\end{remark}

\begin{theorem}[Fotón de una transición]\label{thm:g12:quantum-world:transition}
Un \omterm{def:g11:fundamental-interactions:ladder}{átomo} cambia de nivel únicamente emitiendo o absorbiendo un \omterm{def:g12:quantum-world:photon}{fotón} que se lleve
exactamente la diferencia: al caer del nivel $p$ a otro más bajo $n$ emite
$h\nu = E_p - E_n$; al subir de $n$ a $p$ absorbe esa misma \omterm{def:g10:energy-conservation:energy}{energía}.
\end{theorem}

\begin{proof}
\omterm{thm:g10:energy-conservation:conservation}{Conservación de la energía}, grano a grano: el \omterm{def:g12:quantum-world:photon}{fotón} es la única moneda que acepta la
luz, y las cuentas del \omterm{def:g11:fundamental-interactions:ladder}{átomo} tienen que cuadrar.
\end{proof}

\begin{remark}[Los espectros de líneas, explicados]\label{rem:g12:quantum-world:lines}
Aquí está la respuesta que prometía el \cref{ch:g10:light-spectra}: un gas caliente
emite solo las \omterm{def:g12:oscillators-and-time:oscillator}{frecuencias} $(E_p - E_n)/h$ de su propia escalera --- un código de
barras de líneas brillantes --- y un gas frío roba exactamente esas \omterm{def:g12:oscillators-and-time:oscillator}{frecuencias} a la
\omterm{def:g10:light-spectra:white}{luz blanca}, imprimiendo las líneas oscuras de absorción correspondientes. Cada
elemento tiene su propia escalera y, por tanto, su propio código de barras --- así es
como leemos la luz de las estrellas.
\end{remark}

\begin{example}[Las líneas visibles del hidrógeno]\label{ex:g12:quantum-world:balmer}
Las transiciones que terminan en $n = 2$ caen en el visible. Con
$E \,(\unit{eV}) = 1240/\lambda\,(\unit{nm})$: $3 \to 2$: \qty{1.89}{eV},
\qty{656}{nm} (rojo); $4 \to 2$: \qty{2.55}{eV}, \qty{486}{nm} (azul verdoso);
$5 \to 2$: \qty{2.86}{eV}, \qty{434}{nm} (violeta). Las transiciones a $n = 1$ superan
los \qty{10}{eV}: todas \omterm{def:g10:light-spectra:wavelength}{ultravioletas}.
\end{example}

\begin{remark}[Los láseres, en un suspiro]\label{rem:g12:quantum-world:laser}
Un \omterm{def:g12:quantum-world:photon}{fotón} de \omterm{def:g10:energy-conservation:energy}{energía} exactamente $E_p - E_n$ que pase junto a un \omterm{def:g11:fundamental-interactions:ladder}{átomo} \emph{excitado}
puede provocar que este emita un \omterm{def:g12:quantum-world:photon}{fotón} idéntico --- misma \omterm{def:g10:energy-conservation:energy}{energía}, misma dirección, al
paso. Mantén más \omterm{def:g11:fundamental-interactions:ladder}{átomos} arriba que abajo y un \omterm{def:g12:quantum-world:photon}{fotón} se convierte en una avalancha de
clones: la \emph{emisión estimulada}\index{emisión estimulada} --- y un \omterm{def:g11:color-light-sources:lamps}{láser} es poco
más que eso.
\end{remark}

\section{Ondas de materia}

\begin{proposition}[Los electrones interfieren]\label{prop:g12:quantum-world:interference}
Envía electrones de uno en uno a través de una doble rendija
(\cref{ch:g12:light-as-wave}): cada uno aterriza como un único punto nítido --- una
partícula. Pero, a medida que se acumulan miles, los puntos dibujan \omterm{def:g12:light-as-wave:fringes}{franjas de
interferencia}: cada electrón, él solo, pasa como una \omterm{def:g10:signals-and-waves:wave}{onda} por \emph{las dos}
rendijas. Los cristales también difractan haces de electrones (Davisson y Germer,
1927).
\end{proposition}
\admitted

\begin{definition}[Longitud de onda de De Broglie]\label{def:g12:quantum-world:debroglie}
Toda partícula de \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento} $p = mv$ viaja como una \emph{onda de
materia}\index{onda de materia} de \emph{longitud de onda de De
Broglie}\index{longitud de onda de De Broglie}
\[
\lambda = \frac{h}{p} = \frac{h}{mv}.
\]
\end{definition}
\admitted

\begin{example}[Por qué tú nunca difractas]\label{ex:g12:quantum-world:never}
Un electrón acelerado con \qty{100}{V} tiene $E_k = \qty{1.60e-17}{J}$ y
$p = \sqrt{2m_e E_k} = \qty{5.4e-24}{kg.m/s}$, luego
$\lambda \approx \qty{0.12}{nm}$ --- la separación de los \omterm{def:g11:fundamental-interactions:ladder}{átomos} en un cristal, que
por eso actúa como su doble rendija. Una pelota de tenis de \qty{58}{g} a
\qty{25}{m/s}: $\lambda = 6.63\times10^{-34}/1.45 \approx \qty{4.6e-34}{m}$, unas
$10^{19}$ veces menor que un \omterm{def:g11:fundamental-interactions:ladder}{núcleo} --- ninguna rendija la mostrará jamás formando
franjas. $h$ es tan pequeña que la granularidad cuántica solo asoma a la escala del
\omterm{def:g11:fundamental-interactions:ladder}{átomo}.
\end{example}

\begin{example}[El microscopio electrónico]\label{ex:g12:quantum-world:microscope}
La \omterm{def:g12:light-as-wave:diffraction}{difracción} prohíbe resolver detalles mucho más pequeños que la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}
empleada (\cref{ch:g12:light-as-wave}): la luz visible se detiene cerca de
\qty{0.5}{\micro m}. Los electrones a \qty{10}{kV} tienen
$\lambda \approx \qty{12}{pm}$ --- decenas de miles de veces más corta: los
microscopios electrónicos fotografían virus, la maquinaria celular e incluso \omterm{def:g11:fundamental-interactions:ladder}{átomos}
sueltos. La \omterm{def:g12:quantum-world:debroglie}{onda de materia} le regaló a la física un ojo nuevo.
\end{example}

\section{Problema: El laboratorio de física del panel solar}

\begin{problem}[{Problema de fin de semana --- el laboratorio de física del panel
solar: los $10^{21}$ granos que se bebe un tejado cada segundo, una célula
fotoeléctrica que mide la \omterm{def:g12:quantum-world:photon}{constante de Planck}, una lámpara de hidrógeno leída línea a
línea --- un fin de semana, una $h$}]
\label{pb:g12:quantum-world:1}
Tu clase instrumenta durante un fin de semana el panel solar del tejado del instituto.
Datos: irradiancia solar al mediodía, \qty{1.0e3}{W/m^2}; \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} solar media,
\qty{550}{nm}; superficie del panel, \qty{1.0}{m^2}, con un \omterm{def:g10:energy-conservation:efficiency}{rendimiento} del $20\,\%$;
el día entrega el equivalente de \qty{5.0}{h} de sol pleno.

\medskip
\noindent\textbf{Parte I --- Contabilidad de \omterm{def:g12:quantum-world:photon}{fotones}.}
\begin{enumerate}
  \item Calcula la \omterm{def:g10:energy-conservation:energy}{energía} de un \omterm{def:g12:quantum-world:photon}{fotón} solar medio, en \omterm{def:g11:work-of-force:work}{julios} y en \omterm{def:g12:nuclear-energy:units}{electronvoltios}.
  \item Deduce el número de \omterm{def:g12:quantum-world:photon}{fotones} que golpean el panel por segundo al mediodía.
  \item Calcula la \omterm{prop:g11:circuits-and-power:power}{potencia eléctrica} del panel al mediodía y después la \omterm{def:g10:energy-conservation:energy}{energía}
        producida a lo largo del día, en \unit{kW.h}.
  \item La corriente del panel parece perfectamente lisa. Usando la pregunta 2,
        explica por qué la llegada grano a grano es invisible.
  \item La luz del Sol transporta más \omterm{def:g10:energy-conservation:energy}{energía} infrarroja que \omterm{def:g10:light-spectra:wavelength}{ultravioleta} y, sin
        embargo, solo la \omterm{def:g10:light-spectra:wavelength}{ultravioleta} quema la piel (enlaces de
        $\sim \qtyrange{3}{4}{eV}$). Explícalo en términos de granos.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Medir $h$ con la célula del equipo.}
La célula fotoeléctrica del equipo tiene un cátodo desconocido. Los \omterm{def:g11:color-light-sources:subtractive}{filtros} dan:
\begin{center}\small
\begin{tabular}{lcccc}
$\lambda$ (\unit{nm}) & 365 & 405 & 436 & 546\\
$U_s$ (\unit{V}) & 1.50 & 1.16 & 0.94 & 0.37
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item ¿Qué mide el \omterm{def:g12:quantum-world:stopping}{potencial de frenado}? Relaciona $U_s$, $\nu$, $W_0$ y $h$.
  \item Convierte las cuatro \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} en \omterm{def:g12:oscillators-and-time:oscillator}{frecuencias} (en
        $10^{14}\,\unit{Hz}$).
  \item ¿Por qué debería salir una recta al representar $eU_s$ frente a $\nu$?
        Comprueba la alineación de los puntos.
  \item A partir de los puntos extremos, calcula la pendiente: tu $h$ medida.
  \item Deduce el \omterm{thm:g12:quantum-world:einstein}{trabajo de extracción} del cátodo (en \unit{eV}), su \omterm{prop:g12:quantum-world:facts}{frecuencia
        umbral} y su \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} umbral.
  \item ¿Qué parte de la luz solar podría arrancar electrones de este cátodo --- y qué
        mitad del \omterm{def:g12:sound-acoustics:timbre}{espectro} le resulta inútil?
  \item Compara tu $h$ con \qty{6.63e-34}{J.s}: ¿error relativo en porcentaje?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La lámpara de calibración de hidrógeno.}
La escala de \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} del equipo se comprueba con una lámpara de hidrógeno,
$E_n = -13.6/n^2$ \unit{eV}.
\begin{enumerate}[resume]
  \item Calcula $E_2$, $E_3$, $E_4$ y $E_5$.
  \item Calcula las \omterm{def:g10:energy-conservation:energy}{energías} de los \omterm{def:g12:quantum-world:photon}{fotones} de las transiciones $3 \to 2$, $4 \to 2$ y
        $5 \to 2$.
  \item Deduce las tres \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} y da el color de cada línea.
  \item ¿Por qué emite la lámpara \emph{líneas} y no un arcoíris continuo? En una
        frase, nombrando la idea clave.
  \item ¿Podría el hidrógeno de la lámpara absorber luz de \qty{500}{nm}? Justifícalo
        con la escalera.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Inspeccionar las células: la \omterm{def:g10:signals-and-waves:wave}{onda} del electrón.}
Las microgrietas de una célula quedan muy por debajo del \omterm{def:g12:projectile-motion:range}{alcance} de \qty{0.5}{\micro m}
de los microscopios ópticos; el microscopio electrónico de barrido del laboratorio
funciona a \qty{10}{kV}.
\begin{enumerate}[resume]
  \item Calcula la rapidez de los electrones; comprueba que se mantiene por debajo de
        $0.25\,c$ (fórmula clásica).
  \item Calcula su \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento} y su \omterm{def:g12:quantum-world:debroglie}{longitud de onda de De Broglie}.
  \item La resolución escala con la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}: calcula la ganancia respecto de
        la luz verde (\qty{550}{nm}); concluye con las dos cifras del fin de semana ---
        tu $h$ y esta ganancia.
\end{enumerate}
\end{problem}
