\chapter{Cinemática en dos dimensiones}\label{ch:g12:kinematics-2d}

Observa un coche que toma una rotonda a \qty{30}{km/h} constantes: la
aguja no se mueve y, sin embargo, todos los ocupantes notan un tirón
lateral --- el movimiento está cambiando, y no es la \omterm{def:g12:kinematics-2d:velocity}{rapidez}. Un número
por instante ya no basta: el movimiento en el plano necesita flechas.
Este capítulo equipa a un punto móvil con tres --- posición, velocidad y
aceleración --- encadenadas por la derivada, y lee familias enteras de
movimientos en su geometría.

\section{El vector de posición}

El movimiento solo existe respecto de un \omterm{def:g10:relative-motion:frame}{sistema de referencia} declarado
--- un cuerpo rígido más un reloj (\cref{ch:g10:relative-motion}). La
novedad de este año es la precisión: se le añaden al sistema un origen
$O$ y dos ejes graduados perpendiculares $Ox$ y $Oy$, y se registra
dónde está el punto en cada instante, como una pareja de funciones.

\begin{definition}[Vector de posición]\label{def:g12:kinematics-2d:position}
En un sistema equipado con un origen $O$ y unos ejes $Ox$ y $Oy$, el
\emph{vector de posición}\index{vector de posición} de un punto móvil
$M$ en el instante $t$ es $\vect{OM}(t)$, de coordenadas
$\bigl(x(t),\, y(t)\bigr)$: dos funciones del tiempo, una por eje. La
curva que traza $M$ mientras $t$ corre es su \omterm{def:g10:relative-motion:trajectory}{trayectoria}.
\end{definition}

\section{El vector velocidad}

\begin{definition}[Vector velocidad]\label{def:g12:kinematics-2d:velocity}
Durante un intervalo corto, el punto se desplaza
$\vect{OM}(t+\Delta t) - \vect{OM}(t)$; se divide entre $\Delta t$ y se
deja encoger el intervalo: esta es la derivada de tu curso de
matemáticas, aplicada a cada coordenada. El \emph{vector
velocidad}\index{vector velocidad} de $M$ es la derivada de su \omterm{def:g12:kinematics-2d:position}{vector de
posición}, $\vect v(t) = \bigl(x'(t),\, y'(t)\bigr)$, en \unit{m/s}. Su
norma $v = \sqrt{x'^2 + y'^2}$ es la \emph{rapidez}\index{rapidez} ---
el único número que muestra el velocímetro.
\end{definition}

\begin{example}[Un dron en vuelo horizontal]\label{ex:g12:kinematics-2d:drone}
Un dron vuela con $x(t) = 4.0\,t$ e $y(t) = 3.0\,t$ (\omterm{def:g10:orders-of-magnitude:unit}{metros} y
segundos): \omterm{def:g10:relative-motion:trajectory}{trayectoria} la recta $y = \tfrac34 x$, velocidad
$\vect v = (4.0, 3.0)$ en todo instante y \omterm{def:g12:kinematics-2d:velocity}{rapidez}
$\sqrt{4.0^2 + 3.0^2} = \qty{5.0}{m/s}$, constante.
\end{example}

\begin{proposition}[Tangente a la trayectoria]\label{prop:g12:kinematics-2d:tangent}
En todo instante, $\vect v(t)$ es tangente a la \omterm{def:g10:relative-motion:trajectory}{trayectoria} en la
posición actual de $M$ y apunta en el sentido de la marcha.
\end{proposition}

\begin{proof}
El desplazamiento es una cuerda de la \omterm{def:g10:relative-motion:trajectory}{trayectoria}; a medida que
$\Delta t$ se encoge, la dirección de la cuerda se convierte en la de la
tangente.
\end{proof}

\section{El vector aceleración}

\begin{definition}[Vector aceleración]\label{def:g12:kinematics-2d:acceleration}
El \emph{vector aceleración}\index{vector aceleración} de $M$ es la
derivada de su \omterm{def:g12:kinematics-2d:velocity}{vector velocidad},
$\vect a(t) = \bigl(v_x'(t),\, v_y'(t)\bigr) =
\bigl(x''(t),\, y''(t)\bigr)$, en \unit{m/s^2}: con qué \omterm{def:g12:kinematics-2d:velocity}{rapidez}, y en
qué dirección, está cambiando el \omterm{def:g12:kinematics-2d:velocity}{vector velocidad}.
\end{definition}

\begin{remark}[Acelerar sin ir más deprisa]\label{rem:g12:kinematics-2d:turning}
$\vect a \neq \vect 0$ siempre que cambie el \emph{vector} velocidad ---
en longitud \emph{o en dirección}: un coche que frena acelera (hacia
atrás); un coche que gira a \omterm{def:g12:kinematics-2d:velocity}{rapidez} constante acelera (de lado). Esta es
la magnitud que gobiernan las \omterm{def:g10:inertia:force}{fuerzas}
(\cref{ch:g11:forces-and-motion,ch:g12:newtons-laws}).
\end{remark}

\begin{method}[Leer una cronofotografía]\label{met:g12:kinematics-2d:chrono}
Una \emph{cronofotografía}\index{cronofotografía} registra las
posiciones $M_0, M_1, M_2, \dots$ de un punto móvil a intervalos de
tiempo iguales $\Delta t$.
\begin{enumerate}
  \item La separación entre puntos da la \omterm{def:g12:kinematics-2d:velocity}{rapidez}: separaciones iguales,
        \omterm{def:g12:kinematics-2d:velocity}{rapidez} constante; separaciones que se abren, aceleración; que
        se cierran, frenada.
  \item La velocidad en $M_i$ es la cuerda simétrica,
        $\vect v_i \approx \vect{M_{i-1}M_{i+1}}/(2\,\Delta t)$,
        dibujada en $M_i$ y tangente al camino punteado.
  \item Para la aceleración en $M_i$: se copian $\vect v_{i-1}$ y
        $\vect v_{i+1}$ en $M_i$, se dibuja
        $\Delta\vect v = \vect v_{i+1} - \vect v_{i-1}$ de punta a
        punta; entonces $\vect a_i \approx \Delta\vect v/(2\,\Delta t)$.
\end{enumerate}
\end{method}

\section{Los movimientos rectilíneos}

A lo largo de una recta basta un eje: $x(t)$, $v(t) = x'(t)$,
$a(t) = v'(t)$, todos números con signo.

\begin{definition}[Movimiento rectilíneo uniforme]\label{def:g12:kinematics-2d:uniform}
Un movimiento es \emph{rectilíneo uniforme}\index{movimiento rectilíneo
uniforme} cuando $\vect v$ es un vector constante: \omterm{def:g10:relative-motion:trajectory}{trayectoria} recta,
\omterm{def:g12:kinematics-2d:velocity}{rapidez} constante y $\vect a = \vect 0$. Entonces
$x(t) = v\,t + x_0$.
\end{definition}

\begin{proposition}[Movimiento rectilíneo uniformemente acelerado]\label{prop:g12:kinematics-2d:uarm}
\index{movimiento uniformemente acelerado}%
Si un punto se mueve a lo largo de $Ox$ con aceleración $a$
\emph{constante}, partiendo en $t = 0$ de la posición $x_0$ con
velocidad $v_0$, entonces
\[
v(t) = a\,t + v_0,
\qquad
x(t) = \tfrac12\,a\,t^2 + v_0\,t + x_0 .
\]
\end{proposition}

\begin{proof}
$v$ es una primitiva de la constante $a$: $v(t) = a t + C$, y $t = 0$
obliga a $C = v_0$. A su vez, $x$ es una primitiva de $a t + v_0$:
$x(t) = \tfrac12 a t^2 + v_0 t + C'$, con $C' = x_0$.
\end{proof}

\begin{example}[Distancia de frenado]\label{ex:g12:kinematics-2d:braking}
Un coche que va a la \omterm{def:g12:kinematics-2d:velocity}{rapidez} $v_0$ frena con deceleración constante $a$
(aceleración $-a$). Se detiene cuando $v(t) = -a t + v_0 = 0$, en
$t_s = v_0/a$, tras recorrer
$d = -\tfrac12 a t_s^2 + v_0 t_s = v_0^2/(2a)$. El cuadrado es el
titular de la seguridad vial: duplicar la \omterm{def:g12:kinematics-2d:velocity}{rapidez} \emph{cuadruplica} la
distancia de frenado --- con $a = \qty{6.0}{m/s^2}$, \qty{16}{m} desde
\qty{50}{km/h} y \qty{64}{m} desde \qty{100}{km/h}.
\end{example}

\section{El movimiento circular uniforme y la base de Frenet}

\begin{definition}[Movimiento circular uniforme]\label{def:g12:kinematics-2d:ucm}
Un movimiento es \emph{circular uniforme}\index{movimiento circular
uniforme} cuando la \omterm{def:g10:relative-motion:trajectory}{trayectoria} es una circunferencia de radio $R$ y la
\omterm{def:g12:kinematics-2d:velocity}{rapidez} $v$ es constante. El \emph{periodo}\index{periodo!de rotación}
$T$ es la duración de una vuelta, y la
\emph{frecuencia}\index{frecuencia!de rotación} $f = 1/T$ (en
\unit{Hz}), el número de vueltas por segundo:
$v = 2\pi R/T = 2\pi R f$.
\end{definition}

\omterm{def:g12:kinematics-2d:velocity}{Rapidez} constante --- y, sin embargo, la velocidad gira: \emph{hay}
aceleración. ¿Hacia dónde y de qué tamaño?

\begin{theorem}[Aceleración centrípeta]\label{thm:g12:kinematics-2d:centripetal}
\index{aceleración centrípeta}%
En un \omterm{def:g12:kinematics-2d:ucm}{movimiento circular uniforme} de radio $R$ y \omterm{def:g12:kinematics-2d:velocity}{rapidez} $v$, la
aceleración --- llamada \emph{centrípeta} --- apunta desde el punto
móvil hacia el centro de la circunferencia, con norma constante
\[
a = \frac{v^2}{R}.
\]
\end{theorem}

\begin{proof}
Se centran los ejes en el centro de la circunferencia y se pone
$\omega = v/R$, de modo que el arco $v t$ subtiende el ángulo
$\omega t$: la posición es $x(t) = R\cos(\omega t)$,
$y(t) = R\sin(\omega t)$. Derivando,
$\vect v = (-R\omega\sin(\omega t),\, R\omega\cos(\omega t))$, de norma
constante $R\omega = v$; derivando otra vez,
$\vect a = -\omega^2\,\vect{OM}$: opuesta al \omterm{def:g12:kinematics-2d:position}{vector de posición} ---
hacia el centro --- y de norma $\omega^2 R = v^2/R$.
\end{proof}

\begin{definition}[Base de Frenet]\label{def:g12:kinematics-2d:frenet}
La \emph{base de Frenet}\index{base de Frenet} en la posición actual de
un punto que recorre un camino curvo es la pareja de vectores unitarios
perpendiculares $(\vect{u_t}, \vect{u_n})$: $\vect{u_t}$ tangente, en el
sentido del movimiento; $\vect{u_n}$ normal, hacia el centro de la
circunferencia de radio $R$ que mejor se ajusta allí a la curva.
\end{definition}

\begin{proposition}[La aceleración en la base de Frenet]\label{prop:g12:kinematics-2d:frenetacc}
A lo largo de cualquier camino plano de radio local $R$, recorrido con
\omterm{def:g12:kinematics-2d:velocity}{rapidez} $v(t)$,
\[
\vect a = a_t\,\vect{u_t} + a_n\,\vect{u_n},
\qquad a_t = \frac{\dd v}{\dd t},
\qquad a_n = \frac{v^2}{R}:
\]
la \emph{aceleración tangencial}\index{aceleración tangencial} $a_t$
cambia la \omterm{def:g12:kinematics-2d:velocity}{rapidez}; la \emph{aceleración normal}\index{aceleración
normal} $a_n$ gira la velocidad.
\end{proposition}
\admitted

\begin{remark}[Dónde vive la demostración]\label{rem:g12:kinematics-2d:frenetproof}
Hemos demostrado los extremos: $a_t$ pura sobre una recta
(\cref{prop:g12:kinematics-2d:uarm}) y $a_n$ pura sobre una
circunferencia (\cref{thm:g12:kinematics-2d:centripetal}); la
descomposición general se deduce en el volumen del primer año.
\end{remark}

\begin{example}[Frenar al entrar en una curva]\label{ex:g12:kinematics-2d:bend}
Un coche entra en una curva de radio $R = \qty{50}{m}$ a
$v = \qty{15}{m/s}$ mientras frena con
$\dd v/\dd t = \qty{-3.0}{m/s^2}$: $a_t = \qty{-3.0}{m/s^2}$,
$a_n = 15^2/50 = \qty{4.5}{m/s^2}$, luego
$a = \sqrt{3.0^2 + 4.5^2} \approx \qty{5.4}{m/s^2}$, apuntando hacia
atrás \emph{y} hacia el interior de la curva. Los neumáticos tienen que
dar las dos cosas a la vez --- de ahí que los pilotos de carreras frenen
\emph{antes} de girar.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:1}
Para $x(t) = 3.0\,t$ e $y(t) = 4.0\,t$ (\omterm{def:g10:orders-of-magnitude:unit}{metros} y segundos), da
$\vect v$, la \omterm{def:g12:kinematics-2d:velocity}{rapidez}, la \omterm{def:g10:relative-motion:trajectory}{trayectoria} y la clase de movimiento.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:2}
En una \omterm{met:g12:kinematics-2d:chrono}{cronofotografía} tomada cada \qty{0.10}{s}, los puntos están en
línea recta y separados regularmente \qty{0.35}{m}. Identifica el
movimiento y calcula la \omterm{def:g12:kinematics-2d:velocity}{rapidez}. ¿Qué indicarían unas separaciones que
se abren?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:3}
Para $x(t) = 2.0\,t^2$ a lo largo de $Ox$ (\omterm{def:g10:orders-of-magnitude:unit}{metros} y segundos), deriva
$v(t)$ y $a(t)$, nombra la clase de movimiento y calcula la \omterm{def:g12:kinematics-2d:velocity}{rapidez} en
$t = \qty{3.0}{s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:4}
Un caballito de tiovivo recorre una circunferencia de \qty{4.5}{m} de
radio, una vuelta cada \qty{8.0}{s}: calcula la \omterm{def:g12:kinematics-2d:ucm}{frecuencia}, la \omterm{def:g12:kinematics-2d:velocity}{rapidez} y
la aceleración (norma y dirección).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:5}
Una gráfica de $v(t)$ sube en línea recta de \qty{2.0}{m/s} en $t = 0$ a
\qty{10.0}{m/s} en $t = \qty{4.0}{s}$. Lee la aceleración y después la
distancia recorrida (el área bajo la gráfica).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:6}
Un tren de alta velocidad toma una curva de \qty{6.0}{km} de radio a
\qty{320}{km/h}. Calcula $a_n$ y compárala con $g$; ¿por qué tienen que
evitar las líneas de alta velocidad las curvas cerradas, aunque estén
perfectamente peraltadas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:7}
Una vagoneta sobre un raíl recto tiene
$x(t) = -1.2\,t^2 + 8.0\,t + 3.0$ (\omterm{def:g10:orders-of-magnitude:unit}{metros} y segundos). Halla $v(t)$ y
$a$, el instante y la posición en que da media vuelta, y describe el
movimiento antes y después.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:8}
Una \omterm{met:g12:kinematics-2d:chrono}{cronofotografía} a lo largo de una recta, cada
$\Delta t = \qty{0.20}{s}$, da $x = 0$, $0.10$, $0.28$, $0.54$ y
$\qty{0.88}{m}$. Estima la velocidad en los tres puntos interiores
(\cref{met:g12:kinematics-2d:chrono}) y después la aceleración;
concluye.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:9}
La Luna orbita la Tierra sobre una curva casi circular de radio
\qty{3.84e8}{m} en \qty{27.3}{d}. Calcula su \omterm{def:g12:kinematics-2d:velocity}{rapidez} y su \omterm{thm:g12:kinematics-2d:centripetal}{aceleración
centrípeta}; compara esta última con $g$, sabiendo que la Luna está a

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:10}
El tambor de una lavadora de \qty{25}{cm} de radio gira a $1200$

aceleración del borde en múltiplos de $g$. ¿Por qué se queda la ropa
pegada mientras el agua se va?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:11}
Un coche a \qty{90}{km/h} frena con $a = \qty{7.5}{m/s^2}$ constante:
calcula el tiempo de parada y la distancia de frenado, y esboza $v(t)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:12}
Para $x(t) = R\cos(\omega t)$ e $y(t) = R\sin(\omega t)$: calcula
$\vect v$; comprueba que la \omterm{def:g12:kinematics-2d:velocity}{rapidez} es la constante $R\omega$ y que
$\vect v \perp \vect{OM}$; calcula $\vect a$, demuestra que vale
$-\omega^2\,\vect{OM}$ y recupera $a = v^2/R$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:13}
Una motocicleta entra en una curva de \qty{80}{m} de radio a

norma de $\vect a$ y su ángulo con la dirección del movimiento. Los
neumáticos agarran hasta \qty{7.0}{m/s^2} en total: ¿está el piloto

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:14}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:15}

respecto del centro de la Tierra y un \omterm{def:g12:kinematics-2d:ucm}{periodo} de \qty{86164}{s}.
Calcula su \omterm{def:g12:kinematics-2d:velocity}{rapidez} y su \omterm{thm:g12:kinematics-2d:centripetal}{aceleración centrípeta}. ¿Qué se la
proporciona, y por qué queda el \omterm{def:g10:universal-gravitation:satellite}{satélite} colgado sobre un punto fijo del
ecuador (\cref{ch:g12:satellites-kepler})?
\end{exercise}

\section{Problema: el informe del accidente}

\begin{problem}[{Problema de fin de semana --- el informe del accidente:
la película de una cámara de salpicadero leída fotograma a fotograma,
una ley de frenado reconstruida a partir de las marcas de derrape, una
rotonda que declara sobre el agarre y un veredicto en kilómetros por
hora}]
\label{pb:g12:kinematics-2d:1}
Un coche golpea a una furgoneta que salía de una calle lateral; no hay
heridos, pero las aseguradoras no se ponen de acuerdo. El investigador
dispone de imágenes de una cámara cenital (un fotograma cada
\qty{0.50}{s}), de las marcas de derrape medidas y de la cámara de
salpicadero del coche. El límite de velocidad es \qty{50}{km/h}.

\medskip
\noindent\textbf{Parte I --- Leer la película.}
Los fotogramas anteriores a cualquier reacción dan, a lo largo de la
recta:
\begin{center}\small
\begin{tabular}{lccccc}
$t$ (\unit{s}) & 0.00 & 0.50 & 1.00 & 1.50 & 2.00\\
$x$ (\unit{m}) & 0.0 & 9.0 & 18.0 & 27.0 & 36.0
\end{tabular}
\end{center}
\begin{enumerate}
  \item ¿Por qué tiene que empezar el informe nombrando un sistema y
        unos ejes? ¿Cuáles se usan aquí?
  \item Calcula la \omterm{def:g10:relative-motion:speed}{velocidad media} en cada uno de los cuatro
        intervalos.
  \item ¿Qué clase de movimiento es? Justifícalo con la tabla.
  \item Da la \omterm{def:g12:kinematics-2d:velocity}{rapidez} $v_0$ del coche en \unit{m/s} y en \unit{km/h}.
  \item ¿Cuál es la aceleración durante estos dos segundos?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las marcas de derrape.}
Los ensayos de neumáticos sobre este asfalto seco dan una deceleración
de frenado constante $a = \qty{6.0}{m/s^2}$; las marcas de derrape miden
$d_b = \qty{27}{m}$. Toma $t = 0$ y $x = 0$ al empezar la frenada, con
\omterm{def:g12:kinematics-2d:velocity}{rapidez} desconocida $v_B$.
\begin{enumerate}[resume]
  \item Con primitivas, escribe $v(t)$ y $x(t)$ durante la frenada.
  \item Expresa el tiempo de parada $t_s$; demuestra que
        $d_b = v_B^2/(2a)$.
  \item Deduce $v_B$ a partir de las marcas de derrape. ¿Es coherente
        con la Parte~I?
  \item El tiempo de reacción es de \qty{1.0}{s}: calcula la distancia
        de reacción y la distancia total de parada.
  \item Explica, con la pregunta 7, por qué duplicar la \omterm{def:g12:kinematics-2d:velocity}{rapidez}
        cuadruplica la distancia de frenado; calcula $d_b$ a exactamente
        \qty{50}{km/h}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La rotonda declara.}
Antes, la cámara de salpicadero muestra al coche recorriendo el carril
de una rotonda de radio $R = \qty{12.5}{m}$ a \omterm{def:g12:kinematics-2d:velocity}{rapidez} constante, un
cuarto de vuelta en \qty{2.5}{s}. La calzada puede dar como mucho
$a_n = \mu g$, con $\mu = 0.70$.
\begin{enumerate}[resume]
  \item Allí la \omterm{def:g12:kinematics-2d:velocity}{rapidez} es constante --- ¿lo es la velocidad? ¿Está
        acelerando el coche? Explícalo.
  \item Calcula la \omterm{def:g12:kinematics-2d:velocity}{rapidez} del coche en la rotonda.
  \item Calcula su \omterm{thm:g12:kinematics-2d:centripetal}{aceleración centrípeta} (norma y dirección).
  \item Calcula la \omterm{def:g12:kinematics-2d:velocity}{rapidez} máxima sin derrape. ¿Estaba el coche por
        debajo?
  \item Da el \omterm{def:g12:kinematics-2d:ucm}{periodo} y la \omterm{def:g12:kinematics-2d:ucm}{frecuencia} de una vuelta completa a esa
        \omterm{def:g12:kinematics-2d:velocity}{rapidez}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El veredicto.}
La furgoneta apareció $D = \qty{41}{m}$ por delante del coche en el
instante en que el conductor empezó a reaccionar.
\begin{enumerate}[resume]
  \item ¿En cuánto, en \unit{km/h} y en porcentaje, superaba el límite
        la \omterm{def:g12:kinematics-2d:velocity}{rapidez} anterior a la frenada?
  \item Calcula la \omterm{def:g12:kinematics-2d:velocity}{rapidez} del coche en el momento del impacto.
  \item Rehaz el cálculo de la parada a exactamente \qty{50}{km/h} (con
        la misma reacción y la misma frenada): ¿se detiene el coche, y
        con qué margen?
  \item Da la fórmula general de la distancia de parada $d(v)$; ¿por qué
        entra la \omterm{def:g12:kinematics-2d:velocity}{rapidez} dos veces --- una linealmente y otra al
        cuadrado?
  \item El veredicto, en una frase con sus números: ¿a qué \omterm{def:g12:kinematics-2d:velocity}{rapidez} iba
        el conductor, y habría ocurrido el accidente yendo al límite?
\end{enumerate}
\end{problem}
