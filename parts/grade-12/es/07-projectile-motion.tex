\chapter{Caída libre y tiro parabólico}\label{ch:g12:projectile-motion}

Todo chorro de una manguera de jardín dibuja la misma curva en el aire. Y también un
balón de baloncesto que se arquea hacia la canasta, una saltadora de longitud, una
bala de cañón: cualquier cuerpo abandonado a su \omterm{def:g10:universal-gravitation:weight}{peso}. Este capítulo se gana esa curva
--- una parábola --- a partir de la \omterm{thm:g12:newtons-laws:second}{segunda ley de Newton} en dos líneas de cálculo,
lee en su ecuación la cima, el \omterm{def:g12:projectile-motion:range}{tiempo de vuelo} y el \omterm{def:g12:projectile-motion:range}{alcance}, y después deja entrar de
nuevo el aire para ver qué estropea.

\section{La caída libre}

\begin{definition}[Caída libre]\label{def:g12:projectile-motion:freefall}
Un cuerpo está en \emph{caída libre}\index{caída libre} cuando la única \omterm{def:g10:inertia:force}{fuerza} que
actúa sobre él es su \omterm{def:g10:universal-gravitation:weight}{peso} $\vect P = m\vect g$: sin soporte, sin hilo y con la
resistencia del aire despreciable. Una piedra que se suelta o una pelota que ya ha
dejado la mano están, con buena aproximación, en caída libre --- y ``caída'' incluye
la subida.
\end{definition}

\begin{theorem}[Ley de caída libre de Galileo]\label{thm:g12:projectile-motion:galileo}
\index{ley de caída libre de Galileo}
En \omterm{def:g12:projectile-motion:freefall}{caída libre}, todos los cuerpos comparten la misma aceleración, sea cual sea su
masa:
\[
\vect a = \vect g, \qquad g = \qty{9.81}{m/s^2},
\]
dirigida verticalmente hacia abajo: la intensidad del \omterm{def:g11:electric-gravitational-fields:field}{campo} de \qty{9.81}{N/kg} del
\cref{ch:g10:universal-gravitation} con su otro traje,
$\unit{N/kg} = \unit{m/s^2}$.
\end{theorem}

\begin{proof}
La \omterm{thm:g12:newtons-laws:second}{segunda ley de Newton} (\cref{ch:g12:newtons-laws}) con el \omterm{def:g10:universal-gravitation:weight}{peso} como única \omterm{def:g10:inertia:force}{fuerza}:
$m\vect a = m\vect g$, y la masa se cancela. El martillo y la pluma solo se
diferencian por el aire.
\end{proof}

\begin{proposition}[Caída vertical desde el reposo]\label{prop:g12:projectile-motion:vertical}
Soltado desde el reposo, y contando hacia abajo la profundidad de caída $h$ desde el
punto de suelta, un cuerpo en \omterm{def:g12:projectile-motion:freefall}{caída libre} obedece
\[
a = g, \qquad v = g t, \qquad h = \tfrac12\, g t^2 .
\]
\end{proposition}

\begin{proof}
$v(t)$ es la primitiva de la constante $a = g$ que se anula en $t = 0$
(\cref{ch:g12:kinematics-2d}); $h(t)$, la de $v = gt$ que se anula en $t = 0$.
\end{proof}

\begin{example}[Órdenes de magnitud]\label{ex:g12:projectile-motion:orders}
Tras $1$, $2$ y $3$ segundos de \omterm{def:g12:projectile-motion:freefall}{caída libre}: $v = 9.8$, $19.6$, \qty{29.4}{m/s} ($35$,
$71$, \qty{106}{km/h}) y $h = 4.9$, $19.6$, \qty{44.1}{m}. Desde una azotea de
\qty{30}{m}: $t = \sqrt{2h/g} \approx \qty{2.5}{s}$, y se llega a \qty{24}{m/s}
$\approx \qty{87}{km/h}$. Los tiempos crecen como $\sqrt h$ --- y pasados unos pocos
segundos el aire se niega a seguir siendo despreciable (última sección).
\end{example}

\begin{example}[Lanzamiento vertical]\label{ex:g12:projectile-motion:throw}
Lanzado verticalmente hacia arriba con $v_0$, con el eje $Oy$ hacia arriba desde la
mano: las mismas dos primitivas dan $v = v_0 - gt$ e $y = v_0 t - \tfrac12 gt^2$. La
cima, donde $v = 0$, llega en $t = v_0/g$, a la altura $h = v_0^2/2g$ --- la respuesta
energética $v_0 = \sqrt{2gh}$ del \cref{ch:g11:mechanical-energy} recorrida al revés;
lanzado hacia \emph{abajo}, $v_0$ cambia de signo. Para \qty{15}{m/s}: cima a
\qty{11.5}{m} en \qty{1.53}{s}, de vuelta a \qty{3.06}{s} y a \qty{15}{m/s}.
\end{example}

\section{El proyectil: una ley, dos primitivas}

\begin{definition}[Proyectil]\label{def:g12:projectile-motion:projectile}
Un \emph{proyectil}\index{proyectil} es un cuerpo lanzado con una velocidad $\vect v_0$
y abandonado después a la \omterm{def:g12:projectile-motion:freefall}{caída libre}. Ejes en el punto de lanzamiento: $Ox$
horizontal en el plano vertical de $\vect v_0$, $Oy$ vertical hacia arriba; el
\emph{ángulo de tiro}\index{ángulo de tiro} $\alpha$ de $\vect v_0$ sobre la
horizontal fija sus coordenadas $(v_0\cos\alpha,\; v_0\sin\alpha)$.
\end{definition}

\begin{theorem}[Ecuaciones del movimiento]\label{thm:g12:projectile-motion:equations}
Un \omterm{def:g12:projectile-motion:projectile}{proyectil} lanzado desde el origen con rapidez $v_0$ y ángulo $\alpha$ tiene
$\vect a = (0,\,-g)$ en todo instante, de donde
\[
\vect v(t) = \bigl(v_0\cos\alpha,\; v_0\sin\alpha - g t\bigr),
\qquad
\begin{cases}
x(t) = v_0\cos\alpha\; t,\\[2pt]
y(t) = v_0\sin\alpha\; t - \tfrac12\, g t^2 .
\end{cases}
\]
\end{theorem}

\begin{proof}
La \omterm{thm:g12:newtons-laws:second}{segunda ley de Newton} da $\vect a = \vect g = (0,-g)$. Primitivas eje por eje:
$\vect v(t)$ a partir de $\vect a$ con valor inicial
$(v_0\cos\alpha,\, v_0\sin\alpha)$, y después la posición a partir de $\vect v$, desde
$(0,0)$.
\end{proof}

\begin{proposition}[Independencia de los dos movimientos]\label{prop:g12:projectile-motion:independence}
El movimiento horizontal es uniforme --- $v_x$ no cambia nunca --- y el movimiento
vertical es exactamente el lanzamiento vertical de la sección anterior: ninguna de las
dos ecuaciones menciona la coordenada de la otra. En particular, una bola disparada
horizontalmente y otra soltada en el mismo instante desde la misma altura se mantienen
a la misma altura la una que la otra --- llegan al suelo a la vez.
\end{proposition}

\begin{proof}
En el \cref{thm:g12:projectile-motion:equations}, $x(t)$ solo involucra
$v_0\cos\alpha$, e $y(t)$ solo $v_0\sin\alpha$ y $g$; ambas bolas tienen
$v_y(0) = 0$, así que ambas obedecen $h = \tfrac12 gt^2$.
\end{proof}

\begin{remark}[El mono que se suelta]\label{rem:g12:projectile-motion:monkey}
Un cuidador apunta un dardo \emph{directamente} a un mono, que se suelta de su rama en
el momento del disparo. Mala jugada: el dardo y el mono caen cada uno $\tfrac12 gt^2$
por debajo de la posición que tendrían sin gravedad --- la línea de puntería, la rama
---, así que se encuentran exactamente, sea cual sea la rapidez del dardo
(\cref{exo:g12:projectile-motion:15}).
\end{remark}

\section{La parábola, leída entera}

\begin{theorem}[Trayectoria]\label{thm:g12:projectile-motion:trajectory}
Eliminando $t$ entre $x(t)$ e $y(t)$: la \omterm{def:g10:relative-motion:trajectory}{trayectoria} de un \omterm{def:g12:projectile-motion:projectile}{proyectil}
($0 \leq \alpha < \ang{90}$) es la parábola de ramas hacia abajo
\[
y = x\tan\alpha \;-\; \frac{g}{2 v_0^2 \cos^2\alpha}\, x^2 .
\]
\end{theorem}

\begin{proof}
$t = x/(v_0\cos\alpha)$, sustituido en el $y(t)$ del
\cref{thm:g12:projectile-motion:equations}.
\end{proof}

\begin{definition}[Alcance]\label{def:g12:projectile-motion:range}
Para un \omterm{def:g12:projectile-motion:projectile}{proyectil} sobre terreno horizontal, el \emph{alcance}\index{alcance} $R$ es la
distancia horizontal recorrida cuando vuelve a su altura de lanzamiento; el
\emph{tiempo de vuelo}\index{tiempo de vuelo} $t_f$ es la duración de ese trayecto.
\end{definition}

\begin{proposition}[Cima, tiempo de vuelo y alcance]\label{prop:g12:projectile-motion:range}
Sobre terreno horizontal, con $0 < \alpha < \ang{90}$:
\[
t_f = \frac{2 v_0 \sin\alpha}{g}, \qquad
h_{\text{cima}} = \frac{(v_0\sin\alpha)^2}{2g}
\ \ \text{alcanzada en } t_f/2, \qquad
R = \frac{v_0^2 \sin(2\alpha)}{g}.
\]
El \omterm{def:g12:projectile-motion:range}{alcance} es máximo en $\alpha = \ang{45}$, donde $R = v_0^2/g$; ángulos
complementarios $\alpha$ y $\ang{90} - \alpha$ dan el mismo \omterm{def:g12:projectile-motion:range}{alcance}.
\end{proposition}

\begin{proof}
$y = 0$ da $t\,(v_0\sin\alpha - \tfrac12 gt) = 0$: el lanzamiento y $t_f$. La cima,
$v_y = 0$, cae en $t = v_0\sin\alpha/g = t_f/2$; sustitúyase en $y(t)$. Después,
$R = x(t_f) = 2v_0^2\sin\alpha\cos\alpha/g = v_0^2\sin(2\alpha)/g$. El seno alcanza su
máximo en $2\alpha = \ang{90}$, y
$\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180} - 2\alpha) = \sin(2\alpha)$.
\end{proof}

\begin{example}[Saque de puerta]\label{ex:g12:projectile-motion:kick}
Un portero golpea a $v_0 = \qty{25}{m/s}$ con $\alpha = \ang{45}$:
$R = 25^2/9.81 \approx \qty{64}{m}$; los despejes reales caen más cerca de
\qty{55}{m} --- la parte que se lleva el aire.
\end{example}

\begin{method}[Contabilidad de un proyectil]\label{met:g12:projectile-motion:method}
\begin{enumerate}
  \item Ejes en el punto de lanzamiento, $Ox$ horizontal, $Oy$ hacia arriba;
        coordenadas de $\vect v_0$.
  \item Comprueba la \omterm{def:g12:projectile-motion:freefall}{caída libre} (solo el \omterm{def:g10:universal-gravitation:weight}{peso}): entonces $\vect a = (0,-g)$; dos
        primitivas, con sus condiciones iniciales cada vez.
  \item Pregunta sobre un \emph{lugar}: elimina $t$ (\omterm{def:g10:relative-motion:trajectory}{trayectoria}). Sobre un
        \emph{instante}: conserva $t$.
  \item Diccionario: cima $\Leftrightarrow v_y = 0$; altura de lanzamiento
        $\Leftrightarrow y = 0$; muro a distancia $d \Leftrightarrow x = d$; suelo
        $\ell$ más abajo $\Leftrightarrow y = -\ell$.
  \item Sensatez: el vuelo es simétrico respecto de la cima; $R \leq v_0^2/g$; el aire
        nunca hace otra cosa que acortar.
\end{enumerate}
\end{method}

\section{Lo que cambia el aire}

\begin{definition}[Velocidad límite]\label{def:g12:projectile-motion:terminal}
La resistencia del aire se opone a la velocidad y crece con la rapidez. Por eso un
cuerpo que cae acelera solo hasta que el \omterm{def:g10:inertia:inventory}{rozamiento} del aire equilibra el \omterm{def:g10:universal-gravitation:weight}{peso}; a
partir de ahí $\vect a = \vect 0$: cae con una \emph{velocidad límite}%
\index{velocidad límite} constante.
\end{definition}

\begin{example}[Órdenes de magnitud de la velocidad límite]\label{ex:g12:projectile-motion:airorders}
Un paracaidista en posición extendida: unos \qty{55}{m/s} (\qty{200}{km/h}). Una gota
de lluvia: unos \qty{9}{m/s} --- en el vacío, \qty{2000}{m} de caída la entregarían a
$\sqrt{2gh} \approx \qty{2.0e2}{m/s}$. Una pluma: alrededor de \qty{1}{m/s},
alcanzados en unos centímetros --- el secreto de su lentitud: pasa toda la caída
\emph{a} su \omterm{def:g12:projectile-motion:terminal}{velocidad límite}; el martillo no se acerca jamás a la suya.
\end{example}

\begin{remark}[Balística real]\label{rem:g12:projectile-motion:drag}
El \omterm{def:g10:inertia:inventory}{rozamiento} del aire empuja contra la velocidad durante todo el vuelo: el \omterm{def:g12:projectile-motion:range}{alcance} se
acorta, la cima baja y se rompe la simetría --- el descenso es más empinado que la
subida, y los tiros largos se quedan visiblemente cortos respecto de la parábola. El
tratamiento honesto y cuantitativo es una ecuación diferencial que se resuelve en el
volumen del primer año de universidad; este curso, la parábola es el modelo adecuado
para \omterm{def:g12:projectile-motion:projectile}{proyectiles} densos y moderadamente rápidos --- un balón de baloncesto a
\qty{10}{m/s}, no una pelota de golf a \qty{70}{m/s}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:1}
Una piedra soltada desde un puente llega al agua \qty{2.0}{s} después: ¿altura del
puente y rapidez de impacto en \unit{m/s} y en \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:2}
En la Luna, un astronauta soltó a la vez un martillo y una pluma: llegaron al suelo
juntos. ¿Por qué --- y por qué la pluma pierde tan estrepitosamente en la Tierra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:3}
Despreciando el aire, calcula el tiempo de caída y la rapidez de llegada desde
\qty{1.0}{m}, \qty{10}{m} y \qty{100}{m}. ¿Cómo crecen los tiempos con la altura?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:4}
Se lanza una pelota verticalmente hacia arriba a \qty{15}{m/s}: ¿altura y tiempo de la
cima, tiempo total hasta volver a la mano y rapidez de regreso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:5}
Una canica rueda y se sale de una mesa de \qty{0.80}{m} de altura a \qty{2.5}{m/s}.
Halla el tiempo de caída y la distancia de aterrizaje respecto del borde. ¿Depende el
tiempo de caída de la rapidez de la canica?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:6}
A partir de la figura de los cinco ángulos del curso: ¿qué parejas de \omterm{def:g12:projectile-motion:projectile}{ángulos de tiro}
comparten \omterm{def:g12:projectile-motion:range}{alcance}, y qué ángulo gana? Para un \omterm{def:g12:projectile-motion:range}{alcance} por debajo del máximo, ¿cuál de
los dos ángulos posibles vuela más tiempo? Justifícalo con $t_f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:7}
Desde un balcón a \qty{20}{m} se lanza una pelota verticalmente hacia \emph{abajo} a
\qty{5.0}{m/s}. Halla el tiempo hasta el suelo y la rapidez de impacto; comprueba esta
última con un balance de \omterm{def:g10:energy-conservation:energy}{energía} (\cref{ch:g11:mechanical-energy}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:8}
Un \omterm{def:g12:projectile-motion:projectile}{proyectil} sale del suelo horizontal a \qty{40}{m/s} con $\alpha = \ang{30}$:
¿\omterm{def:g12:projectile-motion:range}{tiempo de vuelo}, altura de la cima y \omterm{def:g12:projectile-motion:range}{alcance}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:9}
Modeliza a una saltadora de longitud como un \omterm{def:g12:projectile-motion:projectile}{proyectil}: batida a \qty{10.5}{m/s} con
$\alpha = \ang{20}$. Calcula el \omterm{def:g12:projectile-motion:range}{alcance}; las campeonas saltan casi \qty{9}{m} ---
nombra dos ingredientes que el modelo puntual deja fuera.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:10}
A \qty{1.2}{m} sobre un suelo llano se suelta una bala y se dispara otra idéntica
horizontalmente a \qty{300}{m/s}. ¿Cuál llega antes al suelo? Calcula el tiempo de
caída y la distancia de aterrizaje de la bala disparada. ¿Qué principio se pone a
prueba?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:11}
Demuestra algebraicamente que $\alpha$ y $\ang{90} - \alpha$ comparten \omterm{def:g12:projectile-motion:range}{alcance} y halla
después los dos ángulos que hacen caer a \qty{50}{m} un \omterm{def:g12:projectile-motion:projectile}{proyectil} lanzado a
\qty{25}{m/s}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:12}
El agua sale de una manguera de jardín a \qty{8.0}{m/s}: ¿mayor \omterm{def:g12:projectile-motion:range}{alcance} horizontal?,
¿mayor altura vertical?, ¿los dos ángulos de la boquilla que riegan un arriate situado
a \qty{5.0}{m}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:13}
Se golpea un balón desde el suelo a \qty{18}{m/s} con $\alpha = \ang{35}$, hacia un

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:15}

\section{Problema: La canasta sobre la bocina}

\begin{problem}[{Problema de fin de semana --- la canasta sobre la bocina: un tiro
libre diseccionado flecha a flecha, unos dedos superados por centímetros, un lanzamiento
de campo entero puesto frente a lo que un brazo puede dar, un globo por encima del
tablero --- y un veredicto sobre el tiro más teatral del deporte}]
\label{pb:g12:projectile-motion:1}
El aro de una canasta de baloncesto es un anillo horizontal a \qty{3.05}{m} del suelo;
Nora suelta todos sus tiros a \qty{2.00}{m}, así que con los ejes en el punto de
suelta ($Ox$ horizontal hacia la canasta, $Oy$ hacia arriba) el centro del aro queda
\qty{1.05}{m} por encima del origen. Se desprecia el aire salvo donde se indique.

\medskip\noindent\textbf{Parte I --- El tiro libre.} Nora tira desde
$d = \qty{4.20}{m}$ (en horizontal) con un \omterm{def:g12:projectile-motion:projectile}{ángulo de tiro} $\alpha = \ang{50}$.
\begin{enumerate}
  \item ¿Qué \omterm{def:g10:inertia:force}{fuerzas} actúan una vez soltado el balón? Da $\vect a$, citando la ley
        empleada.
  \item Deduce las coordenadas de $\vect v(t)$.
  \item Deduce $x(t)$ e $y(t)$.
  \item Elimina $t$ para obtener la \omterm{def:g10:relative-motion:trajectory}{trayectoria} $y(x)$.
  \item El balón debe pasar por el centro del aro, $y(d) = \qty{1.05}{m}$: despeja
        $v_0$, en \unit{m/s} y en \unit{km/h}.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Por encima de los dedos, dentro del aro.}
\begin{enumerate}[resume]
  \item ¿Cuánto tarda el balón en llegar al aro?
  \item Halla el instante y la altura de la cima; comprueba que en el aro el balón ya
        está descendiendo.
  \item Una defensora situada \qty{0.90}{m} por delante de Nora estira los dedos hasta
        \qty{2.60}{m}. ¿Por cuánto los supera el balón?
  \item Calcula $v_x$ y $v_y$ en el aro, la rapidez allí y el ángulo de la velocidad
        por debajo de la horizontal.
  \item ¿Por qué el balón debe llegar \emph{descendiendo} --- y por qué de forma
        empinada?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El lanzamiento de campo entero.} A punto de
\omterm{met:g10:signals-and-waves:echo}{sonar} la bocina, Nora está a \qty{26.0}{m} del aro contrario.
\begin{enumerate}[resume]
  \item La subida de \qty{1.05}{m} es pequeña frente a \qty{26.0}{m}: justifica un
        modelo de terreno horizontal y deduce el \omterm{def:g12:projectile-motion:range}{tiempo de vuelo} de $y(t_f) = 0$.
  \item Deduce $R = v_0^2\sin(2\alpha)/g$ y el ángulo de \omterm{def:g12:projectile-motion:range}{alcance} máximo.
  \item Calcula la $v_0$ necesaria para cubrir \qty{26.0}{m} con $\alpha = \ang{45}$.
  \item Una jugadora potente puede lanzar un balón de baloncesto a unos
        \qty{17.5}{m/s}. ¿Cuál es su \omterm{def:g12:projectile-motion:range}{alcance} máximo --- es humanamente posible el
        tiro?
  \item A \qty{17.5}{m/s}, halla los \emph{dos} ángulos que hacen caer el balón a
        \qty{26.0}{m} y sus \omterm{def:g12:projectile-motion:range}{tiempos de vuelo}: con \qty{2.0}{s} en el reloj, ¿cuál
        llega antes que la bocina?
\end{enumerate}

\medskip\noindent\textbf{Parte IV --- El globo por encima del tablero.} Desde detrás
de la línea de fondo, Nora lanza un globo: suelta a \qty{6.0}{m} del centro del aro,
$\alpha = \ang{75}$; el plano del tablero corta su tiro \qty{0.45}{m} antes del centro
del aro, con el borde superior a \qty{3.95}{m} del suelo.
\begin{enumerate}[resume]
  \item Halla la $v_0$ que hace caer el balón por el centro del aro.
  \item Calcula la altura del balón en el plano del tablero ($x = \qty{5.55}{m}$):
        ¿supera el borde superior?, ¿por cuánto?
  \item Halla la altura de la cima y el \omterm{def:g12:projectile-motion:range}{tiempo de vuelo} --- ¿por qué le encanta al
        público?
  \item Cualitativamente, ¿qué cambia la resistencia del aire en estos tres tiros, y a
        cuál amenaza más?
  \item \emph{El veredicto sobre la canasta desde el propio \omterm{def:g11:electric-gravitational-fields:field}{campo}}, en dos frases:
        compara la rapidez que exige el lanzamiento de campo entero con la que
        proporciona un brazo, da el margen y dictamina --- ¿posible o mito?
\end{enumerate}
\end{problem}
