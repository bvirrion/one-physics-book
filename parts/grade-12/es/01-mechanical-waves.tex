\chapter{Ondas mecánicas}\label{ch:g12:mechanical-waves}

Sábado por la tarde, un estadio lleno: una ondulación de espectadores
que se ponen de pie recorre las gradas en menos de un minuto y, sin
embargo, nadie abandona su asiento. El anillo de una piedra cruza el
estanque mientras el corcho solo cabecea. Este capítulo estudia al
viajero mismo --- la \omterm{def:g10:signals-and-waves:wave}{onda} --- y los cuatro números que lo domestican:
velocidad, \omterm{def:g12:mechanical-waves:delay}{retraso}, \omterm{def:g10:signals-and-waves:period}{periodo} y \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} --- suficientes para leer
un terremoto en una tira de papel.

\section{La perturbación viaja, el medio se queda}

\begin{definition}[Onda mecánica]\label{def:g12:mechanical-waves:wave}
Una \emph{onda mecánica}\index{onda mecánica} es una perturbación que se
propaga por un medio material --- una cuerda, el agua, el aire, la roca
--- transportando \omterm{def:g10:energy-conservation:energy}{energía} \emph{sin transportar materia}: cada trozo del
medio oscila brevemente en su sitio, pasa el movimiento al siguiente y
vuelve a su posición.
\end{definition}

\begin{example}[Sin transporte de materia]\label{ex:g12:mechanical-waves:transport}
En la ola del estadio, el ``medio'' es el público: lo que viaja es el
acto de ponerse de pie, mientras cada espectador conserva su asiento. En
el estanque, el anillo se ensancha, pero un corcho que flote solo
cabecea en su sitio.
\end{example}

\begin{definition}[Ondas transversales y longitudinales]\label{def:g12:mechanical-waves:transverse}
Una \omterm{def:g10:signals-and-waves:wave}{onda} es \emph{transversal}\index{onda transversal} cuando el medio
se mueve \emph{perpendicularmente} a la propagación: una joroba corre a
lo largo de una cuerda sacudida mientras cada punto solo sube y baja. Es
\emph{longitudinal}\index{onda longitudinal} cuando el medio se mueve
\emph{a lo largo} de la propagación: una zona de espiras apretadas
recorre un muelle empujado, mientras cada espira oscila en su sitio.
\end{definition}

\begin{remark}[El sonido es longitudinal]\label{rem:g12:mechanical-waves:sound}
El \omterm{def:g10:signals-and-waves:sound}{sonido} es la \omterm{def:g12:mechanical-waves:transverse}{onda longitudinal} por excelencia: el aire comprimido ---
a mayor \omterm{def:g10:pressure:pressure}{presión} (\cref{ch:g10:pressure}) --- viaja de la fuente al
tímpano; el \cref{ch:g12:sound-acoustics} está dedicado a él. Los
terremotos envían los dos tipos: las \omterm{def:g10:signals-and-waves:wave}{ondas} P comprimen la roca y las
\omterm{def:g10:signals-and-waves:wave}{ondas} S la sacuden de lado.
\end{remark}

\section{Velocidad y retraso}

\begin{definition}[Velocidad de propagación]\label{def:g12:mechanical-waves:speed}
La \emph{velocidad de propagación}\index{velocidad de propagación} de
una \omterm{def:g10:signals-and-waves:wave}{onda} es la distancia recorrida por la perturbación dividida entre el
tiempo empleado: $v = d/t$. Es la velocidad de la \emph{forma} que
viaja, no la del medio.
\end{definition}

\end{omfigure}

\begin{proposition}[El medio fija la velocidad]\label{prop:g12:mechanical-waves:medium}
Para perturbaciones pequeñas, la \omterm{def:g12:mechanical-waves:speed}{velocidad de propagación} depende
únicamente del \emph{medio}, no de la forma ni del tamaño de la
perturbación. En una cuerda tensa, $v$ aumenta con la tensión y
disminuye con la masa por \omterm{def:g10:orders-of-magnitude:unit}{metro}: tensa y ligera es rápida; floja y
pesada, lenta. \admitted
\end{proposition}

\begin{remark}[Dónde vive la fórmula]\label{rem:g12:mechanical-waves:ropeformula}
La fórmula exacta de la velocidad en una cuerda se deduce en el volumen
del primer año; este año bastan los hechos experimentales anteriores.
\end{remark}

\begin{example}[Velocidades del sonido]\label{ex:g12:mechanical-waves:speeds}
El \omterm{def:g10:signals-and-waves:sound}{sonido} viaja a \qty{340}{m/s} en el aire, a \qty{1500}{m/s} en el
agua y a \qty{5900}{m/s} en el acero --- cuanto más rígido, más rápido.
De ahí la oreja sobre el raíl de las películas del oeste: el acero
anuncia el tren antes.
\end{example}

\begin{definition}[Retraso]\label{def:g12:mechanical-waves:delay}
Una \omterm{def:g10:signals-and-waves:wave}{onda} de velocidad $v$ alcanza un punto situado a la distancia $d$
tras el \emph{retraso}\index{retraso} $\tau = d/v$: dos puntos separados
una distancia $d$ repiten el mismo movimiento, y el más lejano se
retrasa $\tau$ --- la medida por \omterm{rem:g10:signals-and-waves:honest}{ecos} del
\cref{ch:g10:signals-and-waves}, leída al revés.
\end{definition}

\begin{proposition}[Superposición sin interacción]\label{prop:g12:mechanical-waves:superposition}
Donde dos \omterm{def:g10:signals-and-waves:wave}{ondas} se solapan, el desplazamiento del medio es en todo
instante la \emph{suma} de los desplazamientos que cada una causaría por
separado; y tras cruzarse, cada \omterm{def:g10:signals-and-waves:wave}{onda} continúa sin cambios --- se deduce
en el volumen del segundo año a partir de la linealidad de las
ecuaciones del movimiento. \admitted
\end{proposition}

\begin{example}[Pulsos que se cruzan]\label{ex:g12:mechanical-waves:crossing}
Envía una joroba hacia arriba desde cada extremo de una cuerda: donde se
encuentran, la cuerda se eleva brevemente hasta la suma de las dos
alturas; después emergen dos jorobas intactas --- como dos
conversaciones que se cruzan y llegan cada una a su oyente.
\end{example}

\section{Ondas periódicas: la doble periodicidad}

\begin{definition}[Onda periódica]\label{def:g12:mechanical-waves:periodic}
Cuando la fuente repite su movimiento con \omterm{def:g10:signals-and-waves:period}{periodo} $T$ y \omterm{def:g10:signals-and-waves:frequency}{frecuencia}
$f = 1/T$ (\cref{ch:g10:signals-and-waves}), alimenta el medio con una
\emph{onda periódica}\index{onda periódica}: cada punto repite su propio
movimiento con \omterm{def:g10:signals-and-waves:period}{periodo} $T$, retrasado respecto de sus vecinos en el
\omterm{def:g12:mechanical-waves:delay}{retraso} de propagación.
\end{definition}

\begin{definition}[Longitud de onda]\label{def:g12:mechanical-waves:wavelength}
Una instantánea de una \omterm{def:g12:mechanical-waves:periodic}{onda periódica} muestra un patrón que se repite en
el espacio; su \omterm{def:g10:signals-and-waves:period}{periodo} espacial --- de cresta a cresta --- es la
\emph{longitud de onda}\index{longitud de onda} $\lambda$, en \omterm{def:g10:orders-of-magnitude:unit}{metros}.
Una \omterm{def:g12:mechanical-waves:periodic}{onda periódica} es \emph{doblemente periódica}\index{doble
periodicidad}: con \omterm{def:g10:signals-and-waves:period}{periodo} $T$ en el tiempo en cada punto, y con \omterm{def:g10:signals-and-waves:period}{periodo}
$\lambda$ en el espacio en cada instante.
\end{definition}

\begin{theorem}[La relación de las ondas]\label{thm:g12:mechanical-waves:relation}
Una \omterm{def:g12:mechanical-waves:periodic}{onda periódica} de velocidad $v$, \omterm{def:g10:signals-and-waves:period}{periodo} $T$ y \omterm{def:g10:signals-and-waves:frequency}{frecuencia} $f$ tiene
\omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} $\lambda = v\,T = v/f$.
\end{theorem}

\begin{proof}
Durante un \omterm{def:g10:signals-and-waves:period}{periodo}, el patrón se desliza hacia adelante $v\,T$; pero
para entonces la fuente --- y con ella cada punto --- ha vuelto a su
estado inicial, así que el patrón desplazado coincide con el original:
la distancia de repetición es $v\,T$.
\end{proof}

\begin{remark}[La frecuencia es de la fuente]\label{rem:g12:mechanical-waves:source}
Al pasar de un medio a otro --- el \omterm{def:g10:signals-and-waves:sound}{sonido} del aire al agua --- una \omterm{def:g10:signals-and-waves:wave}{onda}
conserva su \omterm{def:g10:signals-and-waves:frequency}{frecuencia}, que dicta la fuente, mientras cambia su
velocidad; y por $\lambda = v/f$, la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} cambia con ella.
\end{remark}

\begin{example}[El la de \qty{440}{Hz}, en el aire y bajo el agua]\label{ex:g12:mechanical-waves:concertA}
El la de \qty{440}{Hz} del \cref{ch:g10:signals-and-waves} tiene una
\omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} de $340/440 = \qty{0.77}{m}$ en el aire y de
$1500/440 = \qty{3.4}{m}$ en el agua: la misma nota, estirada por el
medio más rápido.
\end{example}

\begin{definition}[En fase, en oposición de fase]\label{def:g12:mechanical-waves:phase}
Dos puntos separados por un número entero de \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda},
$d = k\lambda$, se mueven de forma idéntica en todo instante: están
\emph{en fase}\index{en fase}. Los puntos separados por
$d = (k + \tfrac12)\lambda$ se mueven siempre en sentidos opuestos ---
cresta contra valle: están en \emph{oposición de fase}\index{oposición
de fase}.
\end{definition}

\begin{method}[Leer la doble periodicidad]\label{met:g12:mechanical-waves:reading}
\begin{enumerate}
  \item En un registro $y(t)$ tomado en un punto fijo, se lee $T$ de
        cresta a cresta; $f = 1/T$.
  \item En una instantánea $y(x)$ tomada en un instante fijo, se lee
        $\lambda$ de cresta a cresta.
  \item Se deduce $v = \lambda/T = \lambda f$; y se comprueba con una
        medida directa $v = d/\tau$ cuando se disponga de ella.
  \item Nunca hay que confundir las dos gráficas: un eje lleva segundos
        y el otro, \omterm{def:g10:orders-of-magnitude:unit}{metros}.
\end{enumerate}
\end{method}

\section{Frentes de onda y atenuación}

\begin{definition}[Frente de onda]\label{def:g12:mechanical-waves:wavefront}
Un \emph{frente de onda}\index{frente de onda} es una línea (o una
superficie) de puntos a los que la \omterm{def:g10:signals-and-waves:wave}{onda} llega en el mismo instante ---
una línea de cresta en el estanque. De una fuente puntual salen frentes
de \omterm{def:g10:signals-and-waves:wave}{onda} circulares que se ensanchan; lejos de cualquier fuente son casi
rectos. La \omterm{def:g10:signals-and-waves:wave}{onda} se propaga perpendicularmente a sus frentes, con las
crestas sucesivas separadas $\lambda$.
\end{definition}

\begin{definition}[Atenuación]\label{def:g12:mechanical-waves:attenuation}
La disminución de la \omterm{def:g10:signals-and-waves:amplitude}{amplitud} de una \omterm{def:g10:signals-and-waves:wave}{onda} a medida que se propaga es su
\emph{atenuación}\index{atenuación}. Se suman dos causas: la \omterm{def:g10:energy-conservation:energy}{energía} de
un frente circular se reparte sobre una cresta cada vez más larga, y el
medio absorbe un poco al pasar. Los frentes rectos escapan de la primera
causa --- y por eso el oleaje de una tormenta lejana puede cruzar un
océano.
\end{definition}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:1}
Una ola de estadio recorre el anillo de \qty{320}{m} de las gradas en
\qty{40}{s}. (a) Calcula su velocidad. (b) ¿Recorre algún espectador el
anillo entero? ¿Qué lo recorre? (c) Cada espectador se levanta
\qty{1.5}{s} después que su vecino situado a \qty{12}{m}: comprueba la
coherencia.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:2}
¿\omterm{def:g12:mechanical-waves:transverse}{Transversal} o \omterm{def:g12:mechanical-waves:transverse}{longitudinal}? (a) un pulso en una cuerda sacudida arriba
y abajo; (b) una compresión enviada a lo largo de un muelle; (c) el
\omterm{def:g10:signals-and-waves:sound}{sonido} en el aire; (d) una ondulación en un estanque (la superficie
cabecea en vertical y el anillo viaja en horizontal); (e) las \omterm{def:g10:signals-and-waves:wave}{ondas}
sísmicas P (que comprimen en el sentido de la marcha) y las S (que
sacuden de lado).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:3}
El trueno te llega \qty{4.0}{s} después del destello. (a) ¿A qué
distancia cayó el rayo ($v = \qty{340}{m/s}$)? (b) Un buceador nota ese
mismo \omterm{def:g12:mechanical-waves:delay}{retraso} entre un destello y un estruendo submarino
($v = \qty{1500}{m/s}$): ¿a qué distancia está esa fuente? (c) ¿Qué hay
que saber antes de convertir un \omterm{def:g12:mechanical-waves:delay}{retraso} en una distancia?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:4}
Calcula la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} de un \omterm{def:g10:signals-and-waves:sound}{sonido} de \qty{440}{Hz} en el aire y
en el agua. Cuando el \omterm{def:g10:signals-and-waves:sound}{sonido} pasa del aire al agua, ¿cuáles de $f$, $v$
y $\lambda$ cambian y cuál se mantiene fija?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:5}
En un estanque, las crestas contiguas de las ondulaciones distan
\qty{12}{cm} y los anillos avanzan a \qty{0.30}{m/s}. Halla la \omterm{def:g12:mechanical-waves:wavelength}{longitud

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:mechanical-waves:6}
Un pulso en una cuerda larga se fotografía dos veces, con \qty{0.30}{s}
de diferencia: su cresta está en $x = \qty{1.0}{m}$ y después en

\begin{exercise}[$\star\star$]\label{exo:g12:mechanical-waves:7}
Dos pulsos hacia arriba, de \qty{3.0}{cm} y \qty{2.0}{cm} de altura,

\begin{exercise}[$\star\star$]\label{exo:g12:mechanical-waves:8}

(c) ¿Sacudirla con el doble de \omterm{def:g10:inertia:force}{fuerza} (un pulso más alto) hace que el

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:mechanical-waves:9}

\begin{exercise}[$\star\star$]\label{exo:g12:mechanical-waves:10}

\begin{exercise}[$\star\star$]\label{exo:g12:mechanical-waves:11}
Un terremoto envía \omterm{def:g10:signals-and-waves:wave}{ondas} P a \qty{6.0}{km/s} y \omterm{def:g10:signals-and-waves:wave}{ondas} S a

\begin{exercise}[$\star\star\star$]\label{exo:g12:mechanical-waves:12}

\section{Problema: leer un sismograma}

\begin{problem}[{Problema de fin de semana --- leer un sismograma: dos
ondas salen disparadas de una falla rota, tres circunferencias fijan el
epicentro y una tercera onda lenta le da su aviso a una costa}]
\label{pb:g12:mechanical-waves:1}
Turno de noche en un observatorio sismológico. A las 09:14:32 la aguja
de la estación A da un salto: temblores rápidos y agudos y, después,
oscilaciones más lentas y anchas. Un terremoto ha roto roca en alguna
parte; dos \omterm{def:g10:signals-and-waves:wave}{ondas} salen disparadas por la corteza: las \omterm{def:g10:signals-and-waves:wave}{ondas} P,
compresiones a $v_P = \qty{6.0}{km/s}$, y las \omterm{def:g10:signals-and-waves:wave}{ondas} S, una sacudida
lateral a $v_S = \qty{3.5}{km/s}$.

\medskip
\noindent\textbf{Parte I --- Dos \omterm{def:g10:signals-and-waves:wave}{ondas}, un reloj.}
\begin{enumerate}
  \item Con las descripciones anteriores, clasifica las \omterm{def:g10:signals-and-waves:wave}{ondas} P y S como
        \omterm{def:g12:mechanical-waves:transverse}{transversales} o \omterm{def:g12:mechanical-waves:transverse}{longitudinales}.
  \item ¿Qué \omterm{def:g10:signals-and-waves:wave}{onda} llega primero a todas las estaciones? Calcula los
        tiempos de viaje de las dos a lo largo de $d = \qty{120}{km}$.
  \item Demuestra que el intervalo entre las llegadas P y S vale
        $\Delta t = d\,(1/v_S - 1/v_P)$ y calcúlalo para
        $d = \qty{120}{km}$.
  \item Deduce la regla de medida del sismólogo
        $d = \dfrac{v_P\,v_S}{v_P - v_S}\,\Delta t$ y demuestra que el
        coeficiente vale unos \qty{8.4}{km} por segundo de intervalo.
  \item ¿Por qué crece el intervalo con la distancia --- y por qué es
        eso un regalo para una estación que nunca vio empezar el
        terremoto?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Circunferencias sobre el mapa.}
\begin{enumerate}[resume]
  \item La estación A mide $\Delta t_A = \qty{25}{s}$. ¿A qué distancia
        está el epicentro de A?
  \item Explica por qué esta única lectura sitúa el epicentro sobre una
        circunferencia y no en un punto.
  \item Las estaciones B y C miden $\Delta t_B = \qty{15}{s}$ y
        $\Delta t_C = \qty{32}{s}$: ¿sus distancias al epicentro?
  \item Explica cómo las tres circunferencias juntas fijan el epicentro.
  \item La \omterm{def:g10:signals-and-waves:wave}{onda} P llegó a A a las 09:14:32. Calcula su tiempo de viaje
        desde el epicentro y la hora de origen del terremoto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La \omterm{def:g10:signals-and-waves:wave}{onda} que cruza el océano.}
Las tres circunferencias se cortan mar adentro: la falla se rompió bajo
el fondo marino y puso en movimiento la columna de agua. Las \omterm{def:g10:signals-and-waves:wave}{ondas}
largas en aguas profundas viajan a una velocidad que fija la profundidad
--- unos \qty{200}{m/s} en mar abierto y unos \qty{10}{m/s} junto a una
playa (admitido aquí; la fórmula se deduce en el volumen del primer
año). Un puerto está a $D = \qty{600}{km}$ del epicentro.
\begin{enumerate}[resume]
  \item Convierte \qty{200}{m/s} a \unit{km/h} y nombra un vehículo con
        una velocidad de crucero comparable.
  \item ¿Cuánto tarda el tsunami en llegar al puerto?
  \item El sismómetro del propio puerto nota antes la \omterm{def:g10:signals-and-waves:wave}{onda} P. ¿Al cabo
        de cuánto? Si la alarma suena en ese instante, ¿de cuánto tiempo
        de aviso dispone la costa antes del tsunami?
  \item En mar abierto, el \omterm{def:g10:signals-and-waves:period}{periodo} del tsunami es de unos \qty{15}{min}.
        Calcula allí su \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} y explica por qué un barco no
        lo nota.
  \item Al acercarse a la playa, la velocidad cae a \qty{10}{m/s}
        mientras el \omterm{def:g10:signals-and-waves:period}{periodo} se mantiene fijo. Calcula la nueva \omterm{def:g12:mechanical-waves:wavelength}{longitud
        de onda}. ¿Qué tiene que pasarle a la \omterm{def:g10:signals-and-waves:wave}{onda} al apretujarse?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La \omterm{def:g10:energy-conservation:energy}{energía} y el informe.}
Un terremoto fuerte libera del orden de \qty{1e15}{J}, que se llevan las
\omterm{def:g10:signals-and-waves:wave}{ondas}.
\begin{enumerate}[resume]
  \item Calcula la longitud de un \omterm{def:g12:mechanical-waves:wavefront}{frente de onda} circular a \qty{10}{km}
        del epicentro y después a \qty{210}{km}; ¿en qué factor ha
        crecido?
  \item Deduce por qué la sacudida se debilita con la distancia incluso
        en una corteza sin pérdidas --- y nombra la segunda causa que
        añade una corteza real.
  \item Un líquido no puede resistir la cizalla. Las estaciones del otro
        lado del planeta registran la \omterm{def:g10:signals-and-waves:wave}{onda} P pero no la S: ¿qué revela
        eso sobre el \omterm{def:g11:fundamental-interactions:ladder}{núcleo} de la Tierra?
  \item El tren de alta velocidad del capítulo de \omterm{def:g10:energy-conservation:energy}{energía} del año
        anterior (\cref{ch:g11:mechanical-energy}) lleva unos
        \qty{1.5e9}{J} de \omterm{def:g11:mechanical-energy:kinetic}{energía cinética}: ¿a cuántos trenes así
        equivalen \qty{1e15}{J}?
  \item El informe, en dos frases: dónde y cuándo fue el terremoto y de
        cuánto aviso dispuso el puerto. Da los tres números.
\end{enumerate}
\end{problem}
