\chapter{El efecto Doppler}\label{ch:g12:doppler-effect}

Una ambulancia se acerca con la sirena a todo trapo y pasa de largo ---
y en ese instante la nota se desliza audiblemente hacia abajo, aunque el
conductor no ha tocado nada. La sirena es honrada; la \omterm{def:g10:signals-and-waves:wave}{onda}, no: el
movimiento la aprieta por delante y la estira por detrás. Este capítulo
mide el apretón y después cabalga la misma idea desde las sirenas hasta
los \omterm{met:g10:signals-and-waves:echo}{radares} de tráfico, la sangre de una arteria y un planeta delatado
por un bamboleo en la luz de una estrella.

\section{El efecto}

\begin{definition}[Efecto Doppler]\label{def:g12:doppler-effect:doppler}
El \emph{efecto Doppler}\index{efecto Doppler} es el cambio de la
\omterm{def:g10:signals-and-waves:frequency}{frecuencia} recibida de una \omterm{def:g10:signals-and-waves:wave}{onda} cuando la fuente y el receptor se mueven
uno respecto del otro: la \omterm{def:g10:signals-and-waves:frequency}{frecuencia} recibida $f'$ es mayor que la
emitida $f$ mientras se acercan, menor mientras se alejan e igual a $f$
solo mientras su distancia deja momentáneamente de cambiar.
\end{definition}

\begin{remark}[Tono, no sonoridad]\label{rem:g12:doppler-effect:pitch}
Al pasar la ambulancia cambian dos cosas. La \emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{sonoridad}} crece y se
apaga porque la \omterm{def:g10:signals-and-waves:wave}{onda} se dispersa con la distancia
(\cref{ch:g12:sound-acoustics}); el \emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{tono}} se desplaza por la
velocidad de acercamiento o de alejamiento, y cae bruscamente al pasar.
La \omterm{def:g12:sound-acoustics:pitch-loudness}{sonoridad} dice ``a qué distancia''; el \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} dice ``a qué
velocidad''.
\end{remark}

\section{Una fuente en movimiento: frentes de onda apiñados}

La imagen clave: en cuanto una cresta abandona la fuente, el medio se
adueña de ella. Cada cresta se ensancha como una circunferencia a la
velocidad $v$ de la \omterm{def:g10:signals-and-waves:wave}{onda} (\cref{ch:g12:mechanical-waves}), centrada en
el punto donde fue \emph{emitida} --- el medio ni sabe ni le importa que
la fuente se haya movido.

\begin{proposition}[Fuente en movimiento]\label{prop:g12:doppler-effect:moving-source}
Una fuente emite crestas con \omterm{def:g10:signals-and-waves:period}{periodo} $T$ (\omterm{def:g10:signals-and-waves:frequency}{frecuencia} $f = 1/T$) mientras
se mueve en línea recta a la velocidad $v_s < v$ por un medio en el que
la \omterm{def:g10:signals-and-waves:wave}{onda} viaja a la velocidad $v$. Un observador en reposo en el medio
recibe
\[
\lambda' = (v \mp v_s)\,T, \qquad f' = f\,\frac{v}{v \mp v_s},
\]
con los signos superiores para una fuente que se acerca y los
inferiores para una que se aleja.
\end{proposition}

\begin{proof}
Entre dos crestas, la primera avanza $vT$ mientras la fuente avanza
$v_s T$. Por delante, la segunda cresta se emite, por tanto, $v_s T$
\emph{más cerca} de la primera, $\lambda' = (v - v_s)T$; por detrás, ese
mismo $v_s T$ se suma. Después las crestas pasan ante un observador en
reposo a la velocidad $v$ del medio, una cada $\lambda'/v$ segundos,
luego $f' = v/\lambda' = f v/(v \mp v_s)$.
\end{proof}

\begin{example}[La ambulancia, con números]\label{ex:g12:doppler-effect:siren}
Una sirena de $f = \qty{700}{Hz}$ se acerca a $v_s = \qty{25}{m/s}$;
tomamos $v = \qty{340}{m/s}$ en el aire
(\cref{ch:g12:sound-acoustics}). Al acercarse:
$f' = 700 \times 340/315 \approx \qty{756}{Hz}$; al alejarse:
$f' = 700 \times 340/365 \approx \qty{652}{Hz}$. El paso baja el \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} en
\qty{104}{Hz} --- una razón $\approx 1.16$, unos dos semitonos y medio,
inconfundible para cualquier oído.
\end{example}

\begin{remark}[Lo que el movimiento no cambia]\label{rem:g12:doppler-effect:speed-unchanged}
La \omterm{def:g10:signals-and-waves:wave}{onda} sigue viajando a $v$: la velocidad la fija el medio y nada más
(\cref{ch:g12:mechanical-waves}). El movimiento de la fuente aprieta
\emph{\omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda}}, nunca velocidades. Y el apretón tiene un
límite: cuando $v_s \to v$, las crestas de delante se amontonan unas
sobre otras y $f' \to \infty$ --- el atasco de \omterm{def:g12:mechanical-waves:wavefront}{frentes de onda} que un
avión supersónico arrastra como una \omterm{def:g10:signals-and-waves:wave}{onda} de choque.
\end{remark}

\section{Un observador en movimiento y el atajo del movimiento lento}

\begin{proposition}[Observador en movimiento]\label{prop:g12:doppler-effect:moving-observer}
Un observador se mueve a la velocidad $v_o$ en línea recta hacia (o
desde) una fuente en reposo en el medio. La \omterm{def:g10:signals-and-waves:frequency}{frecuencia} recibida es
\[ f' = f\left(1 \pm \frac{v_o}{v}\right) \quad (+\ \text{al acercarse},\ -\ \text{al alejarse}). \]
\end{proposition}

\begin{proof}
El patrón de la \omterm{def:g10:signals-and-waves:wave}{onda} no se altera: las crestas, separadas
$\lambda = vT$, viajan a $v$. Un observador que corra hacia ellas se
encuentra crestas a la velocidad relativa $v + v_o$, una cada
$\lambda/(v + v_o)$ segundos: $f' = (v + v_o)/\lambda = f(1 + v_o/v)$.
Corriendo en sentido contrario, $v - v_o$.
\end{proof}

\begin{definition}[Velocidad radial]\label{def:g12:doppler-effect:radial}
La \emph{velocidad radial}\index{velocidad radial} $v_r$ de una fuente
respecto de un observador es la componente de su velocidad relativa
según la recta que los une --- el ritmo al que la distancia se encoge o
crece. Solo esa componente desplaza la \omterm{def:g10:signals-and-waves:frequency}{frecuencia}; el movimiento
\emph{perpendicular} a la línea de visión no produce desplazamiento a
este orden.
\end{definition}

\begin{proposition}[Movimiento lento: una fórmula para todos]\label{prop:g12:doppler-effect:slow}
Cuando la \omterm{def:g12:doppler-effect:radial}{velocidad radial} es pequeña, $v_r \ll v$, ya da igual quién se
mueva:
\[
\frac{\Delta f}{f} = \frac{f' - f}{f} \approx +\frac{v_r}{v} \ \text{(acercamiento)},
\qquad
\frac{\Delta f}{f} \approx -\frac{v_r}{v} \ \text{(alejamiento)}.
\]
\end{proposition}

\begin{proof}
Las fórmulas del observador ya son exactamente $1 \pm v_o/v$. Para la
fuente, escribimos $x = v_s/v$: la identidad
$\frac{1}{1 - x} = 1 + x + \frac{x^2}{1 - x}$ muestra que $f'/f$ se
aparta de $1 + x$ en un término de orden $x^2$ --- para $x = 0.1$, una
corrección del $1\,\%$ sobre un desplazamiento del $10\,\%$. A primer orden,
las cuatro fórmulas se funden en $1 \pm v_r/v$.
\end{proof}

\begin{method}[Leer un paso]\label{met:g12:doppler-effect:pass}
El registro de una fuente que pasa muestra dos mesetas: $f_1$
(acercamiento) y $f_2$ (alejamiento). Entonces
\[
f = \frac{2 f_1 f_2}{f_1 + f_2}, \qquad
v_s = v\,\frac{f_1 - f_2}{f_1 + f_2}.
\]
En efecto, $f_1 + f_2 = \frac{2fv^2}{v^2 - v_s^2}$ y
$f_1 - f_2 = \frac{2fv\,v_s}{v^2 - v_s^2}$: se dividen para obtener
$v_s$ y se calcula $2f_1 f_2/(f_1 + f_2)$ para $f$. Dos lecturas de
\omterm{def:g10:signals-and-waves:frequency}{frecuencia} dan a la vez el \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} verdadero y la velocidad --- sin
cronómetro y sin regla.
\end{method}

\section{Ecos que miden velocidades}

Si se hace rebotar una \omterm{def:g10:signals-and-waves:wave}{onda} en un blanco móvil, el \omterm{def:g12:doppler-effect:doppler}{efecto Doppler}
golpea dos veces.

\begin{proposition}[La reflexión duplica el desplazamiento]\label{prop:g12:doppler-effect:double}
Se envía una \omterm{def:g10:signals-and-waves:wave}{onda} de \omterm{def:g10:signals-and-waves:frequency}{frecuencia} $f$ contra un reflector que se acerca de
frente a la velocidad $u \ll v$. El \omterm{rem:g10:signals-and-waves:honest}{eco} vuelve con
\[ f' = f\,\frac{v + u}{v - u}, \qquad \text{y por tanto} \qquad \Delta f \approx \frac{2u}{v}\,f . \]
\end{proposition}

\begin{proof}
Dos desplazamientos en serie. Como \emph{observador} en movimiento, el
reflector recibe $f(1 + u/v)$; al reemitir lo que recibe, pasa a ser una
\emph{fuente} en movimiento, de modo que el \omterm{rem:g10:signals-and-waves:honest}{eco} llega a
$f' = f(1 + u/v)/(1 - u/v) \approx f(1 + 2u/v)$ para $u \ll v$.
\end{proof}

\begin{example}[El radar de tráfico]\label{ex:g12:doppler-effect:radar}
Un \omterm{met:g10:signals-and-waves:echo}{radar} de tráfico emite microondas a $f = \qty{24.15}{GHz}$
($v = c = \qty{3.00e8}{m/s}$) y superpone el \omterm{rem:g10:signals-and-waves:honest}{eco} a la \omterm{def:g10:signals-and-waves:wave}{onda} saliente; el
batido (\cref{ch:g12:sound-acoustics}) hace \omterm{met:g10:signals-and-waves:echo}{sonar} directamente
$\Delta f = 2uf/c \approx \qty{161}{Hz}$ por cada \unit{m/s} de
velocidad del coche --- una \omterm{def:g10:signals-and-waves:frequency}{frecuencia} de audio trivialmente medible
tallada en una portadora de \qty{24}{GHz}. El factor $2$ no es
opcional: olvidarlo halaga a todos los conductores por la mitad.
\end{example}

\begin{example}[Ecografía Doppler]\label{ex:g12:doppler-effect:ultrasound}
Una sonda médica envía \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonidos} de $f = \qty{5.0}{MHz}$ a una
arteria ($v \approx \qty{1540}{m/s}$ en el tejido); los glóbulos rojos
los reflejan. Una sangre a $u = \qty{0.50}{m/s}$ devuelve
$\Delta f = 2 \times 0.50 \times \num{5.0e6}/1540 \approx \qty{3.2}{kHz}$
--- un batido \emph{audible}: el cardiólogo oye literalmente cómo se
acelera la sangre en cada latido y después mide $u$ a partir de
$\Delta f$.
\end{example}

\section{El Doppler en la luz de las estrellas}

La luz es una \omterm{def:g10:signals-and-waves:wave}{onda} (\cref{ch:g12:light-as-wave}), y su \omterm{def:g12:sound-acoustics:timbre}{espectro} está
rayado por líneas nítidas en \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} que fija cada elemento
químico (\cref{ch:g10:light-spectra}) --- una regla calibrada en el
laboratorio e impresa en todas las estrellas.

\begin{definition}[Corrimiento al rojo y al azul]\label{def:g12:doppler-effect:redshift}
Cuando las \omterm{def:g10:light-spectra:emission}{líneas espectrales} de una fuente aparecen en \omterm{def:g12:mechanical-waves:wavelength}{longitudes de
onda} \emph{mayores} que en el laboratorio, se dice que el \omterm{def:g12:sound-acoustics:timbre}{espectro} está
\emph{corrido al rojo}\index{corrimiento al rojo}; cuando aparecen en
longitudes \emph{menores}, está \emph{corrido al
azul}\index{corrimiento al azul} --- desplazado hacia el extremo rojo o
el azul del \omterm{def:g10:light-spectra:wavelength}{espectro visible}.
\end{definition}

\begin{proposition}[Desplazamiento Doppler de la luz]\label{prop:g12:doppler-effect:light}
Para una fuente que se aleja con \omterm{def:g12:doppler-effect:radial}{velocidad radial} $v_r \ll c$, todas las
\omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} se observan estiradas en
\[ \frac{\Delta \lambda}{\lambda} = \frac{\lambda' - \lambda}{\lambda} \approx \frac{v_r}{c} \quad \text{(corrimiento al rojo)}, \]
y comprimidas en esa misma cantidad,
$\Delta\lambda/\lambda \approx -v_r/c$, para una fuente que se acerca
(\omterm{def:g12:doppler-effect:redshift}{corrimiento al azul}).
\end{proposition}
\admitted

\begin{remark}[Dónde vive la fórmula honrada]\label{rem:g12:doppler-effect:honest}
La luz no necesita medio, así que no se puede copiar la deducción del
\omterm{def:g10:signals-and-waves:sound}{sonido} --- no existe un ``en reposo en el medio''. La fórmula exacta sale
de la relatividad especial, más adelante este mismo año
(\cref{ch:g12:special-relativity}); las correcciones son de orden
$(v_r/c)^2$, despreciables para aviones, estrellas y galaxias cercanas.
Más abajo usamos la fórmula del movimiento lento con la conciencia
tranquila.
\end{remark}

\begin{example}[Medir la huida de una estrella]\label{ex:g12:doppler-effect:vega}
En el \omterm{def:g12:sound-acoustics:timbre}{espectro} de la brillante estrella Vega, la línea del hidrógeno de
\omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} de laboratorio \qty{656.30}{nm} se mide en
\qty{656.27}{nm}: $\Delta\lambda = \qty{-0.03}{nm}$, un \omterm{def:g12:doppler-effect:redshift}{corrimiento al
azul}, así que Vega se acerca a
$v_r = c\,|\Delta\lambda|/\lambda \approx \qty{1.4e4}{m/s}$ ---
\qty{14}{km/s}, leídos en un desplazamiento de una parte en veinte mil.
\end{example}

\begin{example}[Estrellas que se bambolean: binarias y exoplanetas]\label{ex:g12:doppler-effect:wobble}
Cuando dos estrellas se orbitan mutuamente, sus líneas oscilan
periódicamente entre el azul y el rojo --- una \emph{binaria
espectroscópica}\index{binaria espectroscópica}, con \omterm{def:g10:universal-gravitation:satellite}{órbitas} medidas sin
haber resuelto nunca la pareja. Un planeta le hace lo mismo a su
estrella, mucho más débilmente: los dos giran en torno a su \omterm{def:g11:forces-and-motion:com}{centro de
masas} común, de modo que la \omterm{def:g12:doppler-effect:radial}{velocidad radial} de la estrella oscila unas
pocas decenas de \unit{m/s} --- un desplazamiento
$\Delta\lambda/\lambda \sim 10^{-7}$. Detectarlo fue la manera de
encontrar el primer planeta alrededor de una estrella parecida al Sol;
el problema de fin de semana rehace aquel descubrimiento, números
incluidos.
\end{example}

\begin{remark}[Los corrimientos al rojo de las galaxias]\label{rem:g12:doppler-effect:expansion}
Los \omterm{def:g12:sound-acoustics:timbre}{espectros} de las galaxias lejanas están todos corridos al
\emph{rojo}, y cuanto más lejanas, más corridos: las distancias del
universo se están estirando. Para las galaxias cercanas,
$v_r = c\,\Delta\lambda/\lambda$ da directamente la velocidad de
alejamiento; para las más lejanas, los desplazamientos superan lo que la
fórmula del movimiento lento puede tratar honradamente, y la
contabilidad se retoma a partir del
\cref{ch:g12:special-relativity} y en los volúmenes universitarios.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:1}
La sirena de un camión de bomberos emite a \qty{700}{Hz} y circula a
\qty{25}{m/s} ($v = \qty{340}{m/s}$). Calcula la \omterm{def:g10:signals-and-waves:frequency}{frecuencia} que oye un
peatón (a) situado delante de él; (b) situado detrás. (c) ¿Qué oye el
conductor?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:2}
Un amigo afirma: ``el \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} baja cuando la ambulancia se va porque el
\omterm{def:g10:signals-and-waves:sound}{sonido} se debilita con la distancia''. Desenreda los dos efectos que
está confundiendo y di de qué depende cada uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:3}
Un ciclista pedalea a \qty{6.0}{m/s} directamente hacia una sirena fija
que emite a \qty{700}{Hz}. ¿Qué \omterm{def:g10:signals-and-waves:frequency}{frecuencia} oye? ¿Y alejándose en línea
recta? ¿Qué fórmula se aplica, y por qué no la de la fuente en
movimiento?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:4}
La bocina de un tren emite a \qty{400}{Hz} mientras el tren circula a
\qty{30}{m/s}. Calcula la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} por delante del tren, por
detrás y en reposo. ¿Qué observador recibe cada una?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:5}
Un móvil que reproduce un \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} de \qty{440}{Hz} lo lleva un corredor a
\qty{5.0}{m/s}. Con la fórmula del movimiento lento, estima el
desplazamiento que oye alguien hacia quien se acerca el corredor.
Compáralo con un semitono (alrededor del $6\,\%$ en \omterm{def:g10:signals-and-waves:frequency}{frecuencia}): ¿lo
notaría un músico?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:6}
Un murciélago vuela a \qty{6.0}{m/s} directamente hacia una pared
emitiendo a \qty{50.0}{kHz}. Demuestra que el \omterm{rem:g10:signals-and-waves:honest}{eco} que oye vuelve a
$f' = f(v + v_b)/(v - v_b)$ y calcula después $f'$ y el desplazamiento.
¿Por qué es esta la fórmula del \omterm{met:g10:signals-and-waves:echo}{radar} disfrazada?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:7}
Un \omterm{met:g10:signals-and-waves:echo}{radar} de \qty{24.15}{GHz} mide un batido de \qty{4.0}{kHz} de un
coche que se acerca. Halla la velocidad del coche en \unit{m/s} y en
\unit{km/h}. ¿Qué batido daría un coche a exactamente \qty{50}{km/h}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:8}
Una sonda de ecografía Doppler de \qty{5.0}{MHz} apuntada a lo largo de
una arteria ($v = \qty{1540}{m/s}$ en el tejido) oye un batido de
\qty{2.6}{kHz}. Halla la velocidad de la sangre. ¿Por qué hay que
inclinar la sonda \emph{a lo largo} del flujo y no perpendicularmente a
él?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:9}
Toma $f = \qty{1000}{Hz}$, $v = \qty{340}{m/s}$ y una velocidad de

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:10}
El registro de una motocicleta que pasa ante un micrófono muestra

\end{exercise}

\omterm{def:g10:orders-of-magnitude:oom}{Orden de magnitud}: ¿por qué la caza de planetas tuvo que esperar a

\section{Problema: la caza de un exoplaneta}

\begin{problem}[{Problema de fin de semana --- la caza de un exoplaneta:
la bocina de un tren calibra el oído, un radar de tráfico ensaya el
doble desplazamiento y, el domingo por la noche, el espectro de una
estrella que se bambolea una parte en cinco millones ha pesado un
planeta que nadie ha visto jamás}]
\label{pb:g12:doppler-effect:1}
Un club de astronomía dedica un fin de semana a una sola idea --- el
desplazamiento Doppler ---, ensayada en la Tierra y apuntada después a
una estrella parecida al Sol. Datos: $v = \qty{340}{m/s}$ para el \omterm{def:g10:signals-and-waves:sound}{sonido}
en el aire, $c = \qty{3.00e8}{m/s}$,
$G = \qty{6.67e-11}{N.m^2/kg^2}$; la estrella tiene una masa
$M = \qty{2.0e30}{kg}$. Se admite, de la mecánica de las \omterm{def:g10:universal-gravitation:satellite}{órbitas}
circulares (\cref{ch:g12:satellites-kepler}): un planeta de masa $m$ en
una \omterm{def:g10:universal-gravitation:satellite}{órbita} circular de radio $r$ tiene un \omterm{def:g10:signals-and-waves:period}{periodo} y una velocidad dados
por $r^3 = \frac{G M P^2}{4\pi^2}$ y $V = \frac{2\pi r}{P}$, y la
estrella, que gira en torno al \omterm{def:g11:forces-and-motion:com}{centro de masas} común, se bambolea a la
velocidad $K = \frac{m}{M}\,V$.

\medskip
\noindent\textbf{Parte I --- Viernes: el paso a nivel.}
Un tren pasa a velocidad constante con la bocina puesta; un móvil
registra $f_1 = \qty{471}{Hz}$ mientras se acerca y
$f_2 = \qty{411}{Hz}$ después de pasar.
\begin{enumerate}
  \item Explica, con la imagen de los \omterm{def:g12:mechanical-waves:wavefront}{frentes de onda} apiñados, por qué
        $f_1 > f_2$ aunque la bocina no cambie nunca.
  \item Escribe las dos ecuaciones que ligan $f_1$ y $f_2$ con la
        \omterm{def:g10:signals-and-waves:frequency}{frecuencia} verdadera $f$ y con la velocidad $u$ del tren.
  \item Demuestra que $u = v\,\dfrac{f_1 - f_2}{f_1 + f_2}$ y calcúlala
        en \unit{m/s} y en \unit{km/h}.
  \item Demuestra que $f = \dfrac{2 f_1 f_2}{f_1 + f_2}$ y calcúlala.
        ¿Por qué $f$ queda ligeramente \emph{por debajo} de la media de
        $f_1$ y $f_2$?
  \item Calcula las \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} por delante y por detrás del tren
        y la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} en reposo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Sábado: el \omterm{met:g10:signals-and-waves:echo}{radar}.}
Una patrulla de tráfico hace una demostración de su \omterm{met:g10:signals-and-waves:echo}{radar}:
$f = \qty{24.125}{GHz}$, y el aparato lee el batido entre la \omterm{def:g10:signals-and-waves:wave}{onda}
saliente y el \omterm{rem:g10:signals-and-waves:honest}{eco}.
\begin{enumerate}[resume]
  \item Explica por qué el \omterm{rem:g10:signals-and-waves:honest}{eco} de un coche en movimiento está desplazado
        \emph{dos veces} --- nombra el papel que hace el coche en cada
        desplazamiento.
  \item Demuestra que, para un coche a velocidad $u \ll c$, el batido
        vale $\Delta f \approx \dfrac{2u}{c}\,f$.
  \item Un coche devuelve $\Delta f = \qty{3.55}{kHz}$: calcula $u$.
  \item Conviértelo a \unit{km/h} y compáralo con el límite de
        \qty{80}{km/h} que hay señalizado allí.
  \item ¿Cuántos \omterm{def:g10:signals-and-waves:frequency}{hercios} de batido corresponden a \qty{1}{km/h}?
        Coméntalo: ¿qué hace fácil de medir un desplazamiento relativo
        tan diminuto de una \omterm{def:g10:signals-and-waves:wave}{onda} de \qty{24}{GHz}?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Domingo: la estrella que se bambolea.}
El club descarga la \omterm{def:g12:doppler-effect:radial}{velocidad radial} medida de una estrella parecida al
Sol, noche tras noche, representada abajo.

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis, width=8.8cm, height=4.6cm,
    xlabel={$t$ (días)}, ylabel={$v_r$ (\unit{m/s})},
    xmin=0, xmax=8.4, ymin=-72, ymax=72,
    xtick={0,2.1,4.2,6.3,8.4},
    xticklabels={$0$,$2.1$,$4.2$,$6.3$,$8.4$}, ytick={-55,0,55}]
  \addplot[dashed, omExo, domain=0:8.4, samples=2] {55};
  \addplot[dashed, omExo, domain=0:8.4, samples=2] {-55};
  \addplot[omDef, thick, domain=0:8.4, samples=140] {55*sin(360*x/4.2)};
\end{axis}
\end{tikzpicture}

{\small La \omterm{def:g12:doppler-effect:radial}{velocidad radial} de la estrella a lo largo de doce noches:
una sinusoide limpia --- la firma de un cuerpo en \omterm{def:g10:universal-gravitation:satellite}{órbita} circular vista
de canto.}
\end{omfigure}

\begin{enumerate}[resume]
  \item ¿Qué es lo que se mide realmente, noche tras noche, en el
        \emph{\omterm{def:g12:sound-acoustics:timbre}{espectro}} de la estrella? Enuncia la relación que se usa
        para convertirlo en $v_r$.
  \item Lee en la curva la \omterm{def:g10:signals-and-waves:amplitude}{amplitud} $K$ y el \omterm{def:g10:signals-and-waves:period}{periodo} $P$ del bamboleo.
  \item Calcula la oscilación $\Delta\lambda$ correspondiente de una
        línea de \qty{550}{nm}, y $\Delta\lambda/\lambda$. Compáralo con
        la Parte I: ¿cuánto más difícil es esta medida que la del tren?
  \item ¿Por qué hace que la estrella se bambolee un planeta invisible?
        ¿Qué indica la forma sinusoidal sobre la \omterm{def:g10:universal-gravitation:satellite}{órbita}?
  \item ¿Por qué mide este método solo la parte \emph{radial} del
        movimiento de la estrella, y qué se vería si la \omterm{def:g10:universal-gravitation:satellite}{órbita} se nos
        presentara de frente?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Domingo por la noche: pesar lo
invisible.}
Tomamos la \omterm{def:g10:universal-gravitation:satellite}{órbita} circular y vista de canto, de modo que $K$ es la
velocidad orbital completa de la estrella.
\begin{enumerate}[resume]
  \item A partir de $P$ y de la relación admitida
        $r^3 = G M P^2/(4\pi^2)$, calcula el radio orbital $r$ del
        planeta. Compáralo con la distancia Tierra--Sol,
        \qty{1.5e11}{m}.
  \item Calcula la velocidad orbital $V$ del planeta.
  \item A partir de $K = (m/M)\,V$, calcula la masa $m$ del planeta.
  \item Compara $m$ con Júpiter ($\qty{1.9e27}{kg}$) y con la Tierra
        ($\qty{6.0e24}{kg}$), y $r$ con la distancia Sol--Mercurio
        ($\qty{5.8e10}{m}$). ¿Qué clase de mundo es este?
  \item Si la \omterm{def:g10:universal-gravitation:satellite}{órbita} estuviera inclinada en lugar de verse de canto,
        ¿sería la masa verdadera mayor o menor que tu valor? Concluye en
        una frase: ¿qué acaba de entregar un desplazamiento periódico de
        una parte en cinco millones?
\end{enumerate}
\end{problem}
