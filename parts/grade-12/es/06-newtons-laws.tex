\chapter{Las leyes de Newton}\label{ch:g12:newtons-laws}

Cuela una báscula de baño en un ascensor. Cuando la cabina arranca hacia arriba, la
aguja marca tres \omterm{def:g10:orders-of-magnitude:unit}{kilogramos} de más; cerca del último piso marca de menos; entre ambas
cosas, la pura verdad --- a bordo no ha cambiado nada salvo el movimiento. Este
capítulo enuncia las tres leyes que hay detrás de esa aguja y las forja en el método
que gobierna toda la mecánica: elegir un sistema, dibujar sus \omterm{def:g10:inertia:force}{fuerzas}, proyectar
$\sum \vect F = m\vect a$.

\section{La cantidad de movimiento y la primera ley, hecha exacta}

\begin{definition}[Cantidad de movimiento]\label{def:g12:newtons-laws:momentum}
La \emph{cantidad de movimiento}\index{cantidad de movimiento} de un cuerpo de masa
$m$ (\unit{kg}) que se mueve con velocidad $\vect v$ es el vector
\[
\vect p = m \vect v \qquad (\unit{kg.m/s}):
\]
la velocidad, ponderada por la masa; para un sistema, la suma vectorial sobre sus
partes.
\end{definition}

\begin{definition}[Sistema de referencia inercial]\label{def:g12:newtons-laws:inertial}
Un \emph{sistema de referencia inercial}\index{sistema de referencia inercial} es
aquel en el que todo cuerpo aislado --- sin \omterm{def:g10:inertia:force}{fuerzas}, o con \omterm{def:g10:inertia:compensate}{fuerzas que se compensan}
--- se mueve en línea recta con velocidad constante (\cref{ch:g10:inertia}). El suelo
lo es con excelente aproximación, y también cualquier cabina en \omterm{def:g12:kinematics-2d:uniform}{movimiento rectilíneo
uniforme} sobre él; un tren que frena o un tiovivo que gira, no: allí el equipaje
suelto acelera sin que nadie lo \omterm{def:g10:inertia:inventory}{empuje}.
\end{definition}

\begin{theorem}[Primera ley de Newton]\label{thm:g12:newtons-laws:first}\index{primera ley de Newton}
En un sistema inercial, la \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento} de un sistema aislado no cambia:
$\vect p$ es constante, es decir, $d\vect p/dt = \vect 0$.
\end{theorem}
\admitted

\begin{remark}[La inercia, afinada dos veces]\label{rem:g12:newtons-laws:sharpened}
El \omterm{thm:g10:inertia:principle}{principio de inercia} del \cref{ch:g10:inertia}, mejorado dos veces: dice
\emph{dónde} vale --- define los sistemas inerciales como aquellos en los que vale ---
y protege un vector, $m\vect v$, cuya contabilidad rinde sus frutos al final del
capítulo.
\end{remark}

\section{La segunda ley de Newton}

\begin{theorem}[Segunda ley de Newton]\label{thm:g12:newtons-laws:second}\index{segunda ley de Newton}
En un sistema inercial, la \omterm{def:g10:inertia:force}{fuerza} resultante sobre un cuerpo es igual a la rapidez de
variación de su \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento}; con masa constante, como $\vect p = m\vect v$,
\[
\sum \vect F = \frac{d\vect p}{dt} = m\,\frac{d\vect v}{dt} = m \vect a .
\]
\end{theorem}
\admitted

\begin{remark}[Una ley de la naturaleza, y su unidad]\label{rem:g12:newtons-laws:nature}
No hay demostración: es una ley de la naturaleza, contrastada con el experimento
durante tres siglos. El curso pasado (\cref{ch:g11:forces-and-motion}) dio su sombra,
$\Delta v \propto F\,\Delta t/m$; la derivada (\cref{ch:g12:kinematics-2d}) la vuelve
exacta en cada instante --- su letra pequeña espera en el volumen del primer año de
universidad. La \omterm{def:g10:inertia:force}{fuerza} recibe una \omterm{def:g10:orders-of-magnitude:unit}{unidad}, $\unit{N} = \unit{kg.m/s^2}$; y un cuerpo
sometido únicamente a su \omterm{def:g10:universal-gravitation:weight}{peso} obedece $m\vect a = m\vect g$: toda masa cae con
$\vect a = \vect g$, y $g = \qty{9.81}{N/kg} = \qty{9.81}{m/s^2}$ --- las dos lecturas
son una sola.
\end{remark}

\begin{example}[Fuerza media, sin necesidad de detalles]\label{ex:g12:newtons-laws:average}
Un coche de \qty{1300}{kg} pasa del reposo a \qty{100}{km/h} (\qty{27.8}{m/s}) en
\qty{8.0}{s}: sea cual sea la historia instante a instante, la \omterm{def:g10:inertia:force}{fuerza} resultante media
vale $\Delta p/\Delta t = 1300 \times 27.8/8.0 \approx \qty{4.5e3}{N}$ --- la \omterm{def:g12:newtons-laws:momentum}{cantidad
de movimiento} convierte una pregunta sobre \omterm{def:g10:inertia:force}{fuerzas} en una contabilidad de antes y
después.
\end{example}

\section{La tercera ley de Newton}

\begin{theorem}[Tercera ley de Newton]\label{thm:g12:newtons-laws:third}\index{tercera ley de Newton}
Si un cuerpo $A$ ejerce una \omterm{def:g10:inertia:force}{fuerza} $\vect F_{A \to B}$ sobre un cuerpo $B$, entonces
$B$ ejerce sobre $A$ la \omterm{def:g10:inertia:force}{fuerza} $\vect F_{B \to A} = -\vect F_{A \to B}$: misma línea de
acción, mismo módulo, sentidos opuestos --- en todo instante, sea cual sea el
movimiento, por contacto o a distancia.
\end{theorem}
\admitted

\begin{example}[El libro, la mesa y la Tierra]\label{ex:g12:newtons-laws:book}
Un libro reposa sobre una mesa: dos interacciones, dos pares. Gravitatoria: la Tierra
tira del libro hacia abajo ($\vect P$), el libro tira de la Tierra hacia arriba. De
contacto: la mesa empuja el libro hacia arriba ($\vect N$), el libro aprieta la mesa
hacia abajo. Andar es el mismo trato: empujas el suelo hacia atrás; su reacción es la
única \omterm{def:g10:inertia:force}{fuerza} que te impulsa hacia adelante.
\end{example}

\section{El método: elegir, dibujar, proyectar}

\begin{method}[Resolver un problema de dinámica]\label{met:g12:newtons-laws:method}
\begin{enumerate}
  \item \emph{Sistema}: nombra el cuerpo (o el conjunto de cuerpos) que se estudia.
  \item \emph{Referencia}: elige un sistema inercial y unos ejes adaptados al
        movimiento --- ejes del plano inclinado, o la \omterm{def:g12:kinematics-2d:frenet}{base de Frenet} para una curva
        (\cref{ch:g12:kinematics-2d}).
  \item \emph{Inventario}: enumera todas las \omterm{def:g10:inertia:force}{fuerzas} --- el \omterm{def:g10:universal-gravitation:weight}{peso}, luego una \omterm{def:g10:inertia:force}{fuerza} por
        contacto, nada más --- y dibuja el \omterm{def:g10:inertia:diagram}{diagrama de fuerzas}: un punto, una flecha
        por \omterm{def:g10:inertia:force}{fuerza}.
  \item \emph{Proyectar}: escribe $\sum \vect F = m\vect a$ y proyéctala sobre cada
        eje.
  \item \emph{Resolver y comprobar}: \omterm{def:g10:orders-of-magnitude:unit}{unidades}, signos, casos límite ($\mu \to 0$,
        $\alpha \to 0$, \dots).
\end{enumerate}
\end{method}

\begin{definition}[Reacción normal y rozamiento]\label{def:g12:newtons-laws:friction}
Una superficie actúa sobre un cuerpo mediante dos componentes: la \emph{reacción
normal}\index{reacción normal} $\vect N$, perpendicular a la superficie, y el
\omterm{def:g10:inertia:inventory}{rozamiento} $\vect f$, tangencial. La experiencia da las leyes del \omterm{def:g10:inertia:inventory}{rozamiento}: mientras
el cuerpo no desliza, el \omterm{def:g11:forces-and-motion:friction}{rozamiento estático} se ajusta solo hasta un techo,
$f_s \leq \mu_s N$; una vez que desliza, el \omterm{def:g11:forces-and-motion:friction}{rozamiento cinético} vale $f_k = \mu_k N$,
opuesto al deslizamiento. Los \emph{coeficientes de \omterm{def:g10:inertia:inventory}{rozamiento}}%
\index{coeficiente de rozamiento} $\mu_s$ y $\mu_k$ ($\mu_k \leq \mu_s$) dependen solo
de los dos materiales.
\end{definition}

\begin{example}[El plano inclinado, dos veces]\label{ex:g12:newtons-laws:incline}
Un bloque desliza por una pendiente de ángulo $\alpha$; ejes: $x$ pendiente abajo, $y$
perpendicular. Sobre $y$: $N = mg\cos\alpha$. Sobre $x$, sin \omterm{def:g10:inertia:inventory}{rozamiento}:
$ma = mg\sin\alpha$, luego $a = g\sin\alpha$ --- sin masa; con $\alpha = 30^\circ$,
$a = \qty{4.9}{m/s^2}$. Con \omterm{def:g11:forces-and-motion:friction}{rozamiento cinético}:
$a = g(\sin\alpha - \mu_k\cos\alpha)$; para $\mu_k = 0.20$,
$a = 9.81\,(0.500 - 0.173) = \qty{3.2}{m/s^2}$. Y de entrada se quedó quieto solo
mientras el \omterm{def:g11:forces-and-motion:friction}{rozamiento estático} necesario cupo bajo su techo:
$\tan\alpha \leq \mu_s$.
\end{example}

\begin{definition}[Peso aparente]\label{def:g12:newtons-laws:apparent}
El \emph{peso aparente}\index{peso aparente} de un cuerpo es la \omterm{def:g12:newtons-laws:friction}{reacción normal} entre
él y su soporte --- lo que marca una báscula colocada debajo.
\end{definition}

\begin{example}[El ascensor]\label{ex:g12:newtons-laws:elevator}
Un pasajero de \qty{70}{kg} está de pie sobre una báscula en un ascensor de
aceleración vertical $a_z$ (positiva hacia arriba): $N - mg = ma_z$, luego
$N = m(g + a_z)$. Arrancando hacia arriba con $a_z = \qty{1.5}{m/s^2}$: \qty{792}{N},
pesado; frenando cerca del último piso: \qty{582}{N}, ligero; en marcha uniforme:
$mg = \qty{687}{N}$ --- la primera ley; en caída libre, $a_z = -g$: $N = 0$, la
ingravidez sobre una báscula.
\end{example}

\begin{example}[Dos bloques y una polea]\label{ex:g12:newtons-laws:pulley}
Un bloque $m_1 = \qty{4.0}{kg}$ sobre una mesa sin \omterm{def:g10:inertia:inventory}{rozamiento} está unido por una
cuerda ligera, a través de una polea ideal, a un bloque colgante
$m_2 = \qty{1.0}{kg}$. La cuerda transmite la misma tensión $T$ en ambos extremos; los
bloques comparten un único módulo de la aceleración $a$. Una segunda ley para cada
uno, a lo largo de su movimiento: $m_1 a = T$ y $m_2 a = m_2 g - T$, de donde
$a = m_2\,g/(m_1 + m_2) = \qty{1.96}{m/s^2}$ y $T = m_1 a = \qty{7.8}{N}$ --- menor
que $m_2 g = \qty{9.8}{N}$, como tiene que ser para que $m_2$ acelere hacia abajo.
\end{example}

\begin{example}[La curva sin peralte]\label{ex:g12:newtons-laws:curve}
Un coche de masa $m$ toma una curva sin peralte de radio $R$ a rapidez constante $v$.
En la \omterm{def:g12:kinematics-2d:frenet}{base de Frenet} (\cref{ch:g12:kinematics-2d}) la aceleración es puramente normal,
$a_N = v^2/R$, dirigida al centro. En vertical, $N = mg$; en horizontal, la única
\omterm{def:g10:inertia:force}{fuerza} centrípeta disponible es el \omterm{def:g11:forces-and-motion:friction}{rozamiento estático} de la calzada sobre los
neumáticos: $f = mv^2/R \leq \mu_s N = \mu_s mg$, luego $v \leq \sqrt{\mu_s\, g R}$,
sea cual sea la masa. Para $R = \qty{90}{m}$ y calzada seca ($\mu_s = 0.70$):
$v_{\max} \approx \qty{25}{m/s}$ (\qty{89}{km/h}); sobre hielo el techo se desploma
--- el \cref{exo:g12:newtons-laws:13} peralta la carretera y el
\cref{exo:g12:newtons-laws:10} hace oscilar la misma proyección en un hilo.
\end{example}

\section{Sistemas aislados: la cantidad de movimiento se conserva}

\begin{proposition}[Conservación de la cantidad de movimiento]\label{prop:g12:newtons-laws:conservation}\index{conservación de la cantidad de movimiento}
Para dos cuerpos cuyas \omterm{def:g10:inertia:force}{fuerzas} exteriores son nulas o \omterm{def:g10:inertia:compensate}{se compensan}, la \omterm{def:g12:newtons-laws:momentum}{cantidad de
movimiento} total $\vect p_1 + \vect p_2$ es constante, sean cuales sean las \omterm{def:g10:inertia:force}{fuerzas}
que ejercen el uno sobre el otro.
\end{proposition}

\begin{proof}
Sumemos las dos segundas leyes:
$\frac{d}{dt}(\vect p_1 + \vect p_2) = \vect F_{2 \to 1} + \vect F_{1 \to 2} +
\vect F_{\text{ext}}$: el par interno se cancela por la tercera ley, y
$\vect F_{\text{ext}} = \vect 0$ por hipótesis.
\end{proof}

\begin{example}[El retroceso]\label{ex:g12:newtons-laws:recoil}
Una patinadora de \qty{60}{kg} en reposo sobre hielo liso lanza horizontalmente un
balón medicinal de \qty{4.0}{kg} a \qty{6.0}{m/s}. Antes: $\vect p = \vect 0$;
después: $0 = 60\,V + 4.0 \times 6.0$, luego $V = \qty{-0.40}{m/s}$ --- se desliza
hacia atrás. Nada exterior la empujó: fue el balón (tercera ley), y las cuentas siguen
dando cero.
\end{example}

\begin{remark}[Cómo empuja un cohete sobre nada]\label{rem:g12:newtons-laws:rocket}
Un cohete es una máquina de lanzar masa. Cada segundo sus motores arrojan hacia atrás
una tanda de gas; el gas se lleva \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento} hacia atrás, así que el
cohete gana la misma cantidad hacia adelante --- un \omterm{def:g10:inertia:inventory}{empuje} sostenido, aproximadamente
la masa expulsada por segundo multiplicada por la velocidad de eyección. El cohete
empuja sobre sus propios gases, no sobre el aire: funciona mejor en el vacío.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:1}
Calcula la \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento} de (a) un velocista de \qty{70}{kg} a \qty{10}{m/s};
(b) un coche de \qty{1300}{kg} a \qty{50}{km/h}; (c) una bala de \qty{8.0}{g} a
\qty{800}{m/s}. Ordénalas. ¿Por qué la más rápida queda la última?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:2}
Un pasajero de \qty{70}{kg} está de pie sobre una báscula en un ascensor. ¿Qué marca
cuando la cabina (a) acelera hacia arriba a \qty{1.2}{m/s^2}; (b) sube a
\qty{2.0}{m/s} constantes; (c) acelera hacia abajo a \qty{1.2}{m/s^2}? ¿Qué ley
resuelve el caso (b)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:3}
Se suelta un bloque en una pendiente sin \omterm{def:g10:inertia:inventory}{rozamiento} de ángulo $30^\circ$. Calcula su
aceleración y después su rapidez tras deslizar \qty{3.0}{m}. ¿Dónde se ha ido la masa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:4}
Una pelota de tenis de \qty{58}{g} llega a \qty{15}{m/s} y es devuelta a lo largo de
la misma línea a \qty{25}{m/s}; el contacto dura \qty{5.0}{ms}. Calcula la variación
de la \omterm{def:g12:newtons-laws:momentum}{cantidad de movimiento} y la \omterm{def:g10:inertia:force}{fuerza} media de las cuerdas; compara esa \omterm{def:g10:inertia:force}{fuerza} con
el \omterm{def:g10:universal-gravitation:weight}{peso} de la pelota.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:5}
Un libro reposa sobre una mesa. Nombra las dos \omterm{def:g10:inertia:force}{fuerzas} que actúan sobre el libro y,
para cada una, su compañera de acción--reacción (¿sobre qué cuerpo? ¿en qué sentido?).
¿Por qué el \omterm{def:g10:universal-gravitation:weight}{peso} del libro y el \omterm{def:g10:inertia:inventory}{empuje} de la mesa no son compañeras entre sí?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:6}
Una caja de \qty{50}{kg} descansa sobre un suelo horizontal, $\mu_s = 0.45$,
$\mu_k = 0.35$. (a) ¿Qué \omterm{def:g10:inertia:inventory}{empuje} horizontal la pone justo en marcha? (b) Si se mantiene
ese \omterm{def:g10:inertia:inventory}{empuje}, ¿cuál es su aceleración? (c) ¿Qué \omterm{def:g10:inertia:inventory}{empuje} la mantiene a velocidad
constante?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:7}
La misma caja se coloca sobre una rampa de inclinación regulable ($\mu_s = 0.45$,
$\mu_k = 0.35$). (a) ¿A qué ángulo empieza a deslizar? (b) A $30^\circ$, calcula su
aceleración. (c) Una vez iniciado el movimiento al ángulo del apartado (a), ¿sigue
acelerando? ¿Por qué?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:8}
Un bloque de \qty{3.0}{kg} sobre una mesa sin \omterm{def:g10:inertia:inventory}{rozamiento} está unido, a través de una
polea ideal, a un bloque colgante de \qty{2.0}{kg}. Calcula la aceleración y la
tensión. ¿Por qué la tensión tenía que salir menor que el \omterm{def:g10:universal-gravitation:weight}{peso} colgante?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:9}
Una curva de radio \qty{50}{m} no tiene peralte. Calcula la rapidez máxima de un coche
para $\mu_s = 0.70$ (seco) y $0.40$ (mojado), en \unit{km/h}. ¿Por qué la respuesta
vale igual para una motocicleta que para un camión?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:10}

circunferencia horizontal, manteniendo el hilo un ángulo constante de $25^\circ$ con

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:11}

báscula (\unit{N}) & 54.0 & 49.0 & 44.1
\end{tabular}\end{center}
Calcula la aceleración en cada fase y describe un trayecto posible. ¿Podría el

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:12}

¿cuál es su rapidez de retroceso? (b) Los \qty{8.0}{m/s} se miden ahora \emph{respecto

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:13}

\section{Problema: El teleférico}

\begin{problem}[{Problema de fin de semana --- certificar el teleférico: la tensión que
la pendiente exige en reposo, el recargo del arranque, el alivio sobre el pilono, el
precio de una parada de emergencia --- y la única cifra que el cartel de la cabina
puede prometer}]
\label{pb:g12:newtons-laws:1}
Un teleférico de montaña --- cabina más carro, masa en vacío
$M_0 = \qty{2.0e3}{kg}$ --- rueda sobre un cable portador inclinado
$\alpha = 30^\circ$, arrastrado por un cable tractor paralelo a la pendiente con
tensión $\vect T$. El \omterm{def:g10:inertia:inventory}{rozamiento} de rodadura es despreciable; el suelo de la cabina se
mantiene horizontal en todo momento. La cabina admite $25$ pasajeros de masa media
\qty{80}{kg}; tú eres la ingeniera encargada de la certificación. Salvo indicación
contraria, la cabina va llena: $M = \qty{4.0e3}{kg}$.

\medskip
\noindent\textbf{Parte I --- En reposo sobre la pendiente.}
\begin{enumerate}
  \item Elige el sistema; explica por qué el \omterm{def:g10:relative-motion:frame}{sistema de referencia} de la montaña sirve
        como sistema inercial; inventaría las tres \omterm{def:g10:inertia:force}{fuerzas}, con sus direcciones.
  \item ¿Qué ley gobierna la espera? Proyéctala a lo largo de la pendiente y
        perpendicularmente a ella para expresar $T$ y la \omterm{def:g12:newtons-laws:friction}{reacción normal} $N$ del cable
        portador.
  \item Calcula $T$ y $N$.
  \item Recalcula $T$ para la cabina vacía. ¿Por qué $T$ es proporcional a la masa
        total?
  \item El cable tractor se rompe y los frenos fallan: demuestra que la aceleración de
        la caída es $g \sin\alpha$, con carga o sin ella, y calcúlala.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El arranque.} El torno acelera la cabina llena a
$a = \qty{0.60}{m/s^2}$ pendiente arriba hasta alcanzar la marcha de crucero
$v = \qty{6.0}{m/s}$.
\begin{enumerate}[resume]
  \item Expresa y calcula la tensión $T'$ durante esta fase.
  \item ¿En qué fracción eleva el arranque la tensión por encima de su valor en
        reposo?
  \item ¿Cuánto dura la fase y a lo largo de qué distancia?
  \item Una pasajera de \qty{80}{kg} va de pie sobre el suelo horizontal. Proyecta la
        segunda ley en vertical para expresar y calcular su \omterm{def:g12:newtons-laws:apparent}{peso aparente}; compáralo
        con $mg$.
  \item ¿Qué \omterm{def:g10:inertia:force}{fuerza} la acelera horizontalmente y cuánto vale? ¿Por qué se inclinan los
        pasajeros?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Sobre el pilono.} A \qty{6.0}{m/s} constantes, el cable
portador se curva sobre un pilono de apoyo siguiendo un arco de circunferencia
vertical de radio $R = \qty{40}{m}$; en la cresta la velocidad es horizontal.
\begin{enumerate}[resume]
  \item En la \omterm{def:g12:kinematics-2d:frenet}{base de Frenet} en la cresta (\cref{ch:g12:kinematics-2d}), da las dos
        componentes de la aceleración, con sus valores.
  \item Proyecta la segunda ley sobre la normal para expresar y calcular la \omterm{def:g12:newtons-laws:friction}{reacción
        normal} $N'$ del cable sobre el carro.
  \item La misma pregunta para el \omterm{def:g12:newtons-laws:apparent}{peso aparente} de la pasajera en la cresta: ¿en qué
        fracción es más ligera?
  \item ¿A qué rapidez se anularía $N'$? Compárala con la rapidez de crucero y
        comenta.
  \item ¿Qué \omterm{def:g10:inertia:force}{fuerza} ejerce el carro sobre el cable al pasar el pilono --- qué ley lo
        dice, con qué módulo y qué sentido?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Frenada, un salto y el cartel.}
\begin{enumerate}[resume]
  \item En el tramo horizontal de llegada, una parada de emergencia lleva la cabina
        llena de \qty{6.0}{m/s} al reposo en \qty{1.5}{s}: lee
        $\sum \vect F = d\vect p/dt$ como un promedio sobre la parada para calcular la
        \omterm{def:g10:inertia:force}{fuerza} de frenado.
  \item ¿Qué \omterm{def:g10:inertia:force}{fuerza} horizontal debe detener a una pasajera de \qty{80}{kg}, y puede
        proporcionarla el \omterm{def:g11:forces-and-motion:friction}{rozamiento estático} entre el calzado y el suelo
        ($\mu_s = 0.40$)? Concluye.
  \item En el andén, con los frenos sueltos, la cabina vacía cuelga libre e inmóvil; el
        último pasajero (\qty{80}{kg}) salta horizontalmente a \qty{2.5}{m/s}
        \emph{respecto de la cabina}. Calcula la rapidez de retroceso de la cabina y la
        rapidez del pasajero respecto del suelo.
  \item En una frase: ¿qué dispositivo de propulsión vive de este principio de
        retroceso, y qué fija su \omterm{def:g10:inertia:inventory}{empuje}?
  \item \emph{Certificación.} El cable tractor rompe a \qty{100}{kN}; la normativa
        limita la tensión de servicio a la quinta parte de ese valor. Demuestra que el
        peor caso encontrado más arriba es el arranque, deduce la masa total máxima e
        imprime el cartel: ¿cuántos pasajeros de \qty{80}{kg} pueden viajar? Comprueba
        el coeficiente de seguridad.
\end{enumerate}
\end{problem}
