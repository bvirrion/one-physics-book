\chapter{El sonido y la acústica}\label{ch:g12:sound-acoustics}

Pulsa la cuerda grave de una guitarra: la habitación se llena de un la
grave y, sin embargo, hasta tu oído no ha viajado más que aire
comprimido y estirado. Este capítulo sigue ese patrón --- su velocidad,
cómo la \omterm{def:g10:signals-and-waves:frequency}{frecuencia} se convierte en \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} y la \omterm{def:g11:work-of-force:power}{potencia} en \omterm{def:g12:sound-acoustics:pitch-loudness}{sonoridad}, y
por qué una misma nota suena distinta en cada instrumento.

\section{El sonido es una onda de presión}

\begin{definition}[Onda sonora]\label{def:g12:sound-acoustics:sound-wave}
Una \emph{onda sonora}\index{onda sonora} es una \omterm{def:g12:mechanical-waves:wave}{onda mecánica} de
\omterm{def:g10:pressure:pressure}{presión} (\cref{ch:g12:mechanical-waves}): una superficie que vibra crea
\emph{compresiones}\index{compresión} (una ligera sobrepresión) y
\emph{rarefacciones}\index{rarefacción} (una depresión) que viajan. Las
porciones de aire oscilan \emph{a lo largo} de la dirección de
propagación (el \omterm{def:g10:signals-and-waves:sound}{sonido} es \emph{longitudinal}\index{onda longitudinal})
y hace falta un medio: una campana bajo una campana de vacío enmudece
(\cref{ch:g10:signals-and-waves}).
\end{definition}

\begin{proposition}[Velocidad del sonido]\label{prop:g12:sound-acoustics:speed}
\index{velocidad del sonido}%
El \omterm{def:g10:signals-and-waves:sound}{sonido} viaja a unos \qty{340}{m/s} en el aire a
\qty{20}{\degreeCelsius}, a \qty{1500}{m/s} en el agua y a
\qty{5000}{m/s} en el acero: cuanto más rígido es el medio, más rápido.
\admitted
\end{proposition}

\begin{remark}[El aire caliente es más rápido]\label{rem:g12:sound-acoustics:temperature}
Son valores medidos; predecirlos se hace en el volumen del primer año.
El \omterm{def:g10:signals-and-waves:sound}{sonido} se acelera en aire más caliente (\qty{331}{m/s} a
\qty{0}{\degreeCelsius}) --- de ahí que los instrumentos de viento se
desafinen a medida que la sala se calienta.
\end{remark}

\begin{example}[Los tamaños de los sonidos]\label{ex:g12:sound-acoustics:wavelengths}
La relación $\lambda = v/f$ le pone tamaño a cualquier \omterm{def:g10:signals-and-waves:sound}{sonido}: en el
aire, un retumbo de \qty{20}{Hz} abarca $340/20 = \qty{17}{m}$ --- un
edificio --- y un siseo de \qty{20}{kHz} cabe en \qty{1.7}{cm}; los del
tamaño de un \omterm{def:g10:orders-of-magnitude:unit}{metro} se doblan con facilidad al rodear puertas y
esquinas.
\end{example}

\section{Tono, sonoridad y el oído}

\begin{definition}[Tono y sonoridad]\label{def:g12:sound-acoustics:pitch-loudness}
El \emph{tono}\index{tono} es la \omterm{def:g11:color-light-sources:perception}{percepción} de la \omterm{def:g10:signals-and-waves:frequency}{frecuencia}: cuanto
mayor es $f$, más aguda es la nota; el la de afinación es de
\qty{440}{Hz}, y duplicar $f$ sube la nota una octava. La
\emph{sonoridad}\index{sonoridad} es la \omterm{def:g11:color-light-sources:perception}{percepción} de la \omterm{def:g10:signals-and-waves:amplitude}{amplitud}:
cuanto mayor es la oscilación de \omterm{def:g10:pressure:pressure}{presión}, más fuerte suena; amplificar
cambia la sonoridad, no el tono.
\end{definition}

\begin{remark}[La ventana del oído]\label{rem:g12:sound-acoustics:audible}
La audición abarca de unos \qty{20}{Hz} a \qty{20}{kHz}
(\cref{ch:g10:signals-and-waves}); por debajo están los \omterm{def:g10:signals-and-waves:ultrasound}{infrasonidos} y
por encima, los \omterm{def:g10:signals-and-waves:ultrasound}{ultrasonidos} --- los murciélagos a \qty{50}{kHz} y las
sondas médicas en megahercios. Un piano solo toma una muestra del
centro: \qtyrange{27.5}{4186}{Hz}.
\end{remark}

\section{La intensidad sonora y la escala de decibelios}

\begin{definition}[Intensidad sonora]\label{def:g12:sound-acoustics:intensity}
La \emph{intensidad sonora}\index{intensidad sonora} $I$ es la \omterm{def:g11:work-of-force:power}{potencia}
acústica que atraviesa un \omterm{def:g10:orders-of-magnitude:unit}{metro} cuadrado encarado a la \omterm{def:g10:signals-and-waves:wave}{onda}:
$I = P/S$, en \unit{W/m^2}. El \omterm{def:g10:signals-and-waves:sound}{sonido} audible más débil está en torno a
$I_0 = \qty{1.0e-12}{W/m^2}$; el dolor empieza cerca de
\qty{1}{W/m^2}.
\end{definition}

\begin{proposition}[Atenuación con la inversa del cuadrado]\label{prop:g12:sound-acoustics:inverse-square}
\index{ley de la inversa del cuadrado}%
Una fuente pequeña que radia una \omterm{def:g11:work-of-force:power}{potencia} $P$ por igual en todas
direcciones da, a la distancia $r$,
$I = P/(4\pi r^2) \propto 1/r^2$: duplicar la distancia divide la
intensidad entre cuatro.
\end{proposition}

\begin{proof}
Toda la \omterm{def:g11:work-of-force:power}{potencia} $P$ atraviesa la esfera de área $4\pi r^2$; basta
dividir.
\end{proof}

\begin{definition}[Nivel sonoro]\label{def:g12:sound-acoustics:level}
El oído abarca doce potencias de diez, así que las intensidades se dan
en escala logarítmica. El \emph{nivel sonoro}\index{nivel sonoro} de $I$
es
\[ L = 10 \log_{10}(I/I_0), \qquad I_0 = \qty{1.0e-12}{W/m^2}, \]
en \emph{decibelios}\index{decibelio} (\unit{dB}): umbral de audición
\qty{0}{dB}, dolor cerca de \qty{120}{dB}.
\end{definition}

\begin{proposition}[Aritmética de los decibelios]\label{prop:g12:sound-acoustics:db-arithmetic}
Multiplicar $I$ por $k$ añade $10\log_{10}k$ \omterm{def:g12:sound-acoustics:level}{decibelios}. En particular,
$\times 10$ añade \qty{10}{dB}; $\times 2$ añade
$10\log_{10}2 \approx \qty{3.0}{dB}$; $N$ fuentes idénticas e
incoherentes suman \emph{intensidades}, nunca niveles, y dan
$L_1 + 10\log_{10}N$; y duplicar la distancia a una fuente pequeña ($I$
dividida entre cuatro) quita unos \qty{6.0}{dB}.
\end{proposition}

\begin{proof}
$10\log_{10}(kI/I_0) = 10\log_{10}(I/I_0) + 10\log_{10}k$; para la
distancia, la \cref{prop:g12:sound-acoustics:inverse-square} con
$k = 1/4$.
\end{proof}

\begin{method}[Trabajar en decibelios]\label{met:g12:sound-acoustics:db}
\begin{enumerate}
  \item $L = 10\log_{10}(I/I_0)$; y al revés: $I = I_0 \times 10^{L/10}$.
  \item Nunca se suman niveles: hay que pasar a intensidades, sumar y
        volver --- o usar $+\qty{3}{dB}$ por cada duplicación y
        $+\qty{10}{dB}$ por cada factor diez.
  \item Cada duplicación de la distancia a una fuente pequeña cuesta
        \qty{6}{dB}.
\end{enumerate}
\end{method}

\begin{remark}[El daño auditivo]\label{rem:g12:sound-acoustics:damage}
La pérdida de audición es una \omterm{rem:g11:nucleus-radioactivity:dose}{dosis}: \qty{85}{dB} son seguros durante
ocho horas, y cada \qty{3}{dB} de más --- una duplicación de la
intensidad --- reduce a la mitad el tiempo seguro; las células ciliadas
perdidas no vuelven a crecer.
\end{remark}

\section{Timbre y armónicos}

\begin{definition}[Armónicos]\label{def:g12:sound-acoustics:harmonics}
Un \omterm{def:g10:signals-and-waves:sound}{sonido} periódico de \omterm{def:g10:signals-and-waves:frequency}{frecuencia} $f_1$ --- el
\emph{fundamental}\index{fundamental} --- es en general una suma de
sinusoides de \omterm{def:g10:signals-and-waves:frequency}{frecuencias} $f_n = n f_1$, con $n = 1, 2, 3, \dots$, sus
\emph{armónicos}\index{armónico}. Un \omterm{def:g10:signals-and-waves:sound}{sonido} reducido a su fundamental es
un \emph{sonido puro}\index{sonido puro}, como el de un diapasón.
\end{definition}

\begin{definition}[Espectro y timbre]\label{def:g12:sound-acoustics:timbre}
El \emph{espectro}\index{espectro} de un \omterm{def:g10:signals-and-waves:sound}{sonido} es el diagrama de barras
de sus \omterm{def:g12:sound-acoustics:harmonics}{armónicos}: una barra por cada $f_n$, cuya longitud mide la
intensidad de ese \omterm{def:g12:sound-acoustics:harmonics}{armónico}. El oído escucha $f_1$ como el \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} y la
\emph{mezcla} de intensidades como el \emph{timbre}\index{timbre} --- el
color del \omterm{def:g10:signals-and-waves:sound}{sonido}: dos instrumentos que tocan la misma nota comparten
$f_1$ y solo se diferencian en sus espectros.
\end{definition}

\begin{example}[Por qué reconoces quién llama]\label{ex:g12:sound-acoustics:two-instruments}
El la de una flauta es casi un \omterm{def:g12:sound-acoustics:harmonics}{sonido puro}; un violín que frote esos
mismos \qty{220}{Hz} añade \omterm{def:g12:sound-acoustics:harmonics}{armónicos} intensos en
\qtylist{440;660;880}{Hz}. El mismo \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} y la misma \omterm{def:g12:sound-acoustics:pitch-loudness}{sonoridad} y, sin
embargo, nadie los confunde: una voz también es un \omterm{def:g12:sound-acoustics:timbre}{espectro} que te sabes
de memoria.
\end{example}

\section{La cuerda que vibra}

\begin{definition}[Onda estacionaria]\label{def:g12:sound-acoustics:standing-wave}
Una \omterm{def:g10:signals-and-waves:wave}{onda} confinada entre los extremos fijos de una cuerda puede
asentarse en una \emph{onda estacionaria}\index{onda estacionaria}: una
oscilación en el sitio, con puntos inmóviles --- los
\emph{nodos}\index{nodo} --- alternando con puntos de oscilación máxima
--- los \emph{vientres}\index{vientre}. Ya no viaja nada; la cuerda
vibra con un patrón fijo.
\end{definition}

\begin{proposition}[Modos de una cuerda]\label{prop:g12:sound-acoustics:string-modes}
Una cuerda de longitud $L$, fija por los dos extremos y que transporta
\omterm{def:g10:signals-and-waves:wave}{ondas} de velocidad $v$, vibra de forma estable solo con patrones en los
que cabe un número entero $n$ de semilongitudes de \omterm{def:g10:signals-and-waves:wave}{onda} entre los \omterm{def:g12:sound-acoustics:standing-wave}{nodos}
impuestos por los extremos:
\[ \lambda_n = \frac{2L}{n}, \quad f_n = n\,\frac{v}{2L} = n f_1, \quad n = 1, 2, 3, \dots \]
El modo más bajo, $f_1 = v/(2L)$, es el \omterm{def:g12:sound-acoustics:harmonics}{fundamental}; los demás son
exactamente sus \omterm{def:g12:sound-acoustics:harmonics}{armónicos}. \admitted
\end{proposition}

\begin{remark}[Dónde vive la demostración honrada]\label{rem:g12:sound-acoustics:honest}
Que las \omterm{def:g10:signals-and-waves:wave}{ondas} confinadas elijan exactamente estos modos se sigue de
superponer la \omterm{def:g10:signals-and-waves:wave}{onda} con sus reflexiones, un cálculo que se hace en el
volumen del primer año. La geometría convence: los dos extremos tienen
que quedarse quietos, así que ha de caber un número entero de
semilongitudes de \omterm{def:g10:signals-and-waves:wave}{onda}.
\end{remark}

\begin{example}[La cuerda la de un violín]\label{ex:g12:sound-acoustics:violin}
La cuerda la de un violín mide $L = \qty{32.8}{cm}$ y suena a
$f_1 = \qty{440}{Hz}$: las \omterm{def:g10:signals-and-waves:wave}{ondas} la recorren a
$v = 2Lf_1 = 2 \times 0.328 \times 440 \approx \qty{289}{m/s}$ ---
fijada por la tensión y la masa, y afinada con la clavija. Apoyar un
dedo acorta $L$ y sube a la vez todas las $f_n$: el \omterm{def:g12:sound-acoustics:timbre}{espectro} entero se
desliza hacia arriba.
\end{example}

\begin{remark}[Por qué se diferencian los instrumentos]\label{rem:g12:sound-acoustics:instruments}
La manera de excitar la cuerda --- pulsada, frotada, percutida --- fija
cuánto hay de cada modo, y la caja amplifica unos \omterm{def:g12:sound-acoustics:harmonics}{armónicos} más que
otros; la excitación y la caja juntas fijan el \omterm{def:g12:sound-acoustics:timbre}{espectro}, de modo que el
mismo $f_1$ sale de una guitarra, de un violín y de un piano sonando
cada uno como él mismo.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:1}
En el aire ($v = \qty{340}{m/s}$), halla la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} de (a) una
nota grave de \qty{55}{Hz}; (b) un pitido de \qty{1.0}{kHz}; (c) un
chirrido de \qty{15}{kHz}. Compara la mayor con la menor.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:2}
Clasifica como infrasonido, \omterm{def:g10:signals-and-waves:sound}{sonido} audible o ultrasonido: \qty{8}{Hz};
\qty{100}{Hz}; \qty{15}{kHz}; \qty{40}{kHz}; \qty{3}{MHz}. De los
audibles, ¿cuál suena más grave?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:3}
Un operario golpea un raíl a \qty{1.0}{km}. Calcula los tiempos de viaje
por el acero y por el aire. ¿Qué oye alguien con una oreja pegada al
raíl?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:4}
(a) ¿El \omterm{def:g12:sound-acoustics:level}{nivel sonoro} para $I = \qty{1.0e-5}{W/m^2}$? (b) ¿La intensidad
a \qty{60}{dB}? (c) Nombra un \omterm{def:g10:signals-and-waves:sound}{sonido} cotidiano cercano a cada nivel.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sound-acoustics:5}
La cuerda sol de un violín suena a $f_1 = \qty{196}{Hz}$: da sus tres
\omterm{def:g12:sound-acoustics:harmonics}{armónicos} siguientes. ¿Por qué un silbido de \qty{12}{kHz}, por rico que
sea en la fuente, llega al oído como un \omterm{def:g12:sound-acoustics:harmonics}{sonido puro}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:6}
Una sirena radia $P = \qty{0.50}{W}$ por igual en todas direcciones.
Calcula la intensidad y el \omterm{def:g12:sound-acoustics:level}{nivel sonoro} a \qty{5.0}{m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:7}
Un violín solo da \qty{68}{dB} en tu butaca. ¿Qué nivel dan dos? ¿Y
cuatro? ¿Cuántos violines harían falta para llegar a \qty{78}{dB}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:8}
El nivel es de \qty{95}{dB} a \qty{2.0}{m} de un altavoz pequeño.
Predice el nivel a \qty{8.0}{m}: ¿está por debajo de la marca de
\qty{85}{dB}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sound-acoustics:9}
Una nota de \qty{440}{Hz} pasa del aire al agua. Calcula su \omterm{def:g12:mechanical-waves:wavelength}{longitud de
onda} en cada medio. El buceador oye el mismo \omterm{def:g12:sound-acoustics:pitch-loudness}{tono}: ¿qué magnitud fija la
fuente y cuál se ajusta?
\end{exercise}

\section{Problema: el taller del luthier}

\begin{problem}[{Problema de fin de semana --- el taller del luthier:
una cuerda de guitarra repartida traste a traste, sus armónicos afinados
hasta un timbre, un recital mantenido por debajo de la línea de peligro
y una caza por ultrasonidos de un defecto en la madera}]
\label{pb:g12:sound-acoustics:1}
Un luthier está terminando una guitarra. Su cuerda la tiene una longitud
vibrante $L = \qty{65.0}{cm}$ y debe \omterm{met:g10:signals-and-waves:echo}{sonar} a $f_1 = \qty{110}{Hz}$. Toma
\qty{340}{m/s} para el \omterm{def:g10:signals-and-waves:sound}{sonido} en el aire y admite $f_1 = v/(2L)$
(\cref{prop:g12:sound-acoustics:string-modes}), siendo $v$ la velocidad
de la \omterm{def:g10:signals-and-waves:wave}{onda} en la cuerda.

\medskip
\noindent\textbf{Parte I --- Repartir el diapasón.}
\begin{enumerate}
  \item Calcula la velocidad $v$ de la \omterm{def:g10:signals-and-waves:wave}{onda} en la cuerda.
  \item Calcula la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} del \omterm{def:g12:sound-acoustics:harmonics}{fundamental} \emph{en la cuerda}
        y después la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} del \omterm{def:g10:signals-and-waves:sound}{sonido} de \qty{110}{Hz}
        \emph{en el aire}. ¿Por qué son distintas con la misma
        \omterm{def:g10:signals-and-waves:frequency}{frecuencia}?
  \item La octava (\qty{220}{Hz}): ¿qué longitud vibrante la da? ¿Dónde
        tiene que ir el traste duodécimo, como fracción de $L$?
  \item La quinta (\qty{165}{Hz}): ¿qué longitud vibrante la da?
  \item Cada traste sube la nota un semitono, un factor $2^{1/12}$.
        Comprueba con logaritmos cuántos semitonos separan \qty{110}{Hz}
        de \qty{220}{Hz}; ¿longitud en el primer traste?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- \omterm{def:g12:sound-acoustics:harmonics}{Armónicos} y \omterm{def:g12:sound-acoustics:timbre}{timbre}.}
\begin{enumerate}[resume]
  \item Enumera los cinco primeros \omterm{def:g12:sound-acoustics:harmonics}{armónicos} de la cuerda la al aire.
  \item El la de un violinista suena a \qty{440}{Hz}. ¿Qué \omterm{def:g12:sound-acoustics:harmonics}{armónico} de
        la cuerda de guitarra es, y qué intervalo musical lo separa del
        \omterm{def:g12:sound-acoustics:harmonics}{fundamental}? ¿Y los \qty{330}{Hz}?
  \item El luthier pulsa cerca del puente y después sobre el diapasón:
        el mismo \omterm{def:g12:sound-acoustics:pitch-loudness}{tono}, distinto color. Explícalo con los \omterm{def:g12:sound-acoustics:timbre}{espectros}.
  \item Un toque suave en el punto medio mientras se pulsa impone allí
        un \omterm{def:g12:sound-acoustics:standing-wave}{nodo}. ¿Qué \omterm{def:g12:sound-acoustics:harmonics}{armónicos} sobreviven, y qué \omterm{def:g12:sound-acoustics:pitch-loudness}{tono} se oye?
  \item ¿Cuántos \omterm{def:g12:sound-acoustics:harmonics}{armónicos} de la cuerda al aire caen, en principio,
        dentro del intervalo audible?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El recital.}
En el primer concierto de la guitarra, cada instrumento radia una
\omterm{def:g11:work-of-force:power}{potencia} acústica $P = \qty{6.0e-3}{W}$ como fuente pequeña, por igual
en todas direcciones.
\begin{enumerate}[resume]
  \item Calcula la intensidad a $r = \qty{3.0}{m}$ de un instrumento.
  \item Deduce allí el \omterm{def:g12:sound-acoustics:level}{nivel sonoro}.
  \item Toca un cuarteto: ¿qué nivel dan cuatro instrumentos así en el
        mismo sitio? ¿Por qué no es cuatro veces el nivel?
  \item Tocan doce instrumentos: ¿cuál es el nivel a \qty{3.0}{m}?
  \item ¿A qué distancia tiene que sentarse un oyente para que el
        conjunto baje a \qty{85}{dB}?
  \item Regla de seguridad: \qty{85}{dB} son seguros durante ocho horas,
        y cada \qty{+3}{dB} reduce a la mitad el tiempo seguro. ¿Cuánto
        puede quedarse sin riesgo el oyente de primera fila de la
        pregunta 14?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El defecto de la madera.}
Antes de barnizar, el luthier explora una pieza de mástil de \qty{45}{mm}
de grosor con un comprobador de \qty{2.0}{MHz}; en esta madera toma
$v = \qty{5.0e3}{m/s}$.
\begin{enumerate}[resume]
  \item Calcula la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} del ultrasonido en la madera. ¿Por
        qué la \omterm{def:g10:signals-and-waves:wave}{onda} que caza defectos tiene que ser ultrasónica y no
        audible?
  \item Calcula el \omterm{def:g12:mechanical-waves:delay}{retraso} del \omterm{rem:g10:signals-and-waves:honest}{eco} de la cara posterior de la pieza.
  \item Llega un \omterm{rem:g10:signals-and-waves:honest}{eco} al cabo de \qty{7.2}{\micro s}: ¿a qué profundidad
        está el defecto?
  \item Escribe la ficha del taller: velocidad de la \omterm{def:g10:signals-and-waves:wave}{onda} en la cuerda,
        longitudes del traste duodécimo y del primero, distancia de
        escucha segura y profundidad del defecto.
\end{enumerate}
\end{problem}
