\section{نوّاس كهربائي: الدارة LC}

اشحن \omterm{def:g12:rc-rl-circuits:capacitor}{مكثّفة} $C$ إلى \omterm{def:g11:circuits-and-power:voltage}{التوتر} $U_0$ --- فتحمل صفيحتاها عندئذٍ
$\pm Q_0$ مع $Q_0 = C U_0$ --- وعند $t = 0$ صِلها \omterm{def:g12:rc-rl-circuits:inductor}{بوشيعة}
\emph{مثالية} $L$، مقاومتها معدومة. ولا منبع في أي مكان من
الحلقة: فمهما يحدث بعد ذلك، تفعله الدارة من تلقاء نفسها.

\omterm{def:g12:rc-rl-circuits:inductor}{وشيعة} مثالية. وإغلاق $K$ يتيح للشحنة أن تنسكب عبر \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} --- و

\begin{proposition}[معادلة الدارة LC]\label{prop:g12:rlc-oscillations:equation}
\index{دارة LC}
في الحلقة LC المثالية، تخضع شحنة $q(t)$ الصفيحة العليا للمعادلة
\[
L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0,
\qquad\text{أي}\qquad
\frac{d^2q}{dt^2} = -\frac{1}{LC}\,q .
\]
\end{proposition}

\begin{proof}
على امتداد الحلقة تتجمع التوترات إلى الصفر
(\cref{prop:g11:circuits-and-power:laws}): $u_L + u_C = 0$. و
\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} تعطي $u_C = q/C$، \omterm{def:g12:rc-rl-circuits:inductor}{والوشيعة} تعطي $u_L = L\,di/dt$
(\cref{def:g12:rc-rl-circuits:inductor})، مع $i = dq/dt$ عادًّا
الشحنة الواصلة إلى الصفيحة العليا
(\cref{prop:g12:rc-rl-circuits:cdudt})؛ فعوّض.
\end{proof}

\begin{theorem}[التذبذبات الحرة في الدارة LC]\label{thm:g12:rlc-oscillations:oscillation}
\index{تذبذبات كهربائية}
إذا أُطلقت الدارة عند $t = 0$ مع $q(0) = Q_0$ و$i(0) = 0$،
تذبذبت:
\[
q(t) = Q_0 \cos\!\left(\frac{2\pi t}{T_0}\right),
\qquad
i(t) = -I_0 \sin\!\left(\frac{2\pi t}{T_0}\right),
\qquad
T_0 = 2\pi\sqrt{LC},
\]
مع تيار أعظمي $I_0 = 2\pi Q_0/T_0 = Q_0/\sqrt{LC}$.
\end{theorem}

\begin{proof}
تحقّق بالتعويض. فبالاشتقاق،
$i = dq/dt = -\frac{2\pi}{T_0} Q_0 \sin(2\pi t/T_0)$ --- وهو $i(t)$
المذكور --- وبالاشتقاق مرة أخرى $\frac{d^2q}{dt^2} = -(\frac{2\pi}{T_0})^2 q$،
وهي تساوي $-q/(LC)$ بالضبط حين $(2\pi/T_0)^2 = 1/(LC)$، أي\
$T_0 = 2\pi\sqrt{LC}$. و$q(0) = Q_0$، $i(0) = 0$. أما أنه لا توجد
دالة \emph{أخرى} توافق المعادلة والانطلاقة فمقبول،
تمامًا كما في حالة الزنبرك
(\cref{rem:g12:oscillators-and-time:uniqueness}).
\end{proof}

\begin{definition}[الدور الذاتي والتواتر الذاتي]\label{def:g12:rlc-oscillations:period}
\emph{الدور الذاتي}\index{دور ذاتي!لدارة LC} في \omterm{prop:g12:rlc-oscillations:equation}{الدارة
LC} هو $T_0 = 2\pi\sqrt{LC}$؛ و
\emph{تواترها الذاتي}\index{تواتر ذاتي!لدارة LC} هو
\[
f_0 = \frac{1}{T_0} = \frac{1}{2\pi\sqrt{LC}} .
\]
والتحقق البُعدي: $\qty{1}{H} = \qty{1}{V.s/A}$ و
$\qty{1}{F} = \qty{1}{C/V}$، ومن ثم يحمل $LC$
البُعد $\unit{s} \times \unit{C/A} = \unit{s^2}$ --- فالمقدار $\sqrt{LC}$ زمن،
كما يجب.
\end{definition}

\begin{example}[أرقام أولى]\label{ex:g12:rlc-oscillations:firstnumbers}
$L = \qty{10}{mH}$ مع $C = \qty{1.0}{\micro F}$:
$\sqrt{LC} = \qty{1.0e-4}{s}$، ومنه $T_0 = \qty{0.63}{ms}$ و
$f_0 \approx \qty{1.6}{kHz}$ --- أي نغمة مسموعة. و$L = \qty{1.0}{mH}$
مع $C = \qty{100}{pF}$: $T_0 = \qty{2.0}{\micro s}$،
$f_0 \approx \qty{0.50}{MHz}$ --- أي \omterm{def:g12:oscillators-and-time:oscillator}{تواتر} راديوي. ولا يضبط الوتيرة إلا
\emph{الجداء} $LC$.
\end{example}

\begin{remark}[ربع دور خارج الخطى]\label{rem:g12:rlc-oscillations:quarter}
التيار خارج \omterm{def:g12:oscillators-and-time:phase}{الطور} عن الشحنة بربع \omterm{def:g12:oscillators-and-time:oscillator}{دور}: فيكون $i = 0$
حين $q = \pm Q_0$ (فالصفيحتان ممتلئتان، والتدفق ينقلب)، ويكون $i = \pm I_0$
حين $q = 0$ (فالصفيحتان فارغتان، والتدفق في أوجه) --- أي النوّاس
بالضبط، ساكنًا عند الطرفين وأسرع ما يكون عند الأسفل.
\end{remark}

\section{رقصة الطاقة}

\begin{proposition}[انحفاظ الطاقة المخزونة]\label{prop:g12:rlc-oscillations:energy}
\index{طاقة!لدارة LC}
في \omterm{prop:g12:rlc-oscillations:equation}{الدارة LC} المثالية، تكون \omterm{def:g10:energy-conservation:energy}{الطاقة} المخزونة الكلية
(\cref{prop:g12:rc-rl-circuits:energy})
\[
E \;=\; E_C + E_L \;=\; \frac{q^2}{2C} + \tfrac12\,L i^2
\;=\; \frac{Q_0^2}{2C} \;=\; \tfrac12\,L I_0^2
\]
ثابتةً: فهي كلها في \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} حين $q = \pm Q_0$، وكلها في
\omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} حين $i = \pm I_0$.
\end{proposition}

\begin{proof}
اشتقّ: $\frac{dE}{dt} = \frac{q}{C}\frac{dq}{dt} +
Li\frac{di}{dt} = i\left(\frac{q}{C} + L\frac{d^2q}{dt^2}\right) = 0$
بالعلاقة \cref{prop:g12:rlc-oscillations:equation}. وتُقرأ قيمتها عند
$t = 0$ (فكلها في \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة})، ومرة أخرى في لحظة التيار الخالص
$q = 0$ --- ومنه $\tfrac12 L I_0^2 = Q_0^2/2C$، فنستعيد
$I_0 = Q_0/\sqrt{LC}$ \omterm{def:g10:energy-conservation:energy}{بالطاقة} وحدها.
\end{proof}

ذهابًا وإيابًا مرتين في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور}، والمجموع مثبَّت عند $Q_0^2/2C$ --- وهي عين

\begin{example}[الرقصة، مقيسة]\label{ex:g12:rlc-oscillations:waltz}
$C = \qty{1.0}{\micro F}$ مشحونة إلى $U_0 = \qty{10}{V}$، موصولة \omterm{def:g12:rc-rl-circuits:inductor}{بوشيعة}
$L = \qty{10}{mH}$: \omterm{def:g10:energy-conservation:energy}{فالطاقة} $E = \tfrac12 C U_0^2 = \qty{5.0e-5}{J}$،
\omterm{def:g12:oscillators-and-time:oscillator}{والدور} $T_0 = \qty{0.63}{ms}$، وتيار أعظمي
$I_0 = U_0\sqrt{C/L} = \qty{0.10}{A}$ يُبلَغ بعد ربع \omterm{def:g12:oscillators-and-time:oscillator}{دور}
(\qty{0.16}{ms}) من إغلاق القاطعة.
\end{example}

\section{الدارات الحقيقية: التذبذبات المتخامدة}

كل \omterm{def:g12:rc-rl-circuits:inductor}{وشيعة} \omterm{def:g11:lenses-and-eye:images}{حقيقية} ملفوفة من سلك \omterm{def:g11:work-of-force:sign}{مقاوم}: فالحلقة الأمينة دارة
\emph{RLC} على التسلسل، \omterm{def:g11:circuits-and-power:resistance}{والمقاومة} تفرض ضريبتها على الرقصة.

\begin{definition}[النظامان شبه الدوري واللادوري]\label{def:g12:rlc-oscillations:regimes}
\index{دارة RLC}\index{نظام شبه دوري}\index{نظام لادوري}
في دارة RLC على التسلسل، تبدّد \omterm{def:g11:circuits-and-power:resistance}{المقاومة} \omterm{def:g10:energy-conservation:energy}{الطاقة} بمفعول
جول (\cref{ch:g11:circuits-and-power}): فتكون \omterm{def:g12:oscillators-and-time:oscillator}{التذبذبات الحرة}
\emph{\omterm{def:g12:oscillators-and-time:damping}{متخامدة}}. ونظامان، يُقرآن على \omterm{def:g10:signals-and-waves:oscilloscope}{راسم الاهتزاز}:
\begin{itemize}
  \item \emph{شبه دوري} ($R$ صغيرة): إذ تبقى الدارة
        تتذبذب داخل غلاف متقلص، بزمن تكرار ---
        هو \emph{شبه الدور}\index{شبه الدور!الكهربائي} ---
        قريب من $T_0 = 2\pi\sqrt{LC}$ متى كان \omterm{def:g12:oscillators-and-time:damping}{التخامد} خفيفًا؛
  \item \emph{لادوري} ($R$ كبيرة): إذ تزحف الشحنة عائدةً إلى الصفر
        دون أن تتجاوزه قط --- فلا تذبذب البتة.
\end{itemize}
وهما المقابل المضبوط \omterm{def:g12:oscillators-and-time:oscillator}{للمذبذب} الميكانيكي المتخامد
(\cref{def:g12:oscillators-and-time:damping}).
\end{definition}

\begin{remark}[صيانة التذبذبات]\label{rem:g12:rlc-oscillations:maintain}
\index{تذبذبات مصانة}
لإبقاء \omterm{def:g12:oscillators-and-time:oscillator}{السعة} حية، غذِّ الدارة بالضبط \omterm{def:g10:energy-conservation:energy}{بالطاقة} التي
تحرقها \omterm{def:g11:circuits-and-power:resistance}{المقاومة}، مرة في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور} و\emph{على الخطى} مع التأرجح: فالمضخّم
يفعل بالحلقة RLC ما تفعله دفعة الهارب
بالنوّاس (\cref{ch:g12:oscillators-and-time}). وكل ساعة كوارتز
وكل مرسِل راديوي \omterm{def:g12:oscillators-and-time:oscillator}{مذبذب} مصان من هذا النوع، يعمل
على تواتره الذاتي $f_0$.
\end{remark}

\section{التوأم الميكانيكي}

\begin{proposition}[القاموس الكهربائي--الميكانيكي]\label{prop:g12:rlc-oscillations:analogy}
\index{تماثل كهربائي--ميكانيكي}
معادلة \omterm{prop:g12:rlc-oscillations:equation}{الدارة LC} هي معادلة الزنبرك والكتلة
(\cref{prop:g12:oscillators-and-time:spring-period}) كلمةً بكلمة:
\begin{center}\small
\begin{tabular}{lcl}
\omterm{prop:g12:rlc-oscillations:equation}{الدارة LC} & & الزلّاجة على زنبرك\\[2pt]
الشحنة $q$ & $\longleftrightarrow$ & الموضع $x$\\
التيار $i = dq/dt$ & $\longleftrightarrow$ & \omterm{def:g12:kinematics-2d:velocity}{السرعة} $v = dx/dt$\\
\omterm{def:g12:rc-rl-circuits:inductor}{المحاثّة} $L$ & $\longleftrightarrow$ & الكتلة $m$\\
مقلوب السعة $1/C$ & $\longleftrightarrow$ & \omterm{def:g12:oscillators-and-time:spring}{الصلابة} $k$\\
\omterm{def:g11:circuits-and-power:resistance}{المقاومة} $R$ & $\longleftrightarrow$ & \omterm{def:g10:inertia:inventory}{الاحتكاك}\\
$L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0$ & $\longleftrightarrow$ &
  $m\,\frac{d^2x}{dt^2} + kx = 0$\\[2pt]
$T_0 = 2\pi\sqrt{LC}$ & $\longleftrightarrow$ &
  $T_0 = 2\pi\sqrt{m/k}$\\[2pt]
$E_C = q^2/2C$ & $\longleftrightarrow$ & $E_p = \tfrac12 kx^2$\\
$E_L = \tfrac12 Li^2$ & $\longleftrightarrow$ & $E_k = \tfrac12 mv^2$
\end{tabular}
\end{center}
المعادلة نفسها، والحلول نفسها: فكل نتيجة عن أيٍّ من المذبذبين
تُترجَم في الحال إلى اللغة الأخرى.
\end{proposition}

\begin{proof}
قارن \cref{prop:g12:rlc-oscillations:equation} بالمعادلة
$m\,d^2x/dt^2 = -kx$: فبتعويض $q \to x$، $L \to m$، $1/C \to k$
تتحول إحداهما إلى الأخرى، ويحوّل $LC \to m/k$
المقدارَ $2\pi\sqrt{LC}$ إلى $2\pi\sqrt{m/k}$.
\end{proof}

\begin{example}[موالفة راديو]\label{ex:g12:rlc-oscillations:tuning}
يغذّي الهوائي الحلقةَ LC بهمس من \emph{كل} محطة، لكن
لا تدفع على الخطى وتتراكم إلا المحطة التي تبثّ قرب $f_0$ ---
أي \omterm{def:g12:oscillators-and-time:resonance}{الرنين} (\cref{def:g12:oscillators-and-time:resonance}). والقرص
يدير صفائح \omterm{def:g12:rc-rl-circuits:capacitor}{مكثّفة} متغيرة: فتغيير $C$ يزحزح
$f_0 = 1/(2\pi\sqrt{LC})$، وتسلّم الإبرة السماعةَ محطة
واحدة في كل مرة. وفي جهة الإرسال، يضبط \omterm{def:g12:oscillators-and-time:oscillator}{مذبذب} مصان
(\cref{rem:g12:rlc-oscillations:maintain}) \omterm{def:g12:oscillators-and-time:oscillator}{تواتر}
البثّ؛ وفي الساعة، تؤدي بلورة كوارتز --- وهي توأم ميكانيكي
مذهل \omterm{def:g12:oscillators-and-time:spring}{الصلابة} --- الوظيفة نفسها.
\end{example}

\section{مسألة: بناء راديو بلوري}

\begin{problem}[{مسألة نهاية الأسبوع --- بناء راديو بلوري: وشيعة
واحدة، ومكثّفة متغيرة، وثنائي وسماعة أذن --- بلا بطارية،
وبلا مضخّم، وتصل أخبار المساء على طاقة الموجة
نفسها}]
\label{pb:g12:rlc-oscillations:1}
عدّة من علّية أحد الأجداد: \omterm{def:g12:rc-rl-circuits:inductor}{وشيعة} $L = \qty{0.20}{mH}$ ملفوفة على
أنبوب من الورق المقوّى، \omterm{def:g12:rc-rl-circuits:capacitor}{ومكثّفة} متغيرة تدور صفائحها من
\qty{20}{pF} إلى \qty{480}{pF}، وثنائي بلوري، وسماعة أذن، و
\qty{20}{m} من سلك الهوائي. والمحطات المطلوب التقاطها تبثّ بين
\qty{530}{kHz} و\qty{1700}{kHz}.

\medskip
\noindent\textbf{الجزء I --- حلقة الموالفة.}
\begin{enumerate}
  \item \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة}، وهي مشحونة، تُترك لتفرغ في \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة}:
        اكتب قانون الحلقة وحوّله إلى المعادلة التفاضلية
        من أجل $q(t)$.
  \item تحقق بالتعويض من أن $q(t) = Q_0\cos(2\pi t/T_0)$
        تحلّها، بشرط $T_0 = 2\pi\sqrt{LC}$.
  \item تحقق بالتحليل البُعدي من أن $\sqrt{LC}$ زمن.
  \item والقرص مضبوط على $C = \qty{330}{pF}$: احسب $T_0$ و$f_0$. فهل
        الحلقة موالَفة داخل الحزمة؟
  \item جِد السعة التي توالف \qty{1.00}{MHz} بالضبط.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} المتغيرة.}
\begin{enumerate}[resume]
  \item احسب السعة اللازمة من أجل \qty{530}{kHz}.
  \item احسب السعة اللازمة من أجل \qty{1700}{kHz}.
  \item احسب التواترين اللذين تبلغهما فعلًا \omterm{def:g12:rc-rl-circuits:capacitor}{مكثّفة} العدّة
        ذات \qtyrange{20}{480}{pF}. فهل تغطي الحزمة كلها؟
  \item بيّن أن $f_0 \propto 1/\sqrt{C}$ عند ثبات $L$؛ وبأي
        عامل يجب أن يتغير $C$ ليتضاعف $f_0$؟
  \item تنفتح الصفائح انفتاحًا خطيًّا، ومن ثم يهبط $C$ بانتظام مع دوران
        القرص. فسّر لماذا تتزاحم المحطات عندئذٍ عند أحد طرفَي
        القرص --- وأيّ طرف.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- \omterm{def:g10:energy-conservation:energy}{الطاقة} \omterm{def:g12:oscillators-and-time:damping}{والتخامد}.}
وبالموالفة على \qty{1.00}{MHz} ($C = \qty{127}{pF}$)، يشحن الهوائي
\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} لحظيًّا إلى $U_0 = \qty{20}{mV}$.
\begin{enumerate}[resume]
  \item احسب الشحنة $Q_0$ \omterm{def:g10:energy-conservation:energy}{والطاقة} المخزونة $E$.
  \item احسب التيار الأعظمي $I_0$، باستعمال \omterm{thm:g10:energy-conservation:conservation}{انحفاظ الطاقة}.
  \item تبدّد \omterm{def:g11:circuits-and-power:resistance}{مقاومة} \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} $4.0\%$ من \omterm{def:g10:energy-conservation:energy}{الطاقة} المخزونة
        في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور}: فبعد كم \omterm{def:g12:oscillators-and-time:oscillator}{دور} تكون \omterm{def:g10:energy-conservation:energy}{الطاقة} قد انتصفت؟
  \item وكم يبلغ ذلك بالميكروثانية؟ ولماذا لا تقدر هذه ``الرنّة''
        وحدها أن تذيع أخبار المساء؟
  \item فسّر لماذا لا تبقي \omterm{def:g10:signals-and-waves:wave}{الموجة} الواصلة الحلقةَ تتأرجح إلا
        عند الموالفة على تلك المحطة --- وما الذي يصون
        التذبذب في \emph{المرسِل}
        (\cref{rem:g12:rlc-oscillations:maintain}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- التوأم الميكانيكي.}
\begin{enumerate}[resume]
  \item باستعمال القاموس (\cref{prop:g12:rlc-oscillations:analogy})
        مع زلّاجة كتلتها $m = \qty{0.20}{kg}$، احسب
        \omterm{def:g12:oscillators-and-time:spring}{صلابة} $k$ الزنبرك الموافقة للدارة الموالَفة.
  \item ولزنبرك سيارة قاسٍ $k \approx \qty{5e4}{N/m}$. علّق:
        فهل يقدر جهاز زنبرك وكتلة على الطاولة أن يتذبذب بالميغاهرتز؟
  \item واحتفظ بقيمة معقولة $k = \qty{100}{N/m}$ بدلًا من ذلك: فأيّ كتلة
        تتذبذب عند \qty{1.00}{MHz}؟
  \item وأيّ جسم حقيقي هو \omterm{def:g12:oscillators-and-time:oscillator}{مذبذب} ميكانيكي بهذه الخفة
        وهذه \omterm{def:g12:oscillators-and-time:spring}{الصلابة} الفائقة، وأين لاقيته سابقًا هذه
        السنة (\cref{ch:g12:oscillators-and-time})؟
  \item ورقة المواصفات، في جملة واحدة: الحزمة التي يمسحها راديوك، و
        السعة التي تلتقط \qty{1.00}{MHz}، وكم تدوم رنّة وحيدة،
        وما الذي يذيع الأخبار على أي حال.
\end{enumerate}
\end{problem}
