\chapter{الصوت والصوتيات}\label{ch:g12:sound-acoustics}

انقر وتر القيثارة الغليظ: فتمتلئ الغرفة بنغمة لا خفيضة، ومع ذلك
لم ينتقل إلى أذنك إلا هواء مضغوط ومتمدّد. ويتبع هذا
الفصل ذلك النمط --- سرعته، وكيف يصير \omterm{def:g10:signals-and-waves:frequency}{التواتر}
حدّةً \omterm{def:g10:energy-conservation:power}{والاستطاعة} علوًّا، ولماذا تختلف النغمة الواحدة على كل آلة.

\section{الصوت موجة ضغط}

\begin{definition}[الموجة الصوتية]\label{def:g12:sound-acoustics:sound-wave}
\emph{الموجة الصوتية}\index{موجة صوتية} \omterm{def:g10:signals-and-waves:wave}{موجة} \omterm{def:g10:pressure:pressure}{ضغط} ميكانيكية
(\cref{ch:g12:mechanical-waves}): فيُنشئ سطح مهتزّ
\emph{تضاغطات}\index{تضاغط} منتقلة (\omterm{def:g10:pressure:pressure}{بضغط} زائد طفيف) و
\emph{تخلخلات}\index{تخلخل} (\omterm{def:g10:pressure:pressure}{بضغط} ناقص). وتتذبذب طرود
الهواء \emph{على امتداد} جهة الانتقال (\omterm{def:g10:signals-and-waves:sound}{فالصوت} \emph{طولي}\index{موجة طولية})،
ولا بدّ من وسط: \omterm{def:g12:sound-acoustics:timbre}{فجرس} تحت \omterm{def:g12:sound-acoustics:timbre}{جرس} خلاء يصمت
(\cref{ch:g10:signals-and-waves}).
\end{definition}

\begin{proposition}[سرعة الصوت]\label{prop:g12:sound-acoustics:speed}
\index{سرعة الصوت}%
ينتقل \omterm{def:g10:signals-and-waves:sound}{الصوت} بسرعة نحو \qty{340}{m/s} في الهواء عند
\qty{20}{\degreeCelsius}، و\qty{1500}{m/s} في الماء، و
\qty{5000}{m/s} في الفولاذ: فكلما ازداد الوسط صلابة ازداد
سرعة. \admitted
\end{proposition}

\begin{remark}[الهواء الدافئ أسرع]\label{rem:g12:sound-acoustics:temperature}
هذه قيم مقيسة؛ أما التنبؤ بها فيجري في مجلد السنة
الأولى. ويتسارع \omterm{def:g10:signals-and-waves:sound}{الصوت} في الهواء الأدفأ (\qty{331}{m/s} عند
\qty{0}{\degreeCelsius}) --- ولذلك تنحرف آلات النفخ عن الضبط كلما دفئت القاعة.
\end{remark}

\begin{example}[أحجام الأصوات]\label{ex:g12:sound-acoustics:wavelengths}
تقيس العلاقة $\lambda = v/f$ أي \omterm{def:g10:signals-and-waves:sound}{صوت}: ففي الهواء يمتد هدير تواتره \qty{20}{Hz}
على $340/20 = \qty{17}{m}$ --- أي بناية --- ويسع فحيح تواتره
\qty{20}{kHz} في \qty{1.7}{cm}؛ أما ذوات الحجم المتري
فتنعطف بسهولة حول الأبواب والزوايا.
\end{example}

\section{الحدّة والعلوّ والأذن}

\begin{definition}[الحدّة والعلوّ]\label{def:g12:sound-acoustics:pitch-loudness}
\emph{الحدّة}\index{حدّة} \omterm{def:g11:color-light-sources:perception}{إدراك} \omterm{def:g10:signals-and-waves:frequency}{التواتر}: فكلما ارتفع
$f$ ارتفعت النغمة؛ ونغمة لا القياسية \qty{440}{Hz}، ومضاعفة
$f$ ترفع النغمة أوكتافًا واحدًا. و\emph{العلوّ}\index{علوّ}
\omterm{def:g11:color-light-sources:perception}{إدراك} \omterm{def:g10:signals-and-waves:amplitude}{السعة}: فكلما قوي تذبذب \omterm{def:g10:pressure:pressure}{الضغط}
علا \omterm{def:g10:signals-and-waves:sound}{الصوت}؛ والتضخيم يغيّر العلوّ لا الحدّة.
\end{definition}

\begin{remark}[نافذة الأذن]\label{rem:g12:sound-acoustics:audible}
يمتد السمع من نحو \qty{20}{Hz} إلى \qty{20}{kHz}
(\cref{ch:g10:signals-and-waves})؛ وتحت ذلك يقع \omterm{def:g10:signals-and-waves:ultrasound}{تحت الصوتي}، وفوقه يقع
\omterm{def:g10:signals-and-waves:ultrasound}{فوق الصوتي} --- فالخفافيش عند \qty{50}{kHz}، والمسابر الطبية عند الميغاهرتز. أما
البيانو فلا يأخذ إلا عيّنة من الوسط: \qtyrange{27.5}{4186}{Hz}.
\end{remark}

\section{شدة الصوت وسُلّم الديسيبل}

\begin{definition}[شدة الصوت]\label{def:g12:sound-acoustics:intensity}
\emph{شدة الصوت}\index{شدة الصوت} $I$ هي
\omterm{def:g10:energy-conservation:power}{الاستطاعة} الصوتية التي تعبر مترًا مربعًا واحدًا مواجهًا \omterm{def:g10:signals-and-waves:wave}{للموجة}:
$I = P/S$، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{W/m^2}. وأخفت \omterm{def:g10:signals-and-waves:sound}{صوت} مسموع نحو
$I_0 = \qty{1.0e-12}{W/m^2}$؛ ويبدأ الألم قرب \qty{1}{W/m^2}.
\end{definition}

\begin{proposition}[التوهين بالتربيع العكسي]\label{prop:g12:sound-acoustics:inverse-square}
\index{قانون التربيع العكسي}%
منبع صغير يشعّ \omterm{def:g11:work-of-force:power}{استطاعة} $P$ بالتساوي في كل الاتجاهات
يعطي، على المسافة $r$، $I = P/(4\pi r^2) \propto 1/r^2$: فمضاعفة
المسافة تقسم الشدة على أربعة.
\end{proposition}

\begin{proof}
تعبر \omterm{def:g10:energy-conservation:power}{الاستطاعة} $P$ كلها الكرةَ ذات المساحة $4\pi r^2$؛ فاقسم.
\end{proof}

\begin{definition}[مستوى الصوت]\label{def:g12:sound-acoustics:level}
تمتد الأذن على اثنتي عشرة \omterm{def:g10:inertia:force}{قوة} من قوى العشرة، ولذلك تُذكر الشدّات
لوغاريتميًّا. و\emph{مستوى الصوت}\index{مستوى الصوت} للشدة $I$ هو
\[ L = 10 \log_{10}(I/I_0), \qquad I_0 = \qty{1.0e-12}{W/m^2}, \]
\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \emph{الديسيبل}\index{ديسيبل} (\unit{dB}): فعتبة السمع
\qty{0}{dB}، والألم قرب \qty{120}{dB}.
\end{definition}

\begin{proposition}[حساب الديسيبل]\label{prop:g12:sound-acoustics:db-arithmetic}
ضرب $I$ في $k$ يضيف $10\log_{10}k$ ديسيبلًا. وخاصةً
$\times 10$ يضيف \qty{10}{dB}؛ و$\times 2$ يضيف
$10\log_{10}2 \approx \qty{3.0}{dB}$؛ و$N$ منبعًا متطابقًا غير مترابط تتجمع
\emph{شدّاتها}، لا مستوياتها أبدًا، فتعطي
$L_1 + 10\log_{10}N$؛ ومضاعفة المسافة إلى منبع صغير (فتنقسم $I$
على أربعة) تنزع نحو \qty{6.0}{dB}.
\end{proposition}

\begin{proof}
$10\log_{10}(kI/I_0) = 10\log_{10}(I/I_0) + 10\log_{10}k$؛
وفي المسافة: \cref{prop:g12:sound-acoustics:inverse-square}، $k = 1/4$.
\end{proof}

\begin{method}[العمل بالديسيبل]\label{met:g12:sound-acoustics:db}
\begin{enumerate}
  \item $L = 10\log_{10}(I/I_0)$؛ وعكسًا: $I = I_0 \times 10^{L/10}$.
  \item ولا تجمع المستويات أبدًا: حوّل إلى شدّات، واجمع، ثم حوّل عائدًا
        --- أو $+\qty{3}{dB}$ لكل مضاعفة، و$+\qty{10}{dB}$ لكل عشرة أضعاف.
  \item وكل مضاعفة للمسافة إلى منبع صغير تكلّف \qty{6}{dB}.
\end{enumerate}
\end{method}

\begin{remark}[الضرر السمعي]\label{rem:g12:sound-acoustics:damage}
فقدان السمع جرعة: فالمستوى \qty{85}{dB} آمن ثماني ساعات، وكل

الآمن؛ والخلايا الشعرية المفقودة لا تنمو ثانيةً.
\end{remark}

\section{الجرس والتوافقيات}

\begin{definition}[التوافقيات]\label{def:g12:sound-acoustics:harmonics}
\omterm{def:g10:signals-and-waves:sound}{الصوت} الدوري ذو \omterm{def:g10:signals-and-waves:frequency}{التواتر} $f_1$ --- أي
\emph{الأساسي}\index{الأساسي} --- هو عمومًا مجموع
جيوب عند التواترات $f_n = n f_1$ و$n = 1, 2, 3, \dots$، أي
\emph{توافقياته}\index{توافقي}. \omterm{def:g10:signals-and-waves:sound}{والصوت} المختزَل إلى أساسيه
\emph{نغمة نقية}\index{نغمة نقية}، كنغمة شوكة رنّانة.
\end{definition}

\begin{definition}[الطيف والجرس]\label{def:g12:sound-acoustics:timbre}
\emph{طيف}\index{طيف} صوتٍ هو مخطط أعمدة
\omterm{def:g12:sound-acoustics:harmonics}{توافقياته}: عمود واحد لكل $f_n$، وارتفاعه \omterm{def:g10:inertia:force}{قوة} ذلك \omterm{def:g12:sound-acoustics:harmonics}{التوافقي}.
وتسمع الأذن $f_1$ حدّةً، و\emph{مزيج} القوى
جرسًا، هو \emph{الجرس}\index{جرس} --- أي لون \omterm{def:g10:signals-and-waves:sound}{الصوت}:
فالآلات التي تعزف النغمة نفسها تشترك في $f_1$، ولا تختلف إلا في
أطيافها.
\end{definition}

\section{الوتر المهتزّ}

\begin{definition}[الموجة المستقرة]\label{def:g12:sound-acoustics:standing-wave}
\omterm{def:g10:signals-and-waves:wave}{الموجة} المحبوسة بين طرفَي وتر ثابتين تقدر أن تستقرّ في
\emph{موجة مستقرة}\index{موجة مستقرة}: أي تذبذب في المكان،
تتناوب فيه نقاط ساكنة --- هي \emph{العقد}\index{عقدة} --- مع
نقاط أقصى تأرجح --- هي \emph{البطون}\index{بطن}.
ولا ينتقل شيء بعد ذلك؛ إنما يهتزّ الوتر في نمط ثابت.
\end{definition}

\begin{proposition}[أنماط وتر]\label{prop:g12:sound-acoustics:string-modes}
الوتر الذي طوله $L$، والمثبَّت من طرفيه، والذي يحمل أمواجًا سرعتها
$v$، لا يهتزّ اهتزازًا دائمًا إلا في أنماط يسع فيها عدد صحيح $n$
من أنصاف أطوال \omterm{def:g10:signals-and-waves:wave}{الموجة} بين العقدتين المفروضتين عند الطرفين:
\[ \lambda_n = \frac{2L}{n}, \quad f_n = n\,\frac{v}{2L} = n f_1, \quad n = 1, 2, 3, \dots \]
والنمط الأدنى $f_1 = v/(2L)$ هو \omterm{def:g12:sound-acoustics:harmonics}{الأساسي}؛ وسائرها
\omterm{def:g12:sound-acoustics:harmonics}{توافقياته} بالضبط. \admitted
\end{proposition}

\begin{remark}[أين يقيم البرهان الأمين]\label{rem:g12:sound-acoustics:honest}
كون الأمواج المحبوسة تنتقي هذه الأنماط بالضبط ينتج عن
تراكب \omterm{def:g10:signals-and-waves:wave}{الموجة} مع انعكاساتها، وهو تحليل يُجرى في
مجلد السنة الأولى. أما الهندسة فتقنع: إذ يجب أن يبقى الطرفان
ساكنين، فيجب أن يسع عدد صحيح من أنصاف أطوال \omterm{def:g10:signals-and-waves:wave}{الموجة}.
\end{remark}

\begin{example}[وتر لا في الكمان]\label{ex:g12:sound-acoustics:violin}
لوتر لا في كمان $L = \qty{32.8}{cm}$ ويصدر
$f_1 = \qty{440}{Hz}$: فتجري الأمواج عليه بسرعة
$v = 2Lf_1 = 2 \times 0.328 \times 440 \approx \qty{289}{m/s}$ ---
يحدّدها \omterm{def:g10:inertia:inventory}{الشدّ} والكتلة، ويضبطها المِلوى. \omterm{def:g10:pressure:pressure}{وضغط} إصبع
يقصّر $L$ ويرفع كل $f_n$ دفعةً واحدة: فينزلق \omterm{def:g12:sound-acoustics:timbre}{الطيف} صعودًا.
\end{example}

\begin{remark}[لماذا تختلف الآلات]\label{rem:g12:sound-acoustics:instruments}
كيفية إثارة الوتر --- نقرًا أو تقويسًا أو طرقًا --- تحدّد كم
من كل نمط يحضر، ويضخّم الجسم بعض التوافقيات
أكثر من غيرها؛ فالإثارة مع الجسم تثبّتان \omterm{def:g12:sound-acoustics:timbre}{الطيف}، ومن ثم يغادر $f_1$
نفسه القيثارةَ والكمان والبيانو وكلٌّ يبدو على حقيقته.
\end{remark}

\section{مسألة: ورشة صانع الآلات الوترية}

\begin{problem}[{مسألة نهاية الأسبوع --- ورشة صانع الآلات الوترية: وتر قيثارة
موزَّع عتبةً عتبة، وتوافقياته مضبوطة في جرس، وحفل
يُبقى دون خط الخطر، ومطاردة بالموجات فوق الصوتية لعيب
في الخشب}]
\label{pb:g12:sound-acoustics:1}
يُنهي صانع آلات وترية قيثارة. ولوتر لا فيها طول مهتزّ
$L = \qty{65.0}{cm}$ ويجب أن يصدر $f_1 = \qty{110}{Hz}$. خذ
\qty{340}{m/s} \omterm{prop:g12:sound-acoustics:speed}{لسرعة الصوت} في الهواء، واقبل $f_1 = v/(2L)$
(\cref{prop:g12:sound-acoustics:string-modes})، حيث $v$ سرعة
\omterm{def:g10:signals-and-waves:wave}{الموجة} على الوتر.

\medskip
\noindent\textbf{الجزء I --- توزيع لوحة العتبات.}
\begin{enumerate}
  \item احسب سرعة \omterm{def:g10:signals-and-waves:wave}{الموجة} $v$ على الوتر.
  \item احسب طول \omterm{def:g10:signals-and-waves:wave}{موجة} \omterm{def:g12:sound-acoustics:harmonics}{الأساسي} \emph{على الوتر}،
        ثم طول \omterm{def:g10:signals-and-waves:wave}{موجة} \omterm{def:g10:signals-and-waves:sound}{الصوت} \qty{110}{Hz} \emph{في
        الهواء}. ولماذا يختلفان عند \omterm{def:g10:signals-and-waves:frequency}{التواتر} نفسه؟
  \item الأوكتاف (\qty{220}{Hz}): فأيّ طول مهتزّ يعطيه؟
        وأين يجب أن تقع العتبة الثانية عشرة، كسرًا من $L$؟
  \item والخامسة (\qty{165}{Hz}): فأيّ طول مهتزّ يعطيها؟
  \item وكل عتبة ترفع النغمة نصف نغمة، أي بعامل
        $2^{1/12}$. تحقّق باللوغاريتمات كم نصف نغمة يفصل
        \qty{110}{Hz} عن \qty{220}{Hz}؛ وما الطول عند العتبة الأولى؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- التوافقيات \omterm{def:g12:sound-acoustics:timbre}{والجرس}.}
\begin{enumerate}[resume]
  \item أحصِ التوافقيات الخمسة الأولى لوتر لا المفتوح.
  \item نغمة لا عند عازف كمان تصدر \qty{440}{Hz}. فأيّ \omterm{def:g12:sound-acoustics:harmonics}{توافقي} لوتر
        القيثارة هو ذاك، وأيّ فاصل موسيقي يفصله
        عن \omterm{def:g12:sound-acoustics:harmonics}{الأساسي}؟ وماذا عن \qty{330}{Hz}؟
  \item ينقر الصانع قرب الفرس، ثم فوق لوحة
        الأصابع: \omterm{def:g12:sound-acoustics:pitch-loudness}{الحدّة} نفسها ولون مختلف. اشرح بالأطياف.
  \item ولمسة خفيفة عند المنتصف أثناء النقر تفرض \omterm{def:g12:sound-acoustics:standing-wave}{عقدة}
        هناك. فأيّ التوافقيات يبقى، وأيّ \omterm{def:g12:sound-acoustics:pitch-loudness}{حدّة} تُسمع؟
  \item وكم توافقيًّا للوتر المفتوح يقع، من حيث المبدأ،
        داخل \omterm{def:g11:electric-gravitational-fields:field}{المجال} المسموع؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الحفل.}
في أول حفل للقيثارة، تشعّ كل آلة \omterm{def:g11:work-of-force:power}{استطاعة}
صوتية $P = \qty{6.0e-3}{W}$ منبعًا صغيرًا، بالتساوي في كل
الاتجاهات.
\begin{enumerate}[resume]
  \item احسب الشدة على بعد $r = \qty{3.0}{m}$ من آلة واحدة.
  \item استنتج \omterm{def:g12:sound-acoustics:level}{مستوى الصوت} هناك.
  \item ويعزف رباعي: فأيّ مستوى تعطيه أربع آلات كهذه
        عند الموضع نفسه؟ ولماذا ليس أربعة أضعاف المستوى؟
  \item وتعزف اثنتا عشرة آلة: فما المستوى عند \qty{3.0}{m}؟
  \item وكم يجب أن يجلس مستمع إلى الوراء ليهبط مستوى المجموعة
        إلى \qty{85}{dB}؟
  \item وقاعدة السلامة: \qty{85}{dB} آمن ثماني ساعات، وكل
        \qty{+3}{dB} ينصّف الزمن الآمن. فكم يجوز لمستمع
        الصفّ الأول في السؤال 14 أن يبقى بأمان؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- العيب في الخشب.}
قبل التورنيش، يسبر الصانع قطعة عنق خام سُمكها \qty{45}{mm}
بفاحص تواتره \qty{2.0}{MHz}؛ وخذ في هذا الخشب $v = \qty{5.0e3}{m/s}$.
\begin{enumerate}[resume]
  \item احسب \omterm{def:g12:mechanical-waves:wavelength}{طول الموجة} فوق الصوتية في الخشب. ولماذا يجب أن تكون
        \omterm{def:g10:signals-and-waves:wave}{موجة} مطاردة العيوب فوق صوتية لا مسموعة؟
  \item احسب \omterm{def:g12:mechanical-waves:delay}{تأخير} \omterm{rem:g10:signals-and-waves:honest}{الصدى} من الوجه الخلفي للقطعة.
  \item ويصل صدًى بعد \qty{7.2}{\micro s}: فكم يبلغ عمق العيب؟
  \item اكتب بطاقة الورشة: سرعة \omterm{def:g10:signals-and-waves:wave}{الموجة} على الوتر، وطولا العتبة
        الثانية عشرة والعتبة الأولى، ومسافة الاستماع الآمنة، وعمق العيب.
\end{enumerate}
\end{problem}
