\chapter{النسبية الخاصة: تمدد الزمن}\label{ch:g12:special-relativity}

يحمل كل \omterm{def:g10:universal-gravitation:satellite}{تابع} GPS ساعة ذرّية مضبوطة إلى \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة في ثلاثة
ملايين سنة --- وقبل الإطلاق يزيح المهندسون ضبطها عمدًا: فلو تُركت
أمينة لكسبت ثمانية وثلاثين جزءًا من مليون من \omterm{def:g10:orders-of-magnitude:unit}{الثانية} في اليوم، ولوضعك هاتفك
مساءَ الغد على بعد أحد عشر كيلومترًا من حيث تقف.
والساعة ليست المخطئة --- بل الزمن نفسه يدقّ دقًّا مختلفًا للمتحرك و
الساكن: ويتبع هذا الفصل أينشتاين من واقعة عنيدة واحدة إلى تلك النتيجة.

\section{سرعة واحدة ترفض أن تُجمع}

تتوقف السرعات على \omterm{def:g10:relative-motion:frame}{المرجع} وتتركب بالجمع
(\cref{ch:g10:relative-motion}): فالراكب الماشي \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{4}{km/h} نحو
مقدمة قطار يسير \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{300}{km/h} يمرّ بالمراقب على جانب السكة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{304}{km/h}.

\begin{definition}[المرجع العُطالي]\label{def:g12:special-relativity:inertial}
\emph{المرجع العُطالي}\index{مرجع عُطالي} هو \omterm{def:g10:relative-motion:frame}{مرجع}
(\cref{ch:g10:relative-motion}) يحفظ فيه جسمٌ خالٍ من القوى سرعةً
ثابتة: أي الأرض، بتقريب كافٍ، وكل \omterm{def:g10:relative-motion:frame}{مرجع} في \omterm{def:g12:kinematics-2d:uniform}{حركة مستقيمة
منتظمة} بالنسبة إليها.
\end{definition}

ادفع القاعدة إلى الأقصى: فينبغي أن تمرّ حزمة مصباح القطار الأمامي
بالمراقب على الأرض \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $c + \qty{300}{km/h}$. وفي 1887 قارن مايكلسون ومورلي
\omterm{def:g12:kinematics-2d:velocity}{سرعة} الضوء على امتداد اندفاع الأرض المداري وعبره، وهو \qty{30}{km/s}،
بحثًا عن مثل هذه الفروق؛ فلم يجدا شيئًا بدقة بالغة
--- أي $c$ نفسها في كل اتجاه، وفي كل فصل. وفي 1905 صدّق أينشتاين
التجربة وأعاد بناء الزمان والمكان حولها، على مسلَّمتين
يُحكم عليهما بنتائجهما --- وقد اختُبرت منذ قرن دون فشل واحد.

\begin{theorem}[مسلَّمتا النسبية الخاصة]\label{thm:g12:special-relativity:postulates}
\index{نسبية خاصة}\index{مسلَّمتا النسبية}
\begin{enumerate}
  \item \emph{مبدأ النسبية}: تأخذ قوانين الفيزياء الصورة
        نفسها في كل \omterm{def:g12:special-relativity:inertial}{مرجع عُطالي}؛ ولا أحد منها ``ساكن حقًّا''.
  \item \emph{ثبات $c$}: يسافر الضوء في الخلاء \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
        $c = \qty{3.00e8}{m/s}$ في كل \omterm{def:g12:special-relativity:inertial}{مرجع عُطالي}، مهما تكن
        حركة المنبع أو المراقب.
\end{enumerate}
\end{theorem}
\admitted

\section{وداعًا لعبارة ``اللحظة نفسها''}

\begin{proposition}[نسبية التزامن]\label{prop:g12:special-relativity:simultaneity}
\index{تزامن}
الحدثان المتزامنان في \omterm{def:g12:special-relativity:inertial}{مرجع عُطالي} ليسا، في العموم،
متزامنين في \omterm{def:g10:relative-motion:frame}{مرجع} آخر: فعبارة ``الآن، في كل مكان'' تتوقف على \omterm{def:g10:relative-motion:frame}{المرجع}.
\end{proposition}

\begin{proof}
قطار أينشتاين. صاعقتان تضربان طرفَي قطار متحرك،
فتحرقان السكة. والمراقب $M$ على الرصيف، وهو في منتصف المسافة بين
الأثرين، يستقبل الوميضين معًا؛ وقد قطع كلٌّ منهما المسافة نفسها \omterm{def:g12:kinematics-2d:velocity}{بالسرعة}
نفسها $c$، ومن ثم كانت الضربتان متزامنتين. أما الراكب $M'$،
وهو في منتصف القطار، فيركب \emph{نحو} الوميض الأمامي، الذي يبلغه
أولًا؛ لكن في مرجعه أيضًا قطع الوميضان مسافتين متساويتين (نصف
قطار) \omterm{def:g12:kinematics-2d:velocity}{بالسرعة} نفسها $c$ --- أي بالمسلَّمة الثانية! --- ومن ثم وقعت الضربة
الأمامية أولًا. وكل حكم صحيح في مرجعه؛ و
الخلاف، وهو نانوثوانٍ في حالة قطار، هو سبب عدم انتباه أحد له.
\end{proof}

\section{ساعة الضوء وعامل $\gamma$}

\begin{definition}[الحدث والزمن الذاتي]\label{def:g12:special-relativity:proper-time}
\emph{الحدث}\index{حدث} هو وقوع في مكان واحد ولحظة واحدة
--- أي دقّة، أو وميض، أو تفكك. ومتى حضرت ساعة واحدة عند حدثين،
كانت الفترة التي تقرؤها بينهما هي
\emph{الزمن الذاتي}\index{زمن ذاتي} $\Delta t_0$. أما \omterm{def:g10:relative-motion:frame}{المرجع} الذي تتحرك فيه تلك
الساعة فيقيس فترة مختلفة في العموم $\Delta t$.
\end{definition}

\begin{theorem}[تمدد الزمن]\label{thm:g12:special-relativity:dilation}
\index{تمدد الزمن}\index{عامل غاما@عامل $\gamma$ (غاما)}
الساعة المتحركة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} ثابتة $v$ عبر \omterm{def:g12:special-relativity:inertial}{مرجع عُطالي} يُقاس
فيه أنها تتأخر: فبين حدثين عند الساعة،
\[
\Delta t = \gamma\, \Delta t_0, \qquad
\gamma = \frac{1}{\sqrt{1 - v^2/c^2}} \geq 1 .
\]
\end{theorem}

\begin{proof}
ابنِ أبسط ساعة يمكن تصورها: مرآتان متقابلتان تفصل بينهما مسافة $L$،
\omterm{def:g12:quantum-world:photon}{وفوتون} يرتدّ بينهما؛ فالرحلة الكاملة الواحدة دقّة واحدة، وفي
السكون تدوم الدقّة $\Delta t_0 = 2L/c$. والآن ثبّت الساعة في سفينة
تعبر \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v$، والمرآتان عموديتان على الحركة، وقِس دقّة
واحدة من الأرض: ففي أثناء صعود \omterm{def:g12:quantum-world:photon}{الفوتون} تنزلق المرآتان جانبًا،
فيطير \omterm{def:g12:quantum-world:photon}{الفوتون} على \omterm{def:g10:relative-motion:trajectory}{مسار} مائل --- أي حركتين في وقت واحد، \omterm{def:g12:projectile-motion:projectile}{كمقذوف}
\cref{ch:g12:kinematics-2d}. وإذا دامت الدقّة $\Delta t$،
تقدّم كل نصف دقّة $v\,\Delta t/2$ أفقيًّا و$L$ شاقوليًّا؛ ثم ---
وهي الخطوة الحاسمة --- تثبّت المسلَّمة الثانية \omterm{def:g12:kinematics-2d:velocity}{سرعة} \omterm{def:g12:quantum-world:photon}{الفوتون} على الأرض عند
$c$، ومن ثم يقيس نصف \omterm{def:g10:relative-motion:trajectory}{المسار} المائل $c\,\Delta t/2$. وبفيثاغورس:
\[
\Bigl(\frac{c\,\Delta t}{2}\Bigr)^{\!2} = L^2 + \Bigl(\frac{v\,\Delta t}{2}\Bigr)^{\!2}
\;\Longrightarrow\;
\Delta t = \frac{2L/c}{\sqrt{1 - v^2/c^2}} = \gamma\,\Delta t_0 .
\]
وأي ساعة تركب إلى جانبها --- ساعة يد، أو نبض قلب، أو \omterm{def:g11:fundamental-interactions:ladder}{نواة} متفككة ---
يجب أن تبقى على الخطى، وإلا فضحت المقارنةُ حركةَ السفينة وخالفت
المسلَّمة الأولى.
\end{proof}

\begin{remark}[لماذا لم تنتبه الحياة اليومية قط]\label{rem:g12:special-relativity:everyday}
من أجل $v \ll c$، اكتب $x = v^2/c^2$: فبما أن $(1-x)(1+x) \approx 1$ و
$(1 + x/2)^2 \approx 1 + x$، نحصل على $\gamma \approx 1 + v^2/(2c^2)$. والسيارة
عند \qty{130}{km/h} لها $\gamma - 1 \approx \num{7e-15}$ (أي \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة تُفقد
في كل أربعة ملايين سنة من القيادة)؛ والطائرة النفّاثة تبلغ $\num{3.5e-13}$، أي \omterm{def:g10:orders-of-magnitude:unit}{ثانية}
واحدة في كل تسعين ألف سنة في الجو. فساعاتنا كانت ببساطة أخشن من اللازم.
\end{remark}

\section{حكم التجربة}

\begin{example}[الميونات تبلغ الأرض]\label{ex:g12:special-relativity:muons}
الأشعة الكونية التي تضرب الجو العلوي تخلق \emph{ميونات} ---
وهي جسيمات قصيرة العمر عمرها الذاتي $\Delta t_0 = \qty{2.2}{\micro s}$
--- على ارتفاع نحو \qty{15}{km}، متحركةً \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \omterm{def:g11:lenses-and-eye:vergence}{تقارب} $c$. وبلا نسبية يقطع الميون
$c\,\Delta t_0 \approx \qty{660}{m}$ في كل عمر: فيأخذ النزول
نحو $23$ عمرًا، والجزء الناجي
$\mathrm{e}^{-23} \approx \num{e-10}$ ينبغي أن يجعل ميونات مستوى البحر
نادرة إلى حدّ التلاشي. ومع ذلك تعدّ الكواشف نحو واحد في السنتيمتر المربع في
الدقيقة. ويحلّ \omterm{thm:g12:special-relativity:dilation}{تمدد الزمن} الأمر: فالميونات، إذ تصل بعامل $\gamma \approx 30$،
تعيش $30 \times \qty{2.2}{\micro s} = \qty{66}{\micro s}$ على
ساعاتنا --- أي \omterm{def:g12:projectile-motion:range}{مدى} $\approx \qty{20}{km}$، وهو الجو كله.
\end{example}

وتتراكم التأكيدات. ففي 1971 طار هافيل وكيتنغ بساعات سيزيوم ذرّية
(\cref{ch:g12:oscillators-and-time}) حول العالم على متن طائرات ركاب؛
وعند الهبوط اختلفت عن ساعات الأرض بأجزاء من الميكروثانية كما تنبّأ الحساب.
والمسرّعات تعوّل على التمدد يوميًّا، إذ تُدفع جسيماتها
إلى $\gamma$ فتنجو آلاف اللفّات التي لم تكن لتكملها
كلاسيكيًّا أبدًا. أما GPS فتجربة جارية لها ثمن.

\begin{remark}[فاتورة النسبية في GPS]\label{rem:g12:special-relativity:gps}
يدور \omterm{def:g10:universal-gravitation:satellite}{تابع} GPS \omterm{def:g12:kinematics-2d:velocity}{بسرعة} نحو \qty{3.9}{km/s}، ومن ثم تبطّئ \omterm{thm:g12:special-relativity:postulates}{النسبية الخاصة}
ساعته بنحو \qty{7}{\micro s} في اليوم. أما \omterm{def:g11:fundamental-interactions:four}{الجاذبية}، وهي أضعف على
الارتفاع، فتعمل في الاتجاه الآخر وتغلب: فالنسبية العامة (نظرية أينشتاين
في \omterm{def:g11:fundamental-interactions:four}{الجاذبية}، المرويّة في مجلدات الجامعة) تسرّعها بنحو
\qty{45}{\micro s} في اليوم؛ والصافي، \qty{38}{\micro s} في اليوم تقدّمًا.
وتحديد الموضع يحوّل قراءات الساعة إلى
مسافات \omterm{def:g12:kinematics-2d:velocity}{بالسرعة} $c$: فيوم بلا تصحيح يلطّخ المواضع بمقدار
$c \times \qty{38}{\micro s} \approx \qty{11}{km}$ --- ومن هنا الساعات
المزاح ضبطها في الفقرة الافتتاحية.
\end{remark}

\begin{proposition}[تقلص الأطوال]\label{prop:g12:special-relativity:contraction}
\index{تقلص الأطوال}
الجسم ذو الطول $L_0$ في \omterm{def:g10:relative-motion:frame}{مرجع} سكونه يُقاس أقصر على امتداد
اتجاه حركته في \omterm{def:g10:relative-motion:frame}{مرجع} يتحرك فيه \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v$:
$L = L_0/\gamma$.
\end{proposition}
\admitted

\begin{remark}[حكاية الميون كما يرويها هو]\label{rem:g12:special-relativity:muonframe}
\omterm{prop:g12:special-relativity:contraction}{تقلص الأطوال} هو الوجه الآخر \omterm{thm:g12:special-relativity:dilation}{لتمدد الزمن} (وهو مستنتَج استنتاجًا أمينًا في
مجلدات الجامعة): فساعة الميون تدقّ دقًّا عاديًّا في مرجعه، لكن
الجو المندفع صاعدًا متقلص إلى $\qty{15}{km}/30 =
\qty{500}{m}$، ويُقطع في أقل من \qty{2.2}{\micro s} بمريح --- فالمرجعان
يتفقان على أن الميون يبلغ الأرض.
\end{remark}

\section{الكتلة والطاقة والعالم الكلاسيكي مستعادًا}

\begin{theorem}[تكافؤ الكتلة والطاقة]\label{thm:g12:special-relativity:emc2}
\index{تكافؤ الكتلة والطاقة}
الجسم ذو الكتلة $m$ يملك، بكتلته وحدها، \omterm{def:g10:energy-conservation:energy}{الطاقة}
\[ E = m c^2 . \]
فالكتلة صورة مركَّزة من \omterm{def:g10:energy-conservation:energy}{الطاقة}؛ \omterm{def:g10:energy-conservation:energy}{والطاقة}، على العكس، لها كتلة.
\end{theorem}
\admitted

\begin{remark}[أين لاقيتها]\label{rem:g12:special-relativity:emc2seen}
هذا هو الدفتر وراء \omterm{def:g10:energy-conservation:forms}{الطاقة النووية} (\cref{ch:g12:nuclear-energy}):
\omterm{def:g12:nuclear-energy:defect}{فالنقص الكتلي} في تفاعل، مضروبًا في $c^2$، هو \omterm{def:g10:energy-conservation:energy}{الطاقة} المحرَّرة. وغرام
واحد من الكتلة هو \qty{9.0e13}{J} --- أي إنتاج يوم كامل من محطة كهرباء
كبيرة. أما الاستنتاج الأمين فيخصّ مجلدات الجامعة؛ و
المفاعلات والنجوم تعمل عليه على أي حال.
\end{remark}

\begin{remark}[نيوتن لم يكن مخطئًا]\label{rem:g12:special-relativity:newton}
ضَعْ $v \ll c$: فيكون $\gamma \to 1$، وتتفق الأزمنة، وتتوقف الأطوال عن التقلص،
ويصير التزامن مطلقًا --- فتعود ميكانيكا السنوات الثلاث الماضية
تامّة، بوصفها نهاية السرعات المنخفضة. فالنسبية لم تهدم
نيوتن؛ بل رسمت حدود إمبراطوريته. وهكذا تنمو الفيزياء: أي
بتنقيح النظريات، وكل جديدة تعيد القديمة حالةً حدّيّة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:1}
احسب $\gamma$، إلى ثلاثة \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية}، من أجل $v = 0.1c$، $0.6c$، $0.8c$، $0.995c$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:2}
سفينة تبحر \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $0.6c$؛ وميقاتيتها تنبض كل \qty{10.0}{s} بساعة
السفينة. فأيّ \omterm{def:g11:electric-gravitational-fields:field}{مجال} تقيس الأرض؟ وأيّهما \omterm{def:g12:special-relativity:proper-time}{الزمن الذاتي}، ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:3}
عند أي \omterm{def:g12:kinematics-2d:velocity}{سرعة} (كجزء من $c$) يكون $\gamma = 2$؟ و$\gamma = 10$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:4}
سيارة تسير \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{130}{km/h}. احسب $v/c$، ثم $\gamma - 1$
(\cref{rem:g12:special-relativity:everyday})؛ وكم يجب أن تسير حتى تتأخر
ساعتها \qty{1}{s}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:special-relativity:5}
في تجربة قطار أينشتاين، فسّر --- مستشهدًا بالمسلَّمة المستعملة ---
لماذا يستنتج الراكب أن الضربة \emph{الأمامية} جاءت أولًا، ولماذا
لا يكون أيّ من المراقبَين مخطئًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:6}
تعيش البيونات المشحونة $\Delta t_0 = \qty{2.6e-8}{s}$. وتغادر حزمة
مسرّعًا \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $0.99c$. احسب $\gamma$، والعمر في المختبر، و
\omterm{def:g12:projectile-motion:range}{المدى} الكلاسيكي $v\,\Delta t_0$، \omterm{def:g12:projectile-motion:range}{والمدى} المتوسط الفعلي.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:7}
من منحني $\gamma$، اقرأ السرعتين اللتين تعطيان $\gamma = 1.5$ و
$\gamma = 3$ (وتحقق بالحساب). وماذا يقول المقارب الشاقولي عند
$v = c$ عن تسريع جسم ذي كتلة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:8}
لساعة ضوء مرآتان تفصل بينهما $L = \qty{1.5}{m}$. احسب دقّتها في
السكون، ثم دقّتها مقيسةً من الأرض حين تبحر \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $0.8c$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:9}
مَن يقيس \omterm{def:g12:special-relativity:proper-time}{الزمن الذاتي} بين: (أ) دقّتين من ساعة يد --- لابسها،
أم مارّ بها؟ (ب) ولادة ميون وتفككه --- الميون، أم
المختبر؟ (ج) مغادرة الأرض وبلوغ المريخ --- الطيار، أم مركز التحكم بالمهمة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:10}
تدور المحطة الفضائية \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{7.7}{km/s}. احسب
$\gamma - 1 \approx v^2/2c^2$، ثم بكم تتأخر ساعة رائدة فضاء
عن ساعة أرضية بعد ستة أشهر (\qty{1.58e7}{s}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:special-relativity:11}
احسب محتوى \omterm{def:g10:energy-conservation:energy}{الطاقة} $E = mc^2$ في \qty{1.0}{g} من المادة. وقارنه
بالإنتاج اليومي لمحطة كهرباء قدرها \qty{1.0}{GW}، وبالطاقة الحركية
\qty{8.5e5}{J} لسيارة عند \qty{130}{km/h}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:12}
بيّن من $\gamma = 1/\sqrt{1 - v^2/c^2}$ أن $v = c\sqrt{1 - 1/\gamma^2}$؛
واستنتج \omterm{def:g12:kinematics-2d:velocity}{سرعة} ميونات $\gamma = 30$، إلى خمسة \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية}
\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} $c$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:13}
أعِد رواية رحلة الميون من مرجعه هو: فكم يبلغ سُمك الجو
عند $\gamma = 30$، وكم يستغرق مروره عند $0.99944c$، وهل يسع ذلك
في عمر ذاتي واحد؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:14}
مسرّع خطي يدفع إلكترونات إلى $\gamma = \num{1.0e5}$ على امتداد أنبوب طوله
\qty{3.2}{km}. فكم تستغرق الرحلة في المختبر؟ وكم في
\omterm{def:g10:relative-motion:frame}{مرجع} الإلكترون، وكم يبلغ ``طول'' الأنبوب هناك؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:special-relativity:15}
تقع بروكسيما قنطورس على بعد $4.2$ \omterm{def:g10:orders-of-magnitude:lightyear}{سنة ضوئية} (\omterm{def:g10:orders-of-magnitude:lightyear}{والسنة الضوئية} هي المسافة
التي يقطعها الضوء في سنة). ومسبار يبحر إليها \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $0.9c$: فكم تستغرق
الرحلة على تقويم الأرض، وعلى ساعة المسبار؟
\end{exercise}

\section{مسألة: الميون والتوأم}

\begin{problem}[{مسألة نهاية الأسبوع --- رحلة الميون وتوأم رائدة
الفضاء: جسيم يعمّر أطول من ساعته، وساعة ضوء معاد بناؤها، ومسافرة
أصغر سنًّا من توأمها، والمحاسبة التي تُبقي GPS أمينًا}]
\label{pb:g12:special-relativity:1}
المعطيات: العمر الذاتي للميون $\Delta t_0 = \qty{2.2}{\micro s}$؛ والميونات تولد
على ارتفاع نحو \qty{15}{km} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \omterm{def:g11:lenses-and-eye:vergence}{تقارب} $c$؛ وتدفق مستوى البحر نحو ميون واحد في
\unit{cm^2} في الدقيقة؛ $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{الجزء I --- وصول الميون المستحيل.}
\begin{enumerate}
  \item بلا نسبية، كم يسافر الميون في عمر واحد؟
  \item وكم عمرًا يأخذ النزول من \qty{15}{km} عندئذٍ؟
  \item وبعد $n$ عمر ينجو الجزء $\mathrm{e}^{-n}$:
        قدّره، وواجه به التدفق المقيس، وأصدر حكمًا على
        الفيزياء الكلاسيكية في جملة واحدة.
  \item وهذه الميونات تصل في الواقع بعامل $\gamma \approx 30$. احسب
        عمرها مقيسًا من الأرض \omterm{def:g12:projectile-motion:range}{والمدى} الناتج.
  \item احسب سرعتها، مستعملًا $v = c\sqrt{1 - 1/\gamma^2}$، إلى خمسة
        \omterm{def:g10:orders-of-magnitude:sigfig}{أرقام معنوية} \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} $c$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- ساعة الضوء، معاد بناؤها.}
ساعة ضوء ($L = \qty{1.5}{m}$) تركب سفينة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v = 0.6c$.
\begin{enumerate}[resume]
  \item احسب الدقّة $\Delta t_0$ المقروءة على المتن.
  \item ومن الأرض، تدوم الدقّة $\Delta t$: أعطِ التقدّم الأفقي
        في كل نصف دقّة، وصعود \omterm{def:g12:quantum-world:photon}{الفوتون} الشاقولي، و--- مع تسمية
        المسلَّمة التي تثبّته --- طول نصف \omterm{def:g10:relative-motion:trajectory}{المسار} المائل.
  \item طبّق فيثاغورس على مثلث نصف الدقّة وحُلّ لإيجاد
        $\Delta t$، مبيّنًا أن $\Delta t = \gamma\,\Delta t_0$.
  \item احسب $\gamma$ و$\Delta t$ عدديًّا.
  \item وأيّ مراقب يقيس \omterm{def:g12:special-relativity:proper-time}{الزمن الذاتي} بين دقّتين؟ برّر
        انطلاقًا من التعريف.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- المسافرة وتوأمها.}
تترك رائدة فضاء توأمها على الأرض في رحلة ذهاب وإياب \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $0.8c$؛ وتوقّت
ساعات الأرض الرحلة كلها بالمدة \qty{10.0}{years}.
\begin{enumerate}[resume]
  \item احسب $\gamma$ عند $0.8c$ (كسرًا مضبوطًا، ثم عشريًّا).
  \item وكم من الزمن ينقضي على ساعة رائدة الفضاء؟
  \item وعلى أي بعد (في \omterm{def:g10:relative-motion:frame}{مرجع} الأرض، بالسنوات الضوئية) وقعت نقطة الاستدارة؟
  \item وما فرق العمر عند اللقاء، وأيّ التوأمين أصغر؟
  \item ``لكن الحركة نسبية --- فتقدر رائدة الفضاء أن تدّعي أن الأرض هي التي
        سافرت!'' حُلّ الاعتراض في جملة واحدة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- فاتورة GPS.}
\omterm{def:g10:universal-gravitation:satellite}{تابع} GPS يدور بنصف قطر $r = \qty{2.66e7}{m}$، ودوره $T = \qty{12}{h}$.
\begin{enumerate}[resume]
  \item احسب سرعته المدارية $v = 2\pi r/T$.
  \item احسب $\gamma - 1 \approx v^2/2c^2$.
  \item استنتج، بالميكروثانية، بكم تتأخر ساعة \omterm{def:g10:universal-gravitation:satellite}{التابع} عن
        ساعات الأرض في كل يوم (\qty{86400}{s}).
  \item ولو تُرك هذا الانحراف وحده بلا تصحيح، لأخطأت المواضع بمقدار
        $c \times$ (الانحراف): احسب الخطأ بعد يوم واحد.
  \item \omterm{def:g11:fundamental-interactions:four}{والجاذبية}، وهي أضعف على الارتفاع، تشغّل الساعة \emph{متقدمة} بمقدار
        \qty{45}{\micro s} في اليوم (بالنسبية العامة): احسب الانحراف
        الصافي وخطأ الموضع اليومي بالكيلومترات --- أي الرقم
        الذي يُلزم المهندسين بإزاحة ضبط كل ساعة GPS.
\end{enumerate}
\end{problem}

هنا تنتهي فيزياء المدرسة --- لا الفيزياء. ففي مجلدات
الجامعة، تعود الميكانيكا مسلَّحة بالتفاضل والتكامل؛ وتكشف الكهرباء والمغناطيسية
عن نفسيهما بنيةً واحدة كان الضوء، والنسبية نفسها،
مختبئين فيها منذ البداية؛ ويصير العالم الكمومي، الذي لُمح في الفصل السابق،
نظريةً عاملة للمادة. وستُلزَم كل نظرية، كالنسبية، بأن
تعيد ما تعرفه الآن حالةً حدّيّة. فاحملْه معك: فهو
الجزء الذي لن يتغير.
