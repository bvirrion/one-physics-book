\chapter{قوانين نيوتن}\label{ch:g12:newtons-laws}

هرّب ميزان حمّام إلى مصعد. فحين تشدّ المقصورة إلى أعلى تقرأ الإبرة
زيادة ثلاثة كيلوغرامات؛ وقرب القمة تقرأ نقصانًا؛ وبين الحالتين، الحقيقة العارية
--- إذ لم يتغير شيء على المتن إلا الحركة. ويذكر هذا الفصل القوانين الثلاثة وراء
تلك الإبرة ويصوغها في الطريقة التي تدير الميكانيك كله: اختر
جملة، وارسم قواها، وأسقِط $\sum \vect F = m\vect a$.

\section{كمية الحركة، والقانون الأول مضبوطًا}

\begin{definition}[كمية الحركة]\label{def:g12:newtons-laws:momentum}
\emph{كمية الحركة}\index{كمية الحركة} لجسم كتلته $m$ (\unit{kg}) يتحرك \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
$\vect v$ هي المتجهة
\[
\vect p = m \vect v \qquad (\unit{kg.m/s}):
\]
أي \omterm{def:g12:kinematics-2d:velocity}{السرعة} موزونةً بالكتلة؛ وفي جملة، هي المجموع المتجهي على أجزائها.
\end{definition}

\begin{definition}[المرجع العُطالي]\label{def:g12:newtons-laws:inertial}
\emph{المرجع العُطالي}\index{مرجع عطالي} \omterm{def:g10:relative-motion:frame}{مرجع} يتحرك فيه
كل جسم معزول --- لا قوى عليه، أو قوى تتعادل --- في خط مستقيم
\omterm{def:g12:kinematics-2d:velocity}{بسرعة} ثابتة (\cref{ch:g10:inertia}). والأرض \omterm{def:g10:relative-motion:frame}{مرجع} كهذا بدقة ممتازة،
وكذلك أي مقصورة في \omterm{def:g12:kinematics-2d:uniform}{حركة مستقيمة منتظمة} عليها؛ أما قطار يكبح
أو دوّامة تدور فلا: إذ تتسارع فيهما الأمتعة السائبة دون أن يدفعها شيء.
\end{definition}

\begin{theorem}[قانون نيوتن الأول]\label{thm:g12:newtons-laws:first}\index{قانون نيوتن الأول}
في \omterm{def:g10:relative-motion:frame}{مرجع} عُطالي، لا تتغير كمية حركة جملة معزولة: أي أن $\vect p$
ثابتة، أي\ $d\vect p/dt = \vect 0$.
\end{theorem}
\admitted

\begin{remark}[العطالة، مشحوذة مرتين]\label{rem:g12:newtons-laws:sharpened}
مبدأ العطالة في \cref{ch:g10:inertia}، مرقًّى مرتين: فهو يقول
\emph{أين} يصحّ --- إذ يعرّف المراجع العُطالية بأنها التي يصحّ فيها ---
ويحرس متجهةً، هي $m\vect v$، تؤتي محاسبتها ثمارها في آخر الفصل.
\end{remark}

\section{قانون نيوتن الثاني}

\begin{theorem}[قانون نيوتن الثاني]\label{thm:g12:newtons-laws:second}\index{قانون نيوتن الثاني}
في \omterm{def:g10:relative-motion:frame}{مرجع} عُطالي، تساوي \omterm{def:g11:forces-and-motion:netforce}{القوة المحصّلة} على جسم معدلَ تغير
كمية حركته؛ وفي حالة كتلة ثابتة، وبما أن $\vect p = m\vect v$، فإن
\[
\sum \vect F = \frac{d\vect p}{dt} = m\,\frac{d\vect v}{dt} = m \vect a .
\]
\end{theorem}
\admitted

\begin{remark}[قانون طبيعة، ووحدته]\label{rem:g12:newtons-laws:nature}
لا برهان: فهذا قانون طبيعة، وُزن في مواجهة التجربة ثلاثة
قرون. وقد أعطى \cref{ch:g11:forces-and-motion} في السنة الماضية ظلَّه،
$\Delta v \propto F\,\Delta t/m$؛ أما الاشتقاق (\cref{ch:g12:kinematics-2d}) فيجعله
مضبوطًا في كل لحظة --- وتنتظر تفاصيله الدقيقة في مجلد السنة الأولى. وتنال \omterm{def:g10:inertia:force}{القوة}
\omterm{def:g10:orders-of-magnitude:unit}{وحدة}، $\unit{N} = \unit{kg.m/s^2}$؛ ويخضع جسم تحت وزنه وحده للعلاقة
$m\vect a = m\vect g$: فكل كتلة تسقط بتسارع $\vect a = \vect g$، و
$g = \qty{9.81}{N/kg} = \qty{9.81}{m/s^2}$ --- فالقراءتان واحدة.
\end{remark}

\begin{example}[القوة المتوسطة، بلا حاجة إلى التفاصيل]\label{ex:g12:newtons-laws:average}
سيارة كتلتها \qty{1300}{kg} تنتقل من السكون إلى \qty{100}{km/h} (\qty{27.8}{m/s}) في
\qty{8.0}{s}: فمهما تكن القصة لحظةً بلحظة، تكون \omterm{def:g11:forces-and-motion:netforce}{القوة المحصّلة} المتوسطة
$\Delta p/\Delta t = 1300 \times 27.8/8.0 \approx \qty{4.5e3}{N}$ --- \omterm{def:g12:newtons-laws:momentum}{فكمية الحركة} تحوّل
سؤالًا عن \omterm{def:g10:inertia:force}{قوة} إلى محاسبة قبل وبعد.
\end{example}

\section{قانون نيوتن الثالث}

\begin{theorem}[قانون نيوتن الثالث]\label{thm:g12:newtons-laws:third}\index{قانون نيوتن الثالث}
إذا مارس جسم $A$ قوةً $\vect F_{A \to B}$ على جسم $B$، مارس $B$ على
$A$ \omterm{def:g10:inertia:force}{القوة} $\vect F_{B \to A} = -\vect F_{A \to B}$: بمستقيم التأثير نفسه، وبالشدة
نفسها، وباتجاهين متعاكسين --- في كل لحظة، ومهما تكن الحركة، تماسًّا كان أو
تأثيرًا عن بعد.
\end{theorem}
\admitted

\begin{example}[الكتاب والطاولة والأرض]\label{ex:g12:newtons-laws:book}
كتاب يستقر على طاولة: تفاعلان، وزوجان. فالثقالي: الأرض تجذب
الكتاب إلى أسفل ($\vect P$)، والكتاب يجذب الأرض إلى أعلى. والتماسّي: الطاولة تدفع
الكتاب إلى أعلى ($\vect N$)، والكتاب يضغط الطاولة إلى أسفل. والمشي الصفقة نفسها:
ادفع الأرض إلى الوراء؛ فرد فعلها هو \omterm{def:g10:inertia:force}{القوة} الوحيدة التي تدفعك إلى الأمام.
\end{example}

\section{الطريقة: اختر وارسم وأسقِط}

\begin{method}[حلّ مسألة في الديناميك]\label{met:g12:newtons-laws:method}
\begin{enumerate}
  \item \emph{الجملة}: سمِّ الجسم (أو مجموعة الأجسام) المدروس.
  \item \emph{\omterm{def:g10:relative-motion:frame}{المرجع}}: اختر مرجعًا عُطاليًّا ومحاور ملائمة للحركة ---
        محاور المنحدر، أو \omterm{def:g12:kinematics-2d:frenet}{معلم فريني} في منحنٍ (\cref{ch:g12:kinematics-2d}).
  \item \emph{الجرد}: أحصِ كل \omterm{def:g10:inertia:force}{قوة} --- \omterm{def:g10:universal-gravitation:weight}{الوزن}، ثم \omterm{def:g10:inertia:force}{قوة} واحدة لكل
        تماس، ولا شيء غير ذلك --- وارسم \omterm{def:g10:inertia:diagram}{مخطط القوى} المعزول: نقطة واحدة، وسهم
        واحد لكل \omterm{def:g10:inertia:force}{قوة}.
  \item \emph{الإسقاط}: اكتب $\sum \vect F = m\vect a$، وأسقِطها على كل محور.
  \item \emph{حُلّ وتحقّق}: من الوحدات والإشارات والحالات الحدية ($\mu \to 0$،
        $\alpha \to 0$، \dots).
\end{enumerate}
\end{method}

\begin{definition}[القوة الناظمية والاحتكاك]\label{def:g12:newtons-laws:friction}
يؤثر السطح في جسم بمركّبتين: \emph{القوة الناظمية}\index{قوة
ناظمية} $\vect N$، وهي عمودية على السطح، \omterm{def:g10:inertia:inventory}{والاحتكاك} $\vect f$، وهو مماسّي.
وتعطي التجربة قوانين \omterm{def:g10:inertia:inventory}{الاحتكاك}: فما دام الجسم لا ينزلق، يضبط \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني}
نفسه حتى سقف قدره $f_s \leq \mu_s N$؛ ومتى انزلق، صار \omterm{def:g11:forces-and-motion:friction}{الاحتكاك الحركي}
$f_k = \mu_k N$، ضد الانزلاق. و\emph{معاملا \omterm{def:g10:inertia:inventory}{الاحتكاك}}%
\index{معامل الاحتكاك} $\mu_s$ و$\mu_k$ ($\mu_k \leq \mu_s$) لا يتوقفان إلا
على المادتين.
\end{definition}

\begin{example}[المنحدر، مرتين]\label{ex:g12:newtons-laws:incline}
تنزلق كتلة على منحدر زاويته $\alpha$؛ والمحاور: $x$ نازلًا على المنحدر، و$y$
عموديًّا عليه. فعلى $y$: $N = mg\cos\alpha$. وعلى $x$، بلا \omterm{def:g10:inertia:inventory}{احتكاك}: $ma = mg\sin\alpha$،
ومنه $a = g\sin\alpha$ --- بلا كتلة؛ وعند $\alpha = 30^\circ$، $a = \qty{4.9}{m/s^2}$.
ومع \omterm{def:g10:inertia:inventory}{احتكاك} حركي: $a = g(\sin\alpha - \mu_k\cos\alpha)$؛ ومن أجل $\mu_k = 0.20$،
$a = 9.81\,(0.500 - 0.173) = \qty{3.2}{m/s^2}$. أما بقاؤها في مكانها أصلًا فلم يكن
إلا ما دام \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني} اللازم يسع تحت سقفه: $\tan\alpha \leq \mu_s$.
\end{example}

\begin{definition}[الوزن الظاهري]\label{def:g12:newtons-laws:apparent}
\emph{الوزن الظاهري}\index{وزن ظاهري} لجسم هو \omterm{def:g12:newtons-laws:friction}{القوة الناظمية}
بينه وبين سنده --- أي ما يقرؤه ميزان تحته.
\end{definition}

\begin{example}[المصعد]\label{ex:g12:newtons-laws:elevator}
مسافر كتلته \qty{70}{kg} يقف على ميزان في مصعد تسارعه العمودي
$a_z$ (وإلى أعلى موجب): فيكون $N - mg = ma_z$، ومنه $N = m(g + a_z)$. وعند \omterm{def:g10:inertia:inventory}{الشدّ} إلى أعلى
بمقدار $a_z = \qty{1.5}{m/s^2}$: \qty{792}{N}، فيثقل؛ وعند الكبح قرب القمة: \qty{582}{N}،
فيخفّ؛ وفي الانسياب الثابت: $mg = \qty{687}{N}$ --- أي القانون الأول؛ وفي السقوط الحر، $a_z = -g$:
$N = 0$، أي انعدام \omterm{def:g10:universal-gravitation:weight}{وزن} على ميزان.
\end{example}

\begin{example}[كتلتان وبكرة]\label{ex:g12:newtons-laws:pulley}
كتلة $m_1 = \qty{4.0}{kg}$ على طاولة بلا \omterm{def:g10:inertia:inventory}{احتكاك} مربوطة بحبل خفيف، فوق
بكرة مثالية، بكتلة معلّقة $m_2 = \qty{1.0}{kg}$. وينقل الحبل \omterm{def:g10:inertia:inventory}{الشدّ}
$T$ نفسه عند الطرفين؛ وتتقاسم الكتلتان قيمة تسارع واحدة $a$.
وقانون ثانٍ لكلٍّ، على امتداد حركتها: $m_1 a = T$ و$m_2 a = m_2 g - T$، ومنه
$a = m_2\,g/(m_1 + m_2) = \qty{1.96}{m/s^2}$ و$T = m_1 a = \qty{7.8}{N}$ ---
وهو أصغر من $m_2 g = \qty{9.8}{N}$، كما يجب لتتسارع $m_2$ إلى أسفل.
\end{example}

\begin{example}[المنعطف المستوي]\label{ex:g12:newtons-laws:curve}
سيارة كتلتها $m$ تدور في منعطف مستوٍ نصف قطره $R$ \omterm{def:g12:kinematics-2d:velocity}{بسرعة} ثابتة $v$. وفي
\omterm{def:g12:kinematics-2d:frenet}{معلم فريني} (\cref{ch:g12:kinematics-2d}) يكون التسارع ناظميًّا خالصًا،
$a_N = v^2/R$، مصوَّبًا نحو المركز. وعموديًّا، $N = mg$؛ وأفقيًّا، \omterm{def:g10:inertia:force}{القوة}
المركزية الوحيدة المعروضة هي \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني} بين الطريق والإطارات:
$f = mv^2/R \leq \mu_s N = \mu_s mg$، ومنه $v \leq \sqrt{\mu_s\, g R}$، مهما تكن
الكتلة. ومن أجل $R = \qty{90}{m}$، وطريق جافّ ($\mu_s = 0.70$):
$v_{\max} \approx \qty{25}{m/s}$ (\qty{89}{km/h})؛ وعلى الجليد ينهار السقف ---
والعلاقة \cref{exo:g12:newtons-laws:13} تميل الطريق؛ و\cref{exo:g12:newtons-laws:10} يؤرجح
الإسقاط نفسه على سلك.
\end{example}

\section{الجمل المعزولة: كمية الحركة محفوظة}

\begin{proposition}[انحفاظ كمية الحركة]\label{prop:g12:newtons-laws:conservation}\index{انحفاظ كمية الحركة}
في جسمين تنعدم القوى الخارجية عليهما أو تتعادل، تكون \omterm{def:g12:newtons-laws:momentum}{كمية الحركة} الكلية
$\vect p_1 + \vect p_2$ ثابتة، مهما تكن القوى التي يمارسها أحدهما على الآخر.
\end{proposition}

\begin{proof}
اجمع القانونين الثانيين: $\frac{d}{dt}(\vect p_1 + \vect p_2) = \vect F_{2 \to 1} +
\vect F_{1 \to 2} + \vect F_{\text{الخارجية}}$: فيتلاشى الزوج الداخلي بالقانون
الثالث، و$\vect F_{\text{الخارجية}} = \vect 0$ بالفرض.
\end{proof}

\begin{example}[الارتداد]\label{ex:g12:newtons-laws:recoil}
متزلجة كتلتها \qty{60}{kg} ساكنة على جليد أملس تقذف كرة طبية كتلتها \qty{4.0}{kg}
أفقيًّا \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{6.0}{m/s}. فقبل: $\vect p = \vect 0$؛ وبعد:
$0 = 60\,V + 4.0 \times 6.0$، ومنه $V = \qty{-0.40}{m/s}$ --- فتنجرف إلى الوراء.
ولم يدفعها شيء خارجي: بل دفعتها الكرة (بالقانون الثالث)، وتبقى الدفاتر عند الصفر.
\end{example}

\begin{remark}[كيف يدفع صاروخ لا شيء]\label{rem:g12:newtons-laws:rocket}
الصاروخ آلة لرمي الكتلة. ففي كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية} تقذف محركاته دفعة من الغاز
إلى الوراء؛ ويحمل الغاز كمية حركة خلفية، فيكسب الصاروخ القدر
نفسه إلى الأمام --- أي \omterm{def:g10:inertia:force}{قوة} دفع ثابتة، تساوي تقريبًا الكتلة المقذوفة في \omterm{def:g10:orders-of-magnitude:unit}{الثانية} مضروبةً في
\omterm{def:g12:kinematics-2d:velocity}{سرعة} القذف. فالصاروخ يدفع عادمه هو، لا الهواء: ولذلك يعمل أفضل ما يعمل في
الفضاء الخالي.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:1}
احسب كمية حركة (أ) عدّاء كتلته \qty{70}{kg} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{10}{m/s}؛ (ب) سيارة كتلتها
\qty{1300}{kg} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{50}{km/h}؛ (ج) رصاصة كتلتها \qty{8.0}{g} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{800}{m/s}.
ورتّبها. ولماذا يأتي الأسرع في المرتبة الأخيرة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:2}
مسافر كتلته \qty{70}{kg} يقف على ميزان في مصعد. فماذا يقرأ الميزان حين
تتسارع المقصورة (أ) إلى أعلى بمقدار \qty{1.2}{m/s^2}؛ (ب) تصعد \omterm{def:g12:kinematics-2d:velocity}{بسرعة} ثابتة
\qty{2.0}{m/s}؛ (ج) تتسارع إلى أسفل بمقدار \qty{1.2}{m/s^2}؟ وأيّ قانون يحسم (ب)؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:3}
تُطلق كتلة على منحدر بلا \omterm{def:g10:inertia:inventory}{احتكاك} زاويته $30^\circ$. احسب
تسارعها، ثم سرعتها بعد انزلاق \qty{3.0}{m}. وأين ذهبت الكتلة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:4}
تصل كرة تنس كتلتها \qty{58}{g} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{15}{m/s} وتُردّ على المستقيم
نفسه \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{25}{m/s}؛ ويدوم التماس \qty{5.0}{ms}. احسب تغير \omterm{def:g12:newtons-laws:momentum}{كمية الحركة}
\omterm{def:g10:inertia:force}{والقوة} المتوسطة للأوتار؛ وقارن تلك \omterm{def:g10:inertia:force}{القوة} \omterm{def:g10:universal-gravitation:weight}{بوزن} الكرة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:5}
كتاب يستقر على طاولة. سمِّ القوتين المؤثرتين في الكتاب، ولكلٍّ منهما
شريكها في زوج الفعل ورد الفعل (أيّ جسم؟ وأيّ اتجاه؟). ولماذا لا يكون \omterm{def:g10:universal-gravitation:weight}{وزن} الكتاب
ودفع الطاولة شريكين أحدهما للآخر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:6}
صندوق كتلته \qty{50}{kg} يستقر على أرضية أفقية، و$\mu_s = 0.45$، و$\mu_k = 0.35$.
(أ) فأيّ دفعة أفقية تكفي لتحريكه؟ (ب) وإذا حُفظت تلك الدفعة، فما
تسارعه؟ (ج) وأيّ دفعة تبقيه متحركًا \omterm{def:g12:kinematics-2d:velocity}{بسرعة} ثابتة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:7}
يُوضع الصندوق نفسه على منحدر قابل للضبط ($\mu_s = 0.45$، $\mu_k = 0.35$). (أ) فعند
أي زاوية يبدأ بالانزلاق؟ (ب) وعند $30^\circ$، احسب تسارعه.
(ج) ومتى انطلق عند زاوية (أ)، فهل يظل متسارعًا؟ ولماذا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:8}
كتلة \qty{3.0}{kg} على طاولة بلا \omterm{def:g10:inertia:inventory}{احتكاك} مربوطة فوق بكرة مثالية بكتلة
معلّقة \qty{2.0}{kg}. احسب التسارع \omterm{def:g10:inertia:inventory}{والشدّ}. ولماذا وجب أن يخرج
\omterm{def:g10:inertia:inventory}{الشدّ} أصغر من \omterm{def:g10:universal-gravitation:weight}{وزن} الكتلة المعلّقة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:9}
منعطف نصف قطره \qty{50}{m} مستوٍ. احسب أقصى \omterm{def:g12:kinematics-2d:velocity}{سرعة} لسيارة من أجل
$\mu_s = 0.70$ (جافّ) و$0.40$ (مبتلّ)، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{km/h}. ولماذا يصحّ الجواب
لدراجة نارية صغيرة ولشاحنة سواءً؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:10}
كرة كتلتها \qty{0.30}{kg} على سلك طوله \qty{1.2}{m} ترسم دائرة أفقية، والسلك
يحفظ زاوية ثابتة $25^\circ$ مع العمودي (وهو نوّاس مخروطي). أسقِط
القانون الثاني عموديًّا، ثم على امتداد نصف القطر: واحسب \omterm{def:g10:inertia:inventory}{الشدّ} \omterm{def:g12:kinematics-2d:velocity}{والسرعة}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:11}
طرد كتلته \qty{5.0}{kg} يركب ميزانًا في مصعد شحن؛ والقراءات:
\begin{center}\small\begin{tabular}{lccc}
الطور & \qtyrange{0}{2}{s} & \qtyrange{2}{8}{s} & \qtyrange{8}{10}{s}\\
الميزان (\unit{N}) & 54.0 & 49.0 & 44.1
\end{tabular}\end{center}
احسب التسارع في كل طور وصِف رحلة ممكنة. وهل كان يمكن أن يكون المصعد
متحركًا \emph{إلى أسفل} طول الوقت؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:12}
متزلجة كتلتها \qty{60}{kg} ساكنة على جليد أملس تقذف كرة كتلتها \qty{4.0}{kg}
أفقيًّا. (أ) تغادر الكرة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{8.0}{m/s} \emph{بالنسبة إلى الأرض}: فما
\omterm{def:g12:kinematics-2d:velocity}{سرعة} ارتدادها؟ (ب) ثم تُقاس \qty{8.0}{m/s} \emph{بالنسبة إليها}:
أعِد حساب السرعتين. (ج) وأيّ \omterm{def:g10:inertia:force}{قوة} دفعتها إلى الوراء؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:13}
منعطف طريق سريع نصف قطره \qty{80}{m} مصمَّم \omterm{def:g12:kinematics-2d:velocity}{لسرعة} \qty{20}{m/s} بلا مساعدة من
\omterm{def:g10:inertia:inventory}{الاحتكاك}. بيّن أنه يجب إمالة سطح الطريق بالزاوية $\theta$ المعطاة بالعلاقة
$\tan\theta = v^2/(gR)$ واحسبها. وماذا يجب أن يفعل \omterm{def:g10:inertia:inventory}{الاحتكاك} في حالة سيارة تأخذ
المنعطف أبطأ --- أو أسرع؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:14}
تقذف محركات صاروخ \qty{250}{kg} من الغاز في \omterm{def:g10:orders-of-magnitude:unit}{الثانية} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{2500}{m/s}. (أ) فكم
كمية حركة تغادر في \omterm{def:g10:orders-of-magnitude:unit}{الثانية} --- وأيّ \omterm{def:g10:inertia:force}{قوة} دفع تردّها الغازات؟ (ب) وكتلة الصاروخ
\qty{5.0e4}{kg} عند الإشعال: فما وزنه؟ وما تسارعه عند الإقلاع؟ (ج) ولماذا
يعمل المحرك بالكفاءة نفسها في الفضاء الخالي؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:15}
كتلة \qty{4.0}{kg} على طاولة ($\mu_s = 0.35$، $\mu_k = 0.25$) مربوطة فوق
بكرة مثالية بكتلة معلّقة $m_2$. (أ) فأيّ $m_2$ دنيا تُبدئ الحركة؟
(ب) ومن أجل $m_2 = \qty{2.0}{kg}$، احسب التسارع \omterm{def:g10:inertia:inventory}{والشدّ}. (ج) وتحقّق من أن
$\mu_k = 0$ يستعيد صيغة انعدام \omterm{def:g10:inertia:inventory}{الاحتكاك} في \cref{exo:g12:newtons-laws:8}.
\end{exercise}

\section{مسألة: العربة المعلّقة}

\begin{problem}[{مسألة نهاية الأسبوع --- اعتماد العربة المعلّقة: الشدّ الذي يقتضيه المنحدر
في السكون، والزيادة عند الانطلاق، والراحة فوق البرج، وثمن
التوقف الطارئ --- والعدد الوحيد الذي تجوز للافتة في المقصورة أن تَعِد به}]
\label{pb:g12:newtons-laws:1}
عربة جبلية معلّقة --- مقصورة مع عربة، وكتلتها فارغة $M_0 = \qty{2.0e3}{kg}$ ---
تسير على كابل حمل مائل بزاوية $\alpha = 30^\circ$، يجرّها كابل جرّ
موازٍ للمنحدر \omterm{def:g10:inertia:inventory}{بشدّ} $\vect T$. \omterm{def:g10:inertia:inventory}{واحتكاك} التدحرج مهمَل؛ و
تبقى أرضية المقصورة أفقية طول الوقت. وتسع المقصورة $25$ مسافرًا متوسط
كتلتهم \qty{80}{kg}؛ وأنت مهندس الاعتماد. وما لم يُذكر خلاف ذلك، تكون
المقصورة ممتلئة: $M = \qty{4.0e3}{kg}$.

\medskip
\noindent\textbf{الجزء I --- في السكون على المنحدر.}
\begin{enumerate}
  \item اختر الجملة؛ وقل لماذا يفي \omterm{def:g10:relative-motion:frame}{المرجع} الجبلي بأن يكون مرجعًا عُطاليًّا؛
        واجرد القوى الثلاث باتجاهاتها.
  \item وأيّ قانون يحكم الانتظار؟ أسقِطه على امتداد المنحدر وعموديًّا عليه
        للتعبير عن $T$ وعن \omterm{def:g12:newtons-laws:friction}{القوة الناظمية} لكابل الحمل $N$.
  \item احسب $T$ و$N$.
  \item أعِد حساب $T$ للمقصورة الفارغة. ولماذا تتناسب $T$ مع الكتلة الكلية؟
  \item ينقطع كابل الجرّ والمكابح مطفأة: بيّن أن تسارع الانفلات
        $g \sin\alpha$، محمَّلةً كانت أو فارغة، واحسبه.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الانطلاق.} يسرّع الملفاف المقصورة الممتلئة بمقدار
$a = \qty{0.60}{m/s^2}$ صاعدةً على المنحدر حتى تنساب \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v = \qty{6.0}{m/s}$.
\begin{enumerate}[resume]
  \item عبّر عن \omterm{def:g10:inertia:inventory}{الشدّ} $T'$ خلال هذا الطور واحسبه.
  \item وبأي كسر يرفع الانطلاق الشدَّ فوق قيمته في السكون؟
  \item وكم يدوم الطور، وعلى أي مسافة؟
  \item وتقف مسافرة كتلتها \qty{80}{kg} على الأرضية الأفقية. أسقِط القانون
        الثاني عموديًّا للتعبير عن وزنها الظاهري وحسابه؛ وقارنه بالمقدار $mg$.
  \item وأيّ \omterm{def:g10:inertia:force}{قوة} تسرّعها أفقيًّا، وكم مقدارها؟ ولماذا يميل المسافرون؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- فوق البرج.} عند \omterm{def:g12:kinematics-2d:velocity}{سرعة} ثابتة \qty{6.0}{m/s}، ينحني
كابل الحمل فوق برج سند على قوس دائري عمودي نصف قطره
$R = \qty{40}{m}$؛ وعند القمة تكون \omterm{def:g12:kinematics-2d:velocity}{السرعة} أفقية.
\begin{enumerate}[resume]
  \item في \omterm{def:g12:kinematics-2d:frenet}{معلم فريني} عند القمة (\cref{ch:g12:kinematics-2d})، أعطِ مركّبتَي
        التسارع كلتيهما بقيمتيهما.
  \item أسقِط القانون الثاني على \omterm{def:g10:refraction:angles}{الناظم} للتعبير عن \omterm{def:g12:newtons-laws:friction}{القوة الناظمية}
        $N'$ للكابل على العربة وحسابها.
  \item والسؤال نفسه \omterm{def:g12:newtons-laws:apparent}{للوزن الظاهري} للمسافرة عند القمة: بأي كسر
        تخفّ؟
  \item وعند أي \omterm{def:g12:kinematics-2d:velocity}{سرعة} تنعدم $N'$؟ قارنها \omterm{def:g12:kinematics-2d:velocity}{بسرعة} الانسياب وعلّق.
  \item وأيّ \omterm{def:g10:inertia:force}{قوة} تمارسها العربة على الكابل فوق البرج --- وأيّ قانون
        يقول ذلك، وبأي شدة واتجاه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الكبح، وقفزة، واللافتة.}
\begin{enumerate}[resume]
  \item على مقطع الوصول الأفقي، يأخذ توقف طارئ المقصورة الممتلئة
        من \qty{6.0}{m/s} إلى السكون في \qty{1.5}{s}: اقرأ $\sum \vect F = d\vect p/dt$
        متوسطًا على مدى التوقف لحساب \omterm{def:g10:inertia:force}{قوة} الكبح.
  \item وأيّ \omterm{def:g10:inertia:force}{قوة} أفقية يجب أن توقف مسافرة كتلتها \qty{80}{kg}، وهل يقدر
        \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني} بين الحذاء والأرضية ($\mu_s = 0.40$) على توفيرها؟ استنتج.
  \item وعند الرصيف، والمكابح مطفأة، تتدلى المقصورة الفارغة حرةً ساكنة؛ ويقفز آخر
        مسافر (\qty{80}{kg}) أفقيًّا \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{2.5}{m/s}
        \emph{بالنسبة إلى المقصورة}. احسب \omterm{def:g12:kinematics-2d:velocity}{سرعة} ارتداد المقصورة و
        \omterm{def:g12:kinematics-2d:velocity}{سرعة} المسافر بالنسبة إلى الأرض.
  \item وفي جملة واحدة: أيّ جهاز دفع يعيش على مبدأ الارتداد هذا،
        وما الذي يثبّت \omterm{def:g10:inertia:force}{قوة} دفعه؟
  \item \emph{الاعتماد.} ينقطع كابل الجرّ عند \qty{100}{kN}؛ وتحدّ الأنظمة
        \omterm{def:g10:inertia:inventory}{شدّ} التشغيل بخُمس ذلك. بيّن أن أسوأ حالة لقيتها أعلاه
        هي الانطلاق، واستنتج الكتلة الكلية القصوى، واطبع اللافتة: كم
        مسافرًا كتلة كلٍّ \qty{80}{kg} يجوز أن يركبوا؟ وتحقّق من عامل الأمان.
\end{enumerate}
\end{problem}
