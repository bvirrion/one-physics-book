\chapter{السقوط الحر وحركة المقذوفات}\label{ch:g12:projectile-motion}

كل نفّاثة من خرطوم حديقة ترسم المنحني نفسه في الهواء. وكذلك
كرة سلة تقوّس نحو السلة، ولاعب وثب طويل، وكرة مدفع: أي جسم
مُسلَّم إلى وزنه. ويكسب هذا الفصل ذلك المنحني --- وهو قطع مكافئ ---
من \omterm{thm:g12:newtons-laws:second}{قانون نيوتن الثاني} في سطرين من التحليل، ويقرأ الذروة \omterm{def:g12:projectile-motion:range}{وزمن
الطيران} \omterm{def:g12:projectile-motion:range}{والمدى} من معادلته، ثم يدع الهواء يعود ليرى ما
يفسده.

\section{السقوط الحر}

\begin{definition}[السقوط الحر]\label{def:g12:projectile-motion:freefall}
يكون الجسم في \emph{سقوط حرّ}\index{سقوط حرّ} إذا كانت \omterm{def:g10:inertia:force}{القوة} الوحيدة المؤثرة
فيه وزنَه $\vect P = m\vect g$: فلا سند ولا خيط، \omterm{def:g11:circuits-and-power:resistance}{ومقاومة} الهواء
مهمَلة. والحجر المسقَط أو الكرة التي غادرت اليد في سقوط حرّ بدقة
جيدة --- و``السقوط'' يشمل الصعود.
\end{definition}

\begin{theorem}[قانون غاليليو في السقوط الحر]\label{thm:g12:projectile-motion:galileo}
\index{قانون غاليليو في السقوط الحر}
في السقوط الحر تشترك كل الأجسام في التسارع نفسه، مهما تكن كتلتها:
\[
\vect a = \vect g, \qquad g = \qty{9.81}{m/s^2},
\]
موجَّهًا شاقوليًّا إلى أسفل: أي شدة \omterm{def:g11:electric-gravitational-fields:field}{المجال} \qty{9.81}{N/kg} في
\cref{ch:g10:universal-gravitation} بثوبها الآخر، $\unit{N/kg} = \unit{m/s^2}$.
\end{theorem}

\begin{proof}
\omterm{thm:g12:newtons-laws:second}{قانون نيوتن الثاني} (\cref{ch:g12:newtons-laws}) \omterm{def:g10:universal-gravitation:weight}{والوزن} هو \omterm{def:g10:inertia:force}{القوة}
الوحيدة: $m\vect a = m\vect g$، وتتلاشى الكتلة. فلا تختلف المطرقة
والريشة إلا عبر الهواء.
\end{proof}

\begin{proposition}[السقوط الشاقولي من السكون]\label{prop:g12:projectile-motion:vertical}
إذا أُسقط جسم من السكون، وعُدّ العمق المقطوع $h$ إلى أسفل من
نقطة الإطلاق، خضع في السقوط الحر للعلاقتين
\[
a = g, \qquad v = g t, \qquad h = \tfrac12\, g t^2 .
\]
\end{proposition}

\begin{proof}
$v(t)$ دالة أصلية للثابت $a = g$ تنعدم عند $t = 0$
(\cref{ch:g12:kinematics-2d})؛ و$h(t)$ دالة أصلية \omterm{def:g12:kinematics-2d:velocity}{للسرعة} $v = gt$ تنعدم عند $t = 0$.
\end{proof}

\begin{example}[رتب القدر]\label{ex:g12:projectile-motion:orders}
بعد $1$ و$2$ و$3$ ثوانٍ من السقوط الحر: $v = 9.8$، $19.6$، \qty{29.4}{m/s}
($35$، $71$، \qty{106}{km/h}) و$h = 4.9$، $19.6$، \qty{44.1}{m}. ومن سطح
ارتفاعه \qty{30}{m}: $t = \sqrt{2h/g} \approx \qty{2.5}{s}$، فيصل \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
\qty{24}{m/s} $\approx \qty{87}{km/h}$. وتنمو الأزمنة مثل $\sqrt h$ --- و
بعد بضع ثوانٍ يرفض الهواء أن يبقى مهمَلًا (القسم الأخير).
\end{example}

\begin{example}[القذف الشاقولي]\label{ex:g12:projectile-motion:throw}
إذا قُذف شاقوليًّا إلى أعلى \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v_0$، والمحور $Oy$ إلى أعلى من اليد: أعطت الدالّتان
الأصليتان نفساهما $v = v_0 - gt$ و$y = v_0 t - \tfrac12 gt^2$. و
الذروة، حيث $v = 0$، تأتي عند $t = v_0/g$، وارتفاعها $h = v_0^2/2g$ --- أي جواب
\omterm{def:g10:energy-conservation:energy}{الطاقة} $v_0 = \sqrt{2gh}$ في \cref{ch:g11:mechanical-energy} مُجرًى
عكسًا؛ وإذا قُذف \emph{إلى أسفل}، انقلبت \omterm{def:g10:signals-and-waves:signal}{إشارة} $v_0$. ومن أجل \qty{15}{m/s}: الذروة
\qty{11.5}{m} عند \qty{1.53}{s}، والعودة عند \qty{3.06}{s}، \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{15}{m/s}.
\end{example}

\section{المقذوف: قانون واحد ودالّتان أصليتان}

\begin{definition}[المقذوف]\label{def:g12:projectile-motion:projectile}
\emph{المقذوف}\index{مقذوف} جسم يُطلق \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
$\vect v_0$ ثم يُترك في \omterm{def:g12:projectile-motion:freefall}{سقوط حرّ}. والمحاور عند نقطة الإطلاق: $Ox$
أفقيًّا في المستوي العمودي الحاوي $\vect v_0$، و$Oy$ شاقوليًّا إلى أعلى؛ و
\emph{زاوية القذف}\index{زاوية القذف} $\alpha$ للمتجهة $\vect v_0$ فوق
الأفقي تثبّت إحداثييها $(v_0\cos\alpha,\; v_0\sin\alpha)$.
\end{definition}

\begin{theorem}[معادلات الحركة]\label{thm:g12:projectile-motion:equations}
\omterm{def:g12:projectile-motion:projectile}{المقذوف} المطلَق من المبدأ \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v_0$ وزاوية $\alpha$
له $\vect a = (0,\,-g)$ في كل لحظة، ومنه
\[
\vect v(t) = \bigl(v_0\cos\alpha,\; v_0\sin\alpha - g t\bigr),
\qquad
\begin{cases}
x(t) = v_0\cos\alpha\; t,\\[2pt]
y(t) = v_0\sin\alpha\; t - \tfrac12\, g t^2 .
\end{cases}
\]
\end{theorem}

\begin{proof}
يعطي \omterm{thm:g12:newtons-laws:second}{قانون نيوتن الثاني} $\vect a = \vect g = (0,-g)$. وبالدوال الأصلية
محورًا محورًا: $\vect v(t)$ من $\vect a$ بقيمة ابتدائية
$(v_0\cos\alpha,\, v_0\sin\alpha)$، ثم الموضع من $\vect v$، ومن $(0,0)$.
\end{proof}

\begin{proposition}[استقلال الحركتين]\label{prop:g12:projectile-motion:independence}
الحركة الأفقية منتظمة --- إذ لا يتغير $v_x$ قط --- والحركة
الشاقولية هي بالضبط القذف الشاقولي في القسم السابق:
فلا تذكر أيّ من المعادلتين إحداثيّ الأخرى. وخاصةً، فإن كرة
تُطلق أفقيًّا وكرة تُسقط في اللحظة نفسها من الارتفاع
نفسه تبقيان في مستوى واحد --- وتهبطان معًا.
\end{proposition}

\begin{proof}
في \cref{thm:g12:projectile-motion:equations}، لا يحوي $x(t)$ إلا
$v_0\cos\alpha$، ولا يحوي $y(t)$ إلا $v_0\sin\alpha$ و$g$؛ وللكرتين
$v_y(0) = 0$، ومن ثم تخضع كلتاهما للعلاقة $h = \tfrac12 gt^2$.
\end{proof}

\begin{remark}[القرد الساقط]\label{rem:g12:projectile-motion:monkey}
يصوّب حارس حديقة حيوان سهمًا مخدّرًا \emph{مباشرةً نحو} قرد، فيترك
غصنه لحظة الرمية. وهي خطوة سيئة: فيهبط السهم والقرد كلٌّ $\tfrac12 gt^2$
تحت موضعه بلا جاذبية --- أي خط التصويب والغصن --- فيلتقيان
بالضبط، مهما تكن \omterm{def:g12:kinematics-2d:velocity}{سرعة} السهم
(\cref{exo:g12:projectile-motion:15}).
\end{remark}

\section{القطع المكافئ، مقروءًا كاملًا}

\begin{theorem}[المسار]\label{thm:g12:projectile-motion:trajectory}
بحذف $t$ بين $x(t)$ و$y(t)$: يكون \omterm{def:g10:relative-motion:trajectory}{مسار} \omterm{def:g12:projectile-motion:projectile}{مقذوف}
($0 \leq \alpha < \ang{90}$) قطعًا مكافئًا مفتوحًا إلى أسفل
\[
y = x\tan\alpha \;-\; \frac{g}{2 v_0^2 \cos^2\alpha}\, x^2 .
\]
\end{theorem}

\begin{proof}
$t = x/(v_0\cos\alpha)$، مُعوَّضًا في $y(t)$ من
\cref{thm:g12:projectile-motion:equations}.
\end{proof}

\begin{definition}[المدى]\label{def:g12:projectile-motion:range}
في \omterm{def:g12:projectile-motion:projectile}{مقذوف} فوق أرض مستوية، يكون \emph{المدى}\index{مدى} $R$ هو
المسافة الأفقية المقطوعة حين يعود إلى ارتفاع إطلاقه؛ و
\emph{زمن الطيران}\index{زمن الطيران} $t_f$ هو مدة تلك الرحلة.
\end{definition}

\begin{proposition}[الذروة وزمن الطيران والمدى]\label{prop:g12:projectile-motion:range}
فوق أرض مستوية، مع $0 < \alpha < \ang{90}$:
\[
t_f = \frac{2 v_0 \sin\alpha}{g}, \qquad
h_{\text{الذروة}} = \frac{(v_0\sin\alpha)^2}{2g}
\ \ \text{تُبلَغ عند } t_f/2, \qquad
R = \frac{v_0^2 \sin(2\alpha)}{g}.
\]
ويكون \omterm{def:g12:projectile-motion:range}{المدى} أعظميًّا عند $\alpha = \ang{45}$، حيث $R = v_0^2/g$؛
والزاويتان المتتامّتان $\alpha$ و$\ang{90} - \alpha$ تعطيان \omterm{def:g12:projectile-motion:range}{المدى} نفسه.
\end{proposition}

\begin{proof}
يعطي $y = 0$ النتيجة $t\,(v_0\sin\alpha - \tfrac12 gt) = 0$: أي الإطلاق، و$t_f$.
والذروة، $v_y = 0$، تقع عند $t = v_0\sin\alpha/g = t_f/2$؛ عوّض في
$y(t)$. ثم $R = x(t_f) = 2v_0^2\sin\alpha\cos\alpha/g =
v_0^2\sin(2\alpha)/g$. ويبلغ الجيب ذروته عند $2\alpha = \ang{90}$، و
$\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180} - 2\alpha) =
\sin(2\alpha)$.
\end{proof}

\begin{example}[ركلة مرمى]\label{ex:g12:projectile-motion:kick}
يضرب حارس مرمى \omterm{def:g12:kinematics-2d:velocity}{بسرعة} $v_0 = \qty{25}{m/s}$ وزاوية $\alpha = \ang{45}$:
$R = 25^2/9.81 \approx \qty{64}{m}$؛ أما الركلات \omterm{def:g11:lenses-and-eye:images}{الحقيقية} فتهبط أقرب إلى
\qty{55}{m} --- وتلك حصة الهواء.
\end{example}

\begin{method}[محاسبة المقذوفات]\label{met:g12:projectile-motion:method}
\begin{enumerate}
  \item المحاور عند نقطة الإطلاق، $Ox$ أفقيًّا و$Oy$ إلى أعلى؛ وإحداثيا
        $\vect v_0$.
  \item تحقّق من السقوط الحر (\omterm{def:g10:universal-gravitation:weight}{الوزن} وحده): فيكون $\vect a = (0,-g)$؛
        ثم دالّتان أصليتان، وشرطان ابتدائيان في كل مرة.
  \item والسؤال عن \emph{موضع}: احذف $t$ (أي \omterm{def:g10:relative-motion:trajectory}{المسار}). والسؤال عن
        \emph{لحظة}: احتفظ بالمتغير $t$.
  \item والقاموس: الذروة $\Leftrightarrow v_y = 0$؛ وارتفاع الإطلاق
        $\Leftrightarrow y = 0$؛ والجدار عند $d \Leftrightarrow x = d$؛ والأرضية
        $\ell$ تحت $\Leftrightarrow y = -\ell$.
  \item والسلامة: الطيران متناظر حول الذروة؛ و$R \leq v_0^2/g$؛ والهواء
        لا يفعل إلا التقصير.
\end{enumerate}
\end{method}

\section{ما الذي يغيّره الهواء}

\begin{definition}[السرعة الحدية]\label{def:g12:projectile-motion:terminal}
\omterm{def:g11:circuits-and-power:resistance}{مقاومة} الهواء تعاكس \omterm{def:g12:kinematics-2d:velocity}{السرعة} وتنمو معها. ولذلك لا يتسارع جسم
ساقط إلا حتى توازن \omterm{def:g11:circuits-and-power:resistance}{المقاومة} الوزنَ؛ ومن ثم يكون
$\vect a = \vect 0$: فيسقط \omterm{def:g12:kinematics-2d:velocity}{بسرعة} ثابتة هي \emph{السرعة
الحدية}\index{سرعة حدية}.
\end{definition}

\begin{example}[رتب قدر السرعات الحدية]\label{ex:g12:projectile-motion:airorders}
قافز بمظلة منبسط تمامًا: نحو \qty{55}{m/s} (\qty{200}{km/h}). وقطرة مطر:
نحو \qty{9}{m/s} --- ولو في الخلاء لسلّم سقوط \qty{2000}{m} إياها \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
$\sqrt{2gh} \approx \qty{2.0e2}{m/s}$. وريشة: نحو \qty{1}{m/s}،
تُبلغ في حدود سنتيمترات --- وذلك سرّ بطئها: فهي تقضي
سقوطها كله \emph{عند} سرعتها الحدية؛ أما المطرقة فلا تقترب من سرعتها قط.
\end{example}

\begin{remark}[البالستيك الحقيقي]\label{rem:g12:projectile-motion:drag}
تدفع \omterm{def:g11:circuits-and-power:resistance}{المقاومة} عكس \omterm{def:g12:kinematics-2d:velocity}{السرعة} على امتداد الطيران كله: فيقصر \omterm{def:g12:projectile-motion:range}{المدى}،
وتنخفض الذروة، وينكسر التناظر --- فيكون النزول أشدّ انحدارًا من
الصعود، وتقصر الرميات الطويلة عن القطع المكافئ قصورًا مرئيًّا. أما المعالجة الأمينة
الكمّية فمعادلة تفاضلية تُحلّ في مجلد السنة الأولى؛ وهذه
السنة يكون القطع المكافئ النموذجَ الصحيح للمقذوفات الكثيفة المعتدلة
\omterm{def:g12:kinematics-2d:velocity}{السرعة} --- أي كرة سلة عند \qty{10}{m/s}، لا كرة غولف عند
\qty{70}{m/s}.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:1}
حجر يُسقط من جسر يصطدم بالماء بعد \qty{2.0}{s}: فما ارتفاع
الجسر، وما \omterm{def:g12:kinematics-2d:velocity}{سرعة} الارتطام \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{m/s} وبوحدة \unit{km/h}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:2}
على القمر، أسقط رائد فضاء مطرقة وريشة معًا: فهبطتا
معًا. لماذا --- ولماذا تخسر الريشة خسارة فادحة على الأرض؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:3}
بإهمال الهواء، احسب زمن السقوط \omterm{def:g12:kinematics-2d:velocity}{وسرعة} الوصول من \qty{1.0}{m}
و\qty{10}{m} و\qty{100}{m}. وكيف تنمو الأزمنة مع الارتفاع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:4}
تُقذف كرة شاقوليًّا إلى أعلى \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{15}{m/s}: فما ارتفاع الذروة وزمنها،
والزمن الكلي للعودة إلى اليد، \omterm{def:g12:kinematics-2d:velocity}{وسرعة} العودة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:5}
تتدحرج كرة زجاجية عن طاولة ارتفاعها \qty{0.80}{m} \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{2.5}{m/s}. جِد
زمن السقوط ومسافة الهبوط عن الحافة. وهل يتوقف زمن السقوط
على \omterm{def:g12:kinematics-2d:velocity}{سرعة} الكرة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:6}
من شكل الزوايا الخمس في الدرس: أيّ زوجَي زاويتي قذف
يشتركان في \omterm{def:g12:projectile-motion:range}{المدى}، وأيّ زاوية تفوز؟ وفي \omterm{def:g12:projectile-motion:range}{مدى} دون الأعظمي،
أيّ الزاويتين الممكنتين يطير أطول؟ برّر بالعلاقة $t_f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:7}
من شرفة ارتفاعها \qty{20}{m} تُقذف كرة شاقوليًّا \emph{إلى أسفل} \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
\qty{5.0}{m/s}. جِد الزمن إلى الأرض \omterm{def:g12:kinematics-2d:velocity}{وسرعة} الارتطام؛ وتحقّق
من الأخيرة بميزانية \omterm{def:g10:energy-conservation:energy}{طاقة} (\cref{ch:g11:mechanical-energy}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:8}
يغادر \omterm{def:g12:projectile-motion:projectile}{مقذوف} أرضًا مستوية \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{40}{m/s} وزاوية $\alpha = \ang{30}$:
فما \omterm{def:g12:projectile-motion:range}{زمن الطيران} وارتفاع الذروة \omterm{def:g12:projectile-motion:range}{والمدى}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:9}
نمذِج لاعب وثب طويل مقذوفًا: بإقلاع عند \qty{10.5}{m/s}،
$\alpha = \ang{20}$. احسب \omterm{def:g12:projectile-motion:range}{المدى}؛ ويثب الأبطال نحو \qty{9}{m}
--- سمِّ مكوّنين يهملهما النموذج النقطي.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:10}
على ارتفاع \qty{1.2}{m} فوق أرض مستوية، تُسقط رصاصة وتُطلق أخرى
مطابقة أفقيًّا \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{300}{m/s}. فأيّهما تهبط أولًا؟ احسب
زمن السقوط ومسافة هبوط الرصاصة المطلَقة. وأيّ مبدأ
يُختبر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:11}
بيّن جبريًّا أن $\alpha$ و$\ang{90} - \alpha$ تشتركان في
\omterm{def:g12:projectile-motion:range}{المدى}، ثم جِد الزاويتين اللتين تُنزلان مقذوفًا سرعته \qty{25}{m/s}
على بعد \qty{50}{m}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:12}
يغادر الماء خرطوم حديقة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{8.0}{m/s}: فما أقصى \omterm{def:g12:projectile-motion:range}{مدى} أفقي؟
وما أقصى ارتفاع شاقولي؟ وما زاويتا الفوهة اللتان تسقيان حوض زهور
يبعد \qty{5.0}{m}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:13}
تُركل كرة من الأرض \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{18}{m/s} وزاوية $\alpha = \ang{35}$،
نحو سياج نباتي ارتفاعه \qty{4.0}{m} يبعد \qty{15}{m}. باستعمال معادلة
\omterm{def:g10:relative-motion:trajectory}{المسار}، قرّر أتجتازه أم لا، وبكم.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:14}
قافز بمظلة كتلته \qty{80}{kg} يسقط منبسطًا \omterm{def:g12:projectile-motion:terminal}{بسرعة حدية} \qty{55}{m/s}.
(أ) فما \omterm{def:g10:inertia:force}{قوة} \omterm{def:g11:circuits-and-power:resistance}{المقاومة}؟ (ب) ارسم تخطيطيًّا شكل $v(t)$ من لحظة القفز. (ج) وقطرة مطر
تسقط من \qty{2.0}{km}: قارن \omterm{def:g12:kinematics-2d:velocity}{سرعة} الوصول في الخلاء بسرعتها \omterm{def:g11:lenses-and-eye:images}{الحقيقية}\ 
\qty{9}{m/s}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:15}
القرد الساقط، مبرهنًا عليه: يغادر سهم المبدأ مصوَّبًا بالضبط نحو
قرد على المسافة $D$ والارتفاع $H$ ($\tan\alpha = H/D$)، يترك غصنه عند
$t = 0$. بيّن أنه حين يبلغ السهم $x = D$ يساوي ارتفاعه ارتفاع
القرد --- مهما تكن $v_0$.
\end{exercise}

\section{مسألة: تسديدة الصافرة في كرة السلة}

\begin{problem}[{مسألة نهاية الأسبوع --- تسديدة الصافرة في كرة السلة: رمية
حرة مشرَّحة سهمًا سهمًا، وأطراف أصابع تُتجاوَز بسنتيمترات، ورمية
من الملعب كله موزونة في مقابل ما تقدر ذراع على إعطائه، وقذفة عالية فوق اللوح ---
وحكم على أشدّ التسديدات مسرحيةً في الرياضة}]
\label{pb:g12:projectile-motion:1}
طوق سلة كرة السلة حلقة أفقية على ارتفاع \qty{3.05}{m} فوق
الأرضية؛ وتطلق نورة كل تسديدة من \qty{2.00}{m}، ومن ثم يقع مركز الطوق،
بمحاور عند نقطة الإطلاق ($Ox$ أفقيًّا نحو السلة، و$Oy$ إلى أعلى)،
على ارتفاع \qty{1.05}{m} فوق المبدأ. والهواء مهمَل إلا حيث يُذكر.

\medskip\noindent\textbf{الجزء I --- الرمية الحرة.} تسدّد نورة من
$d = \qty{4.20}{m}$ (أفقيًّا) بزاوية قذف $\alpha = \ang{50}$.
\begin{enumerate}
  \item فأيّ قوى تؤثر بعد الإطلاق؟ أعطِ $\vect a$، مستشهدًا بالقانون المستعمل.
  \item اشتقّ إحداثيَي $\vect v(t)$.
  \item اشتقّ $x(t)$ و$y(t)$.
  \item احذف $t$ للحصول على \omterm{def:g10:relative-motion:trajectory}{المسار} $y(x)$.
  \item ويجب أن تمرّ الكرة بمركز الطوق، $y(d) = \qty{1.05}{m}$: حُلّ
        لإيجاد $v_0$، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{m/s} وبوحدة \unit{km/h}.
\end{enumerate}

\medskip\noindent\textbf{الجزء II --- فوق أطراف الأصابع، وداخل الحلقة.}
\begin{enumerate}[resume]
  \item كم تستغرق الكرة لتبلغ الطوق؟
  \item جِد زمن الذروة وارتفاعها؛ وتحقّق من أن الكرة نازلة أصلًا
        عند الطوق.
  \item يمدّ مدافع على بعد \qty{0.90}{m} أمام نورة أطراف أصابعه إلى
        \qty{2.60}{m}. فبكم تتجاوزها الكرة؟
  \item احسب $v_x$ و$v_y$ عند الطوق، \omterm{def:g12:kinematics-2d:velocity}{والسرعة} هناك، وزاوية
        \omterm{def:g12:kinematics-2d:velocity}{السرعة} تحت الأفقي.
  \item ولماذا يجب أن تصل الكرة \emph{نازلة} --- ولماذا شديدة الانحدار؟
\end{enumerate}

\medskip\noindent\textbf{الجزء III --- الرمية من الملعب كله.} والصافرة على وشك
أن تنطلق، ونورة على بعد \qty{26.0}{m} من الطوق البعيد.
\begin{enumerate}[resume]
  \item الصعود \qty{1.05}{m} صغير أمام \qty{26.0}{m}: برّر نموذج
        الأرض المستوية، واشتقّ \omterm{def:g12:projectile-motion:range}{زمن الطيران} من $y(t_f) = 0$.
  \item استنتج $R = v_0^2\sin(2\alpha)/g$ وزاوية \omterm{def:g12:projectile-motion:range}{المدى} الأعظم.
  \item احسب $v_0$ اللازمة لقطع \qty{26.0}{m} عند $\alpha = \ang{45}$.
  \item ويقدر لاعب قوي على قذف كرة سلة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} نحو \qty{17.5}{m/s}.
        فما أقصى \omterm{def:g12:projectile-motion:range}{مدى} له --- وهل التسديدة ممكنة بشريًّا؟
  \item وعند \qty{17.5}{m/s}، جِد \emph{الزاويتين} اللتين تُنزلان الكرة
        على بعد \qty{26.0}{m}، وزمنَي طيرانهما: فمع \qty{2.0}{s} على
        الساعة، أيّهما يسبق الصافرة؟
\end{enumerate}

\medskip\noindent\textbf{الجزء IV --- القذفة العالية فوق اللوح.} من
خلف خط النهاية تقذف نورة قذفة عالية: بإطلاق على بعد \qty{6.0}{m} من مركز الطوق،
و$\alpha = \ang{75}$؛ ويقطع مستوي اللوح تسديدتها على بعد \qty{0.45}{m}
قبل مركز الطوق، وحافته العليا على ارتفاع \qty{3.95}{m} فوق الأرضية.
\begin{enumerate}[resume]
  \item جِد $v_0$ التي تُسقط الكرة عبر مركز الطوق.
  \item احسب ارتفاع الكرة عند مستوي اللوح
        ($x = \qty{5.55}{m}$): فهل تتجاوز الحافة العليا، وبكم؟
  \item جِد ارتفاع الذروة \omterm{def:g12:projectile-motion:range}{وزمن الطيران} --- ولماذا يحبّ الجمهور هذه القذفة؟
  \item وما الذي تغيّره \omterm{def:g11:circuits-and-power:resistance}{مقاومة} الهواء نوعيًّا في هذه التسديدات الثلاث،
        وأيّها تهدّده أكثر؟
  \item \emph{حكم تسديدة الصافرة}، في جملتين: قارن \omterm{def:g12:kinematics-2d:velocity}{السرعة}
        التي تقتضيها رمية الملعب كله بما توفّره ذراع، وأعطِ
        الهامش، واحكم --- أممكنة هي أم أسطورة؟
\end{enumerate}
\end{problem}
