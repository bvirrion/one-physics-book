\chapter{الطاقة النووية: الانشطار والاندماج و$E=mc^2$}\label{ch:g12:nuclear-energy}

كُريّة وقود بحجم ممحاة قلم تغطي كهرباء أسرة
سنةً كاملة؛ والخدمة نفسها من الفحم تأخذ ثلاثة أطنان. وفي الأعلى، تدفع الشمس
ثمن ضوء النهار باختفائها --- أي أربعة ملايين طن من نفسها في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}.
ويفتح هذا الفصل دفتر الحسابات وراء الصفقتين: فالكتلة نفسها
\omterm{def:g10:energy-conservation:energy}{طاقة}، والأنوية (\cref{ch:g11:nucleus-radioactivity}) هي حيث تُصرف.

\section{الكتلة طاقة}

\begin{theorem}[تكافؤ الكتلة والطاقة]\label{thm:g12:nuclear-energy:emc2}\index{تكافؤ الكتلة والطاقة}
الجسم ذو الكتلة $m$، بمجرد وجوده، يحمل \omterm{def:g10:energy-conservation:energy}{الطاقة}
\[
E = m c^2, \qquad c = \qty{3.00e8}{m/s}.
\]
وكلما خسرت جملة كتلةً $\Delta m$، حرّرت \omterm{def:g10:energy-conservation:energy}{الطاقة}
$\Delta m\, c^2$ --- مهما تكن الآلية.
\end{theorem}
\admitted

\begin{remark}[أينشتاين، 1905]\label{rem:g12:nuclear-energy:einstein}
هذه علاقة أينشتاين، وهي مقبولة هنا؛ أما من أين تأتي فشأن
فصل النسبية الخاصة الذي يختم هذه السنة
(\cref{ch:g12:special-relativity}). ولاحِظ السعر: $c^2 = \qty{9.00e16}{J/kg}$
--- فخطأ تدوير في الكتلة ثروة من \omterm{def:g10:energy-conservation:energy}{الطاقة}.
\end{remark}

\begin{definition}[الإلكترون فولط ووحدة الكتلة الذرية]\label{def:g12:nuclear-energy:units}
وحدتان مفصّلتان على قياس \omterm{def:g11:fundamental-interactions:ladder}{النواة}.
\emph{الإلكترون فولط}\index{إلكترون فولط}، وهو \omterm{def:g10:energy-conservation:energy}{الطاقة} التي يكسبها إلكترون
عابرًا \qty{1}{V} (\cref{ch:g11:circuits-and-power}):
$\qty{1}{eV} = \qty{1.60e-19}{J}$، $\qty{1}{MeV} = \qty{1.60e-13}{J}$ ---
فالروابط الكيميائية تتبادل بضعة \unit{eV}، والتفاعلات النووية ميغا إلكترون \omterm{def:g11:circuits-and-power:voltage}{فولط}.
و\emph{وحدة الكتلة الذرية الموحّدة}\index{وحدة الكتلة الذرية الموحّدة}، وهي
جزء من اثني عشر من كتلة \omterm{def:g11:fundamental-interactions:ladder}{ذرة} كربون-12: $\qty{1}{u} = \qty{1.66054e-27}{kg}$،
ومكافئها الطاقي \qty{931.5}{MeV} --- $\qty{1}{u} = \qty{931.5}{MeV}/c^2$.
\end{definition}

\begin{example}[الثروة النائمة]\label{ex:g12:nuclear-energy:fortune}
\omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} واحد من أي شيء، محوَّلًا تحويلًا كاملًا، هو $\qty{9.00e16}{J}$ --- أي ثلاث
سنوات من إنتاج محطة كهرباء كبيرة (\qty{1}{GW}). ولم ينتبه أحد قبل
1905 لأن الفيزياء العادية لا تكاد تمسّ الكتلة: فحرق \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من الفحم
(\qty{3.0e7}{J}) يخفّفه بثلث ميكروغرام. أما الأنوية، كما سنرى،
فتحرّك جزءًا من ألف: وهو ما يزال صغيرًا، لكنه مليون ضعف الكيمياء.
\end{example}

\section{النقص الكتلي}

\begin{definition}[النقص الكتلي وطاقة الربط]\label{def:g12:nuclear-energy:defect}
زِن \omterm{def:g11:fundamental-interactions:ladder}{نواة} ${}^{A}_{Z}\mathrm{X}$ (كتلتها $m$)، ثم زِن بروتوناتها $Z$
ونيوتروناتها $A - Z$ على حدة: فالأجزاء \emph{أثقل} من الكل.
والفرق
\[
\Delta m = Z\,m_p + (A - Z)\,m_n - m > 0,
\qquad m_p = \qty{1.00728}{u},\; m_n = \qty{1.00866}{u},
\]
هو \emph{النقص الكتلي}\index{نقص كتلي}؛ و\emph{طاقة
الربط}\index{طاقة الربط} $E_b = \Delta m\,c^2$ هي \omterm{def:g10:energy-conservation:energy}{الطاقة} اللازمة
لتفكيك \omterm{def:g11:fundamental-interactions:ladder}{النواة} إلى نوكليونات حرة --- وهي نفسها \omterm{def:g10:energy-conservation:energy}{الطاقة} المحرَّرة
حين تُجمَّع. \omterm{def:g11:fundamental-interactions:ladder}{فالنواة} المرتبطة تجلس \emph{دون} أجزائها.
\end{definition}

\begin{example}[الهيليوم-4]\label{ex:g12:nuclear-energy:helium}
\omterm{def:g11:fundamental-interactions:ladder}{لنواة} ${}^{4}_{2}\mathrm{He}$ المقدار $m = \qty{4.00151}{u}$؛ وأجزاؤها:
$2 \times 1.00728 + 2 \times 1.00866 = \qty{4.03188}{u}$. ومنه
$\Delta m = \qty{0.03037}{u}$ و$E_b = 0.03037 \times 931.5 \approx
\qty{28.3}{MeV}$: \omterm{def:g11:fundamental-interactions:ladder}{فالنواة} أخفّ من أجزائها بمقدار $0.75\%$ --- أي
``الجزء من ألف'' في \cref{ex:g12:nuclear-energy:fortune}.
\end{example}

\begin{method}[الميزانية الطاقية لتفاعل نووي]\label{met:g12:nuclear-energy:balance}
\begin{enumerate}
  \item وازِن المعادلة: فمجموع $A$ و$Z$ محفوظ (\cref{ch:g11:nucleus-radioactivity}).
  \item اجمع الكتل قبل، ثم بعد، \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{u} --- واحتفظ بالأرقام العشرية الخمسة كلها.
  \item $\Delta m = m_{\text{قبل}} - m_{\text{بعد}}$، ثم
        $E = \Delta m\,(\text{بوحدة u}) \times \qty{931.5}{MeV}$، محمولةً
        طاقةً حركية للنواتج؛ والسالب يعني أن التفاعل يجب أن يُدفع ثمنه.
\end{enumerate}
\end{method}

\section{المنحني الذي يدير الكون}

\begin{definition}[طاقة الربط لكل نوكليون]\label{def:g12:nuclear-energy:pernucleon}
يُقارَن التماسك مقارنة عادلة بمقدار \emph{طاقة الربط لكل
نوكليون}\index{طاقة الربط لكل نوكليون}، $E_b/A$، وهي التكلفة المتوسطة
لانتزاع \omterm{def:g11:fundamental-interactions:ladder}{نوكليون} واحد: $28.3/4 \approx \qty{7.1}{MeV}$ في الهيليوم-4.
\end{definition}

\begin{proposition}[ذروة الحديد، وطريقان نازلان]\label{prop:g12:nuclear-energy:ironpeak}\index{ذروة الحديد}
مرسومةً بدلالة $A$، تصعد \omterm{def:g12:nuclear-energy:pernucleon}{طاقة الربط لكل نوكليون} صعودًا حادًّا عبر
الأنوية الخفيفة، وتبلغ ذروتها قرب الحديد، ${}^{56}_{26}\mathrm{Fe}$، عند نحو
\qty{8.8}{MeV} لكل \omterm{def:g11:fundamental-interactions:ladder}{نوكليون}، ثم تنزلق برفق إلى نحو \qty{7.6}{MeV} عند
اليورانيوم: فالحديد أشدّ الأنوية ارتباطًا في الطبيعة. ويحرّر التفاعل
طاقةً بالضبط حين تجلس نواتجه \emph{أعلى} على المنحني؛ ومن ثم تربح
استراتيجيتان متعاكستان كلتاهما، وتمشيان معًا نحو الحديد ---
أي \emph{شطر} أثقل الأنوية، و\emph{دمج} أخفّها.
\end{proposition}
\admitted

\begin{remark}[لماذا ذروة]\label{rem:g12:nuclear-energy:whypeak}
إنه شدّ الحبل في \cref{ch:g11:fundamental-interactions}: فالغراء القوي يربط
الجيران المتلامسين وحدهم، وتنفرد كولوم بالتنافر عبر \omterm{def:g11:fundamental-interactions:ladder}{النواة} كلها --- فالأنوية
الصغيرة سطح في معظمها، والكبيرة تنافر في معظمها، والحديد هو الحلّ الوسط.
أما المحاسبة الأمينة فتعيش في مجلدات الجامعة.
\end{remark}

\section{الانشطار والتفاعل المتسلسل}

\begin{definition}[الانشطار]\label{def:g12:nuclear-energy:fission}
\emph{الانشطار}\index{انشطار} هو شطر \omterm{def:g11:fundamental-interactions:ladder}{نواة} ثقيلة إلى
شظيتين متوسطتَي الحجم. واليورانيوم-235 \emph{انشطاري}\index{انشطاري}: فالنيوترون
البطيء، إذا امتُصّ، ترك \omterm{def:g11:fundamental-interactions:ladder}{النواة} من الاضطراب بحيث تتمزق إلى اثنتين،
باصقةً نيوترونين أو ثلاثة جددًا ---
${}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \to \text{شظيتان} +
2\text{--}3\;{}^{1}_{0}\mathrm{n}$، ويتغير الزوج من انشطار إلى آخر.
\end{definition}

\begin{example}[انشطار واحد، موزونًا]\label{ex:g12:nuclear-energy:fissionbalance}
طريق شائع، حسب \cref{met:g12:nuclear-energy:balance}:
\[
{}^{1}_{0}\mathrm{n} + {}^{235}_{92}\mathrm{U} \longrightarrow
{}^{94}_{38}\mathrm{Sr} + {}^{140}_{54}\mathrm{Xe} + 2\,{}^{1}_{0}\mathrm{n}.
\]
قبل: $1.00866 + 234.99350 = \qty{236.00216}{u}$. وبعد:
$93.89450 + 139.89200 + 2 \times 1.00866 = \qty{235.80382}{u}$. ومنه
$\Delta m = \qty{0.19834}{u}$ و$E \approx \qty{185}{MeV}$؛ وبعدّ
تفككات الشظايا المشعّة اللاحقة (\cref{ch:g12:radioactive-decay})،
يساوي كل \omterm{def:g12:nuclear-energy:fission}{انشطار} نحو \qty{200}{MeV} --- فبضعة \unit{eV} تشتري \omterm{def:g11:fundamental-interactions:ladder}{ذرة}
فحم واحدة.
\end{example}

\begin{definition}[التفاعل المتسلسل والكتلة الحرجة]\label{def:g12:nuclear-energy:chain}
كل \omterm{def:g12:nuclear-energy:fission}{انشطار} يشعله نيوترون واحد ويحرّر اثنين أو ثلاثة: ففي كتلة كبيرة بما يكفي
تصيب أنوية أخرى ويغذّي التفاعل نفسه ---
أي \emph{تفاعل متسلسل}\index{تفاعل متسلسل}. وفي كتلة صغيرة تفلت معظم النيوترونات
عبر السطح أولًا؛ وأدنى كمية تديم السلسلة
هي \emph{الكتلة الحرجة}\index{كتلة حرجة} --- فدونها تخبو السلسلة،
وفوقها يتضاعف كل جيل: وهو مبدأ القنبلة.
\end{definition}

\begin{definition}[المفاعل، مروَّضًا]\label{def:g12:nuclear-energy:reactor}
يمسك مفاعل \omterm{def:g10:energy-conservation:energy}{الطاقة} السلسلة عند نيوترون واحد بالضبط لكل \omterm{def:g12:nuclear-energy:fission}{انشطار}. والوقود:
كُريّات يورانيوم مُخصَّبة تخصيبًا متواضعًا باليورانيوم-235. و
\emph{المهدِّئ}\index{مهدِّئ} (وهو ماء عادي في العادة) يبطّئ نيوترونات
\omterm{def:g12:nuclear-energy:fission}{الانشطار} السريعة بالتصادمات --- فالنيوترونات البطيئة أفضل بكثير في إطلاق
\omterm{def:g12:nuclear-energy:fission}{الانشطار} التالي؛ و\emph{قضبان التحكم}\index{قضيب تحكم} (من البورون أو الكادميوم)
تلتهم النيوترونات --- فإذا أُدخلت ماتت السلسلة، وإذا سُحبت تسارعت. و
اضطراب الشظايا يصير حرارة، ثم بخارًا، ثم كهرباء
(\cref{ch:g10:energy-conservation}).
\end{definition}

\begin{example}[حبة فلفل في مقابل شاحنة]\label{ex:g12:nuclear-energy:gram}
غرام واحد من اليورانيوم-235 يحوي
$N = \num{1.0e-3} / (235 \times \num{1.66054e-27}) \approx \num{2.56e21}$
\omterm{def:g11:fundamental-interactions:ladder}{نواة}؛ وعند $\qty{200}{MeV} = \qty{3.2e-11}{J}$ لكل واحدة، يعطي شطرها جميعًا
نحو \qty{8.2e10}{J} --- أي \omterm{def:g10:energy-conservation:forms}{الطاقة الكيميائية} لكتلة $\num{8.2e10} /
\num{3.0e7} \approx \qty{2.7e3}{kg}$ من الفحم: أي حبة فلفل في مقابل ثلاثة أطنان.
\end{example}

\section{الاندماج: نار الشمس نفسها}

\begin{definition}[الاندماج]\label{def:g12:nuclear-energy:fusion}
\emph{الاندماج}\index{اندماج} هو دمج نواتين خفيفتين في \omterm{def:g11:fundamental-interactions:ladder}{نواة} أثقل
أشدّ ارتباطًا. والتفاعل الأيسر منالًا يزوّج الديوتيريوم و
التريتيوم، وهما نظيرا الهيدروجين الثقيلان (\cref{ch:g11:nucleus-radioactivity}):
\[
{}^{2}_{1}\mathrm{H} + {}^{3}_{1}\mathrm{H} \longrightarrow
{}^{4}_{2}\mathrm{He} + {}^{1}_{0}\mathrm{n}.
\]
\end{definition}

\begin{example}[D--T، موزونًا]\label{ex:g12:nuclear-energy:dt}
قبل: $2.01355 + 3.01550 = \qty{5.02905}{u}$؛ وبعد:
$4.00151 + 1.00866 = \qty{5.01017}{u}$؛ ومنه $\Delta m = \qty{0.01888}{u}$ و
$E \approx \qty{17.6}{MeV}$، أربعة أخماسها على النيوترون --- أي أقل من
\omterm{def:g12:nuclear-energy:fission}{انشطار}، لكن من خمسة نوكليونات بدلًا من $236$: أي نحو \qty{3.4e14}{J} لكل
\omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من الوقود، أي أربعة أضعاف \omterm{def:g12:nuclear-energy:fission}{الانشطار}. والهيليوم-4
(\cref{ex:g12:nuclear-energy:helium}) هو الرماد مرة أخرى.
\end{example}

\begin{definition}[حاجز كولوم]\label{def:g12:nuclear-energy:barrier}
النواتان كلتاهما موجبة: فلتتلامسا، عليهما أولًا أن تتسلقا تلّ
تنافرهما الكهربائي (\cref{ch:g11:electric-gravitational-fields}) --- أي
\emph{حاجز كولوم}\index{حاجز كولوم}. ولا تحمل التصادمات \omterm{def:g10:energy-conservation:energy}{الطاقة} اللازمة
لعبوره إلا قرب \qty{1e8}{K}؛ ويكون الوقود عندئذٍ
\emph{بلازما}\index{بلازما} من أنوية عارية وإلكترونات. أما \omterm{def:g12:nuclear-energy:fission}{الانشطار} فلا يدفع
مثل هذه الرسوم: إذ إن مُطلِقه، وهو النيوترون، متعادل، ويعمل على البارد.
\end{definition}

\begin{example}[الشمس تعمل عليه]\label{ex:g12:nuclear-energy:sun}
قلب الشمس، عند $\qty{1.5e7}{K}$، يدمج الهيدروجين العادي خطوة خطوة ---
أي \emph{سلسلة البروتون--البروتون}، ومفعولها الصافي
$4\,{}^{1}_{1}\mathrm{H} \to {}^{4}_{2}\mathrm{He} + 2\,{}^{0}_{+1}\mathrm{e}
+ \text{إشعاع}$، أي نحو \qty{26}{MeV} لكل هيليوم: فتغادر $0.7\%$ من
كتلة الهيدروجين طاقةً. وضوء الشمس الذي قرأنا طيفه في
\cref{ch:g10:light-spectra} هو هذه الكتلة، واصلةً متأخرة ثماني دقائق؛ وكل
نجم يضيء بمشيه على المنحني نحو الحديد.
\end{example}

\begin{remark}[ITER: تعبئة نجم في قنّينة]\label{rem:g12:nuclear-energy:iter}
لا يقدر أي صلب أن يمسك \omterm{def:g12:nuclear-energy:barrier}{بلازما} عند \qty{1.5e8}{K}؛ و\emph{التوكاماك} يحبسها في
\omterm{def:g11:magnetic-fields:bfield}{مجال مغناطيسي} (\cref{ch:g11:magnetic-fields})، أي كعكة من \omterm{def:g11:electric-gravitational-fields:lines}{خطوط المجال}
تلولب الجسيمات المشحونة على امتدادها دون أن تمسّ الجدار؛ وITER، المبني
ليحرّر عشرة أضعاف \omterm{def:g11:work-of-force:power}{استطاعة} تسخينه، هو الخطوة الراهنة. واكسر
الحبس، فتبرد \omterm{def:g12:nuclear-energy:barrier}{البلازما} وتتوقف في ثوانٍ: فلا \omterm{def:g12:nuclear-energy:chain}{تفاعل متسلسل}، ولا
\omterm{def:g12:nuclear-energy:chain}{كتلة حرجة} --- فالجزء الصعب ليس إيقاف النار بل إبقاءها مشتعلة.
\end{remark}

\begin{example}[سُلّم الطاقة، لكل كيلوغرام]\label{ex:g12:nuclear-energy:ladder}
\omterm{def:g10:energy-conservation:energy}{الطاقة} لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من الوقود: الفحم، \qty{3.0e7}{J}؛ واليورانيوم-235 \omterm{def:g12:nuclear-energy:fission}{بالانشطار}،
\qty{8.2e13}{J}؛ والديوتيريوم--التريتيوم \omterm{def:g12:nuclear-energy:fusion}{بالاندماج}، \qty{3.4e14}{J}؛ والتحويل
الكامل ($E = mc^2$)، \qty{9.0e16}{J}. فستّ رتب قدر تفصل
الكيمياء عن \omterm{def:g11:fundamental-interactions:ladder}{النواة} --- وهي سبب إعادة تموين المفاعل بالشاحنة
ومحطة الفحم بحمولات القطارات.
\end{example}

\section{مسألة: تغذية مدينة}

\begin{problem}[{مسألة نهاية الأسبوع --- تغذية مدينة: غيغاواط واحد مستمر من أجل
نصف مليون نسمة، مشترًى بأربع طرائق --- كُريّات يورانيوم، وحمولات قطارات من
الفحم، وبِركة من ماء البحر، ومادة الشمس نفسها --- والكيلوغرام نفسه
من الكتلة المتلاشية مختبئ تحت الأربع جميعًا}]
\label{pb:g12:nuclear-energy:1}
مدينة عدد سكانها $500\,000$ نسمة تسحب \omterm{prop:g11:circuits-and-power:power}{استطاعة كهربائية} مستمرة
$P = \qty{1.0}{GW}$؛ ومحطاتها، مهما يكن الوقود، تحوّل الحرارة إلى
كهرباء \omterm{def:g10:energy-conservation:efficiency}{بمردود} $33\%$ (\cref{ch:g10:energy-conservation}).
والمعطيات: \omterm{def:g12:nuclear-energy:fission}{انشطار} يورانيوم-235 واحد، \qty{200}{MeV} إجمالًا؛ والفحم، \qty{3.0e7}{J/kg}؛
\omterm{def:g12:nuclear-energy:fusion}{واندماج} D--T واحد، \qty{17.6}{MeV} من \qty{5.029}{u} من الوقود؛
$\qty{1}{u} = \qty{1.66054e-27}{kg}$؛ $\qty{1}{eV} = \qty{1.60e-19}{J}$؛ و
السنة $= \qty{3.16e7}{s}$؛ والشمس: \qty{3.9e26}{W}، \qty{2.0e30}{kg}؛ \omterm{def:g10:orders-of-magnitude:unit}{والمتر}
المكعب من ماء البحر يحوي نحو \qty{33}{g} من الديوتيريوم.

\medskip
\noindent\textbf{الجزء I --- محطة \omterm{def:g12:nuclear-energy:fission}{الانشطار}.}
\begin{enumerate}
  \item أيّ \omterm{def:g11:work-of-force:power}{استطاعة} حرارية يجب أن يوصّلها المفاعل؟
  \item عبّر عن \omterm{def:g10:energy-conservation:energy}{طاقة} \omterm{def:g12:nuclear-energy:fission}{انشطار} واحد \omterm{def:g11:work-of-force:work}{بالجول}.
  \item وكم انشطارًا في \omterm{def:g10:orders-of-magnitude:unit}{الثانية} يدير المدينة؟
  \item وما كتلة \omterm{def:g11:fundamental-interactions:ladder}{نواة} يورانيوم-235 واحدة ($A = 235$)، ثم اليورانيوم-235
        المستهلك في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}؟
  \item استنتج اليورانيوم-235 المستهلك في سنة واحدة، بالكيلوغرامات.
  \item والوقود مُخصَّب إلى $4.0\%$ يورانيوم-235: فأيّ كتلة تأخذها سنة،
        وهل تسع شاحنة واحدة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- دفتر الفحم.}
\begin{enumerate}[resume]
  \item احسب \omterm{def:g10:energy-conservation:forms}{الطاقة الحرارية} التي تستهلكها المدينة في سنة واحدة.
  \item وأيّ كتلة من الفحم توصّلها؟
  \item احسب النسبة (كتلة الفحم)/(كتلة اليورانيوم-235) لسنة واحدة، و
        تحقق منها في مقابل سُلّم \cref{ex:g12:nuclear-energy:ladder}.
  \item وقطار شحن يجرّ \qty{3.0e6}{kg}: فكم قطارًا في السنة، وكم
        في اليوم؟ وفي جملة واحدة: تخيّل خطَّي التموين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- حلم \omterm{def:g12:nuclear-energy:fusion}{الاندماج}.}
\begin{enumerate}[resume]
  \item عبّر عن \omterm{def:g10:energy-conservation:energy}{طاقة} \omterm{def:g12:nuclear-energy:fusion}{اندماج} D--T واحد \omterm{def:g11:work-of-force:work}{بالجول}.
  \item احسب \omterm{def:g10:energy-conservation:energy}{الطاقة} المحرَّرة لكل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من وقود D--T؛ وقارنها
        بالسُّلّم.
  \item وأيّ كتلة من وقود D--T تغطي سنة المدينة؟ وأيّ كتلة من ذلك
        ديوتيريوم ($2.014$ من \qty{5.029}{u})؟
  \item وأيّ حجم من ماء البحر يحوي ذلك الديوتيريوم؟ قارنه
        بمسبح أولمبي، أي نحو \qty{2.5e3}{m^3}. (والتريتيوم يولَّد من الليثيوم.)
  \item ولماذا لا تحمل هذه المحطة \omterm{def:g12:nuclear-energy:chain}{كتلة حرجة} ولا تفاعلًا متسلسلًا
        يُخشى؟ فما الجزء الصعب إذًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- النجم الذي يدفع كتلةً.}
\begin{enumerate}[resume]
  \item من $E = mc^2$، أيّ كتلة تحوّل الشمس في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}؟
  \item وأيّ كتلة في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}، وفي السنة، يجب أن يحوّلها \emph{أيّ}
        منبع حراري قدره \qty{3.0}{GW}؟ وتحقق من أن الأنواع الثلاثة من الوقود
        تسلّم هذه الكتلة نفسها.
  \item وأيّ كتلة كلية أشعّتها الشمس في $\num{4.5e9}$
        سنة، وأيّ جزء من الشمس ذلك؟
  \item ويقدر \omterm{def:g12:nuclear-energy:fusion}{الاندماج} أن ينهل نحو $0.07\%$ من كتلة الشمس (أي الحصة القابلة للاحتراق من
        قلبها): قدّر عمرها الكلي نجمًا، وما تبقّى منه.
  \item أغلِق التدقيق في ثلاث جمل: سنة واحدة من المدينة بأطنان
        الفحم، وكيلوغرامات اليورانيوم-235، وكيلوغرامات وقود D--T --- و
        \omterm{def:g10:orders-of-magnitude:unit}{الكيلوغرام} من الكتلة الذي صار، في كل سيناريو، ضوء المدينة.
\end{enumerate}
\end{problem}
