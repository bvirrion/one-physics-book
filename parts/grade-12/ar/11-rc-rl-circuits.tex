\chapter{دارتا RC وRL}\label{ch:g12:rc-rl-circuits}

اضغط مصراع آلة تصوير إلى منتصفه: فيرتفع أزيز رفيع ثانيتين، ويضيء
ضوء أخضر --- وعندئذٍ فقط ينطلق الوماض، فيفرغ في ميلي \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة
ما أمضى ثانيتين في جمعه. ويلاقي هذا الفصل المركّبتين
اللتين تمنحان الدارات ذاكرة وساعة --- أي \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} \omterm{def:g12:rc-rl-circuits:inductor}{والوشيعة} --- و
أول المعادلات التفاضلية في الفيزياء، وهي قابلة للتحقق بمشتقات هذه السنة.

\section{المكثّفة}

\begin{definition}[المكثّفة والسعة الكهربائية]\label{def:g12:rc-rl-circuits:capacitor}
\emph{المكثّفة}\index{مكثّفة} صفيحتان معدنيتان متقابلتان
يفصل بينهما عازل رقيق. وإذا وُصلت بمنبع، حملت الصفيحتان
شحنتين متعاكستين $+q$ و$-q$ --- فيملأ \omterm{def:g11:electric-gravitational-fields:efield}{مجال كهربائي} الفجوة
(\cref{ch:g11:electric-gravitational-fields}) --- ويتناسب \omterm{def:g11:circuits-and-power:voltage}{التوتر} $u$
بين الصفيحتين مع الشحنة المخزونة:
\[ q = C\,u . \]
والثابت $C$ هو \emph{السعة الكهربائية}\index{سعة كهربائية}، وتقاس
\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \emph{الفاراد}\index{فاراد} (\unit{F}): $\qty{1}{F} = \qty{1}{C/V}$.
\end{definition}

\begin{example}[رتب القدر]\label{ex:g12:rc-rl-circuits:orders}
\omterm{def:g12:rc-rl-circuits:capacitor}{الفاراد} الواحد هائل --- أي \omterm{def:g11:fundamental-interactions:charge}{كولوم} واحد مركون لكل \omterm{def:g11:circuits-and-power:voltage}{فولط}. فالمكثّفات الخزفية في دارات
الراديو: من \unit{pF} إلى \unit{nF}؛ والإلكتروليتية في مغذّيات التيار
وفي الأجهزة الوامضة: من \unit{\micro F} إلى \unit{mF}؛ والمكثّفات الفائقة: آلاف الفارادات.
\end{example}

\begin{proposition}[التيار الداخل إلى مكثّفة]\label{prop:g12:rc-rl-circuits:cdudt}
التيار الواصل إلى صفيحة \omterm{def:g12:rc-rl-circuits:capacitor}{مكثّفة} هو الوتيرة التي تنمو بها
شحنتها:
\[
i = \frac{dq}{dt} = C\,\frac{du}{dt} ,
\]
وتشير الحروف الصغيرة إلى المقادير المتغيرة مع الزمن ($U$، $I$ يبقيان ثابتين،
\cref{ch:g11:circuits-and-power}).
\end{proposition}

\begin{proof}
التيار شحنة موصَّلة في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}
(\cref{def:g11:circuits-and-power:current}): وعلى فترة متقلصة،
يكون $i$ مشتقةَ $q$؛ ويعطي $q = Cu$ مع ثبات $C$ العلاقةَ
$C\,du/dt$.
\end{proof}

\begin{remark}[نتيجتان]\label{rem:g12:rc-rl-circuits:continuity}
إذا كان $u$ ثابتًا، فإن $i = 0$: أي إن \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} المشحونة لا تمرّر تيارًا مستمرًّا.
ولا يمكن أبدًا أن يقفز $u$ --- إذ ستقتضي القفزة تيارًا لا نهائيًّا: \omterm{def:g11:circuits-and-power:voltage}{فتوتر}
\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} هو ذاكرة الدارة.
\end{remark}

\section{شحن مكثّفة: الدارة RC}

حلقة واحدة --- منبع $\mathcal{E}$ (\cref{def:g11:circuits-and-power:emf})،
وقاطعة، \omterm{def:g11:circuits-and-power:resistance}{ومقاومة}، \omterm{def:g12:rc-rl-circuits:capacitor}{ومكثّفة} فارغة: أغلِق $K$ عند $t = 0$.

{\small دارة الشحن RC: إغلاق $K$ يتيح للمنبع أن يدفع الشحنة

\begin{theorem}[قانون الشحن في الدارة RC]\label{thm:g12:rc-rl-circuits:charging}
\index{دارة RC}
بعد إغلاق القاطعة، يخضع \omterm{def:g11:circuits-and-power:voltage}{توتر} \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} للمعادلة الآتية، ثم يتبع من
$u(0) = 0$:
\[
RC\,\frac{du}{dt} + u = \mathcal{E},
\qquad
u(t) = \mathcal{E}\left(1 - e^{-t/\tau}\right),
\qquad \tau = RC .
\]
\end{theorem}

\begin{proof}
تتجمع التوترات على امتداد الحلقة (\cref{prop:g11:circuits-and-power:laws}):
$\mathcal{E} = u_R + u = Ri + u$ في كل لحظة؛ فعوّض
$i = C\,du/dt$ (\cref{prop:g12:rc-rl-circuits:cdudt}). والآن \emph{تحقّق} من
\omterm{def:g11:color-light-sources:subtractive}{المرشّح}: $du/dt = (\mathcal{E}/\tau)e^{-t/\tau}$، ومنه
$RC\,(\mathcal{E}/RC)\,e^{-t/\tau} + \mathcal{E}(1 - e^{-t/\tau}) =
\mathcal{E}$، وهي صحيحة من أجل كل $t$؛ و$u(0) = \mathcal{E}(1-1) = 0$. أما أنه
لا يوجد منحنٍ \emph{آخر} يوافق المعادلة والانطلاقة فمبرهَن عليه في درس
الرياضيات هذه السنة في المعادلات التفاضلية.
\end{proof}

\begin{definition}[ثابت الزمن]\label{def:g12:rc-rl-circuits:tau}
\emph{ثابت الزمن}\index{ثابت الزمن} في \omterm{thm:g12:rc-rl-circuits:charging}{الدارة RC} هو
$\tau = RC$. والوحدات: $\unit{\ohm} \times \unit{F} =
(\unit{V/A})(\unit{C/V}) = \unit{C/A} = \unit{s}$ --- أي وتيرة
الدارة الخاصة.
\end{definition}

\begin{method}[قراءة $\tau$ على المنحني]\label{met:g12:rc-rl-circuits:reading}
ثلاث قراءات متكافئة، مأخوذة مباشرة من مخطط:
\begin{enumerate}
  \item \emph{المماسّ في المبدأ:} $u'(0) = \mathcal{E}/\tau$، ومن ثم
        يبلغ المماسّ الابتدائي المقاربَ عند $t = \tau$؛
  \item \emph{63\%:} $u(\tau) = \mathcal{E}(1 - e^{-1}) \approx
        0.63\,\mathcal{E}$؛
  \item \emph{95\%:} $u(3\tau) \approx 0.95\,\mathcal{E}$ --- أي مشحونة،
        عمليًّا ($5\tau$: $99\%$).
\end{enumerate}
\end{method}

\begin{proposition}[التفريغ]\label{prop:g12:rc-rl-circuits:discharge}
\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} المشحونة إلى $U_0$ والمغلقة على \omterm{def:g11:circuits-and-power:resistance}{مقاومة} $R$ وحدها تخضع
للعلاقة $RC\,du/dt + u = 0$ وتتبع
\[
u(t) = U_0\,e^{-t/\tau}, \qquad \tau = RC
\]
--- أي $37\%$ متبقية عند $\tau$، و$5\%$ عند $3\tau$.
\end{proposition}

\begin{proof}
قانون الحلقة نفسه، بلا منبع؛ وبالتعويض مرة أخرى:
$RC(-U_0/\tau)e^{-t/\tau} + U_0 e^{-t/\tau} = 0$، مع $u(0) = U_0$.
\end{proof}

\begin{example}[أرقام وحدة وماضة]\label{ex:g12:rc-rl-circuits:flashrc}
وماض تصوير يشحن $C = \qty{800}{\micro F}$ عبر
$R = \qty{1.0}{k\ohm}$: فيكون $\tau = 10^3 \times 8.0\times10^{-4} =
\qty{0.80}{s}$، ومن ثم ينتظر ضوء ``الجهوزية'' --- المضبوط ليعمل عند $95\%$ ---
مدةَ $3\tau \approx \qty{2.4}{s}$: وهو أزيز الخطّاف. وإذا أُطلق
عبر \qty{0.50}{\ohm} لا غير في أنبوب الوماض، فرغت \emph{\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} نفسها}
بثابت زمن قدره $\tau' = \qty{0.40}{ms}$: أي أسرع خروجًا بألفي مرة منها دخولًا.
\end{example}

\section{الوشيعة: المحاثّة والدارة RL}

\begin{definition}[الوشيعة والمحاثّة]\label{def:g12:rc-rl-circuits:inductor}
\emph{الوشيعة}\index{وشيعة} ملفّ من السلك: فتيارها
يخترق لفّاتها \omterm{def:g11:magnetic-fields:bfield}{بمجال مغناطيسي}
(\cref{ch:g11:magnetic-fields})؛ وكلما \emph{تغيّر}، ظهر \omterm{def:g11:circuits-and-power:voltage}{توتر}
بين طرفي الوشيعة:
\[
u = L\,\frac{di}{dt} .
\]
والثابت $L$ هو \emph{المحاثّة}\index{محاثّة}، وتقاس
\omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \emph{الهنري}\index{هنري} (\unit{H}): $\qty{1}{H} = \qty{1}{V.s/A}$.
\end{definition}
\admitted

\begin{remark}[الوشيعة تقاوم التغيّر]\label{rem:g12:rc-rl-circuits:oppose}
أما لماذا يولّد التيار المتغير توترًا --- أي التحريض --- فمستنتَج في
مجلدات الجامعة؛ وهنا نأخذ القانون وطبعه: \omterm{def:g12:rc-rl-circuits:inductor}{فالوشيعة}
\emph{تعاكس تغيّرات} تيارها. وهما توأمان متقابلان: فالمقدار $u$ في \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة}
لا يقدر أن يقفز، والمقدار $i$ في \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} لا يقدر أن يقفز.
\end{remark}

\begin{omfigure}
\begin{circuitikz}[font=\small]
  \draw (0,0) to[battery1, l=$\mathcal{E}$] (0,2.4) to[nos, l=$K$] (2.1,2.4)
    to[R=$R$] (4.2,2.4) -- (5.4,2.4) to[L=$L$, i=$i$, v=$u$] (5.4,0) -- (0,0);
\end{circuitikz}

{\small \omterm{prop:g12:rc-rl-circuits:rl}{الدارة RL}: بعد إغلاق $K$، لا تدع \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} التيار
يصعد إلا بوتيرتها الخاصة $\tau = L/R$.}
\end{omfigure}

\begin{proposition}[نمو التيار في الدارة RL]\label{prop:g12:rc-rl-circuits:rl}
\index{دارة RL}
إغلاق القاطعة على منبع $\mathcal{E}$، \omterm{def:g11:circuits-and-power:resistance}{ومقاومة} $R$، \omterm{def:g12:rc-rl-circuits:inductor}{ووشيعة}
$L$ على التسلسل يعطي $\mathcal{E} = Ri + L\,di/dt$، أي
\[
\frac{L}{R}\,\frac{di}{dt} + i = \frac{\mathcal{E}}{R},
\qquad\text{يحلّها}\quad
i(t) = \frac{\mathcal{E}}{R}\left(1 - e^{-t/\tau}\right),
\quad \tau = \frac{L}{R} .
\]
\end{proposition}

\begin{proof}
قانون الحلقة مرة أخرى؛ وهي \emph{المعادلة نفسها} الواردة في
\cref{thm:g12:rc-rl-circuits:charging} مع $u \to i$،
$\mathcal{E} \to \mathcal{E}/R$، $RC \to L/R$ --- والتعويض نفسه
يتحقق من الحلّ، وتنتقل كل قراءة من قراءات
\cref{met:g12:rc-rl-circuits:reading}.
\end{proof}

\begin{example}[مغناطيس كهربائي يستيقظ]\label{ex:g12:rc-rl-circuits:rl}
\omterm{def:g12:rc-rl-circuits:inductor}{وشيعة} مرحّل، $L = \qty{0.30}{H}$ و$R = \qty{6.0}{\ohm}$، على
مغذٍّ قدره \qty{12}{V}: فالتيار النهائي $\mathcal{E}/R = \qty{2.0}{A}$، على
الوتيرة $\tau = 0.30/6.0 = \qty{50}{ms}$ --- أي مستيقظة في $3\tau \approx
\qty{0.15}{s}$.
\end{example}

\begin{remark}[الشرارة عند الفتح]\label{rem:g12:rc-rl-circuits:spark}
افتح قاطعة على \omterm{def:g12:rc-rl-circuits:inductor}{وشيعة} عاملة، فيضطر $i$ إلى الانهيار حتى الصفر في جزء
من ميلي \omterm{def:g10:orders-of-magnitude:unit}{ثانية}: فيصير $di/dt$ ضخمًا ويبلغ $u = L\,di/dt$ مئات
الفولطات أو آلافها --- فتقفز شرارة عبر التماس. \omterm{def:g12:rc-rl-circuits:inductor}{ووشيعة} الإشعال في السيارة
تفعل ذلك عمدًا، آلاف المرات في الدقيقة.
\end{remark}

\section{الطاقة المخزونة}

\begin{proposition}[الطاقة في مكثّفة، والطاقة في وشيعة]\label{prop:g12:rc-rl-circuits:energy}
\index{طاقة!في مكثّفة}\index{طاقة!في وشيعة}
\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} $C$ المشحونة إلى \omterm{def:g11:circuits-and-power:voltage}{التوتر} $u$، \omterm{def:g12:rc-rl-circuits:inductor}{والوشيعة} $L$ التي تحمل التيار
$i$، تخزنان
\[
E_C = \tfrac12\,C u^2 = \frac{q^2}{2C},
\qquad
E_L = \tfrac12\,L i^2 .
\]
\end{proposition}

\begin{proof}
إنزال شحنة صغيرة $dq$ على صفيحتين هما أصلًا عند \omterm{def:g11:circuits-and-power:voltage}{التوتر} $u$ يكلّف
$u\,dq$ (\cref{def:g11:circuits-and-power:voltage})؛ وبتكديس
التكاليف، يكون المجموع هو المساحة تحت المستقيم $u = q/C$ --- أي المثلث
الآتي --- ومن ثم $E_C = \tfrac12 qu = \tfrac12 Cu^2$. والمثلث نفسه في حالة
\omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} (فرفع $i$ بمقدار $di$ يكلّف $Li\,di$) يعطي
$E_L = \tfrac12 Li^2$.
\end{proof}

\begin{example}[رتب قدر الطاقة]\label{ex:g12:rc-rl-circuits:energies}
\omterm{def:g12:rc-rl-circuits:capacitor}{مكثّفة} الوماض ذات \cref{ex:g12:rc-rl-circuits:flashrc}:
$\tfrac12 \times 8.0\times10^{-4} \times 300^2 = \qty{36}{J}$. \omterm{def:g12:rc-rl-circuits:capacitor}{ومكثّفة} فائقة
قدرها \qty{3000}{F} عند \qty{2.7}{V}: \qty{1.1e4}{J} --- أي ما يعادل
بطارية AA. \omterm{def:g12:rc-rl-circuits:inductor}{ووشيعة} قدرها \qty{10}{mH} عند \qty{10}{A}: \qty{0.50}{J}.
وهي أقل مما تحمله البطاريات --- لكنها قابلة للتحرير في ميلي \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة.
\end{example}

\begin{remark}[فيمَ يصلح حافظا الزمن هذان]\label{rem:g12:rc-rl-circuits:applications}
\index{تنعيم}\index{دارة توقيت}
البطء دخولًا \omterm{def:g12:kinematics-2d:velocity}{والسرعة} خروجًا يغذّيان وماض آلة التصوير ومزيل الرجفان. و
\omterm{def:g12:mechanical-waves:delay}{التأخير} $\tau = RC$ يشغّل \omterm{rem:g12:rc-rl-circuits:applications}{دارات التوقيت}: أي الماسحات المتقطعة، والمنارات
الوامضة، وإنارة بيت الدرج. \omterm{def:g12:rc-rl-circuits:capacitor}{والمكثّفة} الموضوعة على مغذٍّ \emph{تنعّم}
\omterm{def:g11:circuits-and-power:voltage}{التوتر} (إذ إنّ $u$ لا يقدر أن يقفز)؛ \omterm{def:g12:rc-rl-circuits:inductor}{والوشيعة} على التسلسل تنعّم التيار (إذ إنّ $i$ لا يقدر
أن يقفز). أما \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} مع \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} معًا فتصنعان مذبذبًا
(\cref{ch:g12:rlc-oscillations}).
\end{remark}

\section{مسألة: وماض آلة التصوير}

\begin{problem}[{مسألة نهاية الأسبوع --- وماض آلة التصوير: أزيز يدوم
ثانيتين، وميلي ثانية من المجد، وابن العمّ المعلَّق على الجدار الذي يعيد تشغيل
القلوب --- وأُسّي واحد يديرها جميعًا}]
\label{pb:g12:rc-rl-circuits:1}
\omterm{def:g10:orders-of-magnitude:unit}{وحدة} وماض مدمجة تخزن \omterm{def:g10:energy-conservation:energy}{الطاقة} في \omterm{def:g12:rc-rl-circuits:capacitor}{مكثّفة}
$C = \qty{800}{\micro F}$، مشحونة عبر $R = \qty{1.0}{k\ohm}$ من
مغيّر مُنمذَج بمنبع مثالي $\mathcal{E} = \qty{300}{V}$؛
ويعمل ضوء ``الجهوزية'' عند $95\%$ من \omterm{def:g11:circuits-and-power:voltage}{التوتر} الكامل. وإذا أُطلقت، لم تر
\omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} إلا أنبوب الوماض، $R_f = \qty{0.50}{\ohm}$. والمعطيات: خلية
AA توصّل \qty{1.5}{V} و\qty{2.4}{A.h}؛ $e = \qty{1.60e-19}{C}$.

\medskip
\noindent\textbf{الجزء I --- الخزّان.}
\begin{enumerate}
  \item احسب الشحنة على \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} عند \qty{300}{V}.
  \item وكم \omterm{def:g11:fundamental-interactions:elementary}{شحنة أولية} يعادل ذلك؟
  \item احسب \omterm{def:g10:energy-conservation:energy}{الطاقة} المخزونة $E_C$.
  \item وتحمل خلية AA مقدار $q = It \approx \qty{8.6e3}{C}$، أي نحو
        \qty{1.3e4}{J}. فكم ضعفًا من \omterm{def:g10:energy-conservation:energy}{طاقة} \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} يعادل ذلك؟
  \item فلمَ العناء \omterm{def:g12:rc-rl-circuits:capacitor}{بالمكثّفة} إذًا؟ جملة واحدة، تستعمل ``\omterm{def:g10:energy-conservation:power}{الاستطاعة}''.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- انتظار الضوء الأخضر.}
\begin{enumerate}[resume]
  \item اكتب قانون الحلقة وحوّله إلى المعادلة التفاضلية
        من أجل $u(t)$.
  \item تحقق بالتعويض من أن $u(t) = \mathcal{E}(1 - e^{-t/RC})$
        تحلّها، انطلاقًا من $u(0) = 0$.
  \item احسب \omterm{def:g12:rc-rl-circuits:tau}{ثابت الزمن} $\tau$.
  \item وبعد أي زمن يصير ضوء الجهوزية أخضر، وعند أي
        \omterm{def:g11:circuits-and-power:voltage}{توتر}؟ وتعرّف على أزيز الثانيتين.
  \item احسب تيار الشحن الابتدائي؛ ولماذا يخبو بعد ذلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- ميلي \omterm{def:g10:orders-of-magnitude:unit}{ثانية} المجد.}
\begin{enumerate}[resume]
  \item اكتب قانون التفريغ $u(t)$ عبر الأنبوب واحسب
        \omterm{def:g12:rc-rl-circuits:tau}{ثابت الزمن} الجديد $\tau'$.
  \item احسب التيار الأعظمي عبر الأنبوب.
  \item احسب \omterm{def:g10:energy-conservation:power}{الاستطاعة} العظمى؛ وقارنها بغلاية قدرها \qty{2.2}{kW}.
  \item وبأخذ الوماض منتهيًا عند $3\tau'$، أعطِ مدته و
        استطاعته المتوسطة.
  \item شكّل نسبة زمن الشحن إلى زمن الوماض: \omterm{def:g12:rc-rl-circuits:capacitor}{فالمكثّفة} نفسها،
        والشحنة نفسها --- فما الذي تغيّر، وما الذي يجعل من
        \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة}: خزّان \omterm{def:g10:energy-conservation:energy}{طاقة} أم رافعة \omterm{def:g11:work-of-force:power}{استطاعة}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- ابن العمّ على الجدار.}
يجب أن يوصّل مزيل الرجفان $E_C = \qty{200}{J}$ عند
$U_0 = \qty{1500}{V}$؛ ويقدّم الصدر والصفيحتان نحو \qty{50}{\ohm}.
\begin{enumerate}[resume]
  \item احسب \omterm{def:g12:rc-rl-circuits:capacitor}{السعة الكهربائية} المطلوبة.
  \item احسب الشحنة المخزونة.
  \item احسب \omterm{def:g12:rc-rl-circuits:tau}{ثابت الزمن} ومدة الصدمة ($3\tau$)؛
        وقارنهما بالوماض.
  \item احسب التيار الأعظمي \omterm{def:g10:energy-conservation:power}{والاستطاعة} العظمى في المريض.
  \item ويجب أن تجعل دارة إعادة الشحن الجهاز جاهزًا ($95\%$) في
        \qty{6.0}{s}: جِد \omterm{def:g11:circuits-and-power:resistance}{مقاومة} الشحن المطلوبة و
        تيار الشحن الابتدائي --- أي الأرقام التي سيطلبها مصمّم.
\end{enumerate}
\end{problem}
