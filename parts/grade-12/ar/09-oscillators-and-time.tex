\chapter{المذبذبات الميكانيكية وقياس الزمن}\label{ch:g12:oscillators-and-time}

تدقّ ساعة الجدّ: وكل دقّة نوّاس نحاسي يعبر
الوضع الشاقولي، مكررًا خمسة عشر مليون مرة منذ رأس السنة. وشظية كوارتز
في ساعتك تنثني \num{32768} مرة في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}؛ \omterm{def:g11:fundamental-interactions:ladder}{وذرة} سيزيوم
تدندن أسرع تسعة مليارات مرة. وكل ما يتأرجح يقدر أن يحفظ الوقت:
ويتبع هذا الفصل التأرجح، من هويغنس إلى التوابع.

\section{التذبذبات الحرة}

\begin{definition}[المذبذب والدور والتواتر والسعة]\label{def:g12:oscillators-and-time:oscillator}
\emph{المذبذب الميكانيكي}\index{مذبذب} جملة، إذا أُزيحت
عن \emph{موضع توازن}\index{موضع التوازن} مستقر وأُطلقت،
تأرجحت جيئةً وذهابًا حوله: وتلك هي
\emph{التذبذبات الحرة}\index{تذبذبات حرة}. وتتكرر الحركة
بعد \emph{الدور}\index{دور} $T_0$ (\unit{s})؛ و
\emph{التواتر}\index{تواتر} $f_0 = 1/T_0$
(\unit{Hz}، \cref{ch:g12:mechanical-waves})؛ و
\emph{السعة}\index{سعة} هي الإزاحة القصوى.
\end{definition}

\begin{example}[حديقة حيوان من المذبذبات]\label{ex:g12:oscillators-and-time:zoo}
أرجوحة ملعب: $T_0 \approx \qty{3}{s}$. ووتر لا في قيثارة:
$f_0 = \qty{110}{Hz}$. وبلورة كوارتز: \qty{32768}{Hz}. وتذبذب
السيزيوم الذي يعرّف \omterm{def:g10:orders-of-magnitude:unit}{الثانية}: نحو \qty{9.2e9}{Hz}. أي عشر رتب
من القدر، ووظيفة واحدة: العدّ.
\end{example}

\section{النوّاس البسيط}

\begin{definition}[النوّاس البسيط]\label{def:g12:oscillators-and-time:pendulum}
\emph{النوّاس البسيط}\index{نوّاس بسيط}: كرة صغيرة كتلتها
$m$ على سلك غير قابل للاستطالة طوله $L$ وكتلته مهمَلة،
تتأرجح في مستوٍ شاقولي. ومن القوتين المؤثرتين في الكرة
(\cref{ch:g12:newtons-laws})، يكون \omterm{def:g10:inertia:inventory}{الشدّ} عموديًّا على
الحركة، بينما تتجه مركّبة \omterm{def:g10:universal-gravitation:weight}{الوزن} على امتداد \omterm{def:g10:relative-motion:trajectory}{المسار} عائدةً
نحو أخفض نقطة: \omterm{def:g10:universal-gravitation:weight}{فالوزن} هو \emph{\omterm{def:g10:inertia:force}{القوة} المرجِعة}%
\index{قوة مرجِعة}، وهي تعاكس الإزاحة.
\end{definition}

\begin{proposition}[دور النوّاس البسيط]\label{prop:g12:oscillators-and-time:pendulum-period}
\index{النوّاس البسيط!دوره}
في السعات الصغيرة (دون نحو $15^\circ$)، يكون \omterm{def:g12:oscillators-and-time:oscillator}{الدور}
\[
T_0 = 2\pi \sqrt{\frac{L}{g}}, \qquad g = \qty{9.81}{m/s^2}.
\]
وهو لا يتوقف على كتلة الكرة ولا --- ما دامت الزاوية
صغيرة --- على \omterm{def:g12:oscillators-and-time:oscillator}{السعة}: فالتأرجحات الصغيرة
\emph{متساوية الأزمنة}\index{تساوي الأزمنة}، وتستغرق كلها الزمن نفسه.
\end{proposition}
\admitted

\begin{remark}[التحليل البُعدي، وأين يسكن $2\pi$]\label{rem:g12:oscillators-and-time:dimensional}
يمكن تخمين الصيغة. فمن $m$ (\unit{kg}) و$L$ (\unit{m}) و
$g$ (\unit{m/s^2})، يكون التركيب الوحيد الذي له \omterm{def:g10:orders-of-magnitude:unit}{وحدة} زمن هو
$\sqrt{L/g}$ --- والكتلة \emph{لا تقدر} أن تدخل، إذ لا معطى آخر
يحمل كيلوغرامًا يلغيها. ولا تقدر الأبعاد أن تعطي العدد
الخالص في المقدمة: إذ إنّ $2\pi$ تأتي من حلّ معادلة الحركة،
وهو ما يُجرى في مجلد السنة الأولى. كما أن شرط الزاوية الصغيرة ليس اختياريًّا:
فعند $30^\circ$ يكون \omterm{def:g12:oscillators-and-time:oscillator}{الدور} الحقيقي أطول بمقدار $2\%$، ويخفق \omterm{prop:g12:oscillators-and-time:pendulum-period}{تساوي الأزمنة}.
\end{remark}

\begin{example}[نوّاس الثواني]\label{ex:g12:oscillators-and-time:seconds}
أيّ طول يدقّ \omterm{def:g10:orders-of-magnitude:unit}{الثانية}، $T_0 = \qty{2.00}{s}$، دقةً في كل
اتجاه؟ بالحلّ، $L = g\,T_0^2/4\pi^2 = 9.81 \times 2.00^2/4\pi^2 \approx
\qty{0.994}{m}$. أي قرابة \omterm{def:g10:orders-of-magnitude:unit}{المتر}: ولذلك تكون ساعات النوّاس طويلة،
لأن الثواني طويلة.
\end{example}

\section{مذبذب الزنبرك والكتلة}

\begin{definition}[القوة المرجِعة لزنبرك]\label{def:g12:oscillators-and-time:spring}
منزلقة كتلتها $m$ تنزلق على سكة أفقية بلا \omterm{def:g10:inertia:inventory}{احتكاك}، مربوطة
بزنبرك \emph{صلابته}\index{صلابة} $k$ (\unit{N/m}). وليكن
$x$ إزاحتها عن التوازن. فالزنبرك، ممدودًا كان أو مضغوطًا،
يسحب المنزلقة أو يدفعها عائدةً \omterm{def:g10:inertia:force}{بالقوة} $F = -kx$:
متناسبةً مع الإزاحة ومعاكسةً لها --- أي \omterm{def:g12:oscillators-and-time:pendulum}{قوة مرجِعة}
مرة أخرى، توفّرها المرونة الآن.
\end{definition}

\begin{proposition}[معادلة الحركة وحلّها]\label{prop:g12:oscillators-and-time:spring-period}
\index{مذبذب الزنبرك والكتلة}
يعطي \omterm{thm:g12:newtons-laws:second}{قانون نيوتن الثاني} على امتداد السكة (\cref{ch:g12:newtons-laws})
العلاقة $m \frac{d^2x}{dt^2} = -kx$، أي\ $\frac{d^2x}{dt^2} =
-\frac{k}{m}\,x$. ومهما يكن الثابتان $A$ و$\varphi$، فإن
\[
x(t) = A \cos\!\left(\frac{2\pi t}{T_0} + \varphi\right),
\qquad T_0 = 2\pi\sqrt{\frac{m}{k}},
\]
يحقق هذه المعادلة: أي تذبذب دوره $T_0$، مستقل
عن \omterm{def:g12:oscillators-and-time:oscillator}{السعة}.
\end{proposition}

\begin{proof}
بالاشتقاق، $\frac{dx}{dt} = -\frac{2\pi}{T_0}A \sin(\frac{2\pi
t}{T_0} + \varphi)$؛ وبالاشتقاق مرة أخرى، $\frac{d^2x}{dt^2} = -(\frac{2\pi}{T_0})^2
A\cos(\frac{2\pi t}{T_0} + \varphi) = -(\frac{2\pi}{T_0})^2\,x(t)$،
وهو $-(k/m)\,x$ بالضبط حين $(2\pi/T_0)^2 = k/m$.
\end{proof}

\begin{remark}[كل الحركات الموجودة]\label{rem:g12:oscillators-and-time:uniqueness}
أما كون \emph{كل} حركة للمنزلقة واحدةً من هذه الدوال فمقبول،
ومبرهن عليه في مجلد السنة الأولى. ولا وجود للمقدار $g$ في $2\pi\sqrt{m/k}$:
فالأصلب يتأرجح أسرع، والأثقل أبطأ، والأمر نفسه على القمر.
\end{remark}

\begin{definition}[السعة والطور]\label{def:g12:oscillators-and-time:phase}
في $x(t) = A\cos(2\pi t/T_0 + \varphi)$ يكون الثابت الموجب $A$
هو \omterm{def:g12:oscillators-and-time:oscillator}{السعة} --- فالطرفان $x = \pm A$ --- ويكون $2\pi t/T_0 +
\varphi$ هو \emph{الطور}\index{طور}، وينمو بمقدار $2\pi$ في كل
\omterm{def:g12:oscillators-and-time:oscillator}{دور}؛ و\emph{الطور الابتدائي}\index{طور ابتدائي} $\varphi$
يسجّل البداية: فإذا أُطلقت من السكون عند $x = A$، كان $\varphi = 0$. فمثلًا
$m = \qty{0.20}{kg}$ على $k = \qty{20}{N/m}$ يعطي $T_0 =
2\pi\sqrt{0.20/20} = \qty{0.63}{s}$؛ وإذا سُحبت إلى \qty{5.0}{cm} ثم
أُطلقت، كان $x(t) = 5.0\cos(2\pi t/0.63)$ بالسنتيمترات.
\end{definition}

\section{الطاقة والتخامد والرنين}

\begin{proposition}[طاقة المذبذب]\label{prop:g12:oscillators-and-time:energy}
الزنبرك الممدود بمقدار $x$ يخزّن \omterm{def:g10:energy-conservation:energy}{الطاقة} الكامنة المرنة $E_p =
\tfrac12 kx^2$ (مبرَّرة في \cref{ch:g12:work-and-energy}). وأثناء
التذبذب بلا \omterm{def:g10:inertia:inventory}{احتكاك} تكون \omterm{def:g11:mechanical-energy:em}{الطاقة الميكانيكية}
\[
E_m = E_k + E_p = \tfrac12 m v^2 + \tfrac12 k x^2 = \tfrac12 k A^2
\]
ثابتة: مرنةً كلها عند الطرفين، وحركيةً كلها عند التوازن.
ويعزف النوّاس الثنائية نفسها بين $E_k$ و$E_p$
الثقالية (\cref{ch:g11:mechanical-energy}).
\end{proposition}

\begin{proof}
بأخذ $v = \frac{dx}{dt}$، تكون مشتقة $E_m$ هي $mv\frac{dv}{dt}
+ kx\frac{dx}{dt} = v\,(m\frac{d^2x}{dt^2} + kx) = 0$ بحسب معادلة
الحركة؛ وتُقرأ القيمة $\tfrac12 kA^2$ عند طرف.
\end{proof}

\begin{definition}[التذبذبات المتخامدة]\label{def:g12:oscillators-and-time:damping}
يستنزف \omterm{def:g10:inertia:inventory}{الاحتكاك} \omterm{def:g11:mechanical-energy:em}{الطاقة الميكانيكية} من كل \omterm{def:g12:oscillators-and-time:oscillator}{مذبذب} حقيقي: فتكون
تذبذباته الحرة \emph{متخامدة}\index{تخامد}. وتحت تخامد
خفيف تظل الجملة تتذبذب، \omterm{def:g12:oscillators-and-time:oscillator}{بسعة} تحتضر لكن بزمن تكرار
شبه ثابت، هو \emph{شبه \omterm{def:g12:oscillators-and-time:oscillator}{الدور}}%
\index{شبه الدور}، القريب من $T_0$؛ وتحت تخامد شديد (كالعسل)
تزحف عائدة إلى التوازن دون أن تتجاوزه قط؛ أما \omterm{def:g10:energy-conservation:energy}{الطاقة}،
وهي تتناسب مع مربع \omterm{def:g12:oscillators-and-time:oscillator}{السعة}، فتتفكك إلى حرارة.
\end{definition}

\begin{definition}[التذبذبات القسرية والرنين]\label{def:g12:oscillators-and-time:resonance}
ادفع مذبذبًا دوريًّا --- وهي \emph{التذبذبات القسرية}%
\index{تذبذبات قسرية} --- فيتذبذب بعد \omterm{def:g12:oscillators-and-time:phase}{طور} عابر قصير
\omterm{def:g12:oscillators-and-time:oscillator}{بتواتر} \emph{الإثارة} $f$، لا بتواتره هو. وتبلغ سعته
ذروتها حين يقترب $f$ من \emph{\omterm{def:g12:oscillators-and-time:oscillator}{التواتر} الذاتي}%
\index{تواتر ذاتي} $f_0$ \omterm{def:g12:oscillators-and-time:oscillator}{للمذبذب} الحرّ: وهذا هو
\emph{الرنين}\index{رنين}، إذ تصل كل دفعة على وتيرة
الحركة، فتغذّيها \omterm{def:g10:energy-conservation:energy}{بالطاقة} في كل دورة؛ وكلما خفّ \omterm{def:g12:oscillators-and-time:damping}{التخامد}
ارتفعت الذروة وحدّت.
\end{definition}

\begin{example}[الرنين خيرًا وشرًّا]\label{ex:g12:oscillators-and-time:resonance-zoo}
الوالد الذي يدفع أرجوحة يدفعها بتواترها هي --- أي \omterm{def:g12:oscillators-and-time:resonance}{الرنين}
موضوعًا في العمل. وفي \num{2000} تمايل جسر الألفية الجديد في لندن
تمايلًا مقلقًا حين تعانقت خطى الحشد مع \omterm{def:g12:oscillators-and-time:resonance}{تواتر ذاتي} قرب
\qty{1}{Hz}؛ فأُغلق سنتين لتركيب مخمِدات. وكل ساعة كوارتز تُبقي
بلورتها تغنّي بإثارتها عند \omterm{def:g12:oscillators-and-time:resonance}{الرنين}، \qty{32768}{Hz}.
\end{example}

\section{قياس الزمن}

\begin{method}[توقيت دور]\label{met:g12:oscillators-and-time:timing}
لا توقّت تذبذبة واحدة أبدًا: بل شغّل ساعة الإيقاف عند عبور الكرة
الوضع الشاقولي (فهي أسرع ما تكون، وأسهل ما يكون الحكم عليها)، وعُدّ $N = 50$ تذبذبة،
واقسم على $N$ --- فيصير خطأ ردّ فعل قدره \qty{0.2}{s} مقدارَ \qty{4}{ms}
على $T_0$. ثم كرّر، وقارن التجارب.
\end{method}

\begin{example}[ثلاثة قرون من الدقّ]\label{ex:g12:oscillators-and-time:clocks}
بنى هويغنس أول ساعة نوّاس سنة \num{1657}: فنقل \omterm{prop:g12:oscillators-and-time:pendulum-period}{تساوي الأزمنة}
الساعاتِ من ربع ساعة من الانحراف في اليوم إلى ثوانٍ،
ونقلتها النوّاسات المعوَّضة إلى أجزاء من \omterm{def:g10:orders-of-magnitude:unit}{الثانية} في اليوم. ثم استبدلت ساعة
اليد بالكوارتز (\num{1969}) بالقضيب بلورةً تنثني عند
\qty{32768}{Hz}: أي أعشار \omterm{def:g10:orders-of-magnitude:unit}{الثانية} في الشهر. ومنذ \num{1967} صارت
\omterm{def:g10:orders-of-magnitude:unit}{الثانية} نفسها \emph{معرَّفة} بتذبذب ذرّي: أي
مدة \num{9192631770} \omterm{def:g12:oscillators-and-time:oscillator}{دور} من الإشعاع المكروي لانتقال
داخلي في السيزيوم-133 (\cref{ch:g12:quantum-world}). وأفضل
ساعات السيزيوم تنحرف \omterm{def:g10:orders-of-magnitude:unit}{ثانية} واحدة في مئة مليون سنة.
\end{example}

\begin{remark}[لماذا نطارد النانوثانية]\label{rem:g12:oscillators-and-time:gps}
يجد مستقبِل GPS موضعه بتوقيت إشارات لاسلكية من
التوابع؛ والضوء يقطع \qty{30}{cm} في النانوثانية، ومن ثم فإن خطأ
ساعة قدره ميكروثانية واحدة يخطئ موضعك بمقدار \qty{300}{m}. ولذلك
تحمل التوابع ساعات ذرّية --- وعند تلك الدقة يتوقف معدل
الساعة على \omterm{def:g12:kinematics-2d:velocity}{سرعة} حركتها وعلى ارتفاعها، وهو أثر
نلقاه في \cref{ch:g12:special-relativity}.
\end{remark}

\section{مسألة: طاولة الساعاتي}

\begin{problem}[{مسألة نهاية الأسبوع --- طاولة الساعاتي: ساعة
جدار تُعاد إلى الوقت --- خمسون تأرجحة موقَّتة، وقضيب نحاسي
يتمدد في الصيف، ودفعة آلية الإفلات المهموسة، ومنافس من الكوارتز
--- والحكم يصدر بالثواني في الشهر}]
\label{pb:g12:oscillators-and-time:1}
تصل ساعة جدار بنوّاس إلى الطاولة ``تشتغل خطأً''. عامِل
قضيبها النحاسي وكرتها الثقيلة نوّاسًا بسيطًا طوله الفعّال
$L$، وكتلة كرته $m = \qty{1.2}{kg}$؛ ويعدّ التعشيق كل
تذبذبة كاملة على أنها \qty{2.000}{s} بالضبط.

\medskip
\noindent\textbf{الجزء I --- قياس نبض الساعة.}
\begin{enumerate}
  \item لماذا نوقّت $50$ تذبذبة بدل واحدة؟ قدّر الخطأ
        الباقي على $T_0$ إذا كان زمن ردّ فعلك \qty{0.2}{s}.
  \item تقرأ ساعة الإيقاف \qty{99.4}{s} من أجل $50$ تذبذبة: فما \omterm{def:g12:oscillators-and-time:oscillator}{الدور}؟
  \item استنتج الطول الفعّال الحالي للنوّاس.
  \item وأيّ طول يعطي \qty{2.000}{s} بالضبط؟ وهل يجب أن ترفع
        صامولة الضبط الكرةَ أم تخفضها، وبكم؟
  \item وإذا لم تُضبط، فهل الساعة سريعة أم بطيئة، وبكم \omterm{def:g10:orders-of-magnitude:unit}{ثانية} في اليوم؟
  \item والحساسية للمقدار $g$: ما \omterm{def:g12:oscillators-and-time:oscillator}{الدور} على القمر
        ($g = \qty{1.62}{m/s^2}$)؟ وما الخطأ اليومي في مرصد
        حيث $g = \qty{9.80}{m/s^2}$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- تراخي الصيف.} النحاس يتمدد: فتسخين
قدره $\Delta\theta$ يمدّ القضيب إلى $L' = L\,(1 + \lambda\,
\Delta\theta)$، حيث $\lambda = \num{1.9e-5}$ لكل \omterm{def:g10:light-spectra:kelvin}{كلفن}. وقد ضُبطت الساعة
عند \qty{20}{\celsius}.
\begin{enumerate}[resume]
  \item احسب $\Delta L$ لغرفة صيفية عند \qty{25}{\celsius}.
  \item باستعمال $\sqrt{1+x} \approx 1 + x/2$، بيّن أن
        $\Delta T_0 / T_0 = \tfrac12 \lambda\, \Delta\theta$.
  \item احسب $\Delta T_0/T_0$ والخطأ الصيفي اليومي: أسريعة أم بطيئة؟
  \item والسؤالان نفساهما لممرّ شتوي عند \qty{10}{\celsius}.
  \item استعملت ساعات الدقة القديمة نوّاسات ``المشواة'' التي تمزج قضبانًا
        نحاسية بأخرى فولاذية (وهي تتمدد نحو النصف) في اتجاهين
        متناوبين. اشرح كيف يقدر ذلك على حفظ المعدل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الدفعة المهموسة.} وبفكّ آلية
الإفلات، ينتصف تأرجح بدأ \omterm{def:g12:oscillators-and-time:oscillator}{بسعة} $\theta_m = 2.00^\circ$
في \qty{280}{s}؛ وفي الخدمة تبقي دفعة ضئيلة من
آلية الإفلات في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور} $\theta_m$ ثابتة.
\begin{enumerate}[resume]
  \item كم دورًا يستغرق التنصيف؟
  \item وعند الطرف تكون الكرة قد صعدت $h = L(1 - \cos\theta_m)
        \approx L\,\theta_m^2/2$ (بالراديان): استنتج $E_m = \tfrac12
        m g L\,\theta_m^2$ واحسبها
        (\cref{ch:g11:mechanical-energy}).
  \item \omterm{def:g10:energy-conservation:energy}{والطاقة} تتناسب مع مربع \omterm{def:g12:oscillators-and-time:oscillator}{السعة}: فأيّ جزء من
        $E_m$ يضيع في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور}؟
  \item استنتج \omterm{def:g10:energy-conservation:energy}{طاقة} كل دفعة، \omterm{def:g10:energy-conservation:power}{والاستطاعة} المتوسطة اللازمة.
  \item وينزل ثقل الإدارة $M = \qty{2.5}{kg}$ مسافة $h =
        \qty{1.0}{m}$ في الأسبوع: فهل تكفي تلك \omterm{def:g10:energy-conservation:energy}{الطاقة}، وبأي \omterm{def:g10:energy-conservation:efficiency}{مردود}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- منافس الكوارتز.} حركة كوارتز
تتذبذب عند $f = \qty{32768}{Hz}$ وتنحرف نحو
\qty{0.3}{s} في الشهر؛ والنوّاس المرمَّم ما يزال ينحرف
\qty{0.5}{s} في اليوم.
\begin{enumerate}[resume]
  \item احسب \omterm{def:g12:oscillators-and-time:oscillator}{دور} الكوارتز. وبيّن أن $\num{32768} = 2^{15}$: فلماذا تكون \omterm{def:g10:inertia:force}{قوة}
        من قوى العدد اثنين هي بالضبط ما تريده دارة ساعة؟
  \item احسب الانحرافين النسبيين للكوارتز وللنوّاس المرمَّم،
        وكلًّا منهما بالثواني في الشهر. فمن يفوز، وبأي عامل؟
  \item ومعيار السيزيوم --- إذ يصنع \num{9192631770} \omterm{def:g12:oscillators-and-time:oscillator}{دور} ثانيةً
        واحدة --- ينحرف \qty{1}{s} في مئة مليون سنة: فما
        انحرافه النسبي؟ وبكم رتبة من القدر يقلّ عن الكوارتز،
        ولماذا يقتضيه GPS؟
  \item الحكم، في جملة واحدة بأرقامها: ما الذي تكسبه الساعة المرمَّمة
        على الجدار، ومن يملك \omterm{def:g10:orders-of-magnitude:unit}{الثانية} الآن؟
\end{enumerate}
\end{problem}
