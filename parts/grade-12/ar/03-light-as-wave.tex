\chapter{الضوء موجةً: الحيود والتداخل}\label{ch:g12:light-as-wave}

أمِل غشاءً صابونيًّا نحو نافذة: فتنزلق الألوان عليه، مع أن
لا شيء في الصابون أو الماء ملوّن. وصوّب ليزرًا نحو شعرة واحدة:
فيُظهر الجدار خلفها لا ظلًّا رفيعًا كالشعرة بل شريطًا ساطعًا،
تقطعه خطوط قاتمة، وعرضه آلاف أضعاف عرض الشعرة.
والأشعة السائرة مستقيمة (\cref{ch:g10:refraction}) لا تفسّر أيًّا من
المشهدين. ويرقّي هذا الفصل الضوء إلى \omterm{def:g10:signals-and-waves:wave}{موجة} كاملة --- نصف
ميكرومتر من قمة إلى قمة --- ثم يصرف الفكرة: \omterm{def:g11:color-light-sources:lamps}{فليزر}
وجدار عارٍ سيقيسان شعرة وطول \omterm{def:g10:signals-and-waves:wave}{موجة} وأخاديد قرص CD.

\section{الضوء موجة}

\begin{definition}[الأمواج الضوئية]\label{def:g12:light-as-wave:lightwave}
\omterm{def:g10:light-spectra:white}{الضوء الأحادي اللون} --- أي لون نقي واحد لا يقدر موشور على تحليله
أكثر (\cref{ch:g10:light-spectra}) --- \emph{موجة
ضوئية}\index{موجة ضوئية} \omterm{def:g10:signals-and-waves:period}{دورية}: فهو يعبر الخلاء بسرعة $c = \qty{3.00e8}{m/s}$
ويتميز بتواتره $f$ أو \emph{بطول موجته في
الخلاء}\index{طول الموجة!للضوء} $\lambda = c/f$، وهو المسافة من
قمة إلى قمة في \cref{ch:g12:mechanical-waves}. وتستجيب العين من نحو
\qty{400}{nm} (بنفسجي) إلى \qty{800}{nm} (أحمر)؛
\omterm{def:g10:light-spectra:white}{والضوء الأبيض} يمزج \omterm{def:g11:electric-gravitational-fields:field}{المجال} كله.
\end{definition}

\begin{example}[رتب القدر]\label{ex:g12:light-as-wave:orders}
يصدر \omterm{def:g11:color-light-sources:lamps}{ليزر} الهيليوم-نيون ضوءًا أحمر طوله $\lambda = \qty{633}{nm} =
\qty{6.33e-7}{m}$، ومنه $f = c/\lambda \approx \qty{4.7e14}{Hz}$: فلا
إلكترونيات تعدّ بتلك السرعة --- وسيُقاس كل طول \omterm{def:g10:signals-and-waves:wave}{موجة} في هذا الفصل
بالهندسة. وللمقياس، فإن شعرة (نحو \qty{70}{\micro m})
تساوي مئة طول \omterm{def:g10:signals-and-waves:wave}{موجة} عرضًا. فما الذي يتموّج؟ ليس الهواء: بل \omterm{def:g11:electric-gravitational-fields:efield}{مجال كهربائي}
ومغناطيسي، موصوف وصفًا أمينًا في مجلد السنة الثانية؛ ولا يهمّ فيما يلي إلا
\omterm{def:g12:mechanical-waves:wavelength}{طول الموجة}.
\end{example}

\section{الحيود}

\begin{definition}[الحيود]\label{def:g12:light-as-wave:diffraction}
حين تلاقي \omterm{def:g10:signals-and-waves:wave}{موجة} فتحة أو عائقًا حجمه $a$ مقارب
لطول موجتها، تنتشر في المنطقة التي كانت الأشعة ستتركها مظلمة.
وهذا الانتشار هو \emph{الحيود}\index{حيود} --- وهو مهمَل
ما دام $\lambda/a$ ضئيلًا، ومهيمن كلما تقلص $a$ نحو $\lambda$.
\end{definition}

\begin{proposition}[نصف العرض الزاوي للانتشار]\label{prop:g12:light-as-wave:halfwidth}
شقّ عرضه $a$، مضاء \omterm{def:g12:mechanical-waves:wavelength}{بطول الموجة} $\lambda < a$، ينشر
الضوء حول جهة الورود ضمن نصف عرض زاوي
$\theta$ (بالراديان)؛ وتُظهر شاشة على المسافة $D$ شريطًا
ساطعًا مركزيًّا عرضه $L$، أي ضعف $D \tan\theta \approx D\,\theta$ (في الزوايا
الصغيرة):
\[
\theta \approx \frac{\lambda}{a}, \qquad L = \frac{2\lambda D}{a}.
\]
\end{proposition}
\admitted

\begin{remark}[اقرأ الصيغة معكوسة]\label{rem:g12:light-as-wave:honest}
كلما ازداد الشقّ \emph{ضيقًا} ازداد الانتشار \emph{اتساعًا} --- أي
عكس تنبؤ الأشعة. أما الاشتقاق الأمين (بجمع الموجيات
المعادة الإصدار عبر الشقّ) فيقيم في مجلد السنة الثانية؛ وهنا
تكون الصيغة أداة قياس.
\end{remark}

\begin{example}[ليزر عبر شقّ]\label{ex:g12:light-as-wave:slit}
\omterm{def:g11:color-light-sources:lamps}{ليزر} أحمر، $\lambda = \qty{633}{nm}$؛ وشقّ عرضه $a = \qty{0.10}{mm}$؛
وشاشة على $D = \qty{2.00}{m}$. فيكون $\theta = \num{6.33e-7} /
\num{1.0e-4} = \qty{6.3e-3}{rad}$ --- أي ثلث درجة، لكن
الشريط المركزي على الجدار عرضه $L = 2 \times \num{6.33e-7} \times 2.00 /
\num{1.0e-4} \approx \qty{2.5}{cm}$، تكتنفه خطوط قاتمة
وأشرطة جانبية أخفت. ونصّف الشقّ يتضاعف الشريط.
\end{example}

\begin{proposition}[العائق يحيد كالشقّ]\label{prop:g12:light-as-wave:hair}
العائق المعتم الرفيع الذي عرضه $d$ --- كسلك أو شعرة --- ينتج،
خارج الحزمة الواردة، النمط نفسه الذي ينتجه شقّ عرضه $d$
(مبدأ بابينيه، المشتقّ في مجلد السنة الثانية).
\end{proposition}
\admitted

\begin{method}[قياس عرض ضئيل]\label{met:g12:light-as-wave:hairwidth}
\begin{enumerate}
  \item شُدّ الشعرة عبر حزمة \omterm{def:g11:color-light-sources:lamps}{الليزر}؛ وضع شاشة على مسافة
        مقيسة $D$ (بضعة أمتار).
  \item قِس عرض الشريط المركزي $L$، من خط قاتم إلى خط قاتم.
  \item وبما أن $L = 2\lambda D/d$: فإن $d = 2\lambda D/L$ --- فكلما رقّت الشعرة
        اتسع النمط.
\end{enumerate}
\end{method}

\begin{remark}[الحيود يحدّ كل أداة]\label{rem:g12:light-as-wave:limits}
يدخل الضوء آلة تصوير أو مقرابًا أو عينًا عبر فتحة
عرضها $a$، ومن ثم تُشوّش حتى \omterm{def:g11:lenses-and-eye:lens}{عدسة} تامة النقطةَ إلى بقعة
حجمها الزاوي نحو $\lambda/a$: فتندمج التفاصيل الأقرب. ولذلك لا يفرّق
مجهر بالضوء المرئي شيئًا دون نحو $\lambda \approx
\qty{0.5}{\micro m}$ (فالجراثيم نعم، والفيروسات لا)؛ وتُبنى مرايا
المقاريب واسعة للسبب نفسه.
\end{remark}

\section{التداخل}

\begin{definition}[المنابع المترابطة وفرق المسير]\label{def:g12:light-as-wave:coherent}
يكون منبعا \omterm{def:g10:signals-and-waves:wave}{موجة} \emph{مترابطين}\index{منبعان مترابطان} إذا
كان لهما \omterm{def:g10:signals-and-waves:frequency}{التواتر} نفسه واهتزّا على وتيرة واحدة --- وعمليًّا،
إذا كانا نسختين من \omterm{def:g10:signals-and-waves:wave}{موجة} واحدة، مأخوذتين بتقسيمها. وفي نقطة
$M$ تبلغها الموجتان، يكون \emph{فرق المسير}%
\index{فرق المسير} هو $\delta = S_2M - S_1M$.
\end{definition}

\begin{proposition}[شرطا التداخل]\label{prop:g12:light-as-wave:conditions}
حيث تتراكب موجتان مترابطتان طول موجتهما $\lambda$ تتجمعان ---
أي \emph{تتداخلان}\index{تداخل}:
\begin{itemize}
  \item $\delta = k\lambda$ (حيث $k$ صحيح): فهما على وتيرة واحدة وتتعزّزان ---
        أي تداخل \emph{بنّاء}\index{تداخل بنّاء} (ساطع)؛
  \item $\delta = (k + \tfrac12)\lambda$: فقمة على قاع، وتتلاشيان
        --- أي تداخل \emph{هدّام}\index{تداخل هدّام} (قاتم).
\end{itemize}
\end{proposition}

\begin{proof}
إزاحة \omterm{def:g12:mechanical-waves:periodic}{موجة دورية} بأطوال \omterm{def:g10:signals-and-waves:wave}{موجة} كاملة لا تغيّر شيئًا: فتتراكم القمم
على القمم. أما نصف طول \omterm{def:g10:signals-and-waves:wave}{موجة} زيادةً فيضع قمة على قاع: فتتلاشى
الموجتان المتساويتان.
\end{proof}

\begin{example}[تجربة يونغ بالشقّين]\label{ex:g12:light-as-wave:young}
اثقب شقّين رفيعين $S_1$ و$S_2$، يفصل بينهما جزء من المِلّيمتر،
في لوح معتم، وأضِئ كليهما \omterm{def:g11:color-light-sources:lamps}{بليزر} واحد: فيكون لديك \omterm{def:g12:light-as-wave:coherent}{منبعان
مترابطان}. وعلى شاشة على بعد مترين يكون التراكب مخطَّطًا:
فعند المركز $\delta = 0$، أي ساطع؛ وبالصعود على الشاشة ينمو $\delta$
عبر $\lambda/2$ (قاتم)، و$\lambda$ (ساطع)\,\dots\ أي ظلامًا
حيث يتلاشى الضوء مضافًا إلى الضوء.
\end{example}

\begin{definition}[الأهداب والبعد الهدبي]\label{def:g12:light-as-wave:fringes}
الأشرطة الساطعة والقاتمة المتناوبة في النمط هي
\emph{أهداب التداخل}\index{أهداب التداخل}؛ والمسافة
$i$ بين مركزين ساطعين متتاليين هي \emph{البعد الهدبي}%
\index{البعد الهدبي}.
\end{definition}

\begin{proposition}[البعد الهدبي]\label{prop:g12:light-as-wave:interfringe}
من أجل شقّين تفصل بينهما $a$ وشاشة على المسافة $D \gg a$،
تكون نقطة الشاشة الواقعة على المسافة $x$ من المركز $O$ ذات فرق
مسير $\delta \approx a\,x/D$ (وهي هندسة نقبلها)؛ وبحسب
\cref{prop:g12:light-as-wave:conditions}، تقع الأهداب الساطعة عند
$x_k = k\,\lambda D/a$، متساوية التباعد ببعد هدبي
\[
i = \frac{\lambda D}{a}.
\]
وبأخذ $\lambda = \qty{633}{nm}$ و$a = \qty{0.20}{mm}$ و$D =
\qty{2.0}{m}$: $i = \qty{6.3}{mm}$ --- فالتركيب يضخّم $\lambda$
بعامل $D/a = 10^4$. أما الهندسة وراء ذلك فتُعالَج في مجلد السنة
الثانية.
\end{proposition}
\admitted

\begin{method}[قياس طول موجة]\label{met:g12:light-as-wave:lambda}
\begin{enumerate}
  \item أرسِل \omterm{def:g11:color-light-sources:lamps}{الليزر} المجهول عبر شقّين مزدوجين تباعدهما معلوم
        $a$؛ وضع الشاشة على مسافة مقيسة $D$.
  \item قِس امتداد $10$ بعدًا هدبيًّا؛ واقسم على $10$: فتحصل على
        \omterm{def:g12:light-as-wave:fringes}{البعد الهدبي} $i$، وقد تقلص خطأ المسطرة عشر مرات.
  \item ومنه $\lambda = i\,a/D$.
\end{enumerate}
\end{method}

\begin{remark}[لماذا لا يتداخل مصباحان أبدًا]\label{rem:g12:light-as-wave:lamps}
مصباحا مكتب يضيئان جدارًا واحدًا لا ينتجان أهدابًا: فكلٌّ يصدر
قطارات موجية قصيرة غير مترابطة يقفز توقيتها النسبي عشوائيًّا
مليار مرة في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}، فتنزاح الأهداب بالسرعة نفسها و
تتوسّطها العين إلى ضوء منتظم. فالتداخل يقتضي الترابط
--- أي \omterm{def:g10:signals-and-waves:wave}{موجة} واحدة مقسومة قسمين، كما يفعل شقّا يونغ؛ \omterm{def:g11:color-light-sources:lamps}{والليزر}
هو الاختصار الحديث: \omterm{def:g10:signals-and-waves:wave}{موجة} واحدة نظيفة طويلة.
\end{remark}

\section{ألوان يرسمها التداخل}

\begin{definition}[التقزّح]\label{def:g12:light-as-wave:iridescence}
الألوان التي تنزاح مع زاوية النظر --- كفقاعات الصابون وبقع الزيت
وريش الطاووس وظهر قرص CD --- هي \emph{التقزّح}%
\index{تقزّح}: فلا صِبغة، بل تداخل في \omterm{def:g10:light-spectra:white}{الضوء الأبيض}،
يعزّز بعض أطوال \omterm{def:g10:signals-and-waves:wave}{الموجة} ويلغي غيرها.
\end{definition}

\begin{example}[الأغشية الصابونية وبقع الزيت]\label{ex:g12:light-as-wave:soap}
الضوء الساقط على غشاء صابوني ينعكس جزئيًّا عن وجهه الأمامي، وجزئيًّا
عن وجهه الخلفي: أي نسختان مترابطتان يحدّد فرق مسيرهما
السُّمك وزاوية النظر. وكل سُمك يردّ بعض
أطوال \omterm{def:g10:signals-and-waves:wave}{الموجة} معزَّزة، وبعضها ملغاة: فيعكس الغشاء لونًا
ينزلق كلما نزّ وترقّق. وبقعة زيت على إسفلت مبتلّ تلعب
الحيلة نفسها.
\end{example}

\begin{remark}[الضوء الأحادي اللون في مقابل الضوء الأبيض]\label{rem:g12:light-as-wave:white}
في \omterm{def:g10:light-spectra:white}{الضوء الأبيض} يرسم كل طول \omterm{def:g10:signals-and-waves:wave}{موجة} أهداب يونغ الخاصة به ببعد هدبي
$i = \lambda D/a$ خاص به: فعند المركز تكون كل الألوان ساطعة (أي هدب أبيض
واحد)؛ وبعد بضعة أهداب تتباعد الأنماط --- فحواف متقزّحة،
ثم تشوّش. أما \omterm{def:g10:light-spectra:white}{الضوء الأحادي اللون} فيعطي أهدابًا حادّة بلا
نهاية: ومن هنا \omterm{def:g11:color-light-sources:lamps}{الليزر} في كل قياس دقيق هنا.
\end{remark}

\section{مسألة: القياس بالضوء}

\begin{problem}[{مسألة نهاية الأسبوع --- القياس بالضوء: ليزر
ومسطرة وجدار عارٍ تصير ورشةً ميكرومترية تقيس شقًّا وشعرة
بشرية وطول موجة مجهولًا وأخاديد قرصين فضيّين}]
\label{pb:g12:light-as-wave:1}
عُدّتك: \omterm{def:g11:color-light-sources:lamps}{ليزر} أحمر ($\lambda = \qty{633}{nm}$، من اللافتة)، \omterm{def:g11:color-light-sources:lamps}{وليزر}
أخضر مجهول \omterm{def:g12:mechanical-waves:wavelength}{طول الموجة}، وشقّ معايَر
$a = \qty{100}{\micro m}$، وشقّان مزدوجان تباعدهما $a' =
\qty{0.250}{mm}$، وشريط قياس، وشعرتان، وقرص CD وقرص DVD.

\medskip
\noindent\textbf{الجزء I --- المعايرة على الشقّ المعلوم.}
\omterm{def:g11:color-light-sources:lamps}{الليزر} الأحمر، والشقّ، والجدار على $D = \qty{3.00}{m}$.
\begin{enumerate}
  \item احسب نصف العرض الزاوي $\theta$ للحزمة المحيدة.
  \item بيّن أن عرض الشريط المركزي $L = 2\lambda D/a$؛
        واحسبه.
  \item وتقيس $L = \qty{3.7}{cm}$: احسب الانحراف
        النسبي. فهل صحّت الطريقة؟
  \item وشقّ ثانٍ عرضه النصف، \qty{50}{\micro m}: تنبّأ بالعرض
        $L$ الخاص به. واذكر القاعدة التي تربط عرض الشقّ بعرض النمط.
  \item برّر خطوة السؤال 2 $\tan\theta \approx \theta$: قارن
        $\theta$ و$\tan\theta$ عند هذه الزاوية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الشعرة.} \omterm{def:g11:color-light-sources:lamps}{الليزر} نفسه والجدار نفسه؛ وتحلّ
شعرة مشدودة محلّ الشقّ.
\begin{enumerate}[resume]
  \item أيّ نتيجة مقبولة تتيح لك الاحتفاظ بصيغة الشقّ من أجل
        شعرة، وماذا تقول؟
  \item عرض الشريط المركزي $L = \qty{5.4}{cm}$: احسب
        قطر الشعرة $d$.
  \item ومسطرتك تقرأ $L$ في حدود $\pm\qty{0.2}{cm}$: أعطِ \omterm{def:g11:electric-gravitational-fields:field}{المجال} الناتج
        للقطر $d$، على الصورة $d \pm \Delta d$.
  \item وشعرة صديقك تعطي $L = \qty{7.6}{cm}$: فما قطرها؟
        وأيّ الشعرتين أرقّ، ولماذا يعني الأرقّ الأوسع؟
  \item وهل يقدر مصباح مكتب أن يحلّ محلّ \omterm{def:g11:color-light-sources:lamps}{الليزر}؟ أعطِ سببين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- طول \omterm{def:g10:signals-and-waves:wave}{موجة} \omterm{def:g11:color-light-sources:lamps}{الليزر} الأخضر.} \omterm{def:g11:color-light-sources:lamps}{الليزر}
الأخضر، والشقّان المزدوجان، والجدار على $D = \qty{2.00}{m}$.
\begin{enumerate}[resume]
  \item ولماذا يجب أن يضيء الشقّين \emph{\omterm{def:g11:color-light-sources:lamps}{الليزر} نفسه}؟
  \item ولماذا يكون الهدب المركزي ساطعًا مهما يكن \omterm{def:g12:mechanical-waves:wavelength}{طول الموجة}؟
  \item عشرة أبعاد هدبية تمتد على \qty{4.26}{cm}. أعطِ $i$، وسبب
        قياس عشرة دفعةً واحدة.
  \item استنتج طول \omterm{def:g10:signals-and-waves:wave}{موجة} \omterm{def:g11:color-light-sources:lamps}{الليزر} الأخضر.
  \item وتقول لافتة الصانع $\qty{532 \pm 10}{nm}$: أمتسق ذلك؟
        وأيّ لون تتوقع؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- قراءة الأقراص الفضية.} يصيب \omterm{def:g11:color-light-sources:lamps}{الليزر} الأحمر
قرص CD بورود ناظمي؛ وتعمل مساراته محزّزًا انعكاسيًّا
يخضع (بالقبول) للعلاقة $d \sin\theta = k\lambda$. وعلى الجدار على $D =
\qty{1.00}{m}$، تصنع كل حزمة من الرتبة الأولى بقعة على بعد $x$ من
البقعة المركزية.
\begin{enumerate}[resume]
  \item في قرص CD، $x = \qty{43}{cm}$: احسب $\theta$. ولماذا يُمنع
        اختصار الزوايا الصغيرة في الجزء I الآن؟
  \item استنتج تباعد مسارات القرص $d$.
  \item ويرسل قرص DVD بقعته من الرتبة الأولى إلى $x = \qty{1.65}{m}$:
        احسب $\theta$ وتباعد مساراته.
  \item قارن التباعدين. ولماذا يخزّن قرص DVD أكثر من قرص CD؟
  \item أغلِق الدفتر: أحصِ الأطوال المقيسة هذا الأسبوع،
        وقل في جملة واحدة ما الذي جعل الضوء مسطرةً للجميع.
\end{enumerate}
\end{problem}
