\chapter{التفكك الإشعاعي}\label{ch:g12:radioactive-decay}

في قبو متحف، يطقطق عدّاد غايغر فوق شظية خشب من تابوت
فرعون. وكل طقطقة بلا قانون: فلا شيء أعلن عنها، ولا شيء
يتنبأ بالتالية. ومع ذلك سيطبع المختبر يوم الجمعة تاريخًا مضبوطًا إلى
قرن. ويحلّ هذا الفصل المفارقة --- أي كيف تتجمع أحداث عشوائية تمامًا
واحدًا واحدًا، بالتريليونات، لتصير أثبت الساعات التي نملك.

\section{بلا قانون فرادى، وكآلة الساعة بالمول}

في السنة الماضية (\cref{ch:g11:nucleus-radioactivity}) لاقينا الأنوية
غير المستقرة، ووازنّا معادلات $\alpha$ و$\beta$، ورأينا نصف أي عيّنة
كبيرة يختفي في كل عمر نصف $T_{1/2}$. أما ما لم نستطع قوله فهو
\emph{متى بين الدرجات} --- ولا لماذا تحفظ عملية عشوائية
جدولًا أصلًا. وللسؤالين معًا جواب واحد.

\begin{definition}[ثابت التفكك]\label{def:g12:radioactive-decay:lambda}
لكل \omterm{def:g11:nucleus-radioactivity:notation}{نوكليد} \omterm{prop:g12:work-and-energy:equilibria}{غير مستقر} ثابت $\lambda$، هو \emph{ثابت
التفكك}\index{ثابت التفكك} الخاص به (ووحدته \unit{s^{-1}})، بحيث يكون لكل \omterm{def:g11:fundamental-interactions:ladder}{نواة}، خلال أي
فترة قصيرة $\dd t$، احتمال $p = \lambda\,\dd t$ أن
تتفكك --- وهو نفسه لكل \omterm{def:g11:fundamental-interactions:ladder}{نواة} من النوع، مهما يكن عمرها،
ولا تمسّه الحرارة ولا \omterm{def:g10:pressure:pressure}{الضغط} ولا الكيمياء.
\end{definition}

\begin{remark}[بلا قانون فرادى، مضبوط في الحشود]\label{rem:g12:radioactive-decay:crowds}
\omterm{def:g11:fundamental-interactions:ladder}{النواة} التي انتظرت عشرة آلاف سنة يتساوى احتمال تفككها
في هذه \omterm{def:g10:orders-of-magnitude:unit}{الثانية} مع \omterm{def:g11:fundamental-interactions:ladder}{نواة} صُنعت هذا الصباح: فلا بِلى، ولا إنذار --- وأيّهما
تذهب تاليًا أمر غير قابل للتنبؤ حقًّا. أما الحشود فشأن آخر. ارمِ
$100$ قطعة نقد: فتوقّع $50$ صورة، ولا تُدهش من $43$ ---
أي تقلبات من رتبة $\sqrt n$، وهنا $10\%$. وارمِ $10^{20}$: فيكون
التقلّب النسبي $1/\sqrt n$ مساويًا $10^{-10}$، والنتيجة يقينية إلى
عشرة أرقام عشرية. وغرام واحد من المادة يحوي نحو $10^{22}$ \omterm{def:g11:fundamental-interactions:ladder}{نواة}:
فينحلّ انعدام القانون بالتوسيط إلى آلة ساعة، أي قانونًا مضبوطًا من أجل $N(t)$، عدد الأنوية
التي ما تزال سليمة عند الزمن $t$.
\end{remark}

\section{قانون التفكك}

\begin{proposition}[معادلة التطور]\label{prop:g12:radioactive-decay:ode}
في عيّنة كبيرة من $N$ \omterm{def:g11:fundamental-interactions:ladder}{نواة}، يتغير العدد خلال $\dd t$ بالمقدار
\[
\dd N = -\lambda N\,\dd t, \qquad\text{أي}\qquad \frac{\dd N}{\dd t} = -\lambda N :
\]
فيتقلص العدد بوتيرة تتناسب معه هو نفسه.
\end{proposition}

\begin{proof}
كل \omterm{def:g11:fundamental-interactions:ladder}{نواة} من الأنوية $N$ تتفكك خلال $\dd t$ باحتمال
$\lambda\,\dd t$: فالتفككات المنتظَرة $N\lambda\,\dd t$، ومن أجل $N$ الكبيرة
يلتصق العدد الفعلي بالعدد المنتظَر
(\cref{rem:g12:radioactive-decay:crowds})؛ وكل تفكك يزيل \omterm{def:g11:fundamental-interactions:ladder}{نواة} واحدة.
\end{proof}

\begin{theorem}[قانون التفكك الإشعاعي]\label{thm:g12:radioactive-decay:law}
العيّنة التي تحوي $N_0$ \omterm{def:g11:fundamental-interactions:ladder}{نواة} عند $t = 0$ تحوي، عند الزمن $t$،
\[
N(t) = N_0 \, e^{-\lambda t}.
\]
\end{theorem}

\begin{proof}
تحقّق بالتعويض:
$\dd N/\dd t = N_0(-\lambda)\,e^{-\lambda t} = -\lambda N(t)$، و
$N(0) = N_0 e^0 = N_0$: فالمعادلة والانطلاقة كلتاهما صحيحتان. (أما أنه لا توافق
دالة \emph{أخرى} فمبرهَن عليه، بالتكاملات، في مجلد
الجامعة.)
\end{proof}

\begin{definition}[التناقص الأُسّي]\label{def:g12:radioactive-decay:exponential}
المقدار الذي يتبع $N_0\,e^{-\lambda t}$ يخضع \emph{لتناقص
أُسّي}\index{تناقص أُسّي}: أي إنه يخسر في أزمنة متساوية \emph{أجزاءً}
متساوية. وهو بالضبط \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} المفرِّغة في
\cref{ch:g12:rc-rl-circuits}، مع $\lambda$ مكان $1/(RC)$: فمعادلة
تفاضلية واحدة، تُتعلَّم مرة واحدة، تدير الدارات والأنوية سواء بسواء.
\end{definition}

\begin{proposition}[عمر النصف]\label{prop:g12:radioactive-decay:halflife}
\omterm{def:g11:nucleus-radioactivity:halflife}{عمر النصف} (\cref{def:g11:nucleus-radioactivity:halflife}) في \omterm{def:g11:nucleus-radioactivity:notation}{نوكليد}
يضبطه ثابت تفككه: $T_{1/2} = \ln 2/\lambda$، وبعد $n$
عمر نصف يكون $N = N_0/2^{\,n}$، صحيحًا كان $n$ أم لا.
\end{proposition}

\begin{proof}
يقتضي $N(T_{1/2}) = N_0/2$ أن يكون $e^{-\lambda T_{1/2}} = \tfrac12$، أي\
$\lambda T_{1/2} = \ln 2$؛ و$N(n\,T_{1/2}) = N_0\,e^{-n\ln 2} =
N_0/2^{\,n}$ من أجل أي $n$: فالأُسّي يخيط الدرَج
ويملأ ما بين الدرجات.
\end{proof}

\section{النشاط: عدّ الطقطقات}

\begin{proposition}[النشاط]\label{prop:g12:radioactive-decay:activity}
\omterm{def:g11:nucleus-radioactivity:activity}{نشاط} العيّنة --- أي تفككاتها في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية}، \omterm{def:g11:nucleus-radioactivity:activity}{بالبيكريل}
(\cref{def:g11:nucleus-radioactivity:activity}) --- هو
\[
\mathcal A = \lambda N, \qquad\text{ومنه}\qquad \mathcal A(t) = \mathcal A_0 \, e^{-\lambda t}:
\]
فيتناسب مع العدد الباقي، ويكون أُسّيًّا بالقدر نفسه.
\end{proposition}

\begin{proof}
$N$ \omterm{def:g11:fundamental-interactions:ladder}{نواة}، لكل منها احتمال $\lambda$ في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}: أي $\lambda N$
تفككًا في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}؛ وضرب \cref{thm:g12:radioactive-decay:law} في
$\lambda$ يعطي $\mathcal A(t) = \mathcal A_0 e^{-\lambda t}$.
\end{proof}

\begin{example}[$N$ ضخم، $\lambda$ ضئيل]\label{ex:g12:radioactive-decay:body}
تطقطق موزة عند نحو \qty{15}{Bq}، ويطقطق جسم إنسان عند نحو
\qty{8000}{Bq}، ويطقطق \omterm{def:g10:orders-of-magnitude:unit}{متر} مكعب من الغرانيت عند نحو \qty{e6}{Bq}
(\cref{ch:g11:nucleus-radioactivity}). وخلف الطقطقات المتواضعة تجلس
أعداد فلكية: فبوتاسيوم-40 في الجسم يسهم بالمقدار
$\mathcal A \approx \qty{4.4e3}{Bq}$ مع $\lambda =
\qty{1.8e-17}{s^{-1}}$: أي $N = \mathcal A/\lambda \approx \num{2.5e20}$
\omterm{def:g11:fundamental-interactions:ladder}{نواة}، لا يطلق منها في كل \omterm{def:g10:orders-of-magnitude:unit}{ثانية} إلا بضعة آلاف بالكاد.
\end{example}

\begin{method}[قياس عمر نصف]\label{met:g12:radioactive-decay:semilog}
\begin{enumerate}
  \item عُدّ التفككات بعدّاد غايغر على فترات متساوية
        متتالية: فهذا يعاين $\mathcal A$ عند أزمنة متتالية.
  \item ارسم $\ln \mathcal A$ بدلالة $t$: فبما أن
        $\ln \mathcal A = \ln \mathcal A_0 - \lambda t$، يظهر \omterm{def:g12:radioactive-decay:exponential}{التناقص
        الأُسّي} \emph{مستقيمًا} --- والمعطيات المعوجّة تعني
        قانونًا آخر، أو خليطًا من النوكليدات.
  \item اقرأ $\lambda$ بوصفه سالب الميل؛ $T_{1/2} = \ln 2/\lambda$.
\end{enumerate}
\end{method}

\begin{remark}[البيكريلات ليست ضررًا]\label{rem:g12:radioactive-decay:dose}
\omterm{def:g11:nucleus-radioactivity:activity}{البيكريل} يعدّ التفككات، لا الضرر. أما الضرر فتتعقبه \omterm{rem:g11:nucleus-radioactivity:dose}{الجرعة} \omterm{def:g10:orders-of-magnitude:unit}{بوحدة}
\omterm{rem:g11:nucleus-radioactivity:dose}{السيفرت} (\cref{ch:g11:nucleus-radioactivity})، وهي تزن \omterm{def:g10:energy-conservation:energy}{الطاقة}
المودعة في كل \omterm{def:g10:orders-of-magnitude:unit}{كيلوغرام} من النسيج بمقدار تخريب كل إشعاع:
فمسح تكنيسيومي قدره \qty{5e8}{Bq} أمر روتيني، وأقل منه بكثير من بيكريلات
مُصدر $\alpha$ مستنشَق ليس كذلك؛ وتحويل \unit{Bq} إلى \unit{Sv} مادةٌ
جامعية.
\end{remark}

\section{التأريخ: قراءة الزمن من نسبة}

قانون التفكك مُدارًا إلى الوراء ساعة: فقِس الجزء الباقي،
يسمِّ لك الأُسّي الزمن المنقضي.

\begin{method}[التأريخ بالكربون المشعّ]\label{met:g12:radioactive-decay:carbon}
\begin{enumerate}
  \item \emph{حيًّا:} يتبادل النسيج الحي الكربون مع الجو
        فيحمل نسبته من الكربون-14 --- أي \omterm{def:g11:fundamental-interactions:ladder}{ذرة} واحدة في كل $10^{12}$،
        فتعطي $\mathcal A_0 = 13.6$ تفككًا في الدقيقة لكل غرام من الكربون.
  \item \emph{والموت يشغّل الساعة:} فيتوقف الأخذ، وتتناقص النسبة
        بعمر نصف $T_{1/2} = \num{5730}$ سنة.
  \item قِس \omterm{def:g11:nucleus-radioactivity:activity}{نشاط} العيّنة $\mathcal A$ لكل غرام من الكربون؛
        فينقلب $\mathcal A = \mathcal A_0 e^{-\lambda t}$ إلى
        $t = \frac{1}{\lambda}\ln(\mathcal A_0/\mathcal A)
        = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}
\end{method}

\begin{example}[موقد في كهف]\label{ex:g12:radioactive-decay:hearth}
فحم نباتي من موقد مطمور يعطي $5.0$ تفككًا في الدقيقة لكل غرام:
$\mathcal A_0/\mathcal A = 13.6/5.0 = 2.72$، ومنه
$t = \num{5730} \times \ln(2.72)/\ln 2 \approx \num{8.3e3}$ سنة. فقد
انطفأت النار قبل ثمانية آلاف سنة --- والخشب نفسه هو الأرشيف.
\end{example}

\begin{example}[ساعات للصخور]\label{ex:g12:radioactive-decay:rocks}
يصمت الكربون-14 بعد نحو \num{50000} سنة؛ والجيولوجيا تحفظ ساعات
أبطأ. \emph{البوتاسيوم--الأرغون}: فالبوتاسيوم-40 ($T_{1/2} = \num{1.25e9}$
سنة) يتفكك إلى أرغون، وهو غاز \omterm{rem:g12:work-and-energy:escape}{يفلت} من الحمم المنصهرة لكنه يُحبس متى
تبلور الصخر --- فالتصلب يصفّر الساعة. \emph{واليورانيوم--الرصاص}:
فاليورانيوم-238 يترأس سلسلة تفككات تنتهي عند الرصاص-206 \omterm{prop:g12:work-and-energy:equilibria}{المستقر}
(\cref{ch:g11:nucleus-radioactivity})، ونسبة الرصاص إلى اليورانيوم في
بلورة تؤرّخها. والساعتان تتفقان على أقدم المعادن
($\approx \num{4.4e9}$ سنة)، وتتفقان، عبر النيازك، على عمر المجموعة
الشمسية.
\end{example}

\begin{remark}[السلاسل، والغاز في القبو]\label{rem:g12:radioactive-decay:radon}
بين اليورانيوم والرصاص تمرّ السلسلة بالراديوم، ثم بالرادون-222،
وهو \emph{غاز} مشعّ ($T_{1/2} = 3.8$ يومًا). ويولد في الغرانيت و
التربة، فيتسرّب إلى الأقبية ويتراكم؛ وتتفكك بناته المُصدِرة
للجسيمات $\alpha$ في الرئتين --- وهي أكبر جرعة طبيعية يتلقاها معظم الناس، و
أرخصها خفضًا: فافتح نافذة القبو.
\end{remark}

\section{مسألة: عمر الأشياء}

\begin{problem}[{مسألة نهاية الأسبوع --- عمر الأشياء: مختبر متحف
يعاير ساعة الكربون، ويؤرّخ تمثالًا وبردية، ويتعلم لماذا لا تقرأ أي
ساعة ما وراء قرصها، ويدفع صخرة بركانية إلى الوراء مليار سنة}]
\label{pb:g12:radioactive-decay:1}
أسبوع في مختبر التأريخ في متحف. والمعطيات: النسيج الحي يظهر
$\mathcal A_0 = 13.6$ تفكك كربون-14 في الدقيقة لكل غرام من الكربون؛
$T_{1/2} = \num{5730}$ سنة (للكربون-14)، \num{1.25e9} سنة
(للبوتاسيوم-40)؛ $1$ سنة $= \qty{3.156e7}{s}$؛
$N_A = \qty{6.02e23}{mol^{-1}}$؛ والكتلة المولية للكربون \qty{12}{g/mol}؛
والعدّادات تحلّ نحو $0.2$ تفكك في الدقيقة لكل غرام.

\medskip
\noindent\textbf{الجزء I --- الاثنين: معايرة الساعة.}
\begin{enumerate}
  \item لماذا تكون نسبة الكربون-14 ثابتة في النسيج الحي، و
        ما الذي يتغير عند الموت؟
  \item احسب $\lambda$ للكربون-14 \omterm{def:g10:orders-of-magnitude:unit}{بوحدة} \unit{s^{-1}}.
  \item تحقق بالتعويض من أن $N_0 e^{-\lambda t}$ تحلّ
        $\dd N/\dd t = -\lambda N$.
  \item من $\mathcal A_0$، احسب عدد ذرات الكربون-14 في كل
        غرام من الكربون في النسيج الحي.
  \item وكم \omterm{def:g11:fundamental-interactions:ladder}{ذرة} كربون يحوي غرام واحد؟ استنتج نسبة الكربون-14
        --- وقارنها بعبارة ``واحد في كل $10^{12}$''.
  \item بيّن أن نشاطًا مقيسًا $\mathcal A$ يؤرّخ عيّنة بالعمر
        $t = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الثلاثاء: التمثال والبردية.}
\begin{enumerate}[resume]
  \item تمثال خشبي يعطي $9.1$ تفككًا في الدقيقة لكل غرام. أرّخ
        الخشب.
  \item وأيّ حدث بالضبط يؤرّخه ذلك --- النحت أم
        شيء آخر؟ وأيّ حذر يتبع ذلك؟
  \item بردية مصرية تعطي $10.4$ تفككًا في الدقيقة لكل غرام. أرّخها.
        فهل ناسخ من قبل اثنين وعشرين قرنًا أمر معقول؟
  \item \omterm{def:g11:nucleus-radioactivity:activity}{والنشاط} مقيس إلى $\pm 0.2$ تفكك في الدقيقة. قدّر
        ارتياب عمر البردية، مستعملًا
        $\Delta t \approx \Delta\mathcal A/(\lambda \mathcal A)$.
  \item ورقّ ``قديم'' عند تاجر يعطي $13.4$ تفككًا في الدقيقة
        لكل غرام. فما الحكم؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الخميس: حدود الساعة.}
\begin{enumerate}[resume]
  \item بعد كم عمر نصف ينزل \omterm{def:g11:nucleus-radioactivity:activity}{نشاط} عيّنة دون
        $0.2$ تفكك في الدقيقة لكل غرام عند العدّادات؟ وأيّ عمر ذلك؟
  \item وأفضل المختبرات تبلغ نحو عشرة أعمار نصف. فما
        الأفق العملي للتأريخ بالكربون المشعّ، بالسنوات؟
  \item عظم ديناصور عمره نحو \num{6.6e7} سنة: فكم عمر نصف
        للكربون-14 ذلك؟ وباستعمال السؤال 4، بعد كم عمر نصف
        لا تبقى \omterm{def:g11:fundamental-interactions:ladder}{ذرة} \emph{واحدة} من كربون-14 الغرام؟
        واستنتج بشأن ``تأريخ الديناصورات بالكربون''.
  \item وأيّ نوع من النوكليدات يقدر أن يؤرّخ صخر الديناصور بدلًا من ذلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الجمعة: الصخرة البركانية.}
\begin{enumerate}[resume]
  \item الأرغون غاز، والبوتاسيوم مقيم وفيّ في الصلب: فسّر
        لماذا يصفّر تصلب الحمم ساعة البوتاسيوم--الأرغون.
  \item وبفرض أن كل \omterm{def:g11:fundamental-interactions:ladder}{نواة} بوتاسيوم-40 متفككة تترك \omterm{def:g11:fundamental-interactions:ladder}{ذرة} أرغون
        واحدة محبوسة، بيّن أن
        $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.
  \item والصخر تحت أحفورة المتحف يظهر
        $N_{\mathrm{Ar}}/N_{\mathrm K} = 0.60$. أرّخ الصخر.
  \item فهل كان الكربون-14 يقدر أن يؤرّخ هذا الصخر، أو البوتاسيوم--الأرغون
        أن يؤرّخ التمثال؟ أعطِ نافذة كل ساعة المفيدة بأعمار النصف.
  \item تقرير الجمعة، في جملة واحدة: الأعمار الثلاثة الموجودة هذا الأسبوع، و
        القانون الواحد الذي قرأها جميعًا.
\end{enumerate}
\end{problem}
