\section*{Capítulo \ref{ch:g12:special-relativity} --- Relatividad especial: la dilatación del tiempo}

\begin{solution}{exo:g12:special-relativity:1}
$\gamma = 1/\sqrt{1 - (v/c)^2}$: $1.01$ ($0.1c$); $1.25$ ($0.6c$);
$1.67$ ($0.8c$); $10.0$ ($0.995c$).
\end{solution}

\begin{solution}{exo:g12:special-relativity:2}
$\gamma = 1.25$, así que el suelo mide
$1.25 \times 10.0 = \qty{12.5}{s}$. Los \qty{10.0}{s} de la nave son el
\omterm{def:g12:special-relativity:proper-time}{tiempo propio}: el metrónomo (un solo reloj) está presente en los dos
golpes.
\end{solution}

\begin{solution}{exo:g12:special-relativity:3}
$v = c\sqrt{1 - 1/\gamma^2}$: $\gamma = 2$ da $v = 0.866c$;
$\gamma = 10$ da $v = 0.995c$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:4}
$v = \qty{36.1}{m/s}$, $v/c = \num{1.2e-7}$;
$\gamma - 1 \approx (\num{1.2e-7})^2/2 \approx \num{7.2e-15}$. Pierde un
segundo al cabo de $1/\num{7.2e-15} \approx \qty{1.4e14}{s}$ --- unos
cuatro millones de años de conducción.
\end{solution}

\begin{solution}{exo:g12:special-relativity:5}
El pasajero va al encuentro del destello delantero, así que le llega
primero; por el segundo postulado, ambos destellos viajan a la misma $c$
sobre las mismas distancias de medio tren \emph{en su sistema}, luego
una llegada anterior significa un impacto anterior. La simultaneidad
depende del \omterm{def:g10:relative-motion:frame}{sistema de referencia}: cada veredicto es correcto en el
suyo.
\end{solution}

\begin{solution}{exo:g12:special-relativity:6}
$\gamma = 1/\sqrt{1 - 0.99^2} = 7.09$; vida media en el laboratorio,
$7.09 \times \num{2.6e-8} = \qty{1.8e-7}{s}$. \omterm{def:g12:projectile-motion:range}{Alcance} clásico,
$0.99 \times \num{3.00e8} \times \num{2.6e-8} \approx \qty{7.7}{m}$;
real, $7.09 \times 7.7 \approx \qty{55}{m}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:7}
$\gamma = 1.5$ en $v = c\sqrt{1 - 1/2.25} \approx 0.75c$; $\gamma = 3$
en $\approx 0.94c$. La asíntota: $\gamma$ (y el coste energético de
seguir acelerando) diverge cuando $v \to c$ --- ningún esfuerzo finito
lleva un cuerpo con masa a la velocidad de la luz.
\end{solution}

\begin{solution}{exo:g12:special-relativity:8}
$\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$; a $0.8c$,
$\gamma = 5/3$, luego $\Delta t = \qty{1.7e-8}{s}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:9}
El \omterm{def:g12:special-relativity:proper-time}{tiempo propio} pertenece al reloj presente en los dos \omterm{def:g12:special-relativity:proper-time}{sucesos}: (a)
quien lo lleva; (b) el muón; (c) la piloto --- solo la nave está a la
vez en la salida y en la llegada.
\end{solution}

\begin{solution}{exo:g12:special-relativity:10}
$\gamma - 1 \approx 7700^2/(2 \times \num{9.0e16}) \approx
\num{3.3e-10}$; a lo largo de \qty{1.58e7}{s} la astronauta se retrasa
$\num{3.3e-10} \times \num{1.58e7} \approx \qty{5.2e-3}{s}$ --- unos
cinco milisegundos más joven.
\end{solution}

\begin{solution}{exo:g12:special-relativity:11}
$E = \num{1.0e-3} \times (\num{3.00e8})^2 = \qty{9.0e13}{J}$. Una
central de \qty{1.0}{GW} entrega
$\num{1.0e9} \times 86400 \approx \qty{8.6e13}{J}$ al día --- un gramo
es un día de su producción --- y
$\num{9.0e13}/\num{8.5e5} \approx \num{e8}$: cien millones de coches
lanzados.
\end{solution}

\begin{solution}{exo:g12:special-relativity:12}
Elevando al cuadrado e invirtiendo: $\gamma^2(1 - v^2/c^2) = 1$, luego
$v^2/c^2 = 1 - 1/\gamma^2$ y, por tanto,
$v = c\sqrt{1 - 1/\gamma^2}$. Para $\gamma = 30$:
$v = c\sqrt{1 - 1/900} = 0.99944c$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:13}
Grosor contraído, $L = \qty{15}{km}/30 = \qty{500}{m}$; tiempo de
desfile, $500/(0.99944 \times \num{3.00e8}) \approx \qty{1.7e-6}{s} <
\qty{2.2}{\micro s}$. Los dos sistemas coinciden en que el muón llega al
suelo --- uno se lo achaca a una vida media estirada y el otro, a una
atmósfera encogida.
\end{solution}

\begin{solution}{exo:g12:special-relativity:14}
Laboratorio: $3200/\num{3.00e8} \approx \qty{1.1e-5}{s}$ (rapidez
$\approx c$). Sistema del electrón: el tubo queda contraído a
$3200/\num{1.0e5} = \qty{3.2}{cm}$, recorrido en
$\num{1.1e-5}/\num{1.0e5} \approx \qty{1.1e-10}{s}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:15}
Tierra: $4.2/0.9 \approx \qty{4.7}{años}$. Con
$\gamma = 1/\sqrt{1 - 0.81} = 2.29$, el reloj de la sonda registra
$4.7/2.29 \approx \qty{2.0}{años}$.
\end{solution}

\begin{solution}{pb:g12:special-relativity:1}
\textbf{1.} $c\,\Delta t_0 = \num{3.00e8} \times \num{2.2e-6} \approx
\qty{660}{m}$.

\textbf{2.} $\num{15000}/660 \approx 23$ vidas medias.

\textbf{3.} $\mathrm{e}^{-23} \approx \num{e-10}$: debería llegar menos
de un muón de cada diez mil millones y, sin embargo, el flujo es de uno
por \unit{cm^2} y por minuto --- el cielo contradice de plano la física
clásica.

\textbf{4.} Vida media desde el suelo,
$30 \times \qty{2.2}{\micro s} = \qty{66}{\micro s}$; \omterm{def:g12:projectile-motion:range}{alcance}
$\approx \num{3.00e8} \times \num{6.6e-5} \approx \qty{20}{km} >
\qty{15}{km}$: la llegada queda explicada.

\textbf{5.} $v = c\sqrt{1 - 1/900} = 0.99944c$.

\textbf{6.} $\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$.

\textbf{7.} En horizontal, $v\,\Delta t/2$; en vertical, $L$; y el medio
camino inclinado mide $c\,\Delta t/2$ porque el segundo postulado fija
en $c$ la velocidad del \omterm{def:g12:quantum-world:photon}{fotón} también en el sistema del suelo.

\textbf{8.} $(c\,\Delta t/2)^2 = L^2 + (v\,\Delta t/2)^2$ da
$\Delta t^2(c^2 - v^2) = 4L^2$, luego
$\Delta t = (2L/c)/\sqrt{1 - v^2/c^2} = \gamma\,\Delta t_0$.

\textbf{9.} $\gamma = 1/\sqrt{1 - 0.36} = 1.25$;
$\Delta t = \qty{1.25e-8}{s}$.

\textbf{10.} El de la nave: su único reloj está presente en los dos tics
y, por tanto, marca el \omterm{def:g12:special-relativity:proper-time}{tiempo propio}; el suelo necesita dos relojes
sincronizados.

\textbf{11.} $\gamma = 1/\sqrt{1 - 0.64} = 1/0.6 = 5/3 \approx 1.67$.

\textbf{12.} $10.0/(5/3) = \qty{6.0}{años}$.

\textbf{13.} Ida de $\qty{5.0}{años}$ a $0.8c$: $4.0$ \omterm{def:g10:orders-of-magnitude:lightyear}{años luz}.

\textbf{14.} $10.0 - 6.0 = \qty{4.0}{años}$: la astronauta es cuatro
años más joven que su gemela.

\textbf{15.} La situación no es simétrica: solo la astronauta da media
vuelta --- acelera y cambia de \omterm{def:g12:special-relativity:inertial}{sistema inercial}, mientras que la gemela
terrestre se queda en uno solo ---, así que solo su reloj registra el
tiempo más corto.

\textbf{16.} $v = 2\pi \times \num{2.66e7}/\num{43200} \approx
\qty{3.9e3}{m/s}$.

\textbf{17.} $\gamma - 1 \approx (\num{3.87e3})^2/(2 \times
\num{9.0e16}) \approx \num{8.3e-11}$.

\textbf{18.} $\num{8.3e-11} \times \num{86400} \approx \qty{7.2e-6}{s}
= \qty{7.2}{\micro s}$ de atraso al día.

\textbf{19.} $c \times \num{7.2e-6} \approx \qty{2.2}{km}$ de error de
posición al día.

\textbf{20.} Deriva neta, $45 - 7.2 \approx \qty{38}{\micro s}$ de
adelanto al día; error,
$c \times \num{38e-6} \approx \qty{11}{km}$ al día --- la factura
relativista que todo reloj GPS paga por adelantado, desafinado en tierra
para que marche exacto en \omterm{def:g10:universal-gravitation:satellite}{órbita}.
\end{solution}
