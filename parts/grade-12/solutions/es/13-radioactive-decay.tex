\section*{Capítulo \ref{ch:g12:radioactive-decay} --- La desintegración radiactiva}

\begin{solution}{exo:g12:radioactive-decay:1}
$T_{1/2} = \ln 2/\lambda = 0.693/\num{1.0e-6} = \qty{6.9e5}{s} \approx
8.0$ días --- el reloj del yodo-131.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:2}
$T_{1/2} = 3.8 \times \num{86400} = \qty{3.28e5}{s}$;
$\lambda = 0.693/\num{3.28e5} = \qty{2.1e-6}{s^{-1}}$;
$\mathcal A = \lambda N = \num{2.1e-6} \times \num{1.0e10} \approx
\qty{2.1e4}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:3}
$\dd N/\dd t = N_0(-\lambda)e^{-\lambda t} = -\lambda N$ --- la ecuación
se cumple; $N(0) = N_0 e^0 = N_0$;
$N(T_{1/2})/N_0 = e^{-\lambda \ln 2/\lambda} = e^{-\ln 2} = \tfrac12$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:4}
$90$ años $= 3\,T_{1/2}$:
$\mathcal A = \num{4.0e5}/2^3 = \qty{5.0e4}{Bq}$. $300$ años
$= 10\,T_{1/2}$: $\mathcal A = \num{4.0e5}/1024 \approx
\qty{3.9e2}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:5}
Pendiente $= (18.85 - 20.00)/10 = -0.115$, luego
$\lambda = \qty{0.115}{h^{-1}} = 0.115/3600 = \qty{3.2e-5}{s^{-1}}$;
$T_{1/2} = 0.693/0.115 = 6.0$ horas --- el tecnecio-99m.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:6}
$N = (1.0/226) \times \num{6.02e23} = \num{2.7e21}$;
$T_{1/2} = 1600 \times \num{3.156e7} = \qty{5.05e10}{s}$, luego
$\lambda = \qty{1.37e-11}{s^{-1}}$;
$\mathcal A = \lambda N \approx \qty{3.7e10}{Bq}$ --- un curio, la
\omterm{def:g11:nucleus-radioactivity:activity}{actividad} del gramo de radio de Marie Curie.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:7}
Los dos esperan que quede la mitad: $10$ y \num{1.0e20}. Las
fluctuaciones son $\sim 1/\sqrt{n}$ del recuento: con $10$
desintegraciones, alrededor de un $30\,\%$ --- $7$ o $13$ no
sorprenderían; con \num{1.0e20}, unos $10^{-10}$. Solo la muestra grande
obedece la ley con precisión medible.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:8}
$m = 140 \times \num{1.2e-4} = \qty{1.7e-2}{g}$;
$N = (0.0168/40) \times \num{6.02e23} = \num{2.5e20}$.
$\lambda = 0.693/(\num{1.25e9} \times \num{3.156e7}) =
\qty{1.76e-17}{s^{-1}}$;
$\mathcal A = \lambda N \approx \qty{4.4e3}{Bq}$ --- cerca de la mitad
del total del cuerpo (casi todo el resto es carbono-14).
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:9}
$\lambda = 0.693/6.0 = \qty{0.116}{h^{-1}}$. A las $15$ horas:
$\mathcal A = 500\,e^{-0.116 \times 15} = 500 \times 0.177 \approx
\qty{88}{MBq}$. Por debajo de \qty{1}{MBq}:
$t = \ln(500)/0.116 \approx \qty{54}{horas}$ --- unos $9$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos}.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:10}
Carbón vegetal: $3.4/13.6 = \tfrac14$, exactamente $2$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos}:
$t = \num{1.15e4}$ años. Hueso:
$t = \num{5730} \times \ln(1/0.30)/\ln 2 = \num{5730} \times 1.74
\approx \num{1.0e4}$ años.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:11}
$\lambda = 0.693/3.8 = \qty{0.182}{d^{-1}}$;
$t = \ln(100)/\lambda = 4.61/0.182 \approx 25$ días (unos $6.6$
\omterm{def:g12:oscillators-and-time:oscillator}{periodos}). Pero la cadena del uranio del granito y del suelo de
alrededor engendra radón sin parar: una ventilación puntual queda
deshecha en unas semanas, así que la ventilación tiene que ser
permanente.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:12}
De $N/N_0 = e^{-\lambda t}$ sale $\lambda t = \ln(N_0/N)$, y con
$\lambda = \ln 2/T_{1/2}$ queda $t = T_{1/2}\ln(N_0/N)/\ln 2$. Aquí,
$t = \num{5730} \times \ln(20)/\ln 2 = \num{5730} \times 4.32 \approx
\num{2.5e4}$ años.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:13}
$N_{\mathrm K} = N_0 e^{-\lambda t}$ y
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, luego
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$. Con el cociente
$3.0$: $e^{\lambda t} = 4$, luego $\lambda t = 2\ln 2$ y
$t = 2\,T_{1/2} = \num{2.5e9}$ años.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:14}
$\ln \mathcal A = 5.99$, $5.42$, $4.84$, $4.26$, $3.69$: caídas de
$0.57$--$0.58$ cada \qty{60}{s} --- alineados. Por la pendiente,
$\lambda = 2.30/240 = \qty{9.6e-3}{s^{-1}}$;
$T_{1/2} = 0.693/\num{9.6e-3} \approx \qty{72}{s}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:15}
(a) $2^{-10} \approx \num{1e-3}$. (b) $(1/2)^8 = 1/256 \approx
0.4\,\%$. (c) Se esperan $1000/8 = 125$ --- pero la ley calla sobre
cuáles: predice poblaciones con exactitud e individuos, en absoluto.
\end{solution}

\begin{solution}{pb:g12:radioactive-decay:1}
\textbf{1.} El intercambio con la atmósfera (comer, respirar) mantiene
la proporción rellenada en el valor atmosférico; la muerte detiene la
incorporación y toma el relevo la desintegración.

\textbf{2.} $T_{1/2} = \num{5730} \times \num{3.156e7} =
\qty{1.81e11}{s}$, luego $\lambda = 0.693/\num{1.81e11} =
\qty{3.8e-12}{s^{-1}}$.

\textbf{3.} $\dd N/\dd t = -\lambda N_0 e^{-\lambda t} = -\lambda N$, y
$N(0) = N_0$.

\textbf{4.} $\mathcal A_0 = 13.6/60 = \qty{0.227}{Bq}$ por gramo;
$N = \mathcal A_0/\lambda = 0.227/\num{3.8e-12} \approx \num{5.9e10}$
\omterm{def:g11:fundamental-interactions:ladder}{átomos} por gramo.

\textbf{5.} $(1/12) \times \num{6.02e23} = \num{5.0e22}$ \omterm{def:g11:fundamental-interactions:ladder}{átomos} de
carbono; proporción $\num{5.9e10}/\num{5.0e22} \approx \num{1.2e-12}$
--- uno de cada $10^{12}$, como estaba prometido.

\textbf{6.} De $\mathcal A = \mathcal A_0 e^{-\lambda t}$ sale
$\lambda t = \ln(\mathcal A_0/\mathcal A)$, y
$\lambda = \ln 2/T_{1/2}$ lo convierte en
$t = T_{1/2}\ln(\mathcal A_0/\mathcal A)/\ln 2$.

\textbf{7.} $t = \num{5730} \times \ln(13.6/9.1)/\ln 2 =
\num{5730} \times 0.580 \approx \num{3.3e3}$ años.

\textbf{8.} La muerte de la madera --- la tala del árbol, no la talla
de la estatua; una estatua labrada en madera vieja (o una falsificación
tallada en madera antigua) es anterior o posterior a la fecha de su
material.

\textbf{9.} $t = \num{5730} \times \ln(13.6/10.4)/\ln 2 =
\num{5730} \times 0.387 \approx \num{2.2e3}$ años --- del todo
compatible con un escriba de hace veintidós siglos.

\textbf{10.} $\Delta t \approx \Delta\mathcal A/(\lambda\mathcal A)$ con
$\lambda = \ln 2/5730 = \num{1.21e-4}$ por año:
$\Delta t = (0.2/10.4)/\num{1.21e-4} \approx \num{1.6e2}$ años --- una
fecha buena con un siglo o dos de margen.

\textbf{11.} $t = \num{5730} \times \ln(13.6/13.4)/\ln 2 \approx 120$
años --- indistinguible de lo moderno dentro de la incertidumbre de
$\pm 160$ años: el pergamino es reciente y la ``antigüedad'', una
falsificación.

\textbf{12.} $13.6/2^n < 0.2$ exige $2^n > 68$: $n = 7$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos} (pues
$2^7 = 128$), unos $\num{4.0e4}$ años.

\textbf{13.} Diez \omterm{def:g12:oscillators-and-time:oscillator}{periodos}: unos \num{5.7e4} años --- el horizonte
práctico, unos sesenta mil años.

\textbf{14.} $\num{6.6e7}/\num{5730} \approx \num{11500}$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos}. Los
\num{5.9e10} \omterm{def:g11:fundamental-interactions:ladder}{átomos} de un gramo se agotan cuando
$2^n > \num{5.9e10}$, es decir, con $n \approx 36$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos}
($\approx \num{2e5}$ años): mucho antes de la época de los dinosaurios
no queda ni un \omterm{def:g11:fundamental-interactions:ladder}{átomo} de carbono-14 --- no hay nada que contar.

\textbf{15.} Un reloj mucho más lento: potasio-40 o uranio-238, con
\omterm{def:g12:oscillators-and-time:oscillator}{periodos} de miles de millones de años.

\textbf{16.} La lava fundida deja escapar el argón en burbujas, así que
el cristal recién solidificado alberga potasio pero cero argón: el
cociente arranca en $0$ al solidificarse --- el reloj queda puesto a
cero.

\textbf{17.} Como en el ejercicio: $N_{\mathrm K} = N_0 e^{-\lambda t}$
y $N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, luego
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.

\textbf{18.} $e^{\lambda t} = 1.60$:
$t = \num{1.25e9} \times \ln(1.60)/\ln 2 = \num{1.25e9} \times 0.678
\approx \num{8.5e8}$ años.

\textbf{19.} No, por partida doble: a los \num{8.5e8} años el
carbono-14 lleva corridos $10^5$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos} --- silencio; y a los
\num{3.3e3} años el potasio-40 lleva \num{2.6e-6} de \omterm{def:g12:oscillators-and-time:oscillator}{periodo} --- el
cociente de argón $\approx \lambda t \sim \num{2e-6}$ es de una pequeñez
inmedible. Cada reloj solo marca desde alrededor de una décima de
\omterm{def:g12:oscillators-and-time:oscillator}{periodo} hasta unos diez.

\textbf{20.} Estatua, $\approx \num{3.3e3}$ años; papiro,
$\approx \num{2.2e3}$ años; roca, $\approx \num{8.5e8}$ años --- tres
relojes y una sola ley: $N = N_0\,e^{-\lambda t}$, leída al revés.
\end{solution}
