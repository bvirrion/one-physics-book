\section*{Capítulo \ref{ch:g12:newtons-laws} --- Las leyes de Newton}

\begin{solution}{exo:g12:newtons-laws:1}
(a) $p = 70 \times 10 = \qty{7.0e2}{kg.m/s}$.
(b) $v = \qty{13.9}{m/s}$: $p \approx \qty{1.8e4}{kg.m/s}$.
(c) $p = 0.0080 \times 800 = \qty{6.4}{kg.m/s}$. Coche $>$ velocista $>$
bala: cinco órdenes de magnitud de masa son más de lo que la rapidez
puede comprar.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:2}
$N = m(g + a_z)$: (a) $70 \times 11.01 = \qty{771}{N}$;
(b) $a_z = 0$, luego $N = mg = \qty{687}{N}$ --- la primera ley:
velocidad constante, \omterm{def:g10:inertia:compensate}{fuerzas que se compensan};
(c) $70 \times 8.61 = \qty{603}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:3}
$a = g\sin 30^\circ = \qty{4.9}{m/s^2}$;
$v = \sqrt{2 \times 4.905 \times 3.0} = \qty{5.4}{m/s}$. La masa se
cancela entre $ma$ y $mg\sin\alpha$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:4}
La inversión del sentido:
$\Delta p = 0.058 \times (25 + 15) = \qty{2.3}{kg.m/s}$;
$F = 2.32/0.0050 = \qty{4.6e2}{N}$ --- unas $800$ veces los
\qty{0.57}{N} del \omterm{def:g10:universal-gravitation:weight}{peso}, y por eso durante el contacto la gravedad se
desprecia.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:5}
\omterm{def:g10:universal-gravitation:weight}{Peso} (la Tierra sobre el libro, hacia abajo): compañera = el libro tira
de la Tierra, hacia arriba. \omterm{def:g12:newtons-laws:friction}{Reacción normal} (la mesa sobre el libro,
hacia arriba): compañera = el libro aprieta la mesa, hacia abajo. El
\omterm{def:g10:universal-gravitation:weight}{peso} y la normal actúan sobre el \emph{mismo} cuerpo y pertenecen a dos
interacciones distintas --- aquí son iguales solo porque
$\vect a = \vect 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:6}
$N = mg = \qty{490}{N}$. (a) $F > \mu_s N = \qty{221}{N}$.
(b) $a = (220.7 - 0.35 \times 490.5)/50 = \qty{0.98}{m/s^2}$.
(c) $F = \mu_k N = \qty{172}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:7}
(a) Desliza cuando $\tan\alpha > \mu_s$:
$\alpha = \arctan 0.45 \approx 24^\circ$.
(b) $a = g(\sin 30^\circ - 0.35\cos 30^\circ) = 9.81 \times 0.197 =
\qty{1.9}{m/s^2}$. (c) Sí: una vez en deslizamiento, el \omterm{def:g10:inertia:inventory}{rozamiento} cae a
$\mu_k < \mu_s$, luego
$a = g\cos\alpha\,(\tan\alpha - \mu_k) \approx \qty{0.9}{m/s^2} > 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:8}
$a = m_2 g/(m_1 + m_2) = 19.62/5.0 = \qty{3.9}{m/s^2}$;
$T = m_1 a = \qty{11.8}{N}$. El bloque colgante acelera hacia abajo, así
que la \omterm{def:g10:inertia:force}{fuerza} resultante sobre él apunta hacia abajo: $T$ tiene que
quedarse por debajo de $m_2 g$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:9}
$v_{\max} = \sqrt{\mu_s g R}$: en seco,
$\sqrt{0.70 \times 9.81 \times 50} = \qty{18.5}{m/s} \approx
\qty{67}{km/h}$; mojado, $\qty{14.0}{m/s} = \qty{50}{km/h}$. Tanto
$mv^2/R$ como $\mu_s mg$ llevan $m$: se cancela.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:10}
En vertical: $T\cos\theta = mg$, luego
$T = 2.943/\cos 25^\circ = \qty{3.2}{N}$. Radio
$R = L\sin\theta = \qty{0.507}{m}$; en la dirección radial:
$T\sin\theta = mv^2/R$, luego
$v = \sqrt{gR\tan\theta} = \sqrt{9.81 \times 0.507 \times 0.466}
= \qty{1.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:11}
$a_z = (N - mg)/m$ con $mg = \qty{49.05}{N}$: $+\qty{1.0}{m/s^2}$, $0$,
$-\qty{1.0}{m/s^2}$. Por ejemplo, un trayecto de subida: acelerar,
marcha uniforme, frenar. Sí: si ya venía bajando, las mismas lecturas
significan frenar, marcha uniforme y volver a acelerar hacia abajo --- la
báscula marca la aceleración, no la velocidad.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:12}
(a) $0 = 60V + 4.0 \times 8.0$: $V = \qty{-0.53}{m/s}$.
(b) El balón va a $8.0 + V$ respecto del suelo:
$0 = 60V + 4.0(8.0 + V)$, luego $V = -32/64 = \qty{-0.50}{m/s}$ y el
balón vuela a \qty{7.5}{m/s}. (c) El \omterm{def:g10:inertia:inventory}{empuje} del balón sobre sus manos ---
la compañera de tercera ley de su lanzamiento.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:13}
Sin \omterm{def:g10:inertia:inventory}{rozamiento}: $N\cos\theta = mg$ (vertical) y
$N\sin\theta = mv^2/R$ (radial); dividiendo,
$\tan\theta = v^2/(gR) = 400/784.8$: $\theta \approx 27^\circ$. Más
despacio: el coche tiende a resbalar peralte abajo y el \omterm{def:g10:inertia:inventory}{rozamiento} debe
apuntar pendiente arriba; más deprisa: al revés.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:14}
(a) $250 \times 2500 = \qty{6.3e5}{kg.m/s}$ cada segundo: \omterm{def:g10:inertia:inventory}{empuje}
$\qty{6.3e5}{N}$. (b) $P = \qty{4.9e5}{N}$;
$a = (6.25 - 4.905) \times 10^5 / \num{5.0e4} = \qty{2.7}{m/s^2}$.
(c) Empuja sobre sus propios gases (tercera ley), no sobre el aire.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:15}
(a) $m_2 g > \mu_s m_1 g$: $m_2 > 0.35 \times 4.0 = \qty{1.4}{kg}$.
(b) $a = (m_2 - \mu_k m_1)\,g/(m_1 + m_2) = 1.0 \times 9.81/6.0 =
\qty{1.6}{m/s^2}$; $T = m_2(g - a) = \qty{16.4}{N}$.
(c) Con $\mu_k = 0$ queda $a = m_2 g/(m_1 + m_2)$: la fórmula del
\cref{exo:g12:newtons-laws:8}.
\end{solution}

\begin{solution}{pb:g12:newtons-laws:1}
\textbf{1.} Sistema: cabina $+$ carro ($+$ pasajeros). La montaña está
fija al suelo, inercial con excelente aproximación. \omterm{def:g10:inertia:force}{Fuerzas}: el \omterm{def:g10:universal-gravitation:weight}{peso}
$M\vect g$ (hacia abajo), la \omterm{def:g12:newtons-laws:friction}{reacción normal} $\vect N$ del cable portador
(perpendicular a él) y la tensión $\vect T$ del cable tractor (pendiente
arriba).

\textbf{2.} En reposo, la primera ley: $\sum \vect F = \vect 0$. A lo
largo de la pendiente: $T = Mg\sin\alpha$; perpendicularmente:
$N = Mg\cos\alpha$.

\textbf{3.} $T = 4000 \times 9.81 \times 0.500 = \qty{19.6}{kN}$;
$N = 4000 \times 9.81 \times 0.866 = \qty{34.0}{kN}$.

\textbf{4.} En vacío: $T = \qty{9.8}{kN}$. Todas las \omterm{def:g10:inertia:force}{fuerzas} del
equilibrio son proporcionales a $M$, así que $T$ también.

\textbf{5.} A lo largo de la pendiente, $Ma = Mg\sin\alpha$: $M$ se
cancela, $a = g\sin\alpha = \qty{4.9}{m/s^2}$ pendiente abajo.

\textbf{6.} $T' = M(g\sin\alpha + a) = 4000 \times 5.505 =
\qty{22.0}{kN}$.

\textbf{7.} $a/(g\sin\alpha) = 0.60/4.905 \approx 12\,\%$.

\textbf{8.} $t = v/a = \qty{10}{s}$; $d = v^2/(2a) = \qty{30}{m}$.

\textbf{9.} Componente vertical de $\vect a$:
$a\sin\alpha = \qty{0.30}{m/s^2}$, luego
$N = m(g + a\sin\alpha) = 80 \times 10.11 = \qty{809}{N}$ frente a
$mg = \qty{785}{N}$: un $3\,\%$ más pesada.

\textbf{10.} El \omterm{def:g11:forces-and-motion:friction}{rozamiento estático} del suelo, horizontal:
$ma\cos\alpha = 80 \times 0.52 \approx \qty{42}{N}$. Al inclinarse, el
\omterm{def:g10:inertia:inventory}{empuje} total del suelo le pasa por el \omterm{def:g11:forces-and-motion:com}{centro de masas} y así no la
vuelca.

\textbf{11.} $a_T = dv/dt = 0$ (rapidez constante);
$a_n = v^2/R = 36/40 = \qty{0.90}{m/s^2}$, dirigida al centro ---
verticalmente hacia abajo en la cresta.

\textbf{12.} Eje normal (positivo hacia abajo): $Mg - N' = Mv^2/R$,
luego $N' = M(g - v^2/R) = 4000 \times 8.91 = \qty{35.6}{kN}$.

\textbf{13.} $N = m(g - v^2/R) = 80 \times 8.91 = \qty{713}{N}$: más
ligera en $v^2/(gR) = 0.90/9.81 \approx 9\,\%$.

\textbf{14.} $N' = 0$ cuando
$v = \sqrt{gR} = \qty{19.8}{m/s} \approx \qty{71}{km/h}$ --- más del
triple de la marcha de crucero: el carro va sobre el cable con un amplio
margen.

\textbf{15.} Por la \omterm{thm:g12:newtons-laws:third}{tercera ley de Newton}: \qty{35.6}{kN}, apretando
hacia abajo sobre el cable (y por tanto sobre el pilono) --- algo menos
que el \omterm{def:g10:universal-gravitation:weight}{peso} en reposo, aliviado por el vuelo curvo.

\textbf{16.} $F = \Delta p/\Delta t = 4000 \times 6.0/1.5 =
\qty{16}{kN}$, opuesta al movimiento ($a = \qty{4.0}{m/s^2}$).

\textbf{17.} $f = ma = 80 \times 4.0 = \qty{320}{N}$; el techo vale
$\mu_s mg = 0.40 \times 785 = \qty{314}{N} < \qty{320}{N}$: los pies
solos resbalan --- hay que agarrarse a la barra.

\textbf{18.} $0 = M_0 V + m(v_{\text{rel}} + V)$:
$V = -80 \times 2.5/2080 = \qty{-0.096}{m/s}$; el pasajero se mueve a
$2.5 - 0.096 \approx \qty{2.4}{m/s}$ respecto del suelo.

\textbf{19.} El cohete: su \omterm{def:g10:inertia:inventory}{empuje} queda fijado por la masa expulsada
por segundo multiplicada por la velocidad de eyección respecto de él.

\textbf{20.} La mayor tensión encontrada es la del arranque,
$T' = M \times \qty{5.505}{N/kg}$ (en reposo sale $Mg\sin\alpha$, y en la
cresta aún menos). Límite: $T' \leq \qty{20}{kN}$, luego
$M \leq 20000/5.505 = \qty{3.63e3}{kg}$: carga útil \qty{1.63e3}{kg}, es
decir, $20$ pasajeros de \qty{80}{kg} (no $25$). Cartel: \emph{20
personas}; entonces $M = \qty{3.6e3}{kg}$, $T' = \qty{19.8}{kN}$ y el
coeficiente de seguridad vale $100/19.8 \approx 5.1 \geq 5$.
\end{solution}
