\section*{Capítulo \ref{ch:g12:light-as-wave} --- La luz como onda: difracción e interferencias}

\begin{solution}{exo:g12:light-as-wave:1}
$f = c/\lambda$: rojo, $\num{3.00e8}/\num{7.0e-7} \approx
\qty{4.3e14}{Hz}$; violeta, $\approx \qty{7.5e14}{Hz}$ --- el violeta es
mayor (menor \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} con la misma velocidad $c$).
\end{solution}

\begin{solution}{exo:g12:light-as-wave:3}
Voz: $\lambda = 340/425 = \qty{0.80}{m}$, luego $\lambda/a = 1.0$ ---
\omterm{def:g12:light-as-wave:diffraction}{difracción} máxima, el \omterm{def:g10:signals-and-waves:sound}{sonido} inunda el pasillo. Luz:
$\lambda/a \approx \num{5.5e-7}/0.80 \approx \num{7e-7}$ --- una
extensión despreciable: la luz sigue recta y la esquina la esconde.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:6}

\begin{solution}{exo:g12:light-as-wave:7}
$i = \qty{4.7}{mm}$, luego
$\lambda = i\,a/D = \num{4.7e-3} \times \num{2.0e-4}/1.5 \approx
\qty{6.3e-7}{m} = \qty{630}{nm}$: rojo. Medir diez \omterm{def:g12:light-as-wave:fringes}{interfranjas} divide

\end{solution}

\begin{solution}{exo:g12:light-as-wave:8}
Los flexos tienen \omterm{def:g10:signals-and-waves:frequency}{frecuencias} más o menos parecidas, pero \emph{no van
al unísono}: cada uno emite trenes de \omterm{def:g10:signals-and-waves:wave}{onda} cortos con la fase saltando
al azar, así que las fuentes son incoherentes y las franjas instantáneas
se desplazan miles de millones de veces por segundo --- el ojo ve la
media, uniforme. Young ilumina las dos rendijas con \emph{una sola}
\omterm{def:g10:signals-and-waves:wave}{onda}: las dos copias heredan a la vez todos los saltos de fase y se
mantienen al unísono.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:9}
De la primera a la séptima hay $6i = \qty{18.0}{mm}$, luego
$i = \qty{3.0}{mm}$;
$\lambda = i\,a/D = \num{3.0e-3} \times \num{3.0e-4}/1.4 \approx
\qty{6.4e-7}{m} \approx \qty{640}{nm}$.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:10}
$\sin\theta = k\lambda/d = 0.396\,k$: con $k = 1$,
$\theta = 23^\circ$; con $k = 2$, $\sin\theta = 0.79$,
$\theta = 52^\circ$; con $k = 3$ haría falta $\sin\theta = 1.19 > 1$:
imposible. El orden más alto es $k = 2$.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:11}
$\theta = \num{5.5e-7}/0.10 = \qty{5.5e-6}{rad}$. Los faros subtienden
$1.5/\num{1.0e5} = \qty{1.5e-5}{rad}$, unas tres veces el difuminado:
separados, por poco.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:12}
Las reflexiones de la cara anterior y de la posterior \omterm{prop:g12:light-as-wave:conditions}{interfieren}; en
\omterm{def:g10:light-spectra:white}{luz blanca}, cada grosor refuerza unas \omterm{def:g12:mechanical-waves:wavelength}{longitudes de onda} y cancela
otras, de donde sale un color. El grosor de la lámina solo depende de la
altura, así que los puntos del mismo color forman bandas horizontales.
Al escurrir, la lámina se adelgaza, de modo que el grosor que pintaba un
color dado queda cada vez más abajo: las bandas se arrastran hacia
abajo.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:13}
$i = \lambda D/a$: violeta,
$\num{4.0e-7} \times 1.5/\num{2.0e-4} = \qty{3.0}{mm}$; rojo,
\qty{5.6}{mm}. En el centro: todos los colores brillantes --- una franja
blanca. A unos milímetros, los patrones de los colores se han separado:
franjas irisadas. Más lejos, los máximos de todos los colores se solapan
en todas partes: un difuminado blanco y uniforme.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:14}
$\theta = \num{5.5e-7}/2.4 = \qty{2.3e-7}{rad}$; a \qty{250}{km} eso son
$\num{2.3e-7} \times \num{2.5e5} \approx \qty{6}{cm}$. Puede contar
coches y localizar a una persona, pero la letra impresa (de milímetros)
queda veinte veces por debajo del límite de \omterm{def:g12:light-as-wave:diffraction}{difracción}: la afirmación es
un mito.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:15}
La \omterm{def:g10:signals-and-waves:frequency}{frecuencia} la fija la fuente: no cambia. La velocidad baja a $c/n$,
así que $\lambda' = \lambda/n$ y
$i' = i/n = 4.0/1.33 \approx \qty{3.0}{mm}$: las franjas se aprietan.
\end{solution}

\begin{solution}{pb:g12:light-as-wave:1}
\textbf{1.} $\theta = \lambda/a = \num{6.33e-7}/\num{1.00e-4} =
\qty{6.3e-3}{rad}$.

\textbf{2.} La semianchura vale
$D\tan\theta \approx D\theta = \lambda D/a$, luego
$L = 2\lambda D/a = 2 \times \num{6.33e-7} \times
3.00/\num{1.00e-4} = \qty{3.8}{cm}$.

\textbf{3.} $(3.8 - 3.7)/3.8 \approx 3\,\%$: dentro de la precisión de una
regla --- validado.

\textbf{4.} $L \propto 1/a$: reducir $a$ a la mitad duplica $L$, así que
$L = \qty{7.6}{cm}$. Obstáculo o rendija más estrechos, patrón más
ancho.

\textbf{5.} $\tan(\num{6.33e-3}) = \num{6.33008e-3}$: coinciden con una
diferencia relativa de unos $\num{1e-5}$ --- la aproximación es mucho
mejor que cualquier medida de aquí.

\textbf{6.} Por la \cref{prop:g12:light-as-wave:hair}: un obstáculo de
anchura $d$ difracta como una rendija de anchura $d$, así que
$L = 2\lambda D/d$ sigue valiendo.

\textbf{7.} $d = 2\lambda D/L = 2 \times \num{6.33e-7} \times
3.00/0.054 \approx \qty{70}{\micro m}$.

\textbf{8.} $L = \qty{5.2}{cm} \to d = \qty{73}{\micro m}$;
$L = \qty{5.6}{cm} \to d = \qty{68}{\micro m}$: por tanto,
$d = \qty{70 \pm 3}{\micro m}$.

\textbf{9.} $d = \num{3.80e-6}/0.076 = \qty{50}{\micro m}$: el de tu
amiga es más fino --- y como $L \propto 1/d$, el cabello más fino lanza
el patrón más ancho.

\textbf{10.} No: un flexo es blanco (cada \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} pinta un
patrón distinto y se difuminan) y es incoherente y ancho (no hay un haz
limpio que difractar). El \omterm{def:g11:color-light-sources:lamps}{láser} es monocromático y direccional.

\textbf{11.} Solo son \omterm{def:g12:light-as-wave:coherent}{coherentes} las copias de una misma \omterm{def:g10:signals-and-waves:wave}{onda}; dos
fuentes independientes se desfasan y las franjas se borran
(\cref{rem:g12:light-as-wave:lamps}).

\textbf{12.} En el centro $\delta = 0 = 0 \times \lambda$ para toda
$\lambda$: \omterm{prop:g12:light-as-wave:conditions}{constructiva} sea cual sea la \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda}.

\textbf{13.} $i = 4.26/10 = \qty{4.3}{mm}$; un tramo de diez se lee con
el mismo error de regla que una sola \omterm{def:g12:light-as-wave:fringes}{interfranja}, así que el error sobre
$i$ queda dividido entre diez.

\textbf{14.} $\lambda = i\,a'/D = \num{4.26e-3} \times
\num{2.50e-4}/2.00 = \num{5.33e-7} \approx \qty{533}{nm}$.

\textbf{15.} $\qty{533}{nm}$ cae dentro de $\qty{532 \pm 10}{nm}$: es
coherente; esa \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} es verde.

\textbf{16.} $\tan\theta = 0.43/1.00$, luego $\theta = 23^\circ$. Aquí
$\tan\theta = 0.43$ mientras que $\sin\theta = 0.40$: a $23^\circ$,
identificar $\sin$, $\tan$ y $\theta$ falla casi en un $10\,\%$ --- hay
que usar las funciones exactas.

\textbf{17.} $d = \lambda/\sin\theta = \num{6.33e-7}/0.396 \approx
\qty{1.6}{\micro m}$.

\textbf{18.} $\tan\theta = 1.65$, $\theta = 59^\circ$,
$\sin\theta = 0.855$: $d = \num{6.33e-7}/0.855 \approx
\qty{0.74}{\micro m}$.

\textbf{19.} $1.6/0.74 \approx 2.2$: las pistas del DVD son el doble de
densas, y sus hoyos, en consecuencia, más pequeños (se leen con un \omterm{def:g11:color-light-sources:lamps}{láser}
de \omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} más corta) --- varias veces los mismos datos en el
mismo disco de \qty{12}{cm}.

\textbf{20.} Cabello, \qty{70 \pm 3}{\micro m}; cabello de tu amiga,
\qty{50}{\micro m}; \omterm{def:g11:color-light-sources:lamps}{láser} verde, \qty{533}{nm}; pistas del CD,
\qty{1.6}{\micro m}; pistas del DVD, \qty{0.74}{\micro m} --- una
\omterm{def:g12:mechanical-waves:wavelength}{longitud de onda} conocida convierte la geometría de cualquier patrón en
una longitud, así que una luz de medio micrómetro es una regla graduada
en medio micrómetro.
\end{solution}
