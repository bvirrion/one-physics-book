\section*{Capítulo \ref{ch:g12:rlc-oscillations} --- Oscilaciones eléctricas libres: RLC}

\begin{solution}{exo:g12:rlc-oscillations:1}
(a) $\sqrt{LC} = \qty{1.0e-4}{s}$: $T_0 = \qty{0.63}{ms}$,
$f_0 \approx \qty{1.6}{kHz}$. (b) $\sqrt{LC} = \qty{3.16e-7}{s}$:
$T_0 = \qty{2.0}{\micro s}$, $f_0 \approx \qty{0.50}{MHz}$ --- un factor
$\sqrt{10^5} \approx 316$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:2}
$LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times \unit{C/A} =
\unit{s} \times \unit{s} = \unit{s^2}$, así que $\sqrt{LC}$ va en
segundos. El $2\pi$ es un número puro --- no tiene dimensión que
cambiar.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:3}
$Q_0 = CU_0 = \qty{2.8}{\micro C}$;
$E = \tfrac12 CU_0^2 = \tfrac12 \times \num{4.7e-7} \times 36 \approx
\qty{8.5}{\micro J}$. La carga oscila a $f_0 = 1/(2\pi\sqrt{LC})$, la
corriente va desfasada un cuarto de \omterm{def:g12:oscillators-and-time:oscillator}{periodo} y la \omterm{def:g10:energy-conservation:energy}{energía} se trasvasa
entre las placas y la \omterm{def:g12:rc-rl-circuits:inductor}{bobina}, con el total constante.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:4}
$Q_0 = \qty{2.0}{\micro C}$; $f_0 = 1/T_0 = \qty{250}{Hz}$;
$q(T_0/2) = \qty{-2.0}{\micro C}$; $i(0) = 0$ (el coseno es plano en
$t=0$); $I_0 = 2\pi \times \num{2.0e-6}/\num{4.0e-3} \approx
\qty{3.1}{mA}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:5}
Toda en el \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} cuando $q = \pm Q_0$ e $i = 0$ ($t = 0$,
$T_0/2$, $T_0$, \dots): los extremos del péndulo. Toda en la \omterm{def:g12:rc-rl-circuits:inductor}{bobina}
cuando $q = 0$ e $i = \pm I_0$ ($t = T_0/4$, $3T_0/4$, \dots): el punto
más bajo y más rápido del péndulo.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:6}
$d^2q/dt^2 = -(2\pi/T_0)^2\,Q_0\cos(2\pi t/T_0) = -(2\pi/T_0)^2 q$,
luego $L\,d^2q/dt^2 + q/C = q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$ para toda
$t$ exactamente cuando $(2\pi/T_0)^2 = 1/(LC)$, es decir,
$T_0 = 2\pi\sqrt{LC}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:7}
$L = \dfrac{1}{4\pi^2 f_0^2 C} = \dfrac{1}{39.5 \times
(\num{1.98e5})^2 \times \num{2.2e-10}} \approx \qty{2.9}{mH}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:8}
$E = \tfrac12 CU_0^2 = \qty{5.0e-5}{J}$; de $\tfrac12 LI_0^2 = E$ sale
$I_0 = \sqrt{2E/L} = \sqrt{\num{1.0e-2}} = \qty{0.10}{A}$, alcanzada por
primera vez en $T_0/4 = \tfrac14 \times \qty{0.63}{ms} \approx
\qty{0.16}{ms}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:9}
$E_C \propto \cos^2(2\pi t/T_0)$, y $\cos^2$ se repite cada
\emph{media} vuelta: dos llenados del \omterm{def:g12:rc-rl-circuits:capacitor}{condensador} por \omterm{def:g12:oscillators-and-time:oscillator}{periodo}, de donde
$2f_0$. En $t = T_0/8$ la \omterm{def:g12:oscillators-and-time:phase}{fase} vale $\pi/4$: $\cos^2(\pi/4) = \tfrac12$,
luego $E_C = E_L = E/2$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:10}
Cociente $3.0/4.0 = 0.75$ por \omterm{def:g12:rlc-oscillations:regimes}{seudoperiodo}: sexto máximo
$4.0 \times 0.75^6 \approx \qty{0.71}{V}$. La \omterm{def:g10:energy-conservation:energy}{energía} es proporcional al
cuadrado de la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}: sobrevive el $0.75^2 \approx 56\,\%$ en cada
\omterm{def:g12:oscillators-and-time:oscillator}{periodo}. El \omterm{def:g12:rlc-oscillations:regimes}{seudoperiodo} se queda en $\approx T_0 = 2\pi\sqrt{LC}$: un
\omterm{def:g12:oscillators-and-time:damping}{amortiguamiento} débil encoge la \omterm{def:g12:oscillators-and-time:oscillator}{amplitud}, pero apenas el tempo.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:11}
$T_0 = 2\pi\sqrt{0.50 \times \num{2.0e-5}} = 2\pi \times \num{3.16e-3}
\approx \qty{20}{ms}$. Por el diccionario:
$k = m/(LC) = 0.50/\num{1.0e-5} = \qty{5.0e4}{N/m}$; comprobación,
$2\pi\sqrt{m/k} = 2\pi\sqrt{\num{1.0e-5}}$ --- los mismos \qty{20}{ms}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:12}
$C = \qty{500}{pF}$: $f_0 = 1/(2\pi\sqrt{\num{1.0e-13}}) \approx
\qty{0.50}{MHz}$; $C = \qty{50}{pF}$: $f_0 \approx \qty{1.6}{MHz}$. Con
$L$ fija, $f_0 = (2\pi\sqrt{L})^{-1} C^{-1/2} \propto 1/\sqrt{C}$:
multiplicar $C$ por diez solo divide $f_0$ entre
$\sqrt{10} \approx 3.2$ --- tal como sale.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:13}
$0.95^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.95\rvert \approx 13.5$, o
sea, unos $14$ \omterm{def:g12:oscillators-and-time:oscillator}{periodos}. La \omterm{def:g12:oscillators-and-time:oscillator}{amplitud} es proporcional a $\sqrt{E}$: baja
a $1/\sqrt{2} \approx 0.71$ de la inicial. A \qty{1.0}{MHz},
$T_0 = \qty{1.0}{\micro s}$: unos \qty{14}{\micro s}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:14}
Pérdida por ciclo, $0.020 \times \qty{1.0e-9}{J} = \qty{2.0e-11}{J}$, a
\num{32768} ciclos por segundo: $P = \num{32768} \times \num{2.0e-11}
\approx \qty{6.6e-7}{W}$ --- menos de un microvatio, años con una pila
de botón. Al paso, porque solo un empujón en la \omterm{def:g12:oscillators-and-time:phase}{fase} adecuada
\emph{añade} \omterm{def:g10:energy-conservation:energy}{energía} (\omterm{def:g12:oscillators-and-time:resonance}{resonancia}); a destiempo, frenaría la oscilación.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:15}
$C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
(\num{2.5e7})^2 \times \num{1.0e-6}} \approx \qty{41}{pF}$. El resistor
debería llevar la malla al \omterm{def:g12:rlc-oscillations:regimes}{régimen aperiódico}: la carga se asienta
entonces sin pasarse --- ni oscilación ni repique.
\end{solution}

\begin{solution}{pb:g12:rlc-oscillations:1}
\textbf{1.} Ley de mallas: $u_L + u_C = 0$, con $u_C = q/C$,
$u_L = L\,di/dt$ e $i = dq/dt$: $L\,d^2q/dt^2 + q/C = 0$.

\textbf{2.} $d^2q/dt^2 = -(2\pi/T_0)^2 q$, así que la ecuación queda
$q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$, cierto para toda $t$ exactamente
cuando $T_0 = 2\pi\sqrt{LC}$.

\textbf{3.} $LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times
\unit{C/A} = \unit{s^2}$ --- $\sqrt{LC}$ es un tiempo.

\textbf{4.} $LC = \qty{6.6e-14}{s^2}$: $T_0 = 2\pi \times \num{2.57e-7}
\approx \qty{1.6}{\micro s}$, $f_0 \approx \qty{620}{kHz}$ --- dentro de
\qtyrange{530}{1700}{kHz}.

\textbf{5.} $C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
\num{1.0e12} \times \num{2.0e-4}} \approx \qty{127}{pF}$.

\textbf{6.} La misma fórmula a \qty{530}{kHz}: $C \approx \qty{451}{pF}$.

\textbf{7.} A \qty{1700}{kHz}: $C \approx \qty{44}{pF}$.

\textbf{8.} $C = \qty{480}{pF}$: $f_0 \approx \qty{514}{kHz}$;
$C = \qty{20}{pF}$: $f_0 \approx \qty{2.5}{MHz}$ --- la banda entera,
con margen por los dos extremos.

\textbf{9.} $f_0 = (2\pi\sqrt{L})^{-1}C^{-1/2} \propto 1/\sqrt{C}$;
duplicar $f_0$ exige dividir $C$ entre $4$.

\textbf{10.} Como $f_0 \propto 1/\sqrt{C}$, pasos iguales de $C$ mueven
$f_0$ despacio cuando $C$ es grande y deprisa cuando es pequeña: una
separación dada entre emisoras ocupa entonces poquísimo recorrido de
dial --- se apiñan en el extremo de \omterm{def:g12:oscillators-and-time:oscillator}{frecuencias} altas (el de $C$
pequeña).

\textbf{11.} $Q_0 = CU_0 = \num{1.27e-10} \times 0.020 \approx
\qty{2.5e-12}{C}$; $E = \tfrac12 CU_0^2 \approx \qty{2.5e-14}{J}$.

\textbf{12.} $\tfrac12 LI_0^2 = E$: $I_0 = U_0\sqrt{C/L} =
0.020\sqrt{\num{6.35e-7}} \approx \qty{16}{\micro A}$.

\textbf{13.} $0.96^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.96\rvert
\approx 17$ ciclos.

\textbf{14.} A \qty{1.00}{MHz}, $T_0 = \qty{1.0}{\micro s}$: el repique
se reduce a la mitad en unos \qty{17}{\micro s}. Un repique solitario
muere en microsegundos; la música necesita que la malla se alimente sin
parar.

\textbf{15.} La \omterm{def:g10:signals-and-waves:wave}{onda} empuja millones de veces por segundo; solo cuando
su \omterm{def:g12:oscillators-and-time:oscillator}{frecuencia} coincide con $f_0$ llegan los empujones al paso y
amplifican la oscilación --- la \omterm{def:g12:oscillators-and-time:resonance}{resonancia}; las emisoras desintonizadas
empujan tantas veces en contra como a favor. En el emisor, un
amplificador devuelve cada ciclo la \omterm{def:g10:energy-conservation:energy}{energía} disipada: un \omterm{def:g12:oscillators-and-time:oscillator}{oscilador}
mantenido.

\textbf{16.} $LC = \qty{2.54e-14}{s^2}$: $k = m/(LC) =
0.20/\num{2.54e-14} \approx \qty{7.9e12}{N/m}$.

\textbf{17.} Unas $10^8$ veces el muelle del coche: ningún sistema
masa--resorte de sobremesa llega a los megahercios --- las rigideces y
las masas cotidianas son irremediablemente lentas.

\textbf{18.} $m = \dfrac{k}{(2\pi f_0)^2} = \dfrac{100}
{(\num{6.28e6})^2} \approx \qty{2.5e-12}{kg}$ --- unos pocos
nanogramos, una mota de polvo.

\textbf{19.} Un cristal de cuarzo: una lámina ligerísima y rígida hasta
el extremo que canta a su \omterm{def:g12:rlc-oscillations:period}{frecuencia propia} --- el guardián del tiempo
que ya nos encontramos este curso.

\textbf{20.} Con $L = \qty{0.20}{mH}$ y \qtyrange{20}{480}{pF}, la malla
barre alrededor de \qtyrange{0.51}{2.5}{MHz} y cubre la banda;
$C \approx \qty{127}{pF}$ elige \qty{1.00}{MHz}; abandonado a sí mismo,
el repique se reduce a la mitad en $\approx \qty{17}{\micro s}$, pero la
propia \omterm{def:g10:signals-and-waves:wave}{onda} de la emisora, resonando al paso, realimenta la malla --- el
informativo suena con la \omterm{def:g10:energy-conservation:energy}{energía} de la propia \omterm{def:g10:signals-and-waves:wave}{onda}.
\end{solution}
