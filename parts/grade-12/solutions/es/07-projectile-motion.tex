\begin{solution}{exo:g12:projectile-motion:2}

apenas nota el aire.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:15}
En $t^* = D/(v_0\cos\alpha)$ el dardo está a la altura
$v_0\sin\alpha\, t^* - \tfrac12 g t^{*2} = D\tan\alpha - \tfrac12 g
t^{*2} = H - \tfrac12 g t^{*2}$, exactamente la del mono que se ha
soltado, $y = H - \tfrac12 g t^{*2}$. Los dos cuelgan $\tfrac12 gt^2$
por debajo de su posición sin gravedad --- se encuentran sea cual sea
$v_0$.
\end{solution}

\begin{solution}{pb:g12:projectile-motion:1}
\textbf{1.} Solo el \omterm{def:g10:universal-gravitation:weight}{peso} (el aire se desprecia); por la \omterm{thm:g12:newtons-laws:second}{segunda ley de
Newton}, $m\vect a = m\vect g$, luego $\vect a = (0,\,-g)$.

\textbf{2.} $v_x = v_0\cos\alpha$, $v_y = v_0\sin\alpha - gt$.

\textbf{3.} $x = v_0\cos\alpha\, t$,
$y = v_0\sin\alpha\, t - \tfrac12 gt^2$.

\textbf{4.} $t = x/(v_0\cos\alpha)$:
$y = x\tan\alpha - g x^2/(2v_0^2\cos^2\alpha)$.

\textbf{5.} $v_0^2 = \dfrac{g d^2}{2\cos^2\alpha\,(d\tan\alpha -
1.05)} = \dfrac{173.0}{0.826 \times 3.96} = 52.9$:
$v_0 = \qty{7.28}{m/s} \approx \qty{26}{km/h}$.

\textbf{6.} $t = d/(v_0\cos\alpha) = 4.20/4.68 = \qty{0.90}{s}$.

\textbf{7.} Cima en $t = v_0\sin\alpha/g = \qty{0.57}{s}$, a la altura
$2.00 + (v_0\sin\alpha)^2/2g = 2.00 + 1.58 = \qty{3.58}{m}$;
$0.57 < 0.90$: en el aro va descendiendo.

\textbf{8.} $y(0.90) = 0.90\tan\ang{50} - 0.182 = \qty{0.89}{m}$ por
encima del punto de suelta, es decir, \qty{2.89}{m} sobre el suelo:
\qty{0.29}{m} por encima de los dedos.

\textbf{9.} $v_x = \qty{4.68}{m/s}$;
$v_y = 5.57 - 9.81 \times 0.90 = \qty{-3.24}{m/s}$;
$v = \qty{5.69}{m/s}$, a $\arctan(3.24/4.68) = \ang{34.7}$ por debajo de
la horizontal.

\textbf{10.} El aro es un anillo horizontal: el balón tiene que
atravesar su plano desde arriba. Cuanto más empinada es la entrada, más
grande se ve la abertura del anillo a lo largo de la velocidad --- más
margen de error.

\textbf{11.} $1.05/26.0 = 4\,\%$: despreciable. $y(t_f) = 0$ da
$t_f\,(v_0\sin\alpha - \tfrac12 g t_f) = 0$, luego
$t_f = 2v_0\sin\alpha/g$.

\textbf{12.} $R = v_0\cos\alpha\, t_f = v_0^2\sin(2\alpha)/g$; máximo en
$\alpha = \ang{45}$.

\textbf{13.} $v_0 = \sqrt{gR} = \sqrt{9.81 \times 26.0} =
\qty{16.0}{m/s} \approx \qty{57}{km/h}$.

\textbf{14.} $R_{\max} = 17.5^2/9.81 = \qty{31.2}{m} > \qty{26.0}{m}$:
posible, con \qty{5}{m} de \omterm{def:g12:projectile-motion:range}{alcance} de sobra.

\textbf{15.} $\sin(2\alpha) = 255.1/306.3 = 0.833$: $\alpha =
\ang{28.2}$ o \ang{61.8}; \omterm{def:g12:projectile-motion:range}{tiempos de vuelo} $2v_0\sin\alpha/g =
\qty{1.69}{s}$ y \qty{3.14}{s}. Solo el tiro tenso se adelanta a la
bocina de \qty{2.0}{s}.

\textbf{16.} $v_0^2 = \dfrac{9.81 \times 36}{2\cos^2\ang{75}\,
(6.0\tan\ang{75} - 1.05)} = \dfrac{353.2}{2.86} = 123.5$:
$v_0 = \qty{11.1}{m/s}$.

\textbf{17.} $y(5.55) = 5.55\tan\ang{75} - 18.26 = \qty{2.45}{m}$ por
encima del punto de suelta, es decir, \qty{4.45}{m} sobre el suelo:
\qty{0.50}{m} por encima del borde superior del tablero.

\textbf{18.} Cima: $2.00 + (11.1\sin\ang{75})^2/2g \approx
\qty{7.9}{m}$; vuelo: $6.0/(11.1\cos\ang{75}) = \qty{2.1}{s}$ --- dos
segundos enteros con el balón colgado a \qty{8}{m}.

\textbf{19.} El \omterm{def:g10:inertia:inventory}{rozamiento} del aire acorta todos los tiros y empina el
descenso; el efecto crece con la rapidez, así que el lanzamiento de
campo entero a \qty{17.5}{m/s} es el que más sufre --- su \omterm{def:g12:projectile-motion:range}{alcance} real
se queda unos \omterm{def:g10:orders-of-magnitude:unit}{metros} por debajo de \qty{31}{m} y se come parte del
margen.

\textbf{20.} \emph{El veredicto sobre la canasta desde el propio \omterm{def:g11:electric-gravitational-fields:field}{campo}}:
el tiro de \qty{26.0}{m} exige \qty{16.0}{m/s} y un brazo proporciona
unos \qty{17.5}{m/s}, un margen de \omterm{def:g12:projectile-motion:range}{alcance} de \qty{5}{m}
($\approx 20\,\%$) que la resistencia del aire recorta pero no borra.
Posible --- y por eso, unas cuantas veces por temporada, entra.
\end{solution}
