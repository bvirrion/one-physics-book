\section*{Capítulo \ref{ch:g12:kinematics-2d} --- Cinemática en dos dimensiones}

\begin{solution}{exo:g12:kinematics-2d:1}
$\vect v = (3.0, 4.0)$, constante; \omterm{def:g12:kinematics-2d:velocity}{rapidez}
$\sqrt{3.0^2 + 4.0^2} = \qty{5.0}{m/s}$. \omterm{def:g10:relative-motion:trajectory}{Trayectoria}: la recta
$y = \tfrac43 x$. \omterm{def:g12:kinematics-2d:velocity}{Vector velocidad} constante: \omterm{def:g12:kinematics-2d:uniform}{movimiento rectilíneo
uniforme}.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:2}
Línea recta con separaciones iguales: \omterm{def:g12:kinematics-2d:uniform}{rectilíneo uniforme};
$v = 0.35/0.10 = \qty{3.5}{m/s}$. Unas separaciones que se abrieran
significarían que la \omterm{def:g12:kinematics-2d:velocity}{rapidez} está aumentando.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:3}
$v(t) = 4.0\,t$, $a = \qty{4.0}{m/s^2}$: aceleración constante,
rectilíneo uniformemente acelerado. $v(3.0) = \qty{12}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:4}
$f = 1/8.0 = \qty{0.125}{Hz}$;
$v = 2\pi \times 4.5/8.0 \approx \qty{3.5}{m/s}$;
$a = v^2/R = 3.53^2/4.5 \approx \qty{2.8}{m/s^2}$, apuntando al eje del
tiovivo.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:5}
Pendiente: $a = (10.0 - 2.0)/4.0 = \qty{2.0}{m/s^2}$. Área (un
trapecio): $d = \tfrac12 (2.0 + 10.0) \times 4.0 = \qty{24}{m}$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:6}
$v = \qty{88.9}{m/s}$:
$a_n = 88.9^2/6000 \approx \qty{1.3}{m/s^2} \approx 0.13\,g$. Como
$a_n = v^2/R$, una $v$ alta exige una $R$ enorme para mantener aceptable
la aceleración lateral (y la \omterm{def:g10:inertia:force}{fuerza} sobre la vía y los viajeros) --- el
peralte cambia quién la proporciona, no su tamaño.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:7}
$v(t) = -2.4\,t + 8.0$; $a = \qty{-2.4}{m/s^2}$, constante. Da media
vuelta cuando $v = 0$: $t = 8.0/2.4 \approx \qty{3.3}{s}$, en
$x \approx \qty{16.3}{m}$. Antes: hacia adelante, frenando; después:
hacia atrás, acelerando --- uniformemente acelerado en todo momento.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:8}
$v_1 = 0.28/0.40 = \qty{0.70}{m/s}$;
$v_2 = (0.54 - 0.10)/0.40 = \qty{1.10}{m/s}$;
$v_3 = (0.88 - 0.28)/0.40 = \qty{1.50}{m/s}$. La velocidad gana
\qty{0.40}{m/s} cada \qty{0.20}{s}: $a = \qty{2.0}{m/s^2}$, constante
--- rectilíneo uniformemente acelerado.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:9}
$T = \qty{2.36e6}{s}$:
$v = 2\pi \times \num{3.84e8}/\num{2.36e6} \approx \qty{1.0e3}{m/s}$;
$a_n = v^2/R \approx \qty{2.7e-3}{m/s^2}$. Cociente:
$g/a_n \approx 3600 = 60^2$ --- la gravedad, diluida como la inversa del
cuadrado de la distancia (\cref{ch:g10:universal-gravitation}), da
cuenta exactamente del giro de la Luna.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:10}
$f = 1200/60 = \qty{20}{Hz}$, $T = \qty{0.050}{s}$;
$v = 2\pi R f = 2\pi \times 0.25 \times 20 \approx \qty{31}{m/s}$;
$a_n = v^2/R \approx \qty{3.9e3}{m/s^2} \approx 400\,g$. La pared del
tambor le proporciona a la ropa su $a_n$; a través de los agujeros nada
empuja el agua hacia dentro, así que sale despedida por la tangente.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:11}
$v_0 = \qty{25}{m/s}$: $t_s = 25/7.5 \approx \qty{3.3}{s}$;
$d = v_0^2/(2a) = 625/15 \approx \qty{42}{m}$. Gráfica: un segmento
recto que baja de \qty{25}{m/s} hasta $0$ en $t_s$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:12}
$\vect v = (-R\omega\sin\omega t,\, R\omega\cos\omega t)$, de norma
constante $R\omega$; su producto escalar con $\vect{OM}$ se anula, luego
es perpendicular al radio y, por tanto, tangente. Derivando otra vez,
$\vect a = -\omega^2\vect{OM}$: hacia $O$ y de norma
$\omega^2 R = (v/R)^2 R = v^2/R$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:13}
$a_t = \qty{2.5}{m/s^2}$ (hacia atrás),
$a_n = 20^2/80 = \qty{5.0}{m/s^2}$;
$a = \sqrt{2.5^2 + 5.0^2} \approx \qty{5.6}{m/s^2}$; ángulo con la
dirección del movimiento:
$\theta = 180^\circ - \arctan(5.0/2.5) \approx 117^\circ$ (es decir,
$63^\circ$ respecto de la dirección puramente hacia atrás).
$5.6 < 7.0$: dentro del agarre, con poco margen.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:14}
Arranque: $t_1 = 20/1.2 \approx \qty{17}{s}$,
$d_1 = 20^2/(2 \times 1.2) \approx \qty{167}{m}$. Marcha:
\qty{40}{s}, $\qty{800}{m}$. Frenada:
$t_3 = 20/1.5 \approx \qty{13}{s}$,
$d_3 = 400/3.0 \approx \qty{133}{m}$. Total: $\qty{1.1e3}{m}$ en
\qty{70}{s}; media $1100/70 \approx \qty{16}{m/s}$ (\qty{57}{km/h}).
$v(t)$ es un trapecio cuya área,
$\tfrac12(70 + 40) \times 20 = \qty{1100}{m}$, comprueba el total.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:15}
$v = 2\pi \times \num{4.22e7}/\num{86164} \approx \qty{3.1e3}{m/s}$;
$a_n = v^2/R \approx \qty{0.22}{m/s^2}$. Se la proporciona la gravedad
terrestre a esa distancia; el \omterm{def:g12:kinematics-2d:ucm}{periodo} es igual al \omterm{def:g12:kinematics-2d:ucm}{periodo} de rotación de
la Tierra, así que el \omterm{def:g10:universal-gravitation:satellite}{satélite} y el suelo giran a la vez y queda colgado
sobre un punto del ecuador (\cref{ch:g12:satellites-kepler}).
\end{solution}

\begin{solution}{pb:g12:kinematics-2d:1}
\textbf{1.} El movimiento es relativo
(\cref{ch:g10:relative-motion}): toda posición y toda velocidad tienen
que nombrar su sistema. Aquí: el \omterm{def:g10:relative-motion:frames}{sistema terrestre} (la calzada), con $x$
a lo largo de la calle desde la posición del primer fotograma.

\textbf{2.} En cada intervalo: $9.0/0.50 = \qty{18.0}{m/s}$ --- cuatro
valores iguales.

\textbf{3.} Calle recta y desplazamientos iguales en tiempos iguales:
\omterm{def:g12:kinematics-2d:uniform}{movimiento rectilíneo uniforme}.

\textbf{4.} $v_0 = \qty{18.0}{m/s} = \qty{64.8}{km/h}$.

\textbf{5.} $\vect a = \vect 0$.

\textbf{6.} $v(t) = -a\,t + v_B$; $x(t) = -\tfrac12 a t^2 + v_B\,t$.

\textbf{7.} De $v(t_s) = 0$ sale $t_s = v_B/a$; entonces
$d_b = -\tfrac12 a (v_B/a)^2 + v_B^2/a = v_B^2/(2a)$.

\textbf{8.} $v_B = \sqrt{2 a d_b} = \sqrt{2 \times 6.0 \times 27} =
\sqrt{324} = \qty{18.0}{m/s}$ --- exactamente la \omterm{def:g12:kinematics-2d:velocity}{rapidez} filmada: el
conductor no aminoró antes de frenar.

\textbf{9.} Reacción: $18.0 \times 1.0 = \qty{18}{m}$; en total,
$18 + 27 = \qty{45}{m}$.

\textbf{10.} $d_b = v_B^2/(2a) \propto v_B^2$: duplicar $v_B$
multiplica $d_b$ por $4$. A \qty{50}{km/h} ($\qty{13.9}{m/s}$):
$d_b = 13.9^2/12.0 \approx \qty{16}{m}$.

\textbf{11.} La \omterm{def:g12:kinematics-2d:velocity}{rapidez} es constante, pero la dirección de la velocidad
gira: $\vect v$ no es constante, así que el coche está acelerando.

\textbf{12.} Cuarto de vuelta: $2\pi R/4 = \qty{19.6}{m}$;
$v = 19.6/2.5 \approx \qty{7.9}{m/s}$ (\qty{28}{km/h}).

\textbf{13.} $a_n = 7.85^2/12.5 \approx \qty{4.9}{m/s^2}$, hacia el
centro de la rotonda.

\textbf{14.} $a_{\max} = \mu g = 0.70 \times 9.81 \approx
\qty{6.9}{m/s^2}$;
$v_{\max} = \sqrt{\mu g R} = \sqrt{6.87 \times 12.5} \approx
\qty{9.3}{m/s}$ (\qty{33}{km/h}). $7.9 < 9.3$: dentro del agarre ---
coherente con un carril limpio, sin marcas de derrape.

\textbf{15.} $T = 2\pi R/v = 78.5/7.85 \approx \qty{10.0}{s};
f = \qty{0.10}{Hz}$.

\textbf{16.} $64.8 - 50 \approx \qty{15}{km/h}$ de más --- alrededor de
un $30\,\%$ por encima del límite.

\textbf{17.} La frenada actúa a lo largo de $41 - 18 = \qty{23}{m}$:
$v^2 = v_B^2 - 2ad = 324 - 2 \times 6.0 \times 23 = 48$, luego
$v \approx \qty{6.9}{m/s} \approx \qty{25}{km/h}$ en el impacto.

\textbf{18.} A \qty{13.9}{m/s}:
$13.9 \times 1.0 + 13.9^2/12.0 = 13.9 + 16.1 = \qty{30.0}{m} <
\qty{41}{m}$ --- el coche se para \qty{11}{m} antes de la furgoneta.

\textbf{19.} $d(v) = v\,t_r + v^2/(2a)$: una vez linealmente (la
reacción recorre $v t_r$ antes de que actúen los frenos) y otra al
cuadrado (la parte cinética $v^2/(2a)$) --- y por eso los excesos
moderados salen desproporcionadamente caros.

\textbf{20.} La cámara de salpicadero y las marcas de derrape coinciden:
\qty{65}{km/h} en una zona de \qty{50}{km/h}; al límite legal, el coche
se habría detenido \qty{11}{m} antes de la furgoneta --- el accidente
son los \qty{15}{km/h} de más.
\end{solution}
