\begin{solution}{exo:g12:quantum-world:1}

\qty{2.35}{eV}$. $N = \num{1.0e-3}/\num{3.75e-19} \approx \num{2.7e15}$

\end{solution}

\begin{solution}{exo:g12:quantum-world:2}

\end{solution}

\begin{solution}{exo:g12:quantum-world:3}

\end{solution}

\begin{solution}{exo:g12:quantum-world:7}
Intensiteit verdubbeld: de stroom verdubbelt (twee keer zoveel \omterm{def:g12:quantum-world:photon}{fotonen}), de
\omterm{def:g12:quantum-world:stopping}{remspanning} blijft gelijk (dezelfde \omterm{def:g10:energy-conservation:energy}{energie} per \omterm{def:g12:quantum-world:photon}{foton}). \omterm{def:g12:oscillators-and-time:oscillator}{Frequentie} verhoogd:
de \omterm{def:g12:quantum-world:stopping}{remspanning} stijgt, want $E_{k,\max} = h\nu - W_0$. Het golfbeeld koppelt
de \omterm{def:g10:energy-conservation:energy}{energie} van de elektronen aan de intensiteit en niet aan de \omterm{def:g12:oscillators-and-time:oscillator}{frequentie} ---
precies wat \emph{niet} wordt waargenomen.
\end{solution}

\begin{solution}{exo:g12:quantum-world:8}

ver in het \omterm{def:g10:light-spectra:wavelength}{ultraviolet}. Vanuit $n = 2$: \qty{3.40}{eV},
$\lambda \approx \qty{365}{nm}$, nabij \omterm{def:g10:light-spectra:wavelength}{ultraviolet}.
\end{solution}

\begin{solution}{exo:g12:quantum-world:9}
Vanuit de \omterm{def:g12:quantum-world:levels}{grondtoestand} kost de kleinste sprong $E_2 - E_1 = \qty{10.2}{eV}$
(\qty{122}{nm}): elke absorptielijn is \omterm{def:g10:light-spectra:wavelength}{ultraviolet}, geen enkele zichtbaar. In
een hete steratmosfeer houden botsingen sommige atomen in $n = 2$, van waaruit
zichtbare balmerfotonen (\qtyrange{1.89}{3.40}{eV}) kunnen worden opgenomen.
\end{solution}

\begin{solution}{exo:g12:quantum-world:10}
(a) $E_k = \qty{1.6e-17}{J}$, $p = \sqrt{2m_e E_k} = \qty{5.4e-24}{kg.m/s}$,
$\lambda = h/p \approx \qty{1.2e-10}{m}$ --- zo groot als een \omterm{def:g11:fundamental-interactions:ladder}{atoom}: kristallen
buigen hem af. (b) $p = \num{1.0e-4} \times 1.0 = \qty{1.0e-4}{kg.m/s}$,
$\lambda \approx \qty{6.6e-30}{m}$ --- $10^{20}$ keer kleiner dan een \omterm{def:g11:fundamental-interactions:ladder}{atoom}:
de \omterm{def:g10:signals-and-waves:wave}{golf} van de vlieg blijft voor eeuwig onzichtbaar.
\end{solution}

\begin{solution}{exo:g12:quantum-world:11}
Eén \omterm{def:g12:quantum-world:photon}{foton}: $\num{1.99e-25}/\num{5.05e-7} = \qty{3.94e-19}{J}$; negentig:
$\approx \qty{3.5e-17}{J}$. De mug: $\tfrac12 \times \num{2.5e-6}
\times 0.50^2 = \qty{3.1e-7}{J}$ --- $10^{10}$ keer meer. Het oog werkt tot op
honderd korrels van de absolute grens: bijna een fotonenteller.
\end{solution}

\begin{solution}{exo:g12:quantum-world:12}
$\nu_1 = \qty{7.41e14}{Hz}$, $\nu_2 = \qty{5.49e14}{Hz}$;
$h = e\,\Delta U_s/\Delta\nu = \num{1.60e-19} \times 0.79 /
\num{1.91e14} \approx \qty{6.6e-34}{J.s}$. $W_0 = h\nu_2 - eU_{s,2} =
\num{3.63e-19} - \num{0.64e-19} = \qty{3.0e-19}{J} \approx \qty{1.9}{eV}$:
cesium.
\end{solution}

\begin{solution}{exo:g12:quantum-world:13}

\begin{solution}{exo:g12:quantum-world:14}
Doorsnede $\pi r^2 \approx \qty{3.1e-20}{m^2}$, opgevangen \omterm{def:g11:work-of-force:power}{vermogen}
$\qty{3.1e-26}{W}$; $t = \num{3.7e-19}/\num{3.1e-26} \approx \qty{1.2e7}{s}$

kan niet over het \omterm{def:g12:mechanical-waves:wavefront}{golffront} zijn uitgesmeerd: ze reist geconcentreerd, in
korrels.
\end{solution}

\begin{solution}{exo:g12:quantum-world:15}

\begin{solution}{pb:g12:quantum-world:1}
\textbf{1.} $E = hc/\lambda = \num{1.99e-25}/\num{5.50e-7} =
\qty{3.62e-19}{J} = \qty{2.26}{eV}$.

\textbf{2.} $N = \num{1.0e3}/\num{3.62e-19} \approx \num{2.8e21}$ \omterm{def:g12:quantum-world:photon}{fotonen} per
seconde.

\textbf{3.} $P = 0.20 \times \qty{1.0e3}{W} = \qty{200}{W}$;
$E = 200 \times 5.0 = \qty{1.0}{kW.h}$ ($\qty{3.6e6}{J}$).

\textbf{4.} Met $10^{21}$ korrels per seconde is het statistische flakkeren
fantastisch klein: de korreligheid middelt uit tot een volmaakt gladde
stroom, zoals snel gegoten zand als water stroomt.

\textbf{5.} Schade vraagt één korrel van \qtyrange{3}{4}{eV}: ultraviolette
korrels voldoen, infrarode ($\sim \qty{0.1}{eV}$) nooit --- aantallen kunnen
de klap per korrel niet vervangen.

\textbf{6.} $U_s$ is de tegengestelde \omterm{def:g11:circuits-and-power:voltage}{spanning} die de snelste elektronen net
tegenhoudt: $eU_s = E_{k,\max} = h\nu - W_0$.

\textbf{7.} $\nu = c/\lambda$: $8.22$, $7.41$, $6.88$ en $5.49$
$\times\,10^{14}\,\unit{Hz}$.

\textbf{8.} $eU_s = h\nu - W_0$ is affien in $\nu$, met helling $h$.
Opeenvolgende hellingen: $(0.94 - 0.37)/1.39 = 0.41$ en
$(1.50 - 0.94)/1.34 = 0.42$ \unit{eV} per $10^{14}\,\unit{Hz}$ --- de punten
liggen op één lijn.

\textbf{9.} $h = e(U_{s,1}-U_{s,4})/(\nu_1-\nu_4) = \num{1.60e-19}
\times 1.13/\num{2.72e14} = \qty{6.64e-34}{J.s}$.

\textbf{10.} Bij \qty{546}{nm}: $W_0 = h\nu - eU_s = \num{3.64e-19} -
\num{0.59e-19} = \qty{3.05e-19}{J} = \qty{1.90}{eV}$;
$\nu_0 = W_0/h = \qty{4.60e14}{Hz}$; $\lambda_0 \approx \qty{653}{nm}$.

\textbf{11.} Alleen $\lambda < \qty{653}{nm}$ slaat elektronen los: het
zichtbare van violet tot rood, en het \omterm{def:g10:light-spectra:wavelength}{ultraviolet}. De hele infrarode kant ---
ruwweg de helft van de \omterm{def:g10:energy-conservation:energy}{energie} van de Zon --- is nutteloos voor deze kathode.

\textbf{12.} $(6.64 - 6.63)/6.63 \approx 0.2\%$ --- een meting van een
universele constante op twee promille, in een schoollab.

\textbf{13.} $E_2 = \qty{-3.40}{eV}$, $E_3 = \qty{-1.51}{eV}$,
$E_4 = \qty{-0.85}{eV}$, $E_5 = \qty{-0.54}{eV}$.

\textbf{14.} $3 \to 2$: \qty{1.89}{eV}; $4 \to 2$: \qty{2.55}{eV};
$5 \to 2$: \qty{2.86}{eV}.

\textbf{15.} $\lambda = 1240/E$: \qty{656}{nm} (rood), \qty{486}{nm}
(blauwgroen), \qty{434}{nm} (violet).

\textbf{16.} De energieën van het \omterm{def:g11:fundamental-interactions:ladder}{atoom} zijn \omterm{def:g12:quantum-world:levels}{gekwantiseerd}, dus kan het alleen
de discrete verschillen $E_p - E_n$ uitzenden: een ladder maakt lijnen, geen
regenboog.

\textbf{17.} Nee: \qty{500}{nm} betekent \qty{2.48}{eV}, en dat komt met geen
enkel niveauverschil overeen (vanaf $E_2$ omhoog geeft $1.89$, $2.55$, $2.86$
en \qty{3.40}{eV}; vanaf $E_1$ kost alles minstens \qty{10.2}{eV}) --- het gas
is daar doorzichtig.

\textbf{18.} $eU = \tfrac12 m_e v^2$: $v = \sqrt{2 \times
\num{1.60e-19} \times \num{1.0e4}/\num{9.11e-31}} \approx
\qty{5.9e7}{m/s} \approx 0.20\,c$ --- een klassieke behandeling is
aanvaardbaar.

\textbf{19.} $p = m_e v = \qty{5.4e-23}{kg.m/s}$;
$\lambda = h/p = \qty{1.2e-11}{m} = \qty{12}{pm}$.

\textbf{20.} Winst $= \num{5.50e-7}/\num{1.23e-11} \approx \num{4.5e4}$. De
twee getallen van het weekend: $h = \qty{6.64e-34}{J.s}$, met \omterm{def:g11:color-light-sources:subtractive}{filters} en een
voltmeter tot op $0.2\%$ gemeten, en een microscoop die vijfenveertigduizend
keer scherper is omdat elektronen golven.
\end{solution}
