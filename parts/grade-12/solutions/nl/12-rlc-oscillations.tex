\section*{Hoofdstuk \ref{ch:g12:rlc-oscillations} --- Vrije elektrische trillingen: RLC}

\begin{solution}{exo:g12:rlc-oscillations:1}
(a) $\sqrt{LC} = \qty{1.0e-4}{s}$: $T_0 = \qty{0.63}{ms}$,
$f_0 \approx \qty{1.6}{kHz}$. (b) $\sqrt{LC} = \qty{3.16e-7}{s}$:
$T_0 = \qty{2.0}{\micro s}$, $f_0 \approx \qty{0.50}{MHz}$ --- een factor
$\sqrt{10^5} \approx 316$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:2}
$LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times \unit{C/A} =
\unit{s} \times \unit{s} = \unit{s^2}$, dus staat $\sqrt{LC}$ in seconden.
$2\pi$ is een zuiver getal --- geen dimensie om iets aan te veranderen.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:3}
$Q_0 = CU_0 = \qty{2.8}{\micro C}$; $E = \tfrac12 CU_0^2 =
\tfrac12 \times \num{4.7e-7} \times 36 \approx \qty{8.5}{\micro J}$. De lading
trilt met $f_0 = 1/(2\pi\sqrt{LC})$, de stroom een kwart \omterm{def:g12:oscillators-and-time:oscillator}{periode} uit \omterm{def:g12:oscillators-and-time:phase}{fase}, en
de \omterm{def:g10:energy-conservation:energy}{energie} klotst tussen platen en \omterm{def:g12:rc-rl-circuits:inductor}{spoel}, met een constant totaal.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:4}
$Q_0 = \qty{2.0}{\micro C}$; $f_0 = 1/T_0 = \qty{250}{Hz}$;
$q(T_0/2) = \qty{-2.0}{\micro C}$; $i(0) = 0$ (de cosinus is vlak in $t=0$);
$I_0 = 2\pi \times \num{2.0e-6}/\num{4.0e-3} \approx \qty{3.1}{mA}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:5}
Alles in de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} wanneer $q = \pm Q_0$ en $i = 0$ ($t = 0$, $T_0/2$,
$T_0$, \dots): de uitersten van de slinger. Alles in de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} wanneer $q = 0$
en $i = \pm I_0$ ($t = T_0/4$, $3T_0/4$, \dots): het laagste en snelste punt
van de slinger.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:6}
$d^2q/dt^2 = -(2\pi/T_0)^2\,Q_0\cos(2\pi t/T_0) = -(2\pi/T_0)^2 q$, dus is
$L\,d^2q/dt^2 + q/C = q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$ voor alle $t$ precies
wanneer $(2\pi/T_0)^2 = 1/(LC)$, oftewel $T_0 = 2\pi\sqrt{LC}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:7}
$L = \dfrac{1}{4\pi^2 f_0^2 C} = \dfrac{1}{39.5 \times
(\num{1.98e5})^2 \times \num{2.2e-10}} \approx \qty{2.9}{mH}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:8}
$E = \tfrac12 CU_0^2 = \qty{5.0e-5}{J}$; uit $\tfrac12 LI_0^2 = E$ volgt
$I_0 = \sqrt{2E/L} = \sqrt{\num{1.0e-2}} = \qty{0.10}{A}$, voor het eerst
bereikt op $T_0/4 = \tfrac14 \times \qty{0.63}{ms} \approx \qty{0.16}{ms}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:9}
$E_C \propto \cos^2(2\pi t/T_0)$, en $\cos^2$ herhaalt zich elke \emph{halve}
omwenteling: twee vullingen van de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} per \omterm{def:g12:oscillators-and-time:oscillator}{periode}, dus $2f_0$. Op
$t = T_0/8$ is de \omterm{def:g12:oscillators-and-time:phase}{fase} $\pi/4$: $\cos^2(\pi/4) = \tfrac12$, dus
$E_C = E_L = E/2$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:10}
Verhouding $3.0/4.0 = 0.75$ per \omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode}: het zesde maximum is
$4.0 \times 0.75^6 \approx \qty{0.71}{V}$. De \omterm{def:g10:energy-conservation:energy}{energie} is evenredig met het
kwadraat van de \omterm{def:g12:oscillators-and-time:oscillator}{amplitude}: $0.75^2 \approx 56\%$ overleeft elke \omterm{def:g12:oscillators-and-time:oscillator}{periode}. De
\omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode} blijft $\approx T_0 = 2\pi\sqrt{LC}$: zwakke \omterm{def:g12:oscillators-and-time:damping}{demping} krimpt de
\omterm{def:g12:oscillators-and-time:oscillator}{amplitude}, nauwelijks het tempo.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:11}
$T_0 = 2\pi\sqrt{0.50 \times \num{2.0e-5}} = 2\pi \times \num{3.16e-3}
\approx \qty{20}{ms}$. Woordenboek: $k = m/(LC) = 0.50/\num{1.0e-5} =
\qty{5.0e4}{N/m}$; controle: $2\pi\sqrt{m/k} = 2\pi\sqrt{\num{1.0e-5}}$ ---
dezelfde \qty{20}{ms}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:12}
$C = \qty{500}{pF}$: $f_0 = 1/(2\pi\sqrt{\num{1.0e-13}}) \approx
\qty{0.50}{MHz}$; $C = \qty{50}{pF}$: $f_0 \approx \qty{1.6}{MHz}$. Bij vaste
$L$ is $f_0 = (2\pi\sqrt{L})^{-1} C^{-1/2} \propto 1/\sqrt{C}$: een factor
tien in $C$ geeft maar $\sqrt{10} \approx 3.2$ in $f_0$ --- zoals gevonden.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:13}
$0.95^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.95\rvert \approx 13.5$, dus zo'n
$14$ \omterm{def:g12:oscillators-and-time:oscillator}{perioden}. De \omterm{def:g12:oscillators-and-time:oscillator}{amplitude} is evenredig met $\sqrt{E}$: terug tot
$1/\sqrt{2} \approx 0.71$ van de start. Bij \qty{1.0}{MHz} is
$T_0 = \qty{1.0}{\micro s}$: ongeveer \qty{14}{\micro s}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:14}
Verlies per cyclus $0.020 \times \qty{1.0e-9}{J} = \qty{2.0e-11}{J}$, bij
\num{32768} cycli per seconde: $P = \num{32768} \times \num{2.0e-11}
\approx \qty{6.6e-7}{W}$ --- onder een microwatt, jaren op een knoopcel. In de
pas, omdat alleen een duw op de juiste \omterm{def:g12:oscillators-and-time:phase}{fase} \omterm{def:g10:energy-conservation:energy}{energie} \emph{toevoegt}
(\omterm{def:g12:oscillators-and-time:resonance}{resonantie}); uit de pas zou ze de zwaai afremmen.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:15}
$C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
(\num{2.5e7})^2 \times \num{1.0e-6}} \approx \qty{41}{pF}$. De \omterm{def:g11:circuits-and-power:resistance}{weerstand} moet
de kring in het aperiodische regime duwen: de lading zakt dan zonder over te
schieten --- geen trilling, geen rinkeling.
\end{solution}

\begin{solution}{pb:g12:rlc-oscillations:1}
\textbf{1.} Kringwet: $u_L + u_C = 0$, met $u_C = q/C$, $u_L = L\,di/dt$ en
$i = dq/dt$: $L\,d^2q/dt^2 + q/C = 0$.

\textbf{2.} $d^2q/dt^2 = -(2\pi/T_0)^2 q$, dus luidt de vergelijking
$q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$, waar voor alle $t$ precies wanneer
$T_0 = 2\pi\sqrt{LC}$.

\textbf{3.} $LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times
\unit{C/A} = \unit{s^2}$ --- $\sqrt{LC}$ is een tijd.

\textbf{4.} $LC = \qty{6.6e-14}{s^2}$: $T_0 = 2\pi \times \num{2.57e-7}
\approx \qty{1.6}{\micro s}$, $f_0 \approx \qty{620}{kHz}$ --- binnen
\qtyrange{530}{1700}{kHz}.

\textbf{5.} $C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
\num{1.0e12} \times \num{2.0e-4}} \approx \qty{127}{pF}$.

\textbf{6.} Dezelfde formule op \qty{530}{kHz}: $C \approx \qty{451}{pF}$.

\textbf{7.} Op \qty{1700}{kHz}: $C \approx \qty{44}{pF}$.

\textbf{8.} $C = \qty{480}{pF}$: $f_0 \approx \qty{514}{kHz}$;
$C = \qty{20}{pF}$: $f_0 \approx \qty{2.5}{MHz}$ --- de hele band, met marge
aan beide uiteinden.

\textbf{9.} $f_0 = (2\pi\sqrt{L})^{-1}C^{-1/2} \propto 1/\sqrt{C}$; $f_0$
verdubbelen vraagt $C$ door $4$ te delen.

\textbf{10.} Omdat $f_0 \propto 1/\sqrt{C}$, verschuiven gelijke stappen in
$C$ de $f_0$ traag bij grote $C$ en snel bij kleine $C$: een gegeven afstand
tussen zenders kost dan weinig knopverdraaiing --- ze dringen samen aan het
uiteinde met de hoge \omterm{def:g12:oscillators-and-time:oscillator}{frequenties} (lage $C$).

\textbf{11.} $Q_0 = CU_0 = \num{1.27e-10} \times 0.020 \approx
\qty{2.5e-12}{C}$; $E = \tfrac12 CU_0^2 \approx \qty{2.5e-14}{J}$.

\textbf{12.} $\tfrac12 LI_0^2 = E$: $I_0 = U_0\sqrt{C/L} =
0.020\sqrt{\num{6.35e-7}} \approx \qty{16}{\micro A}$.

\textbf{13.} $0.96^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.96\rvert
\approx 17$ cycli.

\textbf{14.} Bij \qty{1.00}{MHz} is $T_0 = \qty{1.0}{\micro s}$: de rinkeling
halveert in ongeveer \qty{17}{\micro s}. Eén losse rinkeling sterft in
microseconden; muziek vraagt een kring die voortdurend wordt gevoed.

\textbf{15.} De \omterm{def:g10:signals-and-waves:wave}{golf} duwt miljoenen keren per seconde; alleen wanneer haar
\omterm{def:g12:oscillators-and-time:oscillator}{frequentie} met $f_0$ samenvalt, komen de duwen in de pas aan en bouwen ze de
zwaai op --- \omterm{def:g12:oscillators-and-time:resonance}{resonantie}; zenders naast de afstemming duwen even vaak tegen als
mee. Bij de zender geeft een versterker de verstookte \omterm{def:g10:energy-conservation:energy}{energie} elke cyclus
terug: een onderhouden \omterm{def:g12:oscillators-and-time:oscillator}{oscillator}.

\textbf{16.} $LC = \qty{2.54e-14}{s^2}$: $k = m/(LC) =
0.20/\num{2.54e-14} \approx \qty{7.9e12}{N/m}$.

\textbf{17.} Ongeveer $10^8$ keer de autoveer: geen enkele opstelling met een
veer en een massa op tafel haalt megahertz --- alledaagse stijfheden en massa's
zijn hopeloos traag.

\textbf{18.} $m = \dfrac{k}{(2\pi f_0)^2} = \dfrac{100}
{(\num{6.28e6})^2} \approx \qty{2.5e-12}{kg}$ --- een paar nanogram, een
stofje.

\textbf{19.} Een kwartskristal: een vederlicht, superstijf plaatje dat op zijn
\omterm{def:g12:rlc-oscillations:period}{eigenfrequentie} zingt --- de tijdmeter van eerder dit jaar.

\textbf{20.} Met $L = \qty{0.20}{mH}$ en \qtyrange{20}{480}{pF} bestrijkt de
kring ongeveer \qtyrange{0.51}{2.5}{MHz} en dus de hele band;
$C \approx \qty{127}{pF}$ pikt \qty{1.00}{MHz} eruit; op zichzelf gelaten
halveert de rinkeling in $\approx \qty{17}{\micro s}$, maar de \omterm{def:g10:signals-and-waves:wave}{golf} van de
zender zelf, die in de pas resoneert, voedt de kring opnieuw --- het nieuws
speelt op de \omterm{def:g10:energy-conservation:energy}{energie} van de \omterm{def:g10:signals-and-waves:wave}{golf} zelf.
\end{solution}
