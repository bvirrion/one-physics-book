\section*{Hoofdstuk \ref{ch:g12:newtons-laws} --- De wetten van Newton}

\begin{solution}{exo:g12:newtons-laws:1}
(a) $p = 70 \times 10 = \qty{7.0e2}{kg.m/s}$.
(b) $v = \qty{13.9}{m/s}$: $p \approx \qty{1.8e4}{kg.m/s}$.
(c) $p = 0.0080 \times 800 = \qty{6.4}{kg.m/s}$. Auto $>$ sprinter $>$
kogel: vijf \omterm{def:g10:orders-of-magnitude:oom}{ordegrootten} massa zijn meer dan snelheid kan kopen.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:2}
$N = m(g + a_z)$: (a) $70 \times 11.01 = \qty{771}{N}$; (b) $a_z = 0$,
dus $N = mg = \qty{687}{N}$ --- de eerste wet: constante snelheid,
\omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht}; (c) $70 \times 8.61 = \qty{603}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:3}
$a = g\sin 30^\circ = \qty{4.9}{m/s^2}$;
$v = \sqrt{2 \times 4.905 \times 3.0} = \qty{5.4}{m/s}$. De massa valt weg
tussen $ma$ en $mg\sin\alpha$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:4}
Omkering: $\Delta p = 0.058 \times (25 + 15) = \qty{2.3}{kg.m/s}$;
$F = 2.32/0.0050 = \qty{4.6e2}{N}$ --- ongeveer $800$ keer het \omterm{def:g10:universal-gravitation:weight}{gewicht}
van \qty{0.57}{N}, en daarom wordt de zwaartekracht tijdens het contact
genegeerd.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:5}
\omterm{def:g10:universal-gravitation:weight}{Gewicht} (Aarde op boek, omlaag): partner $=$ het boek trekt de Aarde
omhoog. \omterm{def:g12:newtons-laws:friction}{Normaalkracht} (tafel op boek, omhoog): partner $=$ het boek drukt
op de tafel, omlaag. \omterm{def:g10:universal-gravitation:weight}{Gewicht} en \omterm{def:g12:newtons-laws:friction}{normaalkracht} werken op \emph{hetzelfde}
lichaam en horen bij twee verschillende wisselwerkingen --- hier alleen
even groot omdat $\vect a = \vect 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:6}
$N = mg = \qty{490}{N}$. (a) $F > \mu_s N = \qty{221}{N}$.
(b) $a = (220.7 - 0.35 \times 490.5)/50 = \qty{0.98}{m/s^2}$.
(c) $F = \mu_k N = \qty{172}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:7}
(a) Het glijdt zodra $\tan\alpha > \mu_s$:
$\alpha = \arctan 0.45 \approx 24^\circ$.
(b) $a = g(\sin 30^\circ - 0.35\cos 30^\circ) = 9.81 \times 0.197 =
\qty{1.9}{m/s^2}$. (c) Ja: zodra het glijdt, zakt de \omterm{def:g10:inertia:inventory}{wrijving} naar
$\mu_k < \mu_s$, dus
$a = g\cos\alpha\,(\tan\alpha - \mu_k) \approx \qty{0.9}{m/s^2} > 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:8}
$a = m_2 g/(m_1 + m_2) = 19.62/5.0 = \qty{3.9}{m/s^2}$;
$T = m_1 a = \qty{11.8}{N}$. Het hangende blok versnelt omlaag, dus wijst
de \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} erop omlaag: $T$ moet onder $m_2 g$ blijven.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:9}
$v_{\max} = \sqrt{\mu_s g R}$: droog
$\sqrt{0.70 \times 9.81 \times 50} = \qty{18.5}{m/s} \approx
\qty{67}{km/h}$; nat $\qty{14.0}{m/s} = \qty{50}{km/h}$. Zowel
$mv^2/R$ als $\mu_s mg$ draagt $m$: die valt weg.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:10}
Verticaal: $T\cos\theta = mg$, dus
$T = 2.943/\cos 25^\circ = \qty{3.2}{N}$. Straal
$R = L\sin\theta = \qty{0.507}{m}$; radiaal: $T\sin\theta = mv^2/R$, dus
$v = \sqrt{gR\tan\theta} = \sqrt{9.81 \times 0.507 \times 0.466}
= \qty{1.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:11}
$a_z = (N - mg)/m$ met $mg = \qty{49.05}{N}$: $+\qty{1.0}{m/s^2}$, $0$,
$-\qty{1.0}{m/s^2}$. Bijvoorbeeld een rit omhoog: optrekken, doorrijden,
afremmen. En ja: als de lift al daalde, betekenen dezelfde aflezingen
afremmen, doorrijden en weer versnellen omlaag --- de weegschaal leest de
versnelling af, niet de snelheid.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:12}
(a) $0 = 60V + 4.0 \times 8.0$: $V = \qty{-0.53}{m/s}$. (b) De bal gaat
met $8.0 + V$ ten opzichte van de grond: $0 = 60V + 4.0(8.0 + V)$, dus
$V = -32/64 = \qty{-0.50}{m/s}$ en de bal vliegt met \qty{7.5}{m/s}.
(c) De duw van de bal op haar handen --- de partner uit de derde wet van
haar worp.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:13}
Wrijvingsloos: $N\cos\theta = mg$ (verticaal) en $N\sin\theta = mv^2/R$
(radiaal); delen geeft $\tan\theta = v^2/(gR) = 400/784.8$, dus
$\theta \approx 27^\circ$. Trager: de auto neigt de verkanting af te
glijden en moet de \omterm{def:g10:inertia:inventory}{wrijving} de helling op wijzen; sneller: omgekeerd.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:14}
(a) $250 \times 2500 = \qty{6.3e5}{kg.m/s}$ per seconde: een \omterm{def:g10:inertia:inventory}{stuwkracht}
van $\qty{6.3e5}{N}$. (b) $P = \qty{4.9e5}{N}$;
$a = (6.25 - 4.905) \times 10^5 / \num{5.0e4} = \qty{2.7}{m/s^2}$.
(c) Ze duwt op haar eigen uitlaatgassen (derde wet) en niet op de lucht.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:15}
(a) $m_2 g > \mu_s m_1 g$: $m_2 > 0.35 \times 4.0 = \qty{1.4}{kg}$.
(b) $a = (m_2 - \mu_k m_1)\,g/(m_1 + m_2) = 1.0 \times 9.81/6.0 =
\qty{1.6}{m/s^2}$; $T = m_2(g - a) = \qty{16.4}{N}$. (c) $\mu_k = 0$
geeft $a = m_2 g/(m_1 + m_2)$: de formule van
\cref{exo:g12:newtons-laws:8}.
\end{solution}

\begin{solution}{pb:g12:newtons-laws:1}
\textbf{1.} Systeem: cabine $+$ loopwerk ($+$ passagiers). De berg ligt
vast aan de grond en is tot in hoge nauwkeurigheid een \omterm{def:g12:newtons-laws:inertial}{traagheidsstelsel}.
\omterm{def:g10:inertia:force}{Krachten}: het \omterm{def:g10:universal-gravitation:weight}{gewicht} $M\vect g$ (omlaag), de \omterm{def:g12:newtons-laws:friction}{normaalkracht} $\vect N$ van
de draagkabel (loodrecht erop) en de trekspankracht $\vect T$ (de helling
op).

\textbf{2.} In rust geldt de eerste wet: $\sum \vect F = \vect 0$. Langs
de helling: $T = Mg\sin\alpha$; loodrecht erop: $N = Mg\cos\alpha$.

\textbf{3.} $T = 4000 \times 9.81 \times 0.500 = \qty{19.6}{kN}$;
$N = 4000 \times 9.81 \times 0.866 = \qty{34.0}{kN}$.

\textbf{4.} Leeg: $T = \qty{9.8}{kN}$. Elke \omterm{def:g10:inertia:force}{kracht} in het \omterm{def:g10:inertia:compensate}{evenwicht} is
evenredig met $M$, dus $T$ ook.

\textbf{5.} Langs de helling geldt $Ma = Mg\sin\alpha$: $M$ valt weg, dus
$a = g\sin\alpha = \qty{4.9}{m/s^2}$, de helling af.

\textbf{6.} $T' = M(g\sin\alpha + a) = 4000 \times 5.505 =
\qty{22.0}{kN}$.

\textbf{7.} $a/(g\sin\alpha) = 0.60/4.905 \approx 12\%$.

\textbf{8.} $t = v/a = \qty{10}{s}$; $d = v^2/(2a) = \qty{30}{m}$.

\textbf{9.} Verticale component van $\vect a$:
$a\sin\alpha = \qty{0.30}{m/s^2}$, dus
$N = m(g + a\sin\alpha) = 80 \times 10.11 = \qty{809}{N}$ tegenover
$mg = \qty{785}{N}$: $3\%$ zwaarder.

\textbf{10.} De \omterm{def:g11:forces-and-motion:friction}{statische wrijving} van de vloer, horizontaal:
$ma\cos\alpha = 80 \times 0.52 \approx \qty{42}{N}$. Door te leunen laat
ze de gecombineerde duw van de vloer door haar \omterm{def:g11:forces-and-motion:com}{massamiddelpunt} gaan,
zodat die haar niet omkiept.

\textbf{11.} $a_T = dv/dt = 0$ (constante snelheid);
$a_N = v^2/R = 36/40 = \qty{0.90}{m/s^2}$, naar het middelpunt gericht ---
op het hoogste punt dus recht omlaag.

\textbf{12.} Normale as (omlaag positief): $Mg - N' = Mv^2/R$, dus
$N' = M(g - v^2/R) = 4000 \times 8.91 = \qty{35.6}{kN}$.

\textbf{13.} $N = m(g - v^2/R) = 80 \times 8.91 = \qty{713}{N}$: lichter
met $v^2/(gR) = 0.90/9.81 \approx 9\%$.

\textbf{14.} $N' = 0$ bij $v = \sqrt{gR} = \qty{19.8}{m/s} \approx
\qty{71}{km/h}$ --- meer dan drie keer de rijsnelheid: het loopwerk
volgt de kabel met een ruime marge.

\textbf{15.} De derde wet van Newton: \qty{35.6}{kN}, omlaag drukkend op
de kabel (en dus op de mast) --- net iets minder dan het \omterm{def:g10:universal-gravitation:weight}{gewicht} in rust,
verlicht door de gekromde vlucht.

\textbf{16.} $F = \Delta p/\Delta t = 4000 \times 6.0/1.5 = \qty{16}{kN}$,
tegengesteld aan de beweging ($a = \qty{4.0}{m/s^2}$).

\textbf{17.} $f = ma = 80 \times 4.0 = \qty{320}{N}$; het plafond is
$\mu_s mg = 0.40 \times 785 = \qty{314}{N} < \qty{320}{N}$: op de voeten
alleen glijd je weg --- houd je aan de leuning vast.

\textbf{18.} $0 = M_0 V + m(v_{\text{rel}} + V)$:
$V = -80 \times 2.5/2080 = \qty{-0.096}{m/s}$; de passagier beweegt met
$2.5 - 0.096 \approx \qty{2.4}{m/s}$ ten opzichte van de grond.

\textbf{19.} De raket: haar \omterm{def:g10:inertia:inventory}{stuwkracht} ligt vast door de per seconde
uitgestoten massa maal de uitstootsnelheid ten opzichte van haar.

\textbf{20.} De grootste \omterm{def:g10:inertia:inventory}{spankracht} die hierboven voorkomt is die bij het
optrekken, $T' = M \times \qty{5.505}{N/kg}$ (in rust geeft dat
$Mg\sin\alpha$, boven de mast nog minder). Grens:
$T' \leq \qty{20}{kN}$, dus $M \leq 20000/5.505 = \qty{3.63e3}{kg}$: een
nuttige last van \qty{1.63e3}{kg}, oftewel $20$ passagiers van
\qty{80}{kg} (niet $25$). Op het bordje: \emph{20 personen}; dan is
$M = \qty{3.6e3}{kg}$, $T' = \qty{19.8}{kN}$ en de veiligheidsfactor
$100/19.8 \approx 5.1 \geq 5$.
\end{solution}
