\section*{Hoofdstuk \ref{ch:g12:satellites-kepler} --- Satellieten en de beweging van de planeten}

\begin{solution}{exo:g12:satellites-kepler:1}
Op het platform: $F = GMm/R^2 = \num{3.98e17}/\num{4.06e13} \approx
\qty{9.8e3}{N}$. Op $r = \qty{6.77e6}{m}$: $F \approx \qty{8.7e3}{N}$ ---
nog altijd $89\%$ van de waarde op de grond. De inzittenden zweven omdat ze
samen met het station \emph{vallen}, niet omdat de \omterm{def:g11:fundamental-interactions:four}{gravitatie} weg is.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:2}
$v = \sqrt{GM/R} = \sqrt{\num{3.98e14}/\num{6.37e6}} \approx
\qty{7.9e3}{m/s}$; $T = 2\pi R/v \approx \qty{5.1e3}{s} \approx
\qty{84}{min}$. Precies de \qty{7.9}{km/s} van de kanonskogel: een scherende
\omterm{def:g10:universal-gravitation:satellite}{omloopbaan} \emph{is} het schot van Newton.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:3}
$v \propto 1/\sqrt{r}$: gedeeld door $\sqrt 2 \approx 1.41$.
$T \propto r^{3/2}$: vermenigvuldigd met $2^{3/2} \approx 2.8$. Geen van
beide bevat $m$: nee.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:4}
$v = \sqrt{\num{3.98e14}/\num{6.77e6}} \approx \qty{7.67e3}{m/s}$;
$T = 2\pi r/v \approx \qty{5.55e3}{s} \approx \qty{92}{min}$;
$\num{86400}/5546 \approx 15.6$ omloopbanen per dag.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:5}
Ellipsen met de Zon in een \omterm{def:g12:satellites-kepler:ellipse}{brandpunt}; gelijke oppervlakken in gelijke tijden;
$T^2/a^3$ gelijk voor alle planeten. Het snelst in het \omterm{def:g12:satellites-kepler:ellipse}{perihelium}, volgens de
tweede wet (het korte lijnstuk moet snel bestrijken). Nee: de massa van de
\omterm{def:g10:universal-gravitation:satellite}{satelliet} valt weg --- alleen de centrale massa doet mee.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:6}
$T = \qty{2.36e6}{s}$: $M = 4\pi^2 r^3/(GT^2) = \num{2.24e27}/371 \approx
\qty{6.0e24}{kg}$ --- tot op $1\%$ de \qty{5.97e24}{kg} van de kaart.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:7}
De \omterm{def:g11:fundamental-interactions:four}{gravitatie} wijst naar het middelpunt van de Aarde, en bij een
cirkelbeweging wijst de \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} naar het middelpunt van de
cirkel: die twee middelpunten vallen samen. Een cirkel boven de breedtecirkel
van Parijs heeft zijn middelpunt op de as, niet in het middelpunt van de
Aarde --- onmogelijk. Stilhangen lukt alleen waar de breedtecirkel een grote
cirkel om het middelpunt is: boven de \emph{evenaar}.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:8}
$T = \qty{2.754e4}{s}$: $M = 4\pi^2 r^3/(GT^2) =
\num{3.26e22}/\num{5.06e-2} \approx \qty{6.4e23}{kg}$ --- de Mars van de
gegevenskaart.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:9}
Aarde: $T^2/a^3 = \num{9.96e14}/\num{3.35e33} = \qty{2.98e-19}{s^2/m^3}$.
Mars: $T = \qty{5.94e7}{s}$, $\num{3.52e15}/\num{1.18e34} =
\qty{2.98e-19}{s^2/m^3}$ --- gelijk, zoals de derde wet eist. Dan is
$M = 4\pi^2/(G \times \num{2.98e-19}) \approx \qty{1.99e30}{kg}$.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:10}
$v = 2\pi r/T = \qty{1.02e3}{m/s}$; $a = v^2/r = \qty{2.72e-3}{m/s^2}$;
$g/60^2 = 9.81/3600 = \qty{2.73e-3}{m/s^2}$. De Maan staat $60$ aardstralen
ver en haar val is $60^2$ keer zwakker dan die van de appel: precies het
omgekeerde kwadraat --- één wet voor beide.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:11}
Ceres: $T = 2.77^{3/2} = \sqrt{21.3} \approx \qty{4.6}{jaar}$. Halley:
$T = 17.8^{3/2} \approx \qty{75}{jaar}$, dus het volgende \omterm{def:g12:satellites-kepler:ellipse}{perihelium} rond
$1986 + 75 \approx 2061$.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:12}
Op \qty{400}{km}: $v \approx \qty{7.67e3}{m/s}$; op \qty{350}{km}
($r = \qty{6.72e6}{m}$): $v \approx \qty{7.70e3}{m/s}$. De \omterm{def:g10:inertia:inventory}{wrijving} tapt wel
degelijk \omterm{def:g11:mechanical-energy:em}{mechanische energie} af --- de \omterm{def:g10:universal-gravitation:satellite}{omloopbaan} krimpt; maar onderweg naar
beneden vergoedt de \omterm{def:g11:work-of-force:work}{arbeid} van de \omterm{def:g11:fundamental-interactions:four}{gravitatie} meer dan het snelheidsverlies
aan de luchtweerstand: $E_m$ daalt terwijl $E_k$ stijgt.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:13}
$T = \qty{3.024e5}{s}$: $M = 4\pi^2 r^3/(GT^2) = \num{1.65e31}/6.10 \approx
\qty{2.7e30}{kg} \approx 1.4$ zonsmassa's. De massa van de \emph{planeet}
blijft buiten bereik: die viel weg uit de vergelijkingen van de \omterm{def:g10:universal-gravitation:satellite}{omloopbaan}.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:14}
$T = \qty{8.862e4}{s}$, $GM = \qty{4.28e13}{m^3/s^2}$:
$r^3 = GMT^2/(4\pi^2) = \num{8.52e21}$, dus $r \approx \qty{2.04e7}{m}$ en
$h = r - R \approx \qty{1.7e7}{m} \approx \qty{17000}{km}$ boven de grond
van Mars.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:15}
$v_a = r_p v_p / r_a = \num{8.77e10} \times \num{5.45e4} / \num{5.25e12}
\approx \qty{9.1e2}{m/s}$. Omdat $r_a/r_p \approx 60$, staat de komeet $60$
keer verder en gaat hij $60$ keer trager: hij treuzelt decennialang in het
donker en sprint in enkele weken langs de Zon.
\end{solution}

\begin{solution}{pb:g12:satellites-kepler:1}
\textbf{1.} $T = \qty{86164}{s}$: de grond draait met de Aarde mee, één keer
per \omterm{def:g12:satellites-kepler:geo}{siderische dag} \emph{ten opzichte van de sterren}; de dag van \qty{24}{h}
telt er de graad per dag bij die de Aarde om de Zon opschuift.

\textbf{2.} De \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} van een cirkelbeweging mikt op het
middelpunt van de cirkel; de enige \omterm{def:g10:inertia:force}{kracht}, de \omterm{def:g11:fundamental-interactions:four}{gravitatie}, mikt op het
middelpunt van de Aarde: die twee middelpunten zijn hetzelfde punt.

\textbf{3.} Een cirkel die boven Parijs hangt zou zijn middelpunt op de as
hebben, ten noorden van het middelpunt --- uitgesloten door 2. De \omterm{def:g10:universal-gravitation:satellite}{omloopbaan}
moet in het \emph{vlak van de evenaar} liggen.

\textbf{4.} Normale component van de tweede wet van Newton:
$m v^2/r = GmM/r^2$, dus $v = \sqrt{GM/r}$ ($m$ valt weg).

\textbf{5.} $T = 2\pi r/v = 2\pi\sqrt{r^3/(GM)}$; kwadrateren geeft
$T^2/r^3 = 4\pi^2/(GM)$.

\textbf{6.} $r = \left(GMT^2/4\pi^2\right)^{1/3} = (\num{7.49e22})^{1/3}
\approx \qty{4.22e7}{m}$.

\textbf{7.} $h = r - R = \num{4.22e7} - \num{6.37e6} \approx
\qty{3.58e7}{m} \approx \qty{35800}{km}$.

\textbf{8.} $v = 2\pi \times \num{4.22e7}/\num{86164} \approx
\qty{3.07e3}{m/s}$.

\textbf{9.} $\sqrt{\num{3.98e14}/\num{4.22e7}} \approx \qty{3.07e3}{m/s}$ ---
in overeenstemming.

\textbf{10.} Dezelfde straal: de massa viel weg in 4. De zwaardere \omterm{def:g10:universal-gravitation:satellite}{satelliet}
verandert alleen de rekening van de lancering --- meer brandstof voor
dezelfde \omterm{def:g10:universal-gravitation:satellite}{omloopbaan}.

\textbf{11.} $v = \sqrt{\num{3.98e14}/\num{6.77e6}} \approx
\qty{7.67e3}{m/s}$.

\textbf{12.} $T = 2\pi r/v \approx \qty{5.55e3}{s} \approx \qty{92.4}{min}$.

\textbf{13.} $\num{86400}/5546 \approx 15.6$ omloopbanen: ongeveer $16$
zonsopgangen per dag.

\textbf{14.} $r_{\text{geo}}/r_{\text{ISS}} = \num{4.22e7}/\num{6.77e6}
= 6.23$; $6.23^{3/2} \approx 15.5 = \num{86164}/5546$ --- de derde wet van
Kepler, ter plekke geverifieerd.

\textbf{15.} Het station gaat sneller (\qty{7.67}{km/s} tegen
\qty{3.07}{km/s}), en toch kostte de geostationaire meer \omterm{def:g10:energy-conservation:energy}{energie} per
\omterm{def:g10:orders-of-magnitude:unit}{kilogram}: de klim in \omterm{def:g11:mechanical-energy:potential}{potentiële energie} weegt zwaarder dan het verlies aan
snelheid.

\textbf{16.} $M_J = 4\pi^2 r^3/(G T^2)$.

\textbf{17.} $T = \qty{1.529e5}{s}$: $M_J = \num{2.97e27}/1.56 \approx
\qty{1.90e27}{kg}$.

\textbf{18.} $\num{1.90e27}/\num{5.97e24} \approx 318$ aardmassa's --- en
ongeveer $1/1000$ van de Zon.

\textbf{19.} $T_J = \qty{3.576e4}{s}$: $r^3 = GM_J T_J^2/(4\pi^2) =
\num{4.11e24}$, $r \approx \qty{1.60e8}{m}$ --- Io, op \qty{4.22e8}{m},
vliegt $2.6$ keer hoger en drijft dus over de hemel van Jupiter.

\textbf{20.} Parkeer het relais op de cirkel boven de evenaar op hoogte
\qty{35800}{km}, waar het met \qty{3.07}{km/s} meedraait en nooit uit het
vizier van de schotel loopt; en dezelfde wet leverde, gevoed met de maand van
Io, Jupiter gratis op \qty{1.90e27}{kg} --- $318$ aardmassa's.
\end{solution}
