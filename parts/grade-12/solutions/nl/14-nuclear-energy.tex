\section*{Hoofdstuk \ref{ch:g12:nuclear-energy} --- Kernenergie: splijting, fusie, $E=mc^2$}

\begin{solution}{exo:g12:nuclear-energy:1}
(a) $\qty{1}{eV} = \qty{1.60e-19}{J}$; $\qty{1}{MeV} = \qty{1.60e-13}{J}$.
(b) $200 \times \num{1.60e-13} = \qty{3.2e-11}{J}$.
(c) $E = \num{1.66054e-27} \times \num{9.00e16} = \qty{1.49e-10}{J}
= \num{1.49e-10}/\num{1.60e-13} \approx \qty{9.3e2}{MeV}$ --- de
\qty{931.5}{MeV} uit de cursus, op onze afgeronde $c$ na.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:2}
$E = \num{6.0e-3} \times \num{9.00e16} = \qty{5.4e14}{J}$;
$\num{5.4e14}/\num{3.6e7} = \num{1.5e7}$ dagen $\approx \num{4.1e4}$ jaar ---
veertig millennia op één suikerklontje.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:3}

\qty{0.09893}{u}$; $E_b = 0.09893 \times 931.5 \approx \qty{92.2}{MeV}$;
$E_b/A \approx \qty{7.68}{MeV}$ per \omterm{def:g11:fundamental-interactions:ladder}{nucleon}.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:4}
(a) $A = 236 - 144 - 3 = 89$, $Z = 92 - 56 = 36$: ${}^{89}_{36}\mathrm{Kr}$,
krypton. (b) $A = 236 - 97 - 2 = 137$, $Z = 92 - 40 = 52$:
${}^{137}_{52}\mathrm{Te}$, telluur.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:5}
Ongeveer $7.1$, $7.7$, $8.8$ en \qty{7.6}{MeV} per \omterm{def:g11:fundamental-interactions:ladder}{nucleon}; ijzer-56 is het
sterkst gebonden. Uranium kan \omterm{def:g10:energy-conservation:energy}{energie} vrijgeven door \omterm{def:g12:nuclear-energy:fission}{kernsplijting}, helium-4

zelf kan nergens meer heen.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:6}

\begin{solution}{exo:g12:nuclear-energy:7}
Ervóór: $\qty{236.00216}{u}$; erna: $143.89220 + 88.89790 +
3 \times 1.00866 = \qty{235.81608}{u}$; $\Delta m = \qty{0.18608}{u}$, dus

\end{solution}

\begin{solution}{exo:g12:nuclear-energy:8}
$\Delta m = 2 \times 2.01355 - 3.01493 - 1.00866 = \qty{0.00351}{u}$:
$E \approx \qty{3.27}{MeV}$. D--T maakt helium-4, dat uitzonderlijk sterk
gebonden is (\qty{7.1}{MeV} per \omterm{def:g11:fundamental-interactions:ladder}{nucleon}); helium-3 ligt veel lager op de
kromme, dus geeft die stap veel minder vrij.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:9}
(a) Het neutron is neutraal: geen \omterm{def:g12:nuclear-energy:barrier}{coulombbarrière}, dus vuurt \omterm{def:g12:nuclear-energy:fission}{kernsplijting}
koud. Kernen die samensmelten zijn allebei positief en moeten de barrière op,
zodat alleen een opwinding van $\sim \qty{1e8}{K}$ botsingen geeft die heftig
genoeg zijn. (b) De kern van de Zon bevat een astronomisch aantal protonen; de
zeldzame snelste leden van de menigte doen het smeltwerk, en de Zon heeft
miljarden jaren de tijd --- een traag vuur is precies wat een ster wil.
(c) Er is geen neutronenlawine: elke fusie wordt door haar eigen botsing
betaald, en elk verlies aan opsluiting koelt het \omterm{def:g12:nuclear-energy:barrier}{plasma} af en dooft het vuur.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:10}
$\Delta m = 4 \times 1.00728 - 4.00151 = \qty{0.02761}{u}$, dus
$E \approx \qty{25.7}{MeV}$; de fractie is $0.02761/4.02912 \approx 0.7\%$ van
de massa.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:11}
(a) Hij vertraagt de splijtingsneutronen door botsingen; trage neutronen
splijten uranium-235 veel waarschijnlijker, dus heeft de ketting ze nodig.
(b) Ze slurpen neutronen op en houden de ketting op één neutron per splijting
--- ingeschoven dooft ze. (c) Geen water, geen moderatie: de neutronen blijven
snel, missen, en de ketting smoort --- het ontwerp faalt naar ``uit''.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:12}
Het aantal splijtingen verdubbelt per generatie, dus zijn er na $n$ generaties
ongeveer $2^n$ gespleten; $2^{n} = \num{2.56e21} \approx 2 \times (10^{3})^{7}
\approx 2^{71}$ geeft $n \approx 71$. Tijd: $71 \times \qty{10}{ns}
\approx \qty{0.7}{\micro s}$ --- een gram brandstof, goed voor drie ton
steenkool, in minder dan een microseconde: kriticiteit is een klif, geen
helling.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:13}
$\Delta m = 55.92066 - 2 \times 27.96924 = \qty{-0.01782}{u}$:
$E \approx \qty{-16.6}{MeV}$ --- het splijten \emph{slurpt} \omterm{def:g10:energy-conservation:energy}{energie} op. IJzer
ligt op de piek: het splijten of het samensmelten gaat allebei bergaf in
binding en kost allebei \omterm{def:g10:energy-conservation:energy}{energie}. Het is de as van elk kernvuur.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:14}
Fe: $\Delta m = 26 \times 1.00728 + 30 \times 1.00866 - 55.92066 =
\qty{0.52842}{u}$, $E_b \approx \qty{492}{MeV}$, $E_b/A \approx
\qty{8.79}{MeV}$. U: $\Delta m = 92 \times 1.00728 + 143 \times
1.00866 - 234.99350 = \qty{1.91464}{u}$, $E_b \approx \qty{1784}{MeV}$,
$E_b/A \approx \qty{7.59}{MeV}$. $235$ \omterm{def:g11:fundamental-interactions:ladder}{nucleonen} opnieuw inpakken op
\qty{8.5}{MeV} in plaats van \qty{7.59}{MeV} levert $235 \times 0.9
\approx \qty{2e2}{MeV}$ --- de $185$ tot $200$ van het uitgewerkte voorbeeld.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:15}
$\num{5.0e8}/\num{2.82e-12} \approx \num{1.8e20}$ reacties per seconde;
massatempo $\num{1.8e20} \times 5.029 \times \num{1.66054e-27}
\approx \qty{1.5e-6}{kg/s}$, oftewel ongeveer \qty{0.13}{kg} per dag ---
tegenover \num{1.4e6} \omterm{def:g10:orders-of-magnitude:unit}{kilogram} steenkool: tien miljoen tegen één.
\end{solution}

\begin{solution}{pb:g12:nuclear-energy:1}
\textbf{1.} $P_{\text{th}} = \num{1.0e9}/0.33 \approx \qty{3.0}{GW}$.

\textbf{2.} $200 \times \num{1.60e-13} = \qty{3.2e-11}{J}$.

\textbf{3.} $\num{3.0e9}/\num{3.2e-11} \approx \num{9.4e19}$
\omterm{def:g12:nuclear-energy:fission}{kernsplijtingen} per seconde.

\textbf{4.} $m_{\mathrm U} = 235 \times \num{1.66054e-27} =
\qty{3.90e-25}{kg}$; $\num{9.4e19} \times \num{3.90e-25} \approx
\qty{3.7e-5}{kg/s}$ --- een veertigtal microgram per seconde.

\textbf{5.} $\num{3.7e-5} \times \num{3.16e7} \approx \qty{1.2e3}{kg}$:
ongeveer $1.2$ ton uranium-235 per jaar.

\textbf{6.} $\num{1.2e3}/0.040 \approx \qty{2.9e4}{kg}$ --- zo'n $29$ ton
brandstof: één zware vrachtwagen, één keer per jaar.

\textbf{7.} $\num{3.0e9} \times \num{3.16e7} \approx \qty{9.5e16}{J}$.

\textbf{8.} $\num{9.5e16}/\num{3.0e7} \approx \qty{3.2e9}{kg}$: $3.2$ miljoen
ton steenkool.

\textbf{9.} $\num{3.2e9}/\num{1.2e3} \approx \num{2.7e6}$ --- de
$\num{8.2e13}/\num{3.0e7}$ van de ladder, zoals het hoort.

\textbf{10.} $\num{3.2e9}/\num{3.0e6} \approx 1060$ treinen per jaar, ruwweg
drie per dag: een kolentrein bij dageraad, om twaalf uur en bij zonsondergang,
voor altijd, tegenover één vrachtwagen elke lente.

\textbf{11.} $17.6 \times \num{1.60e-13} = \qty{2.82e-12}{J}$.

\textbf{12.} Eén paar weegt $5.029 \times \num{1.66054e-27} =
\qty{8.35e-27}{kg}$: $\num{2.82e-12}/\num{8.35e-27} \approx
\qty{3.4e14}{J/kg}$ --- de fusiesport van de ladder.

\textbf{13.} $\num{9.5e16}/\num{3.4e14} \approx \qty{2.8e2}{kg}$ D--T; het
aandeel deuterium is $280 \times 2.014/5.029 \approx \qty{1.1e2}{kg}$.

\textbf{14.} $\num{1.1e2}/\num{3.3e-2} \approx \qty{3.4e3}{m^3}$ zeewater ---
ongeveer anderhalf zwembad voor het hele jaar van de stad.

\textbf{15.} Niets vermenigvuldigt zich: geen neutron ontsteekt de volgende
fusie, en het verlies van de opsluiting koelt het \omterm{def:g12:nuclear-energy:barrier}{plasma} af en stopt haar
binnen seconden. Het moeilijke is precies het omgekeerde --- \qty{1.5e8}{K}
lang genoeg bijeenhouden om te branden.

\textbf{16.} $\num{3.9e26}/\num{9.00e16} \approx \qty{4.3e9}{kg/s}$: vier
miljoen ton per seconde.

\textbf{17.} $\num{3.0e9}/\num{9.00e16} \approx \qty{3.3e-8}{kg/s}$, oftewel
$\num{3.3e-8} \times \num{3.16e7} \approx \qty{1.0}{kg}$ per jaar --- en ja:
$\qty{1.2e3}{kg} \times 0.00085 \approx 1$, $\num{3.2e9}
\times \num{3.3e-10} \approx 1$, $280 \times 0.00375 \approx 1$. Elke
brandstof staat dezelfde \omterm{def:g10:orders-of-magnitude:unit}{kilogram} af; ze verschillen alleen in hoeveel vracht
die moet dragen.

\textbf{18.} $\num{4.3e9} \times \num{3.16e7} \times \num{4.5e9}
\approx \qty{6.1e26}{kg}$ --- slechts $\num{6.1e26}/\num{2.0e30} \approx
\num{3e-4}$, drie delen op tienduizend.

\textbf{19.} Omzetbare massa $0.0007 \times \num{2.0e30} =
\qty{1.4e27}{kg}$, goed voor $\num{1.4e27} \times \num{9.00e16} \approx
\qty{1.3e44}{J}$; bij \qty{3.9e26}{W} houdt dat $\num{1.3e44}/
\num{3.9e26} \approx \qty{3.2e17}{s} \approx \num{1.0e10}$ jaar vol --- tien
miljard, waarvan er zo'n vijf miljard resten.

\textbf{20.} Eén jaar van de stad: $3.2$ miljoen ton steenkool, of
$\num{1.2e3}$ \omterm{def:g10:orders-of-magnitude:unit}{kilogram} uranium-235, of $\num{2.8e2}$ \omterm{def:g10:orders-of-magnitude:unit}{kilogram} D--T uit
anderhalf zwembad zeewater. Onder elk grootboek dezelfde post: één \omterm{def:g10:orders-of-magnitude:unit}{kilogram}
massa, verdwenen. Het kan $E = mc^2$ niet schelen welke brandstof haar draagt
--- alleen hoe groot de vrachtwagen moet zijn.
\end{solution}
