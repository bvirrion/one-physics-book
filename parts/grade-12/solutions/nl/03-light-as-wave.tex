\section*{Hoofdstuk \ref{ch:g12:light-as-wave} --- Licht als golf: buiging en interferentie}

\begin{solution}{exo:g12:light-as-wave:1}
$f = c/\lambda$: rood $\num{3.00e8}/\num{7.0e-7} \approx
\qty{4.3e14}{Hz}$; violet $\approx \qty{7.5e14}{Hz}$ --- violet is hoger
(kortere \omterm{def:g12:mechanical-waves:wavelength}{golflengte}, dezelfde snelheid $c$).
\end{solution}

\begin{solution}{exo:g12:light-as-wave:2}
$\theta = \num{5.32e-7}/\num{2.0e-4} = \qty{2.7e-3}{rad}$;
$L = 2\theta D = 2 \times \num{2.66e-3} \times 1.5 \approx \qty{8.0}{mm}$.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:3}
Stem: $\lambda = 340/425 = \qty{0.80}{m}$, dus $\lambda/a = 1.0$ ---
maximale \omterm{def:g12:light-as-wave:diffraction}{buiging}, het \omterm{def:g10:signals-and-waves:sound}{geluid} overspoelt de gang. Licht:
$\lambda/a \approx \num{5.5e-7}/0.80 \approx \num{7e-7}$ ---
verwaarloosbare spreiding: het licht gaat rechtdoor, dus verbergt de hoek
het.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:4}
In $M$: $\delta = \qty{6.0}{cm} = 3\lambda$, constructief --- luid. In
$N$: $\delta = \qty{5.0}{cm} = (2 + \tfrac12)\lambda$, destructief ---
(vrijwel) stil.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:5}
$i = \lambda D/a = \num{6.33e-7} \times 1.8/\num{2.5e-4} \approx
\qty{4.6}{mm}$. Binnen $\pm\qty{1.0}{cm}$: $|k| \leq 10/4.6 = 2.2$, dus
$k = -2, \dots, 2$: vijf heldere strepen.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:6}
$d = 2\lambda D/L = 2 \times \num{6.5e-7} \times 1.6/0.028 \approx
\qty{74}{\micro m}$.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:7}
$i = \qty{4.7}{mm}$, dus $\lambda = i\,a/D = \num{4.7e-3} \times
\num{2.0e-4}/1.5 \approx \qty{6.3e-7}{m} = \qty{630}{nm}$: rood. Tien
\omterm{def:g12:light-as-wave:fringes}{strepenafstanden} meten deelt de afleesfout van het meetlint door tien.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:8}
De lampen hebben (grofweg) overeenkomende \omterm{def:g10:signals-and-waves:frequency}{frequenties} maar lopen \emph{niet
in de pas}: elke lamp zendt korte golftreinen uit waarvan de fase
willekeurig verspringt, dus zijn de bronnen incoherent en verschuiven de
momentane strepen miljarden keren per seconde --- het oog ziet het
gemiddelde, gelijkmatig. Young belicht beide spleten vanuit \emph{één}
\omterm{def:g10:signals-and-waves:wave}{golf}: de twee kopieën erven elke fasesprong samen en blijven in de pas.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:9}
Van de eerste tot de zevende is $6i = \qty{18.0}{mm}$, dus
$i = \qty{3.0}{mm}$; $\lambda = i\,a/D = \num{3.0e-3} \times
\num{3.0e-4}/1.4 \approx \qty{6.4e-7}{m} \approx \qty{640}{nm}$.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:10}
$\sin\theta = k\lambda/d = 0.396\,k$: voor $k = 1$ is
$\theta = 23^\circ$; voor $k = 2$ is $\sin\theta = 0.79$, dus
$\theta = 52^\circ$; $k = 3$ zou $\sin\theta = 1.19 > 1$ vergen:
onmogelijk. Hoogste orde: $k = 2$.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:11}
$\theta = \num{5.5e-7}/0.10 = \qty{5.5e-6}{rad}$. De koplampen spannen
$1.5/\num{1.0e5} = \qty{1.5e-5}{rad}$ op, ongeveer drie keer de vlek: ze
worden net gescheiden.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:12}
De weerkaatsingen op het voor- en het achtervlak \omterm{prop:g12:light-as-wave:conditions}{interfereren}; in \omterm{def:g10:light-spectra:white}{wit
licht} versterkt elke dikte sommige \omterm{def:g12:mechanical-waves:wavelength}{golflengten} en dooft andere uit, en
dus ontstaat er een kleur. De dikte van het vlies hangt alleen van de
hoogte af, dus vormen punten met dezelfde kleur horizontale banden. Door
het uitzakken wordt het vlies dunner, zodat de dikte die een bepaalde
kleur schilderde steeds lager komt te liggen: de banden kruipen omlaag.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:13}
$i = \lambda D/a$: violet $\num{4.0e-7} \times 1.5/\num{2.0e-4} =
\qty{3.0}{mm}$; rood \qty{5.6}{mm}. In het midden: alle kleuren helder
--- één witte streep. Een paar millimeter verderop zijn de
kleurpatronen uiteengeschoven: iriserende strepen. Nog verder overlappen
de maxima van alle kleuren overal: een gelijkmatig witte waas.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:14}
$\theta = \num{5.5e-7}/2.4 = \qty{2.3e-7}{rad}$; op \qty{250}{km} is dat
$\num{2.3e-7} \times \num{2.5e5} \approx \qty{6}{cm}$. Ze kan auto's
tellen en een persoon opmerken, maar krantendruk (letters van een
millimeter) ligt twintig keer onder de buigingsgrens: de bewering is een
mythe.
\end{solution}

\begin{solution}{exo:g12:light-as-wave:15}
De \omterm{def:g10:signals-and-waves:frequency}{frequentie} ligt door de bron vast: onveranderd. De snelheid zakt tot
$c/n$, dus is $\lambda' = \lambda/n$ en
$i' = i/n = 4.0/1.33 \approx \qty{3.0}{mm}$: de strepen kruipen dichter
op elkaar.
\end{solution}

\begin{solution}{pb:g12:light-as-wave:1}
\textbf{1.} $\theta = \lambda/a = \num{6.33e-7}/\num{1.00e-4} =
\qty{6.3e-3}{rad}$.

\textbf{2.} De halve breedte is $D\tan\theta \approx D\theta = \lambda
D/a$, dus $L = 2\lambda D/a = 2 \times \num{6.33e-7} \times
3.00/\num{1.00e-4} = \qty{3.8}{cm}$.

\textbf{3.} $(3.8 - 3.7)/3.8 \approx 3\%$: binnen de nauwkeurigheid van
een meetlint --- bevestigd.

\textbf{4.} $L \propto 1/a$: $a$ halveren verdubbelt $L$, dus
$L = \qty{7.6}{cm}$. Smallere hindernis of spleet, breder patroon.

\textbf{5.} $\tan(\num{6.33e-3}) = \num{6.33008e-3}$: ze komen tot op
ongeveer $\num{1e-5}$ relatief overeen --- de benadering is veel beter
dan elke meting hier.

\textbf{6.} Volgens \cref{prop:g12:light-as-wave:hair}: een hindernis met
breedte $d$ buigt als een spleet met breedte $d$, dus blijft
$L = 2\lambda D/d$ gelden.

\textbf{7.} $d = 2\lambda D/L = 2 \times \num{6.33e-7} \times
3.00/0.054 \approx \qty{70}{\micro m}$.

\textbf{8.} $L = \qty{5.2}{cm} \to d = \qty{73}{\micro m}$;
$L = \qty{5.6}{cm} \to d = \qty{68}{\micro m}$: dus
$d = \qty{70 \pm 3}{\micro m}$.

\textbf{9.} $d = \num{3.80e-6}/0.076 = \qty{50}{\micro m}$: die van je
vriendin is fijner --- en omdat $L \propto 1/d$, werpt de fijnere haar het
bredere patroon.

\textbf{10.} Nee: een lamp is wit (elke \omterm{def:g12:mechanical-waves:wavelength}{golflengte} schildert een ander
patroon, en ze vervagen samen) en incoherent én breed (er is geen enkele
zuivere bundel om te buigen). De \omterm{def:g11:color-light-sources:lamps}{laser} is \omterm{def:g10:light-spectra:white}{monochromatisch} en gericht.

\textbf{11.} Alleen kopieën van één \omterm{def:g10:signals-and-waves:wave}{golf} zijn \omterm{def:g12:light-as-wave:coherent}{coherent}; twee
onafhankelijke bronnen lopen uit de pas en de strepen spoelen weg
(\cref{rem:g12:light-as-wave:lamps}).

\textbf{12.} In het midden is $\delta = 0 = 0 \times \lambda$ voor elke
$\lambda$: constructief, wat de \omterm{def:g12:mechanical-waves:wavelength}{golflengte} ook is.

\textbf{13.} $i = 4.26/10 = \qty{4.3}{mm}$; één spanwijdte van tien lees
je met dezelfde meetfout af als één \omterm{def:g12:light-as-wave:fringes}{strepenafstand}, dus wordt de fout op
$i$ door tien gedeeld.

\textbf{14.} $\lambda = i\,a'/D = \num{4.26e-3} \times
\num{2.50e-4}/2.00 = \num{5.33e-7} \approx \qty{533}{nm}$.

\textbf{15.} $\qty{533}{nm}$ ligt binnen $\qty{532 \pm 10}{nm}$: dat
klopt; die \omterm{def:g12:mechanical-waves:wavelength}{golflengte} is groen.

\textbf{16.} $\tan\theta = 0.43/1.00$, dus $\theta = 23^\circ$. Hier is
$\tan\theta = 0.43$ terwijl $\sin\theta = 0.40$: bij $23^\circ$ wijkt het
gelijkstellen van $\sin$, $\tan$ en $\theta$ bijna $10\%$ af --- gebruik
de exacte functies.

\textbf{17.} $d = \lambda/\sin\theta = \num{6.33e-7}/0.396 \approx
\qty{1.6}{\micro m}$.

\textbf{18.} $\tan\theta = 1.65$, $\theta = 59^\circ$,
$\sin\theta = 0.855$: $d = \num{6.33e-7}/0.855 \approx
\qty{0.74}{\micro m}$.

\textbf{19.} $1.6/0.74 \approx 2.2$: de sporen van een dvd liggen twee
keer zo dicht, en haar putjes zijn navenant kleiner (gelezen met een \omterm{def:g11:color-light-sources:lamps}{laser}
met kortere \omterm{def:g12:mechanical-waves:wavelength}{golflengte}) --- meerdere keren zoveel gegevens op hetzelfde
schijfje van \qty{12}{cm}.

\textbf{20.} Haar \qty{70 \pm 3}{\micro m}; haar van je vriendin
\qty{50}{\micro m}; groene \omterm{def:g11:color-light-sources:lamps}{laser} \qty{533}{nm}; cd-sporen
\qty{1.6}{\micro m}; dvd-sporen \qty{0.74}{\micro m} --- een bekende
\omterm{def:g12:mechanical-waves:wavelength}{golflengte} maakt van de meetkunde van elk patroon een lengte, zodat licht
van een halve micrometer een maatlat is met een schaalverdeling van een
halve micrometer.
\end{solution}
