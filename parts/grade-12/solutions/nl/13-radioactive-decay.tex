\section*{Hoofdstuk \ref{ch:g12:radioactive-decay} --- Radioactief verval}

\begin{solution}{exo:g12:radioactive-decay:1}
$T_{1/2} = \ln 2/\lambda = 0.693/\num{1.0e-6} = \qty{6.9e5}{s} \approx 8.0$
dagen --- de klok van jodium-131.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:2}
$T_{1/2} = 3.8 \times \num{86400} = \qty{3.28e5}{s}$;
$\lambda = 0.693/\num{3.28e5} = \qty{2.1e-6}{s^{-1}}$;
$\mathcal A = \lambda N = \num{2.1e-6} \times \num{1.0e10} \approx
\qty{2.1e4}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:3}
$\dd N/\dd t = N_0(-\lambda)e^{-\lambda t} = -\lambda N$ --- de vergelijking
klopt; $N(0) = N_0 e^0 = N_0$; en
$N(T_{1/2})/N_0 = e^{-\lambda \ln 2/\lambda} = e^{-\ln 2} = \tfrac12$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:4}
$90$ jaar $= 3\,T_{1/2}$: $\mathcal A = \num{4.0e5}/2^3 = \qty{5.0e4}{Bq}$.
$300$ jaar $= 10\,T_{1/2}$: $\mathcal A = \num{4.0e5}/1024 \approx
\qty{3.9e2}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:5}
Helling $= (18.85 - 20.00)/10 = -0.115$, dus
$\lambda = \qty{0.115}{h^{-1}} = 0.115/3600 = \qty{3.2e-5}{s^{-1}}$;
$T_{1/2} = 0.693/0.115 = 6.0$ uur --- technetium-99m.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:6}
$N = (1.0/226) \times \num{6.02e23} = \num{2.7e21}$;
$T_{1/2} = 1600 \times \num{3.156e7} = \qty{5.05e10}{s}$, dus
$\lambda = \qty{1.37e-11}{s^{-1}}$; $\mathcal A = \lambda N \approx
\qty{3.7e10}{Bq}$ --- één curie, de \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van de gram radium van Marie
Curie.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:7}
Beiden verwachten dat de helft overblijft: $10$, en \num{1.0e20}. De
fluctuaties zijn $\sim 1/\sqrt{n}$ van de telling: bij $10$ vervallen zo'n
$30\%$ --- $7$ of $13$ zou niet verbazen; bij \num{1.0e20} ongeveer
$10^{-10}$. Alleen het grote monster volgt de wet tot op meetbare
nauwkeurigheid.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:8}
$m = 140 \times \num{1.2e-4} = \qty{1.7e-2}{g}$;
$N = (0.0168/40) \times \num{6.02e23} = \num{2.5e20}$.
$\lambda = 0.693/(\num{1.25e9} \times \num{3.156e7}) =
\qty{1.76e-17}{s^{-1}}$; $\mathcal A = \lambda N \approx \qty{4.4e3}{Bq}$ ---
ongeveer de helft van het totaal van het lichaam (de rest is grotendeels
koolstof-14).
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:9}
$\lambda = 0.693/6.0 = \qty{0.116}{h^{-1}}$. Na $15$ uur:
$\mathcal A = 500\,e^{-0.116 \times 15} = 500 \times 0.177 \approx
\qty{88}{MBq}$. Onder \qty{1}{MBq}: $t = \ln(500)/0.116 \approx \qty{54}{uur}$
--- zo'n $9$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:10}
Houtskool: $3.4/13.6 = \tfrac14$, precies $2$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}:
$t = \num{1.15e4}$ jaar. Bot:
$t = \num{5730} \times \ln(1/0.30)/\ln 2 = \num{5730} \times 1.74
\approx \num{1.0e4}$ jaar.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:11}
$\lambda = 0.693/3.8 = \qty{0.182}{d^{-1}}$;
$t = \ln(100)/\lambda = 4.61/0.182 \approx 25$ dagen (zo'n $6.6$
\omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}). Maar de uraanketen in het omringende graniet en de bodem
kweekt onophoudelijk nieuw radon: eenmaal luchten is binnen enkele weken weer
teniet, dus moet de ventilatie permanent zijn.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:12}
Uit $N/N_0 = e^{-\lambda t}$ volgt $\lambda t = \ln(N_0/N)$, en met
$\lambda = \ln 2/T_{1/2}$ wordt dat $t = T_{1/2}\ln(N_0/N)/\ln 2$. Hier:
$t = \num{5730} \times \ln(20)/\ln 2 = \num{5730} \times 4.32 \approx
\num{2.5e4}$ jaar.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:13}
$N_{\mathrm K} = N_0 e^{-\lambda t}$ en
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, dus
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$. Bij een verhouding $3.0$:
$e^{\lambda t} = 4$, dus $\lambda t = 2\ln 2$ en $t = 2\,T_{1/2} =
\num{2.5e9}$ jaar.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:14}
$\ln \mathcal A = 5.99$, $5.42$, $4.84$, $4.26$, $3.69$: dalingen van $0.57$
tot $0.58$ per \qty{60}{s} --- netjes op één lijn. Helling:
$\lambda = 2.30/240 = \qty{9.6e-3}{s^{-1}}$;
$T_{1/2} = 0.693/\num{9.6e-3} \approx \qty{72}{s}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:15}
(a) $2^{-10} \approx \num{1e-3}$. (b) $(1/2)^8 = 1/256 \approx 0.4\%$.
(c) $1000/8 = 125$ verwacht --- maar de wet zwijgt over welke: ze voorspelt
bevolkingen exact en individuen helemaal niet.
\end{solution}

\begin{solution}{pb:g12:radioactive-decay:1}
\textbf{1.} De uitwisseling met de atmosfeer (eten, ademen) houdt het aandeel
op de waarde van de atmosfeer aangevuld; de dood stopt de opname en het verval
neemt het over.

\textbf{2.} $T_{1/2} = \num{5730} \times \num{3.156e7} = \qty{1.81e11}{s}$,
dus $\lambda = 0.693/\num{1.81e11} = \qty{3.8e-12}{s^{-1}}$.

\textbf{3.} $\dd N/\dd t = -\lambda N_0 e^{-\lambda t} = -\lambda N$, en
$N(0) = N_0$.

\textbf{4.} $\mathcal A_0 = 13.6/60 = \qty{0.227}{Bq}$ per gram;
$N = \mathcal A_0/\lambda = 0.227/\num{3.8e-12} \approx \num{5.9e10}$ atomen
per gram.

\textbf{5.} $(1/12) \times \num{6.02e23} = \num{5.0e22}$ koolstofatomen; het
aandeel is $\num{5.9e10}/\num{5.0e22} \approx \num{1.2e-12}$ --- één op
$10^{12}$, zoals beloofd.

\textbf{6.} Uit $\mathcal A = \mathcal A_0 e^{-\lambda t}$ volgt
$\lambda t = \ln(\mathcal A_0/\mathcal A)$, en met $\lambda = \ln 2/T_{1/2}$
wordt dat $t = T_{1/2}\ln(\mathcal A_0/\mathcal A)/\ln 2$.

\textbf{7.} $t = \num{5730} \times \ln(13.6/9.1)/\ln 2 =
\num{5730} \times 0.580 \approx \num{3.3e3}$ jaar.

\textbf{8.} De dood van het hout --- het vellen van de boom, niet het
snijwerk; een beeld uit oud timmerhout (of een vervalsing uit antiek hout)
ligt vóór of na de datum van zijn materiaal.

\textbf{9.} $t = \num{5730} \times \ln(13.6/10.4)/\ln 2 =
\num{5730} \times 0.387 \approx \num{2.2e3}$ jaar --- volledig in
overeenstemming met een schrijver van tweeëntwintig eeuwen geleden.

\textbf{10.} $\Delta t \approx \Delta\mathcal A/(\lambda\mathcal A)$ met
$\lambda = \ln 2/5730 = \num{1.21e-4}$ per jaar:
$\Delta t = (0.2/10.4)/\num{1.21e-4} \approx \num{1.6e2}$ jaar --- een datum
die tot op een eeuw of twee klopt.

\textbf{11.} $t = \num{5730} \times \ln(13.6/13.4)/\ln 2 \approx 120$ jaar
--- binnen de onzekerheid van $\pm 160$ jaar niet van modern te onderscheiden:
het perkament is recent, de ``antiquiteit'' een vervalsing.

\textbf{12.} $13.6/2^n < 0.2$ vraagt $2^n > 68$: $n = 7$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}
(want $2^7 = 128$), ongeveer $\num{4.0e4}$ jaar.

\textbf{13.} Tien \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}: ongeveer \num{5.7e4} jaar --- de
praktische horizon, zo'n zestigduizend jaar.

\textbf{14.} $\num{6.6e7}/\num{5730} \approx \num{11500}$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}. De
\num{5.9e10} atomen van één gram zijn uitgeput zodra
$2^n > \num{5.9e10}$, oftewel $n \approx 36$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}
($\approx \num{2e5}$ jaar): lang vóór de ouderdom van de dinosauriërs is er
geen enkel \omterm{def:g11:fundamental-interactions:ladder}{atoom} koolstof-14 meer over --- er valt niets te tellen.

\textbf{15.} Een veel tragere klok: kalium-40 of uranium-238, met
\omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} van miljarden jaren.

\textbf{16.} Gesmolten lava laat het argon wegborrelen, dus houdt het vers
gestolde kristal kalium maar nul argon vast: de verhouding start bij het
stollen op $0$ --- de klok staat op nul.

\textbf{17.} Als in de oefening: $N_{\mathrm K} = N_0 e^{-\lambda t}$ en
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, dus
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.

\textbf{18.} $e^{\lambda t} = 1.60$:
$t = \num{1.25e9} \times \ln(1.60)/\ln 2 = \num{1.25e9} \times 0.678
\approx \num{8.5e8}$ jaar.

\textbf{19.} Nee, twee keer niet: op \num{8.5e8} jaar heeft koolstof-14
$10^5$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} gelopen --- stilte; op \num{3.3e3} jaar heeft
kalium-40 \num{2.6e-6} \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd} gelopen --- de argonverhouding
$\approx \lambda t \sim \num{2e-6}$ is onmeetbaar klein. Elke klok leest
alleen van ongeveer een tiende \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd} tot ongeveer tien.

\textbf{20.} Beeld $\approx \num{3.3e3}$ jaar, papyrus $\approx \num{2.2e3}$
jaar, gesteente $\approx \num{8.5e8}$ jaar --- drie klokken, één wet:
$N = N_0\,e^{-\lambda t}$, achterstevoren gelezen.
\end{solution}
