\section*{अध्याय \ref{ch:g12:doppler-effect} --- डॉप्लर प्रभाव}

\begin{solution}{exo:g12:doppler-effect:1}
(क) $f' = 700 \times 340/315 \approx \qty{756}{Hz}$।
(ख) $f' = 700 \times 340/365 \approx \qty{652}{Hz}$।
(ग) \qty{700}{Hz}: चालक और सायरन साथ-साथ चलते हैं, उनकी दूरी कभी बदलती ही नहीं।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:2}
\emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{प्रबलता}} इसलिए मंद पड़ती है कि तरंग फैल जाती है --- यह
दूरी पर निर्भर करती है। और \emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{तारत्व}} पास आने से दूर जाने तक गिर
जाता है --- यह \omterm{def:g12:doppler-effect:doppler}{डॉप्लर प्रभाव} है, जो इस पर निर्भर करता है कि दूरी कितनी
तेज़ी से बदल रही है। दो स्वतंत्र प्रभाव।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:3}
चलता प्रेक्षक: $f' = 700\,(1 + 6.0/340) \approx \qty{712}{Hz}$; दूर जाते समय $700\,(1 - 6.0/340) \approx \qty{688}{Hz}$। सायरन हवा में \omterm{def:g10:relative-motion:frame}{विराम} में है, इसलिए
तरंग का ढर्रा अछूता रहता है; उसमें से दौड़ता केवल प्रेक्षक है।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:4}
आगे: $\lambda' = 310/400 = \qty{0.775}{m}$; पीछे: $370/400 = \qty{0.925}{m}$; \omterm{def:g10:relative-motion:frame}{विराम} में: $340/400 = \qty{0.85}{m}$। छोटा \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} उसे
मिलता है जिसकी ओर गाड़ी बढ़ रही है, लंबा उसे जिसे वह छोड़कर जा रही है।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:5}
$\Delta f/f \approx 5.0/340 \approx 1.5\%$: $\Delta f \approx \qty{6.5}{Hz}$, इसलिए लगभग \qty{446}{Hz} --- यानी अर्धस्वर का चौथाई। कोई संगीतकार उसे
बस-बस पकड़ लेगा।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:6}
दीवार $f\,v/(v - v_b)$ पाती है (चलता स्रोत); और \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} की ओर बढ़ता चमगादड़,
जो अब चलता प्रेक्षक है, उसका $(1 + v_b/v)$ गुना सुनता है: $f' = f(v + v_b)/(v - v_b) = 50.0 \times
346/334 \approx \qty{51.8}{kHz}$, यानी $+\qty{1.8}{kHz}$ का
खिसकाव। यह रडार-बंदूक़ वाला दुहरा खिसकाव ही है, बस उत्सर्जक और परावर्तक की
भूमिकाएँ बदल गई हैं: मायने केवल पास आने की चाल रखती है।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:7}
$u = c\,\Delta f/(2f) = \num{3.00e8} \times 4000/(2 \times \num{24.15e9}) \approx
\qty{24.8}{m/s} \approx \qty{89}{km/h}$। \qty{50}{km/h} ($\qty{13.9}{m/s}$) पर: $\Delta f = 2 \times 13.9 \times \num{24.15e9}/\num{3.00e8} \approx \qty{2.2}{kHz}$।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:8}
$u = v\,\Delta f/(2f) = 1540 \times 2600/\num{1.0e7} \approx \qty{0.40}{m/s}$। लंबवत बहाव का \omterm{def:g12:doppler-effect:radial}{त्रिज्य वेग} शून्य होता है, इसलिए खिसकाव बिलकुल
भी नहीं मिलता।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:9}
(क) स्रोत: $f' = 1000 \times 340/306 \approx \qty{1111}{Hz}$। (ख) प्रेक्षक: $f' = 1000 \times 1.1 = \qty{1100}{Hz}$। धीमी गति \qty{1100}{Hz} बताती है: प्रेक्षक
के लिए ठीक-ठीक सही, और स्रोत के लिए \qty{11}{Hz} कम --- यानी $x^2$ कोटि का सुधार,
जो $x = 0.1$ के लिए $f$ का $1\%$ है।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:10}
$f = 2 \times 465 \times 415/880 \approx \qty{439}{Hz}$; $v_s = 340 \times 50/880 \approx \qty{19.3}{m/s} \approx \qty{70}{km/h}$। सच्ची $f$ \omterm{def:g12:sound-acoustics:harmonics}{संनादी} माध्य है: पास आना
आवृत्ति को जितना चढ़ाता है, दूर जाना उतना नहीं गिराता, इसलिए $f$
समांतर माध्य से नीचे बैठती है (\qty{440}{Hz})।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:11}
\omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य} बड़ा है: \omterm{def:g12:doppler-effect:redshift}{लाल विस्थापन}, यानी तारा दूर जा रहा है।
$v_r = c\,\Delta\lambda/\lambda =
\num{3.00e8} \times 0.2/656.3 \approx \qty{9.1e4}{m/s} \approx \qty{91}{km/s}$।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:12}
$\Delta\lambda = \lambda K/c = 500 \times 55/\num{3.00e8} \approx \qty{9.2e-5}{nm}$; $\Delta\lambda/\lambda = 55/\num{3.00e8} \approx \num{1.8e-7}$। पूरा संकेत एक करोड़ में दो भाग का है: $10^{-7}$ से अधिक
खिसक जाने वाला वर्णक्रममापी उसे दफ़्न कर देगा।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:13}
$v_r = 0.024\,c \approx \qty{7.2e6}{m/s}$। चालें दूरी के समानुपाती, और आकाश की हर दिशा में एक-सी: यानी सारी
दूरियाँ एकसार खिंच रही हैं --- बिना किसी विशेषाधिकारी केंद्र वाला फैलता
ब्रह्मांड। $2.4\%$ पर सुधार $(v_r/c)^2 \approx
\num{6e-4}$ की कोटि के हैं: अब भी भरोसे लायक़।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:14}
दोनों किनारे उलटी दिशाओं में खिसकते हैं, इसलिए पूरी चकती की औसत रेखा
\emph{चौड़ी} होती है, खिसकती नहीं: पूरी चौड़ाई $2\lambda v/c = 2 \times 656.3 \times \num{2.0e3}/\num{3.00e8}
\approx \qty{8.8e-3}{nm}$ (लगभग $\pm\qty{4.4e-3}{nm}$)।
\end{solution}

\begin{solution}{exo:g12:doppler-effect:15}
(क) कार के आगे शिखरों का अंतराल $(v - v_s)T$ है; ट्रक $v_o$ से भागता है और उनसे
सापेक्ष चाल $v - v_o$ पर मिलता है: $f' = (v - v_o)/\bigl((v - v_s)T\bigr) =
f(v - v_o)/(v - v_s)$। (ख) $f' = 700 \times 310/300 \approx \qty{723}{Hz}$। (ग) $v_o = v_s$ रखने पर
$f' = f$ मिलता है: दूरी अचर है, और दूरी न बदले तो परिभाषा से ही कोई डॉप्लर
खिसकाव नहीं होता।
\end{solution}

\begin{solution}{pb:g12:doppler-effect:1}
\textbf{1.} हर शिखर उसी बिंदु से फैलता है जहाँ वह छूटा था: इसलिए शिखर चलते
भोंपू के आगे सट जाते हैं और पीछे खिंच जाते हैं, और पास आता फ़ोन उनसे दूर जाते
फ़ोन ($f_2$) की तुलना में तेज़ी से मिलता है ($f_1$)।

\textbf{2.} $f_1 = f\,\dfrac{v}{v - u}$, \quad $f_2 = f\,\dfrac{v}{v + u}$.

\textbf{3.} $f_1 - f_2 = \frac{2fv\,u}{v^2 - u^2}$ और $f_1 + f_2 =
\frac{2fv^2}{v^2 - u^2}$; भाग देने पर $u = v\,\frac{f_1 - f_2}{f_1 + f_2} =
340 \times 60/882 \approx \qty{23}{m/s} \approx \qty{83}{km/h}$।

\textbf{4.} $\dfrac{2f_1 f_2}{f_1 + f_2} = f$ (साझा गुणक कट जाता है): $f = 2 \times 471 \times 411/882 \approx \qty{439}{Hz}$ --- यानी \omterm{def:g12:sound-acoustics:harmonics}{संनादी}
माध्य, जो औसत \qty{441}{Hz} से नीचे है क्योंकि पास आना $f$ को जितना चढ़ाता है,
दूर जाना उतना नहीं काटता।

\textbf{5.} आगे: $(v - u)/f = 316.9/439 \approx \qty{0.72}{m}$; पीछे: $363.1/439 \approx \qty{0.83}{m}$; \omterm{def:g10:relative-motion:frame}{विराम} में: $340/439 \approx \qty{0.77}{m}$।

\textbf{6.} एक बार तरंग पाते चलते \emph{प्रेक्षक} के रूप में, और एक
बार जो पाया उसे फिर छोड़ते चलते \emph{स्रोत} के रूप में।

\textbf{7.} $f' = f\,(1 + u/c)/(1 - u/c) \approx f(1 + 2u/c)$, इसलिए $\Delta f = f' - f \approx \dfrac{2u}{c}\,f$।

\textbf{8.} $u = c\,\Delta f/(2f) = \num{3.00e8} \times 3550/(2 \times \num{24.125e9})
\approx \qty{22}{m/s}$।

\textbf{9.} $22.1 \times 3.6 \approx \qty{79}{km/h}$: यानी \qty{80}{km/h} की सीमा से ज़रा-सा नीचे।

\textbf{10.} प्रति \unit{km/h} $\Delta f = 2f/(3.6\,c) \approx \qty{45}{Hz}$। \omterm{rem:g10:signals-and-waves:honest}{प्रतिध्वनि} को छोड़ी गई
तरंग से टकराने पर केवल \emph{अंतर} बचता है: एक श्रव्य
आवृत्ति, जिसे गिनना बच्चों का खेल है, भले वह वाहक का $10^{-7}$ भर अंश
हो।

\textbf{11.} रात-दर-रात ज्ञात \omterm{def:g12:sound-acoustics:timbre}{वर्णक्रम} रेखाओं के
\omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य}; और फिर $v_r = c\,\Delta\lambda/\lambda$।

\textbf{12.} $K = \qty{55}{m/s}$, $P = \qty{4.2}{days}$।

\textbf{13.} $\Delta\lambda = 550 \times 55/\num{3.00e8} \approx \qty{1.0e-4}{nm}$, $\Delta\lambda/\lambda \approx \num{1.8e-7}$ --- यानी पचास लाख में एक भाग, जो रेलगाड़ी के
$7\%$ खिसकाव से लगभग $\num{4e5}$ गुना छोटा है।

\textbf{14.} तारा और ग्रह एक-दूसरे को बराबर खींचते हैं, इसलिए दोनों अपने साझा
\omterm{def:g11:forces-and-motion:com}{द्रव्यमान केंद्र} की परिक्रमा करते हैं: तारा ठहरा रह ही नहीं
सकता। और ज्यावक्र जैसा $v_r$ किनारे से देखी गई \omterm{def:g11:forces-and-motion:ucm}{एकसमान वृत्तीय गति}
है --- यानी वृत्तीय कक्षा।

\textbf{15.} तारे और पृथ्वी की दूरी, और इसलिए \omterm{def:g12:mechanical-waves:wavelength}{तरंगदैर्घ्य}, केवल
दृष्टि-रेखा वाला घटक बदलता है। कक्षा सामने चपटी पड़ी हो तो हर समय $v_r = 0$:
कोई संकेत ही नहीं।

\textbf{16.} $P = \qty{3.63e5}{s}$: $r^3 = \num{6.67e-11} \times \num{2.0e30} \times
(\num{3.63e5})^2/(4\pi^2) \approx \num{4.4e29}$, इसलिए $r \approx \qty{7.6e9}{m}$ --- यानी पृथ्वी--सूर्य दूरी का
लगभग $1/20$।

\textbf{17.} $V = 2\pi r/P = 2\pi \times \num{7.6e9}/\num{3.63e5} \approx
\qty{1.3e5}{m/s}$।

\textbf{18.} $m = M K/V = \num{2.0e30} \times 55/\num{1.3e5} \approx \qty{8.3e26}{kg}$।

\textbf{19.} लगभग $0.4$ बृहस्पति-द्रव्यमान ($\approx 140$ पृथ्वियाँ), और वह भी
बुध से $8$ गुना पास परिक्रमा करता हुआ: यानी अपने तारे से सिंकता एक गैस-दानव
--- ``तप्त बृहस्पति''।

\textbf{20.} तिरछापन गति का कुछ हिस्सा छिपा लेता है: नापा गया $K$ केवल
त्रिज्य हिस्सा है, इसलिए सच्चा द्रव्यमान \emph{बड़ा} होगा --- \qty{8.3e26}{kg} एक
न्यूनतम है। पचास लाख में एक भाग के \omterm{def:g10:signals-and-waves:period}{आवर्ती} खिसकाव ने अभी-अभी उस ग्रह
की कक्षा, चाल और न्यूनतम द्रव्यमान सौंप दिए जिसे किसी ने कभी नहीं
देखा।
\end{solution}
