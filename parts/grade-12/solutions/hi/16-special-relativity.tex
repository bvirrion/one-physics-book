\section*{अध्याय \ref{ch:g12:special-relativity} --- विशिष्ट आपेक्षिकता: काल-विस्तारण}

\begin{solution}{exo:g12:special-relativity:1}
$\gamma = 1/\sqrt{1 - (v/c)^2}$: $1.01$ ($0.1c$); $1.25$ ($0.6c$); $1.67$ ($0.8c$); $10.0$ ($0.995c$)।
\end{solution}

\begin{solution}{exo:g12:special-relativity:2}
$\gamma = 1.25$, इसलिए ज़मीन $1.25 \times 10.0 =
\qty{12.5}{s}$ नापती है। जहाज़ का \qty{10.0}{s} ही \omterm{def:g12:special-relativity:proper-time}{निजी काल} है:
ताल-यंत्र (एक ही घड़ी) दोनों धड़कनों पर मौजूद है।
\end{solution}

\begin{solution}{exo:g12:special-relativity:3}
$v = c\sqrt{1 - 1/\gamma^2}$: $\gamma = 2$ से $v = 0.866c$ मिलता है; और $\gamma = 10$ से $v = 0.995c$।
\end{solution}

\begin{solution}{exo:g12:special-relativity:4}
$v = \qty{36.1}{m/s}$, $v/c = \num{1.2e-7}$; $\gamma - 1 \approx (\num{1.2e-7})^2/2 \approx \num{7.2e-15}$। एक \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} की हानि $1/\num{7.2e-15} \approx \qty{1.4e14}{s}$ बाद --- यानी लगभग चालीस लाख
वर्ष कार चलाने पर।
\end{solution}

\begin{solution}{exo:g12:special-relativity:5}
यात्री आगे वाली चमक की ओर बढ़ रहा है, इसलिए वह उस तक पहले पहुँचती है; और दूसरे
अभिगृहीत से \emph{उसके तंत्र में} दोनों चमकें आधी गाड़ी की बराबर दूरियाँ उसी
$c$ पर तय करती हैं, इसलिए पहले पहुँचने का अर्थ पहले गिरना है। युगपतता तंत्र
पर निर्भर है: हर फ़ैसला अपने-अपने तंत्र में सही है।
\end{solution}

\begin{solution}{exo:g12:special-relativity:6}
$\gamma = 1/\sqrt{1 - 0.99^2} = 7.09$; प्रयोगशाला में जीवनकाल $7.09 \times \num{2.6e-8} = \qty{1.8e-7}{s}$। चिरसम्मत \omterm{def:g12:projectile-motion:range}{परास} $0.99 \times \num{3.00e8} \times \num{2.6e-8} \approx \qty{7.7}{m}$; और असली
$7.09 \times 7.7 \approx \qty{55}{m}$।
\end{solution}

\begin{solution}{exo:g12:special-relativity:7}
$v = c\sqrt{1 - 1/2.25} \approx 0.75c$ पर $\gamma = 1.5$; $\approx 0.94c$ पर $\gamma = 3$। अनंतस्पर्शी: $v \to c$ होते-होते $\gamma$ (और
और चाल का ऊर्जा-दाम) \omterm{def:g11:lenses-and-eye:lens}{अपसारी} हो जाता है --- यानी कोई परिमित मेहनत किसी
द्रव्यमान वाले पिंड को प्रकाश की चाल तक नहीं ला सकती।
\end{solution}

\begin{solution}{exo:g12:special-relativity:8}
$\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$; $0.8c$ पर $\gamma = 5/3$, इसलिए $\Delta t = \qty{1.7e-8}{s}$।
\end{solution}

\begin{solution}{exo:g12:special-relativity:9}
\omterm{def:g12:special-relativity:proper-time}{निजी काल} उस घड़ी का है जो दोनों घटनाओं पर मौजूद हो: (क) पहनने
वाला; (ख) म्यूऑन; (ग) चालक --- क्योंकि प्रस्थान और आगमन, दोनों पर केवल यान ही
मौजूद है।
\end{solution}

\begin{solution}{exo:g12:special-relativity:10}
$\gamma - 1 \approx 7700^2/(2 \times \num{9.0e16}) \approx
\num{3.3e-10}$; और \qty{1.58e7}{s} में अंतरिक्ष यात्री $\num{3.3e-10} \times \num{1.58e7} \approx \qty{5.2e-3}{s}$ पीछे रह जाता है --- यानी लगभग
पाँच मिलीसेकंड छोटा।
\end{solution}

\begin{solution}{exo:g12:special-relativity:11}
$E = \num{1.0e-3} \times (\num{3.00e8})^2 = \qty{9.0e13}{J}$। \qty{1.0}{GW} का संयंत्र प्रतिदिन $\num{1.0e9} \times 86400 \approx
\qty{8.6e13}{J}$ देता है --- यानी एक ग्राम उसके एक दिन
के उत्पादन के बराबर है, और $\num{9.0e13}/\num{8.5e5} \approx \num{e8}$: दस करोड़ तेज़ भागती कारें।
\end{solution}

\begin{solution}{exo:g12:special-relativity:12}
वर्ग कीजिए और उलटिए: $\gamma^2(1 - v^2/c^2) = 1$, इसलिए $v^2/c^2 = 1 - 1/\gamma^2$, यानी $v = c\sqrt{1 - 1/\gamma^2}$। और $\gamma = 30$ के लिए:
$v = c\sqrt{1 - 1/900} = 0.99944c$।
\end{solution}

\begin{solution}{exo:g12:special-relativity:13}
संकुचित मोटाई $L = \qty{15}{km}/30 = \qty{500}{m}$; और उससे गुज़रने का समय $500/(0.99944 \times \num{3.00e8}) \approx \qty{1.7e-6}{s} <
\qty{2.2}{\micro s}$। दोनों तंत्र सहमत हैं कि
म्यूऑन ज़मीन तक पहुँच जाता है --- एक इसका दोष खिंचे हुए जीवनकाल पर मढ़ता है,
दूसरा सिकुड़े हुए वायुमंडल पर।
\end{solution}

\begin{solution}{exo:g12:special-relativity:14}
प्रयोगशाला: $3200/\num{3.00e8} \approx \qty{1.1e-5}{s}$ (चाल $\approx c$)। इलेक्ट्रॉन के तंत्र में: नली सिकुड़कर $3200/\num{1.0e5} = \qty{3.2}{cm}$
रह जाती है, जिसे $\num{1.1e-5}/\num{1.0e5} \approx \qty{1.1e-10}{s}$ में पार कर लिया जाता है।
\end{solution}

\begin{solution}{exo:g12:special-relativity:15}
पृथ्वी: $4.2/0.9 \approx \qty{4.7}{years}$। और $\gamma = 1/\sqrt{1 - 0.81} = 2.29$ के साथ यान की घड़ी $4.7/2.29 \approx \qty{2.0}{years}$ दर्ज करती है।
\end{solution}

\begin{solution}{pb:g12:special-relativity:1}
\textbf{1.} $c\,\Delta t_0 = \num{3.00e8} \times \num{2.2e-6} \approx
\qty{660}{m}$।

\textbf{2.} $\num{15000}/660 \approx 23$ जीवनकाल।

\textbf{3.} $\mathrm{e}^{-23} \approx \num{e-10}$: यानी दस अरब में एक से भी कम म्यूऑन पहुँचना चाहिए, जबकि
प्रवाह प्रति \unit{cm^2} प्रति मिनट एक है --- यानी आसमान चिरसम्मत भौतिकी का सीधा
खंडन कर देता है।

\textbf{4.} ज़मीनी जीवनकाल $30 \times \qty{2.2}{\micro s} =
\qty{66}{\micro s}$; \omterm{def:g12:projectile-motion:range}{परास} $\approx \num{3.00e8} \times \num{6.6e-5}
\approx \qty{20}{km} > \qty{15}{km}$: आगमन की व्याख्या हो गई।

\textbf{5.} $v = c\sqrt{1 - 1/900} = 0.99944c$।

\textbf{6.} $\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$।

\textbf{7.} क्षैतिज $v\,\Delta t/2$; \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} $L$; और तिरछा आधा-पथ $c\,\Delta t/2$ नापता है,
क्योंकि दूसरा अभिगृहीत ज़मीनी तंत्र में भी \omterm{def:g12:quantum-world:photon}{फ़ोटॉन} की चाल $c$
पर जमा देता है।

\textbf{8.} $(c\,\Delta t/2)^2 = L^2 + (v\,\Delta t/2)^2$ से $\Delta t^2(c^2 - v^2) = 4L^2$ मिलता है, इसलिए $\Delta t =
(2L/c)/\sqrt{1 - v^2/c^2} = \gamma\,\Delta t_0$।

\textbf{9.} $\gamma = 1/\sqrt{1 - 0.36} = 1.25$; $\Delta t = \qty{1.25e-8}{s}$।

\textbf{10.} जहाज़ वाला: उसकी एक ही घड़ी दोनों टिकों पर मौजूद है, इसलिए वही
\omterm{def:g12:special-relativity:proper-time}{निजी काल} पढ़ती है; ज़मीन को दो तालमेल बिठाई हुई घड़ियाँ चाहिए।

\textbf{11.} $\gamma = 1/\sqrt{1 - 0.64} = 1/0.6 = 5/3 \approx 1.67$।

\textbf{12.} $10.0/(5/3) = \qty{6.0}{years}$।

\textbf{13.} $0.8c$ पर बाहर की ओर $\qty{5.0}{years}$: यानी $4.0$ \omterm{def:g10:orders-of-magnitude:lightyear}{प्रकाश वर्ष}।

\textbf{14.} $10.0 - 6.0 = \qty{4.0}{years}$: अंतरिक्ष यात्री अपनी जुड़वाँ से चार वर्ष छोटी है।

\textbf{15.} स्थिति सममित नहीं है: मुड़ती केवल यात्री है --- वह त्वरित होती है
और \omterm{def:g12:special-relativity:inertial}{जड़त्वीय तंत्र} बदल देती है, जबकि पृथ्वी वाली जुड़वाँ एक ही तंत्र में
बनी रहती है --- इसलिए छोटा समय केवल उसी की घड़ी दर्ज करती है।

\textbf{16.} $v = 2\pi \times \num{2.66e7}/\num{43200} \approx
\qty{3.9e3}{m/s}$।

\textbf{17.} $\gamma - 1 \approx (\num{3.87e3})^2/(2 \times
\num{9.0e16}) \approx \num{8.3e-11}$।

\textbf{18.} प्रतिदिन $\num{8.3e-11} \times \num{86400} \approx \qty{7.2e-6}{s}
= \qty{7.2}{\micro s}$ धीमी।

\textbf{19.} प्रतिदिन $c \times \num{7.2e-6} \approx \qty{2.2}{km}$ की स्थिति-त्रुटि।

\textbf{20.} शुद्ध बहकाव प्रतिदिन $45 - 7.2 \approx \qty{38}{\micro s}$ तेज़; और त्रुटि प्रतिदिन $c \times \num{38e-6} \approx \qty{11}{km}$ ---
यानी आपेक्षिकता का वह बिल जो हर GPS घड़ी पहले ही चुका देती है, ज़मीन पर सुर से
उतारकर, ताकि कक्षा में वह सही टिके।
\end{solution}
