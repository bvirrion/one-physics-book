\section*{अध्याय \ref{ch:g12:newtons-laws} --- न्यूटन के नियम}

\begin{solution}{exo:g12:newtons-laws:1}
(क) $p = 70 \times 10 = \qty{7.0e2}{kg.m/s}$।
(ख) $v = \qty{13.9}{m/s}$: $p \approx \qty{1.8e4}{kg.m/s}$।
(ग) $p = 0.0080 \times 800 = \qty{6.4}{kg.m/s}$। कार $>$ धावक $>$ गोली: द्रव्यमान की पाँच कोटियाँ इतनी बड़ी हैं
कि चाल उन्हें ख़रीद नहीं सकती।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:2}
$N = m(g + a_z)$: (क) $70 \times 11.01 = \qty{771}{N}$; (ख) $a_z = 0$, इसलिए $N = mg = \qty{687}{N}$ --- यानी पहला नियम: अचर वेग,
बल संतुलित; (ग) $70 \times 8.61 = \qty{603}{N}$।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:3}
$a = g\sin 30^\circ = \qty{4.9}{m/s^2}$; $v = \sqrt{2 \times 4.905 \times 3.0} = \qty{5.4}{m/s}$। द्रव्यमान $ma$ और $mg\sin\alpha$ के बीच कट जाता है।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:4}
पलटाव: $\Delta p = 0.058 \times (25 + 15) = \qty{2.3}{kg.m/s}$; $F = 2.32/0.0050 = \qty{4.6e2}{N}$ --- यानी \qty{0.57}{N} के भार का लगभग $800$ गुना, और
इसीलिए संपर्क के दौरान गुरुत्व को अनदेखा किया जाता है।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:5}
भार (पृथ्वी किताब पर, नीचे): साथी = किताब पृथ्वी को ऊपर खींचती है।
\omterm{def:g12:newtons-laws:friction}{अभिलंब बल} (मेज़ किताब पर, ऊपर): साथी = किताब मेज़ को नीचे दबाती है।
भार और \omterm{def:g12:newtons-laws:friction}{अभिलंब बल} \emph{एक ही} पिंड पर लगते हैं और दो अलग अंतःक्रियाओं के हैं
--- यहाँ बराबर केवल इसलिए हैं कि $\vect a = \vect 0$।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:6}
$N = mg = \qty{490}{N}$। (क) $F > \mu_s N = \qty{221}{N}$। (ख) $a = (220.7 - 0.35 \times 490.5)/50 = \qty{0.98}{m/s^2}$। (ग) $F = \mu_k N = \qty{172}{N}$।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:7}
(क) तब सरकता है जब $\tan\alpha > \mu_s$: $\alpha = \arctan 0.45 \approx
24^\circ$। (ख) $a = g(\sin 30^\circ - 0.35\cos 30^\circ) =
9.81 \times 0.197 = \qty{1.9}{m/s^2}$। (ग) हाँ: सरकना शुरू होते ही
\omterm{def:g10:inertia:inventory}{घर्षण} गिरकर $\mu_k < \mu_s$ रह जाता है, इसलिए $a = g\cos\alpha\,(\tan\alpha - \mu_k) \approx \qty{0.9}{m/s^2} > 0$।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:8}
$a = m_2 g/(m_1 + m_2) = 19.62/5.0 = \qty{3.9}{m/s^2}$; $T = m_1 a = \qty{11.8}{N}$। लटकता गुटका नीचे की ओर त्वरित होता है, इसलिए उस पर
\omterm{def:g11:forces-and-motion:netforce}{परिणामी बल} नीचे की ओर है: यानी $T$ को $m_2 g$ से कम रहना ही होगा।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:9}
$v_{\max} = \sqrt{\mu_s g R}$: सूखा $\sqrt{0.70 \times 9.81 \times 50} = \qty{18.5}{m/s} \approx
\qty{67}{km/h}$; गीला $\qty{14.0}{m/s} = \qty{50}{km/h}$। $mv^2/R$ और $\mu_s mg$, दोनों में $m$ है: वह
कट जाता है।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:10}
\omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर}: $T\cos\theta = mg$, इसलिए $T = 2.943/\cos 25^\circ = \qty{3.2}{N}$। त्रिज्या $R = L\sin\theta =
\qty{0.507}{m}$; त्रिज्या के अनुदिश: $T\sin\theta = mv^2/R$,
इसलिए $v = \sqrt{gR\tan\theta} = \sqrt{9.81 \times 0.507 \times 0.466}
= \qty{1.5}{m/s}$।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:11}
$mg = \qty{49.05}{N}$ के साथ $a_z = (N - mg)/m$: $+\qty{1.0}{m/s^2}$, $0$, $-\qty{1.0}{m/s^2}$। जैसे ऊपर जाती एक यात्रा: चाल
बढ़ाना, एक-सी चाल, फिर ब्रेक। और हाँ: पहले से नीचे उतरते हुए घुसने पर वही
पाठ्यांक ब्रेक, एक-सी चाल, और फिर नीचे की ओर तेज़ होना कहते हैं --- तराज़ू
त्वरण पढ़ता है, वेग नहीं।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:12}
(क) $0 = 60V + 4.0 \times 8.0$: $V = \qty{-0.53}{m/s}$। (ख) गेंद ज़मीन के सापेक्ष $8.0 + V$ पर: $0 = 60V + 4.0(8.0 + V)$, इसलिए $V = -32/64 = \qty{-0.50}{m/s}$
और गेंद \qty{7.5}{m/s} से उड़ती है। (ग) उसके हाथों पर गेंद का धक्का --- यानी उसके फेंकने
का तीसरे-नियम वाला साथी।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:13}
घर्षणहीन: $N\cos\theta = mg$ (\omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर}) और $N\sin\theta =
mv^2/R$ (त्रिज्य); भाग देने पर $\tan\theta = v^2/(gR) = 400/784.8$: $\theta \approx 27^\circ$।
धीमे जाने पर कार झुकाव से नीचे फिसलना चाहती है, इसलिए \omterm{def:g10:inertia:inventory}{घर्षण} को ढलान
पर ऊपर की ओर होना पड़ता है; तेज़ जाने पर उलटा।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:14}
(क) हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $250 \times 2500 = \qty{6.3e5}{kg.m/s}$: \omterm{def:g10:inertia:inventory}{प्रणोद} $\qty{6.3e5}{N}$। (ख) $P = \qty{4.9e5}{N}$; $a = (6.25 - 4.905) \times 10^5 / \num{5.0e4} = \qty{2.7}{m/s^2}$। (ग) वह अपने ही
निकास पर धक्का देता है (तीसरा नियम), हवा पर नहीं।
\end{solution}

\begin{solution}{exo:g12:newtons-laws:15}
(क) $m_2 g > \mu_s m_1 g$: $m_2 > 0.35 \times 4.0 = \qty{1.4}{kg}$। (ख) $a = (m_2 - \mu_k m_1)\,g/(m_1 + m_2) = 1.0 \times 9.81/6.0 =
\qty{1.6}{m/s^2}$; $T = m_2(g - a) = \qty{16.4}{N}$। (ग) $\mu_k = 0$ से $a = m_2 g/(m_1 + m_2)$ मिलता है: यानी
\cref{exo:g12:newtons-laws:8} वाला सूत्र।
\end{solution}

\begin{solution}{pb:g12:newtons-laws:1}
\textbf{1.} निकाय: केबिन $+$ गाड़ी ($+$ सवारियाँ)। पहाड़ ज़मीन से जुड़ा
है, और बहुत अच्छी परिशुद्धता तक जड़त्वीय है। बल: भार $M\vect g$
(नीचे), पटरी-रस्सी का \omterm{def:g12:newtons-laws:friction}{अभिलंब बल} $\vect N$ (उसके लंबवत), और खिंचाव का
\omterm{def:g10:inertia:inventory}{तनाव} $\vect T$ (ढलान पर ऊपर की ओर)।

\textbf{2.} \omterm{def:g10:relative-motion:frame}{विराम} में पहला नियम: $\sum \vect F = \vect 0$। ढलान के अनुदिश: $T = Mg\sin\alpha$; लंबवत:
$N = Mg\cos\alpha$।

\textbf{3.} $T = 4000 \times 9.81 \times 0.500 = \qty{19.6}{kN}$; $N = 4000 \times 9.81 \times 0.866 = \qty{34.0}{kN}$।

\textbf{4.} ख़ाली: $T = \qty{9.8}{kN}$। संतुलन में हर बल $M$ के समानुपाती है,
इसलिए $T$ भी।

\textbf{5.} ढलान के अनुदिश $Ma = Mg\sin\alpha$: $M$ कट जाता है और ढलान से नीचे की ओर
$a = g\sin\alpha = \qty{4.9}{m/s^2}$।

\textbf{6.} $T' = M(g\sin\alpha + a) = 4000 \times 5.505 =
\qty{22.0}{kN}$।

\textbf{7.} $a/(g\sin\alpha) = 0.60/4.905 \approx 12\%$।

\textbf{8.} $t = v/a = \qty{10}{s}$; $d = v^2/(2a) = \qty{30}{m}$।

\textbf{9.} $\vect a$ का \omterm{def:g10:universal-gravitation:weight}{ऊर्ध्वाधर} घटक: $a\sin\alpha =
\qty{0.30}{m/s^2}$, इसलिए $mg = \qty{785}{N}$ के मुक़ाबले
$N = m(g + a\sin\alpha) = 80 \times 10.11 =
\qty{809}{N}$: यानी $3\%$ अधिक भारी।

\textbf{10.} फ़र्श का \omterm{def:g11:forces-and-motion:friction}{स्थैतिक घर्षण}, क्षैतिज: $ma\cos\alpha = 80 \times 0.52 \approx \qty{42}{N}$। झुक जाने से
फ़र्श का कुल धक्का उसके \omterm{def:g11:forces-and-motion:com}{द्रव्यमान केंद्र} से होकर गुज़रता है, इसलिए वह
उसे उलट नहीं देता।

\textbf{11.} $a_T = dv/dt = 0$ (अचर चाल); $a_N = v^2/R = 36/40 = \qty{0.90}{m/s^2}$, केंद्र की ओर ताका हुआ --- यानी शिखर
पर सीधे नीचे।

\textbf{12.} \omterm{def:g10:refraction:angles}{अभिलंब} अक्ष (नीचे धनात्मक): $Mg - N' = Mv^2/R$, इसलिए $N' = M(g - v^2/R) = 4000 \times 8.91 = \qty{35.6}{kN}$।

\textbf{13.} $N = m(g - v^2/R) = 80 \times 8.91 = \qty{713}{N}$: यानी $v^2/(gR) = 0.90/9.81 \approx 9\%$ हलकी।

\textbf{14.} $v = \sqrt{gR} = \qty{19.8}{m/s} \approx
\qty{71}{km/h}$ पर $N' = 0$ --- यानी यात्रा-चाल का तिगुने से भी अधिक:
गाड़ी रस्सी पर अच्छे-ख़ासे हाशिये के साथ चलती है।

\textbf{15.} \omterm{thm:g12:newtons-laws:third}{न्यूटन का तीसरा नियम}: \qty{35.6}{kN}, जो रस्सी पर (और इसलिए
मीनार पर) नीचे की ओर दबाता है --- \omterm{def:g10:relative-motion:frame}{विराम} वाले भार से ज़रा कम, क्योंकि
मुड़ी हुई उड़ान उसे कुछ राहत दे देती है।

\textbf{16.} $F = \Delta p/\Delta t = 4000 \times 6.0/1.5 =
\qty{16}{kN}$, गति के विपरीत ($a = \qty{4.0}{m/s^2}$)।

\textbf{17.} $f = ma = 80 \times 4.0 = \qty{320}{N}$; और छत $\mu_s mg = 0.40 \times 785 = \qty{314}{N} < \qty{320}{N}$ है: अकेले पैर फिसल जाएँगे --- रेलिंग
पकड़िए।

\textbf{18.} $0 = M_0 V + m(v_{\text{सापेक्ष}} + V)$: $V = -80 \times 2.5/2080 = \qty{-0.096}{m/s}$; और सवारी ज़मीन के सापेक्ष $2.5 - 0.096 \approx \qty{2.4}{m/s}$ से चलती है।

\textbf{19.} राकेट: उसका \omterm{def:g10:inertia:inventory}{प्रणोद} प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} निकाले गए द्रव्यमान गुणा
उसके सापेक्ष निकास-चाल से तय होता है।

\textbf{20.} मिलता सबसे बड़ा \omterm{def:g10:inertia:inventory}{तनाव} चलना शुरू करने वाला $T' = M \times \qty{5.505}{N/kg}$ है
(\omterm{def:g10:relative-motion:frame}{विराम} $Mg\sin\alpha$ देता है, और शिखर उससे भी कम)। सीमा: $T' \leq \qty{20}{kN}$, इसलिए $M \leq 20000/5.505 = \qty{3.63e3}{kg}$:
यानी भार-क्षमता \qty{1.63e3}{kg}, यानी \qty{80}{kg} की $20$ सवारियाँ ($25$ नहीं)।
तख़्ती: \emph{20 व्यक्ति}; तब $M = \qty{3.6e3}{kg}$, $T' = \qty{19.8}{kN}$, और सुरक्षा-गुणक $100/19.8 \approx 5.1 \geq 5$।
\end{solution}
