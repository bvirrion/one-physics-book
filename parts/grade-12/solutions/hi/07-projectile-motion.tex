\end{solution}

\begin{solution}{exo:g12:projectile-motion:15}
$t^* = D/(v_0\cos\alpha)$ पर तीर की ऊँचाई $v_0\sin\alpha\, t^* - \tfrac12 g t^{*2} = D\tan\alpha - \tfrac12 g
t^{*2} = H - \tfrac12 g t^{*2}$ है, जो छोड़े गए बंदर की $y = H - \tfrac12 g t^{*2}$ से ठीक बराबर है।
दोनों अपने गुरुत्व-रहित बिंदुओं से $\tfrac12 gt^2$ नीचे लटके हैं --- इसलिए वे किसी भी
$v_0$ पर मिल जाते हैं।
\end{solution}

\begin{solution}{pb:g12:projectile-motion:1}
\textbf{1.} केवल भार (हवा छोड़ दी गई); \omterm{thm:g12:newtons-laws:second}{न्यूटन का दूसरा नियम}:
$m\vect a = m\vect g$, इसलिए $\vect a = (0,\,-g)$।

\textbf{2.} $v_x = v_0\cos\alpha$, $v_y = v_0\sin\alpha - gt$।

\textbf{3.} $x = v_0\cos\alpha\, t$, $y = v_0\sin\alpha\, t - \tfrac12 gt^2$।

\textbf{4.} $t = x/(v_0\cos\alpha)$: $y = x\tan\alpha - g x^2/(2v_0^2\cos^2\alpha)$।

\textbf{5.} $v_0^2 = \dfrac{g d^2}{2\cos^2\alpha\,(d\tan\alpha -
1.05)} = \dfrac{173.0}{0.826 \times 3.96} = 52.9$: $v_0 = \qty{7.28}{m/s} \approx \qty{26}{km/h}$।

\textbf{6.} $t = d/(v_0\cos\alpha) = 4.20/4.68 = \qty{0.90}{s}$।

\textbf{7.} शिखर $t = v_0\sin\alpha/g = \qty{0.57}{s}$ पर, ऊँचाई $2.00 + (v_0\sin\alpha)^2/2g = 2.00 + 1.58 = \qty{3.58}{m}$; $0.57 < 0.90$: यानी घेरे पर पहुँचते समय
वह उतर रही है।

\textbf{8.} छोड़ने के बिंदु से $y(0.90) = 0.90\tan\ang{50} - 0.182 = \qty{0.89}{m}$ ऊपर, यानी फ़र्श से \qty{2.89}{m} ऊपर:
उँगलियों से \qty{0.29}{m} ऊपर।

\textbf{9.} $v_x = \qty{4.68}{m/s}$; $v_y = 5.57 - 9.81 \times 0.90 = \qty{-3.24}{m/s}$; $v = \qty{5.69}{m/s}$, और क्षैतिज से $\arctan(3.24/4.68) = \ang{34.7}$ नीचे।

\textbf{10.} घेरा एक क्षैतिज छल्ला है: गेंद को उसका तल ऊपर से पार करना ही
होगा। प्रवेश जितना खड़ा, वेग के अनुदिश छल्ले का मुँह उतना ही बड़ा दिखता है
--- यानी चूक के लिए उतना ही अधिक हाशिया।

\textbf{11.} $1.05/26.0 = 4\%$: नगण्य। $y(t_f) = 0$ से $t_f\,(v_0\sin\alpha - \tfrac12 g t_f) = 0$ मिलता है, इसलिए $t_f = 2v_0\sin\alpha/g$।

\textbf{12.} $R = v_0\cos\alpha\, t_f = v_0^2\sin(2\alpha)/g$; और अधिकतम $\alpha = \ang{45}$ पर।

\textbf{13.} $v_0 = \sqrt{gR} = \sqrt{9.81 \times 26.0} =
\qty{16.0}{m/s} \approx \qty{57}{km/h}$।

\textbf{14.} $R_{\max} = 17.5^2/9.81 = \qty{31.2}{m} >
\qty{26.0}{m}$: संभव है, और \omterm{def:g12:projectile-motion:range}{परास} में \qty{5}{m} बचा भी रहता है।

\textbf{15.} $\sin(2\alpha) = 255.1/306.3 = 0.833$: $\alpha = \ang{28.2}$ या \ang{61.8}; \omterm{def:g12:projectile-motion:range}{उड़ान-काल} $2v_0\sin\alpha/g = \qty{1.69}{s}$ और \qty{3.14}{s}। घंटी
\qty{2.0}{s} पर बजनी है, इसलिए केवल चपटा शॉट ही उससे पहले पहुँचता है।

\textbf{16.} $v_0^2 = \dfrac{9.81 \times 36}{2\cos^2\ang{75}\,
(6.0\tan\ang{75} - 1.05)} = \dfrac{353.2}{2.86} = 123.5$: $v_0 = \qty{11.1}{m/s}$।

\textbf{17.} छोड़ने के बिंदु से $y(5.55) = 5.55\tan\ang{75} - 18.26 = \qty{2.45}{m}$ ऊपर, यानी फ़र्श से \qty{4.45}{m} ऊपर: तख़्ते
के ऊपरी किनारे से \qty{0.50}{m} ऊपर।

\textbf{18.} शिखर $2.00 + (11.1\sin\ang{75})^2/2g \approx
\qty{7.9}{m}$; उड़ान $6.0/(11.1\cos\ang{75}) = \qty{2.1}{s}$ --- यानी पूरे दो \omterm{def:g10:orders-of-magnitude:unit}{सेकंड}, और गेंद
\qty{8}{m} ऊपर टँगी हुई।

\textbf{19.} कर्षण हर शॉट को छोटा करता है और उतार को खड़ा; और उसका असर चाल के
साथ बढ़ता है, इसलिए सबसे अधिक नुक़सान \qty{17.5}{m/s} वाले पूरे-मैदान उछाल को होता है
--- उसका असली \omterm{def:g12:projectile-motion:range}{परास} \qty{31}{m} से कुछ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} पीछे रह जाता है और हाशिया खा
जाता है।

\textbf{20.} \emph{आख़िरी घंटी वाला फ़ैसला}: \qty{26.0}{m} वाला शॉट \qty{16.0}{m/s} माँगता
है और बाँह लगभग \qty{17.5}{m/s} देती है, यानी \qty{5}{m} ($\approx 20\%$) का परास-हाशिया,
जिसे वायु-प्रतिरोध छाँटता तो है पर मिटाता नहीं। संभव --- और ठीक इसीलिए हर
मौसम में दो-चार बार यह शॉट टोकरी में जा भी गिरता है।
\end{solution}
