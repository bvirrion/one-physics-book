\section*{अध्याय \ref{ch:g12:oscillators-and-time} --- यांत्रिक दोलित्र और समय का मापन}

\begin{solution}{exo:g12:oscillators-and-time:1}
(क) $T_0 = 2\pi\sqrt{1.00/9.81} = \qty{2.01}{s}$; $f_0 = 1/T_0 = \qty{0.50}{Hz}$। (ख) $T_0 \propto \sqrt{L}$: \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} दुगुना हो जाता है।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:2}
हृदय: $f = 72/60 = \qty{1.2}{Hz}$, $T = \qty{0.83}{s}$। पंख: $T = 1/30 = \qty{33}{ms}$। छाती का उठाव \emph{आयाम} है।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:3}
$T_0 = 2\pi\sqrt{0.10/25} = \qty{0.40}{s}$, $f_0 = \qty{2.5}{Hz}$। $m$ दुगुना करने पर: $T_0 \times \sqrt{2} \approx 1.4$।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:4}
\omterm{def:g12:oscillators-and-time:damping}{छद्म-आवर्तकाल} \qty{2.0}{s}। हर \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} पर अनुपात $1.5/2.0 = 1.13/1.5 = 0.85/1.13
= 0.75$; अगला
उच्चिष्ठ $0.85 \times 0.75 \approx
\qty{0.64}{cm}$।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:5}
लोलक: $T_0 \propto 1/\sqrt{g}$, इसलिए $\times\sqrt{9.81/1.62} =
2.46$ (धीमा)। स्प्रिंग--द्रव्यमान: अपरिवर्तित --- $2\pi\sqrt{m/k}$
में $g$ है ही नहीं।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:6}
(क) $[L/g] = \unit{m}/(\unit{m/s^2}) = \unit{s^2}$; और उसका वर्गमूल एक समय है। (ख) \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} केवल $m$ में
आता है: उसे काटने वाला और कुछ है ही नहीं, इसलिए $m$ आ ही नहीं सकता।
(ग) कभी नहीं: $2\pi$ जैसी शुद्ध संख्याएँ विमाओं को दिखती ही नहीं।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:7}
$\frac{dx}{dt} = -\frac{2\pi}{T_0}A\sin(2\pi t/T_0 + \varphi)$, इसलिए $\frac{d^2x}{dt^2} = -(2\pi/T_0)^2 x(t)$। सभी $t$ के लिए $-(k/m)x$ से समता ठीक तभी बनती है जब
$(2\pi/T_0)^2 = k/m$, यानी $T_0 = 2\pi\sqrt{m/k}$।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:8}
$E_m = \tfrac12 kA^2 = \tfrac12 \times 40 \times 0.030^2 =
\qty{1.8e-2}{J}$; $v_{\max} = \sqrt{2E_m/m} = \sqrt{0.18} =
\qty{0.42}{m/s}$ --- और वह $x = 0$ पर; छोरों $x = \pm A$ पर शून्य।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:9}
(क) $L = gT_0^2/4\pi^2 = \qty{0.994}{m}$। (ख) $\Delta T_0/T_0 =
\tfrac12 \times 1.0\% = 0.5\%$: $86400 \times 0.005 \approx
\qty{4.3e2}{s}$ --- यानी प्रतिदिन सात मिनट धीमी।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:10}
(क) $A \propto \sqrt{E}$: $\sqrt{0.95} \approx 0.975$, यानी $2.5\%$ की गिरावट। (ख) $0.95^n = 0.5$: $n = \ln 2/\ln(1/0.95) \approx 14$ \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल}।
(ग) नहीं: हलके \omterm{def:g12:oscillators-and-time:damping}{अवमंदन} के लिए वह $T_0$ के एक प्रतिशत के अंश भर के
भीतर रहता है।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:11}
(क) $f_0$ पर हर धक्का क़दम मिलाकर आता है और ऊर्जा जोड़ता है; शिखर से
हटते ही धक्के बारी-बारी मदद और अड़ंगा करने लगते हैं। (ख) क़दम-ताल एक
\omterm{def:g10:signals-and-waves:period}{आवर्ती} प्रणोदन है; और पुल के $f_0$ के पास वह \omterm{def:g12:oscillators-and-time:resonance}{अनुनाद} के शिखर
पर चढ़ जाता। (ग) घूमना बढ़ाते समय ढोल ढाँचे की \omterm{def:g12:oscillators-and-time:resonance}{स्वाभाविक आवृत्ति} से
होकर गुज़रता है: पल भर का \omterm{def:g12:oscillators-and-time:resonance}{अनुनाद}, और फिर शिखर के उस पार।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:12}
(क) $T_0 \propto 1/\sqrt{g}$: धीमी, $\Delta T_0/T_0 = \tfrac12
\times 0.03/9.81 = \num{1.5e-3}$ से, यानी प्रतिदिन $\approx \qty{132}{s}$। (ख) $T_0$ स्थिर रखते हुए
$L \propto g$: $0.994 \times
0.03/9.81 \approx \qty{3.0}{mm}$ छोटा कीजिए।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:13}
$v_{\max} = 2\pi A/T_0 = 2\pi \times 0.040/0.63 = \qty{0.40}{m/s}$; $a_{\max} = (2\pi/T_0)^2 A = \qty{4.0}{m/s^2}$; $k = m(2\pi/T_0)^2
= \qty{20}{N/m}$; $E_m = \tfrac12 kA^2 = \qty{1.6e-2}{J}$।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:14}
$T_0 = 40.1/20 = \qty{2.005}{s}$; $g = 4\pi^2 L/T_0^2 = 4\pi^2
\times 0.995/4.020 = \qty{9.77}{m/s^2}$। अनिश्चितता $\approx 0.2\%
+ 2 \times 0.5\% = 1.2\%$, यानी $\pm\qty{0.12}{m/s^2}$: अंतराल में \qty{9.81}{m/s^2} आ जाता है ---
संगत।
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:15}
लोलक $10/86400 \approx \num{1.2e-4}$; क्वार्ट्ज़ $0.5/\num{2.6e6}
\approx \num{1.9e-7}$; सीज़ियम $1/\num{3.2e15} \approx \num{3e-16}$। \qty{1}{\micro s} की त्रुटि स्थिति में
$\num{1000} \times \qty{30}{cm} =
\qty{300}{m}$ है: GPS को परमाणु-कुल ही चाहिए।
\end{solution}

\begin{solution}{pb:g12:oscillators-and-time:1}
\textbf{1.} $\approx \qty{0.2}{s}$ की प्रतिक्रिया-त्रुटि $50$ से बँट जाती है: $T_0$ पर
लगभग \qty{4}{ms}।

\textbf{2.} $T = 99.4/50 = \qty{1.988}{s}$।

\textbf{3.} $L = gT^2/4\pi^2 = 9.81 \times 1.988^2/4\pi^2 \approx
\qty{0.982}{m}$।

\textbf{4.} $L = \qty{0.994}{m}$ (\omterm{def:g10:orders-of-magnitude:unit}{सेकंड} वाला लोलक): गोलक नीचे कीजिए और लंबाई $\approx \qty{1.2}{cm}$ बढ़ा
दीजिए।

\textbf{5.} बहुत छोटा, इसलिए बहुत जल्दी: तेज़, प्रतिदिन $(2.000 -
1.988)/1.988 \times 86400 \approx \qty{5.2e2}{s} \approx 8.7$ मिनट।

\textbf{6.} चंद्रमा पर: $\times\sqrt{9.81/1.62} = 2.46$ --- यानी बेकार। वेधशाला में: $\Delta T_0/T_0 = \tfrac12 \times 0.01/9.81 =
\num{5.1e-4}$:
प्रतिदिन $\approx \qty{44}{s}$ गँवाती है।

\textbf{7.} $\Delta L = L\lambda\,\Delta\theta = 0.994 \times
\num{1.9e-5} \times 5.0 \approx \qty{9.4e-5}{m} \approx
\qty{0.1}{mm}$।

\textbf{8.} $T' = 2\pi\sqrt{L(1 + \lambda\Delta\theta)/g} =
T_0\sqrt{1 + \lambda\Delta\theta} \approx T_0\,(1 + \tfrac12
\lambda\Delta\theta)$।

\textbf{9.} $\Delta T_0/T_0 = \tfrac12 \times \num{1.9e-5} \times
5.0 = \num{4.8e-5}$: प्रतिदिन $86400 \times \num{4.8e-5} \approx
\qty{4.1}{s}$, धीमी (छड़ लंबी, \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल}
लंबा)।

\textbf{10.} $\Delta\theta = \qty{-10}{K}$: $\Delta T_0/T_0 =
\num{-9.5e-5}$, प्रतिदिन $\approx \qty{8.2}{s}$ कमा लेती है।

\textbf{11.} इस्पात और पीतल अलग-अलग फैलते हैं; और उलटी-सीधी दिशाओं में लटकाने
पर पीतल की छड़ें गोलक को नीचे करती हैं जबकि इस्पात की उसे ऊपर। नाप ऐसा रखिए कि
दोनों खिसकाव कट जाएँ, और प्रभावी लंबाई --- और इसलिए चाल --- ऋतुओं से बची रह
जाती है।

\textbf{12.} $280/2.000 = 140$ \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल}।

\textbf{13.} $\theta_m = 2.00^\circ = \qty{0.0349}{rad}$: $E_m =
\tfrac12 \times 1.2 \times 9.81 \times 0.994 \times 0.0349^2 \approx
\qty{7.1e-3}{J}$।

\textbf{14.} प्रति \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} आयाम $2^{-1/140} \approx
0.9951$ से सिकुड़ता है और
ऊर्जा $0.9951^2 \approx 0.990$ से: यानी प्रति \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} $E_m$ का लगभग
$1.0\%$।

\textbf{15.} धक्का $\approx 0.010 \times \num{7.1e-3} =
\qty{7.1e-5}{J}$; शक्ति $\num{7.1e-5}/2.0 \approx \qty{3.5e-5}{W}$ --- यानी कुछ दर्जन
माइक्रोवाट।

\textbf{16.} एक सप्ताह $604800/2.0 = \num{3.02e5}$ \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} है, जिसके लिए $\num{3.02e5} \times \num{7.1e-5} \approx \qty{21}{J}$ चाहिए;
और भार $Mgh = 2.5 \times 9.81 \times 1.0 = \qty{24.5}{J}$ देता है: पर्याप्त, और वह भी $21/24.5 \approx 88\%$ \omterm{def:g10:energy-conservation:efficiency}{दक्षता}
पर।

\textbf{17.} $T = 1/\num{32768} = \qty{30.5}{\micro s}$। $\num{32768}
= 2^{15}$: दो से भाग देने वाले पंद्रह चरण क्रिस्टल के
संकेत को ठीक \qty{1}{Hz} की टिक में बदल देते हैं।

\textbf{18.} क्वार्ट्ज़: $0.3/\num{2.6e6} \approx \num{1.2e-7}$। लोलक: $0.5/86400 \approx \num{5.8e-6}$, यानी प्रति महीना $\approx
\qty{15}{s}$।
क्वार्ट्ज़ $\approx 50$ गुना से जीतता है।

\textbf{19.} $1/\num{3.2e15} \approx \num{3e-16}$: यानी क्वार्ट्ज़ से नौ कोटि नीचे। GPS संकेतों का
समय नैनोसेकंडों तक नापता है --- और हर नैनोसेकंड प्रकाश की \qty{30}{cm} यात्रा है
--- इसलिए वहाँ परमाणु-घड़ियों के सिवा कुछ नहीं चलेगा।

\textbf{20.} सुधरी हुई दीवार-घड़ी क्वार्ट्ज़ के $\pm\qty{0.3}{s}$ के मुक़ाबले महीने भर में
$\pm\qty{15}{s}$ पर टिकती है: घंटी के लिए उसे रखिए, और रेलगाड़ी के लिए क्वार्ट्ज़ पर
भरोसा कीजिए --- और \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} ख़ुद अब सीज़ियम के \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} का है।
\end{solution}
