\begin{solution}{exo:g12:work-and-energy:1}

\end{solution}

\end{solution}

\begin{solution}{exo:g12:work-and-energy:6}

\begin{solution}{exo:g12:work-and-energy:8}
$\Delta E_m = \tfrac12 \times 55 \times 9.0^2 - 55 \times 9.81 \times

\end{solution}

\begin{solution}{exo:g12:work-and-energy:9}
$H_{\min} = \tfrac52 R = \qty{0.50}{m}$। \qty{60}{cm} से: $v_{\text{शिखर}} = \sqrt{2 \times 9.81 \times (0.60 - 0.40)} \approx
\qty{2.0}{m/s}$, और $v_{\text{शिखर}}^2/(gR) = 2.0$ --- यानी न्यूनतम का दुगुना।
\omterm{def:g10:inertia:inventory}{घर्षण} रास्ते भर ऊर्जा छीलता जाता है, इसलिए असली पटरियों को
अतिरिक्त ऊँचाई चाहिए।
\end{solution}

\begin{solution}{exo:g12:work-and-energy:10}

\end{solution}

\begin{solution}{exo:g12:work-and-energy:11}
$\tfrac12 kx^2 = \tfrac12 mv^2$: $x = v\sqrt{m/k} = 2.0\sqrt{1200/\num{6.0e5}} \approx \qty{8.9}{cm}$। \qty{4.0}{m/s} पर: \qty{18}{cm} --- यानी दुगुना, चौगुना नहीं:
ऊर्जा तो चौगुनी होती है, पर $E_p$ $x^2$ की तरह बढ़ता है, इसलिए
$x$ केवल दुगुना होता है।
\end{solution}

\begin{solution}{exo:g12:work-and-energy:12}

$v \approx \qty{4.5}{m/s}$। (ग) $x = 0$: \omterm{prop:g12:work-and-energy:equilibria}{अस्थायी} (उच्चिष्ठ); $x = \qty{1.5}{m}$: \omterm{prop:g12:work-and-energy:equilibria}{स्थायी}

\end{solution}

\begin{solution}{exo:g12:work-and-energy:13}

\begin{solution}{pb:g12:work-and-energy:1}
\textbf{1.} $v = \sqrt{2gH} = \sqrt{2 \times 9.81 \times 45} \approx
\qty{29.7}{m/s}$: भार \omterm{def:g12:work-and-energy:conservative}{संरक्षी} है और \omterm{def:g12:newtons-laws:friction}{अभिलंब बल}
कार्यहीन --- इसलिए ढलान की काट हिसाब से बाहर हो जाती है।

\textbf{2.} फाटक: $E_m = mgH = 70 \times 9.81 \times 45 \approx
\qty{3.09e4}{J}$। उछाल: $E_m = \tfrac12 \times 70 \times 26^2
\approx \qty{2.37e4}{J}$।

\textbf{3.} $\Delta E_m \approx \qty{-7.2e3}{J}$; यांत्रिक-ऊर्जा प्रमेय इसका बिल \omterm{def:g12:work-and-energy:conservative}{असंरक्षी}
बलों के नाम काट देती है: बर्फ़ का \omterm{def:g10:inertia:inventory}{घर्षण} और वायु-कर्षण।

\textbf{4.} $f = \num{7.2e3}/90 \approx \qty{80}{N}$।

\textbf{5.} $\num{7.2e3}/\num{3.09e4} \approx 23\%$। मोम $f$ घटाता है और दुबकना कर्षण: दोनों $fL$ को
सिकोड़ देते हैं, जो अकेला मोल-भाव लायक़ पद है।

\textbf{6.} क्षैतिज आरंभिक वेग वाला \omterm{def:g12:projectile-motion:freefall}{मुक्त पतन} --- यानी
\omterm{def:g12:projectile-motion:projectile}{प्रक्षेप्य} गति। $v_x = \qty{26}{m/s}$; $v_y = \sqrt{2 \times 9.81 \times 40} \approx \qty{28.0}{m/s}$।

\textbf{7.} $v = \sqrt{26^2 + 28.0^2} = \sqrt{1461} \approx
\qty{38.2}{m/s}$; ऊर्जा से: $v = \sqrt{v_0^2 + 2gh} = \sqrt{676 + 785}$ --- बिलकुल वही। लगभग \qty{138}{km/h}।

\textbf{8.} $t = v_y/g = 28.0/9.81 \approx \qty{2.9}{s}$; $d = v_0 t \approx \qty{74}{m}$।

\textbf{9.} नहीं। उत्थापन वेग के लंबवत है (कार्यहीन) और कर्षण उसका विरोध करता
है (ऋणात्मक कार्य): $\Delta E_m \leq 0$, इसलिए असली उतार-चाल केवल \emph{कम} ही
हो सकती है --- हवा उड़ान लंबी करती है, उसमें ऊर्जा भरती नहीं।

\textbf{10.} क्षैतिज से $\alpha = \arctan(28.0/26) \approx \qty{47}{\degree}$ नीचे।

\textbf{11.} ढलान से कोण: $47 - 35 = \qty{12}{\degree}$। $v_\parallel = 38.2\cos\qty{12}{\degree} \approx \qty{37.4}{m/s}$; $v_\perp = 38.2\sin\qty{12}{\degree} \approx \qty{7.9}{m/s}$।

\textbf{12.} सोखा गया: $\tfrac12 \times 70 \times 7.9^2 \approx
\qty{2.2e3}{J}$, जबकि कुल $\tfrac12 \times 70 \times 38.2^2 \approx
\qty{5.1e4}{J}$: यानी लगभग $4\%$।

\textbf{13.} $h_{\text{तुल्य}} = 7.9^2/19.62 \approx \qty{3.2}{m}$ --- यानी दीवार से एक हिम्मती छलाँग। और समतल पर: $38.2^2/19.62 \approx \qty{74}{m}$
--- जिससे बचना मुश्किल है।

\textbf{14.} उड़ान-पथ के समांतर ढलान $v_\perp$ को छोटा बनाए रखती है, इसलिए
टाँगें कुछ \omterm{def:g10:orders-of-magnitude:unit}{मीटर} सोखती हैं, \omterm{def:g11:mechanical-energy:kinetic}{गतिज ऊर्जा} का पूरा पहाड़ नहीं।

\textbf{15.} $H'_0 = v_0'^2/2g = 576/19.62 \approx \qty{29.4}{m}$।

\textbf{16.} $mgH' - f\,\dfrac{H'}{\sin\qty{35}{\degree}} =
\tfrac12 m v_0'^2$।

\textbf{17.} $H' = \dfrac{\tfrac12 \times 70 \times 576}
{686.7 - 80/0.574} = \dfrac{\num{20160}}{547} \approx \qty{37}{m}$।

\textbf{18.} $36.8/29.4 \approx 1.25$: यानी $25\%$ का अधिभार।

\textbf{19.} \qty{60}{kg} के लिए: $H' = \num{17280}/(588.6 - 139.5)
\approx \qty{38.5}{m}$। ऊँची मीनार \emph{हलके} खिलाड़ी को चाहिए:
\omterm{def:g10:inertia:inventory}{घर्षण} का बिल $fL$ वही रहता है, पर उसका ऊर्जा-बजट $mgH'$
छोटा है।

\textbf{20.} \qty{24}{m/s} की उछाल-चाल के लिए \qty{37}{m} की मीनार की घोषणा कीजिए ---
और यह भी क़बूल कीजिए कि उसका चौथाई हिस्सा, यानी कोई \qty{7}{m} कंक्रीट,
घर्षणहीन सपने पर \omterm{def:g10:inertia:inventory}{घर्षण} का लगाया गया अधिभार है।
\end{solution}
