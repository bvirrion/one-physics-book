\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:2}
\qty{47}{pF}: रेडियो-ट्यूनर; \qty{2200}{\micro F}: चिकनाकार; \qty{10}{F}: सुपर-संधारित्र। $E = \tfrac12 \times 10 \times 2.7^2 \approx \qty{36}{J}$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:3}
$i = C\,\Delta u/\Delta t = 2.2\times10^{-4} \times 5.0/0.010 =
\qty{0.11}{A}$। फिर $i = C\,du/dt = 0$: आवेशित \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} खुला परिपथ है --- कोई स्थिर धारा
नहीं।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:4}
$RC = 10^4 \times 10^{-4} = \qty{1.0}{s}$; $L/R = 0.10/50 = \qty{2.0}{ms}$। $\unit{\ohm}\,\unit{F} =
(\unit{V/A})(\unit{C/V}) = \unit{C/A} = \unit{s}$; $\unit{H}/\unit{\ohm} = (\unit{V.s/A})/(\unit{V/A}) = \unit{s}$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:5}
$3.8/6.0 = 0.63$, इसलिए $\tau = \qty{2.0}{s}$; और $3\tau =
\qty{6.0}{s}$ पर $95\%$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:6}
$\tau = 4700 \times 2.2\times10^{-4} \approx \qty{1.0}{s}$; $u(\tau) = 0.63 \times 9.0 = \qty{5.7}{V}$; $u(3\tau) = 0.95 \times 9.0 = \qty{8.6}{V}$; $I_0 = \mathcal{E}/R = 9.0/4700 \approx \qty{1.9}{mA}$ (ख़ाली \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र}: पूरा $\mathcal{E}$ $R$ के
सिरों पर बैठता है)।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:7}
$du/dt = (\mathcal{E}/RC)e^{-t/RC}$, इसलिए $RC\,du/dt + u =
\mathcal{E}e^{-t/RC} + \mathcal{E}(1 - e^{-t/RC}) = \mathcal{E}$। $u(0) = 0$: \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} ख़ाली शुरू होता है; $u \to \mathcal{E}$:
पूरा आवेशित, और धारा रुक चुकी।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:8}
$\tau = 4.7\times10^{4} \times 10^{-4} = \qty{4.7}{s}$; $u(\tau) = 12\,e^{-1} \approx \qty{4.4}{V}$; \qty{0.60}{V} $5\%$ है, जो $3\tau \approx \qty{14}{s}$ पर पहुँचता है।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:9}
$u = L\,\Delta i/\Delta t = 0.50 \times 2.0/0.040 = \qty{25}{V}$। बंद करने पर $i$ परिपथ की अपनी चाल से चढ़ती है और $di/dt$ मामूली रहता
है; पर खोलने पर $i$ को लगभग तत्काल शून्य होने के लिए \emph{बाध्य}
किया जाता है, $di/dt$ विशाल हो जाता है और $u$ चिंगारी वाले मानों तक पहुँच
जाता है।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:10}
$di/dt = (\mathcal{E}/L)e^{-Rt/L}$, इसलिए $L\,di/dt + Ri =
\mathcal{E}e^{-Rt/L} + \mathcal{E}(1 - e^{-Rt/L}) = \mathcal{E}$। $i(\tau) = 0.63 \times 2.0 = \qty{1.3}{A}$; $i(3\tau) = 0.95 \times 2.0 = \qty{1.9}{A}$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:11}
$u'(t) = (\mathcal{E}/\tau)e^{-t/\tau}$, इसलिए $u'(0) = \mathcal{E}/\tau$: स्पर्शी $u = (\mathcal{E}/\tau)\,t$ $t = \tau$ पर $\mathcal{E}$ तक पहुँच जाती है।
$C = \tau/R = 0.50/2000 = \qty{250}{\micro F}$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:12}
सुपर-संधारित्र $\tfrac12 \times 3000 \times 2.7^2 \approx \qty{1.1e4}{J}$; फ़्लैश $\tfrac12 \times 8.0\times10^{-4} \times 300^2 = \qty{36}{J}$; \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} $\tfrac12 \times 0.010 \times 10^2 = \qty{0.50}{J}$। बैटरी ($\qty{1.3e4}{J}$)
$>$ सुपर-संधारित्र $\gg$ फ़्लैश $\gg$ \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} --- पर तीनों अपनी
ऊर्जा मिलीसेकंडों में उड़ेल सकते हैं: यानी शक्ति, ऊर्जा
नहीं।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:13}
$\tau = \qty{5.0}{s}$, इसलिए $R = \tau/C = 5.0/10^{-4} = \qty{50}{k\ohm}$। $1 - e^{-t/\tau} = 0.50$ से $e^{-t/\tau} = 0.50$, $t = \tau\ln 2
\approx 0.69\,\tau = \qty{3.5}{s}$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:14}
$\Delta u = i\,\Delta t/C = 0.50 \times 0.010/2.2\times10^{-3} \approx
\qty{2.3}{V}$। चाहिए $C \geq i\,\Delta t/\Delta u =
5.0\times10^{-3}/0.50 = \qty{10}{mF}$।
\end{solution}

\begin{solution}{exo:g12:rc-rl-circuits:15}
$u \approx L\,\Delta i/\Delta t = 0.80 \times 2.0/10^{-3} =
\qty{1.6}{kV}$; $E_L = \tfrac12 \times 0.80 \times 2.0^2 =
\qty{1.6}{J}$। धीरे काटिए तो $di/dt$ छोटा रहता है और $u$ कुछ \omterm{def:g11:circuits-and-power:voltage}{वोल्ट}
भर: कोई चिंगारी नहीं।
\end{solution}

\begin{solution}{pb:g12:rc-rl-circuits:1}
\textbf{1.} $q = CU_0 = 8.0\times10^{-4} \times 300 = \qty{0.24}{C}$।

\textbf{2.} $N = 0.24/1.60\times10^{-19} = \num{1.5e18}$।

\textbf{3.} $E_C = \tfrac12 \times 8.0\times10^{-4} \times 300^2 =
\qty{36}{J}$।

\textbf{4.} $\num{1.3e4}/36 \approx 360$ गुना।

\textbf{5.} सेल भरपूर संचित रखता है पर धीरे-धीरे देता है; और
\omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} मामूली ऊर्जा को विशाल \emph{शक्ति} में बदल
देता है।

\textbf{6.} $i = C\,du/dt$ के साथ $\mathcal{E} = Ri + u$: $RC\,du/dt + u = \mathcal{E}$।

\textbf{7.} $du/dt = (\mathcal{E}/RC)e^{-t/RC}$: हर $t$ के लिए $\mathcal{E}e^{-t/RC} + \mathcal{E}(1 - e^{-t/RC}) = \mathcal{E}$, और $u(0) = 0$।

\textbf{8.} $\tau = RC = 10^3 \times 8.0\times10^{-4} = \qty{0.80}{s}$।

\textbf{9.} $3\tau = \qty{2.4}{s}$ पर $u = 0.95 \times 300 =
\qty{285}{V}$ --- यानी दो \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} की वही सीटी।

\textbf{10.} $I_0 = \mathcal{E}/R = \qty{0.30}{A}$; और $u$ के चढ़ते जाने पर $R$ के सिरों पर बचा
हिस्सा $\mathcal{E} - u$ सिकुड़ता जाता है, इसलिए $i = (\mathcal{E}-u)/R$ मरती जाती है।

\textbf{11.} $\tau' = R_f C =
0.50 \times 8.0\times10^{-4} = \qty{0.40}{ms}$ के साथ $u(t) = U_0\,e^{-t/\tau'}$।

\textbf{12.} $I_0 = U_0/R_f = 300/0.50 = \qty{600}{A}$।

\textbf{13.} $P_0 = U_0 I_0 = 300 \times 600 = \qty{180}{kW}$ --- यानी एक साथ लगभग $80$ केतलियाँ।

\textbf{14.} अवधि $3\tau' = \qty{1.2}{ms}$; औसत शक्ति $\approx 36/1.2\times10^{-3} = \qty{30}{kW}$।

\textbf{15.} $2.4/1.2\times10^{-3} = 2000$। बदला केवल प्रतिरोध (भीतर \qty{1.0}{k\ohm}, बाहर \qty{0.50}{\ohm}):
यानी शक्ति का लीवर।

\textbf{16.} $C = 2E_C/U_0^2 = 400/1500^2 \approx \qty{1.8e-4}{F} =
\qty{180}{\micro F}$।

\textbf{17.} $q = CU_0 = 1.78\times10^{-4} \times 1500 \approx
\qty{0.27}{C}$।

\textbf{18.} $\tau = 50 \times 1.78\times10^{-4} \approx \qty{8.9}{ms}$, अवधि $3\tau \approx \qty{27}{ms}$ --- यानी फ़्लैश की लगभग बीस गुनी।

\textbf{19.} $I_0 = 1500/50 = \qty{30}{A}$; $P_0 = 1500 \times 30 = \qty{45}{kW}$।

\textbf{20.} $3\tau_c = \qty{6.0}{s}$ से $\tau_c = \qty{2.0}{s}$ मिलता है, इसलिए $R_c = \tau_c/C = 2.0/1.78\times10^{-4} \approx \qty{11}{k\ohm}$ और $I_0 = 1500/1.12\times10^{4} \approx \qty{0.13}{A}$ --- यानी
छह \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} में तैयार, और वह भी दसवें हिस्से भर \omterm{def:g11:circuits-and-power:current}{ऐंपियर} पर।
\end{solution}
