\section*{अध्याय \ref{ch:g12:nuclear-energy} --- नाभिकीय ऊर्जा: विखंडन, संलयन, $E=mc^2$}

\begin{solution}{exo:g12:nuclear-energy:1}
(क) $\qty{1}{eV} = \qty{1.60e-19}{J}$; $\qty{1}{MeV} = \qty{1.60e-13}{J}$।
(ख) $200 \times \num{1.60e-13} = \qty{3.2e-11}{J}$।
(ग) $E = \num{1.66054e-27} \times \num{9.00e16} = \qty{1.49e-10}{J}
= \num{1.49e-10}/\num{1.60e-13} \approx \qty{9.3e2}{MeV}$ --- यानी पाठ का \qty{931.5}{MeV}, हमारे गोल किए हुए $c$ तक।
\end{solution}

\end{solution}

\begin{solution}{exo:g12:nuclear-energy:8}

\begin{solution}{exo:g12:nuclear-energy:9}
(क) न्यूट्रॉन उदासीन है: कोई \omterm{def:g12:nuclear-energy:barrier}{कूलॉम अवरोध} नहीं, इसलिए \omterm{def:g12:nuclear-energy:fission}{विखंडन}
ठंडे में ही चल पड़ता है। पर संलयित होते दोनों नाभिक धनात्मक हैं और उन्हें अवरोध
चढ़ना ही पड़ता है, इसलिए केवल $\sim \qty{1e8}{K}$ की उत्तेजना ही इतनी हिंसक टक्करें देती है।
(ख) \omterm{def:g10:refraction:fiber}{क्रोड} में प्रोटॉनों की खगोलीय संख्या है; और भीड़ के गिने-चुने सबसे तेज़ सदस्य
ही \omterm{def:g12:nuclear-energy:fusion}{संलयन} करते हैं, जबकि सूर्य के पास अरबों वर्ष हैं --- धीमी आँच ही तो तारे को
चाहिए। (ग) यहाँ न्यूट्रॉनों का कोई हिमस्खलन नहीं होता: हर \omterm{def:g12:nuclear-energy:fusion}{संलयन} अपनी ही
टक्कर से ख़रीदा जाता है, और परिरोध ज़रा भी टूटे तो \omterm{def:g12:nuclear-energy:barrier}{प्लाज़्मा} ठंडा होकर आग
बुझा देता है।
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:10}
$\Delta m = 4 \times 1.00728 - 4.00151 = \qty{0.02761}{u}$, इसलिए $E \approx \qty{25.7}{MeV}$; और द्रव्यमान का बदला हुआ अंश $0.02761/4.02912 \approx 0.7\%$।
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:11}
(क) वह टक्करों से विखंडन-न्यूट्रॉनों को धीमा कर देता है; और धीमे
न्यूट्रॉन यूरेनियम-235 को विखंडित करने में कहीं अधिक सक्षम हैं, इसलिए
शृंखला को उनकी ज़रूरत है। (ख) वे न्यूट्रॉन सोख लेती हैं और शृंखला को प्रति
\omterm{def:g12:nuclear-energy:fission}{विखंडन} एक न्यूट्रॉन पर थामे रखती हैं --- भीतर धकेलिए और वह मर जाती है।
(ग) पानी नहीं तो मंदन नहीं: न्यूट्रॉन तेज़ ही रह जाते हैं, चूक जाते हैं, और
शृंखला घुट जाती है --- यानी डिज़ाइन ``बंद'' की ओर विफल होती है।
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:12}
\omterm{def:g12:nuclear-energy:fission}{विखंडन} हर पीढ़ी पर दुगुने हो जाते हैं, इसलिए $n$ पीढ़ियों बाद लगभग
$2^n$ टूट चुके होते हैं; और $2^{n} = \num{2.56e21} \approx 2 \times (10^{3})^{7}
\approx 2^{71}$ से $n \approx 71$ मिलता है। समय: $71 \times \qty{10}{ns}
\approx \qty{0.7}{\micro s}$ --- यानी
एक ग्राम ईंधन, तीन टन कोयले भर, एक माइक्रोसेकंड से भी कम में: क्रांतिकता कोई
ढलान नहीं, खाई है।
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:13}

बैठा है: उसे विखंडित करना हो या संलयित, दोनों बंधन में नीचे उतरते हैं और दोनों
का दाम ऊर्जा है। वह हर नाभिकीय आग की राख है।
\end{solution}

\begin{solution}{pb:g12:nuclear-energy:1}
\textbf{1.} $P_{\text{ऊष्मीय}} = \num{1.0e9}/0.33 \approx \qty{3.0}{GW}$।

\textbf{2.} $200 \times \num{1.60e-13} = \qty{3.2e-11}{J}$।

\textbf{3.} प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $\num{3.0e9}/\num{3.2e-11} \approx \num{9.4e19}$ \omterm{def:g12:nuclear-energy:fission}{विखंडन}।

\textbf{4.} $m_{\mathrm U} = 235 \times \num{1.66054e-27} =
\qty{3.90e-25}{kg}$; $\num{9.4e19} \times \num{3.90e-25} \approx
\qty{3.7e-5}{kg/s}$ --- यानी प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} कोई चालीस माइक्रोग्राम।

\textbf{5.} $\num{3.7e-5} \times \num{3.16e7} \approx \qty{1.2e3}{kg}$: यानी साल भर में लगभग $1.2$ टन यूरेनियम-235।

\textbf{6.} $\num{1.2e3}/0.040 \approx \qty{2.9e4}{kg}$ --- यानी कोई $29$ टन ईंधन: साल में एक बार, एक भारी ट्रक।

\textbf{7.} $\num{3.0e9} \times \num{3.16e7} \approx \qty{9.5e16}{J}$।

\textbf{8.} $\num{9.5e16}/\num{3.0e7} \approx \qty{3.2e9}{kg}$: $3.2$ लाख टन कोयला।

\textbf{9.} $\num{3.2e9}/\num{1.2e3} \approx \num{2.7e6}$ --- यानी सीढ़ी का $\num{8.2e13}/\num{3.0e7}$, जैसा होना ही चाहिए।

\textbf{10.} साल में $\num{3.2e9}/\num{3.0e6} \approx 1060$ गाड़ियाँ, यानी मोटे तौर पर रोज़ तीन: सुबह, दोपहर
और शाम को एक कोयला-गाड़ी, हमेशा के लिए --- और सामने हर वसंत में एक ट्रक।

\textbf{11.} $17.6 \times \num{1.60e-13} = \qty{2.82e-12}{J}$।

\textbf{12.} एक जोड़े का वज़न $5.029 \times \num{1.66054e-27} =
\qty{8.35e-27}{kg}$ है: $\num{2.82e-12}/\num{8.35e-27} \approx
\qty{3.4e14}{J/kg}$ --- यानी सीढ़ी का
\omterm{def:g12:nuclear-energy:fusion}{संलयन} वाला पायदान।

\textbf{13.} $\num{9.5e16}/\num{3.4e14} \approx \qty{2.8e2}{kg}$ D--T; और उसमें ड्यूटीरियम का हिस्सा $280 \times 2.014/5.029 \approx \qty{1.1e2}{kg}$।

\textbf{14.} $\num{1.1e2}/\num{3.3e-2} \approx \qty{3.4e3}{m^3}$ समुद्री पानी --- यानी पूरे शहर के एक वर्ष के लिए लगभग
डेढ़ ताल।

\textbf{15.} यहाँ कुछ गुणित नहीं होता: कोई न्यूट्रॉन अगला \omterm{def:g12:nuclear-energy:fusion}{संलयन} नहीं
जलाता, और परिरोध छूटते ही \omterm{def:g12:nuclear-energy:barrier}{प्लाज़्मा} ठंडा होकर सेकंडों में रुक जाता है।
कठिन काम उलटा है --- \qty{1.5e8}{K} को इतनी देर एक साथ थामे रखना कि वह जल सके।

\textbf{16.} $\num{3.9e26}/\num{9.00e16} \approx \qty{4.3e9}{kg/s}$: यानी हर \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} चालीस लाख टन।

\textbf{17.} $\num{3.0e9}/\num{9.00e16} \approx \qty{3.3e-8}{kg/s}$, यानी साल भर में $\num{3.3e-8} \times \num{3.16e7} \approx \qty{1.0}{kg}$ --- और हाँ: $\qty{1.2e3}{kg} \times 0.00085 \approx 1$, $\num{3.2e9}
\times \num{3.3e-10} \approx 1$,
$280 \times 0.00375 \approx 1$। हर ईंधन वही एक \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} सौंपता है; फ़र्क़ केवल इतना है कि उसे
ढोने को कितना माल चाहिए।

\textbf{18.} $\num{4.3e9} \times \num{3.16e7} \times \num{4.5e9}
\approx \qty{6.1e26}{kg}$ --- यानी केवल $\num{6.1e26}/\num{2.0e30} \approx
\num{3e-4}$, दस हज़ार में तीन भाग।

\textbf{19.} बदले जा सकने वाला द्रव्यमान $0.0007 \times \num{2.0e30} =
\qty{1.4e27}{kg}$, यानी $\num{1.4e27} \times \num{9.00e16} \approx
\qty{1.3e44}{J}$; और \qty{3.9e26}{W} पर
वह $\num{1.3e44}/
\num{3.9e26} \approx \qty{3.2e17}{s} \approx \num{1.0e10}$ वर्ष चलेगा --- दस अरब, जिनमें से कोई पाँच अरब अभी बाक़ी हैं।

\textbf{20.} शहर का एक वर्ष: $3.2$ लाख टन कोयला, या $\num{1.2e3}$ \omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम}
यूरेनियम-235, या डेढ़ ताल भर समुद्री पानी से खींचा गया $\num{2.8e2}$
\omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} D--T। और हर बहीखाते के नीचे वही एक प्रविष्टि: एक
\omterm{def:g10:orders-of-magnitude:unit}{किलोग्राम} द्रव्यमान, ग़ायब। $E = mc^2$ को इससे कोई मतलब नहीं कि उसे
कौन-सा ईंधन ढोता है --- मतलब केवल इससे है कि उसके लिए कितना बड़ा ट्रक चाहिए।
\end{solution}
