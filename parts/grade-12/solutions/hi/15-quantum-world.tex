\begin{solution}{pb:g12:quantum-world:1}
\textbf{1.} $E = hc/\lambda = \num{1.99e-25}/\num{5.50e-7} =
\qty{3.62e-19}{J} = \qty{2.26}{eV}$।

\textbf{2.} प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $N = \num{1.0e3}/\num{3.62e-19} \approx \num{2.8e21}$ \omterm{def:g12:quantum-world:photon}{फ़ोटॉन}।

\textbf{3.} $P = 0.20 \times \qty{1.0e3}{W} = \qty{200}{W}$; $E = 200 \times 5.0 = \qty{1.0}{kW.h}$ ($\qty{3.6e6}{J}$)।

\textbf{4.} प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} $10^{21}$ दानों पर सांख्यिकीय टिमटिमाहट अद्भुत रूप से
छोटी रह जाती है: दानेदारी औसत होकर बिलकुल चिकनी धारा बन जाती है, ठीक जैसे तेज़
उड़ेली गई रेत पानी की तरह बहती है।

\textbf{5.} नुक़सान के लिए \qtyrange{3}{4}{eV} का एक दाना चाहिए:
\omterm{def:g10:light-spectra:wavelength}{पराबैंगनी} दाने इस कसौटी पर उतरते हैं, \omterm{def:g10:light-spectra:wavelength}{अवरक्त} दाने ($\sim \qty{0.1}{eV}$)
कभी नहीं --- और संख्या प्रति दाना चोट की जगह नहीं ले सकती।

\textbf{6.} $U_s$ वह उलटा \omterm{def:g11:circuits-and-power:voltage}{विभवांतर} है जो सबसे तेज़ इलेक्ट्रॉनों को
बस-बस रोक देता है: $eU_s = E_{k,\max} = h\nu - W_0$।

\textbf{7.} $\nu = c/\lambda$: $8.22$, $7.41$, $6.88$ और $5.49$ $\times\,10^{14}\,\unit{Hz}$।

\textbf{8.} $eU_s = h\nu - W_0$ $\nu$ में ढाल $h$ के साथ एकघाती है। क्रमागत ढालें:
प्रति $10^{14}\,\unit{Hz}$ $(0.94 - 0.37)/1.39 = 0.41$ और $(1.50 - 0.94)/1.34 = 0.42$ \unit{eV} --- यानी तीनों बिंदु रेखा में हैं।

\textbf{9.} $h = e(U_{s,1}-U_{s,4})/(\nu_1-\nu_4) = \num{1.60e-19}
\times 1.13/\num{2.72e14} = \qty{6.64e-34}{J.s}$।

\textbf{10.} \qty{546}{nm} पर: $W_0 = h\nu - eU_s = \num{3.64e-19} -
\num{0.59e-19} = \qty{3.05e-19}{J} = \qty{1.90}{eV}$; $\nu_0 = W_0/h = \qty{4.60e14}{Hz}$; $\lambda_0 \approx \qty{653}{nm}$।

\textbf{11.} केवल $\lambda < \qty{653}{nm}$ ही इलेक्ट्रॉन निकालता है: यानी बैंगनी से लाल तक का
दृश्य भाग और \omterm{def:g10:light-spectra:wavelength}{पराबैंगनी}। पूरा \omterm{def:g10:light-spectra:wavelength}{अवरक्त} पक्ष --- यानी मोटे तौर पर
सूर्य की आधी ऊर्जा --- इस कैथोड के किसी काम का नहीं।

\textbf{12.} $(6.64 - 6.63)/6.63 \approx 0.2\%$ --- यानी किसी सार्वत्रिक नियतांक का दो-प्रति-हज़ार तक सही,
प्रयोगशाला में लिया गया माप।

\textbf{13.} $E_2 = \qty{-3.40}{eV}$, $E_3 = \qty{-1.51}{eV}$, $E_4 = \qty{-0.85}{eV}$, $E_5 = \qty{-0.54}{eV}$।

\textbf{14.} $3 \to 2$: \qty{1.89}{eV}; $4 \to 2$: \qty{2.55}{eV}; $5 \to 2$: \qty{2.86}{eV}।

\textbf{15.} $\lambda = 1240/E$: \qty{656}{nm} (लाल), \qty{486}{nm} (नीला-हरा), \qty{434}{nm} (बैंगनी)।

\textbf{16.} \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} की ऊर्जाएँ \omterm{def:g12:quantum-world:levels}{क्वांटित} हैं, इसलिए वह केवल
विविक्त अंतर $E_p - E_n$ ही छोड़ सकता है: सीढ़ी रेखाएँ बनाती है, इंद्रधनुष नहीं।

\textbf{17.} नहीं: \qty{500}{nm} का अर्थ है \qty{2.48}{eV}, जो किसी भी स्तर-अंतर से मेल नहीं
खाता ($E_2$ से ऊपर की ओर $1.89$, $2.55$, $2.86$, \qty{3.40}{eV} मिलते हैं; और
$E_1$ से हर चीज़ का दाम कम से कम \qty{10.2}{eV} है) --- इसलिए वहाँ गैस पारदर्शी है।

\textbf{18.} $eU = \tfrac12 m_e v^2$: $v = \sqrt{2 \times
\num{1.60e-19} \times \num{1.0e4}/\num{9.11e-31}} \approx
\qty{5.9e7}{m/s} \approx 0.20\,c$ --- यानी चिरसम्मत बरताव स्वीकार्य।

\textbf{19.} $p = m_e v = \qty{5.4e-23}{kg.m/s}$; $\lambda = h/p = \qty{1.2e-11}{m} = \qty{12}{pm}$।

\textbf{20.} लाभ $= \num{5.50e-7}/\num{1.23e-11} \approx \num{4.5e4}$। सप्ताहांत की दो संख्याएँ: $h = \qty{6.64e-34}{J.s}$, जो फ़िल्टरों
और एक वोल्टमीटर से $0.2\%$ तक नापा गया, और वह सूक्ष्मदर्शी जो केवल इसलिए
पैंतालीस हज़ार गुना तीखा है कि इलेक्ट्रॉन तरंग बनकर चलते हैं।
\end{solution}
