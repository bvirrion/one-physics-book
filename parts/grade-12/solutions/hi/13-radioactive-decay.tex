\section*{अध्याय \ref{ch:g12:radioactive-decay} --- रेडियोसक्रिय क्षय}

\begin{solution}{exo:g12:radioactive-decay:1}
$T_{1/2} = \ln 2/\lambda = 0.693/\num{1.0e-6} = \qty{6.9e5}{s}
\approx 8.0$ दिन --- यानी आयोडीन-131 की घड़ी।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:2}
$T_{1/2} = 3.8 \times \num{86400} = \qty{3.28e5}{s}$; $\lambda = 0.693/\num{3.28e5} = \qty{2.1e-6}{s^{-1}}$; $\mathcal A = \lambda N = \num{2.1e-6} \times \num{1.0e10} \approx
\qty{2.1e4}{Bq}$।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:3}
$\dd N/\dd t = N_0(-\lambda)e^{-\lambda t} = -\lambda N$ --- यानी समीकरण सही उतरता है; $N(0) = N_0 e^0 = N_0$; $N(T_{1/2})/N_0 = e^{-\lambda \ln 2/\lambda} = e^{-\ln 2} =
\tfrac12$।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:4}
$90$ वर्ष $= 3\,T_{1/2}$: $\mathcal A = \num{4.0e5}/2^3 =
\qty{5.0e4}{Bq}$। $300$ वर्ष $= 10\,T_{1/2}$: $\mathcal A = \num{4.0e5}/1024 \approx \qty{3.9e2}{Bq}$।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:5}
ढाल $= (18.85 - 20.00)/10 = -0.115$, इसलिए $\lambda = \qty{0.115}{h^{-1}} = 0.115/3600 = \qty{3.2e-5}{s^{-1}}$; $T_{1/2} = 0.693/0.115 = 6.0$ घंटे --- यानी टेक्नीशियम-99m।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:6}
$N = (1.0/226) \times \num{6.02e23} = \num{2.7e21}$; $T_{1/2} = 1600 \times \num{3.156e7} = \qty{5.05e10}{s}$, इसलिए $\lambda = \qty{1.37e-11}{s^{-1}}$; $\mathcal A = \lambda N \approx \qty{3.7e10}{Bq}$ --- यानी एक क्यूरी, मारी क्यूरी के एक
ग्राम रेडियम की \omterm{def:g11:nucleus-radioactivity:activity}{सक्रियता}।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:7}
दोनों आधे बचने की अपेक्षा करते हैं: $10$, और \num{1.0e20}। उतार-चढ़ाव गिनती के
$\sim 1/\sqrt{n}$ होते हैं: $10$ क्षयों पर लगभग $30\%$ --- इसलिए $7$ या $13$ पर
चौंकना नहीं चाहिए; और \num{1.0e20} पर लगभग $10^{-10}$। नापने लायक़ परिशुद्धता तक नियम
केवल बड़ा प्रतिदर्श ही मानता है।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:8}
$m = 140 \times \num{1.2e-4} = \qty{1.7e-2}{g}$; $N = (0.0168/40) \times \num{6.02e23} = \num{2.5e20}$। $\lambda = 0.693/(\num{1.25e9} \times \num{3.156e7}) =
\qty{1.76e-17}{s^{-1}}$; $\mathcal A = \lambda N \approx
\qty{4.4e3}{Bq}$ --- यानी शरीर के कुल का लगभग आधा (बाक़ी अधिकतर
कार्बन-14 है)।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:9}
$\lambda = 0.693/6.0 = \qty{0.116}{h^{-1}}$। $15$ घंटे पर: $\mathcal A = 500\,e^{-0.116 \times 15} = 500 \times 0.177 \approx
\qty{88}{MBq}$। \qty{1}{MBq} से नीचे: $t = \ln(500)/0.116 \approx \qty{54}{hours}$ --- यानी लगभग $9$
\omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:10}
कोयला: $3.4/13.6 = \tfrac14$, ठीक $2$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}: $t = \num{1.15e4}$ वर्ष। हड्डी: $t = \num{5730} \times \ln(1/0.30)/\ln 2 = \num{5730} \times 1.74
\approx \num{1.0e4}$ वर्ष।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:11}
$\lambda = 0.693/3.8 = \qty{0.182}{d^{-1}}$; $t = \ln(100)/\lambda = 4.61/0.182 \approx 25$ दिन (लगभग $6.6$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु})। पर चारों ओर के ग्रेनाइट और मिट्टी
की यूरेनियम-शृंखला लगातार रेडॉन पैदा करती रहती है: एक बार की हवादारी कुछ
हफ़्तों में मिट जाती है, इसलिए हवादारी \omterm{prop:g12:work-and-energy:equilibria}{स्थायी} होनी चाहिए।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:12}
$N/N_0 = e^{-\lambda t}$ से $\lambda t = \ln(N_0/N)$ मिलता है, और $\lambda = \ln 2/T_{1/2}$ से $t = T_{1/2}\ln(N_0/N)/\ln 2$। यहाँ $t = \num{5730} \times \ln(20)/\ln 2 = \num{5730} \times 4.32
\approx \num{2.5e4}$ वर्ष।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:13}
$N_{\mathrm K} = N_0 e^{-\lambda t}$ और $N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, इसलिए $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$। अनुपात $3.0$: $e^{\lambda t} = 4$, इसलिए $\lambda t = 2\ln 2$ और $t = 2\,T_{1/2} = \num{2.5e9}$
वर्ष।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:14}
$\ln \mathcal A = 5.99$, $5.42$, $4.84$, $4.26$, $3.69$: यानी प्रति \qty{60}{s} $0.57$--$0.58$ की
गिरावट --- रेखा में। ढाल: $\lambda = 2.30/240 = \qty{9.6e-3}{s^{-1}}$; $T_{1/2} = 0.693/\num{9.6e-3} \approx \qty{72}{s}$।
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:15}
(क) $2^{-10} \approx \num{1e-3}$। (ख) $(1/2)^8 = 1/256 \approx
0.4\%$। (ग) $1000/8 = 125$ अपेक्षित --- पर नियम इस बारे में चुप है कि
कौन-से: वह आबादियों का ठीक-ठीक पूर्वानुमान लगाता है और व्यक्तियों का बिलकुल
नहीं।
\end{solution}

\begin{solution}{pb:g12:radioactive-decay:1}
\textbf{1.} वायुमंडल से लेन-देन (खाना, साँस) अनुपात को वायुमंडलीय मान पर भरा
रखता है; और मृत्यु आवक रोक देती है, जिसके बाद कमान \omterm{def:g11:mechanical-energy:dissipation}{क्षय} के हाथ आ जाती है।

\textbf{2.} $T_{1/2} = \num{5730} \times \num{3.156e7} =
\qty{1.81e11}{s}$, इसलिए $\lambda = 0.693/\num{1.81e11} =
\qty{3.8e-12}{s^{-1}}$।

\textbf{3.} $\dd N/\dd t = -\lambda N_0 e^{-\lambda t} = -\lambda N$, और $N(0) = N_0$।

\textbf{4.} प्रति ग्राम $\mathcal A_0 = 13.6/60 = \qty{0.227}{Bq}$; $N = \mathcal A_0/\lambda = 0.227/\num{3.8e-12} \approx \num{5.9e10}$ \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} प्रति ग्राम।

\textbf{5.} $(1/12) \times \num{6.02e23} = \num{5.0e22}$ कार्बन-परमाणु; अनुपात $\num{5.9e10}/\num{5.0e22} \approx \num{1.2e-12}$ --- यानी $10^{12}$ में
एक, जैसा वादा था।

\textbf{6.} $\mathcal A = \mathcal A_0 e^{-\lambda t}$ से $\lambda t = \ln(\mathcal A_0/\mathcal A)$ मिलता है, और $\lambda =
\ln 2/T_{1/2}$ उसे $t = T_{1/2}\ln(\mathcal A_0/\mathcal A)/\ln 2$ बना देता है।

\textbf{7.} $t = \num{5730} \times \ln(13.6/9.1)/\ln 2 =
\num{5730} \times 0.580 \approx \num{3.3e3}$ वर्ष।

\textbf{8.} लकड़ी की मृत्यु पर --- यानी पेड़ के काटे जाने पर, तराशे जाने पर
नहीं; और पुरानी लकड़ी से तराशी गई मूर्ति (या प्राचीन लकड़ी पर बनी जालसाज़ी) अपने
पदार्थ की तिथि से पहले या बाद की होती है।

\textbf{9.} $t = \num{5730} \times \ln(13.6/10.4)/\ln 2 =
\num{5730} \times 0.387 \approx \num{2.2e3}$ वर्ष --- यानी बाईस सदी पहले के किसी लेखक से पूरी तरह
संगत।

\textbf{10.} प्रति वर्ष $\lambda = \ln 2/5730 = \num{1.21e-4}$ के साथ $\Delta t \approx \Delta\mathcal A/(\lambda\mathcal A)$: $\Delta t = (0.2/10.4)/\num{1.21e-4} \approx \num{1.6e2}$ वर्ष --- यानी
सदी-दो-सदी तक भरोसे लायक़ तिथि।

\textbf{11.} $t = \num{5730} \times \ln(13.6/13.4)/\ln 2 \approx
120$ वर्ष --- यानी $\pm 160$ वर्ष की अनिश्चितता के भीतर आधुनिक
से अलग नहीं: वह चर्म-पत्र हाल का है, और वह ``पुरातन'' जालसाज़ी।

\textbf{12.} $13.6/2^n < 0.2$ के लिए $2^n > 68$ चाहिए: यानी $n = 7$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} (क्योंकि
$2^7 = 128$), लगभग $\num{4.0e4}$ वर्ष।

\textbf{13.} दस \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}: लगभग \num{5.7e4} वर्ष --- यानी व्यावहारिक क्षितिज, कोई
साठ हज़ार वर्ष।

\textbf{14.} $\num{6.6e7}/\num{5730} \approx \num{11500}$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु}। एक ग्राम के \num{5.9e10} \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} $2^n > \num{5.9e10}$ के
बाद चुक जाते हैं, यानी $n \approx 36$ \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} ($\approx \num{2e5}$ वर्ष): इसलिए डायनासोरों की
उम्र से बहुत पहले ही कार्बन-14 का एक \omterm{def:g11:fundamental-interactions:ladder}{परमाणु} भी नहीं बचता --- गिनने को
कुछ है ही नहीं।

\textbf{15.} कोई कहीं धीमी घड़ी: पोटैशियम-40 या यूरेनियम-238, जिनकी अर्धायुएँ
अरबों वर्ष की हैं।

\textbf{16.} पिघला लावा आर्गन को बुलबुलों में उड़ जाने देता है, इसलिए ताज़ा जमे
क्रिस्टल में पोटैशियम तो होता है पर आर्गन शून्य: यानी जमने के क्षण अनुपात
$0$ से शुरू होता है --- घड़ी शून्य कर दी गई।

\textbf{17.} जैसा अभ्यास में: $N_{\mathrm K} = N_0 e^{-\lambda t}$, $N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, इसलिए $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$।

\textbf{18.} $e^{\lambda t} = 1.60$: $t = \num{1.25e9} \times \ln(1.60)/\ln 2 = \num{1.25e9} \times 0.678
\approx \num{8.5e8}$ वर्ष।

\textbf{19.} नहीं, और दोनों बार नहीं: \num{8.5e8} वर्ष पर कार्बन-14 $10^5$
\omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} चला चुका है --- यानी सन्नाटा; और \num{3.3e3} वर्ष पर पोटैशियम-40 अपनी
\omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} का \num{2.6e-6} ही चला है --- और आर्गन-अनुपात $\approx \lambda t \sim \num{2e-6}$ नापने लायक़
नहीं रह जाता। हर घड़ी लगभग दसवें हिस्से भर \omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} से लेकर लगभग दस
\omterm{def:g11:nucleus-radioactivity:halflife}{अर्धायु} तक ही पढ़ती है।

\textbf{20.} मूर्ति $\approx \num{3.3e3}$ वर्ष, पपाइरस $\approx \num{2.2e3}$ वर्ष, चट्टान $\approx \num{8.5e8}$ वर्ष ---
तीन घड़ियाँ, और नियम एक ही: $N = N_0\,e^{-\lambda t}$, उलटा पढ़ा हुआ।
\end{solution}
