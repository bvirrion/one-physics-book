\section*{अध्याय \ref{ch:g12:kinematics-2d} --- द्विविमीय शुद्धगतिकी}

\begin{solution}{exo:g12:kinematics-2d:1}
$\vect v = (3.0, 4.0)$, अचर; चाल $\sqrt{3.0^2 + 4.0^2} =
\qty{5.0}{m/s}$। \omterm{def:g10:relative-motion:trajectory}{गतिपथ}: रेखा $y = \tfrac43 x$। \omterm{def:g12:kinematics-2d:velocity}{वेग सदिश} अचर
है: यानी \omterm{def:g12:kinematics-2d:uniform}{सरल रेखीय एकसमान गति}।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:2}
सीधी रेखा, बराबर अंतराल: \omterm{def:g12:kinematics-2d:uniform}{सरल रेखीय एकसमान}; $v = 0.35/0.10 = \qty{3.5}{m/s}$। खुलते अंतराल का
अर्थ होता कि चाल बढ़ रही है।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:3}
$v(t) = 4.0\,t$, $a = \qty{4.0}{m/s^2}$: अचर त्वरण, यानी सरल रेखा में \omterm{prop:g12:kinematics-2d:uarm}{एकसमान त्वरित गति}। $v(3.0) = \qty{12}{m/s}$।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:4}
$f = 1/8.0 = \qty{0.125}{Hz}$; $v = 2\pi \times 4.5/8.0 \approx \qty{3.5}{m/s}$; $a = v^2/R = 3.53^2/4.5 \approx \qty{2.8}{m/s^2}$, जो हिंडोले की धुरी की ओर संकेत करता है।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:5}
ढाल: $a = (10.0 - 2.0)/4.0 = \qty{2.0}{m/s^2}$। क्षेत्रफल (समलंब): $d = \tfrac12 (2.0 + 10.0) \times 4.0 = \qty{24}{m}$।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:6}
$v = \qty{88.9}{m/s}$: $a_n = 88.9^2/6000 \approx \qty{1.3}{m/s^2}
\approx 0.13\,g$। चूँकि $a_n = v^2/R$, इसलिए ऊँची $v$ के लिए विशाल $R$ चाहिए,
ताकि बग़ल का त्वरण (और पटरी तथा यात्रियों पर लगता बल) सहने लायक़ रहे
--- झुकाव केवल यह बदलता है कि उसे देता कौन है, उसका आकार नहीं।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:7}
$v(t) = -2.4\,t + 8.0$; $a = \qty{-2.4}{m/s^2}$, अचर। पलटाव $v = 0$ पर: $t = 8.0/2.4 \approx \qty{3.3}{s}$, यानी $x \approx \qty{16.3}{m}$ पर। उससे पहले: आगे
की ओर, धीमी पड़ती हुई; उसके बाद: पीछे की ओर, तेज़ होती हुई --- और पूरे समय
\omterm{def:g11:electric-gravitational-fields:capacitor}{एकसमान} त्वरित।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:8}
$v_1 = 0.28/0.40 = \qty{0.70}{m/s}$; $v_2 = (0.54 - 0.10)/0.40 = \qty{1.10}{m/s}$; $v_3 = (0.88 - 0.28)/0.40 = \qty{1.50}{m/s}$। वेग हर \qty{0.20}{s} में \qty{0.40}{m/s} बढ़ता है: $a = \qty{2.0}{m/s^2}$, अचर ---
यानी सरल रेखा में \omterm{prop:g12:kinematics-2d:uarm}{एकसमान त्वरित गति}।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:9}
$T = \qty{2.36e6}{s}$: $v = 2\pi \times \num{3.84e8}/\num{2.36e6}
\approx \qty{1.0e3}{m/s}$; $a_n = v^2/R \approx \qty{2.7e-3}{m/s^2}$। अनुपात: $g/a_n \approx 3600 = 60^2$ --- यानी दूरी के व्युत्क्रम-वर्ग से
पतला पड़ा गुरुत्व (\cref{ch:g10:universal-gravitation}) चंद्रमा के मुड़ने का ठीक-ठीक हिसाब दे देता है।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:10}
$f = 1200/60 = \qty{20}{Hz}$, $T = \qty{0.050}{s}$; $v = 2\pi R f = 2\pi \times 0.25 \times 20 \approx \qty{31}{m/s}$; $a_n = v^2/R \approx \qty{3.9e3}{m/s^2} \approx 400\,g$। कपड़ों को उनका $a_n$ ढोल की दीवार देती है;
पर छेदों से होकर पानी को भीतर की ओर कोई नहीं धकेलता, इसलिए वह स्पर्शी के
अनुदिश उड़ जाता है।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:11}
$v_0 = \qty{25}{m/s}$: $t_s = 25/7.5 \approx \qty{3.3}{s}$; $d = v_0^2/(2a) = 625/15 \approx \qty{42}{m}$। आलेख: $t_s$ पर \qty{25}{m/s} से $0$ तक उतरता एक सीधा
खंड।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:12}
$\vect v = (-R\omega\sin\omega t,\, R\omega\cos\omega t)$, जिसका मानक अचर $R\omega$ है; और $\vect{OM}$ के साथ उसका अदिश गुणनफल लुप्त हो
जाता है, इसलिए वह त्रिज्या के लंबवत, यानी स्पर्शी है। दुबारा अवकलन करने पर
$\vect a = -\omega^2\vect{OM}$: $O$ की ओर, मानक $\omega^2 R = (v/R)^2 R = v^2/R$।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:13}
$a_t = \qty{2.5}{m/s^2}$ (पीछे की ओर), $a_n = 20^2/80 =
\qty{5.0}{m/s^2}$; $a = \sqrt{2.5^2 + 5.0^2} \approx \qty{5.6}{m/s^2}$; गति की दिशा से कोण: $\theta = 180^\circ - \arctan(5.0/2.5) \approx 117^\circ$ (यानी सीधे
पीछे से $63^\circ$)। $5.6 < 7.0$: पकड़ के भीतर, पर हाशिया बहुत कम।
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:14}
चढ़ाई: $t_1 = 20/1.2 \approx \qty{17}{s}$, $d_1 = 20^2/(2 \times 1.2) \approx \qty{167}{m}$। एक-सी चाल: \qty{40}{s}, $\qty{800}{m}$। ब्रेक: $t_3 = 20/1.5 \approx

\qty{70}{s} में $\qty{1.1e3}{m}$; औसत $1100/70 \approx

\begin{solution}{exo:g12:kinematics-2d:15}
$v = 2\pi \times \num{4.22e7}/\num{86164} \approx \qty{3.1e3}{m/s}$; $a_n = v^2/R \approx \qty{0.22}{m/s^2}$। उस दूरी पर पृथ्वी का गुरुत्व उसे देता है; और

\begin{solution}{pb:g12:kinematics-2d:1}
\textbf{1.} गति सापेक्ष होती है (\cref{ch:g10:relative-motion}): हर स्थिति और हर वेग को अपना तंत्र
बताना ही पड़ता है। यहाँ: ज़मीन (सड़क) का तंत्र, और सड़क के अनुदिश $x$, जो
पहले फ़्रेम की स्थिति से नापा गया है।

\textbf{2.} हर अंतराल पर: $9.0/0.50 = \qty{18.0}{m/s}$ --- चारों बराबर।

\textbf{3.} सीधी सड़क, बराबर समय में बराबर विस्थापन: \omterm{def:g12:kinematics-2d:uniform}{सरल रेखीय एकसमान
गति}।

\textbf{4.} $v_0 = \qty{18.0}{m/s} = \qty{64.8}{km/h}$।

\textbf{5.} $\vect a = \vect 0$।

\textbf{6.} $v(t) = -a\,t + v_B$; $x(t) = -\tfrac12 a t^2 + v_B\,t$।

\textbf{7.} $v(t_s) = 0$ से $t_s = v_B/a$ मिलता है; फिर $d_b = -\tfrac12 a (v_B/a)^2 + v_B^2/a = v_B^2/(2a)$।

\textbf{8.} $v_B = \sqrt{2 a d_b} = \sqrt{2 \times 6.0 \times 27} =
\sqrt{324} = \qty{18.0}{m/s}$ --- यानी ठीक वही चाल जो फ़िल्म में थी: चालक ने
ब्रेक लगाने से पहले गति बिलकुल कम नहीं की।

\textbf{9.} प्रतिक्रिया: $18.0 \times 1.0 = \qty{18}{m}$; कुल $18 + 27 = \qty{45}{m}$।

\textbf{10.} $d_b = v_B^2/(2a) \propto v_B^2$: $v_B$ दुगुना करने पर $d_b$ $4$ से गुणा हो जाता
है। \qty{50}{km/h} ($\qty{13.9}{m/s}$) पर: $d_b = 13.9^2/12.0 \approx \qty{16}{m}$।

\textbf{11.} चाल अचर है पर वेग की दिशा मुड़ती जाती है: इसलिए $\vect v$
अचर नहीं है, और कार त्वरित हो रही है।

\textbf{12.} चौथाई चक्कर: $2\pi R/4 = \qty{19.6}{m}$; $v = 19.6/2.5 \approx \qty{7.9}{m/s}$ (\qty{28}{km/h})।

\textbf{13.} $a_n = 7.85^2/12.5 \approx \qty{4.9}{m/s^2}$, गोल चक्कर के केंद्र की ओर।

\textbf{14.} $a_{\max} = \mu g = 0.70 \times 9.81 \approx
\qty{6.9}{m/s^2}$; $v_{\max} = \sqrt{\mu g R} =
\sqrt{6.87 \times 12.5} \approx \qty{9.3}{m/s}$ (\qty{33}{km/h})। $7.9 < 9.3$: पकड़ के भीतर --- और यह उस
साफ़, बिना घिसटन वाली गली से मेल खाता है।

\textbf{15.} $T = 2\pi R/v = 78.5/7.85 \approx \qty{10.0}{s};
f = \qty{0.10}{Hz}$।

\textbf{16.} $64.8 - 50 \approx \qty{15}{km/h}$ अधिक --- यानी सीमा से लगभग $30\%$ ऊपर।

\textbf{17.} ब्रेक $41 - 18 = \qty{23}{m}$ पर लगता है: $v^2 = v_B^2 - 2ad = 324 - 2 \times 6.0 \times 23 = 48$, इसलिए टक्कर के क्षण $v \approx \qty{6.9}{m/s} \approx \qty{25}{km/h}$।

\textbf{18.} \qty{13.9}{m/s} पर: $13.9 \times 1.0 + 13.9^2/12.0 =
13.9 + 16.1 = \qty{30.0}{m} < \qty{41}{m}$ --- यानी कार वैन से \qty{11}{m} पहले ही रुक
जाती।

\textbf{19.} $d(v) = v\,t_r + v^2/(2a)$: एक बार रैखिक रूप में (प्रतिक्रिया ब्रेक लगने से पहले
$v t_r$ तय कर लेती है) और एक बार वर्ग में (गतिज हिस्सा $v^2/(2a)$) --- और इसीलिए
मामूली-सी अधिकता का दाम अनुपात से कहीं अधिक पड़ता है।

\textbf{20.} डैशकैम और घिसटन के निशान एक ही बात कहते हैं: \qty{50}{km/h} वाले क्षेत्र
में \qty{65}{km/h}; और वैध सीमा पर चलती कार वैन से \qty{11}{m} पहले रुक जाती --- यानी
दुर्घटना उस अतिरिक्त \qty{15}{km/h} का नाम है।
\end{solution}
