\section*{अध्याय \ref{ch:g12:rlc-oscillations} --- मुक्त विद्युत दोलन: RLC}

\begin{solution}{exo:g12:rlc-oscillations:1}
(क) $\sqrt{LC} = \qty{1.0e-4}{s}$: $T_0 = \qty{0.63}{ms}$, $f_0 \approx \qty{1.6}{kHz}$। (ख) $\sqrt{LC} = \qty{3.16e-7}{s}$: $T_0 = \qty{2.0}{\micro s}$, $f_0 \approx \qty{0.50}{MHz}$ --- यानी $\sqrt{10^5} \approx 316$ गुना।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:2}
$LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times \unit{C/A} =
\unit{s} \times \unit{s} = \unit{s^2}$, इसलिए $\sqrt{LC}$ सेकंडों में है। $2\pi$ शुद्ध संख्या है --- उसमें
बदलने को कोई विमा ही नहीं।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:3}
$Q_0 = CU_0 = \qty{2.8}{\micro C}$; $E = \tfrac12 CU_0^2 =
\tfrac12 \times \num{4.7e-7} \times 36 \approx \qty{8.5}{\micro J}$। आवेश $f_0 = 1/(2\pi\sqrt{LC})$ पर दोलन करता है, धारा चौथाई \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} की
\omterm{def:g12:oscillators-and-time:phase}{कला} पीछे रहती है, ऊर्जा पट्टियों और \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} के बीच
छलकती है, और कुल अचर रहता है।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:4}
$Q_0 = \qty{2.0}{\micro C}$; $f_0 = 1/T_0 = \qty{250}{Hz}$; $q(T_0/2) = \qty{-2.0}{\micro C}$; $i(0) = 0$ ($t=0$ पर कोज्या चपटी है); $I_0 = 2\pi \times \num{2.0e-6}/\num{4.0e-3} \approx \qty{3.1}{mA}$।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:5}
पूरी \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} में तब जब $q = \pm Q_0$, $i = 0$ ($t = 0$, $T_0/2$, $T_0$,
\dots): यानी लोलक के छोर। और पूरी \omterm{def:g11:magnetic-fields:solenoid}{कुंडली} में तब जब $q = 0$, $i = \pm I_0$
($t = T_0/4$, $3T_0/4$, \dots): यानी लोलक का सबसे निचला, सबसे तेज़ बिंदु।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:6}
$d^2q/dt^2 = -(2\pi/T_0)^2\,Q_0\cos(2\pi t/T_0) = -(2\pi/T_0)^2 q$, इसलिए हर $t$ के लिए $L\,d^2q/dt^2 + q/C = q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$ ठीक तभी बनता है जब $(2\pi/T_0)^2 = 1/(LC)$, यानी $T_0 = 2\pi\sqrt{LC}$।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:7}
$L = \dfrac{1}{4\pi^2 f_0^2 C} = \dfrac{1}{39.5 \times
(\num{1.98e5})^2 \times \num{2.2e-10}} \approx \qty{2.9}{mH}$।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:8}
$E = \tfrac12 CU_0^2 = \qty{5.0e-5}{J}$; $\tfrac12 LI_0^2 = E$ से $I_0 = \sqrt{2E/L} = \sqrt{\num{1.0e-2}} = \qty{0.10}{A}$ मिलता है, जो पहली बार $T_0/4 = \tfrac14 \times \qty{0.63}{ms} \approx
\qty{0.16}{ms}$ पर पहुँचता है।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:9}
$E_C \propto \cos^2(2\pi t/T_0)$, और $\cos^2$ हर \emph{आधे} चक्कर पर दोहरा जाता है: यानी प्रति
\omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} \omterm{def:g12:rc-rl-circuits:capacitor}{संधारित्र} दो बार भरता है, इसलिए $2f_0$। और $t = T_0/8$ पर
\omterm{def:g12:oscillators-and-time:phase}{कला} $\pi/4$ है: $\cos^2(\pi/4) = \tfrac12$, इसलिए $E_C = E_L = E/2$।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:10}
प्रति \omterm{def:g12:rlc-oscillations:regimes}{छद्म-आवर्तकाल} अनुपात $3.0/4.0 = 0.75$: छठा उच्चिष्ठ $4.0 \times 0.75^6 \approx \qty{0.71}{V}$।
ऊर्जा आयाम के वर्ग के $\propto$ है: हर \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल} में
$0.75^2 \approx 56\%$ बचता है। और \omterm{def:g12:rlc-oscillations:regimes}{छद्म-आवर्तकाल} $\approx T_0 = 2\pi\sqrt{LC}$ ही बना रहता है: हलका
\omterm{def:g12:oscillators-and-time:damping}{अवमंदन} आयाम सिकोड़ता है, लय मुश्किल से।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:11}
$T_0 = 2\pi\sqrt{0.50 \times \num{2.0e-5}} = 2\pi \times
\num{3.16e-3} \approx \qty{20}{ms}$। शब्दावली: $k = m/(LC) =
0.50/\num{1.0e-5} = \qty{5.0e4}{N/m}$; जाँच $2\pi\sqrt{m/k} = 2\pi\sqrt{\num{1.0e-5}}$ --- यानी वही \qty{20}{ms}।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:12}
$C = \qty{500}{pF}$: $f_0 = 1/(2\pi\sqrt{\num{1.0e-13}}) \approx
\qty{0.50}{MHz}$; $C = \qty{50}{pF}$: $f_0 \approx \qty{1.6}{MHz}$। नियत $L$ पर $f_0 = (2\pi\sqrt{L})^{-1} C^{-1/2} \propto 1/\sqrt{C}$: $C$ दस गुना, पर
$f_0$ में केवल $\sqrt{10} \approx 3.2$ --- जैसा मिला।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:13}
$0.95^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.95\rvert \approx 13.5$, इसलिए लगभग $14$ \omterm{def:g12:oscillators-and-time:oscillator}{आवर्तकाल}। आयाम $\propto \sqrt{E}$: यानी
शुरुआत का $1/\sqrt{2} \approx 0.71$। \qty{1.0}{MHz} पर $T_0 = \qty{1.0}{\micro s}$: लगभग \qty{14}{\micro s}।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:14}
प्रति चक्र हानि $0.020 \times \qty{1.0e-9}{J} = \qty{2.0e-11}{J}$, और प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} \num{32768} चक्र: $P = \num{32768} \times \num{2.0e-11}
\approx \qty{6.6e-7}{W}$ --- यानी एक
माइक्रोवाट से भी कम, और किसी बटन-सेल पर सालों। क़दम मिलाकर इसलिए, कि
ऊर्जा केवल ठीक \omterm{def:g12:oscillators-and-time:phase}{कला} पर आया धक्का ही \emph{जोड़ता} है
(\omterm{def:g12:oscillators-and-time:resonance}{अनुनाद}); क़दम से उतरा धक्का झूले पर ब्रेक लगा देता।
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:15}
$C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
(\num{2.5e7})^2 \times \num{1.0e-6}} \approx \qty{41}{pF}$। प्रतिरोधक को फंदे को \omterm{def:g12:rlc-oscillations:regimes}{अनावर्ती व्यवस्था} में धकेलना चाहिए: तब
आवेश बिना छोर पार किए बैठ जाता है --- न दोलन, न झनझनाहट।
\end{solution}

\begin{solution}{pb:g12:rlc-oscillations:1}
\textbf{1.} फंदा-नियम: $u_L + u_C = 0$, जहाँ $u_C = q/C$, $u_L = L\,di/dt$, $i = dq/dt$: $L\,d^2q/dt^2 + q/C = 0$।

\textbf{2.} $d^2q/dt^2 = -(2\pi/T_0)^2 q$, इसलिए समीकरण $q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$ पढ़ता है, जो हर $t$ के लिए ठीक
तभी सही है जब $T_0 = 2\pi\sqrt{LC}$।

\textbf{3.} $LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times
\unit{C/A} = \unit{s^2}$ --- यानी $\sqrt{LC}$ एक समय है।

\textbf{4.} $LC = \qty{6.6e-14}{s^2}$: $T_0 = 2\pi \times \num{2.57e-7}
\approx \qty{1.6}{\micro s}$, $f_0 \approx \qty{620}{kHz}$ --- यानी \qtyrange{530}{1700}{kHz} के
भीतर।

\textbf{5.} $C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
\num{1.0e12} \times \num{2.0e-4}} \approx \qty{127}{pF}$।

\textbf{6.} \qty{530}{kHz} पर वही सूत्र: $C \approx \qty{451}{pF}$।

\textbf{7.} \qty{1700}{kHz} पर: $C \approx \qty{44}{pF}$।

\textbf{8.} $C = \qty{480}{pF}$: $f_0 \approx \qty{514}{kHz}$; $C = \qty{20}{pF}$: $f_0 \approx \qty{2.5}{MHz}$ --- यानी पूरी पट्टी, और दोनों
सिरों पर हाशिया भी।

\textbf{9.} $f_0 = (2\pi\sqrt{L})^{-1}C^{-1/2} \propto 1/\sqrt{C}$; $f_0$ दुगुना करने के लिए $C$ को $4$ से भाग देना
पड़ेगा।

\textbf{10.} चूँकि $f_0 \propto 1/\sqrt{C}$, $C$ के बराबर क़दम बड़े $C$ पर $f_0$ को
धीरे और छोटे $C$ पर तेज़ खिसकाते हैं: इसलिए स्टेशनों के बीच का दिया हुआ
अंतर डायल पर थोड़ी-सी ही यात्रा लेता है --- और वे ऊँची आवृत्ति वाले
(कम $C$ वाले) सिरे पर भीड़ लगा लेते हैं।

\textbf{11.} $Q_0 = CU_0 = \num{1.27e-10} \times 0.020 \approx
\qty{2.5e-12}{C}$; $E = \tfrac12 CU_0^2 \approx \qty{2.5e-14}{J}$।

\textbf{12.} $\tfrac12 LI_0^2 = E$: $I_0 = U_0\sqrt{C/L} =
0.020\sqrt{\num{6.35e-7}} \approx \qty{16}{\micro A}$।

\textbf{13.} $0.96^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.96\rvert
\approx 17$ चक्र।

\textbf{14.} \qty{1.00}{MHz} पर $T_0 = \qty{1.0}{\micro s}$: यानी झनझनाहट लगभग \qty{17}{\micro s} में आधी रह जाती है।
अकेली झनझनाहट माइक्रोसेकंडों में मर जाती है; संगीत के लिए फंदे को लगातार
खिलाना पड़ता है।

\textbf{15.} तरंग प्रति \omterm{def:g10:orders-of-magnitude:unit}{सेकंड} लाखों बार धक्का देती है; पर उसकी
आवृत्ति $f_0$ से मिलती हो तभी वे धक्के क़दम मिलाकर आते हैं और झूला
बढ़ता है --- यानी \omterm{def:g12:oscillators-and-time:resonance}{अनुनाद}; सुर से हटे स्टेशन उसके पक्ष में जितनी बार
धकेलते हैं, विपक्ष में भी उतनी बार। और प्रेषक पर कोई प्रवर्धक हर चक्र में
\omterm{def:g11:mechanical-energy:dissipation}{क्षय} हुई ऊर्जा लौटा देता है: यानी पोषित \omterm{def:g12:oscillators-and-time:oscillator}{दोलित्र}।

\textbf{16.} $LC = \qty{2.54e-14}{s^2}$: $k = m/(LC) =
0.20/\num{2.54e-14} \approx \qty{7.9e12}{N/m}$।

\textbf{17.} कार की स्प्रिंग का लगभग $10^8$ गुना: मेज़ पर रखा कोई
स्प्रिंग--द्रव्यमान निकाय मेगाहर्ट्ज़ तक नहीं पहुँचता --- रोज़मर्रा की दृढ़ताएँ
और द्रव्यमान निराशाजनक रूप से सुस्त हैं।

\textbf{18.} $m = \dfrac{k}{(2\pi f_0)^2} = \dfrac{100}
{(\num{6.28e6})^2} \approx \qty{2.5e-12}{kg}$ --- यानी कुछ नैनोग्राम, धूल का एक कण।

\textbf{19.} क्वार्ट्ज़ का क्रिस्टल: पंख-सा हलका, बेहद कड़ा एक पत्तर जो अपनी
\omterm{def:g12:rlc-oscillations:period}{स्वाभाविक आवृत्ति} पर गाता है --- यानी वही समय-रक्षक जिससे इसी वर्ष
पहले भेंट हुई थी।

\textbf{20.} $L = \qty{0.20}{mH}$ और \qtyrange{20}{480}{pF} के साथ फंदा लगभग
\qtyrange{0.51}{2.5}{MHz} बुहारता है और पूरी पट्टी ढक लेता है; $C \approx \qty{127}{pF}$
\qty{1.00}{MHz} चुनता है; अकेली छोड़ी गई झनझनाहट $\approx \qty{17}{\micro s}$ में आधी रह जाती है, पर स्टेशन
की अपनी तरंग \omterm{def:g12:oscillators-and-time:resonance}{अनुनाद} में क़दम मिलाकर फंदे को फिर-फिर भरती रहती है ---
और समाचार ख़ुद तरंग की ऊर्जा पर बजता रहता है।
\end{solution}
