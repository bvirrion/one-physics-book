\section*{Capítulo \ref{ch:g12:special-relativity} --- Relatividade restrita: a dilatação do tempo}

\begin{solution}{exo:g12:special-relativity:1}
$\gamma = 1/\sqrt{1 - (v/c)^2}$: $1.01$ ($0.1c$); $1.25$ ($0.6c$);
$1.67$ ($0.8c$); $10.0$ ($0.995c$).
\end{solution}

\begin{solution}{exo:g12:special-relativity:2}
$\gamma = 1.25$, de modo que o solo mede
$1.25 \times 10.0 = \qty{12.5}{s}$. Os \qty{10.0}{s} da nave são o \omterm{def:g12:special-relativity:proper-time}{tempo
próprio}: o metrônomo (um único relógio) está presente nas duas batidas.
\end{solution}

\begin{solution}{exo:g12:special-relativity:3}
$v = c\sqrt{1 - 1/\gamma^2}$: $\gamma = 2$ dá $v = 0.866c$;
$\gamma = 10$ dá $v = 0.995c$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:4}
$v = \qty{36.1}{m/s}$, $v/c = \num{1.2e-7}$;
$\gamma - 1 \approx (\num{1.2e-7})^2/2 \approx \num{7.2e-15}$. Um segundo
de \omterm{def:g12:mechanical-waves:delay}{atraso} depois de $1/\num{7.2e-15} \approx \qty{1.4e14}{s}$ --- cerca de
quatro milhões de anos de estrada.
\end{solution}

\begin{solution}{exo:g12:special-relativity:5}
O passageiro corre em direção ao clarão da frente, que por isso o alcança
primeiro; pelo segundo postulado, os dois clarões viajam com a mesma
velocidade $c$ pelas distâncias iguais de meio trem \emph{no \omterm{def:g10:relative-motion:frame}{referencial}
dele}, de modo que uma chegada mais cedo significa uma queda mais cedo. A
simultaneidade depende do \omterm{def:g10:relative-motion:frame}{referencial}: cada veredito está correto no
\omterm{def:g10:relative-motion:frame}{referencial} dele.
\end{solution}

\begin{solution}{exo:g12:special-relativity:6}
$\gamma = 1/\sqrt{1 - 0.99^2} = 7.09$; tempo de vida no laboratório
$7.09 \times \num{2.6e-8} = \qty{1.8e-7}{s}$. \omterm{def:g12:projectile-motion:range}{Alcance} clássico
$0.99 \times \num{3.00e8} \times \num{2.6e-8} \approx \qty{7.7}{m}$; real
$7.09 \times 7.7 \approx \qty{55}{m}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:7}
$\gamma = 1.5$ em $v = c\sqrt{1 - 1/2.25} \approx 0.75c$; $\gamma = 3$ em
$\approx 0.94c$. A assíntota: $\gamma$ (e o custo energético de mais
velocidade) diverge quando $v \to c$ --- esforço finito algum leva um
corpo com massa à velocidade da luz.
\end{solution}

\begin{solution}{exo:g12:special-relativity:8}
$\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$; a $0.8c$,
$\gamma = 5/3$, logo $\Delta t = \qty{1.7e-8}{s}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:9}
O \omterm{def:g12:special-relativity:proper-time}{tempo próprio} pertence ao relógio presente nos dois \omterm{def:g12:special-relativity:proper-time}{eventos}: (a) quem
usa o relógio; (b) o múon; (c) a piloto --- só a nave está tanto na
partida quanto na chegada.
\end{solution}

\begin{solution}{exo:g12:special-relativity:10}
$\gamma - 1 \approx 7700^2/(2 \times \num{9.0e16}) \approx \num{3.3e-10}$;
ao longo de \qty{1.58e7}{s} a astronauta fica atrás
$\num{3.3e-10} \times \num{1.58e7} \approx \qty{5.2e-3}{s}$ --- cerca de
cinco milissegundos mais jovem.
\end{solution}

\begin{solution}{exo:g12:special-relativity:11}
$E = \num{1.0e-3} \times (\num{3.00e8})^2 = \qty{9.0e13}{J}$. Uma usina de
\qty{1.0}{GW} entrega $\num{1.0e9} \times 86400 \approx \qty{8.6e13}{J}$
por dia --- um grama é um dia de produção dela --- e
$\num{9.0e13}/\num{8.5e5} \approx \num{e8}$: cem milhões de carros em
velocidade.
\end{solution}

\begin{solution}{exo:g12:special-relativity:12}
Eleve ao quadrado e inverta: $\gamma^2(1 - v^2/c^2) = 1$, logo
$v^2/c^2 = 1 - 1/\gamma^2$ e, portanto,
$v = c\sqrt{1 - 1/\gamma^2}$. Para $\gamma = 30$:
$v = c\sqrt{1 - 1/900} = 0.99944c$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:13}
Espessura contraída $L = \qty{15}{km}/30 = \qty{500}{m}$; tempo de
passagem $500/(0.99944 \times \num{3.00e8}) \approx \qty{1.7e-6}{s} <
\qty{2.2}{\micro s}$. Os dois referenciais concordam que o múon chega ao
solo --- um culpa um tempo de vida esticado, o outro uma atmosfera
encolhida.
\end{solution}

\begin{solution}{exo:g12:special-relativity:14}
Laboratório: $3200/\num{3.00e8} \approx \qty{1.1e-5}{s}$ (velocidade
$\approx c$). \omterm{def:g10:relative-motion:frame}{Referencial} do elétron: o tubo está contraído a
$3200/\num{1.0e5} = \qty{3.2}{cm}$, atravessado em
$\num{1.1e-5}/\num{1.0e5} \approx \qty{1.1e-10}{s}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:15}
Terra: $4.2/0.9 \approx \qty{4.7}{anos}$. Com
$\gamma = 1/\sqrt{1 - 0.81} = 2.29$, o relógio da sonda registra
$4.7/2.29 \approx \qty{2.0}{anos}$.
\end{solution}

\begin{solution}{pb:g12:special-relativity:1}
\textbf{1.} $c\,\Delta t_0 = \num{3.00e8} \times \num{2.2e-6} \approx
\qty{660}{m}$.

\textbf{2.} $\num{15000}/660 \approx 23$ tempos de vida.

\textbf{3.} $\mathrm{e}^{-23} \approx \num{e-10}$: menos de um múon em
dez bilhões deveria chegar, e no entanto o fluxo é de um por \unit{cm^2}
por minuto --- a física clássica é frontalmente desmentida pelo céu.

\textbf{4.} Tempo de vida no solo
$30 \times \qty{2.2}{\micro s} = \qty{66}{\micro s}$; \omterm{def:g12:projectile-motion:range}{alcance}
$\approx \num{3.00e8} \times \num{6.6e-5} \approx \qty{20}{km} >
\qty{15}{km}$: a chegada está explicada.

\textbf{5.} $v = c\sqrt{1 - 1/900} = 0.99944c$.

\textbf{6.} $\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$.

\textbf{7.} Horizontal $v\,\Delta t/2$; vertical $L$; a meia \omterm{def:g10:relative-motion:trajectory}{trajetória}
inclinada mede $c\,\Delta t/2$ porque o segundo postulado fixa em $c$ a
velocidade do \omterm{def:g12:quantum-world:photon}{fóton} também no \omterm{def:g10:relative-motion:frames}{referencial do solo}.

\textbf{8.} $(c\,\Delta t/2)^2 = L^2 + (v\,\Delta t/2)^2$ dá
$\Delta t^2(c^2 - v^2) = 4L^2$, logo
$\Delta t = (2L/c)/\sqrt{1 - v^2/c^2} = \gamma\,\Delta t_0$.

\textbf{9.} $\gamma = 1/\sqrt{1 - 0.36} = 1.25$;
$\Delta t = \qty{1.25e-8}{s}$.

\textbf{10.} O da nave: o único relógio dela está presente nos dois
tiques, de modo que ele marca o \omterm{def:g12:special-relativity:proper-time}{tempo próprio}; o solo precisa de dois
relógios sincronizados.

\textbf{11.} $\gamma = 1/\sqrt{1 - 0.64} = 1/0.6 = 5/3 \approx 1.67$.

\textbf{12.} $10.0/(5/3) = \qty{6.0}{anos}$.

\textbf{13.} Ida de \qty{5.0}{anos} a $0.8c$: $4.0$ anos-luz.

\textbf{14.} $10.0 - 6.0 = \qty{4.0}{anos}$: a astronauta está quatro
anos mais jovem do que a irmã gêmea.

\textbf{15.} A situação não é simétrica: só a astronauta dá meia-volta ---
ela acelera e troca de \omterm{def:g12:special-relativity:inertial}{referencial inercial}, enquanto a gêmea da Terra
permanece em um só --- de modo que só o relógio dela registra o tempo
menor.

\textbf{16.} $v = 2\pi \times \num{2.66e7}/\num{43200} \approx
\qty{3.9e3}{m/s}$.

\textbf{17.} $\gamma - 1 \approx (\num{3.87e3})^2/(2 \times \num{9.0e16})
\approx \num{8.3e-11}$.

\textbf{18.} $\num{8.3e-11} \times \num{86400} \approx \qty{7.2e-6}{s} =
\qty{7.2}{\micro s}$ de \omterm{def:g12:mechanical-waves:delay}{atraso} por dia.

\textbf{19.} $c \times \num{7.2e-6} \approx \qty{2.2}{km}$ de erro de
posição por dia.

\textbf{20.} Desvio líquido $45 - 7.2 \approx \qty{38}{\micro s}$ de
adiantamento por dia; erro
$c \times \num{38e-6} \approx \qty{11}{km}$ por dia --- a conta
relativística que todo relógio de GPS paga adiantado, desafinado no solo
para bater certo em \omterm{def:g10:universal-gravitation:satellite}{órbita}.
\end{solution}
