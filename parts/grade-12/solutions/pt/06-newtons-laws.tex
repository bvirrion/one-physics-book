\section*{Capítulo \ref{ch:g12:newtons-laws} --- As leis de Newton}

\begin{solution}{exo:g12:newtons-laws:1}
(a) $p = 70 \times 10 = \qty{7.0e2}{kg.m/s}$.
(b) $v = \qty{13.9}{m/s}$: $p \approx \qty{1.8e4}{kg.m/s}$.
(c) $p = 0.0080 \times 800 = \qty{6.4}{kg.m/s}$. Carro $>$ velocista $>$
bala: cinco \omterm{def:g10:orders-of-magnitude:oom}{ordens de grandeza} de massa são mais do que a velocidade
consegue comprar.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:2}
$N = m(g + a_z)$: (a) $70 \times 11.01 = \qty{771}{N}$;
(b) $a_z = 0$, logo $N = mg = \qty{687}{N}$ --- a primeira lei:
velocidade constante, \omterm{def:g10:inertia:compensate}{forças que se compensam}; (c)
$70 \times 8.61 = \qty{603}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:3}
$a = g\sin 30^\circ = \qty{4.9}{m/s^2}$;
$v = \sqrt{2 \times 4.905 \times 3.0} = \qty{5.4}{m/s}$. A massa se
cancela entre $ma$ e $mg\sin\alpha$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:4}
Inversão: $\Delta p = 0.058 \times (25 + 15) = \qty{2.3}{kg.m/s}$;
$F = 2.32/0.0050 = \qty{4.6e2}{N}$ --- cerca de $800$ vezes o \omterm{def:g10:universal-gravitation:weight}{peso} de
\qty{0.57}{N}, e é por isso que a gravidade é ignorada durante o contato.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:5}
\omterm{def:g10:universal-gravitation:weight}{Peso} (a Terra sobre o livro, para baixo): parceiro = o livro puxa a
Terra, para cima. \omterm{def:g12:newtons-laws:friction}{Força normal} (a mesa sobre o livro, para cima):
parceiro = o livro pressiona a mesa, para baixo. O \omterm{def:g10:universal-gravitation:weight}{peso} e a normal agem
sobre o \emph{mesmo} corpo e pertencem a duas interações diferentes ---
iguais aqui apenas porque $\vect a = \vect 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:6}
$N = mg = \qty{490}{N}$. (a) $F > \mu_s N = \qty{221}{N}$.
(b) $a = (220.7 - 0.35 \times 490.5)/50 = \qty{0.98}{m/s^2}$.
(c) $F = \mu_k N = \qty{172}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:7}
(a) Desliza quando $\tan\alpha > \mu_s$: $\alpha = \arctan 0.45 \approx
24^\circ$. (b) $a = g(\sin 30^\circ - 0.35\cos 30^\circ) =
9.81 \times 0.197 = \qty{1.9}{m/s^2}$. (c) Sim: uma vez deslizando, o
\omterm{def:g10:inertia:inventory}{atrito} cai para $\mu_k < \mu_s$, de modo que
$a = g\cos\alpha\,(\tan\alpha - \mu_k) \approx \qty{0.9}{m/s^2} > 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:8}
$a = m_2 g/(m_1 + m_2) = 19.62/5.0 = \qty{3.9}{m/s^2}$;
$T = m_1 a = \qty{11.8}{N}$. O bloco pendurado acelera para baixo, de
modo que a \omterm{def:g11:forces-and-motion:netforce}{força resultante} sobre ele aponta para baixo: $T$ tem de ficar
abaixo de $m_2 g$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:9}
$v_{\max} = \sqrt{\mu_s g R}$: seca
$\sqrt{0.70 \times 9.81 \times 50} = \qty{18.5}{m/s} \approx
\qty{67}{km/h}$; molhada $\qty{14.0}{m/s} = \qty{50}{km/h}$. Tanto
$mv^2/R$ quanto $\mu_s mg$ carregam $m$: ela se cancela.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:10}
Vertical: $T\cos\theta = mg$, logo
$T = 2.943/\cos 25^\circ = \qty{3.2}{N}$. Raio $R = L\sin\theta =
\qty{0.507}{m}$; radial: $T\sin\theta = mv^2/R$, logo
$v = \sqrt{gR\tan\theta} = \sqrt{9.81 \times 0.507 \times 0.466}
= \qty{1.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:11}
$a_z = (N - mg)/m$ com $mg = \qty{49.05}{N}$: $+\qty{1.0}{m/s^2}$,
$0$, $-\qty{1.0}{m/s^2}$. Por exemplo, uma subida: ganhar velocidade,
cruzeiro, frear. Sim: entrando já em descida, as mesmas leituras
significam frear, cruzeiro e voltar a ganhar velocidade descendo --- a
balança marca aceleração, e não velocidade.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:12}
(a) $0 = 60V + 4.0 \times 8.0$: $V = \qty{-0.53}{m/s}$.
(b) A bola vai a $8.0 + V$ em relação ao solo:
$0 = 60V + 4.0(8.0 + V)$, logo $V = -32/64 = \qty{-0.50}{m/s}$ e a bola
voa a \qty{7.5}{m/s}. (c) O empurrão da bola nas mãos dela --- o
parceiro, pela terceira lei, do arremesso.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:13}
Sem \omterm{def:g10:inertia:inventory}{atrito}: $N\cos\theta = mg$ (vertical) e $N\sin\theta =
mv^2/R$ (radial); dividindo, $\tan\theta = v^2/(gR) = 400/784.8$:
$\theta \approx 27^\circ$. Mais devagar: o carro tende a escorregar
rampa abaixo, e o \omterm{def:g10:inertia:inventory}{atrito} precisa apontar rampa acima; mais rápido: o
contrário.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:14}
(a) $250 \times 2500 = \qty{6.3e5}{kg.m/s}$ por segundo: \omterm{def:g10:inertia:inventory}{propulsão} de
$\qty{6.3e5}{N}$. (b) $P = \qty{4.9e5}{N}$;
$a = (6.25 - 4.905) \times 10^5 / \num{5.0e4} = \qty{2.7}{m/s^2}$.
(c) Ele empurra o próprio jato de escape (terceira lei), e não o ar.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:15}
(a) $m_2 g > \mu_s m_1 g$: $m_2 > 0.35 \times 4.0 = \qty{1.4}{kg}$.
(b) $a = (m_2 - \mu_k m_1)\,g/(m_1 + m_2) = 1.0 \times 9.81/6.0 =
\qty{1.6}{m/s^2}$; $T = m_2(g - a) = \qty{16.4}{N}$.
(c) $\mu_k = 0$ dá $a = m_2 g/(m_1 + m_2)$: a fórmula do
\cref{exo:g12:newtons-laws:8}.
\end{solution}

\begin{solution}{pb:g12:newtons-laws:1}
\textbf{1.} Sistema: cabine $+$ carro ($+$ passageiros). A montanha é
fixa ao solo, inercial com excelente precisão. \omterm{def:g10:inertia:force}{Forças}: o \omterm{def:g10:universal-gravitation:weight}{peso}
$M\vect g$ (para baixo), a \omterm{def:g12:newtons-laws:friction}{força normal} $\vect N$ do cabo carregador
(perpendicular a ele) e a tensão $\vect T$ do cabo tractor (rampa
acima).

\textbf{2.} Em repouso, a primeira lei: $\sum \vect F = \vect 0$. Ao
longo da rampa: $T = Mg\sin\alpha$; perpendicularmente:
$N = Mg\cos\alpha$.

\textbf{3.} $T = 4000 \times 9.81 \times 0.500 = \qty{19.6}{kN}$;
$N = 4000 \times 9.81 \times 0.866 = \qty{34.0}{kN}$.

\textbf{4.} Vazia: $T = \qty{9.8}{kN}$. Toda \omterm{def:g10:inertia:force}{força} do equilíbrio é
proporcional a $M$, e portanto $T$ também é.

\textbf{5.} Ao longo da rampa, $Ma = Mg\sin\alpha$: $M$ se cancela,
$a = g\sin\alpha = \qty{4.9}{m/s^2}$ rampa abaixo.

\textbf{6.} $T' = M(g\sin\alpha + a) = 4000 \times 5.505 =
\qty{22.0}{kN}$.

\textbf{7.} $a/(g\sin\alpha) = 0.60/4.905 \approx 12\%$.

\textbf{8.} $t = v/a = \qty{10}{s}$; $d = v^2/(2a) = \qty{30}{m}$.

\textbf{9.} Componente vertical de $\vect a$: $a\sin\alpha =
\qty{0.30}{m/s^2}$, logo $N = m(g + a\sin\alpha) = 80 \times 10.11 =
\qty{809}{N}$ contra $mg = \qty{785}{N}$: $3\%$ mais pesada.

\textbf{10.} O \omterm{def:g11:forces-and-motion:friction}{atrito estático} do piso, horizontal:
$ma\cos\alpha = 80 \times 0.52 \approx \qty{42}{N}$. Inclinar-se faz o
empurrão total do piso passar pelo \omterm{def:g11:forces-and-motion:com}{centro de massa} dela, e assim ele não
a derruba.

\textbf{11.} $a_T = dv/dt = 0$ (\omterm{def:g12:kinematics-2d:velocity}{velocidade escalar} constante);
$a_N = v^2/R = 36/40 = \qty{0.90}{m/s^2}$, apontada para o centro ---
verticalmente para baixo no alto.

\textbf{12.} Eixo normal (para baixo positivo): $Mg - N' = Mv^2/R$,
logo $N' = M(g - v^2/R) = 4000 \times 8.91 = \qty{35.6}{kN}$.

\textbf{13.} $N = m(g - v^2/R) = 80 \times 8.91 = \qty{713}{N}$: mais
leve em $v^2/(gR) = 0.90/9.81 \approx 9\%$.

\textbf{14.} $N' = 0$ em $v = \sqrt{gR} = \qty{19.8}{m/s} \approx
\qty{71}{km/h}$ --- mais do que o triplo do cruzeiro: o carro corre
sobre o cabo com larga margem.

\textbf{15.} \omterm{thm:g12:newtons-laws:third}{Terceira lei de Newton}: \qty{35.6}{kN}, pressionando para
baixo sobre o cabo (e portanto sobre a torre) --- um pouco menos do que
o \omterm{def:g10:universal-gravitation:weight}{peso} em repouso, aliviado pelo voo curvo.

\textbf{16.} $F = \Delta p/\Delta t = 4000 \times 6.0/1.5 =
\qty{16}{kN}$, contra o movimento ($a = \qty{4.0}{m/s^2}$).

\textbf{17.} $f = ma = 80 \times 4.0 = \qty{320}{N}$; o teto é
$\mu_s mg = 0.40 \times 785 = \qty{314}{N} < \qty{320}{N}$: os pés
sozinhos escorregam --- segure no corrimão.

\textbf{18.} $0 = M_0 V + m(v_{\text{rel}} + V)$:
$V = -80 \times 2.5/2080 = \qty{-0.096}{m/s}$; o passageiro se move a
$2.5 - 0.096 \approx \qty{2.4}{m/s}$ em relação ao solo.

\textbf{19.} O foguete: a \omterm{def:g10:inertia:inventory}{propulsão} dele é fixada pela massa expelida
por segundo vezes a velocidade de ejeção em relação a ele.

\textbf{20.} A maior tensão encontrada é a da partida,
$T' = M \times \qty{5.505}{N/kg}$ (o repouso dá $Mg\sin\alpha$, e o alto
da torre, menos ainda). Teto: $T' \leq \qty{20}{kN}$, logo
$M \leq 20000/5.505 = \qty{3.63e3}{kg}$: carga útil de
\qty{1.63e3}{kg}, ou seja, $20$ passageiros de \qty{80}{kg} (e não
$25$). Placa: \emph{20 pessoas}; então $M = \qty{3.6e3}{kg}$,
$T' = \qty{19.8}{kN}$, coeficiente de segurança
$100/19.8 \approx 5.1 \geq 5$.
\end{solution}
