\section*{Capítulo \ref{ch:g12:rlc-oscillations} --- Oscilações elétricas livres: RLC}

\begin{solution}{exo:g12:rlc-oscillations:1}
(a) $\sqrt{LC} = \qty{1.0e-4}{s}$: $T_0 = \qty{0.63}{ms}$,
$f_0 \approx \qty{1.6}{kHz}$. (b) $\sqrt{LC} = \qty{3.16e-7}{s}$:
$T_0 = \qty{2.0}{\micro s}$, $f_0 \approx \qty{0.50}{MHz}$ --- um fator
$\sqrt{10^5} \approx 316$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:2}
$LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times \unit{C/A} =
\unit{s} \times \unit{s} = \unit{s^2}$, de modo que $\sqrt{LC}$ está em
segundos. $2\pi$ é um número puro --- não tem dimensão para mudar.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:3}
$Q_0 = CU_0 = \qty{2.8}{\micro C}$;
$E = \tfrac12 CU_0^2 = \tfrac12 \times \num{4.7e-7} \times 36 \approx
\qty{8.5}{\micro J}$. A carga oscila em $f_0 = 1/(2\pi\sqrt{LC})$, com a
corrente um quarto de \omterm{def:g12:oscillators-and-time:oscillator}{período} defasada, a \omterm{def:g10:energy-conservation:energy}{energia} indo e vindo entre as
placas e a \omterm{def:g11:magnetic-fields:solenoid}{bobina}, e o total constante.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:4}
$Q_0 = \qty{2.0}{\micro C}$; $f_0 = 1/T_0 = \qty{250}{Hz}$;
$q(T_0/2) = \qty{-2.0}{\micro C}$; $i(0) = 0$ (o cosseno tem tangente
horizontal em $t=0$);
$I_0 = 2\pi \times \num{2.0e-6}/\num{4.0e-3} \approx \qty{3.1}{mA}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:5}
Tudo no \omterm{def:g12:rc-rl-circuits:capacitor}{capacitor} quando $q = \pm Q_0$ e $i = 0$ ($t = 0$, $T_0/2$,
$T_0$, \dots): os extremos do pêndulo. Tudo na \omterm{def:g11:magnetic-fields:solenoid}{bobina} quando $q = 0$ e
$i = \pm I_0$ ($t = T_0/4$, $3T_0/4$, \dots): o ponto mais baixo e mais
rápido do pêndulo.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:6}
$d^2q/dt^2 = -(2\pi/T_0)^2\,Q_0\cos(2\pi t/T_0) = -(2\pi/T_0)^2 q$, logo
$L\,d^2q/dt^2 + q/C = q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$ para todo $t$
exatamente quando $(2\pi/T_0)^2 = 1/(LC)$, ou seja,
$T_0 = 2\pi\sqrt{LC}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:7}
$L = \dfrac{1}{4\pi^2 f_0^2 C} = \dfrac{1}{39.5 \times
(\num{1.98e5})^2 \times \num{2.2e-10}} \approx \qty{2.9}{mH}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:8}
$E = \tfrac12 CU_0^2 = \qty{5.0e-5}{J}$; $\tfrac12 LI_0^2 = E$ dá
$I_0 = \sqrt{2E/L} = \sqrt{\num{1.0e-2}} = \qty{0.10}{A}$, atingida pela
primeira vez em $T_0/4 = \tfrac14 \times \qty{0.63}{ms} \approx
\qty{0.16}{ms}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:9}
$E_C \propto \cos^2(2\pi t/T_0)$, e $\cos^2$ se repete a cada
\emph{meia} volta: dois enchimentos do \omterm{def:g12:rc-rl-circuits:capacitor}{capacitor} por \omterm{def:g12:oscillators-and-time:oscillator}{período}, e portanto
$2f_0$. Em $t = T_0/8$ a \omterm{def:g12:oscillators-and-time:phase}{fase} é $\pi/4$: $\cos^2(\pi/4) = \tfrac12$, logo
$E_C = E_L = E/2$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:10}
Razão $3.0/4.0 = 0.75$ por \omterm{def:g12:rlc-oscillations:regimes}{pseudoperíodo}: sexto máximo
$4.0 \times 0.75^6 \approx \qty{0.71}{V}$. A \omterm{def:g10:energy-conservation:energy}{energia} é proporcional ao
quadrado da \omterm{def:g12:oscillators-and-time:oscillator}{amplitude}: sobrevivem $0.75^2 \approx 56\%$ a cada \omterm{def:g12:oscillators-and-time:oscillator}{período}. O
\omterm{def:g12:rlc-oscillations:regimes}{pseudoperíodo} continua $\approx T_0 = 2\pi\sqrt{LC}$: o \omterm{def:g12:oscillators-and-time:damping}{amortecimento}
fraco encolhe a \omterm{def:g12:oscillators-and-time:oscillator}{amplitude}, quase não o compasso.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:11}
$T_0 = 2\pi\sqrt{0.50 \times \num{2.0e-5}} = 2\pi \times \num{3.16e-3}
\approx \qty{20}{ms}$. Pelo dicionário:
$k = m/(LC) = 0.50/\num{1.0e-5} = \qty{5.0e4}{N/m}$; conferindo,
$2\pi\sqrt{m/k} = 2\pi\sqrt{\num{1.0e-5}}$ --- os mesmos \qty{20}{ms}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:12}
$C = \qty{500}{pF}$: $f_0 = 1/(2\pi\sqrt{\num{1.0e-13}}) \approx
\qty{0.50}{MHz}$; $C = \qty{50}{pF}$: $f_0 \approx \qty{1.6}{MHz}$. Com
$L$ fixo, $f_0 = (2\pi\sqrt{L})^{-1} C^{-1/2} \propto 1/\sqrt{C}$: dez
vezes em $C$, apenas $\sqrt{10} \approx 3.2$ em $f_0$ --- como se
encontrou.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:13}
$0.95^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.95\rvert \approx 13.5$, ou
seja, uns $14$ \omterm{def:g12:oscillators-and-time:oscillator}{períodos}. A \omterm{def:g12:oscillators-and-time:oscillator}{amplitude} é proporcional a $\sqrt{E}$: cai
para $1/\sqrt{2} \approx 0.71$ da inicial. A \qty{1.0}{MHz},
$T_0 = \qty{1.0}{\micro s}$: cerca de \qty{14}{\micro s}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:14}
Perda por ciclo $0.020 \times \qty{1.0e-9}{J} = \qty{2.0e-11}{J}$, a
\num{32768} ciclos por segundo:
$P = \num{32768} \times \num{2.0e-11} \approx \qty{6.6e-7}{W}$ ---
menos de um microwatt, anos com uma pilha-botão. Em compasso porque só um
empurrão na \omterm{def:g12:oscillators-and-time:phase}{fase} certa \emph{acrescenta} \omterm{def:g10:energy-conservation:energy}{energia} (\omterm{def:g12:oscillators-and-time:resonance}{ressonância}); fora de
compasso, ele frearia o balanço.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:15}
$C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
(\num{2.5e7})^2 \times \num{1.0e-6}} \approx \qty{41}{pF}$. O resistor
deve empurrar a malha para o \omterm{def:g12:rlc-oscillations:regimes}{regime aperiódico}: a carga então se acomoda
sem passar do ponto --- sem oscilação, sem repique.
\end{solution}

\begin{solution}{pb:g12:rlc-oscillations:1}
\textbf{1.} Lei das malhas: $u_L + u_C = 0$, com $u_C = q/C$,
$u_L = L\,di/dt$ e $i = dq/dt$: $L\,d^2q/dt^2 + q/C = 0$.

\textbf{2.} $d^2q/dt^2 = -(2\pi/T_0)^2 q$, de modo que a equação fica
$q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$, verdadeira para todo $t$ exatamente
quando $T_0 = 2\pi\sqrt{LC}$.

\textbf{3.} $LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times
\unit{C/A} = \unit{s^2}$ --- $\sqrt{LC}$ é um tempo.

\textbf{4.} $LC = \qty{6.6e-14}{s^2}$:
$T_0 = 2\pi \times \num{2.57e-7} \approx \qty{1.6}{\micro s}$,
$f_0 \approx \qty{620}{kHz}$ --- dentro de \qtyrange{530}{1700}{kHz}.

\textbf{5.} $C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
\num{1.0e12} \times \num{2.0e-4}} \approx \qty{127}{pF}$.

\textbf{6.} A mesma fórmula a \qty{530}{kHz}: $C \approx \qty{451}{pF}$.

\textbf{7.} A \qty{1700}{kHz}: $C \approx \qty{44}{pF}$.

\textbf{8.} $C = \qty{480}{pF}$: $f_0 \approx \qty{514}{kHz}$;
$C = \qty{20}{pF}$: $f_0 \approx \qty{2.5}{MHz}$ --- a faixa inteira, com
margem nas duas pontas.

\textbf{9.} $f_0 = (2\pi\sqrt{L})^{-1}C^{-1/2} \propto 1/\sqrt{C}$;
dobrar $f_0$ exige dividir $C$ por $4$.

\textbf{10.} Como $f_0 \propto 1/\sqrt{C}$, passos iguais de $C$ movem
$f_0$ devagar com $C$ grande e depressa com $C$ pequeno: um dado
espaçamento entre estações exige então pouco giro do botão --- elas se
amontoam na ponta de \omterm{def:g12:oscillators-and-time:oscillator}{frequência} alta (e $C$ baixo).

\textbf{11.} $Q_0 = CU_0 = \num{1.27e-10} \times 0.020 \approx
\qty{2.5e-12}{C}$; $E = \tfrac12 CU_0^2 \approx \qty{2.5e-14}{J}$.

\textbf{12.} $\tfrac12 LI_0^2 = E$:
$I_0 = U_0\sqrt{C/L} = 0.020\sqrt{\num{6.35e-7}} \approx
\qty{16}{\micro A}$.

\textbf{13.} $0.96^n = \tfrac12$:
$n = \ln 2/\lvert\ln 0.96\rvert \approx 17$ ciclos.

\textbf{14.} A \qty{1.00}{MHz}, $T_0 = \qty{1.0}{\micro s}$: o repique
cai à metade em cerca de \qty{17}{\micro s}. Um repique solitário morre
em microssegundos; a música exige a malha alimentada continuamente.

\textbf{15.} A \omterm{def:g10:signals-and-waves:wave}{onda} empurra milhões de vezes por segundo; só quando a
\omterm{def:g12:oscillators-and-time:oscillator}{frequência} dela coincide com $f_0$ é que os empurrões chegam em compasso
e constroem o balanço --- \omterm{def:g12:oscillators-and-time:resonance}{ressonância}; as estações fora de sintonia
empurram tanto contra quanto a favor. No transmissor, um amplificador
devolve a \omterm{def:g10:energy-conservation:energy}{energia} \omterm{def:g11:mechanical-energy:dissipation}{dissipada} a cada ciclo: um \omterm{def:g12:oscillators-and-time:oscillator}{oscilador} mantido.

\textbf{16.} $LC = \qty{2.54e-14}{s^2}$:
$k = m/(LC) = 0.20/\num{2.54e-14} \approx \qty{7.9e12}{N/m}$.

\textbf{17.} Cerca de $10^8$ vezes a mola do carro: nenhum sistema
massa--mola de bancada chega a megahertz --- as rigidezes e as massas do
dia a dia são desesperadamente lentas.

\textbf{18.} $m = \dfrac{k}{(2\pi f_0)^2} = \dfrac{100}
{(\num{6.28e6})^2} \approx \qty{2.5e-12}{kg}$ --- alguns nanogramas, um
cisco de poeira.

\textbf{19.} Um cristal de quartzo: uma lâmina leve como pluma e
ultrarrígida, cantando na \omterm{def:g12:rlc-oscillations:period}{frequência própria} dela --- o marcador de tempo
encontrado antes, neste ano.

\textbf{20.} Com $L = \qty{0.20}{mH}$ e \qtyrange{20}{480}{pF} a malha
varre cerca de \qtyrange{0.51}{2.5}{MHz}, cobrindo a faixa;
$C \approx \qty{127}{pF}$ pega \qty{1.00}{MHz}; deixado sozinho, o
repique cai à metade em $\approx \qty{17}{\micro s}$, mas a própria \omterm{def:g10:signals-and-waves:wave}{onda}
da estação, ressoando em compasso, realimenta a malha --- o jornal toca
com a \omterm{def:g10:energy-conservation:energy}{energia} da própria \omterm{def:g10:signals-and-waves:wave}{onda}.
\end{solution}
