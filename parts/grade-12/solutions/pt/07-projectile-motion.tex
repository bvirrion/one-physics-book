\section*{Capítulo \ref{ch:g12:projectile-motion} --- Queda livre e lançamento de projéteis}

\begin{solution}{exo:g12:projectile-motion:1}
$h = \tfrac12 \times 9.81 \times 2.0^2 = \qty{19.6}{m}$;
$v = gt = \qty{19.6}{m/s} \approx \qty{71}{km/h}$.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:2}
Não há ar na Lua: os dois em \omterm{def:g12:projectile-motion:freefall}{queda livre}, e $m\vect a = m\vect g$ dá
$\vect a = \vect g$ para qualquer massa. Na Terra, o arrasto sobre a pena
é comparável ao \omterm{def:g10:universal-gravitation:weight}{peso} minúsculo dela: ela desce com uma \omterm{def:g12:projectile-motion:terminal}{velocidade limite}
de cerca de \qty{1}{m/s}, enquanto o martelo mal percebe o ar.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:3}
$t = \sqrt{2h/g} = 0.45$, $1.4$, \qty{4.5}{s};
$v = gt = 4.4$, $14$, \qty{44}{m/s}. Os tempos crescem como $\sqrt h$:
$\times 100$ na altura é apenas $\times 10$ no tempo.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:4}
$h = v_0^2/2g = \qty{11.5}{m}$ em $t = v_0/g = \qty{1.53}{s}$; de volta
em \qty{3.06}{s}, de novo a \qty{15}{m/s} (simetria).
\end{solution}

\begin{solution}{exo:g12:projectile-motion:5}
$t = \sqrt{2 \times 0.80/9.81} = \qty{0.40}{s}$;
$x = 2.5 \times 0.40 = \qty{1.0}{m}$. Não: o tempo de queda pertence
apenas ao movimento vertical (independência).
\end{solution}

\begin{solution}{exo:g12:projectile-motion:6}
$(\ang{15},\ang{75})$ e $(\ang{30},\ang{60})$ compartilham o \omterm{def:g12:projectile-motion:range}{alcance};
\ang{45} vence. O mais íngreme de um par complementar voa por mais tempo:
$t_f = 2v_0\sin\alpha/g$ cresce com $\alpha$.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:7}
$4.905\,t^2 + 5.0\,t - 20 = 0$ dá
$t = (-5 + \sqrt{25 + 392.4})/9.81 = \qty{1.57}{s}$;
$v = 5.0 + 9.81 \times 1.57 = \qty{20.4}{m/s}$. Pela \omterm{def:g10:energy-conservation:energy}{energia}:
$v = \sqrt{5.0^2 + 2 \times 9.81 \times 20} = \qty{20.4}{m/s}$ --- o
mesmo.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:8}
$t_f = 2 \times 40 \times \sin\ang{30}/9.81 = \qty{4.1}{s}$;
$h = (40\sin\ang{30})^2/(2 \times 9.81) = \qty{20.4}{m}$;
$R = 40^2 \sin\ang{60}/9.81 = \qty{1.4e2}{m}$.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:9}
$R = 10.5^2\sin\ang{40}/9.81 \approx \qty{7.2}{m}$. Ficaram de fora: o
\omterm{def:g11:forces-and-motion:com}{centro de massa} decola alto (corpo estendido) e aterrissa baixo e bem à
frente (pernas jogadas para diante), e o saltador não é um ponto --- os
\qty{1.7}{m} que faltam moram nessa geometria, e não na parábola.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:10}
Juntas: as duas têm $v_y(0) = 0$, logo o mesmo
$t = \sqrt{2 \times 1.2/9.81} = \qty{0.49}{s}$; a disparada cai a
$300 \times 0.49 \approx \qty{1.5e2}{m}$ de distância. Testa-se a
independência dos movimentos horizontal e vertical.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:11}
$\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180}-2\alpha) =
\sin(2\alpha)$: mesmo $R$. Aqui $\sin(2\alpha) = gR/v_0^2 =
9.81 \times 50/625 = 0.785$: $\alpha = \ang{25.9}$ ou \ang{64.1}.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:12}
$R_{\max} = v_0^2/g = \qty{6.5}{m}$ (a \ang{45});
$h_{\max} = v_0^2/2g = \qty{3.3}{m}$ (na vertical).
$\sin(2\alpha) = 9.81 \times 5.0/64 = 0.766$: $\alpha = \ang{25.0}$ ou
\ang{65.0}.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:13}
$y(15) = 15\tan\ang{35} - \dfrac{9.81 \times 15^2}
{2 \times 18^2 \cos^2\ang{35}} = 10.5 - 5.1 = \qty{5.4}{m}$: ela passa
cerca de \qty{1.4}{m} acima da cerca de \qty{4.0}{m}.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:14}
(a) O arrasto equilibra o \omterm{def:g10:universal-gravitation:weight}{peso}: $80 \times 9.81 \approx \qty{7.8e2}{N}$.
(b) $v(t)$ sobe com inclinação cada vez menor e estabiliza em
\qty{55}{m/s}. (c) No vácuo:
$\sqrt{2 \times 9.81 \times 2000} \approx \qty{2.0e2}{m/s}$, vinte vezes
os \qty{9}{m/s} reais --- as gotas de chuva passam quase toda a queda na
\omterm{def:g12:projectile-motion:terminal}{velocidade limite}.
\end{solution}

\begin{solution}{exo:g12:projectile-motion:15}
Em $t^* = D/(v_0\cos\alpha)$ o dardo está na altura
$v_0\sin\alpha\, t^* - \tfrac12 g t^{*2} = D\tan\alpha - \tfrac12 g
t^{*2} = H - \tfrac12 g t^{*2}$, exatamente o
$y = H - \tfrac12 g t^{*2}$ do macaco que se soltou. Os dois pendem
$\tfrac12 gt^2$ abaixo dos pontos que ocupariam sem gravidade --- eles se
encontram para qualquer $v_0$.
\end{solution}

\begin{solution}{pb:g12:projectile-motion:1}
\textbf{1.} Só o \omterm{def:g10:universal-gravitation:weight}{peso} (ar desprezado); pela \omterm{thm:g12:newtons-laws:second}{segunda lei de Newton}:
$m\vect a = m\vect g$, logo $\vect a = (0,\,-g)$.

\textbf{2.} $v_x = v_0\cos\alpha$, $v_y = v_0\sin\alpha - gt$.

\textbf{3.} $x = v_0\cos\alpha\, t$,
$y = v_0\sin\alpha\, t - \tfrac12 gt^2$.

\textbf{4.} $t = x/(v_0\cos\alpha)$:
$y = x\tan\alpha - g x^2/(2v_0^2\cos^2\alpha)$.

\textbf{5.} $v_0^2 = \dfrac{g d^2}{2\cos^2\alpha\,(d\tan\alpha -
1.05)} = \dfrac{173.0}{0.826 \times 3.96} = 52.9$:
$v_0 = \qty{7.28}{m/s} \approx \qty{26}{km/h}$.

\textbf{6.} $t = d/(v_0\cos\alpha) = 4.20/4.68 = \qty{0.90}{s}$.

\textbf{7.} Ponto mais alto em $t = v_0\sin\alpha/g = \qty{0.57}{s}$, à
altura $2.00 + (v_0\sin\alpha)^2/2g = 2.00 + 1.58 = \qty{3.58}{m}$;
$0.57 < 0.90$: descendo no aro.

\textbf{8.} $y(0.90) = 0.90\tan\ang{50} - 0.182 = \qty{0.89}{m}$ acima da
soltura, ou seja, \qty{2.89}{m} acima do piso: \qty{0.29}{m} acima dos
dedos.

\textbf{9.} $v_x = \qty{4.68}{m/s}$;
$v_y = 5.57 - 9.81 \times 0.90 = \qty{-3.24}{m/s}$;
$v = \qty{5.69}{m/s}$, a $\arctan(3.24/4.68) = \ang{34.7}$ abaixo da
horizontal.

\textbf{10.} O aro é um anel horizontal: a bola precisa cruzar o plano
dele vindo de cima. Quanto mais íngreme a entrada, maior parece a abertura
do anel ao longo da velocidade --- mais margem de erro.

\textbf{11.} $1.05/26.0 = 4\%$: desprezível. $y(t_f) = 0$ dá
$t_f\,(v_0\sin\alpha - \tfrac12 g t_f) = 0$, logo
$t_f = 2v_0\sin\alpha/g$.

\textbf{12.} $R = v_0\cos\alpha\, t_f = v_0^2\sin(2\alpha)/g$; máximo em
$\alpha = \ang{45}$.

\textbf{13.} $v_0 = \sqrt{gR} = \sqrt{9.81 \times 26.0} =
\qty{16.0}{m/s} \approx \qty{57}{km/h}$.

\textbf{14.} $R_{\max} = 17.5^2/9.81 = \qty{31.2}{m} > \qty{26.0}{m}$:
viável, com \qty{5}{m} de \omterm{def:g12:projectile-motion:range}{alcance} de sobra.

\textbf{15.} $\sin(2\alpha) = 255.1/306.3 = 0.833$:
$\alpha = \ang{28.2}$ ou \ang{61.8}; \omterm{def:g12:projectile-motion:range}{tempos de voo}
$2v_0\sin\alpha/g = \qty{1.69}{s}$ e \qty{3.14}{s}. Só o arremesso
rasante bate o cronômetro de \qty{2.0}{s}.

\textbf{16.} $v_0^2 = \dfrac{9.81 \times 36}{2\cos^2\ang{75}\,
(6.0\tan\ang{75} - 1.05)} = \dfrac{353.2}{2.86} = 123.5$:
$v_0 = \qty{11.1}{m/s}$.

\textbf{17.} $y(5.55) = 5.55\tan\ang{75} - 18.26 = \qty{2.45}{m}$ acima
da soltura, ou seja, \qty{4.45}{m} acima do piso: \qty{0.50}{m} acima da
borda superior da tabela.

\textbf{18.} Ponto mais alto a
$2.00 + (11.1\sin\ang{75})^2/2g \approx \qty{7.9}{m}$; voo de
$6.0/(11.1\cos\ang{75}) = \qty{2.1}{s}$ --- dois segundos inteiros com a
bola pendurada a \qty{8}{m} de altura.

\textbf{19.} O arrasto encurta todo arremesso e torna a descida mais
íngreme; o efeito cresce com a velocidade, de modo que o arremesso de
quadra inteira a \qty{17.5}{m/s} é o mais castigado --- o \omterm{def:g12:projectile-motion:range}{alcance} real
dele fica alguns \omterm{def:g10:orders-of-magnitude:unit}{metros} aquém de \qty{31}{m}, comendo a margem.

\textbf{20.} \emph{O veredito da cesta no estouro do cronômetro}: o
arremesso de \qty{26.0}{m} exige \qty{16.0}{m/s} e um braço fornece cerca
de \qty{17.5}{m/s}, uma margem de \omterm{def:g12:projectile-motion:range}{alcance} de \qty{5}{m} ($\approx 20\%$)
que a resistência do ar apara, mas não apaga. Possível --- e é
exatamente por isso que, algumas vezes por temporada, ela entra.
\end{solution}
