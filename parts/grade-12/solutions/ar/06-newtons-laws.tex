\section*{الفصل \ref{ch:g12:newtons-laws} --- قوانين نيوتن}

\begin{solution}{exo:g12:newtons-laws:1}
(أ) $p = 70 \times 10 = \qty{7.0e2}{kg.m/s}$.
(ب) $v = \qty{13.9}{m/s}$: $p \approx \qty{1.8e4}{kg.m/s}$.
(ج) $p = 0.0080 \times 800 = \qty{6.4}{kg.m/s}$. فالسيارة $>$ العدّاء $>$
الرصاصة: فخمس رتب من القدر في الكتلة أكثر مما تقدر \omterm{def:g12:kinematics-2d:velocity}{السرعة} على شرائه.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:2}
$N = m(g + a_z)$: (أ) $70 \times 11.01 = \qty{771}{N}$؛
(ب) $a_z = 0$، ومنه $N = mg = \qty{687}{N}$ --- أي القانون الأول: \omterm{def:g12:kinematics-2d:velocity}{فالسرعة}
ثابتة والقوى تتعادل؛ (ج) $70 \times 8.61 = \qty{603}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:3}
$a = g\sin 30^\circ = \qty{4.9}{m/s^2}$؛
$v = \sqrt{2 \times 4.905 \times 3.0} = \qty{5.4}{m/s}$. وتتلاشى الكتلة
بين $ma$ و$mg\sin\alpha$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:4}
الانعكاس: $\Delta p = 0.058 \times (25 + 15) = \qty{2.3}{kg.m/s}$؛
$F = 2.32/0.0050 = \qty{4.6e2}{N}$ --- أي نحو $800$ ضعف
\omterm{def:g10:universal-gravitation:weight}{الوزن} \qty{0.57}{N}، ولهذا تُهمل \omterm{def:g11:fundamental-interactions:four}{الجاذبية} أثناء التماس.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:5}
\omterm{def:g10:universal-gravitation:weight}{الوزن} (الأرض على الكتاب، إلى أسفل): شريكه = الكتاب يجذب الأرض، إلى أعلى.
\omterm{def:g12:newtons-laws:friction}{والقوة الناظمية} (الطاولة على الكتاب، إلى أعلى): شريكها = الكتاب يضغط الطاولة،
إلى أسفل. أما \omterm{def:g10:universal-gravitation:weight}{الوزن} \omterm{def:g12:newtons-laws:friction}{والقوة الناظمية} فيؤثران في الجسم \emph{نفسه} وينتميان إلى
تفاعلين مختلفين --- ولا يتساويان هنا إلا لأن $\vect a = \vect 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:6}
$N = mg = \qty{490}{N}$. (أ) $F > \mu_s N = \qty{221}{N}$.
(ب) $a = (220.7 - 0.35 \times 490.5)/50 = \qty{0.98}{m/s^2}$.
(ج) $F = \mu_k N = \qty{172}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:7}
(أ) تنزلق حين $\tan\alpha > \mu_s$: $\alpha = \arctan 0.45 \approx
24^\circ$. (ب) $a = g(\sin 30^\circ - 0.35\cos 30^\circ) =
9.81 \times 0.197 = \qty{1.9}{m/s^2}$. (ج) نعم: فمتى انزلقت،
هبط \omterm{def:g10:inertia:inventory}{الاحتكاك} إلى $\mu_k < \mu_s$، ومنه
$a = g\cos\alpha\,(\tan\alpha - \mu_k) \approx \qty{0.9}{m/s^2} > 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:8}
$a = m_2 g/(m_1 + m_2) = 19.62/5.0 = \qty{3.9}{m/s^2}$؛
$T = m_1 a = \qty{11.8}{N}$. فالكتلة المعلّقة تتسارع إلى أسفل،
ومن ثم تتجه \omterm{def:g11:forces-and-motion:netforce}{القوة المحصّلة} عليها إلى أسفل: فيجب أن يقلّ $T$ عن $m_2 g$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:9}
$v_{\max} = \sqrt{\mu_s g R}$: الجافّ
$\sqrt{0.70 \times 9.81 \times 50} = \qty{18.5}{m/s} \approx
\qty{67}{km/h}$؛ والمبتلّ $\qty{14.0}{m/s} = \qty{50}{km/h}$. فكلٌّ من
$mv^2/R$ و$\mu_s mg$ يحمل $m$: فتتلاشى.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:10}
عموديًّا: $T\cos\theta = mg$، ومنه
$T = 2.943/\cos 25^\circ = \qty{3.2}{N}$. ونصف القطر $R = L\sin\theta =
\qty{0.507}{m}$؛ وشعاعيًّا: $T\sin\theta = mv^2/R$، ومنه
$v = \sqrt{gR\tan\theta} = \sqrt{9.81 \times 0.507 \times 0.466}
= \qty{1.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:11}
$a_z = (N - mg)/m$ مع $mg = \qty{49.05}{N}$: $+\qty{1.0}{m/s^2}$،
$0$، $-\qty{1.0}{m/s^2}$. فرحلة صاعدة مثلًا\ : تسارع، ثم انسياب، ثم
كبح. ونعم: فلو دخل نازلًا أصلًا، لعنت القراءات نفسها كبحًا، ثم
انسيابًا، ثم تسارعًا مجددًا نازلًا --- فالميزان يقرأ التسارع لا
\omterm{def:g12:kinematics-2d:velocity}{السرعة}.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:12}
(أ) $0 = 60V + 4.0 \times 8.0$: $V = \qty{-0.53}{m/s}$.
(ب) الكرة عند $8.0 + V$ بالنسبة إلى الأرض: $0 = 60V + 4.0(8.0 + V)$، ومنه
$V = -32/64 = \qty{-0.50}{m/s}$ وتطير الكرة \omterm{def:g12:kinematics-2d:velocity}{بسرعة} \qty{7.5}{m/s}.
(ج) دفع الكرة على يديها --- أي شريك رميتها في القانون
الثالث.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:13}
بلا \omterm{def:g10:inertia:inventory}{احتكاك}: $N\cos\theta = mg$ (عموديًّا) و$N\sin\theta =
mv^2/R$ (شعاعيًّا)؛ وبالقسمة، $\tan\theta = v^2/(gR) = 400/784.8$:
$\theta \approx 27^\circ$. وإذا كانت أبطأ: مالت السيارة إلى الانزلاق نازلةً على
الميل، فوجب أن يتجه \omterm{def:g10:inertia:inventory}{الاحتكاك} صاعدًا على المنحدر؛ وإذا كانت أسرع: فالعكس.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:14}

\begin{solution}{exo:g12:newtons-laws:15}
(أ) $m_2 g > \mu_s m_1 g$: $m_2 > 0.35 \times 4.0 = \qty{1.4}{kg}$.
(ب) $a = (m_2 - \mu_k m_1)\,g/(m_1 + m_2) = 1.0 \times 9.81/6.0 =
\qty{1.6}{m/s^2}$؛ $T = m_2(g - a) = \qty{16.4}{N}$.
(ج) يعطي $\mu_k = 0$ النتيجة $a = m_2 g/(m_1 + m_2)$: أي صيغة
\cref{exo:g12:newtons-laws:8}.
\end{solution}

\begin{solution}{pb:g12:newtons-laws:1}
\textbf{1.} الجملة: المقصورة $+$ العربة (مع $+$ مسافرًا). والجبل
مثبَّت على الأرض، وهو عُطالي بدقة ممتازة. والقوى:
\omterm{def:g10:universal-gravitation:weight}{الوزن} $M\vect g$ (إلى أسفل)، \omterm{def:g12:newtons-laws:friction}{والقوة الناظمية} $\vect N$ لكابل الحمل
(عمودية عليه)، \omterm{def:g10:inertia:inventory}{وشدّ} الجرّ $\vect T$ (صاعدًا على المنحدر).

\textbf{2.} في السكون، القانون الأول: $\sum \vect F = \vect 0$. وعلى امتداد
المنحدر: $T = Mg\sin\alpha$؛ وعموديًّا: $N = Mg\cos\alpha$.

\textbf{3.} $T = 4000 \times 9.81 \times 0.500 = \qty{19.6}{kN}$؛
$N = 4000 \times 9.81 \times 0.866 = \qty{34.0}{kN}$.

\textbf{4.} فارغة: $T = \qty{9.8}{kN}$. وكل \omterm{def:g10:inertia:force}{قوة} في التوازن
تتناسب مع $M$، ومن ثم يتناسب $T$ معها أيضًا.

\textbf{5.} على امتداد المنحدر، $Ma = Mg\sin\alpha$: فتتلاشى $M$، و
$a = g\sin\alpha = \qty{4.9}{m/s^2}$ نازلًا على المنحدر.

\textbf{6.} $T' = M(g\sin\alpha + a) = 4000 \times 5.505 =
\qty{22.0}{kN}$.

\textbf{7.} $a/(g\sin\alpha) = 0.60/4.905 \approx 12\%$.

\textbf{8.} $t = v/a = \qty{10}{s}$؛ $d = v^2/(2a) = \qty{30}{m}$.

\textbf{9.} المركّبة العمودية \omterm{def:g10:inertia:force}{للقوة} $\vect a$: $a\sin\alpha =
\qty{0.30}{m/s^2}$، ومنه $N = m(g + a\sin\alpha) = 80 \times 10.11 =
\qty{809}{N}$ في مقابل $mg = \qty{785}{N}$: أي أثقل بمقدار $3\%$.

\textbf{10.} \omterm{def:g11:forces-and-motion:friction}{الاحتكاك السكوني} للأرضية، وهو أفقي:
$ma\cos\alpha = 80 \times 0.52 \approx \qty{42}{N}$. والميل يجعل دفع الأرضية
المجموع يمرّ بمركز كتلتها، فلا يقلبها.

\textbf{11.} $a_T = dv/dt = 0$ (لأن \omterm{def:g12:kinematics-2d:velocity}{السرعة} ثابتة)؛
و$a_N = v^2/R = 36/40 = \qty{0.90}{m/s^2}$، مصوَّبًا نحو المركز ---
أي إلى أسفل مباشرة عند القمة.

\textbf{12.} المحور الناظمي (وإلى أسفل موجب): $Mg - N' = Mv^2/R$، ومنه
$N' = M(g - v^2/R) = 4000 \times 8.91 = \qty{35.6}{kN}$.

\textbf{13.} $N = m(g - v^2/R) = 80 \times 8.91 = \qty{713}{N}$:
أي أخفّ بمقدار $v^2/(gR) = 0.90/9.81 \approx 9\%$.

\textbf{14.} $N' = 0$ عند $v = \sqrt{gR} = \qty{19.8}{m/s} \approx
\qty{71}{km/h}$ --- أي أكثر من ثلاثة أضعاف الانسياب: فتركب العربة
الكابل بهامش واسع.

\textbf{15.} \omterm{thm:g12:newtons-laws:third}{قانون نيوتن الثالث}: \qty{35.6}{kN}، ضاغطةً إلى أسفل على
الكابل (ومن ثم على البرج) --- وهي أقلّ قليلًا من \omterm{def:g10:universal-gravitation:weight}{الوزن} في السكون،
مخفَّفةً بالطيران المنحني.

\textbf{16.} $F = \Delta p/\Delta t = 4000 \times 6.0/1.5 =
\qty{16}{kN}$، عكس الحركة ($a = \qty{4.0}{m/s^2}$).

\textbf{17.} $f = ma = 80 \times 4.0 = \qty{320}{N}$؛ والسقف
$\mu_s mg = 0.40 \times 785 = \qty{314}{N} < \qty{320}{N}$: فالقدمان
وحدهما تنزلقان --- فأمسِك المقبض.

\textbf{18.} $0 = M_0 V + m(v_{\text{النسبية}} + V)$:
$V = -80 \times 2.5/2080 = \qty{-0.096}{m/s}$؛ ويتحرك المسافر \omterm{def:g12:kinematics-2d:velocity}{بسرعة}
$2.5 - 0.096 \approx \qty{2.4}{m/s}$ بالنسبة إلى الأرض.

\textbf{19.} الصاروخ: \omterm{def:g10:inertia:force}{فقوة} دفعه تثبّتها الكتلة المقذوفة في
\omterm{def:g10:orders-of-magnitude:unit}{الثانية} مضروبةً في \omterm{def:g12:kinematics-2d:velocity}{سرعة} القذف بالنسبة إليه.

\textbf{20.} أكبر \omterm{def:g10:inertia:inventory}{شدّ} لقيناه هو \omterm{def:g10:inertia:inventory}{شدّ} الانطلاق
$T' = M \times \qty{5.505}{N/kg}$ (فالسكون يعطي $Mg\sin\alpha$، والقمة أقلّ بعدُ). والحدّ: $T' \leq \qty{20}{kN}$، ومنه
$M \leq 20000/5.505 = \qty{3.63e3}{kg}$: فالحمولة
\qty{1.63e3}{kg}، أي\ $20$ مسافرًا كتلة كلٍّ \qty{80}{kg} (لا $25$).
واللافتة: \emph{20 شخصًا}؛ ومنه $M = \qty{3.6e3}{kg}$،
و$T' = \qty{19.8}{kN}$، وعامل الأمان $100/19.8 \approx 5.1 \geq 5$.
\end{solution}
