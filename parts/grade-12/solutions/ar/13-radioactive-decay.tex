\section*{الفصل \ref{ch:g12:radioactive-decay} --- التفكك الإشعاعي}

\begin{solution}{exo:g12:radioactive-decay:1}
$T_{1/2} = \ln 2/\lambda = 0.693/\num{1.0e-6} = \qty{6.9e5}{s}
\approx 8.0$ يومًا --- أي ساعة اليود-131.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:2}
$T_{1/2} = 3.8 \times \num{86400} = \qty{3.28e5}{s}$؛
$\lambda = 0.693/\num{3.28e5} = \qty{2.1e-6}{s^{-1}}$؛
$\mathcal A = \lambda N = \num{2.1e-6} \times \num{1.0e10} \approx
\qty{2.1e4}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:3}
$\dd N/\dd t = N_0(-\lambda)e^{-\lambda t} = -\lambda N$ ---
فالمعادلة تصحّ؛ $N(0) = N_0 e^0 = N_0$؛
$N(T_{1/2})/N_0 = e^{-\lambda \ln 2/\lambda} = e^{-\ln 2} =
\tfrac12$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:4}
$90$ سنة $= 3\,T_{1/2}$: $\mathcal A = \num{4.0e5}/2^3 =
\qty{5.0e4}{Bq}$. و$300$ سنة $= 10\,T_{1/2}$:
$\mathcal A = \num{4.0e5}/1024 \approx \qty{3.9e2}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:5}
الميل $= (18.85 - 20.00)/10 = -0.115$، ومنه
$\lambda = \qty{0.115}{h^{-1}} = 0.115/3600 = \qty{3.2e-5}{s^{-1}}$؛
$T_{1/2} = 0.693/0.115 = 6.0$ ساعة --- أي التكنيسيوم-99m.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:6}
$N = (1.0/226) \times \num{6.02e23} = \num{2.7e21}$؛
$T_{1/2} = 1600 \times \num{3.156e7} = \qty{5.05e10}{s}$، ومنه
$\lambda = \qty{1.37e-11}{s^{-1}}$؛
$\mathcal A = \lambda N \approx \qty{3.7e10}{Bq}$ --- أي كوري واحد، وهو
\omterm{def:g11:nucleus-radioactivity:activity}{نشاط} غرام الراديوم عند ماري كوري.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:7}
كلاهما يتوقع بقاء النصف: $10$، و\num{1.0e20}. والتقلبات
$\sim 1/\sqrt{n}$ من العدّة: فمع $10$ تفككًا، نحو $30\%$ ---
فلا يفاجئ $7$ ولا $13$؛ ومع \num{1.0e20}، نحو
$10^{-10}$. ولا تطيع القانونَ بدقة قابلة للقياس إلا العيّنة
الكبيرة.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:8}
$m = 140 \times \num{1.2e-4} = \qty{1.7e-2}{g}$؛
$N = (0.0168/40) \times \num{6.02e23} = \num{2.5e20}$.
$\lambda = 0.693/(\num{1.25e9} \times \num{3.156e7}) =
\qty{1.76e-17}{s^{-1}}$؛ $\mathcal A = \lambda N \approx
\qty{4.4e3}{Bq}$ --- أي نحو نصف مجموع الجسم (ومعظم الباقي
كربون-14).
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:9}
$\lambda = 0.693/6.0 = \qty{0.116}{h^{-1}}$. وعند $15$ ساعة:
$\mathcal A = 500\,e^{-0.116 \times 15} = 500 \times 0.177 \approx
\qty{88}{MBq}$. ودون \qty{1}{MBq}:
$t = \ln(500)/0.116 \approx \qty{54}{hours}$ --- أي نحو $9$
عمر نصف.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:10}
الفحم النباتي: $3.4/13.6 = \tfrac14$، أي $2$ عمر نصف بالضبط:
$t = \num{1.15e4}$ سنة. والعظم:
$t = \num{5730} \times \ln(1/0.30)/\ln 2 = \num{5730} \times 1.74
\approx \num{1.0e4}$ سنة.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:11}
$\lambda = 0.693/3.8 = \qty{0.182}{d^{-1}}$؛
$t = \ln(100)/\lambda = 4.61/0.182 \approx 25$ يومًا (أي نحو $6.6$
عمر نصف). لكن سلسلة اليورانيوم في الغرانيت والتربة
المحيطين تولّد الرادون باستمرار: فتهوية مرة واحدة يُبطلها
مرور أسابيع، ومن ثم يجب أن تكون التهوية دائمة.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:12}
يعطي $N/N_0 = e^{-\lambda t}$ المقدارَ $\lambda t = \ln(N_0/N)$، و
يعطي $\lambda = \ln 2/T_{1/2}$ المقدارَ $t = T_{1/2}\ln(N_0/N)/\ln 2$. وهنا
$t = \num{5730} \times \ln(20)/\ln 2 = \num{5730} \times 4.32
\approx \num{2.5e4}$ سنة.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:13}
$N_{\mathrm K} = N_0 e^{-\lambda t}$ و
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$، ومنه
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$. والنسبة $3.0$:
$e^{\lambda t} = 4$، ومنه $\lambda t = 2\ln 2$ و
$t = 2\,T_{1/2} = \num{2.5e9}$ سنة.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:14}
$\ln \mathcal A = 5.99$، $5.42$، $4.84$، $4.26$، $3.69$: أي هبوط قدره
$0.57$--$0.58$ في كل \qty{60}{s} --- فهي مصطفّة. والميل:
$\lambda = 2.30/240 = \qty{9.6e-3}{s^{-1}}$؛
$T_{1/2} = 0.693/\num{9.6e-3} \approx \qty{72}{s}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:15}
(أ) $2^{-10} \approx \num{1e-3}$. (ب) $(1/2)^8 = 1/256 \approx
0.4\%$. (ج) $1000/8 = 125$ منتظَرة --- لكن القانون صامت عن
أيّها: فهو يتنبأ بالأعداد تنبؤًا مضبوطًا، وبالأفراد لا يتنبأ البتة.
\end{solution}

\begin{solution}{pb:g12:radioactive-decay:1}
\textbf{1.} التبادل مع الجو (بالأكل والتنفس) يبقي
النسبة مملوءة عند القيمة الجوّية؛ والموت يوقف
الأخذ فيتولّى التفكك.

\textbf{2.} $T_{1/2} = \num{5730} \times \num{3.156e7} =
\qty{1.81e11}{s}$، ومنه $\lambda = 0.693/\num{1.81e11} =
\qty{3.8e-12}{s^{-1}}$.

\textbf{3.} $\dd N/\dd t = -\lambda N_0 e^{-\lambda t} = -\lambda N$،
و$N(0) = N_0$.

\textbf{4.} $\mathcal A_0 = 13.6/60 = \qty{0.227}{Bq}$ لكل غرام؛
$N = \mathcal A_0/\lambda = 0.227/\num{3.8e-12} \approx \num{5.9e10}$
\omterm{def:g11:fundamental-interactions:ladder}{ذرة} لكل غرام.

\textbf{5.} $(1/12) \times \num{6.02e23} = \num{5.0e22}$ \omterm{def:g11:fundamental-interactions:ladder}{ذرة}
كربون؛ والنسبة $\num{5.9e10}/\num{5.0e22} \approx \num{1.2e-12}$
--- أي واحدة في كل $10^{12}$، كما وُعد.

\textbf{6.} يعطي $\mathcal A = \mathcal A_0 e^{-\lambda t}$ المقدارَ
$\lambda t = \ln(\mathcal A_0/\mathcal A)$، ويحوّله $\lambda =
\ln 2/T_{1/2}$ إلى
$t = T_{1/2}\ln(\mathcal A_0/\mathcal A)/\ln 2$.

\textbf{7.} $t = \num{5730} \times \ln(13.6/9.1)/\ln 2 =
\num{5730} \times 0.580 \approx \num{3.3e3}$ سنة.

\textbf{8.} موت الخشب --- أي قطع الشجرة، لا
النحت؛ فالتمثال المنحوت من خشب قديم (أو المزوَّر المنحوت من خشب
عتيق) يسبق تاريخ مادته أو يتأخر عنه.

\textbf{9.} $t = \num{5730} \times \ln(13.6/10.4)/\ln 2 =
\num{5730} \times 0.387 \approx \num{2.2e3}$ سنة --- وهي متسقة
تمامًا مع ناسخ من قبل اثنين وعشرين قرنًا.

\textbf{10.} $\Delta t \approx \Delta\mathcal A/(\lambda\mathcal A)$
مع $\lambda = \ln 2/5730 = \num{1.21e-4}$ في السنة:
$\Delta t = (0.2/10.4)/\num{1.21e-4} \approx \num{1.6e2}$ سنة ---
أي تاريخ مضبوط إلى قرن أو قرنين.

\textbf{11.} $t = \num{5730} \times \ln(13.6/13.4)/\ln 2 \approx
120$ سنة --- أي لا تُميَّز من الحديث ضمن ارتياب قدره $\pm 160$
سنة: فالرقّ حديث، و``الأثري'' مزوَّر.

\textbf{12.} يقتضي $13.6/2^n < 0.2$ أن $2^n > 68$: أي $n = 7$ عمر نصف
(بما أن $2^7 = 128$)، أي نحو $\num{4.0e4}$ سنة.

\textbf{13.} عشرة أعمار نصف: أي نحو \num{5.7e4} سنة ---
وهو الأفق العملي، أي نحو ستين ألف سنة.

\textbf{14.} $\num{6.6e7}/\num{5730} \approx \num{11500}$ عمر نصف.
وذرات الغرام \num{5.9e10} تنفد بعد
$2^n > \num{5.9e10}$، أي\ $n \approx 36$ عمر نصف
($\approx \num{2e5}$ سنة): فقبل عمر الديناصورات بزمن طويل، لا تبقى \omterm{def:g11:fundamental-interactions:ladder}{ذرة}
كربون-14 واحدة --- فلا شيء يُعدّ.

\textbf{15.} ساعة أبطأ بكثير: أي البوتاسيوم-40 أو اليورانيوم-238،
بأعمار نصف تبلغ مليارات السنين.

\textbf{16.} الحمم المنصهرة تدع الأرغون يتفقّع خارجًا، ومن ثم تحمل البلورة
المتصلبة حديثًا بوتاسيومًا وصفرًا من الأرغون: فتبدأ النسبة
عند $0$ عند التصلب --- أي إن الساعة مصفَّرة.

\textbf{17.} كما في التمرين: $N_{\mathrm K} = N_0 e^{-\lambda t}$،
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$، ومنه
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.

\textbf{18.} $e^{\lambda t} = 1.60$:
$t = \num{1.25e9} \times \ln(1.60)/\ln 2 = \num{1.25e9} \times 0.678
\approx \num{8.5e8}$ سنة.

\textbf{19.} لا، مرتين: فعند \num{8.5e8} سنة يكون الكربون-14 قد قطع
$10^5$ عمر نصف --- أي الصمت؛ وعند \num{3.3e3} سنة يكون البوتاسيوم-40 قد
قطع \num{2.6e-6} من عمر نصف --- ونسبة الأرغون
$\approx \lambda t \sim \num{2e-6}$ صغيرة لا تُقاس. ولا تقرأ كل
ساعة إلا من نحو عُشر عمر نصف إلى نحو عشرة أعمار.

\textbf{20.} التمثال $\approx \num{3.3e3}$ سنة، والبردية
$\approx \num{2.2e3}$ سنة، والصخر $\approx \num{8.5e8}$ سنة ---
أي ثلاث ساعات، وقانون واحد: $N = N_0\,e^{-\lambda t}$، مقروءًا إلى الوراء.
\end{solution}
