\section*{الفصل \ref{ch:g12:kinematics-2d} --- التحريكيات في بعدين}

\begin{solution}{exo:g12:kinematics-2d:1}
$\vect v = (3.0, 4.0)$، وهي ثابتة؛ وقيمة \omterm{def:g12:kinematics-2d:velocity}{السرعة} $\sqrt{3.0^2 + 4.0^2} =
\qty{5.0}{m/s}$. \omterm{def:g10:relative-motion:trajectory}{والمسار}: المستقيم $y = \tfrac43 x$. \omterm{def:g12:kinematics-2d:velocity}{ومتجهة
سرعة} ثابتة: أي \omterm{def:g12:kinematics-2d:uniform}{حركة مستقيمة منتظمة}.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:2}
مستقيم، وتباعدات متساوية: أي \omterm{def:g12:kinematics-2d:uniform}{حركة مستقيمة منتظمة}؛
و$v = 0.35/0.10 = \qty{3.5}{m/s}$. أما التباعدات المتسعة فكانت ستعني أن
\omterm{def:g12:kinematics-2d:velocity}{السرعة} تتزايد.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:3}
$v(t) = 4.0\,t$، $a = \qty{4.0}{m/s^2}$: أي تسارع ثابت،
فحركة مستقيمة متسارعة بانتظام. و$v(3.0) = \qty{12}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:4}
$f = 1/8.0 = \qty{0.125}{Hz}$؛
$v = 2\pi \times 4.5/8.0 \approx \qty{3.5}{m/s}$؛
$a = v^2/R = 3.53^2/4.5 \approx \qty{2.8}{m/s^2}$، متجهًا نحو
محور الدوّامة.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:5}
الميل: $a = (10.0 - 2.0)/4.0 = \qty{2.0}{m/s^2}$. والمساحة (شبه منحرف):
$d = \tfrac12 (2.0 + 10.0) \times 4.0 = \qty{24}{m}$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:6}
$v = \qty{88.9}{m/s}$: $a_n = 88.9^2/6000 \approx \qty{1.3}{m/s^2}
\approx 0.13\,g$. وبما أن $a_n = v^2/R$، فإن $v$ العالية تقتضي $R$ هائلًا
لإبقاء التسارع الجانبي (\omterm{def:g10:inertia:force}{والقوة} على السكة و
الركاب) مقبولًا --- والميل يغيّر من يوفّره، لا مقداره.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:7}
$v(t) = -2.4\,t + 8.0$؛ $a = \qty{-2.4}{m/s^2}$، وهو ثابت. والانعكاس
عند $v = 0$: $t = 8.0/2.4 \approx \qty{3.3}{s}$، عند
$x \approx \qty{16.3}{m}$. وقبله: إلى الأمام متباطئًا؛ وبعده: إلى الوراء
متسارعًا --- وهي متسارعة بانتظام طول الوقت.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:8}
$v_1 = 0.28/0.40 = \qty{0.70}{m/s}$؛
$v_2 = (0.54 - 0.10)/0.40 = \qty{1.10}{m/s}$؛
$v_3 = (0.88 - 0.28)/0.40 = \qty{1.50}{m/s}$. فتكسب \omterm{def:g12:kinematics-2d:velocity}{السرعة}
\qty{0.40}{m/s} كل \qty{0.20}{s}: أي $a = \qty{2.0}{m/s^2}$، وهو ثابت
--- فحركة مستقيمة متسارعة بانتظام.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:9}
$T = \qty{2.36e6}{s}$: $v = 2\pi \times \num{3.84e8}/\num{2.36e6}
\approx \qty{1.0e3}{m/s}$؛
$a_n = v^2/R \approx \qty{2.7e-3}{m/s^2}$. والنسبة:
$g/a_n \approx 3600 = 60^2$ --- \omterm{def:g11:fundamental-interactions:four}{فالجاذبية}، مخفَّفةً كمقلوب
مربع المسافة (\cref{ch:g10:universal-gravitation})، تفسّر
دوران القمر بالضبط.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:10}
$f = 1200/60 = \qty{20}{Hz}$، $T = \qty{0.050}{s}$؛
$v = 2\pi R f = 2\pi \times 0.25 \times 20 \approx \qty{31}{m/s}$؛
$a_n = v^2/R \approx \qty{3.9e3}{m/s^2} \approx 400\,g$. فجدار الحلّة
يوفّر $a_n$ للغسيل؛ أما عبر الثقوب فلا شيء يدفع
الماء إلى الداخل، فيطير على المماسّ.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:11}
$v_0 = \qty{25}{m/s}$: $t_s = 25/7.5 \approx \qty{3.3}{s}$؛
$d = v_0^2/(2a) = 625/15 \approx \qty{42}{m}$. والمخطط: قطعة
مستقيمة من \qty{25}{m/s} نزولًا إلى $0$ عند $t_s$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:12}
$\vect v = (-R\omega\sin\omega t,\, R\omega\cos\omega t)$، وناظمها
ثابت $R\omega$؛ وجداؤها السلّمي مع $\vect{OM}$ ينعدم،
ومن ثم فهي عمودية على نصف القطر، أي مماسّة. وبالاشتقاق
مرة أخرى، $\vect a = -\omega^2\vect{OM}$: أي نحو $O$، وناظمها
$\omega^2 R = (v/R)^2 R = v^2/R$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:13}
$a_t = \qty{2.5}{m/s^2}$ (إلى الوراء)، $a_n = 20^2/80 =
\qty{5.0}{m/s^2}$؛
$a = \sqrt{2.5^2 + 5.0^2} \approx \qty{5.6}{m/s^2}$؛ والزاوية مع
جهة الحركة: $\theta = 180^\circ - \arctan(5.0/2.5) \approx 117^\circ$
(أي\ $63^\circ$ عن الاتجاه الخلفي المستقيم). و$5.6 < 7.0$: أي ضمن التماسك،
ولا يفيض إلا القليل.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:14}
الانطلاق: $t_1 = 20/1.2 \approx \qty{17}{s}$،
$d_1 = 20^2/(2 \times 1.2) \approx \qty{167}{m}$. والانسياب:
\qty{40}{s}، $\qty{800}{m}$. والكبح: $t_3 = 20/1.5 \approx
\qty{13}{s}$، $d_3 = 400/3.0 \approx \qty{133}{m}$. والمجموع:
$\qty{1.1e3}{m}$ في \qty{70}{s}؛ والمتوسط $1100/70 \approx
\qty{16}{m/s}$ (\qty{57}{km/h}). و$v(t)$ شبه منحرف مساحته،
$\tfrac12(70 + 40) \times 20 = \qty{1100}{m}$، تتحقق من المجموع.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:15}
$v = 2\pi \times \num{4.22e7}/\num{86164} \approx \qty{3.1e3}{m/s}$؛
$a_n = v^2/R \approx \qty{0.22}{m/s^2}$. وتوفّره جاذبية الأرض على تلك
المسافة؛ \omterm{def:g12:kinematics-2d:ucm}{والدور} يساوي \omterm{def:g10:signals-and-waves:period}{دور} دوران الأرض، ومن ثم يدور
\omterm{def:g10:universal-gravitation:satellite}{التابع} والأرض معًا فيتعلّق فوق نقطة استوائية
واحدة (\cref{ch:g12:satellites-kepler}).
\end{solution}

\begin{solution}{pb:g12:kinematics-2d:1}
\textbf{1.} الحركة نسبية (\cref{ch:g10:relative-motion}): فكل
موضع وكل \omterm{def:g12:kinematics-2d:velocity}{سرعة} يجب أن يسمّي مرجعه. وهنا: \omterm{def:g10:relative-motion:frames}{المرجع الأرضي}
(الطريق)، و$x$ على امتداد الطريق من موضع الصورة الأولى.

\textbf{2.} في كل فترة: $9.0/0.50 = \qty{18.0}{m/s}$ --- أي أربع
قيم متساوية.

\textbf{3.} طريق مستقيم، وانتقالات متساوية في أزمنة متساوية:
أي \omterm{def:g12:kinematics-2d:uniform}{حركة مستقيمة منتظمة}.

\textbf{4.} $v_0 = \qty{18.0}{m/s} = \qty{64.8}{km/h}$.

\textbf{5.} $\vect a = \vect 0$.

\textbf{6.} $v(t) = -a\,t + v_B$؛ $x(t) = -\tfrac12 a t^2 + v_B\,t$.

\textbf{7.} يعطي $v(t_s) = 0$ النتيجة $t_s = v_B/a$؛ ثم
$d_b = -\tfrac12 a (v_B/a)^2 + v_B^2/a = v_B^2/(2a)$.

\textbf{8.} $v_B = \sqrt{2 a d_b} = \sqrt{2 \times 6.0 \times 27} =
\sqrt{324} = \qty{18.0}{m/s}$ --- أي \omterm{def:g12:kinematics-2d:velocity}{السرعة} المصوَّرة بالضبط: فالسائق
لم يبطّئ قط قبل الكبح.

\textbf{9.} ردّ الفعل: $18.0 \times 1.0 = \qty{18}{m}$؛ والمجموع
$18 + 27 = \qty{45}{m}$.

\textbf{10.} $d_b = v_B^2/(2a) \propto v_B^2$: فمضاعفة $v_B$
تضرب $d_b$ في $4$. وعند \qty{50}{km/h} ($\qty{13.9}{m/s}$):
$d_b = 13.9^2/12.0 \approx \qty{16}{m}$.

\textbf{11.} \omterm{def:g12:kinematics-2d:velocity}{السرعة} ثابتة لكن اتجاه \omterm{def:g12:kinematics-2d:velocity}{متجهة السرعة} يدور:
فالمتجهة $\vect v$ ليست ثابتة، ومن ثم فالسيارة متسارعة.

\textbf{12.} ربع دورة: $2\pi R/4 = \qty{19.6}{m}$؛
$v = 19.6/2.5 \approx \qty{7.9}{m/s}$ (\qty{28}{km/h}).

\textbf{13.} $a_n = 7.85^2/12.5 \approx \qty{4.9}{m/s^2}$، نحو مركز
الدوّار.

\textbf{14.} $a_{\max} = \mu g = 0.70 \times 9.81 \approx
\qty{6.9}{m/s^2}$؛ $v_{\max} = \sqrt{\mu g R} =
\sqrt{6.87 \times 12.5} \approx \qty{9.3}{m/s}$ (\qty{33}{km/h}).
و$7.9 < 9.3$: أي ضمن التماسك --- وهو متسق مع المسرب النظيف الخالي من
الانزلاق.

\textbf{15.} $T = 2\pi R/v = 78.5/7.85 \approx \qty{10.0}{s};
f = \qty{0.10}{Hz}$.

\textbf{16.} $64.8 - 50 \approx \qty{15}{km/h}$ زيادةً --- أي نحو
$30\%$ فوق الحدّ.

\textbf{17.} يؤثر الكبح على مدى $41 - 18 = \qty{23}{m}$:
$v^2 = v_B^2 - 2ad = 324 - 2 \times 6.0 \times 23 = 48$، ومنه
$v \approx \qty{6.9}{m/s} \approx \qty{25}{km/h}$ عند الاصطدام.

\textbf{18.} عند \qty{13.9}{m/s}: $13.9 \times 1.0 + 13.9^2/12.0 =
13.9 + 16.1 = \qty{30.0}{m} < \qty{41}{m}$ --- فتتوقف السيارة
على بعد \qty{11}{m} قبل الشاحنة.

\textbf{19.} $d(v) = v\,t_r + v^2/(2a)$: مرةً خطيًّا (فردّ الفعل
يقطع $v t_r$ قبل أن تعمل المكابح)، ومرةً مربَّعة (فالجزء
الحركي $v^2/(2a)$) --- ولهذا تكلّف التجاوزات المعتدلة
كلفةً غير متناسبة.

\textbf{20.} تتفق كاميرا لوحة القيادة وآثار الانزلاق: \qty{65}{km/h} في
منطقة \qty{50}{km/h}؛ وعند الحدّ القانوني كانت السيارة ستتوقف
على بعد \qty{11}{m} قبل الشاحنة --- فالحادث هو \qty{15}{km/h}
الزائدة.
\end{solution}
