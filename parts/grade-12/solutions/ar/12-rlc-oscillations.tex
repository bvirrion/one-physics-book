\section*{الفصل \ref{ch:g12:rlc-oscillations} --- التذبذبات الكهربائية الحرة: RLC}

\begin{solution}{exo:g12:rlc-oscillations:1}
(أ) $\sqrt{LC} = \qty{1.0e-4}{s}$: $T_0 = \qty{0.63}{ms}$،
$f_0 \approx \qty{1.6}{kHz}$. (ب) $\sqrt{LC} = \qty{3.16e-7}{s}$:
$T_0 = \qty{2.0}{\micro s}$، $f_0 \approx \qty{0.50}{MHz}$ --- أي بعامل
$\sqrt{10^5} \approx 316$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:2}
$LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times \unit{C/A} =
\unit{s} \times \unit{s} = \unit{s^2}$، ومن ثم يكون $\sqrt{LC}$ بالثواني.
و$2\pi$ عدد خالص --- فلا بُعد له ليغيّره.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:3}
$Q_0 = CU_0 = \qty{2.8}{\micro C}$؛ $E = \tfrac12 CU_0^2 =
\tfrac12 \times \num{4.7e-7} \times 36 \approx \qty{8.5}{\micro J}$.
تتذبذب الشحنة عند $f_0 = 1/(2\pi\sqrt{LC})$، والتيار خارج \omterm{def:g12:oscillators-and-time:phase}{الطور} بربع
\omterm{def:g12:oscillators-and-time:oscillator}{دور}، \omterm{def:g10:energy-conservation:energy}{والطاقة} تتقاذف بين الصفيحتين \omterm{def:g12:rc-rl-circuits:inductor}{والوشيعة}، والمجموع
ثابت.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:4}
$Q_0 = \qty{2.0}{\micro C}$؛ $f_0 = 1/T_0 = \qty{250}{Hz}$؛
$q(T_0/2) = \qty{-2.0}{\micro C}$؛ $i(0) = 0$ (فجيب التمام مسطّح عند $t=0$)؛
$I_0 = 2\pi \times \num{2.0e-6}/\num{4.0e-3} \approx \qty{3.1}{mA}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:5}
كلها في \omterm{def:g12:rc-rl-circuits:capacitor}{المكثّفة} حين $q = \pm Q_0$، $i = 0$ ($t = 0$، $T_0/2$،
$T_0$، \dots): أي طرفا النوّاس. وكلها في \omterm{def:g12:rc-rl-circuits:inductor}{الوشيعة} حين $q = 0$،
$i = \pm I_0$ ($t = T_0/4$، $3T_0/4$، \dots): أي أخفض نقاط النوّاس
وأسرعها.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:6}
$d^2q/dt^2 = -(2\pi/T_0)^2\,Q_0\cos(2\pi t/T_0) = -(2\pi/T_0)^2 q$،
ومن ثم $L\,d^2q/dt^2 + q/C = q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$ من أجل كل $t$
بالضبط حين $(2\pi/T_0)^2 = 1/(LC)$، أي\ $T_0 = 2\pi\sqrt{LC}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:7}
$L = \dfrac{1}{4\pi^2 f_0^2 C} = \dfrac{1}{39.5 \times
(\num{1.98e5})^2 \times \num{2.2e-10}} \approx \qty{2.9}{mH}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:8}
$E = \tfrac12 CU_0^2 = \qty{5.0e-5}{J}$؛ ويعطي $\tfrac12 LI_0^2 = E$ المقدارَ
$I_0 = \sqrt{2E/L} = \sqrt{\num{1.0e-2}} = \qty{0.10}{A}$، ويُبلَغ أول مرة
عند $T_0/4 = \tfrac14 \times \qty{0.63}{ms} \approx
\qty{0.16}{ms}$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:9}
$E_C \propto \cos^2(2\pi t/T_0)$، و$\cos^2$ يتكرر في كل
\emph{نصف} دورة: أي امتلاءان \omterm{def:g12:rc-rl-circuits:capacitor}{للمكثّفة} في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور}، ومنه
$2f_0$. وعند $t = T_0/8$ يكون \omterm{def:g12:oscillators-and-time:phase}{الطور} $\pi/4$:
$\cos^2(\pi/4) = \tfrac12$، ومنه $E_C = E_L = E/2$.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:10}
النسبة $3.0/4.0 = 0.75$ في كل شبه \omterm{def:g12:oscillators-and-time:oscillator}{دور}: فالعظمى السادسة
$4.0 \times 0.75^6 \approx \qty{0.71}{V}$. \omterm{def:g10:energy-conservation:energy}{والطاقة} $\propto$ مربع \omterm{def:g12:oscillators-and-time:oscillator}{السعة}:
ومن ثم ينجو $0.75^2 \approx 56\%$ في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور}. أما
\omterm{def:g12:rlc-oscillations:regimes}{شبه الدور} فيبقى $\approx T_0 = 2\pi\sqrt{LC}$: إذ يقلّص \omterm{def:g12:oscillators-and-time:damping}{التخامد} الخفيف
السعةَ، ولا يكاد يمسّ الوتيرة.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:11}
$T_0 = 2\pi\sqrt{0.50 \times \num{2.0e-5}} = 2\pi \times
\num{3.16e-3} \approx \qty{20}{ms}$. وبالقاموس: $k = m/(LC) =
0.50/\num{1.0e-5} = \qty{5.0e4}{N/m}$؛ وبالتحقق
$2\pi\sqrt{m/k} = 2\pi\sqrt{\num{1.0e-5}}$ --- أي \qty{20}{ms} نفسه.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:12}
$C = \qty{500}{pF}$: $f_0 = 1/(2\pi\sqrt{\num{1.0e-13}}) \approx
\qty{0.50}{MHz}$؛ $C = \qty{50}{pF}$: $f_0 \approx \qty{1.6}{MHz}$.
وعند ثبات $L$، يكون $f_0 = (2\pi\sqrt{L})^{-1} C^{-1/2} \propto 1/\sqrt{C}$:
فعامل عشرة في $C$ لا يعطي إلا $\sqrt{10} \approx 3.2$ في $f_0$ --- كما وُجد.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:13}
$0.95^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.95\rvert \approx 13.5$،
أي نحو $14$ \omterm{def:g12:oscillators-and-time:oscillator}{دور}. \omterm{def:g12:oscillators-and-time:oscillator}{والسعة} $\propto \sqrt{E}$: أي نازلة إلى
$1/\sqrt{2} \approx 0.71$ من البداية. وعند \qty{1.0}{MHz}،
$T_0 = \qty{1.0}{\micro s}$: أي نحو \qty{14}{\micro s}.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:14}
الخسارة في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور} $0.020 \times \qty{1.0e-9}{J} = \qty{2.0e-11}{J}$، عند
\num{32768} \omterm{def:g12:oscillators-and-time:oscillator}{دور} في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}: $P = \num{32768} \times \num{2.0e-11}
\approx \qty{6.6e-7}{W}$ --- أي دون الميكروواط، فسنوات على خلية
قرصية. وعلى الخطى، لأن الدفعة عند \omterm{def:g12:oscillators-and-time:phase}{الطور} الصحيح وحدها \emph{تضيف}
\omterm{def:g10:energy-conservation:energy}{طاقة} (أي \omterm{def:g12:oscillators-and-time:resonance}{الرنين})؛ أما خارج الخطى فستكبح التأرجح.
\end{solution}

\begin{solution}{exo:g12:rlc-oscillations:15}
$C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
(\num{2.5e7})^2 \times \num{1.0e-6}} \approx \qty{41}{pF}$. و
ينبغي أن تدفع \omterm{def:g11:circuits-and-power:resistance}{المقاومة} الحلقةَ إلى \omterm{def:g12:rlc-oscillations:regimes}{النظام اللادوري}: فتستقر الشحنة
عندئذٍ دون تجاوز --- فلا تذبذب ولا رنّة.
\end{solution}

\begin{solution}{pb:g12:rlc-oscillations:1}
\textbf{1.} قانون الحلقة: $u_L + u_C = 0$، مع $u_C = q/C$،
$u_L = L\,di/dt$، $i = dq/dt$: $L\,d^2q/dt^2 + q/C = 0$.

\textbf{2.} $d^2q/dt^2 = -(2\pi/T_0)^2 q$، ومن ثم تُقرأ المعادلة
$q\,[\,1/C - L(2\pi/T_0)^2\,] = 0$، وهي صحيحة من أجل كل $t$ بالضبط حين
$T_0 = 2\pi\sqrt{LC}$.

\textbf{3.} $LC$: $(\unit{V.s/A})(\unit{C/V}) = \unit{s} \times
\unit{C/A} = \unit{s^2}$ --- فالمقدار $\sqrt{LC}$ زمن.

\textbf{4.} $LC = \qty{6.6e-14}{s^2}$: $T_0 = 2\pi \times \num{2.57e-7}
\approx \qty{1.6}{\micro s}$، $f_0 \approx \qty{620}{kHz}$ --- أي داخل
\qtyrange{530}{1700}{kHz}.

\textbf{5.} $C = \dfrac{1}{4\pi^2 f_0^2 L} = \dfrac{1}{39.5 \times
\num{1.0e12} \times \num{2.0e-4}} \approx \qty{127}{pF}$.

\textbf{6.} الصيغة نفسها عند \qty{530}{kHz}: $C \approx \qty{451}{pF}$.

\textbf{7.} عند \qty{1700}{kHz}: $C \approx \qty{44}{pF}$.

\textbf{8.} $C = \qty{480}{pF}$: $f_0 \approx \qty{514}{kHz}$؛
$C = \qty{20}{pF}$: $f_0 \approx \qty{2.5}{MHz}$ --- أي الحزمة كلها،
بهامش عند الطرفين.

\textbf{9.} $f_0 = (2\pi\sqrt{L})^{-1}C^{-1/2} \propto 1/\sqrt{C}$؛
ومضاعفة $f_0$ تقتضي قسمة $C$ على $4$.

\textbf{10.} بما أن $f_0 \propto 1/\sqrt{C}$، تزحزح خطوات $C$ المتساوية
المقدارَ $f_0$ ببطء عند $C$ الكبيرة وبسرعة عند $C$ الصغيرة: ومن ثم لا
يستغرق تباعد معطًى بين المحطات إلا مسافة قليلة من دوران القرص --- فتتزاحم عند
الطرف ذي \omterm{def:g12:oscillators-and-time:oscillator}{التواتر} العالي (و$C$ المنخفضة).

\textbf{11.} $Q_0 = CU_0 = \num{1.27e-10} \times 0.020 \approx
\qty{2.5e-12}{C}$؛ $E = \tfrac12 CU_0^2 \approx \qty{2.5e-14}{J}$.

\textbf{12.} $\tfrac12 LI_0^2 = E$: $I_0 = U_0\sqrt{C/L} =
0.020\sqrt{\num{6.35e-7}} \approx \qty{16}{\micro A}$.

\textbf{13.} $0.96^n = \tfrac12$: $n = \ln 2/\lvert\ln 0.96\rvert
\approx 17$ \omterm{def:g12:oscillators-and-time:oscillator}{دور}.

\textbf{14.} عند \qty{1.00}{MHz}، $T_0 = \qty{1.0}{\micro s}$: فتنتصف الرنّة
في نحو \qty{17}{\micro s}. والرنّة الوحيدة تموت في ميكروثوانٍ؛
أما الموسيقى فتحتاج تغذية الحلقة تغذيةً مستمرة.

\textbf{15.} تدفع \omterm{def:g10:signals-and-waves:wave}{الموجة} ملايين المرات في \omterm{def:g10:orders-of-magnitude:unit}{الثانية}؛ ولا تصل الدفعات على الخطى
فتبني التأرجح إلا حين يوافق تواترها $f_0$ --- أي \omterm{def:g12:oscillators-and-time:resonance}{الرنين}؛
أما المحطات خارج الموالفة فتدفع ضده بقدر ما تدفع معه.
وفي المرسِل، يعيد مضخّم الطاقةَ \omterm{def:g11:mechanical-energy:dissipation}{المبدَّدة}
في كل \omterm{def:g12:oscillators-and-time:oscillator}{دور}: أي \omterm{def:g12:oscillators-and-time:oscillator}{مذبذب} مصان.

\textbf{16.} $LC = \qty{2.54e-14}{s^2}$: $k = m/(LC) =
0.20/\num{2.54e-14} \approx \qty{7.9e12}{N/m}$.

\textbf{17.} نحو $10^8$ ضعف زنبرك السيارة: فلا يبلغ أي جهاز
زنبرك وكتلة على الطاولة الميغاهرتز --- إذ إن الصلابات والكتل اليومية
بطيئة بلا أمل.

\textbf{18.} $m = \dfrac{k}{(2\pi f_0)^2} = \dfrac{100}
{(\num{6.28e6})^2} \approx \qty{2.5e-12}{kg}$ --- أي بضعة نانوغرامات، أي
ذرّة غبار.

\textbf{19.} بلورة كوارتز: أي شظية بالغة الخفة، فائقة \omterm{def:g12:oscillators-and-time:spring}{الصلابة}،
تغنّي على \omterm{def:g12:rlc-oscillations:period}{تواترها الذاتي} --- وهي حافظة الزمن التي لُقيت سابقًا هذه
السنة.

\textbf{20.} مع $L = \qty{0.20}{mH}$ و\qtyrange{20}{480}{pF}، تمسح
الحلقة نحو \qtyrange{0.51}{2.5}{MHz}، فتغطي الحزمة؛
و$C \approx \qty{127}{pF}$ تلتقط \qty{1.00}{MHz}؛ وإذا تُركت وشأنها، انتصفت الرنّة
في $\approx \qty{17}{\micro s}$، لكن \omterm{def:g10:signals-and-waves:wave}{موجة} المحطة نفسها،
وهي ترنّ على الخطى، تعيد تغذية الحلقة --- فتذاع الأخبار على \omterm{def:g10:energy-conservation:energy}{طاقة}
\omterm{def:g10:signals-and-waves:wave}{الموجة} نفسها.
\end{solution}
