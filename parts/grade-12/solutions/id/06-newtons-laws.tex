\section*{Bab \ref{ch:g12:newtons-laws} --- Hukum-Hukum Newton}

\begin{solution}{exo:g12:newtons-laws:1}
(a) $p = 70 \times 10 = \qty{7.0e2}{kg.m/s}$.
(b) $v = \qty{13.9}{m/s}$: $p \approx \qty{1.8e4}{kg.m/s}$.
(c) $p = 0.0080 \times 800 = \qty{6.4}{kg.m/s}$. Mobil $>$ pelari $>$
peluru: lima orde besaran massa lebih daripada yang mampu dibeli lajunya.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:2}
$N = m(g + a_z)$: (a) $70 \times 11.01 = \qty{771}{N}$;
(b) $a_z = 0$, sehingga $N = mg = \qty{687}{N}$ --- hukum pertamanya:
kecepatannya tetap, \omterm{def:g10:inertia:force}{gayanya} \omterm{def:g10:inertia:compensate}{saling meniadakan}; (c) $70 \times 8.61 = \qty{603}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:3}
$a = g\sin 30^\circ = \qty{4.9}{m/s^2}$;
$v = \sqrt{2 \times 4.905 \times 3.0} = \qty{5.4}{m/s}$. Massanya
\omterm{def:g10:inertia:compensate}{saling meniadakan} antara $ma$ dan $mg\sin\alpha$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:4}
Berbaliknya: $\Delta p = 0.058 \times (25 + 15) = \qty{2.3}{kg.m/s}$;
$F = 2.32/0.0050 = \qty{4.6e2}{N}$ --- sekitar $800$ kali \omterm{def:g10:universal-gravitation:weight}{berat}
\qty{0.57}{N} bolanya, dan itulah sebabnya gravitasi diabaikan selama sentuhannya.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:5}
\omterm{def:g10:universal-gravitation:weight}{Berat} (Bumi pada bukunya, ke bawah): pasangannya = bukunya menarik Bumi, ke atas.
\omterm{def:g12:newtons-laws:friction}{Gaya normal} (mejanya pada bukunya, ke atas): pasangannya = bukunya menekan mejanya,
ke bawah. \omterm{def:g10:universal-gravitation:weight}{Berat} dan \omterm{def:g12:newtons-laws:friction}{gaya normal} bekerja pada benda yang \emph{sama} dan tergolong dua
interaksi yang berbeda --- sama besar di sini hanya karena $\vect a = \vect 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:6}
$N = mg = \qty{490}{N}$. (a) $F > \mu_s N = \qty{221}{N}$.
(b) $a = (220.7 - 0.35 \times 490.5)/50 = \qty{0.98}{m/s^2}$.
(c) $F = \mu_k N = \qty{172}{N}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:7}
(a) Tergelincir ketika $\tan\alpha > \mu_s$: $\alpha = \arctan 0.45 \approx
24^\circ$. (b) $a = g(\sin 30^\circ - 0.35\cos 30^\circ) =
9.81 \times 0.197 = \qty{1.9}{m/s^2}$. (c) Ya: sekali tergelincir,
\omterm{def:g10:inertia:inventory}{gesekannya} turun ke $\mu_k < \mu_s$, sehingga
$a = g\cos\alpha\,(\tan\alpha - \mu_k) \approx \qty{0.9}{m/s^2} > 0$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:8}
$a = m_2 g/(m_1 + m_2) = 19.62/5.0 = \qty{3.9}{m/s^2}$;
$T = m_1 a = \qty{11.8}{N}$. Balok yang menggantung berpercepatan ke bawah,
sehingga \omterm{def:g11:forces-and-motion:netforce}{gaya total} padanya menunjuk ke bawah: $T$ harus di bawah $m_2 g$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:9}
$v_{\max} = \sqrt{\mu_s g R}$: kering
$\sqrt{0.70 \times 9.81 \times 50} = \qty{18.5}{m/s} \approx
\qty{67}{km/h}$; basah $\qty{14.0}{m/s} = \qty{50}{km/h}$. Baik
$mv^2/R$ maupun $\mu_s mg$ membawa $m$: massanya \omterm{def:g10:inertia:compensate}{saling meniadakan}.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:10}
Secara tegak: $T\cos\theta = mg$, sehingga
$T = 2.943/\cos 25^\circ = \qty{3.2}{N}$. Jari-jarinya $R = L\sin\theta =
\qty{0.507}{m}$; secara radial: $T\sin\theta = mv^2/R$, sehingga
$v = \sqrt{gR\tan\theta} = \sqrt{9.81 \times 0.507 \times 0.466}
= \qty{1.5}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:11}
$a_z = (N - mg)/m$ dengan $mg = \qty{49.05}{N}$: $+\qty{1.0}{m/s^2}$,
$0$, $-\qty{1.0}{m/s^2}$. Misalnya perjalanan naik: bertambah cepat, melaju tetap,
mengerem. Ya: jika masuk dalam keadaan sudah menurun, bacaan yang sama berarti
mengerem, melaju tetap, lalu bertambah cepat lagi ke bawah --- timbangannya membaca
percepatan, bukan kecepatan.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:12}
(a) $0 = 60V + 4.0 \times 8.0$: $V = \qty{-0.53}{m/s}$.
(b) Bolanya pada $8.0 + V$ terhadap tanah: $0 = 60V + 4.0(8.0 + V)$, sehingga
$V = -32/64 = \qty{-0.50}{m/s}$ dan bolanya melesat pada \qty{7.5}{m/s}.
(c) Dorongan bolanya pada tangannya --- pasangan hukum ketiga bagi lemparannya
sendiri.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:13}
Tanpa \omterm{def:g10:inertia:inventory}{gesekan}: $N\cos\theta = mg$ (tegak) dan $N\sin\theta =
mv^2/R$ (radial); dengan membaginya, $\tan\theta = v^2/(gR) = 400/784.8$:
$\theta \approx 27^\circ$. Lebih lambat: mobilnya cenderung merosot menuruni
kemiringannya, \omterm{def:g10:inertia:inventory}{gesekannya} harus menunjuk menaiki lerengnya; lebih cepat: sebaliknya.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:14}
(a) $250 \times 2500 = \qty{6.3e5}{kg.m/s}$ tiap sekon: \omterm{def:g10:inertia:inventory}{gaya dorongnya}
$\qty{6.3e5}{N}$. (b) $P = \qty{4.9e5}{N}$;
$a = (6.25 - 4.905) \times 10^5 / \num{5.0e4} = \qty{2.7}{m/s^2}$.
(c) Roketnya mendorong gas buangannya sendiri (hukum ketiga), bukan udaranya.
\end{solution}

\begin{solution}{exo:g12:newtons-laws:15}
(a) $m_2 g > \mu_s m_1 g$: $m_2 > 0.35 \times 4.0 = \qty{1.4}{kg}$.
(b) $a = (m_2 - \mu_k m_1)\,g/(m_1 + m_2) = 1.0 \times 9.81/6.0 =
\qty{1.6}{m/s^2}$; $T = m_2(g - a) = \qty{16.4}{N}$.
(c) $\mu_k = 0$ memberikan $a = m_2 g/(m_1 + m_2)$: rumus pada
\cref{exo:g12:newtons-laws:8}.
\end{solution}

\begin{solution}{pb:g12:newtons-laws:1}
\textbf{1.} Sistemnya: kabin $+$ kereta ($+$ penumpangnya). Gunungnya
terpaku pada tanah, inersial dengan ketelitian yang sangat baik. \omterm{def:g10:inertia:force}{Gayanya}:
\omterm{def:g10:universal-gravitation:weight}{berat} $M\vect g$ (ke bawah), \omterm{def:g12:newtons-laws:friction}{gaya normal} $\vect N$ kabel jalurnya
(tegak lurus kabelnya), tegangan penghela $\vect T$ (menaiki lerengnya).

\textbf{2.} Saat diam, hukum pertamanya: $\sum \vect F = \vect 0$. Sepanjang
lerengnya: $T = Mg\sin\alpha$; tegak lurus: $N = Mg\cos\alpha$.

\textbf{3.} $T = 4000 \times 9.81 \times 0.500 = \qty{19.6}{kN}$;
$N = 4000 \times 9.81 \times 0.866 = \qty{34.0}{kN}$.

\textbf{4.} Kosong: $T = \qty{9.8}{kN}$. Setiap \omterm{def:g10:inertia:force}{gaya} dalam
keseimbangannya sebanding dengan $M$, sehingga $T$ pun demikian.

\textbf{5.} Sepanjang lerengnya, $Ma = Mg\sin\alpha$: $M$ \omterm{def:g10:inertia:compensate}{saling meniadakan},
$a = g\sin\alpha = \qty{4.9}{m/s^2}$ menuruni lerengnya.

\textbf{6.} $T' = M(g\sin\alpha + a) = 4000 \times 5.505 =
\qty{22.0}{kN}$.

\textbf{7.} $a/(g\sin\alpha) = 0.60/4.905 \approx 12\%$.

\textbf{8.} $t = v/a = \qty{10}{s}$; $d = v^2/(2a) = \qty{30}{m}$.

\textbf{9.} Komponen tegak $\vect a$: $a\sin\alpha =
\qty{0.30}{m/s^2}$, sehingga $N = m(g + a\sin\alpha) = 80 \times 10.11 =
\qty{809}{N}$ terhadap $mg = \qty{785}{N}$: $3\%$ lebih berat.

\textbf{10.} \omterm{def:g10:inertia:inventory}{Gesekan} statis lantainya, mendatar:
$ma\cos\alpha = 80 \times 0.52 \approx \qty{42}{N}$. Dengan condong, dorongan
gabungan lantainya melewati \omterm{def:g11:forces-and-motion:com}{pusat massanya}, sehingga tidak menjungkalkannya.

\textbf{11.} $a_T = dv/dt = 0$ (lajunya tetap);
$a_N = v^2/R = 36/40 = \qty{0.90}{m/s^2}$, membidik pusatnya ---
lurus ke bawah di puncaknya.

\textbf{12.} Sumbu normalnya (ke bawah positif): $Mg - N' = Mv^2/R$, sehingga
$N' = M(g - v^2/R) = 4000 \times 8.91 = \qty{35.6}{kN}$.

\textbf{13.} $N = m(g - v^2/R) = 80 \times 8.91 = \qty{713}{N}$:
lebih ringan sebesar $v^2/(gR) = 0.90/9.81 \approx 9\%$.

\textbf{14.} $N' = 0$ pada $v = \sqrt{gR} = \qty{19.8}{m/s} \approx
\qty{71}{km/h}$ --- lebih dari tiga kali laju lajunya: keretanya menumpang
kabelnya dengan sisa yang lapang.

\textbf{15.} \omterm{thm:g12:newtons-laws:third}{Hukum ketiga Newton}: \qty{35.6}{kN}, menekan ke bawah pada
kabelnya (jadi pada menaranya) --- sedikit kurang daripada \omterm{def:g10:universal-gravitation:weight}{beratnya} saat diam,
yang diringankan oleh \omterm{def:g10:relative-motion:trajectory}{lintasan} yang melengkung itu.

\textbf{16.} $F = \Delta p/\Delta t = 4000 \times 6.0/1.5 =
\qty{16}{kN}$, berlawanan dengan geraknya ($a = \qty{4.0}{m/s^2}$).

\textbf{17.} $f = ma = 80 \times 4.0 = \qty{320}{N}$; langit-langitnya
$\mu_s mg = 0.40 \times 785 = \qty{314}{N} < \qty{320}{N}$: kakinya
saja akan tergelincir --- peganglah pegangannya.

\textbf{18.} $0 = M_0 V + m(v_{\text{rel}} + V)$:
$V = -80 \times 2.5/2080 = \qty{-0.096}{m/s}$; penumpangnya bergerak pada
$2.5 - 0.096 \approx \qty{2.4}{m/s}$ terhadap tanah.

\textbf{19.} Roketnya: \omterm{def:g10:inertia:inventory}{gaya dorongnya} ditetapkan oleh massa yang dilontarkan tiap
sekon dikalikan laju lontarannya terhadap roket itu.

\textbf{20.} Tegangan terbesar yang ditemui adalah saat bertolak,
$T' = M \times \qty{5.505}{N/kg}$ (saat diam memberikan $Mg\sin\alpha$, di
puncaknya bahkan lebih kecil). Batasnya: $T' \leq \qty{20}{kN}$, sehingga
$M \leq 20000/5.505 = \qty{3.63e3}{kg}$: muatannya
\qty{1.63e3}{kg}, yaitu $20$ penumpang \qty{80}{kg} (bukan $25$).
Papannya: \emph{20 orang}; lalu $M = \qty{3.6e3}{kg}$,
$T' = \qty{19.8}{kN}$, faktor keselamatannya $100/19.8 \approx 5.1 \geq 5$.
\end{solution}
