\begin{solution}{exo:g12:sound-acoustics:13}

\begin{solution}{pb:g12:sound-acoustics:1}
\textbf{1.} $v = 2Lf_1 = 2 \times 0.650 \times 110 = \qty{143}{m/s}$.

\textbf{2.} Pada senarnya $\lambda_1 = 2L = \qty{1.30}{m}$; di udaranya
$\lambda = 340/110 \approx \qty{3.1}{m}$. Nilai $f$-nya sama, tetapi
$\lambda = v/f$ dan laju \omterm{def:g10:signals-and-waves:wave}{gelombangnya} berbeda.

\textbf{3.} $L' = v/(2 \times 220) = L/2 = \qty{32.5}{cm}$: fret kedua
belasnya berada di titik tengahnya, $L/2$ dari jembatannya.

\textbf{4.} $L' = 143/(2 \times 165) \approx \qty{43.3}{cm} = 2L/3$.

\textbf{5.} $\log(220/110)/\log 2^{1/12} = 12$ seminada --- dua belas
fret untuk satu oktaf. Fret pertamanya:
$L_1 = L/2^{1/12} = 65.0/1.059
\approx \qty{61.4}{cm}$.

\textbf{6.} \qtylist{110;220;330;440;550}{Hz}.

\textbf{7.} $\qty{440}{Hz} = 4f_1$: \omterm{def:g12:sound-acoustics:harmonics}{harmonik} keempat, yaitu dua oktaf di
atasnya. $\qty{330}{Hz} = 3f_1$: \omterm{def:g12:sound-acoustics:harmonics}{harmonik} ketiga, yaitu satu oktaf
ditambah satu kuint.

\textbf{8.} Kedudukan petikannya menetapkan campuran ragamnya: di dekat
jembatannya \omterm{def:g12:sound-acoustics:harmonics}{harmonik} yang tinggi menjadi kuat (\omterm{def:g12:sound-acoustics:timbre}{spektrumnya} cerah), di
atas papan jarinya \omterm{def:g12:sound-acoustics:harmonics}{nada dasarnya} menguasai (\omterm{def:g10:signals-and-waves:sound}{bunyinya} lembut). Nilai
$f_1$-nya sama, sehingga \omterm{def:g12:sound-acoustics:pitch-loudness}{tinggi nadanya} sama --- \omterm{def:g12:sound-acoustics:timbre}{spektrumnya} berbeda,
\omterm{def:g12:sound-acoustics:timbre}{warna bunyinya} pun berbeda.

\textbf{9.} Hanya \omterm{def:g12:sound-acoustics:harmonics}{harmonik} yang memiliki \omterm{def:g12:sound-acoustics:standing-wave}{simpul} di titik tengahnya yang
bertahan: yang genap, yaitu \qtylist{220;440;660}{Hz}, \dots \omterm{def:g12:sound-acoustics:pitch-loudness}{Tinggi
nadanya} melompat satu oktaf, menjadi \qty{220}{Hz}.

\textbf{10.} $nf_1 \le \qty{20}{kHz}$: $n \le 20000/110 = 181.8$,
sehingga ada $181$ \omterm{def:g12:sound-acoustics:harmonics}{harmonik}.

\textbf{11.} $I = P/(4\pi r^2) = \num{6.0e-3}/(4\pi \times 9.0)
\approx \qty{5.3e-5}{W/m^2}$.

\textbf{12.} $L = 10\log_{10}(\num{5.3e7}) \approx \qty{77}{dB}$.

\textbf{13.} Karena $I \times 4$: $+\qty{6}{dB} \to \qty{83}{dB}$.
Tarafnya berupa logaritma: intensitasnya yang berjumlah, tarafnya
tidak.

\textbf{14.} $+10\log_{10}12 \approx +\qty{10.8}{dB} \to
\qty{88}{dB}$.

\textbf{15.} Sasarannya $I = I_0 \times 10^{8.5} \approx
\qty{3.2e-4}{W/m^2}$ dengan $12P = \qty{7.2e-2}{W}$:
$r = \sqrt{12P/(4\pi I)} \approx \sqrt{18.1} \approx \qty{4.3}{m}$.

\textbf{16.} Taraf \qty{88}{dB} adalah $+\qty{3}{dB}$ di atas batas
delapan jamnya: waktu amannya menjadi separuh satu kali --- yaitu empat
jam. Pergelarannya masih muat.

\textbf{17.} $\lambda = 5000/\num{2.0e6} = \qty{2.5}{mm}$. Sebuah
\omterm{def:g10:signals-and-waves:wave}{gelombang} hanya memantul dari cacat yang lebarnya sekurang-kurangnya
sekitar satu \omterm{def:g12:mechanical-waves:wavelength}{panjang gelombang}: \omterm{def:g10:signals-and-waves:sound}{bunyi} yang terdengar di dalam kayu
memiliki $\lambda$ berukuran puluhan sentimeter dan berbelok mengitari
cacat yang kecil mana pun.

\textbf{18.} $t = 2d/v = 2 \times 0.045/5000 = \qty{18}{\micro s}$.

\textbf{19.} $d = vt/2 = 5000 \times \num{7.2e-6}/2 = \qty{18}{mm}$.

\textbf{20.} Kartunya: $v = \qty{143}{m/s}$ pada senarnya; fretnya pada
$\qty{32.5}{cm}$ (kedua belas) dan $\qty{61.4}{cm}$ (pertama);
kelompoknya aman di luar $\approx \qty{4.3}{m}$; cacatnya sedalam
\qty{18}{mm} --- empat bilangan, satu bab.
\end{solution}
