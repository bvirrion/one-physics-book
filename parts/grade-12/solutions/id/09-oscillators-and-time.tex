\section*{Bab \ref{ch:g12:oscillators-and-time} --- Osilator Mekanik dan Pengukuran Waktu}

\begin{solution}{exo:g12:oscillators-and-time:1}
(a) $T_0 = 2\pi\sqrt{1.00/9.81} = \qty{2.01}{s}$;
$f_0 = 1/T_0 = \qty{0.50}{Hz}$. (b) $T_0 \propto \sqrt{L}$:
\omterm{def:g12:oscillators-and-time:oscillator}{periodenya} menjadi dua kali lipat.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:2}
Jantungnya: $f = 72/60 = \qty{1.2}{Hz}$, $T = \qty{0.83}{s}$. Sayapnya:
$T = 1/30 = \qty{33}{ms}$. Kenaikan dadanya adalah \emph{\omterm{def:g12:oscillators-and-time:oscillator}{amplitudo}}.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:3}
$T_0 = 2\pi\sqrt{0.10/25} = \qty{0.40}{s}$, $f_0 = \qty{2.5}{Hz}$.
Dengan menggandakan $m$: $T_0 \times \sqrt{2} \approx 1.4$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:4}
\omterm{def:g12:oscillators-and-time:damping}{Pseudoperiodenya} \qty{2.0}{s}. Nisbahnya $1.5/2.0 = 1.13/1.5 = 0.85/1.13
= 0.75$ tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode}; puncak berikutnya $0.85 \times 0.75 \approx
\qty{0.64}{cm}$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:5}
Bandulnya: $T_0 \propto 1/\sqrt{g}$, sehingga $\times\sqrt{9.81/1.62} =
2.46$ (lebih lambat). Pegas--massanya: tidak berubah --- tidak ada $g$ pada
$2\pi\sqrt{m/k}$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:6}
(a) $[L/g] = \unit{m}/(\unit{m/s^2}) = \unit{s^2}$; akar kuadratnya
adalah waktu. (b) \omterm{def:g10:orders-of-magnitude:unit}{Kilogramnya} hanya muncul pada $m$: tidak ada lain yang dapat
meniadakannya, sehingga $m$ tidak mungkin masuk. (c) Tidak pernah: bilangan
murni seperti $2\pi$ tak terlihat oleh dimensi.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:7}
$\frac{dx}{dt} = -\frac{2\pi}{T_0}A\sin(2\pi t/T_0 + \varphi)$, sehingga
$\frac{d^2x}{dt^2} = -(2\pi/T_0)^2 x(t)$. Kesamaannya dengan $-(k/m)x$
untuk semua $t$ berlaku tepat ketika $(2\pi/T_0)^2 = k/m$, yaitu
$T_0 = 2\pi\sqrt{m/k}$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:8}
$E_m = \tfrac12 kA^2 = \tfrac12 \times 40 \times 0.030^2 =
\qty{1.8e-2}{J}$; $v_{\max} = \sqrt{2E_m/m} = \sqrt{0.18} =
\qty{0.42}{m/s}$ --- pada $x = 0$; nol di ujung-ujungnya $x = \pm A$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:9}
(a) $L = gT_0^2/4\pi^2 = \qty{0.994}{m}$. (b) $\Delta T_0/T_0 =
\tfrac12 \times 1.0\% = 0.5\%$: $86400 \times 0.005 \approx
\qty{4.3e2}{s}$ --- tujuh menit lambat tiap hari.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:10}
(a) $A \propto \sqrt{E}$: $\sqrt{0.95} \approx 0.975$, turun
$2.5\%$. (b) $0.95^n = 0.5$: $n = \ln 2/\ln(1/0.95) \approx 14$
\omterm{def:g12:oscillators-and-time:oscillator}{periode}. (c) Tidak: untuk \omterm{def:g12:oscillators-and-time:damping}{peredaman} ringan nilainya tetap dalam pecahan
persen dari $T_0$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:11}
(a) Pada $f_0$ tiap dorongannya tiba selaras lalu menambahkan \omterm{def:g10:energy-conservation:energy}{energi}; di luar
puncaknya, dorongannya berganti-ganti membantu dan menghambat. (b) Berbaris
seragam merupakan pemaksaan berkala; di dekat $f_0$ jembatannya hal itu akan
menaiki puncak \omterm{def:g12:oscillators-and-time:resonance}{resonansinya}. (c) Tabungnya menyapu melewati \omterm{def:g12:oscillators-and-time:resonance}{frekuensi alami}
rangkanya saat memulai putaran cepatnya: \omterm{def:g12:oscillators-and-time:resonance}{resonansi} sesaat, lalu lewat puncaknya.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:12}
(a) $T_0 \propto 1/\sqrt{g}$: lambat, sebesar $\Delta T_0/T_0 = \tfrac12
\times 0.03/9.81 = \num{1.5e-3}$, yaitu $\approx \qty{132}{s}$ tiap
hari. (b) $L \propto g$ pada $T_0$ yang tetap: pendekkanlah sebesar $0.994 \times
0.03/9.81 \approx \qty{3.0}{mm}$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:13}
$v_{\max} = 2\pi A/T_0 = 2\pi \times 0.040/0.63 = \qty{0.40}{m/s}$;
$a_{\max} = (2\pi/T_0)^2 A = \qty{4.0}{m/s^2}$; $k = m(2\pi/T_0)^2
= \qty{20}{N/m}$; $E_m = \tfrac12 kA^2 = \qty{1.6e-2}{J}$.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:14}
$T_0 = 40.1/20 = \qty{2.005}{s}$; $g = 4\pi^2 L/T_0^2 = 4\pi^2
\times 0.995/4.020 = \qty{9.77}{m/s^2}$. Ketakpastiannya $\approx 0.2\%
+ 2 \times 0.5\% = 1.2\%$, yaitu $\pm\qty{0.12}{m/s^2}$: selangnya
memuat \qty{9.81}{m/s^2} --- sesuai.
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:15}
Bandulnya $10/86400 \approx \num{1.2e-4}$; kuarsanya $0.5/\num{2.6e6}
\approx \num{1.9e-7}$; sesiumnya $1/\num{3.2e15} \approx \num{3e-16}$.
Galat \qty{1}{\micro s} berarti $\num{1000} \times \qty{30}{cm} =
\qty{300}{m}$ kedudukan: GPS membutuhkan keluarga \omterm{def:g11:fundamental-interactions:ladder}{atomnya}.
\end{solution}

\begin{solution}{pb:g12:oscillators-and-time:1}
\textbf{1.} Galat tanggapan $\approx \qty{0.2}{s}$ itu dibagi
$50$: sekitar \qty{4}{ms} pada $T_0$.

\textbf{2.} $T = 99.4/50 = \qty{1.988}{s}$.

\textbf{3.} $L = gT^2/4\pi^2 = 9.81 \times 1.988^2/4\pi^2 \approx
\qty{0.982}{m}$.

\textbf{4.} $L = \qty{0.994}{m}$ (bandul sesekonnya): turunkanlah
bebannya, memanjangkannya sebesar $\approx \qty{1.2}{cm}$.

\textbf{5.} Terlalu pendek, jadi terlalu gesit: cepat, sebesar $(2.000 -
1.988)/1.988 \times 86400 \approx \qty{5.2e2}{s} \approx 8.7$
menit tiap hari.

\textbf{6.} Bulan: $\times\sqrt{9.81/1.62} = 2.46$ --- tak terpakai.
Observatoriumnya: $\Delta T_0/T_0 = \tfrac12 \times 0.01/9.81 =
\num{5.1e-4}$: kehilangan $\approx \qty{44}{s}$ tiap hari.

\textbf{7.} $\Delta L = L\lambda\,\Delta\theta = 0.994 \times
\num{1.9e-5} \times 5.0 \approx \qty{9.4e-5}{m} \approx
\qty{0.1}{mm}$.

\textbf{8.} $T' = 2\pi\sqrt{L(1 + \lambda\Delta\theta)/g} =
T_0\sqrt{1 + \lambda\Delta\theta} \approx T_0\,(1 + \tfrac12
\lambda\Delta\theta)$.

\textbf{9.} $\Delta T_0/T_0 = \tfrac12 \times \num{1.9e-5} \times
5.0 = \num{4.8e-5}$: $86400 \times \num{4.8e-5} \approx
\qty{4.1}{s}$ tiap hari, lambat (batangnya memanjang, \omterm{def:g12:oscillators-and-time:oscillator}{periodenya} memanjang).

\textbf{10.} $\Delta\theta = \qty{-10}{K}$: $\Delta T_0/T_0 =
\num{-9.5e-5}$, memperoleh $\approx \qty{8.2}{s}$ tiap hari.

\textbf{11.} Baja dan kuningan memuai berbeda; digantung dalam arah yang
berselang-seling, batang kuningannya menurunkan bebannya sementara batang
bajanya menaikkannya. Diukur agar kedua pergeserannya \omterm{def:g10:inertia:compensate}{saling meniadakan},
panjang efektifnya --- dan lajunya --- bertahan melewati musimnya.

\textbf{12.} $280/2.000 = 140$ \omterm{def:g12:oscillators-and-time:oscillator}{periode}.

\textbf{13.} $\theta_m = 2.00^\circ = \qty{0.0349}{rad}$: $E_m =
\tfrac12 \times 1.2 \times 9.81 \times 0.994 \times 0.0349^2 \approx
\qty{7.1e-3}{J}$.

\textbf{14.} Tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode} \omterm{def:g12:oscillators-and-time:oscillator}{amplitudonya} menyusut sebesar $2^{-1/140} \approx
0.9951$, \omterm{def:g10:energy-conservation:energy}{energinya} sebesar $0.9951^2 \approx 0.990$: sekitar $1.0\%$
$E_m$ tiap \omterm{def:g12:oscillators-and-time:oscillator}{periode}.

\textbf{15.} Dorongannya $\approx 0.010 \times \num{7.1e-3} =
\qty{7.1e-5}{J}$; \omterm{def:g11:work-of-force:power}{dayanya} $\num{7.1e-5}/2.0 \approx \qty{3.5e-5}{W}$
--- puluhan mikrowatt.

\textbf{16.} Sepekan adalah $604800/2.0 = \num{3.02e5}$ \omterm{def:g12:oscillators-and-time:oscillator}{periode}, yang menuntut
$\num{3.02e5} \times \num{7.1e-5} \approx \qty{21}{J}$; bebannya
memasok $Mgh = 2.5 \times 9.81 \times 1.0 = \qty{24.5}{J}$: cukup,
pada dayaguna $21/24.5 \approx 88\%$.

\textbf{17.} $T = 1/\num{32768} = \qty{30.5}{\micro s}$. $\num{32768}
= 2^{15}$: lima belas tingkat pembagi dua mengubah \omterm{def:g10:signals-and-waves:signal}{sinyal} hablurnya
menjadi detak \qty{1}{Hz} yang tepat.

\textbf{18.} Kuarsanya: $0.3/\num{2.6e6} \approx \num{1.2e-7}$.
Bandulnya: $0.5/86400 \approx \num{5.8e-6}$, yaitu $\approx
\qty{15}{s}$ tiap bulan. Kuarsanya menang dengan faktor $\approx 50$.

\textbf{19.} $1/\num{3.2e15} \approx \num{3e-16}$: sembilan orde
besaran di bawah kuarsanya. GPS mengukur waktu isyaratnya sampai nanosekon ---
masing-masing \qty{30}{cm} perjalanan cahaya --- sehingga hanya jam \omterm{def:g11:fundamental-interactions:ladder}{atom} yang memadai.

\textbf{20.} Setelah dipulihkan, jam dindingnya bertahan pada $\pm\qty{15}{s}$ tiap
bulan terhadap $\pm\qty{0.3}{s}$ milik kuarsanya: simpanlah ia demi
loncengnya, percayalah kuarsanya demi keretanya --- dan sekon itu sendiri kini
milik \omterm{def:g11:fundamental-interactions:ladder}{atom} sesium.
\end{solution}
