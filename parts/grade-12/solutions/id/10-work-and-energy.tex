\section*{Bab \ref{ch:g12:work-and-energy} --- Usaha dan Energi Mekanik}

\begin{solution}{exo:g12:work-and-energy:1}
$E_p = \tfrac12 \times 200 \times 0.050^2 = \qty{0.25}{J}$. Dengan regangan
yang digandakan: $\times 4$, yaitu \qty{1.0}{J}. \qty{5.0}{cm} yang kedua itu
sendiri berongkos $1.0 - 0.25 = \qty{0.75}{J}$ --- \omterm{def:g10:inertia:force}{gayanya} di sana lebih besar.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:2}
$W = mg(z_A - z_B) = 0.90 \times 9.81 \times (12 - 30) \approx
\qty{-1.6e2}{J}$, apa pun rutenya --- sama saja pada tanjakan yang
lurus. Perjalanan pulang perginya: \qty{0}{J} (\omterm{def:g12:work-and-energy:conservative}{gaya konservatif}).
\end{solution}

\begin{solution}{exo:g12:work-and-energy:3}
(a), (b) \omterm{def:g12:work-and-energy:conservative}{konservatif}: \omterm{def:g11:work-of-force:work}{usahanya} tertetapkan oleh kedua ujungnya (penurunan
ketinggian; regangan). (c), (e) \omterm{def:g12:work-and-energy:conservative}{tak konservatif}: $-fL$ dan \omterm{def:g11:circuits-and-power:resistance}{hambatannya} memungut
tiap \omterm{def:g10:orders-of-magnitude:unit}{meter} yang ditempuh. (d) tanpa \omterm{def:g11:work-of-force:work}{usaha}: selalu tegak lurus geraknya.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:4}
$h = L(1 - \cos\qty{60}{\degree}) = \qty{1.0}{m}$;
$v = \sqrt{2 \times 9.81 \times 1.0} \approx \qty{4.4}{m/s}$.
Tegangannya tegak lurus geraknya pada setiap saat: \omterm{def:g11:work-of-force:power}{dayanya} nol,
\omterm{def:g11:work-of-force:work}{usahanya} nol.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:5}
$F = P/v = 250/9.0 \approx \qty{28}{N}$. Pada \qty{5.0}{m/s}:
$F = 250/5.0 = \qty{50}{N}$ (sebagian besar komponen gravitasi lerengnya).
\end{solution}

\begin{solution}{exo:g12:work-and-energy:6}
$E_p = \tfrac12 \times 450 \times 0.080^2 = \qty{1.44}{J}$;
$v = \sqrt{2 \times 1.44/0.020} = \qty{12}{m/s}$;
$h = E_p/mg = 1.44/(0.020 \times 9.81) \approx \qty{7.3}{m}$.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:7}
Trapesiumnya (langkah \qty{2.0}{cm}): $(0.03 + 0.10 + 0.19 + 0.30) =
\qty{0.62}{J}$. Model garis lurusnya: $k = 18.0/0.080 =
\qty{225}{N/m}$, $\tfrac12 kx^2 = \qty{0.72}{J}$. Kurva terukurnya
melendut di bawah garis lurusnya pada $x$ yang kecil, sehingga luas sejatinya
lebih kecil.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:8}
$\Delta E_m = \tfrac12 \times 55 \times 9.0^2 - 55 \times 9.81 \times
8.0 = 2228 - 4316 \approx \qty{-2.1e3}{J}$;
$f = 2089/25 \approx \qty{84}{N}$. \omterm{def:g10:universal-gravitation:weight}{Beratnya} \omterm{def:g12:work-and-energy:conservative}{konservatif}, \omterm{def:g12:newtons-laws:friction}{gaya normalnya}
tanpa \omterm{def:g11:work-of-force:work}{usaha}: hanya penurunan dan \emph{panjang} \omterm{def:g10:relative-motion:trajectory}{lintasannya} yang masuk, bukan
bentuknya.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:9}
$H_{\min} = \tfrac52 R = \qty{0.50}{m}$. Dari \qty{60}{cm}:
$v_{\text{atas}} = \sqrt{2 \times 9.81 \times (0.60 - 0.40)} \approx
\qty{2.0}{m/s}$, dan $v_{\text{atas}}^2/(gR) = 2.0$ --- dua kali
nilai terkecilnya. \omterm{def:g10:inertia:inventory}{Gesekannya} menyerok \omterm{def:g10:energy-conservation:energy}{energi} sepanjang jalan, sehingga \omterm{def:g10:relative-motion:trajectory}{lintasan}
yang sesungguhnya membutuhkan ketinggian tambahan.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:10}
\omterm{def:g12:work-and-energy:diagram}{Titik baliknya}: $8.0\,x^2 = 2.0$, $x = \pm\qty{0.50}{m}$. Laju
terbesarnya pada $x = 0$: $v = \sqrt{2 \times 2.0/0.100} \approx
\qty{6.3}{m/s}$. Sebuah osilasi maju mundur di dalam sumur parabolik ---
$E_p$ \omterm{prop:g12:oscillators-and-time:spring-period}{osilator pegas--massanya} persis berbentuk begini.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:11}
$\tfrac12 kx^2 = \tfrac12 mv^2$:
$x = v\sqrt{m/k} = 2.0\sqrt{1200/\num{6.0e5}} \approx \qty{8.9}{cm}$.
Pada \qty{4.0}{m/s}: \qty{18}{cm} --- dua kali, bukan empat kali: \omterm{def:g10:energy-conservation:energy}{energinya}
melipatempat, tetapi $E_p$ tumbuh seperti $x^2$, sehingga $x$ hanya berlipat dua.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:12}
(a) $E_m = \qty{5.0}{J} < \qty{6.0}{J}$: keretanya tidak dapat lewat; ia berhenti di
lerengnya tempat $E_p = \qty{5.0}{J}$ lalu menggelinding kembali. (b) Puncak bukitnya:
$E_k = \qty{1.0}{J}$, $v = \sqrt{2 \times 1.0/0.50} = \qty{2.0}{m/s}$;
lembahnya: $E_k = \qty{5.0}{J}$, $v \approx \qty{4.5}{m/s}$. (c) $x = 0$:
\omterm{prop:g12:work-and-energy:equilibria}{goyah} (maksimum); $x = \qty{1.5}{m}$: mantap (minimum).
\end{solution}

\begin{solution}{exo:g12:work-and-energy:13}
$\tfrac12 mv^2 = mgR_E$:
$v = \sqrt{2 \times 9.81 \times \num{6.37e6}} \approx \qty{1.12e4}{m/s}
= \qty{11.2}{km/s}$. Tiap sukunya sebanding dengan $m$: wahana dan
kerikil sama saja. Tepat di bawahnya, peluncurannya masih menanjak amat jauh,
menemui \omterm{def:g12:work-and-energy:diagram}{titik balik}, lalu jatuh kembali: terikat.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:14}
$E_m = \tfrac12 kA^2 = \tfrac12 \times 40 \times 0.060^2 =
\qty{7.2e-2}{J}$; $v_{\max} = \sqrt{2E_m/m} =
\sqrt{2 \times 0.072/0.250} \approx \qty{0.76}{m/s}$;
$E_k = E_p$ di tempat $\tfrac12 kx^2 = E_m/2$: $x = \pm A/\sqrt 2 \approx
\pm\qty{4.2}{cm}$.
\end{solution}

\begin{solution}{exo:g12:work-and-energy:15}
$h = v^2/2g = 9.5^2/19.62 \approx \qty{4.6}{m}$; mistarnya $\approx
4.6 + 1.0 = \qty{5.6}{m}$. Rekornya lebih tinggi: pelompatnya menambahkan \omterm{def:g11:work-of-force:work}{usaha}
dengan lengan dan kakinya pada galah yang melentur. Namun taksiran \omterm{def:g10:energy-conservation:energy}{energinya}
menetapkan skalanya --- tidak ada lompatan pelari cepat yang akan mencapai \qty{10}{m}.
\end{solution}

\begin{solution}{pb:g12:work-and-energy:1}
\textbf{1.} $v = \sqrt{2gH} = \sqrt{2 \times 9.81 \times 45} \approx
\qty{29.7}{m/s}$: \omterm{def:g10:universal-gravitation:weight}{beratnya} \omterm{def:g12:work-and-energy:conservative}{konservatif}, \omterm{def:g12:newtons-laws:friction}{gaya normalnya} tanpa \omterm{def:g11:work-of-force:work}{usaha} ---
tampangnya gugur dari perhitungan.

\textbf{2.} Gerbangnya: $E_m = mgH = 70 \times 9.81 \times 45 \approx
\qty{3.09e4}{J}$. Lepas landasnya: $E_m = \tfrac12 \times 70 \times 26^2
\approx \qty{2.37e4}{J}$.

\textbf{3.} $\Delta E_m \approx \qty{-7.2e3}{J}$; teorema \omterm{def:g11:mechanical-energy:em}{energi
mekaniknya} menagihkannya kepada \omterm{def:g12:work-and-energy:conservative}{gaya tak konservatifnya}:
\omterm{def:g10:inertia:inventory}{gesekan} saljunya dan hambatan udaranya.

\textbf{4.} $f = \num{7.2e3}/90 \approx \qty{80}{N}$.

\textbf{5.} $\num{7.2e3}/\num{3.09e4} \approx 23\%$. Lilinnya menurunkan $f$,
membungkuk menurunkan hambatannya: keduanya menyusutkan $fL$, satu-satunya suku yang dapat ditawar.

\textbf{6.} \omterm{def:g12:projectile-motion:freefall}{Jatuh bebas} dengan kecepatan awal yang mendatar --- sebuah
gerak \omterm{def:g12:projectile-motion:projectile}{proyektil}. $v_x = \qty{26}{m/s}$;
$v_y = \sqrt{2 \times 9.81 \times 40} \approx \qty{28.0}{m/s}$.

\textbf{7.} $v = \sqrt{26^2 + 28.0^2} = \sqrt{1461} \approx
\qty{38.2}{m/s}$; dengan \omterm{def:g10:energy-conservation:energy}{energi}: $v = \sqrt{v_0^2 + 2gh} = \sqrt{676 + 785}$
--- serupa. Sekitar \qty{138}{km/h}.

\textbf{8.} $t = v_y/g = 28.0/9.81 \approx \qty{2.9}{s}$;
$d = v_0 t \approx \qty{74}{m}$.

\textbf{9.} Tidak. \omterm{def:g10:inertia:force}{Gaya} angkatnya tegak lurus kecepatannya (tanpa \omterm{def:g11:work-of-force:work}{usaha}) dan
hambatannya melawannya (\omterm{def:g11:work-of-force:work}{usaha} negatif): $\Delta E_m \leq 0$, sehingga laju
pendaratan yang sesungguhnya hanya dapat menjadi \emph{lebih kecil} --- udaranya
memanjangkan penerbangannya, bukan menyuapinya.

\textbf{10.} $\alpha = \arctan(28.0/26) \approx \qty{47}{\degree}$
di bawah garis mendatarnya.

\textbf{11.} Sudutnya terhadap lerengnya: $47 - 35 = \qty{12}{\degree}$.
$v_\parallel = 38.2\cos\qty{12}{\degree} \approx \qty{37.4}{m/s}$;
$v_\perp = 38.2\sin\qty{12}{\degree} \approx \qty{7.9}{m/s}$.

\textbf{12.} Yang terserap: $\tfrac12 \times 70 \times 7.9^2 \approx
\qty{2.2e3}{J}$, terhadap $\tfrac12 \times 70 \times 38.2^2 \approx
\qty{5.1e4}{J}$ seluruhnya: sekitar $4\%$.

\textbf{13.} $h_{\text{ek}} = 7.9^2/19.62 \approx \qty{3.2}{m}$ ---
lompatan berani dari sebuah tembok. Datar: $38.2^2/19.62 \approx \qty{74}{m}$ ---
tidak terselamatkan.

\textbf{14.} Lereng yang sejajar dengan \omterm{def:g10:relative-motion:trajectory}{lintasan} terbangnya menjaga $v_\perp$
tetap kecil, sehingga kakinya menyerap beberapa \omterm{def:g10:orders-of-magnitude:unit}{meter}, bukan seluruh gunung \omterm{def:g11:mechanical-energy:kinetic}{energi
kinetiknya}.

\textbf{15.} $H'_0 = v_0'^2/2g = 576/19.62 \approx \qty{29.4}{m}$.

\textbf{16.} $mgH' - f\,\dfrac{H'}{\sin\qty{35}{\degree}} =
\tfrac12 m v_0'^2$.

\textbf{17.} $H' = \dfrac{\tfrac12 \times 70 \times 576}
{686.7 - 80/0.574} = \dfrac{\num{20160}}{547} \approx \qty{37}{m}$.

\textbf{18.} $36.8/29.4 \approx 1.25$: tambahan sebesar $25\%$.

\textbf{19.} Untuk \qty{60}{kg}: $H' = \num{17280}/(588.6 - 139.5)
\approx \qty{38.5}{m}$. Pelompat yang \emph{lebih ringan} membutuhkan menara
yang lebih tinggi: tagihan \omterm{def:g10:inertia:inventory}{gesekannya} $fL$ sama saja, tetapi anggaran \omterm{def:g10:energy-conservation:energy}{energinya}
$mgH'$ lebih kecil.

\textbf{20.} Umumkanlah menara \qty{37}{m} untuk lepas landas \qty{24}{m/s}
--- lalu akuilah bahwa seperempatnya, sekitar \qty{7}{m} beton,
adalah tambahan yang dipungut \omterm{def:g10:inertia:inventory}{gesekan} atas impian tanpa \omterm{def:g10:inertia:inventory}{gesekan} itu.
\end{solution}
