\section*{Bab \ref{ch:g12:nuclear-energy} --- Energi Inti: Fisi, Fusi, $E=mc^2$}

\begin{solution}{exo:g12:nuclear-energy:1}
(a) $\qty{1}{eV} = \qty{1.60e-19}{J}$;
$\qty{1}{MeV} = \qty{1.60e-13}{J}$.
(b) $200 \times \num{1.60e-13} = \qty{3.2e-11}{J}$.
(c) $E = \num{1.66054e-27} \times \num{9.00e16} = \qty{1.49e-10}{J}
= \num{1.49e-10}/\num{1.60e-13} \approx \qty{9.3e2}{MeV}$ --- yaitu
\qty{931.5}{MeV} pada babnya, sejauh $c$ yang kita bulatkan.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:2}

$\num{5.4e14}/\num{3.6e7} = \num{1.5e7}$ hari $\approx \num{4.1e4}$
tahun --- empat puluh milenium dengan satu dadu gula.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:5}
Sekitar $7.1$, $7.7$, $8.8$ dan \qty{7.6}{MeV} tiap \omterm{def:g11:fundamental-interactions:ladder}{nukleon}; besi-56
yang paling terikat. Uranium dapat melepaskan \omterm{def:g10:energy-conservation:energy}{energi} lewat \omterm{def:g12:nuclear-energy:fission}{fisi}, helium-4 dan
karbon-12 lewat \omterm{def:g12:nuclear-energy:fusion}{fusi} --- dan kedua langkahnya menanjak menuju besi;
besi sendiri tidak punya tempat untuk pergi.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:7}

\begin{solution}{exo:g12:nuclear-energy:8}
$\Delta m = 2 \times 2.01355 - 3.01493 - 1.00866 = \qty{0.00351}{u}$:
$E \approx \qty{3.27}{MeV}$. D--T membuat helium-4, yang terikat luar biasa erat

\end{solution}

\begin{solution}{exo:g12:nuclear-energy:9}
(a) Neutronnya netral: tanpa \omterm{def:g12:nuclear-energy:barrier}{penghalang Coulomb}, \omterm{def:g12:nuclear-energy:fission}{fisinya} menyala dalam keadaan dingin.
Inti yang berfusi sama-sama positif dan harus menanjaki penghalangnya, sehingga hanya
guncangan $\sim \qty{1e8}{K}$ yang memberikan tumbukan yang cukup ganas. (b)
Terasnya memegang proton dalam jumlah yang mahabesar; anggota terkerap yang tercepat di
dalam kerumunannya yang berfusi, dan Matahari memiliki miliaran tahun yang berlebih
--- pembakaran yang lambat justru yang diinginkan sebuah bintang. (c) Tidak ada longsoran
neutron: tiap \omterm{def:g12:nuclear-energy:fusion}{fusinya} dibeli oleh tumbukannya sendiri, dan hilangnya
pengurungan mendinginkan \omterm{def:g12:nuclear-energy:barrier}{plasmanya} lalu memadamkan apinya.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:10}
$\Delta m = 4 \times 1.00728 - 4.00151 = \qty{0.02761}{u}$, sehingga
$E \approx \qty{25.7}{MeV}$; bagiannya
$0.02761/4.02912 \approx 0.7\%$ dari massanya.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:11}
(a) Ia memperlambat neutron \omterm{def:g12:nuclear-energy:fission}{fisi} lewat tumbukan; neutron yang lambat jauh
lebih mungkin membelah uranium-235, sehingga rantainya membutuhkannya. (b) Batangnya
menyerap neutron, menahan rantainya pada satu neutron tiap \omterm{def:g12:nuclear-energy:fission}{fisi} ---
ketika didorong masuk, rantainya mati. (c) Tanpa air, tanpa moderasi: neutronnya tetap
cepat, meleset, dan rantainya tercekik --- rancangannya gagal ke arah
``mati''.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:12}
\omterm{def:g12:nuclear-energy:fission}{Fisinya} berlipat dua tiap keturunan, sehingga setelah $n$ keturunan sekitar $2^n$
sudah terbelah; $2^{n} = \num{2.56e21} \approx 2 \times (10^{3})^{7}
\approx 2^{71}$ memberikan $n \approx 71$. Waktunya: $71 \times \qty{10}{ns}
\approx \qty{0.7}{\micro s}$ --- segram bahan bakar, senilai tiga ton batu bara,
dalam kurang dari semikrosekon: kekritisan itu tebing, bukan lereng.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:13}
$\Delta m = 55.92066 - 2 \times 27.96924 = \qty{-0.01782}{u}$:
$E \approx \qty{-16.6}{MeV}$ --- pembelahannya justru \emph{menyerap} \omterm{def:g10:energy-conservation:energy}{energi}.
Besi duduk di puncaknya: membelahnya maupun memadukannya sama-sama menuruni
ikatannya, keduanya berongkos \omterm{def:g10:energy-conservation:energy}{energi}. Ia adalah abu setiap api inti.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:14}
Fe: $\Delta m = 26 \times 1.00728 + 30 \times 1.00866 - 55.92066 =
\qty{0.52842}{u}$, $E_b \approx \qty{492}{MeV}$, $E_b/A \approx
\qty{8.79}{MeV}$. U: $\Delta m = 92 \times 1.00728 + 143 \times
1.00866 - 234.99350 = \qty{1.91464}{u}$, $E_b \approx \qty{1784}{MeV}$,
$E_b/A \approx \qty{7.59}{MeV}$. Mengemas ulang $235$ \omterm{def:g11:fundamental-interactions:ladder}{nukleon} pada
\qty{8.5}{MeV} dan bukan \qty{7.59}{MeV} memperoleh $235 \times 0.9
\approx \qty{2e2}{MeV}$ --- yaitu $185$--$200$ pada contoh yang dikerjakan itu.
\end{solution}

\begin{solution}{exo:g12:nuclear-energy:15}
$\num{5.0e8}/\num{2.82e-12} \approx \num{1.8e20}$ reaksi tiap
sekon; laju massanya $\num{1.8e20} \times 5.029 \times \num{1.66054e-27}
\approx \qty{1.5e-6}{kg/s}$, yaitu sekitar \qty{0.13}{kg} tiap hari ---
lawan \num{1.4e6} \omterm{def:g10:orders-of-magnitude:unit}{kilogram} batu bara: sepuluh juta berbanding satu.
\end{solution}

\begin{solution}{pb:g12:nuclear-energy:1}
\textbf{1.} $P_{\text{th}} = \num{1.0e9}/0.33 \approx \qty{3.0}{GW}$.

\textbf{2.} $200 \times \num{1.60e-13} = \qty{3.2e-11}{J}$.

\textbf{3.} $\num{3.0e9}/\num{3.2e-11} \approx \num{9.4e19}$ \omterm{def:g12:nuclear-energy:fission}{fisi}
tiap sekon.

\textbf{4.} $m_{\mathrm U} = 235 \times \num{1.66054e-27} =
\qty{3.90e-25}{kg}$; $\num{9.4e19} \times \num{3.90e-25} \approx
\qty{3.7e-5}{kg/s}$ --- sekitar empat puluh mikrogram tiap sekon.

\textbf{5.} $\num{3.7e-5} \times \num{3.16e7} \approx \qty{1.2e3}{kg}$:
sekitar $1.2$ ton uranium-235 tiap tahun.

\textbf{6.} $\num{1.2e3}/0.040 \approx \qty{2.9e4}{kg}$ --- sekitar $29$
ton bahan bakar: satu truk \omterm{def:g10:universal-gravitation:weight}{berat}, sekali setahun.

\textbf{7.} $\num{3.0e9} \times \num{3.16e7} \approx \qty{9.5e16}{J}$.

\textbf{8.} $\num{9.5e16}/\num{3.0e7} \approx \qty{3.2e9}{kg}$: $3.2$
juta ton batu bara.

\textbf{9.} $\num{3.2e9}/\num{1.2e3} \approx \num{2.7e6}$ --- yaitu
$\num{8.2e13}/\num{3.0e7}$ pada tangganya, sebagaimana seharusnya.

\textbf{10.} $\num{3.2e9}/\num{3.0e6} \approx 1060$ kereta tiap tahun,
kira-kira tiga tiap hari: kereta batu bara saat fajar, tengah hari dan senja selamanya,
lawan satu truk tiap musim semi.

\textbf{11.} $17.6 \times \num{1.60e-13} = \qty{2.82e-12}{J}$.

\textbf{12.} Satu pasangnya bermassa $5.029 \times \num{1.66054e-27} =
\qty{8.35e-27}{kg}$: $\num{2.82e-12}/\num{8.35e-27} \approx
\qty{3.4e14}{J/kg}$ --- anak tangga \omterm{def:g12:nuclear-energy:fusion}{fusi} pada tangganya.

\textbf{13.} $\num{9.5e16}/\num{3.4e14} \approx \qty{2.8e2}{kg}$
D--T; bagian deuteriumnya $280 \times 2.014/5.029 \approx \qty{1.1e2}{kg}$.

\textbf{14.} $\num{1.1e2}/\num{3.3e-2} \approx \qty{3.4e3}{m^3}$
air laut --- sekitar satu setengah kolam untuk setahun seluruh kotanya.

\textbf{15.} Tidak ada yang berlipat: tidak ada neutron yang menyalakan \omterm{def:g12:nuclear-energy:fusion}{fusi} berikutnya,
dan hilangnya pengurungan mendinginkan \omterm{def:g12:nuclear-energy:barrier}{plasmanya} lalu menghentikannya dalam beberapa sekon.
Yang sulit justru sebaliknya --- menahan \qty{1.5e8}{K} tetap bersama cukup
lama untuk terbakar.

\textbf{16.} $\num{3.9e26}/\num{9.00e16} \approx \qty{4.3e9}{kg/s}$:
empat juta ton tiap sekon.

\textbf{17.} $\num{3.0e9}/\num{9.00e16} \approx \qty{3.3e-8}{kg/s}$,
yaitu $\num{3.3e-8} \times \num{3.16e7} \approx \qty{1.0}{kg}$ tiap tahun
--- dan benar: $\qty{1.2e3}{kg} \times 0.00085 \approx 1$, $\num{3.2e9}
\times \num{3.3e-10} \approx 1$, $280 \times 0.00375 \approx 1$. Setiap
bahan bakarnya menyerahkan \omterm{def:g10:orders-of-magnitude:unit}{kilogram} yang sama; keduanya hanya berbeda pada seberapa banyak muatan
yang harus membawanya.

\textbf{18.} $\num{4.3e9} \times \num{3.16e7} \times \num{4.5e9}
\approx \qty{6.1e26}{kg}$ --- hanya $\num{6.1e26}/\num{2.0e30} \approx
\num{3e-4}$, tiga bagian dalam sepuluh ribu.

\textbf{19.} Massa yang dapat diubah $0.0007 \times \num{2.0e30} =
\qty{1.4e27}{kg}$, yang bernilai $\num{1.4e27} \times \num{9.00e16} \approx
\qty{1.3e44}{J}$; pada \qty{3.9e26}{W} nilai itu bertahan $\num{1.3e44}/
\num{3.9e26} \approx \qty{3.2e17}{s} \approx \num{1.0e10}$ tahun ---
sepuluh miliar, yang darinya sekitar lima miliar tersisa.

\textbf{20.} Setahun kotanya: $3.2$ juta ton batu bara, atau
$\num{1.2e3}$ \omterm{def:g10:orders-of-magnitude:unit}{kilogram} uranium-235, atau $\num{2.8e2}$ \omterm{def:g10:orders-of-magnitude:unit}{kilogram}
D--T yang ditimba dari satu setengah kolam air laut. Di bawah setiap buku besarnya
tercatat hal yang sama: satu \omterm{def:g10:orders-of-magnitude:unit}{kilogram} massa, lenyap. $E = mc^2$ tidak peduli bahan bakar
yang mana yang membawanya --- hanya sebesar apa truk yang dituntutnya.
\end{solution}
