\section*{Bab \ref{ch:g12:radioactive-decay} --- Peluruhan Radioaktif}

\begin{solution}{exo:g12:radioactive-decay:1}
$T_{1/2} = \ln 2/\lambda = 0.693/\num{1.0e-6} = \qty{6.9e5}{s}
\approx 8.0$ hari --- jam milik iodium-131.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:2}
$T_{1/2} = 3.8 \times \num{86400} = \qty{3.28e5}{s}$;
$\lambda = 0.693/\num{3.28e5} = \qty{2.1e-6}{s^{-1}}$;
$\mathcal A = \lambda N = \num{2.1e-6} \times \num{1.0e10} \approx
\qty{2.1e4}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:3}
$\dd N/\dd t = N_0(-\lambda)e^{-\lambda t} = -\lambda N$ ---
persamaannya berlaku; $N(0) = N_0 e^0 = N_0$;
$N(T_{1/2})/N_0 = e^{-\lambda \ln 2/\lambda} = e^{-\ln 2} =
\tfrac12$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:4}
$90$ tahun $= 3\,T_{1/2}$: $\mathcal A = \num{4.0e5}/2^3 =
\qty{5.0e4}{Bq}$. $300$ tahun $= 10\,T_{1/2}$:
$\mathcal A = \num{4.0e5}/1024 \approx \qty{3.9e2}{Bq}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:5}
Kemiringannya $= (18.85 - 20.00)/10 = -0.115$, sehingga
$\lambda = \qty{0.115}{h^{-1}} = 0.115/3600 = \qty{3.2e-5}{s^{-1}}$;
$T_{1/2} = 0.693/0.115 = 6.0$ jam --- teknesium-99m.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:6}
$N = (1.0/226) \times \num{6.02e23} = \num{2.7e21}$;
$T_{1/2} = 1600 \times \num{3.156e7} = \qty{5.05e10}{s}$, sehingga
$\lambda = \qty{1.37e-11}{s^{-1}}$;
$\mathcal A = \lambda N \approx \qty{3.7e10}{Bq}$ --- satu curie, yaitu
\omterm{def:g11:nucleus-radioactivity:activity}{aktivitas} segram radium milik Marie Curie.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:7}
Keduanya mengharapkan separuhnya tersisa: $10$, dan \num{1.0e20}. Fluktuasinya
$\sim 1/\sqrt{n}$ dari cacahannya: dengan $10$ peluruhan, sekitar $30\%$ ---
$7$ atau $13$ tidak akan mengejutkan; dengan \num{1.0e20}, sekitar
$10^{-10}$. Hanya contoh yang besar yang menaati hukumnya sampai ketelitian
yang terukur.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:8}
$m = 140 \times \num{1.2e-4} = \qty{1.7e-2}{g}$;
$N = (0.0168/40) \times \num{6.02e23} = \num{2.5e20}$.
$\lambda = 0.693/(\num{1.25e9} \times \num{3.156e7}) =
\qty{1.76e-17}{s^{-1}}$; $\mathcal A = \lambda N \approx
\qty{4.4e3}{Bq}$ --- sekitar separuh jumlah seluruh tubuhnya (sisanya kebanyakan
karbon-14).
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:9}
$\lambda = 0.693/6.0 = \qty{0.116}{h^{-1}}$. Pada $15$ jam:
$\mathcal A = 500\,e^{-0.116 \times 15} = 500 \times 0.177 \approx
\qty{88}{MBq}$. Di bawah \qty{1}{MBq}:
$t = \ln(500)/0.116 \approx \qty{54}{jam}$ --- sekitar $9$
\omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:10}
Arangnya: $3.4/13.6 = \tfrac14$, tepat $2$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}:
$t = \num{1.15e4}$ tahun. Tulangnya:
$t = \num{5730} \times \ln(1/0.30)/\ln 2 = \num{5730} \times 1.74
\approx \num{1.0e4}$ tahun.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:11}
$\lambda = 0.693/3.8 = \qty{0.182}{d^{-1}}$;
$t = \ln(100)/\lambda = 4.61/0.182 \approx 25$ hari (sekitar $6.6$
\omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}). Tetapi rantai uranium di dalam granit dan tanah di sekitarnya
membiakkan radon terus-menerus: pengudaraan sekali jalan terurai dalam
beberapa pekan, sehingga pengudaraannya harus tetap.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:12}
$N/N_0 = e^{-\lambda t}$ memberikan $\lambda t = \ln(N_0/N)$, dan
$\lambda = \ln 2/T_{1/2}$ memberikan $t = T_{1/2}\ln(N_0/N)/\ln 2$. Di sini
$t = \num{5730} \times \ln(20)/\ln 2 = \num{5730} \times 4.32
\approx \num{2.5e4}$ tahun.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:13}
$N_{\mathrm K} = N_0 e^{-\lambda t}$ dan
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, sehingga
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$. Dengan nisbah $3.0$:
$e^{\lambda t} = 4$, sehingga $\lambda t = 2\ln 2$ dan
$t = 2\,T_{1/2} = \num{2.5e9}$ tahun.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:14}
$\ln \mathcal A = 5.99$, $5.42$, $4.84$, $4.26$, $3.69$: turun sebesar
$0.57$--$0.58$ tiap \qty{60}{s} --- segaris. Kemiringannya:
$\lambda = 2.30/240 = \qty{9.6e-3}{s^{-1}}$;
$T_{1/2} = 0.693/\num{9.6e-3} \approx \qty{72}{s}$.
\end{solution}

\begin{solution}{exo:g12:radioactive-decay:15}
(a) $2^{-10} \approx \num{1e-3}$. (b) $(1/2)^8 = 1/256 \approx
0.4\%$. (c) $1000/8 = 125$ yang diharapkan --- tetapi hukumnya bungkam tentang
yang mana: ia meramalkan penduduknya dengan tepat dan perorangannya sama sekali tidak.
\end{solution}

\begin{solution}{pb:g12:radioactive-decay:1}
\textbf{1.} Pertukaran dengan atmosfernya (makan, bernapas) menjaga
perbandingannya tetap terisi pada nilai atmosfernya; kematian menghentikan
asupannya dan peluruhannya mengambil alih.

\textbf{2.} $T_{1/2} = \num{5730} \times \num{3.156e7} =
\qty{1.81e11}{s}$, sehingga $\lambda = 0.693/\num{1.81e11} =
\qty{3.8e-12}{s^{-1}}$.

\textbf{3.} $\dd N/\dd t = -\lambda N_0 e^{-\lambda t} = -\lambda N$,
dan $N(0) = N_0$.

\textbf{4.} $\mathcal A_0 = 13.6/60 = \qty{0.227}{Bq}$ tiap gram;
$N = \mathcal A_0/\lambda = 0.227/\num{3.8e-12} \approx \num{5.9e10}$
\omterm{def:g11:fundamental-interactions:ladder}{atom} tiap gram.

\textbf{5.} $(1/12) \times \num{6.02e23} = \num{5.0e22}$ \omterm{def:g11:fundamental-interactions:ladder}{atom}
karbon; perbandingannya $\num{5.9e10}/\num{5.0e22} \approx \num{1.2e-12}$
--- satu dalam $10^{12}$, seperti yang dijanjikan.

\textbf{6.} $\mathcal A = \mathcal A_0 e^{-\lambda t}$ memberikan
$\lambda t = \ln(\mathcal A_0/\mathcal A)$, dan $\lambda =
\ln 2/T_{1/2}$ mengubahnya menjadi
$t = T_{1/2}\ln(\mathcal A_0/\mathcal A)/\ln 2$.

\textbf{7.} $t = \num{5730} \times \ln(13.6/9.1)/\ln 2 =
\num{5730} \times 0.580 \approx \num{3.3e3}$ tahun.

\textbf{8.} Kematian kayunya --- penebangan pohonnya, bukan
pemahatannya; arca yang dipahat dari kayu tua (atau tiruan yang dipahat dari kayu
kuno) mendahului atau menyusul tarikh bahannya.

\textbf{9.} $t = \num{5730} \times \ln(13.6/10.4)/\ln 2 =
\num{5730} \times 0.387 \approx \num{2.2e3}$ tahun --- sepenuhnya
sesuai dengan seorang juru tulis dua puluh dua abad yang lalu.

\textbf{10.} $\Delta t \approx \Delta\mathcal A/(\lambda\mathcal A)$
dengan $\lambda = \ln 2/5730 = \num{1.21e-4}$ tiap tahun:
$\Delta t = (0.2/10.4)/\num{1.21e-4} \approx \num{1.6e2}$ tahun ---
sebuah tarikh yang tepat sampai satu atau dua abad.

\textbf{11.} $t = \num{5730} \times \ln(13.6/13.4)/\ln 2 \approx
120$ tahun --- tak terbedakan dari zaman kini di dalam ketakpastian
$\pm 160$ tahun itu: perkamennya baru, ``barang antik''-nya palsu.

\textbf{12.} $13.6/2^n < 0.2$ menuntut $2^n > 68$: $n = 7$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}
(karena $2^7 = 128$), sekitar $\num{4.0e4}$ tahun.

\textbf{13.} Sepuluh \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}: sekitar \num{5.7e4} tahun --- cakrawala
praktisnya, sekitar enam puluh ribu tahun.

\textbf{14.} $\num{6.6e7}/\num{5730} \approx \num{11500}$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}.
\num{5.9e10} \omterm{def:g11:fundamental-interactions:ladder}{atom} milik segramnya habis setelah
$2^n > \num{5.9e10}$, yaitu $n \approx 36$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh}
($\approx \num{2e5}$ tahun): jauh sebelum umur dinosaurusnya, tidak satu pun
\omterm{def:g11:fundamental-interactions:ladder}{atom} karbon-14 tersisa --- tidak ada yang dapat dicacah.

\textbf{15.} Sebuah jam yang jauh lebih lambat: kalium-40 atau uranium-238, dengan
\omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} bermiliar tahun.

\textbf{16.} Lava yang meleleh membiarkan argonnya menggelembung pergi, sehingga hablur yang
baru memadat memegang kalium tetapi nol argon: nisbahnya bermula
pada $0$ saat pemadatannya --- jamnya ternolkan.

\textbf{17.} Seperti pada latihannya: $N_{\mathrm K} = N_0 e^{-\lambda t}$,
$N_{\mathrm{Ar}} = N_0 - N_{\mathrm K}$, sehingga
$N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.

\textbf{18.} $e^{\lambda t} = 1.60$:
$t = \num{1.25e9} \times \ln(1.60)/\ln 2 = \num{1.25e9} \times 0.678
\approx \num{8.5e8}$ tahun.

\textbf{19.} Tidak, dua kali: pada \num{8.5e8} tahun karbon-14 sudah menempuh
$10^5$ \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} --- membisu; pada \num{3.3e3} tahun kalium-40 baru
menempuh \num{2.6e-6} \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} --- nisbah argonnya
$\approx \lambda t \sim \num{2e-6}$ terlalu kecil untuk diukur. Tiap
jam hanya membaca dari sekitar sepersepuluh \omterm{def:g11:nucleus-radioactivity:halflife}{waktu paruh} sampai sekitar sepuluh.

\textbf{20.} Arcanya $\approx \num{3.3e3}$ tahun, papirusnya
$\approx \num{2.2e3}$ tahun, batuannya $\approx \num{8.5e8}$ tahun ---
tiga jam, satu hukum: $N = N_0\,e^{-\lambda t}$, yang dibaca mundur.
\end{solution}
