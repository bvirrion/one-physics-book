\section*{Bab \ref{ch:g12:kinematics-2d} --- Kinematika pada Dua Dimensi}

\begin{solution}{exo:g12:kinematics-2d:1}
$\vect v = (3.0, 4.0)$, tetap; lajunya $\sqrt{3.0^2 + 4.0^2} =
\qty{5.0}{m/s}$. \omterm{def:g10:relative-motion:trajectory}{Lintasannya}: garis $y = \tfrac43 x$. \omterm{def:g12:kinematics-2d:velocity}{Vektor
kecepatannya} tetap: \omterm{def:g12:kinematics-2d:uniform}{gerak lurus beraturan}.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:2}
Garis lurus, jaraknya sama: \omterm{def:g12:kinematics-2d:uniform}{lurus beraturan};
$v = 0.35/0.10 = \qty{3.5}{m/s}$. Jarak yang melebar akan berarti
lajunya sedang bertambah.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:3}
$v(t) = 4.0\,t$, $a = \qty{4.0}{m/s^2}$: percepatannya tetap, lurus
berubah beraturan. $v(3.0) = \qty{12}{m/s}$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:4}
$f = 1/8.0 = \qty{0.125}{Hz}$;
$v = 2\pi \times 4.5/8.0 \approx \qty{3.5}{m/s}$;
$a = v^2/R = 3.53^2/4.5 \approx \qty{2.8}{m/s^2}$, yang menunjuk ke arah
sumbu komidi putarnya.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:5}
Kemiringannya: $a = (10.0 - 2.0)/4.0 = \qty{2.0}{m/s^2}$. Luasnya
(trapesium): $d = \tfrac12 (2.0 + 10.0) \times 4.0 = \qty{24}{m}$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:6}
$v = \qty{88.9}{m/s}$: $a_n = 88.9^2/6000 \approx \qty{1.3}{m/s^2}
\approx 0.13\,g$. Karena $a_n = v^2/R$, $v$ yang besar menuntut $R$ yang
raksasa agar percepatan ke sampingnya (dan \omterm{def:g10:inertia:force}{gaya} pada jalur serta
penumpangnya) tetap wajar --- memiringkan jalurnya mengubah siapa yang
memasoknya, bukan besarnya.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:7}
$v(t) = -2.4\,t + 8.0$; $a = \qty{-2.4}{m/s^2}$, tetap. Berbaliknya
pada $v = 0$: $t = 8.0/2.4 \approx \qty{3.3}{s}$, di
$x \approx \qty{16.3}{m}$. Sebelumnya: maju, melambat; sesudahnya:
mundur, bertambah cepat --- berubah beraturan sepanjang waktu.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:8}
$v_1 = 0.28/0.40 = \qty{0.70}{m/s}$;
$v_2 = (0.54 - 0.10)/0.40 = \qty{1.10}{m/s}$;
$v_3 = (0.88 - 0.28)/0.40 = \qty{1.50}{m/s}$. Kecepatannya bertambah
\qty{0.40}{m/s} tiap \qty{0.20}{s}: $a = \qty{2.0}{m/s^2}$, tetap ---
lurus berubah beraturan.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:9}
$T = \qty{2.36e6}{s}$: $v = 2\pi \times \num{3.84e8}/\num{2.36e6}
\approx \qty{1.0e3}{m/s}$;
$a_n = v^2/R \approx \qty{2.7e-3}{m/s^2}$. Perbandingannya:
$g/a_n \approx 3600 = 60^2$ --- gravitasi, yang diencerkan sebagai
kebalikan kuadrat jaraknya (\cref{ch:g10:universal-gravitation}), tepat
menerangkan berbeloknya Bulan.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:10}
$f = 1200/60 = \qty{20}{Hz}$, $T = \qty{0.050}{s}$;
$v = 2\pi R f = 2\pi \times 0.25 \times 20 \approx \qty{31}{m/s}$;
$a_n = v^2/R \approx \qty{3.9e3}{m/s^2} \approx 400\,g$. Dinding
tabungnya memasok $a_n$ untuk cuciannya; melalui lubangnya tidak ada
yang mendorong airnya ke dalam, sehingga airnya terlempar menyinggung.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:11}
$v_0 = \qty{25}{m/s}$: $t_s = 25/7.5 \approx \qty{3.3}{s}$;
$d = v_0^2/(2a) = 625/15 \approx \qty{42}{m}$. Grafiknya: sebuah ruas
lurus dari \qty{25}{m/s} turun ke $0$ pada $t_s$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:12}
$\vect v = (-R\omega\sin\omega t,\, R\omega\cos\omega t)$, yang
normanya tetap $R\omega$; hasil kali titiknya dengan $\vect{OM}$
lenyap, sehingga tegak lurus jari-jarinya, jadi menyinggung. Dengan
menurunkannya sekali lagi, $\vect a = -\omega^2\vect{OM}$: menuju $O$,
dengan norma $\omega^2 R = (v/R)^2 R = v^2/R$.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:13}
$a_t = \qty{2.5}{m/s^2}$ (ke belakang), $a_n = 20^2/80 =
\qty{5.0}{m/s^2}$;
$a = \sqrt{2.5^2 + 5.0^2} \approx \qty{5.6}{m/s^2}$; sudutnya terhadap
arah geraknya: $\theta = 180^\circ - \arctan(5.0/2.5) \approx 117^\circ$
($63^\circ$ dari arah lurus ke belakang). $5.6 < 7.0$: masih tercengkeram,
dengan sisa yang \omterm{def:g11:lenses-and-eye:lens}{tipis}.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:14}
Bertolaknya: $t_1 = 20/1.2 \approx \qty{17}{s}$,
$d_1 = 20^2/(2 \times 1.2) \approx \qty{167}{m}$. Melaju tetap:
\qty{40}{s}, $\qty{800}{m}$. Mengeremnya: $t_3 = 20/1.5 \approx
\qty{13}{s}$, $d_3 = 400/3.0 \approx \qty{133}{m}$. Seluruhnya:
$\qty{1.1e3}{m}$ dalam \qty{70}{s}; rata-ratanya $1100/70 \approx
\qty{16}{m/s}$ (\qty{57}{km/h}). $v(t)$ berupa trapesium yang luasnya,
$\tfrac12(70 + 40) \times 20 = \qty{1100}{m}$, memeriksa jumlahnya.
\end{solution}

\begin{solution}{exo:g12:kinematics-2d:15}
$v = 2\pi \times \num{4.22e7}/\num{86164} \approx \qty{3.1e3}{m/s}$;
$a_n = v^2/R \approx \qty{0.22}{m/s^2}$. Gravitasi Bumi pada jarak itu
memasoknya; \omterm{def:g12:kinematics-2d:ucm}{periodenya} sama dengan \omterm{def:g12:kinematics-2d:ucm}{periode} putaran Bumi, sehingga
\omterm{def:g10:universal-gravitation:satellite}{satelit} dan tanahnya berputar bersama dan \omterm{def:g10:universal-gravitation:satellite}{satelit} itu menggantung di
atas satu titik khatulistiwa (\cref{ch:g12:satellites-kepler}).
\end{solution}

\begin{solution}{pb:g12:kinematics-2d:1}
\textbf{1.} Gerak bersifat nisbi (\cref{ch:g10:relative-motion}): setiap
kedudukan dan kecepatan harus menyebutkan kerangkanya. Di sini: \omterm{def:g10:relative-motion:frames}{kerangka
tanah} (jalannya), $x$ sepanjang jalannya dari kedudukan bingkai pertama.

\textbf{2.} Tiap selangnya: $9.0/0.50 = \qty{18.0}{m/s}$ --- empat nilai
yang sama.

\textbf{3.} Jalannya lurus, perpindahannya sama dalam waktu yang sama:
\omterm{def:g12:kinematics-2d:uniform}{gerak lurus beraturan}.

\textbf{4.} $v_0 = \qty{18.0}{m/s} = \qty{64.8}{km/h}$.

\textbf{5.} $\vect a = \vect 0$.

\textbf{6.} $v(t) = -a\,t + v_B$; $x(t) = -\tfrac12 a t^2 + v_B\,t$.

\textbf{7.} $v(t_s) = 0$ memberikan $t_s = v_B/a$; selanjutnya
$d_b = -\tfrac12 a (v_B/a)^2 + v_B^2/a = v_B^2/(2a)$.

\textbf{8.} $v_B = \sqrt{2 a d_b} = \sqrt{2 \times 6.0 \times 27} =
\sqrt{324} = \qty{18.0}{m/s}$ --- tepat laju terekamnya: pengemudinya
tidak pernah melambat sebelum mengerem.

\textbf{9.} Tanggapannya: $18.0 \times 1.0 = \qty{18}{m}$; seluruhnya
$18 + 27 = \qty{45}{m}$.

\textbf{10.} $d_b = v_B^2/(2a) \propto v_B^2$: menggandakan $v_B$
melipatkan $d_b$ dengan $4$. Pada \qty{50}{km/h} ($\qty{13.9}{m/s}$):
$d_b = 13.9^2/12.0 \approx \qty{16}{m}$.

\textbf{11.} Lajunya tetap tetapi arah kecepatannya berbelok:
$\vect v$ tidaklah tetap, sehingga mobilnya berpercepatan.

\textbf{12.} Seperempat putarannya: $2\pi R/4 = \qty{19.6}{m}$;
$v = 19.6/2.5 \approx \qty{7.9}{m/s}$ (\qty{28}{km/h}).

\textbf{13.} $a_n = 7.85^2/12.5 \approx \qty{4.9}{m/s^2}$, menuju pusat
bundarannya.

\textbf{14.} $a_{\max} = \mu g = 0.70 \times 9.81 \approx
\qty{6.9}{m/s^2}$; $v_{\max} = \sqrt{\mu g R} =
\sqrt{6.87 \times 12.5} \approx \qty{9.3}{m/s}$ (\qty{33}{km/h}).
$7.9 < 9.3$: masih tercengkeram --- sesuai dengan lajur yang bersih,
tanpa bekas tergelincir.

\textbf{15.} $T = 2\pi R/v = 78.5/7.85 \approx \qty{10.0}{s};
f = \qty{0.10}{Hz}$.

\textbf{16.} $64.8 - 50 \approx \qty{15}{km/h}$ di atasnya --- sekitar
$30\%$ melampaui batasnya.

\textbf{17.} Pengeremannya bekerja sepanjang $41 - 18 = \qty{23}{m}$:
$v^2 = v_B^2 - 2ad = 324 - 2 \times 6.0 \times 23 = 48$, sehingga
$v \approx \qty{6.9}{m/s} \approx \qty{25}{km/h}$ saat menabrak.

\textbf{18.} Pada \qty{13.9}{m/s}: $13.9 \times 1.0 + 13.9^2/12.0 =
13.9 + 16.1 = \qty{30.0}{m} < \qty{41}{m}$ --- mobilnya berhenti
\qty{11}{m} sebelum vannya.

\textbf{19.} $d(v) = v\,t_r + v^2/(2a)$: sekali secara linear
(tanggapannya menempuh $v t_r$ sebelum remnya bekerja), sekali secara
kuadrat (bagian kinetiknya $v^2/(2a)$) --- itulah sebabnya kelebihan
yang sedang pun berongkos tak sebanding.

\textbf{20.} Kamera dasbor dan bekas bannya sepakat: \qty{65}{km/h} di
kawasan \qty{50}{km/h}; pada batas kecepatannya mobil itu akan berhenti
\qty{11}{m} sebelum vannya --- kecelakaannya adalah kelebihan
\qty{15}{km/h} itu.
\end{solution}
