\section*{Bab \ref{ch:g12:special-relativity} --- Kerelatifan Khusus: Pemuluran Waktu}

\begin{solution}{exo:g12:special-relativity:1}
$\gamma = 1/\sqrt{1 - (v/c)^2}$: $1.01$ ($0.1c$); $1.25$ ($0.6c$);
$1.67$ ($0.8c$); $10.0$ ($0.995c$).
\end{solution}

\begin{solution}{exo:g12:special-relativity:2}
$\gamma = 1.25$, sehingga tanahnya mengukur $1.25 \times 10.0 =
\qty{12.5}{s}$. \qty{10.0}{s} milik wahananya adalah \omterm{def:g12:special-relativity:proper-time}{waktu dirinya}:
metronomnya (satu jam) hadir pada kedua detakannya.
\end{solution}

\begin{solution}{exo:g12:special-relativity:3}
$v = c\sqrt{1 - 1/\gamma^2}$: $\gamma = 2$ memberikan $v = 0.866c$;
$\gamma = 10$ memberikan $v = 0.995c$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:4}
$v = \qty{36.1}{m/s}$, $v/c = \num{1.2e-7}$;
$\gamma - 1 \approx (\num{1.2e-7})^2/2 \approx \num{7.2e-15}$. Satu
sekon hilang setelah $1/\num{7.2e-15} \approx \qty{1.4e14}{s}$ --- sekitar
empat juta tahun berkendara.
\end{solution}

\begin{solution}{exo:g12:special-relativity:5}
Penumpangnya melaju menuju kilat depannya, sehingga kilat itu mencapainya lebih dahulu;
menurut postulat keduanya kedua kilatnya berjalan pada $c$ yang sama sepanjang
separuh panjang kereta yang sama \emph{pada kerangkanya}, sehingga kedatangan yang lebih awal
berarti sambaran yang lebih awal. Keserentakan bergantung pada kerangkanya: tiap putusannya
benar pada kerangkanya sendiri.
\end{solution}

\begin{solution}{exo:g12:special-relativity:6}
$\gamma = 1/\sqrt{1 - 0.99^2} = 7.09$; umurnya di laboratorium
$7.09 \times \num{2.6e-8} = \qty{1.8e-7}{s}$. \omterm{def:g12:projectile-motion:range}{Jangkauan} klasiknya
$0.99 \times \num{3.00e8} \times \num{2.6e-8} \approx \qty{7.7}{m}$;
yang sesungguhnya $7.09 \times 7.7 \approx \qty{55}{m}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:7}
$\gamma = 1.5$ pada $v = c\sqrt{1 - 1/2.25} \approx 0.75c$;
$\gamma = 3$ pada $\approx 0.94c$. Asimtotnya: $\gamma$ (dan ongkos
\omterm{def:g10:energy-conservation:energy}{energi} untuk laju tambahannya) memencar sembari $v \to c$ --- tidak ada \omterm{def:g11:work-of-force:work}{usaha}
berhingga yang membawa benda bermassa sampai laju cahaya.
\end{solution}

\begin{solution}{exo:g12:special-relativity:8}
$\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$; pada $0.8c$,
$\gamma = 5/3$, sehingga $\Delta t = \qty{1.7e-8}{s}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:9}
\omterm{def:g12:special-relativity:proper-time}{Waktu diri} adalah milik jam yang hadir pada kedua \omterm{def:g12:special-relativity:proper-time}{peristiwanya}: (a) yang
memakainya; (b) muonnya; (c) pilotnya --- hanya wahananya yang hadir pada
keberangkatan dan kedatangannya.
\end{solution}

\begin{solution}{exo:g12:special-relativity:10}
$\gamma - 1 \approx 7700^2/(2 \times \num{9.0e16}) \approx
\num{3.3e-10}$; sepanjang \qty{1.58e7}{s} antariksawannya tertinggal sebesar
$\num{3.3e-10} \times \num{1.58e7} \approx \qty{5.2e-3}{s}$ --- sekitar
lima milisekon lebih muda.
\end{solution}

\begin{solution}{exo:g12:special-relativity:11}
$E = \num{1.0e-3} \times (\num{3.00e8})^2 = \qty{9.0e13}{J}$. Sebuah
pembangkit \qty{1.0}{GW} menyerahkan $\num{1.0e9} \times 86400 \approx
\qty{8.6e13}{J}$ tiap hari --- segram adalah sehari keluarannya, dan
$\num{9.0e13}/\num{8.5e5} \approx \num{e8}$: seratus juta
mobil yang melaju kencang.
\end{solution}

\begin{solution}{exo:g12:special-relativity:12}
Kuadratkan lalu balikkan: $\gamma^2(1 - v^2/c^2) = 1$, sehingga
$v^2/c^2 = 1 - 1/\gamma^2$, jadi $v = c\sqrt{1 - 1/\gamma^2}$. Untuk
$\gamma = 30$: $v = c\sqrt{1 - 1/900} = 0.99944c$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:13}
Tebal yang terkerut $L = \qty{15}{km}/30 = \qty{500}{m}$; waktu sapuannya
$500/(0.99944 \times \num{3.00e8}) \approx \qty{1.7e-6}{s} <
\qty{2.2}{\micro s}$. Kedua kerangkanya sepakat bahwa muonnya mencapai tanah ---
yang satu menyalahkan umur yang teregang, yang lain atmosfer yang menyusut.
\end{solution}

\begin{solution}{exo:g12:special-relativity:14}
Di laboratorium: $3200/\num{3.00e8} \approx \qty{1.1e-5}{s}$ (lajunya
$\approx c$). Pada kerangka elektronnya: tabungnya terkerut menjadi
$3200/\num{1.0e5} = \qty{3.2}{cm}$, yang terlintasi dalam
$\num{1.1e-5}/\num{1.0e5} \approx \qty{1.1e-10}{s}$.
\end{solution}

\begin{solution}{exo:g12:special-relativity:15}
Bumi: $4.2/0.9 \approx \qty{4.7}{tahun}$. Dengan
$\gamma = 1/\sqrt{1 - 0.81} = 2.29$, jam wahananya mencatat
$4.7/2.29 \approx \qty{2.0}{tahun}$.
\end{solution}

\begin{solution}{pb:g12:special-relativity:1}
\textbf{1.} $c\,\Delta t_0 = \num{3.00e8} \times \num{2.2e-6} \approx
\qty{660}{m}$.

\textbf{2.} $\num{15000}/660 \approx 23$ umur.

\textbf{3.} $\mathrm{e}^{-23} \approx \num{e-10}$: kurang dari satu muon
dalam sepuluh miliar yang seharusnya tiba, namun fluksnya satu tiap \unit{cm^2} tiap
menit --- fisika klasik dibantah mentah-mentah oleh langitnya.

\textbf{4.} Umurnya menurut tanah $30 \times \qty{2.2}{\micro s} =
\qty{66}{\micro s}$; \omterm{def:g12:projectile-motion:range}{jangkauannya} $\approx \num{3.00e8} \times \num{6.6e-5}
\approx \qty{20}{km} > \qty{15}{km}$: kedatangannya terjelaskan.

\textbf{5.} $v = c\sqrt{1 - 1/900} = 0.99944c$.

\textbf{6.} $\Delta t_0 = 2L/c = 3.0/\num{3.00e8} = \qty{1.0e-8}{s}$.

\textbf{7.} Mendatarnya $v\,\Delta t/2$; tegaknya $L$; separuh \omterm{def:g10:relative-motion:trajectory}{lintasan}
miringnya berukuran $c\,\Delta t/2$ karena postulat keduanya menetapkan
laju \omterm{def:g12:quantum-world:photon}{fotonnya} pada $c$ di \omterm{def:g10:relative-motion:frames}{kerangka tanahnya} juga.

\textbf{8.} $(c\,\Delta t/2)^2 = L^2 + (v\,\Delta t/2)^2$ memberikan
$\Delta t^2(c^2 - v^2) = 4L^2$, sehingga $\Delta t =
(2L/c)/\sqrt{1 - v^2/c^2} = \gamma\,\Delta t_0$.

\textbf{9.} $\gamma = 1/\sqrt{1 - 0.36} = 1.25$;
$\Delta t = \qty{1.25e-8}{s}$.

\textbf{10.} Milik wahananya: satu jamnya hadir pada kedua detiknya, sehingga jam itu
membaca \omterm{def:g12:special-relativity:proper-time}{waktu dirinya}; tanahnya menuntut dua jam yang terselaraskan.

\textbf{11.} $\gamma = 1/\sqrt{1 - 0.64} = 1/0.6 = 5/3 \approx 1.67$.

\textbf{12.} $10.0/(5/3) = \qty{6.0}{tahun}$.

\textbf{13.} Perginya $\qty{5.0}{tahun}$ pada $0.8c$: $4.0$ \omterm{def:g10:orders-of-magnitude:lightyear}{tahun cahaya}.

\textbf{14.} $10.0 - 6.0 = \qty{4.0}{tahun}$: antariksawannya empat
tahun lebih muda daripada kembarannya.

\textbf{15.} Keadaannya tidak setangkup: hanya antariksawannya yang
berbalik --- ia berpercepatan lalu berganti \omterm{def:g12:special-relativity:inertial}{kerangka inersial}, sementara kembarannya di
Bumi tetap pada satu kerangka --- sehingga hanya jamnya yang mencatat waktu yang lebih pendek.

\textbf{16.} $v = 2\pi \times \num{2.66e7}/\num{43200} \approx
\qty{3.9e3}{m/s}$.

\textbf{17.} $\gamma - 1 \approx (\num{3.87e3})^2/(2 \times
\num{9.0e16}) \approx \num{8.3e-11}$.

\textbf{18.} $\num{8.3e-11} \times \num{86400} \approx \qty{7.2e-6}{s}
= \qty{7.2}{\micro s}$ tiap hari lebih lambat.

\textbf{19.} $c \times \num{7.2e-6} \approx \qty{2.2}{km}$ galat kedudukan
tiap hari.

\textbf{20.} Simpangan bersihnya $45 - 7.2 \approx \qty{38}{\micro s}$ tiap hari
lebih cepat; galatnya $c \times \num{38e-6} \approx \qty{11}{km}$ tiap hari ---
tagihan kerelatifan yang dibayar di muka oleh setiap jam GPS, yang dilaras keliru di
tanah agar ia berdetik tepat di \omterm{def:g10:universal-gravitation:satellite}{orbitnya}.
\end{solution}
