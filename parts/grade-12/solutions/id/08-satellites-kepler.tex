\section*{Bab \ref{ch:g12:satellites-kepler} --- Satelit dan Gerak Planet}

\begin{solution}{exo:g12:satellites-kepler:1}
Di landasannya: $F = GMm/R^2 = \num{3.98e17}/\num{4.06e13} \approx
\qty{9.8e3}{N}$. Pada $r = \qty{6.77e6}{m}$:
$F \approx \qty{8.7e3}{N}$ --- masih $89\%$ nilai di tanahnya. Para
penghuninya melayang karena mereka \emph{jatuh} bersama stasiunnya,
bukan karena gravitasinya lenyap.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:2}
$v = \sqrt{GM/R} = \sqrt{\num{3.98e14}/\num{6.37e6}} \approx
\qty{7.9e3}{m/s}$; $T = 2\pi R/v \approx \qty{5.1e3}{s} \approx
\qty{84}{min}$. Persis \qty{7.9}{km/s} milik peluru meriamnya: \omterm{def:g10:universal-gravitation:satellite}{orbit}
yang menyerempet itu \emph{adalah} tembakan Newton.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:3}
$v \propto 1/\sqrt{r}$: dibagi $\sqrt 2 \approx 1.41$.
$T \propto r^{3/2}$: dikalikan $2^{3/2} \approx 2.8$. Tidak satu pun
memuat $m$: tidak.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:4}
$v = \sqrt{\num{3.98e14}/\num{6.77e6}} \approx \qty{7.67e3}{m/s}$;
$T = 2\pi r/v \approx \qty{5.55e3}{s} \approx \qty{92}{min}$;
$\num{86400}/5546 \approx 15.6$ \omterm{def:g10:universal-gravitation:satellite}{orbit} tiap hari.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:5}
\omterm{def:g12:satellites-kepler:ellipse}{Elips} dengan Matahari pada salah satu fokusnya; luas yang sama dalam waktu
yang sama; $T^2/a^3$ serupa bagi semua planet. Paling cepat di \omterm{def:g12:satellites-kepler:ellipse}{perihelion},
menurut hukum keduanya (ruas sapuan yang pendek harus menyapu cepat). Tidak:
massa \omterm{def:g10:universal-gravitation:satellite}{satelitnya} \omterm{def:g10:inertia:compensate}{saling meniadakan} --- hanya massa pusatnya yang masuk.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:6}
$T = \qty{2.36e6}{s}$: $M = 4\pi^2 r^3/(GT^2) =
\num{2.24e27}/371 \approx \qty{6.0e24}{kg}$ --- dalam selisih $1\%$ dari
\qty{5.97e24}{kg}.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:7}
Gravitasi menunjuk ke pusat Bumi, dan pada gerak melingkar \omterm{def:g11:forces-and-motion:netforce}{gaya
totalnya} menunjuk ke pusat lingkarannya: kedua pusat itu berimpit. Sebuah
lingkaran di atas garis lintang Paris berpusat pada sumbunya, bukan pada
pusat Bumi --- mustahil. Menggantung diam hanya berhasil di tempat garis
lintangnya berupa lingkaran besar yang berpusat di pusatnya: di atas
\emph{khatulistiwa}.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:8}
$T = \qty{2.754e4}{s}$: $M = 4\pi^2 r^3/(GT^2) =
\num{3.26e22}/\num{5.06e-2} \approx \qty{6.4e23}{kg}$ --- Mars pada
kartu datanya.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:9}
Bumi: $T^2/a^3 = \num{9.96e14}/\num{3.35e33} =
\qty{2.98e-19}{s^2/m^3}$. Mars: $T = \qty{5.94e7}{s}$,
$\num{3.52e15}/\num{1.18e34} = \qty{2.98e-19}{s^2/m^3}$ --- sama, seperti
yang dituntut hukum ketiganya. Selanjutnya $M = 4\pi^2/(G \times \num{2.98e-19})
\approx \qty{1.99e30}{kg}$.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:10}
$v = 2\pi r/T = \qty{1.02e3}{m/s}$; $a = v^2/r =
\qty{2.72e-3}{m/s^2}$; $g/60^2 = 9.81/3600 = \qty{2.73e-3}{m/s^2}$.
Bulan terletak $60$ jari-jari Bumi jauhnya, dan jatuhnya $60^2$ kali
lebih lemah daripada jatuh apelnya: persis kebalikan kuadratnya --- satu
hukum bagi keduanya.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:11}
Ceres: $T = 2.77^{3/2} = \sqrt{21.3} \approx \qty{4.6}{tahun}$. Halley:
$T = 17.8^{3/2} \approx \qty{75}{tahun}$, sehingga \omterm{def:g12:satellites-kepler:ellipse}{perihelion} berikutnya sekitar
$1986 + 75 \approx 2061$.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:12}
Pada \qty{400}{km}: $v \approx \qty{7.67e3}{m/s}$; pada \qty{350}{km}
($r = \qty{6.72e6}{m}$): $v \approx \qty{7.70e3}{m/s}$. \omterm{def:g10:inertia:inventory}{Gesekannya} memang
menguras \omterm{def:g11:mechanical-energy:em}{energi mekanik} --- \omterm{def:g10:universal-gravitation:satellite}{orbitnya} menyusut; tetapi dalam perjalanan
turunnya kerja gravitasi lebih daripada melunasi susutnya laju akibat
\omterm{def:g11:circuits-and-power:resistance}{hambatannya}: $E_m$ turun sementara $E_k$ naik.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:13}
$T = \qty{3.024e5}{s}$: $M = 4\pi^2 r^3/(GT^2) =
\num{1.65e31}/6.10 \approx \qty{2.7e30}{kg} \approx 1.4$ kali Matahari. Massa
\emph{planetnya} tidak terjangkau: massa itu sudah \omterm{def:g10:inertia:compensate}{saling meniadakan} dari
persamaan \omterm{def:g10:universal-gravitation:satellite}{orbitnya}.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:14}
$T = \qty{8.862e4}{s}$, $GM = \qty{4.28e13}{m^3/s^2}$:
$r^3 = GMT^2/(4\pi^2) = \num{8.52e21}$, sehingga $r \approx
\qty{2.04e7}{m}$ dan $h = r - R \approx \qty{1.7e7}{m} \approx
\qty{17000}{km}$ di atas tanah Mars.
\end{solution}

\begin{solution}{exo:g12:satellites-kepler:15}
$v_a = r_p v_p / r_a = \num{8.77e10} \times \num{5.45e4} /
\num{5.25e12} \approx \qty{9.1e2}{m/s}$. Karena $r_a/r_p \approx 60$,
kometnya $60$ kali lebih jauh dan $60$ kali lebih lambat: ia menghabiskan
puluhan tahun berlambat di kegelapan dan beberapa pekan melesat melewati Matahari.
\end{solution}

\begin{solution}{pb:g12:satellites-kepler:1}
\textbf{1.} $T = \qty{86164}{s}$: tanahnya berputar bersama Bumi
\emph{terhadap bintang} sekali tiap \omterm{def:g12:satellites-kepler:geo}{hari sideris}; hari \qty{24}{h}
menambahkan satu derajat tiap hari yang ditempuh Bumi mengelilingi Matahari.

\textbf{2.} \omterm{def:g11:forces-and-motion:netforce}{Gaya total} pada gerak melingkar membidik pusat lingkarannya;
satu-satunya \omterm{def:g10:inertia:force}{gaya}, yaitu gravitasi, membidik pusat Bumi: kedua
pusatnya adalah titik yang sama.

\textbf{3.} Lingkaran yang menggantung di atas Paris akan berpusat pada
sumbunya di sebelah utara pusatnya --- tersingkir oleh nomor 2. \omterm{def:g10:universal-gravitation:satellite}{Orbitnya}
harus terletak pada \emph{bidang khatulistiwa}.

\textbf{4.} Komponen normal \omterm{thm:g12:newtons-laws:second}{hukum kedua Newton}:
$m v^2/r = GmM/r^2$, sehingga $v = \sqrt{GM/r}$ ($m$ \omterm{def:g10:inertia:compensate}{saling meniadakan}).

\textbf{5.} $T = 2\pi r/v = 2\pi\sqrt{r^3/(GM)}$; dengan mengkuadratkannya,
$T^2/r^3 = 4\pi^2/(GM)$.

\textbf{6.} $r = \left(GMT^2/4\pi^2\right)^{1/3} =
(\num{7.49e22})^{1/3} \approx \qty{4.22e7}{m}$.

\textbf{7.} $h = r - R = \num{4.22e7} - \num{6.37e6} \approx
\qty{3.58e7}{m} \approx \qty{35800}{km}$.

\textbf{8.} $v = 2\pi \times \num{4.22e7}/\num{86164} \approx
\qty{3.07e3}{m/s}$.

\textbf{9.} $\sqrt{\num{3.98e14}/\num{4.22e7}} \approx
\qty{3.07e3}{m/s}$ --- consistent.

\textbf{10.} Jari-jarinya serupa: massanya sudah \omterm{def:g10:inertia:compensate}{saling meniadakan} pada
nomor 4. \omterm{def:g10:universal-gravitation:satellite}{Satelit} yang lebih berat hanya mengubah ongkos peluncurannya ---
lebih banyak bahan bakar untuk \omterm{def:g10:universal-gravitation:satellite}{orbit} yang sama.

\textbf{11.} $v = \sqrt{\num{3.98e14}/\num{6.77e6}} \approx
\qty{7.67e3}{m/s}$.

\textbf{12.} $T = 2\pi r/v \approx \qty{5.55e3}{s} \approx
\qty{92.4}{min}$.

\textbf{13.} $\num{86400}/5546 \approx 15.6$ \omterm{def:g10:universal-gravitation:satellite}{orbit}: sekitar $16$
matahari terbit tiap hari.

\textbf{14.} $r_{\text{geo}}/r_{\text{sta}} = \num{4.22e7}/\num{6.77e6}
= 6.23$; $6.23^{3/2} \approx 15.5 = \num{86164}/5546$ --- hukum ketiga
Kepler, yang terbukti seketika.

\textbf{15.} Stasiunnya lebih cepat (\qty{7.67}{km/s} lawan
\qty{3.07}{km/s}), namun yang \omterm{def:g12:satellites-kepler:geo}{geostasioner} menuntut lebih banyak \omterm{def:g10:energy-conservation:energy}{energi} tiap
\omterm{def:g10:orders-of-magnitude:unit}{kilogram}: tanjakan \omterm{def:g10:energy-conservation:energy}{energi} potensialnya mengalahkan susutnya lajunya.

\textbf{16.} $M_J = 4\pi^2 r^3/(G T^2)$.

\textbf{17.} $T = \qty{1.529e5}{s}$: $M_J =
\num{2.97e27}/1.56 \approx \qty{1.90e27}{kg}$.

\textbf{18.} $\num{1.90e27}/\num{5.97e24} \approx 318$ kali Bumi ---
dan sekitar $1/1000$ massa Matahari.

\textbf{19.} $T_J = \qty{3.576e4}{s}$: $r^3 = GM_J T_J^2/(4\pi^2) =
\num{4.11e24}$, $r \approx \qty{1.60e8}{m}$ --- Io, pada
\qty{4.22e8}{m}, terbang $2.6$ kali lebih tinggi lalu melayang melintasi
langit Yupiter.

\textbf{20.} Parkirlah penerus isyaratnya pada lingkaran khatulistiwa
berketinggian \qty{35800}{km}, tempat ia melaju pada \qty{3.07}{km/s} dan
tidak pernah meninggalkan bidikan antenanya; dan hukum yang sama, yang diberi
bulan milik Io, menimbang Yupiter pada
$\qty{1.90e27}{kg}$ --- $318$ kali Bumi --- secara cuma-cuma.
\end{solution}
