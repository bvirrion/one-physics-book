\chapter{Vrije val en de worp}\label{ch:g12:projectile-motion}

Elke straal uit een tuinslang tekent dezelfde kromme in de lucht. Een
basketbal op weg naar de ring ook, en een verspringer, en een kanonskogel:
elk lichaam dat aan zijn \omterm{def:g10:universal-gravitation:weight}{gewicht} wordt overgelaten. Dit hoofdstuk verdient
die kromme --- een parabool --- in twee regels rekenwerk uit de tweede wet
van Newton, leest toppunt, \omterm{def:g12:projectile-motion:range}{vluchttijd} en \omterm{def:g12:projectile-motion:range}{dracht} van haar vergelijking af,
en laat dan de lucht weer toe om te zien wat die bederft.

\section{De vrije val}

\begin{definition}[Vrije val]\label{def:g12:projectile-motion:freefall}
Een lichaam is in \emph{vrije val}\index{vrije val} wanneer de enige \omterm{def:g10:inertia:force}{kracht}
die erop werkt zijn \omterm{def:g10:universal-gravitation:weight}{gewicht} $\vect P = m\vect g$ is: geen steun, geen draad,
verwaarloosbare luchtweerstand. Een losgelaten steen of een bal die de hand
heeft verlaten is tot in goede benadering in vrije val --- ``vallen'' sluit
stijgen in.
\end{definition}

\begin{theorem}[Valwet van Galilei]\label{thm:g12:projectile-motion:galileo}
\index{valwet van Galilei}
In \omterm{def:g12:projectile-motion:freefall}{vrije val} delen alle lichamen dezelfde versnelling, wat hun massa ook is:
\[
\vect a = \vect g, \qquad g = \qty{9.81}{m/s^2},
\]
recht omlaag gericht: de veldsterkte \qty{9.81}{N/kg} van
\cref{ch:g10:universal-gravitation} in haar andere kostuum,
$\unit{N/kg} = \unit{m/s^2}$.
\end{theorem}

\begin{proof}
De tweede wet van Newton (\cref{ch:g12:newtons-laws}) met het \omterm{def:g10:universal-gravitation:weight}{gewicht} als
enige \omterm{def:g10:inertia:force}{kracht}: $m\vect a = m\vect g$, en de massa valt weg. Hamer en veer
verschillen alleen door de lucht.
\end{proof}

\begin{proposition}[Verticale val uit rust]\label{prop:g12:projectile-motion:vertical}
Losgelaten uit rust, met de gevallen diepte $h$ omlaag geteld vanaf het
loslaatpunt, voldoet een lichaam in \omterm{def:g12:projectile-motion:freefall}{vrije val} aan
\[
a = g, \qquad v = g t, \qquad h = \tfrac12\, g t^2 .
\]
\end{proposition}

\begin{proof}
$v(t)$ is de primitieve van de constante $a = g$ die nul is in $t = 0$
(\cref{ch:g12:kinematics-2d}); $h(t)$ die van $v = gt$, nul in $t = 0$.
\end{proof}

\begin{example}[Ordegrootten]\label{ex:g12:projectile-motion:orders}
Na $1$, $2$, $3$ seconden \omterm{def:g12:projectile-motion:freefall}{vrije val}: $v = 9.8$, $19.6$, \qty{29.4}{m/s}
($35$, $71$, \qty{106}{km/h}) en $h = 4.9$, $19.6$, \qty{44.1}{m}. Van een
dak op \qty{30}{m}: $t = \sqrt{2h/g} \approx \qty{2.5}{s}$, aankomst met
\qty{24}{m/s} $\approx \qty{87}{km/h}$. De tijden groeien als $\sqrt h$ ---
en na een paar seconden weigert de lucht nog langer verwaarloosbaar te
blijven (laatste paragraaf).
\end{example}

\begin{example}[Verticale worp]\label{ex:g12:projectile-motion:throw}
Recht omhoog gegooid met $v_0$, as $Oy$ omhoog vanaf de hand: dezelfde twee
primitieven geven $v = v_0 - gt$ en $y = v_0 t - \tfrac12 gt^2$. Het toppunt,
waar $v = 0$, valt op $t = v_0/g$, op hoogte $h = v_0^2/2g$ --- het
energieantwoord $v_0 = \sqrt{2gh}$ van \cref{ch:g11:mechanical-energy}
achterstevoren gelezen; naar \emph{beneden} gegooid wisselt $v_0$ van teken.
Voor \qty{15}{m/s}: toppunt \qty{11.5}{m} op \qty{1.53}{s}, terug op
\qty{3.06}{s}, met \qty{15}{m/s}.
\end{example}

\section{De worp: één wet, twee primitieven}

\begin{definition}[Projectiel]\label{def:g12:projectile-motion:projectile}
Een \emph{projectiel}\index{projectiel} is een lichaam dat met een snelheid
$\vect v_0$ wordt weggeworpen en daarna in \omterm{def:g12:projectile-motion:freefall}{vrije val} wordt gelaten. Assen in
het werppunt: $Ox$ horizontaal in het verticale vlak van $\vect v_0$, $Oy$
verticaal omhoog; de \emph{werphoek}\index{werphoek} $\alpha$ van $\vect v_0$
boven de horizontaal legt haar coördinaten
$(v_0\cos\alpha,\; v_0\sin\alpha)$ vast.
\end{definition}

\begin{theorem}[Bewegingsvergelijkingen]\label{thm:g12:projectile-motion:equations}
Een \omterm{def:g12:projectile-motion:projectile}{projectiel} dat vanuit de oorsprong wordt weggeworpen met snelheid $v_0$
en hoek $\alpha$ heeft op elk ogenblik $\vect a = (0,\,-g)$, dus
\[
\vect v(t) = \bigl(v_0\cos\alpha,\; v_0\sin\alpha - g t\bigr),
\qquad
\begin{cases}
x(t) = v_0\cos\alpha\; t,\\[2pt]
y(t) = v_0\sin\alpha\; t - \tfrac12\, g t^2 .
\end{cases}
\]
\end{theorem}

\begin{proof}
De tweede wet van Newton geeft $\vect a = \vect g = (0,-g)$. Primitieven, as
per as: $\vect v(t)$ uit $\vect a$ met beginwaarde
$(v_0\cos\alpha,\, v_0\sin\alpha)$, daarna de plaats uit $\vect v$, vanaf
$(0,0)$.
\end{proof}

\begin{proposition}[Onafhankelijkheid van de twee bewegingen]\label{prop:g12:projectile-motion:independence}
De horizontale beweging is eenparig --- $v_x$ verandert nooit --- en de
verticale beweging is precies de verticale worp van de vorige paragraaf:
geen van beide vergelijkingen noemt de coördinaat van de andere. In het
bijzonder blijven een horizontaal afgeschoten bal en een op hetzelfde
ogenblik van dezelfde hoogte losgelaten bal op gelijke hoogte --- ze landen
samen.
\end{proposition}

\begin{proof}
In \cref{thm:g12:projectile-motion:equations} bevat $x(t)$ alleen
$v_0\cos\alpha$ en $y(t)$ alleen $v_0\sin\alpha$ en $g$; beide ballen hebben
$v_y(0) = 0$, dus voldoen beide aan $h = \tfrac12 gt^2$.
\end{proof}

\begin{remark}[De vallende aap]\label{rem:g12:projectile-motion:monkey}
Een verzorger richt een verdovingspijltje \emph{recht op} een aap, die bij
het schot zijn tak loslaat. Slechte zet: pijltje en aap zakken elk
$\tfrac12 gt^2$ onder hun plaats zonder zwaartekracht --- de vizierlijn, de
tak --- en dus ontmoeten ze elkaar precies, wat de snelheid van het pijltje
ook is (\cref{exo:g12:projectile-motion:15}).
\end{remark}

\section{De parabool, helemaal gelezen}

\begin{theorem}[Baan]\label{thm:g12:projectile-motion:trajectory}
Als je $t$ tussen $x(t)$ en $y(t)$ elimineert: de \omterm{def:g10:relative-motion:trajectory}{baan} van een \omterm{def:g12:projectile-motion:projectile}{projectiel}
($0 \leq \alpha < \ang{90}$) is de naar beneden geopende parabool
\[
y = x\tan\alpha \;-\; \frac{g}{2 v_0^2 \cos^2\alpha}\, x^2 .
\]
\end{theorem}

\begin{proof}
$t = x/(v_0\cos\alpha)$, ingevuld in $y(t)$ van
\cref{thm:g12:projectile-motion:equations}.
\end{proof}

\begin{definition}[Dracht]\label{def:g12:projectile-motion:range}
Voor een \omterm{def:g12:projectile-motion:projectile}{projectiel} boven vlakke grond is de \emph{dracht}\index{dracht} $R$
de horizontale afstand die het aflegt tot het weer op werphoogte komt; de
\emph{vluchttijd}\index{vluchttijd} $t_f$ is de duur van die reis.
\end{definition}

\begin{proposition}[Toppunt, vluchttijd, dracht]\label{prop:g12:projectile-motion:range}
Boven vlakke grond, met $0 < \alpha < \ang{90}$:
\[
t_f = \frac{2 v_0 \sin\alpha}{g}, \qquad
h_{\text{top}} = \frac{(v_0\sin\alpha)^2}{2g}
\ \ \text{bereikt op } t_f/2, \qquad
R = \frac{v_0^2 \sin(2\alpha)}{g}.
\]
De \omterm{def:g12:projectile-motion:range}{dracht} is het grootst bij $\alpha = \ang{45}$, met $R = v_0^2/g$;
complementaire hoeken $\alpha$ en $\ang{90} - \alpha$ geven dezelfde \omterm{def:g12:projectile-motion:range}{dracht}.
\end{proposition}

\begin{proof}
$y = 0$ geeft $t\,(v_0\sin\alpha - \tfrac12 gt) = 0$: de worp, en $t_f$. Het
toppunt, $v_y = 0$, ligt op $t = v_0\sin\alpha/g = t_f/2$; vul dat in
$y(t)$ in. Verder is $R = x(t_f) = 2v_0^2\sin\alpha\cos\alpha/g =
v_0^2\sin(2\alpha)/g$. De sinus is maximaal bij $2\alpha = \ang{90}$, en
$\sin\bigl(2(\ang{90}-\alpha)\bigr) = \sin(\ang{180} - 2\alpha) =
\sin(2\alpha)$.
\end{proof}

\begin{example}[Doeltrap]\label{ex:g12:projectile-motion:kick}
Een doelman trapt met $v_0 = \qty{25}{m/s}$, $\alpha = \ang{45}$:
$R = 25^2/9.81 \approx \qty{64}{m}$; echte uittrappen landen eerder rond
\qty{55}{m} --- het aandeel van de lucht.
\end{example}

\begin{method}[De boekhouding van een worp]\label{met:g12:projectile-motion:method}
\begin{enumerate}
  \item Assen in het werppunt, $Ox$ horizontaal, $Oy$ omhoog; coördinaten
        van $\vect v_0$.
  \item Ga de \omterm{def:g12:projectile-motion:freefall}{vrije val} na (alleen het \omterm{def:g10:universal-gravitation:weight}{gewicht}): dan is $\vect a = (0,-g)$;
        twee keer primitiveren, elke keer met de beginvoorwaarden.
  \item Vraag naar een \emph{plaats}: elimineer $t$ (de \omterm{def:g10:relative-motion:trajectory}{baan}). Naar een
        \emph{ogenblik}: houd $t$.
  \item Woordenboek: toppunt $\Leftrightarrow v_y = 0$; werphoogte
        $\Leftrightarrow y = 0$; muur op $d \Leftrightarrow x = d$; vloer
        $\ell$ lager $\Leftrightarrow y = -\ell$.
  \item Controle: de vlucht is symmetrisch om het toppunt; $R \leq v_0^2/g$;
        de lucht maakt het altijd alleen maar korter.
\end{enumerate}
\end{method}

\section{Wat de lucht verandert}

\begin{definition}[Grenssnelheid]\label{def:g12:projectile-motion:terminal}
De luchtweerstand werkt de snelheid tegen en groeit met de snelheid. Een
vallend lichaam versnelt dus alleen tot de luchtweerstand het \omterm{def:g10:universal-gravitation:weight}{gewicht} opheft;
vanaf dan is $\vect a = \vect 0$: het valt met een constante
\emph{grenssnelheid}\index{grenssnelheid}.
\end{definition}

\begin{example}[Grenssnelheden in ordegrootte]\label{ex:g12:projectile-motion:airorders}
Een parachutespringer, plat uitgespreid: ongeveer \qty{55}{m/s}
(\qty{200}{km/h}). Een regendruppel: ongeveer \qty{9}{m/s} --- in vacuüm zou
\qty{2000}{m} val hem afleveren met $\sqrt{2gh} \approx \qty{2.0e2}{m/s}$.
Een veer: ongeveer \qty{1}{m/s}, binnen enkele centimeters bereikt --- het
geheim van haar traagheid: ze brengt haar hele val \emph{op} de
\omterm{def:g12:projectile-motion:terminal}{grenssnelheid} door; de hamer komt nooit in de buurt van de zijne.
\end{example}

\begin{remark}[Echte ballistiek]\label{rem:g12:projectile-motion:drag}
De luchtweerstand duwt de hele vlucht lang tegen de snelheid in: de \omterm{def:g12:projectile-motion:range}{dracht} wordt
korter, het toppunt lager, en de symmetrie breekt --- de daling steiler dan
de klim, verre schoten zichtbaar korter dan de parabool. De eerlijke,
kwantitatieve behandeling is een differentiaalvergelijking die in het volume
van bachelorjaar 1 wordt opgelost; dit jaar is de parabool het juiste model
voor dichte, matig snelle \omterm{def:g12:projectile-motion:projectile}{projectielen} --- een basketbal met \qty{10}{m/s},
niet een golfbal met \qty{70}{m/s}.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:1}
Een steen die van een brug wordt losgelaten raakt \qty{2.0}{s} later het
water: hoe hoog ligt de brug, en met welke snelheid komt hij aan, in
\unit{m/s} en \unit{km/h}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:2}
Op de Maan liet een astronaut een hamer en een veer samen los: ze landden
samen. Waarom --- en waarom verliest de veer op Aarde zo zwaar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:3}
Bereken, zonder lucht, valtijd en aankomstsnelheid vanaf \qty{1.0}{m},
\qty{10}{m} en \qty{100}{m}. Hoe groeien de tijden met de hoogte?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:4}
Een bal wordt recht omhoog gegooid met \qty{15}{m/s}: hoogte en tijdstip van
het toppunt, totale tijd tot hij terug in de hand ligt, en terugkeersnelheid?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:projectile-motion:5}
Een knikker rolt met \qty{2.5}{m/s} van een tafel van \qty{0.80}{m} hoog.
Bereken de valtijd en de landingsafstand tot de rand. Hangt de valtijd van de
snelheid van de knikker af?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:6}
Uit de figuur met de vijf hoeken uit de cursus: welke paren \omterm{def:g12:projectile-motion:projectile}{werphoeken} delen
hun \omterm{def:g12:projectile-motion:range}{dracht}, en welke hoek wint? Voor een \omterm{def:g12:projectile-motion:range}{dracht} onder de maximale: welke van
de twee mogelijke hoeken vliegt langer? Onderbouw met $t_f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:7}
Van een balkon op \qty{20}{m} wordt een bal recht naar \emph{beneden} gegooid
met \qty{5.0}{m/s}. Bereken de tijd tot de grond en de aankomstsnelheid;
controleer die laatste met een energiebalans
(\cref{ch:g11:mechanical-energy}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:8}
Een \omterm{def:g12:projectile-motion:projectile}{projectiel} verlaat de vlakke grond met \qty{40}{m/s},
$\alpha = \ang{30}$: \omterm{def:g12:projectile-motion:range}{vluchttijd}, hoogte van het toppunt en \omterm{def:g12:projectile-motion:range}{dracht}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:9}
Modelleer een verspringer als een \omterm{def:g12:projectile-motion:projectile}{projectiel}: afzet met \qty{10.5}{m/s},
$\alpha = \ang{20}$. Bereken de \omterm{def:g12:projectile-motion:range}{dracht}; kampioenen springen bijna \qty{9}{m}
--- noem twee ingrediënten die het puntmodel weglaat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:10}
Op \qty{1.2}{m} boven vlakke grond wordt één kogel losgelaten en een
identieke horizontaal afgeschoten met \qty{300}{m/s}. Welke landt het eerst?
Bereken de valtijd en de landingsafstand van de afgeschoten kogel. Welk
beginsel wordt hier getoetst?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:projectile-motion:11}
Toon algebraïsch aan dat $\alpha$ en $\ang{90} - \alpha$ dezelfde \omterm{def:g12:projectile-motion:range}{dracht}
hebben, en bepaal daarna de twee hoeken die een \omterm{def:g12:projectile-motion:projectile}{projectiel} van \qty{25}{m/s}
op \qty{50}{m} laten landen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:12}
Water verlaat een tuinslang met \qty{8.0}{m/s}: grootste horizontale bereik?
grootste hoogte? de twee standen van de spuitmond die een bloembed op
\qty{5.0}{m} water geven?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:13}
Een bal wordt vanaf de grond getrapt met \qty{18}{m/s}, $\alpha = \ang{35}$,
naar een haag van \qty{4.0}{m} op \qty{15}{m} afstand. Beslis met de
baanvergelijking of hij eroverheen gaat, en met hoeveel.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:14}
Een parachutespringer van \qty{80}{kg} valt plat met de \omterm{def:g12:projectile-motion:terminal}{grenssnelheid}

\begin{exercise}[$\star\star\star$]\label{exo:g12:projectile-motion:15}

\section{Probleem: De basketbal in de slotseconde}

\begin{problem}[{Weekendopgave --- de basketbal in de slotseconde: een
vrije worp pijl voor pijl ontleed, vingertoppen met centimeters gemist, een
worp over het hele veld afgewogen tegen wat een arm kan geven, een lob over
het bord --- en een oordeel over het meest theatrale schot in de sport}]
\label{pb:g12:projectile-motion:1}
De ring van een basketbaldoel is een horizontale cirkel op \qty{3.05}{m}
boven de vloer; Nora laat elk schot los op \qty{2.00}{m}, dus met assen in
het loslaatpunt ($Ox$ horizontaal naar het doel, $Oy$ omhoog) ligt het
middelpunt van de ring \qty{1.05}{m} boven de oorsprong. De lucht wordt
verwaarloosd, behalve waar het uitdrukkelijk staat.

\medskip\noindent\textbf{Deel I --- De vrije worp.} Nora schiet van
$d = \qty{4.20}{m}$ (horizontaal) met \omterm{def:g12:projectile-motion:projectile}{werphoek} $\alpha = \ang{50}$.
\begin{enumerate}
  \item Welke \omterm{def:g10:inertia:force}{krachten} werken er zodra de bal los is? Geef $\vect a$, met de
        wet die je gebruikt.
  \item Leid de coördinaten van $\vect v(t)$ af.
  \item Leid $x(t)$ en $y(t)$ af.
  \item Elimineer $t$ om de \omterm{def:g10:relative-motion:trajectory}{baan} $y(x)$ te krijgen.
  \item De bal moet door het middelpunt van de ring, $y(d) = \qty{1.05}{m}$:
        los op naar $v_0$, in \unit{m/s} en \unit{km/h}.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Over de vingertoppen, in de ring.}
\begin{enumerate}[resume]
  \item Hoe lang doet de bal over de weg naar de ring?
  \item Bepaal tijdstip en hoogte van het toppunt; ga na dat de bal bij de
        ring al daalt.
  \item Een verdedigster \qty{0.90}{m} voor Nora reikt met haar
        vingertoppen tot \qty{2.60}{m}. Met hoeveel gaat de bal daaroverheen?
  \item Bereken $v_x$ en $v_y$ bij de ring, de snelheid daar, en de hoek van
        de \omterm{def:g12:kinematics-2d:velocity}{snelheidsvector} onder de horizontaal.
  \item Waarom moet een bal \emph{dalend} aankomen --- en waarom steil?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De worp over het hele \omterm{def:g11:electric-gravitational-fields:field}{veld}.} De zoemer
staat op klinken en Nora staat op \qty{26.0}{m} van de verste ring.
\begin{enumerate}[resume]
  \item De stijging van \qty{1.05}{m} is klein tegenover \qty{26.0}{m}:
        verantwoord een model met vlakke grond, en leid de \omterm{def:g12:projectile-motion:range}{vluchttijd} af uit
        $y(t_f) = 0$.
  \item Leid daaruit $R = v_0^2\sin(2\alpha)/g$ af en de hoek van de
        grootste \omterm{def:g12:projectile-motion:range}{dracht}.
  \item Bereken de $v_0$ die nodig is om \qty{26.0}{m} te halen bij
        $\alpha = \ang{45}$.
  \item Een sterke speelster kan een basketbal met ongeveer \qty{17.5}{m/s}
        wegslingeren. Wat is haar maximale \omterm{def:g12:projectile-motion:range}{dracht} --- is het schot menselijk
        haalbaar?
  \item Bepaal bij \qty{17.5}{m/s} de \emph{twee} hoeken die de bal op
        \qty{26.0}{m} laten landen, en hun \omterm{def:g12:projectile-motion:range}{vluchttijden}: met \qty{2.0}{s} op
        de klok, welke van de twee haalt het vóór de zoemer?
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- De lob over het bord.} Van achter de
achterlijn lobt Nora: loslaten op \qty{6.0}{m} van het middelpunt van de
ring, $\alpha = \ang{75}$; het vlak van het bord kruist haar schot
\qty{0.45}{m} vóór het middelpunt van de ring, met de bovenrand op
\qty{3.95}{m} boven de vloer.
\begin{enumerate}[resume]
  \item Bepaal de $v_0$ die de bal door het middelpunt van de ring laat
        zakken.
  \item Bereken de hoogte van de bal in het vlak van het bord
        ($x = \qty{5.55}{m}$): gaat hij over de bovenrand, en met hoeveel?
  \item Bepaal de hoogte van het toppunt en de \omterm{def:g12:projectile-motion:range}{vluchttijd} --- waarom is het
        publiek er dol op?
  \item Wat verandert de luchtweerstand kwalitatief aan deze drie schoten, en
        welk schot bedreigt ze het meest?
  \item \emph{Het oordeel over de slotworp}, in twee zinnen: vergelijk de
        snelheid die de worp over het hele \omterm{def:g11:electric-gravitational-fields:field}{veld} vraagt met wat een arm
        levert, geef de marge, en beslis --- mogelijk of mythe?
\end{enumerate}
\end{problem}
