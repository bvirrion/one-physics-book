\chapter{Radioactief verval}\label{ch:g12:radioactive-decay}

In de kelder van een museum klikt een geigerteller boven een splinter hout
uit de kist van een farao. Elke klik is wetteloos: niets kondigde hem aan,
niets voorspelt de volgende. En toch drukt het laboratorium vrijdag een datum
af die tot op een eeuw klopt. Dit hoofdstuk lost de tegenspraak op --- hoe
gebeurtenissen die stuk voor stuk volstrekt willekeurig zijn, bij biljoenen
tegelijk optellen tot de gelijkmatigste klokken die we bezitten.

\section{Wetteloos per stuk, uurwerk per mol}

Vorig jaar (\cref{ch:g11:nucleus-radioactivity}) maakten we kennis met de
onstabiele kernen, maakten we hun $\alpha$- en $\beta$-vergelijkingen
kloppend en zagen we per \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd} $T_{1/2}$ de helft van elk groot
monster verdwijnen. Wat we niet konden zeggen was \emph{wanneer tussen de
treden} --- en evenmin waarom een willekeurig proces überhaupt een dienstregeling
aanhoudt. Beide vragen hebben één antwoord.

\begin{definition}[Vervalconstante]\label{def:g12:radioactive-decay:lambda}
Bij elke onstabiele \omterm{def:g11:nucleus-radioactivity:notation}{nuclide} hoort een constante $\lambda$, haar
\emph{vervalconstante}\index{vervalconstante} (\omterm{def:g10:orders-of-magnitude:unit}{eenheid} \unit{s^{-1}}), zodanig
dat elke kern gedurende elk kort interval $\dd t$ een kans
$p = \lambda\,\dd t$ heeft om te vervallen --- dezelfde voor elke kern van de
soort, hoe oud die ook is, en onberoerd door temperatuur, \omterm{def:g10:pressure:pressure}{druk} of chemie.
\end{definition}

\begin{remark}[Wetteloos per stuk, exact in de massa]\label{rem:g12:radioactive-decay:crowds}
Een kern die tienduizend jaar heeft gewacht, vervalt deze seconde met precies
dezelfde kans als een die vanochtend is gemaakt: geen slijtage, geen
waarschuwing --- welke er als volgende gaat, is echt onvoorspelbaar. Met
massa's ligt het anders. Werp $100$ munten op: verwacht $50$ keer kop, maar
schrik niet van $43$ --- fluctuaties van de orde $\sqrt n$, hier $10\%$. Werp
er $10^{20}$: de relatieve fluctuatie $1/\sqrt n$ is $10^{-10}$, en de uitkomst
staat tot op tien decimalen vast. Eén gram materie bevat zo'n $10^{22}$
kernen: wetteloosheid middelt uit tot uurwerk, tot een exacte wet voor $N(t)$,
het aantal kernen dat op tijdstip $t$ nog intact is.
\end{remark}

\section{De vervalwet}

\begin{proposition}[De evolutievergelijking]\label{prop:g12:radioactive-decay:ode}
In een groot monster van $N$ kernen verandert het aantal gedurende $\dd t$ met
\[
\dd N = -\lambda N\,\dd t, \qquad\text{oftewel}\qquad \frac{\dd N}{\dd t} = -\lambda N :
\]
de bevolking krimpt met een tempo dat evenredig is met haarzelf.
\end{proposition}

\begin{proof}
Elk van de $N$ kernen vervalt gedurende $\dd t$ met kans $\lambda\,\dd t$:
verwacht aantal vervallen $N\lambda\,\dd t$, en voor grote $N$ blijft het
werkelijke aantal bij het verwachte
(\cref{rem:g12:radioactive-decay:crowds}); elk verval haalt één kern weg.
\end{proof}

\begin{theorem}[Wet van het radioactieve verval]\label{thm:g12:radioactive-decay:law}
Een monster dat op $t = 0$ nog $N_0$ kernen bevat, bevat er op tijdstip $t$
\[
N(t) = N_0 \, e^{-\lambda t}.
\]
\end{theorem}

\begin{proof}
Controleer door invullen:
$\dd N/\dd t = N_0(-\lambda)\,e^{-\lambda t} = -\lambda N(t)$, en
$N(0) = N_0 e^0 = N_0$: vergelijking en start allebei goed. (Dat geen
\emph{andere} functie past, wordt met integralen in het universitaire volume
bewezen.)
\end{proof}

\begin{definition}[Exponentieel verval]\label{def:g12:radioactive-decay:exponential}
Een grootheid die $N_0\,e^{-\lambda t}$ volgt, ondergaat \emph{exponentieel
verval}\index{exponentieel verval}: in gelijke tijden gaan gelijke
\emph{fracties} verloren. Het is precies de ontladende \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} van
\cref{ch:g12:rc-rl-circuits}, met $\lambda$ in de rol van $1/(RC)$: één
differentiaalvergelijking, één keer geleerd, stuurt schakelingen en kernen
allebei.
\end{definition}

\begin{proposition}[Halveringstijd]\label{prop:g12:radioactive-decay:halflife}
De \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd} (\cref{def:g11:nucleus-radioactivity:halflife}) van een
\omterm{def:g11:nucleus-radioactivity:notation}{nuclide} ligt vast door haar \omterm{def:g12:radioactive-decay:lambda}{vervalconstante}: $T_{1/2} = \ln 2/\lambda$, en na
$n$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} is $N = N_0/2^{\,n}$, of $n$ nu geheel is of niet.
\end{proposition}

\begin{proof}
$N(T_{1/2}) = N_0/2$ vraagt $e^{-\lambda T_{1/2}} = \tfrac12$, oftewel
$\lambda T_{1/2} = \ln 2$; en $N(n\,T_{1/2}) = N_0\,e^{-n\ln 2} =
N_0/2^{\,n}$ voor elke $n$: de exponentiële functie rijgt de trap aaneen en
vult in wat tussen de treden ligt.
\end{proof}

\section{Activiteit: de klikken tellen}

\begin{proposition}[Activiteit]\label{prop:g12:radioactive-decay:activity}
De \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van een monster --- het aantal vervallen per seconde, in
\omterm{def:g11:nucleus-radioactivity:activity}{becquerel} (\cref{def:g11:nucleus-radioactivity:activity}) --- is
\[
\mathcal A = \lambda N, \qquad\text{dus}\qquad \mathcal A(t) = \mathcal A_0 \, e^{-\lambda t}:
\]
evenredig met de overlevende bevolking, en even exponentieel.
\end{proposition}

\begin{proof}
$N$ kernen, elk met kans $\lambda$ per seconde: $\lambda N$ vervallen per
seconde; \cref{thm:g12:radioactive-decay:law} met $\lambda$ vermenigvuldigen
geeft $\mathcal A(t) = \mathcal A_0 e^{-\lambda t}$.
\end{proof}

\begin{example}[Reusachtige $N$, minuscule $\lambda$]\label{ex:g12:radioactive-decay:body}
Een banaan tikt op ongeveer \qty{15}{Bq}, een menselijk lichaam op ongeveer
\qty{8000}{Bq}, een kubieke \omterm{def:g10:orders-of-magnitude:unit}{meter} graniet op zo'n \qty{e6}{Bq}
(\cref{ch:g11:nucleus-radioactivity}). Achter die bescheiden klikken zitten
astronomische bevolkingen: het kalium-40 van het lichaam draagt
$\mathcal A \approx \qty{4.4e3}{Bq}$ bij met
$\lambda = \qty{1.8e-17}{s^{-1}}$: $N = \mathcal A/\lambda \approx
\num{2.5e20}$ kernen, waarvan er per seconde amper een paar duizend afgaan.
\end{example}

\begin{method}[Een halveringstijd meten]\label{met:g12:radioactive-decay:semilog}
\begin{enumerate}
  \item Tel met een geigerteller de vervallen over opeenvolgende gelijke
        intervallen: zo bemonster je $\mathcal A$ op opeenvolgende tijdstippen.
  \item Zet $\ln \mathcal A$ uit tegen $t$: omdat
        $\ln \mathcal A = \ln \mathcal A_0 - \lambda t$, verschijnt
        \omterm{def:g12:radioactive-decay:exponential}{exponentieel verval} als een \emph{rechte lijn} --- kromme meetpunten
        betekenen een andere wet, of een mengsel van \omterm{def:g11:nucleus-radioactivity:notation}{nucliden}.
  \item Lees $\lambda$ af als min de helling; $T_{1/2} = \ln 2/\lambda$.
\end{enumerate}
\end{method}

\begin{remark}[Becquerel is geen schade]\label{rem:g12:radioactive-decay:dose}
De \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} telt vervallen, geen schade. Schade wordt bijgehouden met de
\omterm{rem:g11:nucleus-radioactivity:dose}{dosis} in \omterm{rem:g11:nucleus-radioactivity:dose}{sievert} (\cref{ch:g11:nucleus-radioactivity}), die de per \omterm{def:g10:orders-of-magnitude:unit}{kilogram}
weefsel afgezette \omterm{def:g10:energy-conservation:energy}{energie} weegt met de vernietigingskracht van elke straling:
een technetiumscan van \qty{5e8}{Bq} is routine, veel minder \omterm{def:g11:nucleus-radioactivity:activity}{becquerel} van een
ingeademde $\alpha$-straler is dat niet; \unit{Bq} in \unit{Sv} omrekenen is
universitaire stof.
\end{remark}

\section{Dateren: de tijd van een verhouding aflezen}

De vervalwet achterstevoren gelezen is een klok: meet de overlevende fractie,
en de exponentiële functie noemt de verstreken tijd.

\begin{method}[Koolstofdatering]\label{met:g12:radioactive-decay:carbon}
\begin{enumerate}
  \item \emph{Levend:} levend weefsel wisselt koolstof met de atmosfeer uit en
        draagt haar aandeel koolstof-14 --- één \omterm{def:g11:fundamental-interactions:ladder}{atoom} op $10^{12}$, goed voor
        $\mathcal A_0 = 13.6$ vervallen per minuut per gram koolstof.
  \item \emph{De dood start de klok:} de opname stopt, en het aandeel vervalt
        met $T_{1/2} = \num{5730}$ jaar.
  \item Meet de \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} $\mathcal A$ van het monster per gram koolstof;
        $\mathcal A = \mathcal A_0 e^{-\lambda t}$ keert om tot
        $t = \frac{1}{\lambda}\ln(\mathcal A_0/\mathcal A)
        = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}
\end{method}

\begin{example}[Een haardvuur in een grot]\label{ex:g12:radioactive-decay:hearth}
Houtskool uit een begraven haard geeft $5.0$ vervallen per minuut per gram:
$\mathcal A_0/\mathcal A = 13.6/5.0 = 2.72$, dus
$t = \num{5730} \times \ln(2.72)/\ln 2 \approx \num{8.3e3}$ jaar. Het vuur
doofde achtduizend jaar geleden --- het hout zelf is het archief.
\end{example}

\begin{example}[Klokken voor gesteente]\label{ex:g12:radioactive-decay:rocks}
Koolstof-14 zwijgt voorbij zo'n \num{50000} jaar; de geologie houdt tragere
klokken bij. \emph{Kalium--argon}: kalium-40 ($T_{1/2} = \num{1.25e9}$ jaar)
vervalt tot argon, een gas dat uit gesmolten lava ontsnapt maar gevangen raakt
zodra het gesteente kristalliseert --- het stollen zet de klok op nul.
\emph{Uranium--lood}: uranium-238 leidt een keten van vervallen die bij het
stabiele lood-206 eindigt (\cref{ch:g11:nucleus-radioactivity}), en de
verhouding lood tot uranium in een kristal dateert het. Beide klokken zijn het
eens over de oudste mineralen ($\approx \num{4.4e9}$ jaar) en, via
meteorieten, over de ouderdom van het zonnestelsel.
\end{example}

\begin{remark}[Ketens, en het gas in de kelder]\label{rem:g12:radioactive-decay:radon}
Tussen uranium en lood loopt de keten via radium en dan radon-222, een
radioactief \emph{gas} ($T_{1/2} = 3.8$ dagen). Geboren in graniet en bodem
sijpelt het in kelders en hoopt het zich op; zijn $\alpha$-stralende dochters
vervallen in de longen --- voor de meeste mensen de grootste natuurlijke \omterm{rem:g11:nucleus-radioactivity:dose}{dosis},
en de goedkoopste om te verlagen: zet het kelderraam open.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:1}
Jodium-131 heeft $\lambda = \qty{1.0e-6}{s^{-1}}$. Bereken zijn
\omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd}, in seconden en daarna in dagen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:2}
Radon-222 heeft $T_{1/2} = 3.8$ dagen. Bereken $\lambda$ in \unit{s^{-1}}, en
daarna de \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van een monster van \num{1.0e10} kernen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:3}
Ga door invullen na dat $N(t) = N_0\,e^{-\lambda t}$ aan
$\dd N/\dd t = -\lambda N$ en $N(0) = N_0$ voldoet; bereken daarna

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:4}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:radioactive-decay:5}

Voor één enkele kern is de kans om $n$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} te overleven
$1/2^{\,n}$. Bereken de kans dat (a) één gegeven kern $10$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}
overleeft; (b) alle $8$ bewaakte kernen één \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijd} overleven. (c) Van
\num{1000} kernen: hoeveel worden er na $3$ \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} verwacht --- en
kan de wet zeggen \emph{welke}? Trek een conclusie over wat de wet voorspelt.
\end{exercise}

\section{Probleem: De ouderdom der dingen}

\begin{problem}[{Weekendopgave --- de ouderdom der dingen: een
museumlaboratorium ijkt de koolstofklok, dateert een beeld en een papyrus,
leert waarom geen enkele klok voorbij haar wijzerplaat leest, en duwt een
vulkanisch gesteente een miljard jaar terug}]
\label{pb:g12:radioactive-decay:1}
Een week in het dateringslaboratorium van een museum. Gegevens: levend weefsel
vertoont $\mathcal A_0 = 13.6$ vervallen van koolstof-14 per minuut per gram
koolstof; $T_{1/2} = \num{5730}$ jaar (koolstof-14) en \num{1.25e9} jaar
(kalium-40); $1$ jaar $= \qty{3.156e7}{s}$;
$N_A = \qty{6.02e23}{mol^{-1}}$; molaire massa van koolstof \qty{12}{g/mol};
de tellers onderscheiden ongeveer $0.2$ vervallen per minuut per gram.

\medskip
\noindent\textbf{Deel I --- Maandag: de klok ijken.}
\begin{enumerate}
  \item Waarom is het aandeel koolstof-14 in levend weefsel constant, en wat
        verandert er bij de dood?
  \item Bereken $\lambda$ voor koolstof-14 in \unit{s^{-1}}.
  \item Ga door invullen na dat $N_0 e^{-\lambda t}$ de vergelijking
        $\dd N/\dd t = -\lambda N$ oplost.
  \item Bereken uit $\mathcal A_0$ het aantal koolstof-14-atomen per gram
        koolstof in levend weefsel.
  \item Hoeveel koolstofatomen bevat één gram? Leid het aandeel koolstof-14
        af --- vergelijk met ``één op $10^{12}$''.
  \item Toon aan dat een gemeten \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} $\mathcal A$ een monster dateert
        op $t = T_{1/2}\,\ln(\mathcal A_0/\mathcal A)/\ln 2$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Dinsdag: het beeld en de papyrus.}
\begin{enumerate}[resume]
  \item Een houten beeld geeft $9.1$ vervallen per minuut per gram. Dateer het
        hout.
  \item Welke gebeurtenis markeert die datum precies --- het snijwerk of iets
        anders? Welke voorzichtigheid volgt daaruit?
  \item Een Egyptische papyrus geeft $10.4$ vervallen per minuut per gram.
        Dateer haar. Is een schrijver van tweeëntwintig eeuwen geleden
        aannemelijk?
  \item De \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} wordt gemeten op $\pm 0.2$ vervallen per minuut. Schat de
        onzekerheid op de ouderdom van de papyrus met
        $\Delta t \approx \Delta\mathcal A/(\lambda \mathcal A)$.
  \item Het ``oude'' perkament van een handelaar geeft $13.4$ vervallen per
        minuut per gram. Wat is je oordeel?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Donderdag: de grenzen van de klok.}
\begin{enumerate}[resume]
  \item Na hoeveel \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} zakt de \omterm{def:g11:nucleus-radioactivity:activity}{activiteit} van een monster onder de
        $0.2$ vervallen per minuut per gram van de tellers? Welke ouderdom is
        dat?
  \item De beste laboratoria halen ongeveer tien \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}. Wat is de
        praktische horizon van koolstofdatering, in jaren?
  \item Een dinosaurusbot is ongeveer \num{6.6e7} jaar oud: hoeveel
        \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden} van koolstof-14 zijn dat? Na hoeveel \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}
        is er, volgens vraag 4, zelfs geen \emph{één} \omterm{def:g11:fundamental-interactions:ladder}{atoom} koolstof-14 van de
        gram meer over? Trek een conclusie over ``dinosauriërs met koolstof
        dateren''.
  \item Welk soort \omterm{def:g11:nucleus-radioactivity:notation}{nuclide} zou het gesteente van de dinosaurus dan wel kunnen
        dateren?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Vrijdag: het vulkanische gesteente.}
\begin{enumerate}[resume]
  \item Argon is een gas, kalium een trouwe bewoner van een vaste stof: leg uit
        waarom het stollen van lava de kalium--argonklok op nul zet.
  \item Toon, aangenomen dat elke vervallen kalium-40-kern één gevangen
        argonatoom achterlaat, aan dat
        $N_{\mathrm{Ar}}/N_{\mathrm K} = e^{\lambda t} - 1$.
  \item Het gesteente onder het fossiel van het museum vertoont
        $N_{\mathrm{Ar}}/N_{\mathrm K} = 0.60$. Dateer het gesteente.
  \item Had koolstof-14 dit gesteente kunnen dateren, of kalium--argon het
        beeld? Geef voor elke klok het bruikbare venster in \omterm{def:g11:nucleus-radioactivity:halflife}{halveringstijden}.
  \item Vrijdagrapport, in één zin: de drie ouderdommen van deze week, en de
        ene wet die ze allemaal heeft gelezen.
\end{enumerate}
\end{problem}
