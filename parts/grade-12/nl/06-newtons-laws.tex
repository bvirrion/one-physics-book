\chapter{De wetten van Newton}\label{ch:g12:newtons-laws}

Smokkel een personenweegschaal een lift in. Terwijl de cabine optrekt wijst de naald
drie \omterm{def:g10:orders-of-magnitude:unit}{kilogram} te zwaar; vlak bij de bovenste verdieping wijst ze te licht; daartussen
de nuchtere waarheid --- er is aan boord niets veranderd behalve de beweging. Dit
hoofdstuk formuleert de drie wetten achter die naald en smeedt ze om tot de methode
waarop de hele mechanica draait: kies een systeem, teken de \omterm{def:g10:inertia:force}{krachten}, projecteer
$\sum \vect F = m\vect a$.

\section{Impuls, en de eerste wet exact gemaakt}

\begin{definition}[Impuls]\label{def:g12:newtons-laws:momentum}
De \emph{impuls}\index{impuls} van een lichaam met massa $m$ (\unit{kg}) dat met
snelheid $\vect v$ beweegt, is de vector
\[
\vect p = m \vect v \qquad (\unit{kg.m/s}):
\]
de snelheid, gewogen met de massa; voor een systeem de vectorsom over de delen.
\end{definition}

\begin{definition}[Traagheidsstelsel]\label{def:g12:newtons-laws:inertial}
Een \emph{traagheidsstelsel}\index{traagheidsstelsel} is een stelsel waarin elk
geïsoleerd lichaam --- zonder \omterm{def:g10:inertia:force}{krachten}, of met \omterm{def:g10:inertia:force}{krachten} in \omterm{def:g10:inertia:compensate}{evenwicht} --- rechtlijnig
met constante snelheid beweegt (\cref{ch:g10:inertia}). De grond is er tot in hoge
nauwkeurigheid een, en zo ook elke cabine die er \omterm{def:g12:kinematics-2d:uniform}{eenparig rechtlijnig} overheen
beweegt; een remmende trein of een draaiende draaimolen niet: daarin versnelt losse
bagage zonder dat iemand duwt.
\end{definition}

\begin{theorem}[Eerste wet van Newton]\label{thm:g12:newtons-laws:first}\index{eerste wet van Newton}
In een \omterm{def:g12:newtons-laws:inertial}{traagheidsstelsel} verandert de \omterm{def:g12:newtons-laws:momentum}{impuls} van een geïsoleerd systeem niet:
$\vect p$ is constant, dus $d\vect p/dt = \vect 0$.
\end{theorem}
\admitted

\begin{remark}[Traagheid, twee keer aangescherpt]\label{rem:g12:newtons-laws:sharpened}
Het traagheidsbeginsel van \cref{ch:g10:inertia}, twee keer aangescherpt: het zegt
\emph{waar} het geldt --- het definieert de \omterm{def:g12:newtons-laws:inertial}{traagheidsstelsels} als die waarin het
geldt --- en het bewaakt een vector, $m\vect v$, waarvan de boekhouding aan het einde
van dit hoofdstuk haar vruchten afwerpt.
\end{remark}

\section{De tweede wet van Newton}

\begin{theorem}[Tweede wet van Newton]\label{thm:g12:newtons-laws:second}\index{tweede wet van Newton}
In een \omterm{def:g12:newtons-laws:inertial}{traagheidsstelsel} is de \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} op een lichaam gelijk aan de
verandering per tijdseenheid van zijn \omterm{def:g12:newtons-laws:momentum}{impuls}; bij constante massa geldt, omdat
$\vect p = m\vect v$,
\[
\sum \vect F = \frac{d\vect p}{dt} = m\,\frac{d\vect v}{dt} = m \vect a .
\]
\end{theorem}
\admitted

\begin{remark}[Een natuurwet, en haar eenheid]\label{rem:g12:newtons-laws:nature}
Er is geen bewijs: dit is een natuurwet, drie eeuwen lang aan het experiment
getoetst. Vorig jaar (\cref{ch:g11:forces-and-motion}) gaf haar schaduw,
$\Delta v \propto F\,\Delta t/m$; de afgeleide (\cref{ch:g12:kinematics-2d}) maakt
haar op elk ogenblik exact --- de kleine lettertjes wachten in het volume van
bachelorjaar 1. De \omterm{def:g10:inertia:force}{kracht} krijgt een \omterm{def:g10:orders-of-magnitude:unit}{eenheid}, $\unit{N} = \unit{kg.m/s^2}$; en een
lichaam waarop alleen zijn \omterm{def:g10:universal-gravitation:weight}{gewicht} werkt, voldoet aan $m\vect a = m\vect g$: elke
massa valt met $\vect a = \vect g$, en
$g = \qty{9.81}{N/kg} = \qty{9.81}{m/s^2}$ --- de twee aflezingen zijn één.
\end{remark}

\begin{example}[Gemiddelde kracht, zonder details]\label{ex:g12:newtons-laws:average}
Een auto van \qty{1300}{kg} gaat van stilstand naar \qty{100}{km/h}
(\qty{27.8}{m/s}) in \qty{8.0}{s}: wat er van ogenblik tot ogenblik ook gebeurt, de
gemiddelde \omterm{def:g11:forces-and-motion:netforce}{resulterende kracht} is
$\Delta p/\Delta t = 1300 \times 27.8/8.0 \approx \qty{4.5e3}{N}$ --- de \omterm{def:g12:newtons-laws:momentum}{impuls}
maakt van een krachtenvraag een boekhouding van vóór en na.
\end{example}

\section{De derde wet van Newton}

\begin{theorem}[Derde wet van Newton]\label{thm:g12:newtons-laws:third}\index{derde wet van Newton}
Oefent een lichaam $A$ een \omterm{def:g10:inertia:force}{kracht} $\vect F_{A \to B}$ uit op een lichaam $B$, dan
oefent $B$ op $A$ de \omterm{def:g10:inertia:force}{kracht} $\vect F_{B \to A} = -\vect F_{A \to B}$ uit: dezelfde
werklijn, dezelfde grootte, tegengestelde richtingen --- op elk ogenblik, bij elke
beweging, bij contact of bij werking op afstand.
\end{theorem}
\admitted

\begin{example}[Het boek, de tafel en de Aarde]\label{ex:g12:newtons-laws:book}
Een boek ligt op een tafel: twee wisselwerkingen, twee paren. De \omterm{def:g11:fundamental-interactions:four}{gravitatie}: de Aarde
trekt het boek omlaag ($\vect P$), het boek trekt de Aarde omhoog. Het contact: de
tafel duwt het boek omhoog ($\vect N$), het boek drukt op de tafel. Lopen is dezelfde
afspraak: duw de grond naar achteren; haar reactie is de enige \omterm{def:g10:inertia:force}{kracht} die je vooruit
drijft.
\end{example}

\section{De methode: kiezen, tekenen, projecteren}

\begin{method}[Een dynamicaprobleem oplossen]\label{met:g12:newtons-laws:method}
\begin{enumerate}
  \item \emph{Systeem}: benoem het lichaam (of de verzameling lichamen) die je
        bestudeert.
  \item \emph{Stelsel}: kies een \omterm{def:g12:newtons-laws:inertial}{traagheidsstelsel} en assen die bij de beweging
        passen --- assen langs de helling, of het \omterm{def:g12:kinematics-2d:frenet}{frenetstelsel} voor een kromme
        (\cref{ch:g12:kinematics-2d}).
  \item \emph{Inventaris}: noem elke \omterm{def:g10:inertia:force}{kracht} --- het \omterm{def:g10:universal-gravitation:weight}{gewicht}, daarna één \omterm{def:g10:inertia:force}{kracht} per
        contact, verder niets --- en teken het \omterm{def:g10:inertia:diagram}{krachtenschema}: één punt, één pijl per
        \omterm{def:g10:inertia:force}{kracht}.
  \item \emph{Projecteren}: schrijf $\sum \vect F = m\vect a$ en projecteer die op
        elke as.
  \item \emph{Oplossen en controleren}: eenheden, tekens, grensgevallen
        ($\mu \to 0$, $\alpha \to 0$, \dots).
\end{enumerate}
\end{method}

\begin{definition}[Normaalkracht en wrijving]\label{def:g12:newtons-laws:friction}
Een oppervlak werkt op een lichaam via twee componenten: de
\emph{normaalkracht}\index{normaalkracht} $\vect N$, loodrecht op het oppervlak, en
de \omterm{def:g10:inertia:inventory}{wrijving} $\vect f$, ernaartoe gericht langs het oppervlak. Het experiment geeft de
wrijvingswetten: zolang het lichaam niet glijdt, stelt de \omterm{def:g11:forces-and-motion:friction}{statische wrijving} zich in
tot een plafond, $f_s \leq \mu_s N$; zodra het glijdt, is de \omterm{def:g11:forces-and-motion:friction}{kinetische wrijving}
$f_k = \mu_k N$, tegen het glijden in. De
\emph{wrijvingscoëfficiënten}\index{wrijvingscoëfficiënt} $\mu_s$ en $\mu_k$
($\mu_k \leq \mu_s$) hangen alleen van de twee materialen af.
\end{definition}

\begin{example}[De helling, twee keer]\label{ex:g12:newtons-laws:incline}
Een blok glijdt een helling met hoek $\alpha$ af; assen: $x$ de helling af, $y$
loodrecht erop. Op $y$: $N = mg\cos\alpha$. Op $x$, wrijvingsloos:
$ma = mg\sin\alpha$, dus $a = g\sin\alpha$ --- zonder massa; bij
$\alpha = 30^\circ$ is $a = \qty{4.9}{m/s^2}$. Met \omterm{def:g11:forces-and-motion:friction}{kinetische wrijving}:
$a = g(\sin\alpha - \mu_k\cos\alpha)$; voor $\mu_k = 0.20$ geeft dat
$a = 9.81\,(0.500 - 0.173) = \qty{3.2}{m/s^2}$. En het blok bleef om te beginnen
alleen liggen zolang de benodigde \omterm{def:g11:forces-and-motion:friction}{statische wrijving} onder haar plafond paste:
$\tan\alpha \leq \mu_s$.
\end{example}

\begin{definition}[Schijnbaar gewicht]\label{def:g12:newtons-laws:apparent}
Het \emph{schijnbare gewicht}\index{schijnbaar gewicht} van een lichaam is de
\omterm{def:g12:newtons-laws:friction}{normaalkracht} tussen het lichaam en zijn steunvlak --- wat een weegschaal eronder
aanwijst.
\end{definition}

\begin{example}[De lift]\label{ex:g12:newtons-laws:elevator}
Een passagier van \qty{70}{kg} staat op een weegschaal in een lift met verticale
versnelling $a_z$ (omhoog positief): $N - mg = ma_z$, dus $N = m(g + a_z)$. Bij
optrekken met $a_z = \qty{1.5}{m/s^2}$: \qty{792}{N}, zwaar; bij afremmen vlak onder
de bovenste verdieping: \qty{582}{N}, licht; bij eenparig rijden:
$mg = \qty{687}{N}$ --- de eerste wet; bij vrije val, $a_z = -g$: $N = 0$,
gewichtloosheid op een weegschaal.
\end{example}

\begin{example}[Twee blokken en een katrol]\label{ex:g12:newtons-laws:pulley}
Een blok $m_1 = \qty{4.0}{kg}$ op een wrijvingsloze tafel is met een licht touw over
een ideale katrol verbonden met een hangend blok $m_2 = \qty{1.0}{kg}$. Het touw
draagt aan beide uiteinden dezelfde \omterm{def:g10:inertia:inventory}{spankracht} $T$ over; de blokken delen één
grootte van de versnelling $a$. Eén tweede wet per blok, langs elke beweging:
$m_1 a = T$ en $m_2 a = m_2 g - T$, dus
$a = m_2\,g/(m_1 + m_2) = \qty{1.96}{m/s^2}$ en $T = m_1 a = \qty{7.8}{N}$ ---
kleiner dan $m_2 g = \qty{9.8}{N}$, zoals nodig is om $m_2$ omlaag te laten
versnellen.
\end{example}

\begin{example}[De vlakke bocht]\label{ex:g12:newtons-laws:curve}
Een auto met massa $m$ neemt met constante snelheid $v$ een vlakke bocht met straal
$R$. In het \omterm{def:g12:kinematics-2d:frenet}{frenetstelsel} (\cref{ch:g12:kinematics-2d}) is de versnelling zuiver
normaal, $a_N = v^2/R$, naar het middelpunt gericht. Verticaal geldt $N = mg$;
horizontaal is de enige middelpuntzoekende \omterm{def:g10:inertia:force}{kracht} die beschikbaar is de \omterm{def:g11:forces-and-motion:friction}{statische
wrijving} van de weg op de banden: $f = mv^2/R \leq \mu_s N = \mu_s mg$, dus
$v \leq \sqrt{\mu_s\, g R}$, wat de massa ook is. Voor $R = \qty{90}{m}$ en droog
wegdek ($\mu_s = 0.70$): $v_{\max} \approx \qty{25}{m/s}$ (\qty{89}{km/h}); op ijs
stort het plafond in --- \cref{exo:g12:newtons-laws:13} verkant de weg,
\cref{exo:g12:newtons-laws:10} laat dezelfde projectie aan een draad slingeren.
\end{example}

\section{Geïsoleerde systemen: de impuls blijft behouden}

\begin{proposition}[Behoud van impuls]\label{prop:g12:newtons-laws:conservation}\index{behoud van impuls}
Voor twee lichamen waarop de uitwendige \omterm{def:g10:inertia:force}{krachten} verdwijnen of elkaar opheffen, is de
totale \omterm{def:g12:newtons-laws:momentum}{impuls} $\vect p_1 + \vect p_2$ constant, welke \omterm{def:g10:inertia:force}{krachten} ze ook op elkaar
uitoefenen.
\end{proposition}

\begin{proof}
Tel de twee tweede wetten op:
$\frac{d}{dt}(\vect p_1 + \vect p_2) = \vect F_{2 \to 1} + \vect F_{1 \to 2} +
\vect F_{\text{uitw}}$: het inwendige paar valt weg door de derde wet, en
$\vect F_{\text{uitw}} = \vect 0$ volgens de veronderstelling.
\end{proof}

\begin{example}[Terugstoot]\label{ex:g12:newtons-laws:recoil}
Een schaatsster van \qty{60}{kg} staat stil op glad ijs en gooit een medicijnbal van
\qty{4.0}{kg} horizontaal weg met \qty{6.0}{m/s}. Ervoor: $\vect p = \vect 0$; erna:
$0 = 60\,V + 4.0 \times 6.0$, dus $V = \qty{-0.40}{m/s}$ --- ze drijft achteruit.
Niets van buitenaf duwde haar: de bal deed het (derde wet), en de boeken blijven op
nul.
\end{example}

\begin{remark}[Hoe een raket op niets duwt]\label{rem:g12:newtons-laws:rocket}
Een raket is een machine om massa weg te gooien. Elke seconde slingeren haar motoren
een portie gas naar achteren; het gas voert achterwaartse \omterm{def:g12:newtons-laws:momentum}{impuls} mee, dus wint de
raket dezelfde hoeveelheid vooruit --- een gestage \omterm{def:g10:inertia:inventory}{stuwkracht}, ruwweg de per seconde
uitgestoten massa maal de uitstootsnelheid. De raket duwt op haar eigen uitlaatgassen
en niet op de lucht: ze werkt het best in de lege ruimte.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:1}
Bereken de \omterm{def:g12:newtons-laws:momentum}{impuls} van (a) een sprinter van \qty{70}{kg} met \qty{10}{m/s}; (b) een
auto van \qty{1300}{kg} met \qty{50}{km/h}; (c) een kogel van \qty{8.0}{g} met
\qty{800}{m/s}. Rangschik ze. Waarom komt de snelste als laatste?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:2}
Een passagier van \qty{70}{kg} staat op een weegschaal in een lift. Wat wijst die aan
wanneer de cabine (a) omhoog versnelt met \qty{1.2}{m/s^2}; (b) met een constante
\qty{2.0}{m/s} klimt; (c) omlaag versnelt met \qty{1.2}{m/s^2}? Welke wet beslist
(b)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:3}
Een blok wordt losgelaten op een wrijvingsloze helling met hoek $30^\circ$. Bereken
zijn versnelling, en daarna zijn snelheid na \qty{3.0}{m} glijden. Waar is de massa
gebleven?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:4}
Een tennisbal van \qty{58}{g} komt aan met \qty{15}{m/s} en wordt langs dezelfde lijn
teruggeslagen met \qty{25}{m/s}; het contact duurt \qty{5.0}{ms}. Bereken de
impulsverandering en de gemiddelde \omterm{def:g10:inertia:force}{kracht} van de snaren; vergelijk die \omterm{def:g10:inertia:force}{kracht} met het
\omterm{def:g10:universal-gravitation:weight}{gewicht} van de bal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:newtons-laws:5}
Een boek ligt op een tafel. Noem de twee \omterm{def:g10:inertia:force}{krachten} op het boek en, voor elk, de
partner uit het actie--reactiepaar (welk lichaam? welke richting?). Waarom zijn het
\omterm{def:g10:universal-gravitation:weight}{gewicht} van het boek en de duw van de tafel geen partners van elkaar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:6}
Een krat van \qty{50}{kg} staat op een horizontale vloer, $\mu_s = 0.45$,
$\mu_k = 0.35$. (a) Welke horizontale duw brengt het net in beweging? (b) Als die duw
wordt volgehouden, wat is dan zijn versnelling? (c) Welke duw houdt het bij constante
snelheid in beweging?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:8}
Een blok van \qty{3.0}{kg} op een wrijvingsloze tafel is over een ideale katrol

\omterm{def:g10:inertia:inventory}{spankracht}. Waarom moest de \omterm{def:g10:inertia:inventory}{spankracht} kleiner uitvallen dan het hangende \omterm{def:g10:universal-gravitation:weight}{gewicht}?

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:9}
Een bocht met straal \qty{50}{m} is vlak. Bereken de hoogste snelheid van een auto

antwoord even goed voor een scooter als voor een vrachtwagen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:10}
Een bol van \qty{0.30}{kg} aan een draad van \qty{1.2}{m} beschrijft een horizontale

\begin{exercise}[$\star\star$]\label{exo:g12:newtons-laws:11}

\end{tabular}\end{center}
Bereken de versnelling in elke fase en beschrijf een mogelijke rit. Kan de lift de

Een schaatsster van \qty{60}{kg} staat stil op glad ijs en gooit een bal van
\qty{4.0}{kg} horizontaal weg. (a) De bal vertrekt met \qty{8.0}{m/s} \emph{ten
opzichte van de grond}: hoe groot is haar terugstootsnelheid? (b) De \qty{8.0}{m/s}
is nu \emph{ten opzichte van haar} gemeten: bereken beide snelheden opnieuw.
(c) Welke \omterm{def:g10:inertia:force}{kracht} duwde haar achteruit?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:newtons-laws:13}

groot is haar \omterm{def:g10:universal-gravitation:weight}{gewicht}? En haar versnelling bij het opstijgen? (c) Waarom werkt de

\section{Probleem: De kabelbaan}

\begin{problem}[{Weekendopgave --- de kabelbaan keuren: de spankracht die de helling
in rust vraagt, de toeslag van het optrekken, de opluchting boven de mast, de prijs
van een noodstop --- en het ene getal dat het bordje in de cabine mag beloven}]
\label{pb:g12:newtons-laws:1}
Een kabelbaan in de bergen --- cabine plus loopwerk, lege massa
$M_0 = \qty{2.0e3}{kg}$ --- rolt over een draagkabel die onder $\alpha = 30^\circ$
helt, getrokken door een trekkabel evenwijdig aan de helling met \omterm{def:g10:inertia:inventory}{spankracht}
$\vect T$. De rolwrijving is verwaarloosbaar; de vloer van de cabine blijft de hele
tijd horizontaal. De cabine biedt plaats aan $25$ passagiers met een gemiddelde massa
van \qty{80}{kg}; jij bent de keuringsingenieur. Tenzij anders vermeld is de cabine
vol: $M = \qty{4.0e3}{kg}$.

\medskip
\noindent\textbf{Deel I --- In rust op de helling.}
\begin{enumerate}
  \item Kies het systeem; zeg waarom het stelsel van de berg als \omterm{def:g12:newtons-laws:inertial}{traagheidsstelsel}
        volstaat; inventariseer de drie \omterm{def:g10:inertia:force}{krachten}, met hun richtingen.
  \item Welke wet regeert het wachten? Projecteer die langs en loodrecht op de
        helling om $T$ en de \omterm{def:g12:newtons-laws:friction}{normaalkracht} $N$ van de draagkabel uit te drukken.
  \item Bereken $T$ en $N$.
  \item Bereken $T$ opnieuw voor de lege cabine. Waarom is $T$ evenredig met de
        totale massa?
  \item De trekkabel knapt en de remmen staan af: toon aan dat de versnelling van de
        wegrollende cabine $g \sin\alpha$ is, beladen of leeg, en bereken haar.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Optrekken.} De lier versnelt de volle cabine met
$a = \qty{0.60}{m/s^2}$ de helling op tot ze met $v = \qty{6.0}{m/s}$ doorrijdt.
\begin{enumerate}[resume]
  \item \omterm{def:g10:pressure:pressure}{Druk} de \omterm{def:g10:inertia:inventory}{spankracht} $T'$ tijdens deze fase uit en bereken haar.
  \item Met welk deel tilt het optrekken de \omterm{def:g10:inertia:inventory}{spankracht} boven haar waarde in rust?
  \item Hoe lang duurt de fase, en over welke afstand?
  \item Een passagier van \qty{80}{kg} staat op de horizontale vloer. Projecteer de
        tweede wet verticaal om haar \omterm{def:g12:newtons-laws:apparent}{schijnbare gewicht} uit te drukken en te
        berekenen; vergelijk met $mg$.
  \item Welke \omterm{def:g10:inertia:force}{kracht} versnelt haar horizontaal, en hoe groot is die? Waarom leunen
        passagiers?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Over de mast.} Bij een constante \qty{6.0}{m/s} buigt
de draagkabel over een steunmast langs een verticale cirkelboog met straal
$R = \qty{40}{m}$; op het hoogste punt is de snelheid horizontaal.
\begin{enumerate}[resume]
  \item Geef in het \omterm{def:g12:kinematics-2d:frenet}{frenetstelsel} op het hoogste punt
        (\cref{ch:g12:kinematics-2d}) beide componenten van de versnelling, met hun
        waarden.
  \item Projecteer de tweede wet langs de normaal om de \omterm{def:g12:newtons-laws:friction}{normaalkracht} $N'$ van de
        kabel op het loopwerk uit te drukken en te berekenen.
  \item Dezelfde vraag voor het \omterm{def:g12:newtons-laws:apparent}{schijnbare gewicht} van de passagier op het hoogste
        punt: met welk deel is ze lichter?
  \item Bij welke snelheid zou $N'$ verdwijnen? Vergelijk met de rijsnelheid en
        becommentarieer.
  \item Welke \omterm{def:g10:inertia:force}{kracht} oefent het loopwerk boven de mast op de kabel uit --- welke wet
        zegt dat, met welke grootte en richting?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Remmen, een sprong, en het bordje.}
\begin{enumerate}[resume]
  \item Op het horizontale aankomsttraject brengt een noodstop de volle cabine in
        \qty{1.5}{s} van \qty{6.0}{m/s} tot stilstand: lees
        $\sum \vect F = d\vect p/dt$ als een gemiddelde over de stop om de remkracht
        te berekenen.
  \item Welke horizontale \omterm{def:g10:inertia:force}{kracht} moet een passagier van \qty{80}{kg} tot stilstand
        brengen, en kan de \omterm{def:g11:forces-and-motion:friction}{statische wrijving} tussen schoen en vloer
        ($\mu_s = 0.40$) die leveren? Trek je conclusie.
  \item Op het perron, met de remmen af, hangt de lege cabine vrij en stil; de laatste
        passagier (\qty{80}{kg}) springt er horizontaal uit met \qty{2.5}{m/s}
        \emph{ten opzichte van de cabine}. Bereken de terugstootsnelheid van de
        cabine en de snelheid van de passagier ten opzichte van de grond.
  \item In één zin: welk voortstuwingsapparaat leeft van dit terugstootbeginsel, en
        wat legt zijn \omterm{def:g10:inertia:inventory}{stuwkracht} vast?
  \item \emph{Keuring.} De trekkabel breekt bij \qty{100}{kN}; de voorschriften
        beperken de werkspankracht tot een vijfde daarvan. Toon aan dat het zwaarste
        geval hierboven het optrekken is, leid de maximale totale massa af, en \omterm{def:g10:pressure:pressure}{druk}
        het bordje af: hoeveel passagiers van \qty{80}{kg} mogen mee? Controleer de
        veiligheidsfactor.
\end{enumerate}
\end{problem}
