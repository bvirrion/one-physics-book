\section{De condensator}

\begin{definition}[Condensator en capaciteit]\label{def:g12:rc-rl-circuits:capacitor}
Een \emph{condensator}\index{condensator} is een paar metalen platen die
tegenover elkaar staan, gescheiden door een \omterm{def:g11:lenses-and-eye:lens}{dunne} isolator. Aangesloten op een
bron nemen de platen tegengestelde ladingen $+q$ en $-q$ aan --- een
\omterm{def:g11:electric-gravitational-fields:efield}{elektrisch veld} vult de spleet
(\cref{ch:g11:electric-gravitational-fields}) --- en de \omterm{def:g11:circuits-and-power:voltage}{spanning} $u$ over de
platen is evenredig met de opgeslagen lading:
\[ q = C\,u . \]
De constante $C$ is de \emph{capaciteit}\index{capaciteit}, in
\emph{farad}\index{farad} (\unit{F}): $\qty{1}{F} = \qty{1}{C/V}$.
\end{definition}

\begin{example}[Ordegrootten]\label{ex:g12:rc-rl-circuits:orders}
Eén \omterm{def:g12:rc-rl-circuits:capacitor}{farad} is enorm --- één \omterm{def:g11:fundamental-interactions:charge}{coulomb} geparkeerd per \omterm{def:g11:circuits-and-power:voltage}{volt}. De keramische
\omterm{def:g12:rc-rl-circuits:capacitor}{condensatoren} van radioschakelingen: \unit{pF} tot \unit{nF}; de
elektrolytische van voedingen en flitsers: \unit{\micro F} tot \unit{mF};
supercondensatoren: duizenden \omterm{def:g12:rc-rl-circuits:capacitor}{farad}.
\end{example}

\begin{proposition}[Stroom naar een condensator]\label{prop:g12:rc-rl-circuits:cdudt}
De stroom die op een plaat van een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} aankomt, is het tempo waarmee
haar lading groeit:
\[
i = \frac{dq}{dt} = C\,\frac{du}{dt} ,
\]
waarbij kleine letters de grootheden aangeven die met de tijd variëren ($U$
en $I$ blijven constant, \cref{ch:g11:circuits-and-power}).
\end{proposition}

\begin{proof}
Stroom is lading die per seconde wordt geleverd
(\cref{def:g11:circuits-and-power:current}): over een krimpend interval is
$i$ de afgeleide van $q$; en $q = Cu$ met constante $C$ geeft $C\,du/dt$.
\end{proof}

\begin{remark}[Twee gevolgen]\label{rem:g12:rc-rl-circuits:continuity}
Is $u$ constant, dan is $i = 0$: een geladen \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} laat geen
gelijkstroom door. En $u$ kan nooit springen --- een sprong zou een oneindige
stroom vragen: de \omterm{def:g11:circuits-and-power:voltage}{spanning} van de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} is het geheugen van de
schakeling.
\end{remark}

\section{Een condensator laden: de RC-kring}

Eén kring --- bron $\mathcal{E}$ (\cref{def:g11:circuits-and-power:emf}),
schakelaar, \omterm{def:g11:circuits-and-power:resistance}{weerstand}, lege \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}: sluit $K$ op $t = 0$.

\begin{theorem}[Laadwet van de RC-kring]\label{thm:g12:rc-rl-circuits:charging}
\index{RC-kring}
Nadat de schakelaar sluit, voldoet de \omterm{def:g11:circuits-and-power:voltage}{spanning} over de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} aan de
volgende vergelijking, en volgt uit $u(0) = 0$:
\[
RC\,\frac{du}{dt} + u = \mathcal{E},
\qquad
u(t) = \mathcal{E}\left(1 - e^{-t/\tau}\right),
\qquad \tau = RC .
\]
\end{theorem}

\begin{proof}
De \omterm{def:g11:circuits-and-power:voltage}{spanningen} langs de kring tellen op
(\cref{prop:g11:circuits-and-power:laws}): $\mathcal{E} = u_R + u = Ri + u$ op
elk ogenblik; vul $i = C\,du/dt$ in
(\cref{prop:g12:rc-rl-circuits:cdudt}). \emph{Controleer} nu de kandidaat:
$du/dt = (\mathcal{E}/\tau)e^{-t/\tau}$, dus is
$RC\,(\mathcal{E}/RC)\,e^{-t/\tau} + \mathcal{E}(1 - e^{-t/\tau}) =
\mathcal{E}$, waar voor alle $t$; en $u(0) = \mathcal{E}(1-1) = 0$. Dat geen
\emph{andere} kromme bij de vergelijking en de start past, wordt bewezen in
de wiskundecursus over differentiaalvergelijkingen van dit jaar.
\end{proof}

\begin{definition}[Tijdconstante]\label{def:g12:rc-rl-circuits:tau}
De \emph{tijdconstante}\index{tijdconstante} van een \omterm{thm:g12:rc-rl-circuits:charging}{RC-kring} is
$\tau = RC$. Eenheden: $\unit{\ohm} \times \unit{F} =
(\unit{V/A})(\unit{C/V}) = \unit{C/A} = \unit{s}$ --- het eigen tempo van de
schakeling.
\end{definition}

\begin{method}[$\tau$ van de kromme aflezen]\label{met:g12:rc-rl-circuits:reading}
Drie gelijkwaardige aflezingen, rechtstreeks van een grafiek:
\begin{enumerate}
  \item \emph{raaklijn in de oorsprong:} $u'(0) = \mathcal{E}/\tau$, dus
        bereikt de beginraaklijn de asymptoot op $t = \tau$;
  \item \emph{63\%:} $u(\tau) = \mathcal{E}(1 - e^{-1}) \approx
        0.63\,\mathcal{E}$;
  \item \emph{95\%:} $u(3\tau) \approx 0.95\,\mathcal{E}$ --- in de praktijk
        geladen ($5\tau$: $99\%$).
\end{enumerate}
\end{method}

\begin{proposition}[Ontlading]\label{prop:g12:rc-rl-circuits:discharge}
Een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} die tot $U_0$ is geladen en alleen op een \omterm{def:g11:circuits-and-power:resistance}{weerstand} $R$ wordt
gesloten, voldoet aan $RC\,du/dt + u = 0$ en volgt
\[
u(t) = U_0\,e^{-t/\tau}, \qquad \tau = RC
\]
--- nog $37\%$ over op $\tau$, $5\%$ op $3\tau$.
\end{proposition}

\begin{proof}
Dezelfde kringwet, zonder bron; opnieuw invullen:
$RC(-U_0/\tau)e^{-t/\tau} + U_0 e^{-t/\tau} = 0$, met $u(0) = U_0$.
\end{proof}

\begin{example}[De getallen van een flitser]\label{ex:g12:rc-rl-circuits:flashrc}
Een fotoflitser laadt $C = \qty{800}{\micro F}$ door
$R = \qty{1.0}{k\ohm}$: $\tau = 10^3 \times 8.0\times10^{-4} =
\qty{0.80}{s}$, dus wacht het lampje ``gereed'' --- afgesteld om bij $95\%$
om te slaan --- $3\tau \approx \qty{2.4}{s}$: het gejank van het wachten.
Afgevuurd door de schamele \qty{0.50}{\ohm} van de flitsbuis loopt
\emph{dezelfde} \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} leeg met $\tau' = \qty{0.40}{ms}$: er tweeduizend
keer sneller uit dan erin.
\end{example}

\section{De spoel: zelfinductie en de RL-kring}

\begin{definition}[Spoel en zelfinductie]\label{def:g12:rc-rl-circuits:inductor}
Een \emph{spoel}\index{spoel} is een wikkeling draad: haar stroom rijgt haar
eigen windingen aan een \omterm{def:g11:magnetic-fields:bfield}{magnetisch veld}
(\cref{ch:g11:magnetic-fields}); zodra dat \emph{verandert}, verschijnt er een
\omterm{def:g11:circuits-and-power:voltage}{spanning} over de spoel:
\[
u = L\,\frac{di}{dt} .
\]
De constante $L$ is de \emph{zelfinductie}\index{zelfinductie}, in
\emph{henry}\index{henry} (\unit{H}): $\qty{1}{H} = \qty{1}{V.s/A}$.
\end{definition}
\admitted

\begin{remark}[De spoel verzet zich tegen verandering]\label{rem:g12:rc-rl-circuits:oppose}
Waarom een veranderende stroom een \omterm{def:g11:circuits-and-power:voltage}{spanning} maakt --- inductie --- wordt in de
universitaire volumes afgeleid; hier nemen we de wet en haar karakter aan: de
\omterm{def:g12:rc-rl-circuits:inductor}{spoel} \emph{verzet zich tegen veranderingen} van haar stroom. Spiegelbeelden:
de $u$ van een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} kan niet springen, de $i$ van een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} evenmin.
\end{remark}

    to[R=$R$] (4.2,2.4) -- (5.4,2.4) to[L=$L$, i=$i$, v=$u$] (5.4,0) -- (0,0);
\end{circuitikz}

{\small De \omterm{prop:g12:rc-rl-circuits:rl}{RL-kring}: nadat $K$ sluit, laat de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} de stroom alleen in haar
eigen tempo $\tau = L/R$ klimmen.}
\end{omfigure}

\begin{proposition}[Stroomopbouw in een RL-kring]\label{prop:g12:rc-rl-circuits:rl}
\index{RL-kring}
Door de schakelaar te sluiten op een bron $\mathcal{E}$, een \omterm{def:g11:circuits-and-power:resistance}{weerstand} $R$ en
een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} $L$ in serie krijg je $\mathcal{E} = Ri + L\,di/dt$, oftewel
\[
\frac{L}{R}\,\frac{di}{dt} + i = \frac{\mathcal{E}}{R},
\qquad\text{met als oplossing}\quad
i(t) = \frac{\mathcal{E}}{R}\left(1 - e^{-t/\tau}\right),
\quad \tau = \frac{L}{R} .
\]
\end{proposition}

\begin{proof}
Opnieuw de kringwet; en het is \emph{dezelfde vergelijking} als in
\cref{thm:g12:rc-rl-circuits:charging} met $u \to i$,
$\mathcal{E} \to \mathcal{E}/R$ en $RC \to L/R$ --- dezelfde substitutie
controleert de oplossing, en elke aflezing uit
\cref{met:g12:rc-rl-circuits:reading} draagt over.
\end{proof}

\begin{example}[Een elektromagneet wordt wakker]\label{ex:g12:rc-rl-circuits:rl}
Een relaisspoel met $L = \qty{0.30}{H}$ en $R = \qty{6.0}{\ohm}$ op een
voeding van \qty{12}{V}: eindstroom $\mathcal{E}/R = \qty{2.0}{A}$, in het
tempo $\tau = 0.30/6.0 = \qty{50}{ms}$ --- wakker in
$3\tau \approx \qty{0.15}{s}$.
\end{example}

\begin{remark}[De vonk bij het openen]\label{rem:g12:rc-rl-circuits:spark}
Open een schakelaar op een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} die stroom voert en $i$ moet in een fractie
van een milliseconde naar nul storten: $di/dt$ is reusachtig en
$u = L\,di/dt$ haalt honderden of duizenden \omterm{def:g11:circuits-and-power:voltage}{volt} --- er springt een vonk over
het contact. De bobine van een auto doet dat met opzet, duizenden keren per
minuut.
\end{remark}

\section{Opgeslagen energie}

\begin{proposition}[Energie in een condensator, energie in een spoel]\label{prop:g12:rc-rl-circuits:energy}
\index{energie!van een condensator}\index{energie!van een spoel}
Een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} $C$ die tot \omterm{def:g11:circuits-and-power:voltage}{spanning} $u$ is geladen en een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} $L$ die een
stroom $i$ voert, slaan
\[
E_C = \tfrac12\,C u^2 = \frac{q^2}{2C},
\qquad
E_L = \tfrac12\,L i^2
\]
op.
\end{proposition}

\begin{proof}
Een kleine lading $dq$ neerzetten op platen die al op \omterm{def:g11:circuits-and-power:voltage}{spanning} $u$ staan kost
$u\,dq$ (\cref{def:g11:circuits-and-power:voltage}); door de kosten te
stapelen wordt het totaal de oppervlakte onder de rechte $u = q/C$ --- de
driehoek hieronder --- dus $E_C = \tfrac12 qu = \tfrac12 Cu^2$. Dezelfde
driehoek voor de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} ($i$ met $di$ verhogen kost $Li\,di$) geeft
$E_L = \tfrac12 Li^2$.
\end{proof}

\omterm{def:g11:circuits-and-power:voltage}{spanning}.}
\end{omfigure}

\begin{example}[Ordegrootten van energie]\label{ex:g12:rc-rl-circuits:energies}
De flitscondensator van \cref{ex:g12:rc-rl-circuits:flashrc}:
$\tfrac12 \times 8.0\times10^{-4} \times 300^2 = \qty{36}{J}$. Een
supercondensator van \qty{3000}{F} op \qty{2.7}{V}: \qty{1.1e4}{J} --- zoveel
als een AA-batterij. Een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} van \qty{10}{mH} bij \qty{10}{A}:
\qty{0.50}{J}. Minder dan batterijen bevatten --- maar vrij te geven in een
milliseconde.
\end{example}

\begin{remark}[Waar de twee tijdmeters voor dienen]\label{rem:g12:rc-rl-circuits:applications}
\index{afvlakking}\index{tijdschakeling}
Langzaam erin, snel eruit: dat drijft de cameraflitser en de defibrillator
aan. De \omterm{def:g12:mechanical-waves:delay}{vertraging} $\tau = RC$ laat \omterm{rem:g12:rc-rl-circuits:applications}{tijdschakelingen} lopen:
intervalruitenwissers, knipperende bakens, trappenhuisverlichting. Een
\omterm{def:g12:rc-rl-circuits:capacitor}{condensator} over een voeding \emph{vlakt} de \omterm{def:g11:circuits-and-power:voltage}{spanning} \emph{af} ($u$ kan niet
springen); een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} in serie vlakt de stroom af ($i$ kan niet springen). En
\omterm{def:g12:rc-rl-circuits:inductor}{spoel} plus \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} vormen samen een \omterm{def:g12:oscillators-and-time:oscillator}{oscillator}
(\cref{ch:g12:rlc-oscillations}).
\end{remark}

\section{Probleem: De cameraflitser}

\begin{problem}[{Weekendopgave --- de cameraflitser: twee seconden gejank,
een milliseconde glorie, en de neef aan de muur die harten herstart --- één
exponentiële functie stuurt ze allemaal}]
\label{pb:g12:rc-rl-circuits:1}
Een compacte flitser slaat \omterm{def:g10:energy-conservation:energy}{energie} op in een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}
$C = \qty{800}{\micro F}$, geladen door $R = \qty{1.0}{k\ohm}$ vanuit een
omvormer die als een ideale bron $\mathcal{E} = \qty{300}{V}$ wordt
gemodelleerd; het lampje ``gereed'' slaat om bij $95\%$ van de volle \omterm{def:g11:circuits-and-power:voltage}{spanning}.
Bij het afvuren ziet de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} alleen de flitsbuis,
$R_f = \qty{0.50}{\ohm}$. Gegevens: een AA-cel levert \qty{1.5}{V} en
\qty{2.4}{A.h}; $e = \qty{1.60e-19}{C}$.

\medskip
\noindent\textbf{Deel I --- Het reservoir.}
\begin{enumerate}
  \item Bereken de lading op de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} bij \qty{300}{V}.
  \item Hoeveel \omterm{def:g11:fundamental-interactions:elementary}{elementaire ladingen} is dat?
  \item Bereken de opgeslagen \omterm{def:g10:energy-conservation:energy}{energie} $E_C$.
  \item De AA-cel bevat $q = It \approx \qty{8.6e3}{C}$, dus ongeveer
        \qty{1.3e4}{J}. Hoeveel keer de \omterm{def:g10:energy-conservation:energy}{energie} van de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} is dat?
  \item Waarom dan toch de moeite van de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}? Eén zin, met het woord
        ``\omterm{def:g11:work-of-force:power}{vermogen}''.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het wachten op groen.}
\begin{enumerate}[resume]
  \item Schrijf de kringwet op en maak er de differentiaalvergelijking voor
        $u(t)$ van.
  \item Ga door invullen na dat $u(t) = \mathcal{E}(1 - e^{-t/RC})$ haar
        oplost, startend vanaf $u(0) = 0$.
  \item Bereken de \omterm{def:g12:rc-rl-circuits:tau}{tijdconstante} $\tau$.
  \item Na hoeveel tijd springt het lampje op groen, en bij welke \omterm{def:g11:circuits-and-power:voltage}{spanning}?
        Herken het gejank van twee seconden.
  \item Bereken de beginlaadstroom; waarom sterft die daarna weg?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De milliseconde glorie.}
\begin{enumerate}[resume]
  \item Schrijf de ontlaadwet $u(t)$ door de buis op en bereken de nieuwe
        \omterm{def:g12:rc-rl-circuits:tau}{tijdconstante} $\tau'$.
  \item Bereken de piekstroom door de buis.
  \item Bereken het piekvermogen; vergelijk het met een waterkoker van
        \qty{2.2}{kW}.
  \item Beschouw de flits als voorbij op $3\tau'$ en geef haar duur en haar
        gemiddelde \omterm{def:g11:work-of-force:power}{vermogen}.
  \item Neem de verhouding van laadtijd tot flitstijd: dezelfde \omterm{def:g12:rc-rl-circuits:capacitor}{condensator},
        dezelfde lading --- wat is er veranderd, en wat maakt dat van een
        \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}: een energietank of een vermogenshefboom?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De neef aan de muur.}
Een defibrillator moet $E_C = \qty{200}{J}$ leveren bij
$U_0 = \qty{1500}{V}$; borstkas en peddels bieden ongeveer \qty{50}{\ohm}.
\begin{enumerate}[resume]
  \item Bereken de vereiste \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit}.
  \item Bereken de opgeslagen lading.
  \item Bereken de \omterm{def:g12:rc-rl-circuits:tau}{tijdconstante} en de duur ($3\tau$) van de schok; vergelijk
        met de flitser.
  \item Bereken de piekstroom en het piekvermogen in de patiënt.
  \item De laadschakeling moet het toestel in \qty{6.0}{s} gereed ($95\%$)
        maken: bepaal de vereiste laadweerstand en de beginlaadstroom --- de
        getallen die een ontwerper zou bestellen.
\end{enumerate}
\end{problem}
