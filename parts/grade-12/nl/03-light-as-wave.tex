\chapter{Licht als golf: buiging en interferentie}\label{ch:g12:light-as-wave}

Kantel een zeepvlies naar het raam: er glijden kleuren overheen, terwijl
er in zeep of water niets gekleurd is. Richt een \omterm{def:g11:color-light-sources:lamps}{laser} op één haar: de
muur erachter toont geen haardunne schaduw maar een heldere streep,
doorsneden met donkere lijnen, duizenden keren breder dan de haar.
Rechtlijnig marcherende stralen (\cref{ch:g10:refraction}) verklaren geen
van beide taferelen. Dit hoofdstuk bevordert het licht tot een volwaardige
\omterm{def:g10:signals-and-waves:wave}{golf} --- een halve micrometer van top tot top --- en verzilvert dat idee
daarna: een \omterm{def:g11:color-light-sources:lamps}{laser} en een kale muur meten een haar, een \omterm{def:g12:mechanical-waves:wavelength}{golflengte} en de
sporen van een cd.

\section{Licht is een golf}

\begin{definition}[Lichtgolven]\label{def:g12:light-as-wave:lightwave}
\omterm{def:g10:light-spectra:white}{Monochromatisch licht} --- één zuivere kleur, die geen enkel prisma verder
kan splitsen (\cref{ch:g10:light-spectra}) --- is een periodieke
\emph{lichtgolf}\index{lichtgolf}: ze doorkruist het vacuüm met
$c = \qty{3.00e8}{m/s}$ en wordt gekenmerkt door haar \omterm{def:g10:signals-and-waves:frequency}{frequentie} $f$ of
haar \emph{golflengte in vacuüm}\index{golflengte!van licht}
$\lambda = c/f$, de afstand van top tot top uit
\cref{ch:g12:mechanical-waves}. Het oog reageert van ongeveer
\qty{400}{nm} (violet) tot \qty{800}{nm} (rood); \omterm{def:g10:light-spectra:white}{wit licht} mengt het hele
gebied.
\end{definition}

\begin{example}[Ordegrootten]\label{ex:g12:light-as-wave:orders}
Een helium-neonlaser zendt rood licht uit met
$\lambda = \qty{633}{nm} = \qty{6.33e-7}{m}$, dus
$f = c/\lambda \approx \qty{4.7e14}{Hz}$: geen enkele elektronica telt zo
snel --- elke \omterm{def:g12:mechanical-waves:wavelength}{golflengte} in dit hoofdstuk wordt met meetkunde gemeten.
Ter vergelijking: een haar (ongeveer \qty{70}{\micro m}) is honderd
\omterm{def:g12:mechanical-waves:wavelength}{golflengten} dik. Wat golft er? Niet de lucht: een elektrisch en een
\omterm{def:g11:magnetic-fields:bfield}{magnetisch veld}, eerlijk beschreven in het volume van bachelorjaar 2;
hieronder telt alleen de \omterm{def:g12:mechanical-waves:wavelength}{golflengte}.
\end{example}

\section{Buiging}

\begin{definition}[Buiging]\label{def:g12:light-as-wave:diffraction}
Wanneer een \omterm{def:g10:signals-and-waves:wave}{golf} een opening of een hindernis ontmoet met een afmeting $a$
die vergelijkbaar is met haar \omterm{def:g12:mechanical-waves:wavelength}{golflengte}, spreidt ze zich uit in het
gebied dat stralen donker zouden laten. Die spreiding heet
\emph{buiging}\index{buiging} --- verwaarloosbaar zolang $\lambda/a$
minuscuul is, overheersend zodra $a$ tot bij $\lambda$ krimpt.
\end{definition}

\begin{proposition}[Halve hoekbreedte van de spreiding]\label{prop:g12:light-as-wave:halfwidth}
Een spleet met breedte $a$, belicht met \omterm{def:g12:mechanical-waves:wavelength}{golflengte} $\lambda < a$, spreidt
het licht rond de invalsrichting binnen de halve hoekbreedte $\theta$ (in
radialen); een scherm op afstand $D$ toont een centrale heldere band met
breedte $L$, tweemaal $D \tan\theta \approx D\,\theta$ (kleine hoek):
\[
\theta \approx \frac{\lambda}{a}, \qquad L = \frac{2\lambda D}{a}.
\]
\end{proposition}
\admitted

\begin{remark}[Lees de formule achterstevoren]\label{rem:g12:light-as-wave:honest}
Hoe \emph{smaller} de spleet, hoe \emph{breder} de spreiding --- het
tegenovergestelde van wat stralen voorspellen. De eerlijke afleiding (het
optellen van de golfjes die over de spleet opnieuw worden uitgezonden)
staat in het volume van bachelorjaar 2; hier is de formule een
meetinstrument.
\end{remark}

\begin{example}[Een laser door een spleet]\label{ex:g12:light-as-wave:slit}
Rode \omterm{def:g11:color-light-sources:lamps}{laser}, $\lambda = \qty{633}{nm}$; spleet $a = \qty{0.10}{mm}$; scherm
op $D = \qty{2.00}{m}$. Dan is
$\theta = \num{6.33e-7} / \num{1.0e-4} = \qty{6.3e-3}{rad}$ --- een derde
graad, maar de centrale band op de muur is
$L = 2 \times \num{6.33e-7} \times 2.00 / \num{1.0e-4} \approx
\qty{2.5}{cm}$ breed, geflankeerd door donkere lijnen en zwakkere
zijbanden. Halveer de spleet en de band verdubbelt.
\end{example}

\begin{proposition}[Een hindernis buigt als een spleet]\label{prop:g12:light-as-wave:hair}
Een \omterm{def:g11:lenses-and-eye:lens}{dunne} ondoorzichtige hindernis met breedte $d$ --- een draad, een haar
--- brengt buiten de invallende bundel hetzelfde patroon voort als een
spleet met breedte $d$ (het beginsel van Babinet, afgeleid in het volume
van bachelorjaar 2).
\end{proposition}
\admitted

\begin{method}[Een minieme breedte meten]\label{met:g12:light-as-wave:hairwidth}
\begin{enumerate}
  \item Span de haar dwars door de laserbundel; zet een scherm op een
        gemeten afstand $D$ (enkele \omterm{def:g10:orders-of-magnitude:unit}{meters}).
  \item Meet de breedte $L$ van de centrale band, van donkere lijn tot
        donkere lijn.
  \item Omdat $L = 2\lambda D/d$, geldt $d = 2\lambda D/L$ --- fijnere
        haar, groter patroon.
\end{enumerate}
\end{method}

\begin{remark}[Buiging begrenst elk instrument]\label{rem:g12:light-as-wave:limits}
Licht komt een fototoestel, een telescoop of een oog binnen door een
opening met breedte $a$, dus vervaagt zelfs een volmaakte \omterm{def:g11:lenses-and-eye:lens}{lens} een punt
tot een vlekje met een hoekgrootte van ongeveer $\lambda/a$: dichter bij
elkaar liggende details vloeien samen. Een lichtmicroscoop onderscheidt
daardoor vrijwel niets onder $\lambda \approx \qty{0.5}{\micro m}$
(bacteriën wel, virussen niet); telescoopspiegels worden om dezelfde reden
breed gebouwd.
\end{remark}

\section{Interferentie}

\begin{definition}[Coherente bronnen, weglengteverschil]\label{def:g12:light-as-wave:coherent}
Twee golfbronnen zijn \emph{coherent}\index{coherente bronnen} wanneer ze
dezelfde \omterm{def:g10:signals-and-waves:frequency}{frequentie} hebben en in de pas trillen --- in de praktijk wanneer
het twee kopieën van één \omterm{def:g10:signals-and-waves:wave}{golf} zijn, verkregen door haar te splitsen. Voor
een punt $M$ dat door beide wordt bereikt is het
\emph{weglengteverschil}\index{weglengteverschil} $\delta = S_2M - S_1M$.
\end{definition}

\begin{proposition}[Voorwaarden voor interferentie]\label{prop:g12:light-as-wave:conditions}
Waar twee coherente golven met \omterm{def:g12:mechanical-waves:wavelength}{golflengte} $\lambda$ elkaar overlappen,
tellen ze op --- ze \emph{interfereren}\index{interferentie}:
\begin{itemize}
  \item $\delta = k\lambda$ ($k$ geheel): in de pas, versterkend ---
        \emph{constructieve}\index{constructieve interferentie}
        interferentie (helder);
  \item $\delta = (k + \tfrac12)\lambda$: top op dal, uitdovend ---
        \emph{destructieve}\index{destructieve interferentie}
        interferentie (donker).
\end{itemize}
\end{proposition}

\begin{proof}
Een \omterm{def:g12:mechanical-waves:periodic}{periodieke golf} over gehele \omterm{def:g12:mechanical-waves:wavelength}{golflengten} verschuiven verandert niets:
toppen stapelen op toppen. Een halve \omterm{def:g12:mechanical-waves:wavelength}{golflengte} meer legt een top op een
dal: gelijke golven doven elkaar uit.
\end{proof}

\begin{example}[De tweespletenproef van Young]\label{ex:g12:light-as-wave:young}
Prik twee fijne spleten $S_1$ en $S_2$, een fractie van een millimeter uit
elkaar, in een ondoorzichtige plaat en belicht ze allebei met één \omterm{def:g11:color-light-sources:lamps}{laser}:
twee \omterm{def:g12:light-as-wave:coherent}{coherente bronnen}. Op een scherm een paar \omterm{def:g10:orders-of-magnitude:unit}{meter} verderop is de
overlap gestreept: in het midden is $\delta = 0$ en dus helder; hoger op
het scherm groeit $\delta$ door $\lambda/2$ (donker), $\lambda$
(helder)\,\dots\ duisternis waar licht bij licht opgeteld uitdooft.
\end{example}

\begin{definition}[Strepen en strepenafstand]\label{def:g12:light-as-wave:fringes}
De afwisselend heldere en donkere banden van het patroon zijn de
\emph{interferentiestrepen}\index{interferentiestrepen}; de afstand $i$
tussen de middens van twee opeenvolgende heldere strepen is de
\emph{strepenafstand}\index{strepenafstand}.
\end{definition}

\begin{proposition}[De strepenafstand]\label{prop:g12:light-as-wave:interfringe}
Voor spleten op afstand $a$ en een scherm op afstand $D \gg a$ heeft het
punt van het scherm op afstand $x$ van het midden $O$ het
\omterm{def:g12:light-as-wave:coherent}{weglengteverschil} $\delta \approx a\,x/D$ (aangenomen meetkunde); volgens
\cref{prop:g12:light-as-wave:conditions} liggen de heldere strepen op
$x_k = k\,\lambda D/a$, gelijkmatig verdeeld met \omterm{def:g12:light-as-wave:fringes}{strepenafstand}
\[
i = \frac{\lambda D}{a}.
\]
Met $\lambda = \qty{633}{nm}$, $a = \qty{0.20}{mm}$ en
$D = \qty{2.0}{m}$: $i = \qty{6.3}{mm}$ --- de opstelling vergroot
$\lambda$ met $D/a = 10^4$. De meetkunde erachter wordt uitgewerkt in het
volume van bachelorjaar 2.
\end{proposition}
\admitted

\begin{method}[Een golflengte meten]\label{met:g12:light-as-wave:lambda}
\begin{enumerate}
  \item Stuur de onbekende \omterm{def:g11:color-light-sources:lamps}{laser} door een dubbele spleet met bekende
        afstand $a$; zet het scherm op een gemeten afstand $D$.
  \item Meet de breedte van $10$ \omterm{def:g12:light-as-wave:fringes}{strepenafstanden} en deel door $10$: dat
        is de \omterm{def:g12:light-as-wave:fringes}{strepenafstand} $i$, met een tien keer kleinere meetfout.
  \item Dan is $\lambda = i\,a/D$.
\end{enumerate}
\end{method}

\begin{remark}[Waarom twee lampen nooit interfereren]\label{rem:g12:light-as-wave:lamps}
Twee bureaulampen die één muur beschijnen leveren geen strepen op: elke
lamp zendt korte, ongecorreleerde golftreinen uit waarvan de onderlinge
timing miljarden keren per seconde willekeurig verspringt, zodat de strepen
even snel meeschuiven en het oog ze tot gelijkmatig licht uitmiddelt.
Interferentie vraagt coherentie --- één \omterm{def:g10:signals-and-waves:wave}{golf} in tweeën gesplitst, zoals
de spleten van Young doen; de \omterm{def:g11:color-light-sources:lamps}{laser} is de moderne kortere weg: één
zuivere lange \omterm{def:g10:signals-and-waves:wave}{golf}.
\end{remark}

\section{Kleuren geschilderd door interferentie}

\begin{definition}[Irisatie]\label{def:g12:light-as-wave:iridescence}
Kleuren die met de kijkhoek verschuiven --- zeepbellen, olievlekken,
pauwenveren, de achterkant van een cd --- heten
\emph{irisatie}\index{irisatie}: geen pigment, maar interferentie in \omterm{def:g10:light-spectra:white}{wit
licht}, die sommige \omterm{def:g12:mechanical-waves:wavelength}{golflengten} versterkt en andere uitdooft.
\end{definition}

\begin{example}[Zeepvliezen en olievlekken]\label{ex:g12:light-as-wave:soap}
Licht dat op een zeepvlies valt, weerkaatst deels op het voorvlak en deels
op het achtervlak: twee coherente kopieën waarvan het \omterm{def:g12:light-as-wave:coherent}{weglengteverschil}
door de dikte en de kijkhoek wordt bepaald. Elke dikte stuurt sommige
\omterm{def:g12:mechanical-waves:wavelength}{golflengten} versterkt terug en dooft andere uit: het vlies weerkaatst een
kleur die verschuift terwijl het uitzakt en dunner wordt. Een olievlek op
nat asfalt haalt dezelfde truc uit.
\end{example}

\begin{remark}[Monochromatisch tegenover wit licht]\label{rem:g12:light-as-wave:white}
In \omterm{def:g10:light-spectra:white}{wit licht} tekent elke \omterm{def:g12:mechanical-waves:wavelength}{golflengte} haar eigen strepen van Young met haar
eigen $i = \lambda D/a$: in het midden zijn alle kleuren helder (één witte
streep); een paar strepen verderop lopen de patronen uiteen --- \omterm{def:g12:light-as-wave:iridescence}{irisatie}
langs de randen, en daarna vervaging. \omterm{def:g10:light-spectra:white}{Monochromatisch licht} geeft
eindeloos scherpe strepen: vandaar de \omterm{def:g11:color-light-sources:lamps}{laser} bij elke scherpe meting hier.
\end{remark}

\section{Probleem: Meten met licht}

\begin{problem}[{Weekendopgave --- meten met licht: een laser, een
meetlint en een kale muur worden een micrometerwerkplaats die een spleet,
een mensenhaar, een onbekende golflengte en de sporen van twee zilveren
schijfjes opmeet}]
\label{pb:g12:light-as-wave:1}
Jouw uitrusting: een rode \omterm{def:g11:color-light-sources:lamps}{laser} ($\lambda = \qty{633}{nm}$, volgens het
etiket), een groene \omterm{def:g11:color-light-sources:lamps}{laser} met onbekende \omterm{def:g12:mechanical-waves:wavelength}{golflengte}, een geijkte spleet
$a = \qty{100}{\micro m}$, een dubbele spleet met afstand
$a' = \qty{0.250}{mm}$, een meetlint, twee haren, een cd en een dvd.

\medskip
\noindent\textbf{Deel I --- IJken op de bekende spleet.}
Rode \omterm{def:g11:color-light-sources:lamps}{laser}, spleet, muur op $D = \qty{3.00}{m}$.
\begin{enumerate}
  \item Bereken de halve hoekbreedte $\theta$ van de gebogen bundel.
  \item Toon aan dat de centrale band de breedte $L = 2\lambda D/a$ heeft
        en bereken die.
  \item Je meet $L = \qty{3.7}{cm}$: bereken de relatieve afwijking. Is de
        methode daarmee bevestigd?
  \item Een tweede spleet is half zo breed, \qty{50}{\micro m}: voorspel
        haar $L$. Formuleer de regel die spleetbreedte en patroonbreedte
        verbindt.
  \item Verantwoord de stap $\tan\theta \approx \theta$ uit vraag 2:
        vergelijk $\theta$ met $\tan\theta$ bij deze hoek.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De haar.} Dezelfde \omterm{def:g11:color-light-sources:lamps}{laser} en muur; een
gespannen haar vervangt de spleet.
\begin{enumerate}[resume]
  \item Welk aangenomen resultaat laat je de spleetformule voor een haar
        behouden, en wat zegt het?
  \item De centrale band is $L = \qty{5.4}{cm}$ breed: bereken de diameter
        $d$ van de haar.
  \item Je meetlint leest $L$ op $\pm\qty{0.2}{cm}$ nauwkeurig: geef het
        bijbehorende gebied voor $d$, als $d \pm \Delta d$.
  \item De haar van je vriendin geeft $L = \qty{7.6}{cm}$: welke diameter?
        Wiens haar is fijner, en waarom betekent fijner breder?
  \item Zou een bureaulamp de \omterm{def:g11:color-light-sources:lamps}{laser} kunnen vervangen? Geef twee redenen.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{def:g12:mechanical-waves:wavelength}{golflengte} van de groene \omterm{def:g11:color-light-sources:lamps}{laser}.} Groene
\omterm{def:g11:color-light-sources:lamps}{laser}, dubbele spleet, muur op $D = \qty{2.00}{m}$.
\begin{enumerate}[resume]
  \item Waarom moeten beide spleten door \emph{dezelfde} \omterm{def:g11:color-light-sources:lamps}{laser} worden
        belicht?
  \item Waarom is de middelste streep helder, wat de \omterm{def:g12:mechanical-waves:wavelength}{golflengte} ook is?
  \item Tien \omterm{def:g12:light-as-wave:fringes}{strepenafstanden} beslaan \qty{4.26}{cm}. Geef $i$, en de
        reden om er tien tegelijk te meten.
  \item Leid de \omterm{def:g12:mechanical-waves:wavelength}{golflengte} van de groene \omterm{def:g11:color-light-sources:lamps}{laser} af.
  \item Op het etiket van de fabrikant staat $\qty{532 \pm 10}{nm}$:
        klopt dat? Welke kleur verwacht je?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Zilveren schijfjes lezen.} De rode \omterm{def:g11:color-light-sources:lamps}{laser}
valt loodrecht op een cd; haar sporen werken als een reflectietralie die
(aangenomen) voldoet aan $d \sin\theta = k\lambda$. Op de muur op
$D = \qty{1.00}{m}$ maakt elke bundel van de eerste orde een vlek op
afstand $x$ van de centrale.
\begin{enumerate}[resume]
  \item Voor de cd is $x = \qty{43}{cm}$: bereken $\theta$. Waarom is de
        kleinehoekbenadering uit deel I nu verboden?
  \item Leid de spoorafstand $d$ van de cd af.
  \item De dvd stuurt haar vlek van de eerste orde naar
        $x = \qty{1.65}{m}$: bereken haar $\theta$ en haar spoorafstand.
  \item Vergelijk de twee spoorafstanden. Waarom slaat een dvd meer op dan
        een cd?
  \item Sluit het schrift: noem de lengten die je dit weekend hebt
        gemeten, en zeg in één zin wat het licht tot maatlat voor alles
        maakte.
\end{enumerate}
\end{problem}
