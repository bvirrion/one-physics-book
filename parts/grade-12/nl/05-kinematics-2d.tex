\chapter{Kinematica in twee dimensies}\label{ch:g12:kinematics-2d}

Kijk naar een auto die met een constante \qty{30}{km/h} een rotonde
neemt: de naald beweegt niet, en toch voelt elke passagier zich zijwaarts
getrokken --- de beweging verandert, en het is niet de \omterm{def:g12:kinematics-2d:velocity}{snelheid}. Eén getal
per ogenblik volstaat niet meer: beweging in het vlak vraagt om pijlen.
Dit hoofdstuk rust een bewegend punt uit met drie ervan --- plaats,
\omterm{def:g12:kinematics-2d:velocity}{snelheid}, versnelling --- aaneengeregen door de afgeleide, en leest hele
families van bewegingen af aan hun meetkunde.

\section{De plaatsvector}

Beweging bestaat alleen ten opzichte van een genoemd \omterm{def:g10:relative-motion:frame}{referentiestelsel} ---
een star lichaam plus een klok (\cref{ch:g10:relative-motion}). Nieuw dit
jaar is de precisie: hang aan het stelsel een oorsprong $O$ en twee
loodrechte, geijkte assen $Ox$ en $Oy$, en leg op elk ogenblik vast waar
het punt zich bevindt, als een paar functies.

\begin{definition}[Plaatsvector]\label{def:g12:kinematics-2d:position}
In een stelsel met een oorsprong $O$ en assen $Ox$ en $Oy$ is de
\emph{plaatsvector}\index{plaatsvector} van een bewegend punt $M$ op
tijdstip $t$ de vector $\vect{OM}(t)$, met coördinaten
$\bigl(x(t),\, y(t)\bigr)$: twee functies van de tijd, één per as. De
kromme die $M$ beschrijft terwijl $t$ verloopt, is zijn \omterm{def:g10:relative-motion:trajectory}{baan}.
\end{definition}

\section{De snelheidsvector}

\begin{definition}[Snelheidsvector]\label{def:g12:kinematics-2d:velocity}
Over een kort interval verplaatst het punt zich over
$\vect{OM}(t+\Delta t) - \vect{OM}(t)$; deel door $\Delta t$ en laat het
interval krimpen: dat is de afgeleide uit je wiskundelessen, toegepast op
elke coördinaat. De \emph{snelheidsvector}\index{snelheidsvector} van $M$
is de afgeleide van zijn \omterm{def:g12:kinematics-2d:position}{plaatsvector},
$\vect v(t) = \bigl(x'(t),\, y'(t)\bigr)$, in \unit{m/s}. Haar norm
$v = \sqrt{x'^2 + y'^2}$ is de \emph{snelheid}\index{snelheid} --- het
ene getal dat de snelheidsmeter toont.
\end{definition}

\begin{example}[Een drone in horizontale vlucht]\label{ex:g12:kinematics-2d:drone}
Een drone vliegt met $x(t) = 4.0\,t$ en $y(t) = 3.0\,t$ (\omterm{def:g10:orders-of-magnitude:unit}{meter},
seconde): de \omterm{def:g10:relative-motion:trajectory}{baan} is de rechte $y = \tfrac34 x$, de \omterm{def:g12:kinematics-2d:velocity}{snelheidsvector} is
$\vect v = (4.0, 3.0)$ op elk ogenblik, en de \omterm{def:g12:kinematics-2d:velocity}{snelheid}
$\sqrt{4.0^2 + 3.0^2} = \qty{5.0}{m/s}$ is constant.
\end{example}

\begin{proposition}[Raaklijn aan de baan]\label{prop:g12:kinematics-2d:tangent}
Op elk ogenblik raakt $\vect v(t)$ aan de \omterm{def:g10:relative-motion:trajectory}{baan} in de huidige plaats van
$M$, en wijst hij in de bewegingsrichting.
\end{proposition}

\begin{proof}
De verplaatsing is een koorde van de \omterm{def:g10:relative-motion:trajectory}{baan}; naarmate $\Delta t$ krimpt,
wordt de richting van de koorde die van de raaklijn.
\end{proof}

\section{De versnellingsvector}

\begin{definition}[Versnellingsvector]\label{def:g12:kinematics-2d:acceleration}
De \emph{versnellingsvector}\index{versnellingsvector} van $M$ is de
afgeleide van zijn \omterm{def:g12:kinematics-2d:velocity}{snelheidsvector},
$\vect a(t) = \bigl(v_x'(t),\, v_y'(t)\bigr) =
\bigl(x''(t),\, y''(t)\bigr)$, in \unit{m/s^2}: hoe snel, en in welke
richting, de \omterm{def:g12:kinematics-2d:velocity}{snelheidsvector} op dit ogenblik verandert.
\end{definition}

\begin{remark}[Versnellen zonder sneller te gaan]\label{rem:g12:kinematics-2d:turning}
$\vect a \neq \vect 0$ zodra de \omterm{def:g12:kinematics-2d:velocity}{snelheids}\emph{vector} verandert --- in
lengte \emph{of in richting}: een remmende auto versnelt (achterwaarts);
een auto die bij constante \omterm{def:g12:kinematics-2d:velocity}{snelheid} draait versnelt (zijwaarts). Dit is de
grootheid die \omterm{def:g10:inertia:force}{krachten} besturen
(\cref{ch:g11:forces-and-motion,ch:g12:newtons-laws}).
\end{remark}

\begin{method}[Een chronofotografie lezen]\label{met:g12:kinematics-2d:chrono}
Een \emph{chronofotografie}\index{chronofotografie} legt de posities
$M_0, M_1, M_2, \dots$ van een bewegend punt vast op gelijke
tijdsafstanden $\Delta t$.
\begin{enumerate}
  \item De afstand tussen de stippen verraadt de \omterm{def:g12:kinematics-2d:velocity}{snelheid}: gelijke
        afstanden, constante \omterm{def:g12:kinematics-2d:velocity}{snelheid}; groter wordend, versnellend;
        kleiner wordend, vertragend.
  \item De \omterm{def:g12:kinematics-2d:velocity}{snelheid} in $M_i$ is de symmetrische koorde,
        $\vect v_i \approx \vect{M_{i-1}M_{i+1}}/(2\,\Delta t)$, getekend
        in $M_i$, rakend aan het gestippelde spoor.
  \item Voor de versnelling in $M_i$: kopieer $\vect v_{i-1}$ en
        $\vect v_{i+1}$ naar $M_i$ en teken
        $\Delta\vect v = \vect v_{i+1} - \vect v_{i-1}$ van punt tot
        punt; dan is $\vect a_i \approx \Delta\vect v/(2\,\Delta t)$.
\end{enumerate}
\end{method}

\section{Rechtlijnige bewegingen}

Langs een rechte lijn volstaat één as: $x(t)$, $v(t) = x'(t)$ en
$a(t) = v'(t)$, alle drie getallen met een teken.

\begin{definition}[Eenparige rechtlijnige beweging]\label{def:g12:kinematics-2d:uniform}
Een beweging is \emph{eenparig rechtlijnig}\index{eenparige rechtlijnige
beweging} wanneer $\vect v$ een constante vector is: een rechte \omterm{def:g10:relative-motion:trajectory}{baan}, een
constante \omterm{def:g12:kinematics-2d:velocity}{snelheid}, $\vect a = \vect 0$. Dan is $x(t) = v\,t + x_0$.
\end{definition}

\begin{proposition}[Eenparig versnelde rechtlijnige beweging]\label{prop:g12:kinematics-2d:uarm}
\index{eenparig versnelde beweging}%
Beweegt een punt langs $Ox$ met een \emph{constante} versnelling $a$, en
vertrekt het op $t = 0$ vanuit de plaats $x_0$ met \omterm{def:g12:kinematics-2d:velocity}{snelheid} $v_0$, dan
geldt
\[
v(t) = a\,t + v_0,
\qquad
x(t) = \tfrac12\,a\,t^2 + v_0\,t + x_0 .
\]
\end{proposition}

\begin{proof}
$v$ is een primitieve van de constante $a$: $v(t) = a t + C$, en $t = 0$
legt $C = v_0$ vast. Op haar beurt is $x$ een primitieve van
$a t + v_0$: $x(t) = \tfrac12 a t^2 + v_0 t + C'$, met $C' = x_0$.
\end{proof}

\begin{example}[Remweg]\label{ex:g12:kinematics-2d:braking}
Een auto met \omterm{def:g12:kinematics-2d:velocity}{snelheid} $v_0$ remt met een constante \omterm{def:g12:mechanical-waves:delay}{vertraging} $a$
(versnelling $-a$). Hij staat stil zodra $v(t) = -a t + v_0 = 0$, op
$t_s = v_0/a$, en heeft dan
$d = -\tfrac12 a t_s^2 + v_0 t_s = v_0^2/(2a)$ afgelegd. Het kwadraat is
de krantenkop van de verkeersveiligheid: de \omterm{def:g12:kinematics-2d:velocity}{snelheid} verdubbelen
\emph{verviervoudigt} de remweg --- met $a = \qty{6.0}{m/s^2}$ is dat
\qty{16}{m} vanaf \qty{50}{km/h} en \qty{64}{m} vanaf \qty{100}{km/h}.
\end{example}

\section{Eenparige cirkelbeweging en het frenetstelsel}

\begin{definition}[Eenparige cirkelbeweging]\label{def:g12:kinematics-2d:ucm}
Een beweging is \emph{eenparig cirkelvormig}\index{eenparige
cirkelbeweging} wanneer de \omterm{def:g10:relative-motion:trajectory}{baan} een cirkel met straal $R$ is en de
\omterm{def:g12:kinematics-2d:velocity}{snelheid} $v$ constant is. De \emph{omlooptijd}\index{omlooptijd} $T$ is de
duur van één ronde, en de \emph{omloopfrequentie}\index{omloopfrequentie}
$f = 1/T$ (in \unit{Hz}) het aantal ronden per seconde:
$v = 2\pi R/T = 2\pi R f$.
\end{definition}

Constante \omterm{def:g12:kinematics-2d:velocity}{snelheid} --- en toch draait de \omterm{def:g12:kinematics-2d:velocity}{snelheidsvector}: er \emph{is} een
versnelling. Welke kant op, en hoe groot?

\begin{theorem}[Middelpuntzoekende versnelling]\label{thm:g12:kinematics-2d:centripetal}
\index{middelpuntzoekende versnelling}%
Bij een \omterm{def:g12:kinematics-2d:ucm}{eenparige cirkelbeweging} met straal $R$ en \omterm{def:g12:kinematics-2d:velocity}{snelheid} $v$ wijst de
versnelling --- \emph{middelpuntzoekend} genoemd --- van het bewegende
punt naar het middelpunt van de cirkel, met de constante norm
\[
a = \frac{v^2}{R}.
\]
\end{theorem}

\begin{proof}
Leg de assen in het middelpunt van de cirkel en stel $\omega = v/R$,
zodat de boog $v t$ de hoek $\omega t$ onderspant: de plaats is
$x(t) = R\cos(\omega t)$, $y(t) = R\sin(\omega t)$. Afleiden geeft
$\vect v = (-R\omega\sin(\omega t),\, R\omega\cos(\omega t))$, met
constante norm $R\omega = v$; nogmaals afleiden geeft
$\vect a = -\omega^2\,\vect{OM}$: tegengesteld aan de \omterm{def:g12:kinematics-2d:position}{plaatsvector} ---
dus naar het middelpunt --- met norm $\omega^2 R = v^2/R$.
\end{proof}

\begin{definition}[Frenetstelsel]\label{def:g12:kinematics-2d:frenet}
Het \emph{frenetstelsel}\index{frenetstelsel} in de huidige plaats van een
punt op een gekromde \omterm{def:g10:relative-motion:trajectory}{baan} is het paar loodrechte eenheidsvectoren
$(\vect{u_t}, \vect{u_n})$: $\vect{u_t}$ raakt aan de \omterm{def:g10:relative-motion:trajectory}{baan}, in de
bewegingsrichting; $\vect{u_n}$ staat er loodrecht op en wijst naar het
middelpunt van de cirkel met straal $R$ die de kromme daar het best
benadert.
\end{definition}

\begin{proposition}[Versnelling in het frenetstelsel]\label{prop:g12:kinematics-2d:frenetacc}
Langs elke vlakke \omterm{def:g10:relative-motion:trajectory}{baan} met plaatselijke straal $R$, doorlopen met \omterm{def:g12:kinematics-2d:velocity}{snelheid}
$v(t)$, geldt
\[
\vect a = a_t\,\vect{u_t} + a_n\,\vect{u_n},
\qquad a_t = \frac{\dd v}{\dd t},
\qquad a_n = \frac{v^2}{R}:
\]
de \emph{tangentiële versnelling}\index{tangentiële versnelling} $a_t$
verandert de \omterm{def:g12:kinematics-2d:velocity}{snelheid}; de \emph{normale versnelling}\index{normale
versnelling} $a_n$ draait haar.
\end{proposition}
\admitted

\begin{remark}[Waar het bewijs staat]\label{rem:g12:kinematics-2d:frenetproof}
We hebben de uitersten bewezen: zuivere $a_t$ op een rechte
(\cref{prop:g12:kinematics-2d:uarm}) en zuivere $a_n$ op een cirkel
(\cref{thm:g12:kinematics-2d:centripetal}); de algemene ontbinding wordt
afgeleid in het volume van bachelorjaar 1.
\end{remark}

\begin{example}[Remmen in een bocht]\label{ex:g12:kinematics-2d:bend}
Een auto rijdt een bocht met straal $R = \qty{50}{m}$ binnen met
$v = \qty{15}{m/s}$ terwijl hij remt met
$\dd v/\dd t = \qty{-3.0}{m/s^2}$: $a_t = \qty{-3.0}{m/s^2}$,
$a_n = 15^2/50 = \qty{4.5}{m/s^2}$, dus
$a = \sqrt{3.0^2 + 4.5^2} \approx \qty{5.4}{m/s^2}$, achterwaarts
gericht \emph{én} de bocht in. De banden moeten beide tegelijk leveren ---
en daarom remmen coureurs \emph{vóór} de bocht.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:1}
Geef voor $x(t) = 3.0\,t$ en $y(t) = 4.0\,t$ (\omterm{def:g10:orders-of-magnitude:unit}{meter}, seconde) de vector
$\vect v$, de \omterm{def:g12:kinematics-2d:velocity}{snelheid}, de \omterm{def:g10:relative-motion:trajectory}{baan} en de soort beweging.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:2}
Op een \omterm{met:g12:kinematics-2d:chrono}{chronofotografie} die om de \qty{0.10}{s} is genomen, liggen de
stippen op een rechte, gelijkmatig \qty{0.35}{m} uit elkaar. Benoem de
beweging en bereken de \omterm{def:g12:kinematics-2d:velocity}{snelheid}. Wat zouden groter wordende afstanden
aangeven?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:3}
Bepaal voor $x(t) = 2.0\,t^2$ langs $Ox$ (\omterm{def:g10:orders-of-magnitude:unit}{meter}, seconde) de functies
$v(t)$ en $a(t)$, benoem de soort beweging en bereken de \omterm{def:g12:kinematics-2d:velocity}{snelheid} op
$t = \qty{3.0}{s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:4}
Een paard van een draaimolen doorloopt een cirkel met straal \qty{4.5}{m},
één ronde om de \qty{8.0}{s}: bereken de \omterm{def:g12:kinematics-2d:ucm}{omloopfrequentie}, de \omterm{def:g12:kinematics-2d:velocity}{snelheid} en
de versnelling (norm en richting).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:kinematics-2d:5}
Een grafiek van $v(t)$ klimt rechtlijnig van \qty{2.0}{m/s} op $t = 0$
naar \qty{10.0}{m/s} op $t = \qty{4.0}{s}$. Lees de versnelling af, en
daarna de afgelegde weg (de oppervlakte onder de grafiek).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:6}
Een hogesnelheidstrein neemt een bocht met straal \qty{6.0}{km} bij
\qty{320}{km/h}. Bereken $a_n$ en vergelijk die met $g$; waarom moeten
hogesnelheidslijnen krappe bochten vermijden, zelfs perfect verkante?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:7}
Een wagentje op een rechte rail heeft $x(t) = -1.2\,t^2 + 8.0\,t + 3.0$
(\omterm{def:g10:orders-of-magnitude:unit}{meter}, seconde). Bepaal $v(t)$ en $a$, het tijdstip en de plaats waar het
omkeert, en beschrijf de beweging ervoor en erna.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:8}
Een \omterm{met:g12:kinematics-2d:chrono}{chronofotografie} langs een rechte, om de $\Delta t = \qty{0.20}{s}$,
geeft $x = 0$, $0.10$, $0.28$, $0.54$ en \qty{0.88}{m}. Schat de \omterm{def:g12:kinematics-2d:velocity}{snelheid}
in de drie binnenste stippen
(\cref{met:g12:kinematics-2d:chrono}), en daarna de versnelling; trek je
conclusie.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:9}
De Maan draait om de Aarde langs een bijna-cirkel met straal
\qty{3.84e8}{m} in \qty{27.3}{d}. Bereken haar \omterm{def:g12:kinematics-2d:velocity}{snelheid} en haar
\omterm{thm:g12:kinematics-2d:centripetal}{middelpuntzoekende versnelling}; vergelijk die laatste met $g$, wetend dat
de Maan ongeveer $60$ aardstralen ver staat
(\cref{ch:g10:universal-gravitation}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:10}
De trommel van een wasmachine met straal \qty{25}{cm} draait $1200$
omwentelingen per minuut. Bereken $f$, $T$, de \omterm{def:g12:kinematics-2d:velocity}{snelheid} aan de rand en de
versnelling aan de rand in veelvouden van $g$. Waarom blijft de was
tegen de wand geplakt terwijl het water vertrekt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:kinematics-2d:11}
Een auto met \qty{90}{km/h} remt met een constante $a = \qty{7.5}{m/s^2}$:
bereken de remtijd en de remweg, en schets $v(t)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:12}
Bereken voor $x(t) = R\cos(\omega t)$ en $y(t) = R\sin(\omega t)$ de
vector $\vect v$; ga na dat de \omterm{def:g12:kinematics-2d:velocity}{snelheid} de constante $R\omega$ is en dat
$\vect v \perp \vect{OM}$; bereken $\vect a$, toon aan dat die
$-\omega^2\,\vect{OM}$ is, en vind $a = v^2/R$ terug.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:13}
Een motorrijder rijdt een bocht met straal \qty{80}{m} binnen met
\qty{20}{m/s} en remt daarbij met \qty{2.5}{m/s^2}. Bereken $a_t$, $a_n$,
de norm van $\vect a$ en de hoek daarvan met de bewegingsrichting. De
banden houden in totaal tot \qty{7.0}{m/s^2}: blijft de rijder binnen de
grens?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:14}
Een metro versnelt vanuit rust met \qty{1.2}{m/s^2} tot \qty{20}{m/s},
rijdt \qty{40}{s} door en remt daarna met \qty{1.5}{m/s^2} tot stilstand.
Bereken de duur en de lengte van elke fase, de totale afstand en de
\omterm{def:g10:relative-motion:speed}{gemiddelde snelheid}. Schets $v(t)$ en controleer de afstand als
oppervlakte.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:kinematics-2d:15}
Een geostationaire \omterm{def:g10:universal-gravitation:satellite}{satelliet} cirkelt op een straal van \qty{4.22e7}{m} van
het middelpunt van de Aarde met \omterm{def:g12:kinematics-2d:ucm}{omlooptijd} \qty{86164}{s}. Bereken haar
\omterm{def:g12:kinematics-2d:velocity}{snelheid} en haar \omterm{thm:g12:kinematics-2d:centripetal}{middelpuntzoekende versnelling}. Wat levert die, en waarom
hangt de \omterm{def:g10:universal-gravitation:satellite}{satelliet} boven één vast punt op de evenaar

\end{exercise}

\section{Probleem: Het ongevalsrapport}

\begin{problem}[{Weekendopgave --- het ongevalsrapport: een dashcamfilm
beeldje voor beeldje gelezen, een remwet uit remsporen gereconstrueerd,
een rotonde die over grip getuigt, en een vonnis in kilometer per uur}]
\label{pb:g12:kinematics-2d:1}
Een auto ramt een bestelwagen die vanuit een zijstraat kwam; er raakt
niemand gewond, maar de verzekeraars zijn het oneens. De onderzoeker heeft
camerabeelden van bovenaf (één beeldje om de \qty{0.50}{s}), opgemeten
remsporen en de dashcam van de auto. De maximumsnelheid is
\qty{50}{km/h}.

\medskip
\noindent\textbf{Deel I --- De film lezen.}
De beeldjes vóór elke reactie geven, langs de rechte weg:
\begin{center}\small
\begin{tabular}{lccccc}
$t$ (\unit{s}) & 0.00 & 0.50 & 1.00 & 1.50 & 2.00\\
$x$ (\unit{m}) & 0.0 & 9.0 & 18.0 & 27.0 & 36.0
\end{tabular}
\end{center}
\begin{enumerate}
  \item Waarom moet het rapport eerst een stelsel en assen noemen? Welke
        worden hier gebruikt?
  \item Bereken de \omterm{def:g10:relative-motion:speed}{gemiddelde snelheid} op elk van de vier intervallen.
  \item Wat voor soort beweging is dit? Verantwoord dat uit de tabel.
  \item Geef de \omterm{def:g12:kinematics-2d:velocity}{snelheid} $v_0$ van de auto in \unit{m/s} en \unit{km/h}.
  \item Hoe groot is de versnelling gedurende deze twee seconden?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De remsporen.}
Bandproeven op dit droge asfalt geven een constante remvertraging
$a = \qty{6.0}{m/s^2}$; de remsporen zijn $d_b = \qty{27}{m}$ lang. Neem
$t = 0$ en $x = 0$ bij het begin van het remmen, met onbekende \omterm{def:g12:kinematics-2d:velocity}{snelheid}
$v_B$.
\begin{enumerate}[resume]
  \item Schrijf met primitieven $v(t)$ en $x(t)$ tijdens het remmen op.
  \item \omterm{def:g10:pressure:pressure}{Druk} de remtijd $t_s$ uit; toon aan dat $d_b = v_B^2/(2a)$.
  \item Leid $v_B$ af uit de remsporen. Klopt dat met deel I?
  \item De reactietijd is \qty{1.0}{s}: bereken de reactieafstand en de
        totale stopafstand.
  \item Leg vanuit vraag 7 uit waarom de \omterm{def:g12:kinematics-2d:velocity}{snelheid} verdubbelen de remweg
        verviervoudigt; bereken $d_b$ bij precies \qty{50}{km/h}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De rotonde getuigt.}
Eerder toont de dashcam dat de auto met constante \omterm{def:g12:kinematics-2d:velocity}{snelheid} een
rotondebaan met straal $R = \qty{12.5}{m}$ neemt, een kwartslag in
\qty{2.5}{s}. De weg kan hoogstens $a_n = \mu g$ leveren, met
$\mu = 0.70$.
\begin{enumerate}[resume]
  \item De \omterm{def:g12:kinematics-2d:velocity}{snelheid} is daar constant --- is de \omterm{def:g12:kinematics-2d:velocity}{snelheidsvector} dat ook?
        Versnelt de auto? Verklaar.
  \item Bereken de \omterm{def:g12:kinematics-2d:velocity}{snelheid} van de auto op de rotonde.
  \item Bereken haar \omterm{thm:g12:kinematics-2d:centripetal}{middelpuntzoekende versnelling} (norm en richting).
  \item Bereken de hoogste \omterm{def:g12:kinematics-2d:velocity}{snelheid} zonder slippen. Bleef de auto
        eronder?
  \item Geef de \omterm{def:g12:kinematics-2d:ucm}{omlooptijd} en de \omterm{def:g12:kinematics-2d:ucm}{omloopfrequentie} van een volledige ronde
        bij die \omterm{def:g12:kinematics-2d:velocity}{snelheid}.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het vonnis.}
De bestelwagen dook $D = \qty{41}{m}$ vóór de auto op, op het ogenblik dat
de bestuurder begon te reageren.
\begin{enumerate}[resume]
  \item Met hoeveel, in \unit{km/h} en in procent, lag de \omterm{def:g12:kinematics-2d:velocity}{snelheid} vóór
        het remmen boven de limiet?
  \item Bereken de \omterm{def:g12:kinematics-2d:velocity}{snelheid} van de auto op het ogenblik van de botsing.
  \item Doe de stopberekening over bij precies \qty{50}{km/h} (dezelfde
        reactie, hetzelfde remmen): staat de auto stil, en met welke
        marge?
  \item Geef de algemene formule voor de stopafstand $d(v)$; waarom komt
        de \omterm{def:g12:kinematics-2d:velocity}{snelheid} er twee keer in voor --- één keer lineair en één keer
        gekwadrateerd?
  \item Het vonnis, in één zin met de getallen erbij: welke \omterm{def:g12:kinematics-2d:velocity}{snelheid} reed
        de bestuurder, en zou het ongeval bij de limiet zijn gebeurd?
\end{enumerate}
\end{problem}
