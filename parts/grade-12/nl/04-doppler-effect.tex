\chapter{Het dopplereffect}\label{ch:g12:doppler-effect}

Een ambulance nadert met loeiende sirene en scheert voorbij --- en op dat
ogenblik glijdt de toon hoorbaar omlaag, terwijl de bestuurder nergens
aan heeft gezeten. De sirene is eerlijk; de \omterm{def:g10:signals-and-waves:wave}{golf} niet: beweging perst haar
vooruit samen en rekt haar achteruit uit. Dit hoofdstuk meet die
samenpersing en rijdt daarna met hetzelfde idee van sirenes naar
snelheidsradars, naar bloed in een slagader, en tot ver daarbuiten naar
een planeet die door een wiebeling in sterrenlicht wordt verraden.

\section{Het effect}

\begin{definition}[Dopplereffect]\label{def:g12:doppler-effect:doppler}
Het \emph{dopplereffect}\index{dopplereffect} is de verandering van de
ontvangen \omterm{def:g10:signals-and-waves:frequency}{frequentie} van een \omterm{def:g10:signals-and-waves:wave}{golf} wanneer bron en ontvanger ten opzichte
van elkaar bewegen: de ontvangen \omterm{def:g10:signals-and-waves:frequency}{frequentie} $f'$ is hoger dan de
uitgezonden \omterm{def:g10:signals-and-waves:frequency}{frequentie} $f$ terwijl ze elkaar naderen, lager terwijl ze
zich verwijderen, en alleen gelijk aan $f$ zolang hun afstand op dat
ogenblik niet verandert.
\end{definition}

\begin{remark}[Toonhoogte, niet luidheid]\label{rem:g12:doppler-effect:pitch}
Er veranderen twee dingen wanneer de ambulance passeert. De
\emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{luidheid}} zwelt aan en sterft weg doordat de \omterm{def:g10:signals-and-waves:wave}{golf} zich met de
afstand uitspreidt (\cref{ch:g12:sound-acoustics}); de \emph{\omterm{def:g12:sound-acoustics:pitch-loudness}{toonhoogte}}
verschuift door de snelheid van naderen of verwijderen en zakt bij het
passeren abrupt. De \omterm{def:g12:sound-acoustics:pitch-loudness}{luidheid} zegt ``hoe ver''; de \omterm{def:g12:sound-acoustics:pitch-loudness}{toonhoogte} zegt ``hoe
snel''.
\end{remark}

\section{Een bewegende bron: opeengepakte golffronten}

Het kernbeeld: zodra een top de bron verlaat, is hij eigendom van het
midden. Elke top dijt uit als een cirkel met de golfsnelheid $v$
(\cref{ch:g12:mechanical-waves}), met als middelpunt het punt waar hij is
\emph{uitgezonden} --- het midden weet niet en kan het niets schelen dat
de bron intussen verder is gereden.

\begin{proposition}[Bewegende bron]\label{prop:g12:doppler-effect:moving-source}
Een bron zendt toppen uit met \omterm{def:g10:signals-and-waves:period}{periode} $T$ (\omterm{def:g10:signals-and-waves:frequency}{frequentie} $f = 1/T$) terwijl
ze rechtlijnig met snelheid $v_s < v$ door een midden beweegt waarin de
\omterm{def:g10:signals-and-waves:wave}{golf} met snelheid $v$ reist. Een waarnemer die in het midden in rust is,
ontvangt
\[
\lambda' = (v \mp v_s)\,T, \qquad f' = f\,\frac{v}{v \mp v_s},
\]
met de bovenste tekens voor een naderende bron en de onderste voor een
zich verwijderende.
\end{proposition}

\begin{proof}
Tussen twee toppen legt de eerste $vT$ af terwijl de bron $v_s T$
opschuift. Vooruit wordt de tweede top dus $v_s T$ \emph{dichter} bij de
eerste uitgezonden: $\lambda' = (v - v_s)T$; achteruit komt diezelfde
$v_s T$ erbij. De toppen scheren daarna met de snelheid $v$ van het midden
langs een rustende waarnemer, één om de $\lambda'/v$ seconden, dus
$f' = v/\lambda' = f v/(v \mp v_s)$.
\end{proof}

\begin{example}[De ambulance, in getallen]\label{ex:g12:doppler-effect:siren}
Een sirene op $f = \qty{700}{Hz}$ nadert met $v_s = \qty{25}{m/s}$; neem
$v = \qty{340}{m/s}$ in lucht (\cref{ch:g12:sound-acoustics}). Naderend:
$f' = 700 \times 340/315 \approx \qty{756}{Hz}$; zich verwijderend:
$f' = 700 \times 340/365 \approx \qty{652}{Hz}$. Het passeren laat de
\omterm{def:g12:sound-acoustics:pitch-loudness}{toonhoogte} met \qty{104}{Hz} zakken --- een verhouding
$\approx 1.16$, ongeveer tweeënhalve halve toon, onmiskenbaar voor elk
oor.
\end{example}

\begin{remark}[Wat beweging niet verandert]\label{rem:g12:doppler-effect:speed-unchanged}
De \omterm{def:g10:signals-and-waves:wave}{golf} reist nog altijd met $v$: alleen het midden legt de snelheid vast
(\cref{ch:g12:mechanical-waves}). Beweging van de bron perst
\emph{\omterm{def:g12:mechanical-waves:wavelength}{golflengten}} samen, nooit snelheden. En die samenpersing kent een
grens: als $v_s \to v$ stapelen de toppen vooruit op elkaar en gaat
$f' \to \infty$ --- de file van \omterm{def:g12:mechanical-waves:wavefront}{golffronten} die een supersonisch vliegtuig
als schokgolf achter zich aan sleept.
\end{remark}

\section{Een bewegende waarnemer, en de kortere weg bij trage beweging}

\begin{proposition}[Bewegende waarnemer]\label{prop:g12:doppler-effect:moving-observer}
Een waarnemer beweegt met snelheid $v_o$ recht naar (of recht weg van) een
bron die in het midden in rust is. De ontvangen \omterm{def:g10:signals-and-waves:frequency}{frequentie} is
\[ f' = f\left(1 \pm \frac{v_o}{v}\right) \quad (+\ \text{naderend},\ -\ \text{verwijderend}). \]
\end{proposition}

\begin{proof}
Het golfpatroon zelf blijft ongestoord: toppen op afstand $\lambda = vT$
reizen met $v$. Een waarnemer die erop af rent, ontmoet de toppen met de
relatieve snelheid $v + v_o$, één om de $\lambda/(v + v_o)$ seconden:
$f' = (v + v_o)/\lambda = f(1 + v_o/v)$. Wie wegrent, krijgt $v - v_o$.
\end{proof}

\begin{definition}[Radiale snelheid]\label{def:g12:doppler-effect:radial}
De \emph{radiale snelheid}\index{radiale snelheid} $v_r$ van een bron ten
opzichte van een waarnemer is de component van hun relatieve snelheid
langs de verbindingslijn --- het tempo waarin de afstand krimpt of groeit.
Alleen die component verschuift de \omterm{def:g10:signals-and-waves:frequency}{frequentie}; beweging \emph{dwars} op
de zichtlijn geeft in deze orde geen verschuiving.
\end{definition}

\begin{proposition}[Trage beweging: één formule voor iedereen]\label{prop:g12:doppler-effect:slow}
Wanneer de \omterm{def:g12:doppler-effect:radial}{radiale snelheid} klein is, $v_r \ll v$, maakt het niet langer
uit wie er beweegt:
\[
\frac{\Delta f}{f} = \frac{f' - f}{f} \approx +\frac{v_r}{v} \ \text{(naderen)},
\qquad
\frac{\Delta f}{f} \approx -\frac{v_r}{v} \ \text{(verwijderen)}.
\]
\end{proposition}

\begin{proof}
De formules voor de waarnemer zijn al precies $1 \pm v_o/v$. Schrijf voor
de bron $x = v_s/v$: de identiteit
$\frac{1}{1 - x} = 1 + x + \frac{x^2}{1 - x}$ laat zien dat $f'/f$ van
$1 + x$ afwijkt met een term van de orde $x^2$ --- voor $x = 0.1$ een
correctie van $1\%$ op een verschuiving van $10\%$. Tot op de eerste orde
vallen alle vier de formules samen tot $1 \pm v_r/v$.
\end{proof}

\begin{method}[Een passage aflezen]\label{met:g12:doppler-effect:pass}
Een opname van een passerende bron toont twee plateaus: $f_1$ (naderen) en
$f_2$ (verwijderen). Dan geldt
\[
f = \frac{2 f_1 f_2}{f_1 + f_2}, \qquad
v_s = v\,\frac{f_1 - f_2}{f_1 + f_2}.
\]
Immers $f_1 + f_2 = \frac{2fv^2}{v^2 - v_s^2}$ en
$f_1 - f_2 = \frac{2fv\,v_s}{v^2 - v_s^2}$: deel om $v_s$ te krijgen, en
bereken $2f_1 f_2/(f_1 + f_2)$ voor $f$. Twee frequentieaflezingen leveren
zowel de ware \omterm{def:g12:sound-acoustics:pitch-loudness}{toonhoogte} als de snelheid op --- zonder stopwatch en zonder
meetlint.
\end{method}

\section{Echo's die snelheid meten}

Kaats een \omterm{def:g10:signals-and-waves:wave}{golf} terug van een bewegend doel en het \omterm{def:g12:doppler-effect:doppler}{dopplereffect} slaat twee
keer toe.

\begin{proposition}[Terugkaatsing verdubbelt de verschuiving]\label{prop:g12:doppler-effect:double}
Een \omterm{def:g10:signals-and-waves:wave}{golf} met \omterm{def:g10:signals-and-waves:frequency}{frequentie} $f$ wordt op een reflector gestuurd die frontaal
nadert met snelheid $u \ll v$. De \omterm{rem:g10:signals-and-waves:honest}{echo} komt terug met
\[ f' = f\,\frac{v + u}{v - u}, \qquad \text{dus} \qquad \Delta f \approx \frac{2u}{v}\,f . \]
\end{proposition}

\begin{proof}
Twee verschuivingen achter elkaar. Als bewegende \emph{waarnemer}
ontvangt de reflector $f(1 + u/v)$; door uit te zenden wat hij ontvangt,
is hij nu een bewegende \emph{bron}, dus komt de \omterm{rem:g10:signals-and-waves:honest}{echo} aan met
$f' = f(1 + u/v)/(1 - u/v) \approx f(1 + 2u/v)$ voor $u \ll v$.
\end{proof}

\begin{example}[De snelheidsradar]\label{ex:g12:doppler-effect:radar}
Een verkeersradar zendt microgolven uit op $f = \qty{24.15}{GHz}$
($v = c = \qty{3.00e8}{m/s}$) en legt de \omterm{rem:g10:signals-and-waves:honest}{echo} over de uitgaande \omterm{def:g10:signals-and-waves:wave}{golf}; de
zweving (\cref{ch:g12:sound-acoustics}) laat rechtstreeks
$\Delta f = 2uf/c \approx \qty{161}{Hz}$ per \unit{m/s} autosnelheid
klinken --- een moeiteloos meetbare hoorbare \omterm{def:g10:signals-and-waves:frequency}{frequentie}, uitgesneden uit
een draaggolf van \qty{24}{GHz}. De factor $2$ is niet optioneel: wie hem
vergeet, vleit elke bestuurder met de helft.
\end{example}

\begin{example}[Doppler-echografie]\label{ex:g12:doppler-effect:ultrasound}
Een medische sonde stuurt \omterm{def:g10:signals-and-waves:ultrasound}{ultrageluid} van $f = \qty{5.0}{MHz}$ in een
slagader ($v \approx \qty{1540}{m/s}$ in weefsel); rode bloedcellen
kaatsen het terug. Bloed met $u = \qty{0.50}{m/s}$ levert
$\Delta f = 2 \times 0.50 \times \num{5.0e6}/1540 \approx \qty{3.2}{kHz}$
op --- een \emph{hoorbare} zweving: de cardioloog hoort het bloed bij elke
hartslag letterlijk versnellen, en meet daarna $u$ uit $\Delta f$.
\end{example}

\section{Doppler in sterrenlicht}

Licht is een \omterm{def:g10:signals-and-waves:wave}{golf} (\cref{ch:g12:light-as-wave}) en zijn \omterm{def:g12:sound-acoustics:timbre}{spectrum} is
gestreept met scherpe lijnen op \omterm{def:g12:mechanical-waves:wavelength}{golflengten} die door elk scheikundig
element worden vastgelegd (\cref{ch:g10:light-spectra}) --- een in het
laboratorium geijkte maatlat, afgedrukt op elke ster.

\begin{definition}[Roodverschuiving en blauwverschuiving]\label{def:g12:doppler-effect:redshift}
Wanneer de \omterm{def:g10:light-spectra:emission}{spectraallijnen} van een bron op \emph{langere} \omterm{def:g12:mechanical-waves:wavelength}{golflengten}
verschijnen dan in het laboratorium, is het \omterm{def:g12:sound-acoustics:timbre}{spectrum}
\emph{roodverschoven}\index{roodverschuiving}; verschijnen ze op
\emph{kortere}, dan is het \emph{blauwverschoven}\index{blauwverschuiving}
--- verschoven naar het rode of het blauwe uiteinde van het \omterm{def:g10:light-spectra:wavelength}{zichtbare
spectrum}.
\end{definition}

\begin{proposition}[Dopplerverschuiving van licht]\label{prop:g12:doppler-effect:light}
Voor een bron die zich met \omterm{def:g12:doppler-effect:radial}{radiale snelheid} $v_r \ll c$ verwijdert, wordt
elke \omterm{def:g12:mechanical-waves:wavelength}{golflengte} waargenomen als uitgerekt met
\[ \frac{\Delta \lambda}{\lambda} = \frac{\lambda' - \lambda}{\lambda} \approx \frac{v_r}{c} \quad \text{(roodverschuiving)}, \]
en met evenveel samengeperst, $\Delta\lambda/\lambda \approx -v_r/c$, voor
een naderende bron (\omterm{def:g12:doppler-effect:redshift}{blauwverschuiving}).
\end{proposition}
\admitted

\begin{remark}[Waar de eerlijke formule staat]\label{rem:g12:doppler-effect:honest}
Licht heeft geen tussenstof nodig, dus valt de afleiding voor \omterm{def:g10:signals-and-waves:sound}{geluid} niet
te kopiëren --- er bestaat geen ``in rust in het midden''. De exacte
formule komt uit de speciale relativiteitstheorie, later dit jaar
(\cref{ch:g12:special-relativity}); de correcties zijn van de orde
$(v_r/c)^2$ en verwaarloosbaar voor vliegtuigen, sterren en nabije
melkwegstelsels. Hieronder gebruiken we de formule voor trage beweging met
een gerust hart.
\end{remark}

\begin{example}[De vlucht van een ster wegen]\label{ex:g12:doppler-effect:vega}
In het \omterm{def:g12:sound-acoustics:timbre}{spectrum} van de heldere ster Wega wordt de waterstoflijn met
laboratoriumgolflengte \qty{656.30}{nm} gemeten op \qty{656.27}{nm}:
$\Delta\lambda = \qty{-0.03}{nm}$, dus een \omterm{def:g12:doppler-effect:redshift}{blauwverschuiving}, en Wega
nadert met $v_r = c\,|\Delta\lambda|/\lambda \approx \qty{1.4e4}{m/s}$ ---
\qty{14}{km/s}, afgelezen uit een verschuiving van één deel op
twintigduizend.
\end{example}

\begin{example}[Wiebelende sterren: dubbelsterren en exoplaneten]\label{ex:g12:doppler-effect:wobble}
Wanneer twee sterren om elkaar heen draaien, zwaaien hun lijnen \omterm{def:g10:signals-and-waves:period}{periodiek}
naar blauw en naar rood --- een \emph{spectroscopische
dubbelster}\index{spectroscopische dubbelster}, waarvan de banen worden
gemeten zonder het paar ooit los te zien. Een planeet doet hetzelfde met
haar ster, maar veel zwakker: beide draaien om hun gemeenschappelijke
\omterm{def:g11:forces-and-motion:com}{massamiddelpunt}, zodat de \omterm{def:g12:doppler-effect:radial}{radiale snelheid} van de ster met enkele
tientallen \unit{m/s} slingert --- een verschuiving
$\Delta\lambda/\lambda \sim 10^{-7}$. Die opsporen is de manier waarop de
eerste planeet rond een zonachtige ster is gevonden; de weekendopgave doet
die ontdekking over, met getallen en al.
\end{example}

\begin{remark}[Roodverschuiving van melkwegstelsels]\label{rem:g12:doppler-effect:expansion}
De spectra van verre melkwegstelsels zijn alle \emph{rood}verschoven, en
hoe verder weg, hoe sterker: de afstanden in het heelal rekken uit. Voor
nabije stelsels leest $v_r = c\,\Delta\lambda/\lambda$ de vluchtsnelheid
af; voor de verste groeien de verschuivingen boven wat de formule voor
trage beweging eerlijk aankan, en wordt de boekhouding hervat vanaf
\cref{ch:g12:special-relativity} en in de universitaire volumes.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:1}
De sirene van een brandweerwagen zendt uit op \qty{700}{Hz} en de wagen
rijdt \qty{25}{m/s} ($v = \qty{340}{m/s}$). Bereken de \omterm{def:g10:signals-and-waves:frequency}{frequentie} die een
voetganger hoort (a) ervoor; (b) erachter. (c) Wat hoort de bestuurder?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:2}
Een vriend beweert: ``de \omterm{def:g12:sound-acoustics:pitch-loudness}{toonhoogte} zakt terwijl de ambulance wegrijdt
doordat het \omterm{def:g10:signals-and-waves:sound}{geluid} met de afstand zwakker wordt.'' Ontwar de twee effecten
die hier worden verward, en zeg waarvan elk afhangt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:3}
Een fietser rijdt met \qty{6.0}{m/s} recht op een stilstaande sirene af
die op \qty{700}{Hz} uitzendt. Welke \omterm{def:g10:signals-and-waves:frequency}{frequentie} hoort hij? En als hij er
recht vandaan rijdt? Welke formule geldt, en waarom niet die voor een
bewegende bron?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:4}
Een treinhoorn zendt uit op \qty{400}{Hz} terwijl de trein \qty{30}{m/s}
rijdt. Bereken de \omterm{def:g12:mechanical-waves:wavelength}{golflengte} vóór de trein, erachter en in rust. Welke
waarnemer ontvangt welke?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:doppler-effect:5}
Een telefoon die een toon van \qty{440}{Hz} speelt, wordt door een
hardloper met \qty{5.0}{m/s} meegedragen. Schat met de formule voor trage
beweging de verschuiving die iemand hoort naar wie de hardloper toe rent.
Vergelijk met een halve toon (ongeveer $6\%$ in \omterm{def:g10:signals-and-waves:frequency}{frequentie}): zou een
musicus het merken?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:6}
Een vleermuis vliegt met \qty{6.0}{m/s} recht op een muur af en zendt uit
op \qty{50.0}{kHz}. Toon aan dat de \omterm{rem:g10:signals-and-waves:honest}{echo} die ze hoort terugkomt met
$f' = f(v + v_b)/(v - v_b)$, en bereken daarna $f'$ en de verschuiving.
Waarom is dit de formule van de snelheidsradar in vermomming?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:7}
Een snelheidsradar op \qty{24.15}{GHz} meet een zweving van \qty{4.0}{kHz}
van een naderende auto. Bepaal de snelheid van de auto in \unit{m/s} en in
\unit{km/h}. Welke zweving zou een auto van precies \qty{50}{km/h} geven?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:8}
Een dopplersonde van \qty{5.0}{MHz} die langs een slagader is gericht
($v = \qty{1540}{m/s}$ in weefsel) hoort een zweving van \qty{2.6}{kHz}.
Bepaal de bloedsnelheid. Waarom moet de sonde \emph{langs} de stroming
worden gericht en niet loodrecht erop?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:doppler-effect:9}
Neem $f = \qty{1000}{Hz}$, $v = \qty{340}{m/s}$ en een naderingssnelheid

\section{Probleem: De jacht op een exoplaneet}

\begin{problem}[{Weekendopgave --- de jacht op een exoplaneet: een
treinhoorn ijkt het oor, een snelheidsradar oefent de dubbele
verschuiving, en zondagnacht heeft het spectrum van een ster, wiebelend
met één deel op vijf miljoen, een planeet gewogen die nog nooit iemand
heeft gezien}]
\label{pb:g12:doppler-effect:1}
Een sterrenkundige vereniging besteedt een weekend aan één idee --- de
dopplerverschuiving --- eerst op Aarde geoefend en daarna op een
zonachtige ster gericht. Gegevens: $v = \qty{340}{m/s}$ voor \omterm{def:g10:signals-and-waves:sound}{geluid} in
lucht, $c = \qty{3.00e8}{m/s}$, $G = \qty{6.67e-11}{N.m^2/kg^2}$; de ster
heeft massa $M = \qty{2.0e30}{kg}$. Aangenomen, uit de mechanica van
cirkelvormige omloopbanen (\cref{ch:g12:satellites-kepler}): een planeet
met massa $m$ op een cirkelvormige \omterm{def:g10:relative-motion:trajectory}{baan} met straal $r$ heeft \omterm{def:g10:signals-and-waves:period}{periode} en
snelheid gegeven door $r^3 = \frac{G M P^2}{4\pi^2}$ en
$V = \frac{2\pi r}{P}$, en de ster, die om het gemeenschappelijke
\omterm{def:g11:forces-and-motion:com}{massamiddelpunt} draait, wiebelt met snelheid $K = \frac{m}{M}\,V$.

\medskip
\noindent\textbf{Deel I --- Vrijdag: de spoorwegovergang.}
Een trein passeert met constante snelheid en de hoorn aan; een telefoon
neemt $f_1 = \qty{471}{Hz}$ op terwijl hij nadert en $f_2 = \qty{411}{Hz}$
nadat hij voorbij is.
\begin{enumerate}
  \item Verklaar met het beeld van de opeengepakte \omterm{def:g12:mechanical-waves:wavefront}{golffronten} waarom
        $f_1 > f_2$, ook al verandert de hoorn nooit.
  \item Schrijf de twee vergelijkingen op die $f_1$ en $f_2$ verbinden met
        de ware \omterm{def:g10:signals-and-waves:frequency}{frequentie} $f$ en de treinsnelheid $u$.
  \item Toon aan dat $u = v\,\dfrac{f_1 - f_2}{f_1 + f_2}$ en bereken die
        in \unit{m/s} en \unit{km/h}.
  \item Toon aan dat $f = \dfrac{2 f_1 f_2}{f_1 + f_2}$ en bereken die.
        Waarom ligt $f$ net \emph{onder} het gemiddelde van $f_1$ en
        $f_2$?
  \item Bereken de \omterm{def:g12:mechanical-waves:wavelength}{golflengten} vóór en achter de trein, en de \omterm{def:g12:mechanical-waves:wavelength}{golflengte}
        in rust.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Zaterdag: de snelheidsradar.}
Een verkeerspatrouille demonstreert haar \omterm{met:g10:signals-and-waves:echo}{radar}: $f = \qty{24.125}{GHz}$,
en het toestel leest de zweving tussen de uitgaande \omterm{def:g10:signals-and-waves:wave}{golf} en de \omterm{rem:g10:signals-and-waves:honest}{echo} af.
\begin{enumerate}[resume]
  \item Leg uit waarom de \omterm{rem:g10:signals-and-waves:honest}{echo} van een rijdende auto \emph{twee keer}
        wordt verschoven --- benoem de rol die de auto in elke
        verschuiving speelt.
  \item Toon aan dat de zweving voor een auto met snelheid $u \ll c$
        gelijk is aan $\Delta f \approx \dfrac{2u}{c}\,f$.
  \item Een auto levert $\Delta f = \qty{3.55}{kHz}$ op: bereken $u$.
  \item Zet dat om naar \unit{km/h} en vergelijk met de limiet van
        \qty{80}{km/h} die daar geldt.
  \item Hoeveel \omterm{def:g10:signals-and-waves:frequency}{hertz} zweving komt overeen met \qty{1}{km/h}?
        Becommentarieer: waarom is zo'n minieme relatieve verschuiving van
        een \omterm{def:g10:signals-and-waves:wave}{golf} van \qty{24}{GHz} toch makkelijk te meten?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Zondag: de ster die wiebelt.}
De vereniging haalt de gemeten \omterm{def:g12:doppler-effect:radial}{radiale snelheid} van een zonachtige ster
op, nacht na nacht, hieronder uitgezet.

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis, width=8.8cm, height=4.6cm,
    xlabel={$t$ (dagen)}, ylabel={$v_r$ (\unit{m/s})},
    xmin=0, xmax=8.4, ymin=-72, ymax=72,
    xtick={0,2.1,4.2,6.3,8.4},
    xticklabels={$0$,$2.1$,$4.2$,$6.3$,$8.4$}, ytick={-55,0,55}]
  \addplot[dashed, omExo, domain=0:8.4, samples=2] {55};
  \addplot[dashed, omExo, domain=0:8.4, samples=2] {-55};
  \addplot[omDef, thick, domain=0:8.4, samples=140] {55*sin(360*x/4.2)};
\end{axis}
\end{tikzpicture}

{\small De \omterm{def:g12:doppler-effect:radial}{radiale snelheid} van de ster over twaalf nachten: een zuivere
sinus --- de handtekening van een lichaam op een cirkelvormige \omterm{def:g10:relative-motion:trajectory}{baan}, van
opzij gezien.}
\end{omfigure}

\begin{enumerate}[resume]
  \item Wat wordt er nacht na nacht werkelijk gemeten in het
        \emph{\omterm{def:g12:sound-acoustics:timbre}{spectrum}} van de ster? Noem de betrekking waarmee dat in
        $v_r$ wordt omgezet.
  \item Lees de \omterm{def:g10:signals-and-waves:amplitude}{amplitude} $K$ en de \omterm{def:g10:signals-and-waves:period}{periode} $P$ van de wiebeling van de
        kromme af.
  \item Bereken de bijbehorende slingering $\Delta\lambda$ van een lijn op
        \qty{550}{nm}, en $\Delta\lambda/\lambda$. Vergelijk met deel I:
        hoeveel moeilijker is deze meting dan die aan de trein?
  \item Waarom laat een onzichtbare planeet de ster überhaupt wiebelen?
        Wat geeft de sinusvorm aan over de \omterm{def:g10:relative-motion:trajectory}{baan}?
  \item Waarom meet deze methode alleen het \emph{radiale} deel van de
        beweging van de ster, en wat zou je zien als de \omterm{def:g10:relative-motion:trajectory}{baan} ons frontaal
        toegekeerd was?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Zondagnacht: het onzichtbare wegen.}
Neem de \omterm{def:g10:relative-motion:trajectory}{baan} cirkelvormig en van opzij gezien, zodat $K$ de volledige
baansnelheid van de ster is.
\begin{enumerate}[resume]
  \item Bereken uit $P$ en de aangenomen betrekking
        $r^3 = G M P^2/(4\pi^2)$ de baanstraal $r$ van de planeet.
        Vergelijk die met de afstand Aarde--Zon, \qty{1.5e11}{m}.
  \item Bereken de baansnelheid $V = 2\pi r/P$ van de planeet.
  \item Bereken uit $K = (m/M)\,V$ de massa $m$ van de planeet.
  \item Vergelijk $m$ met Jupiter ($\qty{1.9e27}{kg}$) en met de Aarde
        ($\qty{6.0e24}{kg}$), en $r$ met de afstand Zon--Mercurius
        ($\qty{5.8e10}{m}$). Wat voor wereld is dit?
  \item Zou de ware massa groter of kleiner zijn dan jouw waarde als de
        \omterm{def:g10:relative-motion:trajectory}{baan} gekanteld was in plaats van van opzij gezien? Besluit in één
        zin: wat heeft een periodieke verschuiving van één deel op vijf
        miljoen zojuist opgeleverd?
\end{enumerate}
\end{problem}
