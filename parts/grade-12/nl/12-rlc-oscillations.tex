\chapter{Vrije elektrische trillingen: RLC}\label{ch:g12:rlc-oscillations}

Draai aan de zware knop van een zolderradio en de zenders trekken langs de
naald voorbij: een stem, violen, geruis, weer een stem. Achter het paneel zit
geen motor en geen computer --- een \omterm{def:g12:rc-rl-circuits:inductor}{spoel}, een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} waarvan de knop de
in elkaar geschoven platen draait, en een stuk antennedraad. Dit hoofdstuk
verbindt de twee energietanks van het vorige hoofdstuk
(\cref{ch:g12:rc-rl-circuits}) kop aan kop en vindt de snelste slinger die
ooit is gebouwd: lading die heen en weer zwaait, tot miljoenen keren per
seconde, in een tempo dat de knop kiest.

\section{Een elektrische slinger: de LC-kring}

Laad een \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} $C$ tot een \omterm{def:g11:circuits-and-power:voltage}{spanning} $U_0$ --- zijn platen dragen dan
$\pm Q_0$ met $Q_0 = C U_0$ --- en schakel hem op $t = 0$ op een
\emph{ideale} \omterm{def:g12:rc-rl-circuits:inductor}{spoel} $L$ met \omterm{def:g11:circuits-and-power:resistance}{weerstand} nul. Nergens in de kring een bron: wat
er nu ook gebeurt, de schakeling doet het zelf.

ideale \omterm{def:g12:rc-rl-circuits:inductor}{spoel}. Door $K$ te sluiten laat je de lading door de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} wegstromen
--- en de \omterm{def:g12:rc-rl-circuits:inductor}{spoel}, die verandering haat, laat haar niet stoppen.}
\end{omfigure}

\begin{proposition}[De LC-vergelijking]\label{prop:g12:rlc-oscillations:equation}
\index{LC-kring}
In de ideale \omterm{prop:g12:rlc-oscillations:equation}{LC-kring} voldoet de lading $q(t)$ van de bovenste plaat aan
\[
L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0,
\qquad\text{oftewel}\qquad
\frac{d^2q}{dt^2} = -\frac{1}{LC}\,q .
\]
\end{proposition}

\begin{proof}
Rond de kring tellen de \omterm{def:g11:circuits-and-power:voltage}{spanningen} tot nul op
(\cref{prop:g11:circuits-and-power:laws}): $u_L + u_C = 0$. De \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}
geeft $u_C = q/C$ en de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} $u_L = L\,di/dt$
(\cref{def:g12:rc-rl-circuits:inductor}), met $i = dq/dt$ als de lading die op
de bovenste plaat aankomt
(\cref{prop:g12:rc-rl-circuits:cdudt}); vul in.
\end{proof}

\begin{theorem}[Vrije trillingen van de LC-kring]\label{thm:g12:rlc-oscillations:oscillation}
\index{elektrische trillingen}
Losgelaten op $t = 0$ met $q(0) = Q_0$ en $i(0) = 0$ trilt de kring:
\[
q(t) = Q_0 \cos\!\left(\frac{2\pi t}{T_0}\right),
\qquad
i(t) = -I_0 \sin\!\left(\frac{2\pi t}{T_0}\right),
\qquad
T_0 = 2\pi\sqrt{LC},
\]
met piekstroom $I_0 = 2\pi Q_0/T_0 = Q_0/\sqrt{LC}$.
\end{theorem}

\begin{proof}
Controleer door invullen. Afleiden geeft
$i = dq/dt = -\frac{2\pi}{T_0} Q_0 \sin(2\pi t/T_0)$ --- de opgegeven $i(t)$
--- en nog eens $\frac{d^2q}{dt^2} = -(\frac{2\pi}{T_0})^2 q$, wat gelijk is
aan $-q/(LC)$ precies wanneer $(2\pi/T_0)^2 = 1/(LC)$, oftewel
$T_0 = 2\pi\sqrt{LC}$. En $q(0) = Q_0$, $i(0) = 0$. Dat geen \emph{andere}
functie bij de vergelijking en de start past, wordt toegegeven, net als bij de
veer (\cref{rem:g12:oscillators-and-time:uniqueness}).
\end{proof}

\begin{definition}[Eigenperiode en eigenfrequentie]\label{def:g12:rlc-oscillations:period}
De \emph{eigenperiode}\index{eigenperiode!van een LC-kring} van een \omterm{prop:g12:rlc-oscillations:equation}{LC-kring}
is $T_0 = 2\pi\sqrt{LC}$; haar
\emph{eigenfrequentie}\index{eigenfrequentie!van een LC-kring} is
\[
f_0 = \frac{1}{T_0} = \frac{1}{2\pi\sqrt{LC}} .
\]
Controle van de dimensies: $\qty{1}{H} = \qty{1}{V.s/A}$ en
$\qty{1}{F} = \qty{1}{C/V}$, dus draagt $LC$ de \omterm{def:g10:orders-of-magnitude:unit}{eenheid}
$\unit{s} \times \unit{C/A} = \unit{s^2}$ --- $\sqrt{LC}$ is een tijd, zoals
het hoort.
\end{definition}

\begin{example}[Eerste getallen]\label{ex:g12:rlc-oscillations:firstnumbers}
$L = \qty{10}{mH}$ met $C = \qty{1.0}{\micro F}$:
$\sqrt{LC} = \qty{1.0e-4}{s}$, dus $T_0 = \qty{0.63}{ms}$ en
$f_0 \approx \qty{1.6}{kHz}$ --- een hoorbare toon. $L = \qty{1.0}{mH}$ met
$C = \qty{100}{pF}$: $T_0 = \qty{2.0}{\micro s}$,
$f_0 \approx \qty{0.50}{MHz}$ --- een radiofrequentie. Alleen het
\emph{product} $LC$ legt het tempo vast.
\end{example}

\begin{remark}[Een kwart periode uit de pas]\label{rem:g12:rlc-oscillations:quarter}
De stroom loopt een kwart \omterm{def:g12:oscillators-and-time:oscillator}{periode} uit \omterm{def:g12:oscillators-and-time:phase}{fase} met de lading: $i = 0$ wanneer
$q = \pm Q_0$ (platen vol, stroming keert om) en $i = \pm I_0$ wanneer
$q = 0$ (platen leeg, stroming op volle \omterm{def:g10:inertia:force}{kracht}) --- precies de slinger,
stilstaand in de uitersten en het snelst onderaan.
\end{remark}

\section{De energiewals}

\begin{proposition}[Behoud van de opgeslagen energie]\label{prop:g12:rlc-oscillations:energy}
\index{energie!van een LC-kring}
In de ideale \omterm{prop:g12:rlc-oscillations:equation}{LC-kring} is de totale opgeslagen \omterm{def:g10:energy-conservation:energy}{energie}
(\cref{prop:g12:rc-rl-circuits:energy})
\[
E \;=\; E_C + E_L \;=\; \frac{q^2}{2C} + \tfrac12\,L i^2
\;=\; \frac{Q_0^2}{2C} \;=\; \tfrac12\,L I_0^2
\]
constant: volledig in de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} wanneer $q = \pm Q_0$, volledig in de
\omterm{def:g12:rc-rl-circuits:inductor}{spoel} wanneer $i = \pm I_0$.
\end{proposition}

\begin{proof}
Leid af: $\frac{dE}{dt} = \frac{q}{C}\frac{dq}{dt} +
Li\frac{di}{dt} = i\left(\frac{q}{C} + L\frac{d^2q}{dt^2}\right) = 0$ wegens
\cref{prop:g12:rlc-oscillations:equation}. Haar waarde lees je af op $t = 0$
(alles in de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}), en nogmaals op het ogenblik dat alles stroom is,
$q = 0$ --- waaruit $\tfrac12 L I_0^2 = Q_0^2/2C$ en dus
$I_0 = Q_0/\sqrt{LC}$ volgt, met \omterm{def:g10:energy-conservation:energy}{energie} alleen.
\end{proof}

precies de figuur van de \omterm{prop:g12:oscillators-and-time:spring-period}{veeroscillator}, met andere etiketten.}
\end{omfigure}

\begin{example}[De wals, opgemeten]\label{ex:g12:rlc-oscillations:waltz}
$C = \qty{1.0}{\micro F}$ geladen tot $U_0 = \qty{10}{V}$ en geschakeld op
$L = \qty{10}{mH}$: \omterm{def:g10:energy-conservation:energy}{energie} $E = \tfrac12 C U_0^2 = \qty{5.0e-5}{J}$, \omterm{def:g12:oscillators-and-time:oscillator}{periode}
$T_0 = \qty{0.63}{ms}$, en een piekstroom $I_0 = U_0\sqrt{C/L} =
\qty{0.10}{A}$ die een kwart \omterm{def:g12:oscillators-and-time:oscillator}{periode} (\qty{0.16}{ms}) na het sluiten van de
schakelaar wordt gehaald.
\end{example}

\section{Echte schakelingen: gedempte trillingen}

Elke echte \omterm{def:g12:rc-rl-circuits:inductor}{spoel} is van draad met \omterm{def:g11:circuits-and-power:resistance}{weerstand} gewikkeld: de eerlijke kring is
een serie-\emph{RLC}-kring, en de \omterm{def:g11:circuits-and-power:resistance}{weerstand} belast de wals.

\begin{definition}[Pseudoperiodisch en aperiodisch regime]\label{def:g12:rlc-oscillations:regimes}
\index{RLC-kring}\index{pseudoperiodisch regime}\index{aperiodisch regime}
In een serie-RLC-kring verstookt de \omterm{def:g11:circuits-and-power:resistance}{weerstand} \omterm{def:g10:energy-conservation:energy}{energie} door het \omterm{def:g11:circuits-and-power:joule}{joule-effect}
(\cref{ch:g11:circuits-and-power}): de \omterm{def:g12:oscillators-and-time:oscillator}{vrije trillingen} zijn gedempt. Twee
regimes, af te lezen op een \omterm{def:g10:signals-and-waves:oscilloscope}{oscilloscoop}:
\begin{itemize}
  \item \emph{pseudoperiodisch} (kleine $R$): de kring trilt nog binnen een
        krimpende omhullende, met een herhaaltijd --- de
        \emph{pseudoperiode}\index{pseudoperiode!elektrisch} --- dicht bij
        $T_0 = 2\pi\sqrt{LC}$ zolang de \omterm{def:g12:oscillators-and-time:damping}{demping} zwak is;
  \item \emph{aperiodisch} (grote $R$): de lading kruipt naar nul terug zonder
        er ooit voorbij te schieten --- helemaal geen trilling.
\end{itemize}
Het exacte spiegelbeeld van de gedempte mechanische \omterm{def:g12:oscillators-and-time:oscillator}{oscillator}
(\cref{def:g12:oscillators-and-time:damping}).
\end{definition}

\begin{remark}[Trillingen onderhouden]\label{rem:g12:rlc-oscillations:maintain}
\index{onderhouden trillingen}
Om de \omterm{def:g12:oscillators-and-time:oscillator}{amplitude} in leven te houden, voer je de kring precies de \omterm{def:g10:energy-conservation:energy}{energie} toe
die de \omterm{def:g11:circuits-and-power:resistance}{weerstand} verstookt, één keer per cyclus en \emph{in de pas} met de
zwaai: een versterker doet voor de \omterm{def:g12:rlc-oscillations:regimes}{RLC-kring} wat het duwtje van het
echappement voor de slinger doet (\cref{ch:g12:oscillators-and-time}). Elk
kwartshorloge en elke radiozender is zo'n onderhouden \omterm{def:g12:oscillators-and-time:oscillator}{oscillator}, draaiend op
zijn \omterm{def:g12:rlc-oscillations:period}{eigenfrequentie} $f_0$.
\end{remark}

\section{De mechanische tweelingbroer}

\begin{proposition}[Het elektrisch--mechanische woordenboek]\label{prop:g12:rlc-oscillations:analogy}
\index{elektrisch--mechanische analogie}
De LC-vergelijking is woord voor woord de vergelijking van de veer met massa
(\cref{prop:g12:oscillators-and-time:spring-period}):
\begin{center}\small
\begin{tabular}{lcl}
\omterm{prop:g12:rlc-oscillations:equation}{LC-kring} & & wagentje aan een veer\\[2pt]
lading $q$ & $\longleftrightarrow$ & plaats $x$\\
stroom $i = dq/dt$ & $\longleftrightarrow$ & snelheid $v = dx/dt$\\
\omterm{def:g12:rc-rl-circuits:inductor}{zelfinductie} $L$ & $\longleftrightarrow$ & massa $m$\\
omgekeerde \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit} $1/C$ & $\longleftrightarrow$ & \omterm{def:g12:oscillators-and-time:spring}{veerconstante} $k$\\
\omterm{def:g11:circuits-and-power:resistance}{weerstand} $R$ & $\longleftrightarrow$ & \omterm{def:g10:inertia:inventory}{wrijving}\\
$L\,\frac{d^2q}{dt^2} + \frac{q}{C} = 0$ & $\longleftrightarrow$ &
  $m\,\frac{d^2x}{dt^2} + kx = 0$\\[2pt]
$T_0 = 2\pi\sqrt{LC}$ & $\longleftrightarrow$ &
  $T_0 = 2\pi\sqrt{m/k}$\\[2pt]
$E_C = q^2/2C$ & $\longleftrightarrow$ & $E_p = \tfrac12 kx^2$\\
$E_L = \tfrac12 Li^2$ & $\longleftrightarrow$ & $E_k = \tfrac12 mv^2$
\end{tabular}
\end{center}
Dezelfde vergelijking, dezelfde oplossingen: elk resultaat over de ene
\omterm{def:g12:oscillators-and-time:oscillator}{oscillator} vertaalt zich ogenblikkelijk in de taal van de andere.
\end{proposition}

\begin{proof}
Vergelijk \cref{prop:g12:rlc-oscillations:equation} met
$m\,d^2x/dt^2 = -kx$: de substitutie $q \to x$, $L \to m$, $1/C \to k$ maakt
van de ene de andere, en $LC \to m/k$ maakt van $2\pi\sqrt{LC}$ de uitdrukking
$2\pi\sqrt{m/k}$.
\end{proof}

\begin{example}[Een radio afstemmen]\label{ex:g12:rlc-oscillations:tuning}
Een antenne voert een fluistering van \emph{elke} zender in een \omterm{prop:g12:rlc-oscillations:equation}{LC-kring},
maar alleen die welke rond $f_0$ uitzendt duwt in de pas en bouwt zich op ---
\omterm{def:g12:oscillators-and-time:resonance}{resonantie} (\cref{def:g12:oscillators-and-time:resonance}). De afstemknop
draait de platen van een variabele \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}: door $C$ te veranderen
verschuift $f_0 = 1/(2\pi\sqrt{LC})$, en de naald reikt de oortelefoon één
zender per keer aan. Aan de zendkant legt een onderhouden \omterm{def:g12:oscillators-and-time:oscillator}{oscillator}
(\cref{rem:g12:rlc-oscillations:maintain}) de uitzendfrequentie vast; in een
horloge speelt een kwartskristal --- een mechanische tweelingbroer van
adembenemende stijfheid --- dezelfde rol.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:4}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:rlc-oscillations:5}

En volledig in de \omterm{def:g12:rc-rl-circuits:inductor}{spoel}? Wat is voor een slinger de mechanische tegenhanger

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:6}
Ga door invullen na dat $q(t) = Q_0\cos(2\pi t/T_0)$ aan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:8}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:9}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:10}
Op een \omterm{def:g10:signals-and-waves:oscilloscope}{oscillogram} van een zwak gedempte \omterm{def:g12:rlc-oscillations:regimes}{RLC-kring} lezen opeenvolgende maxima

constant blijft, het zesde maximum na dat van \qty{4.0}{V}. Welke fractie van
de \emph{\omterm{def:g10:energy-conservation:energy}{energie}} overleeft elke \omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode}? Wat kun je over de
\omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode} zelf zeggen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:rlc-oscillations:11}
Een \omterm{prop:g12:rlc-oscillations:equation}{LC-kring} heeft $L = \qty{0.50}{H}$ en $C = \qty{20}{\micro F}$. Bereken
$T_0$; bepaal daarna met het woordenboek
(\cref{prop:g12:rlc-oscillations:analogy}) de \omterm{def:g12:oscillators-and-time:spring}{veerconstante} $k$ die een
wagentje van \qty{0.50}{kg} dezelfde \omterm{def:g12:oscillators-and-time:oscillator}{periode} geeft, en controleer die.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:12}
Een afstemkring gebruikt $L = \qty{0.20}{mH}$ en een variabele \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} die
\qtyrange{50}{500}{pF} bestrijkt. Bereken de twee uiterste \omterm{def:g12:oscillators-and-time:oscillator}{frequenties}. Toon
aan dat $f_0 \propto 1/\sqrt{C}$, zodat een factor tien in $C$ maar
$\sqrt{10} \approx 3.2$ in \omterm{def:g12:oscillators-and-time:oscillator}{frequentie} oplevert.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:13}
Door een echte \omterm{def:g12:rc-rl-circuits:inductor}{spoel} verliest de kring per \omterm{def:g12:rlc-oscillations:regimes}{pseudoperiode} $5.0\%$ van haar
\omterm{def:g10:energy-conservation:energy}{energie}. Na hoeveel \omterm{def:g12:oscillators-and-time:oscillator}{perioden} is de \omterm{def:g10:energy-conservation:energy}{energie} gehalveerd? Hoe groot is de
\omterm{def:g12:oscillators-and-time:oscillator}{amplitude} dan, als fractie van de start? Hoe lang duurt dat bij
$f_0 = \qty{1.0}{MHz}$, in microseconden?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:14}
Een \omterm{def:g12:oscillators-and-time:oscillator}{horloge-oscillator} slaat $E = \qty{1.0e-9}{J}$ op en verliest daarvan
$2.0\%$ per cyclus bij $f_0 = \qty{32768}{Hz}$. Welk gemiddeld \omterm{def:g11:work-of-force:power}{vermogen} moet
de onderhoudsschakeling leveren? Waarom moet het duwtje in de pas met de
trilling aankomen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:rlc-oscillations:15}
Na elke flank van een blokvormig \omterm{def:g10:signals-and-waves:signal}{signaal} ``rinkelt'' de bedrading van een
experimentator op \qty{25}{MHz}; de parasitaire \omterm{def:g12:rc-rl-circuits:inductor}{zelfinductie} is ongeveer
\qty{1.0}{\micro H}. Schat de parasitaire \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit} die daarvoor verantwoordelijk
is. In welk regime zou een kleine \omterm{def:g11:circuits-and-power:resistance}{weerstand} in serie de kring moeten duwen, en
waarom geneest dat het rinkelen?
\end{exercise}

\section{Probleem: Een kristalontvanger bouwen}

\begin{problem}[{Weekendopgave --- een kristalontvanger bouwen: één spoel,
één variabele condensator, een diode en een oortelefoon --- geen batterij,
geen versterker, en het avondnieuws komt binnen op de energie van de golf
zelf}]
\label{pb:g12:rlc-oscillations:1}
Een bouwpakket van zolder bij de grootouders: een \omterm{def:g12:rc-rl-circuits:inductor}{spoel} $L = \qty{0.20}{mH}$
gewikkeld op een kartonnen koker, een variabele \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} waarvan de platen
van \qty{20}{pF} tot \qty{480}{pF} draaien, een kristaldiode, een oortelefoon
en \qty{20}{m} antennedraad. De zenders die je wilt vangen zenden uit tussen
\qty{530}{kHz} en \qty{1700}{kHz}.

\medskip
\noindent\textbf{Deel I --- De afstemkring.}
\begin{enumerate}
  \item De geladen \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} mag zich in de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} ontladen: schrijf de
        kringwet op en maak er de differentiaalvergelijking voor $q(t)$ van.
  \item Ga door invullen na dat $q(t) = Q_0\cos(2\pi t/T_0)$ haar oplost, mits
        $T_0 = 2\pi\sqrt{LC}$.
  \item Controleer met een dimensieanalyse dat $\sqrt{LC}$ een tijd is.
  \item Knop op $C = \qty{330}{pF}$: bereken $T_0$ en $f_0$. Is de kring binnen
        de band afgestemd?
  \item Bepaal de \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit} die precies op \qty{1.00}{MHz} afstemt.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De variabele \omterm{def:g12:rc-rl-circuits:capacitor}{condensator}.}
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit} die nodig is voor \qty{530}{kHz}.
  \item Bereken de \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit} die nodig is voor \qty{1700}{kHz}.
  \item Bereken de twee \omterm{def:g12:oscillators-and-time:oscillator}{frequenties} die de \omterm{def:g12:rc-rl-circuits:capacitor}{condensator} van
        \qtyrange{20}{480}{pF} uit het pakket werkelijk haalt. Bestrijkt hij de
        hele band?
  \item Toon aan dat $f_0 \propto 1/\sqrt{C}$ bij vaste $L$; met welke factor
        moet $C$ veranderen om $f_0$ te verdubbelen?
  \item De platen schuiven lineair open, dus daalt $C$ gestaag terwijl de knop
        draait. Leg uit waarom de zenders dan aan één kant van de schaal op
        elkaar gedrongen staan --- en aan welke kant.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- \omterm{def:g10:energy-conservation:energy}{Energie} en \omterm{def:g12:oscillators-and-time:damping}{demping}.}
Afgestemd op \qty{1.00}{MHz} ($C = \qty{127}{pF}$) laadt de antenne de
\omterm{def:g12:rc-rl-circuits:capacitor}{condensator} even tot $U_0 = \qty{20}{mV}$.
\begin{enumerate}[resume]
  \item Bereken de lading $Q_0$ en de opgeslagen \omterm{def:g10:energy-conservation:energy}{energie} $E$.
  \item Bereken de piekstroom $I_0$ met behulp van het \omterm{thm:g10:energy-conservation:conservation}{behoud van energie}.
  \item De \omterm{def:g11:circuits-and-power:resistance}{weerstand} van de \omterm{def:g12:rc-rl-circuits:inductor}{spoel} verstookt per cyclus $4.0\%$ van de
        opgeslagen \omterm{def:g10:energy-conservation:energy}{energie}: na hoeveel cycli is de \omterm{def:g10:energy-conservation:energy}{energie} gehalveerd?
  \item Hoe lang duurt dat in microseconden? Waarom kan dit ``rinkelen'' in
        zijn eentje het avondnieuws niet laten spelen?
  \item Leg uit waarom de binnenkomende \omterm{def:g10:signals-and-waves:wave}{golf} de kring alleen aan het zwaaien
        houdt als die op die zender is afgestemd --- en wat de trilling bij de
        \emph{zender} onderhoudt
        (\cref{rem:g12:rlc-oscillations:maintain}).
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De mechanische tweelingbroer.}
\begin{enumerate}[resume]
  \item Bereken met het woordenboek (\cref{prop:g12:rlc-oscillations:analogy})
        en een wagentje met massa $m = \qty{0.20}{kg}$ de \omterm{def:g12:oscillators-and-time:spring}{veerconstante} $k$ van
        de veer die bij de afgestemde kring past.
  \item Een stugge autoveer heeft $k \approx \qty{5e4}{N/m}$. Becommentarieer:
        kan een opstelling met een veer en een massa op tafel op megahertz
        trillen?
  \item Houd in plaats daarvan een redelijke $k = \qty{100}{N/m}$ aan: welke
        massa trilt op \qty{1.00}{MHz}?
  \item Welk echt voorwerp is zo'n vederlichte, superstijve mechanische
        \omterm{def:g12:oscillators-and-time:oscillator}{oscillator}, en waar ben je het eerder dit jaar tegengekomen
        (\cref{ch:g12:oscillators-and-time})?
  \item Specificatieblad, in één zin: de band die je radio bestrijkt, de
        \omterm{def:g12:rc-rl-circuits:capacitor}{capaciteit} die \qty{1.00}{MHz} uitkiest, hoe lang één losse rinkeling
        duurt, en wat het nieuws toch laat spelen.
\end{enumerate}
\end{problem}
