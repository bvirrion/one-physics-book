\chapter{Mechanische golven}\label{ch:g12:mechanical-waves}

Zaterdagavond, een vol stadion: een rimpeling van opstaande toeschouwers
raast in minder dan een minuut de tribunes rond, en toch verlaat niemand
zijn stoel. De kring van een kiezelsteen steekt de vijver over terwijl de
kurk alleen op en neer danst. Dit hoofdstuk bestudeert de reiziger zelf
--- de \omterm{def:g10:signals-and-waves:wave}{golf} --- en de vier getallen die haar temmen: snelheid,
\omterm{def:g12:mechanical-waves:delay}{vertraging}, \omterm{def:g10:signals-and-waves:period}{periode} en \omterm{def:g12:mechanical-waves:wavelength}{golflengte} --- genoeg om een aardbeving van een
strook papier af te lezen.

\section{De verstoring reist, het midden blijft}

\begin{definition}[Mechanische golf]\label{def:g12:mechanical-waves:wave}
Een \emph{mechanische golf}\index{mechanische golf} is een verstoring die
zich door een stoffelijk midden voortplant --- touw, water, lucht,
gesteente --- en die \omterm{def:g10:energy-conservation:energy}{energie} draagt \emph{zonder materie te vervoeren}:
elk stukje van het midden zwaait even ter plaatse heen en weer, geeft de
beweging door en komt weer tot rust.
\end{definition}

\begin{example}[Geen vervoer van materie]\label{ex:g12:mechanical-waves:transport}
Bij de stadiongolf is het ``midden'' de menigte: wat reist is de daad van
het opstaan, terwijl elke toeschouwer zijn stoel houdt. Op de vijver
groeit de kring, maar een drijvende kurk danst alleen op en neer.
\end{example}

\begin{definition}[Transversale en longitudinale golven]\label{def:g12:mechanical-waves:transverse}
Een \omterm{def:g10:signals-and-waves:wave}{golf} is \emph{transversaal}\index{transversale golf} wanneer het
midden \emph{loodrecht} op de voortplanting beweegt: een bult snelt langs
een aangetikt touw terwijl elk punt alleen op en neer gaat. Ze is
\emph{longitudinaal}\index{longitudinale golf} wanneer het midden
\emph{langs} de voortplanting beweegt: een zone van samengedrukte
windingen reist door een aangeduwde veer, terwijl elke winding ter plaatse
heen en weer zwaait.
\end{definition}

\begin{remark}[Geluid is longitudinaal]\label{rem:g12:mechanical-waves:sound}
\omterm{def:g10:signals-and-waves:sound}{Geluid} is de \omterm{def:g12:mechanical-waves:transverse}{longitudinale golf} bij uitstek: samengedrukte lucht ---
hogere \omterm{def:g10:pressure:pressure}{druk} (\cref{ch:g10:pressure}) --- reist van de bron naar het
trommelvlies; \cref{ch:g12:sound-acoustics} is eraan gewijd. Aardbevingen
sturen beide soorten uit: P-golven drukken het gesteente samen, S-golven
schudden het zijwaarts.
\end{remark}

\section{Snelheid en vertraging}

\begin{definition}[Voortplantingssnelheid]\label{def:g12:mechanical-waves:speed}
De \emph{voortplantingssnelheid}\index{voortplantingssnelheid} van een
\omterm{def:g10:signals-and-waves:wave}{golf} is de afstand die de verstoring aflegt, gedeeld door de tijd die dat
kost: $v = d/t$. Het is de snelheid van de reizende \emph{vorm}, niet die
van het midden.
\end{definition}

\begin{proposition}[Het midden bepaalt de snelheid]\label{prop:g12:mechanical-waves:medium}
Voor kleine verstoringen hangt de \omterm{def:g12:mechanical-waves:speed}{voortplantingssnelheid} alleen van het
\emph{midden} af, niet van de vorm of de grootte van de verstoring. Op een
gespannen touw groeit $v$ met de \omterm{def:g10:inertia:inventory}{spankracht} en daalt ze met de massa per
\omterm{def:g10:orders-of-magnitude:unit}{meter}: strak en licht is snel, slap en zwaar is traag. \admitted
\end{proposition}

\begin{remark}[Waar de formule staat]\label{rem:g12:mechanical-waves:ropeformula}
De exacte snelheidsformule voor het touw wordt afgeleid in het volume van
bachelorjaar 1; dit jaar volstaan de experimentele feiten hierboven.
\end{remark}

\begin{example}[Geluidssnelheden]\label{ex:g12:mechanical-waves:speeds}
\omterm{def:g10:signals-and-waves:sound}{Geluid} reist met \qty{340}{m/s} in lucht, \qty{1500}{m/s} in water en
\qty{5900}{m/s} in staal --- hoe stijver, hoe sneller. Vandaar het oor op
de rails uit de westerns: het staal kondigt de trein het eerst aan.
\end{example}

\begin{definition}[Vertraging]\label{def:g12:mechanical-waves:delay}
Een \omterm{def:g10:signals-and-waves:wave}{golf} met snelheid $v$ bereikt een punt op afstand $d$ na de
\emph{vertraging}\index{vertraging} $\tau = d/v$: twee punten op afstand
$d$ herhalen dezelfde beweging, waarbij het verste $\tau$ achterloopt ---
de afstandsmeting met \omterm{rem:g10:signals-and-waves:honest}{echo}'s uit \cref{ch:g10:signals-and-waves},
achterstevoren gelezen.
\end{definition}

\begin{proposition}[Superpositie zonder wisselwerking]\label{prop:g12:mechanical-waves:superposition}
Waar twee golven elkaar overlappen, is de uitwijking van het midden op elk
ogenblik de \emph{som} van de uitwijkingen die elk afzonderlijk zou
veroorzaken; na het kruisen loopt elke \omterm{def:g10:signals-and-waves:wave}{golf} onveranderd door --- afgeleid
in het volume van bachelorjaar 2 uit de lineariteit van de
bewegingsvergelijkingen. \admitted
\end{proposition}

\begin{example}[Kruisende pulsen]\label{ex:g12:mechanical-waves:crossing}
Stuur vanaf elk uiteinde van een touw een bult omhoog: waar ze elkaar
ontmoeten, rijst het touw even tot de som van de twee hoogten; daarna
komen er twee ongeschonden bulten uit --- zoals twee kruisende gesprekken
elk hun eigen luisteraar bereiken.
\end{example}

\section{Periodieke golven: de dubbele periodiciteit}

\begin{definition}[Periodieke golf]\label{def:g12:mechanical-waves:periodic}
Wanneer de bron haar beweging herhaalt met \omterm{def:g10:signals-and-waves:period}{periode} $T$ en \omterm{def:g10:signals-and-waves:frequency}{frequentie}
$f = 1/T$ (\cref{ch:g10:signals-and-waves}), voedt ze het midden met een
\emph{periodieke golf}\index{periodieke golf}: elk punt herhaalt zijn
eigen beweging met \omterm{def:g10:signals-and-waves:period}{periode} $T$, en elk loopt op zijn buren achter met de
\omterm{def:g12:mechanical-waves:delay}{vertraging} van de looptijd.
\end{definition}

\begin{definition}[Golflengte]\label{def:g12:mechanical-waves:wavelength}
Een momentopname van een \omterm{def:g12:mechanical-waves:periodic}{periodieke golf} toont een patroon dat zich in de
ruimte herhaalt; de ruimtelijke \omterm{def:g10:signals-and-waves:period}{periode} ervan --- van top tot top --- is
de \emph{golflengte}\index{golflengte} $\lambda$, in \omterm{def:g10:orders-of-magnitude:unit}{meter}. Een \omterm{def:g12:mechanical-waves:periodic}{periodieke
golf} is \emph{dubbel periodiek}\index{dubbele periodiciteit}: met \omterm{def:g10:signals-and-waves:period}{periode}
$T$ in de tijd in elk punt, en met \omterm{def:g10:signals-and-waves:period}{periode} $\lambda$ in de ruimte op elk
ogenblik.
\end{definition}

\begin{theorem}[De golfbetrekking]\label{thm:g12:mechanical-waves:relation}
Een \omterm{def:g12:mechanical-waves:periodic}{periodieke golf} met snelheid $v$, \omterm{def:g10:signals-and-waves:period}{periode} $T$ en \omterm{def:g10:signals-and-waves:frequency}{frequentie} $f$ heeft
\omterm{def:g12:mechanical-waves:wavelength}{golflengte} $\lambda = v\,T = v/f$.
\end{theorem}

\begin{proof}
In één \omterm{def:g10:signals-and-waves:period}{periode} schuift het patroon $v\,T$ naar voren; maar de bron --- en
daarmee elk punt --- is dan terug in haar begintoestand, dus valt het
opgeschoven patroon samen met het oorspronkelijke: de herhalingsafstand is
$v\,T$.
\end{proof}

\begin{remark}[De frequentie hoort bij de bron]\label{rem:g12:mechanical-waves:source}
Bij de overgang van het ene midden naar het andere --- \omterm{def:g10:signals-and-waves:sound}{geluid} van lucht
naar water --- behoudt een \omterm{def:g10:signals-and-waves:wave}{golf} haar \omterm{def:g10:signals-and-waves:frequency}{frequentie}, die de bron oplegt,
terwijl haar snelheid verandert; volgens $\lambda = v/f$ verandert de
\omterm{def:g12:mechanical-waves:wavelength}{golflengte} mee.
\end{remark}

\begin{example}[Kamertoon a, in lucht en onder water]\label{ex:g12:mechanical-waves:concertA}
De kamertoon a van \qty{440}{Hz} uit \cref{ch:g10:signals-and-waves} heeft
\omterm{def:g12:mechanical-waves:wavelength}{golflengte} $340/440 = \qty{0.77}{m}$ in lucht en
$1500/440 = \qty{3.4}{m}$ in water: dezelfde noot, uitgerekt door het
snellere midden.
\end{example}

\begin{definition}[In fase, in tegenfase]\label{def:g12:mechanical-waves:phase}
Twee punten die een geheel aantal \omterm{def:g12:mechanical-waves:wavelength}{golflengten} uit elkaar liggen,
$d = k\lambda$, bewegen op elk ogenblik gelijk: ze zijn \emph{in
fase}\index{in fase}. Punten op afstand $d = (k + \tfrac12)\lambda$
bewegen altijd tegengesteld --- top tegen dal: ze zijn \emph{in
tegenfase}\index{tegenfase}.
\end{definition}

\begin{method}[De dubbele periodiciteit aflezen]\label{met:g12:mechanical-waves:reading}
\begin{enumerate}
  \item Lees op een opname $y(t)$ in een vast punt $T$ af, van top tot
        top; dan is $f = 1/T$.
  \item Lees op een momentopname $y(x)$ op een vast ogenblik $\lambda$ af,
        van top tot top.
  \item Leid $v = \lambda/T = \lambda f$ af; toets die aan een
        rechtstreekse tijdmeting $v = d/\tau$ wanneer je die hebt.
  \item Verwar de twee grafieken nooit: op de ene as staan seconden, op de
        andere \omterm{def:g10:orders-of-magnitude:unit}{meters}.
\end{enumerate}
\end{method}

\section{Golffronten en verzwakking}

\begin{definition}[Golffront]\label{def:g12:mechanical-waves:wavefront}
Een \emph{golffront}\index{golffront} is een lijn (of vlak) van punten die
de \omterm{def:g10:signals-and-waves:wave}{golf} op hetzelfde ogenblik bereikt --- een topkam op de vijver. Vanuit
een puntbron zijn de golffronten uitdijende cirkels; ver van elke bron
zijn ze vrijwel recht. De \omterm{def:g10:signals-and-waves:wave}{golf} plant zich loodrecht op haar golffronten
voort, met opeenvolgende toppen $\lambda$ uit elkaar.
\end{definition}

\begin{definition}[Verzwakking]\label{def:g12:mechanical-waves:attenuation}
De afname van de \omterm{def:g10:signals-and-waves:amplitude}{amplitude} van een \omterm{def:g10:signals-and-waves:wave}{golf} tijdens haar voortplanting is haar
\emph{verzwakking}\index{verzwakking}. Twee oorzaken tellen op: de \omterm{def:g10:energy-conservation:energy}{energie}
van een cirkelvormig \omterm{def:g12:mechanical-waves:wavefront}{golffront} wordt over een steeds langere kam verdeeld,
en het midden absorbeert onderweg een beetje. Rechte \omterm{def:g12:mechanical-waves:wavefront}{golffronten} ontsnappen
aan de eerste oorzaak --- en zo kan de deining van een verre storm een
oceaan oversteken.
\end{definition}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:1}
Een stadiongolf legt de ring van \qty{320}{m} rond de tribunes af in
\qty{40}{s}. (a) Bereken haar snelheid. (b) Reist er een toeschouwer rond
de ring? Wat wel? (c) Elke toeschouwer staat \qty{1.5}{s} na zijn buurman
op \qty{12}{m} afstand op: ga na of dat klopt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:2}
\omterm{def:g12:mechanical-waves:transverse}{Transversaal} of \omterm{def:g12:mechanical-waves:transverse}{longitudinaal}? (a) een puls op een touw dat op en neer

(c) \omterm{def:g10:signals-and-waves:sound}{geluid} in lucht; (d) een rimpeling op een vijver (het oppervlak danst

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:mechanical-waves:3}

\section{Probleem: Een seismogram lezen}

\begin{problem}[{Weekendopgave --- een seismogram lezen: twee golven
racen weg uit een gebroken breuk, drie cirkels prikken het epicentrum
vast, en een trage derde golf geeft een kustlijn haar waarschuwing}]
\label{pb:g12:mechanical-waves:1}
Nachtdienst op een seismologisch observatorium. Om 09:14:32 springt de
naald van station A op: eerst scherpe snelle trillingen, daarna tragere,
bredere uitslagen. Ergens is gesteente gebroken; twee golven racen door de
aardkorst naar buiten: P-golven, verdichtingen met
$v_P = \qty{6.0}{km/s}$, en S-golven, een zijwaartse schudding met
$v_S = \qty{3.5}{km/s}$.

\medskip
\noindent\textbf{Deel I --- Twee golven, één klok.}
\begin{enumerate}
  \item Deel P- en S-golven op grond van de beschrijvingen hierboven in
        als \omterm{def:g12:mechanical-waves:transverse}{transversaal} of \omterm{def:g12:mechanical-waves:transverse}{longitudinaal}.
  \item Welke \omterm{def:g10:signals-and-waves:wave}{golf} komt op elk station het eerst aan? Bereken de
        looptijden van beide over $d = \qty{120}{km}$.
  \item Toon aan dat het gat tussen de aankomsten van P en S gelijk is
        aan $\Delta t = d\,(1/v_S - 1/v_P)$, en bereken het voor
        $d = \qty{120}{km}$.
  \item Leid de afstandsregel van de seismoloog
        $d = \dfrac{v_P\,v_S}{v_P - v_S}\,\Delta t$ af en toon aan dat de
        coëfficiënt ongeveer \qty{8.4}{km} per seconde gat bedraagt.
  \item Waarom groeit het gat met de afstand --- en waarom is dat een
        geschenk voor een station dat het begin van de beving niet heeft
        gezien?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Cirkels op de kaart.}
\begin{enumerate}[resume]
  \item Station A meet $\Delta t_A = \qty{25}{s}$. Hoever ligt het
        epicentrum van A?
  \item Leg uit waarom die ene aflezing het epicentrum op een cirkel legt
        en niet op een punt.
  \item De stations B en C meten $\Delta t_B = \qty{15}{s}$ en
        $\Delta t_C = \qty{32}{s}$: wat zijn hun afstanden tot het
        epicentrum?
  \item Leg uit hoe de drie cirkels samen het epicentrum vastprikken.
  \item De \omterm{def:g10:signals-and-waves:wave}{P-golf} bereikte A om 09:14:32. Bereken haar looptijd vanaf het
        epicentrum, en het tijdstip waarop de beving begon.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{def:g10:signals-and-waves:wave}{golf} die de oceaan oversteekt.}
De drie cirkels snijden elkaar voor de kust: de breuk brak onder de
zeebodem en zette de waterkolom in beweging. Lange golven op diep water
reizen met een snelheid die door de diepte wordt bepaald --- ongeveer
\qty{200}{m/s} op de open oceaan, ongeveer \qty{10}{m/s} vlak bij een
strand (hier aangenomen; de formule wordt afgeleid in het volume van
bachelorjaar 1). Een haven ligt $D = \qty{600}{km}$ van het epicentrum.
\begin{enumerate}[resume]
  \item Zet \qty{200}{m/s} om naar \unit{km/h} en noem een voertuig met
        een vergelijkbare kruissnelheid.
  \item Hoe lang doet de tsunami erover om de haven te bereiken?
  \item De seismometer van de haven voelt eerst de \omterm{def:g10:signals-and-waves:wave}{P-golf}. Na hoeveel
        tijd? Als het alarm op dat ogenblik afgaat, hoeveel
        waarschuwingstijd krijgt de kust dan vóór de tsunami?
  \item Op zee is de \omterm{def:g10:signals-and-waves:period}{periode} van de tsunami ongeveer \qty{15}{min}.
        Bereken haar \omterm{def:g12:mechanical-waves:wavelength}{golflengte} daar, en leg uit waarom een schip haar
        niet opmerkt.
  \item Dicht bij het strand zakt de snelheid tot \qty{10}{m/s} terwijl
        de \omterm{def:g10:signals-and-waves:period}{periode} vastligt. Bereken de nieuwe \omterm{def:g12:mechanical-waves:wavelength}{golflengte}. Wat moet er met
        de \omterm{def:g10:signals-and-waves:wave}{golf} gebeuren nu ze zich opeenhoopt?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- \omterm{def:g10:energy-conservation:energy}{Energie}, en het rapport.}
Een zware beving maakt in de orde van \qty{1e15}{J} vrij, die door de
golven wordt weggedragen.
\begin{enumerate}[resume]
  \item Bereken de lengte van een cirkelvormig \omterm{def:g12:mechanical-waves:wavefront}{golffront} op \qty{10}{km}
        van het epicentrum, en daarna op \qty{210}{km}; met welke factor
        is het gegroeid?
  \item Leid daaruit af waarom het schudden met de afstand afneemt zelfs
        in een verliesvrije aardkorst --- en noem de tweede oorzaak die
        een echte korst eraan toevoegt.
  \item Een vloeistof kan geen schuifspanning verdragen. Stations aan de
        overkant registreren de \omterm{def:g10:signals-and-waves:wave}{P-golf} maar geen \omterm{def:g10:signals-and-waves:wave}{S-golf}: wat verraadt dat
        over de kern van de Aarde?
  \item De hogesnelheidstrein uit het energiehoofdstuk van vorig jaar
        (\cref{ch:g11:mechanical-energy}) draagt ongeveer \qty{1.5e9}{J}
        \omterm{def:g11:mechanical-energy:kinetic}{kinetische energie}: hoeveel van zulke treinen is \qty{1e15}{J}
        waard?
  \item Het rapport, in twee zinnen: waar en wanneer vond de beving
        plaats, en hoeveel waarschuwing kreeg de haven? Geef de drie
        getallen.
\end{enumerate}
\end{problem}
