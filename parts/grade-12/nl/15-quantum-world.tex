\chapter{De kwantumwereld: fotonen en energieniveaus}\label{ch:g12:quantum-world}

Een infraroodlamp kan je huid een hele avond baden zonder een spoor achter te
laten; tien minuten ijle bergzon levert een verbranding op. De lamp levert
veel meer \omterm{def:g10:energy-conservation:energy}{energie} --- maar \omterm{def:g10:energy-conservation:energy}{energie} is niet wat de huid tekent. Licht, zo laat
dit hoofdstuk zien, komt in korrels aan, en alleen een korrel die hard genoeg
inslaat doet chemie: één ultraviolette korrel breekt een binding die een
miljard infrarode korrels alleen maar opwarmen. Volg de korrels en er gaat een
eeuw open: metalen die elektronen lekken, atomen die in \omterm{def:g10:light-spectra:emission}{spectraallijnen}
zingen, elektronen die als golven \omterm{prop:g12:light-as-wave:conditions}{interfereren}.

\section{Licht komt in korrels: het foton}

\begin{definition}[Het foton]\label{def:g12:quantum-world:photon}
Licht met \omterm{def:g12:oscillators-and-time:oscillator}{frequentie} $\nu$ wisselt alleen in hele korrels \omterm{def:g10:energy-conservation:energy}{energie} met materie
uit; die korrels heten \emph{fotonen}\index{foton}, elk met \omterm{def:g10:energy-conservation:energy}{energie}
\[
E = h\nu = \frac{hc}{\lambda}, \qquad h = \qty{6.63e-34}{J.s},
\]
waarin $h$ de \emph{constante van Planck}\index{constante van Planck} is
(Planck, 1900; Einstein, 1905) en $\lambda$ de \omterm{def:g12:light-as-wave:lightwave}{golflengte in vacuüm}. De
intensiteit van een bundel meet het \emph{aantal} fotonen per seconde; de
\omterm{def:g10:energy-conservation:energy}{energie} van elke korrel ligt alleen door de \omterm{def:g12:oscillators-and-time:oscillator}{frequentie} vast.
\end{definition}

\begin{notation}[De elektronvolt]\label{not:g12:quantum-world:ev}
De atoomfysica gebruikt de \omterm{def:g12:nuclear-energy:units}{elektronvolt}, $\qty{1}{eV} = \qty{1.60e-19}{J}$:
de \omterm{def:g10:energy-conservation:energy}{energie} die één \omterm{def:g11:fundamental-interactions:elementary}{elementaire lading} wint door \qty{1}{V} te doorlopen
(\cref{ch:g11:electric-gravitational-fields}). Handig:
$E \,(\unit{eV}) \approx 1240/\lambda\,(\unit{nm})$.
\end{notation}

\section{Het foto-elektrisch effect}

Schijn licht op een schoon metalen plaatje in vacuüm en er springen
elektronen uit --- het \emph{foto-elektrisch
effect}\index{foto-elektrisch effect}. De regels ervan braken de klassieke
natuurkunde.

\begin{proposition}[Wat het experiment laat zien]\label{prop:g12:quantum-world:facts}
Voor elk metaal bestaat er een
\emph{drempelfrequentie}\index{drempelfrequentie} $\nu_0$:
\begin{enumerate}
  \item onder $\nu_0$ vertrekt er geen enkel elektron, hoe fel het licht ook
        is;
  \item boven $\nu_0$ begint de emissie zonder \omterm{def:g12:mechanical-waves:delay}{vertraging}, hoe zwak het licht
        ook is;
  \item de intensiteit verhoogt het \emph{aantal} elektronen per seconde, niet
        hun maximale \omterm{def:g11:mechanical-energy:kinetic}{kinetische energie}; de \omterm{def:g12:oscillators-and-time:oscillator}{frequentie} verhoogt
        $E_{k,\max}$.
\end{enumerate}
Niets daarvan past bij een continue \omterm{def:g10:signals-and-waves:wave}{golf}, waarbij elke kleur zou moeten
kunnen opladen --- desnoods maandenlang in zwak licht
(\cref{exo:g12:quantum-world:14}) --- en waarbij de intensiteit de \omterm{def:g10:energy-conservation:energy}{energie} van
de elektronen zou moeten vastleggen: alle drie de voorspellingen falen.
\end{proposition}
\admitted

\begin{theorem}[Foto-elektrische betrekking van Einstein]\label{thm:g12:quantum-world:einstein}
\index{foto-elektrische betrekking}Een elektron heeft minstens de
\emph{uittreearbeid}\index{uittreearbeid} $W_0$ van het metaal nodig om het
oppervlak te verlaten. Eén \omterm{def:g12:quantum-world:photon}{foton} wordt in zijn geheel door één elektron
opgenomen, dus
\[
h\nu = W_0 + E_{k,\max}, \qquad \text{en dus } \nu_0 = \frac{W_0}{h}.
\]
\end{theorem}

\begin{proof}
Energieboekhouding: $h\nu$ erin, $W_0$ uitgegeven aan het ontsnappen, de rest
kinetisch --- hoogstens $h\nu - W_0$, en minder voor elektronen die dieper
beginnen. Onder $\nu_0$ kan één korrel de uittreetol niet betalen, hoeveel het
er ook zijn.
\end{proof}

\begin{definition}[Remspanning]\label{def:g12:quantum-world:stopping}
Om $E_{k,\max}$ te meten remt een tegengestelde \omterm{def:g11:circuits-and-power:voltage}{spanning} de elektronen af
(\cref{ch:g11:electric-gravitational-fields}); de
\emph{remspanning}\index{remspanning} $U_s$ is de kleinste \omterm{def:g11:circuits-and-power:voltage}{spanning} die de
stroom opheft: de snelste elektronen halen het net niet, dus is
$E_{k,\max} = e\,U_s$.
\end{definition}

\begin{method}[De constante van Planck meten]\label{met:g12:quantum-world:measure-h}
\begin{enumerate}
  \item Belicht een fotocel door \omterm{def:g11:color-light-sources:subtractive}{filters} met bekende \omterm{def:g12:mechanical-waves:wavelength}{golflengten}; bereken
        telkens $\nu = c/\lambda$; noteer elke \omterm{def:g12:quantum-world:stopping}{remspanning}.
  \item Zet $E_{k,\max} = eU_s$ uit tegen $\nu$: Einstein belooft de rechte
        $h\nu - W_0$.
  \item Lees de helling af: die \emph{is} $h$; de snijpunten geven $\nu_0$ en
        $-W_0$.
\end{enumerate}
\end{method}

\section{Energieniveaus: waarom atomen lijnen uitzenden}

\begin{definition}[Energieniveaus]\label{def:g12:quantum-world:levels}
De \omterm{def:g10:energy-conservation:energy}{energie} van een elektron dat in een \omterm{def:g11:fundamental-interactions:ladder}{atoom} gebonden is, is
\emph{gekwantiseerd}\index{kwantisering}: beperkt tot een discrete ladder van
\emph{energieniveaus}\index{energieniveau} $E_1 < E_2 < \dots < 0$, negatief
geteld vanaf het vrije elektron. De laagste sport is de
\emph{grondtoestand}\index{grondtoestand}, de andere zijn \emph{aangeslagen
toestanden}\index{aangeslagen toestand}; $0$ bereiken is
\emph{ionisatie}\index{ionisatie}.
\end{definition}

\begin{proposition}[De waterstofladder]\label{prop:g12:quantum-world:hydrogen}
De \omterm{def:g12:quantum-world:levels}{energieniveaus} van het waterstofatoom zijn
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2}, \qquad n = 1, 2, 3, \dots
\]
dus $E_1 = \qty{-13.6}{eV}$, $E_2 = \qty{-3.40}{eV}$,
$E_3 = \qty{-1.51}{eV}$, $E_4 = \qty{-0.85}{eV}$, samendringend naar $0$.
\end{proposition}
\admitted

\begin{remark}[Waar de ladder vandaan komt]\label{rem:g12:quantum-world:ladder-origin}
Waarom juist deze getallen, is kwantummechanica, bewaard voor de
universitaire volumes --- en zo ook de formule van De Broglie: de \omterm{def:g12:quantum-world:debroglie}{materiegolf}
van het elektron (\cref{def:g12:quantum-world:debroglie} hieronder) moet rond
de kern op zichzelf aansluiten, en alleen bepaalde \omterm{def:g12:mechanical-waves:wavelength}{golflengten} passen --- net
als een gitaarsnaar heeft een \omterm{def:g11:fundamental-interactions:ladder}{atoom} een discrete verzameling noten.
\end{remark}

\begin{theorem}[Foton van een overgang]\label{thm:g12:quantum-world:transition}
Een \omterm{def:g11:fundamental-interactions:ladder}{atoom} verandert alleen van niveau door één \omterm{def:g12:quantum-world:photon}{foton} uit te zenden of op te
nemen dat precies het verschil draagt: vallen van niveau $p$ naar een lager
niveau $n$ zendt $h\nu = E_p - E_n$ uit; klimmen van $n$ naar $p$ neemt
diezelfde \omterm{def:g10:energy-conservation:energy}{energie} op.
\end{theorem}

\begin{proof}
\omterm{thm:g10:energy-conservation:conservation}{Behoud van energie}, korrel per korrel: het \omterm{def:g12:quantum-world:photon}{foton} is de enige munt die licht
aanneemt, en de boeken van het \omterm{def:g11:fundamental-interactions:ladder}{atoom} moeten kloppen.
\end{proof}

\begin{remark}[Lijnenspectra verklaard]\label{rem:g12:quantum-world:lines}
Hier is het antwoord dat \cref{ch:g10:light-spectra} beloofde: een heet gas
zendt alleen de \omterm{def:g12:oscillators-and-time:oscillator}{frequenties} $(E_p - E_n)/h$ van zijn eigen ladder uit --- een
streepjescode van heldere lijnen --- en een koud gas steelt precies die
\omterm{def:g12:oscillators-and-time:oscillator}{frequenties} uit \omterm{def:g10:light-spectra:white}{wit licht} en drukt er passende donkere absorptielijnen voor
terug. Elk element heeft zijn eigen ladder en dus zijn eigen streepjescode ---
zo lezen we het licht van de sterren.
\end{remark}

\begin{example}[De zichtbare lijnen van waterstof]\label{ex:g12:quantum-world:balmer}
Overgangen die op $n = 2$ eindigen, vallen in het zichtbare. Met
$E \,(\unit{eV}) = 1240/\lambda\,(\unit{nm})$: $3 \to 2$: \qty{1.89}{eV},
\qty{656}{nm} (rood); $4 \to 2$: \qty{2.55}{eV}, \qty{486}{nm} (blauwgroen);
$5 \to 2$: \qty{2.86}{eV}, \qty{434}{nm} (violet). Overgangen naar $n = 1$
gaan boven \qty{10}{eV}: allemaal \omterm{def:g10:light-spectra:wavelength}{ultraviolet}.
\end{example}

\begin{remark}[Lasers, in één adem]\label{rem:g12:quantum-world:laser}
Een \omterm{def:g12:quantum-world:photon}{foton} met precies $E_p - E_n$ dat een \emph{aangeslagen} \omterm{def:g11:fundamental-interactions:ladder}{atoom} passeert,
kan het aanzetten om een identiek \omterm{def:g12:quantum-world:photon}{foton} uit te zenden --- dezelfde \omterm{def:g10:energy-conservation:energy}{energie},
dezelfde richting, in de pas. Houd meer atomen boven dan beneden en één \omterm{def:g12:quantum-world:photon}{foton}
wordt een lawine van klonen: \emph{gestimuleerde
emissie}\index{gestimuleerde emissie} --- en een \omterm{def:g11:color-light-sources:lamps}{laser} is weinig meer.
\end{remark}

\section{Materiegolven}

\begin{proposition}[Elektronen interfereren]\label{prop:g12:quantum-world:interference}
Stuur elektronen één voor één door een dubbele spleet
(\cref{ch:g12:light-as-wave}): elk landt als één scherpe stip --- een deeltje.
Maar naarmate er duizenden samenkomen, tekenen de stippen
\omterm{def:g12:light-as-wave:fringes}{interferentiestrepen}: elk elektron gaat in zijn eentje als een \omterm{def:g10:signals-and-waves:wave}{golf} door
\emph{beide} spleten. Kristallen buigen elektronenbundels ook af (Davisson en
Germer, 1927).
\end{proposition}
\admitted

\begin{definition}[Debroglie-golflengte]\label{def:g12:quantum-world:debroglie}
Elk deeltje met \omterm{def:g12:newtons-laws:momentum}{impuls} $p = mv$ reist als een
\emph{materiegolf}\index{materiegolf} met
\emph{debroglie-golflengte}\index{debroglie-golflengte}
\[
\lambda = \frac{h}{p} = \frac{h}{mv}.
\]
\end{definition}
\admitted

\begin{example}[Waarom jij nooit buigt]\label{ex:g12:quantum-world:never}
Een elektron dat door \qty{100}{V} is versneld heeft
$E_k = \qty{1.60e-17}{J}$ en $p = \sqrt{2m_e E_k} = \qty{5.4e-24}{kg.m/s}$,
dus $\lambda \approx \qty{0.12}{nm}$ --- de afstand tussen de atomen in een
kristal, dat daarmee zijn dubbele spleet wordt. Een tennisbal van \qty{58}{g}
met \qty{25}{m/s}: $\lambda = 6.63\times10^{-34}/1.45 \approx
\qty{4.6e-34}{m}$, zo'n $10^{19}$ keer kleiner dan een kern --- geen enkele
spleet zal hem ooit zien \omterm{prop:g12:light-as-wave:conditions}{interfereren}. $h$ is zo klein dat de kwantumkorrel
alleen op de schaal van het \omterm{def:g11:fundamental-interactions:ladder}{atoom} bovenkomt.
\end{example}

\begin{example}[De elektronenmicroscoop]\label{ex:g12:quantum-world:microscope}
\omterm{def:g12:light-as-wave:diffraction}{Buiging} verbiedt het onderscheiden van detail dat veel kleiner is dan de
gebruikte \omterm{def:g12:mechanical-waves:wavelength}{golflengte} (\cref{ch:g12:light-as-wave}): zichtbaar licht houdt op
rond \qty{0.5}{\micro m}. Elektronen bij \qty{10}{kV} hebben
$\lambda \approx \qty{12}{pm}$ --- tienduizenden keren korter:
elektronenmicroscopen beelden virussen af, het raderwerk van cellen, zelfs
losse atomen. De \omterm{def:g12:quantum-world:debroglie}{materiegolf} gaf de natuurkunde een nieuw oog.
\end{example}

\section{Probleem: Het natuurkundelab van het zonnepaneel}

\begin{problem}[{Weekendopgave --- het natuurkundelab van het zonnepaneel:
de $10^{21}$ korrels die een dak elke seconde opdrinkt, een fotocel die de
\omterm{def:g12:quantum-world:photon}{constante van Planck} meet, een waterstoflamp lijn voor lijn gelezen --- één
weekend, één $h$}]
\label{pb:g12:quantum-world:1}
Jullie klas voorziet het zonnepaneel op het schooldak een weekend lang van
meetapparatuur. Gegevens: zonne-instraling om twaalf uur \qty{1.0e3}{W/m^2};
gemiddelde \omterm{def:g12:mechanical-waves:wavelength}{golflengte} van de zon \qty{550}{nm}; oppervlakte van het paneel
\qty{1.0}{m^2}, \omterm{def:g10:energy-conservation:efficiency}{rendement} $20\%$; de dag levert het equivalent van
\qty{5.0}{h} volle zon.

\medskip
\noindent\textbf{Deel I --- Fotonenboekhouding.}
\begin{enumerate}
  \item Bereken de \omterm{def:g10:energy-conservation:energy}{energie} van één gemiddeld zonnefoton, in \omterm{def:g11:work-of-force:work}{joule} en in
        \omterm{def:g12:nuclear-energy:units}{elektronvolt}.
  \item Leid daaruit het aantal \omterm{def:g12:quantum-world:photon}{fotonen} af dat om twaalf uur per seconde op
        het paneel valt.
  \item Bereken het elektrische \omterm{def:g11:work-of-force:power}{vermogen} van het paneel om twaalf uur, en
        daarna de over de dag geproduceerde \omterm{def:g10:energy-conservation:energy}{energie}, in \unit{kW.h}.
  \item De stroom van het paneel oogt volmaakt glad. Leg met vraag 2 uit
        waarom de korrel-voor-korrelaankomst onzichtbaar is.
  \item Zonlicht draagt meer infrarode dan ultraviolette \omterm{def:g10:energy-conservation:energy}{energie}, en toch
        verbrandt alleen het \omterm{def:g10:light-spectra:wavelength}{ultraviolet} de huid (bindingen
        $\sim \qtyrange{3}{4}{eV}$). Leg dat in korrels uit.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- $h$ meten met de fotocel uit de doos.}
De fotocel uit de doos heeft een onbekende kathode. De \omterm{def:g11:color-light-sources:subtractive}{filters} geven:
\begin{center}\small
\begin{tabular}{lcccc}
$\lambda$ (\unit{nm}) & 365 & 405 & 436 & 546\\
$U_s$ (\unit{V}) & 1.50 & 1.16 & 0.94 & 0.37
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item Wat meet de \omterm{def:g12:quantum-world:stopping}{remspanning}? Leg het verband tussen $U_s$, $\nu$, $W_0$ en
        $h$.
  \item Reken de vier \omterm{def:g12:mechanical-waves:wavelength}{golflengten} om naar \omterm{def:g12:oscillators-and-time:oscillator}{frequenties} (in
        $10^{14}\,\unit{Hz}$).
  \item Waarom zou $eU_s$ tegen $\nu$ uitzetten een rechte moeten geven?
        Controleer dat de punten op één lijn liggen.
  \item Bereken uit de uiterste punten de helling: jouw gemeten $h$.
  \item Leid de \omterm{thm:g12:quantum-world:einstein}{uittreearbeid} van de kathode af (in \unit{eV}), en haar
        \omterm{prop:g12:quantum-world:facts}{drempelfrequentie} en drempelgolflengte.
  \item Welk deel van het zonlicht zou elektronen uit deze kathode kunnen
        losslaan --- en welke helft van het \omterm{def:g12:sound-acoustics:timbre}{spectrum} is er nutteloos voor?
  \item Vergelijk jouw $h$ met \qty{6.63e-34}{J.s}: hoe groot is de relatieve
        fout in procent?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De waterstoflamp als ijkbron.}
De golflengteschaal van de doos wordt aan een waterstoflamp getoetst,
$E_n = -13.6/n^2$ \unit{eV}.
\begin{enumerate}[resume]
  \item Bereken $E_2$, $E_3$, $E_4$ en $E_5$.
  \item Bereken de fotonenergieën van de overgangen $3 \to 2$, $4 \to 2$ en
        $5 \to 2$.
  \item Leid de drie \omterm{def:g12:mechanical-waves:wavelength}{golflengten} af en geef de kleur van elke lijn.
  \item Waarom zendt de lamp \emph{lijnen} uit en geen doorlopende regenboog?
        Eén zin, met het kernidee erin.
  \item Zou het waterstof van de lamp licht van \qty{500}{nm} kunnen opnemen?
        Onderbouw met de ladder.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De cellen inspecteren: de elektronengolf.}
Microscheurtjes in een cel liggen ver onder het bereik van \qty{0.5}{\micro m}
van lichtmicroscopen; de rasterelektronenmicroscoop van het lab werkt op
\qty{10}{kV}.
\begin{enumerate}[resume]
  \item Bereken de snelheid van de elektronen; ga na dat die onder $0.25\,c$
        blijft (klassieke formule).
  \item Bereken hun \omterm{def:g12:newtons-laws:momentum}{impuls} en hun \omterm{def:g12:quantum-world:debroglie}{debroglie-golflengte}.
  \item Het oplossend \omterm{def:g11:work-of-force:power}{vermogen} schaalt met de \omterm{def:g12:mechanical-waves:wavelength}{golflengte}: bereken de winst ten
        opzichte van groen licht (\qty{550}{nm}); sluit af met de twee getallen
        van het weekend --- jouw $h$ en deze winst.
\end{enumerate}
\end{problem}
