\chapter{Arbeid en mechanische energie}\label{ch:g12:work-and-energy}

Laat een knikker in een slakom vallen: hij duikt, schiet de bodem voorbij,
klimt tot zijn beginhoogte, aarzelt, keert terug --- en elke omkering laat
zich voorspellen uit één kromme (het profiel van de kom) en één horizontale
lijn (de \omterm{def:g10:energy-conservation:energy}{energie} van de knikker), zonder ook maar één
bewegingsvergelijking op te lossen. Vorig jaar zetten we een prijs op
\omterm{def:g10:inertia:force}{krachten} langs rechte lijnen (\cref{ch:g11:work-of-force}) en wogen we
kinetische tegen \omterm{def:g12:work-and-energy:ep}{potentiële energie} (\cref{ch:g11:mechanical-energy}); dit
jaar gaat de boekhouding overal mee naartoe: kromme wegen, veren, landschappen
in één oogopslag.

\section{Arbeid langs welke weg dan ook}

\begin{definition}[Arbeid langs een weg]\label{def:g12:work-and-energy:path}
Op een lichaam dat langs een willekeurige weg van $A$ naar $B$ beweegt, werkt een
\omterm{def:g10:inertia:force}{kracht} $\vect F$: knip de weg in verplaatsingen $\vect{\dd\ell}$ die
zo kort zijn dat elke recht is en $\vect F$ er nauwelijks langs verandert.
Elke stap levert de kleine \omterm{def:g11:work-of-force:work}{arbeid} $\scal{F}{\dd\ell} = F\,\dd\ell\cos\theta$
op (\cref{ch:g11:work-of-force}), en de \emph{arbeid langs de
weg}\index{arbeid!langs een weg} is hun som,
$W_{AB}(\vect F) = \sum \scal{F}{\dd\ell}$ --- wat op een rechte weg met een
constante \omterm{def:g10:inertia:force}{kracht} inklapt tot de $F \times AB \times \cos\theta$ van vorig
jaar.
\end{definition}

\begin{proposition}[Arbeid van het gewicht, langs elke weg]\label{prop:g12:work-and-energy:weight}
Langs \emph{elke} weg van $A$ (hoogte $z_A$) naar $B$ (hoogte $z_B$) verricht
het \omterm{def:g10:universal-gravitation:weight}{gewicht} van een massa $m$ de \omterm{def:g11:work-of-force:work}{arbeid} $W_{AB}(\vect P) = mg\,(z_A - z_B)$:
de daling telt mee, de route nooit.
\end{proposition}

\begin{proof}
Elke korte stap draagt $mg \times (\text{kleine daling})$ bij (de berekening
op rechte stukken van vorig jaar); de dalingen tellen telescopisch op tot
$z_A - z_B$. Het kromme geval, toen toegegeven, is hiermee geleverd.
\end{proof}

\begin{definition}[Conservatieve en niet-conservatieve krachten]\label{def:g12:work-and-energy:conservative}
Een \omterm{def:g10:inertia:force}{kracht} heet een \emph{conservatieve kracht}\index{conservatieve kracht}
wanneer haar \omterm{def:g11:work-of-force:work}{arbeid} tussen twee punten langs elke weg dezelfde is (nul op elke
rondreis), en anders een \emph{niet-conservatieve
kracht}\index{niet-conservatieve kracht}. Het \omterm{def:g10:universal-gravitation:weight}{gewicht} en de veerkracht
(volgende paragraaf) zijn conservatief. Glijdende \omterm{def:g10:inertia:inventory}{wrijving} is dat niet: ze
werkt de beweging tegen en verricht $W = -fL$ op een weg met lengte $L$ ---
tol per \omterm{def:g10:orders-of-magnitude:unit}{meter}, nooit terugbetaald.
\end{definition}

\begin{example}[Twee paden naar de hut]\label{ex:g12:work-and-energy:trails}
Een wandelaar van \qty{75}{kg} wint \qty{300}{m} hoogte via een steil pad van
\qty{800}{m} of via \qty{2.4}{km} haarspeldbochten. Het \omterm{def:g10:universal-gravitation:weight}{gewicht} rekent op
beide $-75 \times 9.81 \times 300 \approx \qty{-2.2e5}{J}$; een
wrijvingskracht van \qty{30}{N} rekent \qty{-2.4e4}{J} op het ene en
\qty{-7.2e4}{J} op het andere. Het \omterm{def:g10:universal-gravitation:weight}{gewicht} betaalt bij de afdaling terug; de
rekening van de \omterm{def:g10:inertia:inventory}{wrijving} is als warmte weg, beide kanten op.
\end{example}

\section{Potentiële energie}

\begin{definition}[Potentiële energie]\label{def:g12:work-and-energy:ep}
Aan elke \omterm{def:g12:work-and-energy:conservative}{conservatieve kracht} hangt een \emph{potentiële
energie}\index{potentiële energie!uit conservatieve arbeid} $E_p$, een getal
dat alleen van de plaats afhangt, vastgelegd door
$W_{AB} = E_p(A) - E_p(B)$: geleverde \omterm{def:g11:work-of-force:work}{arbeid} is uitgegeven potentiële
\omterm{def:g10:energy-conservation:energy}{energie}. Door $E_p = 0$ in een gekozen referentiepunt vast te leggen, ligt ze
overal vast; de keuze verschuift alle waarden met een constante die uit elk
verschil wegvalt. Voor het \omterm{def:g10:universal-gravitation:weight}{gewicht} geeft dat de $E_p = mgz$ van vorig jaar
terug, nu langs elke weg geldig. \omterm{def:g10:inertia:inventory}{Wrijving} kan geen $E_p$ bezitten: een
rekening die van de route afhangt is geen verschil van twee getallen aan de
uiteinden --- alleen \omterm{def:g12:work-and-energy:conservative}{conservatieve krachten} slaan terughaalbaar op, precies
wat de naam bewaart.
\end{definition}

\begin{proposition}[Elastische potentiële energie]\label{prop:g12:work-and-energy:elastic}
Een veer met \omterm{def:g12:oscillators-and-time:spring}{veerconstante} $k$ (\unit{N/m}) die over $x$ is uitgerekt of
ingedrukt, trekt terug met de \omterm{def:g10:inertia:force}{kracht} $F = kx$
(\cref{ch:g12:oscillators-and-time}); de \omterm{def:g11:work-of-force:work}{arbeid} die ze levert als ze naar de
rustlengte terugkeert --- haar \emph{elastische potentiële
energie}\index{potentiële energie!elastisch} --- is $E_p = \tfrac12\, k x^2$.
\end{proposition}

\begin{proof}
De \omterm{def:g12:oscillators-and-time:pendulum}{terugdrijvende kracht} werkt langs de beweging, dus is $\sum F\,\dd\ell$ de
\emph{oppervlakte onder de $F$--$x$-grafiek}: een driehoek met basis $x$ en
hoogte $kx$, met oppervlakte $\tfrac12 x \times kx$. De \omterm{def:g10:inertia:force}{kracht} bij elke
uitrekking is heen en terug dezelfde: de \omterm{def:g11:work-of-force:work}{arbeid} ligt vast door de uiteinden,
de veerkracht is conservatief, en $E_p$ is goed gedefinieerd.
\end{proof}

\section{De twee energiestellingen}

\begin{theorem}[Kinetische-energiestelling]\label{thm:g12:work-and-energy:ketheorem}\index{kinetische-energiestelling}
Voor elke beweging van een lichaam met massa $m$ van $A$ naar $B$ --- krom,
in lussen, driedimensionaal --- geldt
\[
\Delta E_k = E_k(B) - E_k(A) = \sum W_{AB}(\vect F),
\]
waarbij de som over alle \omterm{def:g10:inertia:force}{krachten} op het lichaam loopt.
\end{theorem}

\begin{proof}
Omdat $v^2 = \scal{v}{v}$, geeft de productregel uit de wiskunde van dit jaar
de afgeleide van $v^2$ als $2\,\scal{a}{v}$, dus is
$\dd E_k/\dd t = m\,\scal{a}{v} = \scal{F_{\text{tot}}}{v}$ volgens de tweede
wet van Newton (\cref{ch:g12:newtons-laws}). In een korte $\dd t$ verplaatst
het lichaam zich over $\vect{\dd\ell} = \vect v\,\dd t$: $E_k$ wint de kleine
\omterm{def:g11:work-of-force:work}{arbeid} van \cref{def:g12:work-and-energy:path}; de stappen optellen telt de
\omterm{def:g11:work-of-force:work}{arbeiden} op --- de vorig jaar toegegeven uitspraak, nu volledig verdiend.
\end{proof}

\begin{proposition}[Ogenblikkelijk vermogen]\label{prop:g12:work-and-energy:power}
Op elk ogenblik levert een \omterm{def:g10:inertia:force}{kracht} $\vect F$ op een lichaam met snelheid
$\vect v$ het \omterm{def:g11:work-of-force:power}{vermogen} $P = \scal{F}{v} = Fv\cos\theta$, en is
$\dd E_k/\dd t$ de som van de \omterm{def:g11:work-of-force:power}{vermogens} van alle \omterm{def:g10:inertia:force}{krachten}: de formule van
vorig jaar bij constante snelheid, nu exact op elk ogenblik van elke beweging.
\end{proposition}

\begin{proof}
Lees het vorige bewijs af: elke \omterm{def:g10:inertia:force}{kracht} voert tijdens $\dd t$ de bijdrage
$\scal{F}{\dd\ell} = \scal{F}{v}\,\dd t$ aan $E_k$ toe.
\end{proof}

\begin{theorem}[Mechanische-energiestelling]\label{thm:g12:work-and-energy:emtheorem}\index{mechanische-energiestelling}
Zij $E_m = E_k + E_p$, waarbij $E_p$ de potentiële energieën van alle
\omterm{def:g12:work-and-energy:conservative}{conservatieve krachten} die \omterm{def:g11:work-of-force:work}{arbeid} verrichten verzamelt. Dan geldt, langs elke
weg,
\[
\Delta E_m = W_{AB}(\text{niet-conservatieve krachten}),
\]
en blijft $E_m$ behouden zodra de \omterm{def:g12:work-and-energy:conservative}{niet-conservatieve krachten} geen \omterm{def:g11:work-of-force:work}{arbeid}
verrichten.
\end{theorem}

\begin{proof}
Splits de kinetische-energiestelling: $\Delta E_k = W_{\text{cons}} +
W_{\text{nc}} = -\Delta E_p + W_{\text{nc}}$; breng $\Delta E_p$ naar de
andere kant.
\end{proof}

\begin{example}[De slinger, dit keer eerlijk]\label{ex:g12:work-and-energy:pendulum}
Een slingerlichaam aan een draad van lengte $L = \qty{2.5}{m}$ wordt uit rust
losgelaten op $\qty{45}{\degree}$ met de verticaal. De \omterm{def:g10:inertia:inventory}{spankracht} staat
loodrecht op de beweging en is machteloos; het \omterm{def:g10:universal-gravitation:weight}{gewicht} is conservatief:
$E_m$ blijft behouden \emph{op de kromme boog}, wat vorig jaar alleen
gecontroleerd kon worden. Het lichaam begint $h = L(1 -
\cos\qty{45}{\degree}) = \qty{0.73}{m}$ hoger, dus is onderaan
$v = \sqrt{2gh} \approx \qty{3.8}{m/s}$.
\end{example}

\begin{method}[Energieaudit van elke beweging]\label{met:g12:work-and-energy:audit}
\begin{enumerate}
  \item Kies het systeem en noem de \omterm{def:g10:inertia:force}{krachten} erop.
  \item Sorteer: loodrecht op de beweging $\to$ geen \omterm{def:g11:work-of-force:work}{arbeid}; conservatief
        $\to$ een $E_p$ in $E_m$; de rest $\to$ $W_{\text{nc}}$ (glijdende
        \omterm{def:g10:inertia:inventory}{wrijving}: $-fL$, met $L$ de lengte van de weg).
  \item Schrijf $\Delta E_m = W_{\text{nc}}$ tussen het gegeven punt en het
        gevraagde punt; los op. \omterm{def:g10:energy-conservation:energy}{Energie} antwoordt op ``hoe snel, hoe hoog'';
        alleen tijd en versnelling vragen de wetten van Newton.
\end{enumerate}
\end{method}

\section{Energiediagrammen}

\begin{definition}[Energiediagram, keerpunten]\label{def:g12:work-and-energy:diagram}
Voor een beweging langs één as $x$ met behouden $E_m$ zet het
\emph{energiediagram}\index{energiediagram} de kromme $E_p(x)$ en de
horizontale lijn $E_m$ uit. Omdat $E_k = E_m - E_p \geq 0$, blijft de beweging
beperkt tot waar de kromme onder de lijn ligt; de kloof ertussen is $E_k$,
rechtstreeks af te lezen. In een \emph{keerpunt}\index{keerpunt}, waar kromme
en lijn elkaar raken, is $v = 0$ en keert de beweging om.
\end{definition}

\begin{proposition}[Kracht uit het landschap; evenwichten]\label{prop:g12:work-and-energy:equilibria}
De \omterm{def:g12:work-and-energy:conservative}{conservatieve kracht} langs $x$ is min de helling van $E_p$:
$F = -\,\dd E_p/\dd x$ --- ze wijst op het diagram \emph{bergaf}, en verdwijnt
waar de raaklijn horizontaal is: een \omterm{def:g10:inertia:compensate}{evenwicht}. Een minimum van $E_p$ is een
\emph{stabiel evenwicht}\index{stabiel evenwicht} --- het uitgeweken lichaam
wordt teruggeduwd, een bal in een dal; een maximum is een \emph{labiel
evenwicht}\index{labiel evenwicht} --- het wordt weggeduwd, een bal op een
heuveltop.
\end{proposition}

\begin{proof}
Op een kleine stap $\dd x$ verricht de \omterm{def:g10:inertia:force}{kracht} $F\,\dd x = E_p(x) -
E_p(x + \dd x) = -\dd E_p$; deel door $\dd x$. Aan beide zijden van een dal
wijst bergaf naar binnen; rond een heuveltop naar buiten.
\end{proof}

\begin{remark}[Trillingen en ontsnappen]\label{rem:g12:work-and-energy:escape}
De kleine trillingen van \cref{ch:g12:oscillators-and-time} wonen op de bodem
van de dalen van $E_p$. En ver van de Aarde \omterm{def:g11:nucleus-radioactivity:abg}{verzwakt} de \omterm{def:g11:fundamental-interactions:four}{gravitatie}: de kromme
$E_p(r)$ klimt steeds trager naar een eindig plafond (de oppervlakte onder de
krimpende krachtgrafiek is eindig). Een raket waarvan $E_m$ dat plafond haalt,
komt nooit een \omterm{def:g12:work-and-energy:diagram}{keerpunt} tegen: ze haalt de
\emph{ontsnappingssnelheid}\index{ontsnappingssnelheid} --- ongeveer
\qty{11.2}{km/s} vanaf de grond (\cref{exo:g12:work-and-energy:13}); daaronder
blijft ze gebonden. Het plafond zelf wordt in het volume van bachelorjaar 1
berekend.
\end{remark}

\begin{example}[De looping]\label{ex:g12:work-and-energy:loop}
Een knikker die van hoogte $H$ wordt losgelaten, rolt een verticale lus met
straal $R$ in. De dynamica van de cirkelbeweging
(\cref{ch:g12:newtons-laws}) eist bovenin $v_{\text{top}}^2 \geq gR$ --- het
gegeven dat het achtbaanprobleem van vorig jaar moest lenen en dat nu van ons
is. Behoud van rust op $H$ tot de top (hoogte $2R$) geeft
$v_{\text{top}}^2 = 2g(H - 2R)$, dus $H \geq \tfrac52 R$: een halve straal
boven de top van de lus, wat de massa ook is --- iets meer in de echte,
schurende wereld.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:1}
Een veer met \omterm{def:g12:oscillators-and-time:spring}{veerconstante} $k = \qty{200}{N/m}$ wordt \qty{5.0}{cm}

verdubbelt? Hoeveel kost de \emph{tweede} \qty{5.0}{cm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:2}
Een drone van \qty{0.90}{kg} vliegt langs een grillige route van een terras op
$z = \qty{12}{m}$ naar een balkon op $z = \qty{30}{m}$. Hoe groot is de \omterm{def:g11:work-of-force:work}{arbeid}
van zijn \omterm{def:g10:universal-gravitation:weight}{gewicht}? En langs de rechte klim? En op de heen-en-terugvlucht?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:3}
Conservatief, niet-conservatief of arbeidsloos? Onderbouw met de wegen:
(a) het \omterm{def:g10:universal-gravitation:weight}{gewicht}; (b) de veerkracht; (c) glijdende \omterm{def:g10:inertia:inventory}{wrijving}; (d) de
\omterm{def:g12:newtons-laws:friction}{normaalkracht} op een glijdende doos; (e) de luchtweerstand.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:4}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:work-and-energy:5}

\section{Probleem: De schans}

\begin{problem}[{Weekendopgave --- de schans: een aanloop doorgelicht met
een snelheidsmeter, een vlucht in vrije val, de zachte meetkunde van een
steile landing, en de toren die een ontwerper moet kopen zodra de wrijving
haar rekening stuurt}]
\label{pb:g12:work-and-energy:1}
Een olympische schansspringer ($m = \qty{70}{kg}$, ski's inbegrepen) vertrekt
uit rust van een startpoort $H = \qty{45}{m}$ boven de schanstafel, glijdt
$L = \qty{90}{m}$ kromme aanloop en verlaat de tafel horizontaal; een
snelheidsmeter leest daar $v_0 = \qty{26}{m/s}$. Het landingspunt ligt
$h = \qty{40}{m}$ lager, op een helling van $\qty{35}{\degree}$. De delen II
en III verwaarlozen de luchtweerstand.

\medskip\noindent\textbf{Deel I --- De aanloop.}
\begin{enumerate}
  \item Voorspel de wrijvingsloze snelheid op de tafel; waarom doet de kromming
        van de aanloop niet ter zake?
  \item Met het \omterm{def:g11:mechanical-energy:potential}{referentieniveau} op de tafel: bereken $E_m$ in de startpoort en
        op de tafel.
  \item Bereken $\Delta E_m$; welke stelling noemt de schuldigen, en wie zijn
        het?
  \item Leid de gemiddelde remmende \omterm{def:g10:inertia:force}{kracht} (sneeuw plus lucht) over de aanloop
        af.
  \item Welke fractie van $E_m$ gaat verloren? Waarom de ski's waxen en
        ineengedoken zitten?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De vlucht.}
\begin{enumerate}[resume]
  \item In de vlucht werkt alleen het \omterm{def:g10:universal-gravitation:weight}{gewicht}: welke beweging uit
        \cref{ch:g12:projectile-motion} is dit? Wat zijn de componenten van de
        snelheid bij de landing?
  \item Bereken de landingssnelheid twee keer: uit de componenten, en uit het
        behoud van $E_m$. Hoeveel is dat in \unit{km/h}?
  \item Bereken de \omterm{def:g12:projectile-motion:range}{vluchttijd} en de horizontale afstand die hij vliegt.
  \item Echte springers berijden de lucht als een vleugel en vliegen veel
        verder: kan de lucht de landingssnelheid \emph{verhogen}? Onderbouw met
        de mechanische-energiestelling.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De landing.}
\begin{enumerate}[resume]
  \item Bereken de hoek van de landingssnelheid onder de horizontaal.
  \item Ontbind haar in componenten evenwijdig aan en loodrecht op de helling.
  \item De benen vangen alleen het loodrechte deel op: bereken die \omterm{def:g11:mechanical-energy:kinetic}{kinetische
        energie} en vergelijk haar met het totaal.
  \item Vlakke grond zou alles moeten opvangen: met welke verticale val komt
        elke landing overeen ($h_{\text{eq}} = v_\perp^2/2g$, respectievelijk
        $v^2/2g$)?
  \item In één zin: waarom zijn landingsheuvels steil en gekromd?
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- De toren van de ontwerper.}
Een kleinere schans vraagt een tafelsnelheid van maar
$v_0' = \qty{24}{m/s}$; haar rechte aanloop daalt onder $\qty{35}{\degree}$,
zodat een poorthoogte $H'$ een baanlengte $H'/\sin\qty{35}{\degree}$ betekent;
dezelfde remmende \omterm{def:g10:inertia:force}{kracht} van \qty{80}{N}.
\begin{enumerate}[resume]
  \item Bereken de poorthoogte die een wrijvingsloze ontwerper zou bouwen.
  \item Schrijf de mechanische-energiestelling van poort tot tafel op voor
        $H'$.
  \item Los op naar $H'$.
  \item Met welk percentage heeft de \omterm{def:g10:inertia:inventory}{wrijving} de toren verhoogd?
  \item De massa valt nu niet meer weg: bereken $H'$ opnieuw voor een springer
        van \qty{60}{kg}. Wie heeft de hoogste toren nodig, en waarom?
  \item Slotzin: de aangekondigde toren, en het aandeel van de \omterm{def:g10:inertia:inventory}{wrijving} erin.
\end{enumerate}
\end{problem}
