\chapter{Optical Instruments and the Eye}\label{ch:b1:optical-instruments}

From the back row of a concert hall a pair of binoculars pulls the
singer's face to within arm's reach; a \omterm{def:b1:optical-instruments:microscope}{microscope} turns a drop of pond
water into a zoo; a phone the thickness of a finger photographs the
Milky Way. Every one of these instruments is a few thin lenses
arranged for one purpose: to present to the \omterm{def:b1:optical-instruments:eye}{eye}, or to a sensor, an
image larger, brighter or sharper than the \omterm{def:b1:optical-instruments:eye}{eye} could form alone. So the
chapter begins with the \omterm{def:b1:optical-instruments:eye}{eye} itself --- what it can and cannot do --- and
then builds the \omterm{def:b1:optical-instruments:G}{magnifier}, the \omterm{def:b1:optical-instruments:microscope}{microscope}, the telescope and the \omterm{def:b1:optical-instruments:camera}{camera}
out of the conjugation relation of the previous chapter.

\section{The eye}

\begin{definition}[Reduced eye]\label{def:b1:optical-instruments:eye}
\index{eye}\index{reduced eye}\index{retina}\index{accommodation}
The \emph{reduced eye} models the cornea and the crystalline \omterm{def:b1:mirrors-thin-lenses:lens}{lens} as a
single thin \omterm{def:b1:mirrors-thin-lenses:lens}{converging lens} of \emph{variable} \omterm{def:b1:mirrors-thin-lenses:lens}{focal length}, and the
\emph{retina} as a screen at a fixed distance $d \approx \qty{17}{mm}$
behind it. Muscles change the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}'s curvature to keep the image of the
object looked at on the retina: this is \emph{accommodation}. The iris
in front of the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} is a variable aperture of diameter \qty{2}{} to
\qty{8}{mm}, the \emph{pupil}.
\end{definition}

\begin{definition}[Far point, near point]\label{def:b1:optical-instruments:punctum}
\index{far point}\index{near point}\index{amplitude of accommodation}
The \emph{far point} (punctum remotum) is the object point seen
sharply without accommodating; the \emph{near point} (punctum proximum)
the closest point seen sharply with maximal \omterm{def:b1:optical-instruments:eye}{accommodation}. For a normal
(emmetropic) \omterm{def:b1:optical-instruments:eye}{eye} the far point is at infinity and the near point at
$d_m \approx \qty{25}{cm}$, the conventional \emph{distance of distinct
vision}. The difference of \omterm{def:b1:mirrors-thin-lenses:lens}{vergences} needed for the two is the
\emph{amplitude of \omterm{def:b1:optical-instruments:eye}{accommodation}}: about \qty{4}{\delta} for a young
adult, falling with age.
\end{definition}

\begin{example}[The eye's vergence]\label{ex:b1:optical-instruments:vergence}
Distant object: image at $d = \qty{17}{mm}$ means $f' = \qty{17}{mm}$,
$V = 1/0.017 = \qty{59}{\delta}$. Object at \qty{25}{cm}:
$V = 1/0.017 + 1/0.25 = 59 + 4 = \qty{63}{\delta}$. Four diopters of
\omterm{def:b1:optical-instruments:eye}{accommodation} out of sixty: the crystalline \omterm{def:b1:mirrors-thin-lenses:lens}{lens} adds only a few percent
to the cornea's job, but those few percent are everything for reading.
\end{example}

\begin{proposition}[Defects and corrections]\label{prop:b1:optical-instruments:defects}
\index{myopia}\index{hyperopia}\index{presbyopia}
A \emph{myopic} \omterm{def:b1:optical-instruments:eye}{eye} is too convergent: its \omterm{def:b1:optical-instruments:punctum}{far point} is at a finite
distance $D_r$; a \omterm{def:b1:mirrors-thin-lenses:lens}{diverging lens} of \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} $f' = -D_r$ (worn
close to the \omterm{def:b1:optical-instruments:eye}{eye}) images infinity at the \omterm{def:b1:optical-instruments:punctum}{far point} and restores distant
vision. A \emph{hyperopic} \omterm{def:b1:optical-instruments:eye}{eye} is not convergent enough: its \omterm{def:b1:optical-instruments:punctum}{far point}
is virtual (behind the \omterm{def:b1:optical-instruments:eye}{eye}) and its \omterm{def:b1:optical-instruments:punctum}{near point} too far; a \omterm{def:b1:mirrors-thin-lenses:lens}{converging
lens} corrects it. \emph{Presbyopia} is the loss of \omterm{def:b1:optical-instruments:eye}{accommodation} with
age: the \omterm{def:b1:optical-instruments:punctum}{near point} recedes and reading needs converging lenses,
whatever the \omterm{def:b1:optical-instruments:punctum}{far point}.
\end{proposition}

\begin{proof}
A \omterm{def:b1:mirrors-thin-lenses:lens}{lens} close to the \omterm{def:b1:optical-instruments:eye}{eye} and the \omterm{def:b1:optical-instruments:eye}{eye} form a system whose object must be
what the \omterm{def:b1:optical-instruments:eye}{eye} can see: the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} must turn an object at infinity into a
\omterm{def:b1:mirrors-thin-lenses:image}{virtual image} at the \omterm{def:b1:optical-instruments:punctum}{far point}, $\overline{OA'} = -D_r$ for
$\overline{OA} = -\infty$, so $f' = \overline{OA'} = -D_r$. The other
cases are the same argument with the relevant points.
\end{proof}

\begin{definition}[Angular size and resolution]\label{def:b1:optical-instruments:angular}
\index{angular size}\index{angular resolution}
The \emph{angular size} of an object of height $h$ at distance $D$ is
$\theta \approx h/D$ (small angles). The \omterm{def:b1:optical-instruments:eye}{eye} distinguishes two points
only if their angular separation exceeds its \emph{angular resolution},
about $\epsilon \approx \qty{3e-4}{rad}$ (one arcminute): the spacing
of the cone cells and diffraction at the pupil both set this limit.
\end{definition}

\begin{remark}[What an instrument is for]\label{rem:b1:optical-instruments:purpose}
Brought to the \omterm{def:b1:optical-instruments:punctum}{near point}, a \qty{1}{mm} detail subtends $1/250 =
\qty{4e-3}{rad}$: thirteen times the resolution limit; a \qty{0.05}{mm}
detail subtends $\qty{2e-4}{rad}$ and is invisible. Instruments that
look at near objects (\omterm{def:b1:optical-instruments:G}{magnifier}, \omterm{def:b1:optical-instruments:microscope}{microscope}) increase the \omterm{def:b1:optical-instruments:angular}{angular size}
beyond what bringing the object closer can do; instruments for distant
objects (telescope) increase the \omterm{def:b1:optical-instruments:angular}{angular size} of things one cannot
approach; the \omterm{def:b1:optical-instruments:camera}{camera} replaces the \omterm{def:b1:optical-instruments:eye}{retina} by a sensor that can integrate
light for seconds and be enlarged afterwards.
\end{remark}

\section{The magnifier}

\begin{definition}[Angular magnification]\label{def:b1:optical-instruments:G}
\index{angular magnification}\index{magnifier}\index{commercial magnification}
For an instrument used by the \omterm{def:b1:optical-instruments:eye}{eye}, the \emph{angular magnification} is
$G = \theta'/\theta$, where $\theta'$ is the \omterm{def:b1:optical-instruments:angular}{angular size} of the image
seen through the instrument and $\theta$ the \omterm{def:b1:optical-instruments:angular}{angular size} of the object
seen with the naked \omterm{def:b1:optical-instruments:eye}{eye} --- at the \omterm{def:b1:optical-instruments:punctum}{near point} $d_m = \qty{25}{cm}$ for
a near object, in place for a distant one.
\end{definition}

\begin{proposition}[The magnifier]\label{prop:b1:optical-instruments:loupe}
A \omterm{def:b1:mirrors-thin-lenses:lens}{converging lens} of \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} $f' < d_m$ with the object in its
object focal plane gives an image at infinity, seen without
\omterm{def:b1:optical-instruments:eye}{accommodation}, with
\[
G = \frac{d_m}{f'} = d_m V .
\]
\end{proposition}

\begin{proof}
Object $AB$ of height $h$ in the focal plane: rays from $B$ leave
parallel, at the angle of the ray through $O$, $\theta' = h/f'$. Naked
\omterm{def:b1:optical-instruments:eye}{eye} at best: $\theta = h/d_m$. Ratio $d_m/f'$.
\end{proof}

\begin{example}[A watchmaker's loupe]\label{ex:b1:optical-instruments:loupe}
$f' = \qty{25}{mm}$: $G = 250/25 = 10$, sold as ``$10\times$''. A
\qty{0.05}{mm} detail then subtends $\qty{2e-3}{rad}$, comfortably
resolved. Pushing $G$ higher means $f'$ of a few millimeters, a tiny
\omterm{def:b1:mirrors-thin-lenses:lens}{lens} held against the \omterm{def:b1:optical-instruments:eye}{eye} --- the limit of the single \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, and the
reason the \omterm{def:b1:optical-instruments:microscope}{microscope} exists.
\end{example}

\section{The microscope}

\begin{definition}[Compound microscope]\label{def:b1:optical-instruments:microscope}
\index{microscope}\index{objective}\index{eyepiece}\index{optical interval}
A \emph{microscope} is an \emph{objective} $L_1$ of short \omterm{def:b1:mirrors-thin-lenses:lens}{focal length}
$f_1'$ forming a real, enlarged intermediate image of a nearby object
in the object focal plane of an \emph{eyepiece} $L_2$ (\omterm{def:b1:mirrors-thin-lenses:lens}{focal length}
$f_2'$), which acts as a \omterm{def:b1:optical-instruments:G}{magnifier}. The distance $\Delta =
\overline{F_1'F_2}$ between the objective's image focus and the
eyepiece's object focus is the \emph{optical interval}, a standardized
\qty{160}{mm} on classical instruments.
\end{definition}

\begin{proposition}[Magnification of a microscope]\label{prop:b1:optical-instruments:microscope}
With the final image at infinity,
\[
\gamma_1 = -\frac{\Delta}{f_1'}, \qquad G_2 = \frac{d_m}{f_2'},
\qquad G = \abs{\gamma_1}\,G_2 = \frac{\Delta\,d_m}{f_1' f_2'} .
\]
\end{proposition}

\begin{proof}
The intermediate image $A_1B_1$ lies in the \omterm{def:b1:optical-instruments:microscope}{eyepiece}'s object focal
plane, i.e.\ at $\overline{F_1'A_1} = \Delta$; Newton's relation gives
$\gamma_1 = -\overline{F_1'A_1}/f_1' = -\Delta/f_1'$. The \omterm{def:b1:optical-instruments:microscope}{eyepiece} then
shows $A_1B_1$ (height $\abs{\gamma_1} h$) at infinity under the angle
$\theta' = \abs{\gamma_1} h/f_2'$, against $\theta = h/d_m$ for the
naked \omterm{def:b1:optical-instruments:eye}{eye}.
\end{proof}

\begin{example}[A student microscope]\label{ex:b1:optical-instruments:micronum}
\omterm{def:b1:optical-instruments:microscope}{Objective} $f_1' = \qty{4.0}{mm}$, \omterm{def:b1:optical-instruments:microscope}{eyepiece} $f_2' = \qty{25}{mm}$,
$\Delta = \qty{160}{mm}$: $\gamma_1 = -40$, $G_2 = 10$, $G = 400$. The
object sits at $\overline{F_1A} = -f_1'^2/\Delta = -16/160 =
\qty{-0.10}{mm}$ from $F_1$: \qty{4.1}{mm} from the \omterm{def:b1:optical-instruments:microscope}{objective}. Focusing
means moving the whole tube by fractions of a millimeter.
\end{example}

\begin{remark}[The resolution limit]\label{rem:b1:optical-instruments:abbe}
\index{resolving power}
Magnification without limit would be useless: light is a wave, and an
\omterm{def:b1:optical-instruments:microscope}{objective} accepting rays within a half-angle $u$ (in a medium of index
$n$) cannot separate two points closer than about
$\lambda/(2n\sin u)$ --- the wave treatment of the Year 2 volume derives
it. With $n\sin u \approx 1.4$ (oil immersion) and $\lambda =
\qty{0.5}{\micro m}$, the limit is near \qty{0.2}{\micro m}. To make
that detail visible to the \omterm{def:b1:optical-instruments:eye}{eye} ($\epsilon = \qty{3e-4}{rad}$ at
\qty{25}{cm}, i.e.\ \qty{75}{\micro m}), a magnification of about $400$
suffices; beyond $1000$ the image is only larger, not richer:
\emph{empty magnification}.
\end{remark}

\section{The astronomical telescope}

\begin{proposition}[Refracting telescope]\label{prop:b1:optical-instruments:telescope}
\index{telescope}\index{exit pupil}
An \omterm{def:b1:optical-instruments:microscope}{objective} $L_1$ and an \omterm{def:b1:optical-instruments:microscope}{eyepiece} $L_2$ with $F_1' = F_2$ form an
\omterm{prop:b1:mirrors-thin-lenses:doublet}{afocal system}: a star (at infinity) is seen at infinity, the relaxed
\omterm{def:b1:optical-instruments:eye}{eye} views it, and
\[
G = -\frac{f_1'}{f_2'} .
\]
The image of the \omterm{def:b1:optical-instruments:microscope}{objective}'s rim through the \omterm{def:b1:optical-instruments:microscope}{eyepiece} is the
\emph{\omterm{prop:b1:optical-instruments:telescope}{exit pupil}}: a bright disk of diameter $D/\abs G$ located
$\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ behind the \omterm{def:b1:optical-instruments:microscope}{eyepiece}, where
the \omterm{def:b1:optical-instruments:eye}{eye}'s pupil must be placed to receive all the light.
\end{proposition}

\begin{proof}
$G$ was derived in \cref{prop:b1:mirrors-thin-lenses:doublet}. The
\omterm{def:b1:optical-instruments:microscope}{objective}, seen from $L_2$, is an object at $\overline{O_2O_1} = -(f_1'
+ f_2')$; Descartes gives $1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2')
= f_1'/[f_2'(f_1' + f_2')]$, and the magnification $\overline{O_2P'}/
\overline{O_2O_1} = -f_2'/f_1' = 1/G$ scales the diameter $D$ to
$D/\abs G$. Every ray entering the \omterm{def:b1:optical-instruments:microscope}{objective} leaves through the \omterm{prop:b1:optical-instruments:telescope}{exit
pupil}, since each point of the \omterm{def:b1:optical-instruments:microscope}{objective} is imaged there.
\end{proof}

\begin{proposition}[What a telescope gains]\label{prop:b1:optical-instruments:gains}
\index{light-gathering power}\index{Rayleigh criterion}
Compared with the naked \omterm{def:b1:optical-instruments:eye}{eye} (pupil $d_p$):
\begin{itemize}
  \item \emph{\omterm{def:b1:optical-instruments:angular}{angular size}}: multiplied by $\abs G$;
  \item \emph{light gathered} from a point source: multiplied by
        $(D/d_p)^2$, provided the \omterm{prop:b1:optical-instruments:telescope}{exit pupil} is not larger than the
        \omterm{def:b1:optical-instruments:eye}{eye}'s pupil, i.e.\ $\abs G \geq D/d_p$;
  \item \emph{\omterm{def:b1:optical-instruments:angular}{angular resolution}}: diffraction at the \omterm{def:b1:optical-instruments:microscope}{objective} limits
        it to $\theta_{\min} \approx 1.22\,\lambda/D$ (Rayleigh
        criterion), far below the \omterm{def:b1:optical-instruments:eye}{eye}'s $\epsilon$; the magnification
        that makes this limit visible is $G_r = \epsilon/\theta_{\min}$,
        and larger magnifications only enlarge the blur.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[An amateur's 150-millimeter refractor]\label{ex:b1:optical-instruments:telnum}
$D = \qty{150}{mm}$, $f_1' = \qty{1200}{mm}$, \omterm{def:b1:optical-instruments:microscope}{eyepiece} \qty{25}{mm}:
$G = -48$; \omterm{prop:b1:optical-instruments:telescope}{exit pupil} \qty{3.1}{mm}; light gain $(150/7)^2 \approx
460$ against a dark-adapted pupil of \qty{7}{mm}; resolution $1.22 \times
\qty{550}{nm}/\qty{0.15}{m} = \qty{4.5e-6}{rad} = 0.9''$, so
craters \qty{1.7}{km} wide on the Moon; $G_r = 3\times10^{-4}/4.5\times
10^{-6} \approx 67$ --- the \qty{6}{mm} \omterm{def:b1:optical-instruments:microscope}{eyepiece} ($G = 200$) shows no
more detail, only a dimmer, bigger disk. The weekend problem designs
this instrument.
\end{example}

\begin{remark}[Reflectors, binoculars, spyglasses]\label{rem:b1:optical-instruments:variants}
A \omterm{def:b1:mirrors-thin-lenses:mirror}{concave mirror} of \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} $f_1'$ replaces the \omterm{def:b1:optical-instruments:microscope}{objective} \omterm{def:b1:mirrors-thin-lenses:lens}{lens} in
the \emph{reflecting} telescope (no chromatic dispersion, and a mirror
can be made meters wide and supported from behind): everything above
holds with $f_1'$ the mirror's \omterm{def:b1:mirrors-thin-lenses:lens}{focal length}. The image is inverted ---
harmless for stars; for terrestrial use, binoculars insert a pair of
total-reflection prisms to erect it, and Galileo's spyglass uses a
diverging \omterm{def:b1:optical-instruments:microscope}{eyepiece} ($f_2' < 0$, $G > 0$), shorter but with a narrow
field.
\end{remark}

\section{The camera}

\begin{definition}[Camera; f-number]\label{def:b1:optical-instruments:camera}
\index{camera}\index{f-number}\index{depth of field}
A \emph{camera} is a \omterm{def:b1:mirrors-thin-lenses:lens}{converging lens} of \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} $f'$ forming a \omterm{def:b1:mirrors-thin-lenses:image}{real
image} on a sensor, at $\overline{OA'} \approx f'$ for distant subjects;
focusing moves the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}. A diaphragm of diameter $D$ limits the beam;
the \emph{f-number} is $N = f'/D$ (written $f/N$). The \emph{angle of
view} is $2\arctan(\ell/2f')$ for a sensor of width $\ell$.
\end{definition}

\begin{proposition}[Exposure and depth of field]\label{prop:b1:optical-instruments:dof}
\index{hyperfocal distance}
The irradiance on the sensor scales as $1/N^2$, so the exposure time
for a given image brightness scales as $N^2$. A point at a distance
other than the focused one forms a blur disk; requiring it to stay
below a tolerated diameter $c$, a \omterm{def:b1:mirrors-thin-lenses:lens}{lens} focused at the \emph{\omterm{prop:b1:optical-instruments:dof}{hyperfocal
distance}}
\[
H = \frac{f'^2}{N c}
\]
renders everything from $H/2$ to infinity acceptably sharp.
\end{proposition}

\begin{proof}
Irradiance $=$ power/area $\propto D^2/f'^2 = 1/N^2$ (the image of a
given subject has a size fixed by $f'$, and the collected power grows
as $D^2$). Focus at distance $p$ with the image at $q \approx f'$;
a point at infinity focuses at $f'$, i.e.\ $q - f'$ before the sensor;
its cone of rays, of base $D$ at the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, has diameter $D(q - f')/q$
on the sensor (similar triangles). With $1/q = 1/f' - 1/p$ paraxially,
$q - f' \approx f'^2/p$ for $p \gg f'$, so the blur is
$Df'/p = f'^2/(Np)$; it equals $c$ for $p = H$. The same geometry on the
near side gives the limit $H/2$.
\end{proof}

\begin{example}[Full frame and phone]\label{ex:b1:optical-instruments:cameras}
A \qty{50}{mm} \omterm{def:b1:mirrors-thin-lenses:lens}{lens} on a \qty{36}{mm} wide sensor: angle of view
$2\arctan(18/50) = 40^\circ$; at $f/8$ with $c = \qty{0.03}{mm}$,
$H = 2500/(8 \times 0.03) = \qty{10}{m}$: focused at \qty{10}{m},
everything beyond \qty{5}{m} is sharp. A phone \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, $f' = \qty{4.3}{mm}$
at $f/1.8$ on a \qty{6}{mm} sensor with $c = \qty{5}{\micro m}$: same
angle of view (\qty{70}{\degree} wide), $H = 18.5/(1.8 \times 0.005) =
\qty{2.1}{m}$ --- almost everything is in focus at once, the reason
phones fake background blur in software.
\end{example}

\section{Problem: Designing a telescope for the Moon}

\begin{problem}[{Weekend problem --- a 150-millimeter refractor on a
garden lawn: choose the eyepieces, place the eye, count the light, and
find out how small a crater it can show}]
\label{pb:b1:optical-instruments:1}
The \omterm{def:b1:optical-instruments:microscope}{objective} is a \omterm{def:b1:mirrors-thin-lenses:lens}{converging lens} $L_1$ of \omterm{def:b1:mirrors-thin-lenses:lens}{focal length}
$f_1' = \qty{1200}{mm}$ and diameter $D = \qty{150}{mm}$; \omterm{def:b1:optical-instruments:microscope}{eyepieces}
$L_2$ of \omterm{def:b1:mirrors-thin-lenses:lens}{focal lengths} \qty{25}{mm} and \qty{6}{mm} are available. The
Moon is \qty{3.84e5}{km} away and subtends $0.52^\circ$; the \omterm{def:b1:optical-instruments:eye}{eye}'s
night pupil is \qty{7}{mm}, its resolution $\epsilon = \qty{3.0e-4}{rad}$;
$\lambda = \qty{550}{nm}$.

\medskip
\noindent\textbf{Part I --- The afocal arrangement.}
\begin{enumerate}
  \item Where must the \omterm{def:b1:optical-instruments:microscope}{eyepiece} be placed for a star to be seen without
        \omterm{def:b1:optical-instruments:eye}{accommodation}? Give the objective--eyepiece distance for each
        \omterm{def:b1:optical-instruments:microscope}{eyepiece}.
  \item Check with Descartes's relation, applied to each \omterm{def:b1:mirrors-thin-lenses:lens}{lens} in turn,
        that a point at infinity on the axis is imaged at infinity.
  \item Show that a beam from a star at angle $\alpha$ from the axis
        emerges as a parallel beam at angle $\alpha' = -(f_1'/f_2')\alpha$.
  \item Compute $G$ for each \omterm{def:b1:optical-instruments:microscope}{eyepiece} and the apparent diameter of the
        Moon through each.
  \item The image is inverted. Why is that acceptable here, and what
        would a terrestrial spotting scope add?
  \item Where is the real intermediate image of the Moon, and what is
        its diameter? (It can be photographed there.)
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Where the \omterm{def:b1:optical-instruments:eye}{eye} goes.}
\begin{enumerate}[resume]
  \item Show that the \omterm{def:b1:optical-instruments:microscope}{eyepiece} forms a \omterm{def:b1:mirrors-thin-lenses:image}{real image} of the \omterm{def:b1:optical-instruments:microscope}{objective}'s rim
        at $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ behind it, of
        diameter $D/\abs G$ (the \omterm{prop:b1:optical-instruments:telescope}{exit pupil}).
  \item Compute the position and diameter of the \omterm{prop:b1:optical-instruments:telescope}{exit pupil} for each
        \omterm{def:b1:optical-instruments:microscope}{eyepiece}.
  \item Why must the \omterm{def:b1:optical-instruments:eye}{eye}'s pupil be placed at the \omterm{prop:b1:optical-instruments:telescope}{exit pupil}? What
        happens to the field if it is placed farther back?
  \item For the light of a star to be fully used, the \omterm{prop:b1:optical-instruments:telescope}{exit pupil} must
        not exceed the \omterm{def:b1:optical-instruments:eye}{eye}'s pupil: deduce the minimum useful
        magnification of this telescope at night.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Resolution.}
\begin{enumerate}[resume]
  \item Compute the diffraction limit $\theta_{\min} = 1.22\lambda/D$
        in radians and arcseconds.
  \item What is the smallest crater this angle corresponds to on the
        Moon? And for the naked \omterm{def:b1:optical-instruments:eye}{eye}?
  \item Define and compute the resolving magnification $G_r =
        \epsilon/\theta_{\min}$. Which \omterm{def:b1:optical-instruments:microscope}{eyepiece} is closest to it?
  \item With the \qty{6}{mm} \omterm{def:b1:optical-instruments:microscope}{eyepiece} the image looks blurrier and
        dimmer than with the \qty{25}{mm}. Explain both effects.
  \item Atmospheric turbulence smears star images to about $1''$. For
        which \omterm{def:b1:optical-instruments:microscope}{objective} diameters does the Rayleigh limit stop
        mattering from the ground, and what does a larger $D$ still buy?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Light.}
\begin{enumerate}[resume]
  \item By what factor does the \omterm{def:b1:optical-instruments:microscope}{objective} collect more light than the
        night pupil?
  \item Convert this to magnitudes ($\Delta m = 2.5\log_{10}$ of the
        ratio); if the \omterm{def:b1:optical-instruments:eye}{eye} reaches magnitude $6$, what magnitude does
        the telescope reach?
  \item The full Moon delivers an irradiance of about
        \qty{2e-3}{W/m^2} at the ground. What power enters the
        \omterm{def:b1:optical-instruments:microscope}{objective}, and how is it spread over the \omterm{def:b1:optical-instruments:eye}{retina} compared with
        \omterm{def:b1:optical-instruments:eye}{naked-eye} viewing (same power per unit solid angle argument:
        compare surface brightness through the two \omterm{def:b1:optical-instruments:microscope}{eyepieces})?
  \item Why does a telescope not make the Moon's surface brighter per
        unit area than the naked \omterm{def:b1:optical-instruments:eye}{eye}, whereas it does make stars
        brighter?
\end{enumerate}

\medskip
\noindent\textbf{Part V --- Field of view.}
The \omterm{def:b1:optical-instruments:microscope}{eyepiece} carries, in its focal plane, a circular field stop of
diameter \qty{20}{mm} (\qty{25}{mm} \omterm{def:b1:optical-instruments:microscope}{eyepiece}) or \qty{5}{mm}
(\qty{6}{mm} \omterm{def:b1:optical-instruments:microscope}{eyepiece}).
\begin{enumerate}[resume]
  \item Show that the true field of view is $2\arctan(\phi/2f_1')
        \approx \phi/f_1'$ for a stop of diameter $\phi$.
  \item Compute it for both \omterm{def:b1:optical-instruments:microscope}{eyepieces} and say whether the whole Moon
        fits in the view.
  \item Compute the apparent field (the angle under which the \omterm{def:b1:optical-instruments:eye}{eye} sees
        the stop through the \omterm{def:b1:optical-instruments:microscope}{eyepiece}) for both.
  \item The Earth turns at $15''$ per second of time: how long does the
        Moon take to cross the field of each \omterm{def:b1:optical-instruments:microscope}{eyepiece} without tracking?
  \item Summarize the design: which \omterm{def:b1:optical-instruments:microscope}{eyepiece} for the whole Moon, which
        for craters, and the smallest crater size the instrument can
        reveal --- its one number.
  \item The same \omterm{def:b1:optical-instruments:microscope}{objective} is replaced by a \qty{150}{mm} \omterm{def:b1:mirrors-thin-lenses:mirror}{concave mirror}
        of the same \omterm{def:b1:mirrors-thin-lenses:lens}{focal length}. Which of the above results change?
\end{enumerate}
\end{problem}
