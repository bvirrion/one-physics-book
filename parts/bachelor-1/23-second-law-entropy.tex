\chapter{The Second Law: Entropy}\label{ch:b1:second-law-entropy}

Drop a hot spoon into a cup of cold water: the spoon cools, the water
warms, and the two end at one temperature. The first law would allow
the reverse just as well --- the water cooling further while the spoon
heated up, energy conserved to the joule --- yet it never happens. A
film run backward is instantly recognizable; the molecules obey laws
that have no arrow, and the world has one. The second law of
thermodynamics names the quantity that tells the two directions apart:
entropy, which a \omterm{def:b1:newton-dynamics:conservation}{closed system} can only create, never destroy. This
chapter states the law, computes entropy for the systems of this
volume, shows how it measures irreversibility, and gives it, at the
end, its meaning in terms of counting.

\section{The second law}

\begin{theorem}[Second law of thermodynamics]\label{thm:b1:second-law-entropy:secondlaw}
\index{second law of thermodynamics}\index{entropy}\index{entropy!created}\index{entropy!exchanged}
Every \omterm{def:b1:newton-dynamics:conservation}{closed system} possesses an extensive state function, the
\emph{entropy} $S$ (\unit{J/K}), such that in any transformation
\[
\Delta S = S_{\mathrm{exch}} + S_{\mathrm{created}}, \qquad
S_{\mathrm{exch}} = \int\frac{\delta Q}{T_{\mathrm{ext}}}, \qquad
S_{\mathrm{created}} \geq 0 ,
\]
where $\delta Q$ is the heat received from the surroundings at the
temperature $T_{\mathrm{ext}}$ of the place where it enters, and
$S_{\mathrm{created}} = 0$ if and only if the transformation is
reversible. An isolated system ($\delta Q = 0$) can only see its entropy
grow, and its equilibrium is the state of maximum entropy.
\end{theorem}
\admitted

\begin{remark}[Reading the law]\label{rem:b1:second-law-entropy:reading}
Unlike energy, entropy is not conserved: it is \emph{exchanged} with heat
(and only with heat --- work carries no entropy) and \emph{created} by
every irreversible process: friction, heat flowing across a temperature
difference, a gas rushing into a vacuum, mixing. The created term is
the measure of irreversibility, and a transformation is possible only if
it is non-negative. ``Reversible'' is an idealization --- quasi-static,
frictionless, heat exchanged across vanishing temperature gaps --- that
real processes approach but never reach.
\end{remark}

\begin{proposition}[Fundamental identity; entropy of the perfect gas and of condensed phases]\label{prop:b1:second-law-entropy:identity}
\index{fundamental identity of thermodynamics}\index{entropy!of a perfect gas}
For a \omterm{def:b1:newton-dynamics:conservation}{closed system} of fixed composition, along any reversible path,
$\delta Q_{\mathrm{rev}} = T\,\dd S$ and $\delta W_{\mathrm{rev}} = -P\,\dd V$, so
\[
\dd U = T\,\dd S - P\,\dd V ,
\]
a relation between state functions that holds for any infinitesimal
change. Consequently, for $n$ \omterm{def:b1:kinetic-theory:scales}{moles} of \omterm{def:b1:kinetic-theory:model}{perfect gas},
\[
S(T, V) = nC_{V,m}\ln T + nR\ln V + \text{const}, \qquad
S(T, P) = nC_{P,m}\ln T - nR\ln P + \text{const},
\]
and for a \omterm{prop:b1:kinetic-theory:condensed}{condensed phase} of heat capacity $C$, $S(T) = C\ln T + \text{const}$.
\end{proposition}

\begin{proof}
Reversible: $T_{\mathrm{ext}} = T$ and $S_{\mathrm{created}} = 0$ give $\dd S =
\delta Q/T$; the first law then gives the identity, which no longer
refers to a path. \omterm{def:b1:kinetic-theory:model}{Perfect gas}: $\dd S = (\dd U + P\,\dd V)/T = nC_{V,m}\dd T/T
+ nR\,\dd V/V$; integrate; use $PV = nRT$ for the other form. \omterm{prop:b1:kinetic-theory:condensed}{Condensed
phase}: $\dd V = 0$, $\dd S = C\,\dd T/T$.
\end{proof}

\begin{example}[Reading the formulas]\label{ex:b1:second-law-entropy:formulas}
An adiabatic \omterm{def:b1:first-law:transformation}{reversible transformation} is \emph{isentropic}: $S(T, V)$
constant gives $T^{C_{V,m}}V^R$ constant, i.e.\ $TV^{\gamma - 1}$ constant ---
\omterm{prop:b1:first-law:transformations}{Laplace's law} again. Doubling the volume of a \omterm{def:b1:kinetic-theory:scales}{mole} of gas at constant
temperature raises its entropy by $R\ln2 = \qty{5.8}{J/K}$, whether done
reversibly (the entropy comes in as heat from the bath) or by a \omterm{rem:b1:first-law:irreversible}{Joule
expansion} into vacuum (no heat: the same \qty{5.8}{J/K} are then
\emph{created}). The state function does not care how it got there;
the balance does.
\end{example}

\section{Entropy balances}

\begin{proposition}[Heat exchanged with a thermostat]\label{prop:b1:second-law-entropy:thermostat}
A system receiving the heat $Q$ from a thermostat at $T_0$ (whatever
happens inside) has $S_{\mathrm{exch}} = Q/T_0$, and the thermostat's own
entropy changes by $-Q/T_0$: a thermostat exchanges entropy reversibly.
\end{proposition}

\begin{proof}
$T_{\mathrm{ext}} = T_0$ is constant; the thermostat's internal state
changes infinitesimally, so its own creation is negligible.
\end{proof}

\begin{example}[A hot body in a cold lake]\label{ex:b1:second-law-entropy:quench}
An iron block ($m = \qty{1.0}{kg}$, $c = \qty{450}{J/(kg.K)}$) at $T_1 =
\qty{373}{K}$ dropped into a lake at $T_0 = \qty{293}{K}$: it ends at $T_0$,
with $\Delta S_{\text{block}} = mc\ln(T_0/T_1) = \qty{-109}{J/K}$ and $Q =
mc(T_0 - T_1) = \qty{-36}{kJ}$ given to the lake, so $S_{\mathrm{exch}} =
Q/T_0 = \qty{-123}{J/K}$ and $S_{\mathrm{created}} = -109 + 123 = \qty{+14}{J/K}$.
The block's entropy fell, the lake's rose by more: the process is
irreversible, and the number measures by how much.
\end{example}

\begin{proposition}[Two consequences]\label{prop:b1:second-law-entropy:clausiuskelvin}
\index{Clausius statement}\index{Kelvin statement}
\begin{enumerate}
  \item \emph{(Clausius)} Heat does not flow spontaneously from a cold
        body to a hotter one: a quantity $Q > 0$ passing from $T_h$ to
        $T_c < T_h$ creates $S = Q(1/T_c - 1/T_h) > 0$; the reverse would
        destroy entropy.
  \item \emph{(Kelvin)} No cyclic machine can produce work from a single
        thermostat: over a cycle $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$
        forces $Q \leq 0$, hence $W = -Q \geq 0$ --- the machine can only
        \emph{receive} work and reject heat.
\end{enumerate}
\end{proposition}

\begin{proof}
Both are the balance applied to the system ``the two thermostats'' or
``the machine over a cycle'', with $S$ a state function so that $\Delta S
= 0$ over a cycle.
\end{proof}

\begin{definition}[Entropy diagram]\label{def:b1:second-law-entropy:TS}
\index{entropy diagram}
In the $(S, T)$ plane a \omterm{def:b1:first-law:transformation}{reversible transformation} is a curve, and the
heat received is the area under it: $Q_{\mathrm{rev}} = \int T\,\dd S$. A
reversible isotherm is horizontal, an isentropic (reversible adiabatic)
vertical; a reversible cycle encloses an area equal to the heat received
and hence, by the first law, to the work delivered
(\cref{ch:b1:heat-engines}).
\end{definition}

\begin{method}[Making an entropy balance]\label{met:b1:second-law-entropy:method}
\begin{enumerate}
  \item Compute $\Delta S$ of the system from the state functions
        (\cref{prop:b1:second-law-entropy:identity}): initial and final
        states only.
  \item Compute $S_{\mathrm{exch}}$ from the heat actually received and the
        temperature of the source: $\sum Q_i/T_i$ for thermostats,
        $\int\delta Q/T_{\mathrm{ext}}$ otherwise.
  \item Deduce $S_{\mathrm{created}} = \Delta S - S_{\mathrm{exch}}$; it must be
        $\geq 0$ --- if not, the transformation is impossible (or an error
        was made); $= 0$ for a reversible one.
\end{enumerate}
\end{method}

\begin{example}[Mixing hot and cold]\label{ex:b1:second-law-entropy:mixing}
Two equal masses $m$ of water at $T_1$ and $T_2$ mixed in a thermos end at
$(T_1 + T_2)/2$ (first law); $\Delta S = mc[\ln\frac{T_f}{T_1} + \ln\frac{T_f}{T_2}]
= mc\ln\frac{(T_1 + T_2)^2}{4T_1T_2} \geq 0$ (arithmetic mean above geometric
mean), with no exchange: all created. For \qty{1}{kg} at \qty{300}{K} and
\qty{400}{K}: \qty{86}{J/K}. The weekend problem shows that by mixing them
\emph{reversibly} --- through an ideal engine --- one would instead
recover work and end at $\sqrt{T_1T_2}$.
\end{example}

\section{What entropy counts}

\begin{proposition}[Boltzmann's formula]\label{prop:b1:second-law-entropy:boltzmann}
\index{Boltzmann's formula}\index{microstate}
A macroscopic state (given $U$, $V$, $N$) can be realized by a number
$\Omega$ of microscopic configurations (\emph{microstates}). Its entropy is
\[
S = k_B\ln\Omega .
\]
An isolated system evolves toward the macroscopic state that has, by
far, the most microstates: the growth of entropy is the march toward
the overwhelmingly probable.
\end{proposition}
\admitted

\begin{example}[Why the gas never comes back]\label{ex:b1:second-law-entropy:count}
$N$ molecules free to be in either half of a box: $2^N$ arrangements,
all equally likely; only one has them all on the left. Letting a gas
confined to one half expand multiplies $\Omega$ by $2^N$, so $\Delta S =
Nk_B\ln2 = nR\ln2$ --- exactly the Joule-expansion result above, now with
a meaning: the molecules have $2^N$ times more ways to be. The chance of
their spontaneous return is $2^{-N}$: one in a thousand for ten
molecules, one in $10^{30}$ for a hundred, and for a \omterm{def:b1:kinetic-theory:scales}{mole} a number with
$10^{23}$ zeros. The second law is not a prohibition; it is a probability
so lopsided that it has never once been caught failing.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:1}
Entropy change of \qty{1.0}{kg} of ice melting at \qty{0}{\degreeCelsius}
($L_f = \qty{334}{kJ/kg}$), and of \qty{1.0}{kg} of water boiling at
\qty{100}{\degreeCelsius} ($L_v = \qty{2260}{kJ/kg}$). Why is the second so
much larger?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:2}
Entropy change of \qty{1.0}{kg} of water heated from \qty{20}{} to
\qty{80}{\degreeCelsius}. Does it depend on how the heating is done?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:3}
One \omterm{def:b1:kinetic-theory:scales}{mole} of \omterm{def:b1:kinetic-theory:model}{perfect gas} doubles its volume at constant temperature.
$\Delta S$; entropy exchanged and created if the expansion is reversible;
if it is a \omterm{rem:b1:first-law:irreversible}{Joule expansion} into vacuum.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:4}
Show from $S(T, V)$ that a reversible \omterm{prop:b1:first-law:transformations}{adiabatic transformation} of a
\omterm{def:b1:kinetic-theory:model}{perfect gas} obeys $TV^{\gamma - 1} = $ const. What is the entropy change in
a \emph{sudden} adiabatic compression (\cref{exo:b1:first-law:11})?
Compute it for that exercise's numbers.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:5}
A \qty{1.0}{kg} iron block ($c = \qty{450}{J/(kg.K)}$) at \qty{100}{\degreeCelsius}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:6}
Two \qty{1.0}{kg} masses of water at \qty{300}{K} and \qty{400}{K} are mixed

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:7}
A \qty{1.0}{kJ} heat leak crosses a wall from a room at \qty{300}{K} to the

greatest.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:8}
Prove Kelvin's statement from the entropy balance of a machine running

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:9}
One \omterm{def:b1:kinetic-theory:scales}{mole} of nitrogen and one \omterm{def:b1:kinetic-theory:scales}{mole} of oxygen, at the same $T$ and $P$,

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:10}
A \qty{100}{W} bulb burns for one hour; all its energy ends up as heat in

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:11}
A body of heat capacity $C$ at $T_1$ is cooled to $T_0 < T_1$ (a) by
direct contact with a thermostat at $T_0$, (b) in two steps through an

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:12}

\end{exercise}

\section{Problem: The cup of coffee and the arrow of time}

\begin{problem}[{Weekend problem --- a cooling cup, a mixed bath, a
refrigerator and a box of molecules: four entropy balances, and what
the created entropy would have been worth in work}]
\label{pb:b1:second-law-entropy:1}
Water: $c = \qty{4180}{J/(kg.K)}$; room (a thermostat): $T_0 = \qty{293}{K}$;
$R = \qty{8.314}{J/(mol.K)}$, $k_B = \qty{1.38e-23}{J/K}$.

\medskip
\noindent\textbf{Part I --- The cooling cup.}
A cup of \qty{0.25}{kg} of coffee at $T_1 = \qty{363}{K}$ cools to room
temperature.
\begin{enumerate}
  \item Heat given to the room, and the room's entropy change.
  \item Entropy change of the coffee.
  \item Entropy created; check its sign.
  \item The cup is drunk when it reaches \qty{330}{K}: entropy created up
        to that moment.
  \item Show that for any $T_1 > T_0$, $S_{\mathrm{created}} = mc[T_1/T_0 - 1 -
        \ln(T_1/T_0)] \geq 0$ (study $f(x) = x - 1 - \ln x$).
  \item The coffee cools faster if stirred or blown upon: does that change
        the created entropy? Why or why not?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Mixing, badly and well.}
Two \qty{1.0}{kg} masses of water at $T_1 = \qty{400}{K}$ and $T_2 =
\qty{300}{K}$, in a thermos.
\begin{enumerate}[resume]
  \item Direct mixing: final temperature $T_f$ and entropy created.
  \item Instead, an ideal engine takes heat from the hot water, rejects
        heat to the cold one and delivers work, until both are at the same
        temperature $T_f'$; the whole process is reversible. Write the
        entropy balance and show that $T_f' = \sqrt{T_1T_2}$.
  \item Compute $T_f'$ and the work delivered (first law on the two
        waters).
  \item Check the reversible scheme's entropy balance numerically: the
        entropy lost by the hot water equals that gained by the cold.
  \item Compare the work with $T_0 \times S_{\mathrm{created}}$ of question 6
        (take the room as the reference temperature here, $T_0 \approx T_2$):
        what does the created entropy of the careless mixing measure?
  \item Why does no kitchen recover that work?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The refrigerator.}
A refrigerator keeps its interior at $T_c = \qty{278}{K}$ in the room at
$T_0$; heat leaks into it at \qty{40}{W}, which the machine pumps back out.
\begin{enumerate}[resume]
  \item Over one second, write the entropy balance of the machine (a
        cyclic device): entropy received from the cold interior, given to
        the room, created inside.
  \item Deduce the minimum work per second (the \omterm{thm:b1:dc-circuits:kirchhoff}{electric power}) needed,
        and the corresponding maximal coefficient of performance
        $Q_c/W$.
  \item Express that bound in terms of $T_c$ and $T_0$ alone, and evaluate
        it for a freezer at $T_c = \qty{255}{K}$.
  \item A real refrigerator of this size draws about \qty{15}{W}: entropy
        created per second.
  \item Left open, the refrigerator's door lets heat in at \qty{200}{W}:
        minimum power now, and why an open refrigerator warms a room.
  \item What would the second law say about a refrigerator that needed
        no electricity at all?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Counting.}
A box is divided in two equal halves by a wall; $N$ molecules of a
\omterm{def:b1:kinetic-theory:model}{perfect gas} are on the left. The wall is removed.
\begin{enumerate}[resume]
  \item Number of microstates before and after (each molecule may be in
        either half), and $\Delta S$ by Boltzmann's formula.
  \item Check against $\Delta S = nR\ln2$ from the thermodynamic formula
        for a \omterm{rem:b1:first-law:irreversible}{Joule expansion}.
  \item For $N = 50$, probability of finding all molecules back on the
        left at a random instant; for one \omterm{def:b1:kinetic-theory:scales}{mole}, the \omterm{def:b1:units-dimensions:oom}{order of magnitude} of
        that probability as a power of ten.
  \item How many yes/no questions (bits) specify which half each of the
        $N$ molecules is in? Compare with $\Delta S/(k_B\ln2)$.
  \item Entropy exchanged and created in the \omterm{rem:b1:first-law:irreversible}{Joule expansion}; why is it
        the textbook example of irreversibility?
  \item The same gas is compressed back reversibly and isothermally to the
        left half: work received, heat given to the room, entropy
        exchanged; what must be supplied to undo the free expansion?
  \item Summarize in three lines: what the second law conserves
