\chapter{Newton's Laws of Dynamics}\label{ch:b1:newton-dynamics}

A skydiver steps out of the plane and, within a dozen seconds, stops
accelerating: at two hundred kilometers per hour the air pushes back
exactly as hard as the Earth pulls. A car takes a bend too fast and
the tires give way; a truck on a mountain road crawls in low gear,
held by the same friction that lets it stop on the flat. Kinematics
described motions; dynamics explains them, from a single law linking
the acceleration of a body to the \omterm{def:b1:newton-dynamics:momentum}{forces} acting on it. This chapter
states Newton's three laws, catalogues the \omterm{def:b1:newton-dynamics:momentum}{forces} one meets at the
level of this volume --- weight, springs, strings, contacts, friction
dry and fluid --- and shows how to go from a drawing of the \omterm{def:b1:newton-dynamics:momentum}{forces} to
a differential equation and its solution.

\section{Mass, momentum, and the three laws}

\begin{definition}[Mass, momentum, force]\label{def:b1:newton-dynamics:momentum}
\index{mass}\index{momentum}\index{force}
A \omterm{def:b1:point-kinematics:frame}{point particle} has an \emph{inertial mass} $m > 0$ (kilograms), a
measure of its resistance to changes of velocity, independent of the
frame. Its \emph{momentum} in a frame is
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
A \emph{force} $\vect F$ (newtons) is the action of another body on the
particle; forces add as vectors, and the \emph{resultant} is their sum
$\sum\vect F$.
\end{definition}

\begin{theorem}[Newton's laws]\label{thm:b1:newton-dynamics:laws}
\index{Newton's laws}\index{inertial frame}\index{principle of inertia}\index{action and reaction}
\begin{enumerate}
  \item \emph{Principle of inertia.} There exist frames, called
        \emph{inertial} (or Galilean), in which a particle subject to
        no \omterm{def:b1:newton-dynamics:momentum}{force}, or to \omterm{def:b1:newton-dynamics:momentum}{forces} of zero resultant, moves in a straight
        line at constant velocity. Any frame in uniform rectilinear
        translation with respect to an \omterm{thm:b1:newton-dynamics:laws}{inertial frame} is inertial.
  \item \emph{Law of \omterm{def:b1:newton-dynamics:momentum}{momentum}.} In an \omterm{thm:b1:newton-dynamics:laws}{inertial frame},
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{i.e.}\qquad
        m\,\vect a = \sum\vect F \ \text{ for constant } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{Action and reaction}.} If body $A$ exerts $\vect F_{A\to B}$
        on body $B$, then $B$ exerts $\vect F_{B\to A} = -\vect F_{A\to B}$
        on $A$, along the line joining them.
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[What the laws say]\label{rem:b1:newton-dynamics:meaning}
The first law is not a special case of the second: it \emph{defines}
the frames in which the second holds. In practice a laboratory on the
Earth is inertial to excellent accuracy for experiments lasting minutes
(\cref{ch:b1:non-inertial-frames} quantifies the error); a frame
attached to the Sun and the distant stars is better still. The second
law is a vector equation: three scalar equations, one per axis, and
the choice of axes (\cref{ch:b1:point-kinematics}) is the first
decision of every problem. The third law holds at each instant for
the contact and gravitational \omterm{def:b1:newton-dynamics:momentum}{forces} of this volume.
\end{remark}

\begin{definition}[Closed system; momentum conservation]\label{def:b1:newton-dynamics:conservation}
\index{closed system}\index{conservation of momentum}
A \emph{closed system} exchanges no matter with the outside. For a
system of particles, the internal \omterm{def:b1:newton-dynamics:momentum}{forces} cancel in pairs (third law),
and the total \omterm{def:b1:newton-dynamics:momentum}{momentum} $\vect P = \sum m_i\vect v_i$ obeys
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$: it is conserved when
the external \omterm{def:b1:newton-dynamics:momentum}{forces} vanish (\cref{ch:b1:systems-of-points} develops
this).
\end{definition}

\section{The usual forces}

\begin{proposition}[Weight and gravitation]\label{prop:b1:newton-dynamics:weight}
\index{weight}\index{gravitational field}
Near the Earth's surface a body of \omterm{def:b1:newton-dynamics:momentum}{mass} $m$ is pulled by its
\emph{weight} $\vect P = m\vect g$, with $\vect g$ the local \omterm{prop:b1:newton-dynamics:weight}{gravitational
field}, $g \approx \qty{9.81}{m/s^2}$, directed downward (toward the
center, very nearly). More generally a \omterm{def:b1:newton-dynamics:momentum}{mass} $M$ at distance $r$ attracts
$m$ with $\vect F = -GMm/r^2\,\vect e_r$ (\cref{ch:b1:central-forces});
at the surface of a sphere of radius $R$, $g = GM/R^2$. Note the unit:
\unit{N/kg} and \unit{m/s^2} are the same, which is why all bodies fall
alike.
\end{proposition}

\begin{proof}
Newton's law of gravitation is admitted and explored in
\cref{ch:b1:central-forces}; $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$.
\end{proof}

\begin{proposition}[Spring; string]\label{prop:b1:newton-dynamics:springstring}
\index{Hooke's law}\index{spring}\index{tension}
An ideal spring of stiffness $k$ and natural length $\ell_0$, stretched
to length $\ell$ along the unit vector $\vect u$ pointing from its fixed
end to the body, exerts on the body the \emph{restoring \omterm{def:b1:newton-dynamics:momentum}{force}}
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{Hooke's law}),
\]
toward its natural length. An ideal string (massless, inextensible)
exerts a \emph{tension} $\vect T$ along itself, pulling, of the same
magnitude at both ends and all along it (even over a massless,
frictionless pulley).
\end{proposition}

\begin{proof}
\omterm{prop:b1:newton-dynamics:springstring}{Hooke's law} is an empirical model valid for small deformations. The
string: a massless element obeys $0 = \vect T_1 + \vect T_2$ from
Newton's second law, so the tension is transmitted unchanged.
\end{proof}

\begin{proposition}[Contact forces: normal reaction and dry friction]\label{prop:b1:newton-dynamics:friction}
\index{normal reaction}\index{dry friction}\index{Coulomb's laws of friction}\index{coefficient of friction}
A solid surface exerts on a body touching it a \emph{\omterm{prop:b1:newton-dynamics:friction}{normal reaction}}
$\vect N$, perpendicular to the surface and repulsive ($N \geq 0$: a
surface cannot pull), and a tangential \emph{friction \omterm{def:b1:newton-dynamics:momentum}{force}} $\vect T$.
Coulomb's laws: if the body does not slide, $\abs{\vect T} \leq \mu_s N$
(\emph{static} friction adjusts to what is needed, up to that limit);
if it slides at velocity $\vect v_g$ relative to the surface,
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (\emph{kinetic} friction,
opposing the sliding), with $\mu_d \leq \mu_s$ coefficients of order
$0.1$ to $1$ depending on the materials, independent of the contact
area and of the speed.
\end{proposition}
\admitted

\begin{proposition}[Fluid friction]\label{prop:b1:newton-dynamics:drag}
\index{fluid friction}\index{drag}\index{Stokes's law}
A body moving at velocity $\vect v$ through a fluid at rest feels a
\emph{drag} opposite to $\vect v$: at low speed (small size, viscous
fluid) it is linear, $\vect F = -\alpha\vect v$ --- for a sphere of
radius $r$ in a fluid of viscosity $\eta$, $\alpha = 6\pi\eta r$
(Stokes); at high speed (large, fast bodies in air or water) it is
quadratic,
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
with $\rho$ the fluid's density, $S$ the body's cross-section facing
the flow and $C_x$ a \omterm{def:b1:units-dimensions:dimension}{dimensionless} \emph{drag coefficient} ($0.5$ for a
sphere, $0.3$ for a car, $1$ for a flat disk). The Reynolds number
$\rho vL/\eta$ (\cref{rem:b1:units-dimensions:pi}) decides: linear
below $1$, quadratic above about $1000$.
\end{proposition}
\admitted

\begin{remark}[Other forces]\label{rem:b1:newton-dynamics:others}
Buoyancy (Archimedes) is a resultant of pressure \omterm{def:b1:newton-dynamics:momentum}{forces}, derived in
\cref{ch:b1:fluid-statics}; the electric and magnetic \omterm{def:b1:newton-dynamics:momentum}{forces} on a
charge are the subject of \cref{ch:b1:charged-particles}. All the
\omterm{def:b1:newton-dynamics:momentum}{forces} of this chapter are \emph{models} of contact or field
interactions, valid within stated limits --- Hooke's spring breaks,
Coulomb's friction fails at high pressure, Stokes's drag fails at high
speed. Part of every solution is checking that the model used applies.
\end{remark}

\section{Solving a problem of dynamics}

\begin{method}[From the situation to the equation]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{System}: name the body treated as a point.
  \item \emph{Frame}: name it and check it is inertial (or treat it as
        such to the accuracy required).
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{Forces}}: list every body touching or attracting the
        system, and draw each \omterm{def:b1:newton-dynamics:momentum}{force} on a diagram (the free-body
        diagram); nothing else enters.
  \item \emph{Coordinates}: choose axes adapted to the motion
        (\cref{met:b1:point-kinematics:choose}) and express $\vect a$.
  \item \emph{Project} $m\vect a = \sum\vect F$ on the axes; use the
        constraints (a body on a surface stays on it: $a_\perp = 0$; a
        string of fixed length: equal speeds at both ends).
  \item \emph{Solve} and \emph{check}: dimensions, limiting cases,
        signs ($N \geq 0$, $\abs T \leq \mu_sN$ if at rest).
\end{enumerate}
\end{method}

\begin{example}[Block on an incline]\label{ex:b1:newton-dynamics:incline}
Block of \omterm{def:b1:newton-dynamics:momentum}{mass} $m$ on a plane inclined at $\alpha$, coefficients $\mu_s$,
$\mu_d$. Axes: $x$ down the slope, $y$ along the normal. Weight
$(mg\sin\alpha, -mg\cos\alpha)$, reaction $(0, N)$, friction $(-T, 0)$
if sliding down. Along $y$: $N = mg\cos\alpha$. At rest along $x$:
$T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$, i.e.\ $\tan\alpha
\leq \mu_s$: the block holds up to the angle $\arctan\mu_s$ --- a
measurement of $\mu_s$ needing no balance. Sliding: $ma = mg\sin\alpha -
\mu_dmg\cos\alpha$, so $a = g(\sin\alpha - \mu_d\cos\alpha)$, independent
of $m$: \qty{3.2}{m/s^2} for $\alpha = 30^\circ$, $\mu_d = 0.2$.
\end{example}

\begin{example}[The simple pendulum]\label{ex:b1:newton-dynamics:pendulum}
\index{simple pendulum}
\omterm{def:b1:newton-dynamics:momentum}{Mass} $m$ on a string of length $\ell$, angle $\theta$ from the downward
vertical. \omterm{def:b1:point-kinematics:cylindrical}{Polar coordinates} at the suspension point: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$; \omterm{def:b1:newton-dynamics:momentum}{forces}:
weight $mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$ and
tension $-T\vect e_r$. Along $\vect e_\theta$: $m\ell\ddot\theta =
-mg\sin\theta$,
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
which for small angles ($\sin\theta \approx \theta$) is the harmonic
equation with $\omega_0 = \sqrt{g/\ell}$: period $T_0 = 2\pi\sqrt{\ell/g}$,
the $k = 2\pi$ that \cref{ex:b1:units-dimensions:pendulum} could not
supply. Along $\vect e_r$: $T = mg\cos\theta + m\ell\dot\theta^2$ --- the
string pulls hardest at the bottom, where the speed is greatest.
\end{example}

\begin{proposition}[Fall with drag; terminal velocity]\label{prop:b1:newton-dynamics:terminal}
\index{terminal velocity}
A body of \omterm{def:b1:newton-dynamics:momentum}{mass} $m$ falling from rest under its weight and a drag \omterm{def:b1:newton-dynamics:momentum}{force}
reaches a \emph{\omterm{prop:b1:newton-dynamics:terminal}{terminal velocity}} at which the two balance. Linear
drag: $m\dot v = mg - \alpha v$, $v = v_\infty(1 - \eu^{-t/\tau})$ with
$v_\infty = mg/\alpha$, $\tau = m/\alpha$. Quadratic drag: $m\dot v = mg -
\tfrac12\rho C_xSv^2$, $v = v_\infty\tanh(t/\tau)$ with
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
Linear: a first-order equation, $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder}). Quadratic: with
$u = v/v_\infty$, $\dot u = (1 - u^2)/\tau$; separating variables,
$\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$ (the rational
fraction $1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$ integrates to
$\tfrac12\ln\frac{1 + u}{1 - u}$), so $u = \tanh(t/\tau)$.
\end{proof}

\begin{example}[Two terminal velocities]\label{ex:b1:newton-dynamics:terminal}
A skydiver, $m = \qty{80}{kg}$, belly down, $C_xS \approx \qty{0.8}{m^2}$
in air ($\rho = \qty{1.2}{kg/m^3}$): $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$, reached (to $90\%$) in about $1.5\tau =
\qty{6}{s}$. A fog droplet, $r = \qty{10}{\micro m}$, in Stokes's regime
($\eta_{\text{air}} = \qty{1.8e-5}{Pa.s}$): $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- which is why fog hangs in the air.
\end{example}

\section{Motion on a circle: constraint forces}

\begin{proposition}[Centripetal requirement]\label{prop:b1:newton-dynamics:centripetal}
\index{centripetal force}
A particle moving on a circle of radius $R$ at speed $v$ must receive
from the resultant of the \omterm{def:b1:newton-dynamics:momentum}{forces} a component $mv^2/R$ directed toward
the center. Whatever provides it --- a string's tension, friction, a
banked road's reaction, gravity --- sets the maximum speed or the
minimum speed at which the motion is possible.
\end{proposition}

\begin{proof}
Newton's second law with the Frenet acceleration: the normal component
of $\sum\vect F$ equals $mv^2/\rho$.
\end{proof}

\begin{example}[Three circles]\label{ex:b1:newton-dynamics:circles}
\emph{Flat bend}: friction must supply $mv^2/R \leq \mu_sN = \mu_smg$,
so $v \leq \sqrt{\mu_sgR}$: \qty{20}{m/s} for $R = \qty{50}{m}$,
$\mu_s = 0.8$. \emph{Banked bend} at angle $\beta$, no friction: the
reaction $\vect N$ is normal to the road; vertically $N\cos\beta = mg$,
horizontally $N\sin\beta = mv^2/R$, so $v^2 = gR\tan\beta$ --- the one
speed at which the bend is taken without any friction. \emph{Loop}: at
the top, with $\vect N$ pointing down toward the center, $N + mg =
mv^2/R$, and $N \geq 0$ requires $v \geq \sqrt{gR}$: below that, the car
leaves the track (\cref{pb:b1:point-kinematics:1} again, now explained).
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:1}
A person of \omterm{def:b1:newton-dynamics:momentum}{mass} \qty{70}{kg} stands on a scale in an elevator. What
does the scale read (in newtons and in ``kilograms'') when the elevator
accelerates upward at \qty{2.0}{m/s^2}, downward at \qty{2.0}{m/s^2},
moves at constant velocity, and falls freely?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:2}
A block $m_1 = \qty{2.0}{kg}$ on a frictionless table is tied by a
string over a frictionless pulley to a hanging \omterm{def:b1:newton-dynamics:momentum}{mass} $m_2 = \qty{1.0}{kg}$.
Find the acceleration and the tension.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:3}
A block slides down a $30^\circ$ incline with $\mu_d = 0.20$. Compute
its acceleration and its speed after \qty{5.0}{m} from rest. Does the
answer depend on its \omterm{def:b1:newton-dynamics:momentum}{mass}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:4}
A \qty{0.50}{kg} \omterm{def:b1:newton-dynamics:momentum}{mass} hangs from a spring of stiffness \qty{200}{N/m}.
Find the static elongation; then the \omterm{def:b1:wave-propagation:signal}{angular frequency} and period of
its vertical oscillations about equilibrium (show the equation of
motion is harmonic).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:5}
A crate sits on a tilting platform; it starts to slide at $22^\circ$.
Find $\mu_s$. A car on a flat road with $\mu_s = 0.80$: maximum speed

\end{exercise}

\section{Problem: The parachutist}

\begin{problem}[{Weekend problem --- from the door of the plane to the
ground: free fall, the air that pushes back, a canopy that opens too
fast, and the landing --- one law of motion, two drag laws}]
\label{pb:b1:newton-dynamics:1}
A parachutist with gear has \omterm{def:b1:newton-dynamics:momentum}{mass} $m = \qty{80}{kg}$. Air:
$\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$. Falling belly
down, $C_xS = \qty{0.80}{m^2}$; under the open canopy, $C_xS =
\qty{25}{m^2}$. $g = \qty{9.81}{m/s^2}$; $z$ is counted downward from
the jump point.

\medskip
\noindent\textbf{Part I --- Without air.}
\begin{enumerate}
  \item Write Newton's second law and give $v(t)$ and $z(t)$; compute
        both at $t = \qty{10}{s}$.
  \item Compute the weight.
  \item The parachutist feels ``weightless'' during this phase although
        gravity acts fully. Explain with the \omterm{def:b1:newton-dynamics:momentum}{forces} (what is missing
        compared with standing on the ground).
  \item Is the Earth's frame inertial enough for this problem? Name what
        is neglected.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Falling through air.}
\begin{enumerate}[resume]
  \item Write the drag \omterm{def:b1:newton-dynamics:momentum}{force} for belly-down fall as a function of $v$.
  \item Write the equation of motion for $v(t)$.
  \item Find the \omterm{prop:b1:newton-dynamics:terminal}{terminal velocity} $v_\infty$ (in \unit{m/s} and
        \unit{km/h}).
  \item With $u = v/v_\infty$ and $\tau = v_\infty/g$, show that
        $\dd u/\dd t = (1 - u^2)/\tau$; compute $\tau$.
  \item Solve by separation of variables and show that
        $v = v_\infty\tanh(t/\tau)$.
  \item After how long is $v = 0.9v_\infty$? $0.99v_\infty$?
  \item Show that $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ and compute the
        distance fallen at $t = \qty{10}{s}$; compare with Part I.
  \item Estimate the Reynolds number at $v_\infty$ with $L = \qty{0.5}{m}$
        and justify the quadratic law.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The canopy opens.}
\begin{enumerate}[resume]
  \item Compute the new \omterm{prop:b1:newton-dynamics:terminal}{terminal velocity} under the open canopy.
  \item If the canopy opened instantaneously at $v = \qty{40}{m/s}$, what
        would the deceleration be, in \unit{m/s^2} and in $g$? Why is
        this unacceptable?
  \item Real canopies open over about \qty{3}{s}. Estimate the mean
        deceleration and the mean \omterm{def:b1:newton-dynamics:momentum}{force} of the harness on the
        parachutist.
  \item Once the canopy is fully open the speed decays toward the new
        $v_\infty$: what is the new \omterm{def:b1:transient-regimes:firstorder}{time constant} $\tau' = v_\infty'/g$,
        and what does it mean for the approach to steady descent?
  \item In steady descent, what is the harness \omterm{def:b1:newton-dynamics:momentum}{force}? Justify with the
        second law.
  \item Explain why the \omterm{prop:b1:newton-dynamics:terminal}{terminal velocity} varies as $1/\sqrt{C_xS}$: by
        what factor would a canopy of half the area raise it?
  \item A heavier jumper (\qty{100}{kg}) under the same canopy: \omterm{prop:b1:newton-dynamics:terminal}{terminal
        velocity}?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Landing.}
\begin{enumerate}[resume]
  \item The landing speed is the canopy's $v_\infty$. From what height
        would one have to jump, with no air, to hit the ground at that
        speed?
  \item A stiff-legged landing stops the body over \qty{0.5}{m}; a roll
        over \qty{1.5}{m}. Estimate the mean deceleration (in $g$) and
        the mean \omterm{def:b1:newton-dynamics:momentum}{force} on the legs in each case, and conclude.
  \item Just before touching down the parachutist pulls both toggles,
        which briefly increases the canopy's drag. Explain qualitatively
        the effect on the speed.
\end{enumerate}

\medskip
\noindent\textbf{Part V --- Small things fall differently.}
\begin{enumerate}[resume]
  \item A raindrop of radius \qty{1.0}{mm} ($C_x = 0.5$): compute its
        \omterm{prop:b1:newton-dynamics:terminal}{terminal velocity} with the quadratic law, and check the Reynolds
        number.
  \item A fog droplet of radius \qty{10}{\micro m}: use \omterm{prop:b1:newton-dynamics:drag}{Stokes's law}
        for $v_\infty$ and check that the Reynolds number is indeed
        small.
  \item Summarize: one equation of motion, two drag regimes, and the
        three numbers (\unit{m/s}) that decide whether something falls
        like a stone, like a leaf or not at all.
\end{enumerate}
\end{problem}
