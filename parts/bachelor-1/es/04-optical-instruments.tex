\chapter{Instrumentos ópticos y el ojo}\label{ch:b1:optical-instruments}

Desde la última fila de una sala de conciertos, unos prismáticos acercan
el rostro de la cantante hasta el alcance del brazo; un \omterm{def:b1:optical-instruments:microscope}{microscopio}
convierte una gota de agua de estanque en un zoo; un teléfono del grosor
de un dedo fotografía la Vía Láctea. Cada uno de esos instrumentos son
unas pocas \omterm{def:b1:mirrors-thin-lenses:lens}{lentes delgadas} dispuestas con un único fin: presentar al \omterm{def:b1:optical-instruments:eye}{ojo},
o a un sensor, una imagen mayor, más luminosa o más nítida que la que el
\omterm{def:b1:optical-instruments:eye}{ojo} podría formar por sí solo. Por eso el capítulo empieza por el \omterm{def:b1:optical-instruments:eye}{ojo}
mismo --- lo que puede y lo que no puede hacer --- y después construye la
\omterm{def:b1:optical-instruments:G}{lupa}, el \omterm{def:b1:optical-instruments:microscope}{microscopio}, el anteojo y la \omterm{def:b1:optical-instruments:camera}{cámara} a partir de la relación de
conjugación del capítulo anterior.

\section{El ojo}

\begin{definition}[Ojo reducido]\label{def:b1:optical-instruments:eye}
\index{ojo}\index{ojo reducido}\index{retina}\index{acomodación}
El \emph{ojo reducido} modela la córnea y el cristalino como una única
\omterm{def:b1:mirrors-thin-lenses:lens}{lente delgada} convergente de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} \emph{variable}, y la
\emph{retina} como una pantalla situada a una distancia fija
$d \approx \qty{17}{mm}$ por detrás. Unos músculos cambian la curvatura
del cristalino para mantener sobre la retina la imagen del objeto
observado: es la \emph{acomodación}. El iris, delante del cristalino, es
un diafragma variable de \qty{2}{} a \qty{8}{mm} de diámetro, la
\emph{pupila}.
\end{definition}

\begin{definition}[Punto remoto, punto próximo]\label{def:b1:optical-instruments:punctum}
\index{punto remoto}\index{punto próximo}\index{amplitud de acomodación}
El \emph{punto remoto} (punctum remotum) es el punto objeto que se ve
nítido sin acomodar; el \emph{punto próximo} (punctum proximum), el punto
más cercano que se ve nítido con la \omterm{def:b1:optical-instruments:eye}{acomodación} máxima. Para un \omterm{def:b1:optical-instruments:eye}{ojo} normal
(emétrope) el punto remoto está en el infinito y el punto próximo a
$d_m \approx \qty{25}{cm}$, la \emph{distancia convencional de visión
distinta}. La diferencia de \omterm{def:b1:mirrors-thin-lenses:lens}{vergencias} necesaria para pasar de uno a otro
es la \emph{amplitud de \omterm{def:b1:optical-instruments:eye}{acomodación}}: unas \qty{4}{\delta} en un adulto
joven, que disminuyen con la edad.
\end{definition}

\begin{example}[La vergencia del ojo]\label{ex:b1:optical-instruments:vergence}
Objeto lejano: imagen a $d = \qty{17}{mm}$ significa $f' = \qty{17}{mm}$,
$V = 1/0.017 = \qty{59}{\delta}$. Objeto a \qty{25}{cm}:
$V = 1/0.017 + 1/0.25 = 59 + 4 = \qty{63}{\delta}$. Cuatro dioptrías de
\omterm{def:b1:optical-instruments:eye}{acomodación} sobre sesenta: el cristalino añade solo un pequeño porcentaje
al trabajo de la córnea, pero ese pequeño porcentaje lo es todo para leer.
\end{example}

\begin{proposition}[Defectos y correcciones]\label{prop:b1:optical-instruments:defects}
\index{miopía}\index{hipermetropía}\index{presbicia}
Un \omterm{def:b1:optical-instruments:eye}{ojo} \emph{miope} es demasiado convergente: su \omterm{def:b1:optical-instruments:punctum}{punto remoto} está a una
distancia finita $D_r$; una \omterm{def:b1:mirrors-thin-lenses:lens}{lente divergente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal}
$f' = -D_r$ (llevada junto al \omterm{def:b1:optical-instruments:eye}{ojo}) conjuga el infinito con el \omterm{def:b1:optical-instruments:punctum}{punto remoto}
y restituye la visión lejana. Un \omterm{def:b1:optical-instruments:eye}{ojo} \emph{hipermétrope} no es bastante
convergente: su \omterm{def:b1:optical-instruments:punctum}{punto remoto} es virtual (detrás del \omterm{def:b1:optical-instruments:eye}{ojo}) y su \omterm{def:b1:optical-instruments:punctum}{punto
próximo} está demasiado lejos; lo corrige una \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente}. La
\emph{presbicia} es la pérdida de \omterm{def:b1:optical-instruments:eye}{acomodación} con la edad: el \omterm{def:b1:optical-instruments:punctum}{punto
próximo} se aleja y la lectura exige \omterm{def:b1:mirrors-thin-lenses:lens}{lentes convergentes}, sea cual sea el
\omterm{def:b1:optical-instruments:punctum}{punto remoto}.
\end{proposition}

\begin{proof}
Una \omterm{def:b1:mirrors-thin-lenses:lens}{lente} pegada al \omterm{def:b1:optical-instruments:eye}{ojo} y el \omterm{def:b1:optical-instruments:eye}{ojo} forman un sistema cuyo objeto ha de ser
algo que el \omterm{def:b1:optical-instruments:eye}{ojo} pueda ver: la \omterm{def:b1:mirrors-thin-lenses:lens}{lente} debe convertir un objeto en el
infinito en una \omterm{def:b1:mirrors-thin-lenses:image}{imagen virtual} en el \omterm{def:b1:optical-instruments:punctum}{punto remoto}, $\overline{OA'} = -D_r$
para $\overline{OA} = -\infty$, luego $f' = \overline{OA'} = -D_r$. Los
demás casos son el mismo argumento con los puntos pertinentes.
\end{proof}

\begin{definition}[Tamaño angular y resolución]\label{def:b1:optical-instruments:angular}
\index{tamaño angular}\index{resolución angular}
El \emph{tamaño angular} de un objeto de altura $h$ situado a una
distancia $D$ es $\theta \approx h/D$ (ángulos pequeños). El \omterm{def:b1:optical-instruments:eye}{ojo} distingue
dos puntos solo si su separación angular supera su \emph{resolución
angular}, de unos $\epsilon \approx \qty{3e-4}{rad}$ (un minuto de arco):
la separación de los conos y la difracción en la pupila fijan por igual
ese límite.
\end{definition}

\begin{remark}[Para qué sirve un instrumento]\label{rem:b1:optical-instruments:purpose}
Llevado al \omterm{def:b1:optical-instruments:punctum}{punto próximo}, un detalle de \qty{1}{mm} subtiende $1/250 =
\qty{4e-3}{rad}$: trece veces el límite de resolución; un detalle de
\qty{0.05}{mm} subtiende $\qty{2e-4}{rad}$ y es invisible. Los
instrumentos que miran objetos próximos (\omterm{def:b1:optical-instruments:G}{lupa}, \omterm{def:b1:optical-instruments:microscope}{microscopio}) aumentan el
\omterm{def:b1:optical-instruments:angular}{tamaño angular} más allá de lo que permite acercar el objeto; los
instrumentos para objetos lejanos (anteojo) aumentan el \omterm{def:b1:optical-instruments:angular}{tamaño angular} de
lo que no se puede acercar; la \omterm{def:b1:optical-instruments:camera}{cámara} sustituye la \omterm{def:b1:optical-instruments:eye}{retina} por un sensor
capaz de acumular luz durante segundos y de ampliarse después.
\end{remark}

\section{La lupa}

\begin{definition}[Aumento angular]\label{def:b1:optical-instruments:G}
\index{aumento angular}\index{lupa}\index{aumento comercial}
Para un instrumento utilizado por el \omterm{def:b1:optical-instruments:eye}{ojo}, el \emph{aumento angular} es
$G = \theta'/\theta$, donde $\theta'$ es el \omterm{def:b1:optical-instruments:angular}{tamaño angular} de la imagen
vista a través del instrumento y $\theta$ el \omterm{def:b1:optical-instruments:angular}{tamaño angular} del objeto
visto a simple vista --- en el \omterm{def:b1:optical-instruments:punctum}{punto próximo} $d_m = \qty{25}{cm}$ para un
objeto cercano, y en su sitio para uno lejano.
\end{definition}

\begin{proposition}[La lupa]\label{prop:b1:optical-instruments:loupe}
Una \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} $f' < d_m$ con el objeto en su
plano focal objeto da una imagen en el infinito, vista sin acomodar,
con
\[
G = \frac{d_m}{f'} = d_m V .
\]
\end{proposition}

\begin{proof}
Objeto $AB$ de altura $h$ en el plano focal: los rayos que salen de $B$
emergen paralelos, con el ángulo del rayo que pasa por $O$,
$\theta' = h/f'$. A simple vista, en el mejor caso: $\theta = h/d_m$. El
cociente vale $d_m/f'$.
\end{proof}

\begin{example}[La lupa de un relojero]\label{ex:b1:optical-instruments:loupe}
$f' = \qty{25}{mm}$: $G = 250/25 = 10$, que se vende como ``$10\times$''.
Un detalle de \qty{0.05}{mm} subtiende entonces $\qty{2e-3}{rad}$, de
sobra resuelto. Subir más $G$ exige una $f'$ de pocos milímetros, una
\omterm{def:b1:mirrors-thin-lenses:lens}{lente} diminuta pegada al \omterm{def:b1:optical-instruments:eye}{ojo} --- el límite de la \omterm{def:b1:mirrors-thin-lenses:lens}{lente} única y la razón de
que exista el \omterm{def:b1:optical-instruments:microscope}{microscopio}.
\end{example}

\section{El microscopio}

\begin{definition}[Microscopio compuesto]\label{def:b1:optical-instruments:microscope}
\index{microscopio}\index{objetivo}\index{ocular}\index{intervalo óptico}
Un \emph{microscopio} es un \emph{objetivo} $L_1$ de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} corta
$f_1'$ que forma una imagen intermedia real y agrandada de un objeto
próximo en el plano focal objeto de un \emph{ocular} $L_2$ (de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia
focal} $f_2'$), que actúa como \omterm{def:b1:optical-instruments:G}{lupa}. La distancia $\Delta =
\overline{F_1'F_2}$ entre el foco imagen del objetivo y el foco objeto del
ocular es el \emph{intervalo óptico}, normalizado en \qty{160}{mm} en los
instrumentos clásicos.
\end{definition}

\begin{proposition}[Aumento de un microscopio]\label{prop:b1:optical-instruments:microscope}
Con la imagen final en el infinito,
\[
\gamma_1 = -\frac{\Delta}{f_1'}, \qquad G_2 = \frac{d_m}{f_2'},
\qquad G = \abs{\gamma_1}\,G_2 = \frac{\Delta\,d_m}{f_1' f_2'} .
\]
\end{proposition}

\begin{proof}
La imagen intermedia $A_1B_1$ está en el plano focal objeto del \omterm{def:b1:optical-instruments:microscope}{ocular},
es decir en $\overline{F_1'A_1} = \Delta$; la \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relación de Newton} da
$\gamma_1 = -\overline{F_1'A_1}/f_1' = -\Delta/f_1'$. El \omterm{def:b1:optical-instruments:microscope}{ocular} muestra
entonces $A_1B_1$ (de altura $\abs{\gamma_1} h$) en el infinito bajo el
ángulo $\theta' = \abs{\gamma_1} h/f_2'$, frente a $\theta = h/d_m$ a
simple vista.
\end{proof}

\begin{example}[Un microscopio escolar]\label{ex:b1:optical-instruments:micronum}
\omterm{def:b1:optical-instruments:microscope}{Objetivo} $f_1' = \qty{4.0}{mm}$, \omterm{def:b1:optical-instruments:microscope}{ocular} $f_2' = \qty{25}{mm}$,
$\Delta = \qty{160}{mm}$: $\gamma_1 = -40$, $G_2 = 10$, $G = 400$. El
objeto se sitúa a $\overline{F_1A} = -f_1'^2/\Delta = -16/160 =
\qty{-0.10}{mm}$ de $F_1$: a \qty{4.1}{mm} del \omterm{def:b1:optical-instruments:microscope}{objetivo}. Enfocar consiste
en mover el tubo entero fracciones de milímetro.
\end{example}

\begin{remark}[El límite de resolución]\label{rem:b1:optical-instruments:abbe}
\index{poder de resolución}
Aumentar sin límite no serviría de nada: la luz es una onda, y un \omterm{def:b1:optical-instruments:microscope}{objetivo}
que acepta rayos dentro de un semiángulo $u$ (en un medio de índice $n$)
no puede separar dos puntos más próximos que
$\lambda/(2n\sin u)$ --- el tratamiento ondulatorio del volumen del Año 2
lo deduce. Con $n\sin u \approx 1.4$ (inmersión en aceite) y $\lambda =
\qty{0.5}{\micro m}$, el límite ronda \qty{0.2}{\micro m}. Para hacer
visible ese detalle al \omterm{def:b1:optical-instruments:eye}{ojo} ($\epsilon = \qty{3e-4}{rad}$ a
\qty{25}{cm}, es decir \qty{75}{\micro m}) basta un aumento de unos $400$;
más allá de $1000$ la imagen solo es mayor, no más rica: es
\emph{aumento vacío}.
\end{remark}

\section{El anteojo astronómico}

\begin{proposition}[Anteojo refractor]\label{prop:b1:optical-instruments:telescope}
\index{anteojo}\index{pupila de salida}
Un \omterm{def:b1:optical-instruments:microscope}{objetivo} $L_1$ y un \omterm{def:b1:optical-instruments:microscope}{ocular} $L_2$ con $F_1' = F_2$ forman un \omterm{prop:b1:mirrors-thin-lenses:doublet}{sistema
afocal}: una estrella (en el infinito) se ve en el infinito, el \omterm{def:b1:optical-instruments:eye}{ojo}
relajado la observa, y
\[
G = -\frac{f_1'}{f_2'} .
\]
La imagen del borde del \omterm{def:b1:optical-instruments:microscope}{objetivo} a través del \omterm{def:b1:optical-instruments:microscope}{ocular} es la \emph{\omterm{prop:b1:optical-instruments:telescope}{pupila de
salida}}: un disco brillante de diámetro $D/\abs G$ situado
$\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ detrás del \omterm{def:b1:optical-instruments:microscope}{ocular}, donde ha de
colocarse la pupila del \omterm{def:b1:optical-instruments:eye}{ojo} para recibir toda la luz.
\end{proposition}

\begin{proof}
$G$ se dedujo en \cref{prop:b1:mirrors-thin-lenses:doublet}. El
\omterm{def:b1:optical-instruments:microscope}{objetivo}, visto desde $L_2$, es un objeto situado en $\overline{O_2O_1} = -(f_1'
+ f_2')$; Descartes da $1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2')
= f_1'/[f_2'(f_1' + f_2')]$, y el aumento $\overline{O_2P'}/
\overline{O_2O_1} = -f_2'/f_1' = 1/G$ lleva el diámetro $D$ a
$D/\abs G$. Todo rayo que entra por el \omterm{def:b1:optical-instruments:microscope}{objetivo} sale por la \omterm{prop:b1:optical-instruments:telescope}{pupila de
salida}, porque cada punto del \omterm{def:b1:optical-instruments:microscope}{objetivo} tiene allí su imagen.
\end{proof}

\begin{proposition}[Lo que gana un anteojo]\label{prop:b1:optical-instruments:gains}
\index{poder colector de luz}\index{criterio de Rayleigh}
Frente al \omterm{def:b1:optical-instruments:eye}{ojo} desnudo (pupila $d_p$):
\begin{itemize}
  \item \emph{\omterm{def:b1:optical-instruments:angular}{tamaño angular}}: multiplicado por $\abs G$;
  \item \emph{luz recogida} de una fuente puntual: multiplicada por
        $(D/d_p)^2$, siempre que la \omterm{prop:b1:optical-instruments:telescope}{pupila de salida} no sea mayor que la
        pupila del \omterm{def:b1:optical-instruments:eye}{ojo}, es decir $\abs G \geq D/d_p$;
  \item \emph{\omterm{def:b1:optical-instruments:angular}{resolución angular}}: la difracción en el \omterm{def:b1:optical-instruments:microscope}{objetivo} la limita
        a $\theta_{\min} \approx 1.22\,\lambda/D$ (\omterm{prop:b1:optical-instruments:gains}{criterio de Rayleigh}),
        muy por debajo del $\epsilon$ del \omterm{def:b1:optical-instruments:eye}{ojo}; el aumento que hace visible
        ese límite es $G_r = \epsilon/\theta_{\min}$, y aumentos mayores
        solo agrandan la mancha.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[El refractor de 150 milímetros de un aficionado]\label{ex:b1:optical-instruments:telnum}
$D = \qty{150}{mm}$, $f_1' = \qty{1200}{mm}$, \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}:
$G = -48$; \omterm{prop:b1:optical-instruments:telescope}{pupila de salida} \qty{3.1}{mm}; ganancia de luz $(150/7)^2 \approx
460$ frente a una pupila adaptada a la oscuridad de \qty{7}{mm};
resolución $1.22 \times
\qty{550}{nm}/\qty{0.15}{m} = \qty{4.5e-6}{rad} = 0.9''$, es decir
cráteres de \qty{1.7}{km} en la Luna; $G_r = 3\times10^{-4}/4.5\times
10^{-6} \approx 67$ --- el \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{6}{mm} ($G = 200$) no enseña
más detalle, solo un disco mayor y más apagado. El problema de fin de
semana diseña este instrumento.
\end{example}

\begin{remark}[Reflectores, prismáticos, catalejos]\label{rem:b1:optical-instruments:variants}
Un \omterm{def:b1:mirrors-thin-lenses:mirror}{espejo cóncavo} de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} $f_1'$ sustituye a la \omterm{def:b1:mirrors-thin-lenses:lens}{lente} \omterm{def:b1:optical-instruments:microscope}{objetivo}
en el telescopio \emph{reflector} (sin dispersión cromática, y un espejo
puede tener metros de anchura y sostenerse por detrás): todo lo anterior
vale con $f_1'$ la \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} del espejo. La imagen está invertida
--- inofensivo para las estrellas; para uso terrestre, los prismáticos
intercalan un par de prismas de \omterm{thm:b1:rays-reflection-refraction:tir}{reflexión total} que la enderezan, y el
catalejo de Galileo usa un \omterm{def:b1:optical-instruments:microscope}{ocular} divergente ($f_2' < 0$, $G > 0$), más
corto pero de campo estrecho.
\end{remark}

\section{La cámara fotográfica}

\begin{definition}[Cámara; número de diafragma]\label{def:b1:optical-instruments:camera}
\index{cámara fotográfica}\index{número de diafragma}\index{profundidad de campo}
Una \emph{cámara} es una \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} $f'$ que
forma una \omterm{def:b1:mirrors-thin-lenses:image}{imagen real} sobre un sensor, en $\overline{OA'} \approx f'$ para
motivos lejanos; enfocar consiste en mover la \omterm{def:b1:mirrors-thin-lenses:lens}{lente}. Un diafragma de
diámetro $D$ limita el haz; el \emph{número de diafragma} es $N = f'/D$
(se escribe $f/N$). El \emph{ángulo de campo} vale
$2\arctan(\ell/2f')$ para un sensor de anchura $\ell$.
\end{definition}

\begin{proposition}[Exposición y profundidad de campo]\label{prop:b1:optical-instruments:dof}
\index{distancia hiperfocal}
La irradiancia sobre el sensor va como $1/N^2$, de modo que el tiempo de
exposición para un brillo de imagen dado va como $N^2$. Un punto situado a
una distancia distinta de la enfocada forma un disco de desenfoque; si se
le exige quedar por debajo de un diámetro tolerado $c$, una \omterm{def:b1:mirrors-thin-lenses:lens}{lente} enfocada
a la \emph{\omterm{prop:b1:optical-instruments:dof}{distancia hiperfocal}}
\[
H = \frac{f'^2}{N c}
\]
da por nítido todo lo que hay entre $H/2$ y el infinito.
\end{proposition}

\begin{proof}
Irradiancia $=$ potencia/área $\propto D^2/f'^2 = 1/N^2$ (la imagen de un
motivo dado tiene un tamaño fijado por $f'$, y la potencia recogida crece
como $D^2$). Enfóquese a la distancia $p$, con la imagen en $q \approx f'$;
un punto en el infinito se enfoca en $f'$, es decir $q - f'$ antes del
sensor; su cono de rayos, de base $D$ en la \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, tiene diámetro
$D(q - f')/q$ sobre el sensor (triángulos semejantes). Con
$1/q = 1/f' - 1/p$ en el marco paraxial, $q - f' \approx f'^2/p$ para
$p \gg f'$, así que el desenfoque vale $Df'/p = f'^2/(Np)$; iguala a $c$
para $p = H$. La misma geometría por el lado cercano da el límite $H/2$.
\end{proof}

\section{Problema: diseñar un anteojo para la Luna}

\begin{problem}[{Problema de fin de semana --- un refractor de 150
milímetros sobre el césped del jardín: elegir los oculares, colocar el
ojo, contar la luz y averiguar qué cráter tan pequeño puede mostrar}]
\label{pb:b1:optical-instruments:1}
El \omterm{def:b1:optical-instruments:microscope}{objetivo} es una \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente} $L_1$ de \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal}
$f_1' = \qty{1200}{mm}$ y diámetro $D = \qty{150}{mm}$; se dispone de
\omterm{def:b1:optical-instruments:microscope}{oculares} $L_2$ de \omterm{def:b1:mirrors-thin-lenses:lens}{distancias focales} \qty{25}{mm} y \qty{6}{mm}. La Luna
está a \qty{3.84e5}{km} y subtiende $0.52^\circ$; la pupila nocturna del
\omterm{def:b1:optical-instruments:eye}{ojo} mide \qty{7}{mm} y su resolución es $\epsilon = \qty{3.0e-4}{rad}$;
$\lambda = \qty{550}{nm}$.

\medskip
\noindent\textbf{Parte I --- El montaje afocal.}
\begin{enumerate}
  \item ¿Dónde debe colocarse el \omterm{def:b1:optical-instruments:microscope}{ocular} para ver una estrella sin
        acomodar? Dese la distancia objetivo--ocular para cada \omterm{def:b1:optical-instruments:microscope}{ocular}.
  \item Compruébese con la \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relación de Descartes}, aplicada a cada \omterm{def:b1:mirrors-thin-lenses:lens}{lente}
        por turno, que un punto del eje en el infinito se conjuga con el
        infinito.
  \item Muéstrese que un haz procedente de una estrella con ángulo
        $\alpha$ respecto del eje emerge como haz paralelo con ángulo
        $\alpha' = -(f_1'/f_2')\alpha$.
  \item Calcúlense $G$ para cada \omterm{def:b1:optical-instruments:microscope}{ocular} y el diámetro aparente de la Luna
        a través de cada uno.
  \item La imagen está invertida. ¿Por qué resulta aceptable aquí, y qué
        añadiría un anteojo terrestre?
  \item ¿Dónde está la imagen intermedia real de la Luna y cuál es su
        diámetro? (Ahí puede fotografiarse.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dónde va el \omterm{def:b1:optical-instruments:eye}{ojo}.}
\begin{enumerate}[resume]
  \item Muéstrese que el \omterm{def:b1:optical-instruments:microscope}{ocular} forma una \omterm{def:b1:mirrors-thin-lenses:image}{imagen real} del borde del
        \omterm{def:b1:optical-instruments:microscope}{objetivo} a $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ por detrás
        de él, de diámetro $D/\abs G$ (la \omterm{prop:b1:optical-instruments:telescope}{pupila de salida}).
  \item Calcúlense la posición y el diámetro de la \omterm{prop:b1:optical-instruments:telescope}{pupila de salida} para
        cada \omterm{def:b1:optical-instruments:microscope}{ocular}.
  \item ¿Por qué debe situarse la pupila del \omterm{def:b1:optical-instruments:eye}{ojo} en la \omterm{prop:b1:optical-instruments:telescope}{pupila de salida}?
        ¿Qué le ocurre al campo si se coloca más atrás?
  \item Para aprovechar íntegra la luz de una estrella, la \omterm{prop:b1:optical-instruments:telescope}{pupila de
        salida} no debe superar la del \omterm{def:b1:optical-instruments:eye}{ojo}: dedúzcase el aumento útil
        mínimo de este anteojo de noche.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Resolución.}
\begin{enumerate}[resume]
  \item Calcúlese el límite de difracción $\theta_{\min} = 1.22\lambda/D$
        en radianes y en segundos de arco.
  \item ¿A qué cráter mínimo corresponde ese ángulo sobre la Luna? ¿Y para
        el \omterm{def:b1:optical-instruments:eye}{ojo} desnudo?
  \item Defínase y calcúlese el aumento resolvente $G_r =
        \epsilon/\theta_{\min}$. ¿Qué \omterm{def:b1:optical-instruments:microscope}{ocular} se le acerca más?
  \item Con el \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{6}{mm} la imagen se ve más borrosa y más
        apagada que con el de \qty{25}{mm}. Explíquense ambos efectos.
  \item La turbulencia atmosférica difumina las imágenes estelares hasta
        cerca de $1''$. ¿Para qué diámetros de \omterm{def:b1:optical-instruments:microscope}{objetivo} deja de importar
        el límite de Rayleigh desde el suelo, y qué sigue aportando un $D$
        mayor?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La luz.}
\begin{enumerate}[resume]
  \item ¿Por qué factor recoge el \omterm{def:b1:optical-instruments:microscope}{objetivo} más luz que la pupila
        nocturna?
  \item Conviértase a magnitudes ($\Delta m = 2.5\log_{10}$ del cociente);
        si el \omterm{def:b1:optical-instruments:eye}{ojo} llega a la magnitud $6$, ¿a qué magnitud llega el
        anteojo?
  \item La Luna llena entrega en el suelo una irradiancia de unos
        \qty{2e-3}{W/m^2}. ¿Qué potencia entra por el \omterm{def:b1:optical-instruments:microscope}{objetivo} y cómo se
        reparte sobre la \omterm{def:b1:optical-instruments:eye}{retina} en comparación con la visión a simple
        vista (mismo argumento de potencia por unidad de ángulo sólido:
        compárense los brillos superficiales a través de los dos \omterm{def:b1:optical-instruments:microscope}{oculares})?
  \item ¿Por qué un anteojo no hace más brillante la superficie lunar por
        unidad de área que el \omterm{def:b1:optical-instruments:eye}{ojo} desnudo, mientras que sí hace más
        brillantes las estrellas?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- El campo de visión.}
El \omterm{def:b1:optical-instruments:microscope}{ocular} lleva, en su plano focal, un diafragma de campo circular de
\qty{20}{mm} de diámetro (\omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}) o de \qty{5}{mm}
(\omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{6}{mm}).
\begin{enumerate}[resume]
  \item Muéstrese que el campo real vale $2\arctan(\phi/2f_1')
        \approx \phi/f_1'$ para un diafragma de diámetro $\phi$.
  \item Calcúlese para ambos \omterm{def:b1:optical-instruments:microscope}{oculares} y dígase si la Luna entera cabe en
        el campo.
  \item Calcúlese el campo aparente (el ángulo bajo el que el \omterm{def:b1:optical-instruments:eye}{ojo} ve el
        diafragma a través del \omterm{def:b1:optical-instruments:microscope}{ocular}) para ambos.
  \item La Tierra gira $15''$ por segundo de tiempo: ¿cuánto tarda la Luna
        en cruzar el campo de cada \omterm{def:b1:optical-instruments:microscope}{ocular} sin seguimiento?
  \item Resúmase el diseño: qué \omterm{def:b1:optical-instruments:microscope}{ocular} para la Luna entera, cuál para los
        cráteres y el tamaño del menor cráter que el instrumento puede
        revelar --- su número.
  \item El mismo \omterm{def:b1:optical-instruments:microscope}{objetivo} se sustituye por un \omterm{def:b1:mirrors-thin-lenses:mirror}{espejo cóncavo} de
        \qty{150}{mm} de la misma \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal}. ¿Cuáles de los
        resultados anteriores cambian?
\end{enumerate}
\end{problem}
