\chapter{El potencial eléctrico; conductores y condensadores}\label{ch:b1:potential-capacitors}

Un campo son tres números en cada punto; un potencial, uno solo. Como la
fuerza electrostática es conservativa, todo el campo de una distribución
de cargas puede plegarse en una única función escalar, el potencial, del
que se recupera derivando --- la misma economía que convirtió las fuerzas
en \omterm{def:b1:work-and-energy:conservative}{energía potencial} en mecánica. El potencial es lo que mide un
voltímetro, lo que mantiene una pila, lo que acelera los electrones en un
tubo; sus superficies de nivel dibujan un campo de un vistazo. Este
capítulo lo construye, lo aplica al dipolo (el primer modelo eléctrico de
una molécula) y se vuelve después hacia los dos objetos con los que está
hecho todo circuito: los conductores, en los que la carga se ordena de
modo que mata el campo interior, y los \omterm{def:b1:potential-capacitors:capacitance}{condensadores}, que almacenan carga
y energía en un campo.

\section{El potencial electrostático}

\begin{theorem}[El campo electrostático es conservativo]\label{thm:b1:potential-capacitors:potential}
\index{potencial electrostático}\index{voltio}\index{circulación}
Existe una función escalar $V(M)$, el \emph{\omterm{thm:b1:potential-capacitors:potential}{potencial electrostático}}
(voltio, \unit{V}), tal que
\[
\vect E = -\overrightarrow{\operatorname{grad}}\,V , \qquad
V_A - V_B = \int_A^B \vect E\cdot\dd\vect l
\]
a lo largo de cualquier camino de $A$ a $B$ (la \emph{circulación} de
$\vect E$ de $A$ a $B$ no depende del camino y es nula sobre un circuito
cerrado). Para una carga puntual $q$ situada en $O$, $V(M) = \dfrac{q}{4\pi\varepsilon_0\,OM}$
con el convenio $V(\infty) = 0$; para varias cargas los potenciales se
suman. La \omterm{def:b1:work-and-energy:conservative}{energía potencial} de una carga $q'$ en $M$ es $E_p = q'V(M)$, y
el trabajo de la fuerza eléctrica de $A$ a $B$ vale $q'(V_A - V_B)$.
\end{theorem}

\begin{proof}
La fuerza de Coulomb de una carga fija $q$ sobre $q'$ tiene la forma de
la \omterm{def:b1:central-forces:gravitation}{gravitación} de Newton con $-Gm_1m_2$ sustituido por $qq'/4\pi\varepsilon_0$;
el mismo cálculo que dio $E_p = -Gm_1m_2/r$
(\cref{prop:b1:work-and-energy:eps}) da $E_p = qq'/4\pi\varepsilon_0r$: al
dividir por $q'$ se obtiene $V$, y $\vect E = \vect F/q' =
-\overrightarrow{\operatorname{grad}}(E_p)/q'$. La superposición lleva el
resultado a cualquier distribución; la fórmula de la circulación es la
definición del gradiente integrada a lo largo del camino.
\end{proof}

\begin{proposition}[Cómo leer un potencial]\label{prop:b1:potential-capacitors:reading}
\index{superficie equipotencial}
\begin{enumerate}
  \item Las \emph{\omterm{prop:b1:potential-capacitors:reading}{superficies equipotenciales}} $V = \text{cte}$ son en
        todo punto perpendiculares a las \omterm{def:b1:electrostatics-gauss:field}{líneas de campo}, y el campo
        apunta hacia los potenciales \emph{decrecientes}.
  \item $|\vect E|$ es la rapidez de variación de $V$ en la dirección de
        mayor pendiente: donde las equipotenciales se aprietan, el campo
        es intenso.
  \item $V$ está definido salvo una constante aditiva; solo son medibles
        las diferencias (las tensiones). La tierra o el infinito se toman
        como cero por convenio.
  \item En \omterm{def:b1:point-kinematics:spherical}{coordenadas esféricas} con simetría en torno a $O$, $E_r = -\dd V/
        \dd r$; en \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares} $(r, \theta)$ en un plano,
        $E_r = -\partial V/\partial r$ y $E_\theta = -\dfrac1r\,\partial V/\partial\theta$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sobre una equipotencial $\dd V = -\vect E\cdot\dd\vect l = 0$ para todo
desplazamiento contenido en la superficie, luego $\vect E$ es normal a
ella; al avanzar según $\vect E$, $\dd V = -E\,\dd l < 0$. Las componentes
polares salen de
$\dd V = (\partial V/\partial r)\dd r + (\partial V/\partial\theta)\dd\theta$ y
$\dd\vect l = \dd r\,\vect e_r + r\,\dd\theta\,\vect e_\theta$.
\end{proof}

\begin{method}[Cómo calcular un potencial]\label{meth:b1:potential-capacitors:compute}
O bien se suma $\dd q/4\pi\varepsilon_0r$ sobre la distribución (cuando la suma
es manejable y la distribución es finita, de modo que $V(\infty) = 0$
tenga sentido), o bien se integra $\dd V = -\vect E\cdot\dd\vect l$ por un
camino cómodo desde un punto donde se conoce $V$, usando el campo hallado
con el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss}. Para las distribuciones infinitas (hilo, plano)
solo sirve la segunda vía, y el cero de $V$ ha de elegirse en un punto finito.
\end{method}

\begin{proposition}[Potenciales de las distribuciones clásicas]\label{prop:b1:potential-capacitors:classic}
\begin{enumerate}
  \item Esfera de radio $R$ con $Q$ en su volumen: $V(r) = \dfrac{Q}{4\pi\varepsilon_0r}$
        fuera; $V(r) = \dfrac{Q}{4\pi\varepsilon_0}\,\dfrac{3R^2 - r^2}{2R^3}$
        dentro, luego $V(0) = \tfrac32\,Q/4\pi\varepsilon_0R$. Con $Q$ en la superficie,
        $V$ es uniforme dentro e igual a $Q/4\pi\varepsilon_0R$.
  \item Hilo infinito $\lambda$: $V(r) = -\dfrac{\lambda}{2\pi\varepsilon_0}\ln\dfrac{r}{r_0}$.
  \item \omterm{prop:b1:potential-capacitors:three}{Condensador plano}, con campo $E$ de la armadura~1 a la
        armadura~2, separadas $d$: $V$ decrece linealmente en el hueco y $V_1 - V_2 = Ed$.
\end{enumerate}
\end{proposition}

\begin{proof}
Intégrese $E_r = -\dd V/\dd r$ con los campos de
\cref{prop:b1:electrostatics-gauss:three}: desde el infinito hacia dentro
para la esfera (la continuidad en $R$ fija la constante interior), desde
$r_0$ para el hilo y a través del hueco para el \omterm{def:b1:potential-capacitors:capacitance}{condensador}.
\end{proof}

\begin{example}[El electronvoltio, otra vez]\label{ex:b1:potential-capacitors:eV}
Un electrón que atraviesa un hueco de \qty{100}{V} gana $eU = \qty{100}{eV} =
\qty{1.6e-17}{J}$, sea cual sea la forma del campo entre los electrodos:
el potencial hace irrelevante el camino
(\cref{def:b1:charged-particles:eV}). Una carga de \qty{1}{\micro C} a
\qty{1}{m} de otra tiene $V = \qty{9}{kV}$ y el par almacena
\qty{9}{mJ}; el potencial en el centro de un núcleo de oro vale
\qty{24}{MV} (\cref{exo:b1:potential-capacitors:6}).
\end{example}

\section{El dipolo eléctrico}

\begin{definition}[Dipolo eléctrico]\label{def:b1:potential-capacitors:dipole}
\index{dipolo eléctrico}\index{momento dipolar}
Dos cargas opuestas, $-q$ en $N$ y $+q$ en $P$, separadas una distancia
$a = NP$, forman un \emph{dipolo eléctrico} de \emph{momento dipolar}
$\vect p = q\,\vect{NP}$ (\unit{C.m}), dirigido del $-$ al $+$. La
aproximación dipolar es el estudio de su campo a distancias $r \gg a$.
Una molécula cuyos centros de carga positiva y negativa no coinciden
(el agua, $p = \qty{6.2e-30}{C.m}$) es un dipolo; un átomo neutro en un
campo se convierte en uno (dipolo inducido).
\end{definition}

\begin{proposition}[Campo de un dipolo]\label{prop:b1:potential-capacitors:dipolefield}
En $M$ con $OM = r \gg a$ ($O$ el punto medio de $NP$) y $\theta$ el
ángulo entre $\vect p$ y $\vect{OM}$,
\[
V(M) = \frac{p\cos\theta}{4\pi\varepsilon_0 r^2}, \qquad
E_r = \frac{2p\cos\theta}{4\pi\varepsilon_0r^3}, \qquad
E_\theta = \frac{p\sin\theta}{4\pi\varepsilon_0r^3} .
\]
El campo de un dipolo cae como $1/r^3$, más deprisa que el $1/r^2$ de una
carga: de lejos las dos cargas casi se cancelan, y lo que queda es su
separación.
\end{proposition}

\begin{proof}
$V = \dfrac{q}{4\pi\varepsilon_0}\Bigl(\dfrac1{PM} - \dfrac1{NM}\Bigr)$ con
$PM \approx r - \tfrac a2\cos\theta$ y $NM \approx r + \tfrac a2\cos\theta$
(proyección de $\pm\tfrac a2$ sobre $\vect{OM}$); a primer orden en $a/r$,
$1/PM - 1/NM \approx a\cos\theta/r^2$. Después $E_r = -\partial V/\partial r$ y
$E_\theta = -(1/r)\,\partial V/\partial\theta$.
\end{proof}

\begin{proposition}[Dipolo en un campo exterior]\label{prop:b1:potential-capacitors:dipoleinfield}
\index{dipolo!par de fuerzas}\index{dipolo!energía}
Un dipolo $\vect p$ en un campo uniforme $\vect E$ no siente fuerza neta,
sino un par $\vect\Gamma = \vect p\wedge\vect E$ que tiende a alinearlo
con el campo, y tiene la \omterm{def:b1:work-and-energy:conservative}{energía potencial} $E_p = -\vect p\cdot\vect E$,
mínima cuando está alineado. En un campo no uniforme es además atraído
hacia las regiones de campo intenso (en una dimensión, $F_x = p\,\dd E/\dd x$
para un dipolo alineado).
\end{proposition}

\begin{proof}
Las fuerzas $\pm q\vect E$ se cancelan; su momento respecto de $O$ vale
$\vect{OP}\wedge
q\vect E + \vect{ON}\wedge(-q\vect E) = q\vect{NP}\wedge\vect E$. Energía:
$qV(P) - qV(N) = -q\,\vect{NP}\cdot\vect E$ en un campo uniforme (ya que $V(P)
- V(N) = -\vect E\cdot\vect{NP}$). No uniforme y alineado según $x$: $F_x =
qE(x + a) - qE(x) \approx qa\,\dd E/\dd x$.
\end{proof}

\begin{example}[Por qué el peine atrae al papel]\label{ex:b1:potential-capacitors:comb}
Un peine cargado crea un campo que decrece con la distancia; un trocito
de papel, neutro, queda polarizado por él (dipolo inducido según
$\vect E$), y la fuerza $p\,\dd E/\dd x$ sobre un dipolo alineado lo atrae
hacia la región de campo intenso: el peine. El signo de la carga del
peine es indiferente; el mismo mecanismo mantiene el globo en la pared y
desvía el hilo de agua. Para las moléculas de agua, $pE$ en un campo de
\qty{1}{MV/m} vale \qty{6e-24}{J}, $600$ veces menos que $k_BT$ a
temperatura ambiente: solo sobrevive una ligera alineación neta al
revoltijo térmico (\cref{exo:b1:potential-capacitors:7}).
\end{example}

\section{Los conductores en equilibrio electrostático}

\begin{theorem}[Propiedades de un conductor en equilibrio]\label{thm:b1:potential-capacitors:conductor}
\index{conductor en equilibrio}\index{teorema de Coulomb}\index{jaula de Faraday}
En un conductor cuyas cargas libres están en reposo:
\begin{enumerate}
  \item $\vect E = \vect 0$ en el interior, y el conductor es un volumen
        equipotencial: $V$ uniforme dentro y en su superficie.
  \item Toda carga neta se sitúa en la superficie, con una densidad
        superficial $\sigma$; el interior es neutro.
  \item Justo fuera, el campo es normal a la superficie y vale
        $\vect E = \dfrac{\sigma}{\varepsilon_0}\,\vect n$ (\emph{\omterm{thm:b1:potential-capacitors:conductor}{teorema de Coulomb}}).
  \item Una cavidad interior a un conductor, vacía de carga, tiene $\vect E =
        \vect 0$ y $V$ uniforme pase lo que pase fuera (\emph{\omterm{thm:b1:potential-capacitors:conductor}{jaula de
        Faraday}}).
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Un campo movería las cargas libres: el equilibrio exige $\vect E =
\vect 0$, luego $\dd V = 0$ entre dos puntos interiores cualesquiera; la
superficie es el límite. (2) El \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss} sobre cualquier
superficie cerrada trazada dentro del conductor: flujo nulo, carga
encerrada nula. (3) La superficie es una equipotencial, luego $\vect E$
justo fuera es normal a ella; un cilindro achatado que la atraviesa tiene
un flujo $E\,\dd S$ solo por su cara exterior (dentro $\vect E = \vect 0$)
y encierra la carga $\sigma\,\dd S$. (4) La pared de la cavidad es una
equipotencial; una \omterm{def:b1:electrostatics-gauss:field}{línea de campo} interior a la cavidad tendría que ir de
la pared a la pared salvando una caída de potencial --- imposible sobre
una equipotencial; así que no hay línea ni campo (admitido en esta forma).
\end{proof}

\begin{example}[Poder de las puntas y pararrayos]\label{ex:b1:potential-capacitors:point}
\index{poder de las puntas}
Dos esferas de radios $R_1 > R_2$, muy separadas y unidas por un hilo,
comparten el mismo $V = Q_i/4\pi\varepsilon_0R_i$, luego $\sigma_i = Q_i/4\pi R_i^2 = \varepsilon_0V/R_i$:
cuanto \emph{menor} es el \omterm{def:b1:point-kinematics:frenet}{radio de curvatura}, \emph{mayor} son la carga
superficial y el campo $\sigma/\varepsilon_0 = V/R$. En una punta afilada el
campo supera el valor de ruptura del aire (\qty{3}{MV/m}) mucho antes que
en ningún otro sitio: el aire se ioniza y la carga se fuga en silencio. Un
pararrayos funciona por este \emph{\omterm{ex:b1:potential-capacitors:point}{poder de las puntas}} --- no atrae tanto
la descarga como le ofrece un camino, y descarga el suelo bajo una nube
antes de que la carga se acumule.
\end{example}

la cavidad (la \omterm{thm:b1:potential-capacitors:conductor}{jaula de Faraday}). A la derecha: el \omterm{ex:b1:potential-capacitors:point}{poder de las puntas}
--- sobre un conductor, que es una sola equipotencial, la carga
superficial y el campo exterior son máximos donde la curvatura es mayor.}
\end{omfigure}

\section{Los condensadores}

\begin{definition}[Capacidad]\label{def:b1:potential-capacitors:capacitance}
\index{capacidad}\index{condensador}\index{faradio}
Un conductor aislado a potencial $V$ lleva una carga $Q$ proporcional a
$V$: $Q = CV$, con $C$ su \emph{capacidad} (faradio, \unit{F}); para una
esfera $C = 4\pi\varepsilon_0R$. Un \emph{condensador} es un par de conductores
(las \emph{armaduras}) que llevan cargas opuestas $\pm Q$ y se enfrentan
de modo que el campo queda confinado entre ellas; su capacidad es
$C = Q/U$, con $U = V_1 - V_2$ la tensión entre las armaduras. El símbolo
y la ley de circuito $i = C\,\dd u/\dd t$ se usaron en
\cref{ch:b1:transient-regimes}.
\end{definition}

\begin{proposition}[Tres condensadores]\label{prop:b1:potential-capacitors:three}
\index{condensador plano}\index{condensador esférico}\index{condensador cilíndrico}
\begin{enumerate}
  \item Plano: armaduras de área $S$ separadas $d \ll \sqrt S$:
        $C = \dfrac{\varepsilon_0S}{d}$.
  \item Esférico: esferas concéntricas de radios $R_1 < R_2$:
        $C = \dfrac{4\pi\varepsilon_0R_1R_2}{R_2 - R_1}$.
  \item Cilíndrico (coaxial), radios $a < b$, longitud $L \gg b$:
        $C = \dfrac{2\pi\varepsilon_0L}{\ln(b/a)}$.
\end{enumerate}
Llenar el hueco con un aislante de permitividad relativa
$\varepsilon_r$ multiplica $C$ por $\varepsilon_r$ (las moléculas del dieléctrico
se polarizan y debilitan el campo, $\varepsilon_r = 2$ a $10$ en los
plásticos y el vidrio corrientes, $80$ en el agua).
\end{proposition}

\begin{proof}
En cada caso el campo entre las armaduras es el de
\cref{prop:b1:electrostatics-gauss:three}: $E = \sigma/\varepsilon_0 = Q/\varepsilon_0S$,
$U = Ed$; $E = Q/4\pi\varepsilon_0r^2$, $U = \int_{R_1}^{R_2}E\,\dd r = \dfrac{Q}{4\pi\varepsilon_0}
\Bigl(\dfrac1{R_1} - \dfrac1{R_2}\Bigr)$; $E = Q/2\pi\varepsilon_0Lr$, $U =
\dfrac{Q}{2\pi\varepsilon_0L}\ln\dfrac ba$. Divídase $Q$ por $U$. El factor del
dieléctrico se admite.
\end{proof}

\begin{theorem}[Energía de un condensador]\label{thm:b1:potential-capacitors:energy}
\index{energía de un condensador}\index{densidad de energía del campo eléctrico}
Un \omterm{def:b1:potential-capacitors:capacitance}{condensador} cargado con $Q$ bajo la tensión $U$ almacena
\[
W = \tfrac12 CU^2 = \tfrac12 QU = \frac{Q^2}{2C} ,
\]
el trabajo necesario para llevar la carga de una armadura a la otra
contra la tensión creciente. En un \omterm{prop:b1:potential-capacitors:three}{condensador plano} $W = \tfrac12\varepsilon_0E^2
\times Sd$: la energía está en el campo, con una densidad
$\tfrac12\varepsilon_0E^2$ por unidad de volumen --- afirmación válida para
todo campo electrostático.
\end{theorem}

\begin{proof}
Llevar $\dd q$ de la armadura negativa a la positiva cuando la carga vale
$q$ cuesta $\dd W = (q/C)\,\dd q$; intégrese de $0$ a $Q$. Sustitúyanse
después $C = \varepsilon_0S/d$ y $U = Ed$; la densidad general se admite.
\end{proof}

\begin{example}[Tres condensadores y sus energías]\label{ex:b1:potential-capacitors:energies}
Un metro cuadrado de par de láminas separadas \qty{1}{mm}: $C = 8.85 \times 10^{-12}/
10^{-3} = \qty{8.9}{nF}$ --- el \omterm{def:b1:potential-capacitors:capacitance}{faradio} es una unidad enorme. Los
\qty{150}{\micro F} de un desfibrilador a \qty{2}{kV}: \qty{300}{J},
entregados en milisegundos. Un supercondensador de \qty{3000}{F} a
\qty{2.7}{V}: \qty{11}{kJ} en un bote del tamaño de una botella de
refresco --- su \omterm{def:b1:potential-capacitors:capacitance}{capacidad} viene de un hueco $d$ molecular y de un $S$
poroso enorme. La densidad de energía de un campo en la ruptura del aire,
$\tfrac12\varepsilon_0(3 \times 10^6)^2 = \qty{40}{J/m^3}$, explica por qué los
\omterm{def:b1:potential-capacitors:capacitance}{condensadores} almacenan tan poco frente a las baterías.
\end{example}

\begin{remark}[Asociación de condensadores]\label{rem:b1:potential-capacitors:association}
En paralelo (misma $U$) las cargas se suman: $C = C_1 + C_2$; en serie
(misma $Q$) las tensiones se suman: $1/C = 1/C_1 + 1/C_2$ --- al revés que
las resistencias, porque $C$ mide una facilidad (carga por voltio) allí
donde $R$ mide una dificultad. Estas reglas salen de $Q = CU$ y de las
leyes de los circuitos de \cref{ch:b1:dc-circuits}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:1}
Potencial a \qty{1.0}{m} de una carga de \qty{1.0}{\micro C}; energía
necesaria para traer una segunda carga de \qty{1.0}{\micro C} desde el
infinito hasta ese punto; rapidez con la que saldría despedida si se la
soltara, siendo su masa de \qty{1}{g}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:2}
Armaduras separadas \qty{1.0}{mm} bajo \qty{100}{V}: campo; \omterm{thm:b1:work-and-energy:ke}{energía
cinética} y rapidez que gana un electrón que cruza el hueco partiendo del
reposo; tiempo de vuelo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:3}
En un mapa de equipotenciales trazadas cada \qty{10}{V}, las líneas
distan \qty{2}{mm} cerca de $A$ y \qty{8}{mm} cerca de $B$. Campo en $A$ y
en $B$; ¿hacia dónde apunta el campo respecto de los potenciales
crecientes?; ¿hacia dónde se acelera un electrón?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:4}
Una esfera de Van de Graaff de \qty{15}{cm} de radio se carga a \qty{200}{kV}.
\omterm{def:b1:potential-capacitors:capacitance}{Capacidad}, carga y campo en su superficie. Potencial máximo antes de que
el aire de alrededor rompa (\qty{3}{MV/m}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:5}
Potencial en el eje de un anillo de radio $R$ y carga $Q$; recupérese el
campo axial derivando. Lo mismo para el disco cargado uniformemente
(úsese \cref{exo:b1:electrostatics-gauss:5}): $V(z) = \dfrac{\sigma}{2\varepsilon_0}
\bigl(\sqrt{z^2 + R^2} - z\bigr)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:6}
Dedúzcase el potencial dentro de una esfera cargada uniformemente a
partir de su campo. Potencial en el centro de un núcleo de oro ($Z = 79$,
$R = \qty{7.0}{fm}$); energía que necesitaría un protón para llegar al
centro desde muy lejos, en \unit{MeV}; coméntense las partículas alfa de
\qty{5}{MeV} de Rutherford.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:7}
Molécula de agua, $p = \qty{6.2e-30}{C.m}$. Campo que crea a \qty{1.0}{nm}
en su eje y perpendicularmente a él; par y diferencia de energía entre
las posiciones alineada y antialineada en $E = \qty{1.0}{MV/m}$;
compárese con $k_BT$ a \qty{300}{K}. Campo de un ion de sodio a
\qty{0.3}{nm} y energía del dipolo de agua alineado en él (hidratación).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:8}
Una esfera conductora de \qty{1.0}{cm} de radio en aire: carga y potencial
máximos antes de la ruptura. Dos esferas conductoras, $R_1 = \qty{10}{cm}$
y $R_2 = \qty{1.0}{mm}$, unidas por un hilo fino y llevadas a
\qty{20}{kV}: campo superficial en cada una; ¿cuál descarga en el aire?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:9}
\omterm{def:b1:potential-capacitors:capacitance}{Capacidades}: dos láminas de \qty{1.0}{m^2} separadas \qty{10}{\micro m}
por una película de polímero con $\varepsilon_r = 2.2$ (un \omterm{def:b1:potential-capacitors:capacitance}{condensador} de
película); las mismas láminas enrolladas: ¿cambia $C$ al enrollarlas?
Energía a \qty{400}{V}. Un \omterm{def:b1:potential-capacitors:capacitance}{condensador} de \qty{150}{\micro F} a
\qty{2.0}{kV} (desfibrilador): carga, energía y \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} si se
entrega en \qty{5}{ms}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:10}
\emph{\omterm{prop:b1:potential-capacitors:three}{Condensadores esférico} y cilíndrico.} (a) Dedúzcase $C$ para dos
esferas concéntricas y compruébese que, si $R_2 - R_1 = d \ll R_1$, se
reduce a la fórmula del plano, y que si $R_2 \to \infty$ da la esfera
aislada. (b) Cable coaxial, $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, con
polietileno $\varepsilon_r = 2.3$: \omterm{def:b1:potential-capacitors:capacitance}{capacidad} por metro (compárese con los
\qty{100}{pF/m} de los cables corrientes). (c) Campo en el conductor
interior bajo \qty{1}{kV}; ¿es seguro?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:11}
\emph{La energía del campo.} (a) Energía de una esfera conductora cargada
($Q$, $R$) a partir de $Q^2/2C$. (b) Iguálese a la energía en reposo del
electrón $m_ec^2 = \qty{0.511}{MeV}$ y despéjese $R$: el ``radio clásico
del electrón'' (salvo un factor). (c) A un \omterm{prop:b1:potential-capacitors:three}{condensador plano} aislado
cargado con $Q$ se le separan las armaduras de $d$ a $2d$: variación de la
energía almacenada y fuerza entre las armaduras deducida de ella;
compruébese frente a \cref{exo:b1:electrostatics-gauss:8}. (d) Lo mismo
con el \omterm{def:b1:potential-capacitors:capacitance}{condensador} mantenido a $U$ fija por una pila: variación de energía
y adónde va la diferencia.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:12}
\emph{Energía coulombiana de un núcleo.} (a) Constrúyase capa a capa una
esfera cargada uniformemente ($Q$, $R$) y muéstrese que su energía
electrostática vale $W = \dfrac{3Q^2}{20\pi\varepsilon_0R}$. (b) Uranio-238: $Z = 92$, $R =
\qty{7.4}{fm}$: $W$ en \unit{MeV}. (c) Divídase en dos fragmentos iguales
($Z/2$, $A/2$, radio $R/2^{1/3}$) muy separados: energía coulombiana
liberada. (d) La energía medida de la fisión ronda los \qty{200}{MeV}:
¿qué se opone a la ganancia coulombiana?
\end{exercise}

\section{Problema: la gota de aceite de Millikan}

\begin{problem}[{Problema de fin de semana --- pesar la carga del electrón
con una gota de aceite: Stokes, el condensador plano y una tabla de
números que se niegan a ser otra cosa que múltiplos de uno solo}]
\label{pb:b1:potential-capacitors:1}
Se pulverizan gotas de aceite entre dos armaduras horizontales separadas
$d = \qty{16}{mm}$, de \qty{20}{cm} de diámetro, y se observan con un
\omterm{def:b1:optical-instruments:microscope}{microscopio}. Datos: densidad del aceite $\rho = \qty{900}{kg/m^3}$,
densidad del aire \qty{1.2}{kg/m^3}, viscosidad del aire $\eta = \qty{1.8e-5}{Pa.s}$, $g =
\qty{9.8}{m/s^2}$, $\varepsilon_0 = \qty{8.85e-12}{F/m}$. Una esfera pequeña de
radio $r$ que se mueve a $v$ en el aire sufre la fuerza de Stokes $6\pi\eta rv$.

\medskip
\noindent\textbf{Parte I --- La gota que cae.}
\begin{enumerate}
  \item Fuerzas sobre una gota que cae sin tensión aplicada; muéstrese que
        alcanza una \omterm{prop:b1:newton-dynamics:terminal}{velocidad límite} $v_1 = 2\rho gr^2/9\eta$ (despréciese
        el empuje del aire, y justifíquese).
  \item Una gota cae \qty{1.0}{mm} en \qty{10}{s}: su radio y su masa.
  \item \omterm{def:b1:transient-regimes:firstorder}{Constante de tiempo} con que se aproxima a $v_1$; distancia
        recorrida entretanto. ¿Se ve alguna vez acelerar la gota?
  \item Número de Reynolds $\rho_{\text{aire}}v_1r/\eta$ del movimiento:
        ¿es legítima la \omterm{prop:b1:newton-dynamics:drag}{ley de Stokes}?
  \item ¿Por qué han de ser tan pequeñas las gotas? (Compárese el peso con
        la fuerza eléctrica sobre una \omterm{def:b1:electrostatics-gauss:charge}{carga elemental} en un campo de
        \qty{3e5}{V/m}.)
  \item Se ve que la gota tiembla ligeramente en torno a su caída media:
        ¿qué es esto y por qué limita la precisión?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El campo.}
Se aplica una tensión $U$, con la armadura superior positiva.
\begin{enumerate}[resume]
  \item Campo entre las armaduras para $U = \qty{5.0}{kV}$; ¿por qué es
        uniforme y cuáles son las equipotenciales?
  \item \omterm{def:b1:potential-capacitors:capacitance}{Capacidad} de las armaduras, carga que llevan a \qty{5.0}{kV} y
        energía almacenada.
  \item La gota de la pregunta~2 queda inmóvil con $U_0 = \qty{1080}{V}$:
        signo y valor de su carga, en culombios y en unidades de $e$.
  \item Con $U = \qty{5.0}{kV}$ la misma gota sube a $v_2$: muéstrese que
        $q = 6\pi\eta r(v_1 + v_2)d/U$ y calcúlese $v_2$.
  \item ¿Por qué es mejor cronometrar una subida y una caída que buscar la
        tensión de equilibrio?
  \item \omterm{def:b1:work-and-energy:conservative}{Energía potencial} de la carga de la gota entre las armaduras a
        \qty{5}{kV}, y energía disipada por el rozamiento durante una
        subida de \qty{1}{cm}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La tabla de cargas.}
Expuesta entre medidas a una fuente débil de rayos X, la misma gota
cambia de carga; las medidas sucesivas dan (en $10^{-19}$~C):
$4.80$, $8.01$, $6.42$, $3.19$, $9.63$.
\begin{enumerate}[resume]
  \item Muéstrese que las cinco están cerca de múltiplos enteros de una
        misma cantidad, y dense los enteros.
  \item Mejor estimación de la \omterm{def:b1:electrostatics-gauss:charge}{carga elemental} a partir de los cinco
        valores; \omterm{def:b1:units-dimensions:uncertainty}{incertidumbre típica} de la media (\cref{ch:b1:units-dimensions}).
  \item ¿Por qué cambia la carga en números enteros de $e$ entre dos
        medidas?
  \item Muéstrese que, con este método, $q \propto \eta^{3/2}$: ¿cuánto
        desplaza un error del $1\%$ en la viscosidad el valor de $e$? (El
        valor original de Millikan era un $0.6\%$ bajo, justamente por eso.)
  \item Los quarks llevan $\pm e/3$ y $\pm 2e/3$: ¿por qué ninguna gota de
        aceite ha mostrado nunca un tercio de $e$?
  \item La constante de Faraday, la carga de un \omterm{def:b1:kinetic-theory:scales}{mol} de electrones, vale
        $F = \qty{96485}{C/mol}$: dedúzcase el número de Avogadro.
  \item Thomson había medido $e/m_e = \qty{1.76e11}{C/kg}$ para el
        electrón: dedúzcase su masa y compárese con un átomo de hidrógeno
        (\qty{1.67e-27}{kg}).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El aparato, visto por el potencial.}
\begin{enumerate}[resume]
  \item Esbócense las equipotenciales y las \omterm{def:b1:electrostatics-gauss:field}{líneas de campo} entre las
        armaduras, incluidos los efectos de borde; justifíquese que se
        desprecien.
  \item Una gota cargada positivamente a la altura $z$ sobre la armadura
        inferior (a \qty{0}{V}): \omterm{def:b1:work-and-energy:conservative}{energía potencial} en función de $z$;
        \omterm{def:b1:work-and-energy:conservative}{energía potencial} total incluida la gravedad; condición sobre $U$
        para el punto de equilibrio.
  \item Compárese la energía de la carga de la gota (pregunta~12) con la
        energía almacenada en el \omterm{def:b1:potential-capacitors:capacitance}{condensador}: ¿perturba la gota el campo?
  \item Se invierte la tensión (armadura inferior positiva) con
        $U = \qty{5.0}{kV}$: ¿qué hace la gota de la pregunta~9 y a qué
        rapidez?
  \item Un grano de polvo de \qty{1}{\micro g} que lleva $10^4e$ en el
        mismo campo: cociente entre la fuerza eléctrica y el peso; por qué
        no pueden levitarse esos granos y por qué eligió Millikan una
        niebla de aceite.
  \item Resúmase: qué se midió (tres veces), qué se dedujo, con qué
        incertidumbre y qué dos constantes se siguieron de ello.
\end{enumerate}
\end{problem}
