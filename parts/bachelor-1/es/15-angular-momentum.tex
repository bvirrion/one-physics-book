\chapter{Momento angular}\label{ch:b1:angular-momentum}

Una patinadora que gira con los brazos extendidos los recoge y, sin que
nadie la empuje, gira el doble de deprisa. Una puerta cede ante la yema de
un dedo en el picaporte y resiste un hombro junto a la bisagra. Los
planetas barren áreas iguales en tiempos iguales --- Kepler lo advirtió
hace cuatro siglos, mucho antes de que nadie supiera decir por qué. Y la
Luna, tironeada por las mareas que levanta, se aleja de la Tierra unos
centímetros al año mientras nuestros días se alargan de forma
imperceptible. Detrás de los cuatro casos hay una magnitud vectorial, el
\omterm{def:b1:angular-momentum:L}{momento angular}, y una ley: su ritmo de variación es igual al \omterm{def:b1:angular-momentum:moment}{momento} de
las fuerzas. Este capítulo define ambos, demuestra la ley para un punto y
para un sistema, y la pone a trabajar en péndulos, órbitas y cosas que giran.

\section{Momento de una fuerza}

\begin{definition}[Momento respecto de un punto; respecto de un eje]\label{def:b1:angular-momentum:moment}
\index{momento de una fuerza}\index{par}\index{brazo de palanca}
El \emph{momento} (o \emph{par}) respecto de un punto $O$ de una fuerza
$\vect F$ aplicada en $M$ es el vector
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
perpendicular al plano de $\vect{OM}$ y $\vect F$, de norma
$OM\cdot F\sin\alpha = F\,d$, donde $d$ --- el \emph{brazo de palanca} ---
es la distancia de $O$ a la recta de acción de $\vect F$. El momento
respecto de un eje orientado $\Delta$ (de vector unitario $\vect e_\Delta$)
que pasa por $O$ es el escalar $\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$,
independiente de la elección de $O$ sobre $\Delta$; se anula si $\vect F$
es paralela a $\Delta$ o lo corta. Dos fuerzas opuestas con rectas de
acción distintas forman un \emph{par de fuerzas}, cuyo momento
$\vect F\wedge\vect{AB}$ es el mismo respecto de todo punto.
\end{definition}

\begin{example}[La puerta]\label{ex:b1:angular-momentum:door}
Un empujón de \qty{20}{N} perpendicular a una puerta a \qty{0.80}{m} de
las bisagras tiene un \omterm{def:b1:angular-momentum:moment}{momento} de $\qty{16}{N.m}$ respecto del eje de las
bisagras; el mismo empujón a \qty{0.10}{m}, \qty{2}{N.m}; a lo largo de la
puerta (con la recta de acción pasando por las bisagras), cero. La propia
reacción de la bisagra, que corta al eje, no tiene \omterm{def:b1:angular-momentum:moment}{momento} respecto de él:
eso hace del eje el lugar natural para tomar \omterm{def:b1:angular-momentum:moment}{momentos}.
\end{example}

\section{Momento angular de un punto material}

\begin{definition}[Momento angular]\label{def:b1:angular-momentum:L}
\index{momento angular}
El \emph{\omterm{def:b1:angular-momentum:moment}{momento} angular} respecto de $O$ de un \omterm{def:b1:point-kinematics:frame}{punto material} de masa $m$
situado en $M$ con velocidad $\vect v$ es
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
y, respecto de un eje $\Delta$ que pasa por $O$, $L_\Delta = \vect L_O\cdot\vect e_\Delta$.
Para un movimiento en un plano perpendicular a $\Delta$, en \omterm{def:b1:point-kinematics:cylindrical}{coordenadas
polares} en torno al eje, $L_\Delta = mr^2\dot\theta$; para una
circunferencia de radio $R$, $L_\Delta = mR^2\omega = mRv$.
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$, $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[Teorema del momento angular]\label{thm:b1:angular-momentum:theorem}
\index{teorema del momento angular}
En un \omterm{thm:b1:newton-dynamics:laws}{sistema inercial}, para un punto $O$ fijo en ese sistema,
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
para un eje fijo $\Delta$. Si el \omterm{def:b1:angular-momentum:moment}{momento} total respecto de $O$ se anula,
$\vect L_O$ se conserva.
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$, usando la segunda ley de Newton
($O$ fijo, de modo que $\dd\vect{OM}/\dd t = \vect v$). Proyéctese sobre $\vect e_\Delta$.
\end{proof}

\begin{corollary}[Fuerzas centrales: movimiento plano y ley de las áreas]\label{cor:b1:angular-momentum:central}
\index{fuerza central}\index{ley de las áreas}\index{velocidad areolar}
Un \omterm{def:b1:point-kinematics:frame}{punto material} sometido únicamente a una \emph{fuerza central}
(siempre dirigida según $\vect{OM}$, hacia el centro fijo $O$ o en sentido
contrario) mantiene $\vect L_O$ constante. Su movimiento se queda en el
plano que pasa por $O$ perpendicular a $\vect L_O$, y en ese plano
$r^2\dot\theta = C$ (constante): el radio vector $\vect{OM}$ barre área
con la \emph{\omterm{cor:b1:angular-momentum:central}{velocidad areolar}} constante
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$ para $\vect F \parallel \vect{OM}$:
$\vect L_O$ es constante. $\vect{OM}\perp\vect L_O$ siempre, de ahí el
plano; $L_O =
mr^2\dot\theta$ es constante. El área barrida durante $\dd t$ es la del
triángulo de lados $\vect{OM}$ y $\vect v\dd t$: $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$.
\end{proof}

\begin{example}[El péndulo por momentos]\label{ex:b1:angular-momentum:pendulum}
\omterm{ex:b1:newton-dynamics:pendulum}{Péndulo simple}, con el eje $\Delta$ por el pivote perpendicular al plano
de oscilación: $L_\Delta = m\ell^2\dot\theta$; la tensión corta al eje (no
tiene \omterm{def:b1:angular-momentum:moment}{momento}) y el \omterm{def:b1:angular-momentum:moment}{momento} del peso vale $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{brazo de
palanca} $\ell\sin\theta$, recuperador). El teorema da $m\ell^2\ddot\theta =
-mg\ell\sin\theta$, la ecuación de \cref{ex:b1:newton-dynamics:pendulum}
sin escribir jamás la tensión --- la principal ventaja de los \omterm{def:b1:angular-momentum:moment}{momentos}:
las fuerzas que pasan por el eje desaparecen.
\end{example}

\begin{example}[Recoger un disco hacia dentro]\label{ex:b1:angular-momentum:puck}
Un disco desliza sobre hielo sin rozamiento describiendo una
circunferencia, sujeto por un hilo que pasa por un agujero en el centro;
se tira del hilo hasta reducir el radio a la mitad. La tensión es central:
$L = mr^2\omega$ se conserva, luego $\omega$ queda multiplicada por $4$, la
rapidez $r\omega$ por $2$ y la \omterm{thm:b1:work-and-energy:ke}{energía cinética} por $4$ --- la energía de
más es el trabajo del tirón, al que el disco se resiste durante todo el
recorrido. Los brazos de la patinadora son la misma física con la masa
repartida (\cref{exo:b1:angular-momentum:4}).
\end{example}

\section{Sistemas de puntos y rotación en torno a un eje fijo}

\begin{theorem}[Momento angular de un sistema]\label{thm:b1:angular-momentum:system}
\index{momento angular!de un sistema}
Para un sistema de \omterm{def:b1:point-kinematics:frame}{puntos materiales}, con $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
y $O$ fijo en un \omterm{thm:b1:newton-dynamics:laws}{sistema inercial},
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
solo cuentan los \omterm{def:b1:angular-momentum:moment}{momentos} de las fuerzas \emph{exteriores}; las fuerzas
interiores, que cumplen la ley de \omterm{thm:b1:newton-dynamics:laws}{acción y reacción} a lo largo de la recta
que une los puntos, no aportan nada.
\end{theorem}

\begin{proof}
Súmense los teoremas de cada punto. Para una pareja interior, $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$, ya que la fuerza va
según $\vect{M_jM_i}$.
\end{proof}

\begin{proposition}[Rotación rígida en torno a un eje fijo]\label{prop:b1:angular-momentum:inertia}
\index{momento de inercia}
Si los puntos mantienen distancias $r_i$ fijas a un eje $\Delta$ y todos
giran en torno a él con la misma \omterm{cor:b1:point-kinematics:circular}{velocidad angular} $\omega$ (rotación rígida),
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
donde $J_\Delta$ es el \emph{\omterm{prop:b1:angular-momentum:inertia}{momento de inercia}} respecto de $\Delta$, y el
teorema se lee $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$.
\end{proposition}

\begin{proof}
Cada punto aporta $m_ir_i^2\omega$ a $L_\Delta$ y $\tfrac12 m_i(r_i\omega)^2$
a $E_k$. \cref{ch:b1:systems-of-points} lo extiende a los sólidos continuos.
\end{proof}

\begin{example}[La patinadora]\label{ex:b1:angular-momentum:skater}
Con los brazos extendidos, el cuerpo de una patinadora tiene
$J \approx \qty{4}{kg.m^2}$ más dos brazos de \qty{3}{kg} a \qty{0.8}{m}: $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$. Con los brazos recogidos (\qty{0.2}{m}):
$J_2 = \qty{4.2}{kg.m^2}$. La reacción del hielo y el peso no tienen
\omterm{def:b1:angular-momentum:moment}{momento} respecto del eje vertical: $J\omega$ se conserva y el ritmo de giro
sube en $J_1/J_2 = 1.85$ --- de $2$ a $3.7$ vueltas por segundo ---
mientras la \omterm{thm:b1:work-and-energy:ke}{energía cinética} sube en ese mismo factor, pagada por los
músculos que recogen los brazos.
\end{example}

\begin{remark}[Estática]\label{rem:b1:angular-momentum:statics}
\index{equilibrio de un sólido}
Un sólido en reposo tiene $\vect L_O = \vect 0$ constante: las fuerzas
exteriores han de tener resultante nula \emph{y} \omterm{def:b1:angular-momentum:moment}{momento} total nulo
respecto de cualquier punto. La segunda condición es la ley de la palanca
de la balanza, el balancín y la palanqueta: una fuerza pequeña lejos del
pivote equilibra a una grande cerca de él.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:1}
Una fuerza de \qty{50}{N} sobre una llave a \qty{0.30}{m} de la tuerca:
\omterm{def:b1:angular-momentum:moment}{momento} cuando la fuerza es perpendicular al mango; a $60^\circ$ de él; a
lo largo de él.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:2}
\omterm{def:b1:angular-momentum:L}{Momento angular} de la Tierra respecto del Sol (órbita circular,
$r = \qty{1.50e11}{m}$, $v = \qty{29.8}{km/s}$, $m = \qty{5.97e24}{kg}$) y
su \omterm{cor:b1:angular-momentum:central}{velocidad areolar}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:3}
Un péndulo cónico (hilo de \qty{1.0}{m}, a $30^\circ$ de la vertical,
\cref{exo:b1:newton-dynamics:9}). Calcúlese $L$ respecto del eje vertical
que pasa por el pivote, por unidad de masa, y explíquese por qué se
conserva aunque actúen la tensión del hilo y el peso.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:4}
El tronco de una patinadora tiene $J = \qty{4.0}{kg.m^2}$ respecto del eje
de giro; cada brazo (\qty{3.0}{kg}) se trata como un punto a \qty{0.80}{m}
y después a \qty{0.20}{m}. Ritmo de giro tras recoger los brazos,
partiendo de \qty{2.0}{} vueltas por segundo; cociente de \omterm{thm:b1:work-and-energy:ke}{energías
cinéticas}; ¿de dónde sale la energía de más?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:5}
Dedúzcase la ecuación del péndulo $\ell\ddot\theta + g\sin\theta = 0$ a
partir del \omterm{thm:b1:angular-momentum:theorem}{teorema del momento angular} respecto del eje del pivote, y dese
el \omterm{def:b1:angular-momentum:moment}{momento} de cada fuerza.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:6}
La distancia de la Tierra al Sol varía de \qty{1.471e11}{m} (perihelio) a
\qty{1.521e11}{m} (afelio). Con la \omterm{cor:b1:angular-momentum:central}{ley de las áreas}, hállese el cociente
de sus rapideces en esos puntos; con una rapidez media de \qty{29.8}{km/s},
dense ambas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:7}
Un disco de \qty{0.20}{kg} gira a \qty{2.0}{m/s} atado a un hilo de
\qty{0.50}{m} que pasa por un agujero; se tira del hilo hasta que el radio
vale \qty{0.25}{m}. Nueva rapidez y nueva \omterm{cor:b1:point-kinematics:circular}{velocidad angular}; \omterm{thm:b1:work-and-energy:ke}{energías
cinéticas} antes y después; trabajo realizado por el tirón. ¿Por qué puede
aquí la tensión realizar trabajo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:8}
Un tablón homogéneo de \qty{20}{kg} y \qty{4.0}{m} de largo se apoya en un
pivote a \qty{1.5}{m} de su extremo izquierdo; un niño de \qty{30}{kg} se
sienta en ese extremo. ¿Dónde debe sentarse un niño de \qty{45}{kg} para
equilibrarlo? ¿Qué soporta el pivote?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:9}
Una máquina de Atwood ($m_1 = \qty{2.0}{kg}$, $m_2 = \qty{1.0}{kg}$) cuelga
de una polea de radio \qty{0.10}{m} y \omterm{prop:b1:angular-momentum:inertia}{momento de inercia} \qty{0.010}{kg.m^2}
sobre la que el hilo no desliza. Con el \omterm{thm:b1:angular-momentum:theorem}{teorema del momento angular} para
la polea y la ley de Newton para las masas, hállense la aceleración y las
dos tensiones. Compárese con la polea sin masa.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:10}
Un cometa pasa por su perihelio a \qty{0.50}{AU} del Sol a \qty{40}{km/s}.
Cuando está a \qty{10}{AU}, ¿cuál es la componente de su velocidad
perpendicular al radio vector? ¿Puede la \omterm{cor:b1:angular-momentum:central}{ley de las áreas} dar su rapidez
completa?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:11}
Una estrella como el Sol ($R = \qty{7.0e8}{m}$, periodo de rotación
\qty{25}{d}) se colapsa en una estrella de neutrones de radio \qty{10}{km},
conservando su masa y su \omterm{def:b1:angular-momentum:L}{momento angular} ($J \propto MR^2$). ¿Nuevo
periodo? ¿Rapidez ecuatorial? ¿Qué dice esta última sobre la hipótesis?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:12}
Una cuenta desliza sin rozamiento por una varilla recta que gira en un
plano horizontal con $\omega$ constante en torno a un eje vertical que pasa
por uno de sus extremos. Escríbase $L_z$ de la cuenta; muéstrese que
\emph{no} se conserva y calcúlese el \omterm{def:b1:angular-momentum:moment}{momento} que debe ejercer la varilla;
dedúzcase la reacción transversal de la varilla sobre la cuenta.
\end{exercise}

\section{Problema: por qué se va la Luna}

\begin{problem}[{Problema de fin de semana --- las mareas que la Luna
levanta en la Tierra tiran de ella hacia delante y de la Tierra hacia
atrás: cómo fluye el momento angular de nuestros días a la órbita lunar,
centímetro a centímetro, y adónde va la energía}]
\label{pb:b1:angular-momentum:1}
Datos: $G = \qty{6.67e-11}{SI}$; Tierra $M = \qty{5.97e24}{kg}$, $R =
\qty{6.37e6}{m}$, rotación $\Omega = 2\pi/(\qty{86164}{s})$, \omterm{prop:b1:angular-momentum:inertia}{momento de
inercia} $J = 0.33\,MR^2$; Luna $m = \qty{7.35e22}{kg}$, con la órbita
tratada como circular, $r = \qty{3.84e8}{m}$; la telemetría láser lunar
mide $\dd r/\dd t = \qty{3.8}{cm/yr}$; el día se alarga \qty{2.3}{ms} por
siglo; $1$ siglo $= \qty{3.16e9}{s}$.

\medskip
\noindent\textbf{Parte I --- Los \omterm{def:b1:angular-momentum:L}{momentos angulares}.}
\begin{enumerate}
  \item Calcúlese la \omterm{cor:b1:point-kinematics:circular}{velocidad angular} orbital $\omega$ de la Luna con la
        ley de Newton (órbita circular) y compruébese frente al mes
        sidéreo de \qty{27.3}{d}.
  \item Calcúlese el \omterm{def:b1:angular-momentum:L}{momento angular} orbital de la Luna $L_{\text{orb}} =
        mr^2\omega$.
  \item Muéstrese que, para una órbita circular, $L_{\text{orb}} = m\sqrt{GMr}$,
        y dedúzcase $\dd L_{\text{orb}}/\dd r = L_{\text{orb}}/2r$.
  \item Calcúlese el \omterm{def:b1:angular-momentum:L}{momento angular} de rotación de la Tierra $L_{\text{rot}} =
        J\Omega$ y compárese con $L_{\text{orb}}$.
  \item La Luna gira sobre sí misma una vez por mes: estímese su \omterm{def:b1:angular-momentum:L}{momento
        angular} de rotación ($J_m \approx 0.4\,mR_m^2$, $R_m = \qty{1.74e6}{m}$)
        y justifíquese despreciarlo.
  \item ¿Qué \omterm{def:b1:angular-momentum:moment}{momentos} exteriores actúan sobre el sistema Tierra--Luna
        respecto de su centro de masas? Arguméntese que, con buena
        aproximación, el total $L = L_{\text{rot}} + L_{\text{orb}}$ se conserva.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El \omterm{def:b1:angular-momentum:moment}{par} de marea.}
La Luna levanta dos abultamientos de agua en la Tierra; como la Tierra
gira más deprisa de lo que la Luna orbita, el rozamiento arrastra los
abultamientos por delante de la línea Tierra--Luna en un ángulo pequeño.
\begin{enumerate}[resume]
  \item Esbócense la Tierra, el abultamiento adelantado y la Luna. ¿En qué
        sentido actúa la atracción de la Luna sobre el abultamiento más
        cercano, respecto de la rotación de la Tierra?
  \item Dedúzcase el signo del \omterm{def:b1:angular-momentum:moment}{momento} sobre la rotación terrestre: ¿se
        alarga o se acorta el día?
  \item Por \omterm{thm:b1:newton-dynamics:laws}{acción y reacción}, ¿qué \omterm{def:b1:angular-momentum:moment}{momento} actúa sobre el movimiento
        orbital de la Luna? ¿Crece o disminuye $L_{\text{orb}}$?
  \item A partir de la pregunta 3, ¿crece o disminuye $r$? ¿Y la rapidez
        orbital de la Luna $v = \sqrt{GM/r}$?
  \item Resuélvase la paradoja aparente: se empuja a la Luna hacia
        delante y, sin embargo, acaba moviéndose más despacio.
  \item ¿Hacia qué estado final tiende el intercambio?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Los números.}
\begin{enumerate}[resume]
  \item A partir del alargamiento del día, calcúlese la variación relativa
        $\Delta\Omega/\Omega$ por siglo.
  \item Dedúzcase la variación de $L_{\text{rot}}$ por siglo.
  \item Dedúzcase la variación de $L_{\text{orb}}$ por siglo.
  \item Con la pregunta 3, conviértase en una variación de $r$ por siglo
        y por año. Compárese con el valor de la telemetría láser.
  \item ¿En qué fracción, y en cuántos segundos, se alarga el mes sidéreo
        por siglo ($T \propto r^{3/2}$)?
  \item Calcúlese la variación de la \omterm{thm:b1:work-and-energy:ke}{energía cinética} de rotación de la
        Tierra por siglo ($\Delta E = L_{\text{rot}}\Delta\Omega$ a primer
        orden).
  \item Calcúlese la variación de la \omterm{thm:b1:work-and-energy:em}{energía mecánica} orbital de la Luna,
        $E_{\text{orb}} = -GMm/2r$, por siglo.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La energía.}
\begin{enumerate}[resume]
  \item Muéstrese que la Tierra pierde más energía de la que gana la Luna,
        y calcúlense la diferencia por siglo y la potencia correspondiente.
        ¿Adónde va?
  \item Compárese esa potencia con el consumo de la humanidad, unos
        \qty{18}{TW}, y con el flujo de calor interno de la Tierra, unos
        \qty{47}{TW}.
  \item En el estado final de la pregunta 12, día y mes son iguales:
        $\Omega = \omega = \sqrt{GM/r^3}$. Escribiendo la conservación de
        $L_{\text{rot}} + L_{\text{orb}}$, compruébese que $r \approx
        \qty{5.5e8}{m}$ la satisface y dese el periodo común en días.
  \item ¿Por qué no se alcanzará nunca en realidad ese estado? (Piénsese
        en las mareas solares.)
  \item La Luna ya muestra a la Tierra una sola cara. Explíquese con el
        mismo mecanismo y dígase por qué le ocurrió antes a la Luna.
  \item Resúmase en tres líneas: qué se conserva, qué se intercambia y qué
        se disipa.
\end{enumerate}
\end{problem}
