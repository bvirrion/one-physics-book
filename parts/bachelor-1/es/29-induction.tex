\chapter{La inducción electromagnética y sus aplicaciones}\label{ch:b1:induction}

Acérquese un imán a una bobina y por la bobina pasa una corriente;
párese el imán y la corriente cesa. Faraday lo descubrió en 1831, tras
diez años de búsqueda, y es el hecho sobre el que funciona el mundo
eléctrico: toda central, todo transformador, todo motor, altavoz y
micrófono, toda placa de inducción y todo cargador inalámbrico convierten
energía por la ley de que un \omterm{def:b1:induction:flux}{flujo magnético} \emph{variable} impulsa una
\omterm{def:b1:dc-circuits:sources}{fuerza electromotriz}. Este capítulo enuncia la ley con sus signos, le da
las dos caras que muestra (un campo variable en un circuito fijo, un
circuito que se mueve en un campo fijo), define la \omterm{def:b1:induction:inductance}{autoinducción} y la
\omterm{def:b1:induction:inductance}{inducción mutua}, y recorre después las máquinas electromecánicas --- la
barra deslizante, el alternador, el transformador, el altavoz --- donde
la misma ley convierte potencia mecánica en \omterm{thm:b1:dc-circuits:kirchhoff}{potencia eléctrica} y al
revés, nunca gratis.

\section{La ley de Faraday}

\begin{definition}[Flujo magnético a través de un circuito]\label{def:b1:induction:flux}
\index{flujo magnético}\index{weber}
Oriéntese un circuito cerrado $\Gamma$ (un sentido de recorrido); su
superficie recibe la normal $\vect n$ que da la \omterm{prop:b1:magnetostatics:maps}{regla de la mano derecha}.
El \emph{flujo magnético} a través del circuito es
$\Phi = \sum\vect B\cdot\vect n\,\dd S$ sobre cualquier superficie apoyada
en $\Gamma$ (weber, $\qty{1}{Wb} = \qty{1}{T.m^2}$) --- ``cualquiera'',
porque el flujo de $\vect B$ a través de una superficie cerrada es nulo.
Para un campo uniforme y una bobina plana de $N$ espiras y área $S$, $\Phi =
NBS\cos\alpha$, siendo $\alpha$ el ángulo entre $\vect B$ y $\vect n$.
\end{definition}

\begin{theorem}[Ley de Faraday de la inducción]\label{thm:b1:induction:faraday}
\index{ley de Faraday}\index{fuerza electromotriz (inducida)}\index{ley de Lenz}
Siempre que varía el flujo a través de un circuito, el circuito es sede
de una \emph{\omterm{def:b1:dc-circuits:sources}{fuerza electromotriz} inducida}
\[
e = -\frac{\dd\Phi}{\dd t} ,
\]
contada en la orientación del circuito, es decir, actuando como una
\omterm{def:b1:dc-circuits:sources}{fuente de tensión} ideal de fem $e$ en ese sentido; en un circuito de
resistencia $R$ impulsa la corriente $i = e/R$. El signo menos es la
\emph{\omterm{thm:b1:induction:faraday}{ley de Lenz}}: la corriente inducida circula de modo que se opone a
la variación de flujo que la produce --- su propio campo, y las fuerzas
que siente, combaten la causa.
\end{theorem}
\admitted

\begin{remark}[Dos caras de la misma ley]\label{rem:b1:induction:twofaces}
\index{inducción de Neumann}\index{inducción de Lorentz}
El flujo varía porque $\vect B$ varía en el tiempo a través de un
circuito fijo (\emph{\omterm{rem:b1:induction:twofaces}{inducción de Neumann}}: transformador, placa de
inducción) o porque el circuito se mueve o se deforma en un campo
estacionario (\emph{\omterm{rem:b1:induction:twofaces}{inducción de Lorentz}}: generador, altavoz). En el
segundo caso la fem puede calcularse también directamente: los
portadores de un elemento de conductor $\dd\vect l$ que se mueve a
$\vect v$ sienten la fuerza $q\vect v\wedge
\vect B$, equivalente a un campo eléctrico $\vect v\wedge\vect B$, de modo
que $e = \oint(\vect v\wedge\vect B)\cdot\dd\vect l$ --- y esto coincide
siempre con $-\dd\Phi/\dd t$. La \omterm{thm:b1:induction:faraday}{ley de Faraday} cubre ambos casos; el
volumen del Año 2 explica por qué los dos mecanismos dan una sola fórmula.
\end{remark}

\begin{example}[Órdenes de magnitud]\label{ex:b1:induction:orders}
Una bobina de $100$ espiras y \qty{10}{cm^2} en el campo terrestre, volteada en
\qty{0.1}{s}: $\Delta\Phi = 2 \times 100 \times 10^{-3} \times 5 \times 10^{-5} = \qty{1e-5}{Wb}$,
$e \sim \qty{0.1}{mV}$ --- medible con un galvanómetro sensible, como se
cartografió antaño el campo terrestre. La misma bobina en el entrehierro
de \qty{1}{T} de un imán, sacada en \qty{0.1}{s}: \qty{1}{V}. Una bobina de
transformador de $1000$ espiras con \qty{1}{T} a \qty{50}{Hz} sobre \qty{10}{cm^2}: $e_{\max} =
1000 \times 10^{-3} \times 314 = \qty{314}{V}$ --- la red.
\end{example}

\section{Autoinducción e inducción mutua}

\begin{definition}[Coeficientes de inducción]\label{def:b1:induction:inductance}
\index{autoinducción}\index{henrio}\index{inducción mutua}
El flujo del campo propio de un circuito a través de sí mismo es
proporcional a su corriente: $\Phi_{\text{propio}} = Li$, con $L > 0$ su
\emph{coeficiente de autoinducción} (henrio, \unit{H}). El flujo que el
circuito~1 envía a través del circuito~2 vale
$\Phi_{1 \to 2} = Mi_1$, y simétricamente $\Phi_{2 \to 1} = Mi_2$, con $M$
su \emph{coeficiente de inducción mutua} (el signo lo fijan las
orientaciones). El flujo total del circuito~1 es entonces $\Phi_1 = L_1i_1 + Mi_2$.
\end{definition}

\begin{proposition}[La bobina; el solenoide]\label{prop:b1:induction:inductor}
\index{bobina}\index{energía magnética}\index{densidad de energía del campo magnético}
Un circuito de coeficiente de \omterm{def:b1:induction:inductance}{autoinducción} $L$ recorrido por una
corriente variable es sede de la fem $e = -L\,\dd i/\dd t$: en el \omterm{thm:b1:dc-circuits:kirchhoff}{convenio
receptor}, esta es la tensión $u = L\,\dd i/\dd t$ de la bobina usada desde
\cref{ch:b1:transient-regimes}. Almacena la energía $W = \tfrac12Li^2$.
Un solenoide largo de $n$ espiras por metro, longitud $\ell$ y sección
$S$, tiene $L = \mu_0n^2S\ell$; su energía almacenada vale $\tfrac12\mu_0n^2I^2S\ell =
\dfrac{B^2}{2\mu_0}\,S\ell$: la \omterm{prop:b1:induction:inductor}{energía magnética} vive en el campo con la
densidad $B^2/2\mu_0$, exactamente como la eléctrica con $\varepsilon_0E^2/2$.
\end{proposition}

\begin{proof}
Faraday con $\Phi = Li$ y $L$ constante. Energía: la potencia que recibe
la bobina vale $ui = Li\,\dd i/\dd t = \dd(\tfrac12Li^2)/\dd t$. Solenoide:
$B = \mu_0nI$ dentro, flujo $BS$ por cada una de las $n\ell$ espiras, luego $\Phi =
\mu_0n^2S\ell I$. La densidad general se admite.
\end{proof}

\begin{example}[Un imán de resonancia magnética]\label{ex:b1:induction:mri}
\qty{1.5}{T} sobre alrededor de un metro cúbico: $B^2/2\mu_0 = 2.25/2.5 \times 10^{-6} =
\qty{0.9}{MJ/m^3}$ --- la energía de la batería de un coche, almacenada
en el vacío. Con una corriente de bobina de \qty{500}{A},
$L = 2W/I^2 = \qty{7}{H}$. Si el superconductor se vuelve resistivo de
golpe (un \emph{apagado brusco}), ese megajulio se convierte en calor en
segundos y evapora centenares de litros de helio líquido: la sala del
imán tiene un respiradero para eso.
\end{example}

\begin{proposition}[El transformador ideal]\label{prop:b1:induction:transformer}
\index{transformador}
Dos bobinas de $N_1$ y $N_2$ espiras devanadas sobre el mismo núcleo de
hierro cerrado, de modo que el mismo flujo $\varphi$ atraviesa todas las
espiras, cumplen
\[
\frac{u_2}{u_1} = \frac{N_2}{N_1} , \qquad N_1i_1 + N_2i_2 = 0 \quad\text{(sin pérdidas en vacío, sin corriente magnetizante)},
\]
de modo que $u_2i_2 = -u_1i_1$: la potencia tomada en el primario se
entrega en el secundario; las tensiones cambian con la razón de espiras y
las corrientes en razón inversa. Solo funciona con corrientes variables.
\end{proposition}

\begin{proof}
$u_1 = N_1\,\dd\varphi/\dd t$ y $u_2 = N_2\,\dd\varphi/\dd t$ (\omterm{thm:b1:dc-circuits:kirchhoff}{convenio
receptor} en ambos, con las orientaciones tomadas según el flujo común):
divídase. La relación entre corrientes es el \omterm{thm:b1:magnetostatics:ampere}{teorema de Ampère} en el
núcleo: su circulación, $\mu_0\mu_r$ veces los amperios-vuelta totales,
tendría que crear el flujo; en un núcleo ideal ($\mu_r \to \infty$) eso
exige amperios-vuelta que tienden a cero, luego $N_1i_1 + N_2i_2 = 0$. Se
admite en esta forma.
\end{proof}

\section{Conductores en movimiento: generadores y motores}

\begin{proposition}[La barra deslizante]\label{prop:b1:induction:rails}
\index{raíles de Laplace}\index{conversión electromecánica}
Una barra de longitud $\ell$ desliza a la rapidez $v$ sobre dos raíles
cerrados por una resistencia $R$, en un campo uniforme $B$ normal al
plano. Es sede de la fem $e = B\ell v$, impulsa $i = B\ell v/R$ y siente
la \omterm{thm:b1:magnetostatics:laplace}{fuerza de Laplace} $F = B\ell i = B^2\ell^2v/R$ que se \emph{opone} a su
movimiento. La potencia mecánica que se gasta en empujarla, $Fv$, iguala
a la \omterm{thm:b1:dc-circuits:kirchhoff}{potencia eléctrica} $ei = Ri^2$ disipada: la barra es un generador.
Alimentada en cambio por una fuente $E$, es un motor: la corriente
$(E - B\ell v)/R$ la impulsa hasta que la fem contraria $B\ell v$
equilibra a $E$, a la rapidez en vacío $E/B\ell$.
\end{proposition}

\begin{proof}
Flujo $B\ell x$, $e = -B\ell\dot x$ en la orientación para la que $\vect n$
va según $\vect B$; la \omterm{thm:b1:magnetostatics:laplace}{fuerza de Laplace} $i\vect\ell\wedge\vect B$ sobre la
corriente así orientada apunta en contra de $\vect v$
(\cref{thm:b1:induction:faraday}, Lenz). Multiplíquese la ecuación
eléctrica $e = Ri$ por $i$ y la mecánica por $v$: $Fv = B\ell iv = ei$. La
potencia electromecánica $ei$ y la potencia de Laplace $-Fv$ son siempre
opuestas --- la ley de la conversión.
\end{proof}

\begin{proposition}[La espira que gira: el alternador]\label{prop:b1:induction:alternator}
\index{alternador}
Una bobina de $N$ espiras y área $S$ que gira a $\omega$ alrededor de un
eje perpendicular a un $B$ uniforme tiene $\Phi = NBS\cos\omega t$ y
\[
e = NBS\omega\sin\omega t :
\]
una fem sinusoidal de \omterm{def:b1:wave-propagation:signal}{amplitud} $NBS\omega$ y de la frecuencia de giro.
Sobre una resistencia $R$ entrega la \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} $(NBS\omega)^2/2R$,
suministrada por el par que la mantiene girando contra el par de Laplace
$-\vect m\wedge\vect B$ de su propia corriente.
\end{proposition}

\begin{proof}
Derívese el flujo; $\langle\sin^2\rangle = \tfrac12$. El par es el de
\cref{thm:b1:magnetostatics:dipoleinfield} con $m = Ni S$; su \omterm{thm:b1:sinusoidal-impedance:power}{potencia
media} $\langle\Gamma\omega\rangle$ iguala a $\langle ei\rangle$ por el mismo
cálculo que en la barra.
\end{proof}

\begin{example}[Corrientes de Foucault]\label{ex:b1:induction:eddy}
\index{corrientes de Foucault}
Un conductor macizo que se mueve en un campo no uniforme, o que está en
uno variable, queda atravesado por un flujo variable a lo largo de
innumerables caminos cerrados: por él circulan \emph{\omterm{ex:b1:induction:eddy}{corrientes de
Foucault}} que lo calientan y --- por Lenz --- lo frenan. Un imán que se
deja caer por un tubo de cobre baja despacio; los frenos de camiones y
montañas rusas lo usan (sin contacto, sin desgaste, sin fallo de freno
--- pero sin frenada en reposo); la placa de inducción calienta la cazuela
con esas mismas corrientes, a \qty{25}{kHz}; y los núcleos de los
transformadores se laminan precisamente para cortarlas.
\end{example}

\section{El altavoz: un sistema acoplado}

\begin{proposition}[Acoplamiento electromecánico de una bobina en un campo radial]\label{prop:b1:induction:loudspeaker}
\index{altavoz}\index{micrófono}
Una bobina de longitud total de hilo $\ell$ está en un campo radial $B$
de modo que todo elemento de hilo es perpendicular a $\vect B$; se mueve
según su eje a la rapidez $v$ y lleva $i$. Siente entonces la \omterm{thm:b1:magnetostatics:laplace}{fuerza de
Laplace} axial $F = B\ell i$ y es sede de la fem contraria $e = -B\ell v$
(en \omterm{thm:b1:dc-circuits:kirchhoff}{convenio receptor}, una tensión $B\ell v$). Con una masa $m$, una
suspensión de rigidez $k$ y un amortiguamiento mecánico $h$, alimentada
por una tensión $u$ a través de la resistencia $R$ y la inductancia $L$
de la bobina:
\[
u = Ri + L\frac{\dd i}{\dd t} + B\ell v , \qquad
m\frac{\dd v}{\dd t} = -kx - hv + B\ell i .
\]
Al multiplicar por $i$ y por $v$ y sumar: la \omterm{thm:b1:dc-circuits:kirchhoff}{potencia eléctrica} $ui$
iguala a la pérdida Joule $Ri^2$ más el crecimiento de las energías
almacenadas $\tfrac12Li^2 + \tfrac12mv^2 + \tfrac12kx^2$ más la pérdida
mecánica $hv^2$ (que incluye el sonido): el término de conversión
$B\ell iv$ se cancela entre las dos ecuaciones. Al revés (impuesto el
movimiento y leído $u$), el mismo aparato es un \emph{micrófono} dinámico.
\end{proposition}

\begin{proof}
\omterm{thm:b1:magnetostatics:laplace}{Fuerza de Laplace} sobre cada elemento $i\,\dd\vect l\wedge\vect B$, todas
axiales y del mismo signo; fem a partir de
$(\vect v\wedge\vect B)\cdot\dd\vect l$ sumado a lo largo del hilo. El
balance energético es la suma de las dos ecuaciones multiplicadas como se
ha dicho.
\end{proof}

\begin{remark}[Por qué amortigua el acoplamiento]\label{rem:b1:induction:damping}
A baja frecuencia ($L\omega \ll R$), $i \approx (u - B\ell v)/R$ y la fuerza
vale $B\ell u/R - (B^2\ell^2/R)\,v$: el circuito eléctrico añade el
amortiguamiento $B^2\ell^2/R$ al mecánico. Un altavoz conectado a un
amplificador (de baja resistencia de salida) está muy amortiguado --- su
cono se para cuando cesa la \omterm{def:b1:wave-propagation:signal}{señal} --- y ese mismo acoplamiento es lo que
permite al micrófono tomar energía del sonido (\cref{pb:b1:induction:1}).
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:induction:1}
Una bobina de $100$ espiras y \qty{10}{cm^2} en un campo uniforme de
\qty{0.20}{T} a \qty{30}{\degree} de su normal. Flujo; fem media si el
campo se anula en \qty{10}{ms}; y en \qty{1}{ms}; sentido de la corriente
inducida respecto del campo (Lenz).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:2}
Una barra de \qty{20}{cm} desliza a \qty{2.0}{m/s} sobre raíles cerrados
por \qty{0.10}{\ohm}, en \qty{0.50}{T}. Fem, corriente, \omterm{thm:b1:magnetostatics:laplace}{fuerza de Laplace},
fuerza que ha de aplicar quien empuja, potencia mecánica y \omterm{thm:b1:dc-circuits:kirchhoff}{potencia eléctrica}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:3}
Solenoide de \qty{1000}{espiras/m}, longitud \qty{20}{cm}, sección \qty{4.0}{cm^2}:
coeficiente de \omterm{def:b1:induction:inductance}{autoinducción}; energía a \qty{5.0}{A}; \omterm{def:b1:transient-regimes:firstorder}{constante de tiempo}
con una resistencia de \qty{1.0}{\ohm}; campo interior y densidad de
energía, para comprobar $B^2/2\mu_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:4}
Un transformador de \qty{230}{V} a \qty{12}{V}: razón de espiras; espiras
del primario para un secundario de $50$ espiras; corriente del primario
cuando el secundario entrega \qty{2.0}{A}. ¿Por qué no funciona con la
batería de un coche?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:5}
\omterm{thm:b1:induction:faraday}{Ley de Lenz}: dese el sentido de la corriente inducida y la dirección de
la fuerza en cada caso --- (a) un polo norte se acerca a una espira según
su eje; (b) una espira está en un campo que crece; (c) se deja caer una
espira sobre un imán; (d) un imán cae por un tubo largo de cobre. En (d),
explíquese con la energía por qué cae a una rapidez pequeña y constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:6}
Alternador: $N = 200$, $S = \qty{20}{cm^2}$, $B = \qty{0.10}{T}$, \qty{50}{Hz}.
Fem de pico y eficaz; corriente y par de pico sobre \qty{10}{\ohm};
\omterm{thm:b1:dc-circuits:kirchhoff}{potencia eléctrica} media; compárese con la potencia mecánica media y con
el par instantáneo en el momento en que la fem se anula.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:7}
\emph{Freno magnético.} Una espira cuadrada de lado $a = \qty{10}{cm}$ y
resistencia $R = \qty{1.0}{m\ohm}$, de masa \qty{1.0}{kg}, sale de una
región de campo uniforme $B = \qty{1.0}{T}$ a la rapidez $v$ (con un lado
todavía dentro del campo). Fem, corriente, fuerza de frenado; ecuación
del movimiento y \omterm{def:b1:transient-regimes:firstorder}{constante de tiempo}; distancia recorrida antes de
pararse desde \qty{1.0}{m/s}; ¿adónde fue la \omterm{thm:b1:work-and-energy:ke}{energía cinética}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:8}
Una bobina pequeña ($N_2 = 50$ espiras, sección \qty{2.0}{cm^2}) está
dentro de un solenoide largo ($n_1 = \qty{2000}{m^{-1}}$), con los ejes
alineados. \omterm{def:b1:induction:inductance}{Inducción mutua}; fem en la bobina cuando la corriente del
solenoide crece a razón de \qty{100}{A/s}, y cuando vale $\qty{1.0}{A}$ a
\qty{10}{kHz}. ¿Depende $M$ de cuál de las dos bobinas lleva la corriente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:9}
Energía en un campo: $B = \qty{1.5}{T}$ sobre \qty{1.0}{m^3}, como en un
imán de resonancia magnética. Energía; coeficiente de \omterm{def:b1:induction:inductance}{autoinducción} para
$I = \qty{500}{A}$; altura a la que esa energía elevaría un coche de
\qty{1}{t}; volumen de helio líquido que evapora en un apagado brusco
(\qty{2.6}{kJ} por litro). Energía del campo terrestre
(\qty{5e-5}{T}) en una habitación de \qty{50}{m^3}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:10}
\emph{Los raíles como motor.} La barra de \cref{exo:b1:induction:2} (masa
$m$, sin rozamiento) se conecta a una fuente $E$ a través de la
resistencia $R$. (a) Ecuaciones eléctrica y mecánica; muéstrese que
$m\,\dd v/\dd t = B\ell(E -
B\ell v)/R$. (b) Rapidez límite y \omterm{def:b1:transient-regimes:firstorder}{constante de tiempo}. (c) Balance de
potencias: potencia de la fuente $=$ Joule $+$ mecánica; \omterm{def:b1:heat-engines:efficiency}{rendimiento} en
función de $v$. (d) Cifras: $E = \qty{2.0}{V}$, $m = \qty{50}{g}$, lo
demás como en \cref{exo:b1:induction:2}. (e) Un cañón de raíles acelera
un proyectil de \qty{10}{g} con \qty{1}{MA} a lo largo de \qty{5}{m} en
\qty{1}{T}: rapidez de salida (\omterm{def:b1:units-dimensions:oom}{orden de magnitud}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:11}
\emph{El disco de Faraday.} Un disco de cobre de radio $R$ gira a
$\omega$ alrededor de su eje en un $B$ uniforme según el eje; unas
escobillas tocan el centro y el borde. (a) Fem entre el centro y el borde
sumando $(\vect v\wedge
\vect B)\cdot\dd\vect l$ a lo largo de un radio: $e = \tfrac12B\omega R^2$.
(b) Cifras: $R = \qty{10}{cm}$, \qty{3000}{rpm}, \qty{1.0}{T}. (c) ¿Por
qué es una máquina de poca tensión y mucha corriente?; corriente sobre
\qty{1}{m\ohm} y par de frenado. (d) Recupérese $e$ con la \omterm{thm:b1:induction:faraday}{ley de Faraday}
aplicada al circuito escobilla--radio--borde--escobilla: ¿qué flujo varía?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:12}
\emph{Placa de inducción.} La bobina que hay bajo el vidrio lleva $i_1 =
I_1\cos\omega t$ con $I_1 = \qty{30}{A}$ a \qty{25}{kHz}; el fondo de la
cazuela actúa como una sola espira de resistencia $R_p = \qty{10}{m\ohm}$
acoplada por $M = \qty{1.0}{\micro H}$ (despreciada la \omterm{def:b1:induction:inductance}{autoinducción} de
esa espira). (a) Fem inducida en la cazuela, valor de pico. (b) Corriente
y \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} disipada en la cazuela. (c) ¿Por qué una frecuencia alta
y por qué aparece el calor en la cazuela y no en el vidrio? (d) La
corriente de la cazuela reacciona sobre la bobina: fem que induce allí
(de pico) y potencia adicional que ha de dar el generador --- compruébese
que coincide con (b).
\end{exercise}

\section{Problema: la dinamo de la bicicleta y el altavoz}

\begin{problem}[{Problema de fin de semana --- dos convertidores
electromecánicos desmontados: una dinamo que se regula sola y un altavoz
cuyo amplificador es también su freno}]
\label{pb:b1:induction:1}

\medskip
\noindent\textbf{Parte I --- La dinamo de la bicicleta.}
Una bobina de $N = 300$ espiras y área $S = \qty{3.0}{cm^2}$ gira en un
campo uniforme $B = \qty{0.30}{T}$; su eje lleva un rodillo de radio
$r = \qty{1.0}{cm}$ apoyado en el neumático, de modo que la bobina gira a
$\omega = v/r$ para una rapidez $v$ de la bicicleta. Resistencia de la bobina $r_c = \qty{4.0}{\ohm}$,
coeficiente de \omterm{def:b1:induction:inductance}{autoinducción} $L = \qty{20}{mH}$; la lámpara es una resistencia $R = \qty{12}{\ohm}$.
\begin{enumerate}
  \item Flujo a través de la bobina y fem inducida en función del tiempo;
        \omterm{def:b1:wave-propagation:signal}{amplitud} y frecuencia a $v = \qty{18}{km/h}$.
  \item Despreciando primero $L$: corriente eficaz y potencia en la
        lámpara a \qty{18}{km/h}; ¿está cómoda una lámpara de
        ``\qty{6}{V}, \qty{3}{W}''?
  \item Potencia mecánica media que el ciclista entrega a la dinamo, y
        fuerza correspondiente sobre el neumático (compárese con el
        rozamiento de rodadura, de unos \qty{2}{N}).
  \item Explíquese con la \omterm{thm:b1:induction:faraday}{ley de Lenz} por qué se resiste la dinamo, y por
        qué una lámpara que luce ``gratis'' no contradice la conservación
        de la energía.
  \item Inclúyase ahora $L$: \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} compleja del circuito; muéstrese
        que la \omterm{def:b1:wave-propagation:signal}{amplitud} de la corriente vale $I = NBS\omega/\sqrt{(R + r_c)^2 + L^2\omega^2}$
        y que se satura a gran rapidez; dese su límite y la rapidez a la
        que alcanza el $70\%$ del límite.
  \item Sin la inductancia, ¿qué recibiría la lámpara a \qty{54}{km/h}
        cuesta abajo? ¿Y con ella?
  \item Corriente y potencia de la lámpara a \qty{9}{km/h} y a \qty{36}{km/h};
        coméntese la autorregulación.
  \item Las dinamos reales hacen girar un imán dentro de una bobina fija.
        ¿Cambia el análisis? ¿Qué problema práctico elimina?
  \item ¿Qué hace la lámpara cuando la bicicleta se para en un semáforo, y
        cómo lo resuelven las luces modernas?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El altavoz.}
Bobina móvil: longitud de hilo $\ell = \qty{5.0}{m}$ en un campo radial $B =
\qty{1.0}{T}$, resistencia $R = \qty{8.0}{\ohm}$, inductancia $L = \qty{0.50}{mH}$;
masa móvil (bobina y cono) $m = \qty{10}{g}$; rigidez de la suspensión
$k = \qty{1000}{N/m}$, amortiguamiento mecánico $h = \qty{1.0}{N.s/m}$.
Régimen sinusoidal, \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas}.
\begin{enumerate}[resume]
  \item Fuerza sobre la bobina para una corriente $i$; fem contraria para
        una rapidez $v$ (dedúzcanse ambas, con sus signos).
  \item Escríbanse las ecuaciones eléctrica y mecánica.
  \item Elimínese $i$ y muéstrese que
        $\underline v = \dfrac{B\ell\,\underline u}{(R + jL\omega)(jm\omega + h + k/j\omega) + B^2\ell^2}$.
  \item En las frecuencias en que $L\omega \ll R$, muéstrese que el
        acoplamiento actúa como un amortiguamiento mecánico adicional
        $B^2\ell^2/R$; su valor; frecuencia de resonancia del cono y su
        \omterm{def:b1:transient-regimes:secondorder}{factor de calidad} con y sin el amortiguamiento eléctrico.
  \item El amplificador entrega una $i$ de \omterm{def:b1:wave-propagation:signal}{amplitud} \qty{1.0}{A} a \qty{1}{kHz}:
        \omterm{def:b1:wave-propagation:signal}{amplitudes} de la fuerza, la aceleración, el desplazamiento y la
        velocidad del cono (régimen dominado por la masa); \omterm{def:b1:wave-propagation:signal}{amplitud} de la
        fem contraria comparada con $Ri$.
  \item La misma corriente a la frecuencia de resonancia: \omterm{def:b1:wave-propagation:signal}{amplitud} de la
        velocidad (dominada por el amortiguamiento) y fem contraria;
        coméntese.
  \item \omterm{thm:b1:dc-circuits:kirchhoff}{Potencia eléctrica} a \qty{1}{kHz}; potencia mecánica disipada en
        $h$ (tomada como el sonido y las pérdidas de la suspensión);
        \omterm{def:b1:heat-engines:efficiency}{rendimiento}.
  \item ¿Por qué es la presión sonora proporcional a la aceleración del
        cono (admítase) la razón de que un cono dominado por la masa dé una
        respuesta plana por encima de la resonancia?
  \item ¿Por qué ha de ser radial el campo y por qué es pequeño un tuíter?
  \item Esbócese el módulo de la \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} $\underline u/\underline i$ que ve
        el amplificador de \qty{10}{Hz} a \qty{10}{kHz}: valor a baja
        frecuencia, en la resonancia (úsese la velocidad dominada por el
        amortiguamiento) y a \qty{10}{kHz}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Al revés, y el balance.}
\begin{enumerate}[resume]
  \item El mismo aparato como micrófono: un sonido mueve el cono a
        $v = \qty{1.0}{mm/s}$; tensión en circuito abierto.
  \item Cargado con $R_L = \qty{600}{\ohm}$: corriente y fuerza de frenado
        sobre el cono; muéstrese que el micrófono toma potencia del sonido
        a través de un amortiguamiento efectivo $B^2\ell^2/(R + R_L)$.
  \item La bobina está devanada sobre un cilindro de aluminio (el
        soporte), una espira cerrada de longitud \qty{0.10}{m} y
        resistencia \qty{1.0}{m\ohm}: amortiguamiento que añade; por qué se
        ranuran los soportes.
  \item Golpéese el cono de un altavoz con los bornes al aire y después el
        de otro con los bornes en cortocircuito: ¿cuál resuena más tiempo
        y por qué?
  \item Escríbase el balance energético del altavoz a lo largo de un
        periodo en \omterm{def:b1:transient-regimes:firstorder}{régimen permanente}, nombrando cada término.
  \item Resúmase: para cada aparato, la ley empleada y las dos cifras que
        vale la pena recordar.
\end{enumerate}
\end{problem}
