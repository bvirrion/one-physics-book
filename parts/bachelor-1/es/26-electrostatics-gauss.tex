\chapter{Electrostática: el campo y el teorema de Gauss}\label{ch:b1:electrostatics-gauss}

Frótese un globo con un jersey y se pega a la pared; péinese el pelo seco
y el peine desvía un hilo fino de agua. Detrás de ambas cosas está la
fuerza entre \omterm{def:b1:electrostatics-gauss:charge}{cargas eléctricas} en reposo, la \omterm{thm:b1:electrostatics-gauss:coulomb}{ley de Coulomb} --- una ley
en el inverso del cuadrado como la \omterm{def:b1:central-forces:gravitation}{gravitación}, pero $10^{36}$ veces más
intensa entre dos protones y de los dos signos, de modo que la materia es
neutra con una precisión fantástica y los efectos residuales son lo que
llamamos química, materiales y rayos. Este capítulo construye el concepto
que hace manejable la ley, el \omterm{def:b1:electrostatics-gauss:field}{campo eléctrico}, muestra cómo leer un campo
a partir de sus simetrías y deduce el único teorema que calcula en tres
líneas los campos de las distribuciones simétricas: el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss}.
Su gemelo gravitatorio sale gratis al final.

\section{Las cargas y la ley de Coulomb}

\begin{definition}[Carga eléctrica]\label{def:b1:electrostatics-gauss:charge}
\index{carga eléctrica}\index{carga elemental}\index{densidad de carga}
La \emph{carga eléctrica} $q$ (culombio, \unit{C}) es una propiedad de la
materia que puede tener uno u otro signo; se conserva (la carga total de
un sistema aislado no cambia nunca) y está cuantizada en unidades de la
\emph{carga elemental} $e = \qty{1.602e-19}{C}$ (protón $+e$, electrón
$-e$). Una distribución macroscópica se describe con una \emph{densidad
de carga}: lineal $\lambda$ (\unit{C/m}), superficial $\sigma$
(\unit{C/m^2}) o volumétrica $\rho$ (\unit{C/m^3}), de modo que un
pequeño elemento lleva $\dd q = \lambda\,\dd l$, $\sigma\,\dd S$ o $\rho\,\dd\tau$.
\end{definition}

\begin{theorem}[Ley de Coulomb]\label{thm:b1:electrostatics-gauss:coulomb}
\index{ley de Coulomb}\index{permitividad del vacío}
Dos cargas puntuales $q_1$ en $M_1$ y $q_2$ en $M_2$ en reposo en el vacío
ejercen una sobre otra las fuerzas
\[
\vect F_{1 \to 2} = \frac{q_1q_2}{4\pi\varepsilon_0\,r^2}\,\vect e_{12} = -\vect F_{2 \to 1},
\qquad r = M_1M_2,\quad \vect e_{12} = \frac{\vect{M_1M_2}}{r},
\]
repulsivas para cargas del mismo signo y atractivas para cargas de signos
opuestos, con la \emph{\omterm{thm:b1:electrostatics-gauss:coulomb}{permitividad del vacío}} $\varepsilon_0 = \qty{8.854e-12}{F/m}$, es decir
$1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$. Las fuerzas de varias cargas se suman
(superposición).
\end{theorem}
\admitted

\begin{example}[Cuán intensa]\label{ex:b1:electrostatics-gauss:strong}
Dos protones separados \qty{1}{fm} se repelen con $8.99 \times 10^9 \times (1.6 \times
10^{-19})^2/10^{-30} = \qty{230}{N}$ --- el peso de un niño, entre dos
partículas de $10^{-27}$~kg; su atracción gravitatoria es
$10^{36}$ veces menor. Dos gramos de electrones en un extremo de una
habitación y los protones correspondientes en el otro se atraerían con $10^{25}$~N.
La materia es neutra porque no le queda otra: todo desequilibrio lo
corrigen fuerzas a las que ninguna estructura puede resistir.
\end{example}

\section{El campo eléctrico}

\begin{definition}[Campo electrostático]\label{def:b1:electrostatics-gauss:field}
\index{campo eléctrico}\index{línea de campo}
Una distribución de cargas en reposo crea en cada punto $M$ del espacio
un \emph{campo eléctrico} $\vect E(M)$, definido por la fuerza que ejerce
sobre una carga de prueba $q$ colocada en $M$: $\vect F = q\vect E(M)$ (\unit{V/m} $=$
\unit{N/C}). Una carga puntual $q_1$ situada en $O$ crea
\[
\vect E(M) = \frac{q_1}{4\pi\varepsilon_0 r^2}\,\vect e_r , \qquad r = OM ,
\]
radial, saliente si $q_1 > 0$; una distribución crea la suma vectorial de
los campos de sus elementos. Una \emph{línea de campo} es una curva
tangente en cada punto a $\vect E$ y orientada según él.
\end{definition}

\begin{proposition}[Cómo leer un mapa de campo]\label{prop:b1:electrostatics-gauss:lines}
Las \omterm{def:b1:electrostatics-gauss:field}{líneas de campo} salen de las cargas positivas y acaban en las
negativas (o en el infinito); nunca se cortan (el campo es único en cada
punto) y se aprietan donde el campo es intenso. Cuando hay dos cargas
presentes, el número de líneas dibujadas en cada una es proporcional a su carga.
\end{proposition}

\begin{example}[Anillo y segmento por suma directa]\label{ex:b1:electrostatics-gauss:ring}
En el eje de un anillo de radio $R$ que lleva $Q$ uniformemente, a la
distancia $z$ de su centro, cada elemento $\dd q$ aporta $\dd q/4\pi\varepsilon_0(R^2 +
z^2)$ a lo largo de la recta que lo une con $M$; las componentes
perpendiculares al eje se cancelan de dos en dos y las axiales se suman
con el factor $z/\sqrt{R^2 + z^2}$:
\[
E_z = \frac{Qz}{4\pi\varepsilon_0(R^2 + z^2)^{3/2}} ,
\]
nulo en el centro, máximo en $z = R/\sqrt2$ y $Q/4\pi\varepsilon_0z^2$ muy lejos.
Un disco cargado uniformemente es un montón de anillos (\cref{pb:b1:electrostatics-gauss:1}).
La suma directa funciona siempre que la geometría permita hacer la
integral; el resto del capítulo trata de evitarla.
\end{example}

\section{Simetrías e invariancias}

\begin{proposition}[Principio de simetría]\label{prop:b1:electrostatics-gauss:symmetry}
\index{simetría (de un campo)}\index{plano de simetría}\index{plano de antisimetría}
El campo tiene las simetrías de sus fuentes. Si un plano $\Pi$ es
\emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{plano de simetría}} de la distribución de cargas (la imagen
especular de la distribución es ella misma), entonces en todo punto de
$\Pi$ el campo está \emph{contenido} en $\Pi$; si $\Pi$ es un \emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{plano
de antisimetría}} (la imagen especular es la distribución con todas las
cargas cambiadas de signo), el campo en los puntos de $\Pi$ es
\emph{perpendicular} a $\Pi$. Si la distribución es invariante en una
traslación o en una rotación, también lo es el campo: sus componentes no
dependen de la coordenada correspondiente.
\end{proposition}

\begin{proof}
$\vect E$ es una suma de términos $\dd q\,\vect e/4\pi\varepsilon_0r^2$; la imagen
especular de la suma es la suma de las imágenes, y la imagen del campo de
una distribución es el campo de la distribución imagen (un vector
construido a partir de posiciones y cargas se transforma como las
posiciones). En un \omterm{prop:b1:electrostatics-gauss:symmetry}{plano de simetría} el campo es igual a su propia imagen
especular, luego está contenido en el plano; en un \omterm{prop:b1:electrostatics-gauss:symmetry}{plano de antisimetría}
es igual al opuesto de su imagen, luego es normal a él. La invariancia es
el mismo argumento con una traslación o una rotación.
\end{proof}

\begin{method}[Hallar la forma de un campo antes de calcularlo]\label{meth:b1:electrostatics-gauss:symmetry}
Para el punto $M$ donde se quiere el campo: (1) enumérense los \omterm{prop:b1:electrostatics-gauss:symmetry}{planos de
simetría} de las fuentes que contienen a $M$ --- el campo está en su
intersección; dos de esos planos fijan la dirección. (2) Enumérense las
invariancias --- dicen de qué coordenadas depende $\vect E$. Así, un hilo
infinito cargado uniformemente da $\vect E = E(r)\,\vect e_r$ en
\omterm{def:b1:point-kinematics:cylindrical}{coordenadas cilíndricas}, una esfera $\vect E = E(r)\,\vect e_r$ en
esféricas, y un plano infinito $\vect E = E(z)\,\vect e_z$ con
$E(-z) = -E(z)$. Solo entonces se calcula.
\end{method}

\section{El flujo y el teorema de Gauss}

\begin{definition}[Flujo del campo]\label{def:b1:electrostatics-gauss:flux}
\index{flujo del campo eléctrico}
Córtese una superficie $S$ en pequeños elementos de área $\dd S$, cada uno
con una normal unitaria $\vect n$ (para una superficie cerrada, la normal
saliente). El \emph{flujo} de $\vect E$ a través de $S$ es la suma
\[
\Phi = \sum \vect E\cdot\vect n\,\dd S = \sum E_n\,\dd S \qquad (\unit{V.m}):
\]
la componente normal del campo ponderada por el área --- el ``número de
\omterm{def:b1:electrostatics-gauss:field}{líneas de campo}'' que atraviesan $S$, contadas positivamente cuando lo
hacen en el sentido de $\vect n$.
\end{definition}

\begin{theorem}[Teorema de Gauss]\label{thm:b1:electrostatics-gauss:gauss}
\index{teorema de Gauss}
El \omterm{def:b1:electrostatics-gauss:flux}{flujo} del campo electrostático a través de cualquier superficie cerrada
$S$ vale la carga total encerrada por $S$ dividida por $\varepsilon_0$:
\[
\Phi_{\text{sal}} = \frac{Q_{\text{int}}}{\varepsilon_0} .
\]
Las cargas exteriores a $S$ no aportan nada al \omterm{def:b1:electrostatics-gauss:flux}{flujo} (sus líneas entran y
vuelven a salir).
\end{theorem}

\begin{proof}
Para una carga puntual $q$ en el centro de una esfera de radio $r$:
$\vect E$ es radial y de módulo $q/4\pi\varepsilon_0r^2$ en toda la esfera, luego
$\Phi = E \times 4\pi r^2 = q/\varepsilon_0$ --- independiente de $r$, porque el
campo decrece como $1/r^2$ exactamente igual que crece el área. Que lo
mismo valga para cualquier superficie cerrada que rodee a $q$ (el \omterm{def:b1:electrostatics-gauss:flux}{flujo} a
través de un elemento de superficie solo depende del ángulo sólido que
subtiende desde $q$), que una carga exterior dé cero (subtiende dos veces
cada ángulo sólido, con signos opuestos) y que los \omterm{def:b1:electrostatics-gauss:flux}{flujos} de varias cargas
se sumen, se admite aquí; la demostración completa está en el volumen del Año 2.
\end{proof}

\begin{method}[Cómo usar el teorema de Gauss]\label{meth:b1:electrostatics-gauss:gauss}
(1) A partir de las simetrías, escríbase la forma de $\vect E$ (dirección
y variable de la que depende). (2) Elíjase una \emph{superficie de Gauss}
cerrada que pase por el punto $M$ y sobre la cual $\vect E$ sea en todas
partes o bien normal y del mismo módulo $E(M)$, o bien tangente: el \omterm{def:b1:electrostatics-gauss:flux}{flujo}
es entonces $E(M)$ por el área de la parte normal. (3) Cuéntese la carga
interior. (4) Despéjese $E(M)$. Funciona con esferas, cilindros infinitos
y planos infinitos --- y con nada más sin herramientas adicionales.
\end{method}

\begin{proposition}[Los tres campos clásicos]\label{prop:b1:electrostatics-gauss:three}
\index{campo de una esfera cargada}\index{campo de un hilo cargado}\index{campo de un plano cargado}
\begin{enumerate}
  \item Esfera de radio $R$ con la carga $Q$ repartida uniformemente en su
        volumen: $E(r) = \dfrac{Q}{4\pi\varepsilon_0r^2}$ fuera ($r > R$) ---
        como si toda la carga estuviera en el centro --- y $E(r) =
        \dfrac{Qr}{4\pi\varepsilon_0R^3}$ dentro, creciendo linealmente desde cero;
        continuo en $r = R$. Si la carga está solo en la superficie, el
        campo interior es nulo.
  \item Hilo infinito de carga lineal $\lambda$: $E(r) =
        \dfrac{\lambda}{2\pi\varepsilon_0 r}$, radial.
  \item Plano infinito de carga superficial $\sigma$: $E = \dfrac{\sigma}{2\varepsilon_0}$
        a cada lado, normal al plano y saliendo de él si
        $\sigma > 0$, \emph{independiente de la distancia}. Al atravesar el
        plano, la componente normal salta en $\sigma/\varepsilon_0$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Esfera de radio $r$: \omterm{def:b1:electrostatics-gauss:flux}{flujo} $4\pi r^2E(r)$; carga interior $Q$ si
$r > R$ y $Q(r/R)^3$ si $r < R$. (2) Cilindro de radio $r$ y longitud $h$:
\omterm{def:b1:electrostatics-gauss:flux}{flujo} $2\pi rh\,E(r)$ por la superficie lateral y ninguno por las bases
(campo tangente); carga $\lambda h$. (3) Cilindro achatado de sección $S$
que atraviesa el plano: \omterm{def:b1:electrostatics-gauss:flux}{flujo} $2SE$ por las dos bases (el campo es impar
en $z$, de modo que ambas cuentan positivamente) y ninguno por el
costado; carga $\sigma S$.
\end{proof}

\begin{example}[Dos planos: el condensador plano]\label{ex:b1:electrostatics-gauss:twoplanes}
\index{condensador plano!campo}
Dos planos paralelos con $+\sigma$ y $-\sigma$: cada uno crea
$\sigma/2\varepsilon_0$ saliendo de él (entrando en el negativo). Entre los
planos los dos campos se suman y dan $\sigma/\varepsilon_0$, dirigido del $+$ al
$-$; fuera se cancelan. Con $\sigma = \qty{1}{\micro C/m^2}$, $E =
10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$ entre ellos y cero fuera: el
campo de un condensador cargado está confinado en su entrehierro
(\cref{ch:b1:potential-capacitors}).
\end{example}

\section{La analogía gravitatoria}

\begin{proposition}[Teorema de Gauss para la gravitación]\label{prop:b1:electrostatics-gauss:gravity}
\index{teorema de Gauss!para la gravitación}\index{campo gravitatorio}
La ley de Newton $\vect F = -Gm_1m_2\,\vect e_{12}/r^2$ tiene la forma de
la de Coulomb con $q \to m$ y $1/4\pi\varepsilon_0 \to -G$. El \omterm{prop:b1:newton-dynamics:weight}{campo
gravitatorio} $\vect g$ (fuerza por unidad de masa) obedece por tanto a las
mismas reglas de simetría y a
\[
\Phi_{\text{sal}}(\vect g) = -4\pi G\,M_{\text{int}} .
\]
En particular, un cuerpo de simetría esférica atrae a los puntos
exteriores como si su masa estuviera en su centro, y el campo dentro de
una esfera uniforme de radio $R$ y masa $M$ vale $g(r) = GMr/R^3$, lineal en $r$.
\end{proposition}

\begin{proof}
Sustitúyase $Q/\varepsilon_0$ por $-4\pi GM$ en la demostración de
\cref{prop:b1:electrostatics-gauss:three}; el signo menos dice que el
\omterm{def:b1:electrostatics-gauss:flux}{flujo} entra (la gravedad atrae).
\end{proof}

\begin{example}[Dentro de la Tierra]\label{ex:b1:electrostatics-gauss:earth}
Para una Tierra uniforme, $g$ caería linealmente desde \qty{9.8}{m/s^2} en
la superficie hasta cero en el centro; una piedra soltada por un túnel
que atravesara el planeta oscilaría armónicamente con periodo $2\pi\sqrt{R/g} =
\qty{84}{min}$ --- el periodo de un satélite rasante. La Tierra real
tiene un núcleo denso, y $g$ en realidad \emph{sube} hasta
\qty{10.7}{m/s^2} en la frontera núcleo--manto antes de caer (\cref{pb:b1:electrostatics-gauss:1}).
Newton necesitó el teorema de las capas para aplicar su ley a los
planetas; el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss} lo da en una línea.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:1}
Fuerza eléctrica y fuerza gravitatoria entre dos protones separados
\qty{2.0}{fm}; su cociente. ¿Por qué se mantienen unidos entonces los núcleos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:2}
Campo a \qty{1.0}{m} de una carga puntual de \qty{1.0}{\micro C}. La
superficie de la Tierra soporta un campo de \qty{100}{V/m} dirigido hacia
abajo: signo y valor de la carga de la Tierra, tratada como una esfera de
radio \qty{6370}{km}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:3}
Dos cargas $+q$ en $x = \pm a$ sobre el eje $x$. Campo sobre el eje $x$
para $|x| > a$ y sobre el eje $y$; muéstrese que sobre el eje $y$ el $E_y$
es máximo en $y = a/\sqrt2$. Esbócense las líneas cerca del origen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:4}
Cargas $+2q$ en $x = 0$ y $-q$ en $x = a$. ¿Dónde se anula el campo sobre
el eje $x$? ¿Cuántas \omterm{def:b1:electrostatics-gauss:field}{líneas de campo} salen de $+2q$ por cada una que llega
a $-q$, y adónde van las demás?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:5}
Un disco de radio $R$ lleva $\sigma$ uniformemente. A partir del resultado
del anillo, muéstrese que en su eje $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 +
R^2}}\Bigr)$ para $z > 0$; compruébense los dos límites $z \ll R$ y $z \gg R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:6}
Un núcleo de oro ($Z = 79$, $R = \qty{7.0}{fm}$) como esfera cargada
uniformemente: campo en la superficie, en $R/2$ y a \qty{1}{pm}.
Compárese con el campo de ruptura del aire, \qty{3e6}{V/m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:7}
Campo de un hilo infinito por el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss}. Un conductor de alta
tensión de radio \qty{1.0}{cm}: carga lineal a la que el campo en su
superficie alcanza el valor de ruptura \qty{3e6}{V/m} (inicio del efecto
corona). Campo a \qty{5}{m} entonces.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:8}
Dos placas paralelas de \qty{1.0}{m^2} llevan $\pm\qty{1.0}{\micro C}$.
Campo entre ellas; fuerza sobre una placa (el campo que actúa sobre ella
es solo el de la otra placa: ¿por qué?); presión sobre las placas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:9}
Cable coaxial: conductor interior de radio $a$ con $+\lambda$ y malla
exterior de radio $b$ con $-\lambda$. Campo en las tres regiones por el
\omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss}. $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, $\lambda = \qty{10}{nC/m}$:
campo en la superficie del conductor interior.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:10}
Una corona esférica gruesa $R_1 < r < R_2$ lleva $Q$ uniformemente en su
volumen. Campo en las tres regiones; compruébese la continuidad en $R_1$ y
$R_2$; ¿dónde es máximo? Esbócese $E(r)$. Una carga positiva colocada en
la cavidad: ¿qué fuerza siente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:11}
\emph{La gravedad dentro de la Tierra.} Núcleo: radio \qty{3480}{km},
densidad \qty{11000}{kg/m^3}; manto hasta \qty{6370}{km}, densidad \qty{4500}{kg/m^3}.
(a) Masa del núcleo, del manto y total (compárese con \qty{5.97e24}{kg}).
(b) $g$ en la superficie y en la frontera núcleo--manto. (c) Muéstrese que
$g$ crece con la profundidad en el manto si la densidad de este es menor
que $\tfrac23$ de la densidad media de lo que hay debajo, y compruébese.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:12}
\emph{El átomo de Thomson.} Modélese el átomo de hidrógeno como una
esfera positiva uniforme de carga $+e$ y radio $R = \qty{0.10}{nm}$ con el
electrón libre en su interior. (a) Fuerza sobre el electrón a la
distancia $r$ del centro. (b) Muéstrese que oscila armónicamente y
calcúlense la frecuencia y la \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} de la luz que emitiría.
(c) Compárese con las rayas ultravioletas del hidrógeno (\qty{122}{nm} y
menores) y coméntese en qué acierta y en qué falla el modelo (\cref{ch:b1:quantum-introduction}).
\end{exercise}

\section{Problema: la Tierra eléctrica}

\begin{problem}[{Problema de fin de semana --- el campo de buen tiempo, la
nube de tormenta, la descarga del rayo y el teorema de Gauss aplicado a la gravitación}]
\label{pb:b1:electrostatics-gauss:1}
Datos: $\varepsilon_0 = \qty{8.85e-12}{F/m}$, $1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$,
$e = \qty{1.6e-19}{C}$, radio de la Tierra $R_E = \qty{6.37e6}{m}$, superficie
$\qty{5.1e14}{m^2}$, $G = \qty{6.67e-11}{SI}$, $M_E = \qty{5.97e24}{kg}$.

\medskip
\noindent\textbf{Parte I --- El campo de buen tiempo.}
Con tiempo despejado el campo en el suelo vale $E_0 = \qty{100}{V/m}$ y
apunta hacia \emph{abajo}.
\begin{enumerate}
  \item Sentido de la fuerza sobre un electrón libre en el suelo; signo de
        la carga de la Tierra.
  \item Trátese la Tierra como una esfera cargada uniformemente: dese, por
        el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss}, su carga total.
  \item Densidad superficial de carga del suelo; número de electrones en
        exceso por centímetro cuadrado.
  \item Fuerza sobre un grano de polvo de \qty{1}{\micro g} que lleva $100$
        \omterm{def:b1:electrostatics-gauss:charge}{cargas elementales}; compárese con su peso.
  \item Las medidas muestran que el campo baja a unos \qty{5}{V/m} a
        \qty{10}{km}. Aplíquese el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss} a una columna vertical
        de sección \qty{1}{m^2} entre el suelo y \qty{10}{km}: signo y
        cantidad de la carga que lleva el aire por metro cuadrado.
  \item Densidad volumétrica media de carga de ese aire y número
        correspondiente de \omterm{def:b1:electrostatics-gauss:charge}{cargas elementales} por metro cúbico. (El aire
        contiene unos $10^9$ iones de cada signo por metro cúbico: ¿qué
        fracción de desequilibrio es esta?)
  \item El aire tiene una pequeña conductividad $\gamma = \qty{2e-14}{S/m}$,
        de modo que baja una densidad de corriente $\gamma E_0$. Corriente
        total de buen tiempo en todo el globo; tiempo que tardaría en
        neutralizar la carga de la Tierra. Conclúyase.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La nube de tormenta.}
Modélese una nube de tormenta como dos discos horizontales de radio
$R = \qty{3}{km}$: $-Q$ a \qty{3}{km} de altitud y $+Q$ a \qty{9}{km}, con $Q = \qty{40}{C}$.
\begin{enumerate}[resume]
  \item Campo en el eje de un anillo de radio $a$ que lleva $q$, a la
        distancia $z$ de su centro (simetría y después suma).
  \item Dedúzcase, sumando anillos, el campo en el eje de un disco de
        radio $R$ y carga superficial $\sigma$:
        $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$.
  \item Límites $z \ll R$ y $z \gg R$; interprétese cada uno.
  \item Carga superficial del disco inferior; campo que crea en el suelo
        justo debajo (dirección y valor).
  \item Campo del disco superior en el mismo punto; campo total en el
        suelo bajo la tormenta. El suelo es un conductor y sus cargas
        inducidas duplican aproximadamente el resultado (admitido):
        compárese con los \qty{10}{kV/m} medidos bajo las tormentas y con
        el valor de buen tiempo.
  \item \omterm{def:b1:electrostatics-gauss:flux}{Flujo} del campo a través de una superficie cerrada que encierra
        toda la nube; y a través de otra que encierra solo el disco inferior.
  \item Campo a media distancia entre los dos discos (\qty{6}{km}) y su
        dirección; compárese con el campo de ruptura del aire a esa
        altitud, de unos \qty{1.5e6}{V/m}.
  \item Si los \qty{40}{C} estuvieran concentrados en una esfera de radio
        \qty{500}{m}: campo en su superficie. ¿Por qué sigue siendo una
        cuestión abierta el inicio del rayo?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La descarga.}
Una descarga de retorno lleva \qty{5}{C} al suelo en \qty{50}{\micro s} por
un canal de \qty{3}{km} de largo; tómese el campo a lo largo del canal
igual a $10^5$~V/m.
\begin{enumerate}[resume]
  \item Corriente media.
  \item Trabajo de la fuerza eléctrica sobre la carga transferida;
        potencia durante la descarga; compárese con el consumo medio de
        potencia de la humanidad, de unos \qty{2e13}{W}.
  \item El canal tiene un radio de \qty{2}{cm}: masa de aire que contiene
        ($\rho = \qty{1.2}{kg/m^3}$) y temperatura que alcanzaría si toda
        la energía lo calentara a volumen constante ($c_V =
        \qty{720}{J/(kg.K)}$). Las temperaturas reales del canal rondan los
        \qty{30000}{K}: ¿adónde va el resto?
  \item El trueno: retraso por kilómetro y razón de que el sonido de un
        canal de \qty{3}{km} retumbe durante segundos.
  \item En todo el mundo se producen unas $50$ descargas de nube a tierra
        por segundo: corriente que llevan hacia abajo; compárese con la
        Parte~I y complétese la imagen del ``circuito global''.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss} aplicado a la \omterm{def:b1:central-forces:gravitation}{gravitación}.}
\begin{enumerate}[resume]
  \item Escríbase el \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss} para el \omterm{prop:b1:newton-dynamics:weight}{campo gravitatorio} $\vect g$
        y justifíquese la constante con el caso de una masa puntual.
  \item Tierra uniforme: $g(r)$ en el interior; valor en $r = R_E/2$.
  \item Tierra real, con un núcleo de radio \qty{3480}{km} y densidad
        \qty{11000}{kg/m^3}: $g$ en la frontera núcleo--manto; compárese con
        la superficie.
  \item Una piedra soltada por un túnel sin rozamiento que atraviesa una
        Tierra uniforme: ecuación del movimiento, periodo de la oscilación
        y rapidez en el centro.
  \item Resúmase: en cada una de las cuatro partes, ¿qué sustituyó el
        \omterm{thm:b1:electrostatics-gauss:gauss}{teorema de Gauss} y qué cuatro números vale la pena recordar?
\end{enumerate}
\end{problem}
