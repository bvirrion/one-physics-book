\chapter{El primer principio de la termodinámica}\label{ch:b1:first-law}

Hínchese un neumático de bicicleta y el cuerpo de la bomba se pone
demasiado caliente para sujetarlo. Un motor diésel no tiene bujía:
comprime su aire veinte veces y el aire se calienta lo bastante para
encender el combustible por sí solo. Una botella de gas comprimido se
escarcha mientras se vacía. En todos los casos la energía cambia de
forma --- el trabajo se vuelve \omterm{def:b1:first-law:heat}{calor}, el \omterm{def:b1:first-law:heat}{calor} trabajo, ambos \omterm{def:b1:kinetic-theory:U}{energía
interna} --- y el \omterm{thm:b1:first-law:firstlaw}{primer principio de la termodinámica} es la contabilidad
que cuadra las cuentas. Este capítulo define el trabajo y el \omterm{def:b1:first-law:heat}{calor} de un
\omterm{def:b1:first-law:transformation}{sistema termodinámico}, enuncia el primer principio, presenta la \omterm{def:b1:kinetic-theory:U}{energía
interna} y la \omterm{def:b1:first-law:enthalpy}{entalpía} como las magnitudes que conserva, y lo aplica a las
transformaciones del \omterm{def:b1:kinetic-theory:model}{gas perfecto} y a la calorimetría.

\section{Transformaciones, trabajo y calor}

\begin{definition}[Sistema, transformación]\label{def:b1:first-law:transformation}
\index{sistema termodinámico}\index{transformación}\index{transformación cuasiestática}\index{transformación reversible}
Un \emph{sistema termodinámico} es la materia encerrada dentro de una
frontera elegida; todo lo demás es el \emph{entorno}. Una
\emph{transformación} lo lleva de un estado de equilibrio a otro. Es
\emph{cuasiestática} si el sistema pasa por una sucesión de estados de
equilibrio (bastante lenta para que $P$ y $T$ estén definidas en todo
él), y \emph{reversible} si además invertir las condiciones exteriores
invierte el camino (sin rozamiento, sin salto finito de temperatura o de
presión con el entorno). Se nombran por lo que permanece constante:
\emph{isoterma} ($T$), \emph{isobara} ($P$), \emph{isócora} ($V$);
\emph{adiabática} si no hay intercambio de \omterm{def:b1:first-law:heat}{calor}; \emph{monobara}
(\emph{monoterma}) si la presión (la temperatura) exterior es constante,
pase lo que pase dentro.
\end{definition}

\begin{proposition}[Trabajo de las fuerzas de presión]\label{prop:b1:first-law:work}
\index{trabajo!de las fuerzas de presión}
Un sistema cuyo volumen varía en $\dd V$ contra una presión exterior
$P_{\mathrm{ext}}$ recibe el trabajo
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
En una \omterm{def:b1:first-law:transformation}{transformación cuasiestática} $P_{\mathrm{ext}} = P$ (la presión del
propio sistema) y $W = -\int P\,\dd V$ es el opuesto del área bajo el
camino en el diagrama de Clapeyron: positivo al comprimir, negativo al
expandirse.
\end{proposition}

\begin{proof}
Un pistón de área $S$ empujado por el entorno con la fuerza
$P_{\mathrm{ext}}S$ se desplaza $\dd x$ hacia fuera: el entorno realiza sobre
el sistema el trabajo $-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$.
Toda frontera es un conjunto de pistones así. Cuasiestática: el pistón
está en equilibrio, $P_{\mathrm{ext}} = P$ salvo una diferencia que se anula.
\end{proof}

\begin{definition}[Calor]\label{def:b1:first-law:heat}
\index{calor}\index{termostato}
El \emph{calor} $Q$ es la energía transferida al sistema por una vía
distinta del trabajo macroscópico --- por choques moleculares en una
pared (conducción), por la circulación de un fluido (convección), por
radiación. Un \emph{termostato} es un cuerpo tan grande que su
temperatura no cambia sea cual sea el calor que intercambie. Convenio de
signos en todo el libro: $W$ y $Q$ son \emph{recibidos} por el sistema,
positivos cuando entran.
\end{definition}

\section{El primer principio}

\begin{theorem}[Primer principio de la termodinámica]\label{thm:b1:first-law:firstlaw}
\index{primer principio de la termodinámica}\index{energía interna}
Todo \omterm{def:b1:newton-dynamics:conservation}{sistema cerrado} posee una función de estado, su \emph{\omterm{def:b1:kinetic-theory:U}{energía interna}}
$U$ (extensiva), tal que en cualquier \omterm{def:b1:first-law:transformation}{transformación}
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
donde $E_k$ es la \omterm{thm:b1:work-and-energy:ke}{energía cinética} macroscópica (nula para un sistema en
reposo): la energía recibida como trabajo y como \omterm{def:b1:first-law:heat}{calor} se almacena en
forma de \omterm{def:b1:kinetic-theory:U}{energía interna}. $W$ y $Q$ dependen cada uno del camino; su
suma, no.
\end{theorem}
\admitted

\begin{remark}[Qué dice]\label{rem:b1:first-law:meaning}
Es la conservación de la energía, con el \omterm{def:b1:first-law:heat}{calor} reconocido como una
transferencia de energía (la experiencia de la rueda de paletas de Joule:
un trabajo medido eleva siempre la temperatura del agua tanto como un
\omterm{def:b1:first-law:heat}{calor} medido). Que ``$U$ es una función de estado'' significa que
$\Delta U$ queda fijado por los estados inicial y final --- de modo que un
ciclo tiene $\Delta U = 0$ y $W + Q = 0$: una máquina que entrega trabajo
ha de recibir \omterm{def:b1:first-law:heat}{calor}. El contenido microscópico de $U$ es el de
\cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[Entalpía; capacidades caloríficas]\label{def:b1:first-law:enthalpy}
\index{entalpía}\index{capacidad calorífica a presión constante}
La \emph{entalpía} $H = U + PV$ es una función de estado adaptada a las
transformaciones a presión constante. Las \emph{capacidades caloríficas a
volumen y a presión constantes} son
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
en \unit{J/K}; por \omterm{def:b1:kinetic-theory:scales}{mol} $C_{V,m}$, $C_{P,m}$; por kilogramo, los \omterm{def:b1:first-law:heat}{calores}
específicos $c_V$, $c_P$.
\end{definition}

\begin{proposition}[Calor a volumen constante y a presión constante]\label{prop:b1:first-law:qvqp}
Para un sistema en reposo que solo intercambia trabajo por fuerzas de presión:
\begin{itemize}
  \item \omterm{def:b1:first-law:transformation}{transformación} isócora: $W = 0$ y $Q_V = \Delta U$;
  \item \omterm{def:b1:first-law:transformation}{transformación} monobara entre dos estados a la presión exterior
        $P_0$: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
Isócora: $\dd V = 0$. Monobara: $W = -P_0(V_2 - V_1)$, luego $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$ puesto que $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[Gas perfecto: $U$, $H$, relación de Mayer]\label{prop:b1:first-law:gas}
\index{relación de Mayer}\index{índice adiabático}\index{leyes de Joule}
En un \omterm{def:b1:kinetic-theory:model}{gas perfecto}, $U$ y $H$ solo dependen de $T$ (\omterm{prop:b1:kinetic-theory:Ugas}{leyes de Joule}):
$\dd U = C_V\,\dd T$, $\dd H = C_P\,\dd T$, y
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
$\gamma = 5/3$ para los gases monoatómicos y $7/5$ para los diatómicos a
temperaturas ordinarias. Para una \omterm{prop:b1:kinetic-theory:condensed}{fase condensada}, en el modelo de
\cref{prop:b1:kinetic-theory:condensed}, $C_P \approx C_V = C$ y
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
$H = U + nRT$; derívese respecto de $T$. $C_{V,m} = \tfrac32 R$ o
$\tfrac52 R$ según \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{Transformaciones del gas perfecto}

\begin{proposition}[Las cuatro transformaciones clásicas]\label{prop:b1:first-law:transformations}
\index{transformación isoterma}\index{transformación adiabática}\index{ley de Laplace}
Para $n$ \omterm{def:b1:kinetic-theory:scales}{moles} de \omterm{def:b1:kinetic-theory:model}{gas perfecto}:
\begin{itemize}
  \item isócora: $W = 0$, $Q = \Delta U = nC_{V,m}\Delta T$;
  \item isobara (cuasiestática): $W = -P\Delta V = -nR\Delta T$, $Q = \Delta H =
        nC_{P,m}\Delta T$;
  \item isoterma reversible: $\Delta U = 0$, $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$;
  \item adiabática reversible: $Q = 0$, $W = \Delta U = nC_{V,m}(T_2 - T_1)$,
        y a lo largo del camino (\emph{\omterm{prop:b1:first-law:transformations}{ley de Laplace}})
        \[
        PV^\gamma = \text{cte}, \qquad TV^{\gamma - 1} = \text{cte}, \qquad
        T^\gamma P^{1 - \gamma} = \text{cte} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Isócora e isobara: las proposiciones anteriores. Isoterma: $W =
-\int nRT\,\dd V/V$. Adiabática: $\dd U = \delta W$, es decir $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$, luego $\dd T/T = -(\gamma - 1)\dd V/V$ (usando
$R/C_{V,m} = \gamma - 1$); intégrese: $TV^{\gamma - 1}$ constante; las demás
formas salen de $PV = nRT$.
\end{proof}

\begin{example}[Bomba y diésel]\label{ex:b1:first-law:diesel}
Aire ($\gamma = 1.4$) comprimido adiabáticamente de \qty{1}{bar} y \qty{300}{K}
a \qty{7}{bar} (una bomba de bicicleta): $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- \qty{250}{\degreeCelsius} por un instante, que el cuerpo de la bomba
nota. Con la relación de compresión $20$ de un diésel:
$T_2 = 300 \times 20^{0.4} = \qty{994}{K}$,
$P_2 = 20^{1.4} = \qty{66}{bar}$ --- por encima de la temperatura de
autoinflamación del combustible: el motor no necesita bujía (el problema
de fin de semana recorre el ciclo entero).
\end{example}

\begin{remark}[Transformaciones irreversibles]\label{rem:b1:first-law:irreversible}
\index{expansión de Joule}\index{transformación irreversible}
Cuando la \omterm{def:b1:first-law:transformation}{transformación} no es cuasiestática, la \omterm{prop:b1:first-law:transformations}{ley de Laplace} y $W =
-\int P\,\dd V$ no valen; solo valen $W = -\int P_{\mathrm{ext}}\,\dd V$ y el
primer principio. Dos casos clásicos: un gas que se expande en el vacío
(\omterm{rem:b1:first-law:irreversible}{expansión de Joule}) no recibe trabajo y, con paredes adiabáticas,
tampoco \omterm{def:b1:first-law:heat}{calor} --- $\Delta U
= 0$, y un \omterm{def:b1:kinetic-theory:model}{gas perfecto} conserva su temperatura; un gas comprimido de
golpe por una presión exterior que salta a $P_2$ recibe $W = -P_2\Delta V$,
más que el trabajo reversible hasta la misma presión, y acaba más caliente
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{Calorimetría}

\begin{method}[Balance calorimétrico]\label{met:b1:first-law:calorimetry}
\index{calorimetría}\index{calor latente}
Los cuerpos puestos en contacto térmico dentro de un recipiente aislado a
presión constante (la atmosférica) solo intercambian \omterm{def:b1:first-law:heat}{calor} entre sí: el
total $\Delta H = \sum Q_P = 0$. Para cada cuerpo $\Delta H = mc\,\Delta T$
(sin cambio de fase) o $\pm mL$ para un cambio de estado a su temperatura
de transición, con $L$ el \emph{\omterm{def:b1:first-law:heat}{calor} latente} (\omterm{def:b1:first-law:enthalpy}{entalpía} de fusión o de
vaporización por unidad de masa: agua, $L_f = \qty{334}{kJ/kg}$ a
\qty{0}{\degreeCelsius}, $L_v =
\qty{2260}{kJ/kg}$ a \qty{100}{\degreeCelsius}). Escríbase el balance,
despéjese la temperatura final y compruébese que la fase supuesta para
cada cuerpo es coherente con ella.
\end{method}

\begin{example}[Hielo en agua]\label{ex:b1:first-law:ice}
\qty{50}{g} de hielo a \qty{0}{\degreeCelsius} en \qty{200}{g} de agua a
\qty{30}{\degreeCelsius}: fundirlo exige \qty{16.7}{kJ}; el agua puede ceder
\qty{25.1}{kJ} al enfriarse hasta \qty{0}{\degreeCelsius}, de modo que se
funde todo el hielo y el balance $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ kJ da
$T_f = \qty{8}{\degreeCelsius}$. Con \qty{100}{g} de hielo harían falta
\qty{33.4}{kJ}: solo se funde una parte y $T_f = \qty{0}{\degreeCelsius}$
--- la comprobación importa.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:first-law:1}
Un \omterm{def:b1:kinetic-theory:scales}{mol} de aire ($C_{P,m} = \qty{29.1}{J/(mol.K)}$) se calienta de
\qty{300}{K} a \qty{400}{K} a presión constante. \omterm{def:b1:first-law:heat}{Calor} recibido, trabajo
recibido y variación de \omterm{def:b1:kinetic-theory:U}{energía interna}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:2}
Un \omterm{def:b1:kinetic-theory:scales}{mol} de gas a \qty{300}{K} se comprime isoterma y reversiblemente de
\qty{2.0}{L} a \qty{0.50}{L}. Trabajo recibido, \omterm{def:b1:first-law:heat}{calor} intercambiado y
$\Delta U$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:3}
Aire a \qty{300}{K} y \qty{1}{bar} se comprime adiabática y reversiblemente
hasta una décima parte de su volumen. Temperatura y presión finales.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:4}
Dense $C_{V,m}$, $C_{P,m}$ y $\gamma$ para el argón y para el nitrógeno.
En el agua líquida ($c = \qty{4180}{J/(kg.K)}$), ¿por qué se prescinde de
la distinción entre $c_P$ y $c_V$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:5}
Se mezclan \qty{200}{g} de agua a \qty{80}{\degreeCelsius} con \qty{300}{g}
a \qty{20}{\degreeCelsius} en un vaso aislado. Temperatura final; ¿qué
cambia si el propio vaso tiene una capacidad calorífica de \qty{80}{J/K} y
parte de \qty{20}{\degreeCelsius}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:6}
Se echan \qty{50}{g} de hielo a \qty{-10}{\degreeCelsius} ($c_{\text{hielo}} = \qty{2100}{J/(kg.K)}$)
en \qty{200}{g} de agua a \qty{30}{\degreeCelsius}. Estado y temperatura
finales. La misma pregunta con \qty{150}{g} de hielo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:7}
Un gas recorre un ciclo rectangular en el diagrama de Clapeyron: $(V_1, P_1)
\to (V_2, P_1) \to (V_2, P_2) \to (V_1, P_2) \to (V_1, P_1)$ con $V_2 > V_1$,
$P_2 > P_1$. Trabajo recibido a lo largo del ciclo en función del área;
signo y significado del signo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:8}
Una bomba de bicicleta comprime aire de \qty{1}{bar} y \qty{300}{K} a
\qty{7}{bar} lo bastante deprisa para ser adiabática. Temperatura
alcanzada; por qué se calienta la bomba; y por qué baja un poco la presión
del neumático cuando se deja de bombear.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:9}
Un \omterm{def:b1:kinetic-theory:model}{gas perfecto} se expande en un recipiente aislado en el que se ha hecho
el vacío (\omterm{rem:b1:first-law:irreversible}{expansión de Joule}). Trabajo, \omterm{def:b1:first-law:heat}{calor}, $\Delta U$, $\Delta T$. ¿Es
reversible la \omterm{def:b1:first-law:transformation}{transformación}? ¿Qué haría un gas real?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:10}
Ebullición de \qty{1.0}{kg} de agua a \qty{100}{\degreeCelsius} y \qty{1}{atm}
(volumen del vapor \qty{1.67}{m^3}). \omterm{def:b1:first-law:heat}{Calor} aportado, trabajo intercambiado
con la atmósfera y $\Delta U$. ¿Adónde va la mayor parte del \omterm{def:b1:first-law:heat}{calor} latente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:11}
Un \omterm{def:b1:kinetic-theory:scales}{mol} de \omterm{prop:b1:kinetic-theory:Ugas}{gas diatómico} a \qty{300}{K} y \qty{1}{bar}, en un cilindro
aislado, se comprime llevando de golpe la presión exterior a \qty{5}{bar} y
esperando el equilibrio. Temperatura y volumen finales (primer principio
con $W = -P_{\mathrm{ext}}\Delta V$); compárese con la compresión adiabática
reversible hasta \qty{5}{bar}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:12}
El sonido es una compresión rápida, luego adiabática: su velocidad vale $c = \sqrt{\gamma P/\rho}$
(el valor isotermo $\sqrt{P/\rho}$ fue el error de Newton). Calcúlense
ambos para el aire a \qty{20}{\degreeCelsius} y compárense con los
\qty{343}{m/s} medidos. Muéstrese que $c = \sqrt{\gamma RT/M}$ y compárese
con la \omterm{def:b1:kinetic-theory:kinetictemp}{velocidad cuadrática media} de \cref{ch:b1:kinetic-theory}.
\end{exercise}

\section{Problema: de la bomba de bicicleta al motor diésel}

\begin{problem}[{Problema de fin de semana --- una bomba de mano que
quema los dedos, un cilindro que enciende su propio combustible y una
bola de acero que rebota sobre un colchón de aire: el primer principio en
tres compresiones}]
\label{pb:b1:first-law:1}
Aire: \omterm{def:b1:kinetic-theory:model}{gas perfecto} diatómico, $\gamma = 1.40$, $C_{V,m} = \tfrac52 R$, $C_{P,m} =
\tfrac72 R$, $M = \qty{29}{g/mol}$; $R = \qty{8.314}{J/(mol.K)}$. Estado
inicial en todas partes: $P_1 = \qty{1.00}{bar}$, $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{Parte I --- La bomba de bicicleta.}
El cilindro de la bomba tiene un volumen $V_1 = \qty{212}{cm^3}$; el aire
que contiene se comprime reversible y adiabáticamente hasta alcanzar la
presión del neumático, \qty{7.0}{bar}.
\begin{enumerate}
  \item Cantidad de aire en el cilindro.
  \item Volumen al final de la compresión.
  \item Temperatura al final de la compresión.
  \item Trabajo recibido por el aire en un bombeo.
  \item El neumático (\qty{2.0}{L}) ha de pasar de \qty{1}{bar} a
        \qty{7}{bar} a \qty{300}{K}: ¿cuántos bombeos y cuánto trabajo en
        total (tómese constante el trabajo por bombeo)?
  \item El aire caliente se enfría después en el neumático hasta
        \qty{300}{K}: ¿qué ocurre con la presión del neumático y qué hace
        el ciclista?
  \item Compárese el trabajo adiabático por bombeo con el de una
        compresión isoterma hasta los mismos \qty{7}{bar}. ¿Por qué es
        mayor el trabajo isotermo aunque el gas acabe más frío?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La compresión del diésel.}
Un cilindro de $V_1 = \qty{0.50}{L}$ comprime su aire reversible y
adiabáticamente en un factor $20$.
\begin{enumerate}[resume]
  \item Cantidad de aire y temperatura $T_2$ tras la compresión.
  \item Presión $P_2$.
  \item El combustible se autoinflama por encima de unos \qty{500}{K}:
        explíquese por qué un diésel no necesita bujía, y por qué un motor
        de gasolina (relación $\approx 10$, véase \cref{exo:b1:first-law:3})
        no debe autoinflamarse.
  \item Trabajo recibido por el aire durante la compresión.
  \item A \qty{3000}{rpm} un cilindro comprime $25$ veces por segundo:
        potencia absorbida por la compresión (que se devuelve más tarde en
        el ciclo).
  \item Las compresiones reales son rápidas (irreversibilidad) y pierden
        \omterm{def:b1:first-law:heat}{calor} por las paredes: ¿en qué sentido mueve a $T_2$ cada efecto?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Combustión y expansión.}
En el punto muerto superior arden \qty{20}{mg} de combustible (poder
calorífico \qty{43}{MJ/kg}) mientras el pistón empieza a bajar, a presión
constante $P_2$; después el gas se expande reversible y adiabáticamente
hasta $V_1$. Trátese el gas como aire en todo momento.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{Calor} liberado por el combustible.
  \item Temperatura $T_3$ y volumen $V_3$ al final de la combustión
        isobara.
  \item Trabajo intercambiado durante la fase isobara (¡signo!).
  \item Temperatura $T_4$ al final de la expansión adiabática y trabajo
        intercambiado en ella.
  \item Trabajo neto entregado por el gas en los cuatro tiempos y cociente
        con el \omterm{def:b1:first-law:heat}{calor} liberado (el rendimiento ideal).
  \item Compárese ese rendimiento con $1 - T_1/T_3$, la cota que
        \cref{ch:b1:heat-engines} establecerá para toda máquina que trabaje
        entre esas temperaturas extremas.
  \item \omterm{def:b1:first-law:heat}{Calor} cedido con los gases de escape, por el primer principio a lo
        largo del ciclo; compruébese el balance.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Medida de $\gamma$ con una bola que rebota.}
Una bola de acero de masa $m = \qty{16.5}{g}$ ajusta en un tubo vertical de
vidrio de sección $S = \qty{2.0}{cm^2}$ cerrado por abajo por un matraz de
volumen $V = \qty{10}{L}$ de aire a $P = \qty{1.0}{bar}$; apartada $x$
de su posición de equilibrio, oscila.
\begin{enumerate}[resume]
  \item ¿Por qué pueden tomarse por adiabáticas las compresiones y
        expansiones del aire? Escríbase la relación entre un pequeño
        desplazamiento $x$ y la variación de presión $\dd P$ (\omterm{prop:b1:first-law:transformations}{ley de
        Laplace}, linealizada).
  \item Dedúzcase la fuerza recuperadora sobre la bola y muéstrese que su
        movimiento es armónico, con $\omega_0^2 = \gamma PS^2/mV$.
  \item Calcúlese el periodo, y el valor de $\gamma$ que se deduciría de un
        periodo medido de \qty{1.10}{s}.
  \item ¿Por qué sesgan el resultado el rozamiento entre la bola y el tubo
        y las fugas de aire, y en qué sentido?
  \item Resúmanse las tres compresiones: qué magnitud conservó en cada una
        el primer principio y qué gobernó el exponente adiabático $\gamma$.
\end{enumerate}
\end{problem}
