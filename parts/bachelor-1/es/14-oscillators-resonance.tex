\chapter{Osciladores mecánicos: amortiguamiento y resonancia}\label{ch:b1:oscillators-resonance}

Empújese el columpio de un niño en el momento equivocado y no pasará gran
cosa; empújese una vez por periodo, con suavidad, y en una docena de
vaivenes el niño chilla en lo más alto del arco. Una lavadora vibra con
violencia a una velocidad concreta de su centrifugado y se calma por
encima de ella. Una copa de vino canta cuando un dedo húmedo recorre su
borde al ritmo justo, y estalla cuando un cantante mantiene esa misma nota
con bastante fuerza. Todos esos casos son una \emph{resonancia}: un
sistema con una \omterm{def:b1:oscillators-resonance:harmonic}{frecuencia propia}, excitado cerca de esa frecuencia,
acumula energía ciclo tras ciclo hasta que el rozamiento la drena tan
deprisa como entra. Este capítulo toma el oscilador mecánico --- libre,
amortiguado y después forzado --- y deduce las curvas de resonancia cuyas
gemelas eléctricas aparecieron en \cref{ch:b1:sinusoidal-impedance}.

\section{El oscilador armónico}

\begin{definition}[Oscilador armónico]\label{def:b1:oscillators-resonance:harmonic}
\index{oscilador armónico}\index{frecuencia propia}
Un sistema cuyo desplazamiento $x$ respecto de un equilibrio estable cumple
\[
\ddot x + \omega_0^2\,x = 0
\]
es un \emph{oscilador armónico} de \emph{\omterm{def:b1:transient-regimes:secondorder}{frecuencia angular propia}}
$\omega_0$ (periodo $T_0 = 2\pi/\omega_0$). La solución $x = A\cos(\omega_0t
+ \varphi)$ tiene una \omterm{def:b1:wave-propagation:signal}{amplitud} y una fase fijadas por las condiciones
iniciales y un periodo independiente de la \omterm{def:b1:wave-propagation:signal}{amplitud} (\emph{isocronismo}).
\end{definition}

\begin{proposition}[De dónde salen los osciladores armónicos]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item Masa $m$ sobre un resorte $k$: $\omega_0^2 = k/m$ (\omterm{prop:b1:newton-dynamics:springstring}{ley de Hooke}).
  \item \omterm{ex:b1:newton-dynamics:pendulum}{Péndulo simple} de longitud $\ell$, ángulos pequeños: $\omega_0^2 = g/\ell$.
  \item Cualquier \omterm{def:b1:point-kinematics:frame}{punto material} cerca de un equilibrio estable de un
        potencial $E_p$: $\omega_0^2 = E_p''(x_0)/m$
        (\cref{prop:b1:work-and-energy:stability}).
  \item Circuito LC: $\omega_0^2 = 1/LC$, con $q \leftrightarrow x$,
        $L \leftrightarrow m$, $1/C \leftrightarrow k$.
\end{itemize}
La energía $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ es constante y va y viene entre la forma cinética y la
potencial dos veces por periodo.
\end{proposition}

\begin{proof}
La segunda ley de Newton con $F = -kx$; la ecuación del péndulo
$\ddot\theta + (g/\ell)\sin\theta = 0$ con $\sin\theta \approx \theta$; la
fórmula de Taylor en el mínimo de $E_p$; la \omterm{thm:b1:dc-circuits:kirchhoff}{ley de las mallas} del circuito
LC. Energía: derívese y úsese la ecuación.
\end{proof}

\begin{example}[Un cilindro que cabecea]\label{ex:b1:oscillators-resonance:bob}
Un cilindro vertical de sección $S$ flota con una profundidad $h$
sumergida ($mg = \rho_wgSh$, empuje según \cref{ch:b1:fluid-statics}). Si
se hunde una cantidad $x$, recibe una fuerza adicional hacia arriba
$\rho_wgSx$: $m\ddot x = -\rho_wgSx$, luego $\omega_0^2 = \rho_wgS/m = g/h$
--- lo mismo que un péndulo de longitud $h$. Una boya sumergida
\qty{1}{m} cabecea con un periodo de \qty{2}{s}, sea cual sea su masa.
\end{example}

\section{El oscilador amortiguado}

\begin{proposition}[Oscilador amortiguado]\label{prop:b1:oscillators-resonance:damped}
\index{oscilador amortiguado}\index{factor de calidad!mecánico}
Con un rozamiento viscoso $-\alpha\dot x$, la ecuación pasa a ser
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
la forma canónica de segundo orden de \cref{ch:b1:transient-regimes}: para
$Q > \tfrac12$, un decaimiento seudoperiódico $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$ con $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$; la
energía decae como $\eu^{-2t/\tau}$ y
\[
Q = 2\pi\,\frac{\text{energía almacenada}}{\text{energía perdida por periodo}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
Divídase $m\ddot x + \alpha\dot x + kx = 0$ por $m$. Para la energía: $E
\propto A^2\eu^{-2t/\tau}$ pierde la fracción $2T/\tau = 2\pi/Q$ por
periodo cuando esta es pequeña.
\end{proof}

\begin{example}[Tiempos de extinción]\label{ex:b1:oscillators-resonance:ringing}
Un diapasón de \qty{440}{Hz} todavía suena diez segundos después de
golpearlo: su \omterm{def:b1:wave-propagation:signal}{amplitud} cayó a un $5\%$ en $3\tau = \qty{10}{s}$, luego
$\tau = \qty{3.3}{s}$ y $Q = \omega_0\tau/2 \approx 4600$. La suspensión de
un coche ($Q \approx 0.7$) se para en menos de un periodo; el columpio de
un niño ($Q \approx 20$) pierde un tercio de su energía por vaivén y
necesita un empujón cada vez.
\end{example}

\section{Oscilaciones forzadas y resonancia}

\begin{theorem}[Oscilador excitado sinusoidalmente]\label{thm:b1:oscillators-resonance:forced}
\index{oscilaciones forzadas}\index{resonancia!de amplitud}\index{resonancia!de velocidad}
Excitado por una fuerza $F_0\cos\omega t$, el \omterm{prop:b1:oscillators-resonance:damped}{oscilador amortiguado}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
se instala, tras un transitorio de duración $\sim\tau$, en el \omterm{prop:b1:sinusoidal-impedance:forced}{régimen
forzado} $x = X\cos(\omega t + \varphi)$ con, escribiendo $x_r = \omega/\omega_0$
y $X_0 = F_0/k$ el desplazamiento estático,
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
La \emph{\omterm{def:b1:wave-propagation:signal}{amplitud}} $X$ presenta un máximo, si $Q > 1/\sqrt2$, en $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$, con $X_{\max} \approx QX_0$; la \emph{\omterm{def:b1:wave-propagation:signal}{amplitud}
de velocidad} $V = \omega X$ es máxima exactamente en $\omega_0$, donde $V_{\max} =
F_0/\alpha$, y su curva de resonancia
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
tiene el \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda} $\Delta\omega = \omega_0/Q$. En $\omega_0$ el
desplazamiento se retrasa $\pi/2$ respecto de la fuerza y la velocidad
está en fase con ella.
\end{theorem}

\begin{proof}
Método complejo (\cref{ch:b1:sinusoidal-impedance}): $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$, luego $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$; divídanse numerador y denominador por
$\omega_0^2 = k/m$. El módulo y el argumento se siguen; el máximo de $X$ se
localizó en \cref{prop:b1:sinusoidal-impedance:overvoltage} (la misma
función). Velocidad: $\underline V = j\omega\underline X$ da $V = \omega X$;
escribir $(1 - x_r^2)/x_r = 1/x_r - x_r$ convierte el denominador en
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$, es decir la forma del RLC. En $x_r = 1$,
$\underline X = -jQX_0$: un retraso de $\pi/2$.
\end{proof}

\begin{remark}[Dos resonancias, un circuito]\label{rem:b1:oscillators-resonance:analogy}
\index{analogía electromecánica}
La correspondencia $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$,
$F \leftrightarrow e$ lleva el oscilador forzado al RLC serie de
\cref{ch:b1:sinusoidal-impedance}: la resonancia de velocidad es la
resonancia de corriente (siempre en $\omega_0$, con \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda}
$\omega_0/Q$), y la resonancia de \omterm{def:b1:wave-propagation:signal}{amplitud} es la de la carga, es decir la
de la tensión del condensador (algo por debajo de $\omega_0$, ausente para
$Q$ bajo). La potencia absorbida, $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$, es máxima en $\omega_0$ y vale $F_0^2/2\alpha$,
y los puntos de potencia mitad son el \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda} de velocidad ---
exactamente como en el RLC.
\end{remark}

\begin{example}[Velocidades de resonancia]\label{ex:b1:oscillators-resonance:road}
La suspensión de un coche tiene $f_0 \approx \qty{1.3}{Hz}$ (\cref{pb:b1:transient-regimes:1}).
En una carretera con ondulaciones cada $\lambda = \qty{10}{m}$, la
frecuencia de excitación es $v/\lambda$: resonancia en
$v = \lambda f_0 = \qty{13}{m/s}$, unos \qty{47}{km/h} --- la velocidad a
la que una pista ondulada más sacude el coche, y la razón de montar un
amortiguador con $Q \approx 0.7$: con $Q = 5$ la carrocería rebotaría con
cinco veces la \omterm{def:b1:wave-propagation:signal}{amplitud} de la carretera. Por encima de la resonancia
($x_r \gg 1$) la \omterm{def:b1:wave-propagation:signal}{amplitud} cae como $X_0/x_r^2$: la carrocería ``flota''
sobre los baches rápidos --- el aislamiento, la otra cara de la resonancia.
\end{example}

\section{Excitación por la base: aislamiento y sismómetro}

\begin{proposition}[Oscilador sacudido por su soporte]\label{prop:b1:oscillators-resonance:base}
\index{excitación por la base}\index{aislamiento de vibraciones}\index{sismómetro}
Supóngase que el soporte de un conjunto masa--resorte--amortiguador se
desplaza $x_g(t)$ (la carretera bajo un coche, el suelo bajo un
sismómetro), y sea $u$ el desplazamiento de la masa respecto del soporte y
$x = x_g + u$ su desplazamiento absoluto (desde el equilibrio). Entonces
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
y para $x_g = X_g\cos\omega t$, con $x_r = \omega/\omega_0$:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
Para $x_r \gg 1$: $U \to X_g$ (la masa se queda quieta en el espacio ---
un \emph{sismómetro} lee el desplazamiento del suelo) y $X \to 0$ (la masa
queda \emph{aislada} de la sacudida); para $x_r \ll 1$: $U \approx
\ddot x_g/\omega_0^2$ (un \emph{acelerómetro}) y $X \approx X_g$ (la masa
sigue al soporte).
\end{proposition}

\begin{proof}
El resorte y el amortiguador actúan sobre el movimiento relativo:
$m\ddot x = -ku -
\alpha\dot u$, y $\ddot x = \ddot x_g + \ddot u$. En forma compleja
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$, de donde
$U/X_g$; después $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$.
\end{proof}

\begin{example}[Por qué las máquinas se montan sobre soportes blandos]\label{ex:b1:oscillators-resonance:mounts}
Un motor de \qty{50}{Hz} sobre soportes de goma ajustados a
$f_0 = \qty{10}{Hz}$ ($x_r = 5$, $Q \approx 2$):
$X/X_g = \sqrt{(1 + 6.25)/(576 + 6.25)} = 0.11$ --- nueve décimas partes
de la vibración no llegan al suelo. Los soportes han de ser \emph{más
blandos} de lo que sugiere la intuición: endurecerlos sube $f_0$ hacia los
$50$ Hz y destruye el aislamiento. La misma fórmula, leída al revés, es el
coche de \cref{ex:b1:oscillators-resonance:road} en una carretera ondulada.
\end{example}

\section{Energía: cómo se construye una resonancia}

\begin{proposition}[Balance energético en régimen forzado]\label{prop:b1:oscillators-resonance:energy}
En el \omterm{prop:b1:sinusoidal-impedance:forced}{régimen forzado} estacionario, la \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} que suministra la
fuerza excitadora es igual a la \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} que disipa el rozamiento,
$\tfrac12\alpha V^2$. En la resonancia, la energía almacenada
$\tfrac12 mV_{\max}^2$ vale $Q/2\pi$ veces la energía suministrada por
periodo: el oscilador acumula el trabajo de la excitación durante unos $Q$
ciclos antes de que las pérdidas igualen a la entrada, que es también el
número de ciclos ($\sim\tau/T$) que dura el transitorio.
\end{proposition}

\begin{proof}
Promédiese el teorema de la energía sobre un periodo: $\langle\dd E/\dd t\rangle = 0$
en régimen estacionario, luego $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$. En la resonancia $V_{\max} = F_0/\alpha$, la energía
almacenada vale $\tfrac12 m(F_0/\alpha)^2$ y la suministrada por periodo
$\tfrac12\alpha(F_0/\alpha)^2T$; el cociente es $m/(\alpha T) = Q/2\pi$.
\end{proof}

\begin{example}[El columpio]\label{ex:b1:oscillators-resonance:swing}
Un columpio de \qty{2.5}{m} de largo ($T_0 = \qty{3.2}{s}$, $Q \approx 20$)
recibe un empujón de \qty{50}{N} a lo largo de \qty{0.5}{m} en cada
periodo: \qty{25}{J} por ciclo. Crece hasta que la pérdida por periodo,
$2\pi E/Q$, iguala los \qty{25}{J}: $E = \qty{80}{J}$; con un niño de
\qty{30}{kg}, $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ da $\theta_{\max} \approx 27^\circ$.
Empujando a cualquier otro ritmo, el trabajo de los empujones alternaría
de signo y promediaría cero.
\end{example}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:3}

\section{Problema: el sismógrafo}

\begin{problem}[{Problema de fin de semana --- una masa cuelga de un
resorte dentro de una caja atornillada al suelo; el suelo tiembla: qué
registra la plumilla, por qué el mismo instrumento mide desplazamiento a
alta frecuencia y aceleración a baja frecuencia, y por qué un sismómetro
de periodo largo no puede ser simplemente un resorte largo}]
\label{pb:b1:oscillators-resonance:1}
Una masa $m = \qty{1.0}{kg}$ cuelga de un resorte de rigidez $k$ dentro de
un bastidor rígido fijado al suelo; un amortiguador ejerce $-\alpha\dot u$,
donde $u$ es el desplazamiento de la masa respecto del bastidor, medido
desde su posición de equilibrio. El suelo, y con él el bastidor, se
desplaza verticalmente $x_g(t)$. El sistema de las estrellas lejanas es inercial.

\medskip
\noindent\textbf{Parte I --- La ecuación.}
\begin{enumerate}
  \item Escríbase la posición absoluta de la masa como $x = x_g + u +
        \text{cte}$, enumérense las fuerzas sobre la masa (en el \omterm{thm:b1:newton-dynamics:laws}{sistema
        inercial}) y muéstrese que
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item Pásese a la forma canónica; identifíquense $\omega_0$ y $Q$.
  \item El instrumento se construye con $f_0 = \qty{0.50}{Hz}$ y $Q = 0.70$:
        calcúlense $k$ y $\alpha$.
  \item Una aceleración constante del suelo $a_g$ (por ejemplo, el
        bastidor que se inclina) ¿qué $u$ estacionario da? Interprétese.
  \item ¿Qué registra la plumilla sujeta a la masa, que escribe sobre un
        tambor fijado al bastidor?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- \omterm{prop:b1:point-kinematics:harmonic}{Movimiento armónico} del suelo.}
El suelo se mueve como $x_g = X_g\cos\omega t$; escríbase $x_r = \omega/\omega_0$.
\begin{enumerate}[resume]
  \item Con \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas}, muéstrese que $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$.
  \item Dedúzcase $U/X_g$ en función de $x_r$ y de $Q$.
  \item Límite $x_r \gg 1$: muéstrese que $U \to X_g$ y explíquese
        físicamente qué hace la masa (¿qué mide el instrumento?).
  \item Límite $x_r \ll 1$: muéstrese que $U \approx x_r^2X_g = a_g/\omega_0^2$,
        con $a_g$ la \omterm{def:b1:wave-propagation:signal}{amplitud} de la aceleración del suelo: ¿qué es ahora
        el instrumento?
  \item Valor en $x_r = 1$; ¿por qué es $Q \approx 0.7$ una elección sensata?
  \item Esbócese $U/X_g$ frente a $x_r$ (escalas logarítmicas) para
        $Q = 0.7$ y $Q = 5$.
  \item Fase de $u$ respecto de $x_g$ a alta frecuencia; interprétese.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Diseñar para terremotos.}
Los terremotos lejanos producen ondas superficiales de \qty{0.05}{} a
\qty{0.1}{Hz} con \omterm{def:b1:wave-propagation:signal}{amplitudes} milimétricas; los temblores locales llegan a \qty{5}{} a \qty{10}{Hz}.
\begin{enumerate}[resume]
  \item Para registrar como desplazamientos las ondas de \qty{0.05}{Hz}
        conviene $f_0 \ll \qty{0.05}{Hz}$, digamos \qty{0.02}{Hz}. ¿Qué
        alargamiento estático tendría el resorte bajo el peso de la masa?
        Coméntese.
  \item Un \omterm{ex:b1:newton-dynamics:pendulum}{péndulo simple} de la misma frecuencia: ¿qué longitud?
  \item Los instrumentos reales de periodo largo usan un péndulo de
        ``puerta de jardín'', casi horizontal, o una realimentación
        electrónica de fuerza. Explíquese en una frase qué consigue cada uno.
  \item Para el instrumento de \qty{0.50}{Hz}, calcúlese el alargamiento
        estático: ¿es viable?
  \item Con él, una onda de \qty{0.10}{Hz} y \omterm{def:b1:wave-propagation:signal}{amplitud} \qty{1.0}{mm} ¿qué
        $U$ da? ¿En qué régimen está el instrumento?
  \item Un temblor local de \qty{5.0}{Hz} y \omterm{def:b1:wave-propagation:signal}{amplitud} \qty{1.0}{mm}: ¿$U$?
        ¿Régimen?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El acelerómetro de un teléfono.}
Una masa micromecanizada sobre resortes de silicio tiene
$f_0 = \qty{1.0}{kHz}$ y $Q = 0.7$.
\begin{enumerate}[resume]
  \item Una sacudida de \qty{10}{Hz} con \omterm{def:b1:wave-propagation:signal}{amplitud} de aceleración $g$:
        calcúlese $U$. ¿Cómo se lee un desplazamiento así?
  \item ¿Hasta qué frecuencia se mantiene la lectura dentro del $10\%$ de
        $a_g/\omega_0^2$?
  \item ¿Qué \omterm{def:b1:transient-regimes:firstorder}{constante de tiempo} $\tau = 2Q/\omega_0$ gobierna su respuesta
        a una sacudida, y por qué importa eso para detectar una caída?
  \item La $\tau$ del sismómetro: valor y consecuencia tras un escalón
        brusco del suelo.
  \item ¿Por qué se elige un amortiguamiento crítico (o casi) en ambos
        instrumentos, en lugar de un $Q$ alto?
  \item El teléfono se agita justo a \qty{1.0}{kHz}: ¿en qué factor se
        equivoca la lectura?
  \item Resúmase: un solo aparato, dos regímenes, decididos por una única
        comparación.
\end{enumerate}
\end{problem}
