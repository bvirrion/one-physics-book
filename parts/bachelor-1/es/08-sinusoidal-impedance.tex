\chapter{Régimen sinusoidal permanente e impedancia}\label{ch:b1:sinusoidal-impedance}

Se gira el dial de una radio antigua y, de las cien emisoras que llegan a
la vez a la antena, se oye una: una bobina y un condensador variable se
han ajustado para que resuenen exactamente a una frecuencia. La red
eléctrica que zumba en cada pared es una sinusoide de \qty{50}{Hz}; la
fábrica que mueve con ella motores grandes paga un recargo si su corriente
se retrasa demasiado respecto de la tensión. Las sinusoides son el idioma
natural de los \omterm{def:b1:dc-circuits:linear}{circuitos lineales} --- salen tal como entran, solo
escaladas y desplazadas --- y este capítulo les da su álgebra: \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes
complejas}, \omterm{def:b1:sinusoidal-impedance:impedance}{impedancias}, la resonancia del \omterm{prop:b1:transient-regimes:rlc}{circuito RLC} y la potencia que
entregan realmente las corrientes sinusoidales.

\section{El régimen sinusoidal permanente}

\begin{proposition}[Régimen sinusoidal forzado]\label{prop:b1:sinusoidal-impedance:forced}
\index{régimen sinusoidal permanente}\index{régimen forzado}
En un \omterm{def:b1:dc-circuits:linear}{circuito lineal} excitado por una fuente sinusoidal de \omterm{def:b1:wave-propagation:signal}{frecuencia
angular} $\omega$, toda corriente y toda tensión son, una vez extinguidos
los transitorios, sinusoides de la misma \omterm{def:b1:wave-propagation:signal}{frecuencia angular} $\omega$, cada
una con su propia \omterm{def:b1:wave-propagation:signal}{amplitud} y su propia fase:
$x(t) = X\cos(\omega t + \varphi)$.
\end{proposition}

\begin{proof}
Las ecuaciones del circuito son ecuaciones diferenciales lineales con
coeficientes constantes; su solución general es una solución particular
más las soluciones de la ecuación homogénea, que son los transitorios de
\cref{ch:b1:transient-regimes} y decaen (todo circuito real tiene algo de
resistencia). Existe una solución particular sinusoidal porque derivar
una sinusoide de frecuencia $\omega$ da una sinusoide de la misma
frecuencia: las ecuaciones pueden satisfacerse término a término --- el
método complejo de más abajo la construye explícitamente.
\end{proof}

\begin{definition}[Amplitud compleja]\label{def:b1:sinusoidal-impedance:complex}
\index{amplitud compleja}\index{fasor}
A la sinusoide $x(t) = X\cos(\omega t + \varphi)$ se le asocia la \omterm{def:b1:wave-propagation:signal}{señal}
compleja $\underline{x}(t) = X\eu^{j(\omega t + \varphi)}$, de modo que
$x = \Rea(\underline{x})$, y la \emph{\omterm{def:b1:wave-propagation:signal}{amplitud} compleja}
\[
\underline{X} = X\eu^{j\varphi}, \qquad \underline{x}(t) = \underline{X}\,\eu^{j\omega t},
\]
que lleva la \omterm{def:b1:wave-propagation:signal}{amplitud} $X = \abs{\underline{X}}$ y la fase
$\varphi = \arg\underline{X}$. Aquí $j$ designa la unidad imaginaria,
$j^2 = -1$, ya que la letra $i$ queda reservada para las corrientes.
Dibujada como un vector de longitud $X$ y ángulo $\varphi$,
$\underline{X}$ es un \emph{fasor} (vector de Fresnel).
\end{definition}

\begin{proposition}[Reglas del método complejo]\label{prop:b1:sinusoidal-impedance:rules}
\begin{enumerate}
  \item Las operaciones lineales conmutan con la toma de parte real: la
        \omterm{def:b1:sinusoidal-impedance:complex}{amplitud compleja} de una suma es la suma de las \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes
        complejas}.
  \item Derivar respecto del tiempo multiplica la \omterm{def:b1:sinusoidal-impedance:complex}{amplitud compleja} por
        $j\omega$; integrar la divide por $j\omega$.
  \item Las \omterm{thm:b1:dc-circuits:kirchhoff}{leyes de Kirchhoff}, por ser lineales, valen para las
        \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas}.
\end{enumerate}
\end{proposition}

\begin{proof}
$\Rea(\underline a + \underline b) = \Rea\underline a + \Rea\underline b$;
$\dd(\underline X\eu^{j\omega t})/\dd t = j\omega\underline X\eu^{j\omega t}$,
y $\dd x/\dd t = \Rea(\dd\underline x/\dd t)$ porque la derivación es
lineal sobre los reales. Las sumas de corrientes en un nudo y de
tensiones en una malla son lineales.
\end{proof}

\begin{example}[Sumar dos sinusoides]\label{ex:b1:sinusoidal-impedance:add}
$3\cos\omega t + 4\sin\omega t$: las \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas} son $3$ y
$4\eu^{-j\pi/2} = -4j$ (pues $\sin\omega t = \cos(\omega t - \pi/2)$); su
suma $3 - 4j$ tiene módulo $5$ y argumento $-0.927\ \unit{rad}$:
$5\cos(\omega t - 0.927)$. Sin ninguna identidad trigonométrica.
\end{example}

\section{Impedancia}

\begin{definition}[Impedancia, admitancia]\label{def:b1:sinusoidal-impedance:impedance}
\index{impedancia}\index{admitancia}\index{reactancia}
Para un \omterm{def:b1:dc-circuits:linear}{dipolo lineal} en \omterm{prop:b1:sinusoidal-impedance:forced}{régimen sinusoidal permanente} (\omterm{thm:b1:dc-circuits:kirchhoff}{convenio
receptor}), la \emph{impedancia} es el cociente de \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas}
\[
\underline Z = \frac{\underline U}{\underline I} = Z\eu^{j\varphi}
\quad (\text{ohmios}),
\]
cuyo módulo $Z = U/I$ relaciona las \omterm{def:b1:wave-propagation:signal}{amplitudes} y cuyo argumento
$\varphi = \varphi_u - \varphi_i$ es el adelanto de fase de la tensión
sobre la corriente. Su inversa $\underline Y = 1/\underline Z$ es la
\emph{admitancia} (siemens). La parte real de $\underline Z$ es su
resistencia, y la parte imaginaria su \emph{reactancia}.
\end{definition}

\begin{proposition}[Impedancias de R, L y C]\label{prop:b1:sinusoidal-impedance:rlc}
\[
\underline Z_R = R, \qquad \underline Z_L = jL\omega, \qquad
\underline Z_C = \frac{1}{jC\omega} = -\frac{j}{C\omega} .
\]
Un \omterm{def:b1:dc-circuits:resistor}{resistor} mantiene en fase la tensión y la corriente; en una bobina la
tensión se adelanta $\pi/2$ a la corriente, y en un condensador se
retrasa $\pi/2$. A baja frecuencia una bobina tiende a un hilo y un
condensador a un circuito abierto; a alta frecuencia, al revés.
\end{proposition}

\begin{proof}
$u = Ri$; $u = L\,\dd i/\dd t \to \underline U = jL\omega\,\underline I$;
$i = C\,\dd u/\dd t \to \underline I = jC\omega\,\underline U$. Los
argumentos de $1$, $j$ y $-j$ son $0$, $\pi/2$ y $-\pi/2$.
\end{proof}

\begin{proposition}[Asociación; divisores; teoremas]\label{prop:b1:sinusoidal-impedance:assoc}
Las \omterm{def:b1:sinusoidal-impedance:impedance}{impedancias} en serie se suman; las \omterm{def:b1:sinusoidal-impedance:impedance}{admitancias} en paralelo se suman;
los \omterm{prop:b1:dc-circuits:dividers}{divisores de tensión} y de corriente, los teoremas de Thévenin y de
Norton, la superposición y el \omterm{prop:b1:dc-circuits:millman}{teorema de Millman} de
\cref{ch:b1:dc-circuits} valen todos con \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas} e
\omterm{def:b1:sinusoidal-impedance:impedance}{impedancias} en lugar de tensiones, corrientes y resistencias reales.
\end{proposition}

\begin{proof}
Todas las demostraciones de \cref{ch:b1:dc-circuits} usaron solo las \omterm{thm:b1:dc-circuits:kirchhoff}{leyes
de Kirchhoff} y la linealidad de $u = Ri$, válidas ambas en forma compleja.
\end{proof}

\begin{example}[El divisor RC]\label{ex:b1:sinusoidal-impedance:rcdivider}
Un \omterm{def:b1:dc-circuits:resistor}{resistor} $R$ y un condensador $C$ en serie sometidos a $\underline E$,
con la salida tomada en bornes de $C$: $\underline U_C = \underline E\,\dfrac{1/jC\omega}
{R + 1/jC\omega} = \dfrac{\underline E}{1 + jRC\omega}$. \omterm{def:b1:wave-propagation:signal}{Amplitud}
$E/\sqrt{1 + (RC\omega)^2}$, fase $-\arctan(RC\omega)$: transmisión
completa a baja frecuencia, $1/\sqrt2$ y $-45^\circ$ en
$\omega = 1/RC$, y una caída como $1/\omega$ más allá --- un filtro de
paso bajo, objeto de \cref{ch:b1:filters-transfer-functions}.
\end{example}

\section{El circuito RLC serie: resonancia}

\begin{theorem}[Resonancia de corriente]\label{thm:b1:sinusoidal-impedance:resonance}
\index{resonancia}\index{resonancia!de corriente}\index{ancho de banda}
Un \omterm{def:b1:dc-circuits:resistor}{resistor} $R$, una bobina $L$ y un condensador $C$ en serie sometidos a
$e(t) = E\cos\omega t$ llevan la corriente de \omterm{def:b1:sinusoidal-impedance:complex}{amplitud compleja}
\[
\underline I = \frac{E}{R + j\left(L\omega - \dfrac{1}{C\omega}\right)},
\qquad
I = \frac{E}{\sqrt{R^2 + \left(L\omega - \dfrac{1}{C\omega}\right)^2}},
\qquad
\varphi_i = -\arctan\frac{L\omega - 1/C\omega}{R} .
\]
La \omterm{def:b1:wave-propagation:signal}{amplitud} es máxima, $I_{\max} = E/R$, con la corriente y la tensión en
fase, en la \emph{resonancia} $\omega_0 = 1/\sqrt{LC}$. Con
$Q = L\omega_0/R = 1/(RC\omega_0)$ y $x = \omega/\omega_0$,
\[
\frac{I}{I_{\max}} = \frac{1}{\sqrt{1 + Q^2\left(x - \dfrac1x\right)^2}} ,
\]
y el \emph{\omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda}} --- el intervalo de $\omega$ en el que
$I \geq I_{\max}/\sqrt2$ --- vale
\[
\Delta\omega = \omega_2 - \omega_1 = \frac{\omega_0}{Q} = \frac{R}{L} .
\]
\end{theorem}

\begin{proof}
Las \omterm{def:b1:sinusoidal-impedance:impedance}{impedancias} en serie se suman: $\underline Z = R + jL\omega + 1/jC\omega$, y
$\underline I = \underline E/\underline Z$. El módulo es máximo cuando se
anula la \omterm{def:b1:sinusoidal-impedance:impedance}{reactancia} $L\omega - 1/C\omega$, es decir en $\omega_0$.
Factorícese $L\omega - 1/C\omega = L\omega_0(x - 1/x)$ y divídase por $R$:
$L\omega_0/R = Q$. En los bordes de la banda, $Q(x - 1/x) = \pm 1$, es decir
$x^2 \mp x/Q - 1 = 0$, cuyas raíces positivas son $x_{1,2} = \mp\frac{1}{2Q}
+ \sqrt{1 + \frac{1}{4Q^2}}$; su diferencia vale $1/Q$.
\end{proof}

\begin{proposition}[Sobretensión]\label{prop:b1:sinusoidal-impedance:overvoltage}
\index{resonancia!de tensión}\index{sobretensión}
En la resonancia, las tensiones en bornes de la bobina y del condensador
tienen la misma \omterm{def:b1:wave-propagation:signal}{amplitud} $QE$ y fases opuestas: para $Q \gg 1$ el
condensador soporta una tensión $Q$ veces mayor que la de la fuente. La
propia \omterm{def:b1:wave-propagation:signal}{amplitud} $U_C(\omega)$ presenta un máximo, si $Q > 1/\sqrt2$, en $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$, algo por debajo de $\omega_0$, con
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$.
\end{proposition}

\begin{proof}
En $\omega_0$, $\underline U_C = \underline I/jC\omega_0 = -jQE$ y
$\underline U_L = jL\omega_0\underline I = jQE$, puesto que $I = E/R$. En
general $U_C = I/C\omega = E/\sqrt{(1 - x^2)^2 + x^2/Q^2}$ (multiplíquense
numerador y denominador por $1/C\omega_0$); el cuadrado del denominador
vale $(1 - x^2)^2 + x^2/Q^2$, un polinomio de segundo grado en $x^2$ que
se minimiza en $x^2 =
1 - 1/2Q^2$ cuando esto es positivo, con mínimo $1/Q^2 - 1/4Q^4$.
\end{proof}

\begin{example}[Sintonizar una radio]\label{ex:b1:sinusoidal-impedance:radio}
Una bobina $L = \qty{200}{\micro H}$ y un condensador variable: $C =
\qty{127}{pF}$ sintoniza $f_0 = \qty{1.00}{MHz}$. Con la resistencia del
hilo de la bobina $R = \qty{12.6}{\ohm}$, $L\omega_0 = \qty{1257}{\ohm}$ y
$Q = 100$: \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda} $\Delta f = f_0/Q = \qty{10}{kHz}$ --- la
anchura de un canal de AM --- y los \qty{10}{\micro V} inducidos por un
emisor lejano se convierten en $QE = \qty{1}{mV}$ en bornes del
condensador, de balde. El problema de fin de semana construye este receptor.
\end{example}

\begin{remark}[Por qué de nuevo $Q$]\label{rem:b1:sinusoidal-impedance:Q}
El $Q$ de este capítulo es el $Q$ de \cref{ch:b1:transient-regimes}: el
circuito que oscila durante unos $Q$ periodos cuando se le golpea es el
que resuena en una banda de anchura $\omega_0/Q$ cuando se le excita.
Ambos expresan un mismo cociente: $2\pi$ veces la energía almacenada
dividida por la energía disipada en un periodo
(\cref{exo:b1:sinusoidal-impedance:12}). Una resonancia aguda y una
oscilación larga son la misma propiedad, y también lo es la respuesta
lenta: un filtro de \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda} $\Delta\omega$ tarda un tiempo del
orden de $1/\Delta\omega$ en estabilizarse.
\end{remark}

\section{Potencia en régimen sinusoidal}

\begin{definition}[Valor eficaz]\label{def:b1:sinusoidal-impedance:rms}
\index{valor eficaz}\index{valor cuadrático medio}
El \emph{valor eficaz} (rms, o valor cuadrático medio) de una
\omterm{def:b1:wave-propagation:signal}{señal} periódica $x(t)$ es $X_{\mathrm{rms}} = \sqrt{\langle x^2\rangle}$,
la raíz cuadrada del promedio temporal de $x^2$ sobre un periodo. Para
una sinusoide de \omterm{def:b1:wave-propagation:signal}{amplitud} $X$, $X_{\mathrm{rms}} = X/\sqrt2$. Los
\qty{230}{V} de la red son un valor eficaz; su \omterm{def:b1:wave-propagation:signal}{amplitud} es \qty{325}{V}.
\end{definition}

\begin{theorem}[Potencia media]\label{thm:b1:sinusoidal-impedance:power}
\index{potencia media}\index{factor de potencia}\index{potencia reactiva}
Un dipolo con $u = U\cos\omega t$ e $i = I\cos(\omega t - \varphi)$
(\omterm{thm:b1:dc-circuits:kirchhoff}{convenio receptor}) recibe la potencia instantánea $p = ui$, de valor
medio
\[
P = \tfrac12\,UI\cos\varphi = U_{\mathrm{rms}}I_{\mathrm{rms}}\cos\varphi
= \tfrac12\Rea\!\big(\underline U\,\underline I^{\,*}\big) = \tfrac12\Rea(\underline Z)\,I^2,
\]
donde $\cos\varphi$ es el \emph{\omterm{thm:b1:sinusoidal-impedance:power}{factor de potencia}}. Un \omterm{def:b1:dc-circuits:resistor}{resistor} recibe
$\tfrac12 RI^2 = RI_{\mathrm{rms}}^2$; una bobina o un condensador ideales
($\varphi = \pm\pi/2$) no reciben \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} --- almacenan energía
durante medio periodo y la devuelven durante el otro medio.
\end{theorem}

\begin{proof}
$p = UI\cos\omega t\cos(\omega t - \varphi) = \tfrac12 UI[\cos\varphi +
\cos(2\omega t - \varphi)]$; el segundo término promedia cero. Con
$\underline U = U$ e $\underline I = I\eu^{-j\varphi}$,
$\underline U\,\underline I^* = UI\eu^{j\varphi}$, cuya parte real vale
$UI\cos\varphi$; y $\underline U = \underline Z\,\underline I$ da
$\underline U\,\underline I^* = \underline Z I^2$.
\end{proof}

\begin{example}[Corrección del factor de potencia]\label{ex:b1:sinusoidal-impedance:pf}
El motor de un taller a \qty{230}{V} eficaces consume \qty{2.0}{kW} con
$\cos\varphi = 0.70$ (inductivo): $I_{\mathrm{rms}} = 2000/(230 \times
0.70) = \qty{12.4}{A}$, frente a \qty{8.7}{A} si el \omterm{thm:b1:sinusoidal-impedance:power}{factor de potencia}
fuese $1$ --- la corriente sobrante calienta en balde los cables de la
compañía, de ahí el recargo. Un condensador en paralelo aporta localmente
los vaivenes de energía de un cuarto de periodo de la bobina: la
\emph{\omterm{thm:b1:sinusoidal-impedance:power}{potencia reactiva}} $P\tan\varphi =
\qty{2.0}{kvar}$ exige $C = P\tan\varphi/(\omega U_{\mathrm{rms}}^2)
\approx \qty{120}{\micro F}$ a \qty{50}{Hz}, y la corriente de línea baja
a \qty{8.7}{A}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:1}
Dese la \omterm{def:b1:sinusoidal-impedance:complex}{amplitud compleja} de $u = 5\cos(\omega t + \pi/3)$ (V), en forma
polar y cartesiana. Usando \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complejas}, escríbase $3\cos\omega t
+ 4\sin\omega t$ como un único coseno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:2}

\section{Problema: el receptor de radio resonante}

\begin{problem}[{Problema de fin de semana --- una bobina, un condensador
variable y una antena: cómo una resonancia serie escoge una emisora entre
cien, multiplica por cien su señal débil y por qué no puede hacerse
arbitrariamente selectiva}]
\label{pb:b1:sinusoidal-impedance:1}
La antena se modela por una fem sinusoidal ideal $e(t) = E\cos\omega t$ en
serie con una bobina $L = \qty{200}{\micro H}$, cuyo hilo tiene una
resistencia $R = \qty{12.6}{\ohm}$, y un condensador variable $C$. El
receptor lee la tensión en bornes de $C$. Más adelante (Parte IV) se tiene
en cuenta la propia resistencia de la antena $R_a = \qty{50}{\ohm}$.

\medskip
\noindent\textbf{Parte I --- La corriente resonante.}
\begin{enumerate}
  \item Escríbase la \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} compleja del circuito serie.
  \item Dese la \omterm{def:b1:sinusoidal-impedance:complex}{amplitud compleja} de la corriente y, después, su \omterm{def:b1:wave-propagation:signal}{amplitud}
        $I(\omega)$ y su fase $\varphi_i(\omega)$ respecto de $e$.
  \item Muéstrese que $I$ es máxima en $\omega_0 = 1/\sqrt{LC}$; dense
        $I_{\max}$ y la fase en ese caso.
  \item ¿Qué capacidad sintoniza $f_0 = \qty{1.000}{MHz}$? Calcúlese
        $L\omega_0$.
  \item Defínase $Q = L\omega_0/R$ y calcúlese.
  \item Muéstrese que $I/I_{\max} = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ con
        $x = \omega/\omega_0$.
  \item Dígase el signo de $\varphi_i$ por debajo y por encima de la
        resonancia e interprétese (comportamiento capacitivo o inductivo).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Selectividad.}
\begin{enumerate}[resume]
  \item Escríbase la condición que define las dos frecuencias a las que
        $I = I_{\max}/\sqrt2$.
  \item Resuélvase y muéstrese que $\Delta\omega = \omega_2 - \omega_1 =
        \omega_0/Q$.
  \item Calcúlese el \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda} $\Delta f$ de este receptor.
  \item Una segunda emisora emite a \qty{1.010}{MHz} y una tercera a
        \qty{1.020}{MHz}: calcúlese la razón de \omterm{def:b1:wave-propagation:signal}{amplitudes} $I/I_{\max}$ de
        cada una y la razón de potencias correspondiente en decibelios.
  \item Las emisoras de AM se separan de \qty{9}{} a \qty{10}{kHz}. ¿Es
        bastante selectivo el receptor? ¿Qué lo mejoraría?
  \item El condensador puede ajustarse entre \qty{50}{} y \qty{500}{pF}:
        ¿qué banda de frecuencias cubre el receptor?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La sobretensión.}
\begin{enumerate}[resume]
  \item Exprésese la \omterm{def:b1:sinusoidal-impedance:complex}{amplitud compleja} de la tensión del condensador y
        muéstrese que en la resonancia su \omterm{def:b1:wave-propagation:signal}{amplitud} vale $QE$.
  \item La fem de antena vale $E = \qty{10}{\micro V}$: calcúlense
        $I_{\max}$ y la tensión del condensador en la resonancia.
        Coméntese.
  \item Muéstrese que la tensión de la bobina tiene la misma \omterm{def:b1:wave-propagation:signal}{amplitud} y
        está en oposición de fase con la del condensador en la resonancia:
        ¿qué ``ve'' la fuente?
  \item Calcúlense la energía almacenada en el circuito en la resonancia y
        la energía disipada por periodo; compruébese que su cociente por
        $2\pi$ vale $Q$.
  \item A partir de \cref{ch:b1:transient-regimes}, ¿cuál es la \omterm{def:b1:transient-regimes:firstorder}{constante
        de tiempo} $\tau = 2Q/\omega_0$ de este circuito? ¿Por qué un $Q$
        muy alto acaba distorsionando la voz y la música (\omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda}
        del audio: unos \qty{5}{kHz} en AM)?
  \item Si la resistencia del hilo de la bobina se duplicase (hilo más
        fino), ¿cómo cambiarían $Q$, el \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda} y la tensión del
        condensador?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La resistencia de la antena, y la potencia.}
\begin{enumerate}[resume]
  \item Con $R_a = \qty{50}{\ohm}$ en serie, recalcúlense $Q$, el \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de
        banda} y la tensión del condensador para $E = \qty{10}{\micro V}$.
        Conclúyase.
  \item Calcúlense la \omterm{thm:b1:sinusoidal-impedance:power}{potencia media} entregada por la fem de antena en la
        resonancia y la fracción disipada en $R_a$.
  \item El teorema de máxima potencia pediría una resistencia de carga
        igual a $R_a$: ¿por qué choca ese objetivo con la selectividad?
  \item Los receptores reales conectan la antena a una toma intermedia de
        la bobina, o a través de un pequeño condensador de acoplamiento.
        Explíquese cualitativamente qué se consigue con ello.
  \item Usando $P = \tfrac12 UI\cos\varphi$, calcúlese el \omterm{thm:b1:sinusoidal-impedance:power}{factor de
        potencia} en las dos frecuencias de potencia mitad y explíquese el
        nombre ``potencia mitad''.
  \item Resúmase el diseño: qué aporta $Q$, qué cuesta y qué valor se
        elige para un receptor de \qty{1}{MHz} con canales de
        \qty{10}{kHz}.
\end{enumerate}
\end{problem}
