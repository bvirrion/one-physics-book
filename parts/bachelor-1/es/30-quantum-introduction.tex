\chapter{Introducción a la física cuántica}\label{ch:b1:quantum-introduction}

Ilumínese con luz ultravioleta un metal limpio y saldrán electrones al
instante, con una energía que depende del color de la luz y en absoluto
de su brillo; ilumínese con luz roja y no saldrá nada por brillante que
sea la lámpara. Envíense electrones de uno en uno a través de dos
rendijas y llegarán como puntos, cada uno en un solo sitio --- y los
puntos van formando, al cabo de miles, las franjas de una onda. Ninguno
de los dos hechos encaja en la física de los veintinueve capítulos
anteriores. Este último capítulo presenta las ideas que sí los explican:
la luz se intercambia en cuantos, los fotones; la materia tiene una
\omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda}; lo que se propaga es una \omterm{def:b1:wave-propagation:signal}{amplitud} cuyo cuadrado es una
probabilidad; la posición y la cantidad de movimiento no pueden estar
ambas bien definidas; y una partícula confinada solo puede tener ciertas
energías --- por lo que los átomos tienen tamaño, los espectros tienen
rayas y un grano de semiconductor de unos nanómetros brilla en un color
que fija su diámetro.

\section{El fotón}

\begin{theorem}[Relaciones de Planck--Einstein]\label{thm:b1:quantum-introduction:photon}
\index{fotón}\index{constante de Planck}
La luz de frecuencia $\nu$ y \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} $\lambda$ intercambia
energía con la materia solo en cuantos, los \emph{fotones}, de energía y
cantidad de movimiento
\[
E = h\nu = \frac{hc}{\lambda} , \qquad p = \frac{h}{\lambda} = \frac{E}{c} ,
\]
con la \emph{\omterm{thm:b1:quantum-introduction:photon}{constante de Planck}} $h = \qty{6.626e-34}{J.s}$ ($\hbar = h/2\pi =
\qty{1.055e-34}{J.s}$). Útil: $hc = \qty{1240}{eV.nm}$ --- un fotón de
\qty{620}{nm} lleva \qty{2}{eV}.
\end{theorem}
\admitted

\begin{proposition}[El efecto fotoeléctrico]\label{prop:b1:quantum-introduction:photoelectric}
\index{efecto fotoeléctrico}\index{trabajo de extracción}\index{potencial de frenado}
La luz que incide sobre un metal de \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{trabajo de extracción}} $W$ (la
energía que liga a sus electrones menos ligados, unos pocos eV) arranca
electrones de \omterm{thm:b1:work-and-energy:ke}{energía cinética} máxima
\[
E_{k,\max} = h\nu - W ,
\]
luego solo por encima del \emph{umbral} $\nu_0 = W/h$, al instante y con
una energía independiente de la intensidad; la intensidad fija el
\emph{número} de electrones por segundo. El \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{potencial de frenado}}
$V_s$ que justo anula la corriente mide $E_{k,\max} = eV_s$, y la gráfica
de $V_s$ frente a $\nu$ es una recta de pendiente $h/e$ --- así midió
Millikan $h$.
\end{proposition}

\begin{proof}
Cada fotón lo absorbe un solo electrón, que se queda a lo sumo con
$h\nu - W$; una onda entregaría energía de forma continua a todos los
electrones, la energía crecería con la intensidad y una luz débil
tardaría horas en acumular $W$ sobre un átomo
(\cref{pb:b1:quantum-introduction:1}) --- nada de lo cual se observa. La
interpretación de Einstein (1905) se admite como ley.
\end{proof}

\begin{example}[Órdenes de magnitud]\label{ex:b1:quantum-introduction:orders}
Un fotón visible: \qty{2}{eV}; un fotón de rayos X de \qty{0.1}{nm}: \qty{12}{keV};
un fotón de radio a \qty{100}{MHz}: \qty{4e-7}{eV}. Un láser rojo de \qty{1}{mW}
emite $10^{-3}/(3.1 \times 10^{-19}) = 3 \times 10^{15}$ fotones por segundo;
una antena de radio que recoge \qty{1}{pW} a \qty{100}{MHz} recibe $10^{13}$
por segundo --- tantos y tan pequeños que la descripción ondulatoria es
exacta a todos los efectos; el \omterm{def:b1:optical-instruments:eye}{ojo}, en cambio, puede registrar un
destello de cien fotones. La naturaleza cuántica de la luz se manifiesta
cuando los fotones son pocos o energéticos: células fotoeléctricas,
detectores de rayos X, el grano de una fotografía poco expuesta.
\end{example}

\begin{remark}[La dispersión Compton]\label{rem:b1:quantum-introduction:compton}
\index{dispersión Compton}
Un fotón que rebota en un electrón libre le transfiere cantidad de
movimiento como una bola de billar: su \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} crece en
$\Delta\lambda = \frac{h}{m_ec}(1 -
\cos\theta)$, con $h/m_ec = \qty{2.43}{pm}$ (admitido --- es un choque
relativista, Año 2). Invisible para la luz ($\Delta\lambda/\lambda \sim 10^{-5}$),
es un efecto del $20\%$ para los rayos X, y fue el experimento que
convenció a los escépticos (1923) de que el fotón lleva la cantidad de
movimiento $h/\lambda$.
\end{remark}

\section{Las ondas de materia}

\begin{theorem}[De Broglie]\label{thm:b1:quantum-introduction:debroglie}
\index{longitud de onda de De Broglie}\index{onda de materia}
Una partícula de cantidad de movimiento $p$ se propaga como una onda de
\omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda}
\[
\lambda = \frac{h}{p} ,
\]
y muestra en consecuencia difracción e interferencias: electrones en un
cristal (Davisson y Germer, 1927), a través de dos rendijas (Jönsson,
1961), neutrones, átomos y moléculas de sesenta átomos de carbono. Para
un electrón acelerado por $U$, $\lambda = h/\sqrt{2m_eeU} = \qty{1.23}{nm}/\sqrt{U/\unit{V}}$.
\end{theorem}
\admitted

\begin{example}[Por qué nunca lo vimos]\label{ex:b1:quantum-introduction:wavelengths}
Un electrón de \qty{100}{eV}: $\lambda = \qty{0.12}{nm}$, la separación de
los átomos en un cristal --- de ahí la difracción y el \omterm{def:b1:optical-instruments:microscope}{microscopio}
electrónico. Un neutrón térmico (\qty{0.025}{eV}): \qty{0.18}{nm}. Una
bola de \qty{1}{g} a \qty{1}{m/s}: $\lambda = \qty{7e-31}{m}$, $10^{20}$ veces
menor que un núcleo: ninguna rendija ni ningún cristal podrían jamás
difractarla. La naturaleza ondulatoria de la materia es universal; es su
\omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} la que la hace visible solo en lo ligero y lo lento.
\end{example}

\begin{proposition}[Las partículas de una en una construyen las franjas]\label{prop:b1:quantum-introduction:buildup}
\index{dualidad onda--corpúsculo}
Los electrones enviados de uno en uno a través de dos rendijas
(Tonomura, 1989) hacen cada uno un solo punto en la pantalla, en un
lugar que no puede predecirse; los puntos se acumulan formando la figura
de interferencias de dos rendijas de una onda de \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} $h/p$.
Cerrar una rendija borra las franjas --- la figura no la hacen unos
electrones interfiriendo con otros, sino la onda de cada electrón al
pasar por las dos rendijas. Detectar por qué rendija pasó una partícula
también borra las franjas.
\end{proposition}
\admitted

\section{Amplitudes, probabilidades y principio de indeterminación}

\begin{definition}[Función de onda]\label{def:b1:quantum-introduction:wavefunction}
\index{función de onda}\index{densidad de probabilidad}
El estado de una partícula lo describe una \emph{función de onda}
$\psi(x, t)$ (compleja), tal que $|\psi(x, t)|^2\,\dd x$ es la
probabilidad de encontrar la partícula entre $x$ y $x + \dd x$ cuando se
mira: $|\psi|^2$ es una \emph{densidad de probabilidad} y $\int|\psi|^2\dd x =
1$. Las \omterm{def:b1:wave-propagation:signal}{amplitudes} se suman (superposición): si dos caminos llevan al
mismo punto, $\psi = \psi_1 + \psi_2$ y $|\psi|^2 = |\psi_1|^2 + |\psi_2|^2 +
2\,\mathrm{Re}(\psi_1^\ast\psi_2)$ --- el último término es la
interferencia. Una medida da uno de los resultados posibles, al azar con
las probabilidades que prescriben las \omterm{def:b1:wave-propagation:signal}{amplitudes}, y deja a la partícula
en el estado hallado.
\end{definition}

\begin{remark}[Qué es lo que ondula]\label{rem:b1:quantum-introduction:whatwaves}
$\psi$ no es un campo físico como $\vect E$: no se mide, solo se mide su
cuadrado, y describe el mejor conocimiento posible de dónde se
encontrará la partícula. Un electrón aislado no está repartido por la
pantalla; se detecta entero, en un punto; lo que está repartido es la
probabilidad. Ese es el contenido de la teoría, y nunca ha fallado en
una prueba.
\end{remark}

\begin{theorem}[Principio de indeterminación de Heisenberg]\label{thm:b1:quantum-introduction:heisenberg}
\index{principio de indeterminación}
La posición y la cantidad de movimiento de una partícula según un mismo
eje no pueden estar ambas bien definidas: sus desviaciones típicas, en
cualquier estado, cumplen
\[
\Delta x\,\Delta p_x \geq \frac{\hbar}{2} .
\]
Una partícula confinada en una región de tamaño $\ell$ tiene por tanto
una dispersión de cantidad de movimiento de al menos $\hbar/2\ell$, y
con ella una \omterm{thm:b1:work-and-energy:ke}{energía cinética} del orden de $\hbar^2/2m\ell^2$ que ningún
enfriamiento puede quitar: el confinamiento cuesta energía.
\end{theorem}
\admitted

\begin{example}[El tamaño del átomo de hidrógeno]\label{ex:b1:quantum-introduction:hsize}
\index{radio de Bohr}
Un electrón a menos de $r$ del protón tiene $p \gtrsim \hbar/r$ y energía
$E(r) \approx \dfrac{\hbar^2}{2m_er^2} - \dfrac{e^2}{4\pi\varepsilon_0r}$: el primer
término se dispara si $r$ disminuye, y el segundo si crece. $\dd E/\dd r = 0$ en
\[
a_0 = \frac{4\pi\varepsilon_0\hbar^2}{m_ee^2} = \qty{53}{pm} , \qquad
E(a_0) = -\frac{m_ee^4}{2(4\pi\varepsilon_0)^2\hbar^2} = \qty{-13.6}{eV} :
\]
el \emph{\omterm{ex:b1:quantum-introduction:hsize}{radio de Bohr}} y la energía de ionización del hidrógeno, hasta
la última cifra. La física clásica no podía explicar por qué el electrón
no cae en espiral sobre el núcleo (radia) ni por qué todos los átomos de
un elemento son idénticos; el \omterm{thm:b1:quantum-introduction:heisenberg}{principio de indeterminación} responde a
ambas cosas en una línea: el átomo es tan pequeño como puede serlo sin
que la \omterm{thm:b1:work-and-energy:ke}{energía cinética} supere a la de enlace.
\end{example}

\section{La cuantización de la energía}

\begin{proposition}[Partícula en una caja]\label{prop:b1:quantum-introduction:box}
\index{pozo infinito}\index{cuantización}\index{estado fundamental}
Una partícula de masa $m$ confinada entre dos paredes impenetrables
separadas una distancia $L$ solo admite, como la cuerda de Melde
(\cref{ch:b1:wave-propagation}), las \omterm{thm:b1:wave-propagation:standing}{ondas estacionarias} $\lambda_n = 2L/n$,
luego solo las cantidades de movimiento $p_n = nh/2L$ y las energías
\[
E_n = \frac{n^2h^2}{8mL^2} , \qquad n = 1, 2, 3, \ldots
\]
con funciones de onda $\psi_n(x) \propto \sin(n\pi x/L)$. La energía está
\emph{cuantizada}; la menor, el \emph{\omterm{prop:b1:quantum-introduction:box}{estado fundamental}} $E_1 > 0$, es la
energía de confinamiento del \omterm{thm:b1:quantum-introduction:heisenberg}{principio de indeterminación} hecha exacta;
la partícula salta de un nivel a otro absorbiendo o emitiendo un fotón
de energía $h\nu = E_p - E_n$.
\end{proposition}

\begin{proof}
$\psi$ ha de anularse en las paredes (la partícula no puede estar ahí),
de modo que la onda encaja un número entero de semilongitudes de onda en
$L$ --- la condición de la cuerda; De Broglie da $p$, y $E = p^2/2m$. Que
$\psi$ obedezca a una ecuación de ondas con esas condiciones de contorno
se admite (la ecuación de Schrödinger, Año 2).
\end{proof}

\begin{proposition}[El átomo de hidrógeno]\label{prop:b1:quantum-introduction:hydrogen}
\index{átomo de hidrógeno}\index{niveles de energía}\index{rayas espectrales}
El electrón del \omterm{prop:b1:quantum-introduction:hydrogen}{átomo de hidrógeno} tiene las energías discretas
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2} , \qquad n = 1, 2, 3, \ldots ,
\]
siendo $n = 1$ el \omterm{prop:b1:quantum-introduction:box}{estado fundamental} y $E \geq 0$ el átomo ionizado. Su
espectro consta de rayas en $h\nu = E_p - E_n$: la serie de Lyman, hasta
$n = 1$ (ultravioleta, \qty{122}{nm} para $2 \to 1$), la serie de Balmer,
hasta $n = 2$ (visible: \qty{656}{nm}, \qty{486}{nm}, \qty{434}{nm}, \ldots),
y así sucesivamente. El modelo de Bohr (1913) obtiene estas energías
cuantizando el \omterm{def:b1:angular-momentum:L}{momento angular} de una \omterm{prop:b1:central-forces:effective}{órbita circular}, $m_evr = n\hbar$
(\cref{exo:b1:quantum-introduction:9}); la teoría completa sustituye las
órbitas por funciones de onda, pero conserva las energías.
\end{proposition}
\admitted

\begin{example}[Colores a partir del tamaño: el punto cuántico]\label{ex:b1:quantum-introduction:dot}
\index{punto cuántico}
Un nanocristal de semiconductor de unos pocos nanómetros confina sus
electrones como una caja: cuanto menor es el cristal, mayor es la energía
de confinamiento que se añade a la banda prohibida del semiconductor y
más azul la luz que emite. Los puntos de seleniuro de cadmio de radio
\qty{2}{nm} brillan en azul, los de \qty{3}{nm} en amarillo y los de
\qty{4}{nm} en rojo --- un solo material, con los colores elegidos con
una regla (\cref{pb:b1:quantum-introduction:1}). Las pantallas de
televisión los usan.
\end{example}

\begin{remark}[Qué añade el resto de la teoría]\label{rem:b1:quantum-introduction:rest}
\index{efecto túnel}\index{láser}
La ecuación de Schrödinger vuelve exactos la caja, el átomo y la
molécula; predice que una partícula puede atravesar una barrera que no
tiene energía para escalar (el \emph{\omterm{rem:b1:quantum-introduction:rest}{efecto túnel}}, detrás de la
radiactividad y del \omterm{def:b1:optical-instruments:microscope}{microscopio} de \omterm{rem:b1:quantum-introduction:rest}{efecto túnel}); que las partículas
idénticas son o bien sociables (los bosones --- de ahí el \emph{láser},
donde los fotones se amontonan en un solo modo) o bien exclusivas (los
fermiones --- de ahí la tabla periódica y la rigidez de la materia); y
que el espín del electrón es un \omterm{def:b1:magnetostatics:moment}{momento magnético} sin equivalente
clásico. Todo eso es del Año 2 y más allá; todo ello se apoya en las
pocas ideas de este capítulo.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:1}
Energía de un fotón de: radio a \qty{100}{MHz}; luz de \qty{500}{nm};
rayos X de \qty{0.10}{nm}; rayos gamma de \qty{1}{MeV} (dese su \omterm{prop:b1:wave-propagation:sinusoidal}{longitud
de onda}). Número de fotones por segundo de un láser de \qty{1.0}{mW} a
\qty{633}{nm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:2}
Sodio, $W = \qty{2.3}{eV}$: \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} umbral; \omterm{thm:b1:work-and-energy:ke}{energía cinética}
máxima y \omterm{prop:b1:quantum-introduction:photoelectric}{potencial de frenado} con luz de \qty{400}{nm}; ¿qué cambia si se
duplica la intensidad? ¿Y con luz de \qty{600}{nm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:3}
\omterm{thm:b1:quantum-introduction:debroglie}{Longitud de onda de De Broglie} de un electrón de \qty{100}{eV} y de
\qty{1.0}{keV}; de un neutrón de \qty{0.025}{eV}; de una bola de \qty{1}{g}
a \qty{1}{m/s}. ¿Cuáles pueden difractarse en un cristal (separación
\qty{0.2}{nm})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:4}
Un electrón en una caja de \qty{1.0}{nm}: $E_1$, $E_2$ y \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda}
del fotón emitido en $2 \to 1$. Un protón en una caja de \qty{5}{fm}
(un núcleo): $E_1$ en \unit{MeV}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:5}
\omterm{prop:b1:quantum-introduction:photoelectric}{Potenciales de frenado} medidos en un metal: \qty{0.40}{V} a
\qty{6.0e14}{Hz}, \qty{1.23}{V} a \qty{8.0e14}{Hz} y \qty{2.05}{V} a
\qty{1.0e15}{Hz}. Dedúzcanse $h$ (con $e$ conocido) y el \omterm{prop:b1:quantum-introduction:photoelectric}{trabajo de
extracción}; \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} umbral; ¿qué metal podría ser (cesio
\qty{2.1}{eV}, potasio \qty{2.3}{eV}, cinc \qty{4.3}{eV})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:6}
Compton: un fotón de rayos X de \qty{10}{pm} se dispersa \qty{90}{\degree}
en un electrón. Nueva \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda}; energías del fotón antes y
después; energía cedida al electrón. Desplazamiento relativo para luz de
\qty{500}{nm} --- por qué Compton necesitó rayos X.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:7}
\emph{La experiencia de Jönsson.} Electrones de \qty{50}{keV} (úsese $p =
\sqrt{2m_eE}$), dos rendijas separadas \qty{2.0}{\micro m} y pantalla a
\qty{35}{cm}. \omterm{prop:b1:wave-propagation:sinusoidal}{Longitud de onda}; interfranja; por qué hizo falta una \omterm{def:b1:mirrors-thin-lenses:lens}{lente}
electrónica de aumento. En la versión de Tonomura, los electrones cruzan
el aparato de uno en uno: ¿qué muestra la pantalla tras 10 y tras $10^5$
electrones, y qué enseña eso?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:8}
\omterm{thm:b1:work-and-energy:ke}{Energía cinética} mínima, según el \omterm{thm:b1:quantum-introduction:heisenberg}{principio de indeterminación}, de: un
electrón confinado en \qty{0.1}{nm} (un átomo); un electrón confinado en
\qty{5}{fm} (un núcleo --- úsese $E \approx pc$ si el resultado es
relativista, y conclúyase si los núcleos pueden contener electrones); un
grano de polvo de \qty{1}{\micro g} localizado en \qty{1}{\micro m} (dese $\Delta v$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:9}
\emph{El modelo de Bohr.} Un electrón en una \omterm{prop:b1:central-forces:effective}{órbita circular} de radio $r$
alrededor de un protón: (a) rapidez y energía en función de $r$ (dinámica
coulombiana, \cref{ch:b1:central-forces}); (b) impóngase $m_evr = n\hbar$ y
dedúzcanse $r_n = n^2a_0$, $E_n = E_1/n^2$ con $a_0$ y $E_1$ como en
\cref{ex:b1:quantum-introduction:hsize}; (c) $v_1$ y $v_1/c$; (d)
\omterm{prop:b1:wave-propagation:sinusoidal}{longitudes de onda} de Lyman $\alpha$ ($2 \to 1$) y de H$\alpha$ ($3 \to 2$);
(e) muéstrese que la órbita $n$ contiene exactamente $n$ \omterm{thm:b1:quantum-introduction:debroglie}{longitudes de onda de De Broglie}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:10}
\emph{Sistemas hidrogenoides.} (a) Para un núcleo de carga $Ze$,
muéstrese que $E_n = -Z^2 \times \qty{13.6}{eV}/n^2$ y $r_n = n^2a_0/Z$:
energía de ionización del He$^+$ y del último electrón del uranio ($Z = 92$; coméntese).
(b) Hidrógeno muónico: el muon tiene $207$ veces la masa del electrón:
radio y energía fundamental; \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} de su raya Lyman $\alpha$;
por qué se usa para medir el tamaño del protón (radio \qty{0.84}{fm}).
(c) Positronio (un electrón y un positrón, de masas iguales): muéstrese
que la \omterm{prop:b1:systems-of-points:twobody}{masa reducida} divide por dos las energías: su Lyman $\alpha$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:11}
\emph{\omterm{ex:b1:quantum-introduction:dot}{Puntos cuánticos}.} En un nanocristal esférico de radio $R$, la
energía de confinamiento de una partícula de masa efectiva $m^\ast$ vale
$h^2/8m^\ast R^2$ (admitido). CdSe: banda prohibida $E_g = \qty{1.74}{eV}$, $m_e^\ast =
0.13\,m_e$, $m_h^\ast = 0.45\,m_e$ (el ``hueco'' que deja el electrón
excitado también queda confinado). (a) Energía del fotón emitido $E_g + E_e + E_h$
y \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} para $R = \qty{2.0}{nm}$, \qty{3.0}{nm} y \qty{4.0}{nm}; colores.
(b) Radio para emitir a \qty{520}{nm} (verde). (c) ¿Por qué el CdSe
masivo emite solo a \qty{713}{nm}? (d) Una dispersión de radios del
\qty{10}{\percent} en un lote: dispersión de \omterm{prop:b1:wave-propagation:sinusoidal}{longitudes de onda} a \qty{3}{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:12}
\emph{Energía de punto cero.} (a) Para un \omterm{def:b1:oscillators-resonance:harmonic}{oscilador armónico} $E = p^2/2m +
\tfrac12m\omega^2x^2$, tómese $\Delta x\,\Delta p = \hbar/2$ y minimícese $E \approx
\Delta p^2/2m + \tfrac12m\omega^2\Delta x^2$ respecto de $\Delta x$: muéstrese que $E_{\min} = \hbar\omega/2$
(la energía fundamental exacta). (b) La molécula de H$_2$ vibra a $\omega =
\qty{8.3e14}{rad/s}$: energía de punto cero en eV y temperatura más baja
a la que su vibración es ``térmica'' ($k_BT \sim \hbar\omega$).
(c) El helio líquido no se congela nunca a presión atmosférica: estímese
la energía de punto cero de un átomo de helio localizado en $\Delta x = \qty{0.05}{nm}$
(una fracción de la distancia interatómica) en un cristal, y compárese
con su energía de enlace en el sólido, de aproximadamente \qty{1}{meV}. (d) Compárese la fórmula de la caja $E_1 = h^2/8mL^2$ con la
cota de indeterminación para $\Delta x = L/2$: ¿cuál es mayor y por qué ha de serlo?
\end{exercise}

\section{Problema: el espectro del hidrógeno y un punto cuántico}

\begin{problem}[{Problema de fin de semana --- medir $h$ con luz y un
voltímetro, construir el átomo de hidrógeno a partir de tres constantes y
elegir con el tamaño el color de un cristal}]
\label{pb:b1:quantum-introduction:1}
Constantes: $h = \qty{6.626e-34}{J.s}$, $\hbar = \qty{1.055e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$, $e = \qty{1.602e-19}{C}$, $m_e = \qty{9.11e-31}{kg}$,
$1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$, $hc = \qty{1240}{eV.nm}$.

\medskip
\noindent\textbf{Parte I --- La medida de la \omterm{thm:b1:quantum-introduction:photon}{constante de Planck}.}
Se ilumina un fotocátodo de potasio con luz monocromática y se mide el
\omterm{prop:b1:quantum-introduction:photoelectric}{potencial de frenado}: \qty{0.30}{V} a \qty{6.0e14}{Hz}, \qty{0.69}{V} a
\qty{7.0e14}{Hz}, \qty{1.13}{V} a \qty{8.0e14}{Hz}, \qty{1.50}{V} a
\qty{9.0e14}{Hz} y \qty{2.36}{V} a \qty{1.1e15}{Hz}.
\begin{enumerate}
  \item Descríbase la célula y explíquese por qué una tensión inversa
        puede cortar la corriente y qué mide $eV_s$.
  \item Escríbase la relación de Einstein y dígase qué dan la pendiente y
        la ordenada en el origen de $V_s(\nu)$.
  \item A partir de los datos (por ajuste lineal o con los dos puntos
        extremos), dedúzcase $h$; compárese con el valor aceptado.
  \item \omterm{prop:b1:quantum-introduction:photoelectric}{Trabajo de extracción} del potasio en eV; frecuencia y \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de
        onda} umbral; ¿podría arrancar electrones un puntero láser rojo?
  \item Se duplica la intensidad de la lámpara: ¿qué le ocurre a $V_s$? ¿Y
        a la corriente? ¿Por qué es esto incompatible con una imagen
        puramente ondulatoria?
  \item Estimación clásica: una luz de \qty{1}{\micro W/m^2} incide sobre el
        metal, y un átomo ofrece un área de unos \qty{1e-20}{m^2}: ¿cuánto
        tardaría en acumular \qty{2.2}{eV} sobre un átomo? ¿Qué se observa
        en cambio?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El \omterm{prop:b1:quantum-introduction:hydrogen}{átomo de hidrógeno} de Bohr.}
\begin{enumerate}[resume]
  \item Un electrón en una \omterm{prop:b1:central-forces:effective}{órbita circular} de radio $r$ alrededor de un
        protón fijo: rapidez $v(r)$ y \omterm{thm:b1:work-and-energy:em}{energía mecánica} $E(r)$.
  \item Condición de Bohr $m_evr = n\hbar$: muéstrese que $r_n = n^2a_0$ y
        dese $a_0 = 4\pi\varepsilon_0\hbar^2/m_ee^2$ numéricamente.
  \item $E_n = E_1/n^2$ con $E_1 = -m_ee^4/[2(4\pi\varepsilon_0)^2\hbar^2]$;
        valor numérico en eV.
  \item Rapidez en la primera órbita y cociente $v_1/c$ (la constante de
        estructura fina $\alpha \approx 1/137$).
  \item \omterm{prop:b1:wave-propagation:sinusoidal}{Longitudes de onda} de las rayas $2 \to 1$ (Lyman $\alpha$), $3 \to 2$ y
        $4 \to 2$ (Balmer H$\alpha$, H$\beta$): ¿cuáles son visibles?
  \item Muéstrese que $1/\lambda = R_H(1/n^2 - 1/p^2)$ y calcúlese la
        constante de Rydberg $R_H$; la \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} más corta de Balmer.
  \item Energía de ionización del hidrógeno desde el \omterm{prop:b1:quantum-introduction:box}{estado fundamental};
        \omterm{prop:b1:wave-propagation:sinusoidal}{longitud de onda} del fotón que apenas lo ioniza.
  \item De Broglie: muéstrese que la circunferencia de la órbita $n$ mide
        $n$ \omterm{prop:b1:wave-propagation:sinusoidal}{longitudes de onda} --- la condición de Bohr es una condición de
        \omterm{thm:b1:wave-propagation:standing}{onda estacionaria}.
  \item Dos cosas en las que falla el modelo, una en la que acierta
        exactamente, y qué sustituye a la órbita en la teoría completa.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El \omterm{ex:b1:quantum-introduction:dot}{punto cuántico}.}
Un nanocristal de CdSe de radio $R$ confina un electrón excitado
(masa efectiva $m_e^\ast = 0.13\,m_e$) y el hueco que deja
($m_h^\ast = 0.45\,m_e$); la energía de confinamiento de cada uno en una
esfera de radio $R$ vale $h^2/8m^\ast R^2$ (admitido), y la banda prohibida
del CdSe masivo es $E_g = \qty{1.74}{eV}$.
\begin{enumerate}[resume]
  \item Dedúzcanse las energías $E_n = n^2h^2/8mL^2$ de una partícula en
        una caja unidimensional a partir de la condición de \omterm{thm:b1:wave-propagation:standing}{onda
        estacionaria}, y coméntese la analogía con la fórmula de la esfera.
  \item Energías de confinamiento del electrón y del hueco para
        $R = \qty{2.0}{nm}$.
