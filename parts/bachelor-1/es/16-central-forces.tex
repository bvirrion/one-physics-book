\chapter{Fuerzas centrales: planetas y satélites}\label{ch:b1:central-forces}

Treinta y un satélites dan dos vueltas al día a la Tierra, cada uno un
reloj y una radio, y un teléfono que oye a cuatro de ellos sabe dónde está
con un error de pocos metros. El cometa Halley cae hacia el Sol durante
treinta y ocho años, lo rodea en unas semanas y vuelve a alejarse otros
treinta y ocho. Una sonda con destino a Marte sale de la Tierra
exactamente con la rapidez que hará que su \omterm{def:b1:point-kinematics:frame}{trayectoria} roce la órbita de
Marte. Todo eso es el movimiento de un cuerpo atraído hacia un punto fijo
por una fuerza que decrece como el inverso del cuadrado de la distancia, y
todo se sigue de dos magnitudes conservadas --- la energía y el \omterm{def:b1:angular-momentum:L}{momento
angular} --- más una única idea nueva, el potencial efectivo. Este capítulo
deduce las formas y las rapideces de las órbitas, las tres \omterm{thm:b1:central-forces:conics}{leyes de Kepler}
y la aritmética de poner un satélite donde se quiere.

\section{Fuerzas centrales conservativas}

\begin{definition}[Fuerza central; fuerza central conservativa]\label{def:b1:central-forces:central}
\index{fuerza central}\index{energía potencial efectiva}
Una fuerza es \emph{central} con centro $O$ si está siempre dirigida según
$\vect{OM}$: $\vect F = F(M)\,\vect e_r$. Es \emph{central conservativa}
si su módulo depende solo de $r = OM$: $\vect F =
F(r)\,\vect e_r$, que deriva de una \omterm{def:b1:work-and-energy:conservative}{energía potencial} $E_p(r)$ con
$F(r) = -\dd E_p/\dd r$.
\end{definition}

\begin{theorem}[Movimiento bajo una fuerza central conservativa]\label{thm:b1:central-forces:reduction}
\index{ley de las áreas}
Un \omterm{def:b1:point-kinematics:frame}{punto material} de masa $m$ sometido únicamente a una \omterm{def:b1:central-forces:central}{fuerza central} conservativa:
\begin{enumerate}
  \item conserva su \omterm{def:b1:angular-momentum:L}{momento angular} $\vect L_O$; su movimiento es plano
        y, en \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares} de ese plano, $r^2\dot\theta = C$ es
        constante (\omterm{cor:b1:angular-momentum:central}{ley de las áreas}, \cref{cor:b1:angular-momentum:central});
  \item conserva su \omterm{thm:b1:work-and-energy:em}{energía mecánica} $E_m = \tfrac12 m(\dot r^2 + r^2\dot\theta^2)
        + E_p(r)$, que puede escribirse
        \[
        E_m = \tfrac12 m\dot r^2 + E_{p,\mathrm{eff}}(r), \qquad
        E_{p,\mathrm{eff}}(r) = E_p(r) + \frac{mC^2}{2r^2} = E_p(r) + \frac{L^2}{2mr^2} .
        \]
\end{enumerate}
El movimiento radial es, pues, el de un punto unidimensional en el
\emph{potencial efectivo} $E_{p,\mathrm{eff}}$: los \omterm{def:b1:work-and-energy:turning}{puntos de retroceso}
radiales son las raíces de $E_{p,\mathrm{eff}}(r) = E_m$, el movimiento es
ligado si queda atrapado entre dos de ellos, y circular si $r$ se sitúa en
un mínimo de $E_{p,\mathrm{eff}}$.
\end{theorem}

\begin{proof}
La parte 1 es \cref{cor:b1:angular-momentum:central}. La fuerza es
conservativa y no actúa ninguna otra, luego $E_m$ se conserva; en el
plano, $v^2 = \dot r^2 + r^2\dot\theta^2$ y $r^2\dot\theta^2 = C^2/r^2$.
El término $mC^2/2r^2$, \omterm{thm:b1:work-and-energy:ke}{energía cinética} del movimiento angular, actúa
como una \emph{\omterm{prop:b1:central-forces:effective}{barrera centrífuga}} repulsiva en el problema radial, al que
se aplica el análisis unidimensional de \cref{ch:b1:work-and-energy}.
\end{proof}

\section{Gravitación newtoniana}

\begin{definition}[Gravitación universal]\label{def:b1:central-forces:gravitation}
\index{gravitación}\index{energía potencial gravitatoria}
Una masa puntual $M$ situada en $O$ atrae a una masa puntual $m$ en $M$ con
\[
\vect F = -\frac{GMm}{r^2}\,\vect e_r, \qquad E_p = -\frac{GMm}{r} \quad (E_p \to 0 \text{ en el infinito}),
\]
donde $G = \qty{6.67e-11}{N.m^2/kg^2}$. Un cuerpo de simetría esférica
atrae fuera de sí mismo como si toda su masa estuviera en su centro
(admitido; es consecuencia del teorema de Gauss,
\cref{ch:b1:electrostatics-gauss}).
\end{definition}

\begin{proposition}[El potencial efectivo de la gravitación]\label{prop:b1:central-forces:effective}
\index{barrera centrífuga}\index{órbita circular}
Para $E_p = -GMm/r$,
\[
E_{p,\mathrm{eff}}(r) = -\frac{GMm}{r} + \frac{L^2}{2mr^2}
\]
tiende a $+\infty$ cuando $r \to 0$ (si $L \neq 0$), a $0^-$ en el
infinito, y tiene un único mínimo en $r_c = L^2/GMm^2$, de valor
$-G^2M^2m^3/2L^2$. Por tanto: $E_m < 0$ --- \omterm{def:b1:work-and-energy:turning}{movimiento ligado} entre un
radio mínimo y uno máximo (perigeo y apogeo); $E_m \geq 0$ --- no ligado,
el cuerpo viene del infinito y vuelve a él; $E_m$ igual al mínimo ---
\omterm{prop:b1:central-forces:effective}{órbita circular} de radio $r_c$.
\end{proposition}

\begin{proof}
Los límites, por inspección; $E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$
en $r_c$; sustitúyase. El caso circular es $\dot r = 0$ para siempre, solo
posible en el mínimo.
\end{proof}

\begin{theorem}[Órbitas circulares]\label{thm:b1:central-forces:circular}
\index{rapidez orbital}\index{tercera ley de Kepler}
En una \omterm{prop:b1:central-forces:effective}{órbita circular} de radio $r$ en torno a una masa $M$:
\[
v = \sqrt{\frac{GM}{r}}, \qquad
T = 2\pi\sqrt{\frac{r^3}{GM}}, \qquad
E_m = -\frac{GMm}{2r} = -E_k = \tfrac12 E_p .
\]
En particular, $T^2/r^3 = 4\pi^2/GM$ es el mismo para todas las \omterm{prop:b1:central-forces:effective}{órbitas
circulares} en torno al mismo centro (\omterm{thm:b1:central-forces:circular}{tercera ley de Kepler}).
\end{theorem}

\begin{proof}
El \omterm{cor:b1:point-kinematics:circular}{movimiento circular} uniforme necesita la \omterm{prop:b1:newton-dynamics:centripetal}{fuerza centrípeta} $mv^2/r = GMm/r^2$;
$T = 2\pi r/v$; $E_k = \tfrac12 mv^2 = GMm/2r$ y $E_p = -GMm/r$.
\end{proof}

\begin{example}[Tres órbitas]\label{ex:b1:central-forces:orbits}
$GM_\oplus = \qty{3.99e14}{m^3/s^2}$. Órbita baja ($r = \qty{6770}{km}$,
a \qty{400}{km} de altura): $v = \qty{7.7}{km/s}$, $T = \qty{92}{min}$.
Geoestacionaria ($T = \qty{86164}{s}$, un día sidéreo): $r = (GMT^2/4\pi^2)^{1/3} =
\qty{42200}{km}$, $v = \qty{3.1}{km/s}$. La Luna ($r = \qty{384000}{km}$):
$T = \qty{27.4}{d}$ --- y su \omterm{cor:b1:point-kinematics:circular}{aceleración centrípeta} $v^2/r =
\qty{2.7e-3}{m/s^2}$ vale $g/3600$ a sesenta radios terrestres: el
cuadrado inverso, comprobado por el propio Newton.
\end{example}

\section{Las leyes de Kepler y la forma de las órbitas}

\begin{theorem}[Trayectorias en un campo newtoniano]\label{thm:b1:central-forces:conics}
\index{leyes de Kepler}\index{cónica}\index{elipse (órbita)}\index{excentricidad}
La \omterm{def:b1:point-kinematics:frame}{trayectoria} de un cuerpo en el campo de una masa fija $M$ es una cónica
con $O$ en uno de sus focos: una elipse si $E_m < 0$, una parábola si
$E_m = 0$ y una hipérbola si $E_m > 0$. Para la elipse de semieje mayor
$a$ y excentricidad $e$:
\begin{itemize}
  \item (Kepler I) el centro de fuerzas está en un foco; el perigeo y el
        apogeo se sitúan en $r_p = a(1 - e)$ y $r_a = a(1 + e)$;
  \item (Kepler II) la \omterm{cor:b1:angular-momentum:central}{velocidad areolar} es constante (\omterm{cor:b1:angular-momentum:central}{ley de las áreas});
  \item (Kepler III) el periodo cumple $T^2 = 4\pi^2a^3/GM$;
  \item la energía depende solo de $a$, $E_m = -GMm/2a$, de modo que la
        rapidez a la distancia $r$ viene dada por
        \[
        v^2 = GM\left(\frac{2}{r} - \frac{1}{a}\right) .
        \]
\end{itemize}
\end{theorem}
\admitted

\begin{remark}[Qué se demuestra y qué se admite]\label{rem:b1:central-forces:admitted}
Kepler II y la clasificación por la energía se demostraron más arriba. Que
las órbitas ligadas sean exactamente elipses con $O$ en un foco, junto con
Kepler III y $E_m = -GMm/2a$ en general, exige integrar la ecuación radial
(lo hace el volumen del Año 2); para las \omterm{prop:b1:central-forces:effective}{órbitas circulares} ($a = r$,
$e = 0$) todos los enunciados se reducen a \cref{thm:b1:central-forces:circular},
que es la comprobación posible aquí. La relación $v^2 = GM(2/r - 1/a)$ se
sigue de $E_m = \tfrac12 mv^2 - GMm/r = -GMm/2a$ una vez admitida esa
fórmula de la energía.
\end{remark}

\begin{example}[El cometa Halley]\label{ex:b1:central-forces:halley}
Perihelio \qty{0.586}{AU}, afelio \qty{35.1}{AU}: $a = \qty{17.8}{AU}$,
$e = 0.967$, $T = \sqrt{17.8^3} = 75$ años (Kepler III en unidades
astronómicas y años, donde $T^2 = a^3$ para el Sol). Con
$GM_\odot = \qty{1.33e20}{m^3/s^2}$, la rapidez en el perihelio vale
$\sqrt{GM(2/r_p - 1/a)} = \qty{54}{km/s}$ y en el afelio \qty{0.9}{km/s}
--- sesenta veces menor, como exige la \omterm{cor:b1:angular-momentum:central}{ley de las áreas} ($r_pv_p = r_av_a$).
\end{example}

\section{Satélites}

\begin{proposition}[Velocidad de escape; órbita geoestacionaria]\label{prop:b1:central-forces:escape}
\index{velocidad de escape}\index{órbita geoestacionaria}
Desde la superficie de un cuerpo de masa $M$ y radio $R$, la rapidez
mínima de lanzamiento para llegar al infinito es
$v_e = \sqrt{2GM/R} = \sqrt2\,v_c$, donde $v_c = \sqrt{GM/R}$ es la
rapidez circular a ras de suelo: \qty{11.2}{km/s} para la Tierra. Un
satélite \emph{geoestacionario} orbita en el plano ecuatorial con el
periodo sidéreo terrestre, a $r = \qty{42200}{km}$ (\qty{35800}{km} sobre
el suelo).
\end{proposition}

\begin{proof}
$E_m \geq 0$: $\tfrac12 mv_e^2 - GMm/R = 0$. Kepler III con $T = \qty{86164}{s}$.
\end{proof}

\begin{proposition}[Transferencia de Hohmann]\label{prop:b1:central-forces:hohmann}
\index{transferencia de Hohmann}
Para llevar un satélite de una \omterm{prop:b1:central-forces:effective}{órbita circular} de radio $r_1$ a otra
circular y coplanaria de radio $r_2 > r_1$ con dos impulsos breves, se
enciende el motor en $r_1$ para entrar en la \emph{elipse de
transferencia} de perigeo $r_1$ y apogeo $r_2$ ($a = (r_1 + r_2)/2$), se
deja derivar medio periodo y se vuelve a encender en $r_2$ para
circularizar:
\[
\Delta v_1 = \sqrt{\frac{GM}{r_1}}\left(\sqrt{\frac{2r_2}{r_1 + r_2}} - 1\right),
\qquad
\Delta v_2 = \sqrt{\frac{GM}{r_2}}\left(1 - \sqrt{\frac{2r_1}{r_1 + r_2}}\right),
\qquad
t = \pi\sqrt{\frac{(r_1 + r_2)^3}{8GM}} .
\]
\end{proposition}

\begin{proof}
Las rapideces sobre la elipse de transferencia, a partir de
$v^2 = GM(2/r - 1/a)$ en $r_1$ y en $r_2$; réstense las rapideces
circulares; medio periodo de \cref{thm:b1:central-forces:conics}.
\end{proof}

\begin{example}[De la órbita baja a la geoestacionaria]\label{ex:b1:central-forces:leogeo}
$r_1 = \qty{6770}{km}$, $r_2 = \qty{42200}{km}$: $\Delta v_1 = \qty{2.4}{km/s}$,
$\Delta v_2 = \qty{1.5}{km/s}$, con \qty{5.3}{h} de deriva. El total,
\qty{3.9}{km/s}, es la mitad de lo que ya costó el lanzamiento a la órbita
baja: la ``última milla'' hasta una órbita alta sale cara, y los satélites
llevan su propio motor para recorrerla.
\end{example}

\begin{remark}[Centros repulsivos]\label{rem:b1:central-forces:rutherford}
\index{dispersión de Rutherford}
Para una fuerza repulsiva en cuadrado inverso (dos cargas del mismo signo,
\cref{ch:b1:electrostatics-gauss}) el potencial efectivo no tiene pozo:
toda \omterm{def:b1:point-kinematics:frame}{trayectoria} es una hipérbola, la partícula se acerca, se desvía y se
marcha. Al apuntar una partícula $\alpha$ directamente a un núcleo de oro,
su distancia de máxima aproximación se sigue solo de la energía, $E_k =
2Ze^2/4\pi\varepsilon_0 r_{\min}$ --- la manera que tuvo Rutherford de acotar
el tamaño del núcleo (\cref{exo:b1:central-forces:11}).
\end{remark}

y que las \omterm{prop:b1:work-and-energy:stability}{pequeñas oscilaciones} radiales en torno a él tienen \omterm{def:b1:wave-propagation:signal}{frecuencia
angular} $\sqrt{GM/r_c^3}$ --- la propia frecuencia orbital. ¿Qué dice esto
de una \omterm{prop:b1:central-forces:effective}{órbita circular} ligeramente perturbada?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:9}
Los astronautas a bordo de la ISS flotan. Explíquese con las fuerzas;
¿cuánto vale $g$ a su altura y por qué no contradice el hecho de que
floten?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:10}
La ``prueba de la Luna'' de Newton: calcúlese la \omterm{cor:b1:point-kinematics:circular}{aceleración centrípeta} de
la Luna a partir de $r = \qty{3.84e8}{m}$ y $T = \qty{27.3}{d}$, y
compárese con $g(R/r)^2$. Conclúyase.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:11}
Una partícula $\alpha$ ($q = 2e$, $E_k = \qty{5.0}{MeV}$) se dispara
directamente contra un núcleo de oro ($Z = 79$), supuesto fijo. Con la
conservación de la energía y $E_p = 2Ze^2/4\pi\varepsilon_0r$
($1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$), hállese la distancia de máxima
aproximación; compárese con el radio nuclear, de unos \qty{7}{fm}. ¿Qué
energía haría falta para tocar el núcleo?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:12}
La atmósfera residual frena a un satélite en órbita baja, de modo que su
\omterm{thm:b1:work-and-energy:em}{energía mecánica} disminuye. Muéstrese, con $E_m = -GMm/2r$ y
$v = \sqrt{GM/r}$, que entonces se mueve \emph{más deprisa}. ¿Adónde va la
energía y qué fracción de la \omterm{def:b1:work-and-energy:conservative}{energía potencial} perdida se convierte en calor?
\end{exercise}

\section{Problema: los satélites que dicen dónde estás}

\begin{problem}[{Problema de fin de semana --- una constelación de
navegación a veinte mil kilómetros de altura: su órbita, el combustible
para llegar, cuántos satélites ve un teléfono y por qué hay que ordenar a
sus relojes que atrasen}]
\label{pb:b1:central-forces:1}
Datos: $GM_\oplus = \qty{3.986e14}{m^3/s^2}$, $R_\oplus = \qty{6371}{km}$,
día sidéreo \qty{86164}{s}, $c = \qty{2.998e8}{m/s}$. Los satélites dan la
vuelta a la Tierra en exactamente medio día sidéreo, en \omterm{prop:b1:central-forces:effective}{órbitas circulares}.

\medskip
\noindent\textbf{Parte I --- La órbita.}
\begin{enumerate}
  \item Calcúlense el radio orbital y la altura.
  \item Calcúlese la \omterm{thm:b1:central-forces:circular}{rapidez orbital}.
  \item Calcúlese la aceleración del satélite y compárese con $g$ en la
        superficie.
  \item Calcúlense la \omterm{thm:b1:work-and-energy:em}{energía mecánica} por kilogramo y sus partes cinética
        y potencial.
  \item ¿Por qué elegir medio día sidéreo y no medio día solar? ¿Qué ve
        repetirse un observador en el suelo?
  \item Visto desde un satélite, ¿qué ángulo subtiende el disco terrestre?
  \item Compárese el radio de la órbita con el de la geoestacionaria:
        ¿cuál es más alta, y en qué factor?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Llegar hasta allí.}
\begin{enumerate}[resume]
  \item Energía por kilogramo para pasar del reposo en la superficie a la
        órbita final (despréciense el aire y la rotación terrestre).
  \item El lanzador alcanza primero una \omterm{prop:b1:central-forces:effective}{órbita circular} baja a
        \qty{400}{km}. ¿Rapidez allí?
  \item \omterm{prop:b1:central-forces:hohmann}{Transferencia de Hohmann} desde esa órbita hasta la final: los dos
        impulsos y su suma.
  \item Duración de la transferencia.
  \item Compárese la suma de los impulsos con la \omterm{prop:b1:central-forces:escape}{velocidad de escape} desde
        la órbita baja: ¿cuán lejos está esta misión de ``abandonar la Tierra''?
  \item La etapa superior y el satélite tienen una masa de \qty{2000}{kg}
        antes de la transferencia; la velocidad de eyección del motor es
        \qty{3.0}{km/s} y la ecuación del cohete da $\Delta v = u\ln(m_0/m_1)$.
        ¿Propulsante necesario para los dos impulsos?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Cobertura y sincronización.}
\begin{enumerate}[resume]
  \item Un satélite es visible desde todo punto de la Tierra situado
        dentro del cono de semiángulo $\theta$ con $\cos\theta = R/r$ en
        torno a la línea Tierra--satélite. Calcúlense $\theta$ y la
        fracción $(1 - \cos\theta)/2$ de la superficie terrestre que cubre.
  \item Con $24$ satélites repartidos uniformemente, ¿cuántos son visibles
        en promedio desde un punto? ¿Es verosímil el mínimo de $4$
        necesario para determinar la posición?
  \item Tiempo de propagación de la luz desde un satélite en el cenit; y
        en el horizonte.
  \item Para situar el receptor con un error de \qty{3}{m}, ¿con qué
        precisión hay que medir los tiempos de las \omterm{def:b1:wave-propagation:signal}{señales}?
  \item El reloj del propio receptor es un cuarzo barato, con errores de
        milisegundos: ¿por qué hacen falta cuatro satélites y no tres?
  \item Los relojes de los satélites son atómicos. Dese la estabilidad
        relativa necesaria para un error de \qty{3}{m} al cabo de un día.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La relatividad, brevemente.}
Dos efectos desplazan los relojes de los satélites respecto de los del
suelo (fórmulas admitidas, del volumen del Año 3): el movimiento atrasa un
reloj en la fracción $v^2/2c^2$; la altura lo adelanta en $GM(1/R - 1/r)/c^2$.
\begin{enumerate}[resume]
  \item Calcúlese el atraso debido a la rapidez, en microsegundos por día.
  \item Calcúlese el adelanto debido a la altura, en microsegundos por
        día.
  \item Efecto neto y error de posición que causaría al día si no se
        corrigiese.
  \item Los relojes de los satélites se ajustan antes del lanzamiento a
        \qty{10.22999999543}{MHz} en lugar de \qty{10.23}{MHz}.
        Compruébese que eso compensa el efecto neto.
  \item Las órbitas no son exactamente keplerianas: nómbrense tres
        perturbaciones y dígase cómo las afronta el sistema.
  \item Resúmase la cadena: dos leyes de conservación, una órbita, una
        constelación y la única corrección que nadie habría adivinado.
\end{enumerate}
\end{problem}
