\chapter{Sistemas de puntos; introducción al sólido rígido}\label{ch:b1:systems-of-points}

Una saltadora abandona el trampolín, se encoge, da dos volteretas y se
estira; a lo largo de todo ello, un punto de su cuerpo --- su \omterm{def:b1:systems-of-points:cm}{centro de
masas} --- traza la misma parábola sencilla que trazaría una piedra. Un
yoyó soltado de la mano tarda tres veces más que una piedra en llegar al
final de su cuerda y llega girando a dos mil vueltas por minuto. Una
esfera maciza gana a una hueca por cualquier plano inclinado, sean cuales
sean sus tamaños y sus masas. La mecánica de un solo punto, desarrollada
en los capítulos anteriores, se extiende a sistemas de muchos puntos ---
y a los \omterm{def:b1:systems-of-points:solid}{sólidos rígidos} de la vida diaria --- mediante un puñado de
teoremas: el \omterm{def:b1:systems-of-points:cm}{centro de masas} se mueve como un punto, la energía y el
\omterm{def:b1:angular-momentum:L}{momento angular} se parten en una contribución del \omterm{def:b1:systems-of-points:cm}{centro de masas} y otra
``en torno'' a él, y un cuerpo que gira sobre un eje fijo es un sistema
de un grado de libertad gobernado por su \omterm{prop:b1:angular-momentum:inertia}{momento de inercia}.

\section{Centro de masas y cantidad de movimiento}

\begin{definition}[Centro de masas; cantidad de movimiento de un sistema]\label{def:b1:systems-of-points:cm}
\index{centro de masas}\index{baricentro}
Para \omterm{def:b1:point-kinematics:frame}{puntos materiales} de masas $m_i$ situados en $M_i$, con masa total
$M = \sum m_i$, el \emph{centro de masas} (baricentro) $G$ se define por
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{es decir}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
sea cual sea el origen $O$. La \emph{cantidad de movimiento} del sistema
es $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[Teorema del centro de masas]\label{thm:b1:systems-of-points:cmtheorem}
\index{teorema del centro de masas}\index{conservación de la cantidad de movimiento}
En un \omterm{thm:b1:newton-dynamics:laws}{sistema inercial},
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
el \omterm{def:b1:systems-of-points:cm}{centro de masas} se mueve como un \omterm{def:b1:point-kinematics:frame}{punto material} de masa $M$ sometido
solo a la resultante de las fuerzas \emph{exteriores}. Para un \omterm{def:b1:newton-dynamics:conservation}{sistema
cerrado} sin fuerza exterior, $\vect P$ se conserva y $G$ se mueve
uniformemente.
\end{theorem}

\begin{proof}
Súmese la segunda ley de Newton sobre todos los puntos; las fuerzas
interiores se cancelan por parejas de \omterm{thm:b1:newton-dynamics:laws}{acción y reacción}; y
$\sum m_i\vect a_i = M\vect a_G$ al derivar dos veces la definición de $G$.
\end{proof}

\begin{example}[Saltadora, retroceso, choque]\label{ex:b1:systems-of-points:cm}
\index{choque}
El $G$ de la saltadora sigue una parábola bajo su solo peso, gire como
gire --- sus músculos son fuerzas interiores. Un fusil de \qty{4.0}{kg}
que dispara una bala de \qty{10}{g} a \qty{800}{m/s}: $\vect P = \vect 0$
antes y después, así que el fusil retrocede a \qty{2.0}{m/s}. Dos coches,
uno de \qty{1000}{kg} a \qty{20}{m/s} y otro de \qty{1500}{kg} en reposo,
que quedan enganchados: $v =
20000/2500 = \qty{8.0}{m/s}$ --- cantidad de movimiento conservada,
mientras que el $60\%$ de la \omterm{thm:b1:work-and-energy:ke}{energía cinética} se ha ido en chapa abollada
y calor. La cantidad de movimiento se conserva en todo choque; la \omterm{thm:b1:work-and-energy:ke}{energía
cinética}, solo en los \emph{elásticos}
(\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{Los teoremas de König}

\begin{definition}[Sistema baricéntrico]\label{def:b1:systems-of-points:barycentric}
\index{sistema baricéntrico}
El \emph{sistema baricéntrico} $\mathcal R^*$ tiene su origen en $G$ y está
en traslación respecto del \omterm{thm:b1:newton-dynamics:laws}{sistema inercial} $\mathcal R$ (sus ejes conservan
direcciones fijas). Las magnitudes medidas en él llevan asterisco: $\vect v_i^*
= \vect v_i - \vect v_G$, y $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[Teoremas de König]\label{thm:b1:systems-of-points:koenig}
\index{teoremas de König}
El \omterm{def:b1:angular-momentum:L}{momento angular} respecto de un punto $O$ y la \omterm{thm:b1:work-and-energy:ke}{energía cinética} de un
sistema se parten en la contribución de su \omterm{def:b1:systems-of-points:cm}{centro de masas}, que lleva toda
la masa, y las contribuciones en el \omterm{def:b1:systems-of-points:barycentric}{sistema baricéntrico}:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
donde $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ y $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ no dependen del origen del sistema.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ y $\vect v_i = \vect v_G + \vect v_i^*$;
desarróllense $\sum\vect{OM_i}\wedge m_i\vect v_i$ y $\sum\tfrac12 m_iv_i^2$: los
términos cruzados contienen $\sum m_i\vect{GM_i} = \vect 0$ o $\sum m_i\vect v_i^* = \vect 0$
y se anulan.
\end{proof}

\begin{example}[Una rueda que rueda]\label{ex:b1:systems-of-points:rolling}
\index{rodadura sin deslizamiento}
Una rueda de masa $M$ y radio $R$, con \omterm{prop:b1:angular-momentum:inertia}{momento de inercia} $J$ respecto de
su eje, que rueda sin deslizar con rapidez $v$: el punto de contacto está
en reposo, luego $\omega = v/R$, y $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- tres cuartos de $Mv^2$ para un disco homogéneo ($J = \tfrac12 MR^2$)
y $Mv^2$ para un aro. El segundo término explica por qué las ruedas de una
bicicleta se hacen ligeras: su rotación cuesta una energía que la masa del
cuadro no cuesta.
\end{example}

\section{El problema de dos cuerpos}

\begin{proposition}[Masa reducida]\label{prop:b1:systems-of-points:twobody}
\index{problema de dos cuerpos}\index{masa reducida}
Dos \omterm{def:b1:point-kinematics:frame}{puntos materiales} que interaccionan solo entre sí (fuerza $\vect F$
sobre $m_2$ debida a $m_1$, y $-\vect F$ sobre $m_1$): $G$ se mueve
uniformemente y la posición relativa $\vect r = \vect{M_1M_2}$ cumple
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
la ecuación de un único punto ficticio de \emph{\omterm{prop:b1:systems-of-points:twobody}{masa reducida}} $\mu$ que
se mueve bajo $\vect F$ en torno a un centro fijo. En $\mathcal R^*$, $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ y $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. En $\mathcal R^*$, $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ y
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$; sustitúyase en $E_k^*$ y en $\vect L^*$.
\end{proof}

\begin{example}[Cuando el centro también se mueve]\label{ex:b1:systems-of-points:reduced}
En \cref{ch:b1:central-forces} el Sol estaba fijo: exacto solo en el
límite $m \ll M$; en general la \omterm{thm:b1:central-forces:circular}{tercera ley de Kepler} se lee $T^2 = 4\pi^2a^3/G(M +
m)$. Para la molécula de HCl que vibra en \cref{pb:b1:work-and-energy:1},
la masa que oscila en el pozo es $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$, lo que sube la frecuencia un $1.4\%$; para el átomo de
hidrógeno, la $\mu$ del electrón difiere de $m_e$ en una parte entre
$1836$ --- medible en el espectro.
\end{example}

\section{Sólido rígido que gira en torno a un eje fijo}

\begin{definition}[Sólido; rotación en torno a un eje fijo]\label{def:b1:systems-of-points:solid}
\index{sólido rígido}\index{rotación en torno a un eje fijo}
Un \emph{sólido rígido} es un sistema cuyos puntos mantienen distancias
mutuas fijas. Si gira en torno a un eje $\Delta$ fijo en el sistema, todo
punto describe una circunferencia centrada en $\Delta$ con la misma
\omterm{cor:b1:point-kinematics:circular}{velocidad angular} $\omega = \dot\theta$; un punto a distancia $r_i$ del eje
tiene rapidez $r_i\omega$.
\end{definition}

\begin{proposition}[Momento de inercia; momento angular; energía cinética]\label{prop:b1:systems-of-points:inertia}
\index{momento de inercia!de sólidos}\index{teorema de Huygens}
Con el \emph{\omterm{prop:b1:angular-momentum:inertia}{momento de inercia}} $J_\Delta = \sum m_ir_i^2$ (una integral
$\int r^2\,\dd m$ para un cuerpo continuo):
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
Valores habituales (masa $M$): anillo delgado o cilindro hueco respecto de
su eje, $MR^2$; disco homogéneo o cilindro macizo, $\tfrac12 MR^2$; varilla
delgada de longitud $L$ respecto de su centro, $\tfrac{1}{12}ML^2$, y
respecto de un extremo, $\tfrac13 ML^2$; esfera maciza respecto de un
diámetro, $\tfrac25 MR^2$. \emph{\omterm{prop:b1:systems-of-points:inertia}{Teorema de Huygens}}: respecto de un eje
$\Delta$ paralelo a un eje $\Delta_G$ que pasa por $G$ a distancia $d$,
$J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
$L_\Delta$ y $E_k$: \cref{prop:b1:angular-momentum:inertia}. Varilla
respecto de un extremo: $\int_0^L x^2(M/L)\,\dd x = ML^2/3$; respecto del
centro, $\int_{-L/2}^{L/2}
x^2(M/L)\dd x = ML^2/12$. Disco: anillo de radio $r$ y anchura $\dd r$, de
masa $(2Mr/R^2)\dd r$: $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$; el anillo y la
esfera se admiten (la esfera exige una integral doble). Huygens: con
$r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2 = r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$, el término cruzado suma cero por definición de $G$.
\end{proof}

\begin{theorem}[Dinámica de un sólido en torno a un eje fijo]\label{thm:b1:systems-of-points:rotation}
\index{pivote}\index{potencia de un momento}
Para un sólido que gira en torno a un eje fijo $\Delta$ en un \omterm{thm:b1:newton-dynamics:laws}{sistema inercial},
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
los momentos de las fuerzas exteriores respecto del eje mandan sobre la
aceleración angular, y un momento $\mathcal M_\Delta$ entrega la potencia
$\mathcal M_\Delta\omega$. Un \emph{pivote} ideal (cojinete sin rozamiento) no
ejerce momento alguno respecto de su eje y, por tanto, ni acelera ni frena
la rotación.
\end{theorem}

\begin{proof}
\cref{thm:b1:angular-momentum:system} proyectado sobre $\Delta$, con $L_\Delta
= J_\Delta\dot\theta$ y $J_\Delta$ constante para un \omterm{def:b1:systems-of-points:solid}{sólido rígido};
multiplíquese por $\dot\theta$ para la forma energética (las fuerzas
interiores de un \omterm{def:b1:systems-of-points:solid}{sólido rígido} no trabajan: las distancias no cambian).
\end{proof}

\begin{example}[Péndulo compuesto; péndulo de torsión]\label{ex:b1:systems-of-points:pendulums}
\index{péndulo compuesto}\index{péndulo de torsión}
Un sólido de masa $M$ que pivota en torno a un eje horizontal a distancia
$d$ de $G$: el momento del peso vale $-Mgd\sin\theta$, luego $J_\Delta\ddot\theta =
-Mgd\sin\theta$ y las oscilaciones pequeñas tienen el periodo
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
el de un \omterm{ex:b1:newton-dynamics:pendulum}{péndulo simple} de longitud $J_\Delta/Md$ --- para una varilla
homogénea que pivota por un extremo, $2L/3$. Un disco colgado de un hilo
de constante de torsión $C$ (momento recuperador $-C\theta$):
$J\ddot\theta = -C\theta$, $T = 2\pi\sqrt{J/C}$ --- la balanza de torsión
que midió $G$ y la ley de Coulomb.
\end{example}

\begin{proposition}[Rodadura sin deslizamiento por un plano inclinado]\label{prop:b1:systems-of-points:rollingincline}
Un sólido de revolución ($J = kMR^2$ respecto de su eje) soltado en un
plano inclinado de ángulo $\alpha$ rueda sin deslizar con la aceleración
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
$\tfrac57 g\sin\alpha$ para una esfera maciza, $\tfrac23 g\sin\alpha$ para un
disco y $\tfrac12 g\sin\alpha$ para un aro --- sin depender de la masa ni del radio.
\end{proposition}

\begin{proof}
Energía: $Mgh = \tfrac12 Mv^2(1 + k)$ a lo largo de la pendiente (el
rozamiento estático en el punto de contacto, que está en reposo, no
trabaja), luego $v^2 = 2ah$ con la $a$ indicada; o bien Newton para $G$ y
el teorema del momento respecto de $G$ con el rozamiento $f$:
$Ma = Mg\sin\alpha - f$, $kMR^2\dot\omega = fR$, $a = R\dot\omega$.
\end{proof}

\begin{remark}[Sistemas deformables]\label{rem:b1:systems-of-points:deformable}
\index{sistema deformable}
Para un sistema que no es rígido, el \omterm{thm:b1:work-and-energy:ke}{teorema de la energía cinética} se lee
$\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$: las fuerzas interiores
trabajan (músculos, resortes, rozamiento dentro del sistema). La
patinadora que recoge los brazos, la saltadora que se encoge y el ciclista
que se levanta sobre los pedales ganan \omterm{thm:b1:work-and-energy:ke}{energía cinética} sin trabajo
exterior --- el \omterm{thm:b1:systems-of-points:cmtheorem}{teorema del centro de masas} sigue siendo cierto, y la
energía viene de dentro.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:1}
Masas de \qty{2.0}{kg} en $x = 0$ y de \qty{3.0}{kg} en $x = \qty{1.0}{m}$:
posición de $G$. El sistema Tierra--Luna ($81:1$, \qty{384000}{km}):
distancia de $G$ al centro de la Tierra; compárese con el radio
terrestre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:2}
Un fusil de \qty{4.0}{kg} dispara una bala de \qty{10}{g} a \qty{800}{m/s}.
Rapidez de retroceso; \omterm{thm:b1:work-and-energy:ke}{energías cinéticas} de la bala y del fusil; ¿de dónde
salió la energía?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:3}
Un coche de \qty{1000}{kg} a \qty{20}{m/s} choca con otro de \qty{1500}{kg}
en reposo y quedan enganchados. Rapidez común; fracción de \omterm{thm:b1:work-and-energy:ke}{energía cinética} perdida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:4}
Dedúzcase el \omterm{prop:b1:angular-momentum:inertia}{momento de inercia} de una varilla homogénea de masa $M$ y
longitud $L$ respecto de un eje perpendicular por su extremo, y después
por su centro; compruébese entre ambos el \omterm{prop:b1:systems-of-points:inertia}{teorema de Huygens}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:5}
\omterm{thm:b1:work-and-energy:ke}{Energía cinética} de un cilindro homogéneo que rueda y de un aro que rueda,
de la misma masa y la misma rapidez, como múltiplos de $\tfrac12 Mv^2$.
¿Cuál se detiene antes en un suelo plano rugoso y cuál sube más por una pendiente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:6}
\omterm{prop:b1:systems-of-points:twobody}{Masa reducida} del HCl ($m_{Cl} = 35m_H$) y del átomo de hidrógeno
($m_p = 1836m_e$); corrección relativa a una frecuencia de vibración o
de órbita calculada solo con la masa ligera.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:7}
Una varilla homogénea de \qty{1.0}{m} pivota libremente por un extremo.
Periodo de las \omterm{prop:b1:work-and-energy:stability}{pequeñas oscilaciones}; longitud del \omterm{ex:b1:newton-dynamics:pendulum}{péndulo simple} del
mismo periodo; el punto de la varilla (el centro de percusión) en el que
un golpe seco no produce reacción en el pivote está a esa distancia ---
¿por qué lo busca un bateador?
\end{exercise}

Un disco ($M = \qty{0.20}{kg}$, $R = \qty{10}{cm}$) cuelga de un hilo y

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:9}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:systems-of-points:10}

\begin{exercise}[$\star\star\star$]\label{exo:b1:systems-of-points:11}

\section{Problema: el yoyó}

\begin{problem}[{Problema de fin de semana --- un disco sobre un eje fino
al final de una cuerda: por qué cae tan despacio, a qué velocidad gira
abajo, por qué ``duerme'' allí y hacia dónde rueda un carrete cuando se
tira de su hilo}]
\label{pb:b1:systems-of-points:1}
Un yoyó es un disco homogéneo de masa $M = \qty{50}{g}$ y radio $R =
\qty{3.0}{cm}$ con un eje fino de radio $r = \qty{0.40}{cm}$ en torno al
cual se enrolla una cuerda de longitud $h = \qty{1.0}{m}$; el extremo
superior de la cuerda se mantiene fijo. $g = \qty{9.81}{m/s^2}$;
$J = \tfrac12 MR^2$ respecto del eje.

\medskip
\noindent\textbf{Parte I --- Cinemática y energía.}
\begin{enumerate}
  \item ¿Dónde está el \omterm{def:b1:systems-of-points:cm}{centro de masas} $G$ del yoyó? ¿Por qué?
  \item Muéstrese, con el teorema de König, que su \omterm{thm:b1:work-and-energy:ke}{energía cinética} vale $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$.
  \item Dedúzcase $J = \tfrac12 MR^2$ integrando anillo a anillo, y
        calcúlese $J$.
  \item A medida que la cuerda se desenrolla del eje sin deslizar,
        relaciónense $v_G$ y $\omega$.
  \item Dedúzcase $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ y calcúlese el
        factor entre paréntesis. Coméntese.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La caída.}
\begin{enumerate}[resume]
  \item Fuerzas sobre el yoyó; escríbase el \omterm{thm:b1:systems-of-points:cmtheorem}{teorema del centro de masas}
        (eje vertical hacia abajo).
  \item Escríbase el \omterm{thm:b1:angular-momentum:theorem}{teorema del momento angular} respecto del eje que pasa
        por $G$ (¿qué fuerzas tienen momento?).
  \item Combínese con la pregunta 4 para hallar la aceleración $a_G = g/(1 +
        R^2/2r^2)$; calcúlese.
  \item Calcúlese la tensión de la cuerda y compárese con el peso.
  \item Tiempo que tarda en desenrollarse el metro de cuerda y rapidez
        entonces; compárese con una piedra soltada desde la misma altura.
  \item \omterm{cor:b1:point-kinematics:circular}{Velocidad angular} abajo, en rad/s y en vueltas por minuto.
  \item Sepárese la \omterm{thm:b1:work-and-energy:ke}{energía cinética} abajo en sus partes de traslación y
        de rotación.
  \item Compruébese el balance energético: compárese $Mgh$ con la \omterm{thm:b1:work-and-energy:ke}{energía
        cinética} abajo y dígase por qué la tensión de la cuerda, aun siendo
        grande, no trabaja.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Abajo: el yoyó que duerme.}
Cuando la cuerda está del todo desenrollada, detiene el \omterm{def:b1:systems-of-points:cm}{centro de masas} en
unos pocos milisegundos; el yoyó sigue girando en el lazo del extremo de
la cuerda (``durmiendo'').
\begin{enumerate}[resume]
  \item ¿Por qué sobrevive el giro a la detención de $G$? (¿Qué teorema, y
        qué fuerza habría hecho falta para detenerlo?)
  \item Estímese la tensión media durante la detención si $G$ se para en
        \qty{5}{ms}.
  \item El lazo roza contra el eje con un momento de rozamiento $\Gamma =
        \qty{1.0e-4}{N.m}$: ¿cuánto tiempo duerme el yoyó?
  \item Un tirón seco hacia arriba hace que la cuerda vuelva a agarrar el
        eje; el yoyó, girando, sube entonces. Con la energía, ¿hasta qué
        altura podría subir si no se perdiera nada? ¿Por qué menos en la
        práctica?
  \item Durante la subida, ¿es la tensión mayor o menor que el peso?
        (Signo de $a_G$.)
  \item ¿Qué ocurriría con un yoyó cuyo eje tuviera radio $R$ (cuerda
        enrollada en el borde)? Calcúlese su aceleración de caída.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El carrete sobre la mesa.}
El mismo objeto se apoya en una mesa rugosa, con el eje horizontal, y su
hilo sale horizontalmente por la parte \emph{inferior} del eje; se tira
con una fuerza $F$. Rueda sin deslizar.
\begin{enumerate}[resume]
  \item En el punto de contacto $C$ con la mesa, ¿cuál es la velocidad del
        material del carrete? Dedúzcase que el movimiento es, en cada
        instante, una rotación en torno a $C$.
  \item Calcúlese el momento de $F$ respecto de $C$ (\omterm{def:b1:angular-momentum:moment}{brazo de palanca}
        $R - r$) y el del peso y el de la reacción de la mesa.
  \item Con el \omterm{thm:b1:angular-momentum:theorem}{teorema del momento angular} respecto de $C$ y $J_C = J +
        MR^2$ (Huygens), hállese $a_G$ y muéstrese que el carrete rueda
        \emph{hacia} la mano.
  \item Ahora se tira del hilo con un ángulo $\beta$ sobre la horizontal.
        Muéstrese que el carrete no rueda cuando la recta de acción de $F$
        pasa por $C$, es decir cuando $\cos\beta = r/R$, y que rueda
        alejándose de la mano con tirones más inclinados. Calcúlese ese
        ángulo.
  \item Calcúlese $a_G$ para $F = \qty{0.20}{N}$ tirando en horizontal, y
        la fuerza de rozamiento necesaria; ¿qué limita $F$ antes de que el
        carrete deslice?
  \item Resúmase: los tres teoremas usados (\omterm{def:b1:systems-of-points:cm}{centro de masas}, \omterm{def:b1:angular-momentum:L}{momento
        angular} respecto de un eje y energía) y qué decidió cada uno.
\end{enumerate}
\end{problem}
