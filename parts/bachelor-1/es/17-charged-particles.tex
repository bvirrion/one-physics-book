\chapter{Partículas cargadas en campos E y B}\label{ch:b1:charged-particles}

Dentro del osciloscopio que dibujó los registros de
\cref{ch:b1:transient-regimes}, un haz de electrones es desviado a lo
ancho de una pantalla por la tensión entre dos placas; dentro del imán de
un \omterm{prop:b1:charged-particles:spectrometer}{espectrómetro de masas}, los iones de dos isótopos se separan unos
milímetros porque uno pesa un ocho por ciento más que el otro; dentro del
ciclotrón de un hospital, los protones describen una espiral de doscientas
vueltas antes de salir a un tercio de la velocidad de la luz para quemar
un tumor. Los tres casos son la misma dinámica: una carga en un campo
uniforme, eléctrico o magnético, y la ley de Newton. Este capítulo la
desarrolla --- la parábola en un \omterm{def:b1:charged-particles:lorentz}{campo eléctrico}, la circunferencia y la
hélice en un \omterm{def:b1:charged-particles:lorentz}{campo magnético} --- y la convierte en los instrumentos que
clasifican, aceleran y miden partículas cargadas.

\section{La fuerza de Lorentz}

\begin{definition}[Fuerza de Lorentz]\label{def:b1:charged-particles:lorentz}
\index{fuerza de Lorentz}\index{campo eléctrico}\index{campo magnético}
Una partícula de carga $q$ que se mueve con velocidad $\vect v$ en un
campo eléctrico $\vect E$ y un campo magnético $\vect B$ recibe la fuerza
\[
\vect F = q\big(\vect E + \vect v\wedge\vect B\big) .
\]
La parte eléctrica va según $\vect E$ (o en sentido contrario, si $q < 0$)
y no depende del movimiento; la parte magnética es perpendicular a la vez
a $\vect v$ y a $\vect B$, proporcional a la rapidez, y se anula para una
partícula en reposo. Unidades: $\vect E$ en \unit{V/m} (o \unit{N/C}) y
$\vect B$ en teslas (\unit{T}): $\qty{1}{T} = \qty{1}{N/(A.m)}$.
\end{definition}

\begin{proposition}[Trabajo de la fuerza de Lorentz]\label{prop:b1:charged-particles:work}
La fuerza magnética no trabaja: su potencia $q(\vect v\wedge\vect B)\cdot\vect v$
es nula, así que un \omterm{def:b1:charged-particles:lorentz}{campo magnético} solo cambia la dirección de la
velocidad, pero nunca la rapidez ni la \omterm{thm:b1:work-and-energy:ke}{energía cinética}. La fuerza
eléctrica tiene potencia $q\vect E\cdot\vect v$, y entre dos puntos su
trabajo vale $q(V_A - V_B) =
qU_{AB}$ (\cref{ch:b1:potential-capacitors}): una partícula de carga $q$
que atraviesa una tensión $U$ gana la \omterm{thm:b1:work-and-energy:ke}{energía cinética} $\abs{qU}$.
\end{proposition}

\begin{proof}
$\vect v\wedge\vect B \perp \vect v$. El campo electrostático deriva del
potencial, $\vect E = -\overrightarrow{\operatorname{grad}}V$, luego $q\vect E$
es conservativa con $E_p = qV$.
\end{proof}

\begin{definition}[El electronvoltio]\label{def:b1:charged-particles:eV}
\index{electronvoltio}
El \emph{electronvoltio} es la \omterm{thm:b1:work-and-energy:ke}{energía cinética} que gana una carga
elemental al atravesar un voltio: $\qty{1}{eV} = \qty{1.60e-19}{J}$. La
química vive en unos pocos eV, los tubos de rayos X en decenas de keV, la
física nuclear en MeV y la física de partículas en GeV y TeV.
\end{definition}

\begin{remark}[Por qué se ignora la gravedad]\label{rem:b1:charged-particles:gravity}
Para un electrón en un campo de \qty{1}{kV/m} --- un valor modesto de
laboratorio --- la fuerza eléctrica vale $eE = \qty{1.6e-16}{N}$, frente a
un peso $mg = \qty{9e-30}{N}$: $10^{13}$ veces menor. El peso se descarta
en todos los cálculos de este capítulo sin pensarlo dos veces. (El límite
relativista es otra cosa: las fórmulas siguientes valen para
$\tfrac12 mv^2 \ll mc^2$, es decir por debajo de unos \qty{50}{keV} para
electrones y \qty{100}{MeV} para protones; véase el volumen del Año 3.)
\end{remark}

\section{Campo eléctrico uniforme}

\begin{proposition}[Aceleración y desviación]\label{prop:b1:charged-particles:efield}
\index{tensión aceleradora}
En un campo uniforme $\vect E$, la aceleración $\vect a = q\vect E/m$ es
constante: el movimiento es el de un proyectil, con $q\vect E/m$ en lugar
de $\vect g$.
\begin{itemize}
  \item Soltada desde el reposo y acelerada a través de una tensión $U$,
        la partícula alcanza $v = \sqrt{2\abs{qU}/m}$.
  \item Al entrar con rapidez $v_0$ según $x$ en una región de longitud
        $L$ donde $\vect E = E\vect e_y$, sigue la parábola
        $y = (qE/2mv_0^2)x^2$ y sale desviada un ángulo $\alpha$ con $\tan\alpha =
        qEL/mv_0^2$, como si hubiera venido en línea recta desde el centro
        de las placas.
\end{itemize}
\end{proposition}

\begin{proof}
Energía: $\tfrac12 mv^2 = \abs{qU}$. Desviación: $x = v_0t$, $y =
\tfrac12(qE/m)t^2$; a la salida, $\dd y/\dd x = (qE/m)(x/v_0^2)$ evaluada
en $L$; la tangente en la salida, $y = (qEL/mv_0^2)(x - L/2)$, corta al eje
en $x = L/2$.
\end{proof}

\begin{example}[El tubo del osciloscopio]\label{ex:b1:charged-particles:crt}
Electrones acelerados con $U_0 = \qty{2.0}{kV}$ ($v_0 = \qty{2.65e7}{m/s}$,
una décima parte de $c$ --- justo en el límite no relativista) atraviesan
placas de \qty{4.0}{cm} separadas \qty{1.0}{cm} y sometidas a \qty{100}{V}:
$E = \qty{1.0e4}{V/m}$, $\tan\alpha = eEL/mv_0^2 = EL/2U_0 = 0.10$, y en una
pantalla situada \qty{30}{cm} más allá del centro de las placas la mancha
se desplaza \qty{3.0}{cm} --- \qty{0.3}{mm} por voltio, la sensibilidad del
tubo, fijada solo por la geometría y por la \omterm{prop:b1:charged-particles:efield}{tensión aceleradora}.
\end{example}

\section{Campo magnético uniforme}

\begin{theorem}[Movimiento en un campo magnético uniforme]\label{thm:b1:charged-particles:bfield}
\index{movimiento ciclotrónico}\index{radio de Larmor}\index{frecuencia ciclotrónica}\index{movimiento helicoidal}
Una partícula de carga $q$ y masa $m$, con velocidad $\vect v_0$
perpendicular a un $\vect B$ uniforme, describe una circunferencia con
rapidez constante $v_0$, de radio y \omterm{def:b1:wave-propagation:signal}{frecuencia angular}
\[
R = \frac{mv_0}{\abs q B}, \qquad \omega_c = \frac{\abs q B}{m},
\]
siendo la \emph{\omterm{thm:b1:charged-particles:bfield}{frecuencia ciclotrónica}} $\omega_c$ independiente de la
rapidez; el sentido de giro depende del signo de $q$. Si $\vect v_0$ tiene
además una componente $v_\parallel$ según $\vect B$, esa componente no
cambia y la \omterm{def:b1:point-kinematics:frame}{trayectoria} es una \emph{hélice} de radio $mv_\perp/\abs qB$ y
paso $2\pi mv_\parallel/\abs qB$ enrollada en torno a las líneas de campo.
\end{theorem}

\begin{proof}
La fuerza $q\vect v\wedge\vect B$ es perpendicular a $\vect v$ y a $\vect B$;
la rapidez es constante (no hay trabajo) y la componente según $\vect B$ no
siente fuerza alguna. En el plano perpendicular a $\vect B$, la fuerza
tiene norma constante $\abs qv_\perp B$ y es siempre perpendicular a la
velocidad: \omterm{cor:b1:point-kinematics:circular}{movimiento circular} uniforme
(\cref{thm:b1:point-kinematics:frenet}) con $mv_\perp^2/R =
\abs qv_\perp B$. Después $\omega_c = v_\perp/R$; el periodo $2\pi m/\abs qB$
por $v_\parallel$ es el paso.
\end{proof}

\begin{example}[Órdenes de magnitud]\label{ex:b1:charged-particles:orders}
Un electrón de \qty{1}{keV} ($v = \qty{1.9e7}{m/s}$) en \qty{1}{mT}:
$R = \qty{11}{cm}$, $\omega_c = \qty{1.8e8}{rad/s}$, una vuelta en
\qty{36}{ns} --- el mismo periodo a cualquier energía, que es lo que hace
posible el ciclotrón. Un protón del viento solar (\qty{1}{keV},
\qty{4.4e5}{m/s}) en los \qty{50}{\micro T} terrestres: $R = \qty{92}{m}$;
no puede cruzar las líneas de campo, solo describir una espiral a lo largo
de ellas hacia los polos --- las auroras.
\end{example}

\section{Instrumentos}

\begin{proposition}[Selector de velocidades y espectrómetro de masas]\label{prop:b1:charged-particles:spectrometer}
\index{selector de velocidades}\index{espectrómetro de masas}
En campos uniformes cruzados $\vect E \perp \vect B$, una partícula que se
mueve perpendicularmente a ambos pasa sin desviarse si y solo si
$v = E/B$, sea cual sea su carga y su masa (\emph{\omterm{prop:b1:charged-particles:spectrometer}{selector de
velocidades}}). En un \omterm{def:b1:charged-particles:lorentz}{campo magnético} solo, una partícula de rapidez
conocida describe una semicircunferencia de diámetro $2mv/\abs qB$:
medirla da $m/q$ (\emph{\omterm{prop:b1:charged-particles:spectrometer}{espectrómetro de masas}}). Los iones acelerados con
una tensión $U$ y desviados después en $B$ caen en
\[
R = \frac{1}{B}\sqrt{\frac{2mU}{\abs q}} :
\]
dos isótopos de masas $m$ y $m + \Delta m$ se separan en $\Delta R \approx
R\,\Delta m/2m$.
\end{proposition}

\begin{proof}
$q\vect E + q\vect v\wedge\vect B = \vect 0$ exige $vB = E$ con la
orientación adecuada. $R = mv/qB$ con $v = \sqrt{2qU/m}$; derívese
$\ln R = \tfrac12\ln m + \text{cte}$.
\end{proof}

\begin{example}[Isótopos del carbono]\label{ex:b1:charged-particles:isotopes}
Iones de $^{12}$C y $^{13}$C con una sola carga, acelerados con
\qty{20}{kV} en \qty{0.50}{T}: $R = \qty{14.1}{cm}$ y \qty{14.7}{cm},
separados \qty{5.8}{mm} --- dos manchas distintas en el detector, y el
cociente de sus intensidades es la abundancia isotópica, materia prima de
la datación por radiocarbono y de todo análisis de isótopos.
\end{example}

\begin{proposition}[Ciclotrón]\label{prop:b1:charged-particles:cyclotron}
\index{ciclotrón}
Dos semicilindros huecos (las ``des'') en un $\vect B$ uniforme, con una
tensión alterna $U$ entre ellos a la \omterm{thm:b1:charged-particles:bfield}{frecuencia ciclotrónica}: la partícula
gira dentro de cada de (allí no hay campo), es acelerada en $\abs qU$ en
cada uno de los dos cruces del hueco por vuelta y describe una espiral
hacia fuera con $R \propto v$; en el radio de extracción $R_{\max}$ su
\omterm{thm:b1:work-and-energy:ke}{energía cinética} vale
\[
E_k = \frac{q^2B^2R_{\max}^2}{2m},
\]
con independencia de la tensión, que solo fija el número de vueltas.
\end{proposition}

\begin{proof}
Como $\omega_c$ no depende de la rapidez, una tensión de frecuencia fija se
mantiene al paso de la partícula vuelta tras vuelta; $v = \abs qBR/m$ en el
radio $R$.
\end{proof}

\begin{example}[Un ciclotrón de hospital]\label{ex:b1:charged-particles:medical}
Protones, $B = \qty{1.5}{T}$, $R_{\max} = \qty{0.50}{m}$: $E_k =
(1.6\times10^{-19} \times 1.5 \times 0.5)^2/(2 \times 1.67\times10^{-27}) =
\qty{4.3e-12}{J} = \qty{27}{MeV}$, a $f_c = \omega_c/2\pi = \qty{23}{MHz}$.
Con \qty{50}{kV} en las des, \qty{100}{keV} por vuelta: $270$ vueltas en
\qty{12}{\micro s}. Por encima de unas decenas de MeV, el aumento
relativista de la masa (volumen del Año 3) desincroniza los protones; las
máquinas de terapia a \qty{230}{MeV} modulan la frecuencia o el campo para
compensarlo --- el asunto del problema de fin de semana.
\end{example}

\begin{proposition}[Efecto Hall]\label{prop:b1:charged-particles:hall}
\index{efecto Hall}
Una cinta de espesor $d$ que lleva una corriente $I$ de portadores de carga
$q$ y densidad $n$, en un campo $B$ perpendicular a la cinta, desarrolla a
lo ancho la \emph{tensión Hall}
\[
U_H = \frac{IB}{nqd} :
\]
proporcional a $B$ (un magnetómetro) e inversamente proporcional a $n$ (una
sonda de la densidad y del signo de los portadores).
\end{proposition}

\begin{proof}
La fuerza magnética $qv_dB$ sobre los portadores (de velocidad de arrastre
$v_d$) los empuja hacia un borde hasta que el \omterm{def:b1:charged-particles:lorentz}{campo eléctrico} transversal
$E_H$ la equilibra: $qE_H = qv_dB$, y $U_H = E_Hw$ a lo ancho $w$; con
$I = nqv_dwd$, $E_Hw = v_dBw = IB/(nqd)$.
\end{proof}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:1}
Conviértanse \qty{1}{eV}, \qty{13.6}{eV} y \qty{1}{MeV} a julios. Rapidez

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:2}
Un electrón de \qty{1.0}{keV} entra perpendicularmente en un campo de
\qty{1.0}{mT}. Radio, \omterm{def:b1:wave-propagation:signal}{frecuencia angular} ciclotrónica y periodo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:3}
Un protón a \qty{1.0e5}{m/s} en el campo terrestre (\qty{50}{\micro T}):
radio de su circunferencia. ¿Qué ocurre si su velocidad forma $30^\circ$
con el campo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:4}
Un \omterm{prop:b1:charged-particles:spectrometer}{selector de velocidades} tiene $E = \qty{10}{kV/m}$ y $B = \qty{20}{mT}$.
¿Qué rapidez pasa? ¿Qué les ocurre a los iones más rápidos, y a los de la
misma rapidez con carga opuesta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:5}
Osciloscopio: electrones a \qty{2.0}{kV}, placas de \qty{4.0}{cm} de largo
separadas \qty{1.0}{cm}, con \qty{100}{V} entre ellas, y pantalla a
\qty{30}{cm} del centro de las placas. Desviación en la pantalla y
sensibilidad en mm/V. ¿Por qué una \omterm{prop:b1:charged-particles:efield}{tensión aceleradora} mayor reduce la
sensibilidad?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:6}
Iones de $^{12}$C y $^{13}$C con una sola carga, acelerados con
\qty{20}{kV}, entran en un campo de \qty{0.50}{T} ($\qty{1}{u} = \qty{1.66e-27}{kg}$).
Radios y separación de las dos manchas tras media vuelta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:7}
Un ciclotrón de protones tiene $B = \qty{1.5}{T}$ y $R_{\max} = \qty{0.50}{m}$.
Frecuencia de la tensión de las des, \omterm{thm:b1:work-and-energy:ke}{energía cinética} máxima, número de
vueltas con \qty{50}{kV} en las des y tiempo que pasa dentro. ¿En qué
fracción ha aumentado la masa del protón a la salida
($\gamma - 1 \approx E_k/mc^2$, $mc^2 = \qty{938}{MeV}$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:8}
Un electrón a \qty{1.0e6}{m/s} entra en un campo de \qty{2.0}{mT} a
$30^\circ$ de las líneas de campo. Radio y paso de su hélice; número de
vueltas por metro a lo largo del campo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:9}
Una lámina de cobre de \qty{1.0}{mm} de espesor lleva \qty{5.0}{A} en
\qty{1.0}{T} ($n = \qty{8.5e28}{m^{-3}}$). Tensión Hall. ¿Por qué se hacen

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:10}
Pruébese, a partir de la ley de Newton, que una carga en cualquier \omterm{def:b1:charged-particles:lorentz}{campo
magnético} (uniforme o no), sin \omterm{def:b1:charged-particles:lorentz}{campo eléctrico}, conserva la rapidez. Un
electrón de \qty{10}{keV} entra en una región de campo intenso y no
uniforme: ¿qué puede cambiar y qué no?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:11}
Una carga $q$ parte del reposo en el origen con $\vect E = E\vect e_y$ y
$\vect B = B\vect e_z$. Escríbanse las ecuaciones del movimiento,
muéstrese que la partícula deriva en promedio con la velocidad $E/B$ según
$\vect e_x$ (indicación: búsquese una solución de la forma
$\vect v = \vect u + \vect w$ con $\vect u$ constante) y descríbase la
\omterm{def:b1:point-kinematics:frame}{trayectoria} (una cicloide). Datos para un electrón con
$E = \qty{1.0e4}{V/m}$ y $B = \qty{0.10}{T}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:12}
En el osciloscopio de \cref{exo:b1:charged-particles:5}, sustitúyanse las
placas por una bobina de desviación magnética que produzca \qty{1.0}{mT}

qué los tubos de televisión, con sus ángulos grandes, usan desviación
magnética y no eléctrica?
\end{exercise}

\section{Problema: protones contra un tumor}

\begin{problem}[{Problema de fin de semana --- el ciclotrón tras el muro
de un hospital: cómo se lleva a los protones a un tercio de la velocidad
de la luz, por qué la máquina sencilla deja de funcionar antes de la
energía que un terapeuta necesita y cuántos protones hacen falta para
tratar un tumor}]
\label{pb:b1:charged-particles:1}
Protones: $m = \qty{1.67e-27}{kg}$, $q = e = \qty{1.60e-19}{C}$, $mc^2 =
\qty{938}{MeV}$; $c = \qty{3.00e8}{m/s}$. Ciclotrón: $B = \qty{1.5}{T}$,
radio de extracción $R_{\max} = \qty{0.50}{m}$, \omterm{def:b1:wave-propagation:signal}{amplitud} de la tensión de
las des $U = \qty{50}{kV}$.

\medskip
\noindent\textbf{Parte I --- Una vuelta.}
\begin{enumerate}
  \item Escríbase la \omterm{def:b1:charged-particles:lorentz}{fuerza de Lorentz} sobre un protón que se mueve con
        $\vect v$ perpendicular a $\vect B$, y muéstrese que su rapidez no
        puede cambiar dentro de una de.
  \item Dedúzcanse el radio de su circunferencia y su periodo; muéstrese
        que el periodo no depende de la rapidez.
  \item Calcúlese la \omterm{thm:b1:charged-particles:bfield}{frecuencia ciclotrónica} $f_c$ de estos protones.
  \item Energía ganada en cada cruce del hueco; y por vuelta.
  \item ¿Por qué debe alternar la tensión de las des exactamente a $f_c$?
        ¿Qué ocurre si se desvía un poco?
  \item Los protones se inyectan en el centro con rapidez despreciable:
        ¿radio tras la primera vuelta? ¿Y tras $n$ vueltas?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Hasta el borde.}
\begin{enumerate}[resume]
  \item \omterm{thm:b1:work-and-energy:ke}{Energía cinética} en la extracción (en J y en MeV) y rapidez, como
        fracción de $c$.
  \item Número de vueltas necesarias y tiempo que pasa en la máquina.
  \item Longitud total del camino en espiral (suma de las circunferencias
        --- aproxímese la suma por una integral).
  \item ¿Depende de $U$ el resultado de la pregunta 7? ¿Sobre qué actúa $U$?
  \item El vacío interior ha de ser bueno: estímese la distancia que
        recorre un protón entre choques con moléculas de gas residual si
        el recorrido libre medio debe superar la longitud de la espiral;
        el recorrido libre medio va como $1/p$ y vale unos \qty{70}{nm} a
        presión atmosférica. ¿Qué presión hace falta?
  \item Calcúlense $\gamma - 1 \approx E_k/mc^2$ en la extracción y la
        variación relativa del periodo que eso supone. A lo largo del
        número de vueltas hallado antes, ¿en qué fracción de periodo se ha
        desfasado el protón respecto de la tensión? Coméntese.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- 230 MeV.}
La protonterapia necesita unos \qty{230}{MeV} para llegar a tumores a
\qty{30}{cm} de profundidad.
\begin{enumerate}[resume]
  \item Con la fórmula no relativista, ¿qué producto $BR$ daría
        \qty{230}{MeV}? Con $B = \qty{2.0}{T}$, ¿qué radio?
  \item A esa energía, $\gamma - 1 = 0.245$: ¿qué le ocurre al periodo
        ciclotrónico a medida que el protón gana energía y por qué falla
        un ciclotrón de frecuencia fija?
  \item Hay dos remedios: variar la frecuencia de la tensión durante la
        aceleración (sincrociclotrón) o hacer que $B$ crezca con el radio
        de modo que $B/\gamma$ se mantenga constante (ciclotrón isócrono).
        Explíquese cada uno en una frase y nómbrese un inconveniente del
        primero.
  \item La cantidad de movimiento relativista es $p = \gamma mv$ y el $R =
        p/qB$ sigue valiendo: con $\gamma = 1.245$ y $v = 0.60c$,
        calcúlese $R$ para $B = \qty{2.0}{T}$ y compárese con la pregunta 13.
  \item El haz extraído pasa entre dos placas de \qty{1.0}{m} de largo con
        $E = \qty{1.0e5}{V/m}$ para desviarlo un ángulo pequeño: estímese
        el ángulo de desviación (de forma no relativista, con la $p$ de la
        pregunta anterior en lugar de $mv$). ¿Resulta práctica aquí la
        desviación eléctrica?
  \item La misma desviación con un imán de \qty{1.0}{T} a lo largo de
        \qty{1.0}{m}: ¿ángulo? Conclúyase por qué las líneas de haz usan imanes.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La dosis.}
\begin{enumerate}[resume]
  \item Un protón de \qty{230}{MeV} deposita casi toda su energía cerca
        del final de su alcance (el pico de Bragg). Si toda ella se
        absorbe en un tumor de \qty{0.50}{kg} de masa, ¿cuántos protones
        entregan una dosis de \qty{2.0}{Gy} ($\qty{1}{Gy} = \qty{1}{J/kg}$)?
  \item La corriente del haz es de \qty{1.0}{nA}: protones por segundo y
        duración de la sesión.
  \item Potencia que transporta el haz; compárese con una bombilla.
  \item Los protones se frenan hasta detenerse dentro del paciente: ¿cuál
        es, a grandes rasgos, la energía depositada por protón en el
        último centímetro, si allí se pierde una cuarta parte de la
        energía? ¿En \omterm{def:b1:charged-particles:eV}{electronvoltios} por micrómetro?
  \item Cada ionización en el tejido cuesta unos \qty{30}{eV}: ¿cuántas
        ionizaciones causa un protón a lo largo de todo su recorrido?
  \item ¿Por qué es preferible un haz de protones a los rayos X para un
        tumor profundo, en una frase sobre adónde va la energía?
  \item Resúmase la cadena que va de la \omterm{def:b1:charged-particles:lorentz}{fuerza de Lorentz} a la dosis: qué
        ecuaciones fijaron la energía, el radio, el número de vueltas y el
        tiempo de tratamiento.
\end{enumerate}
\end{problem}
