\chapter{Forces centrales~: planètes et satellites}\label{ch:b1:central-forces}

Trente et un satellites font deux fois le tour de la Terre par jour,
chacun une horloge et une radio, et un téléphone qui en entend quatre
sait où il se trouve à quelques mètres près. La comète de Halley tombe
vers le Soleil pendant trente-huit ans, en fait le tour en quelques
semaines, puis s'en éloigne pendant trente-huit ans encore. Une sonde à
destination de Mars quitte la Terre exactement à la vitesse qui fera
effleurer à sa \omterm{def:b1:point-kinematics:frame}{trajectoire} l'orbite de Mars. Tout cela est le mouvement
d'un corps attiré vers un point fixe par une force qui décroît comme
l'inverse du carré de la distance, et tout cela découle de deux
grandeurs conservées --- l'énergie et le \omterm{def:b1:angular-momentum:L}{moment cinétique} --- plus une
idée nouvelle, le potentiel effectif. Ce chapitre en tire la forme et
les vitesses des orbites, les trois \omterm{thm:b1:central-forces:conics}{lois de Kepler}, et l'arithmétique
qui place un satellite là où on le veut.

\section{Forces centrales conservatives}

\begin{definition}[Force centrale~; force centrale conservative]\label{def:b1:central-forces:central}
\index{force centrale}\index{énergie potentielle effective}
Une force est \emph{centrale} de centre $O$ si elle est toujours portée
par $\vect{OM}$~: $\vect F = F(M)\,\vect e_r$. Elle est \emph{centrale
conservative} si son intensité ne dépend que de $r = OM$~: $\vect F =
F(r)\,\vect e_r$, ce qui la fait dériver d'une \omterm{def:b1:work-and-energy:conservative}{énergie potentielle}
$E_p(r)$ avec $F(r) = -\dd E_p/\dd r$.
\end{definition}

\begin{theorem}[Mouvement sous une force centrale conservative]\label{thm:b1:central-forces:reduction}
\index{loi des aires}
Un point de masse $m$ soumis à une seule \omterm{def:b1:central-forces:central}{force centrale} conservative~:
\begin{enumerate}
  \item conserve son \omterm{def:b1:angular-momentum:L}{moment cinétique} $\vect L_O$~; son mouvement est
        plan et, en \omterm{def:b1:point-kinematics:cylindrical}{coordonnées polaires} dans ce plan, $r^2\dot\theta = C$
        est constant (\omterm{cor:b1:angular-momentum:central}{loi des aires}, \cref{cor:b1:angular-momentum:central})~;
  \item conserve son \omterm{thm:b1:work-and-energy:em}{énergie mécanique} $E_m = \tfrac12 m(\dot r^2 + r^2\dot\theta^2)
        + E_p(r)$, que l'on peut écrire
        \[
        E_m = \tfrac12 m\dot r^2 + E_{p,\mathrm{eff}}(r), \qquad
        E_{p,\mathrm{eff}}(r) = E_p(r) + \frac{mC^2}{2r^2} = E_p(r) + \frac{L^2}{2mr^2} .
        \]
\end{enumerate}
Le mouvement radial est donc celui d'un point à une dimension dans le
\emph{potentiel effectif} $E_{p,\mathrm{eff}}$~: les \omterm{def:b1:work-and-energy:turning}{points de
rebroussement} radiaux sont les racines de
$E_{p,\mathrm{eff}}(r) = E_m$, le mouvement est lié s'il est piégé entre
deux d'entre eux, et circulaire si $r$ se place en un minimum de
$E_{p,\mathrm{eff}}$.
\end{theorem}

\begin{proof}
La partie~1 est le \cref{cor:b1:angular-momentum:central}. La force est
conservative et aucune autre force n'agit, donc $E_m$ se conserve~; dans
le plan, $v^2 = \dot r^2 + r^2\dot\theta^2$ et $r^2\dot\theta^2 = C^2/r^2$.
Le terme $mC^2/2r^2$, \omterm{thm:b1:work-and-energy:ke}{énergie cinétique} du mouvement angulaire, joue le
rôle d'une \emph{\omterm{prop:b1:central-forces:effective}{barrière centrifuge}} répulsive dans le problème radial,
auquel s'applique l'analyse à une dimension du \cref{ch:b1:work-and-energy}.
\end{proof}

\section{La gravitation newtonienne}

\begin{definition}[Gravitation universelle]\label{def:b1:central-forces:gravitation}
\index{gravitation}\index{énergie potentielle de gravitation}
Une masse ponctuelle $M$ en $O$ attire une masse ponctuelle $m$ en $M$ avec
\[
\vect F = -\frac{GMm}{r^2}\,\vect e_r, \qquad E_p = -\frac{GMm}{r} \quad (E_p \to 0 \text{ à l'infini}),
\]
avec $G = \qty{6.67e-11}{N.m^2/kg^2}$. Un corps à symétrie sphérique
attire, à l'extérieur de lui-même, comme si toute sa masse était en son
centre (admis~; c'est une conséquence du théorème de Gauss,
\cref{ch:b1:electrostatics-gauss}).
\end{definition}

\begin{proposition}[Le potentiel effectif de la gravitation]\label{prop:b1:central-forces:effective}
\index{barrière centrifuge}\index{orbite circulaire}
Pour $E_p = -GMm/r$,
\[
E_{p,\mathrm{eff}}(r) = -\frac{GMm}{r} + \frac{L^2}{2mr^2}
\]
tend vers $+\infty$ quand $r \to 0$ (pour $L \neq 0$), vers $0^-$ à
l'infini, et présente un minimum unique en $r_c = L^2/GMm^2$, de valeur
$-G^2M^2m^3/2L^2$. D'où~: $E_m < 0$ --- \omterm{def:b1:work-and-energy:turning}{mouvement lié} entre un rayon
minimal et un rayon maximal (périgée et apogée)~; $E_m \geq 0$ --- non
lié, le corps vient de l'infini et y retourne~; $E_m$ égale au minimum
--- \omterm{prop:b1:central-forces:effective}{orbite circulaire} de rayon $r_c$.
\end{proposition}

\begin{proof}
Les limites se lisent~; $E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ en
$r_c$, puis on substitue. Le cas circulaire est $\dot r = 0$ pour
toujours, ce qui n'est possible qu'au minimum.
\end{proof}

\begin{theorem}[Orbites circulaires]\label{thm:b1:central-forces:circular}
\index{vitesse orbitale}\index{troisième loi de Kepler}
Sur une \omterm{prop:b1:central-forces:effective}{orbite circulaire} de rayon $r$ autour d'une masse $M$~:
\[
v = \sqrt{\frac{GM}{r}}, \qquad
T = 2\pi\sqrt{\frac{r^3}{GM}}, \qquad
E_m = -\frac{GMm}{2r} = -E_k = \tfrac12 E_p .
\]
En particulier, $T^2/r^3 = 4\pi^2/GM$ est le même pour toutes les \omterm{prop:b1:central-forces:effective}{orbites
circulaires} autour du même centre (\omterm{thm:b1:central-forces:circular}{troisième loi de Kepler}).
\end{theorem}

\begin{proof}
Le \omterm{cor:b1:point-kinematics:circular}{mouvement circulaire} uniforme réclame la \omterm{prop:b1:newton-dynamics:centripetal}{force centripète}
$mv^2/r = GMm/r^2$~; puis $T = 2\pi r/v$~; enfin
$E_k = \tfrac12 mv^2 = GMm/2r$ et $E_p = -GMm/r$.
\end{proof}

\begin{example}[Trois orbites]\label{ex:b1:central-forces:orbits}
$GM_\oplus = \qty{3.99e14}{m^3/s^2}$. Orbite basse ($r = \qty{6770}{km}$,
\qty{400}{km} d'altitude)~: $v = \qty{7.7}{km/s}$, $T = \qty{92}{min}$.
Géostationnaire ($T = \qty{86164}{s}$, un jour sidéral)~: $r = (GMT^2/4\pi^2)^{1/3} =
\qty{42200}{km}$, $v = \qty{3.1}{km/s}$. La Lune ($r = \qty{384000}{km}$)~:
$T = \qty{27.4}{d}$ --- et son \omterm{cor:b1:point-kinematics:circular}{accélération centripète} $v^2/r =
\qty{2.7e-3}{m/s^2}$ vaut $g/3600$ à soixante rayons terrestres~: la loi
en carré inverse, vérifiée par Newton lui-même.
\end{example}

\section{Les lois de Kepler et la forme des orbites}

\begin{theorem}[Trajectoires dans un champ newtonien]\label{thm:b1:central-forces:conics}
\index{lois de Kepler}\index{conique}\index{ellipse (orbite)}\index{excentricité}
La \omterm{def:b1:point-kinematics:frame}{trajectoire} d'un corps dans le champ d'une masse fixe $M$ est une
conique dont $O$ est un foyer~: une ellipse si $E_m < 0$, une parabole
si $E_m = 0$, une hyperbole si $E_m > 0$. Pour l'ellipse de demi-grand
axe $a$ et d'excentricité $e$~:
\begin{itemize}
  \item (Kepler~I) le centre de force est en un foyer~; le périgée et
        l'apogée sont en $r_p = a(1 - e)$ et $r_a = a(1 + e)$~;
  \item (Kepler~II) la \omterm{cor:b1:angular-momentum:central}{vitesse aréolaire} est constante (\omterm{cor:b1:angular-momentum:central}{loi des aires})~;
  \item (Kepler~III) la période vérifie $T^2 = 4\pi^2a^3/GM$~;
  \item l'énergie ne dépend que de $a$, $E_m = -GMm/2a$, de sorte que la
        vitesse à la distance $r$ est donnée par
        \[
        v^2 = GM\left(\frac{2}{r} - \frac{1}{a}\right) .
        \]
\end{itemize}
\end{theorem}
\admitted

\begin{remark}[Ce qui est démontré et ce qui est admis]\label{rem:b1:central-forces:admitted}
Kepler~II et la classification par l'énergie ont été démontrées
ci-dessus. Que les orbites liées soient exactement des ellipses dont $O$
est un foyer, avec Kepler~III et $E_m = -GMm/2a$ en général, demande
l'intégration de l'équation radiale (le volume de deuxième année la
fait)~; pour les \omterm{prop:b1:central-forces:effective}{orbites circulaires} ($a = r$, $e = 0$), chaque énoncé
se réduit au \cref{thm:b1:central-forces:circular}, ce qui est la
vérification que nous pouvons faire ici. La relation
$v^2 = GM(2/r - 1/a)$ découle de
$E_m = \tfrac12 mv^2 - GMm/r = -GMm/2a$ une fois cette formule d'énergie
accordée.
\end{remark}

\begin{example}[La comète de Halley]\label{ex:b1:central-forces:halley}
Périhélie \qty{0.586}{AU}, aphélie \qty{35.1}{AU}~: $a = \qty{17.8}{AU}$,
$e = 0.967$, $T = \sqrt{17.8^3} = 75$~ans (Kepler~III en unités
astronomiques et en années, où $T^2 = a^3$ pour le Soleil). Avec
$GM_\odot = \qty{1.33e20}{m^3/s^2}$, la vitesse au périhélie vaut
$\sqrt{GM(2/r_p - 1/a)} = \qty{54}{km/s}$ et à l'aphélie \qty{0.9}{km/s}
--- soixante fois moins, comme l'exige la \omterm{cor:b1:angular-momentum:central}{loi des aires} ($r_pv_p = r_av_a$).
\end{example}

\section{Satellites}

\begin{proposition}[Vitesse de libération~; orbite géostationnaire]\label{prop:b1:central-forces:escape}
\index{vitesse de libération}\index{orbite géostationnaire}
Depuis la surface d'un corps de masse $M$ et de rayon $R$, la vitesse de
lancement minimale pour atteindre l'infini est
$v_e = \sqrt{2GM/R} = \sqrt2\,v_c$, où $v_c = \sqrt{GM/R}$ est la vitesse
circulaire au ras du sol~: \qty{11.2}{km/s} pour la Terre. Un satellite
\emph{géostationnaire} orbite dans le plan équatorial avec la période
sidérale de la Terre, à $r = \qty{42200}{km}$ (\qty{35800}{km}
au-dessus du sol).
\end{proposition}

\begin{proof}
$E_m \geq 0$~: $\tfrac12 mv_e^2 - GMm/R = 0$. Kepler~III avec $T = \qty{86164}{s}$.
\end{proof}

\begin{proposition}[Transfert de Hohmann]\label{prop:b1:central-forces:hohmann}
\index{transfert de Hohmann}
Pour faire passer un satellite d'une \omterm{prop:b1:central-forces:effective}{orbite circulaire} de rayon $r_1$ à
une \omterm{prop:b1:central-forces:effective}{orbite circulaire} coplanaire de rayon $r_2 > r_1$ par deux
impulsions brèves, on allume en $r_1$ pour entrer sur l'\emph{ellipse de
transfert} de périgée $r_1$ et d'apogée $r_2$ ($a = (r_1 + r_2)/2$), on
se laisse porter une demi-période, puis on rallume en $r_2$ pour
circulariser~:
\[
\Delta v_1 = \sqrt{\frac{GM}{r_1}}\left(\sqrt{\frac{2r_2}{r_1 + r_2}} - 1\right),
\qquad
\Delta v_2 = \sqrt{\frac{GM}{r_2}}\left(1 - \sqrt{\frac{2r_1}{r_1 + r_2}}\right),
\qquad
t = \pi\sqrt{\frac{(r_1 + r_2)^3}{8GM}} .
\]
\end{proposition}

\begin{proof}
Les vitesses sur l'ellipse de transfert par $v^2 = GM(2/r - 1/a)$ en
$r_1$ et $r_2$~; on retranche les vitesses circulaires~; et la demi-période
du \cref{thm:b1:central-forces:conics}.
\end{proof}

\begin{example}[De l'orbite basse à la géostationnaire]\label{ex:b1:central-forces:leogeo}
$r_1 = \qty{6770}{km}$, $r_2 = \qty{42200}{km}$~: $\Delta v_1 = \qty{2.4}{km/s}$,
$\Delta v_2 = \qty{1.5}{km/s}$, et \qty{5.3}{h} de transit. Le total,
\qty{3.9}{km/s}, vaut la moitié de ce qu'a déjà coûté la mise en orbite
basse~: le «~dernier kilomètre~» vers une orbite haute est cher, et les
satellites emportent leur propre moteur pour le franchir.
\end{example}

\begin{remark}[Centres répulsifs]\label{rem:b1:central-forces:rutherford}
\index{diffusion de Rutherford}
Pour une force répulsive en carré inverse (deux charges de même signe,
\cref{ch:b1:electrostatics-gauss}), le potentiel effectif n'a pas de
puits~: toute \omterm{def:b1:point-kinematics:frame}{trajectoire} est une hyperbole, la particule s'approche,
est déviée et repart. Une particule $\alpha$ lancée droit sur un noyau
d'or a une distance d'approche minimale que l'énergie seule fixe, $E_k =
2Ze^2/4\pi\varepsilon_0 r_{\min}$ --- c'est ainsi que Rutherford a
majoré la taille du noyau (\cref{exo:b1:central-forces:11}).
\end{remark}

\section{Problème~: les satellites qui vous disent où vous êtes}

\begin{problem}[{Problème du week-end --- une constellation de
navigation à vingt mille kilomètres~: son orbite, les ergols pour y
aller, combien de satellites un téléphone peut voir, et pourquoi leurs
horloges doivent être réglées pour retarder}]
\label{pb:b1:central-forces:1}
Données~: $GM_\oplus = \qty{3.986e14}{m^3/s^2}$, $R_\oplus = \qty{6371}{km}$,
jour sidéral \qty{86164}{s}, $c = \qty{2.998e8}{m/s}$. Les satellites
font le tour de la Terre en exactement un demi-jour sidéral, sur des
\omterm{prop:b1:central-forces:effective}{orbites circulaires}.

\medskip
\noindent\textbf{Partie I --- L'orbite.}
\begin{enumerate}
  \item Calculer le rayon orbital et l'altitude.
  \item Calculer la \omterm{thm:b1:central-forces:circular}{vitesse orbitale}.
  \item Calculer l'accélération du satellite et la comparer à $g$ à la
        surface.
  \item Calculer l'\omterm{thm:b1:work-and-energy:em}{énergie mécanique} par kilogramme, ainsi que ses parts
        cinétique et potentielle.
  \item Pourquoi choisir un demi-jour sidéral plutôt qu'un demi-jour
        solaire~? Que voit se répéter un observateur au sol~?
  \item Vu d'un satellite, sous quel angle le disque terrestre est-il vu~?
  \item Comparer le rayon de cette orbite à celui de l'\omterm{prop:b1:central-forces:escape}{orbite
        géostationnaire}~: laquelle est la plus haute, et de quel facteur~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Y aller.}
\begin{enumerate}[resume]
  \item Énergie par kilogramme pour passer du repos à la surface à
        l'orbite finale (on néglige l'air et la rotation terrestre).
  \item Le lanceur atteint d'abord une \omterm{prop:b1:central-forces:effective}{orbite circulaire} basse à
        \qty{400}{km}. Quelle vitesse y a-t-il~?
  \item \omterm{prop:b1:central-forces:hohmann}{Transfert de Hohmann} de cette orbite vers l'orbite finale~: les
        deux impulsions et leur somme.
  \item Durée du transfert.
  \item Comparer la somme des impulsions à la \omterm{prop:b1:central-forces:escape}{vitesse de libération}
        depuis l'orbite basse~: à quelle distance de «~quitter la Terre~»
        cette mission se trouve-t-elle~?
  \item L'étage supérieur et le satellite pèsent \qty{2000}{kg} avant le
        transfert~; la vitesse d'éjection du moteur vaut \qty{3.0}{km/s}
        et l'équation de Tsiolkovski donne $\Delta v = u\ln(m_0/m_1)$.
        Quelle masse d'ergols les deux impulsions demandent-elles~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Couverture et datation.}
\begin{enumerate}[resume]
  \item Un satellite est visible de tout point de la Terre situé dans le
        cône de demi-angle $\theta$ tel que $\cos\theta = R/r$ autour de
        la ligne Terre--satellite. Calculer $\theta$ et la fraction
        $(1 - \cos\theta)/2$ de la surface terrestre qu'il couvre.
  \item Avec $24$ satellites répartis uniformément, combien en voit-on en
        moyenne depuis un point~? Le minimum de $4$ nécessaire à un point
        fixe est-il plausible~?
  \item Durée de propagation de la lumière depuis un satellite au
        zénith~; puis à l'horizon.
  \item Pour situer le récepteur à \qty{3}{m} près, avec quelle précision
        faut-il dater les signaux~?
  \item L'horloge propre du récepteur est un quartz bon marché, faux de
        quelques millisecondes~: pourquoi faut-il quatre satellites au
        lieu de trois~?
  \item Les horloges des satellites sont atomiques. Donner la stabilité
        relative nécessaire pour une erreur de \qty{3}{m} au bout d'un jour.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Un peu de relativité.}
Deux effets décalent les horloges des satellites par rapport à celles du
sol (formules admises, du volume de troisième année)~: le mouvement
ralentit une horloge de la fraction $v^2/2c^2$~; l'altitude l'accélère
de $GM(1/R - 1/r)/c^2$.
\begin{enumerate}[resume]
  \item Calculer le ralentissement dû à la vitesse, en microsecondes par
        jour.
  \item Calculer l'accélération due à l'altitude, en microsecondes par
        jour.
  \item Effet net, et erreur de position qu'il causerait par jour s'il
        n'était pas corrigé.
  \item Les horloges des satellites sont réglées avant le lancement sur
        \qty{10.22999999543}{MHz} au lieu de \qty{10.23}{MHz}. Vérifier
        que cela compense l'effet net.
  \item Les orbites ne sont pas exactement képlériennes~: nommer trois
        perturbations et dire comment le système y fait face.
  \item Récapituler l'enchaînement~: deux lois de conservation, une
        orbite, une constellation, et l'unique correction que personne
        n'aurait devinée.
\end{enumerate}
\end{problem}
