\chapter{Systèmes de points~; introduction aux solides}\label{ch:b1:systems-of-points}

Une plongeuse quitte le tremplin, se groupe, fait deux saltos et se
déplie~; pendant tout ce temps, un point de son corps --- son \omterm{def:b1:systems-of-points:cm}{centre de
masse} --- décrit la même parabole banale qu'une pierre. Un yo-yo lâché
de la main met trois fois plus de temps qu'une pierre à atteindre le
bout de sa ficelle, et y arrive en tournant à deux mille tours par
minute. Une sphère pleine bat une sphère creuse sur n'importe quel plan
incliné, quelles que soient leurs tailles et leurs masses. La mécanique
du point, développée dans les chapitres précédents, s'étend aux systèmes
de plusieurs points --- et aux \omterm{def:b1:systems-of-points:solid}{solides} de la vie courante --- par une
poignée de théorèmes~: le \omterm{def:b1:systems-of-points:cm}{centre de masse} se déplace comme un point,
l'énergie et le \omterm{def:b1:angular-momentum:L}{moment cinétique} se scindent en une part de \omterm{def:b1:systems-of-points:cm}{centre de
masse} et une part «~autour~» de lui, et un \omterm{def:b1:systems-of-points:solid}{solide} qui tourne autour d'un
axe fixe est un système à un degré de liberté que gouverne son \omterm{prop:b1:angular-momentum:inertia}{moment
d'inertie}.

\section{Centre de masse et quantité de mouvement}

\begin{definition}[Centre de masse~; quantité de mouvement d'un système]\label{def:b1:systems-of-points:cm}
\index{centre de masse}\index{barycentre}
Pour des points de masses $m_i$ situés en $M_i$, de masse totale
$M = \sum m_i$, le \emph{centre de masse} (barycentre) $G$ est défini par
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{c.-à-d.}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
quelle que soit l'origine $O$. La \emph{quantité de mouvement} du
système vaut $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[Théorème du centre de masse]\label{thm:b1:systems-of-points:cmtheorem}
\index{théorème du centre de masse}\index{conservation de la quantité de mouvement}
Dans un \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen},
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
le \omterm{def:b1:systems-of-points:cm}{centre de masse} se déplace comme un \omterm{def:b1:point-kinematics:frame}{point matériel} de masse $M$
soumis à la résultante des seules forces \emph{extérieures}. Pour un
\omterm{def:b1:newton-dynamics:conservation}{système fermé} sans force extérieure, $\vect P$ se conserve et $G$ se
déplace uniformément.
\end{theorem}

\begin{proof}
On somme la deuxième loi de Newton sur les points~; les forces
intérieures se compensent par paires d'action et de réaction~; et
$\sum m_i\vect a_i = M\vect a_G$ en dérivant deux fois la définition de $G$.
\end{proof}

\begin{example}[Plongeuse, recul, choc]\label{ex:b1:systems-of-points:cm}
\index{choc}
Le $G$ de la plongeuse suit une parabole sous son seul poids, quelles que
soient ses vrilles --- ses muscles sont des forces intérieures. Un fusil
de \qty{4.0}{kg} qui tire une balle de \qty{10}{g} à \qty{800}{m/s}~:
$\vect P = \vect 0$ avant comme après, donc le fusil recule à
\qty{2.0}{m/s}. Deux voitures, \qty{1000}{kg} à \qty{20}{m/s} et
\qty{1500}{kg} à l'arrêt, qui s'accrochent~: $v =
20000/2500 = \qty{8.0}{m/s}$ --- quantité de mouvement conservée, tandis
que $60\%$ de l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} est passée en tôles froissées et en
chaleur. La quantité de mouvement se conserve dans tout choc~; l'\omterm{thm:b1:work-and-energy:ke}{énergie
cinétique} seulement dans les chocs \emph{élastiques}
(\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{Les théorèmes de König}

\begin{definition}[Référentiel barycentrique]\label{def:b1:systems-of-points:barycentric}
\index{référentiel barycentrique}
Le \emph{\omterm{def:b1:point-kinematics:frame}{référentiel} barycentrique} $\mathcal R^*$ a son origine en $G$
et est en translation par rapport au \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen} $\mathcal R$
(ses axes gardent des directions fixes). Les grandeurs qu'on y mesure
sont étoilées~: $\vect v_i^*
= \vect v_i - \vect v_G$, et $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[Théorèmes de König]\label{thm:b1:systems-of-points:koenig}
\index{théorèmes de König}
Le \omterm{def:b1:angular-momentum:L}{moment cinétique} en un point $O$ et l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} d'un système
se scindent en la contribution de leur \omterm{def:b1:systems-of-points:cm}{centre de masse}, portant toute la
masse, et les contributions dans le \omterm{def:b1:systems-of-points:barycentric}{référentiel barycentrique}~:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
où $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ et $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ ne dépendent pas de l'origine du \omterm{def:b1:point-kinematics:frame}{référentiel}.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ et $\vect v_i = \vect v_G + \vect v_i^*$~;
on développe $\sum\vect{OM_i}\wedge m_i\vect v_i$ et $\sum\tfrac12 m_iv_i^2$~: les
termes croisés contiennent $\sum m_i\vect{GM_i} = \vect 0$ ou $\sum m_i\vect v_i^* = \vect 0$
et s'annulent.
\end{proof}

\begin{example}[Une roue qui roule]\label{ex:b1:systems-of-points:rolling}
\index{roulement sans glissement}
Une roue de masse $M$, de rayon $R$, de \omterm{prop:b1:angular-momentum:inertia}{moment d'inertie} $J$ par rapport
à son axe, qui roule sans glisser à la vitesse $v$~: le point de contact
est immobile, donc $\omega = v/R$, et $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- soit les trois quarts de $Mv^2$ pour un disque homogène
($J = \tfrac12 MR^2$) et $Mv^2$ pour un cerceau. C'est ce second terme qui
fait qu'on allège les roues d'un vélo~: leur rotation coûte une énergie
que la masse du cadre ne coûte pas.
\end{example}

\section{Le problème à deux corps}

\begin{proposition}[Masse réduite]\label{prop:b1:systems-of-points:twobody}
\index{problème à deux corps}\index{masse réduite}
Deux points qui n'interagissent qu'entre eux (force $\vect F$ exercée sur
$m_2$ par $m_1$, et $-\vect F$ sur $m_1$)~: $G$ se déplace uniformément,
et la position relative $\vect r = \vect{M_1M_2}$ obéit à
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
c'est l'équation d'un unique point fictif de \emph{\omterm{prop:b1:systems-of-points:twobody}{masse réduite}} $\mu$
soumis à $\vect F$ autour d'un centre fixe. Dans $\mathcal R^*$, $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ et $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. Dans $\mathcal R^*$, $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ et
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$~; on substitue dans $E_k^*$ et $\vect L^*$.
\end{proof}

\begin{example}[Quand le centre bouge aussi]\label{ex:b1:systems-of-points:reduced}
Au \cref{ch:b1:central-forces}, le Soleil était fixe~: ce n'est exact
qu'à la limite $m \ll M$~; en général, la \omterm{thm:b1:central-forces:circular}{troisième loi de Kepler} s'écrit $T^2 = 4\pi^2a^3/G(M +
m)$. Pour la molécule de HCl qui vibre au \cref{pb:b1:work-and-energy:1},
la masse qui oscille dans le puits est $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$, ce qui élève la fréquence de $1.4\%$~; pour l'atome
d'hydrogène, le $\mu$ de l'électron diffère de $m_e$ d'une part sur
$1836$ --- mesurable dans le spectre.
\end{example}

\section{Solide en rotation autour d'un axe fixe}

\begin{definition}[Solide~; rotation autour d'un axe fixe]\label{def:b1:systems-of-points:solid}
\index{solide}\index{rotation autour d'un axe fixe}
Un \emph{solide} est un système dont les points gardent des distances
mutuelles fixes. S'il tourne autour d'un axe $\Delta$ fixe dans le
\omterm{def:b1:point-kinematics:frame}{référentiel}, chaque point décrit un cercle centré sur $\Delta$ à la même
\omterm{cor:b1:point-kinematics:circular}{vitesse angulaire} $\omega = \dot\theta$~; un point situé à la distance
$r_i$ de l'axe a pour vitesse $r_i\omega$.
\end{definition}

\begin{proposition}[Moment d'inertie~; moment cinétique~; énergie cinétique]\label{prop:b1:systems-of-points:inertia}
\index{moment d'inertie!des solides}\index{théorème de Huygens}
Avec le \emph{\omterm{prop:b1:angular-momentum:inertia}{moment d'inertie}} $J_\Delta = \sum m_ir_i^2$ (une intégrale
$\int r^2\,\dd m$ pour un corps continu)~:
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
Valeurs classiques (masse $M$)~: anneau mince ou cylindre creux autour de
son axe, $MR^2$~; disque homogène ou cylindre plein, $\tfrac12 MR^2$~;
tige mince de longueur $L$ autour de son centre, $\tfrac{1}{12}ML^2$,
autour d'une extrémité, $\tfrac13 ML^2$~; sphère pleine autour d'un
diamètre, $\tfrac25 MR^2$. \emph{Théorème de Huygens}~: par rapport à un
axe $\Delta$ parallèle à un axe $\Delta_G$ passant par $G$ et distant de
$d$, $J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
$L_\Delta$ et $E_k$~: \cref{prop:b1:angular-momentum:inertia}. Tige
autour d'une extrémité~: $\int_0^L x^2(M/L)\,\dd x = ML^2/3$~; autour du
centre, $\int_{-L/2}^{L/2}
x^2(M/L)\dd x = ML^2/12$. Disque~: un anneau de rayon $r$, de largeur
$\dd r$ et de masse $(2Mr/R^2)\dd r$ donne $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$~;
l'anneau et la sphère sont admis (la sphère demande une intégrale
double). Huygens~: avec $r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2 = r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$, le terme croisé se somme à zéro par définition de $G$.
\end{proof}

\begin{theorem}[Dynamique d'un solide autour d'un axe fixe]\label{thm:b1:systems-of-points:rotation}
\index{liaison pivot}\index{puissance d'un moment}
Pour un \omterm{def:b1:systems-of-points:solid}{solide} en \omterm{def:b1:systems-of-points:solid}{rotation autour d'un axe fixe} $\Delta$ dans un
\omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen},
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
les moments des forces extérieures par rapport à l'axe entraînent
l'accélération angulaire, et un moment $\mathcal M_\Delta$ délivre la
puissance $\mathcal M_\Delta\omega$. Une \emph{\omterm{thm:b1:systems-of-points:rotation}{liaison pivot}} idéale
(palier sans frottement) n'exerce aucun moment par rapport à son axe et
n'accélère donc ni ne freine la rotation.
\end{theorem}

\begin{proof}
Le \cref{thm:b1:angular-momentum:system} projeté sur $\Delta$, avec $L_\Delta
= J_\Delta\dot\theta$ et $J_\Delta$ constant pour un \omterm{def:b1:systems-of-points:solid}{solide}~; on multiplie
par $\dot\theta$ pour la forme énergétique (les forces intérieures d'un
\omterm{def:b1:systems-of-points:solid}{solide} ne travaillent pas~: les distances ne changent pas).
\end{proof}

\begin{example}[Pendule pesant~; pendule de torsion]\label{ex:b1:systems-of-points:pendulums}
\index{pendule pesant}\index{pendule de torsion}
Un \omterm{def:b1:systems-of-points:solid}{solide} de masse $M$ qui pivote autour d'un axe horizontal situé à la
distance $d$ de $G$~: le moment du poids vaut $-Mgd\sin\theta$, donc $J_\Delta\ddot\theta =
-Mgd\sin\theta$ et les \omterm{prop:b1:work-and-energy:stability}{petites oscillations} ont pour période
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
soit celle d'un \omterm{ex:b1:newton-dynamics:pendulum}{pendule simple} de longueur $J_\Delta/Md$ --- pour une
tige homogène pivotant à une extrémité, $2L/3$. Un disque suspendu à un
fil de constante de torsion $C$ (moment de rappel $-C\theta$)~:
$J\ddot\theta = -C\theta$, $T = 2\pi\sqrt{J/C}$ --- c'est la balance de
torsion qui a mesuré $G$ et la loi de Coulomb.
\end{example}

\begin{proposition}[Roulement sans glissement sur un plan incliné]\label{prop:b1:systems-of-points:rollingincline}
Un \omterm{def:b1:systems-of-points:solid}{solide} de révolution ($J = kMR^2$ par rapport à son axe) lâché sur un
plan incliné d'angle $\alpha$ roule sans glisser avec l'accélération
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
soit $\tfrac57 g\sin\alpha$ pour une sphère pleine, $\tfrac23 g\sin\alpha$
pour un disque et $\tfrac12 g\sin\alpha$ pour un cerceau ---
indépendamment de la masse et du rayon.
\end{proposition}

\begin{proof}
Énergie~: $Mgh = \tfrac12 Mv^2(1 + k)$ le long de la pente (le frottement
statique au point de contact, immobile, ne travaille pas), donc
$v^2 = 2ah$ avec le $a$ annoncé~; ou bien Newton pour $G$ et le théorème
du \omterm{def:b1:angular-momentum:L}{moment cinétique} en $G$ avec le frottement $f$~: $Ma = Mg\sin\alpha - f$,
$kMR^2\dot\omega = fR$, $a = R\dot\omega$.
\end{proof}

\begin{remark}[Systèmes déformables]\label{rem:b1:systems-of-points:deformable}
\index{système déformable}
Pour un système qui n'est pas rigide, le théorème de l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique}
s'écrit $\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$~: les forces
intérieures travaillent (muscles, ressorts, frottements internes). La
patineuse qui ramène ses bras, la plongeuse qui se groupe, le cycliste
qui se met en danseuse gagnent tous de l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} sans travail
extérieur --- le théorème du \omterm{def:b1:systems-of-points:cm}{centre de masse} reste vrai, l'énergie vient
de l'intérieur.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:1}
Des masses de \qty{2.0}{kg} en $x = 0$ et de \qty{3.0}{kg} en $x = \qty{1.0}{m}$~:
position de $G$. Le système Terre--Lune ($81:1$, \qty{384000}{km})~:
distance de $G$ au centre de la Terre~; comparer au rayon terrestre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:2}
Un fusil de \qty{4.0}{kg} tire une balle de \qty{10}{g} à \qty{800}{m/s}.
Vitesse de recul~; \omterm{thm:b1:work-and-energy:ke}{énergies cinétiques} de la balle et du fusil~; d'où
vient cette énergie~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:3}
Une voiture de \qty{1000}{kg} à \qty{20}{m/s} percute une voiture de
\qty{1500}{kg} à l'arrêt~; elles restent accrochées. Vitesse commune~;
fraction d'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} perdue.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:4}
Établir le \omterm{prop:b1:angular-momentum:inertia}{moment d'inertie} d'une tige homogène de masse $M$ et de
longueur $L$ par rapport à un axe perpendiculaire passant par une
extrémité, puis par son centre~; vérifier le théorème de Huygens entre
les deux.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:5}
\omterm{thm:b1:work-and-energy:ke}{Énergie cinétique} d'un cylindre homogène qui roule et d'un cerceau qui
roule, de même masse et de même vitesse, en multiples de $\tfrac12 Mv^2$.
Lequel s'arrête le premier sur un sol plat rugueux, et lequel monte le
plus haut sur une pente~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:6}
\omterm{prop:b1:systems-of-points:twobody}{Masse réduite} de HCl ($m_{Cl} = 35m_H$) et de l'atome d'hydrogène
($m_p = 1836m_e$)~; correction relative à une fréquence de vibration ou
orbitale calculée avec la seule masse légère.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:7}
Une tige homogène de \qty{1.0}{m} pivote librement autour d'une

\section{Problème~: le yo-yo}

\begin{problem}[{Problème du week-end --- un disque sur un axe fin au
bout d'une ficelle~: pourquoi il tombe si lentement, à quelle vitesse il
tourne en bas, pourquoi il y «~dort~», et de quel côté roule une bobine
dont on tire le fil}]
\label{pb:b1:systems-of-points:1}
Un yo-yo est un disque homogène de masse $M = \qty{50}{g}$ et de rayon $R =
\qty{3.0}{cm}$ muni d'un axe fin de rayon $r = \qty{0.40}{cm}$ autour
duquel est enroulée une ficelle de longueur $h = \qty{1.0}{m}$~;
l'extrémité supérieure de la ficelle est tenue fixe.
$g = \qty{9.81}{m/s^2}$~; $J = \tfrac12 MR^2$ par rapport à l'axe.

\medskip
\noindent\textbf{Partie I --- Cinématique et énergie.}
\begin{enumerate}
  \item Où se trouve le \omterm{def:b1:systems-of-points:cm}{centre de masse} $G$ du yo-yo~? Pourquoi~?
  \item Montrer, par le théorème de König, que son \omterm{thm:b1:work-and-energy:ke}{énergie cinétique} vaut $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$.
  \item Établir $J = \tfrac12 MR^2$ en intégrant sur des anneaux, et
        calculer $J$.
  \item La ficelle se déroulant sans glisser de l'axe, relier $v_G$ et
        $\omega$.
  \item En déduire $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ et calculer le
        facteur entre parenthèses. Commenter.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La descente.}
\begin{enumerate}[resume]
  \item Bilan des forces sur le yo-yo~; écrire le théorème du \omterm{def:b1:systems-of-points:cm}{centre de
        masse} (axe vertical dirigé vers le bas).
  \item Écrire le théorème du \omterm{def:b1:angular-momentum:L}{moment cinétique} par rapport à l'axe
        passant par $G$ (quelles forces ont un moment~?).
  \item Combiner avec la question~4 pour trouver l'accélération $a_G = g/(1 +
        R^2/2r^2)$~; la calculer.
  \item Calculer la tension de la ficelle et la comparer au poids.
  \item Temps de déroulement des \qty{1.0}{m} de ficelle, et vitesse
        atteinte~; comparer à une pierre lâchée de la même hauteur.
  \item \omterm{cor:b1:point-kinematics:circular}{Vitesse angulaire} en bas, en rad/s et en tours par minute.
  \item Séparer l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} en bas en ses parts de translation et
        de rotation.
  \item Vérifier le bilan énergétique~: comparer $Mgh$ à l'\omterm{thm:b1:work-and-energy:ke}{énergie
        cinétique} en bas, et dire pourquoi la tension de la ficelle,
        pourtant grande, ne travaille pas.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- En bas~: le «~dormeur~».}
Lorsque la ficelle est entièrement déroulée, elle arrête le \omterm{def:b1:systems-of-points:cm}{centre de
masse} en quelques millisecondes~; le yo-yo continue de tourner dans la
boucle au bout de la ficelle (il «~dort~»).
\begin{enumerate}[resume]
  \item Pourquoi la rotation survit-elle à l'arrêt de $G$~? (Quel
        théorème, et quelle force aurait-il fallu pour l'arrêter~?)
  \item Estimer la tension moyenne pendant l'arrêt si $G$ est stoppé en
        \qty{5}{ms}.
  \item La boucle frotte sur l'axe avec un moment de frottement $\Gamma =
        \qty{1.0e-4}{N.m}$~: combien de temps le yo-yo dort-il~?
  \item Une secousse sèche vers le haut fait mordre la ficelle sur
        l'axe~; le yo-yo, qui tourne, remonte alors. Par l'énergie,
        jusqu'à quelle hauteur pourrait-il monter si rien n'était perdu~?
        Pourquoi, en pratique, moins~?
  \item Pendant la remontée, la tension est-elle plus grande ou plus
        petite que le poids~? (Signe de $a_G$.)
  \item Que se passerait-il avec un yo-yo dont l'axe aurait le rayon $R$
        (ficelle enroulée sur la jante)~? Calculer son accélération de chute.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La bobine sur la table.}
Le même objet repose sur une table rugueuse, axe horizontal, et sa
ficelle quitte le \emph{bas} de l'axe horizontalement~; on tire avec une
force $F$. Il roule sans glisser.
\begin{enumerate}[resume]
  \item Au point de contact $C$ avec la table, quelle est la vitesse de
        la matière de la bobine~? En déduire que le mouvement est, à
        chaque instant, une rotation autour de $C$.
  \item Calculer le moment de $F$ par rapport à $C$ (\omterm{def:b1:angular-momentum:moment}{bras de levier}
        $R - r$), puis ceux du poids et de la réaction de la table.
  \item À l'aide du théorème du \omterm{def:b1:angular-momentum:L}{moment cinétique} en $C$ avec $J_C = J +
        MR^2$ (Huygens), trouver $a_G$ et montrer que la bobine roule
        \emph{vers} la main.
  \item On tire maintenant la ficelle sous un angle $\beta$ au-dessus de
        l'horizontale. Montrer que la bobine ne roule pas lorsque la
        droite d'action de $F$ passe par $C$, c'est-à-dire
        $\cos\beta = r/R$, et qu'elle s'éloigne de la main pour des
        tractions plus redressées. Calculer cet angle.
  \item Calculer $a_G$ pour $F = \qty{0.20}{N}$ tirée horizontalement, et
        la force de frottement nécessaire~; qu'est-ce qui limite $F$
        avant que la bobine ne glisse~?
  \item Récapituler~: les trois théorèmes employés (\omterm{def:b1:systems-of-points:cm}{centre de masse},
        \omterm{def:b1:angular-momentum:L}{moment cinétique} par rapport à un axe, énergie) et ce que chacun
        a tranché.
\end{enumerate}
\end{problem}
