\chapter{Optique géométrique~: rayons, réflexion, réfraction}\label{ch:b1:rays-reflection-refraction}

Une paille plantée dans un verre d'eau paraît brisée à la surface~; un
nageur qui regarde vers le haut depuis le fond d'un bassin voit tout le
ciel comprimé dans un disque brillant, cerné par un miroir du fond~; une
fibre de verre plus fine qu'un cheveu porte une communication
téléphonique d'un bout à l'autre d'un océan avec moins de pertes qu'un
fil de cuivre n'en subit en traversant une rue. Trois lois --- la
lumière va tout droit, se réfléchit à angles égaux, se courbe selon la
règle de Snell-Descartes --- rendent compte de tout cela~; ce chapitre
les énonce précisément, marque où elles cessent d'être vraies, et les
met au travail.

\section{Le modèle de l'optique géométrique}

\begin{definition}[Rayon lumineux~; milieu homogène]\label{def:b1:rays-reflection-refraction:ray}
\index{rayon lumineux}\index{optique géométrique}\index{milieu homogène}
Dans le modèle de l'\emph{optique géométrique}, la lumière se propage le
long de \emph{rayons lumineux}~: des courbes orientées, rectilignes dans
un milieu \emph{homogène} (mêmes propriétés en tout point), qui
transportent de l'énergie indépendamment les uns des autres. Un faisceau
est un ensemble de rayons~; une source ponctuelle émet des rayons dans
toutes les directions.
\end{definition}

\begin{definition}[Indice de réfraction]\label{def:b1:rays-reflection-refraction:index}
\index{indice de réfraction}\index{milieu transparent}
Un milieu \emph{transparent} laisse passer la lumière~; celle-ci y
progresse à la vitesse $v < c$. Son \emph{indice de réfraction} vaut
\[
n = \frac{c}{v} \geq 1 .
\]
Vide~: $n = 1$~; air~: $1.0003$~; eau~: $1.33$~; verre ordinaire~: $1.5$
à $1.6$~; diamant~: $2.42$. Une onde de fréquence $\nu$ garde sa
fréquence en entrant dans un milieu, si bien que sa longueur d'onde se
contracte de $\lambda_0 =
c/\nu$ dans le vide à $\lambda = \lambda_0/n$.
\end{definition}

\begin{remark}[Dispersion]\label{rem:b1:rays-reflection-refraction:dispersion}
\index{dispersion (optique)}\index{loi de Cauchy}
Dans la matière, l'indice dépend légèrement de la longueur d'onde --- le
milieu est \emph{dispersif}. Pour le verre dans le visible, la loi
empirique de Cauchy $n(\lambda) = A + B/\lambda^2$ convient bien~: $n$
est plus grand pour le bleu que pour le rouge, de $0.01$ à $0.02$
environ. Le prisme et l'arc-en-ciel, à la fin de ce chapitre, vivent de
cette différence.
\end{remark}

\begin{remark}[Limites du modèle]\label{rem:b1:rays-reflection-refraction:limits}
La lumière est une onde, et une onde qui franchit une ouverture de
largeur $a$ s'étale d'un angle de l'ordre de $\lambda/a$ (diffraction,
\cref{ch:b1:wave-propagation}). Les rayons en donnent une bonne
description tant que toute ouverture, toute lentille et tout obstacle
sont bien plus grands que la longueur d'onde (\qty{0.4}{} à
\qty{0.8}{\micro m} pour la lumière visible)~: un trou de \qty{1}{mm}
étale un faisceau de $10^{-3}$ radian, négligeable sur un banc~; un trou
de \qty{1}{\micro m} l'étale sur tout le demi-espace, et aucune
description en termes de rayons ne survit. Le \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} de \qty{9}{\micro m}
d'une fibre monomode de télécommunication est déjà hors du modèle.
\end{remark}

\begin{proposition}[Retour inverse de la lumière]\label{prop:b1:rays-reflection-refraction:reversibility}
\index{principe du retour inverse}
Le trajet d'un rayon ne dépend pas du sens dans lequel la lumière le
parcourt~: si la lumière va de $A$ à $B$ le long d'un trajet, la lumière
issue de $B$ suit le même trajet jusqu'à $A$.
\end{proposition}
\admitted

\section{Réflexion et réfraction}

\begin{definition}[Incidence]\label{def:b1:rays-reflection-refraction:incidence}
\index{plan d'incidence}\index{angle d'incidence}\index{dioptre}
Un rayon qui rencontre en un point $I$ la surface séparant deux milieux
(un \emph{dioptre}) fait avec la \emph{normale} à cette surface en $I$
l'\emph{angle d'incidence} $i_1$. Le \emph{plan d'incidence} contient le
rayon incident et la normale.
\end{definition}

\begin{theorem}[Lois de la réflexion et de la réfraction (Snell-Descartes)]\label{thm:b1:rays-reflection-refraction:snell}
\index{réflexion!lois de la}\index{réfraction!lois de la}\index{loi de Snell-Descartes}
À la surface séparant deux milieux d'indices $n_1$ (côté incident) et
$n_2$~:
\begin{enumerate}
  \item les rayons réfléchi et réfracté appartiennent au \omterm{def:b1:rays-reflection-refraction:incidence}{plan d'incidence}~;
  \item le rayon réfléchi fait avec la normale l'angle $i_1' = i_1$, de
        l'autre côté de la normale~;
  \item le rayon réfracté fait avec la normale un angle $i_2$ tel que
        \[
        n_1 \sin i_1 = n_2 \sin i_2 .
        \]
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[D'où viennent ces lois]\label{rem:b1:rays-reflection-refraction:fermat}
\index{principe de Fermat}
Les deux lois découlent du \emph{principe de Fermat}~: entre deux
points, la lumière suit le trajet de durée minimale. La réflexion à
angles égaux est le plus court chemin brisé touchant le miroir~; la \omterm{thm:b1:rays-reflection-refraction:snell}{loi
de Snell-Descartes} est le trajet qui échange un parcours plus long dans
le milieu rapide contre un parcours plus court dans le milieu lent
(\cref{exo:b1:rays-reflection-refraction:12}). Le volume de deuxième
année démontre les deux lois à partir de la nature ondulatoire de la
lumière à une interface.
\end{remark}

\begin{proposition}[Dans quel sens le rayon se courbe]\label{prop:b1:rays-reflection-refraction:bending}
En entrant dans un milieu plus réfringent ($n_2 > n_1$), un rayon
\emph{se rapproche} de la normale ($i_2 < i_1$)~; en entrant dans un
milieu moins réfringent, il \emph{s'en écarte} ($i_2 > i_1$). Un rayon
porté par la normale traverse sans être dévié.
\end{proposition}

\begin{proof}
$\sin i_2 = (n_1/n_2)\sin i_1$, et $\sin$ est croissant sur
$[0, \pi/2]$.
\end{proof}

\begin{example}[Profondeur apparente]\label{ex:b1:rays-reflection-refraction:depth}
\index{profondeur apparente}
Une pièce de monnaie repose à la profondeur $h$ sous l'eau ($n = 1.33$),
observée presque à la verticale. Un rayon issu de la pièce atteint la
surface sous un petit angle $i_2$ (dans l'eau) et en sort sous $i_1$
avec $\sin i_1 = n\sin i_2$~; pour de petits angles, $i_1 \approx n\,i_2$.
L'œil prolonge le rayon émergent vers l'arrière~: la pièce semble venir
d'une profondeur $h'$ telle que $\tan i_1 \approx x/h'$ et
$\tan i_2 \approx x/h$ pour le même point de la surface, situé à la
distance $x$ de la verticale~; d'où $h' = h\,i_2/i_1 = h/n$. Un bassin de
\qty{2.0}{m} de profondeur en paraît \qty{1.5}{m}~; un poisson voit le
pêcheur $1.33$ fois plus haut qu'il n'est.
\end{example}

\begin{theorem}[Réflexion totale]\label{thm:b1:rays-reflection-refraction:tir}
\index{réflexion totale}\index{angle limite}
La lumière qui passe d'un milieu d'indice $n_1$ vers un milieu moins
réfringent ($n_2 < n_1$) n'est réfractée que si $i_1 \leq i_c$, où
l'\emph{\omterm{thm:b1:rays-reflection-refraction:tir}{angle limite}} vérifie
\[
\sin i_c = \frac{n_2}{n_1} .
\]
Pour $i_1 > i_c$, il n'y a pas de rayon réfracté~: la surface renvoie
toute la lumière --- c'est la \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{réflexion totale}}.
\end{theorem}

\begin{proof}
La \omterm{thm:b1:rays-reflection-refraction:snell}{loi de Snell-Descartes} exige $\sin i_2 = (n_1/n_2)\sin i_1 > \sin i_1$~;
cette quantité dépasse $1$ dès que $\sin i_1 > n_2/n_1$, et alors aucun
angle $i_2$ n'existe. Pour $i_1 = i_c$, le rayon réfracté rase la
surface ($i_2 = 90^\circ$). Que \emph{toute} l'énergie soit alors
réfléchie est un énoncé sur les intensités, dont l'\omterm{def:b1:rays-reflection-refraction:ray}{optique géométrique}
ne traite pas~; le traitement ondulatoire du volume de deuxième année le
confirme.
\end{proof}

\begin{example}[Diamants et nageurs]\label{ex:b1:rays-reflection-refraction:tir}
Verre--air~: $\sin i_c = 1/1.5$, $i_c = 41.8^\circ$, ce qui explique
qu'un prisme à $45^\circ$ réfléchisse totalement et remplace un miroir
dans les jumelles. Diamant--air~: $i_c = 24.4^\circ$~; un diamant taillé
piège la lumière qui entre par sa table et la renvoie presque tout
entière par la table --- l'éclat vient de la \omterm{thm:b1:rays-reflection-refraction:tir}{réflexion totale}, le feu de
la dispersion. Eau--air~: $i_c = 48.8^\circ$~; en regardant vers le
haut, un plongeur voit tout le ciel dans un cône de demi-angle
$48.8^\circ$ (la «~fenêtre de Snell~») et, au-delà, le fond du bassin
réfléchi.
\end{example}

\section{La fibre optique}

\begin{definition}[Fibre à saut d'indice]\label{def:b1:rays-reflection-refraction:fiber}
\index{fibre optique}\index{cœur}\index{gaine}\index{ouverture numérique}
Une \emph{fibre optique à saut d'indice} est un \emph{cœur} de verre
cylindrique d'indice $n_1$ entouré d'une \emph{gaine} d'indice $n_2$
légèrement plus faible. La lumière qui entre dans le cœur à l'intérieur
d'un cône de demi-angle $\theta_a$ autour de l'axe est guidée par
\omterm{thm:b1:rays-reflection-refraction:tir}{réflexions totales} successives sur la surface cœur--gaine.
L'\emph{ouverture numérique} vaut $\mathrm{NA} = \sin\theta_a$.
\end{definition}

\begin{proposition}[Cône d'acceptance et étalement intermodal]\label{prop:b1:rays-reflection-refraction:fiber}
\index{angle d'acceptance}\index{dispersion intermodale}
Pour une fibre dont la face d'entrée est dans l'air,
\[
\sin\theta_a = \sqrt{n_1^2 - n_2^2} .
\]
Sur une longueur $L$, le rayon guidé selon l'axe et le rayon guidé le
plus incliné arrivent avec un décalage
\[
\Delta t = \frac{L\,n_1}{c}\left(\frac{n_1}{n_2} - 1\right)
\]
entre eux (\emph{\omterm{prop:b1:rays-reflection-refraction:fiber}{dispersion intermodale}}).
\end{proposition}

\begin{proof}
Un rayon entrant à l'angle $\theta$ de l'axe est réfracté sous $\theta'$
avec $\sin\theta = n_1\sin\theta'$ et rencontre la \omterm{def:b1:rays-reflection-refraction:fiber}{gaine} sous
l'incidence $i = 90^\circ - \theta'$. Le guidage impose
$\sin i \geq n_2/n_1$, c'est-à-dire $\cos\theta' \geq n_2/n_1$, soit
$\sin\theta' \leq
\sqrt{1 - n_2^2/n_1^2}$~; il suffit de multiplier par $n_1$. Le rayon
axial parcourt $L$ à la vitesse $c/n_1$~: $t_0 = Ln_1/c$. Le rayon le
plus incliné zigzague sous l'angle $\theta'_{\max}$ tel que
$\cos\theta'_{\max} = n_2/n_1$~; son trajet vaut
$L/\cos\theta'_{\max} = L n_1/n_2$, donc $t_1 = L n_1^2/(n_2 c)$, et
$\Delta t = t_1 - t_0$.
\end{proof}

\begin{example}[Les chiffres d'une fibre de télécommunication]\label{ex:b1:rays-reflection-refraction:fibernum}
$n_1 = 1.480$, $n_2 = 1.465$~: $\mathrm{NA} = \sqrt{2.190 - 2.146} =
0.21$, $\theta_a = 12^\circ$~; $i_c = 81.8^\circ$, donc les rayons
guidés restent à moins de $8.2^\circ$ de l'axe. Étalement intermodal~:
$\Delta t/L =
(1.48/c)(1.48/1.465 - 1) = \qty{51}{ns/km}$. Sur \qty{2}{km} une
impulsion s'étale de \qty{0.1}{\micro s}, ce qui plafonne le débit vers
\qty{5}{Mbit/s}~: le problème du week-end suit cette limitation jusqu'à
son remède, la fibre monomode.
\end{example}

\section{Le miroir plan}

\begin{proposition}[Image dans un miroir plan]\label{prop:b1:rays-reflection-refraction:mirror}
\index{miroir plan}\index{image virtuelle}\index{stigmatisme}
Tous les rayons issus d'un point $A$ et réfléchis par un \omterm{prop:b1:rays-reflection-refraction:mirror}{miroir plan}
semblent provenir du point $A'$ symétrique de $A$ par rapport au plan du
miroir~: $A'$ est l'\emph{image} de $A$. Elle est \emph{virtuelle}
(aucune lumière n'y passe) et le miroir est \emph{rigoureusement
stigmatique}~: tout point a une image ponctuelle exacte.
\end{proposition}

\begin{proof}
Un rayon issu de $A$ qui frappe le miroir en $I$ sous l'incidence $i$
repart sous $i$ de l'autre côté de la normale. La droite réfléchie et la
normale en $I$ font l'angle $i$, tout comme la droite $IA$~; par
symétrie par rapport au plan du miroir, la droite réfléchie est la
symétrique de la droite $IA$, et passe donc par le point symétrique $A'$
--- et cela pour tout $I$.
\end{proof}

\begin{proposition}[Rotation d'un miroir]\label{prop:b1:rays-reflection-refraction:rotation}
Faire tourner un \omterm{prop:b1:rays-reflection-refraction:mirror}{miroir plan} d'un angle $\alpha$ autour d'un axe de son
plan, perpendiculaire au \omterm{def:b1:rays-reflection-refraction:incidence}{plan d'incidence}, fait tourner le rayon
réfléchi de $2\alpha$.
\end{proposition}

\begin{proof}
La normale tourne de $\alpha$, donc l'incidence passe de $i$ à
$i + \alpha$~; le rayon réfléchi, à $i + \alpha$ de l'autre côté de la
nouvelle normale, s'est écarté de $\alpha$ (la normale) $+ \alpha$
(l'angle) $=
2\alpha$ de son ancienne direction.
\end{proof}

\begin{example}[Le levier du galvanomètre]\label{ex:b1:rays-reflection-refraction:galvanometer}
Un minuscule miroir collé sur le cadre d'un galvanomètre, éclairé par
une lampe, projette une tache sur une règle placée à \qty{2}{m}~: une
rotation de $\qty{1}{mrad}$ du cadre déplace la tache de $2 \times 10^{-3} \times
\qty{2}{m} = \qty{4}{mm}$ --- un levier optique qui amplifie sans
frottement, principe du galvanomètre à miroir et du scanner laser.
\end{example}

\section{Le prisme}

\begin{proposition}[Formules du prisme]\label{prop:b1:rays-reflection-refraction:prism}
\index{prisme}\index{déviation}\index{déviation minimale}
Un rayon qui traverse un prisme d'angle au sommet $A$ et d'indice $n$
dans l'air, dans le plan perpendiculaire à l'arête, avec les angles $i$
(entrée), $r$, $r'$ (à l'intérieur) et $i'$ (sortie) comptés depuis les
normales, vérifie
\[
\sin i = n \sin r, \qquad r + r' = A, \qquad \sin i' = n \sin r',
\]
et il est dévié de $D = i + i' - A$. La déviation est minimale lorsque
le trajet est symétrique, $i = i'$, $r = r' = A/2$~; alors
\[
n = \frac{\sin\frac{A + D_{\min}}{2}}{\sin\frac{A}{2}} .
\]
\end{proposition}

\begin{proof}
La \omterm{thm:b1:rays-reflection-refraction:snell}{loi de Snell-Descartes} à chaque face~; $r + r' = A$ parce que les
deux normales font l'angle $A$ (dans le quadrilatère formé par le
sommet, les deux points d'incidence et l'intersection des normales). La
déviation totale vaut $(i - r) + (i' - r') = i + i' - A$. Par le retour
inverse de la lumière, la déviation est la même fonction de $i$ et de
$i'$~; si le minimum était atteint en $i_0 \neq i_0'$, il le serait
aussi à l'incidence renversée $i_0'$, ce qui donnerait deux minima, que
la monotonie de $D(i)$ de part et d'autre de son minimum interdit~: le
minimum est en $i = i'$. Alors $r = r' = A/2$ et $D_{\min} = 2i - A$~;
en reportant $i = (A +
D_{\min})/2$ dans $\sin i = n\sin(A/2)$, on obtient la formule.
\end{proof}

\begin{example}[Mesurer un indice]\label{ex:b1:rays-reflection-refraction:prismnum}
Un prisme de $60^\circ$ que l'on tourne jusqu'à ce que la déviation d'un
faisceau jaune soit la plus faible~: $D_{\min} = 38.9^\circ$. Alors
$n = \sin 49.45^\circ/\sin 30^\circ
= 0.760/0.500 = 1.52$. Le minimum de déviation est la manière classique
de mesurer l'indice d'un verre --- ce minimum, net, se trouve en faisant
osciller le prisme et reste insensible aux petits défauts de réglage.
\end{example}

\begin{remark}[Prismes minces et spectroscopes]\label{rem:b1:rays-reflection-refraction:thin}
Pour $A$ petit et une incidence petite, les sinus valent les angles et
$D \approx (n - 1)A$, indépendamment de $i$~: un prisme mince dévie tous
les rayons du même angle --- un coin de verre est un «~prisme~» qui
courbe la lumière, et une lentille se révélera être un empilement de
tels coins. Comme $n$ dépend de $\lambda$, la déviation en dépend
aussi~: un prisme étale la lumière blanche en un spectre (le bleu le
plus dévié), c'est le \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} du spectroscope à prisme.
\end{remark}

\section{L'arc-en-ciel}

\begin{example}[L'arc-en-ciel de Descartes]\label{ex:b1:rays-reflection-refraction:rainbow}
\index{arc-en-ciel}
Un rayon de soleil entre dans une goutte de pluie sphérique sous
l'incidence $i$, s'y réfracte ($\sin i = n\sin r$), se réfléchit une
fois sur la face arrière, puis ressort après une seconde réfraction.
Chaque réfraction le dévie de $i - r$, la réflexion interne de
$\pi - 2r$~: la déviation totale vaut $D(i) = \pi + 2i - 4r$. Les rayons
remplissent la goutte pour toutes les valeurs de $i$~; l'observateur
voit la lumière la plus vive là où beaucoup de rayons repartent dans
presque la même direction, c'est-à-dire là où $D$ est stationnaire~:
$\dd D/\dd i = 2 - 4\,\dd r/\dd i
= 0$. En différentiant la \omterm{thm:b1:rays-reflection-refraction:snell}{loi de Snell-Descartes},
$\cos i = n\cos r\,\dd r/\dd i$, si bien que la condition s'écrit
$2\cos i = n\cos r$~; en élevant au carré et en utilisant
$n^2\cos^2 r = n^2 - \sin^2 i = n^2 - 1 + \cos^2 i$, il vient
\[
\cos^2 i = \frac{n^2 - 1}{3} .
\]
Pour l'eau, $n = 1.333$~: $i = 59.4^\circ$, $r = 40.2^\circ$,
$D_{\min} = 137.9^\circ$. La lumière revient du côté du Soleil, à
$180^\circ - 137.9^\circ = 42^\circ$ de la direction antisolaire~:
l'arc-en-ciel est un cône de demi-angle $42^\circ$ autour de l'ombre de
la tête de l'observateur. Le rouge ($n = 1.331$) est à $42.4^\circ$, le
violet ($n = 1.343$) à $40.6^\circ$ --- rouge à l'extérieur, violet à
l'intérieur.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:1}
Calculer la vitesse de la lumière dans l'eau ($n = 1.33$), dans le verre
($1.50$) et dans le diamant ($2.42$). Combien de temps la lumière
met-elle à traverser \qty{1.0}{km} d'une fibre d'indice $1.48$~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:2}
Un rayon dans l'air rencontre l'eau ($n = 1.33$) sous $30^\circ$, puis
sous $60^\circ$ de la normale~: trouver les angles réfractés. Un rayon
dans l'eau rencontre la surface sous $50^\circ$~: que se passe-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:3}
Calculer l'\omterm{thm:b1:rays-reflection-refraction:tir}{angle limite} pour verre--air ($n = 1.50$), diamant--air
($2.42$) et eau--air ($1.33$). Quel est le diamètre angulaire du disque
de ciel que voit un plongeur~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:4}
Une personne de \qty{1.70}{m}, les yeux \qty{10}{cm} sous le sommet du
crâne, se tient devant un miroir mural vertical. Trouver le plus petit
miroir (hauteur, et position de ses bords au-dessus du sol) dans lequel
elle se voit tout entière. La réponse dépend-elle de la distance au
miroir~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:5}
Une pièce repose à \qty{1.2}{m} de profondeur dans un bassin~: à quelle
profondeur paraît-elle à qui la regarde d'en haut~? Les yeux d'un
pêcheur sont à \qty{1.8}{m} au-dessus de l'eau~: à quelle hauteur
paraissent-ils à un poisson qui regarde vers le haut~? Justifier les
deux par des rayons peu inclinés.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:6}
Un prisme a $A = 60^\circ$ et $n = 1.52$. Calculer la \omterm{prop:b1:rays-reflection-refraction:prism}{déviation
minimale}. Montrer qu'un rayon ne peut émerger par la seconde face que si
$A \leq 2 i_c$, et vérifier que ce prisme convient.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:7}
Une fibre à saut d'indice a $n_1 = 1.50$, $n_2 = 1.48$. Calculer son
\omterm{def:b1:rays-reflection-refraction:fiber}{ouverture numérique}, son \omterm{prop:b1:rays-reflection-refraction:fiber}{angle d'acceptance}, le décalage intermodal par
kilomètre et le débit maximal qui en résulte sur \qty{1}{km}
(prendre $B_{\max} \approx 1/(2\Delta t)$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:8}
Un prisme mince d'angle au sommet $10^\circ$ a $n = 1.510$ pour le rouge
et $1.530$ pour le bleu. Calculer la déviation de chaque couleur et la
largeur angulaire du spectre qu'il produit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:9}
Un rayon rencontre une lame de verre à faces parallèles ($n = 1.50$,
épaisseur $e = \qty{2.0}{cm}$) sous $45^\circ$. Montrer qu'il émerge
parallèlement à sa direction incidente, décalé latéralement de
$d = e\,\sin(i - r)/\cos r$, et calculer $d$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:10}
L'arc secondaire vient des rayons réfléchis \emph{deux fois} à
l'intérieur de la goutte. Montrer que $D = 2\pi + 2i - 6r$, que la
déviation stationnaire vérifie $\cos^2 i = (n^2 - 1)/8$, et calculer
l'angle de l'arc secondaire depuis la direction antisolaire pour
$n = 1.333$. Où se situe-t-il par rapport à l'arc primaire, et dans quel
ordre ses couleurs se succèdent-elles~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:11}
Sur une route chaude, l'air juste au-dessus du bitume a pour indice
$n_{\text{chaud}} = 1.00026$, l'air plus frais au-dessus $n_{\text{froid}} =
1.00029$. En modélisant les couches par une interface nette, trouver
l'angle rasant limite (compté depuis la route) au-dessous duquel un
rayon venu du ciel est totalement réfléchi vers le haut. L'œil d'un
conducteur est à \qty{1.2}{m} au-dessus de la route~: au-delà de quelle
distance la route ressemble-t-elle à de l'eau~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:12}
Les points $A$ (à la hauteur $a$ au-dessus d'une interface) et $B$ (à la
profondeur $b$ en dessous) sont séparés horizontalement de $d$~; la
lumière va à la vitesse $v_1$ au-dessus et $v_2$ en dessous. Écrire la
durée du trajet brisé passant par le point d'interface d'abscisse $x$,
et montrer que cette durée est stationnaire exactement lorsque
$\sin i_1/v_1 = \sin i_2/v_2$ --- la \omterm{thm:b1:rays-reflection-refraction:snell}{loi de Snell-Descartes}.
\end{exercise}

\section{Problème~: la liaison par fibre optique}

\begin{problem}[{Problème du week-end --- un fil de verre sous la mer~:
jusqu'où et à quelle cadence une impulsion lumineuse peut-elle porter un
bit avant que la géométrie, le verre et la dispersion ne le brouillent}]
\label{pb:b1:rays-reflection-refraction:1}
Une liaison utilise une fibre à saut d'indice de \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} $n_1 = 1.480$, de
\omterm{def:b1:rays-reflection-refraction:fiber}{gaine} $n_2 = 1.465$, de rayon de \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} $a = \qty{25}{\micro m}$, à la
longueur d'onde $\lambda_0 = \qty{1550}{nm}$. Donnée~: $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Partie I --- La lumière dans le verre.}
\begin{enumerate}
  \item Définir l'indice et calculer la vitesse de la lumière dans le
        \omterm{def:b1:rays-reflection-refraction:fiber}{cœur}.
  \item Combien de temps une impulsion met-elle à parcourir \qty{50}{km}
        selon l'axe~?
  \item Calculer la fréquence de la lumière et sa longueur d'onde dans
        le \omterm{def:b1:rays-reflection-refraction:fiber}{cœur}.
  \item Pourquoi le \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} est-il gainé de verre d'indice plus faible
        plutôt que laissé nu dans l'air, ce qui donnerait un saut
        d'indice plus grand~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le guidage.}
\begin{enumerate}[resume]
  \item Calculer l'\omterm{thm:b1:rays-reflection-refraction:tir}{angle limite} à la surface cœur--gaine.
  \item En déduire le plus grand angle qu'un rayon guidé peut faire avec
        l'axe.
  \item Un rayon entre depuis l'air par la face d'entrée plane, sous
        l'angle $\theta$ avec l'axe. Montrer qu'il est guidé si
        $\sin\theta \leq
        \sqrt{n_1^2 - n_2^2}$, et calculer l'\omterm{prop:b1:rays-reflection-refraction:fiber}{angle d'acceptance}
        $\theta_a$.
  \item Donner l'\omterm{def:b1:rays-reflection-refraction:fiber}{ouverture numérique}. Pourquoi un saut d'indice plus
        grand rend-il la fibre plus facile à alimenter en lumière~?
  \item La fibre est courbée selon un cercle de rayon $R_b$. On
        considère un rayon parallèle à l'axe longeant le bord intérieur
        du \omterm{def:b1:rays-reflection-refraction:fiber}{cœur}~: montrer qu'il rencontre la paroi extérieure sous
        l'incidence $i$ telle que $\sin i = R_b/(R_b + 2a)$, et trouver
        le plus petit $R_b$ pour lequel il reste guidé.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La \omterm{prop:b1:rays-reflection-refraction:fiber}{dispersion intermodale}.}
\begin{enumerate}[resume]
  \item Exprimer les durées de parcours du rayon axial et du rayon guidé
        le plus incliné sur une longueur $L$.
  \item En déduire l'étalement $\Delta t$ par kilomètre, et sa valeur
        sur $L = \qty{2}{km}$.
  \item Un bit est une impulsion brève~; deux impulsions successives
        doivent rester séparées d'au moins $2\Delta t$ pour être
        distinguées. Estimer le débit maximal sur \qty{2}{km}.
  \item Montrer que le produit (débit) $\times$ (longueur) est une
        constante pour cette fibre, et le calculer en
        \unit{Mbit.s^{-1}.km}.
  \item Dans une fibre à gradient d'indice, l'indice décroît
        continûment de l'axe vers l'extérieur. Expliquer
        qualitativement pourquoi cela réduit l'étalement intermodal.
  \item Une fibre monomode a un diamètre de \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} d'environ
        \qty{9}{\micro m}. Comparer à $\lambda_0$ et expliquer pourquoi
        le modèle des rayons ne peut pas la décrire~; qu'y gagne-t-on~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Atténuation et dispersion chromatique.}
La fibre atténue la puissance de $\alpha = \qty{0.20}{dB/km}$ à
\qty{1550}{nm}~: $P(L) = P_0\,10^{-\alpha L/10}$. L'émetteur envoie
$P_0 = \qty{1.0}{mW}$~; le récepteur exige au moins
$P_{\min} = \qty{1.0}{\micro W}$. Comme $n$ dépend de $\lambda$, deux
longueurs d'onde voyagent à des vitesses légèrement différentes~: une
source de largeur spectrale $\Delta\lambda$ donne un étalement
$\Delta t_c = D\,\Delta\lambda\,L$ avec $D = \qty{17}{ps/(nm.km)}$.
\begin{enumerate}[resume]
  \item Quelle fraction de la puissance survit à \qty{100}{km}~?
  \item Trouver la longueur maximale avant qu'un amplificateur ne soit
        nécessaire.
  \item Les amplificateurs sont placés tous les \qty{100}{km}~: quelle
        marge, en dB et en rapport de puissances, cela laisse-t-il pour
        les connecteurs et les épissures~?
  \item Calculer $\Delta t_c$ sur \qty{100}{km} pour une source à
        diode électroluminescente ($\Delta\lambda = \qty{40}{nm}$) et
        pour un laser ($\Delta\lambda = \qty{0.10}{nm}$)~; en déduire le
        débit maximal de chacune sur cette longueur (même critère que
        ci-dessus).
  \item À \qty{850}{nm}, la silice atténue \qty{2.0}{dB/km}. Quelle
        distance le même bilan de puissance permettrait-il alors~?
        Pourquoi \qty{1550}{nm} est-elle la longueur d'onde des
        télécommunications~?
  \item Un câble coaxial de cuivre atténue environ \qty{30}{dB/km} aux
        fréquences nécessaires à \qty{1}{Gbit/s}. Comparer l'espacement
        des amplificateurs à celui de la fibre.
\end{enumerate}

\medskip
\noindent\textbf{Partie V --- La liaison complète.}
\begin{enumerate}[resume]
  \item Pour la fibre à saut d'indice sur \qty{2}{km}, quel effet limite
        le débit~: l'étalement intermodal, l'étalement chromatique
        (source laser) ou l'atténuation~? Donner le débit limite.
  \item Pour une fibre monomode sur \qty{100}{km} avec la source laser~:
        quel effet limite, et à quel débit~?
  \item La liaison monomode transporte $80$ lasers dont les longueurs
        d'onde sont espacées de \qty{0.8}{nm} (multiplexage en longueur
        d'onde), chacun au débit de la question précédente. Capacité
        totale~?
  \item Former le produit débit--longueur de la liaison à saut d'indice
        et d'un canal monomode~; de quel facteur la fibre monomode
        l'emporte-t-elle, et quel fait physique unique en est
        responsable~?
\end{enumerate}
\end{problem}
