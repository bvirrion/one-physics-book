\chapter{Moment cinétique}\label{ch:b1:angular-momentum}

Une patineuse qui tourne bras écartés les ramène contre elle et, sans
que personne ne la pousse, tourne deux fois plus vite. Une porte cède
sous un doigt posé sur la poignée et résiste à une épaule appuyée près
des gonds. Les planètes balaient des aires égales en des temps égaux ---
Kepler l'a remarqué il y a quatre siècles, bien avant que quiconque
puisse en dire la raison. Et la Lune, tirée par les marées qu'elle
soulève, s'éloigne de la Terre de quelques centimètres par an tandis que
nos journées s'allongent imperceptiblement. Derrière ces quatre
phénomènes se tiennent une seule grandeur vectorielle, le \omterm{def:b1:angular-momentum:L}{moment
cinétique}, et une seule loi~: sa dérivée égale le \omterm{def:b1:angular-momentum:moment}{moment} des forces. Ce
chapitre définit l'une et l'autre, démontre la loi pour un point puis
pour un système, et la met au travail sur des pendules, des orbites et
des objets qui tournent.

\section{Moment d'une force}

\begin{definition}[Moment en un point~; par rapport à un axe]\label{def:b1:angular-momentum:moment}
\index{moment d'une force}\index{couple}\index{bras de levier}
Le \emph{moment} (ou \emph{couple}) en un point $O$ d'une force
$\vect F$ appliquée en $M$ est le vecteur
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
perpendiculaire au plan de $\vect{OM}$ et $\vect F$, de norme
$OM\cdot F\sin\alpha = F\,d$, où $d$ --- le \emph{bras de levier} --- est
la distance de $O$ à la droite d'action de $\vect F$. Le moment par
rapport à un axe orienté $\Delta$ (de vecteur unitaire
$\vect e_\Delta$) passant par $O$ est le scalaire
$\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$, indépendant
du choix de $O$ sur $\Delta$~; il est nul si $\vect F$ est parallèle à
$\Delta$ ou la rencontre. Deux forces opposées portées par des droites
d'action différentes forment un \emph{couple de forces}, dont le moment
$\vect F\wedge\vect{AB}$ est le même en tout point.
\end{definition}

\begin{example}[La porte]\label{ex:b1:angular-momentum:door}
Une poussée de \qty{20}{N} perpendiculaire à une porte, à \qty{0.80}{m}
des gonds, a un \omterm{def:b1:angular-momentum:moment}{moment} de $\qty{16}{N.m}$ par rapport à l'axe des
gonds~; la même poussée à \qty{0.10}{m}, \qty{2}{N.m}~; dans le plan de
la porte (droite d'action passant par les gonds), zéro. La réaction des
gonds eux-mêmes, qui rencontre l'axe, n'a pas de \omterm{def:b1:angular-momentum:moment}{moment} par rapport à
lui~: c'est ce qui fait de l'axe l'endroit naturel où prendre les \omterm{def:b1:angular-momentum:moment}{moments}.
\end{example}

\section{Moment cinétique d'un point matériel}

\begin{definition}[Moment cinétique]\label{def:b1:angular-momentum:L}
\index{moment cinétique}
Le \emph{\omterm{def:b1:angular-momentum:moment}{moment} cinétique} en $O$ d'un \omterm{def:b1:point-kinematics:frame}{point matériel} de masse $m$ situé
en $M$ et de vitesse $\vect v$ est
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
et, par rapport à un axe $\Delta$ passant par $O$, $L_\Delta = \vect L_O\cdot\vect e_\Delta$.
Pour un mouvement dans un plan perpendiculaire à $\Delta$, en
\omterm{def:b1:point-kinematics:cylindrical}{coordonnées polaires} autour de l'axe, $L_\Delta = mr^2\dot\theta$~; pour
un cercle de rayon $R$, $L_\Delta = mR^2\omega = mRv$.
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$, $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$~:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[Théorème du moment cinétique]\label{thm:b1:angular-momentum:theorem}
\index{théorème du moment cinétique}
Dans un \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen}, pour un point $O$ fixe dans ce \omterm{def:b1:point-kinematics:frame}{référentiel},
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
pour un axe fixe $\Delta$. Si le \omterm{def:b1:angular-momentum:moment}{moment} total en $O$ est nul,
$\vect L_O$ se conserve.
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$, par la deuxième loi de Newton
($O$ fixe, de sorte que $\dd\vect{OM}/\dd t = \vect v$). On projette ensuite sur $\vect e_\Delta$.
\end{proof}

\begin{corollary}[Forces centrales~: mouvement plan et loi des aires]\label{cor:b1:angular-momentum:central}
\index{force centrale}\index{loi des aires}\index{vitesse aréolaire}
Un point soumis à une seule \emph{force centrale} (toujours portée par
$\vect{OM}$, vers le centre fixe $O$ ou à l'opposé) garde $\vect L_O$
constant. Son mouvement reste dans le plan passant par $O$ et
perpendiculaire à $\vect L_O$, et dans ce plan $r^2\dot\theta = C$
(constante)~: le rayon vecteur $\vect{OM}$ balaie de l'aire à la
\emph{\omterm{cor:b1:angular-momentum:central}{vitesse aréolaire}} constante
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$ pour $\vect F \parallel \vect{OM}$~:
$\vect L_O$ est constant. Comme $\vect{OM}\perp\vect L_O$ toujours, on
obtient le plan~; et $L_O =
mr^2\dot\theta$ est constant. L'aire balayée pendant $\dd t$ est celle du
triangle de côtés $\vect{OM}$ et $\vect v\dd t$~: $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$.
\end{proof}

\begin{example}[Le pendule par les moments]\label{ex:b1:angular-momentum:pendulum}
\omterm{ex:b1:newton-dynamics:pendulum}{Pendule simple}, axe $\Delta$ passant par le point de suspension et
perpendiculaire au plan d'oscillation~: $L_\Delta = m\ell^2\dot\theta$~;
la tension rencontre l'axe (\omterm{def:b1:angular-momentum:moment}{moment} nul), le \omterm{def:b1:angular-momentum:moment}{moment} du poids vaut
$-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{bras de levier} $\ell\sin\theta$, effet de rappel).
Le théorème donne $m\ell^2\ddot\theta =
-mg\ell\sin\theta$, l'équation de l'\cref{ex:b1:newton-dynamics:pendulum}
sans jamais écrire la tension --- c'est le principal avantage des
\omterm{def:b1:angular-momentum:moment}{moments}~: les forces qui passent par l'axe disparaissent.
\end{example}

\begin{example}[Tirer un palet vers le centre]\label{ex:b1:angular-momentum:puck}
Un palet glisse en cercle sur une glace sans frottement, retenu par un
fil qui passe par un trou au centre~; on tire le fil de sorte que le
rayon soit divisé par deux. La tension est centrale~: $L = mr^2\omega$
se conserve, donc $\omega$ est multipliée par $4$, la vitesse $r\omega$
par $2$, l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} par $4$ --- l'énergie supplémentaire est le
travail de la traction, à laquelle le palet résiste tout au long. Les
bras de la patineuse relèvent de la même physique, avec une masse
répartie (\cref{exo:b1:angular-momentum:4}).
\end{example}

\section{Systèmes de points et rotation autour d'un axe fixe}

\begin{theorem}[Moment cinétique d'un système]\label{thm:b1:angular-momentum:system}
\index{moment cinétique!d'un système}
Pour un système de points, avec $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
et $O$ fixe dans un \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen},
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
seuls comptent les \omterm{def:b1:angular-momentum:moment}{moments} des forces \emph{extérieures}~; les forces
intérieures, qui obéissent à la loi de l'action et de la réaction selon
la droite qui joint les points, ne contribuent en rien.
\end{theorem}

\begin{proof}
On somme les théorèmes ponctuels. Pour un \omterm{def:b1:angular-momentum:moment}{couple} de points intérieurs, $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$, puisque la force est portée par
$\vect{M_jM_i}$.
\end{proof}

\begin{proposition}[Rotation rigide autour d'un axe fixe]\label{prop:b1:angular-momentum:inertia}
\index{moment d'inertie}
Si les points gardent des distances $r_i$ fixes à un axe $\Delta$ et
tournent tous autour de lui à la même \omterm{cor:b1:point-kinematics:circular}{vitesse angulaire} $\omega$ (une
rotation rigide),
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
où $J_\Delta$ est le \emph{\omterm{prop:b1:angular-momentum:inertia}{moment d'inertie}} par rapport à $\Delta$, et
le théorème s'écrit $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$.
\end{proposition}

\begin{proof}
Chaque point contribue pour $m_ir_i^2\omega$ à $L_\Delta$ et pour
$\tfrac12 m_i(r_i\omega)^2$ à $E_k$. Le \cref{ch:b1:systems-of-points}
étend ce résultat aux solides continus.
\end{proof}

\begin{example}[La patineuse]\label{ex:b1:angular-momentum:skater}
Bras écartés, le corps d'une patineuse a $J \approx \qty{4}{kg.m^2}$ plus
deux bras de \qty{3}{kg} à \qty{0.8}{m}~: $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$. Bras ramenés (\qty{0.2}{m})~: $J_2 = \qty{4.2}{kg.m^2}$.
La réaction de la glace et le poids n'ont pas de \omterm{def:b1:angular-momentum:moment}{moment} par rapport à
l'axe vertical~: $J\omega$ se conserve et la vitesse de rotation croît
de $J_1/J_2 = 1.85$ --- de $2$ à $3.7$ tours par seconde --- tandis que
l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} croît du même facteur, payée par les muscles qui
ramènent les bras.
\end{example}

\begin{remark}[Statique]\label{rem:b1:angular-momentum:statics}
\index{équilibre d'un solide}
Un solide au repos a $\vect L_O = \vect 0$ constant~: les forces
extérieures doivent avoir une résultante nulle \emph{et} un \omterm{def:b1:angular-momentum:moment}{moment} total
nul en tout point. La seconde condition est la loi du levier, celle de
la balance, de la bascule et du pied-de-biche~: une petite force loin du
pivot équilibre une grande force près de lui.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:1}
Une force de \qty{50}{N} sur une clé, à \qty{0.30}{m} de l'écrou~:
\omterm{def:b1:angular-momentum:moment}{moment} lorsque la force est perpendiculaire au manche~; à $60^\circ$ de
celui-ci~; dans son prolongement.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:2}
\omterm{def:b1:angular-momentum:L}{Moment cinétique} de la Terre par rapport au Soleil (orbite circulaire,
$r = \qty{1.50e11}{m}$, $v = \qty{29.8}{km/s}$, $m = \qty{5.97e24}{kg}$),
et sa \omterm{cor:b1:angular-momentum:central}{vitesse aréolaire}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:3}
Un pendule conique (fil de \qty{1.0}{m}, à $30^\circ$ de la verticale,
\cref{exo:b1:newton-dynamics:9}). Calculer $L$ par unité de masse par
rapport à l'axe vertical passant par le point de suspension, et
expliquer pourquoi il se conserve alors que la tension du fil et le
poids agissent.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:4}
Le tronc d'une patineuse a $J = \qty{4.0}{kg.m^2}$ par rapport à l'axe de
rotation~; chaque bras (\qty{3.0}{kg}) est assimilé à un point à
\qty{0.80}{m}, puis à \qty{0.20}{m}. Vitesse de rotation après avoir
ramené les bras, en partant de \qty{2.0}{} tours par seconde~; rapport
des \omterm{thm:b1:work-and-energy:ke}{énergies cinétiques}~; d'où vient l'énergie supplémentaire~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:5}
Établir l'équation du pendule $\ell\ddot\theta + g\sin\theta = 0$ à
partir du théorème du \omterm{def:b1:angular-momentum:L}{moment cinétique} par rapport à l'axe de
suspension, et donner le \omterm{def:b1:angular-momentum:moment}{moment} de chaque force.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:6}
La distance de la Terre au Soleil varie de \qty{1.471e11}{m} (périhélie)
à \qty{1.521e11}{m} (aphélie). À l'aide de la \omterm{cor:b1:angular-momentum:central}{loi des aires}, trouver le
rapport de ses vitesses en ces points~; avec une vitesse moyenne de
\qty{29.8}{km/s}, donner les deux vitesses.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:7}
Un palet de \qty{0.20}{kg} tourne à \qty{2.0}{m/s} au bout d'un fil de
\qty{0.50}{m} passant par un trou~; on tire le fil jusqu'à ce que le
rayon vaille \qty{0.25}{m}. Nouvelle vitesse et nouvelle \omterm{cor:b1:point-kinematics:circular}{vitesse
angulaire}~; \omterm{thm:b1:work-and-energy:ke}{énergies cinétiques} avant et après~; travail de la traction.
Pourquoi la tension peut-elle travailler ici~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:8}
Une planche homogène de \qty{20}{kg} et de \qty{4.0}{m} repose sur un
pivot situé à \qty{1.5}{m} de son extrémité gauche~; un enfant de
\qty{30}{kg} s'assoit à cette extrémité. Où doit s'asseoir un enfant de
\qty{45}{kg} pour équilibrer~? Que supporte le pivot~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:9}
Une machine d'Atwood ($m_1 = \qty{2.0}{kg}$, $m_2 = \qty{1.0}{kg}$) est
suspendue à une poulie de rayon \qty{0.10}{m} et de \omterm{prop:b1:angular-momentum:inertia}{moment d'inertie}
\qty{0.010}{kg.m^2} sur laquelle le fil ne glisse pas. À l'aide du
théorème du \omterm{def:b1:angular-momentum:L}{moment cinétique} pour la poulie et de la loi de Newton pour
les masses, trouver l'accélération et les deux tensions. Comparer au cas
de la poulie sans masse.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:10}
Une comète passe à son périhélie à \qty{0.50}{AU} du Soleil à
\qty{40}{km/s}. Lorsqu'elle est à \qty{10}{AU}, quelle est la composante
de sa vitesse perpendiculaire au rayon vecteur~? La \omterm{cor:b1:angular-momentum:central}{loi des aires}
peut-elle donner sa vitesse entière~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:11}
Une étoile semblable au Soleil ($R = \qty{7.0e8}{m}$, période de rotation
\qty{25}{d}) s'effondre en une étoile à neutrons de rayon \qty{10}{km},
en gardant sa masse et son \omterm{def:b1:angular-momentum:L}{moment cinétique} ($J \propto MR^2$). Nouvelle
période~? Vitesse à l'équateur~? Que dit cette dernière de l'hypothèse~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:12}
Une perle glisse sans frottement le long d'une tige rectiligne qui
tourne dans un plan horizontal à $\omega$ constante autour d'un axe
vertical passant par une de ses extrémités. Écrire le $L_z$ de la
perle~; montrer qu'il n'est \emph{pas} conservé et calculer le \omterm{def:b1:angular-momentum:moment}{moment} que
la tige doit exercer~; en déduire la réaction transversale de la tige
sur la perle.
\end{exercise}

\section{Problème~: pourquoi la Lune s'en va}

\begin{problem}[{Problème du week-end --- les marées que la Lune
soulève sur la Terre la tirent en avant et freinent la Terre~: comment
le moment cinétique s'écoule de nos journées vers l'orbite lunaire,
centimètre par centimètre, et où passe l'énergie}]
\label{pb:b1:angular-momentum:1}
Données~: $G = \qty{6.67e-11}{SI}$~; Terre $M = \qty{5.97e24}{kg}$, $R =
\qty{6.37e6}{m}$, rotation $\Omega = 2\pi/(\qty{86164}{s})$, \omterm{prop:b1:angular-momentum:inertia}{moment
d'inertie} $J = 0.33\,MR^2$~; Lune $m = \qty{7.35e22}{kg}$, orbite traitée
comme circulaire, $r = \qty{3.84e8}{m}$~; la télémétrie laser lunaire
mesure $\dd r/\dd t = \qty{3.8}{cm/yr}$~; le jour s'allonge de
\qty{2.3}{ms} par siècle~; $1$ siècle $= \qty{3.16e9}{s}$.

\medskip
\noindent\textbf{Partie I --- Les \omterm{def:b1:angular-momentum:L}{moments cinétiques}.}
\begin{enumerate}
  \item Calculer la \omterm{cor:b1:point-kinematics:circular}{vitesse angulaire} orbitale $\omega$ de la Lune par la
        loi de Newton (orbite circulaire), et la vérifier sur le mois
        sidéral de \qty{27.3}{d}.
  \item Calculer le \omterm{def:b1:angular-momentum:L}{moment cinétique} orbital de la Lune $L_{\text{orb}} =
        mr^2\omega$.
  \item Montrer que, pour une orbite circulaire, $L_{\text{orb}} = m\sqrt{GMr}$,
        et en déduire $\dd L_{\text{orb}}/\dd r = L_{\text{orb}}/2r$.
  \item Calculer le \omterm{def:b1:angular-momentum:L}{moment cinétique} de rotation de la Terre $L_{\text{rot}} =
        J\Omega$ et le comparer à $L_{\text{orb}}$.
  \item La Lune tourne une fois par mois sur elle-même~: estimer son
        \omterm{def:b1:angular-momentum:L}{moment cinétique} propre ($J_m \approx 0.4\,mR_m^2$, $R_m = \qty{1.74e6}{m}$)
        et justifier qu'on le néglige.
  \item Quels \omterm{def:b1:angular-momentum:moment}{moments} extérieurs s'exercent sur le système Terre--Lune
        par rapport à son centre de masse~? Montrer qu'en bonne
        approximation le total $L = L_{\text{rot}} + L_{\text{orb}}$ se conserve.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le \omterm{def:b1:angular-momentum:moment}{couple} des marées.}
La Lune soulève sur la Terre deux bourrelets d'eau~; comme la Terre
tourne plus vite que la Lune n'orbite, le frottement entraîne ces
bourrelets en avant de la ligne Terre--Lune, d'un petit angle.
\begin{enumerate}[resume]
  \item Faire un schéma de la Terre, du bourrelet en avance et de la
        Lune. Dans quel sens agit l'attraction de la Lune sur le
        bourrelet le plus proche, par rapport à la rotation terrestre~?
  \item En déduire le signe du \omterm{def:b1:angular-momentum:moment}{moment} exercé sur la rotation de la
        Terre~: le jour s'allonge-t-il ou se raccourcit-il~?
  \item Par l'action et la réaction, quel \omterm{def:b1:angular-momentum:moment}{moment} s'exerce sur le
        mouvement orbital de la Lune~? $L_{\text{orb}}$ croît-il ou
        décroît-il~?
  \item D'après la question~3, $r$ croît-il ou décroît-il~? Et la vitesse
        orbitale de la Lune $v = \sqrt{GM/r}$~?
  \item Lever le paradoxe apparent~: la Lune est poussée en avant, et
        pourtant elle finit par aller moins vite.
  \item Vers quel état final tend cet échange~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Les nombres.}
\begin{enumerate}[resume]
  \item À partir de l'allongement du jour, calculer la variation
        relative $\Delta\Omega/\Omega$ par siècle.
  \item En déduire la variation de $L_{\text{rot}}$ par siècle.
  \item En déduire la variation de $L_{\text{orb}}$ par siècle.
  \item À l'aide de la question~3, la convertir en variation de $r$ par
        siècle puis par an. Comparer à la valeur de la télémétrie laser.
  \item De quelle fraction, et de combien de secondes, le mois sidéral
        s'allonge-t-il par siècle ($T \propto r^{3/2}$)~?
  \item Calculer la variation de l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} de rotation de la
        Terre par siècle ($\Delta E = L_{\text{rot}}\Delta\Omega$ au
        premier ordre).
  \item Calculer la variation de l'\omterm{thm:b1:work-and-energy:em}{énergie mécanique} orbitale de la Lune,
        $E_{\text{orb}} = -GMm/2r$, par siècle.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- L'énergie.}
\begin{enumerate}[resume]
  \item Montrer que la Terre perd plus d'énergie que la Lune n'en gagne,
        et calculer la différence par siècle ainsi que la puissance
        correspondante. Où passe-t-elle~?
  \item Comparer cette puissance à la consommation de l'humanité,
        environ \qty{18}{TW}, et au flux de chaleur interne de la Terre,
        environ \qty{47}{TW}.
  \item Dans l'état final de la question~12, le jour et le mois sont
        égaux~: $\Omega = \omega = \sqrt{GM/r^3}$. En écrivant la
        conservation de $L_{\text{rot}} + L_{\text{orb}}$, vérifier que
        $r \approx
        \qty{5.5e8}{m}$ convient, et donner la période commune en jours.
  \item Pourquoi cet état ne sera-t-il en fait jamais atteint~? (Penser
        aux marées solaires.)
  \item La Lune montre déjà toujours la même face à la Terre. Expliquer
        par le même mécanisme, et dire pourquoi cela lui est arrivé
        d'abord.
  \item Récapituler en trois lignes~: ce qui se conserve, ce qui
        s'échange, ce qui se dissipe.
\end{enumerate}
\end{problem}
