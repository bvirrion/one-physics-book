\chapter{Induction électromagnétique et applications}\label{ch:b1:induction}

Approchez un aimant d'une bobine et un courant apparaît dans la bobine~;
arrêtez l'aimant et le courant cesse. Faraday l'a découvert en 1831, après dix
ans de recherches, et c'est le fait sur lequel repose tout le monde
électrique~: chaque centrale, chaque transformateur, chaque moteur,
haut-parleur et microphone, chaque plaque à induction et chaque chargeur sans
fil convertit de l'énergie par la loi selon laquelle un \omterm{def:b1:induction:flux}{flux magnétique}
\emph{qui varie} engendre une \omterm{def:b1:dc-circuits:sources}{force électromotrice}. Ce chapitre énonce la loi
avec ses signes, lui donne les deux visages qu'elle prend (un champ variable
dans un circuit fixe, un circuit mobile dans un champ fixe), définit
l'\omterm{def:b1:induction:inductance}{inductance propre} et l'\omterm{def:b1:induction:inductance}{inductance mutuelle}, puis parcourt les machines
électromécaniques --- la barre glissante, l'alternateur, le transformateur, le
haut-parleur --- où la même loi transforme la puissance mécanique en \omterm{thm:b1:dc-circuits:kirchhoff}{puissance
électrique} et réciproquement, jamais gratuitement.

\section{La loi de Faraday}

\begin{definition}[Flux magnétique à travers un circuit]\label{def:b1:induction:flux}
\index{flux magnétique}\index{weber}
Orientons un circuit fermé $\Gamma$ (en choisissant un sens de parcours)~; sa
surface reçoit la normale $\vect n$ donnée par la règle de la main droite. Le
\emph{flux magnétique} à travers le circuit est $\Phi = \sum\vect B\cdot\vect n\,\dd S$ sur
une surface quelconque s'appuyant sur $\Gamma$ (le weber, $\qty{1}{Wb} = \qty{1}{T.m^2}$) ---
«~quelconque~», parce que le flux de $\vect B$ à travers une surface fermée est nul.
Pour un champ uniforme et une bobine plane de $N$ spires et d'aire $S$, $\Phi =
NBS\cos\alpha$, où $\alpha$ est l'angle entre $\vect B$ et $\vect n$.
\end{definition}

\begin{theorem}[Loi de l'induction de Faraday]\label{thm:b1:induction:faraday}
\index{loi de Faraday}\index{force électromotrice (induite)}\index{loi de Lenz}
Dès que le flux à travers un circuit varie, celui-ci est le siège d'une
\emph{\omterm{def:b1:dc-circuits:sources}{force électromotrice} induite}
\[
e = -\frac{\dd\Phi}{\dd t} ,
\]
comptée dans l'orientation du circuit, c'est-à-dire agissant comme une source
idéale de tension de f.é.m. $e$ dans ce sens~; dans un circuit de résistance $R$
elle fait passer le courant $i = e/R$. Le signe moins est la \emph{\omterm{thm:b1:induction:faraday}{loi de Lenz}}~:
le courant induit circule de manière à s'opposer à la variation de flux qui lui
donne naissance --- son propre champ, et les forces qu'il subit, combattent la cause.
\end{theorem}
\admitted

\begin{remark}[Deux visages d'une même loi]\label{rem:b1:induction:twofaces}
\index{induction de Neumann}\index{induction de Lorentz}
Le flux varie soit parce que $\vect B$ varie dans le temps à travers un circuit
fixe (\emph{\omterm{rem:b1:induction:twofaces}{induction de Neumann}}~: transformateur, plaque à induction), soit
parce que le circuit se déplace ou se déforme dans un champ permanent
(\emph{\omterm{rem:b1:induction:twofaces}{induction de Lorentz}}~: alternateur, haut-parleur). Dans le second cas,
la f.é.m. se calcule aussi directement~: les porteurs d'un élément conducteur
$\dd\vect l$ animé de la vitesse $\vect v$ subissent la force $q\vect v\wedge
\vect B$, équivalente à un champ électrique $\vect v\wedge\vect B$, de sorte que
$e = \oint(\vect v\wedge\vect B)\cdot\dd\vect l$ --- et cela s'accorde toujours
avec $-\dd\Phi/\dd t$. La \omterm{thm:b1:induction:faraday}{loi de Faraday} couvre les deux~; le volume de
deuxième année explique pourquoi les deux mécanismes donnent une seule formule.
\end{remark}

\begin{example}[Ordres de grandeur]\label{ex:b1:induction:orders}
Une bobine de $100$ spires et de \qty{10}{cm^2} dans le champ terrestre, que
l'on retourne en \qty{0.1}{s}~: $\Delta\Phi = 2 \times 100 \times 10^{-3} \times 5 \times 10^{-5} = \qty{1e-5}{Wb}$,
$e \sim \qty{0.1}{mV}$ --- mesurable avec un galvanomètre sensible, comme on a
autrefois cartographié le champ terrestre. La même bobine dans l'entrefer de
\qty{1}{T} d'un aimant, retirée en \qty{0.1}{s}~: \qty{1}{V}. Une bobine de
transformateur de $1000$ spires avec \qty{1}{T} à \qty{50}{Hz} sur \qty{10}{cm^2}~: $e_{\max} =
1000 \times 10^{-3} \times 314 = \qty{314}{V}$ --- le secteur.
\end{example}

\section{Inductance propre et inductance mutuelle}

\begin{definition}[Inductance]\label{def:b1:induction:inductance}
\index{inductance propre}\index{henry}\index{inductance mutuelle}
Le flux du champ propre d'un circuit à travers lui-même est proportionnel à son
courant~: $\Phi_{\text{propre}} = Li$, où $L > 0$ est son \emph{inductance propre}
(le henry, \unit{H}). Le flux que le circuit~1 envoie dans le circuit~2 vaut
$\Phi_{1 \to 2} = Mi_1$, et symétriquement $\Phi_{2 \to 1} = Mi_2$, où $M$ est
leur \emph{inductance mutuelle} (dont le signe dépend des orientations). Le
flux total du circuit~1 vaut alors $\Phi_1 = L_1i_1 + Mi_2$.
\end{definition}

\begin{proposition}[La bobine~; le solénoïde]\label{prop:b1:induction:inductor}
\index{bobine}\index{énergie magnétique}\index{densité d'énergie du champ magnétique}
Un circuit d'inductance $L$ parcouru par un courant variable est le siège de la
f.é.m. $e = -L\,\dd i/\dd t$~: en \omterm{thm:b1:dc-circuits:kirchhoff}{convention récepteur}, c'est la tension
$u = L\,\dd i/\dd t$ de la bobine utilisée depuis le
\cref{ch:b1:transient-regimes}. Il stocke l'énergie $W = \tfrac12Li^2$.
Un long solénoïde de $n$ spires par mètre, de longueur $\ell$ et de section $S$,
a $L = \mu_0n^2S\ell$~; l'énergie qu'il stocke vaut $\tfrac12\mu_0n^2I^2S\ell =
\dfrac{B^2}{2\mu_0}\,S\ell$~: l'\omterm{prop:b1:induction:inductor}{énergie magnétique} réside dans le champ, à la
densité $B^2/2\mu_0$, exactement comme l'énergie électrique à $\varepsilon_0E^2/2$.
\end{proposition}

\begin{proof}
Faraday avec $\Phi = Li$ et $L$ constante. Énergie~: la puissance reçue par
la bobine vaut $ui = Li\,\dd i/\dd t = \dd(\tfrac12Li^2)/\dd t$. Solénoïde~:
$B = \mu_0nI$ à l'intérieur, flux $BS$ à travers chacune des $n\ell$ spires, donc $\Phi =
\mu_0n^2S\ell I$. La densité générale est admise.
\end{proof}

\begin{example}[Un aimant d'IRM]\label{ex:b1:induction:mri}
\qty{1.5}{T} sur environ un mètre cube~: $B^2/2\mu_0 = 2.25/2.5 \times 10^{-6} =
\qty{0.9}{MJ/m^3}$ --- l'énergie d'une batterie de voiture, stockée dans du
vide. Avec un courant de \qty{500}{A} dans la bobine, $L = 2W/I^2 = \qty{7}{H}$. Si le
supraconducteur redevient brusquement résistif (une \emph{transition
résistive}), ce mégajoule devient de la chaleur en quelques secondes et fait
bouillir des centaines de litres d'hélium liquide~: la salle de l'aimant a un
évent pour cela.
\end{example}

\begin{proposition}[Le transformateur idéal]\label{prop:b1:induction:transformer}
\index{transformateur}
Deux bobines de $N_1$ et $N_2$ spires enroulées sur un même circuit magnétique
fermé en fer, de sorte que le même flux $\varphi$ traverse chaque spire, vérifient
\[
\frac{u_2}{u_1} = \frac{N_2}{N_1} , \qquad N_1i_1 + N_2i_2 = 0 \quad\text{(sans pertes en charge, sans courant magnétisant)},
\]
de sorte que $u_2i_2 = -u_1i_1$~: la puissance prise au primaire est fournie au
secondaire~; les tensions se transforment dans le rapport des nombres de
spires, les courants en raison inverse. Cela ne fonctionne qu'avec des courants variables.
\end{proposition}

\begin{proof}
$u_1 = N_1\,\dd\varphi/\dd t$ et $u_2 = N_2\,\dd\varphi/\dd t$ (\omterm{thm:b1:dc-circuits:kirchhoff}{convention
récepteur} des deux côtés, orientations choisies selon le flux commun)~: on
divise. La relation entre les courants est le \omterm{thm:b1:magnetostatics:ampere}{théorème d'Ampère} sur le circuit
magnétique~: sa circulation, soit $\mu_0\mu_r$ fois le total des ampères-tours,
devrait créer le flux~; pour un circuit magnétique idéal ($\mu_r \to \infty$),
il y faut des ampères-tours qui tendent vers zéro, d'où $N_1i_1 + N_2i_2 = 0$.
Admis sous cette forme.
\end{proof}

\section{Conducteurs en mouvement~: génératrices et moteurs}

\begin{proposition}[La barre glissante]\label{prop:b1:induction:rails}
\index{rails de Laplace}\index{conversion électromécanique}
Une barre de longueur $\ell$ glisse à la vitesse $v$ sur deux rails fermés par
une résistance $R$, dans un champ uniforme $B$ normal au plan. Elle est le
siège de la f.é.m. $e = B\ell v$, débite $i = B\ell v/R$, et subit la \omterm{thm:b1:magnetostatics:laplace}{force de
Laplace} $F = B\ell i = B^2\ell^2v/R$ qui \emph{s'oppose} à son mouvement.
La puissance mécanique dépensée pour la pousser, $Fv$, égale la \omterm{thm:b1:dc-circuits:kirchhoff}{puissance
électrique} $ei = Ri^2$ dissipée~: la barre est une génératrice. Alimentée au
contraire par une source $E$, c'est un moteur~: le courant $(E - B\ell v)/R$ la
pousse jusqu'à ce que la f.c.é.m. $B\ell v$ équilibre $E$, à la vitesse à vide $E/B\ell$.
\end{proposition}

\begin{proof}
Le flux vaut $B\ell x$, et $e = -B\ell\dot x$ dans l'orientation pour laquelle $\vect n$ est
selon $\vect B$~; la \omterm{thm:b1:magnetostatics:laplace}{force de Laplace} $i\vect\ell\wedge\vect B$ sur le courant
ainsi orienté est dirigée contre $\vect v$ (\cref{thm:b1:induction:faraday},
Lenz). En multipliant l'équation électrique $e = Ri$ par $i$ et l'équation
mécanique par $v$~: $Fv = B\ell iv = ei$. La puissance électromécanique
$ei$ et la puissance de Laplace $-Fv$ sont toujours opposées --- c'est la loi
de la conversion.
\end{proof}

\begin{proposition}[La spire tournante~: l'alternateur]\label{prop:b1:induction:alternator}
\index{alternateur}
Une bobine de $N$ spires et d'aire $S$ qui tourne à $\omega$ autour d'un axe
perpendiculaire à un champ uniforme $B$ a $\Phi = NBS\cos\omega t$ et
\[
e = NBS\omega\sin\omega t :
\]
soit une f.é.m. sinusoïdale d'\omterm{def:b1:wave-propagation:signal}{amplitude} $NBS\omega$ et de fréquence égale à
celle de la rotation. Dans une résistance $R$, elle délivre la \omterm{thm:b1:sinusoidal-impedance:power}{puissance moyenne}
$(NBS\omega)^2/2R$, fournie par le couple qui la maintient en rotation contre le
couple de Laplace $-\vect m\wedge\vect B$ de son propre courant.
\end{proposition}

\begin{proof}
On dérive le flux~; $\langle\sin^2\rangle = \tfrac12$. Le couple est celui du
\cref{thm:b1:magnetostatics:dipoleinfield} avec $m = Ni S$~; sa \omterm{thm:b1:sinusoidal-impedance:power}{puissance
moyenne} $\langle\Gamma\omega\rangle$ égale $\langle ei\rangle$ par le même
calcul que pour la barre.
\end{proof}

\begin{example}[Courants de Foucault]\label{ex:b1:induction:eddy}
\index{courants de Foucault}
Un conducteur massif qui se déplace dans un champ non uniforme, ou qui baigne
dans un champ variable, est traversé par un flux changeant le long d'une
infinité de chemins fermés~: des \emph{\omterm{ex:b1:induction:eddy}{courants de Foucault}} y circulent, qui
l'échauffent et --- par Lenz --- le freinent. Un aimant lâché dans un tube de
cuivre descend lentement~; les freins des camions et des montagnes russes
utilisent ce phénomène (sans contact, sans usure, sans défaillance --- mais
sans freinage à l'arrêt)~; la plaque à induction chauffe la casserole par les
mêmes courants, à \qty{25}{kHz}~; et les circuits magnétiques des
transformateurs sont feuilletés précisément pour les briser.
\end{example}

\section{Le haut-parleur~: un système couplé}

\begin{proposition}[Couplage électromécanique d'une bobine dans un champ radial]\label{prop:b1:induction:loudspeaker}
\index{haut-parleur}\index{microphone}
Une bobine dont le fil a la longueur totale $\ell$ est placée dans un champ
radial $B$, de sorte que chaque élément de fil soit perpendiculaire à $\vect B$~;
elle se déplace selon son axe à la vitesse $v$ et porte le courant $i$. Elle
subit alors la \omterm{thm:b1:magnetostatics:laplace}{force de Laplace} axiale $F = B\ell i$ et est le siège de la
f.c.é.m. $e = -B\ell v$ (en \omterm{thm:b1:dc-circuits:kirchhoff}{convention récepteur}, une tension $B\ell v$). Avec
une masse $m$, une suspension de raideur $k$ et un amortissement mécanique $h$,
alimentée sous la tension $u$ à travers la résistance $R$ et l'inductance $L$
de la bobine~:
\[
u = Ri + L\frac{\dd i}{\dd t} + B\ell v , \qquad
m\frac{\dd v}{\dd t} = -kx - hv + B\ell i .
\]
En multipliant par $i$ et par $v$ puis en ajoutant~: la \omterm{thm:b1:dc-circuits:kirchhoff}{puissance électrique} $ui$
égale les pertes Joule $Ri^2$, plus l'accroissement des énergies stockées
$\tfrac12Li^2 + \tfrac12mv^2 + \tfrac12kx^2$, plus les pertes mécaniques
$hv^2$ (qui contiennent le son)~: le terme de conversion $B\ell iv$ se compense
entre les deux équations. Pris à l'envers (mouvement imposé, $u$ lu), le même
appareil est un \emph{microphone} dynamique.
\end{proposition}

\begin{proof}
\omterm{thm:b1:magnetostatics:laplace}{Force de Laplace} sur chaque élément, $i\,\dd\vect l\wedge\vect B$, toutes
axiales et de même sens~; f.é.m. par $(\vect v\wedge\vect B)\cdot\dd\vect l$
sommé le long du fil. Le bilan énergétique est la somme des deux équations
multipliées comme il vient d'être dit.
\end{proof}

\begin{remark}[Pourquoi le couplage amortit]\label{rem:b1:induction:damping}
À basse fréquence ($L\omega \ll R$), $i \approx (u - B\ell v)/R$ et la force
vaut $B\ell u/R - (B^2\ell^2/R)\,v$~: le circuit électrique ajoute
l'amortissement $B^2\ell^2/R$ à l'amortissement mécanique. Un haut-parleur
branché sur un amplificateur (de faible résistance de sortie) est fortement
amorti --- sa membrane s'arrête quand le \omterm{def:b1:wave-propagation:signal}{signal} s'arrête --- et c'est le même
couplage qui permet au microphone de prélever de l'énergie au son
(\cref{pb:b1:induction:1}).
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:induction:1}
Une bobine de $100$ spires et de \qty{10}{cm^2} dans un champ uniforme de
\qty{0.20}{T} faisant \qty{30}{\degree} avec sa normale. Flux~; f.é.m. moyenne
si le champ est coupé en \qty{10}{ms}~; puis en \qty{1}{ms}~; sens du courant
induit par rapport au champ (Lenz).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:2}
Une barre de \qty{20}{cm} glisse à \qty{2.0}{m/s} sur des rails fermés par
\qty{0.10}{\ohm}, dans \qty{0.50}{T}. F.é.m., courant, \omterm{thm:b1:magnetostatics:laplace}{force de Laplace}, force
que doit exercer celui qui pousse, puissance mécanique, \omterm{thm:b1:dc-circuits:kirchhoff}{puissance électrique}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:3}
Solénoïde de \qty{1000}{spires/m}, de longueur \qty{20}{cm} et de section \qty{4.0}{cm^2}~:
inductance~; énergie sous \qty{5.0}{A}~; \omterm{def:b1:transient-regimes:firstorder}{constante de temps} avec une résistance
de \qty{1.0}{\ohm}~; champ intérieur et densité d'énergie, pour vérifier $B^2/2\mu_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:4}
Un transformateur de \qty{230}{V} vers \qty{12}{V}~: rapport des nombres de
spires~; nombre de spires au primaire pour un secondaire de $50$ spires~;
courant au primaire lorsque le secondaire débite \qty{2.0}{A}. Pourquoi ne
fonctionne-t-il pas sur une batterie de voiture~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:5}
\omterm{thm:b1:induction:faraday}{Loi de Lenz}~: donner le sens du courant induit et la direction de la force dans
chaque cas --- (a) un pôle nord s'approche d'un anneau selon son axe~; (b) un
anneau est plongé dans un champ qui croît~; (c) on laisse tomber un anneau sur
un aimant~; (d) un aimant tombe dans un long tube de cuivre. Pour (d),
expliquer par l'énergie pourquoi il tombe à petite vitesse constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:6}
Alternateur~: $N = 200$, $S = \qty{20}{cm^2}$, $B = \qty{0.10}{T}$, \qty{50}{Hz}.
F.é.m. maximale et efficace~; courant maximal et couple maximal dans
\qty{10}{\ohm}~; \omterm{thm:b1:dc-circuits:kirchhoff}{puissance électrique} moyenne~; comparer à la puissance
mécanique moyenne et au couple instantané à l'instant où la f.é.m. s'annule.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:7}
\emph{Frein magnétique.} Une spire carrée de côté $a = \qty{10}{cm}$ et de
résistance $R = \qty{1.0}{m\ohm}$, de masse \qty{1.0}{kg}, quitte une région de
champ uniforme $B = \qty{1.0}{T}$ à la vitesse $v$ (un côté est encore dans le
champ). F.é.m., courant, force de freinage~; équation du mouvement et \omterm{def:b1:transient-regimes:firstorder}{constante
de temps}~; distance parcourue avant l'arrêt depuis \qty{1.0}{m/s}~; où est
passée l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:8}
Une petite bobine ($N_2 = 50$ spires, section \qty{2.0}{cm^2}) est placée dans
un long solénoïde ($n_1 = \qty{2000}{m^{-1}}$), axes confondus. \omterm{def:b1:induction:inductance}{Inductance
mutuelle}~; f.é.m. dans la bobine lorsque le courant du solénoïde croît à
\qty{100}{A/s}, puis lorsqu'il vaut $\qty{1.0}{A}$ à \qty{10}{kHz}. $M$
dépend-il de celle des deux bobines qui porte le courant~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:9}
Énergie dans un champ~: $B = \qty{1.5}{T}$ sur \qty{1.0}{m^3}, comme dans un
aimant d'IRM. Énergie~; inductance pour $I = \qty{500}{A}$~; hauteur à laquelle
cette énergie élèverait une voiture de \qty{1}{t}~; volume d'hélium liquide
qu'elle fait bouillir lors d'une transition résistive (\qty{2.6}{kJ} par litre).
Énergie du champ terrestre (\qty{5e-5}{T}) dans une pièce de \qty{50}{m^3}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:10}
\emph{Les rails en moteur.} La barre de l'\cref{exo:b1:induction:2} (masse $m$,
sans frottement) est reliée à une source $E$ à travers la résistance $R$.
(a) Équations électrique et mécanique~; montrer que $m\,\dd v/\dd t = B\ell(E -
B\ell v)/R$. (b) \omterm{prop:b1:newton-dynamics:terminal}{Vitesse limite} et \omterm{def:b1:transient-regimes:firstorder}{constante de temps}. (c) Bilan de puissance~:
puissance de la source $=$ Joule $+$ mécanique~; \omterm{def:b1:heat-engines:efficiency}{rendement} en fonction de
$v$. (d) Application~: $E = \qty{2.0}{V}$, $m = \qty{50}{g}$, le reste comme
dans l'\cref{exo:b1:induction:2}. (e) Un canon électromagnétique accélère un
projectile de \qty{10}{g} avec \qty{1}{MA} sur \qty{5}{m} dans \qty{1}{T}~:
vitesse de sortie (\omterm{def:b1:units-dimensions:oom}{ordre de grandeur}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:11}
\emph{Le disque de Faraday.} Un disque de cuivre de rayon $R$ tourne à $\omega$
autour de son axe dans un champ uniforme $B$ porté par l'axe~; des balais
touchent le centre et le bord. (a) F.é.m. entre le centre et le bord, en sommant $(\vect v\wedge
\vect B)\cdot\dd\vect l$ le long d'un rayon~: $e = \tfrac12B\omega R^2$. (b) Application~:
$R = \qty{10}{cm}$, \qty{3000}{tr/min}, \qty{1.0}{T}. (c) Pourquoi est-ce une
machine à basse tension et à fort courant~; courant dans \qty{1}{m\ohm} et
couple de freinage. (d) Retrouver $e$ par la \omterm{thm:b1:induction:faraday}{loi de Faraday} appliquée au
circuit balai--rayon--bord--balai~: quel flux varie~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:12}
\emph{Plaque à induction.} La bobine placée sous le verre porte $i_1 =
I_1\cos\omega t$ avec $I_1 = \qty{30}{A}$ à \qty{25}{kHz}~; le fond de la
casserole se comporte comme une spire unique de résistance $R_p = \qty{10}{m\ohm}$
couplée par $M = \qty{1.0}{\micro H}$ (on néglige l'inductance de cette spire).
(a) F.é.m. induite dans la casserole, valeur maximale. (b) Courant et \omterm{thm:b1:sinusoidal-impedance:power}{puissance
moyenne} dissipée dans la casserole. (c) Pourquoi une fréquence élevée, et
pourquoi la chaleur apparaît-elle dans la casserole et non dans le verre~?
(d) Le courant de la casserole réagit sur la bobine~: f.é.m. qu'il y induit
(valeur maximale), et supplément de puissance que le générateur doit fournir
--- vérifier que cela vaut (b).
\end{exercise}

\section{Problème~: la dynamo de bicyclette et le haut-parleur}

\begin{problem}[{Problème du week-end --- deux convertisseurs électromécaniques
démontés~: une dynamo qui se régule toute seule, et un haut-parleur dont
l'amplificateur est aussi le frein}]
\label{pb:b1:induction:1}

\medskip
\noindent\textbf{Partie I --- La dynamo de bicyclette.}
Une bobine de $N = 300$ spires et d'aire $S = \qty{3.0}{cm^2}$ tourne dans un
champ uniforme $B = \qty{0.30}{T}$~; son axe porte un galet de rayon
$r = \qty{1.0}{cm}$ pressé contre le pneu, de sorte que la bobine tourne à
$\omega = v/r$ pour une vitesse $v$ du vélo. Résistance de la bobine $r_c = \qty{4.0}{\ohm}$,
inductance $L = \qty{20}{mH}$~; la lampe est une résistance $R = \qty{12}{\ohm}$.
\begin{enumerate}
  \item Flux à travers la bobine et f.é.m. induite en fonction du temps~;
        \omterm{def:b1:wave-propagation:signal}{amplitude} et fréquence à $v = \qty{18}{km/h}$.
  \item En négligeant d'abord $L$~: courant efficace et puissance dans la
        lampe à \qty{18}{km/h}~; une lampe «~\qty{6}{V}, \qty{3}{W}~» est-elle à l'aise~?
  \item Puissance mécanique moyenne que le cycliste fournit à la dynamo, et
        force correspondante sur le pneu (comparer au frottement de
        roulement, environ \qty{2}{N}).
  \item Expliquer par la \omterm{thm:b1:induction:faraday}{loi de Lenz} pourquoi la dynamo résiste, et pourquoi
        une lampe qui s'allume «~gratuitement~» ne contredit pas la
        conservation de l'énergie.
  \item Tenir compte de $L$ à présent~: \omterm{def:b1:sinusoidal-impedance:impedance}{impédance} complexe du circuit~;
        montrer que l'\omterm{def:b1:wave-propagation:signal}{amplitude} du courant vaut $I = NBS\omega/\sqrt{(R + r_c)^2 + L^2\omega^2}$
        et qu'elle sature à grande vitesse~; sa limite, et la vitesse à
        laquelle elle atteint $70\%$ de cette limite.
  \item Sans l'inductance, que recevrait la lampe à \qty{54}{km/h} dans une
        descente~? Et avec elle~?
  \item Courant et puissance dans la lampe à \qty{9}{km/h} puis à \qty{36}{km/h}~;
        commenter l'autorégulation.
  \item Les dynamos réelles font tourner un aimant à l'intérieur d'une bobine
        fixe. L'analyse change-t-elle~? Quel problème pratique cela supprime-t-il~?
  \item Que fait la lampe quand le vélo s'arrête à un feu rouge, et comment
        les éclairages modernes y remédient-ils~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le haut-parleur.}
Bobine mobile~: longueur de fil $\ell = \qty{5.0}{m}$ dans un champ radial $B =
\qty{1.0}{T}$, résistance $R = \qty{8.0}{\ohm}$, inductance $L = \qty{0.50}{mH}$~;
masse mobile (bobine et membrane) $m = \qty{10}{g}$~; raideur de la suspension
$k = \qty{1000}{N/m}$, amortissement mécanique $h = \qty{1.0}{N.s/m}$. Régime
sinusoïdal, \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes}.
\begin{enumerate}[resume]
  \item Force sur la bobine pour un courant $i$~; f.c.é.m. pour une vitesse $v$
        (établir les deux, avec les signes).
  \item Écrire l'équation électrique et l'équation mécanique.
  \item Éliminer $i$ et montrer que
        $\underline v = \dfrac{B\ell\,\underline u}{(R + jL\omega)(jm\omega + h + k/j\omega) + B^2\ell^2}$.
  \item Aux fréquences où $L\omega \ll R$, montrer que le couplage agit comme
        un amortissement mécanique supplémentaire $B^2\ell^2/R$~; sa valeur~;
        fréquence de résonance de la membrane et \omterm{def:b1:transient-regimes:secondorder}{facteur de qualité} avec et
        sans l'amortissement électrique.
  \item L'amplificateur délivre un courant $i$ d'\omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{A} à \qty{1}{kHz}~:
        force, accélération, élongation et vitesse de la membrane en
        \omterm{def:b1:wave-propagation:signal}{amplitude} (régime contrôlé par la masse)~; \omterm{def:b1:wave-propagation:signal}{amplitude} de la f.c.é.m.
        comparée à $Ri$.
  \item Même courant à la fréquence de résonance~: \omterm{def:b1:wave-propagation:signal}{amplitude} de la vitesse
        (régime contrôlé par l'amortissement) et f.c.é.m.~; commenter.
  \item \omterm{thm:b1:dc-circuits:kirchhoff}{Puissance électrique} à \qty{1}{kHz}~; puissance mécanique dissipée dans
        $h$ (assimilée au son et aux pertes de la suspension)~; \omterm{def:b1:heat-engines:efficiency}{rendement}.
  \item Pourquoi le fait que la pression acoustique soit proportionnelle à
        l'accélération de la membrane (on l'admettra) explique-t-il qu'une
        membrane contrôlée par sa masse donne une réponse plate au-dessus de
        la résonance~?
  \item Pourquoi le champ doit-il être radial, et pourquoi un tweeter est-il petit~?
  \item Esquisser le module de l'\omterm{def:b1:sinusoidal-impedance:impedance}{impédance} $\underline u/\underline i$ vue par
        l'amplificateur de \qty{10}{Hz} à \qty{10}{kHz}~: valeur à basse
        fréquence, à la résonance (en utilisant la vitesse contrôlée par
        l'amortissement), et à \qty{10}{kHz}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- À l'envers, et les comptes.}
\begin{enumerate}[resume]
  \item Le même appareil en microphone~: un son fait bouger la membrane à
        $v = \qty{1.0}{mm/s}$~; tension en circuit ouvert.
  \item Chargé par $R_L = \qty{600}{\ohm}$~: courant, et force de freinage sur
        la membrane~; montrer que le microphone prélève de la puissance au son
        par un amortissement effectif $B^2\ell^2/(R + R_L)$.
  \item La bobine est enroulée sur un cylindre d'aluminium (la carcasse), une
        spire fermée de longueur \qty{0.10}{m} et de résistance \qty{1.0}{m\ohm}~:
        amortissement qu'elle ajoute~; pourquoi les carcasses sont fendues.
  \item On frappe la membrane d'un haut-parleur dont les bornes sont laissées
        ouvertes, puis celle d'un haut-parleur dont les bornes sont
        court-circuitées~: lequel résonne le plus longtemps, et pourquoi~?
  \item Écrire le bilan énergétique du haut-parleur sur une période en régime
        établi, en nommant chaque terme.
  \item Récapituler~: pour chaque appareil, la loi employée et les deux
        nombres à retenir.
\end{enumerate}
\end{problem}
