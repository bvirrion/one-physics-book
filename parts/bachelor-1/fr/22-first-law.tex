\chapter{Le premier principe de la thermodynamique}\label{ch:b1:first-law}

Gonflez un pneu de vélo et le corps de la pompe devient trop chaud pour
qu'on le tienne. Un moteur Diesel n'a pas de bougie~: il comprime son air
vingt fois, et l'air devient assez chaud pour enflammer le carburant tout
seul. Une bouteille de gaz comprimé se couvre de givre en se vidant. Dans
chaque cas l'énergie change de forme --- le travail devient \omterm{def:b1:first-law:heat}{chaleur}, la
\omterm{def:b1:first-law:heat}{chaleur} devient travail, tous deux deviennent \omterm{def:b1:kinetic-theory:U}{énergie interne} --- et le
premier principe de la thermodynamique est la comptabilité qui fait tomber
les comptes juste. Ce chapitre définit le travail et la \omterm{def:b1:first-law:heat}{chaleur} d'un
\omterm{def:b1:first-law:transformation}{système thermodynamique}, énonce le premier principe, introduit l'\omterm{def:b1:kinetic-theory:U}{énergie
interne} et l'\omterm{def:b1:first-law:enthalpy}{enthalpie} comme les grandeurs qu'il conserve, et l'applique
aux \omterm{def:b1:first-law:transformation}{transformations} du \omterm{def:b1:kinetic-theory:model}{gaz parfait} et à la calorimétrie.

\section{Transformations, travail et chaleur}

\begin{definition}[Système, transformation]\label{def:b1:first-law:transformation}
\index{système thermodynamique}\index{transformation}\index{transformation quasi statique}\index{transformation réversible}
Un \emph{système thermodynamique} est la matière contenue à l'intérieur
d'une frontière choisie~; tout le reste est le \emph{milieu extérieur}. Une
\emph{transformation} le fait passer d'un état d'équilibre à un autre. Elle
est \emph{quasi statique} si le système traverse une succession d'états
d'équilibre (assez lentement pour que $P$ et $T$ soient définis partout),
et \emph{réversible} si, de plus, inverser les conditions extérieures
inverse le chemin suivi (ni frottement, ni écart fini de température ou de
pression avec l'extérieur). On les nomme d'après ce qui reste constant~:
\emph{isotherme} ($T$), \emph{isobare} ($P$), \emph{isochore} ($V$)~;
\emph{adiabatique} si aucune \omterm{def:b1:first-law:heat}{chaleur} n'est échangée~; \emph{monobare}
(\emph{monotherme}) si la pression (la température) extérieure est
constante, quoi qu'il arrive à l'intérieur.
\end{definition}

\begin{proposition}[Travail des forces de pression]\label{prop:b1:first-law:work}
\index{travail!des forces de pression}
Un système dont le volume varie de $\dd V$ contre une pression extérieure
$P_{\mathrm{ext}}$ reçoit le travail
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
Pour une \omterm{def:b1:first-law:transformation}{transformation quasi statique}, $P_{\mathrm{ext}} = P$ (la pression
du système lui-même) et $W = -\int P\,\dd V$ est l'opposé de l'aire sous le
chemin dans le diagramme de Clapeyron~: positif à la compression, négatif
à la détente.
\end{proposition}

\begin{proof}
Un piston d'aire $S$ poussé par l'extérieur avec la force
$P_{\mathrm{ext}}S$ se déplace de $\dd x$ vers l'extérieur~: le milieu
extérieur fournit au système le travail
$-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$. Toute frontière est
une réunion de tels pistons. En quasi statique, le piston est en équilibre
et $P_{\mathrm{ext}} = P$ à une différence infinitésimale près.
\end{proof}

\begin{definition}[Chaleur]\label{def:b1:first-law:heat}
\index{chaleur}\index{thermostat}
La \emph{chaleur} $Q$ est l'énergie transférée au système autrement que par
un travail macroscopique --- par chocs moléculaires sur une paroi
(conduction), par circulation d'un fluide (convection), par rayonnement. Un
\emph{thermostat} est un corps assez grand pour que sa température ne change
pas, quelle que soit la chaleur qu'il échange. Convention de signe dans tout
l'ouvrage~: $W$ et $Q$ sont \emph{reçus} par le système, positifs lorsqu'ils entrent.
\end{definition}

\section{Le premier principe}

\begin{theorem}[Premier principe de la thermodynamique]\label{thm:b1:first-law:firstlaw}
\index{premier principe de la thermodynamique}\index{énergie interne}
Un \omterm{def:b1:newton-dynamics:conservation}{système fermé} possède une fonction d'état, son \emph{\omterm{def:b1:kinetic-theory:U}{énergie interne}}
$U$ (extensive), telle que pour toute \omterm{def:b1:first-law:transformation}{transformation}
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
où $E_k$ est l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} macroscopique (nulle pour un système au
repos)~: l'énergie reçue sous forme de travail et de \omterm{def:b1:first-law:heat}{chaleur} est stockée en
\omterm{def:b1:kinetic-theory:U}{énergie interne}. $W$ et $Q$ dépendent chacun du chemin~; leur somme, non.
\end{theorem}
\admitted

\begin{remark}[Ce qu'il dit]\label{rem:b1:first-law:meaning}
C'est la conservation de l'énergie, la \omterm{def:b1:first-law:heat}{chaleur} étant reconnue comme un
transfert d'énergie (expérience des palettes de Joule~: un travail mesuré
élève toujours autant la température de l'eau qu'une \omterm{def:b1:first-law:heat}{chaleur} mesurée). Dire
que «~$U$ est une fonction d'état~» signifie que $\Delta U$ est fixé par les
états initial et final --- de sorte qu'un cycle a $\Delta U = 0$ et
$W + Q = 0$~: un moteur qui fournit du travail doit recevoir de la \omterm{def:b1:first-law:heat}{chaleur}.
Le contenu microscopique de $U$ est celui de la
\cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[Enthalpie~; capacités thermiques]\label{def:b1:first-law:enthalpy}
\index{enthalpie}\index{capacité thermique à pression constante}
L'\emph{enthalpie} $H = U + PV$ est une fonction d'état adaptée aux
\omterm{def:b1:first-law:transformation}{transformations} à pression constante. Les \emph{capacités thermiques à
volume et à pression constants} valent
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
en \unit{J/K}~; par \omterm{def:b1:kinetic-theory:scales}{mole} $C_{V,m}$, $C_{P,m}$~; par kilogramme, les
capacités thermiques massiques $c_V$, $c_P$.
\end{definition}

\begin{proposition}[Chaleur à volume constant et à pression constante]\label{prop:b1:first-law:qvqp}
Pour un système au repos qui n'échange de travail que par les forces de pression~:
\begin{itemize}
  \item \omterm{def:b1:first-law:transformation}{transformation} isochore~: $W = 0$ et $Q_V = \Delta U$~;
  \item \omterm{def:b1:first-law:transformation}{transformation} monobare entre deux états à la pression extérieure
        $P_0$~: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
Isochore~: $\dd V = 0$. Monobare~: $W = -P_0(V_2 - V_1)$, donc $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$ puisque $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[Gaz parfait~: $U$, $H$, relation de Mayer]\label{prop:b1:first-law:gas}
\index{relation de Mayer}\index{coefficient de Laplace}\index{lois de Joule}
Pour un \omterm{def:b1:kinetic-theory:model}{gaz parfait}, $U$ et $H$ ne dépendent que de $T$ (\omterm{prop:b1:kinetic-theory:Ugas}{lois de Joule})~:
$\dd U = C_V\,\dd T$, $\dd H = C_P\,\dd T$, et
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
$\gamma = 5/3$ pour les \omterm{prop:b1:kinetic-theory:Ugas}{gaz monoatomiques}, $7/5$ pour les diatomiques aux
températures ordinaires. Pour une \omterm{prop:b1:kinetic-theory:condensed}{phase condensée}, dans le modèle de la
\cref{prop:b1:kinetic-theory:condensed}, $C_P \approx C_V = C$ et
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
$H = U + nRT$~; on dérive par rapport à $T$. $C_{V,m} = \tfrac32 R$ ou
$\tfrac52 R$ d'après la \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{Transformations du gaz parfait}

\begin{proposition}[Les quatre transformations usuelles]\label{prop:b1:first-law:transformations}
\index{transformation isotherme}\index{transformation adiabatique}\index{loi de Laplace}
Pour $n$ \omterm{def:b1:kinetic-theory:scales}{moles} de \omterm{def:b1:kinetic-theory:model}{gaz parfait}~:
\begin{itemize}
  \item isochore~: $W = 0$, $Q = \Delta U = nC_{V,m}\Delta T$~;
  \item isobare (quasi statique)~: $W = -P\Delta V = -nR\Delta T$, $Q = \Delta H =
        nC_{P,m}\Delta T$~;
  \item isotherme réversible~: $\Delta U = 0$, $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$~;
  \item adiabatique réversible~: $Q = 0$, $W = \Delta U = nC_{V,m}(T_2 - T_1)$,
        et, le long du chemin (\emph{\omterm{prop:b1:first-law:transformations}{loi de Laplace}}),
        \[
        PV^\gamma = \text{const}, \qquad TV^{\gamma - 1} = \text{const}, \qquad
        T^\gamma P^{1 - \gamma} = \text{const} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Isochore et isobare~: les propositions précédentes. Isotherme~: $W =
-\int nRT\,\dd V/V$. Adiabatique~: $\dd U = \delta W$, c'est-à-dire $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$, donc $\dd T/T = -(\gamma - 1)\dd V/V$ (en utilisant
$R/C_{V,m} = \gamma - 1$)~; on intègre~: $TV^{\gamma - 1}$ est constant~; les autres
formes viennent de $PV = nRT$.
\end{proof}

\begin{example}[Pompe et moteur Diesel]\label{ex:b1:first-law:diesel}
De l'air ($\gamma = 1.4$) comprimé adiabatiquement de \qty{1}{bar},
\qty{300}{K} à \qty{7}{bar} (une pompe à vélo)~: $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- \qty{250}{\degreeCelsius} l'espace d'un instant, ce que le corps de la
pompe ressent. Avec le taux de compression $20$ d'un Diesel~: $T_2 = 300 \times 20^{0.4} = \qty{994}{K}$,
$P_2 = 20^{1.4} = \qty{66}{bar}$ --- au-dessus de la température
d'auto-inflammation du carburant~: le moteur n'a pas besoin de bougie (le
problème du week-end parcourt tout le cycle).
\end{example}

\begin{remark}[Transformations irréversibles]\label{rem:b1:first-law:irreversible}
\index{détente de Joule--Gay-Lussac}\index{transformation irréversible}
Lorsque la \omterm{def:b1:first-law:transformation}{transformation} n'est pas quasi statique, la \omterm{prop:b1:first-law:transformations}{loi de Laplace} et $W =
-\int P\,\dd V$ ne s'appliquent plus~; seuls valent $W = -\int P_{\mathrm{ext}}\,\dd V$ et le
premier principe. Deux cas classiques~: un gaz qui se détend dans le vide
(\omterm{rem:b1:first-law:irreversible}{détente de Joule--Gay-Lussac}) ne reçoit aucun travail et, entre parois adiabatiques, aucune \omterm{def:b1:first-law:heat}{chaleur} --- $\Delta U
= 0$, et un \omterm{def:b1:kinetic-theory:model}{gaz parfait} garde sa température~; un gaz brusquement comprimé
par une pression extérieure portée d'un coup à $P_2$ reçoit $W = -P_2\Delta V$, plus
que le travail réversible jusqu'à la même pression, et finit plus chaud
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{Calorimétrie}

\begin{method}[Bilan calorimétrique]\label{met:b1:first-law:calorimetry}
\index{calorimétrie}\index{chaleur latente}
Des corps mis en contact thermique dans une enceinte isolée à pression
(atmosphérique) constante n'échangent de la \omterm{def:b1:first-law:heat}{chaleur} qu'entre eux~: au total
$\Delta H = \sum Q_P = 0$. Pour chaque corps, $\Delta H = mc\,\Delta T$ (sans
changement d'état) ou $\pm mL$ pour un changement d'état à sa température de
transition, où $L$ est la \emph{\omterm{def:b1:first-law:heat}{chaleur} latente} (\omterm{def:b1:first-law:enthalpy}{enthalpie} de fusion ou de
vaporisation par unité de masse~: pour l'eau, $L_f = \qty{334}{kJ/kg}$ à \qty{0}{\degreeCelsius}, $L_v =
\qty{2260}{kJ/kg}$ à \qty{100}{\degreeCelsius}). Écrire le bilan, en tirer la
température finale, et vérifier que la phase supposée de chaque corps est
compatible avec elle.
\end{method}

\begin{example}[De la glace dans l'eau]\label{ex:b1:first-law:ice}
\qty{50}{g} de glace à \qty{0}{\degreeCelsius} dans \qty{200}{g} d'eau à
\qty{30}{\degreeCelsius}~: la fusion demande \qty{16.7}{kJ}~; l'eau peut
céder \qty{25.1}{kJ} en se refroidissant jusqu'à \qty{0}{\degreeCelsius},
donc toute la glace fond et le bilan $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ kJ donne
$T_f = \qty{8}{\degreeCelsius}$. Avec \qty{100}{g} de glace, il aurait fallu
\qty{33.4}{kJ}~: une partie seulement fond et $T_f = \qty{0}{\degreeCelsius}$
--- la vérification n'est pas facultative.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:first-law:1}
Une \omterm{def:b1:kinetic-theory:scales}{mole} d'air ($C_{P,m} = \qty{29.1}{J/(mol.K)}$) est chauffée de
\qty{300}{K} à \qty{400}{K} à pression constante. \omterm{def:b1:first-law:heat}{Chaleur} reçue, travail
reçu, variation d'\omterm{def:b1:kinetic-theory:U}{énergie interne}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:2}
Une \omterm{def:b1:kinetic-theory:scales}{mole} de gaz à \qty{300}{K} est comprimée de façon isotherme et
réversible de \qty{2.0}{L} à \qty{0.50}{L}. Travail reçu, \omterm{def:b1:first-law:heat}{chaleur} échangée,
$\Delta U$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:3}
De l'air à \qty{300}{K}, \qty{1}{bar} est comprimé de façon adiabatique et
réversible jusqu'au dixième de son volume. Température et pression finales.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:4}
Donner $C_{V,m}$, $C_{P,m}$ et $\gamma$ pour l'argon et pour l'azote. Pour
l'eau liquide ($c = \qty{4180}{J/(kg.K)}$), pourquoi ne distingue-t-on plus
$c_P$ de $c_V$~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:5}
\qty{200}{g} d'eau à \qty{80}{\degreeCelsius} sont mélangés à \qty{300}{g}
à \qty{20}{\degreeCelsius} dans un récipient isolé. Température finale~; qu'est-ce
qui change si le récipient lui-même a une capacité thermique de \qty{80}{J/K}
et part de \qty{20}{\degreeCelsius}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:6}
\qty{50}{g} de glace à \qty{-10}{\degreeCelsius} ($c_{\text{glace}} = \qty{2100}{J/(kg.K)}$)
tombent dans \qty{200}{g} d'eau à \qty{30}{\degreeCelsius}. État et
température finals. Même question avec \qty{150}{g} de glace.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:7}
Un gaz décrit un cycle rectangulaire dans le diagramme de Clapeyron~: $(V_1, P_1)
\to (V_2, P_1) \to (V_2, P_2) \to (V_1, P_2) \to (V_1, P_1)$ avec $V_2 > V_1$,
$P_2 > P_1$. Travail reçu sur le cycle, en fonction de l'aire~; signe, et
signification de ce signe.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:8}
Une pompe à vélo comprime de l'air de \qty{1}{bar}, \qty{300}{K} à
\qty{7}{bar}, assez vite pour que ce soit adiabatique. Température atteinte~;
pourquoi la pompe s'échauffe~; et pourquoi la pression du pneu baisse un peu
une fois le gonflage terminé.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:9}
Un \omterm{def:b1:kinetic-theory:model}{gaz parfait} se détend dans une enceinte vide et isolée (\omterm{rem:b1:first-law:irreversible}{détente de
Joule--Gay-Lussac}). Travail, \omterm{def:b1:first-law:heat}{chaleur}, $\Delta U$, $\Delta T$. La
\omterm{def:b1:first-law:transformation}{transformation} est-elle réversible~? Que ferait un gaz réel~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:10}
Ébullition de \qty{1.0}{kg} d'eau à \qty{100}{\degreeCelsius} et \qty{1}{atm}
(volume de vapeur \qty{1.67}{m^3}). \omterm{def:b1:first-law:heat}{Chaleur} fournie, travail échangé avec
l'atmosphère, $\Delta U$. Où passe l'essentiel de la \omterm{def:b1:first-law:heat}{chaleur} latente~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:11}
Une \omterm{def:b1:kinetic-theory:scales}{mole} de \omterm{prop:b1:kinetic-theory:Ugas}{gaz diatomique} à \qty{300}{K}, \qty{1}{bar}, dans un cylindre
isolé, est comprimée en portant brusquement la pression extérieure à
\qty{5}{bar}, puis on attend l'équilibre. Température et volume finals
(premier principe avec $W = -P_{\mathrm{ext}}\Delta V$)~; comparer à la
compression adiabatique réversible jusqu'à \qty{5}{bar}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:12}
Le son est une compression rapide, donc adiabatique~: sa \omterm{def:b1:wave-propagation:travelling}{célérité} vaut $c = \sqrt{\gamma P/\rho}$
(la valeur isotherme $\sqrt{P/\rho}$ fut l'erreur de Newton). Calculer les
deux pour l'air à \qty{20}{\degreeCelsius} et comparer aux \qty{343}{m/s}
mesurés. Montrer que $c = \sqrt{\gamma RT/M}$ et comparer à la \omterm{def:b1:kinetic-theory:kinetictemp}{vitesse
quadratique moyenne} du \cref{ch:b1:kinetic-theory}.
\end{exercise}

\section{Problème~: de la pompe à vélo au moteur Diesel}

\begin{problem}[{Problème du week-end --- une pompe à main qui brûle les
doigts, un cylindre qui enflamme lui-même son carburant et une bille
d'acier qui rebondit sur un coussin d'air~: le premier principe en trois compressions}]
\label{pb:b1:first-law:1}
Air~: \omterm{def:b1:kinetic-theory:model}{gaz parfait} diatomique, $\gamma = 1.40$, $C_{V,m} = \tfrac52 R$, $C_{P,m} =
\tfrac72 R$, $M = \qty{29}{g/mol}$~; $R = \qty{8.314}{J/(mol.K)}$. État initial
partout~: $P_1 = \qty{1.00}{bar}$, $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{Partie I --- La pompe à vélo.}
Le corps de la pompe a un volume $V_1 = \qty{212}{cm^3}$~; l'air qu'il
contient est comprimé de façon réversible et adiabatique jusqu'à la pression
du pneu, \qty{7.0}{bar}.
\begin{enumerate}
  \item Quantité d'air dans le corps de la pompe.
  \item Volume à la fin de la compression.
  \item Température à la fin de la compression.
  \item Travail reçu par l'air pendant un coup de pompe.
  \item Le pneu (\qty{2.0}{L}) doit passer de \qty{1}{bar} à \qty{7}{bar}
        à \qty{300}{K}~: combien de coups de pompe, et quel travail au
        total (on prendra le travail par coup constant)~?
  \item L'air chaud se refroidit ensuite dans le pneu jusqu'à \qty{300}{K}~:
        qu'advient-il de la pression du pneu, et que fait le cycliste~?
  \item Comparer le travail adiabatique par coup au travail d'une
        compression isotherme jusqu'aux mêmes \qty{7}{bar}. Pourquoi le
        travail isotherme est-il plus grand alors que le gaz finit plus froid~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La compression Diesel.}
Un cylindre de $V_1 = \qty{0.50}{L}$ comprime son air de façon réversible
et adiabatique d'un facteur $20$.
\begin{enumerate}[resume]
  \item Quantité d'air, et température $T_2$ après compression.
  \item Pression $P_2$.
  \item Le carburant s'enflamme spontanément au-dessus de \qty{500}{K}
        environ~: expliquer pourquoi un moteur Diesel n'a pas besoin de
        bougie, et pourquoi un moteur à essence (taux $\approx 10$, voir
        l'\cref{exo:b1:first-law:3}) ne doit pas s'auto-enflammer.
  \item Travail reçu par l'air pendant la compression.
  \item À \qty{3000}{tr/min}, un cylindre comprime $25$ fois par seconde~:
        puissance absorbée par la compression (elle est restituée plus loin
        dans le cycle).
  \item Les compressions réelles sont rapides (irréversibilité) et perdent
        de la \omterm{def:b1:first-law:heat}{chaleur} aux parois~: dans quel sens chacun de ces effets
        déplace-t-il $T_2$~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Combustion et détente.}
En haut de course, \qty{20}{mg} de carburant (pouvoir calorifique
\qty{43}{MJ/kg}) brûlent tandis que le piston commence à descendre, à
pression constante $P_2$~; le gaz se détend ensuite de façon réversible et
adiabatique jusqu'à $V_1$. On traitera le gaz comme de l'air d'un bout à l'autre.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{Chaleur} libérée par le carburant.
  \item Température $T_3$ et volume $V_3$ à la fin de la combustion isobare.
  \item Travail échangé pendant la phase isobare (attention au signe~!).
  \item Température $T_4$ à la fin de la détente adiabatique, et travail
        échangé pendant celle-ci.
  \item Travail net fourni par le gaz sur les quatre temps, et rapport à la
        \omterm{def:b1:first-law:heat}{chaleur} libérée (le rendement idéal).
  \item Comparer ce rendement à $1 - T_1/T_3$, la borne que le
        \cref{ch:b1:heat-engines} établira pour tout moteur fonctionnant
        entre ces températures extrêmes.
  \item \omterm{def:b1:first-law:heat}{Chaleur} évacuée avec les gaz d'échappement, par le premier principe
        sur le cycle~; vérifier le bilan.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Mesurer $\gamma$ avec une bille qui rebondit.}
Une bille d'acier de masse $m = \qty{16.5}{g}$ s'ajuste dans un tube de
verre vertical de section $S = \qty{2.0}{cm^2}$, fermé en bas par un ballon
de volume $V = \qty{10}{L}$ d'air à $P = \qty{1.0}{bar}$~; écartée de $x$ de
sa position d'équilibre, elle oscille.
\begin{enumerate}[resume]
  \item Pourquoi peut-on considérer les compressions et les détentes de
        l'air comme adiabatiques~? Écrire la relation entre un petit
        déplacement $x$ et la variation de pression $\dd P$ (\omterm{prop:b1:first-law:transformations}{loi de Laplace}, linéarisée).
  \item En déduire la force de rappel sur la bille et montrer que son
        mouvement est harmonique, avec $\omega_0^2 = \gamma PS^2/mV$.
  \item Calculer la période, et la valeur de $\gamma$ que l'on déduirait
        d'une période mesurée de \qty{1.10}{s}.
  \item Pourquoi le frottement entre la bille et le tube, ainsi que les
        fuites d'air, biaisent-ils le résultat, et dans quel sens~?
  \item Récapituler les trois compressions~: quelle grandeur le premier
        principe a conservée dans chacune, et ce que le coefficient
        adiabatique $\gamma$ y gouvernait.
\end{enumerate}
\end{problem}
