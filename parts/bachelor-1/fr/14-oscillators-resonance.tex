\chapter{Oscillateurs mécaniques~: amortissement et résonance}\label{ch:b1:oscillators-resonance}

Poussez une balançoire au mauvais moment et il ne se passe pas
grand-chose~; poussez-la une fois par période, doucement, et en une
douzaine d'allers-retours l'enfant crie de joie tout en haut de l'arc.
Une machine à laver tremble violemment à une vitesse précise de son
essorage, et redevient silencieuse au-dessus. Un verre à pied chante
quand un doigt mouillé en parcourt le bord à la bonne cadence, et se
brise quand une chanteuse tient la même note assez fort. Chacun de ces
phénomènes est une \emph{résonance}~: un système doté d'une \omterm{def:b1:oscillators-resonance:harmonic}{fréquence
propre}, excité près de cette fréquence, accumule de l'énergie cycle
après cycle jusqu'à ce que le frottement en évacue autant qu'il en
entre. Ce chapitre reprend l'oscillateur mécanique --- libre, amorti,
puis forcé --- et en tire les courbes de résonance dont les jumelles
électriques sont apparues au \cref{ch:b1:sinusoidal-impedance}.

\section{L'oscillateur harmonique}

\begin{definition}[Oscillateur harmonique]\label{def:b1:oscillators-resonance:harmonic}
\index{oscillateur harmonique}\index{fréquence propre}
Un système dont l'écart $x$ à un équilibre stable obéit à
\[
\ddot x + \omega_0^2\,x = 0
\]
est un \emph{oscillateur harmonique} de \emph{\omterm{def:b1:transient-regimes:secondorder}{pulsation propre}}
$\omega_0$ (de période $T_0 = 2\pi/\omega_0$). La solution $x = A\cos(\omega_0t
+ \varphi)$ a une \omterm{def:b1:wave-propagation:signal}{amplitude} et une phase fixées par les conditions
initiales, et une période indépendante de l'\omterm{def:b1:wave-propagation:signal}{amplitude}
(\emph{isochronisme}).
\end{definition}

\begin{proposition}[D'où viennent les oscillateurs harmoniques]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item Masse $m$ sur un ressort $k$~: $\omega_0^2 = k/m$ (\omterm{prop:b1:newton-dynamics:springstring}{loi de Hooke}).
  \item \omterm{ex:b1:newton-dynamics:pendulum}{Pendule simple} de longueur $\ell$, petits angles~: $\omega_0^2 = g/\ell$.
  \item Tout point au voisinage d'un équilibre stable d'un potentiel $E_p$~:
        $\omega_0^2 = E_p''(x_0)/m$ (\cref{prop:b1:work-and-energy:stability}).
  \item Circuit LC~: $\omega_0^2 = 1/LC$, avec $q \leftrightarrow x$,
        $L \leftrightarrow m$, $1/C \leftrightarrow k$.
\end{itemize}
L'énergie $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ est constante et passe d'une forme, cinétique, à
l'autre, potentielle, deux fois par période.
\end{proposition}

\begin{proof}
La deuxième loi de Newton avec $F = -kx$~; l'équation du pendule
$\ddot\theta + (g/\ell)\sin\theta = 0$ avec $\sin\theta \approx \theta$~;
la formule de Taylor au minimum de $E_p$~; la \omterm{thm:b1:dc-circuits:kirchhoff}{loi des mailles} du circuit
LC. Pour l'énergie~: on dérive et on utilise l'équation.
\end{proof}

\begin{example}[Un cylindre qui tangue]\label{ex:b1:oscillators-resonance:bob}
Un cylindre vertical de section $S$ flotte avec une hauteur $h$ immergée
($mg = \rho_wgSh$, poussée d'Archimède du
\cref{ch:b1:fluid-statics}). Enfoncé de $x$, il reçoit une force
supplémentaire vers le haut $\rho_wgSx$~: $m\ddot x = -\rho_wgSx$, donc
$\omega_0^2 = \rho_wgS/m = g/h$ --- comme un pendule de longueur $h$. Une
bouée immergée de \qty{1}{m} tangue avec une période de \qty{2}{s},
quelle que soit sa masse.
\end{example}

\section{L'oscillateur amorti}

\begin{proposition}[Oscillateur amorti]\label{prop:b1:oscillators-resonance:damped}
\index{oscillateur amorti}\index{facteur de qualité!mécanique}
Avec un frottement visqueux $-\alpha\dot x$, l'équation devient
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
c'est la forme canonique du second ordre du
\cref{ch:b1:transient-regimes}~: pour $Q > \tfrac12$, une décroissance
pseudo-périodique $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$ avec $\tau = 2Q/\omega_0$ et $\omega = \omega_0\sqrt{1 - 1/4Q^2}$~;
l'énergie décroît en $\eu^{-2t/\tau}$ et
\[
Q = 2\pi\,\frac{\text{énergie stockée}}{\text{énergie perdue par période}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
On divise $m\ddot x + \alpha\dot x + kx = 0$ par $m$. Pour l'énergie~: $E
\propto A^2\eu^{-2t/\tau}$ perd la fraction $2T/\tau = 2\pi/Q$ par
période lorsque celle-ci est petite.
\end{proof}

\begin{example}[Durées de sonnerie]\label{ex:b1:oscillators-resonance:ringing}
Un diapason à \qty{440}{Hz} s'entend encore dix secondes après avoir été
frappé~: son \omterm{def:b1:wave-propagation:signal}{amplitude} est tombée à environ $5\%$ en $3\tau = \qty{10}{s}$,
donc $\tau = \qty{3.3}{s}$ et $Q = \omega_0\tau/2 \approx 4600$. La
suspension d'une voiture ($Q \approx 0.7$) s'arrête en moins d'une
période~; une balançoire ($Q \approx 20$) perd un tiers de son énergie
par aller-retour et réclame une poussée à chaque fois.
\end{example}

\section{Oscillations forcées et résonance}

\begin{theorem}[Oscillateur excité sinusoïdalement]\label{thm:b1:oscillators-resonance:forced}
\index{oscillations forcées}\index{résonance!en amplitude}\index{résonance!en vitesse}
Excité par une force $F_0\cos\omega t$, l'\omterm{prop:b1:oscillators-resonance:damped}{oscillateur amorti}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
s'installe, après un transitoire de durée $\sim\tau$, dans le \omterm{prop:b1:sinusoidal-impedance:forced}{régime
forcé} $x = X\cos(\omega t + \varphi)$ avec, en posant $x_r = \omega/\omega_0$
et $X_0 = F_0/k$ la déformation statique,
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
L'\emph{\omterm{def:b1:wave-propagation:signal}{amplitude}} $X$ passe par un maximum, pour $Q > 1/\sqrt2$, en $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$ avec $X_{\max} \approx QX_0$~; l'\emph{\omterm{def:b1:wave-propagation:signal}{amplitude}
de vitesse} $V = \omega X$ est maximale exactement en $\omega_0$, où $V_{\max} =
F_0/\alpha$, et sa courbe de résonance
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
a pour \omterm{def:b1:filters-transfer-functions:bode}{bande passante} $\Delta\omega = \omega_0/Q$. En $\omega_0$, le
déplacement est en retard de $\pi/2$ sur la force et la vitesse est en
phase avec elle.
\end{theorem}

\begin{proof}
Méthode complexe (\cref{ch:b1:sinusoidal-impedance})~: $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$, donc $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$~; on divise numérateur et dénominateur
par $\omega_0^2 = k/m$. Le module et l'argument suivent~; le maximum de
$X$ a été localisé à la \cref{prop:b1:sinusoidal-impedance:overvoltage}
(même fonction). Vitesse~: $\underline V = j\omega\underline X$ donne $V = \omega X$~;
en écrivant $(1 - x_r^2)/x_r = 1/x_r - x_r$, le dénominateur devient
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$, c'est-à-dire la forme du RLC. En $x_r = 1$,
$\underline X = -jQX_0$~: un retard de $\pi/2$.
\end{proof}

\begin{remark}[Deux résonances, un seul circuit]\label{rem:b1:oscillators-resonance:analogy}
\index{analogie électromécanique}
La correspondance $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$,
$F \leftrightarrow e$ transporte l'oscillateur forcé sur le RLC série du
\cref{ch:b1:sinusoidal-impedance}~: la résonance en vitesse est la
résonance en intensité (toujours en $\omega_0$, de \omterm{def:b1:filters-transfer-functions:bode}{bande passante}
$\omega_0/Q$), la résonance en \omterm{def:b1:wave-propagation:signal}{amplitude} est celle de la charge,
c'est-à-dire de la tension du condensateur (un peu au-dessous de
$\omega_0$, absente pour $Q$ faible). La puissance absorbée, $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$, est maximale en $\omega_0$ et y vaut $F_0^2/2\alpha$,
et les points de demi-puissance délimitent la \omterm{def:b1:filters-transfer-functions:bode}{bande passante} en vitesse
--- exactement comme pour le RLC.
\end{remark}

\begin{example}[Vitesses de résonance]\label{ex:b1:oscillators-resonance:road}
La suspension d'une voiture a $f_0 \approx \qty{1.3}{Hz}$ (\cref{pb:b1:transient-regimes:1}).
Sur une route à tôle ondulée de période $\lambda = \qty{10}{m}$, la
fréquence d'excitation vaut $v/\lambda$~: résonance à
$v = \lambda f_0 = \qty{13}{m/s}$, environ \qty{47}{km/h} --- la vitesse
à laquelle une telle route secoue le plus la voiture, et la raison pour
laquelle on monte un amortisseur de $Q \approx 0.7$~: avec $Q = 5$, la
caisse rebondirait avec cinq fois l'\omterm{def:b1:wave-propagation:signal}{amplitude} de la route. Au-dessus de
la résonance ($x_r \gg 1$), l'\omterm{def:b1:wave-propagation:signal}{amplitude} tombe comme
$X_0/x_r^2$~: la caisse «~flotte~» au-dessus des bosses rapides ---
l'isolation, l'autre visage de la résonance.
\end{example}

\section{Excitation par la base~: isolation et sismomètre}

\begin{proposition}[Oscillateur secoué par son support]\label{prop:b1:oscillators-resonance:base}
\index{excitation par la base}\index{isolation vibratoire}\index{sismomètre}
Soit un système masse--ressort--amortisseur dont le support se déplace
de $x_g(t)$ (la route sous une voiture, le sol sous un sismomètre)~;
soit $u$ le déplacement de la masse par rapport au support et
$x = x_g + u$ son déplacement absolu (compté depuis l'équilibre). Alors
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
et, pour $x_g = X_g\cos\omega t$, avec $x_r = \omega/\omega_0$~:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
Pour $x_r \gg 1$~: $U \to X_g$ (la masse reste immobile dans l'espace ---
un \emph{sismomètre} lit le déplacement du sol) et $X \to 0$ (la masse
est \emph{isolée} des secousses)~; pour $x_r \ll 1$~: $U \approx
\ddot x_g/\omega_0^2$ (un \emph{accéléromètre}) et $X \approx X_g$ (la
masse suit le support).
\end{proposition}

\begin{proof}
Le ressort et l'amortisseur agissent sur le mouvement relatif~: $m\ddot x = -ku -
\alpha\dot u$, et $\ddot x = \ddot x_g + \ddot u$. En complexes,
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$, d'où
$U/X_g$~; puis $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$.
\end{proof}

\begin{example}[Pourquoi les machines reposent sur des supports souples]\label{ex:b1:oscillators-resonance:mounts}
Un moteur à \qty{50}{Hz} sur des plots de caoutchouc accordés à
$f_0 = \qty{10}{Hz}$ ($x_r = 5$, $Q \approx 2$)~: $X/X_g = \sqrt{(1 + 6.25)/(576 + 6.25)} = 0.11$
--- neuf dixièmes de la vibration ne passent pas dans le plancher. Les
plots doivent être \emph{plus souples} que l'intuition ne le suggère~:
les raidir remonte $f_0$ vers $50$ Hz et détruit l'isolation. La même
formule, lue à l'envers, décrit la voiture de
l'\cref{ex:b1:oscillators-resonance:road} sur sa tôle ondulée.
\end{example}

\section{Énergie~: comment une résonance s'installe}

\begin{proposition}[Bilan d'énergie en régime forcé]\label{prop:b1:oscillators-resonance:energy}
En \omterm{prop:b1:sinusoidal-impedance:forced}{régime forcé} établi, la \omterm{thm:b1:sinusoidal-impedance:power}{puissance moyenne} fournie par la force
d'excitation égale la \omterm{thm:b1:sinusoidal-impedance:power}{puissance moyenne} dissipée par le frottement,
$\tfrac12\alpha V^2$. À la résonance, l'énergie stockée
$\tfrac12 mV_{\max}^2$ vaut $Q/2\pi$ fois l'énergie fournie par
période~: l'oscillateur accumule le travail de l'excitation sur environ
$Q$ cycles avant que les pertes n'égalent l'apport, ce qui est aussi le
nombre de cycles ($\sim\tau/T$) que dure le transitoire.
\end{proposition}

\begin{proof}
Moyennons le théorème de l'énergie sur une période~: $\langle\dd E/\dd t\rangle = 0$
en régime établi, donc $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$. À la résonance, $V_{\max} = F_0/\alpha$, l'énergie
stockée vaut $\tfrac12 m(F_0/\alpha)^2$, l'énergie fournie par période $\tfrac12\alpha(F_0/\alpha)^2T$~;
le rapport vaut $m/(\alpha T) = Q/2\pi$.
\end{proof}

\begin{example}[La balançoire]\label{ex:b1:oscillators-resonance:swing}
Une balançoire de \qty{2.5}{m} ($T_0 = \qty{3.2}{s}$, $Q \approx 20$)
reçoit une poussée de \qty{50}{N} sur \qty{0.5}{m} à chaque période~:
\qty{25}{J} par cycle. L'\omterm{def:b1:wave-propagation:signal}{amplitude} croît jusqu'à ce que la perte par
période, $2\pi E/Q$, égale \qty{25}{J}~: $E = \qty{80}{J}$~; avec un
enfant de \qty{30}{kg}, $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ donne $\theta_{\max} \approx 27^\circ$.
Poussé sur tout autre rythme, le travail des poussées changerait de
signe et se moyennerait à rien.
\end{example}

\section{Exercices}

Une masse de \qty{0.20}{kg} sur un ressort de raideur \qty{80}{N/m}
oscille avec une \omterm{def:b1:wave-propagation:signal}{amplitude} de \qty{5.0}{cm}. \omterm{def:b1:wave-propagation:signal}{Pulsation}, fréquence,

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:2}

\end{exercise}

\section{Problème~: le sismographe}

\begin{problem}[{Problème du week-end --- une masse pend à un ressort
dans un boîtier boulonné au sol~; le sol tremble~: qu'écrit la plume,
pourquoi le même instrument mesure un déplacement à haute fréquence et
une accélération à basse fréquence, et pourquoi un sismomètre longue
période ne peut pas être simplement un long ressort}]
\label{pb:b1:oscillators-resonance:1}
Une masse $m = \qty{1.0}{kg}$ pend à un ressort de raideur $k$ à
l'intérieur d'un bâti rigide fixé au sol~; un amortisseur exerce
$-\alpha\dot u$, où $u$ est le déplacement de la masse par rapport au
bâti, compté depuis sa position d'équilibre. Le sol, et donc le bâti, se
déplace verticalement de $x_g(t)$. Le \omterm{def:b1:point-kinematics:frame}{référentiel} des étoiles lointaines
est galiléen.

\medskip
\noindent\textbf{Partie I --- L'équation.}
\begin{enumerate}
  \item Écrire la position absolue de la masse sous la forme $x = x_g + u +
        \text{const}$, faire le bilan des forces sur la masse (dans le
        \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen}), et montrer que
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item La mettre sous forme canonique~; identifier $\omega_0$ et $Q$.
  \item L'instrument est construit avec $f_0 = \qty{0.50}{Hz}$ et $Q = 0.70$~:
        calculer $k$ et $\alpha$.
  \item Une accélération constante du sol $a_g$ (le bâti qui bascule,
        par exemple) donne quel $u$ permanent~? Interpréter.
  \item Qu'enregistre la plume liée à la masse, qui écrit sur un tambour
        solidaire du bâti~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- \omterm{prop:b1:point-kinematics:harmonic}{Mouvement harmonique} du sol.}
Le sol se déplace selon $x_g = X_g\cos\omega t$~; on pose $x_r = \omega/\omega_0$.
\begin{enumerate}[resume]
  \item À l'aide des \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes}, montrer que $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$.
  \item En déduire $U/X_g$ en fonction de $x_r$ et de $Q$.
  \item Limite $x_r \gg 1$~: montrer que $U \to X_g$ et expliquer
        physiquement ce que fait la masse (que mesure alors
        l'instrument~?).
  \item Limite $x_r \ll 1$~: montrer que $U \approx x_r^2X_g = a_g/\omega_0^2$, où
        $a_g$ est l'\omterm{def:b1:wave-propagation:signal}{amplitude} d'accélération du sol~: qu'est devenu
        l'instrument~?
  \item Valeur en $x_r = 1$~; pourquoi $Q \approx 0.7$ est-il un choix
        raisonnable~?
  \item Esquisser $U/X_g$ en fonction de $x_r$ (échelles logarithmiques)
        pour $Q = 0.7$ et $Q = 5$.
  \item Phase de $u$ par rapport à $x_g$ à haute fréquence~; interpréter.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Dimensionner pour les séismes.}
Les séismes lointains produisent des ondes de surface de \qty{0.05}{} à
\qty{0.1}{Hz} et d'\omterm{def:b1:wave-propagation:signal}{amplitude} millimétrique~; les secousses locales
atteignent \qty{5}{} à \qty{10}{Hz}.
\begin{enumerate}[resume]
  \item Pour enregistrer les ondes à \qty{0.05}{Hz} comme des
        déplacements, on veut $f_0 \ll \qty{0.05}{Hz}$, disons
        \qty{0.02}{Hz}. Quel allongement statique le ressort
        prendrait-il sous le poids de la masse~? Commenter.
  \item Un \omterm{ex:b1:newton-dynamics:pendulum}{pendule simple} de même fréquence~: quelle longueur~?
  \item Les instruments longue période réels emploient un pendule
        presque horizontal, dit «~de porte de jardin~», ou une
        contre-réaction électronique de force. Expliquer en une phrase ce
        qu'apporte chacun.
  \item Pour l'instrument à \qty{0.50}{Hz}, calculer l'allongement
        statique~: est-ce réalisable~?
  \item Avec lui, une onde à \qty{0.10}{Hz} d'\omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm}
        donne quel $U$~? Dans quel régime est l'instrument~?
  \item Une secousse locale à \qty{5.0}{Hz} d'\omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm}~:
        quel $U$~? Quel régime~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- L'accéléromètre d'un téléphone.}
Une masse micro-usinée sur des ressorts de silicium a
$f_0 = \qty{1.0}{kHz}$ et $Q = 0.7$.
\begin{enumerate}[resume]
  \item Une secousse à \qty{10}{Hz} d'\omterm{def:b1:wave-propagation:signal}{amplitude} d'accélération $g$~:
        calculer $U$. Comment lit-on un tel déplacement~?
  \item Jusqu'à quelle fréquence la lecture reste-t-elle à $10\%$ près de
        $a_g/\omega_0^2$~?
  \item Quelle \omterm{def:b1:transient-regimes:firstorder}{constante de temps} $\tau = 2Q/\omega_0$ gouverne sa réponse
        à un choc, et pourquoi cela importe-t-il pour détecter une chute~?
  \item Le $\tau$ du sismomètre~: valeur et conséquence après un échelon
        brusque du sol.
  \item Pourquoi choisit-on un amortissement critique (ou presque) pour
        les deux instruments, plutôt qu'un $Q$ élevé~?
  \item Le téléphone est secoué exactement à \qty{1.0}{kHz}~: de quel
        facteur la lecture est-elle fausse~?
  \item Récapituler~: un seul dispositif, deux régimes, tranchés par
        quelle unique comparaison~?
\end{enumerate}
\end{problem}
