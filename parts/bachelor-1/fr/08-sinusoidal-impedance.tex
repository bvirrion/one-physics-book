\chapter{Régime sinusoïdal forcé et impédance}\label{ch:b1:sinusoidal-impedance}

Tournez le bouton d'un vieux poste de radio et, sur les cent stations
qui atteignent l'antenne à la fois, une seule passe~: une bobine et un
condensateur variable ont été accordés pour qu'une fréquence, et une
seule, les fasse résonner. Le secteur qui ronronne dans les murs est une
sinusoïde à \qty{50}{Hz}~; l'usine qui y fait tourner de gros moteurs
paie une pénalité si son courant retarde trop sur sa tension. Les
sinusoïdes sont la langue naturelle des \omterm{def:b1:dc-circuits:linear}{circuits linéaires} --- elles en
ressortent telles qu'elles y sont entrées, à un facteur et un décalage
près --- et ce chapitre leur donne leur algèbre~: \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes},
\omterm{def:b1:sinusoidal-impedance:impedance}{impédances}, résonance du \omterm{prop:b1:transient-regimes:rlc}{circuit RLC}, et puissance que les courants
sinusoïdaux délivrent réellement.

\section{Le régime sinusoïdal forcé}

\begin{proposition}[Régime sinusoïdal forcé]\label{prop:b1:sinusoidal-impedance:forced}
\index{régime sinusoïdal forcé}\index{régime forcé}
Dans un \omterm{def:b1:dc-circuits:linear}{circuit linéaire} alimenté par une source sinusoïdale de
\omterm{def:b1:wave-propagation:signal}{pulsation} $\omega$, tout courant et toute tension est, une fois les
transitoires éteints, une sinusoïde de même \omterm{def:b1:wave-propagation:signal}{pulsation} $\omega$, chacune
avec son \omterm{def:b1:wave-propagation:signal}{amplitude} et sa phase propres~: $x(t) = X\cos(\omega t + \varphi)$.
\end{proposition}

\begin{proof}
Les équations du circuit sont des équations différentielles linéaires à
coefficients constants~; leur solution générale est une solution
particulière plus les solutions de l'équation homogène, qui sont les
transitoires du \cref{ch:b1:transient-regimes} et s'amortissent (tout
circuit réel possède une résistance). Une solution particulière
sinusoïdale existe parce que dériver une sinusoïde de \omterm{def:b1:wave-propagation:signal}{pulsation} $\omega$
donne une sinusoïde de même \omterm{def:b1:wave-propagation:signal}{pulsation}~: les équations peuvent être
satisfaites terme à terme --- la méthode complexe ci-dessous la
construit explicitement.
\end{proof}

\begin{definition}[Amplitude complexe]\label{def:b1:sinusoidal-impedance:complex}
\index{amplitude complexe}\index{vecteur de Fresnel}
À la sinusoïde $x(t) = X\cos(\omega t + \varphi)$ on associe le \omterm{def:b1:wave-propagation:signal}{signal}
complexe $\underline{x}(t) = X\eu^{j(\omega t + \varphi)}$, de sorte que
$x = \Rea(\underline{x})$, et l'\emph{\omterm{def:b1:wave-propagation:signal}{amplitude} complexe}
\[
\underline{X} = X\eu^{j\varphi}, \qquad \underline{x}(t) = \underline{X}\,\eu^{j\omega t},
\]
qui porte l'\omterm{def:b1:wave-propagation:signal}{amplitude} $X = \abs{\underline{X}}$ et la phase
$\varphi = \arg\underline{X}$. Ici $j$ désigne l'unité imaginaire,
$j^2 = -1$, la lettre $i$ étant réservée aux courants. Représentée par
un vecteur de longueur $X$ faisant l'angle $\varphi$, $\underline{X}$
est un \emph{vecteur de Fresnel}.
\end{definition}

\begin{proposition}[Règles de la méthode complexe]\label{prop:b1:sinusoidal-impedance:rules}
\begin{enumerate}
  \item Les opérations linéaires commutent avec la prise de la partie
        réelle~: l'\omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexe} d'une somme est la somme des
        \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes}.
  \item Dériver par rapport au temps multiplie l'\omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexe} par
        $j\omega$~; intégrer la divise par $j\omega$.
  \item Les \omterm{thm:b1:dc-circuits:kirchhoff}{lois de Kirchhoff}, étant linéaires, valent pour les
        \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes}.
\end{enumerate}
\end{proposition}

\begin{proof}
$\Rea(\underline a + \underline b) = \Rea\underline a + \Rea\underline b$~;
$\dd(\underline X\eu^{j\omega t})/\dd t = j\omega\underline X\eu^{j\omega t}$,
et $\dd x/\dd t = \Rea(\dd\underline x/\dd t)$ puisque la dérivation est
linéaire sur les réels. Les sommes de courants en un nœud et de tensions
le long d'une maille sont linéaires.
\end{proof}

\begin{example}[Additionner deux sinusoïdes]\label{ex:b1:sinusoidal-impedance:add}
$3\cos\omega t + 4\sin\omega t$~: les \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes} valent $3$ et
$4\eu^{-j\pi/2} = -4j$ (puisque $\sin\omega t = \cos(\omega t - \pi/2)$)~;
leur somme $3 - 4j$ a pour module $5$ et pour argument $-0.927\ \unit{rad}$~:
$5\cos(\omega t - 0.927)$. Aucune identité trigonométrique n'a servi.
\end{example}

\section{Impédance}

\begin{definition}[Impédance, admittance]\label{def:b1:sinusoidal-impedance:impedance}
\index{impédance}\index{admittance}\index{réactance}
Pour un \omterm{def:b1:dc-circuits:linear}{dipôle linéaire} en \omterm{prop:b1:sinusoidal-impedance:forced}{régime sinusoïdal forcé} (\omterm{thm:b1:dc-circuits:kirchhoff}{convention
récepteur}), l'\emph{impédance} est le rapport des \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes}
\[
\underline Z = \frac{\underline U}{\underline I} = Z\eu^{j\varphi}
\quad (\text{en ohms}),
\]
dont le module $Z = U/I$ relie les \omterm{def:b1:wave-propagation:signal}{amplitudes} et dont l'argument
$\varphi = \varphi_u - \varphi_i$ est l'avance de phase de la tension
sur le courant. Son inverse $\underline Y = 1/\underline Z$ est
l'\emph{admittance} (en siemens). La partie réelle de $\underline Z$ est
sa résistance, la partie imaginaire sa \emph{réactance}.
\end{definition}

\begin{proposition}[Impédances de R, L, C]\label{prop:b1:sinusoidal-impedance:rlc}
\[
\underline Z_R = R, \qquad \underline Z_L = jL\omega, \qquad
\underline Z_C = \frac{1}{jC\omega} = -\frac{j}{C\omega} .
\]
Un \omterm{def:b1:dc-circuits:resistor}{conducteur ohmique} garde tension et courant en phase~; dans une
bobine la tension est en avance de $\pi/2$ sur le courant~; dans un
condensateur elle est en retard de $\pi/2$. À basse fréquence une bobine
tend vers un fil et un condensateur vers un circuit ouvert~; à haute
fréquence, l'inverse.
\end{proposition}

\begin{proof}
$u = Ri$~; $u = L\,\dd i/\dd t \to \underline U = jL\omega\,\underline I$~;
$i = C\,\dd u/\dd t \to \underline I = jC\omega\,\underline U$. Les
arguments de $1$, $j$ et $-j$ sont $0$, $\pi/2$, $-\pi/2$.
\end{proof}

\begin{proposition}[Associations~; diviseurs~; théorèmes]\label{prop:b1:sinusoidal-impedance:assoc}
Les \omterm{def:b1:sinusoidal-impedance:impedance}{impédances} en série s'ajoutent~; les \omterm{def:b1:sinusoidal-impedance:impedance}{admittances} en dérivation
s'ajoutent~; les \omterm{prop:b1:dc-circuits:dividers}{diviseurs de tension} et de courant, les théorèmes de
Thévenin et de Norton, la superposition et le théorème de Millman du
\cref{ch:b1:dc-circuits} valent tous avec \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes} et
\omterm{def:b1:sinusoidal-impedance:impedance}{impédances} à la place des tensions, courants et résistances réels.
\end{proposition}

\begin{proof}
Toutes les démonstrations du \cref{ch:b1:dc-circuits} n'utilisaient que
les \omterm{thm:b1:dc-circuits:kirchhoff}{lois de Kirchhoff} et la linéarité de $u = Ri$, valables l'une et
l'autre sous forme complexe.
\end{proof}

\begin{example}[Le diviseur RC]\label{ex:b1:sinusoidal-impedance:rcdivider}
Une résistance $R$ et un condensateur $C$ en série sous $\underline E$,
la sortie étant prise aux bornes de $C$~: $\underline U_C = \underline E\,\dfrac{1/jC\omega}
{R + 1/jC\omega} = \dfrac{\underline E}{1 + jRC\omega}$. \omterm{def:b1:wave-propagation:signal}{Amplitude}
$E/\sqrt{1 + (RC\omega)^2}$, phase $-\arctan(RC\omega)$~: transmission
totale à basse fréquence, $1/\sqrt2$ et $-45^\circ$ en
$\omega = 1/RC$, une décroissance en $1/\omega$ au-delà --- un filtre
passe-bas, objet du \cref{ch:b1:filters-transfer-functions}.
\end{example}

\section{Le circuit RLC série~: la résonance}

\begin{theorem}[Résonance en intensité]\label{thm:b1:sinusoidal-impedance:resonance}
\index{résonance}\index{résonance!en intensité}\index{bande passante}
Une résistance $R$, une bobine $L$ et un condensateur $C$ en série sous
$e(t) = E\cos\omega t$ sont parcourus par le courant d'\omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexe}
\[
\underline I = \frac{E}{R + j\left(L\omega - \dfrac{1}{C\omega}\right)},
\qquad
I = \frac{E}{\sqrt{R^2 + \left(L\omega - \dfrac{1}{C\omega}\right)^2}},
\qquad
\varphi_i = -\arctan\frac{L\omega - 1/C\omega}{R} .
\]
L'\omterm{def:b1:wave-propagation:signal}{amplitude} est maximale, $I_{\max} = E/R$, courant et tension en phase,
à la \emph{résonance} $\omega_0 = 1/\sqrt{LC}$. Avec
$Q = L\omega_0/R = 1/(RC\omega_0)$ et $x = \omega/\omega_0$,
\[
\frac{I}{I_{\max}} = \frac{1}{\sqrt{1 + Q^2\left(x - \dfrac1x\right)^2}} ,
\]
et la \emph{bande passante} --- l'intervalle de $\omega$ sur lequel
$I \geq I_{\max}/\sqrt2$ --- vaut
\[
\Delta\omega = \omega_2 - \omega_1 = \frac{\omega_0}{Q} = \frac{R}{L} .
\]
\end{theorem}

\begin{proof}
Les \omterm{def:b1:sinusoidal-impedance:impedance}{impédances} en série s'ajoutent~: $\underline Z = R + jL\omega + 1/jC\omega$,
et $\underline I = \underline E/\underline Z$. Le module est le plus
grand lorsque la \omterm{def:b1:sinusoidal-impedance:impedance}{réactance} $L\omega - 1/C\omega$ s'annule, c'est-à-dire
en $\omega_0$. En factorisant $L\omega - 1/C\omega = L\omega_0(x - 1/x)$
et en divisant par $R$~: $L\omega_0/R = Q$. Aux bords de la bande,
$Q(x - 1/x) = \pm 1$, c'est-à-dire
$x^2 \mp x/Q - 1 = 0$, dont les racines positives sont $x_{1,2} = \mp\frac{1}{2Q}
+ \sqrt{1 + \frac{1}{4Q^2}}$~; leur différence vaut $1/Q$.
\end{proof}

\begin{proposition}[Surtension]\label{prop:b1:sinusoidal-impedance:overvoltage}
\index{résonance!en tension}\index{surtension}
À la résonance, les tensions aux bornes de la bobine et du condensateur
ont la même \omterm{def:b1:wave-propagation:signal}{amplitude} $QE$ et des phases opposées~: pour $Q \gg 1$, le
condensateur porte une tension $Q$ fois plus grande que celle de la
source. L'\omterm{def:b1:wave-propagation:signal}{amplitude} $U_C(\omega)$ elle-même passe par un maximum, pour
$Q > 1/\sqrt2$, en $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$, légèrement sous $\omega_0$, avec
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$.
\end{proposition}

\begin{proof}
En $\omega_0$, $\underline U_C = \underline I/jC\omega_0 = -jQE$ et
$\underline U_L = jL\omega_0\underline I = jQE$ puisque $I = E/R$. En
général $U_C = I/C\omega = E/\sqrt{(1 - x^2)^2 + x^2/Q^2}$ (multiplier
numérateur et dénominateur par $1/C\omega_0$)~; le carré du dénominateur
vaut $(1 - x^2)^2 + x^2/Q^2$, trinôme en $x^2$ minimal en $x^2 =
1 - 1/2Q^2$ lorsque cette quantité est positive, de minimum $1/Q^2 - 1/4Q^4$.
\end{proof}

\begin{example}[Accorder une radio]\label{ex:b1:sinusoidal-impedance:radio}
Une bobine $L = \qty{200}{\micro H}$ et un condensateur variable~: $C =
\qty{127}{pF}$ accorde $f_0 = \qty{1.00}{MHz}$. Avec la résistance du
fil de la bobine $R = \qty{12.6}{\ohm}$, $L\omega_0 = \qty{1257}{\ohm}$
et $Q = 100$~: bande passante $\Delta f = f_0/Q = \qty{10}{kHz}$ --- la
largeur d'un canal AM --- et les \qty{10}{\micro V} induits par un
émetteur lointain deviennent $QE = \qty{1}{mV}$ aux bornes du
condensateur, gratuitement. Le problème du week-end construit ce récepteur.
\end{example}

\begin{remark}[Encore $Q$]\label{rem:b1:sinusoidal-impedance:Q}
Le $Q$ de ce chapitre est le $Q$ du \cref{ch:b1:transient-regimes}~: le
circuit qui sonne pendant environ $Q$ oscillations lorsqu'on le
percute est celui qui résonne sur une bande large de $\omega_0/Q$
lorsqu'on l'excite. Tous deux expriment un même rapport~: $2\pi$ fois
l'énergie stockée sur l'énergie dissipée par période
(\cref{exo:b1:sinusoidal-impedance:12}). Une résonance aiguë et une
sonnerie longue sont une seule et même propriété, tout comme la lenteur
de la réponse~: un filtre de bande passante $\Delta\omega$ met un temps
de l'ordre de $1/\Delta\omega$ à s'établir.
\end{remark}

\section{Puissance en régime sinusoïdal}

\begin{definition}[Valeur efficace]\label{def:b1:sinusoidal-impedance:rms}
\index{valeur efficace}\index{valeur quadratique moyenne}
La \emph{valeur efficace} (ou valeur quadratique moyenne) d'un \omterm{def:b1:wave-propagation:signal}{signal}
périodique $x(t)$ est $X_{\mathrm{eff}} = \sqrt{\langle x^2\rangle}$, la
racine carrée de la moyenne temporelle de $x^2$ sur une période. Pour
une sinusoïde d'\omterm{def:b1:wave-propagation:signal}{amplitude} $X$, $X_{\mathrm{eff}} = X/\sqrt2$. Les
\qty{230}{V} du secteur sont une valeur efficace~; l'\omterm{def:b1:wave-propagation:signal}{amplitude} vaut
\qty{325}{V}.
\end{definition}

\begin{theorem}[Puissance moyenne]\label{thm:b1:sinusoidal-impedance:power}
\index{puissance moyenne}\index{facteur de puissance}\index{puissance réactive}
Un dipôle sous $u = U\cos\omega t$ parcouru par $i = I\cos(\omega t - \varphi)$
(\omterm{thm:b1:dc-circuits:kirchhoff}{convention récepteur}) reçoit la puissance instantanée $p = ui$, de
valeur moyenne
\[
P = \tfrac12\,UI\cos\varphi = U_{\mathrm{eff}}I_{\mathrm{eff}}\cos\varphi
= \tfrac12\Rea\!\big(\underline U\,\underline I^{\,*}\big) = \tfrac12\Rea(\underline Z)\,I^2,
\]
où $\cos\varphi$ est le \emph{\omterm{thm:b1:sinusoidal-impedance:power}{facteur de puissance}}. Une résistance
reçoit $\tfrac12 RI^2 = RI_{\mathrm{eff}}^2$~; une bobine ou un
condensateur idéal ($\varphi = \pm\pi/2$) ne reçoit aucune \omterm{thm:b1:sinusoidal-impedance:power}{puissance
moyenne} --- il stocke de l'énergie pendant une demi-période et la rend
pendant l'autre.
\end{theorem}

\begin{proof}
$p = UI\cos\omega t\cos(\omega t - \varphi) = \tfrac12 UI[\cos\varphi +
\cos(2\omega t - \varphi)]$~; le second terme est de moyenne nulle. Avec
$\underline U = U$ et $\underline I = I\eu^{-j\varphi}$,
$\underline U\,\underline I^* = UI\eu^{j\varphi}$, dont la partie réelle
vaut $UI\cos\varphi$~; et $\underline U = \underline Z\,\underline I$ donne
$\underline U\,\underline I^* = \underline Z I^2$.
\end{proof}

\begin{example}[Relèvement du facteur de puissance]\label{ex:b1:sinusoidal-impedance:pf}
Le moteur d'un atelier, sous \qty{230}{V} efficaces, consomme
\qty{2.0}{kW} avec $\cos\varphi = 0.70$ (inductif)~: $I_{\mathrm{eff}} = 2000/(230 \times
0.70) = \qty{12.4}{A}$, contre \qty{8.7}{A} si le \omterm{thm:b1:sinusoidal-impedance:power}{facteur de puissance}
valait $1$ --- le courant en trop chauffe pour rien les câbles du
distributeur, d'où la pénalité. Un condensateur en dérivation fournit
localement les échanges d'énergie par quart de période de la bobine~: la
\emph{\omterm{thm:b1:sinusoidal-impedance:power}{puissance réactive}} $P\tan\varphi =
\qty{2.0}{kvar}$ demande $C = P\tan\varphi/(\omega U_{\mathrm{eff}}^2)
\approx \qty{120}{\micro F}$ à \qty{50}{Hz}, et le courant de ligne
retombe à \qty{8.7}{A}.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:1}
Donner l'\omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexe} de $u = 5\cos(\omega t + \pi/3)$ (V), sous
forme polaire et cartésienne. À l'aide des \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexes}, écrire $3\cos\omega t
+ 4\sin\omega t$ comme un unique cosinus.
\end{exercise}

\section{Problème~: le récepteur radio à résonance}

\begin{problem}[{Problème du week-end --- une bobine, un condensateur
variable et une antenne~: comment une résonance série choisit une
station sur cent, multiplie son signal ténu par cent, et pourquoi on ne
peut pas la rendre arbitrairement sélective}]
\label{pb:b1:sinusoidal-impedance:1}
L'antenne est modélisée par une fém sinusoïdale idéale
$e(t) = E\cos\omega t$ en série avec une bobine $L = \qty{200}{\micro H}$,
dont le fil a une résistance $R = \qty{12.6}{\ohm}$, et un condensateur
variable $C$. Le récepteur lit la tension aux bornes de $C$. Plus loin
(partie~IV), on tiendra compte de la résistance propre de l'antenne
$R_a = \qty{50}{\ohm}$.

\medskip
\noindent\textbf{Partie I --- Le courant résonnant.}
\begin{enumerate}
  \item Écrire l'\omterm{def:b1:sinusoidal-impedance:impedance}{impédance} complexe du circuit série.
  \item Donner l'\omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexe} du courant, puis son \omterm{def:b1:wave-propagation:signal}{amplitude}
        $I(\omega)$ et sa phase $\varphi_i(\omega)$ par rapport à $e$.
  \item Montrer que $I$ est maximal en $\omega_0 = 1/\sqrt{LC}$~; donner
        $I_{\max}$ et la phase à cet endroit.
  \item Quelle capacité accorde $f_0 = \qty{1.000}{MHz}$~? Calculer
        $L\omega_0$.
  \item Définir $Q = L\omega_0/R$ et le calculer.
  \item Montrer que $I/I_{\max} = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ avec
        $x = \omega/\omega_0$.
  \item Donner le signe de $\varphi_i$ au-dessous puis au-dessus de la
        résonance et l'interpréter (comportement capacitif ou inductif).
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La sélectivité.}
\begin{enumerate}[resume]
  \item Écrire la condition qui définit les deux fréquences auxquelles
        $I = I_{\max}/\sqrt2$.
  \item La résoudre et montrer que $\Delta\omega = \omega_2 - \omega_1 =
        \omega_0/Q$.
  \item Calculer la bande passante $\Delta f$ de ce récepteur.
  \item Une deuxième station émet à \qty{1.010}{MHz}, une troisième à
        \qty{1.020}{MHz}~: calculer le rapport d'\omterm{def:b1:wave-propagation:signal}{amplitudes} $I/I_{\max}$
        pour chacune, et le rapport de puissances correspondant en décibels.
  \item Les stations AM sont espacées de \qty{9}{} à \qty{10}{kHz}. Le
        récepteur est-il assez sélectif~? Qu'est-ce qui l'améliorerait~?
  \item Le condensateur se règle entre \qty{50}{} et \qty{500}{pF}~:
        quelle bande de fréquences le récepteur couvre-t-il~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La surtension.}
\begin{enumerate}[resume]
  \item Exprimer l'\omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexe} de la tension aux bornes du
        condensateur et montrer qu'à la résonance son \omterm{def:b1:wave-propagation:signal}{amplitude} vaut $QE$.
  \item La fém d'antenne vaut $E = \qty{10}{\micro V}$~: calculer
        $I_{\max}$ et la tension du condensateur à la résonance.
        Commenter.
  \item Montrer que la tension de la bobine a la même \omterm{def:b1:wave-propagation:signal}{amplitude} et est
        en opposition de phase avec celle du condensateur à la
        résonance~: que «~voit~» la source~?
  \item Calculer l'énergie stockée dans le circuit à la résonance et
        l'énergie dissipée par période~; vérifier que leur rapport,
        multiplié par $2\pi$, vaut $Q$.
  \item D'après le \cref{ch:b1:transient-regimes}, que vaut la \omterm{def:b1:transient-regimes:firstorder}{constante
        de temps} $\tau = 2Q/\omega_0$ de ce circuit~? Pourquoi un $Q$
        très élevé finit-il par déformer la parole et la musique (bande
        passante audio~: environ \qty{5}{kHz} en AM)~?
  \item Si la résistance du fil de la bobine doublait (fil plus fin),
        comment $Q$, la bande passante et la tension du condensateur
        changeraient-ils~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La résistance de l'antenne, et la puissance.}
\begin{enumerate}[resume]
  \item Avec $R_a = \qty{50}{\ohm}$ en série, recalculer $Q$, la bande
        passante et la tension du condensateur pour
        $E = \qty{10}{\micro V}$. Conclure.
  \item Calculer la \omterm{thm:b1:sinusoidal-impedance:power}{puissance moyenne} délivrée par la fém d'antenne à la
        résonance et la fraction dissipée dans $R_a$.
  \item Le théorème du transfert maximal de puissance réclamerait une
        résistance de charge égale à $R_a$~: pourquoi cet objectif
        entre-t-il en conflit avec la sélectivité~?
  \item Les récepteurs réels relient l'antenne à une prise intermédiaire
        de la bobine, ou par un petit condensateur de couplage.
        Expliquer qualitativement ce que cela apporte.
  \item À l'aide de $P = \tfrac12 UI\cos\varphi$, calculer le \omterm{thm:b1:sinusoidal-impedance:power}{facteur de
        puissance} aux deux fréquences de coupure et expliquer le nom de
        «~demi-puissance~».
  \item Récapituler la conception~: ce qu'apporte $Q$, ce qu'il coûte,
        et la valeur retenue pour un récepteur à \qty{1}{MHz} avec des
        canaux de \qty{10}{kHz}.
\end{enumerate}
\end{problem}
