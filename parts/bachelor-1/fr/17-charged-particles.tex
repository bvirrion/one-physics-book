\chapter{Particules chargées dans les champs E et B}\label{ch:b1:charged-particles}

Dans l'oscilloscope qui traçait les courbes du
\cref{ch:b1:transient-regimes}, un faisceau d'électrons est dirigé sur un
écran par la tension entre deux plaques~; dans l'aimant d'un
\omterm{prop:b1:charged-particles:spectrometer}{spectromètre de masse}, les ions de deux isotopes se séparent de quelques
millimètres parce que l'un est huit pour cent plus lourd que l'autre~;
dans le cyclotron d'un hôpital, des protons s'enroulent vers l'extérieur
sur deux cents tours avant d'en sortir au tiers de la vitesse de la
lumière pour brûler une tumeur. Ces trois situations relèvent de la même
dynamique~: une charge dans un champ uniforme, électrique ou magnétique,
et la loi de Newton. Ce chapitre la met en œuvre --- la parabole dans un
\omterm{def:b1:charged-particles:lorentz}{champ électrique}, le cercle et l'hélice dans un \omterm{def:b1:charged-particles:lorentz}{champ magnétique} --- et
en tire les instruments qui trient, accélèrent et mesurent les
particules chargées.

\section{La force de Lorentz}

\begin{definition}[Force de Lorentz]\label{def:b1:charged-particles:lorentz}
\index{force de Lorentz}\index{champ électrique}\index{champ magnétique}
Une particule de charge $q$ animée de la vitesse $\vect v$ dans un champ
électrique $\vect E$ et un champ magnétique $\vect B$ subit la force
\[
\vect F = q\big(\vect E + \vect v\wedge\vect B\big) .
\]
La part électrique est portée par $\vect E$ (ou par son opposé, si
$q < 0$) et ne dépend pas du mouvement~; la part magnétique est
perpendiculaire à la fois à $\vect v$ et à $\vect B$, proportionnelle à
la vitesse, et s'annule pour une particule au repos. Unités~: $\vect E$
en \unit{V/m} (ou \unit{N/C}), $\vect B$ en teslas (\unit{T})~:
$\qty{1}{T} = \qty{1}{N/(A.m)}$.
\end{definition}

\begin{proposition}[Travail de la force de Lorentz]\label{prop:b1:charged-particles:work}
La force magnétique ne travaille pas~: sa puissance $q(\vect v\wedge\vect B)\cdot\vect v$
est nulle, si bien qu'un \omterm{def:b1:charged-particles:lorentz}{champ magnétique} seul change la direction de la
vitesse mais jamais sa valeur ni l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique}. La force
électrique a pour puissance $q\vect E\cdot\vect v$, et entre deux points
son travail vaut $q(V_A - V_B) =
qU_{AB}$ (\cref{ch:b1:potential-capacitors})~: une particule de charge
$q$ qui franchit une tension $U$ gagne l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} $\abs{qU}$.
\end{proposition}

\begin{proof}
$\vect v\wedge\vect B \perp \vect v$. Le champ électrostatique dérive du
potentiel, $\vect E = -\overrightarrow{\operatorname{grad}}V$, donc $q\vect E$
est conservative avec $E_p = qV$.
\end{proof}

\begin{definition}[L'électronvolt]\label{def:b1:charged-particles:eV}
\index{électronvolt}
L'\emph{électronvolt} est l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} que gagne une charge
élémentaire en franchissant un volt~: $\qty{1}{eV} = \qty{1.60e-19}{J}$.
La chimie vit à quelques eV, les tubes à rayons X à des dizaines de keV,
la physique nucléaire au MeV, la physique des particules au GeV et au TeV.
\end{definition}

\begin{remark}[Pourquoi on néglige la pesanteur]\label{rem:b1:charged-particles:gravity}
Pour un électron dans un champ de \qty{1}{kV/m} --- une valeur modeste
de laboratoire --- la force électrique vaut $eE = \qty{1.6e-16}{N}$,
contre un poids $mg = \qty{9e-30}{N}$~: $10^{13}$ fois plus petit. Le
poids est écarté de tous les calculs de ce chapitre sans autre forme de
procès. (La limite relativiste est une autre affaire~: les formules qui
suivent valent pour $\tfrac12 mv^2 \ll mc^2$, c'est-à-dire en dessous
d'environ \qty{50}{keV} pour les électrons et \qty{100}{MeV} pour les
protons~; voir le volume de troisième année.)
\end{remark}

\section{Champ électrique uniforme}

\begin{proposition}[Accélération et déviation]\label{prop:b1:charged-particles:efield}
\index{tension accélératrice}
Dans un champ uniforme $\vect E$, l'accélération $\vect a = q\vect E/m$
est constante~: le mouvement est celui d'un projectile, avec
$q\vect E/m$ à la place de $\vect g$.
\begin{itemize}
  \item Lâchée sans vitesse et accélérée sous une tension $U$, la
        particule atteint $v = \sqrt{2\abs{qU}/m}$.
  \item Entrant à la vitesse $v_0$ selon $x$ dans une région de longueur
        $L$ où $\vect E = E\vect e_y$, elle suit la parabole $y = (qE/2mv_0^2)x^2$
        et en sort déviée de l'angle $\alpha$ tel que $\tan\alpha =
        qEL/mv_0^2$, comme si elle était venue en ligne droite du milieu
        des plaques.
\end{itemize}
\end{proposition}

\begin{proof}
Énergie~: $\tfrac12 mv^2 = \abs{qU}$. Déviation~: $x = v_0t$, $y =
\tfrac12(qE/m)t^2$~; à la sortie, $\dd y/\dd x = (qE/m)(x/v_0^2)$ évaluée
en $L$~; la tangente à la sortie, $y = (qEL/mv_0^2)(x - L/2)$, coupe
l'axe en $x = L/2$.
\end{proof}

\begin{example}[Le tube d'oscilloscope]\label{ex:b1:charged-particles:crt}
Des électrons accélérés sous $U_0 = \qty{2.0}{kV}$ ($v_0 = \qty{2.65e7}{m/s}$,
le dixième de $c$ --- tout juste non relativiste) traversent des plaques
de \qty{4.0}{cm} distantes de \qty{1.0}{cm} et portant \qty{100}{V}~:
$E = \qty{1.0e4}{V/m}$, $\tan\alpha = eEL/mv_0^2 = EL/2U_0 = 0.10$, et sur
un écran situé \qty{30}{cm} au-delà du centre des plaques la tache se
déplace de \qty{3.0}{cm} --- \qty{0.3}{mm} par volt, la sensibilité du
tube, fixée uniquement par la géométrie et la tension d'accélération.
\end{example}

\section{Champ magnétique uniforme}

\begin{theorem}[Mouvement dans un champ magnétique uniforme]\label{thm:b1:charged-particles:bfield}
\index{mouvement cyclotron}\index{rayon de Larmor}\index{pulsation cyclotron}\index{mouvement hélicoïdal}
Une particule de charge $q$ et de masse $m$, de vitesse $\vect v_0$
perpendiculaire à un champ $\vect B$ uniforme, décrit un cercle à vitesse
constante $v_0$, de rayon et de \omterm{def:b1:wave-propagation:signal}{pulsation}
\[
R = \frac{mv_0}{\abs q B}, \qquad \omega_c = \frac{\abs q B}{m},
\]
la \emph{\omterm{thm:b1:charged-particles:bfield}{pulsation cyclotron}} $\omega_c$ étant indépendante de la
vitesse~; le sens de rotation dépend du signe de $q$. Si $\vect v_0$ a
en outre une composante $v_\parallel$ selon $\vect B$, cette composante
reste inchangée et la \omterm{def:b1:point-kinematics:frame}{trajectoire} est une \emph{hélice} de rayon
$mv_\perp/\abs qB$ et de pas $2\pi mv_\parallel/\abs qB$ enroulée autour
des lignes de champ.
\end{theorem}

\begin{proof}
La force $q\vect v\wedge\vect B$ est perpendiculaire à $\vect v$ et à $\vect B$~;
la vitesse est constante (pas de travail) et la composante selon
$\vect B$ ne subit aucune force. Dans le plan perpendiculaire à
$\vect B$, la force a la norme constante $\abs qv_\perp B$, toujours
perpendiculaire à la vitesse~: c'est un \omterm{cor:b1:point-kinematics:circular}{mouvement circulaire} uniforme
(\cref{thm:b1:point-kinematics:frenet}) avec $mv_\perp^2/R =
\abs qv_\perp B$. Puis $\omega_c = v_\perp/R$~; la période $2\pi m/\abs qB$
multipliée par $v_\parallel$ donne le pas.
\end{proof}

\begin{example}[Ordres de grandeur]\label{ex:b1:charged-particles:orders}
Un électron de \qty{1}{keV} ($v = \qty{1.9e7}{m/s}$) dans \qty{1}{mT}~:
$R = \qty{11}{cm}$, $\omega_c = \qty{1.8e8}{rad/s}$, un tour en
\qty{36}{ns} --- la même période à toute énergie, et c'est ce qui rend
le cyclotron possible. Un proton du vent solaire (\qty{1}{keV},
\qty{4.4e5}{m/s}) dans les \qty{50}{\micro T} terrestres~: $R = \qty{92}{m}$~;
il ne peut pas traverser les lignes de champ, seulement s'enrouler le
long d'elles vers les pôles --- c'est l'aurore polaire.
\end{example}

\section{Instruments}

\begin{proposition}[Sélecteur de vitesse et spectromètre de masse]\label{prop:b1:charged-particles:spectrometer}
\index{sélecteur de vitesse}\index{spectromètre de masse}
Dans des champs uniformes croisés $\vect E \perp \vect B$, une particule
qui se déplace perpendiculairement aux deux passe sans être déviée si et
seulement si $v = E/B$, quelles que soient sa charge et sa masse
(\emph{\omterm{prop:b1:charged-particles:spectrometer}{sélecteur de vitesse}}). Dans un \omterm{def:b1:charged-particles:lorentz}{champ magnétique} seul, une
particule de vitesse connue décrit un demi-cercle de diamètre
$2mv/\abs qB$~: le mesurer donne $m/q$ (\emph{\omterm{prop:b1:charged-particles:spectrometer}{spectromètre de masse}}).
Des ions accélérés sous une tension $U$ puis courbés dans $B$ arrivent en
\[
R = \frac{1}{B}\sqrt{\frac{2mU}{\abs q}} :
\]
et deux isotopes de masses $m$ et $m + \Delta m$ se séparent de $\Delta R \approx
R\,\Delta m/2m$.
\end{proposition}

\begin{proof}
$q\vect E + q\vect v\wedge\vect B = \vect 0$ exige $vB = E$ avec la bonne
orientation. $R = mv/qB$ avec $v = \sqrt{2qU/m}$~; puis on différentie
$\ln R = \tfrac12\ln m + \text{const}$.
\end{proof}

\begin{example}[Les isotopes du carbone]\label{ex:b1:charged-particles:isotopes}
Des ions $^{12}$C et $^{13}$C une fois chargés, accélérés sous \qty{20}{kV}
dans \qty{0.50}{T}~: $R = \qty{14.1}{cm}$ et \qty{14.7}{cm}, soit
\qty{5.8}{mm} d'écart --- deux taches distinctes sur le détecteur, et le
rapport de leurs intensités donne l'abondance isotopique, matière
première de la datation au carbone et de toute analyse isotopique.
\end{example}

\begin{proposition}[Cyclotron]\label{prop:b1:charged-particles:cyclotron}
\index{cyclotron}
Deux demi-cylindres creux (les «~dees~») dans un champ $\vect B$
uniforme, avec entre eux une tension alternative $U$ à la \omterm{thm:b1:charged-particles:bfield}{pulsation
cyclotron}~: la particule décrit des cercles à l'intérieur de chaque dee
(où le \omterm{def:b1:charged-particles:lorentz}{champ électrique} est nul), est accélérée de $\abs qU$ à chacune
des deux traversées de l'intervalle par tour, et s'enroule vers
l'extérieur avec $R \propto v$~; au rayon d'extraction $R_{\max}$, son
\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} vaut
\[
E_k = \frac{q^2B^2R_{\max}^2}{2m},
\]
indépendante de la tension, qui ne fixe que le nombre de tours.
\end{proposition}

\begin{proof}
Comme $\omega_c$ ne dépend pas de la vitesse, une tension de fréquence
fixe reste en phase avec la particule tour après tour~; et $v = \abs qBR/m$
au rayon $R$.
\end{proof}

\begin{example}[Un cyclotron d'hôpital]\label{ex:b1:charged-particles:medical}
Protons, $B = \qty{1.5}{T}$, $R_{\max} = \qty{0.50}{m}$~: $E_k =
(1.6\times10^{-19} \times 1.5 \times 0.5)^2/(2 \times 1.67\times10^{-27}) =
\qty{4.3e-12}{J} = \qty{27}{MeV}$, à $f_c = \omega_c/2\pi = \qty{23}{MHz}$.
Avec \qty{50}{kV} sur les dees, \qty{100}{keV} par tour~: $270$ tours en
\qty{12}{\micro s}. Au-delà de quelques dizaines de MeV, l'augmentation
relativiste de la masse (volume de troisième année) désynchronise les
protons~; les machines de thérapie à \qty{230}{MeV} modulent la fréquence
ou le champ pour compenser --- c'est le sujet du problème du week-end.
\end{example}

\begin{proposition}[Effet Hall]\label{prop:b1:charged-particles:hall}
\index{effet Hall}
Un ruban d'épaisseur $d$ parcouru par un courant $I$ de porteurs de
charge $q$ et de densité $n$, plongé dans un champ $B$ perpendiculaire au
ruban, développe en travers de sa largeur la \emph{tension de Hall}
\[
U_H = \frac{IB}{nqd} :
\]
proportionnelle à $B$ (un magnétomètre) et inversement proportionnelle à
$n$ (une sonde de la densité et du signe des porteurs).
\end{proposition}

\begin{proof}
La force magnétique $qv_dB$ sur les porteurs (de vitesse de dérive
$v_d$) les pousse vers un bord jusqu'à ce que le \omterm{def:b1:charged-particles:lorentz}{champ électrique}
transversal $E_H$ l'équilibre~: $qE_H = qv_dB$, et $U_H = E_Hw$ en
travers de la largeur $w$~; avec $I = nqv_dwd$, $E_Hw = v_dBw = IB/(nqd)$.
\end{proof}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:1}
Convertir \qty{1}{eV}, \qty{13.6}{eV} et \qty{1}{MeV} en joules. Vitesse
d'un électron puis d'un proton accélérés du repos sous \qty{1.0}{kV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:2}
Un électron de \qty{1.0}{keV} entre perpendiculairement dans un champ de
\qty{1.0}{mT}. Rayon, \omterm{thm:b1:charged-particles:bfield}{pulsation cyclotron}, période.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:3}
Un proton à \qty{1.0e5}{m/s} dans le champ terrestre (\qty{50}{\micro T})~:
rayon de son cercle. Que se passe-t-il si sa vitesse fait $30^\circ$ avec
le champ~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:4}
Un \omterm{prop:b1:charged-particles:spectrometer}{sélecteur de vitesse} a $E = \qty{10}{kV/m}$ et $B = \qty{20}{mT}$.
Quelle vitesse passe~? Qu'advient-il des ions plus rapides, et de la même
vitesse avec la charge opposée~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:5}
Oscilloscope~: électrons sous \qty{2.0}{kV}, plaques de \qty{4.0}{cm} de
long distantes de \qty{1.0}{cm}, \qty{100}{V} entre elles, écran à
\qty{30}{cm} au-delà du centre des plaques. Déviation sur l'écran et
sensibilité en mm/V. Pourquoi une tension d'accélération plus grande
réduit-elle la sensibilité~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:6}
Des ions $^{12}$C et $^{13}$C une fois chargés, accélérés sous
\qty{20}{kV}, entrent dans un champ de \qty{0.50}{T} ($\qty{1}{u} = \qty{1.66e-27}{kg}$).
Rayons et écart des deux taches après un demi-tour.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:7}
Un cyclotron à protons a $B = \qty{1.5}{T}$ et $R_{\max} = \qty{0.50}{m}$.
Fréquence de la tension des dees, \omterm{thm:b1:work-and-energy:ke}{énergie cinétique} maximale, nombre de
tours pour \qty{50}{kV} sur les dees, temps passé à l'intérieur. De
quelle fraction la masse du proton a-t-elle augmenté à la sortie
($\gamma - 1 \approx E_k/mc^2$, $mc^2 = \qty{938}{MeV}$)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:8}
Un électron à \qty{1.0e6}{m/s} entre dans un champ de \qty{2.0}{mT} à
$30^\circ$ des lignes de champ. Rayon et pas de son hélice~; nombre de
tours par mètre le long du champ.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:9}
Un ruban de cuivre de \qty{1.0}{mm} d'épaisseur porte \qty{5.0}{A} dans
\qty{1.0}{T} ($n = \qty{8.5e28}{m^{-3}}$). Tension de Hall. Pourquoi les
capteurs à \omterm{prop:b1:charged-particles:hall}{effet Hall} sont-ils faits de semi-conducteurs
($n \sim \qty{1e22}{m^{-3}}$)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:10}
Démontrer, à partir de la loi de Newton, qu'une charge placée dans un

garde une vitesse de norme constante. Un électron de \qty{10}{keV} entre
dans une région de champ intense et non uniforme~: qu'est-ce qui peut
changer, et qu'est-ce qui ne le peut pas~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:11}
Une charge $q$ part du repos à l'origine dans $\vect E = E\vect e_y$ et

(indication~: chercher une solution de la forme
$\vect v = \vect u + \vect w$ avec $\vect u$ constant), et décrire la
\omterm{def:b1:point-kinematics:frame}{trajectoire} (une cycloïde). Application numérique pour un électron avec
$E = \qty{1.0e4}{V/m}$ et $B = \qty{0.10}{T}$.
\end{exercise}

\section{Problème~: des protons contre une tumeur}

\begin{problem}[{Problème du week-end --- le cyclotron derrière le mur
d'un hôpital~: comment des protons sont lancés au tiers de la vitesse de
la lumière, pourquoi la machine simple cesse de fonctionner avant
l'énergie dont le thérapeute a besoin, et combien de protons il faut
pour traiter une tumeur}]
\label{pb:b1:charged-particles:1}
Protons~: $m = \qty{1.67e-27}{kg}$, $q = e = \qty{1.60e-19}{C}$, $mc^2 =
\qty{938}{MeV}$~; $c = \qty{3.00e8}{m/s}$. Cyclotron~: $B = \qty{1.5}{T}$,
rayon d'extraction $R_{\max} = \qty{0.50}{m}$, \omterm{def:b1:wave-propagation:signal}{amplitude} de la tension
des dees $U = \qty{50}{kV}$.

\medskip
\noindent\textbf{Partie I --- Un tour.}
\begin{enumerate}
  \item Écrire la \omterm{def:b1:charged-particles:lorentz}{force de Lorentz} sur un proton animé de la vitesse
        $\vect v$ perpendiculaire à $\vect B$, et montrer que sa vitesse
        ne peut pas changer à l'intérieur d'un dee.
  \item Établir le rayon de son cercle et sa période~; montrer que la
        période ne dépend pas de la vitesse.
  \item Calculer la fréquence cyclotron $f_c$ pour ces protons.
  \item Énergie gagnée à chaque traversée de l'intervalle~; puis par tour.
  \item Pourquoi la tension des dees doit-elle alterner exactement à
        $f_c$~? Que se passe-t-il si elle en est un peu écartée~?
  \item Les protons sont injectés au centre avec une vitesse
        négligeable~: quel rayon après le premier tour~? Après $n$ tours~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Jusqu'au bord.}
\begin{enumerate}[resume]
  \item \omterm{thm:b1:work-and-energy:ke}{Énergie cinétique} à l'extraction (en J et en MeV), et vitesse en
        fraction de $c$.
  \item Nombre de tours nécessaires et temps passé dans la machine.
  \item Longueur totale du trajet en spirale (somme des circonférences
        --- on approchera la somme par une intégrale).
  \item Le résultat de la question~7 dépend-il de $U$~? Sur quoi $U$
        agit-il~?
  \item Le vide intérieur doit être poussé~: estimer la distance qu'un
        proton parcourt entre deux chocs avec les molécules du gaz
        résiduel si le libre parcours moyen doit dépasser la longueur de
        la spirale~; ce libre parcours varie comme $1/p$ et vaut environ
        \qty{70}{nm} à la pression atmosphérique. Quelle pression faut-il~?
  \item Calculer $\gamma - 1 \approx E_k/mc^2$ à l'extraction et la
        variation relative de période qui en résulte. Sur le nombre de
        tours trouvé plus haut, de quelle fraction de période le proton
        a-t-il glissé par rapport à la tension~? Commenter.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- 230 MeV.}
La protonthérapie demande environ \qty{230}{MeV} pour atteindre des
tumeurs à \qty{30}{cm} de profondeur.
\begin{enumerate}[resume]
  \item Avec la formule non relativiste, quel produit $BR$ donnerait
        \qty{230}{MeV}~? Avec $B = \qty{2.0}{T}$, quel rayon~?
  \item À cette énergie, $\gamma - 1 = 0.245$~: qu'advient-il de la
        période cyclotron à mesure que le proton gagne de l'énergie, et
        pourquoi un cyclotron à fréquence fixe échoue-t-il~?
  \item Deux remèdes existent~: faire varier la fréquence de la tension
        pendant l'accélération (synchrocyclotron), ou faire croître $B$
        avec le rayon de sorte que $B/\gamma$ reste constant (cyclotron
        isochrone). Expliquer chacun en une phrase et nommer un
        inconvénient du premier.
  \item La quantité de mouvement relativiste vaut $p = \gamma mv$ et le
        rayon $R =
        p/qB$ reste valable~: avec $\gamma = 1.245$ et $v = 0.60c$,
        calculer $R$ pour $B = \qty{2.0}{T}$ et comparer à la question~13.
  \item Le faisceau extrait passe entre deux plaques de \qty{1.0}{m} de
        long sous $E = \qty{1.0e5}{V/m}$ pour être dévié d'un petit
        angle~: estimer cet angle (non relativistement, avec le $p$ de la
        question précédente à la place de $mv$). Le pilotage électrique
        est-il praticable ici~?
  \item Le même pilotage avec un aimant de \qty{1.0}{T} sur \qty{1.0}{m}~:
        quel angle~? Conclure sur les raisons qui font des aimants la
        règle dans les lignes de faisceau.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La dose.}
\begin{enumerate}[resume]
  \item Un proton de \qty{230}{MeV} dépose l'essentiel de son énergie
        près de la fin de son parcours (le pic de Bragg). Si tout est
        absorbé dans une tumeur de \qty{0.50}{kg}, combien de protons
        délivrent une dose de \qty{2.0}{Gy} ($\qty{1}{Gy} = \qty{1}{J/kg}$)~?
  \item Le courant du faisceau vaut \qty{1.0}{nA}~: protons par seconde,
        et durée de la séance.
  \item Puissance transportée par le faisceau~; la comparer à celle d'une
        ampoule.
  \item Les protons sont arrêtés dans le patient~: quelle énergie chaque
        proton dépose-t-il grosso modo dans le dernier centimètre, si un
        quart de l'énergie s'y perd~? En \omterm{def:b1:charged-particles:eV}{électronvolts} par micromètre~?
  \item Chaque ionisation dans les tissus coûte environ \qty{30}{eV}~:
        combien d'ionisations un proton provoque-t-il sur tout son
        parcours~?
  \item Pourquoi un faisceau de protons vaut-il mieux que des rayons X
        pour une tumeur profonde, en une phrase sur l'endroit où va
        l'énergie~?
  \item Récapituler l'enchaînement, de la \omterm{def:b1:charged-particles:lorentz}{force de Lorentz} à la dose~:
        quelles équations ont fixé l'énergie, le rayon, le nombre de tours
        et la durée du traitement.
\end{enumerate}
\end{problem}
