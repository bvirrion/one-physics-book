\chapter{Introduction à la physique quantique}\label{ch:b1:quantum-introduction}

Éclairez un métal propre en ultraviolet et des électrons s'échappent
aussitôt, avec une énergie qui dépend de la couleur de la lumière et
nullement de son éclat~; éclairez-le en rouge et il ne sort rien, si
puissante que soit la lampe. Envoyez des électrons un à un à travers
deux fentes~: chacun arrive en un point, un seul --- et les points, par
milliers, dessinent les franges d'une onde. Ni l'un ni l'autre de ces
faits ne tient dans la physique des vingt-neuf premiers chapitres. Ce
dernier chapitre introduit les idées qui les accueillent~: la lumière
s'échange par quanta, les photons~; la matière a une \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde}~;
ce qui se propage est une \omterm{def:b1:wave-propagation:signal}{amplitude} dont le carré est une probabilité~;
position et quantité de mouvement ne peuvent être nettes ensemble~; et
une particule confinée n'a que certaines énergies --- d'où la taille des
atomes, les raies des spectres, et l'éclat d'un grain de
semi-conducteur de quelques nanomètres, dont le diamètre choisit la couleur.

\section{Le photon}

\begin{theorem}[Relations de Planck--Einstein]\label{thm:b1:quantum-introduction:photon}
\index{photon}\index{constante de Planck}
La lumière de fréquence $\nu$ et de \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} $\lambda$ n'échange
d'énergie avec la matière que par quanta entiers, les \emph{photons},
d'énergie et de quantité de mouvement
\[
E = h\nu = \frac{hc}{\lambda} , \qquad p = \frac{h}{\lambda} = \frac{E}{c} ,
\]
avec la \emph{\omterm{thm:b1:quantum-introduction:photon}{constante de Planck}} $h = \qty{6.626e-34}{J.s}$ ($\hbar = h/2\pi =
\qty{1.055e-34}{J.s}$). Utile~: $hc = \qty{1240}{eV.nm}$ --- un photon de
\qty{620}{nm} transporte \qty{2}{eV}.
\end{theorem}
\admitted

\begin{proposition}[L'effet photoélectrique]\label{prop:b1:quantum-introduction:photoelectric}
\index{effet photoélectrique}\index{travail d'extraction}\index{potentiel d'arrêt}
La lumière qui tombe sur un métal de \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{travail d'extraction}} $W$
(l'énergie qui retient ses électrons les moins liés, quelques eV) en
arrache des électrons d'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} maximale
\[
E_{k,\max} = h\nu - W ,
\]
donc seulement au-dessus du \emph{seuil} $\nu_0 = W/h$, instantanément,
avec une énergie indépendante de l'intensité~; l'intensité, elle, fixe le
\emph{nombre} d'électrons par seconde. Le \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{potentiel d'arrêt}}
$V_s$ qui annule tout juste le courant mesure $E_{k,\max} = eV_s$, et
le tracé de $V_s$ en fonction de $\nu$ est une droite de pente $h/e$ ---
c'est ainsi que Millikan a mesuré $h$.
\end{proposition}

\begin{proof}
Chaque photon est absorbé par un électron, qui en retient au mieux
$h\nu - W$~; une onde livrerait l'énergie continûment à tous les
électrons, l'énergie croîtrait avec l'intensité, et une lumière faible
mettrait des heures à accumuler $W$ sur un seul atome
(\cref{pb:b1:quantum-introduction:1}) --- rien de tout cela ne s'observe.
L'interprétation d'Einstein (1905) est admise comme loi.
\end{proof}

\begin{example}[Ordres de grandeur]\label{ex:b1:quantum-introduction:orders}
Un photon visible~: \qty{2}{eV}~; un photon X de \qty{0.1}{nm}~: \qty{12}{keV}~;
un photon radio à \qty{100}{MHz}~: \qty{4e-7}{eV}. Un laser rouge de \qty{1}{mW}
émet $10^{-3}/(3.1 \times 10^{-19}) = 3 \times 10^{15}$ photons par seconde~;
une antenne qui capte \qty{1}{pW} à \qty{100}{MHz} en reçoit $10^{13}$
par seconde --- si nombreux, si petits, que la description ondulatoire est
exacte à toute fin utile~; l'\omterm{def:b1:optical-instruments:eye}{œil}, lui, enregistre un éclair d'une
centaine de photons. La nature quantique de la lumière se manifeste
quand les photons sont rares ou énergétiques~: cellules photoélectriques,
détecteurs de rayons X, grain d'une photographie sous-exposée.
\end{example}

\begin{remark}[Diffusion Compton]\label{rem:b1:quantum-introduction:compton}
\index{diffusion Compton}
Un photon qui rebondit sur un électron libre lui cède de la quantité de
mouvement comme une boule de billard~: sa \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} augmente de
$\Delta\lambda = \frac{h}{m_ec}(1 -
\cos\theta)$, avec $h/m_ec = \qty{2.43}{pm}$ (admis --- c'est un choc
relativiste, volume de deuxième année). Invisible pour la lumière ($\Delta\lambda/\lambda \sim 10^{-5}$),
l'effet atteint $20\%$ pour les rayons X~; c'est l'expérience qui a
convaincu les sceptiques (1923) que le photon porte la quantité de
mouvement $h/\lambda$.
\end{remark}

\section{Les ondes de matière}

\begin{theorem}[de Broglie]\label{thm:b1:quantum-introduction:debroglie}
\index{longueur d'onde de de Broglie}\index{onde de matière}
Une particule de quantité de mouvement $p$ se propage comme une onde de \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde}
\[
\lambda = \frac{h}{p} ,
\]
et donne diffraction et interférences en conséquence~: des électrons sur un
cristal (Davisson et Germer, 1927), à travers deux fentes (Jönsson, 1961),
des neutrons, des atomes, des molécules de soixante atomes de carbone. Pour un
électron accéléré sous $U$, $\lambda = h/\sqrt{2m_eeU} = \qty{1.23}{nm}/\sqrt{U/\unit{V}}$.
\end{theorem}
\admitted

\begin{example}[Pourquoi on ne l'a jamais vue]\label{ex:b1:quantum-introduction:wavelengths}
Un électron de \qty{100}{eV}~: $\lambda = \qty{0.12}{nm}$, la distance entre
atomes d'un cristal --- d'où la diffraction, et le \omterm{def:b1:optical-instruments:microscope}{microscope}
électronique. Un neutron thermique (\qty{0.025}{eV})~: \qty{0.18}{nm}. Une
bille de \qty{1}{g} à \qty{1}{m/s}~: $\lambda = \qty{7e-31}{m}$, $10^{20}$ fois
plus petite qu'un noyau~: aucune fente, aucun cristal ne la diffractera
jamais. La nature ondulatoire de la matière est universelle~; c'est sa
\omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} qui ne la rend visible que pour le léger et le lent.
\end{example}

\begin{proposition}[Les particules construisent les franges une à une]\label{prop:b1:quantum-introduction:buildup}
\index{dualité onde--corpuscule}
Des électrons envoyés un à un à travers deux fentes (Tonomura, 1989)
font chacun un point unique sur l'écran, en un lieu imprévisible~; les
points s'accumulent en la figure d'interférences à deux fentes d'une onde
de \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} $h/p$. Boucher une fente efface les franges --- la
figure n'est pas due à des électrons qui interféreraient entre eux, mais à
l'onde de chaque électron, qui passe par les deux fentes. Détecter par
quelle fente une particule est passée efface aussi les franges.
\end{proposition}
\admitted

\section{Amplitudes, probabilités et principe d'incertitude}

\begin{definition}[Fonction d'onde]\label{def:b1:quantum-introduction:wavefunction}
\index{fonction d'onde}\index{densité de probabilité}
L'état d'une particule est décrit par une \emph{fonction d'onde}
$\psi(x, t)$ (complexe), telle que $|\psi(x, t)|^2\,\dd x$ soit la
probabilité de trouver la particule entre $x$ et $x + \dd x$ lorsqu'on
regarde~: $|\psi|^2$ est une \emph{densité de probabilité}, et $\int|\psi|^2\dd x =
1$. Les \omterm{def:b1:wave-propagation:signal}{amplitudes} s'ajoutent (superposition)~: si deux chemins mènent au
même point, $\psi = \psi_1 + \psi_2$ et $|\psi|^2 = |\psi_1|^2 + |\psi_2|^2 +
2\,\mathrm{Re}(\psi_1^\ast\psi_2)$ --- le dernier terme est l'interférence.
Une mesure donne l'un des résultats possibles, au hasard avec les
probabilités que prescrivent les \omterm{def:b1:wave-propagation:signal}{amplitudes}, et laisse la particule dans
l'état trouvé.
\end{definition}

\begin{remark}[Ce qui ondule]\label{rem:b1:quantum-introduction:whatwaves}
$\psi$ n'est pas un champ physique comme $\vect E$~: on ne le mesure pas,
on ne mesure que son carré, et il décrit le meilleur savoir possible sur
le lieu où la particule sera trouvée. Un électron unique n'est pas étalé
sur l'écran~; il est détecté entier, en un point~; ce qui est étalé, c'est
la probabilité. Tel est le contenu de la théorie, et il n'a jamais été
pris en défaut.
\end{remark}

\begin{theorem}[Principe d'incertitude de Heisenberg]\label{thm:b1:quantum-introduction:heisenberg}
\index{principe d'incertitude}
La position et la quantité de mouvement d'une particule selon un même axe
ne peuvent être toutes deux nettes~: leurs écarts types, dans n'importe
quel état, vérifient
\[
\Delta x\,\Delta p_x \geq \frac{\hbar}{2} .
\]
Une particule confinée dans une région de taille $\ell$ a donc une
dispersion en quantité de mouvement au moins égale à $\hbar/2\ell$, donc une
\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} de l'ordre de $\hbar^2/2m\ell^2$ qu'aucun refroidissement
n'enlève~: confiner coûte de l'énergie.
\end{theorem}
\admitted

\begin{example}[La taille de l'atome d'hydrogène]\label{ex:b1:quantum-introduction:hsize}
\index{rayon de Bohr}
Un électron situé à moins de $r$ du proton a $p \gtrsim \hbar/r$ et l'énergie
$E(r) \approx \dfrac{\hbar^2}{2m_er^2} - \dfrac{e^2}{4\pi\varepsilon_0r}$~: le premier
terme explose si $r$ diminue, le second s'il augmente. $\dd E/\dd r = 0$ pour
\[
a_0 = \frac{4\pi\varepsilon_0\hbar^2}{m_ee^2} = \qty{53}{pm} , \qquad
E(a_0) = -\frac{m_ee^4}{2(4\pi\varepsilon_0)^2\hbar^2} = \qty{-13.6}{eV} :
\]
le \emph{\omterm{ex:b1:quantum-introduction:hsize}{rayon de Bohr}} et l'énergie d'ionisation de l'hydrogène, jusqu'au
dernier chiffre. La physique classique n'expliquait ni pourquoi l'électron
ne tombe pas en spirale sur le noyau (il rayonne), ni pourquoi tous les
atomes d'un élément sont identiques~; le principe d'incertitude répond aux
deux en une ligne~: l'atome est aussi petit qu'il peut l'être sans que
l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} l'emporte sur la liaison.
\end{example}

\section{La quantification de l'énergie}

\begin{proposition}[Particule dans une boîte]\label{prop:b1:quantum-introduction:box}
\index{puits infini}\index{quantification}\index{état fondamental}
Une particule de masse $m$ confinée entre deux parois impénétrables
distantes de $L$ n'admet, comme la corde de Melde
(\cref{ch:b1:wave-propagation}), que les \omterm{thm:b1:wave-propagation:standing}{ondes stationnaires} $\lambda_n = 2L/n$,
donc les seules quantités de mouvement $p_n = nh/2L$ et les seules énergies
\[
E_n = \frac{n^2h^2}{8mL^2} , \qquad n = 1, 2, 3, \ldots
\]
de \omterm{def:b1:quantum-introduction:wavefunction}{fonctions d'onde} $\psi_n(x) \propto \sin(n\pi x/L)$. L'énergie est
\emph{quantifiée}~; la plus basse, l'\emph{\omterm{prop:b1:quantum-introduction:box}{état fondamental}} $E_1 > 0$, est
l'énergie de confinement du principe d'incertitude rendue exacte~; la
particule saute d'un niveau à l'autre en absorbant ou en émettant un
photon d'énergie $h\nu = E_p - E_n$.
\end{proposition}

\begin{proof}
$\psi$ doit s'annuler aux parois (la particule ne peut y être), de sorte
que l'onde loge un nombre entier de demi-longueurs d'onde dans $L$ --- la
condition de la corde~; de Broglie donne $p$, et $E = p^2/2m$. Que $\psi$
obéisse à une équation d'onde avec ces conditions aux limites est admis
(l'équation de Schrödinger, volume de deuxième année).
\end{proof}

\begin{proposition}[L'atome d'hydrogène]\label{prop:b1:quantum-introduction:hydrogen}
\index{atome d'hydrogène}\index{niveaux d'énergie}\index{raies spectrales}
L'électron de l'\omterm{prop:b1:quantum-introduction:hydrogen}{atome d'hydrogène} possède les énergies discrètes
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2} , \qquad n = 1, 2, 3, \ldots ,
\]
$n = 1$ étant l'\omterm{prop:b1:quantum-introduction:box}{état fondamental} et $E \geq 0$ l'atome ionisé. Son
spectre est fait de raies en $h\nu = E_p - E_n$~: la série de Lyman qui
descend vers $n = 1$ (ultraviolet, \qty{122}{nm} pour $2 \to 1$), la série
de Balmer qui descend vers $n = 2$ (visible~: \qty{656}{nm}, \qty{486}{nm}, \qty{434}{nm}, \ldots),
et ainsi de suite. Le modèle de Bohr (1913) obtient ces énergies en
quantifiant le \omterm{def:b1:angular-momentum:L}{moment cinétique} d'une \omterm{prop:b1:central-forces:effective}{orbite circulaire}, $m_evr = n\hbar$
(\cref{exo:b1:quantum-introduction:9})~; la théorie complète remplace les
orbites par des \omterm{def:b1:quantum-introduction:wavefunction}{fonctions d'onde}, mais garde les énergies.
\end{proposition}
\admitted

\begin{example}[La couleur par la taille~: la boîte quantique]\label{ex:b1:quantum-introduction:dot}
\index{boîte quantique}
Un nanocristal de semi-conducteur de quelques nanomètres confine ses
électrons comme une boîte~: plus le cristal est petit, plus l'énergie de
confinement qui s'ajoute à la bande interdite du semi-conducteur est
grande, et plus la lumière émise est bleue. Des boîtes de séléniure de
cadmium de rayon \qty{2}{nm} brillent en bleu, \qty{3}{nm} en jaune,
\qty{4}{nm} en rouge --- un seul matériau, des couleurs choisies à la
règle (\cref{pb:b1:quantum-introduction:1}). Les écrans de télévision en usent.
\end{example}

\begin{remark}[Ce qu'ajoute le reste de la théorie]\label{rem:b1:quantum-introduction:rest}
\index{effet tunnel}\index{laser}
L'équation de Schrödinger rend exacts la boîte, l'atome et la molécule~;
elle prédit qu'une particule peut franchir une barrière qu'elle n'a pas
l'énergie d'escalader (l'\emph{\omterm{rem:b1:quantum-introduction:rest}{effet tunnel}}, derrière la radioactivité et
le \omterm{def:b1:optical-instruments:microscope}{microscope} à \omterm{rem:b1:quantum-introduction:rest}{effet tunnel})~; que les particules identiques sont soit
sociables (les bosons --- d'où le \emph{laser}, où les photons s'entassent
dans un même mode), soit exclusives (les fermions --- d'où la
classification périodique et la rigidité de la matière)~; et que le spin
de l'électron est un \omterm{def:b1:magnetostatics:moment}{moment magnétique} sans équivalent classique. Tout
cela relève du volume de deuxième année et au-delà~; tout cela repose sur
les quelques idées de ce chapitre.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:1}
Énergie d'un photon~: radio à \qty{100}{MHz}~; lumière à \qty{500}{nm}~;
rayons X à \qty{0.10}{nm}~; rayons gamma de \qty{1}{MeV} (en donner la
\omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde}). Nombre de photons par seconde d'un laser de \qty{1.0}{mW} à
\qty{633}{nm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:2}
Sodium, $W = \qty{2.3}{eV}$~: \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} seuil~; \omterm{thm:b1:work-and-energy:ke}{énergie cinétique}
maximale et \omterm{prop:b1:quantum-introduction:photoelectric}{potentiel d'arrêt} pour une lumière de \qty{400}{nm}~; qu'est-ce
qui change si l'on double l'intensité~? et avec une lumière de \qty{600}{nm}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:3}
\omterm{thm:b1:quantum-introduction:debroglie}{Longueur d'onde de de Broglie} d'un électron de \qty{100}{eV}, de \qty{1.0}{keV}~;
d'un neutron de \qty{0.025}{eV}~; d'une bille de \qty{1}{g} à \qty{1}{m/s}.
Lesquelles un cristal (pas de \qty{0.2}{nm}) peut-il diffracter~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:4}
Un électron dans une boîte de \qty{1.0}{nm}~: $E_1$, $E_2$, \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde}
du photon émis lors de $2 \to 1$. Un proton dans une boîte de \qty{5}{fm}
(un noyau)~: $E_1$ en \unit{MeV}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:5}
\omterm{prop:b1:quantum-introduction:photoelectric}{Potentiels d'arrêt} mesurés pour un métal~: \qty{0.40}{V} à
\qty{6.0e14}{Hz}, \qty{1.23}{V} à \qty{8.0e14}{Hz}, \qty{2.05}{V} à
\qty{1.0e15}{Hz}. En déduire $h$ ($e$ étant connu) et le \omterm{prop:b1:quantum-introduction:photoelectric}{travail
d'extraction}~; la \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} seuil~; de quel métal peut-il s'agir
(césium \qty{2.1}{eV}, potassium \qty{2.3}{eV}, zinc \qty{4.3}{eV})~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:6}
Compton~: un photon X de \qty{10}{pm} est diffusé à \qty{90}{\degree} par un
électron. Nouvelle \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde}~; énergies du photon avant et après~;
énergie cédée à l'électron. Décalage relatif pour une lumière de
\qty{500}{nm} --- pourquoi il fallait des rayons X à Compton.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:7}
\emph{L'expérience de Jönsson.} Des électrons de \qty{50}{keV} (prendre $p =
\sqrt{2m_eE}$), deux fentes distantes de \qty{2.0}{\micro m}, écran à
\qty{35}{cm}. \omterm{prop:b1:wave-propagation:sinusoidal}{Longueur d'onde}~; interfrange~; pourquoi il a fallu une
\omterm{def:b1:mirrors-thin-lenses:lens}{lentille} électronique grossissante. Dans la version de Tonomura, un seul
électron à la fois traverse le dispositif~: que montre l'écran après 10,
puis après $10^5$ électrons, et qu'est-ce que cela enseigne~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:8}
\omterm{thm:b1:work-and-energy:ke}{Énergie cinétique} minimale, d'après le principe d'incertitude~: d'un
électron confiné dans \qty{0.1}{nm} (un atome)~; d'un électron confiné dans
\qty{5}{fm} (un noyau --- prendre $E \approx pc$ si le résultat est
relativiste, et conclure si un noyau peut contenir des électrons)~; d'un
grain de poussière de \qty{1}{\micro g} localisé à \qty{1}{\micro m} (donner $\Delta v$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:9}
\emph{Le modèle de Bohr.} Un électron sur une \omterm{prop:b1:central-forces:effective}{orbite circulaire} de rayon $r$
autour d'un proton~: (a) vitesse et énergie en fonction de $r$ (dynamique
coulombienne, \cref{ch:b1:central-forces})~; (b) imposer $m_evr = n\hbar$ et en déduire
$r_n = n^2a_0$, $E_n = E_1/n^2$ avec $a_0$ et $E_1$ comme dans
\cref{ex:b1:quantum-introduction:hsize}~; (c) $v_1$ et $v_1/c$~; (d)
\omterm{prop:b1:wave-propagation:sinusoidal}{longueurs d'onde} de Lyman $\alpha$ ($2 \to 1$) et de H$\alpha$ ($3 \to 2$)~;
(e) montrer que l'orbite $n$ contient exactement $n$ \omterm{thm:b1:quantum-introduction:debroglie}{longueurs d'onde de de Broglie}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:10}
\emph{Les hydrogénoïdes.} (a) Pour un noyau de charge $Ze$, montrer que
$E_n = -Z^2 \times \qty{13.6}{eV}/n^2$ et $r_n = n^2a_0/Z$~: énergie
d'ionisation de He$^+$ et du dernier électron de l'uranium ($Z = 92$~; commenter).
(b) Hydrogène muonique~: le muon a $207$ fois la masse de l'électron~:
rayon et énergie fondamentale~; \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} de sa raie Lyman $\alpha$~;
pourquoi on s'en sert pour mesurer la taille du proton (rayon \qty{0.84}{fm}).
(c) Positronium (un électron et un positron, de masses égales)~: montrer que
la \omterm{prop:b1:systems-of-points:twobody}{masse réduite} divise les énergies par deux~: sa raie Lyman $\alpha$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:11}
\emph{Les \omterm{ex:b1:quantum-introduction:dot}{boîtes quantiques}.} Dans un nanocristal sphérique de rayon $R$,
l'énergie de confinement d'une particule de masse effective $m^\ast$ vaut
$h^2/8m^\ast R^2$ (admis). CdSe~: bande interdite $E_g = \qty{1.74}{eV}$, $m_e^\ast =
0.13\,m_e$, $m_h^\ast = 0.45\,m_e$ (le «~trou~» laissé par l'électron
excité confine lui aussi). (a) Énergie du photon émis $E_g + E_e + E_h$
et \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} pour $R = \qty{2.0}{nm}$, \qty{3.0}{nm}, \qty{4.0}{nm}~; couleurs.
(b) Rayon pour une émission à \qty{520}{nm} (vert). (c) Pourquoi le CdSe
massif n'émet-il qu'à \qty{713}{nm}~? (d) Une dispersion de \qty{10}{\percent} sur
le rayon dans un lot~: dispersion en \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} à \qty{3}{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:12}
\emph{L'énergie de point zéro.} (a) Pour un \omterm{def:b1:oscillators-resonance:harmonic}{oscillateur harmonique} $E = p^2/2m +
\tfrac12m\omega^2x^2$, prendre $\Delta x\,\Delta p = \hbar/2$ et minimiser $E \approx
\Delta p^2/2m + \tfrac12m\omega^2\Delta x^2$ par rapport à $\Delta x$~: montrer que $E_{\min} = \hbar\omega/2$
(c'est l'énergie fondamentale exacte). (b) La molécule H$_2$ vibre à $\omega =
\qty{8.3e14}{rad/s}$~: énergie de point zéro en eV, et température la plus
basse à laquelle sa vibration est «~thermique~» ($k_BT \sim \hbar\omega$).
(c) L'hélium liquide ne gèle jamais sous la pression atmosphérique~: estimer
l'énergie de point zéro d'un atome d'hélium localisé à $\Delta x = \qty{0.05}{nm}$
(une fraction de la distance interatomique) dans un cristal, et la comparer à
son énergie de liaison dans le solide, environ \qty{1}{meV}. (d) Comparer la formule de la boîte $E_1 = h^2/8mL^2$ à la
borne d'incertitude pour $\Delta x = L/2$~: laquelle est la plus grande, et pourquoi doit-elle l'être~?
\end{exercise}

\section{Problème~: le spectre de l'hydrogène et une boîte quantique}

\begin{problem}[{Problème du week-end --- mesurer $h$ avec de la lumière et
un voltmètre, construire l'atome d'hydrogène à partir de trois constantes,
et choisir la couleur d'un cristal par sa taille}]
\label{pb:b1:quantum-introduction:1}
Constantes~: $h = \qty{6.626e-34}{J.s}$, $\hbar = \qty{1.055e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$, $e = \qty{1.602e-19}{C}$, $m_e = \qty{9.11e-31}{kg}$,
$1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$, $hc = \qty{1240}{eV.nm}$.

\medskip
\noindent\textbf{Partie I --- Mesurer la \omterm{thm:b1:quantum-introduction:photon}{constante de Planck}.}
Une photocathode au potassium est éclairée en lumière monochromatique~; on
mesure le \omterm{prop:b1:quantum-introduction:photoelectric}{potentiel d'arrêt}~: \qty{0.30}{V} à \qty{6.0e14}{Hz}, \qty{0.69}{V} à
\qty{7.0e14}{Hz}, \qty{1.13}{V} à \qty{8.0e14}{Hz}, \qty{1.50}{V} à
\qty{9.0e14}{Hz}, \qty{2.36}{V} à \qty{1.1e15}{Hz}.
\begin{enumerate}
  \item Décrire la cellule et expliquer pourquoi une tension inverse peut
        annuler le courant, et ce que mesure $eV_s$.
  \item Écrire la relation d'Einstein et dire ce que donnent la pente et
        l'ordonnée à l'origine de $V_s(\nu)$.
  \item À partir des mesures (\omterm{prop:b1:units-dimensions:regression}{régression linéaire}, ou les deux points
        extrêmes), en déduire $h$~; comparer à la valeur admise.
  \item \omterm{prop:b1:quantum-introduction:photoelectric}{Travail d'extraction} du potassium en eV~; fréquence et \omterm{prop:b1:wave-propagation:sinusoidal}{longueur
        d'onde} seuils~; un pointeur laser rouge arracherait-il des
        électrons~?
  \item On double l'intensité de la lampe~: qu'advient-il de $V_s$, du
        courant~? Pourquoi est-ce incompatible avec une image purement
        ondulatoire~?
  \item Estimation classique~: une lumière de \qty{1}{\micro W/m^2} tombe sur
        le métal~; un atome offre une aire d'environ \qty{1e-20}{m^2}~:
        combien de temps faudrait-il pour accumuler \qty{2.2}{eV} sur un
        atome~? Qu'observe-t-on à la place~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- L'\omterm{prop:b1:quantum-introduction:hydrogen}{atome d'hydrogène} de Bohr.}
\begin{enumerate}[resume]
  \item Un électron sur une \omterm{prop:b1:central-forces:effective}{orbite circulaire} de rayon $r$ autour d'un
        proton fixe~: vitesse $v(r)$ et \omterm{thm:b1:work-and-energy:em}{énergie mécanique} $E(r)$.
  \item Condition de Bohr $m_evr = n\hbar$~: montrer que $r_n = n^2a_0$ et
        donner $a_0 = 4\pi\varepsilon_0\hbar^2/m_ee^2$ numériquement.
  \item $E_n = E_1/n^2$ avec $E_1 = -m_ee^4/[2(4\pi\varepsilon_0)^2\hbar^2]$~;
        valeur numérique en eV.
  \item Vitesse sur la première orbite et rapport $v_1/c$ (la constante
        de structure fine $\alpha \approx 1/137$).
  \item \omterm{prop:b1:wave-propagation:sinusoidal}{Longueurs d'onde} des raies $2 \to 1$ (Lyman $\alpha$), $3 \to 2$ et
        $4 \to 2$ (Balmer H$\alpha$, H$\beta$)~: lesquelles sont visibles~?
  \item Montrer que $1/\lambda = R_H(1/n^2 - 1/p^2)$ et calculer la constante
        de Rydberg $R_H$~; la plus courte \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} de Balmer.
  \item Énergie d'ionisation de l'hydrogène depuis l'\omterm{prop:b1:quantum-introduction:box}{état fondamental}~;
        \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} du photon qui l'ionise tout juste.
  \item De Broglie~: montrer que la circonférence de l'orbite $n$ vaut $n$
        \omterm{prop:b1:wave-propagation:sinusoidal}{longueurs d'onde} --- la condition de Bohr est une condition d'\omterm{thm:b1:wave-propagation:standing}{onde
        stationnaire}.
  \item Deux choses que le modèle rate, une qu'il donne exactement juste,
        et ce qui remplace l'orbite dans la théorie complète.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La \omterm{ex:b1:quantum-introduction:dot}{boîte quantique}.}
Un nanocristal de CdSe de rayon $R$ confine un électron excité
(masse effective $m_e^\ast = 0.13\,m_e$) et le trou qu'il laisse
($m_h^\ast = 0.45\,m_e$)~; l'énergie de confinement de chacun dans une
sphère de rayon $R$ vaut $h^2/8m^\ast R^2$ (admis)~; la bande interdite du CdSe massif est
$E_g = \qty{1.74}{eV}$.
\begin{enumerate}[resume]
  \item Établir les énergies $E_n = n^2h^2/8mL^2$ d'une particule dans une
        boîte à une dimension à partir de la condition d'\omterm{thm:b1:wave-propagation:standing}{onde
        stationnaire}, et commenter l'analogie avec la formule de la sphère.
  \item Énergies de confinement de l'électron et du trou pour
        $R = \qty{2.0}{nm}$.
  \item Énergie et \omterm{prop:b1:wave-propagation:sinusoidal}{longueur d'onde} du photon émis pour $R = \qty{2.0}{nm}$,
        \qty{3.0}{nm} et \qty{4.0}{nm}~; couleurs.
  \item Expliquer en une phrase pourquoi les petites boîtes sont plus bleues.
  \item Vérifier l'\omterm{def:b1:units-dimensions:oom}{ordre de grandeur} avec le principe d'incertitude~:
        $\Delta p \sim \hbar/R$ pour l'électron à $R = \qty{2}{nm}$, et
        l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} qui en découle.
  \item Une boîte émet \qty{1.0}{nW} à \qty{577}{nm}~: photons par seconde.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Compter les quanta.}
\begin{enumerate}[resume]
  \item Photons par seconde d'un laser de \qty{1.0}{mW} à \qty{633}{nm}.
  \item L'\omterm{def:b1:optical-instruments:eye}{œil} adapté à l'obscurité détecte un éclair d'environ $100$ photons
        à \qty{500}{nm} arrivant en \qty{0.1}{s}~: la puissance correspondante.
  \item Une antenne FM reçoit \qty{1.0}{pW} à \qty{100}{MHz}~: photons par
        seconde~; peut-on espérer les détecter un à un~?
  \item Bilan~: à partir de $h$, $e$, $m_e$ seuls, quelles trois grandeurs du
        monde atomique ce problème a-t-il produites, et avec quels \omterm{def:b1:units-dimensions:oom}{ordres
        de grandeur}~?
\end{enumerate}
\end{problem}
