\chapter{Les lois de Newton}\label{ch:b1:newton-dynamics}

Un parachutiste saute de l'avion et, en une douzaine de secondes, cesse
d'accélérer~: à deux cents kilomètres à l'heure, l'air le repousse
exactement aussi fort que la Terre l'attire. Une voiture prend un virage
trop vite et les pneus lâchent~; un camion en montagne rampe en
première, retenu par le frottement même qui lui permet de s'arrêter sur
le plat. La cinématique décrivait les mouvements~; la dynamique les
explique, à partir d'une seule loi qui relie l'accélération d'un corps
aux \omterm{def:b1:newton-dynamics:momentum}{forces} qu'il subit. Ce chapitre énonce les trois \omterm{thm:b1:newton-dynamics:laws}{lois de Newton},
recense les \omterm{def:b1:newton-dynamics:momentum}{forces} que l'on rencontre au niveau de ce volume --- poids,
ressorts, fils, contacts, \omterm{prop:b1:newton-dynamics:friction}{frottements solides} et fluides --- et montre
comment passer d'un schéma des \omterm{def:b1:newton-dynamics:momentum}{forces} à une équation différentielle et à
sa solution.

\section{Masse, quantité de mouvement et les trois lois}

\begin{definition}[Masse, quantité de mouvement, force]\label{def:b1:newton-dynamics:momentum}
\index{masse}\index{quantité de mouvement}\index{force}
Un \omterm{def:b1:point-kinematics:frame}{point matériel} possède une \emph{masse inertielle} $m > 0$ (en
kilogrammes), mesure de sa résistance aux changements de vitesse, et
indépendante du \omterm{def:b1:point-kinematics:frame}{référentiel}. Sa \emph{quantité de mouvement} dans un
\omterm{def:b1:point-kinematics:frame}{référentiel} vaut
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
Une \emph{force} $\vect F$ (en newtons) est l'action d'un autre corps sur
le point~; les forces s'ajoutent comme des vecteurs, et la
\emph{résultante} est leur somme $\sum\vect F$.
\end{definition}

\begin{theorem}[Lois de Newton]\label{thm:b1:newton-dynamics:laws}
\index{lois de Newton}\index{référentiel galiléen}\index{principe d'inertie}\index{action et réaction}
\begin{enumerate}
  \item \emph{Principe d'inertie.} Il existe des \omterm{def:b1:point-kinematics:frame}{référentiels}, dits
        \emph{galiléens}, dans lesquels un point soumis à aucune \omterm{def:b1:newton-dynamics:momentum}{force},
        ou à des \omterm{def:b1:newton-dynamics:momentum}{forces} de résultante nulle, se déplace en ligne droite
        à vitesse constante. Tout \omterm{def:b1:point-kinematics:frame}{référentiel} en translation rectiligne
        uniforme par rapport à un \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen} est galiléen.
  \item \emph{Loi de la \omterm{def:b1:newton-dynamics:momentum}{quantité de mouvement}.} Dans un \omterm{thm:b1:newton-dynamics:laws}{référentiel galiléen},
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{c.-à-d.}\qquad
        m\,\vect a = \sum\vect F \ \text{ pour une masse constante } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{Action et réaction}.} Si un corps $A$ exerce $\vect F_{A\to B}$
        sur un corps $B$, alors $B$ exerce $\vect F_{B\to A} = -\vect F_{A\to B}$
        sur $A$, selon la droite qui les joint.
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[Ce que disent ces lois]\label{rem:b1:newton-dynamics:meaning}
La première loi n'est pas un cas particulier de la deuxième~: elle
\emph{définit} les \omterm{def:b1:point-kinematics:frame}{référentiels} dans lesquels la deuxième vaut. En
pratique, un laboratoire terrestre est galiléen avec une excellente
précision pour des expériences de quelques minutes (le
\cref{ch:b1:non-inertial-frames} en chiffre l'erreur)~; un \omterm{def:b1:point-kinematics:frame}{référentiel}
lié au Soleil et aux étoiles lointaines l'est mieux encore. La deuxième
loi est une équation vectorielle~: trois équations scalaires, une par
axe, et le choix des axes (\cref{ch:b1:point-kinematics}) est la
première décision de tout problème. La troisième loi vaut à chaque
instant pour les \omterm{def:b1:newton-dynamics:momentum}{forces} de contact et de gravitation de ce volume.
\end{remark}

\begin{definition}[Système fermé~; conservation de la quantité de mouvement]\label{def:b1:newton-dynamics:conservation}
\index{système fermé}\index{conservation de la quantité de mouvement}
Un \emph{système fermé} n'échange pas de matière avec l'extérieur. Pour
un système de points, les \omterm{def:b1:newton-dynamics:momentum}{forces} intérieures se compensent deux à deux
(troisième loi), et la \omterm{def:b1:newton-dynamics:momentum}{quantité de mouvement} totale
$\vect P = \sum m_i\vect v_i$ obéit à
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$~: elle se conserve
lorsque les \omterm{def:b1:newton-dynamics:momentum}{forces} extérieures s'annulent (le
\cref{ch:b1:systems-of-points} développe ce point).
\end{definition}

\section{Les forces usuelles}

\begin{proposition}[Poids et gravitation]\label{prop:b1:newton-dynamics:weight}
\index{poids}\index{champ de pesanteur}
Au voisinage de la surface terrestre, un corps de \omterm{def:b1:newton-dynamics:momentum}{masse} $m$ est attiré
par son \emph{poids} $\vect P = m\vect g$, où $\vect g$ est le \omterm{prop:b1:newton-dynamics:weight}{champ de
pesanteur} local, $g \approx \qty{9.81}{m/s^2}$, dirigé vers le bas (vers
le centre, à très peu près). Plus généralement, une \omterm{def:b1:newton-dynamics:momentum}{masse} $M$ à la
distance $r$ attire $m$ avec $\vect F = -GMm/r^2\,\vect e_r$
(\cref{ch:b1:central-forces})~; à la surface d'une sphère de rayon $R$,
$g = GM/R^2$. Noter l'unité~: \unit{N/kg} et \unit{m/s^2} sont la même
chose, et c'est pourquoi tous les corps tombent de la même façon.
\end{proposition}

\begin{proof}
La loi de la gravitation de Newton est admise et explorée au
\cref{ch:b1:central-forces}~; $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$.
\end{proof}

\begin{proposition}[Ressort~; fil]\label{prop:b1:newton-dynamics:springstring}
\index{loi de Hooke}\index{ressort}\index{tension}
Un ressort idéal de raideur $k$ et de longueur à vide $\ell_0$, étiré à
la longueur $\ell$ le long du vecteur unitaire $\vect u$ dirigé de son
extrémité fixe vers le corps, exerce sur ce corps la \emph{\omterm{def:b1:newton-dynamics:momentum}{force} de
rappel}
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{loi de Hooke}),
\]
dirigée vers sa longueur à vide. Un fil idéal (sans \omterm{def:b1:newton-dynamics:momentum}{masse},
inextensible) exerce une \emph{tension} $\vect T$ selon sa propre
direction, en tirant, de même intensité aux deux extrémités et tout le
long (même par-dessus une poulie sans \omterm{def:b1:newton-dynamics:momentum}{masse} ni frottement).
\end{proposition}

\begin{proof}
La \omterm{prop:b1:newton-dynamics:springstring}{loi de Hooke} est un modèle empirique valable pour de petites
déformations. Pour le fil~: un élément sans \omterm{def:b1:newton-dynamics:momentum}{masse} obéit à
$0 = \vect T_1 + \vect T_2$ par la deuxième loi de Newton, donc la
tension se transmet inchangée.
\end{proof}

\begin{proposition}[Forces de contact~: réaction normale et frottement solide]\label{prop:b1:newton-dynamics:friction}
\index{réaction normale}\index{frottement solide}\index{lois de Coulomb du frottement}\index{coefficient de frottement}
Une surface solide exerce sur un corps qui la touche une \emph{\omterm{prop:b1:newton-dynamics:friction}{réaction
normale}} $\vect N$, perpendiculaire à la surface et répulsive
($N \geq 0$~: une surface ne peut pas tirer), et une \emph{\omterm{def:b1:newton-dynamics:momentum}{force} de
frottement} tangentielle $\vect T$. Lois de Coulomb~: si le corps ne
glisse pas, $\abs{\vect T} \leq \mu_s N$ (le frottement \emph{statique}
s'ajuste à ce qu'il faut, jusqu'à cette limite)~; s'il glisse à la
vitesse $\vect v_g$ par rapport à la surface,
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (frottement
\emph{dynamique}, qui s'oppose au glissement), avec des coefficients
$\mu_d \leq \mu_s$ de l'ordre de $0.1$ à $1$ selon les matériaux,
indépendants de l'aire de contact et de la vitesse.
\end{proposition}
\admitted

\begin{proposition}[Frottement fluide]\label{prop:b1:newton-dynamics:drag}
\index{frottement fluide}\index{traînée}\index{loi de Stokes}
Un corps qui se déplace à la vitesse $\vect v$ dans un fluide au repos
subit une \emph{traînée} opposée à $\vect v$~: à faible vitesse (petite
taille, fluide visqueux) elle est linéaire, $\vect F = -\alpha\vect v$
--- pour une sphère de rayon $r$ dans un fluide de viscosité $\eta$,
$\alpha = 6\pi\eta r$ (Stokes)~; à grande vitesse (corps gros et
rapides dans l'air ou l'eau) elle est quadratique,
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
où $\rho$ est la \omterm{def:b1:newton-dynamics:momentum}{masse} volumique du fluide, $S$ la section du corps face
à l'écoulement et $C_x$ un \emph{coefficient de traînée} \omterm{def:b1:units-dimensions:dimension}{sans dimension}
($0.5$ pour une sphère, $0.3$ pour une voiture, $1$ pour un disque
plat). Le nombre de Reynolds $\rho vL/\eta$
(\cref{rem:b1:units-dimensions:pi}) tranche~: linéaire en dessous de
$1$, quadratique au-dessus de $1000$ environ.
\end{proposition}
\admitted

\begin{remark}[D'autres forces]\label{rem:b1:newton-dynamics:others}
La poussée d'Archimède est une résultante de \omterm{def:b1:newton-dynamics:momentum}{forces} de pression,
établie au \cref{ch:b1:fluid-statics}~; les \omterm{def:b1:newton-dynamics:momentum}{forces} électrique et
magnétique sur une charge font l'objet du
\cref{ch:b1:charged-particles}. Toutes les \omterm{def:b1:newton-dynamics:momentum}{forces} de ce chapitre sont
des \emph{modèles} d'interactions de contact ou de champ, valables dans
des limites énoncées --- le ressort de Hooke casse, le frottement de
Coulomb échoue à haute pression, la traînée de Stokes échoue à grande
vitesse. Vérifier que le modèle employé s'applique fait partie de toute
solution.
\end{remark}

\section{Résoudre un problème de dynamique}

\begin{method}[De la situation à l'équation]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{Système}~: nommer le corps traité comme un point.
  \item \emph{\omterm{def:b1:point-kinematics:frame}{Référentiel}}~: le nommer et vérifier qu'il est galiléen
        (ou le traiter comme tel à la précision requise).
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{Forces}}~: lister tous les corps qui touchent ou attirent
        le système, et dessiner chaque \omterm{def:b1:newton-dynamics:momentum}{force} sur un schéma (le bilan des
        \omterm{def:b1:newton-dynamics:momentum}{forces})~; rien d'autre n'intervient.
  \item \emph{Coordonnées}~: choisir des axes adaptés au mouvement
        (\cref{met:b1:point-kinematics:choose}) et exprimer $\vect a$.
  \item \emph{Projeter} $m\vect a = \sum\vect F$ sur les axes~; utiliser
        les contraintes (un corps sur une surface y reste~:
        $a_\perp = 0$~; un fil de longueur fixe~: mêmes vitesses aux deux
        extrémités).
  \item \emph{Résoudre} et \emph{vérifier}~: dimensions, cas limites,
        signes ($N \geq 0$, $\abs T \leq \mu_sN$ au repos).
\end{enumerate}
\end{method}

\begin{example}[Bloc sur un plan incliné]\label{ex:b1:newton-dynamics:incline}
Un bloc de \omterm{def:b1:newton-dynamics:momentum}{masse} $m$ sur un plan incliné de $\alpha$, de coefficients
$\mu_s$ et $\mu_d$. Axes~: $x$ vers le bas de la pente, $y$ selon la
normale. Poids $(mg\sin\alpha, -mg\cos\alpha)$, réaction $(0, N)$,
frottement $(-T, 0)$ s'il glisse vers le bas. Selon $y$~:
$N = mg\cos\alpha$. Au repos, selon $x$~:
$T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$, c'est-à-dire $\tan\alpha
\leq \mu_s$~: le bloc tient jusqu'à l'angle $\arctan\mu_s$ --- une
mesure de $\mu_s$ qui ne demande aucune balance. En glissement~: $ma = mg\sin\alpha -
\mu_dmg\cos\alpha$, donc $a = g(\sin\alpha - \mu_d\cos\alpha)$,
indépendante de $m$~: \qty{3.2}{m/s^2} pour $\alpha = 30^\circ$ et $\mu_d = 0.2$.
\end{example}

\begin{example}[Le pendule simple]\label{ex:b1:newton-dynamics:pendulum}
\index{pendule simple}
Une \omterm{def:b1:newton-dynamics:momentum}{masse} $m$ au bout d'un fil de longueur $\ell$, faisant l'angle
$\theta$ avec la verticale descendante. En \omterm{def:b1:point-kinematics:cylindrical}{coordonnées polaires} au point
de suspension~: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$~; les
\omterm{def:b1:newton-dynamics:momentum}{forces}~: le poids $mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$
et la tension $-T\vect e_r$. Selon $\vect e_\theta$~: $m\ell\ddot\theta =
-mg\sin\theta$,
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
qui, pour de petits angles ($\sin\theta \approx \theta$), est l'équation
harmonique de \omterm{def:b1:wave-propagation:signal}{pulsation} $\omega_0 = \sqrt{g/\ell}$~: période
$T_0 = 2\pi\sqrt{\ell/g}$, le $k = 2\pi$ que
l'\cref{ex:b1:units-dimensions:pendulum} ne pouvait pas fournir. Selon
$\vect e_r$~: $T = mg\cos\theta + m\ell\dot\theta^2$ --- le fil tire le
plus fort en bas, là où la vitesse est la plus grande.
\end{example}

\begin{proposition}[Chute avec traînée~; vitesse limite]\label{prop:b1:newton-dynamics:terminal}
\index{vitesse limite}
Un corps de \omterm{def:b1:newton-dynamics:momentum}{masse} $m$ qui tombe du repos sous son poids et une traînée
atteint une \emph{\omterm{prop:b1:newton-dynamics:terminal}{vitesse limite}} pour laquelle les deux se compensent.
Traînée linéaire~: $m\dot v = mg - \alpha v$, $v = v_\infty(1 - \eu^{-t/\tau})$
avec $v_\infty = mg/\alpha$ et $\tau = m/\alpha$. Traînée quadratique~: $m\dot v = mg -
\tfrac12\rho C_xSv^2$, $v = v_\infty\tanh(t/\tau)$ avec
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
Linéaire~: une équation du premier ordre, $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder}). Quadratique~: avec
$u = v/v_\infty$, $\dot u = (1 - u^2)/\tau$~; en séparant les variables,
$\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$ (la fraction
rationnelle $1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$ s'intègre en
$\tfrac12\ln\frac{1 + u}{1 - u}$), d'où $u = \tanh(t/\tau)$.
\end{proof}

\begin{example}[Deux vitesses limites]\label{ex:b1:newton-dynamics:terminal}
Un parachutiste, $m = \qty{80}{kg}$, à plat ventre, $C_xS \approx \qty{0.8}{m^2}$
dans l'air ($\rho = \qty{1.2}{kg/m^3}$)~: $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$, atteinte (à $90\%$) en environ $1.5\tau =
\qty{6}{s}$. Une gouttelette de brouillard, $r = \qty{10}{\micro m}$, en
régime de Stokes ($\eta_{\text{air}} = \qty{1.8e-5}{Pa.s}$)~: $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- et c'est pourquoi le brouillard reste suspendu.
\end{example}

\section{Mouvement sur un cercle~: les forces de liaison}

\begin{proposition}[L'exigence centripète]\label{prop:b1:newton-dynamics:centripetal}
\index{force centripète}
Un point qui se déplace sur un cercle de rayon $R$ à la vitesse $v$ doit
recevoir de la résultante des \omterm{def:b1:newton-dynamics:momentum}{forces} une composante $mv^2/R$ dirigée
vers le centre. Ce qui la fournit --- la tension d'un fil, le
frottement, la réaction d'une route relevée, la pesanteur --- fixe la
vitesse maximale, ou la vitesse minimale, pour laquelle le mouvement est
possible.
\end{proposition}

\begin{proof}
La deuxième loi de Newton avec l'accélération de Frenet~: la composante
normale de $\sum\vect F$ vaut $mv^2/\rho$.
\end{proof}

\begin{example}[Trois cercles]\label{ex:b1:newton-dynamics:circles}
\emph{Virage à plat}~: le frottement doit fournir $mv^2/R \leq \mu_sN = \mu_smg$,
donc $v \leq \sqrt{\mu_sgR}$~: \qty{20}{m/s} pour $R = \qty{50}{m}$ et
$\mu_s = 0.8$. \emph{Virage relevé} d'un angle $\beta$, sans
frottement~: la réaction $\vect N$ est normale à la route~; à la
verticale $N\cos\beta = mg$, à l'horizontale $N\sin\beta = mv^2/R$, donc
$v^2 = gR\tan\beta$ --- l'unique vitesse à laquelle le virage se prend
sans aucun frottement. \emph{Boucle}~: au sommet, avec $\vect N$ dirigée
vers le bas vers le centre, $N + mg =
mv^2/R$, et $N \geq 0$ exige $v \geq \sqrt{gR}$~: en dessous, la voiture
quitte le rail (le \cref{pb:b1:point-kinematics:1} à nouveau, expliqué cette fois).
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:1}
Une personne de \qty{70}{kg} se tient sur un pèse-personne dans un
ascenseur. Qu'indique l'appareil (en newtons et en «~kilogrammes~»)
lorsque l'ascenseur accélère vers le haut de \qty{2.0}{m/s^2}, vers le
bas de \qty{2.0}{m/s^2}, se déplace à vitesse constante, et tombe en
chute libre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:2}
Un bloc $m_1 = \qty{2.0}{kg}$ posé sur une table sans frottement est
relié par un fil, passant sur une poulie sans frottement, à une \omterm{def:b1:newton-dynamics:momentum}{masse}
suspendue $m_2 = \qty{1.0}{kg}$. Trouver l'accélération et la tension.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:3}
Un bloc glisse sur un plan incliné à $30^\circ$ avec $\mu_d = 0.20$.
Calculer son accélération et sa vitesse après \qty{5.0}{m} depuis le
repos. La réponse dépend-elle de sa \omterm{def:b1:newton-dynamics:momentum}{masse}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:4}
Une \omterm{def:b1:newton-dynamics:momentum}{masse} de \qty{0.50}{kg} est suspendue à un ressort de raideur
\qty{200}{N/m}. Trouver l'allongement statique~; puis la \omterm{def:b1:wave-propagation:signal}{pulsation} et la
période de ses oscillations verticales autour de l'équilibre (montrer
que l'équation du mouvement est harmonique).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:5}
Une caisse posée sur un plateau que l'on incline se met à glisser à
$22^\circ$. Trouver $\mu_s$. Une voiture sur route plate avec
$\mu_s = 0.80$~: vitesse maximale dans un virage de rayon \qty{50}{m},
puis sur route mouillée avec $\mu_s = 0.40$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:6}
\omterm{prop:b1:newton-dynamics:terminal}{Vitesses limites}~: un parachutiste ($m = \qty{80}{kg}$, $C_xS = \qty{0.70}{m^2}$,
$\rho = \qty{1.2}{kg/m^3}$)~; une gouttelette de brouillard de rayon
\qty{10}{\micro m} dans l'air ($\eta = \qty{1.8e-5}{Pa.s}$), par la \omterm{prop:b1:newton-dynamics:drag}{loi
de Stokes}. Vérifier par un nombre de Reynolds que le régime retenu est
le bon dans chaque cas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:7}
Un pendule de longueur $\ell$ est lâché sans vitesse initiale depuis
$\theta_0 = 60^\circ$. Sachant que sa vitesse en bas vaut
$\sqrt{2g\ell(1 - \cos\theta_0)}$ (énergie, chapitre suivant), trouver
la tension du fil en bas, en unités de $mg$, puis au point de lâcher.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:8}
Un virage de rayon \qty{200}{m} est relevé de $15^\circ$. À quelle
vitesse peut-il être pris sans aucun frottement~? Que doit faire le
frottement à plus basse et à plus haute vitesse~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:9}
Pendule conique~: une \omterm{def:b1:newton-dynamics:momentum}{masse} au bout d'un fil de longueur
$\ell = \qty{1.0}{m}$ décrit un cercle horizontal, le fil faisant
$30^\circ$ avec la verticale. Trouver la \omterm{cor:b1:point-kinematics:circular}{vitesse angulaire}, la période
et la tension (en unités de $mg$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:10}
Une voiture de \qty{1000}{kg} atteint \qty{100}{km/h} depuis le repos en
\qty{8.0}{s} à accélération constante. Trouver la \omterm{def:b1:newton-dynamics:momentum}{force} résultante, et
le \omterm{prop:b1:newton-dynamics:friction}{coefficient de frottement} minimal entre les pneus moteurs et la route
si les roues motrices portent tout le poids. Qu'est-ce qui limite une
voiture de sport sur route mouillée~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:11}
Un corps tombe du repos avec une traînée linéaire, $\tau = m/\alpha = \qty{0.50}{s}$.
Écrire $v(t)$ et en déduire la distance de chute $z(t)$~; calculer
$v_\infty$, puis $z$ après \qty{2.0}{s}~; comparer à la chute libre.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:12}
Une voiture de \omterm{def:b1:newton-dynamics:momentum}{masse} $m$ roule à l'intérieur d'une boucle circulaire
verticale de rayon $R$ (\cref{pb:b1:point-kinematics:1}). Exprimer la
\omterm{prop:b1:newton-dynamics:friction}{réaction normale} du rail en fonction de l'angle $\theta$ compté depuis
le bas et de la vitesse $v(\theta)$~; en déduire la vitesse minimale au
sommet~; à l'aide de $v^2 = v_0^2 - 2gR(1 - \cos\theta)$, trouver la
vitesse d'entrée minimale et la réaction en bas dans ce cas.
\end{exercise}

\section{Problème~: le parachutiste}

\begin{problem}[{Problème du week-end --- de la porte de l'avion au
sol~: la chute libre, l'air qui repousse, une voilure qui s'ouvre trop
vite, et l'atterrissage --- une loi du mouvement, deux lois de traînée}]
\label{pb:b1:newton-dynamics:1}
Un parachutiste équipé a une \omterm{def:b1:newton-dynamics:momentum}{masse} $m = \qty{80}{kg}$. Air~:
$\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$. En chute à plat
ventre, $C_xS = \qty{0.80}{m^2}$~; sous la voilure ouverte, $C_xS =
\qty{25}{m^2}$. $g = \qty{9.81}{m/s^2}$~; $z$ est compté vers le bas
depuis le point de saut.

\medskip
\noindent\textbf{Partie I --- Sans l'air.}
\begin{enumerate}
  \item Écrire la deuxième loi de Newton et donner $v(t)$ et $z(t)$~;
        calculer les deux à $t = \qty{10}{s}$.
  \item Calculer le poids.
  \item Le parachutiste se sent «~sans poids~» pendant cette phase, bien
        que la pesanteur agisse pleinement. Expliquer par les \omterm{def:b1:newton-dynamics:momentum}{forces} (ce
        qui manque par rapport à quelqu'un debout sur le sol).
  \item Le \omterm{def:b1:point-kinematics:frame}{référentiel} terrestre est-il assez galiléen pour ce problème~?
        Nommer ce que l'on néglige.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La chute dans l'air.}
\begin{enumerate}[resume]
  \item Écrire la \omterm{def:b1:newton-dynamics:momentum}{force} de traînée pour la chute à plat ventre en
        fonction de $v$.
  \item Écrire l'équation du mouvement pour $v(t)$.
  \item Trouver la \omterm{prop:b1:newton-dynamics:terminal}{vitesse limite} $v_\infty$ (en \unit{m/s} et en
        \unit{km/h}).
  \item Avec $u = v/v_\infty$ et $\tau = v_\infty/g$, montrer que
        $\dd u/\dd t = (1 - u^2)/\tau$~; calculer $\tau$.
  \item Résoudre par séparation des variables et montrer que
        $v = v_\infty\tanh(t/\tau)$.
  \item Au bout de combien de temps a-t-on $v = 0.9v_\infty$~? Et
        $0.99v_\infty$~?
  \item Montrer que $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ et calculer la
        distance de chute à $t = \qty{10}{s}$~; comparer à la partie~I.
  \item Estimer le nombre de Reynolds à $v_\infty$ avec $L = \qty{0.5}{m}$
        et justifier la loi quadratique.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La voilure s'ouvre.}
\begin{enumerate}[resume]
  \item Calculer la nouvelle \omterm{prop:b1:newton-dynamics:terminal}{vitesse limite} sous la voilure ouverte.
  \item Si la voilure s'ouvrait instantanément à $v = \qty{40}{m/s}$,
        quelle serait la décélération, en \unit{m/s^2} et en $g$~?
        Pourquoi est-ce inacceptable~?
  \item Les voilures réelles s'ouvrent en \qty{3}{s} environ. Estimer la
        décélération moyenne et la \omterm{def:b1:newton-dynamics:momentum}{force} moyenne du harnais sur le
        parachutiste.
  \item Une fois la voilure entièrement ouverte, la vitesse décroît vers
        la nouvelle $v_\infty$~: quelle est la nouvelle \omterm{def:b1:transient-regimes:firstorder}{constante de
        temps} $\tau' = v_\infty'/g$, et que signifie-t-elle pour
        l'approche du \omterm{def:b1:transient-regimes:firstorder}{régime permanent}~?
  \item En descente établie, que vaut la \omterm{def:b1:newton-dynamics:momentum}{force} du harnais~? Justifier
        par la deuxième loi.
  \item Expliquer pourquoi la \omterm{prop:b1:newton-dynamics:terminal}{vitesse limite} varie comme $1/\sqrt{C_xS}$~:
        de quel facteur une voilure de moitié d'aire l'élèverait-elle~?
  \item Un sauteur plus lourd (\qty{100}{kg}) sous la même voilure~:
        \omterm{prop:b1:newton-dynamics:terminal}{vitesse limite}~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- L'atterrissage.}
\begin{enumerate}[resume]
  \item La vitesse d'atterrissage est la $v_\infty$ de la voilure. De
        quelle hauteur faudrait-il sauter, sans air, pour toucher le sol
        à cette vitesse~?
  \item Un atterrissage jambes raides arrête le corps sur \qty{0.5}{m}~;
        un roulé-boulé sur \qty{1.5}{m}. Estimer la décélération moyenne
        (en $g$) et la \omterm{def:b1:newton-dynamics:momentum}{force} moyenne sur les jambes dans chaque cas, puis
        conclure.
  \item Juste avant de toucher, le parachutiste tire les deux
        commandes, ce qui augmente brièvement la traînée de la voilure.
        Expliquer qualitativement l'effet sur la vitesse.
\end{enumerate}

\medskip
\noindent\textbf{Partie V --- Les petits objets tombent autrement.}
\begin{enumerate}[resume]
  \item Une goutte de pluie de rayon \qty{1.0}{mm} ($C_x = 0.5$)~:
        calculer sa \omterm{prop:b1:newton-dynamics:terminal}{vitesse limite} par la loi quadratique, et vérifier
        le nombre de Reynolds.
  \item Une gouttelette de brouillard de rayon \qty{10}{\micro m}~:
        utiliser la \omterm{prop:b1:newton-dynamics:drag}{loi de Stokes} pour $v_\infty$ et vérifier que le
        nombre de Reynolds est bien petit.
  \item Récapituler~: une équation du mouvement, deux régimes de
        traînée, et les trois nombres (en \unit{m/s}) qui décident si
        une chose tombe comme une pierre, comme une feuille, ou pas du tout.
\end{enumerate}
\end{problem}
