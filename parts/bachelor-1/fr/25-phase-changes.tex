\chapter{Changements d'état d'un corps pur}\label{ch:b1:phase-changes}

L'eau est le seul corps que tout le monde ait vu dans les trois états~;
c'est aussi le fluide de la plupart des centrales électriques du monde, le
frigorigène des premières machines à glace et le moteur du temps qu'il fait.
La glace fond à une température qui ne bouge pas tant que la chaleur
arrive~; une bouilloire bout à \qty{100}{\degreeCelsius} au bord de la mer et
à \qty{70}{\degreeCelsius} sur l'Everest~; une bouteille d'eau très froide,
bien fermée, gèle d'un coup à la moindre secousse. Ce chapitre dresse la
carte des \omterm{def:b1:phase-changes:phase}{phases} d'un \omterm{def:b1:phase-changes:phase}{corps pur} --- les diagrammes $(T, P)$ et $(v, P)$ ---
et donne les trois outils qui l'accompagnent~: la règle des moments (quelle
proportion de chaque \omterm{def:b1:phase-changes:phase}{phase}), la \omterm{def:b1:phase-changes:phase}{chaleur latente} (ce que coûte un changement,
en énergie et en entropie) et la \omterm{thm:b1:phase-changes:clapeyron}{relation de Clausius--Clapeyron} (la pente
des frontières). Il finit là où la carte échoue~: les \omterm{def:b1:phase-changes:metastable}{états métastables}, d'où
viennent le temps qu'il fait et les surprises de l'autocuiseur.

\section{Les phases et le diagramme $(T, P)$}

\begin{definition}[Corps pur, phase, changement d'état]\label{def:b1:phase-changes:phase}
\index{phase}\index{changement d'état}\index{corps pur}\index{chaleur latente}
Un \emph{corps pur} est un corps formé d'une seule espèce chimique. Une
\emph{phase} en est une partie homogène par ses propriétés physiques~:
solide, liquide, gaz (on dit \emph{vapeur} lorsqu'il coexiste avec son
liquide). Un \emph{changement d'état} (fusion/solidification, vaporisation/
liquéfaction, sublimation/condensation) est le passage d'une phase à une
autre~; à pression fixée il se produit à température fixée, avec échange de
chaleur mais sans variation de température~: c'est la \emph{chaleur latente}.
\end{definition}

\begin{proposition}[Diagramme d'équilibre]\label{prop:b1:phase-changes:PTdiagram}
\index{diagramme des phases}\index{point triple}\index{point critique}\index{pression de vapeur saturante}
Dans le plan $(T, P)$, chaque \omterm{def:b1:phase-changes:phase}{phase} occupe un domaine~; deux \omterm{def:b1:phase-changes:phase}{phases} ne
coexistent que le long d'une courbe (sublimation, fusion, vaporisation), les
trois en un seul point, le \emph{\omterm{prop:b1:phase-changes:PTdiagram}{point triple}}. La courbe de vaporisation
$P = P_s(T)$ --- la \emph{\omterm{prop:b1:phase-changes:PTdiagram}{pression de vapeur saturante}} --- s'arrête au
\emph{\omterm{prop:b1:phase-changes:PTdiagram}{point critique}} $(T_c, P_c)$, au-delà duquel liquide et gaz ne se
distinguent plus. Pour l'eau~: \omterm{prop:b1:phase-changes:PTdiagram}{point triple} \qty{273.16}{K}, \qty{611}{Pa}~;
\omterm{prop:b1:phase-changes:PTdiagram}{point critique} \qty{647}{K}, \qty{221}{bar}. La courbe de fusion de l'eau
penche vers la gauche (la glace est moins dense que l'eau)~; pour presque
tout autre corps, elle penche vers la droite.
\end{proposition}
\admitted

\begin{remark}[Ébullition, évaporation et autocuiseur]\label{rem:b1:phase-changes:boiling}
Un liquide s'\emph{évapore} à toute température par sa surface, tant que la
pression partielle de sa vapeur au-dessus de lui reste inférieure à $P_s(T)$~;
il \emph{bout} lorsque $P_s(T)$ atteint la pression ambiante, de sorte que
des bulles de vapeur peuvent grossir en son sein. L'eau bout à
\qty{100}{\degreeCelsius} parce que
$P_s(\qty{100}{\degreeCelsius}) = \qty{1.013}{bar}$~; à \qty{0.33}{bar} sur
l'Everest elle bout vers \qty{70}{\degreeCelsius}, dans un autocuiseur à
\qty{1.8}{bar} vers \qty{117}{\degreeCelsius}. La température d'un liquide qui
bout est imposée par la pression --- ce qui a servi deux siècles durant à
définir l'échelle Celsius.
\end{remark}

\section{Le diagramme $(v, P)$ et la règle des moments}

\begin{proposition}[Isothermes d'Andrews]\label{prop:b1:phase-changes:andrews}
\index{isothermes d'Andrews}\index{courbe de saturation}
Dans le plan de Clapeyron $(v, P)$ du volume massique, une isotherme
$T < T_c$ comporte trois parties~: une branche très raide (le liquide,
presque incompressible), un \emph{palier} horizontal à $P = P_s(T)$ où
liquide et vapeur coexistent, et une branche d'allure hyperbolique (la
vapeur). Les extrémités du palier sont les volumes massiques $v_l(T)$ et
$v_v(T)$ du liquide et de la vapeur saturants~; leur lieu est la
\emph{\omterm{prop:b1:phase-changes:andrews}{courbe de saturation}} (la cloche), dont le sommet est le \omterm{prop:b1:phase-changes:PTdiagram}{point
critique}, où l'isotherme $T_c$ présente une inflexion à tangente
horizontale. Au-dessus de $T_c$, il n'y a plus de palier.
\end{proposition}
\admitted

\begin{theorem}[Règle des moments]\label{thm:b1:phase-changes:lever}
\index{règle des moments}\index{titre massique en vapeur}
Une masse $m$ de volume massique $v$ située sur le palier à $T$ se partage en
une fraction massique de vapeur $x = m_v/m$ (le \emph{titre}) et une fraction
liquide $1 - x$, avec
\[
v = (1 - x)v_l + x\,v_v , \qquad\text{c.-à-d.}\qquad
x = \frac{v - v_l}{v_v - v_l} = \frac{LM}{LV} ,
\]
soit le rapport des segments découpés sur le palier par le point d'état $M$.
La même règle vaut pour toute grandeur extensive par unité de masse ($u$,
$h$, $s$)~: $h = (1 - x)h_l + xh_v$.
\end{theorem}

\begin{proof}
Les volumes s'ajoutent~: $V = m_lv_l + m_vv_v$~; on divise par $m$. Énergie,
\omterm{def:b1:first-law:enthalpy}{enthalpie} et entropie sont extensives elles aussi.
\end{proof}

\begin{example}[Une bouilloire et une chaudière]\label{ex:b1:phase-changes:lever}
Eau à \qty{100}{\degreeCelsius}, \qty{1}{atm}~: $v_l = \qty{1.04e-3}{m^3/kg}$,
$v_v = \qty{1.67}{m^3/kg}$ --- un rapport de $1600$. Un récipient fermé de
\qty{1.0}{L} contenant \qty{0.50}{kg} d'eau à cette température a $v =
\qty{2.0e-3}{m^3/kg}$ et $x = (2.0 - 1.04) \times 10^{-3}/1.67 = 5.8 \times 10^{-4}$~:
\qty{0.29}{g} de vapeur occupent la moitié du récipient. Doubler la masse de
vapeur dans un récipient scellé ne déplace presque pas le point d'état~; la
pression est fixée par la seule température.
\end{example}

\section{Énergétique d'un changement d'état}

\begin{proposition}[Chaleur latente, enthalpie et entropie de transition]\label{prop:b1:phase-changes:latent}
\index{enthalpie de transition}\index{entropie de transition}
À la température $T$ et à la pression $P_s(T)$ de coexistence, le passage de
l'unité de masse de la \omterm{def:b1:phase-changes:phase}{phase} 1 à la \omterm{def:b1:phase-changes:phase}{phase} 2 est réversible, isotherme et
isobare, et
\[
L_{12}(T) = h_2 - h_1 , \qquad s_2 - s_1 = \frac{L_{12}}{T} , \qquad
u_2 - u_1 = L_{12} - P_s(v_2 - v_1) .
\]
$L_{12} > 0$ dans le sens de la \omterm{def:b1:phase-changes:phase}{phase} la moins ordonnée (fusion,
vaporisation, sublimation), et $L_{21} = -L_{12}$. Pour l'eau sous \qty{1}{atm}~:
$L_f = \qty{334}{kJ/kg}$ à \qty{0}{\degreeCelsius}, $L_v = \qty{2257}{kJ/kg}$ à
\qty{100}{\degreeCelsius}.
\end{proposition}

\begin{proof}
À pression constante, la chaleur reçue vaut $\Delta H$
(\cref{prop:b1:first-law:qvqp})~; le passage étant réversible et isotherme,
elle vaut $T\Delta S$ (\cref{prop:b1:second-law-entropy:identity})~; et $U = H - PV$.
\end{proof}

\begin{example}[Où passe la chaleur de vaporisation]\label{ex:b1:phase-changes:whereheat}
Pour l'eau à \qty{100}{\degreeCelsius}~: $P_s(v_v - v_l) = 1.013 \times 10^5 \times
1.67 = \qty{169}{kJ/kg}$, donc $u_v - u_l = 2257 - 169 = \qty{2088}{kJ/kg}$~:
$93\%$ de la \omterm{def:b1:phase-changes:phase}{chaleur latente} sert à arracher les molécules les unes aux
autres, $7\%$ à repousser l'atmosphère. L'entropie de vaporisation,
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, écrase celle de fusion, $334/273 =
\qty{1.22}{kJ/(kg.K)}$~: fondre desserre les molécules, bouillir les libère.
\end{example}

\begin{method}[Calorimétrie avec changements d'état]\label{meth:b1:phase-changes:calorimetry}
Dans une enceinte isolée à pression constante, $\sum\Delta H_i = 0$. On devine
l'état final (quelles \omterm{def:b1:phase-changes:phase}{phases} sont présentes, à quelle température)~; on écrit
le $\Delta H$ de chaque corps comme des chaleurs sensibles $mc\,\Delta T$ plus
des \omterm{def:b1:phase-changes:phase}{chaleurs latentes} $\pm mL$ pour les changements qu'il subit~; on résout~;
puis on \emph{vérifie la supposition}~: un corps qui finit au-dessus de sa
température d'ébullition ou au-dessous de sa température de fusion, ou une
masse négative d'une \omterm{def:b1:phase-changes:phase}{phase}, signale une supposition fausse --- l'état final
est alors un mélange à la température de transition, et l'inconnue devient la
masse transformée (\cref{ex:b1:first-law:ice}).
\end{method}

\begin{example}[La vapeur brûle]\label{ex:b1:phase-changes:steam}
\qty{10}{g} de vapeur à \qty{100}{\degreeCelsius} qui se condensent sur la
peau cèdent $0.010 \times 2257 = \qty{22.6}{kJ}$ avant même d'avoir refroidi
d'un degré --- autant que \qty{10}{g} d'eau bouillante en se refroidissant de
\qty{540}{K}. Voilà pourquoi la vapeur brûle bien plus que l'eau à la même
température, et pourquoi un radiateur à vapeur demande si peu de débit.
\end{example}

\section{La relation de Clausius--Clapeyron}

\begin{theorem}[Clausius--Clapeyron]\label{thm:b1:phase-changes:clapeyron}
\index{relation de Clausius--Clapeyron}
Le long d'une courbe de coexistence des \omterm{def:b1:phase-changes:phase}{phases} 1 et 2,
\[
\frac{\dd P_s}{\dd T} = \frac{L_{12}(T)}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
Faisons décrire à l'unité de masse un \omterm{def:b1:heat-engines:carnotcycle}{cycle de Carnot} infinitésimal à
l'intérieur de la cloche~: transition complète $1 \to 2$ le long du palier à
$T + \dd T$, sous la pression $P_s + \dd P_s$ (chaleur reçue $L_{12}$ au
premier ordre)~; une détente adiabatique infinitésimale jusqu'à $T$~; la
transition inverse le long du palier à $T$, $P_s$~; une compression
adiabatique pour revenir. Le cycle est réversible, ditherme entre
$T + \dd T$ et $T$, donc son \omterm{def:b1:heat-engines:efficiency}{rendement} vaut $\dd T/T$
(\cref{thm:b1:heat-engines:carnot})~; son travail, l'aire du fin rectangle,
vaut $\dd P_s\,(v_2 - v_1)$. D'où $\dd P_s\,(v_2 - v_1)/L_{12} =
\dd T/T$.
\end{proof}

\begin{corollary}[Pression de vapeur saturante d'un liquide]\label{cor:b1:phase-changes:vapourpressure}
Si la vapeur est un \omterm{def:b1:kinetic-theory:model}{gaz parfait} et si $v_v \gg v_l$, de sorte que $v_v - v_l \approx
RT/(MP_s)$, et si $L_v$ varie peu,
\[
\frac{\dd P_s}{P_s} = \frac{ML_v}{R}\,\frac{\dd T}{T^2} , \qquad
\ln\frac{P_s(T)}{P_s(T_0)} = \frac{ML_v}{R}\Bigl(\frac{1}{T_0} - \frac{1}{T}\Bigr) :
\]
la pression de vapeur croît à peu près exponentiellement avec la
température. Pour l'eau vers \qty{100}{\degreeCelsius}, $ML_v/R = 0.018 \times 2.257 \times 10^6/
8.314 = \qty{4890}{K}$ et $\dd P_s/\dd T = 2.257 \times 10^6/(373 \times 1.67) =
\qty{3.6}{kPa/K}$~: chaque \omterm{def:b1:kinetic-theory:temperature}{kelvin} de température d'ébullition en plus coûte
\qty{36}{mbar} de pression en plus.
\end{corollary}

\begin{proof}
On reporte $v_v - v_l \approx RT/(MP_s)$ dans le théorème et l'on sépare les
variables~; on intègre à $L_v$ constante.
\end{proof}

\begin{example}[La glace sous un patin]\label{ex:b1:phase-changes:skate}
Pour l'équilibre glace--eau, $L_f = \qty{334}{kJ/kg}$ et $v_l - v_s = (1.000 - 1.091)

(-9.1 \times 10^{-5})) = \qty{-1.3e7}{Pa/K} = \qty{-130}{bar/K}$. Un patineur de \qty{70}{kg}
sur une lame de \qty{30}{cm} par \qty{3}{mm} ($\qty{9}{cm^2}$) exerce
environ \qty{8}{bar}, ce qui abaisse le point de fusion de \qty{0.06}{K} --- la fusion sous pression n'est donc pas la raison pour laquelle les patins glissent (une mince

frottement échauffe)~; mais la pente négative est bien réelle, et c'est elle
qui fait qu'un fil lesté traverse un bloc de glace qui se ressoude derrière lui.
\end{example}

\section{Les états métastables}

\begin{definition}[Surfusion, surchauffe]\label{def:b1:phase-changes:metastable}
\index{surfusion}\index{surchauffe}\index{état métastable}\index{germination}
Une \omterm{def:b1:phase-changes:phase}{phase} peut subsister au-delà de sa frontière d'équilibre~: de l'eau
liquide refroidie au-dessous de \qty{0}{\degreeCelsius} sans geler (la
\emph{surfusion}), chauffée au-dessus de son point d'ébullition sans bouillir
(la \emph{surchauffe}), une vapeur comprimée au-delà de $P_s$ sans se
condenser (la sursaturation). De tels \emph{états métastables} exigent
l'absence de sites de \emph{germination} --- poussières, rayures, gaz dissous,
cristaux de glace --- sur lesquels la nouvelle \omterm{def:b1:phase-changes:phase}{phase} peut naître~; une
perturbation les fait basculer d'un coup, irréversiblement, vers l'équilibre.
\end{definition}

\begin{example}[La bouteille en surfusion]\label{ex:b1:phase-changes:supercooled}
De l'eau immobile à \qty{-8}{\degreeCelsius} dans une bouteille propre, que
l'on heurte~: des cristaux de glace la parcourent et la température saute à
\qty{0}{\degreeCelsius}. Le processus est adiabatique et rapide, de sorte que
la chaleur sensible $c\,\Delta T$ gèle une fraction $x = c\,\Delta T/L_f = 4.18 \times 8/334 =
0.10$ de l'eau~; l'entropie créée, $c\ln(273/265) - xL_f/273 =
124 - 122 = \qty{+2}{J/(kg.K)}$, est bien positive. Les nuages sont pleins de
gouttelettes en \omterm{def:b1:phase-changes:metastable}{surfusion} jusqu'à \qty{-40}{\degreeCelsius}~; le givrage des
avions et l'ensemencement des nuages exploitent l'un et l'autre ce phénomène.
\end{example}

\begin{remark}[Régulateurs d'ébullition et chambres à bulles]\label{rem:b1:phase-changes:nucleation}
Une bulle de rayon $r$ dans un liquide à la pression $P$ contient de la vapeur
à $P + 2\sigma/r$ (tension superficielle $\sigma$, volume de deuxième année)~;
elle ne peut grossir que si $P_s(T) > P + 2\sigma/r$, de sorte qu'un liquide
propre à son point d'ébullition n'a aucune bulle assez petite pour démarrer et
se \omterm{def:b1:phase-changes:metastable}{surchauffe} de plusieurs \omterm{def:b1:kinetic-theory:temperature}{kelvins} avant d'exploser --- les soubresauts que
les grains régulateurs d'ébullition empêchent en offrant des cavités remplies
de gaz piégé. La chambre à bulles de la physique des particules vivait de la
même instabilité renversée~: un liquide surchauffé bout d'abord le long de la
trace d'une particule ionisante.
\end{remark}

\begin{example}[Humidité et rosée]\label{ex:b1:phase-changes:dew}
\index{humidité relative}\index{point de rosée}
L'air contient de la vapeur d'eau sous une pression partielle $P_v \leq P_s(T)$~;
l'\emph{\omterm{ex:b1:phase-changes:dew}{humidité relative}} est $P_v/P_s(T)$, le \emph{\omterm{ex:b1:phase-changes:dew}{point de rosée}} la
température à laquelle $P_s$ descend jusqu'à $P_v$. Un air à
\qty{25}{\degreeCelsius} ($P_s = \qty{3.17}{kPa}$) à $60\%$ d'humidité a
$P_v = \qty{1.9}{kPa}$, $\rho_v = P_vM/(RT) = \qty{14}{g/m^3}$, et se dépose
en rosée sur toute surface au-dessous de \qty{17}{\degreeCelsius} environ ---
l'herbe à l'aube, la bouteille froide, le miroir de la salle de bains. Soulevé,
ce même air se refroidit adiabatiquement de \qty{9.8}{K/km}
(\cref{pb:b1:phase-changes:1}) et atteint son \omterm{ex:b1:phase-changes:dew}{point de rosée} un kilomètre plus
haut~: c'est là que se trouve la base plate des cumulus.
\end{example}

\end{exercise}

\section{Problème~: l'eau en quatre scènes}

\begin{problem}[{Problème du week-end --- l'autocuiseur, le nuage, le
récipient scellé et la bouteille en surfusion~: un seul corps, quatre usages
de son diagramme des phases}]
\label{pb:b1:phase-changes:1}
Données pour l'eau~: $L_v = \qty{2257}{kJ/kg}$ à \qty{100}{\degreeCelsius} et
environ \qty{2.2}{MJ/kg} à \qty{117}{\degreeCelsius}, \qty{2.45}{MJ/kg} vers
\qty{20}{\degreeCelsius}~; $L_f = \qty{334}{kJ/kg}$~; $c = \qty{4.18}{kJ/(kg.K)}$
(liquide)~; $M = \qty{18}{g/mol}$~; $ML_v/R = \qty{4890}{K}$ vers
\qty{100}{\degreeCelsius}. Pressions de saturation~: \qty{0.61}{kPa}
(\qty{0}{\degreeCelsius}), \qty{1.23}{kPa} (\qty{10}{\degreeCelsius}), \qty{1.70}{kPa} (\qty{15}{\degreeCelsius}),
\qty{2.34}{kPa} (\qty{20}{\degreeCelsius}), \qty{3.17}{kPa} (\qty{25}{\degreeCelsius}), \qty{4.24}{kPa}
(\qty{30}{\degreeCelsius}), \qty{101.3}{kPa} (\qty{100}{\degreeCelsius}).
Volumes massiques saturants~: $v_l = \qty{1.0e-3}{m^3/kg}$ (eau froide),
\qty{1.9e-3}{m^3/kg} à \qty{360}{\degreeCelsius}~; $v_v = \qty{0.0217}{m^3/kg}$
à \qty{300}{\degreeCelsius}, $0.0108$ à \qty{340}{\degreeCelsius}, $0.0088$
à \qty{350}{\degreeCelsius}~; \omterm{prop:b1:phase-changes:PTdiagram}{point critique} \qty{374}{\degreeCelsius},
\qty{221}{bar}, $v_c = \qty{3.1e-3}{m^3/kg}$. Air~: $M = \qty{29}{g/mol}$, $c_p =
\qty{1.0}{kJ/(kg.K)}$, $\gamma = 1.4$~; $g = \qty{9.8}{m/s^2}$~; $R =
\qty{8.314}{J/(mol.K)}$.

\medskip
\noindent\textbf{Partie I --- L'autocuiseur.}
\begin{enumerate}
  \item Pourquoi l'eau d'une casserole ouverte, au niveau de la mer, ne
        peut-elle pas dépasser \qty{100}{\degreeCelsius}, si vive que soit la flamme~?
  \item À partir de la \omterm{thm:b1:phase-changes:clapeyron}{relation de Clausius--Clapeyron}, établir
        $\ln[P_s(T)/P_s(T_0)] = (ML_v/R)(1/T_0 - 1/T)$, en précisant les
        approximations.
  \item Température dans un autocuiseur dont la soupape s'ouvre à
        \qty{1.8}{bar} absolus.
  \item La soupape est un poids posé sur un orifice de \qty{20}{mm^2}~;
        pression atmosphérique \qty{1.0}{bar}~: masse du poids.
  \item \qty{2.0}{L} d'eau sont portés de \qty{20}{\degreeCelsius} à la
        température de fonctionnement par une plaque de \qty{2.0}{kW} (on
        néglige la cocotte)~: chaleur et durée.
  \item Une fois là, la plaque reste à \qty{2.0}{kW}~: masse de vapeur qui
        sort de la soupape par minute. Que doit faire le cuisinier~?
  \item Les réactions de cuisson doublent à peu près de vitesse tous les
        \qty{10}{K}~: par quel facteur l'autocuiseur raccourcit-il une
        recette de \qty{30}{min} au niveau de la mer~? Et combien de temps
        durerait-elle sur l'Everest, où l'eau bout à \qty{70}{\degreeCelsius}~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le nuage.}
\begin{enumerate}[resume]
  \item Définir l'\omterm{ex:b1:phase-changes:dew}{humidité relative}. Pour un air à \qty{25}{\degreeCelsius} et $60\%$~:
        pression partielle de la vapeur, et masse de vapeur par mètre cube
        (\omterm{def:b1:kinetic-theory:model}{gaz parfait}).
  \item \omterm{ex:b1:phase-changes:dew}{Point de rosée} de cet air (interpoler dans le tableau). Expliquer la
        rosée sur l'herbe à l'aube et les gouttes sur une bouteille froide.
  \item Une masse d'air sec s'élève sans échanger de chaleur. À partir de la
        loi adiabatique réversible et de la \omterm{thm:b1:fluid-statics:fundamental}{relation hydrostatique}
        $\dd P/\dd z = -\rho g$, montrer que $\dd T/\dd z = -g/c_p$ et
        l'évaluer.
  \item Le \omterm{ex:b1:phase-changes:dew}{point de rosée} de la masse d'air qui s'élève baisse lui-même
        d'environ \qty{2}{K/km} à mesure que la pression tombe. Altitude à
        laquelle l'air de la question~8 devient saturé --- la base du nuage.
  \item Au-dessus de la base, la vapeur qui se condense libère sa \omterm{def:b1:phase-changes:phase}{chaleur
        latente} dans la masse d'air. De combien la condensation de \qty{1}{g}
        d'eau par kilogramme d'air réchauffe-t-elle cet air~? Qu'est-ce que
        cela fait au taux de refroidissement de la masse d'air qui continue de
        monter, et pourquoi est-ce décisif pour les orages~?
  \item Un cumulus de \qty{1}{km^3} contient \qty{1}{g} d'eau liquide par mètre
        cube~: masse d'eau, et énergie libérée par sa condensation, en joules
        et en \unit{GW h}.
  \item Un miroir de salle de bains à \qty{18}{\degreeCelsius}, un air à
        \qty{25}{\degreeCelsius} et $80\%$~: s'embue-t-il~? À partir de quelle
        humidité cesserait-il de s'embuer~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le récipient scellé.}
Un récipient rigide en acier de \qty{1.0}{L} est rempli d'une masse $m$ d'eau
et scellé sans air, puis chauffé lentement.
\begin{enumerate}[resume]
  \item $m = \qty{0.50}{kg}$ à \qty{20}{\degreeCelsius}~: pression intérieure,
        volume massique, masse de vapeur (prendre le \omterm{def:b1:kinetic-theory:model}{gaz parfait} pour $v_v$),
        fraction du volume occupée par le liquide.
  \item On chauffe~: le niveau du liquide monte-t-il ou descend-il~? Vers
        quelle température le liquide remplit-il le récipient, et qu'arrive-t-il
        ensuite à la pression~? (Prendre $v_l$ à \qty{360}{\degreeCelsius}.)
  \item $m = \qty{0.10}{kg}$~: le niveau du liquide monte-t-il ou descend-il au
        chauffage~? Vers quelle température la dernière goutte s'évapore-t-elle
        (utiliser le tableau des $v_v$)~?
  \item $m = \qty{0.31}{kg}$~: décrire ce qu'un observateur voit au ménisque
        quand la température approche \qty{374}{\degreeCelsius}.
  \item Tracer les trois chemins de chauffage dans le diagramme $(v, P)$ avec
        la cloche de saturation, et dire de quel côté du \omterm{prop:b1:phase-changes:PTdiagram}{point critique} passe
        chacun.
  \item Le même récipient de \qty{1.0}{L} avec \qty{0.50}{kg} d'eau est
        maintenant scellé \emph{avec} l'air qu'il contenait à
        \qty{20}{\degreeCelsius}, \qty{1.0}{bar}, puis chauffé à
        \qty{100}{\degreeCelsius}~: pression totale intérieure (Dalton), et
        force sur un couvercle de \qty{100}{cm^2}.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La bouteille en \omterm{def:b1:phase-changes:metastable}{surfusion}.}
\begin{enumerate}[resume]
  \item Une bouteille d'eau propre et immobile a refroidi jusqu'à
        \qty{-8}{\degreeCelsius} sans geler. Comment nomme-t-on cet état,
        pourquoi est-il possible, et que se passe-t-il si l'on heurte la
        bouteille~? Pourquoi la température se fixe-t-elle alors exactement à
        \qty{0}{\degreeCelsius}~?
  \item Fraction de l'eau qui gèle (le processus est rapide, donc adiabatique).
  \item Entropie créée par kilogramme~; vérifier le signe.
  \item La surprise inverse~: une tasse d'eau chauffée à \qty{105}{\degreeCelsius}
        dans un four à micro-ondes sans bouillir, puis perturbée. Fraction de
        l'eau qui se vaporise brutalement, volume de vapeur produit par litre
        ($v_v = \qty{1.67}{m^3/kg}$), et pourquoi la tasse déborde violemment.
  \item \omterm{thm:b1:heat-engines:lostwork}{Travail perdu}~: le litre en \omterm{def:b1:phase-changes:metastable}{surfusion} aurait pu actionner un moteur
        réversible avec le milieu extérieur à \qty{0}{\degreeCelsius}~; en
        prenant $T_0 = \qty{273}{K}$, le travail jeté en le laissant geler sur
        un choc vaut $T_0S_{\text{créée}}$. L'évaluer par litre et le comparer
        au travail nécessaire pour élever la bouteille d'un mètre. Conclure~:
        que stocke un \omterm{def:b1:phase-changes:metastable}{état métastable}~?
\end{enumerate}
\end{problem}
