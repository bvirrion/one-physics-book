\chapter{Introdução à física quântica}\label{ch:b1:quantum-introduction}

Ilumine com luz ultravioleta um metal limpo e elétrons saem voando na
hora, com uma energia que depende da cor da luz e absolutamente não do
seu brilho; ilumine com luz vermelha e nada sai, por mais forte que seja a
lâmpada. Envie elétrons um a um por duas fendas e eles chegam como pontos,
cada um num lugar --- e os pontos se acumulam, aos milhares, nas franjas de
uma onda. Nenhum dos dois fatos cabe na física dos vinte e nove capítulos
anteriores. Este último capítulo introduz as ideias em que eles cabem: a
luz vem em quanta, os fótons; a matéria tem um \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda}; o que
se propaga é uma \omterm{def:b1:wave-propagation:signal}{amplitude} cujo quadrado é uma probabilidade; posição e
quantidade de movimento não podem ser ambas exatas; e uma partícula
confinada só pode ter certas energias --- e é por isso que os átomos têm
tamanho, os espectros têm linhas e um grão de semicondutor de alguns
nanômetros brilha numa cor fixada pelo seu diâmetro.

\section{O fóton}

\begin{theorem}[Relações de Planck--Einstein]\label{thm:b1:quantum-introduction:photon}
\index{fóton}\index{constante de Planck}
A luz de frequência $\nu$ e \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} $\lambda$ troca energia com
a matéria apenas em quanta, os \emph{fótons}, de energia e quantidade de movimento
\[
E = h\nu = \frac{hc}{\lambda} , \qquad p = \frac{h}{\lambda} = \frac{E}{c} ,
\]
com a \emph{\omterm{thm:b1:quantum-introduction:photon}{constante de Planck}} $h = \qty{6.626e-34}{J.s}$ ($\hbar = h/2\pi =
\qty{1.055e-34}{J.s}$). Útil: $hc = \qty{1240}{eV.nm}$ --- um fóton de
\qty{620}{nm} carrega \qty{2}{eV}.
\end{theorem}
\admitted

\begin{proposition}[O efeito fotoelétrico]\label{prop:b1:quantum-introduction:photoelectric}
\index{efeito fotoelétrico}\index{função trabalho}\index{potencial de corte}
A luz que incide num metal de \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{função trabalho}} $W$ (a energia
que liga os seus elétrons menos ligados, alguns eV) arranca elétrons de
\omterm{thm:b1:work-and-energy:ke}{energia cinética} máxima
\[
E_{k,\max} = h\nu - W ,
\]
e portanto só acima do \emph{limiar} $\nu_0 = W/h$, instantaneamente, com uma
energia independente da intensidade; a intensidade fixa o
\emph{número} de elétrons por segundo. O \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{potencial de corte}}
$V_s$ que anula exatamente a corrente mede $E_{k,\max} = eV_s$, e
o gráfico de $V_s$ contra $\nu$ é uma reta de inclinação $h/e$ --- foi assim que
Millikan mediu $h$.
\end{proposition}

\begin{proof}
Cada fóton é absorvido por um elétron, que fica, no melhor dos casos, com
$h\nu - W$; uma onda entregaria energia continuamente a cada elétron, a
energia cresceria com a intensidade, e uma luz fraca levaria horas para
acumular $W$ num átomo (\cref{pb:b1:quantum-introduction:1}) ---
e nada disso se observa. A interpretação de Einstein (1905) fica admitida
como lei.
\end{proof}

\begin{example}[Ordens de grandeza]\label{ex:b1:quantum-introduction:orders}
Um fóton visível: \qty{2}{eV}; um fóton de raios X de \qty{0.1}{nm}: \qty{12}{keV};
um fóton de rádio a \qty{100}{MHz}: \qty{4e-7}{eV}. Um laser vermelho de \qty{1}{mW}
emite $10^{-3}/(3.1 \times 10^{-19}) = 3 \times 10^{15}$ fótons por segundo;
uma antena de rádio que capta \qty{1}{pW} a \qty{100}{MHz} recebe $10^{13}$
por segundo --- tantos e tão pequenos que a descrição ondulatória é exata
para todos os fins; o \omterm{def:b1:optical-instruments:eye}{olho}, ao contrário, consegue registrar um clarão de
uma centena de fótons. A natureza quântica da luz aparece quando os fótons são
poucos ou energéticos: células fotoelétricas, detectores de raios X, o grão de
uma fotografia fraca.
\end{example}

\begin{remark}[Espalhamento Compton]\label{rem:b1:quantum-introduction:compton}
\index{espalhamento Compton}
Um fóton que ricocheteia num elétron livre transfere quantidade de movimento como uma
bola de bilhar: o seu \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} cresce de $\Delta\lambda = \frac{h}{m_ec}(1 -
\cos\theta)$, com $h/m_ec = \qty{2.43}{pm}$ (admitido --- uma colisão
relativista, do segundo ano). Invisível para a luz ($\Delta\lambda/\lambda \sim 10^{-5}$),
é um efeito de $20\%$ para os raios X, e foi a experiência que convenceu
os céticos (1923) de que o fóton carrega a quantidade de movimento $h/\lambda$.
\end{remark}

\section{Ondas de matéria}

\begin{theorem}[de Broglie]\label{thm:b1:quantum-introduction:debroglie}
\index{comprimento de onda de De Broglie}\index{onda de matéria}
Uma partícula de quantidade de movimento $p$ se propaga como uma onda de \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda}
\[
\lambda = \frac{h}{p} ,
\]
e apresenta, em consequência, difração e interferência: elétrons num
cristal (Davisson e Germer, 1927), através de duas fendas (Jönsson, 1961),
nêutrons, átomos e moléculas de sessenta átomos de carbono. Para um elétron
acelerado por $U$, $\lambda = h/\sqrt{2m_eeU} = \qty{1.23}{nm}/\sqrt{U/\unit{V}}$.
\end{theorem}
\admitted

\begin{example}[Por que nunca víamos isso]\label{ex:b1:quantum-introduction:wavelengths}
Um elétron de \qty{100}{eV}: $\lambda = \qty{0.12}{nm}$, o espaçamento dos
átomos num cristal --- daí a difração e o \omterm{def:b1:optical-instruments:microscope}{microscópio}
eletrônico. Um nêutron térmico (\qty{0.025}{eV}): \qty{0.18}{nm}. Uma bola
de \qty{1}{g} a \qty{1}{m/s}: $\lambda = \qty{7e-31}{m}$, $10^{20}$ vezes
menor que um núcleo: fenda nenhuma, cristal nenhum poderia difratá-la. A
natureza ondulatória da matéria é universal; é o seu \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} que a torna
visível só para o que é leve e lento.
\end{example}

\begin{proposition}[Partículas isoladas constroem as franjas]\label{prop:b1:quantum-introduction:buildup}
\index{dualidade onda--partícula}
Elétrons enviados um de cada vez através de duas fendas (Tonomura, 1989)
fazem cada um um único ponto na tela, num lugar que não se pode prever;
os pontos se acumulam na figura de interferência de duas fendas de uma onda
de \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} $h/p$. Fechar uma das fendas apaga as franjas --- a
figura não é feita por elétrons que interferem uns com os outros, mas por
a onda de cada elétron passar pelas duas fendas. Detectar por qual fenda
uma partícula passou também apaga as franjas.
\end{proposition}
\admitted

\section{Amplitudes, probabilidades e o princípio da incerteza}

\begin{definition}[Função de onda]\label{def:b1:quantum-introduction:wavefunction}
\index{função de onda}\index{densidade de probabilidade}
O estado de uma partícula é descrito por uma \emph{função de onda}
$\psi(x, t)$ (complexa), tal que $|\psi(x, t)|^2\,\dd x$ é a
probabilidade de encontrar a partícula entre $x$ e $x + \dd x$ quando se
olha: $|\psi|^2$ é uma \emph{densidade de probabilidade}, e $\int|\psi|^2\dd x =
1$. As \omterm{def:b1:wave-propagation:signal}{amplitudes} se somam (superposição): se dois caminhos levam ao mesmo
ponto, $\psi = \psi_1 + \psi_2$ e $|\psi|^2 = |\psi_1|^2 + |\psi_2|^2 +
2\,\mathrm{Re}(\psi_1^\ast\psi_2)$ --- o último termo é a interferência.
Uma medida dá um dos resultados possíveis, ao acaso, com as
probabilidades que as \omterm{def:b1:wave-propagation:signal}{amplitudes} prescrevem, e deixa a partícula no
estado encontrado.
\end{definition}

\begin{remark}[O que é que ondula]\label{rem:b1:quantum-introduction:whatwaves}
$\psi$ não é um campo físico como $\vect E$: ele não é medido, só
o seu quadrado o é, e descreve o melhor conhecimento possível de onde
a partícula será encontrada. Um único elétron não está espalhado pela
tela; ele é detectado inteiro, num ponto; o que está espalhado é a
probabilidade. É esse o conteúdo da teoria, e ela nunca falhou
num teste.
\end{remark}

\begin{theorem}[Princípio da incerteza de Heisenberg]\label{thm:b1:quantum-introduction:heisenberg}
\index{princípio da incerteza}
A posição e a quantidade de movimento de uma partícula ao longo do mesmo eixo não podem
ser ambas exatamente definidas: os seus desvios padrão, em qualquer estado, obedecem a
\[
\Delta x\,\Delta p_x \geq \frac{\hbar}{2} .
\]
Uma partícula confinada a uma região de tamanho $\ell$ tem, portanto, uma dispersão de
quantidade de movimento de pelo menos $\hbar/2\ell$, e daí uma \omterm{thm:b1:work-and-energy:ke}{energia cinética} da ordem de
$\hbar^2/2m\ell^2$ que resfriamento nenhum remove: confinar custa energia.
\end{theorem}
\admitted

\begin{example}[O tamanho do átomo de hidrogênio]\label{ex:b1:quantum-introduction:hsize}
\index{raio de Bohr}
Um elétron a menos de $r$ do próton tem $p \gtrsim \hbar/r$ e energia
$E(r) \approx \dfrac{\hbar^2}{2m_er^2} - \dfrac{e^2}{4\pi\varepsilon_0r}$: o primeiro
termo explode se $r$ encolhe, e o segundo, se ele cresce. $\dd E/\dd r = 0$ em
\[
a_0 = \frac{4\pi\varepsilon_0\hbar^2}{m_ee^2} = \qty{53}{pm} , \qquad
E(a_0) = -\frac{m_ee^4}{2(4\pi\varepsilon_0)^2\hbar^2} = \qty{-13.6}{eV} :
\]
o \emph{\omterm{ex:b1:quantum-introduction:hsize}{raio de Bohr}} e a energia de ionização do hidrogênio, até o
último algarismo. A física clássica não conseguia explicar por que o elétron não
espirala até o núcleo (ele irradia) nem por que todos os átomos de um
elemento são idênticos; o princípio da incerteza responde às duas coisas numa
linha: o átomo é tão pequeno quanto pode ser sem que a \omterm{thm:b1:work-and-energy:ke}{energia cinética}
supere a de ligação.
\end{example}

\section{Quantização da energia}

\begin{proposition}[Partícula numa caixa]\label{prop:b1:quantum-introduction:box}
\index{poço infinito}\index{quantização}\index{estado fundamental}
Uma partícula de massa $m$ confinada entre duas paredes impenetráveis à
distância $L$ tem, como a corda de Melde
(\cref{ch:b1:wave-propagation}), apenas as \omterm{thm:b1:wave-propagation:standing}{ondas estacionárias} $\lambda_n = 2L/n$,
e portanto apenas as quantidades de movimento $p_n = nh/2L$ e as energias
\[
E_n = \frac{n^2h^2}{8mL^2} , \qquad n = 1, 2, 3, \ldots
\]
com as \omterm{def:b1:quantum-introduction:wavefunction}{funções de onda} $\psi_n(x) \propto \sin(n\pi x/L)$. A energia é
\emph{quantizada}; a mais baixa, o \emph{\omterm{prop:b1:quantum-introduction:box}{estado fundamental}} $E_1 > 0$, é a
energia de confinamento do princípio da incerteza tornada exata; a
partícula salta entre níveis absorvendo ou emitindo um fóton de
energia $h\nu = E_p - E_n$.
\end{proposition}

\begin{proof}
$\psi$ tem de se anular nas paredes (a partícula não pode estar lá), de modo que a
onda encaixa um número inteiro de meios \omterm{prop:b1:wave-propagation:sinusoidal}{comprimentos de onda} em $L$ --- é a
condição da corda; de Broglie dá $p$, e $E = p^2/2m$. Que $\psi$ obedeça
a uma equação de onda com essas condições de contorno fica admitido (a
equação de Schrödinger, do segundo ano).
\end{proof}

\begin{proposition}[O átomo de hidrogênio]\label{prop:b1:quantum-introduction:hydrogen}
\index{átomo de hidrogênio}\index{níveis de energia}\index{linhas espectrais}
O elétron do \omterm{prop:b1:quantum-introduction:hydrogen}{átomo de hidrogênio} tem as energias discretas
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2} , \qquad n = 1, 2, 3, \ldots ,
\]
sendo $n = 1$ o \omterm{prop:b1:quantum-introduction:box}{estado fundamental} e $E \geq 0$ o átomo ionizado. O seu
espectro é feito de linhas em $h\nu = E_p - E_n$: a série de Lyman até
$n = 1$ (ultravioleta, \qty{122}{nm} para $2 \to 1$), a série de Balmer
até $n = 2$ (visível: \qty{656}{nm}, \qty{486}{nm}, \qty{434}{nm}, \ldots),
e assim por diante. O modelo de Bohr (1913) obtém essas energias quantizando o
\omterm{def:b1:angular-momentum:L}{momento angular} de uma \omterm{prop:b1:central-forces:effective}{órbita circular}, $m_evr = n\hbar$
(\cref{exo:b1:quantum-introduction:9}); a teoria completa substitui as órbitas por
\omterm{def:b1:quantum-introduction:wavefunction}{funções de onda}, mas mantém as energias.
\end{proposition}
\admitted

\begin{example}[Cores a partir do tamanho: o ponto quântico]\label{ex:b1:quantum-introduction:dot}
\index{ponto quântico}
Um nanocristal de semicondutor de alguns nanômetros confina os seus
elétrons como uma caixa: quanto menor o cristal, maior a
energia de confinamento somada à banda proibida do semicondutor, e mais azul a
luz que ele emite. Pontos de seleneto de cádmio de raio \qty{2}{nm} brilham em azul,
os de \qty{3}{nm} em amarelo e os de \qty{4}{nm} em vermelho --- um só material, com as cores escolhidas com
uma régua (\cref{pb:b1:quantum-introduction:1}). As telas de televisão os usam.
\end{example}

\begin{remark}[O que o resto da teoria acrescenta]\label{rem:b1:quantum-introduction:rest}
\index{efeito túnel}\index{laser}
A equação de Schrödinger torna exatos a caixa, o átomo e a molécula;
ela prevê que uma partícula pode atravessar uma barreira que não tem
energia para escalar (o \emph{\omterm{rem:b1:quantum-introduction:rest}{efeito túnel}}, por trás da radioatividade e do
\omterm{def:b1:optical-instruments:microscope}{microscópio} de varredura por tunelamento); que as partículas idênticas são ou
sociáveis (bósons --- daí o \emph{laser}, em que os fótons se amontoam num só
modo) ou exclusivas (férmions --- daí a tabela periódica e a
rigidez da matéria); e que o spin do elétron é um \omterm{def:b1:magnetostatics:moment}{momento magnético}
sem contrapartida clássica. Tudo isso é do segundo ano em diante;
tudo isso repousa sobre as poucas ideias deste capítulo.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:1}
Energia de um fóton de: rádio a \qty{100}{MHz}; luz a \qty{500}{nm};
raios X a \qty{0.10}{nm}; raios gama de \qty{1}{MeV} (dê o seu
\omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda}). Número de fótons por segundo de um laser de \qty{1.0}{mW} a
\qty{633}{nm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:2}
Sódio, $W = \qty{2.3}{eV}$: \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} limiar; \omterm{thm:b1:work-and-energy:ke}{energia cinética} máxima
e \omterm{prop:b1:quantum-introduction:photoelectric}{potencial de corte} para luz de \qty{400}{nm}; o que muda se a
intensidade for dobrada? e com luz de \qty{600}{nm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:3}
\omterm{thm:b1:quantum-introduction:debroglie}{Comprimento de onda de De Broglie} de um elétron de \qty{100}{eV} e de \qty{1.0}{keV};
de um nêutron de \qty{0.025}{eV}; de uma bola de \qty{1}{g} a \qty{1}{m/s}.
Quais deles podem ser difratados por um cristal (espaçamento de \qty{0.2}{nm})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:4}
Um elétron numa caixa de \qty{1.0}{nm}: $E_1$, $E_2$, \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} do
fóton emitido em $2 \to 1$. Um próton numa caixa de \qty{5}{fm}
(um núcleo): $E_1$ em \unit{MeV}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:5}
Potenciais de corte medidos para um metal: \qty{0.40}{V} a
\qty{6.0e14}{Hz}, \qty{1.23}{V} a \qty{8.0e14}{Hz}, \qty{2.05}{V} a
\qty{1.0e15}{Hz}. Deduza $h$ (com $e$ conhecido) e a \omterm{prop:b1:quantum-introduction:photoelectric}{função trabalho};
\omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} limiar; que metal poderia ser (césio \qty{2.1}{eV},
potássio \qty{2.3}{eV}, zinco \qty{4.3}{eV})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:6}
Compton: um fóton de raios X de \qty{10}{pm} é espalhado a \qty{90}{\degree} por um
elétron. Novo \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda}; energias do fóton antes e depois; energia
cedida ao elétron. Deslocamento relativo para luz de \qty{500}{nm} --- por que
Compton precisou de raios X.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:7}
\emph{A experiência de Jönsson.} Elétrons de \qty{50}{keV} (use $p =
\sqrt{2m_eE}$), duas fendas a \qty{2.0}{\micro m} uma da outra, tela a
\qty{35}{cm}. \omterm{prop:b1:wave-propagation:sinusoidal}{Comprimento de onda}; espaçamento das franjas; por que foi preciso uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente}
eletrônica de aumento. Na versão de Tonomura, um elétron de cada vez atravessa
o aparelho: o que a tela mostra depois de 10 e depois de $10^5$ elétrons,
e o que isso ensina?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:8}
\omterm{thm:b1:work-and-energy:ke}{Energia cinética} mínima, pelo princípio da incerteza, de: um
elétron confinado a \qty{0.1}{nm} (um átomo); um elétron confinado a
\qty{5}{fm} (um núcleo --- use $E \approx pc$ se o resultado for
relativista, e conclua se os núcleos podem conter elétrons); um
grão de poeira de \qty{1}{\micro g} localizado em \qty{1}{\micro m} (dê $\Delta v$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:9}
\emph{O modelo de Bohr.} Um elétron numa \omterm{prop:b1:central-forces:effective}{órbita circular} de raio $r$ em torno de um
próton: (a) velocidade e energia em função de $r$ (dinâmica coulombiana,
\cref{ch:b1:central-forces}); (b) imponha $m_evr = n\hbar$ e deduza
$r_n = n^2a_0$, $E_n = E_1/n^2$ com $a_0$ e $E_1$ como no
\cref{ex:b1:quantum-introduction:hsize}; (c) $v_1$ e $v_1/c$; (d)
\omterm{prop:b1:wave-propagation:sinusoidal}{comprimentos de onda} da Lyman $\alpha$ ($2 \to 1$) e da H$\alpha$ ($3 \to 2$);
(e) mostre que a órbita $n$ contém exatamente $n$ \omterm{thm:b1:quantum-introduction:debroglie}{comprimentos de onda de De Broglie}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:10}
\emph{Sistemas hidrogenoides.} (a) Para um núcleo de carga $Ze$, mostre que
$E_n = -Z^2 \times \qty{13.6}{eV}/n^2$ e $r_n = n^2a_0/Z$: energia de ionização
do He$^+$ e do último elétron do urânio ($Z = 92$; comente).
(b) Hidrogênio muônico: o múon tem $207$ vezes a massa do elétron:
raio e energia fundamental; \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} da sua linha Lyman $\alpha$; por que ele
é usado para medir o tamanho do próton (raio \qty{0.84}{fm}).
(c) Positrônio (elétron e pósitron, de massas iguais): mostre que a
\omterm{prop:b1:systems-of-points:twobody}{massa reduzida} divide as energias por dois: a sua Lyman $\alpha$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:11}
\emph{\omterm{ex:b1:quantum-introduction:dot}{Pontos quânticos}.} Num nanocristal esférico de raio $R$ a
energia de confinamento de uma partícula de massa efetiva $m^\ast$ vale
$h^2/8m^\ast R^2$ (admitido). CdSe: banda proibida $E_g = \qty{1.74}{eV}$, $m_e^\ast =
0.13\,m_e$, $m_h^\ast = 0.45\,m_e$ (a ``lacuna'' deixada pelo elétron excitado
também fica confinada). (a) Energia do fóton emitido $E_g + E_e + E_h$
e \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} para $R = \qty{2.0}{nm}$, \qty{3.0}{nm} e \qty{4.0}{nm}; cores.
(b) Raio para emissão a \qty{520}{nm} (verde). (c) Por que o CdSe maciço
emite só a \qty{713}{nm}? (d) Uma dispersão de \qty{10}{\percent} nos raios de um
lote: dispersão em \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} a \qty{3}{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:12}
\emph{Energia de ponto zero.} (a) Para um \omterm{def:b1:oscillators-resonance:harmonic}{oscilador harmônico} $E = p^2/2m +
\tfrac12m\omega^2x^2$, tome $\Delta x\,\Delta p = \hbar/2$ e minimize $E \approx
\Delta p^2/2m + \tfrac12m\omega^2\Delta x^2$ em $\Delta x$: mostre que $E_{\min} = \hbar\omega/2$
(a energia fundamental exata). (b) A molécula de H$_2$ vibra a $\omega =
\qty{8.3e14}{rad/s}$: energia de ponto zero em eV, e a menor
temperatura em que a sua vibração é ``térmica'' ($k_BT \sim \hbar\omega$).
(c) O hélio líquido nunca congela à pressão atmosférica: estime a
energia de ponto zero de um átomo de hélio localizado em $\Delta x = \qty{0.05}{nm}$
(uma fração da distância interatômica) num cristal, e compare com a
sua energia de ligação no sólido, cerca de \qty{1}{meV}. (d) Compare a fórmula da caixa $E_1 = h^2/8mL^2$ com o
limite da incerteza para $\Delta x = L/2$: qual é maior, e por que tem de ser?
\end{exercise}

\section{Problema: o espectro do hidrogênio e um ponto quântico}

\begin{problem}[{Problema de fim de semana --- medir $h$ com luz e um
voltímetro, construir o átomo de hidrogênio a partir de três constantes e
escolher a cor de um cristal pelo seu tamanho}]
\label{pb:b1:quantum-introduction:1}
Constantes: $h = \qty{6.626e-34}{J.s}$, $\hbar = \qty{1.055e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$, $e = \qty{1.602e-19}{C}$, $m_e = \qty{9.11e-31}{kg}$,
$1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$, $hc = \qty{1240}{eV.nm}$.

\medskip
\noindent\textbf{Parte I --- Medindo a \omterm{thm:b1:quantum-introduction:photon}{constante de Planck}.}
Um fotocátodo de potássio é iluminado com luz monocromática; mede-se o
\omterm{prop:b1:quantum-introduction:photoelectric}{potencial de corte}: \qty{0.30}{V} a \qty{6.0e14}{Hz}, \qty{0.69}{V} a
\qty{7.0e14}{Hz}, \qty{1.13}{V} a \qty{8.0e14}{Hz}, \qty{1.50}{V} a
\qty{9.0e14}{Hz} e \qty{2.36}{V} a \qty{1.1e15}{Hz}.
\begin{enumerate}
  \item Descreva a célula e explique por que uma \omterm{def:b1:dc-circuits:voltage}{tensão} inversa consegue parar
        a corrente, e o que $eV_s$ mede.
  \item Escreva a relação de Einstein e diga o que dão a inclinação e o
        intercepto de $V_s(\nu)$.
  \item A partir dos dados (um ajuste linear, ou os dois pontos extremos),
        deduza $h$; compare com o valor aceito.
  \item \omterm{prop:b1:quantum-introduction:photoelectric}{Função trabalho} do potássio em eV; frequência e \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda}
        limiares; uma caneta laser vermelha conseguiria arrancar elétrons?
  \item A intensidade da lâmpada é dobrada: o que acontece com $V_s$ e com a
        corrente? Por que isso é incompatível com uma imagem puramente ondulatória?
  \item Estimativa clássica: uma luz de \qty{1}{\micro W/m^2} incide no
        metal; um átomo oferece uma área de cerca de \qty{1e-20}{m^2}: quanto tempo
        levaria para acumular \qty{2.2}{eV} num átomo? O que se observa
        em vez disso?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O \omterm{prop:b1:quantum-introduction:hydrogen}{átomo de hidrogênio} de Bohr.}
\begin{enumerate}[resume]
  \item Um elétron numa \omterm{prop:b1:central-forces:effective}{órbita circular} de raio $r$ em torno de um próton
        fixo: velocidade $v(r)$ e \omterm{thm:b1:work-and-energy:em}{energia mecânica} $E(r)$.
  \item Condição de Bohr $m_evr = n\hbar$: mostre que $r_n = n^2a_0$ e dê
        $a_0 = 4\pi\varepsilon_0\hbar^2/m_ee^2$ numericamente.
  \item $E_n = E_1/n^2$ com $E_1 = -m_ee^4/[2(4\pi\varepsilon_0)^2\hbar^2]$;
        valor numérico em eV.
  \item Velocidade na primeira órbita e a razão $v_1/c$ (a
        constante de estrutura fina $\alpha \approx 1/137$).
  \item \omterm{prop:b1:wave-propagation:sinusoidal}{Comprimentos de onda} das linhas $2 \to 1$ (Lyman $\alpha$), $3 \to 2$ e
        $4 \to 2$ (Balmer H$\alpha$ e H$\beta$): quais são visíveis?
  \item Mostre que $1/\lambda = R_H(1/n^2 - 1/p^2)$ e calcule a constante de
        Rydberg $R_H$; o menor \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} de Balmer.
  \item Energia de ionização do hidrogênio a partir do \omterm{prop:b1:quantum-introduction:box}{estado fundamental}; \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda}
        do fóton que apenas o ioniza.
  \item De Broglie: mostre que a circunferência da órbita $n$ vale $n$
        \omterm{prop:b1:wave-propagation:sinusoidal}{comprimentos de onda} --- a condição de Bohr é uma condição de \omterm{thm:b1:wave-propagation:standing}{onda estacionária}.
  \item Duas coisas que o modelo erra, uma que ele acerta exatamente
        e o que substitui a órbita na teoria completa.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O \omterm{ex:b1:quantum-introduction:dot}{ponto quântico}.}
Um nanocristal de CdSe de raio $R$ confina um elétron excitado
(massa efetiva $m_e^\ast = 0.13\,m_e$) e a lacuna que ele deixa
($m_h^\ast = 0.45\,m_e$); a energia de confinamento de cada um numa esfera de
raio $R$ vale $h^2/8m^\ast R^2$ (admitido); a banda proibida do CdSe maciço vale
$E_g = \qty{1.74}{eV}$.
\begin{enumerate}[resume]
  \item Deduza as energias $E_n = n^2h^2/8mL^2$ de uma partícula numa
        caixa unidimensional a partir da condição de \omterm{thm:b1:wave-propagation:standing}{onda estacionária}, e
        comente a analogia com a fórmula da esfera.
  \item Energias de confinamento do elétron e da lacuna para
        $R = \qty{2.0}{nm}$.
