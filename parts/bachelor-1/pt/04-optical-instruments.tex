\chapter{Instrumentos ópticos e o olho}\label{ch:b1:optical-instruments}

Da última fileira de uma sala de concertos, um binóculo traz o rosto do
cantor à distância de um braço; um \omterm{def:b1:optical-instruments:microscope}{microscópio} transforma uma gota de água
de lagoa num zoológico; um celular da espessura de um dedo fotografa a
Via Láctea. Cada um desses instrumentos é um punhado de \omterm{def:b1:mirrors-thin-lenses:lens}{lentes delgadas}
dispostas com um único fim: apresentar ao \omterm{def:b1:optical-instruments:eye}{olho}, ou a um sensor, uma
imagem maior, mais brilhante ou mais nítida do que o \omterm{def:b1:optical-instruments:eye}{olho} formaria sozinho. Por isso o
capítulo começa pelo próprio \omterm{def:b1:optical-instruments:eye}{olho} --- o que ele pode e o que não pode fazer --- e
depois constrói a \omterm{def:b1:optical-instruments:G}{lupa}, o \omterm{def:b1:optical-instruments:microscope}{microscópio}, a luneta e a \omterm{def:b1:optical-instruments:camera}{câmera}
a partir da relação de conjugação do capítulo anterior.

\section{O olho}

\begin{definition}[Olho reduzido]\label{def:b1:optical-instruments:eye}
\index{olho}\index{olho reduzido}\index{retina}\index{acomodação}
O \emph{olho reduzido} modela a córnea e o cristalino como uma
única \omterm{def:b1:mirrors-thin-lenses:lens}{lente delgada} convergente de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} \emph{variável}, e a
\emph{retina} como um anteparo a uma distância fixa $d \approx \qty{17}{mm}$
atrás dela. Os músculos mudam a curvatura do cristalino para manter na retina a imagem do
objeto observado: é a \emph{acomodação}. A íris,
diante do cristalino, é um diafragma variável de diâmetro \qty{2}{} a
\qty{8}{mm}, a \emph{pupila}.
\end{definition}

\begin{definition}[Ponto remoto, ponto próximo]\label{def:b1:optical-instruments:punctum}
\index{ponto remoto}\index{ponto próximo}\index{amplitude de acomodação}
O \emph{ponto remoto} (punctum remotum) é o ponto-objeto visto
nitidamente sem \omterm{def:b1:optical-instruments:eye}{acomodação}; o \emph{ponto próximo} (punctum proximum),
o ponto mais próximo visto nitidamente com \omterm{def:b1:optical-instruments:eye}{acomodação} máxima. Para um \omterm{def:b1:optical-instruments:eye}{olho} normal
(emétrope) o ponto remoto está no infinito e o ponto próximo em
$d_m \approx \qty{25}{cm}$, a \emph{distância mínima de visão
distinta} convencional. A diferença de \omterm{def:b1:mirrors-thin-lenses:lens}{vergências} necessária para os dois é a
\emph{amplitude de \omterm{def:b1:optical-instruments:eye}{acomodação}}: cerca de \qty{4}{\delta} para um
adulto jovem, decrescente com a idade.
\end{definition}

\begin{example}[A vergência do olho]\label{ex:b1:optical-instruments:vergence}
Objeto distante: imagem a $d = \qty{17}{mm}$ significa $f' = \qty{17}{mm}$,
$V = 1/0.017 = \qty{59}{\delta}$. Objeto a \qty{25}{cm}:
$V = 1/0.017 + 1/0.25 = 59 + 4 = \qty{63}{\delta}$. Quatro dioptrias de
\omterm{def:b1:optical-instruments:eye}{acomodação} em sessenta: o cristalino acrescenta apenas alguns por cento
ao trabalho da córnea, mas esses poucos por cento são tudo para a leitura.
\end{example}

\begin{proposition}[Defeitos e correções]\label{prop:b1:optical-instruments:defects}
\index{miopia}\index{hipermetropia}\index{presbiopia}
Um \omterm{def:b1:optical-instruments:eye}{olho} \emph{míope} é convergente demais: o seu \omterm{def:b1:optical-instruments:punctum}{ponto remoto} está a uma
distância finita $D_r$; uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente divergente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} $f' = -D_r$ (usada
junto ao \omterm{def:b1:optical-instruments:eye}{olho}) conjuga o infinito no \omterm{def:b1:optical-instruments:punctum}{ponto remoto} e restaura a visão de
longe. Um \omterm{def:b1:optical-instruments:eye}{olho} \emph{hipermetrope} não é convergente o bastante: o seu \omterm{def:b1:optical-instruments:punctum}{ponto remoto}
é virtual (atrás do \omterm{def:b1:optical-instruments:eye}{olho}) e o seu \omterm{def:b1:optical-instruments:punctum}{ponto próximo}, longe demais; uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente
convergente} o corrige. A \emph{presbiopia} é a perda de \omterm{def:b1:optical-instruments:eye}{acomodação} com
a idade: o \omterm{def:b1:optical-instruments:punctum}{ponto próximo} se afasta e a leitura exige \omterm{def:b1:mirrors-thin-lenses:lens}{lentes convergentes},
qualquer que seja o \omterm{def:b1:optical-instruments:punctum}{ponto remoto}.
\end{proposition}

\begin{proof}
Uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente} junto ao \omterm{def:b1:optical-instruments:eye}{olho} e o \omterm{def:b1:optical-instruments:eye}{olho} formam um sistema cujo objeto deve ser
o que o \omterm{def:b1:optical-instruments:eye}{olho} consegue ver: a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} precisa transformar um objeto no infinito numa
\omterm{def:b1:mirrors-thin-lenses:image}{imagem virtual} no \omterm{def:b1:optical-instruments:punctum}{ponto remoto}, $\overline{OA'} = -D_r$ para
$\overline{OA} = -\infty$, de modo que $f' = \overline{OA'} = -D_r$. Os outros
casos seguem do mesmo argumento com os pontos pertinentes.
\end{proof}

\begin{definition}[Tamanho angular e resolução]\label{def:b1:optical-instruments:angular}
\index{tamanho angular}\index{resolução angular}
O \emph{tamanho angular} de um objeto de altura $h$ à distância $D$ é
$\theta \approx h/D$ (ângulos pequenos). O \omterm{def:b1:optical-instruments:eye}{olho} distingue dois pontos
apenas se a sua separação angular exceder a sua \emph{resolução angular},
cerca de $\epsilon \approx \qty{3e-4}{rad}$ (um minuto de arco): o espaçamento
das células cones e a difração na pupila fixam ambos esse limite.
\end{definition}

\begin{remark}[Para que serve um instrumento]\label{rem:b1:optical-instruments:purpose}
Levado ao \omterm{def:b1:optical-instruments:punctum}{ponto próximo}, um detalhe de \qty{1}{mm} subtende $1/250 =
\qty{4e-3}{rad}$: treze vezes o limite de resolução; um detalhe de \qty{0.05}{mm}
subtende $\qty{2e-4}{rad}$ e é invisível. Os instrumentos que
olham objetos próximos (\omterm{def:b1:optical-instruments:G}{lupa}, \omterm{def:b1:optical-instruments:microscope}{microscópio}) aumentam o \omterm{def:b1:optical-instruments:angular}{tamanho angular}
além do que aproximar o objeto conseguiria; os instrumentos para objetos
distantes (luneta) aumentam o \omterm{def:b1:optical-instruments:angular}{tamanho angular} do que não se pode
aproximar; a \omterm{def:b1:optical-instruments:camera}{câmera} substitui a \omterm{def:b1:optical-instruments:eye}{retina} por um sensor capaz de integrar
luz durante segundos e de ser ampliado depois.
\end{remark}

\section{A lupa}

\begin{definition}[Aumento angular]\label{def:b1:optical-instruments:G}
\index{aumento angular}\index{lupa}\index{aumento comercial}
Para um instrumento usado pelo \omterm{def:b1:optical-instruments:eye}{olho}, o \emph{aumento angular} é
$G = \theta'/\theta$, em que $\theta'$ é o \omterm{def:b1:optical-instruments:angular}{tamanho angular} da imagem
vista através do instrumento e $\theta$ o \omterm{def:b1:optical-instruments:angular}{tamanho angular} do objeto
visto a \omterm{def:b1:optical-instruments:eye}{olho} nu --- no \omterm{def:b1:optical-instruments:punctum}{ponto próximo} $d_m = \qty{25}{cm}$ para
um objeto próximo, e no lugar em que está para um objeto distante.
\end{definition}

\begin{proposition}[A lupa]\label{prop:b1:optical-instruments:loupe}
Uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} $f' < d_m$ com o objeto no seu
plano focal objeto dá uma imagem no infinito, vista sem
\omterm{def:b1:optical-instruments:eye}{acomodação}, com
\[
G = \frac{d_m}{f'} = d_m V .
\]
\end{proposition}

\begin{proof}
Objeto $AB$ de altura $h$ no plano focal: os raios que partem de $B$ saem
paralelos, com o ângulo do raio que passa por $O$, $\theta' = h/f'$. A \omterm{def:b1:optical-instruments:eye}{olho}
nu, no melhor caso: $\theta = h/d_m$. Razão $d_m/f'$.
\end{proof}

\begin{example}[A lupa do relojoeiro]\label{ex:b1:optical-instruments:loupe}
$f' = \qty{25}{mm}$: $G = 250/25 = 10$, vendida como ``$10\times$''. Um
detalhe de \qty{0.05}{mm} subtende então $\qty{2e-3}{rad}$, confortavelmente
resolvido. Levar $G$ mais alto significa $f'$ de poucos milímetros, uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente}
minúscula encostada no \omterm{def:b1:optical-instruments:eye}{olho} --- o limite da \omterm{def:b1:mirrors-thin-lenses:lens}{lente} única, e a
razão de existir do \omterm{def:b1:optical-instruments:microscope}{microscópio}.
\end{example}

\section{O microscópio}

\begin{definition}[Microscópio composto]\label{def:b1:optical-instruments:microscope}
\index{microscópio}\index{objetiva}\index{ocular}\index{intervalo óptico}
Um \emph{microscópio} é uma \emph{objetiva} $L_1$ de curta \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal}
$f_1'$ que forma uma imagem intermediária real e ampliada de um objeto próximo
no plano focal objeto de uma \emph{ocular} $L_2$ (\omterm{def:b1:mirrors-thin-lenses:lens}{distância focal}
$f_2'$), que funciona como \omterm{def:b1:optical-instruments:G}{lupa}. A distância $\Delta =
\overline{F_1'F_2}$ entre o foco imagem da objetiva e o foco objeto
da ocular é o \emph{intervalo óptico}, padronizado em
\qty{160}{mm} nos instrumentos clássicos.
\end{definition}

\begin{proposition}[Aumento de um microscópio]\label{prop:b1:optical-instruments:microscope}
Com a imagem final no infinito,
\[
\gamma_1 = -\frac{\Delta}{f_1'}, \qquad G_2 = \frac{d_m}{f_2'},
\qquad G = \abs{\gamma_1}\,G_2 = \frac{\Delta\,d_m}{f_1' f_2'} .
\]
\end{proposition}

\begin{proof}
A imagem intermediária $A_1B_1$ fica no plano focal objeto da \omterm{def:b1:optical-instruments:microscope}{ocular},
isto é,\ em $\overline{F_1'A_1} = \Delta$; a \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relação de Newton} dá
$\gamma_1 = -\overline{F_1'A_1}/f_1' = -\Delta/f_1'$. A \omterm{def:b1:optical-instruments:microscope}{ocular} então
mostra $A_1B_1$ (altura $\abs{\gamma_1} h$) no infinito sob o ângulo
$\theta' = \abs{\gamma_1} h/f_2'$, contra $\theta = h/d_m$ a \omterm{def:b1:optical-instruments:eye}{olho}
nu.
\end{proof}

\begin{example}[Um microscópio escolar]\label{ex:b1:optical-instruments:micronum}
\omterm{def:b1:optical-instruments:microscope}{Objetiva} $f_1' = \qty{4.0}{mm}$, \omterm{def:b1:optical-instruments:microscope}{ocular} $f_2' = \qty{25}{mm}$,
$\Delta = \qty{160}{mm}$: $\gamma_1 = -40$, $G_2 = 10$, $G = 400$. O
objeto fica a $\overline{F_1A} = -f_1'^2/\Delta = -16/160 =
\qty{-0.10}{mm}$ de $F_1$: \qty{4.1}{mm} da \omterm{def:b1:optical-instruments:microscope}{objetiva}. Focalizar
significa mover o tubo inteiro por frações de milímetro.
\end{example}

\begin{remark}[O limite de resolução]\label{rem:b1:optical-instruments:abbe}
\index{poder de resolução}
Aumentar sem limite seria inútil: a luz é uma onda, e uma
\omterm{def:b1:optical-instruments:microscope}{objetiva} que aceita raios dentro de um semiângulo $u$ (num meio de índice
$n$) não consegue separar dois pontos mais próximos que cerca de
$\lambda/(2n\sin u)$ --- o tratamento ondulatório do volume do segundo ano o
deduz. Com $n\sin u \approx 1.4$ (imersão em óleo) e $\lambda =
\qty{0.5}{\micro m}$, o limite fica perto de \qty{0.2}{\micro m}. Para tornar
esse detalhe visível ao \omterm{def:b1:optical-instruments:eye}{olho} ($\epsilon = \qty{3e-4}{rad}$ a
\qty{25}{cm}, isto é,\ \qty{75}{\micro m}), um aumento de cerca de $400$
basta; além de $1000$ a imagem fica apenas maior, não mais rica:
é o \emph{aumento vazio}.
\end{remark}

\section{A luneta astronômica}

\begin{proposition}[Luneta refratora]\label{prop:b1:optical-instruments:telescope}
\index{luneta}\index{pupila de saída}
Uma \omterm{def:b1:optical-instruments:microscope}{objetiva} $L_1$ e uma \omterm{def:b1:optical-instruments:microscope}{ocular} $L_2$ com $F_1' = F_2$ formam um
\omterm{prop:b1:mirrors-thin-lenses:doublet}{sistema afocal}: uma estrela (no infinito) é vista no infinito, o \omterm{def:b1:optical-instruments:eye}{olho}
relaxado a observa, e
\[
G = -\frac{f_1'}{f_2'} .
\]
A imagem do bordo da \omterm{def:b1:optical-instruments:microscope}{objetiva} através da \omterm{def:b1:optical-instruments:microscope}{ocular} é a
\emph{\omterm{prop:b1:optical-instruments:telescope}{pupila de saída}}: um disco luminoso de diâmetro $D/\abs G$ situado
$\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ atrás da \omterm{def:b1:optical-instruments:microscope}{ocular}, onde
a pupila do \omterm{def:b1:optical-instruments:eye}{olho} deve ser colocada para receber toda a luz.
\end{proposition}

\begin{proof}
$G$ foi deduzido na \cref{prop:b1:mirrors-thin-lenses:doublet}. A
\omterm{def:b1:optical-instruments:microscope}{objetiva}, vista de $L_2$, é um objeto em $\overline{O_2O_1} = -(f_1'
+ f_2')$; Descartes dá $1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2')
= f_1'/[f_2'(f_1' + f_2')]$, e o aumento $\overline{O_2P'}/
\overline{O_2O_1} = -f_2'/f_1' = 1/G$ leva o diâmetro $D$ a
$D/\abs G$. Todo raio que entra na \omterm{def:b1:optical-instruments:microscope}{objetiva} sai pela \omterm{prop:b1:optical-instruments:telescope}{pupila
de saída}, pois cada ponto da \omterm{def:b1:optical-instruments:microscope}{objetiva} é ali conjugado.
\end{proof}

\begin{proposition}[O que uma luneta ganha]\label{prop:b1:optical-instruments:gains}
\index{poder coletor de luz}\index{critério de Rayleigh}
Comparada ao \omterm{def:b1:optical-instruments:eye}{olho} nu (pupila $d_p$):
\begin{itemize}
  \item \emph{\omterm{def:b1:optical-instruments:angular}{tamanho angular}}: multiplicado por $\abs G$;
  \item \emph{luz recolhida} de uma fonte pontual: multiplicada por
        $(D/d_p)^2$, desde que a \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} não seja maior que a
        pupila do \omterm{def:b1:optical-instruments:eye}{olho}, isto é, $\abs G \geq D/d_p$;
  \item \emph{\omterm{def:b1:optical-instruments:angular}{resolução angular}}: a difração na \omterm{def:b1:optical-instruments:microscope}{objetiva} a limita
        a $\theta_{\min} \approx 1.22\,\lambda/D$ (critério de
        Rayleigh), muito abaixo do $\epsilon$ do \omterm{def:b1:optical-instruments:eye}{olho}; o aumento
        que torna esse limite visível é $G_r = \epsilon/\theta_{\min}$,
        e aumentos maiores só ampliam a mancha.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Um refrator amador de 150 milímetros]\label{ex:b1:optical-instruments:telnum}
$D = \qty{150}{mm}$, $f_1' = \qty{1200}{mm}$, \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}:
$G = -48$; \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} \qty{3.1}{mm}; ganho de luz $(150/7)^2 \approx
460$ frente a uma pupila adaptada ao escuro de \qty{7}{mm}; resolução $1.22 \times
\qty{550}{nm}/\qty{0.15}{m} = \qty{4.5e-6}{rad} = 0.9''$, ou seja,
crateras de \qty{1.7}{km} na Lua; $G_r = 3\times10^{-4}/4.5\times
10^{-6} \approx 67$ --- a \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{6}{mm} ($G = 200$) não mostra
mais detalhe, apenas um disco maior e mais fraco. O problema de fim de semana projeta
este instrumento.
\end{example}

\begin{remark}[Refletores, binóculos, lunetas de Galileu]\label{rem:b1:optical-instruments:variants}
Um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho côncavo} de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} $f_1'$ substitui a \omterm{def:b1:optical-instruments:microscope}{objetiva} de \omterm{def:b1:mirrors-thin-lenses:lens}{lente} no
telescópio \emph{refletor} (sem dispersão cromática, e um espelho
pode ter metros de largura e ser sustentado por trás): tudo o que foi dito acima
vale com $f_1'$ igual à \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} do espelho. A imagem é invertida ---
inofensivo para estrelas; para uso terrestre, os binóculos inserem um par de
prismas de \omterm{thm:b1:rays-reflection-refraction:tir}{reflexão total} para endireitá-la, e a luneta de Galileu usa uma
\omterm{def:b1:optical-instruments:microscope}{ocular} divergente ($f_2' < 0$, $G > 0$), mais curta, mas com um campo
estreito.
\end{remark}

\section{A câmera}

\begin{definition}[Câmera; número f]\label{def:b1:optical-instruments:camera}
\index{câmera}\index{número f}\index{profundidade de campo}
Uma \emph{câmera} é uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} $f'$ que forma uma \omterm{def:b1:mirrors-thin-lenses:image}{imagem
real} sobre um sensor, em $\overline{OA'} \approx f'$ para motivos distantes;
focalizar desloca a \omterm{def:b1:mirrors-thin-lenses:lens}{lente}. Um diafragma de diâmetro $D$ limita o feixe;
o \emph{número f} é $N = f'/D$ (escrito $f/N$). O \emph{ângulo de
campo} vale $2\arctan(\ell/2f')$ para um sensor de largura $\ell$.
\end{definition}

\begin{proposition}[Exposição e profundidade de campo]\label{prop:b1:optical-instruments:dof}
\index{distância hiperfocal}
A irradiância sobre o sensor varia como $1/N^2$, de modo que o tempo de exposição
para um dado brilho de imagem varia como $N^2$. Um ponto a uma distância
diferente daquela em que se focalizou forma um disco de confusão; exigindo que ele fique
abaixo de um diâmetro tolerado $c$, uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente} focalizada na \emph{\omterm{prop:b1:optical-instruments:dof}{distância
hiperfocal}}
\[
H = \frac{f'^2}{N c}
\]
mostra aceitavelmente nítido tudo o que estiver entre $H/2$ e o infinito.
\end{proposition}

\begin{proof}
Irradiância $=$ potência/área $\propto D^2/f'^2 = 1/N^2$ (a imagem de um
dado motivo tem tamanho fixado por $f'$, e a potência recolhida cresce
como $D^2$). Focalize à distância $p$, com a imagem em $q \approx f'$;
um ponto no infinito focaliza em $f'$, isto é,\ $q - f'$ antes do sensor;
o seu cone de raios, de base $D$ na \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, tem diâmetro $D(q - f')/q$
sobre o sensor (triângulos semelhantes). Com $1/q = 1/f' - 1/p$ paraxialmente,
$q - f' \approx f'^2/p$ para $p \gg f'$, de modo que o borrão vale
$Df'/p = f'^2/(Np)$; ele iguala $c$ para $p = H$. A mesma geometria do
lado próximo dá o limite $H/2$.
\end{proof}

\section{Problema: projetando uma luneta para a Lua}

\begin{problem}[{Problema de fim de semana --- um refrator de 150 milímetros no
gramado do quintal: escolher as oculares, posicionar o olho, contar a luz e
descobrir qual a menor cratera que ele consegue mostrar}]
\label{pb:b1:optical-instruments:1}
A \omterm{def:b1:optical-instruments:microscope}{objetiva} é uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente} $L_1$ de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal}
$f_1' = \qty{1200}{mm}$ e diâmetro $D = \qty{150}{mm}$; há \omterm{def:b1:optical-instruments:microscope}{oculares}
$L_2$ de distâncias focais \qty{25}{mm} e \qty{6}{mm}. A
Lua está a \qty{3.84e5}{km} e subtende $0.52^\circ$; a pupila
noturna do \omterm{def:b1:optical-instruments:eye}{olho} vale \qty{7}{mm} e a sua resolução $\epsilon = \qty{3.0e-4}{rad}$;
$\lambda = \qty{550}{nm}$.

\medskip
\noindent\textbf{Parte I --- O arranjo afocal.}
\begin{enumerate}
  \item Onde a \omterm{def:b1:optical-instruments:microscope}{ocular} deve ser colocada para que uma estrela seja vista sem
        \omterm{def:b1:optical-instruments:eye}{acomodação}? Dê a distância objetiva--ocular para cada
        \omterm{def:b1:optical-instruments:microscope}{ocular}.
  \item Verifique, com a \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relação de Descartes} aplicada a cada \omterm{def:b1:mirrors-thin-lenses:lens}{lente} por sua vez,
        que um ponto no infinito sobre o eixo é conjugado no infinito.
  \item Mostre que um feixe vindo de uma estrela a um ângulo $\alpha$ do eixo
        emerge como feixe paralelo a um ângulo $\alpha' = -(f_1'/f_2')\alpha$.
  \item Calcule $G$ para cada \omterm{def:b1:optical-instruments:microscope}{ocular} e o diâmetro aparente da
        Lua através de cada uma.
  \item A imagem é invertida. Por que isso é aceitável aqui, e o que
        acrescentaria uma luneta terrestre?
  \item Onde está a imagem intermediária real da Lua, e qual é o
        seu diâmetro? (Ela pode ser fotografada ali.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Onde fica o \omterm{def:b1:optical-instruments:eye}{olho}.}
\begin{enumerate}[resume]
  \item Mostre que a \omterm{def:b1:optical-instruments:microscope}{ocular} forma uma \omterm{def:b1:mirrors-thin-lenses:image}{imagem real} do bordo da \omterm{def:b1:optical-instruments:microscope}{objetiva}
        em $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ atrás dela, de
        diâmetro $D/\abs G$ (a \omterm{prop:b1:optical-instruments:telescope}{pupila de saída}).
  \item Calcule a posição e o diâmetro da \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} para cada
        \omterm{def:b1:optical-instruments:microscope}{ocular}.
  \item Por que a pupila do \omterm{def:b1:optical-instruments:eye}{olho} deve ser colocada na \omterm{prop:b1:optical-instruments:telescope}{pupila de saída}? O que
        acontece com o campo se ela for colocada mais atrás?
  \item Para que a luz de uma estrela seja plenamente aproveitada, a \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} não pode
        exceder a pupila do \omterm{def:b1:optical-instruments:eye}{olho}: deduza o aumento útil mínimo
        desta luneta à noite.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Resolução.}
\begin{enumerate}[resume]
  \item Calcule o limite de difração $\theta_{\min} = 1.22\lambda/D$
        em radianos e em segundos de arco.
  \item Qual é a menor cratera a que esse ângulo corresponde na
        Lua? E a \omterm{def:b1:optical-instruments:eye}{olho} nu?
  \item Defina e calcule o aumento resolvente $G_r =
        \epsilon/\theta_{\min}$. Qual \omterm{def:b1:optical-instruments:microscope}{ocular} chega mais perto dele?
  \item Com a \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{6}{mm} a imagem parece mais borrada e mais
        fraca do que com a de \qty{25}{mm}. Explique os dois efeitos.
  \item A turbulência atmosférica espalha as imagens das estrelas em cerca de $1''$. Para
        que diâmetros de \omterm{def:b1:optical-instruments:microscope}{objetiva} o limite de Rayleigh deixa de
        importar a partir do solo, e o que um $D$ maior ainda traz?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A luz.}
\begin{enumerate}[resume]
  \item Por qual fator a \omterm{def:b1:optical-instruments:microscope}{objetiva} recolhe mais luz do que a
        pupila noturna?
  \item Converta isso em magnitudes ($\Delta m = 2.5\log_{10}$ da
        razão); se o \omterm{def:b1:optical-instruments:eye}{olho} alcança a magnitude $6$, que magnitude
        a luneta alcança?
  \item A Lua cheia entrega ao solo uma irradiância de cerca de
        \qty{2e-3}{W/m^2}. Que potência entra na
        \omterm{def:b1:optical-instruments:microscope}{objetiva}, e como ela se distribui sobre a \omterm{def:b1:optical-instruments:eye}{retina} em comparação com
        a observação a \omterm{def:b1:optical-instruments:eye}{olho} nu (mesmo argumento de potência por unidade de ângulo sólido:
        compare o brilho superficial através das duas \omterm{def:b1:optical-instruments:microscope}{oculares})?
  \item Por que uma luneta não torna a superfície da Lua mais brilhante por
        unidade de área do que o \omterm{def:b1:optical-instruments:eye}{olho} nu, ao passo que torna as estrelas
        mais brilhantes?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Campo de visão.}
A \omterm{def:b1:optical-instruments:microscope}{ocular} traz, no seu plano focal, um diafragma de campo circular de
diâmetro \qty{20}{mm} (\omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}) ou \qty{5}{mm}
(\omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{6}{mm}).
\begin{enumerate}[resume]
  \item Mostre que o campo de visão real vale $2\arctan(\phi/2f_1')
        \approx \phi/f_1'$ para um diafragma de diâmetro $\phi$.
  \item Calcule-o para as duas \omterm{def:b1:optical-instruments:microscope}{oculares} e diga se a Lua inteira
        cabe no campo.
  \item Calcule o campo aparente (o ângulo sob o qual o \omterm{def:b1:optical-instruments:eye}{olho} vê
        o diafragma através da \omterm{def:b1:optical-instruments:microscope}{ocular}) para as duas.
  \item A Terra gira $15''$ por segundo de tempo: quanto tempo a
        Lua leva para atravessar o campo de cada \omterm{def:b1:optical-instruments:microscope}{ocular} sem acompanhamento?
  \item Resuma o projeto: que \omterm{def:b1:optical-instruments:microscope}{ocular} para a Lua inteira, qual
        para as crateras, e o menor tamanho de cratera que o instrumento consegue
        revelar --- o seu único número.
  \item A mesma \omterm{def:b1:optical-instruments:microscope}{objetiva} é substituída por um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho côncavo} de \qty{150}{mm}
        de mesma \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal}. Quais dos resultados acima mudam?
\end{enumerate}
\end{problem}
