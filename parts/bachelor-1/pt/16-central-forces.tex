\chapter{Forças centrais: planetas e satélites}\label{ch:b1:central-forces}

Trinta e um satélites dão duas voltas por dia em torno da Terra, cada um um relógio
e um rádio, e um celular que ouve quatro deles sabe onde está com
alguns metros de precisão. O cometa Halley cai rumo ao Sol durante trinta e oito
anos, contorna-o em poucas semanas e volta a subir por mais
trinta e oito. Uma sonda rumo a Marte deixa a Terra exatamente com a
velocidade que fará o seu trajeto apenas tangenciar a órbita de Marte. Tudo isso
é o movimento de um corpo atraído para um ponto fixo por uma força que
cai com o inverso do quadrado da distância, e tudo isso decorre
de duas grandezas conservadas --- a energia e o \omterm{def:b1:angular-momentum:L}{momento angular} --- mais
uma única ideia nova, o potencial efetivo. Este capítulo deduz a
forma e as velocidades das órbitas, as três \omterm{thm:b1:central-forces:conics}{leis de Kepler} e a aritmética de
pôr um satélite onde se quer.

\section{Forças centrais conservativas}

\begin{definition}[Força central; força central conservativa]\label{def:b1:central-forces:central}
\index{força central}\index{energia potencial efetiva}
Uma força é \emph{central} com centro $O$ se está sempre dirigida
ao longo de $\vect{OM}$: $\vect F = F(M)\,\vect e_r$. Ela é \emph{central
conservativa} se o seu módulo depende apenas de $r = OM$: $\vect F =
F(r)\,\vect e_r$, o que decorre de uma \omterm{def:b1:work-and-energy:conservative}{energia potencial} $E_p(r)$ com
$F(r) = -\dd E_p/\dd r$.
\end{definition}

\begin{theorem}[Movimento sob uma força central conservativa]\label{thm:b1:central-forces:reduction}
\index{lei das áreas}
Um \omterm{def:b1:point-kinematics:frame}{ponto material} de massa $m$ submetido apenas a uma \omterm{def:b1:central-forces:central}{força central} conservativa:
\begin{enumerate}
  \item conserva o seu \omterm{def:b1:angular-momentum:L}{momento angular} $\vect L_O$; o seu movimento é plano e,
        em \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares} nesse plano, $r^2\dot\theta = C$ é
        constante (\omterm{cor:b1:angular-momentum:central}{lei das áreas}, \cref{cor:b1:angular-momentum:central});
  \item conserva a sua \omterm{thm:b1:work-and-energy:em}{energia mecânica} $E_m = \tfrac12 m(\dot r^2 + r^2\dot\theta^2)
        + E_p(r)$, que se pode escrever
        \[
        E_m = \tfrac12 m\dot r^2 + E_{p,\mathrm{eff}}(r), \qquad
        E_{p,\mathrm{eff}}(r) = E_p(r) + \frac{mC^2}{2r^2} = E_p(r) + \frac{L^2}{2mr^2} .
        \]
\end{enumerate}
O movimento radial é, portanto, o de um \omterm{def:b1:point-kinematics:frame}{ponto material} unidimensional no
\emph{potencial efetivo} $E_{p,\mathrm{eff}}$: os \omterm{def:b1:work-and-energy:turning}{pontos de retorno}
radiais são as raízes de $E_{p,\mathrm{eff}}(r) = E_m$, o movimento é
ligado se ficar preso entre dois deles, e circular se $r$ se situar num
mínimo de $E_{p,\mathrm{eff}}$.
\end{theorem}

\begin{proof}
A parte 1 é o \cref{cor:b1:angular-momentum:central}. A força é
conservativa e nenhuma outra atua, de modo que $E_m$ se conserva; no
plano, $v^2 = \dot r^2 + r^2\dot\theta^2$ e $r^2\dot\theta^2 = C^2/r^2$.
O termo $mC^2/2r^2$, \omterm{thm:b1:work-and-energy:ke}{energia cinética} do movimento angular, age como uma
\emph{\omterm{prop:b1:central-forces:effective}{barreira centrífuga}} repulsiva no problema radial, ao qual se aplica
a análise unidimensional do \cref{ch:b1:work-and-energy}.
\end{proof}

\section{A gravitação newtoniana}

\begin{definition}[Gravitação universal]\label{def:b1:central-forces:gravitation}
\index{gravitação}\index{energia potencial gravitacional}
Uma massa pontual $M$ em $O$ atrai uma massa pontual $m$ em $M$ com
\[
\vect F = -\frac{GMm}{r^2}\,\vect e_r, \qquad E_p = -\frac{GMm}{r} \quad (E_p \to 0 \text{ no infinito}),
\]
com $G = \qty{6.67e-11}{N.m^2/kg^2}$. Um corpo de simetria esférica atrai,
fora de si, como se toda a sua massa estivesse no centro (admitido; é
consequência do teorema de Gauss, \cref{ch:b1:electrostatics-gauss}).
\end{definition}

\begin{proposition}[O potencial efetivo da gravitação]\label{prop:b1:central-forces:effective}
\index{barreira centrífuga}\index{órbita circular}
Para $E_p = -GMm/r$,
\[
E_{p,\mathrm{eff}}(r) = -\frac{GMm}{r} + \frac{L^2}{2mr^2}
\]
tende a $+\infty$ quando $r \to 0$ (para $L \neq 0$) e a $0^-$ no infinito,
e tem um único mínimo em $r_c = L^2/GMm^2$, de valor $-G^2M^2m^3/2L^2$.
Daí: $E_m < 0$ --- \omterm{def:b1:work-and-energy:turning}{movimento ligado} entre um raio mínimo e um
raio máximo (perigeu e apogeu); $E_m \geq 0$ --- não ligado, o corpo vem
do infinito e a ele retorna; $E_m$ igual ao mínimo --- \omterm{prop:b1:central-forces:effective}{órbita
circular} de raio $r_c$.
\end{proposition}

\begin{proof}
Os limites por inspeção; $E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ em
$r_c$; substitua. O caso circular é $\dot r = 0$ para sempre, possível
apenas no mínimo.
\end{proof}

\begin{theorem}[Órbitas circulares]\label{thm:b1:central-forces:circular}
\index{velocidade orbital}\index{terceira lei de Kepler}
Numa \omterm{prop:b1:central-forces:effective}{órbita circular} de raio $r$ em torno de uma massa $M$:
\[
v = \sqrt{\frac{GM}{r}}, \qquad
T = 2\pi\sqrt{\frac{r^3}{GM}}, \qquad
E_m = -\frac{GMm}{2r} = -E_k = \tfrac12 E_p .
\]
Em particular, $T^2/r^3 = 4\pi^2/GM$ é o mesmo para todas as \omterm{prop:b1:central-forces:effective}{órbitas circulares}
em torno do mesmo centro (\omterm{thm:b1:central-forces:circular}{terceira lei de Kepler}).
\end{theorem}

\begin{proof}
O \omterm{cor:b1:point-kinematics:circular}{movimento circular} uniforme exige a \omterm{prop:b1:newton-dynamics:centripetal}{força centrípeta} $mv^2/r = GMm/r^2$;
$T = 2\pi r/v$; $E_k = \tfrac12 mv^2 = GMm/2r$ e $E_p = -GMm/r$.
\end{proof}

\begin{example}[Três órbitas]\label{ex:b1:central-forces:orbits}
$GM_\oplus = \qty{3.99e14}{m^3/s^2}$. Órbita baixa ($r = \qty{6770}{km}$,
\qty{400}{km} de altitude): $v = \qty{7.7}{km/s}$, $T = \qty{92}{min}$. Geoestacionária
($T = \qty{86164}{s}$, um dia sideral): $r = (GMT^2/4\pi^2)^{1/3} =
\qty{42200}{km}$, $v = \qty{3.1}{km/s}$. A Lua ($r = \qty{384000}{km}$):
$T = \qty{27.4}{d}$ --- e a sua \omterm{cor:b1:point-kinematics:circular}{aceleração centrípeta} $v^2/r =
\qty{2.7e-3}{m/s^2}$ vale $g/3600$ a sessenta raios terrestres: o inverso
do quadrado, verificado pelo próprio Newton.
\end{example}

\section{As leis de Kepler e a forma das órbitas}

\begin{theorem}[Trajetórias num campo newtoniano]\label{thm:b1:central-forces:conics}
\index{leis de Kepler}\index{cônica}\index{elipse (órbita)}\index{excentricidade}
A \omterm{def:b1:point-kinematics:frame}{trajetória} de um corpo no campo de uma massa fixa $M$ é uma cônica
com $O$ num dos focos: uma elipse se $E_m < 0$, uma parábola se $E_m = 0$
e uma hipérbole se $E_m > 0$. Para a elipse de semieixo maior $a$ e
excentricidade $e$:
\begin{itemize}
  \item (Kepler I) o centro de força está num foco; perigeu e apogeu
        ficam em $r_p = a(1 - e)$ e $r_a = a(1 + e)$;
  \item (Kepler II) a \omterm{cor:b1:angular-momentum:central}{velocidade areolar} é constante (\omterm{cor:b1:angular-momentum:central}{lei das áreas});
  \item (Kepler III) o período obedece a $T^2 = 4\pi^2a^3/GM$;
  \item a energia depende apenas de $a$, $E_m = -GMm/2a$, de modo que a
        velocidade à distância $r$ é dada por
        \[
        v^2 = GM\left(\frac{2}{r} - \frac{1}{a}\right) .
        \]
\end{itemize}
\end{theorem}
\admitted

\begin{remark}[O que é demonstrado e o que é admitido]\label{rem:b1:central-forces:admitted}
Kepler II e a classificação por energia foram demonstrados acima. Que as
órbitas ligadas sejam exatamente elipses com $O$ num foco, com Kepler III
e $E_m = -GMm/2a$ em geral, exige a integração da equação
radial (o volume do segundo ano a faz); para \omterm{prop:b1:central-forces:effective}{órbitas circulares} ($a = r$,
$e = 0$) toda afirmação se reduz ao \cref{thm:b1:central-forces:circular},
que é a verificação possível aqui. A relação $v^2 = GM(2/r - 1/a)$
decorre de $E_m = \tfrac12 mv^2 - GMm/r = -GMm/2a$, uma vez concedida essa
fórmula da energia.
\end{remark}

\begin{example}[O cometa Halley]\label{ex:b1:central-forces:halley}
Periélio \qty{0.586}{AU}, afélio \qty{35.1}{AU}: $a = \qty{17.8}{AU}$,
$e = 0.967$, $T = \sqrt{17.8^3} = 75$ anos (Kepler III em
unidades astronômicas e anos, em que $T^2 = a^3$ para o Sol). Com
$GM_\odot = \qty{1.33e20}{m^3/s^2}$, a velocidade no periélio vale
$\sqrt{GM(2/r_p - 1/a)} = \qty{54}{km/s}$ e no afélio \qty{0.9}{km/s}
--- sessenta vezes mais lenta, como a \omterm{cor:b1:angular-momentum:central}{lei das áreas} exige ($r_pv_p = r_av_a$).
\end{example}

\section{Satélites}

\begin{proposition}[Velocidade de escape; órbita geoestacionária]\label{prop:b1:central-forces:escape}
\index{velocidade de escape}\index{órbita geoestacionária}
A partir da superfície de um corpo de massa $M$ e raio $R$, a menor
velocidade de lançamento capaz de alcançar o infinito é $v_e = \sqrt{2GM/R} = \sqrt2\,v_c$,
em que $v_c = \sqrt{GM/R}$ é a velocidade circular ao nível do solo:
\qty{11.2}{km/s} para a Terra. Um satélite \emph{geoestacionário} orbita
no plano equatorial com o período sideral da Terra, a
$r = \qty{42200}{km}$ (\qty{35800}{km} de altitude).
\end{proposition}

\begin{proof}
$E_m \geq 0$: $\tfrac12 mv_e^2 - GMm/R = 0$. Kepler III com $T = \qty{86164}{s}$.
\end{proof}

\begin{proposition}[Transferência de Hohmann]\label{prop:b1:central-forces:hohmann}
\index{transferência de Hohmann}
Para levar um satélite de uma \omterm{prop:b1:central-forces:effective}{órbita circular} de raio $r_1$ a uma \omterm{prop:b1:central-forces:effective}{órbita
circular} coplanar de raio $r_2 > r_1$ com duas queimas breves, acione o motor em $r_1$
para entrar na \emph{elipse de transferência} de perigeu $r_1$ e apogeu $r_2$
($a = (r_1 + r_2)/2$), percorra meio período e acione de novo em $r_2$ para
circularizar:
\[
\Delta v_1 = \sqrt{\frac{GM}{r_1}}\left(\sqrt{\frac{2r_2}{r_1 + r_2}} - 1\right),
\qquad
\Delta v_2 = \sqrt{\frac{GM}{r_2}}\left(1 - \sqrt{\frac{2r_1}{r_1 + r_2}}\right),
\qquad
t = \pi\sqrt{\frac{(r_1 + r_2)^3}{8GM}} .
\]
\end{proposition}

\begin{proof}
As velocidades na elipse de transferência vêm de $v^2 = GM(2/r - 1/a)$ em $r_1$ e
$r_2$; subtraia as velocidades circulares; e tome metade do período do
\cref{thm:b1:central-forces:conics}.
\end{proof}

\begin{example}[Da órbita baixa à geoestacionária]\label{ex:b1:central-forces:leogeo}
$r_1 = \qty{6770}{km}$, $r_2 = \qty{42200}{km}$: $\Delta v_1 = \qty{2.4}{km/s}$,
$\Delta v_2 = \qty{1.5}{km/s}$, com \qty{5.3}{h} de percurso. O total,
\qty{3.9}{km/s}, é metade do que o lançamento até a órbita baixa já custou: a
``última milha'' rumo a uma órbita alta é cara, e os satélites levam o próprio
motor para ela.
\end{example}

\begin{remark}[Centros repulsivos]\label{rem:b1:central-forces:rutherford}
\index{espalhamento de Rutherford}
Para uma força repulsiva em inverso do quadrado (duas cargas de mesmo sinal,
\cref{ch:b1:electrostatics-gauss}) o potencial efetivo não tem poço:
toda \omterm{def:b1:point-kinematics:frame}{trajetória} é uma hipérbole, a partícula se aproxima, é desviada
e vai embora. Lançando uma partícula $\alpha$ direto contra um núcleo de ouro, a sua
distância de máxima aproximação decorre só da energia, $E_k =
2Ze^2/4\pi\varepsilon_0 r_{\min}$ --- o modo como Rutherford limitou o tamanho do
núcleo (\cref{exo:b1:central-forces:11}).
\end{remark}

e que as \omterm{prop:b1:work-and-energy:stability}{pequenas oscilações} radiais em torno dele têm \omterm{def:b1:wave-propagation:signal}{frequência angular}
$\sqrt{GM/r_c^3}$ --- a própria frequência orbital. O que isso diz
sobre uma \omterm{prop:b1:central-forces:effective}{órbita circular} ligeiramente perturbada?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:9}
Os astronautas a bordo da EEI flutuam. Explique pelas forças; qual é o
valor de $g$ na sua altitude, e por que isso não contradiz a
flutuação?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:10}
O ``teste da Lua'' de Newton: calcule a \omterm{cor:b1:point-kinematics:circular}{aceleração centrípeta} da Lua a partir de
$r = \qty{3.84e8}{m}$ e $T = \qty{27.3}{d}$, e compare com
$g(R/r)^2$. Conclua.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:11}
Uma partícula $\alpha$ ($q = 2e$, $E_k = \qty{5.0}{MeV}$) é lançada direto
contra um núcleo de ouro ($Z = 79$), tomado como fixo. Usando a conservação da energia
com $E_p = 2Ze^2/4\pi\varepsilon_0r$ ($1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$),
encontre a distância de máxima aproximação; compare com o raio nuclear,
cerca de \qty{7}{fm}. Que energia seria necessária para tocar o núcleo?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:12}
Um satélite em órbita baixa é freado pela atmosfera residual, de modo que a sua

energia, e que fração da \omterm{def:b1:work-and-energy:conservative}{energia potencial} perdida vira calor?
\end{exercise}

\section{Problema: os satélites que dizem onde você está}

\begin{problem}[{Problema de fim de semana --- uma constelação de navegação a vinte
mil quilômetros de altitude: a sua órbita, o combustível para chegar lá, quantos
satélites um celular consegue ver e por que os seus relógios precisam ser ajustados para
atrasar}]
\label{pb:b1:central-forces:1}
Dados: $GM_\oplus = \qty{3.986e14}{m^3/s^2}$, $R_\oplus = \qty{6371}{km}$,
dia sideral \qty{86164}{s}, $c = \qty{2.998e8}{m/s}$. Os satélites
dão a volta na Terra em exatamente meio dia sideral, em \omterm{prop:b1:central-forces:effective}{órbitas circulares}.

\medskip
\noindent\textbf{Parte I --- A órbita.}
\begin{enumerate}
  \item Calcule o raio orbital e a altitude.
  \item Calcule a \omterm{thm:b1:central-forces:circular}{velocidade orbital}.
  \item Calcule a aceleração do satélite e compare-a com $g$ na
        superfície.
  \item Calcule a \omterm{thm:b1:work-and-energy:em}{energia mecânica} por quilograma, e as partes cinética e
        potencial.
  \item Por que escolher meio dia sideral e não meio dia solar?
        O que um observador em terra vê se repetir?
  \item Visto de um satélite, que ângulo o disco da Terra subtende?
  \item Compare o raio dessa órbita com o da geoestacionária; qual é
        mais alta, e por qual fator?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Chegar lá.}
\begin{enumerate}[resume]
  \item Energia por quilograma para ir do repouso na superfície até a órbita
        final (despreze o ar e a rotação da Terra).
  \item O lançador atinge primeiro uma \omterm{prop:b1:central-forces:effective}{órbita circular} baixa a \qty{400}{km}.
        Qual a velocidade ali?
  \item \omterm{prop:b1:central-forces:hohmann}{Transferência de Hohmann} daquela órbita para a final: as duas queimas
        e a sua soma.
  \item Duração da transferência.
  \item Compare a soma das queimas com a \omterm{prop:b1:central-forces:escape}{velocidade de escape} a partir da órbita
        baixa: quão longe de ``deixar a Terra'' está esta missão?
  \item O estágio superior e o satélite têm \qty{2000}{kg} antes da
        transferência; a velocidade de exaustão do motor vale \qty{3.0}{km/s} e a
        equação do foguete dá $\Delta v = u\ln(m_0/m_1)$. Quanto propelente
        é preciso para as duas queimas?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Cobertura e cronometragem.}
\begin{enumerate}[resume]
  \item Um satélite é visível de todo ponto da Terra que fique
        dentro do cone de semiângulo $\theta$ com $\cos\theta = R/r$
        em torno da linha Terra--satélite. Calcule $\theta$ e a
        fração $(1 - \cos\theta)/2$ da superfície da Terra que ele cobre.
  \item Com $24$ satélites distribuídos uniformemente, quantos são visíveis em
        média de um dado ponto? O mínimo de $4$ necessário para uma posição é
        plausível?
  \item Tempo de percurso da luz de um satélite no zênite; e no
        horizonte.
  \item Para localizar o receptor com \qty{3}{m} de precisão, com que precisão os
        sinais devem ser cronometrados?
  \item O relógio do próprio receptor é um quartzo barato, errado em
        milissegundos: por que são necessários quatro satélites em vez de três?
  \item Os relógios dos satélites são atômicos. Dê a estabilidade relativa
        necessária para um erro de \qty{3}{m} após um dia.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A relatividade, em resumo.}
Dois efeitos deslocam os relógios dos satélites em relação aos relógios em terra
(fórmulas admitidas, do volume do terceiro ano): o movimento atrasa um relógio na
fração $v^2/2c^2$; a altitude o adianta em $GM(1/R - 1/r)/c^2$.
\begin{enumerate}[resume]
  \item Calcule o atraso devido à velocidade, em microssegundos por dia.
  \item Calcule o adiantamento devido à altitude, em microssegundos por
        dia.
  \item Efeito líquido e erro de posição que ele causaria por dia se não
        fosse corrigido.
  \item Os relógios dos satélites são ajustados antes do lançamento para
        \qty{10.22999999543}{MHz} em vez de \qty{10.23}{MHz}. Verifique que
        isso compensa o efeito líquido.
  \item As órbitas não são exatamente keplerianas: nomeie três perturbações
        e diga como o sistema lida com elas.
  \item Resuma a cadeia: duas leis de conservação, uma órbita, uma
        constelação e a única correção que ninguém teria adivinhado.
\end{enumerate}
\end{problem}
