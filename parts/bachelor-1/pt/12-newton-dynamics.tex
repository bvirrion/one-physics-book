\chapter{As leis de Newton da dinâmica}\label{ch:b1:newton-dynamics}

Um paraquedista sai do avião e, em uma dezena de segundos, para de
acelerar: a duzentos quilômetros por hora o ar empurra de volta
exatamente com a mesma \omterm{def:b1:newton-dynamics:momentum}{força} com que a Terra puxa. Um carro faz uma curva rápido demais e
os pneus cedem; um caminhão numa estrada de serra sobe em marcha reduzida,
segurado pelo mesmo atrito que lhe permite parar no plano. A cinemática
descreveu movimentos; a dinâmica os explica, a partir de uma única lei que liga
a aceleração de um corpo às \omterm{def:b1:newton-dynamics:momentum}{forças} que agem sobre ele. Este capítulo
enuncia as três \omterm{thm:b1:newton-dynamics:laws}{leis de Newton}, cataloga as \omterm{def:b1:newton-dynamics:momentum}{forças} que se encontram no
nível deste volume --- peso, molas, fios, contatos, \omterm{prop:b1:newton-dynamics:friction}{atrito
seco} e fluido --- e mostra como ir de um desenho das \omterm{def:b1:newton-dynamics:momentum}{forças} até
uma equação diferencial e a sua solução.

\section{Massa, quantidade de movimento e as três leis}

\begin{definition}[Massa, quantidade de movimento, força]\label{def:b1:newton-dynamics:momentum}
\index{massa}\index{quantidade de movimento}\index{força}
Um \omterm{def:b1:point-kinematics:frame}{ponto material} tem uma \emph{massa inercial} $m > 0$ (quilogramas), uma
medida da sua resistência às variações de velocidade, independente do
\omterm{def:b1:point-kinematics:frame}{referencial}. A sua \emph{quantidade de movimento} num \omterm{def:b1:point-kinematics:frame}{referencial} vale
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
Uma \emph{força} $\vect F$ (newtons) é a ação de outro corpo sobre o
\omterm{def:b1:point-kinematics:frame}{ponto material}; as forças se somam como vetores, e a \emph{resultante} é a sua soma
$\sum\vect F$.
\end{definition}

\begin{theorem}[Leis de Newton]\label{thm:b1:newton-dynamics:laws}
\index{leis de Newton}\index{referencial inercial}\index{princípio da inércia}\index{ação e reação}
\begin{enumerate}
  \item \emph{Princípio da inércia.} Existem referenciais, ditos
        \emph{inerciais} (ou galileanos), nos quais um \omterm{def:b1:point-kinematics:frame}{ponto material} submetido
        a nenhuma \omterm{def:b1:newton-dynamics:momentum}{força}, ou a \omterm{def:b1:newton-dynamics:momentum}{forças} de resultante nula, se move em linha
        reta com velocidade constante. Todo \omterm{def:b1:point-kinematics:frame}{referencial} em translação
        retilínea uniforme em relação a um \omterm{thm:b1:newton-dynamics:laws}{referencial inercial} é inercial.
  \item \emph{Lei da \omterm{def:b1:newton-dynamics:momentum}{quantidade de movimento}.} Num \omterm{thm:b1:newton-dynamics:laws}{referencial inercial},
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{isto é,}\qquad
        m\,\vect a = \sum\vect F \ \text{ para massa constante } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{Ação e reação}.} Se o corpo $A$ exerce $\vect F_{A\to B}$
        sobre o corpo $B$, então $B$ exerce $\vect F_{B\to A} = -\vect F_{A\to B}$
        sobre $A$, ao longo da reta que os une.
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[O que as leis dizem]\label{rem:b1:newton-dynamics:meaning}
A primeira lei não é um caso particular da segunda: ela \emph{define}
os referenciais em que a segunda vale. Na prática um laboratório na
Terra é inercial com excelente precisão para experimentos que duram minutos
(o \cref{ch:b1:non-inertial-frames} quantifica o erro); um \omterm{def:b1:point-kinematics:frame}{referencial}
ligado ao Sol e às estrelas distantes é ainda melhor. A segunda
lei é uma equação vetorial: três equações escalares, uma por eixo, e
a escolha dos eixos (\cref{ch:b1:point-kinematics}) é a primeira
decisão de todo problema. A terceira lei vale a cada instante para
as \omterm{def:b1:newton-dynamics:momentum}{forças} de contato e gravitacionais deste volume.
\end{remark}

\begin{definition}[Sistema fechado; conservação da quantidade de movimento]\label{def:b1:newton-dynamics:conservation}
\index{sistema fechado}\index{conservação da quantidade de movimento}
Um \emph{sistema fechado} não troca matéria com o exterior. Para um
sistema de \omterm{def:b1:point-kinematics:frame}{pontos materiais}, as \omterm{def:b1:newton-dynamics:momentum}{forças} internas se cancelam aos pares (terceira lei),
e a \omterm{def:b1:newton-dynamics:momentum}{quantidade de movimento} total $\vect P = \sum m_i\vect v_i$ obedece a
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$: ela se conserva quando
as \omterm{def:b1:newton-dynamics:momentum}{forças} externas se anulam (o \cref{ch:b1:systems-of-points} desenvolve
isso).
\end{definition}

\section{As forças usuais}

\begin{proposition}[Peso e gravitação]\label{prop:b1:newton-dynamics:weight}
\index{peso}\index{campo gravitacional}
Perto da superfície da Terra um corpo de \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ é puxado pelo seu
\emph{peso} $\vect P = m\vect g$, sendo $\vect g$ o \omterm{prop:b1:newton-dynamics:weight}{campo gravitacional}
local, $g \approx \qty{9.81}{m/s^2}$, dirigido para baixo (rumo ao
centro, com ótima aproximação). Mais geralmente, uma \omterm{def:b1:newton-dynamics:momentum}{massa} $M$ à distância $r$ atrai
$m$ com $\vect F = -GMm/r^2\,\vect e_r$ (\cref{ch:b1:central-forces});
na superfície de uma esfera de raio $R$, $g = GM/R^2$. Note a unidade:
\unit{N/kg} e \unit{m/s^2} são a mesma coisa, e é por isso que todos os corpos caem
do mesmo jeito.
\end{proposition}

\begin{proof}
A lei da gravitação de Newton é admitida e explorada no
\cref{ch:b1:central-forces}; $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$.
\end{proof}

\begin{proposition}[Mola; fio]\label{prop:b1:newton-dynamics:springstring}
\index{lei de Hooke}\index{mola}\index{tensão}
Uma mola ideal de constante elástica $k$ e comprimento natural $\ell_0$, esticada
até o comprimento $\ell$ ao longo do vetor unitário $\vect u$ que aponta da sua extremidade
fixa para o corpo, exerce sobre o corpo a \emph{\omterm{def:b1:newton-dynamics:momentum}{força} restauradora}
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{lei de Hooke}),
\]
rumo ao seu comprimento natural. Um fio ideal (sem \omterm{def:b1:newton-dynamics:momentum}{massa}, inextensível)
exerce uma \emph{tensão} $\vect T$ ao longo de si mesmo, puxando, de mesmo
módulo nas duas extremidades e em todo o seu comprimento (mesmo passando por uma polia
sem \omterm{def:b1:newton-dynamics:momentum}{massa} e sem atrito).
\end{proposition}

\begin{proof}
A \omterm{prop:b1:newton-dynamics:springstring}{lei de Hooke} é um modelo empírico válido para pequenas deformações. O
fio: um elemento sem \omterm{def:b1:newton-dynamics:momentum}{massa} obedece a $0 = \vect T_1 + \vect T_2$ pela
segunda lei de Newton, de modo que a tensão é transmitida sem alteração.
\end{proof}

\begin{proposition}[Forças de contato: reação normal e atrito seco]\label{prop:b1:newton-dynamics:friction}
\index{reação normal}\index{atrito seco}\index{leis de Coulomb do atrito}\index{coeficiente de atrito}
Uma superfície sólida exerce sobre um corpo que a toca uma \emph{\omterm{prop:b1:newton-dynamics:friction}{reação normal}}
$\vect N$, perpendicular à superfície e repulsiva ($N \geq 0$: uma
superfície não pode puxar), e uma \emph{\omterm{def:b1:newton-dynamics:momentum}{força} de atrito} tangencial $\vect T$.
Leis de Coulomb: se o corpo não desliza, $\abs{\vect T} \leq \mu_s N$
(o atrito \emph{estático} se ajusta ao que for necessário, até esse limite);
se ele desliza com velocidade $\vect v_g$ em relação à superfície,
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (atrito \emph{cinético},
que se opõe ao deslizamento), com $\mu_d \leq \mu_s$ coeficientes da ordem de
$0.1$ a $1$ conforme os materiais, independentes da área de
contato e da velocidade.
\end{proposition}
\admitted

\begin{proposition}[Atrito fluido]\label{prop:b1:newton-dynamics:drag}
\index{atrito fluido}\index{arrasto}\index{lei de Stokes}
Um corpo que se move com velocidade $\vect v$ através de um fluido em repouso sente um
\emph{arrasto} oposto a $\vect v$: em baixa velocidade (corpo pequeno, fluido
viscoso) ele é linear, $\vect F = -\alpha\vect v$ --- para uma esfera de
raio $r$ num fluido de viscosidade $\eta$, $\alpha = 6\pi\eta r$
(Stokes); em alta velocidade (corpos grandes e rápidos no ar ou na água) ele é
quadrático,
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
com $\rho$ a densidade do fluido, $S$ a seção do corpo voltada para
o escoamento e $C_x$ um \emph{coeficiente de arrasto} \omterm{def:b1:units-dimensions:dimension}{adimensional} ($0.5$ para uma
esfera, $0.3$ para um carro, $1$ para um disco plano). O número de Reynolds
$\rho vL/\eta$ (\cref{rem:b1:units-dimensions:pi}) decide: linear
abaixo de $1$, quadrático acima de cerca de $1000$.
\end{proposition}
\admitted

\begin{remark}[Outras forças]\label{rem:b1:newton-dynamics:others}
O empuxo (Arquimedes) é uma resultante de \omterm{def:b1:newton-dynamics:momentum}{forças} de pressão, deduzida no
\cref{ch:b1:fluid-statics}; as \omterm{def:b1:newton-dynamics:momentum}{forças} elétrica e magnética sobre uma
carga são assunto do \cref{ch:b1:charged-particles}. Todas as
\omterm{def:b1:newton-dynamics:momentum}{forças} deste capítulo são \emph{modelos} de interações de contato ou de
campo, válidos dentro de limites declarados --- a mola de Hooke arrebenta,
o atrito de Coulomb falha sob alta pressão, o arrasto de Stokes falha em alta
velocidade. Parte de toda solução é verificar que o modelo usado se aplica.
\end{remark}

\section{Resolvendo um problema de dinâmica}

\begin{method}[Da situação à equação]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{Sistema}: nomeie o corpo tratado como um ponto.
  \item \emph{\omterm{def:b1:point-kinematics:frame}{Referencial}}: nomeie-o e verifique que é inercial (ou trate-o
        como tal com a precisão exigida).
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{Forças}}: liste todo corpo que toca ou atrai o
        sistema e desenhe cada \omterm{def:b1:newton-dynamics:momentum}{força} num esquema (o diagrama de
        \omterm{def:b1:newton-dynamics:momentum}{forças}); nada mais entra.
  \item \emph{Coordenadas}: escolha eixos adaptados ao movimento
        (\cref{met:b1:point-kinematics:choose}) e exprima $\vect a$.
  \item \emph{Projete} $m\vect a = \sum\vect F$ nos eixos; use os
        vínculos (um corpo sobre uma superfície permanece nela: $a_\perp = 0$; um
        fio de comprimento fixo: velocidades iguais nas duas extremidades).
  \item \emph{Resolva} e \emph{verifique}: dimensões, casos-limite,
        sinais ($N \geq 0$, $\abs T \leq \mu_sN$ se em repouso).
\end{enumerate}
\end{method}

\begin{example}[Bloco num plano inclinado]\label{ex:b1:newton-dynamics:incline}
Bloco de \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ num plano inclinado de $\alpha$, com coeficientes $\mu_s$,
$\mu_d$. Eixos: $x$ descendo a rampa, $y$ ao longo da normal. Peso
$(mg\sin\alpha, -mg\cos\alpha)$, reação $(0, N)$, atrito $(-T, 0)$
se ele desliza para baixo. Ao longo de $y$: $N = mg\cos\alpha$. Em repouso, ao longo de $x$:
$T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$, isto é,\ $\tan\alpha
\leq \mu_s$: o bloco se sustenta até o ângulo $\arctan\mu_s$ --- uma
medida de $\mu_s$ que dispensa balança. Deslizando: $ma = mg\sin\alpha -
\mu_dmg\cos\alpha$, logo $a = g(\sin\alpha - \mu_d\cos\alpha)$, independente
de $m$: \qty{3.2}{m/s^2} para $\alpha = 30^\circ$, $\mu_d = 0.2$.
\end{example}

\begin{example}[O pêndulo simples]\label{ex:b1:newton-dynamics:pendulum}
\index{pêndulo simples}
\omterm{def:b1:newton-dynamics:momentum}{Massa} $m$ num fio de comprimento $\ell$, com ângulo $\theta$ em relação à vertical
descendente. \omterm{def:b1:point-kinematics:cylindrical}{Coordenadas polares} no ponto de suspensão: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$; \omterm{def:b1:newton-dynamics:momentum}{forças}:
peso $mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$ e
tensão $-T\vect e_r$. Ao longo de $\vect e_\theta$: $m\ell\ddot\theta =
-mg\sin\theta$,
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
que para pequenos ângulos ($\sin\theta \approx \theta$) é a equação
harmônica com $\omega_0 = \sqrt{g/\ell}$: período $T_0 = 2\pi\sqrt{\ell/g}$,
o $k = 2\pi$ que o \cref{ex:b1:units-dimensions:pendulum} não pôde
fornecer. Ao longo de $\vect e_r$: $T = mg\cos\theta + m\ell\dot\theta^2$ --- o
fio puxa com mais \omterm{def:b1:newton-dynamics:momentum}{força} embaixo, onde a velocidade é maior.
\end{example}

\begin{proposition}[Queda com arrasto; velocidade limite]\label{prop:b1:newton-dynamics:terminal}
\index{velocidade limite}
Um corpo de \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ que cai do repouso sob o seu peso e uma \omterm{def:b1:newton-dynamics:momentum}{força} de arrasto
atinge uma \emph{\omterm{prop:b1:newton-dynamics:terminal}{velocidade limite}} em que os dois se equilibram. Arrasto
linear: $m\dot v = mg - \alpha v$, $v = v_\infty(1 - \eu^{-t/\tau})$ com
$v_\infty = mg/\alpha$, $\tau = m/\alpha$. Arrasto quadrático: $m\dot v = mg -
\tfrac12\rho C_xSv^2$, $v = v_\infty\tanh(t/\tau)$ com
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
Linear: uma equação de primeira ordem, $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder}). Quadrático: com
$u = v/v_\infty$, $\dot u = (1 - u^2)/\tau$; separando as variáveis,
$\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$ (a fração
racional $1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$ se integra em
$\tfrac12\ln\frac{1 + u}{1 - u}$), logo $u = \tanh(t/\tau)$.
\end{proof}

\begin{example}[Duas velocidades limites]\label{ex:b1:newton-dynamics:terminal}
Um paraquedista, $m = \qty{80}{kg}$, de barriga para baixo, $C_xS \approx \qty{0.8}{m^2}$
no ar ($\rho = \qty{1.2}{kg/m^3}$): $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$, alcançada (a $90\%$) em cerca de $1.5\tau =
\qty{6}{s}$. Uma gotícula de neblina, $r = \qty{10}{\micro m}$, em regime de Stokes
($\eta_{\text{ar}} = \qty{1.8e-5}{Pa.s}$): $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- e é por isso que a neblina fica suspensa no ar.
\end{example}

\section{Movimento num círculo: forças de vínculo}

\begin{proposition}[Exigência centrípeta]\label{prop:b1:newton-dynamics:centripetal}
\index{força centrípeta}
Um \omterm{def:b1:point-kinematics:frame}{ponto material} que se move num círculo de raio $R$ com velocidade $v$ deve receber
da resultante das \omterm{def:b1:newton-dynamics:momentum}{forças} uma componente $mv^2/R$ dirigida ao
centro. O que quer que a forneça --- a tensão de um fio, o atrito, a
reação de uma pista sobrelevada, a gravidade --- fixa a velocidade máxima ou a
velocidade mínima com que o movimento é possível.
\end{proposition}

\begin{proof}
Segunda lei de Newton com a aceleração de Frenet: a componente normal
de $\sum\vect F$ vale $mv^2/\rho$.
\end{proof}

\begin{example}[Três círculos]\label{ex:b1:newton-dynamics:circles}
\emph{Curva plana}: o atrito deve fornecer $mv^2/R \leq \mu_sN = \mu_smg$,
logo $v \leq \sqrt{\mu_sgR}$: \qty{20}{m/s} para $R = \qty{50}{m}$,
$\mu_s = 0.8$. \emph{Curva sobrelevada} de ângulo $\beta$, sem atrito: a
reação $\vect N$ é normal à pista; na vertical $N\cos\beta = mg$,
na horizontal $N\sin\beta = mv^2/R$, logo $v^2 = gR\tan\beta$ --- a única
velocidade com que a curva é feita sem atrito algum. \emph{Loop}: no
alto, com $\vect N$ apontando para baixo, rumo ao centro, $N + mg =
mv^2/R$, e $N \geq 0$ exige $v \geq \sqrt{gR}$: abaixo disso, o carro
deixa a pista (o \cref{pb:b1:point-kinematics:1} de novo, agora explicado).
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:1}
Uma pessoa de \omterm{def:b1:newton-dynamics:momentum}{massa} \qty{70}{kg} está sobre uma balança num elevador. O que
a balança indica (em newtons e em ``quilogramas'') quando o elevador
acelera para cima a \qty{2.0}{m/s^2}, para baixo a \qty{2.0}{m/s^2},
move-se com velocidade constante e cai livremente?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:2}
Um bloco $m_1 = \qty{2.0}{kg}$ sobre uma mesa sem atrito é ligado por um
fio que passa por uma polia sem atrito a uma \omterm{def:b1:newton-dynamics:momentum}{massa} pendurada $m_2 = \qty{1.0}{kg}$.
Encontre a aceleração e a tensão.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:3}
Um bloco desliza por um plano inclinado de $30^\circ$ com $\mu_d = 0.20$. Calcule
a sua aceleração e a sua velocidade após \qty{5.0}{m} a partir do repouso. A
resposta depende da sua \omterm{def:b1:newton-dynamics:momentum}{massa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:4}
Uma \omterm{def:b1:newton-dynamics:momentum}{massa} de \qty{0.50}{kg} pende de uma mola de constante elástica \qty{200}{N/m}.
Encontre o alongamento estático; depois a \omterm{def:b1:wave-propagation:signal}{frequência angular} e o período das
suas oscilações verticais em torno do equilíbrio (mostre que a equação do
movimento é harmônica).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:5}
Uma caixa está sobre uma plataforma inclinável; ela começa a deslizar a $22^\circ$.
Encontre $\mu_s$. Um carro numa pista plana com $\mu_s = 0.80$: velocidade máxima
numa curva de raio \qty{50}{m}, e numa pista molhada com $\mu_s = 0.40$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:6}
\omterm{prop:b1:newton-dynamics:terminal}{Velocidades limites}: um paraquedista ($m = \qty{80}{kg}$, $C_xS = \qty{0.70}{m^2}$,
$\rho = \qty{1.2}{kg/m^3}$); uma gotícula de neblina de raio \qty{10}{\micro m}
no ar ($\eta = \qty{1.8e-5}{Pa.s}$), usando a \omterm{prop:b1:newton-dynamics:drag}{lei de Stokes}. Verifique com um
número de Reynolds que cada regime é o correto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:7}
Um pêndulo de comprimento $\ell$ é solto do repouso em $\theta_0 = 60^\circ$.
Dado que a sua velocidade embaixo vale $\sqrt{2g\ell(1 - \cos\theta_0)}$
(energia, próximo capítulo), encontre a tensão do fio embaixo, em unidades
de $mg$, e no ponto de soltura.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:8}
Uma curva de raio \qty{200}{m} é sobrelevada de $15^\circ$. Com que velocidade ela
pode ser feita sem atrito algum? O que o atrito deve fazer em velocidades menores e
maiores?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:9}
Um pêndulo cônico: uma \omterm{def:b1:newton-dynamics:momentum}{massa} num fio de comprimento $\ell = \qty{1.0}{m}$
descreve um círculo horizontal, com o fio formando $30^\circ$ com a
vertical. Encontre a \omterm{cor:b1:point-kinematics:circular}{velocidade angular}, o período e a tensão (em
unidades de $mg$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:10}
Um carro de \qty{1000}{kg} atinge \qty{100}{km/h} a partir do repouso em \qty{8.0}{s}
com aceleração constante. Encontre a \omterm{def:b1:newton-dynamics:momentum}{força} resultante e o \omterm{prop:b1:newton-dynamics:friction}{coeficiente de
atrito} mínimo entre os pneus de tração e a pista, supondo que as rodas de
tração suportem todo o peso. O que limita um carro esportivo numa pista molhada?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:11}
Um corpo cai do repouso com arrasto linear, $\tau = m/\alpha = \qty{0.50}{s}$.
Escreva $v(t)$ e deduza a distância percorrida $z(t)$; calcule $v_\infty$
e $z$ após \qty{2.0}{s}; compare com a queda livre.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:12}
Um carro de \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ percorre o interior de um loop circular vertical de
raio $R$ (\cref{pb:b1:point-kinematics:1}). Exprima a \omterm{prop:b1:newton-dynamics:friction}{reação normal}
da pista em função do ângulo $\theta$ medido a partir do ponto mais baixo e
da velocidade $v(\theta)$; deduza a velocidade mínima no alto; usando
$v^2 = v_0^2 - 2gR(1 - \cos\theta)$, encontre a velocidade mínima de entrada e
a reação embaixo nesse caso.
\end{exercise}

\section{Problema: o paraquedista}

\begin{problem}[{Problema de fim de semana --- da porta do avião até o
solo: queda livre, o ar que empurra de volta, um velame que se abre depressa
demais e a aterrissagem --- uma lei do movimento, duas leis de arrasto}]
\label{pb:b1:newton-dynamics:1}
Um paraquedista com equipamento tem \omterm{def:b1:newton-dynamics:momentum}{massa} $m = \qty{80}{kg}$. Ar:
$\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$. Caindo de barriga
para baixo, $C_xS = \qty{0.80}{m^2}$; sob o velame aberto, $C_xS =
\qty{25}{m^2}$. $g = \qty{9.81}{m/s^2}$; $z$ é contado para baixo a partir
do ponto de salto.

\medskip
\noindent\textbf{Parte I --- Sem ar.}
\begin{enumerate}
  \item Escreva a segunda lei de Newton e dê $v(t)$ e $z(t)$; calcule
        ambos em $t = \qty{10}{s}$.
  \item Calcule o peso.
  \item O paraquedista se sente ``sem peso'' durante essa fase, embora
        a gravidade aja plenamente. Explique pelas \omterm{def:b1:newton-dynamics:momentum}{forças} (o que falta
        em comparação com estar de pé no solo).
  \item O \omterm{def:b1:point-kinematics:frame}{referencial} da Terra é suficientemente inercial para este problema? Nomeie o que
        está sendo desprezado.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Caindo através do ar.}
\begin{enumerate}[resume]
  \item Escreva a \omterm{def:b1:newton-dynamics:momentum}{força} de arrasto na queda de barriga para baixo em função de $v$.
  \item Escreva a equação de movimento para $v(t)$.
  \item Encontre a \omterm{prop:b1:newton-dynamics:terminal}{velocidade limite} $v_\infty$ (em \unit{m/s} e em
        \unit{km/h}).
  \item Com $u = v/v_\infty$ e $\tau = v_\infty/g$, mostre que
        $\dd u/\dd t = (1 - u^2)/\tau$; calcule $\tau$.
  \item Resolva por separação de variáveis e mostre que
        $v = v_\infty\tanh(t/\tau)$.
  \item Após quanto tempo $v = 0.9v_\infty$? E $0.99v_\infty$?
  \item Mostre que $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ e calcule a
        distância percorrida em $t = \qty{10}{s}$; compare com a Parte I.
  \item Estime o número de Reynolds em $v_\infty$ com $L = \qty{0.5}{m}$
        e justifique a lei quadrática.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O velame se abre.}
\begin{enumerate}[resume]
  \item Calcule a nova \omterm{prop:b1:newton-dynamics:terminal}{velocidade limite} sob o velame aberto.
  \item Se o velame abrisse instantaneamente a $v = \qty{40}{m/s}$, qual
        seria a desaceleração, em \unit{m/s^2} e em $g$? Por que
        isso é inaceitável?
  \item Os velames reais se abrem em cerca de \qty{3}{s}. Estime a
        desaceleração média e a \omterm{def:b1:newton-dynamics:momentum}{força} média do arnês sobre o
        paraquedista.
  \item Uma vez plenamente aberto o velame, a velocidade decai rumo à nova
        $v_\infty$: qual é a nova \omterm{def:b1:transient-regimes:firstorder}{constante de tempo} $\tau' = v_\infty'/g$,
        e o que ela significa para a aproximação à descida estacionária?
  \item Na descida estacionária, qual é a \omterm{def:b1:newton-dynamics:momentum}{força} do arnês? Justifique com a
        segunda lei.
  \item Explique por que a \omterm{prop:b1:newton-dynamics:terminal}{velocidade limite} varia como $1/\sqrt{C_xS}$: por
        qual fator um velame de metade da área a aumentaria?
  \item Um saltador mais pesado (\qty{100}{kg}) sob o mesmo velame: qual a \omterm{prop:b1:newton-dynamics:terminal}{velocidade
        limite}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A aterrissagem.}
\begin{enumerate}[resume]
  \item A velocidade de aterrissagem é a $v_\infty$ do velame. De que altura
        seria preciso saltar, sem ar, para chegar ao solo com essa
        velocidade?
  \item Uma aterrissagem de pernas rígidas para o corpo em \qty{0.5}{m}; um rolamento,
        em \qty{1.5}{m}. Estime a desaceleração média (em $g$) e
        a \omterm{def:b1:newton-dynamics:momentum}{força} média sobre as pernas em cada caso, e conclua.
  \item Pouco antes de tocar o solo o paraquedista puxa os dois comandos,
        o que aumenta brevemente o arrasto do velame. Explique qualitativamente
        o efeito sobre a velocidade.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Coisas pequenas caem de outro jeito.}
\begin{enumerate}[resume]
  \item Uma gota de chuva de raio \qty{1.0}{mm} ($C_x = 0.5$): calcule a sua
        \omterm{prop:b1:newton-dynamics:terminal}{velocidade limite} pela lei quadrática e verifique o número de
        Reynolds.
  \item Uma gotícula de neblina de raio \qty{10}{\micro m}: use a \omterm{prop:b1:newton-dynamics:drag}{lei de Stokes}
        para $v_\infty$ e verifique que o número de Reynolds é de fato
        pequeno.
  \item Resuma: uma equação de movimento, dois regimes de arrasto e os
        três números (\unit{m/s}) que decidem se algo cai
        como uma pedra, como uma folha ou não cai.
\end{enumerate}
\end{problem}
