\chapter{Espelhos e lentes delgadas}\label{ch:b1:mirrors-thin-lenses}

Um espelho de maquiagem mostra um rosto com o dobro do tamanho; o dorso de
uma colher o mostra minúsculo e direito, a concavidade o mostra minúsculo e
de cabeça para baixo; uma lupa segurada com o braço estendido vira a sala ao
contrário e, aproximada, aumenta um selo; a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} de um celular, do tamanho
de uma lentilha, pinta a rua inteira sobre um sensor. Uma única relação
algébrica entre três distâncias governa todas essas imagens, e este capítulo
a estabelece --- primeiro para os espelhos curvos, depois para a \omterm{def:b1:mirrors-thin-lenses:lens}{lente
delgada} com que quase todos os instrumentos ópticos são construídos.

\section{Imagens e condições de Gauss}

\begin{definition}[Objeto, imagem, estigmatismo]\label{def:b1:mirrors-thin-lenses:image}
\index{imagem (óptica)}\index{objeto (óptica)}\index{estigmatismo}\index{imagem real}\index{imagem virtual}
Um sistema óptico recebe raios vindos de um ponto $A$ (o \emph{ponto
objeto}). Se todos os raios emergentes passam por um único ponto $A'$, ou todos
parecem vir de um único ponto $A'$, o sistema é \emph{estigmático} para
$A$, e $A'$ é a \emph{imagem} de $A$. A imagem é \emph{real} se os
raios emergentes de fato se cruzam em $A'$ (ela pode ser recolhida num anteparo),
e \emph{virtual} se apenas os seus prolongamentos se cruzam (ela é vista através
do sistema, não projetada). Do mesmo modo, o objeto é real se os raios incidentes
divergem de $A$, e virtual se convergem para $A$ depois da entrada do
sistema. O par $(A, A')$ é \emph{conjugado}.
\end{definition}

\begin{definition}[Sistema centrado; condições de Gauss]\label{def:b1:mirrors-thin-lenses:gauss}
\index{eixo óptico}\index{condições de Gauss}\index{raio paraxial}
Um \emph{sistema centrado} tem um eixo de simetria de revolução, o
\emph{eixo óptico}. Os raios que ficam próximos do eixo e fazem ângulos
pequenos com ele são \emph{paraxiais}; diz-se que um sistema usado apenas com
raios paraxiais trabalha nas \emph{condições de Gauss}. Nessas
condições um sistema centrado é aproximadamente estigmático para todo
ponto, e um plano-objeto perpendicular ao eixo tem um \omterm{def:b1:mirrors-thin-lenses:image}{plano-imagem}
perpendicular ao eixo (\emph{aplanetismo}).
\end{definition}
\admitted

\begin{remark}[Por que ângulos pequenos]\label{rem:b1:mirrors-thin-lenses:why}
O \omterm{prop:b1:rays-reflection-refraction:mirror}{espelho plano} é o único sistema simples rigorosamente
estigmático para todo ponto. Um espelho ou uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente} esférica é estigmático
apenas na medida em que $\sin\theta \approx \tan\theta \approx \theta$:
é essa aproximação que transforma a \omterm{thm:b1:rays-reflection-refraction:snell}{lei de Snell} em relações lineares
entre distâncias. Fora dela, a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} de um ponto é uma mancha
(\emph{aberrações}), e o trabalho do projetista --- diafragmas, combinações de
\omterm{def:b1:mirrors-thin-lenses:lens}{lentes}, superfícies asféricas --- é empurrar essa mancha para baixo do que o olho ou
o sensor conseguem resolver.
\end{remark}

\begin{notation}[Medidas algébricas e aumento]\label{not:b1:mirrors-thin-lenses:signs}
\index{medida algébrica}\index{aumento transversal}
O \omterm{def:b1:mirrors-thin-lenses:gauss}{eixo óptico} é orientado no sentido da luz incidente;
a direção transversal é orientada para cima. Para pontos $P$, $Q$ do
eixo, $\overline{PQ}$ designa a \omterm{rem:b1:mirrors-thin-lenses:why}{medida algébrica} ($\overline{PQ} > 0$
se $Q$ está a jusante de $P$); para um objeto $AB$ perpendicular ao
eixo e a sua \omterm{def:b1:mirrors-thin-lenses:image}{imagem} $A'B'$, o \emph{\omterm{rem:b1:mirrors-thin-lenses:why}{aumento transversal}} é
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}} ,
\]
negativo para uma \omterm{def:b1:mirrors-thin-lenses:image}{imagem} invertida, de valor absoluto maior que $1$ para uma
\omterm{def:b1:mirrors-thin-lenses:image}{imagem} ampliada.
\end{notation}

\section{Espelhos esféricos}

\begin{definition}[Espelho esférico]\label{def:b1:mirrors-thin-lenses:mirror}
\index{espelho esférico}\index{espelho côncavo}\index{espelho convexo}\index{foco}
Um \emph{espelho esférico} é uma calota esférica refletora de centro $C$
e vértice $S$ (a sua interseção com o eixo). Ele é \emph{côncavo}
se a luz encontra o lado oco ($C$ do lado da luz
incidente), e \emph{convexo} em caso contrário. O seu \emph{foco} $F$ é a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} de um
ponto do infinito sobre o eixo; a sua \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} é $f = \overline{SF}$.
\end{definition}

\begin{proposition}[Foco de um espelho esférico]\label{prop:b1:mirrors-thin-lenses:focus}
Nas \omterm{def:b1:mirrors-thin-lenses:gauss}{condições de Gauss}, o \omterm{def:b1:mirrors-thin-lenses:mirror}{foco} é o ponto médio de $SC$:
$\overline{SF} = \overline{SC}/2$. Um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho côncavo} tem $f < 0$ (\omterm{def:b1:mirrors-thin-lenses:mirror}{foco}
real, à frente); um convexo tem $f > 0$ (\omterm{def:b1:mirrors-thin-lenses:mirror}{foco} virtual, atrás).
\end{proposition}

\begin{proof}
Um raio paralelo ao eixo, à altura $h$, encontra o espelho em $I$; a
normal em $I$ é o raio $CI$, que faz um ângulo $\alpha$ com o
eixo, $\sin\alpha = h/R$. Refletido com o mesmo ângulo $\alpha$ em relação à
normal, o raio cruza o eixo em $F$; o triângulo $CIF$ tem ângulos
iguais $\alpha$ em $C$ e em $I$, de modo que $CF = IF$ e, projetando,
$CF = R/(2\cos\alpha)$. Para raios paraxiais $\cos\alpha \to 1$ e
$CF = R/2$: todos esses raios cruzam o eixo no mesmo ponto, o
ponto médio de $SC$. Sinal: para um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho côncavo}, $C$ e $F$ ficam à frente
(a montante) e $\overline{SC} < 0$.
\end{proof}

\begin{theorem}[Conjugação de um espelho esférico]\label{thm:b1:mirrors-thin-lenses:mirrorconj}
\index{relação de conjugação!espelho esférico}
Nas \omterm{def:b1:mirrors-thin-lenses:gauss}{condições de Gauss}, a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} $A'$ de um ponto $A$ do eixo
satisfaz
\[
\frac{1}{\overline{SA'}} + \frac{1}{\overline{SA}} = \frac{2}{\overline{SC}}
= \frac{1}{\overline{SF}},
\qquad
\gamma = -\frac{\overline{SA'}}{\overline{SA}} .
\]
\end{theorem}

\begin{proof}
Tome um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho côncavo} e um objeto real $A$ além de $C$; a forma
algébrica cobre então todos os casos. Um raio que parte de $A$ atinge o espelho em $I$
(altura $h$) e volta a $A'$ sobre o eixo. Sejam $u$, $u'$, $\omega$
os ângulos de $AI$, $A'I$ e $CI$ com o eixo; paraxialmente
$u = h/AS$, $u' = h/A'S$, $\omega = h/CS$ (distâncias). A lei da
reflexão diz que a normal $CI$ bissecta o ângulo entre $AI$ e
$A'I$: $\omega - u = u' - \omega$, isto é,\ $u + u' = 2\omega$, isto é,\
$1/AS + 1/A'S = 2/CS$; com $\overline{SA}$, $\overline{SA'}$,
$\overline{SC}$ todos negativos, essa é a relação exibida. Quanto ao
aumento, o raio $B \to S \to B'$ refletido no vértice faz
ângulos iguais com o eixo: $A'B'/SA' = AB/SA$ em comprimentos, e a \omterm{def:b1:mirrors-thin-lenses:image}{imagem}
é invertida, $\gamma = -\overline{SA'}/\overline{SA}$.
\end{proof}

\begin{method}[Construindo uma imagem num espelho]\label{met:b1:mirrors-thin-lenses:mirrorrays}
Para um ponto-objeto $B$ fora do eixo, trace dois dos quatro raios notáveis;
a sua interseção (ou a dos seus prolongamentos) é $B'$:
\begin{enumerate}
  \item paralelo ao eixo $\to$ refletido passando por $F$;
  \item passando por $F$ $\to$ refletido paralelo ao eixo;
  \item passando por $C$ $\to$ refletido sobre si mesmo;
  \item dirigido ao vértice $S$ $\to$ refletido simetricamente em relação ao eixo.
\end{enumerate}
Então $A'$ é o pé da perpendicular baixada de $B'$ ao eixo.
\end{method}

\begin{example}[Espelho de barbear, espelho retrovisor]\label{ex:b1:mirrors-thin-lenses:mirrors}
Um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho côncavo} de raio \qty{60}{cm} ($f = \qty{-30}{cm}$) com um
rosto a \qty{20}{cm}: $1/\overline{SA'} = -1/30 + 1/20 = 1/60$,
$\overline{SA'} = \qty{+60}{cm}$ --- atrás do espelho, virtual ---
e $\gamma = -60/(-20) = +3$: direita, três vezes maior, o espelho de
barbear. Um \omterm{def:b1:mirrors-thin-lenses:mirror}{espelho convexo} de raio \qty{2.0}{m} ($f = \qty{+1.0}{m}$)
com um carro \qty{20}{m} atrás: $1/\overline{SA'} = 1 + 1/20$,
$\overline{SA'} = \qty{0.95}{m}$ (virtual, atrás), $\gamma = +0.048$:
direita, vinte vezes menor e, portanto, com campo amplo --- o
espelho retrovisor, com o seu aviso de que os objetos estão mais perto do que
parecem.
\end{example}

\section{Lentes delgadas}

\begin{definition}[Lente delgada]\label{def:b1:mirrors-thin-lenses:lens}
\index{lente delgada}\index{lente convergente}\index{lente divergente}\index{centro óptico}\index{distância focal}\index{vergência}
Uma \emph{lente} é um \omterm{def:b1:rays-reflection-refraction:index}{meio transparente} limitado por duas superfícies esféricas (ou
planas); ela é \emph{delgada} quando a sua espessura é desprezível
frente aos raios de curvatura e às distâncias em jogo, de modo que os dois
vértices se confundem num só ponto, o \emph{centro óptico} $O$. Um raio
que passa por $O$ não é desviado. O \emph{\omterm{def:b1:mirrors-thin-lenses:mirror}{foco} \omterm{def:b1:mirrors-thin-lenses:image}{imagem}} $F'$ é a \omterm{def:b1:mirrors-thin-lenses:image}{imagem}
de um ponto do infinito sobre o eixo; o \emph{\omterm{def:b1:mirrors-thin-lenses:mirror}{foco} objeto} $F$ é o
ponto cuja \omterm{def:b1:mirrors-thin-lenses:image}{imagem} está no infinito; ambos estão à mesma distância de
$O$: $\overline{OF'} = -\overline{OF} = f'$, a \emph{distância focal}.
A lente é \emph{convergente} se $f' > 0$ ($F'$ real, a jusante; bordas
mais finas que o centro), e \emph{divergente} se $f' < 0$. A sua
\emph{vergência} $V = 1/f'$ é medida em \emph{dioptrias}
($\qty{1}{\delta} = \qty{1}{m^{-1}}$).
\end{definition}

\begin{theorem}[Conjugação de uma lente delgada]\label{thm:b1:mirrors-thin-lenses:lensconj}
\index{relação de conjugação!lente delgada}\index{relação de Descartes}\index{relação de Newton}
Nas \omterm{def:b1:mirrors-thin-lenses:gauss}{condições de Gauss}, a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} $A'$ de um ponto $A$ do eixo
satisfaz a \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relação de Descartes}
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'},
\qquad
\gamma = \frac{\overline{OA'}}{\overline{OA}},
\]
e, de modo equivalente, a \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relação de Newton}
\[
\overline{FA}\cdot\overline{F'A'} = -f'^2,
\qquad
\gamma = \frac{f'}{\overline{FA}} = -\frac{\overline{F'A'}}{f'} .
\]
\end{theorem}

\begin{proof}[Demonstração parcial]
O fato central é que uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente delgada} desvia um raio que a atravessa à altura
$h$ do ângulo $h/f'$ em direção ao eixo, qualquer que seja a direção do raio
--- cada zona da \omterm{def:b1:mirrors-thin-lenses:lens}{lente} age como um prisma delgado (\cref{rem:b1:rays-reflection-refraction:thin})
cujo ângulo de refringência cresce linearmente com $h$ para superfícies esféricas; essa
linearidade, exata para raios paraxiais, é admitida aqui e deduzida da
\omterm{thm:b1:rays-reflection-refraction:snell}{lei de Snell} em duas superfícies esféricas no volume do segundo ano. Admitindo-a:
um raio que parte de $A$ (à distância $p = -\overline{OA} > 0$ para um objeto
real) e chega à \omterm{def:b1:mirrors-thin-lenses:lens}{lente} à altura $h$ tem inclinação $h/p$, é
desviado de $h/f'$ e sai com inclinação $h/p - h/f'$; ele cruza o eixo
em $A'$ com $h/\overline{OA'} = h/f' - h/p$, isto é,\ $1/\overline{OA'}
- 1/\overline{OA} = 1/f'$, independentemente de $h$: todos os raios vindos de $A$ se encontram
em $A'$. O raio $B \to O \to B'$ que passa pelo centro é retilíneo, de modo que
$\overline{A'B'}/\overline{AB} = \overline{OA'}/\overline{OA}$. A forma de Newton
segue substituindo $\overline{OA} = \overline{OF} + \overline{FA}
= -f' + \overline{FA}$ e $\overline{OA'} = f' + \overline{F'A'}$.
\end{proof}

\begin{method}[Construindo uma imagem através de uma lente delgada]\label{met:b1:mirrors-thin-lenses:lensrays}
Três raios notáveis a partir de $B$:
\begin{enumerate}
  \item passando por $O$: não desviado;
  \item paralelo ao eixo: emerge passando por $F'$;
  \item passando por $F$: emerge paralelo ao eixo.
\end{enumerate}
Dois bastam; para uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente divergente}, use os prolongamentos (o raio dirigido a
$F$ atrás da \omterm{def:b1:mirrors-thin-lenses:lens}{lente} emerge paralelo; o raio paralelo emerge como se viesse
de $F'$, à frente).
\end{method}

\begin{example}[A mesma lente, três usos]\label{ex:b1:mirrors-thin-lenses:lensuses}
Uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente convergente}, $f' = \qty{5.0}{cm}$. Selo a \qty{3.0}{cm}:
$1/\overline{OA'} = 1/5 - 1/3 = -2/15$, $\overline{OA'} = \qty{-7.5}{cm}$,
virtual, $\gamma = +2.5$: a lupa. Lâmpada a \qty{20}{cm}:
$\overline{OA'} = \qty{6.7}{cm}$, real, $\gamma = -1/3$: uma \omterm{def:b1:mirrors-thin-lenses:image}{imagem}
invertida e reduzida sobre um cartão. Janela a \qty{4}{m}: $\overline{OA'}
\approx f'$, \omterm{def:b1:mirrors-thin-lenses:image}{imagem} invertida minúscula no plano focal --- a câmera.
Três usos, uma relação.
\end{example}

\begin{proposition}[Aumento longitudinal]\label{prop:b1:mirrors-thin-lenses:longitudinal}
\index{aumento longitudinal}
Se o objeto se desloca ao longo do eixo de $\dd\overline{OA}$, a \omterm{def:b1:mirrors-thin-lenses:image}{imagem}
se desloca de $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$, no mesmo
sentido.
\end{proposition}

\begin{proof}
Derive a \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relação de Descartes} com $f'$ fixo:
$-\dd\overline{OA'}/\overline{OA'}^2 + \dd\overline{OA}/\overline{OA}^2 = 0$,
de modo que $\dd\overline{OA'} = (\overline{OA'}/\overline{OA})^2\,\dd\overline{OA}$.
\end{proof}

\begin{remark}[Tolerância de foco]\label{rem:b1:mirrors-thin-lenses:depth}
Um projetor com $\gamma = -50$ tem $\gamma^2 = 2500$: um slide que
se deforma \qty{0.1}{mm} sob o calor da lâmpada joga a sua \omterm{def:b1:mirrors-thin-lenses:image}{imagem}
\qty{25}{cm} para fora do plano da tela. Inversamente, uma câmera ($\abs\gamma
\ll 1$) tolera grandes deslocamentos do objeto para um \omterm{prop:b1:rays-reflection-refraction:prism}{desvio mínimo} da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} ---
é a profundidade de campo.
\end{remark}

\section{Medindo uma distância focal}

\begin{method}[Autocolimação]\label{met:b1:mirrors-thin-lenses:autocollimation}
\index{autocolimação}
Coloque um \omterm{prop:b1:rays-reflection-refraction:mirror}{espelho plano} logo atrás da \omterm{def:b1:mirrors-thin-lenses:lens}{lente} e desloque um objeto iluminado (uma
cruz num vidro fosco) ao longo do eixo até que a sua \omterm{def:b1:mirrors-thin-lenses:image}{imagem}, refletida de volta
através da \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, se forme nítida sobre o próprio objeto, invertida e do
mesmo tamanho. O objeto está então no plano focal: os raios saem da
\omterm{def:b1:mirrors-thin-lenses:lens}{lente} paralelos, voltam paralelos do espelho e são refocalizados no
plano focal. A distância objeto--lente vale $f'$.
\end{method}

\begin{proposition}[Método de Bessel]\label{prop:b1:mirrors-thin-lenses:bessel}
\index{método de Bessel}\index{método de Silbermann}
Um objeto e um anteparo estão fixos a uma distância $D$ um do outro. Uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente
convergente} forma uma \omterm{def:b1:mirrors-thin-lenses:image}{imagem real} nítida sobre o anteparo em exatamente duas posições
se $D > 4f'$, numa única se $D = 4f'$, e em nenhuma se $D < 4f'$. As duas
posições distam $d$ uma da outra, com
\[
f' = \frac{D^2 - d^2}{4D} .
\]
Em $D = 4f'$ (Silbermann) a única posição dá $\gamma = -1$.
\end{proposition}

\begin{proof}
Seja $p = -\overline{OA} > 0$ a distância objeto--lente; então
$\overline{OA'} = D - p$ e Descartes dá $1/(D - p) + 1/p = 1/f'$,
isto é,\ $p^2 - Dp + Df' = 0$: discriminante $D^2 - 4Df'$, positivo se e só se
$D > 4f'$. As duas raízes $p_{1,2} = (D \mp \sqrt{D^2 - 4Df'})/2$ somam
$D$ (as posições são simétricas: as distâncias do objeto e da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} se
trocam) e diferem de $d = \sqrt{D^2 - 4Df'}$; basta isolar $f'$. Em
$D = 4f'$, $p = 2f' = \overline{OA'}$ e $\gamma = -1$.
\end{proof}

\section{Duas lentes delgadas}

\begin{proposition}[Lentes em contato; dubleto afocal]\label{prop:b1:mirrors-thin-lenses:doublet}
\index{sistema afocal}\index{aumento angular}
Duas \omterm{def:b1:mirrors-thin-lenses:lens}{lentes delgadas} em contato se comportam como uma única \omterm{def:b1:mirrors-thin-lenses:lens}{lente delgada} de \omterm{def:b1:mirrors-thin-lenses:lens}{vergência}
$V = V_1 + V_2$. Duas \omterm{def:b1:mirrors-thin-lenses:lens}{lentes} dispostas de modo que $F_1' = F_2$ formam um
sistema \emph{afocal}: um feixe paralelo emerge paralelo, e um feixe
inclinado de $\alpha$ emerge inclinado de $\alpha' = G\alpha$, com o
\emph{aumento angular} $G = -f_1'/f_2'$.
\end{proposition}

\begin{proof}
A \omterm{def:b1:mirrors-thin-lenses:image}{imagem} $A_1$ dada por $L_1$ é o objeto para $L_2$. Em contato,
$O_1 = O_2 = O$: $1/\overline{OA_1} - 1/\overline{OA} = V_1$ e
$1/\overline{OA'} - 1/\overline{OA_1} = V_2$; some. Afocal: um feixe
paralelo com ângulo $\alpha$ focaliza no plano focal comum à altura
$h = f_1'\alpha$ do eixo (raio que passa por $O_1$); visto de $L_2$, esse
ponto está no seu plano focal objeto, de modo que o feixe emerge paralelo, com o
ângulo do raio que vai desse ponto até $O_2$: $\alpha' = -h/f_2'$.
\end{proof}

\begin{example}[Uma luneta em uma linha]\label{ex:b1:mirrors-thin-lenses:afocal}
$f_1' = \qty{1000}{mm}$, $f_2' = \qty{25}{mm}$: $G = -40$; a Lua,
de meio grau de largura, vira um disco de $20^\circ$ --- a luneta
astronômica do próximo capítulo. Com uma ocular divergente,
$f_2' = \qty{-25}{mm}$, $G = +40$: direita, a luneta de Galileu.
\end{example}

\section{Problema: o projetor de slides}

\begin{problem}[{Problema de fim de semana --- uma lente, medida numa bancada,
depois encarregada de lançar um slide de 35 mm através de um salão: como um projetor é
projetado, focalizado e equipado com um conversor de longo alcance}]
\label{pb:b1:mirrors-thin-lenses:1}
Uma objetiva de projeção $L_1$ é uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente delgada} convergente de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal}
$f_1'$ desconhecida. O slide tem \qty{24}{mm} de altura e \qty{36}{mm} de largura.

\medskip
\noindent\textbf{Parte I --- A \omterm{def:b1:mirrors-thin-lenses:lens}{lente} na bancada.}
\begin{enumerate}
  \item Enuncie a \omterm{thm:b1:mirrors-thin-lenses:lensconj}{relação de Descartes} e o aumento de uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente
        delgada}, com as convenções de sinal.
  \item O objeto e o anteparo estão fixos a uma distância $D$ um do outro. Mostre que uma
        \omterm{def:b1:mirrors-thin-lenses:image}{imagem real} sobre o anteparo exige $D \geq 4f_1'$.
  \item Mostre que, quando $D > 4f_1'$, as duas posições da \omterm{def:b1:mirrors-thin-lenses:lens}{lente} que dão uma
        \omterm{def:b1:mirrors-thin-lenses:image}{imagem} nítida são simétricas em relação ao ponto médio entre
        o objeto e o anteparo, e que $f_1' = (D^2 - d^2)/4D$, sendo $d$
        a separação entre elas.
  \item Com $D = \qty{80.0}{cm}$ e $d = \qty{40.0}{cm}$, calcule
        $f_1'$.
  \item A \omterm{def:b1:mirrors-thin-lenses:lens}{lente} fica num barrilete espesso cujo \omterm{def:b1:mirrors-thin-lenses:lens}{centro óptico} não pode ser
        localizado com precisão. Por que isso não afeta o método?
  \item As distâncias carregam $u(D) = \qty{0.2}{cm}$ e
        $u(d) = \qty{0.5}{cm}$. Calcule $u(f_1')$ e escreva o resultado.
  \item Dê o aumento em cada uma das duas posições. Qual delas
        corresponde a um projetor?
  \item Qual é o menor $D$ para o qual existe \omterm{def:b1:mirrors-thin-lenses:image}{imagem} nítida, e o
        aumento nesse caso (Silbermann)?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Lançando a \omterm{def:b1:mirrors-thin-lenses:image}{imagem}.}
A \omterm{def:b1:mirrors-thin-lenses:image}{imagem} na tela deve ter \qty{1.80}{m} de largura. Tome
$f_1' = \qty{15.0}{cm}$.
\begin{enumerate}[resume]
  \item Que aumento é necessário (com o seu sinal)?
  \item Calcule a distância lente--tela e a distância slide--lente.
  \item Por que o slide é inserido de cabeça para baixo e invertido da esquerda para a direita?
  \item A tela é levada a \qty{10.0}{m}. De quanto, e em que
        sentido, a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} precisa ser deslocada em relação ao slide para
        refocalizar? Qual é a largura da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} agora?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Tolerância de \omterm{def:b1:mirrors-thin-lenses:mirror}{foco} e luz.}
\begin{enumerate}[resume]
  \item Mostre que um pequeno deslocamento $\dd\overline{OA}$ do slide
        desloca a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} de $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$.
  \item O filme se deforma \qty{0.10}{mm} sob o calor da lâmpada. De quanto
        se desloca o plano da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} (tela a \qty{7.65}{m})?
        Comente.
  \item A abertura da \omterm{def:b1:mirrors-thin-lenses:lens}{lente} tem \qty{30}{mm} de diâmetro. Se a tela
        estiver \qty{20}{cm} fora do plano da \omterm{def:b1:mirrors-thin-lenses:image}{imagem}, qual é o diâmetro do
        círculo de confusão de um ponto do slide? Visto de \qty{5}{m},
        o olho (resolução angular de cerca de $\qty{3e-4}{rad}$) percebe?
  \item O slide recebe \qty{2.0}{W} de luz, que chegam integralmente à
        tela. Calcule a irradiância (potência por unidade de área) sobre o
        slide e sobre a tela, e mostre que a razão entre elas é $\gamma^2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O conversor de longo alcance.}
Num salão maior a tela fica a \qty{12.0}{m}. Uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente divergente}
$L_2$, $f_2' = \qty{-30.0}{cm}$, é encaixada \qty{8.0}{cm} atrás de
$L_1$; a tela está a \qty{12.0}{m} de $L_2$.
\begin{enumerate}[resume]
  \item Para que a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} final seja real e esteja sobre a tela, onde (a partir de
        $L_2$) deve ficar a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} intermediária dada por $L_1$? Ela é um
        objeto real ou virtual para $L_2$?
  \item Deduza a posição do slide $\overline{O_1A}$.
  \item Calcule $\gamma_1$, $\gamma_2$ e o aumento total.
  \item Dê a largura da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} e compare com o que $L_1$ sozinha daria
        na mesma tela (\qty{12.08}{m} de $L_1$).
  \item Para um objeto no infinito, mostre que o sistema equivale
        a uma única \omterm{def:b1:mirrors-thin-lenses:lens}{lente} de \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} $f_{\mathrm{eq}}$ com
        \[
        \frac{1}{f_{\mathrm{eq}}} = \frac{1}{f_1'} + \frac{1}{f_2'} - \frac{e}{f_1' f_2'},
        \]
        $e = \overline{O_1O_2}$, no sentido de que um feixe inclinado de
        $\alpha$ focaliza à altura $f_{\mathrm{eq}}\,\alpha$ do
        eixo. Calcule $f_{\mathrm{eq}}$.
  \item Verifique que a razão entre as larguras da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} (com/sem conversor)
        está próxima de $f_1'/f_{\mathrm{eq}}$. Por que apenas aproximadamente?
  \item Por que a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} intermediária deve ficar a menos de $\abs{f_2'}$
        além de $L_2$ para que a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} final seja real?
  \item Com a mesma lâmpada, por qual fator a irradiância na tela
        muda quando o conversor é montado (mesma distância de tela)?
  \item Resuma: o conversor multiplica a \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} por qual
        fator, e o que ele faz com o tamanho e o brilho da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} a uma
        dada distância de projeção?
\end{enumerate}
\end{problem}
