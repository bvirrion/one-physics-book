\chapter{Regime senoidal permanente e impedância}\label{ch:b1:sinusoidal-impedance}

Gire o dial de um rádio antigo e, das cem estações que chegam todas
à antena ao mesmo tempo, uma se faz ouvir: uma bobina e um capacitor
variável foram ajustados de modo que exatamente uma frequência os faça
ressoar. A rede elétrica que zumbe em cada parede é uma senoide de \qty{50}{Hz};
a fábrica que faz girar grandes motores com ela paga uma multa se a sua corrente
se atrasa demais em relação à \omterm{def:b1:dc-circuits:voltage}{tensão}. As senoides são a linguagem natural dos
\omterm{def:b1:dc-circuits:linear}{circuitos lineares} --- saem como entraram, apenas escaladas e defasadas
--- e este capítulo lhes dá a sua álgebra: \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas},
\omterm{def:b1:sinusoidal-impedance:impedance}{impedâncias}, a ressonância do \omterm{prop:b1:transient-regimes:rlc}{circuito RLC} e a potência que as
correntes senoidais realmente entregam.

\section{O regime senoidal permanente}

\begin{proposition}[Regime senoidal forçado]\label{prop:b1:sinusoidal-impedance:forced}
\index{regime senoidal permanente}\index{regime forçado}
Num \omterm{def:b1:dc-circuits:linear}{circuito linear} excitado por uma fonte senoidal de \omterm{def:b1:wave-propagation:signal}{frequência angular}
$\omega$, toda corrente e toda \omterm{def:b1:dc-circuits:voltage}{tensão} são, uma vez extintos os
transitórios, senoides de mesma \omterm{def:b1:wave-propagation:signal}{frequência angular} $\omega$, cada uma com a sua
própria \omterm{def:b1:wave-propagation:signal}{amplitude} e fase: $x(t) = X\cos(\omega t + \varphi)$.
\end{proposition}

\begin{proof}
As equações do circuito são equações diferenciais lineares com coeficientes
constantes; a sua solução geral é uma solução particular mais as
soluções da equação homogênea, que são os transitórios do
\cref{ch:b1:transient-regimes} e decaem (todo circuito real tem alguma
resistência). Existe uma solução particular senoidal porque
derivar uma senoide de frequência $\omega$ dá uma senoide da
mesma frequência: as equações podem ser satisfeitas termo a termo --- o
método complexo abaixo a constrói explicitamente.
\end{proof}

\begin{definition}[Amplitude complexa]\label{def:b1:sinusoidal-impedance:complex}
\index{amplitude complexa}\index{fasor}
À senoide $x(t) = X\cos(\omega t + \varphi)$ associa-se o
sinal complexo $\underline{x}(t) = X\eu^{j(\omega t + \varphi)}$, de modo que
$x = \Rea(\underline{x})$, e a \emph{\omterm{def:b1:wave-propagation:signal}{amplitude} complexa}
\[
\underline{X} = X\eu^{j\varphi}, \qquad \underline{x}(t) = \underline{X}\,\eu^{j\omega t},
\]
que carrega a \omterm{def:b1:wave-propagation:signal}{amplitude} $X = \abs{\underline{X}}$ e a fase
$\varphi = \arg\underline{X}$. Aqui $j$ designa a unidade imaginária,
$j^2 = -1$, ficando a letra $i$ reservada às correntes. Desenhada como um
vetor de comprimento $X$ e ângulo $\varphi$, $\underline{X}$ é um
\emph{fasor} (vetor de Fresnel).
\end{definition}

\begin{proposition}[Regras do método complexo]\label{prop:b1:sinusoidal-impedance:rules}
\begin{enumerate}
  \item As operações lineares comutam com a tomada da parte real: a
        \omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexa} de uma soma é a soma das \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas}.
  \item Derivar em relação ao tempo multiplica a \omterm{def:b1:sinusoidal-impedance:complex}{amplitude
        complexa} por $j\omega$; integrar a divide por $j\omega$.
  \item As \omterm{thm:b1:dc-circuits:kirchhoff}{leis de Kirchhoff}, por serem lineares, valem para \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas}.
\end{enumerate}
\end{proposition}

\begin{proof}
$\Rea(\underline a + \underline b) = \Rea\underline a + \Rea\underline b$;
$\dd(\underline X\eu^{j\omega t})/\dd t = j\omega\underline X\eu^{j\omega t}$,
e $\dd x/\dd t = \Rea(\dd\underline x/\dd t)$, pois a derivação
é linear sobre os reais. As somas de correntes num nó e de tensões numa
malha são lineares.
\end{proof}

\begin{example}[Somando duas senoides]\label{ex:b1:sinusoidal-impedance:add}
$3\cos\omega t + 4\sin\omega t$: as \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas} são $3$ e
$4\eu^{-j\pi/2} = -4j$ (pois $\sin\omega t = \cos(\omega t - \pi/2)$);
a sua soma $3 - 4j$ tem módulo $5$ e argumento $-0.927\ \unit{rad}$:
$5\cos(\omega t - 0.927)$. Nenhuma identidade trigonométrica é necessária.
\end{example}

\section{Impedância}

\begin{definition}[Impedância, admitância]\label{def:b1:sinusoidal-impedance:impedance}
\index{impedância}\index{admitância}\index{reatância}
Para um \omterm{def:b1:dc-circuits:linear}{dipolo linear} em \omterm{prop:b1:sinusoidal-impedance:forced}{regime senoidal permanente} (\omterm{thm:b1:dc-circuits:kirchhoff}{convenção
receptor}), a \emph{impedância} é a razão das \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas}
\[
\underline Z = \frac{\underline U}{\underline I} = Z\eu^{j\varphi}
\quad (\text{ohms}),
\]
cujo módulo $Z = U/I$ relaciona as \omterm{def:b1:wave-propagation:signal}{amplitudes} e cujo argumento
$\varphi = \varphi_u - \varphi_i$ é o avanço de fase da \omterm{def:b1:dc-circuits:voltage}{tensão} sobre
a corrente. O seu inverso $\underline Y = 1/\underline Z$ é a
\emph{admitância} (siemens). A parte real de $\underline Z$ é a sua
resistência, e a parte imaginária a sua \emph{reatância}.
\end{definition}

\begin{proposition}[Impedâncias de R, L, C]\label{prop:b1:sinusoidal-impedance:rlc}
\[
\underline Z_R = R, \qquad \underline Z_L = jL\omega, \qquad
\underline Z_C = \frac{1}{jC\omega} = -\frac{j}{C\omega} .
\]
Um \omterm{def:b1:dc-circuits:resistor}{resistor} mantém \omterm{def:b1:dc-circuits:voltage}{tensão} e corrente em fase; numa bobina a \omterm{def:b1:dc-circuits:voltage}{tensão}
adianta a corrente de $\pi/2$; num capacitor ela atrasa de $\pi/2$. Em baixa
frequência uma bobina tende a um fio e um capacitor a um circuito aberto; em
alta frequência é o contrário.
\end{proposition}

\begin{proof}
$u = Ri$; $u = L\,\dd i/\dd t \to \underline U = jL\omega\,\underline I$;
$i = C\,\dd u/\dd t \to \underline I = jC\omega\,\underline U$. Os
argumentos de $1$, $j$ e $-j$ valem $0$, $\pi/2$, $-\pi/2$.
\end{proof}

\begin{proposition}[Associação; divisores; teoremas]\label{prop:b1:sinusoidal-impedance:assoc}
\omterm{def:b1:sinusoidal-impedance:impedance}{Impedâncias} em série somam-se; \omterm{def:b1:sinusoidal-impedance:impedance}{admitâncias} em paralelo somam-se; os \omterm{prop:b1:dc-circuits:dividers}{divisores
de tensão} e de corrente, os teoremas de Thévenin e de Norton, a superposição e
o teorema de Millman do \cref{ch:b1:dc-circuits} valem todos com \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes
complexas} e \omterm{def:b1:sinusoidal-impedance:impedance}{impedâncias} no lugar de tensões, correntes e
resistências reais.
\end{proposition}

\begin{proof}
Toda demonstração do \cref{ch:b1:dc-circuits} usou apenas as \omterm{thm:b1:dc-circuits:kirchhoff}{leis de Kirchhoff} e
a linearidade de $u = Ri$, ambas válidas em forma complexa.
\end{proof}

\begin{example}[O divisor RC]\label{ex:b1:sinusoidal-impedance:rcdivider}
Um \omterm{def:b1:dc-circuits:resistor}{resistor} $R$ e um capacitor $C$ em série sob $\underline E$, com a
saída tomada em $C$: $\underline U_C = \underline E\,\dfrac{1/jC\omega}
{R + 1/jC\omega} = \dfrac{\underline E}{1 + jRC\omega}$. \omterm{def:b1:wave-propagation:signal}{Amplitude}
$E/\sqrt{1 + (RC\omega)^2}$, fase $-\arctan(RC\omega)$: transmissão
total em baixa frequência, $1/\sqrt2$ e $-45^\circ$ em
$\omega = 1/RC$, e uma queda em $1/\omega$ além dali --- um filtro passa-baixa,
objeto do \cref{ch:b1:filters-transfer-functions}.
\end{example}

\section{O circuito RLC série: a ressonância}

\begin{theorem}[Ressonância em corrente]\label{thm:b1:sinusoidal-impedance:resonance}
\index{ressonância}\index{ressonância!em corrente}\index{largura de banda}
Um \omterm{def:b1:dc-circuits:resistor}{resistor} $R$, uma bobina $L$ e um capacitor $C$ em série sob
$e(t) = E\cos\omega t$ conduzem a corrente de \omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexa}
\[
\underline I = \frac{E}{R + j\left(L\omega - \dfrac{1}{C\omega}\right)},
\qquad
I = \frac{E}{\sqrt{R^2 + \left(L\omega - \dfrac{1}{C\omega}\right)^2}},
\qquad
\varphi_i = -\arctan\frac{L\omega - 1/C\omega}{R} .
\]
A \omterm{def:b1:wave-propagation:signal}{amplitude} é máxima, $I_{\max} = E/R$, com corrente e \omterm{def:b1:dc-circuits:voltage}{tensão} em
fase, na \emph{ressonância} $\omega_0 = 1/\sqrt{LC}$. Com
$Q = L\omega_0/R = 1/(RC\omega_0)$ e $x = \omega/\omega_0$,
\[
\frac{I}{I_{\max}} = \frac{1}{\sqrt{1 + Q^2\left(x - \dfrac1x\right)^2}} ,
\]
e a \emph{\omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda}} --- o intervalo de $\omega$ em que
$I \geq I_{\max}/\sqrt2$ --- vale
\[
\Delta\omega = \omega_2 - \omega_1 = \frac{\omega_0}{Q} = \frac{R}{L} .
\]
\end{theorem}

\begin{proof}
As \omterm{def:b1:sinusoidal-impedance:impedance}{impedâncias} em série somam-se: $\underline Z = R + jL\omega + 1/jC\omega$, e
$\underline I = \underline E/\underline Z$. O módulo é máximo quando
a \omterm{def:b1:sinusoidal-impedance:impedance}{reatância} $L\omega - 1/C\omega$ se anula, isto é,\ em $\omega_0$.
Fatore $L\omega - 1/C\omega = L\omega_0(x - 1/x)$ e divida por $R$:
$L\omega_0/R = Q$. Nas bordas da banda, $Q(x - 1/x) = \pm 1$, isto é,\
$x^2 \mp x/Q - 1 = 0$, cujas raízes positivas são $x_{1,2} = \mp\frac{1}{2Q}
+ \sqrt{1 + \frac{1}{4Q^2}}$; a sua diferença vale $1/Q$.
\end{proof}

\begin{proposition}[Sobretensão]\label{prop:b1:sinusoidal-impedance:overvoltage}
\index{ressonância!em tensão}\index{sobretensão}
Na ressonância, as tensões na bobina e no capacitor têm a
mesma \omterm{def:b1:wave-propagation:signal}{amplitude} $QE$ e fases opostas: para $Q \gg 1$ o capacitor
suporta uma \omterm{def:b1:dc-circuits:voltage}{tensão} $Q$ vezes maior que a da fonte. A própria \omterm{def:b1:wave-propagation:signal}{amplitude}
$U_C(\omega)$ passa por um máximo, para $Q > 1/\sqrt2$, em $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$, ligeiramente abaixo de $\omega_0$, com
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$.
\end{proposition}

\begin{proof}
Em $\omega_0$, $\underline U_C = \underline I/jC\omega_0 = -jQE$ e
$\underline U_L = jL\omega_0\underline I = jQE$, pois $I = E/R$. Em
geral $U_C = I/C\omega = E/\sqrt{(1 - x^2)^2 + x^2/Q^2}$ (multiplique
numerador e denominador por $1/C\omega_0$); o quadrado do denominador
vale $(1 - x^2)^2 + x^2/Q^2$, um trinômio em $x^2$ mínimo em $x^2 =
1 - 1/2Q^2$ quando este é positivo, com mínimo $1/Q^2 - 1/4Q^4$.
\end{proof}

\begin{example}[Sintonizando um rádio]\label{ex:b1:sinusoidal-impedance:radio}
Uma bobina $L = \qty{200}{\micro H}$ e um capacitor variável: $C =
\qty{127}{pF}$ sintoniza $f_0 = \qty{1.00}{MHz}$. Com a resistência do fio
da bobina $R = \qty{12.6}{\ohm}$, tem-se $L\omega_0 = \qty{1257}{\ohm}$ e
$Q = 100$: \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} $\Delta f = f_0/Q = \qty{10}{kHz}$ --- a largura
de um canal AM --- e os \qty{10}{\micro V} induzidos por um transmissor
distante viram $QE = \qty{1}{mV}$ no capacitor, de
graça. O problema de fim de semana constrói esse receptor.
\end{example}

\begin{remark}[Por que $Q$ de novo]\label{rem:b1:sinusoidal-impedance:Q}
O $Q$ deste capítulo é o $Q$ do \cref{ch:b1:transient-regimes}:
o circuito que oscila por cerca de $Q$ períodos quando percutido é o
que ressoa numa banda de largura $\omega_0/Q$ quando excitado. Ambos
exprimem uma mesma razão: $2\pi$ vezes a energia armazenada dividida pela energia
dissipada por período (\cref{exo:b1:sinusoidal-impedance:12}). Uma
ressonância aguda e uma oscilação longa são a mesma propriedade, e o mesmo vale para a
resposta lenta: um filtro de \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} $\Delta\omega$ leva um tempo da
ordem de $1/\Delta\omega$ para se estabelecer.
\end{remark}

\section{Potência em regime senoidal}

\begin{definition}[Valor eficaz]\label{def:b1:sinusoidal-impedance:rms}
\index{valor eficaz}\index{valor RMS}
O \emph{valor eficaz} (ou valor RMS, da raiz da média quadrática) de um
sinal periódico $x(t)$ é $X_{\mathrm{rms}} = \sqrt{\langle x^2\rangle}$, a
raiz quadrada da média temporal de $x^2$ sobre um período. Para uma senoide
de \omterm{def:b1:wave-propagation:signal}{amplitude} $X$, $X_{\mathrm{rms}} = X/\sqrt2$. Os \qty{230}{V} da
rede elétrica são um valor eficaz; a sua \omterm{def:b1:wave-propagation:signal}{amplitude} vale \qty{325}{V}.
\end{definition}

\begin{theorem}[Potência média]\label{thm:b1:sinusoidal-impedance:power}
\index{potência média}\index{fator de potência}\index{potência reativa}
Um dipolo com $u = U\cos\omega t$ e $i = I\cos(\omega t - \varphi)$
(\omterm{thm:b1:dc-circuits:kirchhoff}{convenção receptor}) recebe a potência instantânea $p = ui$, de
média
\[
P = \tfrac12\,UI\cos\varphi = U_{\mathrm{rms}}I_{\mathrm{rms}}\cos\varphi
= \tfrac12\Rea\!\big(\underline U\,\underline I^{\,*}\big) = \tfrac12\Rea(\underline Z)\,I^2,
\]
em que $\cos\varphi$ é o \emph{\omterm{thm:b1:sinusoidal-impedance:power}{fator de potência}}. Um \omterm{def:b1:dc-circuits:resistor}{resistor} recebe
$\tfrac12 RI^2 = RI_{\mathrm{rms}}^2$; uma bobina ou um capacitor ideais
($\varphi = \pm\pi/2$) não recebem \omterm{thm:b1:sinusoidal-impedance:power}{potência média} --- armazenam energia
durante meio período e a devolvem durante a outra metade.
\end{theorem}

\begin{proof}
$p = UI\cos\omega t\cos(\omega t - \varphi) = \tfrac12 UI[\cos\varphi +
\cos(2\omega t - \varphi)]$; o segundo termo tem média nula. Com
$\underline U = U$ e $\underline I = I\eu^{-j\varphi}$,
$\underline U\,\underline I^* = UI\eu^{j\varphi}$, cuja parte real vale
$UI\cos\varphi$; e $\underline U = \underline Z\,\underline I$ dá
$\underline U\,\underline I^* = \underline Z I^2$.
\end{proof}

\begin{example}[Correção do fator de potência]\label{ex:b1:sinusoidal-impedance:pf}
Um motor de oficina em \qty{230}{V} eficazes consome \qty{2.0}{kW} com
$\cos\varphi = 0.70$ (indutivo): $I_{\mathrm{rms}} = 2000/(230 \times
0.70) = \qty{12.4}{A}$, contra \qty{8.7}{A} se o \omterm{thm:b1:sinusoidal-impedance:power}{fator de potência} fosse
$1$ --- a corrente excedente aquece os cabos da concessionária à toa, daí
a multa. Um capacitor em paralelo fornece localmente as trocas de energia
de um quarto de período da bobina: a \emph{\omterm{thm:b1:sinusoidal-impedance:power}{potência reativa}} $P\tan\varphi =
\qty{2.0}{kvar}$ pede $C = P\tan\varphi/(\omega U_{\mathrm{rms}}^2)
\approx \qty{120}{\micro F}$ a \qty{50}{Hz}, e a corrente de linha cai
para \qty{8.7}{A}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:1}
Dê a \omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexa} de $u = 5\cos(\omega t + \pi/3)$ (V), nas formas
polar e cartesiana. Usando \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas}, escreva $3\cos\omega t
+ 4\sin\omega t$ como um único cosseno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:2}

\section{Problema: o receptor de rádio ressonante}

\begin{problem}[{Problema de fim de semana --- uma bobina, um capacitor variável e
uma antena: como uma ressonância série escolhe uma estação entre cem,
multiplica o seu sinal fraco por cem e por que ela não pode ser feita
arbitrariamente seletiva}]
\label{pb:b1:sinusoidal-impedance:1}
A antena é modelada por uma fem senoidal ideal $e(t) = E\cos\omega t$
em série com uma bobina $L = \qty{200}{\micro H}$, cujo fio tem
resistência $R = \qty{12.6}{\ohm}$, e um capacitor variável $C$. O
receptor lê a \omterm{def:b1:dc-circuits:voltage}{tensão} em $C$. Mais adiante (Parte IV), leva-se em conta a
resistência própria da antena, $R_a = \qty{50}{\ohm}$.

\medskip
\noindent\textbf{Parte I --- A corrente na ressonância.}
\begin{enumerate}
  \item Escreva a \omterm{def:b1:sinusoidal-impedance:impedance}{impedância} complexa do circuito série.
  \item Dê a \omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexa} da corrente, depois a sua \omterm{def:b1:wave-propagation:signal}{amplitude}
        $I(\omega)$ e a sua fase $\varphi_i(\omega)$ em relação a $e$.
  \item Mostre que $I$ é máxima em $\omega_0 = 1/\sqrt{LC}$; dê
        $I_{\max}$ e a fase nesse caso.
  \item Que capacitância sintoniza $f_0 = \qty{1.000}{MHz}$? Calcule
        $L\omega_0$.
  \item Defina $Q = L\omega_0/R$ e calcule-o.
  \item Mostre que $I/I_{\max} = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ com
        $x = \omega/\omega_0$.
  \item Enuncie o sinal de $\varphi_i$ abaixo e acima da ressonância e
        interprete (comportamento capacitivo ou indutivo).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A seletividade.}
\begin{enumerate}[resume]
  \item Escreva a condição que define as duas frequências em que
        $I = I_{\max}/\sqrt2$.
  \item Resolva-a e mostre que $\Delta\omega = \omega_2 - \omega_1 =
        \omega_0/Q$.
  \item Calcule a \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} $\Delta f$ deste receptor.
  \item Uma segunda estação transmite em \qty{1.010}{MHz} e uma terceira em
        \qty{1.020}{MHz}: calcule a razão de \omterm{def:b1:wave-propagation:signal}{amplitudes} $I/I_{\max}$ para
        cada uma e a razão de potências correspondente em decibéis.
  \item As estações AM ficam a \qty{9}{} ou \qty{10}{kHz} umas das outras. O
        receptor é suficientemente seletivo? O que o melhoraria?
  \item O capacitor pode ser ajustado entre \qty{50}{} e \qty{500}{pF}:
        que faixa de frequências o receptor cobre?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A sobretensão.}
\begin{enumerate}[resume]
  \item Exprima a \omterm{def:b1:sinusoidal-impedance:complex}{amplitude complexa} da \omterm{def:b1:dc-circuits:voltage}{tensão} no capacitor e mostre
        que na ressonância a sua \omterm{def:b1:wave-propagation:signal}{amplitude} vale $QE$.
  \item A fem da antena vale $E = \qty{10}{\micro V}$: calcule $I_{\max}$
        e a \omterm{def:b1:dc-circuits:voltage}{tensão} no capacitor na ressonância. Comente.
  \item Mostre que a \omterm{def:b1:dc-circuits:voltage}{tensão} na bobina tem a mesma \omterm{def:b1:wave-propagation:signal}{amplitude} e está em
        oposição de fase com a \omterm{def:b1:dc-circuits:voltage}{tensão} no capacitor na ressonância: o que
        a fonte ``enxerga''?
  \item Calcule a energia armazenada no circuito na ressonância e a
        energia dissipada por período; verifique que a razão entre elas, vezes
        $2\pi$, vale $Q$.
  \item A partir do \cref{ch:b1:transient-regimes}, qual é a \omterm{def:b1:transient-regimes:firstorder}{constante de tempo}
        $\tau = 2Q/\omega_0$ deste circuito? Por que um $Q$ muito alto
        acaba distorcendo a fala e a música (\omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} do áudio:
        cerca de \qty{5}{kHz} em AM)?
  \item Se a resistência do fio da bobina dobrasse (fio mais fino), como
        mudariam $Q$, a \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} e a \omterm{def:b1:dc-circuits:voltage}{tensão} no capacitor?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A resistência da antena e a potência.}
\begin{enumerate}[resume]
  \item Com $R_a = \qty{50}{\ohm}$ em série, recalcule $Q$, a
        \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} e a \omterm{def:b1:dc-circuits:voltage}{tensão} no capacitor para $E = \qty{10}{\micro V}$.
        Conclua.
  \item Calcule a \omterm{thm:b1:sinusoidal-impedance:power}{potência média} entregue pela fem da antena na
        ressonância e a fração dissipada em $R_a$.
  \item O teorema da máxima potência pediria uma resistência de carga igual
        a $R_a$: por que esse objetivo entra em conflito com a seletividade?
  \item Os receptores reais ligam a antena a uma derivação da bobina, ou
        através de um pequeno capacitor de acoplamento. Explique qualitativamente o que
        isso consegue.
  \item Usando $P = \tfrac12 UI\cos\varphi$, calcule o \omterm{thm:b1:sinusoidal-impedance:power}{fator de potência} nas
        duas frequências de meia potência e explique o nome
        ``meia potência''.
  \item Resuma o projeto: o que $Q$ traz, o que ele custa e o
        valor escolhido para um receptor de \qty{1}{MHz} com canais de \qty{10}{kHz}.
\end{enumerate}
\end{problem}
