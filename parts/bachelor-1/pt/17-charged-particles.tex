\chapter{Partículas carregadas em campos E e B}\label{ch:b1:charged-particles}

Dentro do osciloscópio que traçou as curvas do \cref{ch:b1:transient-regimes},
um feixe de elétrons é dirigido pela tela pela \omterm{def:b1:dc-circuits:voltage}{tensão} entre
duas placas; dentro do ímã de um \omterm{prop:b1:charged-particles:spectrometer}{espectrômetro de massa}, íons de dois
isótopos se separam por milímetros porque um é oito por cento mais pesado
que o outro; dentro do cíclotron de um hospital, prótons descrevem espirais
por duzentas voltas antes de sair a um terço da velocidade da
luz para queimar um tumor. Os três casos são a mesma dinâmica: uma carga num
campo uniforme, elétrico ou magnético, e a lei de Newton. Este capítulo
a desenvolve --- a parábola num \omterm{def:b1:charged-particles:lorentz}{campo elétrico}, o círculo e a
hélice num \omterm{def:b1:charged-particles:lorentz}{campo magnético} --- e a transforma nos instrumentos que
separam, aceleram e medem partículas carregadas.

{\small O cíclotron de 60 polegadas de Lawrence, em Berkeley, 1939 (Departamento de Energia dos Estados Unidos): entre os polos do ímã, os prótons descrevem espirais para fora à medida que recebem dois impulsos por volta, exatamente como no problema de fim de semana.}

\section{A força de Lorentz}

\begin{definition}[Força de Lorentz]\label{def:b1:charged-particles:lorentz}
\index{força de Lorentz}\index{campo elétrico}\index{campo magnético}
Uma partícula de carga $q$ que se move com velocidade $\vect v$ num campo
elétrico $\vect E$ e num campo magnético $\vect B$ recebe a força
\[
\vect F = q\big(\vect E + \vect v\wedge\vect B\big) .
\]
A parte elétrica é dirigida ao longo de $\vect E$ (ou em sentido oposto, se $q < 0$) e
independe do movimento; a parte magnética é perpendicular a
$\vect v$ e a $\vect B$, proporcional à velocidade, e se anula para uma
partícula em repouso. Unidades: $\vect E$ em \unit{V/m} (ou \unit{N/C}) e $\vect B$
em teslas (\unit{T}): $\qty{1}{T} = \qty{1}{N/(A.m)}$.
\end{definition}

\begin{proposition}[Trabalho da força de Lorentz]\label{prop:b1:charged-particles:work}
A força magnética não realiza trabalho: a sua potência $q(\vect v\wedge\vect B)\cdot\vect v$
é nula, de modo que um \omterm{def:b1:charged-particles:lorentz}{campo magnético} sozinho muda a direção da velocidade,
mas nunca o seu módulo nem a \omterm{thm:b1:work-and-energy:ke}{energia cinética}. A força elétrica tem potência
$q\vect E\cdot\vect v$ e, entre dois pontos, o seu trabalho vale $q(V_A - V_B) =
qU_{AB}$ (\cref{ch:b1:potential-capacitors}): uma partícula de carga $q$
que atravessa uma \omterm{def:b1:dc-circuits:voltage}{tensão} $U$ ganha a \omterm{thm:b1:work-and-energy:ke}{energia cinética} $\abs{qU}$.
\end{proposition}

\begin{proof}
$\vect v\wedge\vect B \perp \vect v$. O campo eletrostático deriva do
potencial, $\vect E = -\overrightarrow{\operatorname{grad}}V$, de modo que $q\vect E$
é conservativa, com $E_p = qV$.
\end{proof}

\begin{definition}[O elétron-volt]\label{def:b1:charged-particles:eV}
\index{elétron-volt}
O \emph{elétron-volt} é a \omterm{thm:b1:work-and-energy:ke}{energia cinética} ganha por uma carga
elementar ao atravessar um volt: $\qty{1}{eV} = \qty{1.60e-19}{J}$. A química
vive em alguns eV, os tubos de raios X em dezenas de keV, a física nuclear em MeV
e a física de partículas em GeV e TeV.
\end{definition}

\begin{remark}[Por que a gravidade é ignorada]\label{rem:b1:charged-particles:gravity}
Para um elétron num campo de \qty{1}{kV/m} --- um valor modesto de
laboratório --- a força elétrica vale $eE = \qty{1.6e-16}{N}$, contra um peso
$mg = \qty{9e-30}{N}$: $10^{13}$ vezes menor. O peso é descartado de
todo cálculo deste capítulo sem hesitação. (O limite
relativístico é outra história: as fórmulas abaixo valem para
$\tfrac12 mv^2 \ll mc^2$, isto é,\ abaixo de cerca de \qty{50}{keV} para elétrons
e \qty{100}{MeV} para prótons; veja o volume do terceiro ano.)
\end{remark}

\section{Campo elétrico uniforme}

\begin{proposition}[Aceleração e deflexão]\label{prop:b1:charged-particles:efield}
\index{tensão de aceleração}
Num campo uniforme $\vect E$ a aceleração $\vect a = q\vect E/m$ é
constante: o movimento é o de um projétil, com $q\vect E/m$ no lugar
de $\vect g$.
\begin{itemize}
  \item Solta do repouso e acelerada por uma \omterm{def:b1:dc-circuits:voltage}{tensão} $U$, a
        partícula atinge $v = \sqrt{2\abs{qU}/m}$.
  \item Entrando com velocidade $v_0$ ao longo de $x$ numa região de comprimento $L$ em que
        $\vect E = E\vect e_y$, ela segue a parábola $y = (qE/2mv_0^2)x^2$
        e sai desviada do ângulo $\alpha$, com $\tan\alpha =
        qEL/mv_0^2$, como se tivesse vindo em linha reta do
        meio das placas.
\end{itemize}
\end{proposition}

\begin{proof}
Energia: $\tfrac12 mv^2 = \abs{qU}$. Deflexão: $x = v_0t$, $y =
\tfrac12(qE/m)t^2$; na saída, $\dd y/\dd x = (qE/m)(x/v_0^2)$ calculada
em $L$; a tangente na saída, $y = (qEL/mv_0^2)(x - L/2)$, cruza o
eixo em $x = L/2$.
\end{proof}

\begin{example}[O tubo do osciloscópio]\label{ex:b1:charged-particles:crt}
Elétrons acelerados por $U_0 = \qty{2.0}{kV}$ ($v_0 = \qty{2.65e7}{m/s}$,
um décimo de $c$ --- ainda não relativísticos) atravessam placas de \qty{4.0}{cm}
separadas por \qty{1.0}{cm} sob \qty{100}{V}: $E = \qty{1.0e4}{V/m}$,
$\tan\alpha = eEL/mv_0^2 = EL/2U_0 = 0.10$, e numa tela a \qty{30}{cm}
do centro das placas a mancha se desloca \qty{3.0}{cm} --- \qty{0.3}{mm}
por volt, a sensibilidade do tubo, fixada apenas pela geometria e pela
\omterm{prop:b1:charged-particles:efield}{tensão de aceleração}.
\end{example}

\section{Campo magnético uniforme}

\begin{theorem}[Movimento num campo magnético uniforme]\label{thm:b1:charged-particles:bfield}
\index{movimento ciclotrônico}\index{raio de Larmor}\index{frequência ciclotrônica}\index{movimento helicoidal}
Uma partícula de carga $q$ e massa $m$, com velocidade $\vect v_0$ perpendicular a
um $\vect B$ uniforme, descreve um círculo com velocidade constante $v_0$, de raio
e \omterm{def:b1:wave-propagation:signal}{frequência angular}
\[
R = \frac{mv_0}{\abs q B}, \qquad \omega_c = \frac{\abs q B}{m},
\]
sendo a \emph{\omterm{thm:b1:charged-particles:bfield}{frequência ciclotrônica}} $\omega_c$ independente da velocidade;
o sentido de rotação depende do sinal de $q$. Se $\vect v_0$ tiver também
uma componente $v_\parallel$ ao longo de $\vect B$, essa componente não muda e
a \omterm{def:b1:point-kinematics:frame}{trajetória} é uma \emph{hélice} de raio $mv_\perp/\abs qB$ e passo
$2\pi mv_\parallel/\abs qB$ enrolada em torno das linhas de campo.
\end{theorem}

\begin{proof}
A força $q\vect v\wedge\vect B$ é perpendicular a $\vect v$ e a $\vect B$;
o módulo da velocidade é constante (sem trabalho) e a componente ao longo de $\vect B$ não sente
força alguma. No plano perpendicular a $\vect B$ a força tem norma constante
$\abs qv_\perp B$, sempre perpendicular à velocidade: \omterm{cor:b1:point-kinematics:circular}{movimento circular}
uniforme (\cref{thm:b1:point-kinematics:frenet}) com $mv_\perp^2/R =
\abs qv_\perp B$. Então $\omega_c = v_\perp/R$; o período $2\pi m/\abs qB$ vezes
$v_\parallel$ é o passo.
\end{proof}

\begin{example}[Ordens de grandeza]\label{ex:b1:charged-particles:orders}
Um elétron de \qty{1}{keV} ($v = \qty{1.9e7}{m/s}$) em \qty{1}{mT}:
$R = \qty{11}{cm}$, $\omega_c = \qty{1.8e8}{rad/s}$, uma volta em
\qty{36}{ns} --- o mesmo período em qualquer energia, e é isso que torna o
cíclotron possível. Um próton do vento solar (\qty{1}{keV},
\qty{4.4e5}{m/s}) nos \qty{50}{\micro T} da Terra: $R = \qty{92}{m}$;
ele não consegue atravessar as linhas de campo, apenas descrever espirais ao longo delas rumo aos
polos --- a aurora.
\end{example}

\section{Instrumentos}

\begin{proposition}[Seletor de velocidades e espectrômetro de massa]\label{prop:b1:charged-particles:spectrometer}
\index{seletor de velocidades}\index{espectrômetro de massa}
Em campos uniformes cruzados $\vect E \perp \vect B$, uma partícula que se move
perpendicularmente aos dois passa sem desvio se e só se $v = E/B$, quaisquer que sejam a sua
carga e a sua massa (\emph{\omterm{prop:b1:charged-particles:spectrometer}{seletor de velocidades}}). Num \omterm{def:b1:charged-particles:lorentz}{campo magnético} sozinho, uma
partícula de velocidade conhecida descreve um semicírculo de diâmetro $2mv/\abs qB$:
medi-lo dá $m/q$ (\emph{\omterm{prop:b1:charged-particles:spectrometer}{espectrômetro de massa}}). Íons acelerados
por uma \omterm{def:b1:dc-circuits:voltage}{tensão} $U$ e depois curvados em $B$ chegam a
\[
R = \frac{1}{B}\sqrt{\frac{2mU}{\abs q}} :
\]
dois isótopos de massas $m$ e $m + \Delta m$ se separam por $\Delta R \approx
R\,\Delta m/2m$.
\end{proposition}

\begin{proof}
$q\vect E + q\vect v\wedge\vect B = \vect 0$ exige $vB = E$ com a orientação
correta. $R = mv/qB$ com $v = \sqrt{2qU/m}$; derive
$\ln R = \tfrac12\ln m + \text{const}$.
\end{proof}

\begin{example}[Isótopos do carbono]\label{ex:b1:charged-particles:isotopes}
Íons $^{12}$C e $^{13}$C de carga unitária acelerados por \qty{20}{kV}
em \qty{0.50}{T}: $R = \qty{14.1}{cm}$ e \qty{14.7}{cm}, a \qty{5.8}{mm}
um do outro --- duas manchas separadas no detector, e a razão das suas
intensidades é a abundância isotópica, matéria-prima da datação por
radiocarbono e de toda análise de isótopos.
\end{example}

\begin{proposition}[Cíclotron]\label{prop:b1:charged-particles:cyclotron}
\index{cíclotron}
Dois semicilindros ocos (``dês'') num $\vect B$ uniforme, com uma
\omterm{def:b1:dc-circuits:voltage}{tensão} alternada $U$ entre eles na \omterm{thm:b1:charged-particles:bfield}{frequência ciclotrônica}: a
partícula descreve círculos dentro de cada dê (onde não há campo), é acelerada por
$\abs qU$ em cada uma das duas travessias do intervalo por volta, e descreve uma espiral para fora
com $R \propto v$; no raio de extração $R_{\max}$ a sua \omterm{thm:b1:work-and-energy:ke}{energia
cinética} vale
\[
E_k = \frac{q^2B^2R_{\max}^2}{2m},
\]
independente da \omterm{def:b1:dc-circuits:voltage}{tensão}, que apenas fixa o número de voltas.
\end{proposition}

\begin{proof}
Como $\omega_c$ não depende da velocidade, uma \omterm{def:b1:dc-circuits:voltage}{tensão} de frequência fixa
permanece em passo com a partícula volta após volta; $v = \abs qBR/m$ no raio
$R$.
\end{proof}

\begin{example}[Um cíclotron hospitalar]\label{ex:b1:charged-particles:medical}
Prótons, $B = \qty{1.5}{T}$, $R_{\max} = \qty{0.50}{m}$: $E_k =
(1.6\times10^{-19} \times 1.5 \times 0.5)^2/(2 \times 1.67\times10^{-27}) =
\qty{4.3e-12}{J} = \qty{27}{MeV}$, com $f_c = \omega_c/2\pi = \qty{23}{MHz}$.
Com \qty{50}{kV} nos dês, \qty{100}{keV} por volta: $270$ voltas em
\qty{12}{\micro s}. Acima de algumas dezenas de MeV o aumento relativístico
da massa (volume do terceiro ano) tira os prótons de sincronia; as máquinas de terapia
a \qty{230}{MeV} modulam a frequência ou o campo para compensar ---
o assunto do problema de fim de semana.
\end{example}

\begin{proposition}[Efeito Hall]\label{prop:b1:charged-particles:hall}
\index{efeito Hall}
Uma fita de espessura $d$ que conduz uma corrente $I$ de portadores de carga
$q$ e densidade $n$, num campo $B$ perpendicular à fita,
desenvolve ao longo da sua largura a \emph{\omterm{def:b1:dc-circuits:voltage}{tensão} Hall}
\[
U_H = \frac{IB}{nqd} :
\]
proporcional a $B$ (um magnetômetro) e inversamente proporcional a $n$ (uma sonda da
densidade e do sinal dos portadores).
\end{proposition}

\begin{proof}
A força magnética $qv_dB$ sobre os portadores (velocidade de deriva $v_d$) os empurra
para uma das bordas até que o \omterm{def:b1:charged-particles:lorentz}{campo elétrico} transversal $E_H$ a equilibre:
$qE_H = qv_dB$, e $U_H = E_Hw$ ao longo da largura $w$; com $I = nqv_dwd$,
$E_Hw = v_dBw = IB/(nqd)$.
\end{proof}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:1}
Converta \qty{1}{eV}, \qty{13.6}{eV} e \qty{1}{MeV} em joules. Velocidade de um
elétron e de um próton acelerados do repouso por \qty{1.0}{kV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:2}
Um elétron de \qty{1.0}{keV} entra perpendicularmente num campo de \qty{1.0}{mT}.
Raio, \omterm{def:b1:wave-propagation:signal}{frequência angular} ciclotrônica, período.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:3}
Um próton a \qty{1.0e5}{m/s} no campo da Terra (\qty{50}{\micro T}):
raio do seu círculo. O que acontece se a sua velocidade fizer $30^\circ$ com
o campo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:4}
Um \omterm{prop:b1:charged-particles:spectrometer}{seletor de velocidades} tem $E = \qty{10}{kV/m}$ e $B = \qty{20}{mT}$. Que
velocidade passa? O que acontece com íons mais rápidos, e com a mesma velocidade e
carga oposta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:5}
Osciloscópio: elétrons a \qty{2.0}{kV}, placas de \qty{4.0}{cm} de comprimento,
separadas por \qty{1.0}{cm}, com \qty{100}{V} entre elas, e tela a \qty{30}{cm} do
centro das placas. Deflexão na tela e sensibilidade em mm/V.
Por que uma \omterm{prop:b1:charged-particles:efield}{tensão de aceleração} maior reduz a sensibilidade?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:6}
Íons $^{12}$C e $^{13}$C de carga unitária, acelerados por
\qty{20}{kV}, entram num campo de \qty{0.50}{T} ($\qty{1}{u} = \qty{1.66e-27}{kg}$).
Raios e separação das duas manchas após meia volta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:7}
Um cíclotron de prótons tem $B = \qty{1.5}{T}$ e $R_{\max} = \qty{0.50}{m}$.
Frequência da \omterm{def:b1:dc-circuits:voltage}{tensão} dos dês, \omterm{thm:b1:work-and-energy:ke}{energia cinética} máxima, número de voltas
para \qty{50}{kV} nos dês, e tempo passado dentro. Em que fração
a massa do próton aumentou na saída ($\gamma - 1 \approx E_k/mc^2$,
$mc^2 = \qty{938}{MeV}$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:8}
Um elétron a \qty{1.0e6}{m/s} entra num campo de \qty{2.0}{mT} a $30^\circ$
das linhas de campo. Raio e passo da sua hélice; número de voltas por
metro ao longo do campo.
\end{exercise}

\section{Problema: prótons contra um tumor}

\begin{problem}[{Problema de fim de semana --- o cíclotron atrás da parede de um
hospital: como os prótons são levados a um terço da velocidade da luz,
por que a máquina simples deixa de funcionar antes da energia de que um terapeuta
precisa, e quantos prótons são necessários para tratar um tumor}]
\label{pb:b1:charged-particles:1}
Prótons: $m = \qty{1.67e-27}{kg}$, $q = e = \qty{1.60e-19}{C}$, $mc^2 =
\qty{938}{MeV}$; $c = \qty{3.00e8}{m/s}$. Cíclotron: $B = \qty{1.5}{T}$,
raio de extração $R_{\max} = \qty{0.50}{m}$, \omterm{def:b1:wave-propagation:signal}{amplitude} da \omterm{def:b1:dc-circuits:voltage}{tensão} dos dês
$U = \qty{50}{kV}$.

\medskip
\noindent\textbf{Parte I --- Uma volta.}
\begin{enumerate}
  \item Escreva a \omterm{def:b1:charged-particles:lorentz}{força de Lorentz} sobre um próton que se move com $\vect v$
        perpendicular a $\vect B$, e mostre que o módulo da sua velocidade não pode mudar
        dentro de um dê.
  \item Deduza o raio do seu círculo e o seu período; mostre que o
        período não depende da velocidade.
  \item Calcule a \omterm{thm:b1:charged-particles:bfield}{frequência ciclotrônica} $f_c$ para esses prótons.
  \item Energia ganha em cada travessia do intervalo; e por volta.
  \item Por que a \omterm{def:b1:dc-circuits:voltage}{tensão} dos dês deve alternar exatamente em $f_c$? O que acontece
        se ela estiver ligeiramente fora?
  \item Os prótons são injetados no centro com velocidade desprezível:
        qual o raio após a primeira volta? E após $n$ voltas?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Até a borda.}
\begin{enumerate}[resume]
  \item \omterm{thm:b1:work-and-energy:ke}{Energia cinética} na extração (em J e em MeV), e a velocidade, como
        fração de $c$.
  \item Número de voltas necessárias e tempo passado na máquina.
  \item Comprimento total do trajeto em espiral (soma das circunferências ---
        aproxime a soma por uma integral).
  \item O resultado da questão 7 depende de $U$? Sobre o que $U$
        age?
  \item O vácuo interno precisa ser bom: estime a distância que um próton
        percorre entre colisões com moléculas de gás residual, se o
        livre caminho médio deve exceder o comprimento da espiral; o livre caminho médio
        varia como $1/p$ e vale cerca de \qty{70}{nm} à pressão
        atmosférica. Que pressão é necessária?
  \item Calcule $\gamma - 1 \approx E_k/mc^2$ na extração e a
        variação relativa do período que dela resulta. Ao longo do número de
        voltas encontrado acima, de que fração de período o próton
        escorregou em relação à \omterm{def:b1:dc-circuits:voltage}{tensão}? Comente.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- 230 MeV.}
A protonterapia precisa de cerca de \qty{230}{MeV} para alcançar tumores a \qty{30}{cm}
de profundidade.
\begin{enumerate}[resume]
  \item Com a fórmula não relativística, que produto $BR$ daria
        \qty{230}{MeV}? Com $B = \qty{2.0}{T}$, que raio?
  \item Nessa energia $\gamma - 1 = 0.245$: o que acontece com o
        período ciclotrônico à medida que o próton ganha energia, e por que um
        cíclotron de frequência fixa falha?
  \item Existem dois remédios: variar a frequência da \omterm{def:b1:dc-circuits:voltage}{tensão} durante a
        aceleração (sincrocíclotron) ou fazer $B$ crescer com o raio de modo
        que $B/\gamma$ permaneça constante (cíclotron isócrono). Explique
        cada um em uma frase e aponte um inconveniente do primeiro.
  \item A quantidade de movimento relativística vale $p = \gamma mv$ e o raio $R =
        p/qB$ continua valendo: com $\gamma = 1.245$ e $v = 0.60c$, calcule
        $R$ para $B = \qty{2.0}{T}$ e compare com a questão 13.
  \item O feixe extraído passa entre duas placas de \qty{1.0}{m} de comprimento
        com $E = \qty{1.0e5}{V/m}$ para ser desviado de um pequeno ângulo:
        estime o ângulo de deflexão (de modo não relativístico, usando o $p$
        da questão anterior no lugar de $mv$). O desvio elétrico é prático aqui?
  \item O mesmo desvio com um ímã de \qty{1.0}{T} ao longo de \qty{1.0}{m}:
        qual o ângulo? Conclua por que as linhas de feixe usam ímãs.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A dose.}
\begin{enumerate}[resume]
  \item Um próton de \qty{230}{MeV} deposita a maior parte da sua energia perto do fim
        do seu alcance (o pico de Bragg). Se toda ela for absorvida num
        tumor de massa \qty{0.50}{kg}, quantos prótons entregam uma dose de
        \qty{2.0}{Gy} ($\qty{1}{Gy} = \qty{1}{J/kg}$)?
  \item A corrente do feixe vale \qty{1.0}{nA}: prótons por segundo e
        duração da sessão.
  \item Potência transportada pelo feixe; compare com uma lâmpada.
  \item Os prótons são freados até parar dentro do paciente: qual é a
        energia depositada por próton no último centímetro, grosso modo, se
        um quarto da energia se perde ali? Em \omterm{def:b1:charged-particles:eV}{elétron-volts} por
        micrômetro?
  \item Cada ionização no tecido custa cerca de \qty{30}{eV}: quantas
        ionizações um próton provoca ao longo de todo o seu trajeto?
  \item Por que um feixe de prótons é preferível a raios X para um tumor profundo, em
        uma frase sobre para onde vai a energia?
  \item Resuma a cadeia, da \omterm{def:b1:charged-particles:lorentz}{força de Lorentz} à dose: que
        equações fixaram a energia, o raio, o número de voltas e
        o tempo de tratamento.
\end{enumerate}
\end{problem}
