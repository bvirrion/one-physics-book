\chapter{Mudanças de fase de um corpo puro}\label{ch:b1:phase-changes}

A água é a única substância que todo mundo já viu nos três estados, e é o
fluido de trabalho da maior parte das usinas do mundo, o refrigerante das
primeiras máquinas de gelo e o motor do tempo meteorológico. O gelo derrete
a uma temperatura que não se mexe enquanto o calor entra; uma chaleira ferve
a \qty{100}{\degreeCelsius} à beira-mar e a \qty{70}{\degreeCelsius} no
Everest; uma garrafa lacrada de água muito fria congela de uma vez a uma
batidinha. Este capítulo desenha o mapa das \omterm{def:b1:phase-changes:phase}{fases} de um \omterm{def:b1:phase-changes:phase}{corpo puro} --- os
diagramas $(T, P)$ e $(v, P)$ --- e dá as três ferramentas que o acompanham:
a regra da alavanca (quanto há de cada \omterm{def:b1:phase-changes:phase}{fase}), o \omterm{def:b1:phase-changes:phase}{calor latente} (o que uma
mudança custa, em energia e em entropia) e a \omterm{thm:b1:phase-changes:clapeyron}{relação de
Clausius--Clapeyron} (como se inclinam as linhas de fronteira). E termina
onde o mapa falha: nos \omterm{def:b1:phase-changes:metastable}{estados metaestáveis}, que é de onde vêm o tempo
meteorológico e as surpresas da panela de pressão.

\section{Fases e o diagrama $(T, P)$}

\begin{definition}[Corpo puro, fase, mudança de fase]\label{def:b1:phase-changes:phase}
\index{fase}\index{mudança de fase}\index{corpo puro}\index{calor latente}
Um \emph{corpo puro} é um corpo de uma única espécie química. Uma
\emph{fase} é uma parte dele homogênea nas suas propriedades físicas:
sólido, líquido, gás (\emph{vapor} quando coexiste com o seu líquido). Uma
\emph{mudança de fase} (fusão/solidificação, vaporização/condensação,
sublimação/deposição) é a passagem de uma fase a outra; a pressão fixa ela
ocorre a uma temperatura fixa, com troca de calor mas sem variação de
temperatura: é o \emph{calor latente}.
\end{definition}

\begin{proposition}[Diagrama de equilíbrio]\label{prop:b1:phase-changes:PTdiagram}
\index{diagrama de fases}\index{ponto triplo}\index{ponto crítico}\index{pressão de vapor saturante}
No plano $(T, P)$ cada \omterm{def:b1:phase-changes:phase}{fase} ocupa uma região; duas \omterm{def:b1:phase-changes:phase}{fases} coexistem
apenas ao longo de uma curva (sublimação, fusão, vaporização), e as três
apenas num ponto, o \emph{\omterm{prop:b1:phase-changes:PTdiagram}{ponto triplo}}. A curva de vaporização
$P = P_s(T)$ --- a \emph{\omterm{prop:b1:phase-changes:PTdiagram}{pressão de vapor saturante}} --- termina no
\emph{\omterm{prop:b1:phase-changes:PTdiagram}{ponto crítico}} $(T_c, P_c)$, além do qual líquido e gás já não se
distinguem. Para a água: \omterm{prop:b1:phase-changes:PTdiagram}{ponto triplo} \qty{273.16}{K}, \qty{611}{Pa};
\omterm{prop:b1:phase-changes:PTdiagram}{ponto crítico} \qty{647}{K}, \qty{221}{bar}. A curva de fusão da água
pende para a esquerda (o gelo é menos denso que a água); para quase todas
as outras substâncias ela pende para a direita.
\end{proposition}
\admitted

\begin{remark}[Ebulição, evaporação e a panela de pressão]\label{rem:b1:phase-changes:boiling}
Um líquido \emph{evapora} a qualquer temperatura pela sua superfície,
desde que a pressão parcial do seu vapor acima dela esteja abaixo de
$P_s(T)$; ele \emph{ferve} quando $P_s(T)$ alcança a pressão ambiente, de
modo que bolhas de vapor podem crescer no seu interior. A água ferve a
\qty{100}{\degreeCelsius} porque $P_s(\qty{100}{\degreeCelsius}) = \qty{1.013}{bar}$;
a \qty{0.33}{bar} no Everest ela ferve perto de \qty{70}{\degreeCelsius}, e
numa panela de pressão a \qty{1.8}{bar}, perto de \qty{117}{\degreeCelsius}. A
temperatura de um líquido em ebulição é fixada pela pressão --- fato que
serviu para definir a escala Celsius durante dois séculos.
\end{remark}

\section{O diagrama $(v, P)$ e a regra da alavanca}

\begin{proposition}[Isotermas de Andrews]\label{prop:b1:phase-changes:andrews}
\index{isotermas de Andrews}\index{curva de saturação}
No plano de Clapeyron $(v, P)$ do volume específico, uma isoterma
$T < T_c$ tem três partes: um ramo íngreme (líquido, quase
incompressível), um \emph{patamar} horizontal em $P = P_s(T)$ onde líquido
e vapor coexistem, e um ramo parecido com uma hipérbole (vapor). As
extremidades do patamar são os volumes específicos $v_l(T)$ e $v_v(T)$ do
líquido e do vapor saturados; o seu lugar geométrico é a \emph{\omterm{prop:b1:phase-changes:andrews}{curva de
saturação}} (a cúpula), cujo cume é o \omterm{prop:b1:phase-changes:PTdiagram}{ponto crítico}, onde a isoterma $T_c$
tem uma inflexão horizontal. Acima de $T_c$ não existe patamar.
\end{proposition}
\admitted

\begin{theorem}[Regra da alavanca]\label{thm:b1:phase-changes:lever}
\index{regra da alavanca}\index{título (fração mássica de vapor)}
Uma massa $m$ de volume específico $v$ sobre o patamar a $T$ se reparte numa
fração mássica de vapor $x = m_v/m$ (o \emph{título}) e numa fração
líquida $1 - x$, com
\[
v = (1 - x)v_l + x\,v_v , \qquad\text{isto é,}\qquad
x = \frac{v - v_l}{v_v - v_l} = \frac{LM}{LV} ,
\]
a razão entre os segmentos que o ponto de estado $M$ corta no patamar.
A mesma regra vale para toda grandeza extensiva por unidade de massa ($u$,
$h$, $s$): $h = (1 - x)h_l + xh_v$.
\end{theorem}

\begin{proof}
Os volumes se somam: $V = m_lv_l + m_vv_v$; divida por $m$. Energia, \omterm{def:b1:first-law:enthalpy}{entalpia}
e entropia também são extensivas.
\end{proof}

\begin{example}[Uma chaleira e uma caldeira]\label{ex:b1:phase-changes:lever}
Água a \qty{100}{\degreeCelsius} e \qty{1}{atm}: $v_l = \qty{1.04e-3}{m^3/kg}$,
$v_v = \qty{1.67}{m^3/kg}$ --- uma razão de $1600$. Um recipiente fechado de \qty{1.0}{L}
que contém \qty{0.50}{kg} de água a essa temperatura tem $v =
\qty{2.0e-3}{m^3/kg}$ e $x = (2.0 - 1.04) \times 10^{-3}/1.67 = 5.8 \times 10^{-4}$:
\qty{0.29}{g} de vapor ocupando metade do recipiente. Dobrar a massa de
vapor num recipiente lacrado mal desloca o ponto de estado; a pressão é
fixada só pela temperatura.
\end{example}

\section{Energética de uma mudança de fase}

\begin{proposition}[Calor latente, entalpia e entropia de transição]\label{prop:b1:phase-changes:latent}
\index{entalpia de transição}\index{entropia de transição}
À temperatura $T$ e à pressão $P_s(T)$ de coexistência, a
transição de uma unidade de massa da \omterm{def:b1:phase-changes:phase}{fase} 1 para a \omterm{def:b1:phase-changes:phase}{fase} 2 é reversível,
isotérmica e isobárica, e
\[
L_{12}(T) = h_2 - h_1 , \qquad s_2 - s_1 = \frac{L_{12}}{T} , \qquad
u_2 - u_1 = L_{12} - P_s(v_2 - v_1) .
\]
$L_{12} > 0$ quando se vai para a \omterm{def:b1:phase-changes:phase}{fase} menos ordenada (fusão,
vaporização, sublimação), e $L_{21} = -L_{12}$. Para a água a \qty{1}{atm}:
$L_f = \qty{334}{kJ/kg}$ a \qty{0}{\degreeCelsius} e $L_v = \qty{2257}{kJ/kg}$ a
\qty{100}{\degreeCelsius}.
\end{proposition}

\begin{proof}
A pressão constante o calor recebido é $\Delta H$
(\cref{prop:b1:first-law:qvqp}); sendo reversível e isotérmico, ele vale
$T\Delta S$ (\cref{prop:b1:second-law-entropy:identity}); e $U = H - PV$.
\end{proof}

\begin{example}[Para onde vai o calor de vaporização]\label{ex:b1:phase-changes:whereheat}
Para a água a \qty{100}{\degreeCelsius}: $P_s(v_v - v_l) = 1.013 \times 10^5 \times
1.67 = \qty{169}{kJ/kg}$, logo $u_v - u_l = 2257 - 169 = \qty{2088}{kJ/kg}$:
$93\%$ do \omterm{def:b1:phase-changes:phase}{calor latente} vai para separar as moléculas e $7\%$
para empurrar a atmosfera. A entropia de vaporização,
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, é muito maior que a de fusão, $334/273 =
\qty{1.22}{kJ/(kg.K)}$: derreter afrouxa as moléculas, ferver as liberta.
\end{example}

\begin{method}[Calorimetria com mudanças de fase]\label{meth:b1:phase-changes:calorimetry}
Num vaso isolado a pressão constante, $\sum\Delta H_i = 0$. Suponha
o estado final (que \omterm{def:b1:phase-changes:phase}{fases} estão presentes, a que temperatura);
escreva o $\Delta H$ de cada corpo como calores sensíveis $mc\,\Delta T$ mais
\omterm{def:b1:phase-changes:phase}{calores latentes} $\pm mL$ para as mudanças por que ele passa; resolva; e
\emph{verifique a suposição}: um corpo que termine acima do seu ponto de
ebulição ou abaixo do seu ponto de fusão, ou uma massa negativa de uma
\omterm{def:b1:phase-changes:phase}{fase}, significa que a suposição estava errada --- o estado final é então uma
mistura à temperatura de transição, e a incógnita passa a ser a massa
transformada (\cref{ex:b1:first-law:ice}).
\end{method}

\begin{example}[Queimaduras de vapor]\label{ex:b1:phase-changes:steam}
\qty{10}{g} de vapor a \qty{100}{\degreeCelsius} que condensam na pele cedem
$0.010 \times 2257 = \qty{22.6}{kJ}$ antes de ter esfriado um grau ---
tanto quanto \qty{10}{g} de água fervente cedem ao esfriar
\qty{540}{K}. É por isso que o vapor escalda muito pior do que a água à
mesma temperatura, e por que um radiador a vapor precisa de tão pouca vazão.
\end{example}

\section{A relação de Clausius--Clapeyron}

\begin{theorem}[Clausius--Clapeyron]\label{thm:b1:phase-changes:clapeyron}
\index{relação de Clausius--Clapeyron}
Ao longo de uma curva de coexistência das \omterm{def:b1:phase-changes:phase}{fases} 1 e 2,
\[
\frac{\dd P_s}{\dd T} = \frac{L_{12}(T)}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
Leve uma unidade de massa por um \omterm{def:b1:heat-engines:carnotcycle}{ciclo de Carnot} infinitesimal dentro da
cúpula: transição completa $1 \to 2$ ao longo do patamar a $T + \dd T$, pressão
$P_s + \dd P_s$ (calor recebido $L_{12}$ em primeira ordem); um passo
adiabático infinitesimal para baixo até $T$; a transição inversa ao longo do patamar a
$T$, $P_s$; e um passo adiabático de volta. O ciclo é reversível e ditérmico
entre $T + \dd T$ e $T$, de modo que o seu \omterm{def:b1:heat-engines:efficiency}{rendimento} vale $\dd T/T$
(\cref{thm:b1:heat-engines:carnot}); o seu trabalho, a área do
retângulo fino, vale $\dd P_s\,(v_2 - v_1)$. Logo $\dd P_s\,(v_2 - v_1)/L_{12} =
\dd T/T$.
\end{proof}

\begin{corollary}[Pressão de vapor saturante de um líquido]\label{cor:b1:phase-changes:vapourpressure}
Se o vapor for um \omterm{def:b1:kinetic-theory:model}{gás perfeito} e $v_v \gg v_l$, de modo que $v_v - v_l \approx
RT/(MP_s)$, e se $L_v$ variar pouco,
\[
\frac{\dd P_s}{P_s} = \frac{ML_v}{R}\,\frac{\dd T}{T^2} , \qquad
\ln\frac{P_s(T)}{P_s(T_0)} = \frac{ML_v}{R}\Bigl(\frac{1}{T_0} - \frac{1}{T}\Bigr) :
\]
a pressão de vapor cresce aproximadamente de modo exponencial com a
temperatura. Para a água perto de \qty{100}{\degreeCelsius}, $ML_v/R = 0.018 \times 2.257 \times 10^6/
8.314 = \qty{4890}{K}$ e $\dd P_s/\dd T = 2.257 \times 10^6/(373 \times 1.67) =
\qty{3.6}{kPa/K}$: cada \omterm{def:b1:kinetic-theory:temperature}{kelvin} a mais na temperatura de ebulição custa
\qty{36}{mbar} de pressão a mais.
\end{corollary}

\begin{proof}
Substitua $v_v - v_l \approx RT/(MP_s)$ no teorema e separe as
variáveis; integre com $L_v$ constante.
\end{proof}

\begin{example}[O gelo sob um patim]\label{ex:b1:phase-changes:skate}
Para o par gelo--água, $L_f = \qty{334}{kJ/kg}$ e $v_l - v_s = (1.000 - 1.091)

fina camada superficial do gelo já é quase líquida por si só, e o atrito

\end{example}

\section{Estados metaestáveis}

\begin{definition}[Super-resfriamento, superaquecimento]\label{def:b1:phase-changes:metastable}
\index{super-resfriamento}\index{superaquecimento}\index{estado metaestável}\index{nucleação}
Uma \omterm{def:b1:phase-changes:phase}{fase} pode persistir além da sua fronteira de equilíbrio: água líquida
esfriada abaixo de \qty{0}{\degreeCelsius} sem congelar (\emph{super-resfriamento}),
aquecida acima do seu ponto de ebulição sem ferver (\emph{superaquecimento}),
vapor comprimido além de $P_s$ sem condensar (supersaturação).
Tais \emph{estados metaestáveis} exigem a ausência de sítios de
\emph{nucleação} --- poeira, riscos, gás dissolvido, cristais de gelo --- em
que a nova \omterm{def:b1:phase-changes:phase}{fase} possa começar; uma perturbação os leva de repente e
irreversivelmente ao equilíbrio.
\end{definition}

\begin{example}[A garrafa super-resfriada]\label{ex:b1:phase-changes:supercooled}
Água parada a \qty{-8}{\degreeCelsius} numa garrafa limpa, ao levar uma
batida: cristais de gelo a atravessam e a temperatura salta para
\qty{0}{\degreeCelsius}. O processo é adiabático e rápido, de modo que o calor
sensível $c\,\Delta T$ congela uma fração $x = c\,\Delta T/L_f = 4.18 \times 8/334 =
0.10$ da água; a entropia criada, $c\ln(273/265) - xL_f/273 =
124 - 122 = \qty{+2}{J/(kg.K)}$, é positiva, como tem de ser. As nuvens estão
cheias de gotículas super-resfriadas até \qty{-40}{\degreeCelsius}; a formação
de gelo nas aeronaves e a semeadura de nuvens exploram as duas esse fato.
\end{example}

\begin{remark}[Pedras de ebulição e câmaras de bolhas]\label{rem:b1:phase-changes:nucleation}
Uma bolha de raio $r$ num líquido à pressão $P$ contém vapor a
$P + 2\sigma/r$ (tensão superficial $\sigma$, volume do segundo ano); ela só pode
crescer se $P_s(T) > P + 2\sigma/r$, de modo que um líquido limpo no seu ponto
de ebulição não tem bolha pequena o bastante para começar e superaquece
alguns \omterm{def:b1:kinetic-theory:temperature}{kelvins} antes de irromper --- são os solavancos que as pedras de
ebulição evitam, oferecendo cavidades cheias de gás preso. A câmara de
bolhas da física de partículas funcionava com a mesma instabilidade ao
contrário: um líquido superaquecido ferve primeiro ao longo do rastro de uma
partícula ionizante.
\end{remark}

\begin{example}[Umidade e orvalho]\label{ex:b1:phase-changes:dew}
\index{umidade relativa}\index{ponto de orvalho}
O ar contém vapor de água a uma pressão parcial $P_v \leq P_s(T)$; a
\emph{\omterm{ex:b1:phase-changes:dew}{umidade relativa}} é $P_v/P_s(T)$, e o \emph{\omterm{ex:b1:phase-changes:dew}{ponto de orvalho}}, a
temperatura em que $P_s$ cai até $P_v$. O ar a \qty{25}{\degreeCelsius}
($P_s = \qty{3.17}{kPa}$) com $60\%$ de umidade tem $P_v = \qty{1.9}{kPa}$,
$\rho_v = P_vM/(RT) = \qty{14}{g/m^3}$, e orvalha em qualquer superfície abaixo
de uns \qty{17}{\degreeCelsius} --- a grama ao amanhecer, a garrafa gelada, o
espelho do banheiro. Ao subir, esse mesmo ar esfria adiabaticamente a
\qty{9.8}{K/km} (\cref{pb:b1:phase-changes:1}) e chega ao seu \omterm{ex:b1:phase-changes:dew}{ponto de orvalho}
mais ou menos um quilômetro acima: é ali que ficam as bases planas dos cúmulos.
\end{example}

\end{exercise}

\section{Problema: a água em quatro cenas}

\begin{problem}[{Problema de fim de semana --- a panela de pressão, a nuvem,
o recipiente lacrado e a garrafa super-resfriada: uma substância, quatro
usos do seu diagrama de fases}]
\label{pb:b1:phase-changes:1}
Dados da água: $L_v = \qty{2257}{kJ/kg}$ a \qty{100}{\degreeCelsius} e
cerca de \qty{2.2}{MJ/kg} a \qty{117}{\degreeCelsius}, \qty{2.45}{MJ/kg} perto de
\qty{20}{\degreeCelsius}; $L_f = \qty{334}{kJ/kg}$; $c = \qty{4.18}{kJ/(kg.K)}$
(líquido); $M = \qty{18}{g/mol}$; $ML_v/R = \qty{4890}{K}$ perto de
\qty{100}{\degreeCelsius}. Pressões de saturação: \qty{0.61}{kPa}
(\qty{0}{\degreeCelsius}), \qty{1.23}{kPa} (\qty{10}{\degreeCelsius}), \qty{1.70}{kPa} (\qty{15}{\degreeCelsius}),
\qty{2.34}{kPa} (\qty{20}{\degreeCelsius}), \qty{3.17}{kPa} (\qty{25}{\degreeCelsius}), \qty{4.24}{kPa}
(\qty{30}{\degreeCelsius}), \qty{101.3}{kPa} (\qty{100}{\degreeCelsius}).
Volumes específicos saturados: $v_l = \qty{1.0e-3}{m^3/kg}$ (água fria),
\qty{1.9e-3}{m^3/kg} a \qty{360}{\degreeCelsius}; $v_v = \qty{0.0217}{m^3/kg}$
a \qty{300}{\degreeCelsius}, $0.0108$ a \qty{340}{\degreeCelsius}, $0.0088$
a \qty{350}{\degreeCelsius}; \omterm{prop:b1:phase-changes:PTdiagram}{ponto crítico} \qty{374}{\degreeCelsius},
\qty{221}{bar}, $v_c = \qty{3.1e-3}{m^3/kg}$. Ar: $M = \qty{29}{g/mol}$, $c_p =
\qty{1.0}{kJ/(kg.K)}$, $\gamma = 1.4$; $g = \qty{9.8}{m/s^2}$; $R =
\qty{8.314}{J/(mol.K)}$.

\medskip
\noindent\textbf{Parte I --- A panela de pressão.}
\begin{enumerate}
  \item Por que a água numa panela aberta ao nível do mar não pode ser
        aquecida acima de \qty{100}{\degreeCelsius}, por mais forte que seja a chama?
  \item Partindo da \omterm{thm:b1:phase-changes:clapeyron}{relação de Clausius--Clapeyron}, deduza
        $\ln[P_s(T)/P_s(T_0)] = (ML_v/R)(1/T_0 - 1/T)$, dizendo quais são as
        aproximações.
  \item Temperatura numa panela cuja válvula abre a \qty{1.8}{bar}
        absolutos.
  \item A válvula é um peso apoiado sobre um orifício de \qty{20}{mm^2};
        pressão atmosférica \qty{1.0}{bar}: massa do peso.
  \item \qty{2.0}{L} de água são levados de \qty{20}{\degreeCelsius} à
        temperatura de trabalho por uma chapa de \qty{2.0}{kW} (despreze a panela):
        calor e tempo.
  \item Uma vez ali, a chapa é deixada em \qty{2.0}{kW}: massa de vapor
        que sai pela válvula por minuto. O que o cozinheiro deve fazer?
  \item As reações do cozimento mais ou menos dobram de velocidade a cada \qty{10}{K}:
        por que fator a panela encurta uma receita de \qty{30}{min} ao nível
        do mar? E quanto tempo levaria a receita no Everest, onde a
        água ferve a \qty{70}{\degreeCelsius}?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A nuvem.}
\begin{enumerate}[resume]
  \item Defina \omterm{ex:b1:phase-changes:dew}{umidade relativa}. Ar a \qty{25}{\degreeCelsius} com $60\%$:
        pressão parcial do vapor e massa de vapor por metro cúbico
        (\omterm{def:b1:kinetic-theory:model}{gás perfeito}).
  \item \omterm{ex:b1:phase-changes:dew}{Ponto de orvalho} desse ar (interpole a tabela). Explique o orvalho
        na grama ao amanhecer e as gotas numa garrafa gelada.
  \item Uma porção de ar seco sobe sem trocar calor. A partir da lei
        adiabática reversível e da \omterm{thm:b1:fluid-statics:fundamental}{relação hidrostática}
        $\dd P/\dd z = -\rho g$, mostre que $\dd T/\dd z = -g/c_p$ e
        avalie o valor.
  \item O próprio \omterm{ex:b1:phase-changes:dew}{ponto de orvalho} da porção que sobe cai cerca de
        \qty{2}{K/km} à medida que a pressão baixa. Altura em que a porção da
        questão~8 fica saturada --- a base da nuvem.
  \item Acima da base, o vapor que condensa libera o seu \omterm{def:b1:phase-changes:phase}{calor latente}
        dentro da porção. De quanto condensar \qty{1}{g} de
        água por quilograma de ar aquece esse ar? O que isso faz com a
        taxa de resfriamento da porção enquanto ela continua a subir, e por que
        isso importa para as tempestades?
  \item Um cúmulo de \qty{1}{km^3} contém \qty{1}{g} de água líquida por metro
        cúbico: massa de água e energia liberada pela sua
        condensação, em joules e em \unit{GW h}.
  \item Um espelho de banheiro a \qty{18}{\degreeCelsius}, ar a
        \qty{25}{\degreeCelsius} com $80\%$: ele embaça? Com que umidade
        ele deixaria de embaçar?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O recipiente lacrado.}
Um recipiente rígido de aço de \qty{1.0}{L} é enchido com uma massa $m$ de água
e lacrado sem ar, e depois aquecido lentamente.
\begin{enumerate}[resume]
  \item $m = \qty{0.50}{kg}$ a \qty{20}{\degreeCelsius}: pressão dentro,
        volume específico, massa de vapor (use o \omterm{def:b1:kinetic-theory:model}{gás perfeito} para $v_v$)
        e fração do volume ocupada pelo líquido.
  \item Aquecido: o nível do líquido sobe ou desce? A que temperatura,
        mais ou menos, o líquido enche o recipiente, e o que acontece com
        a pressão depois disso? (Use $v_l$ a \qty{360}{\degreeCelsius}.)
  \item $m = \qty{0.10}{kg}$: o nível do líquido sobe ou desce com o
        aquecimento? A que temperatura, mais ou menos, evapora a última
        gota (use a tabela de $v_v$)?
  \item $m = \qty{0.31}{kg}$: descreva o que um observador vê no
        menisco quando a temperatura se aproxima de \qty{374}{\degreeCelsius}.
  \item Esboce os três caminhos de aquecimento no diagrama $(v, P)$ com a
        cúpula de saturação, e diga de que lado do \omterm{prop:b1:phase-changes:PTdiagram}{ponto crítico} cada um
        passa.
  \item O mesmo recipiente de \qty{1.0}{L} com \qty{0.50}{kg} de água é agora
        lacrado \emph{com} o ar que ele continha a \qty{20}{\degreeCelsius} e
        \qty{1.0}{bar}, e aquecido a \qty{100}{\degreeCelsius}: pressão total
        interna (Dalton) e força sobre uma tampa de \qty{100}{cm^2}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A garrafa super-resfriada.}
\begin{enumerate}[resume]
  \item Uma garrafa de água limpa e parada esfriou até \qty{-8}{\degreeCelsius}
        sem congelar. Como se chama esse estado, por que ele é possível
        e o que acontece quando a garrafa leva uma batida? Por que a
        temperatura se estabiliza então exatamente em \qty{0}{\degreeCelsius}?
  \item Fração da água que congela (o processo é rápido, portanto
        adiabático).
  \item Entropia criada por quilograma; verifique o sinal.
  \item A surpresa inversa: uma xícara de água aquecida a \qty{105}{\degreeCelsius}
        num forno de micro-ondas sem ferver, e depois perturbada. Fração
        da água que vira vapor de repente, volume de vapor produzido por
        litro ($v_v = \qty{1.67}{m^3/kg}$) e por que a xícara transborda com violência.
  \item \omterm{thm:b1:heat-engines:lostwork}{Trabalho perdido}: o litro super-resfriado poderia ter acionado uma
        máquina reversível com a vizinhança a \qty{0}{\degreeCelsius}; tomando
        $T_0 = \qty{273}{K}$, o trabalho jogado fora ao deixá-lo congelar com uma
        batida vale $T_0S_{\mathrm{created}}$. Avalie-o por litro e
        compare com erguer a garrafa em um metro. Conclua: o que armazena
        um \omterm{def:b1:phase-changes:metastable}{estado metaestável}?
\end{enumerate}
\end{problem}
