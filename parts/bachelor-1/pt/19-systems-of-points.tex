\chapter{Sistemas de pontos; introdução aos corpos rígidos}\label{ch:b1:systems-of-points}

Uma saltadora deixa o trampolim, encolhe-se, dá dois mortais e se estica;
durante tudo isso um ponto do seu corpo --- o seu \omterm{def:b1:systems-of-points:cm}{centro de massa} ---
descreve a mesma parábola simples que uma pedra descreveria. Um ioiô solto
da mão leva três vezes mais tempo para chegar ao fim do seu cordão do que
levaria uma pedra, e chega girando a duas mil voltas por minuto. Uma esfera
maciça vence uma esfera oca em qualquer rampa, sejam quais forem os seus
tamanhos ou as suas massas. A mecânica de um único ponto, desenvolvida nos
capítulos anteriores, estende-se a sistemas de muitos pontos --- e aos
\omterm{def:b1:systems-of-points:solid}{corpos rígidos} da vida cotidiana --- por meio de um punhado de teoremas: o
\omterm{def:b1:systems-of-points:cm}{centro de massa} se move como um ponto, a energia e o \omterm{def:b1:angular-momentum:L}{momento angular} se
separam numa parte do \omterm{def:b1:systems-of-points:cm}{centro de massa} e numa parte ``em torno'' dele, e um
corpo que gira em torno de um eixo fixo é um sistema de um grau de
liberdade governado pelo seu \omterm{prop:b1:angular-momentum:inertia}{momento de inércia}.

\section{Centro de massa e quantidade de movimento}

\begin{definition}[Centro de massa; quantidade de movimento de um sistema]\label{def:b1:systems-of-points:cm}
\index{centro de massa}\index{baricentro}
Para partículas de massas $m_i$ em $M_i$, de massa total $M = \sum m_i$, o
\emph{centro de massa} (baricentro) $G$ é definido por
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{isto é,}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
qualquer que seja a origem $O$. A \emph{quantidade de movimento} do sistema é $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[Teorema do centro de massa]\label{thm:b1:systems-of-points:cmtheorem}
\index{teorema do centro de massa}\index{conservação da quantidade de movimento}
Num \omterm{thm:b1:newton-dynamics:laws}{referencial inercial},
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
o \omterm{def:b1:systems-of-points:cm}{centro de massa} se move como um \omterm{def:b1:point-kinematics:frame}{ponto material} de massa $M$ submetido à
resultante das forças \emph{externas} apenas. Para um \omterm{def:b1:newton-dynamics:conservation}{sistema fechado} sem
força externa, $\vect P$ se conserva e $G$ se move uniformemente.
\end{theorem}

\begin{proof}
Some a segunda lei de Newton sobre as partículas; as forças internas se
cancelam aos pares de \omterm{thm:b1:newton-dynamics:laws}{ação e reação}; $\sum m_i\vect a_i = M\vect a_G$
derivando duas vezes a definição de $G$.
\end{proof}

\begin{example}[Saltadora, recuo, colisão]\label{ex:b1:systems-of-points:cm}
\index{colisão}
O $G$ da saltadora segue uma parábola só sob o seu peso, por mais que ela
gire --- os seus músculos são forças internas. Um fuzil de \qty{4.0}{kg}
que dispara uma bala de \qty{10}{g} a \qty{800}{m/s}: $\vect P = \vect 0$ antes
e depois, de modo que o fuzil recua a \qty{2.0}{m/s}. Dois carros, \qty{1000}{kg}
a \qty{20}{m/s} e \qty{1500}{kg} parado, que ficam engatados: $v =
20000/2500 = \qty{8.0}{m/s}$ --- a quantidade de movimento se conserva, enquanto $60\%$ da
\omterm{thm:b1:work-and-energy:ke}{energia cinética} foi para o metal amassado e para o calor. A quantidade de
movimento se conserva em toda colisão; a \omterm{thm:b1:work-and-energy:ke}{energia cinética}, só nas
\emph{elásticas} (\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{Os teoremas de Koenig}

\begin{definition}[Referencial baricêntrico]\label{def:b1:systems-of-points:barycentric}
\index{referencial baricêntrico}
O \emph{\omterm{def:b1:point-kinematics:frame}{referencial} baricêntrico} $\mathcal R^*$ tem origem em $G$ e está
em translação em relação ao \omterm{thm:b1:newton-dynamics:laws}{referencial inercial} $\mathcal R$ (os seus eixos
mantêm direções fixas). As grandezas nele medidas levam asterisco: $\vect v_i^*
= \vect v_i - \vect v_G$, e $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[Teoremas de Koenig]\label{thm:b1:systems-of-points:koenig}
\index{teoremas de Koenig}
O \omterm{def:b1:angular-momentum:L}{momento angular} em relação a um ponto $O$ e a \omterm{thm:b1:work-and-energy:ke}{energia cinética} de um
sistema se separam na contribuição do seu \omterm{def:b1:systems-of-points:cm}{centro de massa}, que carrega
toda a massa, e nas contribuições no \omterm{def:b1:systems-of-points:barycentric}{referencial baricêntrico}:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
onde $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ e $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ não dependem da origem do \omterm{def:b1:point-kinematics:frame}{referencial}.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ e $\vect v_i = \vect v_G + \vect v_i^*$;
desenvolva $\sum\vect{OM_i}\wedge m_i\vect v_i$ e $\sum\tfrac12 m_iv_i^2$: os termos
cruzados contêm $\sum m_i\vect{GM_i} = \vect 0$ ou $\sum m_i\vect v_i^* = \vect 0$
e se anulam.
\end{proof}

\begin{example}[Uma roda que rola]\label{ex:b1:systems-of-points:rolling}
\index{rolamento sem deslizamento}
Uma roda de massa $M$, raio $R$ e \omterm{prop:b1:angular-momentum:inertia}{momento de inércia} $J$ em relação ao seu
eixo, que rola sem deslizar com velocidade $v$: o ponto de contato está em repouso, logo
$\omega = v/R$, e $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- três quartos de $Mv^2$ para um disco homogêneo ($J = \tfrac12 MR^2$),
$Mv^2$ para um aro. É por causa do segundo termo que as rodas de uma
bicicleta são feitas leves: a sua rotação custa uma energia que a massa do quadro não custa.
\end{example}

\section{O problema de dois corpos}

\begin{proposition}[Massa reduzida]\label{prop:b1:systems-of-points:twobody}
\index{problema de dois corpos}\index{massa reduzida}
Duas partículas que interagem apenas uma com a outra (força $\vect F$ sobre $m_2$
vinda de $m_1$, $-\vect F$ sobre $m_1$): $G$ se move uniformemente, e a posição
relativa $\vect r = \vect{M_1M_2}$ obedece a
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
que é a equação de uma única partícula fictícia de \emph{\omterm{prop:b1:systems-of-points:twobody}{massa reduzida}} $\mu$
que se move sob $\vect F$ em torno de um centro fixo. Em $\mathcal R^*$, $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ e $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. Em $\mathcal R^*$, $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ e
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$; substitua em $E_k^*$ e $\vect L^*$.
\end{proof}

\begin{example}[Quando o centro também se move]\label{ex:b1:systems-of-points:reduced}
No \cref{ch:b1:central-forces} o Sol estava fixo: exato apenas no
limite $m \ll M$; em geral a \omterm{thm:b1:central-forces:circular}{terceira lei de Kepler} se escreve $T^2 = 4\pi^2a^3/G(M +
m)$. Para a molécula de HCl em vibração do \cref{pb:b1:work-and-energy:1},
a massa que oscila no poço vale $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$, o que eleva a frequência de $1.4\%$; para o átomo de
hidrogênio, o $\mu$ do elétron difere de $m_e$ de uma parte em $1836$ ---
mensurável no espectro.
\end{example}

\section{Sólido em rotação em torno de um eixo fixo}

\begin{definition}[Sólido; rotação em torno de um eixo fixo]\label{def:b1:systems-of-points:solid}
\index{corpo rígido}\index{rotação em torno de um eixo fixo}
Um \emph{corpo rígido} (sólido) é um sistema cujos pontos mantêm fixas as
distâncias mútuas. Se ele gira em torno de um eixo $\Delta$ fixo no
\omterm{def:b1:point-kinematics:frame}{referencial}, todo ponto descreve um círculo centrado em $\Delta$ com a mesma
\omterm{cor:b1:point-kinematics:circular}{velocidade angular} $\omega = \dot\theta$; um ponto à distância $r_i$ do eixo
tem velocidade $r_i\omega$.
\end{definition}

\begin{proposition}[Momento de inércia; momento angular; energia cinética]\label{prop:b1:systems-of-points:inertia}
\index{momento de inércia!dos sólidos}\index{teorema de Huygens}
Com o \emph{\omterm{prop:b1:angular-momentum:inertia}{momento de inércia}} $J_\Delta = \sum m_ir_i^2$ (uma integral
$\int r^2\,\dd m$ para um corpo contínuo):
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
Valores usuais (massa $M$): anel fino ou cilindro oco em torno do seu eixo,
$MR^2$; disco homogêneo ou cilindro maciço, $\tfrac12 MR^2$; barra fina de
comprimento $L$ em torno do seu centro, $\tfrac{1}{12}ML^2$, e em torno de uma
extremidade, $\tfrac13 ML^2$; esfera maciça em torno de um diâmetro, $\tfrac25 MR^2$.
\emph{Teorema de Huygens}: em torno de um eixo $\Delta$ paralelo a um eixo
$\Delta_G$ que passa por $G$ à distância $d$, $J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
$L_\Delta$ e $E_k$: \cref{prop:b1:angular-momentum:inertia}. Barra em torno de
uma extremidade: $\int_0^L x^2(M/L)\,\dd x = ML^2/3$; em torno do centro, $\int_{-L/2}^{L/2}
x^2(M/L)\dd x = ML^2/12$. Disco: anel de raio $r$, largura $\dd r$, massa
$(2Mr/R^2)\dd r$: $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$; o anel e a
esfera são admitidos (a esfera exige uma integral dupla). Huygens: com
$r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2 = r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$, o termo cruzado soma zero pela definição de $G$.
\end{proof}

\begin{theorem}[Dinâmica de um sólido em torno de um eixo fixo]\label{thm:b1:systems-of-points:rotation}
\index{pivô}\index{potência de um momento}
Para um sólido que gira em torno de um eixo fixo $\Delta$ num \omterm{thm:b1:newton-dynamics:laws}{referencial inercial},
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
os momentos das forças externas em relação ao eixo governam a aceleração
angular, e um momento $\mathcal M_\Delta$ fornece a potência
$\mathcal M_\Delta\omega$. Um \emph{pivô} ideal (mancal sem atrito) não exerce
momento algum em torno do seu eixo e portanto nem acelera nem freia a
rotação.
\end{theorem}

\begin{proof}
\cref{thm:b1:angular-momentum:system} projetado sobre $\Delta$, com $L_\Delta
= J_\Delta\dot\theta$ e $J_\Delta$ constante para um \omterm{def:b1:systems-of-points:solid}{corpo rígido}; multiplique por
$\dot\theta$ para a forma em energia (as forças internas de um \omterm{def:b1:systems-of-points:solid}{corpo rígido} não
realizam trabalho: as distâncias não mudam).
\end{proof}

\begin{example}[Pêndulo físico; pêndulo de torção]\label{ex:b1:systems-of-points:pendulums}
\index{pêndulo físico}\index{pêndulo de torção}
Um sólido de massa $M$ que gira em torno de um eixo horizontal à distância $d$
de $G$: o momento do peso vale $-Mgd\sin\theta$, de modo que $J_\Delta\ddot\theta =
-Mgd\sin\theta$ e as \omterm{prop:b1:work-and-energy:stability}{pequenas oscilações} têm o período
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
o de um \omterm{ex:b1:newton-dynamics:pendulum}{pêndulo simples} de comprimento $J_\Delta/Md$ --- para uma barra
homogênea presa por uma extremidade, $2L/3$. Um disco pendurado num fio de
constante de torção $C$ (momento restaurador $-C\theta$): $J\ddot\theta = -C\theta$, $T = 2\pi\sqrt{J/C}$
--- é a balança de torção que mediu $G$ e a lei de Coulomb.
\end{example}

\begin{proposition}[Rolamento sem deslizamento numa rampa]\label{prop:b1:systems-of-points:rollingincline}
Um sólido de revolução ($J = kMR^2$ em torno do seu eixo) solto numa rampa
de ângulo $\alpha$ rola sem deslizar com a aceleração
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
$\tfrac57 g\sin\alpha$ para uma esfera maciça, $\tfrac23 g\sin\alpha$ para um disco,
$\tfrac12 g\sin\alpha$ para um aro --- independentemente da massa e do raio.
\end{proposition}

\begin{proof}
Energia: $Mgh = \tfrac12 Mv^2(1 + k)$ ao longo da rampa (o atrito estático
no ponto de contato, que está em repouso, não realiza trabalho), logo $v^2 = 2ah$ com o $a$
do enunciado; ou Newton para $G$ e o teorema do momento em relação a $G$ com o
atrito $f$: $Ma = Mg\sin\alpha - f$, $kMR^2\dot\omega = fR$, $a = R\dot\omega$.
\end{proof}

\begin{remark}[Sistemas deformáveis]\label{rem:b1:systems-of-points:deformable}
\index{sistema deformável}
Para um sistema que não é rígido, o teorema da \omterm{thm:b1:work-and-energy:ke}{energia cinética} se escreve
$\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$: as forças internas
realizam trabalho (músculos, molas, atrito dentro do sistema). A patinadora
que recolhe os braços, a saltadora que se encolhe, o ciclista que pedala em
pé ganham \omterm{thm:b1:work-and-energy:ke}{energia cinética} sem trabalho externo --- o teorema do \omterm{def:b1:systems-of-points:cm}{centro de
massa} continua verdadeiro, e a energia vem de dentro.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:1}
Massas de \qty{2.0}{kg} em $x = 0$ e de \qty{3.0}{kg} em $x = \qty{1.0}{m}$:
posição de $G$. O sistema Terra--Lua ($81:1$, \qty{384000}{km}):
distância de $G$ ao centro da Terra; compare com o raio da
Terra.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:2}
Um fuzil de \qty{4.0}{kg} dispara uma bala de \qty{10}{g} a \qty{800}{m/s}.
Velocidade de recuo; \omterm{thm:b1:work-and-energy:ke}{energias cinéticas} da bala e do fuzil; de onde veio a
energia?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:3}
Um carro de \qty{1000}{kg} a \qty{20}{m/s} bate num carro de \qty{1500}{kg} parado;
os dois ficam engatados. Velocidade comum; fração da \omterm{thm:b1:work-and-energy:ke}{energia cinética} perdida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:4}
Deduza o \omterm{prop:b1:angular-momentum:inertia}{momento de inércia} de uma barra homogênea de massa $M$ e
comprimento $L$ em torno de um eixo perpendicular que passa por uma
extremidade, e depois pelo centro; verifique entre os dois o teorema de Huygens.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:5}
\omterm{thm:b1:work-and-energy:ke}{Energia cinética} de um cilindro homogêneo que rola e de um aro que rola com
a mesma massa e a mesma velocidade, em múltiplos de $\tfrac12 Mv^2$. Qual
deles para primeiro num piso plano e rugoso, e qual sobe mais alto numa rampa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:6}
\omterm{prop:b1:systems-of-points:twobody}{Massa reduzida} do HCl ($m_{Cl} = 35m_H$) e do átomo de hidrogênio
($m_p = 1836m_e$); correção relativa a uma frequência de vibração ou de
órbita calculada só com a massa leve.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:7}
Uma barra homogênea de \qty{1.0}{m} gira livremente em torno de uma
extremidade. Período das \omterm{prop:b1:work-and-energy:stability}{pequenas oscilações}; comprimento do \omterm{ex:b1:newton-dynamics:pendulum}{pêndulo
simples} de mesmo período; o ponto da barra (o centro de percussão) em que
um golpe seco não produz reação no pivô fica a essa distância --- por que
um batedor o procura?
\end{exercise}

\section{Problema: o ioiô}

\begin{problem}[{Problema de fim de semana --- um disco num eixo fino na
ponta de um cordão: por que ele cai tão devagar, com que rapidez gira lá
embaixo, por que ele ``dorme'' ali e para que lado rola um carretel quando
se puxa o seu fio}]
\label{pb:b1:systems-of-points:1}
Um ioiô é um disco homogêneo de massa $M = \qty{50}{g}$ e raio $R =
\qty{3.0}{cm}$ com um eixo fino de raio $r = \qty{0.40}{cm}$ em torno do
qual se enrola um cordão de comprimento $h = \qty{1.0}{m}$; a ponta de cima
do cordão é mantida fixa. $g = \qty{9.81}{m/s^2}$; $J = \tfrac12 MR^2$ em torno do
eixo.

\medskip
\noindent\textbf{Parte I --- Cinemática e energia.}
\begin{enumerate}
  \item Onde fica o \omterm{def:b1:systems-of-points:cm}{centro de massa} $G$ do ioiô? Por quê?
  \item Mostre, pelo teorema de Koenig, que a sua \omterm{thm:b1:work-and-energy:ke}{energia cinética} é $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$.
  \item Deduza $J = \tfrac12 MR^2$ integrando sobre anéis e calcule
        $J$.
  \item À medida que o cordão se desenrola do eixo sem deslizar, relacione
        $v_G$ e $\omega$.
  \item Deduza $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ e calcule o fator
        entre parênteses. Comente.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A queda.}
\begin{enumerate}[resume]
  \item Forças sobre o ioiô; escreva o teorema do \omterm{def:b1:systems-of-points:cm}{centro de massa}
        (eixo vertical para baixo).
  \item Escreva o teorema do \omterm{def:b1:angular-momentum:L}{momento angular} em relação ao eixo que passa
        por $G$ (que forças têm momento?).
  \item Combine com a questão 4 para achar a aceleração $a_G = g/(1 +
        R^2/2r^2)$; calcule-a.
  \item Calcule a tração do cordão e compare com o peso.
  \item Tempo para desenrolar o \qty{1.0}{m} de cordão e a velocidade nesse
        instante; compare com uma pedra solta da mesma altura.
  \item \omterm{cor:b1:point-kinematics:circular}{Velocidade angular} lá embaixo, em rad/s e em voltas por minuto.
  \item Separe a \omterm{thm:b1:work-and-energy:ke}{energia cinética} lá embaixo nas suas partes de translação
        e de rotação.
  \item Verifique o balanço de energia: compare $Mgh$ com a \omterm{thm:b1:work-and-energy:ke}{energia cinética}
        lá embaixo, e diga por que a tração do cordão, embora grande,
        não realiza trabalho.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Lá embaixo: o ioiô que dorme.}
Quando o cordão está todo desenrolado, ele para o \omterm{def:b1:systems-of-points:cm}{centro de massa} em alguns
milissegundos; o ioiô continua girando na alça da ponta do cordão
(``dormindo'').
\begin{enumerate}[resume]
  \item Por que a rotação sobrevive à parada de $G$? (Que teorema, e que
        força teria sido necessária para pará-la?)
  \item Estime a tração média durante a parada, se $G$ é detido em
        \qty{5}{ms}.
  \item A alça roça no eixo com um momento de atrito $\Gamma =
        \qty{1.0e-4}{N.m}$: quanto tempo o ioiô dorme?
  \item Um puxão seco para cima no cordão faz o cordão agarrar de novo o
        eixo, e o ioiô girante então sobe. Pela energia, até que altura ele
        poderia subir se nada se perdesse? Por que, na prática, menos?
  \item Durante a subida, a tração é maior ou menor do que o peso? (Sinal
        de $a_G$.)
  \item O que aconteceria com um ioiô cujo eixo tivesse raio igual a $R$
        (cordão enrolado na borda)? Calcule a sua aceleração de queda.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O carretel sobre a mesa.}
O mesmo objeto está deitado sobre uma mesa rugosa, com o eixo horizontal, e
o seu fio sai horizontalmente por \emph{baixo} do eixo; puxa-se com uma
força $F$. Ele rola sem deslizar.
\begin{enumerate}[resume]
  \item No ponto de contato $C$ com a mesa, qual é a velocidade da matéria
        do carretel? Deduza que o movimento é, a cada instante, uma
        rotação em torno de $C$.
  \item Calcule o momento de $F$ em relação a $C$ (\omterm{def:b1:angular-momentum:moment}{braço de alavanca} $R - r$)
        e os do peso e da reação da mesa.
  \item Usando o teorema do \omterm{def:b1:angular-momentum:L}{momento angular} em relação a $C$ com $J_C = J +
        MR^2$ (Huygens), ache $a_G$ e mostre que o carretel rola
        \emph{na direção} da mão.
  \item Agora o fio é puxado formando um ângulo $\beta$ acima da
        horizontal. Mostre que o carretel não rola quando a linha de ação
        de $F$ passa por $C$, isto é, $\cos\beta = r/R$, e que ele se
        afasta da mão para puxões mais inclinados. Calcule esse ângulo.
  \item Calcule $a_G$ para $F = \qty{0.20}{N}$ puxado horizontalmente e a
        força de atrito necessária; o que limita $F$ antes que o carretel
        deslize?
  \item Resuma: os três teoremas usados (\omterm{def:b1:systems-of-points:cm}{centro de massa}, \omterm{def:b1:angular-momentum:L}{momento angular}
        em relação a um eixo, energia) e o que cada um decidiu.
\end{enumerate}
\end{problem}
