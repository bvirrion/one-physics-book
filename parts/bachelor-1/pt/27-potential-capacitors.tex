\chapter{Potencial elétrico; condutores e capacitores}\label{ch:b1:potential-capacitors}

Um campo são três números em cada ponto; um potencial é um só. Como a
força eletrostática é conservativa, todo o campo de uma distribuição de
cargas pode ser dobrado numa única função escalar, o potencial, do qual ele
se recupera por derivação --- a mesma economia que transformou forças em
\omterm{def:b1:work-and-energy:conservative}{energia potencial} na mecânica. O potencial é o que um voltímetro lê, o que
uma bateria mantém, o que acelera elétrons num tubo; as suas superfícies de
nível mapeiam um campo num relance. Este capítulo o constrói, usa-o no
dipolo (o primeiro modelo elétrico de uma molécula) e depois se volta para
os dois objetos de que todo circuito é feito: os condutores, em que a carga
se arranja de modo a matar o campo interno, e os \omterm{def:b1:potential-capacitors:capacitance}{capacitores}, que armazenam
carga e energia num campo.

\section{O potencial eletrostático}

\begin{theorem}[O campo eletrostático é conservativo]\label{thm:b1:potential-capacitors:potential}
\index{potencial eletrostático}\index{volt}\index{circulação}
Existe uma função escalar $V(M)$, o \emph{\omterm{thm:b1:potential-capacitors:potential}{potencial eletrostático}}
(volt, \unit{V}), tal que
\[
\vect E = -\overrightarrow{\operatorname{grad}}\,V , \qquad
V_A - V_B = \int_A^B \vect E\cdot\dd\vect l
\]
ao longo de qualquer caminho de $A$ a $B$ (a \emph{circulação} de $\vect E$
de $A$ a $B$ não depende do caminho e é nula ao longo de um circuito
fechado). Para uma carga pontual $q$ em $O$, $V(M) = \dfrac{q}{4\pi\varepsilon_0\,OM}$
com a convenção $V(\infty) = 0$; para várias cargas os potenciais
se somam. A \omterm{def:b1:work-and-energy:conservative}{energia potencial} de uma carga $q'$ em $M$ vale $E_p = q'V(M)$, e
o trabalho da força elétrica de $A$ a $B$ vale $q'(V_A - V_B)$.
\end{theorem}

\begin{proof}
A força de Coulomb de uma carga fixa $q$ sobre $q'$ tem a forma da
\omterm{def:b1:central-forces:gravitation}{gravitação} de Newton com $-Gm_1m_2$ substituído por $qq'/4\pi\varepsilon_0$; o mesmo
cálculo que deu $E_p = -Gm_1m_2/r$
(\cref{prop:b1:work-and-energy:eps}) dá $E_p = qq'/4\pi\varepsilon_0r$:
dividindo por $q'$ obtém-se $V$, e $\vect E = \vect F/q' =
-\overrightarrow{\operatorname{grad}}(E_p)/q'$. A superposição leva o
resultado a qualquer distribuição; a fórmula da circulação é a definição
do gradiente integrada ao longo do caminho.
\end{proof}

\begin{proposition}[Ler um potencial]\label{prop:b1:potential-capacitors:reading}
\index{superfície equipotencial}
\begin{enumerate}
  \item As \emph{\omterm{prop:b1:potential-capacitors:reading}{superfícies equipotenciais}} $V = \text{const}$ são
        perpendiculares em toda parte às \omterm{def:b1:electrostatics-gauss:field}{linhas de campo}, e o campo
        aponta para os potenciais \emph{decrescentes}.
  \item $|\vect E|$ é a taxa de variação de $V$ ao longo da direção de
        maior declive: onde as equipotenciais estão próximas, o campo é forte.
  \item $V$ é definido a menos de uma constante aditiva; só as diferenças
        (as tensões) são mensuráveis. A terra ou o infinito é posto em zero por
        convenção.
  \item Em \omterm{def:b1:point-kinematics:spherical}{coordenadas esféricas} com simetria em torno de $O$, $E_r = -\dd V/
        \dd r$; em \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares} $(r, \theta)$ num plano,
        $E_r = -\partial V/\partial r$ e $E_\theta = -\dfrac1r\,\partial V/\partial\theta$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sobre uma equipotencial, $\dd V = -\vect E\cdot\dd\vect l = 0$ para todo
deslocamento na superfície, de modo que $\vect E$ é normal a ela; deslocando-se ao longo de
$\vect E$, $\dd V = -E\,\dd l < 0$. As componentes polares seguem de
$\dd V = (\partial V/\partial r)\dd r + (\partial V/\partial\theta)\dd\theta$ e
$\dd\vect l = \dd r\,\vect e_r + r\,\dd\theta\,\vect e_\theta$.
\end{proof}

\begin{method}[Calcular um potencial]\label{meth:b1:potential-capacitors:compute}
Ou some $\dd q/4\pi\varepsilon_0r$ sobre a distribuição (quando a soma é
calculável e a distribuição é finita, de modo que $V(\infty) = 0$ faça
sentido), ou integre $\dd V = -\vect E\cdot\dd\vect l$ ao longo de um caminho
conveniente a partir de um ponto onde $V$ é conhecido, usando o campo achado pela \omterm{thm:b1:electrostatics-gauss:gauss}{lei de
Gauss}. Para distribuições infinitas (fio, plano) só o segundo caminho
funciona, e o zero de $V$ tem de ser escolhido num ponto finito.
\end{method}

\begin{proposition}[Potenciais das distribuições clássicas]\label{prop:b1:potential-capacitors:classic}
\begin{enumerate}
  \item Esfera de raio $R$ com $Q$ no seu volume: $V(r) = \dfrac{Q}{4\pi\varepsilon_0r}$
        fora; $V(r) = \dfrac{Q}{4\pi\varepsilon_0}\,\dfrac{3R^2 - r^2}{2R^3}$
        dentro, de modo que $V(0) = \tfrac32\,Q/4\pi\varepsilon_0R$. Com $Q$ na superfície,
        $V$ é uniforme dentro, igual a $Q/4\pi\varepsilon_0R$.
  \item Fio infinito $\lambda$: $V(r) = -\dfrac{\lambda}{2\pi\varepsilon_0}\ln\dfrac{r}{r_0}$.
  \item \omterm{prop:b1:potential-capacitors:three}{Capacitor plano}, com campo $E$ da placa~1 para a placa~2, separadas de $d$:
        $V$ decresce linearmente através do espaçamento e $V_1 - V_2 = Ed$.
\end{enumerate}
\end{proposition}

\begin{proof}
Integre $E_r = -\dd V/\dd r$ com os campos da
\cref{prop:b1:electrostatics-gauss:three}, do infinito para dentro no caso da
esfera (a continuidade em $R$ fixa a constante interna), a partir de $r_0$ no caso do
fio, e através do espaçamento no caso do \omterm{def:b1:potential-capacitors:capacitance}{capacitor}.
\end{proof}

\begin{example}[O elétron-volt, de novo]\label{ex:b1:potential-capacitors:eV}
Um elétron que atravessa um espaçamento de \qty{100}{V} ganha $eU = \qty{100}{eV} =
\qty{1.6e-17}{J}$, qualquer que seja a forma do campo entre os
eletrodos: o potencial torna o caminho irrelevante
(\cref{def:b1:charged-particles:eV}). Uma carga de \qty{1}{\micro C} a
\qty{1}{m} de outra tem $V = \qty{9}{kV}$ e o par armazena
\qty{9}{mJ}; o potencial no centro de um núcleo de ouro vale
\qty{24}{MV} (\cref{exo:b1:potential-capacitors:6}).
\end{example}

\section{O dipolo elétrico}

\begin{definition}[Dipolo elétrico]\label{def:b1:potential-capacitors:dipole}
\index{dipolo elétrico}\index{momento dipolar}
Duas cargas opostas, $-q$ em $N$ e $+q$ em $P$, separadas de $a = NP$,
formam um \emph{dipolo elétrico} de \emph{momento dipolar}
$\vect p = q\,\vect{NP}$ (\unit{C.m}), apontando do $-$ para o $+$. A aproximação
dipolar é o estudo do seu campo a distâncias $r \gg a$. Uma
molécula cujos centros de carga positiva e negativa não coincidem
(a água, $p = \qty{6.2e-30}{C.m}$) é um dipolo; um átomo neutro num campo
se torna um (dipolo induzido).
\end{definition}

\begin{proposition}[Campo de um dipolo]\label{prop:b1:potential-capacitors:dipolefield}
Em $M$ com $OM = r \gg a$ ($O$ o meio de $NP$) e $\theta$ o
ângulo entre $\vect p$ e $\vect{OM}$,
\[
V(M) = \frac{p\cos\theta}{4\pi\varepsilon_0 r^2}, \qquad
E_r = \frac{2p\cos\theta}{4\pi\varepsilon_0r^3}, \qquad
E_\theta = \frac{p\sin\theta}{4\pi\varepsilon_0r^3} .
\]
O campo de um dipolo cai como $1/r^3$, mais depressa que o $1/r^2$ de uma
carga: de longe as duas cargas quase se cancelam, e o que resta é a
separação entre elas.
\end{proposition}

\begin{proof}
$V = \dfrac{q}{4\pi\varepsilon_0}\Bigl(\dfrac1{PM} - \dfrac1{NM}\Bigr)$ com
$PM \approx r - \tfrac a2\cos\theta$ e $NM \approx r + \tfrac a2\cos\theta$
(projeção de $\pm\tfrac a2$ sobre $\vect{OM}$); em primeira ordem em $a/r$,
$1/PM - 1/NM \approx a\cos\theta/r^2$. Depois, $E_r = -\partial V/\partial r$ e
$E_\theta = -(1/r)\,\partial V/\partial\theta$.
\end{proof}

\begin{proposition}[Dipolo num campo externo]\label{prop:b1:potential-capacitors:dipoleinfield}
\index{dipolo!torque}\index{dipolo!energia}
Um dipolo $\vect p$ num campo uniforme $\vect E$ não sente força resultante, mas
sim um \omterm{def:b1:angular-momentum:moment}{torque} $\vect\Gamma = \vect p\wedge\vect E$ que tende a alinhá-lo com
o campo, e tem a \omterm{def:b1:work-and-energy:conservative}{energia potencial} $E_p = -\vect p\cdot\vect E$,
mínima quando alinhado. Num campo não uniforme ele é, além disso, puxado
para as regiões de campo intenso (em uma dimensão, $F_x = p\,\dd E/\dd x$
para um dipolo alinhado).
\end{proposition}

\begin{proof}
As forças $\pm q\vect E$ se cancelam; o seu momento em relação a $O$ vale $\vect{OP}\wedge
q\vect E + \vect{ON}\wedge(-q\vect E) = q\vect{NP}\wedge\vect E$. Energia:
$qV(P) - qV(N) = -q\,\vect{NP}\cdot\vect E$ para um campo uniforme (pois $V(P)
- V(N) = -\vect E\cdot\vect{NP}$). Não uniforme, alinhado ao longo de $x$: $F_x =
qE(x + a) - qE(x) \approx qa\,\dd E/\dd x$.
\end{proof}

\begin{example}[Por que o pente atrai o papel]\label{ex:b1:potential-capacitors:comb}
Um pente carregado cria um campo que decresce com a distância; um pedacinho de
papel, neutro, é polarizado por ele (dipolo induzido ao longo de $\vect E$), e
a força $p\,\dd E/\dd x$ sobre um dipolo alinhado o puxa para a
região de campo intenso: o pente. O sinal da carga do pente não importa;
o mesmo mecanismo segura o balão na parede e faz o filete de
água entortar. Para as moléculas de água, $pE$ num campo de \qty{1}{MV/m}
vale \qty{6e-24}{J}, $600$ vezes menos que $k_BT$ à temperatura ambiente:
só um leve alinhamento líquido sobrevive à agitação térmica
(\cref{exo:b1:potential-capacitors:7}).
\end{example}

\section{Condutores em equilíbrio eletrostático}

\begin{theorem}[Propriedades de um condutor em equilíbrio]\label{thm:b1:potential-capacitors:conductor}
\index{condutor em equilíbrio}\index{teorema de Coulomb}\index{gaiola de Faraday}
Num condutor em que as cargas livres estão em repouso:
\begin{enumerate}
  \item $\vect E = \vect 0$ dentro, e o condutor é um volume
        equipotencial: $V$ uniforme nele e na sua superfície.
  \item Toda carga líquida fica na superfície, com uma densidade superficial
        $\sigma$; o interior é neutro.
  \item Logo do lado de fora, o campo é normal à superfície e vale
        $\vect E = \dfrac{\sigma}{\varepsilon_0}\,\vect n$ (\emph{teorema de Coulomb}).
  \item Uma cavidade dentro de um condutor, vazia de carga, tem $\vect E =
        \vect 0$ e $V$ uniforme, aconteça o que acontecer fora (\emph{\omterm{thm:b1:potential-capacitors:conductor}{gaiola
        de Faraday}}).
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Um campo faria as cargas livres se moverem: o equilíbrio exige $\vect E =
\vect 0$, donde $\dd V = 0$ entre dois pontos internos quaisquer; a superfície é
o limite. (2) \omterm{thm:b1:electrostatics-gauss:gauss}{Lei de Gauss} sobre qualquer superfície fechada desenhada dentro do
condutor: fluxo nulo, carga encerrada nula. (3) A superfície é uma
equipotencial, de modo que $\vect E$ logo do lado de fora é normal a ela; uma caixinha
cilíndrica que atravessa a superfície tem fluxo $E\,\dd S$ apenas pela face externa
(dentro $\vect E = \vect 0$) e carga $\sigma\,\dd S$. (4) A parede da
cavidade é uma equipotencial; uma \omterm{def:b1:electrostatics-gauss:field}{linha de campo} dentro da cavidade teria de
ir da parede à parede vencendo uma queda de potencial --- impossível sobre
uma equipotencial; logo, nenhuma linha e nenhum campo (admitido nesta forma).
\end{proof}

\begin{example}[Poder das pontas e para-raios]\label{ex:b1:potential-capacitors:point}
\index{poder das pontas}
Duas esferas de raios $R_1 > R_2$, bem afastadas e ligadas por um fio, têm
o mesmo $V = Q_i/4\pi\varepsilon_0R_i$, logo $\sigma_i = Q_i/4\pi R_i^2 = \varepsilon_0V/R_i$:
quanto \emph{menor} o \omterm{def:b1:point-kinematics:frenet}{raio de curvatura}, \emph{maior} a carga superficial
e o campo $\sigma/\varepsilon_0 = V/R$. Numa ponta afiada o campo
supera o valor de ruptura do ar (\qty{3}{MV/m}) muito antes de superá-lo
em qualquer outro lugar: o ar ioniza e a carga escoa em silêncio. Um
para-raios funciona por esse \emph{\omterm{ex:b1:potential-capacitors:point}{poder das pontas}} --- ele não atrai
tanto a descarga quanto lhe oferece um caminho, e escoa a carga sob uma
nuvem antes que ela se acumule.
\end{example}

\omterm{thm:b1:potential-capacitors:conductor}{gaiola de Faraday}). À direita: o \omterm{ex:b1:potential-capacitors:point}{poder das pontas} --- num condutor

a curvatura é mais aguda.}
\end{omfigure}

\section{Capacitores}

\begin{definition}[Capacitância]\label{def:b1:potential-capacitors:capacitance}
\index{capacitância}\index{capacitor}\index{farad}
Um condutor isolado ao potencial $V$ leva uma carga $Q$
proporcional a $V$: $Q = CV$, sendo $C$ a sua \emph{capacitância} (farad,
\unit{F}); para uma esfera $C = 4\pi\varepsilon_0R$. Um \emph{capacitor} é um par
de condutores (as \emph{placas}) que levam cargas opostas $\pm Q$ e
ficam frente a frente, de modo que o campo fica confinado entre eles; a sua
capacitância vale $C = Q/U$, sendo $U = V_1 - V_2$ a \omterm{def:b1:dc-circuits:voltage}{tensão} entre as
placas. O símbolo e a lei de circuito $i = C\,\dd u/\dd t$ foram usados no
\cref{ch:b1:transient-regimes}.
\end{definition}

\begin{proposition}[Três capacitores]\label{prop:b1:potential-capacitors:three}
\index{capacitor plano}\index{capacitor esférico}\index{capacitor cilíndrico}
\begin{enumerate}
  \item Plano: placas de área $S$ separadas de $d \ll \sqrt S$:
        $C = \dfrac{\varepsilon_0S}{d}$.
  \item Esférico: esferas concêntricas de raios $R_1 < R_2$:
        $C = \dfrac{4\pi\varepsilon_0R_1R_2}{R_2 - R_1}$.
  \item Cilíndrico (coaxial), raios $a < b$, comprimento $L \gg b$:
        $C = \dfrac{2\pi\varepsilon_0L}{\ln(b/a)}$.
\end{enumerate}
Preencher o espaçamento com um isolante de permissividade relativa
$\varepsilon_r$ multiplica $C$ por $\varepsilon_r$ (as moléculas do dielétrico se
polarizam e enfraquecem o campo; $\varepsilon_r = 2$ a $10$ para os plásticos e
o vidro comuns, e $80$ para a água).
\end{proposition}

\begin{proof}
Em cada caso o campo entre as placas é o da
\cref{prop:b1:electrostatics-gauss:three}: $E = \sigma/\varepsilon_0 = Q/\varepsilon_0S$,
$U = Ed$; $E = Q/4\pi\varepsilon_0r^2$, $U = \int_{R_1}^{R_2}E\,\dd r = \dfrac{Q}{4\pi\varepsilon_0}
\Bigl(\dfrac1{R_1} - \dfrac1{R_2}\Bigr)$; $E = Q/2\pi\varepsilon_0Lr$, $U =
\dfrac{Q}{2\pi\varepsilon_0L}\ln\dfrac ba$. Divida $Q$ por $U$. O fator do
dielétrico fica admitido.
\end{proof}

\begin{theorem}[Energia de um capacitor]\label{thm:b1:potential-capacitors:energy}
\index{energia de um capacitor}\index{densidade de energia do campo elétrico}
Um \omterm{def:b1:potential-capacitors:capacitance}{capacitor} carregado com $Q$ sob $U$ armazena
\[
W = \tfrac12 CU^2 = \tfrac12 QU = \frac{Q^2}{2C} ,
\]
o trabalho necessário para levar a carga de uma placa à outra
contra a \omterm{def:b1:dc-circuits:voltage}{tensão} que cresce. Num \omterm{prop:b1:potential-capacitors:three}{capacitor plano} $W = \tfrac12\varepsilon_0E^2
\times Sd$: a energia está no campo, com densidade $\tfrac12\varepsilon_0E^2$
por unidade de volume --- afirmação que vale para todo campo
eletrostático.
\end{theorem}

\begin{proof}
Levar $\dd q$ da placa negativa à positiva quando a carga
vale $q$ custa $\dd W = (q/C)\,\dd q$; integre de $0$ a $Q$. Depois
substitua $C = \varepsilon_0S/d$ e $U = Ed$; a densidade geral fica admitida.
\end{proof}

\begin{example}[Três capacitores e as suas energias]\label{ex:b1:potential-capacitors:energies}
Um metro quadrado de par de folhas metálicas a \qty{1}{mm}: $C = 8.85 \times 10^{-12}/
10^{-3} = \qty{8.9}{nF}$ --- o \omterm{def:b1:potential-capacitors:capacitance}{farad} é uma unidade enorme. Os
\qty{150}{\micro F} de um desfibrilador a \qty{2}{kV}: \qty{300}{J}, entregues em milissegundos.
Um supercapacitor de \qty{3000}{F} a \qty{2.7}{V}: \qty{11}{kJ} numa lata
do tamanho de uma garrafa de refrigerante --- a sua \omterm{def:b1:potential-capacitors:capacitance}{capacitância} vem de um
espaçamento molecular $d$ e de um $S$ poroso enorme. A densidade de energia de um campo na
ruptura do ar, $\tfrac12\varepsilon_0(3 \times 10^6)^2 = \qty{40}{J/m^3}$, é
a razão de os \omterm{def:b1:potential-capacitors:capacitance}{capacitores} armazenarem tão pouco em comparação com as baterias.
\end{example}

\begin{remark}[Associação de capacitores]\label{rem:b1:potential-capacitors:association}
Em paralelo (mesmo $U$) as cargas se somam: $C = C_1 + C_2$; em série (mesma
$Q$) as tensões se somam: $1/C = 1/C_1 + 1/C_2$ --- o contrário dos
\omterm{def:b1:dc-circuits:resistor}{resistores}, porque $C$ mede uma facilidade (carga por volt) onde $R$
mede uma dificuldade. Essas regras seguem de $Q = CU$ e das
leis de circuito do \cref{ch:b1:dc-circuits}.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:1}
Potencial a \qty{1.0}{m} de uma carga de \qty{1.0}{\micro C}; energia necessária
para trazer uma segunda carga de \qty{1.0}{\micro C} do infinito até esse ponto; velocidade
com que ela sairia se fosse solta, sendo a sua massa \qty{1}{g}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:2}
Placas a \qty{1.0}{mm} uma da outra sob \qty{100}{V}: campo; \omterm{thm:b1:work-and-energy:ke}{energia cinética} e
velocidade ganhas por um elétron que atravessa o espaçamento a partir do repouso; tempo de
voo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:3}
Num mapa de equipotenciais desenhadas a cada \qty{10}{V}, as linhas estão a
\qty{2}{mm} umas das outras perto de $A$ e a \qty{8}{mm} perto de $B$. Campo em $A$
e em $B$; para que lado o campo aponta em relação aos potenciais crescentes;
para que lado um elétron acelera?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:4}
Uma esfera de Van de Graaff de raio \qty{15}{cm} é carregada a \qty{200}{kV}.
\omterm{def:b1:potential-capacitors:capacitance}{Capacitância}, carga e campo na sua superfície. Potencial máximo antes que
o ar em volta rompa (\qty{3}{MV/m}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:5}
Potencial no eixo de um anel de raio $R$ e carga $Q$; recupere
o campo axial por derivação. O mesmo para o disco carregado uniformemente
(use o \cref{exo:b1:electrostatics-gauss:5}): $V(z) = \dfrac{\sigma}{2\varepsilon_0}
\bigl(\sqrt{z^2 + R^2} - z\bigr)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:6}
Deduza o potencial dentro de uma esfera carregada uniformemente a partir do seu campo.
Potencial no centro de um núcleo de ouro ($Z = 79$, $R = \qty{7.0}{fm}$);
energia de que um próton precisaria para chegar ao centro vindo de longe, em
\unit{MeV}; comente as partículas alfa de \qty{5}{MeV} de Rutherford.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:7}
Molécula de água, $p = \qty{6.2e-30}{C.m}$. Campo que ela cria a \qty{1.0}{nm}
ao longo do seu eixo e perpendicularmente a ele; \omterm{def:b1:angular-momentum:moment}{torque} e diferença de energia
entre as posições alinhada e antialinhada em $E = \qty{1.0}{MV/m}$;
compare com $k_BT$ a \qty{300}{K}. Campo de um íon de sódio a
\qty{0.3}{nm} e energia do dipolo de água alinhado nele (hidratação).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:8}
Uma esfera condutora de raio \qty{1.0}{cm} no ar: carga e potencial
máximos antes da ruptura. Duas esferas condutoras, $R_1 = \qty{10}{cm}$
e $R_2 = \qty{1.0}{mm}$, ligadas por um fio fino e levadas a
\qty{20}{kV}: campo na superfície de cada uma; qual delas descarrega no ar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:9}
\omterm{def:b1:potential-capacitors:capacitance}{Capacitâncias}: duas folhas de \qty{1.0}{m^2} a \qty{10}{\micro m} uma da outra com um
filme polimérico $\varepsilon_r = 2.2$ (um \omterm{def:b1:potential-capacitors:capacitance}{capacitor} de filme); as mesmas folhas enroladas
--- enrolar muda $C$? Energia a \qty{400}{V}. Um \omterm{def:b1:potential-capacitors:capacitance}{capacitor} de \qty{150}{\micro F}
a \qty{2.0}{kV} (desfibrilador): carga, energia e \omterm{thm:b1:sinusoidal-impedance:power}{potência média} se ele for
descarregado em \qty{5}{ms}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:10}
\emph{\omterm{prop:b1:potential-capacitors:three}{Capacitores esférico} e cilíndrico.} (a) Deduza $C$ para
esferas concêntricas e verifique que, para $R_2 - R_1 = d \ll R_1$, ele se reduz
à fórmula do plano, e que, para $R_2 \to \infty$, dá a esfera isolada.
(b) Cabo coaxial, $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, polietileno
$\varepsilon_r = 2.3$: \omterm{def:b1:potential-capacitors:capacitance}{capacitância} por metro (compare com os \qty{100}{pF/m}
dos cabos usuais). (c) Campo no condutor interno sob
\qty{1}{kV}; é seguro?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:11}
\emph{A energia no campo.} (a) Energia de uma esfera condutora carregada
($Q$, $R$) a partir de $Q^2/2C$. (b) Iguale-a à energia de repouso do elétron,
$m_ec^2 = \qty{0.511}{MeV}$, e resolva para $R$: é o ``raio clássico do
elétron'' (a menos de um fator). (c) Um \omterm{prop:b1:potential-capacitors:three}{capacitor plano} isolado carregado com
$Q$ tem as suas placas afastadas de $d$ a $2d$: variação da energia armazenada e
força entre as placas deduzida dela; confira com o
\cref{exo:b1:electrostatics-gauss:8}. (d) O mesmo com o \omterm{def:b1:potential-capacitors:capacitance}{capacitor} mantido
a $U$ fixo por uma bateria: variação da energia, e para onde vai a diferença.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:12}
\emph{Energia coulombiana de um núcleo.} (a) Construa uma esfera carregada
uniformemente ($Q$, $R$) casca por casca e mostre que a sua energia eletrostática
vale $W = \dfrac{3Q^2}{20\pi\varepsilon_0R}$. (b) Urânio-238: $Z = 92$, $R =
\qty{7.4}{fm}$: $W$ em \unit{MeV}. (c) Divida-o em dois fragmentos iguais
($Z/2$, $A/2$, raio $R/2^{1/3}$) bem afastados: energia coulombiana liberada.
(d) A energia medida da fissão vale cerca de \qty{200}{MeV}: o que se opõe
ao ganho coulombiano?
\end{exercise}

\section{Problema: a gota de óleo de Millikan}

\begin{problem}[{Problema de fim de semana --- pesar a carga do elétron com
uma gota de óleo: Stokes, o capacitor plano e uma tabela de números que
se recusam a ser outra coisa senão múltiplos de um só}]
\label{pb:b1:potential-capacitors:1}
Gotas de óleo são pulverizadas entre duas placas horizontais separadas de $d = \qty{16}{mm}$,
com \qty{20}{cm} de diâmetro, e observadas ao \omterm{def:b1:optical-instruments:microscope}{microscópio}.
Dados: densidade do óleo $\rho = \qty{900}{kg/m^3}$, densidade do ar
\qty{1.2}{kg/m^3}, viscosidade do ar $\eta = \qty{1.8e-5}{Pa.s}$, $g =
\qty{9.8}{m/s^2}$, $\varepsilon_0 = \qty{8.85e-12}{F/m}$. Uma pequena esfera de
raio $r$ que se move com velocidade $v$ no ar sente o arrasto de Stokes $6\pi\eta rv$.

\medskip
\noindent\textbf{Parte I --- A gota que cai.}
\begin{enumerate}
  \item Forças sobre uma gota que cai sem \omterm{def:b1:dc-circuits:voltage}{tensão} aplicada; mostre que ela
        atinge uma \omterm{prop:b1:newton-dynamics:terminal}{velocidade limite} $v_1 = 2\rho gr^2/9\eta$ (despreze o
        empuxo do ar --- justifique).
  \item Uma gota cai \qty{1.0}{mm} em \qty{10}{s}: o seu raio e a sua massa.
  \item \omterm{def:b1:transient-regimes:firstorder}{Constante de tempo} da aproximação de $v_1$; distância percorrida
        nesse intervalo. Alguma vez se vê a gota acelerando?
  \item Número de Reynolds $\rho_{\text{ar}}v_1r/\eta$ do movimento: a \omterm{prop:b1:newton-dynamics:drag}{lei de
        Stokes} é legítima?
  \item Por que as gotas têm de ser tão pequenas? (Compare o peso com a
        força elétrica sobre uma \omterm{def:b1:electrostatics-gauss:charge}{carga elementar} num campo de
        \qty{3e5}{V/m}.)
  \item Vê-se a gota tremular ligeiramente em torno da sua queda média: o que
        é isso, e por que limita a precisão?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O campo.}
Aplica-se uma \omterm{def:b1:dc-circuits:voltage}{tensão} $U$, com a placa de cima positiva.
\begin{enumerate}[resume]
  \item Campo entre as placas para $U = \qty{5.0}{kV}$; por que ele é
        uniforme, e quais são as equipotenciais?
  \item \omterm{def:b1:potential-capacitors:capacitance}{Capacitância} das placas, carga nelas a \qty{5.0}{kV} e
        energia armazenada.
  \item A gota da questão~2 fica parada com $U_0 = \qty{1080}{V}$:
        sinal e valor da sua carga, em coulombs e em unidades de $e$.
  \item Com $U = \qty{5.0}{kV}$ a mesma gota sobe com $v_2$: mostre que
        $q = 6\pi\eta r(v_1 + v_2)d/U$ e calcule $v_2$.
  \item Por que cronometrar uma subida e uma queda é melhor do que caçar a
        \omterm{def:b1:dc-circuits:voltage}{tensão} de equilíbrio?
  \item \omterm{def:b1:work-and-energy:conservative}{Energia potencial} da carga da gota entre as placas a
        \qty{5}{kV}, e energia dissipada pelo arrasto numa subida de
        \qty{1}{cm}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A tabela de cargas.}
Exposta a uma fonte fraca de raios X entre as medidas, a mesma gota
muda de carga; medidas sucessivas dão (em $10^{-19}$~C):
$4.80$, $8.01$, $6.42$, $3.19$, $9.63$.
\begin{enumerate}[resume]
  \item Mostre que os cinco valores estão perto de múltiplos inteiros de uma
        mesma quantidade, e dê os inteiros.
  \item Melhor estimativa da \omterm{def:b1:electrostatics-gauss:charge}{carga elementar} a partir dos cinco valores;
        incerteza padrão da média (\cref{ch:b1:units-dimensions}).
  \item Por que a carga muda por números inteiros de $e$ entre as
        medidas?
  \item Mostre que, neste método, $q \propto \eta^{3/2}$: de quanto um erro
        de $1\%$ na viscosidade desloca $e$? (O valor original de Millikan era
        $0.6\%$ baixo, exatamente por isso.)
  \item Os quarks têm carga $\pm e/3$ e $\pm 2e/3$: por que nenhuma gota de óleo
        jamais mostrou um terço de $e$?
  \item A constante de Faraday, a carga de um \omterm{def:b1:kinetic-theory:scales}{mol} de elétrons, vale
        $F = \qty{96485}{C/mol}$: deduza o número de Avogadro.
  \item Thomson havia medido $e/m_e = \qty{1.76e11}{C/kg}$ para o
        elétron: deduza a sua massa e compare com um átomo de hidrogênio
        (\qty{1.67e-27}{kg}).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O aparelho, pelo potencial.}
\begin{enumerate}[resume]
  \item Esboce as equipotenciais e as \omterm{def:b1:electrostatics-gauss:field}{linhas de campo} entre as placas,
        incluindo o efeito de borda perto das extremidades; justifique desprezá-lo.
  \item Uma gota carregada positivamente à altura $z$ acima da placa de baixo
        (a \qty{0}{V}): \omterm{def:b1:work-and-energy:conservative}{energia potencial} em função de $z$; \omterm{def:b1:work-and-energy:conservative}{energia
        potencial} total incluindo a gravidade; condição sobre $U$ para o
        ponto de equilíbrio.
  \item Compare a energia da carga da gota (questão~12) com a
        energia armazenada no \omterm{def:b1:potential-capacitors:capacitance}{capacitor}: a gota perturba o campo?
  \item A \omterm{def:b1:dc-circuits:voltage}{tensão} é invertida (placa de baixo positiva) com
        $U = \qty{5.0}{kV}$: o que faz a gota da questão~9, e com
        que velocidade?
  \item Um grão de poeira de \qty{1}{\micro g} que leva $10^4e$ no mesmo
        campo: razão entre a força elétrica e o peso; por que tais grãos não podem
        ser levitados e por que Millikan escolheu uma névoa de óleo.
  \item Resuma: o que foi medido (três vezes), o que foi deduzido,
        com que incerteza, e que duas constantes daí resultaram.
\end{enumerate}
\end{problem}
