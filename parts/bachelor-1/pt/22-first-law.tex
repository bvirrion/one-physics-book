\chapter{A primeira lei da termodinâmica}\label{ch:b1:first-law}

Encha o pneu de uma bicicleta e o corpo da bomba fica quente demais para
segurar. Um motor diesel não tem vela de ignição: ele comprime o seu ar
vinte vezes, e o ar fica quente o bastante para acender o combustível
sozinho. Um cilindro de gás comprimido cobre-se de gelo enquanto se
esvazia. Em cada caso a energia muda de forma --- trabalho vira \omterm{def:b1:first-law:heat}{calor},
\omterm{def:b1:first-law:heat}{calor} vira trabalho, os dois viram \omterm{def:b1:kinetic-theory:U}{energia interna} --- e a \omterm{thm:b1:first-law:firstlaw}{primeira lei da
termodinâmica} é a contabilidade que fecha as contas. Este capítulo define
trabalho e \omterm{def:b1:first-law:heat}{calor} para um \omterm{def:b1:first-law:transformation}{sistema termodinâmico}, enuncia a primeira lei,
introduz a \omterm{def:b1:kinetic-theory:U}{energia interna} e a \omterm{def:b1:first-law:enthalpy}{entalpia} como as grandezas que ela conserva
e a aplica às transformações de um \omterm{def:b1:kinetic-theory:model}{gás perfeito} e à calorimetria.

\section{Transformações, trabalho e calor}

\begin{definition}[Sistema, transformação]\label{def:b1:first-law:transformation}
\index{sistema termodinâmico}\index{transformação}\index{transformação quase-estática}\index{transformação reversível}
Um \emph{sistema termodinâmico} é a matéria contida numa fronteira
escolhida; todo o resto é o \emph{exterior}. Uma \emph{transformação} o leva
de um estado de equilíbrio a outro. Ela é \emph{quase-estática} se o
sistema passa por uma sucessão de estados de equilíbrio (lenta o bastante
para que $P$ e $T$ estejam definidos em todo o sistema), e \emph{reversível}
se, além disso, inverter as condições externas inverte o caminho (sem
atrito, sem diferença finita de temperatura ou de pressão com o exterior).
Os nomes vêm do que fica constante: \emph{isotérmica} ($T$), \emph{isobárica}
($P$), \emph{isocórica} ($V$); \emph{adiabática} se não há troca de \omterm{def:b1:first-law:heat}{calor};
\emph{monobárica} (\emph{monotérmica}) se a pressão (a temperatura) externa
é constante, aconteça o que acontecer lá dentro.
\end{definition}

\begin{proposition}[Trabalho das forças de pressão]\label{prop:b1:first-law:work}
\index{trabalho!das forças de pressão}
Um sistema cujo volume varia de $\dd V$ contra uma pressão externa
$P_{\mathrm{ext}}$ recebe o trabalho
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
Para uma \omterm{def:b1:first-law:transformation}{transformação quase-estática} $P_{\mathrm{ext}} = P$ (a pressão do
próprio sistema) e $W = -\int P\,\dd V$ é menos a área sob o caminho no
diagrama de Clapeyron: positivo na compressão, negativo na expansão.
\end{proposition}

\begin{proof}
Um pistão de área $S$ empurrado pelo exterior com a força
$P_{\mathrm{ext}}S$ se desloca de $\dd x$ para fora: o exterior realiza sobre o
sistema o trabalho $-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$. Toda
fronteira é um conjunto de pistões como esse. Quase-estática: o pistão está
em equilíbrio, $P_{\mathrm{ext}} = P$ a menos de uma diferença que tende a zero.
\end{proof}

\begin{definition}[Calor]\label{def:b1:first-law:heat}
\index{calor}\index{termostato}
\emph{Calor} $Q$ é a energia transferida ao sistema de outro modo que não
por trabalho macroscópico --- por colisões moleculares numa parede
(condução), pela circulação de um fluido (convecção), por radiação. Um
\emph{termostato} é um corpo tão grande que a sua temperatura não muda,
qualquer que seja o calor que ele troque. Convenção de sinais em todo o
livro: $W$ e $Q$ são \emph{recebidos} pelo sistema, positivos quando entram.
\end{definition}

\section{A primeira lei}

\begin{theorem}[Primeira lei da termodinâmica]\label{thm:b1:first-law:firstlaw}
\index{primeira lei da termodinâmica}\index{energia interna}
Um \omterm{def:b1:newton-dynamics:conservation}{sistema fechado} possui uma função de estado, a sua \emph{\omterm{def:b1:kinetic-theory:U}{energia interna}}
$U$ (extensiva), tal que, para qualquer \omterm{def:b1:first-law:transformation}{transformação},
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
sendo $E_k$ a \omterm{thm:b1:work-and-energy:ke}{energia cinética} macroscópica (nula para um sistema em
repouso): a energia recebida como trabalho e como \omterm{def:b1:first-law:heat}{calor} fica armazenada como
\omterm{def:b1:kinetic-theory:U}{energia interna}. $W$ e $Q$ dependem cada um do caminho; a sua soma, não.
\end{theorem}
\admitted

\begin{remark}[O que ela diz]\label{rem:b1:first-law:meaning}
É a conservação da energia, com o \omterm{def:b1:first-law:heat}{calor} reconhecido como uma transferência
de energia (a experiência da roda de pás de Joule: um trabalho medido eleva
sempre a temperatura da água tanto quanto um \omterm{def:b1:first-law:heat}{calor} medido). ``$U$ é uma
função de estado'' quer dizer que $\Delta U$ está fixado pelos estados
inicial e final --- de modo que um ciclo tem $\Delta U = 0$ e $W + Q = 0$: uma
máquina que fornece trabalho tem de receber \omterm{def:b1:first-law:heat}{calor}. O conteúdo microscópico
de $U$ é o da \cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[Entalpia; capacidades térmicas]\label{def:b1:first-law:enthalpy}
\index{entalpia}\index{capacidade térmica a pressão constante}
A \emph{entalpia} $H = U + PV$ é uma função de estado adaptada às
transformações a pressão constante. As \emph{capacidades térmicas a volume
e a pressão constantes} são
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
em \unit{J/K}; por \omterm{def:b1:kinetic-theory:scales}{mol}, $C_{V,m}$ e $C_{P,m}$; por quilograma, os \omterm{def:b1:first-law:heat}{calores}
específicos $c_V$ e $c_P$.
\end{definition}

\begin{proposition}[Calor a volume constante e a pressão constante]\label{prop:b1:first-law:qvqp}
Para um sistema em repouso que troca trabalho apenas por forças de pressão:
\begin{itemize}
  \item \omterm{def:b1:first-law:transformation}{transformação} isocórica: $W = 0$ e $Q_V = \Delta U$;
  \item \omterm{def:b1:first-law:transformation}{transformação} monobárica entre dois estados à pressão externa
        $P_0$: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
Isocórica: $\dd V = 0$. Monobárica: $W = -P_0(V_2 - V_1)$, logo $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$, pois $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[Gás perfeito: $U$, $H$, relação de Mayer]\label{prop:b1:first-law:gas}
\index{relação de Mayer}\index{coeficiente adiabático}\index{leis de Joule}
Para um \omterm{def:b1:kinetic-theory:model}{gás perfeito}, $U$ e $H$ dependem só de $T$ (\omterm{prop:b1:kinetic-theory:Ugas}{leis de Joule}):
$\dd U = C_V\,\dd T$, $\dd H = C_P\,\dd T$, e
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
$\gamma = 5/3$ para os gases monoatômicos e $7/5$ para os diatômicos a
temperaturas comuns. Para uma \omterm{prop:b1:kinetic-theory:condensed}{fase condensada} no modelo da
\cref{prop:b1:kinetic-theory:condensed}, $C_P \approx C_V = C$ e
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
$H = U + nRT$; derive em relação a $T$. $C_{V,m} = \tfrac32 R$ ou
$\tfrac52 R$ pela \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{Transformações de um gás perfeito}

\begin{proposition}[As quatro transformações usuais]\label{prop:b1:first-law:transformations}
\index{transformação isotérmica}\index{transformação adiabática}\index{lei de Laplace}
Para $n$ \omterm{def:b1:kinetic-theory:scales}{mols} de \omterm{def:b1:kinetic-theory:model}{gás perfeito}:
\begin{itemize}
  \item isocórica: $W = 0$, $Q = \Delta U = nC_{V,m}\Delta T$;
  \item isobárica (quase-estática): $W = -P\Delta V = -nR\Delta T$, $Q = \Delta H =
        nC_{P,m}\Delta T$;
  \item isotérmica reversível: $\Delta U = 0$, $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$;
  \item adiabática reversível: $Q = 0$, $W = \Delta U = nC_{V,m}(T_2 - T_1)$,
        e ao longo do caminho (\emph{\omterm{prop:b1:first-law:transformations}{lei de Laplace}})
        \[
        PV^\gamma = \text{const}, \qquad TV^{\gamma - 1} = \text{const}, \qquad
        T^\gamma P^{1 - \gamma} = \text{const} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Isocórica e isobárica: as proposições anteriores. Isotérmica: $W =
-\int nRT\,\dd V/V$. Adiabática: $\dd U = \delta W$, isto é, $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$, logo $\dd T/T = -(\gamma - 1)\dd V/V$ (usando
$R/C_{V,m} = \gamma - 1$); integre: $TV^{\gamma - 1}$ constante; as outras
formas saem de $PV = nRT$.
\end{proof}

\begin{example}[A bomba e o diesel]\label{ex:b1:first-law:diesel}
Ar ($\gamma = 1.4$) comprimido adiabaticamente de \qty{1}{bar} e \qty{300}{K}
a \qty{7}{bar} (uma bomba de bicicleta): $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- \qty{250}{\degreeCelsius} por um instante, o que o corpo da bomba sente.
Uma taxa de compressão de $20$ num diesel: $T_2 = 300 \times 20^{0.4} = \qty{994}{K}$,
$P_2 = 20^{1.4} = \qty{66}{bar}$ --- acima da temperatura de autoignição do
combustível: o motor não precisa de faísca (o problema de fim de semana
percorre o ciclo inteiro).
\end{example}

\begin{remark}[Transformações irreversíveis]\label{rem:b1:first-law:irreversible}
\index{expansão de Joule}\index{transformação irreversível}
Quando a \omterm{def:b1:first-law:transformation}{transformação} não é quase-estática, a \omterm{prop:b1:first-law:transformations}{lei de Laplace} e $W =
-\int P\,\dd V$ não se aplicam; só valem $W = -\int P_{\mathrm{ext}}\,\dd V$ e a
primeira lei. Dois casos clássicos: um gás que se expande no vácuo (\omterm{rem:b1:first-law:irreversible}{expansão
de Joule}) não recebe trabalho e, em paredes adiabáticas, nenhum \omterm{def:b1:first-law:heat}{calor} --- $\Delta U
= 0$, e um \omterm{def:b1:kinetic-theory:model}{gás perfeito} mantém a sua temperatura; um gás subitamente comprimido
por uma pressão externa que salta para $P_2$ recebe $W = -P_2\Delta V$, mais
do que o trabalho reversível até a mesma pressão, e termina mais quente
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{Calorimetria}

\begin{method}[Balanço calorimétrico]\label{met:b1:first-law:calorimetry}
\index{calorimetria}\index{calor latente}
Corpos postos em contato térmico dentro de um vaso isolado a pressão
constante (a atmosférica) trocam \omterm{def:b1:first-law:heat}{calor} apenas entre si: o total
$\Delta H = \sum Q_P = 0$. Para cada corpo, $\Delta H = mc\,\Delta T$ (sem
mudança de fase) ou $\pm mL$ para uma mudança de estado na sua temperatura
de transição, sendo $L$ o \emph{\omterm{def:b1:first-law:heat}{calor} latente} (\omterm{def:b1:first-law:enthalpy}{entalpia} de fusão ou de
vaporização por unidade de massa: água, $L_f = \qty{334}{kJ/kg}$ a
\qty{0}{\degreeCelsius}, $L_v =
\qty{2260}{kJ/kg}$ a \qty{100}{\degreeCelsius}).
Escreva o balanço, resolva para a temperatura final e verifique que a fase
suposta para cada corpo é compatível com ela.
\end{method}

\begin{example}[Gelo na água]\label{ex:b1:first-law:ice}
\qty{50}{g} de gelo a \qty{0}{\degreeCelsius} em \qty{200}{g} de água a
\qty{30}{\degreeCelsius}: a fusão exige \qty{16.7}{kJ}; a água pode dar
\qty{25.1}{kJ} ao esfriar até \qty{0}{\degreeCelsius}, de modo que todo o gelo
derrete e o balanço $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ kJ dá
$T_f = \qty{8}{\degreeCelsius}$. Se houvesse \qty{100}{g} de gelo, seriam
precisos \qty{33.4}{kJ}: só uma parte derrete e $T_f = \qty{0}{\degreeCelsius}$
--- a verificação importa.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:first-law:1}
Um \omterm{def:b1:kinetic-theory:scales}{mol} de ar ($C_{P,m} = \qty{29.1}{J/(mol.K)}$) é aquecido de
\qty{300}{K} a \qty{400}{K} a pressão constante. \omterm{def:b1:first-law:heat}{Calor} recebido, trabalho
recebido, variação da \omterm{def:b1:kinetic-theory:U}{energia interna}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:2}
Um \omterm{def:b1:kinetic-theory:scales}{mol} de gás a \qty{300}{K} é comprimido isotérmica e reversivelmente
de \qty{2.0}{L} a \qty{0.50}{L}. Trabalho recebido, \omterm{def:b1:first-law:heat}{calor} trocado,
$\Delta U$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:3}
Ar a \qty{300}{K} e \qty{1}{bar} é comprimido adiabática e
reversivelmente até um décimo do seu volume. Temperatura e pressão finais.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:4}
Dê $C_{V,m}$, $C_{P,m}$ e $\gamma$ para o argônio e para o nitrogênio. Para a
água líquida ($c = \qty{4180}{J/(kg.K)}$), por que se abandona a distinção
entre $c_P$ e $c_V$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:5}
\qty{200}{g} de água a \qty{80}{\degreeCelsius} são misturados com \qty{300}{g}
a \qty{20}{\degreeCelsius} num copo isolado. Temperatura final; o que muda se
o próprio copo tiver capacidade térmica de \qty{80}{J/K} e começar a
\qty{20}{\degreeCelsius}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:6}
\qty{50}{g} de gelo a \qty{-10}{\degreeCelsius} ($c_{\text{gelo}} = \qty{2100}{J/(kg.K)}$)
são jogados em \qty{200}{g} de água a \qty{30}{\degreeCelsius}. Estado e
temperatura finais. Mesma pergunta com \qty{150}{g} de gelo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:7}
Um gás percorre um ciclo retangular no diagrama de Clapeyron: $(V_1, P_1)
\to (V_2, P_1) \to (V_2, P_2) \to (V_1, P_2) \to (V_1, P_1)$ com $V_2 > V_1$ e
$P_2 > P_1$. Trabalho recebido ao longo do ciclo, em função da área;
sinal, e significado do sinal.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:8}
Uma bomba de bicicleta comprime ar de \qty{1}{bar} e \qty{300}{K} a
\qty{7}{bar} depressa o bastante para ser adiabática. Temperatura atingida; por que a
bomba esquenta; e por que a pressão do pneu cai um pouco depois que se para
de bombear.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:9}
Um \omterm{def:b1:kinetic-theory:model}{gás perfeito} se expande num vaso isolado e previamente evacuado (\omterm{rem:b1:first-law:irreversible}{expansão
de Joule}). Trabalho, \omterm{def:b1:first-law:heat}{calor}, $\Delta U$, $\Delta T$. A \omterm{def:b1:first-law:transformation}{transformação} é
reversível? O que faria um gás real?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:10}
Fervura de \qty{1.0}{kg} de água a \qty{100}{\degreeCelsius} e \qty{1}{atm}
(volume do vapor \qty{1.67}{m^3}). \omterm{def:b1:first-law:heat}{Calor} fornecido, trabalho trocado com a
atmosfera, $\Delta U$. Para onde vai a maior parte do \omterm{def:b1:first-law:heat}{calor} latente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:11}
Um \omterm{def:b1:kinetic-theory:scales}{mol} de \omterm{prop:b1:kinetic-theory:Ugas}{gás diatômico} a \qty{300}{K} e \qty{1}{bar}, num cilindro
isolado, é comprimido pondo-se subitamente a pressão externa em
\qty{5}{bar} e esperando o equilíbrio. Temperatura e volume finais
(primeira lei com $W = -P_{\mathrm{ext}}\Delta V$); compare com a compressão
adiabática reversível até \qty{5}{bar}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:12}
O som é uma compressão rápida, portanto adiabática: a sua velocidade vale $c = \sqrt{\gamma P/\rho}$
(o valor isotérmico $\sqrt{P/\rho}$ foi o erro de Newton). Calcule os dois
para o ar a \qty{20}{\degreeCelsius} e compare com os \qty{343}{m/s} medidos.
Mostre que $c = \sqrt{\gamma RT/M}$ e compare com a \omterm{def:b1:kinetic-theory:kinetictemp}{velocidade quadrática
média} do \cref{ch:b1:kinetic-theory}.
\end{exercise}

\section{Problema: da bomba de bicicleta ao motor diesel}

\begin{problem}[{Problema de fim de semana --- uma bomba manual que queima
os dedos, um cilindro que acende o seu próprio combustível e uma esfera de
aço que quica sobre um colchão de ar: a primeira lei em três compressões}]
\label{pb:b1:first-law:1}
Ar: \omterm{def:b1:kinetic-theory:model}{gás perfeito} diatômico, $\gamma = 1.40$, $C_{V,m} = \tfrac52 R$, $C_{P,m} =
\tfrac72 R$, $M = \qty{29}{g/mol}$; $R = \qty{8.314}{J/(mol.K)}$. Estado
inicial em toda parte: $P_1 = \qty{1.00}{bar}$, $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{Parte I --- A bomba de bicicleta.}
O cilindro da bomba tem volume $V_1 = \qty{212}{cm^3}$; o ar que ele contém
é comprimido reversível e adiabaticamente até atingir a pressão do pneu,
\qty{7.0}{bar}.
\begin{enumerate}
  \item Quantidade de ar no cilindro.
  \item Volume no fim da compressão.
  \item Temperatura no fim da compressão.
  \item Trabalho recebido pelo ar durante um curso.
  \item O pneu (\qty{2.0}{L}) tem de passar de \qty{1}{bar} a
        \qty{7}{bar} a \qty{300}{K}: quantos cursos, e quanto trabalho
        ao todo (tome o trabalho por curso como constante)?
  \item O ar quente esfria depois no pneu até \qty{300}{K}: o que acontece
        com a pressão do pneu, e o que o ciclista faz?
  \item Compare o trabalho adiabático por curso com o trabalho de uma
        compressão isotérmica até os mesmos \qty{7}{bar}. Por que o
        trabalho isotérmico é maior, embora o gás termine mais frio?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A compressão do diesel.}
Um cilindro de $V_1 = \qty{0.50}{L}$ comprime o seu ar reversível e
adiabaticamente por um fator $20$.
\begin{enumerate}[resume]
  \item Quantidade de ar e temperatura $T_2$ depois da compressão.
  \item Pressão $P_2$.
  \item O combustível se autoinflama acima de uns \qty{500}{K}: explique
        por que um diesel não precisa de vela de ignição, e por que um
        motor a gasolina (taxa $\approx 10$, veja o
        \cref{exo:b1:first-law:3}) não pode autoinflamar.
  \item Trabalho recebido pelo ar durante a compressão.
  \item A \qty{3000}{rpm} um cilindro comprime $25$ vezes por segundo:
        potência absorvida pela compressão (ela é devolvida mais adiante
        no ciclo).
  \item As compressões reais são rápidas (irreversibilidade) e perdem
        \omterm{def:b1:first-law:heat}{calor} para as paredes: em que sentido cada efeito desloca $T_2$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Combustão e expansão.}
No alto do curso, \qty{20}{mg} de combustível (poder calorífico
\qty{43}{MJ/kg}) queimam enquanto o pistão começa a descer, à pressão
constante $P_2$; depois o gás se expande reversível e adiabaticamente de
volta a $V_1$. Trate o gás como ar o tempo todo.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{Calor} liberado pelo combustível.
  \item Temperatura $T_3$ e volume $V_3$ no fim da combustão
        isobárica.
  \item Trabalho trocado durante a fase isobárica (atenção ao sinal!).
  \item Temperatura $T_4$ no fim da expansão adiabática e o trabalho
        trocado durante ela.
  \item Trabalho líquido fornecido pelo gás nos quatro tempos e razão
        entre ele e o \omterm{def:b1:first-law:heat}{calor} liberado (o rendimento ideal).
  \item Compare esse rendimento com $1 - T_1/T_3$, o limite que o
        \cref{ch:b1:heat-engines} estabelecerá para qualquer máquina que
        funcione entre essas temperaturas extremas.
  \item \omterm{def:b1:first-law:heat}{Calor} rejeitado com o escapamento, pela primeira lei ao longo do
        ciclo; verifique o balanço.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Medindo $\gamma$ com uma esfera que quica.}
Uma esfera de aço de massa $m = \qty{16.5}{g}$ se ajusta a um tubo vertical
de vidro de seção $S = \qty{2.0}{cm^2}$ fechado embaixo por um balão de
volume $V = \qty{10}{L}$ de ar a $P = \qty{1.0}{bar}$; deslocada de $x$ da
sua posição de equilíbrio, ela oscila.
\begin{enumerate}[resume]
  \item Por que as compressões e expansões do ar podem ser tomadas como
        adiabáticas? Escreva a relação entre um pequeno deslocamento $x$
        e a variação de pressão $\dd P$ (\omterm{prop:b1:first-law:transformations}{lei de Laplace}, linearizada).
  \item Deduza a força de restituição sobre a esfera e mostre que o seu
        movimento é harmônico, com $\omega_0^2 = \gamma PS^2/mV$.
  \item Calcule o período e o valor de $\gamma$ que se deduziria de um
        período medido de \qty{1.10}{s}.
  \item Por que o atrito entre a esfera e o tubo, e o vazamento de ar,
        enviesam o resultado, e em que sentido?
  \item Resuma as três compressões: que grandeza a primeira lei conservou
        em cada uma, e o que o expoente adiabático $\gamma$ governou.
\end{enumerate}
\end{problem}
