\chapter{Momento angular}\label{ch:b1:angular-momentum}

Uma patinadora que gira de braços abertos os recolhe e, sem
que ninguém a empurre, roda duas vezes mais depressa. Uma porta cede à ponta de um dedo
na maçaneta e resiste a um ombro junto à dobradiça. Os planetas varrem
áreas iguais em tempos iguais --- Kepler notou isso há quatro séculos, muito
antes que alguém pudesse dizer por quê. E a Lua, puxada pelas marés que
levanta, se afasta da Terra alguns centímetros por ano enquanto os nossos
dias se alongam imperceptivelmente. Por trás dos quatro casos está uma única grandeza
vetorial, o \omterm{def:b1:angular-momentum:L}{momento angular}, e uma única lei: a sua taxa de variação é igual
ao \omterm{def:b1:angular-momentum:moment}{momento} das forças. Este capítulo define os dois, demonstra a lei para
um \omterm{def:b1:point-kinematics:frame}{ponto material} e para um sistema, e a põe a trabalhar em pêndulos, órbitas
e coisas que giram.

\section{Momento de uma força}

\begin{definition}[Momento em relação a um ponto; em relação a um eixo]\label{def:b1:angular-momentum:moment}
\index{momento de uma força}\index{torque}\index{braço de alavanca}
O \emph{momento} (ou \emph{torque}) em relação a um ponto $O$ de uma força
$\vect F$ aplicada em $M$ é o vetor
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
perpendicular ao plano de $\vect{OM}$ e $\vect F$, de norma
$OM\cdot F\sin\alpha = F\,d$, em que $d$ --- o \emph{braço de alavanca} --- é a
distância de $O$ à linha de ação de $\vect F$. O momento em relação a um
eixo orientado $\Delta$ (vetor unitário $\vect e_\Delta$) que passa por $O$ é o
escalar $\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$, independente
da escolha de $O$ sobre $\Delta$; ele se anula se $\vect F$ for paralela a
$\Delta$ ou o encontrar. Duas forças opostas sobre linhas de ação diferentes
formam um \emph{binário}, cujo momento $\vect F\wedge\vect{AB}$ é o mesmo
em relação a todos os pontos.
\end{definition}

\begin{example}[A porta]\label{ex:b1:angular-momentum:door}
Um empurrão de \qty{20}{N} perpendicular a uma porta, a \qty{0.80}{m} das
dobradiças, tem \omterm{def:b1:angular-momentum:moment}{momento} $\qty{16}{N.m}$ em relação ao eixo das dobradiças; o mesmo empurrão
a \qty{0.10}{m}, \qty{2}{N.m}; ao longo da porta (linha de ação passando
pelas dobradiças), zero. A reação da própria dobradiça, que encontra o eixo, não tem
\omterm{def:b1:angular-momentum:moment}{momento} em relação a ele: é isso que faz do eixo o lugar natural para tomar
\omterm{def:b1:angular-momentum:moment}{momentos}.
\end{example}

\section{Momento angular de um ponto material}

\begin{definition}[Momento angular]\label{def:b1:angular-momentum:L}
\index{momento angular}
O \emph{\omterm{def:b1:angular-momentum:moment}{momento} angular} em relação a $O$ de um \omterm{def:b1:point-kinematics:frame}{ponto material} de massa $m$ em $M$
com velocidade $\vect v$ é
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
e, em relação a um eixo $\Delta$ que passa por $O$, $L_\Delta = \vect L_O\cdot\vect e_\Delta$.
Para um movimento num plano perpendicular a $\Delta$, em \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares}
em torno do eixo, $L_\Delta = mr^2\dot\theta$; num círculo de raio $R$,
$L_\Delta = mR^2\omega = mRv$.
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$, $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[Teorema do momento angular]\label{thm:b1:angular-momentum:theorem}
\index{teorema do momento angular}
Num \omterm{thm:b1:newton-dynamics:laws}{referencial inercial}, para um ponto $O$ fixo nesse \omterm{def:b1:point-kinematics:frame}{referencial},
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
para um eixo fixo $\Delta$. Se o \omterm{def:b1:angular-momentum:moment}{momento} total em relação a $O$ se anula,
$\vect L_O$ se conserva.
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$, usando a segunda lei de Newton
($O$ fixo, de modo que $\dd\vect{OM}/\dd t = \vect v$). Projete sobre $\vect e_\Delta$.
\end{proof}

\begin{corollary}[Forças centrais: movimento plano e lei das áreas]\label{cor:b1:angular-momentum:central}
\index{força central}\index{lei das áreas}\index{velocidade areolar}
Um \omterm{def:b1:point-kinematics:frame}{ponto material} submetido apenas a uma \emph{força central} (sempre dirigida
ao longo de $\vect{OM}$, para o centro fixo $O$ ou para longe dele) mantém
$\vect L_O$ constante. O seu movimento permanece no plano que passa por $O$
perpendicular a $\vect L_O$ e, nesse plano, $r^2\dot\theta = C$
(constante): o raio vetor $\vect{OM}$ varre área com a
\emph{\omterm{cor:b1:angular-momentum:central}{velocidade areolar}} constante
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$ para $\vect F \parallel \vect{OM}$: $\vect L_O$
é constante. Sempre $\vect{OM}\perp\vect L_O$, donde o plano; $L_O =
mr^2\dot\theta$ é constante. A área varrida durante $\dd t$ é a do
triângulo de lados $\vect{OM}$ e $\vect v\dd t$: $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$.
\end{proof}

\begin{example}[O pêndulo pelos momentos]\label{ex:b1:angular-momentum:pendulum}
\omterm{ex:b1:newton-dynamics:pendulum}{Pêndulo simples}, com eixo $\Delta$ pelo pivô e perpendicular ao
plano de oscilação: $L_\Delta = m\ell^2\dot\theta$; a tensão encontra o
eixo (\omterm{def:b1:angular-momentum:moment}{momento} nulo) e o \omterm{def:b1:angular-momentum:moment}{momento} do peso vale $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{braço de alavanca}
$\ell\sin\theta$, restaurador). O teorema dá $m\ell^2\ddot\theta =
-mg\ell\sin\theta$, a equação do \cref{ex:b1:newton-dynamics:pendulum}
sem jamais escrever a tensão --- a principal vantagem dos \omterm{def:b1:angular-momentum:moment}{momentos}:
as forças que passam pelo eixo desaparecem.
\end{example}

\begin{example}[Puxando um disco para dentro]\label{ex:b1:angular-momentum:puck}
Um disco desliza sem atrito no gelo em círculo, preso a um fio que passa por
um furo no centro; puxa-se o fio de modo que o raio caia à metade.
A tensão é central: $L = mr^2\omega$ se conserva, de modo que $\omega$ fica
multiplicada por $4$, a velocidade $r\omega$ por $2$ e a \omterm{thm:b1:work-and-energy:ke}{energia cinética} por $4$
--- a energia extra é o trabalho da puxada, à qual o disco resiste
todo o caminho para dentro. Os braços da patinadora são a mesma física, com a massa
distribuída (\cref{exo:b1:angular-momentum:4}).
\end{example}

\section{Sistemas de pontos e rotação em torno de um eixo fixo}

\begin{theorem}[Momento angular de um sistema]\label{thm:b1:angular-momentum:system}
\index{momento angular!de um sistema}
Para um sistema de \omterm{def:b1:point-kinematics:frame}{pontos materiais}, com $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
e $O$ fixo num \omterm{thm:b1:newton-dynamics:laws}{referencial inercial},
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
só contam os \omterm{def:b1:angular-momentum:moment}{momentos} das forças \emph{externas}; as forças internas,
que obedecem à lei da \omterm{thm:b1:newton-dynamics:laws}{ação e reação} ao longo da reta que une os
pontos, nada contribuem.
\end{theorem}

\begin{proof}
Some os teoremas de cada ponto. Para um par interno, $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$, pois a força é dirigida ao longo de
$\vect{M_jM_i}$.
\end{proof}

\begin{proposition}[Rotação rígida em torno de um eixo fixo]\label{prop:b1:angular-momentum:inertia}
\index{momento de inércia}
Se os pontos mantêm distâncias fixas $r_i$ a um eixo $\Delta$ e todos
giram em torno dele com a mesma \omterm{cor:b1:point-kinematics:circular}{velocidade angular} $\omega$ (uma rotação rígida),
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
sendo $J_\Delta$ o \emph{\omterm{prop:b1:angular-momentum:inertia}{momento de inércia}} em relação a $\Delta$, e o
teorema se lê $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$.
\end{proposition}

\begin{proof}
Cada ponto contribui com $m_ir_i^2\omega$ para $L_\Delta$ e com
$\tfrac12 m_i(r_i\omega)^2$ para $E_k$. O \cref{ch:b1:systems-of-points}
estende isso a sólidos contínuos.
\end{proof}

\begin{example}[A patinadora]\label{ex:b1:angular-momentum:skater}
De braços abertos, o corpo de uma patinadora tem $J \approx \qty{4}{kg.m^2}$ mais dois
braços de \qty{3}{kg} a \qty{0.8}{m}: $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$. Braços recolhidos (\qty{0.2}{m}): $J_2 = \qty{4.2}{kg.m^2}$.
A reação do gelo e o peso não têm \omterm{def:b1:angular-momentum:moment}{momento} em relação ao eixo
vertical: $J\omega$ se conserva e a taxa de giro sobe por $J_1/J_2 = 1.85$
--- de $2$ para $3.7$ voltas por segundo --- enquanto a \omterm{thm:b1:work-and-energy:ke}{energia cinética} sobe
pelo mesmo fator, paga pelos músculos que recolhem os braços.
\end{example}

\begin{remark}[Estática]\label{rem:b1:angular-momentum:statics}
\index{equilíbrio de um sólido}
Um sólido em repouso tem $\vect L_O = \vect 0$ constante: as forças externas
devem ter resultante nula \emph{e} \omterm{def:b1:angular-momentum:moment}{momento} total nulo em relação a qualquer ponto.
A segunda condição é a lei da alavanca, da balança, da gangorra e
do pé de cabra: uma força pequena longe do pivô equilibra uma grande perto
dele.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:1}
Uma força de \qty{50}{N} numa chave a \qty{0.30}{m} da porca: \omterm{def:b1:angular-momentum:moment}{momento} quando
a força é perpendicular ao cabo; a $60^\circ$ dele; ao longo
dele.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:2}
\omterm{def:b1:angular-momentum:L}{Momento angular} da Terra em relação ao Sol (órbita circular,
$r = \qty{1.50e11}{m}$, $v = \qty{29.8}{km/s}$, $m = \qty{5.97e24}{kg}$),
e a sua \omterm{cor:b1:angular-momentum:central}{velocidade areolar}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:3}
Um pêndulo cônico (fio de \qty{1.0}{m}, $30^\circ$ com a vertical,
\cref{exo:b1:newton-dynamics:9}). Calcule $L$ em relação ao eixo vertical
que passa pelo pivô, por unidade de massa, e explique por que ele se conserva
embora a tensão do fio e o peso atuem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:4}
O tronco de uma patinadora tem $J = \qty{4.0}{kg.m^2}$ em relação ao eixo de giro; cada
braço (\qty{3.0}{kg}) é tratado como um ponto a \qty{0.80}{m} e depois a
\qty{0.20}{m}. Taxa de giro depois de recolher os braços, partindo de
\qty{2.0}{} voltas por segundo; razão das \omterm{thm:b1:work-and-energy:ke}{energias cinéticas}; de onde vem a
energia extra?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:5}
Deduza a equação do pêndulo $\ell\ddot\theta + g\sin\theta = 0$ a partir do
teorema do \omterm{def:b1:angular-momentum:L}{momento angular} em relação ao eixo do pivô, e dê o \omterm{def:b1:angular-momentum:moment}{momento} de
cada força.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:6}
A distância da Terra ao Sol varia de \qty{1.471e11}{m} (periélio)
a \qty{1.521e11}{m} (afélio). Pela \omterm{cor:b1:angular-momentum:central}{lei das áreas}, encontre a razão

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:7}
Um disco de \qty{0.20}{kg} gira a \qty{2.0}{m/s} preso a um fio de \qty{0.50}{m}

Nova velocidade e nova \omterm{cor:b1:point-kinematics:circular}{velocidade angular}; \omterm{thm:b1:work-and-energy:ke}{energias cinéticas} antes e depois; trabalho
realizado pela puxada. Por que a tensão consegue realizar trabalho aqui?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:9}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:11}
Uma estrela como o Sol ($R = \qty{7.0e8}{m}$, período de rotação \qty{25}{d})
colapsa numa estrela de nêutrons de raio \qty{10}{km}, conservando a sua massa

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:12}

\section{Problema: por que a Lua está indo embora}

\begin{problem}[{Problema de fim de semana --- as marés que a Lua levanta na
Terra a puxam para a frente e freiam a Terra: como o momento angular flui
dos nossos dias para a órbita da Lua, centímetro a centímetro, e
para onde vai a energia}]
\label{pb:b1:angular-momentum:1}
Dados: $G = \qty{6.67e-11}{SI}$; Terra $M = \qty{5.97e24}{kg}$, $R =
\qty{6.37e6}{m}$, rotação $\Omega = 2\pi/(\qty{86164}{s})$, \omterm{prop:b1:angular-momentum:inertia}{momento de inércia}
$J = 0.33\,MR^2$; Lua $m = \qty{7.35e22}{kg}$, órbita tratada como
circular, $r = \qty{3.84e8}{m}$; a telemetria laser lunar mede
$\dd r/\dd t = \qty{3.8}{cm/yr}$; o dia se alonga \qty{2.3}{ms} por
século; $1$ século $= \qty{3.16e9}{s}$.

\medskip
\noindent\textbf{Parte I --- Os \omterm{def:b1:angular-momentum:L}{momentos angulares}.}
\begin{enumerate}
  \item Calcule a \omterm{cor:b1:point-kinematics:circular}{velocidade angular} orbital $\omega$ da Lua pela
        lei de Newton (órbita circular) e confira com o
        mês sideral de \qty{27.3}{d}.
  \item Calcule o \omterm{def:b1:angular-momentum:L}{momento angular} orbital da Lua, $L_{\text{orb}} =
        mr^2\omega$.
  \item Mostre que, para uma órbita circular, $L_{\text{orb}} = m\sqrt{GMr}$, e
        deduza $\dd L_{\text{orb}}/\dd r = L_{\text{orb}}/2r$.
  \item Calcule o \omterm{def:b1:angular-momentum:L}{momento angular} de rotação da Terra, $L_{\text{rot}} =
        J\Omega$, e compare com $L_{\text{orb}}$.
  \item A Lua gira uma vez por mês: estime o seu \omterm{def:b1:angular-momentum:L}{momento angular} de
        rotação ($J_m \approx 0.4\,mR_m^2$, $R_m = \qty{1.74e6}{m}$) e
        justifique desprezá-lo.
  \item Que \omterm{def:b1:angular-momentum:moment}{momentos} externos agem sobre o sistema Terra--Lua em relação ao seu
        centro de massa? Argumente que, com boa aproximação, o total
        $L = L_{\text{rot}} + L_{\text{orb}}$ se conserva.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O \omterm{def:b1:angular-momentum:moment}{torque} de maré.}
A Lua levanta duas protuberâncias de água na Terra; como a Terra
gira mais depressa do que a Lua orbita, o atrito arrasta as protuberâncias à frente
da linha Terra--Lua por um pequeno ângulo.
\begin{enumerate}[resume]
  \item Esboce a Terra, a protuberância adiantada e a Lua. Em que
        sentido age a atração da Lua sobre a protuberância mais próxima,
        em relação à rotação da Terra?
  \item Deduza o sinal do \omterm{def:b1:angular-momentum:moment}{momento} sobre a rotação da Terra: o
        dia se alonga ou encurta?
  \item Por \omterm{thm:b1:newton-dynamics:laws}{ação e reação}, que \omterm{def:b1:angular-momentum:moment}{momento} age sobre o movimento orbital
        da Lua? $L_{\text{orb}}$ cresce ou diminui?
  \item Pela questão 3, $r$ cresce ou diminui? E a velocidade orbital
        da Lua, $v = \sqrt{GM/r}$?
  \item Resolva o aparente paradoxo: a Lua é empurrada para a frente e, ainda
        assim, acaba se movendo mais devagar.
  \item Para que estado final tende essa troca?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Os números.}
\begin{enumerate}[resume]
  \item A partir do alongamento do dia, calcule a variação relativa
        $\Delta\Omega/\Omega$ por século.
  \item Deduza a variação de $L_{\text{rot}}$ por século.
  \item Deduza a variação de $L_{\text{orb}}$ por século.
  \item Usando a questão 3, converta-a numa variação de $r$ por século
        e por ano. Compare com o valor da telemetria laser.
  \item Em que fração, e em quantos segundos, o mês sideral
        se alonga por século ($T \propto r^{3/2}$)?
  \item Calcule a variação da \omterm{thm:b1:work-and-energy:ke}{energia cinética} de rotação da Terra
        por século ($\Delta E = L_{\text{rot}}\Delta\Omega$ em primeira
        ordem).
  \item Calcule a variação da \omterm{thm:b1:work-and-energy:em}{energia mecânica} orbital da Lua,
        $E_{\text{orb}} = -GMm/2r$, por século.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A energia.}
\begin{enumerate}[resume]
  \item Mostre que a Terra perde mais energia do que a Lua ganha, e
        calcule a diferença por século e a potência correspondente.
        Para onde ela vai?
  \item Compare essa potência com o consumo da humanidade, cerca de
        \qty{18}{TW}, e com o fluxo de calor interno da Terra, cerca de
        \qty{47}{TW}.
  \item No estado final da questão 12, dia e mês são iguais:
        $\Omega = \omega = \sqrt{GM/r^3}$. Escrevendo a conservação de
        $L_{\text{rot}} + L_{\text{orb}}$, verifique que $r \approx
        \qty{5.5e8}{m}$ a satisfaz, e dê o período comum em dias.
  \item Por que esse estado nunca será de fato alcançado? (Pense nas
        marés do Sol.)
  \item A Lua já mostra sempre a mesma face à Terra. Explique pelo mesmo
        mecanismo e diga por que isso aconteceu primeiro com a Lua.
  \item Resuma em três linhas: o que se conserva, o que é trocado,
        o que é dissipado.
\end{enumerate}
\end{problem}
