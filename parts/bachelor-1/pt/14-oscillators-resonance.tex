\chapter{Osciladores mecânicos: amortecimento e ressonância}\label{ch:b1:oscillators-resonance}

Empurre o balanço de uma criança na hora errada e quase nada acontece; empurre-o
uma vez por período, com suavidade, e em uma dezena de idas e vindas a criança já
grita no alto do arco. Uma máquina de lavar treme violentamente
a uma velocidade determinada da centrifugação e fica quieta acima dela. Uma
taça de cristal canta quando um dedo molhado percorre a sua borda no ritmo
certo, e se estilhaça quando um cantor sustenta a mesma nota alto o bastante.
Todos esses casos são uma \emph{ressonância}: um sistema com uma \omterm{def:b1:oscillators-resonance:harmonic}{frequência
própria}, excitado perto dessa frequência, acumula energia ciclo após
ciclo até que o atrito a drene tão depressa quanto ela entra. Este
capítulo toma o oscilador mecânico --- livre, amortecido, depois forçado
--- e deduz as curvas de ressonância cujas gêmeas elétricas apareceram
no \cref{ch:b1:sinusoidal-impedance}.

\section{O oscilador harmônico}

\begin{definition}[Oscilador harmônico]\label{def:b1:oscillators-resonance:harmonic}
\index{oscilador harmônico}\index{frequência própria}
Um sistema cujo deslocamento $x$ em relação a um equilíbrio estável obedece a
\[
\ddot x + \omega_0^2\,x = 0
\]
é um \emph{oscilador harmônico} de \emph{\omterm{def:b1:transient-regimes:secondorder}{frequência angular própria}}
$\omega_0$ (período $T_0 = 2\pi/\omega_0$). A solução $x = A\cos(\omega_0t
+ \varphi)$ tem \omterm{def:b1:wave-propagation:signal}{amplitude} e fase fixadas pelas condições iniciais
e um período independente da \omterm{def:b1:wave-propagation:signal}{amplitude} (\emph{isocronismo}).
\end{definition}

\begin{proposition}[De onde vêm os osciladores harmônicos]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item Massa $m$ numa mola $k$: $\omega_0^2 = k/m$ (\omterm{prop:b1:newton-dynamics:springstring}{lei de Hooke}).
  \item \omterm{ex:b1:newton-dynamics:pendulum}{Pêndulo simples} de comprimento $\ell$, ângulos pequenos: $\omega_0^2 = g/\ell$.
  \item Qualquer \omterm{def:b1:point-kinematics:frame}{ponto material} perto de um equilíbrio estável de um potencial $E_p$:
        $\omega_0^2 = E_p''(x_0)/m$ (\cref{prop:b1:work-and-energy:stability}).
  \item Circuito LC: $\omega_0^2 = 1/LC$, com $q \leftrightarrow x$,
        $L \leftrightarrow m$, $1/C \leftrightarrow k$.
\end{itemize}
A energia $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ é constante e vai e volta entre as formas cinética e
potencial duas vezes por período.
\end{proposition}

\begin{proof}
Segunda lei de Newton com $F = -kx$; a equação do pêndulo
$\ddot\theta + (g/\ell)\sin\theta = 0$ com $\sin\theta \approx \theta$;
a fórmula de Taylor no mínimo de $E_p$; a \omterm{thm:b1:dc-circuits:kirchhoff}{lei das malhas} do circuito
LC. Energia: derive e use a equação.
\end{proof}

\begin{example}[Um cilindro que bóia e balança]\label{ex:b1:oscillators-resonance:bob}
Um cilindro vertical de seção $S$ flutua com uma profundidade $h$
imersa ($mg = \rho_wgSh$, empuxo do \cref{ch:b1:fluid-statics}).
Afundado de $x$, ele recebe uma força adicional para cima $\rho_wgSx$:
$m\ddot x = -\rho_wgSx$, logo $\omega_0^2 = \rho_wgS/m = g/h$ --- o mesmo
que um pêndulo de comprimento $h$. Uma boia imersa \qty{1}{m} balança com
período de \qty{2}{s}, qualquer que seja a sua massa.
\end{example}

\section{O oscilador amortecido}

\begin{proposition}[Oscilador amortecido]\label{prop:b1:oscillators-resonance:damped}
\index{oscilador amortecido}\index{fator de qualidade!mecânico}
Com um atrito viscoso $-\alpha\dot x$, a equação se torna
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
a forma canônica de segunda ordem do \cref{ch:b1:transient-regimes}: para
$Q > \tfrac12$, um decaimento pseudoperiódico $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$ com $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$; a
energia decai como $\eu^{-2t/\tau}$ e
\[
Q = 2\pi\,\frac{\text{energia armazenada}}{\text{energia perdida por período}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
Divida $m\ddot x + \alpha\dot x + kx = 0$ por $m$. Para a energia: $E
\propto A^2\eu^{-2t/\tau}$ perde a fração $2T/\tau = 2\pi/Q$ por
período quando esta é pequena.
\end{proof}

\begin{example}[Tempos de extinção]\label{ex:b1:oscillators-resonance:ringing}
Um diapasão de \qty{440}{Hz} ainda soa dez segundos depois de
percutido: a sua \omterm{def:b1:wave-propagation:signal}{amplitude} caiu a cerca de $5\%$ em $3\tau = \qty{10}{s}$, logo
$\tau = \qty{3.3}{s}$ e $Q = \omega_0\tau/2 \approx 4600$. A suspensão de um
carro ($Q \approx 0.7$) para em menos de um período; um balanço de criança
($Q \approx 20$) perde um terço da sua energia por vaivém e precisa de um empurrão
a cada vez.
\end{example}

\section{Oscilações forçadas e ressonância}

\begin{theorem}[Oscilador excitado senoidalmente]\label{thm:b1:oscillators-resonance:forced}
\index{oscilações forçadas}\index{ressonância!em amplitude}\index{ressonância!em velocidade}
Excitado por uma força $F_0\cos\omega t$, o \omterm{prop:b1:oscillators-resonance:damped}{oscilador amortecido}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
se estabelece, após um transitório de duração $\sim\tau$, no \omterm{prop:b1:sinusoidal-impedance:forced}{regime
forçado} $x = X\cos(\omega t + \varphi)$ com, escrevendo $x_r = \omega/\omega_0$
e $X_0 = F_0/k$ a deflexão estática,
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
A \emph{\omterm{def:b1:wave-propagation:signal}{amplitude}} $X$ passa por um máximo, para $Q > 1/\sqrt2$, em $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$, com $X_{\max} \approx QX_0$; a \emph{\omterm{def:b1:wave-propagation:signal}{amplitude}
da velocidade} $V = \omega X$ é máxima exatamente em $\omega_0$, onde $V_{\max} =
F_0/\alpha$, e a sua curva de ressonância
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
tem \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} $\Delta\omega = \omega_0/Q$. Em $\omega_0$ o deslocamento
atrasa $\pi/2$ em relação à força, e a velocidade está em fase com ela.
\end{theorem}

\begin{proof}
Método complexo (\cref{ch:b1:sinusoidal-impedance}): $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$, logo $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$; divida numerador e denominador por
$\omega_0^2 = k/m$. O módulo e o argumento decorrem daí; o máximo de $X$ foi
localizado na \cref{prop:b1:sinusoidal-impedance:overvoltage} (mesma
função). Velocidade: $\underline V = j\omega\underline X$ dá $V = \omega X$;
escrever $(1 - x_r^2)/x_r = 1/x_r - x_r$ transforma o denominador em
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$, isto é,\ a forma do RLC. Em $x_r = 1$,
$\underline X = -jQX_0$: um atraso de $\pi/2$.
\end{proof}

\begin{remark}[Duas ressonâncias, um circuito]\label{rem:b1:oscillators-resonance:analogy}
\index{analogia eletromecânica}
A correspondência $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$,
$F \leftrightarrow e$ leva o oscilador forçado ao RLC série do
\cref{ch:b1:sinusoidal-impedance}: a ressonância em velocidade é a ressonância
em corrente (sempre em $\omega_0$, com \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} $\omega_0/Q$), e a ressonância
em \omterm{def:b1:wave-propagation:signal}{amplitude} é a ressonância da carga, isto é,\ da \omterm{def:b1:dc-circuits:voltage}{tensão} no capacitor (ligeiramente
abaixo de $\omega_0$, ausente para $Q$ baixo). A potência absorvida, $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$, é máxima em $\omega_0$, com o valor $F_0^2/2\alpha$,
e os pontos de meia potência delimitam a \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} em velocidade --- exatamente como no
RLC.
\end{remark}

\begin{example}[Velocidades de ressonância]\label{ex:b1:oscillators-resonance:road}
A suspensão de um carro tem $f_0 \approx \qty{1.3}{Hz}$ (\cref{pb:b1:transient-regimes:1}).
Numa estrada com ondulações a cada $\lambda = \qty{10}{m}$, a frequência de
excitação vale $v/\lambda$: ressonância em $v = \lambda f_0 = \qty{13}{m/s}$,
cerca de \qty{47}{km/h} --- a velocidade em que uma estrada ondulada sacode mais o
carro, e a razão de se montar um amortecedor com $Q \approx 0.7$:
com $Q = 5$ a carroceria oscilaria com cinco vezes a \omterm{def:b1:wave-propagation:signal}{amplitude} da
estrada. Acima da ressonância ($x_r \gg 1$) a \omterm{def:b1:wave-propagation:signal}{amplitude} cai como
$X_0/x_r^2$: a carroceria ``flutua'' sobre as ondulações rápidas --- o isolamento, a outra
face da ressonância.
\end{example}

\section{Excitação pela base: isolamento e o sismômetro}

\begin{proposition}[Oscilador sacudido pelo seu suporte]\label{prop:b1:oscillators-resonance:base}
\index{excitação pela base}\index{isolamento de vibrações}\index{sismômetro}
Seja o suporte de um conjunto massa--mola--amortecedor deslocado de $x_g(t)$ (a estrada
sob um carro, o solo sob um sismômetro), e seja $u$ o
deslocamento da massa em relação ao suporte e $x = x_g + u$ o seu
deslocamento absoluto (a partir do equilíbrio). Então
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
e para $x_g = X_g\cos\omega t$, com $x_r = \omega/\omega_0$:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
Para $x_r \gg 1$: $U \to X_g$ (a massa fica parada no espaço --- um
\emph{sismômetro} lê o deslocamento do solo) e $X \to 0$ (a
massa fica \emph{isolada} do tremor); para $x_r \ll 1$: $U \approx
\ddot x_g/\omega_0^2$ (um \emph{acelerômetro}) e $X \approx X_g$ (a massa
acompanha o suporte).
\end{proposition}

\begin{proof}
A mola e o amortecedor agem sobre o movimento relativo: $m\ddot x = -ku -
\alpha\dot u$, e $\ddot x = \ddot x_g + \ddot u$. Em forma complexa,
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$, donde
$U/X_g$; depois $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$.
\end{proof}

\begin{example}[Por que as máquinas ficam sobre coxins macios]\label{ex:b1:oscillators-resonance:mounts}
Um motor de \qty{50}{Hz} sobre coxins de borracha sintonizados em $f_0 = \qty{10}{Hz}$
($x_r = 5$, $Q \approx 2$): $X/X_g = \sqrt{(1 + 6.25)/(576 + 6.25)} = 0.11$
--- nove décimos da vibração não chegam ao piso. Os coxins precisam ser
\emph{mais macios} do que a intuição sugere: enrijecê-los eleva $f_0$
rumo aos $50$ Hz e destrói o isolamento. A mesma fórmula, lida ao
contrário, é o carro do \cref{ex:b1:oscillators-resonance:road} numa
estrada ondulada.
\end{example}

\section{Energia: como uma ressonância se constrói}

\begin{proposition}[Balanço energético no regime forçado]\label{prop:b1:oscillators-resonance:energy}
No \omterm{prop:b1:sinusoidal-impedance:forced}{regime forçado} permanente, a \omterm{thm:b1:sinusoidal-impedance:power}{potência média} fornecida pela força
excitadora iguala a \omterm{thm:b1:sinusoidal-impedance:power}{potência média} dissipada pelo atrito, $\tfrac12\alpha V^2$.
Na ressonância a energia armazenada $\tfrac12 mV_{\max}^2$ vale $Q/2\pi$ vezes
a energia fornecida por período: o oscilador acumula o trabalho da excitação
durante cerca de $Q$ ciclos antes que as perdas igualem a entrada, que é também
o número de ciclos ($\sim\tau/T$) que o transitório dura.
\end{proposition}

\begin{proof}
Tome a média do teorema da energia sobre um período: $\langle\dd E/\dd t\rangle = 0$
em \omterm{def:b1:transient-regimes:firstorder}{regime permanente}, logo $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$. Na ressonância $V_{\max} = F_0/\alpha$, energia armazenada
$\tfrac12 m(F_0/\alpha)^2$, energia fornecida por período $\tfrac12\alpha(F_0/\alpha)^2T$;
razão $m/(\alpha T) = Q/2\pi$.
\end{proof}

\begin{example}[O balanço]\label{ex:b1:oscillators-resonance:swing}
Um balanço de \qty{2.5}{m} ($T_0 = \qty{3.2}{s}$, $Q \approx 20$) recebe
um empurrão de \qty{50}{N} ao longo de \qty{0.5}{m} por período: \qty{25}{J} por
ciclo. Ele cresce até que a perda por período, $2\pi E/Q$, iguale
\qty{25}{J}: $E = \qty{80}{J}$; com uma criança de \qty{30}{kg}, $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ dá $\theta_{\max} \approx 27^\circ$.
Empurrando em qualquer outro ritmo, o trabalho dos empurrões alternaria de
sinal e daria média nula.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:1}
Uma massa de \qty{0.20}{kg} numa mola de constante elástica \qty{80}{N/m} oscila
com \omterm{def:b1:wave-propagation:signal}{amplitude} de \qty{5.0}{cm}. \omterm{def:b1:wave-propagation:signal}{Frequência angular}, frequência, período,
velocidade máxima, energia total.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:2}
Período de um pêndulo de \qty{1.0}{m} na Terra e na Lua
($g = \qty{1.62}{m/s^2}$). O que acontece com o período do sistema massa--mola na
Lua?
\end{exercise}

\section{Problema: o sismógrafo}

\begin{problem}[{Problema de fim de semana --- uma massa pende de uma mola dentro
de uma caixa aparafusada ao solo; o solo treme: o que a pena
registra, por que o mesmo instrumento mede deslocamento em alta
frequência e aceleração em baixa frequência, e por que um sismômetro de
longo período não pode ser simplesmente uma mola longa}]
\label{pb:b1:oscillators-resonance:1}
Uma massa $m = \qty{1.0}{kg}$ pende de uma mola de constante elástica $k$ dentro
de uma armação rígida presa ao solo; um amortecedor exerce $-\alpha\dot u$, sendo
$u$ o deslocamento da massa em relação à armação, medido
a partir da sua posição de equilíbrio. O solo, e portanto a armação, se move
verticalmente segundo $x_g(t)$. O \omterm{def:b1:point-kinematics:frame}{referencial} das estrelas distantes é inercial.

\medskip
\noindent\textbf{Parte I --- A equação.}
\begin{enumerate}
  \item Escreva a posição absoluta da massa como $x = x_g + u +
        \text{const}$, liste as forças sobre a massa (no \omterm{thm:b1:newton-dynamics:laws}{referencial
        inercial}) e mostre que
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item Ponha-a na forma canônica; identifique $\omega_0$ e $Q$.
  \item O instrumento é construído com $f_0 = \qty{0.50}{Hz}$ e $Q = 0.70$:
        calcule $k$ e $\alpha$.
  \item Uma aceleração constante do solo $a_g$ (a armação se inclinando, digamos)
        dá que $u$ permanente? Interprete.
  \item O que registra a pena presa à massa, escrevendo num tambor fixo
        à armação?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- \omterm{prop:b1:point-kinematics:harmonic}{Movimento harmônico} do solo.}
O solo se move segundo $x_g = X_g\cos\omega t$; escreva $x_r = \omega/\omega_0$.
\begin{enumerate}[resume]
  \item Usando \omterm{def:b1:sinusoidal-impedance:complex}{amplitudes complexas}, mostre que $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$.
  \item Deduza $U/X_g$ em função de $x_r$ e de $Q$.
  \item Limite $x_r \gg 1$: mostre que $U \to X_g$ e explique fisicamente o que
        a massa faz (o que o instrumento está medindo?).
  \item Limite $x_r \ll 1$: mostre que $U \approx x_r^2X_g = a_g/\omega_0^2$, sendo
        $a_g$ a \omterm{def:b1:wave-propagation:signal}{amplitude} da aceleração do solo: o que é o instrumento
        agora?
  \item Valor em $x_r = 1$; por que $Q \approx 0.7$ é uma escolha sensata?
  \item Esboce $U/X_g$ contra $x_r$ (escalas logarítmicas) para $Q = 0.7$ e
        $Q = 5$.
  \item Fase de $u$ em relação a $x_g$ em alta frequência; interprete.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Projetando para terremotos.}
Terremotos distantes produzem ondas de superfície de \qty{0.05}{} a \qty{0.1}{Hz}
com \omterm{def:b1:wave-propagation:signal}{amplitudes} milimétricas; tremores locais chegam a \qty{5}{} ou \qty{10}{Hz}.
\begin{enumerate}[resume]
  \item Para registrar ondas de \qty{0.05}{Hz} como deslocamentos quer-se
        $f_0 \ll \qty{0.05}{Hz}$, digamos \qty{0.02}{Hz}. Que alongamento
        estático a mola teria sob o peso da massa? Comente.
  \item Um \omterm{ex:b1:newton-dynamics:pendulum}{pêndulo simples} de mesma frequência: que comprimento?
  \item Os instrumentos reais de longo período usam um pêndulo quase horizontal
        do tipo ``porteira de jardim'', ou uma realimentação eletrônica de força. Explique
        em uma frase o que cada solução consegue.
  \item Para o instrumento de \qty{0.50}{Hz}, calcule o alongamento estático:
        é viável?
  \item Com ele, uma onda de \qty{0.10}{Hz} e \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm} dá
        que $U$? Em que regime está o instrumento?
  \item Um tremor local de \qty{5.0}{Hz} e \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm}: qual $U$?
        Qual regime?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O acelerômetro de um celular.}
Uma massa micro-usinada sobre molas de silício tem $f_0 = \qty{1.0}{kHz}$ e
$Q = 0.7$.
\begin{enumerate}[resume]
  \item Um tremor de \qty{10}{Hz} com \omterm{def:b1:wave-propagation:signal}{amplitude} de aceleração $g$: calcule
        $U$. Como se lê um deslocamento desses?
  \item Até que frequência a leitura fica a menos de $10\%$ de $a_g/\omega_0^2$?
  \item Que \omterm{def:b1:transient-regimes:firstorder}{constante de tempo} $\tau = 2Q/\omega_0$ governa a sua resposta a um
        solavanco, e por que isso importa para detectar uma queda?
  \item O $\tau$ do sismômetro: valor e consequência após um degrau súbito
        do solo.
  \item Por que se escolhe o amortecimento crítico (ou quase crítico) para os dois
        instrumentos, em vez de um $Q$ alto?
  \item O celular é sacudido exatamente a \qty{1.0}{kHz}: por qual fator
        a leitura se afasta do valor correto?
  \item Resuma: um único dispositivo, dois regimes, decididos por qual única
        comparação.
\end{enumerate}
\end{problem}
