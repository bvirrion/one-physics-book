\chapter{Estática dos fluidos}\label{ch:b1:fluid-statics}

Um submarino a trezentos metros carrega, em cada metro quadrado do seu
casco, o peso de um caminhão carregado. Uma barragem segura um lago com
uma parede grossa embaixo e fina em cima, por uma razão que nada tem a ver
com o comprimento do lago. Um navio de cem mil toneladas flutua porque
empurra para o lado cem mil toneladas de água, e afunda um palmo a mais
quando passa do mar para um rio. Cada um desses fatos é consequência de uma
única relação --- num fluido em repouso a pressão cresce para baixo à taxa
$\rho g$ --- e este capítulo a deduz, aplica-a aos líquidos e à atmosfera e
tira dela a força sobre uma parede e o teorema de Arquimedes.

\section{A pressão num fluido em repouso}

\begin{proposition}[Isotropia da pressão]\label{prop:b1:fluid-statics:isotropy}
\index{pressão!num fluido}
Num fluido em repouso, a força exercida pelo fluido sobre uma pequena
superfície $\dd S$ colocada num ponto $M$ vale $\dd\vect F = P(M)\,\dd S\,\vect n$, normal
à superfície e dirigida para ela, com uma pressão $P(M)$ que depende do
ponto mas \emph{não} da orientação da superfície.
\end{proposition}

\begin{proof}[Demonstração parcial]
Um fluido em repouso não transmite força tangencial (senão escoaria): a
força é normal. Considere um pequeníssimo prisma triangular de fluido em $M$
com faces de orientações diferentes: ele está em equilíbrio sob as forças
de pressão nas suas faces e sob o seu peso; o peso varia como o volume
(cubo do tamanho) e as forças de pressão como áreas (quadrado), de modo que,
para um prisma bastante pequeno, só as forças de pressão contam, e o seu
equilíbrio em toda direção exige o mesmo $P$ em todas as faces.
\end{proof}

\begin{theorem}[Relação fundamental da estática dos fluidos]\label{thm:b1:fluid-statics:fundamental}
\index{relação hidrostática}
Num fluido em repouso na gravidade uniforme $\vect g = -g\vect e_z$ ($z$
para cima), a pressão depende só de $z$ e
\[
\frac{\dd P}{\dd z} = -\rho g ,
\]
sendo $\rho$ a densidade do fluido nesse nível. A pressão cresce para
baixo; as superfícies de pressão igual (as \emph{isóbaras}) são horizontais,
e a superfície livre de um líquido também.
\end{theorem}

\begin{proof}
Tome uma fatia horizontal de fluido de área $S$ entre $z$ e $z + \dd z$:
ela é empurrada para cima por $P(z)S$ de baixo, para baixo por $P(z + \dd z)S$ de cima,
e puxada para baixo pelo seu peso $\rho gS\,\dd z$. Em repouso:
$P(z)S - P(z + \dd z)S - \rho gS\,\dd z = 0$. As fatias horizontais de fluido
não recebem força horizontal da gravidade, logo na horizontal $P$ é uniforme.
\end{proof}

\begin{corollary}[Fluido incompressível]\label{cor:b1:fluid-statics:incompressible}
\index{pressão hidrostática}
Num líquido de densidade uniforme $\rho$, a pressão à profundidade $h$
abaixo de um ponto onde ela vale $P_0$ é
\[
P = P_0 + \rho gh .
\]
Água: \qty{1}{bar} a mais a cada \qty{10}{m}. Dois pontos de um líquido
conectado no mesmo nível estão à mesma pressão; as superfícies livres de
vasos comunicantes ficam num só nível; uma pressão aplicada em qualquer
ponto de um líquido confinado é transmitida sem atenuação a todos os
pontos (princípio de Pascal).
\end{corollary}

\begin{proof}
Integre $\dd P/\dd z = -\rho g$ com $\rho$ constante; o resto é a
horizontalidade das isóbaras e a unicidade de $P$ num dado nível.
\end{proof}

\begin{example}[Manômetros e prensas]\label{ex:b1:fluid-statics:manometer}
\index{manômetro}\index{prensa hidráulica}
Um tubo em U de mercúrio ($\rho = \qty{13600}{kg/m^3}$) cujas colunas diferem de
$h = \qty{760}{mm}$ mede uma diferença de pressão $\rho gh = \qty{1.013e5}{Pa}$
--- a atmosfera, que sustenta uma coluna de \qty{10.3}{m} de água. Uma
\omterm{ex:b1:fluid-statics:manometer}{prensa hidráulica}: uma força $F_1$ sobre um pistão de área $S_1$ eleva a
pressão de $F_1/S_1$ em todo o líquido; sobre o pistão grande $S_2$ ela dá
$F_2 = F_1S_2/S_1$ --- cinquenta vezes mais para cinquenta vezes a área, ao
preço de cinquenta vezes menos curso (a energia se conserva: $F_1d_1 = F_2d_2$).
\end{example}

\section{A atmosfera}

\begin{proposition}[Atmosfera isotérmica]\label{prop:b1:fluid-statics:atmosphere}
\index{fórmula barométrica}\index{altura de escala}
Para um \omterm{def:b1:kinetic-theory:model}{gás perfeito} de massa molar $M$ à temperatura uniforme $T$ numa
gravidade uniforme,
\[
P(z) = P_0\,\eu^{-z/H}, \qquad H = \frac{RT}{Mg} ,
\]
a \emph{\omterm{prop:b1:fluid-statics:atmosphere}{altura de escala}}: cerca de \qty{8}{km} para o ar; a pressão cai pela
metade a cada $H\ln2 \approx \qty{5.5}{km}$.
\end{proposition}

\begin{proof}
$\rho = PM/RT$ (equação de estado), logo $\dd P/\dd z = -(Mg/RT)P$: uma
equação linear de primeira ordem, $P = P_0\eu^{-Mgz/RT}$.
\end{proof}

\begin{example}[Ar rarefeito]\label{ex:b1:fluid-statics:altitude}
Com $T = \qty{260}{K}$ (uma média sobre a baixa atmosfera), $H =
\qty{7.6}{km}$: $P = 0.52P_0$ a \qty{5000}{m} e $0.31P_0$ no cume do
Everest --- perto dos valores medidos, $0.54$ e $0.33$; a atmosfera real
esfria com a altitude, o que o volume do segundo ano leva em conta. O
modelo também diz por que o mar é diferente: a água é oitocentas vezes mais
densa e quase incompressível, de modo que a sua pressão cresce linearmente,
um bar a cada dez metros, sem \omterm{prop:b1:fluid-statics:atmosphere}{altura de escala}.
\end{example}

\section{Forças sobre paredes}

\begin{proposition}[Força sobre uma parede vertical]\label{prop:b1:fluid-statics:wall}
\index{centro de pressão}
Um líquido de profundidade $h$ contra uma parede vertical retangular de
largura $L$ (a pressão atmosférica agindo nas duas faces e se cancelando)
exerce a força horizontal
\[
F = \tfrac12\rho gh^2L ,
\]
equivalente a uma única força aplicada no \emph{\omterm{prop:b1:fluid-statics:wall}{centro de pressão}}, à
profundidade $2h/3$ --- a resultante age embaixo, onde a pressão é
maior.
\end{proposition}

\begin{proof}
À profundidade $x$ a pressão manométrica vale $\rho gx$; sobre a faixa de
altura $\dd x$ a força vale $\rho gxL\,\dd x$; integre de $0$ a $h$. O seu
momento em relação à linha da superfície é $\int_0^h\rho gx^2L\,\dd x = \tfrac13\rho gh^3L$;
dividindo por $F$ obtém-se o \omterm{def:b1:angular-momentum:moment}{braço de alavanca} $2h/3$.
\end{proof}

\end{example}

\section{O teorema de Arquimedes}

\begin{theorem}[Arquimedes]\label{thm:b1:fluid-statics:archimedes}
\index{teorema de Arquimedes}\index{empuxo}\index{centro de empuxo}
Um corpo imerso num fluido em repouso recebe das forças de pressão uma
resultante --- o \emph{empuxo} --- igual e oposta ao peso do fluido que ele
desloca,
\[
\vect\Pi = -\rho_{\text{fluido}}V_{\text{imerso}}\,\vect g ,
\]
aplicada no \omterm{def:b1:systems-of-points:cm}{centro de massa} do fluido deslocado (o \emph{\omterm{thm:b1:fluid-statics:archimedes}{centro de
empuxo}}).
\end{theorem}

\begin{proof}
As forças de pressão sobre a superfície do corpo dependem apenas da forma
dessa superfície e do fluido de fora. Substitua o corpo, em pensamento, por
fluido em repouso ocupando o mesmo volume: esse fluido está em equilíbrio
sob o seu peso e sob as mesmas forças de pressão, de modo que as forças de
pressão equilibram o seu peso --- a soma delas vale $-\rho V\vect g$, aplicada no
seu \omterm{def:b1:systems-of-points:cm}{centro de massa}. Desfaça a experiência de pensamento; as forças de
pressão não mudaram.
\end{proof}

\begin{corollary}[Flutuação]\label{cor:b1:fluid-statics:floating}
\index{flutuação}
Um corpo de densidade $\rho_b$ flutua num fluido de densidade $\rho_f > \rho_b$
com a fração $\rho_b/\rho_f$ do seu volume imersa; se $\rho_b >
\rho_f$ ele afunda, com um peso aparente $(\rho_b - \rho_f)Vg$.
\end{corollary}

\begin{proof}
O peso $\rho_bVg$ é igual ao empuxo $\rho_fV_{\text{im}}g$.
\end{proof}

\begin{example}[Três empuxos]\label{ex:b1:fluid-statics:buoy}
Um densímetro é um tubo lastreado que afunda até deslocar o seu próprio
peso: quanto mais denso o líquido, menos ele afunda --- um medidor de densidade
que se lê na haste. Um balão de \qty{2800}{m^3} de ar quente a \qty{373}{K} desloca
\qty{3.4}{t} de ar frio e pesa \qty{2.6}{t} de ar quente: \qty{0.7}{t}
de sustentação (\cref{pb:b1:kinetic-theory:1}). Um navio de \qty{100000}{t}
desloca \qty{97600}{m^3} de água do mar; em água doce ($\qty{1000}{kg/m^3}$)
ele precisa deslocar $2.5\%$ a mais e afunda mais --- daí as marcas de Plimsoll
no seu casco.
\end{example}

\begin{remark}[Estabilidade]\label{rem:b1:fluid-statics:stability}
\index{metacentro}
Um corpo que flutua, inclinado de um pequeno ângulo, é restaurado se o
empuxo, agora aplicado no centro deslocado do novo volume imerso, produz
um momento de endireitamento --- o que acontece quando o \emph{metacentro} (o
ponto em que a linha de ação do empuxo encontra o eixo do corpo) fica acima
de $G$. Lastro no fundo do casco baixa $G$ e estabiliza o navio; um barco
pesado em cima emborca.
\end{remark}

\section{Problema: o submarino}

\begin{problem}[{Problema de fim de semana --- um casco de aço a trezentos
metros de profundidade: a pressão sobre as suas chapas, a água que ele tem

\label{pb:b1:fluid-statics:1}
Água do mar: $\rho = \qty{1025}{kg/m^3}$; água doce \qty{1000}{kg/m^3};
$P_0 = \qty{1.013e5}{Pa}$; $g = \qty{9.81}{m/s^2}$. O casco do submarino
encerra $V = \qty{3000}{m^3}$; com os tanques de lastro vazios a sua massa é
$m_0 = \qty{2800}{t}$.

\medskip
\noindent\textbf{Parte I --- A pressão sobre o casco.}
\begin{enumerate}
  \item Pressão absoluta a \qty{300}{m}, em \omterm{def:b1:kinetic-theory:pressure}{pascals} e em bars.
  \item Força sobre uma escotilha circular de \qty{0.80}{m} de diâmetro
        nessa profundidade; compare com o peso de um caminhão.
  \item Força sobre um metro quadrado de casco na quilha, se o casco tem
        \qty{10}{m} de altura e o topo está a \qty{300}{m}: quão diferente
        do topo?
  \item A água é ligeiramente compressível ($\chi_T = \qty{4.6e-10}{Pa^{-1}}$):
        de que fração a água do mar é mais densa a \qty{300}{m} do que na
        superfície? Isso importa para a pressão calculada acima?
  \item A tripulação lá dentro respira ar a \qty{1}{atm}: por que o casco
        precisa ser um cilindro espesso, e por que um cilindro resiste
        melhor do que uma caixa de faces planas?
  \item Uma casca cilíndrica fina de raio $R$ e espessura de parede $e$
        sob uma sobrepressão externa $\Delta P$ suporta na parede uma
        tensão de compressão $\Delta P\,R/e$. Para $R = \qty{5.0}{m}$ e
        $e = \qty{3.0}{cm}$ a \qty{300}{m}, calcule-a e compare com o
        limite de escoamento do aço, cerca de \qty{600}{MPa}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Afundar e flutuar.}
\begin{enumerate}[resume]
  \item Na superfície, com os tanques vazios, que volume do casco emerge?
  \item Que massa de água do mar os tanques de lastro precisam admitir para
        o submarino ficar em equilíbrio indiferente?
  \item Uma vez em equilíbrio indiferente a \qty{300}{m}, o casco é
        comprimido de $0.1\%$ pela pressão. Calcule a força resultante e
        diga se o equilíbrio é estável em relação à profundidade.
  \item O barco navega em equilíbrio indiferente do mar para um estuário de
        água doce. Força resultante? Para que lado ele vai, e o que a
        tripulação tem de fazer?
  \item Um torpedo de \qty{20}{t} é disparado. O que o sistema de lastro
        tem de fazer imediatamente, e de quanto?
  \item Por que um submarino usa ar comprimido para esvaziar os seus
        tanques, e por que a pressão desse ar é o limite da profundidade
        em que ele pode esvaziá-los?
  \item Estime o volume de ar a \qty{1}{atm} necessário para esvaziar
        \qty{275}{m^3} de tanques a \qty{300}{m} (lei de Boyle, $PV$
        constante).
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A mergulhadora.}
Uma mergulhadora deixa a superfície com \qty{6.0}{L} de ar nos pulmões.
\begin{enumerate}[resume]
  \item Pressão absoluta a \qty{10}{m} e a \qty{30}{m}.
  \item Se ela desce prendendo a respiração, qual é o volume dos seus
        pulmões a \qty{30}{m}?
  \item Uma mergulhadora com cilindro respira ar à pressão ambiente a
        \qty{30}{m} e sobe prendendo a respiração: volume que os seus
        pulmões precisariam ter na superfície, e a lição.
  \item Um snorkel de \qty{1.0}{m}: diferença de pressão entre a água
        sobre o peito da mergulhadora e o ar dos seus pulmões; força sobre
        um peito de \qty{0.05}{m^2}; ela consegue inspirar?
  \item O cilindro contém \qty{12}{L} a \qty{200}{bar}: quantos litros
        de ar a \qty{30}{m} isso dá, e por quanto tempo a
        \qty{20}{L} por minuto?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O petroleiro lá em cima.}
\begin{enumerate}[resume]
  \item Um petroleiro de \qty{100000}{t} flutua em água do mar: volume de
        água deslocado.
  \item A sua área de flutuação é \qty{8000}{m^2}: de quanto ele sobe ou
        desce ao entrar em água doce?
  \item Carregando \qty{10000}{t} de petróleo: variação do calado.
  \item O petroleiro vazio tem o \omterm{def:b1:systems-of-points:cm}{centro de massa} alto e leva água do mar
        como lastro quando está vazio. Por quê, em termos do metacentro?
  \item O casco do petroleiro na quilha (calado de \qty{20}{m}): pressão e
        força por metro quadrado, comparadas com as do submarino a
        \qty{300}{m}.
  \item O petroleiro passa exatamente por cima do submarino. O submarino
        sente as suas \qty{100000}{t}? Explique com o campo de pressão.
  \item Resuma: a única relação que fixou todos os números deste problema,
        e o teorema que dela decorreu.
\end{enumerate}
\end{problem}
