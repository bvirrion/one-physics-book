\chapter{Máquinas térmicas}\label{ch:b1:heat-engines}

O motor de um carro, uma usina elétrica, uma geladeira e uma \omterm{def:b1:heat-engines:machine}{bomba de calor}
são o mesmo objeto visto de quatro lados: um fluido levado por um ciclo, em
contato alternadamente com uma fonte quente e uma fria, trocando calor com
as duas e trabalho com um eixo. A primeira lei faz a contabilidade da
energia; a segunda, pelo balanço de entropia ao longo de um ciclo, fixa o
limite que nenhuma esperteza vence --- o teorema de Carnot, a desigualdade
de maiores consequências de toda a engenharia. Este capítulo deduz esse
limite, calcula o ciclo ideal e os ciclos reais que dele se aproximam
(Otto, Diesel, Stirling), inverte a máquina para fazer frio e para aquecer
casas, e mostra como a entropia criada por uma máquina real é paga, joule a
joule, em \omterm{thm:b1:heat-engines:lostwork}{trabalho perdido}.

\section{Máquinas cíclicas e as duas leis}

\begin{definition}[Máquina termodinâmica]\label{def:b1:heat-engines:machine}
\index{máquina térmica}\index{refrigerador}\index{bomba de calor}\index{termostato}
Uma \emph{máquina termodinâmica} é um \omterm{def:b1:newton-dynamics:conservation}{sistema fechado} (o fluido de
trabalho) levado por um \emph{ciclo}, de modo que ele volta
periodicamente ao mesmo estado, enquanto troca trabalho $W$ com o exterior
e calor $Q_i$ com fontes às temperaturas $T_i$ --- os \emph{termostatos},
cuja temperatura a troca não altera. Uma máquina \emph{ditérmica} usa dois
termostatos, a fonte quente a $T_h$ e a fonte fria a $T_c < T_h$.
Ela é uma \emph{máquina motriz} se fornece trabalho ($W < 0$), um
\emph{refrigerador} se é acionada ($W > 0$) para tirar calor da fonte
fria, e uma \emph{bomba de calor} se é acionada para ceder calor à fonte
quente. Todas as grandezas são contadas ao longo de um ciclo,
algebricamente, como recebidas pelo fluido.
\end{definition}

\begin{theorem}[As duas leis num ciclo; desigualdade de Clausius]\label{thm:b1:heat-engines:clausius}
\index{desigualdade de Clausius}
Ao longo de um ciclo de uma máquina ditérmica,
\[
W + Q_h + Q_c = 0 , \qquad
\frac{Q_h}{T_h} + \frac{Q_c}{T_c} = -S_{\mathrm{created}} \leq 0 ,
\]
com igualdade na segunda relação se e somente se o ciclo for
reversível. Mais geralmente, para um número qualquer de \omterm{def:b1:heat-engines:machine}{termostatos},
$\sum_i Q_i/T_i \leq 0$ (\emph{desigualdade de Clausius}).
\end{theorem}

\begin{proof}
$U$ e $S$ são funções de estado, logo $\Delta U = 0$ e $\Delta S = 0$ ao longo
de um ciclo. A primeira lei dá $0 = W + Q_h + Q_c$; o balanço de entropia
(\cref{thm:b1:second-law-entropy:secondlaw}) dá $0 = S_{\mathrm{exch}} +
S_{\mathrm{created}}$ com $S_{\mathrm{exch}} = \sum Q_i/T_i$, pois cada
\omterm{def:b1:heat-engines:machine}{termostato} troca à sua própria temperatura fixa
(\cref{prop:b1:second-law-entropy:thermostat}).
\end{proof}

\begin{corollary}[Kelvin de novo]\label{cor:b1:heat-engines:kelvin}
Uma máquina \emph{monotérmica} ($Q_c = 0$) tem $Q_h \leq 0$, donde $W \geq 0$:
nenhum ciclo pode converter em trabalho o calor de uma única fonte. Um
navio não pode navegar esfriando o mar --- uma máquina motriz precisa de uma
fonte fria onde rejeitar calor, e a segunda lei diz quanto ela tem de rejeitar.
\end{corollary}

\section{O teorema de Carnot}

\begin{definition}[Rendimento e coeficientes de desempenho]\label{def:b1:heat-engines:efficiency}
\index{rendimento!de uma máquina térmica}\index{coeficiente de desempenho}
O \emph{rendimento} de uma máquina motriz é o trabalho fornecido por
unidade de calor tomado da fonte quente, $\eta = -W/Q_h$; o
\emph{coeficiente de desempenho} (COP) de um \omterm{def:b1:heat-engines:machine}{refrigerador} é o frio
produzido por unidade de trabalho, $e_f = Q_c/W$; o de uma \omterm{def:b1:heat-engines:machine}{bomba de calor} é
o calor cedido por unidade de trabalho, $e_p = -Q_h/W$. Cada um é a razão
entre ``o que se quer'' e ``o que se paga''; um rendimento vale no máximo~1,
mas um COP pode passar de~1.
\end{definition}

\begin{theorem}[Carnot]\label{thm:b1:heat-engines:carnot}
\index{teorema de Carnot}\index{rendimento de Carnot}
Entre \omterm{def:b1:heat-engines:machine}{termostatos} $T_h > T_c$,
\[
\eta \leq \eta_C = 1 - \frac{T_c}{T_h}, \qquad
e_f \leq \frac{T_c}{T_h - T_c}, \qquad
e_p \leq \frac{T_h}{T_h - T_c} = 1 + \frac{T_c}{T_h - T_c},
\]
com igualdade se e somente se o ciclo for reversível. Os limites dependem
só das temperaturas --- não do fluido, não do mecanismo.
\end{theorem}

\begin{proof}
Para uma máquina motriz ($Q_h > 0$): $\eta = -W/Q_h = (Q_h + Q_c)/Q_h = 1 + Q_c/Q_h$,
e a desigualdade de Clausius dá $Q_c/Q_h \leq -T_c/T_h$. Para um
\omterm{def:b1:heat-engines:machine}{refrigerador} ($Q_c > 0$, $W > 0$): $W = -Q_h - Q_c$ com $-Q_h \geq Q_c T_h/T_c$,
logo $W \geq Q_c(T_h/T_c - 1)$ e $e_f = Q_c/W \leq T_c/(T_h - T_c)$.
Para a \omterm{def:b1:heat-engines:machine}{bomba de calor}, $e_p = -Q_h/W = (W + Q_c)/W = 1 + e_f$. A igualdade é
o caso reversível $S_{\mathrm{created}} = 0$ em toda parte.
\end{proof}

\begin{example}[Três números]\label{ex:b1:heat-engines:numbers}
Uma turbina a vapor entre $T_h = \qty{800}{K}$ e um rio a
$T_c = \qty{300}{K}$: $\eta_C = 0.625$; as usinas reais chegam a uns $0.40$.
Uma geladeira doméstica entre $\qty{275}{K}$ dentro e uma cozinha a
$\qty{298}{K}$: $e_f \leq 275/23 = 12$; as máquinas reais dão $2$ a $4$. Uma
\omterm{def:b1:heat-engines:machine}{bomba de calor} que aquece uma casa a $\qty{293}{K}$ a partir do ar a $\qty{273}{K}$:
$e_p \leq 293/20 = 14.7$; as bombas reais dão $3$ a $5$ --- ainda assim, três a
cinco vezes o que a mesma eletricidade daria numa resistência
(\cref{pb:b1:heat-engines:1}).
\end{example}

\begin{remark}[Por que o rendimento não pode ser~1]\label{rem:b1:heat-engines:why}
Uma máquina motriz recebe a entropia $Q_h/T_h$ com o calor que toma; o
trabalho não leva nenhuma embora; a única saída é o calor rejeitado à fonte
fria, que leva $|Q_c|/T_c$. Como a entropia não pode ser destruída, pelo
menos $Q_h T_c/T_h$ de calor tem de ser jogado fora, e o trabalho é o que
sobra. Quanto mais baixa a fonte fria e mais alta a quente, melhor --- daí
as torres de resfriamento e a busca de pás de turbina cada vez mais quentes.
\end{remark}

\section{O ciclo de Carnot}

\begin{definition}[Ciclo de Carnot]\label{def:b1:heat-engines:carnotcycle}
\index{ciclo de Carnot}
O \emph{ciclo de Carnot} é o ciclo ditérmico reversível: duas isotermas
reversíveis, a $T_h$ (em contato com a fonte quente) e a $T_c$ (com a
fria), ligadas por dois trechos adiabáticos reversíveis (isentrópicos,
sem fonte em contato). Percorrido no sentido horário no plano $(V, P)$ ele é
uma máquina motriz; no sentido anti-horário, um \omterm{def:b1:heat-engines:machine}{refrigerador} ou uma \omterm{def:b1:heat-engines:machine}{bomba
de calor}. No \omterm{def:b1:second-law-entropy:TS}{diagrama entrópico} ele é um retângulo.
\end{definition}

\begin{proposition}[Ciclo de Carnot de um gás perfeito]\label{prop:b1:heat-engines:carnotgas}
Para $n$ \omterm{def:b1:kinetic-theory:scales}{mols} de \omterm{def:b1:kinetic-theory:model}{gás perfeito} levados por $A \to B$ (isoterma $T_h$,
expansão), $B \to C$ (adiabática), $C \to D$ (isoterma $T_c$,
compressão), $D \to A$ (adiabática):
\[
Q_h = nRT_h\ln\frac{V_B}{V_A}, \qquad
Q_c = -nRT_c\ln\frac{V_B}{V_A}, \qquad
\eta = 1 - \frac{T_c}{T_h} ,
\]
o valor de Carnot --- como tem de ser para qualquer ciclo ditérmico reversível.
\end{proposition}

\begin{proof}
Numa isoterma $\Delta U = 0$, logo $Q = -W = nRT\ln(V_{\text{final}}/
V_{\text{inicial}})$. Nas adiabáticas $TV^{\gamma-1}$ é constante:
$T_hV_B^{\gamma-1} = T_cV_C^{\gamma-1}$ e $T_hV_A^{\gamma-1} =
T_cV_D^{\gamma-1}$, donde $V_C/V_D = V_B/V_A$ e $Q_c = nRT_c\ln(V_D/V_C) =
-nRT_c\ln(V_B/V_A)$. Então $\eta = 1 + Q_c/Q_h = 1 - T_c/T_h$.
\end{proof}

\begin{remark}[O ciclo de Carnot é um teorema, não um motor]\label{rem:b1:heat-engines:notengine}
As isotermas reversíveis exigem uma transferência de calor infinitamente
lenta através de uma diferença de temperatura que tende a zero: uma máquina
de Carnot fornece o seu trabalho máximo com potência nula. As máquinas
reais trocam \omterm{def:b1:heat-engines:efficiency}{rendimento} por potência aceitando diferenças finitas de
temperatura nos seus trocadores (\cref{exo:b1:heat-engines:10}); o valor de
Carnot continua sendo o teto com que todo projeto é comparado.
\end{remark}

\section{Ciclos de máquinas reais}

\begin{proposition}[Ciclo Otto]\label{prop:b1:heat-engines:otto}
\index{ciclo Otto}\index{taxa de compressão}
O motor a gasolina idealizado leva o ar ($\gamma$) por: $1 \to 2$
compressão adiabática de $V_1$ a $V_2$ (\emph{\omterm{prop:b1:heat-engines:otto}{taxa de compressão}}
$r = V_1/V_2$); $2 \to 3$ aquecimento isocórico (a combustão); $3 \to 4$
expansão adiabática de volta a $V_1$; $4 \to 1$ resfriamento isocórico (o
escapamento). O seu \omterm{def:b1:heat-engines:efficiency}{rendimento} vale
\[
\eta_{\text{Otto}} = 1 - \frac{1}{r^{\gamma-1}} .
\]
\end{proposition}

\begin{proof}
O calor só é trocado nas isocóricas: $Q_h = nC_{V,m}(T_3 - T_2)$ e
$Q_c = nC_{V,m}(T_1 - T_4)$. Nas adiabáticas $TV^{\gamma-1}$ é constante:
$T_2 = T_1r^{\gamma-1}$ e $T_3 = T_4r^{\gamma-1}$, logo $T_3 - T_2 =
(T_4 - T_1)r^{\gamma-1}$ e
$\eta = 1 + Q_c/Q_h = 1 - (T_4 - T_1)/(T_3 - T_2) = 1 - r^{1-\gamma}$.
\end{proof}

\begin{example}[Números para um motor a gasolina]\label{ex:b1:heat-engines:ottonumbers}
$r = 9$, $\gamma = 1.4$: $9^{0.4} = 2.41$ e $\eta = 0.58$. Um motor real dá
uns $0.30$: a combustão não é instantânea, o gás não é perfeito a
\qty{2500}{K}, o calor foge para as paredes, e o atrito e o bombeamento
cobram a sua parte. Aumentar $r$ ajuda --- até que a mistura ar--combustível
se acenda sozinha no fim da compressão (a detonação); o motor diesel
transforma esse vício no seu princípio.
\end{example}

\begin{proposition}[Ciclo Diesel]\label{prop:b1:heat-engines:diesel}
\index{ciclo Diesel}
Substitua o aquecimento isocórico do \omterm{prop:b1:heat-engines:otto}{ciclo Otto} por um isobárico, de
$V_2$ a $V_3 = \rho V_2$ (\emph{razão de corte} $\rho > 1$): o combustível
injetado no ar já quente por uma \omterm{prop:b1:heat-engines:otto}{taxa de compressão} $r$ de $15$ a $22$ queima
à medida que chega. Então
\[
\eta_{\text{Diesel}} = 1 - \frac{1}{r^{\gamma-1}}\,\frac{\rho^{\gamma} - 1}{\gamma(\rho - 1)} ,
\]
menor que o valor de Otto para o mesmo $r$, mas atingido com $r$ muito
maiores, e por isso maior na prática ($0.40$ a $0.45$ nos motores grandes).
\end{proposition}

\begin{proof}
$Q_h = nC_{P,m}(T_3 - T_2)$, $Q_c = nC_{V,m}(T_1 - T_4)$, logo $\eta = 1 -
(T_4 - T_1)/[\gamma(T_3 - T_2)]$. Com $T_2 = T_1r^{\gamma-1}$, $T_3 =
\rho T_2$ e, na adiabática $3 \to 4$ ($V_4 = V_1 = rV_2$), $T_4 =
T_3(V_3/V_4)^{\gamma-1} = \rho T_1 r^{\gamma-1}(\rho/r)^{\gamma-1} = T_1\rho^{\gamma}$:
$\eta = 1 - (\rho^\gamma - 1)/[\gamma r^{\gamma-1}(\rho - 1)]$.
\end{proof}

compressão}; as linhas tracejadas marcam um motor a gasolina ($r = 9$) e um
diesel ($r = 18$, $\rho = 2$).}
\end{omfigure}

\begin{proposition}[Ciclo Stirling e o regenerador]\label{prop:b1:heat-engines:stirling}
\index{ciclo Stirling}\index{regenerador}
Duas isotermas ($T_c$, $T_h$) ligadas por duas isocóricas. O calor dos
trechos isocóricos, $\pm nC_{V,m}(T_h - T_c)$, é igual e de sinais opostos; se ele
for armazenado no trecho de resfriamento num \emph{regenerador} (uma massa
porosa através da qual o gás escoa) e devolvido no trecho de aquecimento, só
as isotermas trocam com as fontes, e $\eta = 1 - T_c/T_h$: o motor Stirling
é um motor de Carnot em tudo menos na forma do seu ciclo. Sem o
regenerador o calor das isocóricas tem de vir da fonte quente e
$\eta = \dfrac{nR(T_h - T_c)\ln(V_1/V_2)}{nRT_h\ln(V_1/V_2) + nC_{V,m}(T_h - T_c)}$,
bem menor (\cref{exo:b1:heat-engines:8}).
\end{proposition}

\begin{proof}
Nas isotermas $Q_h = nRT_h\ln(V_1/V_2)$ e $Q_c = -nRT_c\ln(V_1/V_2)$, logo
$-W = Q_h + Q_c + 0$ assim que os calores isocóricos se cancelam; divida pelo calor
efetivamente tirado da fonte quente em cada caso.
\end{proof}

\begin{remark}[Turbinas a vapor e a gás]\label{rem:b1:heat-engines:turbines}
As usinas elétricas não usam um gás num cilindro, mas água fervida,
expandida através de uma turbina, condensada e bombeada de volta (o ciclo
Rankine, com duas mudanças de fase --- \cref{ch:b1:phase-changes}), ou ar
comprimido, aquecido pela combustão e expandido numa turbina a gás (o ciclo
Brayton, duas adiabáticas e duas isobáricas). A sua análise exige o balanço
de energia de um fluido em escoamento, dado no volume do segundo ano; o
limite de Carnot e a lógica deste capítulo se aplicam sem mudança.
\end{remark}

\section{Refrigeradores e bombas de calor}

\begin{definition}[Ciclo de compressão de vapor]\label{def:b1:heat-engines:vapourcycle}
\index{ciclo de compressão de vapor}\index{fluido refrigerante}
Quase toda geladeira, congelador, ar-condicionado e \omterm{def:b1:heat-engines:machine}{bomba de calor} faz um
\emph{fluido refrigerante} circular por quatro componentes: um \emph{compressor}
(recebe o trabalho e eleva a pressão do vapor), um
\emph{condensador} (o vapor quente se liquefaz a alta pressão, liberando
$|Q_h|$ para o lado quente), uma \emph{válvula de expansão} (o líquido cai à
baixa pressão sem trabalho nem calor, e em parte vira vapor) e um
\emph{evaporador} (o resto ferve a baixa pressão, absorvendo $Q_c$ do
lado frio). O calor é transportado como calor latente de vaporização, e
as duas pressões fixam as duas temperaturas de ebulição
(\cref{ch:b1:phase-changes}).
\end{definition}

\begin{proposition}[O preço das diferenças de temperatura]\label{prop:b1:heat-engines:gaps}
Uma máquina reversível que trabalha entre as suas próprias temperaturas
internas $T_{\mathrm{ev}} < T_c$ e $T_{\mathrm{cd}} > T_h$ tem $e_p = T_{\mathrm{cd}}/(T_{\mathrm{cd}} - T_{\mathrm{ev}})$,
menor que o valor de Carnot entre as fontes; uma máquina real atinge
uma fração (tipicamente $0.4$ a $0.6$) até mesmo desse valor. O
desempenho de uma \omterm{def:b1:heat-engines:machine}{bomba de calor} cai, portanto, quando a temperatura
externa baixa e quando a temperatura exigida pelo circuito de aquecimento sobe.
\end{proposition}

\begin{proof}
Aplique o \cref{thm:b1:heat-engines:carnot} ao fluido, cujos \omterm{def:b1:heat-engines:machine}{termostatos}
são, de fato, as temperaturas dos trocadores; $T_{\mathrm{cd}} - T_{\mathrm{ev}} >
T_h - T_c$ baixa a razão.
\end{proof}

\begin{example}[Bomba de calor ou resistência?]\label{ex:b1:heat-engines:hpvsresistor}
Uma casa que perde \qty{6}{kW} com \qty{-5}{\degreeCelsius} lá fora precisa de
\qty{6}{kW} de eletricidade com resistências; uma \omterm{def:b1:heat-engines:machine}{bomba de calor} ideal de
\qty{268}{K} a \qty{293}{K} precisaria de $6/11.7 = \qty{0.51}{kW}$; uma real
com temperaturas de trocador de \qty{260}{K} e \qty{313}{K} e um \omterm{def:b1:transient-regimes:secondorder}{fator de

O problema de fim de semana percorre a estação inteira, entropia incluída.
\end{example}

\section{Entropia e trabalho perdido}

\begin{theorem}[Trabalho perdido]\label{thm:b1:heat-engines:lostwork}
\index{trabalho perdido}
Uma máquina motriz ditérmica que cria a entropia $S_{\mathrm{created}}$ por ciclo
fornece
\[
-W = Q_h\Bigl(1 - \frac{T_c}{T_h}\Bigr) - T_c\,S_{\mathrm{created}} ,
\]
e uma \omterm{def:b1:heat-engines:machine}{bomba de calor} ditérmica que fornece $|Q_h|$ consome
$W = |Q_h|(1 - T_c/T_h) + T_c\,S_{\mathrm{created}}$: cada joule por \omterm{def:b1:kinetic-theory:temperature}{kelvin}
criado custa $T_c$ joules de trabalho, sendo a temperatura da fonte fria
a taxa de câmbio entre entropia e trabalho.
\end{theorem}

\begin{proof}
Pelo \cref{thm:b1:heat-engines:clausius}, $Q_c = -T_c(Q_h/T_h + S_{\mathrm{created}})$;
substitua em $-W = Q_h + Q_c$. Para a bomba, escreva as mesmas duas relações
com $Q_h < 0$.
\end{proof}

\begin{method}[Auditar uma máquina]\label{meth:b1:heat-engines:audit}
Meça (ou calcule) os calores trocados com cada fonte e o trabalho;
verifique a primeira lei; calcule $S_{\mathrm{created}} = -\sum Q_i/T_i$ por ciclo
ou por segundo; a potência perdida vale $T_c\dot S_{\mathrm{created}}$; localize a
criação (diferenças finitas de temperatura nos trocadores, atrito,
estrangulamento, compressão não quase-estática) aplicando o balanço de
entropia a cada componente, um a um. Um componente que não cria entropia
não vale a pena melhorar.
\end{method}

\begin{example}[A auditoria de uma usina]\label{ex:b1:heat-engines:audit}
Fonte quente \qty{800}{K}, rio \qty{300}{K}, $\qty{1.0}{GW}$ de eletricidade
a $\eta = 0.40$: $\dot Q_h = \qty{2.5}{GW}$, $\dot Q_c = \qty{1.5}{GW}$, logo
$\dot S_{\mathrm{created}} = 1.5/300 - 2.5/800 = \qty{1.9}{MW/K}$ e a potência
perdida vale $300 \times 1.9 = \qty{0.56}{GW}$: exatamente o intervalo entre a
produção de Carnot $0.625 \times 2.5 = \qty{1.56}{GW}$ e o \qty{1.0}{GW} real.
O rio esquenta com os \qty{1.5}{GW} que tem de levar embora; no verão, quando
ele está quente e baixo, as usinas reduzem a carga.
\end{example}

\begin{exercise}[$\star\star\star$]\label{exo:b1:heat-engines:10}

\section{Problema: a bomba de calor da casa}

\begin{problem}[{Problema de fim de semana --- aquecer uma casa durante um
inverno: a conta com resistências, com gás, com uma bomba de calor ideal e
com uma real, e para onde vai a eletricidade da real}]
\label{pb:b1:heat-engines:1}
Uma casa perde calor à taxa $K(T_{\mathrm{in}} - T_{\mathrm{out}})$ com
$K = \qty{250}{W/K}$; ela é mantida a $T_{\mathrm{in}} = \qty{20}{\degreeCelsius}$.
Temperatura externa de projeto: $\qty{-5}{\degreeCelsius}$; estação de
aquecimento: $200$ dias com temperatura externa média de \qty{7}{\degreeCelsius}.
Preços: eletricidade 0.25~\texteuro{} por \unit{kW h}, gás 0.10~\texteuro{} por
\unit{kW h}; $\qty{1}{kW h} = \qty{3.6}{MJ}$.

\medskip
\noindent\textbf{Parte I --- A casa e os meios antigos.}
\begin{enumerate}
  \item Potência de aquecimento necessária na temperatura de projeto.
  \item Energia necessária ao longo da estação, em joules e em \unit{kW h}.
  \item Custo da estação com resistências elétricas.
  \item Custo com uma caldeira a gás de \omterm{def:b1:heat-engines:efficiency}{rendimento} $0.90$.
  \item A eletricidade vem de uma usina térmica de \omterm{def:b1:heat-engines:efficiency}{rendimento}
        $0.40$. Energia de combustível consumida, por estação, pelas
        resistências e pela caldeira; comente a frase ``o aquecimento
        elétrico é um desperdício termodinâmico''.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A \omterm{def:b1:heat-engines:machine}{bomba de calor} ideal.}
\begin{enumerate}[resume]
  \item Defina $e_p$ e demonstre, a partir das duas leis num ciclo, que
        $e_p \leq T_{\mathrm{in}}/(T_{\mathrm{in}} - T_{\mathrm{out}})$.
  \item $e_p$ ideal na temperatura de projeto e a \omterm{thm:b1:dc-circuits:kirchhoff}{potência elétrica} então
        necessária.
  \item $e_p$ ideal na temperatura média da estação e (tomando esse valor
        para toda a estação) a eletricidade sazonal.
  \item Explique fisicamente por que $e_p$ cai quando a temperatura externa
        cai, e por que esse é o pior comportamento possível para um
        aparelho de aquecimento.
  \item No ponto de projeto a bomba ideal consome \qty{0.53}{kW} e
        fornece \qty{6.25}{kW}: de onde vêm os outros \qty{5.7}{kW}, e
        por que esfriar o ar externo frio para aquecer uma casa quente
        não viola a segunda lei?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A máquina real.}
A bomba funciona num \omterm{def:b1:heat-engines:vapourcycle}{ciclo de compressão de vapor}. O \omterm{def:b1:heat-engines:vapourcycle}{fluido refrigerante}
evapora a $T_{\mathrm{ev}} = T_{\mathrm{out}} - \qty{8}{K}$ e condensa a
$T_{\mathrm{cd}} = \qty{313}{K}$ (a água de um circuito de piso aquecido
circula a \qty{35}{\degreeCelsius}); a máquina atinge $0.55$ do valor de
Carnot calculado entre $T_{\mathrm{ev}}$ e $T_{\mathrm{cd}}$. Calor latente do
\omterm{def:b1:heat-engines:vapourcycle}{fluido refrigerante}: $L = \qty{200}{kJ/kg}$.
\begin{enumerate}[resume]
  \item Nomeie os quatro componentes do ciclo, diga em qual deles o fluido
        recebe $Q_c$, cede $|Q_h|$ e recebe $W$, e por que o
        evaporador tem de ser mais frio que o ar externo e o
        condensador mais quente que a casa.
  \item Na temperatura de projeto: $T_{\mathrm{ev}}$ e o $e_p$ de Carnot
        entre $T_{\mathrm{ev}}$ e $T_{\mathrm{cd}}$.
  \item $e_p$ real, \omterm{thm:b1:dc-circuits:kirchhoff}{potência elétrica} consumida e calor tirado do
        ar externo, na temperatura de projeto.
  \item Vazão mássica de \omterm{def:b1:heat-engines:vapourcycle}{fluido refrigerante} através do evaporador.
  \item As mesmas perguntas ($e_p$ real, \omterm{thm:b1:dc-circuits:kirchhoff}{potência elétrica}) na
        temperatura média de \qty{7}{\degreeCelsius}, em que a casa perde
        \qty{3.25}{kW}.
  \item Dê $e_p$ em função de $T_{\mathrm{out}}$; eletricidade e custo
        sazonais, tomando para a estação o $e_p$ da temperatura média.
        Compare com a Parte~I.
  \item Energia de combustível consumida na usina para a estação com \omterm{def:b1:heat-engines:machine}{bomba
        de calor}; compare com as resistências e com a caldeira.
  \item Uma onda de frio a \qty{-15}{\degreeCelsius}: demanda de calor, $e_p$
        real, \omterm{thm:b1:dc-circuits:kirchhoff}{potência elétrica} se a bomba pudesse fornecê-la. A capacidade

  \item Se a casa tivesse radiadores antigos que exigem água a
        \qty{60}{\degreeCelsius} ($T_{\mathrm{cd}} = \qty{338}{K}$): $e_p$ real

        escolha dos emissores.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Para onde vai a eletricidade.}
\begin{enumerate}[resume]
  \item Mostre que, para uma \omterm{def:b1:heat-engines:machine}{bomba de calor} que fornece $|Q_h|$ à casa a
        $T_{\mathrm{in}}$ a partir do ar externo a $T_{\mathrm{out}}$,
        $W = |Q_h|(1 - T_{\mathrm{out}}/T_{\mathrm{in}}) + T_{\mathrm{out}}S_{\mathrm{created}}$.
  \item No ponto de projeto, entropia criada por segundo pela
        transferência de calor no condensador (de \qty{313}{K} à casa) e
        no evaporador (do ar externo a \qty{260}{K}).
  \item \omterm{thm:b1:dc-circuits:kirchhoff}{Potência elétrica} que essas duas diferenças de temperatura custam;
        fração dos \qty{1.92}{kW} consumidos.
  \item Entropia total criada por segundo pela máquina real (compare
        os seus \qty{1.92}{kW} com os \qty{0.53}{kW} ideais); parcela dos
        trocadores; cite as outras fontes.
  \item Reduza à metade as duas diferenças (\qty{4}{K} e \qty{10}{K}): entropia criada
        nos trocadores e \omterm{thm:b1:dc-circuits:kirchhoff}{potência elétrica} economizada; o que custa ao
        projetista reduzir uma diferença à metade?
  \item Resuma em três números: quilowatt-horas de calor por
        quilowatt-hora de eletricidade ao longo da estação; custo sazonal
        contra o do gás; e, com a eletricidade a \qty{60}{g} de CO$_2$ por
        \unit{kW h} e o gás a \qty{200}{g}, o carbono emitido por cada
        via.
\end{enumerate}
\end{problem}
