\chapter{Geometric Optics: Rays, Reflection, Refraction}\label{ch:b1:rays-reflection-refraction}

A straw standing in a glass of water looks broken at the surface; a
swimmer looking up from the bottom of a pool sees the whole sky squeezed
into a bright disk, ringed by a mirror of the pool floor; a hair-thin
glass fiber carries a telephone call across an ocean with less loss than
a copper wire suffers across a street. Three laws --- light travels
straight, reflects at equal angles, bends by Snell's rule --- account for
all of it, and this chapter states them precisely, marks where they
stop being true, and puts them to work.

\section{The geometric-optics model}

\begin{definition}[Light ray; homogeneous medium]\label{def:b1:rays-reflection-refraction:ray}
\index{light ray}\index{geometric optics}\index{homogeneous medium}
In the model of \emph{geometric optics}, light propagates along
\emph{light rays}: oriented curves that are straight lines in a
\emph{homogeneous} medium (same properties at every point) and that
carry energy independently of one another. A beam is a bundle of rays; a
point source emits rays in every direction.
\end{definition}

\begin{definition}[Refractive index]\label{def:b1:rays-reflection-refraction:index}
\index{refractive index}\index{transparent medium}
A \emph{transparent} medium lets light through; light travels in it at a
speed $v < c$. Its \emph{refractive index} is
\[
n = \frac{c}{v} \geq 1 .
\]
Vacuum: $n = 1$; air: $1.0003$; water: $1.33$; ordinary glass: $1.5$
to $1.6$; diamond: $2.42$. A wave of frequency $\nu$ keeps its frequency
when it enters a medium, so its wavelength shrinks from $\lambda_0 =
c/\nu$ in vacuum to $\lambda = \lambda_0/n$.
\end{definition}

\begin{remark}[Dispersion]\label{rem:b1:rays-reflection-refraction:dispersion}
\index{dispersion (optical)}\index{Cauchy's law}
In matter the index depends slightly on the wavelength --- the medium is
\emph{dispersive}. For glass in the visible, Cauchy's empirical law
$n(\lambda) = A + B/\lambda^2$ fits well: $n$ is larger for blue than for
red by about $0.01$ to $0.02$. The prism and the rainbow, at the end of
this chapter, live on that difference.
\end{remark}

\begin{remark}[Limits of the model]\label{rem:b1:rays-reflection-refraction:limits}
Light is a wave, and a wave passing through an opening of width $a$
spreads by an angle of order $\lambda/a$ (diffraction,
\cref{ch:b1:wave-propagation}). Rays are a good description as long as
every aperture, lens and obstacle is much larger than the wavelength
(\qty{0.4}{} to \qty{0.8}{\micro m} for visible light): a \qty{1}{mm}
hole spreads a beam by $10^{-3}$ radians, negligible on a bench; a
\qty{1}{\micro m} hole spreads it over the whole half-space, and no ray
picture survives. The \qty{9}{\micro m} \omterm{def:b1:rays-reflection-refraction:fiber}{core} of a single-mode telecom
fiber is already beyond the model.
\end{remark}

\begin{proposition}[Return of light]\label{prop:b1:rays-reflection-refraction:reversibility}
\index{principle of inverse return}
The path of a ray does not depend on the direction in which the light
travels it: if light goes from $A$ to $B$ along a path, light from $B$
follows the same path to $A$.
\end{proposition}
\admitted

\section{Reflection and refraction}

\begin{definition}[Incidence]\label{def:b1:rays-reflection-refraction:incidence}
\index{plane of incidence}\index{angle of incidence}\index{dioptric surface}
A ray meeting the surface separating two media (a \emph{diopter}) at a
point $I$ makes with the \emph{normal} to the surface at $I$ the
\emph{angle of incidence} $i_1$. The \emph{plane of incidence} contains
the incident ray and the normal.
\end{definition}

\begin{theorem}[Laws of reflection and refraction (Snell--Descartes)]\label{thm:b1:rays-reflection-refraction:snell}
\index{reflection!law of}\index{refraction!law of}\index{Snell's law}
At the surface between media of indices $n_1$ (incident side) and
$n_2$:
\begin{enumerate}
  \item the reflected and refracted rays lie in the \omterm{def:b1:rays-reflection-refraction:incidence}{plane of incidence};
  \item the reflected ray makes with the normal the angle $i_1' = i_1$,
        on the other side of the normal;
  \item the refracted ray makes with the normal an angle $i_2$ such that
        \[
        n_1 \sin i_1 = n_2 \sin i_2 .
        \]
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[Where the laws come from]\label{rem:b1:rays-reflection-refraction:fermat}
\index{Fermat's principle}
Both laws follow from \emph{Fermat's principle}: between two points,
light follows the path of least travel time. Reflection at equal angles
is the shortest broken path touching the mirror; \omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law} is the
path that trades a longer route in the fast medium against a shorter
one in the slow medium (\cref{exo:b1:rays-reflection-refraction:12}).
In the Year 2 volume both laws are derived from the wave nature of light
at an interface.
\end{remark}

\begin{proposition}[Which way the ray bends]\label{prop:b1:rays-reflection-refraction:bending}
Entering a more refractive medium ($n_2 > n_1$), a ray bends
\emph{toward} the normal ($i_2 < i_1$); entering a less refractive one,
\emph{away} from it ($i_2 > i_1$). A ray along the normal goes through
undeviated.
\end{proposition}

\begin{proof}
$\sin i_2 = (n_1/n_2)\sin i_1$ with $\sin$ increasing on
$[0, \pi/2]$.
\end{proof}

\begin{example}[Apparent depth]\label{ex:b1:rays-reflection-refraction:depth}
\index{apparent depth}
A coin lies at depth $h$ under water ($n = 1.33$), seen from almost
straight above. A ray from the coin reaching the surface at small angle
$i_2$ (in water) leaves at $i_1$ with $\sin i_1 = n\sin i_2$; for small
angles $i_1 \approx n\,i_2$. The eye extends the emerging ray back: it
seems to come from a depth $h'$ with $\tan i_1 \approx x/h'$ and
$\tan i_2 \approx x/h$ for the same surface point at distance $x$ from
the vertical, so $h' = h\,i_2/i_1 = h/n$. A pool \qty{2.0}{m} deep looks
\qty{1.5}{m} deep; a fish sees the fisherman $1.33$ times higher than
he is.
\end{example}

\begin{theorem}[Total internal reflection]\label{thm:b1:rays-reflection-refraction:tir}
\index{total internal reflection}\index{critical angle}
Light going from a medium of index $n_1$ toward a less refractive one
($n_2 < n_1$) is refracted only if $i_1 \leq i_c$, where the
\emph{\omterm{thm:b1:rays-reflection-refraction:tir}{critical angle}} satisfies
\[
\sin i_c = \frac{n_2}{n_1} .
\]
For $i_1 > i_c$ there is no refracted ray: the surface reflects all the
light --- \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{total internal reflection}}.
\end{theorem}

\begin{proof}
\omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law} asks for $\sin i_2 = (n_1/n_2)\sin i_1 > \sin i_1$; this
exceeds $1$ as soon as $\sin i_1 > n_2/n_1$, and then no angle $i_2$
exists. At $i_1 = i_c$ the refracted ray grazes the surface
($i_2 = 90^\circ$). That \emph{all} the energy is then reflected is a
statement about intensities, which \omterm{def:b1:rays-reflection-refraction:ray}{geometric optics} does not handle; the
wave treatment of the Year 2 volume confirms it.
\end{proof}

\begin{example}[Diamonds and swimmers]\label{ex:b1:rays-reflection-refraction:tir}
Glass--air: $\sin i_c = 1/1.5$, $i_c = 41.8^\circ$, which is why a
$45^\circ$ prism reflects totally and replaces a mirror in binoculars.
Diamond--air: $i_c = 24.4^\circ$; a cut diamond traps light that enters
its top and sends almost all of it back out through the top --- the
brilliance is total reflection, the fire is dispersion. Water--air:
$i_c = 48.8^\circ$; looking up, a diver sees the entire sky inside a
cone of half-angle $48.8^\circ$ (``Snell's window'') and the reflected
pool floor outside it.
\end{example}

\section{The optical fiber}

\begin{definition}[Step-index fiber]\label{def:b1:rays-reflection-refraction:fiber}
\index{optical fiber}\index{core}\index{cladding}\index{numerical aperture}
A \emph{step-index optical fiber} is a cylindrical glass \emph{core} of
index $n_1$ sheathed in a \emph{cladding} of slightly lower index $n_2$.
Light entering the core within a cone of half-angle $\theta_a$ around
the axis is guided by successive \omterm{thm:b1:rays-reflection-refraction:tir}{total internal reflections} at the
core--cladding surface. The \emph{numerical aperture} is
$\mathrm{NA} = \sin\theta_a$.
\end{definition}

\begin{proposition}[Acceptance cone and intermodal spread]\label{prop:b1:rays-reflection-refraction:fiber}
\index{acceptance angle}\index{intermodal dispersion}
For a fiber whose entry face is in air,
\[
\sin\theta_a = \sqrt{n_1^2 - n_2^2} .
\]
Over a length $L$, the guided ray along the axis and the steepest
guided ray arrive with a delay
\[
\Delta t = \frac{L\,n_1}{c}\left(\frac{n_1}{n_2} - 1\right)
\]
between them (\emph{\omterm{prop:b1:rays-reflection-refraction:fiber}{intermodal dispersion}}).
\end{proposition}

\begin{proof}
A ray entering at angle $\theta$ to the axis is refracted to $\theta'$
with $\sin\theta = n_1\sin\theta'$ and meets the \omterm{def:b1:rays-reflection-refraction:fiber}{cladding} at incidence
$i = 90^\circ - \theta'$. Guiding requires $\sin i \geq n_2/n_1$, i.e.\
$\cos\theta' \geq n_2/n_1$, i.e.\ $\sin\theta' \leq
\sqrt{1 - n_2^2/n_1^2}$; multiply by $n_1$. The axial ray travels $L$ at
speed $c/n_1$: $t_0 = Ln_1/c$. The steepest ray zigzags at angle
$\theta'_{\max}$ with $\cos\theta'_{\max} = n_2/n_1$; its path is
$L/\cos\theta'_{\max} = L n_1/n_2$, so $t_1 = L n_1^2/(n_2 c)$, and
$\Delta t = t_1 - t_0$.
\end{proof}

\begin{example}[A telecom fiber's numbers]\label{ex:b1:rays-reflection-refraction:fibernum}
$n_1 = 1.480$, $n_2 = 1.465$: $\mathrm{NA} = \sqrt{2.190 - 2.146} =
0.21$, $\theta_a = 12^\circ$; $i_c = 81.8^\circ$, so guided rays stay
within $8.2^\circ$ of the axis. Intermodal spread: $\Delta t/L =
(1.48/c)(1.48/1.465 - 1) = \qty{51}{ns/km}$. Over \qty{2}{km} a pulse
spreads by \qty{0.1}{\micro s}, which caps the bit rate near
\qty{5}{Mbit/s}: the weekend problem follows this limitation to its
remedy, the single-mode fiber.
\end{example}

\section{The plane mirror}

\begin{proposition}[Image in a plane mirror]\label{prop:b1:rays-reflection-refraction:mirror}
\index{plane mirror}\index{virtual image}\index{stigmatism}
All rays from a point $A$ reflected by a \omterm{prop:b1:rays-reflection-refraction:mirror}{plane mirror} seem to come from
the point $A'$ symmetric to $A$ with respect to the mirror's plane: $A'$
is the \emph{image} of $A$. It is \emph{virtual} (no light passes through
it) and the mirror is \emph{rigorously stigmatic}: every point has an
exact point image.
\end{proposition}

\begin{proof}
A ray from $A$ hitting the mirror at $I$ with incidence $i$ leaves at
$i$ on the other side of the normal. The reflected line and the normal
at $I$ make the angle $i$, as does the line $IA$; by symmetry with
respect to the mirror plane, the reflected line is the mirror image of
line $IA$, hence passes through the symmetric point $A'$ --- for every
$I$.
\end{proof}

\begin{proposition}[Rotating a mirror]\label{prop:b1:rays-reflection-refraction:rotation}
Turning a \omterm{prop:b1:rays-reflection-refraction:mirror}{plane mirror} by an angle $\alpha$ about an axis in its plane,
perpendicular to the \omterm{def:b1:rays-reflection-refraction:incidence}{plane of incidence}, turns the reflected ray by
$2\alpha$.
\end{proposition}

\begin{proof}
The normal turns by $\alpha$, so the incidence changes from $i$ to
$i + \alpha$; the reflected ray, at $i + \alpha$ on the other side of
the new normal, has moved by $\alpha$ (normal) $+ \alpha$ (angle) $=
2\alpha$ from its old direction.
\end{proof}

\begin{example}[The galvanometer's lever]\label{ex:b1:rays-reflection-refraction:galvanometer}
A tiny mirror glued to the coil of a galvanometer, lit by a lamp and
throwing a spot on a scale \qty{2}{m} away: a rotation of
$\qty{1}{mrad}$ of the coil moves the spot by $2 \times 10^{-3} \times
\qty{2}{m} = \qty{4}{mm}$ --- an optical lever that magnifies without
friction, the principle of the mirror galvanometer and of the laser
scanner.
\end{example}

\section{The prism}

\begin{proposition}[Prism formulas]\label{prop:b1:rays-reflection-refraction:prism}
\index{prism}\index{deviation}\index{minimum deviation}
A ray crossing a prism of apex angle $A$ and index $n$ in air, in the
plane perpendicular to the edge, with angles $i$ (entry), $r$, $r'$
(inside) and $i'$ (exit) measured from the normals, obeys
\[
\sin i = n \sin r, \qquad r + r' = A, \qquad \sin i' = n \sin r',
\]
and is deviated by $D = i + i' - A$. The deviation is minimal when the
path is symmetric, $i = i'$, $r = r' = A/2$; then
\[
n = \frac{\sin\frac{A + D_{\min}}{2}}{\sin\frac{A}{2}} .
\]
\end{proposition}

\begin{proof}
\omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law} at each face; $r + r' = A$ because the two normals make the
angle $A$ (the quadrilateral formed by the apex, the two incidence
points and the intersection of the normals). The total deviation is
$(i - r) + (i' - r') = i + i' - A$. By the return of light, the
deviation is the same function of $i$ and of $i'$; if the minimum
occurred at $i_0 \neq i_0'$ it would also occur at the reversed
incidence $i_0'$, giving two minima, which the monotonic behavior of
$D(i)$ on each side of its minimum forbids: the minimum is at $i = i'$.
Then $r = r' = A/2$ and $D_{\min} = 2i - A$; substituting $i = (A +
D_{\min})/2$ into $\sin i = n\sin(A/2)$ gives the formula.
\end{proof}

\begin{example}[Measuring an index]\label{ex:b1:rays-reflection-refraction:prismnum}
A $60^\circ$ prism turned until the deviation of a yellow beam is
smallest: $D_{\min} = 38.9^\circ$. Then $n = \sin 49.45^\circ/\sin 30^\circ
= 0.760/0.500 = 1.52$. \omterm{prop:b1:rays-reflection-refraction:prism}{Minimum deviation} is the standard way to measure
the index of a glass --- the sharp minimum, found by turning the prism
back and forth, is insensitive to small misadjustments.
\end{example}

\begin{remark}[Thin prisms and spectroscopes]\label{rem:b1:rays-reflection-refraction:thin}
For small $A$ and small incidence, the sines equal the angles and
$D \approx (n - 1)A$, independent of $i$: a thin prism deviates every
ray by the same angle --- a wedge of glass is a ``prism'' that bends
light, and a lens will turn out to be a stack of such wedges. Because
$n$ depends on $\lambda$, the deviation does too: a prism spreads white
light into a spectrum (blue deviated most), the heart of the prism
spectroscope.
\end{remark}

\section{The rainbow}

\begin{example}[Descartes's rainbow]\label{ex:b1:rays-reflection-refraction:rainbow}
\index{rainbow}
A ray of sunlight enters a spherical raindrop at incidence $i$, is
refracted ($\sin i = n\sin r$), reflects once on the back surface, and
exits after a second refraction. Each refraction deviates it by
$i - r$, the internal reflection by $\pi - 2r$: total deviation
$D(i) = \pi + 2i - 4r$. Rays fill the drop at every $i$; the observer
sees the brightest light where many rays leave in nearly the same
direction, i.e.\ where $D$ is stationary: $\dd D/\dd i = 2 - 4\,\dd r/\dd i
= 0$. Differentiating \omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law}, $\cos i = n\cos r\,\dd r/\dd i$, so
the condition reads $2\cos i = n\cos r$; squaring and using
$n^2\cos^2 r = n^2 - \sin^2 i = n^2 - 1 + \cos^2 i$ gives
\[
\cos^2 i = \frac{n^2 - 1}{3} .
\]
For water, $n = 1.333$: $i = 59.4^\circ$, $r = 40.2^\circ$,
$D_{\min} = 137.9^\circ$. The light comes back toward the Sun's side
at $180^\circ - 137.9^\circ = 42^\circ$ from the anti-solar direction:
the rainbow is a cone of half-angle $42^\circ$ around the shadow of the
observer's head. Red ($n = 1.331$) sits at $42.4^\circ$, violet
($n = 1.343$) at $40.6^\circ$ --- red outside, violet inside.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:1}
Compute the speed of light in water ($n = 1.33$), in glass ($1.50$) and
in diamond ($2.42$). How long does light take to cross \qty{1.0}{km} of
a fiber of index $1.48$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:2}
A ray in air meets water ($n = 1.33$) at $30^\circ$, then at
$60^\circ$ from the normal: find the refracted angles. A ray in water
meets the surface at $50^\circ$: what happens?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:3}
Compute the \omterm{thm:b1:rays-reflection-refraction:tir}{critical angle} for glass--air ($n = 1.50$), diamond--air
($2.42$) and water--air ($1.33$). What is the angular diameter of the
disk of sky a diver sees?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:4}
A person \qty{1.70}{m} tall, eyes \qty{10}{cm} below the top of the
head, stands before a vertical wall mirror. Find the smallest mirror
(height, and position of its edges above the floor) in which the whole
body is visible. Does the answer depend on the distance to the mirror?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:5}
A coin lies \qty{1.2}{m} deep in a pool; at what depth does it appear
to someone looking from above? A fisherman's eyes are \qty{1.8}{m}
above the water: how high do they appear to a fish looking up? Justify
both with small-angle rays.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:6}
A prism has $A = 60^\circ$ and $n = 1.52$. Compute the \omterm{prop:b1:rays-reflection-refraction:prism}{minimum
deviation}. Show that a ray can emerge from the second face only if
$A \leq 2 i_c$, and check this prism qualifies.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:7}
A step-index fiber has $n_1 = 1.50$, $n_2 = 1.48$. Compute its
\omterm{def:b1:rays-reflection-refraction:fiber}{numerical aperture}, its \omterm{prop:b1:rays-reflection-refraction:fiber}{acceptance angle}, the intermodal delay per
kilometer, and the resulting maximum bit rate over \qty{1}{km}
(take $B_{\max} \approx 1/(2\Delta t)$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:8}
A thin prism of apex $10^\circ$ has $n = 1.510$ for red and $1.530$ for
blue. Compute the deviation of each color and the angular width of the
spectrum it produces.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:9}
A ray meets a parallel-faced glass plate ($n = 1.50$, thickness
$e = \qty{2.0}{cm}$) at $45^\circ$. Show that it emerges parallel to
its incident direction, shifted sideways by $d = e\,\sin(i - r)/\cos r$,
and compute $d$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:10}
The secondary rainbow comes from rays reflected \emph{twice} inside the
drop. Show that $D = 2\pi + 2i - 6r$, that the stationary deviation
satisfies $\cos^2 i = (n^2 - 1)/8$, and compute the angle of the
secondary bow from the anti-solar direction for $n = 1.333$. Where is
it relative to the primary, and in which order are its colors?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:11}
On a hot road the air just above the asphalt has index
$n_{\text{hot}} = 1.00026$, the cooler air above it $n_{\text{cool}} =
1.00029$. Model the layers as a sharp interface: find the critical
grazing angle (measured from the road) below which a ray from the sky
is totally reflected upward. A driver's eye is \qty{1.2}{m} above the
road: beyond what distance does the road look like water?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:12}
Points $A$ (height $a$ above an interface) and $B$ (depth $b$ below it)
are separated horizontally by $d$; light travels at $v_1$ above and
$v_2$ below. Write the travel time of the broken path through the
interface point at horizontal position $x$, and show that the time is
stationary exactly when $\sin i_1/v_1 = \sin i_2/v_2$ --- \omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law}.
\end{exercise}

\section{Problem: The optical fiber link}

\begin{problem}[{Weekend problem --- a glass thread under the sea: how
far and how fast can a pulse of light carry a bit before geometry,
glass and dispersion blur it away}]
\label{pb:b1:rays-reflection-refraction:1}
A link uses a step-index fiber with \omterm{def:b1:rays-reflection-refraction:fiber}{core} $n_1 = 1.480$, \omterm{def:b1:rays-reflection-refraction:fiber}{cladding}
$n_2 = 1.465$, \omterm{def:b1:rays-reflection-refraction:fiber}{core} radius $a = \qty{25}{\micro m}$, at the wavelength
$\lambda_0 = \qty{1550}{nm}$. Data: $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Part I --- Light in glass.}
\begin{enumerate}
  \item Define the index and compute the speed of light in the \omterm{def:b1:rays-reflection-refraction:fiber}{core}.
  \item How long does a pulse take to travel \qty{50}{km} along the
        axis?
  \item Compute the frequency of the light and its wavelength inside the
        \omterm{def:b1:rays-reflection-refraction:fiber}{core}.
  \item Why is the \omterm{def:b1:rays-reflection-refraction:fiber}{core} clad in glass of lower index rather than left
        bare in air, which would give a larger index step?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Guiding.}
\begin{enumerate}[resume]
  \item Compute the \omterm{thm:b1:rays-reflection-refraction:tir}{critical angle} at the core--cladding surface.
  \item Deduce the largest angle a guided ray can make with the axis.
  \item A ray enters the flat end face from air at angle $\theta$ to
        the axis. Show that it is guided if $\sin\theta \leq
        \sqrt{n_1^2 - n_2^2}$, and compute the \omterm{prop:b1:rays-reflection-refraction:fiber}{acceptance angle}
        $\theta_a$.
  \item Give the \omterm{def:b1:rays-reflection-refraction:fiber}{numerical aperture}. Why does a larger index step make
        the fiber easier to feed with light?
  \item The fiber is bent along a circle of radius $R_b$. Consider a
        ray travelling parallel to the axis along the inner edge of the
        \omterm{def:b1:rays-reflection-refraction:fiber}{core}: show that it meets the outer wall at incidence $i$ with
        $\sin i = R_b/(R_b + 2a)$, and find the smallest $R_b$ for which
        it is still guided.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{prop:b1:rays-reflection-refraction:fiber}{Intermodal dispersion}.}
\begin{enumerate}[resume]
  \item Express the travel times of the axial ray and of the steepest
        guided ray over a length $L$.
  \item Deduce the spread $\Delta t$ per kilometer, and its value over
        $L = \qty{2}{km}$.
  \item A bit is a short pulse; two successive pulses must stay
        separated by at least $2\Delta t$ to be told apart. Estimate the
        maximum bit rate over \qty{2}{km}.
  \item Show that the product (bit rate) $\times$ (length) is a
        constant for this fiber, and compute it in \unit{Mbit.s^{-1}.km}.
  \item In a graded-index fiber the index decreases smoothly from the
        axis outward. Explain qualitatively why this reduces the
        intermodal spread.
  \item A single-mode fiber has a \omterm{def:b1:rays-reflection-refraction:fiber}{core} diameter of about \qty{9}{\micro m}.
        Compare with $\lambda_0$ and explain why the ray model cannot
        describe it; what is gained?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Attenuation and chromatic dispersion.}
The fiber attenuates the power by $\alpha = \qty{0.20}{dB/km}$ at
\qty{1550}{nm}: $P(L) = P_0\,10^{-\alpha L/10}$. The transmitter
launches $P_0 = \qty{1.0}{mW}$; the receiver needs at least
$P_{\min} = \qty{1.0}{\micro W}$. Because $n$ depends on $\lambda$, two
wavelengths travel at slightly different speeds: a source of spectral
width $\Delta\lambda$ gives a pulse spread $\Delta t_c = D\,\Delta\lambda\,L$
with $D = \qty{17}{ps/(nm.km)}$.
\begin{enumerate}[resume]
  \item What fraction of the power survives \qty{100}{km}?
  \item Find the maximum length before an amplifier is needed.
  \item Amplifiers are placed every \qty{100}{km}: what margin, in dB
        and as a power ratio, does this leave for connectors and
        splices?
  \item Compute $\Delta t_c$ over \qty{100}{km} for an LED source
        ($\Delta\lambda = \qty{40}{nm}$) and for a laser
        ($\Delta\lambda = \qty{0.10}{nm}$); deduce the maximum bit rate
        of each over that length (same criterion as above).
  \item At \qty{850}{nm} silica attenuates \qty{2.0}{dB/km}. What
        distance would the same power budget allow there? Why is
        \qty{1550}{nm} the telecom wavelength?
  \item Copper coaxial cable attenuates about \qty{30}{dB/km} at the
        frequencies needed for \qty{1}{Gbit/s}. Compare the amplifier
        spacing with the fiber's.
\end{enumerate}

\medskip
\noindent\textbf{Part V --- The whole link.}
\begin{enumerate}[resume]
  \item For the step-index fiber over \qty{2}{km}, which effect limits
        the bit rate: intermodal spread, chromatic spread (laser source)
        or attenuation? Give the limiting bit rate.
  \item For a single-mode fiber over \qty{100}{km} with the laser
        source: which effect limits, and at what bit rate?
  \item The single-mode link carries $80$ lasers at wavelengths
        \qty{0.8}{nm} apart (wavelength multiplexing), each at the bit
        rate of the previous question. Total capacity?
  \item Form the bit-rate--length product of the step-index link and of
        one single-mode channel; by what factor does the single-mode
        fiber win, and which single physical fact is responsible?
\end{enumerate}
\end{problem}
