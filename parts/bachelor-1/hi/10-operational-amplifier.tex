\chapter{संक्रियात्मक प्रवर्धक}\label{ch:b1:operational-amplifier}

किसी पुल की धरन पर चिपका विकृति-मापी ट्रक गुज़रने पर अपना प्रतिरोध हज़ारवें
हिस्से भर बदलता है; और उसके चारों ओर बना \omterm{prop:b1:dc-circuits:wheatstone}{व्हीटस्टोन सेतु}
(\cref{ch:b1:dc-circuits}) उसे दो मिलीवोल्ट में बदल देता है। दो
मिलीवोल्ट कुछ भी नहीं चला सकते। मापी और उस चेतावनी-यंत्र के बीच, जो पुल को
यातायात के लिए बंद कर देगा, नाख़ून जितनी एक चिप बैठी है, जो कुछ ही पैसों में
बिकती है और संकेत को हज़ार गुना कर देती है, उसे \omterm{def:b1:filters-transfer-functions:H}{फ़िल्टर} करती है, किसी देहली से
उसकी तुलना करती है और एक दीपक जला देती है --- चार काम, एक अवयव, चार भिन्न
जोड़-तोड़। यह अध्याय \omterm{def:b1:operational-amplifier:opamp}{संक्रियात्मक प्रवर्धक} को उसके आदर्श प्रतिरूप के द्वारा
प्रस्तुत करता है, उन मुट्ठी भर परिपथों को व्युत्पन्न करता है जिनसे हर उपकरण
बनता है, और दिखाता है कि पुनर्भरण हटा देने पर क्या होता है।

\section{अवयव और उसका आदर्श प्रतिरूप}

\begin{definition}[संक्रियात्मक प्रवर्धक]\label{def:b1:operational-amplifier:opamp}
\index{संक्रियात्मक प्रवर्धक}\index{संतृप्ति विभवांतर}\index{खुले-पाश की लब्धि}
\emph{संक्रियात्मक प्रवर्धक} एक एकीकृत परिपथ है, जिसके दो
निवेश हैं --- विभव $V_+$ वाला \emph{अ-प्रतिलोमी} निवेश और
विभव $V_-$ वाला \emph{प्रतिलोमी} निवेश --- तथा विभव $V_s$ वाला एक निर्गम,
और जो दो आपूर्तियों $\pm V_{\mathrm{cc}}$ (प्रायः
$\pm\qty{15}{V}$) से चलता है, जिन्हें आरेखों में आमतौर पर छोड़ दिया जाता है। वह
अंतर $\epsilon = V_+ - V_-$ को प्रवर्धित करता है:
\[
V_s = \mu\,\epsilon \quad\text{जब तक } \abs{V_s} < V_{\mathrm{sat}},
\]
जहाँ \emph{खुले-पाश की \omterm{def:b1:filters-transfer-functions:H}{लब्धि}} $\mu$ लगभग $10^5$ कोटि की है और
\emph{संतृप्ति \omterm{def:b1:dc-circuits:voltage}{विभवांतर}} $V_{\mathrm{sat}}$
$V_{\mathrm{cc}}$ से ज़रा नीचे। जब $\mu\abs\epsilon$ $V_{\mathrm{sat}}$ से आगे
निकलने लगे, तब निर्गम $+V_{\mathrm{sat}}$ या $-V_{\mathrm{sat}}$ पर अटक जाता
है: यही \emph{संतृप्त} व्यवस्था है।
\end{definition}

\begin{definition}[आदर्श संक्रियात्मक प्रवर्धक]\label{def:b1:operational-amplifier:ideal}
\index{आदर्श संक्रियात्मक प्रवर्धक}\index{आभासी लघुपथ}\index{रैखिक व्यवस्था}
\emph{आदर्श} प्रतिरूप यह मान लेता है: (क) किसी भी निवेश में कोई धारा नहीं
घुसती, $i_+ = i_- = 0$ (अनंत निवेश \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा}); (ख) निर्गम अपना
\omterm{def:b1:dc-circuits:voltage}{विभवांतर} लाद देता है, चाहे उससे कितनी भी धारा खींची जाए (शून्य निर्गम
\omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा}); (ग) अनंत \omterm{def:b1:filters-transfer-functions:H}{लब्धि}, $\mu \to \infty$। \emph{रैखिक व्यवस्था}
($\abs{V_s}
< V_{\mathrm{sat}}$) में तीसरी मान्यता यह लाद देती है कि
\[
\epsilon = V_+ - V_- = 0 :
\]
यानी दोनों निवेश एक ही विभव पर बैठते हैं (एक ``आभासी लघुपथ''),
हालाँकि उनके बीच कोई धारा नहीं बहती।
\end{definition}

$\mu \sim 10^5$ का एक तीखा रैखिक खंड (जो यहाँ कहीं अधिक चपटा खींचा गया है),
और फिर $\pm V_{\mathrm{sat}}$ पर संतृप्ति। आदर्श प्रतिरूप में रैखिक खंड

\begin{proposition}[ऋणात्मक पुनर्भरण और रैखिक व्यवस्था]\label{prop:b1:operational-amplifier:feedback}
\index{ऋणात्मक पुनर्भरण}\index{धनात्मक पुनर्भरण}
यदि निर्गम किसी जाल के द्वारा वापस \emph{प्रतिलोमी} निवेश से जोड़ दिया जाए
(\omterm{prop:b1:operational-amplifier:feedback}{ऋणात्मक पुनर्भरण}), तो परिपथ का $\epsilon = 0$ पर एक स्थायी रैखिक \omterm{met:b1:dc-circuits:loadline}{क्रिया
बिंदु} होता है और आदर्श प्रवर्धक के नियम लागू होते हैं। और यदि
निर्गम अ-प्रतिलोमी निवेश को लौटाया जाए (\omterm{prop:b1:operational-amplifier:feedback}{धनात्मक पुनर्भरण}), या बिलकुल
न लौटाया जाए, तो प्रवर्धक संतृप्त हो जाता है: $V_s = \pm V_{\mathrm{sat}}$,
$\epsilon$ के चिह्न के अनुसार।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
\omterm{prop:b1:operational-amplifier:feedback}{ऋणात्मक पुनर्भरण} के साथ मान लीजिए $V_s$ अपने साम्य मान से थोड़ा ऊपर
चढ़ता है; तब पुनर्भरण जाल के द्वारा $V_-$ चढ़ता है, $\epsilon$
गिरता है, और प्रवर्धक $V_s$ को वापस नीचे खींच लेता है: यानी विक्षोभ सुधर जाता
है। \omterm{prop:b1:operational-amplifier:feedback}{धनात्मक पुनर्भरण} के साथ वही विक्षोभ $V_+$ को चढ़ाता है,
$\epsilon$ बढ़ाता है और $V_s$ को और दूर धकेलता है, जब तक वह संतृप्त न हो जाए।
पूरा तर्क (प्रवर्धक का प्रथम कोटि प्रतिरूप, $\mu(j\omega)
= \mu_0/(1 + j\omega/\omega_0)$, और उससे बनते अवकल समीकरण की स्थिरता) वर्ष 2
के खंड का विषय है; और ऊपर दिया नियम ही वह है जिसे काम में लिया जाता है।
\end{proof}

\begin{remark}[असली प्रवर्धक]\label{rem:b1:operational-amplifier:real}
\index{लब्धि--बैंड-चौड़ाई गुणनफल}\index{आरोहण दर}\index{निवेश विस्थापन विभवांतर}
व्यवहार में आदर्श से तीन विचलन मायने रखते हैं। \omterm{def:b1:operational-amplifier:opamp}{खुले-पाश की लब्धि}
आवृत्ति के साथ $\mu_0/(1 + jf/f_0)$ की तरह गिरती है, जहाँ $f_0 \sim \qty{10}{Hz}$:
और गुणनफल $\mu_0 f_0 = f_T \sim \qty{1}{MHz}$ (यानी \emph{\omterm{rem:b1:operational-amplifier:real}{लब्धि--बैंड-चौड़ाई
गुणनफल}}) किसी अनुगामी को उपलब्ध बैंड-चौड़ाई है, जबकि बंद-पाश \omterm{def:b1:filters-transfer-functions:H}{लब्धि}
$G$ वाले किसी परिपथ की बैंड-चौड़ाई $f_T/G$ होती है। निर्गम
\emph{\omterm{rem:b1:operational-amplifier:real}{आरोहण दर}} से तेज़ नहीं बदल सकता, जो $\sim\qty{1}{V/\micro s}$ होती है।
और कोई छोटा-सा \emph{\omterm{rem:b1:operational-amplifier:real}{निवेश विस्थापन विभवांतर}} ($\sim\qty{1}{mV}$) भी संकेत की
तरह ही प्रवर्धित होता है --- मिलीवोल्ट के निवेशों के लिए घातक, जब तक उसे
साधा न जाए।
\end{remark}

\section{बुनियादी रैखिक परिपथ}

\begin{method}[रैखिक व्यवस्था में किसी आदर्श ऑप-ऐंप परिपथ का विश्लेषण]\label{met:b1:operational-amplifier:method}
\begin{enumerate}
  \item जाँचिए कि पुनर्भरण प्रतिलोमी निवेश को जाता है।
  \item $i_+ = i_- = 0$ लिखिए: निवेश-संधियाँ किरचॉफ़ के \omterm{thm:b1:dc-circuits:kirchhoff}{संधि नियम} का पालन
        करती हैं, और प्रवर्धक में कोई धारा नहीं जाती (विभाजक तथा \omterm{prop:b1:dc-circuits:millman}{मिलमैन की
        प्रमेय} लागू होते हैं)।
  \item $V_+ = V_-$ लिखिए और $V_s$ के लिए हल कीजिए।
  \item बाद में जाँचिए कि $\abs{V_s} < V_{\mathrm{sat}}$ है; यदि नहीं, तो
        प्रवर्धक संतृप्त है और परिणाम $\pm V_{\mathrm{sat}}$ है।
\end{enumerate}
\end{method}

\begin{proposition}[अनुगामी, अ-प्रतिलोमी और प्रतिलोमी प्रवर्धक]\label{prop:b1:operational-amplifier:basic}
\index{विभव अनुगामी}\index{अ-प्रतिलोमी प्रवर्धक}\index{प्रतिलोमी प्रवर्धक}\index{आभासी भू}
\begin{itemize}
  \item \emph{अनुगामी}: निर्गम $V_-$ से जुड़ा, निवेश $V_+$ पर:
        $V_s = V_e$; अनंत निवेश \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा}, शून्य निर्गम \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा}।
  \item \emph{\omterm{prop:b1:operational-amplifier:basic}{अ-प्रतिलोमी प्रवर्धक}}: निर्गम से $V_-$ तक एक
        विभाजक $R_2$ (पुनर्भरण) और $R_1$ (भू तक), निवेश $V_+$ पर:
        \[
        V_s = \left(1 + \frac{R_2}{R_1}\right)V_e .
        \]
  \item \emph{\omterm{prop:b1:operational-amplifier:basic}{प्रतिलोमी प्रवर्धक}}: $V_+$ भूसंपर्कित, निवेश
        $R_1$ के द्वारा $V_-$ तक, पुनर्भरण $R_2$ निर्गम से $V_-$ तक:
        \[
        V_s = -\frac{R_2}{R_1}\,V_e ,
        \]
        जहाँ $V_-$ भू के विभव पर बना रहता है (एक \emph{\omterm{prop:b1:operational-amplifier:basic}{आभासी भू}})
        और निवेश \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} $R_1$ होती है।
\end{itemize}
\end{proposition}

\begin{proof}
अनुगामी: $V_- = V_s$ और $V_+ = V_e$; $\epsilon = 0$ से $V_s = V_e$ मिलता है।
अ-प्रतिलोमी: $V_-$ में कोई धारा नहीं जाती, अतः विभाजक देता है $V_- =
V_sR_1/(R_1 + R_2)$; अब इसे $V_+ = V_e$ के बराबर रखिए। प्रतिलोमी: $V_- = V_+ = 0$;
और $V_-$ पर \omterm{thm:b1:dc-circuits:kirchhoff}{संधि नियम}, $i_- = 0$ के साथ: $(V_e - 0)/R_1 + (V_s - 0)/R_2 = 0$।
स्रोत $V_e/R_1$ देता है: अतः निवेश \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} $R_1$।
\end{proof}

\begin{example}[अनुगामी क्यों]\label{ex:b1:operational-amplifier:follower}
तेवेनिन प्रतिरोध \qty{10}{k\ohm} का कोई संवेदक, जो \qty{1}{k\ohm} के
भार को खिलाए, अपने खुले परिपथ के \omterm{def:b1:dc-circuits:voltage}{विभवांतर} का केवल $1/11$ पहुँचाता है
(\cref{rem:b1:dc-circuits:loading})। उनके बीच लगा कोई अनुगामी संवेदक से कोई
धारा नहीं खींचता और भार को शून्य-प्रतिबाधा वाले निर्गम से चलाता है:
अतः पूरा \omterm{def:b1:dc-circuits:voltage}{विभवांतर} पहुँच जाता है। अनुगामी एक \emph{\omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा}
अनुकूलक} है --- और वही \cref{prop:b1:filters-transfer-functions:cascade} के
शृंखलन-नियम को यथार्थ बना देता है।
\end{example}

\begin{proposition}[योग और अंतर प्रवर्धक]\label{prop:b1:operational-amplifier:sumdiff}
\index{योग प्रवर्धक}\index{अंतर प्रवर्धक}
निवेश $V_1, V_2$ $R_1, R_2$ के द्वारा प्रतिलोमी संधि तक, पुनर्भरण
$R_f$, और $V_+$ भूसंपर्कित:
\[
V_s = -R_f\left(\frac{V_1}{R_1} + \frac{V_2}{R_2}\right)
\quad (\text{योग प्रवर्धक}).
\]
$V_1$ $R_1$ के द्वारा $V_-$ तक (पुनर्भरण $R_2$) और $V_2$
$R_1$ के द्वारा $V_+$ तक ($R_2$ $V_+$ से भू तक) हो, तो:
\[
V_s = \frac{R_2}{R_1}\,(V_2 - V_1)
\quad (\text{अंतर प्रवर्धक}).
\]
\end{proposition}

\begin{proof}
योग: \omterm{prop:b1:operational-amplifier:basic}{आभासी भू} पर \omterm{thm:b1:dc-circuits:kirchhoff}{संधि नियम}, $V_1/R_1 + V_2/R_2 + V_s/R_f =
0$। अंतर: $V_+ = V_2R_2/(R_1 + R_2)$ (विभाजक); $V_-$ पर \omterm{thm:b1:dc-circuits:kirchhoff}{संधि नियम}:
$(V_1 - V_-)/R_1 + (V_s - V_-)/R_2 = 0$, अतः $V_s = V_-(1 + R_2/R_1) -
V_1R_2/R_1$; अब $V_- = V_+$ रखिए और सरल कीजिए।
\end{proof}

\section{समाकलक और अवकलक}

\begin{proposition}[समाकलक]\label{prop:b1:operational-amplifier:integrator}
\index{समाकलक (ऑप-ऐंप)}
निवेश $R$ के द्वारा $V_-$ तक, पुनर्भरण के रूप में संधारित्र $C$, और $V_+$
भूसंपर्कित:
\[
V_s(t) = -\frac{1}{RC}\int_0^t V_e(t')\,\dd t' + V_s(0),
\qquad
\underline H = -\frac{1}{jRC\omega} .
\]
\omterm{def:b1:filters-transfer-functions:H}{लब्धि} हर आवृत्ति पर प्रति दशक \qty{20}{dB} गिरती है और \omterm{def:b1:wave-propagation:signal}{कला}
$+90^\circ$ रहती है: यानी एक यथार्थ समाकलक --- जब तक कि कोई नन्हा-सा विस्थापन
या अभिनति, हमेशा के लिए समाकलित होकर, निर्गम को संतृप्ति में न धकेल दे।
\end{proposition}

\begin{proof}
\omterm{prop:b1:operational-amplifier:basic}{आभासी भू}: धारा $V_e/R$ संधारित्र में बहती है, जिसका
\omterm{def:b1:dc-circuits:voltage}{विभवांतर} ($V_-$ से निर्गम) $-V_s$ है;
अतः $C\,\dd(-V_s)/\dd t = V_e/R$,
और सम्मिश्र संकेतन में $\underline H = -1/(jRC\omega)$।
\end{proof}

\begin{remark}[समाकलक को साधना]\label{rem:b1:operational-amplifier:pseudo}
\index{छद्म-समाकलक}
कोई \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} $R'$, जो $C$ के समांतर में जुड़ा हो, देता है
\[
\underline H = -\frac{R'/R}{1 + jR'C\omega},
\]
यानी दिष्ट \omterm{def:b1:filters-transfer-functions:H}{लब्धि} $-R'/R$ और कटऑफ़ $1/R'C$ वाला प्रथम कोटि का निम्न-पारक, जो
$\omega \gg 1/R'C$ के लिए समाकलन करता है पर संतृप्ति की ओर बह नहीं सकता। इसी तरह अवकलक (निवेश पर संधारित्र,
पुनर्भरण के रूप में \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक}: $V_s = -RC\,\dd V_e/\dd t$, $\underline H =
-jRC\omega$) ऊँची आवृत्ति के रव को बिना सीमा के प्रवर्धित करता है, जब तक
कोई छोटा श्रेणी \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} उसकी \omterm{def:b1:filters-transfer-functions:H}{लब्धि} पर टोपी न लगा दे।
\end{remark}

लिए शून्य-प्रतिबाधा निर्गम के साथ।
\end{example}

\section{सक्रिय फ़िल्टर}

\begin{proposition}[प्रथम कोटि का सक्रिय निम्न-पारक]\label{prop:b1:operational-amplifier:activelp}
\index{सक्रिय फ़िल्टर}
\omterm{prop:b1:operational-amplifier:basic}{प्रतिलोमी प्रवर्धक}, जिसमें संधारित्र $C$ अपने पुनर्भरण \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} $R_2$ के
आरपार लगा हो, उसके लिए
\[
\underline H = -\frac{R_2/R_1}{1 + jR_2C\omega}:
\]
यानी दिष्ट \omterm{def:b1:filters-transfer-functions:H}{लब्धि} $-R_2/R_1$ (जो $1$ से बड़ी हो सकती है) और कटऑफ़
$\omega_c = 1/R_2C$ वाला निम्न-पारक, साथ में शून्य निर्गम \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} --- अतः
चरण साधारण गुणा से शृंखलित हो जाते हैं।
\end{proposition}

\begin{proof}
\omterm{prop:b1:operational-amplifier:basic}{प्रतिलोमी प्रवर्धक}, जिसमें पुनर्भरण \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} $R_2 \parallel 1/jC\omega
= R_2/(1 + jR_2C\omega)$ है, $R_2$ की जगह।
\end{proof}

\section{ऋणात्मक पुनर्भरण के बिना: तुलनित्र}

\begin{proposition}[तुलनित्र]\label{prop:b1:operational-amplifier:comparator}
\index{तुलनित्र}
बिना किसी पुनर्भरण के आदर्श प्रवर्धक संतृप्त रहता है:
$V_s = +V_{\mathrm{sat}}$ जब $V_+ > V_-$ हो, और
$-V_{\mathrm{sat}}$ जब $V_+ < V_-$ हो। किसी एक निवेश पर संदर्भ $V_{\mathrm{ref}}$
और दूसरे पर संकेत रखने से निर्गम बता देता है कि संकेत संदर्भ से आगे निकला या
नहीं --- यानी एक-बिट का परिवर्तक।
\end{proposition}

\begin{proof}
$\mu\epsilon$ $V_{\mathrm{sat}}$ से आगे निकल जाता है, किसी भी $\abs\epsilon >
V_{\mathrm{sat}}/\mu \sim \qty{0.1}{mV}$ के लिए।
\end{proof}

\begin{proposition}[श्मिट ट्रिगर]\label{prop:b1:operational-amplifier:schmitt}
\index{श्मिट ट्रिगर}\index{शैथिल्य}
निर्गम का अंश $\beta = R_1/(R_1 + R_2)$ वापस $V_+$ को दीजिए
(विभाजक: $R_1$ भू तक, $R_2$ निर्गम से) और संकेत
$V_e$ $V_-$ पर लगाइए। निर्गम $+V_{\mathrm{sat}}$ से
$-V_{\mathrm{sat}}$ पर तब पलटता है जब $V_e$ चढ़कर $+\beta V_{\mathrm{sat}}$ पार
करे, और वापस तब जब $V_e$ गिरकर $-\beta V_{\mathrm{sat}}$ से नीचे जाए: यानी दो
देहलियाँ, चौड़ाई $2\beta V_{\mathrm{sat}}$ का एक \emph{शैथिल्य}, और उस चौड़ाई
से छोटे रव के प्रति निरापद।
\end{proposition}

\begin{proof}
\omterm{prop:b1:operational-amplifier:feedback}{धनात्मक पुनर्भरण}: अतः संतृप्त। यदि $V_s = +V_{\mathrm{sat}}$ हो तो $V_+ =
\beta V_{\mathrm{sat}}$, और $\epsilon = \beta V_{\mathrm{sat}} - V_e$
तब तक धनात्मक रहता है जब तक $V_e$ $\beta V_{\mathrm{sat}}$ से आगे न निकले; तब
निर्गम पलट जाता है, $V_+$ गिरकर $-\beta V_{\mathrm{sat}}$ हो जाता है, और नई
अवस्था तब तक टिकी रहती है जब तक $V_e$ उससे नीचे न गिरे।
\end{proof}

\begin{example}[शिथिलन दोलित्र]\label{ex:b1:operational-amplifier:astable}
\index{शिथिलन दोलित्र}
\omterm{prop:b1:operational-amplifier:schmitt}{श्मिट ट्रिगर} के निर्गम को $R$ के द्वारा उसी के प्रतिलोमी निवेश से जोड़
दीजिए, और $C$ को $V_-$ से भू तक लगा दीजिए। संधारित्र
$\pm V_{\mathrm{sat}}$ की ओर आवेशित होता है और हर बार
$\pm\beta V_{\mathrm{sat}}$ पर पहुँचकर ट्रिगर को पलट देता है: अतः निर्गम एक
वर्ग तरंग है, संधारित्र का \omterm{def:b1:dc-circuits:voltage}{विभवांतर} चरघातांकी चापों की शृंखला, और आवर्तकाल है
\[
T = 2RC\ln\frac{1 + \beta}{1 - \beta}
\]
(\cref{exo:b1:operational-amplifier:12}): $2.2\,RC$, जब $\beta = \tfrac12$ हो।
कोई निवेश नहीं, एक संधारित्र, और एक घड़ी --- यानी ऐसा निकाय जो इसलिए दोलन करता
है कि वह थम ही नहीं सकता, और यह विषय वर्ष 2 के खंड के पुनर्भरण-दोलित्रों के
साथ लौटता है।
\end{example}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:operational-amplifier:7}
पिछले समाकलक के संधारित्र के आरपार \qty{1.0}{M\ohm} का \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} जोड़ दिया
जाता है। \omterm{def:b1:filters-transfer-functions:H}{स्थानांतरण फलन}, दिष्ट \omterm{def:b1:filters-transfer-functions:H}{लब्धि}, \omterm{def:b1:filters-transfer-functions:bode}{कटऑफ़
आवृत्ति}, और \qty{1.0}{kHz} पर \omterm{def:b1:filters-transfer-functions:H}{लब्धि} दीजिए; किस परास में परिपथ अब भी
समाकलन करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:operational-amplifier:8}
$R_1 = \qty{1.0}{k\ohm}$ का उपयोग करके ऐसा प्रतिलोमी सक्रिय निम्न-पारक
गढ़िए जिसकी \omterm{def:b1:filters-transfer-functions:bode}{पारक बैंड} में \omterm{def:b1:filters-transfer-functions:H}{लब्धि} \qty{20}{dB} और कटऑफ़ \qty{500}{Hz} हो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:operational-amplifier:9}
किसी असली प्रवर्धक के लिए $\mu_0 = 10^5$ और $f_T = \qty{1.0}{MHz}$ है।
$100$ की \omterm{def:b1:filters-transfer-functions:H}{लब्धि} के लिए गढ़े गए किसी \omterm{prop:b1:operational-amplifier:basic}{अ-प्रतिलोमी प्रवर्धक} के लिए: वास्तविक दिष्ट
\omterm{def:b1:filters-transfer-functions:H}{लब्धि} (विश्लेषण में $\mu_0$ को परिमित रखिए), और बैंड-चौड़ाई। और किसी
अनुगामी की बैंड-चौड़ाई?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:operational-amplifier:10}
$V_{\mathrm{ref}} = \qty{2.0}{V}$ वाला कोई तुलनित्र, जो $V_-$ पर लगा है, ऐसे
धीरे-धीरे चढ़ते संकेत को देखता है जिसमें \qty{50}{mV} का रव है। जब संकेत \qty{2.0}{V}
पार करे तब निर्गम का वर्णन कीजिए। उपाय?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:operational-amplifier:11}
$R_1 = \qty{1.0}{k\ohm}$, $R_2 = \qty{9.0}{k\ohm}$,
$V_{\mathrm{sat}} = \qty{15}{V}$ वाला \omterm{prop:b1:operational-amplifier:schmitt}{श्मिट ट्रिगर}: दोनों देहलियाँ व्युत्पन्न कीजिए और निकालिए।
$V_s$ खींचिए, \omterm{def:b1:wave-propagation:signal}{आयाम} \qty{3}{V} की त्रिभुजाकार $V_e$ के लिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:operational-amplifier:12}
$R_1 = R_2$ ($\beta = \tfrac12$) वाला \omterm{ex:b1:operational-amplifier:astable}{शिथिलन दोलित्र}। आवर्तकाल
$T = 2RC\ln\frac{1 + \beta}{1 - \beta}$ को $C$ के आवेशन से, दोनों देहलियों
के बीच $R$ के द्वारा, व्युत्पन्न कीजिए; \qty{1.0}{kHz} के लिए $R$ चुनिए,
$C = \qty{100}{nF}$ के साथ; और $V_s(t)$ तथा $V_C(t)$ खींचिए।
\end{exercise}

\section{समस्या: विकृति-मापी के लिए एक उपकरण-शृंखला}

\begin{problem}[{सप्ताहांत समस्या --- पुल की धरन पर दो मिलीवोल्ट से लेकर एक
लाल दीपक तक: अनुगामी, एक अंतर प्रवर्धक, एक सक्रिय
फ़िल्टर और एक श्मिट ट्रिगर, तथा वे दो अपूर्णताएँ जो तय करती हैं कि
शृंखला काम करेगी या नहीं}]
\label{pb:b1:operational-amplifier:1}
प्रतिरोध $R(1 + \epsilon)$ का कोई विकृति-मापी, $R = \qty{1.00}{k\ohm}$,
$0 \leq \epsilon \leq \num{2.0e-3}$, तीन जड़े \qty{1.00}{k\ohm} प्रतिरोधकों
के साथ \omterm{prop:b1:dc-circuits:wheatstone}{व्हीटस्टोन सेतु} बनाता है, जिसे $E = \qty{5.00}{V}$ खिलाता है।
सेतु का निर्गम $u = V_B - V_A$ (दोनों मध्यबिंदुओं के बीच)
$u \approx E\epsilon/4$ है (चिह्न तारों से चुना जाता है)। प्रवर्धक:
जब तक कुछ और न कहा जाए, आदर्श; $V_{\mathrm{sat}} = \qty{15}{V}$; और जहाँ ज़रूरी हो,
$f_T = \qty{1.0}{MHz}$, \omterm{rem:b1:operational-amplifier:real}{आरोहण दर} \qty{0.5}{V/\micro s}, निवेश विस्थापन
\qty{1}{mV}।

\medskip
\noindent\textbf{भाग I --- सेतु और उस पर भार।}
\begin{enumerate}
  \item याद कीजिए कि $u \approx E\epsilon/4$ क्यों है, छोटे $\epsilon$ के लिए।
  \item $u$ निकालिए, पूर्ण पैमाने पर ($\epsilon = \num{2.0e-3}$)।
  \item शून्य विकृति पर और पूर्ण पैमाने पर हर मध्यबिंदु का विभव दीजिए
        (सेतु के ऋण सिरे के सापेक्ष)। उनके साझे मान को क्या कहते हैं?
  \item हर मध्यबिंदु दो \qty{1}{k\ohm} प्रतिरोधकों वाले किसी विभाजक का
        निर्गम है: दोनों मध्यबिंदुओं के बीच तेवेनिन प्रतिरोध क्या दिखता है?
  \item सेतु के निर्गम के आरपार ऐसा प्रवर्धक जोड़ा जाता है जिसका निवेश
        प्रतिरोध \qty{10}{k\ohm} है: उसे $u$ का कितना अंश मिलता है?
  \item हर मध्यबिंदु पर लगा अनुगामी इसे क्यों ठीक कर देता है? काम में आए
        दोनों गुण बताइए।
  \item अनुगामियों के निर्गम \omterm{def:b1:dc-circuits:voltage}{विभवांतर} क्या हैं ($V_A$, $V_B$ के पदों में),
        और वे सेतु से कितनी धारा खींचते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- अंतर को प्रवर्धित करना।}
\begin{enumerate}[resume]
  \item अनुगामी किसी \omterm{prop:b1:operational-amplifier:sumdiff}{अंतर प्रवर्धक} को खिलाते हैं (दोनों निवेशों पर $R_1$,
        और $R_2$ पुनर्भरण तथा भू तक)। $V_s =
        (R_2/R_1)(V_B - V_A)$ व्युत्पन्न कीजिए।
  \item $R_1 = \qty{10}{k\ohm}$ लीजिए और $R_2$ चुनिए, $100$ की \omterm{def:b1:filters-transfer-functions:H}{लब्धि} के लिए।
  \item दूसरा, अ-प्रतिलोमी चरण $10$ से गुणा करता है: उसके \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक}, कुल
        \omterm{def:b1:filters-transfer-functions:H}{लब्धि}, और पूर्ण पैमाने पर निर्गम दीजिए।
  \item दोनों मध्यबिंदु $E/2 = \qty{2.5}{V}$ के पास बैठते हैं (यानी
        उभयनिष्ठ-विधा \omterm{def:b1:dc-circuits:voltage}{विभवांतर})। ठीक-ठीक मिलाया गया \omterm{prop:b1:operational-amplifier:sumdiff}{अंतर प्रवर्धक} उसके
        साथ क्या करता है? यदि कोई एक प्रतिरोध-अनुपात $1\%$ चूक जाए, तो
        निवेश पर लौटाए गए मिथ्या \omterm{def:b1:dc-circuits:voltage}{विभवांतर} का आकलन कीजिए और संकेत से तुलना कीजिए।
  \item पहले प्रवर्धक का निवेश विस्थापन \qty{1}{mV} है। कुल \omterm{def:b1:filters-transfer-functions:H}{लब्धि} के बाद
        वह निर्गम में कितनी त्रुटि लाता है? निष्कर्ष निकालिए।
  \item $f_T = \qty{1}{MHz}$ के साथ हर चरण की बैंड-चौड़ाई क्या है?
        क्या वह \qty{5}{Hz} से नीचे के संकेत के लिए पर्याप्त है?
  \item निर्गम को लगभग \qty{10}{ms} में \qty{2.5}{V} झूलना है, जब कोई ट्रक
        आए: क्या \omterm{rem:b1:operational-amplifier:real}{आरोहण दर} सीमित करती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- फ़िल्टरन।}
प्रवर्धित संकेत बिजली की लाइन से पकड़ी गई \qty{50}{Hz} की गुनगुनाहट ढो रहा
है।
\begin{enumerate}[resume]
  \item \omterm{def:b1:filters-transfer-functions:H}{लब्धि} $-1$ और कटऑफ़ \qty{10}{Hz} वाला प्रतिलोमी सक्रिय निम्न-पारक
        गढ़िए, $R_1 = R_2 = \qty{16}{k\ohm}$ का उपयोग करके: $C$ निकालिए।
  \item \qty{50}{Hz} पर उसका क्षीणन \omterm{def:b1:filters-transfer-functions:bode}{डेसिबल} में निकालिए।
  \item \qty{5}{Hz} पर कला-खिसकाव और उससे मिलता विलंब निकालिए। क्या यह
        किसी चेतावनी-यंत्र के लिए स्वीकार्य है?
  \item यहाँ किसी निष्क्रिय RC की तुलना में \omterm{prop:b1:operational-amplifier:activelp}{सक्रिय फ़िल्टर} क्यों बेहतर है
        (दो कारण)?
  \item दूसरा एक जैसा चरण शृंखलित किया जाता है: \qty{50}{Hz} पर क्षीणन?
        यहाँ गुणनफल का नियम यथार्थ क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- देहली और चेतावनी।}
चेतावनी तब बजनी चाहिए जब विकृति \num{1.6e-3} से आगे निकले।
\begin{enumerate}[resume]
  \item यह शृंखला के किस निर्गम \omterm{def:b1:dc-circuits:voltage}{विभवांतर} से मेल खाता है? इसी संदर्भ वाला
        कोई तुलनित्र: निर्गम की अवस्थाएँ।
  \item \omterm{def:b1:filters-transfer-functions:H}{फ़िल्टर} किया संकेत देहली के पास अब भी \qty{50}{mV} का रव ढोता है।
        सादा तुलनित्र क्या करता है?
  \item किसी एक निवेश पर संदर्भ और पुनर्भरण $\beta$ वाला \omterm{prop:b1:operational-amplifier:schmitt}{श्मिट ट्रिगर}:
        दिखाइए कि देहलियाँ $V_{\mathrm{ref}} \pm
        \beta V_{\mathrm{sat}}$ हैं; $\beta$ चुनिए, $\pm\qty{0.10}{V}$ के लिए,
        और $R_1$, $R_2$ दीजिए।
  \item दोनों देहलियाँ वोल्ट में और विकृति में बताइए।
  \item निर्गम किसी \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} के द्वारा एक LED (\qty{2.0}{V}, \qty{13}{mA})
        चलाता है: उसे निकालिए। जब निर्गम $-V_{\mathrm{sat}}$ पर हो तब
        LED की रक्षा कौन करता है?
  \item शृंखला का चरण-दर-चरण सार लिखिए, और उन दो अ-आदर्शताओं के नाम लीजिए
        जो उसकी असली सीमाएँ तय करती हैं।
\end{enumerate}
\end{problem}
