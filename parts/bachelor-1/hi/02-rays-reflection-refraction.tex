\chapter{ज्यामितीय प्रकाशिकी: किरणें, परावर्तन, अपवर्तन}\label{ch:b1:rays-reflection-refraction}

पानी के गिलास में खड़ा तिनका सतह पर टूटा हुआ दिखता है; तालाब की तली से ऊपर
देखता तैराक पूरे आकाश को एक चमकीली चकती में सिमटा हुआ पाता है, जिसके चारों ओर
तली का दर्पण फैला होता है; और बाल जितना पतला काँच का तंतु एक टेलीफ़ोन वार्ता को
पूरे महासागर के पार इतनी कम हानि से पहुँचा देता है जितनी ताँबे का तार एक सड़क
पार करने में झेल लेता है। तीन नियम --- प्रकाश सीधा चलता है, बराबर कोणों पर
परावर्तित होता है, स्नेल के नियम से मुड़ता है --- इन सबका हिसाब दे देते हैं; यह
अध्याय उन्हें ठीक-ठीक कहता है, बताता है कि वे कहाँ सच होना बंद कर देते हैं, और
फिर उन्हें काम पर लगा देता है।

\section{ज्यामितीय प्रकाशिकी का प्रतिरूप}

\begin{definition}[प्रकाश किरण; समांगी माध्यम]\label{def:b1:rays-reflection-refraction:ray}
\index{प्रकाश किरण}\index{ज्यामितीय प्रकाशिकी}\index{समांगी माध्यम}
\emph{ज्यामितीय प्रकाशिकी} के प्रतिरूप में प्रकाश \emph{प्रकाश किरणों} के
अनुदिश संचरित होता है: ये दिशायुक्त वक्र हैं, जो \emph{समांगी} माध्यम (जिसके
हर बिंदु पर गुण समान हों) में सरल रेखाएँ होते हैं और एक-दूसरे से स्वतंत्र रूप
से ऊर्जा ले जाते हैं। पुंज किरणों का गट्ठर है; और कोई बिंदु स्रोत हर दिशा में
किरणें उत्सर्जित करता है।
\end{definition}

\begin{definition}[अपवर्तनांक]\label{def:b1:rays-reflection-refraction:index}
\index{अपवर्तनांक}\index{पारदर्शी माध्यम}
\emph{पारदर्शी} माध्यम प्रकाश को अपने भीतर से गुज़र जाने देता है; उसमें उसकी चाल
$v < c$ होती है। उसका \emph{अपवर्तनांक} है
\[
n = \frac{c}{v} \geq 1 .
\]
निर्वात: $n = 1$; हवा: $1.0003$; पानी: $1.33$; साधारण काँच: $1.5$ से
$1.6$; हीरा: $2.42$। आवृत्ति $\nu$ की कोई तरंग किसी माध्यम में प्रवेश करने
पर अपनी आवृत्ति बनाए रखती है, अतः उसका तरंगदैर्घ्य निर्वात के $\lambda_0 =
c/\nu$ से घटकर $\lambda = \lambda_0/n$ रह जाता है।
\end{definition}

\begin{remark}[वर्ण-विक्षेपण]\label{rem:b1:rays-reflection-refraction:dispersion}
\index{वर्ण-विक्षेपण (प्रकाशिक)}\index{कोशी का नियम}
पदार्थ में \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} तरंगदैर्घ्य पर थोड़ा-बहुत निर्भर करता है --- ऐसा माध्यम
\emph{विक्षेपी} कहलाता है। दृश्य परास में काँच के लिए कोशी का आनुभविक नियम
$n(\lambda) = A + B/\lambda^2$ अच्छा बैठता है: $n$ नीले के लिए लाल की अपेक्षा
लगभग $0.01$ से $0.02$ बड़ा होता है। इस अध्याय के अंत में आने वाले प्रिज़्म और
इंद्रधनुष, दोनों इसी अंतर पर जीते हैं।
\end{remark}

\begin{remark}[प्रतिरूप की सीमाएँ]\label{rem:b1:rays-reflection-refraction:limits}
प्रकाश एक तरंग है, और चौड़ाई $a$ के किसी छिद्र से गुज़रती तरंग लगभग
$\lambda/a$ कोटि के कोण से फैल जाती है (विवर्तन,
\cref{ch:b1:wave-propagation})। किरणें तब तक अच्छा वर्णन देती हैं जब तक हर
द्वारक, लेंस और अवरोध तरंगदैर्घ्य से कहीं बड़ा हो (दृश्य प्रकाश के लिए
\qty{0.4}{} से \qty{0.8}{\micro m}): \qty{1}{mm} का छिद्र किसी पुंज को
$10^{-3}$ रेडियन फैलाता है, जो प्रयोग-मेज़ पर नगण्य है; जबकि
\qty{1}{\micro m} का छिद्र उसे पूरे अर्ध-आकाश में बिखेर देता है, और किरण-चित्र
बचता ही नहीं। एकल-मोड दूरसंचार तंतु का \qty{9}{\micro m} \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} पहले ही इस
प्रतिरूप के परे है।
\end{remark}

\begin{proposition}[प्रकाश की वापसी]\label{prop:b1:rays-reflection-refraction:reversibility}
\index{उत्क्रम-वापसी का सिद्धांत}
किसी किरण का पथ इस पर निर्भर नहीं करता कि प्रकाश उसे किस दिशा में तय करता है:
यदि प्रकाश $A$ से $B$ तक किसी पथ पर जाता है, तो $B$ से चला प्रकाश उसी पथ पर
$A$ तक पहुँचता है।
\end{proposition}
\admitted

\section{परावर्तन और अपवर्तन}

\begin{definition}[आपतन]\label{def:b1:rays-reflection-refraction:incidence}
\index{आपतन तल}\index{आपतन कोण}\index{अपवर्तक पृष्ठ}
दो माध्यमों को अलग करते पृष्ठ (एक \emph{अपवर्तक पृष्ठ}) से किसी बिंदु $I$ पर
मिलती किरण उस पृष्ठ के $I$ पर खींचे \emph{अभिलंब} के साथ जो कोण बनाती है, वही
\emph{आपतन कोण} $i_1$ है। \emph{आपतन तल} में आपतित किरण और अभिलंब, दोनों
रहते हैं।
\end{definition}

\begin{theorem}[परावर्तन और अपवर्तन के नियम (स्नेल--देकार्त)]\label{thm:b1:rays-reflection-refraction:snell}
\index{परावर्तन!का नियम}\index{अपवर्तन!का नियम}\index{स्नेल का नियम}
\omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $n_1$ (आपतन की ओर) और $n_2$ वाले माध्यमों के बीच के पृष्ठ पर:
\begin{enumerate}
  \item परावर्तित और अपवर्तित किरणें \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन तल} में ही रहती हैं;
  \item परावर्तित किरण अभिलंब के साथ कोण $i_1' = i_1$ बनाती है, अभिलंब के
        दूसरी ओर;
  \item अपवर्तित किरण अभिलंब के साथ ऐसा कोण $i_2$ बनाती है कि
        \[
        n_1 \sin i_1 = n_2 \sin i_2 .
        \]
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[ये नियम कहाँ से आते हैं]\label{rem:b1:rays-reflection-refraction:fermat}
\index{फ़र्मा का सिद्धांत}
दोनों नियम \emph{फ़र्मा के सिद्धांत} से निकलते हैं: दो बिंदुओं के बीच प्रकाश
वही पथ लेता है जिस पर यात्रा का समय न्यूनतम हो। बराबर कोणों वाला परावर्तन
दर्पण को छूता सबसे छोटा टूटा हुआ पथ है; और \omterm{thm:b1:rays-reflection-refraction:snell}{स्नेल का नियम} वह पथ है जो तेज़
माध्यम में लंबे रास्ते के बदले धीमे माध्यम में छोटा रास्ता ख़रीदता है
(\cref{exo:b1:rays-reflection-refraction:12})। वर्ष 2 के खंड में दोनों नियम
किसी अंतरापृष्ठ पर प्रकाश की तरंग-प्रकृति से व्युत्पन्न किए जाते हैं।
\end{remark}

\begin{proposition}[किरण किस ओर मुड़ती है]\label{prop:b1:rays-reflection-refraction:bending}
अधिक अपवर्तक माध्यम में प्रवेश करते हुए ($n_2 > n_1$) किरण अभिलंब की
\emph{ओर} मुड़ती है ($i_2 < i_1$); और कम अपवर्तक माध्यम में प्रवेश करते हुए
उससे \emph{दूर} ($i_2 > i_1$)। अभिलंब के अनुदिश चलती किरण बिना विचलित हुए
निकल जाती है।
\end{proposition}

\begin{proof}
$\sin i_2 = (n_1/n_2)\sin i_1$, और $\sin$ अंतराल $[0, \pi/2]$ पर वर्धमान
है।
\end{proof}

\begin{example}[आभासी गहराई]\label{ex:b1:rays-reflection-refraction:depth}
\index{आभासी गहराई}
एक सिक्का गहराई $h$ पर पानी ($n = 1.33$) के नीचे पड़ा है और उसे लगभग ठीक ऊपर
से देखा जा रहा है। सिक्के से चली जो किरण सतह पर छोटे कोण $i_2$ (पानी में) पर
मिलती है, वह $i_1$ पर बाहर निकलती है, जहाँ $\sin i_1 = n\sin i_2$; छोटे कोणों
के लिए $i_1 \approx n\,i_2$। आँख निकलती हुई किरण को पीछे बढ़ा लेती है: सिक्का
गहराई $h'$ से आता जान पड़ता है, जहाँ $\tan i_1 \approx x/h'$ और
$\tan i_2 \approx x/h$ उसी सतह-बिंदु के लिए हैं जो ऊर्ध्वाधर से दूरी $x$ पर
है, अतः
$h' = h\,i_2/i_1 = h/n$। \qty{2.0}{m} गहरा तालाब \qty{1.5}{m} गहरा दिखता है;
और मछली को मछुआरा जितना है उससे $1.33$ गुना ऊँचा दिखता है।
\end{example}

\begin{theorem}[पूर्ण आंतरिक परावर्तन]\label{thm:b1:rays-reflection-refraction:tir}
\index{पूर्ण आंतरिक परावर्तन}\index{क्रांतिक कोण}
\omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $n_1$ के माध्यम से कम अपवर्तक माध्यम ($n_2 < n_1$) की ओर जाता
प्रकाश तभी अपवर्तित होता है जब $i_1 \leq i_c$ हो, जहाँ \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{क्रांतिक कोण}}
इस शर्त का पालन करता है:
\[
\sin i_c = \frac{n_2}{n_1} .
\]
$i_1 > i_c$ के लिए कोई अपवर्तित किरण नहीं रहती: पृष्ठ सारा प्रकाश परावर्तित
कर देता है --- यही \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{पूर्ण आंतरिक परावर्तन}} है।
\end{theorem}

\begin{proof}
\omterm{thm:b1:rays-reflection-refraction:snell}{स्नेल का नियम} $\sin i_2 = (n_1/n_2)\sin i_1 > \sin i_1$ माँगता है; और यह $1$
से बड़ा हो जाता है जैसे ही $\sin i_1 > n_2/n_1$ हो जाए, जिसके बाद ऐसा कोई कोण
$i_2$ बचता ही नहीं। $i_1 = i_c$ पर अपवर्तित किरण पृष्ठ को स्पर्श करती हुई
निकलती है ($i_2 = 90^\circ$)। तब \emph{सारी} ऊर्जा परावर्तित हो जाती है ---
यह तीव्रताओं का कथन है, जिसे \omterm{def:b1:rays-reflection-refraction:ray}{ज्यामितीय प्रकाशिकी} सँभालती नहीं; वर्ष 2 के खंड
का तरंग-विवेचन इसकी पुष्टि करता है।
\end{proof}

\begin{example}[हीरे और तैराक]\label{ex:b1:rays-reflection-refraction:tir}
काँच--हवा: $\sin i_c = 1/1.5$, $i_c = 41.8^\circ$, और इसीलिए
$45^\circ$ वाला प्रिज़्म पूर्ण परावर्तन करता है तथा दूरबीनों में दर्पण की जगह
ले लेता है। हीरा--हवा: $i_c = 24.4^\circ$; तराशा हुआ हीरा अपने ऊपरी फलक से
घुसे प्रकाश को क़ैद कर लेता है और उसका लगभग सारा भाग ऊपर से ही लौटा देता है
--- उसकी चमक पूर्ण परावर्तन है, उसकी आग वर्ण-विक्षेपण। पानी--हवा:
$i_c = 48.8^\circ$; ऊपर देखता गोताख़ोर पूरे आकाश को अर्ध-कोण $48.8^\circ$ के
शंकु के भीतर देखता है (``स्नेल की खिड़की'') और उसके बाहर तली का परावर्तित
प्रतिबिंब।
\end{example}

\section{प्रकाशिक तंतु}

\begin{definition}[सोपान-अपवर्तनांक तंतु]\label{def:b1:rays-reflection-refraction:fiber}
\index{प्रकाशिक तंतु}\index{क्रोड}\index{आवरण}\index{संख्यात्मक द्वारक}
\emph{सोपान-अपवर्तनांक प्रकाशिक तंतु} \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $n_1$ के काँच का एक बेलनाकार
\emph{क्रोड} है, जिस पर थोड़े कम \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $n_2$ का \emph{आवरण} चढ़ा होता
है। अक्ष के चारों ओर अर्ध-कोण $\theta_a$ के शंकु के भीतर क्रोड में घुसा
प्रकाश क्रोड--आवरण पृष्ठ पर लगातार होते पूर्ण आंतरिक परावर्तनों से निर्देशित
हो जाता है। \emph{संख्यात्मक द्वारक} है
$\mathrm{NA} = \sin\theta_a$।
\end{definition}

\begin{proposition}[ग्राह्यता शंकु और अंतरमोड प्रसार]\label{prop:b1:rays-reflection-refraction:fiber}
\index{ग्राह्यता कोण}\index{अंतरमोड विक्षेपण}
ऐसे तंतु के लिए जिसका प्रवेश-फलक हवा में हो,
\[
\sin\theta_a = \sqrt{n_1^2 - n_2^2} .
\]
लंबाई $L$ पर, अक्ष के अनुदिश चलती निर्देशित किरण और सबसे तिरछी निर्देशित किरण
के बीच पहुँचने में जो विलंब पड़ता है वह है
\[
\Delta t = \frac{L\,n_1}{c}\left(\frac{n_1}{n_2} - 1\right)
\]
(\emph{\omterm{prop:b1:rays-reflection-refraction:fiber}{अंतरमोड विक्षेपण}})।
\end{proposition}

\begin{proof}
अक्ष से कोण $\theta$ पर घुसती किरण $\theta'$ पर अपवर्तित होती है, जहाँ
$\sin\theta = n_1\sin\theta'$, और वह \omterm{def:b1:rays-reflection-refraction:fiber}{आवरण} से \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन कोण}
$i = 90^\circ - \theta'$ पर मिलती है। निर्देशन के लिए
$\sin i \geq n_2/n_1$ चाहिए, यानी $\cos\theta' \geq n_2/n_1$, यानी
$\sin\theta' \leq
\sqrt{1 - n_2^2/n_1^2}$; अब $n_1$ से गुणा कीजिए। अक्षीय
किरण $L$ को चाल $c/n_1$ से तय करती है: $t_0 = Ln_1/c$। सबसे तिरछी किरण कोण
$\theta'_{\max}$ पर टेढ़ी-मेढ़ी चलती है, जहाँ
$\cos\theta'_{\max} = n_2/n_1$; उसका पथ
$L/\cos\theta'_{\max} = L n_1/n_2$ है, अतः $t_1 = L n_1^2/(n_2 c)$ और
$\Delta t = t_1 - t_0$।
\end{proof}

\section{समतल दर्पण}

\begin{proposition}[समतल दर्पण में प्रतिबिंब]\label{prop:b1:rays-reflection-refraction:mirror}
\index{समतल दर्पण}\index{आभासी प्रतिबिंब}\index{बिंदुसंगतता}
किसी बिंदु $A$ से चलकर \omterm{prop:b1:rays-reflection-refraction:mirror}{समतल दर्पण} से परावर्तित सभी किरणें उस बिंदु $A'$ से आती
जान पड़ती हैं जो दर्पण के तल के सापेक्ष $A$ का प्रतिबिंबित बिंदु है: $A'$ ही
$A$ का \emph{प्रतिबिंब} है। यह \emph{आभासी} है (इसमें से कोई प्रकाश नहीं
गुज़रता) और दर्पण \emph{कठोरता से बिंदुसंगत} है: हर बिंदु का ठीक-ठीक बिंदु
प्रतिबिंब बनता है।
\end{proposition}

\begin{proof}
$A$ से चली जो किरण दर्पण से बिंदु $I$ पर \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन कोण} $i$ के साथ टकराती है, वह
अभिलंब के दूसरी ओर $i$ पर ही लौटती है। परावर्तित रेखा और $I$ पर खींचा अभिलंब
कोण $i$ बनाते हैं, और उतना ही कोण रेखा $IA$ भी बनाती है; दर्पण-तल के सापेक्ष
प्रतिबिंबन लेने पर परावर्तित रेखा ठीक रेखा $IA$ का दर्पण-प्रतिबिंब निकलती है,
अतः वह प्रतिबिंबित बिंदु $A'$ से होकर जाती है --- और यह हर
$I$ के लिए सच है।
\end{proof}

\begin{proposition}[दर्पण को घुमाना]\label{prop:b1:rays-reflection-refraction:rotation}
\omterm{prop:b1:rays-reflection-refraction:mirror}{समतल दर्पण} को उसके अपने तल में स्थित, \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन तल} के लंबवत किसी अक्ष के परितः कोण
$\alpha$ से घुमाने पर परावर्तित किरण $2\alpha$ से घूम जाती
है।
\end{proposition}

\begin{proof}
अभिलंब $\alpha$ से घूमता है, अतः \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन कोण} $i$ से बदलकर
$i + \alpha$ हो जाता है; नए अभिलंब के दूसरी ओर $i + \alpha$ पर स्थित
परावर्तित किरण अपनी पुरानी दिशा से $\alpha$ (अभिलंब) $+ \alpha$ (कोण) $=
2\alpha$ हट चुकी होती है।
\end{proof}

\begin{example}[धारामापी का उत्तोलक]\label{ex:b1:rays-reflection-refraction:galvanometer}
धारामापी की कुंडली पर चिपका एक नन्हा दर्पण, जिस पर दीपक की रोशनी पड़ती है और
जो \qty{2}{m} दूर रखे पैमाने पर एक धब्बा फेंकता है: कुंडली के
$\qty{1}{mrad}$ घूमने पर धब्बा $2 \times 10^{-3} \times
\qty{2}{m} = \qty{4}{mm}$ खिसक जाता है --- यह एक प्रकाशिक उत्तोलक है जो बिना
घर्षण के आवर्धन करता है, और यही दर्पण-धारामापी तथा लेज़र क्रमवीक्षक का
सिद्धांत है।
\end{example}

\section{प्रिज़्म}

\begin{proposition}[प्रिज़्म के सूत्र]\label{prop:b1:rays-reflection-refraction:prism}
\index{प्रिज़्म}\index{विचलन}\index{न्यूनतम विचलन}
हवा में रखे शीर्ष कोण $A$ और \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $n$ के किसी प्रिज़्म को, कोर के लंबवत
तल में, कोणों $i$ (प्रवेश), $r$, $r'$ (भीतर) और $i'$ (निर्गम) के साथ, जो
अभिलंबों से नापे गए हैं, पार करती किरण के लिए
\[
\sin i = n \sin r, \qquad r + r' = A, \qquad \sin i' = n \sin r',
\]
और वह $D = i + i' - A$ से विचलित होती है। विचलन तब न्यूनतम होता है जब पथ
सममित हो, यानी $i = i'$, $r = r' = A/2$; तब
\[
n = \frac{\sin\frac{A + D_{\min}}{2}}{\sin\frac{A}{2}} .
\]
\end{proposition}

\begin{proof}
हर फलक पर \omterm{thm:b1:rays-reflection-refraction:snell}{स्नेल का नियम}; और $r + r' = A$ इसलिए कि दोनों अभिलंब आपस में कोण
$A$ बनाते हैं (शीर्ष, दोनों आपतन बिंदुओं और अभिलंबों के प्रतिच्छेद से बनने
वाला चतुर्भुज)। कुल विचलन $(i - r) + (i' - r') = i + i' - A$ है। प्रकाश की
वापसी के कारण विचलन $i$ का और $i'$ का एक ही फलन है; यदि न्यूनतम
$i_0 \neq i_0'$ पर आता, तो वह उलटे आपतन $i_0'$ पर भी आता और दो न्यूनतम बन
जाते, जिसकी अनुमति अपने न्यूनतम के दोनों ओर $D(i)$ का एकदिष्ट व्यवहार नहीं
देता: अतः न्यूनतम $i = i'$ पर ही है।
तब $r = r' = A/2$ और $D_{\min} = 2i - A$; अब $i = (A +
D_{\min})/2$ को $\sin i = n\sin(A/2)$ में रखने पर सूत्र मिल जाता है।
\end{proof}

विचलन} ही है --- प्रिज़्म को आगे-पीछे घुमाकर पाया गया यह तीखा न्यूनतम छोटी-मोटी
ग़लत सज्जा के प्रति असंवेदनशील रहता है।
\end{example}

\begin{remark}[पतले प्रिज़्म और वर्णक्रमदर्शी]\label{rem:b1:rays-reflection-refraction:thin}
छोटे $A$ और छोटे आपतन के लिए ज्याएँ कोणों के बराबर हो जाती हैं और
$D \approx (n - 1)A$ रह जाता है, जो $i$ से स्वतंत्र है: पतला प्रिज़्म हर किरण
को एक ही कोण से विचलित करता है --- काँच की कोई फाँक ऐसा ``प्रिज़्म'' है जो
प्रकाश को मोड़ देता है, और आगे चलकर लेंस ऐसी ही फाँकों की चट्टी सिद्ध होगा।
चूँकि $n$ $\lambda$ पर निर्भर करता है, इसलिए विचलन भी निर्भर करता है: प्रिज़्म
श्वेत प्रकाश को वर्णक्रम में फैला देता है (नीला सबसे अधिक विचलित), और यही
प्रिज़्म वर्णक्रमदर्शी का हृदय है।
\end{remark}

\section{इंद्रधनुष}

\begin{example}[देकार्त का इंद्रधनुष]\label{ex:b1:rays-reflection-refraction:rainbow}
\index{इंद्रधनुष}
सूर्य की एक किरण वर्षा की गोलाकार बूँद में \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन कोण} $i$ पर घुसती है, अपवर्तित
होती है ($\sin i = n\sin r$), पिछली सतह पर एक बार परावर्तित होती है, और दूसरे
अपवर्तन के बाद बाहर निकल जाती है। हर अपवर्तन उसे $i - r$ से विचलित करता है
और आंतरिक परावर्तन $\pi - 2r$ से: कुल विचलन
$D(i) = \pi + 2i - 4r$। किरणें हर $i$ पर बूँद को भरती हैं; प्रेक्षक को सबसे
अधिक चमक वहाँ दिखती है जहाँ बहुत-सी किरणें लगभग एक ही दिशा में निकलती हैं,
यानी जहाँ $D$ अचर-कोटि का हो: $\dd D/\dd i = 2 - 4\,\dd r/\dd i
= 0$। स्नेल के नियम का अवकलन करने पर $\cos i = n\cos r\,\dd r/\dd i$, अतः
शर्त $2\cos i = n\cos r$ बन जाती है; वर्ग करके और
$n^2\cos^2 r = n^2 - \sin^2 i = n^2 - 1 + \cos^2 i$ का उपयोग करके मिलता है
\[
\cos^2 i = \frac{n^2 - 1}{3} .
\]
पानी के लिए, $n = 1.333$: $i = 59.4^\circ$, $r = 40.2^\circ$,
$D_{\min} = 137.9^\circ$। प्रकाश सूर्य की ओर लौटता है, सूर्य-विपरीत दिशा से
$180^\circ - 137.9^\circ = 42^\circ$ पर: इस प्रकार इंद्रधनुष प्रेक्षक के सिर
की छाया के चारों ओर अर्ध-कोण $42^\circ$ का शंकु है। लाल ($n = 1.331$)
$42.4^\circ$ पर बैठता है, बैंगनी ($n = 1.343$) $40.6^\circ$ पर --- लाल बाहर,
बैंगनी भीतर।
\end{example}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:1}
पानी ($n = 1.33$), काँच ($1.50$) और हीरे ($2.42$) में प्रकाश की चाल निकालिए।
\omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $1.48$ के तंतु की \qty{1.0}{km} लंबाई पार करने में प्रकाश को कितना
समय लगता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:2}
हवा में चलती एक किरण पानी ($n = 1.33$) से अभिलंब से $30^\circ$ पर, फिर
$60^\circ$ पर मिलती है: अपवर्तन कोण निकालिए। पानी में चलती एक किरण सतह से
$50^\circ$ पर मिलती है: क्या होता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:3}
काँच--हवा ($n = 1.50$), हीरा--हवा ($2.42$) और पानी--हवा ($1.33$) के लिए
\omterm{thm:b1:rays-reflection-refraction:tir}{क्रांतिक कोण} निकालिए। गोताख़ोर को दिखती आकाश की चकती का कोणीय व्यास कितना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:4}
\qty{1.70}{m} लंबा एक व्यक्ति, जिसकी आँखें सिर के ऊपरी सिरे से \qty{10}{cm}
नीचे हैं, दीवार पर लगे ऊर्ध्वाधर दर्पण के सामने खड़ा है। वह सबसे छोटा दर्पण
निकालिए (ऊँचाई, और फ़र्श से उसके किनारों की स्थिति) जिसमें पूरा शरीर दिखाई दे।
क्या उत्तर दर्पण से दूरी पर निर्भर करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:5}
एक सिक्का तालाब में \qty{1.2}{m} गहराई पर पड़ा है; ऊपर से देखने वाले को वह
किस गहराई पर दिखेगा? किसी मछुआरे की आँखें पानी से \qty{1.8}{m} ऊपर हैं: ऊपर
देखती मछली को वे कितनी ऊँची दिखेंगी? दोनों का औचित्य छोटे-कोण वाली किरणों से
सिद्ध कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:6}
किसी प्रिज़्म के लिए $A = 60^\circ$ और $n = 1.52$ है। \omterm{prop:b1:rays-reflection-refraction:prism}{न्यूनतम विचलन}
निकालिए। दिखाइए कि किरण दूसरे फलक से तभी निकल सकती है जब
$A \leq 2 i_c$ हो, और जाँचिए कि यह प्रिज़्म इस शर्त पर खरा उतरता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:7}
किसी सोपान-अपवर्तनांक तंतु में $n_1 = 1.50$, $n_2 = 1.48$ है। उसका
\omterm{def:b1:rays-reflection-refraction:fiber}{संख्यात्मक द्वारक}, उसका \omterm{prop:b1:rays-reflection-refraction:fiber}{ग्राह्यता कोण}, प्रति किलोमीटर अंतरमोड विलंब, और
\qty{1}{km} पर उससे मिलने वाली अधिकतम बिट दर निकालिए
($B_{\max} \approx 1/(2\Delta t)$ लीजिए)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:8}
शीर्ष कोण $10^\circ$ के किसी पतले प्रिज़्म के लिए लाल का $n = 1.510$ और नीले
का $1.530$ है। हर रंग का विचलन और उससे बनने वाले वर्णक्रम की कोणीय चौड़ाई
निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:9}
एक किरण समांतर फलकों वाली काँच की पट्टिका ($n = 1.50$, मोटाई
$e = \qty{2.0}{cm}$) से $45^\circ$ पर मिलती है। दिखाइए कि वह अपनी आपतन दिशा
के समांतर ही निकलती है, बग़ल में $d = e\,\sin(i - r)/\cos r$ खिसककर, और
$d$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:10}
द्वितीयक इंद्रधनुष उन किरणों से बनता है जो बूँद के भीतर \emph{दो बार}
परावर्तित होती हैं। दिखाइए कि $D = 2\pi + 2i - 6r$ है, कि अचर-कोटि विचलन
$\cos^2 i = (n^2 - 1)/8$ का पालन करता है, और $n = 1.333$ के लिए सूर्य-विपरीत
दिशा से द्वितीयक धनुष का कोण निकालिए। वह प्राथमिक के सापेक्ष कहाँ है, और उसके
रंग किस क्रम में हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:11}
गरम सड़क पर डामर के ठीक ऊपर की हवा का \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक}
$n_{\text{गरम}} = 1.00026$ है और उसके ऊपर की ठंडी हवा का $n_{\text{ठंडी}} =
1.00029$। दोनों परतों को एक तीखे अंतरापृष्ठ के रूप में लीजिए: वह क्रांतिक
स्पर्श कोण (सड़क से नापा हुआ) निकालिए जिससे नीचे आकाश से आती किरण पूरी तरह ऊपर
परावर्तित हो जाती है। किसी चालक की आँख सड़क से \qty{1.2}{m} ऊपर है: कितनी दूरी
के परे सड़क पानी जैसी दिखने लगती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:12}
बिंदु $A$ (किसी अंतरापृष्ठ से ऊँचाई $a$ पर) और $B$ (उसके नीचे गहराई $b$ पर)
क्षैतिज रूप से $d$ से अलग हैं; प्रकाश ऊपर $v_1$ चाल से और नीचे $v_2$ चाल से
चलता है। क्षैतिज स्थिति $x$ वाले अंतरापृष्ठ-बिंदु से होकर जाते टूटे पथ का
यात्रा-समय लिखिए, और दिखाइए कि समय ठीक तभी अचर-कोटि का होता है जब
$\sin i_1/v_1 = \sin i_2/v_2$ --- यानी \omterm{thm:b1:rays-reflection-refraction:snell}{स्नेल का नियम}।
\end{exercise}

\section{समस्या: प्रकाशिक तंतु की कड़ी}

\begin{problem}[{सप्ताहांत समस्या --- समुद्र के नीचे काँच का एक धागा: प्रकाश
का एक स्पंद कितनी दूर और कितनी तेज़ी से एक बिट ढो सकता है, इससे पहले कि
ज्यामिति, काँच और विक्षेपण उसे धुँधला कर दें}]
\label{pb:b1:rays-reflection-refraction:1}
एक कड़ी में सोपान-अपवर्तनांक तंतु काम में लिया जाता है, जिसका \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} $n_1 = 1.480$,
\omterm{def:b1:rays-reflection-refraction:fiber}{आवरण} $n_2 = 1.465$, क्रोड-त्रिज्या $a = \qty{25}{\micro m}$ और तरंगदैर्घ्य
$\lambda_0 = \qty{1550}{nm}$ है। आँकड़े: $c = \qty{3.00e8}{m/s}$।

\medskip
\noindent\textbf{भाग I --- काँच में प्रकाश।}
\begin{enumerate}
  \item \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} परिभाषित कीजिए और \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} में प्रकाश की चाल निकालिए।
  \item अक्ष के अनुदिश \qty{50}{km} चलने में किसी स्पंद को कितना समय लगता
        है?
  \item प्रकाश की आवृत्ति और \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} के भीतर उसका तरंगदैर्घ्य निकालिए।
  \item \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} पर कम \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} वाले काँच का \omterm{def:b1:rays-reflection-refraction:fiber}{आवरण} क्यों चढ़ाया जाता है, उसे
        हवा में नंगा क्यों नहीं छोड़ दिया जाता, जबकि उससे \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} की छलाँग
        और बड़ी मिलती?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- निर्देशन।}
\begin{enumerate}[resume]
  \item क्रोड--आवरण पृष्ठ पर \omterm{thm:b1:rays-reflection-refraction:tir}{क्रांतिक कोण} निकालिए।
  \item इससे वह अधिकतम कोण निकालिए जो कोई निर्देशित किरण अक्ष के साथ बना
        सकती है।
  \item एक किरण हवा से समतल सिरे-फलक में अक्ष से कोण $\theta$ पर घुसती है।
        दिखाइए कि वह तभी निर्देशित होती है जब $\sin\theta \leq
        \sqrt{n_1^2 - n_2^2}$ हो, और \omterm{prop:b1:rays-reflection-refraction:fiber}{ग्राह्यता कोण}
        $\theta_a$ निकालिए।
  \item \omterm{def:b1:rays-reflection-refraction:fiber}{संख्यात्मक द्वारक} दीजिए। \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} की बड़ी छलाँग से तंतु में प्रकाश
        डालना आसान क्यों हो जाता है?
  \item तंतु को त्रिज्या $R_b$ के वृत्त के अनुदिश मोड़ा जाता है। \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} के
        भीतरी किनारे पर अक्ष के समांतर चलती किसी किरण पर विचार कीजिए:
        दिखाइए कि वह बाहरी दीवार से \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन कोण} $i$ पर मिलती है, जहाँ
        $\sin i = R_b/(R_b + 2a)$, और वह न्यूनतम $R_b$ निकालिए जिसके लिए
        वह अब भी निर्देशित रहती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{prop:b1:rays-reflection-refraction:fiber}{अंतरमोड विक्षेपण}।}
\begin{enumerate}[resume]
  \item लंबाई $L$ पर अक्षीय किरण और सबसे तिरछी निर्देशित किरण के यात्रा-समय
        व्यक्त कीजिए।
  \item इनसे प्रति किलोमीटर प्रसार $\Delta t$ निकालिए, और
        $L = \qty{2}{km}$ पर उसका मान।
  \item एक बिट एक छोटा स्पंद है; दो क्रमागत स्पंदों को अलग-अलग पहचानने के
        लिए उनके बीच कम से कम $2\Delta t$ का अंतर रहना चाहिए। \qty{2}{km}
        पर अधिकतम बिट दर का आकलन कीजिए।
  \item दिखाइए कि इस तंतु के लिए (बिट दर) $\times$ (लंबाई) एक नियतांक है,
        और उसे \unit{Mbit.s^{-1}.km} में निकालिए।
  \item क्रमिक-अपवर्तनांक तंतु में \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} अक्ष से बाहर की ओर धीरे-धीरे
        घटता है। गुणात्मक रूप से समझाइए कि इससे अंतरमोड प्रसार क्यों घट
        जाता है।
  \item एकल-मोड तंतु के \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} का व्यास लगभग \qty{9}{\micro m} होता है।
        उसकी तुलना $\lambda_0$ से कीजिए और समझाइए कि किरण-प्रतिरूप उसका
        वर्णन क्यों नहीं कर सकता; इससे क्या मिलता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- क्षीणन और वर्णिक विक्षेपण।}
यह तंतु \qty{1550}{nm} पर शक्ति को $\alpha = \qty{0.20}{dB/km}$ की दर से
क्षीण करता है: $P(L) = P_0\,10^{-\alpha L/10}$। प्रेषक
$P_0 = \qty{1.0}{mW}$ भेजता है; अभिग्राही को कम से कम
$P_{\min} = \qty{1.0}{\micro W}$ चाहिए। चूँकि $n$ $\lambda$ पर निर्भर है,
दो तरंगदैर्घ्य थोड़ी भिन्न चालों से चलते हैं: वर्णक्रमी चौड़ाई
$\Delta\lambda$ का कोई स्रोत स्पंद को $\Delta t_c = D\,\Delta\lambda\,L$
जितना फैला देता है, जहाँ $D = \qty{17}{ps/(nm.km)}$।
\begin{enumerate}[resume]
  \item \qty{100}{km} के बाद शक्ति का कितना अंश बचता है?
  \item प्रवर्धक की ज़रूरत पड़ने से पहले की अधिकतम लंबाई निकालिए।
  \item प्रवर्धक हर \qty{100}{km} पर रखे जाते हैं: इससे संयोजकों और जोड़ों
        के लिए कितना अवकाश बचता है, dB में और शक्ति-अनुपात के रूप में?
  \item \qty{100}{km} पर $\Delta t_c$ निकालिए, किसी LED स्रोत
        ($\Delta\lambda = \qty{40}{nm}$) के लिए और किसी लेज़र
        ($\Delta\lambda = \qty{0.10}{nm}$) के लिए; इनसे उस लंबाई पर हर एक की अधिकतम बिट दर
        निकालिए (कसौटी ऊपर वाली ही)।
  \item \qty{850}{nm} पर सिलिका \qty{2.0}{dB/km} क्षीण करती है। वही
        शक्ति-बजट वहाँ कितनी दूरी की अनुमति देता? \qty{1550}{nm} ही
        दूरसंचार का तरंगदैर्घ्य क्यों है?
  \item \qty{1}{Gbit/s} के लिए ज़रूरी आवृत्तियों पर ताँबे की समाक्ष केबल
        लगभग \qty{30}{dB/km} क्षीण करती है। उसके प्रवर्धकों के अंतराल की
        तुलना तंतु के अंतराल से कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- पूरी कड़ी।}
\begin{enumerate}[resume]
  \item \qty{2}{km} पर सोपान-अपवर्तनांक तंतु के लिए बिट दर को कौन सीमित
        करता है: अंतरमोड प्रसार, वर्णिक प्रसार (लेज़र स्रोत) या क्षीणन?
        सीमित करने वाली बिट दर दीजिए।
  \item लेज़र स्रोत के साथ \qty{100}{km} पर एकल-मोड तंतु के लिए: कौन सीमित
        करता है, और किस बिट दर पर?
  \item एकल-मोड कड़ी \qty{0.8}{nm} के अंतर पर रखे $80$ लेज़र ढोती है
        (तरंगदैर्घ्य बहुसंकेतन), और हर एक पिछले प्रश्न वाली बिट दर पर।
        कुल क्षमता?
  \item सोपान-अपवर्तनांक कड़ी का और एकल-मोड की एक चैनल का बिट-दर--लंबाई
        गुणनफल बनाइए; एकल-मोड तंतु कितने गुना जीतता है, और इसके पीछे कौन-सा
        एक भौतिक तथ्य है?
\end{enumerate}
\end{problem}
