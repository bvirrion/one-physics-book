समस्याओं को एक पंक्ति के बीजगणित में बदल देता है और किसी \omterm{def:b1:work-and-energy:turning}{स्थितिज कूप} के

\section{शक्ति और कार्य}

\begin{definition}[किसी बल की शक्ति और कार्य]\label{def:b1:work-and-energy:work}
\index{शक्ति (बल की)}\index{कार्य}
किसी ऐसे बल $\vect F$ की \emph{शक्ति}, जो $\vect v$ वेग से चलते किसी बिंदु
पर लग रहा हो, है
\[
\mathcal P = \vect F\cdot\vect v \qquad (\text{वाट}),
\]
और समय $t_1$ तथा $t_2$ के बीच (स्थितियाँ $A$ और $B$) उसका \emph{कार्य}
संचित शक्ति है,
\[
W_{A\to B} = \int_{t_1}^{t_2}\vect F\cdot\vect v\,\dd t
= \int_A^B \vect F\cdot\dd\vect\ell \qquad (\text{जूल}),
\]
जो \emph{अल्प कार्यों} $\delta W = \vect F\cdot\dd\vect\ell$ का योग है,
जिन्हें \omterm{def:b1:point-kinematics:frame}{गतिपथ} के अल्प विस्थापनों $\dd\vect\ell = \vect v\,\dd t$ के अनुदिश
जोड़ा जाता है। कार्य धनात्मक है यदि बल गति की दिशा में धकेले
(\emph{प्रेरक}), ऋणात्मक यदि उसके विरुद्ध हो (\emph{प्रतिरोधी}), और शून्य
यदि वह लंब हो।
\end{definition}

\begin{proposition}[जिन कार्यों से सामना होता है]\label{prop:b1:work-and-energy:works}
\begin{itemize}
  \item अचर बल: $W_{A\to B} = \vect F\cdot\vect{AB}$, पथ चाहे कोई भी हो।
        भार: $W = mg(z_A - z_B)$, जहाँ $z$ ऊपर की ओर गिना जाता है।
  \item $k$ दृढ़ता की स्प्रिंग, वृद्धि $x_A$ से $x_B$ तक:
        $W = \tfrac12 k(x_A^2 - x_B^2)$।
  \item किसी स्थिर पृष्ठ की \omterm{prop:b1:newton-dynamics:friction}{अभिलंब प्रतिक्रिया}, और उस पर लंब चलते पिंड पर
        डोरी का तनाव: $W = 0$ (बल $\perp$ विस्थापन)।
  \item गतिज घर्षण और कर्षण: सदा $W < 0$ (बल सापेक्ष गति का विरोध करता है),
        और यह पथ की लंबाई पर निर्भर करता है।
\end{itemize}
\end{proposition}

\begin{proof}
अचर $\vect F$: $\int\vect F\cdot\dd\vect\ell = \vect F\cdot\int\dd\vect\ell
= \vect F\cdot\vect{AB}$; और $\vect F = -mg\vect e_z$ के साथ $\vect F\cdot\vect{AB}
= -mg(z_B - z_A)$। स्प्रिंग: $\vect F\cdot\dd\vect\ell = -kx\,\dd x$, और
$\int_{x_A}^{x_B}-kx\,\dd x = \tfrac12 k(x_A^2 - x_B^2)$। घर्षण: $\vect T
= -\mu N\vect v/v$, अतः $\vect T\cdot\vect v = -\mu Nv < 0$।
\end{proof}

\section{गतिज-ऊर्जा प्रमेय}

\begin{theorem}[गतिज-ऊर्जा और शक्ति प्रमेय]\label{thm:b1:work-and-energy:ke}
\index{गतिज ऊर्जा}\index{गतिज-ऊर्जा प्रमेय}
किसी \omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} में किसी \omterm{def:b1:point-kinematics:frame}{बिंदु कण} की \emph{\omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा}}
$E_k = \tfrac12 mv^2$ इस नियम का पालन करती है
\[
\frac{\dd E_k}{\dd t} = \sum\mathcal P_i = \Big(\sum\vect F_i\Big)\cdot\vect v,
\qquad
E_k(B) - E_k(A) = \sum_i W_{i,A\to B}:
\]
अर्थात् दो स्थितियों के बीच \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} का परिवर्तन रास्ते भर लगे सभी बलों
के कुल \omterm{def:b1:work-and-energy:work}{कार्य} के बराबर होता है।
\end{theorem}

\begin{proof}
न्यूटन के दूसरे नियम का $\vect v$ के साथ अदिश गुणन कीजिए: $m\vect a\cdot\vect v =
\tfrac{\dd}{\dd t}(\tfrac12 m\vect v\cdot\vect v) = \dd E_k/\dd t$, जबकि
$(\sum\vect F_i)\cdot\vect v = \sum\mathcal P_i$। अब समय पर समाकलन कीजिए।
\end{proof}

\begin{example}[ब्रेक-दूरी]\label{ex:b1:work-and-energy:braking}
\qty{100}{km/h} पर चलती \qty{1000}{kg} की कोई कार ($E_k = \qty{386}{kJ}$),
जिसे अचर \qty{7.0}{kN} बल से रोका जाए, $d = E_k/F = \qty{55}{m}$ के बाद
रुक जाती है: यह दूरी चाल के \emph{वर्ग} की तरह बढ़ती है। चूँकि अधिकतम
ब्रेक-बल $\mu_smg$ है (\cref{ch:b1:newton-dynamics}), रुकने की न्यूनतम दूरी
$v^2/2\mu_sg$ होती है, कार का द्रव्यमान चाहे जो हो --- और गीली सड़क पर,
जहाँ $\mu_s$ आधा रह जाता है, वह दुगुनी हो जाती है।
\end{example}

\section{संरक्षी बल और स्थितिज ऊर्जा}

\begin{definition}[संरक्षी बल; स्थितिज ऊर्जा]\label{def:b1:work-and-energy:conservative}
\index{संरक्षी बल}\index{स्थितिज ऊर्जा}
कोई बल \emph{संरक्षी} है यदि दो बिंदुओं के बीच उसका \omterm{def:b1:work-and-energy:work}{कार्य} अपनाए गए पथ पर
निर्भर न करे। समान रूप से, स्थिति का ऐसा कोई फलन होता है, \emph{स्थितिज
ऊर्जा} $E_p$, कि
\[
W_{A\to B} = E_p(A) - E_p(B) = -\Delta E_p ,
\qquad \delta W = -\dd E_p ,
\]
जहाँ $E_p$ किसी योज्य नियतांक तक (यानी मूल के चुनाव तक) परिभाषित है।
\end{definition}

\begin{proposition}[स्थितिज ऊर्जा से बल]\label{prop:b1:work-and-energy:gradient}
\index{प्रवणता}
कोई \omterm{def:b1:work-and-energy:conservative}{संरक्षी बल} अपनी \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} से व्युत्पन्न होता है:
\[
\vect F = -\overrightarrow{\operatorname{grad}}\,E_p
= -\frac{\partial E_p}{\partial x}\,\vect e_x
- \frac{\partial E_p}{\partial y}\,\vect e_y
- \frac{\partial E_p}{\partial z}\,\vect e_z ,
\]
और एक विमा में $F_x = -\dd E_p/\dd x$: बल घटती \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} की ओर इंगित
करता है, यानी $E_p$ के आलेख पर ``ढलान से नीचे''।
\end{proposition}

\begin{proof}
हर विस्थापन के लिए $\delta W = F_x\dd x + F_y\dd y + F_z\dd z = -\dd E_p = -(\partial_xE_p\,\dd x
+ \partial_yE_p\,\dd y + \partial_zE_p\,\dd z)$: अब गुणांकों की तुलना
कीजिए (कई चरों वाले किसी फलन का अवकल, गणित के पाठ्यक्रम से)।
\end{proof}

\begin{proposition}[सामान्य स्थितिज ऊर्जाएँ]\label{prop:b1:work-and-energy:eps}
\index{स्थितिज ऊर्जा!गुरुत्वीय}\index{स्थितिज ऊर्जा!प्रत्यास्थ}
\begin{itemize}
  \item भार: $E_p = mgz + \text{नियतांक}$ ($z$ ऊपर की ओर)।
  \item स्प्रिंग: $E_p = \tfrac12 k(\ell - \ell_0)^2 + \text{नियतांक}$।
  \item $M$ द्रव्यमान का न्यूटनी गुरुत्वाकर्षण: $E_p = -GMm/r + \text{नियतांक}$,
        जहाँ नियतांक प्रायः ऐसा चुना जाता है कि अनंत पर $E_p \to 0$ हो।
  \item किसी आवेश $q$ पर, जो विभव $V$ में हो, लगता स्थिरवैद्युत बल:
        $E_p = qV$ (\cref{ch:b1:potential-capacitors})।
\end{itemize}
गतिज घर्षण और तरल कर्षण संरक्षी \emph{नहीं} हैं (उनका \omterm{def:b1:work-and-energy:work}{कार्य} पथ पर निर्भर
करता है, और सदा ऋणात्मक होता है)।
\end{proposition}

\begin{proof}
हर एक \cref{prop:b1:work-and-energy:works} के कार्यों से पढ़ा जाता है:
$W = -\Delta E_p$। गुरुत्व के लिए $\vect F = -GMm/r^2\,\vect e_r$ और
$\delta W = -GMm\,\dd r/r^2 = -\dd(-GMm/r)$।
\end{proof}

\section{यांत्रिक ऊर्जा}

\begin{theorem}[यांत्रिक ऊर्जा]\label{thm:b1:work-and-energy:em}
\index{यांत्रिक ऊर्जा}\index{यांत्रिक ऊर्जा का संरक्षण}
मान लीजिए $E_m = E_k + E_p$ \emph{\omterm{thm:b1:work-and-energy:em}{यांत्रिक ऊर्जा}} है, जहाँ $E_p$ संरक्षी
बलों की स्थितिज ऊर्जाओं का योग है। तब
\[
\Delta E_m = W_{\text{असं}} ,
\]
यानी असंरक्षी बलों का कुल \omterm{def:b1:work-and-energy:work}{कार्य}। यदि वे कोई \omterm{def:b1:work-and-energy:work}{कार्य} न करें (घर्षण न हो, या
प्रतिबंध-बल गति पर लंब हों), तो $E_m$ \emph{संरक्षित} रहती है: गति
\emph{संरक्षी} होती है।
\end{theorem}

\begin{proof}
$\Delta E_k = W_{\text{सं}} + W_{\text{असं}} = -\Delta E_p + W_{\text{असं}}$।
\end{proof}

\begin{method}[ऊर्जा का प्रयोग]\label{met:b1:work-and-energy:method}
जब कोई समस्या \emph{किसी स्थिति पर चाल} पूछे (समय नहीं, बल नहीं), तो पहले
ऊर्जा आज़माइए:
\begin{enumerate}
  \item बलों की सूची बनाइए; देखिए कौन \omterm{def:b1:work-and-energy:work}{कार्य} करते हैं और कौन संरक्षी हैं;
  \item दोनों स्थितियों पर $E_m$ लिखिए, और $E_p$ के लिए एक स्पष्ट मूल तय
        कीजिए;
  \item दोनों को बराबर कीजिए (घर्षण लग रहा हो तो $W_{\text{असं}}$ जोड़कर)
        और अज्ञात चाल या स्थिति के लिए हल कीजिए।
\end{enumerate}
ऊर्जा कभी समय या प्रतिबंध-बल नहीं देती; उनके लिए \omterm{thm:b1:newton-dynamics:laws}{न्यूटन के नियम} पर लौटिए
--- प्रायः उसी चाल के साथ जो ऊर्जा ने अभी-अभी दी है।
\end{method}

\begin{example}[पाश का चाल-नियम]\label{ex:b1:work-and-energy:loop}
\cref{pb:b1:point-kinematics:1} के घर्षणहीन पाश पर पटरी की प्रतिक्रिया गति
पर अभिलंब है और कोई \omterm{def:b1:work-and-energy:work}{कार्य} नहीं करती: अतः $E_m$ संरक्षित है। तल से ऊपर
$z = R(1 - \cos\theta)$ लेने पर $\tfrac12 mv^2 + mgR(1 -
\cos\theta) = \tfrac12 mv_0^2$: वहाँ प्रयुक्त चाल-नियम $v^2 = v_0^2 - 2gR(1 -
\cos\theta)$ अब एक ही पंक्ति में व्युत्पन्न हो गया। न्यूनतम प्रवेश-चाल
$\sqrt{5gR}$ इसके बाद \cref{ex:b1:newton-dynamics:circles} की शिखर-शर्त
$v^2 \geq gR$ से निकल आती है।
\end{example}

\begin{example}[घर्षण में खोई ऊर्जा]\label{ex:b1:work-and-energy:friction}
कोई गुटका $\alpha$ कोण और $h$ ऊँचाई की ढलान पर गतिज घर्षण $\mu_d$ के साथ
नीचे फिसलता है: $W_{\text{असं}} = -\mu_dmg\cos\alpha \times
h/\sin\alpha = -\mu_dmgh\cot\alpha$, अतः $\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha)$।
$h = \qty{2.0}{m}$, $\alpha = 30^\circ$, $\mu_d = 0.20$ के लिए: $v =
\qty{5.1}{m/s}$, न कि $\qty{6.3}{m/s}$; \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} का एक तिहाई ऊष्मा बन
गया।
\end{example}

\section{साम्यावस्था, स्थायित्व और कला-आरेख}

\begin{proposition}[एक विमा में साम्यावस्था और स्थायित्व]\label{prop:b1:work-and-energy:stability}
\index{साम्यावस्था!स्थायी, अस्थायी}\index{छोटे दोलन}
\omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} $E_p(x)$ वाले किसी \omterm{def:b1:work-and-energy:conservative}{संरक्षी बल} के अधीन किसी अक्ष पर चलता कोई
कण:
\begin{itemize}
  \item $x_0$ पर \emph{साम्यावस्था} में है यदि और केवल यदि
        $F(x_0) = -E_p'(x_0) = 0$: यानी $E_p$ के चरम बिंदुओं पर;
  \item साम्यावस्था \emph{स्थायी} है यदि वहाँ $E_p$ का कोई स्थानीय
        निम्निष्ठ हो ($E_p''(x_0) > 0$): छोटा-सा विस्थापन प्रत्यानयन बल
        पैदा करता है; और किसी उच्चिष्ठ पर वह \emph{अस्थायी} होती है;
  \item किसी स्थायी साम्यावस्था के पास छोटी गतियाँ आवर्त होती हैं, जिनमें
        \[
        \omega_0^2 = \frac{E_p''(x_0)}{m} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
टेलर का सूत्र (गणित का पाठ्यक्रम): $E_p(x) \approx E_p(x_0) +
\tfrac12 E_p''(x_0)(x - x_0)^2$, अतः $F \approx -E_p''(x_0)(x - x_0)$ और
$m\ddot x = -E_p''(x_0)(x - x_0)$ --- यानी $E_p''(x_0)$ दृढ़ता वाली एक
स्प्रिंग। $E_p'' < 0$ के लिए ``दृढ़ता'' ऋणात्मक है: चरघातांकी पलायन।
\end{proof}

\begin{definition}[पलटाव बिंदु; बद्ध और मुक्त गति]\label{def:b1:work-and-energy:turning}
\index{पलटाव बिंदु}\index{बद्ध गति}\index{स्थितिज कूप}\index{स्थितिज अवरोध}
ऊर्जा $E_m$ वाली किसी संरक्षी एक-विमीय गति में कण केवल वहीं हो सकता है जहाँ
$E_p(x) \leq E_m$ (क्योंकि $E_k \geq 0$); और जिन बिंदुओं पर $E_p = E_m$ है वे
\emph{पलटाव बिंदु} कहलाते हैं, जहाँ वेग लुप्त होकर उलट जाता है। किसी
\emph{स्थितिज कूप} में दो पलटाव बिंदुओं के बीच फँसी गति \emph{बद्ध} और
आवर्ती होती है; यदि वह अनंत तक पहुँच सके तो \emph{मुक्त} होती है; और $E_m$
से ऊँचा कोई \emph{स्थितिज अवरोध} पार नहीं किया जा
सकता।
\end{definition}

\begin{definition}[कला-आरेख]\label{def:b1:work-and-energy:phase}
\index{कला-आरेख}
किसी एक-विमीय गति का \emph{कला-आरेख} उन वक्रों का कुल है जिन्हें बिंदु
$(x, \dot x)$ \emph{कला-तल} में खींचता है। किसी संरक्षी निकाय के लिए हर
वक्र एक स्तर-रेखा $\tfrac12 m\dot x^2 +
E_p(x) = E_m$ है: स्थायी साम्यावस्थाओं के चारों ओर बंद वक्र (दोलन), मुक्त
गति के लिए खुले वक्र, और अस्थायी साम्यावस्थाओं में से होकर जाते वे
\emph{विभाजक वक्र} जो दोनों को अलग करते हैं। वक्र दक्षिणावर्त तय होते हैं
($\dot x > 0$ का अर्थ है $x$ का बढ़ना) और कभी एक-दूसरे को काटते नहीं।
\end{definition}

\begin{example}[लोलक का परिदृश्य]\label{ex:b1:work-and-energy:pendulum}
$E_p = -mg\ell\cos\theta$: $\theta = 0$ पर निम्निष्ठ, $E_p'' = mg\ell$, अतः
$\omega_0^2 = g\ell/(m\ell^2) = g/\ell$ --- यानी
\cref{ex:b1:newton-dynamics:pendulum} का छोटे-दोलन वाला परिणाम, जो कूप की
वक्रता से ही पढ़ लिया गया; और $\theta = \pi$ पर उच्चिष्ठ, यानी उलटा लोलक,
अस्थायी। ऊर्जा $E_m < mg\ell$: $\pm\theta_{\max}$ के बीच दोलन;
$E_m > mg\ell$: वह शिखर के पार जाकर घूमता रहता है; $E_m = mg\ell$: विभाजक
वक्र, यानी शिखर तक अनंत रूप से धीमा पहुँचना।
\end{example}

\section{समस्या: किसी रासायनिक बंध का ऊर्जा-कूप}

\begin{problem}[{सप्ताहांत समस्या --- दो परमाणु, एक वक्र: किसी बंध की
स्थितिज ऊर्जा को एक परिदृश्य की तरह पढ़ना --- उसका फ़र्श, उसकी दीवारें, तल
पर के छोटे कंपन, और उससे बाहर निकलने में लगती ऊर्जा}]
\label{pb:b1:work-and-energy:1}
किसी द्विपरमाणुक अणु के दो परमाणुओं की अंतःक्रिया को इस \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} से
प्रतिरूपित किया जाता है
\[
E_p(r) = \epsilon\left[\left(\frac{\sigma}{r}\right)^{12} - 2\left(\frac{\sigma}{r}\right)^{6}\right],
\]
जहाँ $r$ नाभिकों के बीच की दूरी है। हाइड्रोजन क्लोराइड के लिए
$\epsilon = \qty{4.4}{eV}$, $\sigma = \qty{0.127}{nm}$ लीजिए, और मानिए कि
हल्का हाइड्रोजन परमाणु ($m = \qty{1.67e-27}{kg}$) चलता है जबकि भारी क्लोरीन
टिका रहता है। $\qty{1}{eV} = \qty{1.60e-19}{J}$; $k_B = \qty{1.38e-23}{J/K}$;
$N_A = \num{6.02e23}$।

\medskip
\noindent\textbf{भाग I --- परिदृश्य।}
\begin{enumerate}
  \item $E_p$ की सीमाएँ $r \to 0$ और $r \to \infty$ पर दीजिए, और वक्र का
        रेखाचित्र बनाइए।
  \item दिखाइए कि $E_p$ का एक ही निम्निष्ठ है, $r = \sigma$ पर, जिसका मान
        $-\epsilon$ है।
  \item $u = (\sigma/r)^6$ के साथ दिखाइए कि $E_p = \epsilon(u^2 - 2u)$: यानी
        $u$ में एक परवलय। उससे निम्निष्ठ फिर से निकालिए।
  \item बल $F(r)$ व्यक्त कीजिए (प्रतिकर्षी हो तो धनात्मक) और $\sigma$ के
        दोनों ओर उसका चिह्न दीजिए।
  \item $E_p$ और $F$ को $r = 1.2\sigma$ पर निकालिए (eV में और nN में)।
  \item अणु की वियोजन ऊर्जा eV में और \unit{kJ/mol} में क्या है?
        (हाइड्रोजन क्लोराइड की मापी गई बंध-ऊर्जा: \qty{431}{kJ/mol}।)
  \item $r = \sigma$ पर स्थायित्व की कसौटी जाँचिए: $E_p''(\sigma)$
        निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- छोटे कंपन।}
\begin{enumerate}[resume]
  \item $E_p$ का $\sigma$ के आसपास द्वितीय कोटि का टेलर प्रसार लिखिए और
        कोई प्रभावी दृढ़ता $k$ पहचानिए।
  \item $k$ निकालिए (\unit{N/m} में)।
  \item कंपन की \omterm{def:b1:wave-propagation:signal}{कोणीय आवृत्ति}, आवृत्ति और तरंगदैर्घ्य निकालिए, और
        \unit{cm^{-1}} में उसकी \omterm{prop:b1:wave-propagation:sinusoidal}{तरंग संख्या} $1/\lambda$ भी।
  \item हाइड्रोजन क्लोराइड का मापा गया कंपन \qty{2990}{cm^{-1}} पर है। मेल
        पर, और इस पर कि प्रतिरूप क्या ठीक पकड़ता है, टिप्पणी कीजिए।
  \item \qty{300}{K} पर औसत कंपन ऊर्जा $k_BT$ की कोटि की होती है। कंपन का
        \omterm{def:b1:wave-propagation:signal}{आयाम} आँकिए और $\sigma$ से तुलना कीजिए।
  \item उस कंपन में हाइड्रोजन परमाणु की अधिकतम चाल निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बड़े दोलन और बाहर का रास्ता।}
\begin{enumerate}[resume]
  \item \omterm{def:b1:work-and-energy:phase}{कला-आरेख} का वर्णन कीजिए: किन ऊर्जाओं पर बंद वक्र मिलते हैं, किन पर
        खुले, और उन्हें अलग क्या करता है?
  \item $E_m = -\epsilon/2$ के लिए दोनों \omterm{def:b1:work-and-energy:turning}{पलटाव बिंदु} निकालिए ($\sigma$ के
        मात्रकों में)।
  \item दिखाइए कि उनका मध्यबिंदु $\sigma$ से आगे पड़ता है, और समझाइए कि
        गरम करने से (यानी कंपन ऊर्जा बढ़ाने से) औसत बंध-लंबाई क्यों बढ़ती
        है --- यही तापीय प्रसार का मूल है।
  \item बड़े दोलनों का आवर्तकाल $2\pi/\omega_0$ से बड़ा होता है या छोटा?
        दीवारों के आकार से तर्क दीजिए।
  \item $E_m = -\epsilon/2$ से आरंभ करके अणु को वियोजित करने के लिए कितनी
        ऊर्जा देनी पड़ेगी? यदि वह $r = \sigma$ पर हाइड्रोजन परमाणु की \omterm{thm:b1:work-and-energy:ke}{गतिज
        ऊर्जा} के रूप में दी जाए, तो वह कितनी चाल हुई?
  \item दूर से धनात्मक कुल ऊर्जा लेकर पास आते दो परमाणु स्वयं कोई बद्ध अणु
        नहीं बना सकते। क्यों? इसके लिए क्या चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- \omterm{def:b1:work-and-energy:work}{कार्य}, बल और ताप।}
\begin{enumerate}[resume]
  \item $r$ के $2\sigma$ से $\sigma$ तक जाने पर बंध-बल का \omterm{def:b1:work-and-energy:work}{कार्य} निकालिए।
        वहाँ वह बल प्रेरक है या प्रतिरोधी?
  \item बंध को $\sigma$ से $1.5\sigma$ तक अत्यंत धीरे-धीरे तानने के लिए
        किसी बाहरी कारक को जो \omterm{def:b1:work-and-energy:work}{कार्य} करना पड़ेगा, वह निकालिए।
  \item $F$ को $r = 0.9\sigma$ पर निकालिए और $r = 1.2\sigma$ से तुलना कीजिए:
        यह विषमता दबाने बनाम तानने के बारे में क्या बताती है?
  \item किस ताप पर $k_BT$ $\epsilon$ के बराबर होगा? व्यवहार में अणु उससे
        कहीं कम ताप पर क्यों वियोजित हो जाते हैं?
  \item कोई बेहतर प्रतिरूप (मोर्स) दीवारों का आकार बदल देता है पर
        बंध-लंबाई पर निम्निष्ठ बनाए रखता है। ऊपर के कौन-से परिणाम विधि के
        स्तर पर अपरिवर्तित रहेंगे, और कौन-सी संख्याएँ बदलेंगी?
  \item सार दीजिए कि ऊर्जा-विधियाँ किन तीन तरीकों से काम में आईं:
        साम्यावस्था ढूँढ़ने में, कंपन आवृत्ति निकालने में, और यह जानने में
        कि छूटने के लिए क्या चाहिए।
\end{enumerate}
\end{problem}
