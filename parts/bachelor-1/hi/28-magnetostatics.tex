\chapter{चुंबकस्थैतिकी: क्षेत्र, बल और द्विध्रुव}\label{ch:b1:magnetostatics}

पास से गुज़रते किसी तार में धारा बहते ही दिक्सूचक की सुई घूम जाती है:
ओर्स्टेड के 1820 के उस प्रेक्षण ने दो ऐसे विज्ञानों को जोड़ दिया जो अलग-अलग
बड़े हुए थे, यानी विद्युत और चुंबकत्व, और कुछ ही महीनों में \omterm{def:b1:dc-circuits:current}{ऐंपियर} के पास
धाराओं के बीच के बल का नियम था तथा बायो और सावार के पास किसी तार का क्षेत्र।
यह अध्याय \omterm{def:b1:magnetostatics:field}{चुंबकीय क्षेत्र} की स्थैतिकी है: धाराएँ क्या बनाती हैं
(क्षेत्र-चित्र, \omterm{thm:b1:magnetostatics:biotsavart}{बायो--सावार का नियम}, और \omterm{def:b1:dc-circuits:current}{ऐंपियर} की प्रमेय, यानी गाउस के नियम
की चुंबकीय जुड़वाँ), क्षेत्र धाराओं के साथ क्या करता है (\omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल}, जो हर
मोटर और हर मापी को चलाता है), और \omterm{def:b1:magnetostatics:moment}{चुंबकीय द्विध्रुव}, यानी वह धारा-पाश जो
दिक्सूचक, चुंबक और पृथ्वी के अपने क्षेत्र की व्याख्या कर देता है।

\section{स्रोत और क्षेत्र-चित्र}

\begin{definition}[चुंबकस्थैतिक क्षेत्र]\label{def:b1:magnetostatics:field}
\index{चुंबकीय क्षेत्र}\index{क्षेत्र-रेखा!चुंबकीय}\index{टेस्ला}
स्थायी धाराएँ (यानी स्थायी गति में आवेश) हर बिंदु पर एक
\emph{चुंबकीय क्षेत्र} $\vect B$ (टेस्ला, \unit{T}) बनाती हैं, जो उस बल
$q\vect v\wedge\vect B$ से परिभाषित होता है जो वह किसी चलते परीक्षण आवेश पर
लगाता है (\cref{def:b1:charged-particles:lorentz})। उसकी
\emph{क्षेत्र-रेखाएँ} $\vect B$ की स्पर्शरेखा होती हैं और उसी के अनुदिश
दिशित। स्थायी चुंबक भी अपने परमाणुओं की सूक्ष्म धाराओं से इसी प्रकृति के
क्षेत्र बनाते हैं। परिमाण की कोटियाँ: पृथ्वी \qty{5e-5}{T}; रेफ़्रिजरेटर का
चुंबक \qty{0.01}{T}; चुंबकीय अनुनाद प्रतिबिंबन यंत्र \qty{1.5}{T} से
\qty{3}{T} तक; और प्रयोगशाला के सबसे बड़े स्थायी क्षेत्र \qty{45}{T}।
\end{definition}

\begin{proposition}[क्षेत्र-चित्र]\label{prop:b1:magnetostatics:maps}
\index{दाएँ हाथ का नियम}\index{अभिवाह संरक्षण (चुंबकीय)}
\begin{enumerate}
  \item कोई सीधा तार: तार पर केंद्रित वृत्त, उस पर लंब तलों में, जिनकी
        दिशा \emph{दाएँ हाथ के नियम} से मिलती है
        (अँगूठा धारा के अनुदिश, उँगलियाँ $\vect B$ के अनुदिश)।
  \item कोई वृत्ताकार पाश: रेखाएँ पाश में से उसकी अक्ष के अनुदिश निकलती
        हैं और बाहर घूमकर बंद हो जाती हैं; और दूर से, किसी छोटे चुंबक का
        चित्र।
  \item कोई लंबी परिनालिका: भीतर समांतर, सघन, एकसमान रेखाएँ; बाहर लगभग कोई
        क्षेत्र नहीं; और वही चित्र जो किसी छड़-चुंबक का है, जिसका
        ``उत्तरी'' फलक वही है जिससे रेखाएँ निकलती हैं।
  \item चुंबकीय क्षेत्र-रेखाओं का न कोई आरंभ है, न अंत: वे या तो अपने ऊपर
        बंद हो जाती हैं या अनंत तक चली जाती हैं। इसके तुल्य,
        \emph{किसी भी बंद पृष्ठ से होकर $\vect B$ का अभिवाह शून्य है} ---
        यानी कोई चुंबकीय आवेश होते ही नहीं।
\end{enumerate}
\end{proposition}
\admitted

\begin{proposition}[$\vect B$ की सममितियाँ]\label{prop:b1:magnetostatics:symmetry}
\index{सममिति (किसी क्षेत्र की)!चुंबकीय}
दर्पण-परावर्तन के अधीन $\vect B$ $\vect E$ से \emph{उलटा} बरतता है
(वह एक \emph{छद्म-सदिश} है, जो किसी सदिश गुणनफल से बना है): धाराओं के किसी
\emph{सममिति} तल के किसी बिंदु पर $\vect B$ उस तल पर
\emph{लंब} होता है; और किसी \emph{प्रतिसममिति} तल के किसी बिंदु पर
(जहाँ दर्पण-प्रतिबिंब धाराओं को उलट देता है) $\vect B$
तल \emph{में} ही रहता है। अपरिवर्तिताएँ वैसे ही चलती हैं जैसे $\vect E$ के लिए।
\end{proposition}

\begin{proof}
किसी धारा-अवयव का क्षेत्र $\propto I\,\dd\vect l\wedge\vect e_r$ होता है
(नीचे देखिए); और कोई दर्पण किसी सदिश गुणनफल के एक गुणक को, किसी सच्चे सदिश
के प्रतिबिंब के सापेक्ष, उलट देता है, अतः $\vect B$ का प्रतिबिंब उस तीर के
दर्पण-प्रतिबिंब का विपरीत होता है। किसी \omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल} पर $\vect B$ को अपने ही
दर्पण-प्रतिबिंब के विपरीत के बराबर होना पड़ता है, अतः वह तल पर अभिलंब होता
है; और किसी \omterm{prop:b1:electrostatics-gauss:symmetry}{प्रतिसममिति तल} पर, विपरीत का विपरीत: यानी तल में।
\end{proof}

\begin{example}[सममितियों को काम में लेना]\label{ex:b1:magnetostatics:symmetryuse}
किसी सीधे तार के लिए, तार को रखने वाला हर तल कोई \omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल} है, अतः हर बिंदु
पर $\vect B$ उस तल पर लंब है जो तार और उस बिंदु से जाता है: यानी
लंबत्रिज्य, और तार के अनुदिश तथा उसके चारों ओर की अपरिवर्तिता से
$\vect B = B(r)\,\vect e_\theta$। किसी पाश के लिए, पाश का तल कोई \omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल}
है: उस पर $\vect B$ पाश पर अभिलंब है; और अक्ष उसे रखने वाले हर \omterm{prop:b1:electrostatics-gauss:symmetry}{प्रतिसममिति
तल} में है: अतः अक्ष पर $\vect B$ अक्ष के अनुदिश है।
\end{example}

\section{बायो--सावार का नियम}

\begin{theorem}[बायो--सावार]\label{thm:b1:magnetostatics:biotsavart}
\index{बायो--सावार का नियम}\index{निर्वात की चुंबकशीलता}
धारा $I$ ले जाता कोई परिपथ $M$ पर यह क्षेत्र बनाता है
\[
\vect B(M) = \frac{\mu_0}{4\pi}\sum_{\text{परिपथ}} \frac{I\,\dd\vect l\wedge\vect e_{PM}}{PM^2} ,
\]
जो परिपथ के उन अवयवों $\dd\vect l$ पर योग है जो बिंदुओं $P$ पर हैं और धारा
के अनुदिश दिशित हैं, जहाँ $\vect e_{PM}$ $P$ से $M$ तक का एकांक सदिश है,
और $\mu_0 = 4\pi \times 10^{-7}\,\unit{T.m/A}$
\emph{\omterm{thm:b1:magnetostatics:biotsavart}{निर्वात की चुंबकशीलता}}। हर अवयव ऐसा क्षेत्र देता है जो उस पर और रेखा
$PM$ पर लंब है, और जो $1/PM^2$ की तरह गिरता है।
\end{theorem}
\admitted

\begin{proposition}[तार, पाश, परिनालिका]\label{prop:b1:magnetostatics:three}
\index{सीधे तार का क्षेत्र}\index{पाश का क्षेत्र}\index{परिनालिका का क्षेत्र}
\begin{enumerate}
  \item अनंत सीधा तार, $r$ दूरी पर: $B = \dfrac{\mu_0I}{2\pi r}$।
  \item $R$ त्रिज्या का वृत्ताकार पाश, उसकी अक्ष पर केंद्र से $z$ दूरी पर:
        $B_z = \dfrac{\mu_0IR^2}{2(R^2 + z^2)^{3/2}} = \dfrac{\mu_0I}{2R}
        \sin^3\alpha$, जहाँ $\alpha$ वह अर्ध-कोण है जिस पर पाश
        $M$ से दिखता है; और केंद्र पर $B = \mu_0I/2R$।
  \item प्रति मीटर $n$ फेरों की परिनालिका, उसकी अक्ष पर: $B = \tfrac12\mu_0nI
        (\cos\alpha_1 - \cos\alpha_2)$, जहाँ $\alpha_1, \alpha_2$ वे कोण हैं जिन
        पर दोनों सिरे दिखते हैं; और किसी लंबी परिनालिका के लिए भीतर
        $B = \mu_0nI$, तथा किसी सिरे पर उसका आधा।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $M$ तार से $r$ दूरी पर, और $P$ उसके अनुदिश भुज $z$ पर:
$\dd\vect l\wedge\vect e_{PM}$ का परिमाण $\dd z\,\sin\beta = \dd z\cdot r/PM$
है, जहाँ $PM = \sqrt{r^2 + z^2}$, और सारे योगदान लंबत्रिज्य हैं:
$B = \dfrac{\mu_0I}{4\pi}\displaystyle\int_{-\infty}^{\infty}\dfrac{r\,\dd z}{(r^2 + z^2)^{3/2}}
= \dfrac{\mu_0I}{4\pi}\,\dfrac{2}{r}$। (2) पाश का हर अवयव एक ही दूरी
$\sqrt{R^2 + z^2}$ पर है, $M$ से, और
$\vect e_{PM}$ पर लंब; उसके क्षेत्र का परिमाण $\mu_0I\,\dd l/4\pi(R^2 + z^2)$ है, और
पाश के चारों ओर के योग में केवल अक्षीय घटक, यानी उसका अंश $R/\sqrt{R^2 + z^2}$,
बचता है: $B_z = \mu_0I\cdot2\pi R\cdot R/4\pi(R^2 +
z^2)^{3/2}$। (3) परिनालिका का कोई फाँक $\dd z'$ धारा
$nI\,\dd z'$ का एक पाश है; और $z' - z = R/\tan\alpha$, $\dd z' = -R\,\dd\alpha/\sin^2\alpha$
के साथ, परिनालिका पर $B = \dfrac{\mu_0nI}{2}\displaystyle\int\sin^3\alpha\,\dfrac{\dd\alpha}{\sin^2\alpha}$
$= \tfrac12\mu_0nI(\cos\alpha_1 - \cos\alpha_2)$; और अपरिमित रूप से लंबी होने पर,
$\alpha_1 \to 0$, $\alpha_2 \to \pi$।
\end{proof}

\section{ऐंपियर का नियम}

\begin{theorem}[ऐंपियर]\label{thm:b1:magnetostatics:ampere}
\index{ऐंपियर का नियम}\index{परिचालन!चुंबकीय क्षेत्र का}
$\vect B$ का परिचालन, किसी भी बंद दिशित वक्र $\Gamma$ के अनुदिश,
$\mu_0$ गुना उस कुल धारा के बराबर होता है जो $\Gamma$ से घिरे किसी भी पृष्ठ
को पार करती है, और जिसे तब धनात्मक गिना जाता है जब वह $\Gamma$ की दिशा से
दाएँ हाथ के नियम द्वारा मिली दिशा में पार करे:
\[
\oint_\Gamma \vect B\cdot\dd\vect l = \mu_0 I_{\text{घिरी}} .
\]
\end{theorem}

\begin{proof}
किसी सीधे तार पर केंद्रित $r$ त्रिज्या के किसी वृत्त के लिए, $\vect B$
स्पर्शरेखीय है और हर जगह $\mu_0I/2\pi r$ परिमाण का, अतः परिचालन
$2\pi r \times \mu_0I/2\pi r = \mu_0I$ है, जो $r$ से स्वतंत्र है; और तार को न
घेरते किसी वक्र के लिए योगदान कट जाते हैं। सामान्य स्थिति स्वीकृत है
(वर्ष 2 का खंड)।
\end{proof}

\begin{method}[ऐंपियर का नियम काम में लेना]\label{meth:b1:magnetostatics:ampere}
गाउस की तरह ही: (1) सममितियाँ $\vect B$ की दिशा और वह चर दे देती हैं जिस पर
वह निर्भर है; (2) $M$ से होकर कोई \emph{ऐंपियरी पाश} चुनिए,
जिस पर $\vect B$ या तो स्पर्शरेखीय हो और अचर परिमाण का, या फिर
लंब; (3) उसमें से जाती धारा गिनिए; (4) हल कीजिए। यह अनंत तारों तथा
बेलनों, अनंत परिनालिकाओं तथा टोरॉइडों, और अनंत धारा-चादरों के लिए चलता है।
\end{method}

\begin{proposition}[बेलन, परिनालिका, टोरॉइड]\label{prop:b1:magnetostatics:ampereapps}
\begin{enumerate}
  \item $a$ त्रिज्या का कोई बेलनाकार तार, जो $I$ एकसमान ले जाए:
        भीतर $B = \dfrac{\mu_0Ir}{2\pi a^2}$, और बाहर $\dfrac{\mu_0I}{2\pi r}$
        --- अधिकतम पृष्ठ पर है।
  \item प्रति मीटर $n$ फेरों की कोई अनंत परिनालिका: भीतर $\vect B = \mu_0nI\,
        \vect e_z$ एकसमान, और बाहर $\vect 0$, चाहे उसकी अनुप्रस्थ काट का
        आकार कुछ भी हो।
  \item $N$ फेरों का कोई टोरॉइड: कुंडलनों के भीतर $B = \dfrac{\mu_0NI}{2\pi r}$,
        और बाहर शून्य --- यानी अपने ऊपर बंद हुई कोई परिनालिका, जिसका कोई
        आवारा क्षेत्र नहीं।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $r$ त्रिज्या का वृत्त: $2\pi rB = \mu_0I(r/a)^2$ या $\mu_0I$। (2) $\vect B$
सममिति से अक्षीय है और, दोनों लंबी भुजाएँ बाहर रखते किसी आयत पर \omterm{def:b1:dc-circuits:current}{ऐंपियर} से,
बाहर एकसमान, अतः शून्य (वह अनंत पर लुप्त हो जाता है); और एक लंबी भुजा भीतर
रखते, $\ell$ लंबाई के किसी आयत से $B\ell =
\mu_0n\ell I$। (3) टोरस के भीतर $r$ त्रिज्या का वृत्त: $2\pi rB =
\mu_0NI$; और बाहर घिरी हुई धारा शून्य है।
\end{proof}

\section{लाप्लास बल}

\begin{theorem}[लाप्लास बल]\label{thm:b1:magnetostatics:laplace}
\index{लाप्लास बल}
कोई चालक अवयव $\dd\vect l$, जो $I$ ले जा रहा हो और किसी क्षेत्र $\vect B$ में हो,
बल $\dd\vect F = I\,\dd\vect l\wedge\vect B$ पाता है; और किसी एकसमान
क्षेत्र में कोई सीधा खंड $\vect L$ बल $\vect F = I\vect L\wedge\vect B$ पाता
है, जिसका परिमाण $ILB\sin\theta$ है और जो तार तथा क्षेत्र दोनों पर लंब
होता है।
\end{theorem}

\begin{proof}
$\dd\vect l$ के आवेश-वाहक (संख्या $n\,S\,\dd l$, आवेश $q$, और अपवाह वेग
$\vect v$) हर एक $q\vect v\wedge\vect B$ अनुभव करते हैं; और उनका योग
$nSq\,\dd l\,\vect v\wedge\vect B = I\,\dd\vect l\wedge\vect B$ है, क्योंकि $I = nqvS$
है और $\vect v$ $\dd\vect l$ के अनुदिश। वाहक यह बल अपनी टक्करों के ज़रिए तार
की जालक तक पहुँचा देते हैं।
\end{proof}

\begin{proposition}[समांतर तारों के बीच बल]\label{prop:b1:magnetostatics:parallel}
\index{ऐंपियर (परिभाषा)}
$d$ दूरी पर के दो लंबे समांतर तार, जो $I_1$ और $I_2$ ले जा रहे हों, धाराओं
के समांतर होने पर एक-दूसरे को आकर्षित करते हैं और प्रतिसमांतर होने पर
प्रतिकर्षित, और प्रति एकांक लंबाई बल यह है
\[
\frac{F}{\ell} = \frac{\mu_0I_1I_2}{2\pi d} .
\]
\qty{1}{m} की दूरी पर \qty{1}{A} ले जाते दो तार प्रति मीटर $2 \times 10^{-7}$~न्यूटन
अनुभव करते हैं --- यही 1948 से 2019 तक ऐंपियर की परिभाषा थी, और यहीं से
$\mu_0 = 4\pi \times 10^{-7}$ का मान आया।
\end{proposition}

\begin{proof}
तार~1 तार~2 पर $\mu_0I_1/2\pi d$ बनाता है, जो उस पर लंब है; और तार~2 की
$\ell$ लंबाई पर \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल} $I_2\ell \times \mu_0I_1/2\pi d$ है, जो धाराओं के
समांतर होने पर तार~1 की ओर होता है (दाएँ हाथ के नियम से जाँचिए)।
\end{proof}

\section{चुंबकीय द्विध्रुव}

\begin{definition}[चुंबकीय आघूर्ण]\label{def:b1:magnetostatics:moment}
\index{चुंबकीय आघूर्ण}\index{चुंबकीय द्विध्रुव}
$S$ क्षेत्रफल का कोई तलीय पाश, जो $I$ ले जा रहा हो और दाएँ हाथ के नियम से
दिशित हो (अभिलंब $\vect n$ अँगूठे के अनुदिश, जब उँगलियाँ $I$ के पीछे चलें),
\emph{चुंबकीय आघूर्ण} $\vect m = IS\,\vect n$ (\unit{A.m^2}) रखता है; और $N$
फेरों के लिए, $NIS\,\vect n$। $r \gg \sqrt S$ दूरियों से देखने पर वह पाश कोई
\emph{चुंबकीय द्विध्रुव} है: उसके क्षेत्र का रूप ठीक \omterm{def:b1:potential-capacitors:dipole}{विद्युत द्विध्रुव} जैसा
है, जहाँ $\vect p/\varepsilon_0 \to \mu_0\vect m$:
\[
B_r = \frac{\mu_0}{4\pi}\,\frac{2m\cos\theta}{r^3}, \qquad
B_\theta = \frac{\mu_0}{4\pi}\,\frac{m\sin\theta}{r^3} .
\]
कोई छड़-चुंबक, कोई परमाणु, और पृथ्वी --- सब चुंबकीय द्विध्रुव हैं: पृथ्वी का
आघूर्ण \qty{8e22}{A.m^2} है, और किसी इलेक्ट्रॉन का \qty{9.3e-24}{A.m^2} (यानी
बोर मैग्नेटॉन)।
\end{definition}

\begin{proof}
अक्ष पर \omterm{prop:b1:magnetostatics:three}{पाश का क्षेत्र} $\mu_0IR^2/2z^3$, जो $z \gg R$ के लिए है,
$\mu_0 \cdot 2m/4\pi z^3$ के बराबर बैठता है, जहाँ $m = I\pi R^2$, और $B_r$ से
मेल खाता है, $\theta = 0$ पर;
और पूरा कोणीय रूप स्वीकृत है।
\end{proof}

\begin{theorem}[किसी क्षेत्र में द्विध्रुव]\label{thm:b1:magnetostatics:dipoleinfield}
\index{चुंबकीय द्विध्रुव!बलाघूर्ण}\index{चुंबकीय द्विध्रुव!ऊर्जा}
कोई \omterm{def:b1:magnetostatics:moment}{चुंबकीय आघूर्ण} $\vect m$, जो किसी एकसमान क्षेत्र $\vect B$ में हो, कोई
परिणामी बल नहीं पाता, पर एक \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} $\vect\Gamma = \vect m\wedge\vect B$
पाता है जो उसे क्षेत्र के अनुदिश पंक्तिबद्ध कर देता है, और उसकी \omterm{def:b1:work-and-energy:conservative}{स्थितिज
ऊर्जा} $E_p = -\vect m\cdot\vect B$ होती है। किसी असमान क्षेत्र में,
पंक्तिबद्ध होने पर वह प्रबल क्षेत्र वाले भाग की ओर खिंचता है (और
विपरीत-पंक्तिबद्ध होने पर दूर धकेला जाता है)।
\end{theorem}

\begin{proof}
$a \times b$ के किसी आयताकार पाश के लिए, जिसका अभिलंब $\vect n$ कोण
$\theta$ बनाता हो $\vect B$ के साथ: घूर्णन-अक्ष के समांतर $b$ लंबाई की दोनों
भुजाओं पर \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल} $\pm IbB$ हैं, जो विपरीत हैं और $a\sin\theta$ की दूरी
पर: यानी $IabB\sin\theta = mB\sin\theta$ आघूर्ण का एक युग्म, जो
$\theta$ घटाना चाहता है; और बाकी दोनों भुजाओं पर बल विपरीत तथा संरेख हैं।
ऊर्जा: $\Gamma = -\dd E_p/\dd\theta$, जहाँ $\Gamma = -mB\sin\theta$, जिससे
$E_p = -mB\cos\theta$ मिलता है। कोई भी तलीय पाश छोटे आयतों का योग है;
और असमान वाली स्थिति स्वीकृत है।
\end{proof}

\section{समस्या: तार, लाइन और मापी}

\begin{problem}[{सप्ताहांत समस्या --- किसी समाक्ष तार पर ऐंपियर का नियम,
किसी विद्युत लाइन पर लाप्लास बल, और किसी चल-कुंडली धारामापी का
अभिकल्प}]
\label{pb:b1:magnetostatics:1}
$\mu_0 = 4\pi \times 10^{-7}$~SI; ताँबे की प्रतिरोधकता $\rho = \qty{1.7e-8}{\ohm.m}$।

\medskip
\noindent\textbf{भाग I --- समाक्ष तार।}
कोई तार $I = \qty{1.0}{A}$ को $a = \qty{0.50}{mm}$ त्रिज्या के ठोस ताँबे के
क्रोड के अनुदिश ले जाता है और ऐसे आवरण के अनुदिश वापस लाता है जिसकी भीतरी
त्रिज्या $b =
\qty{2.5}{mm}$ और बाहरी $c = \qty{3.0}{mm}$ है; और हर
चालक की काट पर धाराएँ एकसमान हैं।
\begin{enumerate}
  \item सममितियाँ और अपरिवर्तिताएँ: $\vect B$ की दिशा और वह चर जिस पर वह
        निर्भर है।
  \item \omterm{def:b1:dc-circuits:current}{ऐंपियर} के नियम से, क्रोड में, दोनों चालकों के बीच, और तार के बाहर
        $B(r)$।
  \item $r = a$ और $r = b$ पर मान।
  \item क्रोड की जगह उसी त्रिज्या $a$ की ताँबे की कोई पतली नली रख दी जाती
        है, जो वही धारा ले जाती है: हर क्षेत्र में क्या बदलता है?
  \item आवरण के भीतर $B(r)$ ($b < r < c$); $b$ और $c$ पर मान जाँचिए; और
        $B(r)$ का आलेख $0$ से \qty{4}{mm} तक खींचिए।
  \item आवरण पर \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल}: दिशा (क्रोड की ओर आकर्षण या उससे प्रतिकर्षण?)
        और प्रति एकांक क्षेत्रफल परिमाण, जहाँ आवरण पर क्षेत्र $b$ तथा
        $c$ पर के मानों का औसत लिया जाए।
        \qty{10}{kA} की किसी स्पंदित धारा के लिए भी वही।
  \item संकेतों के लिए समाक्ष ज्यामिति ही क्यों काम में ली जाती है? $d = \qty{2}{mm}$
        की दूरी पर के समांतर तारों के किसी जोड़े से तुलना कीजिए: उस जोड़े से
        $D = \qty{10}{cm}$ दूरी पर क्षेत्र (दो तारों का अध्यारोपण लीजिए)।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- विद्युत लाइन।}
$d = \qty{1.0}{m}$ की दूरी पर दो क्षैतिज चालक, जो भूमि से
$h = \qty{10}{m}$ ऊपर हैं, $I = \qty{500}{A}$ विपरीत दिशाओं में ले जाते हैं।
\begin{enumerate}[resume]
  \item मध्यबिंदु के नीचे भूमि के बिंदु पर एक चालक का क्षेत्र; दिखाइए कि
        दोनों क्षेत्र मिलकर कोई ऊर्ध्वाधर क्षेत्र बना देते हैं और उसका मान
        दीजिए; पृथ्वी के क्षेत्र से और \qty{100}{\micro T} की
        संस्तुत सीमा से तुलना कीजिए।
  \item दिखाइए कि $D \gg d$ दूरियों पर उस जोड़े का क्षेत्र
        $\mu_0Id/2\pi D^2$ की तरह गिरता है: किसी अकेले तार से तेज़ी से
        क्यों?
  \item किसी एक चालक के \qty{1.0}{m} नीचे क्षेत्र, जहाँ कोई तकनीशियन काम
        कर सकता है।
  \item चालकों के बीच प्रति मीटर बल; आकर्षण या प्रतिकर्षण? और
        \qty{20}{kA} की किसी लघुपथन धारा के अधीन वही।
  \item \qty{300}{m} के किसी फैलाव पर बल, सामान्य में और लघुपथन में;
        फैलाव के भार (\qty{1.0}{kg/m}) से तुलना कीजिए; और लघुपथन के दौरान
        लाइनें क्या करती हैं?
  \item कोई त्रिकला लाइन एक ही \omterm{def:b1:wave-propagation:signal}{आयाम} की तीन धाराएँ ले जाती है, जो
        \qty{120}{\degree} से खिसकी हुई हैं: दिखाइए कि हर क्षण उनका योग
        लुप्त हो जाता है, और दूर के क्षेत्र के लिए इसका क्या अर्थ है।
  \item प्रश्न~8 वाले बिंदु पर कोई दिक्सूचक, जहाँ पृथ्वी का क्षैतिज
        क्षेत्र \qty{2.0e-5}{T} है: क्या लाइन का क्षेत्र उसे विक्षेपित
        करता है? (उसकी दिशा पर विचार कीजिए।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चल-कुंडली धारामापी।}
कोई आयताकार कुंडली, $N = 200$ फेरे, भुजाएँ $a = \qty{2.0}{cm}$ (जो
घूर्णन-अक्ष के समांतर है) और $b = \qty{3.0}{cm}$, बेलनाकार ध्रुव-खंडों के
बीच की दरार में अपनी अक्ष के परितः घूमती है, जहाँ क्षेत्र $B = \qty{0.20}{T}$
\emph{त्रिज्य} है: यानी हमेशा कुंडली के तल में, और उसकी भुजाओं $a$ पर लंब।
कोई कुंडलित स्प्रिंग प्रत्यानयन \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} $-C\theta$ लगाती है, जहाँ
$C = \qty{1.0e-5}{N.m/rad}$। कुंडली और सुई का \omterm{prop:b1:angular-momentum:inertia}{जड़त्व आघूर्ण}:
$J = \qty{6.7e-8}{kg.m^2}$।
\begin{enumerate}[resume]
  \item चारों भुजाओं पर \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल}; उनमें से कौन-से अक्ष के परितः
        \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} देते हैं?
  \item किसी धारा $I$ के लिए \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण}; त्रिज्य क्षेत्र उसे $\theta$ से
        स्वतंत्र क्यों कर देता है?
  \item साम्य पर विक्षेप $\theta$; प्रति मिलीऐंपियर रेडियन में सुग्राहिता;
        और \qty{1.2}{rad} के पूर्ण-पैमाना विक्षेप के लिए धारा।
  \item कुंडली \qty{0.10}{mm} व्यास के ताँबे के तार से लपेटी गई है:
        कुंडली का प्रतिरोध; पूर्ण पैमाने पर उस पर वोल्टता और क्षय हुई
        शक्ति।
  \item कुंडली की गति का समीकरण; और साम्य के आसपास दोलनों का स्वाभाविक
        आवर्तकाल।
  \item क्षेत्र में कुंडली की गति एक रोकने वाला \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण}
        $-\alpha\dot\theta$ प्रेरित करती है (\cref{ch:b1:induction});
        क्रांतिक अवमंदन के लिए $\alpha$ का मान, और वह क्यों चाहिए।
  \item \qty{10}{V} पूर्ण-पैमाना वोल्टमापी बनाने के लिए श्रेणी-प्रतिरोध; और
        \qty{1.0}{A} पूर्ण-पैमाना धारामापी बनाने के लिए शंट प्रतिरोध।
  \item पूर्ण पैमाने पर कुंडली का \omterm{def:b1:magnetostatics:moment}{चुंबकीय आघूर्ण}; और \qty{0.20}{T} के किसी
        एकसमान क्षेत्र में पंक्तिबद्ध होने पर उसकी ऊर्जा $-\vect m\cdot
        \vect B$; पूर्ण पैमाने पर स्प्रिंग में संचित ऊर्जा से तुलना कीजिए।
  \item यदि क्षेत्र त्रिज्य के बजाय \emph{एकसमान} होता, तो दिखाइए कि पैमाना
        अब रैखिक नहीं रहता: साम्य की शर्त लिखिए, जहाँ
        $\theta$ उस स्थिति से नापा जाए जिसमें कुंडली का तल
        $\vect B$ को रखता है।
  \item \qty{0.5}{mA} \omterm{def:b1:wave-propagation:signal}{आयाम} की कोई \qty{50}{Hz} धारा मापी को दी जाती है:
        सुई क्या दिखाती है, और क्यों? (\qty{50}{Hz} की तुलना \omterm{def:b1:oscillators-resonance:harmonic}{स्वाभाविक
        आवृत्ति} से कीजिए।)
  \item तीनों दृश्यों का सार तीन आँकड़ों में दीजिए, और बताइए कि हर एक को
        कौन-से एक नियम ने पैदा किया।
\end{enumerate}
\end{problem}
