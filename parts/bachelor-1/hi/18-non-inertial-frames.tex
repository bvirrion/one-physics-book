\chapter{अजड़त्वीय तंत्र; पृथ्वी पर गतिकी}\label{ch:b1:non-inertial-frames}

बाईं ओर मुड़ती किसी बस में खड़े होकर आप दाईं ओर झुक जाते हैं, किसी ऐसे बल
से धकेले हुए जिसे कोई लगाता ही नहीं। घूमते चक्र पर नचाई गई बाल्टी का पानी
अपनी सतह को कटोरे-सा खोखला कर लेता है। किसी संग्रहालय के गुंबद के नीचे कोई
लोलक घंटों झूलता रहता है और उसके झूलने की रेखा धीरे-धीरे घूमती जाती है,
घंटे भर में कुछ अंश, मानो कोई अदृश्य हाथ उसे घुमा रहा हो। इनमें से कोई भी
नया बल नहीं है: यह बस वह रूप है जो \omterm{thm:b1:newton-dynamics:laws}{न्यूटन के नियम} तब ले लेते हैं जब गति का
वर्णन जिस तंत्र में किया जा रहा है वही स्वयं त्वरित हो रहा हो या घूम रहा
हो। यह अध्याय दिखाता है कि वेग और त्वरण एक तंत्र से दूसरे में कैसे बदलते
हैं, गतिकी के नियम \emph{\omterm{thm:b1:non-inertial-frames:dynamics}{जड़त्वीय बल}} जोड़ देने से कैसे सुधर जाते हैं, और
वे बल सबसे महत्वपूर्ण घूर्णी तंत्र पर --- यानी पृथ्वी पर --- क्या करते हैं।

\section{तंत्र बदलना: शुद्धगतिकी}

\begin{definition}[सापेक्ष गति; स्थानांतरण में और घूर्णन में तंत्र]\label{def:b1:non-inertial-frames:frames}
\index{सापेक्ष गति}\index{स्थानांतरण में तंत्र}\index{घूर्णन में तंत्र}
मान लीजिए $\mathcal R$ कोई तंत्र है (मूलबिंदु $O$) और $\mathcal R'$ कोई
दूसरा (मूलबिंदु $O'$), जो उसके सापेक्ष चल रहा है। $\mathcal R'$
\emph{स्थानांतरण में} है यदि उसकी अक्षें $\mathcal R$ में स्थिर दिशाएँ बनाए
रखें ($\mathcal R'$ के हर बिंदु का वेग $O'$ का वेग हो); और वह
\emph{किसी स्थिर अक्ष $\Delta$ के परितः घूर्णन में} है यदि $\Delta$ दोनों
तंत्रों में स्थिर हो और $\mathcal R'$ उसके परितः \omterm{cor:b1:point-kinematics:circular}{कोणीय वेग} $\Omega(t)$ से
घूमता हो। किसी बिंदु $M$ के, $\mathcal R'$ के \emph{सापेक्ष}, वेग और त्वरण
$\vect v'$ तथा $\vect a'$ वही हैं जो $\mathcal R'$ का कोई स्थिर प्रेक्षक
मापता है।
\end{definition}

\begin{theorem}[वेगों और त्वरणों का संयोजन]\label{thm:b1:non-inertial-frames:composition}
\index{वाहक वेग}\index{वाहक त्वरण}\index{कोरिओलिस त्वरण}
\begin{itemize}
  \item $\mathcal R'$ जो $\vect v_{O'}$ वेग और $\vect a_{O'}$ त्वरण से
        स्थानांतरण कर रहा हो:
        \[
        \vect v = \vect v' + \vect v_{O'}, \qquad \vect a = \vect a' + \vect a_{O'} .
        \]
  \item $\mathcal R'$ जो स्थिर अक्ष $\Delta$ के परितः $\Omega$ से घूम रहा हो
        (अक्ष के अनुदिश सदिश $\vect\Omega = \Omega\vect e_z$); और $H$ अक्ष
        पर $M$ का प्रक्षेप हो:
        \[
        \vect v = \vect v' + \vect\Omega\wedge\vect{HM}, \qquad
        \vect a = \vect a' + \underbrace{\big(-\Omega^2\vect{HM} + \dot\Omega\,\vect e_z\wedge\vect{HM}\big)}_{\vect a_e}
        + \underbrace{2\vect\Omega\wedge\vect v'}_{\vect a_C} .
        \]
\end{itemize}
$\vect a_e$ \emph{\omterm{thm:b1:non-inertial-frames:composition}{वाहक त्वरण}} है ($\mathcal R'$ के उस बिंदु का त्वरण जो
$M$ से संपाती है: एकसमान घूर्णन के लिए अक्ष की ओर अभिकेंद्र), और
$\vect a_C = 2\vect\Omega\wedge\vect v'$ \emph{\omterm{thm:b1:non-inertial-frames:composition}{कोरिओलिस त्वरण}}, जो केवल तभी
होता है जब $M$ घूर्णी तंत्र के सापेक्ष चल रहा हो।
\end{theorem}

\begin{proof}
स्थानांतरण: $\vect{OM} = \vect{OO'} + \vect{O'M}$, जहाँ
$\vect{O'M}$ के घटक उभयनिष्ठ आधार में लिए गए हैं; दो बार अवकलित कीजिए।
घूर्णन: $\Delta$ के परितः \omterm{def:b1:point-kinematics:cylindrical}{बेलनाकार निर्देशांक} लीजिए, $\mathcal R$ में
$(r, \theta, z)$ और $\mathcal R'$ में $(r, \theta', z)$, जहाँ
$\theta = \theta' + \int\Omega\,\dd t$ है।
फिर $\mathcal R$ में \cref{thm:b1:point-kinematics:cylindrical} से,
$\dot\theta = \dot\theta' + \Omega$ के साथ:
$\vect v = \dot r\vect e_r + r(\dot\theta' + \Omega)\vect e_\theta + \dot z\vect e_z
= \vect v' + r\Omega\vect e_\theta$, और $r\Omega\vect e_\theta = \vect\Omega\wedge\vect{HM}$।
त्वरण के लिए, $\ddot\theta = \ddot\theta' + \dot\Omega$:
$\vect a = [\ddot r - r(\dot\theta' + \Omega)^2]\vect e_r + [r(\ddot\theta' + \dot\Omega)
+ 2\dot r(\dot\theta' + \Omega)]\vect e_\theta + \ddot z\vect e_z$; फैलाकर और
समूहित करने पर, $\vect a = \vect a' - r\Omega^2\vect e_r + r\dot\Omega\vect e_\theta
+ 2\Omega(-r\dot\theta'\vect e_r + \dot r\vect e_\theta)$, और अंतिम कोष्ठक
$2\vect\Omega\wedge\vect v'$ है, क्योंकि $\vect e_z\wedge\vect e_r = \vect e_\theta$,
$\vect e_z\wedge\vect e_\theta = -\vect e_r$।
\end{proof}

\begin{example}[घूमते चक्र पर चलना]\label{ex:b1:non-inertial-frames:turntable}
\cref{exo:b1:point-kinematics:12} में ऐसे किसी व्यक्ति का त्वरण निकाला गया
था जो $u$ चाल से त्रिज्या के अनुदिश ऐसे चक्र पर चल रहा है जो $\Omega$ से
घूम रहा है: $-ut\Omega^2\vect e_r
+ 2u\Omega\vect e_\theta$। नई भाषा में: सापेक्ष त्वरण शून्य, \omterm{thm:b1:non-inertial-frames:composition}{वाहक त्वरण}
$-r\Omega^2\vect e_r$, और कोरिओलिस $2\vect\Omega\wedge u\vect e_r = 2u\Omega\vect e_\theta$।
\end{example}

\section{अजड़त्वीय तंत्र में गतिकी: जड़त्वीय बल}

\begin{theorem}[अजड़त्वीय तंत्र में न्यूटन का नियम]\label{thm:b1:non-inertial-frames:dynamics}
\index{जड़त्वीय बल}\index{अपकेंद्र बल}\index{कोरिओलिस बल}
जो तंत्र $\mathcal R'$ जड़त्वीय नहीं है, उसमें किसी कण की गति इस नियम का
पालन करती है
\[
m\vect a' = \sum\vect F + \vect F_e + \vect F_C, \qquad
\vect F_e = -m\vect a_e, \qquad \vect F_C = -m\vect a_C = -2m\vect\Omega\wedge\vect v' :
\]
न्यूटन का दूसरा नियम तभी तक लागू रहता है जब तक वास्तविक बलों में
\emph{वाहक बल} जोड़ दिया जाए (एकसमान घूर्णी तंत्र के लिए यह
\emph{\omterm{thm:b1:non-inertial-frames:dynamics}{अपकेंद्र बल}} $+m\Omega^2\vect{HM}$ है, जो अक्ष से दूर की ओर होता है;
और स्थानांतरण करते तंत्र के लिए $-m\vect a_{O'}$), तथा साथ में
\emph{\omterm{thm:b1:non-inertial-frames:dynamics}{कोरिओलिस बल}} भी। ये \emph{\omterm{thm:b1:non-inertial-frames:dynamics}{जड़त्वीय बल}} अन्योन्यक्रियाएँ नहीं हैं:
इन्हें कोई पिंड नहीं लगाता, इनकी कोई प्रतिक्रिया नहीं होती, और किसी
\omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} में ये लुप्त हो जाते हैं।
\end{theorem}

\begin{proof}
$m\vect a = \sum\vect F$ \omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} $\mathcal R$ में लागू है; उसमें
$\vect a = \vect a' + \vect a_e + \vect a_C$ प्रतिस्थापित कीजिए और दोनों
पदों को दाईं ओर ले जाइए।
\end{proof}

\begin{example}[झुकता यात्री, साहुल]\label{ex:b1:non-inertial-frames:bus}
$a = \qty{3.0}{m/s^2}$ पर ब्रेक लगाती किसी बस में लटकता पट्टा, बस के तंत्र
में, गुरुत्व के साथ-साथ आगे की ओर एक \omterm{thm:b1:non-inertial-frames:dynamics}{जड़त्वीय बल} $ma$ भी अनुभव करता है: वह
\emph{आभासी गुरुत्व} $\vect g' = \vect g - \vect a_{O'}$ के अनुदिश लटकता
है, यानी आगे की ओर $\arctan(a/g) = 17^\circ$ झुका हुआ। स्थिर खड़ा रहना
चाहने वाले यात्री को उसी आभासी ऊर्ध्वाधर की ओर झुकना पड़ता है। मुक्त रूप
से गिरती किसी लिफ़्ट में $\vect g' = \vect 0$: भीतर से देखने पर भारहीनता।
\end{example}

\begin{example}[घूमती बाल्टी]\label{ex:b1:non-inertial-frames:bucket}
अपने ऊर्ध्वाधर अक्ष के परितः $\Omega$ से घूमती किसी बाल्टी का पानी घूर्णी
तंत्र में गुरुत्व और \omterm{thm:b1:non-inertial-frames:dynamics}{अपकेंद्र बल} के अधीन विराम में रहता है: उसकी मुक्त सतह
आभासी गुरुत्व $\vect g' = -g\vect e_z +
\Omega^2r\vect e_r$ पर लंब होती है (विराम में पड़े किसी तरल की सतह क्षेत्र
के अभिलंब होती है, \cref{ch:b1:fluid-statics})। उसकी ढाल
$\dd z/\dd r = \Omega^2r/g$ है, अतः
$z = \Omega^2r^2/2g$: यानी एक परवलयज। \qty{10}{rad/s} पर \qty{10}{cm} की
बाल्टी का किनारा केंद्र से \qty{5}{cm} ऊपर खड़ा रहता है --- यही
द्रव-दर्पण दूरदर्शी का सिद्धांत है, जिसका घूमता पारद ठीक वही परवलयी आकार
ले लेता है जो किसी दर्पण को चाहिए।
\end{example}

\begin{method}[अजड़त्वीय तंत्र में काम करना]\label{met:b1:non-inertial-frames:method}
वह तंत्र चुनिए जिसमें प्रश्न सबसे सरल हो (बस, घूमता चक्र, पृथ्वी);
वास्तविक बलों की सूची बनाइए; $-m\vect a_e$ जोड़िए और, यदि पिंड उस तंत्र में
चल रहा हो, तो $-2m\vect\Omega\wedge\vect v'$ भी; फिर
\cref{met:b1:newton-dynamics:method} की तरह आगे बढ़िए। जाँच: \omterm{thm:b1:non-inertial-frames:dynamics}{जड़त्वीय बल}
ऊर्जा-संतुलन में कभी ``किसी के द्वारा किया गया कार्य'' बनकर नहीं आते ---
\omterm{thm:b1:non-inertial-frames:dynamics}{कोरिओलिस बल} $\vect v'$ पर लंब है, अतः कोई कार्य नहीं करता; और \omterm{thm:b1:non-inertial-frames:dynamics}{अपकेंद्र बल}
किसी एकसमान घूर्णी तंत्र में \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} $-\tfrac12 m\Omega^2r^2$ से
व्युत्पन्न होता है।
\end{method}

\section{पार्थिव तंत्र}

\begin{definition}[भूकेंद्रीय और पार्थिव तंत्र]\label{def:b1:non-inertial-frames:earth}
\index{भूकेंद्रीय तंत्र}\index{पार्थिव तंत्र}
\emph{भूकेंद्रीय तंत्र} का मूलबिंदु पृथ्वी के केंद्र पर है और उसकी अक्षें
स्थिर तारों की ओर संकेत करती हैं; एक दिन या उससे कम की गतियों के लिए वह
उत्कृष्ट सन्निकटन तक जड़त्वीय है (सूर्य के परितः उसका अपना त्वरण केवल
\qty{6e-3}{m/s^2} है, और पूरी पृथ्वी पर लगभग एक-सा)। भूमि से जुड़ा
\emph{पार्थिव तंत्र} उसके सापेक्ष ध्रुवीय अक्ष के परितः इस दर से घूमता है
\[
\Omega = \frac{2\pi}{\qty{86164}{s}} = \qty{7.29e-5}{rad/s} .
\]
\end{definition}

\begin{proposition}[भार, और $g$ का परिवर्तन]\label{prop:b1:non-inertial-frames:weight}
\index{भार!और पृथ्वी का घूर्णन}
\omterm{def:b1:non-inertial-frames:earth}{पार्थिव तंत्र} में विराम में पड़ा कोई पिंड गुरुत्वीय आकर्षण
$m\vect{\mathcal G}$ और \omterm{thm:b1:non-inertial-frames:dynamics}{अपकेंद्र बल} $m\Omega^2\vect{HM}$ अनुभव करता है; इनका
योग ही वह है जो तराजू मापता है और साहुल-रेखा दिखाती है: यानी \emph{भार}
$m\vect g = m\vect{\mathcal G} + m\Omega^2\vect{HM}$। अपकेंद्र पद भूमध्य
रेखा पर सबसे बड़ा है, $\Omega^2R = \qty{0.034}{m/s^2}$, यानी $g$ का
$0.35\%$; वह ध्रुवों से भूमध्य रेखा तक $g$ को घटाता जाता है
और मध्य अक्षांशों पर ऊर्ध्वाधर को पृथ्वी के केंद्र से $0.1^\circ$ तक हटा
देता है।
\end{proposition}

\begin{proof}
\cref{thm:b1:non-inertial-frames:dynamics} को $\vect v' = \vect 0$ के साथ
लगाइए (कोई \omterm{thm:b1:non-inertial-frames:dynamics}{कोरिओलिस बल} नहीं); $\Omega^2R = (7.29\times10^{-5})^2 \times 6.37\times10^6$।
विचलन-कोण $m\vect{\mathcal G}$ और \omterm{thm:b1:non-inertial-frames:dynamics}{अपकेंद्र बल} के त्रिभुज से निकल आता है
(\cref{exo:b1:non-inertial-frames:6})।
\end{proof}

\begin{proposition}[पृथ्वी पर कोरिओलिस बल]\label{prop:b1:non-inertial-frames:coriolis}
अक्षांश $\lambda$ पर क्षैतिज दिशा में $v'$ चाल से चलता कोई पिंड इस परिमाण
का क्षैतिज \omterm{thm:b1:non-inertial-frames:dynamics}{कोरिओलिस बल} अनुभव करता है
\[
F_C = 2m\Omega v'\sin\lambda ,
\]
जो उत्तरी गोलार्ध में गति के \emph{दाईं} ओर होता है और दक्षिणी में बाईं
ओर, चाहे यात्रा की दिशा कोई भी हो; और ऊर्ध्वाधर गिरता कोई पिंड
\emph{पूर्व की ओर} $2m\Omega v'\cos\lambda$ से विक्षेपित होता है।
\end{proposition}

\begin{proof}
$\vect\Omega = \Omega(\cos\lambda\,\vect e_N + \sin\lambda\,\vect e_z)$ में वियोजित
कीजिए (उत्तर और स्थानीय ऊर्ध्वाधर)। क्षैतिज $\vect v'$ के लिए $-2m\vect\Omega\wedge\vect v'$
का एक क्षैतिज भाग $-2m\Omega\sin\lambda\,\vect e_z\wedge\vect v'$ है --- यानी
$90^\circ$ घड़ी की दिशा में $\vect v'$ से घूमा हुआ, ऊपर से देखने पर, $\lambda > 0$ के लिए ---
और एक ऊर्ध्वाधर भाग (गुरुत्व के सामने नगण्य)। ऊर्ध्वाधर $\vect v' =
-v'\vect e_z$ के लिए: $-2m\Omega\cos\lambda\,\vect e_N\wedge(-v'\vect e_z) = 2m\Omega v'\cos\lambda\,
\vect e_E$, यानी पूर्व की ओर।
\end{proof}

\begin{example}[छोटे बल, बड़े परिणाम]\label{ex:b1:non-inertial-frames:coriolisnum}
अक्षांश $45^\circ$ पर \qty{100}{km/h} की कोई रेलगाड़ी: $a_C = 2 \times 7.29\times
10^{-5} \times 27.8 \times 0.71 = \qty{2.9e-3}{m/s^2}$, यानी
$g$ का तीन-हज़ारवाँ भाग --- एकतरफ़ा पटरी की दाहिनी रेल कुछ तेज़ी से घिसती
है। किसी निम्न-दाब केंद्र की ओर \qty{20}{m/s} से बहती हवा भीतर आते-आते
दाईं ओर मुड़ती जाती है और अंत में उस केंद्र का चक्कर घड़ी की उलटी दिशा में
लगाने लगती है: उत्तरी गोलार्ध का हर चक्रवात उसी दिशा में घूमता है और हर
दक्षिणी उसकी उलटी दिशा में। $45^\circ$ पर \qty{100}{m} गहरे किसी कूप में
गिराया गया पत्थर साहुल-रेखा से \qty{1.6}{cm} पूर्व की ओर गिरता है
(\cref{exo:b1:non-inertial-frames:7})।
\end{example}

\begin{proposition}[फूको का लोलक]\label{prop:b1:non-inertial-frames:foucault}
\index{फूको लोलक}
अक्षांश $\lambda$ पर छोटे \omterm{def:b1:wave-propagation:signal}{आयाम} से झूलता कोई लोलक अपने दोलन-तल को \omterm{def:b1:non-inertial-frames:earth}{भूकेंद्रीय
तंत्र} के केवल स्थानीय ऊर्ध्वाधर घूर्णन के सापेक्ष स्थिर रखता है: \omterm{def:b1:non-inertial-frames:earth}{पार्थिव
तंत्र} में देखने पर वह तल (ऊपर से, उत्तरी गोलार्ध में) \emph{घड़ी की दिशा
में} कोणीय दर $\Omega\sin\lambda$ से घूमता है, अर्थात् एक पूरा चक्कर इतने
समय में
\[
T_F = \frac{\qty{23.93}{h}}{\sin\lambda} .
\]
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
ध्रुव पर ($\lambda = 90^\circ$) तर्क ज्यामितीय है: तनाव को छोड़कर गोलक पर
कोई क्षैतिज बल नहीं लगता, और तनाव का क्षैतिज घटक झूलने के अनुदिश ही होता
है; अतः \omterm{def:b1:non-inertial-frames:earth}{भूकेंद्रीय तंत्र} में झूलने का तल स्थिर रहता है जबकि पृथ्वी उसके
नीचे $\Omega$ से घूमती जाती है; भूमि से देखने पर वह तल
$-\Omega$ से घूमता है। अक्षांश $\lambda$ पर क्षैतिज \omterm{thm:b1:non-inertial-frames:dynamics}{कोरिओलिस बल}
$-2m\Omega\sin\lambda\,\vect e_z\wedge\vect v'$ है: ठीक वही, जो स्थानीय
ऊर्ध्वाधर के परितः $\Omega\sin\lambda$ से घूमता कोई तंत्र उत्पन्न करता। तल
के साथ $-\Omega\sin\lambda$ से घूमते किसी तंत्र में वह बल प्रथम कोटि तक
लुप्त हो जाता है और लोलक एक साधारण तलीय दोलक भर रह जाता है; अतः \omterm{def:b1:non-inertial-frames:earth}{पार्थिव
तंत्र} में तल $\Omega\sin\lambda$ से पुरस्सरण करता है। उपेक्षित
द्वितीय-कोटि पदों सहित पूरी गणना सप्ताहांत समस्या में है।
\end{proof}

\begin{remark}[ज्वार-भाटा, दो पंक्तियों में]\label{rem:b1:non-inertial-frames:tides}
\index{ज्वार-भाटा}
पृथ्वी के केंद्र के तंत्र में, जो चंद्रमा की ओर मुक्त रूप से गिर रहा है,
चंद्रमा का आकर्षण एकसमान नहीं है: निकट की ओर कुछ अधिक प्रबल, दूर की ओर कुछ
क्षीण। यह अंतर, कोटि $2GM_{\text{चंद्र}}R/d^3 \approx \qty{1e-6}{m/s^2}$,
महासागरों को दो उभारों में खींच देता है ---
यानी दिन में दो ज्वार --- और सूर्य का योगदान, जो चंद्रमा के योगदान का
$46\%$ है, दोनों के एक रेखा में आने या समकोण पर पड़ने के अनुसार बृहत् और
लघु ज्वार बनाता है (\cref{exo:b1:non-inertial-frames:12})।
\end{remark}

दूर एक बच्चा बैठा है। घूर्णी तंत्र में अपकेंद्र त्वरण, $g$ के अंश के रूप

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:3}
पृथ्वी का \omterm{cor:b1:point-kinematics:circular}{कोणीय वेग} $\Omega$, भूमध्य रेखा पर अपकेंद्र त्वरण, और $g$ की

उड़ जातीं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:4}

रेल दबती है? और पूर्व की ओर जाते हुए?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:5}
\qty{10}{cm} त्रिज्या की कोई बाल्टी \qty{2.0}{rad/s} से घूमती है, फिर

घूमता पारद-कुंड अच्छा दूरदर्शी दर्पण क्यों बनता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:6}
दिखाइए कि अक्षांश $\lambda$ पर साहुल-रेखा पृथ्वी के केंद्र से लगभग

चाप-मिनटों में निकालिए। केवल घूर्णन के कारण भूमध्य रेखा पर $g$ ध्रुव की

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:7}
अक्षांश $45^\circ$ पर \qty{100}{m} गहरे किसी कूप में एक पत्थर विराम से

$v = gt$ के साथ एक छोटा विक्षोभ मानिए; तली पर पूर्व की ओर विचलन निकालने के

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:8}

का पुरस्सरण-आवर्तकाल। पेरिस में एक घंटे में तल कितने कोण से घूम जाता है?

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:9}
$45^\circ$ उत्तर पर किसी निम्न-दाब केंद्र की ओर \qty{20}{m/s} से हवा बहती

छोटे भँवर (चौबच्चे का नाला, बवंडर) इस नियम का पालन क्यों नहीं करते?

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:10}
डोरी से बँधा हीलियम का कोई गुब्बारा किसी कार के भीतर तैर रहा है। कार जब
आगे की ओर त्वरित होती है, तो गुब्बारा किस ओर झुकता है? आभासी गुरुत्व और

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:11}
कोई अंतरिक्ष स्थानक \qty{100}{m} त्रिज्या का एक वलय है, जिसे इतना घुमाया

\section{समस्या: फूको का लोलक}

\begin{problem}[{सप्ताहांत समस्या --- किसी गुंबद के नीचे 67 मीटर तार पर
28 किलोग्राम का गोलक: उसके झूलने का तल घूमता क्यों है, कितनी तेज़ी से, वह
सिद्ध क्या करता है, और यह करने वाला बल कितना छोटा है}]
\label{pb:b1:non-inertial-frames:1}
1851 में $\ell = \qty{67}{m}$ लंबाई और $m = \qty{28}{kg}$ द्रव्यमान का कोई
लोलक पेरिस के पैंथियन के गुंबद के नीचे, अक्षांश $\lambda =
48.85^\circ$ पर, लगभग $x_0 = \qty{3.0}{m}$ के \omterm{def:b1:wave-propagation:signal}{आयाम} से झूला था। $\Omega =
\qty{7.29e-5}{rad/s}$, $g = \qty{9.81}{m/s^2}$।

\medskip
\noindent\textbf{भाग I --- लोलक स्वयं।}
\begin{enumerate}
  \item छोटे दोलनों का आवर्तकाल; जाँचिए कि \omterm{def:b1:wave-propagation:signal}{आयाम} छोटा है
        ($x_0/\ell$)।
  \item गोलक की अधिकतम चाल, और निम्नतम बिंदु से उसकी अधिकतम ऊँचाई।
  \item निम्नतम बिंदु पर तनाव, $mg$ के मात्रक में।
  \item अधिकतम चाल पर गोलक पर हवा का कर्षण आँकिए (द्विघाती नियम,
        $C_xS \approx \qty{0.05}{m^2}$, $\rho = \qty{1.2}{kg/m^3}$) और
        अधिकतम प्रत्यानयन बल $m\omega_0^2x_0$ से तुलना कीजिए।
  \item लंबा तार और भारी गोलक क्यों? (दो कारण, एक आवर्तकाल का, एक
        अवमंदन का।)
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ध्रुव पर।}
वही लोलक उत्तरी ध्रुव पर कल्पित कीजिए।
\begin{enumerate}[resume]
  \item \omterm{def:b1:non-inertial-frames:earth}{भूकेंद्रीय तंत्र} में गोलक पर लगने वाले बलों की सूची बनाइए और
        दिखाइए कि कोई भी उसके झूलने का तल नहीं घुमा सकता।
  \item इससे निकालिए कि बर्फ़ पर खड़ा कोई प्रेक्षक उस तल को क्या करते
        देखता है, और वह कितने समय में अपनी आरंभिक दिशा में लौट आता है।
  \item भूमि पर गोलक के पथ का वर्णन कीजिए (एक गुलाब-आकृति): एक दिन में
        कितनी पंखुड़ियाँ?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- अक्षांश $\lambda$ पर: कोरिओलिस पद।}
स्थानीय अक्षें लीजिए: $x$ पूर्व, $y$ उत्तर, $z$ ऊपर; दोलन छोटे हैं, अतः
गति लगभग क्षैतिज है, $\vect v' \approx (\dot x, \dot y, 0)$; और प्रत्यानयन
बल $-m\omega_0^2(x, y, 0)$ है, जहाँ $\omega_0 = \sqrt{g/\ell}$।
\begin{enumerate}[resume]
  \item इन अक्षों में $\vect\Omega$ लिखिए।
  \item $-2m\vect\Omega\wedge\vect v'$ निकालिए और उसके क्षैतिज घटक रखिए;
        दिखाइए कि उनमें केवल $\Omega\sin\lambda$ आता है।
  \item $x$ और $y$ के लिए गति के दोनों समीकरण लिखिए;
        $\omega_F = \Omega\sin\lambda$ रखिए और $\omega_F$ की $\omega_0$ से
        तुलना कीजिए।
  \item $u = x + jy$ रखिए और दिखाइए कि $\ddot u + 2j\omega_F\dot u +
        \omega_0^2u = 0$।
  \item $u = \eu^{j\alpha t}$ आज़माइए: $\alpha$ के दोनों मान निकालिए, और
        दिखाइए कि $\omega_F \ll \omega_0$ के लिए वे $\alpha \approx \pm\omega_0
        - \omega_F$ हैं।
  \item इससे निकालिए कि $u \approx \eu^{-j\omega_Ft}(A\eu^{j\omega_0t} + B\eu^{-j\omega_0t})$:
        गुणक $\eu^{-j\omega_Ft}$ की व्याख्या कीजिए --- झूलने का तल किस
        दिशा में, और किस दर से घूमता है?
  \item पेरिस में पुरस्सरण-आवर्तकाल और एक घंटे में घूमा हुआ कोण निकालिए।
  \item भूमध्य रेखा पर तल क्यों नहीं घूमता?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह बल कितना छोटा है।}
\begin{enumerate}[resume]
  \item गोलक पर अधिकतम \omterm{thm:b1:non-inertial-frames:dynamics}{कोरिओलिस बल} निकालिए और उसकी तुलना भार से तथा
        अधिकतम प्रत्यानयन बल से कीजिए।
  \item आधे आवर्तकाल में झूले का अंतिम बिंदु पार्श्व में कितना खिसक जाता
        है? (तल $\omega_FT/2$ से घूमता है।)
  \item 1851 में गोलक पर एक शलाका लगी थी, जो \omterm{def:b1:wave-propagation:signal}{आयाम} पर वृत्त में रखी छोटी
        खूँटियाँ गिराती जाती थी: अगली खूँटी गिरने में लगभग कितना समय
        लगता था?
  \item लोलक को उस धागे को जलाकर छोड़ा गया था जो गोलक को एक ओर थामे था।
        उसे हाथ से धकेलकर क्यों नहीं छोड़ा गया?
  \item हवा का कर्षण किसी लोलक को कुछ घंटों में रोक देता है। आधुनिक
        संग्रहालयों के लोलक नीचे लगे एक छोटे विद्युत्चुंबक से ऊर्जा
        लौटाते हैं, बिना झूलने के तल को छुए: वह धक्का कड़ाई से त्रिज्य ही
        क्यों होना चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- वह सिद्ध क्या करता है, और और क्या-क्या करता है।}
\begin{enumerate}[resume]
  \item फूको के प्रदर्शन ने जनता को पृथ्वी के घूर्णन का विश्वास दिलाया।
        पुरस्सरण-दर ठीक-ठीक किस राशि को मापती है, और क्या यह प्रयोग
        अक्षांश निर्धारित कर सकता था?
  \item किस अक्षांश पर तल ठीक \qty{48}{h} में एक चक्कर लगाता?
  \item कोई घूर्णदर्शी (जिम्बलों में तेज़ी से घूमता पहिया) अपनी अक्ष को
        \omterm{def:b1:non-inertial-frames:earth}{भूकेंद्रीय तंत्र} में स्थिर रखता है। एक वाक्य में समझाइए कि वह भी
        घूर्णन को क्यों उजागर करता है, और उसका एक अनुप्रयोग बताइए।
  \item सार दीजिए: इसके लिए ज़िम्मेदार वह एक \omterm{thm:b1:non-inertial-frames:dynamics}{जड़त्वीय बल}, अक्षांश पर उसकी
        निर्भरता, और पेरिस में वह जो आवर्तकाल देता है।
\end{enumerate}
\end{problem}
