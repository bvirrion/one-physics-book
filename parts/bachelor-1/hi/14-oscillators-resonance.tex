\chapter{यांत्रिक दोलित्र: अवमंदन और अनुनाद}\label{ch:b1:oscillators-resonance}

किसी बच्चे के झूले को गलत क्षण पर धकेलिए और कुछ ख़ास नहीं होता; उसे प्रति
आवर्तकाल एक बार, हलके से धकेलिए, और दस-बारह झूलों में बच्चा चाप के शिखर पर
चीख़ रहा होता है। कपड़े धोने की कोई मशीन अपने घुमाव-चक्र की एक ख़ास चाल पर
ज़ोर से थरथराती है और उससे ऊपर शांत हो जाती है। शराब का कोई गिलास तब गाने
लगता है जब गीली उँगली ठीक उपयुक्त दर से उसकी कोर पर घूमती है, और तब चटक
जाता है जब कोई गायक वही स्वर पर्याप्त ऊँची आवाज़ में पकड़े रखता है। इनमें
से हर एक \emph{अनुनाद} है: कोई निकाय, जिसकी अपनी एक \omterm{def:b1:oscillators-resonance:harmonic}{स्वाभाविक आवृत्ति} है,
उसी के आसपास प्रणोदित होने पर चक्र दर चक्र ऊर्जा जमा करता जाता है, जब तक
घर्षण उसे उतनी ही तेज़ी से बहा न ले जाए जितनी तेज़ी से वह आ रही है। यह
अध्याय यांत्रिक दोलित्र को लेता है --- मुक्त, अवमंदित, फिर प्रणोदित --- और
वे अनुनाद-वक्र व्युत्पन्न करता है जिनके विद्युत जुड़वाँ
\cref{ch:b1:sinusoidal-impedance} में आ चुके हैं।

\section{आवर्त दोलित्र}

\begin{definition}[आवर्त दोलित्र]\label{def:b1:oscillators-resonance:harmonic}
\index{आवर्त दोलित्र}\index{स्वाभाविक आवृत्ति}
कोई निकाय, जिसका किसी स्थायी साम्यावस्था से विस्थापन $x$ इसका पालन करता है
\[
\ddot x + \omega_0^2\,x = 0
\]
वह \emph{स्वाभाविक \omterm{def:b1:wave-propagation:signal}{कोणीय आवृत्ति}} $\omega_0$ (आवर्तकाल
$T_0 = 2\pi/\omega_0$) वाला \emph{आवर्त दोलित्र} है। हल $x = A\cos(\omega_0t
+ \varphi)$ का \omterm{def:b1:wave-propagation:signal}{आयाम} और \omterm{def:b1:wave-propagation:signal}{कला} आरंभिक शर्तों से तय होते हैं, और उसका आवर्तकाल
\omterm{def:b1:wave-propagation:signal}{आयाम} से स्वतंत्र होता है (\emph{समकालिकता})।
\end{definition}

\begin{proposition}[आवर्त दोलित्र कहाँ-कहाँ से आते हैं]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item $m$ द्रव्यमान, $k$ दृढ़ता की स्प्रिंग पर: $\omega_0^2 = k/m$ (\omterm{prop:b1:newton-dynamics:springstring}{हुक का
        नियम})।
  \item $\ell$ लंबाई का \omterm{ex:b1:newton-dynamics:pendulum}{सरल लोलक}, छोटे कोणों पर: $\omega_0^2 = g/\ell$।
  \item किसी \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} $E_p$ की किसी स्थायी साम्यावस्था के पास कोई भी
        कण: $\omega_0^2 = E_p''(x_0)/m$ (\cref{prop:b1:work-and-energy:stability})।
  \item LC परिपथ: $\omega_0^2 = 1/LC$, जहाँ $q \leftrightarrow x$,
        $L \leftrightarrow m$, $1/C \leftrightarrow k$।
\end{itemize}
ऊर्जा $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ अचर रहती है, और प्रति आवर्तकाल दो बार गतिज तथा स्थितिज रूप
के बीच छलकती है।
\end{proposition}

\begin{proof}
$F = -kx$ के साथ न्यूटन का दूसरा नियम; लोलक का समीकरण
$\ddot\theta + (g/\ell)\sin\theta = 0$, जिसमें $\sin\theta \approx \theta$;
$E_p$ के निम्निष्ठ पर टेलर का सूत्र; और LC परिपथ का \omterm{thm:b1:dc-circuits:kirchhoff}{पाश नियम}। ऊर्जा:
अवकलन कीजिए और समीकरण का प्रयोग कीजिए।
\end{proof}

\begin{example}[डोलता बेलन]\label{ex:b1:oscillators-resonance:bob}
$S$ अनुप्रस्थ काट का कोई ऊर्ध्वाधर बेलन $h$ गहराई तक डूबा हुआ तैरता है
($mg = \rho_wgSh$, उत्प्लावन \cref{ch:b1:fluid-statics} से)। उसे $x$ नीचे
दबाने पर उस पर ऊपर की ओर अतिरिक्त बल $\rho_wgSx$ लगता है:
$m\ddot x = -\rho_wgSx$, अतः $\omega_0^2 = \rho_wgS/m = g/h$ --- ठीक उतना ही
जितना $h$ लंबाई के किसी लोलक का। \qty{1}{m} तक डूबा कोई प्लव \qty{2}{s} के
आवर्तकाल से डोलता है, उसका द्रव्यमान चाहे जो हो।
\end{example}

\section{अवमंदित दोलित्र}

\begin{proposition}[अवमंदित दोलित्र]\label{prop:b1:oscillators-resonance:damped}
\index{अवमंदित दोलित्र}\index{गुणता गुणक!यांत्रिक}
श्यान घर्षण $-\alpha\dot x$ के साथ समीकरण यह हो जाता है
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
यानी \cref{ch:b1:transient-regimes} का प्रामाणिक द्वितीय कोटि का रूप:
$Q > \tfrac12$ के लिए एक छद्म-आवर्ती क्षय $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$, जिसमें $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$; और
ऊर्जा $\eu^{-2t/\tau}$ की तरह क्षीण होती है तथा
\[
Q = 2\pi\,\frac{\text{संचित ऊर्जा}}{\text{प्रति आवर्तकाल खोई ऊर्जा}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
$m\ddot x + \alpha\dot x + kx = 0$ को $m$ से भाग दीजिए। ऊर्जा के लिए: $E
\propto A^2\eu^{-2t/\tau}$ प्रति आवर्तकाल $2T/\tau = 2\pi/Q$ अंश खो देती
है, जब वह छोटा हो।
\end{proof}

\begin{example}[गूँजने के समय]\label{ex:b1:oscillators-resonance:ringing}
\qty{440}{Hz} का कोई स्वरित्र द्विभुज ठोके जाने के दस सेकंड बाद भी सुनाई
देता है: उसका \omterm{def:b1:wave-propagation:signal}{आयाम} लगभग $5\%$ तक गिर गया, वह भी $3\tau = \qty{10}{s}$ में, अतः
$\tau = \qty{3.3}{s}$ और $Q = \omega_0\tau/2 \approx 4600$। किसी कार का
निलंबन ($Q \approx 0.7$) एक ही आवर्तकाल में थम जाता है; किसी बच्चे का झूला
($Q \approx 20$) प्रति झूले अपनी ऊर्जा का एक तिहाई खो देता है और हर बार
धक्का माँगता है।
\end{example}

\section{प्रणोदित दोलन और अनुनाद}

\begin{theorem}[ज्यावक्रीय रूप से प्रणोदित दोलित्र]\label{thm:b1:oscillators-resonance:forced}
\index{प्रणोदित दोलन}\index{अनुनाद!आयाम का}\index{अनुनाद!वेग का}
बल $F_0\cos\omega t$ से प्रणोदित होने पर \omterm{prop:b1:oscillators-resonance:damped}{अवमंदित दोलित्र}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
$\sim\tau$ अवधि के किसी क्षणिक चरण के बाद \omterm{prop:b1:sinusoidal-impedance:forced}{प्रणोदित व्यवस्था}
$x = X\cos(\omega t + \varphi)$ में बैठ जाता है, जहाँ $x_r = \omega/\omega_0$
और स्थैतिक विक्षेप $X_0 = F_0/k$ लिखने पर,
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
\emph{\omterm{def:b1:wave-propagation:signal}{आयाम}} $X$ का शिखर, $Q > 1/\sqrt2$ के लिए, $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$ पर आता है, जहाँ $X_{\max} \approx QX_0$;
\emph{वेग-आयाम} $V = \omega X$ का शिखर ठीक $\omega_0$ पर आता है, जहाँ $V_{\max} =
F_0/\alpha$, और उसके अनुनाद-वक्र
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
की बैंड-चौड़ाई $\Delta\omega = \omega_0/Q$ है। $\omega_0$ पर विस्थापन बल से
$\pi/2$ पीछे रहता है और वेग उसके साथ समकला रहता है।
\end{theorem}

\begin{proof}
सम्मिश्र विधि (\cref{ch:b1:sinusoidal-impedance}): $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$, अतः $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$; अंश और हर दोनों को $\omega_0^2 = k/m$ से भाग
दीजिए। मापांक और कोणांक इसके बाद निकल आते हैं; $X$ का उच्चिष्ठ
\cref{prop:b1:sinusoidal-impedance:overvoltage} में निकाला जा चुका है (वही
फलन)। वेग: $\underline V = j\omega\underline X$ से $V = \omega X$ मिलता है;
और $(1 - x_r^2)/x_r = 1/x_r - x_r$ लिखने पर हर बदलकर
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$ हो जाता है, अर्थात् RLC वाला रूप। $x_r = 1$
पर $\underline X = -jQX_0$: यानी $\pi/2$ का पिछड़ाव।
\end{proof}

\begin{remark}[दो अनुनाद, एक परिपथ]\label{rem:b1:oscillators-resonance:analogy}
\index{विद्युत--यांत्रिक सादृश्य}
संगति $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$,
$F \leftrightarrow e$ प्रणोदित दोलित्र को \cref{ch:b1:sinusoidal-impedance}
के श्रेणी RLC पर प्रतिचित्रित कर देती है: वेग-अनुनाद ही धारा का अनुनाद है
(सदा $\omega_0$ पर, बैंड-चौड़ाई $\omega_0/Q$), और आयाम-अनुनाद आवेश का, यानी
संधारित्र-विभवांतर का अनुनाद है ($\omega_0$ से थोड़ा नीचे, और कम $Q$ पर
होता ही नहीं)। अवशोषित शक्ति, $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$, $\omega_0$ पर शिखर पर पहुँचती है, जहाँ उसका मान
$F_0^2/2\alpha$ है, और अर्ध-शक्ति बिंदु वही वेग-बैंड-चौड़ाई देते हैं ---
ठीक जैसे RLC के लिए।
\end{remark}

\begin{example}[अनुनादी चालें]\label{ex:b1:oscillators-resonance:road}
किसी कार के निलंबन का $f_0 \approx \qty{1.3}{Hz}$ होता है (\cref{pb:b1:transient-regimes:1})।
हर $\lambda = \qty{10}{m}$ पर नालियों वाली किसी सड़क पर प्रणोदन आवृत्ति
$v/\lambda$ होती है: अतः अनुनाद $v = \lambda f_0 = \qty{13}{m/s}$ पर, यानी
लगभग \qty{47}{km/h} --- वही चाल जिस पर नालीदार सड़क कार को सबसे अधिक हिलाती
है, और यही कारण है कि $Q \approx 0.7$ वाला अवमंदक लगाया जाता है: $Q = 5$
होता तो पिंड सड़क के \omterm{def:b1:wave-propagation:signal}{आयाम} से पाँच गुना उछलता। अनुनाद से ऊपर ($x_r \gg 1$)
\omterm{def:b1:wave-propagation:signal}{आयाम} $X_0/x_r^2$ की तरह गिरता है: तेज़ी से आते उभारों के ऊपर से पिंड
``तैर'' जाता है --- यानी पृथक्करण, अनुनाद का दूसरा चेहरा।
\end{example}

\section{आधार-प्रणोदन: पृथक्करण और भूकंपमापी}

\begin{proposition}[अपने आधार से हिलाया गया दोलित्र]\label{prop:b1:oscillators-resonance:base}
\index{आधार-प्रणोदन}\index{कंपन-पृथक्करण}\index{भूकंपमापी}
मान लीजिए किसी द्रव्यमान--स्प्रिंग--अवमंदक का आधार $x_g(t)$ से चलता है
(कार के नीचे की सड़क, भूकंपमापी के नीचे की भूमि), और $u$ आधार के सापेक्ष
द्रव्यमान का विस्थापन हो तथा $x = x_g + u$ उसका निरपेक्ष विस्थापन
(साम्यावस्था से)। तब
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
और $x_g = X_g\cos\omega t$ के लिए, जहाँ $x_r = \omega/\omega_0$:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
$x_r \gg 1$ के लिए: $U \to X_g$ (द्रव्यमान अंतरिक्ष में स्थिर रहता है ---
कोई \emph{भूकंपमापी} भूमि का विस्थापन पढ़ता है) और $X \to 0$ (द्रव्यमान
हिलने से \emph{पृथक} रहता है); और $x_r \ll 1$ के लिए: $U \approx
\ddot x_g/\omega_0^2$ (कोई \emph{त्वरणमापी}) तथा $X \approx X_g$ (द्रव्यमान
आधार के पीछे-पीछे चलता है)।
\end{proposition}

\begin{proof}
स्प्रिंग और अवमंदक सापेक्ष गति पर लगते हैं: $m\ddot x = -ku -
\alpha\dot u$, और $\ddot x = \ddot x_g + \ddot u$। सम्मिश्र रूप में
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$, जिससे
$U/X_g$ मिलता है; और फिर $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$।
\end{proof}

\section{ऊर्जा: कोई अनुनाद कैसे बनता है}

\begin{proposition}[प्रणोदित व्यवस्था में ऊर्जा-संतुलन]\label{prop:b1:oscillators-resonance:energy}
स्थायी \omterm{prop:b1:sinusoidal-impedance:forced}{प्रणोदित व्यवस्था} में प्रणोदक बल द्वारा दी गई \omterm{thm:b1:sinusoidal-impedance:power}{औसत शक्ति} घर्षण
द्वारा क्षय की गई \omterm{thm:b1:sinusoidal-impedance:power}{औसत शक्ति} $\tfrac12\alpha V^2$ के बराबर होती है।
अनुनाद पर संचित ऊर्जा $\tfrac12 mV_{\max}^2$ प्रति आवर्तकाल दी गई ऊर्जा की
$Q/2\pi$ गुनी होती है: दोलित्र प्रणोदन का कार्य लगभग $Q$ चक्रों तक जमा
करता है, तब कहीं जाकर हानियाँ आवक के बराबर होती हैं --- और उतने ही चक्र
($\sim\tau/T$) क्षणिक चरण भी चलता है।
\end{proposition}

\begin{proof}
ऊर्जा प्रमेय का एक आवर्तकाल पर औसत लीजिए: \omterm{def:b1:transient-regimes:firstorder}{स्थायी अवस्था} में
$\langle\dd E/\dd t\rangle = 0$, अतः $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$। अनुनाद पर $V_{\max} = F_0/\alpha$,
संचित $\tfrac12 m(F_0/\alpha)^2$, प्रति आवर्त दी गई $\tfrac12\alpha(F_0/\alpha)^2T$;
अनुपात $m/(\alpha T) = Q/2\pi$।
\end{proof}

\begin{example}[झूला]\label{ex:b1:oscillators-resonance:swing}
\qty{2.5}{m} लंबाई का कोई झूला ($T_0 = \qty{3.2}{s}$, $Q \approx 20$) हर
आवर्तकाल में \qty{0.5}{m} पर \qty{50}{N} का धक्का पाता है: यानी प्रति चक्र
\qty{25}{J}। वह तब तक बढ़ता है जब तक प्रति आवर्तकाल की हानि $2\pi E/Q$
\qty{25}{J} के बराबर न हो जाए: $E = \qty{80}{J}$; और \qty{30}{kg} के बच्चे
के साथ $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ से $\theta_{\max} \approx 27^\circ$ मिलता
है। किसी और लय पर धकेलने से धक्कों का कार्य चिह्न बदलता रहता और औसत में
कुछ भी न बचता।
\end{example}

\section{समस्या: भूकंपलेखी}

\begin{problem}[{सप्ताहांत समस्या --- भूमि से कसे किसी डिब्बे के भीतर
स्प्रिंग से एक द्रव्यमान लटका है; भूमि हिलती है: कलम क्या दर्ज करती है,
वही यंत्र ऊँची आवृत्ति पर विस्थापन और नीची आवृत्ति पर त्वरण क्यों मापता
है, और लंबे आवर्तकाल का कोई भूकंपमापी केवल एक लंबी स्प्रिंग क्यों नहीं हो
सकता}]
\label{pb:b1:oscillators-resonance:1}
भूमि से जुड़े किसी दृढ़ ढाँचे के भीतर $m = \qty{1.0}{kg}$ का कोई द्रव्यमान
$k$ दृढ़ता की स्प्रिंग से लटका है; कोई अवमंदक $-\alpha\dot u$ लगाता है, जहाँ
$u$ ढाँचे के सापेक्ष द्रव्यमान का विस्थापन है, जिसे उसकी साम्यावस्था की
स्थिति से नापा जाता है। भूमि, और इसलिए ढाँचा, $x_g(t)$ से ऊर्ध्वाधर दिशा
में चलते हैं। दूर के तारों का तंत्र जड़त्वीय है।

\medskip
\noindent\textbf{भाग I --- समीकरण।}
\begin{enumerate}
  \item द्रव्यमान की निरपेक्ष स्थिति $x = x_g + u +
        \text{नियतांक}$ लिखिए, द्रव्यमान पर लगते बलों की सूची बनाइए
        (\omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} में), और दिखाइए कि
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item उसे प्रामाणिक रूप में लाइए; $\omega_0$ और $Q$ पहचानिए।
  \item यह यंत्र $f_0 = \qty{0.50}{Hz}$ और $Q = 0.70$ के साथ बनाया गया है:
        $k$ और $\alpha$ निकालिए।
  \item कोई अचर भूमि-त्वरण $a_g$ (मान लीजिए ढाँचे का झुक जाना) कौन-सा
        स्थायी $u$ देता है? व्याख्या कीजिए।
  \item द्रव्यमान से जुड़ी कलम, जो ढाँचे पर टिके किसी ढोल पर लिखती है, क्या
        दर्ज करती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- आवर्त भूमि-गति।}
भूमि $x_g = X_g\cos\omega t$ की तरह चलती है; $x_r = \omega/\omega_0$ लिखिए।
\begin{enumerate}[resume]
  \item सम्मिश्र आयामों का प्रयोग करके दिखाइए कि $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$।
  \item $U/X_g$ को $x_r$ और $Q$ के फलन के रूप में निकालिए।
  \item सीमा $x_r \gg 1$: दिखाइए कि $U \to X_g$, और भौतिक रूप से समझाइए कि
        द्रव्यमान करता क्या है (यंत्र माप क्या रहा है?)।
  \item सीमा $x_r \ll 1$: दिखाइए कि $U \approx x_r^2X_g = a_g/\omega_0^2$,
        जहाँ $a_g$ भूमि के त्वरण का \omterm{def:b1:wave-propagation:signal}{आयाम} है: अब यंत्र क्या है?
  \item $x_r = 1$ पर मान; $Q \approx 0.7$ समझदारी भरा चुनाव क्यों है?
  \item $U/X_g$ का $x_r$ के सापेक्ष रेखाचित्र बनाइए (लघुगणकीय पैमाने),
        $Q = 0.7$ और $Q = 5$ के लिए।
  \item ऊँची आवृत्ति पर $u$ की \omterm{def:b1:wave-propagation:signal}{कला}, $x_g$ के सापेक्ष; व्याख्या कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- भूकंपों के लिए अभिकल्प।}
दूर के भूकंप \qty{0.05}{} से \qty{0.1}{Hz} तक की पृष्ठ-तरंगें पैदा करते हैं,
जिनके \omterm{def:b1:wave-propagation:signal}{आयाम} मिलीमीटर भर के होते हैं; स्थानीय झटके \qty{5}{} से \qty{10}{Hz}
तक पहुँचते हैं।
\begin{enumerate}[resume]
  \item \qty{0.05}{Hz} की तरंगों को विस्थापन के रूप में दर्ज करने के लिए
        $f_0 \ll \qty{0.05}{Hz}$ चाहिए, मान लीजिए \qty{0.02}{Hz}।
        द्रव्यमान के भार के नीचे स्प्रिंग कितनी स्थैतिक रूप से खिंचेगी?
        टिप्पणी कीजिए।
  \item उसी आवृत्ति का कोई \omterm{ex:b1:newton-dynamics:pendulum}{सरल लोलक}: कितनी लंबाई?
  \item असली लंबे-आवर्तकाल वाले यंत्र लगभग क्षैतिज ``बाग़-फाटक'' लोलक काम
        में लेते हैं, या इलेक्ट्रॉनिक बल-पुनर्भरण। एक-एक वाक्य में समझाइए
        कि इनमें से हर एक क्या हासिल करता है।
  \item \qty{0.50}{Hz} वाले यंत्र के लिए स्थैतिक खिंचाव निकालिए: क्या यह
        व्यवहार्य है?
  \item उससे, \qty{1.0}{mm} \omterm{def:b1:wave-propagation:signal}{आयाम} की कोई \qty{0.10}{Hz} तरंग कौन-सा $U$ देती
        है? यंत्र किस व्यवस्था में है?
  \item \qty{1.0}{mm} \omterm{def:b1:wave-propagation:signal}{आयाम} का कोई \qty{5.0}{Hz} स्थानीय झटका: $U$? और
        व्यवस्था?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- फ़ोन का त्वरणमापी।}
सिलिकॉन की स्प्रिंगों पर रखे किसी सूक्ष्म-गढ़े द्रव्यमान का
$f_0 = \qty{1.0}{kHz}$ और $Q = 0.7$ है।
\begin{enumerate}[resume]
  \item $g$ त्वरण-आयाम वाला कोई \qty{10}{Hz} का झटका: $U$ निकालिए। ऐसा
        विस्थापन पढ़ा कैसे जाता है?
  \item पाठ किस आवृत्ति तक $10\%$ के भीतर रहता है, $a_g/\omega_0^2$ की तुलना
        में?
  \item किसी झटके के प्रति उसकी अनुक्रिया को कौन-सा \omterm{def:b1:transient-regimes:firstorder}{कालांक}
        $\tau = 2Q/\omega_0$ नियंत्रित करता है, और गिरने का पता लगाने के लिए
        उसका क्या महत्व है?
  \item भूकंपमापी का $\tau$: मान, और किसी अचानक भूमि-सोपान के बाद उसका
        परिणाम।
  \item दोनों यंत्रों के लिए ऊँचे $Q$ के बजाय क्रांतिक (या लगभग क्रांतिक)
        अवमंदन क्यों चुना जाता है?
  \item फ़ोन को ठीक \qty{1.0}{kHz} पर हिलाया जाता है: पाठ कितने गुना ग़लत
        होगा?
  \item सार दीजिए: एक यंत्र, दो व्यवस्थाएँ, और किस एक तुलना से यह तय होता
        है।
\end{enumerate}
\end{problem}
