\chapter{अणुगति सिद्धांत और आदर्श गैस}\label{ch:b1:kinetic-theory}

किसी कक्षा की हवा का भार किसी वयस्क आदमी जितना होता है, और उसमें अणु
पृथ्वी की सारी रेत के कणों से भी अधिक होते हैं; हर अणु किसी राइफ़ल की गोली
की चाल से उड़ता है और सेकंड में दस अरब \omterm{def:b1:kinetic-theory:pressure}{बार} अपने पड़ोसियों से टकराता है।
उनमें से किसी एक का पीछा कोई नहीं कर सकता; और करने की ज़रूरत भी नहीं। कुछ
ही औसत --- \omterm{def:b1:kinetic-theory:pressure}{दाब}, \omterm{def:b1:kinetic-theory:temperature}{ताप}, आयतन, मात्रा --- अभियांत्रिकी के हर प्रयोजन के लिए उस
हवा का पूरा वर्णन कर देते हैं, और दीवारों से टकराकर लौटते अणुओं का एक
सीधा-सादा चित्र बता देता है कि वे औसत आते कहाँ से हैं और आपस में कैसे जुड़े
हैं। यह अध्याय पदार्थ को उसके दोनों सिरों से देखकर इस खंड का ऊष्मागतिकी
वाला भाग खोलता है: सूक्ष्म अव्यवस्था और स्थूल शांति, तथा वह अणुगति
सिद्धांत जो \omterm{def:b1:kinetic-theory:model}{आदर्श गैस} की स्थिति में दोनों को जोड़ देता है।

\section{वर्णन के दो पैमाने}

\begin{definition}[सूक्ष्म और स्थूल पैमाने]\label{def:b1:kinetic-theory:scales}
\index{सूक्ष्म पैमाना}\index{स्थूल पैमाना}\index{आवोगाद्रो नियतांक}\index{मोल}
\emph{सूक्ष्म} पैमाने पर पदार्थ अणुओं से बना है: $\qty{1}{mol}$ में
$N_A = \qty{6.02e23}{}$ अणु होते हैं, और हर एक लगभग \qty{0.3}{nm} चौड़ा;
साधारण परिस्थितियों की किसी गैस में वे \qty{3}{nm} की दूरी पर रहते हैं,
सैकड़ों मीटर प्रति सेकंड से चलते हैं और लगभग हर \qty{70}{nm} पर टकराते
हैं। \emph{स्थूल} पैमाने पर किसी निकाय का वर्णन कुछ ही
\emph{अवस्था चरों} से होता है --- \omterm{def:b1:kinetic-theory:pressure}{दाब} $P$, आयतन $V$, \omterm{def:b1:kinetic-theory:temperature}{ताप} $T$, पदार्थ की
मात्रा $n$ --- जो अणुओं की विशाल संख्याओं पर लिए गए औसत हैं। कोई चर
\emph{विस्तीर्ण} कहलाता है यदि निकाय दुगुना करने पर वह भी दुगुना हो जाए
($V$, $n$, द्रव्यमान, ऊर्जा), और \emph{गहन} यदि न हो
($P$, $T$, घनत्व)।
\end{definition}

\begin{definition}[ऊष्मागतिक साम्य]\label{def:b1:kinetic-theory:equilibrium}
\index{ऊष्मागतिक साम्य}\index{अवस्था चर}
कोई निकाय \emph{ऊष्मागतिक साम्य} में तब होता है जब उसके अवस्था चर एकसमान
और समय में अचर हों और उसे पार करता पदार्थ या ऊर्जा का कोई स्थूल प्रवाह न
हो। तभी पूरे निकाय के लिए $P$ और $T$ परिभाषित होते हैं; और
\emph{अवस्था समीकरण} उन्हें $V$ तथा $n$ से जोड़ता है।
\end{definition}

\begin{definition}[दाब]\label{def:b1:kinetic-theory:pressure}
\index{दाब}\index{पास्कल}\index{बार}
विराम में पड़ा कोई तरल किसी दीवार के हर पृष्ठ-अवयव $\dd S$ पर उस पर लंब
और दीवार की ओर बल $\dd\vect F = P\,\dd S\,\vect n$ लगाता है; $P$ ही
\emph{दाब} है, जो पास्कल में मापा जाता है ($\qty{1}{Pa} = \qty{1}{N/m^2}$); $\qty{1}{bar}
= \qty{1e5}{Pa}$, $\qty{1}{atm} = \qty{1.013e5}{Pa}$। किसी दिए हुए बिंदु पर
वह पृष्ठ की दिशा पर निर्भर नहीं करता (\cref{ch:b1:fluid-statics})।
\end{definition}

\begin{definition}[ताप]\label{def:b1:kinetic-theory:temperature}
\index{ताप}\index{केल्विन}\index{शून्यवाँ नियम}
ऊर्जा को गुज़रने देने वाली किसी दीवार से जुड़े दो निकाय समय के साथ एक
उभयनिष्ठ अवस्था तक पहुँच जाते हैं: तब वे एक ही \emph{ताप} पर होते हैं, और
जो दो पिंड किसी तीसरे के साथ साम्य में हों वे आपस में भी साम्य में रहते
हैं (\emph{शून्यवाँ नियम})। केल्विन मापनी ठीक-ठीक
$k_B = \qty{1.380649e-23}{J/K}$ से नियत है; सेल्सियस: $\theta = T - 273.15$। नीचे
का अणुगति सिद्धांत $T$ को उसका सूक्ष्म अर्थ देता है।
\end{definition}

\section{आदर्श गैस का अणुगति प्रतिरूप}

\begin{definition}[आदर्श गैस, सूक्ष्म प्रतिरूप]\label{def:b1:kinetic-theory:model}
\index{आदर्श गैस}\index{परिपूर्ण गैस}
\emph{आदर्श गैस} (परिपूर्ण गैस) $N$ अणुओं का ऐसा समूह है, जिन्हें $m$
द्रव्यमान के बिंदु मान लिया जाता है और जो $V$ आयतन के किसी पात्र में
स्वतंत्र रूप से तथा यादृच्छिक ढंग से चलते हैं --- छोटी-छोटी टक्करों को
छोड़कर कोई अन्योन्यक्रिया नहीं। साम्य में वेग-वितरण समदैशिक और अचर रहता है;
$n^* = N/V$ संख्या घनत्व है और $\langle v^2\rangle$ माध्य वर्ग चाल।
\end{definition}

\begin{theorem}[गतिज दाब]\label{thm:b1:kinetic-theory:pressure}
\index{गतिज दाब}
गैस दीवारों पर जो \omterm{def:b1:kinetic-theory:pressure}{दाब} लगाती है वह यह है
\[
P = \tfrac13\,n^*m\langle v^2\rangle = \tfrac23\,n^*\langle\epsilon_k\rangle ,
\]
जहाँ $\langle\epsilon_k\rangle = \tfrac12 m\langle v^2\rangle$ किसी अणु की
माध्य \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} है।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
वह सरलीकृत प्रतिरूप लीजिए जिसमें सभी अणुओं की चाल $u$ है और वे तीनों
अक्षों के अनुदिश चलते हैं, हर दिशा में छठा हिस्सा। दीवार $x = 0$ से सीधे
टकराता कोई अणु प्रत्यास्थ रूप से लौट आता है और उसे $2mu$ संवेग दे जाता है।
समय $\dd t$ में, दीवार की ओर चलते जो अणु $S$ क्षेत्रफल तक पहुँचते हैं वे
$u\,\dd t$ मोटी उस परत के भीतर वाले हैं, यानी वहाँ के $n^*Su\,\dd t$ अणुओं
का छठा हिस्सा। बल $=$
प्रति समय संवेग $= (n^*Su\,\dd t/6)(2mu)/\dd t = \tfrac13 n^*mu^2S$, अतः
$P = \tfrac13 n^*mu^2$। चालों और दिशाओं के वास्तविक वितरण पर औसत लेने से
$u^2$ की जगह $\langle v^2\rangle$ आ जाती है (समदैशिकता से
$\langle v_x^2\rangle = \tfrac13\langle v^2\rangle$); वर्ष 2 का खंड इसे
पूरा-पूरा करता है।
\end{proof}

\begin{definition}[गतिज ताप; अवस्था समीकरण]\label{def:b1:kinetic-theory:kinetictemp}
\index{गतिज ताप}\index{अवस्था समीकरण!आदर्श गैस का}\index{गैस नियतांक}\index{वर्ग-माध्य-मूल चाल}
\omterm{def:b1:kinetic-theory:model}{आदर्श गैस} का \omterm{def:b1:kinetic-theory:temperature}{ताप} उसके अणुओं की माध्य \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} से \emph{परिभाषित} होता
है,
\[
\langle\epsilon_k\rangle = \tfrac12 m\langle v^2\rangle = \tfrac32 k_BT ,
\]
अतः \cref{thm:b1:kinetic-theory:pressure} के साथ मिलाने पर,
\[
PV = Nk_BT = nRT, \qquad R = N_Ak_B = \qty{8.314}{J/(mol.K)} ,
\]
यही \emph{\omterm{def:b1:kinetic-theory:model}{आदर्श गैस} का अवस्था समीकरण} है। \emph{वर्ग-माध्य-मूल चाल}
$u = \sqrt{\langle v^2\rangle} = \sqrt{3k_BT/m} = \sqrt{3RT/M}$ है, जहाँ
$M$ मोलर द्रव्यमान है।
\end{definition}

\begin{example}[हवा के आँकड़े]\label{ex:b1:kinetic-theory:air}
\qty{300}{K} पर नाइट्रोजन: $u = \sqrt{3 \times 8.314 \times 300/0.028} =
\qty{517}{m/s}$; हाइड्रोजन, $\qty{1930}{m/s}$; जो जितना भारी वह उतना ही
धीमा, $1/\sqrt M$ की तरह। \qty{1}{bar} और \qty{300}{K} पर, $n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$:
यानी प्रति \omterm{def:b1:kinetic-theory:scales}{मोल} \qty{24}{L}, और एक लीटर में \qty{2.7e22}{} अणु। \qty{200}{m^3}
की किसी कक्षा में: \qty{8300}{mol}, यानी \qty{240}{kg} हवा।
\end{example}

\begin{remark}[प्रतिरूप काम क्यों करता है, और कब विफल होता है]\label{rem:b1:kinetic-theory:validity}
\index{माध्य मुक्त पथ}
दो टक्करों के बीच हवा का कोई अणु \emph{\omterm{rem:b1:kinetic-theory:validity}{माध्य मुक्त पथ}}
$\lambda \approx 1/(\sqrt2\pi d^2n^*) \approx \qty{70}{nm}$ उड़ता है --- यानी
अपने आकार से दो सौ गुना: गैस अधिकतर खाली है, अन्योन्यक्रियाएँ विरल हैं, और
साधारण दाबों पर $PV = nRT$ एक प्रतिशत से भी अच्छी यथार्थता तक लागू रहता
है। ऊँचे \omterm{def:b1:kinetic-theory:pressure}{दाब} या नीचे \omterm{def:b1:kinetic-theory:temperature}{ताप} पर अणुओं का अपना आयतन और उनका आकर्षण मायने रखने
लगते हैं: वान डेर वाल्स का समीकरण $(P + an^2/V^2)(V - nb) = nRT$
दोनों का संशोधन कर देता है, और किसी क्रांतिक \omterm{def:b1:kinetic-theory:temperature}{ताप} से नीचे गैस द्रवित हो
जाती है (\cref{ch:b1:phase-changes})।
\end{remark}

\section{आदर्श गैस की आंतरिक ऊर्जा}

\begin{definition}[आंतरिक ऊर्जा]\label{def:b1:kinetic-theory:U}
\index{आंतरिक ऊर्जा}
किसी निकाय की \emph{आंतरिक ऊर्जा} $U$ उस तंत्र में, जिसमें निकाय विराम में
है, उसके अणुओं की गतिज ऊर्जाओं तथा उनकी पारस्परिक स्थितिज ऊर्जाओं का योग
है; यह एक विस्तीर्ण अवस्था फलन है।
\end{definition}

\begin{proposition}[आदर्श गैसों की आंतरिक ऊर्जा]\label{prop:b1:kinetic-theory:Ugas}
\index{एकपरमाणुक गैस}\index{द्विपरमाणुक गैस}\index{स्थिर आयतन पर ऊष्मा धारिता}\index{जूल के नियम}
\omterm{def:b1:kinetic-theory:model}{आदर्श गैस} के लिए $U$ केवल $T$ पर निर्भर करता है (जूल का पहला नियम):
\begin{gather*}
U = \tfrac32 nRT \quad (\text{एकपरमाणुक: He, Ne, Ar}), \\
U = \tfrac52 nRT \quad (\text{साधारण तापों पर द्विपरमाणुक: N}_2, \text{O}_2, \text{H}_2),
\end{gather*}
अतः स्थिर आयतन पर मोलर ऊष्मा धारिता $C_{V,m} = \dd U_m/\dd T$ क्रमशः
$\tfrac32 R = \qty{12.5}{J/(mol.K)}$ और $\tfrac52 R = \qty{20.8}{J/(mol.K)}$
होती है।
\end{proposition}

\begin{proof}[आंशिक उपपत्ति]
परिकल्पना से कोई पारस्परिक \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} नहीं है, अतः बिंदु-अणुओं के लिए $U = N\langle\epsilon_k\rangle
= \tfrac32 Nk_BT$। कोई द्विपरमाणुक अणु घूमता भी है: साम्य में उसकी दोनों
घूर्णी स्वातंत्र्य-कोटियों में से हर एक उतनी ही $\tfrac12 k_BT$ लिए रहती
है जितनी हर स्थानांतरी कोटि (\emph{ऊर्जा-समविभाजन} प्रमेय, स्वीकृत; वर्ष 3
का खंड उसे व्युत्पन्न करता है और समझाता है कि कंपन एक हज़ार \omterm{def:b1:kinetic-theory:temperature}{केल्विन} से ऊपर
ही क्यों शामिल होता है)।
\end{proof}

\begin{example}[एक कमरे भर की ऊर्जा]\label{ex:b1:kinetic-theory:room}
\qty{300}{K} पर कक्षा की \qty{8300}{mol} हवा: $U = \tfrac52 \times
8300 \times 8.314 \times 300 = \qty{5.2e7}{J}$ --- यानी डेढ़ लीटर पेट्रोल
जितनी ऊर्जा, जो आणविक गति में धरी है। कमरे को $1$ \omterm{def:b1:kinetic-theory:temperature}{केल्विन} गरम करने में
$\tfrac52 nR = \qty{170}{kJ}$ लगते हैं (स्थिर आयतन पर)।
\end{example}

\section{संघनित प्रावस्थाएँ}

\begin{proposition}[असंपीड्य, अप्रसार्य प्रावस्था का प्रतिरूप]\label{prop:b1:kinetic-theory:condensed}
\index{संघनित प्रावस्था}\index{तापीय प्रसार}\index{संपीड्यता}
द्रवों और ठोसों को, इस स्तर पर, स्थिर आयतन की ऐसी प्रावस्थाएँ मान लिया
जाता है --- न संपीड्य, न प्रसार्य --- जिनकी \omterm{def:b1:kinetic-theory:U}{आंतरिक ऊर्जा} केवल $T$ पर
निर्भर करती है: $\dd U = C\,\dd T$, जहाँ $C = mc$ है और $c$ विशिष्ट ऊष्मा
धारिता (पानी के लिए \qty{4180}{J/(kg.K)}, इस्पात के लिए
\qty{450}{J/(kg.K)})। वास्तविक, छोटे विचलन \omterm{prop:b1:kinetic-theory:condensed}{तापीय प्रसार} गुणांक
$\alpha = (1/V)(\partial V/\partial T)_P$ से मापे जाते हैं (धातुओं के लिए
$\sim\qty{1e-5}{K^{-1}}$, द्रवों के लिए \qty{2e-4}{K^{-1}}) और समतापी
संपीड्यता $\chi_T = -(1/V)(\partial V/\partial P)_T$ से (पानी के लिए
$\sim\qty{5e-10}{Pa^{-1}}$, जबकि किसी गैस के लिए $1/P \approx \qty{1e-5}{Pa^{-1}}$)।
\end{proposition}
\admitted

\begin{example}[कितनी असंपीड्य]\label{ex:b1:kinetic-theory:water}
मारियाना खाई की तली का पानी (\qty{1100}{bar})
$\chi_T\Delta P \approx 5\times10^{-10} \times 1.1\times10^8 = 5\%$ संपीडित
होता है; और $50$ \omterm{def:b1:kinetic-theory:temperature}{केल्विन} गरम की गई \qty{30}{m} की इस्पात-पटरी $\alpha L\Delta T =
\qty{18}{mm}$ लंबी हो जाती है --- यही प्रसार-संधियों का कारण है। \omterm{def:b1:kinetic-theory:pressure}{दाब} दुगुना
करने पर जिसका आयतन आधा हो जाता है, उस गैस के सामने स्थिर आयतन का प्रतिरूप
इस खंड की अधिकांश स्थितियों में कुछ प्रतिशत तक ठीक बैठता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:1}
\qty{0}{\degreeCelsius} और \qty{1}{atm} पर \qty{1.0}{L} गैस में अणुओं की
संख्या; पड़ोसियों के बीच माध्य दूरी; और \qty{0.3}{nm} के आणविक आकार से
तुलना।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:2}
\qty{300}{K} पर H$_2$, N$_2$, O$_2$ और CO$_2$ की, तथा \qty{77}{K}
(उबलती नाइट्रोजन) पर N$_2$ की वर्ग-माध्य-मूल चालें।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:3}
\qty{30}{L} आयतन का कोई कार-टायर \qty{20}{\degreeCelsius} पर
\qty{2.5}{bar} के प्रमापी \omterm{def:b1:kinetic-theory:pressure}{दाब} तक भरा जाता है। भीतर की हवा की मात्रा और
द्रव्यमान; और उसे \qty{50}{\degreeCelsius} तक गरम कर देने वाली यात्रा के
बाद निरपेक्ष \omterm{def:b1:kinetic-theory:pressure}{दाब}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:4}
\qty{1}{bar}, \qty{300}{K} पर हवा के अणुओं का \omterm{rem:b1:kinetic-theory:validity}{माध्य मुक्त पथ}
($d = \qty{0.37}{nm}$) आँकिए, और किसी एक अणु की टक्कर-आवृत्ति। किस \omterm{def:b1:kinetic-theory:pressure}{दाब} पर
\omterm{rem:b1:kinetic-theory:validity}{माध्य मुक्त पथ} \qty{1}{m} तक पहुँच जाता है (यानी ``निर्वात'')?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:5}
\qty{1.013}{bar} और \qty{20}{\degreeCelsius} पर हवा का घनत्व
($M = \qty{29}{g/mol}$), फिर \qty{100}{\degreeCelsius} पर। \qty{2800}{m^3}
के किसी गर्म-वायु गुब्बारे का उत्थापन (हवा के दोनों भारों का अंतर)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:6}
हवा में अणुओं के हिसाब से $21\%$ O$_2$ है। समुद्र-तल पर ऑक्सीजन का आंशिक
\omterm{def:b1:kinetic-theory:pressure}{दाब}; \qty{5000}{m} पर, जहाँ $P = \qty{0.54}{atm}$ है; और एवरेस्ट के शिखर
पर (\qty{0.33}{atm})। वहाँ शरीर को कठिनाई क्यों होती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:7}
\qty{1.0}{L} में \qty{300}{K} पर CO$_2$ का एक \omterm{def:b1:kinetic-theory:scales}{मोल}: पहले आदर्श-गैस नियम से
\omterm{def:b1:kinetic-theory:pressure}{दाब}, फिर वान डेर वाल्स से ($a = \qty{0.364}{Pa.m^6/mol^2}$,
$b = \qty{4.27e-5}{m^3/mol}$)। यहाँ कौन-सा संशोधन प्रबल है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:8}
\qty{1}{bar}, \qty{300}{K} पर \qty{50}{m^3} के किसी कमरे की हवा की \omterm{def:b1:kinetic-theory:U}{आंतरिक
ऊर्जा}; उसे स्थिर आयतन पर \qty{5}{K} गरम करने की ऊर्जा; और \qty{2}{kW} के
किसी तापक से लगा समय (कोई हानि नहीं)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:9}
\qty{-10}{\degreeCelsius} और \qty{40}{\degreeCelsius} के बीच \qty{30}{m}
की कोई इस्पात-पटरी ($\alpha = \qty{1.2e-5}{K^{-1}}$): लंबाई में परिवर्तन।
\qty{49.95}{mm} भीतरी व्यास के किसी पीतल के छल्ले
($\alpha = \qty{1.9e-5}{K^{-1}}$) को \qty{50.00}{mm} की धुरी पर चढ़ाना है:
उसे कितना गरम करना पड़ेगा?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:kinetic-theory:10}
\qty{300}{K} पर प्रति अणु ऊर्जा ($\tfrac32 k_BT$), जूल में और
\omterm{def:b1:charged-particles:eV}{इलेक्ट्रॉन-वोल्ट} में; आर्गन तथा नाइट्रोजन की मोलर ऊष्मा धारिताएँ
$C_{V,m}$; और नाइट्रोजन की धारिता बड़ी क्यों है, जबकि उसके अणु हलके हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:kinetic-theory:11}
छह-दिशा वाले प्रतिरूप में \omterm{thm:b1:kinetic-theory:pressure}{गतिज दाब} की व्युत्पत्ति दुहराइए, फिर दिखाइए कि
वेगों के किसी समदैशिक वितरण के लिए $\langle v_x^2\rangle
= \tfrac13\langle v^2\rangle$ होता है और \omterm{def:b1:kinetic-theory:pressure}{दाब} का सूत्र अपरिवर्तित रहता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:kinetic-theory:12}
\qty{300}{K} पर H$_2$ और O$_2$ की वर्ग-माध्य-मूल चालों की तुलना पृथ्वी के
\omterm{prop:b1:central-forces:escape}{पलायन वेग} से, और चंद्रमा के (\qty{2.4}{km/s}) से कीजिए। यह जानते हुए कि
$3u$ से तेज़ अणुओं का अंश छोटा तो है पर शून्य नहीं, समझाइए कि पृथ्वी ने
अपनी हाइड्रोजन क्यों खो दी और चंद्रमा ने अपना पूरा वायुमंडल।
\end{exercise}

\section{समस्या: कमरे की हवा}

\begin{problem}[{सप्ताहांत समस्या --- एक कक्षा भर हवा: उसके अणु गिनना, उसे
तौलना, उनकी उड़ानों का समय निकालना, और उसी प्रतिरूप से गुब्बारे का
उत्थापन, पहाड़ी हवा का विरलपन तथा ध्वनि की चाल फिर से पा
लेना}]
\label{pb:b1:kinetic-theory:1}
कोई कक्षा $10 \times 8 \times \qty{2.5}{m}$ की है; उसकी हवा
$P = \qty{1.013e5}{Pa}$, $T = \qty{293}{K}$ पर है, मोलर द्रव्यमान $M = \qty{29.0}{g/mol}$,
और अणुओं के हिसाब से $79\%$ N$_2$ तथा $21\%$ O$_2$। $R = \qty{8.314}{J/(mol.K)}$,
$k_B = \qty{1.38e-23}{J/K}$, $N_A = \num{6.02e23}$, $g = \qty{9.81}{m/s^2}$।

\medskip
\noindent\textbf{भाग I --- गिनना और तौलना।}
\begin{enumerate}
  \item कमरे में हवा की मात्रा (मोलों में) और अणुओं की संख्या।
  \item हवा का द्रव्यमान; किसी व्यक्ति से तुलना कीजिए।
  \item संख्या घनत्व $n^*$ और अणुओं के बीच माध्य दूरी
        ($n^{*-1/3}$); \qty{0.37}{nm} के आणविक व्यास से तुलना कीजिए।
  \item N$_2$ और O$_2$ की \omterm{def:b1:kinetic-theory:kinetictemp}{वर्ग-माध्य-मूल चाल}; उसी \omterm{def:b1:kinetic-theory:temperature}{ताप} पर भारी गैस अधिक
        धीमी क्यों चलती है?
  \item किसी एक अणु की माध्य \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा}; कमरे की हवा की कुल स्थानांतरी
        \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा}; और \qty{100}{km/h} पर किसी कार की \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} से तुलना
        कीजिए।
  \item \omterm{rem:b1:kinetic-theory:validity}{माध्य मुक्त पथ} $\lambda \approx 1/(\sqrt2\pi d^2n^*)$ और किसी अणु
        की दो टक्करों के बीच माध्य समय।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- टक्करों से \omterm{def:b1:kinetic-theory:pressure}{दाब}।}
\begin{enumerate}[resume]
  \item $u = \qty{500}{m/s}$ वाले छह-दिशा प्रतिरूप से, दीवार के
        \qty{1}{cm^2} पर प्रति सेकंड टक्करों की संख्या निकालिए।
  \item प्रति सेकंड दिया गया संवेग निकालिए, और जाँचिए कि वह वायुमंडलीय
        \omterm{def:b1:kinetic-theory:pressure}{दाब} को फिर से दे देता है।
  \item \qty{8}{m} गुणा \qty{2.5}{m} की एक दीवार पर हवा का कुल बल; दीवार
        हिलती क्यों नहीं?
  \item खिड़कियाँ बंद और सील करके कमरा \qty{303}{K} तक गरम किया जाता है:
        नया \omterm{def:b1:kinetic-theory:pressure}{दाब}, और अब दीवार पर परिणामी बल।
  \item वही गरमाहट, पर खिड़की खुली रहने पर: क्या बाहर जाता है, और कितना?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- उत्प्लावन।}
\begin{enumerate}[resume]
  \item कमरे की हवा का घनत्व, और उसी \omterm{def:b1:kinetic-theory:pressure}{दाब} पर \qty{373}{K} की हवा का।
  \item \qty{2800}{m^3} आयतन का कोई गर्म-वायु गुब्बारा \qty{373}{K} की
        हवा से भरा जाता है: गरम हवा का भार, हटाई गई ठंडी हवा का भार, और
        उपलब्ध उत्थापन (आर्किमिडीज़,
        \cref{ch:b1:fluid-statics})।
  \item आवरण, टोकरी, ज्वालक और गैस का भार \qty{300}{kg} है: वह
        \qty{75}{kg} के कितने यात्री उठा सकता है?
  \item उत्थापन दुगुना करने के लिए हवा को किस \omterm{def:b1:kinetic-theory:temperature}{ताप} तक गरम करना चाहिए, और
        ऐसा किया क्यों नहीं जाता (आवरण नायलॉन का है)?
  \item उसी आयतन का कोई हीलियम गुब्बारा \qty{293}{K} पर
        ($M = \qty{4}{g/mol}$): उत्थापन? फिर भी पर्यटक गुब्बारे गरम हवा ही
        क्यों काम में लेते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विरल हवा और ध्वनि की चाल।}
\begin{enumerate}[resume]
  \item कमरे में ऑक्सीजन का आंशिक \omterm{def:b1:kinetic-theory:pressure}{दाब}।
  \item एवरेस्ट के शिखर पर $P = \qty{0.33}{atm}$ और $T \approx \qty{250}{K}$ है:
        ऑक्सीजन अणुओं का संख्या घनत्व, कमरे वाले से तुलना करके।
  \item वायुमंडल का \omterm{def:b1:kinetic-theory:pressure}{दाब} ऊँचाई के साथ मोटे तौर पर
        $P(z) = P_0\eu^{-z/H}$ की तरह घटता है, जहाँ $H = RT/Mg$ (\cref{ch:b1:fluid-statics}):
        \qty{260}{K} के लिए $H$ निकालिए, और वह ऊँचाई जिस पर $P = P_0/2$ हो।
  \item किसी \omterm{def:b1:kinetic-theory:model}{आदर्श गैस} में ध्वनि की चाल $c_s = \sqrt{\gamma RT/M}$ होती
        है, जहाँ हवा के लिए $\gamma = 1.4$ (\cref{ch:b1:first-law}): उसे
        \qty{293}{K} पर निकालिए और \omterm{def:b1:kinetic-theory:kinetictemp}{वर्ग-माध्य-मूल चाल} से तुलना कीजिए; इस
        निकटता की व्याख्या कीजिए।
  \item $c_s$ का $T$ के साथ बदलाव कैसा है? \qty{303}{K} पर कमरे में ध्वनि
        की चाल, और \qty{293}{K} पर हीलियम में ($\gamma = 5/3$): हीलियम
        वाली आवाज़ ऊँची क्यों सुनाई देती है?
  \item एवरेस्ट के शिखर पर ध्वनि कमरे की तुलना में तेज़ चलती है या धीमी?
        कितनी?
  \item कमरे की हवा उसी $P$ और $T$ पर आर्गन से बदल दी जाती है: गिनती,
        द्रव्यमान, चालों और \omterm{def:b1:kinetic-theory:U}{आंतरिक ऊर्जा} में क्या बदलता है?
  \item कमरे की हवा की \omterm{def:b1:kinetic-theory:U}{आंतरिक ऊर्जा} ($U = \tfrac52 nRT$) और उसे स्थिर
        आयतन पर \qty{10}{K} गरम करने की ऊर्जा।
  \item तीन पंक्तियों में सार दीजिए कि अणुगति प्रतिरूप ने क्या दिया: किन
        स्थूल राशियों की व्याख्या की, और किस एक सूक्ष्म औसत से।
\end{enumerate}
\end{problem}
