\chapter{प्रकाशिक उपकरण और नेत्र}\label{ch:b1:optical-instruments}

संगीत-सभा की पिछली पंक्ति से दूरबीन की एक जोड़ी गायक का चेहरा बाँह भर की दूरी
पर खींच लाती है; \omterm{def:b1:optical-instruments:microscope}{सूक्ष्मदर्शी} तालाब के पानी की एक बूँद को चिड़ियाघर बना देता
है; और उँगली जितना मोटा फ़ोन आकाशगंगा का छायाचित्र ले लेता है। इनमें से हर
उपकरण कुछ पतले लेंसों की ऐसी सज्जा है जिसका एक ही प्रयोजन है: \omterm{def:b1:optical-instruments:eye}{नेत्र} के सामने,
या किसी संवेदक के सामने, ऐसा प्रतिबिंब रखना जो \omterm{def:b1:optical-instruments:eye}{नेत्र} अकेला बना पाता उससे बड़ा,
अधिक चमकीला या अधिक तीखा हो। इसलिए यह अध्याय स्वयं \omterm{def:b1:optical-instruments:eye}{नेत्र} से शुरू होता है ---
वह क्या कर सकता है और क्या नहीं --- और फिर पिछले अध्याय के संयुग्मन संबंध से
\omterm{def:b1:optical-instruments:G}{आवर्धक}, \omterm{def:b1:optical-instruments:microscope}{सूक्ष्मदर्शी}, दूरदर्शी और \omterm{def:b1:optical-instruments:camera}{कैमरा} खड़े करता है।

\section{नेत्र}

\begin{definition}[सरलीकृत नेत्र]\label{def:b1:optical-instruments:eye}
\index{नेत्र}\index{सरलीकृत नेत्र}\index{दृष्टिपटल}\index{समंजन}
\emph{सरलीकृत नेत्र} कॉर्निया और नेत्र-लेंस को \emph{परिवर्तनशील} \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} के
एक ही पतले \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस} के रूप में लेता है, और \emph{दृष्टिपटल} को उसके पीछे
नियत दूरी $d \approx \qty{17}{mm}$ पर रखे परदे के रूप में। पेशियाँ \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} की
वक्रता बदलकर देखी जा रही वस्तु का प्रतिबिंब दृष्टिपटल पर बनाए रखती हैं: यही
\emph{समंजन} है। \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के आगे बैठी परितारिका \qty{2}{} से
\qty{8}{mm} व्यास का परिवर्ती द्वारक है, यानी \emph{पुतली}।
\end{definition}

\begin{definition}[दूर बिंदु, निकट बिंदु]\label{def:b1:optical-instruments:punctum}
\index{दूर बिंदु}\index{निकट बिंदु}\index{समंजन का आयाम}
\emph{दूर बिंदु} (पंक्टम रेमोटम) वह वस्तु बिंदु है जो बिना \omterm{def:b1:optical-instruments:eye}{समंजन} किए तीखा
दिखता है; और \emph{निकट बिंदु} (पंक्टम प्रॉक्सिमम) वह निकटतम बिंदु जो पूरे
\omterm{def:b1:optical-instruments:eye}{समंजन} के साथ तीखा दिखता है। किसी सामान्य (एमीट्रोपिक) \omterm{def:b1:optical-instruments:eye}{नेत्र} के लिए दूर बिंदु
अनंत पर होता है और निकट बिंदु
$d_m \approx \qty{25}{cm}$ पर, जो \emph{स्पष्ट दृष्टि की परंपरागत दूरी} है।
दोनों के लिए चाहिए अभिसारिताओं का अंतर ही \emph{\omterm{def:b1:optical-instruments:eye}{समंजन} का आयाम} है: किसी युवा
वयस्क के लिए लगभग \qty{4}{\delta}, जो आयु के साथ घटता जाता है।
\end{definition}

\begin{example}[नेत्र की अभिसारिता]\label{ex:b1:optical-instruments:vergence}
दूरस्थ वस्तु: प्रतिबिंब $d = \qty{17}{mm}$ पर होने का अर्थ है $f' = \qty{17}{mm}$,
$V = 1/0.017 = \qty{59}{\delta}$। \qty{25}{cm} पर वस्तु:
$V = 1/0.017 + 1/0.25 = 59 + 4 = \qty{63}{\delta}$। साठ में से चार डाइऑप्टर
का \omterm{def:b1:optical-instruments:eye}{समंजन}: नेत्र-लेंस कॉर्निया के काम में कुछ ही प्रतिशत जोड़ता है, पर पढ़ने के
लिए वही कुछ प्रतिशत सब कुछ हैं।
\end{example}

\begin{proposition}[दोष और उनका संशोधन]\label{prop:b1:optical-instruments:defects}
\index{निकट-दृष्टि दोष}\index{दूर-दृष्टि दोष}\index{जरा-दृष्टि दोष}
\emph{निकट-दृष्टि} वाला \omterm{def:b1:optical-instruments:eye}{नेत्र} आवश्यकता से अधिक अभिसारी है: उसका \omterm{def:b1:optical-instruments:punctum}{दूर बिंदु}
परिमित दूरी $D_r$ पर होता है; \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} $f' = -D_r$ का \omterm{def:b1:mirrors-thin-lenses:lens}{अपसारी लेंस} (\omterm{def:b1:optical-instruments:eye}{नेत्र} के
पास पहना हुआ) अनंत का प्रतिबिंब \omterm{def:b1:optical-instruments:punctum}{दूर बिंदु} पर बना देता है और दूर की दृष्टि लौटा
देता है। \emph{दूर-दृष्टि} वाला \omterm{def:b1:optical-instruments:eye}{नेत्र} पर्याप्त अभिसारी नहीं है: उसका \omterm{def:b1:optical-instruments:punctum}{दूर बिंदु}
आभासी है (\omterm{def:b1:optical-instruments:eye}{नेत्र} के पीछे) और उसका \omterm{def:b1:optical-instruments:punctum}{निकट बिंदु} बहुत दूर; उसे \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस}
संशोधित करता है। \emph{\omterm{prop:b1:optical-instruments:defects}{जरा-दृष्टि दोष}} आयु के साथ \omterm{def:b1:optical-instruments:eye}{समंजन} का लोप है: \omterm{def:b1:optical-instruments:punctum}{निकट बिंदु}
पीछे हटता जाता है और पढ़ने के लिए \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस} चाहिए होते हैं, \omterm{def:b1:optical-instruments:punctum}{दूर बिंदु} चाहे
कहीं भी हो।
\end{proposition}

\begin{proof}
\omterm{def:b1:optical-instruments:eye}{नेत्र} के पास रखा \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} और \omterm{def:b1:optical-instruments:eye}{नेत्र} मिलकर ऐसा निकाय बनाते हैं जिसकी वस्तु वही होनी
चाहिए जो \omterm{def:b1:optical-instruments:eye}{नेत्र} देख सके: अतः \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} को अनंत पर रखी वस्तु का \omterm{def:b1:mirrors-thin-lenses:image}{आभासी प्रतिबिंब} \omterm{def:b1:optical-instruments:punctum}{दूर
बिंदु} पर बनाना होगा, यानी $\overline{OA'} = -D_r$, जब
$\overline{OA} = -\infty$ हो; अतः $f' = \overline{OA'} = -D_r$। शेष
स्थितियाँ वही तर्क हैं, बस संबंधित बिंदुओं के साथ।
\end{proof}

\begin{definition}[कोणीय आकार और विभेदन]\label{def:b1:optical-instruments:angular}
\index{कोणीय आकार}\index{कोणीय विभेदन}
ऊँचाई $h$ की किसी वस्तु का, जो दूरी $D$ पर रखी हो, \emph{कोणीय आकार}
$\theta \approx h/D$ होता है (छोटे कोण)। \omterm{def:b1:optical-instruments:eye}{नेत्र} दो बिंदुओं में भेद तभी कर पाता
है जब उनका कोणीय अंतराल उसके \emph{कोणीय विभेदन} से बड़ा हो, जो लगभग
$\epsilon \approx \qty{3e-4}{rad}$ (एक कलामिनट) है: शंकु कोशिकाओं का अंतराल
और पुतली पर विवर्तन, दोनों यही सीमा बाँधते हैं।
\end{definition}

\begin{remark}[उपकरण किसलिए है]\label{rem:b1:optical-instruments:purpose}
\omterm{def:b1:optical-instruments:punctum}{निकट बिंदु} तक लाया गया \qty{1}{mm} का ब्योरा $1/250 =
\qty{4e-3}{rad}$ का कोण घेरता है: यानी विभेदन सीमा का तेरह गुना; जबकि
\qty{0.05}{mm} का ब्योरा $\qty{2e-4}{rad}$ घेरता है और अदृश्य रहता है। पास की
वस्तुओं को देखने वाले उपकरण (\omterm{def:b1:optical-instruments:G}{आवर्धक}, \omterm{def:b1:optical-instruments:microscope}{सूक्ष्मदर्शी}) \omterm{def:b1:optical-instruments:angular}{कोणीय आकार} को उससे भी आगे
बढ़ा देते हैं जितना वस्तु को पास लाकर बढ़ाया जा सकता है; दूर की वस्तुओं के
उपकरण (दूरदर्शी) उन चीज़ों का \omterm{def:b1:optical-instruments:angular}{कोणीय आकार} बढ़ाते हैं जिनके पास जाया नहीं जा सकता;
और \omterm{def:b1:optical-instruments:camera}{कैमरा} \omterm{def:b1:optical-instruments:eye}{दृष्टिपटल} की जगह ऐसा संवेदक रख देता है जो कई सेकंड तक प्रकाश जमा कर
सकता है और जिसे बाद में बड़ा किया जा सकता है।
\end{remark}

\section{आवर्धक}

\begin{definition}[कोणीय आवर्धन]\label{def:b1:optical-instruments:G}
\index{कोणीय आवर्धन}\index{आवर्धक}\index{व्यावसायिक आवर्धन}
\omterm{def:b1:optical-instruments:eye}{नेत्र} से काम में लिए जाते किसी उपकरण के लिए \emph{कोणीय आवर्धन}
$G = \theta'/\theta$ है, जहाँ $\theta'$ उपकरण से देखे गए प्रतिबिंब का \omterm{def:b1:optical-instruments:angular}{कोणीय
आकार} है और $\theta$ नंगी आँख से देखी गई वस्तु का \omterm{def:b1:optical-instruments:angular}{कोणीय आकार} --- पास की वस्तु
के लिए \omterm{def:b1:optical-instruments:punctum}{निकट बिंदु} $d_m = \qty{25}{cm}$ पर, और दूर की वस्तु के लिए जहाँ वह है
वहीं।
\end{definition}

\begin{proposition}[आवर्धक]\label{prop:b1:optical-instruments:loupe}
\omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} $f' < d_m$ का \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस}, जिसके वस्तु फोकस तल में वस्तु रखी हो,
अनंत पर प्रतिबिंब देता है, जो बिना \omterm{def:b1:optical-instruments:eye}{समंजन} के दिखता है, और तब
\[
G = \frac{d_m}{f'} = d_m V .
\]
\end{proposition}

\begin{proof}
फोकस तल में वस्तु $AB$, ऊँचाई $h$ की: $B$ से चली किरणें समांतर निकलती हैं, उस
किरण के कोण पर जो $O$ से होकर जाती है, यानी $\theta' = h/f'$। नंगी आँख से,
सर्वोत्तम स्थिति में: $\theta = h/d_m$। अनुपात $d_m/f'$।
\end{proof}

\begin{example}[घड़ीसाज़ का आवर्धक]\label{ex:b1:optical-instruments:loupe}
$f' = \qty{25}{mm}$: $G = 250/25 = 10$, जो ``$10\times$'' के नाम से बिकता
है। तब \qty{0.05}{mm} का ब्योरा $\qty{2e-3}{rad}$ घेरता है, यानी आराम से
विभेदित। $G$ को और ऊपर ले जाने का अर्थ है कुछ ही मिलीमीटर का $f'$, यानी \omterm{def:b1:optical-instruments:eye}{नेत्र}
से सटाकर पकड़ा नन्हा \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} --- यही अकेले \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} की सीमा है, और यही कारण कि
\omterm{def:b1:optical-instruments:microscope}{सूक्ष्मदर्शी} का अस्तित्व है।
\end{example}

\section{सूक्ष्मदर्शी}

\begin{definition}[संयुक्त सूक्ष्मदर्शी]\label{def:b1:optical-instruments:microscope}
\index{सूक्ष्मदर्शी}\index{अभिदृश्यक}\index{नेत्रिका}\index{प्रकाशिक अंतराल}
\emph{सूक्ष्मदर्शी} में एक \emph{अभिदृश्यक} $L_1$ होता है, जिसकी \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी}
$f_1'$ छोटी है और जो पास रखी किसी वस्तु का वास्तविक, बड़ा मध्यवर्ती प्रतिबिंब किसी
\emph{नेत्रिका} $L_2$ (\omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी}
$f_2'$) के वस्तु फोकस तल में बनाता है, और नेत्रिका \omterm{def:b1:optical-instruments:G}{आवर्धक} की तरह काम करती है।
अभिदृश्यक के प्रतिबिंब फोकस और नेत्रिका के वस्तु फोकस के बीच की दूरी $\Delta =
\overline{F_1'F_2}$ \emph{प्रकाशिक अंतराल} कहलाती है, जो पारंपरिक उपकरणों पर
मानकीकृत \qty{160}{mm} होती है।
\end{definition}

\begin{proposition}[सूक्ष्मदर्शी का आवर्धन]\label{prop:b1:optical-instruments:microscope}
अंतिम प्रतिबिंब अनंत पर होने पर,
\[
\gamma_1 = -\frac{\Delta}{f_1'}, \qquad G_2 = \frac{d_m}{f_2'},
\qquad G = \abs{\gamma_1}\,G_2 = \frac{\Delta\,d_m}{f_1' f_2'} .
\]
\end{proposition}

\begin{proof}
मध्यवर्ती प्रतिबिंब $A_1B_1$ \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} के वस्तु फोकस तल में पड़ता है, यानी
$\overline{F_1'A_1} = \Delta$ पर; न्यूटन का संबंध देता है
$\gamma_1 = -\overline{F_1'A_1}/f_1' = -\Delta/f_1'$। फिर \omterm{def:b1:optical-instruments:microscope}{नेत्रिका}
$A_1B_1$ को (ऊँचाई $\abs{\gamma_1} h$) अनंत पर कोण
$\theta' = \abs{\gamma_1} h/f_2'$ के नीचे दिखाती है, जबकि नंगी आँख के लिए वह
कोण $\theta = h/d_m$ होता।
\end{proof}

\begin{example}[विद्यार्थी का सूक्ष्मदर्शी]\label{ex:b1:optical-instruments:micronum}
\omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} $f_1' = \qty{4.0}{mm}$, \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} $f_2' = \qty{25}{mm}$,

$\overline{F_1A} = -f_1'^2/\Delta = -16/160 =
\qty{-0.10}{mm}$ पर बैठती है, $F_1$ से नापी हुई: यानी \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} से \qty{4.1}{mm}। फोकसन का अर्थ
है पूरी नली को मिलीमीटर के अंशों में खिसकाना।
\end{example}

\begin{remark}[विभेदन की सीमा]\label{rem:b1:optical-instruments:abbe}
\index{विभेदन क्षमता}
बिना सीमा का आवर्धन बेकार होता: प्रकाश एक तरंग है, और अर्ध-कोण $u$ के भीतर की
किरणें स्वीकारता कोई \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} (\omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक} $n$ के माध्यम में) दो ऐसे
बिंदुओं को अलग नहीं कर सकता जो लगभग
$\lambda/(2n\sin u)$ से पास हों --- वर्ष 2 के खंड का तरंग-विवेचन इसे
व्युत्पन्न करता है। $n\sin u \approx 1.4$ (तेल-निमज्जन) और $\lambda =
\qty{0.5}{\micro m}$ के साथ यह सीमा \qty{0.2}{\micro m} के पास बैठती है। उस
ब्योरे को \omterm{def:b1:optical-instruments:eye}{नेत्र} के लिए दृश्य बनाने के वास्ते (\qty{25}{cm} पर
$\epsilon = \qty{3e-4}{rad}$, यानी \qty{75}{\micro m}) लगभग $400$ का आवर्धन
काफ़ी है; $1000$ के परे प्रतिबिंब केवल बड़ा होता है, समृद्ध नहीं: यही
\emph{रिक्त आवर्धन} है।
\end{remark}

\section{खगोलीय दूरदर्शी}

\begin{proposition}[अपवर्ती दूरदर्शी]\label{prop:b1:optical-instruments:telescope}
\index{दूरदर्शी}\index{निर्गम पुतली}
कोई \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} $L_1$ और कोई \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} $L_2$, जिनके लिए $F_1' = F_2$ हो, मिलकर
\omterm{prop:b1:mirrors-thin-lenses:doublet}{अफोकल निकाय} बनाते हैं: तारा (अनंत पर) अनंत पर ही दिखता है, शिथिल \omterm{def:b1:optical-instruments:eye}{नेत्र} उसे देखता है, और
\[
G = -\frac{f_1'}{f_2'} .
\]
\omterm{def:b1:optical-instruments:microscope}{नेत्रिका} से होकर बना \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} की परिधि का प्रतिबिंब \emph{\omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली}}
कहलाता है: यह $D/\abs G$ व्यास की चमकीली चकती है, जो \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} से
$\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ पीछे पड़ती है, और वहीं \omterm{def:b1:optical-instruments:eye}{नेत्र} की
पुतली रखनी चाहिए ताकि सारा प्रकाश मिले।
\end{proposition}

\begin{proof}
$G$ \cref{prop:b1:mirrors-thin-lenses:doublet} में व्युत्पन्न हो चुका है।
\omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक}, $L_2$ से देखने पर, $\overline{O_2O_1} = -(f_1'
+ f_2')$ पर रखी वस्तु है; देकार्त देता है $1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2')
= f_1'/[f_2'(f_1' + f_2')]$, और आवर्धन $\overline{O_2P'}/
\overline{O_2O_1} = -f_2'/f_1' = 1/G$ व्यास $D$ को
$D/\abs G$ तक समेट देता है। \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} में घुसती हर किरण \omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली} से होकर
निकलती है, क्योंकि \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} का हर बिंदु वहीं प्रतिबिंबित होता है।
\end{proof}

\begin{proposition}[दूरदर्शी से क्या मिलता है]\label{prop:b1:optical-instruments:gains}
\index{प्रकाश-संग्रह क्षमता}\index{रैले कसौटी}
नंगी आँख (पुतली $d_p$) की तुलना में:
\begin{itemize}
  \item \emph{\omterm{def:b1:optical-instruments:angular}{कोणीय आकार}}: $\abs G$ गुना;
  \item किसी बिंदु स्रोत से \emph{बटोरा गया प्रकाश}:
        $(D/d_p)^2$ गुना, बशर्ते \omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली} \omterm{def:b1:optical-instruments:eye}{नेत्र} की पुतली से बड़ी न हो,
        यानी $\abs G \geq D/d_p$;
  \item \emph{\omterm{def:b1:optical-instruments:angular}{कोणीय विभेदन}}: \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} पर विवर्तन इसे
        $\theta_{\min} \approx 1.22\,\lambda/D$ तक बाँध देता है (\omterm{prop:b1:optical-instruments:gains}{रैले
        कसौटी}), जो \omterm{def:b1:optical-instruments:eye}{नेत्र} के $\epsilon$ से कहीं नीचे है; इस सीमा को दृश्य
        बनाने वाला आवर्धन $G_r = \epsilon/\theta_{\min}$ है,
        और इससे बड़े आवर्धन केवल धुँधलेपन को बड़ा करते हैं।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[शौक़ीन का 150-मिलीमीटर अपवर्ती दूरदर्शी]\label{ex:b1:optical-instruments:telnum}
$D = \qty{150}{mm}$, $f_1' = \qty{1200}{mm}$, \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} \qty{25}{mm}:
$G = -48$; \omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली} \qty{3.1}{mm}; \qty{7}{mm} की अँधेरे में खुली पुतली की
तुलना में प्रकाश-लाभ $(150/7)^2 \approx
460$; विभेदन $1.22 \times
\qty{550}{nm}/\qty{0.15}{m} = \qty{4.5e-6}{rad} = 0.9''$, अतः चंद्रमा पर
\qty{1.7}{km} चौड़े क्रेटर; $G_r = 3\times10^{-4}/4.5\times
10^{-6} \approx 67$ --- अतः \qty{6}{mm} की \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} ($G = 200$) कोई नया
ब्योरा नहीं दिखाती, केवल एक मंद, बड़ी चकती। सप्ताहांत समस्या इसी उपकरण की
अभिकल्पना करती है।
\end{example}

\begin{remark}[परावर्ती दूरदर्शी, दूरबीन, नाट्य-दूरबीन]\label{rem:b1:optical-instruments:variants}
\emph{परावर्ती} दूरदर्शी में \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} की जगह \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} $f_1'$ का \omterm{def:b1:mirrors-thin-lenses:mirror}{अवतल
दर्पण} ले लेता है (कोई वर्णिक विक्षेपण नहीं, और दर्पण मीटरों चौड़ा बनाया तथा
पीछे से टिकाया जा सकता है): ऊपर की हर बात लागू रहती है, बस $f_1'$ दर्पण की
\omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} होती है। प्रतिबिंब उलटा होता है --- तारों के लिए यह हानिरहित है;
पार्थिव उपयोग के लिए दूरबीनें उसे सीधा करने को पूर्ण-परावर्तन प्रिज़्मों की एक
जोड़ी बिठा देती हैं, और गैलीलियो की नाट्य-दूरबीन अपसारी \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} काम में लेती
है ($f_2' < 0$, $G > 0$), जो छोटी तो है पर जिसका क्षेत्र सँकरा
होता है।
\end{remark}

\section{कैमरा}

\begin{definition}[कैमरा; f-संख्या]\label{def:b1:optical-instruments:camera}
\index{कैमरा}\index{f-संख्या}\index{क्षेत्र-गहराई}
\emph{कैमरा} \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} $f'$ का \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस} है जो किसी संवेदक पर \omterm{def:b1:mirrors-thin-lenses:image}{वास्तविक
प्रतिबिंब} बनाता है, दूर के विषयों के लिए $\overline{OA'} \approx f'$ पर;
फोकसन में \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} खिसकता है। व्यास $D$ का डायाफ़्राम पुंज को सीमित करता है;
और \emph{f-संख्या} $N = f'/D$ है (जिसे $f/N$ लिखा जाता है)। \emph{दृष्टि कोण}
$2\arctan(\ell/2f')$ होता है, चौड़ाई $\ell$ के किसी संवेदक के लिए।
\end{definition}

\begin{proposition}[उद्भासन और क्षेत्र-गहराई]\label{prop:b1:optical-instruments:dof}
\index{अतिफोकसी दूरी}
संवेदक पर विकिरण-तीव्रता $1/N^2$ की तरह बदलती है, अतः किसी दी गई प्रतिबिंब-चमक
के लिए उद्भासन काल $N^2$ की तरह बदलता है। फोकस की गई दूरी से भिन्न दूरी पर
पड़ा कोई बिंदु धुँधलेपन की चकती बनाता है; उसे किसी सह्य व्यास $c$ से नीचे रखने
की माँग करने पर, \emph{\omterm{prop:b1:optical-instruments:dof}{अतिफोकसी दूरी}}
पर फोकस किया \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस}
\[
H = \frac{f'^2}{N c}
\]
$H/2$ से लेकर अनंत तक सब कुछ स्वीकार्य रूप से तीखा दिखाता है।
\end{proposition}

\begin{proof}
विकिरण-तीव्रता $=$ शक्ति/क्षेत्रफल $\propto D^2/f'^2 = 1/N^2$ (किसी दिए गए
विषय के प्रतिबिंब का आकार $f'$ से बँधा है, और बटोरी गई शक्ति
$D^2$ की तरह बढ़ती है)। दूरी $p$ पर फोकस, प्रतिबिंब $q \approx f'$ पर;
अनंत का बिंदु $f'$ पर फोकस होता है, यानी संवेदक से $q - f'$ पहले;
उसका किरण-शंकु, जिसका आधार \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} पर $D$ है, संवेदक पर व्यास $D(q - f')/q$
बनाता है (समरूप त्रिभुज)। उपाक्षीय रूप में $1/q = 1/f' - 1/p$ के साथ,
$q - f' \approx f'^2/p$ हो जाता है जब $p \gg f'$ हो, अतः धुँधलापन
$Df'/p = f'^2/(Np)$ है; और वह $c$ के बराबर होता है जब $p = H$ हो। पास की ओर
वही ज्यामिति सीमा $H/2$ देती है।
\end{proof}

\section{समस्या: चंद्रमा के लिए दूरदर्शी की अभिकल्पना}

\begin{problem}[{सप्ताहांत समस्या --- बग़ीचे के लॉन पर 150 मिलीमीटर का अपवर्ती
दूरदर्शी: नेत्रिकाएँ चुनिए, नेत्र को उसकी जगह रखिए, प्रकाश गिनिए, और पता
लगाइए कि वह कितना छोटा क्रेटर दिखा सकता है}]
\label{pb:b1:optical-instruments:1}
\omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} एक \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस} $L_1$ है, जिसकी \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} $f_1' = \qty{1200}{mm}$
और व्यास $D = \qty{150}{mm}$ है; \qty{25}{mm} और \qty{6}{mm} फोकस दूरियों की
नेत्रिकाएँ $L_2$ उपलब्ध हैं। चंद्रमा
\qty{3.84e5}{km} दूर है और $0.52^\circ$ का कोण घेरता है; \omterm{def:b1:optical-instruments:eye}{नेत्र} की
रात्रि-पुतली \qty{7}{mm} है, उसका विभेदन $\epsilon = \qty{3.0e-4}{rad}$;
$\lambda = \qty{550}{nm}$।

\medskip
\noindent\textbf{भाग I --- अफोकल सज्जा।}
\begin{enumerate}
  \item किसी तारे को बिना \omterm{def:b1:optical-instruments:eye}{समंजन} के देखने के लिए \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} कहाँ रखनी होगी?
        हर \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} के लिए अभिदृश्यक--नेत्रिका दूरी दीजिए।
  \item हर \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} पर बारी-बारी देकार्त का संबंध लगाकर जाँचिए कि अक्ष पर अनंत
        का बिंदु अनंत पर ही प्रतिबिंबित होता है।
  \item दिखाइए कि अक्ष से कोण $\alpha$ पर किसी तारे से आता पुंज
        कोण $\alpha' = -(f_1'/f_2')\alpha$ पर समांतर पुंज बनकर निकलता है।
  \item हर \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} के लिए $G$ और उससे दिखते चंद्रमा का आभासी व्यास
        निकालिए।
  \item प्रतिबिंब उलटा है। यहाँ यह स्वीकार्य क्यों है, और कोई पार्थिव
        प्रेक्षण-दूरदर्शी इसमें क्या जोड़ता?
  \item चंद्रमा का वास्तविक मध्यवर्ती प्रतिबिंब कहाँ है, और उसका व्यास
        कितना है? (वहीं उसका छायाचित्र लिया जा सकता है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:b1:optical-instruments:eye}{नेत्र} कहाँ जाता है।}
\begin{enumerate}[resume]
  \item दिखाइए कि \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} की परिधि का \omterm{def:b1:mirrors-thin-lenses:image}{वास्तविक प्रतिबिंब} अपने
        पीछे $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ पर बनाती है, जिसका
        व्यास $D/\abs G$ है (\omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली})।
  \item हर \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} के लिए \omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली} की स्थिति और व्यास निकालिए।
  \item \omterm{def:b1:optical-instruments:eye}{नेत्र} की पुतली \omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली} पर ही क्यों रखनी चाहिए? यदि उसे और
        पीछे रखा जाए तो क्षेत्र का क्या होता है?
  \item किसी तारे का प्रकाश पूरी तरह काम में आने के लिए \omterm{prop:b1:optical-instruments:telescope}{निर्गम पुतली} \omterm{def:b1:optical-instruments:eye}{नेत्र}
        की पुतली से बड़ी नहीं होनी चाहिए: इससे रात में इस दूरदर्शी का
        न्यूनतम उपयोगी आवर्धन निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विभेदन।}
\begin{enumerate}[resume]
  \item विवर्तन सीमा $\theta_{\min} = 1.22\lambda/D$ रेडियन में और
        कलासेकंडों में निकालिए।
  \item यह कोण चंद्रमा पर सबसे छोटे किस क्रेटर से मेल खाता है? और नंगी
        आँख के लिए?
  \item विभेदक आवर्धन $G_r =
        \epsilon/\theta_{\min}$ को परिभाषित कीजिए और निकालिए। कौन-सी
        \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} उसके सबसे पास है?
  \item \qty{6}{mm} की \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} के साथ प्रतिबिंब \qty{25}{mm} वाली की
        तुलना में अधिक धुँधला और मंद दिखता है। दोनों प्रभाव समझाइए।
  \item वायुमंडलीय विक्षोभ तारों के प्रतिबिंबों को लगभग $1''$ तक फैला देता
        है। किन अभिदृश्यक-व्यासों के लिए ज़मीन से रैले सीमा मायने रखनी बंद
        कर देती है, और बड़ा $D$ फिर भी क्या देता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- प्रकाश।}
\begin{enumerate}[resume]
  \item \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} रात्रि-पुतली से कितने गुना अधिक प्रकाश बटोरता है?
  \item इसे कांतिमानों में बदलिए (अनुपात का $\Delta m = 2.5\log_{10}$);
        यदि नंगी आँख कांतिमान $6$ तक पहुँचती है, तो दूरदर्शी कौन-से
        कांतिमान तक पहुँचता है?
  \item पूर्णिमा का चंद्रमा ज़मीन पर लगभग
        \qty{2e-3}{W/m^2} की विकिरण-तीव्रता पहुँचाता है। \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} में
        कितनी शक्ति घुसती है, और नंगी आँख की तुलना में वह \omterm{def:b1:optical-instruments:eye}{दृष्टिपटल} पर
        कैसे फैलती है (प्रति एकांक घन कोण समान शक्ति वाला वही तर्क: दोनों
        नेत्रिकाओं से पृष्ठ-चमक की तुलना कीजिए)?
  \item दूरदर्शी चंद्रमा के पृष्ठ को प्रति एकांक क्षेत्रफल नंगी आँख से
        अधिक चमकीला क्यों नहीं बनाता, जबकि तारों को वह अधिक चमकीला बना
        देता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- दृष्टि क्षेत्र।}
\omterm{def:b1:optical-instruments:microscope}{नेत्रिका} अपने फोकस तल में \qty{20}{mm} व्यास (\qty{25}{mm} वाली \omterm{def:b1:optical-instruments:microscope}{नेत्रिका}) या
\qty{5}{mm} व्यास (\qty{6}{mm} वाली \omterm{def:b1:optical-instruments:microscope}{नेत्रिका}) का एक वृत्ताकार क्षेत्र-रोधक
रखती है।
\begin{enumerate}[resume]
  \item दिखाइए कि वास्तविक दृष्टि क्षेत्र $2\arctan(\phi/2f_1')
        \approx \phi/f_1'$ होता है, व्यास $\phi$ के किसी रोधक के लिए।
  \item दोनों नेत्रिकाओं के लिए उसे निकालिए और बताइए कि पूरा चंद्रमा
        दृष्टि में समाता है या नहीं।
  \item दोनों के लिए आभासी क्षेत्र निकालिए (वह कोण जिसके नीचे \omterm{def:b1:optical-instruments:eye}{नेत्र}
        \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} से होकर रोधक को देखता है)।
  \item पृथ्वी प्रति सेकंड $15''$ घूमती है: बिना अनुसरण के चंद्रमा को हर
        \omterm{def:b1:optical-instruments:microscope}{नेत्रिका} का क्षेत्र पार करने में कितना समय लगता है?
  \item अभिकल्पना का सार लिखिए: पूरे चंद्रमा के लिए कौन-सी \omterm{def:b1:optical-instruments:microscope}{नेत्रिका}, क्रेटरों
        के लिए कौन-सी, और यह उपकरण कितने छोटे क्रेटर तक दिखा सकता है ---
        यही उसकी इकलौती संख्या है।
  \item वही \omterm{def:b1:optical-instruments:microscope}{अभिदृश्यक} हटाकर उसी \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} का \qty{150}{mm} \omterm{def:b1:mirrors-thin-lenses:mirror}{अवतल दर्पण} रख
        दिया जाता है। ऊपर के कौन-से परिणाम बदल जाते हैं?
\end{enumerate}
\end{problem}
