\chapter{विद्युत विभव; चालक और संधारित्र}\label{ch:b1:potential-capacitors}

कोई क्षेत्र हर बिंदु पर तीन संख्याएँ है; कोई विभव एक। स्थिरवैद्युत बल
संरक्षी है, इसलिए आवेशों के किसी भी वितरण का पूरा क्षेत्र एक ही अदिश फलन
में समेटा जा सकता है, यानी विभव में, जिससे वह अवकलन द्वारा फिर पा लिया
जाता है --- वही मितव्ययिता जिसने यांत्रिकी में बलों को \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} बना
दिया था। विभव ही वह है जो कोई वोल्टमापी पढ़ता है, जो कोई बैटरी बनाए रखती
है, और जो किसी नलिका में इलेक्ट्रॉनों को त्वरित करता है; और उसके समतल-पृष्ठ
किसी क्षेत्र का नक़्शा एक ही नज़र में खींच देते हैं। यह अध्याय उसे गढ़ता है,
उसे द्विध्रुव पर लगाता है (यानी किसी अणु का पहला विद्युत प्रतिरूप), और फिर
उन दोनों वस्तुओं की ओर मुड़ता है जिनसे हर परिपथ बना है: चालक, जिन पर आवेश
अपने आप ऐसे बैठ जाता है कि भीतर का क्षेत्र मर जाए, और \omterm{def:b1:potential-capacitors:capacitance}{संधारित्र}, जो आवेश
तथा ऊर्जा किसी क्षेत्र में संचित रखते हैं।

\section{स्थिरवैद्युत विभव}

\begin{theorem}[स्थिरवैद्युत क्षेत्र संरक्षी है]\label{thm:b1:potential-capacitors:potential}
\index{स्थिरवैद्युत विभव}\index{वोल्ट}\index{परिचालन}
कोई अदिश फलन $V(M)$ होता है, यानी \emph{\omterm{thm:b1:potential-capacitors:potential}{स्थिरवैद्युत
विभव}} (वोल्ट, \unit{V}), जिससे
\[
\vect E = -\overrightarrow{\operatorname{grad}}\,V , \qquad
V_A - V_B = \int_A^B \vect E\cdot\dd\vect l
\]
$A$ से $B$ तक के किसी भी पथ के अनुदिश ($\vect E$ का \emph{परिचालन},
$A$ से $B$ तक, पथ पर निर्भर नहीं करता, और किसी बंद पाश पर शून्य होता है)।
किसी बिंदु आवेश $q$ के लिए, जो $O$ पर हो, $V(M) = \dfrac{q}{4\pi\varepsilon_0\,OM}$
है, जहाँ रूढ़ि $V(\infty) = 0$ ली गई है; और कई आवेशों के विभव जुड़ जाते
हैं। किसी आवेश $q'$ की, जो $M$ पर हो, \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} $E_p = q'V(M)$ है, और
$A$ से $B$ तक विद्युत बल का कार्य $q'(V_A - V_B)$।
\end{theorem}

\begin{proof}
किसी स्थिर आवेश $q$ का $q'$ पर कूलॉम बल न्यूटन के \omterm{def:b1:central-forces:gravitation}{गुरुत्वाकर्षण} जैसा ही रूप
रखता है, जहाँ $-Gm_1m_2$ की जगह $qq'/4\pi\varepsilon_0$ है; और वही
गणना, जिसने $E_p = -Gm_1m_2/r$ दिया था
(\cref{prop:b1:work-and-energy:eps}), $E_p = qq'/4\pi\varepsilon_0r$ देती है:
$q'$ से भाग देने पर $V$ मिलता है, और $\vect E = \vect F/q' =
-\overrightarrow{\operatorname{grad}}(E_p)/q'$। अध्यारोपण इस परिणाम को किसी
भी वितरण तक ले जाता है; और परिचालन वाला सूत्र प्रवणता की परिभाषा को ही पथ
के अनुदिश समाकलित करना है।
\end{proof}

\begin{proposition}[किसी विभव को पढ़ना]\label{prop:b1:potential-capacitors:reading}
\index{समविभव पृष्ठ}
\begin{enumerate}
  \item \emph{\omterm{prop:b1:potential-capacitors:reading}{समविभव पृष्ठ}} $V = \text{अचर}$ हर जगह क्षेत्र-रेखाओं पर लंब
        होते हैं, और क्षेत्र \emph{घटते} विभव की ओर संकेत करता है।
  \item $|\vect E|$ सबसे खड़ी दिशा के अनुदिश $V$ के बदलने की दर है: जहाँ
        समविभव पास-पास हों, वहाँ क्षेत्र प्रबल है।
  \item $V$ किसी योज्य अचर तक ही परिभाषित है; केवल अंतर (यानी वोल्टता)
        मापे जा सकते हैं। भूमि या अनंत को रूढ़ि से शून्य रख लिया जाता
        है।
  \item $O$ के परितः सममिति वाले गोलीय निर्देशांकों में $E_r = -\dd V/
        \dd r$; और किसी तल में ध्रुवीय निर्देशांकों $(r, \theta)$ में,
        $E_r = -\partial V/\partial r$ तथा $E_\theta = -\dfrac1r\,\partial V/\partial\theta$।
\end{enumerate}
\end{proposition}

\begin{proof}
किसी समविभव पर, पृष्ठ में के हर विस्थापन के लिए $\dd V = -\vect E\cdot\dd\vect l = 0$,
अतः $\vect E$ उस पर अभिलंब है; और $\vect E$ के अनुदिश चलने पर
$\dd V = -E\,\dd l < 0$। ध्रुवीय घटक
$\dd V = (\partial V/\partial r)\dd r + (\partial V/\partial\theta)\dd\theta$ और
$\dd\vect l = \dd r\,\vect e_r + r\,\dd\theta\,\vect e_\theta$ से निकलते हैं।
\end{proof}

\begin{method}[किसी विभव की गणना]\label{meth:b1:potential-capacitors:compute}
या तो वितरण पर $\dd q/4\pi\varepsilon_0r$ का योग लीजिए (जब वह योग सँभल जाए और
वितरण परिमित हो, ताकि $V(\infty) = 0$ का कोई अर्थ बने), या फिर
$\dd V = -\vect E\cdot\dd\vect l$ को किसी सुविधाजनक पथ के अनुदिश, ऐसे किसी
बिंदु से चलकर जहाँ $V$ ज्ञात हो, समाकलित कीजिए, और इसके लिए गाउस के नियम से
मिला क्षेत्र काम में लीजिए।
अनंत वितरणों (तार, तल) के लिए केवल दूसरा रास्ता चलता है, और $V$ का शून्य
किसी परिमित बिंदु पर चुनना पड़ता है।
\end{method}

\begin{proposition}[चिरपरिचित वितरणों के विभव]\label{prop:b1:potential-capacitors:classic}
\begin{enumerate}
  \item $R$ त्रिज्या का गोला, जिसके आयतन में $Q$ हो: बाहर $V(r) = \dfrac{Q}{4\pi\varepsilon_0r}$;
        और भीतर $V(r) = \dfrac{Q}{4\pi\varepsilon_0}\,\dfrac{3R^2 - r^2}{2R^3}$,
        अतः $V(0) = \tfrac32\,Q/4\pi\varepsilon_0R$। यदि $Q$ पृष्ठ पर हो, तो भीतर
        $V$ एकसमान रहता है, और $Q/4\pi\varepsilon_0R$ के बराबर।
  \item अनंत तार $\lambda$: $V(r) = -\dfrac{\lambda}{2\pi\varepsilon_0}\ln\dfrac{r}{r_0}$।
  \item \omterm{prop:b1:potential-capacitors:three}{समतल संधारित्र}, जिसमें क्षेत्र $E$ पट्टिका~1 से पट्टिका~2 तक है और
        दोनों $d$ दूरी पर हैं: दरार के पार $V$ रैखिक रूप से घटता है और $V_1 - V_2 = Ed$।
\end{enumerate}
\end{proposition}

\begin{proof}
\cref{prop:b1:electrostatics-gauss:three} के क्षेत्रों के साथ
$E_r = -\dd V/\dd r$ समाकलित कीजिए: गोले के लिए अनंत से भीतर की ओर ($R$ पर
संततता भीतर वाला अचर तय कर देती है), तार के लिए $r_0$ से, और \omterm{def:b1:potential-capacitors:capacitance}{संधारित्र} के
लिए दरार के पार।
\end{proof}

\section{विद्युत द्विध्रुव}

\begin{definition}[विद्युत द्विध्रुव]\label{def:b1:potential-capacitors:dipole}
\index{विद्युत द्विध्रुव}\index{द्विध्रुव आघूर्ण}
दो विपरीत आवेश, $-q$ जो $N$ पर हो और $+q$ जो $P$ पर, $a = NP$ की दूरी पर,
मिलकर \emph{द्विध्रुव आघूर्ण}
$\vect p = q\,\vect{NP}$ (\unit{C.m}) का एक \emph{विद्युत द्विध्रुव} बनाते
हैं, जो $-$ से $+$ की ओर संकेत करता है। द्विध्रुव सन्निकटन उसके क्षेत्र का
अध्ययन दूरियों $r \gg a$ पर करना है। जिस अणु के धनात्मक और ऋणात्मक आवेश के
केंद्र संपाती न हों (पानी, $p = \qty{6.2e-30}{C.m}$) वह कोई द्विध्रुव है;
और किसी क्षेत्र में रखा कोई उदासीन परमाणु द्विध्रुव बन जाता है (प्रेरित
द्विध्रुव)।
\end{definition}

\begin{proposition}[किसी द्विध्रुव का क्षेत्र]\label{prop:b1:potential-capacitors:dipolefield}
$M$ पर, जहाँ $OM = r \gg a$ हो ($O$ यानी $NP$ का मध्य) और $\theta$
$\vect p$ तथा $\vect{OM}$ के बीच का कोण,
\[
V(M) = \frac{p\cos\theta}{4\pi\varepsilon_0 r^2}, \qquad
E_r = \frac{2p\cos\theta}{4\pi\varepsilon_0r^3}, \qquad
E_\theta = \frac{p\sin\theta}{4\pi\varepsilon_0r^3} .
\]
किसी द्विध्रुव का क्षेत्र $1/r^3$ की तरह गिरता है, यानी किसी आवेश के
$1/r^2$ से तेज़: दूर से दोनों आवेश लगभग कट जाते हैं, और जो बच रहता है वह
उनका अलगाव है।
\end{proposition}

\begin{proof}
$V = \dfrac{q}{4\pi\varepsilon_0}\Bigl(\dfrac1{PM} - \dfrac1{NM}\Bigr)$, जहाँ
$PM \approx r - \tfrac a2\cos\theta$ और $NM \approx r + \tfrac a2\cos\theta$
($\pm\tfrac a2$ का $\vect{OM}$ पर प्रक्षेप); और $a/r$ में प्रथम कोटि तक,
$1/PM - 1/NM \approx a\cos\theta/r^2$। फिर $E_r = -\partial V/\partial r$ और
$E_\theta = -(1/r)\,\partial V/\partial\theta$।
\end{proof}

\begin{proposition}[किसी बाह्य क्षेत्र में द्विध्रुव]\label{prop:b1:potential-capacitors:dipoleinfield}
\index{द्विध्रुव!बलाघूर्ण}\index{द्विध्रुव!ऊर्जा}
कोई द्विध्रुव $\vect p$, जो किसी एकसमान क्षेत्र $\vect E$ में हो, कोई
परिणामी बल नहीं पाता, पर एक \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} $\vect\Gamma = \vect p\wedge\vect E$ पाता है, जो
उसे क्षेत्र के अनुदिश पंक्तिबद्ध करना चाहता है, और उसकी \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा}
$E_p = -\vect p\cdot\vect E$ होती है, जो पंक्तिबद्ध होने पर निम्नतम है।
किसी असमान क्षेत्र में वह इसके अलावा प्रबल क्षेत्र वाले भागों की ओर खिंचता
भी है (एक विमा में, किसी पंक्तिबद्ध द्विध्रुव के लिए $F_x = p\,\dd E/\dd x$)।
\end{proposition}

\begin{proof}
बल $\pm q\vect E$ कट जाते हैं; और $O$ के परितः उनका आघूर्ण $\vect{OP}\wedge
q\vect E + \vect{ON}\wedge(-q\vect E) = q\vect{NP}\wedge\vect E$ है। ऊर्जा:
किसी एकसमान क्षेत्र के लिए $qV(P) - qV(N) = -q\,\vect{NP}\cdot\vect E$ (क्योंकि $V(P)
- V(N) = -\vect E\cdot\vect{NP}$)। असमान, और $x$ के अनुदिश पंक्तिबद्ध: $F_x =
qE(x + a) - qE(x) \approx qa\,\dd E/\dd x$।
\end{proof}

\begin{example}[कंघी काग़ज़ को क्यों खींचती है]\label{ex:b1:potential-capacitors:comb}
कोई आवेशित कंघी ऐसा क्षेत्र बनाती है जो दूरी के साथ गिरता जाता है; और
काग़ज़ का कोई उदासीन टुकड़ा उससे ध्रुवित हो जाता है ($\vect E$ के अनुदिश
प्रेरित द्विध्रुव), और किसी पंक्तिबद्ध द्विध्रुव पर बल $p\,\dd E/\dd x$ उसे
प्रबल क्षेत्र वाले भाग की ओर खींच लेता है: यानी कंघी की ओर। कंघी के आवेश का
चिह्न इसमें कोई भूमिका नहीं निभाता; वही क्रियाविधि गुब्बारे को दीवार पर
थामे रखती है और पानी की धार को मोड़ देती है। पानी के अणुओं के लिए,
\qty{1}{MV/m} के किसी क्षेत्र में $pE$
\qty{6e-24}{J} है, यानी कमरे के ताप पर के $k_BT$ से $600$ गुना कम:
अतः तापीय उछल-कूद के बाद केवल थोड़ा-सा परिणामी पंक्तिबद्ध होना ही बचता है
(\cref{exo:b1:potential-capacitors:7})।
\end{example}

\section{स्थिरवैद्युत साम्य में चालक}

\begin{theorem}[साम्य में किसी चालक के गुण]\label{thm:b1:potential-capacitors:conductor}
\index{साम्य में चालक}\index{कूलॉम की प्रमेय}\index{फ़ैराडे पिंजरा}
जिस चालक में मुक्त आवेश विराम में हों, उसमें:
\begin{enumerate}
  \item भीतर $\vect E = \vect 0$, और चालक कोई समविभव आयतन है:
        उसमें और उसके पृष्ठ पर $V$ एकसमान।
  \item कोई भी परिणामी आवेश पृष्ठ पर बैठता है, जिसका पृष्ठीय घनत्व
        $\sigma$ है; और भीतर उदासीन रहता है।
  \item ठीक बाहर क्षेत्र पृष्ठ पर अभिलंब होता है और उसका मान
        $\vect E = \dfrac{\sigma}{\varepsilon_0}\,\vect n$ है (\emph{\omterm{thm:b1:potential-capacitors:conductor}{कूलॉम की प्रमेय}})।
  \item किसी चालक के भीतर की कोई गुहा, जिसमें कोई आवेश न हो, $\vect E =
        \vect 0$ और एकसमान $V$ रखती है, बाहर चाहे कुछ भी होता रहे
        (\emph{\omterm{thm:b1:potential-capacitors:conductor}{फ़ैराडे पिंजरा}})।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) कोई क्षेत्र मुक्त आवेशों को चला देता: अतः साम्य के लिए $\vect E =
\vect 0$ चाहिए, जिससे भीतर के किन्हीं दो बिंदुओं के बीच $\dd V = 0$; और
पृष्ठ वही सीमा है। (2) चालक के भीतर खींचे किसी भी बंद पृष्ठ पर \omterm{thm:b1:electrostatics-gauss:gauss}{गाउस का
नियम}: शून्य अभिवाह, शून्य घिरा हुआ आवेश। (3) पृष्ठ एक समविभव है, अतः ठीक
बाहर $\vect E$ उस पर अभिलंब है; और पृष्ठ पर टिकी किसी डिबिया से केवल उसके
बाहरी फलक से ही अभिवाह $E\,\dd S$ जाता है (भीतर $\vect E = \vect 0$), और
आवेश $\sigma\,\dd S$ है। (4) गुहा की दीवार एक समविभव है; और गुहा के भीतर
किसी \omterm{def:b1:electrostatics-gauss:field}{क्षेत्र-रेखा} को दीवार से दीवार तक किसी विभव-गिरावट के पार जाना पड़ता
--- जो किसी समविभव पर असंभव है; अतः न कोई रेखा, न कोई क्षेत्र (इस रूप में
स्वीकृत)।
\end{proof}

\begin{example}[नोक-प्रभाव और तड़ित-चालक]\label{ex:b1:potential-capacitors:point}
\index{नोक-प्रभाव}
$R_1 > R_2$ त्रिज्याओं के दो गोले, जो दूर-दूर हों और किसी तार से जुड़े हों,
वही $V = Q_i/4\pi\varepsilon_0R_i$ साझा करते हैं, अतः $\sigma_i = Q_i/4\pi R_i^2 = \varepsilon_0V/R_i$:

आयनित हो जाती है और आवेश चुपचाप रिस जाता है। कोई तड़ित-चालक इसी

देता है।

\section{संधारित्र}

\begin{definition}[धारिता]\label{def:b1:potential-capacitors:capacitance}
\index{धारिता}\index{संधारित्र}\index{फ़ैरड}
विभव $V$ पर का कोई विलगित चालक ऐसा आवेश $Q$ लिए रहता है जो
$V$ के समानुपाती है: $Q = CV$, जहाँ $C$ उसकी \emph{धारिता} है (फ़ैरड,
\unit{F}); और किसी गोले के लिए $C = 4\pi\varepsilon_0R$। कोई \emph{संधारित्र}
चालकों का ऐसा युग्म (\emph{पट्टिकाएँ}) है जो विपरीत आवेश $\pm Q$ लिए हों
और एक-दूसरे के सामने हों, ताकि क्षेत्र उन्हीं के बीच सीमित रहे; उसकी
धारिता $C = Q/U$ है, जहाँ $U = V_1 - V_2$ पट्टिकाओं के बीच की वोल्टता है।
संकेत और परिपथ-नियम $i = C\,\dd u/\dd t$
\cref{ch:b1:transient-regimes} में काम आ चुके हैं।
\end{definition}

\begin{proposition}[तीन संधारित्र]\label{prop:b1:potential-capacitors:three}
\index{समतल संधारित्र}\index{गोलीय संधारित्र}\index{बेलनाकार संधारित्र}
\begin{enumerate}
  \item समतल: $S$ क्षेत्रफल की पट्टिकाएँ, जो $d \ll \sqrt S$ दूरी पर हों:
        $C = \dfrac{\varepsilon_0S}{d}$।
  \item गोलीय: $R_1 < R_2$ त्रिज्याओं के संकेंद्री गोले:
        $C = \dfrac{4\pi\varepsilon_0R_1R_2}{R_2 - R_1}$।
  \item बेलनाकार (समाक्ष), त्रिज्याएँ $a < b$, लंबाई $L \gg b$:
        $C = \dfrac{2\pi\varepsilon_0L}{\ln(b/a)}$।
\end{enumerate}
दरार को आपेक्षिक विद्युतशीलता
$\varepsilon_r$ के किसी विद्युतरोधी से भर देने पर $C$ $\varepsilon_r$ गुना हो जाता है
(परावैद्युत के अणु ध्रुवित होकर क्षेत्र कमज़ोर कर देते हैं; आम प्लास्टिकों
और काँच के लिए $\varepsilon_r = 2$ से $10$, और पानी के लिए $80$)।
\end{proposition}

\begin{proof}
हर स्थिति में पट्टिकाओं के बीच क्षेत्र वही है जो
\cref{prop:b1:electrostatics-gauss:three} में है: $E = \sigma/\varepsilon_0 = Q/\varepsilon_0S$,
$U = Ed$; $E = Q/4\pi\varepsilon_0r^2$, $U = \int_{R_1}^{R_2}E\,\dd r = \dfrac{Q}{4\pi\varepsilon_0}
\Bigl(\dfrac1{R_1} - \dfrac1{R_2}\Bigr)$; $E = Q/2\pi\varepsilon_0Lr$, $U =
\dfrac{Q}{2\pi\varepsilon_0L}\ln\dfrac ba$। $Q$ को $U$ से भाग दीजिए। परावैद्युत
वाला गुणक स्वीकृत है।
\end{proof}

\begin{theorem}[किसी संधारित्र की ऊर्जा]\label{thm:b1:potential-capacitors:energy}
\index{संधारित्र की ऊर्जा}\index{विद्युत क्षेत्र का ऊर्जा घनत्व}
$Q$ तक आवेशित कोई \omterm{def:b1:potential-capacitors:capacitance}{संधारित्र}, $U$ के नीचे, इतना संचित रखता है
\[
W = \tfrac12 CU^2 = \tfrac12 QU = \frac{Q^2}{2C} ,
\]
यानी वह कार्य जो आवेश को बढ़ती वोल्टता के विरुद्ध एक पट्टिका से दूसरी तक ले
जाने में लगता है। किसी \omterm{prop:b1:potential-capacitors:three}{समतल संधारित्र} में $W = \tfrac12\varepsilon_0E^2
\times Sd$: यानी ऊर्जा क्षेत्र में है, प्रति एकांक आयतन
$\tfrac12\varepsilon_0E^2$ घनत्व पर --- और यह कथन हर स्थिरवैद्युत क्षेत्र के लिए
सही है।
\end{theorem}

\begin{proof}
$\dd q$ को ऋणात्मक से धनात्मक पट्टिका तक ले जाने में, जब आवेश $q$ हो,
$\dd W = (q/C)\,\dd q$ लगता है; इसे $0$ से $Q$ तक समाकलित कीजिए। फिर
$C = \varepsilon_0S/d$ और $U = Ed$ रखिए; और सामान्य घनत्व स्वीकृत है।
\end{proof}

\begin{example}[तीन संधारित्र और उनकी ऊर्जाएँ]\label{ex:b1:potential-capacitors:energies}
\qty{1}{mm} की दूरी पर पन्नी के एक वर्ग मीटर के जोड़े के लिए: $C = 8.85 \times 10^{-12}/
10^{-3} = \qty{8.9}{nF}$ --- यानी \omterm{def:b1:potential-capacitors:capacitance}{फ़ैरड} एक बहुत बड़ा मात्रक है। किसी
डिफ़िब्रिलेटर के \qty{150}{\micro F}, \qty{2}{kV} पर: \qty{300}{J}, जो
मिलीसेकंडों में दे दिए जाते हैं। \qty{2.7}{V} पर \qty{3000}{F} का कोई
अतिसंधारित्र: किसी शीतल-पेय की बोतल जितने डिब्बे में \qty{11}{kJ} ---
उसकी \omterm{def:b1:potential-capacitors:capacitance}{धारिता} किसी आणविक दरार $d$ और किसी विशाल सरंध्र $S$ से आती है। और हवा
के भंजन पर किसी क्षेत्र का ऊर्जा घनत्व, $\tfrac12\varepsilon_0(3 \times 10^6)^2 = \qty{40}{J/m^3}$,
वही कारण है जिससे \omterm{def:b1:potential-capacitors:capacitance}{संधारित्र} बैटरियों के सामने इतना कम संचित रखते हैं।
\end{example}

\begin{remark}[संधारित्रों का संयोजन]\label{rem:b1:potential-capacitors:association}
समांतर में (वही $U$) आवेश जुड़ जाते हैं: $C = C_1 + C_2$; और श्रेणी में (वही
$Q$) वोल्टताएँ जुड़ जाती हैं: $1/C = 1/C_1 + 1/C_2$ --- यानी प्रतिरोधों से
उलटा, क्योंकि $C$ एक सुगमता नापता है (प्रति वोल्ट आवेश) जहाँ $R$ एक कठिनाई
नापता है। ये नियम $Q = CU$ और \cref{ch:b1:dc-circuits} के परिपथ-नियमों से
निकलते हैं।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:1}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:2}
\qty{100}{V} के नीचे \qty{1.0}{mm} की दूरी पर पट्टिकाएँ: क्षेत्र; विराम से

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:4}
\qty{15}{cm} त्रिज्या का कोई वान डे ग्राफ़ गोला \qty{200}{kV} तक आवेशित है।
\omterm{def:b1:potential-capacitors:capacitance}{धारिता}, आवेश, और उसके पृष्ठ पर क्षेत्र। उसके चारों ओर की हवा के भंजन
(\qty{3}{MV/m}) से पहले अधिकतम विभव।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:5}
$R$ त्रिज्या और $Q$ आवेश के किसी वलय की अक्ष पर विभव; और अवकलन से अक्षीय
क्षेत्र फिर से पाइए। एकसमान आवेशित चकती के लिए भी वही
(\cref{exo:b1:electrostatics-gauss:5} लीजिए): $V(z) = \dfrac{\sigma}{2\varepsilon_0}
\bigl(\sqrt{z^2 + R^2} - z\bigr)$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:6}
एकसमान आवेशित किसी गोले के क्षेत्र से उसके भीतर का विभव व्युत्पन्न कीजिए।
सोने के किसी नाभिक ($Z = 79$, $R = \qty{7.0}{fm}$) के केंद्र पर विभव;
और दूर से केंद्र तक पहुँचने के लिए किसी प्रोटॉन को कितनी ऊर्जा चाहिए,
\unit{MeV} में; रदरफ़ोर्ड के \qty{5}{MeV} अल्फ़ा कणों पर टिप्पणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:7}
पानी का अणु, $p = \qty{6.2e-30}{C.m}$। वह अपनी अक्ष के अनुदिश और उस पर लंब,
\qty{1.0}{nm} पर जो क्षेत्र बनाता है; $E = \qty{1.0}{MV/m}$ में पंक्तिबद्ध
तथा विपरीत-पंक्तिबद्ध स्थितियों के बीच \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} और ऊर्जा-अंतर;
\qty{300}{K} पर $k_BT$ से तुलना कीजिए। \qty{0.3}{nm} पर किसी सोडियम आयन का
क्षेत्र, और उसमें पंक्तिबद्ध पानी के द्विध्रुव की ऊर्जा (जलयोजन)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:8}
हवा में \qty{1.0}{cm} त्रिज्या का कोई चालक गोला: भंजन से पहले अधिकतम आवेश
और विभव। दो चालक गोले, $R_1 = \qty{10}{cm}$
और $R_2 = \qty{1.0}{mm}$, जो किसी पतले तार से जुड़े हों और \qty{20}{kV} तक

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:9}
धारिताएँ: \qty{10}{\micro m} की दूरी पर \qty{1.0}{m^2} की दो पन्नियाँ, बीच

पन्नियाँ लपेटकर --- क्या लपेटने से $C$ बदलता है? \qty{400}{V} पर ऊर्जा। और
\qty{2.0}{kV} पर \qty{150}{\micro F} का कोई
\omterm{def:b1:potential-capacitors:capacitance}{संधारित्र} (डिफ़िब्रिलेटर): आवेश, ऊर्जा, और \qty{5}{ms} में दे देने पर माध्य
शक्ति।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:10}
\emph{गोलीय और \omterm{prop:b1:potential-capacitors:three}{बेलनाकार संधारित्र}।} (क) संकेंद्री गोलों के लिए $C$
व्युत्पन्न कीजिए और जाँचिए कि $R_2 - R_1 = d \ll R_1$ के लिए वह समतल सूत्र
पर उतर आता है, और $R_2 \to \infty$ के लिए विलगित गोले पर।
(ख) समाक्ष तार, $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, पॉलिएथिलीन
$\varepsilon_r = 2.3$: प्रति मीटर \omterm{def:b1:potential-capacitors:capacitance}{धारिता} (मानक तारों के \qty{100}{pF/m}
से तुलना कीजिए)। (ग) \qty{1}{kV} के नीचे भीतरी चालक पर क्षेत्र;

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:11}

विलगित \omterm{prop:b1:potential-capacitors:three}{समतल संधारित्र} की पट्टिकाएँ $d$ से $2d$ तक खींची जाती हैं: संचित
ऊर्जा में परिवर्तन, और उससे निकला पट्टिकाओं के बीच का बल;
\cref{exo:b1:electrostatics-gauss:8} से जाँचिए। (घ) वही, पर \omterm{def:b1:potential-capacitors:capacitance}{संधारित्र} किसी
बैटरी से नियत $U$ पर रखा हुआ: ऊर्जा में परिवर्तन, और अंतर जाता कहाँ है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:12}

ऊर्जा। (घ) विखंडन की मापी गई ऊर्जा लगभग \qty{200}{MeV} है: कूलॉम वाले लाभ
का विरोध कौन करता है?
\end{exercise}

\section{समस्या: मिलिकन की तेल-बूँद}

\begin{problem}[{सप्ताहांत समस्या --- तेल की एक बूँद से इलेक्ट्रॉन का आवेश
तौलना: स्टोक्स, समतल संधारित्र, और आँकड़ों की एक ऐसी सारणी जो किसी एक के
गुणजों के सिवा कुछ बनने से इनकार कर देती है}]
\label{pb:b1:potential-capacitors:1}
तेल की बूँदें $d = \qty{16}{mm}$ की दूरी पर रखी दो क्षैतिज पट्टिकाओं के बीच
छिड़की जाती हैं, जो \qty{20}{cm} व्यास की हैं, और किसी \omterm{def:b1:optical-instruments:microscope}{सूक्ष्मदर्शी} से
देखी जाती हैं। आँकड़े: तेल का घनत्व $\rho = \qty{900}{kg/m^3}$, हवा का घनत्व
\qty{1.2}{kg/m^3}, हवा की श्यानता $\eta = \qty{1.8e-5}{Pa.s}$, $g =
\qty{9.8}{m/s^2}$, $\varepsilon_0 = \qty{8.85e-12}{F/m}$। हवा में
$r$ त्रिज्या का कोई छोटा गोला, जो $v$ से चल रहा हो, स्टोक्स कर्षण $6\pi\eta rv$ अनुभव करता है।

\medskip
\noindent\textbf{भाग I --- गिरती बूँद।}
\begin{enumerate}
  \item बिना कोई वोल्टता लगाए गिरती किसी बूँद पर लगने वाले बल; दिखाइए कि
        वह \omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग} $v_1 = 2\rho gr^2/9\eta$ तक पहुँच जाती है (हवा के
        उत्प्लावन की उपेक्षा कीजिए --- और उसे न्यायसंगत ठहराइए)।
  \item कोई बूँद \qty{10}{s} में \qty{1.0}{mm} गिरती है: उसकी त्रिज्या और
        द्रव्यमान।
  \item $v_1$ तक पहुँचने का \omterm{def:b1:transient-regimes:firstorder}{कालांक}; और इस बीच तय की गई दूरी। क्या बूँद कभी
        त्वरित होती दिखती है?
  \item गति की रेनॉल्ड्स संख्या $\rho_{\text{हवा}}v_1r/\eta$: क्या \omterm{prop:b1:newton-dynamics:drag}{स्टोक्स
        का नियम} यहाँ जायज़ है?
  \item बूँदों का इतना छोटा होना क्यों ज़रूरी है? (भार की तुलना
        \qty{3e5}{V/m} के किसी क्षेत्र में एक \omterm{def:b1:electrostatics-gauss:charge}{मूल आवेश} पर लगने वाले विद्युत
        बल से कीजिए।)
  \item बूँद अपने माध्य गिरने के आसपास ज़रा-सा काँपती दिखती है: यह क्या है,
        और इससे यथार्थता क्यों सीमित हो जाती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- क्षेत्र।}
कोई वोल्टता $U$ लगाई जाती है, जिसमें ऊपरी पट्टिका धनात्मक है।
\begin{enumerate}[resume]
  \item $U = \qty{5.0}{kV}$ के लिए पट्टिकाओं के बीच क्षेत्र; वह एकसमान
        क्यों है, और समविभव क्या हैं?
  \item पट्टिकाओं की \omterm{def:b1:potential-capacitors:capacitance}{धारिता}, \qty{5.0}{kV} पर उन पर आवेश, और संचित
        ऊर्जा।
  \item प्रश्न~2 वाली बूँद $U_0 = \qty{1080}{V}$ पर स्थिर थमी रहती है:
        उसके आवेश का चिह्न और मान, कूलॉम में और $e$ के मात्रकों में।
  \item $U = \qty{5.0}{kV}$ के साथ वही बूँद $v_2$ से ऊपर उठती है: दिखाइए
        कि $q = 6\pi\eta r(v_1 + v_2)d/U$ है और $v_2$ निकालिए।
  \item संतुलन-वोल्टता ढूँढ़ते रहने से चढ़ाई और गिरावट का समय लेना बेहतर
        क्यों है?
  \item \qty{5}{kV} पर पट्टिकाओं के बीच बूँद के आवेश की \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा}, और
        \qty{1}{cm} की किसी चढ़ाई के दौरान कर्षण से क्षय हुई ऊर्जा।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- आवेशों की सारणी।}
मापों के बीच-बीच में X-किरणों के किसी कमज़ोर स्रोत के सामने रखी जाने पर वही
बूँद अपना आवेश बदल लेती है; और क्रमिक मापों से मिलते हैं (यहाँ $10^{-19}$~कूलॉम में):
$4.80$, $8.01$, $6.42$, $3.19$, $9.63$।
\begin{enumerate}[resume]
  \item दिखाइए कि पाँचों किसी एक राशि के पूर्णांक गुणजों के पास हैं, और वे
        पूर्णांक दीजिए।
  \item इन पाँचों मानों से \omterm{def:b1:electrostatics-gauss:charge}{मूल आवेश} का सर्वोत्तम आकलन; और माध्य की \omterm{def:b1:units-dimensions:uncertainty}{मानक
        अनिश्चितता} (\cref{ch:b1:units-dimensions})।
  \item मापों के बीच आवेश $e$ की पूरी-पूरी संख्याओं से ही क्यों बदलता है?
  \item दिखाइए कि इस विधि में $q \propto \eta^{3/2}$ है: तो श्यानता पर
        $1\%$ की त्रुटि $e$ को कितना खिसका देती है? (मिलिकन का मूल मान
        इसी कारण $0.6\%$ कम था।)
  \item क्वार्क $\pm e/3$ और $\pm 2e/3$ लिए रहते हैं: फिर किसी तेल-बूँद ने
        $e$ का तिहाई भाग कभी क्यों नहीं दिखाया?
  \item फ़ैराडे नियतांक, यानी इलेक्ट्रॉनों के एक \omterm{def:b1:kinetic-theory:scales}{मोल} का आवेश,
        $F = \qty{96485}{C/mol}$ है: इससे आवोगाद्रो संख्या निकालिए।
  \item टॉमसन ने इलेक्ट्रॉन के लिए $e/m_e = \qty{1.76e11}{C/kg}$ मापा था:
        इससे उसका द्रव्यमान निकालिए, और किसी हाइड्रोजन परमाणु
        (\qty{1.67e-27}{kg}) से तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विभव की दृष्टि से उपकरण।}
\begin{enumerate}[resume]
  \item पट्टिकाओं के बीच समविभव और क्षेत्र-रेखाएँ खींचिए, किनारों के पास
        के फैलाव सहित; और उसकी उपेक्षा न्यायसंगत ठहराइए।
  \item निचली पट्टिका (जो \qty{0}{V} पर है) से $z$ ऊँचाई पर कोई धनात्मक
        आवेशित बूँद: $z$ के फलन के रूप में \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा}; गुरुत्व सहित कुल
        \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा}; और स्थिर बिंदु के लिए $U$ पर शर्त।
  \item बूँद के आवेश की ऊर्जा (प्रश्न~12) की तुलना \omterm{def:b1:potential-capacitors:capacitance}{संधारित्र} में संचित
        ऊर्जा से कीजिए: क्या बूँद क्षेत्र को बिगाड़ती है?
  \item वोल्टता उलट दी जाती है (निचली पट्टिका धनात्मक), जहाँ
        $U = \qty{5.0}{kV}$: तो प्रश्न~9 वाली बूँद क्या करती है, और किस
        चाल से?
  \item उसी क्षेत्र में \qty{1}{\micro g} का कोई धूल-कण, जो $10^4e$ लिए हो:
        विद्युत बल और भार का अनुपात; ऐसे कण तैराए क्यों नहीं जा सकते और
        मिलिकन ने तेल की फुहार क्यों चुनी।
  \item सार दीजिए: क्या-क्या मापा गया (तीन बार), क्या निकाला गया, किस
        अनिश्चितता के साथ, और उससे कौन-से दो नियतांक निकल आए।
\end{enumerate}
\end{problem}
