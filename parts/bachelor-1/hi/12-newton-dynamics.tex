\chapter{गतिकी: न्यूटन के नियम}\label{ch:b1:newton-dynamics}

कोई पैराशूट-यात्री विमान से बाहर कदम रखता है और दस-बारह सेकंड के भीतर उसका
त्वरण थम जाता है: दो सौ किलोमीटर प्रति घंटा पर हवा ठीक उतने ही ज़ोर से पीछे
धकेलती है जितने से पृथ्वी खींचती है। कोई कार मोड़ को बहुत तेज़ी से लेती है
और टायर साथ छोड़ देते हैं; पहाड़ी सड़क पर कोई ट्रक नीची गियर में रेंगता है,
उसी घर्षण के सहारे जो उसे समतल पर रोक देता है। शुद्धगतिकी ने गतियों का
वर्णन किया; गतिकी उन्हें समझाती है, और वह भी एक ही नियम से, जो किसी पिंड के
त्वरण को उस पर लगते बलों से जोड़ता है। यह अध्याय न्यूटन के तीन नियम कहता है,
इस खंड के स्तर पर मिलने वाले बलों की सूची बनाता है --- भार, स्प्रिंग,
डोरियाँ, संस्पर्श, शुष्क और \omterm{prop:b1:newton-dynamics:drag}{तरल घर्षण} --- और दिखाता है कि बलों के चित्र से
किसी अवकल समीकरण और उसके हल तक कैसे पहुँचा जाए।

\section{द्रव्यमान, संवेग और तीन नियम}

\begin{definition}[द्रव्यमान, संवेग, बल]\label{def:b1:newton-dynamics:momentum}
\index{द्रव्यमान}\index{संवेग}\index{बल}
किसी \omterm{def:b1:point-kinematics:frame}{बिंदु कण} का एक \emph{जड़त्वीय द्रव्यमान} $m > 0$ (किलोग्राम) होता है, जो
वेग के परिवर्तनों के प्रति उसके प्रतिरोध का माप है और तंत्र से स्वतंत्र है।
किसी तंत्र में उसका \emph{संवेग} है
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
\emph{बल} $\vect F$ (न्यूटन) किसी दूसरे पिंड की उस कण पर की गई क्रिया है;
बल सदिशों की तरह जुड़ते हैं, और उनका योग $\sum\vect F$ उनका \emph{परिणामी}
कहलाता है।
\end{definition}

\begin{theorem}[न्यूटन के नियम]\label{thm:b1:newton-dynamics:laws}
\index{न्यूटन के नियम}\index{जड़त्वीय तंत्र}\index{जड़त्व का सिद्धांत}\index{क्रिया और प्रतिक्रिया}
\begin{enumerate}
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{जड़त्व का सिद्धांत}।} ऐसे तंत्र होते हैं, जिन्हें
        \emph{जड़त्वीय} (या गैलीलीय) कहा जाता है, जिनमें किसी \omterm{def:b1:newton-dynamics:momentum}{बल} के अधीन
        नहीं, अथवा शून्य परिणामी वाले बलों के अधीन, कोई कण सरल रेखा में
        अचर वेग से चलता है। किसी \omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} के सापेक्ष एकसमान सरल
        रेखीय स्थानांतरण करता हर तंत्र स्वयं जड़त्वीय होता है।
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{संवेग} का नियम।} किसी \omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} में,
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{अर्थात्}\qquad
        m\,\vect a = \sum\vect F \ \text{ अचर द्रव्यमान } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{क्रिया और प्रतिक्रिया}।} यदि पिंड $A$ \omterm{def:b1:newton-dynamics:momentum}{बल} $\vect F_{A\to B}$ को
        पिंड $B$ पर लगाता है, तो $B$ भी $\vect F_{B\to A} = -\vect F_{A\to B}$
        को $A$ पर लगाता है, और वह भी उन्हें जोड़ने वाली रेखा के अनुदिश।
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[नियम कहते क्या हैं]\label{rem:b1:newton-dynamics:meaning}
पहला नियम दूसरे का कोई विशेष प्रसंग नहीं है: वह उन्हीं तंत्रों को
\emph{परिभाषित} करता है जिनमें दूसरा लागू होता है। व्यवहार में पृथ्वी पर की
कोई प्रयोगशाला कुछ मिनट चलने वाले प्रयोगों के लिए उत्कृष्ट यथार्थता तक
जड़त्वीय है (\cref{ch:b1:non-inertial-frames} उस त्रुटि को माप देता है);
सूर्य और दूर के तारों से जुड़ा तंत्र और भी अच्छा है। दूसरा नियम एक सदिश
समीकरण है: तीन अदिश समीकरण, प्रति अक्ष एक, और अक्षों का चुनाव
(\cref{ch:b1:point-kinematics}) हर समस्या का पहला निर्णय होता है। तीसरा नियम
इस खंड के संस्पर्श और गुरुत्वीय बलों के लिए हर क्षण लागू होता है।
\end{remark}

\begin{definition}[संवृत निकाय; संवेग संरक्षण]\label{def:b1:newton-dynamics:conservation}
\index{संवृत निकाय}\index{संवेग संरक्षण}
\emph{संवृत निकाय} बाहर के साथ पदार्थ का कोई आदान-प्रदान नहीं करता। कणों के
किसी निकाय में आंतरिक \omterm{def:b1:newton-dynamics:momentum}{बल} जोड़ों में कट जाते हैं (तीसरा नियम), और कुल \omterm{def:b1:newton-dynamics:momentum}{संवेग}
$\vect P = \sum m_i\vect v_i$ इस नियम का पालन करता है:
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$; बाह्य बलों के लुप्त होने पर
वह संरक्षित रहता है (\cref{ch:b1:systems-of-points} इसे आगे बढ़ाता है)।
\end{definition}

\section{सामान्य बल}

\begin{proposition}[भार और गुरुत्वाकर्षण]\label{prop:b1:newton-dynamics:weight}
\index{भार}\index{गुरुत्वीय क्षेत्र}
पृथ्वी के पृष्ठ के पास $m$ \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान} का कोई पिंड अपने \emph{भार}
$\vect P = m\vect g$ से खींचा जाता है, जहाँ $\vect g$ स्थानीय \omterm{prop:b1:newton-dynamics:weight}{गुरुत्वीय
क्षेत्र} है, $g \approx \qty{9.81}{m/s^2}$, और दिशा नीचे की ओर (लगभग ठीक
केंद्र की ओर)। और अधिक व्यापक रूप से, $M$ \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान} की कोई वस्तु $r$ दूरी
पर स्थित $m$ को $\vect F = -GMm/r^2\,\vect e_r$ से आकर्षित करती है
(\cref{ch:b1:central-forces}); $R$ त्रिज्या के गोले के पृष्ठ पर
$g = GM/R^2$। मात्रक पर ध्यान दीजिए: \unit{N/kg} और \unit{m/s^2} एक ही हैं,
और इसीलिए सभी पिंड एक-से गिरते हैं।
\end{proposition}

\begin{proof}
न्यूटन का गुरुत्वाकर्षण नियम यहाँ स्वीकृत है और
\cref{ch:b1:central-forces} में उसकी पड़ताल की गई है; $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$।
\end{proof}

\begin{proposition}[स्प्रिंग; डोरी]\label{prop:b1:newton-dynamics:springstring}
\index{हुक का नियम}\index{स्प्रिंग}\index{तनाव}
$k$ दृढ़ता और $\ell_0$ स्वाभाविक लंबाई वाली कोई आदर्श स्प्रिंग, जिसे $\ell$
लंबाई तक उस एकांक सदिश $\vect u$ के अनुदिश तान दिया गया हो जो उसके स्थिर
सिरे से पिंड की ओर इंगित करता है, पिंड पर यह \emph{प्रत्यानयन \omterm{def:b1:newton-dynamics:momentum}{बल}} लगाती है
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{हुक का नियम}),
\]
जो उसकी स्वाभाविक लंबाई की ओर होता है। कोई आदर्श डोरी (द्रव्यमानहीन,
अवितान्य) अपने ही अनुदिश एक \emph{तनाव} $\vect T$ लगाती है, जो खींचता है,
और जो दोनों सिरों पर तथा पूरी लंबाई में एक ही परिमाण का होता है (चाहे बीच
में कोई द्रव्यमानहीन, घर्षणहीन घिरनी क्यों न हो)।
\end{proposition}

\begin{proof}
\omterm{prop:b1:newton-dynamics:springstring}{हुक का नियम} एक आनुभविक प्रतिरूप है जो छोटे विरूपणों के लिए वैध है। डोरी:
कोई द्रव्यमानहीन अवयव न्यूटन के दूसरे नियम से $0 = \vect T_1 + \vect T_2$
मानता है, अतः तनाव बिना बदले संचरित होता है।
\end{proof}

\begin{proposition}[संस्पर्श बल: अभिलंब प्रतिक्रिया और शुष्क घर्षण]\label{prop:b1:newton-dynamics:friction}
\index{अभिलंब प्रतिक्रिया}\index{शुष्क घर्षण}\index{घर्षण के कूलॉम नियम}\index{घर्षण गुणांक}
कोई ठोस पृष्ठ अपने संपर्क में आए पिंड पर एक \emph{\omterm{prop:b1:newton-dynamics:friction}{अभिलंब प्रतिक्रिया}}
$\vect N$ लगाता है, जो पृष्ठ पर लंब और प्रतिकर्षी होती है ($N \geq 0$: कोई
पृष्ठ खींच नहीं सकता), और साथ ही एक स्पर्शरेखीय \emph{घर्षण \omterm{def:b1:newton-dynamics:momentum}{बल}} $\vect T$।
कूलॉम के नियम: यदि पिंड फिसलता नहीं, तो $\abs{\vect T} \leq \mu_s N$
(\emph{स्थैतिक} घर्षण जितना आवश्यक हो उतना समायोजित हो जाता है, उस सीमा
तक); और यदि वह पृष्ठ के सापेक्ष $\vect v_g$ वेग से फिसलता है, तो
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (\emph{गतिज} घर्षण, जो
फिसलन का विरोध करता है), जहाँ गुणांक $\mu_d \leq \mu_s$ पदार्थों के अनुसार
$0.1$ से $1$ तक की कोटि के होते हैं, और संपर्क क्षेत्रफल तथा चाल दोनों से
स्वतंत्र होते हैं।
\end{proposition}
\admitted

\begin{proposition}[तरल घर्षण]\label{prop:b1:newton-dynamics:drag}
\index{तरल घर्षण}\index{कर्षण}\index{स्टोक्स का नियम}
विराम में पड़े किसी तरल में से $\vect v$ वेग से गुजरता कोई पिंड $\vect v$ के
विपरीत एक \emph{कर्षण} अनुभव करता है: कम चाल पर (छोटा आकार, श्यान तरल) वह
रैखिक होता है, $\vect F = -\alpha\vect v$ --- $r$ त्रिज्या के ऐसे गोले के
लिए जो $\eta$ श्यानता वाले तरल में हो, $\alpha = 6\pi\eta r$ (स्टोक्स); और
ऊँची चाल पर (हवा या पानी में बड़े, तेज़ पिंड) वह द्विघाती होता है,
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
जहाँ $\rho$ तरल का घनत्व है, $S$ प्रवाह के सामने पड़ती पिंड की अनुप्रस्थ
काट, और $C_x$ एक \omterm{def:b1:units-dimensions:dimension}{विमाहीन} \emph{कर्षण गुणांक} (गोले के लिए $0.5$, कार के
लिए $0.3$, और किसी चपटी चकती के लिए $1$)। रेनॉल्ड्स संख्या
$\rho vL/\eta$ (\cref{rem:b1:units-dimensions:pi}) तय करती है: $1$ से नीचे
रैखिक, और लगभग $1000$ से ऊपर द्विघाती।
\end{proposition}
\admitted

\begin{remark}[अन्य बल]\label{rem:b1:newton-dynamics:others}
उत्प्लावन (आर्किमिडीज़) दाब-बलों का परिणामी है, जो
\cref{ch:b1:fluid-statics} में व्युत्पन्न किया गया है; किसी आवेश पर लगते
विद्युत और चुंबकीय \omterm{def:b1:newton-dynamics:momentum}{बल} \cref{ch:b1:charged-particles} का विषय हैं। इस अध्याय
के सारे \omterm{def:b1:newton-dynamics:momentum}{बल} संस्पर्श या क्षेत्र-अंतःक्रियाओं के \emph{प्रतिरूप} हैं, जो कही
गई सीमाओं के भीतर ही वैध हैं --- हुक की स्प्रिंग टूट जाती है, कूलॉम का
घर्षण ऊँचे दाब पर काम नहीं करता, स्टोक्स का कर्षण ऊँची चाल पर विफल हो जाता
है। हर हल का एक हिस्सा यह जाँचना है कि प्रयुक्त प्रतिरूप लागू होता है या नहीं।
\end{remark}

\section{गतिकी की किसी समस्या का हल}

\begin{method}[स्थिति से समीकरण तक]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{निकाय}: उस पिंड का नाम लीजिए जिसे बिंदु माना जा रहा है।
  \item \emph{तंत्र}: उसका नाम लीजिए और जाँचिए कि वह जड़त्वीय है (या
        अपेक्षित यथार्थता तक उसे वैसा मान लीजिए)।
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{बल}}: निकाय को छूते या आकर्षित करते हर पिंड की सूची बनाइए, और
        हर \omterm{def:b1:newton-dynamics:momentum}{बल} को एक आरेख पर खींचिए (\omterm{def:b1:newton-dynamics:momentum}{बल} आरेख); इसके अलावा और कुछ नहीं आता।
  \item \emph{निर्देशांक}: गति के अनुकूल अक्ष चुनिए
        (\cref{met:b1:point-kinematics:choose}) और $\vect a$ व्यक्त कीजिए।
  \item $m\vect a = \sum\vect F$ को अक्षों पर \emph{प्रक्षिप्त} कीजिए;
        प्रतिबंधों का प्रयोग कीजिए (पृष्ठ पर रखा पिंड उसी पर बना रहता है:
        $a_\perp = 0$; नियत लंबाई की डोरी: दोनों सिरों पर बराबर चाल)।
  \item \emph{हल कीजिए} और \emph{जाँचिए}: विमाएँ, सीमांत प्रसंग, चिह्न
        ($N \geq 0$, और विराम में हो तो $\abs T \leq \mu_sN$)।
\end{enumerate}
\end{method}

\begin{example}[नत तल पर गुटका]\label{ex:b1:newton-dynamics:incline}
$m$ \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान} का गुटका $\alpha$ पर झुके किसी तल पर, गुणांक $\mu_s$,
$\mu_d$। अक्ष: $x$ ढाल के नीचे की ओर, $y$ अभिलंब के अनुदिश। भार
$(mg\sin\alpha, -mg\cos\alpha)$, प्रतिक्रिया $(0, N)$, और नीचे फिसलने पर
घर्षण $(-T, 0)$। $y$ के अनुदिश: $N = mg\cos\alpha$। विराम में $x$ के
अनुदिश: $T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$, अर्थात् $\tan\alpha
\leq \mu_s$: गुटका $\arctan\mu_s$ कोण तक टिका रहता है --- यह $\mu_s$ का
ऐसा मापन है जिसमें किसी तुला की आवश्यकता नहीं। फिसलते हुए: $ma = mg\sin\alpha -
\mu_dmg\cos\alpha$, अतः $a = g(\sin\alpha - \mu_d\cos\alpha)$, जो $m$ से
स्वतंत्र है: $\alpha = 30^\circ$, $\mu_d = 0.2$ के लिए \qty{3.2}{m/s^2}।
\end{example}

\begin{example}[सरल लोलक]\label{ex:b1:newton-dynamics:pendulum}
\index{सरल लोलक}
$m$ \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान}, $\ell$ लंबाई की डोरी पर, नीचे की ओर की ऊर्ध्वाधर से
$\theta$ कोण पर। निलंबन बिंदु पर \omterm{def:b1:point-kinematics:cylindrical}{ध्रुवीय निर्देशांक}: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$; \omterm{def:b1:newton-dynamics:momentum}{बल}: भार
$mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$ और तनाव
$-T\vect e_r$। $\vect e_\theta$ के अनुदिश: $m\ell\ddot\theta =
-mg\sin\theta$,
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
जो छोटे कोणों के लिए ($\sin\theta \approx \theta$) वह आवर्त समीकरण है
जिसमें $\omega_0 = \sqrt{g/\ell}$: आवर्तकाल $T_0 = 2\pi\sqrt{\ell/g}$, यानी
वही $k = 2\pi$ जो \cref{ex:b1:units-dimensions:pendulum} नहीं दे सका था।
$\vect e_r$ के अनुदिश: $T = mg\cos\theta + m\ell\dot\theta^2$ --- डोरी सबसे
अधिक तल पर खींचती है, जहाँ चाल सबसे बड़ी होती है।
\end{example}

\begin{proposition}[कर्षण के साथ पतन; सीमांत वेग]\label{prop:b1:newton-dynamics:terminal}
\index{सीमांत वेग}
$m$ \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान} का कोई पिंड, जो अपने भार और किसी कर्षण \omterm{def:b1:newton-dynamics:momentum}{बल} के अधीन विराम से
गिरता है, उस \emph{\omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग}} तक पहुँच जाता है जहाँ दोनों संतुलित हो जाते
हैं। रैखिक कर्षण: $m\dot v = mg - \alpha v$, $v = v_\infty(1 - \eu^{-t/\tau})$,
जहाँ $v_\infty = mg/\alpha$, $\tau = m/\alpha$। द्विघाती कर्षण: $m\dot v = mg -
\tfrac12\rho C_xSv^2$, $v = v_\infty\tanh(t/\tau)$, जहाँ
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
रैखिक: यह प्रथम कोटि का समीकरण है, $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder})। द्विघाती: $u = v/v_\infty$ के
साथ $\dot u = (1 - u^2)/\tau$; चरों को पृथक करने पर
$\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$ (परिमेय भिन्न
$1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$ का समाकलन
$\tfrac12\ln\frac{1 + u}{1 - u}$ है), अतः $u = \tanh(t/\tau)$।
\end{proof}

\begin{example}[दो सीमांत वेग]\label{ex:b1:newton-dynamics:terminal}
कोई पैराशूट-यात्री, $m = \qty{80}{kg}$, पेट के \omterm{def:b1:newton-dynamics:momentum}{बल}, हवा में
$C_xS \approx \qty{0.8}{m^2}$ ($\rho = \qty{1.2}{kg/m^3}$): $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$, जो ($90\%$ तक) लगभग $1.5\tau =
\qty{6}{s}$ में आ जाता है। कोहरे की कोई बूँद, $r = \qty{10}{\micro m}$,
स्टोक्स की व्यवस्था में ($\eta_{\text{हवा}} = \qty{1.8e-5}{Pa.s}$): $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- और इसीलिए कोहरा हवा में टँगा रहता है।
\end{example}

\section{वृत्त पर गति: प्रतिबंध-बल}

\begin{proposition}[अभिकेंद्र की माँग]\label{prop:b1:newton-dynamics:centripetal}
\index{अभिकेंद्र बल}
$R$ त्रिज्या के वृत्त पर $v$ चाल से चलते किसी कण को बलों के परिणामी से
केंद्र की ओर $mv^2/R$ घटक मिलना ही चाहिए। जो भी उसे देता है --- किसी डोरी
का तनाव, घर्षण, किसी बंकित सड़क की प्रतिक्रिया, गुरुत्व --- वही अधिकतम चाल
या न्यूनतम चाल तय कर देता है जिस पर वह गति संभव है।
\end{proposition}

\begin{proof}
फ्रेने त्वरण के साथ न्यूटन का दूसरा नियम: $\sum\vect F$ का अभिलंब घटक
$mv^2/\rho$ के बराबर है।
\end{proof}

\begin{example}[तीन वृत्त]\label{ex:b1:newton-dynamics:circles}
\emph{समतल मोड़}: घर्षण को $mv^2/R \leq \mu_sN = \mu_smg$ देना पड़ता है,
अतः $v \leq \sqrt{\mu_sgR}$: $R = \qty{50}{m}$, $\mu_s = 0.8$ के लिए
\qty{20}{m/s}। $\beta$ कोण पर \emph{बंकित मोड़}, बिना घर्षण के: प्रतिक्रिया
$\vect N$ सड़क पर अभिलंब है; ऊर्ध्वाधर दिशा में $N\cos\beta = mg$, और क्षैतिज
दिशा में $N\sin\beta = mv^2/R$, अतः $v^2 = gR\tan\beta$ --- यही वह एक चाल है
जिस पर मोड़ बिना किसी घर्षण के लिया जा सकता है। \emph{पाश}: शिखर पर, जहाँ
$\vect N$ नीचे केंद्र की ओर इंगित करता है, $N + mg =
mv^2/R$, और $N \geq 0$ के लिए $v \geq \sqrt{gR}$ चाहिए: उससे कम पर डिब्बी
पटरी छोड़ देती है (\cref{pb:b1:point-kinematics:1} फिर से, अब समझाई हुई)।
\end{example}

\section{अभ्यास}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:9}
शंक्वाकार लोलक: $\ell = \qty{1.0}{m}$ लंबाई की डोरी पर लटका कोई \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान}
क्षैतिज वृत्त बनाता है, और डोरी ऊर्ध्वाधर से $30^\circ$ बनाती है। \omterm{cor:b1:point-kinematics:circular}{कोणीय
वेग}, आवर्तकाल और तनाव ($mg$ के मात्रकों में) निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:10}
\qty{1000}{kg} की कोई कार अचर त्वरण के साथ विराम से \qty{8.0}{s} में
\qty{100}{km/h} तक पहुँचती है। परिणामी \omterm{def:b1:newton-dynamics:momentum}{बल} निकालिए, और चालक-टायरों तथा सड़क
के बीच न्यूनतम \omterm{prop:b1:newton-dynamics:friction}{घर्षण गुणांक} भी, यदि चालक-पहिये पूरा भार उठाते हों। गीली
सड़क पर किसी खेल-कार को क्या सीमित करता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:11}
कोई पिंड रैखिक कर्षण के साथ विराम से गिरता है, $\tau = m/\alpha = \qty{0.50}{s}$।
$v(t)$ लिखिए और गिरी हुई दूरी $z(t)$ व्युत्पन्न कीजिए; $v_\infty$ निकालिए,
और \qty{2.0}{s} के बाद $z$ भी; मुक्त पतन से तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:12}
$m$ \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान} की कोई डिब्बी $R$ त्रिज्या के किसी ऊर्ध्वाधर वृत्तीय पाश के
भीतरी ओर दौड़ती है (\cref{pb:b1:point-kinematics:1})। पटरी की \omterm{prop:b1:newton-dynamics:friction}{अभिलंब
प्रतिक्रिया} को तल से नापे कोण $\theta$ और चाल $v(\theta)$ के फलन के रूप में
व्यक्त कीजिए; उससे शिखर पर न्यूनतम चाल निकालिए;
$v^2 = v_0^2 - 2gR(1 - \cos\theta)$ का प्रयोग करके न्यूनतम प्रवेश-चाल और
उस स्थिति में तल पर प्रतिक्रिया निकालिए।
\end{exercise}

\section{समस्या: पैराशूट-यात्री}

\begin{problem}[{सप्ताहांत समस्या --- विमान के दरवाज़े से भूमि तक: मुक्त
पतन, पीछे धकेलती हवा, बहुत तेज़ी से खुलता छत्र, और उतराई --- गति का एक
नियम, कर्षण के दो नियम}]
\label{pb:b1:newton-dynamics:1}
साज़-सामान समेत किसी पैराशूट-यात्री का \omterm{def:b1:newton-dynamics:momentum}{द्रव्यमान} $m = \qty{80}{kg}$ है।
हवा: $\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$। पेट के \omterm{def:b1:newton-dynamics:momentum}{बल}
गिरते हुए $C_xS = \qty{0.80}{m^2}$; खुले छत्र के नीचे $C_xS =
\qty{25}{m^2}$। $g = \qty{9.81}{m/s^2}$; $z$ को छलाँग-बिंदु से नीचे की ओर
गिना जाता है।

\medskip
\noindent\textbf{भाग I --- हवा के बिना।}
\begin{enumerate}
  \item न्यूटन का दूसरा नियम लिखिए और $v(t)$ तथा $z(t)$ दीजिए; दोनों को
        $t = \qty{10}{s}$ पर निकालिए।
  \item भार निकालिए।
  \item इस चरण में पैराशूट-यात्री स्वयं को ``भारहीन'' अनुभव करता है, यद्यपि
        गुरुत्व पूरी तरह काम कर रहा होता है। बलों की सहायता से समझाइए (भूमि
        पर खड़े होने की तुलना में क्या नहीं है)।
  \item क्या इस समस्या के लिए पृथ्वी का तंत्र पर्याप्त रूप से जड़त्वीय है?
        जो उपेक्षित है उसे नाम दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हवा में से गिरते हुए।}
\begin{enumerate}[resume]
  \item पेट के \omterm{def:b1:newton-dynamics:momentum}{बल} गिरने के लिए कर्षण \omterm{def:b1:newton-dynamics:momentum}{बल} को $v$ के फलन के रूप में लिखिए।
  \item $v(t)$ के लिए गति का समीकरण लिखिए।
  \item \omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग} $v_\infty$ निकालिए (\unit{m/s} और \unit{km/h} में)।
  \item $u = v/v_\infty$ तथा $\tau = v_\infty/g$ के साथ दिखाइए कि
        $\dd u/\dd t = (1 - u^2)/\tau$; और $\tau$ निकालिए।
  \item चरों को पृथक करके हल कीजिए और दिखाइए कि
        $v = v_\infty\tanh(t/\tau)$।
  \item कितनी देर बाद $v = 0.9v_\infty$ होता है? और $0.99v_\infty$?
  \item दिखाइए कि $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ और
        $t = \qty{10}{s}$ पर गिरी हुई दूरी निकालिए; भाग I से तुलना कीजिए।
  \item $v_\infty$ पर रेनॉल्ड्स संख्या का आकलन $L = \qty{0.5}{m}$ के साथ
        कीजिए और द्विघाती नियम को उचित ठहराइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- छत्र खुलता है।}
\begin{enumerate}[resume]
  \item खुले छत्र के नीचे नया \omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग} निकालिए।
  \item यदि छत्र $v = \qty{40}{m/s}$ पर तत्क्षण खुल जाता, तो मंदन कितना
        होता, \unit{m/s^2} में और $g$ में? यह अस्वीकार्य क्यों है?
  \item असली छत्र लगभग \qty{3}{s} में खुलते हैं। औसत मंदन और
        पैराशूट-यात्री पर बंधन-पट्टे के औसत \omterm{def:b1:newton-dynamics:momentum}{बल} का आकलन कीजिए।
  \item छत्र के पूरी तरह खुल जाने पर चाल नए $v_\infty$ की ओर क्षीण होती है:
        नया \omterm{def:b1:transient-regimes:firstorder}{कालांक} $\tau' = v_\infty'/g$ क्या है, और स्थायी अवतरण तक
        पहुँचने के लिए उसका क्या अर्थ है?
  \item स्थायी अवतरण में बंधन-पट्टे का \omterm{def:b1:newton-dynamics:momentum}{बल} क्या है? दूसरे नियम से उचित
        ठहराइए।
  \item समझाइए कि \omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग} $1/\sqrt{C_xS}$ की तरह क्यों बदलता है: आधे
        क्षेत्रफल वाला छत्र उसे कितने गुना बढ़ा देगा?
  \item उसी छत्र के नीचे कोई भारी कूदने वाला (\qty{100}{kg}): \omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग}?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- उतराई।}
\begin{enumerate}[resume]
  \item उतराई की चाल छत्र का $v_\infty$ है। हवा के बिना, उसी चाल से भूमि
        पर पहुँचने के लिए कितनी ऊँचाई से कूदना पड़ता?
  \item अकड़ी टाँगों वाली उतराई शरीर को \qty{0.5}{m} में रोक देती है; और
        लुढ़ककर उतरना \qty{1.5}{m} में। दोनों प्रसंगों में औसत मंदन ($g$ में)
        और टाँगों पर औसत \omterm{def:b1:newton-dynamics:momentum}{बल} का आकलन कीजिए, और निष्कर्ष निकालिए।
  \item भूमि छूने से ठीक पहले पैराशूट-यात्री दोनों डोरियाँ खींचता है, जिससे
        छत्र का कर्षण थोड़ी देर के लिए बढ़ जाता है। चाल पर पड़ते प्रभाव को
        गुणात्मक रूप से समझाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- छोटी चीज़ें अलग तरह से गिरती हैं।}
\begin{enumerate}[resume]
  \item \qty{1.0}{mm} त्रिज्या की कोई वर्षा-बूँद ($C_x = 0.5$): द्विघाती
        नियम से उसका \omterm{prop:b1:newton-dynamics:terminal}{सीमांत वेग} निकालिए, और रेनॉल्ड्स संख्या भी जाँचिए।
  \item \qty{10}{\micro m} त्रिज्या की कोहरे की कोई बूँद: $v_\infty$ के
        लिए \omterm{prop:b1:newton-dynamics:drag}{स्टोक्स का नियम} लगाइए और जाँचिए कि रेनॉल्ड्स संख्या सचमुच छोटी
        है।
  \item सार दीजिए: गति का एक समीकरण, कर्षण की दो व्यवस्थाएँ, और वे तीन
        संख्याएँ (\unit{m/s}) जो तय करती हैं कि कोई चीज़ पत्थर की तरह
        गिरेगी, पत्ते की तरह गिरेगी, या गिरेगी ही नहीं।
\end{enumerate}
\end{problem}
