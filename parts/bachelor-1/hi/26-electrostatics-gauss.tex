\chapter{स्थिरवैद्युतिकी: क्षेत्र और गाउस का नियम}\label{ch:b1:electrostatics-gauss}

किसी स्वेटर पर गुब्बारा रगड़िए और वह दीवार से चिपक जाता है; सूखे बालों में
कंघी कीजिए और वह पानी की पतली धार को मोड़ देती है। दोनों के पीछे विराम में
पड़े विद्युत आवेशों के बीच का बल है, यानी \omterm{thm:b1:electrostatics-gauss:coulomb}{कूलॉम का नियम} --- गुरुत्व की तरह
कोई व्युत्क्रम-वर्ग नियम, पर दो प्रोटॉनों के बीच उससे $10^{36}$ गुना प्रबल
और दोनों चिह्नों वाला, जिससे पदार्थ अद्भुत यथार्थता तक उदासीन रहता है और
जो थोड़ा-बहुत बच रहता है उसी को हम रसायन, पदार्थ-विज्ञान और तड़ित कहते हैं।
यह अध्याय वह संकल्पना गढ़ता है जो इस नियम को सँभालने लायक बना देती है,
यानी \omterm{def:b1:electrostatics-gauss:field}{विद्युत क्षेत्र}, दिखाता है कि किसी क्षेत्र को उसकी सममितियों से कैसे
पढ़ा जाए, और वह एक प्रमेय व्युत्पन्न करता है जो सममित वितरणों के क्षेत्र
तीन पंक्तियों में निकाल देती है: \omterm{thm:b1:electrostatics-gauss:gauss}{गाउस का नियम}। और उसका गुरुत्वीय जुड़वाँ
अंत में मुफ़्त मिल जाता है।

\section{आवेश और कूलॉम का नियम}

\begin{definition}[विद्युत आवेश]\label{def:b1:electrostatics-gauss:charge}
\index{विद्युत आवेश}\index{मूल आवेश}\index{आवेश घनत्व}
\emph{विद्युत आवेश} $q$ (कूलॉम, \unit{C}) पदार्थ का एक गुण है, जो दोनों
चिह्नों का हो सकता है; वह संरक्षित रहता है (किसी विलगित निकाय का कुल आवेश
कभी नहीं बदलता) और \emph{मूल आवेश}
$e = \qty{1.602e-19}{C}$ के मात्रकों में क्वांटित है (प्रोटॉन $+e$,
इलेक्ट्रॉन $-e$)। किसी स्थूल वितरण का वर्णन किसी \emph{आवेश घनत्व} से
होता है: रैखिक $\lambda$
(\unit{C/m}), पृष्ठीय $\sigma$ (\unit{C/m^2}) या आयतनी $\rho$
(\unit{C/m^3}), जिससे कोई छोटा अवयव $\dd q = \lambda\,\dd l$,
$\sigma\,\dd S$ या $\rho\,\dd\tau$ लिए रहता है।
\end{definition}

\begin{theorem}[कूलॉम का नियम]\label{thm:b1:electrostatics-gauss:coulomb}
\index{कूलॉम का नियम}\index{निर्वात की विद्युतशीलता}
निर्वात में विराम में पड़े दो बिंदु आवेश, $q_1$ जो $M_1$ पर है और $q_2$ जो
$M_2$ पर, एक-दूसरे पर ये बल लगाते हैं
\[
\vect F_{1 \to 2} = \frac{q_1q_2}{4\pi\varepsilon_0\,r^2}\,\vect e_{12} = -\vect F_{2 \to 1},
\qquad r = M_1M_2,\quad \vect e_{12} = \frac{\vect{M_1M_2}}{r},
\]
जो समान आवेशों के लिए प्रतिकर्षी और असमान के लिए आकर्षी हैं, जहाँ
\emph{\omterm{thm:b1:electrostatics-gauss:coulomb}{निर्वात की विद्युतशीलता}} $\varepsilon_0 = \qty{8.854e-12}{F/m}$ है, अर्थात्
$1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$। कई आवेशों के बल जुड़ जाते हैं
(अध्यारोपण)।
\end{theorem}
\admitted

\begin{example}[कितना प्रबल]\label{ex:b1:electrostatics-gauss:strong}
\qty{1}{fm} की दूरी पर दो प्रोटॉन $8.99 \times 10^9 \times (1.6 \times
10^{-19})^2/10^{-30} = \qty{230}{N}$ से प्रतिकर्षित होते हैं --- यानी किसी
बच्चे का भार, $10^{-27}$~किलोग्राम के दो कणों के बीच; और उनका गुरुत्वीय
आकर्षण $10^{36}$ गुना छोटा है। किसी कमरे के एक सिरे पर दो ग्राम
इलेक्ट्रॉन और दूसरे सिरे पर उनके जोड़ के प्रोटॉन $10^{25}$~न्यूटन से
आकर्षित होते। पदार्थ उदासीन इसलिए है कि उसके पास कोई चारा नहीं: कोई भी
असंतुलन ऐसे बलों से सुधार दिया जाता है जिनका सामना कोई संरचना नहीं कर
सकती।
\end{example}

\section{विद्युत क्षेत्र}

\begin{definition}[स्थिरवैद्युत क्षेत्र]\label{def:b1:electrostatics-gauss:field}
\index{विद्युत क्षेत्र}\index{क्षेत्र-रेखा}
विराम में पड़े आवेशों का कोई वितरण आकाश के हर बिंदु $M$ पर एक
\emph{विद्युत क्षेत्र} $\vect E(M)$ बनाता है, जो उस बल से परिभाषित होता है
जो वह किसी परीक्षण आवेश $q$ पर लगाता है, जब वह $M$ पर रखा हो: $\vect F = q\vect E(M)$ (\unit{V/m} $=$
\unit{N/C})। कोई बिंदु आवेश $q_1$, जो $O$ पर हो, यह बनाता है
\[
\vect E(M) = \frac{q_1}{4\pi\varepsilon_0 r^2}\,\vect e_r , \qquad r = OM ,
\]
जो त्रिज्य है, और $q_1 > 0$ के लिए बाहर की ओर; और कोई वितरण अपने अवयवों के
क्षेत्रों का सदिश योग बनाता है। कोई \emph{क्षेत्र-रेखा} ऐसा वक्र है जो हर
बिंदु पर $\vect E$ की स्पर्शरेखा हो और उसी के अनुदिश दिशित हो।
\end{definition}

\begin{proposition}[किसी क्षेत्र-चित्र को पढ़ना]\label{prop:b1:electrostatics-gauss:lines}
क्षेत्र-रेखाएँ धनात्मक आवेशों से निकलती हैं और ऋणात्मक पर समाप्त होती हैं
(या अनंत पर); वे कभी नहीं कटतीं (क्षेत्र का एक ही मान होता है), और जहाँ
क्षेत्र प्रबल है वहाँ वे सट जाती हैं। जहाँ दो आवेश हों, वहाँ हर एक से खींची
गई रेखाओं की संख्या उसके आवेश के समानुपाती होती है।
\end{proposition}

\begin{example}[वलय और खंड, सीधे योग से]\label{ex:b1:electrostatics-gauss:ring}
$R$ त्रिज्या के किसी वलय की अक्ष पर, जिस पर $Q$ एकसमान फैला हो, उसके केंद्र
से $z$ दूरी पर, हर अवयव $\dd q$ का योगदान $\dd q/4\pi\varepsilon_0(R^2 +
z^2)$ है, उस रेखा के अनुदिश जो उसे $M$ से जोड़ती है; अक्ष पर लंब घटक जोड़ों में कट जाते हैं और अक्षीय
घटक गुणक
$z/\sqrt{R^2 + z^2}$ के साथ जुड़ जाते हैं:
\[
E_z = \frac{Qz}{4\pi\varepsilon_0(R^2 + z^2)^{3/2}} ,
\]
जो केंद्र पर शून्य है, $z = R/\sqrt2$ पर अधिकतम, और दूर जाकर $Q/4\pi\varepsilon_0z^2$।
एकसमान आवेशित कोई चकती वलयों का ढेर है (\cref{pb:b1:electrostatics-gauss:1})।
सीधा योग तभी तक चलता है जब तक ज्यामिति समाकल करने दे; और शेष अध्याय उसी से
बचने के बारे में है।
\end{example}

\section{सममितियाँ और अपरिवर्तिताएँ}

\begin{proposition}[सममिति का सिद्धांत]\label{prop:b1:electrostatics-gauss:symmetry}
\index{सममिति (किसी क्षेत्र की)}\index{सममिति तल}\index{प्रतिसममिति तल}
क्षेत्र में वही सममितियाँ होती हैं जो उसके स्रोतों में। यदि कोई तल $\Pi$
आवेश-वितरण का \emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल}} हो (यानी वितरण का दर्पण-प्रतिबिंब वही
वितरण हो), तो $\Pi$ के हर बिंदु पर क्षेत्र $\Pi$ \emph{में} ही रहता है; और
यदि $\Pi$ कोई \emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{प्रतिसममिति तल}} हो (यानी दर्पण-प्रतिबिंब वही वितरण हो
पर सारे आवेश उलटे हों), तो $\Pi$ के बिंदुओं पर क्षेत्र $\Pi$ पर
\emph{लंब} होता है। यदि वितरण किसी स्थानांतरण या घूर्णन के अधीन अपरिवर्तित
रहे, तो क्षेत्र भी: उसके घटक उस निर्देशांक पर निर्भर नहीं करते।
\end{proposition}

\begin{proof}
$\vect E$ पदों $\dd q\,\vect e/4\pi\varepsilon_0r^2$ का योग है; योग का
दर्पण-प्रतिबिंब प्रतिबिंबों का योग है, और किसी वितरण के क्षेत्र का
प्रतिबिंब प्रतिबिंब-वितरण का क्षेत्र है (स्थितियों और आवेशों से बना कोई
सदिश स्थितियों की तरह रूपांतरित होता है)। किसी \omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल} पर क्षेत्र अपने ही
दर्पण-प्रतिबिंब के बराबर होता है, अतः तल में ही रहता है; और किसी
\omterm{prop:b1:electrostatics-gauss:symmetry}{प्रतिसममिति तल} पर वह अपने प्रतिबिंब के विपरीत के बराबर होता है, अतः उस पर
अभिलंब होता है। अपरिवर्तिता के लिए वही तर्क किसी स्थानांतरण या घूर्णन के
साथ चलता है।
\end{proof}

\begin{method}[क्षेत्र निकालने से पहले उसका रूप पता करना]\label{meth:b1:electrostatics-gauss:symmetry}
जिस बिंदु $M$ पर क्षेत्र चाहिए, उसके लिए: (1) स्रोतों के उन सममिति तलों की
सूची बनाइए जिनमें $M$ हो --- क्षेत्र उनके प्रतिच्छेदन पर होगा; और ऐसे दो तल
दिशा तय कर देते हैं। (2) अपरिवर्तिताओं की सूची बनाइए --- वे बताती हैं कि
$\vect E$ किन निर्देशांकों पर निर्भर करता है। इस तरह एकसमान आवेशित कोई
अनंत तार बेलनाकार निर्देशांकों में $\vect E = E(r)\,\vect e_r$ देता है, कोई
गोला गोलीय निर्देशांकों में $\vect E = E(r)\,\vect e_r$, और कोई अनंत तल
$\vect E = E(z)\,\vect e_z$, जहाँ $E(-z) = -E(z)$। तभी गणना कीजिए।
\end{method}

हर एक पर क्षेत्र या तो अभिलंब और एक ही परिमाण का है, या स्पर्शरेखीय।}

\section{अभिवाह और गाउस का नियम}

\begin{definition}[क्षेत्र का अभिवाह]\label{def:b1:electrostatics-gauss:flux}
\index{विद्युत क्षेत्र का अभिवाह}
किसी पृष्ठ $S$ को $\dd S$ क्षेत्रफल के छोटे अवयवों में काटिए, जिनमें से हर
एक का एक एकांक अभिलंब $\vect n$ हो (किसी बंद पृष्ठ के लिए, बाहर की ओर का
अभिलंब)। $\vect E$ का \emph{अभिवाह}, $S$ से होकर, यह योग है
\[
\Phi = \sum \vect E\cdot\vect n\,\dd S = \sum E_n\,\dd S \qquad (\unit{V.m}):
\]
यानी क्षेत्रफल से भारित क्षेत्र का अभिलंब घटक --- अर्थात् $S$ को पार करती
``क्षेत्र-रेखाओं की संख्या'', जो $\vect n$ के अनुदिश पार करने पर धनात्मक
गिनी जाती है।
\end{definition}

\begin{theorem}[गाउस का नियम]\label{thm:b1:electrostatics-gauss:gauss}
\index{गाउस का नियम}
किसी भी बंद पृष्ठ $S$ से होकर स्थिरवैद्युत क्षेत्र का \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} $S$ के भीतर
घिरे कुल आवेश के बराबर होता है, $\varepsilon_0$ से भाग देकर:
\[
\Phi_{\text{बाहर}} = \frac{Q_{\text{भीतर}}}{\varepsilon_0} .
\]
$S$ के बाहर के आवेश \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} में कुछ भी नहीं जोड़ते (उनकी रेखाएँ भीतर जाती
हैं और फिर बाहर आ जाती हैं)।
\end{theorem}

\begin{proof}
किसी बिंदु आवेश $q$ के लिए, जो $r$ त्रिज्या के किसी गोले के केंद्र पर हो:
$\vect E$ त्रिज्य है और गोले पर हर जगह $q/4\pi\varepsilon_0r^2$ परिमाण का, अतः
$\Phi = E \times 4\pi r^2 = q/\varepsilon_0$ --- जो $r$ से स्वतंत्र है,
क्योंकि क्षेत्र $1/r^2$ की तरह ठीक उतना ही गिरता है जितना क्षेत्रफल बढ़ता
है। यह कि वही बात $q$ के चारों ओर के किसी भी बंद पृष्ठ के लिए सही है (किसी
पृष्ठ-अवयव से होकर \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} केवल उस घन कोण पर निर्भर करता है जो वह $q$ से
बनाता है), कि बाहर का कोई आवेश शून्य देता है (वह हर घन कोण को दो बार, और
विपरीत चिह्नों से, बनाता है), और कि कई आवेशों के \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} जुड़ जाते हैं ---
ये यहाँ स्वीकृत मान लिए गए हैं; पूरी उपपत्ति वर्ष 2 के खंड में है।
\end{proof}

\begin{method}[गाउस का नियम काम में लेना]\label{meth:b1:electrostatics-gauss:gauss}
(1) सममितियों से $\vect E$ का रूप लिखिए (दिशा, और वह चर जिस पर वह निर्भर
है)। (2) बिंदु $M$ से होकर कोई बंद \emph{गाउसीय पृष्ठ} चुनिए, जिस पर
$\vect E$ हर जगह या तो अभिलंब हो और एक ही परिमाण $E(M)$ का, या स्पर्शरेखीय:
तब \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} $E(M)$ गुणा अभिलंब भाग का क्षेत्रफल हो जाता है। (3) भीतर का आवेश
गिनिए। (4) $E(M)$ के लिए हल कीजिए। यह गोलों, अनंत बेलनों और अनंत तलों के लिए
चलता है --- और आगे के औज़ारों के बिना किसी और के लिए नहीं।
\end{method}

\begin{proposition}[तीन चिरपरिचित क्षेत्र]\label{prop:b1:electrostatics-gauss:three}
\index{आवेशित गोले का क्षेत्र}\index{आवेशित तार का क्षेत्र}\index{आवेशित तल का क्षेत्र}
\begin{enumerate}
  \item $R$ त्रिज्या का गोला, जिसके आयतन में $Q$ आवेश एकसमान फैला हो:
        $E(r) = \dfrac{Q}{4\pi\varepsilon_0r^2}$ बाहर ($r > R$) ---
        मानो सारा आवेश केंद्र पर बैठा हो --- और भीतर $E(r) =
        \dfrac{Qr}{4\pi\varepsilon_0R^3}$, जो शून्य से रैखिक रूप से बढ़ता है;
        और $r = R$ पर संतत। यदि आवेश केवल पृष्ठ पर बैठा हो, तो भीतर का
        क्षेत्र शून्य है।
  \item रैखिक आवेश $\lambda$ का अनंत तार: $E(r) =
        \dfrac{\lambda}{2\pi\varepsilon_0 r}$, त्रिज्य।
  \item पृष्ठीय आवेश $\sigma$ का अनंत तल: दोनों ओर $E = \dfrac{\sigma}{2\varepsilon_0}$,
        तल पर अभिलंब, और $\sigma > 0$ के लिए तल से दूर की ओर, तथा
        \emph{दूरी से स्वतंत्र}। तल के पार अभिलंब घटक
        $\sigma/\varepsilon_0$ से उछल जाता है।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $r$ त्रिज्या का गोला: \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} $4\pi r^2E(r)$; और भीतर का आवेश $Q$ यदि
$r > R$, तथा $Q(r/R)^3$ यदि $r < R$। (2) $r$ त्रिज्या और $h$ लंबाई का बेलन:
पार्श्व पृष्ठ से \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} $2\pi rh\,E(r)$, सिरों से कुछ नहीं
(क्षेत्र स्पर्शरेखीय); आवेश $\lambda h$। (3) तल पर टिकी $S$ अनुप्रस्थ काट
की डिबिया: दोनों सिरों से \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} $2SE$ (क्षेत्र $z$ में विषम है, अतः दोनों
धनात्मक गिने जाते हैं), पार्श्व से कुछ नहीं; और आवेश
$\sigma S$।
\end{proof}

\section{गुरुत्वीय सादृश्य}

\begin{proposition}[गुरुत्व के लिए गाउस का नियम]\label{prop:b1:electrostatics-gauss:gravity}
\index{गाउस का नियम!गुरुत्व के लिए}\index{गुरुत्वीय क्षेत्र}
\omterm{thm:b1:newton-dynamics:laws}{न्यूटन के नियम} $\vect F = -Gm_1m_2\,\vect e_{12}/r^2$ का रूप कूलॉम वाले जैसा
ही है, जहाँ $q \to m$ और $1/4\pi\varepsilon_0 \to -G$। अतः \omterm{prop:b1:newton-dynamics:weight}{गुरुत्वीय
क्षेत्र} $\vect g$ (प्रति एकांक द्रव्यमान बल) उन्हीं सममिति-नियमों का पालन
करता है, और
\[
\Phi_{\text{बाहर}}(\vect g) = -4\pi G\,M_{\text{भीतर}} .
\]
विशेष रूप से, गोलीय सममिति वाला कोई पिंड बाहर के बिंदुओं को ऐसे आकर्षित
करता है मानो उसका सारा द्रव्यमान उसके केंद्र पर हो, और $R$ त्रिज्या तथा
$M$ द्रव्यमान के किसी एकसमान गोले के भीतर क्षेत्र $g(r) = GMr/R^3$ है, जो $r$ में रैखिक है।
\end{proposition}

\begin{proof}
\cref{prop:b1:electrostatics-gauss:three} की उपपत्ति में $Q/\varepsilon_0$ की जगह
$-4\pi GM$ रखिए; और ऋण चिह्न कहता है कि \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} भीतर जाता है (गुरुत्व
आकर्षित करता है)।
\end{proof}

\begin{example}[पृथ्वी के भीतर]\label{ex:b1:electrostatics-gauss:earth}
किसी एकसमान पृथ्वी के लिए $g$ सतह की \qty{9.8}{m/s^2} से घटकर केंद्र पर
रैखिक रूप से शून्य हो जाता; और ग्रह के भीतर से गुज़रती किसी सुरंग में गिराया गया
पत्थर $2\pi\sqrt{R/g} =
\qty{84}{min}$ आवर्तकाल से सरल आवर्त दोलन करता --- यानी किसी सतह-छूती
उपग्रह-कक्षा का आवर्तकाल। असली पृथ्वी का क्रोड सघन है, और $g$ वास्तव में
क्रोड--प्रावार सीमा पर \qty{10.7}{m/s^2} तक \emph{चढ़} जाता है, और उसके बाद
गिरता है (\cref{pb:b1:electrostatics-gauss:1})। न्यूटन को अपना नियम ग्रहों
पर लगाने के लिए कोश प्रमेय चाहिए थी; \omterm{thm:b1:electrostatics-gauss:gauss}{गाउस का नियम} उसे एक ही पंक्ति में दे
देता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:1}
\qty{2.0}{fm} की दूरी पर दो प्रोटॉनों के बीच विद्युत बल और गुरुत्वीय बल;
उनका अनुपात। फिर नाभिक टिके कैसे रहते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:2}
\qty{1.0}{\micro C} के किसी बिंदु आवेश से \qty{1.0}{m} पर क्षेत्र। पृथ्वी
की सतह पर नीचे की ओर संकेत करता \qty{100}{V/m} का क्षेत्र है: पृथ्वी को
\qty{6370}{km} त्रिज्या का गोला मानकर उसके आवेश का चिह्न और मान।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:3}
दो आवेश $+q$, $x = \pm a$ पर, $x$ अक्ष पर। $x$ अक्ष पर, $|x| > a$ के लिए,
और $y$ अक्ष पर क्षेत्र; दिखाइए कि $y$ अक्ष पर $E_y$
$y = a/\sqrt2$ पर अधिकतम है। मूलबिंदु के पास की रेखाएँ खींचिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:4}
आवेश $+2q$, $x = 0$ पर, और $-q$, $x = a$ पर। $x$ अक्ष पर क्षेत्र कहाँ लुप्त
होता है? $+2q$ से कितनी रेखाएँ निकलती हैं, $-q$ पर पहुँचती हर एक रेखा के
बदले, और बाकी जाती कहाँ हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:5}
$R$ त्रिज्या की कोई चकती एकसमान $\sigma$ लिए है। वलय वाले परिणाम से
दिखाइए कि उसकी अक्ष पर $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 +
R^2}}\Bigr)$ है, $z > 0$ के लिए; और दोनों सीमाएँ $z \ll R$ तथा $z \gg R$ जाँचिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:6}
सोने का कोई नाभिक ($Z = 79$, $R = \qty{7.0}{fm}$), जिसे एकसमान आवेशित गोला
माना जाए: पृष्ठ पर, $R/2$ पर, और \qty{1}{pm} पर क्षेत्र। हवा के भंजन
क्षेत्र, \qty{3e6}{V/m}, से तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:7}
गाउस के नियम से किसी अनंत तार का क्षेत्र। \qty{1.0}{cm} त्रिज्या का कोई
उच्च-वोल्टता चालक: वह रैखिक आवेश जिस पर उसके पृष्ठ का क्षेत्र भंजन मान
\qty{3e6}{V/m} तक पहुँच जाए (कोरोना का आरंभ)। तब \qty{5}{m} पर
क्षेत्र।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:8}
\qty{1.0}{m^2} क्षेत्रफल की दो समांतर पट्टिकाएँ $\pm\qty{1.0}{\micro C}$ लिए
हैं। उनके बीच क्षेत्र; एक पट्टिका पर बल (उस पर लगने वाला क्षेत्र केवल दूसरी
पट्टिका का है --- क्यों?); और पट्टिकाओं पर दाब।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:9}
समाक्ष तार: $a$ त्रिज्या का भीतरी चालक, जिस पर $+\lambda$, और $b$ त्रिज्या
का बाहरी आवरण, जिस पर $-\lambda$। गाउस के नियम से तीनों क्षेत्रों में
क्षेत्र। $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, $\lambda = \qty{10}{nC/m}$:
भीतरी चालक के पृष्ठ पर क्षेत्र।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:10}
कोई मोटा गोलीय कोश $R_1 < r < R_2$ अपने आयतन में $Q$ एकसमान लिए है। तीनों
क्षेत्रों में क्षेत्र; $R_1$ और $R_2$ पर संततता जाँचिए; वह अधिकतम कहाँ है?
$E(r)$ का आलेख खींचिए। गुहा में रखा कोई धनात्मक आवेश: उसे कौन-सा बल अनुभव
होता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:11}
\emph{पृथ्वी के भीतर गुरुत्व।} क्रोड: त्रिज्या \qty{3480}{km}, घनत्व
\qty{11000}{kg/m^3}; प्रावार \qty{6370}{km} तक, घनत्व \qty{4500}{kg/m^3}।
(क) क्रोड का, प्रावार का और कुल द्रव्यमान (\qty{5.97e24}{kg} से तुलना कीजिए)।
(ख) सतह पर और क्रोड--प्रावार सीमा पर $g$। (ग) दिखाइए कि
प्रावार में गहराई के साथ $g$ तभी बढ़ता है जब प्रावार का घनत्व उसके नीचे
पड़े पदार्थ के माध्य घनत्व के $\tfrac23$ से कम हो, और उसे जाँचिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:12}
\emph{टॉमसन का परमाणु।} हाइड्रोजन परमाणु का प्रतिरूप ऐसे लीजिए: $+e$ आवेश
और $R = \qty{0.10}{nm}$ त्रिज्या का कोई एकसमान धनात्मक गोला, जिसके भीतर
इलेक्ट्रॉन स्वतंत्र हो। (क) केंद्र से $r$ दूरी पर इलेक्ट्रॉन पर बल।
(ख) दिखाइए कि वह सरल आवर्त दोलन करता है, और आवृत्ति तथा उससे निकलने वाले
प्रकाश की तरंगदैर्घ्य निकालिए। (ग) हाइड्रोजन की पराबैंगनी रेखाओं
(\qty{122}{nm} और उससे छोटी) से तुलना कीजिए और टिप्पणी कीजिए कि यह प्रतिरूप
क्या ठीक पकड़ता है और क्या नहीं (\cref{ch:b1:quantum-introduction})।
\end{exercise}

\section{समस्या: विद्युत पृथ्वी}

\begin{problem}[{सप्ताहांत समस्या --- साफ़ मौसम का क्षेत्र, तड़ित-मेघ,
बिजली का आघात, और गुरुत्व पर घुमाया गया गाउस का नियम}]
\label{pb:b1:electrostatics-gauss:1}
आँकड़े: $\varepsilon_0 = \qty{8.85e-12}{F/m}$, $1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$,
$e = \qty{1.6e-19}{C}$, पृथ्वी की त्रिज्या $R_E = \qty{6.37e6}{m}$, सतह
$\qty{5.1e14}{m^2}$, $G = \qty{6.67e-11}{SI}$, $M_E = \qty{5.97e24}{kg}$।

\medskip
\noindent\textbf{भाग I --- साफ़ मौसम का क्षेत्र।}
साफ़ मौसम में भूमि पर क्षेत्र $E_0 = \qty{100}{V/m}$ होता है, जो
\emph{नीचे} की ओर संकेत करता है।
\begin{enumerate}
  \item भूमि पर के किसी मुक्त इलेक्ट्रॉन पर बल की दिशा; और पृथ्वी के
        आवेश का चिह्न।
  \item पृथ्वी को एकसमान आवेशित गोला मानिए: गाउस के नियम से उसका कुल
        आवेश।
  \item भूमि का पृष्ठीय \omterm{def:b1:electrostatics-gauss:charge}{आवेश घनत्व}; और प्रति वर्ग सेंटीमीटर अतिरिक्त
        इलेक्ट्रॉनों की संख्या।
  \item \qty{1}{\micro g} द्रव्यमान के उस धूल-कण पर बल जो $100$ \omterm{def:b1:electrostatics-gauss:charge}{मूल आवेश}
        लिए हो; उसके भार से तुलना कीजिए।
  \item मापों से पता चलता है कि क्षेत्र \qty{10}{km} पर गिरकर लगभग
        \qty{5}{V/m} रह जाता है। भूमि और \qty{10}{km} के बीच
        \qty{1}{m^2} अनुप्रस्थ काट के किसी ऊर्ध्वाधर स्तंभ पर \omterm{thm:b1:electrostatics-gauss:gauss}{गाउस का नियम}
        लगाइए: हवा प्रति वर्ग मीटर जो आवेश रखती है उसका चिह्न और मात्रा।
  \item उस हवा का माध्य आयतनी \omterm{def:b1:electrostatics-gauss:charge}{आवेश घनत्व}, और उससे संगत प्रति घन मीटर मूल
        आवेशों की संख्या। (हवा में हर चिह्न के लगभग $10^9$ आयन प्रति घन
        मीटर होते हैं: यह असंतुलन उसका कितना अंश है?)
  \item हवा की एक छोटी चालकता $\gamma = \qty{2e-14}{S/m}$ होती है, अतः
        नीचे की ओर $\gamma E_0$ धारा-घनत्व बहता है। पूरे भूमंडल पर साफ़
        मौसम की कुल धारा; और पृथ्वी का आवेश उदासीन करने में वह कितना समय
        लेती। निष्कर्ष निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तड़ित-मेघ।}
किसी तूफ़ानी बादल का प्रतिरूप $R = \qty{3}{km}$ त्रिज्या की दो क्षैतिज
चकतियों से लीजिए: \qty{3}{km} ऊँचाई पर $-Q$, \qty{9}{km} पर $+Q$, जहाँ $Q = \qty{40}{C}$।
\begin{enumerate}[resume]
  \item $a$ त्रिज्या के किसी वलय की अक्ष पर, जिस पर $q$ हो, उसके केंद्र
        से $z$ दूरी पर क्षेत्र (सममिति, फिर योग)।
  \item वलयों पर योग करके, $R$ त्रिज्या और पृष्ठीय आवेश $\sigma$ की किसी
        चकती की अक्ष पर क्षेत्र निकालिए:
        $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$।
  \item सीमाएँ $z \ll R$ और $z \gg R$; हर एक की व्याख्या कीजिए।
  \item निचली चकती का पृष्ठीय आवेश; और वह ठीक नीचे भूमि पर जो क्षेत्र
        बनाती है (दिशा और मान)।
  \item उसी बिंदु पर ऊपरी चकती का क्षेत्र; और तूफ़ान के नीचे भूमि पर
        परिणामी क्षेत्र। भूमि एक चालक है और उसके प्रेरित आवेश परिणाम को
        मोटे तौर पर दुगुना कर देते हैं (स्वीकृत): तूफ़ानों के नीचे मापे गए
        \qty{10}{kV/m} से और साफ़ मौसम वाले मान से तुलना कीजिए।
  \item पूरे बादल को घेरते किसी बंद पृष्ठ से होकर क्षेत्र का \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह}; और
        केवल निचली चकती को घेरते किसी पृष्ठ से होकर।
  \item दोनों चकतियों के ठीक बीच (\qty{6}{km}) क्षेत्र, और उसकी दिशा; उस
        ऊँचाई पर हवा के भंजन क्षेत्र, लगभग \qty{1.5e6}{V/m}, से तुलना
        कीजिए।
  \item यदि वे \qty{40}{C} \qty{500}{m} त्रिज्या के किसी गोले में
        केंद्रित होते: उसके पृष्ठ पर क्षेत्र। तड़ित का आरंभ आज भी कोई खुला
        प्रश्न क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- आघात।}
कोई वापसी आघात \qty{3}{km} लंबी किसी नाली से होकर \qty{50}{\micro s} में
\qty{5}{C} भूमि तक ले जाता है; नाली के अनुदिश क्षेत्र
$10^5$~वोल्ट प्रति मीटर लीजिए।
\begin{enumerate}[resume]
  \item माध्य धारा।
  \item हस्तांतरित आवेश पर विद्युत बल का कार्य; आघात के दौरान शक्ति; और
        मानवजाति की औसत शक्ति-खपत, लगभग \qty{2e13}{W}, से तुलना कीजिए।
  \item नाली की त्रिज्या \qty{2}{cm} है: उसमें हवा का द्रव्यमान
        ($\rho = \qty{1.2}{kg/m^3}$), और वह ताप जिस तक वह पहुँचती यदि सारी
        ऊर्जा उसे स्थिर आयतन पर गरम करती ($c_V =
        \qty{720}{J/(kg.K)}$)। असली नाली के ताप \qty{30000}{K} के पास होते
        हैं: बाकी ऊर्जा जाती कहाँ है?
  \item गरज: प्रति किलोमीटर विलंब, और \qty{3}{km} लंबी किसी नाली की ध्वनि
        सेकंडों तक क्यों गड़गड़ाती रहती है।
  \item पूरी दुनिया में हर सेकंड लगभग $50$ बादल-से-भूमि आघात होते हैं: वे
        नीचे की ओर कितनी धारा ले जाते हैं; भाग~I से तुलना कीजिए और
        ``वैश्विक परिपथ'' का चित्र पूरा कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- गुरुत्व पर घुमाया गया \omterm{thm:b1:electrostatics-gauss:gauss}{गाउस का नियम}।}
\begin{enumerate}[resume]
  \item \omterm{prop:b1:newton-dynamics:weight}{गुरुत्वीय क्षेत्र} $\vect g$ के लिए \omterm{thm:b1:electrostatics-gauss:gauss}{गाउस का नियम} लिखिए और किसी
        बिंदु द्रव्यमान की स्थिति से उसका नियतांक न्यायसंगत ठहराइए।
  \item एकसमान पृथ्वी: भीतर $g(r)$; और $r = R_E/2$ पर मान।
  \item असली पृथ्वी, जिसका क्रोड \qty{3480}{km} त्रिज्या और
        \qty{11000}{kg/m^3} घनत्व का है: क्रोड--प्रावार सीमा पर $g$; सतह से
        तुलना कीजिए।
  \item किसी एकसमान पृथ्वी के भीतर से गुज़रती घर्षणरहित सुरंग में गिराया गया
        पत्थर: गति का समीकरण, दोलन का आवर्तकाल, और केंद्र पर चाल।
  \item सार दीजिए: चारों भागों में से हर एक में गाउस के नियम ने किसकी जगह
        ली, और कौन-से चार आँकड़े याद रखने लायक हैं?
\end{enumerate}
\end{problem}
