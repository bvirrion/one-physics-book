\chapter{केंद्रीय बल: ग्रह और उपग्रह}\label{ch:b1:central-forces}

इकतीस उपग्रह दिन में दो बार पृथ्वी का चक्कर लगाते हैं, हर एक एक घड़ी भी है
और एक रेडियो भी, और जो फ़ोन उनमें से चार को सुन लेता है वह कुछ मीटर तक जान
जाता है कि वह कहाँ है। हैली का धूमकेतु अड़तीस वर्ष तक सूर्य की ओर गिरता है,
कुछ ही सप्ताहों में उसका चक्कर काटता है, और फिर अड़तीस वर्ष तक ऊपर चढ़ता
जाता है। मंगल की ओर जाता कोई यान पृथ्वी को ठीक उसी चाल से छोड़ता है जो
उसके पथ को मंगल की कक्षा से बस छूने भर देगी। यह सब किसी ऐसे पिंड की गति है
जो किसी स्थिर बिंदु की ओर ऐसे बल से खिंचता है जो दूरी के व्युत्क्रम वर्ग
की तरह घटता है, और यह सब दो संरक्षित राशियों --- ऊर्जा और \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} ---
तथा एक ही नए विचार, प्रभावी स्थितिज, से निकल आता है। यह अध्याय कक्षाओं के
आकार और चालें, केप्लर के तीनों नियम, और किसी उपग्रह को मनचाही जगह रखने का
अंकगणित व्युत्पन्न करता है।

\section{केंद्रीय संरक्षी बल}

\begin{definition}[केंद्रीय बल; केंद्रीय संरक्षी बल]\label{def:b1:central-forces:central}
\index{केंद्रीय बल}\index{प्रभावी स्थितिज ऊर्जा}
कोई बल $O$ केंद्र वाला \emph{केंद्रीय} बल है यदि वह सदा $\vect{OM}$ के
अनुदिश हो: $\vect F = F(M)\,\vect e_r$। और वह \emph{केंद्रीय संरक्षी} है यदि
उसका परिमाण केवल $r = OM$ पर निर्भर करे: $\vect F =
F(r)\,\vect e_r$, जो किसी \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} $E_p(r)$ से व्युत्पन्न होता है, जहाँ
$F(r) = -\dd E_p/\dd r$।
\end{definition}

\begin{theorem}[केंद्रीय संरक्षी बल के अधीन गति]\label{thm:b1:central-forces:reduction}
\index{क्षेत्रफल का नियम}
केवल किसी \omterm{def:b1:central-forces:central}{केंद्रीय संरक्षी} बल के अधीन $m$ द्रव्यमान का कोई कण:
\begin{enumerate}
  \item अपना \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} $\vect L_O$ बनाए रखता है; उसकी गति तलीय होती है
        और, उस तल में ध्रुवीय निर्देशांकों में, $r^2\dot\theta = C$ अचर
        रहता है (\omterm{cor:b1:angular-momentum:central}{क्षेत्रफल का नियम}, \cref{cor:b1:angular-momentum:central});
  \item अपनी \omterm{thm:b1:work-and-energy:em}{यांत्रिक ऊर्जा} $E_m = \tfrac12 m(\dot r^2 + r^2\dot\theta^2)
        + E_p(r)$ बनाए रखता है, जिसे यों लिखा जा सकता है
        \[
        E_m = \tfrac12 m\dot r^2 + E_{p,\mathrm{eff}}(r), \qquad
        E_{p,\mathrm{eff}}(r) = E_p(r) + \frac{mC^2}{2r^2} = E_p(r) + \frac{L^2}{2mr^2} .
        \]
\end{enumerate}
अतः त्रिज्य गति ठीक उस एक-विमीय कण जैसी है जो \emph{प्रभावी स्थितिज}
$E_{p,\mathrm{eff}}$ में हो: त्रिज्य \omterm{def:b1:work-and-energy:turning}{पलटाव बिंदु}
$E_{p,\mathrm{eff}}(r) = E_m$ के मूल हैं, गति बद्ध है यदि वह उनमें से दो के
बीच फँसी हो, और वृत्तीय है यदि $r$ $E_{p,\mathrm{eff}}$ के किसी निम्निष्ठ
पर बैठा हो।
\end{theorem}

\begin{proof}
भाग 1 तो \cref{cor:b1:angular-momentum:central} ही है। बल संरक्षी है और
कोई दूसरा बल नहीं लगता, अतः $E_m$ संरक्षित है; और उस तल में
$v^2 = \dot r^2 + r^2\dot\theta^2$ तथा $r^2\dot\theta^2 = C^2/r^2$।
पद $mC^2/2r^2$, जो कोणीय गति की \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} है, त्रिज्य समस्या में एक
प्रतिकर्षी \emph{\omterm{prop:b1:central-forces:effective}{अपकेंद्र अवरोध}} की तरह काम करता है, और उस पर
\cref{ch:b1:work-and-energy} का एक-विमीय विश्लेषण लागू होता है।
\end{proof}

\section{न्यूटनी गुरुत्वाकर्षण}

\begin{definition}[सार्वत्रिक गुरुत्वाकर्षण]\label{def:b1:central-forces:gravitation}
\index{गुरुत्वाकर्षण}\index{गुरुत्वीय स्थितिज ऊर्जा}
कोई बिंदु-द्रव्यमान $M$, जो $O$ पर है, बिंदु-द्रव्यमान $m$ को, जो $M$ पर है, इस बल से आकर्षित करता है
\[
\vect F = -\frac{GMm}{r^2}\,\vect e_r, \qquad E_p = -\frac{GMm}{r} \quad (E_p \to 0 \text{ अनंत पर}),
\]
जहाँ $G = \qty{6.67e-11}{N.m^2/kg^2}$। कोई गोलीय सममित पिंड अपने बाहर ऐसे
आकर्षित करता है मानो उसका सारा द्रव्यमान उसके केंद्र पर हो (स्वीकृत; यह
गाउस की प्रमेय का परिणाम है, \cref{ch:b1:electrostatics-gauss})।
\end{definition}

\begin{proposition}[गुरुत्वाकर्षण की प्रभावी स्थितिज]\label{prop:b1:central-forces:effective}
\index{अपकेंद्र अवरोध}\index{वृत्तीय कक्षा}
$E_p = -GMm/r$ के लिए,
\[
E_{p,\mathrm{eff}}(r) = -\frac{GMm}{r} + \frac{L^2}{2mr^2}
\]
यह $+\infty$ की ओर जाता है जब $r \to 0$ ($L \neq 0$ के लिए), और अनंत पर
$0^-$ की ओर; तथा उसका एक निम्निष्ठ $r_c = L^2/GMm^2$ पर है, जिसका मान
$-G^2M^2m^3/2L^2$ है। अतः: $E_m < 0$ --- किसी न्यूनतम और किसी अधिकतम
त्रिज्या (उपभू और अपभू) के बीच \omterm{def:b1:work-and-energy:turning}{बद्ध गति}; $E_m \geq 0$ --- अबद्ध, पिंड अनंत
से आता है और अनंत को लौट जाता है; और $E_m$ उस निम्निष्ठ के बराबर --- $r_c$
त्रिज्या की \omterm{prop:b1:central-forces:effective}{वृत्तीय कक्षा}।
\end{proposition}

\begin{proof}
सीमाएँ देखने भर से; $E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ का हल
$r_c$ है; अब प्रतिस्थापित कीजिए। वृत्तीय प्रसंग में सदा $\dot r = 0$ रहता
है, जो केवल निम्निष्ठ पर संभव है।
\end{proof}

\begin{theorem}[वृत्तीय कक्षाएँ]\label{thm:b1:central-forces:circular}
\index{कक्षीय चाल}\index{केप्लर का तीसरा नियम}
$r$ त्रिज्या की उस \omterm{prop:b1:central-forces:effective}{वृत्तीय कक्षा} पर, जो किसी द्रव्यमान $M$ के परितः है:
\[
v = \sqrt{\frac{GM}{r}}, \qquad
T = 2\pi\sqrt{\frac{r^3}{GM}}, \qquad
E_m = -\frac{GMm}{2r} = -E_k = \tfrac12 E_p .
\]
विशेष रूप से $T^2/r^3 = 4\pi^2/GM$ एक ही केंद्र के परितः सभी वृत्तीय
कक्षाओं के लिए एक-सा होता है (\omterm{thm:b1:central-forces:circular}{केप्लर का तीसरा नियम})।
\end{theorem}

\begin{proof}
एकसमान \omterm{cor:b1:point-kinematics:circular}{वृत्तीय गति} के लिए \omterm{prop:b1:newton-dynamics:centripetal}{अभिकेंद्र बल} चाहिए, $mv^2/r = GMm/r^2$;
$T = 2\pi r/v$; $E_k = \tfrac12 mv^2 = GMm/2r$ और $E_p = -GMm/r$।
\end{proof}

\begin{example}[तीन कक्षाएँ]\label{ex:b1:central-forces:orbits}
$GM_\oplus = \qty{3.99e14}{m^3/s^2}$। नीची कक्षा ($r = \qty{6770}{km}$,
यानी \qty{400}{km} ऊपर): $v = \qty{7.7}{km/s}$, $T = \qty{92}{min}$। भूस्थिर
($T = \qty{86164}{s}$, एक नाक्षत्र दिवस): $r = (GMT^2/4\pi^2)^{1/3} =
\qty{42200}{km}$, $v = \qty{3.1}{km/s}$। चंद्रमा ($r = \qty{384000}{km}$):
$T = \qty{27.4}{d}$ --- और उसका \omterm{cor:b1:point-kinematics:circular}{अभिकेंद्र त्वरण} $v^2/r =
\qty{2.7e-3}{m/s^2}$ साठ पृथ्वी-त्रिज्याओं पर $g/3600$ है: व्युत्क्रम
वर्ग, जिसे स्वयं न्यूटन ने जाँचा था।
\end{example}

\section{केप्लर के नियम और कक्षाओं का आकार}

\begin{theorem}[न्यूटनी क्षेत्र में गतिपथ]\label{thm:b1:central-forces:conics}
\index{केप्लर के नियम}\index{शांकव}\index{दीर्घवृत्त (कक्षा)}\index{उत्केंद्रता}
किसी स्थिर द्रव्यमान $M$ के क्षेत्र में किसी पिंड का \omterm{def:b1:point-kinematics:frame}{गतिपथ} कोई शांकव होता
है, जिसकी एक नाभि पर $O$ है: $E_m < 0$ हो तो दीर्घवृत्त, $E_m = 0$ हो तो
परवलय, और $E_m > 0$ हो तो अतिपरवलय। $a$ अर्ध-दीर्घ अक्ष और $e$ उत्केंद्रता
वाले दीर्घवृत्त के लिए:
\begin{itemize}
  \item (केप्लर I) बल का केंद्र किसी नाभि पर होता है; उपभू और अपभू
        $r_p = a(1 - e)$ और $r_a = a(1 + e)$ पर हैं;
  \item (केप्लर II) \omterm{cor:b1:angular-momentum:central}{क्षेत्रीय वेग} अचर है (\omterm{cor:b1:angular-momentum:central}{क्षेत्रफल का नियम});
  \item (केप्लर III) आवर्तकाल $T^2 = 4\pi^2a^3/GM$ मानता है;
  \item ऊर्जा केवल $a$ पर निर्भर है, $E_m = -GMm/2a$, जिससे दूरी $r$ पर
        चाल यह मिलती है
        \[
        v^2 = GM\left(\frac{2}{r} - \frac{1}{a}\right) .
        \]
\end{itemize}
\end{theorem}
\admitted

\begin{remark}[क्या सिद्ध है और क्या स्वीकृत]\label{rem:b1:central-forces:admitted}
केप्लर II और ऊर्जा के आधार पर किया गया वर्गीकरण ऊपर सिद्ध हो चुका है। यह
कि बद्ध कक्षाएँ ठीक-ठीक दीर्घवृत्त हैं जिनकी एक नाभि पर $O$ है, और साथ ही
व्यापक रूप में केप्लर III तथा $E_m = -GMm/2a$ --- इसके लिए त्रिज्य समीकरण
का समाकलन चाहिए (वर्ष 2 का खंड वह करता है); वृत्तीय कक्षाओं के लिए
($a = r$, $e = 0$) हर कथन \cref{thm:b1:central-forces:circular} पर सिमट आता
है, और यही वह जाँच है जो यहाँ की जा सकती है। संबंध $v^2 = GM(2/r - 1/a)$
उस ऊर्जा-सूत्र को मान लेने पर
$E_m = \tfrac12 mv^2 - GMm/r = -GMm/2a$ से निकल आता है।
\end{remark}

\begin{example}[हैली का धूमकेतु]\label{ex:b1:central-forces:halley}
उपसौर \qty{0.586}{AU}, अपसौर \qty{35.1}{AU}: $a = \qty{17.8}{AU}$,
$e = 0.967$, $T = \sqrt{17.8^3} = 75$ वर्ष (खगोलीय मात्रकों और
वर्षों में केप्लर III, जहाँ सूर्य के लिए $T^2 = a^3$)।
$GM_\odot = \qty{1.33e20}{m^3/s^2}$ के साथ उपसौर पर चाल
$\sqrt{GM(2/r_p - 1/a)} = \qty{54}{km/s}$ है और अपसौर पर \qty{0.9}{km/s}
--- साठ गुना धीमी, ठीक जैसा \omterm{cor:b1:angular-momentum:central}{क्षेत्रफल का नियम} माँगता है ($r_pv_p = r_av_a$)।
\end{example}

\section{उपग्रह}

\begin{proposition}[पलायन वेग; भूस्थिर कक्षा]\label{prop:b1:central-forces:escape}
\index{पलायन वेग}\index{भूस्थिर कक्षा}
$M$ द्रव्यमान और $R$ त्रिज्या के किसी पिंड के पृष्ठ से अनंत तक पहुँचने के
लिए न्यूनतम प्रक्षेप-चाल $v_e = \sqrt{2GM/R} = \sqrt2\,v_c$ है, जहाँ
$v_c = \sqrt{GM/R}$ भूमि-तल पर वृत्तीय चाल है: पृथ्वी के लिए
\qty{11.2}{km/s}। कोई \emph{भूस्थिर} उपग्रह भूमध्य-रेखीय तल में पृथ्वी के
नाक्षत्र आवर्तकाल से परिक्रमा करता है, $r = \qty{42200}{km}$ पर (भूमि से
\qty{35800}{km} ऊपर)।
\end{proposition}

\begin{proof}
$E_m \geq 0$: $\tfrac12 mv_e^2 - GMm/R = 0$। और $T = \qty{86164}{s}$ के साथ केप्लर III।
\end{proof}

\begin{proposition}[होमान अंतरण]\label{prop:b1:central-forces:hohmann}
\index{होमान अंतरण}
किसी उपग्रह को $r_1$ त्रिज्या की \omterm{prop:b1:central-forces:effective}{वृत्तीय कक्षा} से $r_2 > r_1$ त्रिज्या की
समतलीय \omterm{prop:b1:central-forces:effective}{वृत्तीय कक्षा} तक दो छोटे प्रज्वलनों से ले जाने के लिए, $r_1$ पर
इंजन चलाकर उस \emph{अंतरण दीर्घवृत्त} में घुसिए जिसकी उपभू $r_1$ और अपभू
$r_2$ है ($a = (r_1 + r_2)/2$), आधा आवर्तकाल बिना ज़ोर के चलिए, और फिर
$r_2$ पर दुबारा इंजन चलाकर कक्षा को वृत्तीय कीजिए:
\[
\Delta v_1 = \sqrt{\frac{GM}{r_1}}\left(\sqrt{\frac{2r_2}{r_1 + r_2}} - 1\right),
\qquad
\Delta v_2 = \sqrt{\frac{GM}{r_2}}\left(1 - \sqrt{\frac{2r_1}{r_1 + r_2}}\right),
\qquad
t = \pi\sqrt{\frac{(r_1 + r_2)^3}{8GM}} .
\]
\end{proposition}

\begin{proof}
अंतरण दीर्घवृत्त पर चालें $v^2 = GM(2/r - 1/a)$ से $r_1$ और
$r_2$ पर निकालिए; उनमें से वृत्तीय चालें घटाइए; और आधा आवर्तकाल
\cref{thm:b1:central-forces:conics} से।
\end{proof}

\begin{example}[नीची कक्षा से भूस्थिर तक]\label{ex:b1:central-forces:leogeo}
$r_1 = \qty{6770}{km}$, $r_2 = \qty{42200}{km}$: $\Delta v_1 = \qty{2.4}{km/s}$,
$\Delta v_2 = \qty{1.5}{km/s}$, और बिना ज़ोर के \qty{5.3}{h}। कुल
\qty{3.9}{km/s}, यानी नीची कक्षा तक के प्रक्षेपण की लागत का आधा: किसी ऊँची
कक्षा तक की ``आख़िरी मील'' महँगी पड़ती है, और उपग्रह उसके लिए अपना इंजन साथ
लेकर चलते हैं।
\end{example}

\begin{remark}[प्रतिकर्षी केंद्र]\label{rem:b1:central-forces:rutherford}
\index{रदरफ़ोर्ड प्रकीर्णन}
किसी प्रतिकर्षी व्युत्क्रम-वर्ग बल के लिए (दो समान आवेश,
\cref{ch:b1:electrostatics-gauss}) प्रभावी स्थितिज में कोई कूप नहीं होता:
हर \omterm{def:b1:point-kinematics:frame}{गतिपथ} एक अतिपरवलय है, कण पास आता है, विक्षेपित होता है और चला जाता है।
किसी $\alpha$ कण को सोने के किसी नाभिक पर सीधा दागा जाए, तो उसकी निकटतम
पहुँच की दूरी अकेले ऊर्जा से निकल आती है, $E_k =
2Ze^2/4\pi\varepsilon_0 r_{\min}$ --- यही रदरफ़ोर्ड का नाभिक के आकार को बाँधने
का तरीक़ा था (\cref{exo:b1:central-forces:11})।
\end{remark}

विचलित \omterm{prop:b1:central-forces:effective}{वृत्तीय कक्षा} के बारे में क्या कहता है?
\end{exercise}

\end{exercise}

\end{exercise}

नाभिकीय त्रिज्या से कीजिए, जो लगभग \qty{7}{fm} है। नाभिक को छूने के लिए
कितनी ऊर्जा चाहिए होगी?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:12}

खोई \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} का कितना अंश ऊष्मा बनता है?
\end{exercise}

\section{समस्या: वे उपग्रह जो बताते हैं आप कहाँ हैं}

\begin{problem}[{सप्ताहांत समस्या --- बीस हज़ार किलोमीटर ऊपर एक
दिक्चालन-मंडल: उसकी कक्षा, वहाँ तक पहुँचने का ईंधन, कोई फ़ोन कितने उपग्रह
देख सकता है, और उनकी घड़ियों को धीमा चलने के लिए क्यों कहना पड़ता है}]
\label{pb:b1:central-forces:1}
आँकड़े: $GM_\oplus = \qty{3.986e14}{m^3/s^2}$, $R_\oplus = \qty{6371}{km}$,
नाक्षत्र दिवस \qty{86164}{s}, $c = \qty{2.998e8}{m/s}$। ये उपग्रह वृत्तीय
कक्षाओं पर ठीक आधे नाक्षत्र दिवस में पृथ्वी का चक्कर लगाते हैं।

\medskip
\noindent\textbf{भाग I --- कक्षा।}
\begin{enumerate}
  \item कक्षीय त्रिज्या और ऊँचाई निकालिए।
  \item \omterm{thm:b1:central-forces:circular}{कक्षीय चाल} निकालिए।
  \item उपग्रह का त्वरण निकालिए और उसकी तुलना पृष्ठ पर के $g$ से कीजिए।
  \item प्रति किलोग्राम \omterm{thm:b1:work-and-energy:em}{यांत्रिक ऊर्जा} निकालिए, और उसके गतिज तथा स्थितिज
        भाग भी।
  \item आधा सौर दिवस नहीं, आधा नाक्षत्र दिवस क्यों चुना जाता है? भूमि पर
        खड़े किसी प्रेक्षक को क्या दुहराता दिखाई देता है?
  \item किसी उपग्रह से देखने पर पृथ्वी की चकती कितना कोण घेरती है?
  \item इस कक्षा की त्रिज्या की तुलना \omterm{prop:b1:central-forces:escape}{भूस्थिर कक्षा} से कीजिए; कौन-सी ऊँची
        है, और कितने गुना?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वहाँ तक पहुँचना।}
\begin{enumerate}[resume]
  \item पृष्ठ पर विराम से अंतिम कक्षा तक जाने के लिए प्रति किलोग्राम
        ऊर्जा (हवा और पृथ्वी के चक्रण की उपेक्षा कीजिए)।
  \item प्रक्षेपक पहले \qty{400}{km} पर किसी नीची \omterm{prop:b1:central-forces:effective}{वृत्तीय कक्षा} तक पहुँचता
        है। वहाँ चाल?
  \item उस कक्षा से अंतिम कक्षा तक \omterm{prop:b1:central-forces:hohmann}{होमान अंतरण}: दोनों प्रज्वलन और उनका
        योग।
  \item अंतरण की अवधि।
  \item प्रज्वलनों के योग की तुलना नीची कक्षा से \omterm{prop:b1:central-forces:escape}{पलायन वेग} से कीजिए: यह
        अभियान ``पृथ्वी छोड़ने'' से कितना दूर है?
  \item अंतरण से पहले ऊपरी चरण और उपग्रह का द्रव्यमान \qty{2000}{kg} है;
        इंजन की निष्कासन चाल \qty{3.0}{km/s} है और रॉकेट समीकरण
        $\Delta v = u\ln(m_0/m_1)$ देता है। दोनों प्रज्वलनों के लिए कितना
        प्रणोदक चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- आच्छादन और समय।}
\begin{enumerate}[resume]
  \item कोई उपग्रह पृथ्वी के उन सभी बिंदुओं से दिखता है जो पृथ्वी--उपग्रह
        रेखा के परितः उस शंकु के भीतर पड़ते हैं जिसका अर्ध-कोण $\theta$ है,
        जहाँ $\cos\theta = R/r$। $\theta$ निकालिए, और पृथ्वी के पृष्ठ का
        वह अंश $(1 - \cos\theta)/2$ भी जिसे वह ढकता है।
  \item $24$ उपग्रह एकसमान रूप से बिखरे हों, तो किसी एक बिंदु से औसतन
        कितने दिखेंगे? स्थान-निर्धारण के लिए आवश्यक न्यूनतम $4$ की संख्या
        विश्वसनीय लगती है?
  \item शिरोबिंदु पर स्थित किसी उपग्रह से प्रकाश का यात्रा-काल; और क्षितिज
        पर के उपग्रह से।
  \item अभिग्राही की स्थिति \qty{3}{m} तक जानने के लिए संकेतों का समय किस
        यथार्थता से नापना पड़ेगा?
  \item अभिग्राही की अपनी घड़ी सस्ता क्वार्ट्ज़ है, जो मिलीसेकंडों तक
        ग़लत रहती है: तीन के बजाय चार उपग्रह क्यों चाहिए?
  \item उपग्रहों की घड़ियाँ परमाणु घड़ियाँ हैं। एक दिन बाद \qty{3}{m} की
        त्रुटि के लिए आवश्यक सापेक्ष स्थिरता दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- संक्षेप में आपेक्षिकता।}
दो प्रभाव उपग्रहों की घड़ियों को भूमि की घड़ियों के सापेक्ष खिसका देते हैं
(सूत्र स्वीकृत, वर्ष 3 के खंड से): गति किसी घड़ी को अंश $v^2/2c^2$ से धीमा
कर देती है; और ऊँचाई उसे $GM(1/R - 1/r)/c^2$ से तेज़ कर देती है।
\begin{enumerate}[resume]
  \item चाल के कारण होती धीमी चाल प्रति दिन माइक्रोसेकंडों में निकालिए।
  \item ऊँचाई के कारण होती तेज़ी प्रति दिन माइक्रोसेकंडों में निकालिए।
  \item शुद्ध प्रभाव, और यदि सुधार न किया जाए तो प्रति दिन उससे बनती
        स्थिति-त्रुटि।
  \item उपग्रहों की घड़ियाँ प्रक्षेपण से पहले \qty{10.23}{MHz} के बजाय
        \qty{10.22999999543}{MHz} पर सधी दी जाती हैं। जाँचिए कि इससे शुद्ध
        प्रभाव की भरपाई हो जाती है।
  \item कक्षाएँ ठीक-ठीक केप्लरीय नहीं हैं: तीन विक्षोभ नाम से बताइए, और यह
        भी कि तंत्र उनसे कैसे निपटता है।
  \item पूरी शृंखला का सार दीजिए: दो संरक्षण नियम, एक कक्षा, एक
        उपग्रह-मंडल, और वह एक सुधार जिसका किसी ने अंदाज़ा भी न लगाया होता।
\end{enumerate}
\end{problem}
