\chapter{विद्युत्चुंबकीय प्रेरण और उसके अनुप्रयोग}\label{ch:b1:induction}

किसी कुंडली की ओर कोई चुंबक खिसकाइए और कुंडली में धारा बहने लगती है; चुंबक
रोक दीजिए और धारा रुक जाती है। फ़ैराडे ने दस बरस की खोज के बाद 1831 में यही
पाया, और विद्युत की सारी दुनिया इसी तथ्य पर चलती है: हर बिजलीघर, हर
परिणामित्र, हर मोटर, हर ध्वनिविस्तारक और ध्वनिग्राही, हर प्रेरण-चूल्हा और
हर बेतार आवेशक ऊर्जा को उसी नियम से बदलता है कि \emph{बदलता} हुआ \omterm{def:b1:induction:flux}{चुंबकीय
अभिवाह} कोई \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} चला देता है। यह अध्याय उस नियम को उसके चिह्नों
समेत बताता है, उसके दोनों रूप देता है (किसी स्थिर परिपथ में बदलता क्षेत्र,
और किसी स्थिर क्षेत्र में चलता परिपथ), \omterm{def:b1:induction:inductance}{स्वप्रेरकत्व} तथा \omterm{def:b1:induction:inductance}{अन्योन्य प्रेरकत्व}
परिभाषित करता है, और फिर उन विद्युत्-यांत्रिक मशीनों से होकर गुज़रता है ---
सरकती छड़, प्रत्यावर्तित्र, परिणामित्र, ध्वनिविस्तारक --- जहाँ वही नियम
यांत्रिक शक्ति को \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति} में बदल देता है और वापस भी, पर कभी मुफ़्त
नहीं।

\section{फ़ैराडे का नियम}

\begin{definition}[किसी परिपथ से होकर चुंबकीय अभिवाह]\label{def:b1:induction:flux}
\index{चुंबकीय अभिवाह}\index{वेबर}
किसी बंद परिपथ $\Gamma$ को दिशा दीजिए (यानी चलने की एक दिशा); तब उसके पृष्ठ
को दाएँ हाथ के नियम से अभिलंब $\vect n$ मिल जाता है। परिपथ से होकर
\emph{चुंबकीय अभिवाह} $\Phi = \sum\vect B\cdot\vect n\,\dd S$ है, जो
$\Gamma$ से घिरे किसी भी पृष्ठ पर लिया जाए (वेबर, $\qty{1}{Wb} = \qty{1}{T.m^2}$) ---
``किसी भी'', क्योंकि किसी बंद पृष्ठ से होकर $\vect B$ का अभिवाह शून्य है।
किसी एकसमान क्षेत्र और $N$ फेरों तथा $S$ क्षेत्रफल की किसी तलीय कुंडली के लिए, $\Phi =
NBS\cos\alpha$, जहाँ $\alpha$ $\vect B$ और $\vect n$ के बीच का कोण है।
\end{definition}

\begin{theorem}[फ़ैराडे का प्रेरण-नियम]\label{thm:b1:induction:faraday}
\index{फ़ैराडे का नियम}\index{विद्युत वाहक बल (प्रेरित)}\index{लेंज़ का नियम}
जब भी किसी परिपथ से होकर अभिवाह बदलता है, वह परिपथ किसी
\emph{प्रेरित \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल}} का आसन बन जाता है
\[
e = -\frac{\dd\Phi}{\dd t} ,
\]
जो परिपथ की दिशा में गिना जाता है, अर्थात् वह उसी दिशा में \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल}
$e$ के किसी आदर्श वोल्टता-स्रोत की तरह काम करता है; और $R$ प्रतिरोध के किसी
परिपथ में वह धारा $i = e/R$ चला देता है। ऋण चिह्न ही \emph{\omterm{thm:b1:induction:faraday}{लेंज़ का नियम}}
है: प्रेरित धारा इस तरह बहती है कि जिस अभिवाह-परिवर्तन ने उसे पैदा किया उसी
का विरोध करे --- उसका अपना क्षेत्र, और उस पर लगने वाले बल, कारण से ही लड़ते हैं।
\end{theorem}
\admitted

\begin{remark}[एक ही नियम के दो रूप]\label{rem:b1:induction:twofaces}
\index{नॉयमान प्रेरण}\index{लोरेंत्स प्रेरण}
अभिवाह या तो इसलिए बदलता है कि किसी स्थिर परिपथ से होकर $\vect B$ समय के
साथ बदल रहा हो (\emph{\omterm{rem:b1:induction:twofaces}{नॉयमान प्रेरण}}: परिणामित्र, प्रेरण-चूल्हा), या इसलिए
कि परिपथ किसी स्थायी क्षेत्र में चल या विरूपित हो रहा हो
(\emph{\omterm{rem:b1:induction:twofaces}{लोरेंत्स प्रेरण}}: जनित्र, ध्वनिविस्तारक)। दूसरी स्थिति में \omterm{def:b1:dc-circuits:sources}{विद्युत
वाहक बल} सीधे भी निकाला जा सकता है: किसी चालक अवयव $\dd\vect l$ के वाहक, जब
वह $\vect v$ से चल रहा हो, बल $q\vect v\wedge
\vect B$ अनुभव करते हैं, जो किसी विद्युत क्षेत्र $\vect v\wedge\vect B$ के
तुल्य है, जिससे $e = \oint(\vect v\wedge\vect B)\cdot\dd\vect l$ --- और यह
हमेशा $-\dd\Phi/\dd t$ से मेल खाता है। \omterm{thm:b1:induction:faraday}{फ़ैराडे का नियम} दोनों को समेट लेता है;
और वर्ष 2 का खंड समझाता है कि दोनों क्रियाविधियाँ एक ही सूत्र क्यों देती हैं।
\end{remark}

\begin{example}[परिमाण की कोटियाँ]\label{ex:b1:induction:orders}
पृथ्वी के क्षेत्र में $100$ फेरों और \qty{10}{cm^2} की कोई कुंडली, जो
\qty{0.1}{s} में पलट दी जाए: $\Delta\Phi = 2 \times 100 \times 10^{-3} \times 5 \times 10^{-5} = \qty{1e-5}{Wb}$,
और $e \sim \qty{0.1}{mV}$ --- जो किसी सुग्राही धारामापी से मापा जा सकता है,
और इसी तरह कभी पृथ्वी का क्षेत्र नापा गया था। वही कुंडली किसी चुंबक की
\qty{1}{T} दरार में, जो \qty{0.1}{s} में बाहर खींच ली जाए: \qty{1}{V}। और
$1000$ फेरों की किसी परिणामित्र-कुंडली में \qty{10}{cm^2} पर \qty{50}{Hz} पर \qty{1}{T}: $e_{\max} =
1000 \times 10^{-3} \times 314 = \qty{314}{V}$ --- यानी घरेलू बिजली।
\end{example}

\section{स्वप्रेरकत्व और अन्योन्य प्रेरकत्व}

\begin{definition}[प्रेरकत्व]\label{def:b1:induction:inductance}
\index{स्वप्रेरकत्व}\index{हेनरी}\index{अन्योन्य प्रेरकत्व}
किसी परिपथ के अपने क्षेत्र का उसी में से अभिवाह उसकी धारा के समानुपाती होता
है: $\Phi_{\text{अपना}} = Li$, जहाँ $L > 0$ उसका \emph{स्वप्रेरकत्व} है
(हेनरी, \unit{H})। परिपथ~1 परिपथ~2 में से जो अभिवाह भेजता है वह
$\Phi_{1 \to 2} = Mi_1$ है, और सममित रूप से $\Phi_{2 \to 1} = Mi_2$, जहाँ $M$
उनका \emph{अन्योन्य प्रेरकत्व} है (जिसका चिह्न दिशाओं से तय होता है)। तब
परिपथ~1 का कुल अभिवाह $\Phi_1 = L_1i_1 + Mi_2$ है।
\end{definition}

\begin{proposition}[प्रेरक; परिनालिका]\label{prop:b1:induction:inductor}
\index{प्रेरक}\index{चुंबकीय ऊर्जा}\index{चुंबकीय क्षेत्र का ऊर्जा घनत्व}
$L$ प्रेरकत्व का कोई परिपथ, जो बदलती धारा ले जा रहा हो, \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल}
$e = -L\,\dd i/\dd t$ का आसन होता है: ग्राही रूढ़ि में यही वह वोल्टता
$u = L\,\dd i/\dd t$ है जो \cref{ch:b1:transient-regimes} से प्रेरक के लिए
काम आती रही है। वह ऊर्जा $W = \tfrac12Li^2$ संचित रखता है।
प्रति मीटर $n$ फेरों, $\ell$ लंबाई और $S$ काट की किसी लंबी परिनालिका का
$L = \mu_0n^2S\ell$ होता है; और उसकी संचित ऊर्जा $\tfrac12\mu_0n^2I^2S\ell =
\dfrac{B^2}{2\mu_0}\,S\ell$: यानी \omterm{prop:b1:induction:inductor}{चुंबकीय ऊर्जा} क्षेत्र में ही रहती है,
घनत्व $B^2/2\mu_0$ पर, ठीक वैसे ही जैसे विद्युत ऊर्जा $\varepsilon_0E^2/2$ पर।
\end{proposition}

\begin{proof}
$\Phi = Li$ के साथ फ़ैराडे, जहाँ $L$ अचर है। ऊर्जा: प्रेरक को मिली शक्ति
$ui = Li\,\dd i/\dd t = \dd(\tfrac12Li^2)/\dd t$ है। परिनालिका:
भीतर $B = \mu_0nI$, और अभिवाह $BS$ $n\ell$ फेरों में से हर एक में से, अतः $\Phi =
\mu_0n^2S\ell I$। सामान्य घनत्व स्वीकृत है।
\end{proof}

\begin{example}[किसी चुंबकीय अनुनाद यंत्र का चुंबक]\label{ex:b1:induction:mri}
लगभग एक घन मीटर पर \qty{1.5}{T}: $B^2/2\mu_0 = 2.25/2.5 \times 10^{-6} =
\qty{0.9}{MJ/m^3}$ --- यानी किसी कार-बैटरी जितनी ऊर्जा, जो ख़ाली जगह में
संचित है। \qty{500}{A} की कुंडली-धारा के साथ, $L = 2W/I^2 = \qty{7}{H}$। और
यदि अतिचालक अचानक प्रतिरोधी हो जाए (यानी कोई \emph{शमन}), तो वह मेगाजूल
सेकंडों में ऊष्मा बन जाता है और सैकड़ों लीटर द्रव हीलियम उड़ा देता है:
इसीलिए चुंबक-कक्ष में उसके लिए एक निकास बना रहता है।
\end{example}

\begin{proposition}[आदर्श परिणामित्र]\label{prop:b1:induction:transformer}
\index{परिणामित्र}
एक ही बंद लोहे के क्रोड पर लपेटी गई $N_1$ और $N_2$ फेरों की दो कुंडलियाँ,
जिनसे होकर वही अभिवाह $\varphi$ हर फेरे में से गुज़रे, यह मानती हैं
\[
\frac{u_2}{u_1} = \frac{N_2}{N_1} , \qquad N_1i_1 + N_2i_2 = 0 \quad\text{(न भार-हानियाँ, न चुंबकन-धारा)},
\]
जिससे $u_2i_2 = -u_1i_1$: यानी प्राथमिक पर ली गई शक्ति द्वितीयक पर दे दी
जाती है; वोल्टताएँ फेरों के अनुपात में चढ़ती-उतरती हैं और धाराएँ उलटे
अनुपात में। यह केवल बदलती धाराओं के साथ ही चलता है।
\end{proposition}

\begin{proof}
$u_1 = N_1\,\dd\varphi/\dd t$ और $u_2 = N_2\,\dd\varphi/\dd t$ (दोनों पर ग्राही
रूढ़ि, और दिशाएँ उभयनिष्ठ अभिवाह के अनुदिश चुनी हुई): भाग दीजिए।
धारा वाला संबंध क्रोड पर \omterm{thm:b1:magnetostatics:ampere}{ऐंपियर का नियम} है: उसके परिचालन को
$\mu_0\mu_r$ गुना कुल ऐंपियर-फेरों के बराबर होकर वह अभिवाह बनाना पड़ता; और
किसी आदर्श क्रोड ($\mu_r \to \infty$) के लिए उसमें लुप्त होते ऐंपियर-फेरे ही
लगते हैं, अतः $N_1i_1 + N_2i_2 = 0$। इस रूप में स्वीकृत।
\end{proof}

\section{चलते चालक: जनित्र और मोटर}

\begin{proposition}[सरकती छड़]\label{prop:b1:induction:rails}
\index{लाप्लास की पटरियाँ}\index{विद्युत्-यांत्रिक रूपांतरण}
$\ell$ लंबाई की कोई छड़ $v$ चाल से दो पटरियों पर सरकती है, जो किसी
प्रतिरोध $R$ से बंद हैं, और तल पर अभिलंब किसी एकसमान क्षेत्र $B$ में। वह
\omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} $e = B\ell v$ का आसन है, धारा $i = B\ell v/R$ चलाती है, और
\omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल} $F = B\ell i = B^2\ell^2v/R$ अनुभव करती है, जो उसकी गति का
\emph{विरोध} करता है। उसे धकेलने में लगी यांत्रिक शक्ति, $Fv$, उस \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत
शक्ति} $ei = Ri^2$ के बराबर है जो क्षय होती है: यानी छड़ एक जनित्र है। और
इसकी जगह किसी स्रोत $E$ से खिलाने पर वह एक मोटर है: धारा $(E - B\ell v)/R$
उसे तब तक धकेलती है जब तक प्रति-विद्युत वाहक बल $B\ell v$ $E$ को संतुलित न
कर दे, यानी बिना भार की चाल $E/B\ell$ तक।
\end{proposition}

\begin{proof}
अभिवाह $B\ell x$, और उस दिशा में $e = -B\ell\dot x$ जिसमें $\vect n$
$\vect B$ के अनुदिश हो; और इस तरह दिशित धारा पर \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल} $i\vect\ell\wedge\vect B$
$\vect v$ के विरुद्ध संकेत करता है (\cref{thm:b1:induction:faraday},
लेंज़)। विद्युत समीकरण $e = Ri$ को $i$ से और यांत्रिक को $v$ से गुणा
कीजिए: $Fv = B\ell iv = ei$। विद्युत्-यांत्रिक शक्ति
$ei$ और लाप्लास शक्ति $-Fv$ हमेशा विपरीत होती हैं --- यही रूपांतरण का
नियम है।
\end{proof}

\begin{proposition}[घूमता पाश: प्रत्यावर्तित्र]\label{prop:b1:induction:alternator}
\index{प्रत्यावर्तित्र}
$N$ फेरों और $S$ क्षेत्रफल की कोई कुंडली, जो $\omega$ से किसी ऐसी अक्ष के
परितः घूमे जो किसी एकसमान $B$ पर लंब हो, उसका $\Phi = NBS\cos\omega t$ होता है और
\[
e = NBS\omega\sin\omega t :
\]
यानी $NBS\omega$ \omterm{def:b1:wave-propagation:signal}{आयाम} का कोई ज्यावक्रीय \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल}, जिसकी आवृत्ति घूर्णन
की है। किसी प्रतिरोध $R$ में वह माध्य शक्ति $(NBS\omega)^2/2R$ देती है, जिसे
वह \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} जुटाता है जो उसे उसकी अपनी धारा के लाप्लास युग्म
$-\vect m\wedge\vect B$ के विरुद्ध घुमाता रहता है।
\end{proposition}

\begin{proof}
अभिवाह को अवकलित कीजिए; और $\langle\sin^2\rangle = \tfrac12$। युग्म
$m = Ni S$ के साथ \cref{thm:b1:magnetostatics:dipoleinfield} है; और उसकी माध्य
शक्ति $\langle\Gamma\omega\rangle$ $\langle ei\rangle$ के बराबर है, उसी
गणना से जो छड़ के लिए की गई थी।
\end{proof}

\section{ध्वनिविस्तारक: एक युग्मित निकाय}

\begin{proposition}[त्रिज्य क्षेत्र में किसी कुंडली का विद्युत्-यांत्रिक युग्मन]\label{prop:b1:induction:loudspeaker}
\index{ध्वनिविस्तारक}\index{ध्वनिग्राही}
कुल तार-लंबाई $\ell$ की कोई कुंडली किसी त्रिज्य क्षेत्र $B$ में बैठी है,
जिससे तार का हर अवयव $\vect B$ पर लंब रहे; वह अपनी अक्ष के अनुदिश
$v$ चाल से चलती है और $i$ ले जाती है। तब वह अक्षीय \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल}
$F = B\ell i$ अनुभव करती है और प्रति-विद्युत वाहक बल $e = -B\ell v$ का आसन
होती है (ग्राही रूढ़ि में: वोल्टता $B\ell v$)। द्रव्यमान $m$, दृढ़ता
$k$ के किसी निलंबन और यांत्रिक अवमंदन $h$ के साथ, जब उसे वोल्टता $u$ खिलाई
जाए, कुंडली के प्रतिरोध $R$ तथा प्रेरकत्व $L$ से होकर:
\[
u = Ri + L\frac{\dd i}{\dd t} + B\ell v , \qquad
m\frac{\dd v}{\dd t} = -kx - hv + B\ell i .
\]
$i$ से और $v$ से गुणा करके जोड़ने पर: \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति} $ui$
जूल-हानि $Ri^2$ के, संचित ऊर्जाओं
$\tfrac12Li^2 + \tfrac12mv^2 + \tfrac12kx^2$ की वृद्धि के, और यांत्रिक हानि
$hv^2$ के (जिसमें ध्वनि भी है) बराबर बैठती है: और रूपांतरण-पद $B\ell iv$
दोनों समीकरणों के बीच कट जाता है। उलटा चलाने पर (यानी गति थोपकर, $u$ पढ़कर)
वही युक्ति कोई गतिक \emph{ध्वनिग्राही} है।
\end{proposition}

\begin{proof}
हर अवयव पर \omterm{thm:b1:magnetostatics:laplace}{लाप्लास बल} $i\,\dd\vect l\wedge\vect B$, जो सब अक्षीय हैं और एक ही
चिह्न के; और \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} तार पर $(\vect v\wedge\vect B)\cdot\dd\vect l$
जोड़कर। ऊर्जा-संतुलन दोनों समीकरणों का वही योग है जो ऊपर कहा गया।
\end{proof}

\begin{remark}[युग्मन अवमंदन क्यों करता है]\label{rem:b1:induction:damping}
नीची आवृत्ति पर ($L\omega \ll R$), $i \approx (u - B\ell v)/R$ होता है और बल
$B\ell u/R - (B^2\ell^2/R)\,v$: यानी विद्युत परिपथ यांत्रिक अवमंदन में
$B^2\ell^2/R$ और जोड़ देता है। किसी प्रवर्धक से जुड़ा कोई ध्वनिविस्तारक
(जिसका निर्गत प्रतिरोध कम है) ख़ूब अवमंदित रहता है --- संकेत रुकते ही उसका
शंकु ठहर जाता है --- और वही युग्मन ध्वनिग्राही को ध्वनि से ऊर्जा लेने देता
है (\cref{pb:b1:induction:1})।
\end{remark}

धकेलने वाले को लगाना पड़ेगा, यांत्रिक शक्ति, \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:3}

\omterm{def:b1:transient-regimes:firstorder}{कालांक}; और भीतर क्षेत्र तथा ऊर्जा घनत्व, ताकि $B^2/2\mu_0$ जाँचा जा सके।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:4}
\qty{230}{V} से \qty{12}{V} तक का कोई परिणामित्र: फेरों का अनुपात;
$50$-फेरों के किसी द्वितीयक के लिए प्राथमिक के फेरे; और जब द्वितीयक

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:5}
\omterm{thm:b1:induction:faraday}{लेंज़ का नियम}: हर स्थिति में प्रेरित धारा की दिशा और बल की दिशा दीजिए ---

जाता है; (घ) कोई चुंबक ताँबे की किसी लंबी नली में से गिरता है। (घ) के लिए
ऊर्जा से समझाइए कि वह किसी छोटी अचर चाल से क्यों गिरता है।

\begin{exercise}[$\star\star$]\label{exo:b1:induction:6}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:7}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:8}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:9}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:10}

(क) विद्युत और यांत्रिक समीकरण; दिखाइए कि $m\,\dd v/\dd t = B\ell(E -

\end{exercise}

किनारे को छूते हैं। (क) किसी त्रिज्या के अनुदिश $(\vect v\wedge

का नियम} लगाकर $e$ फिर से पाइए: कौन-सा अभिवाह बदलता है?

\section{समस्या: साइकिल का डायनमो और ध्वनिविस्तारक}

\begin{problem}[{सप्ताहांत समस्या --- खोलकर देखे गए दो विद्युत्-यांत्रिक
रूपांतरक: एक डायनमो जो अपने आप को नियंत्रित रखता है, और एक ध्वनिविस्तारक
जिसका प्रवर्धक ही उसका ब्रेक भी है}]
\label{pb:b1:induction:1}

\medskip
\noindent\textbf{भाग I --- साइकिल का डायनमो।}
$N = 300$ फेरों और $S = \qty{3.0}{cm^2}$ क्षेत्रफल की कोई कुंडली किसी
एकसमान क्षेत्र $B = \qty{0.30}{T}$ में घूमती है; उसकी धुरी पर
$r = \qty{1.0}{cm}$ त्रिज्या का कोई बेलन टायर से सटा है, ताकि कुंडली
$\omega = v/r$ से घूमे, जब साइकिल की चाल $v$ हो। कुंडली का प्रतिरोध $r_c = \qty{4.0}{\ohm}$,
प्रेरकत्व $L = \qty{20}{mH}$; और बत्ती $R = \qty{12}{\ohm}$ का कोई प्रतिरोध है।
\begin{enumerate}
  \item समय के फलन के रूप में कुंडली से होकर अभिवाह और प्रेरित \omterm{def:b1:dc-circuits:sources}{विद्युत
        वाहक बल}; और $v = \qty{18}{km/h}$ पर \omterm{def:b1:wave-propagation:signal}{आयाम} तथा आवृत्ति।
  \item पहले $L$ की उपेक्षा करके: \qty{18}{km/h} पर बत्ती में
        वर्ग-माध्य-मूल धारा और शक्ति; क्या ``\qty{6}{V}, \qty{3}{W}'' की
        कोई बत्ती सुखी रहेगी?
  \item साइकिल-सवार डायनमो को जो माध्य यांत्रिक शक्ति देता है, और टायर पर
        उससे संगत बल (लोटनी घर्षण, लगभग \qty{2}{N}, से तुलना कीजिए)।
  \item लेंज़ के नियम से समझाइए कि डायनमो विरोध क्यों करता है, और ``मुफ़्त
        में'' जलती कोई बत्ती ऊर्जा-संरक्षण का खंडन क्यों नहीं करती।
  \item अब $L$ भी लीजिए: परिपथ की सम्मिश्र \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा}; दिखाइए कि धारा का
        \omterm{def:b1:wave-propagation:signal}{आयाम} $I = NBS\omega/\sqrt{(R + r_c)^2 + L^2\omega^2}$ है और वह ऊँची चाल
        पर संतृप्त हो जाता है; उसकी सीमा, और वह चाल जिस पर वह सीमा का
        $70\%$ पा लेता है।
  \item प्रेरकत्व के बिना, ढलान पर \qty{54}{km/h} पर बत्ती को क्या मिलता?
        और उसके साथ?
  \item \qty{9}{km/h} पर और \qty{36}{km/h} पर धारा तथा बत्ती की शक्ति; और
        इस स्वयं-नियंत्रण पर टिप्पणी कीजिए।
  \item असली डायनमो किसी स्थिर कुंडली के भीतर चुंबक घुमाते हैं। क्या
        विश्लेषण बदलता है? इससे कौन-सी व्यावहारिक कठिनाई हट जाती है?
  \item लाल बत्ती पर साइकिल रुकने पर बत्ती क्या करती है, और आधुनिक
        बत्तियाँ इसे कैसे सुलझाती हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ध्वनिविस्तारक।}
ध्वनि-कुंडली: तार की लंबाई $\ell = \qty{5.0}{m}$, किसी त्रिज्य क्षेत्र $B =
\qty{1.0}{T}$ में, प्रतिरोध $R = \qty{8.0}{\ohm}$, प्रेरकत्व $L = \qty{0.50}{mH}$;
चलता द्रव्यमान (कुंडली और शंकु) $m = \qty{10}{g}$; निलंबन की दृढ़ता
$k = \qty{1000}{N/m}$, यांत्रिक अवमंदन $h = \qty{1.0}{N.s/m}$। ज्यावक्रीय
अवस्था, \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम}।
\begin{enumerate}[resume]
  \item किसी धारा $i$ के लिए कुंडली पर बल; और किसी चाल $v$ के लिए
        प्रति-विद्युत वाहक बल (दोनों व्युत्पन्न कीजिए, चिह्नों सहित)।
  \item विद्युत और यांत्रिक समीकरण लिखिए।
  \item $i$ हटाइए और दिखाइए कि
        $\underline v = \dfrac{B\ell\,\underline u}{(R + jL\omega)(jm\omega + h + k/j\omega) + B^2\ell^2}$।
  \item जिन आवृत्तियों पर $L\omega \ll R$ है, वहाँ दिखाइए कि युग्मन एक
        अतिरिक्त यांत्रिक अवमंदन $B^2\ell^2/R$ की तरह काम करता है; उसका
        मान; और शंकु की अनुनाद-आवृत्ति तथा उसका गुणता गुणक, विद्युत अवमंदन
        के साथ और उसके बिना।
  \item प्रवर्धक \qty{1}{kHz} पर \qty{1.0}{A} \omterm{def:b1:wave-propagation:signal}{आयाम} की $i$ देता है:
        शंकु के बल, त्वरण, विस्थापन और वेग के \omterm{def:b1:wave-propagation:signal}{आयाम} (द्रव्यमान-शासित
        अवस्था); और प्रति-विद्युत वाहक बल का \omterm{def:b1:wave-propagation:signal}{आयाम}, $Ri$ से तुलना करके।
  \item अनुनाद-आवृत्ति पर वही धारा: वेग का \omterm{def:b1:wave-propagation:signal}{आयाम} (अवमंदन-शासित) और
        प्रति-विद्युत वाहक बल; टिप्पणी कीजिए।
  \item \qty{1}{kHz} पर \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}; $h$ में क्षय हुई यांत्रिक शक्ति
        (जिसे ध्वनि तथा निलंबन की हानियाँ माना जाए); और \omterm{def:b1:heat-engines:efficiency}{दक्षता}।
  \item ध्वनि-दाब शंकु के त्वरण के समानुपाती होता है (इसे स्वीकार लीजिए):
        यही वह कारण क्यों है जिससे कोई द्रव्यमान-शासित शंकु अनुनाद के ऊपर
        चपटी अनुक्रिया देता है?
  \item क्षेत्र त्रिज्य क्यों होना चाहिए, और कोई ट्वीटर छोटा क्यों होता है?
  \item प्रवर्धक को दिखती \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} $\underline u/\underline i$ का मापांक
        \qty{10}{Hz} से \qty{10}{kHz} तक खींचिए: नीची आवृत्ति पर मान,
        अनुनाद पर (अवमंदन-शासित वेग लीजिए), और \qty{10}{kHz} पर।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- उलटा, और हिसाब-किताब।}
\begin{enumerate}[resume]
  \item वही युक्ति ध्वनिग्राही के रूप में: कोई ध्वनि शंकु को
        $v = \qty{1.0}{mm/s}$ से चलाती है; खुले परिपथ की वोल्टता।
  \item $R_L = \qty{600}{\ohm}$ से भारित: धारा, और शंकु पर रोकने वाला बल;
        दिखाइए कि ध्वनिग्राही ध्वनि से शक्ति किसी प्रभावी अवमंदन
        $B^2\ell^2/(R + R_L)$ के ज़रिए लेता है।
  \item कुंडली अल्युमिनियम के किसी बेलन (यानी ढाँचे) पर लपेटी गई है, जो
        \qty{0.10}{m} लंबाई और \qty{1.0}{m\ohm} प्रतिरोध का एक बंद फेरा है:
        वह कितना अवमंदन जोड़ता है; और ढाँचों में चीरा क्यों लगाया जाता है।
  \item किसी ध्वनिविस्तारक के शंकु को ठोकिए, पहले तब जब उसके सिरे खुले
        हों, फिर तब जब वे लघुपथित हों: कौन-सा अधिक देर गूँजता है, और
        क्यों?
  \item \omterm{def:b1:transient-regimes:firstorder}{स्थायी अवस्था} में एक आवर्तकाल पर ध्वनिविस्तारक का ऊर्जा-संतुलन
        लिखिए, और हर पद का नाम बताइए।
  \item सार दीजिए: हर युक्ति के लिए, प्रयुक्त नियम और याद रखने लायक दो
        आँकड़े।
\end{enumerate}
\end{problem}
