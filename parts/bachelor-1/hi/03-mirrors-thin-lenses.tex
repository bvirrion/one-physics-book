\chapter{दर्पण और पतले लेंस}\label{ch:b1:mirrors-thin-lenses}

शृंगार-दर्पण चेहरे को दुगुना बड़ा दिखाता है; चम्मच की पीठ उसे नन्हा और सीधा
दिखाती है, जबकि उसका कटोरा नन्हा और उलटा; बाँह भर दूर पकड़ा आवर्धक काँच पूरे
कमरे को सिर के बल खड़ा कर देता है, और पास लाने पर डाक-टिकट को बड़ा कर देता है;
मसूर के दाने जितना फ़ोन का \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} पूरी गली को एक संवेदक पर उतार देता है। तीन
दूरियों के बीच एक ही बीजीय संबंध इन सब प्रतिबिंबों को साधता है, और यह अध्याय
उसी को स्थापित करता है --- पहले वक्र दर्पणों के लिए, फिर उस पतले \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के लिए
जिससे लगभग सारे प्रकाशिक उपकरण बने हैं।

\section{प्रतिबिंब और गाउस की शर्तें}

\begin{definition}[वस्तु, प्रतिबिंब, बिंदुसंगतता]\label{def:b1:mirrors-thin-lenses:image}
\index{प्रतिबिंब (प्रकाशिकी)}\index{वस्तु (प्रकाशिकी)}\index{बिंदुसंगतता}\index{वास्तविक प्रतिबिंब}\index{आभासी प्रतिबिंब}
किसी प्रकाशिक निकाय पर किसी बिंदु $A$ (\emph{वस्तु बिंदु}) से किरणें पड़ती
हैं। यदि निकलती हुई सारी किरणें किसी एक बिंदु $A'$ से गुज़रें, या सब उसी एक
बिंदु $A'$ से आती जान पड़ें, तो निकाय $A$ के लिए \emph{बिंदुसंगत} है और $A'$
ही $A$ का \emph{प्रतिबिंब} है। प्रतिबिंब \emph{वास्तविक} है यदि निकलती किरणें
सचमुच $A'$ पर मिलती हों (तब उसे परदे पर पकड़ा जा सकता है), और \emph{आभासी} है
यदि केवल उनके पीछे की ओर बढ़ाए गए विस्तार मिलते हों (तब वह निकाय के भीतर से
दिखता है, प्रक्षेपित नहीं होता)। इसी तरह वस्तु वास्तविक है यदि आती किरणें $A$
से अपसरित होती हों, और आभासी यदि वे निकाय के प्रवेश के पीछे $A$ की ओर अभिसरित
होती हों। युग्म $(A, A')$ \emph{संयुग्मी} कहलाता है।
\end{definition}

\begin{definition}[केंद्रित निकाय; गाउस की शर्तें]\label{def:b1:mirrors-thin-lenses:gauss}
\index{प्रकाशिक अक्ष}\index{गाउस की शर्तें}\index{उपाक्षीय किरण}
\emph{केंद्रित निकाय} में घूर्णन-सममिति का एक अक्ष होता है, जिसे
\emph{प्रकाशिक अक्ष} कहते हैं। जो किरणें अक्ष के पास बनी रहती हैं और उसके साथ
छोटे कोण बनाती हैं, वे \emph{उपाक्षीय} कहलाती हैं; और जो निकाय केवल उपाक्षीय
किरणों के साथ काम में लिया जाता है, उसके बारे में कहा जाता है कि वह
\emph{गाउस की शर्तों} में काम कर रहा है। इन शर्तों में केंद्रित निकाय हर बिंदु
के लिए लगभग बिंदुसंगत होता है, और अक्ष के लंबवत किसी वस्तु-तल का प्रतिबिंब-तल
भी अक्ष के लंबवत होता है (\emph{समतल-संगतता})।
\end{definition}
\admitted

\begin{remark}[छोटे कोण ही क्यों]\label{rem:b1:mirrors-thin-lenses:why}
\omterm{prop:b1:rays-reflection-refraction:mirror}{समतल दर्पण} अकेला ऐसा सरल निकाय है जो हर बिंदु के लिए कठोरता से बिंदुसंगत है।
कोई \omterm{def:b1:mirrors-thin-lenses:mirror}{गोलीय दर्पण} या \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} केवल उतना ही बिंदुसंगत है जितना
$\sin\theta \approx \tan\theta \approx \theta$ सही ठहरता है: यही सन्निकटन
स्नेल के नियम को दूरियों के बीच रैखिक संबंधों में बदल देता है। इसके बाहर किसी
बिंदु का \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} एक धब्बा बन जाता है (\emph{विपथन}), और अभिकल्पक का काम ---
रोधक, लेंसों के संयोजन, अगोलीय पृष्ठ --- उस धब्बे को आँख या संवेदक की विभेदन
सीमा से नीचे धकेल देना है।
\end{remark}

\begin{notation}[बीजीय माप और आवर्धन]\label{not:b1:mirrors-thin-lenses:signs}
\index{बीजीय माप}\index{अनुप्रस्थ आवर्धन}
\omterm{def:b1:mirrors-thin-lenses:gauss}{प्रकाशिक अक्ष} आती हुई रोशनी की दिशा में दिष्ट किया जाता है, और अनुप्रस्थ दिशा
ऊपर की ओर। अक्ष पर स्थित बिंदुओं $P$, $Q$ के लिए $\overline{PQ}$ \omterm{rem:b1:mirrors-thin-lenses:why}{बीजीय माप} को
दर्शाता है ($\overline{PQ} > 0$ यदि $Q$ $P$ के अनुप्रवाह में हो); और अक्ष के
लंबवत वस्तु $AB$ तथा उसके \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $A'B'$ के लिए \emph{\omterm{rem:b1:mirrors-thin-lenses:why}{अनुप्रस्थ आवर्धन}} है
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}} ,
\]
जो उलटे \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} के लिए ऋणात्मक होता है और बड़े हुए \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} के लिए $1$ से
बड़े निरपेक्ष मान का।
\end{notation}

\section{गोलीय दर्पण}

\begin{definition}[गोलीय दर्पण]\label{def:b1:mirrors-thin-lenses:mirror}
\index{गोलीय दर्पण}\index{अवतल दर्पण}\index{उत्तल दर्पण}\index{फोकस}
\emph{गोलीय दर्पण} केंद्र $C$ और शीर्ष $S$ (अक्ष के साथ उसका प्रतिच्छेद) वाली
परावर्तक गोलीय टोपी है। वह \emph{अवतल} है यदि प्रकाश उसके खोखले पक्ष से
मिलता हो ($C$ आती हुई रोशनी की ओर), अन्यथा \emph{उत्तल}। उसका \emph{फोकस}
$F$ अक्ष पर अनंत दूरी के किसी बिंदु का \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} है; और उसकी \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी}
$f = \overline{SF}$ है।
\end{definition}

\begin{proposition}[गोलीय दर्पण का फोकस]\label{prop:b1:mirrors-thin-lenses:focus}
\omterm{def:b1:mirrors-thin-lenses:gauss}{गाउस की शर्तों} में \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} $SC$ का मध्यबिंदु होता है:
$\overline{SF} = \overline{SC}/2$। \omterm{def:b1:mirrors-thin-lenses:mirror}{अवतल दर्पण} के लिए $f < 0$ (वास्तविक
\omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस}, आगे की ओर); उत्तल के लिए $f > 0$ (आभासी \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस}, पीछे की ओर)।
\end{proposition}

\begin{proof}
अक्ष के समांतर, ऊँचाई $h$ पर चलती कोई किरण दर्पण से $I$ पर मिलती है; $I$ पर
अभिलंब त्रिज्या $CI$ है, जो अक्ष के साथ कोण $\alpha$ बनाती है, जहाँ
$\sin\alpha = h/R$। अभिलंब के साथ उसी कोण $\alpha$ पर परावर्तित होकर किरण अक्ष
को $F$ पर काटती है; त्रिभुज $CIF$ में कोण $\alpha$ $C$ पर और $I$ पर बराबर है,
अतः $CF = IF$ और प्रक्षेप लेने पर
$CF = R/(2\cos\alpha)$। उपाक्षीय किरणों के लिए $\cos\alpha \to 1$ और
$CF = R/2$: ऐसी सारी किरणें अक्ष को एक ही बिंदु पर काटती हैं, यानी
$SC$ के मध्यबिंदु पर। चिह्न: \omterm{def:b1:mirrors-thin-lenses:mirror}{अवतल दर्पण} के लिए $C$ और $F$ आगे की ओर
(अनुप्रवाह से पहले) पड़ते हैं, अतः $\overline{SC} < 0$।
\end{proof}

\begin{theorem}[गोलीय दर्पण के लिए संयुग्मन]\label{thm:b1:mirrors-thin-lenses:mirrorconj}
\index{संयुग्मन संबंध!गोलीय दर्पण}
\omterm{def:b1:mirrors-thin-lenses:gauss}{गाउस की शर्तों} में, \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $A'$ अक्ष पर स्थित अपने वस्तु बिंदु $A$ के साथ
इस संबंध का पालन करता है:
\[
\frac{1}{\overline{SA'}} + \frac{1}{\overline{SA}} = \frac{2}{\overline{SC}}
= \frac{1}{\overline{SF}},
\qquad
\gamma = -\frac{\overline{SA'}}{\overline{SA}} .
\]
\end{theorem}

\begin{proof}
कोई \omterm{def:b1:mirrors-thin-lenses:mirror}{अवतल दर्पण} और एक वास्तविक वस्तु $A$ लीजिए, जो $C$ के परे रखी हो; बीजीय
रूप फिर हर स्थिति को समेट लेता है। $A$ से चली एक किरण दर्पण से $I$ पर टकराती है
(ऊँचाई $h$) और अक्ष पर $A'$ को लौटती है। अक्ष के साथ बने कोणों को $u$, $u'$,
$\omega$ कहिए, जो क्रमशः $AI$, $A'I$ और $CI$ के हैं; उपाक्षीय रूप में
$u = h/AS$, $u' = h/A'S$, $\omega = h/CS$ (दूरियाँ)। परावर्तन का
नियम कहता है कि अभिलंब $CI$ $AI$ और $A'I$ के बीच के कोण को समद्विभाजित करता
है: $\omega - u = u' - \omega$, यानी $u + u' = 2\omega$, यानी
$1/AS + 1/A'S = 2/CS$; और चूँकि $\overline{SA}$, $\overline{SA'}$,
$\overline{SC}$ तीनों ऋणात्मक हैं, यही ऊपर दर्शाया गया संबंध बन जाता है।
आवर्धन के लिए, शीर्ष पर परावर्तित किरण $B \to S \to B'$ अक्ष के साथ बराबर
कोण बनाती है: लंबाइयों में $A'B'/SA' = AB/SA$, और \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} उलटा होता है,
अतः $\gamma = -\overline{SA'}/\overline{SA}$।
\end{proof}

\begin{method}[दर्पण में प्रतिबिंब की रचना]\label{met:b1:mirrors-thin-lenses:mirrorrays}
अक्ष से हटकर स्थित किसी वस्तु बिंदु $B$ के लिए चार उल्लेखनीय किरणों में से दो
खींचिए; उनका (या उनके विस्तारों का) प्रतिच्छेद ही $B'$ है:
\begin{enumerate}
  \item अक्ष के समांतर $\to$ $F$ से होकर परावर्तित;
  \item $F$ से होकर $\to$ अक्ष के समांतर परावर्तित;
  \item $C$ से होकर $\to$ अपने ही ऊपर वापस परावर्तित;
  \item शीर्ष $S$ की ओर $\to$ अक्ष के परितः सममित रूप से परावर्तित।
\end{enumerate}
फिर $A'$ अक्ष पर $B'$ से डाले गए लंब का पाद है।
\end{method}

\begin{example}[दाढ़ी का दर्पण, पश्च-दृश्य दर्पण]\label{ex:b1:mirrors-thin-lenses:mirrors}
\qty{60}{cm} त्रिज्या का \omterm{def:b1:mirrors-thin-lenses:mirror}{अवतल दर्पण} ($f = \qty{-30}{cm}$), और उससे
\qty{20}{cm} पर चेहरा: $1/\overline{SA'} = -1/30 + 1/20 = 1/60$,
$\overline{SA'} = \qty{+60}{cm}$ --- दर्पण के पीछे, आभासी ---
और $\gamma = -60/(-20) = +3$: सीधा, तीन गुना बड़ा, यानी दाढ़ी का
दर्पण। \qty{2.0}{m} त्रिज्या का \omterm{def:b1:mirrors-thin-lenses:mirror}{उत्तल दर्पण} ($f = \qty{+1.0}{m}$),
और उससे \qty{20}{m} पीछे एक कार: $1/\overline{SA'} = 1 + 1/20$,
$\overline{SA'} = \qty{0.95}{m}$ (आभासी, पीछे), $\gamma = +0.048$:
सीधा, बीस गुना छोटा, और इसीलिए चौड़ा क्षेत्र --- यही पश्च-दृश्य दर्पण है, उस
चेतावनी के साथ कि वस्तुएँ दिखने से अधिक पास हैं।
\end{example}

\section{पतले लेंस}

\begin{definition}[पतला लेंस]\label{def:b1:mirrors-thin-lenses:lens}
\index{पतला लेंस}\index{अभिसारी लेंस}\index{अपसारी लेंस}\index{प्रकाशिक केंद्र}\index{फोकस दूरी}\index{अभिसारिता}
\emph{लेंस} दो गोलीय (या समतल) पृष्ठों से घिरा \omterm{def:b1:rays-reflection-refraction:index}{पारदर्शी माध्यम} है; वह
\emph{पतला} तब है जब उसकी मोटाई त्रिज्याओं के और खेल में आती दूरियों के सामने
नगण्य हो, ताकि दोनों शीर्ष एक ही बिंदु में मिल जाएँ, जिसे
\emph{प्रकाशिक केंद्र} $O$ कहते हैं। $O$ से होकर जाती किरण विचलित नहीं होती।
\emph{\omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस}} $F'$ अक्ष पर अनंत दूरी के किसी बिंदु का \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} है;
\emph{वस्तु \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस}} $F$ वह बिंदु है जिसका \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} अनंत पर बनता है; दोनों $O$
से समान दूरी पर हैं: $\overline{OF'} = -\overline{OF} = f'$, यानी
\emph{\omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} दूरी}। लेंस \emph{अभिसारी} है यदि $f' > 0$ हो ($F'$ वास्तविक,
अनुप्रवाह में; किनारे केंद्र से पतले), और \emph{अपसारी} यदि $f' < 0$ हो।
उसकी \emph{अभिसारिता} $V = 1/f'$ \emph{डाइऑप्टर} में नापी जाती है
($\qty{1}{\delta} = \qty{1}{m^{-1}}$)।
\end{definition}

\begin{theorem}[पतले लेंस के लिए संयुग्मन]\label{thm:b1:mirrors-thin-lenses:lensconj}
\index{संयुग्मन संबंध!पतला लेंस}\index{देकार्त संबंध}\index{न्यूटन संबंध}
\omterm{def:b1:mirrors-thin-lenses:gauss}{गाउस की शर्तों} में, \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $A'$ अक्ष पर स्थित अपने वस्तु बिंदु $A$ के लिए
देकार्त के संबंध का पालन करता है:
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'},
\qquad
\gamma = \frac{\overline{OA'}}{\overline{OA}},
\]
और, उसी के तुल्य, न्यूटन के संबंध का:
\[
\overline{FA}\cdot\overline{F'A'} = -f'^2,
\qquad
\gamma = \frac{f'}{\overline{FA}} = -\frac{\overline{F'A'}}{f'} .
\]
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
मुख्य तथ्य यह है कि \omterm{def:b1:mirrors-thin-lenses:lens}{पतला लेंस} ऊँचाई $h$ पर उसे पार करती किसी किरण को अक्ष की
ओर कोण $h/f'$ से विचलित कर देता है, चाहे किरण की दिशा कुछ भी हो
--- \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} का हर क्षेत्र एक पतले प्रिज़्म की तरह बरतता है
(\cref{rem:b1:rays-reflection-refraction:thin}), जिसका शीर्ष कोण गोलीय
पृष्ठों के लिए $h$ के साथ रैखिक रूप से बढ़ता है; उपाक्षीय किरणों के लिए यथार्थ
यह रैखिकता यहाँ मान ली गई है और वर्ष 2 के खंड में दो गोलीय पृष्ठों पर स्नेल के
नियम से व्युत्पन्न की जाती है। इसे मान लेने पर: $A$ से चली किरण (वास्तविक
वस्तु के लिए दूरी $p = -\overline{OA} > 0$) \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} तक ऊँचाई $h$ पर
ढाल $h/p$ के साथ पहुँचती है, $h/f'$ से मुड़ती है और ढाल $h/p - h/f'$ के साथ
निकल जाती है; वह अक्ष को $A'$ पर काटती है, जहाँ $h/\overline{OA'} = h/f' - h/p$, यानी $1/\overline{OA'}
- 1/\overline{OA} = 1/f'$, और यह $h$ से स्वतंत्र है: अतः $A$ से चली सारी
किरणें $A'$ पर मिलती हैं। केंद्र से होकर जाती किरण $B \to O \to B'$ सीधी है,
अतः $\overline{A'B'}/\overline{AB} = \overline{OA'}/\overline{OA}$। न्यूटन का
रूप $\overline{OA} = \overline{OF} + \overline{FA}
= -f' + \overline{FA}$ और $\overline{OA'} = f' + \overline{F'A'}$ रखने पर
निकल आता है।
\end{proof}

\begin{method}[पतले लेंस से प्रतिबिंब की रचना]\label{met:b1:mirrors-thin-lenses:lensrays}
$B$ से चलती तीन उल्लेखनीय किरणें:
\begin{enumerate}
  \item $O$ से होकर: अविचलित;
  \item अक्ष के समांतर: $F'$ से होकर निकलती है;
  \item $F$ से होकर: अक्ष के समांतर निकलती है।
\end{enumerate}
दो ही काफ़ी हैं; \omterm{def:b1:mirrors-thin-lenses:lens}{अपसारी लेंस} के लिए विस्तार काम में लीजिए (\omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के पीछे $F$ की
ओर तानी गई किरण समांतर निकलती है; और समांतर किरण ऐसे निकलती है मानो सामने के
$F'$ से आ रही हो)।
\end{method}

\begin{proposition}[अनुदैर्घ्य आवर्धन]\label{prop:b1:mirrors-thin-lenses:longitudinal}
\index{अनुदैर्घ्य आवर्धन}
यदि वस्तु अक्ष के अनुदिश $\dd\overline{OA}$ खिसके, तो \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} उसी दिशा में
$\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$ खिसकता
है।
\end{proposition}

\begin{proof}
देकार्त के संबंध का $f'$ नियत रखकर अवकलन कीजिए:
$-\dd\overline{OA'}/\overline{OA'}^2 + \dd\overline{OA}/\overline{OA}^2 = 0$,
अतः $\dd\overline{OA'} = (\overline{OA'}/\overline{OA})^2\,\dd\overline{OA}$।
\end{proof}

\begin{remark}[फोकस की गहराई]\label{rem:b1:mirrors-thin-lenses:depth}
$\gamma = -50$ वाले प्रक्षेपक के लिए $\gamma^2 = 2500$: दीपक की गर्मी से
\qty{0.1}{mm} टेढ़ी हुई स्लाइड अपने \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} को परदे के तल से
\qty{25}{cm} बाहर फेंक देती है। इसके उलट कैमरा ($\abs\gamma
\ll 1$) \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} के नन्हे खिसकाव के बदले वस्तु के बड़े खिसकाव सह लेता है ---
यही क्षेत्र-गहराई है।
\end{remark}

\begin{method}[स्वसंरेखण]\label{met:b1:mirrors-thin-lenses:autocollimation}
\index{स्वसंरेखण}
\omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के ठीक पीछे एक \omterm{prop:b1:rays-reflection-refraction:mirror}{समतल दर्पण} रखिए और किसी प्रकाशित वस्तु (धुँधले काँच पर बने
क्रॉस) को अक्ष के अनुदिश तब तक खिसकाइए जब तक \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} से होकर लौटा उसका \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब}
वस्तु पर ही तीखा, उलटा और उसी आकार का न बन जाए। तब वस्तु \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} तल में है:
किरणें \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} से समांतर निकलती हैं, दर्पण से समांतर ही लौटती हैं, और फिर \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस}
तल में केंद्रित हो जाती हैं। वस्तु--लेंस दूरी ही $f'$ है।
\end{method}

\begin{proposition}[बेसेल की विधि]\label{prop:b1:mirrors-thin-lenses:bessel}
\index{बेसेल की विधि}\index{सिल्बरमान की विधि}
कोई वस्तु और कोई परदा एक-दूसरे से दूरी $D$ पर जड़े हैं। \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारी लेंस} परदे पर
तीखा \omterm{def:b1:mirrors-thin-lenses:image}{वास्तविक प्रतिबिंब} ठीक दो स्थितियों में बनाता है यदि $D > 4f'$ हो, एक
में यदि $D = 4f'$ हो, और किसी में नहीं यदि $D < 4f'$ हो। दोनों स्थितियाँ
आपस में दूरी $d$ पर हैं, जहाँ
\[
f' = \frac{D^2 - d^2}{4D} .
\]
$D = 4f'$ पर (सिल्बरमान) इकलौती स्थिति $\gamma = -1$ देती है।
\end{proposition}

\begin{proof}
मान लीजिए वस्तु--लेंस दूरी $p = -\overline{OA} > 0$ है; तब
$\overline{OA'} = D - p$ और देकार्त देता है $1/(D - p) + 1/p = 1/f'$,
यानी $p^2 - Dp + Df' = 0$: विविक्तकर $D^2 - 4Df'$, जो तभी धनात्मक है जब
$D > 4f'$ हो। दोनों मूल $p_{1,2} = (D \mp \sqrt{D^2 - 4Df'})/2$ जुड़कर
$D$ देते हैं (स्थितियाँ सममित हैं: वस्तु और \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} की दूरियाँ आपस में बदल
जाती हैं) और उनका अंतर $d = \sqrt{D^2 - 4Df'}$ है; अब $f'$ के लिए हल कीजिए।
$D = 4f'$ पर $p = 2f' = \overline{OA'}$ और $\gamma = -1$।
\end{proof}

\section{दो पतले लेंस}

\begin{proposition}[सटे हुए लेंस; अफोकल युग्म]\label{prop:b1:mirrors-thin-lenses:doublet}
\index{अफोकल निकाय}\index{कोणीय आवर्धन}
आपस में सटे दो पतले \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} \omterm{def:b1:mirrors-thin-lenses:lens}{अभिसारिता} $V = V_1 + V_2$ के एक ही पतले \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} की तरह
बरतते हैं। और ऐसे रखे गए दो \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} कि $F_1' = F_2$ हो, एक
\emph{अफोकल} निकाय बनाते हैं: समांतर पुंज समांतर ही निकलता है, और $\alpha$
से झुका पुंज $\alpha' = G\alpha$ से झुका निकलता है, जहाँ
\emph{कोणीय आवर्धन} $G = -f_1'/f_2'$ है।
\end{proposition}

\begin{proof}
\omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $A_1$, जो $L_1$ देता है, $L_2$ के लिए वस्तु है। सटे होने पर
$O_1 = O_2 = O$: $1/\overline{OA_1} - 1/\overline{OA} = V_1$ और
$1/\overline{OA'} - 1/\overline{OA_1} = V_2$; अब जोड़ दीजिए। अफोकल स्थिति में:
कोण $\alpha$ पर आता समांतर पुंज साझे \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} तल में अक्ष से ऊँचाई
$h = f_1'\alpha$ पर केंद्रित होता है ($O_1$ से होकर जाती किरण); $L_2$ की
दृष्टि से यह बिंदु उसके वस्तु \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} तल में है, अतः पुंज समांतर निकलता है, उस
किरण के कोण पर जो उस बिंदु से $O_2$ से होकर जाती है: $\alpha' = -h/f_2'$।
\end{proof}

\begin{example}[एक पंक्ति में दूरदर्शी]\label{ex:b1:mirrors-thin-lenses:afocal}
$f_1' = \qty{1000}{mm}$, $f_2' = \qty{25}{mm}$: $G = -40$; और आधा अंश चौड़ा
चंद्रमा $20^\circ$ की चकती बन जाता है --- यही अगले अध्याय का खगोलीय
दूरदर्शी है। अपसारी नेत्रिका के साथ,
$f_2' = \qty{-25}{mm}$, $G = +40$: सीधा, यानी गैलीलियो की दूरबीन।
\end{example}

\section{समस्या: स्लाइड प्रक्षेपक}

\begin{problem}[{सप्ताहांत समस्या --- एक लेंस, प्रयोग-मेज़ पर नापा गया, और
फिर एक हॉल के पार 35-mm स्लाइड फेंकने को कहा गया: प्रक्षेपक की अभिकल्पना,
फोकसन, और उस पर चढ़ने वाला दीर्घ-प्रक्षेप परिवर्तक}]
\label{pb:b1:mirrors-thin-lenses:1}
प्रक्षेपण \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} $L_1$ अज्ञात \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} $f_1'$ वाला अभिसारी \omterm{def:b1:mirrors-thin-lenses:lens}{पतला लेंस} है।
स्लाइड \qty{24}{mm} ऊँची और \qty{36}{mm} चौड़ी है।

\medskip
\noindent\textbf{भाग I --- प्रयोग-मेज़ पर \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस}।}
\begin{enumerate}
  \item पतले \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के लिए देकार्त का संबंध और आवर्धन, चिह्न-रूढ़ियों सहित,
        लिखिए।
  \item वस्तु और परदा एक-दूसरे से दूरी $D$ पर जड़े हैं। दिखाइए कि परदे पर
        \omterm{def:b1:mirrors-thin-lenses:image}{वास्तविक प्रतिबिंब} के लिए $D \geq 4f_1'$ आवश्यक है।
  \item दिखाइए कि जब $D > 4f_1'$ हो, तब तीखा \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} देने वाली दोनों
        लेंस-स्थितियाँ वस्तु और परदे के मध्यबिंदु के सापेक्ष सममित होती हैं,
        और $f_1' = (D^2 - d^2)/4D$, जहाँ $d$ उनका अंतराल है।
  \item $D = \qty{80.0}{cm}$ और $d = \qty{40.0}{cm}$ के साथ
        $f_1'$ निकालिए।
  \item \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} एक मोटी नली में बैठा है, जिसका \omterm{def:b1:mirrors-thin-lenses:lens}{प्रकाशिक केंद्र} ठीक-ठीक नहीं
        ढूँढ़ा जा सकता। इससे यह विधि प्रभावित क्यों नहीं होती?
  \item दूरियों के साथ $u(D) = \qty{0.2}{cm}$ और
        $u(d) = \qty{0.5}{cm}$ हैं। $u(f_1')$ निकालिए और परिणाम लिखिए।
  \item दोनों स्थितियों में से हर एक पर आवर्धन दीजिए। इनमें से कौन-सी
        प्रक्षेपक से मेल खाती है?
  \item वह सबसे छोटा $D$ कौन-सा है जिसके लिए तीखा \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} बनता है, और
        तब आवर्धन कितना होता है (सिल्बरमान)?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- चित्र फेंकना।}
परदे पर चित्र \qty{1.80}{m} चौड़ा होना चाहिए।
$f_1' = \qty{15.0}{cm}$ लीजिए।
\begin{enumerate}[resume]
  \item कितना आवर्धन चाहिए (चिह्न सहित)?
  \item लेंस--परदा दूरी और स्लाइड--लेंस दूरी निकालिए।
  \item स्लाइड को उलटा और दर्पण-उलट क्यों डाला जाता है?
  \item परदा \qty{10.0}{m} पर हटा दिया जाता है। फिर से \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} करने के लिए
        स्लाइड के सापेक्ष \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} को कितना और किस दिशा में खिसकाना होगा? अब
        चित्र कितना चौड़ा है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} की सह्यता और प्रकाश।}
\begin{enumerate}[resume]
  \item दिखाइए कि स्लाइड के छोटे खिसकाव $\dd\overline{OA}$ से \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब}
        $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$ खिसकता है।
  \item दीपक की गर्मी से फ़िल्म \qty{0.10}{mm} टेढ़ी हो जाती है। प्रतिबिंब-तल
        कितना खिसकता है (परदा \qty{7.65}{m} पर)? टिप्पणी कीजिए।
  \item \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के द्वारक का व्यास \qty{30}{mm} है। यदि परदा प्रतिबिंब-तल से
        \qty{20}{cm} हटकर बैठा हो, तो स्लाइड के किसी बिंदु के धुँधलेपन के
        धब्बे का व्यास कितना होगा? \qty{5}{m} से देखने पर क्या आँख (कोणीय
        विभेदन लगभग $\qty{3e-4}{rad}$) उसे पकड़ लेगी?
  \item स्लाइड को \qty{2.0}{W} प्रकाश मिलता है, और वह सारा परदे तक पहुँचता
        है। स्लाइड पर और परदे पर विकिरण-तीव्रता (प्रति एकांक क्षेत्रफल
        शक्ति) निकालिए, और दिखाइए कि उनका अनुपात $\gamma^2$ है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दीर्घ-प्रक्षेप परिवर्तक।}
किसी बड़े हॉल में परदा \qty{12.0}{m} दूर है। एक \omterm{def:b1:mirrors-thin-lenses:lens}{अपसारी लेंस}
$L_2$, $f_2' = \qty{-30.0}{cm}$, $L_1$ से \qty{8.0}{cm} पीछे चढ़ा दिया
जाता है; परदा $L_2$ से \qty{12.0}{m} दूर है।
\begin{enumerate}[resume]
  \item अंतिम \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} के वास्तविक और परदे पर बनने के लिए, ($L_2$ से) वह
        मध्यवर्ती \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} कहाँ पड़ना चाहिए जो $L_1$ देता है? वह $L_2$ के
        लिए वास्तविक वस्तु है या आभासी?
  \item इससे स्लाइड की स्थिति $\overline{O_1A}$ निकालिए।
  \item $\gamma_1$, $\gamma_2$ और कुल आवर्धन निकालिए।
  \item चित्र की चौड़ाई दीजिए, और उसकी तुलना उससे कीजिए जो अकेला $L_1$ उसी
        परदे पर ($L_1$ से \qty{12.08}{m}) देता।
  \item अनंत पर रखी वस्तु के लिए दिखाइए कि यह निकाय \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी}
        $f_{\mathrm{eq}}$ वाले एक ही \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के तुल्य है, जहाँ
        \[
        \frac{1}{f_{\mathrm{eq}}} = \frac{1}{f_1'} + \frac{1}{f_2'} - \frac{e}{f_1' f_2'},
        \]
        $e = \overline{O_1O_2}$, इस अर्थ में कि $\alpha$ से झुका पुंज
        अक्ष से ऊँचाई $f_{\mathrm{eq}}\,\alpha$ पर केंद्रित होता है।
        $f_{\mathrm{eq}}$ निकालिए।
  \item जाँचिए कि चित्र-चौड़ाइयों का अनुपात (परिवर्तक के साथ/बिना)
        $f_1'/f_{\mathrm{eq}}$ के पास है। यह केवल लगभग ही क्यों है?
  \item अंतिम \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} के वास्तविक होने के लिए मध्यवर्ती \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} को
        $\abs{f_2'}$ से कम दूरी पर, $L_2$ के आगे, क्यों पड़ना चाहिए?
  \item उसी दीपक के साथ, परिवर्तक चढ़ाने पर परदे की विकिरण-तीव्रता कितने
        गुना बदल जाती है (परदे की दूरी वही)?
  \item सार लिखिए: परिवर्तक \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} को कितने गुना कर देता है, और किसी दिए
        गए प्रक्षेप पर वह चित्र के आकार तथा चमक के साथ क्या करता है?
\end{enumerate}
\end{problem}
