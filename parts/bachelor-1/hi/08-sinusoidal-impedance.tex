\chapter{ज्यावक्रीय स्थायी व्यवस्था और प्रतिबाधा}\label{ch:b1:sinusoidal-impedance}

किसी पुराने रेडियो का घुंडी घुमाइए और उन सौ केंद्रों में से, जो सब एक साथ
ऐंटेना तक पहुँचते हैं, एक ही सुनाई देने लगता है: एक कुंडली और एक परिवर्ती
संधारित्र इस तरह सुर में बिठा दिए गए हैं कि ठीक एक ही आवृत्ति उन्हें अनुनाद
में ले आती है। हर दीवार में गुनगुनाती बिजली \qty{50}{Hz} की ज्या है; और जो
कारख़ाना उस पर बड़ी मोटरें चलाता है, यदि उसकी धारा \omterm{def:b1:dc-circuits:voltage}{विभवांतर} से बहुत पिछड़ जाए तो
उसे जुर्माना भरना पड़ता है। ज्याएँ रैखिक परिपथों की स्वाभाविक भाषा हैं --- वे
जैसी घुसती हैं वैसी ही निकलती हैं, बस पैमाना और खिसकाव बदलकर --- और यह अध्याय
उन्हें उनका बीजगणित देता है: \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम}, प्रतिबाधाएँ, \omterm{prop:b1:transient-regimes:rlc}{RLC परिपथ} का अनुनाद,
और वह शक्ति जो ज्यावक्रीय धाराएँ सचमुच पहुँचाती हैं।

\section{ज्यावक्रीय स्थायी व्यवस्था}

\begin{proposition}[प्रणोदित ज्यावक्रीय व्यवस्था]\label{prop:b1:sinusoidal-impedance:forced}
\index{ज्यावक्रीय स्थायी व्यवस्था}\index{प्रणोदित व्यवस्था}
\omterm{def:b1:wave-propagation:signal}{कोणीय आवृत्ति} $\omega$ के किसी ज्यावक्रीय स्रोत से चलाए जाते \omterm{def:b1:dc-circuits:linear}{रैखिक परिपथ} में,
क्षणिक अवस्थाएँ मर जाने के बाद, हर धारा और हर \omterm{def:b1:dc-circuits:voltage}{विभवांतर} उसी \omterm{def:b1:wave-propagation:signal}{कोणीय आवृत्ति}
$\omega$ की ज्या होता है, और हर एक का अपना
\omterm{def:b1:wave-propagation:signal}{आयाम} तथा अपनी \omterm{def:b1:wave-propagation:signal}{कला} होती है: $x(t) = X\cos(\omega t + \varphi)$।
\end{proposition}

\begin{proof}
परिपथ के समीकरण नियत गुणांकों वाले रैखिक अवकल समीकरण हैं; उनका सामान्य हल कोई
एक विशेष हल जमा समांगी समीकरण के हल है, और वे हल
\cref{ch:b1:transient-regimes} की क्षणिक अवस्थाएँ हैं, जो क्षीण हो जाती हैं
(हर वास्तविक परिपथ में कुछ न कुछ प्रतिरोध होता है)। कोई ज्यावक्रीय विशेष हल
इसलिए मिलता है कि आवृत्ति $\omega$ की ज्या का अवकलन उसी आवृत्ति की ज्या
देता है: अतः समीकरणों को पद-दर-पद संतुष्ट किया जा सकता है --- और नीचे की
सम्मिश्र विधि उसे स्पष्ट रूप से रच देती है।
\end{proof}

\begin{definition}[सम्मिश्र आयाम]\label{def:b1:sinusoidal-impedance:complex}
\index{सम्मिश्र आयाम}\index{कला-सदिश}
ज्या $x(t) = X\cos(\omega t + \varphi)$ के साथ
सम्मिश्र संकेत $\underline{x}(t) = X\eu^{j(\omega t + \varphi)}$ जोड़ा जाता है, ताकि
$x = \Rea(\underline{x})$ हो, और \emph{सम्मिश्र \omterm{def:b1:wave-propagation:signal}{आयाम}}
\[
\underline{X} = X\eu^{j\varphi}, \qquad \underline{x}(t) = \underline{X}\,\eu^{j\omega t},
\]
जो \omterm{def:b1:wave-propagation:signal}{आयाम} $X = \abs{\underline{X}}$ और \omterm{def:b1:wave-propagation:signal}{कला}
$\varphi = \arg\underline{X}$, दोनों ढोता है। यहाँ $j$ काल्पनिक इकाई को
दर्शाता है, $j^2 = -1$, क्योंकि अक्षर $i$ धाराओं के लिए सुरक्षित है। लंबाई $X$ के सदिश के रूप में, कोण
$\varphi$ पर, खींचा गया $\underline{X}$ एक
\emph{कला-सदिश} (फ़्रेनल सदिश) है।
\end{definition}

\begin{proposition}[सम्मिश्र विधि के नियम]\label{prop:b1:sinusoidal-impedance:rules}
\begin{enumerate}
  \item रैखिक संक्रियाएँ वास्तविक भाग लेने के साथ क्रम बदल सकती हैं: किसी
        योग का \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम} सम्मिश्र आयामों का योग है।
  \item समय के सापेक्ष अवकलन \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम} को
        $j\omega$ से गुणा कर देता है; और समाकलन उसे $j\omega$ से भाग देता है।
  \item \omterm{thm:b1:dc-circuits:kirchhoff}{किरचॉफ़ के नियम}, रैखिक होने के कारण, सम्मिश्र आयामों पर भी टिकते हैं।
\end{enumerate}
\end{proposition}

\begin{proof}
$\Rea(\underline a + \underline b) = \Rea\underline a + \Rea\underline b$;
$\dd(\underline X\eu^{j\omega t})/\dd t = j\omega\underline X\eu^{j\omega t}$,
और $\dd x/\dd t = \Rea(\dd\underline x/\dd t)$, क्योंकि अवकलन
वास्तविक-रैखिक है। किसी संधि पर धाराओं के और किसी पाश के चारों ओर विभवांतरों
के योग रैखिक हैं।
\end{proof}

\begin{example}[दो ज्याओं को जोड़ना]\label{ex:b1:sinusoidal-impedance:add}
$3\cos\omega t + 4\sin\omega t$: \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम} $3$ और
$4\eu^{-j\pi/2} = -4j$ हैं (क्योंकि $\sin\omega t = \cos(\omega t - \pi/2)$);
उनके योग $3 - 4j$ का मापांक $5$ और कोणांक $-0.927\ \unit{rad}$ है:
अतः $5\cos(\omega t - 0.927)$। किसी त्रिकोणमितीय सर्वसमिका की ज़रूरत नहीं।
\end{example}

\section{प्रतिबाधा}

\begin{definition}[प्रतिबाधा, प्रवेश्यता]\label{def:b1:sinusoidal-impedance:impedance}
\index{प्रतिबाधा}\index{प्रवेश्यता}\index{प्रतिघात}
\omterm{prop:b1:sinusoidal-impedance:forced}{ज्यावक्रीय स्थायी व्यवस्था} में किसी \omterm{def:b1:dc-circuits:linear}{रैखिक द्विसिरा} के लिए (\omterm{thm:b1:dc-circuits:kirchhoff}{अभिग्राही
रूढ़ि} में) \emph{प्रतिबाधा} सम्मिश्र आयामों का अनुपात है:
\[
\underline Z = \frac{\underline U}{\underline I} = Z\eu^{j\varphi}
\quad (\text{ओम}),
\]
जिसका मापांक $Z = U/I$ आयामों को जोड़ता है और जिसका कोणांक
$\varphi = \varphi_u - \varphi_i$ धारा पर \omterm{def:b1:dc-circuits:voltage}{विभवांतर} की कला-अग्रता है। उसका
व्युत्क्रम $\underline Y = 1/\underline Z$ \emph{प्रवेश्यता} है (सीमेंस में)।
$\underline Z$ का वास्तविक भाग उसका प्रतिरोध है, और काल्पनिक भाग उसका
\emph{प्रतिघात}।
\end{definition}

\begin{proposition}[R, L, C की प्रतिबाधाएँ]\label{prop:b1:sinusoidal-impedance:rlc}
\[
\underline Z_R = R, \qquad \underline Z_L = jL\omega, \qquad
\underline Z_C = \frac{1}{jC\omega} = -\frac{j}{C\omega} .
\]
\omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} \omterm{def:b1:dc-circuits:voltage}{विभवांतर} और धारा को समकला में रखता है; कुंडली में \omterm{def:b1:dc-circuits:voltage}{विभवांतर} धारा से
$\pi/2$ आगे रहता है; और संधारित्र में वह $\pi/2$ पीछे। कम
आवृत्ति पर कुंडली तार जैसी हो जाती है और संधारित्र खुला परिपथ; और ऊँची
आवृत्ति पर उलटा।
\end{proposition}

\begin{proof}
$u = Ri$; $u = L\,\dd i/\dd t \to \underline U = jL\omega\,\underline I$;
$i = C\,\dd u/\dd t \to \underline I = jC\omega\,\underline U$। और
$1$, $j$ तथा $-j$ के कोणांक $0$, $\pi/2$, $-\pi/2$ हैं।
\end{proof}

\begin{proposition}[संयोजन; विभाजक; प्रमेयें]\label{prop:b1:sinusoidal-impedance:assoc}
श्रेणी में प्रतिबाधाएँ जुड़ती हैं; समांतर में प्रवेश्यताएँ जुड़ती हैं; और
\cref{ch:b1:dc-circuits} के विभव तथा
\omterm{prop:b1:dc-circuits:dividers}{धारा विभाजक}, तेवेनिन और नॉर्टन की प्रमेयें, अध्यारोपण तथा
\omterm{prop:b1:dc-circuits:millman}{मिलमैन की प्रमेय} --- ये सब सम्मिश्र आयामों और प्रतिबाधाओं के साथ भी टिकते
हैं, वास्तविक विभवांतरों, धाराओं और प्रतिरोधों की जगह।
\end{proposition}

\begin{proof}
\cref{ch:b1:dc-circuits} की हर उपपत्ति ने केवल \omterm{thm:b1:dc-circuits:kirchhoff}{किरचॉफ़ के नियम} और
$u = Ri$ की रैखिकता का उपयोग किया था, और दोनों सम्मिश्र रूप में भी टिकते हैं।
\end{proof}

\begin{example}[RC विभाजक]\label{ex:b1:sinusoidal-impedance:rcdivider}
एक \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} $R$ और एक संधारित्र $C$, श्रेणी में $\underline E$ के आरपार, और
निर्गम $C$ के आरपार: $\underline U_C = \underline E\,\dfrac{1/jC\omega}
{R + 1/jC\omega} = \dfrac{\underline E}{1 + jRC\omega}$। \omterm{def:b1:wave-propagation:signal}{आयाम}
$E/\sqrt{1 + (RC\omega)^2}$, \omterm{def:b1:wave-propagation:signal}{कला} $-\arctan(RC\omega)$: यानी कम आवृत्ति पर
पूरा संचरण, $1/\sqrt2$ और $-45^\circ$ तब जब $\omega = 1/RC$ हो,
और उसके परे $1/\omega$ जैसी गिरावट --- यानी निम्न-पारक फ़िल्टर, जो
\cref{ch:b1:filters-transfer-functions} का विषय है।
\end{example}

\section{श्रेणी RLC परिपथ: अनुनाद}

\begin{theorem}[धारा का अनुनाद]\label{thm:b1:sinusoidal-impedance:resonance}
\index{अनुनाद}\index{अनुनाद!धारा का}\index{बैंड-चौड़ाई}
\omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} $R$, कुंडली $L$ और संधारित्र $C$, जो $e(t) = E\cos\omega t$ के
आरपार श्रेणी में जुड़े हों, जो धारा ढोते हैं उसका \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम} है
\[
\underline I = \frac{E}{R + j\left(L\omega - \dfrac{1}{C\omega}\right)},
\qquad
I = \frac{E}{\sqrt{R^2 + \left(L\omega - \dfrac{1}{C\omega}\right)^2}},
\qquad
\varphi_i = -\arctan\frac{L\omega - 1/C\omega}{R} .
\]
\omterm{def:b1:wave-propagation:signal}{आयाम} अधिकतम, $I_{\max} = E/R$, तब होता है जब धारा और \omterm{def:b1:dc-circuits:voltage}{विभवांतर} समकला में हों,
यानी \emph{अनुनाद} $\omega_0 = 1/\sqrt{LC}$ पर।
$Q = L\omega_0/R = 1/(RC\omega_0)$ और $x = \omega/\omega_0$ के साथ,
\[
\frac{I}{I_{\max}} = \frac{1}{\sqrt{1 + Q^2\left(x - \dfrac1x\right)^2}} ,
\]
और \emph{बैंड-चौड़ाई} --- $\omega$ का वह अंतराल जिस पर
$I \geq I_{\max}/\sqrt2$ रहे --- है
\[
\Delta\omega = \omega_2 - \omega_1 = \frac{\omega_0}{Q} = \frac{R}{L} .
\]
\end{theorem}

\begin{proof}
श्रेणी में प्रतिबाधाएँ जुड़ती हैं: $\underline Z = R + jL\omega + 1/jC\omega$, और
$\underline I = \underline E/\underline Z$। मापांक तब सबसे बड़ा होता है जब
\omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिघात} $L\omega - 1/C\omega$ लुप्त हो जाए, यानी $\omega_0$ पर।
$L\omega - 1/C\omega = L\omega_0(x - 1/x)$ गुणनखंडित कीजिए और $R$ से भाग दीजिए:
$L\omega_0/R = Q$। बैंड के सिरों पर $Q(x - 1/x) = \pm 1$, यानी
$x^2 \mp x/Q - 1 = 0$, जिसके धनात्मक मूल $x_{1,2} = \mp\frac{1}{2Q}
+ \sqrt{1 + \frac{1}{4Q^2}}$ हैं; और उनका अंतर $1/Q$ है।
\end{proof}

\begin{proposition}[विभव-आवर्धन]\label{prop:b1:sinusoidal-impedance:overvoltage}
\index{अनुनाद!विभव का}\index{अधिविभव}
अनुनाद पर कुंडली और संधारित्र के आरपार विभवांतरों का \omterm{def:b1:wave-propagation:signal}{आयाम}
समान $QE$ होता है और उनकी कलाएँ विपरीत: अतः $Q \gg 1$ के लिए संधारित्र
स्रोत से $Q$ गुना बड़ा \omterm{def:b1:dc-circuits:voltage}{विभवांतर} ढोता है। \omterm{def:b1:wave-propagation:signal}{आयाम}
$U_C(\omega)$ ख़ुद भी, $Q > 1/\sqrt2$ के लिए, $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$ पर शिखर पर आता है, जो $\omega_0$ से ज़रा नीचे है,
और वहाँ $U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$।
\end{proposition}

\begin{proof}
$\omega_0$ पर $\underline U_C = \underline I/jC\omega_0 = -jQE$ और
$\underline U_L = jL\omega_0\underline I = jQE$, क्योंकि $I = E/R$।
सामान्य रूप में $U_C = I/C\omega = E/\sqrt{(1 - x^2)^2 + x^2/Q^2}$ (अंश और
हर को $1/C\omega_0$ से गुणा कीजिए); हर का वर्ग
$(1 - x^2)^2 + x^2/Q^2$ है, जो $x^2$ में द्विघात है और $x^2 =
1 - 1/2Q^2$ पर न्यूनतम होता है, जब वह धनात्मक हो, और न्यूनतम $1/Q^2 - 1/4Q^4$ है।
\end{proof}

\begin{example}[रेडियो सुर में बिठाना]\label{ex:b1:sinusoidal-impedance:radio}
एक कुंडली $L = \qty{200}{\micro H}$ और एक परिवर्ती संधारित्र: $C =
\qty{127}{pF}$ $f_0 = \qty{1.00}{MHz}$ पर सुर बिठा देता है। कुंडली के तार के
प्रतिरोध $R = \qty{12.6}{\ohm}$ के साथ $L\omega_0 = \qty{1257}{\ohm}$ और
$Q = 100$: अतः बैंड-चौड़ाई $\Delta f = f_0/Q = \qty{10}{kHz}$ --- यानी किसी
AM चैनल की चौड़ाई --- और किसी दूर के प्रेषक से प्रेरित \qty{10}{\micro V}
संधारित्र के आरपार $QE = \qty{1}{mV}$ बन जाते हैं, बिना कोई क़ीमत
चुकाए। सप्ताहांत समस्या यही अभिग्राही बनाती है।
\end{example}

\begin{remark}[$Q$ फिर से क्यों]\label{rem:b1:sinusoidal-impedance:Q}
इस अध्याय का $Q$ वही $Q$ है जो \cref{ch:b1:transient-regimes} का है:
जो परिपथ चोट खाकर लगभग $Q$ दोलनों तक झनझनाता है, वही चलाए जाने पर
$\omega_0/Q$ चौड़े बैंड पर अनुनाद करता है। दोनों एक ही अनुपात कहते हैं:
$2\pi$ गुना संचित ऊर्जा बटा प्रति आवर्तकाल क्षय हुई ऊर्जा
(\cref{exo:b1:sinusoidal-impedance:12})। तीखा अनुनाद और लंबी झनझनाहट एक ही
गुण हैं, और वैसी ही है धीमी अनुक्रिया: बैंड-चौड़ाई $\Delta\omega$ का कोई
फ़िल्टर थमने में लगभग $1/\Delta\omega$ कोटि का समय लेता है।
\end{remark}

\section{ज्यावक्रीय व्यवस्था में शक्ति}

\begin{definition}[वर्ग-माध्य-मूल मान]\label{def:b1:sinusoidal-impedance:rms}
\index{वर्ग-माध्य-मूल मान}\index{प्रभावी मान}
किसी आवर्ती संकेत $x(t)$ का \emph{वर्ग-माध्य-मूल} (या प्रभावी)
मान $X_{\mathrm{rms}} = \sqrt{\langle x^2\rangle}$ है, यानी एक
आवर्तकाल पर $x^2$ के काल-औसत का वर्गमूल। \omterm{def:b1:wave-propagation:signal}{आयाम} $X$ की किसी ज्या
के लिए $X_{\mathrm{rms}} = X/\sqrt2$। घरेलू बिजली के \qty{230}{V}
प्रभावी मान हैं; उनका \omterm{def:b1:wave-propagation:signal}{आयाम} \qty{325}{V} है।
\end{definition}

\begin{theorem}[औसत शक्ति]\label{thm:b1:sinusoidal-impedance:power}
\index{औसत शक्ति}\index{शक्ति गुणक}\index{प्रतिघाती शक्ति}
जिस द्विसिरा के लिए $u = U\cos\omega t$ और $i = I\cos(\omega t - \varphi)$ हो
(\omterm{thm:b1:dc-circuits:kirchhoff}{अभिग्राही रूढ़ि}), वह तात्क्षणिक शक्ति $p = ui$ पाता है, जिसका
औसत है
\[
P = \tfrac12\,UI\cos\varphi = U_{\mathrm{rms}}I_{\mathrm{rms}}\cos\varphi
= \tfrac12\Rea\!\big(\underline U\,\underline I^{\,*}\big) = \tfrac12\Rea(\underline Z)\,I^2,
\]
जहाँ $\cos\varphi$ \emph{\omterm{thm:b1:sinusoidal-impedance:power}{शक्ति गुणक}} है।
कोई \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} $\tfrac12 RI^2 = RI_{\mathrm{rms}}^2$ पाता है;
और कोई आदर्श कुंडली या संधारित्र ($\varphi = \pm\pi/2$)
कोई \omterm{thm:b1:sinusoidal-impedance:power}{औसत शक्ति} नहीं पाता --- वह आधे आवर्तकाल में ऊर्जा
संचित करता है और दूसरे आधे में लौटा देता है।
\end{theorem}

\begin{proof}
$p = UI\cos\omega t\cos(\omega t - \varphi) = \tfrac12 UI[\cos\varphi +
\cos(2\omega t - \varphi)]$; और दूसरा पद औसत में शून्य हो जाता है।
$\underline U = U$ तथा $\underline I = I\eu^{-j\varphi}$ के साथ
$\underline U\,\underline I^* = UI\eu^{j\varphi}$, जिसका वास्तविक भाग
$UI\cos\varphi$ है; और $\underline U = \underline Z\,\underline I$ से
$\underline U\,\underline I^* = \underline Z I^2$ मिलता है।
\end{proof}

\begin{example}[शक्ति गुणक का संशोधन]\label{ex:b1:sinusoidal-impedance:pf}
\qty{230}{V} प्रभावी पर चलती किसी कार्यशाला की मोटर \qty{2.0}{kW} खींचती है, जहाँ
$\cos\varphi = 0.70$ (प्रेरकीय): अतः $I_{\mathrm{rms}} = 2000/(230 \times
0.70) = \qty{12.4}{A}$, जबकि \omterm{thm:b1:sinusoidal-impedance:power}{शक्ति गुणक} $1$ होने पर \qty{8.7}{A} होती
--- अतिरिक्त धारा बिजली कंपनी के तारों को यों ही गरम करती है, और इसीलिए
जुर्माना लगता है। समांतर में लगा कोई संधारित्र कुंडली के चौथाई-आवर्तकाल वाले
ऊर्जा-झूले स्थानीय रूप से जुटा देता है: \emph{\omterm{thm:b1:sinusoidal-impedance:power}{प्रतिघाती शक्ति}} $P\tan\varphi =
\qty{2.0}{kvar}$ के लिए \qty{50}{Hz} पर $C = P\tan\varphi/(\omega U_{\mathrm{rms}}^2)
\approx \qty{120}{\micro F}$ चाहिए, और तब रेखा-धारा गिरकर
\qty{8.7}{A} रह जाती है।
\end{example}

\end{exercise}

\section{समस्या: अनुनादी रेडियो अभिग्राही}

\begin{problem}[{सप्ताहांत समस्या --- एक कुंडली, एक परिवर्ती संधारित्र और एक
ऐंटेना: कोई श्रेणी अनुनाद सौ में से एक केंद्र कैसे चुन लेता है, उसके मंद
संकेत को सौ गुना कैसे कर देता है, और उसे मनमाने ढंग से वरणात्मक क्यों नहीं
बनाया जा सकता}]
\label{pb:b1:sinusoidal-impedance:1}
ऐंटेना का प्रतिरूप एक आदर्श ज्यावक्रीय \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} $e(t) = E\cos\omega t$ है,
जो कुंडली $L = \qty{200}{\micro H}$ के साथ श्रेणी में है, जिसके तार का
प्रतिरोध $R = \qty{12.6}{\ohm}$ है, और साथ में परिवर्ती संधारित्र $C$।
अभिग्राही $C$ के आरपार \omterm{def:b1:dc-circuits:voltage}{विभवांतर} पढ़ता है। आगे (भाग IV में) ऐंटेना का
अपना प्रतिरोध $R_a = \qty{50}{\ohm}$ भी हिसाब में लिया जाता है।

\medskip
\noindent\textbf{भाग I --- अनुनादी धारा।}
\begin{enumerate}
  \item श्रेणी परिपथ की सम्मिश्र \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} लिखिए।
  \item धारा का \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम} दीजिए, फिर उसका \omterm{def:b1:wave-propagation:signal}{आयाम}
        $I(\omega)$ और उसकी \omterm{def:b1:wave-propagation:signal}{कला} $\varphi_i(\omega)$, $e$ के सापेक्ष।
  \item दिखाइए कि $I$ $\omega_0 = 1/\sqrt{LC}$ पर अधिकतम है; तब
        $I_{\max}$ और \omterm{def:b1:wave-propagation:signal}{कला} दीजिए।
  \item कौन-सी धारिता $f_0 = \qty{1.000}{MHz}$ पर सुर बिठाती है?
        $L\omega_0$ निकालिए।
  \item $Q = L\omega_0/R$ परिभाषित कीजिए और उसे निकालिए।
  \item दिखाइए कि $I/I_{\max} = 1/\sqrt{1 + Q^2(x - 1/x)^2}$, जहाँ
        $x = \omega/\omega_0$।
  \item अनुनाद से नीचे और ऊपर $\varphi_i$ का चिह्न बताइए और उसकी व्याख्या
        कीजिए (धारिता जैसा या प्रेरकत्व जैसा व्यवहार)।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वरणात्मकता।}
\begin{enumerate}[resume]
  \item वह शर्त लिखिए जो उन दो आवृत्तियों को परिभाषित करती है जिन पर
        $I = I_{\max}/\sqrt2$ हो।
  \item उसे हल कीजिए और दिखाइए कि $\Delta\omega = \omega_2 - \omega_1 =
        \omega_0/Q$।
  \item इस अभिग्राही की बैंड-चौड़ाई $\Delta f$ निकालिए।
  \item कोई दूसरा केंद्र \qty{1.010}{MHz} पर प्रसारित करता है और तीसरा
        \qty{1.020}{MHz} पर: हर एक के लिए आयाम-अनुपात $I/I_{\max}$ और
        उससे मिलता शक्ति-अनुपात डेसिबल में निकालिए।
  \item AM केंद्र आपस में \qty{9}{} से \qty{10}{kHz} दूर होते हैं। क्या यह
        अभिग्राही पर्याप्त वरणात्मक है? उसे क्या बेहतर बनाएगा?
  \item संधारित्र \qty{50}{} और \qty{500}{pF} के बीच बिठाया जा सकता है:
        अभिग्राही कौन-सा आवृत्ति-बैंड ढकता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विभव-आवर्धन।}
\begin{enumerate}[resume]
  \item संधारित्र के \omterm{def:b1:dc-circuits:voltage}{विभवांतर} का \omterm{def:b1:sinusoidal-impedance:complex}{सम्मिश्र आयाम} व्यक्त कीजिए और दिखाइए
        कि अनुनाद पर उसका \omterm{def:b1:wave-propagation:signal}{आयाम} $QE$ है।
  \item ऐंटेना का \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} $E = \qty{10}{\micro V}$ है: $I_{\max}$
        और अनुनाद पर संधारित्र का \omterm{def:b1:dc-circuits:voltage}{विभवांतर} निकालिए। टिप्पणी कीजिए।
  \item दिखाइए कि अनुनाद पर कुंडली के \omterm{def:b1:dc-circuits:voltage}{विभवांतर} का \omterm{def:b1:wave-propagation:signal}{आयाम} वही है और वह
        संधारित्र के \omterm{def:b1:dc-circuits:voltage}{विभवांतर} से \omterm{def:b1:wave-propagation:signal}{कला} में विपरीत है: तब स्रोत को क्या
        ``दिखता'' है?
  \item अनुनाद पर परिपथ में संचित ऊर्जा और प्रति आवर्तकाल क्षय हुई
        ऊर्जा निकालिए; जाँचिए कि उनका अनुपात गुणा
        $2\pi$ $Q$ है।
  \item \cref{ch:b1:transient-regimes} के अनुसार इस परिपथ का \omterm{def:b1:transient-regimes:firstorder}{कालांक}
        $\tau = 2Q/\omega_0$ क्या है? बहुत ऊँचा $Q$ आख़िरकार वाणी और संगीत
        को विकृत क्यों कर देता है (श्रव्य बैंड-चौड़ाई:
        AM के लिए लगभग \qty{5}{kHz})?
  \item यदि कुंडली के तार का प्रतिरोध दुगुना हो जाए (पतला तार), तो
        $Q$, बैंड-चौड़ाई और संधारित्र का \omterm{def:b1:dc-circuits:voltage}{विभवांतर} कैसे बदलेंगे?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- ऐंटेना का प्रतिरोध, और शक्ति।}
\begin{enumerate}[resume]
  \item श्रेणी में $R_a = \qty{50}{\ohm}$ के साथ $Q$,
        बैंड-चौड़ाई और $E = \qty{10}{\micro V}$ के लिए संधारित्र का \omterm{def:b1:dc-circuits:voltage}{विभवांतर}
        फिर से निकालिए। निष्कर्ष निकालिए।
  \item अनुनाद पर ऐंटेना के \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} से दी गई \omterm{thm:b1:sinusoidal-impedance:power}{औसत शक्ति} और
        $R_a$ में क्षय हुआ अंश निकालिए।
  \item अधिकतम-शक्ति प्रमेय $R_a$ के बराबर भार-प्रतिरोध माँगती:
        यह लक्ष्य वरणात्मकता से टकराव में क्यों है?
  \item वास्तविक अभिग्राही ऐंटेना को कुंडली के किसी टैप से जोड़ते हैं, या
        किसी छोटे युग्मन संधारित्र से। गुणात्मक रूप से समझाइए कि इससे क्या
        हासिल होता है।
  \item $P = \tfrac12 UI\cos\varphi$ का उपयोग करके दोनों अर्ध-शक्ति
        आवृत्तियों पर \omterm{thm:b1:sinusoidal-impedance:power}{शक्ति गुणक} निकालिए और नाम
        ``अर्ध-शक्ति'' समझाइए।
  \item अभिकल्पना का सार लिखिए: $Q$ क्या देता है, उसकी क़ीमत क्या है, और
        \qty{10}{kHz} चैनलों वाले \qty{1}{MHz} अभिग्राही के लिए कौन-सा मान
        चुना जाता है।
\end{enumerate}
\end{problem}
