\chapter{कोणीय संवेग}\label{ch:b1:angular-momentum}

बाँहें फैलाकर घूमती कोई नृत्य-स्केटर उन्हें भीतर खींच लेती है और, बिना किसी
के धक्के के, दुगुनी तेज़ी से नाचने लगती है। कोई दरवाज़ा हत्थे पर उँगली की
नोक से खुल जाता है और कब्ज़े के पास कंधे का ज़ोर भी सह जाता है। ग्रह बराबर
समय में बराबर क्षेत्रफल बुहारते हैं --- केप्लर ने यह चार सदी पहले देख लिया
था, उससे बहुत पहले कि कोई कारण बता पाता। और चंद्रमा, अपने उठाए ज्वारों से
खिंचता हुआ, हर वर्ष कुछ सेंटीमीटर पृथ्वी से दूर सरक जाता है, जबकि हमारे दिन
अगोचर रूप से लंबे होते जाते हैं। इन चारों के पीछे एक ही सदिश राशि है, \omterm{def:b1:angular-momentum:L}{कोणीय
संवेग}, और एक ही नियम: उसके परिवर्तन की दर बलों के \omterm{def:b1:angular-momentum:moment}{आघूर्ण} के बराबर होती है।
यह अध्याय दोनों को परिभाषित करता है, किसी कण के लिए और किसी निकाय के लिए वह
नियम सिद्ध करता है, और उसे लोलकों, कक्षाओं तथा घूमती चीज़ों पर काम में लाता
है।

\section{बल का आघूर्ण}

\begin{definition}[किसी बिंदु के परितः आघूर्ण; किसी अक्ष के परितः]\label{def:b1:angular-momentum:moment}
\index{बल का आघूर्ण}\index{बलाघूर्ण}\index{उत्तोलक भुजा}
किसी बिंदु $O$ के परितः उस बल $\vect F$ का \emph{आघूर्ण} (या
\emph{बलाघूर्ण}), जो $M$ पर लगाया गया है, वह सदिश है
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
जो $\vect{OM}$ और $\vect F$ के तल पर लंब है और जिसका परिमाण
$OM\cdot F\sin\alpha = F\,d$ है, जहाँ $d$ --- \emph{उत्तोलक भुजा} --- $O$ से
$\vect F$ की क्रिया-रेखा तक की दूरी है। किसी दिष्ट अक्ष $\Delta$ (एकांक
सदिश $\vect e_\Delta$) के परितः, जो $O$ से होकर जाती है, आघूर्ण वह अदिश
$\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$ है, जो $O$ के
चुनाव से स्वतंत्र है, चाहे वह $\Delta$ पर कहीं भी हो; और यदि $\vect F$
$\Delta$ के समांतर हो या उससे मिलता हो तो वह लुप्त हो जाता है। भिन्न
क्रिया-रेखाओं पर लगते दो विपरीत बल एक \emph{बल-युग्म} बनाते हैं, जिसका
आघूर्ण $\vect F\wedge\vect{AB}$ हर बिंदु के परितः एक-सा होता है।
\end{definition}

\begin{example}[दरवाज़ा]\label{ex:b1:angular-momentum:door}
कब्ज़ों से \qty{0.80}{m} दूर दरवाज़े पर लंबवत लगाया गया \qty{20}{N} का धक्का
कब्ज़ा-अक्ष के परितः $\qty{16}{N.m}$ का \omterm{def:b1:angular-momentum:moment}{आघूर्ण} देता है; वही धक्का
\qty{0.10}{m} पर \qty{2}{N.m} का; और दरवाज़े के अनुदिश (क्रिया-रेखा कब्ज़ों
से होकर) शून्य। कब्ज़े की अपनी प्रतिक्रिया, जो अक्ष से मिलती है, उसके परितः
कोई \omterm{def:b1:angular-momentum:moment}{आघूर्ण} नहीं देती: यही बात अक्ष को \omterm{def:b1:angular-momentum:moment}{आघूर्ण} लेने की स्वाभाविक जगह बनाती है।
\end{example}

\section{किसी कण का कोणीय संवेग}

\begin{definition}[कोणीय संवेग]\label{def:b1:angular-momentum:L}
\index{कोणीय संवेग}
$O$ के परितः किसी ऐसे कण का \emph{कोणीय संवेग}, जिसका द्रव्यमान $m$ है, जो
$M$ पर है और जिसका वेग $\vect v$ है, यह होता है
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
और किसी अक्ष $\Delta$ के परितः, जो $O$ से होकर जाती है,
$L_\Delta = \vect L_O\cdot\vect e_\Delta$। $\Delta$ पर लंब किसी तल में की गति
के लिए, अक्ष के परितः ध्रुवीय निर्देशांकों में, $L_\Delta = mr^2\dot\theta$;
और $R$ त्रिज्या के वृत्त के लिए $L_\Delta = mR^2\omega = mRv$।
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$, $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[कोणीय संवेग प्रमेय]\label{thm:b1:angular-momentum:theorem}
\index{कोणीय संवेग प्रमेय}
किसी \omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} में, उसी तंत्र में स्थिर किसी बिंदु $O$ के लिए,
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
जहाँ $\Delta$ कोई स्थिर अक्ष है। यदि $O$ के परितः कुल \omterm{def:b1:angular-momentum:moment}{आघूर्ण} लुप्त हो जाए,
तो $\vect L_O$ संरक्षित रहता है।
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$, जिसमें न्यूटन का दूसरा नियम काम
में लिया गया है ($O$ स्थिर, ताकि $\dd\vect{OM}/\dd t = \vect v$)। अब $\vect e_\Delta$ पर प्रक्षेप कीजिए।
\end{proof}

\begin{corollary}[केंद्रीय बल: तलीय गति और क्षेत्रफल का नियम]\label{cor:b1:angular-momentum:central}
\index{केंद्रीय बल}\index{क्षेत्रफल का नियम}\index{क्षेत्रीय वेग}
केवल किसी \emph{केंद्रीय बल} के अधीन कोई कण (यानी ऐसा बल जो सदा
$\vect{OM}$ के अनुदिश हो, स्थिर केंद्र $O$ की ओर या उससे दूर)
$\vect L_O$ को अचर रखता है। उसकी गति $O$ से होकर जाते, $\vect L_O$ पर लंब
तल में ही बनी रहती है, और उस तल में $r^2\dot\theta = C$ (अचर): यानी
त्रिज्या-सदिश $\vect{OM}$ अचर \emph{\omterm{cor:b1:angular-momentum:central}{क्षेत्रीय वेग}} से क्षेत्रफल
बुहारता है
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$, क्योंकि $\vect F \parallel \vect{OM}$: अतः $\vect L_O$
अचर है। $\vect{OM}\perp\vect L_O$ सदा, इसलिए वह तल; और $L_O =
mr^2\dot\theta$ अचर है। $\dd t$ में बुहारा गया क्षेत्रफल वह त्रिभुज है
जिसकी भुजाएँ $\vect{OM}$ और $\vect v\dd t$ हैं: $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$।
\end{proof}

\begin{example}[आघूर्णों से लोलक]\label{ex:b1:angular-momentum:pendulum}
\omterm{ex:b1:newton-dynamics:pendulum}{सरल लोलक}, और धुरी से होकर जाती, झूलने के तल पर लंब अक्ष $\Delta$:
$L_\Delta = m\ell^2\dot\theta$; तनाव अक्ष से मिलता है (कोई \omterm{def:b1:angular-momentum:moment}{आघूर्ण} नहीं), और
भार का \omterm{def:b1:angular-momentum:moment}{आघूर्ण} $-mg\ell\sin\theta$ है (\omterm{def:b1:angular-momentum:moment}{उत्तोलक भुजा}
$\ell\sin\theta$, प्रत्यानयक)। प्रमेय से $m\ell^2\ddot\theta =
-mg\ell\sin\theta$ मिलता है, यानी \cref{ex:b1:newton-dynamics:pendulum} वाला
समीकरण, वह भी तनाव लिखे बिना --- आघूर्णों का यही सबसे बड़ा लाभ है: अक्ष से
होकर जाते बल ग़ायब हो जाते हैं।
\end{example}

\begin{example}[चकती को भीतर खींचना]\label{ex:b1:angular-momentum:puck}
कोई चकती घर्षणहीन बर्फ़ पर वृत्त में फिसलती है, और उसे केंद्र के किसी छेद
से होकर जाती डोरी थामे हुए है; डोरी इतनी खींची जाती है कि त्रिज्या आधी रह
जाए। तनाव केंद्रीय है: $L = mr^2\omega$ संरक्षित है, अतः $\omega$ $4$ से
गुणित हो जाता है, चाल $r\omega$ $2$ से, और \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} $4$ से
--- यह अतिरिक्त ऊर्जा उस खिंचाव का कार्य है, जिसका चकती भीतर आते-आते विरोध
करती रहती है। स्केटर की बाँहें भी वही भौतिकी हैं, बस द्रव्यमान बँटा हुआ है
(\cref{exo:b1:angular-momentum:4})।
\end{example}

\section{कणों के निकाय और स्थिर अक्ष के परितः घूर्णन}

\begin{theorem}[किसी निकाय का कोणीय संवेग]\label{thm:b1:angular-momentum:system}
\index{कोणीय संवेग!किसी निकाय का}
कणों के किसी निकाय के लिए, $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
के साथ और $O$ को किसी \omterm{thm:b1:newton-dynamics:laws}{जड़त्वीय तंत्र} में स्थिर लेने पर,
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
यानी केवल \emph{बाह्य} बलों के \omterm{def:b1:angular-momentum:moment}{आघूर्ण} गिने जाते हैं; आंतरिक बल, जो कणों को
जोड़ने वाली रेखा के अनुदिश क्रिया--प्रतिक्रिया नियम मानते हैं, कुछ भी योगदान
नहीं करते।
\end{theorem}

\begin{proof}
कणों की प्रमेयों को जोड़ दीजिए। किसी आंतरिक युग्म के लिए, $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$, क्योंकि बल
$\vect{M_jM_i}$ के अनुदिश है।
\end{proof}

\begin{proposition}[स्थिर अक्ष के परितः दृढ़ घूर्णन]\label{prop:b1:angular-momentum:inertia}
\index{जड़त्व आघूर्ण}
यदि कण अपनी दूरियाँ $r_i$ किसी अक्ष $\Delta$ से स्थिर रखें और सब उसके परितः
एक ही \omterm{cor:b1:point-kinematics:circular}{कोणीय वेग} $\omega$ से घूमें (यानी दृढ़ घूर्णन), तो
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
जहाँ $J_\Delta$ $\Delta$ के परितः \emph{\omterm{prop:b1:angular-momentum:inertia}{जड़त्व आघूर्ण}} है, और प्रमेय इस रूप
में पढ़ी जाती है: $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$।
\end{proposition}

\begin{proof}
हर कण $m_ir_i^2\omega$ को $L_\Delta$ में और
$\tfrac12 m_i(r_i\omega)^2$ को $E_k$ में जोड़ता है। \cref{ch:b1:systems-of-points}
इसे सतत ठोसों तक बढ़ाता है।
\end{proof}

\begin{example}[स्केटर]\label{ex:b1:angular-momentum:skater}
बाँहें फैलाए किसी स्केटर के धड़ का $J \approx \qty{4}{kg.m^2}$ है, और साथ में
\qty{0.8}{m} पर दो \qty{3}{kg} की बाँहें: $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$। बाँहें भीतर (\qty{0.2}{m}): $J_2 = \qty{4.2}{kg.m^2}$।
बर्फ़ की प्रतिक्रिया और भार का ऊर्ध्वाधर अक्ष के परितः कोई \omterm{def:b1:angular-momentum:moment}{आघूर्ण} नहीं है:
अतः $J\omega$ संरक्षित है और घूर्णन-दर $J_1/J_2 = 1.85$ गुना बढ़ जाती है
--- प्रति सेकंड $2$ से $3.7$ चक्कर --- जबकि \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} भी उतनी ही गुनी हो
जाती है, जिसका मूल्य बाँहें भीतर खींचती माँसपेशियाँ चुकाती हैं।
\end{example}

\begin{remark}[स्थैतिकी]\label{rem:b1:angular-momentum:statics}
\index{ठोस की साम्यावस्था}
विराम में पड़े किसी ठोस का $\vect L_O = \vect 0$ अचर होता है: अतः बाह्य
बलों का परिणामी शून्य होना चाहिए \emph{और} किसी भी बिंदु के परितः उनका कुल
\omterm{def:b1:angular-momentum:moment}{आघूर्ण} भी शून्य। दूसरी शर्त ही तुला, ढेंकुली और सब्बल का उत्तोलक-नियम है:
धुरी से दूर लगा छोटा बल उसके पास लगे किसी बड़े बल को संतुलित कर देता है।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:1}
नट से \qty{0.30}{m} दूर किसी पाने पर \qty{50}{N} का बल: \omterm{def:b1:angular-momentum:moment}{आघूर्ण}, जब बल हत्थे

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:2}
सूर्य के परितः पृथ्वी का \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} (वृत्तीय कक्षा,

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:3}
कोई शंक्वाकार लोलक (डोरी \qty{1.0}{m}, ऊर्ध्वाधर से $30^\circ$,
\cref{exo:b1:newton-dynamics:9})। धुरी से होकर जाती ऊर्ध्वाधर अक्ष के परितः
प्रति एकांक द्रव्यमान $L$ निकालिए, और समझाइए कि डोरी का तनाव और भार लगते
रहने पर भी वह संरक्षित क्यों है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:4}
किसी स्केटर के धड़ का घूर्णन-अक्ष के परितः $J = \qty{4.0}{kg.m^2}$ है; हर
बाँह (\qty{3.0}{kg}) को \qty{0.80}{m} पर, और फिर \qty{0.20}{m} पर, एक बिंदु
माना जाता है। प्रति सेकंड \qty{2.0}{} चक्करों से आरंभ करके बाँहें भीतर
खींचने के बाद घूर्णन-दर; गतिज ऊर्जाओं का अनुपात; और वह अतिरिक्त ऊर्जा कहाँ
से आती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:5}
\omterm{thm:b1:angular-momentum:theorem}{कोणीय संवेग प्रमेय} से, धुरी-अक्ष के परितः, लोलक का समीकरण
$\ell\ddot\theta + g\sin\theta = 0$ व्युत्पन्न कीजिए, और हर \omterm{def:b1:angular-momentum:moment}{बल का आघूर्ण}
दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:6}
सूर्य से पृथ्वी की दूरी \qty{1.471e11}{m} (उपसौर) से \qty{1.521e11}{m}
(अपसौर) तक बदलती है। क्षेत्रफल के नियम से इन बिंदुओं पर उसकी चालों का
अनुपात निकालिए; और \qty{29.8}{km/s} की माध्य चाल के साथ दोनों चालें दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:7}
\qty{0.20}{kg} की कोई चकती \qty{0.50}{m} की उस डोरी पर \qty{2.0}{m/s} से
चक्कर लगाती है जो किसी छेद से होकर जाती है; डोरी तब तक खींची जाती है जब तक
त्रिज्या \qty{0.25}{m} न रह जाए। नई चाल और \omterm{cor:b1:point-kinematics:circular}{कोणीय वेग}; पहले और बाद की गतिज
ऊर्जाएँ; खिंचाव का किया गया कार्य। यहाँ तनाव कार्य कर क्यों पाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:8}
\qty{20}{kg} द्रव्यमान और \qty{4.0}{m} लंबाई का कोई एकसमान तख़्ता अपने बाएँ
सिरे से \qty{1.5}{m} दूर किसी धुरी पर टिका है; \qty{30}{kg} का कोई बच्चा
बाएँ सिरे पर बैठा है। संतुलन के लिए \qty{45}{kg} के बच्चे को कहाँ बैठना
चाहिए? धुरी किसका भार सँभालती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:9}
कोई ऐटवुड मशीन ($m_1 = \qty{2.0}{kg}$, $m_2 = \qty{1.0}{kg}$) \qty{0.10}{m}
त्रिज्या और \qty{0.010}{kg.m^2} \omterm{prop:b1:angular-momentum:inertia}{जड़त्व आघूर्ण} की उस घिरनी पर लटकी है जिस पर
डोरी फिसलती नहीं। घिरनी के लिए \omterm{thm:b1:angular-momentum:theorem}{कोणीय संवेग प्रमेय} और द्रव्यमानों के लिए
\omterm{thm:b1:newton-dynamics:laws}{न्यूटन के नियम} का प्रयोग करके त्वरण तथा दोनों तनाव निकालिए। द्रव्यमानहीन
घिरनी से तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:10}
कोई धूमकेतु सूर्य से \qty{0.50}{AU} की दूरी पर, अपने उपसौर से,
\qty{40}{km/s} की चाल से गुजरता है। जब वह \qty{10}{AU} दूर हो, तब
त्रिज्या-सदिश पर लंब उसके वेग का घटक क्या होगा? क्या \omterm{cor:b1:angular-momentum:central}{क्षेत्रफल का नियम}
उसकी पूरी चाल दे सकता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:11}
सूर्य जैसा कोई तारा ($R = \qty{7.0e8}{m}$, घूर्णन आवर्तकाल \qty{25}{d})
\qty{10}{km} त्रिज्या के किसी न्यूट्रॉन तारे में संकुचित हो जाता है, और
अपना द्रव्यमान तथा \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} बनाए रखता है ($J \propto MR^2$)। नया
आवर्तकाल? भूमध्य-रेखीय चाल? यह दूसरा उत्तर उस मान्यता के बारे में क्या
बताता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:12}
कोई मनका किसी सीधी छड़ के अनुदिश बिना घर्षण के फिसलता है, और वह छड़ किसी
क्षैतिज तल में अपने एक सिरे से होकर जाती ऊर्ध्वाधर अक्ष के परितः अचर
$\omega$ से घूम रही है। मनके का $L_z$ लिखिए; दिखाइए कि वह संरक्षित
\emph{नहीं} है, और छड़ को जो \omterm{def:b1:angular-momentum:moment}{आघूर्ण} लगाना पड़ता है वह निकालिए; उससे मनके पर
छड़ की अनुप्रस्थ प्रतिक्रिया निकालिए।
\end{exercise}

\section{समस्या: चंद्रमा क्यों दूर जा रहा है}

\begin{problem}[{सप्ताहांत समस्या --- चंद्रमा पृथ्वी पर जो ज्वार उठाता है
वे उसे आगे खींचते हैं और पृथ्वी को पीछे: कोणीय संवेग हमारे दिनों से
चंद्रमा की कक्षा में सेंटीमीटर-सेंटीमीटर करके कैसे बहता है, और ऊर्जा कहाँ
जाती है}]
\label{pb:b1:angular-momentum:1}
आँकड़े: $G = \qty{6.67e-11}{SI}$; पृथ्वी $M = \qty{5.97e24}{kg}$, $R =
\qty{6.37e6}{m}$, चक्रण $\Omega = 2\pi/(\qty{86164}{s})$, \omterm{prop:b1:angular-momentum:inertia}{जड़त्व आघूर्ण}
$J = 0.33\,MR^2$; चंद्रमा $m = \qty{7.35e22}{kg}$, कक्षा वृत्तीय मानी गई है,
$r = \qty{3.84e8}{m}$; चंद्र लेज़र परास-मापन से
$\dd r/\dd t = \qty{3.8}{cm/yr}$; दिन प्रति शताब्दी \qty{2.3}{ms} लंबा होता
है; $1$ शताब्दी $= \qty{3.16e9}{s}$।

\medskip
\noindent\textbf{भाग I --- \omterm{def:b1:angular-momentum:L}{कोणीय संवेग}।}
\begin{enumerate}
  \item \omterm{thm:b1:newton-dynamics:laws}{न्यूटन के नियम} से (वृत्तीय कक्षा) चंद्रमा का कक्षीय \omterm{cor:b1:point-kinematics:circular}{कोणीय वेग}
        $\omega$ निकालिए, और उसे \qty{27.3}{d} के नाक्षत्र मास से जाँचिए।
  \item चंद्रमा का कक्षीय \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} $L_{\text{कक्षा}} =
        mr^2\omega$ निकालिए।
  \item दिखाइए कि वृत्तीय कक्षा के लिए $L_{\text{कक्षा}} = m\sqrt{GMr}$, और
        उससे $\dd L_{\text{कक्षा}}/\dd r = L_{\text{कक्षा}}/2r$ निकालिए।
  \item पृथ्वी का चक्रण \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} $L_{\text{चक्रण}} =
        J\Omega$ निकालिए और $L_{\text{कक्षा}}$ से तुलना कीजिए।
  \item चंद्रमा मास में एक बार घूमता है: उसका चक्रण \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} आँकिए
        ($J_m \approx 0.4\,mR_m^2$, $R_m = \qty{1.74e6}{m}$) और उसे उपेक्षित
        करना उचित ठहराइए।
  \item पृथ्वी--चंद्र निकाय पर उसके द्रव्यमान केंद्र के परितः कौन-से बाह्य
        \omterm{def:b1:angular-momentum:moment}{आघूर्ण} लगते हैं? तर्क दीजिए कि अच्छे सन्निकटन में कुल
        $L = L_{\text{चक्रण}} + L_{\text{कक्षा}}$ संरक्षित है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ज्वारीय \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण}।}
चंद्रमा पृथ्वी पर पानी के दो उभार उठाता है; और चूँकि पृथ्वी चंद्रमा की
परिक्रमा से तेज़ घूमती है, घर्षण उन उभारों को पृथ्वी--चंद्र रेखा से थोड़े-से
कोण भर आगे घसीट देता है।
\begin{enumerate}[resume]
  \item पृथ्वी, आगे चलते उभार और चंद्रमा का रेखाचित्र बनाइए। पास वाले
        उभार पर चंद्रमा का आकर्षण पृथ्वी के चक्रण के सापेक्ष किस दिशा में
        लगता है?
  \item उससे पृथ्वी के चक्रण पर लगते \omterm{def:b1:angular-momentum:moment}{आघूर्ण} का चिह्न निकालिए: दिन लंबा
        होता है या छोटा?
  \item \omterm{thm:b1:newton-dynamics:laws}{क्रिया और प्रतिक्रिया} से, चंद्रमा की कक्षीय गति पर कौन-सा \omterm{def:b1:angular-momentum:moment}{आघूर्ण}
        लगता है? $L_{\text{कक्षा}}$ बढ़ता है या घटता है?
  \item प्रश्न 3 से, $r$ बढ़ता है या घटता है? और चंद्रमा की कक्षीय चाल
        $v = \sqrt{GM/r}$?
  \item इस आभासी विरोधाभास को सुलझाइए: चंद्रमा को आगे धकेला जाता है, फिर
        भी अंत में वह और धीमा चलने लगता है।
  \item यह आदान-प्रदान किस अंतिम अवस्था की ओर बढ़ता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- संख्याएँ।}
\begin{enumerate}[resume]
  \item दिन के लंबे होने से प्रति शताब्दी सापेक्ष परिवर्तन
        $\Delta\Omega/\Omega$ निकालिए।
  \item उससे प्रति शताब्दी $L_{\text{चक्रण}}$ का परिवर्तन निकालिए।
  \item उससे प्रति शताब्दी $L_{\text{कक्षा}}$ का परिवर्तन निकालिए।
  \item प्रश्न 3 का प्रयोग करके उसे प्रति शताब्दी और प्रति वर्ष $r$ के
        परिवर्तन में बदलिए। लेज़र परास-मापन के मान से तुलना कीजिए।
  \item नाक्षत्र मास प्रति शताब्दी किस अंश से, और कितने सेकंड, लंबा होता
        है ($T \propto r^{3/2}$)?
  \item पृथ्वी की घूर्णन \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} का प्रति शताब्दी परिवर्तन निकालिए
        (प्रथम कोटि में $\Delta E = L_{\text{चक्रण}}\Delta\Omega$)।
  \item चंद्रमा की कक्षीय \omterm{thm:b1:work-and-energy:em}{यांत्रिक ऊर्जा}, $E_{\text{कक्षा}} = -GMm/2r$, का
        प्रति शताब्दी परिवर्तन निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- ऊर्जा।}
\begin{enumerate}[resume]
  \item दिखाइए कि पृथ्वी जितनी ऊर्जा खोती है वह चंद्रमा को मिलती ऊर्जा से
        अधिक है, और प्रति शताब्दी अंतर तथा उससे बनती शक्ति निकालिए। वह
        कहाँ जाती है?
  \item उस शक्ति की तुलना मानवता की खपत, लगभग \qty{18}{TW}, से और पृथ्वी
        के आंतरिक ऊष्मा-प्रवाह, लगभग \qty{47}{TW}, से कीजिए।
  \item प्रश्न 12 की अंतिम अवस्था में दिन और मास बराबर हैं:
        $\Omega = \omega = \sqrt{GM/r^3}$। $L_{\text{चक्रण}} + L_{\text{कक्षा}}$
        का संरक्षण लिखकर जाँचिए कि $r \approx
        \qty{5.5e8}{m}$ उसे संतुष्ट करता है, और साझा आवर्तकाल दिनों में
        दीजिए।
  \item वह अवस्था वास्तव में कभी क्यों नहीं आएगी? (सूर्य के ज्वारों के
        बारे में सोचिए।)
  \item चंद्रमा पहले ही पृथ्वी को एक ही मुख दिखाता है। उसी क्रियाविधि से
        समझाइए, और बताइए कि ऐसा पहले चंद्रमा के साथ ही क्यों हुआ।
  \item तीन पंक्तियों में सार दीजिए: क्या संरक्षित है, क्या बदला जाता है,
        और क्या क्षय हो जाता है।
\end{enumerate}
\end{problem}
