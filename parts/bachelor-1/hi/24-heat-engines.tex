\chapter{ऊष्मा इंजन और मशीनें}\label{ch:b1:heat-engines}

कार का इंजन, कोई बिजलीघर, कोई \omterm{def:b1:heat-engines:machine}{प्रशीतक} और कोई \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप} --- ये चारों एक ही
वस्तु के चार पहलू हैं: कोई तरल किसी चक्र पर घुमाया जाता है, बारी-बारी से
किसी गरम स्रोत और किसी ठंडे स्रोत के संस्पर्श में आता है, दोनों के साथ
ऊष्मा और किसी दंड के साथ कार्य का आदान-प्रदान करता है। पहला नियम ऊर्जा का
बहीखाता रखता है; और दूसरा नियम, एक चक्र पर के एन्ट्रॉपी-संतुलन के ज़रिए, वह
सीमा बाँध देता है जिसे कोई चतुराई नहीं हरा सकती --- यानी \omterm{thm:b1:heat-engines:carnot}{कार्नो की प्रमेय},
अभियांत्रिकी की सबसे दूरगामी असमिका। यह अध्याय उस सीमा को व्युत्पन्न करता
है, आदर्श चक्र और उसके पास आते वास्तविक चक्रों (ओटो, डीज़ल, स्टर्लिंग) की
गणना करता है, मशीन को उलटकर ठंडक बनाता तथा घर गरम करता है, और दिखाता है कि
किसी वास्तविक मशीन की सृजित एन्ट्रॉपी की क़ीमत, जूल-दर-जूल, खोए हुए कार्य
में चुकाई जाती है।

\section{चक्रीय मशीनें और दोनों नियम}

\begin{definition}[ऊष्मागतिक मशीन]\label{def:b1:heat-engines:machine}
\index{ऊष्मा इंजन}\index{प्रशीतक}\index{ऊष्मा पंप}\index{ऊष्मा-भंडार}
\emph{ऊष्मागतिक मशीन} कोई ऐसा \omterm{def:b1:newton-dynamics:conservation}{संवृत निकाय} (कार्यकारी तरल) है जिसे किसी
\emph{चक्र} पर घुमाया जाता है, ताकि वह आवर्ती रूप से उसी अवस्था में लौट
आए, और इस बीच वह बाहर के साथ कार्य $W$ का तथा ऊष्मा $Q_i$ का आदान-प्रदान उन
स्रोतों के साथ करता रहे जो तापों $T_i$ पर हैं --- ये स्रोत \emph{ऊष्मा-भंडार} हैं,
जिनका ताप इस आदान-प्रदान से नहीं बदलता। कोई \emph{द्वितापी} मशीन दो
ऊष्मा-भंडार काम में लेती है: $T_h$ पर गरम स्रोत और $T_c < T_h$ पर ठंडा
स्रोत। वह \emph{इंजन} है यदि वह कार्य देती हो ($W < 0$),
\emph{प्रशीतक} यदि उसे चलाकर ($W > 0$) ठंडे स्रोत से ऊष्मा ली जाती हो, और
\emph{ऊष्मा पंप} यदि उसे चलाकर गरम स्रोत को ऊष्मा दी जाती हो। सारी राशियाँ
एक चक्र पर, बीजगणितीय रूप से, तरल को मिली हुई के रूप में गिनी जाती हैं।
\end{definition}

\begin{theorem}[एक चक्र पर दोनों नियम; क्लॉज़ियस असमिका]\label{thm:b1:heat-engines:clausius}
\index{क्लॉज़ियस असमिका}
किसी द्वितापी मशीन के एक चक्र पर,
\[
W + Q_h + Q_c = 0 , \qquad
\frac{Q_h}{T_h} + \frac{Q_c}{T_c} = -S_{\mathrm{created}} \leq 0 ,
\]
जहाँ दूसरे संबंध में समता तभी और केवल तभी होती है जब चक्र उत्क्रमणीय हो।
और आम तौर पर, कितने भी ऊष्मा-भंडारों के लिए,
$\sum_i Q_i/T_i \leq 0$ (\emph{\omterm{thm:b1:heat-engines:clausius}{क्लॉज़ियस असमिका}})।
\end{theorem}

\begin{proof}
$U$ और $S$ अवस्था फलन हैं, अतः किसी चक्र पर $\Delta U = 0$ और $\Delta S = 0$
रहते हैं। पहला नियम $0 = W + Q_h + Q_c$ देता है; और एन्ट्रॉपी-संतुलन
(\cref{thm:b1:second-law-entropy:secondlaw}) $0 = S_{\mathrm{exch}} +
S_{\mathrm{created}}$ देता है, जहाँ $S_{\mathrm{exch}} = \sum Q_i/T_i$ है,
क्योंकि हर \omterm{def:b1:heat-engines:machine}{ऊष्मा-भंडार} अपने ही स्थिर ताप पर आदान-प्रदान करता है
(\cref{prop:b1:second-law-entropy:thermostat})।
\end{proof}

\begin{corollary}[केल्विन, फिर से]\label{cor:b1:heat-engines:kelvin}
किसी \emph{एकतापी} मशीन ($Q_c = 0$) के लिए $Q_h \leq 0$ होता है, अतः
$W \geq 0$: यानी कोई चक्र किसी अकेले स्रोत की ऊष्मा को कार्य में नहीं बदल
सकता। कोई जहाज़ समुद्र को ठंडा करके नहीं चल सकता --- किसी इंजन को ऊष्मा
छोड़ने के लिए कोई ठंडा स्रोत चाहिए, और दूसरा नियम बताता है कि कितनी छोड़नी
पड़ेगी।
\end{corollary}

\section{कार्नो की प्रमेय}

\begin{definition}[दक्षता और निष्पादन गुणांक]\label{def:b1:heat-engines:efficiency}
\index{दक्षता!ऊष्मा इंजन की}\index{निष्पादन गुणांक}
किसी इंजन की \emph{दक्षता} वह कार्य है जो वह गरम स्रोत से ली गई प्रति एकांक
ऊष्मा पर देता है, $\eta = -W/Q_h$; किसी \omterm{def:b1:heat-engines:machine}{प्रशीतक} का
\emph{निष्पादन गुणांक} प्रति एकांक कार्य पर बनी ठंडक है, $e_f = Q_c/W$; और
किसी \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप} का, प्रति एकांक कार्य पर दी गई ऊष्मा, $e_p = -Q_h/W$। हर एक
``जो हमें चाहिए'' और ``जो हम चुकाते हैं'' का अनुपात है; कोई दक्षता अधिक से
अधिक~1 हो सकती है, पर कोई निष्पादन गुणांक~1 से बड़ा भी हो सकता है।
\end{definition}

\begin{theorem}[कार्नो]\label{thm:b1:heat-engines:carnot}
\index{कार्नो की प्रमेय}\index{कार्नो दक्षता}
ऊष्मा-भंडारों $T_h > T_c$ के बीच,
\[
\eta \leq \eta_C = 1 - \frac{T_c}{T_h}, \qquad
e_f \leq \frac{T_c}{T_h - T_c}, \qquad
e_p \leq \frac{T_h}{T_h - T_c} = 1 + \frac{T_c}{T_h - T_c},
\]
जहाँ समता तभी और केवल तभी होती है जब चक्र उत्क्रमणीय हो। ये सीमाएँ केवल
तापों पर निर्भर करती हैं --- न तरल पर, न क्रियाविधि पर।
\end{theorem}

\begin{proof}
किसी इंजन के लिए ($Q_h > 0$): $\eta = -W/Q_h = (Q_h + Q_c)/Q_h = 1 + Q_c/Q_h$,
और \omterm{thm:b1:heat-engines:clausius}{क्लॉज़ियस असमिका} $Q_c/Q_h \leq -T_c/T_h$ देती है। किसी
\omterm{def:b1:heat-engines:machine}{प्रशीतक} के लिए ($Q_c > 0$, $W > 0$): $W = -Q_h - Q_c$, जहाँ $-Q_h \geq Q_c T_h/T_c$,
अतः $W \geq Q_c(T_h/T_c - 1)$ और $e_f = Q_c/W \leq T_c/(T_h - T_c)$।
\omterm{def:b1:heat-engines:machine}{ऊष्मा पंप} के लिए, $e_p = -Q_h/W = (W + Q_c)/W = 1 + e_f$। समता वही
उत्क्रमणीय स्थिति है, जिसमें हर जगह $S_{\mathrm{created}} = 0$ हो।
\end{proof}

\begin{example}[तीन आँकड़े]\label{ex:b1:heat-engines:numbers}
$T_h = \qty{800}{K}$ और $T_c = \qty{300}{K}$ की किसी नदी के बीच चलती कोई
भाप-टरबाइन: $\eta_C = 0.625$; वास्तविक संयंत्र लगभग $0.40$ तक पहुँचते हैं।
भीतर $\qty{275}{K}$ और $\qty{298}{K}$ की किसी रसोई के बीच का कोई घरेलू
\omterm{def:b1:heat-engines:machine}{प्रशीतक}: $e_f \leq 275/23 = 12$; वास्तविक मशीनें $2$ से $4$ देती हैं। और
किसी घर को $\qty{293}{K}$ पर गरम करता कोई \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप}, $\qty{273}{K}$ की हवा
से: $e_p \leq 293/20 = 14.7$; वास्तविक पंप $3$ से $5$ देते हैं --- फिर भी
उतनी ही बिजली किसी \omterm{def:b1:dc-circuits:resistor}{प्रतिरोधक} में जितनी देती, उससे तीन से पाँच गुना
(\cref{pb:b1:heat-engines:1})।
\end{example}

\begin{remark}[दक्षता~1 क्यों नहीं हो सकती]\label{rem:b1:heat-engines:why}
कोई इंजन जो ऊष्मा भीतर लेता है उसके साथ एन्ट्रॉपी $Q_h/T_h$ भी पा लेता है;
कार्य कुछ भी बाहर नहीं ले जाता; और अकेला निकास वही ऊष्मा है जो ठंडे स्रोत
को छोड़ी जाती है और $|Q_c|/T_c$ ले जाती है। एन्ट्रॉपी नष्ट नहीं की जा सकती,
अतः कम से कम $Q_h T_c/T_h$ ऊष्मा फेंकनी ही पड़ती है, और कार्य वही है जो बच
रहे। ठंडा स्रोत जितना नीचा और गरम स्रोत जितना ऊँचा, उतना अच्छा --- इसी से
शीतलन-मीनारें हैं, और और भी गरम टरबाइन-पंखुड़ियों की खोज।
\end{remark}

\section{कार्नो चक्र}

\begin{definition}[कार्नो चक्र]\label{def:b1:heat-engines:carnotcycle}
\index{कार्नो चक्र}
\emph{कार्नो चक्र} वही उत्क्रमणीय द्वितापी चक्र है: दो उत्क्रमणीय समतापी
वक्र, एक $T_h$ पर (गरम स्रोत के संस्पर्श में) और एक $T_c$ पर (ठंडे स्रोत
के साथ), जिन्हें दो उत्क्रमणीय रुद्धोष्म चरण (समएन्ट्रॉपी, किसी स्रोत के
संस्पर्श के बिना) जोड़ते हैं। $(V, P)$ तल में घड़ी की दिशा में घुमाने पर वह
इंजन है; उलटी दिशा में, \omterm{def:b1:heat-engines:machine}{प्रशीतक} या \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप}। और \omterm{def:b1:second-law-entropy:TS}{एन्ट्रॉपी आरेख} में वह एक
आयत है।
\end{definition}

\begin{proposition}[आदर्श गैस का कार्नो चक्र]\label{prop:b1:heat-engines:carnotgas}
\omterm{def:b1:kinetic-theory:model}{आदर्श गैस} के $n$ मोलों को $A \to B$ (समतापी $T_h$, प्रसार), $B \to C$
(रुद्धोष्म), $C \to D$ (समतापी $T_c$, संपीडन), $D \to A$ (रुद्धोष्म) पर
घुमाने पर:
\[
Q_h = nRT_h\ln\frac{V_B}{V_A}, \qquad
Q_c = -nRT_c\ln\frac{V_B}{V_A}, \qquad
\eta = 1 - \frac{T_c}{T_h} ,
\]
यानी कार्नो का मान --- जैसा किसी भी उत्क्रमणीय द्वितापी चक्र के लिए होना ही चाहिए।
\end{proposition}

\begin{proof}
किसी समतापी वक्र पर $\Delta U = 0$, अतः $Q = -W = nRT\ln(V_{\text{अंतिम}}/
V_{\text{आरंभिक}})$।
रुद्धोष्म चरणों पर $TV^{\gamma-1}$ अचर रहता है:
यानी $T_hV_B^{\gamma-1} = T_cV_C^{\gamma-1}$
और $T_hV_A^{\gamma-1} = T_cV_D^{\gamma-1}$, जिससे $V_C/V_D = V_B/V_A$ और $Q_c = nRT_c\ln(V_D/V_C) =
-nRT_c\ln(V_B/V_A)$। फिर $\eta = 1 + Q_c/Q_h = 1 - T_c/T_h$।
\end{proof}

\begin{remark}[कार्नो का चक्र एक प्रमेय है, कोई इंजन नहीं]\label{rem:b1:heat-engines:notengine}
उत्क्रमणीय समतापी वक्रों के लिए ऊष्मा का हस्तांतरण अपरिमित रूप से धीमा और
लुप्त होते ताप-अंतर के पार होना चाहिए: यानी कोई कार्नो इंजन अपना अधिकतम
कार्य शून्य शक्ति पर देता है। वास्तविक मशीनें अपने विनिमयकों में परिमित
ताप-अंतर स्वीकार करके \omterm{def:b1:heat-engines:efficiency}{दक्षता} के बदले शक्ति ख़रीदती हैं
(\cref{exo:b1:heat-engines:10}); और कार्नो का मान वही छत बना रहता है जिससे
हर अभिकल्प नापा जाता है।
\end{remark}

\section{वास्तविक इंजन-चक्र}

\begin{proposition}[ओटो चक्र]\label{prop:b1:heat-engines:otto}
\index{ओटो चक्र}\index{संपीडन अनुपात}
आदर्शीकृत पेट्रोल इंजन हवा ($\gamma$) को इस तरह घुमाता है: $1 \to 2$
$V_1$ से $V_2$ तक रुद्धोष्म संपीडन (\emph{\omterm{prop:b1:heat-engines:otto}{संपीडन अनुपात}}
$r = V_1/V_2$); $2 \to 3$ समआयतनिक तापन (यानी दहन); $3 \to 4$
वापस $V_1$ तक रुद्धोष्म प्रसार; $4 \to 1$ समआयतनिक शीतलन (यानी
निकास)। उसकी \omterm{def:b1:heat-engines:efficiency}{दक्षता} यह है
\[
\eta_{\text{ओटो}} = 1 - \frac{1}{r^{\gamma-1}} .
\]
\end{proposition}

\begin{proof}
ऊष्मा का आदान-प्रदान केवल समआयतनिक चरणों पर:
$Q_h = nC_{V,m}(T_3 - T_2)$ और $Q_c = nC_{V,m}(T_1 - T_4)$। रुद्धोष्म चरणों पर $TV^{\gamma-1}$ अचर है:
$T_2 = T_1r^{\gamma-1}$ और $T_3 = T_4r^{\gamma-1}$, अतः $T_3 - T_2 =
(T_4 - T_1)r^{\gamma-1}$ और
$\eta = 1 + Q_c/Q_h = 1 - (T_4 - T_1)/(T_3 - T_2) = 1 - r^{1-\gamma}$।
\end{proof}

\begin{example}[किसी पेट्रोल इंजन के आँकड़े]\label{ex:b1:heat-engines:ottonumbers}
$r = 9$, $\gamma = 1.4$: $9^{0.4} = 2.41$ और $\eta = 0.58$। कोई वास्तविक
इंजन लगभग $0.30$ देता है: दहन तात्क्षणिक नहीं होता, \qty{2500}{K} पर गैस
आदर्श नहीं रहती, दीवारों को ऊष्मा रिसती है, और घर्षण तथा पंपन अपना हिस्सा
ले जाते हैं। $r$ बढ़ाना मदद करता है --- जब तक हवा--ईंधन मिश्रण संपीडन के अंत
में अपने आप न जल उठे (खटखट); और डीज़ल इंजन इसी अवगुण को अपना सिद्धांत बना
लेता है।
\end{example}

\begin{proposition}[डीज़ल चक्र]\label{prop:b1:heat-engines:diesel}
\index{डीज़ल चक्र}
\omterm{prop:b1:heat-engines:otto}{ओटो चक्र} के समआयतनिक तापन की जगह समदाबी तापन रख दीजिए, $V_2$ से
$V_3 = \rho V_2$ तक (\emph{अंतःक्षेप अनुपात} $\rho > 1$): \omterm{prop:b1:heat-engines:otto}{संपीडन अनुपात} $r$
के $15$ से $22$ तक होने से हवा पहले ही गरम हो चुकी होती है, और उसमें
अंतःक्षिप्त ईंधन आते ही जल जाता है। तब
\[
\eta_{\text{डीज़ल}} = 1 - \frac{1}{r^{\gamma-1}}\,\frac{\rho^{\gamma} - 1}{\gamma(\rho - 1)} ,
\]
जो उसी $r$ के लिए ओटो के मान से छोटा है, पर कहीं बड़े $r$ के साथ पाया जाता
है, अतः व्यवहार में बड़ा (बड़े इंजनों के लिए $0.40$ से $0.45$)।
\end{proposition}

\begin{proof}
$Q_h = nC_{P,m}(T_3 - T_2)$, $Q_c = nC_{V,m}(T_1 - T_4)$, अतः $\eta = 1 -
(T_4 - T_1)/[\gamma(T_3 - T_2)]$। $T_2 = T_1r^{\gamma-1}$, $T_3 =
\rho T_2$ के साथ, और रुद्धोष्म चरण $3 \to 4$ पर ($V_4 = V_1 = rV_2$), $T_4 =
T_3(V_3/V_4)^{\gamma-1} = \rho T_1 r^{\gamma-1}(\rho/r)^{\gamma-1} = T_1\rho^{\gamma}$:
$\eta = 1 - (\rho^\gamma - 1)/[\gamma r^{\gamma-1}(\rho - 1)]$।
\end{proof}

\begin{proposition}[स्टर्लिंग चक्र और पुनर्जनक]\label{prop:b1:heat-engines:stirling}
\index{स्टर्लिंग चक्र}\index{पुनर्जनक}
दो समतापी वक्र ($T_c$, $T_h$), जिन्हें दो समआयतनिक वक्र जोड़ते हैं।
समआयतनिक चरणों की ऊष्मा, $\pm nC_{V,m}(T_h - T_c)$, बराबर और विपरीत है; और
यदि उसे ठंडे होते चरण पर किसी \emph{पुनर्जनक} में (यानी किसी सरंध्र पिंड
में, जिसमें से गैस बहती है) संचित करके गरम होते चरण पर लौटा दिया जाए, तो
स्रोतों के साथ आदान-प्रदान केवल समतापी वक्र करते हैं, और
$\eta = 1 - T_c/T_h$: यानी स्टर्लिंग इंजन अपने चक्र के आकार को छोड़कर हर
बात में कार्नो इंजन ही है। पुनर्जनक के बिना समआयतनिक ऊष्मा गरम स्रोत से आनी
पड़ती है और
$\eta = \dfrac{nR(T_h - T_c)\ln(V_1/V_2)}{nRT_h\ln(V_1/V_2) + nC_{V,m}(T_h - T_c)}$,
जो काफ़ी कम है (\cref{exo:b1:heat-engines:8})।
\end{proposition}

\begin{proof}
समतापी वक्रों पर $Q_h = nRT_h\ln(V_1/V_2)$ और $Q_c = -nRT_c\ln(V_1/V_2)$, अतः
समआयतनिक ऊष्माओं के कट जाने पर $-W = Q_h + Q_c + 0$; और हर स्थिति में उसे
उस ऊष्मा से भाग दीजिए जो वास्तव में गरम स्रोत से ली गई।
\end{proof}

\begin{remark}[भाप और गैस टरबाइन]\label{rem:b1:heat-engines:turbines}
बिजलीघर किसी बेलन में गैस नहीं, बल्कि पानी काम में लेते हैं, जिसे उबाला
जाता है, किसी टरबाइन में फैलाया जाता है, संघनित करके वापस पंप कर दिया जाता
है (यानी रैंकिन चक्र, जिसमें दो प्रावस्था-परिवर्तन हैं ---
\cref{ch:b1:phase-changes}); या फिर हवा, जिसे संपीडित किया जाता है, दहन से
गरम किया जाता है और किसी गैस-टरबाइन में फैलाया जाता है (यानी ब्रेटन चक्र,
दो रुद्धोष्म और दो समदाबी वक्र)। इनके विश्लेषण के लिए किसी बहते तरल का
ऊर्जा-संतुलन चाहिए, जो वर्ष 2 के खंड में है; कार्नो की सीमा और इस अध्याय
का तर्क ज्यों-का-त्यों लागू रहते हैं।
\end{remark}

\section{प्रशीतक और ऊष्मा पंप}

\begin{definition}[वाष्प-संपीडन चक्र]\label{def:b1:heat-engines:vapourcycle}

लगभग हर \omterm{def:b1:heat-engines:machine}{प्रशीतक}, हिमकारक, वातानुकूलक और \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप} किसी \emph{शीतलक} को चार

बढ़ाता है), कोई \emph{संघनित्र} (जिसमें गरम वाष्प ऊँचे दाब पर द्रवित होकर

दोनों दाब ही दोनों क्वथन-ताप तय करते हैं

\begin{proposition}[ताप-अंतरों की क़ीमत]\label{prop:b1:heat-engines:gaps}
जो उत्क्रमणीय मशीन अपने ही भीतरी तापों
$T_{\mathrm{ev}} < T_c$ और $T_{\mathrm{cd}} > T_h$ के बीच काम करे, उसका $e_p = T_{\mathrm{cd}}/(T_{\mathrm{cd}} - T_{\mathrm{ev}})$ होता है,
जो स्रोतों के बीच के कार्नो मान से छोटा है; और कोई वास्तविक मशीन उसका भी
एक अंश (प्रायः $0.4$ से $0.6$) ही पाती है। अतः किसी \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप} का निष्पादन
बाहर का ताप गिरने के साथ और तापन-परिपथ की माँग का ताप बढ़ने के साथ गिरता
जाता है।
\end{proposition}

\begin{proof}
\cref{thm:b1:heat-engines:carnot} को उस तरल पर लगाइए जिसके \omterm{def:b1:heat-engines:machine}{ऊष्मा-भंडार}
प्रभावी रूप से विनिमयकों के ताप हैं; और $T_{\mathrm{cd}} - T_{\mathrm{ev}} >
T_h - T_c$ अनुपात को घटा देता है।
\end{proof}

\section{एन्ट्रॉपी और खोया कार्य}

\begin{theorem}[खोया कार्य]\label{thm:b1:heat-engines:lostwork}
\index{खोया कार्य}
प्रति चक्र एन्ट्रॉपी $S_{\mathrm{created}}$ सृजित करता कोई द्वितापी इंजन
इतना देता है
\[
-W = Q_h\Bigl(1 - \frac{T_c}{T_h}\Bigr) - T_c\,S_{\mathrm{created}} ,
\]
और $|Q_h|$ देता कोई द्वितापी \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप}
$W = |Q_h|(1 - T_c/T_h) + T_c\,S_{\mathrm{created}}$ खपाता है: यानी प्रति
\omterm{def:b1:kinetic-theory:temperature}{केल्विन} सृजित हर जूल $T_c$ जूल कार्य की क़ीमत माँगता है, और ठंडे स्रोत का
ताप ही एन्ट्रॉपी तथा कार्य के बीच की विनिमय-दर है।
\end{theorem}

\begin{proof}
\cref{thm:b1:heat-engines:clausius} से, $Q_c = -T_c(Q_h/T_h + S_{\mathrm{created}})$;
इसे $-W = Q_h + Q_c$ में रखिए। पंप के लिए वही दोनों संबंध
$Q_h < 0$ के साथ लिखिए।
\end{proof}

\begin{method}[किसी मशीन का लेखा-जोखा]\label{meth:b1:heat-engines:audit}
हर स्रोत के साथ आदान-प्रदान की गई ऊष्माएँ और कार्य नापिए (या निकालिए);
पहला नियम जाँचिए; प्रति चक्र या प्रति सेकंड $S_{\mathrm{created}} = -\sum Q_i/T_i$
निकालिए; खोई हुई शक्ति $T_c\dot S_{\mathrm{created}}$ है; और सृजन कहाँ हो रहा
है यह पता करने के लिए (विनिमयकों में परिमित ताप-अंतर, घर्षण, अवरोधन,
अनर्ध-स्थैतिक संपीडन) एन्ट्रॉपी-संतुलन को बारी-बारी से हर घटक पर लगाइए। जो
घटक कोई एन्ट्रॉपी नहीं सृजित करता, उसे सुधारने का कोई लाभ नहीं।
\end{method}

\begin{example}[किसी बिजलीघर का लेखा]\label{ex:b1:heat-engines:audit}
गरम स्रोत \qty{800}{K}, नदी \qty{300}{K}, $\qty{1.0}{GW}$ बिजली,
$\eta = 0.40$ पर: $\dot Q_h = \qty{2.5}{GW}$, $\dot Q_c = \qty{1.5}{GW}$, अतः
$\dot S_{\mathrm{created}} = 1.5/300 - 2.5/800 = \qty{1.9}{MW/K}$ और खोई हुई
शक्ति $300 \times 1.9 = \qty{0.56}{GW}$: यानी ठीक वह अंतर जो कार्नो के
निर्गत $0.625 \times 2.5 = \qty{1.56}{GW}$ और वास्तविक \qty{1.0}{GW} के बीच है।
नदी उस \qty{1.5}{GW} से गरम होती है जो उसे ढोनी पड़ती है; और गर्मियों में,
जब वह गरम तथा उथली रहती है, संयंत्र अपना निर्गत घटा लेते हैं।
\end{example}

$x$ और $y$ के बीच काम करता है, और स्रोतों के साथ ऊष्मा का आदान-प्रदान

\end{exercise}

\section{समस्या: घर का ऊष्मा पंप}

\begin{problem}[{सप्ताहांत समस्या --- पूरी सर्दी एक घर को गरम रखना:
प्रतिरोधकों से, गैस से, किसी आदर्श ऊष्मा पंप से और किसी वास्तविक पंप से
बिल, और वास्तविक पंप की बिजली जाती कहाँ है}]
\label{pb:b1:heat-engines:1}
कोई घर $K(T_{\mathrm{in}} - T_{\mathrm{out}})$ की दर से ऊष्मा खोता है, जहाँ
$K = \qty{250}{W/K}$; और उसे $T_{\mathrm{in}} = \qty{20}{\degreeCelsius}$ पर
रखा जाता है। अभिकल्प के लिए बाहर का ताप $\qty{-5}{\degreeCelsius}$;
तापन-मौसम: $200$ दिन, जिनमें बाहर का माध्य ताप \qty{7}{\degreeCelsius} हो।
क़ीमतें: बिजली 0.25~\texteuro{} प्रति \unit{kW h}, गैस 0.10~\texteuro{} प्रति
\unit{kW h}; $\qty{1}{kW h} = \qty{3.6}{MJ}$।

\medskip
\noindent\textbf{भाग I --- घर और पुराने तरीक़े।}
\begin{enumerate}
  \item अभिकल्प-ताप पर आवश्यक तापन-शक्ति।
  \item पूरे मौसम में आवश्यक ऊर्जा, जूल में और \unit{kW h} में।
  \item विद्युत प्रतिरोधकों के साथ मौसम भर की लागत।
  \item $0.90$ \omterm{def:b1:heat-engines:efficiency}{दक्षता} के किसी गैस-बॉयलर के साथ लागत।
  \item बिजली $0.40$ \omterm{def:b1:heat-engines:efficiency}{दक्षता} के किसी तापीय बिजलीघर से आती है। प्रति मौसम,
        प्रतिरोधकों से और बॉयलर से खपी ईंधन-ऊर्जा; और ``विद्युत तापन
        ऊष्मागतिक रूप से फ़िज़ूलख़र्ची है'' पर टिप्पणी कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- आदर्श \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप}।}
\begin{enumerate}[resume]
  \item $e_p$ को परिभाषित कीजिए और एक चक्र पर दोनों नियमों से सिद्ध कीजिए
        कि $e_p \leq T_{\mathrm{in}}/(T_{\mathrm{in}} - T_{\mathrm{out}})$।
  \item अभिकल्प-ताप पर आदर्श $e_p$, और तब आवश्यक \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}।
  \item मौसम के माध्य ताप पर आदर्श $e_p$, और (उसी मान को पूरे मौसम के लिए
        लेकर) मौसम भर की बिजली।
  \item भौतिक रूप से समझाइए कि बाहर का ताप गिरने पर $e_p$ क्यों गिरता है,
        और किसी तापन-युक्ति के लिए यह सबसे बुरा संभव व्यवहार क्यों है।
  \item अभिकल्प-बिंदु पर आदर्श पंप \qty{0.53}{kW} खींचता है और
        \qty{6.25}{kW} देता है: बाकी \qty{5.7}{kW} आते कहाँ से हैं, और
        बाहर की ठंडी हवा को ठंडा करके किसी गरम घर को गरम करना दूसरे नियम
        का उल्लंघन क्यों नहीं है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- वास्तविक मशीन।}
पंप कोई \omterm{def:b1:heat-engines:vapourcycle}{वाष्प-संपीडन चक्र} चलाता है। \omterm{def:b1:heat-engines:vapourcycle}{शीतलक}
$T_{\mathrm{ev}} = T_{\mathrm{out}} - \qty{8}{K}$ पर वाष्पित होता है और
$T_{\mathrm{cd}} = \qty{313}{K}$ पर संघनित (किसी फ़र्श-तापन परिपथ का पानी
\qty{35}{\degreeCelsius} पर चलता है); और मशीन उस कार्नो मान का $0.55$ पा
लेती है जो $T_{\mathrm{ev}}$ तथा $T_{\mathrm{cd}}$ के बीच निकाला जाए। \omterm{def:b1:heat-engines:vapourcycle}{शीतलक} की
गुप्त ऊष्मा: $L = \qty{200}{kJ/kg}$।
\begin{enumerate}[resume]
  \item चक्र के चारों घटकों के नाम बताइए, बताइए कि तरल किसमें $Q_c$ पाता
        है, किसमें $|Q_h|$ देता है और किसमें $W$ पाता है, और वाष्पित्र को
        बाहर की हवा से ठंडा तथा संघनित्र को घर से गरम क्यों होना चाहिए।
  \item अभिकल्प-ताप पर: $T_{\mathrm{ev}}$, और कार्नो $e_p$,
        $T_{\mathrm{ev}}$ तथा $T_{\mathrm{cd}}$ के बीच।
  \item अभिकल्प-ताप पर वास्तविक $e_p$, खींची गई \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}, और बाहर की
        हवा से ली गई ऊष्मा।
  \item वाष्पित्र से होकर \omterm{def:b1:heat-engines:vapourcycle}{शीतलक} की द्रव्यमान-प्रवाह दर।
  \item वही प्रश्न (वास्तविक $e_p$, \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}) \qty{7}{\degreeCelsius}
        के माध्य ताप पर, जहाँ घर \qty{3.25}{kW} खोता है।
  \item $e_p$ को $T_{\mathrm{out}}$ के फलन के रूप में दीजिए; और माध्य-ताप
        वाला $e_p$ पूरे मौसम के लिए लेकर मौसम भर की बिजली तथा लागत।
        भाग~I से तुलना कीजिए।
  \item ऊष्मा-पंप वाले मौसम के लिए बिजलीघर में खपी ईंधन-ऊर्जा; प्रतिरोधकों
        और बॉयलर से तुलना कीजिए।
  \item \qty{-15}{\degreeCelsius} की कोई शीत-लहर: ऊष्मा की माँग, वास्तविक
        $e_p$, और पंप उसे दे पाता तो कितनी \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}। असल में संपीडक
        की क्षमता $T_{\mathrm{ev}}$ के साथ गिरती है और पंप वहाँ केवल
        \qty{3.8}{kW} दे पाता है: आवश्यक सहायक प्रतिरोधक-शक्ति, और ``\omterm{def:b1:heat-engines:machine}{ऊष्मा
        पंप} का आकार तय करने'' का अर्थ।
  \item यदि घर में \qty{60}{\degreeCelsius} का पानी माँगते पुराने रेडिएटर
        होते ($T_{\mathrm{cd}} = \qty{338}{K}$): अभिकल्प-ताप पर वास्तविक
        $e_p$ और \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}; और उत्सर्जकों के चुनाव पर निष्कर्ष
        निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- बिजली जाती कहाँ है।}
\begin{enumerate}[resume]
  \item दिखाइए कि जो \omterm{def:b1:heat-engines:machine}{ऊष्मा पंप} घर को $|Q_h|$ देता हो, $T_{\mathrm{in}}$ पर,
        और $T_{\mathrm{out}}$ की बाहरी हवा से लेता हो, उसके लिए
        $W = |Q_h|(1 - T_{\mathrm{out}}/T_{\mathrm{in}}) + T_{\mathrm{out}}S_{\mathrm{created}}$।
  \item अभिकल्प-बिंदु पर, संघनित्र के पार (\qty{313}{K} से घर तक) और
        वाष्पित्र के पार (बाहर की हवा से \qty{260}{K} तक) ऊष्मा के
        हस्तांतरण से प्रति सेकंड सृजित एन्ट्रॉपी।
  \item ये दोनों ताप-अंतर कितनी \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति} की क़ीमत माँगते हैं; और खींची
        गई \qty{1.92}{kW} का कितना अंश।
  \item वास्तविक मशीन द्वारा प्रति सेकंड सृजित कुल एन्ट्रॉपी (उसकी
        \qty{1.92}{kW} की तुलना आदर्श \qty{0.53}{kW} से कीजिए); उसमें
        विनिमयकों का हिस्सा; और बाकी स्रोतों के नाम बताइए।
  \item दोनों अंतर आधे कर दीजिए (\qty{4}{K} और \qty{10}{K}): विनिमयकों में
        सृजित एन्ट्रॉपी और बची हुई \omterm{thm:b1:dc-circuits:kirchhoff}{विद्युत शक्ति}; किसी अंतर को आधा करना
        अभिकल्पक को क्या क़ीमत देता है?
  \item तीन आँकड़ों में सार दीजिए: मौसम भर में बिजली के प्रति किलोवाट-घंटे
        पर ऊष्मा के कितने किलोवाट-घंटे; गैस के सामने मौसम भर की लागत; और,
        बिजली के प्रति \unit{kW h} पर \qty{60}{g} CO$_2$ तथा गैस के
        \qty{200}{g} के साथ, हर रास्ते से निकला कार्बन।
\end{enumerate}
\end{problem}
