\chapter{किसी शुद्ध पदार्थ के प्रावस्था-परिवर्तन}\label{ch:b1:phase-changes}

पानी वही एक पदार्थ है जिसे सबने तीनों अवस्थाओं में देखा है, और वही दुनिया
के अधिकांश बिजलीघरों का कार्यकारी तरल है, पहली बर्फ़-मशीनों का \omterm{def:b1:heat-engines:vapourcycle}{शीतलक}, और
मौसम का इंजन भी। बर्फ़ ऐसे ताप पर गलती है जो ऊष्मा उँडेलते रहने पर भी टस से
मस नहीं होता; कोई केतली समुद्र-तट पर \qty{100}{\degreeCelsius} पर उबलती है
और एवरेस्ट पर \qty{70}{\degreeCelsius} पर; और बहुत ठंडे पानी की कोई बंद
बोतल एक थपकी से जम जाती है। यह अध्याय किसी \omterm{def:b1:phase-changes:phase}{शुद्ध पदार्थ} की प्रावस्थाओं का
नक़्शा खींचता है --- $(T, P)$ और $(v, P)$ आरेख --- और उसके साथ के तीनों
औज़ार देता है: \omterm{thm:b1:phase-changes:lever}{उत्तोलक नियम} (हर \omterm{def:b1:phase-changes:phase}{प्रावस्था} कितनी है), \omterm{def:b1:phase-changes:phase}{गुप्त ऊष्मा} (किसी
परिवर्तन की क़ीमत, ऊर्जा में और एन्ट्रॉपी में) तथा \omterm{thm:b1:phase-changes:clapeyron}{क्लॉज़ियस--क्लैपेरॉन
संबंध} (सीमा-रेखाओं की ढाल कितनी है)। और वह वहीं समाप्त होता है जहाँ नक़्शा
चूक जाता है: \omterm{def:b1:phase-changes:metastable}{उपस्थायी अवस्थाओं} पर, जहाँ से मौसम और प्रेशर कुकर के
आश्चर्य आते हैं।

\section{प्रावस्थाएँ और $(T, P)$ आरेख}

\begin{definition}[शुद्ध पदार्थ, प्रावस्था, प्रावस्था-परिवर्तन]\label{def:b1:phase-changes:phase}
\index{प्रावस्था}\index{प्रावस्था-परिवर्तन}\index{शुद्ध पदार्थ}\index{गुप्त ऊष्मा}
\emph{शुद्ध पदार्थ} किसी एक ही रासायनिक जाति का पिंड है। कोई
\emph{प्रावस्था} उसका वह भाग है जो अपने भौतिक गुणों में समांगी हो: ठोस,
द्रव, गैस (\emph{वाष्प}, जब वह अपने द्रव के साथ सह-अस्तित्व में हो)। कोई
\emph{प्रावस्था-परिवर्तन} (गलन/जमाव, वाष्पन/
संघनन, ऊर्ध्वपातन/निक्षेपण) एक प्रावस्था से दूसरी में जाना है; नियत दाब पर
वह किसी नियत ताप पर होता है, जिसमें ऊष्मा का आदान-प्रदान तो होता है पर ताप
नहीं बदलता: यही \emph{गुप्त ऊष्मा} है।
\end{definition}

\begin{proposition}[साम्य आरेख]\label{prop:b1:phase-changes:PTdiagram}
\index{प्रावस्था आरेख}\index{त्रिक बिंदु}\index{क्रांतिक बिंदु}\index{संतृप्त वाष्प दाब}
$(T, P)$ तल में हर \omterm{def:b1:phase-changes:phase}{प्रावस्था} एक क्षेत्र घेरती है; दो प्रावस्थाएँ केवल किसी
वक्र के अनुदिश साथ रहती हैं (ऊर्ध्वपातन, गलन, वाष्पन), और तीनों केवल एक ही
बिंदु पर, जिसे \emph{\omterm{prop:b1:phase-changes:PTdiagram}{त्रिक बिंदु}} कहते हैं। वाष्पन-वक्र
$P = P_s(T)$ --- यानी \emph{\omterm{prop:b1:phase-changes:PTdiagram}{संतृप्त वाष्प दाब}} --- \emph{\omterm{prop:b1:phase-changes:PTdiagram}{क्रांतिक बिंदु}}
$(T_c, P_c)$ पर समाप्त हो जाता है, जिसके आगे द्रव और गैस में कोई भेद नहीं
रह जाता। पानी के लिए: \omterm{prop:b1:phase-changes:PTdiagram}{त्रिक बिंदु} \qty{273.16}{K}, \qty{611}{Pa}; और
\omterm{prop:b1:phase-changes:PTdiagram}{क्रांतिक बिंदु} \qty{647}{K}, \qty{221}{bar}। पानी का गलन-वक्र बाईं ओर झुका
है (बर्फ़ पानी से कम सघन है); और लगभग हर दूसरे पदार्थ के लिए वह दाईं ओर
झुकता है।
\end{proposition}
\admitted

\begin{remark}[क्वथन, वाष्पीकरण और प्रेशर कुकर]\label{rem:b1:phase-changes:boiling}
कोई द्रव अपने पृष्ठ से किसी भी ताप पर \emph{वाष्पित} होता रहता है, जब तक
उसके ऊपर उसकी वाष्प का आंशिक दाब $P_s(T)$ से नीचे हो; और वह तब
\emph{उबलता} है जब $P_s(T)$ परिवेश के दाब तक पहुँच जाए, ताकि वाष्प के
बुलबुले उसके भीतर ही बढ़ सकें। पानी \qty{100}{\degreeCelsius} पर इसलिए
उबलता है कि $P_s(\qty{100}{\degreeCelsius}) = \qty{1.013}{bar}$ है; एवरेस्ट
पर \qty{0.33}{bar} पर वह लगभग \qty{70}{\degreeCelsius} पर उबलता है, और किसी
प्रेशर कुकर में \qty{1.8}{bar} पर लगभग \qty{117}{\degreeCelsius} पर। किसी
उबलते द्रव का ताप दाब से जकड़ा रहता है --- यही तथ्य दो सदियों तक सेल्सियस
मापनी की परिभाषा में काम आया।
\end{remark}

\section{$(v, P)$ आरेख और उत्तोलक नियम}

\begin{proposition}[ऐंड्रूज़ के समतापी वक्र]\label{prop:b1:phase-changes:andrews}
\index{ऐंड्रूज़ के समतापी वक्र}\index{संतृप्ति वक्र}
विशिष्ट आयतन के क्लैपेरॉन तल $(v, P)$ में, कोई समतापी वक्र
$T < T_c$ तीन भागों में बँटा होता है: एक खड़ी शाखा (द्रव, जो लगभग
असंपीड्य है), $P = P_s(T)$ पर एक क्षैतिज \emph{पठार}, जहाँ द्रव और वाष्प
साथ रहते हैं, और एक अतिपरवलय-सी शाखा (वाष्प)। पठार के सिरे संतृप्त द्रव तथा
वाष्प के विशिष्ट आयतन $v_l(T)$ और $v_v(T)$ हैं; और उनका बिंदुपथ
\emph{\omterm{prop:b1:phase-changes:andrews}{संतृप्ति वक्र}} (यानी गुंबद) है, जिसका शिखर \omterm{prop:b1:phase-changes:PTdiagram}{क्रांतिक बिंदु} है, जहाँ
समतापी वक्र $T_c$ का नति-परिवर्तन क्षैतिज होता है। $T_c$ के ऊपर कोई पठार
होता ही नहीं।
\end{proposition}
\admitted

\begin{theorem}[उत्तोलक नियम]\label{thm:b1:phase-changes:lever}
\index{उत्तोलक नियम}\index{शुष्कता अंश}
कोई द्रव्यमान $m$, जिसका विशिष्ट आयतन $v$ हो और जो $T$ पर पठार पर हो, वाष्प के
द्रव्यमान-अंश $x = m_v/m$ (यानी \emph{\omterm{thm:b1:phase-changes:lever}{शुष्कता अंश}}) और द्रव के अंश
$1 - x$ में बँट जाता है, जहाँ
\[
v = (1 - x)v_l + x\,v_v , \qquad\text{अर्थात्}\qquad
x = \frac{v - v_l}{v_v - v_l} = \frac{LM}{LV} ,
\]
यानी पठार पर अवस्था-बिंदु $M$ से कटे खंडों का अनुपात। यही नियम प्रति एकांक
द्रव्यमान की हर विस्तीर्ण राशि के लिए लागू है ($u$,
$h$, $s$): $h = (1 - x)h_l + xh_v$।
\end{theorem}

\begin{proof}
आयतन जुड़ते हैं: $V = m_lv_l + m_vv_v$; इसे $m$ से भाग दीजिए। ऊर्जा,
\omterm{def:b1:first-law:enthalpy}{एन्थैल्पी} और एन्ट्रॉपी भी विस्तीर्ण हैं।
\end{proof}

\begin{example}[केतली और बॉयलर]\label{ex:b1:phase-changes:lever}
\qty{100}{\degreeCelsius}, \qty{1}{atm} पर पानी: $v_l = \qty{1.04e-3}{m^3/kg}$,
$v_v = \qty{1.67}{m^3/kg}$ --- यानी $1600$ का अनुपात। उसी ताप पर
\qty{0.50}{kg} पानी रखे किसी बंद \qty{1.0}{L} पात्र के लिए $v =
\qty{2.0e-3}{m^3/kg}$ और $x = (2.0 - 1.04) \times 10^{-3}/1.67 = 5.8 \times 10^{-4}$:
यानी \qty{0.29}{g} वाष्प, जो पात्र का आधा भाग घेरे रहती है। किसी बंद पात्र
में वाष्प का द्रव्यमान दुगुना करने से अवस्था-बिंदु मुश्किल से हिलता है; दाब
केवल ताप से तय होता है।
\end{example}

\section{किसी प्रावस्था-परिवर्तन की ऊर्जा}

\begin{proposition}[गुप्त ऊष्मा, संक्रमण की एन्थैल्पी और एन्ट्रॉपी]\label{prop:b1:phase-changes:latent}
\index{संक्रमण की एन्थैल्पी}\index{संक्रमण की एन्ट्रॉपी}
सह-अस्तित्व के ताप $T$ और दाब $P_s(T)$ पर, एकांक द्रव्यमान का \omterm{def:b1:phase-changes:phase}{प्रावस्था} 1
से \omterm{def:b1:phase-changes:phase}{प्रावस्था} 2 में संक्रमण उत्क्रमणीय, समतापी और समदाबी होता है, और
\[
L_{12}(T) = h_2 - h_1 , \qquad s_2 - s_1 = \frac{L_{12}}{T} , \qquad
u_2 - u_1 = L_{12} - P_s(v_2 - v_1) .
\]
कम व्यवस्थित \omterm{def:b1:phase-changes:phase}{प्रावस्था} की ओर जाने पर $L_{12} > 0$ होता है (गलन,
वाष्पन, ऊर्ध्वपातन), और $L_{21} = -L_{12}$। \qty{1}{atm} पर पानी के लिए:
\qty{0}{\degreeCelsius} पर $L_f = \qty{334}{kJ/kg}$, और
\qty{100}{\degreeCelsius} पर $L_v = \qty{2257}{kJ/kg}$।
\end{proposition}

\begin{proof}
अचर दाब पर मिली ऊष्मा $\Delta H$ है (\cref{prop:b1:first-law:qvqp});
उत्क्रमणीय तथा समतापी होने से वह $T\Delta S$
(\cref{prop:b1:second-law-entropy:identity}); और $U = H - PV$।
\end{proof}

\begin{example}[वाष्पन की ऊष्मा जाती कहाँ है]\label{ex:b1:phase-changes:whereheat}
\qty{100}{\degreeCelsius} पर पानी के लिए: $P_s(v_v - v_l) = 1.013 \times 10^5 \times
1.67 = \qty{169}{kJ/kg}$, अतः $u_v - u_l = 2257 - 169 = \qty{2088}{kJ/kg}$:
यानी \omterm{def:b1:phase-changes:phase}{गुप्त ऊष्मा} का $93\%$ अणुओं को एक-दूसरे से नोचने में जाता है और $7\%$
वायुमंडल को पीछे धकेलने में। वाष्पन की एन्ट्रॉपी,
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, गलन की $334/273 =
\qty{1.22}{kJ/(kg.K)}$ के सामने बहुत बड़ी है: गलन अणुओं की जकड़ ढीली करता है,
क्वथन उन्हें मुक्त कर देता है।
\end{example}

\begin{method}[प्रावस्था-परिवर्तनों सहित ऊष्मामिति]\label{meth:b1:phase-changes:calorimetry}
अचर दाब पर किसी रोधी पात्र में, $\sum\Delta H_i = 0$। अंतिम अवस्था का
अनुमान लगाइए (कौन-सी प्रावस्थाएँ मौजूद हैं, किस ताप पर); हर पिंड का
$\Delta H$ सुबोध ऊष्माओं $mc\,\Delta T$ के रूप में लिखिए, साथ में उन
परिवर्तनों की गुप्त ऊष्माएँ $\pm mL$ जो वह झेलता है; हल कीजिए; और
\emph{अनुमान जाँचिए}: कोई पिंड यदि अपने क्वथनांक से ऊपर या गलनांक से नीचे
समाप्त हो, या किसी \omterm{def:b1:phase-changes:phase}{प्रावस्था} का द्रव्यमान ऋणात्मक निकले, तो अनुमान ग़लत था
--- तब अंतिम अवस्था संक्रमण-ताप पर कोई मिश्रण है, और अज्ञात राशि वह
द्रव्यमान बन जाती है जो बदला (\cref{ex:b1:first-law:ice})।
\end{method}

\begin{example}[भाप जलाती है]\label{ex:b1:phase-changes:steam}
\qty{100}{\degreeCelsius} की \qty{10}{g} भाप त्वचा पर संघनित होकर एक डिग्री
ठंडी होने से पहले ही $0.010 \times 2257 = \qty{22.6}{kJ}$ दे देती है ---
यानी उतना ही जितना \qty{10}{g} खौलता पानी \qty{540}{K} ठंडा होकर देता।
इसीलिए भाप उसी ताप के पानी से कहीं बुरी तरह झुलसाती है, और इसीलिए किसी
भाप-रेडिएटर को इतना कम द्रव्यमान-प्रवाह चाहिए।
\end{example}

\section{क्लॉज़ियस--क्लैपेरॉन संबंध}

\begin{theorem}[क्लॉज़ियस--क्लैपेरॉन]\label{thm:b1:phase-changes:clapeyron}
\index{क्लॉज़ियस--क्लैपेरॉन संबंध}
प्रावस्थाओं 1 और 2 के किसी सह-अस्तित्व वक्र के अनुदिश,
\[
\frac{\dd P_s}{\dd T} = \frac{L_{12}(T)}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
एकांक द्रव्यमान को गुंबद के भीतर किसी अत्यल्प \omterm{def:b1:heat-engines:carnotcycle}{कार्नो चक्र} पर घुमाइए: पूरा
संक्रमण $1 \to 2$, $T + \dd T$ ताप और $P_s + \dd P_s$ दाब पर के पठार के
अनुदिश (प्रथम कोटि तक मिली ऊष्मा $L_{12}$); फिर $T$ तक कोई अत्यल्प
रुद्धोष्म चरण; फिर $T$, $P_s$ पर के पठार के अनुदिश उलटा संक्रमण; और फिर
वापस कोई रुद्धोष्म चरण। यह चक्र उत्क्रमणीय है, $T + \dd T$ और $T$ के बीच
द्वितापी, अतः उसकी \omterm{def:b1:heat-engines:efficiency}{दक्षता} $\dd T/T$ है
(\cref{thm:b1:heat-engines:carnot}); और उसका कार्य, यानी उस पतले आयत का
क्षेत्रफल, $\dd P_s\,(v_2 - v_1)$ है। अतः $\dd P_s\,(v_2 - v_1)/L_{12} =
\dd T/T$।
\end{proof}

\begin{corollary}[किसी द्रव का संतृप्त वाष्प दाब]\label{cor:b1:phase-changes:vapourpressure}
यदि वाष्प कोई \omterm{def:b1:kinetic-theory:model}{आदर्श गैस} हो और $v_v \gg v_l$, ताकि $v_v - v_l \approx
RT/(MP_s)$, और यदि $L_v$ कम ही बदले, तो
\[
\frac{\dd P_s}{P_s} = \frac{ML_v}{R}\,\frac{\dd T}{T^2} , \qquad
\ln\frac{P_s(T)}{P_s(T_0)} = \frac{ML_v}{R}\Bigl(\frac{1}{T_0} - \frac{1}{T}\Bigr) :
\]
यानी वाष्प दाब ताप के साथ मोटे तौर पर चरघातांकी रूप से बढ़ता है।
\qty{100}{\degreeCelsius} के पास पानी के लिए, $ML_v/R = 0.018 \times 2.257 \times 10^6/
8.314 = \qty{4890}{K}$ और $\dd P_s/\dd T = 2.257 \times 10^6/(373 \times 1.67) =
\qty{3.6}{kPa/K}$: यानी क्वथन-ताप के हर अतिरिक्त \omterm{def:b1:kinetic-theory:temperature}{केल्विन} की क़ीमत
\qty{36}{mbar} अतिरिक्त दाब है।
\end{corollary}

\begin{proof}
प्रमेय में $v_v - v_l \approx RT/(MP_s)$ रखिए और चर अलग कीजिए; फिर
$L_v$ को अचर मानकर समाकलित कीजिए।
\end{proof}

\end{example}

\section{उपस्थायी अवस्थाएँ}

\begin{definition}[अतिशीतलन, अतितापन]\label{def:b1:phase-changes:metastable}
\index{अतिशीतलन}\index{अतितापन}\index{उपस्थायी अवस्था}\index{नाभिकन}
कोई \omterm{def:b1:phase-changes:phase}{प्रावस्था} अपनी साम्य-सीमा के आगे भी बनी रह सकती है: द्रव पानी
\qty{0}{\degreeCelsius} से नीचे बिना जमे ठंडा (\emph{अतिशीतलन}), अपने
क्वथनांक से ऊपर बिना उबले गरम (\emph{अतितापन}), और वाष्प $P_s$ से आगे बिना
संघनित हुए संपीडित (अतिसंतृप्ति)। ऐसी \emph{उपस्थायी अवस्थाओं} के लिए
\emph{नाभिकन} स्थलों का न होना ज़रूरी है --- धूल, खरोंच, घुली हुई गैस,
बर्फ़ के रवे --- जिन पर नई \omterm{def:b1:phase-changes:phase}{प्रावस्था} शुरू हो सके; और कोई भी विक्षोभ उन्हें
अचानक तथा अनुत्क्रमणीय ढंग से साम्य की ओर ढकेल देता है।
\end{definition}

\begin{example}[अतिशीतित बोतल]\label{ex:b1:phase-changes:supercooled}
किसी साफ़ बोतल में \qty{-8}{\degreeCelsius} पर शांत पड़ा पानी, थपकी लगने पर:
बर्फ़ के रवे उसमें दौड़ जाते हैं और ताप उछलकर
\qty{0}{\degreeCelsius} पर आ जाता है। यह प्रक्रिया रुद्धोष्म और तेज़ है,
अतः सुबोध ऊष्मा $c\,\Delta T$ पानी के $x = c\,\Delta T/L_f = 4.18 \times 8/334 =
0.10$ अंश को जमा देती है; और सृजित एन्ट्रॉपी, $c\ln(273/265) - xL_f/273 =
124 - 122 = \qty{+2}{J/(kg.K)}$, धनात्मक है, जैसा होना ही चाहिए। बादल
\qty{-40}{\degreeCelsius} तक की अतिशीतित बूँदों से भरे रहते हैं; और
विमानों पर बर्फ़ जमना तथा बादलों का बीजारोपण, दोनों इसी का लाभ उठाते हैं।
\end{example}

\begin{remark}[क्वथन-कण और बुलबुला कक्ष]\label{rem:b1:phase-changes:nucleation}
किसी द्रव में $r$ त्रिज्या का कोई बुलबुला, जब द्रव दाब $P$ पर हो,
$P + 2\sigma/r$ पर वाष्प रखता है (पृष्ठ तनाव $\sigma$, वर्ष 2 का खंड); और
वह तभी बढ़ सकता है जब $P_s(T) > P + 2\sigma/r$ हो, अतः अपने क्वथनांक पर पड़े
किसी साफ़ द्रव में शुरू होने लायक कोई छोटा बुलबुला होता ही नहीं और वह
फूटने से पहले कई \omterm{def:b1:kinetic-theory:temperature}{केल्विन} तक अतितापित हो जाता है --- यही वह ``उछाल'' है जिसे
क्वथन-कण गैस से भरी गुहाएँ देकर रोक लेते हैं। कण-भौतिकी का बुलबुला कक्ष
उसी अस्थायित्व को उलटकर चलता था: कोई अतितापित द्रव सबसे पहले किसी आयनकारी
कण के पथ के अनुदिश उबलता है।
\end{remark}

\begin{example}[आर्द्रता और ओस]\label{ex:b1:phase-changes:dew}
\index{आपेक्षिक आर्द्रता}\index{ओसांक}
हवा जल-वाष्प को किसी आंशिक दाब $P_v \leq P_s(T)$ पर रखती है;
\emph{\omterm{ex:b1:phase-changes:dew}{आपेक्षिक आर्द्रता}} $P_v/P_s(T)$ है, और \emph{ओसांक} वह ताप जिस पर
$P_s$ गिरकर $P_v$ हो जाए। \qty{25}{\degreeCelsius} की हवा
($P_s = \qty{3.17}{kPa}$), जिसकी आर्द्रता $60\%$ हो, उसमें $P_v = \qty{1.9}{kPa}$,
$\rho_v = P_vM/(RT) = \qty{14}{g/m^3}$ होता है, और वह लगभग
\qty{17}{\degreeCelsius} से नीचे के किसी भी पृष्ठ पर ओस बना देती है ---
भोर की घास, ठंडी बोतल, स्नानघर का दर्पण। ऊपर उठने पर वही हवा रुद्धोष्म रूप
से \qty{9.8}{K/km} की दर से ठंडी होती है (\cref{pb:b1:phase-changes:1}) और
लगभग एक किलोमीटर ऊपर अपने ओसांक पर पहुँच जाती है: वहीं कपासी मेघों के चपटे
तल होते हैं।
\end{example}

\section{समस्या: चार दृश्यों में पानी}

\begin{problem}[{सप्ताहांत समस्या --- प्रेशर कुकर, बादल, बंद पात्र और
अतिशीतित बोतल: एक ही पदार्थ, और उसके प्रावस्था आरेख के चार
उपयोग}]
\label{pb:b1:phase-changes:1}
पानी के आँकड़े: \qty{100}{\degreeCelsius} पर $L_v = \qty{2257}{kJ/kg}$,
\qty{117}{\degreeCelsius} पर लगभग \qty{2.2}{MJ/kg}, और \qty{20}{\degreeCelsius}
के पास \qty{2.45}{MJ/kg}; $L_f = \qty{334}{kJ/kg}$; $c = \qty{4.18}{kJ/(kg.K)}$
(द्रव); $M = \qty{18}{g/mol}$; \qty{100}{\degreeCelsius} के पास
$ML_v/R = \qty{4890}{K}$। संतृप्ति दाब: \qty{0.61}{kPa}
(\qty{0}{\degreeCelsius}), \qty{1.23} (\qty{10}), \qty{1.70} (\qty{15}),
\qty{2.34} (\qty{20}), \qty{3.17} (\qty{25}), \qty{4.24}{kPa}
(\qty{30}{\degreeCelsius}), \qty{101.3}{kPa} (\qty{100}{\degreeCelsius})।
संतृप्त विशिष्ट आयतन: $v_l = \qty{1.0e-3}{m^3/kg}$ (ठंडा पानी),
\qty{360}{\degreeCelsius} पर \qty{1.9e-3}{m^3/kg}; \qty{300}{\degreeCelsius}
पर $v_v = \qty{0.0217}{m^3/kg}$, \qty{340}{\degreeCelsius} पर $0.0108$,
\qty{350}{\degreeCelsius} पर $0.0088$; \omterm{prop:b1:phase-changes:PTdiagram}{क्रांतिक बिंदु} \qty{374}{\degreeCelsius},
\qty{221}{bar}, $v_c = \qty{3.1e-3}{m^3/kg}$। हवा: $M = \qty{29}{g/mol}$, $c_p =
\qty{1.0}{kJ/(kg.K)}$, $\gamma = 1.4$; $g = \qty{9.8}{m/s^2}$; $R =
\qty{8.314}{J/(mol.K)}$।

\medskip
\noindent\textbf{भाग I --- प्रेशर कुकर।}
\begin{enumerate}
  \item समुद्र-तल पर खुले किसी बरतन में पानी \qty{100}{\degreeCelsius} से
        ऊपर क्यों नहीं गरम किया जा सकता, चाहे लौ कितनी भी तेज़ हो?
  \item \omterm{thm:b1:phase-changes:clapeyron}{क्लॉज़ियस--क्लैपेरॉन संबंध} से चलकर
        $\ln[P_s(T)/P_s(T_0)] = (ML_v/R)(1/T_0 - 1/T)$ व्युत्पन्न कीजिए, और
        सन्निकटन बताइए।
  \item उस कुकर के भीतर का ताप जिसका कपाट \qty{1.8}{bar} निरपेक्ष पर खुलता
        है।
  \item कपाट \qty{20}{mm^2} के किसी छिद्र पर टिका कोई भार है; वायुमंडलीय
        दाब \qty{1.0}{bar}: भार का द्रव्यमान।
  \item \qty{2.0}{L} पानी \qty{20}{\degreeCelsius} से काम के ताप तक
        \qty{2.0}{kW} की किसी प्लेट से लाया जाता है (बरतन की उपेक्षा
        कीजिए): ऊष्मा और समय।
  \item वहाँ पहुँचने के बाद प्लेट \qty{2.0}{kW} पर ही छोड़ दी जाती है:
        प्रति मिनट कपाट से निकलती भाप का द्रव्यमान। रसोइये को क्या करना
        चाहिए?
  \item पकाने की अभिक्रियाएँ हर \qty{10}{K} पर मोटे तौर पर दुगुनी तेज़ हो
        जाती हैं: कुकर समुद्र-तल की किसी \qty{30}{min} वाली विधि को कितने
        गुना छोटा कर देता है? और वही विधि एवरेस्ट पर, जहाँ पानी
        \qty{70}{\degreeCelsius} पर उबलता है, कितनी देर लेती?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- बादल।}
\begin{enumerate}[resume]
  \item \omterm{ex:b1:phase-changes:dew}{आपेक्षिक आर्द्रता} को परिभाषित कीजिए। \qty{25}{\degreeCelsius} और $60\%$
        की हवा: वाष्प का आंशिक दाब, और प्रति घन मीटर वाष्प का द्रव्यमान
        (\omterm{def:b1:kinetic-theory:model}{आदर्श गैस})।
  \item उस हवा का ओसांक (सारणी में अंतर्वेशन कीजिए)। भोर में घास पर ओस और
        ठंडी बोतल पर बूँदों को समझाइए।
  \item सूखी हवा का कोई पिंड बिना ऊष्मा का आदान-प्रदान किए ऊपर उठता है।
        उत्क्रमणीय रुद्धोष्म नियम और \omterm{thm:b1:fluid-statics:fundamental}{द्रवस्थैतिक संबंध}
        $\dd P/\dd z = -\rho g$ से दिखाइए कि $\dd T/\dd z = -g/c_p$ है, और
        उसका मान निकालिए।
  \item ऊपर उठते पिंड का अपना ओसांक भी, दाब गिरने के साथ, लगभग
        \qty{2}{K/km} से गिरता है। वह ऊँचाई जिस पर प्रश्न~8 का पिंड
        संतृप्त हो जाता है --- यानी बादल का तल।
  \item तल के ऊपर, जो वाष्प संघनित होती है वह अपनी \omterm{def:b1:phase-changes:phase}{गुप्त ऊष्मा} पिंड में
        छोड़ देती है। प्रति किलोग्राम हवा पर \qty{1}{g} पानी के संघनन से वह
        हवा कितनी गरम हो जाती है? ऊपर उठते रहने पर इससे पिंड के ठंडा होने
        की दर पर क्या असर पड़ता है, और यह गरजते तूफ़ानों के लिए क्यों
        मायने रखता है?
  \item \qty{1}{km^3} का कोई कपासी मेघ प्रति घन मीटर \qty{1}{g} द्रव पानी
        रखता है: पानी का द्रव्यमान, और उसके संघनन से निकली ऊर्जा, जूल में
        तथा \unit{GW h} में।
  \item स्नानघर का कोई दर्पण \qty{18}{\degreeCelsius} पर, और हवा
        \qty{25}{\degreeCelsius} तथा $80\%$ पर: क्या उस पर भाप जमेगी? किस
        आर्द्रता पर वह रुक जाती?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बंद पात्र।}
\qty{1.0}{L} का कोई दृढ़ इस्पात-पात्र $m$ द्रव्यमान के पानी से भरकर बिना
किसी हवा के बंद कर दिया जाता है, फिर धीरे-धीरे गरम किया जाता है।
\begin{enumerate}[resume]
  \item \qty{20}{\degreeCelsius} पर $m = \qty{0.50}{kg}$: भीतर का दाब,
        विशिष्ट आयतन, वाष्प का द्रव्यमान ($v_v$ के लिए \omterm{def:b1:kinetic-theory:model}{आदर्श गैस} लीजिए),
        और आयतन का वह अंश जो द्रव भरता है।
  \item गरम करने पर: द्रव का तल ऊपर उठता है या नीचे गिरता है? मोटे तौर पर
        किस ताप पर द्रव पूरा पात्र भर देता है, और उसके बाद दाब का क्या
        होता है? (\qty{360}{\degreeCelsius} पर $v_l$ लीजिए।)
  \item $m = \qty{0.10}{kg}$: गरम करने पर द्रव का तल ऊपर उठता है या नीचे
        गिरता है? मोटे तौर पर किस ताप पर आख़िरी बूँद वाष्पित हो जाती है
        ($v_v$ की सारणी लीजिए)?
  \item $m = \qty{0.31}{kg}$: वर्णन कीजिए कि ताप के \qty{374}{\degreeCelsius}
        के पास पहुँचते ही कोई प्रेक्षक नवचंद्रक पर क्या देखता है।
  \item संतृप्ति गुंबद सहित $(v, P)$ आरेख में तीनों तापन-पथ खींचिए, और
        बताइए कि हर एक \omterm{prop:b1:phase-changes:PTdiagram}{क्रांतिक बिंदु} की किस ओर से गुज़रता है।
  \item वही \qty{1.0}{L} पात्र \qty{0.50}{kg} पानी के साथ अब उस हवा
        \emph{सहित} बंद किया जाता है जो उसमें \qty{20}{\degreeCelsius},
        \qty{1.0}{bar} पर थी, और \qty{100}{\degreeCelsius} तक गरम किया जाता
        है: भीतर का कुल दाब (डाल्टन), और \qty{100}{cm^2} के किसी ढक्कन पर बल।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- अतिशीतित बोतल।}
\begin{enumerate}[resume]
  \item पानी की कोई साफ़, शांत बोतल बिना जमे \qty{-8}{\degreeCelsius} तक
        ठंडी हो चुकी है। इस अवस्था को क्या कहते हैं, वह संभव क्यों है, और
        बोतल पर थपकी लगने पर क्या होता है? तब ताप ठीक
        \qty{0}{\degreeCelsius} पर ही क्यों ठहरता है?
  \item पानी का वह अंश जो जम जाता है (प्रक्रिया तेज़ है, अतः रुद्धोष्म)।
  \item प्रति किलोग्राम सृजित एन्ट्रॉपी; चिह्न जाँचिए।
  \item उलटा आश्चर्य: किसी माइक्रोवेव भट्ठी में \qty{105}{\degreeCelsius}
        तक बिना उबले गरम किए गए पानी का कोई प्याला, फिर छेड़ा गया। पानी का
        वह अंश जो झटके में भाप बन जाता है, प्रति लीटर बनी भाप का आयतन
        ($v_v = \qty{1.67}{m^3/kg}$), और प्याला क्यों फूट पड़ता है।
  \item \omterm{thm:b1:heat-engines:lostwork}{खोया कार्य}: वह अतिशीतित लीटर \qty{0}{\degreeCelsius} के परिवेश के
        साथ कोई उत्क्रमणीय इंजन चला सकता था; और
        $T_0 = \qty{273}{K}$ लेने पर, किसी थपकी से उसे जम जाने देकर फेंका
        गया कार्य $T_0S_{\mathrm{created}}$ है। उसे प्रति लीटर निकालिए और
        बोतल को एक मीटर उठाने से तुलना कीजिए। निष्कर्ष निकालिए: कोई
        \omterm{def:b1:phase-changes:metastable}{उपस्थायी अवस्था} संचित क्या रखती है?
\end{enumerate}
\end{problem}
