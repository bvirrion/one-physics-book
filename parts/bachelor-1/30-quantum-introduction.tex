\chapter{Introduction to Quantum Physics}\label{ch:b1:quantum-introduction}

Shine ultraviolet light on a clean metal and electrons fly out at
once, with an energy that depends on the colour of the light and not at
all on its brightness; shine red light and nothing comes out however
bright the lamp. Send electrons one by one through two slits and they
land as dots, each in one place --- and the dots build up, over
thousands, into the fringes of a wave. Neither fact fits the physics
of the first twenty-nine chapters. This last chapter introduces the
ideas that do fit them: light comes in quanta, the photons; matter has
a wavelength; what propagates is an \omterm{def:b1:wave-propagation:signal}{amplitude} whose square is a
probability; position and momentum cannot both be sharp; and a
confined particle can only have certain energies --- which is why atoms
have sizes, spectra have lines, and a grain of semiconductor a few
nanometres across glows in a colour set by its diameter.

\section{The photon}

\begin{theorem}[Planck--Einstein relations]\label{thm:b1:quantum-introduction:photon}
\index{photon}\index{Planck's constant}
Light of frequency $\nu$ and wavelength $\lambda$ exchanges energy with
matter only in quanta, the \emph{photons}, of energy and momentum
\[
E = h\nu = \frac{hc}{\lambda} , \qquad p = \frac{h}{\lambda} = \frac{E}{c} ,
\]
with \emph{\omterm{thm:b1:quantum-introduction:photon}{Planck's constant}} $h = \qty{6.626e-34}{J.s}$ ($\hbar = h/2\pi =
\qty{1.055e-34}{J.s}$). Useful: $hc = \qty{1240}{eV.nm}$ --- a photon of
\qty{620}{nm} carries \qty{2}{eV}.
\end{theorem}
\admitted

\begin{proposition}[The photoelectric effect]\label{prop:b1:quantum-introduction:photoelectric}
\index{photoelectric effect}\index{work function}\index{stopping potential}
Light falling on a metal of \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{work function}} $W$ (the energy
binding its least-bound electrons, a few eV) extracts electrons of
maximal \omterm{thm:b1:work-and-energy:ke}{kinetic energy}
\[
E_{k,\max} = h\nu - W ,
\]
hence only above the \emph{threshold} $\nu_0 = W/h$, instantly, with an
energy independent of the intensity; the intensity fixes the
\emph{number} of electrons per second. The \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{stopping potential}}
$V_s$ that just cancels the current measures $E_{k,\max} = eV_s$, and
the plot of $V_s$ against $\nu$ is a straight line of slope $h/e$ --- how
Millikan measured $h$.
\end{proposition}

\begin{proof}
Each photon is absorbed by one electron, which keeps $h\nu - W$ at
best; a wave would deliver energy continuously to every electron, the
energy would grow with intensity, and a weak light would need hours to
accumulate $W$ on one atom (\cref{pb:b1:quantum-introduction:1}) ---
none of which is observed. Einstein's interpretation (1905) is admitted
as the law.
\end{proof}

\begin{example}[Orders of magnitude]\label{ex:b1:quantum-introduction:orders}
A visible photon: \qty{2}{eV}; an X-ray photon of \qty{0.1}{nm}: \qty{12}{keV};
a radio photon at \qty{100}{MHz}: \qty{4e-7}{eV}. A \qty{1}{mW} red laser
emits $10^{-3}/(3.1 \times 10^{-19}) = 3 \times 10^{15}$ photons per second;
a radio antenna picking up \qty{1}{pW} at \qty{100}{MHz} receives $10^{13}$
per second --- so many, so small, that the wave description is exact
for every purpose; the \omterm{def:b1:optical-instruments:eye}{eye}, by contrast, can register a flash of a
hundred photons. The quantum nature of light shows when photons are
few or energetic: photoelectric cells, X-ray detectors, the grain of a
faint photograph.
\end{example}

\begin{remark}[Compton scattering]\label{rem:b1:quantum-introduction:compton}
\index{Compton scattering}
A photon bouncing off a free electron transfers momentum like a
billiard ball: its wavelength grows by $\Delta\lambda = \frac{h}{m_ec}(1 -
\cos\theta)$, with $h/m_ec = \qty{2.43}{pm}$ (admitted --- a relativistic
collision, Year 2). Invisible for light ($\Delta\lambda/\lambda \sim 10^{-5}$),
it is a $20\%$ effect for X-rays, and was the experiment that convinced
the sceptics (1923) that the photon carries momentum $h/\lambda$.
\end{remark}

\section{Matter waves}

\begin{theorem}[de Broglie]\label{thm:b1:quantum-introduction:debroglie}
\index{de Broglie wavelength}\index{matter wave}
A particle of momentum $p$ propagates as a wave of wavelength
\[
\lambda = \frac{h}{p} ,
\]
and shows diffraction and interference accordingly: electrons on a
crystal (Davisson and Germer, 1927), through two slits (Jönsson, 1961),
neutrons, atoms, and molecules of sixty carbon atoms. For an electron
accelerated by $U$, $\lambda = h/\sqrt{2m_eeU} = \qty{1.23}{nm}/\sqrt{U/\unit{V}}$.
\end{theorem}
\admitted

\begin{example}[Why we never saw it]\label{ex:b1:quantum-introduction:wavelengths}
An electron of \qty{100}{eV}: $\lambda = \qty{0.12}{nm}$, the spacing of
atoms in a crystal --- hence the diffraction, and the electron
\omterm{def:b1:optical-instruments:microscope}{microscope}. A thermal neutron (\qty{0.025}{eV}): \qty{0.18}{nm}. A ball
of \qty{1}{g} at \qty{1}{m/s}: $\lambda = \qty{7e-31}{m}$, $10^{20}$ times
smaller than a nucleus: no slit, no crystal could ever diffract it. The
wave nature of matter is universal; its wavelength is what makes it
visible only for the light and the slow.
\end{example}

\begin{proposition}[Single particles build the fringes]\label{prop:b1:quantum-introduction:buildup}
\index{wave--particle duality}
Electrons sent one at a time through two slits (Tonomura, 1989) each
make a single dot on the screen, at a place that cannot be predicted;
the dots accumulate into the two-slit interference pattern of a wave
of wavelength $h/p$. Closing one slit erases the fringes --- the
pattern is not made by electrons interfering with each other, but by
each electron's wave passing through both slits. Detecting which slit
a particle went through also erases the fringes.
\end{proposition}
\admitted

\section{Amplitudes, probabilities, and the uncertainty principle}

\begin{definition}[Wavefunction]\label{def:b1:quantum-introduction:wavefunction}
\index{wavefunction}\index{probability density}
The state of a particle is described by a \emph{wavefunction}
$\psi(x, t)$ (complex), such that $|\psi(x, t)|^2\,\dd x$ is the
probability of finding the particle between $x$ and $x + \dd x$ when one
looks: $|\psi|^2$ is a \emph{probability density}, and $\int|\psi|^2\dd x =
1$. \omterm{def:b1:wave-propagation:signal}{Amplitudes} add (superposition): if two paths lead to the same
point, $\psi = \psi_1 + \psi_2$ and $|\psi|^2 = |\psi_1|^2 + |\psi_2|^2 +
2\,\mathrm{Re}(\psi_1^\ast\psi_2)$ --- the last term is the interference.
A measurement gives one of the possible results, at random with the
probabilities the \omterm{def:b1:wave-propagation:signal}{amplitudes} prescribe, and leaves the particle in the
state found.
\end{definition}

\begin{remark}[What is waving]\label{rem:b1:quantum-introduction:whatwaves}
$\psi$ is not a physical field like $\vect E$: it is not measured, only
its square is, and it describes our best possible knowledge of where
the particle will be found. A single electron is not spread over the
screen; it is detected whole, at one point; what is spread is the
probability. This is the content of the theory, and it has never failed
a test.
\end{remark}

\begin{theorem}[Heisenberg's uncertainty principle]\label{thm:b1:quantum-introduction:heisenberg}
\index{uncertainty principle}
The position and the momentum of a particle along the same axis cannot
both be sharply defined: their standard deviations in any state obey
\[
\Delta x\,\Delta p_x \geq \frac{\hbar}{2} .
\]
A particle confined to a region of size $\ell$ therefore has a momentum
spread of at least $\hbar/2\ell$, hence a \omterm{thm:b1:work-and-energy:ke}{kinetic energy} of order
$\hbar^2/2m\ell^2$ that no cooling can remove: confinement costs energy.
\end{theorem}
\admitted

\begin{example}[The size of the hydrogen atom]\label{ex:b1:quantum-introduction:hsize}
\index{Bohr radius}
An electron within $r$ of the proton has $p \gtrsim \hbar/r$ and energy
$E(r) \approx \dfrac{\hbar^2}{2m_er^2} - \dfrac{e^2}{4\pi\varepsilon_0r}$: the first
term blows up if $r$ shrinks, the second if it grows. $\dd E/\dd r = 0$ at
\[
a_0 = \frac{4\pi\varepsilon_0\hbar^2}{m_ee^2} = \qty{53}{pm} , \qquad
E(a_0) = -\frac{m_ee^4}{2(4\pi\varepsilon_0)^2\hbar^2} = \qty{-13.6}{eV} :
\]
the \emph{\omterm{ex:b1:quantum-introduction:hsize}{Bohr radius}} and the ionization energy of hydrogen, to the
last digit. Classical physics could not explain why the electron does
not spiral into the nucleus (it radiates) nor why all atoms of an
element are identical; the uncertainty principle answers both in one
line: the atom is as small as it can be without the \omterm{thm:b1:work-and-energy:ke}{kinetic energy}
exceeding the binding.
\end{example}

\section{Quantization of energy}

\begin{proposition}[Particle in a box]\label{prop:b1:quantum-introduction:box}
\index{infinite well}\index{quantization}\index{ground state}
A particle of mass $m$ confined between two impenetrable walls a
distance $L$ apart has, like Melde's string
(\cref{ch:b1:wave-propagation}), only the \omterm{thm:b1:wave-propagation:standing}{standing waves} $\lambda_n = 2L/n$,
hence only the momenta $p_n = nh/2L$ and the energies
\[
E_n = \frac{n^2h^2}{8mL^2} , \qquad n = 1, 2, 3, \ldots
\]
with \omterm{def:b1:quantum-introduction:wavefunction}{wavefunctions} $\psi_n(x) \propto \sin(n\pi x/L)$. The energy is
\emph{quantized}; the lowest, the \emph{\omterm{prop:b1:quantum-introduction:box}{ground state}} $E_1 > 0$, is the
confinement energy of the uncertainty principle made exact; the
particle jumps between levels by absorbing or emitting a photon of
energy $h\nu = E_p - E_n$.
\end{proposition}

\begin{proof}
$\psi$ must vanish at the walls (the particle cannot be there), so the
wave fits a whole number of half-wavelengths in $L$ --- the string
condition; de Broglie gives $p$, and $E = p^2/2m$. That $\psi$ obeys
a wave equation with these boundary conditions is admitted (the
Schrödinger equation, Year 2).
\end{proof}

\begin{proposition}[The hydrogen atom]\label{prop:b1:quantum-introduction:hydrogen}
\index{hydrogen atom}\index{energy levels}\index{spectral lines}
The electron of the \omterm{prop:b1:quantum-introduction:hydrogen}{hydrogen atom} has the discrete energies
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2} , \qquad n = 1, 2, 3, \ldots ,
\]
$n = 1$ being the \omterm{prop:b1:quantum-introduction:box}{ground state} and $E \geq 0$ the ionized atom. Its
spectrum consists of lines at $h\nu = E_p - E_n$: the Lyman series down
to $n = 1$ (ultraviolet, \qty{122}{nm} for $2 \to 1$), the Balmer series
down to $n = 2$ (visible: \qty{656}{nm}, \qty{486}{nm}, \qty{434}{nm}, \ldots),
and so on. Bohr's model (1913) obtains these energies by quantizing the
\omterm{def:b1:angular-momentum:L}{angular momentum} of a \omterm{prop:b1:central-forces:effective}{circular orbit}, $m_evr = n\hbar$
(\cref{exo:b1:quantum-introduction:9}); the full theory replaces orbits by
\omterm{def:b1:quantum-introduction:wavefunction}{wavefunctions} but keeps the energies.
\end{proposition}
\admitted

\begin{example}[Colours from size: the quantum dot]\label{ex:b1:quantum-introduction:dot}
\index{quantum dot}
A nanocrystal of semiconductor a few nanometres across confines its
electrons like a box: the smaller the crystal, the larger the
confinement energy added to the semiconductor's gap, the bluer the
light it emits. Cadmium selenide dots of radius \qty{2}{nm} glow blue,
\qty{3}{nm} yellow, \qty{4}{nm} red --- one material, colours chosen with
a ruler (\cref{pb:b1:quantum-introduction:1}). Television screens use them.
\end{example}

\begin{remark}[What the rest of the theory adds]\label{rem:b1:quantum-introduction:rest}
\index{tunnel effect}\index{laser}
The Schrödinger equation makes the box, the atom and the molecule
exact; it predicts that a particle can cross a barrier it has not the
energy to climb (the \emph{\omterm{rem:b1:quantum-introduction:rest}{tunnel effect}}, behind radioactivity and the
scanning tunnelling \omterm{def:b1:optical-instruments:microscope}{microscope}); that identical particles are either
sociable (bosons --- hence the \emph{laser}, where photons pile into one
mode) or exclusive (fermions --- hence the periodic table and the
stiffness of matter); and that the spin of the electron is a \omterm{def:b1:magnetostatics:moment}{magnetic
moment} with no classical counterpart. All of it is Year 2 and beyond;
all of it rests on the few ideas of this chapter.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:1}
Energy of a photon of: radio at \qty{100}{MHz}; light at \qty{500}{nm};
X-rays at \qty{0.10}{nm}; gamma rays of \qty{1}{MeV} (give its
wavelength). Number of photons per second from a \qty{1.0}{mW} laser at
\qty{633}{nm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:2}
Sodium, $W = \qty{2.3}{eV}$: threshold wavelength; maximal \omterm{thm:b1:work-and-energy:ke}{kinetic energy}
and \omterm{prop:b1:quantum-introduction:photoelectric}{stopping potential} for \qty{400}{nm} light; what changes if the
intensity is doubled? and with \qty{600}{nm} light?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:3}
\omterm{thm:b1:quantum-introduction:debroglie}{De Broglie wavelength} of an electron of \qty{100}{eV}, of \qty{1.0}{keV};
of a neutron of \qty{0.025}{eV}; of a \qty{1}{g} ball at \qty{1}{m/s}.
Which of these can be diffracted by a crystal (\qty{0.2}{nm} spacing)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:4}
An electron in a box of \qty{1.0}{nm}: $E_1$, $E_2$, wavelength of the
photon emitted from $2 \to 1$. A proton in a box of \qty{5}{fm}
(a nucleus): $E_1$ in \unit{MeV}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:5}
\omterm{prop:b1:quantum-introduction:photoelectric}{Stopping potentials} measured for a metal: \qty{0.40}{V} at
\qty{6.0e14}{Hz}, \qty{1.23}{V} at \qty{8.0e14}{Hz}, \qty{2.05}{V} at
\qty{1.0e15}{Hz}. Deduce $h$ (with $e$ known) and the \omterm{prop:b1:quantum-introduction:photoelectric}{work function};
threshold wavelength; which metal could it be (caesium \qty{2.1}{eV},
potassium \qty{2.3}{eV}, zinc \qty{4.3}{eV})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:6}
Compton: an X-ray photon of \qty{10}{pm} scatters at \qty{90}{\degree} off an
electron. New wavelength; energies of the photon before and after; energy
given to the electron. Relative shift for \qty{500}{nm} light --- why
Compton needed X-rays.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:7}
\emph{Jönsson's experiment.} Electrons of \qty{50}{keV} (use $p =
\sqrt{2m_eE}$), two slits \qty{2.0}{\micro m} apart, screen at
\qty{35}{cm}. Wavelength; \omterm{ex:b1:wave-propagation:young}{fringe spacing}; why a magnifying electron \omterm{def:b1:mirrors-thin-lenses:lens}{lens}
was needed. In Tonomura's version one electron at a time crosses the
apparatus: what does the screen show after 10, after $10^5$ electrons,
and what does that teach?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:8}
Minimum \omterm{thm:b1:work-and-energy:ke}{kinetic energy}, from the uncertainty principle, of: an
electron confined to \qty{0.1}{nm} (an atom); an electron confined to
\qty{5}{fm} (a nucleus --- use $E \approx pc$ if the result is
relativistic, and conclude whether nuclei can contain electrons); a
dust grain of \qty{1}{\micro g} localized to \qty{1}{\micro m} (give $\Delta v$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:9}
\emph{Bohr's model.} Electron on a \omterm{prop:b1:central-forces:effective}{circular orbit} of radius $r$ round a
proton: (a) speed and energy as functions of $r$ (Coulomb dynamics,
\cref{ch:b1:central-forces}); (b) impose $m_evr = n\hbar$ and deduce
$r_n = n^2a_0$, $E_n = E_1/n^2$ with $a_0$ and $E_1$ as in
\cref{ex:b1:quantum-introduction:hsize}; (c) $v_1$ and $v_1/c$; (d)
wavelengths of Lyman $\alpha$ ($2 \to 1$) and of H$\alpha$ ($3 \to 2$);
(e) show that the orbit $n$ holds exactly $n$ \omterm{thm:b1:quantum-introduction:debroglie}{de Broglie wavelengths}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:10}
\emph{Hydrogen-like systems.} (a) For a nucleus of charge $Ze$ show
$E_n = -Z^2 \times \qty{13.6}{eV}/n^2$ and $r_n = n^2a_0/Z$: ionization
energy of He$^+$ and of the last electron of uranium ($Z = 92$; comment).
(b) Muonic hydrogen: the muon has $207$ times the electron's mass:
radius and ground energy; wavelength of its Lyman $\alpha$ line; why it
is used to measure the size of the proton (radius \qty{0.84}{fm}).
(c) Positronium (electron and positron, equal masses): show that the
\omterm{prop:b1:systems-of-points:twobody}{reduced mass} halves the energies: its Lyman $\alpha$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:11}
\emph{\omterm{ex:b1:quantum-introduction:dot}{Quantum dots}.} In a spherical nanocrystal of radius $R$ the
confinement energy of a particle of effective mass $m^\ast$ is
$h^2/8m^\ast R^2$ (admitted). CdSe: gap $E_g = \qty{1.74}{eV}$, $m_e^\ast =
0.13\,m_e$, $m_h^\ast = 0.45\,m_e$ (the ``hole'' left by the excited
electron also confines). (a) Emitted photon energy $E_g + E_e + E_h$
and wavelength for $R = \qty{2.0}{nm}$, \qty{3.0}{nm}, \qty{4.0}{nm}; colours.
(b) Radius for emission at \qty{520}{nm} (green). (c) Why does bulk CdSe
emit at \qty{713}{nm} only? (d) A \qty{10}{\percent} spread in radius in a
batch: spread in wavelength at \qty{3}{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:12}
\emph{Zero-point energy.} (a) For a \omterm{def:b1:oscillators-resonance:harmonic}{harmonic oscillator} $E = p^2/2m +
\tfrac12m\omega^2x^2$, take $\Delta x\,\Delta p = \hbar/2$ and minimize $E \approx
\Delta p^2/2m + \tfrac12m\omega^2\Delta x^2$ over $\Delta x$: show $E_{\min} = \hbar\omega/2$
(the exact ground energy). (b) The H$_2$ molecule vibrates at $\omega =
\qty{8.3e14}{rad/s}$: zero-point energy in eV, and the lowest
temperature at which its vibration is ``thermal'' ($k_BT \sim \hbar\omega$).
(c) Liquid helium never freezes at atmospheric pressure: estimate the
zero-point energy of a helium atom localized to $\Delta x = \qty{0.05}{nm}$
(a fraction of the interatomic distance) in a crystal, and compare with
its binding energy in the solid, about \qty{1}{meV}. (d) Compare the box formula $E_1 = h^2/8mL^2$ with the
uncertainty bound for $\Delta x = L/2$: which is larger, and why must it be?
\end{exercise}

\section{Problem: The hydrogen spectrum and a quantum dot}

\begin{problem}[{Weekend problem --- measuring $h$ with light and a
voltmeter, building the hydrogen atom from three constants, and
choosing the colour of a crystal with its size}]
\label{pb:b1:quantum-introduction:1}
Constants: $h = \qty{6.626e-34}{J.s}$, $\hbar = \qty{1.055e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$, $e = \qty{1.602e-19}{C}$, $m_e = \qty{9.11e-31}{kg}$,
$1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$, $hc = \qty{1240}{eV.nm}$.

\medskip
\noindent\textbf{Part I --- Measuring \omterm{thm:b1:quantum-introduction:photon}{Planck's constant}.}
A potassium photocathode is lit with monochromatic light; the \omterm{prop:b1:quantum-introduction:photoelectric}{stopping
potential} is measured: \qty{0.30}{V} at \qty{6.0e14}{Hz}, \qty{0.69}{V} at
\qty{7.0e14}{Hz}, \qty{1.13}{V} at \qty{8.0e14}{Hz}, \qty{1.50}{V} at
\qty{9.0e14}{Hz}, \qty{2.36}{V} at \qty{1.1e15}{Hz}.
\begin{enumerate}
  \item Describe the cell and explain why a reverse \omterm{def:b1:dc-circuits:voltage}{voltage} can stop
        the current, and what $eV_s$ measures.
  \item Write Einstein's relation and say what the slope and the
        intercept of $V_s(\nu)$ give.
  \item From the data (a linear fit, or the two extreme points),
        deduce $h$; compare with the accepted value.
  \item \omterm{prop:b1:quantum-introduction:photoelectric}{Work function} of potassium in eV; threshold frequency and
        wavelength; could a red laser pointer extract electrons?
  \item The lamp's intensity is doubled: what happens to $V_s$, to the
        current? Why is this incompatible with a purely wave picture?
  \item Classical estimate: a light of \qty{1}{\micro W/m^2} falls on the
        metal; an atom offers an area of about \qty{1e-20}{m^2}: how long
        would it take to accumulate \qty{2.2}{eV} on one atom? What is
        observed instead?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Bohr's \omterm{prop:b1:quantum-introduction:hydrogen}{hydrogen atom}.}
\begin{enumerate}[resume]
  \item An electron on a \omterm{prop:b1:central-forces:effective}{circular orbit} of radius $r$ round a fixed
        proton: speed $v(r)$ and \omterm{thm:b1:work-and-energy:em}{mechanical energy} $E(r)$.
  \item Bohr's condition $m_evr = n\hbar$: show $r_n = n^2a_0$ and give
        $a_0 = 4\pi\varepsilon_0\hbar^2/m_ee^2$ numerically.
  \item $E_n = E_1/n^2$ with $E_1 = -m_ee^4/[2(4\pi\varepsilon_0)^2\hbar^2]$;
        numerical value in eV.
  \item Speed on the first orbit and the ratio $v_1/c$ (the
        fine-structure constant $\alpha \approx 1/137$).
  \item Wavelengths of the lines $2 \to 1$ (Lyman $\alpha$), $3 \to 2$ and
        $4 \to 2$ (Balmer H$\alpha$, H$\beta$): which are visible?
  \item Show that $1/\lambda = R_H(1/n^2 - 1/p^2)$ and compute the Rydberg
        constant $R_H$; the shortest Balmer wavelength.
  \item Ionization energy of hydrogen from the \omterm{prop:b1:quantum-introduction:box}{ground state}; wavelength
        of the photon that just ionizes it.
  \item De Broglie: show that the circumference of orbit $n$ is $n$
        wavelengths --- Bohr's condition is a standing-wave condition.
  \item Two things the model gets wrong, one thing it gets exactly
        right, and what replaces the orbit in the full theory.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The \omterm{ex:b1:quantum-introduction:dot}{quantum dot}.}
A CdSe nanocrystal of radius $R$ confines an excited electron
(effective mass $m_e^\ast = 0.13\,m_e$) and the hole it leaves
($m_h^\ast = 0.45\,m_e$); the confinement energy of each in a sphere of
radius $R$ is $h^2/8m^\ast R^2$ (admitted); the gap of bulk CdSe is
$E_g = \qty{1.74}{eV}$.
\begin{enumerate}[resume]
  \item Derive the energies $E_n = n^2h^2/8mL^2$ of a particle in a
        one-dimensional box from the standing-wave condition, and
        comment on the analogy with the sphere formula.
  \item Confinement energies of the electron and of the hole for
        $R = \qty{2.0}{nm}$.
  \item Energy and wavelength of the emitted photon for $R = \qty{2.0}{nm}$,
        \qty{3.0}{nm} and \qty{4.0}{nm}; colours.
  \item Explain in one sentence why smaller dots are bluer.
  \item Check the \omterm{def:b1:units-dimensions:oom}{order of magnitude} with the uncertainty principle:
        $\Delta p \sim \hbar/R$ for the electron at $R = \qty{2}{nm}$, and the
        \omterm{thm:b1:work-and-energy:ke}{kinetic energy} it implies.
  \item A dot emits \qty{1.0}{nW} at \qty{577}{nm}: photons per second.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Counting quanta.}
\begin{enumerate}[resume]
  \item Photons per second from a \qty{1.0}{mW} laser at \qty{633}{nm}.
  \item The dark-adapted \omterm{def:b1:optical-instruments:eye}{eye} detects a flash of about $100$ photons at
        \qty{500}{nm} arriving within \qty{0.1}{s}: the corresponding power.
  \item An FM antenna receives \qty{1.0}{pW} at \qty{100}{MHz}: photons per
        second; is there any hope of detecting them one by one?
  \item Sum up: from $h$, $e$, $m_e$ alone, which three quantities of the
        atomic world did this problem produce, and with which
        magnitudes?
\end{enumerate}
\end{problem}
