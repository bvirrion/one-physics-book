\chapter{Mirrors and Thin Lenses}\label{ch:b1:mirrors-thin-lenses}

A make-up mirror shows a face twice its size; the back of a spoon shows
it tiny and upright, the bowl shows it tiny and upside down; a
magnifying glass held at arm's length turns the room on its head, held
close it enlarges a stamp; a phone \omterm{def:b1:mirrors-thin-lenses:lens}{lens} the size of a lentil paints the
whole street onto a sensor. One algebraic relation between three
distances governs every one of these \omterm{def:b1:mirrors-thin-lenses:image}{images}, and this chapter
establishes it --- for curved mirrors first, then for the \omterm{def:b1:mirrors-thin-lenses:lens}{thin lens} that
almost all optical instruments are built from.

\section{Images and the conditions of Gauss}

\begin{definition}[Object, image, stigmatism]\label{def:b1:mirrors-thin-lenses:image}
\index{image (optics)}\index{object (optics)}\index{stigmatism}\index{real image}\index{virtual image}
An optical system receives rays from a point $A$ (the \emph{object
point}). If all the emerging rays pass through one point $A'$, or all
seem to come from one point $A'$, the system is \emph{stigmatic} for
$A$ and $A'$ is the \emph{image} of $A$. The image is \emph{real} if the
emerging rays actually cross at $A'$ (it can be caught on a screen),
\emph{virtual} if only their backward extensions do (it is seen through
the system, not projected). Likewise the object is real if the incoming
rays diverge from $A$, virtual if they converge toward $A$ behind the
system's entrance. The pair $(A, A')$ is \emph{conjugate}.
\end{definition}

\begin{definition}[Centered system; Gauss conditions]\label{def:b1:mirrors-thin-lenses:gauss}
\index{optical axis}\index{Gauss conditions}\index{paraxial ray}
A \emph{centered system} has an axis of rotational symmetry, the
\emph{optical axis}. Rays that stay close to the axis and make small
angles with it are \emph{paraxial}; a system used with paraxial rays
only is said to work under the \emph{Gauss conditions}. Under those
conditions a centered system is approximately stigmatic for every
point, and an object plane perpendicular to the axis has an \omterm{def:b1:mirrors-thin-lenses:image}{image}
plane perpendicular to the axis (\emph{aplanatism}).
\end{definition}
\admitted

\begin{remark}[Why small angles]\label{rem:b1:mirrors-thin-lenses:why}
The \omterm{prop:b1:rays-reflection-refraction:mirror}{plane mirror} is the only simple system that is rigorously
stigmatic for every point. A \omterm{def:b1:mirrors-thin-lenses:mirror}{spherical mirror} or \omterm{def:b1:mirrors-thin-lenses:lens}{lens} is stigmatic
only to the extent that $\sin\theta \approx \tan\theta \approx \theta$:
that approximation is what turns \omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law} into linear relations
between distances. Outside it the \omterm{def:b1:mirrors-thin-lenses:image}{image} of a point is a blur
(\emph{aberrations}), and the designer's job --- stops, combinations of
lenses, aspheric surfaces --- is to push the blur below what the eye or
the sensor can resolve.
\end{remark}

\begin{notation}[Algebraic measures and magnification]\label{not:b1:mirrors-thin-lenses:signs}
\index{algebraic measure}\index{transverse magnification}
The \omterm{def:b1:mirrors-thin-lenses:gauss}{optical axis} is oriented in the direction of the incoming light;
the transverse direction is oriented upward. For points $P$, $Q$ on the
axis, $\overline{PQ}$ denotes the \omterm{rem:b1:mirrors-thin-lenses:why}{algebraic measure} ($\overline{PQ} > 0$
if $Q$ is downstream of $P$); for an object $AB$ perpendicular to the
axis and its \omterm{def:b1:mirrors-thin-lenses:image}{image} $A'B'$, the \emph{\omterm{rem:b1:mirrors-thin-lenses:why}{transverse magnification}} is
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}} ,
\]
negative for an inverted \omterm{def:b1:mirrors-thin-lenses:image}{image}, of absolute value greater than $1$ for an
enlarged one.
\end{notation}

\section{Spherical mirrors}

\begin{definition}[Spherical mirror]\label{def:b1:mirrors-thin-lenses:mirror}
\index{spherical mirror}\index{concave mirror}\index{convex mirror}\index{focus}
A \emph{spherical mirror} is a reflecting spherical cap of center $C$
and vertex $S$ (its intersection with the axis). It is \emph{concave}
if the light meets the hollow side ($C$ on the side of the incoming
light), \emph{convex} otherwise. Its \emph{focus} $F$ is the \omterm{def:b1:mirrors-thin-lenses:image}{image} of a
point at infinity on the axis; its \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} is $f = \overline{SF}$.
\end{definition}

\begin{proposition}[Focus of a spherical mirror]\label{prop:b1:mirrors-thin-lenses:focus}
Under the \omterm{def:b1:mirrors-thin-lenses:gauss}{Gauss conditions}, the \omterm{def:b1:mirrors-thin-lenses:mirror}{focus} is the midpoint of $SC$:
$\overline{SF} = \overline{SC}/2$. A \omterm{def:b1:mirrors-thin-lenses:mirror}{concave mirror} has $f < 0$ (real
\omterm{def:b1:mirrors-thin-lenses:mirror}{focus}, in front); a convex one has $f > 0$ (virtual \omterm{def:b1:mirrors-thin-lenses:mirror}{focus}, behind).
\end{proposition}

\begin{proof}
A ray parallel to the axis at height $h$ meets the mirror at $I$; the
normal at $I$ is the radius $CI$, making an angle $\alpha$ with the
axis, $\sin\alpha = h/R$. Reflected at the same angle $\alpha$ to the
normal, the ray crosses the axis at $F$; triangle $CIF$ has equal
angles $\alpha$ at $C$ and at $I$, so $CF = IF$ and, projecting,
$CF = R/(2\cos\alpha)$. For \omterm{def:b1:mirrors-thin-lenses:gauss}{paraxial rays} $\cos\alpha \to 1$ and
$CF = R/2$: all such rays cross the axis at the same point, the
midpoint of $SC$. Sign: for a \omterm{def:b1:mirrors-thin-lenses:mirror}{concave mirror} $C$ and $F$ lie in front
(upstream), $\overline{SC} < 0$.
\end{proof}

\begin{theorem}[Conjugation for a spherical mirror]\label{thm:b1:mirrors-thin-lenses:mirrorconj}
\index{conjugation relation!spherical mirror}
Under the \omterm{def:b1:mirrors-thin-lenses:gauss}{Gauss conditions}, the \omterm{def:b1:mirrors-thin-lenses:image}{image} $A'$ of a point $A$ on the axis
satisfies
\[
\frac{1}{\overline{SA'}} + \frac{1}{\overline{SA}} = \frac{2}{\overline{SC}}
= \frac{1}{\overline{SF}},
\qquad
\gamma = -\frac{\overline{SA'}}{\overline{SA}} .
\]
\end{theorem}

\begin{proof}
Take a \omterm{def:b1:mirrors-thin-lenses:mirror}{concave mirror} and a real object $A$ beyond $C$; the algebraic
form then covers every case. A ray from $A$ hits the mirror at $I$
(height $h$) and returns to $A'$ on the axis. Call $u$, $u'$, $\omega$
the angles of $AI$, $A'I$ and $CI$ with the axis; paraxially
$u = h/AS$, $u' = h/A'S$, $\omega = h/CS$ (distances). The law of
reflection says the normal $CI$ bisects the angle between $AI$ and
$A'I$: $\omega - u = u' - \omega$, i.e.\ $u + u' = 2\omega$, i.e.\
$1/AS + 1/A'S = 2/CS$; with $\overline{SA}$, $\overline{SA'}$,
$\overline{SC}$ all negative this is the displayed relation. For the
magnification, the ray $B \to S \to B'$ reflected at the vertex makes
equal angles with the axis: $A'B'/SA' = AB/SA$ in lengths, and the \omterm{def:b1:mirrors-thin-lenses:image}{image}
is inverted, $\gamma = -\overline{SA'}/\overline{SA}$.
\end{proof}

\begin{method}[Constructing an image in a mirror]\label{met:b1:mirrors-thin-lenses:mirrorrays}
For an object point $B$ off the axis, draw two of the four remarkable
rays; their intersection (or that of their extensions) is $B'$:
\begin{enumerate}
  \item parallel to the axis $\to$ reflected through $F$;
  \item through $F$ $\to$ reflected parallel to the axis;
  \item through $C$ $\to$ reflected back on itself;
  \item to the vertex $S$ $\to$ reflected symmetrically about the axis.
\end{enumerate}
Then $A'$ is the foot of the perpendicular from $B'$ to the axis.
\end{method}

\begin{example}[Shaving mirror, rear-view mirror]\label{ex:b1:mirrors-thin-lenses:mirrors}
A \omterm{def:b1:mirrors-thin-lenses:mirror}{concave mirror} of radius \qty{60}{cm} ($f = \qty{-30}{cm}$) with a
face at \qty{20}{cm}: $1/\overline{SA'} = -1/30 + 1/20 = 1/60$,
$\overline{SA'} = \qty{+60}{cm}$ --- behind the mirror, virtual ---
and $\gamma = -60/(-20) = +3$: upright, three times larger, the shaving
mirror. A \omterm{def:b1:mirrors-thin-lenses:mirror}{convex mirror} of radius \qty{2.0}{m} ($f = \qty{+1.0}{m}$)
with a car \qty{20}{m} behind: $1/\overline{SA'} = 1 + 1/20$,
$\overline{SA'} = \qty{0.95}{m}$ (virtual, behind), $\gamma = +0.048$:
upright, twenty times smaller, and therefore a wide field --- the
rear-view mirror, with its warning that objects are closer than they
appear.
\end{example}

\section{Thin lenses}

\begin{definition}[Thin lens]\label{def:b1:mirrors-thin-lenses:lens}
\index{thin lens}\index{converging lens}\index{diverging lens}\index{optical center}\index{focal length}\index{vergence}
A \emph{lens} is a \omterm{def:b1:rays-reflection-refraction:index}{transparent medium} bounded by two spherical (or
plane) surfaces; it is \emph{thin} when its thickness is negligible
against the radii and against the distances in play, so that the two
vertices merge into one point, the \emph{optical center} $O$. A ray
through $O$ is not deviated. The \emph{\omterm{def:b1:mirrors-thin-lenses:image}{image} \omterm{def:b1:mirrors-thin-lenses:mirror}{focus}} $F'$ is the \omterm{def:b1:mirrors-thin-lenses:image}{image}
of a point at infinity on the axis; the \emph{object \omterm{def:b1:mirrors-thin-lenses:mirror}{focus}} $F$ is the
point whose \omterm{def:b1:mirrors-thin-lenses:image}{image} is at infinity; both are at the same distance from
$O$: $\overline{OF'} = -\overline{OF} = f'$, the \emph{focal length}.
The lens is \emph{converging} if $f' > 0$ ($F'$ real, downstream; edges
thinner than the center), \emph{diverging} if $f' < 0$. Its
\emph{vergence} $V = 1/f'$ is measured in \emph{diopters}
($\qty{1}{\delta} = \qty{1}{m^{-1}}$).
\end{definition}

\begin{theorem}[Conjugation for a thin lens]\label{thm:b1:mirrors-thin-lenses:lensconj}
\index{conjugation relation!thin lens}\index{Descartes relation}\index{Newton relation}
Under the \omterm{def:b1:mirrors-thin-lenses:gauss}{Gauss conditions}, the \omterm{def:b1:mirrors-thin-lenses:image}{image} $A'$ of a point $A$ on the axis
satisfies the relation of Descartes
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'},
\qquad
\gamma = \frac{\overline{OA'}}{\overline{OA}},
\]
and, equivalently, the relation of Newton
\[
\overline{FA}\cdot\overline{F'A'} = -f'^2,
\qquad
\gamma = \frac{f'}{\overline{FA}} = -\frac{\overline{F'A'}}{f'} .
\]
\end{theorem}

\begin{proof}[Partial proof]
The key fact is that a \omterm{def:b1:mirrors-thin-lenses:lens}{thin lens} deviates a ray crossing it at height
$h$ by the angle $h/f'$ toward the axis, whatever the ray's direction
--- each zone of the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} acts as a thin prism (\cref{rem:b1:rays-reflection-refraction:thin})
whose apex angle grows linearly with $h$ for spherical surfaces; this
linearity, exact for \omterm{def:b1:mirrors-thin-lenses:gauss}{paraxial rays}, is admitted here and derived from
\omterm{thm:b1:rays-reflection-refraction:snell}{Snell's law} at two spherical surfaces in the Year 2 volume. Granting
it: a ray from $A$ (at distance $p = -\overline{OA} > 0$ for a real
object) reaching the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} at height $h$ arrives with slope $h/p$, is
bent by $h/f'$ and leaves with slope $h/p - h/f'$; it crosses the axis
at $A'$ with $h/\overline{OA'} = h/f' - h/p$, i.e.\ $1/\overline{OA'}
- 1/\overline{OA} = 1/f'$, independently of $h$: all rays from $A$ meet
at $A'$. The ray $B \to O \to B'$ through the center is straight, so
$\overline{A'B'}/\overline{AB} = \overline{OA'}/\overline{OA}$. Newton's
form follows by substituting $\overline{OA} = \overline{OF} + \overline{FA}
= -f' + \overline{FA}$ and $\overline{OA'} = f' + \overline{F'A'}$.
\end{proof}

\begin{method}[Constructing an image through a thin lens]\label{met:b1:mirrors-thin-lenses:lensrays}
Three remarkable rays from $B$:
\begin{enumerate}
  \item through $O$: undeviated;
  \item parallel to the axis: emerges through $F'$;
  \item through $F$: emerges parallel to the axis.
\end{enumerate}
Two suffice; for a \omterm{def:b1:mirrors-thin-lenses:lens}{diverging lens}, use the extensions (the ray aimed at
$F$ behind the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} emerges parallel; the parallel ray emerges as if
from $F'$ in front).
\end{method}

\end{example}

\begin{proposition}[Longitudinal magnification]\label{prop:b1:mirrors-thin-lenses:longitudinal}
\index{longitudinal magnification}
If the object moves along the axis by $\dd\overline{OA}$, the \omterm{def:b1:mirrors-thin-lenses:image}{image}
moves by $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$, in the same
direction.
\end{proposition}

\begin{proof}
Differentiate Descartes's relation at fixed $f'$:
$-\dd\overline{OA'}/\overline{OA'}^2 + \dd\overline{OA}/\overline{OA}^2 = 0$,
so $\dd\overline{OA'} = (\overline{OA'}/\overline{OA})^2\,\dd\overline{OA}$.
\end{proof}

\begin{remark}[Depth of focus]\label{rem:b1:mirrors-thin-lenses:depth}
A projector with $\gamma = -50$ has $\gamma^2 = 2500$: a slide that
buckles by \qty{0.1}{mm} under the lamp's heat throws its \omterm{def:b1:mirrors-thin-lenses:image}{image}
\qty{25}{cm} out of the screen plane. Conversely a camera ($\abs\gamma
\ll 1$) tolerates large object displacements for a tiny \omterm{def:b1:mirrors-thin-lenses:image}{image} shift ---
the depth of field.
\end{remark}

\section{Measuring a focal length}

\begin{method}[Autocollimation]\label{met:b1:mirrors-thin-lenses:autocollimation}
\index{autocollimation}
Place a \omterm{prop:b1:rays-reflection-refraction:mirror}{plane mirror} right behind the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} and move a lit object (a
cross on a frosted glass) along the axis until its \omterm{def:b1:mirrors-thin-lenses:image}{image}, reflected
back through the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, forms sharp on the object itself, inverted and
the same size. The object is then in the focal plane: rays leave the
\omterm{def:b1:mirrors-thin-lenses:lens}{lens} parallel, return parallel from the mirror, and are refocused in
the focal plane. The object--lens distance is $f'$.
\end{method}

\begin{proposition}[Bessel's method]\label{prop:b1:mirrors-thin-lenses:bessel}
\index{Bessel's method}\index{Silbermann's method}
An object and a screen are fixed a distance $D$ apart. A \omterm{def:b1:mirrors-thin-lenses:lens}{converging
lens} forms a sharp \omterm{def:b1:mirrors-thin-lenses:image}{real image} on the screen for exactly two positions
if $D > 4f'$, for one if $D = 4f'$, for none if $D < 4f'$. The two
positions are a distance $d$ apart with
\[
f' = \frac{D^2 - d^2}{4D} .
\]
At $D = 4f'$ (Silbermann) the single position gives $\gamma = -1$.
\end{proposition}

\begin{proof}
Let $p = -\overline{OA} > 0$ be the object--lens distance; then
$\overline{OA'} = D - p$ and Descartes gives $1/(D - p) + 1/p = 1/f'$,
i.e.\ $p^2 - Dp + Df' = 0$: discriminant $D^2 - 4Df'$, positive iff
$D > 4f'$. The two roots $p_{1,2} = (D \mp \sqrt{D^2 - 4Df'})/2$ add up
to $D$ (the positions are symmetric: object and \omterm{def:b1:mirrors-thin-lenses:image}{image} distances are
exchanged) and differ by $d = \sqrt{D^2 - 4Df'}$; solve for $f'$. At
$D = 4f'$, $p = 2f' = \overline{OA'}$ and $\gamma = -1$.
\end{proof}

\section{Two thin lenses}

\begin{proposition}[Lenses in contact; afocal doublet]\label{prop:b1:mirrors-thin-lenses:doublet}
\index{afocal system}\index{angular magnification}
Two thin lenses in contact behave as one \omterm{def:b1:mirrors-thin-lenses:lens}{thin lens} of \omterm{def:b1:mirrors-thin-lenses:lens}{vergence}
$V = V_1 + V_2$. Two lenses placed so that $F_1' = F_2$ form an
\emph{afocal} system: a parallel beam emerges parallel, and a beam
inclined by $\alpha$ emerges inclined by $\alpha' = G\alpha$ with the
\emph{angular magnification} $G = -f_1'/f_2'$.
\end{proposition}

\begin{proof}
The \omterm{def:b1:mirrors-thin-lenses:image}{image} $A_1$ given by $L_1$ is the object for $L_2$. In contact,
$O_1 = O_2 = O$: $1/\overline{OA_1} - 1/\overline{OA} = V_1$ and
$1/\overline{OA'} - 1/\overline{OA_1} = V_2$; add. Afocal: a parallel
beam at angle $\alpha$ focuses in the common focal plane at height
$h = f_1'\alpha$ from the axis (ray through $O_1$); seen from $L_2$ this
point is in its object focal plane, so the beam emerges parallel, at the
angle of the ray from the point through $O_2$: $\alpha' = -h/f_2'$.
\end{proof}

\begin{example}[A telescope in one line]\label{ex:b1:mirrors-thin-lenses:afocal}
$f_1' = \qty{1000}{mm}$, $f_2' = \qty{25}{mm}$: $G = -40$; the Moon,
half a degree wide, becomes a $20^\circ$ disk --- the astronomical
telescope of the next chapter. With a diverging eyepiece,
$f_2' = \qty{-25}{mm}$, $G = +40$: upright, Galileo's spyglass.
\end{example}

\section{Problem: The slide projector}

\begin{problem}[{Weekend problem --- one lens, measured on a bench,
then asked to throw a 35-mm slide across a hall: how a projector is
designed, focused, and fitted with a long-throw converter}]
\label{pb:b1:mirrors-thin-lenses:1}
A projection \omterm{def:b1:mirrors-thin-lenses:lens}{lens} $L_1$ is a converging \omterm{def:b1:mirrors-thin-lenses:lens}{thin lens} of unknown \omterm{def:b1:mirrors-thin-lenses:lens}{focal
length} $f_1'$. The slide is \qty{24}{mm} high and \qty{36}{mm} wide.

\medskip
\noindent\textbf{Part I --- The \omterm{def:b1:mirrors-thin-lenses:lens}{lens} on the bench.}
\begin{enumerate}
  \item State Descartes's relation and the magnification for a \omterm{def:b1:mirrors-thin-lenses:lens}{thin
        lens}, with the sign conventions.
  \item Object and screen are fixed a distance $D$ apart. Show that a
        \omterm{def:b1:mirrors-thin-lenses:image}{real image} on the screen requires $D \geq 4f_1'$.
  \item Show that, when $D > 4f_1'$, the two \omterm{def:b1:mirrors-thin-lenses:lens}{lens} positions giving a
        sharp \omterm{def:b1:mirrors-thin-lenses:image}{image} are symmetric with respect to the midpoint of
        object and screen, and that $f_1' = (D^2 - d^2)/4D$ with $d$
        their separation.
  \item With $D = \qty{80.0}{cm}$ and $d = \qty{40.0}{cm}$, compute
        $f_1'$.
  \item The \omterm{def:b1:mirrors-thin-lenses:lens}{lens} sits in a thick barrel whose \omterm{def:b1:mirrors-thin-lenses:lens}{optical center} cannot be
        located precisely. Why does this not affect the method?
  \item The distances carry $u(D) = \qty{0.2}{cm}$ and
        $u(d) = \qty{0.5}{cm}$. Compute $u(f_1')$ and write the result.
  \item Give the magnification at each of the two positions. Which one
        corresponds to a projector?
  \item What is the smallest $D$ for which a sharp \omterm{def:b1:mirrors-thin-lenses:image}{image} exists, and
        the magnification then (Silbermann)?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Throwing the picture.}
The picture on the screen must be \qty{1.80}{m} wide. Take
$f_1' = \qty{15.0}{cm}$.
\begin{enumerate}[resume]
  \item What magnification is required (with its sign)?
  \item Compute the lens--screen distance and the slide--lens distance.
  \item Why is the slide inserted upside down and mirror-reversed?
  \item The screen is moved to \qty{10.0}{m}. By how much, and in which
        direction, must the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} be moved relative to the slide to
        refocus? How wide is the picture now?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{def:b1:mirrors-thin-lenses:mirror}{Focus} tolerance and light.}
\begin{enumerate}[resume]
  \item Show that a small displacement $\dd\overline{OA}$ of the slide
        shifts the \omterm{def:b1:mirrors-thin-lenses:image}{image} by $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$.
  \item The film buckles by \qty{0.10}{mm} under the lamp's heat. By how
        much does the \omterm{def:b1:mirrors-thin-lenses:image}{image} plane move (screen at \qty{7.65}{m})?
        Comment.
  \item The \omterm{def:b1:mirrors-thin-lenses:lens}{lens} aperture is \qty{30}{mm} in diameter. If the screen
        sits \qty{20}{cm} off the \omterm{def:b1:mirrors-thin-lenses:image}{image} plane, what is the diameter of
        the blur spot of a point of the slide? Seen from \qty{5}{m},
        does the eye (angular resolution about $\qty{3e-4}{rad}$) notice?
  \item The slide receives \qty{2.0}{W} of light, all of which reaches
        the screen. Compute the irradiance (power per unit area) on the
        slide and on the screen, and show that their ratio is $\gamma^2$.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The long-throw converter.}
In a larger hall the screen is \qty{12.0}{m} away. A \omterm{def:b1:mirrors-thin-lenses:lens}{diverging lens}
$L_2$, $f_2' = \qty{-30.0}{cm}$, is clipped on \qty{8.0}{cm} behind
$L_1$; the screen is \qty{12.0}{m} from $L_2$.
\begin{enumerate}[resume]
  \item For the final \omterm{def:b1:mirrors-thin-lenses:image}{image} to be real and on the screen, where (from
        $L_2$) must the intermediate \omterm{def:b1:mirrors-thin-lenses:image}{image} given by $L_1$ lie? Is it a
        real or a virtual object for $L_2$?
  \item Deduce the slide position $\overline{O_1A}$.
  \item Compute $\gamma_1$, $\gamma_2$ and the total magnification.
  \item Give the picture width, and compare with what $L_1$ alone would
        give on the same screen (\qty{12.08}{m} from $L_1$).
  \item For an object at infinity, show that the system is equivalent
        to a single \omterm{def:b1:mirrors-thin-lenses:lens}{lens} of \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} $f_{\mathrm{eq}}$ with
        \[
        \frac{1}{f_{\mathrm{eq}}} = \frac{1}{f_1'} + \frac{1}{f_2'} - \frac{e}{f_1' f_2'},
        \]
        $e = \overline{O_1O_2}$, in the sense that a beam inclined by
        $\alpha$ focuses at height $f_{\mathrm{eq}}\,\alpha$ from the
        axis. Compute $f_{\mathrm{eq}}$.
  \item Check that the ratio of picture widths (with/without converter)
        is close to $f_1'/f_{\mathrm{eq}}$. Why only approximately?
  \item Why must the intermediate \omterm{def:b1:mirrors-thin-lenses:image}{image} lie less than $\abs{f_2'}$
        beyond $L_2$ for the final \omterm{def:b1:mirrors-thin-lenses:image}{image} to be real?
  \item With the same lamp, by what factor does the screen irradiance
        change when the converter is fitted (same screen distance)?
  \item Summarize: the converter multiplies the \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} by which
        factor, and what does it do to picture size and brightness at a
        given throw?
\end{enumerate}
\end{problem}
