\chapter{Kinematics of a Point}\label{ch:b1:point-kinematics}

The car on a roller coaster enters the loop at \qty{90}{km/h}, slows
as it climbs, hangs for an instant at the top, and roars out again;
its riders feel crushed into their seats at the bottom and almost
weightless at the top. Describing that ride --- where the car is, how
fast it moves, how its \omterm{def:b1:point-kinematics:velocity}{velocity} turns and changes --- is kinematics,
and it needs more than the $x$, $y$, $z$ of a straight road: \omterm{def:b1:point-kinematics:cylindrical}{polar
coordinates} for anything that turns, and a frame that travels with the
car itself. This chapter sets up those tools, proves the formulas for
\omterm{def:b1:point-kinematics:velocity}{velocity} and \omterm{def:b1:point-kinematics:velocity}{acceleration} in each of them, and applies them to the
motions that every later chapter will call upon: circular, helical,
harmonic, and the motion along any curve.

\section{Position, velocity, acceleration}

\begin{definition}[Reference frame; position; trajectory]\label{def:b1:point-kinematics:frame}
\index{reference frame}\index{position vector}\index{trajectory}\index{point particle}
A \emph{reference frame} is a rigid body taken as fixed (the
laboratory, the Earth, a train), together with a clock. A \emph{point
particle} $M$ --- a body whose size is irrelevant to the motion studied
--- is located at each instant by its \emph{position vector}
$\vect{OM}(t)$ from an origin $O$ attached to the frame; the curve
described by $M$ is its \emph{trajectory}. Motion is always motion
\emph{relative to a frame}: the same passenger is at rest in the train
and moving in the station.
\end{definition}

\begin{definition}[Velocity and acceleration]\label{def:b1:point-kinematics:velocity}
\index{velocity}\index{acceleration}
The \emph{velocity} and \emph{acceleration} of $M$ in the frame are
\[
\vect v = \frac{\dd\vect{OM}}{\dd t}, \qquad
\vect a = \frac{\dd\vect v}{\dd t} = \frac{\dd^2\vect{OM}}{\dd t^2},
\]
derivatives of vector functions: the derivative of a vector is
obtained by differentiating its components in a basis that is
\emph{fixed in the frame}. The velocity is tangent to the \omterm{def:b1:point-kinematics:frame}{trajectory}
and points along the motion; its norm $v$ is the \emph{speed}. Units:
\unit{m/s}, \unit{m/s^2}.
\end{definition}

\begin{proposition}[Cartesian coordinates]\label{prop:b1:point-kinematics:cartesian}
\index{Cartesian coordinates}
With a fixed orthonormal basis $(\vect e_x, \vect e_y, \vect e_z)$ and
$\vect{OM} = x\,\vect e_x + y\,\vect e_y + z\,\vect e_z$,
\[
\vect v = \dot x\,\vect e_x + \dot y\,\vect e_y + \dot z\,\vect e_z, \qquad
\vect a = \ddot x\,\vect e_x + \ddot y\,\vect e_y + \ddot z\,\vect e_z,
\]
where the dot denotes $\dd/\dd t$.
\end{proposition}

\begin{proof}
The basis vectors are constant; only the components vary.
\end{proof}

\begin{example}[Projectile]\label{ex:b1:point-kinematics:projectile}
Launched from $O$ at speed $v_0$ and angle $\alpha$ above the
horizontal, a ball's coordinates (dynamics, next chapter) are
$x = v_0\cos\alpha\,t$, $z = v_0\sin\alpha\,t - \tfrac12 gt^2$. Then
$\vect v = v_0\cos\alpha\,\vect e_x + (v_0\sin\alpha - gt)\vect e_z$,
$\vect a = -g\vect e_z$: the \omterm{def:b1:point-kinematics:velocity}{acceleration} is constant while the
\omterm{def:b1:point-kinematics:velocity}{velocity} turns; at the apex ($\dot z = 0$, $t = v_0\sin\alpha/g$) the
\omterm{def:b1:point-kinematics:velocity}{velocity} is horizontal and the \omterm{def:b1:point-kinematics:velocity}{acceleration} perpendicular to it.
Eliminating $t$: $z = x\tan\alpha - gx^2/(2v_0^2\cos^2\alpha)$, a
parabola.
\end{example}

\section{Cylindrical coordinates}

\begin{definition}[Cylindrical coordinates and basis]\label{def:b1:point-kinematics:cylindrical}
\index{cylindrical coordinates}\index{polar coordinates}\index{local basis}
Point $M$ is located by $r \geq 0$ (distance to the $z$ axis), the
angle $\theta$ of its projection on the $xy$ plane with $\vect e_x$,
and $z$: $\vect{OM} = r\,\vect e_r + z\,\vect e_z$, where
\[
\vect e_r = \cos\theta\,\vect e_x + \sin\theta\,\vect e_y, \qquad
\vect e_\theta = -\sin\theta\,\vect e_x + \cos\theta\,\vect e_y,
\]
and $\vect e_z$ form the \emph{local basis} $(\vect e_r, \vect e_\theta,
\vect e_z)$, orthonormal and direct, which \emph{moves with $M$}. In the
plane ($z$ fixed) these are the \emph{polar coordinates} $(r, \theta)$.
\end{definition}

\begin{lemma}[Derivatives of the local basis]\label{lem:b1:point-kinematics:basisderiv}
\[
\frac{\dd\vect e_r}{\dd t} = \dot\theta\,\vect e_\theta, \qquad
\frac{\dd\vect e_\theta}{\dd t} = -\dot\theta\,\vect e_r, \qquad
\frac{\dd\vect e_z}{\dd t} = \vect 0 .
\]
\end{lemma}

\begin{proof}
Differentiate the components: $\dd\vect e_r/\dd t = \dot\theta(-\sin\theta\,
\vect e_x + \cos\theta\,\vect e_y) = \dot\theta\vect e_\theta$, and
similarly for $\vect e_\theta$. A rotating unit vector's derivative is
perpendicular to it, of norm the angular rate.
\end{proof}

\begin{theorem}[Velocity and acceleration in cylindrical coordinates]\label{thm:b1:point-kinematics:cylindrical}
\[
\vect v = \dot r\,\vect e_r + r\dot\theta\,\vect e_\theta + \dot z\,\vect e_z,
\qquad
\vect a = \big(\ddot r - r\dot\theta^2\big)\vect e_r
+ \big(r\ddot\theta + 2\dot r\dot\theta\big)\vect e_\theta + \ddot z\,\vect e_z .
\]
\end{theorem}

\begin{proof}
Differentiate $\vect{OM} = r\vect e_r + z\vect e_z$ with the lemma:
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta + \dot z\vect e_z$.
Differentiate again: $\ddot r\vect e_r + \dot r\dot\theta\vect e_\theta +
\dot r\dot\theta\vect e_\theta + r\ddot\theta\vect e_\theta - r\dot\theta^2
\vect e_r + \ddot z\vect e_z$; collect.
\end{proof}

\begin{corollary}[Circular motion]\label{cor:b1:point-kinematics:circular}
\index{circular motion}\index{centripetal acceleration}\index{angular velocity}
On a circle of radius $R$ ($r = R$, $z = 0$), with \emph{\omterm{cor:b1:point-kinematics:circular}{angular
velocity}} $\omega = \dot\theta$:
\[
\vect v = R\omega\,\vect e_\theta, \qquad
\vect a = -R\omega^2\,\vect e_r + R\dot\omega\,\vect e_\theta
= -\frac{v^2}{R}\,\vect e_r + \frac{\dd v}{\dd t}\,\vect e_\theta .
\]
For \emph{uniform} \omterm{cor:b1:point-kinematics:circular}{circular motion} ($\omega$ constant) the \omterm{def:b1:point-kinematics:velocity}{acceleration}
is purely \emph{centripetal}, of norm $v^2/R = R\omega^2$, although the
speed is constant.
\end{corollary}

\begin{proof}
Set $\dot r = \ddot r = 0$ in the theorem; $v = R\omega$.
\end{proof}

\begin{example}[Two circles]\label{ex:b1:point-kinematics:circles}
A car rounds a roundabout of radius \qty{20}{m} at \qty{36}{km/h}
(\qty{10}{m/s}): $a = v^2/R = \qty{5.0}{m/s^2}$, half a $g$, directed
toward the center --- what the tires must supply. The Moon
($R = \qty{3.84e8}{m}$, $T = \qty{27.3}{d}$): $\omega = 2\pi/T =
\qty{2.66e-6}{rad/s}$, $a = R\omega^2 = \qty{2.7e-3}{m/s^2}$, one
$3600$th of $g$ at the Earth's surface, at sixty Earth radii --- the
comparison Newton made (\cref{ch:b1:central-forces}).
\end{example}

\section{Spherical coordinates}

\begin{definition}[Spherical coordinates]\label{def:b1:point-kinematics:spherical}
\index{spherical coordinates}
Point $M$ is located by its distance $r = OM$, the \emph{colatitude}
$\theta \in [0, \pi]$ between $\vect{OM}$ and $\vect e_z$, and the
\emph{azimuth} $\varphi$ of its projection on the $xy$ plane:
$\vect{OM} = r\,\vect e_r$ with
\[
\vect e_r = \sin\theta\cos\varphi\,\vect e_x + \sin\theta\sin\varphi\,\vect e_y + \cos\theta\,\vect e_z ;
\]
$\vect e_\theta$ (tangent to the meridian, toward increasing $\theta$)
and $\vect e_\varphi$ (tangent to the parallel) complete the local
orthonormal basis. On the Earth, $\theta$ is $90^\circ$ minus the
latitude and $\varphi$ the longitude.
\end{definition}

\begin{proposition}[Velocity in spherical coordinates]\label{prop:b1:point-kinematics:sphericalv}
\[
\vect v = \dot r\,\vect e_r + r\dot\theta\,\vect e_\theta + r\sin\theta\,\dot\varphi\,\vect e_\varphi .
\]
\end{proposition}

\begin{proof}
$\dd\vect e_r/\dd t = \dot\theta\,\vect e_\theta + \sin\theta\,\dot\varphi\,
\vect e_\varphi$: the first term is the rotation in the meridian plane
(as in the polar case), the second the rotation about $\vect e_z$ at
rate $\dot\varphi$ of a vector whose distance to the axis is
$\sin\theta$. Then $\vect v = \dd(r\vect e_r)/\dd t$. The \omterm{def:b1:point-kinematics:velocity}{acceleration} is
not needed this year.
\end{proof}

\section{The Frenet frame}

\begin{definition}[Curvilinear abscissa; Frenet basis]\label{def:b1:point-kinematics:frenet}
\index{curvilinear abscissa}\index{Frenet basis}\index{radius of curvature}
Along a \omterm{def:b1:point-kinematics:frame}{trajectory}, the \emph{curvilinear abscissa} $s(t)$ is the arc
length from a reference point, counted algebraically; $v = \dot s$. At
$M$, the unit tangent $\vect T = \dd\vect{OM}/\dd s$ points along
increasing $s$; the \emph{radius of curvature} $\rho > 0$ and the unit
normal $\vect N$, pointing toward the concave side (the center of
curvature), are defined by
\[
\frac{\dd\vect T}{\dd s} = \frac{1}{\rho}\,\vect N .
\]
$(\vect T, \vect N)$ is the \emph{Frenet basis}; $\rho = \infty$ on a
straight line, $\rho = R$ on a circle of radius $R$.
\end{definition}

\begin{theorem}[Acceleration in the Frenet basis]\label{thm:b1:point-kinematics:frenet}
\index{tangential acceleration}\index{normal acceleration}
\[
\vect v = v\,\vect T, \qquad
\vect a = \frac{\dd v}{\dd t}\,\vect T + \frac{v^2}{\rho}\,\vect N .
\]
The \emph{tangential} component $\dd v/\dd t$ measures how fast the
speed changes; the \emph{normal} component $v^2/\rho$, always toward
the center of curvature, how fast the direction turns. The motion is
uniform iff $\vect a \perp \vect v$; it is rectilinear iff
$\vect a \parallel \vect v$.
\end{theorem}

\begin{proof}
$\vect v = (\dd\vect{OM}/\dd s)(\dd s/\dd t) = v\vect T$. Then $\vect a =
\dot v\vect T + v\,\dd\vect T/\dd t = \dot v\vect T + v(\dd\vect T/\dd s)\dot s
= \dot v\vect T + (v^2/\rho)\vect N$. That $\dd\vect T/\dd s \perp \vect T$
follows from $\vect T\cdot\vect T = 1$ differentiated.
\end{proof}

\begin{example}[Braking in a bend]\label{ex:b1:point-kinematics:bend}
A car at \qty{90}{km/h} (\qty{25}{m/s}) in a bend of radius \qty{100}{m}
brakes at \qty{2.0}{m/s^2}: $a_T = \qty{-2.0}{m/s^2}$, $a_N = 625/100 =
\qty{6.25}{m/s^2}$, $a = \sqrt{4 + 39} = \qty{6.6}{m/s^2}$. The normal
part dominates: a bend is first of all a turning, and the tires' grip
(next chapter) limits $\sqrt{a_T^2 + a_N^2}$, which is why one brakes
\emph{before} the bend.
\end{example}

\section{Standard motions}

\begin{proposition}[Uniformly accelerated rectilinear motion]\label{prop:b1:point-kinematics:uarm}
\index{uniformly accelerated motion}
On an axis, $\ddot x = a$ constant, from $x_0$ and $v_0$:
$v = v_0 + at$, $x = x_0 + v_0t + \tfrac12 at^2$, and
$v^2 - v_0^2 = 2a(x - x_0)$.
\end{proposition}

\begin{proof}
Integrate twice; eliminate $t$ between the first two relations.
\end{proof}

\begin{proposition}[Helical motion]\label{prop:b1:point-kinematics:helix}
\index{helical motion}
$r = R$, $\theta = \omega t$, $z = v_zt$ (constants $R$, $\omega$, $v_z$):
the speed $v = \sqrt{R^2\omega^2 + v_z^2}$ is constant, the \omterm{def:b1:point-kinematics:velocity}{acceleration}
$\vect a = -R\omega^2\vect e_r$ is centripetal of constant norm, and the
\omterm{def:b1:point-kinematics:frenet}{radius of curvature} is $\rho = v^2/a = R(1 + v_z^2/R^2\omega^2) > R$.
\end{proposition}

\begin{proof}
\cref{thm:b1:point-kinematics:cylindrical} with $\dot r = 0$,
$\ddot\theta = 0$, $\ddot z = 0$; uniform motion, so $\vect a$ is purely
normal and $v^2/\rho = R\omega^2$.
\end{proof}

\begin{proposition}[Harmonic motion]\label{prop:b1:point-kinematics:harmonic}
\index{harmonic motion}
$x = A\cos(\omega t + \varphi)$: $\dot x = -A\omega\sin(\omega t + \varphi)$,
$\ddot x = -\omega^2 x$. The \omterm{def:b1:point-kinematics:velocity}{velocity} leads the position by a quarter
period and the \omterm{def:b1:point-kinematics:velocity}{acceleration} is opposite to the position; $v_{\max} =
A\omega$, $a_{\max} = A\omega^2$. This is the projection on a diameter of
a uniform \omterm{cor:b1:point-kinematics:circular}{circular motion} of radius $A$ and \omterm{cor:b1:point-kinematics:circular}{angular velocity} $\omega$.
\end{proposition}

\begin{proof}
Differentiate; for the projection, take $x = A\cos\theta$ with $\theta =
\omega t + \varphi$.
\end{proof}

\begin{method}[Choosing coordinates]\label{met:b1:point-kinematics:choose}
\begin{itemize}
  \item Straight-line or parabolic motion, constant \omterm{def:b1:point-kinematics:velocity}{acceleration}:
        Cartesian.
  \item Motion around an axis or a point (circles, spirals, orbits,
        turntables): cylindrical/polar; the basis moves, so use
        \cref{thm:b1:point-kinematics:cylindrical}, never naive
        component differentiation.
  \item A given curve (track, bend, loop) when the speed law is known:
        Frenet, which separates ``how fast'' from ``which way''.
\end{itemize}
Whatever the choice, $\vect v$ and $\vect a$ are frame-dependent
vectors, and a result is complete only with the frame named.
\end{method}

\begin{exercise}[$\star$]\label{exo:b1:point-kinematics:4}

\section{Problem: The roller-coaster loop}

\begin{problem}[{Weekend problem --- a car enters a vertical loop at
ninety kilometers per hour: where it slows, where its acceleration
points, how many $g$ the riders feel, and why modern loops are not
circles}]
\label{pb:b1:point-kinematics:1}
A car, treated as a point $M$, runs on a vertical circular loop of
radius $R = \qty{10}{m}$, center $C$. Its position is given by the
angle $\theta$ between $\vect{CM}$ and the downward vertical ($\theta =
0$ at the bottom, $\pi$ at the top). Friction is neglected, and the
speed law (derived in \cref{ch:b1:work-and-energy}) is
\[
v^2 = v_0^2 - 2gR(1 - \cos\theta),
\]
with $v_0 = \qty{25}{m/s}$ the speed at the bottom; $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Part I --- Kinematics on the circle.}
\begin{enumerate}
  \item In \omterm{def:b1:point-kinematics:cylindrical}{polar coordinates} centered at $C$ (with $\vect e_r$ from $C$
        to $M$), write $\vect v$ and $\vect a$ for motion on the circle.
  \item Identify the Frenet vectors $\vect T$ and $\vect N$ in terms of
        $\vect e_r$ and $\vect e_\theta$, and the \omterm{def:b1:point-kinematics:frenet}{radius of curvature}.
  \item Compute the speed at the top, and at the side points
        $\theta = \pi/2$ and $3\pi/2$.
  \item Differentiate the speed law with respect to time to show that
        the \omterm{thm:b1:point-kinematics:frenet}{tangential acceleration} is $a_T = -g\sin\theta$. Interpret
        its sign on the way up and on the way down.
  \item Express the \omterm{thm:b1:point-kinematics:frenet}{normal acceleration} $a_N(\theta)$ and compute it at
        the bottom, the sides and the top.
  \item Give the norm of $\vect a$ at the four points and the angle it
        makes with the vertical at the top.
  \item At which points is $\vect a$ perpendicular to $\vect v$? Parallel?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Angular rates and timing.}
\begin{enumerate}[resume]
  \item Express $\dot\theta$ as a function of $\theta$ and compute it
        at the bottom and the top.
  \item Show that $\ddot\theta = -(g/R)\sin\theta$ (the equation of a
        pendulum) and comment.
  \item For comparison, compute the period of small oscillations of a
        simple pendulum of length $R$ (\cref{ex:b1:units-dimensions:pendulum}
        gave its form; $k = 2\pi$).
  \item Estimate the time to go round the loop, using the average of
        the speeds at the four reference points.
  \item The minimum speed at the top for the car to stay on the track
        (next chapter) is $\sqrt{gR}$. Is it satisfied? What minimum
        $v_0$ would be needed?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- What the riders feel.}
The apparent weight per unit mass felt by a rider is $\vect a - \vect g$
(admitted here, derived in \cref{ch:b1:newton-dynamics}); its norm in
units of $g$ is the ``$g$-load''.
\begin{enumerate}[resume]
  \item Compute the $g$-load at the bottom.
  \item Compute it at the top, and say which way the rider is pressed
        (seat or restraint).
  \item Compute it at the side points.
  \item Riders tolerate about $4g$ briefly. Does this loop pass? Where
        is the problem?
  \item Show that, for a circular loop entered just fast enough to
        clear the top (zero apparent weight there), the bottom $g$-load
        is necessarily $6g$, whatever $R$.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The clothoid loop.}
Modern loops are ``teardrops'': the \omterm{def:b1:point-kinematics:frenet}{radius of curvature} is large at the
bottom and small at the top.
\begin{enumerate}[resume]
  \item Using the Frenet form of $\vect a$, explain why varying $\rho$
        along the track changes $a_N$ without changing the speed law
        (which depends only on height).
  \item Choose $\rho_{\text{bottom}}$ so that the $g$-load at the
        bottom is $3g$ with $v_0 = \qty{25}{m/s}$.
  \item At the top, the height is still $2R = \qty{20}{m}$ above the
        bottom. Choose $\rho_{\text{top}}$ so that the $g$-load there is
        $1.5g$.
  \item A clothoid has curvature proportional to arc length,
        $1/\rho = s/A^2$. Along such an entry, with the speed nearly
        constant, how does $a_N$ grow with time, and why is that gentler
        on the neck than a circle (where $a_N$ jumps at the entry)?
  \item Sketch the resulting loop shape qualitatively and summarize in
        two sentences why it replaced the circle.
\end{enumerate}

\medskip
\noindent\textbf{Part V --- Seen from the ground.}
A \omterm{def:b1:optical-instruments:camera}{camera} at ground level, \qty{50}{m} from the loop's center in its
plane, follows the car.
\begin{enumerate}[resume]
  \item When the car is at the top, moving horizontally at
        \qty{15}{m/s}, at what angular rate (rad/s) must the \omterm{def:b1:optical-instruments:camera}{camera} turn?
        (\omterm{def:b1:point-kinematics:cylindrical}{Polar coordinates} from the \omterm{def:b1:optical-instruments:camera}{camera}.)
  \item Why is a \omterm{def:b1:optical-instruments:camera}{camera} that turns at constant rate a poor tracker of
        uniform \omterm{cor:b1:point-kinematics:circular}{circular motion}?
  \item Summarize the chapter's lesson from the loop: which coordinate
        system answered which question.
\end{enumerate}
\end{problem}
