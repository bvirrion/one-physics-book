\chapter{Hydrostatica}\label{ch:b1:fluid-statics}

Een onderzeeër op driehonderd meter draagt op elke vierkante meter van
zijn romp het gewicht van een volgeladen vrachtwagen. Een stuwdam houdt een

\section{Druk in een fluïdum in rust}

\begin{proposition}[Isotropie van de druk]\label{prop:b1:fluid-statics:isotropy}
\index{druk!in een fluïdum}
In een fluïdum in rust is de kracht die het fluïdum uitoefent op een klein oppervlak
$\dd S$ in een punt $M$ gelijk aan $\dd\vect F = P(M)\,\dd S\,\vect n$, loodrecht op
het oppervlak en ernaartoe gericht, met een druk $P(M)$ die van
het punt afhangt maar \emph{niet} van de oriëntatie van het oppervlak.
\end{proposition}

\begin{proof}[Gedeeltelijk bewijs]
Een fluïdum in rust draagt geen tangentiële kracht over (anders zou het stromen):
de kracht staat loodrecht. Beschouw een piepklein driehoekig prisma fluïdum in $M$
met vlakken van verschillende oriëntatie: het is in evenwicht onder de
drukkrachten op zijn vlakken en zijn gewicht; het gewicht schaalt met het
volume (derde macht van de afmeting), de drukkrachten met oppervlakken (tweede
macht), zodat voor een klein genoeg prisma alleen de drukkrachten tellen, en
hun evenwicht in elke richting dezelfde $P$ op elk vlak vereist.
\end{proof}

\begin{theorem}[Grondbetrekking van de hydrostatica]\label{thm:b1:fluid-statics:fundamental}
\index{hydrostatische betrekking}
In een fluïdum in rust in het homogene \omterm{prop:b1:newton-dynamics:weight}{zwaartekrachtveld} $\vect g = -g\vect e_z$ ($z$
naar boven) hangt de druk alleen van $z$ af en geldt
\[
\frac{\dd P}{\dd z} = -\rho g ,
\]
waarin $\rho$ de dichtheid van het fluïdum op dat niveau is. De druk neemt naar
beneden toe; vlakken van gelijke druk (\emph{isobaren}) zijn horizontaal,
en het vrije oppervlak van een vloeistof dus ook.
\end{theorem}

\begin{proof}
Neem een horizontale schijf fluïdum met oppervlak $S$ tussen $z$ en $z + \dd z$:
zij wordt van onderen omhoog geduwd door $P(z)S$, van boven omlaag door $P(z + \dd z)S$,
en omlaag getrokken door haar gewicht $\rho gS\,\dd z$. In rust:
$P(z)S - P(z + \dd z)S - \rho gS\,\dd z = 0$. Horizontale schijven fluïdum
ondervinden geen horizontale zwaartekracht, dus is $P$ horizontaal uniform.
\end{proof}

\begin{corollary}[Onsamendrukbaar fluïdum]\label{cor:b1:fluid-statics:incompressible}
\index{hydrostatische druk}
In een vloeistof met uniforme dichtheid $\rho$ is de druk op diepte $h$ onder een
punt waar zij $P_0$ is
\[
P = P_0 + \rho gh .
\]
Water: \qty{1}{bar} meer per \qty{10}{m}. Twee punten van een samenhangende
vloeistof op hetzelfde niveau staan onder dezelfde druk; de vrije oppervlakken van
communicerende vaten staan op één niveau; een druk die ergens op een
afgesloten vloeistof wordt uitgeoefend plant zich onverzwakt naar elk punt voort
(het beginsel van Pascal).
\end{corollary}

\begin{proof}
Integreer $\dd P/\dd z = -\rho g$ met constante $\rho$; de rest volgt uit de
horizontaliteit van de isobaren en de eenduidigheid van $P$ op een niveau.
\end{proof}

\begin{example}[Manometers en persen]\label{ex:b1:fluid-statics:manometer}
\index{manometer}\index{hydraulische pers}
Een U-buis met kwik ($\rho = \qty{13600}{kg/m^3}$) waarvan de kolommen
$h = \qty{760}{mm}$ verschillen meet een drukverschil $\rho gh = \qty{1.013e5}{Pa}$
--- de atmosfeer, die een waterkolom van \qty{10.3}{m} draagt. Een
\omterm{ex:b1:fluid-statics:manometer}{hydraulische pers}: een kracht $F_1$ op een zuiger met oppervlak $S_1$ verhoogt de
druk overal met $F_1/S_1$; op de grote zuiger $S_2$ levert dat
$F_2 = F_1S_2/S_1$ --- vijftig keer meer bij vijftig keer het oppervlak, tegen de
prijs van vijftig keer minder slag (de energie blijft behouden: $F_1d_1 = F_2d_2$).
\end{example}

\section{De atmosfeer}

\begin{proposition}[Isotherme atmosfeer]\label{prop:b1:fluid-statics:atmosphere}
\index{barometrische hoogteformule}\index{schaalhoogte}
Voor een \omterm{def:b1:kinetic-theory:model}{ideaal gas} met molaire massa $M$ bij uniforme temperatuur $T$ in een
homogeen \omterm{prop:b1:newton-dynamics:weight}{zwaartekrachtveld} geldt
\[
P(z) = P_0\,\eu^{-z/H}, \qquad H = \frac{RT}{Mg} ,
\]
de \emph{\omterm{prop:b1:fluid-statics:atmosphere}{schaalhoogte}}: ongeveer \qty{8}{km} voor lucht; de druk halveert
elke $H\ln2 \approx \qty{5.5}{km}$.
\end{proposition}

\begin{proof}
$\rho = PM/RT$ (toestandsvergelijking), dus $\dd P/\dd z = -(Mg/RT)P$: een
lineaire eerste-ordevergelijking, $P = P_0\eu^{-Mgz/RT}$.
\end{proof}

\begin{example}[IJle lucht]\label{ex:b1:fluid-statics:altitude}
Met $T = \qty{260}{K}$ (een gemiddelde over de onderste atmosfeer) is $H =
\qty{7.6}{km}$: $P = 0.52P_0$ op \qty{5000}{m}, $0.31P_0$ op de top van de
Everest --- dicht bij de gemeten $0.54$ en $0.33$; de werkelijke atmosfeer
koelt af met de hoogte, waarmee het volume van jaar 2 rekening houdt. Het
model zegt ook waarom de zee anders is: water is achthonderd keer
dichter en vrijwel onsamendrukbaar, zodat zijn druk lineair stijgt, een bar
per tien meter, zonder \omterm{prop:b1:fluid-statics:atmosphere}{schaalhoogte}.
\end{example}

\section{Krachten op wanden}

\begin{proposition}[Kracht op een verticale wand]\label{prop:b1:fluid-statics:wall}
\index{drukmiddelpunt}
Een vloeistof met diepte $h$ tegen een verticale rechthoekige wand met breedte $L$
(de atmosferische druk werkt op beide zijden en valt weg) oefent de
horizontale kracht
\[
F = \tfrac12\rho gh^2L ,
\]
uit, gelijkwaardig aan één kracht die aangrijpt in het \emph{\omterm{prop:b1:fluid-statics:wall}{drukmiddelpunt}},
op diepte $2h/3$ --- de resultante grijpt laag aan, waar de druk het
grootst is.
\end{proposition}

\begin{proof}
Op diepte $x$ is de overdruk $\rho gx$; op de strook met hoogte
$\dd x$ is de kracht $\rho gxL\,\dd x$; integreer van $0$ tot $h$. Haar
moment om de lijn van het oppervlak is $\int_0^h\rho gx^2L\,\dd x = \tfrac13\rho gh^3L$;
deling door $F$ geeft de \omterm{def:b1:angular-momentum:moment}{arm} $2h/3$.
\end{proof}

\begin{example}[Een stuwdam]\label{ex:b1:fluid-statics:dam}
$h = \qty{20}{m}$, $L = \qty{100}{m}$: $F = \tfrac12 \times 1000 \times 9.81 \times
400 \times 100 = \qty{2.0e8}{N}$ --- twintigduizend ton, aangrijpend
\qty{13}{m} onder het oppervlak, en even groot of het meer erachter nu een vijver
of een fjord is: de \emph{hydrostatische paradox}.
\end{example}

\section{De wet van Archimedes}

\begin{theorem}[Archimedes]\label{thm:b1:fluid-statics:archimedes}
\index{wet van Archimedes}\index{opwaartse kracht}\index{drukkingspunt}
Een lichaam dat in een fluïdum in rust is ondergedompeld ondervindt van de drukkrachten
een resultante --- de \emph{\omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht}} --- die even groot is als en tegengesteld
aan het gewicht van het fluïdum dat het verdringt,
\[
\vect\Pi = -\rho_{\text{fluïdum}}V_{\text{ondergedompeld}}\,\vect g ,
\]
aangrijpend in het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} van het verdrongen fluïdum (het
\emph{drukkingspunt}).
\end{theorem}

\begin{proof}
De drukkrachten op het oppervlak van het lichaam hangen alleen af van de vorm van
dat oppervlak en van het fluïdum eromheen. Vervang het lichaam in gedachten door
fluïdum in rust dat hetzelfde volume vult: dat fluïdum is in evenwicht
onder zijn gewicht en dezelfde drukkrachten, dus houden de drukkrachten
zijn gewicht in evenwicht --- zij tellen op tot $-\rho V\vect g$, aangrijpend door zijn
\omterm{def:b1:systems-of-points:cm}{massamiddelpunt}. Haal het gedachte-experiment weg; de drukkrachten zijn
onveranderd.
\end{proof}

\begin{corollary}[Drijven]\label{cor:b1:fluid-statics:floating}
\index{drijven}
Een lichaam met dichtheid $\rho_b$ drijft in een fluïdum met dichtheid $\rho_f > \rho_b$
met het deel $\rho_b/\rho_f$ van zijn volume ondergedompeld; is $\rho_b >
\rho_f$, dan zinkt het, met een schijnbaar gewicht $(\rho_b - \rho_f)Vg$.
\end{corollary}

\begin{proof}
Gewicht $\rho_bVg$ gelijk aan de \omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} $\rho_fV_{\text{ond}}g$.
\end{proof}

\begin{example}[Drie opwaartse krachten]\label{ex:b1:fluid-statics:buoy}
Een areometer is een verzwaarde buis die zakt tot zij haar eigen gewicht
verdringt: hoe dichter de vloeistof, hoe minder zij zakt --- een dichtheidsmeter
die men op de steel afleest. Een heteluchtballon van \qty{2800}{m^3} bij \qty{373}{K} verdringt
\qty{3.4}{t} koude lucht en weegt \qty{2.6}{t} hete lucht: \qty{0.7}{t}
draagkracht (\cref{pb:b1:kinetic-theory:1}). Een schip van \qty{100000}{t}
verdringt \qty{97600}{m^3} zeewater; in zoet water ($\qty{1000}{kg/m^3}$)
moet het $2.5\%$ meer verdringen en zakt het dieper --- vandaar de lastlijnmerken op
zijn romp.
\end{example}

\begin{remark}[Stabiliteit]\label{rem:b1:fluid-statics:stability}
\index{metacentrum}
Een drijvend lichaam dat over een kleine hoek kantelt richt zich weer op als de
\omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht}, die nu aangrijpt in het verschoven midden van het nieuwe
ondergedompelde volume, een oprichtend moment geeft --- wat gebeurt wanneer het
\emph{metacentrum} (het punt waar de werklijn van de \omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} de as van
het lichaam snijdt) boven $G$ ligt. Ballast laag in de romp verlaagt $G$ en maakt
het schip stabiel; een topzware boot slaat om.
\end{remark}

\end{exercise}

\section{Probleem: De onderzeeër}

\begin{problem}[{Weekendopgave --- een stalen romp op driehonderd meter
diepte: de druk op zijn platen, het water dat hij moet slikken om te zakken
en uitspugen om te stijgen, de duikster die hem verlaat, en de tanker die
overvaart}]
\label{pb:b1:fluid-statics:1}
Zeewater: $\rho = \qty{1025}{kg/m^3}$; zoet water \qty{1000}{kg/m^3};
$P_0 = \qty{1.013e5}{Pa}$; $g = \qty{9.81}{m/s^2}$. De romp van de onderzeeër
omsluit $V = \qty{3000}{m^3}$; met lege ballasttanks is zijn massa
$m_0 = \qty{2800}{t}$.

\medskip
\noindent\textbf{Deel I --- Druk op de romp.}
\begin{enumerate}
  \item Absolute druk op \qty{300}{m}, in \omterm{def:b1:kinetic-theory:pressure}{pascal} en in bar.
  \item Kracht op een rond luik met diameter \qty{0.80}{m} op die diepte;
        vergelijk met het gewicht van een vrachtwagen.
  \item Kracht op één vierkante meter romp bij de kiel als de romp
        \qty{10}{m} hoog is en de bovenkant op \qty{300}{m} ligt: hoeveel verschilt
        dat van de bovenkant?
  \item Water is licht samendrukbaar ($\chi_T = \qty{4.6e-10}{Pa^{-1}}$):
        met welk deel is zeewater op \qty{300}{m} dichter dan aan het
        oppervlak? Maakt dat iets uit voor de hierboven berekende druk?
  \item De bemanning binnen ademt lucht van \qty{1}{atm}: waarom moet de romp
        een dikke cilinder zijn, en waarom weerstaat een cilinder de druk
        beter dan een platte doos?
  \item Een dunne cilindrische schaal met straal $R$ en wanddikte $e$
        onder een uitwendige overdruk $\Delta P$ draagt in haar wand een
        drukspanning $\Delta P\,R/e$. Bereken die voor $R = \qty{5.0}{m}$ en
        $e = \qty{3.0}{cm}$ op \qty{300}{m} en vergelijk met de
        vloeigrens van het staal, ongeveer \qty{600}{MPa}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Zinken en drijven.}
\begin{enumerate}[resume]
  \item Welk volume van de romp steekt aan de oppervlakte met lege tanks boven water uit?
  \item Welke massa zeewater moeten de ballasttanks innemen om de
        onderzeeër te laten zweven (neutraal drijfvermogen)?
  \item Eenmaal zwevend op \qty{300}{m} wordt de romp door de druk met
        $0.1\%$ samengedrukt. Bereken de resulterende nettokracht en zeg
        of het evenwicht stabiel is ten opzichte van de diepte.
  \item De boot vaart zwevend van zee een zoetwaterestuarium
        binnen. Nettokracht? Welke kant gaat hij op, en wat moet de bemanning
        doen?
  \item Er wordt een torpedo van \qty{20}{t} afgevuurd. Wat moet het ballastsysteem
        onmiddellijk doen, en met hoeveel?
  \item Waarom gebruikt een onderzeeër perslucht om zijn tanks leeg te blazen, en
        waarom begrenst de druk van die lucht de diepte waarop dat nog kan?
  \item Schat het volume lucht van \qty{1}{atm} dat nodig is om
        \qty{275}{m^3} tanks op \qty{300}{m} leeg te blazen (de wet van Boyle, $PV$
        constant).
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De duikster.}
Een duikster verlaat het oppervlak met \qty{6.0}{L} lucht in haar longen.
\begin{enumerate}[resume]
  \item Absolute druk op \qty{10}{m} en \qty{30}{m}.
  \item Daalt zij met ingehouden adem, wat is dan haar longvolume op
        \qty{30}{m}?
  \item Een duikster met persluchtfles ademt op \qty{30}{m} lucht van de
        omgevingsdruk en stijgt met ingehouden adem: welk volume zouden haar
        longen aan het oppervlak nodig hebben, en wat is de les?
  \item Een snorkel van \qty{1.0}{m}: drukverschil tussen het
        water op de borst van de duikster en de lucht in haar longen; kracht op een
        borstkas van \qty{0.05}{m^2}; kan zij inademen?
  \item De duikfles bevat \qty{12}{L} bij \qty{200}{bar}: hoeveel liter
        lucht levert dat op \qty{30}{m}, en hoe lang duurt dat bij
        \qty{20}{L} per minuut?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De tanker erboven.}
\begin{enumerate}[resume]
  \item Een tanker van \qty{100000}{t} drijft in zeewater: volume verdrongen
        water.
  \item Zijn waterlijnoppervlak is \qty{8000}{m^2}: hoeveel rijst of zakt hij
        wanneer hij zoet water binnenvaart?
  \item Er wordt \qty{10000}{t} olie geladen: verandering van de diepgang.
  \item De lege tanker heeft een hoog \omterm{def:b1:systems-of-points:cm}{massamiddelpunt}; leeg neemt hij zeewater
        als ballast in. Waarom, in termen van het metacentrum?
  \item De romp van de tanker bij de kiel (diepgang \qty{20}{m}): druk en
        kracht per vierkante meter, vergeleken met de onderzeeër op
        \qty{300}{m}.
  \item De tanker vaart recht over de onderzeeër heen. Voelt de
        onderzeeër zijn \qty{100000}{t}? Verklaar dat met het drukveld.
  \item Vat samen: de ene betrekking die elk getal in deze opgave heeft
        vastgelegd, en de stelling die eruit volgde.
\end{enumerate}
\end{problem}
