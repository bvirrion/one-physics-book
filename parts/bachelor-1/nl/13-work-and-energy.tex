\chapter{Arbeid, energie en potentiële energie}\label{ch:b1:work-and-energy}

Een fietser stopt met trappen boven aan een heuvel en komt beneden aan
met een snelheid die van het hoogteverschil afhangt en niet van de vorm van de weg. Een
bungeejumper valt zestig meter en staat een ogenblik stil, precies
waar de trek van het koord alle snelheid heeft opgegeten die de val had gegeven. Een
waterstofatoom gebonden aan een chlooratoom trilt in een put waarvan de diepte
de energie is die het kost om het molecuul uiteen te rukken. In elk geval wordt
één getal tussen vormen uitgewisseld --- beweging, hoogte, rek,
binding --- en opgeteld verandert het niet. Dit hoofdstuk bouwt die
boekhouding op uit de tweede wet van Newton: \omterm{def:b1:work-and-energy:work}{arbeid} en vermogen, de
stelling van de \omterm{thm:b1:work-and-energy:ke}{kinetische energie}, potentiële energieën voor de krachten die er een toelaten, en
het \omterm{thm:b1:work-and-energy:em}{behoud van mechanische energie} dat veel dynamicavraagstukken
tot algebra van één regel maakt en, via de vorm van een \omterm{def:b1:work-and-energy:turning}{potentiaalput},
verklaart welke standen stabiel zijn.

\section{Vermogen en arbeid}

\begin{definition}[Vermogen en arbeid van een kracht]\label{def:b1:work-and-energy:work}
\index{vermogen (van een kracht)}\index{arbeid}
Het \emph{vermogen} van een kracht $\vect F$ die op een punt werkt dat met
snelheid $\vect v$ beweegt, is
\[
\mathcal P = \vect F\cdot\vect v \qquad (\text{watt}),
\]
en haar \emph{arbeid} tussen de tijdstippen $t_1$ en $t_2$ (posities $A$ en $B$)
is het opgehoopte vermogen,
\[
W_{A\to B} = \int_{t_1}^{t_2}\vect F\cdot\vect v\,\dd t
= \int_A^B \vect F\cdot\dd\vect\ell \qquad (\text{joule}),
\]
de som van de \emph{elementaire arbeidsbijdragen} $\delta W = \vect F\cdot\dd\vect\ell$
langs de elementaire verplaatsingen $\dd\vect\ell = \vect v\,\dd t$ van de
\omterm{def:b1:point-kinematics:frame}{baan}. De arbeid is positief als de kracht met de beweging mee duwt
(\emph{drijvend}), negatief als zij tegenwerkt (\emph{remmend}), en nul als zij er
loodrecht op staat.
\end{definition}

\begin{proposition}[Arbeid die men tegenkomt]\label{prop:b1:work-and-energy:works}
\begin{itemize}
  \item Constante kracht: $W_{A\to B} = \vect F\cdot\vect{AB}$, wat de
        weg ook is. Gewicht: $W = mg(z_A - z_B)$, met $z$ naar boven geteld.
  \item Veer met veerconstante $k$, van uitrekking $x_A$ tot $x_B$:
        $W = \tfrac12 k(x_A^2 - x_B^2)$.
  \item \omterm{prop:b1:newton-dynamics:friction}{Normaalkracht} van een vast oppervlak, spankracht van een koord op een lichaam
        dat er loodrecht op beweegt: $W = 0$ (kracht $\perp$ verplaatsing).
  \item Dynamische wrijving en luchtweerstand: altijd $W < 0$ (de kracht werkt de
        relatieve beweging tegen), en de \omterm{def:b1:work-and-energy:work}{arbeid} hangt af van de weglengte.
\end{itemize}
\end{proposition}

\begin{proof}
Constante $\vect F$: $\int\vect F\cdot\dd\vect\ell = \vect F\cdot\int\dd\vect\ell
= \vect F\cdot\vect{AB}$; met $\vect F = -mg\vect e_z$ is $\vect F\cdot\vect{AB}
= -mg(z_B - z_A)$. Veer: $\vect F\cdot\dd\vect\ell = -kx\,\dd x$, en
$\int_{x_A}^{x_B}-kx\,\dd x = \tfrac12 k(x_A^2 - x_B^2)$. Wrijving: $\vect T
= -\mu N\vect v/v$, dus $\vect T\cdot\vect v = -\mu Nv < 0$.
\end{proof}

\section{De stelling van de kinetische energie}

\begin{theorem}[Stellingen van de kinetische energie en het vermogen]\label{thm:b1:work-and-energy:ke}
\index{kinetische energie}\index{stelling van de kinetische energie}
In een inertiaalstelsel voldoet de \emph{\omterm{thm:b1:work-and-energy:ke}{kinetische energie}} $E_k = \tfrac12 mv^2$ van
een \omterm{def:b1:point-kinematics:frame}{puntmassa} aan
\[
\frac{\dd E_k}{\dd t} = \sum\mathcal P_i = \Big(\sum\vect F_i\Big)\cdot\vect v,
\qquad
E_k(B) - E_k(A) = \sum_i W_{i,A\to B}:
\]
de verandering van de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} tussen twee posities is gelijk aan de totale \omterm{def:b1:work-and-energy:work}{arbeid}
van alle krachten die onderweg aangrijpen.
\end{theorem}

\begin{proof}
Neem het inproduct van de tweede wet van Newton met $\vect v$: $m\vect a\cdot\vect v =
\tfrac{\dd}{\dd t}(\tfrac12 m\vect v\cdot\vect v) = \dd E_k/\dd t$, terwijl
$(\sum\vect F_i)\cdot\vect v = \sum\mathcal P_i$. Integreer over de tijd.
\end{proof}

\begin{example}[Remweg]\label{ex:b1:work-and-energy:braking}
Een auto van \qty{1000}{kg} bij \qty{100}{km/h} ($E_k = \qty{386}{kJ}$), geremd door
een constante kracht van \qty{7.0}{kN}, staat na $d = E_k/F = \qty{55}{m}$ stil:
de afstand groeit met het \emph{kwadraat} van de snelheid. Omdat de
grootste remkracht $\mu_smg$ is (\cref{ch:b1:newton-dynamics}), is de
kortste remweg $v^2/2\mu_sg$, wat de massa van de auto ook is ---
en hij verdubbelt op een natte weg, waar $\mu_s$ halveert.
\end{example}

\section{Conservatieve krachten en potentiële energie}

\begin{definition}[Conservatieve kracht; potentiële energie]\label{def:b1:work-and-energy:conservative}
\index{conservatieve kracht}\index{potentiële energie}
Een kracht heet \emph{conservatief} als haar \omterm{def:b1:work-and-energy:work}{arbeid} tussen twee punten niet
van de gevolgde weg afhangt. Gelijkwaardig: er bestaat een functie van
de plaats, de \emph{potentiële energie} $E_p$, zodanig dat
\[
W_{A\to B} = E_p(A) - E_p(B) = -\Delta E_p ,
\qquad \delta W = -\dd E_p ,
\]
waarbij $E_p$ op een additieve constante na is bepaald (een keuze van nulpunt).
\end{definition}

\begin{proposition}[Kracht uit de potentiële energie]\label{prop:b1:work-and-energy:gradient}
\index{gradiënt}
Een \omterm{def:b1:work-and-energy:conservative}{conservatieve kracht} is af te leiden uit haar \omterm{def:b1:work-and-energy:conservative}{potentiële energie}:
\[
\vect F = -\overrightarrow{\operatorname{grad}}\,E_p
= -\frac{\partial E_p}{\partial x}\,\vect e_x
- \frac{\partial E_p}{\partial y}\,\vect e_y
- \frac{\partial E_p}{\partial z}\,\vect e_z ,
\]
en in één dimensie $F_x = -\dd E_p/\dd x$: de kracht wijst naar
afnemende \omterm{def:b1:work-and-energy:conservative}{potentiële energie}, ``bergafwaarts'' op de grafiek van $E_p$.
\end{proposition}

\begin{proof}
$\delta W = F_x\dd x + F_y\dd y + F_z\dd z = -\dd E_p = -(\partial_xE_p\,\dd x
+ \partial_yE_p\,\dd y + \partial_zE_p\,\dd z)$ voor elke verplaatsing:
identificeer de coëfficiënten (de differentiaal van een functie van meerdere
variabelen, uit de wiskunde).
\end{proof}

\begin{proposition}[De gebruikelijke potentiële energieën]\label{prop:b1:work-and-energy:eps}
\index{potentiële energie!zwaarte-}\index{potentiële energie!elastische}
\begin{itemize}
  \item Gewicht: $E_p = mgz + \text{const}$ ($z$ omhoog).
  \item Veer: $E_p = \tfrac12 k(\ell - \ell_0)^2 + \text{const}$.
  \item Newtoniaanse gravitatie van een massa $M$: $E_p = -GMm/r + \text{const}$,
        waarbij de constante meestal zo wordt gekozen dat $E_p \to 0$ op oneindig.
  \item Elektrostatische kracht op een lading $q$ in een potentiaal $V$:
        $E_p = qV$ (\cref{ch:b1:potential-capacitors}).
\end{itemize}
Dynamische wrijving en vloeistofweerstand zijn \emph{niet} conservatief (hun \omterm{def:b1:work-and-energy:work}{arbeid}
hangt van de weg af en is altijd negatief).
\end{proposition}

\begin{proof}
Elk ervan is af te lezen uit de \omterm{def:b1:work-and-energy:work}{arbeid} in \cref{prop:b1:work-and-energy:works}:
$W = -\Delta E_p$. Voor de gravitatie is $\vect F = -GMm/r^2\,\vect e_r$ en
$\delta W = -GMm\,\dd r/r^2 = -\dd(-GMm/r)$.
\end{proof}

\section{Mechanische energie}

\begin{theorem}[Mechanische energie]\label{thm:b1:work-and-energy:em}
\index{mechanische energie}\index{behoud van mechanische energie}
Zij $E_m = E_k + E_p$ de \emph{\omterm{thm:b1:work-and-energy:em}{mechanische energie}}, waarin $E_p$ de
som is van de potentiële energieën van de \omterm{def:b1:work-and-energy:conservative}{conservatieve krachten}. Dan geldt
\[
\Delta E_m = W_{\text{nc}} ,
\]
de totale \omterm{def:b1:work-and-energy:work}{arbeid} van de niet-conservatieve krachten. Verrichten die geen \omterm{def:b1:work-and-energy:work}{arbeid}
(geen wrijving, of dwangkrachten loodrecht op de beweging), dan blijft $E_m$
\emph{behouden}: de beweging is \emph{conservatief}.
\end{theorem}

\begin{proof}
$\Delta E_k = W_{\text{c}} + W_{\text{nc}} = -\Delta E_p + W_{\text{nc}}$.
\end{proof}

\begin{method}[Energie gebruiken]\label{met:b1:work-and-energy:method}
Vraagt een vraagstuk om een \emph{snelheid op een plaats} (geen tijd, geen
kracht), probeer dan eerst de energie:
\begin{enumerate}
  \item som de krachten op; ga na welke \omterm{def:b1:work-and-energy:work}{arbeid} verrichten en welke conservatief zijn;
  \item schrijf $E_m$ op de twee plaatsen op, met een duidelijk nulpunt voor $E_p$;
  \item stel ze gelijk (tel $W_{\text{nc}}$ erbij als er wrijving is) en los
        op naar de onbekende snelheid of plaats.
\end{enumerate}
Energie levert nooit de tijd of de dwangkrachten; daarvoor keer je terug
naar de wet van Newton --- vaak met de snelheid die de energie zojuist gaf.
\end{method}

\begin{example}[De snelheidswet van de looping]\label{ex:b1:work-and-energy:loop}
Op de wrijvingsloze looping van \cref{pb:b1:point-kinematics:1} staat de
\omterm{prop:b1:newton-dynamics:friction}{normaalkracht} van de \omterm{def:b1:point-kinematics:frame}{baan} loodrecht op de beweging en verricht geen \omterm{def:b1:work-and-energy:work}{arbeid}: $E_m$ blijft behouden.
Met $z = R(1 - \cos\theta)$ boven de onderkant geldt $\tfrac12 mv^2 + mgR(1 -
\cos\theta) = \tfrac12 mv_0^2$: de snelheidswet $v^2 = v_0^2 - 2gR(1 -
\cos\theta)$ die daar werd gebruikt, nu in één regel afgeleid. De kleinste
inrijsnelheid $\sqrt{5gR}$ volgt dan uit de voorwaarde bovenin,
$v^2 \geq gR$, uit \cref{ex:b1:newton-dynamics:circles}.
\end{example}

\begin{example}[Energie verloren aan wrijving]\label{ex:b1:work-and-energy:friction}
Een blok glijdt een helling met hoek $\alpha$ en hoogte $h$ af met
dynamische wrijving $\mu_d$: $W_{\text{nc}} = -\mu_dmg\cos\alpha \times
h/\sin\alpha = -\mu_dmgh\cot\alpha$, dus $\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha)$.
Voor $h = \qty{2.0}{m}$, $\alpha = 30^\circ$, $\mu_d = 0.20$: $v =
\qty{5.1}{m/s}$ in plaats van $\qty{6.3}{m/s}$; een derde van de \omterm{def:b1:work-and-energy:conservative}{potentiële
energie} werd warmte.
\end{example}

\section{Evenwicht, stabiliteit en faseportretten}

\begin{proposition}[Evenwicht en stabiliteit in één dimensie]\label{prop:b1:work-and-energy:stability}
\index{evenwicht!stabiel, labiel}\index{kleine trillingen}
Voor een \omterm{def:b1:point-kinematics:frame}{puntmassa} die op een as beweegt onder een \omterm{def:b1:work-and-energy:conservative}{conservatieve kracht} met \omterm{def:b1:work-and-energy:conservative}{potentiële
energie} $E_p(x)$ geldt:
\begin{itemize}
  \item zij is in \emph{evenwicht} in $x_0$ precies als $F(x_0) = -E_p'(x_0) = 0$:
        in de extrema van $E_p$;
  \item het evenwicht is \emph{stabiel} als $E_p$ daar een lokaal minimum
        heeft ($E_p''(x_0) > 0$): een kleine uitwijking levert een
        terugdrijvende kracht; \emph{labiel} in een maximum;
  \item vlak bij een stabiel evenwicht zijn kleine bewegingen harmonisch, met
        \[
        \omega_0^2 = \frac{E_p''(x_0)}{m} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Taylorformule (wiskunde): $E_p(x) \approx E_p(x_0) +
\tfrac12 E_p''(x_0)(x - x_0)^2$, dus $F \approx -E_p''(x_0)(x - x_0)$ en
$m\ddot x = -E_p''(x_0)(x - x_0)$ --- een veer met veerconstante $E_p''(x_0)$.
Voor $E_p'' < 0$ is de ``veerconstante'' negatief: exponentieel weglopen.
\end{proof}

\begin{definition}[Keerpunten; gebonden en vrije beweging]\label{def:b1:work-and-energy:turning}
\index{keerpunt}\index{gebonden beweging}\index{potentiaalput}\index{potentiaalbarrière}
Bij een conservatieve eendimensionale beweging met energie $E_m$ kan het deeltje
alleen zijn waar $E_p(x) \leq E_m$ (want $E_k \geq 0$); de punten waar
$E_p = E_m$ zijn \emph{keerpunten}, waar de snelheid nul wordt en
omkeert. Gevangen tussen twee keerpunten in een \emph{potentiaalput}
is de beweging \emph{gebonden} en periodiek; kan zij het oneindige bereiken, dan is zij
\emph{vrij}; een \emph{potentiaalbarrière} hoger dan $E_m$ is niet te
passeren.
\end{definition}

\begin{definition}[Faseportret]\label{def:b1:work-and-energy:phase}
\index{faseportret}
Het \emph{faseportret} van een eendimensionale beweging is de verzameling
krommen die het punt $(x, \dot x)$ in het \emph{fasevlak} beschrijft. Voor
een conservatief systeem is elke kromme een niveaulijn $\tfrac12 m\dot x^2 +
E_p(x) = E_m$: gesloten krommen rond stabiele evenwichten (trillingen),
open krommen bij vrije beweging, en, door de labiele evenwichten heen, de
\emph{separatrices} die de twee scheiden. De krommen worden met de klok mee doorlopen
($\dot x > 0$ betekent toenemende $x$) en snijden elkaar nooit.
\end{definition}

\begin{example}[Het landschap van de slinger]\label{ex:b1:work-and-energy:pendulum}
$E_p = -mg\ell\cos\theta$: minimum bij $\theta = 0$, $E_p'' = mg\ell$, dus
$\omega_0^2 = g\ell/(m\ell^2) = g/\ell$ --- het resultaat voor \omterm{prop:b1:work-and-energy:stability}{kleine trillingen} uit
\cref{ex:b1:newton-dynamics:pendulum}, nu afgelezen van de kromming van de put;
maximum bij $\theta = \pi$, de omgekeerde slinger, labiel. Bij energie
$E_m < mg\ell$: trilling tussen $\pm\theta_{\max}$; bij $E_m > mg\ell$: zij
gaat over de top en draait rond; bij $E_m = mg\ell$: de separatrix, een
oneindig traag naderen van de top.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:work-and-energy:1}
Een kist van \qty{20}{kg} wordt in \qty{5.0}{s} met constante snelheid
\qty{3.0}{m} opgetild. \omterm{def:b1:work-and-energy:work}{Arbeid} van de hefkracht, van het gewicht, en geleverd
vermogen. Dezelfde kist \qty{3.0}{m} over een vloer geduwd met

\section{Probleem: De energieput van een chemische binding}

\begin{problem}[{Weekendopgave --- twee atomen, één kromme: de potentiële
energie van een binding gelezen als een landschap --- haar bodem, haar wanden, de
kleine trillingen onderin, en de energie die het kost om eruit te klimmen}]
\label{pb:b1:work-and-energy:1}
De wisselwerking van de twee atomen van een tweeatomig molecuul wordt gemodelleerd door
de \omterm{def:b1:work-and-energy:conservative}{potentiële energie}
\[
E_p(r) = \epsilon\left[\left(\frac{\sigma}{r}\right)^{12} - 2\left(\frac{\sigma}{r}\right)^{6}\right],
\]
met $r$ de afstand tussen de kernen. Neem voor waterstofchloride
$\epsilon = \qty{4.4}{eV}$, $\sigma = \qty{0.127}{nm}$, en laat het lichte
waterstofatoom ($m = \qty{1.67e-27}{kg}$) bewegen terwijl het zware chloor
blijft staan. $\qty{1}{eV} = \qty{1.60e-19}{J}$; $k_B = \qty{1.38e-23}{J/K}$;
$N_A = \num{6.02e23}$.

\medskip
\noindent\textbf{Deel I --- Het landschap.}
\begin{enumerate}
  \item Geef de limieten van $E_p$ voor $r \to 0$ en $r \to \infty$, en
        schets de kromme.
  \item Toon aan dat $E_p$ één minimum heeft, bij $r = \sigma$, met waarde
        $-\epsilon$.
  \item Toon met $u = (\sigma/r)^6$ aan dat $E_p = \epsilon(u^2 - 2u)$: een
        parabool in $u$. Vind het minimum daaruit terug.
  \item Druk de kracht $F(r)$ uit (positief als zij afstoot) en geef haar
        teken aan weerszijden van $\sigma$.
  \item Bereken $E_p$ en $F$ bij $r = 1.2\sigma$ (in eV en in nN).
  \item Wat is de dissociatie-energie van het molecuul, in eV en in
        \unit{kJ/mol}? (Gemeten bindingsenergie van HCl: \qty{431}{kJ/mol}.)
  \item Controleer het stabiliteitscriterium bij $r = \sigma$: bereken
        $E_p''(\sigma)$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- \omterm{prop:b1:work-and-energy:stability}{Kleine trillingen}.}
\begin{enumerate}[resume]
  \item Schrijf de taylorontwikkeling tot tweede orde van $E_p$ rond $\sigma$
        op en benoem een effectieve veerconstante $k$.
  \item Bereken $k$ (in \unit{N/m}).
  \item Leid de \omterm{def:b1:wave-propagation:signal}{hoekfrequentie}, de frequentie en de golflengte
        van de trilling af, en haar golfgetal $1/\lambda$ in \unit{cm^{-1}}.
  \item De gemeten trilling van HCl ligt bij \qty{2990}{cm^{-1}}. Becommentarieer
        de overeenkomst en wat het model wel goed doet.
  \item Bij \qty{300}{K} is de gemiddelde trillingsenergie van de orde $k_BT$.
        Schat de trillingsamplitude en vergelijk haar met $\sigma$.
  \item Leid daaruit de grootste snelheid van het waterstofatoom in die
        trilling af.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Grote trillingen en de uitweg.}
\begin{enumerate}[resume]
  \item Beschrijf het \omterm{def:b1:work-and-energy:phase}{faseportret}: welke energieën geven gesloten krommen,
        welke open krommen, en wat scheidt ze?
  \item Bepaal voor $E_m = -\epsilon/2$ de twee \omterm{def:b1:work-and-energy:turning}{keerpunten} (in eenheden
        van $\sigma$).
  \item Toon aan dat hun midden voorbij $\sigma$ ligt, en verklaar waarom
        verwarmen (de trillingsenergie verhogen) de gemiddelde
        bindingslengte doet groeien --- de oorsprong van de thermische uitzetting.
  \item Is de periode van grote trillingen langer of korter dan
        $2\pi/\omega_0$? Beargumenteer dat uit de vorm van de wanden.
  \item Welke energie moet er vanaf $E_m = -\epsilon/2$ worden toegevoerd om
        het molecuul te dissociëren? Als zij als \omterm{thm:b1:work-and-energy:ke}{kinetische energie} van het
        waterstofatoom bij $r = \sigma$ wordt gegeven, welke snelheid is dat?
  \item Twee atomen die van ver naderen met positieve totale energie
        kunnen uit zichzelf geen gebonden molecuul vormen. Waarom? Wat is er nodig?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- \omterm{def:b1:work-and-energy:work}{Arbeid}, kracht en temperatuur.}
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:b1:work-and-energy:work}{arbeid} van de bindingskracht wanneer $r$ van $2\sigma$
        naar $\sigma$ gaat. Is de kracht daar drijvend of remmend?
  \item Bereken de \omterm{def:b1:work-and-energy:work}{arbeid} die een uitwendige agens moet leveren om de binding
        oneindig langzaam van $\sigma$ tot $1.5\sigma$ uit te rekken.
  \item Bereken $F$ bij $r = 0.9\sigma$ en vergelijk met $r = 1.2\sigma$:
        wat zegt de asymmetrie over samendrukken tegenover uitrekken?
  \item Bij welke temperatuur zou $k_BT$ gelijk zijn aan $\epsilon$? Waarom
        dissociëren moleculen in de praktijk bij veel lagere temperaturen?
  \item Een beter model (Morse) verandert de vorm van de wanden maar houdt
        een minimum bij de bindingslengte. Welke van de bovenstaande resultaten blijven
        qua methode onveranderd, en welke getallen zouden veranderen?
  \item Vat de drie manieren samen waarop energiemethoden zijn gebruikt: om het
        evenwicht te vinden, om de trillingsfrequentie te vinden, en om te vinden wat
        er nodig is om te ontsnappen.
\end{enumerate}
\end{problem}
