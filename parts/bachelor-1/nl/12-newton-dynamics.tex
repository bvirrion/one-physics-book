\chapter{De bewegingswetten van Newton}\label{ch:b1:newton-dynamics}

Een parachutist stapt uit het vliegtuig en houdt binnen een tiental seconden
op met versnellen: bij tweehonderd kilometer per uur duwt de lucht precies
even hard terug als de aarde trekt. Een auto neemt een bocht te snel en
de banden begeven het; een vrachtwagen op een bergweg kruipt in zijn laagste
versnelling, gehouden door dezelfde wrijving die hem op vlak wegdek laat stoppen. De kinematica
beschreef bewegingen; de dynamica verklaart ze, met één wet die
de versnelling van een lichaam verbindt met de \omterm{def:b1:newton-dynamics:momentum}{krachten} die erop werken. Dit hoofdstuk
formuleert de drie \omterm{thm:b1:newton-dynamics:laws}{wetten van Newton}, brengt de \omterm{def:b1:newton-dynamics:momentum}{krachten} in kaart die men op het
niveau van dit volume tegenkomt --- gewicht, veren, koorden, contact, droge
en vloeistofwrijving --- en laat zien hoe men van een tekening van de \omterm{def:b1:newton-dynamics:momentum}{krachten} naar
een differentiaalvergelijking en haar oplossing gaat.

\section{Massa, impuls en de drie wetten}

\begin{definition}[Massa, impuls, kracht]\label{def:b1:newton-dynamics:momentum}
\index{massa}\index{impuls}\index{kracht}
Een \omterm{def:b1:point-kinematics:frame}{puntmassa} heeft een \emph{traagheidsmassa} $m > 0$ (kilogram), een
maat voor haar weerstand tegen snelheidsveranderingen, onafhankelijk van het
stelsel. Haar \emph{impuls} in een stelsel is
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
Een \emph{kracht} $\vect F$ (newton) is de werking van een ander lichaam op de
massa; krachten tellen op als vectoren, en de \emph{resultante} is hun som
$\sum\vect F$.
\end{definition}

\begin{theorem}[De wetten van Newton]\label{thm:b1:newton-dynamics:laws}
\index{wetten van Newton}\index{inertiaalstelsel}\index{traagheidsbeginsel}\index{actie en reactie}
\begin{enumerate}
  \item \emph{Traagheidsbeginsel.} Er bestaan stelsels, \emph{inertiaal}
        (of galileïsch) genoemd, waarin een \omterm{def:b1:point-kinematics:frame}{puntmassa} waarop geen \omterm{def:b1:newton-dynamics:momentum}{kracht}
        werkt, of \omterm{def:b1:newton-dynamics:momentum}{krachten} met resultante nul, in een rechte
        lijn met constante snelheid beweegt. Elk stelsel dat eenparig rechtlijnig
        transleert ten opzichte van een inertiaalstelsel is zelf inertiaal.
  \item \emph{Impulswet.} In een inertiaalstelsel geldt
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{oftewel}\qquad
        m\,\vect a = \sum\vect F \ \text{ bij constante } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{Actie en reactie}.} Oefent lichaam $A$ de \omterm{def:b1:newton-dynamics:momentum}{kracht} $\vect F_{A\to B}$
        uit op lichaam $B$, dan oefent $B$ de \omterm{def:b1:newton-dynamics:momentum}{kracht} $\vect F_{B\to A} = -\vect F_{A\to B}$
        uit op $A$, langs de verbindingslijn.
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[Wat de wetten zeggen]\label{rem:b1:newton-dynamics:meaning}
De eerste wet is geen bijzonder geval van de tweede: zij \emph{definieert}
de stelsels waarin de tweede geldt. In de praktijk is een laboratorium op
aarde uitstekend inertiaal voor proeven die minuten duren
(\cref{ch:b1:non-inertial-frames} maakt de fout kwantitatief); een stelsel
dat aan de zon en de verre sterren vastzit is nog beter. De tweede
wet is een vectorvergelijking: drie scalaire vergelijkingen, één per as, en
de keuze van de assen (\cref{ch:b1:point-kinematics}) is de eerste
beslissing van elk vraagstuk. De derde wet geldt op elk moment voor
de contact- en zwaartekrachten van dit volume.
\end{remark}

\begin{definition}[Gesloten systeem; behoud van impuls]\label{def:b1:newton-dynamics:conservation}
\index{gesloten systeem}\index{behoud van impuls}
Een \emph{gesloten systeem} wisselt geen materie uit met de omgeving. Voor een
systeem van \omterm{def:b1:point-kinematics:frame}{puntmassa}'s vallen de inwendige \omterm{def:b1:newton-dynamics:momentum}{krachten} paarsgewijs weg (derde wet),
en voldoet de totale \omterm{def:b1:newton-dynamics:momentum}{impuls} $\vect P = \sum m_i\vect v_i$ aan
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$: zij blijft behouden wanneer
de uitwendige \omterm{def:b1:newton-dynamics:momentum}{krachten} verdwijnen (\cref{ch:b1:systems-of-points} werkt dit
uit).
\end{definition}

\section{De gebruikelijke krachten}

\begin{proposition}[Gewicht en gravitatie]\label{prop:b1:newton-dynamics:weight}
\index{gewicht}\index{zwaartekrachtveld}
Dicht bij het aardoppervlak wordt een lichaam met \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ aangetrokken door zijn
\emph{gewicht} $\vect P = m\vect g$, met $\vect g$ het plaatselijke \omterm{prop:b1:newton-dynamics:weight}{zwaartekrachtveld},
$g \approx \qty{9.81}{m/s^2}$, naar beneden gericht (bij zeer goede benadering
naar het middelpunt). Algemener trekt een \omterm{def:b1:newton-dynamics:momentum}{massa} $M$ op afstand $r$ de \omterm{def:b1:newton-dynamics:momentum}{massa}
$m$ aan met $\vect F = -GMm/r^2\,\vect e_r$ (\cref{ch:b1:central-forces});
aan het oppervlak van een bol met straal $R$ is $g = GM/R^2$. Let op de eenheid:
\unit{N/kg} en \unit{m/s^2} zijn hetzelfde, en daarom vallen alle lichamen
gelijk.
\end{proposition}

\begin{proof}
De gravitatiewet van Newton wordt aangenomen en in
\cref{ch:b1:central-forces} verkend; $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$.
\end{proof}

\begin{proposition}[Veer; koord]\label{prop:b1:newton-dynamics:springstring}
\index{wet van Hooke}\index{veer}\index{spankracht}
Een ideale veer met veerconstante $k$ en rustlengte $\ell_0$, uitgerekt
tot lengte $\ell$ langs de eenheidsvector $\vect u$ die van haar vaste
uiteinde naar het lichaam wijst, oefent op dat lichaam de \emph{terugdrijvende \omterm{def:b1:newton-dynamics:momentum}{kracht}}
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{wet van Hooke}),
\]
uit, naar haar rustlengte toe. Een ideaal koord (massaloos, onrekbaar)
oefent een \emph{spankracht} $\vect T$ in zijn eigen richting uit, trekkend, met dezelfde
grootte aan beide uiteinden en overal ertussen (ook over een massaloze,
wrijvingsloze katrol).
\end{proposition}

\begin{proof}
De \omterm{prop:b1:newton-dynamics:springstring}{wet van Hooke} is een empirisch model dat geldt bij kleine vervormingen. Het
koord: een massaloos elementje voldoet aan $0 = \vect T_1 + \vect T_2$ uit de
tweede wet van Newton, zodat de spankracht onveranderd wordt doorgegeven.
\end{proof}

\begin{proposition}[Contactkrachten: normaalkracht en droge wrijving]\label{prop:b1:newton-dynamics:friction}
\index{normaalkracht}\index{droge wrijving}\index{wrijvingswetten van Coulomb}\index{wrijvingscoëfficiënt}
Een vast oppervlak oefent op een lichaam dat het raakt een \emph{\omterm{prop:b1:newton-dynamics:friction}{normaalkracht}}
$\vect N$ uit, loodrecht op het oppervlak en afstotend ($N \geq 0$: een
oppervlak kan niet trekken), en een tangentiële \emph{wrijvingskracht} $\vect T$.
De wetten van Coulomb: glijdt het lichaam niet, dan is $\abs{\vect T} \leq \mu_s N$
(de \emph{statische} wrijving past zich aan wat nodig is, tot die grens);
glijdt het met snelheid $\vect v_g$ ten opzichte van het oppervlak, dan is
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (\emph{dynamische} wrijving,
die het glijden tegenwerkt), met $\mu_d \leq \mu_s$ coëfficiënten van de orde
$0.1$ tot $1$ afhankelijk van de materialen, onafhankelijk van het contactoppervlak
en van de snelheid.
\end{proposition}
\admitted

\begin{proposition}[Vloeistofwrijving]\label{prop:b1:newton-dynamics:drag}
\index{vloeistofwrijving}\index{weerstandskracht}\index{wet van Stokes}
Een lichaam dat met snelheid $\vect v$ door een stilstaande vloeistof beweegt ondervindt een
\emph{weerstandskracht} tegengesteld aan $\vect v$: bij lage snelheid (kleine afmeting, stroperige
vloeistof) is zij lineair, $\vect F = -\alpha\vect v$ --- voor een bol met
straal $r$ in een vloeistof met viscositeit $\eta$ is $\alpha = 6\pi\eta r$
(Stokes); bij hoge snelheid (grote, snelle lichamen in lucht of water) is zij
kwadratisch,
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
met $\rho$ de dichtheid van de vloeistof, $S$ de doorsnede van het lichaam dwars op
de stroming en $C_x$ een dimensieloze \emph{weerstandscoëfficiënt} ($0.5$ voor een
bol, $0.3$ voor een auto, $1$ voor een vlakke schijf). Het getal van Reynolds
$\rho vL/\eta$ (\cref{rem:b1:units-dimensions:pi}) beslist: lineair
onder $1$, kwadratisch boven ongeveer $1000$.
\end{proposition}
\admitted

\begin{remark}[Andere krachten]\label{rem:b1:newton-dynamics:others}
De opwaartse \omterm{def:b1:newton-dynamics:momentum}{kracht} (Archimedes) is een resultante van drukkrachten, afgeleid in
\cref{ch:b1:fluid-statics}; de elektrische en magnetische \omterm{def:b1:newton-dynamics:momentum}{krachten} op een
lading zijn het onderwerp van \cref{ch:b1:charged-particles}. Alle
\omterm{def:b1:newton-dynamics:momentum}{krachten} van dit hoofdstuk zijn \emph{modellen} van contact- of
veldwisselwerkingen, geldig binnen aangegeven grenzen --- de veer van Hooke breekt,
de wrijving van Coulomb faalt bij hoge druk, de weerstandskracht van Stokes faalt bij hoge
snelheid. Bij elke oplossing hoort de controle dat het gebruikte model van toepassing is.
\end{remark}

\section{Een dynamicavraagstuk oplossen}

\begin{method}[Van de situatie naar de vergelijking]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{Systeem}: benoem het lichaam dat als punt wordt behandeld.
  \item \emph{Stelsel}: benoem het en ga na dat het inertiaal is (of het als
        zodanig mag gelden binnen de vereiste nauwkeurigheid).
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{Krachten}}: som elk lichaam op dat het systeem raakt of aantrekt,
        en teken elke \omterm{def:b1:newton-dynamics:momentum}{kracht} in een schema (het
        krachtenschema); er komt niets anders bij.
  \item \emph{Coördinaten}: kies assen die bij de beweging passen
        (\cref{met:b1:point-kinematics:choose}) en druk $\vect a$ uit.
  \item \emph{Projecteer} $m\vect a = \sum\vect F$ op de assen; gebruik de
        randvoorwaarden (een lichaam op een oppervlak blijft erop: $a_\perp = 0$; een
        koord met vaste lengte: gelijke snelheden aan beide uiteinden).
  \item \emph{Los op} en \emph{controleer}: dimensies, grensgevallen,
        tekens ($N \geq 0$, $\abs T \leq \mu_sN$ in rust).
\end{enumerate}
\end{method}

\begin{example}[Blok op een helling]\label{ex:b1:newton-dynamics:incline}
Een blok met \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ op een vlak met hellingshoek $\alpha$, coëfficiënten $\mu_s$
en $\mu_d$. Assen: $x$ langs de helling omlaag, $y$ langs de normaal. Gewicht
$(mg\sin\alpha, -mg\cos\alpha)$, \omterm{prop:b1:newton-dynamics:friction}{normaalkracht} $(0, N)$, wrijving $(-T, 0)$
als het naar beneden glijdt. Langs $y$: $N = mg\cos\alpha$. In rust langs $x$:
$T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$, dus $\tan\alpha
\leq \mu_s$: het blok blijft liggen tot de hoek $\arctan\mu_s$ --- een
meting van $\mu_s$ waarvoor geen weegschaal nodig is. Glijdend: $ma = mg\sin\alpha -
\mu_dmg\cos\alpha$, dus $a = g(\sin\alpha - \mu_d\cos\alpha)$, onafhankelijk
van $m$: \qty{3.2}{m/s^2} voor $\alpha = 30^\circ$, $\mu_d = 0.2$.
\end{example}

\begin{example}[De wiskundige slinger]\label{ex:b1:newton-dynamics:pendulum}
\index{wiskundige slinger}
\omterm{def:b1:newton-dynamics:momentum}{Massa} $m$ aan een koord met lengte $\ell$, hoek $\theta$ met de neerwaartse
verticaal. \omterm{def:b1:point-kinematics:cylindrical}{Poolcoördinaten} in het ophangpunt: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$; \omterm{def:b1:newton-dynamics:momentum}{krachten}:
het gewicht $mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$ en de
spankracht $-T\vect e_r$. Langs $\vect e_\theta$: $m\ell\ddot\theta =
-mg\sin\theta$,
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
wat voor kleine hoeken ($\sin\theta \approx \theta$) de harmonische
vergelijking is met $\omega_0 = \sqrt{g/\ell}$: periode $T_0 = 2\pi\sqrt{\ell/g}$,
de $k = 2\pi$ die \cref{ex:b1:units-dimensions:pendulum} niet kon
leveren. Langs $\vect e_r$: $T = mg\cos\theta + m\ell\dot\theta^2$ --- het
koord trekt het hardst onderin, waar de snelheid het grootst is.
\end{example}

\begin{proposition}[Val met weerstand; eindsnelheid]\label{prop:b1:newton-dynamics:terminal}
\index{eindsnelheid}
Een lichaam met \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ dat vanuit rust valt onder zijn gewicht en een weerstandskracht
bereikt een \emph{\omterm{prop:b1:newton-dynamics:terminal}{eindsnelheid}} waarbij de twee elkaar in evenwicht houden. Lineaire
weerstand: $m\dot v = mg - \alpha v$, $v = v_\infty(1 - \eu^{-t/\tau})$ met
$v_\infty = mg/\alpha$, $\tau = m/\alpha$. Kwadratische weerstand: $m\dot v = mg -
\tfrac12\rho C_xSv^2$, $v = v_\infty\tanh(t/\tau)$ met
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
Lineair: een vergelijking van de eerste orde, $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder}). Kwadratisch: met
$u = v/v_\infty$ is $\dot u = (1 - u^2)/\tau$; scheiding van variabelen geeft
$\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$ (de rationale
breuk $1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$ integreert tot
$\tfrac12\ln\frac{1 + u}{1 - u}$), dus $u = \tanh(t/\tau)$.
\end{proof}

\begin{example}[Twee eindsnelheden]\label{ex:b1:newton-dynamics:terminal}
Een parachutist, $m = \qty{80}{kg}$, buikwaarts, $C_xS \approx \qty{0.8}{m^2}$
in lucht ($\rho = \qty{1.2}{kg/m^3}$): $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$, bereikt (tot op $90\%$) in ongeveer $1.5\tau =
\qty{6}{s}$. Een mistdruppeltje, $r = \qty{10}{\micro m}$, in het regime van Stokes
($\eta_{\text{air}} = \qty{1.8e-5}{Pa.s}$): $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- en daarom blijft mist in de lucht hangen.
\end{example}

\section{Beweging op een cirkel: dwangkrachten}

\begin{proposition}[De centripetale eis]\label{prop:b1:newton-dynamics:centripetal}
\index{centripetale kracht}
Een \omterm{def:b1:point-kinematics:frame}{puntmassa} die op een cirkel met straal $R$ met snelheid $v$ beweegt, moet
van de resultante van de \omterm{def:b1:newton-dynamics:momentum}{krachten} een component $mv^2/R$ naar het
middelpunt ontvangen. Wat die ook levert --- de spankracht van een koord, wrijving, de
\omterm{prop:b1:newton-dynamics:friction}{normaalkracht} van een verkante weg, de zwaartekracht --- het legt de hoogste snelheid of de
laagste snelheid vast waarbij de beweging mogelijk is.
\end{proposition}

\begin{proof}
De tweede wet van Newton met de frenetversnelling: de normale component
van $\sum\vect F$ is gelijk aan $mv^2/\rho$.
\end{proof}

\begin{example}[Drie cirkels]\label{ex:b1:newton-dynamics:circles}
\emph{Vlakke bocht}: de wrijving moet $mv^2/R \leq \mu_sN = \mu_smg$ leveren,
dus $v \leq \sqrt{\mu_sgR}$: \qty{20}{m/s} voor $R = \qty{50}{m}$,
$\mu_s = 0.8$. \emph{Verkante bocht} met hoek $\beta$, zonder wrijving: de
\omterm{prop:b1:newton-dynamics:friction}{normaalkracht} $\vect N$ staat loodrecht op het wegdek; verticaal $N\cos\beta = mg$,
horizontaal $N\sin\beta = mv^2/R$, dus $v^2 = gR\tan\beta$ --- de ene
snelheid waarbij de bocht zonder enige wrijving wordt genomen. \emph{Looping}: bovenin,
met $\vect N$ omlaag naar het middelpunt gericht, geldt $N + mg =
mv^2/R$, en $N \geq 0$ vraagt $v \geq \sqrt{gR}$: daaronder verlaat het wagentje
de \omterm{def:b1:point-kinematics:frame}{baan} (\cref{pb:b1:point-kinematics:1} opnieuw, nu verklaard).
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:1}
Iemand met een \omterm{def:b1:newton-dynamics:momentum}{massa} van \qty{70}{kg} staat op een weegschaal in een lift. Wat
wijst de weegschaal aan (in newton en in ``kilogram'') wanneer de lift
omhoog versnelt met \qty{2.0}{m/s^2}, omlaag met \qty{2.0}{m/s^2},
met constante snelheid beweegt, en vrij valt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:2}
Een blok $m_1 = \qty{2.0}{kg}$ op een wrijvingsloze tafel is met een
koord over een wrijvingsloze katrol verbonden met een hangende \omterm{def:b1:newton-dynamics:momentum}{massa} $m_2 = \qty{1.0}{kg}$.
Bepaal de versnelling en de spankracht.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:3}
Een blok glijdt langs een helling van $30^\circ$ met $\mu_d = 0.20$. Bereken
zijn versnelling en zijn snelheid na \qty{5.0}{m} vanuit rust. Hangt het
antwoord af van zijn \omterm{def:b1:newton-dynamics:momentum}{massa}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:4}
Een \omterm{def:b1:newton-dynamics:momentum}{massa} van \qty{0.50}{kg} hangt aan een veer met veerconstante \qty{200}{N/m}.
Bepaal de statische uitrekking; daarna de \omterm{def:b1:wave-propagation:signal}{hoekfrequentie} en de periode van
haar verticale trillingen rond het evenwicht (toon aan dat de bewegingsvergelijking
harmonisch is).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:5}
Een kist staat op een kantelbaar plateau; zij begint te glijden bij $22^\circ$.
Bepaal $\mu_s$. Een auto op een vlakke weg met $\mu_s = 0.80$: maximale snelheid
in een bocht met straal \qty{50}{m}, en op een natte weg met $\mu_s = 0.40$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:6}
Eindsnelheden: een parachutist ($m = \qty{80}{kg}$, $C_xS = \qty{0.70}{m^2}$,
$\rho = \qty{1.2}{kg/m^3}$); een mistdruppeltje met straal \qty{10}{\micro m}
in lucht ($\eta = \qty{1.8e-5}{Pa.s}$), met de \omterm{prop:b1:newton-dynamics:drag}{wet van Stokes}. Controleer met een
getal van Reynolds dat elk regime het juiste is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:7}
Een slinger met lengte $\ell$ wordt uit rust losgelaten bij $\theta_0 = 60^\circ$.
Gegeven dat haar snelheid onderin $\sqrt{2g\ell(1 - \cos\theta_0)}$ is
(energie, volgend hoofdstuk), bepaal de spankracht in het koord onderin in eenheden
van $mg$, en in het loslaatpunt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:8}
Een bocht met straal \qty{200}{m} is verkant onder $15^\circ$. Bij welke snelheid kan
zij zonder enige wrijving worden genomen? Wat moet de wrijving doen bij lagere en
bij hogere snelheden?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:9}
Een kegelslinger: een \omterm{def:b1:newton-dynamics:momentum}{massa} aan een koord met lengte $\ell = \qty{1.0}{m}$
beschrijft een horizontale cirkel, waarbij het koord $30^\circ$ met de
verticaal maakt. Bepaal de hoeksnelheid, de periode en de spankracht (in
eenheden van $mg$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:10}
Een auto van \qty{1000}{kg} haalt vanuit stilstand \qty{100}{km/h} in \qty{8.0}{s}
met constante versnelling. Bepaal de resulterende \omterm{def:b1:newton-dynamics:momentum}{kracht} en de kleinste
wrijvingscoëfficiënt tussen aandrijfbanden en wegdek als de aangedreven
wielen het hele gewicht dragen. Wat begrenst een sportauto op een natte weg?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:11}
Een lichaam valt vanuit rust met lineaire weerstand, $\tau = m/\alpha = \qty{0.50}{s}$.
Schrijf $v(t)$ op en leid de gevallen afstand $z(t)$ af; bereken $v_\infty$
en $z$ na \qty{2.0}{s}; vergelijk met de vrije val.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:12}
Een wagentje met \omterm{def:b1:newton-dynamics:momentum}{massa} $m$ rijdt aan de binnenkant van een verticale cirkelvormige looping met
straal $R$ (\cref{pb:b1:point-kinematics:1}). Druk de \omterm{prop:b1:newton-dynamics:friction}{normaalkracht} van de \omterm{def:b1:point-kinematics:frame}{baan}
uit als functie van de hoek $\theta$ vanaf de onderkant en
van de snelheid $v(\theta)$; leid de kleinste snelheid bovenin af; bepaal met
$v^2 = v_0^2 - 2gR(1 - \cos\theta)$ de kleinste inrijsnelheid en
de \omterm{prop:b1:newton-dynamics:friction}{normaalkracht} onderin in dat geval.
\end{exercise}

\section{Probleem: De parachutist}

\begin{problem}[{Weekendopgave --- van de deur van het vliegtuig tot de
grond: vrije val, de lucht die terugduwt, een koepel die te snel
opengaat, en de landing --- één bewegingswet, twee weerstandswetten}]
\label{pb:b1:newton-dynamics:1}
Een parachutist met uitrusting heeft \omterm{def:b1:newton-dynamics:momentum}{massa} $m = \qty{80}{kg}$. Lucht:
$\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$. Buikwaarts
vallend is $C_xS = \qty{0.80}{m^2}$; onder de open koepel is $C_xS =
\qty{25}{m^2}$. $g = \qty{9.81}{m/s^2}$; $z$ wordt naar beneden geteld vanaf
het sprongpunt.

\medskip
\noindent\textbf{Deel I --- Zonder lucht.}
\begin{enumerate}
  \item Schrijf de tweede wet van Newton op en geef $v(t)$ en $z(t)$; bereken
        beide op $t = \qty{10}{s}$.
  \item Bereken het gewicht.
  \item De parachutist voelt zich in deze fase ``gewichtloos'', hoewel de
        zwaartekracht volledig werkt. Verklaar dat met de \omterm{def:b1:newton-dynamics:momentum}{krachten} (wat ontbreekt er
        vergeleken met staan op de grond).
  \item Is het aardse stelsel inertiaal genoeg voor dit vraagstuk? Noem wat
        er wordt verwaarloosd.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Vallen door lucht.}
\begin{enumerate}[resume]
  \item Schrijf de weerstandskracht voor de buikwaartse val op als functie van $v$.
  \item Schrijf de bewegingsvergelijking voor $v(t)$ op.
  \item Bepaal de \omterm{prop:b1:newton-dynamics:terminal}{eindsnelheid} $v_\infty$ (in \unit{m/s} en
        \unit{km/h}).
  \item Toon met $u = v/v_\infty$ en $\tau = v_\infty/g$ aan dat
        $\dd u/\dd t = (1 - u^2)/\tau$; bereken $\tau$.
  \item Los op door scheiding van variabelen en toon aan dat
        $v = v_\infty\tanh(t/\tau)$.
  \item Na hoeveel tijd is $v = 0.9v_\infty$? En $0.99v_\infty$?
  \item Toon aan dat $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ en bereken de
        gevallen afstand op $t = \qty{10}{s}$; vergelijk met deel I.
  \item Schat het getal van Reynolds bij $v_\infty$ met $L = \qty{0.5}{m}$
        en verantwoord de kwadratische wet.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De koepel gaat open.}
\begin{enumerate}[resume]
  \item Bereken de nieuwe \omterm{prop:b1:newton-dynamics:terminal}{eindsnelheid} onder de open koepel.
  \item Als de koepel ogenblikkelijk zou opengaan bij $v = \qty{40}{m/s}$, hoe
        groot zou de vertraging dan zijn, in \unit{m/s^2} en in $g$? Waarom is
        dat onaanvaardbaar?
  \item Echte koepels gaan in ongeveer \qty{3}{s} open. Schat de gemiddelde
        vertraging en de gemiddelde \omterm{def:b1:newton-dynamics:momentum}{kracht} van het harnas op de
        parachutist.
  \item Zodra de koepel helemaal open is zakt de snelheid naar de nieuwe
        $v_\infty$: wat is de nieuwe \omterm{def:b1:transient-regimes:firstorder}{tijdconstante} $\tau' = v_\infty'/g$,
        en wat betekent zij voor het bereiken van de stationaire daling?
  \item Welke \omterm{def:b1:newton-dynamics:momentum}{kracht} oefent het harnas uit bij een stationaire daling? Verantwoord dat met de
        tweede wet.
  \item Verklaar waarom de \omterm{prop:b1:newton-dynamics:terminal}{eindsnelheid} gaat als $1/\sqrt{C_xS}$: met welke
        factor zou een koepel met de halve oppervlakte haar verhogen?
  \item Een zwaardere springer (\qty{100}{kg}) onder dezelfde koepel: welke
        \omterm{prop:b1:newton-dynamics:terminal}{eindsnelheid}?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De landing.}
\begin{enumerate}[resume]
  \item De landingssnelheid is de $v_\infty$ van de koepel. Van welke hoogte
        zou men zonder lucht moeten springen om met die snelheid de grond te
        raken?
  \item Een landing op gestrekte benen stopt het lichaam over \qty{0.5}{m}; een rol
        over \qty{1.5}{m}. Schat de gemiddelde vertraging (in $g$) en
        de gemiddelde \omterm{def:b1:newton-dynamics:momentum}{kracht} op de benen in beide gevallen, en trek een conclusie.
  \item Vlak voor het neerkomen trekt de parachutist beide stuurlijnen aan,
        wat de weerstand van de koepel kortstondig vergroot. Verklaar kwalitatief
        het effect op de snelheid.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Kleine dingen vallen anders.}
\begin{enumerate}[resume]
  \item Een regendruppel met straal \qty{1.0}{mm} ($C_x = 0.5$): bereken haar
        \omterm{prop:b1:newton-dynamics:terminal}{eindsnelheid} met de kwadratische wet en controleer het getal van
        Reynolds.
  \item Een mistdruppeltje met straal \qty{10}{\micro m}: gebruik de \omterm{prop:b1:newton-dynamics:drag}{wet van Stokes}
        voor $v_\infty$ en ga na dat het getal van Reynolds inderdaad
        klein is.
  \item Vat samen: één bewegingsvergelijking, twee weerstandsregimes en de
        drie getallen (\unit{m/s}) die bepalen of iets valt
        als een steen, als een blad of helemaal niet.
\end{enumerate}
\end{problem}
