\chapter{Impulsmoment}\label{ch:b1:angular-momentum}

Een kunstschaatsster die met gestrekte \omterm{def:b1:angular-momentum:moment}{armen} tolt trekt ze in en wervelt,
zonder dat iemand haar duwt, twee keer zo snel rond. Een deur wijkt voor een
vingertop bij de klink en weerstaat een schouder bij het scharnier. Planeten
bestrijken gelijke oppervlakken in gelijke tijden --- Kepler zag het vier eeuwen
geleden, lang voordat iemand kon zeggen waarom. En de maan, getrokken door de
getijden die zij opwekt, drijft jaarlijks enkele centimeters van de aarde weg terwijl onze
dagen onmerkbaar langer worden. Achter alle vier staat één vectoriële
grootheid, het \omterm{def:b1:angular-momentum:L}{impulsmoment}, en één wet: haar veranderingssnelheid is gelijk aan
het \omterm{def:b1:angular-momentum:moment}{moment} van de krachten. Dit hoofdstuk definieert beide, bewijst de wet voor
een \omterm{def:b1:point-kinematics:frame}{puntmassa} en voor een systeem, en zet haar aan het werk op slingers, banen
en draaiende dingen.

\section{Moment van een kracht}

\begin{definition}[Moment om een punt; om een as]\label{def:b1:angular-momentum:moment}
\index{moment van een kracht}\index{koppel}\index{arm}
Het \emph{moment} (of \emph{koppel}) om een punt $O$ van een kracht
$\vect F$ die in $M$ aangrijpt is de vector
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
loodrecht op het vlak van $\vect{OM}$ en $\vect F$, met norm
$OM\cdot F\sin\alpha = F\,d$, waarin $d$ --- de \emph{arm} --- de
afstand is van $O$ tot de werklijn van $\vect F$. Het moment om een
georiënteerde as $\Delta$ (eenheidsvector $\vect e_\Delta$) door $O$ is de
scalar $\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$, onafhankelijk
van de keuze van $O$ op $\Delta$; het verdwijnt als $\vect F$ evenwijdig is aan
$\Delta$ of die snijdt. Twee tegengestelde krachten op verschillende werklijnen
vormen een \emph{krachtenkoppel}, waarvan het moment $\vect F\wedge\vect{AB}$ om elk
punt hetzelfde is.
\end{definition}

\begin{example}[De deur]\label{ex:b1:angular-momentum:door}
Een duw van \qty{20}{N} loodrecht op een deur, \qty{0.80}{m} van de
scharnieren, heeft \omterm{def:b1:angular-momentum:moment}{moment} $\qty{16}{N.m}$ om de scharnieras; dezelfde duw
op \qty{0.10}{m} geeft \qty{2}{N.m}; langs de deur (werklijn door
de scharnieren) nul. De reactie van het scharnier zelf, die de as snijdt, heeft er geen
\omterm{def:b1:angular-momentum:moment}{moment} om: dat maakt de as de natuurlijke plek om \omterm{def:b1:angular-momentum:moment}{momenten} te nemen.
\end{example}

\section{Impulsmoment van een puntmassa}

\begin{definition}[Impulsmoment]\label{def:b1:angular-momentum:L}
\index{impulsmoment}
Het \emph{impulsmoment} om $O$ van een \omterm{def:b1:point-kinematics:frame}{puntmassa} $m$ in $M$
met snelheid $\vect v$ is
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
en om een as $\Delta$ door $O$: $L_\Delta = \vect L_O\cdot\vect e_\Delta$.
Voor een beweging in een vlak loodrecht op $\Delta$ geldt, in \omterm{def:b1:point-kinematics:cylindrical}{poolcoördinaten}
om de as, $L_\Delta = mr^2\dot\theta$; voor een cirkel met straal $R$ geldt
$L_\Delta = mR^2\omega = mRv$.
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$, $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[Impulsmomentstelling]\label{thm:b1:angular-momentum:theorem}
\index{impulsmomentstelling}
In een inertiaalstelsel geldt, voor een punt $O$ dat in dat stelsel vastligt,
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
voor een vaste as $\Delta$. Verdwijnt het totale \omterm{def:b1:angular-momentum:moment}{moment} om $O$, dan blijft
$\vect L_O$ behouden.
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$, met de tweede wet van Newton
($O$ vast, zodat $\dd\vect{OM}/\dd t = \vect v$). Projecteer op $\vect e_\Delta$.
\end{proof}

\begin{corollary}[Centrale krachten: vlakke beweging en de perkenwet]\label{cor:b1:angular-momentum:central}
\index{centrale kracht}\index{perkenwet}\index{perksnelheid}
Een \omterm{def:b1:point-kinematics:frame}{puntmassa} die alleen aan een \emph{centrale kracht} is onderworpen (altijd gericht
langs $\vect{OM}$, naar het vaste centrum $O$ toe of ervandaan), houdt
$\vect L_O$ constant. Haar beweging blijft in het vlak door $O$
loodrecht op $\vect L_O$, en in dat vlak is $r^2\dot\theta = C$
(constant): de voerstraal $\vect{OM}$ bestrijkt oppervlak met de
constante \emph{perksnelheid}
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$ voor $\vect F \parallel \vect{OM}$: $\vect L_O$
is constant. Steeds geldt $\vect{OM}\perp\vect L_O$, vandaar het vlak; $L_O =
mr^2\dot\theta$ is constant. Het oppervlak bestreken tijdens $\dd t$ is de
driehoek met zijden $\vect{OM}$ en $\vect v\dd t$: $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$.
\end{proof}

\begin{example}[De slinger met momenten]\label{ex:b1:angular-momentum:pendulum}
\omterm{ex:b1:newton-dynamics:pendulum}{Wiskundige slinger}, as $\Delta$ door het draaipunt loodrecht op het
zwaaivlak: $L_\Delta = m\ell^2\dot\theta$; de spankracht snijdt de
as (geen \omterm{def:b1:angular-momentum:moment}{moment}), het \omterm{def:b1:angular-momentum:moment}{moment} van het gewicht is $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{arm}
$\ell\sin\theta$, terugdrijvend). De stelling geeft $m\ell^2\ddot\theta =
-mg\ell\sin\theta$, de vergelijking van \cref{ex:b1:newton-dynamics:pendulum}
zonder ooit de spankracht op te schrijven --- het grote voordeel van \omterm{def:b1:angular-momentum:moment}{momenten}:
krachten door de as verdwijnen.
\end{example}

\begin{example}[Een puck naar binnen trekken]\label{ex:b1:angular-momentum:puck}
Een puck glijdt wrijvingsloos over ijs in een cirkel, gehouden door een koord door
een gaatje in het midden; aan het koord wordt getrokken tot de straal halveert.
De spankracht is centraal: $L = mr^2\omega$ blijft behouden, dus wordt $\omega$
met $4$ vermenigvuldigd, de snelheid $r\omega$ met $2$ en de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} met $4$
--- de extra energie is de arbeid van het trekken, waar de puck de hele weg naar
binnen tegen in werkt. De \omterm{def:b1:angular-momentum:moment}{armen} van de schaatsster zijn dezelfde natuurkunde met de massa
verdeeld (\cref{exo:b1:angular-momentum:4}).
\end{example}

\section{Systemen van punten en rotatie om een vaste as}

\begin{theorem}[Impulsmoment van een systeem]\label{thm:b1:angular-momentum:system}
\index{impulsmoment!van een systeem}
Voor een systeem van \omterm{def:b1:point-kinematics:frame}{puntmassa}'s geldt, met $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
en $O$ vast in een inertiaalstelsel,
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
alleen de \omterm{def:b1:angular-momentum:moment}{momenten} van de \emph{uitwendige} krachten tellen mee; de inwendige krachten,
die de wet van \omterm{thm:b1:newton-dynamics:laws}{actie en reactie} langs de verbindingslijn van de
\omterm{def:b1:point-kinematics:frame}{puntmassa}'s volgen, dragen niets bij.
\end{theorem}

\begin{proof}
Tel de stellingen voor de \omterm{def:b1:point-kinematics:frame}{puntmassa}'s op. Voor een inwendig paar is $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$, want de kracht ligt langs
$\vect{M_jM_i}$.
\end{proof}

\begin{proposition}[Starre rotatie om een vaste as]\label{prop:b1:angular-momentum:inertia}
\index{traagheidsmoment}
Houden de \omterm{def:b1:point-kinematics:frame}{puntmassa}'s vaste afstanden $r_i$ tot een as $\Delta$ en draaien zij
er allemaal met dezelfde hoeksnelheid $\omega$ omheen (een starre rotatie), dan is
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
waarin $J_\Delta$ het \emph{\omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment}} om $\Delta$ is, en luidt de
stelling $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$.
\end{proposition}

\begin{proof}
Elke \omterm{def:b1:point-kinematics:frame}{puntmassa} draagt $m_ir_i^2\omega$ bij aan $L_\Delta$ en
$\tfrac12 m_i(r_i\omega)^2$ aan $E_k$. \cref{ch:b1:systems-of-points}
breidt dit uit tot continue vaste lichamen.
\end{proof}

\begin{example}[De schaatsster]\label{ex:b1:angular-momentum:skater}
Met de \omterm{def:b1:angular-momentum:moment}{armen} uit heeft het lichaam van een schaatsster $J \approx \qty{4}{kg.m^2}$ plus twee
\omterm{def:b1:angular-momentum:moment}{armen} van \qty{3}{kg} op \qty{0.8}{m}: $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$. \omterm{def:b1:angular-momentum:moment}{Armen} in (\qty{0.2}{m}): $J_2 = \qty{4.2}{kg.m^2}$.
De reactie van het ijs en het gewicht hebben geen \omterm{def:b1:angular-momentum:moment}{moment} om de verticale
as: $J\omega$ blijft behouden en het toerental stijgt met $J_1/J_2 = 1.85$
--- van $2$ naar $3.7$ omwentelingen per seconde --- terwijl de \omterm{thm:b1:work-and-energy:ke}{kinetische energie}
met dezelfde factor stijgt, betaald door de spieren die de \omterm{def:b1:angular-momentum:moment}{armen} naar binnen trekken.
\end{example}

\begin{remark}[Statica]\label{rem:b1:angular-momentum:statics}
\index{evenwicht van een vast lichaam}
Een vast lichaam in rust heeft $\vect L_O = \vect 0$ constant: de uitwendige krachten
moeten resultante nul \emph{en} totaal \omterm{def:b1:angular-momentum:moment}{moment} nul om elk punt hebben.
De tweede voorwaarde is de hefboomwet van de balans, de wip en
de koevoet: een kleine kracht ver van het draaipunt houdt een grote kracht dichtbij
in evenwicht.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:1}
Een kracht van \qty{50}{N} op een sleutel \qty{0.30}{m} van de moer: \omterm{def:b1:angular-momentum:moment}{moment} wanneer
de kracht loodrecht op de steel staat; onder $60^\circ$ ermee; en erlangs.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:2}
\omterm{def:b1:angular-momentum:L}{Impulsmoment} van de aarde om de zon (cirkelvormige \omterm{def:b1:point-kinematics:frame}{baan},
$r = \qty{1.50e11}{m}$, $v = \qty{29.8}{km/s}$, $m = \qty{5.97e24}{kg}$),
en haar perksnelheid.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:3}
Een kegelslinger (koord \qty{1.0}{m}, $30^\circ$ met de verticaal,
\cref{exo:b1:newton-dynamics:9}). Bereken $L$ om de verticale as
door het draaipunt per eenheid massa, en leg uit waarom het behouden blijft
hoewel de spankracht van het koord en het gewicht werken.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:angular-momentum:4}
De romp van een schaatsster heeft $J = \qty{4.0}{kg.m^2}$ om de tolas; elke
\omterm{def:b1:angular-momentum:moment}{arm} (\qty{3.0}{kg}) wordt behandeld als een punt op \qty{0.80}{m}, en daarna op
\qty{0.20}{m}. Toerental na het intrekken van de \omterm{def:b1:angular-momentum:moment}{armen}, uitgaand van
\qty{2.0}{} omwentelingen per seconde; verhouding van de kinetische energieën; waar komt de
extra energie vandaan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:5}
Leid de slingervergelijking $\ell\ddot\theta + g\sin\theta = 0$ af uit de
impulsmomentstelling om de as door het draaipunt, en geef het \omterm{def:b1:angular-momentum:moment}{moment} van
elke kracht.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:6}
De afstand van de aarde tot de zon varieert van \qty{1.471e11}{m} (perihelium)
tot \qty{1.521e11}{m} (aphelium). Bepaal met de \omterm{cor:b1:angular-momentum:central}{perkenwet} de verhouding
van haar snelheden in die punten; geef beide snelheden bij een gemiddelde snelheid van \qty{29.8}{km/s}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:7}
Een puck van \qty{0.20}{kg} cirkelt met \qty{2.0}{m/s} aan een koord van \qty{0.50}{m}
door een gaatje; het koord wordt aangetrokken tot de straal \qty{0.25}{m} is.
Nieuwe snelheid en hoeksnelheid; kinetische energieën vóór en na; verrichte arbeid
door het trekken. Waarom kan de spankracht hier arbeid verrichten?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:8}
Een homogene plank van \qty{20}{kg} en \qty{4.0}{m} lang rust op een
draaipunt \qty{1.5}{m} van haar linkeruiteinde; een kind van \qty{30}{kg} zit op
het linkeruiteinde. Waar moet een kind van \qty{45}{kg} zitten voor evenwicht? Wat draagt
het draaipunt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:9}
Een machine van Atwood ($m_1 = \qty{2.0}{kg}$, $m_2 = \qty{1.0}{kg}$) hangt over
een katrol met straal \qty{0.10}{m} en \omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment} \qty{0.010}{kg.m^2}
waarop het koord niet slipt. Bepaal met de impulsmomentstelling
voor de katrol en de wet van Newton voor de massa's de versnelling
en de twee spankrachten. Vergelijk met de massaloze katrol.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:10}
Een komeet passeert haar perihelium op \qty{0.50}{AU} van de zon met
\qty{40}{km/s}. Wat is, wanneer zij \qty{10}{AU} ver is, de component van
haar snelheid loodrecht op de voerstraal? Kan de \omterm{cor:b1:angular-momentum:central}{perkenwet}
haar volledige snelheid geven?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:11}
Een ster als de zon ($R = \qty{7.0e8}{m}$, omwentelingstijd \qty{25}{d})
stort in tot een neutronenster met straal \qty{10}{km} en behoudt haar massa
en \omterm{def:b1:angular-momentum:L}{impulsmoment} ($J \propto MR^2$). Nieuwe periode? Snelheid aan de evenaar?
Wat zegt dat laatste over de aanname?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:12}
Een kraal glijdt zonder wrijving langs een rechte staaf die in een
horizontaal vlak met constante $\omega$ om een verticale as door één
uiteinde draait. Schrijf $L_z$ van de kraal op; toon aan dat het \emph{niet} behouden blijft
en bereken het \omterm{def:b1:angular-momentum:moment}{moment} dat de staaf moet uitoefenen; leid daaruit de dwarse reactie
van de staaf op de kraal af.
\end{exercise}

\section{Probleem: Waarom de maan wegdrijft}

\begin{problem}[{Weekendopgave --- de getijden die de maan op de
aarde opwekt trekken haar vooruit en de aarde terug: hoe impulsmoment
uit onze dagen naar de baan van de maan stroomt, centimeter voor centimeter, en
waar de energie blijft}]
\label{pb:b1:angular-momentum:1}
Gegevens: $G = \qty{6.67e-11}{SI}$; aarde $M = \qty{5.97e24}{kg}$, $R =
\qty{6.37e6}{m}$, draaiing $\Omega = 2\pi/(\qty{86164}{s})$, \omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment}
$J = 0.33\,MR^2$; maan $m = \qty{7.35e22}{kg}$, \omterm{def:b1:point-kinematics:frame}{baan} als
cirkelvormig behandeld, $r = \qty{3.84e8}{m}$; laserafstandsmeting naar de maan geeft
$\dd r/\dd t = \qty{3.8}{cm/yr}$; de dag wordt \qty{2.3}{ms} per
eeuw langer; $1$ eeuw $= \qty{3.16e9}{s}$.

\medskip
\noindent\textbf{Deel I --- De \omterm{def:b1:angular-momentum:L}{impulsmomenten}.}
\begin{enumerate}
  \item Bereken de hoeksnelheid $\omega$ van de maan in haar \omterm{def:b1:point-kinematics:frame}{baan} uit
        de wet van Newton (cirkelvormige \omterm{def:b1:point-kinematics:frame}{baan}), en controleer haar aan de
        siderische maand van \qty{27.3}{d}.
  \item Bereken het baanimpulsmoment van de maan $L_{\text{baan}} =
        mr^2\omega$.
  \item Toon aan dat voor een cirkelvormige \omterm{def:b1:point-kinematics:frame}{baan} $L_{\text{baan}} = m\sqrt{GMr}$ geldt, en
        leid $\dd L_{\text{baan}}/\dd r = L_{\text{baan}}/2r$ af.
  \item Bereken het draaiimpulsmoment van de aarde $L_{\text{spin}} =
        J\Omega$ en vergelijk het met $L_{\text{baan}}$.
  \item De maan draait één keer per maand om haar as: schat haar
        draaiimpulsmoment ($J_m \approx 0.4\,mR_m^2$, $R_m = \qty{1.74e6}{m}$) en
        verantwoord dat het verwaarloosd wordt.
  \item Welke uitwendige \omterm{def:b1:angular-momentum:moment}{momenten} werken er op het stelsel aarde--maan om zijn
        massamiddelpunt? Beargumenteer dat het totaal
        $L = L_{\text{spin}} + L_{\text{baan}}$ bij goede benadering behouden blijft.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het getijdenkoppel.}
De maan wekt op de aarde twee vloedbergen op; omdat de aarde
sneller draait dan de maan haar \omterm{def:b1:point-kinematics:frame}{baan} doorloopt, sleept de wrijving de vloedbergen over een kleine hoek
vooruit ten opzichte van de lijn aarde--maan.
\begin{enumerate}[resume]
  \item Schets de aarde, de voorste vloedberg en de maan. In welke
        richting werkt de aantrekking van de maan op de nabije vloedberg,
        ten opzichte van de draaiing van de aarde?
  \item Leid het teken af van het \omterm{def:b1:angular-momentum:moment}{moment} op de draaiing van de aarde: wordt de
        dag langer of korter?
  \item Welk \omterm{def:b1:angular-momentum:moment}{moment} werkt door \omterm{thm:b1:newton-dynamics:laws}{actie en reactie} op de baanbeweging van
        de maan? Groeit of krimpt $L_{\text{baan}}$?
  \item Groeit of krimpt $r$ volgens vraag 3? En de baansnelheid
        $v = \sqrt{GM/r}$ van de maan?
  \item Los de schijnbare tegenspraak op: de maan wordt vooruit geduwd en
        beweegt toch trager.
  \item Naar welke \omterm{def:b1:transient-regimes:firstorder}{eindtoestand} streeft deze uitwisseling?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De getallen.}
\begin{enumerate}[resume]
  \item Bereken uit het langer worden van de dag de relatieve verandering
        $\Delta\Omega/\Omega$ per eeuw.
  \item Leid de verandering van $L_{\text{spin}}$ per eeuw af.
  \item Leid de verandering van $L_{\text{baan}}$ per eeuw af.
  \item Reken haar met vraag 3 om in een verandering van $r$ per eeuw
        en per jaar. Vergelijk met de waarde uit de laserafstandsmeting.
  \item Met welke fractie, en met hoeveel seconden, wordt de siderische
        maand per eeuw langer ($T \propto r^{3/2}$)?
  \item Bereken de verandering van de rotatie-energie van de aarde
        per eeuw ($\Delta E = L_{\text{spin}}\Delta\Omega$ tot op eerste
        orde).
  \item Bereken de verandering van de mechanische baanenergie van de maan,
        $E_{\text{baan}} = -GMm/2r$, per eeuw.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De energie.}
\begin{enumerate}[resume]
  \item Toon aan dat de aarde meer energie verliest dan de maan wint, en
        bereken het verschil per eeuw en het bijbehorende vermogen.
        Waar blijft het?
  \item Vergelijk dat vermogen met het verbruik van de mensheid, ongeveer
        \qty{18}{TW}, en met de inwendige warmtestroom van de aarde, ongeveer
        \qty{47}{TW}.
  \item In de \omterm{def:b1:transient-regimes:firstorder}{eindtoestand} van vraag 12 zijn dag en maand gelijk:
        $\Omega = \omega = \sqrt{GM/r^3}$. Ga met het behoud van
        $L_{\text{spin}} + L_{\text{baan}}$ na dat $r \approx
        \qty{5.5e8}{m}$ daaraan voldoet, en geef de gemeenschappelijke periode in dagen.
  \item Waarom zal die toestand in werkelijkheid nooit worden bereikt? (Denk aan de
        getijden van de zon.)
  \item De maan keert de aarde al één zijde toe. Verklaar dat met hetzelfde
        mechanisme, en zeg waarom het de maan als eerste overkwam.
  \item Vat in drie regels samen: wat behouden blijft, wat wordt uitgewisseld,
        wat wordt gedissipeerd.
\end{enumerate}
\end{problem}
