\chapter{Kinetische theorie en het ideale gas}\label{ch:b1:kinetic-theory}

De lucht in een klaslokaal weegt evenveel als een volwassen man en bevat meer
moleculen dan er zandkorrels op aarde zijn, elk vliegend met de snelheid van
een geweerkogel en tien miljard keer per seconde botsend met zijn buren.
Niemand kan er één van volgen; niemand hoeft dat. Een handvol gemiddelden ---
\omterm{def:b1:kinetic-theory:pressure}{druk}, \omterm{def:b1:kinetic-theory:temperature}{temperatuur}, volume, hoeveelheid --- beschrijft de lucht volledig voor
elk technisch doel, en een eenvoudig beeld van moleculen die van wanden
terugkaatsen verklaart waar die gemiddelden vandaan komen en hoe zij
samenhangen. Dit hoofdstuk opent het thermodynamische deel van dit volume door
de materie van twee kanten te bekijken: de microscopische chaos en de
macroscopische rust, en de kinetische theorie die beide verbindt in het geval
van het ideale gas.

\section{Twee beschrijvingsschalen}

\begin{definition}[Microscopische en macroscopische schaal]\label{def:b1:kinetic-theory:scales}
\index{microscopische schaal}\index{macroscopische schaal}\index{constante van Avogadro}\index{mol}
Op de \emph{microscopische} schaal bestaat materie uit moleculen: $\qty{1}{mol}$
bevat er $N_A = \qty{6.02e23}{}$, elk ongeveer \qty{0.3}{nm} groot;
in een gas onder gewone omstandigheden liggen zij \qty{3}{nm} uit elkaar, bewegen zij met
honderden meters per seconde en botsen zij ongeveer elke \qty{70}{nm}.
Op de \emph{macroscopische} schaal wordt een systeem beschreven door enkele
\emph{toestandsgrootheden} --- \omterm{def:b1:kinetic-theory:pressure}{druk} $P$, volume $V$, \omterm{def:b1:kinetic-theory:temperature}{temperatuur} $T$,
hoeveelheid stof $n$ --- die gemiddelden zijn over enorme aantallen
moleculen. Een grootheid is \emph{extensief} als zij verdubbelt wanneer het systeem
verdubbelt ($V$, $n$, massa, energie), en \emph{intensief} als zij dat niet doet
($P$, $T$, dichtheid).
\end{definition}

\begin{definition}[Thermodynamisch evenwicht]\label{def:b1:kinetic-theory:equilibrium}
\index{thermodynamisch evenwicht}\index{toestandsgrootheid}
Een systeem is in \emph{thermodynamisch evenwicht} wanneer zijn toestandsgrootheden
uniform en constant in de tijd zijn en er geen macroscopische stroom van materie of
energie doorheen gaat. Alleen dan zijn $P$ en $T$ voor het systeem als
geheel gedefinieerd; de \emph{toestandsvergelijking} verbindt ze met $V$ en $n$.
\end{definition}

\begin{definition}[Druk]\label{def:b1:kinetic-theory:pressure}
\index{druk}\index{pascal}\index{bar}
Een vloeistof of gas in rust duwt op elk oppervlakte-element $\dd S$ van een wand met een
kracht $\dd\vect F = P\,\dd S\,\vect n$ loodrecht daarop, naar de wand toe; $P$ is
de \emph{druk}, in pascal ($\qty{1}{Pa} = \qty{1}{N/m^2}$); $\qty{1}{bar}
= \qty{1e5}{Pa}$, $\qty{1}{atm} = \qty{1.013e5}{Pa}$. In een gegeven punt hangt zij
niet van de oriëntatie van het oppervlak af (\cref{ch:b1:fluid-statics}).
\end{definition}

\begin{definition}[Temperatuur]\label{def:b1:kinetic-theory:temperature}
\index{temperatuur}\index{kelvin}\index{nulde hoofdwet}
Twee systemen die in contact staan door een wand die energie doorlaat bereiken na
verloop van tijd een gemeenschappelijke toestand: zij zijn dan op dezelfde \emph{temperatuur}, en
twee lichamen die elk met een derde in evenwicht zijn, zijn dat ook met
elkaar (de \emph{nulde hoofdwet}). De kelvinschaal ligt vast door
$k_B = \qty{1.380649e-23}{J/K}$ exact; Celsius: $\theta = T - 273.15$. De
kinetische theorie hieronder geeft $T$ zijn microscopische betekenis.
\end{definition}

\section{Het kinetische model van het ideale gas}

\begin{definition}[Ideaal gas, microscopisch model]\label{def:b1:kinetic-theory:model}
\index{ideaal gas}\index{perfect gas}
Een \emph{ideaal gas} (perfect gas) is een verzameling van $N$ moleculen,
opgevat als punten met massa $m$, die vrij en willekeurig bewegen --- geen
wisselwerking behalve korte botsingen --- in een vat met volume $V$.
In evenwicht is de snelheidsverdeling isotroop en stationair;
$n^* = N/V$ is de deeltjesdichtheid en $\langle v^2\rangle$ de gemiddelde
kwadratische snelheid.
\end{definition}

\begin{theorem}[Kinetische druk]\label{thm:b1:kinetic-theory:pressure}
\index{kinetische druk}
De \omterm{def:b1:kinetic-theory:pressure}{druk} die het gas op de wanden uitoefent is
\[
P = \tfrac13\,n^*m\langle v^2\rangle = \tfrac23\,n^*\langle\epsilon_k\rangle ,
\]
waarin $\langle\epsilon_k\rangle = \tfrac12 m\langle v^2\rangle$ de gemiddelde
\omterm{thm:b1:work-and-energy:ke}{kinetische energie} van een molecuul is.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Neem het vereenvoudigde model waarin de moleculen alle de snelheid $u$ hebben en
langs de drie assen bewegen, een zesde in elke richting. Een molecuul
dat de wand $x = 0$ recht raakt kaatst elastisch terug en draagt er
de impuls $2mu$ aan over. In de tijd $\dd t$ zijn de moleculen die naar de
wand toe bewegen en een oppervlak $S$ bereiken die welke in de laag $u\,\dd t$
dik liggen, een zesde van de $n^*Su\,\dd t$ moleculen daarin. Kracht $= $
impuls per tijd $= (n^*Su\,\dd t/6)(2mu)/\dd t = \tfrac13 n^*mu^2S$, dus
$P = \tfrac13 n^*mu^2$. Middelen over de werkelijke verdeling van snelheden
en richtingen vervangt $u^2$ door $\langle v^2\rangle$ (de isotropie geeft
$\langle v_x^2\rangle = \tfrac13\langle v^2\rangle$); het volume van jaar 2 doet
dat volledig.
\end{proof}

\begin{definition}[Kinetische temperatuur; toestandsvergelijking]\label{def:b1:kinetic-theory:kinetictemp}
\index{kinetische temperatuur}\index{toestandsvergelijking!ideaal gas}\index{gasconstante}\index{kwadratisch gemiddelde snelheid}
De \omterm{def:b1:kinetic-theory:temperature}{temperatuur} van een \omterm{def:b1:kinetic-theory:model}{ideaal gas} wordt \emph{gedefinieerd} door de gemiddelde
\omterm{thm:b1:work-and-energy:ke}{kinetische energie} van zijn moleculen,
\[
\langle\epsilon_k\rangle = \tfrac12 m\langle v^2\rangle = \tfrac32 k_BT ,
\]
zodat, gecombineerd met \cref{thm:b1:kinetic-theory:pressure},
\[
PV = Nk_BT = nRT, \qquad R = N_Ak_B = \qty{8.314}{J/(mol.K)} ,
\]
de \emph{toestandsvergelijking van het ideale gas}. De \emph{kwadratisch
gemiddelde snelheid} is $u = \sqrt{\langle v^2\rangle} = \sqrt{3k_BT/m} = \sqrt{3RT/M}$,
met $M$ de molaire massa.
\end{definition}

\begin{example}[Getallen voor lucht]\label{ex:b1:kinetic-theory:air}
Stikstof bij \qty{300}{K}: $u = \sqrt{3 \times 8.314 \times 300/0.028} =
\qty{517}{m/s}$; waterstof, $\qty{1930}{m/s}$; hoe zwaarder, hoe trager, als
$1/\sqrt M$. Bij \qty{1}{\omterm{def:b1:kinetic-theory:pressure}{bar}} en \qty{300}{K} is $n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$:
\qty{24}{L} per \omterm{def:b1:kinetic-theory:scales}{mol}, \qty{2.7e22}{} moleculen in een liter. Een klaslokaal van
\qty{200}{m^3}: \qty{8300}{\omterm{def:b1:kinetic-theory:scales}{mol}}, \qty{240}{kg} lucht.
\end{example}

\begin{remark}[Waarom het model werkt, en wanneer het faalt]\label{rem:b1:kinetic-theory:validity}
\index{gemiddelde vrije weglengte}
Tussen twee botsingen legt een luchtmolecuul een \emph{\omterm{rem:b1:kinetic-theory:validity}{gemiddelde vrije weglengte}}
$\lambda \approx 1/(\sqrt2\pi d^2n^*) \approx \qty{70}{nm}$ af --- tweehonderd
keer zijn eigen grootte: het gas is grotendeels leeg, wisselwerkingen zijn zeldzaam, en
$PV = nRT$ klopt bij gewone drukken tot op minder dan een procent. Bij hoge
\omterm{def:b1:kinetic-theory:pressure}{druk} of lage \omterm{def:b1:kinetic-theory:temperature}{temperatuur} tellen het eigen volume van de moleculen en hun
aantrekking wel mee: de vergelijking van Van der Waals $(P + an^2/V^2)(V - nb) = nRT$
corrigeert voor beide, en onder een kritische \omterm{def:b1:kinetic-theory:temperature}{temperatuur} wordt het gas vloeibaar
(\cref{ch:b1:phase-changes}).
\end{remark}

\section{Inwendige energie van het ideale gas}

\begin{definition}[Inwendige energie]\label{def:b1:kinetic-theory:U}
\index{inwendige energie}
De \emph{inwendige energie} $U$ van een systeem is de som van de kinetische
energieën van zijn moleculen in het stelsel waarin het systeem in rust is en
van hun onderlinge potentiële energieën; zij is een extensieve toestandsfunctie.
\end{definition}

\begin{proposition}[Inwendige energie van ideale gassen]\label{prop:b1:kinetic-theory:Ugas}
\index{eenatomig gas}\index{tweeatomig gas}\index{warmtecapaciteit bij constant volume}\index{wetten van Joule}
Voor een \omterm{def:b1:kinetic-theory:model}{ideaal gas} hangt $U$ alleen van $T$ af (de eerste wet van Joule):
\begin{gather*}
U = \tfrac32 nRT \quad (\text{eenatomig: He, Ne, Ar}), \\
U = \tfrac52 nRT \quad (\text{tweeatomig bij gewone temperaturen: N}_2, \text{O}_2, \text{H}_2),
\end{gather*}
zodat de molaire \omterm{prop:b1:kinetic-theory:Ugas}{warmtecapaciteit bij constant volume} $C_{V,m} = \dd U_m/\dd T$
respectievelijk $\tfrac32 R = \qty{12.5}{J/(mol.K)}$ en $\tfrac52 R = \qty{20.8}{J/(mol.K)}$
is.
\end{proposition}

\begin{proof}[Gedeeltelijk bewijs]
Per aanname geen onderlinge \omterm{def:b1:work-and-energy:conservative}{potentiële energie}, dus $U = N\langle\epsilon_k\rangle
= \tfrac32 Nk_BT$ voor puntmoleculen. Een tweeatomig molecuul draait bovendien:
elk van zijn twee rotatievrijheidsgraden draagt in evenwicht
dezelfde $\tfrac12 k_BT$ als elke translatievrijheidsgraad (de \emph{equipartitiestelling},
aangenomen; het volume van jaar 3 leidt haar af en legt uit waarom de
trilling pas boven duizend \omterm{def:b1:kinetic-theory:temperature}{kelvin} meedoet).
\end{proof}

\begin{example}[Wat een lokaal bevat]\label{ex:b1:kinetic-theory:room}
De \qty{8300}{\omterm{def:b1:kinetic-theory:scales}{mol}} lucht in het klaslokaal bij \qty{300}{K}: $U = \tfrac52 \times
8300 \times 8.314 \times 300 = \qty{5.2e7}{J}$ --- de energie van anderhalve
liter benzine, opgeslagen in moleculaire beweging. Het lokaal $1$ K warmer maken
kost $\tfrac52 nR = \qty{170}{kJ}$ (bij constant volume).
\end{example}

\section{Gecondenseerde fasen}

\begin{proposition}[Model van de onsamendrukbare, niet-uitzettende fase]\label{prop:b1:kinetic-theory:condensed}
\index{gecondenseerde fase}\index{thermische uitzetting}\index{samendrukbaarheid}
Vloeistoffen en vaste stoffen worden op dit niveau gemodelleerd als fasen met vast
volume --- niet samendrukbaar en niet uitzettend --- waarvan de \omterm{def:b1:kinetic-theory:U}{inwendige energie}
alleen van $T$ afhangt: $\dd U = C\,\dd T$ met $C = mc$, waarin $c$ de soortelijke
warmte is (\qty{4180}{J/(kg.K)} voor water, \qty{450}{J/(kg.K)} voor
staal). De werkelijke, kleine afwijkingen worden gemeten door de thermische
uitzettingscoëfficiënt $\alpha = (1/V)(\partial V/\partial T)_P$ ($\sim\qty{1e-5}{K^{-1}}$
voor metalen, \qty{2e-4}{K^{-1}} voor vloeistoffen) en de isotherme
samendrukbaarheid $\chi_T = -(1/V)(\partial V/\partial P)_T$ ($\sim\qty{5e-10}{Pa^{-1}}$
voor water, tegen $1/P \approx \qty{1e-5}{Pa^{-1}}$ voor een gas).
\end{proposition}
\admitted

\begin{example}[Hoe onsamendrukbaar]\label{ex:b1:kinetic-theory:water}
Water op de bodem van de Marianentrog (\qty{1100}{\omterm{def:b1:kinetic-theory:pressure}{bar}}) wordt samengedrukt
met $\chi_T\Delta P \approx 5\times10^{-10} \times 1.1\times10^8 = 5\%$; een
stalen rail van \qty{30}{m} die $50$ K warmer wordt, wordt $\alpha L\Delta T =
\qty{18}{mm}$ langer --- vandaar de uitzettingsvoegen. Tegenover een gas, waarvan
het volume bij een verdubbelde \omterm{def:b1:kinetic-theory:pressure}{druk} halveert, is het model van een vast volume
in de meeste situaties van dit volume goed tot op enkele procenten.
\end{example}

\section{Probleem: De lucht in het lokaal}

\begin{problem}[{Weekendopgave --- een klaslokaal vol lucht: haar moleculen
tellen, haar wegen, hun vluchten klokken, en uit hetzelfde model de
draagkracht van een ballon, de ijlheid van berglucht en de geluidssnelheid
terugvinden}]
\label{pb:b1:kinetic-theory:1}
Een klaslokaal meet $10 \times 8 \times \qty{2.5}{m}$; zijn lucht is op
$P = \qty{1.013e5}{Pa}$, $T = \qty{293}{K}$, molaire massa $M = \qty{29.0}{g/mol}$,
$79\%$ N$_2$ en $21\%$ O$_2$ naar moleculen. $R = \qty{8.314}{J/(mol.K)}$,
$k_B = \qty{1.38e-23}{J/K}$, $N_A = \num{6.02e23}$, $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Deel I --- Tellen en wegen.}
\begin{enumerate}
  \item Hoeveelheid lucht (in \omterm{def:b1:kinetic-theory:scales}{mol}) en aantal moleculen in het lokaal.
  \item Massa van de lucht; vergelijk met een mens.
  \item Deeltjesdichtheid $n^*$ en gemiddelde afstand tussen moleculen
        ($n^{*-1/3}$); vergelijk met de moleculaire diameter \qty{0.37}{nm}.
  \item \omterm{def:b1:kinetic-theory:kinetictemp}{Kwadratisch gemiddelde snelheid} van N$_2$ en van O$_2$; waarom beweegt het
        zwaardere gas bij dezelfde \omterm{def:b1:kinetic-theory:temperature}{temperatuur} trager?
  \item Gemiddelde \omterm{thm:b1:work-and-energy:ke}{kinetische energie} van één molecuul; totale translatie-energie
        van de lucht in het lokaal; vergelijk met de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} van een
        auto met \qty{100}{km/h}.
  \item \omterm{rem:b1:kinetic-theory:validity}{Gemiddelde vrije weglengte} $\lambda \approx 1/(\sqrt2\pi d^2n^*)$ en de gemiddelde
        tijd tussen twee botsingen van een molecuul.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- \omterm{def:b1:kinetic-theory:pressure}{Druk} uit botsingen.}
\begin{enumerate}[resume]
  \item Bereken met het model met zes richtingen en $u = \qty{500}{m/s}$
        het aantal botsingen per seconde op \qty{1}{cm^2} wand.
  \item Bereken de per seconde afgegeven impuls en ga na dat die
        de atmosferische \omterm{def:b1:kinetic-theory:pressure}{druk} oplevert.
  \item Totale kracht van de lucht op één wand van \qty{8}{m} bij \qty{2.5}{m};
        waarom beweegt de wand niet?
  \item Het lokaal wordt tot \qty{303}{K} verwarmd met de ramen dicht en
        afgedicht: nieuwe \omterm{def:b1:kinetic-theory:pressure}{druk}, en de resulterende kracht op de wand nu.
  \item Dezelfde verwarming met een raam open: wat verdwijnt er, en hoeveel?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Opwaartse kracht.}
\begin{enumerate}[resume]
  \item Dichtheid van de lucht in het lokaal, en van lucht bij \qty{373}{K} bij
        dezelfde \omterm{def:b1:kinetic-theory:pressure}{druk}.
  \item Een heteluchtballon met volume \qty{2800}{m^3} wordt gevuld met lucht van
        \qty{373}{K}: gewicht van de hete lucht, gewicht van de verdrongen koude
        lucht, en de beschikbare draagkracht (Archimedes,
        \cref{ch:b1:fluid-statics}).
  \item Omhulsel, mand, brander en gas wegen samen \qty{300}{kg}: hoeveel
        passagiers van \qty{75}{kg} kan hij dragen?
  \item Tot welke \omterm{def:b1:kinetic-theory:temperature}{temperatuur} zou de lucht moeten worden verhit om de draagkracht
        te verdubbelen, en waarom gebeurt dat niet (het omhulsel is van nylon)?
  \item Een heliumballon met hetzelfde volume bij \qty{293}{K}
        ($M = \qty{4}{g/mol}$): draagkracht? Waarom gebruiken toeristenballonnen
        toch hete lucht?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- IJle lucht en de geluidssnelheid.}
\begin{enumerate}[resume]
  \item Partiële zuurstofdruk in het lokaal.
  \item Op de top van de Everest is $P = \qty{0.33}{atm}$ en $T \approx \qty{250}{K}$:
        deeltjesdichtheid van de zuurstofmoleculen vergeleken met die in het lokaal.
  \item De \omterm{def:b1:kinetic-theory:pressure}{druk} van de atmosfeer daalt met de hoogte ruwweg als
        $P(z) = P_0\eu^{-z/H}$ met $H = RT/Mg$ (\cref{ch:b1:fluid-statics}):
        bereken $H$ voor \qty{260}{K} en de hoogte waarop $P = P_0/2$.
  \item De geluidssnelheid in een \omterm{def:b1:kinetic-theory:model}{ideaal gas} is $c_s = \sqrt{\gamma RT/M}$
        met $\gamma = 1.4$ voor lucht (\cref{ch:b1:first-law}): bereken haar bij
        \qty{293}{K} en vergelijk met de \omterm{def:b1:kinetic-theory:kinetictemp}{kwadratisch gemiddelde snelheid}; duid
        hoe dicht die bij elkaar liggen.
  \item Hoe hangt $c_s$ van $T$ af? Geluidssnelheid in het lokaal bij
        \qty{303}{K}, en in helium bij \qty{293}{K} ($\gamma = 5/3$): waarom
        klinkt een heliumstem hoog?
  \item Is het geluid op de top van de Everest sneller of trager dan in het
        lokaal? Hoeveel?
  \item De lucht in het lokaal wordt vervangen door argon bij dezelfde $P$ en $T$:
        wat verandert er aan het aantal, de massa, de snelheden en de \omterm{def:b1:kinetic-theory:U}{inwendige
        energie}?
  \item \omterm{def:b1:kinetic-theory:U}{Inwendige energie} van de lucht in het lokaal ($U = \tfrac52 nRT$) en de
        energie om haar bij constant volume \qty{10}{K} op te warmen.
  \item Vat in drie regels samen wat het kinetische model heeft opgeleverd: welke
        macroscopische grootheden het verklaarde, en uit welk enkele
        microscopische gemiddelde.
\end{enumerate}
\end{problem}
