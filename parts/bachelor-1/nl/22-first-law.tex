\chapter{De eerste hoofdwet van de thermodynamica}\label{ch:b1:first-law}

Pomp een fietsband op en de cilinder van de pomp wordt te heet om vast te houden.
Een dieselmotor heeft geen bougie: hij perst zijn lucht twintig keer
kleiner, en de lucht wordt heet genoeg om de brandstof vanzelf te ontsteken. Een
fles samengeperst gas beslaat met rijp terwijl zij leegloopt. In elk geval
verandert energie van vorm --- arbeid wordt \omterm{def:b1:first-law:heat}{warmte}, \omterm{def:b1:first-law:heat}{warmte} wordt arbeid, beide
worden \omterm{def:b1:kinetic-theory:U}{inwendige energie} --- en de \omterm{thm:b1:first-law:firstlaw}{eerste hoofdwet van de thermodynamica} is de
boekhouding die de rekening laat kloppen. Dit hoofdstuk definieert arbeid
en \omterm{def:b1:first-law:heat}{warmte} voor een \omterm{def:b1:first-law:transformation}{thermodynamisch systeem}, formuleert de eerste hoofdwet, voert
de \omterm{def:b1:kinetic-theory:U}{inwendige energie} en de \omterm{def:b1:first-law:enthalpy}{enthalpie} in als de grootheden die zij behoudt, en past
haar toe op de \omterm{def:b1:first-law:transformation}{toestandsveranderingen} van een \omterm{def:b1:kinetic-theory:model}{ideaal gas} en op de calorimetrie.

\section{Toestandsveranderingen, arbeid en warmte}

\begin{definition}[Systeem, toestandsverandering]\label{def:b1:first-law:transformation}
\index{thermodynamisch systeem}\index{toestandsverandering}\index{quasistatische toestandsverandering}\index{omkeerbare toestandsverandering}
Een \emph{thermodynamisch systeem} is de materie binnen een gekozen grens;
al het overige is de \emph{omgeving}. Een \emph{toestandsverandering} brengt
het van de ene evenwichtstoestand naar de andere. Zij is \emph{quasistatisch} als het
systeem een opeenvolging van evenwichtstoestanden doorloopt (traag genoeg
opdat $P$ en $T$ overal gedefinieerd zijn), en \emph{omkeerbaar} als bovendien
het omkeren van de uitwendige omstandigheden de weg omkeert (geen
wrijving, geen eindig temperatuur- of drukverschil met de omgeving).
Namen naar wat constant blijft: \emph{isotherm} ($T$), \emph{isobaar}
($P$), \emph{isochoor} ($V$); \emph{adiabatisch} als er geen \omterm{def:b1:first-law:heat}{warmte} wordt uitgewisseld;
\emph{monobaar} (\emph{monotherm}) als de uitwendige druk
(temperatuur) constant is, wat er binnen ook gebeurt.
\end{definition}

\begin{proposition}[Arbeid van de drukkrachten]\label{prop:b1:first-law:work}
\index{arbeid!van drukkrachten}
Een systeem waarvan het volume met $\dd V$ verandert tegen een uitwendige druk
$P_{\mathrm{ext}}$ ontvangt de arbeid
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
Bij een \omterm{def:b1:first-law:transformation}{quasistatische toestandsverandering} is $P_{\mathrm{ext}} = P$ (de eigen druk
van het systeem) en is $W = -\int P\,\dd V$ min de oppervlakte onder de weg in het
clapeyrondiagram: positief bij samendrukken, negatief bij uitzetten.
\end{proposition}

\begin{proof}
Een zuiger met oppervlak $S$ die door de omgeving met de kracht
$P_{\mathrm{ext}}S$ wordt geduwd verplaatst zich $\dd x$ naar buiten: de omgeving verricht de arbeid
$-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$ op het systeem. Elke
grens is een verzameling van zulke zuigers. Quasistatisch: de zuiger is in
evenwicht, $P_{\mathrm{ext}} = P$ op een verwaarloosbaar verschil na.
\end{proof}

\begin{definition}[Warmte]\label{def:b1:first-law:heat}
\index{warmte}\index{warmtereservoir}
\emph{Warmte} $Q$ is energie die anders dan door macroscopische arbeid
aan het systeem wordt overgedragen --- door moleculaire botsingen op een wand (geleiding),
door de stroming van een fluïdum (convectie), door straling. Een
\emph{warmtereservoir} is een lichaam dat zo groot is dat zijn temperatuur niet verandert,
hoeveel warmte het ook uitwisselt. Tekenafspraak overal: $W$ en $Q$ worden
door het systeem \emph{ontvangen} en zijn positief wanneer zij binnenkomen.
\end{definition}

\section{De eerste hoofdwet}

\begin{theorem}[Eerste hoofdwet van de thermodynamica]\label{thm:b1:first-law:firstlaw}
\index{eerste hoofdwet van de thermodynamica}\index{inwendige energie}
Een \omterm{def:b1:newton-dynamics:conservation}{gesloten systeem} bezit een toestandsfunctie, zijn \emph{\omterm{def:b1:kinetic-theory:U}{inwendige energie}}
$U$ (extensief), zodanig dat voor elke \omterm{def:b1:first-law:transformation}{toestandsverandering}
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
waarin $E_k$ de macroscopische \omterm{thm:b1:work-and-energy:ke}{kinetische energie} is (nul voor een systeem in rust):
de als arbeid en \omterm{def:b1:first-law:heat}{warmte} ontvangen energie wordt als \omterm{def:b1:kinetic-theory:U}{inwendige energie} opgeslagen. $W$
en $Q$ hangen elk van de weg af; hun som niet.
\end{theorem}
\admitted

\begin{remark}[Wat zij zegt]\label{rem:b1:first-law:meaning}
Zij is het behoud van energie, met \omterm{def:b1:first-law:heat}{warmte} erkend als een overdracht van
energie (het schoepenradexperiment van Joule: een gemeten arbeid verhoogt de
temperatuur van water altijd evenveel als een gemeten \omterm{def:b1:first-law:heat}{warmte}). ``$U$ is een
toestandsfunctie'' betekent dat $\Delta U$ door begin- en \omterm{def:b1:transient-regimes:firstorder}{eindtoestand} vastligt ---
zodat een kringproces $\Delta U = 0$ en $W + Q = 0$ heeft: een motor die
arbeid levert moet \omterm{def:b1:first-law:heat}{warmte} ontvangen. De microscopische inhoud van $U$ is die van
\cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[Enthalpie; warmtecapaciteiten]\label{def:b1:first-law:enthalpy}
\index{enthalpie}\index{warmtecapaciteit bij constante druk}
De \emph{enthalpie} $H = U + PV$ is een toestandsfunctie die past bij
\omterm{def:b1:first-law:transformation}{toestandsveranderingen} bij constante druk. De \emph{warmtecapaciteiten bij
constant volume en bij constante druk} zijn
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
in \unit{J/K}; per \omterm{def:b1:kinetic-theory:scales}{mol} $C_{V,m}$, $C_{P,m}$; per kilogram de soortelijke
\omterm{def:b1:first-law:heat}{warmten} $c_V$, $c_P$.
\end{definition}

\begin{proposition}[Warmte bij constant volume en bij constante druk]\label{prop:b1:first-law:qvqp}
Voor een systeem in rust dat alleen via drukkrachten arbeid uitwisselt:
\begin{itemize}
  \item isochore \omterm{def:b1:first-law:transformation}{toestandsverandering}: $W = 0$ en $Q_V = \Delta U$;
  \item monobare \omterm{def:b1:first-law:transformation}{toestandsverandering} tussen twee toestanden bij de uitwendige
        druk $P_0$: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
Isochoor: $\dd V = 0$. Monobaar: $W = -P_0(V_2 - V_1)$, dus $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$ omdat $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[Ideaal gas: $U$, $H$, de betrekking van Mayer]\label{prop:b1:first-law:gas}
\index{betrekking van Mayer}\index{adiabatische exponent}\index{wetten van Joule}
Voor een \omterm{def:b1:kinetic-theory:model}{ideaal gas} hangen $U$ en $H$ alleen van $T$ af (de \omterm{prop:b1:kinetic-theory:Ugas}{wetten van Joule}):
$\dd U = C_V\,\dd T$, $\dd H = C_P\,\dd T$, en
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
$\gamma = 5/3$ voor eenatomige gassen, $7/5$ voor tweeatomige bij gewone
temperaturen. Voor een \omterm{prop:b1:kinetic-theory:condensed}{gecondenseerde fase} in het model van
\cref{prop:b1:kinetic-theory:condensed} geldt $C_P \approx C_V = C$ en
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
$H = U + nRT$; differentieer naar $T$. $C_{V,m} = \tfrac32 R$ of
$\tfrac52 R$ volgens \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{Toestandsveranderingen van een ideaal gas}

\begin{proposition}[De vier standaardtoestandsveranderingen]\label{prop:b1:first-law:transformations}
\index{isotherme toestandsverandering}\index{adiabatische toestandsverandering}\index{wet van Laplace}
Voor $n$ \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{def:b1:kinetic-theory:model}{ideaal gas}:
\begin{itemize}
  \item isochoor: $W = 0$, $Q = \Delta U = nC_{V,m}\Delta T$;
  \item isobaar (quasistatisch): $W = -P\Delta V = -nR\Delta T$, $Q = \Delta H =
        nC_{P,m}\Delta T$;
  \item isotherm omkeerbaar: $\Delta U = 0$, $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$;
  \item adiabatisch omkeerbaar: $Q = 0$, $W = \Delta U = nC_{V,m}(T_2 - T_1)$,
        en langs de weg (de \emph{\omterm{prop:b1:first-law:transformations}{wet van Laplace}})
        \[
        PV^\gamma = \text{const}, \qquad TV^{\gamma - 1} = \text{const}, \qquad
        T^\gamma P^{1 - \gamma} = \text{const} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Isochoor en isobaar: de vorige proposities. Isotherm: $W =
-\int nRT\,\dd V/V$. Adiabatisch: $\dd U = \delta W$, dus $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$, zodat $\dd T/T = -(\gamma - 1)\dd V/V$ (met
$R/C_{V,m} = \gamma - 1$); integreren geeft $TV^{\gamma - 1}$ constant; de andere
vormen volgen uit $PV = nRT$.
\end{proof}

\begin{example}[Pomp en diesel]\label{ex:b1:first-law:diesel}
Lucht ($\gamma = 1.4$) adiabatisch samengedrukt van \qty{1}{bar}, \qty{300}{K}
naar \qty{7}{bar} (een fietspomp): $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- even \qty{250}{\degreeCelsius}, wat de cilinder voelt. Bij een
compressieverhouding van $20$ in een diesel: $T_2 = 300 \times 20^{0.4} = \qty{994}{K}$,
$P_2 = 20^{1.4} = \qty{66}{bar}$ --- boven de zelfontbrandingstemperatuur van
de brandstof: de motor heeft geen vonk nodig (de weekendopgave doorloopt het
hele kringproces).
\end{example}

\begin{remark}[Onomkeerbare toestandsveranderingen]\label{rem:b1:first-law:irreversible}
\index{expansie van Joule}\index{onomkeerbare toestandsverandering}
Is de \omterm{def:b1:first-law:transformation}{toestandsverandering} niet quasistatisch, dan gelden de \omterm{prop:b1:first-law:transformations}{wet van Laplace} en $W =
-\int P\,\dd V$ niet; alleen $W = -\int P_{\mathrm{ext}}\,\dd V$ en de
eerste hoofdwet wel. Twee klassieke gevallen: een gas dat in vacuüm uitzet (\omterm{rem:b1:first-law:irreversible}{expansie
van Joule}) ontvangt geen arbeid en, tussen adiabatische wanden, geen \omterm{def:b1:first-law:heat}{warmte} --- $\Delta U
= 0$, en een \omterm{def:b1:kinetic-theory:model}{ideaal gas} houdt zijn temperatuur; een gas dat plotseling wordt samengedrukt
door een uitwendige druk die naar $P_2$ springt ontvangt $W = -P_2\Delta V$, meer
dan de omkeerbare arbeid naar dezelfde druk, en eindigt warmer
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{Calorimetrie}

\begin{method}[Calorimetrische balans]\label{met:b1:first-law:calorimetry}
\index{calorimetrie}\index{latente warmte}
Lichamen die in een geïsoleerd vat bij constante (atmosferische) druk in
thermisch contact worden gebracht wisselen alleen onderling \omterm{def:b1:first-law:heat}{warmte} uit: in totaal
$\Delta H = \sum Q_P = 0$. Voor elk lichaam is $\Delta H = mc\,\Delta T$ (geen
faseovergang) of $\pm mL$ bij een faseovergang op de bijbehorende temperatuur,
met $L$ de \emph{latente \omterm{def:b1:first-law:heat}{warmte}} (smelt- of verdampingsenthalpie per
massa-eenheid: water, $L_f = \qty{334}{kJ/kg}$ bij \qty{0}{\degreeCelsius}, $L_v =
\qty{2260}{kJ/kg}$ bij \qty{100}{\degreeCelsius}). Schrijf de balans op, los
op naar de eindtemperatuur, en ga na dat de aangenomen fase van elk lichaam
daarmee verenigbaar is.
\end{method}

\begin{example}[IJs in water]\label{ex:b1:first-law:ice}
\qty{50}{g} ijs van \qty{0}{\degreeCelsius} in \qty{200}{g} water van
\qty{30}{\degreeCelsius}: het smelten vergt \qty{16.7}{kJ}; het water kan
\qty{25.1}{kJ} afstaan door tot \qty{0}{\degreeCelsius} af te koelen, dus smelt al het ijs
en geeft de balans $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ kJ
$T_f = \qty{8}{\degreeCelsius}$. Was er \qty{100}{g} ijs geweest, dan zou
\qty{33.4}{kJ} nodig zijn: er smelt maar een deel en $T_f = \qty{0}{\degreeCelsius}$
--- de controle doet ertoe.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:first-law:1}
Eén \omterm{def:b1:kinetic-theory:scales}{mol} lucht ($C_{P,m} = \qty{29.1}{J/(mol.K)}$) wordt bij constante druk van
\qty{300}{K} tot \qty{400}{K} verwarmd. Ontvangen \omterm{def:b1:first-law:heat}{warmte}, ontvangen arbeid
en verandering van de \omterm{def:b1:kinetic-theory:U}{inwendige energie}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:2}
Eén \omterm{def:b1:kinetic-theory:scales}{mol} gas bij \qty{300}{K} wordt isotherm en omkeerbaar samengedrukt
van \qty{2.0}{L} tot \qty{0.50}{L}. Ontvangen arbeid, uitgewisselde \omterm{def:b1:first-law:heat}{warmte}
en $\Delta U$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:3}
Lucht van \qty{300}{K} en \qty{1}{bar} wordt adiabatisch en omkeerbaar
tot een tiende van haar volume samengedrukt. Eindtemperatuur en einddruk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:4}
Geef $C_{V,m}$, $C_{P,m}$ en $\gamma$ voor argon en voor stikstof. Waarom laat
men voor vloeibaar water ($c = \qty{4180}{J/(kg.K)}$) het onderscheid tussen
$c_P$ en $c_V$ vallen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:5}
\qty{200}{g} water van \qty{80}{\degreeCelsius} wordt gemengd met \qty{300}{g}
van \qty{20}{\degreeCelsius} in een geïsoleerde beker. Eindtemperatuur; wat
verandert er als de beker zelf een warmtecapaciteit van \qty{80}{J/K} heeft en op
\qty{20}{\degreeCelsius} begint?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:6}
\qty{50}{g} ijs van \qty{-10}{\degreeCelsius} ($c_{\text{ijs}} = \qty{2100}{J/(kg.K)}$)
wordt in \qty{200}{g} water van \qty{30}{\degreeCelsius} gedaan. \omterm{def:b1:transient-regimes:firstorder}{Eindtoestand}
en eindtemperatuur. Dezelfde vraag met \qty{150}{g} ijs.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:7}
Een gas doorloopt een rechthoekig kringproces in het clapeyrondiagram: $(V_1, P_1)
\to (V_2, P_1) \to (V_2, P_2) \to (V_1, P_2) \to (V_1, P_1)$ met $V_2 > V_1$,
$P_2 > P_1$. Ontvangen arbeid over het kringproces, als functie van de oppervlakte;
teken, en de betekenis van dat teken.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:8}
Een fietspomp perst lucht van \qty{1}{bar}, \qty{300}{K} naar
\qty{7}{bar}, snel genoeg om adiabatisch te zijn. Bereikte temperatuur; waarom de
pomp warm wordt; en waarom de banddruk na het pompen iets
zakt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:9}
Een \omterm{def:b1:kinetic-theory:model}{ideaal gas} zet uit in een leeggepompt, geïsoleerd vat (\omterm{rem:b1:first-law:irreversible}{expansie
van Joule}). Arbeid, \omterm{def:b1:first-law:heat}{warmte}, $\Delta U$, $\Delta T$. Is de \omterm{def:b1:first-law:transformation}{toestandsverandering}
omkeerbaar? Wat zou een echt gas doen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:10}
\qty{1.0}{kg} water koken bij \qty{100}{\degreeCelsius} en \qty{1}{atm}
(stoomvolume \qty{1.67}{m^3}). Toegevoerde \omterm{def:b1:first-law:heat}{warmte}, met de atmosfeer uitgewisselde
arbeid, $\Delta U$. Waar gaat het grootste deel van de latente \omterm{def:b1:first-law:heat}{warmte} heen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:11}
Eén \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{prop:b1:kinetic-theory:Ugas}{tweeatomig gas} bij \qty{300}{K} en \qty{1}{bar}, in een geïsoleerde
cilinder, wordt samengedrukt door de uitwendige druk plotseling op
\qty{5}{bar} te zetten en op evenwicht te wachten. Eindtemperatuur en eindvolume
(eerste hoofdwet met $W = -P_{\mathrm{ext}}\Delta V$); vergelijk met de omkeerbare
adiabatische compressie naar \qty{5}{bar}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:12}
Geluid is een snelle compressie en dus adiabatisch: zijn snelheid is $c = \sqrt{\gamma P/\rho}$
(de isotherme waarde $\sqrt{P/\rho}$ was de vergissing van Newton). Bereken beide
voor lucht bij \qty{20}{\degreeCelsius} en vergelijk met de gemeten
\qty{343}{m/s}. Toon aan dat $c = \sqrt{\gamma RT/M}$ en vergelijk met de \omterm{def:b1:kinetic-theory:kinetictemp}{kwadratisch
gemiddelde snelheid} uit \cref{ch:b1:kinetic-theory}.
\end{exercise}

\section{Probleem: Van de fietspomp naar de dieselmotor}

\begin{problem}[{Weekendopgave --- een handpomp die de vingers verbrandt,
een cilinder die zijn eigen brandstof ontsteekt, en een stalen kogel die op een
luchtkussen stuitert: de eerste hoofdwet in drie compressies}]
\label{pb:b1:first-law:1}
Lucht: ideaal \omterm{prop:b1:kinetic-theory:Ugas}{tweeatomig gas}, $\gamma = 1.40$, $C_{V,m} = \tfrac52 R$, $C_{P,m} =
\tfrac72 R$, $M = \qty{29}{g/mol}$; $R = \qty{8.314}{J/(mol.K)}$. Begintoestand
overal: $P_1 = \qty{1.00}{bar}$, $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{Deel I --- De fietspomp.}
De cilinder van de pomp heeft een volume $V_1 = \qty{212}{cm^3}$; de lucht erin
wordt omkeerbaar en adiabatisch samengedrukt tot zij de banddruk
van \qty{7.0}{bar} bereikt.
\begin{enumerate}
  \item Hoeveelheid lucht in de cilinder.
  \item Volume aan het eind van de compressie.
  \item Temperatuur aan het eind van de compressie.
  \item Arbeid die de lucht bij één slag ontvangt.
  \item De band (\qty{2.0}{L}) moet bij \qty{300}{K} van \qty{1}{bar} naar
        \qty{7}{bar} worden gebracht: hoeveel slagen, en hoeveel arbeid
        in totaal (neem de arbeid per slag constant)?
  \item De hete lucht koelt daarna in de band af tot \qty{300}{K}: wat gebeurt er
        met de banddruk, en wat doet de fietser?
  \item Vergelijk de adiabatische arbeid per slag met de arbeid van een
        isotherme compressie naar dezelfde \qty{7}{bar}. Waarom is de
        isotherme arbeid groter, hoewel het gas kouder eindigt?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De dieselcompressie.}
Een cilinder met $V_1 = \qty{0.50}{L}$ drukt zijn lucht omkeerbaar en
adiabatisch met een factor $20$ samen.
\begin{enumerate}[resume]
  \item Hoeveelheid lucht, en temperatuur $T_2$ na de compressie.
  \item Druk $P_2$.
  \item De brandstof ontbrandt vanzelf boven ongeveer \qty{500}{K}: verklaar waarom een
        diesel geen bougie nodig heeft, en waarom een benzinemotor (verhouding
        $\approx 10$, zie \cref{exo:b1:first-law:3}) niet vanzelf mag ontbranden.
  \item Arbeid die de lucht tijdens de compressie ontvangt.
  \item Bij \qty{3000}{rpm} comprimeert één cilinder $25$ keer per seconde:
        vermogen dat de compressie opneemt (het wordt later in het
        kringproces teruggegeven).
  \item Echte compressies zijn snel (onomkeerbaarheid) en verliezen \omterm{def:b1:first-law:heat}{warmte} aan de
        wanden: welke kant op verschuift elk effect $T_2$?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Verbranding en expansie.}
Boven in de slag verbrandt \qty{20}{mg} brandstof (verbrandingswaarde
\qty{43}{MJ/kg}) terwijl de zuiger begint te dalen, bij constante druk
$P_2$; daarna zet het gas omkeerbaar en adiabatisch weer uit tot $V_1$.
Behandel het gas overal als lucht.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{Warmte} die de brandstof vrijmaakt.
  \item Temperatuur $T_3$ en volume $V_3$ aan het eind van de isobare
        verbranding.
  \item Arbeid die tijdens de isobare fase wordt uitgewisseld (let op het teken!).
  \item Temperatuur $T_4$ aan het eind van de adiabatische expansie, en de
        arbeid die daarbij wordt uitgewisseld.
  \item Nettoarbeid die het gas over de vier slagen levert, en de
        verhouding tot de vrijgemaakte \omterm{def:b1:first-law:heat}{warmte} (het ideale rendement).
  \item Vergelijk dat rendement met $1 - T_1/T_3$, de grens die
        \cref{ch:b1:heat-engines} zal vaststellen voor elke motor die tussen
        deze uiterste temperaturen werkt.
  \item \omterm{def:b1:first-law:heat}{Warmte} die met de uitlaat wordt afgevoerd, met de eerste hoofdwet over het
        kringproces; controleer de balans.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- $\gamma$ meten met een stuiterende kogel.}
Een stalen kogel met massa $m = \qty{16.5}{g}$ past in een verticale glazen buis met
doorsnede $S = \qty{2.0}{cm^2}$ die onderaan wordt afgesloten door een vat met
$V = \qty{10}{L}$ lucht bij $P = \qty{1.0}{bar}$; over $x$ uit
zijn evenwichtsstand verplaatst, gaat hij trillen.
\begin{enumerate}[resume]
  \item Waarom mogen de compressies en expansies van de lucht als
        adiabatisch worden opgevat? Schrijf het verband op tussen een kleine verplaatsing $x$
        en de drukverandering $\dd P$ (de \omterm{prop:b1:first-law:transformations}{wet van Laplace}, gelineariseerd).
  \item Leid daaruit de terugdrijvende kracht op de kogel af en toon aan dat zijn beweging
        harmonisch is, met $\omega_0^2 = \gamma PS^2/mV$.
  \item Bereken de periode, en de waarde van $\gamma$ die men uit een
        gemeten periode van \qty{1.10}{s} zou afleiden.
  \item Waarom vertekenen wrijving tussen kogel en buis, en lekkende lucht,
        het resultaat, en welke kant op?
  \item Vat de drie compressies samen: welke grootheid de eerste hoofdwet
        in elk ervan behield, en wat de \omterm{prop:b1:first-law:gas}{adiabatische exponent} $\gamma$
        bepaalde.
\end{enumerate}
\end{problem}
