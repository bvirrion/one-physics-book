\chapter{Systemen van punten; inleiding tot vaste lichamen}\label{ch:b1:systems-of-points}

Een schoonspringster verlaat de plank, trekt zich samen, maakt twee salto's en
strekt zich weer; door dit alles heen tekent één punt van haar lichaam --- haar
\omterm{def:b1:systems-of-points:cm}{massamiddelpunt} --- dezelfde eenvoudige parabool als een steen. Een jojo die
uit de hand valt doet er drie keer zo lang over om het eind van zijn touwtje te
bereiken als een steen, en komt daar aan met tweeduizend omwentelingen per
minuut. Een massieve bol wint van een holle op elke helling, hoe groot of zwaar
zij ook zijn. De mechanica van één punt, in de vorige hoofdstukken uitgewerkt,
strekt zich uit tot systemen van vele punten --- en tot de vaste lichamen van
alledag --- via een handvol stellingen: het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} beweegt als een punt,
energie en \omterm{def:b1:angular-momentum:L}{impulsmoment} splitsen in een deel van het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} en een deel
``eromheen'', en een lichaam dat om een vaste as draait is een systeem met één
vrijheidsgraad, beheerst door zijn \omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment}.

\section{Massamiddelpunt en impuls}

\begin{definition}[Massamiddelpunt; impuls van een systeem]\label{def:b1:systems-of-points:cm}
\index{massamiddelpunt}\index{zwaartepunt}
Voor \omterm{def:b1:point-kinematics:frame}{puntmassa}'s $m_i$ in $M_i$, met totale massa $M = \sum m_i$, is het
\emph{massamiddelpunt} (zwaartepunt) $G$ gedefinieerd door
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{d.w.z.}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
voor elke oorsprong $O$. De \emph{impuls} van het systeem is $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[Stelling van het massamiddelpunt]\label{thm:b1:systems-of-points:cmtheorem}
\index{stelling van het massamiddelpunt}\index{behoud van impuls}
In een inertiaalstelsel geldt
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} beweegt als een \omterm{def:b1:point-kinematics:frame}{puntmassa} met massa $M$ die alleen aan de
resultante van de \emph{uitwendige} krachten onderworpen is. Voor een \omterm{def:b1:newton-dynamics:conservation}{gesloten
systeem} zonder uitwendige kracht blijft $\vect P$ behouden en beweegt $G$ eenparig.
\end{theorem}

\begin{proof}
Som de tweede wet van Newton over de \omterm{def:b1:point-kinematics:frame}{puntmassa}'s; de inwendige krachten vallen
paarsgewijs weg (actie--reactie); $\sum m_i\vect a_i = M\vect a_G$ door
de definitie van $G$ twee keer te differentiëren.
\end{proof}

\begin{example}[Schoonspringster, terugslag, botsing]\label{ex:b1:systems-of-points:cm}
\index{botsing}
De $G$ van de springster volgt een parabool onder haar gewicht alleen, hoe zij
zich ook wendt --- haar spieren zijn inwendige krachten. Een geweer van \qty{4.0}{kg}
dat een kogel van \qty{10}{g} met \qty{800}{m/s} afvuurt: $\vect P = \vect 0$ ervoor
en erna, dus slaat het geweer met \qty{2.0}{m/s} terug. Twee auto's, \qty{1000}{kg}
met \qty{20}{m/s} en \qty{1500}{kg} in rust, die in elkaar haken: $v =
20000/2500 = \qty{8.0}{m/s}$ --- de impuls blijft behouden, terwijl $60\%$ van de
\omterm{thm:b1:work-and-energy:ke}{kinetische energie} in verkreukeld plaatwerk en warmte is gegaan. De impuls blijft
in elke botsing behouden; de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} alleen in \emph{elastische}
botsingen (\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{De stellingen van Koenig}

\begin{definition}[Barycentrisch stelsel]\label{def:b1:systems-of-points:barycentric}
\index{barycentrisch stelsel}
Het \emph{barycentrische stelsel} $\mathcal R^*$ heeft zijn oorsprong in $G$ en
transleert ten opzichte van het inertiaalstelsel $\mathcal R$ (zijn assen
houden vaste richtingen). Grootheden die erin worden gemeten krijgen een ster: $\vect v_i^*
= \vect v_i - \vect v_G$, en $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[De stellingen van Koenig]\label{thm:b1:systems-of-points:koenig}
\index{stellingen van Koenig}
Het \omterm{def:b1:angular-momentum:L}{impulsmoment} om een punt $O$ en de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} van een
systeem splitsen in de bijdrage van het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt}, dat de hele
massa draagt, en de bijdragen in het \omterm{def:b1:systems-of-points:barycentric}{barycentrische stelsel}:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
waarin $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ en $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ niet van de oorsprong van het stelsel afhangen.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ en $\vect v_i = \vect v_G + \vect v_i^*$;
werk $\sum\vect{OM_i}\wedge m_i\vect v_i$ en $\sum\tfrac12 m_iv_i^2$ uit: de gemengde
termen bevatten $\sum m_i\vect{GM_i} = \vect 0$ of $\sum m_i\vect v_i^* = \vect 0$
en verdwijnen.
\end{proof}

\begin{example}[Een rollend wiel]\label{ex:b1:systems-of-points:rolling}
\index{rollen zonder slippen}
Een wiel met massa $M$, straal $R$ en \omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment} $J$ om zijn as,
dat zonder slippen met snelheid $v$ rolt: het contactpunt is in rust, dus
$\omega = v/R$, en $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- driekwart van $Mv^2$ voor een homogene schijf ($J = \tfrac12 MR^2$),
$Mv^2$ voor een hoepel. Die tweede term is de reden dat fietswielen licht
worden gemaakt: hun rotatie kost energie die de massa van het frame niet kost.
\end{example}

\section{Het tweelichamenprobleem}

\begin{proposition}[Gereduceerde massa]\label{prop:b1:systems-of-points:twobody}
\index{tweelichamenprobleem}\index{gereduceerde massa}
Twee \omterm{def:b1:point-kinematics:frame}{puntmassa}'s die alleen met elkaar wisselwerken (kracht $\vect F$ op $m_2$
vanwege $m_1$, $-\vect F$ op $m_1$): $G$ beweegt eenparig, en de relatieve
positie $\vect r = \vect{M_1M_2}$ voldoet aan
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
de vergelijking van één fictieve \omterm{def:b1:point-kinematics:frame}{puntmassa} met \emph{\omterm{prop:b1:systems-of-points:twobody}{gereduceerde massa}} $\mu$
die onder $\vect F$ om een vast centrum beweegt. In $\mathcal R^*$ is $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ en $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. In $\mathcal R^*$ is $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ en
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$; vul in in $E_k^*$ en $\vect L^*$.
\end{proof}

\begin{example}[Als het centrum meebeweegt]\label{ex:b1:systems-of-points:reduced}
In \cref{ch:b1:central-forces} lag de zon vast: exact alleen in de
limiet $m \ll M$; in het algemeen luidt de \omterm{thm:b1:central-forces:circular}{derde wet van Kepler} $T^2 = 4\pi^2a^3/G(M +
m)$. Voor het trillende HCl-molecuul uit \cref{pb:b1:work-and-energy:1}
is de massa die in de kuil trilt $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$, wat de frequentie met $1.4\%$ verhoogt; voor het
waterstofatoom verschilt de $\mu$ van het elektron van $m_e$ met één deel op $1836$ ---
meetbaar in het spectrum.
\end{example}

\section{Vast lichaam dat om een vaste as draait}

\begin{definition}[Vast lichaam; rotatie om een vaste as]\label{def:b1:systems-of-points:solid}
\index{star lichaam}\index{rotatie om een vaste as}
Een \emph{star lichaam} (vast lichaam) is een systeem waarvan de punten vaste
onderlinge afstanden houden. Draait het om een as $\Delta$ die in het stelsel
vastligt, dan beschrijft elk punt een cirkel met middelpunt op $\Delta$, met dezelfde
hoeksnelheid $\omega = \dot\theta$; een punt op afstand $r_i$ van de as heeft snelheid
$r_i\omega$.
\end{definition}

\begin{proposition}[Traagheidsmoment; impulsmoment; kinetische energie]\label{prop:b1:systems-of-points:inertia}
\index{traagheidsmoment!van vaste lichamen}\index{stelling van Huygens}
Met het \emph{\omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment}} $J_\Delta = \sum m_ir_i^2$ (een integraal
$\int r^2\,\dd m$ voor een continu lichaam):
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
Standaardwaarden (massa $M$): dunne ring of holle cilinder om zijn as,
$MR^2$; homogene schijf of massieve cilinder, $\tfrac12 MR^2$; dunne staaf met
lengte $L$ om haar midden, $\tfrac{1}{12}ML^2$, om een uiteinde,
$\tfrac13 ML^2$; massieve bol om een middellijn, $\tfrac25 MR^2$.
\emph{Stelling van Huygens}: om een as $\Delta$ evenwijdig aan een as
$\Delta_G$ door $G$ op afstand $d$ geldt $J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
$L_\Delta$ en $E_k$: \cref{prop:b1:angular-momentum:inertia}. Staaf om
een uiteinde: $\int_0^L x^2(M/L)\,\dd x = ML^2/3$; om het midden, $\int_{-L/2}^{L/2}
x^2(M/L)\dd x = ML^2/12$. Schijf: ring met straal $r$, breedte $\dd r$, massa
$(2Mr/R^2)\dd r$: $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$; de ring en de
bol worden aangenomen (de bol vergt een dubbele integraal). Huygens: met
$r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2 = r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$ sommeert de gemengde term tot nul per definitie van $G$.
\end{proof}

\begin{theorem}[Dynamica van een vast lichaam om een vaste as]\label{thm:b1:systems-of-points:rotation}
\index{draaipunt}\index{vermogen van een moment}
Voor een vast lichaam dat in een inertiaalstelsel om een vaste as $\Delta$ draait geldt
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
de momenten van de uitwendige krachten om de as drijven de hoekversnelling
aan, en een moment $\mathcal M_\Delta$ levert het vermogen
$\mathcal M_\Delta\omega$. Een ideaal \emph{draaipunt} (een wrijvingsloos lager) oefent
geen moment om zijn as uit en versnelt of remt de rotatie dus niet.
\end{theorem}

\begin{proof}
\cref{thm:b1:angular-momentum:system} geprojecteerd op $\Delta$, met $L_\Delta
= J_\Delta\dot\theta$ en $J_\Delta$ constant voor een \omterm{def:b1:systems-of-points:solid}{star lichaam}; vermenigvuldig met
$\dot\theta$ voor de energievorm (de inwendige krachten van een \omterm{def:b1:systems-of-points:solid}{star lichaam}
verrichten geen arbeid: de afstanden veranderen niet).
\end{proof}

\begin{example}[Fysische slinger; torsieslinger]\label{ex:b1:systems-of-points:pendulums}
\index{fysische slinger}\index{torsieslinger}
Een vast lichaam met massa $M$ dat om een horizontale as op afstand $d$
van $G$ draait: het moment van het gewicht is $-Mgd\sin\theta$, dus $J_\Delta\ddot\theta =
-Mgd\sin\theta$, en kleine zwaaien hebben de periode
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
die van een \omterm{ex:b1:newton-dynamics:pendulum}{wiskundige slinger} met lengte $J_\Delta/Md$ --- voor een homogene staaf
die aan één uiteinde draait, $2L/3$. Een schijf aan een draad met torsieconstante
$C$ (terugdrijvend moment $-C\theta$): $J\ddot\theta = -C\theta$, $T = 2\pi\sqrt{J/C}$
--- de torsiebalans die $G$ en de wet van Coulomb heeft gemeten.
\end{example}

\begin{proposition}[Zonder slippen een helling af rollen]\label{prop:b1:systems-of-points:rollingincline}
Een omwentelingslichaam ($J = kMR^2$ om zijn as) dat op een helling met
hoek $\alpha$ wordt losgelaten rolt zonder slippen met de versnelling
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
$\tfrac57 g\sin\alpha$ voor een massieve bol, $\tfrac23 g\sin\alpha$ voor een schijf,
$\tfrac12 g\sin\alpha$ voor een hoepel --- onafhankelijk van massa en straal.
\end{proposition}

\begin{proof}
Energie: $Mgh = \tfrac12 Mv^2(1 + k)$ langs de helling (de statische wrijving
in het rustende contactpunt verricht geen arbeid), dus $v^2 = 2ah$ met $a$ zoals
aangegeven; of Newton voor $G$ en de momentstelling om $G$ met de
wrijving $f$: $Ma = Mg\sin\alpha - f$, $kMR^2\dot\omega = fR$, $a = R\dot\omega$.
\end{proof}

\begin{remark}[Vervormbare systemen]\label{rem:b1:systems-of-points:deformable}
\index{vervormbaar systeem}
Voor een systeem dat niet star is luidt de stelling van de \omterm{thm:b1:work-and-energy:ke}{kinetische energie}
$\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$: de inwendige krachten
verrichten arbeid (spieren, veren, wrijving binnen het systeem). De schaatsster
die haar \omterm{def:b1:angular-momentum:moment}{armen} intrekt, de springster die zich samentrekt, de fietser die op de
pedalen gaat staan winnen alle \omterm{thm:b1:work-and-energy:ke}{kinetische energie} zonder uitwendige arbeid --- de
stelling van het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} blijft gelden, de energie komt van binnenuit.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:1}
Massa's \qty{2.0}{kg} in $x = 0$ en \qty{3.0}{kg} in $x = \qty{1.0}{m}$:
plaats van $G$. Het stelsel aarde--maan ($81:1$, \qty{384000}{km}):
afstand van $G$ tot het middelpunt van de aarde; vergelijk met de straal
van de aarde.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:2}
Een geweer van \qty{4.0}{kg} vuurt een kogel van \qty{10}{g} af met \qty{800}{m/s}.
Terugslagsnelheid; \omterm{thm:b1:work-and-energy:ke}{kinetische energie} van kogel en geweer; waar kwam die energie
vandaan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:3}
Een auto van \qty{1000}{kg} met \qty{20}{m/s} raakt een auto van \qty{1500}{kg} in rust;
zij haken in elkaar. Gemeenschappelijke snelheid; verloren deel van de \omterm{thm:b1:work-and-energy:ke}{kinetische energie}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:4}
Leid het \omterm{prop:b1:angular-momentum:inertia}{traagheidsmoment} af van een homogene staaf met massa $M$ en lengte $L$,
om een loodrechte as door haar uiteinde, en daarna door haar midden;
controleer de stelling van Huygens tussen beide.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:5}
\omterm{thm:b1:work-and-energy:ke}{Kinetische energie} van een rollende homogene cilinder en van een rollende hoepel
met dezelfde massa en snelheid, als veelvoud van $\tfrac12 Mv^2$. Welke stopt
het eerst op een ruwe vlakke vloer, en welke klimt het hoogst tegen een helling op?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:6}
\omterm{prop:b1:systems-of-points:twobody}{Gereduceerde massa} van HCl ($m_{Cl} = 35m_H$) en van het waterstofatoom
($m_p = 1836m_e$); relatieve correctie op een trillings- of baanfrequentie
die met de lichte massa alleen is berekend.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:7}
Een homogene staaf van \qty{1.0}{m} draait vrij om één uiteinde. Periode
van \omterm{prop:b1:work-and-energy:stability}{kleine trillingen}; lengte van de \omterm{ex:b1:newton-dynamics:pendulum}{wiskundige slinger} met dezelfde
periode; het punt van de staaf (het slagmiddelpunt) waar een scherpe
klap geen reactie in het draaipunt geeft ligt op die afstand --- waarom zoekt
een slagman het op?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:8}
Een schijf ($M = \qty{0.20}{kg}$, $R = \qty{10}{cm}$) hangt aan een draad en
trilt in torsie met een periode van \qty{2.0}{s}. Torsieconstante van
de draad.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:9}

\section{Probleem: De jojo}

\begin{problem}[{Weekendopgave --- een schijf op een dunne as aan het eind
van een touwtje: waarom hij zo traag valt, hoe snel hij onderaan draait,
waarom hij daar ``slaapt'', en welke kant een klos op rolt als je aan zijn
draad trekt}]
\label{pb:b1:systems-of-points:1}
Een jojo is een homogene schijf met massa $M = \qty{50}{g}$ en straal $R =
\qty{3.0}{cm}$ met een dunne as met straal $r = \qty{0.40}{cm}$ waaromheen
een touwtje van lengte $h = \qty{1.0}{m}$ is gewonden; het bovenste eind van
het touwtje wordt vastgehouden. $g = \qty{9.81}{m/s^2}$; $J = \tfrac12 MR^2$ om de
as.

\medskip
\noindent\textbf{Deel I --- Kinematica en energie.}
\begin{enumerate}
  \item Waar ligt het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} $G$ van de jojo? Waarom?
  \item Toon met de stelling van Koenig aan dat zijn \omterm{thm:b1:work-and-energy:ke}{kinetische energie} $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$ is.
  \item Leid $J = \tfrac12 MR^2$ af door over ringen te integreren, en bereken
        $J$.
  \item Het touwtje wikkelt zich zonder slippen van de as af: leg het verband tussen
        $v_G$ en $\omega$.
  \item Leid $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ af en bereken de
        factor tussen haakjes. Becommentarieer.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De val.}
\begin{enumerate}[resume]
  \item Krachten op de jojo; schrijf de stelling van het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} op
        (verticale as naar beneden).
  \item Schrijf de impulsmomentstelling op om de as door $G$
        (welke krachten hebben een moment?).
  \item Combineer met vraag 4 en vind de versnelling $a_G = g/(1 +
        R^2/2r^2)$; bereken haar.
  \item Bereken de spankracht in het touwtje en vergelijk haar met het gewicht.
  \item Tijd om de \qty{1.0}{m} touwtje af te wikkelen, en de snelheid dan;
        vergelijk met een steen die van dezelfde hoogte valt.
  \item Hoeksnelheid onderaan, in rad/s en in omwentelingen per minuut.
  \item Splits de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} onderaan in haar translatie- en
        haar rotatiedeel.
  \item Controleer de energiebalans: vergelijk $Mgh$ met de \omterm{thm:b1:work-and-energy:ke}{kinetische energie}
        onderaan, en zeg waarom de spankracht in het touwtje, hoe groot ook,
        geen arbeid verricht.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Onderaan: de slaper.}
Is het touwtje helemaal afgewikkeld, dan stopt het het \omterm{def:b1:systems-of-points:cm}{massamiddelpunt} in enkele
milliseconden; de jojo blijft draaien in de lus aan het eind van het
touwtje (``slapen'').
\begin{enumerate}[resume]
  \item Waarom overleeft de draaiing het stoppen van $G$? (Welke stelling, en
        welke kracht zou nodig zijn geweest om haar te stoppen?)
  \item Schat de gemiddelde spankracht tijdens het stoppen als $G$ in
        \qty{5}{ms} tot stilstand komt.
  \item De lus schuurt op de as met een wrijvingsmoment $\Gamma =
        \qty{1.0e-4}{N.m}$: hoe lang slaapt de jojo?
  \item Een korte ruk aan het touwtje laat het touwtje de as weer grijpen;
        de draaiende jojo klimt dan omhoog. Tot welke hoogte zou hij met
        energiebehoud kunnen stijgen als er niets verloren ging? Waarom in de
        praktijk minder?
  \item Is de spankracht tijdens het klimmen groter of kleiner dan het
        gewicht? (Teken van $a_G$.)
  \item Wat zou er gebeuren met een jojo waarvan de asstraal gelijk was aan $R$
        (touwtje op de rand gewonden)? Bereken zijn valversnelling.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De klos op de tafel.}
Hetzelfde voorwerp ligt op een ruwe tafel, met de as horizontaal, en zijn draad
verlaat de \emph{onderkant} van de as horizontaal; men trekt met een
kracht $F$. Hij rolt zonder slippen.
\begin{enumerate}[resume]
  \item Welke snelheid heeft het materiaal van de klos in het contactpunt $C$
        met de tafel? Leid daaruit af dat de beweging op elk
        ogenblik een rotatie om $C$ is.
  \item Bereken het moment van $F$ om $C$ (\omterm{def:b1:angular-momentum:moment}{arm} $R - r$) en dat
        van het gewicht en van de reactie van de tafel.
  \item Vind met de impulsmomentstelling om $C$ met $J_C = J +
        MR^2$ (Huygens) de versnelling $a_G$ en toon aan dat de klos
        \emph{naar} de hand toe rolt.
  \item Nu wordt er aan de draad getrokken onder een hoek $\beta$ boven de
        horizontaal. Toon aan dat de klos niet rolt wanneer de werklijn
        van $F$ door $C$ gaat, dat wil zeggen $\cos\beta = r/R$, en
        bij steilere trekrichtingen van de hand weg rolt. Bereken die hoek.
  \item Bereken $a_G$ voor $F = \qty{0.20}{N}$ horizontaal getrokken, en
        de benodigde wrijvingskracht; wat begrenst $F$ voordat de klos
        slipt?
  \item Vat samen: de drie gebruikte stellingen (\omterm{def:b1:systems-of-points:cm}{massamiddelpunt}, \omterm{def:b1:angular-momentum:L}{impulsmoment}
        om een as, energie) en wat elk ervan besliste.
\end{enumerate}
\end{problem}
