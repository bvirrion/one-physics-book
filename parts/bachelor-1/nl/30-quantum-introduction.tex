\chapter{Inleiding tot de kwantumfysica}\label{ch:b1:quantum-introduction}

Schijn ultraviolet licht op een schoon metaal en er vliegen meteen elektronen
uit, met een energie die van de kleur van het licht afhangt en helemaal niet
van de helderheid; schijn er rood licht op en er komt niets uit, hoe fel
de lamp ook is. Stuur elektronen één voor één door twee spleten en zij
landen als stippen, elk op één plaats --- en die stippen bouwen zich over
duizenden heen op tot de strepen van een golf. Geen van beide feiten past bij de
fysica van de eerste negenentwintig hoofdstukken. Dit laatste hoofdstuk voert de
denkbeelden in die er wel bij passen: licht komt in kwanta, de fotonen; materie
heeft een golflengte; wat zich voortplant is een \omterm{def:b1:wave-propagation:signal}{amplitude} waarvan het kwadraat
een kans is; plaats en impuls kunnen niet allebei scherp zijn; en een
opgesloten deeltje kan alleen bepaalde energieën hebben --- en daarom hebben
atomen een grootte, hebben spectra lijnen, en gloeit een korreltje halfgeleider
van een paar nanometer in een kleur die zijn diameter bepaalt.

\section{Het foton}

\begin{theorem}[De betrekkingen van Planck en Einstein]\label{thm:b1:quantum-introduction:photon}
\index{foton}\index{constante van Planck}
Licht met frequentie $\nu$ en golflengte $\lambda$ wisselt alleen in kwanta
energie met materie uit, de \emph{fotonen}, met energie en impuls
\[
E = h\nu = \frac{hc}{\lambda} , \qquad p = \frac{h}{\lambda} = \frac{E}{c} ,
\]
met de \emph{\omterm{thm:b1:quantum-introduction:photon}{constante van Planck}} $h = \qty{6.626e-34}{J.s}$ ($\hbar = h/2\pi =
\qty{1.055e-34}{J.s}$). Handig: $hc = \qty{1240}{eV.nm}$ --- een foton van
\qty{620}{nm} draagt \qty{2}{eV}.
\end{theorem}
\admitted

\begin{proposition}[Het foto-elektrisch effect]\label{prop:b1:quantum-introduction:photoelectric}
\index{foto-elektrisch effect}\index{uittree-arbeid}\index{remspanning}
Licht dat op een metaal met \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{uittree-arbeid}} $W$ valt (de energie die
zijn zwakst gebonden elektronen vasthoudt, enkele eV) maakt elektronen vrij met
maximale \omterm{thm:b1:work-and-energy:ke}{kinetische energie}
\[
E_{k,\max} = h\nu - W ,
\]
dus alleen boven de \emph{drempel} $\nu_0 = W/h$, ogenblikkelijk, met een
energie die niet van de intensiteit afhangt; de intensiteit bepaalt het
\emph{aantal} elektronen per seconde. De \emph{\omterm{prop:b1:quantum-introduction:photoelectric}{remspanning}}
$V_s$ die de stroom net doet verdwijnen meet $E_{k,\max} = eV_s$, en
de grafiek van $V_s$ tegen $\nu$ is een rechte met helling $h/e$ --- zo
mat Millikan $h$.
\end{proposition}

\begin{proof}
Elk foton wordt door één elektron opgenomen, dat er hoogstens $h\nu - W$ van
overhoudt; een golf zou continu energie aan elk elektron leveren, de
energie zou met de intensiteit groeien, en zwak licht zou uren nodig hebben om
$W$ op één atoom te verzamelen (\cref{pb:b1:quantum-introduction:1}) ---
en niets daarvan wordt waargenomen. De uitleg van Einstein (1905) wordt als
wet aangenomen.
\end{proof}

\begin{example}[Ordes van grootte]\label{ex:b1:quantum-introduction:orders}
Een zichtbaar foton: \qty{2}{eV}; een röntgenfoton van \qty{0.1}{nm}: \qty{12}{keV};
een radiofoton bij \qty{100}{MHz}: \qty{4e-7}{eV}. Een rode laser van \qty{1}{mW}
zendt $10^{-3}/(3.1 \times 10^{-19}) = 3 \times 10^{15}$ fotonen per seconde uit;
een radioantenne die \qty{1}{pW} bij \qty{100}{MHz} opvangt ontvangt er $10^{13}$
per seconde --- zoveel en zo klein dat de golfbeschrijving voor elk
doel exact is; het \omterm{def:b1:optical-instruments:eye}{oog} daarentegen kan een flits van
honderd fotonen registreren. Het kwantumkarakter van licht komt tevoorschijn wanneer fotonen
schaars of energierijk zijn: foto-elektrische cellen, röntgendetectoren, de korrel van een
zwakke foto.
\end{example}

\begin{remark}[Comptonverstrooiing]\label{rem:b1:quantum-introduction:compton}
\index{comptonverstrooiing}
Een foton dat van een vrij elektron afketst draagt impuls over als een
biljartbal: zijn golflengte groeit met $\Delta\lambda = \frac{h}{m_ec}(1 -
\cos\theta)$, met $h/m_ec = \qty{2.43}{pm}$ (aangenomen --- een relativistische
botsing, jaar 2). Onzichtbaar voor licht ($\Delta\lambda/\lambda \sim 10^{-5}$),
is het voor röntgenstraling een effect van $20\%$, en het was het experiment dat
de sceptici (1923) overtuigde dat het foton de impuls $h/\lambda$ draagt.
\end{remark}

\section{Materiegolven}

\begin{theorem}[De Broglie]\label{thm:b1:quantum-introduction:debroglie}
\index{debroglie-golflengte}\index{materiegolf}
Een deeltje met impuls $p$ plant zich voort als een golf met golflengte
\[
\lambda = \frac{h}{p} ,
\]
en vertoont dienovereenkomstig buiging en interferentie: elektronen op een
kristal (Davisson en Germer, 1927), door twee spleten (Jönsson, 1961),
neutronen, atomen, en moleculen van zestig koolstofatomen. Voor een elektron
dat door $U$ is versneld is $\lambda = h/\sqrt{2m_eeU} = \qty{1.23}{nm}/\sqrt{U/\unit{V}}$.
\end{theorem}
\admitted

\begin{example}[Waarom wij het nooit zagen]\label{ex:b1:quantum-introduction:wavelengths}
Een elektron van \qty{100}{eV}: $\lambda = \qty{0.12}{nm}$, de afstand tussen
de atomen in een kristal --- vandaar de buiging, en vandaar de
elektronenmicroscoop. Een thermisch neutron (\qty{0.025}{eV}): \qty{0.18}{nm}. Een bal
van \qty{1}{g} met \qty{1}{m/s}: $\lambda = \qty{7e-31}{m}$, $10^{20}$ keer
kleiner dan een kern: geen spleet en geen kristal zou hem ooit kunnen buigen. Het
golfkarakter van materie is universeel; zijn golflengte maakt het alleen
zichtbaar voor het lichte en het trage.
\end{example}

\begin{proposition}[Losse deeltjes bouwen de strepen op]\label{prop:b1:quantum-introduction:buildup}
\index{golf-deeltjedualiteit}
Elektronen die één voor één door twee spleten worden gestuurd (Tonomura, 1989) maken
elk één stip op het scherm, op een plaats die niet te voorspellen is;
de stippen stapelen zich op tot het tweespletenpatroon van een golf
met golflengte $h/p$. Sluit men één spleet, dan verdwijnen de strepen --- het
patroon ontstaat niet doordat elektronen met elkaar interfereren, maar doordat
de golf van elk elektron door beide spleten gaat. Waarnemen door welke spleet
een deeltje ging wist de strepen eveneens uit.
\end{proposition}
\admitted

\section{Amplitudes, kansen en de onzekerheidsrelatie}

\begin{definition}[Golffunctie]\label{def:b1:quantum-introduction:wavefunction}
\index{golffunctie}\index{kansdichtheid}
De toestand van een deeltje wordt beschreven door een \emph{golffunctie}
$\psi(x, t)$ (complex), zodanig dat $|\psi(x, t)|^2\,\dd x$ de
kans is om het deeltje tussen $x$ en $x + \dd x$ aan te treffen wanneer men
kijkt: $|\psi|^2$ is een \emph{kansdichtheid}, en $\int|\psi|^2\dd x =
1$. \omterm{def:b1:wave-propagation:signal}{Amplitudes} tellen op (superpositie): leiden twee wegen naar hetzelfde
punt, dan is $\psi = \psi_1 + \psi_2$ en $|\psi|^2 = |\psi_1|^2 + |\psi_2|^2 +
2\,\mathrm{Re}(\psi_1^\ast\psi_2)$ --- die laatste term is de interferentie.
Een meting geeft een van de mogelijke uitkomsten, willekeurig met de
kansen die de \omterm{def:b1:wave-propagation:signal}{amplitudes} voorschrijven, en laat het deeltje achter in de
gevonden toestand.
\end{definition}

\begin{remark}[Wat er golft]\label{rem:b1:quantum-introduction:whatwaves}
$\psi$ is geen fysisch veld zoals $\vect E$: zij wordt niet gemeten, alleen
haar kwadraat, en zij beschrijft onze best mogelijke kennis van waar
het deeltje zal worden aangetroffen. Eén elektron ligt niet uitgesmeerd over het
scherm; het wordt in zijn geheel op één punt gedetecteerd; wat uitgesmeerd is, is de
kans. Dat is de inhoud van de theorie, en zij is nog nooit door een
proef weerlegd.
\end{remark}

\begin{theorem}[De onzekerheidsrelatie van Heisenberg]\label{thm:b1:quantum-introduction:heisenberg}
\index{onzekerheidsrelatie}
De plaats en de impuls van een deeltje langs dezelfde as kunnen niet
allebei scherp zijn: hun standaardafwijkingen voldoen in elke toestand aan
\[
\Delta x\,\Delta p_x \geq \frac{\hbar}{2} .
\]
Een deeltje dat in een gebied met afmeting $\ell$ is opgesloten heeft dus een
impulsspreiding van ten minste $\hbar/2\ell$, en dus een \omterm{thm:b1:work-and-energy:ke}{kinetische energie} van de orde
$\hbar^2/2m\ell^2$ die geen koeling kan wegnemen: opsluiting kost energie.
\end{theorem}
\admitted

\begin{example}[De grootte van het waterstofatoom]\label{ex:b1:quantum-introduction:hsize}
\index{bohrstraal}
Een elektron binnen $r$ van het proton heeft $p \gtrsim \hbar/r$ en energie
$E(r) \approx \dfrac{\hbar^2}{2m_er^2} - \dfrac{e^2}{4\pi\varepsilon_0r}$: de eerste
term explodeert als $r$ krimpt, de tweede als hij groeit. $\dd E/\dd r = 0$ bij
\[
a_0 = \frac{4\pi\varepsilon_0\hbar^2}{m_ee^2} = \qty{53}{pm} , \qquad
E(a_0) = -\frac{m_ee^4}{2(4\pi\varepsilon_0)^2\hbar^2} = \qty{-13.6}{eV} :
\]
de \emph{\omterm{ex:b1:quantum-introduction:hsize}{bohrstraal}} en de ionisatie-energie van waterstof, tot op het
laatste cijfer. De klassieke fysica kon niet verklaren waarom het elektron niet
de kern in spiraliseert (het straalt) en evenmin waarom alle atomen van een
element gelijk zijn; de onzekerheidsrelatie beantwoordt beide in één
regel: het atoom is zo klein als het kan zijn zonder dat de \omterm{thm:b1:work-and-energy:ke}{kinetische energie}
de binding overtreft.
\end{example}

\section{Kwantisering van de energie}

\begin{proposition}[Deeltje in een doos]\label{prop:b1:quantum-introduction:box}
\index{oneindige put}\index{kwantisering}\index{grondtoestand}
Een deeltje met massa $m$ dat tussen twee ondoordringbare wanden op afstand
$L$ is opgesloten heeft, net als de snaar van Melde
(\cref{ch:b1:wave-propagation}), alleen de staande golven $\lambda_n = 2L/n$,
dus alleen de impulsen $p_n = nh/2L$ en de energieën
\[
E_n = \frac{n^2h^2}{8mL^2} , \qquad n = 1, 2, 3, \ldots
\]
met \omterm{def:b1:quantum-introduction:wavefunction}{golffuncties} $\psi_n(x) \propto \sin(n\pi x/L)$. De energie is
\emph{gekwantiseerd}; de laagste, de \emph{\omterm{prop:b1:quantum-introduction:box}{grondtoestand}} $E_1 > 0$, is de
opsluitingsenergie van de onzekerheidsrelatie, nu exact; het
deeltje springt tussen niveaus door een foton met energie
$h\nu = E_p - E_n$ te absorberen of uit te zenden.
\end{proposition}

\begin{proof}
$\psi$ moet aan de wanden verdwijnen (daar kan het deeltje niet zijn), zodat de
golf een geheel aantal halve golflengten in $L$ kwijt kan --- de
snaarvoorwaarde; De Broglie geeft $p$, en $E = p^2/2m$. Dat $\psi$ met deze
randvoorwaarden aan een golfvergelijking voldoet wordt aangenomen (de
vergelijking van Schrödinger, jaar 2).
\end{proof}

\begin{proposition}[Het waterstofatoom]\label{prop:b1:quantum-introduction:hydrogen}
\index{waterstofatoom}\index{energieniveaus}\index{spectraallijnen}
Het elektron van het \omterm{prop:b1:quantum-introduction:hydrogen}{waterstofatoom} heeft de discrete energieën
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2} , \qquad n = 1, 2, 3, \ldots ,
\]
waarbij $n = 1$ de \omterm{prop:b1:quantum-introduction:box}{grondtoestand} is en $E \geq 0$ het geïoniseerde atoom. Zijn
spectrum bestaat uit lijnen bij $h\nu = E_p - E_n$: de lymanreeks omlaag
naar $n = 1$ (ultraviolet, \qty{122}{nm} voor $2 \to 1$), de balmerreeks
omlaag naar $n = 2$ (zichtbaar: \qty{656}{nm}, \qty{486}{nm}, \qty{434}{nm}, \ldots),
enzovoort. Het model van Bohr (1913) verkrijgt deze energieën door het
\omterm{def:b1:angular-momentum:L}{impulsmoment} van een cirkelbaan te kwantiseren, $m_evr = n\hbar$
(\cref{exo:b1:quantum-introduction:9}); de volledige theorie vervangt banen door
\omterm{def:b1:quantum-introduction:wavefunction}{golffuncties} maar houdt de energieën.
\end{proposition}
\admitted

\begin{example}[Kleuren uit grootte: de kwantumdot]\label{ex:b1:quantum-introduction:dot}
\index{kwantumdot}
Een nanokristal halfgeleider van een paar nanometer sluit zijn
elektronen op als een doos: hoe kleiner het kristal, hoe groter de
opsluitingsenergie die bij de bandkloof van de halfgeleider komt, hoe blauwer het
licht dat het uitzendt. Cadmiumselenidedots met straal \qty{2}{nm} gloeien blauw,
\qty{3}{nm} geel, \qty{4}{nm} rood --- één materiaal, kleuren gekozen met
een liniaal (\cref{pb:b1:quantum-introduction:1}). Televisieschermen gebruiken ze.
\end{example}

\begin{remark}[Wat de rest van de theorie toevoegt]\label{rem:b1:quantum-introduction:rest}
\index{tunneleffect}\index{laser}
De vergelijking van Schrödinger maakt de doos, het atoom en het molecuul
exact; zij voorspelt dat een deeltje een barrière kan oversteken waar het niet de
energie voor heeft om eroverheen te klimmen (het \emph{\omterm{rem:b1:quantum-introduction:rest}{tunneleffect}}, achter de radioactiviteit en de
rastertunnelmicroscoop); dat gelijke deeltjes ofwel gezellig zijn
(bosonen --- vandaar de \emph{laser}, waar fotonen zich in één
mode ophopen) ofwel exclusief (fermionen --- vandaar het periodiek systeem en de
stijfheid van materie); en dat de spin van het elektron een \omterm{def:b1:magnetostatics:moment}{magnetisch
moment} is zonder klassieke tegenhanger. Dat alles is jaar 2 en verder;
het rust allemaal op de weinige denkbeelden van dit hoofdstuk.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:1}
Energie van een foton van: radio bij \qty{100}{MHz}; licht bij \qty{500}{nm};
röntgenstraling bij \qty{0.10}{nm}; gammastraling van \qty{1}{MeV} (geef haar

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:2}
Natrium, $W = \qty{2.3}{eV}$: drempelgolflengte; maximale \omterm{thm:b1:work-and-energy:ke}{kinetische energie}
en \omterm{prop:b1:quantum-introduction:photoelectric}{remspanning} bij licht van \qty{400}{nm}; wat verandert er als de
intensiteit wordt verdubbeld? En bij licht van \qty{600}{nm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:3}
\omterm{thm:b1:quantum-introduction:debroglie}{Debroglie-golflengte} van een elektron van \qty{100}{eV} en van \qty{1.0}{keV};
van een neutron van \qty{0.025}{eV}; van een bal van \qty{1}{g} met \qty{1}{m/s}.
Welke daarvan kan door een kristal (afstand \qty{0.2}{nm}) worden gebogen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:4}
Een elektron in een doos van \qty{1.0}{nm}: $E_1$, $E_2$, golflengte van het
foton dat bij $2 \to 1$ wordt uitgezonden. Een proton in een doos van \qty{5}{fm}
(een kern): $E_1$ in \unit{MeV}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:5}
\omterm{prop:b1:quantum-introduction:photoelectric}{Remspanningen} die voor een metaal zijn gemeten: \qty{0.40}{V} bij
\qty{6.0e14}{Hz}, \qty{1.23}{V} bij \qty{8.0e14}{Hz}, \qty{2.05}{V} bij
\qty{1.0e15}{Hz}. Leid $h$ af (met $e$ bekend) en de \omterm{prop:b1:quantum-introduction:photoelectric}{uittree-arbeid};
drempelgolflengte; welk metaal zou het kunnen zijn (cesium \qty{2.1}{eV},
kalium \qty{2.3}{eV}, zink \qty{4.3}{eV})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:6}
Compton: een röntgenfoton van \qty{10}{pm} verstrooit onder \qty{90}{\degree} aan een
elektron. Nieuwe golflengte; energie van het foton ervoor en erna; energie
die aan het elektron wordt gegeven. Relatieve verschuiving voor licht van \qty{500}{nm} --- waarom
Compton röntgenstraling nodig had.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:7}
\emph{Het experiment van Jönsson.} Elektronen van \qty{50}{keV} (gebruik $p =
\sqrt{2m_eE}$), twee spleten op \qty{2.0}{\micro m} van elkaar, scherm op
\qty{35}{cm}. Golflengte; streepafstand; waarom er een vergrotende elektronenlens
nodig was. In de versie van Tonomura steekt één elektron tegelijk het
apparaat over: wat toont het scherm na 10 en na $10^5$ elektronen,
en wat leert dat?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:8}
Minimale \omterm{thm:b1:work-and-energy:ke}{kinetische energie}, uit de onzekerheidsrelatie, van: een
elektron dat tot \qty{0.1}{nm} is opgesloten (een atoom); een elektron dat tot
\qty{5}{fm} is opgesloten (een kern --- gebruik $E \approx pc$ als het resultaat
relativistisch is, en concludeer of kernen elektronen kunnen bevatten); een
stofdeeltje van \qty{1}{\micro g} dat tot \qty{1}{\micro m} is gelokaliseerd (geef $\Delta v$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:9}
\emph{Het model van Bohr.} Een elektron op een cirkelbaan met straal $r$ om een
proton: (a) snelheid en energie als functie van $r$ (coulombdynamica,
\cref{ch:b1:central-forces}); (b) leg $m_evr = n\hbar$ op en leid
$r_n = n^2a_0$, $E_n = E_1/n^2$ af met $a_0$ en $E_1$ als in
\cref{ex:b1:quantum-introduction:hsize}; (c) $v_1$ en $v_1/c$; (d)
golflengten van Lyman $\alpha$ ($2 \to 1$) en van H$\alpha$ ($3 \to 2$);
(e) toon aan dat \omterm{def:b1:point-kinematics:frame}{baan} $n$ precies $n$ \omterm{thm:b1:quantum-introduction:debroglie}{debroglie-golflengten} bevat.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:10}
\emph{Waterstofachtige systemen.} (a) Toon voor een kern met lading $Ze$ aan dat
$E_n = -Z^2 \times \qty{13.6}{eV}/n^2$ en $r_n = n^2a_0/Z$: bepaal de
ionisatie-energie van He$^+$ en van het laatste elektron van uranium ($Z = 92$; becommentarieer).
(b) Muonisch waterstof: het muon heeft $207$ keer de massa van het elektron:
straal en grondenergie; golflengte van zijn Lyman-$\alpha$-lijn; waarom het
wordt gebruikt om de grootte van het proton te meten (straal \qty{0.84}{fm}).
(c) Positronium (elektron en positron, gelijke massa's): toon aan dat de
\omterm{prop:b1:systems-of-points:twobody}{gereduceerde massa} de energieën halveert: zijn Lyman $\alpha$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:11}
\emph{\omterm{ex:b1:quantum-introduction:dot}{Kwantumdots}.} In een bolvormig nanokristal met straal $R$ is de
opsluitingsenergie van een deeltje met effectieve massa $m^\ast$ gelijk aan
$h^2/8m^\ast R^2$ (aangenomen). CdSe: bandkloof $E_g = \qty{1.74}{eV}$, $m_e^\ast =
0.13\,m_e$, $m_h^\ast = 0.45\,m_e$ (het ``gat'' dat het aangeslagen
elektron achterlaat wordt ook opgesloten). (a) Energie $E_g + E_e + E_h$ van het uitgezonden foton
en de golflengte voor $R = \qty{2.0}{nm}$, \qty{3.0}{nm}, \qty{4.0}{nm}; kleuren.
(b) Straal voor emissie bij \qty{520}{nm} (groen). (c) Waarom zendt massief CdSe
alleen bij \qty{713}{nm} uit? (d) Een spreiding van \qty{10}{\percent} in de straal binnen een
partij: spreiding in golflengte bij \qty{3}{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:12}
\emph{Nulpuntsenergie.} (a) Neem voor een \omterm{def:b1:oscillators-resonance:harmonic}{harmonische oscillator} $E = p^2/2m +
\tfrac12m\omega^2x^2$ aan dat $\Delta x\,\Delta p = \hbar/2$ en minimaliseer $E \approx
\Delta p^2/2m + \tfrac12m\omega^2\Delta x^2$ over $\Delta x$: toon aan dat $E_{\min} = \hbar\omega/2$
(de exacte grondenergie). (b) Het H$_2$-molecuul trilt met $\omega =
\qty{8.3e14}{rad/s}$: nulpuntsenergie in eV, en de laagste
temperatuur waarbij zijn trilling ``thermisch'' is ($k_BT \sim \hbar\omega$).
(c) Vloeibaar helium bevriest bij atmosferische druk nooit: schat de
nulpuntsenergie van een heliumatoom dat in een kristal tot $\Delta x = \qty{0.05}{nm}$
is gelokaliseerd (een deel van de afstand tussen de atomen), en vergelijk met
zijn bindingsenergie in de vaste stof, ongeveer \qty{1}{meV}. (d) Vergelijk de doosformule $E_1 = h^2/8mL^2$ met de
grens uit de onzekerheidsrelatie voor $\Delta x = L/2$: welke is groter, en waarom moet dat zo zijn?
\end{exercise}

\section{Probleem: Het waterstofspectrum en een kwantumdot}

\begin{problem}[{Weekendopgave --- $h$ meten met licht en een
voltmeter, het waterstofatoom uit drie constanten opbouwen, en
de kleur van een kristal met zijn grootte kiezen}]
\label{pb:b1:quantum-introduction:1}
Constanten: $h = \qty{6.626e-34}{J.s}$, $\hbar = \qty{1.055e-34}{J.s}$, $c =
\qty{3.00e8}{m/s}$, $e = \qty{1.602e-19}{C}$, $m_e = \qty{9.11e-31}{kg}$,
$1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$, $hc = \qty{1240}{eV.nm}$.

\medskip
\noindent\textbf{Deel I --- De \omterm{thm:b1:quantum-introduction:photon}{constante van Planck} meten.}
Een kaliumfotokathode wordt met monochromatisch licht beschenen; de
\omterm{prop:b1:quantum-introduction:photoelectric}{remspanning} wordt gemeten: \qty{0.30}{V} bij \qty{6.0e14}{Hz}, \qty{0.69}{V} bij
\qty{7.0e14}{Hz}, \qty{1.13}{V} bij \qty{8.0e14}{Hz}, \qty{1.50}{V} bij
\qty{9.0e14}{Hz}, \qty{2.36}{V} bij \qty{1.1e15}{Hz}.
\begin{enumerate}
  \item Beschrijf de cel en leg uit waarom een tegenspanning de stroom
        kan stoppen, en wat $eV_s$ meet.
  \item Schrijf de betrekking van Einstein op en zeg wat de helling en het
        snijpunt van $V_s(\nu)$ geven.
  \item Leid $h$ uit de gegevens af (met een lineaire fit of met de twee
        uiterste punten); vergelijk met de aanvaarde waarde.
  \item \omterm{prop:b1:quantum-introduction:photoelectric}{Uittree-arbeid} van kalium in eV; drempelfrequentie en
        -golflengte; zou een rode laserpen elektronen kunnen vrijmaken?
  \item De intensiteit van de lamp wordt verdubbeld: wat gebeurt er met $V_s$ en met de
        stroom? Waarom is dat onverenigbaar met een zuiver golfbeeld?
  \item Klassieke schatting: er valt licht van \qty{1}{\micro W/m^2} op het
        metaal; een atoom biedt een oppervlak van ongeveer \qty{1e-20}{m^2}: hoe lang
        zou het duren om \qty{2.2}{eV} op één atoom te verzamelen? Wat wordt er
        in plaats daarvan waargenomen?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het \omterm{prop:b1:quantum-introduction:hydrogen}{waterstofatoom} van Bohr.}
\begin{enumerate}[resume]
  \item Een elektron op een cirkelbaan met straal $r$ om een vast
        proton: snelheid $v(r)$ en \omterm{thm:b1:work-and-energy:em}{mechanische energie} $E(r)$.
  \item De voorwaarde van Bohr $m_evr = n\hbar$: toon aan dat $r_n = n^2a_0$ en geef
        $a_0 = 4\pi\varepsilon_0\hbar^2/m_ee^2$ als getal.
  \item $E_n = E_1/n^2$ met $E_1 = -m_ee^4/[2(4\pi\varepsilon_0)^2\hbar^2]$;
        getalwaarde in eV.
  \item Snelheid op de eerste \omterm{def:b1:point-kinematics:frame}{baan} en de verhouding $v_1/c$ (de
        fijnstructuurconstante $\alpha \approx 1/137$).
  \item Golflengten van de lijnen $2 \to 1$ (Lyman $\alpha$), $3 \to 2$ en
        $4 \to 2$ (Balmer H$\alpha$, H$\beta$): welke zijn zichtbaar?
  \item Toon aan dat $1/\lambda = R_H(1/n^2 - 1/p^2)$ en bereken de
        rydbergconstante $R_H$; de kortste balmergolflengte.
  \item Ionisatie-energie van waterstof vanuit de \omterm{prop:b1:quantum-introduction:box}{grondtoestand}; golflengte
        van het foton dat het net ioniseert.
  \item De Broglie: toon aan dat de omtrek van \omterm{def:b1:point-kinematics:frame}{baan} $n$ gelijk is aan $n$
        golflengten --- de voorwaarde van Bohr is een staandegolfvoorwaarde.
  \item Twee dingen die het model fout heeft, één ding dat het precies
        goed heeft, en wat in de volledige theorie de \omterm{def:b1:point-kinematics:frame}{baan} vervangt.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{ex:b1:quantum-introduction:dot}{kwantumdot}.}
Een CdSe-nanokristal met straal $R$ sluit een aangeslagen elektron op
(effectieve massa $m_e^\ast = 0.13\,m_e$) en het gat dat het achterlaat
($m_h^\ast = 0.45\,m_e$); de opsluitingsenergie van elk in een bol met
straal $R$ is $h^2/8m^\ast R^2$ (aangenomen); de bandkloof van massief CdSe is
$E_g = \qty{1.74}{eV}$.
\begin{enumerate}[resume]
  \item Leid de energieën $E_n = n^2h^2/8mL^2$ van een deeltje in een
        eendimensionale doos af uit de staandegolfvoorwaarde, en
        becommentarieer de analogie met de formule voor de bol.
  \item Opsluitingsenergieën van het elektron en van het gat voor
        $R = \qty{2.0}{nm}$.
  \item Energie en golflengte van het uitgezonden foton voor $R = \qty{2.0}{nm}$,
        \qty{3.0}{nm} en \qty{4.0}{nm}; kleuren.
  \item Leg in één zin uit waarom kleinere dots blauwer zijn.
  \item Controleer de \omterm{def:b1:units-dimensions:oom}{orde van grootte} met de onzekerheidsrelatie:
        $\Delta p \sim \hbar/R$ voor het elektron bij $R = \qty{2}{nm}$, en de
        \omterm{thm:b1:work-and-energy:ke}{kinetische energie} die daaruit volgt.
  \item Een dot zendt \qty{1.0}{nW} uit bij \qty{577}{nm}: fotonen per seconde.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Kwanta tellen.}
\begin{enumerate}[resume]
