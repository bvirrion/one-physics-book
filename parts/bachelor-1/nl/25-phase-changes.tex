\chapter{Faseovergangen van een zuivere stof}\label{ch:b1:phase-changes}

Water is de ene stof die iedereen in alle drie de toestanden heeft gezien,
en het is het werkfluïdum van de meeste elektriciteitscentrales ter wereld, het
\omterm{def:b1:heat-engines:vapourcycle}{koudemiddel} van de eerste ijsmachines en de motor van het weer.
IJs smelt bij een temperatuur die niet wijkt terwijl de warmte binnenstroomt;
een ketel kookt bij \qty{100}{\degreeCelsius} aan zee en bij
\qty{70}{\degreeCelsius} op de Everest; een gesloten fles zeer koud water
bevriest bij een tik in één klap. Dit hoofdstuk tekent de kaart van de \omterm{def:b1:phase-changes:phase}{fasen} van een
\omterm{def:b1:phase-changes:phase}{zuivere stof} --- de $(T, P)$- en $(v, P)$-diagrammen --- en geeft de
drie gereedschappen die erbij horen: de hefboomregel (hoeveel van elke \omterm{def:b1:phase-changes:phase}{fase}),
de \omterm{def:b1:phase-changes:phase}{latente warmte} (wat een overgang kost, in energie en in entropie) en de
\omterm{thm:b1:phase-changes:clapeyron}{betrekking van Clausius--Clapeyron} (hoe steil de grenslijnen lopen). Het eindigt
waar de kaart tekortschiet: bij \omterm{def:b1:phase-changes:metastable}{metastabiele toestanden}, waar het weer
en de verrassingen van de snelkookpan vandaan komen.

\section{Fasen en het $(T, P)$-diagram}

\begin{definition}[Zuivere stof, fase, faseovergang]\label{def:b1:phase-changes:phase}
\index{fase}\index{faseovergang}\index{zuivere stof}\index{latente warmte}
Een \emph{zuivere stof} is een lichaam van één chemische soort. Een
\emph{fase} is een deel ervan dat homogeen is in zijn fysische
eigenschappen: vast, vloeibaar, gasvormig (\emph{damp} wanneer zij naast haar
vloeistof bestaat). Een \emph{faseovergang} (smelten/stollen, verdampen/
condenseren, sublimeren/rijpen) is de gang van de ene fase naar de
andere; bij vaste druk gebeurt hij bij een vaste temperatuur, met uitgewisselde
warmte maar zonder temperatuurverandering: de \emph{latente warmte}.
\end{definition}

\begin{proposition}[Evenwichtsdiagram]\label{prop:b1:phase-changes:PTdiagram}
\index{fasediagram}\index{tripelpunt}\index{kritisch punt}\index{verzadigingsdampdruk}
In het $(T, P)$-vlak beslaat elke \omterm{def:b1:phase-changes:phase}{fase} een gebied; twee \omterm{def:b1:phase-changes:phase}{fasen} bestaan alleen
langs een kromme naast elkaar (sublimatie, smelten, verdampen), alle drie alleen
in één punt, het \emph{tripelpunt}. De verdampingskromme
$P = P_s(T)$ --- de \emph{verzadigingsdampdruk} --- eindigt in het
\emph{\omterm{prop:b1:phase-changes:PTdiagram}{kritische punt}} $(T_c, P_c)$, waarboven vloeistof en gas niet langer
te onderscheiden zijn. Voor water: tripelpunt \qty{273.16}{K}, \qty{611}{Pa};
\omterm{prop:b1:phase-changes:PTdiagram}{kritisch punt} \qty{647}{K}, \qty{221}{bar}. De smeltkromme van water
helt naar links (ijs is minder dicht dan water); bij vrijwel elke
andere stof helt zij naar rechts.
\end{proposition}
\admitted

\begin{remark}[Koken, verdampen en de snelkookpan]\label{rem:b1:phase-changes:boiling}
Een vloeistof \emph{verdampt} bij elke temperatuur vanaf haar oppervlak, zolang
de partiële druk van haar damp erboven onder $P_s(T)$ ligt; zij
\emph{kookt} wanneer $P_s(T)$ de omgevingsdruk bereikt, zodat er
dampbellen binnenin kunnen groeien. Water kookt bij \qty{100}{\degreeCelsius}
omdat $P_s(\qty{100}{\degreeCelsius}) = \qty{1.013}{bar}$; bij
\qty{0.33}{bar} op de Everest kookt het rond \qty{70}{\degreeCelsius}, in een
snelkookpan bij \qty{1.8}{bar} rond \qty{117}{\degreeCelsius}. De
temperatuur van een kokende vloeistof wordt door de druk vastgepind --- een feit
dat twee eeuwen lang de celsiusschaal heeft gedefinieerd.
\end{remark}

\section{Het $(v, P)$-diagram en de hefboomregel}

\begin{proposition}[Isothermen van Andrews]\label{prop:b1:phase-changes:andrews}
\index{isothermen van Andrews}\index{verzadigingskromme}
In het clapeyronvlak $(v, P)$ van het soortelijk volume heeft een isotherm
met $T < T_c$ drie delen: een steile tak (vloeistof, vrijwel
onsamendrukbaar), een horizontaal \emph{plateau} bij $P = P_s(T)$ waar vloeistof
en damp naast elkaar bestaan, en een hyperboolachtige tak (damp). De uiteinden van het
plateau zijn de soortelijke volumes $v_l(T)$ en $v_v(T)$ van de verzadigde
vloeistof en damp; hun meetkundige plaats is de \emph{verzadigingskromme} (de
klok), waarvan de top het \omterm{prop:b1:phase-changes:PTdiagram}{kritische punt} is, waar de isotherm $T_c$
een horizontaal buigpunt heeft. Boven $T_c$ bestaat er geen plateau.
\end{proposition}
\admitted

\begin{theorem}[Hefboomregel]\label{thm:b1:phase-changes:lever}
\index{hefboomregel}\index{dampfractie}
Een massa $m$ met soortelijk volume $v$ op het plateau bij $T$ splitst in een
massafractie damp $x = m_v/m$ (de \emph{dampfractie}) en een vloeistoffractie
$1 - x$ met
\[
v = (1 - x)v_l + x\,v_v , \qquad\text{d.w.z.}\qquad
x = \frac{v - v_l}{v_v - v_l} = \frac{LM}{LV} ,
\]
de verhouding van de stukken die het toestandspunt $M$ op het plateau afsnijdt.
Dezelfde regel geldt voor elke extensieve grootheid per massa-eenheid ($u$,
$h$, $s$): $h = (1 - x)h_l + xh_v$.
\end{theorem}

\begin{proof}
Volumes tellen op: $V = m_lv_l + m_vv_v$; deel door $m$. Energie, \omterm{def:b1:first-law:enthalpy}{enthalpie} en
entropie zijn eveneens extensief.
\end{proof}

\begin{example}[Een ketel en een stoomketel]\label{ex:b1:phase-changes:lever}
Water bij \qty{100}{\degreeCelsius}, \qty{1}{atm}: $v_l = \qty{1.04e-3}{m^3/kg}$,
$v_v = \qty{1.67}{m^3/kg}$ --- een verhouding van $1600$. Een gesloten vat van \qty{1.0}{L}
met \qty{0.50}{kg} water op die temperatuur heeft $v =
\qty{2.0e-3}{m^3/kg}$ en $x = (2.0 - 1.04) \times 10^{-3}/1.67 = 5.8 \times 10^{-4}$:
\qty{0.29}{g} damp die de helft van het vat vult. De dampmassa in een gesloten
vat verdubbelen verschuift het toestandspunt nauwelijks; de druk ligt
alleen door de temperatuur vast.
\end{example}

\section{Energetica van een faseovergang}

\begin{proposition}[Latente warmte, overgangsenthalpie en -entropie]\label{prop:b1:phase-changes:latent}
\index{overgangsenthalpie}\index{overgangsentropie}
Bij de temperatuur $T$ en de druk $P_s(T)$ waarbij de \omterm{def:b1:phase-changes:phase}{fasen} naast elkaar bestaan is de
overgang van een massa-eenheid van \omterm{def:b1:phase-changes:phase}{fase} 1 naar \omterm{def:b1:phase-changes:phase}{fase} 2 omkeerbaar,
isotherm en isobaar, en geldt
\[
L_{12}(T) = h_2 - h_1 , \qquad s_2 - s_1 = \frac{L_{12}}{T} , \qquad
u_2 - u_1 = L_{12} - P_s(v_2 - v_1) .
\]
$L_{12} > 0$ wanneer men naar de minder geordende \omterm{def:b1:phase-changes:phase}{fase} gaat (smelten,
verdampen, sublimeren), en $L_{21} = -L_{12}$. Voor water bij \qty{1}{atm}:
$L_f = \qty{334}{kJ/kg}$ bij \qty{0}{\degreeCelsius}, $L_v = \qty{2257}{kJ/kg}$ bij
\qty{100}{\degreeCelsius}.
\end{proposition}

\begin{proof}
Bij constante druk is de ontvangen warmte $\Delta H$
(\cref{prop:b1:first-law:qvqp}); omkeerbaar en isotherm is zij
$T\Delta S$ (\cref{prop:b1:second-law-entropy:identity}); en $U = H - PV$.
\end{proof}

\begin{example}[Waar de verdampingswarmte blijft]\label{ex:b1:phase-changes:whereheat}
Voor water bij \qty{100}{\degreeCelsius}: $P_s(v_v - v_l) = 1.013 \times 10^5 \times
1.67 = \qty{169}{kJ/kg}$, dus $u_v - u_l = 2257 - 169 = \qty{2088}{kJ/kg}$:
$93\%$ van de \omterm{def:b1:phase-changes:phase}{latente warmte} gaat naar het uit elkaar trekken van de moleculen, $7\%$
naar het wegduwen van de atmosfeer. De verdampingsentropie,
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, overtreft die van het smelten, $334/273 =
\qty{1.22}{kJ/(kg.K)}$, ruimschoots: smelten maakt de moleculen los, koken maakt ze vrij.
\end{example}

\begin{method}[Calorimetrie met faseovergangen]\label{meth:b1:phase-changes:calorimetry}
In een geïsoleerd vat bij constante druk geldt $\sum\Delta H_i = 0$. Raad
de \omterm{def:b1:transient-regimes:firstorder}{eindtoestand} (welke \omterm{def:b1:phase-changes:phase}{fasen} aanwezig zijn, bij welke temperatuur);
schrijf de $\Delta H$ van elk lichaam als voelbare warmten $mc\,\Delta T$ plus \omterm{def:b1:phase-changes:phase}{latente
warmten} $\pm mL$ voor de overgangen die het ondergaat; los op; \emph{controleer de
gok}: een lichaam dat boven zijn kookpunt of onder zijn smeltpunt
eindigt, of een negatieve massa van een \omterm{def:b1:phase-changes:phase}{fase}, betekent dat de gok fout was --- de
\omterm{def:b1:transient-regimes:firstorder}{eindtoestand} is dan een mengsel bij de overgangstemperatuur, en de
onbekende wordt de omgezette massa (\cref{ex:b1:first-law:ice}).
\end{method}

\begin{example}[Stoom verbrandt]\label{ex:b1:phase-changes:steam}
\qty{10}{g} stoom van \qty{100}{\degreeCelsius} die op de huid condenseert geeft
$0.010 \times 2257 = \qty{22.6}{kJ}$ af voordat zij ook maar één graad is afgekoeld ---
evenveel als \qty{10}{g} kokend water afgeeft bij een afkoeling van
\qty{540}{K}. Daarom brandt stoom zoveel erger dan water van dezelfde
temperatuur, en daarom heeft een stoomradiator zo weinig massastroom nodig.
\end{example}

\section{De betrekking van Clausius--Clapeyron}

\begin{theorem}[Clausius--Clapeyron]\label{thm:b1:phase-changes:clapeyron}
\index{betrekking van Clausius--Clapeyron}
Langs een coëxistentiekromme van de \omterm{def:b1:phase-changes:phase}{fasen} 1 en 2 geldt
\[
\frac{\dd P_s}{\dd T} = \frac{L_{12}(T)}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
Voer een massa-eenheid rond langs een infinitesimaal \omterm{def:b1:heat-engines:carnotcycle}{kringproces van Carnot} binnen de klok:
volledige overgang $1 \to 2$ langs het plateau bij $T + \dd T$, druk
$P_s + \dd P_s$ (ontvangen warmte $L_{12}$ in eerste orde); een infinitesimale
adiabatische stap omlaag naar $T$; de omgekeerde overgang langs het plateau bij
$T$, $P_s$; een adiabatische stap terug. Het kringproces is omkeerbaar en dithermisch
tussen $T + \dd T$ en $T$, dus is zijn \omterm{def:b1:heat-engines:efficiency}{rendement} $\dd T/T$
(\cref{thm:b1:heat-engines:carnot}); zijn arbeid, de oppervlakte van de dunne
rechthoek, is $\dd P_s\,(v_2 - v_1)$. Dus $\dd P_s\,(v_2 - v_1)/L_{12} =
\dd T/T$.
\end{proof}

\begin{corollary}[Verzadigingsdampdruk van een vloeistof]\label{cor:b1:phase-changes:vapourpressure}
Is de damp een \omterm{def:b1:kinetic-theory:model}{ideaal gas} en $v_v \gg v_l$, zodat $v_v - v_l \approx
RT/(MP_s)$, en verandert $L_v$ weinig, dan is
\[
\frac{\dd P_s}{P_s} = \frac{ML_v}{R}\,\frac{\dd T}{T^2} , \qquad
\ln\frac{P_s(T)}{P_s(T_0)} = \frac{ML_v}{R}\Bigl(\frac{1}{T_0} - \frac{1}{T}\Bigr) :
\]
de dampdruk stijgt ruwweg exponentieel met de temperatuur. Voor
water rond \qty{100}{\degreeCelsius} is $ML_v/R = 0.018 \times 2.257 \times 10^6/
8.314 = \qty{4890}{K}$ en $\dd P_s/\dd T = 2.257 \times 10^6/(373 \times 1.67) =
\qty{3.6}{kPa/K}$: elke \omterm{def:b1:kinetic-theory:temperature}{kelvin} extra kooktemperatuur kost
\qty{36}{mbar} extra druk.
\end{corollary}

\begin{proof}
Vul $v_v - v_l \approx RT/(MP_s)$ in in de stelling en scheid de
variabelen; integreer met constante $L_v$.
\end{proof}

\begin{example}[IJs onder een schaats]\label{ex:b1:phase-changes:skate}
Voor ijs--water is $L_f = \qty{334}{kJ/kg}$ en $v_l - v_s = (1.000 - 1.091)

op één ijzer van \qty{30}{cm} bij \qty{3}{mm} ($\qty{9}{cm^2}$) oefent

maar de negatieve helling is echt, en zij verklaart waarom een draad met
gewichten eraan dwars door een blok ijs zakt dat achter hem weer aanvriest.
\end{example}

\section{Metastabiele toestanden}

\begin{definition}[Onderkoeling, oververhitting]\label{def:b1:phase-changes:metastable}
\index{onderkoeling}\index{oververhitting}\index{metastabiele toestand}\index{kiemvorming}
Een \omterm{def:b1:phase-changes:phase}{fase} kan voorbij haar evenwichtsgrens blijven bestaan: vloeibaar water
dat onder \qty{0}{\degreeCelsius} wordt gekoeld zonder te bevriezen (\emph{onderkoeling}),
boven zijn kookpunt wordt verhit zonder te koken (\emph{oververhitting}),
damp die voorbij $P_s$ wordt samengeperst zonder te condenseren (oververzadiging).
Zulke \emph{metastabiele toestanden} vergen de afwezigheid van plaatsen voor
\emph{kiemvorming} --- stof, krassen, opgelost gas, ijskristallen --- waarop de
nieuwe \omterm{def:b1:phase-changes:phase}{fase} kan beginnen; een verstoring kantelt ze plotseling en onomkeerbaar
naar het evenwicht.
\end{definition}

\begin{example}[De onderkoelde fles]\label{ex:b1:phase-changes:supercooled}
Stilstaand water van \qty{-8}{\degreeCelsius} in een schone fles, aangetikt: er
schieten ijskristallen doorheen en de temperatuur springt naar
\qty{0}{\degreeCelsius}. Het proces is adiabatisch en snel, dus bevriest de voelbare
warmte $c\,\Delta T$ een fractie $x = c\,\Delta T/L_f = 4.18 \times 8/334 =
0.10$ van het water; de geproduceerde entropie, $c\ln(273/265) - xL_f/273 =
124 - 122 = \qty{+2}{J/(kg.K)}$, is positief zoals het moet. Wolken zitten
vol onderkoelde druppels tot \qty{-40}{\degreeCelsius}; ijsafzetting op
vliegtuigen en het bezaaien van wolken maken er beide gebruik van.
\end{example}

\begin{remark}[Kooksteentjes en bellenvaten]\label{rem:b1:phase-changes:nucleation}
Een bel met straal $r$ in een vloeistof onder druk $P$ bevat damp met
$P + 2\sigma/r$ (oppervlaktespanning $\sigma$, volume van jaar 2); zij kan alleen
groeien als $P_s(T) > P + 2\sigma/r$, zodat een schone vloeistof op haar kookpunt
geen bel heeft die klein genoeg is om te beginnen en enkele \omterm{def:b1:kinetic-theory:temperature}{kelvin} oververhit
voordat zij uitbarst --- het ``stoten'' dat kooksteentjes voorkomen door
holtes met ingesloten gas aan te bieden. Het bellenvat van de deeltjesfysica
draaide op dezelfde instabiliteit in omgekeerde richting: een oververhitte vloeistof
kookt het eerst langs het spoor van een ioniserend deeltje.
\end{remark}

\begin{example}[Luchtvochtigheid en dauw]\label{ex:b1:phase-changes:dew}
\index{relatieve luchtvochtigheid}\index{dauwpunt}
Lucht bevat waterdamp met een partiële druk $P_v \leq P_s(T)$; de
\emph{\omterm{ex:b1:phase-changes:dew}{relatieve luchtvochtigheid}} is $P_v/P_s(T)$, het \emph{dauwpunt} de
temperatuur waarbij $P_s$ tot $P_v$ zakt. Lucht van \qty{25}{\degreeCelsius}
($P_s = \qty{3.17}{kPa}$) met $60\%$ luchtvochtigheid heeft $P_v = \qty{1.9}{kPa}$,
$\rho_v = P_vM/(RT) = \qty{14}{g/m^3}$, en slaat neer op elk oppervlak onder
ongeveer \qty{17}{\degreeCelsius} --- het gras bij dageraad, de koude fles, de
badkamerspiegel. Stijgt diezelfde lucht op, dan koelt zij adiabatisch af met
\qty{9.8}{K/km} (\cref{pb:b1:phase-changes:1}) en bereikt zij haar dauwpunt een
kilometer of zo hoger: daar liggen de vlakke onderkanten van de stapelwolken.
\end{example}

\section{Probleem: Water in vier taferelen}

\begin{problem}[{Weekendopgave --- de snelkookpan, de wolk, het
gesloten vat en de onderkoelde fles: één stof, vier toepassingen van
haar fasediagram}]
\label{pb:b1:phase-changes:1}
Gegevens voor water: $L_v = \qty{2257}{kJ/kg}$ bij \qty{100}{\degreeCelsius} en
ongeveer \qty{2.2}{MJ/kg} bij \qty{117}{\degreeCelsius}, \qty{2.45}{MJ/kg} rond
\qty{20}{\degreeCelsius}; $L_f = \qty{334}{kJ/kg}$; $c = \qty{4.18}{kJ/(kg.K)}$
(vloeibaar); $M = \qty{18}{g/mol}$; $ML_v/R = \qty{4890}{K}$ rond
\qty{100}{\degreeCelsius}. Verzadigingsdrukken: \qty{0.61}{kPa}
(\qty{0}{\degreeCelsius}), \qty{1.23} (\qty{10}), \qty{1.70} (\qty{15}),
\qty{2.34} (\qty{20}), \qty{3.17} (\qty{25}), \qty{4.24}{kPa}
(\qty{30}{\degreeCelsius}), \qty{101.3}{kPa} (\qty{100}{\degreeCelsius}).
Verzadigde soortelijke volumes: $v_l = \qty{1.0e-3}{m^3/kg}$ (koud water),
\qty{1.9e-3}{m^3/kg} bij \qty{360}{\degreeCelsius}; $v_v = \qty{0.0217}{m^3/kg}$
bij \qty{300}{\degreeCelsius}, $0.0108$ bij \qty{340}{\degreeCelsius}, $0.0088$
bij \qty{350}{\degreeCelsius}; \omterm{prop:b1:phase-changes:PTdiagram}{kritisch punt} \qty{374}{\degreeCelsius},
\qty{221}{bar}, $v_c = \qty{3.1e-3}{m^3/kg}$. Lucht: $M = \qty{29}{g/mol}$, $c_p =
\qty{1.0}{kJ/(kg.K)}$, $\gamma = 1.4$; $g = \qty{9.8}{m/s^2}$; $R =
\qty{8.314}{J/(mol.K)}$.

\medskip
\noindent\textbf{Deel I --- De snelkookpan.}
\begin{enumerate}
  \item Waarom kan water in een open pan op zeeniveau niet boven
        \qty{100}{\degreeCelsius} worden verwarmd, hoe sterk de vlam ook is?
  \item Leid uit de \omterm{thm:b1:phase-changes:clapeyron}{betrekking van Clausius--Clapeyron}
        $\ln[P_s(T)/P_s(T_0)] = (ML_v/R)(1/T_0 - 1/T)$ af, en noem de
        benaderingen.
  \item Temperatuur in een pan waarvan het ventiel bij \qty{1.8}{bar}
        absoluut opengaat.
  \item Het ventiel is een gewicht op een opening van \qty{20}{mm^2};
        atmosferische druk \qty{1.0}{bar}: massa van het gewicht.
  \item \qty{2.0}{L} water wordt door een plaat van \qty{2.0}{kW} van
        \qty{20}{\degreeCelsius} tot de werktemperatuur gebracht (verwaarloos de pan):
        warmte en tijd.
  \item Eenmaal daar blijft de plaat op \qty{2.0}{kW}: massa stoom
        die per minuut door het ventiel ontsnapt. Wat moet de kok doen?
  \item De snelheid van kookreacties verdubbelt ruwweg per \qty{10}{K}:
        met welke factor bekort de pan een recept van \qty{30}{min} op zeeniveau?
        En hoe lang zou het recept op de Everest duren, waar
        water bij \qty{70}{\degreeCelsius} kookt?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De wolk.}
\begin{enumerate}[resume]
  \item Definieer de \omterm{ex:b1:phase-changes:dew}{relatieve luchtvochtigheid}. Lucht van \qty{25}{\degreeCelsius} bij $60\%$:
        partiële dampdruk, en massa damp per kubieke meter
        (\omterm{def:b1:kinetic-theory:model}{ideaal gas}).
  \item Dauwpunt van die lucht (interpoleer de tabel). Verklaar de dauw
        op het gras bij dageraad en de druppels op een koude fles.
  \item Een pakketje droge lucht stijgt op zonder warmte uit te wisselen. Toon met
        de omkeerbare adiabatische wet en de \omterm{thm:b1:fluid-statics:fundamental}{hydrostatische betrekking}
        $\dd P/\dd z = -\rho g$ aan dat $\dd T/\dd z = -g/c_p$, en
        bereken die waarde.
  \item Het dauwpunt van het stijgende pakketje zelf daalt met ongeveer
        \qty{2}{K/km} naarmate de druk zakt. Hoogte waarop het pakketje uit
        vraag~8 verzadigd raakt --- de basis van de wolk.
  \item Boven de basis staat de damp die condenseert haar \omterm{def:b1:phase-changes:phase}{latente warmte}
        aan het pakketje af. Hoeveel warmer wordt die lucht als er \qty{1}{g}
        water per kilogram lucht condenseert? Wat doet dat met de
        afkoelsnelheid van het pakketje terwijl het verder stijgt, en waarom is dat
        van belang voor onweersbuien?
  \item Een stapelwolk van \qty{1}{km^3} bevat \qty{1}{g} vloeibaar water per kubieke
        meter: massa water, en de energie die de condensatie ervan
        heeft vrijgemaakt, in joule en in \unit{GW h}.
  \item Een badkamerspiegel van \qty{18}{\degreeCelsius}, lucht van
        \qty{25}{\degreeCelsius} bij $80\%$: beslaat hij? Bij welke luchtvochtigheid
        zou dat ophouden?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het gesloten vat.}
Een stijf stalen vat van \qty{1.0}{L} wordt met een massa $m$ water
gevuld en zonder lucht gesloten, en daarna langzaam verwarmd.
\begin{enumerate}[resume]
  \item $m = \qty{0.50}{kg}$ bij \qty{20}{\degreeCelsius}: druk binnenin,
        soortelijk volume, massa damp (gebruik het ideale gas voor $v_v$),
        deel van het volume dat door de vloeistof wordt gevuld.
  \item Bij verwarming: stijgt of daalt het vloeistofniveau? Bij ongeveer welke
        temperatuur vult de vloeistof het vat, en wat gebeurt er daarna met
        de druk? (Gebruik $v_l$ bij \qty{360}{\degreeCelsius}.)
  \item $m = \qty{0.10}{kg}$: stijgt of daalt het vloeistofniveau bij
        verwarming? Bij ongeveer welke temperatuur verdampt de laatste
        druppel (gebruik de tabel met $v_v$)?
  \item $m = \qty{0.31}{kg}$: beschrijf wat een waarnemer bij de
        meniscus ziet als de temperatuur \qty{374}{\degreeCelsius} nadert.
  \item Schets de drie verwarmingswegen in het $(v, P)$-diagram met de
        verzadigingsklok, en zeg aan welke kant van het \omterm{prop:b1:phase-changes:PTdiagram}{kritische punt} elk
        passeert.
  \item Hetzelfde vat van \qty{1.0}{L} met \qty{0.50}{kg} water wordt nu
        \emph{met} de lucht erin gesloten bij \qty{20}{\degreeCelsius},
        \qty{1.0}{bar}, en tot \qty{100}{\degreeCelsius} verwarmd: totale
        druk binnenin (Dalton), en de kracht op een deksel van \qty{100}{cm^2}.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De onderkoelde fles.}
\begin{enumerate}[resume]
  \item Een schone, stilstaande fles water is tot \qty{-8}{\degreeCelsius} afgekoeld
        zonder te bevriezen. Hoe heet die toestand, waarom is hij mogelijk,
        en wat gebeurt er als de fles wordt aangetikt? Waarom blijft de
        temperatuur dan precies op \qty{0}{\degreeCelsius} staan?
  \item Fractie van het water die bevriest (het proces is snel en dus
        adiabatisch).
  \item Geproduceerde entropie per kilogram; controleer het teken.
  \item De omgekeerde verrassing: een kop water die in een magnetron tot
