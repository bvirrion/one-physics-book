\chapter{Kinematica van een punt}\label{ch:b1:point-kinematics}

Het wagentje van een achtbaan gaat de looping in met \qty{90}{km/h}, vertraagt
tijdens het klimmen, hangt een ogenblik stil bovenin en dendert er weer uit;
de inzittenden voelen zich onderin in hun stoel geperst en bovenin bijna
gewichtloos. Die rit beschrijven --- waar het wagentje is, hoe
snel het gaat, hoe zijn \omterm{def:b1:point-kinematics:velocity}{snelheid} draait en verandert --- is kinematica,
en daar is meer voor nodig dan de $x$, $y$, $z$ van een rechte weg:
\omterm{def:b1:point-kinematics:cylindrical}{poolcoördinaten} voor alles wat draait, en een stelsel dat met het wagentje
zelf meereist. Dit hoofdstuk zet dat gereedschap op, bewijst de formules voor
\omterm{def:b1:point-kinematics:velocity}{snelheid} en \omterm{def:b1:point-kinematics:velocity}{versnelling} in elk daarvan, en past ze toe op de
bewegingen waar elk later hoofdstuk een beroep op doet: de eenparige cirkelbeweging, de
schroeflijn, de \omterm{prop:b1:point-kinematics:harmonic}{harmonische beweging} en de beweging langs een willekeurige kromme.

\section{Plaats, snelheid, versnelling}

\begin{definition}[Referentiestelsel; plaats; baan]\label{def:b1:point-kinematics:frame}
\index{referentiestelsel}\index{plaatsvector}\index{baan}\index{puntmassa}
Een \emph{referentiestelsel} is een star lichaam dat als vast wordt genomen (het
laboratorium, de aarde, een trein), samen met een klok. Een \emph{puntmassa}
$M$ --- een lichaam waarvan de afmeting er voor de bestudeerde beweging niet toe doet
--- wordt op elk moment vastgelegd door haar \emph{plaatsvector}
$\vect{OM}(t)$ vanuit een oorsprong $O$ die aan het stelsel vastzit; de kromme
die $M$ beschrijft is haar \emph{baan}. Beweging is altijd beweging
\emph{ten opzichte van een stelsel}: dezelfde reiziger is in rust in de trein
en in beweging in het station.
\end{definition}

\begin{definition}[Snelheid en versnelling]\label{def:b1:point-kinematics:velocity}
\index{snelheid}\index{versnelling}
De \emph{snelheid} en de \emph{versnelling} van $M$ in het stelsel zijn
\[
\vect v = \frac{\dd\vect{OM}}{\dd t}, \qquad
\vect a = \frac{\dd\vect v}{\dd t} = \frac{\dd^2\vect{OM}}{\dd t^2},
\]
afgeleiden van vectorfuncties: de afgeleide van een vector krijgt men
door zijn componenten af te leiden in een basis die \emph{vast ligt in het
stelsel}. De snelheidsvector raakt aan de \omterm{def:b1:point-kinematics:frame}{baan}
en wijst met de beweging mee; zijn norm $v$ is de \emph{baansnelheid}. Eenheden:
\unit{m/s}, \unit{m/s^2}.
\end{definition}

\begin{proposition}[Cartesische coördinaten]\label{prop:b1:point-kinematics:cartesian}
\index{cartesische coördinaten}
Met een vaste orthonormale basis $(\vect e_x, \vect e_y, \vect e_z)$ en
$\vect{OM} = x\,\vect e_x + y\,\vect e_y + z\,\vect e_z$ geldt
\[
\vect v = \dot x\,\vect e_x + \dot y\,\vect e_y + \dot z\,\vect e_z, \qquad
\vect a = \ddot x\,\vect e_x + \ddot y\,\vect e_y + \ddot z\,\vect e_z,
\]
waarin de punt $\dd/\dd t$ aangeeft.
\end{proposition}

\begin{proof}
De basisvectoren zijn constant; alleen de componenten veranderen.
\end{proof}

\begin{example}[Worp]\label{ex:b1:point-kinematics:projectile}
Vanuit $O$ afgeschoten met \omterm{def:b1:point-kinematics:velocity}{snelheid} $v_0$ onder een hoek $\alpha$ boven de
horizontaal heeft een bal de coördinaten (dynamica, volgend hoofdstuk)
$x = v_0\cos\alpha\,t$, $z = v_0\sin\alpha\,t - \tfrac12 gt^2$. Dan is
$\vect v = v_0\cos\alpha\,\vect e_x + (v_0\sin\alpha - gt)\vect e_z$,
$\vect a = -g\vect e_z$: de \omterm{def:b1:point-kinematics:velocity}{versnelling} is constant terwijl de
\omterm{def:b1:point-kinematics:velocity}{snelheid} draait; in het hoogste punt ($\dot z = 0$, $t = v_0\sin\alpha/g$) is de
\omterm{def:b1:point-kinematics:velocity}{snelheid} horizontaal en staat de \omterm{def:b1:point-kinematics:velocity}{versnelling} er loodrecht op.
$t$ elimineren geeft $z = x\tan\alpha - gx^2/(2v_0^2\cos^2\alpha)$, een
parabool.
\end{example}

\section{Cilindercoördinaten}

\begin{definition}[Cilindercoördinaten en lokale basis]\label{def:b1:point-kinematics:cylindrical}
\index{cilindercoördinaten}\index{poolcoördinaten}\index{lokale basis}
Het punt $M$ wordt vastgelegd door $r \geq 0$ (de afstand tot de $z$-as), de
hoek $\theta$ die zijn projectie op het $xy$-vlak met $\vect e_x$ maakt,
en $z$: $\vect{OM} = r\,\vect e_r + z\,\vect e_z$, waarin
\[
\vect e_r = \cos\theta\,\vect e_x + \sin\theta\,\vect e_y, \qquad
\vect e_\theta = -\sin\theta\,\vect e_x + \cos\theta\,\vect e_y,
\]
samen met $\vect e_z$ de \emph{lokale basis} $(\vect e_r, \vect e_\theta,
\vect e_z)$ vormen, orthonormaal en rechtsdraaiend, die \emph{met $M$ meebeweegt}. In het
vlak ($z$ vast) zijn dit de \emph{poolcoördinaten} $(r, \theta)$.
\end{definition}

\begin{lemma}[Afgeleiden van de lokale basis]\label{lem:b1:point-kinematics:basisderiv}
\[
\frac{\dd\vect e_r}{\dd t} = \dot\theta\,\vect e_\theta, \qquad
\frac{\dd\vect e_\theta}{\dd t} = -\dot\theta\,\vect e_r, \qquad
\frac{\dd\vect e_z}{\dd t} = \vect 0 .
\]
\end{lemma}

\begin{proof}
Leid de componenten af: $\dd\vect e_r/\dd t = \dot\theta(-\sin\theta\,
\vect e_x + \cos\theta\,\vect e_y) = \dot\theta\vect e_\theta$, en
evenzo voor $\vect e_\theta$. De afgeleide van een draaiende eenheidsvector staat
er loodrecht op en heeft als norm de hoeksnelheid.
\end{proof}

\begin{theorem}[Snelheid en versnelling in cilindercoördinaten]\label{thm:b1:point-kinematics:cylindrical}
\[
\vect v = \dot r\,\vect e_r + r\dot\theta\,\vect e_\theta + \dot z\,\vect e_z,
\qquad
\vect a = \big(\ddot r - r\dot\theta^2\big)\vect e_r
+ \big(r\ddot\theta + 2\dot r\dot\theta\big)\vect e_\theta + \ddot z\,\vect e_z .
\]
\end{theorem}

\begin{proof}
Leid $\vect{OM} = r\vect e_r + z\vect e_z$ af met het lemma:
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta + \dot z\vect e_z$.
Leid nogmaals af: $\ddot r\vect e_r + \dot r\dot\theta\vect e_\theta +
\dot r\dot\theta\vect e_\theta + r\ddot\theta\vect e_\theta - r\dot\theta^2
\vect e_r + \ddot z\vect e_z$; groepeer.
\end{proof}

\begin{corollary}[Cirkelbeweging]\label{cor:b1:point-kinematics:circular}
\index{cirkelbeweging}\index{centripetale versnelling}\index{hoeksnelheid}
Op een cirkel met straal $R$ ($r = R$, $z = 0$), met \emph{hoeksnelheid}
$\omega = \dot\theta$, geldt
\[
\vect v = R\omega\,\vect e_\theta, \qquad
\vect a = -R\omega^2\,\vect e_r + R\dot\omega\,\vect e_\theta
= -\frac{v^2}{R}\,\vect e_r + \frac{\dd v}{\dd t}\,\vect e_\theta .
\]
Bij een \emph{eenparige} cirkelbeweging ($\omega$ constant) is de \omterm{def:b1:point-kinematics:velocity}{versnelling}
zuiver \emph{centripetaal}, met norm $v^2/R = R\omega^2$, hoewel de
baansnelheid constant is.
\end{corollary}

\begin{proof}
Stel $\dot r = \ddot r = 0$ in de stelling; $v = R\omega$.
\end{proof}

\begin{example}[Twee cirkels]\label{ex:b1:point-kinematics:circles}
Een auto rijdt een rotonde met straal \qty{20}{m} rond met \qty{36}{km/h}
(\qty{10}{m/s}): $a = v^2/R = \qty{5.0}{m/s^2}$, een halve $g$, gericht
naar het middelpunt --- wat de banden moeten leveren. De maan
($R = \qty{3.84e8}{m}$, $T = \qty{27.3}{d}$): $\omega = 2\pi/T =
\qty{2.66e-6}{rad/s}$, $a = R\omega^2 = \qty{2.7e-3}{m/s^2}$, een
$3600$ste van $g$ aan het aardoppervlak, op zestig aardstralen --- de
vergelijking die Newton maakte (\cref{ch:b1:central-forces}).
\end{example}

\section{Bolcoördinaten}

\begin{definition}[Bolcoördinaten]\label{def:b1:point-kinematics:spherical}
\index{bolcoördinaten}
Het punt $M$ wordt vastgelegd door zijn afstand $r = OM$, de \emph{poolafstand}
$\theta \in [0, \pi]$ tussen $\vect{OM}$ en $\vect e_z$, en de
\emph{azimut} $\varphi$ van zijn projectie op het $xy$-vlak:
$\vect{OM} = r\,\vect e_r$ met
\[
\vect e_r = \sin\theta\cos\varphi\,\vect e_x + \sin\theta\sin\varphi\,\vect e_y + \cos\theta\,\vect e_z ;
\]
$\vect e_\theta$ (rakend aan de meridiaan, naar toenemende $\theta$)
en $\vect e_\varphi$ (rakend aan de parallel) maken de lokale
orthonormale basis compleet. Op aarde is $\theta$ gelijk aan $90^\circ$ min de
breedtegraad en $\varphi$ de lengtegraad.
\end{definition}

\begin{proposition}[Snelheid in bolcoördinaten]\label{prop:b1:point-kinematics:sphericalv}
\[
\vect v = \dot r\,\vect e_r + r\dot\theta\,\vect e_\theta + r\sin\theta\,\dot\varphi\,\vect e_\varphi .
\]
\end{proposition}

\begin{proof}
$\dd\vect e_r/\dd t = \dot\theta\,\vect e_\theta + \sin\theta\,\dot\varphi\,
\vect e_\varphi$: de eerste term is de draaiing in het meridiaanvlak
(zoals in het poolgeval), de tweede de draaiing om $\vect e_z$ met
\omterm{def:b1:point-kinematics:velocity}{snelheid} $\dot\varphi$ van een vector waarvan de afstand tot de as
$\sin\theta$ is. Vervolgens $\vect v = \dd(r\vect e_r)/\dd t$. De \omterm{def:b1:point-kinematics:velocity}{versnelling} is
dit jaar niet nodig.
\end{proof}

\section{De frenetbasis}

\begin{definition}[Kromlijnige abscis; frenetbasis]\label{def:b1:point-kinematics:frenet}
\index{kromlijnige abscis}\index{frenetbasis}\index{kromtestraal}
Langs een \omterm{def:b1:point-kinematics:frame}{baan} is de \emph{kromlijnige abscis} $s(t)$ de booglengte
vanaf een referentiepunt, algebraïsch geteld; $v = \dot s$. In
$M$ wijst de raakeenheidsvector $\vect T = \dd\vect{OM}/\dd s$ naar
toenemende $s$; de \emph{kromtestraal} $\rho > 0$ en de
normaaleenheidsvector $\vect N$, die naar de holle kant wijst (het
kromtemiddelpunt), worden gedefinieerd door
\[
\frac{\dd\vect T}{\dd s} = \frac{1}{\rho}\,\vect N .
\]
$(\vect T, \vect N)$ is de \emph{frenetbasis}; $\rho = \infty$ op een
rechte lijn, $\rho = R$ op een cirkel met straal $R$.
\end{definition}

\begin{theorem}[Versnelling in de frenetbasis]\label{thm:b1:point-kinematics:frenet}
\index{tangentiële versnelling}\index{normale versnelling}
\[
\vect v = v\,\vect T, \qquad
\vect a = \frac{\dd v}{\dd t}\,\vect T + \frac{v^2}{\rho}\,\vect N .
\]
De \emph{tangentiële} component $\dd v/\dd t$ meet hoe snel de
baansnelheid verandert; de \emph{normale} component $v^2/\rho$, altijd naar
het kromtemiddelpunt, hoe snel de richting draait. De beweging is
eenparig dan en slechts dan als $\vect a \perp \vect v$; zij is rechtlijnig dan en slechts dan als
$\vect a \parallel \vect v$.
\end{theorem}

\begin{proof}
$\vect v = (\dd\vect{OM}/\dd s)(\dd s/\dd t) = v\vect T$. Dan is $\vect a =
\dot v\vect T + v\,\dd\vect T/\dd t = \dot v\vect T + v(\dd\vect T/\dd s)\dot s
= \dot v\vect T + (v^2/\rho)\vect N$. Dat $\dd\vect T/\dd s \perp \vect T$
volgt uit het afleiden van $\vect T\cdot\vect T = 1$.
\end{proof}

\begin{example}[Remmen in een bocht]\label{ex:b1:point-kinematics:bend}
Een auto met \qty{90}{km/h} (\qty{25}{m/s}) in een bocht met straal \qty{100}{m}
remt met \qty{2.0}{m/s^2}: $a_T = \qty{-2.0}{m/s^2}$, $a_N = 625/100 =
\qty{6.25}{m/s^2}$, $a = \sqrt{4 + 39} = \qty{6.6}{m/s^2}$. Het normale
deel overheerst: een bocht is in de eerste plaats een draaiing, en de grip van de banden
(volgend hoofdstuk) begrenst $\sqrt{a_T^2 + a_N^2}$, en daarom remt men
\emph{vóór} de bocht.
\end{example}

\section{Standaardbewegingen}

\begin{proposition}[Eenparig versnelde rechtlijnige beweging]\label{prop:b1:point-kinematics:uarm}
\index{eenparig versnelde beweging}
Op een as, met $\ddot x = a$ constant, vanaf $x_0$ en $v_0$:
$v = v_0 + at$, $x = x_0 + v_0t + \tfrac12 at^2$, en
$v^2 - v_0^2 = 2a(x - x_0)$.
\end{proposition}

\begin{proof}
Twee keer integreren; elimineer $t$ tussen de eerste twee betrekkingen.
\end{proof}

\begin{proposition}[Schroeflijnbeweging]\label{prop:b1:point-kinematics:helix}
\index{schroeflijnbeweging}
$r = R$, $\theta = \omega t$, $z = v_zt$ (constanten $R$, $\omega$, $v_z$):
de baansnelheid $v = \sqrt{R^2\omega^2 + v_z^2}$ is constant, de \omterm{def:b1:point-kinematics:velocity}{versnelling}
$\vect a = -R\omega^2\vect e_r$ is centripetaal met constante norm, en de
\omterm{def:b1:point-kinematics:frenet}{kromtestraal} is $\rho = v^2/a = R(1 + v_z^2/R^2\omega^2) > R$.
\end{proposition}

\begin{proof}
\cref{thm:b1:point-kinematics:cylindrical} met $\dot r = 0$,
$\ddot\theta = 0$, $\ddot z = 0$; eenparige beweging, dus is $\vect a$ zuiver
normaal en $v^2/\rho = R\omega^2$.
\end{proof}

\begin{proposition}[Harmonische beweging]\label{prop:b1:point-kinematics:harmonic}
\index{harmonische beweging}
$x = A\cos(\omega t + \varphi)$: $\dot x = -A\omega\sin(\omega t + \varphi)$,
$\ddot x = -\omega^2 x$. De \omterm{def:b1:point-kinematics:velocity}{snelheid} loopt een kwart periode voor op de plaats
en de \omterm{def:b1:point-kinematics:velocity}{versnelling} is tegengesteld aan de plaats; $v_{\max} =
A\omega$, $a_{\max} = A\omega^2$. Dit is de projectie op een middellijn van
een eenparige cirkelbeweging met straal $A$ en hoeksnelheid $\omega$.
\end{proposition}

\begin{proof}
Afleiden; neem voor de projectie $x = A\cos\theta$ met $\theta =
\omega t + \varphi$.
\end{proof}

\begin{method}[Coördinaten kiezen]\label{met:b1:point-kinematics:choose}
\begin{itemize}
  \item Rechtlijnige of parabolische beweging, constante \omterm{def:b1:point-kinematics:velocity}{versnelling}:
        cartesisch.
  \item Beweging om een as of een punt (cirkels, spiralen, banen,
        draaischijven): cilinder- of \omterm{def:b1:point-kinematics:cylindrical}{poolcoördinaten}; de basis beweegt mee, dus gebruik
        \cref{thm:b1:point-kinematics:cylindrical} en nooit naïef
        de componenten afleiden.
  \item Een gegeven kromme (spoor, bocht, looping) wanneer de snelheidswet bekend is:
        de \omterm{def:b1:point-kinematics:frenet}{frenetbasis}, die ``hoe snel'' van ``welke kant op'' scheidt.
\end{itemize}
Wat de keuze ook is, $\vect v$ en $\vect a$ zijn stelselafhankelijke
vectoren, en een resultaat is pas volledig wanneer het stelsel erbij genoemd is.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:point-kinematics:1}
Een auto vertrekt uit rust met $x(t) = 2.0\,t^2$ (SI-eenheden). \omterm{def:b1:point-kinematics:velocity}{Snelheid} en
\omterm{def:b1:point-kinematics:velocity}{versnelling} op $t = \qty{3.0}{s}$; afgelegde weg in de eerste
\qty{5}{s}; tijd om \qty{100}{km/h} te bereiken.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:point-kinematics:2}
Een punt beweegt in een vlak met $r = \qty{2.0}{m}$ en $\theta = 3.0\,t$
(rad, SI). Geef $\vect v$ en $\vect a$ in de poolbasis, hun normen
en de aard van de beweging.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:point-kinematics:3}
Een punt op de evenaar ($R = \qty{6.37e6}{m}$, sterrendag
\qty{86164}{s}): \omterm{def:b1:point-kinematics:velocity}{snelheid} en \omterm{def:b1:point-kinematics:velocity}{versnelling} in het geocentrische stelsel.
Vergelijk de \omterm{def:b1:point-kinematics:velocity}{versnelling} met $g$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:point-kinematics:4}
Een bal wordt geworpen met $v_0 = \qty{12}{m/s}$ onder $\alpha = 40^\circ$; zijn
coördinaten zijn die van \cref{ex:b1:point-kinematics:projectile}.
Bepaal de tijd tot het hoogste punt, de maximale hoogte, de dracht en de
\omterm{def:b1:point-kinematics:velocity}{snelheid} bij de landing.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:point-kinematics:5}
Een fietser met \qty{10}{m/s} rijdt een bocht met straal \qty{25}{m} in en
versnelt met \qty{1.0}{m/s^2}. Geef $\vect a$ in de \omterm{def:b1:point-kinematics:frenet}{frenetbasis} en
zijn norm; en na \qty{5}{s} met dezelfde \omterm{def:b1:point-kinematics:velocity}{versnelling} in dezelfde bocht?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:point-kinematics:6}
Een punt volgt de spiraal $r = b\theta$, $\theta = \omega t$ ($b$,
$\omega$ constant). Bereken $\vect v$, de baansnelheid en $\vect a$ in de
poolbasis. Waar komt de term $2\dot r\dot\theta$ vandaan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:point-kinematics:7}
Schroeflijn: $R = \qty{2.0}{m}$, $\omega = \qty{1.0}{rad/s}$, $v_z = \qty{0.50}{m/s}$.
Baansnelheid, \omterm{def:b1:point-kinematics:velocity}{versnelling}, spoed van de schroeflijn, \omterm{def:b1:point-kinematics:frenet}{kromtestraal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:point-kinematics:8}
Een zuiger beweegt als $x = 0.10\cos(10t)$ (SI). Geef $v_{\max}$, $a_{\max}$,
de faserelaties tussen $x$, $v$, $a$, en de plaatsen waar
$\abs v$ en $\abs a$ het grootst zijn.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:point-kinematics:9}
Leid, uitgaand van $\vect e_r = \cos\theta\,\vect e_x + \sin\theta\,\vect e_y$,
de uitdrukkingen $\dd\vect e_r/\dd t$ en $\dd\vect e_\theta/\dd t$ af, en daarna de
poolvorm van \omterm{def:b1:point-kinematics:velocity}{snelheid} en \omterm{def:b1:point-kinematics:velocity}{versnelling} uit \cref{thm:b1:point-kinematics:cylindrical}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:point-kinematics:10}
Bepaal voor de worp van \cref{ex:b1:point-kinematics:projectile}
de \omterm{def:b1:point-kinematics:frenet}{kromtestraal} van de \omterm{def:b1:point-kinematics:frame}{baan} in het hoogste punt en in het
vertrekpunt (gebruik de normale component van $\vect g$). Welke is kleiner,
en waarom?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:point-kinematics:11}
Een wiel met straal $R$ rolt zonder slippen met \omterm{def:b1:point-kinematics:velocity}{snelheid} $v$; een punt van
zijn velg heeft coördinaten $x = vt - R\sin(vt/R)$, $y = R - R\cos(vt/R)$
(een cycloïde). Bereken zijn \omterm{def:b1:point-kinematics:velocity}{snelheid} en \omterm{def:b1:point-kinematics:velocity}{versnelling}; toon aan dat zijn \omterm{def:b1:point-kinematics:velocity}{snelheid}
nul is op het moment dat het de grond raakt en zijn \omterm{def:b1:point-kinematics:velocity}{versnelling} dan
$v^2/R$ omhoog bedraagt; bepaal zijn \omterm{def:b1:point-kinematics:velocity}{snelheid} bovenin.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:point-kinematics:12}
Iemand loopt naar buiten langs een straal van een draaischijf die met
constante $\omega$ draait, met constante \omterm{def:b1:point-kinematics:velocity}{snelheid} $u$ ten opzichte van de schijf:
$r = ut$, $\theta = \omega t$. Bereken $\vect a$ in de poolbasis,
benoem de twee termen en werk ze uit voor $u = \qty{1.0}{m/s}$,
$\omega = \qty{1.0}{rad/s}$ op $t = \qty{2.0}{s}$. Welke term heeft geen
tegenhanger voor iemand die over de grond loopt?
\end{exercise}

\section{Probleem: De looping van de achtbaan}

\begin{problem}[{Weekendopgave --- een wagentje gaat met negentig
kilometer per uur een verticale looping in: waar het vertraagt, waarheen zijn
versnelling wijst, hoeveel $g$ de inzittenden voelen, en waarom moderne loopings
geen cirkels zijn}]
\label{pb:b1:point-kinematics:1}
Een wagentje, opgevat als een punt $M$, rijdt op een verticale cirkelvormige looping met
straal $R = \qty{10}{m}$ en middelpunt $C$. Zijn plaats wordt gegeven door de
hoek $\theta$ tussen $\vect{CM}$ en de neerwaartse verticaal ($\theta =
0$ onderin, $\pi$ bovenin). Wrijving wordt verwaarloosd, en de
snelheidswet (afgeleid in \cref{ch:b1:work-and-energy}) luidt
\[
v^2 = v_0^2 - 2gR(1 - \cos\theta),
\]
met $v_0 = \qty{25}{m/s}$ de \omterm{def:b1:point-kinematics:velocity}{snelheid} onderin; $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Deel I --- Kinematica op de cirkel.}
\begin{enumerate}
  \item Schrijf in \omterm{def:b1:point-kinematics:cylindrical}{poolcoördinaten} met middelpunt $C$ (met $\vect e_r$ van $C$
        naar $M$) $\vect v$ en $\vect a$ op voor beweging op de cirkel.
  \item Benoem de frenetvectoren $\vect T$ en $\vect N$ in termen van
        $\vect e_r$ en $\vect e_\theta$, en de \omterm{def:b1:point-kinematics:frenet}{kromtestraal}.
  \item Bereken de \omterm{def:b1:point-kinematics:velocity}{snelheid} bovenin en in de zijpunten
        $\theta = \pi/2$ en $3\pi/2$.
  \item Leid de snelheidswet naar de tijd af om aan te tonen dat
        de \omterm{thm:b1:point-kinematics:frenet}{tangentiële versnelling} $a_T = -g\sin\theta$ is. Duid
        haar teken op de weg omhoog en op de weg omlaag.
  \item Druk de \omterm{thm:b1:point-kinematics:frenet}{normale versnelling} $a_N(\theta)$ uit en bereken haar
        onderin, in de zijpunten en bovenin.
  \item Geef de norm van $\vect a$ in de vier punten en de hoek die zij
        bovenin met de verticaal maakt.
  \item In welke punten staat $\vect a$ loodrecht op $\vect v$? Evenwijdig?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Hoeksnelheden en tijdsduur.}
\begin{enumerate}[resume]
  \item Druk $\dot\theta$ uit als functie van $\theta$ en bereken haar
        onderin en bovenin.
  \item Toon aan dat $\ddot\theta = -(g/R)\sin\theta$ (de vergelijking van een
        slinger) en becommentarieer dat.
  \item Bereken ter vergelijking de periode van kleine trillingen van een
        wiskundige slinger met lengte $R$ (\cref{ex:b1:units-dimensions:pendulum}
        gaf de vorm; $k = 2\pi$).
  \item Schat de tijd om de looping rond te gaan met het gemiddelde van
        de snelheden in de vier referentiepunten.
  \item De kleinste \omterm{def:b1:point-kinematics:velocity}{snelheid} bovenin waarbij het wagentje op de \omterm{def:b1:point-kinematics:frame}{baan} blijft
        (volgend hoofdstuk) is $\sqrt{gR}$. Wordt daaraan voldaan? Welke minimale
        $v_0$ zou nodig zijn?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Wat de inzittenden voelen.}
Het schijnbare gewicht per massa-eenheid dat een inzittende voelt is $\vect a - \vect g$
(hier aangenomen, afgeleid in \cref{ch:b1:newton-dynamics}); de norm daarvan in
eenheden van $g$ is de ``$g$-belasting''.
\begin{enumerate}[resume]
  \item Bereken de $g$-belasting onderin.
  \item Bereken haar bovenin en zeg welke kant de inzittende op wordt
        gedrukt (stoel of beugel).
  \item Bereken haar in de zijpunten.
  \item Inzittenden verdragen kortstondig ongeveer $4g$. Haalt deze looping dat? Waar
        zit het probleem?
  \item Toon aan dat voor een cirkelvormige looping die net snel genoeg wordt ingereden
        om bovenin rond te komen (daar schijnbare gewichtloosheid), de $g$-belasting
        onderin noodzakelijk $6g$ is, wat $R$ ook is.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De clothoïdelooping.}
Moderne loopings zijn ``druppels'': de \omterm{def:b1:point-kinematics:frenet}{kromtestraal} is groot onderin
en klein bovenin.
\begin{enumerate}[resume]
  \item Leg met de frenetvorm van $\vect a$ uit waarom het variëren van $\rho$
        langs de \omterm{def:b1:point-kinematics:frame}{baan} $a_N$ verandert zonder de snelheidswet te veranderen
        (die alleen van de hoogte afhangt).
  \item Kies $\rho_{\text{onder}}$ zo dat de $g$-belasting
        onderin $3g$ is bij $v_0 = \qty{25}{m/s}$.
  \item Bovenin is de hoogte nog altijd $2R = \qty{20}{m}$ boven de
        onderkant. Kies $\rho_{\text{boven}}$ zo dat de $g$-belasting daar
        $1.5g$ is.
  \item Een clothoïde heeft een kromming die evenredig is met de booglengte,
        $1/\rho = s/A^2$. Hoe groeit $a_N$ met de tijd langs zo'n inloop bij
        vrijwel constante \omterm{def:b1:point-kinematics:velocity}{snelheid}, en waarom is dat zachter
        voor de nek dan een cirkel (waar $a_N$ bij het inrijden springt)?
  \item Schets de resulterende vorm van de looping kwalitatief en vat in
        twee zinnen samen waarom zij de cirkel heeft vervangen.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Gezien vanaf de grond.}
Een \omterm{def:b1:optical-instruments:camera}{camera} op grondniveau, \qty{50}{m} van het middelpunt van de looping in haar
vlak, volgt het wagentje.
\begin{enumerate}[resume]
  \item Met welke hoeksnelheid (rad/s) moet de \omterm{def:b1:optical-instruments:camera}{camera} draaien wanneer het
        wagentje bovenin is en horizontaal met \qty{15}{m/s} beweegt?
        (\omterm{def:b1:point-kinematics:cylindrical}{Poolcoördinaten} vanuit de \omterm{def:b1:optical-instruments:camera}{camera}.)
  \item Waarom volgt een \omterm{def:b1:optical-instruments:camera}{camera} die met constante \omterm{def:b1:point-kinematics:velocity}{snelheid} draait een eenparige
        cirkelbeweging slecht?
  \item Vat de les van dit hoofdstuk uit de looping samen: welk
        coördinatenstelsel welke vraag beantwoordde.
\end{enumerate}
\end{problem}
