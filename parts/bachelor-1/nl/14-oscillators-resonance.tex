\chapter{Mechanische oscillatoren: demping en resonantie}\label{ch:b1:oscillators-resonance}

Duw een kinderschommel op het verkeerde moment aan en er gebeurt weinig; duw
één keer per periode, zacht, en binnen een tiental slagen gilt het kind
boven in de boog. Een wasmachine schudt hevig bij één bepaald toerental
van haar centrifuge en is daarboven stil. Een wijnglas zingt wanneer een natte
vinger met precies het juiste tempo langs de rand cirkelt, en springt aan
scherven wanneer een zangeres diezelfde toon luid genoeg aanhoudt.
Elk daarvan is een \emph{resonantie}: een systeem met een eigen
frequentie, aangedreven dicht bij die frequentie, hoopt cyclus na cyclus
energie op tot de wrijving haar even snel afvoert als zij binnenkomt. Dit
hoofdstuk neemt de mechanische oscillator --- vrij, gedempt en dan gedwongen
--- en leidt de resonantiekrommen af waarvan de elektrische tweelingen in
\cref{ch:b1:sinusoidal-impedance} verschenen.

\section{De harmonische oscillator}

\begin{definition}[Harmonische oscillator]\label{def:b1:oscillators-resonance:harmonic}
\index{harmonische oscillator}\index{eigenfrequentie}
Een systeem waarvan de uitwijking $x$ uit een stabiel evenwicht voldoet aan
\[
\ddot x + \omega_0^2\,x = 0
\]
is een \emph{harmonische oscillator} met \emph{\omterm{def:b1:transient-regimes:secondorder}{eigenhoekfrequentie}}
$\omega_0$ (periode $T_0 = 2\pi/\omega_0$). De oplossing $x = A\cos(\omega_0t
+ \varphi)$ heeft een \omterm{def:b1:wave-propagation:signal}{amplitude} en een fase die door de beginvoorwaarden zijn vastgelegd
en een periode die niet van de \omterm{def:b1:wave-propagation:signal}{amplitude} afhangt (\emph{isochronie}).
\end{definition}

\begin{proposition}[Waar harmonische oscillatoren vandaan komen]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item Massa $m$ aan een veer $k$: $\omega_0^2 = k/m$ (\omterm{prop:b1:newton-dynamics:springstring}{wet van Hooke}).
  \item \omterm{ex:b1:newton-dynamics:pendulum}{Wiskundige slinger} met lengte $\ell$, kleine hoeken: $\omega_0^2 = g/\ell$.
  \item Elke \omterm{def:b1:point-kinematics:frame}{puntmassa} vlak bij een stabiel evenwicht van een potentiaal $E_p$:
        $\omega_0^2 = E_p''(x_0)/m$ (\cref{prop:b1:work-and-energy:stability}).
  \item LC-kring: $\omega_0^2 = 1/LC$, met $q \leftrightarrow x$,
        $L \leftrightarrow m$, $1/C \leftrightarrow k$.
\end{itemize}
De energie $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ is constant en klotst twee keer per periode heen en weer tussen
kinetische en potentiële vorm.
\end{proposition}

\begin{proof}
De tweede wet van Newton met $F = -kx$; de slingervergelijking
$\ddot\theta + (g/\ell)\sin\theta = 0$ met $\sin\theta \approx \theta$;
de formule van Taylor in het minimum van $E_p$; de maasregel van de
LC-kring. Energie: afleiden en de vergelijking gebruiken.
\end{proof}

\begin{example}[Een dobberende cilinder]\label{ex:b1:oscillators-resonance:bob}
Een verticale cilinder met doorsnede $S$ drijft met een diepte $h$
ondergedompeld ($mg = \rho_wgSh$, opwaartse kracht uit \cref{ch:b1:fluid-statics}).
Over $x$ omlaag geduwd ondervindt hij een extra opwaartse kracht $\rho_wgSx$:
$m\ddot x = -\rho_wgSx$, dus $\omega_0^2 = \rho_wgS/m = g/h$ --- hetzelfde
als een slinger met lengte $h$. Een boei die \qty{1}{m} diep ligt dobbert met een
periode van \qty{2}{s}, wat haar massa ook is.
\end{example}

\section{De gedempte oscillator}

\begin{proposition}[Gedempte oscillator]\label{prop:b1:oscillators-resonance:damped}
\index{gedempte oscillator}\index{kwaliteitsfactor!mechanische}
Met een viskeuze wrijving $-\alpha\dot x$ wordt de vergelijking
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
de canonieke tweede-ordevorm van \cref{ch:b1:transient-regimes}: voor
$Q > \tfrac12$ een pseudoperiodieke uitdemping $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$ met $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$; de
energie neemt af als $\eu^{-2t/\tau}$ en
\[
Q = 2\pi\,\frac{\text{opgeslagen energie}}{\text{energie verloren per periode}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
Deel $m\ddot x + \alpha\dot x + kx = 0$ door $m$. Voor de energie: $E
\propto A^2\eu^{-2t/\tau}$ verliest per periode de fractie $2T/\tau = 2\pi/Q$
wanneer die klein is.
\end{proof}

\begin{example}[Uitklinktijden]\label{ex:b1:oscillators-resonance:ringing}
Een stemvork van \qty{440}{Hz} klinkt tien seconden na de aanslag nog:
haar \omterm{def:b1:wave-propagation:signal}{amplitude} is tot ongeveer $5\%$ gezakt in $3\tau = \qty{10}{s}$, dus
$\tau = \qty{3.3}{s}$ en $Q = \omega_0\tau/2 \approx 4600$. De vering van een
auto ($Q \approx 0.7$) komt binnen één periode tot rust; een kinderschommel
($Q \approx 20$) verliest een derde van haar energie per slag en heeft elke keer
een duw nodig.
\end{example}

\section{Gedwongen trillingen en resonantie}

\begin{theorem}[Sinusvormig aangedreven oscillator]\label{thm:b1:oscillators-resonance:forced}
\index{gedwongen trillingen}\index{resonantie!amplitude}\index{resonantie!snelheid}
Aangedreven door een kracht $F_0\cos\omega t$ komt de \omterm{prop:b1:oscillators-resonance:damped}{gedempte oscillator}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
na een overgang met duur $\sim\tau$ in het \omterm{prop:b1:sinusoidal-impedance:forced}{gedwongen
regime} $x = X\cos(\omega t + \varphi)$ terecht, met, schrijvend $x_r = \omega/\omega_0$
en $X_0 = F_0/k$ voor de statische uitwijking,
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
De \emph{\omterm{def:b1:wave-propagation:signal}{amplitude}} $X$ piekt, voor $Q > 1/\sqrt2$, bij $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$ met $X_{\max} \approx QX_0$; de \emph{\omterm{def:b1:wave-propagation:signal}{amplitude} van de
snelheid} $V = \omega X$ piekt precies bij $\omega_0$, waar $V_{\max} =
F_0/\alpha$, en haar resonantiekromme
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
heeft bandbreedte $\Delta\omega = \omega_0/Q$. Bij $\omega_0$ loopt de uitwijking
$\pi/2$ achter op de kracht en is de snelheid ermee in fase.
\end{theorem}

\begin{proof}
Complexe methode (\cref{ch:b1:sinusoidal-impedance}): $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$, dus $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$; deel teller en noemer door
$\omega_0^2 = k/m$. Modulus en argument volgen; het maximum van $X$ is
bepaald in \cref{prop:b1:sinusoidal-impedance:overvoltage} (dezelfde
functie). Snelheid: $\underline V = j\omega\underline X$ geeft $V = \omega X$;
$(1 - x_r^2)/x_r = 1/x_r - x_r$ schrijven maakt van de noemer
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$, dus de RLC-vorm. Bij $x_r = 1$ is
$\underline X = -jQX_0$: een achterstand van $\pi/2$.
\end{proof}

\begin{remark}[Twee resonanties, één kring]\label{rem:b1:oscillators-resonance:analogy}
\index{elektromechanische analogie}
De overeenkomst $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$,
$F \leftrightarrow e$ beeldt de aangedreven oscillator af op de RLC in serie van
\cref{ch:b1:sinusoidal-impedance}: de snelheidsresonantie is de
stroomresonantie (altijd bij $\omega_0$, bandbreedte $\omega_0/Q$), de
amplituderesonantie is de ladingsresonantie, dus die van de condensatorspanning (iets
onder $\omega_0$, afwezig bij lage $Q$). Het opgenomen vermogen, $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$, piekt bij $\omega_0$ met de waarde $F_0^2/2\alpha$,
en de halfvermogenpunten zijn de bandbreedte van de snelheid --- precies zoals bij
de RLC.
\end{remark}

\begin{example}[Resonantiesnelheden]\label{ex:b1:oscillators-resonance:road}
De vering van een auto heeft $f_0 \approx \qty{1.3}{Hz}$ (\cref{pb:b1:transient-regimes:1}).
Op een weg met ribbels om de $\lambda = \qty{10}{m}$ is de aandrijffrequentie
$v/\lambda$: resonantie bij $v = \lambda f_0 = \qty{13}{m/s}$,
ongeveer \qty{47}{km/h} --- de snelheid waarbij een wasbordweg de
auto het hardst schudt, en de reden dat er een demper met $Q \approx 0.7$ wordt gemonteerd:
met $Q = 5$ zou de carrosserie met vijf keer de \omterm{def:b1:wave-propagation:signal}{amplitude} van de weg
wippen. Boven de resonantie ($x_r \gg 1$) daalt de \omterm{def:b1:wave-propagation:signal}{amplitude} als
$X_0/x_r^2$: de carrosserie ``zweeft'' over snelle hobbels --- isolatie, de andere
kant van de resonantie.
\end{example}

\section{Voetpuntexcitatie: isolatie en de seismometer}

\begin{proposition}[Oscillator geschud door zijn steun]\label{prop:b1:oscillators-resonance:base}
\index{voetpuntexcitatie}\index{trillingsisolatie}\index{seismometer}
Laat de steun van een stelsel massa--veer--demper over $x_g(t)$ bewegen (de weg
onder een auto, de grond onder een seismometer), zij $u$ de
uitwijking van de massa ten opzichte van de steun en $x = x_g + u$ haar
absolute uitwijking (vanuit het evenwicht). Dan geldt
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
en voor $x_g = X_g\cos\omega t$, met $x_r = \omega/\omega_0$:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
Voor $x_r \gg 1$ geldt $U \to X_g$ (de massa blijft in de ruimte stilstaan --- een
\emph{seismometer} leest de uitwijking van de grond af) en $X \to 0$ (de
massa is van het schudden \emph{geïsoleerd}); voor $x_r \ll 1$ geldt $U \approx
\ddot x_g/\omega_0^2$ (een \emph{versnellingsopnemer}) en $X \approx X_g$ (de massa
volgt de steun).
\end{proposition}

\begin{proof}
De veer en de demper werken op de relatieve beweging: $m\ddot x = -ku -
\alpha\dot u$, en $\ddot x = \ddot x_g + \ddot u$. In complexe vorm geldt
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$, waaruit
$U/X_g$ volgt; vervolgens $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$.
\end{proof}

\begin{example}[Waarom machines op zachte voetjes staan]\label{ex:b1:oscillators-resonance:mounts}
Een motor van \qty{50}{Hz} op rubberen voetjes afgestemd op $f_0 = \qty{10}{Hz}$
($x_r = 5$, $Q \approx 2$): $X/X_g = \sqrt{(1 + 6.25)/(576 + 6.25)} = 0.11$
--- negen tiende van de trilling blijft uit de vloer. De voetjes moeten
\emph{zachter} zijn dan de intuïtie ingeeft: ze verstijven verhoogt $f_0$
naar $50$ Hz toe en vernietigt de isolatie. Dezelfde formule, andersom
gelezen, is de auto van \cref{ex:b1:oscillators-resonance:road} op een
wasbordweg.
\end{example}

\section{Energie: hoe een resonantie opbouwt}

\begin{proposition}[Energiebalans in het gedwongen regime]\label{prop:b1:oscillators-resonance:energy}
In het stationaire \omterm{prop:b1:sinusoidal-impedance:forced}{gedwongen regime} is het \omterm{thm:b1:sinusoidal-impedance:power}{gemiddelde vermogen} dat de aandrijvende
kracht levert gelijk aan het \omterm{thm:b1:sinusoidal-impedance:power}{gemiddelde vermogen} dat de wrijving dissipeert, $\tfrac12\alpha V^2$.
Bij resonantie is de opgeslagen energie $\tfrac12 mV_{\max}^2$ gelijk aan $Q/2\pi$ keer
de per periode toegevoerde energie: de oscillator hoopt de arbeid van de aandrijving
over ongeveer $Q$ cycli op voordat de verliezen de toevoer evenaren, en dat is ook
het aantal cycli ($\sim\tau/T$) dat de overgang duurt.
\end{proposition}

\begin{proof}
Middel de energiestelling over een periode: $\langle\dd E/\dd t\rangle = 0$
in de stationaire toestand, dus $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$. Bij resonantie is $V_{\max} = F_0/\alpha$, de opgeslagen energie
$\tfrac12 m(F_0/\alpha)^2$ en de per periode toegevoerde energie $\tfrac12\alpha(F_0/\alpha)^2T$;
de verhouding is $m/(\alpha T) = Q/2\pi$.
\end{proof}

\begin{example}[De schommel]\label{ex:b1:oscillators-resonance:swing}
Een schommel met lengte \qty{2.5}{m} ($T_0 = \qty{3.2}{s}$, $Q \approx 20$) krijgt
elke periode een duw van \qty{50}{N} over \qty{0.5}{m}: \qty{25}{J} per
cyclus. Zij groeit tot het verlies per periode, $2\pi E/Q$, gelijk is aan
\qty{25}{J}: $E = \qty{80}{J}$; bij een kind van \qty{30}{kg} geeft $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ dat $\theta_{\max} \approx 27^\circ$.
Bij elk ander ritme zou de arbeid van de duwen van teken wisselen en
gemiddeld niets opleveren.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:1}
Een massa van \qty{0.20}{kg} aan een veer met veerconstante \qty{80}{N/m} trilt
met \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{5.0}{cm}. \omterm{def:b1:wave-propagation:signal}{Hoekfrequentie}, frequentie, periode,
grootste snelheid, totale energie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:2}
Periode van een slinger van \qty{1.0}{m} op aarde en op de maan
($g = \qty{1.62}{m/s^2}$). Wat gebeurt er met de periode van massa en veer op de
maan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:3}
Een oscillator heeft $f_0 = \qty{2.0}{Hz}$ en $Q = 50$. Uitdooftijd $\tau$
van de \omterm{def:b1:wave-propagation:signal}{amplitude}; tijd waarin de energie halveert; aantal trillingen
voordat de \omterm{def:b1:wave-propagation:signal}{amplitude} tot $5\%$ is gezakt.
\end{exercise}

\section{Probleem: De seismograaf}

\begin{problem}[{Weekendopgave --- een massa hangt aan een veer in
een kast die aan de grond is verankerd; de grond schudt: wat schrijft de pen
op, waarom hetzelfde instrument bij hoge frequentie uitwijking en bij lage
frequentie versnelling meet, en waarom een seismometer voor lange perioden
niet gewoon een lange veer kan zijn}]
\label{pb:b1:oscillators-resonance:1}
Een massa $m = \qty{1.0}{kg}$ hangt aan een veer met veerconstante $k$ binnen
een star frame dat aan de grond is bevestigd; een demper oefent $-\alpha\dot u$ uit, waarin
$u$ de uitwijking van de massa ten opzichte van het frame is, gemeten
vanaf haar evenwichtsstand. De grond, en dus het frame, beweegt
verticaal over $x_g(t)$. Het stelsel van de verre sterren is inertiaal.

\medskip
\noindent\textbf{Deel I --- De vergelijking.}
\begin{enumerate}
  \item Schrijf de absolute plaats van de massa als $x = x_g + u +
        \text{const}$, som de krachten op de massa op (in het inertiaalstelsel)
        en toon aan dat
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item Breng haar in canonieke vorm; benoem $\omega_0$ en $Q$.
  \item Het instrument wordt gebouwd met $f_0 = \qty{0.50}{Hz}$ en $Q = 0.70$:
        bereken $k$ en $\alpha$.
  \item Een constante grondversnelling $a_g$ (bijvoorbeeld doordat het frame
        kantelt) geeft welke stationaire $u$? Duid dat.
  \item Wat registreert de pen die aan de massa vastzit en op een trommel
        schrijft die aan het frame vastzit?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Harmonische grondbeweging.}
De grond beweegt als $x_g = X_g\cos\omega t$; schrijf $x_r = \omega/\omega_0$.
\begin{enumerate}[resume]
  \item Toon met \omterm{def:b1:sinusoidal-impedance:complex}{complexe amplitudes} aan dat $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$.
  \item Leid $U/X_g$ af als functie van $x_r$ en $Q$.
  \item Limiet $x_r \gg 1$: toon aan dat $U \to X_g$ en leg fysisch uit wat
        de massa doet (wat meet het instrument?).
  \item Limiet $x_r \ll 1$: toon aan dat $U \approx x_r^2X_g = a_g/\omega_0^2$ met
        $a_g$ de \omterm{def:b1:wave-propagation:signal}{amplitude} van de grondversnelling: wat is het instrument
        nu?
  \item Waarde bij $x_r = 1$; waarom is $Q \approx 0.7$ een verstandige keuze?
  \item Schets $U/X_g$ tegen $x_r$ (logaritmische schalen) voor $Q = 0.7$ en
        $Q = 5$.
  \item Fase van $u$ ten opzichte van $x_g$ bij hoge frequentie; duid dat.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Ontwerpen voor aardbevingen.}
Verre aardbevingen leveren oppervlaktegolven van \qty{0.05}{} tot \qty{0.1}{Hz}
met \omterm{def:b1:wave-propagation:signal}{amplitudes} van millimeters; plaatselijke trillingen halen \qty{5}{} tot \qty{10}{Hz}.
\begin{enumerate}[resume]
  \item Om golven van \qty{0.05}{Hz} als uitwijkingen te registreren wil men
        $f_0 \ll \qty{0.05}{Hz}$, zeg \qty{0.02}{Hz}. Welke statische
        uitrekking zou de veer onder het gewicht van de massa hebben? Becommentarieer.
  \item Een \omterm{ex:b1:newton-dynamics:pendulum}{wiskundige slinger} met dezelfde frequentie: welke lengte?
  \item Werkelijke instrumenten voor lange perioden gebruiken een bijna horizontale
        ``tuinhekslinger'', of elektronische krachtterugkoppeling. Leg
        in één zin uit wat elk bereikt.
  \item Bereken voor het instrument van \qty{0.50}{Hz} de statische uitrekking:
        haalbaar?
  \item Welke $U$ geeft daarmee een golf van \qty{0.10}{Hz} met \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm}?
        In welk regime zit het instrument?
  \item Een plaatselijke trilling van \qty{5.0}{Hz} met \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm}: $U$?
        Regime?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De versnellingsopnemer in een telefoon.}
Een microbewerkte massa op siliciumveren heeft $f_0 = \qty{1.0}{kHz}$ en
$Q = 0.7$.
\begin{enumerate}[resume]
  \item Een schudding van \qty{10}{Hz} met versnellingsamplitude $g$: bereken
        $U$. Hoe wordt zo'n uitwijking uitgelezen?
  \item Tot welke frequentie blijft de aflezing binnen $10\%$ van $a_g/\omega_0^2$?
  \item Welke \omterm{def:b1:transient-regimes:firstorder}{tijdconstante} $\tau = 2Q/\omega_0$ beheerst haar reactie op een
        schok, en waarom telt dat voor het opmerken van een val?
  \item De $\tau$ van de seismometer: waarde en gevolg na een plotselinge
        stap van de grond.
  \item Waarom wordt voor beide instrumenten kritische (of bijna kritische)
        demping gekozen in plaats van een hoge $Q$?
  \item De telefoon wordt precies op \qty{1.0}{kHz} geschud: met welke factor
        zit de aflezing ernaast?
  \item Vat samen: één apparaat, twee regimes, bepaald door welke ene
        vergelijking.
\end{enumerate}
\end{problem}
