\chapter{Optische instrumenten en het oog}\label{ch:b1:optical-instruments}

Vanaf de achterste rij van een concertzaal trekt een verrekijker het gezicht
van de zangeres tot op armlengte; een \omterm{def:b1:optical-instruments:microscope}{microscoop} maakt van een druppel
slootwater een dierentuin; een telefoon zo dik als een vinger fotografeert de
Melkweg. Elk van deze instrumenten is een handvol dunne lenzen, geschikt met
één doel: het \omterm{def:b1:optical-instruments:eye}{oog}, of een sensor, een beeld voorleggen dat groter, helderder
of scherper is dan het \omterm{def:b1:optical-instruments:eye}{oog} alleen kan vormen. Het hoofdstuk begint dus bij het
\omterm{def:b1:optical-instruments:eye}{oog} zelf --- wat het wel en niet kan --- en bouwt daarna de \omterm{def:b1:optical-instruments:G}{loep}, de
\omterm{def:b1:optical-instruments:microscope}{microscoop}, de kijker en de \omterm{def:b1:optical-instruments:camera}{camera} op uit de conjugatiebetrekking van het
vorige hoofdstuk.

\section{Het oog}

\begin{definition}[Gereduceerd oog]\label{def:b1:optical-instruments:eye}
\index{oog}\index{gereduceerd oog}\index{netvlies}\index{accommodatie}
Het \emph{gereduceerde oog} modelleert het hoornvlies en de ooglens als één
dunne \omterm{def:b1:mirrors-thin-lenses:lens}{convergerende lens} met \emph{veranderlijke} \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand}, en het
\emph{netvlies} als een scherm op een vaste afstand $d \approx \qty{17}{mm}$
erachter. Spieren veranderen de kromming van de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} om het beeld van het
bekeken voorwerp op het netvlies te houden: dat is \emph{accommodatie}. De
iris vóór de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} is een regelbare opening met diameter \qty{2}{} tot
\qty{8}{mm}, de \emph{pupil}.
\end{definition}

\begin{definition}[Verpunt, nabijheidspunt]\label{def:b1:optical-instruments:punctum}
\index{verpunt}\index{nabijheidspunt}\index{accommodatiebreedte}
Het \emph{verpunt} (punctum remotum) is het voorwerpspunt dat zonder
accommoderen scherp wordt gezien; het \emph{nabijheidspunt} (punctum proximum)
het dichtstbijzijnde punt dat bij maximale \omterm{def:b1:optical-instruments:eye}{accommodatie} scherp wordt gezien.
Voor een normaal (emmetroop) \omterm{def:b1:optical-instruments:eye}{oog} ligt het verpunt op oneindig en het
nabijheidspunt op $d_m \approx \qty{25}{cm}$, de conventionele
\emph{afstand van duidelijk zien}. Het verschil in \omterm{def:b1:mirrors-thin-lenses:lens}{vergentie} tussen beide is de
\emph{accommodatiebreedte}: ongeveer \qty{4}{\delta} voor een jonge
volwassene, afnemend met de leeftijd.
\end{definition}

\begin{example}[De vergentie van het oog]\label{ex:b1:optical-instruments:vergence}
Ver voorwerp: beeld op $d = \qty{17}{mm}$ betekent $f' = \qty{17}{mm}$,
$V = 1/0.017 = \qty{59}{\delta}$. Voorwerp op \qty{25}{cm}:
$V = 1/0.017 + 1/0.25 = 59 + 4 = \qty{63}{\delta}$. Vier dioptrieën
\omterm{def:b1:optical-instruments:eye}{accommodatie} op zestig: de ooglens voegt maar enkele procenten toe aan het
werk van het hoornvlies, maar die paar procent zijn alles om te lezen.
\end{example}

\begin{proposition}[Afwijkingen en correcties]\label{prop:b1:optical-instruments:defects}
\index{bijziendheid}\index{verziendheid}\index{ouderdomsverziendheid}
Een \emph{bijziend} \omterm{def:b1:optical-instruments:eye}{oog} is te sterk convergerend: zijn \omterm{def:b1:optical-instruments:punctum}{verpunt} ligt op een
eindige afstand $D_r$; een \omterm{def:b1:mirrors-thin-lenses:lens}{divergerende lens} met \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand}
$f' = -D_r$ (vlak voor het \omterm{def:b1:optical-instruments:eye}{oog} gedragen) beeldt oneindig af in het \omterm{def:b1:optical-instruments:punctum}{verpunt} en
herstelt het zicht in de verte. Een \emph{verziend} \omterm{def:b1:optical-instruments:eye}{oog} is niet convergerend
genoeg: zijn \omterm{def:b1:optical-instruments:punctum}{verpunt} is virtueel (achter het \omterm{def:b1:optical-instruments:eye}{oog}) en zijn \omterm{def:b1:optical-instruments:punctum}{nabijheidspunt} ligt
te ver; een \omterm{def:b1:mirrors-thin-lenses:lens}{convergerende lens} corrigeert het. \emph{Ouderdomsverziendheid}
is het verlies van \omterm{def:b1:optical-instruments:eye}{accommodatie} met de jaren: het \omterm{def:b1:optical-instruments:punctum}{nabijheidspunt} wijkt terug
en lezen vraagt convergerende lenzen, waar het \omterm{def:b1:optical-instruments:punctum}{verpunt} ook ligt.
\end{proposition}

\begin{proof}
Een \omterm{def:b1:mirrors-thin-lenses:lens}{lens} vlak voor het \omterm{def:b1:optical-instruments:eye}{oog} vormt met het \omterm{def:b1:optical-instruments:eye}{oog} een systeem waarvan het voorwerp
moet zijn wat het \omterm{def:b1:optical-instruments:eye}{oog} kan zien: de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} moet van een voorwerp op oneindig een
\omterm{def:b1:mirrors-thin-lenses:image}{virtueel beeld} in het \omterm{def:b1:optical-instruments:punctum}{verpunt} maken, $\overline{OA'} = -D_r$ voor
$\overline{OA} = -\infty$, dus $f' = \overline{OA'} = -D_r$. De overige
gevallen zijn hetzelfde argument met de bijbehorende punten.
\end{proof}

\begin{definition}[Hoekgrootte en oplossend vermogen]\label{def:b1:optical-instruments:angular}
\index{hoekgrootte}\index{hoekoplossend vermogen}
De \emph{hoekgrootte} van een voorwerp met hoogte $h$ op afstand $D$ is
$\theta \approx h/D$ (kleine hoeken). Het \omterm{def:b1:optical-instruments:eye}{oog} onderscheidt twee punten
alleen als hun hoekafstand groter is dan zijn \emph{hoekoplossend vermogen},
ongeveer $\epsilon \approx \qty{3e-4}{rad}$ (één boogminuut): zowel de
onderlinge afstand van de kegeltjes als de buiging aan de pupil stelt die grens.
\end{definition}

\begin{remark}[Waar een instrument voor dient]\label{rem:b1:optical-instruments:purpose}
Op het \omterm{def:b1:optical-instruments:punctum}{nabijheidspunt} beslaat een detail van \qty{1}{mm} $1/250 =
\qty{4e-3}{rad}$: dertien keer de grens van het \omterm{rem:b1:optical-instruments:abbe}{oplossend vermogen}; een detail
van \qty{0.05}{mm} beslaat $\qty{2e-4}{rad}$ en is onzichtbaar. Instrumenten
die naar nabije voorwerpen kijken (\omterm{def:b1:optical-instruments:G}{loep}, \omterm{def:b1:optical-instruments:microscope}{microscoop}) vergroten de \omterm{def:b1:optical-instruments:angular}{hoekgrootte}
verder dan het voorwerp dichterbij brengen kan; instrumenten voor verre
voorwerpen (de kijker) vergroten de \omterm{def:b1:optical-instruments:angular}{hoekgrootte} van dingen die men niet kan
naderen; de \omterm{def:b1:optical-instruments:camera}{camera} vervangt het \omterm{def:b1:optical-instruments:eye}{netvlies} door een sensor die seconden lang
licht kan verzamelen en die achteraf kan worden uitvergroot.
\end{remark}

\section{De loep}

\begin{definition}[Hoekvergroting]\label{def:b1:optical-instruments:G}
\index{hoekvergroting}\index{loep}\index{handelsvergroting}
Voor een instrument dat door het \omterm{def:b1:optical-instruments:eye}{oog} wordt gebruikt is de \emph{hoekvergroting}
$G = \theta'/\theta$, waarin $\theta'$ de \omterm{def:b1:optical-instruments:angular}{hoekgrootte} is van het beeld
door het instrument gezien en $\theta$ de \omterm{def:b1:optical-instruments:angular}{hoekgrootte} van het voorwerp met het
blote \omterm{def:b1:optical-instruments:eye}{oog} --- op het \omterm{def:b1:optical-instruments:punctum}{nabijheidspunt} $d_m = \qty{25}{cm}$ voor een nabij
voorwerp, ter plaatse voor een ver voorwerp.
\end{definition}

\begin{proposition}[De loep]\label{prop:b1:optical-instruments:loupe}
Een \omterm{def:b1:mirrors-thin-lenses:lens}{convergerende lens} met \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand} $f' < d_m$ met het voorwerp in
haar voorwerpsbrandvlak geeft een beeld op oneindig, zonder \omterm{def:b1:optical-instruments:eye}{accommodatie}
gezien, met
\[
G = \frac{d_m}{f'} = d_m V .
\]
\end{proposition}

\begin{proof}
Voorwerp $AB$ met hoogte $h$ in het brandvlak: de stralen vanuit $B$ vertrekken
evenwijdig, onder de hoek van de straal door $O$, $\theta' = h/f'$. Met het
blote \omterm{def:b1:optical-instruments:eye}{oog} op zijn best: $\theta = h/d_m$. Verhouding $d_m/f'$.
\end{proof}

\begin{example}[De loep van de horlogemaker]\label{ex:b1:optical-instruments:loupe}
$f' = \qty{25}{mm}$: $G = 250/25 = 10$, verkocht als ``$10\times$''. Een
detail van \qty{0.05}{mm} beslaat dan $\qty{2e-3}{rad}$, ruim opgelost.
$G$ hoger opdrijven betekent $f'$ van enkele millimeters, een piepklein
lensje tegen het \omterm{def:b1:optical-instruments:eye}{oog} gehouden --- de grens van de enkele \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, en de reden
dat de \omterm{def:b1:optical-instruments:microscope}{microscoop} bestaat.
\end{example}

\section{De microscoop}

\begin{definition}[Samengestelde microscoop]\label{def:b1:optical-instruments:microscope}
\index{microscoop}\index{objectief}\index{oculair}\index{optisch interval}
Een \emph{microscoop} bestaat uit een \emph{objectief} $L_1$ met kleine
\omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand} $f_1'$, dat een reëel, vergroot tussenbeeld van een nabij
voorwerp vormt in het voorwerpsbrandvlak van een \emph{oculair} $L_2$
(\omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand} $f_2'$), dat als \omterm{def:b1:optical-instruments:G}{loep} werkt. De afstand $\Delta =
\overline{F_1'F_2}$ tussen het beeldbrandpunt van het objectief en het
voorwerpsbrandpunt van het oculair is het \emph{optisch interval}, op
klassieke instrumenten gestandaardiseerd op \qty{160}{mm}.
\end{definition}

\begin{proposition}[Vergroting van een microscoop]\label{prop:b1:optical-instruments:microscope}
Met het eindbeeld op oneindig geldt
\[
\gamma_1 = -\frac{\Delta}{f_1'}, \qquad G_2 = \frac{d_m}{f_2'},
\qquad G = \abs{\gamma_1}\,G_2 = \frac{\Delta\,d_m}{f_1' f_2'} .
\]
\end{proposition}

\begin{proof}
Het tussenbeeld $A_1B_1$ ligt in het voorwerpsbrandvlak van het \omterm{def:b1:optical-instruments:microscope}{oculair},
dus op $\overline{F_1'A_1} = \Delta$; de \omterm{thm:b1:mirrors-thin-lenses:lensconj}{betrekking van Newton} geeft
$\gamma_1 = -\overline{F_1'A_1}/f_1' = -\Delta/f_1'$. Het \omterm{def:b1:optical-instruments:microscope}{oculair} toont
$A_1B_1$ (hoogte $\abs{\gamma_1} h$) vervolgens op oneindig onder de hoek
$\theta' = \abs{\gamma_1} h/f_2'$, tegenover $\theta = h/d_m$ voor het
blote \omterm{def:b1:optical-instruments:eye}{oog}.
\end{proof}

\begin{example}[Een schoolmicroscoop]\label{ex:b1:optical-instruments:micronum}
\omterm{def:b1:optical-instruments:microscope}{Objectief} $f_1' = \qty{4.0}{mm}$, \omterm{def:b1:optical-instruments:microscope}{oculair} $f_2' = \qty{25}{mm}$,
$\Delta = \qty{160}{mm}$: $\gamma_1 = -40$, $G_2 = 10$, $G = 400$. Het
voorwerp ligt op $\overline{F_1A} = -f_1'^2/\Delta = -16/160 =
\qty{-0.10}{mm}$ van $F_1$: \qty{4.1}{mm} van het \omterm{def:b1:optical-instruments:microscope}{objectief}. Scherpstellen
betekent de hele buis over fracties van een millimeter verschuiven.
\end{example}

\begin{remark}[De grens van het oplossend vermogen]\label{rem:b1:optical-instruments:abbe}
\index{oplossend vermogen}
Onbeperkt vergroten zou nutteloos zijn: licht is een golf, en een \omterm{def:b1:optical-instruments:microscope}{objectief} dat
stralen binnen een halve hoek $u$ opvangt (in een medium met index $n$)
kan twee punten die dichter bij elkaar liggen dan ongeveer
$\lambda/(2n\sin u)$ niet scheiden --- de golfbehandeling in het volume van
jaar 2 leidt dat af. Met $n\sin u \approx 1.4$ (olie-immersie) en $\lambda =
\qty{0.5}{\micro m}$ ligt de grens rond \qty{0.2}{\micro m}. Om dat
detail zichtbaar te maken voor het \omterm{def:b1:optical-instruments:eye}{oog} ($\epsilon = \qty{3e-4}{rad}$ op
\qty{25}{cm}, dus \qty{75}{\micro m}) volstaat een vergroting van ongeveer $400$;
boven $1000$ is het beeld alleen groter, niet rijker:
\emph{lege vergroting}.
\end{remark}

\section{De astronomische kijker}

\begin{proposition}[Lenzenkijker]\label{prop:b1:optical-instruments:telescope}
\index{kijker}\index{uittredepupil}
Een \omterm{def:b1:optical-instruments:microscope}{objectief} $L_1$ en een \omterm{def:b1:optical-instruments:microscope}{oculair} $L_2$ met $F_1' = F_2$ vormen een
\omterm{prop:b1:mirrors-thin-lenses:doublet}{afocaal systeem}: een ster (op oneindig) wordt op oneindig gezien, het
ontspannen \omterm{def:b1:optical-instruments:eye}{oog} bekijkt haar, en
\[
G = -\frac{f_1'}{f_2'} .
\]
Het beeld van de rand van het \omterm{def:b1:optical-instruments:microscope}{objectief} door het \omterm{def:b1:optical-instruments:microscope}{oculair} is de
\emph{\omterm{prop:b1:optical-instruments:telescope}{uittredepupil}}: een heldere schijf met diameter $D/\abs G$ op
$\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ achter het \omterm{def:b1:optical-instruments:microscope}{oculair}, waar de
pupil van het \omterm{def:b1:optical-instruments:eye}{oog} moet staan om al het licht op te vangen.
\end{proposition}

\begin{proof}
$G$ werd afgeleid in \cref{prop:b1:mirrors-thin-lenses:doublet}. Het
\omterm{def:b1:optical-instruments:microscope}{objectief} is, gezien vanuit $L_2$, een voorwerp op $\overline{O_2O_1} = -(f_1'
+ f_2')$; Descartes geeft $1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2')
= f_1'/[f_2'(f_1' + f_2')]$, en de vergroting $\overline{O_2P'}/
\overline{O_2O_1} = -f_2'/f_1' = 1/G$ schaalt de diameter $D$ tot
$D/\abs G$. Elke straal die het \omterm{def:b1:optical-instruments:microscope}{objectief} binnenkomt vertrekt door de
\omterm{prop:b1:optical-instruments:telescope}{uittredepupil}, want elk punt van het \omterm{def:b1:optical-instruments:microscope}{objectief} wordt daar afgebeeld.
\end{proof}

\begin{proposition}[Wat een kijker oplevert]\label{prop:b1:optical-instruments:gains}
\index{lichtverzamelend vermogen}\index{criterium van Rayleigh}
Vergeleken met het blote \omterm{def:b1:optical-instruments:eye}{oog} (pupil $d_p$):
\begin{itemize}
  \item \emph{\omterm{def:b1:optical-instruments:angular}{hoekgrootte}}: vermenigvuldigd met $\abs G$;
  \item \emph{opgevangen licht} van een puntbron: vermenigvuldigd met
        $(D/d_p)^2$, mits de \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} niet groter is dan de pupil
        van het \omterm{def:b1:optical-instruments:eye}{oog}, dus $\abs G \geq D/d_p$;
  \item \emph{\omterm{def:b1:optical-instruments:angular}{hoekoplossend vermogen}}: buiging aan het \omterm{def:b1:optical-instruments:microscope}{objectief} begrenst
        het tot $\theta_{\min} \approx 1.22\,\lambda/D$ (criterium van
        Rayleigh), ver onder de $\epsilon$ van het \omterm{def:b1:optical-instruments:eye}{oog}; de vergroting
        die deze grens zichtbaar maakt is $G_r = \epsilon/\theta_{\min}$,
        en grotere vergrotingen maken alleen de vlek groter.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Een refractor van 150 millimeter voor de liefhebber]\label{ex:b1:optical-instruments:telnum}
$D = \qty{150}{mm}$, $f_1' = \qty{1200}{mm}$, \omterm{def:b1:optical-instruments:microscope}{oculair} \qty{25}{mm}:
$G = -48$; \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} \qty{3.1}{mm}; lichtwinst $(150/7)^2 \approx
460$ tegenover een aan het donker aangepaste pupil van \qty{7}{mm};
\omterm{rem:b1:optical-instruments:abbe}{oplossend vermogen} $1.22 \times
\qty{550}{nm}/\qty{0.15}{m} = \qty{4.5e-6}{rad} = 0.9''$, dus
kraters van \qty{1.7}{km} breed op de maan; $G_r = 3\times10^{-4}/4.5\times
10^{-6} \approx 67$ --- het \omterm{def:b1:optical-instruments:microscope}{oculair} van \qty{6}{mm} ($G = 200$) toont geen
detail meer, alleen een fletsere, grotere schijf. De weekendopgave ontwerpt
dit instrument.
\end{example}

\begin{remark}[Spiegeltelescopen, verrekijkers, Hollandse kijkers]\label{rem:b1:optical-instruments:variants}
In de \emph{spiegeltelescoop} vervangt een \omterm{def:b1:mirrors-thin-lenses:mirror}{holle spiegel} met \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand}
$f_1'$ de objectieflens (geen kleurdispersie, en een spiegel kan meters breed
worden gemaakt en van achteren worden gesteund): al het bovenstaande blijft
gelden met $f_1'$ de \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand} van de spiegel. Het beeld staat
omgekeerd --- onschadelijk voor sterren; voor gebruik op aarde zetten
verrekijkers er een paar totaalreflecterende prisma's in om het recht te
zetten, en de kijker van Galilei gebruikt een divergerend \omterm{def:b1:optical-instruments:microscope}{oculair}
($f_2' < 0$, $G > 0$), korter maar met een smal veld.
\end{remark}

\section{De camera}

\begin{definition}[Camera; diafragmagetal]\label{def:b1:optical-instruments:camera}
\index{camera}\index{diafragmagetal}\index{scherptediepte}
Een \emph{camera} is een \omterm{def:b1:mirrors-thin-lenses:lens}{convergerende lens} met \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand} $f'$ die een
\omterm{def:b1:mirrors-thin-lenses:image}{reëel beeld} op een sensor vormt, op $\overline{OA'} \approx f'$ voor verre
onderwerpen; scherpstellen verplaatst de \omterm{def:b1:mirrors-thin-lenses:lens}{lens}. Een diafragma met diameter $D$
begrenst de bundel; het \emph{diafragmagetal} is $N = f'/D$ (geschreven
$f/N$). De \emph{beeldhoek} is $2\arctan(\ell/2f')$ voor een sensor met
breedte $\ell$.
\end{definition}

\begin{proposition}[Belichting en scherptediepte]\label{prop:b1:optical-instruments:dof}
\index{hyperfocale afstand}
De bestralingssterkte op de sensor schaalt als $1/N^2$, zodat de belichtingstijd
voor een gegeven beeldhelderheid als $N^2$ schaalt. Een punt op een andere
afstand dan de scherpgestelde vormt een onscherpteschijf; eist men dat die
onder een toegelaten diameter $c$ blijft, dan geeft een \omterm{def:b1:mirrors-thin-lenses:lens}{lens} die op de
\emph{\omterm{prop:b1:optical-instruments:dof}{hyperfocale afstand}}
\[
H = \frac{f'^2}{N c}
\]
is scherpgesteld alles van $H/2$ tot oneindig aanvaardbaar scherp weer.
\end{proposition}

\begin{proof}
Bestralingssterkte $=$ vermogen/oppervlak $\propto D^2/f'^2 = 1/N^2$ (het beeld
van een gegeven onderwerp heeft een grootte die door $f'$ is vastgelegd, en het
opgevangen vermogen groeit als $D^2$). Stel scherp op afstand $p$, met het beeld op $q \approx f'$;
een punt op oneindig komt samen op $f'$, dus $q - f'$ vóór de sensor;
zijn stralenkegel, met grondvlak $D$ aan de \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, heeft diameter $D(q - f')/q$
op de sensor (gelijkvormige driehoeken). Met $1/q = 1/f' - 1/p$ paraxiaal is
$q - f' \approx f'^2/p$ voor $p \gg f'$, zodat de onscherpte
$Df'/p = f'^2/(Np)$ bedraagt; zij is gelijk aan $c$ voor $p = H$. Dezelfde
meetkunde aan de nabije kant geeft de grens $H/2$.
\end{proof}

\begin{example}[Kleinbeeld en telefoon]\label{ex:b1:optical-instruments:cameras}
Een \omterm{def:b1:mirrors-thin-lenses:lens}{lens} van \qty{50}{mm} op een sensor van \qty{36}{mm} breed: beeldhoek

telefoons de achtergrondonscherpte in software nabootsen.
\end{example}

\section{Probleem: Een kijker voor de maan ontwerpen}

\begin{problem}[{Weekendopgave --- een refractor van 150 millimeter op een
gazon: kies de oculairs, plaats het oog, tel het licht, en zoek uit hoe
klein een krater is die hij nog kan tonen}]
\label{pb:b1:optical-instruments:1}
Het \omterm{def:b1:optical-instruments:microscope}{objectief} is een \omterm{def:b1:mirrors-thin-lenses:lens}{convergerende lens} $L_1$ met \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand}
$f_1' = \qty{1200}{mm}$ en diameter $D = \qty{150}{mm}$; er zijn \omterm{def:b1:optical-instruments:microscope}{oculairs}
$L_2$ met \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstanden} \qty{25}{mm} en \qty{6}{mm} beschikbaar. De
maan staat \qty{3.84e5}{km} ver en beslaat $0.52^\circ$; de nachtpupil van
het \omterm{def:b1:optical-instruments:eye}{oog} is \qty{7}{mm}, haar \omterm{rem:b1:optical-instruments:abbe}{oplossend vermogen} $\epsilon = \qty{3.0e-4}{rad}$;
$\lambda = \qty{550}{nm}$.

\medskip
\noindent\textbf{Deel I --- De afocale opstelling.}
\begin{enumerate}
  \item Waar moet het \omterm{def:b1:optical-instruments:microscope}{oculair} staan opdat een ster zonder \omterm{def:b1:optical-instruments:eye}{accommodatie}
        wordt gezien? Geef de afstand objectief--oculair voor elk
        \omterm{def:b1:optical-instruments:microscope}{oculair}.
  \item Ga met de \omterm{thm:b1:mirrors-thin-lenses:lensconj}{betrekking van Descartes}, op elke \omterm{def:b1:mirrors-thin-lenses:lens}{lens} om beurten
        toegepast, na dat een punt op oneindig op de as op oneindig wordt afgebeeld.
  \item Toon aan dat een bundel van een ster onder hoek $\alpha$ met de as
        uittreedt als evenwijdige bundel onder hoek $\alpha' = -(f_1'/f_2')\alpha$.
  \item Bereken $G$ voor elk \omterm{def:b1:optical-instruments:microscope}{oculair} en de schijnbare diameter van de
        maan door elk.
  \item Het beeld staat omgekeerd. Waarom is dat hier aanvaardbaar, en wat
        zou een aardse spotting scope toevoegen?
  \item Waar ligt het reële tussenbeeld van de maan, en wat is
        zijn diameter? (Daar kan het worden gefotografeerd.)
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Waar het \omterm{def:b1:optical-instruments:eye}{oog} komt.}
\begin{enumerate}[resume]
  \item Toon aan dat het \omterm{def:b1:optical-instruments:microscope}{oculair} een \omterm{def:b1:mirrors-thin-lenses:image}{reëel beeld} van de rand van het \omterm{def:b1:optical-instruments:microscope}{objectief}
        vormt op $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1'$ erachter, met
        diameter $D/\abs G$ (de \omterm{prop:b1:optical-instruments:telescope}{uittredepupil}).
  \item Bereken plaats en diameter van de \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} voor elk
        \omterm{def:b1:optical-instruments:microscope}{oculair}.
  \item Waarom moet de pupil van het \omterm{def:b1:optical-instruments:eye}{oog} op de \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} staan? Wat
        gebeurt er met het veld als zij verder naar achteren staat?
  \item Om het licht van een ster volledig te benutten mag de \omterm{prop:b1:optical-instruments:telescope}{uittredepupil}
        de pupil van het \omterm{def:b1:optical-instruments:eye}{oog} niet overtreffen: leid de kleinste bruikbare
        vergroting van deze kijker bij nacht af.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- \omterm{rem:b1:optical-instruments:abbe}{Oplossend vermogen}.}
\begin{enumerate}[resume]
  \item Bereken de buigingsgrens $\theta_{\min} = 1.22\lambda/D$
        in radialen en in boogseconden.
  \item Met welke kleinste krater op de maan komt deze hoek overeen? En
        voor het blote \omterm{def:b1:optical-instruments:eye}{oog}?
  \item Definieer en bereken de oplossende vergroting $G_r =
        \epsilon/\theta_{\min}$. Welk \omterm{def:b1:optical-instruments:microscope}{oculair} ligt daar het dichtst bij?
  \item Met het \omterm{def:b1:optical-instruments:microscope}{oculair} van \qty{6}{mm} oogt het beeld waziger en
        fletser dan met dat van \qty{25}{mm}. Verklaar beide effecten.
  \item Luchtonrust smeert sterbeelden uit tot ongeveer $1''$. Vanaf
        welke objectiefdiameters houdt de grens van Rayleigh vanaf de
        grond op te tellen, en wat levert een grotere $D$ dan nog op?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Licht.}
\begin{enumerate}[resume]
  \item Met welke factor vangt het \omterm{def:b1:optical-instruments:microscope}{objectief} meer licht op dan de
        nachtpupil?
  \item Zet dit om in magnituden ($\Delta m = 2.5\log_{10}$ van de

  \item Bereken het voor beide \omterm{def:b1:optical-instruments:microscope}{oculairs} en zeg of de hele maan in het
