\section{Het model van de geometrische optica}

\begin{definition}[Lichtstraal; homogeen medium]\label{def:b1:rays-reflection-refraction:ray}
\index{lichtstraal}\index{geometrische optica}\index{homogeen medium}
In het model van de \emph{geometrische optica} plant licht zich voort langs
\emph{lichtstralen}: georiënteerde krommen die in een \emph{homogeen} medium
(overal dezelfde eigenschappen) rechte lijnen zijn en die onafhankelijk van
elkaar energie dragen. Een bundel is een bos stralen; een puntbron zendt
stralen uit in alle richtingen.
\end{definition}

\begin{definition}[Brekingsindex]\label{def:b1:rays-reflection-refraction:index}
\index{brekingsindex}\index{doorzichtig medium}
Een \emph{doorzichtig} medium laat licht door; het licht loopt daarin met een
snelheid $v < c$. Zijn \emph{brekingsindex} is
\[
n = \frac{c}{v} \geq 1 .
\]
Vacuüm: $n = 1$; lucht: $1.0003$; water: $1.33$; gewoon glas: $1.5$
tot $1.6$; diamant: $2.42$. Een golf met frequentie $\nu$ behoudt haar
frequentie bij het binnentreden van een medium, zodat haar golflengte krimpt
van $\lambda_0 =
c/\nu$ in vacuüm tot $\lambda = \lambda_0/n$.
\end{definition}

\begin{remark}[Dispersie]\label{rem:b1:rays-reflection-refraction:dispersion}
\index{dispersie (optische)}\index{wet van Cauchy}
In materie hangt de index licht af van de golflengte --- het medium is
\emph{dispersief}. Voor glas in het zichtbare gebied past de empirische \omterm{rem:b1:rays-reflection-refraction:dispersion}{wet
van Cauchy} $n(\lambda) = A + B/\lambda^2$ goed: $n$ is voor blauw ongeveer
$0.01$ tot $0.02$ groter dan voor rood. Het prisma en de regenboog, aan het
eind van dit hoofdstuk, leven van dat verschil.
\end{remark}

\begin{remark}[Grenzen van het model]\label{rem:b1:rays-reflection-refraction:limits}
Licht is een golf, en een golf die door een opening met breedte $a$ gaat,
spreidt zich over een hoek van de orde $\lambda/a$ (diffractie,
\cref{ch:b1:wave-propagation}). Stralen zijn een goede beschrijving zolang
elke opening, lens en hindernis veel groter is dan de golflengte
(\qty{0.4}{} tot \qty{0.8}{\micro m} voor zichtbaar licht): een gaatje van
\qty{1}{mm} spreidt een bundel over $10^{-3}$ radialen, verwaarloosbaar op een
optische bank; een gaatje van \qty{1}{\micro m} spreidt hem over de hele
halfruimte, en geen stralenbeeld overleeft dat. De kern van \qty{9}{\micro m}
van een monomode telecomvezel ligt al buiten het model.
\end{remark}

\begin{proposition}[Omkeerbaarheid van de lichtweg]\label{prop:b1:rays-reflection-refraction:reversibility}
\index{omkeerbaarheid van de lichtweg}
De weg van een straal hangt niet af van de richting waarin het licht hem
aflegt: gaat licht van $A$ naar $B$ langs een weg, dan volgt licht vanuit $B$
diezelfde weg naar $A$.
\end{proposition}
\admitted

\section{Reflectie en breking}

\begin{definition}[Inval]\label{def:b1:rays-reflection-refraction:incidence}
\index{invalsvlak}\index{invalshoek}\index{brekend vlak}
Een straal die het scheidingsvlak tussen twee media (een \emph{brekend vlak})
in een punt $I$ treft, maakt met de \emph{normaal} op dat vlak in $I$ de
\emph{invalshoek} $i_1$. Het \emph{invalsvlak} bevat de invallende straal en
de normaal.
\end{definition}

\begin{theorem}[Wetten van reflectie en breking (Snellius--Descartes)]\label{thm:b1:rays-reflection-refraction:snell}
\index{reflectie!wet van}\index{breking!wet van}\index{wet van Snellius}
Aan het scheidingsvlak tussen media met indices $n_1$ (kant van inval) en
$n_2$ geldt:
\begin{enumerate}
  \item de gereflecteerde en de gebroken straal liggen in het \omterm{def:b1:rays-reflection-refraction:incidence}{invalsvlak};
  \item de gereflecteerde straal maakt met de normaal de hoek $i_1' = i_1$,
        aan de andere kant van de normaal;
  \item de gebroken straal maakt met de normaal een hoek $i_2$ zodanig dat
        \[
        n_1 \sin i_1 = n_2 \sin i_2 .
        \]
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[Waar de wetten vandaan komen]\label{rem:b1:rays-reflection-refraction:fermat}
\index{beginsel van Fermat}
Beide wetten volgen uit het \emph{\omterm{rem:b1:rays-reflection-refraction:fermat}{beginsel van Fermat}}: tussen twee punten
volgt licht de weg met de kortste looptijd. Reflectie onder gelijke hoeken is
de kortste geknikte weg die de spiegel raakt; de \omterm{thm:b1:rays-reflection-refraction:snell}{wet van Snellius} is de weg
die een langere route in het snelle medium inruilt tegen een kortere in het
trage (\cref{exo:b1:rays-reflection-refraction:12}). In het volume van jaar 2
worden beide wetten afgeleid uit het golfkarakter van licht aan een grensvlak.
\end{remark}

\begin{proposition}[Welke kant de straal op buigt]\label{prop:b1:rays-reflection-refraction:bending}
Bij het binnengaan van een sterker brekend medium ($n_2 > n_1$) buigt een
straal \emph{naar} de normaal toe ($i_2 < i_1$); bij het binnengaan van een
zwakker brekend medium \emph{van} de normaal af ($i_2 > i_1$). Een straal
langs de normaal gaat onafgebogen door.
\end{proposition}

\begin{proof}
$\sin i_2 = (n_1/n_2)\sin i_1$, met $\sin$ stijgend op
$[0, \pi/2]$.
\end{proof}

\begin{example}[Schijnbare diepte]\label{ex:b1:rays-reflection-refraction:depth}
\index{schijnbare diepte}
Een muntstuk ligt op diepte $h$ onder water ($n = 1.33$) en wordt bijna
loodrecht van boven bekeken. Een straal vanaf de munt die het oppervlak onder
een kleine hoek $i_2$ (in het water) bereikt, verlaat het onder $i_1$ met
$\sin i_1 = n\sin i_2$; voor kleine hoeken is $i_1 \approx n\,i_2$. Het oog
verlengt de uittredende straal naar achteren: zij lijkt van een diepte $h'$ te
komen, met $\tan i_1 \approx x/h'$ en $\tan i_2 \approx x/h$ voor hetzelfde
oppervlaktepunt op afstand $x$ van de verticaal, dus $h' = h\,i_2/i_1 = h/n$.
Een bad van \qty{2.0}{m} diep lijkt \qty{1.5}{m} diep; een vis ziet de visser
$1.33$ keer hoger dan hij is.
\end{example}

\begin{theorem}[Totale interne reflectie]\label{thm:b1:rays-reflection-refraction:tir}
\index{totale interne reflectie}\index{grenshoek}
Licht dat vanuit een medium met index $n_1$ naar een zwakker brekend medium
gaat ($n_2 < n_1$), wordt alleen gebroken als $i_1 \leq i_c$, waarin de
\emph{grenshoek} voldoet aan
\[
\sin i_c = \frac{n_2}{n_1} .
\]
Voor $i_1 > i_c$ is er geen gebroken straal: het vlak kaatst al het licht
terug --- \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{totale interne reflectie}}.
\end{theorem}

\begin{proof}
De \omterm{thm:b1:rays-reflection-refraction:snell}{wet van Snellius} vraagt $\sin i_2 = (n_1/n_2)\sin i_1 > \sin i_1$; dat
overschrijdt $1$ zodra $\sin i_1 > n_2/n_1$, en dan bestaat er geen hoek $i_2$.
Bij $i_1 = i_c$ scheert de gebroken straal langs het oppervlak
($i_2 = 90^\circ$). Dat dan \emph{alle} energie wordt gereflecteerd, is een
uitspraak over intensiteiten, waarmee de \omterm{def:b1:rays-reflection-refraction:ray}{geometrische optica} niets kan; de
golfbehandeling in het volume van jaar 2 bevestigt haar.
\end{proof}

\begin{example}[Diamanten en zwemmers]\label{ex:b1:rays-reflection-refraction:tir}
Glas--lucht: $\sin i_c = 1/1.5$, $i_c = 41.8^\circ$, en daarom kaatst een
prisma van $45^\circ$ totaal terug en vervangt het een spiegel in een verrekijker.
Diamant--lucht: $i_c = 24.4^\circ$; een geslepen diamant vangt licht dat door
zijn bovenkant binnenkomt en stuurt het er bijna volledig weer uit --- de
schittering is totale reflectie, het vuur is dispersie. Water--lucht:
$i_c = 48.8^\circ$; wie omhoogkijkt onder water ziet de hele hemel binnen een
kegel met halve tophoek $48.8^\circ$ (``het venster van Snellius'') en
daarbuiten de weerspiegelde bodem.
\end{example}

\section{De glasvezel}

\begin{definition}[Stapindexvezel]\label{def:b1:rays-reflection-refraction:fiber}
\index{glasvezel}\index{kern (vezel)}\index{mantel (vezel)}\index{numerieke apertuur}
Een \emph{glasvezel met stapindex} is een cilindrische glazen \emph{kern} met
index $n_1$, omhuld door een \emph{mantel} met een iets lagere index $n_2$.
Licht dat de kern binnenkomt binnen een kegel met halve tophoek $\theta_a$
rond de as, wordt geleid door opeenvolgende \omterm{thm:b1:rays-reflection-refraction:tir}{totale interne reflecties} aan het
scheidingsvlak kern--mantel. De \emph{numerieke apertuur} is
$\mathrm{NA} = \sin\theta_a$.
\end{definition}

\begin{proposition}[Acceptatiekegel en intermodale spreiding]\label{prop:b1:rays-reflection-refraction:fiber}
\index{acceptatiehoek}\index{intermodale dispersie}
Voor een vezel waarvan het intreevlak zich in lucht bevindt geldt
\[
\sin\theta_a = \sqrt{n_1^2 - n_2^2} .
\]
Over een lengte $L$ komen de geleide straal langs de as en de steilste
geleide straal aan met een onderling tijdsverschil
\[
\Delta t = \frac{L\,n_1}{c}\left(\frac{n_1}{n_2} - 1\right)
\]
(\emph{\omterm{prop:b1:rays-reflection-refraction:fiber}{intermodale dispersie}}).
\end{proposition}

\begin{proof}
Een straal die onder een hoek $\theta$ met de as binnenkomt, wordt gebroken tot
$\theta'$ met $\sin\theta = n_1\sin\theta'$ en treft de mantel onder inval
$i = 90^\circ - \theta'$. Geleiding vereist $\sin i \geq n_2/n_1$, dus
$\cos\theta' \geq n_2/n_1$, dus $\sin\theta' \leq
\sqrt{1 - n_2^2/n_1^2}$; vermenigvuldig met $n_1$. De straal langs de as legt
$L$ af met snelheid $c/n_1$: $t_0 = Ln_1/c$. De steilste straal zigzagt onder
hoek $\theta'_{\max}$ met $\cos\theta'_{\max} = n_2/n_1$; haar weg is
$L/\cos\theta'_{\max} = L n_1/n_2$, dus $t_1 = L n_1^2/(n_2 c)$, en
$\Delta t = t_1 - t_0$.
\end{proof}

\begin{example}[Getallen van een telecomvezel]\label{ex:b1:rays-reflection-refraction:fibernum}
$n_1 = 1.480$, $n_2 = 1.465$: $\mathrm{NA} = \sqrt{2.190 - 2.146} =
0.21$, $\theta_a = 12^\circ$; $i_c = 81.8^\circ$, zodat geleide stralen binnen
$8.2^\circ$ van de as blijven. Intermodale spreiding: $\Delta t/L =
(1.48/c)(1.48/1.465 - 1) = \qty{51}{ns/km}$. Over \qty{2}{km} spreidt een puls
\qty{0.1}{\micro s} uit, wat de bitsnelheid begrenst rond
\qty{5}{Mbit/s}: de weekendopgave volgt deze beperking tot haar remedie, de
monomodevezel.
\end{example}

\section{De vlakke spiegel}

\begin{proposition}[Beeld in een vlakke spiegel]\label{prop:b1:rays-reflection-refraction:mirror}
\index{vlakke spiegel}\index{virtueel beeld}\index{stigmatisme}
Alle stralen vanuit een punt $A$ die door een \omterm{prop:b1:rays-reflection-refraction:mirror}{vlakke spiegel} worden
gereflecteerd, lijken te komen uit het punt $A'$ dat symmetrisch met $A$ ligt
ten opzichte van het spiegelvlak: $A'$ is het \emph{beeld} van $A$. Het is
\emph{virtueel} (er gaat geen licht doorheen) en de spiegel is \emph{streng
stigmatisch}: elk punt heeft een scherp puntbeeld.
\end{proposition}

\begin{proof}
Een straal vanuit $A$ die de spiegel in $I$ treft onder inval $i$, vertrekt
onder $i$ aan de andere kant van de normaal. De gereflecteerde lijn en de
normaal in $I$ maken de hoek $i$, net als de lijn $IA$; door spiegeling in het
spiegelvlak is de gereflecteerde lijn het spiegelbeeld van de lijn $IA$ en
gaat zij dus door het symmetrische punt $A'$ --- voor elke $I$.
\end{proof}

\begin{proposition}[Een spiegel draaien]\label{prop:b1:rays-reflection-refraction:rotation}
Een \omterm{prop:b1:rays-reflection-refraction:mirror}{vlakke spiegel} over een hoek $\alpha$ draaien om een as in zijn vlak,
loodrecht op het \omterm{def:b1:rays-reflection-refraction:incidence}{invalsvlak}, draait de gereflecteerde straal over $2\alpha$.
\end{proposition}

\begin{proof}
De normaal draait over $\alpha$, zodat de inval verandert van $i$ in
$i + \alpha$; de gereflecteerde straal, onder $i + \alpha$ aan de andere kant
van de nieuwe normaal, is over $\alpha$ (normaal) $+ \alpha$ (hoek) $=
2\alpha$ ten opzichte van haar oude richting verdraaid.
\end{proof}

\begin{example}[De hefboom van de galvanometer]\label{ex:b1:rays-reflection-refraction:galvanometer}
Een spiegeltje op de spoel van een galvanometer, beschenen door een lamp en een
lichtvlek werpend op een schaal \qty{2}{m} verderop: een draaiing van
$\qty{1}{mrad}$ van de spoel verplaatst de vlek over $2 \times 10^{-3} \times
\qty{2}{m} = \qty{4}{mm}$ --- een optische hefboom die vergroot zonder
wrijving, het beginsel van de spiegelgalvanometer en van de laserscanner.
\end{example}

\section{Het prisma}

\begin{proposition}[Prismaformules]\label{prop:b1:rays-reflection-refraction:prism}
\index{prisma}\index{deviatie}\index{minimale deviatie}
Een straal die in lucht een prisma met tophoek $A$ en index $n$ doorloopt, in
het vlak loodrecht op de ribbe, met hoeken $i$ (intrede), $r$, $r'$
(binnenin) en $i'$ (uittrede) gemeten vanaf de normalen, voldoet aan
\[
\sin i = n \sin r, \qquad r + r' = A, \qquad \sin i' = n \sin r',
\]
en wordt afgebogen over $D = i + i' - A$. De deviatie is minimaal wanneer de
weg symmetrisch is, $i = i'$, $r = r' = A/2$; dan geldt
\[
n = \frac{\sin\frac{A + D_{\min}}{2}}{\sin\frac{A}{2}} .
\]
\end{proposition}

\begin{proof}
De \omterm{thm:b1:rays-reflection-refraction:snell}{wet van Snellius} aan elk vlak; $r + r' = A$ omdat de twee normalen de
hoek $A$ maken (de vierhoek gevormd door de tophoek, de twee invalspunten en
het snijpunt van de normalen). De totale deviatie is
$(i - r) + (i' - r') = i + i' - A$. Door de \omterm{prop:b1:rays-reflection-refraction:reversibility}{omkeerbaarheid van de lichtweg} is
de deviatie dezelfde functie van $i$ als van $i'$; lag het minimum bij
$i_0 \neq i_0'$, dan lag het ook bij de omgekeerde inval $i_0'$, wat twee
minima zou geven --- en dat verbiedt het monotone gedrag van $D(i)$ aan
weerszijden van haar minimum: het minimum ligt bij $i = i'$.
Dan is $r = r' = A/2$ en $D_{\min} = 2i - A$; invullen van $i = (A +
D_{\min})/2$ in $\sin i = n\sin(A/2)$ geeft de formule.
\end{proof}

\begin{example}[Een index meten]\label{ex:b1:rays-reflection-refraction:prismnum}
Een prisma van $60^\circ$ wordt gedraaid tot de deviatie van een gele bundel
het kleinst is: $D_{\min} = 38.9^\circ$. Dan is $n = \sin 49.45^\circ/\sin 30^\circ
= 0.760/0.500 = 1.52$. \omterm{prop:b1:rays-reflection-refraction:prism}{Minimale deviatie} is de standaardmanier om de index van
een glassoort te meten --- het scherpe minimum, gevonden door het prisma heen
en weer te draaien, is ongevoelig voor kleine instelfouten.
\end{example}

\begin{remark}[Dunne prisma's en spectroscopen]\label{rem:b1:rays-reflection-refraction:thin}
Voor kleine $A$ en kleine inval zijn de sinussen gelijk aan de hoeken en is
$D \approx (n - 1)A$, onafhankelijk van $i$: een dun prisma buigt elke straal
over dezelfde hoek af --- een glazen wig is een ``prisma'' dat licht afbuigt,
en een lens zal een stapel van zulke wiggen blijken te zijn. Omdat
$n$ van $\lambda$ afhangt, doet de deviatie dat ook: een prisma spreidt wit
licht tot een spectrum (blauw het sterkst afgebogen), het hart van de
prismaspectroscoop.
\end{remark}

\section{De regenboog}

\begin{example}[De regenboog van Descartes]\label{ex:b1:rays-reflection-refraction:rainbow}
\index{regenboog}
Een zonnestraal treedt onder inval $i$ een bolvormige regendruppel binnen,
wordt gebroken ($\sin i = n\sin r$), kaatst één keer terug tegen de
achterwand en treedt na een tweede breking weer uit. Elke breking buigt hem
over $i - r$ af, de interne reflectie over $\pi - 2r$: totale deviatie
$D(i) = \pi + 2i - 4r$. Stralen vullen de druppel bij elke $i$; de waarnemer
ziet het helderste licht daar waar veel stralen in vrijwel dezelfde richting
vertrekken, dus waar $D$ stationair is: $\dd D/\dd i = 2 - 4\,\dd r/\dd i
= 0$. Differentiëren van de \omterm{thm:b1:rays-reflection-refraction:snell}{wet van Snellius} geeft $\cos i = n\cos r\,\dd r/\dd i$,
zodat de voorwaarde luidt $2\cos i = n\cos r$; kwadrateren en gebruik van
$n^2\cos^2 r = n^2 - \sin^2 i = n^2 - 1 + \cos^2 i$ geeft
\[
\cos^2 i = \frac{n^2 - 1}{3} .
\]
Voor water, $n = 1.333$: $i = 59.4^\circ$, $r = 40.2^\circ$,
$D_{\min} = 137.9^\circ$. Het licht komt terug naar de kant van de zon
onder $180^\circ - 137.9^\circ = 42^\circ$ vanaf de richting tegenover de zon:
de regenboog is een kegel met halve tophoek $42^\circ$ rond de schaduw van het
hoofd van de waarnemer. Rood ($n = 1.331$) ligt op $42.4^\circ$, violet
($n = 1.343$) op $40.6^\circ$ --- rood buiten, violet binnen.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:1}
Bereken de lichtsnelheid in water ($n = 1.33$), in glas ($1.50$) en in
diamant ($2.42$). Hoe lang doet licht erover om \qty{1.0}{km} vezel met index
$1.48$ te doorlopen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:2}
Een straal in lucht treft water ($n = 1.33$) onder $30^\circ$, daarna onder
$60^\circ$ met de normaal: bepaal de brekingshoeken. Een straal in water
treft het oppervlak onder $50^\circ$: wat gebeurt er?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:3}
Bereken de grenshoek voor glas--lucht ($n = 1.50$), diamant--lucht
($2.42$) en water--lucht ($1.33$). Hoe groot is de hoekdiameter van de
hemelschijf die een duiker ziet?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:4}
Iemand van \qty{1.70}{m} lang, met de ogen \qty{10}{cm} onder de kruin, staat
voor een verticale wandspiegel. Bepaal de kleinste spiegel (hoogte, en de
plaats van haar randen boven de vloer) waarin het hele lichaam zichtbaar is.
Hangt het antwoord af van de afstand tot de spiegel?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:5}
Een muntstuk ligt \qty{1.2}{m} diep in een zwembad; op welke diepte lijkt het
te liggen voor wie van boven kijkt? De ogen van een visser bevinden zich
\qty{1.8}{m} boven het water: hoe hoog lijken ze voor een vis die omhoogkijkt?
Verantwoord beide met stralen onder kleine hoeken.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:6}
Een prisma heeft $A = 60^\circ$ en $n = 1.52$. Bereken de \omterm{prop:b1:rays-reflection-refraction:prism}{minimale
deviatie}. Toon aan dat een straal alleen uit het tweede vlak kan treden als
$A \leq 2 i_c$, en ga na dat dit prisma daaraan voldoet.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:7}
Een stapindexvezel heeft $n_1 = 1.50$, $n_2 = 1.48$. Bereken haar
\omterm{def:b1:rays-reflection-refraction:fiber}{numerieke apertuur}, haar acceptatiehoek, het intermodale tijdsverschil per
kilometer en de daaruit volgende maximale bitsnelheid over \qty{1}{km}
(neem $B_{\max} \approx 1/(2\Delta t)$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:8}
Een dun prisma met tophoek $10^\circ$ heeft $n = 1.510$ voor rood en $1.530$
voor blauw. Bereken de deviatie van elke kleur en de hoekbreedte van het
spectrum dat het levert.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:9}
Een straal treft een glazen plaat met evenwijdige vlakken ($n = 1.50$, dikte
$e = \qty{2.0}{cm}$) onder $45^\circ$. Toon aan dat hij evenwijdig aan zijn
invalsrichting uittreedt, zijdelings verschoven over $d = e\,\sin(i - r)/\cos r$,
en bereken $d$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:10}
De secundaire regenboog komt van stralen die \emph{twee keer} binnen de
druppel worden gereflecteerd. Toon aan dat $D = 2\pi + 2i - 6r$, dat de
stationaire deviatie voldoet aan $\cos^2 i = (n^2 - 1)/8$, en bereken de hoek
van de secundaire boog vanaf de richting tegenover de zon voor $n = 1.333$.
Waar ligt hij ten opzichte van de primaire, en in welke volgorde staan zijn kleuren?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:11}
Op een heet wegdek heeft de lucht vlak boven het asfalt index
$n_{\text{heet}} = 1.00026$, de koelere lucht erboven $n_{\text{koel}} =
1.00029$. Modelleer de lagen als één scherp grensvlak: bepaal de kritieke
scherende hoek (gemeten vanaf de weg) waaronder een straal uit de hemel
volledig omhoog wordt gereflecteerd. Het oog van een bestuurder zit
\qty{1.2}{m} boven de weg: vanaf welke afstand lijkt de weg op water?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:12}
De punten $A$ (op hoogte $a$ boven een grensvlak) en $B$ (op diepte $b$
eronder) liggen horizontaal $d$ uit elkaar; licht loopt met $v_1$ erboven en
$v_2$ eronder. Schrijf de looptijd op van de geknikte weg door het
grensvlakpunt op horizontale positie $x$, en toon aan dat de tijd precies dan
stationair is wanneer $\sin i_1/v_1 = \sin i_2/v_2$ --- de \omterm{thm:b1:rays-reflection-refraction:snell}{wet van Snellius}.
\end{exercise}

\section{Probleem: De glasvezelverbinding}

\begin{problem}[{Weekendopgave --- een glazen draad onder zee: hoe ver en
hoe snel kan een lichtpuls een bit dragen voordat meetkunde, glas en
dispersie hem uitsmeren}]
\label{pb:b1:rays-reflection-refraction:1}
Een verbinding gebruikt een stapindexvezel met kern $n_1 = 1.480$, mantel
$n_2 = 1.465$, kernstraal $a = \qty{25}{\micro m}$, bij de golflengte
$\lambda_0 = \qty{1550}{nm}$. Gegeven: $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Deel I --- Licht in glas.}
\begin{enumerate}
  \item Definieer de index en bereken de lichtsnelheid in de kern.
  \item Hoe lang doet een puls erover om \qty{50}{km} langs de as af te
        leggen?
  \item Bereken de frequentie van het licht en zijn golflengte binnen de
        kern.
  \item Waarom wordt de kern in glas met lagere index gehuld in plaats van
        bloot in lucht gelaten, wat een grotere indexsprong zou geven?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Geleiding.}
\begin{enumerate}[resume]
  \item Bereken de grenshoek aan het scheidingsvlak kern--mantel.
  \item Leid daaruit de grootste hoek af die een geleide straal met de as
        kan maken.
  \item Een straal komt vanuit lucht het vlakke eindvlak binnen onder een
        hoek $\theta$ met de as. Toon aan dat hij geleid wordt als
        $\sin\theta \leq
        \sqrt{n_1^2 - n_2^2}$, en bereken de acceptatiehoek
        $\theta_a$.
  \item Geef de \omterm{def:b1:rays-reflection-refraction:fiber}{numerieke apertuur}. Waarom is een vezel met een grotere
        indexsprong gemakkelijker met licht te voeden?
  \item De vezel wordt gebogen langs een cirkel met straal $R_b$. Beschouw
        een straal die evenwijdig aan de as langs de binnenrand van de kern
        loopt: toon aan dat hij de buitenwand treft onder inval $i$ met
        $\sin i = R_b/(R_b + 2a)$, en bepaal de kleinste $R_b$ waarvoor hij
        nog geleid wordt.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- \omterm{prop:b1:rays-reflection-refraction:fiber}{Intermodale dispersie}.}
\begin{enumerate}[resume]
  \item Druk de looptijden uit van de straal langs de as en van de steilste
        geleide straal over een lengte $L$.
  \item Leid de spreiding $\Delta t$ per kilometer af, en haar waarde over
        $L = \qty{2}{km}$.
  \item Een bit is een korte puls; twee opeenvolgende pulsen moeten
        minstens $2\Delta t$ uit elkaar blijven om te onderscheiden zijn.
        Schat de maximale bitsnelheid over \qty{2}{km}.
  \item Toon aan dat het product (bitsnelheid) $\times$ (lengte) voor deze
        vezel constant is, en bereken het in \unit{Mbit.s^{-1}.km}.
  \item In een gradiëntindexvezel neemt de index vanaf de as geleidelijk af
        naar buiten toe. Leg kwalitatief uit waarom dit de intermodale
        spreiding verkleint.
  \item Een monomodevezel heeft een kerndiameter van ongeveer \qty{9}{\micro m}.
        Vergelijk met $\lambda_0$ en leg uit waarom het stralenmodel haar
        niet kan beschrijven; wat wordt er gewonnen?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Demping en chromatische dispersie.}
De vezel verzwakt het vermogen met $\alpha = \qty{0.20}{dB/km}$ bij
\qty{1550}{nm}: $P(L) = P_0\,10^{-\alpha L/10}$. De zender
zendt $P_0 = \qty{1.0}{mW}$ uit; de ontvanger heeft minstens
$P_{\min} = \qty{1.0}{\micro W}$ nodig. Omdat $n$ van $\lambda$ afhangt,
lopen twee golflengtes met licht verschillende snelheden: een bron met
spectrale breedte $\Delta\lambda$ geeft een pulsspreiding $\Delta t_c = D\,\Delta\lambda\,L$
met $D = \qty{17}{ps/(nm.km)}$.
\begin{enumerate}[resume]
  \item Welk deel van het vermogen overleeft \qty{100}{km}?
  \item Bepaal de maximale lengte voordat een versterker nodig is.
  \item Er staan versterkers om de \qty{100}{km}: welke marge, in dB en
        als vermogensverhouding, laat dat voor connectoren en lassen?
  \item Bereken $\Delta t_c$ over \qty{100}{km} voor een ledbron
        ($\Delta\lambda = \qty{40}{nm}$) en voor een laser
        ($\Delta\lambda = \qty{0.10}{nm}$); leid de maximale bitsnelheid
        van elk over die lengte af (hetzelfde criterium als hierboven).
  \item Bij \qty{850}{nm} verzwakt kwartsglas \qty{2.0}{dB/km}. Welke
        afstand zou hetzelfde vermogensbudget daar toelaten? Waarom is
        \qty{1550}{nm} de telecomgolflengte?
  \item Coaxiale koperkabel verzwakt ongeveer \qty{30}{dB/km} bij de
        frequenties die \qty{1}{Gbit/s} vereist. Vergelijk de afstand
        tussen versterkers met die van de vezel.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De hele verbinding.}
\begin{enumerate}[resume]
  \item Wat begrenst voor de stapindexvezel over \qty{2}{km} de
        bitsnelheid: de intermodale spreiding, de chromatische spreiding
        (laserbron) of de demping? Geef de begrenzende bitsnelheid.
  \item Wat begrenst voor een monomodevezel over \qty{100}{km} met de
        laserbron, en bij welke bitsnelheid?
  \item De monomodeverbinding draagt $80$ lasers met golflengtes
        \qty{0.8}{nm} uit elkaar (golflengtemultiplexing), elk op de
        bitsnelheid van de vorige vraag. Totale capaciteit?
  \item Vorm het product bitsnelheid maal lengte van de stapindexverbinding
        en van één monomodekanaal; met welke factor wint de monomodevezel,
        en welk enkel natuurkundig feit is daarvoor verantwoordelijk?
\end{enumerate}
\end{problem}
