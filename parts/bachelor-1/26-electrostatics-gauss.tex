\chapter{Electrostatics: Field and Gauss's Law}\label{ch:b1:electrostatics-gauss}

Rub a balloon on a sweater and it sticks to the wall; comb dry hair and
the comb bends a thin stream of water. Behind both is the force between
\omterm{def:b1:electrostatics-gauss:charge}{electric charges} at rest, \omterm{thm:b1:electrostatics-gauss:coulomb}{Coulomb's law} --- an inverse-square law like
gravity, but $10^{36}$ times stronger between two protons and of both
signs, so that matter is neutral to fantastic precision and the
residual effects are what we call chemistry, materials and lightning.
This chapter builds the concept that makes the law tractable, the
\omterm{def:b1:electrostatics-gauss:field}{electric field}, shows how to read a field from its symmetries, and
derives the one theorem that computes fields of symmetric
distributions in three lines: \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law}. Its gravitational twin
comes free at the end.

\section{Charges and Coulomb's law}

\begin{definition}[Electric charge]\label{def:b1:electrostatics-gauss:charge}
\index{electric charge}\index{elementary charge}\index{charge density}
\emph{Electric charge} $q$ (coulomb, \unit{C}) is a property of matter
of either sign; it is conserved (the total charge of an isolated system
never changes) and quantized in units of the \emph{elementary charge}
$e = \qty{1.602e-19}{C}$ (proton $+e$, electron $-e$). A macroscopic
distribution is described by a \emph{charge density}: linear $\lambda$
(\unit{C/m}), surface $\sigma$ (\unit{C/m^2}) or volume $\rho$
(\unit{C/m^3}), so that a small element carries $\dd q = \lambda\,\dd l$,
$\sigma\,\dd S$ or $\rho\,\dd\tau$.
\end{definition}

\begin{theorem}[Coulomb's law]\label{thm:b1:electrostatics-gauss:coulomb}
\index{Coulomb's law}\index{permittivity of vacuum}
Two point charges $q_1$ at $M_1$ and $q_2$ at $M_2$ at rest in vacuum
exert on each other the forces
\[
\vect F_{1 \to 2} = \frac{q_1q_2}{4\pi\varepsilon_0\,r^2}\,\vect e_{12} = -\vect F_{2 \to 1},
\qquad r = M_1M_2,\quad \vect e_{12} = \frac{\vect{M_1M_2}}{r},
\]
repulsive for like charges, attractive for unlike ones, with the
\emph{\omterm{thm:b1:electrostatics-gauss:coulomb}{permittivity of vacuum}} $\varepsilon_0 = \qty{8.854e-12}{F/m}$, i.e.\
$1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$. Forces from several charges add
(superposition).
\end{theorem}
\admitted

\begin{example}[How strong]\label{ex:b1:electrostatics-gauss:strong}
Two protons \qty{1}{fm} apart repel with $8.99 \times 10^9 \times (1.6 \times
10^{-19})^2/10^{-30} = \qty{230}{N}$ --- the weight of a child, between two
particles of $10^{-27}$~kg; their gravitational attraction is
$10^{36}$ times smaller. Two grams of electrons at one end of a room
and the matching protons at the other would attract with $10^{25}$~N.
Matter is neutral because it has no choice: any imbalance is corrected
by forces no structure can resist.
\end{example}

\section{The electric field}

\begin{definition}[Electrostatic field]\label{def:b1:electrostatics-gauss:field}
\index{electric field}\index{field line}
A distribution of charges at rest creates at every point $M$ of space
an \emph{electric field} $\vect E(M)$, defined by the force it exerts on
a test charge $q$ placed at $M$: $\vect F = q\vect E(M)$ (\unit{V/m} $=$
\unit{N/C}). A point charge $q_1$ at $O$ creates
\[
\vect E(M) = \frac{q_1}{4\pi\varepsilon_0 r^2}\,\vect e_r , \qquad r = OM ,
\]
radial, outward for $q_1 > 0$; a distribution creates the vector sum of
the fields of its elements. A \emph{field line} is a curve tangent at
each point to $\vect E$, oriented along it.
\end{definition}

\begin{proposition}[Reading a field map]\label{prop:b1:electrostatics-gauss:lines}
\omterm{def:b1:electrostatics-gauss:field}{Field lines} leave positive charges and end on negative ones (or at
infinity); they never cross (the field is single-valued), and they
crowd where the field is strong. Where two charges are present, the
number of lines drawn from each is proportional to its charge.
\end{proposition}

\begin{example}[Ring and segment by direct summation]\label{ex:b1:electrostatics-gauss:ring}
On the axis of a ring of radius $R$ carrying $Q$ uniformly, at distance
$z$ from its centre, each element $\dd q$ contributes $\dd q/4\pi\varepsilon_0(R^2 +
z^2)$ along the line joining it to $M$; the components perpendicular
to the axis cancel in pairs and the axial ones add with the factor
$z/\sqrt{R^2 + z^2}$:
\[
E_z = \frac{Qz}{4\pi\varepsilon_0(R^2 + z^2)^{3/2}} ,
\]
zero at the centre, maximal at $z = R/\sqrt2$, and $Q/4\pi\varepsilon_0z^2$ far
away. A uniformly charged disk is a pile of rings (\cref{pb:b1:electrostatics-gauss:1}).
Direct summation works whenever the geometry allows the integral; the
rest of the chapter is about avoiding it.
\end{example}

\section{Symmetries and invariances}

\begin{proposition}[Symmetry principle]\label{prop:b1:electrostatics-gauss:symmetry}
\index{symmetry (of a field)}\index{plane of symmetry}\index{plane of antisymmetry}
The field has the symmetries of its sources. If a plane $\Pi$ is a
\emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{plane of symmetry}} of the charge distribution (the mirror image
of the distribution is itself), then at every point of $\Pi$ the field
lies \emph{in} $\Pi$; if $\Pi$ is a \emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{plane of antisymmetry}}
(the mirror image is the distribution with all charges reversed), the
field at points of $\Pi$ is \emph{perpendicular} to $\Pi$. If the
distribution is invariant under a translation or a rotation, so is the
field: its components do not depend on the corresponding coordinate.
\end{proposition}

\begin{proof}
$\vect E$ is a sum of terms $\dd q\,\vect e/4\pi\varepsilon_0r^2$; the mirror
image of the sum is the sum of the images, and the image of the field
of a distribution is the field of the image distribution (a vector
built from positions and charges transforms like the positions). On a
symmetry plane the field equals its own mirror image, hence lies in the
plane; on an antisymmetry plane it equals the opposite of its image,
hence is normal to it. Invariance is the same argument with a
translation or rotation.
\end{proof}

\begin{method}[Finding the shape of a field before computing it]\label{meth:b1:electrostatics-gauss:symmetry}
For the point $M$ where the field is wanted: (1) list the planes of
symmetry of the sources that contain $M$ --- the field lies on their
intersection; two such planes fix the direction. (2) List the
invariances --- they tell which coordinates $\vect E$ depends on. Thus a
uniformly charged infinite wire gives $\vect E = E(r)\,\vect e_r$ in
\omterm{def:b1:point-kinematics:cylindrical}{cylindrical coordinates}, a sphere $\vect E = E(r)\,\vect e_r$ in spherical
ones, an infinite plane $\vect E = E(z)\,\vect e_z$ with $E(-z) = -E(z)$.
Only then compute.
\end{method}

\section{Flux and Gauss's law}

\begin{definition}[Flux of the field]\label{def:b1:electrostatics-gauss:flux}
\index{flux of the electric field}
Cut a surface $S$ into small elements of area $\dd S$, each with a unit
normal $\vect n$ (for a closed surface, the outward normal). The
\emph{flux} of $\vect E$ through $S$ is the sum
\[
\Phi = \sum \vect E\cdot\vect n\,\dd S = \sum E_n\,\dd S \qquad (\unit{V.m}):
\]
the field's normal component weighted by area --- the ``number of \omterm{def:b1:electrostatics-gauss:field}{field
lines}'' crossing $S$, counted positively when they cross along
$\vect n$.
\end{definition}

\begin{theorem}[Gauss's law]\label{thm:b1:electrostatics-gauss:gauss}
\index{Gauss's law}
The \omterm{def:b1:electrostatics-gauss:flux}{flux} of the electrostatic field through any closed surface $S$
equals the total charge enclosed by $S$ divided by $\varepsilon_0$:
\[
\Phi_{\text{out}} = \frac{Q_{\text{inside}}}{\varepsilon_0} .
\]
Charges outside $S$ contribute nothing to the \omterm{def:b1:electrostatics-gauss:flux}{flux} (their lines go in
and come out again).
\end{theorem}

\begin{proof}
For a point charge $q$ at the centre of a sphere of radius $r$:
$\vect E$ is radial, of magnitude $q/4\pi\varepsilon_0r^2$ everywhere on the
sphere, so $\Phi = E \times 4\pi r^2 = q/\varepsilon_0$ --- independent of $r$,
because the field falls as $1/r^2$ exactly as the area grows. That the
same holds for any closed surface around $q$ (the \omterm{def:b1:electrostatics-gauss:flux}{flux} through a
surface element depends only on the solid angle it subtends from $q$),
that an outside charge gives zero (it subtends every solid angle twice,
with opposite signs), and that the \omterm{def:b1:electrostatics-gauss:flux}{fluxes} of several charges add, is
admitted here; the full proof is in the Year 2 volume.
\end{proof}

\begin{method}[Using Gauss's law]\label{meth:b1:electrostatics-gauss:gauss}
(1) From the symmetries, write the form of $\vect E$ (direction and the
variable it depends on). (2) Choose a closed \emph{Gaussian surface}
through the point $M$ on which $\vect E$ is everywhere either normal with
the same magnitude $E(M)$, or tangent: the \omterm{def:b1:electrostatics-gauss:flux}{flux} is then $E(M)$ times the
area of the normal part. (3) Count the charge inside. (4) Solve for
$E(M)$. It works for spheres, infinite cylinders and infinite planes
--- and for nothing else without further tools.
\end{method}

\begin{proposition}[The three classic fields]\label{prop:b1:electrostatics-gauss:three}
\index{field of a charged sphere}\index{field of a charged wire}\index{field of a charged plane}
\begin{enumerate}
  \item Sphere of radius $R$, charge $Q$ uniformly spread in its
        volume: $E(r) = \dfrac{Q}{4\pi\varepsilon_0r^2}$ outside ($r > R$) ---
        as if all the charge sat at the centre --- and $E(r) =
        \dfrac{Qr}{4\pi\varepsilon_0R^3}$ inside, growing linearly from zero;
        continuous at $r = R$. If the charge sits on the surface only,
        the inside field is zero.
  \item Infinite wire of linear charge $\lambda$: $E(r) =
        \dfrac{\lambda}{2\pi\varepsilon_0 r}$, radial.
  \item Infinite plane of surface charge $\sigma$: $E = \dfrac{\sigma}{2\varepsilon_0}$
        on either side, normal to the plane, pointing away from it for
        $\sigma > 0$, \emph{independent of the distance}. Across the plane
        the normal component jumps by $\sigma/\varepsilon_0$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Sphere of radius $r$: \omterm{def:b1:electrostatics-gauss:flux}{flux} $4\pi r^2E(r)$; charge inside $Q$ if
$r > R$, $Q(r/R)^3$ if $r < R$. (2) Cylinder of radius $r$, length $h$:
\omterm{def:b1:electrostatics-gauss:flux}{flux} through the lateral surface $2\pi rh\,E(r)$, none through the ends
(field tangent); charge $\lambda h$. (3) Pillbox of cross-section $S$
straddling the plane: \omterm{def:b1:electrostatics-gauss:flux}{flux} $2SE$ through the two ends (the field is odd
in $z$, so both count positively), none through the side; charge
$\sigma S$.
\end{proof}

\begin{example}[Two planes: the plane capacitor]\label{ex:b1:electrostatics-gauss:twoplanes}
\index{plane capacitor!field}
Two parallel planes carrying $+\sigma$ and $-\sigma$: each creates
$\sigma/2\varepsilon_0$ pointing away from it (toward it for the negative one).
Between the planes the two fields add to $\sigma/\varepsilon_0$, directed from
$+$ to $-$; outside they cancel. With $\sigma = \qty{1}{\micro C/m^2}$, $E =
10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$ between, zero outside: the
field of a charged capacitor is confined to its gap
(\cref{ch:b1:potential-capacitors}).
\end{example}

\section{The gravitational analogy}

\begin{proposition}[Gauss's law for gravity]\label{prop:b1:electrostatics-gauss:gravity}
\index{Gauss's law!for gravitation}\index{gravitational field}
Newton's law $\vect F = -Gm_1m_2\,\vect e_{12}/r^2$ has the form of
Coulomb's with $q \to m$ and $1/4\pi\varepsilon_0 \to -G$. The \omterm{prop:b1:newton-dynamics:weight}{gravitational
field} $\vect g$ (force per unit mass) therefore obeys the same symmetry
rules and
\[
\Phi_{\text{out}}(\vect g) = -4\pi G\,M_{\text{inside}} .
\]
In particular a spherically symmetric body attracts outside points as
if its mass were at its centre, and the field inside a uniform sphere
of radius $R$ and mass $M$ is $g(r) = GMr/R^3$, linear in $r$.
\end{proposition}

\begin{proof}
Replace $Q/\varepsilon_0$ by $-4\pi GM$ in the proof of
\cref{prop:b1:electrostatics-gauss:three}; the minus sign says the \omterm{def:b1:electrostatics-gauss:flux}{flux}
enters (gravity attracts).
\end{proof}

\begin{example}[Inside the Earth]\label{ex:b1:electrostatics-gauss:earth}
For a uniform Earth, $g$ would fall linearly from \qty{9.8}{m/s^2} at the
surface to zero at the centre; a stone dropped down a tunnel through
the planet would oscillate harmonically with period $2\pi\sqrt{R/g} =
\qty{84}{min}$ --- the period of a grazing satellite. The real Earth has
a dense core, and $g$ actually \emph{rises} to \qty{10.7}{m/s^2} at the
core--mantle boundary before falling (\cref{pb:b1:electrostatics-gauss:1}).
Newton needed the shell theorem to apply his law to planets; \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's
law} gives it in a line.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:1}
Electric force and gravitational force between two protons
\qty{2.0}{fm} apart; their ratio. Why then do nuclei hold together?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:2}
Field at \qty{1.0}{m} from a \qty{1.0}{\micro C} point charge. The Earth's
surface carries a field of \qty{100}{V/m} pointing down: sign and value of
the Earth's charge, treating it as a sphere of radius \qty{6370}{km}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:3}
Two charges $+q$ at $x = \pm a$ on the $x$ axis. Field on the $x$ axis
for $|x| > a$ and on the $y$ axis; show that on the $y$ axis $E_y$ is
maximal at $y = a/\sqrt2$. Sketch the lines near the origin.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:4}
Charges $+2q$ at $x = 0$ and $-q$ at $x = a$. Where on the $x$ axis does
the field vanish? How many \omterm{def:b1:electrostatics-gauss:field}{field lines} leave $+2q$ for each one
arriving at $-q$, and where do the others go?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:5}
A disk of radius $R$ carries $\sigma$ uniformly. From the ring result,
show that on its axis $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 +
R^2}}\Bigr)$ for $z > 0$; check the two limits $z \ll R$ and $z \gg R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:6}
A gold nucleus ($Z = 79$, $R = \qty{7.0}{fm}$) as a uniformly charged
sphere: field at the surface, at $R/2$, at \qty{1}{pm}. Compare with the

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:7}
Field of an infinite wire by \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law}. A \omterm{def:b1:dc-circuits:voltage}{high-voltage} conductor of
radius \qty{1.0}{cm}: linear charge at which the field at its surface
reaches the breakdown value \qty{3e6}{V/m} (corona onset). Field at

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:8}
Two parallel plates of area \qty{1.0}{m^2} carry $\pm\qty{1.0}{\micro C}$.
Field between them; force on one plate (the field acting on it is that
of the other plate only --- why?); pressure on the plates.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:9}
Coaxial cable: inner conductor of radius $a$ with $+\lambda$, outer

\omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law}. $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, $\lambda = \qty{10}{nC/m}$:

in the cavity: what force does it feel?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:11}
\emph{Gravity inside the Earth.} Core: radius \qty{3480}{km}, density
\qty{11000}{kg/m^3}; mantle to \qty{6370}{km}, density \qty{4500}{kg/m^3}.
(a) Mass of core, of mantle, total (compare with \qty{5.97e24}{kg}).
(b) $g$ at the surface and at the core--mantle boundary. (c) Show that

\end{exercise}

\section{Problem: The electric Earth}

\begin{problem}[{Weekend problem --- the fair-weather field, the
thundercloud, the lightning stroke, and Gauss's law turned on gravity}]
\label{pb:b1:electrostatics-gauss:1}
Data: $\varepsilon_0 = \qty{8.85e-12}{F/m}$, $1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$,
$e = \qty{1.6e-19}{C}$, Earth's radius $R_E = \qty{6.37e6}{m}$, surface
$\qty{5.1e14}{m^2}$, $G = \qty{6.67e-11}{SI}$, $M_E = \qty{5.97e24}{kg}$.

\medskip
\noindent\textbf{Part I --- The fair-weather field.}
In clear weather the field at the ground is $E_0 = \qty{100}{V/m}$,
pointing \emph{down}.
\begin{enumerate}
  \item Direction of the force on a free electron at the ground; sign
        of the Earth's charge.
  \item Treat the Earth as a uniformly charged sphere: from \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law},
        its total charge.
  \item Surface \omterm{def:b1:electrostatics-gauss:charge}{charge density} of the ground; number of excess electrons
        per square centimetre.
  \item Force on a dust grain of mass \qty{1}{\micro g} carrying $100$
        \omterm{def:b1:electrostatics-gauss:charge}{elementary charges}; compare with its weight.
  \item Measurements show the field falling to about \qty{5}{V/m} at
        \qty{10}{km}. Apply \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law} to a vertical column of
        cross-section \qty{1}{m^2} between the ground and \qty{10}{km}: sign
        and amount of the charge the air holds per square metre.
  \item Mean volume \omterm{def:b1:electrostatics-gauss:charge}{charge density} of that air, and the corresponding
        number of \omterm{def:b1:electrostatics-gauss:charge}{elementary charges} per cubic metre. (The air contains
        about $10^9$ ions of each sign per cubic metre: what fraction of
        imbalance is this?)
  \item The air has a small conductivity $\gamma = \qty{2e-14}{S/m}$, so a
        current density $\gamma E_0$ flows down. Total fair-weather
        current over the globe; time it would take to neutralize the
        Earth's charge. Conclude.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The thundercloud.}
Model a storm cloud as two horizontal disks of radius $R = \qty{3}{km}$:
$-Q$ at altitude \qty{3}{km}, $+Q$ at \qty{9}{km}, with $Q = \qty{40}{C}$.
\begin{enumerate}[resume]
  \item Field on the axis of a ring of radius $a$ carrying $q$, at
        distance $z$ from its centre (symmetry, then summation).
  \item Deduce, by summing over rings, the field on the axis of a disk
        of radius $R$ and surface charge $\sigma$:
        $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$.
  \item Limits $z \ll R$ and $z \gg R$; interpret each.
  \item Surface charge of the lower disk; field it creates at the ground
        directly beneath (direction and value).
  \item Field of the upper disk at the same point; net field at the
        ground under the storm. The ground is a conductor and its induced
        charges roughly double the result (admitted): compare with the
        \qty{10}{kV/m} measured under storms and with the fair-weather
        value.
  \item \omterm{def:b1:electrostatics-gauss:flux}{Flux} of the field through a closed surface enclosing the whole
        cloud; through one enclosing the lower disk only.
  \item Field midway between the two disks (\qty{6}{km}), and its
        direction; compare with the breakdown field of air at that
        altitude, about \qty{1.5e6}{V/m}.
  \item If the \qty{40}{C} were concentrated in a sphere of radius
        \qty{500}{m}: field at its surface. Why is lightning initiation
        still an open question?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The stroke.}
A return stroke carries \qty{5}{C} to the ground in \qty{50}{\micro s}
through a channel \qty{3}{km} long; take the field along the channel
as $10^5$~V/m.
\begin{enumerate}[resume]
  \item Mean current.
  \item Work of the electric force on the charge transferred; power
        during the stroke; compare with humanity's \omterm{thm:b1:sinusoidal-impedance:power}{average power}
        consumption, about \qty{2e13}{W}.
  \item The channel has a radius of \qty{2}{cm}: mass of air in it
        ($\rho = \qty{1.2}{kg/m^3}$), and the temperature it would reach if
        all the energy heated it at constant volume ($c_V =
        \qty{720}{J/(kg.K)}$). Actual channel temperatures are near
        \qty{30000}{K}: where does the rest go?
  \item Thunder: delay per kilometre, and why the sound of a
        \qty{3}{km} channel rumbles for seconds.
  \item Worldwide about $50$ cloud-to-ground strokes occur every second:
        current they carry downward; compare with Part~I and complete the
        picture of the ``global circuit''.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law} turned on gravity.}
\begin{enumerate}[resume]
  \item Write \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law} for the \omterm{prop:b1:newton-dynamics:weight}{gravitational field} $\vect g$ and
        justify the constant by the case of a point mass.
  \item Uniform Earth: $g(r)$ inside; value at $r = R_E/2$.
  \item Real Earth, core of radius \qty{3480}{km} and density
        \qty{11000}{kg/m^3}: $g$ at the core--mantle boundary; compare with
        the surface.
  \item A stone dropped down a frictionless tunnel through a uniform
        Earth: equation of motion, period of the oscillation, speed at
        the centre.
  \item Sum up: in each of the four parts, what did \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law} replace,
        and which four numbers are worth remembering?
\end{enumerate}
\end{problem}
