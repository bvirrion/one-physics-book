\chapter{Work, Energy, and Potential Energy}\label{ch:b1:work-and-energy}

A cyclist stops pedaling at the top of a hill and arrives at the bottom
at a speed that depends on the drop, not on the shape of the road. A
bungee jumper falls sixty meters and stops, for an instant, exactly
where the cord's pull has eaten all the speed the fall had given. A
hydrogen atom bound to a chlorine atom vibrates in a well whose depth
is the energy it costs to tear the molecule apart. In each case a
single number is exchanged between forms --- motion, height, stretch,
bond --- and summed, it does not change. This chapter builds that
bookkeeping from Newton's second law: \omterm{def:b1:work-and-energy:work}{work} and power, the \omterm{thm:b1:work-and-energy:ke}{kinetic
energy} theorem, potential energies for the forces that allow one, and
the \omterm{thm:b1:work-and-energy:em}{conservation of mechanical energy} that turns many dynamical problems
into one-line algebra and explains, through the shape of a \omterm{def:b1:work-and-energy:turning}{potential
well}, which positions are stable.

\section{Power and work}

\begin{definition}[Power and work of a force]\label{def:b1:work-and-energy:work}
\index{power (of a force)}\index{work}
The \emph{power} of a force $\vect F$ acting on a point moving at
velocity $\vect v$ is
\[
\mathcal P = \vect F\cdot\vect v \qquad (\text{watts}),
\]
and its \emph{work} between times $t_1$ and $t_2$ (positions $A$ and $B$)
is the accumulated power,
\[
W_{A\to B} = \int_{t_1}^{t_2}\vect F\cdot\vect v\,\dd t
= \int_A^B \vect F\cdot\dd\vect\ell \qquad (\text{joules}),
\]
the sum of the \emph{elementary works} $\delta W = \vect F\cdot\dd\vect\ell$
along the elementary displacements $\dd\vect\ell = \vect v\,\dd t$ of the
\omterm{def:b1:point-kinematics:frame}{trajectory}. Work is positive if the force pushes along the motion
(\emph{motive}), negative if against it (\emph{resistive}), zero if
perpendicular.
\end{definition}

\begin{proposition}[Works one meets]\label{prop:b1:work-and-energy:works}
\begin{itemize}
  \item Constant force: $W_{A\to B} = \vect F\cdot\vect{AB}$, whatever the
        path. Weight: $W = mg(z_A - z_B)$, $z$ counted upward.
  \item Spring of stiffness $k$, from elongation $x_A$ to $x_B$:
        $W = \tfrac12 k(x_A^2 - x_B^2)$.
  \item \omterm{prop:b1:newton-dynamics:friction}{Normal reaction} of a fixed surface, string tension on a body
        moving perpendicular to it: $W = 0$ (force $\perp$ displacement).
  \item Kinetic friction and drag: $W < 0$ always (the force opposes the
        relative motion), and depends on the path length.
\end{itemize}
\end{proposition}

\begin{proof}
Constant $\vect F$: $\int\vect F\cdot\dd\vect\ell = \vect F\cdot\int\dd\vect\ell
= \vect F\cdot\vect{AB}$; with $\vect F = -mg\vect e_z$, $\vect F\cdot\vect{AB}
= -mg(z_B - z_A)$. Spring: $\vect F\cdot\dd\vect\ell = -kx\,\dd x$, and
$\int_{x_A}^{x_B}-kx\,\dd x = \tfrac12 k(x_A^2 - x_B^2)$. Friction: $\vect T
= -\mu N\vect v/v$, so $\vect T\cdot\vect v = -\mu Nv < 0$.
\end{proof}

\section{The kinetic energy theorem}

\begin{theorem}[Kinetic energy and power theorems]\label{thm:b1:work-and-energy:ke}
\index{kinetic energy}\index{kinetic energy theorem}
In an \omterm{thm:b1:newton-dynamics:laws}{inertial frame}, the \emph{\omterm{thm:b1:work-and-energy:ke}{kinetic energy}} $E_k = \tfrac12 mv^2$ of
a \omterm{def:b1:point-kinematics:frame}{point particle} obeys
\[
\frac{\dd E_k}{\dd t} = \sum\mathcal P_i = \Big(\sum\vect F_i\Big)\cdot\vect v,
\qquad
E_k(B) - E_k(A) = \sum_i W_{i,A\to B}:
\]
the change of \omterm{thm:b1:work-and-energy:ke}{kinetic energy} between two positions equals the total \omterm{def:b1:work-and-energy:work}{work}
of all the forces applied along the way.
\end{theorem}

\begin{proof}
Dot Newton's second law with $\vect v$: $m\vect a\cdot\vect v =
\tfrac{\dd}{\dd t}(\tfrac12 m\vect v\cdot\vect v) = \dd E_k/\dd t$, while
$(\sum\vect F_i)\cdot\vect v = \sum\mathcal P_i$. Integrate over time.
\end{proof}

\begin{example}[Braking distance]\label{ex:b1:work-and-energy:braking}
A \qty{1000}{kg} car at \qty{100}{km/h} ($E_k = \qty{386}{kJ}$) braked by
a constant \qty{7.0}{kN} force stops after $d = E_k/F = \qty{55}{m}$:
the distance grows as the \emph{square} of the speed. Since the
maximal braking force is $\mu_smg$ (\cref{ch:b1:newton-dynamics}), the
shortest stopping distance is $v^2/2\mu_sg$ whatever the car's mass ---
and it doubles on a wet road where $\mu_s$ halves.
\end{example}

\section{Conservative forces and potential energy}

\begin{definition}[Conservative force; potential energy]\label{def:b1:work-and-energy:conservative}
\index{conservative force}\index{potential energy}
A force is \emph{conservative} if its \omterm{def:b1:work-and-energy:work}{work} between two points does not
depend on the path followed. Equivalently, there exists a function of
position, the \emph{potential energy} $E_p$, such that
\[
W_{A\to B} = E_p(A) - E_p(B) = -\Delta E_p ,
\qquad \delta W = -\dd E_p ,
\]
$E_p$ being defined up to an additive constant (a choice of origin).
\end{definition}

\begin{proposition}[Force from potential energy]\label{prop:b1:work-and-energy:gradient}
\index{gradient}
A \omterm{def:b1:work-and-energy:conservative}{conservative force} derives from its \omterm{def:b1:work-and-energy:conservative}{potential energy}:
\[
\vect F = -\overrightarrow{\operatorname{grad}}\,E_p
= -\frac{\partial E_p}{\partial x}\,\vect e_x
- \frac{\partial E_p}{\partial y}\,\vect e_y
- \frac{\partial E_p}{\partial z}\,\vect e_z ,
\]
and in one dimension $F_x = -\dd E_p/\dd x$: the force points toward
decreasing \omterm{def:b1:work-and-energy:conservative}{potential energy}, ``downhill'' on the graph of $E_p$.
\end{proposition}

\begin{proof}
$\delta W = F_x\dd x + F_y\dd y + F_z\dd z = -\dd E_p = -(\partial_xE_p\,\dd x
+ \partial_yE_p\,\dd y + \partial_zE_p\,\dd z)$ for every displacement:
identify the coefficients (the differential of a function of several
variables, from the mathematics course).
\end{proof}

\begin{proposition}[The usual potential energies]\label{prop:b1:work-and-energy:eps}
\index{potential energy!gravitational}\index{potential energy!elastic}
\begin{itemize}
  \item Weight: $E_p = mgz + \text{const}$ ($z$ upward).
  \item Spring: $E_p = \tfrac12 k(\ell - \ell_0)^2 + \text{const}$.
  \item Newtonian gravitation of a mass $M$: $E_p = -GMm/r + \text{const}$,
        the constant usually chosen so that $E_p \to 0$ at infinity.
  \item Electrostatic force on a charge $q$ in a potential $V$:
        $E_p = qV$ (\cref{ch:b1:potential-capacitors}).
\end{itemize}
Kinetic friction and fluid drag are \emph{not} \omterm{def:b1:work-and-energy:conservative}{conservative} (their \omterm{def:b1:work-and-energy:work}{work}
depends on the path, and is always negative).
\end{proposition}

\begin{proof}
Each is read off the \omterm{def:b1:work-and-energy:work}{works} of \cref{prop:b1:work-and-energy:works}:
$W = -\Delta E_p$. For gravitation, $\vect F = -GMm/r^2\,\vect e_r$ and
$\delta W = -GMm\,\dd r/r^2 = -\dd(-GMm/r)$.
\end{proof}

\section{Mechanical energy}

\begin{theorem}[Mechanical energy]\label{thm:b1:work-and-energy:em}
\index{mechanical energy}\index{conservation of mechanical energy}
Let $E_m = E_k + E_p$ be the \emph{\omterm{thm:b1:work-and-energy:em}{mechanical energy}}, where $E_p$ is the
sum of the potential energies of the \omterm{def:b1:work-and-energy:conservative}{conservative forces}. Then
\[
\Delta E_m = W_{\text{nc}} ,
\]
the total \omterm{def:b1:work-and-energy:work}{work} of the \omterm{def:b1:work-and-energy:conservative}{non-conservative} forces. If those do no \omterm{def:b1:work-and-energy:work}{work}
(no friction, or constraint forces perpendicular to the motion), $E_m$
is \emph{conserved}: the motion is \emph{\omterm{def:b1:work-and-energy:conservative}{conservative}}.
\end{theorem}

\begin{proof}
$\Delta E_k = W_{\text{c}} + W_{\text{nc}} = -\Delta E_p + W_{\text{nc}}$.
\end{proof}

\begin{method}[Using energy]\label{met:b1:work-and-energy:method}
When a problem asks for a \emph{speed at a position} (not a time, not a
force), try energy first:
\begin{enumerate}
  \item list the forces; check which \omterm{def:b1:work-and-energy:work}{work} and which are \omterm{def:b1:work-and-energy:conservative}{conservative};
  \item write $E_m$ at the two positions, with a clear origin for $E_p$;
  \item equate them (adding $W_{\text{nc}}$ if friction acts) and solve
        for the unknown speed or position.
\end{enumerate}
Energy never gives the time or the constraint forces; for those, go
back to Newton's law --- often with the speed that energy just provided.
\end{method}

\begin{example}[The loop's speed law]\label{ex:b1:work-and-energy:loop}
On the frictionless loop of \cref{pb:b1:point-kinematics:1} the track's
reaction is normal to the motion and \omterm{def:b1:work-and-energy:work}{works} not: $E_m$ is conserved.
With $z = R(1 - \cos\theta)$ above the bottom, $\tfrac12 mv^2 + mgR(1 -
\cos\theta) = \tfrac12 mv_0^2$: the speed law $v^2 = v_0^2 - 2gR(1 -
\cos\theta)$ used there, now derived in one line. The minimum entry
speed $\sqrt{5gR}$ then follows from the top condition
$v^2 \geq gR$ of \cref{ex:b1:newton-dynamics:circles}.
\end{example}

\begin{example}[Energy lost to friction]\label{ex:b1:work-and-energy:friction}
A block slides down a slope of angle $\alpha$ and height $h$ with
kinetic friction $\mu_d$: $W_{\text{nc}} = -\mu_dmg\cos\alpha \times
h/\sin\alpha = -\mu_dmgh\cot\alpha$, so $\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha)$.
For $h = \qty{2.0}{m}$, $\alpha = 30^\circ$, $\mu_d = 0.20$: $v =
\qty{5.1}{m/s}$ instead of $\qty{6.3}{m/s}$; a third of the \omterm{def:b1:work-and-energy:conservative}{potential
energy} became heat.
\end{example}

\section{Equilibrium, stability, and phase portraits}

\begin{proposition}[Equilibrium and stability in one dimension]\label{prop:b1:work-and-energy:stability}
\index{equilibrium!stable, unstable}\index{small oscillations}
A particle moving on an axis under a \omterm{def:b1:work-and-energy:conservative}{conservative force} of \omterm{def:b1:work-and-energy:conservative}{potential
energy} $E_p(x)$:
\begin{itemize}
  \item is in \emph{equilibrium} at $x_0$ iff $F(x_0) = -E_p'(x_0) = 0$:
        the extrema of $E_p$;
  \item the equilibrium is \emph{stable} if $E_p$ has a local minimum
        there ($E_p''(x_0) > 0$): a small displacement produces a
        restoring force; \emph{unstable} at a maximum;
  \item near a stable equilibrium, small motions are harmonic, with
        \[
        \omega_0^2 = \frac{E_p''(x_0)}{m} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Taylor's formula (mathematics course): $E_p(x) \approx E_p(x_0) +
\tfrac12 E_p''(x_0)(x - x_0)^2$, so $F \approx -E_p''(x_0)(x - x_0)$ and
$m\ddot x = -E_p''(x_0)(x - x_0)$ --- a spring of stiffness $E_p''(x_0)$.
For $E_p'' < 0$ the ``stiffness'' is negative: exponential departure.
\end{proof}

\begin{definition}[Turning points; bound and free motion]\label{def:b1:work-and-energy:turning}
\index{turning point}\index{bound motion}\index{potential well}\index{potential barrier}
For a \omterm{def:b1:work-and-energy:conservative}{conservative} one-dimensional motion of energy $E_m$, the particle
can only be where $E_p(x) \leq E_m$ (since $E_k \geq 0$); the points where
$E_p = E_m$ are \emph{turning points}, where the velocity vanishes and
reverses. Trapped between two turning points in a \emph{potential well},
the motion is \emph{bound} and periodic; able to reach infinity, it is
\emph{free}; a \emph{potential barrier} higher than $E_m$ cannot be
crossed.
\end{definition}

\begin{definition}[Phase portrait]\label{def:b1:work-and-energy:phase}
\index{phase portrait}
The \emph{phase portrait} of a one-dimensional motion is the family of
curves traced by the point $(x, \dot x)$ in the \emph{phase plane}. For
a \omterm{def:b1:work-and-energy:conservative}{conservative} system each curve is a level line $\tfrac12 m\dot x^2 +
E_p(x) = E_m$: closed curves around stable equilibria (oscillations),
open curves for free motion, and, through unstable equilibria, the
\emph{separatrices} that divide the two. Curves are traversed clockwise
($\dot x > 0$ means $x$ increasing) and never cross.
\end{definition}

\begin{example}[The pendulum's landscape]\label{ex:b1:work-and-energy:pendulum}
$E_p = -mg\ell\cos\theta$: minimum at $\theta = 0$, $E_p'' = mg\ell$, so
$\omega_0^2 = g\ell/(m\ell^2) = g/\ell$ --- the small-oscillation result of
\cref{ex:b1:newton-dynamics:pendulum} read off the well's curvature;
maximum at $\theta = \pi$, the inverted pendulum, unstable. Energy
$E_m < mg\ell$: oscillation between $\pm\theta_{\max}$; $E_m > mg\ell$: it
goes over the top and rotates; $E_m = mg\ell$: the separatrix, an
infinitely slow approach to the top.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:work-and-energy:1}
A \qty{20}{kg} crate is lifted \qty{3.0}{m} at constant speed in
\qty{5.0}{s}. \omterm{def:b1:work-and-energy:work}{Work} of the lifting force, of the weight, and power
supplied. The same crate pushed \qty{3.0}{m} along a floor with
$\mu_d = 0.30$: \omterm{def:b1:work-and-energy:work}{work} of friction.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:work-and-energy:2}
\omterm{thm:b1:work-and-energy:ke}{Kinetic energy} of a \qty{1000}{kg} car at \qty{100}{km/h}; stopping
distance with a constant \qty{7.0}{kN} braking force; what happens to
the distance if the speed is doubled?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:work-and-energy:3}
A spring of stiffness \qty{500}{N/m} is compressed by \qty{10}{cm} and
released against a \qty{50}{g} ball on a frictionless horizontal track.
Energy stored; speed of the ball.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:work-and-energy:4}
A pendulum of length \qty{1.0}{m} is released from rest at $60^\circ$:
speed at the bottom by energy; then, with Newton's law, the tension
there (in units of $mg$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:work-and-energy:5}
A block slides from rest down a $30^\circ$ slope of height \qty{2.0}{m},
$\mu_d = 0.20$. Speed at the bottom by the energy theorem; fraction of
the initial \omterm{def:b1:work-and-energy:conservative}{potential energy} lost to friction.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:work-and-energy:6}
Using $E_p = -GMm/r$, show that the speed needed to escape the Earth's
attraction from its surface is $v_e = \sqrt{2GM/R}$, and compute it
($M = \qty{5.97e24}{kg}$, $R = \qty{6.37e6}{m}$). What is the escape
speed from a body with the same density but twice the radius?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:work-and-energy:7}
Bungee: a \qty{70}{kg} jumper, cord of natural length \qty{20}{m} and
stiffness \qty{50}{N/m}, jumps from rest. Find the maximal stretch of
the cord, the total drop, and the maximal tension (in units of the
weight). Where is the speed greatest?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:work-and-energy:8}
A cyclist (\qty{80}{kg} with bike) climbs a $5\%$ slope at \qty{5.0}{m/s}
against a drag $\tfrac12\rho C_xSv^2$ with $C_xS = \qty{0.50}{m^2}$.
Power against gravity, against drag, total.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:work-and-energy:9}
A particle of mass $m$ has $E_p(x) = E_0[(x/a)^4 - 2(x/a)^2]$. Find the
equilibria and their stability, the depth of the wells, and the \omterm{def:b1:wave-propagation:signal}{angular
frequency} of \omterm{prop:b1:work-and-energy:stability}{small oscillations} about a stable one.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:work-and-energy:10}
For the harmonic oscillator $E_p = \tfrac12 kx^2$, show that the phase
curves are ellipses, give their semi-axes for energy $E$, and explain
why all are traversed in the same time.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:work-and-energy:11}
A small object slides from rest at the top of a frictionless hemisphere
of radius $R$. Using energy and Newton's law, find the angle from the
vertical at which it leaves the surface, the height, and its speed
then.
\end{exercise}

\section{Problem: The energy well of a chemical bond}

\begin{problem}[{Weekend problem --- two atoms, one curve: the potential
energy of a bond read as a landscape --- its floor, its walls, the
small vibrations at the bottom, and the energy it takes to climb out}]
\label{pb:b1:work-and-energy:1}
The interaction of the two atoms of a diatomic molecule is modeled by
the \omterm{def:b1:work-and-energy:conservative}{potential energy}
\[
E_p(r) = \epsilon\left[\left(\frac{\sigma}{r}\right)^{12} - 2\left(\frac{\sigma}{r}\right)^{6}\right],
\]
$r$ the distance between the nuclei. For hydrogen chloride take
$\epsilon = \qty{4.4}{eV}$, $\sigma = \qty{0.127}{nm}$, and let the light
hydrogen atom ($m = \qty{1.67e-27}{kg}$) move while the heavy chlorine
stays put. $\qty{1}{eV} = \qty{1.60e-19}{J}$; $k_B = \qty{1.38e-23}{J/K}$;
$N_A = \num{6.02e23}$.

\medskip
\noindent\textbf{Part I --- The landscape.}
\begin{enumerate}
  \item Give the limits of $E_p$ as $r \to 0$ and $r \to \infty$, and
        sketch the curve.
  \item Show that $E_p$ has a single minimum, at $r = \sigma$, of value
        $-\epsilon$.
  \item With $u = (\sigma/r)^6$, show that $E_p = \epsilon(u^2 - 2u)$: a
        parabola in $u$. Recover the minimum from it.
  \item Express the force $F(r)$ (positive if repulsive) and give its
        sign on each side of $\sigma$.
  \item Compute $E_p$ and $F$ at $r = 1.2\sigma$ (in eV and in nN).
  \item What is the dissociation energy of the molecule, in eV and in
        \unit{kJ/mol}? (Measured bond energy of HCl: \qty{431}{kJ/mol}.)
  \item Check the stability criterion at $r = \sigma$: compute
        $E_p''(\sigma)$.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Small vibrations.}
\begin{enumerate}[resume]
  \item Write the second-order Taylor expansion of $E_p$ around $\sigma$
        and identify an effective stiffness $k$.
  \item Compute $k$ (in \unit{N/m}).
  \item Deduce the \omterm{def:b1:wave-propagation:signal}{angular frequency}, the frequency and the wavelength
        of the vibration, and its wavenumber $1/\lambda$ in \unit{cm^{-1}}.
  \item The measured vibration of HCl is at \qty{2990}{cm^{-1}}. Comment
        on the agreement and on what the model gets right.
  \item At \qty{300}{K} the mean vibrational energy is of order $k_BT$.
        Estimate the \omterm{def:b1:wave-propagation:signal}{amplitude} of vibration and compare with $\sigma$.
  \item Deduce the maximal speed of the hydrogen atom in that
        vibration.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Large oscillations and the way out.}
\begin{enumerate}[resume]
  \item Describe the \omterm{def:b1:work-and-energy:phase}{phase portrait}: which energies give closed curves,
        which open ones, and what separates them?
  \item For $E_m = -\epsilon/2$, find the two \omterm{def:b1:work-and-energy:turning}{turning points} (in units
        of $\sigma$).
  \item Show that their midpoint lies beyond $\sigma$, and explain why
        heating (raising the vibrational energy) makes the mean bond
        length grow --- the origin of thermal expansion.
  \item Is the period of large oscillations longer or shorter than
        $2\pi/\omega_0$? Argue from the shape of the walls.
  \item From $E_m = -\epsilon/2$, what energy must be supplied to
        dissociate the molecule? If given as \omterm{thm:b1:work-and-energy:ke}{kinetic energy} of the
        hydrogen atom at $r = \sigma$, what speed is that?
  \item Two atoms approaching from far apart with positive total energy
        cannot form a bound molecule by themselves. Why? What is needed?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- \omterm{def:b1:work-and-energy:work}{Work}, force, and temperature.}
\begin{enumerate}[resume]
  \item Compute the \omterm{def:b1:work-and-energy:work}{work} of the bond force when $r$ goes from $2\sigma$
        to $\sigma$. Is the force motive or resistive there?
  \item Compute the \omterm{def:b1:work-and-energy:work}{work} an external agent must do to stretch the bond
        from $\sigma$ to $1.5\sigma$ at vanishing speed.
  \item Compute $F$ at $r = 0.9\sigma$ and compare with $r = 1.2\sigma$:
        what does the asymmetry tell about compressing versus stretching?
  \item At what temperature would $k_BT$ equal $\epsilon$? Why do
        molecules dissociate at far lower temperatures in practice?
  \item A better model (Morse) changes the shape of the walls but keeps
        a minimum at the bond length. Which of the results above survive
        unchanged in method, and which numbers would change?
  \item Summarize the three ways energy methods were used: to find
        equilibrium, to find the vibration frequency, and to find what
        is needed to escape.
\end{enumerate}
\end{problem}
