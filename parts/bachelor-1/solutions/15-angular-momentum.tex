\section*{Chapter \ref{ch:b1:angular-momentum} --- Angular Momentum}

\begin{solution}{exo:b1:angular-momentum:1}
$Fd = 50 \times 0.30 = \qty{15}{N.m}$; $Fd\sin 60^\circ = \qty{13}{N.m}$;
along the handle the line of action passes through the nut: $0$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:2}
$L = mrv = 5.97\times10^{24} \times 1.50\times10^{11} \times 2.98\times10^4 =
\qty{2.7e40}{kg.m^2/s}$; $\dd A/\dd t = L/2m = rv/2 = \qty{2.2e15}{m^2/s}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:3}
$r = \ell\sin 30^\circ = \qty{0.50}{m}$, $\omega = \qty{3.4}{rad/s}$:
$L_z/m = r^2\omega = \qty{0.84}{m^2/s}$. The tension's line of action
passes through the pivot (zero \omterm{def:b1:angular-momentum:moment}{moment} about any axis through it); the
weight is parallel to the vertical axis (zero \omterm{def:b1:angular-momentum:moment}{moment} about it): $L_z$
is conserved.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:4}
$J_1 = 4.0 + 2 \times 3.0 \times 0.64 = \qty{7.8}{kg.m^2}$, $J_2 = 4.0 +
2 \times 3.0 \times 0.04 = \qty{4.2}{kg.m^2}$; $\omega_2 = \omega_1J_1/J_2 =
1.85\omega_1$: \qty{3.7}{} turns per second. $E_k = L^2/2J$: ratio
$J_1/J_2 = 1.85$; the muscles, pulling the arms inward against the
centrifugal tendency, do the work.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:5}
$L_\Delta = m\ell^2\dot\theta$; tension: zero \omterm{def:b1:angular-momentum:moment}{moment} (meets the axis);
weight: $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{lever arm} $\ell\sin\theta$, restoring).
$m\ell^2\ddot\theta = -mg\ell\sin\theta$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:6}
$r_pv_p = r_av_a$: $v_p/v_a = r_a/r_p = 1.034$; with $(v_p + v_a)/2 \approx
\qty{29.8}{km/s}$: $v_p = \qty{30.3}{km/s}$, $v_a = \qty{29.3}{km/s}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:7}
$L = mrv$ conserved: $v_2 = v_1r_1/r_2 = \qty{4.0}{m/s}$, $\omega_2 = v_2/r_2
= \qty{16}{rad/s}$ (from \qty{4}{rad/s}). $E_1 = \qty{0.40}{J}$, $E_2 =
\qty{1.6}{J}$: the pull does \qty{1.2}{J}. The tension is radial, but the
puck's displacement now has a radial component (it spirals in): the work
$\vect T\cdot\dd\vect\ell$ is no longer zero.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:8}
\omterm{def:b1:angular-momentum:moment}{Moments} about the pivot (lengths from the pivot): child $30 \times 1.5 =
45$ (left); plank's weight at its center, $0.5$ to the right: $20 \times
0.5 = 10$ (right); second child at $x$ to the right: $45x$. Balance:
$45 = 10 + 45x$, $x = \qty{0.78}{m}$. The pivot supports the total weight,
$95 \times 9.81 = \qty{932}{N}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:9}
Masses: $m_1a = m_1g - T_1$, $m_2a = T_2 - m_2g$; pulley: $J\dot\omega =
(T_1 - T_2)R$ with $a = R\dot\omega$. Adding: $a = (m_1 - m_2)g/(m_1 + m_2 +
J/R^2) = 9.81/(3.0 + 1.0) = \qty{2.5}{m/s^2}$ (against \qty{3.3}{m/s^2});
$T_1 = m_1(g - a) = \qty{14.7}{N}$, $T_2 = m_2(g + a) = \qty{12.3}{N}$ ---
unequal, as the pulley needs a net \omterm{def:b1:angular-momentum:moment}{moment}.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:10}
$r_1v_1 = r_2v_{\perp 2}$: $v_{\perp 2} = 40 \times 0.5/10 = \qty{2.0}{km/s}$.
The law of areas gives only the perpendicular component; the radial one
needs energy conservation (\cref{ch:b1:central-forces}).
\end{solution}

\begin{solution}{exo:b1:angular-momentum:11}
$J\omega$ constant with $J \propto R^2$: $T' = T(R'/R)^2 = 25 \times 86400
\times (10^4/7\times10^8)^2 = \qty{4.4e-4}{s}$. Equatorial speed $2\pi R'/T'
= \qty{1.4e8}{m/s}$, half the speed of light: the collapse cannot keep
all the \omterm{def:b1:angular-momentum:L}{angular momentum} --- much is shed (and relativity enters).
\end{solution}

\begin{solution}{exo:b1:angular-momentum:12}
$L_z = mr^2\omega$, and $r$ changes: $\dd L_z/\dd t = 2mr\dot r\omega \neq 0$.
This must be the \omterm{def:b1:angular-momentum:moment}{moment} of the rod's transverse reaction $N$: $rN =
2mr\dot r\omega$, so $N = 2m\dot r\omega$ --- the rod pushes the bead sideways
as it slides out (the Coriolis term of \cref{ch:b1:non-inertial-frames}).
\end{solution}

\begin{solution}{pb:b1:angular-momentum:1}
\textbf{1.} $m\omega^2r = GMm/r^2$: $\omega = \sqrt{GM/r^3} = \sqrt{3.98\times
10^{14}/5.66\times10^{25}} = \qty{2.65e-6}{rad/s}$, period $2\pi/\omega =
\qty{2.37e6}{s} = \qty{27.4}{d}$.

\textbf{2.} $L_{\text{orb}} = 7.35\times10^{22} \times 1.475\times10^{17}
\times 2.65\times10^{-6} = \qty{2.9e34}{kg.m^2/s}$.

\textbf{3.} $v = \sqrt{GM/r}$, $L = mrv = m\sqrt{GMr}$; $\dd L/\dd r =
\tfrac12 m\sqrt{GM/r} = L/2r$.

\textbf{4.} $J = 0.33 \times 5.97\times10^{24} \times 4.06\times10^{13} =
\qty{8.0e37}{kg.m^2}$; $\Omega = \qty{7.29e-5}{rad/s}$; $L_{\text{spin}} =
\qty{5.8e33}{kg.m^2/s}$, a fifth of $L_{\text{orb}}$.

\textbf{5.} $J_m \approx 0.4 \times 7.35\times10^{22} \times 3.0\times10^{12} =
\qty{8.9e34}{kg.m^2}$, times $\omega$: \qty{2.4e29}{kg.m^2/s}, $10^{-5}$ of
the orbit's: negligible.

\textbf{6.} The Sun's attraction acts on the system's center of mass
(no \omterm{def:b1:angular-momentum:moment}{moment} about it to first order); its tidal effect on the pair is
small: the total $L$ is conserved to the accuracy needed here.

\textbf{7.} The nearer bulge, ahead of the Earth--Moon line, is pulled
by the Moon with a force whose tangential component opposes the Earth's
spin.

\textbf{8.} Negative \omterm{def:b1:angular-momentum:moment}{moment} on the spin: the Earth slows, the day
lengthens.

\textbf{9.} The bulge pulls the Moon forward: positive \omterm{def:b1:angular-momentum:moment}{moment} on the
orbit, $L_{\text{orb}}$ grows.

\textbf{10.} $L_{\text{orb}} \propto \sqrt r$ grows, so $r$ grows; $v \propto
1/\sqrt r$ decreases.

\textbf{11.} The forward pull does positive work, raising the Moon's
energy; a higher orbit is a slower orbit: the energy goes into height
(potential), more than the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} lost.

\textbf{12.} Tidal locking of the Earth: day equal to month, no leading
bulge, no \omterm{def:b1:angular-momentum:moment}{torque}.

\textbf{13.} $\Delta\Omega/\Omega = -\qty{2.3e-3}{s}/\qty{86164}{s} =
\num{-2.7e-8}$ per century.

\textbf{14.} $\Delta L_{\text{spin}} = L_{\text{spin}}\Delta\Omega/\Omega =
\qty{-1.55e26}{kg.m^2/s}$ per century.

\textbf{15.} $\Delta L_{\text{orb}} = +\qty{1.55e26}{kg.m^2/s}$ per century.

\textbf{16.} $\Delta r = 2r\,\Delta L/L = 2 \times 3.84\times10^8 \times 1.55
\times10^{26}/2.9\times10^{34} = \qty{4.1}{m}$ per century, \qty{4.1}{cm/yr}:
close to the measured \qty{3.8}{cm/yr} (the day's lengthening also has
a non-tidal part).

\textbf{17.} $\Delta T/T = \tfrac32\Delta r/r = 1.6\times10^{-8}$: $2.37\times
10^6 \times 1.6\times10^{-8} = \qty{0.04}{s}$ per century.

\textbf{18.} $\Delta E_{\text{rot}} = L_{\text{spin}}\Delta\Omega = 5.8\times10^{33}
\times 7.29\times10^{-5} \times (-2.7\times10^{-8}) = \qty{-1.1e22}{J}$ per
century.

\textbf{19.} $\Delta E_{\text{orb}} = GMm\Delta r/2r^2 = 2.93\times10^{37}
\times 4.1/(2 \times 1.475\times10^{17}) = \qty{+4.1e20}{J}$ per century.

\textbf{20.} Net $-1.1\times10^{22} + 4\times10^{20} \approx \qty{-1.06e22}{J}$
per century: $3.4\times10^{12}\ \unit{W}$, dissipated as heat by tidal
friction in the oceans and the solid Earth.

\textbf{21.} A fifth of humanity's power, a fourteenth of the geothermal
flow --- a small but permanent heater.

\textbf{22.} $L_{\text{tot}} = 3.5\times10^{34}$. At $r = 5.5\times10^8$:
$\omega = \sqrt{GM/r^3} = \qty{1.55e-6}{rad/s}$, $J\omega = 1.2\times10^{32}$,
$m\sqrt{GMr} = 7.35\times10^{22} \times 4.68\times10^{11} = 3.44\times10^{34}$;
sum $3.45\times10^{34}$: consistent. Period $2\pi/\omega = \qty{4.1e6}{s}
= \qty{47}{d}$.

\textbf{23.} The Sun's tides keep slowing the Earth below the Moon's
orbital rate, so the Earth--Moon \omterm{def:b1:angular-momentum:moment}{torque} would then reverse; besides,
the Sun will have evolved long before (the time scale is tens of
billions of years).

\textbf{24.} The Earth raised tides on the Moon, which braked its spin
until one face stayed toward us; being lighter and closer to a heavier
partner, the Moon's locking was far faster.

\textbf{25.} Conserved: the total \omterm{def:b1:angular-momentum:L}{angular momentum}. Exchanged: \omterm{def:b1:angular-momentum:L}{angular
momentum} from the Earth's spin to the Moon's orbit, about \qty{1.6e26}{SI}
per century. Dissipated: \omterm{thm:b1:work-and-energy:em}{mechanical energy}, a few terawatts, as tidal
heat.
\end{solution}
