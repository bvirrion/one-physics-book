\section*{Chapter \ref{ch:b1:magnetostatics} --- Magnetostatics: Field, Forces, and Dipoles}

\begin{solution}{exo:b1:magnetostatics:1}
$B = \mu_0I/2\pi r = 2 \times 10^{-7} \times 10/0.01 = \qty{2e-4}{T}$. For \qty{1}{T}:
$I = 2\pi rB/\mu_0 = \qty{5e4}{A}$. Earth's value at $r = 2 \times 10^{-7} \times
100/5 \times 10^{-5} = \qty{0.4}{m}$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:2}
Centre: $\mu_0NI/2R = 4\pi \times 10^{-7} \times 200/0.1 = \qty{2.5}{mT}$. At $z = R$:
$\times(R^2/2R^2)^{3/2} = 2^{-3/2}$: \qty{0.89}{mT}. At $z = 10R$: $\times(1/101)^{3/2}
= 10^{-3}$: \qty{2.5}{\micro T}, already the $1/z^3$ dipole law.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:3}
$B = \mu_0nI = 4\pi \times 10^{-7} \times 1000 \times 5 = \qty{6.3}{mT}$; at the end,
half: \qty{3.1}{mT}. Wire of \qty{0.5}{mm}: $2000$ turns per metre, so
\qty{12.6}{mT} at \qty{5}{A}.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:4}
$F = ILB = 10 \times 1 \times 0.1 = \qty{1.0}{N}$; at \qty{30}{\degree}, $\sin30^\circ$:
\qty{0.5}{N}. Motor: $100 \times \qty{1}{N} \times \qty{0.05}{m} = \qty{5}{N.m}$ ---
a small industrial motor.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:5}
With $z = d\tan\alpha$: $B = \dfrac{\mu_0I}{4\pi}\displaystyle\int\dfrac{d\,\dd z}{(d^2 + z^2)^{3/2}}
= \dfrac{\mu_0I}{4\pi d}\displaystyle\int_{\alpha_1}^{\alpha_2}\cos\alpha\,\dd\alpha = \dfrac{\mu_0I}{4\pi d}
(\sin\alpha_2 - \sin\alpha_1)$; infinite wire, $\alpha_{1,2} = \mp\pi/2$: $\mu_0I/2\pi d$.
Square: each side at $d = a/2$ seen under $\pm45^\circ$ gives $\mu_0I\sqrt2/2\pi a$;
four sides: $B = 2\sqrt2\,\mu_0I/\pi a = 0.90\,\mu_0I/a$ (a circle of the same
``radius'' $a/2$ would give $\mu_0I/a$).
\end{solution}

\begin{solution}{exo:b1:magnetostatics:6}
$B(z) = f(z - R/2) + f(z + R/2)$ with $f(u) = \mu_0NIR^2/2(R^2 + u^2)^{3/2}$.
Midpoint: $2f(R/2) = \mu_0NIR^2/(5R^2/4)^{3/2} = (4/5)^{3/2}\mu_0NI/R$. $f'$ is
odd, so $B'(0) = f'(-R/2) + f'(R/2) = 0$; $f''(u) \propto (4u^2 - R^2)(R^2 +
u^2)^{-7/2}$ vanishes at $u = R/2$, so $B''(0) = 2f''(R/2) = 0$: the field is
uniform to third order. Numbers: $0.716 \times 4\pi \times 10^{-7} \times 100/0.15 =
\qty{0.60}{mT}$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:7}
Surface: $\mu_0I/2\pi a = 2 \times 10^{-7} \times 100/0.002 = \qty{10}{mT}$; at
\qty{1}{mm}, inside, $\times r/a$: \qty{5}{mT}; at \qty{1}{cm}: \qty{2}{mT}. $B$ rises
linearly to the surface, then falls as $1/r$. Coaxial: outside the sheath
the enclosed current is zero, $B = 0$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:8}
$2 \times 10^{-7}$~N/m. Bus bars: $2 \times 10^{-7} \times 10^8/0.1 = \qty{200}{N/m}$;
at \qty{100}{kA}: $\qty{2e4}{N/m}$ --- two tonnes per metre. The insulating
supports of switchgear are sized for the fault current, not the
working one.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:9}
$\Gamma = mB_h = \qty{2e-6}{N.m}$; $W = 2mB_h = \qty{4e-6}{J}$; $J\ddot\theta =
-mB_h\theta$: $T = 2\pi\sqrt{J/mB_h} = 2\pi\sqrt{10^{-7}/2 \times 10^{-6}} =
\qty{1.4}{s}$. Time the oscillations: $B_h = 4\pi^2J/mT^2$ once $m$ and $J$
are known (Gauss measured $m$ separately from the deflection it gives
a second needle).
\end{solution}

\begin{solution}{exo:b1:magnetostatics:10}
(a) $\mu_0IR^2/2z^3 = \mu_0(I\pi R^2)/2\pi z^3 = \mu_0m/2\pi z^3 = (\mu_0/4\pi)\,2m/z^3$:
the electric $2p/4\pi\varepsilon_0z^3$ with $1/\varepsilon_0 \to \mu_0$. (b) $m = 2\pi R_E^3B/
\mu_0 = 2\pi \times 2.58 \times 10^{20} \times 6 \times 10^{-5}/1.26 \times 10^{-6} = \qty{7.7e22}{A.m^2}$.
(c) $I = m/\pi R^2 = 7.7 \times 10^{22}/(\pi \times 9 \times 10^{12}) = \qty{2.7e9}{A}$. (d)
Equator ($\theta = 90^\circ$): $B_\theta = \mu_0m/4\pi R_E^3 = \qty{3e-5}{T}$, half the
polar value. At $\theta = 45^\circ$: $\tan(\text{dip}) = B_r/B_\theta = 2\cot\theta = 2$,
dip $= \qty{63}{\degree}$ --- the field plunges steeply into the ground at
mid-latitudes.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:11}
$\mu_0nI = 4\pi \times 10^{-7} \times 2000 \times 3 = \qty{7.5}{mT}$. Centre: $\tan\alpha_1 =
R/(L/2) = 0.2$, $\cos\alpha_1 = -\cos\alpha_2 = 0.981$: $B = 0.981\,\mu_0nI =
\qty{7.4}{mT}$ ($98\%$). End: $\cos\alpha_1 = 0$, $\tan\alpha_2 = R/L$, $\cos\alpha_2 =
-0.995$: $B = 0.497\,\mu_0nI = \qty{3.7}{mT}$ ($50\%$). Outside at $x$ beyond the
end, both ends on the same side: $B = \tfrac12\mu_0nI(\cos\alpha_1 - \cos\alpha_2)
\approx \tfrac12\mu_0nI\,\tfrac{R^2}{2}[1/x^2 - 1/(x + L)^2]$; $1\%$ of the centre
value at $x \approx \qty{10}{cm}$, half a length out. $B(z)$: a flat top over
most of the length, falling to half at the ends and to a few percent a
length away.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:12}
(a) The windings' own field jumps from $0$ to $B$ across them; the
field acting on them is the mean, $B/2$ (the sheet cannot push on
itself). Force on an element of winding of length $\dd l$ and width $\dd z$
carrying $K\,\dd z$: $K\,\dd z\,\dd l\,B/2$, radial outward (check with $I\,\dd\vect l
\wedge\vect B$): pressure $p = KB/2 = B^2/2\mu_0$ since $K = nI = B/\mu_0$.
(b) \qty{1}{T}: \qty{4e5}{Pa} = \qty{4}{bar}; \qty{10}{T}: \qty{400}{bar}; \qty{45}{T}:
\qty{8e8}{Pa} = \qty{8000}{bar}, close to the \qty{1}{GPa} of steel. (c) The
windings must hold a pressure growing as $B^2$: above \qty{40}{T} or so
no material and no cooling scheme copes --- the limit of steady magnets
is mechanical (and, for superconductors, their critical field).
\end{solution}

\begin{solution}{pb:b1:magnetostatics:1}
\textbf{1.} Every plane containing the axis is a \omterm{prop:b1:electrostatics-gauss:symmetry}{plane of symmetry} of
the currents: $\vect B$ is perpendicular to it, orthoradial; invariance
along and around the axis: $\vect B = B(r)\,\vect e_\theta$.

\textbf{2.} Circle of radius $r$: $2\pi rB = \mu_0I_{\text{enc}}$. Core:
$I_{\text{enc}} = Ir^2/a^2$, $B = \mu_0Ir/2\pi a^2$; between: $\mu_0I/2\pi r$;
outside ($r > c$): $I - I = 0$, $B = 0$.

\textbf{3.} $B(a) = 2 \times 10^{-7}/5 \times 10^{-4} = \qty{4.0e-4}{T}$; $B(b) = 2 \times
10^{-7}/2.5 \times 10^{-3} = \qty{8.0e-5}{T}$.

\textbf{4.} Inside the tube no current is enclosed: $B = 0$ there instead
of the linear rise; everywhere else nothing changes --- the field outside
a cylindrical current depends only on the total current.

\textbf{5.} $I_{\text{enc}} = I - I(r^2 - b^2)/(c^2 - b^2) = I(c^2 - r^2)/(c^2 -
b^2)$: $B = \dfrac{\mu_0I}{2\pi r}\,\dfrac{c^2 - r^2}{c^2 - b^2}$, equal to $\mu_0I/2\pi b$ at
$b$ and to $0$ at $c$: continuous. Sketch: linear to \qty{4e-4}{T} at
\qty{0.5}{mm}, $1/r$ down to \qty{8e-5}{T} at \qty{2.5}{mm}, falling to zero at
\qty{3}{mm}, zero beyond.

\textbf{6.} The sheath's current is antiparallel to the core's: repulsion,
an outward pressure. Surface current $K = I/2\pi b = \qty{64}{A/m}$; mean field
$\tfrac12(8 \times 10^{-5} + 0) = \qty{4e-5}{T}$: $p = KB = \qty{2.5e-3}{Pa}$. At
\qty{10}{kA}: $\times10^8$, \qty{2.5e5}{Pa} = \qty{2.5}{bar} --- pulsed-power
cables are built to take it.

\textbf{7.} No field outside: the cable neither radiates nor picks up
magnetic interference from its neighbours. A pair of wires: the two
fields nearly cancel, leaving $\approx \mu_0Id/2\pi D^2 = 2 \times 10^{-7} \times 2 \times
10^{-3}/10^{-2} = \qty{4e-8}{T}$ at \qty{10}{cm} --- small, but the coax gives
zero.

\textbf{8.} $D = \sqrt{h^2 + d^2/4} = \qty{10.0}{m}$: each gives $\mu_0I/2\pi D =
\qty{1.0e-5}{T}$, perpendicular to the line from the conductor. The
currents being opposite, the components parallel to the line joining
the conductors cancel and the vertical ones add: $B = 2 \times 10^{-5} \times
(d/2)/D = \qty{1.0e-6}{T}$, vertical. Fifty times below the Earth's field,
a hundred times below the guideline.

\textbf{9.} For $D \gg d$, $B \approx \dfrac{\mu_0I}{2\pi}\Bigl(\dfrac1{D_1} - \dfrac1{D_2}\Bigr)
\sim \dfrac{\mu_0I}{2\pi}\dfrac{d}{D^2}$: the two $1/D$ fields cancel to first
order and only their difference, of relative size $d/D$, survives --- a
``two-wire dipole''.

\textbf{10.} Own conductor at \qty{1}{m}: \qty{1.0e-4}{T} horizontal; the other
at \qty{1.41}{m}: \qty{7.1e-5}{T} at \qty{45}{\degree}, opposing: resultant
$(5 \times 10^{-5}, 5 \times 10^{-5})$, \qty{7e-5}{T} --- \qty{70}{\micro T}, below the
public guideline and far below the occupational one.

\textbf{11.} $F/\ell = \mu_0I^2/2\pi d = 2 \times 10^{-7} \times 2.5 \times 10^5 =
\qty{0.05}{N/m}$, repulsive; at \qty{20}{kA}: \qty{80}{N/m}.

\textbf{12.} \qty{15}{N} normally; \qty{24}{kN} in short circuit, eight times
the span's \qty{2.9}{kN} weight: the conductors are flung apart, swing
and may clash when they fall back --- hence spacers and breakers that
open within a few cycles.

\textbf{13.} $I\cos\omega t + I\cos(\omega t - 2\pi/3) + I\cos(\omega t + 2\pi/3) =
I\cos\omega t\,(1 + 2\cos2\pi/3) = 0$: the three \omterm{def:b1:sinusoidal-impedance:complex}{phasors} sum to zero. The
$1/D$ fields cancel as for the coax; the far field falls as $1/D^2$ or
faster.

\textbf{14.} The line's field there is vertical; a compass responds to
the horizontal component only: no deflection (the dip changes by
$10^{-6}/(2 \times 10^{-5})$, three degrees, which the compass cannot show).

\textbf{15.} Sides $a$ (parallel to the axis): $\vect B$ radial is
perpendicular to them, force $NIaB$ on each, perpendicular to the
coil's plane, opposite on the two sides: a couple. Sides $b$: $\dd\vect l$
is radial, parallel to $\vect B$: no force.

\textbf{16.} $\Gamma = 2 \times NIaB \times b/2 = NIabB = NISB = 0.024\,I$ (N\,m, $I$
in A). The radial field is always in the coil's plane and perpendicular
to the sides $a$: the forces are always normal to the plane and their
\omterm{def:b1:angular-momentum:moment}{lever arm} $b/2$ never changes.

\textbf{17.} $C\theta = NISB$: $\theta = 2400\,I$ rad, i.e.\ \qty{2.4}{rad/mA}; full
scale \qty{1.2}{rad} at $I = \qty{0.50}{mA}$.

\textbf{18.} Length $200 \times 2(0.02 + 0.03) = \qty{20}{m}$, section $\pi(5 \times
10^{-5})^2 = \qty{7.9e-9}{m^2}$: $R = 1.7 \times 10^{-8} \times 20/7.9 \times 10^{-9} =
\qty{43}{\ohm}$; $U = 43 \times 5 \times 10^{-4} = \qty{22}{mV}$; $P = UI = \qty{11}{\micro W}$.

\textbf{19.} $J\ddot\theta = -C\theta + NISB\,I$: $\omega_0 = \sqrt{C/J} = \sqrt{10^{-5}/6.7
\times 10^{-8}} = \qty{12}{rad/s}$, $T = \qty{0.51}{s}$.

\textbf{20.} $J\ddot\theta + \alpha\dot\theta + C\theta = NISB\,I$: critical for
$\alpha = 2\sqrt{JC} = 2\sqrt{6.7 \times 10^{-13}} = \qty{1.6e-6}{N.m.s}$; the needle
then reaches its reading fastest without overshoot.

\textbf{21.} Voltmeter: $R_s = 10/5 \times 10^{-4} - 43 \approx \qty{20}{k\ohm}$.
Ammeter: the shunt carries \qty{0.9995}{A} under \qty{22}{mV}: $R_{\text{sh}} =
\qty{22}{m\ohm}$.

\textbf{22.} $m = NIS = 200 \times 5 \times 10^{-4} \times 6 \times 10^{-4} = \qty{6e-5}{A.m^2}$;
$-mB = \qty{-1.2e-5}{J}$; spring $\tfrac12C\theta^2 = \tfrac12 \times 10^{-5} \times 1.44 =
\qty{7.2e-6}{J}$: the same order --- the spring stores what the magnetic
\omterm{def:b1:angular-momentum:moment}{torque} worked while the coil turned.

\textbf{23.} In a uniform field the \omterm{def:b1:angular-momentum:moment}{torque} is $mB\cos\theta$ (maximal when
the plane contains $\vect B$, zero when the normal is along it): $C\theta =
NISB\,I\cos\theta$ --- linear only for small $\theta$, compressed at the
top of the scale.

\textbf{24.} \qty{50}{Hz} against $\omega_0/2\pi = \qty{2}{Hz}$: the coil cannot
follow; it sits at the mean of the current, zero, with a quiver of
\omterm{def:b1:wave-propagation:signal}{amplitude} $\approx \theta_{\text{static}}(\omega_0/\omega)^2 = 1.2 \times (12/314)^2 \approx
\qty{2e-3}{rad}$, invisible. A moving-coil meter reads the mean --- for
alternating current it needs a rectifier.

\textbf{25.} \qty{4e-4}{T} at the core of a \qty{1}{A} coax and nothing outside
(\omterm{thm:b1:magnetostatics:ampere}{Ampère's law}); \qty{1}{\micro T} under a \qty{500}{A} line and \qty{0.05}{N/m}
between its conductors (Biot--Savart plus Laplace); \qty{2.4}{rad/mA} for
the meter (Laplace couple against a spring).
\end{solution}
