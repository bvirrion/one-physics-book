\section*{Chapter \ref{ch:b1:filters-transfer-functions} --- Filters and Transfer Functions}

\begin{solution}{exo:b1:filters-transfer-functions:1}
$0.5 \to \qty{-6.0}{dB}$; $10 \to \qty{20}{dB}$; $1/\sqrt2 \to \qty{-3.0}{dB}$;
$100 \to \qty{40}{dB}$. $\qty{-40}{dB} \to 0.01$; $\qty{+6}{dB} \to 2.0$.
Cascade: $-6 - 20 = \qty{-26}{dB}$, a ratio $0.05$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:2}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 10^4 \times 1.6\times10^{-8}) =
\qty{1.0}{kHz}$. \omterm{def:b1:filters-transfer-functions:H}{Gains} $-10\log(1 + (f/f_c)^2)$: \qty{100}{Hz}:
\qty{-0.04}{dB}; \qty{1}{kHz}: \qty{-3}{dB}; \qty{10}{kHz}: \qty{-20}{dB};
\qty{100}{kHz}: \qty{-40}{dB}.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:3}
(a) low-pass ($C$ shorts the output at high $f$); (b) high-pass ($C$
opens at low $f$); (c) second-order low-pass ($L$ opens at high $f$,
$C$ shorts); (d) band-pass ($C$ blocks low $f$, $L$ blocks high $f$).
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:4}
$4E/n\pi$: \qty{6.37}{V} at \qty{1}{kHz}, \qty{2.12}{V} at \qty{3}{kHz},
\qty{1.27}{V} at \qty{5}{kHz}. The square wave is antisymmetric about
a half-period shift ($s(t + T/2) = -s(t)$), which even harmonics are
not.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:5}
At \qty{100}{Hz} and its useful harmonics, $\omega \ll \omega_c$:
$\underline H \approx j\omega/\omega_c$, a derivative divided by
$\omega_c$. The triangle's slope is $\pm 4Ef = \pm\qty{400}{V/s}$, so
the output is a square wave of \omterm{def:b1:wave-propagation:signal}{amplitude} $400/(2\pi \times 10^4) =
\qty{6.4}{mV}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:6}
$\omega \gg \omega_c$: $u_s \approx (1/\tau)\int u_e\,\dd t$, $\tau =
1/(2\pi \times 10) = \qty{16}{ms}$. The integral of $\pm E$ is a
triangle rising by $ET/2$ per half period: swing $ET/2\tau$, \omterm{def:b1:wave-propagation:signal}{amplitude}
$ET/4\tau = 1.0 \times 10^{-3}/(4 \times 0.016) = \qty{16}{mV}$ around
zero.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:7}
$f_0 = \qty{5.03}{kHz}$. $Q = \sqrt{L/C}/R = 1/\sqrt2$: $R = \sqrt2 \times
316 = \qty{447}{\ohm}$. At $f_0$: \qty{-3}{dB}; at $10f_0$: $1/\sqrt{1 +
10^4} \to \qty{-40}{dB}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:8}
$G = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ with $x = f/3$: \qty{1}{kHz}: $x = 1/3$,
$Q(x - 1/x) = -53$, $G = 0.019$; \qty{3}{kHz}: $G = 1$; \qty{5}{kHz}:
$x = 5/3$, $21$, $G = 0.047$. Output: \qty{2.1}{V} at \qty{3}{kHz}, with
\qty{0.12}{V} at \qty{1}{kHz} and \qty{0.06}{V} at \qty{5}{kHz} --- a
nearly pure \qty{3}{kHz} sinusoid.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:9}
Nodes: with $a = jRC\omega$, the first capacitor sees $R$ upstream and
$R + 1/jC\omega$ downstream: $\underline H = 1/(1 + 3a + a^2)$, i.e.\
$1/[1 - (\omega/\omega_c)^2 + 3j\omega/\omega_c]$. At $\omega_c$: $1/3j$,
$G = 1/3$, \qty{-9.5}{dB}; the product of two separate \omterm{def:b1:filters-transfer-functions:H}{gains} would be
$1/2$, \qty{-6}{dB}. $f_c = \qty{1.0}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:10}
Below $\omega_1$: \qty{0}{dB}; between $\omega_1$ and $\omega_2$: rising
\qty{20}{dB/decade}; above $\omega_2$: flat at $20\log(\omega_2/\omega_1) =
\qty{40}{dB}$. Phase $\arctan(\omega/\omega_1) - \arctan(\omega/\omega_2)$:
at $\omega = 10^3$, $\arctan 10 - \arctan 0.1 = 84^\circ - 6^\circ =
78^\circ$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:11}
Far above cutoff, $G \approx f_c/f$; the ripple ratio after filtering
is
\[
\frac{(4E/3\pi)(f_c/100)}{2E/\pi} = \frac{2}{3}\,\frac{f_c}{100} < 0.01,
\]
so $f_c < \qty{1.5}{Hz}$ and $\tau = 1/2\pi f_c > \qty{0.1}{s}$. The load itself is the
$R$: a capacitor across it costs nothing in \omterm{def:b1:dc-circuits:voltage}{voltage} drop and power,
whereas a series \omterm{def:b1:dc-circuits:resistor}{resistor} would waste both.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:12}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 1.6\times10^5 \times 10^{-7}) =
\qty{10}{Hz}$, $\tau = RC = \qty{16}{ms}$. During a flat top the
capacitor charges through $R$: the output decays as $\eu^{-t/\tau}$
from each edge. Sag over $T/2$: $1 - \eu^{-T/2\tau}$: \qty{50}{Hz}
($T/2 = \qty{10}{ms}$): $1 - \eu^{-0.63} = 47\%$; \qty{1}{kHz}
($T/2 = \qty{0.5}{ms}$): $3\%$.
\end{solution}

\begin{solution}{pb:b1:filters-transfer-functions:1}
\textbf{1.} $\underline H = \underline U_{\text{driver}}/\underline U_e$;
$G_{\mathrm{dB}} = 20\log\abs{\underline H}$.

\textbf{2.} $\underline H_L = R/(R + jL\omega) = 1/(1 + j\omega/\omega_c)$,
$\omega_c = R/L$.

\textbf{3.} $L = R/\omega_c = 8.0/(2\pi \times 2000) = \qty{0.64}{mH}$.

\textbf{4.} $\underline H_H = R/(R + 1/jC\omega) = jRC\omega/(1 + jRC\omega)$,
$\omega_c = 1/RC$; $C = 1/(R\omega_c) = \qty{9.9}{\micro F} \approx
\qty{10}{\micro F}$.

\textbf{5.} At $f_c$: \qty{-3}{dB} each. \qty{200}{Hz} ($x = 0.1$): woofer
\qty{-0.04}{dB}, tweeter \qty{-20}{dB}. \qty{20}{kHz} ($x = 10$): woofer
\qty{-20}{dB}, tweeter \qty{-0.04}{dB}.

\textbf{6.} $G_L^2 + G_H^2 = (1 + x^2)/(1 + x^2) = 1$: with equal loads
the two powers add up to the power a single \qty{8}{\ohm} driver would
take --- shared, not lost.

\textbf{7.} $-45^\circ$ (woofer) and $+45^\circ$ (tweeter): $90^\circ$
apart; at $f_c$ the two cones do not move together.

\textbf{8.} \qty{\pm 6}{dB} per octave (\qty{20}{dB} per decade).

\textbf{9.} Divider between $jL\omega$ and $R \parallel 1/jC\omega =
R/(1 + jRC\omega)$: $\underline H = R/[R + jL\omega(1 + jRC\omega)] =
1/(1 - LC\omega^2 + jL\omega/R)$.

\textbf{10.} $\omega_0 = 1/\sqrt{LC}$; $L\omega/R = x/Q$ gives $Q =
R/L\omega_0 = R\sqrt{C/L}$.

\textbf{11.} $\abs{\underline H}^{-2} = (1 - x^2)^2 + x^2/Q^2 = 1 + x^4 +
x^2(1/Q^2 - 2) = 1 + x^4$ for $Q^2 = \tfrac12$; at $x = 1$, $G = 1/\sqrt2$.

\textbf{12.} $L = R/(Q\omega_0) = 8\sqrt2/(1.257\times10^4) = \qty{0.90}{mH}$;
$C = 1/(L\omega_0^2) = \qty{7.0}{\micro F}$.

\textbf{13.} $x = 2$: $1/\sqrt{17}$, \qty{-12.3}{dB}; $x = 4$: $1/\sqrt{257}$,
\qty{-24.1}{dB}: \qty{12}{dB} per octave.

\textbf{14.} Exchanging $L$ and $C$ exchanges $x \leftrightarrow 1/x$ (and
$Q$ stays): $\underline H_H = -x^2/(1 - x^2 + jx/Q)$.

\textbf{15.} At $x = 1$: $\underline H_L = Q/j = -jQ$ ($-90^\circ$),
$\underline H_H = jQ$ ($+90^\circ$): the drivers are in phase
opposition at the crossover and would cancel acoustically; the tweeter
is wired with its polarity reversed.

\textbf{16.} Harmonics above \qty{2}{kHz}: $n \geq 10$ (\qty{2.2}{kHz}
and up) go mostly to the tweeter.

\textbf{17.} $9 \times 220 = \qty{1980}{Hz} \approx f_c$: split equally,
\qty{-3}{dB} to each.

\textbf{18.} Woofer ($G_L$): $x = 0.5$: $0.89$; $x = 1.5$: $0.55$; $x = 2.5$:
$0.37$. Tweeter ($G_H$): $0.45$, $0.83$, $0.93$.

\textbf{19.} The tweeter: its series capacitor blocks DC. The woofer
takes the offset (\qty{1}{V} across \qty{8}{\ohm}: \qty{125}{mA},
\qty{0.125}{W} of heat and a displaced cone).

\textbf{20.} First order, $x = 0.25$: $G_H = 0.25/\sqrt{1.0625} = 0.24$
(\qty{-12}{dB}), power fraction $6\%$. Second order: $0.0625/\sqrt{1 +
0.0039} = 0.06$ (\qty{-24}{dB}), $0.4\%$. A tweeter rated a few watts
receives $6\%$ of a loud bass note with the first-order design --- it can
burn.

\textbf{21.} $\underline Z = 8 + jL_{vc}\omega$: \qty{2}{kHz}: $8 + 6.3j$,
$\abs{\underline Z} = \qty{10.2}{\ohm}$; \qty{8}{kHz}: $8 + 25.1j$,
\qty{26.3}{\ohm}.

\textbf{22.} $\underline H = \underline Z/(\underline Z + jL\omega)$:
\qty{2}{kHz}: $10.2/\abs{8 + 14.3j} = 10.2/16.4 = 0.62$ (\qty{-4.2}{dB});
\qty{8}{kHz}: $26.3/\abs{8 + 57.3j} = 0.45$ (\qty{-6.9}{dB}) instead of
\qty{-12.3}{dB}: the rising coil \omterm{def:b1:sinusoidal-impedance:impedance}{impedance} fights the series coil, and
the roll-off flattens to a few dB per octave.

\textbf{23.} \omterm{def:b1:sinusoidal-impedance:impedance}{Admittances}: $1/(R + jL_{vc}\omega) + jC_z\omega/(1 + jRC_z\omega)$;
with $C_z = L_{vc}/R^2$ the second term is $jL_{vc}\omega/[R(R +
jL_{vc}\omega)]$, and the sum is $(R + jL_{vc}\omega)/[R(R + jL_{vc}\omega)]
= 1/R$. $C_z = 0.50\times10^{-3}/64 = \qty{7.8}{\micro F}$.

\textbf{24.} $U = \sqrt{PR} = \sqrt{400} = \qty{20}{V}$ rms. At $f_c$ each
driver gets half: \qty{25}{W}. At \qty{500}{Hz} the tweeter gets $6\%$:
\qty{3}{W}.

\textbf{25.} First order: two parts, power-complementary, gentle
\qty{6}{dB/oct} slopes, $90^\circ$ between drivers, bass leaks into the
tweeter. Second order: four parts, \qty{12}{dB/oct}, $180^\circ$ at
crossover (invert the tweeter), far better tweeter protection. Either
way the woofer's inductance must be tamed by a Zobel network for the
design to hold.
\end{solution}
