\section*{अध्याय \ref{ch:b1:charged-particles} --- विद्युत और चुंबकीय क्षेत्रों में आवेशित कण}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$। $v = \sqrt{2eU/m}$: इलेक्ट्रॉन
\qty{1.9e7}{m/s}; प्रोटॉन \qty{4.4e5}{m/s} ($\sqrt{1836}$ गुना धीमा)।
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$; $T = 2\pi/\omega_c = \qty{36}{ns}$।
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$। $30^\circ$ पर: $R\sin 30^\circ
= \qty{10}{m}$ त्रिज्या और $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$ पिच
वाली एक कुंडलिनी, जो क्षेत्र-रेखा के अनुदिश बहती जाती है।
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$। तेज़ आयन: चुंबकीय बल जीत जाता है,
और वे चुंबकीय ओर मुड़ जाते हैं; उसी चाल पर विपरीत आवेश: दोनों बल उलट जाते
हैं, फिर भी संतुलित --- चयनक चिह्न और द्रव्यमान दोनों के प्रति अंधा है।
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$; विक्षेपण
$0.30 \times 0.10 = \qty{3.0}{cm}$; सुग्राहिता \qty{0.30}{mm/V}। तेज़
इलेक्ट्रॉन पट्टिकाओं के बीच कम समय बिताते हैं: $\tan\alpha \propto 1/U_0$।
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$; गिरने के बिंदुओं के बीच दूरी $2\Delta R =
\qty{1.2}{cm}$ (त्रिज्याएँ \qty{5.8}{mm} से भिन्न हैं, और व्यास उससे
दुगुने)।
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$; प्रति चक्कर \qty{100}{keV}: यानी $270$ चक्कर; समय $270/f_c =
\qty{12}{\micro s}$; $\gamma - 1 = 27/938 = 2.9\%$।
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$;
आवर्तकाल $2\pi m/eB = \qty{18}{ns}$, पिच $v_\parallel T = \qty{1.5}{cm}$:
यानी प्रति मीटर लगभग $64$ चक्कर।
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: यानी माइक्रोवोल्ट से भी कम। $n$ के एक करोड़ गुना छोटा
होने पर कोई अर्धचालक मिलीवोल्ट देता है --- यानी काम का एक संवेदक।
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$; इसका $\vect v$ के साथ अदिश गुणन
कीजिए: $\dd(\tfrac12 mv^2)/
\dd t = 0$, चाहे $\vect B(\vect r, t)$ कुछ भी हो, बशर्ते कोई $\vect E$ न हो।
वेग की दिशा, \omterm{def:b1:point-kinematics:frenet}{वक्रता त्रिज्या}, पिच --- सब बदल सकते हैं; चाल और \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा}
(\qty{10}{keV}) नहीं बदल सकते।
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$। और $\vect v =
(E/B)\vect e_x + \vect w$ के साथ: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ ---
यानी $\vect w$ का $\omega_c$ पर विशुद्ध साइक्लोट्रॉन घूर्णन। विराम से आरंभ
करने पर $\vect w(0) = -(E/B)\vect e_x$: $\vect w$ अचर परिमाण $E/B$ से घूमता
है, अतः वेग $E/B$ का अपवाह और उतने ही परिमाण के किसी घूमते सदिश का योग है
--- यानी एक चक्रज, जिसके उभयाग्र वहाँ हैं जहाँ $\vect v = \vect 0$।
इलेक्ट्रॉन: अपवाह $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, मेहराब की ऊँचाई $2mE/qB^2 =
\qty{11}{\micro m}$।
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (लगभग $15^\circ$):
यानी एक मामूली मिलीटेस्ला के लिए विक्षेपण
$\approx 0.30 \times 0.27 = \qty{8}{cm}$। पट्टिकाओं से बड़े कोण पाने के लिए
किलोवोल्ट और लंबी पट्टिकाएँ चाहिए होतीं; जबकि कोई कुंडली बिना किसी
\omterm{def:b1:dc-circuits:voltage}{विभवांतर} के पुंज को दसियों अंश तक मोड़ देती है।
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: कोई शक्ति नहीं,
अतः अचर चाल।

\textbf{2.} $mv^2/R = evB$: $R = mv/eB$; $T = 2\pi R/v = 2\pi m/eB$, जो
$v$ से मुक्त है।

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$।

\textbf{4.} प्रति पार $eU = \qty{50}{keV}$, प्रति चक्कर \qty{100}{keV}।

\textbf{5.} जब-जब प्रोटॉन पहुँचे, तब-तब अंतराल त्वरक ध्रुवता पर होना
चाहिए, यानी आधे-आधे आवर्तकाल पर; आवृत्ति हटते ही \omterm{def:b1:wave-propagation:signal}{कला} खिसकती है और धक्के
अंततः मंदन करने लगते हैं।

\textbf{6.} एक चक्कर के बाद $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$; और $n$ चक्करों के बाद $E_k =
n \times \qty{100}{keV}$ तथा $R = \qty{3.0}{cm}\sqrt n$।

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$।

\textbf{8.} $270$ चक्कर; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$।

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$।

\textbf{10.} नहीं: $E_k$ केवल $B$ और $R_{\max}$ पर निर्भर है; $U$ चक्करों
की संख्या और समय तय करता है।

\textbf{11.} पथ \qty{560}{m}: अतः $\lambda \gtrsim \qty{1}{km}$ चाहिए,
अर्थात् $p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$
--- यानी एक उच्च निर्वात।

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$; अतः आवर्तकाल $2.9\%$ बढ़ जाता
है। $270$ चक्करों पर यह खिसकाव जुड़कर कई आवर्तकाल हो जाता है --- अंतिम
चक्कर \omterm{def:b1:wave-propagation:signal}{कला} से बाहर होते; व्यवहार में $B$ और अंतःक्षेपण की \omterm{def:b1:wave-propagation:signal}{कला} दोनों साध ली
जाती हैं, और \qty{27}{MeV} इस सीधी-सादी मशीन की सीमा के पास ही है।

\textbf{13.} $E_k = (eBR)^2/2m$: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$; और \qty{2}{T} पर $R =
\qty{1.1}{m}$।

\textbf{14.} $T = 2\pi\gamma m/eB$ केंद्र से कोर तक $24\%$ बढ़ जाता है: कोई
नियत आवृत्ति निराशाजनक रूप से ताल से बाहर हो जाती है।

\textbf{15.} सिंक्रोसाइक्लोट्रॉन: पुंजक के ऊर्जा पाने के साथ आवृत्ति घटाते
जाइए --- पर एक बार में एक ही पुंजक, अतः औसत धारा कम। समकालिक: $B(R) \propto \gamma$
ऐसा गढ़िए कि आवर्तकाल अचर रहे --- सतत पुंज, और यही आधुनिक चुनाव है।

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$; $R = p/eB = \qty{1.2}{m}$, यानी
आपेक्षिकता-रहित \qty{1.1}{m} से थोड़ा-सा अधिक।

\textbf{17.} अनुप्रस्थ आवेग $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$; कोण $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: हँकाई के लिए निराशाजनक।

\textbf{18.} चुंबकीय: $R = p/eB = \qty{2.3}{m}$, कोण $L/R = \qty{0.43}{rad}
\approx 25^\circ$: यानी हज़ार गुना अधिक --- अतः चुंबक ही।

\textbf{19.} प्रति प्रोटॉन ऊर्जा $\qty{3.68e-11}{J}$; मात्रा $2.0 \times 0.50 =
\qty{1.0}{J}$: अतः $2.7\times10^{10}$ प्रोटॉन।

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \num{6.2e9}$ प्रोटॉन प्रति सेकंड: यानी लगभग
\qty{4.4}{s}।

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
यानी एक रात-दीप, पर ठीक वहीं पहुँचाया गया जहाँ चाहिए।

\textbf{22.} \qty{1}{cm} में $\qty{58}{MeV}$: यानी लगभग
\qty{6}{keV/\micro m} --- अंतिम पथ पर सघन आयनन।

\textbf{23.} प्रति प्रोटॉन $230\times10^6/30 \approx \num{8e6}$ आयनन ---
यही आणविक क्षति कोशिका को मार देती है।

\textbf{24.} X-किरणें अपने पूरे पथ पर ऊर्जा जमा करती हैं, और सबसे अधिक
प्रवेश के पास; जबकि प्रोटॉन अपनी अधिकांश ऊर्जा अपनी परास के अंत में जमा
करते हैं, आगे के ऊतक को बचाते हुए और पीछे कुछ भी न छोड़ते हुए।

\textbf{25.} $R = mv/eB$ और $T = 2\pi m/eB$ (वृत्त), $E_k = (eBR)^2/2m$
(ऊर्जा), $E_k/2eU$ (चक्कर), मात्रा $=$ (प्रोटॉन $\times$ ऊर्जा)/द्रव्यमान,
और धारा $= e \times$ दर (समय)।
\end{solution}
