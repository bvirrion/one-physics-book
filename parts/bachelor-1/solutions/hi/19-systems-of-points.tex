\section*{अध्याय \ref{ch:b1:systems-of-points} --- कणों के निकाय; दृढ़ पिंडों का परिचय}

\begin{solution}{exo:b1:systems-of-points:1}
$x_G = (2.0 \times 0 + 3.0 \times 1.0)/5.0 = \qty{0.60}{m}$। पृथ्वी--चंद्र:
पृथ्वी के केंद्र से $384000/82 = \qty{4700}{km}$, यानी पृथ्वी के भीतर ही
(\qty{6370}{km}): पृथ्वी अपनी सतह से \qty{1700}{km} नीचे के किसी बिंदु के
परितः डोलती है।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:2}
$4.0v = 0.010 \times 800$: $v = \qty{2.0}{m/s}$। गोली $\tfrac12 \times 0.010
\times 800^2 = \qty{3.2}{kJ}$, राइफ़ल \qty{8}{J}: यह ऊर्जा बारूद की
रासायनिक ऊर्जा से आई; हलका पिंड उसका लगभग सारा हिस्सा ले जाता है
($E_k = p^2/2m$, जहाँ $p$ दोनों के लिए समान है)।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:3}
$v = 20000/2500 = \qty{8.0}{m/s}$; $E_k$: पहले \qty{200}{kJ}, बाद में $\tfrac12 \times
2500 \times 64 = \qty{80}{kJ}$: यानी $60\%$ खो गया।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:4}
सिरा: $\int_0^L x^2(M/L)\dd x = ML^2/3$; केंद्र: $\int_{-L/2}^{L/2}x^2(M/L)
\dd x = ML^2/12$; और $d = L/2$ के साथ ह्यूगेंस: $ML^2/12 + ML^2/4 = ML^2/3$।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:5}
$E_k = \tfrac12 Mv^2(1 + k)$: बेलन के लिए $1.5$, छल्ले के लिए $2$ गुना $\tfrac12 Mv^2$।
समान चाल पर छल्ले के पास अधिक ऊर्जा होती है: वह समतल फ़र्श पर अधिक दूर तक
लुढ़कता है (घर्षण समान) और अधिक ऊँचा चढ़ता है ($h = v^2(1 + k)/2g$)।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:6}
हाइड्रोजन क्लोराइड: $\mu = 35m_H/36 = 0.972m_H$; परमाणु: $\mu = m_e/(1 + 1/1836) =
0.99946m_e$। आवृत्तियाँ $\propto 1/\sqrt\mu$: हाइड्रोजन क्लोराइड के लिए $+1.4\%$, और
हाइड्रोजन के लिए $+0.027\%$ (H तथा D के वर्णक्रमों का समस्थानिक विस्थापन
इसी पर टिका है)।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:7}
$J = ML^2/3$, $d = L/2$: $T = 2\pi\sqrt{(ML^2/3)/(MgL/2)} = 2\pi\sqrt{2L/3g}
= \qty{1.64}{s}$; तुल्य लंबाई $2L/3 = \qty{0.67}{m}$। आघात-केंद्र पर पड़ा
प्रहार छड़ को धुरी के परितः घुमा देता है, बिना वहाँ कोई आवेगी प्रतिक्रिया
दिए: गेंद को बल्ले के इसी ``मीठे बिंदु'' पर मारिए और हाथों को कोई झनझनाहट
नहीं होगी।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:8}
$J = \tfrac12 MR^2 = \qty{1.0e-3}{kg.m^2}$; $C = 4\pi^2J/T^2 = \qty{9.9e-3}{N.m/rad}$।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:9}
$\omega = \qty{314}{rad/s}$, $E = \tfrac12 J\omega^2 = \qty{4.9e5}{J}$;
$\Gamma = J\omega/t = \qty{314}{N.m}$; चक्कर: $\tfrac12\omega t/2\pi = 250$।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:10}
$a = g\sin\alpha/(1 + k)$: गोला ($k = 2/5$) $0.714$, बेलन ($1/2$)
$0.667$, छल्ला ($1$) $0.500$ गुना $g\sin\alpha$: यानी गोला पहले, छल्ला अंत में;
और कहीं कोई द्रव्यमान या त्रिज्या नहीं।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:11}
संवेग: $0.010 \times 400 = 2.01\,v$: $v = \qty{2.0}{m/s}$; फिर ऊर्जा:
$h = v^2/2g = \qty{0.20}{m}$। $E_k$: पहले \qty{800}{J}, बाद में $\tfrac12 \times
2.01 \times 4.0 = \qty{4.0}{J}$: यानी $99.5\%$ विरूपण और ऊष्मा में खो गया।
टक्कर अप्रत्यास्थ है --- उसके पार ऊर्जा संरक्षित नहीं रहती, संवेग रहता है
(डोरी का तनाव ऊर्ध्वाधर है और टक्कर बहुत छोटी)।
\end{solution}

\begin{solution}{exo:b1:systems-of-points:12}
$m_1v_1 = m_1v_1' + m_2v_2'$ और $m_1v_1^2 = m_1v_1'^2 + m_2v_2'^2$ से
$v_1 + v_1' = v_2'$ मिलता है (ऊर्जा वाले समीकरण को संवेग वाले से भाग
दीजिए), और वहीं से सूत्र। लक्ष्य तक पहुँचा ऊर्जा-अंश $4m_1m_2/(m_1 + m_2)^2$:
प्रोटॉन $100\%$; कार्बन $4 \times 12/169 = 28\%$; सीसा $4 \times 207/208^2 =
1.9\%$। हलके नाभिक न्यूट्रॉन की ऊर्जा कुछ ही संघट्टों में ले लेते हैं
(पानी, ग्रैफ़ाइट); सीसे को सैकड़ों चाहिए होते।
\end{solution}

\begin{solution}{pb:b1:systems-of-points:1}
\textbf{1.} अक्ष पर, चकती के केंद्र में, सममिति से।

\textbf{2.} कोनिग: $E_k = \tfrac12 Mv_G^2 + E_k^*$, और \omterm{def:b1:systems-of-points:barycentric}{भारकेंद्रीय तंत्र} में
यो-यो अपनी अक्ष के परितः घूमता है: $E_k^* = \tfrac12 J\omega^2$।

\textbf{3.} $\rho$ त्रिज्या और $2M\rho\,\dd\rho/R^2$ द्रव्यमान का छल्ला: $J = \int_0^R
\rho^2\cdot 2M\rho\,\dd\rho/R^2 = MR^2/2 = 0.5 \times 0.050 \times 9\times10^{-4}
= \qty{2.25e-5}{kg.m^2}$।

\textbf{4.} डोरी स्थिर है और धुरी से खुलती है: धुरी का जो बिंदु उससे
संस्पर्श में है वह क्षण भर के लिए विराम में होता है, अतः $v_G = r\omega$।

\textbf{5.} $E_k = \tfrac12 Mv_G^2 + \tfrac12(\tfrac12 MR^2)(v_G/r)^2 = \tfrac12
Mv_G^2(1 + R^2/2r^2)$; $R^2/2r^2 = 9/(2 \times 0.16) = 28$: यानी गुणक $29$ ---
लगभग सारी ऊर्जा घूर्णन में है।

\textbf{6.} भार $Mg$ नीचे की ओर, तनाव $T$ ऊपर की ओर: $Ma_G = Mg - T$।

\textbf{7.} अक्ष के परितः केवल तनाव का आघूर्ण है (\omterm{def:b1:angular-momentum:moment}{उत्तोलक भुजा} $r$):
$J\dot\omega = Tr$।

\textbf{8.} $\dot\omega = a_G/r$: $T = Ja_G/r^2 = (R^2/2r^2)Ma_G$; फिर
$Ma_G = Mg - (R^2/2r^2)Ma_G$, $a_G = g/29 = \qty{0.34}{m/s^2}$।

\textbf{9.} $T = M(g - a_G) = 0.050 \times 9.47 = \qty{0.47}{N}$, यानी भार का
$97\%$: डोरी उसे लगभग थामे ही रखती है।

\textbf{10.} $t = \sqrt{2h/a_G} = \qty{2.4}{s}$; $v_G = a_Gt = \qty{0.82}{m/s}$।
पत्थर के लिए: \qty{0.45}{s} और \qty{4.4}{m/s}।

\textbf{11.} $\omega = v_G/r = \qty{205}{rad/s} = \qty{1960}{rpm}$।

\textbf{12.} $\tfrac12 Mv_G^2 = \qty{0.017}{J}$ जबकि $\tfrac12 J\omega^2 =
\qty{0.47}{J}$: यानी गिरने से $28$ गुना अधिक ऊर्जा घूर्णन में।

\textbf{13.} $Mgh = \qty{0.49}{J}$; $\tfrac12 Mv_G^2 \times 29 = 0.5 \times 0.050
\times 0.67 \times 29 = \qty{0.49}{J}$। तनाव डोरी के उस बिंदु पर लगता है जो
विराम में है (डोरी हिलती ही नहीं): यानी शून्य शक्ति।

\textbf{14.} रोकने वाला बल डोरी के अनुदिश, अक्ष से होकर लगता है: उसका उस
अक्ष के परितः कोई आघूर्ण नहीं; अतः \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} $J\omega$ अछूता रहता है। केवल
कोई स्पर्शरेखीय बल (धुरी पर घर्षण) ही घूर्णन को धीमा कर सकता है।

\textbf{15.} $T \approx Mv_G/\Delta t + Mg = 0.050 \times 0.82/0.005 + 0.49 =
\qty{8.7}{N}$, यानी भार से अठारह गुना --- इसीलिए उँगली को वह झटका महसूस
होता है।

\textbf{16.} $J\dot\omega = -\Gamma$: $t = J\omega/\Gamma = 2.25\times10^{-5} \times
205/10^{-4} = \qty{46}{s}$।

\textbf{17.} घूर्णन की पूरी ऊर्जा $\tfrac12 J\omega^2 = \qty{0.47}{J}$
$Mgh'$ में बदलती है: $h' = \qty{0.96}{m}$, यानी लगभग पूरी ऊँचाई; व्यवहार
में इससे कम, क्योंकि सोते समय धुरी पर घर्षण होता है और झटके में भी हानि
होती है।

\textbf{18.} $a_G$ नीचे की ओर है (ऊपर जाते हुए यो-यो मंदित होता है) जबकि
$v_G$ ऊपर की ओर: $T = M(g - a_G) < Mg$ --- यानी भार से छोटा।

\textbf{19.} $r = R$: गुणक $1 + 1/2 = 1.5$, $a_G = 2g/3 = \qty{6.5}{m/s^2}$
--- यानी तेज़ गिरना, थोड़ा घूर्णन: पतली धुरी ही यो-यो को यो-यो बनाती है।

\textbf{20.} \omterm{ex:b1:systems-of-points:rolling}{बिना फिसले लुढ़कना}: $C$ पर के पदार्थ-बिंदु का वेग शून्य है;
और जिस \omterm{def:b1:systems-of-points:solid}{दृढ़ पिंड} का एक बिंदु विराम में हो, उसका वेग-क्षेत्र उसी बिंदु के
परितः एक घूर्णन होता है ($C$ से जाती अक्ष, वही $\omega$)।

\textbf{21.} $F$ धुरी के नीचे, मेज़ से $R - r$ ऊँचाई पर, क्षैतिज दिशा में
हाथ की ओर लगता है: उसका आघूर्ण $F(R - r)$ है, $C$ के परितः, और उसी दिशा में
जो फिरकी को हाथ की ओर लुढ़काती है; भार और प्रतिक्रिया $C$ से होकर जाते हैं
(कोई आघूर्ण नहीं)।

\textbf{22.} $(J + MR^2)\dot\omega = F(R - r)$, $a_G = R\dot\omega = FR(R - r)/
(\tfrac32 MR^2) = 2F(R - r)/3MR > 0$: यानी हाथ की ओर।

\textbf{23.} $F$ की क्रिया-रेखा $C$ से तब होकर जाती है जब धुरी के नीचे से
आती डोरी मेज़ को $C$ पर मिले: $\cos\beta = r/R$,
$\beta = 82^\circ$। इससे खड़ी खिंचाई पर: $C$ के परितः आघूर्ण उलट जाता है और फिरकी
दूर लुढ़कती है (और $\beta$ के पर्याप्त बड़े होने पर उठ जाती है)।

\textbf{24.} $a_G = 2 \times 0.20 \times 0.026/(3 \times 0.050 \times 0.030) =
\qty{2.3}{m/s^2}$; घर्षण $Ma_G = F - f$ से (घर्षण $F$ का विरोध करता है):
$f = 0.20 - 0.050 \times 2.3 = \qty{0.085}{N}$; उसे $\mu_sMg
\approx \qty{0.2}{N}$ से नीचे रहना चाहिए ($\mu_s = 0.4$ के लिए), अर्थात् $F \lesssim \qty{0.5}{N}$।

\textbf{25.} \omterm{def:b1:systems-of-points:cm}{द्रव्यमान केंद्र}: $G$ का स्थानांतरण और तनाव; किसी अक्ष के
परितः \omterm{def:b1:angular-momentum:L}{कोणीय संवेग}: घूर्णन और लुढ़कने की दिशा; ऊर्जा: तली पर चाल और चढ़ाई
की ऊँचाई।
\end{solution}
