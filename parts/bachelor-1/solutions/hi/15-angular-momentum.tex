\section*{अध्याय \ref{ch:b1:angular-momentum} --- कोणीय संवेग}

\begin{solution}{exo:b1:angular-momentum:1}
$Fd = 50 \times 0.30 = \qty{15}{N.m}$; $Fd\sin 60^\circ = \qty{13}{N.m}$;
हत्थे के अनुदिश क्रिया-रेखा नट से होकर जाती है: $0$।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:2}
$L = mrv = 5.97\times10^{24} \times 1.50\times10^{11} \times 2.98\times10^4 =
\qty{2.7e40}{kg.m^2/s}$; $\dd A/\dd t = L/2m = rv/2 = \qty{2.2e15}{m^2/s}$।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:3}
$r = \ell\sin 30^\circ = \qty{0.50}{m}$, $\omega = \qty{3.4}{rad/s}$:
$L_z/m = r^2\omega = \qty{0.84}{m^2/s}$। तनाव की क्रिया-रेखा धुरी से होकर
जाती है (उससे होकर जाती किसी भी अक्ष के परितः शून्य \omterm{def:b1:angular-momentum:moment}{आघूर्ण}); और भार
ऊर्ध्वाधर अक्ष के समांतर है (उसके परितः शून्य \omterm{def:b1:angular-momentum:moment}{आघूर्ण}): अतः $L_z$
संरक्षित है।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:4}
$J_1 = 4.0 + 2 \times 3.0 \times 0.64 = \qty{7.8}{kg.m^2}$, $J_2 = 4.0 +
2 \times 3.0 \times 0.04 = \qty{4.2}{kg.m^2}$; $\omega_2 = \omega_1J_1/J_2 =
1.85\omega_1$: यानी प्रति सेकंड \qty{3.7}{} चक्कर। $E_k = L^2/2J$: अनुपात
$J_1/J_2 = 1.85$; और यह कार्य माँसपेशियाँ करती हैं, जो बाँहों को अपकेंद्र
प्रवृत्ति के विरुद्ध भीतर खींचती हैं।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:5}
$L_\Delta = m\ell^2\dot\theta$; तनाव: शून्य \omterm{def:b1:angular-momentum:moment}{आघूर्ण} (अक्ष से मिलता है);
भार: $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{उत्तोलक भुजा} $\ell\sin\theta$, प्रत्यानयक)।
$m\ell^2\ddot\theta = -mg\ell\sin\theta$।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:6}
$r_pv_p = r_av_a$: $v_p/v_a = r_a/r_p = 1.034$; और $(v_p + v_a)/2 \approx
\qty{29.8}{km/s}$ के साथ: $v_p = \qty{30.3}{km/s}$, $v_a = \qty{29.3}{km/s}$।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:7}
$L = mrv$ संरक्षित: $v_2 = v_1r_1/r_2 = \qty{4.0}{m/s}$, $\omega_2 = v_2/r_2
= \qty{16}{rad/s}$ (\qty{4}{rad/s} से)। $E_1 = \qty{0.40}{J}$, $E_2 =
\qty{1.6}{J}$: अतः खिंचाव \qty{1.2}{J} का कार्य करता है। तनाव त्रिज्य है,
पर अब चकती के विस्थापन का एक त्रिज्य घटक भी है (वह भीतर की ओर सर्पिल बनाती
है): इसलिए कार्य $\vect T\cdot\dd\vect\ell$ अब शून्य नहीं रहता।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:8}
धुरी के परितः \omterm{def:b1:angular-momentum:moment}{आघूर्ण} (लंबाइयाँ धुरी से नापी गई हैं): बच्चा $30 \times 1.5 =
45$ (बाएँ); तख़्ते का भार उसके केंद्र पर, दाईं ओर $0.5$: $20 \times
0.5 = 10$ (दाएँ); दूसरा बच्चा दाईं ओर $x$ पर: $45x$। संतुलन:
$45 = 10 + 45x$, $x = \qty{0.78}{m}$। धुरी कुल भार सँभालती है,
$95 \times 9.81 = \qty{932}{N}$।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:9}
द्रव्यमान: $m_1a = m_1g - T_1$, $m_2a = T_2 - m_2g$; घिरनी: $J\dot\omega =
(T_1 - T_2)R$, जहाँ $a = R\dot\omega$। जोड़ने पर: $a = (m_1 - m_2)g/(m_1 + m_2 +
J/R^2) = 9.81/(3.0 + 1.0) = \qty{2.5}{m/s^2}$ (जबकि \qty{3.3}{m/s^2});
$T_1 = m_1(g - a) = \qty{14.7}{N}$, $T_2 = m_2(g + a) = \qty{12.3}{N}$ ---
असमान, क्योंकि घिरनी को कोई शुद्ध \omterm{def:b1:angular-momentum:moment}{आघूर्ण} चाहिए।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:10}
$r_1v_1 = r_2v_{\perp 2}$: $v_{\perp 2} = 40 \times 0.5/10 = \qty{2.0}{km/s}$।
\omterm{cor:b1:angular-momentum:central}{क्षेत्रफल का नियम} केवल लंब घटक देता है; त्रिज्य घटक के लिए ऊर्जा-संरक्षण
चाहिए (\cref{ch:b1:central-forces})।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:11}
$J\omega$ अचर, और $J \propto R^2$: $T' = T(R'/R)^2 = 25 \times 86400
\times (10^4/7\times10^8)^2 = \qty{4.4e-4}{s}$। भूमध्य-रेखीय चाल $2\pi R'/T'
= \qty{1.4e8}{m/s}$, यानी प्रकाश की चाल की आधी: संकुचन पूरा \omterm{def:b1:angular-momentum:L}{कोणीय संवेग} रख
नहीं सकता --- बहुत कुछ छूट जाता है (और आपेक्षिकता भी आ जाती है)।
\end{solution}

\begin{solution}{exo:b1:angular-momentum:12}
$L_z = mr^2\omega$, और $r$ बदलता है: $\dd L_z/\dd t = 2mr\dot r\omega \neq 0$।
यह छड़ की अनुप्रस्थ प्रतिक्रिया $N$ का \omterm{def:b1:angular-momentum:moment}{आघूर्ण} ही होना चाहिए: $rN =
2mr\dot r\omega$, अतः $N = 2m\dot r\omega$ --- यानी बाहर की ओर फिसलते मनके को
छड़ बग़ल से धकेलती है (\cref{ch:b1:non-inertial-frames} का कोरिओलिस पद)।
\end{solution}

\begin{solution}{pb:b1:angular-momentum:1}
\textbf{1.} $m\omega^2r = GMm/r^2$: $\omega = \sqrt{GM/r^3} = \sqrt{3.98\times
10^{14}/5.66\times10^{25}} = \qty{2.65e-6}{rad/s}$, आवर्तकाल $2\pi/\omega =
\qty{2.37e6}{s} = \qty{27.4}{d}$।

\textbf{2.} $L_{\text{कक्षा}} = 7.35\times10^{22} \times 1.475\times10^{17}
\times 2.65\times10^{-6} = \qty{2.9e34}{kg.m^2/s}$।

\textbf{3.} $v = \sqrt{GM/r}$, $L = mrv = m\sqrt{GMr}$; $\dd L/\dd r =
\tfrac12 m\sqrt{GM/r} = L/2r$।

\textbf{4.} $J = 0.33 \times 5.97\times10^{24} \times 4.06\times10^{13} =
\qty{8.0e37}{kg.m^2}$; $\Omega = \qty{7.29e-5}{rad/s}$; $L_{\text{चक्रण}} =
\qty{5.8e33}{kg.m^2/s}$, यानी $L_{\text{कक्षा}}$ का पाँचवाँ भाग।

\textbf{5.} $J_m \approx 0.4 \times 7.35\times10^{22} \times 3.0\times10^{12} =
\qty{8.9e34}{kg.m^2}$, और $\omega$ से गुणा करने पर \qty{2.4e29}{kg.m^2/s}, यानी
कक्षा वाले का $10^{-5}$: अतः नगण्य।

\textbf{6.} सूर्य का आकर्षण निकाय के द्रव्यमान केंद्र पर लगता है (प्रथम
कोटि में उसके परितः कोई \omterm{def:b1:angular-momentum:moment}{आघूर्ण} नहीं); और उस युग्म पर उसका ज्वारीय प्रभाव
छोटा है: अतः यहाँ अपेक्षित यथार्थता तक कुल $L$ संरक्षित है।

\textbf{7.} पास वाला उभार, जो पृथ्वी--चंद्र रेखा से आगे है, चंद्रमा द्वारा
ऐसे बल से खींचा जाता है जिसका स्पर्शरेखीय घटक पृथ्वी के चक्रण का विरोध
करता है।

\textbf{8.} चक्रण पर ऋणात्मक \omterm{def:b1:angular-momentum:moment}{आघूर्ण}: पृथ्वी धीमी पड़ती है, दिन लंबा होता
है।

\textbf{9.} उभार चंद्रमा को आगे खींचता है: कक्षा पर धनात्मक \omterm{def:b1:angular-momentum:moment}{आघूर्ण}, अतः
$L_{\text{कक्षा}}$ बढ़ता है।

\textbf{10.} $L_{\text{कक्षा}} \propto \sqrt r$ बढ़ता है, अतः $r$ बढ़ता है; और $v \propto
1/\sqrt r$ घटता है।

\textbf{11.} आगे की ओर का खिंचाव धनात्मक कार्य करता है और चंद्रमा की ऊर्जा
बढ़ा देता है; और ऊँची कक्षा धीमी कक्षा होती है: ऊर्जा ऊँचाई में (स्थितिज
रूप में) जाती है, खोई गई \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} से भी अधिक।

\textbf{12.} पृथ्वी का ज्वारीय बंधन: दिन मास के बराबर, कोई आगे बढ़ा उभार
नहीं, कोई \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} नहीं।

\textbf{13.} $\Delta\Omega/\Omega = -\qty{2.3e-3}{s}/\qty{86164}{s} =
\num{-2.7e-8}$ प्रति शताब्दी।

\textbf{14.} $\Delta L_{\text{चक्रण}} = L_{\text{चक्रण}}\Delta\Omega/\Omega =
\qty{-1.55e26}{kg.m^2/s}$ प्रति शताब्दी।

\textbf{15.} $\Delta L_{\text{कक्षा}} = +\qty{1.55e26}{kg.m^2/s}$ प्रति शताब्दी।

\textbf{16.} $\Delta r = 2r\,\Delta L/L = 2 \times 3.84\times10^8 \times 1.55
\times10^{26}/2.9\times10^{34} = \qty{4.1}{m}$ प्रति शताब्दी, यानी \qty{4.1}{cm/yr}:
जो मापे गए \qty{3.8}{cm/yr} के निकट है (दिन के लंबे होने का एक ग़ैर-ज्वारीय
हिस्सा भी है)।

\textbf{17.} $\Delta T/T = \tfrac32\Delta r/r = 1.6\times10^{-8}$: $2.37\times
10^6 \times 1.6\times10^{-8} = \qty{0.04}{s}$ प्रति शताब्दी।

\textbf{18.} $\Delta E_{\text{घूर्णन}} = L_{\text{चक्रण}}\Delta\Omega = 5.8\times10^{33}
\times 7.29\times10^{-5} \times (-2.7\times10^{-8}) = \qty{-1.1e22}{J}$ प्रति
शताब्दी।

\textbf{19.} $\Delta E_{\text{कक्षा}} = GMm\Delta r/2r^2 = 2.93\times10^{37}
\times 4.1/(2 \times 1.475\times10^{17}) = \qty{+4.1e20}{J}$ प्रति शताब्दी।

\textbf{20.} शुद्ध $-1.1\times10^{22} + 4\times10^{20} \approx \qty{-1.06e22}{J}$
प्रति शताब्दी: यानी $3.4\times10^{12}\ \unit{W}$, जो महासागरों और ठोस पृथ्वी
में ज्वारीय घर्षण से ऊष्मा बनकर क्षय हो जाता है।

\textbf{21.} मानवता की शक्ति का पाँचवाँ भाग, और भूतापीय प्रवाह का चौदहवाँ
--- छोटा, पर स्थायी एक तापक।

\textbf{22.} $L_{\text{कुल}} = 3.5\times10^{34}$। $r = 5.5\times10^8$ पर:
$\omega = \sqrt{GM/r^3} = \qty{1.55e-6}{rad/s}$, $J\omega = 1.2\times10^{32}$,
$m\sqrt{GMr} = 7.35\times10^{22} \times 4.68\times10^{11} = 3.44\times10^{34}$;
योग $3.45\times10^{34}$: यानी संगत। आवर्तकाल $2\pi/\omega = \qty{4.1e6}{s}
= \qty{47}{d}$।

\textbf{23.} सूर्य के ज्वार पृथ्वी को चंद्रमा की कक्षीय दर से भी नीचे धीमा
करते रहते हैं, इसलिए तब पृथ्वी--चंद्र \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} उलट जाएगा; और वैसे भी सूर्य
उससे बहुत पहले बदल चुका होगा (काल-मापक दसियों अरब वर्ष का है)।

\textbf{24.} पृथ्वी ने चंद्रमा पर ज्वार उठाए, जिन्होंने उसके चक्रण को तब
तक रोका जब तक एक ही मुख हमारी ओर न टिक गया; और हलका होने तथा किसी भारी
साथी के पास होने के कारण चंद्रमा का बंधन कहीं अधिक तेज़ हुआ।

\textbf{25.} संरक्षित: कुल \omterm{def:b1:angular-momentum:L}{कोणीय संवेग}। बदला गया: \omterm{def:b1:angular-momentum:L}{कोणीय संवेग}, पृथ्वी के
चक्रण से चंद्रमा की कक्षा में, लगभग \qty{1.6e26}{SI}
प्रति शताब्दी। क्षय हुआ: \omterm{thm:b1:work-and-energy:em}{यांत्रिक ऊर्जा}, कुछ टेरावाट, ज्वारीय ऊष्मा के
रूप में।
\end{solution}
