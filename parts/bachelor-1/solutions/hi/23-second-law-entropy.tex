\section*{अध्याय \ref{ch:b1:second-law-entropy} --- दूसरा नियम: एन्ट्रॉपी}

\begin{solution}{exo:b1:second-law-entropy:1}
$\Delta S = mL/T$: गलन $334000/273 = \qty{1.22}{kJ/K}$; क्वथन $2260000/
373 = \qty{6.06}{kJ/K}$। वाष्पन अणुओं को एक-दूसरे से पूरी तरह मुक्त कर देता
है --- यानी कहीं अधिक सूक्ष्म अव्यवस्था, और प्रति \omterm{def:b1:kinetic-theory:temperature}{केल्विन} कहीं अधिक ऊष्मा।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:2}
$\Delta S = mc\ln(T_2/T_1) = 4180\ln(353/293) = \qty{0.78}{kJ/K}$ --- यह एक
अवस्था फलन है: विधि से स्वतंत्र (\emph{सृजित} एन्ट्रॉपी नहीं)।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:3}
$\Delta S = nR\ln2 = \qty{5.8}{J/K}$। उत्क्रमणीय समतापी: $Q = nRT\ln2$, जो
$T$ पर के ऊष्मा-भंडार से आती है, $S_{\mathrm{exch}} = nR\ln2$, सृजित $0$। जूल:
$Q = 0$, $S_{\mathrm{exch}} = 0$, सृजित \qty{5.8}{J/K}।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:4}
$S = nC_{V,m}\ln T + nR\ln V$ अचर: यानी $T^{C_{V,m}}V^R$ अचर, अर्थात्
$TV^{R/C_{V,m}} = TV^{\gamma - 1}$ अचर। अचानक किया गया संपीडन: $Q = 0$, अतः
$S_{\mathrm{exch}} = 0$ और $\Delta S = S_{\mathrm{created}}$; $T$ के साथ: $300 \to
643$ K, $V$: $24.9 \to 10.7$ L: $\Delta S = 20.8\ln(643/300) + 8.314\ln(10.7/
24.9) = 15.9 - 7.0 = \qty{+8.9}{J/K}$ सृजित।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:5}
गुटका: $450\ln(293/373) = \qty{-109}{J/K}$; झील: $+36000/293 = \qty{+123}{J/K}$;
विश्व: $+\qty{14}{J/K}$।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:6}
$T_f = \qty{350}{K}$; $S = 4180[\ln(350/400) + \ln(350/300)] = 4180 \times
0.0206 = \qty{86}{J/K}$, सब सृजित। $(T_1 + T_2)^2 - 4T_1T_2 = (T_1 - T_2)^2
\geq 0$, अतः लघुगणक का प्रांतांक $\geq 1$ रहता है।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:7}
$S = Q(1/T_c - 1/T_h)$: $1000(1/280 - 1/300) = \qty{0.24}{J/K}$; भट्ठी से
कमरे तक: $1000(1/300 - 1/1000) = \qty{2.3}{J/K}$, यानी दस गुना अधिक: कोई
ऊष्मा-प्रवाह ताप का जितना बड़ा अंतर पार करे वह उतना ही अनुत्क्रमणीय होता है
--- भट्ठी-से-कमरे वाली गिरावट ही वह जगह है जहाँ उच्च कोटि की ऊर्जा उड़ा दी
जाती है।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:8}
चक्र: $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$, जहाँ $S_{\mathrm{created}} \geq 0$,
अतः $Q \leq 0$ और $W = -Q \geq 0$: यानी मशीन कोई कार्य दे ही नहीं सकती।
समुद्र एक अकेला ऊष्मा-भंडार है: उसकी ऊष्मा को कार्य में तब तक नहीं बदला जा
सकता जब तक एन्ट्रॉपी उँडेलने के लिए कोई ठंडा पिंड न हो।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:9}
हर गैस $V$ से $2V$ तक फैलती है, अचर $T$ पर: $\Delta S = 2 \times R\ln2
= \qty{11.5}{J/K}$, सृजित (कोई ऊष्मा का आदान-प्रदान नहीं, डिब्बा विलगित
है)। दोनों ओर नाइट्रोजन: दीवार हटाने से स्थूल रूप से कुछ नहीं बदलता ---
$\Delta S = 0$ (गिब्स का विरोधाभास: एक-से अणु ``मिलते'' ही नहीं)।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:10}
\qty{293}{K} के कमरे में $Q = \qty{3.6e5}{J}$: $S = \qty{1230}{J/K}$
सृजित --- तंतु में (विद्युत कार्य ऊष्मा में बदला) और गरम तंतु से ठंडे कमरे
तक उस ऊष्मा के बहने में; विद्युत कार्य स्वयं कोई एन्ट्रॉपी नहीं लाया।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:11}
हर स्थिति में $\Delta S_{\text{पिंड}} = C\ln(T_0/T_1)$। (क) आदान-प्रदान
$C(T_0 - T_1)/T_0$: सृजित $C[T_1/T_0 - 1 - \ln(T_1/T_0)]$। (ख) हर एक में
अनुपात $r = \sqrt{T_1/T_0}$ वाले दो चरण: सृजित $2C[r - 1 - \ln r]$, जो कम है
($T_1/T_0 = 2$ के लिए: $0.193C$, जबकि पहले $0.307C$)। (ग) अपरिमित चरण:
हर एक $C[(1 + \epsilon) - 1 - \ln(1 + \epsilon)] \approx C\epsilon^2/2$ सृजित करता
है, जिनका योग $\epsilon \to 0$ पर शून्य हो जाता है: उत्क्रमणीय शीतलन के लिए
हर बीच के ताप पर एक ऊष्मा-भंडार चाहिए --- ऊष्मा को कभी कोई परिमित अंतर पार
नहीं करना चाहिए।
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:12}
$2^{-10} = 10^{-3}$; $2^{-100} = \num{8e-31}$; $2^{-10^{22}} = 10^{-3\times10^{21}}$।
$k_B\ln2^N = Nk_B\ln2 = nR\ln2$। ``असंभव'' का कड़ा अर्थ है प्रायिकता
शून्य; यहाँ वह $10^{-3\times10^{21}}$ है --- शून्य नहीं, पर इतनी छोटी संख्या कि
कोई भेद कभी देखा ही न जा सके: अनुत्क्रमणीयता $10^{23}$ के पैमाने पर
सांख्यिकी है।
\end{solution}

\begin{solution}{pb:b1:second-law-entropy:1}
\textbf{1.} कॉफ़ी के लिए $Q = mc(T_0 - T_1) = 0.25 \times 4180 \times (-70) = \qty{-73}{kJ}$:
यानी कमरे को \qty{73}{kJ} मिलते हैं, $\Delta S_{\text{कमरा}} = 73150/293
= \qty{+250}{J/K}$।

\textbf{2.} $mc\ln(T_0/T_1) = 1045\ln(293/363) = \qty{-224}{J/K}$।

\textbf{3.} $S_{\mathrm{created}} = -224 - (-250) = \qty{+26}{J/K} > 0$।

\textbf{4.} $\Delta S = 1045\ln(330/363) = \qty{-99.6}{J/K}$; $Q = 1045 \times
(-33) = \qty{-34.5}{kJ}$, $S_{\mathrm{exch}} = \qty{-117.7}{J/K}$: अब तक सृजित
\qty{18}{J/K} --- यानी अधिकांश अनुत्क्रमणीयता तभी होती है जब कॉफ़ी सबसे गरम
रहती है।

\textbf{5.} $\Delta S - Q/T_0 = mc[\ln(T_0/T_1) - (T_0 - T_1)/T_0] = mc[x - 1 -
\ln x]$, जहाँ $x = T_1/T_0$; $f'(x) = 1 - 1/x$ $x = 1$ पर लुप्त होता है, जहाँ
$f = 0$ है, और $f'' = 1/x^2 > 0$: अतः निम्निष्ठ $0$।

\textbf{6.} नहीं: आरंभिक और अंतिम अवस्थाएँ वही हैं, ऊष्मा वही है, कमरा वही
ऊष्मा-भंडार है --- सृजित एन्ट्रॉपी अवस्थाओं से तय होती है, गति से नहीं।

\textbf{7.} $T_f = \qty{350}{K}$; सृजित $4180\ln(350^2/120000) = \qty{86}{J/K}$।

\textbf{8.} उत्क्रमणीय: कुल एन्ट्रॉपी संरक्षित: $mc\ln(T_f'/T_1) + mc\ln(T_f'/
T_2) = 0$, अतः $T_f'^2 = T_1T_2$।

\textbf{9.} $T_f' = \sqrt{120000} = \qty{346.4}{K}$; दोनों पानी $mc[(T_1 -
T_f') + (T_2 - T_f')] = 4180 \times (53.6 - 46.4) = \qty{30}{kJ}$ खोते हैं, जो
कार्य के रूप में मिलते हैं।

\textbf{10.} गरम: $4180\ln(346.4/400) = \qty{-601}{J/K}$; ठंडा: $4180\ln(346.4/
300) = \qty{+601}{J/K}$: योग शून्य, जैसा उत्क्रमणीयता माँगती है।

\textbf{11.} $T_0S_{\mathrm{created}} = 300 \times 86 = \qty{26}{kJ}$, जो वसूली
जा सकने वाली \qty{30}{kJ} के पास है: सृजित एन्ट्रॉपी, गुणा संदर्भ ताप, वही
कार्य है जिसे अनुत्क्रमणीय पथ ने फेंक दिया।

\textbf{12.} उसके लिए ऐसा इंजन चाहिए जो दो गुनगुने पानियों के बीच काम करे,
जिनके ताप लगातार बदलते रहते हैं --- किसी बड़ी मशीन के लिए कुछ दसियों
किलोजूल: उतने के लायक नहीं, पर दूसरा नियम कहता है कि वह कार्य वहाँ था।

\textbf{13.} भीतरी भाग से मिली $Q_c/T_c = 40/278 = \qty{0.144}{W/K}$;
कमरे को दी गई $Q_0/T_0 = (40 + W)/293$; और चक्र पर संतुलन:
$0 = 0.144 - (40 + W)/293 + S_c$।

\textbf{14.} $S_c \geq 0$: $(40 + W)/293 \geq 0.144$, $W \geq 42.2 - 40 =
\qty{2.2}{W}$; निष्पादन गुणांक $\leq 40/2.2 = 18$ ($= T_c/(T_0 - T_c)$)।

\textbf{15.} $\mathrm{COP}_{\max} = T_c/(T_0 - T_c)$: किसी हिमकारक के लिए $255/38 = 6.7$
--- डिब्बा जितना ठंडा, हटाया गया हर जूल उतना ही महँगा।

\textbf{16.} $W = 15$ के साथ: $S_c = (55/293) - 0.144 = \qty{0.044}{W/K}$ हर
सेकंड सृजित।

\textbf{17.} $W \geq 200 \times 15/278 = \qty{10.8}{W}$; मशीन \qty{200}{W} और
अपना कार्य, दोनों कमरे में उँडेल देती है, जबकि भीतरी भाग ठंडा बना रहता है:
कमरे का शुद्ध लाभ $W$ है --- यानी खुला प्रशीतक एक तापक है।

\textbf{18.} $W = 0$ से $S_c = Q_c(1/T_0 - 1/T_c) < 0$ मिलता: यानी वर्जित ---
\omterm{prop:b1:second-law-entropy:clausiuskelvin}{क्लॉज़ियस का कथन}।

\textbf{19.} पहले: $1$ (सब बाईं ओर); बाद में: $2^N$; अतः $\Delta S = k_B\ln2^N =
Nk_B\ln2$।

\textbf{20.} $Nk_B = nR$: अतः $\Delta S = nR\ln2$, यानी जूल-प्रसार वाला परिणाम।

\textbf{21.} प्रति अणु एक बिट: यानी $N$ बिट; और $\Delta S/(k_B\ln2) = N$ ---
एन्ट्रॉपी वही सूचना गिनती है जो अणुओं को ठीक-ठीक तय करने के लिए चाहिए
होती।

\textbf{22.} $2^{-50} = \num{9e-16}$; और किसी \omterm{def:b1:kinetic-theory:scales}{मोल} के लिए $2^{-6\times10^{23}} =
10^{-1.8\times10^{23}}$।

\textbf{23.} कोई ऊष्मा नहीं ($Q = 0$): आदान-प्रदान $0$, सृजित $nR\ln2$ ---
यानी सबसे शुद्ध स्थिति: न कार्य, न ऊष्मा, केवल गैस का फैल जाना।

\textbf{24.} मिला $W = nRT\ln2$, कमरे को दी गई $Q = -nRT\ln2$,
$S_{\mathrm{exch}} = -nR\ln2$ (गैस की एन्ट्रॉपी उत्क्रमणीय ढंग से अपने आरंभिक
मान पर लौट आती है): मुक्त प्रसार को उलटने में कार्य $nRT\ln2$ लगता है और
कमरे में $nR\ln2$ एन्ट्रॉपी उँडेल दी जाती है --- विश्व एक बार सृजित हुई वह
$nR\ln2$ हमेशा के लिए अपने पास रख लेता है।

\textbf{25.} वास्तविक कुछ भी एन्ट्रॉपी को संरक्षित नहीं रखता --- केवल
आदर्शीकृत उत्क्रमणीय सीमा में; नियम हर उस प्रक्रिया को मना करता है जो उसे
नष्ट कर दे (ऊष्मा का ऊपर चढ़ना, एक ही ऊष्मा-भंडार से कार्य, मुफ़्त में
अ-मिश्रण); और सृजित हर बिट खोया हुआ कार्य है, $T_0S_{\mathrm{created}}$ जितना।
\end{solution}
