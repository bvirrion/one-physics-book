\begin{solution}{exo:b1:central-forces:7}
$a = \qty{24490}{km}$। $v_1 = \qty{7.67}{km/s}$; दीर्घवृत्त पर $r_1$ पर:
$\sqrt{GM(2/r_1 - 1/a)} = \qty{10.07}{km/s}$: अतः $\Delta v_1 = \qty{2.4}{km/s}$।
$r_2$ पर: $\sqrt{GM(2/r_2 - 1/a)} = \qty{1.62}{km/s}$, और वृत्तीय
\qty{3.07}{km/s}: $\Delta v_2 = \qty{1.45}{km/s}$। कुल \qty{3.85}{km/s};
अवधि $\pi\sqrt{a^3/GM} = \qty{1.9e4}{s} = \qty{5.3}{h}$।
\end{solution}

\begin{solution}{exo:b1:central-forces:8}
$E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ का हल $r_c = L^2/GMm^2$ है।
$E_{p,\mathrm{eff}}'' = -2GMm/r^3 + 3L^2/mr^4$; और $r_c$ पर $L^2 = GMm^2r_c$
से $E'' = GMm/r_c^3$ मिलता है, अतः $\omega_r^2 = E''/m = GM/r_c^3 = \omega_{\text{कक्षा}}^2$।
कोई छोटा त्रिज्य विक्षोभ ठीक प्रति परिक्रमा एक बार दोलन करता है: अतः कक्षा
स्वयं पर बंद हो जाती है --- यानी थोड़ी उत्केंद्र एक दीर्घवृत्त, न कि कोई
गुलाब-वक्र।
\end{solution}

\begin{solution}{exo:b1:central-forces:9}
स्टेशन और दल दोनों पर केवल गुरुत्व लगता है; दोनों एक ही कक्षा पर गिरते हैं,
इसलिए उनके बीच किसी संस्पर्श बल की आवश्यकता ही नहीं --- कोई फ़र्श नहीं
धकेलता, और इसीलिए ``भारहीनता''। $g = GM/r^2 = 3.99\times10^{14}/(6.77\times10^6)^2
= \qty{8.7}{m/s^2}$: गुरुत्व लगभग पूरी ताक़त पर है; भार अनुपस्थित नहीं है,
वह पूरा का पूरा पथ मोड़ने में लग रहा है।
\end{solution}

\begin{solution}{exo:b1:central-forces:12}
$E_m = -GMm/2r$ घटता है $\Rightarrow$ $r$ घटता है $\Rightarrow$ $v =
\sqrt{GM/r}$ बढ़ता है। \omterm{thm:b1:work-and-energy:ke}{गतिज ऊर्जा} $GMm/2r$ $\abs{\Delta E}$ से बढ़ती है
जबकि \omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} $-GMm/r$ $2\abs{\Delta E}$ से गिरती है: यानी मुक्त हुई
\omterm{def:b1:work-and-energy:conservative}{स्थितिज ऊर्जा} का आधा गतिज बनता है और आधा कर्षण से ऊष्मा के रूप में क्षय
हो जाता है।
\end{solution}

\begin{solution}{pb:b1:central-forces:1}
