\section*{अध्याय \ref{ch:b1:point-kinematics} --- बिंदु की शुद्धगतिकी}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: \qty{3.0}{s} पर \qty{12}{m/s}; $a = \qty{4.0}{m/s^2}$;
$x(5) = \qty{50}{m}$; $\qty{100}{km/h} = \qty{27.8}{m/s}$ के लिए $t = 27.8/4.0
= \qty{6.9}{s}$।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s}); $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2})। एकसमान \omterm{cor:b1:point-kinematics:circular}{वृत्तीय गति}: अचर चाल, \omterm{cor:b1:point-kinematics:circular}{अभिकेंद्र
त्वरण}।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$;
$a = v^2/R = \qty{0.034}{m/s^2}$, यानी $g/290$।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$।
शिखर $t = 7.71/9.81 = \qty{0.79}{s}$ पर, ऊँचाई $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$। परास $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$।
भूमि पर: $(9.19, -7.71)$, यानी \qty{12}{m/s}, क्षैतिज से $40^\circ$
नीचे।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$। \qty{5}{s} बाद: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$। $2\dot r\dot\theta$ पद के दो बराबर आधे हैं: त्रिज्य \omterm{def:b1:point-kinematics:velocity}{वेग} की
दिशा $\dot\theta$ दर से मुड़ती है (जिससे
$\dot r\dot\theta\,\vect e_\theta$ मिलता है), और बाहर की ओर बढ़ने से
लंबत्रिज्य चाल $r\dot\theta$ बढ़ जाती है (एक और $\dot r\dot\theta$)।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ अक्ष की ओर; पिच $v_z \times 2\pi/\omega =
\qty{3.1}{m}$; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$;
$v$ \omterm{def:b1:wave-propagation:signal}{कला} में $x$ से चौथाई आवर्तकाल आगे है, और $a = -\omega^2x$ $x$ के विपरीत है।
$\abs v$ सबसे बड़ा $x = 0$ पर होता है, और $\abs a$ चरम बिंदुओं $x = \pm A$ पर।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$। तब
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, और गुणनफल-नियम से एक
बार और अवकलन करने पर \cref{thm:b1:point-kinematics:cylindrical} का \omterm{def:b1:point-kinematics:velocity}{त्वरण}
मिल जाता है।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
शिखर: $v = v_0\cos\alpha$, $a_N = g$: $\rho = v_0^2\cos^2\alpha/g$।
प्रक्षेप बिंदु: $v = v_0$, $a_N = g\cos\alpha$: $\rho = v_0^2/(g\cos\alpha)$।
शिखर वाली त्रिज्या $\cos^3\alpha$ गुना छोटी है: परवलय वहीं सबसे अधिक मुड़ता
है जहाँ वह सबसे धीमा है और जहाँ पूरा $\vect g$ अभिलंब है।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$। भूमि पर ($vt/R = 2\pi n$):
$\dot x = \dot y = 0$ और $\vect a = (0, v^2/R)$, ऊपर की ओर। शिखर पर
($vt/R = \pi$): $\dot x = 2v$, $\dot y = 0$: चाल $2v$।
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$। $t = \qty{2.0}{s}$ पर: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2})। पहला घूर्णन का अभिकेंद्र पद है;
दूसरा, $2\dot r\dot\theta$, केवल इसलिए है क्योंकि त्रिज्य दिशा स्वयं घूम
रही है --- भूमि पर अचर चाल से सीधे चलने में तो कोई \omterm{def:b1:point-kinematics:velocity}{त्वरण} होता ही नहीं।
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$।

\textbf{2.} $\vect T = \vect e_\theta$ ($\theta$ बढ़ते हुए गति),
$\vect N = -\vect e_r$ ($C$ की ओर), $\rho = R$।

\textbf{3.} शिखर: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$। पार्श्व:
$625 - 196 = 429$, $v = \qty{20.7}{m/s}$।

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ और $v = R\dot\theta$:
$\dot v = -g\sin\theta$। $0 < \theta < \pi$ के लिए ऋणात्मक (ऊपर चढ़ते समय
धीमा पड़ना), और नीचे उतरते समय धनात्मक।

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: तल पर
\qty{62.5}{m/s^2}; पार्श्वों पर $62.5 - 19.6 = \qty{42.9}{m/s^2}$; शिखर पर $62.5 -
39.2 = \qty{23.3}{m/s^2}$।

\textbf{6.} तल पर: \qty{62.5}{m/s^2} (स्पर्शरेखीय भाग शून्य); पार्श्वों पर:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$; शिखर पर: \qty{23.3}{m/s^2},
जो सीधे नीचे $C$ की ओर है: ऊर्ध्वाधर से कोण $0$।

\textbf{7.} तल और शिखर पर लंब ($a_T = 0$); समांतर कहीं नहीं
($a_N > 0$ सर्वत्र)।

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
तल पर \qty{2.50}{rad/s}, शिखर पर \qty{1.53}{rad/s}।

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: यह $R$ लंबाई के
सरल लोलक का समीकरण है --- पटरी पर दौड़ती डिब्बी वही लोलक-गोलक है जो शिखर के
पार निकल जाता है।

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$।

\textbf{11.} $25$, $20.7$, $15.3$, $20.7$ का माध्य: लगभग \qty{20}{m/s};
लंबाई $2\pi R = \qty{63}{m}$: यानी मोटे तौर पर \qty{3}{s}।

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: हाँ। न्यूनतम: $v_0^2 \geq
gR + 4gR = 5gR = 490$, $v_0 \geq \qty{22.1}{m/s}$।

\textbf{13.} तल पर: $\vect a = 62.5$ ऊपर की ओर, $\vect g$ नीचे की ओर:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$।

\textbf{14.} शिखर पर: दोनों नीचे की ओर, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
जो सवार को सीट में दबाता है (केंद्र की ओर, यानी उलटे बैठे सवार के लिए ऊपर
की ओर)।

\textbf{15.} पार्श्वों पर: $\vect a = 42.9$ क्षैतिज, $C$ की ओर, और साथ में
$9.8$ नीचे की ओर (स्पर्शरेखीय); $\vect a - \vect g$ क्षैतिज $42.9$ है: यानी
केंद्र की ओर बग़ल में $4.4g$।

\textbf{16.} विफल: तल पर $7.4g$ (शिखर और पार्श्व ठीक हैं)।

\textbf{17.} शिखर पर आभासी भार शून्य: $v_{\text{शिखर}}^2 = gR$;
तब $v_0^2 = gR + 4gR = 5gR$, तल पर $a_N = 5g$ और भार
$5g + g = 6g$, जो $R$ से स्वतंत्र है।

\textbf{18.} $a_N = v^2/\rho$: $v$ ऊँचाई से तय होता है (ऊर्जा), और $\rho$
पटरी की ज्यामिति से; उसी ऊँचाई पर भिन्न $\rho$ केवल $a_N$ को बदलता है।

\textbf{19.} $a_N + g = 3g$: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$।

\textbf{20.} $v_{\text{शिखर}} = \qty{15.3}{m/s}$; $a_N - g = 1.5g$: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$।

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: यह शून्य से रैखिक रूप
से बढ़ता है --- \omterm{def:b1:point-kinematics:velocity}{त्वरण} के परिवर्तन की दर अचर --- न कि वृत्त के प्रवेश पर
$0$ से $v^2/R$ तक एकाएक कूदता है, जिसे गरदन एक चोट की तरह अनुभव करती है।

\textbf{22.} एक अश्रु-बूँद: तल पर चौड़ा और हलके मोड़ वाला, जहाँ डिब्बी तेज़
है; शिखर पर कसा हुआ, जहाँ वह धीमी है। इससे भार पूरे रास्ते $3g$ के आसपास
रहता है और धीरे-धीरे चढ़ता है।

\textbf{23.} दृष्टि-रेखा: क्षैतिज दूरी \qty{50}{m}, ऊँचाई
\qty{20}{m}: $r = \qty{53.9}{m}$, उन्नयन $\beta = 21.8^\circ$। क्षैतिज \omterm{def:b1:point-kinematics:velocity}{वेग}
का अनुप्रस्थ घटक $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$ है; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$।

\textbf{24.} किसी अ-केंद्रीय बिंदु से देखने पर दृष्टि-रेखा की कोणीय दर अचर
नहीं रहती (डिब्बी पास हो तो तेज़, दूर हो तो धीमी): अचर दर से घूमता \omterm{def:b1:optical-instruments:camera}{कैमरा}
डिब्बी से भटक जाता है।

\textbf{25.} वृत्त पर \omterm{def:b1:point-kinematics:velocity}{वेग} और \omterm{def:b1:point-kinematics:velocity}{त्वरण} के लिए $C$ पर \omterm{def:b1:point-kinematics:cylindrical}{ध्रुवीय निर्देशांक};
``तेज़ होना'' बनाम ``मुड़ना'' के लिए तथा क्लोथॉयड के लिए फ्रेने विभाजन;
पीछा करने की दर के लिए कैमरे पर \omterm{def:b1:point-kinematics:cylindrical}{ध्रुवीय निर्देशांक}; और चाल के नियम के लिए
ऊर्जा (अगले अध्याय)।
\end{solution}
