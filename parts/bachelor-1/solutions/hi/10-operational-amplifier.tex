\section*{अध्याय \ref{ch:b1:operational-amplifier} --- संक्रियात्मक प्रवर्धक}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$; $V_s = \qty{5.0}{V}$; और संतृप्ति $V_e = 15/10 =
\qty{1.5}{V}$ पर।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$; निवेश \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} $R_1 = \qty{10}{k\ohm}$; $i = 1.0/10^4 =
\qty{0.10}{mA}$; $V_s = \qty{-10}{V}$।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
बिना अनुगामी के: विभाजक $1/(1 + 10) = 0.091$, यानी \qty{91}{mV}। अनुगामी के
साथ: \qty{1.0}{V} --- अनुगामी संवेदक से कुछ नहीं खींचता और भार को जो
\qty{1}{mA} चाहिए वह जुटा देता है।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$; और $V_-$ पर: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$,
अतः $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$; और $V_- = V_+$ के साथ:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$। आँकड़े: $10 \times 0.05 = \qty{0.50}{V}$;
और साझी \qty{1}{V} कुछ नहीं जोड़ती --- यही उभयनिष्ठ-विधा निराकरण है।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$: ढाल $-1000 \times 1.0 = \qty{-1000}{V/s}$;
और संतृप्ति $15/1000 = \qty{15}{ms}$ के बाद। वर्ग तरंग: \omterm{def:b1:wave-propagation:signal}{आयाम}
$ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$ की त्रिभुज
(और साथ में वह नियतांक जो आरंभिक आवेश छोड़ गया हो)।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$: दिष्ट \omterm{def:b1:filters-transfer-functions:H}{लब्धि} $-100$; कटऑफ़
$1/(2\pi R'C) = \qty{1.6}{Hz}$; और \qty{1}{kHz} पर $R'C\omega = 628$, $G =
100/628 = 0.16$ --- यानी समाकलक वाला मान $1/RC\omega$। वह
$f \gg \qty{1.6}{Hz}$ के लिए समाकलन करता है।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$: अतः $R_2 = \qty{10}{k\ohm}$; और $C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
परिमित $\mu$ के साथ: $V_s = \mu(V_e - \beta V_s)$, $\beta = 1/100$, अतः
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$। बैंड-चौड़ाई $f_T/G =
\qty{10}{kHz}$; और अनुगामी की: \qty{1}{MHz}।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
रव का हर उछाल \qty{2.0}{V} पार करते ही निर्गम पलट देता है: अतः पार करने के
आसपास तेज़ पलटावों की झड़ी (``चटचटाहट'')। उपाय: रव से चौड़ा शैथिल्य, यानी
कोई \omterm{prop:b1:operational-amplifier:schmitt}{श्मिट ट्रिगर}।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$: अतः देहलियाँ $\pm\qty{1.5}{V}$।
निर्गम $+\qty{15}{V}$ रहता है जब तक $V_e$ चढ़कर \qty{1.5}{V} न पहुँचे, फिर
$-\qty{15}{V}$ पर पलटता है, और वहीं रहता है जब तक $V_e$ \qty{-1.5}{V} से नीचे
न गिरे, फिर वापस पलटता है: यानी त्रिभुज की ही आवृत्ति की वर्ग तरंग, जिसके
किनारे देहली-पारण पर पड़ते हैं।
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
$V_s = +V_{\mathrm{sat}}$ के साथ $V_C$ $-\beta V_{\mathrm{sat}}$ से
$+V_{\mathrm{sat}}$ की ओर आवेशित होता है: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$, और $+\beta V_{\mathrm{sat}}$ तब पहुँचता है जब $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$ हो: अतः अर्ध-आवर्तकाल $RC\ln\frac{1 + \beta}{1 - \beta}$,
और इससे $T$। $\beta = \tfrac12$: $T = 2RC\ln 3 = 2.2RC$; $RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$, $R = \qty{4.6}{k\ohm}$। $V_s$: $\pm\qty{15}{V}$ की
वर्ग तरंग; और $V_C$: $\pm\qty{7.5}{V}$ के बीच चरघातांकी चाप।
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
दो विभाजक: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone}); और तारें
चिह्न तय कर देती हैं।

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$।

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ हमेशा; और पूर्ण पैमाने पर $V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$। दोनों में साझी
\qty{2.5}{V} ही उभयनिष्ठ-विधा \omterm{def:b1:dc-circuits:voltage}{विभवांतर} है।

\textbf{4.} हर मध्यबिंदु पर: $1 \parallel 1 = \qty{500}{\ohm}$; और दोनों
मध्यबिंदुओं के बीच \qty{1.0}{k\ohm}।

\textbf{5.} $10/(10 + 1) = 0.91$: यानी $9\%$ की हानि।

\textbf{6.} अनंत निवेश \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} (सेतु से कोई धारा नहीं, अतः
मध्यबिंदु अपने खुले परिपथ वाले विभव बनाए रखते हैं) और शून्य निर्गम
\omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} (अगला चरण जितना चाहे खींच सकता है)।

\textbf{7.} ठीक-ठीक $V_A$ और $V_B$; और धारा शून्य।

\textbf{8.} जैसा \cref{prop:b1:operational-amplifier:sumdiff} में: $V_+ =
V_BR_2/(R_1 + R_2)$, $V_-$ पर \omterm{thm:b1:dc-circuits:kirchhoff}{संधि नियम}, और $V_- = V_+$।

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$।

\textbf{10.} $1 + R_2/R_1 = 10$: जैसे \qty{1}{k\ohm} और \qty{9}{k\ohm}।
कुल \omterm{def:b1:filters-transfer-functions:H}{लब्धि} $1000$; और पूर्ण पैमाने पर $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$।

\textbf{11.} मिलाया हुआ: साझी \qty{2.5}{V} ठीक-ठीक कट जाती है (केवल
$V_B - V_A$ बचता है)। और $1\%$ की चूक उसका लगभग $1\%$ जाने देती है: यानी
निवेश पर लौटाई गई लगभग \qty{25}{mV} की त्रुटि, जो पूर्ण-पैमाने के संकेत से
दस गुना है --- अतः चारों प्रतिरोधकों को $0.01\%$ से भी बेहतर मिलाना पड़ता है
(या पहले उभयनिष्ठ-विधा \omterm{def:b1:dc-circuits:voltage}{विभवांतर} हटाना पड़ता है)।

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$, यानी पूर्ण पैमाने का
चालीस प्रतिशत: अतः विस्थापन साधे बिना, या कम-विस्थापन वाले (चॉपर)
प्रवर्धक के बिना, या धरन पर बिना भार के शून्य-अंशांकन के बिना, यह अनुपयोगी है।

\textbf{13.} चरण 1: $f_T/100 = \qty{10}{kHz}$; चरण 2: \qty{100}{kHz};
यानी \qty{5}{Hz} से कहीं ऊपर।

\textbf{14.} आवश्यक $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$,
जो \omterm{rem:b1:operational-amplifier:real}{आरोहण दर} से दो हज़ार गुना नीचे है: अतः नहीं।

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$।

\textbf{16.} $x = 5$: $G = 1/\sqrt{26}$, यानी \qty{-14}{dB}।

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$; विलंब $\varphi/\omega =
\qty{15}{ms}$ --- ट्रक के गुज़रने के सामने कुछ भी नहीं।

\textbf{18.} अगले चरण पर कोई भार नहीं (शून्य निर्गम \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा})
और \omterm{def:b1:filters-transfer-functions:H}{लब्धि} भी उपलब्ध; साथ ही कटऑफ़ से नीचे अभीष्ट संकेत का कोई क्षीणन नहीं।

\textbf{19.} \qty{-28}{dB} (लब्धियाँ गुणित, \omterm{def:b1:filters-transfer-functions:bode}{डेसिबल} जुड़ते हैं), और यथार्थ
इसलिए कि पहले चरण की निर्गम \omterm{def:b1:sinusoidal-impedance:impedance}{प्रतिबाधा} शून्य है: अतः दूसरा चरण उस पर भार नहीं
डालता।

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$; और संकेत के \qty{2.0}{V} से आगे निकलने पर निर्गम
$+V_{\mathrm{sat}}$, नीचे $-V_{\mathrm{sat}}$ (या तारों के अनुसार उलटा)।

\textbf{21.} वह चटचटाता है: रव के संदर्भ को पार करते ही दीपक टिमटिमाने
लगता है।

\textbf{22.} $V_{\mathrm{ref}}$ $R_2$ के द्वारा और निर्गम
$R_1$ के द्वारा हो\dots{} मानक रूप में अ-प्रतिलोमी निवेश
$V_{\mathrm{ref}}(1 - \beta) + \beta V_s$ पर बैठता है; और संदर्भ को उसी के
अनुसार खिसकाकर लेने पर पलटाव के स्तर $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$ होते हैं।
$\beta V_{\mathrm{sat}} = \qty{0.10}{V}$: अतः $\beta = 1/150$, जैसे $R_1 =
\qty{1.0}{k\ohm}$, $R_2 = \qty{149}{k\ohm}$।

\textbf{23.} \qty{2.1}{V} (चालू) और \qty{1.9}{V} (बंद): यानी विकृतियाँ
\num{1.68e-3} और \num{1.52e-3}।

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$। और
$-V_{\mathrm{sat}}$ पर LED \qty{15}{V} से उत्क्रम-अभिनत हो जाता है, जो उसकी
सह्यता के परे है: अतः प्रति-समांतर (या श्रेणी में) लगा कोई साधारण डायोड उसकी रक्षा करता है।

\textbf{25.} सेतु (पूर्ण पैमाने पर \qty{2.5}{mV}) $\to$ दो अनुगामी (कोई
भार नहीं) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{अंतर प्रवर्धक} $\times100$ $\to$ अ-प्रतिलोमी
$\times10$ $\to$ \qty{10}{Hz} का सक्रिय निम्न-पारक $\to$ \omterm{prop:b1:operational-amplifier:schmitt}{श्मिट ट्रिगर}
$\qty{2.0}{V} \pm \qty{0.1}{V}$ पर $\to$ LED। और असली सीमाएँ: \omterm{rem:b1:operational-amplifier:real}{निवेश
विस्थापन विभवांतर} तथा \omterm{prop:b1:operational-amplifier:sumdiff}{अंतर प्रवर्धक} के प्रतिरोधकों का मिलान, दोनों ही
मिलीवोल्ट के संकेत से तुलनीय।
