\begin{solution}{exo:b1:electrostatics-gauss:3}
अक्ष पर, $|x| > a$: $E_x = \dfrac{q}{4\pi\varepsilon_0}\Bigl[\dfrac{1}{(x - a)^2} +
\dfrac{1}{(x + a)^2}\Bigr]$, बाहर की ओर। $y$ अक्ष पर $x$ घटक
कट जाते हैं: $E_y = \dfrac{2qy}{4\pi\varepsilon_0(y^2 + a^2)^{3/2}}$; और $\dd E_y/\dd y \propto
(y^2 + a^2) - 3y^2 = 0$ $y = a/\sqrt2$ पर। मूलबिंदु पर क्षेत्र लुप्त हो
जाता है; और हर आवेश से दूसरे की ओर निकलती रेखाएँ मुड़कर
$\pm y$ के अनुदिश दूर चली जाती हैं --- यानी एक पल्याण-बिंदु।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:4}
दोनों आवेशों के बीच दोनों क्षेत्र $-q$ की ओर संकेत करते हैं: कोई शून्य नहीं।
$+2q$ के बाईं ओर, $2q/x^2 > q/(x + a)^2$ हमेशा। और $-q$ के आगे, उससे
$d$ दूरी पर: $2q/(a + d)^2 = q/d^2$, $a + d = \sqrt2\,d$, $d = a/(\sqrt2 - 1) = 2.4a$।
$+2q$ से दो रेखाएँ निकलती हैं, $-q$ पर समाप्त होती हर एक रेखा के बदले; बाकी
आधी अनंत तक जाती हैं, जहाँ से यह युग्म किसी आवेश $+q$ जैसा दिखता है।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:5}
$a$ त्रिज्या और $\dd a$ चौड़ाई का वलय: $\dd q = 2\pi\sigma a\,\dd a$, $\dd E_z =
\dfrac{\sigma z}{2\varepsilon_0}\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}}$; समाकलित कीजिए:
\[
\dfrac{\sigma z}{2\varepsilon_0}\Bigl[-\dfrac{1}{\sqrt{a^2 + z^2}}\Bigr]_0^R
= \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr) .
\]
और $z \ll R$: $\sigma/2\varepsilon_0$, यानी
अनंत तल। $z \gg R$: $z/\sqrt{z^2 + R^2} \approx 1 - R^2/2z^2$, अतः $E_z
\approx \sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, यानी बिंदु आवेश।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:6}

\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:7}
$r$ त्रिज्या और $h$ लंबाई का बेलन: $2\pi rh\,E = \lambda h/\varepsilon_0$, $E =
\lambda/2\pi\varepsilon_0r$। कोरोना तब, जब $\lambda = 2\pi\varepsilon_0 rE = 2\pi \times 8.85 \times
10^{-12} \times 0.01 \times 3 \times 10^6 = \qty{1.7}{\micro C/m}$; और \qty{5}{m} पर,
$E = 3 \times 10^6 \times 0.01/5 = \qty{6}{kV/m}$।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:8}
$E = \sigma/\varepsilon_0 = 10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$। कोई
पट्टिका अपने ऊपर कोई परिणामी बल नहीं लगाती (उसके अपने क्षेत्र के योगदान
जोड़ों में कट जाते हैं); और उसकी जगह पर दूसरी पट्टिका का क्षेत्र $\sigma/2\varepsilon_0$ है, अतः
$F = Q\sigma/2\varepsilon_0 = 10^{-6} \times 10^{-6}/(2 \times 8.85 \times 10^{-12}) =
\qty{5.6e-2}{N}$, आकर्षी; और दाब $\sigma^2/2\varepsilon_0 = \qty{0.056}{Pa}$।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:9}
$r < a$: भीतरी चालक का आवेश उसके पृष्ठ पर बैठा रहता है, भीतर कोई आवेश नहीं,
अतः $E = 0$। $a < r < b$: $E = \lambda/2\pi\varepsilon_0r$। $r > b$: घिरा हुआ
आवेश $\lambda - \lambda = 0$, अतः $E = 0$ --- यानी तार बाहर कोई स्थैतिक
क्षेत्र नहीं फैलाता। और $r = a$ पर: $2 \times 8.99 \times 10^9 \times 10^{-8}/(5 \times 10^{-4}) =
\qty{3.6e5}{V/m}$।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:10}
$\rho = 3Q/4\pi(R_2^3 - R_1^3)$। $r < R_1$: $E = 0$। $R_1 < r < R_2$: $4\pi r^2E =
\rho\,\tfrac43\pi(r^3 - R_1^3)/\varepsilon_0$, अतः $E = \dfrac{Q}{4\pi\varepsilon_0r^2}\,
\dfrac{r^3 - R_1^3}{R_2^3 - R_1^3}$। $r > R_2$: $Q/4\pi\varepsilon_0r^2$। $R_1$ पर दोनों
ओर से $0$ मिलता है; और $R_2$ पर दोनों से $Q/4\pi\varepsilon_0R_2^2$। कोश में $E
\propto r - R_1^3/r^2$ बढ़ता जाता है, अतः उच्चिष्ठ $R_2$ पर है, जिसके बाद
वह $1/r^2$ की तरह गिरता है। गुहा में क्षेत्र शून्य है: वहाँ रखे किसी आवेश
पर कोई बल नहीं लगता --- चाहे वह कहीं भी बैठे।
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:11}
(क) क्रोड $\tfrac43\pi(3.48 \times 10^6)^3 \times 11000 = \qty{1.94e24}{kg}$; प्रावार
$\tfrac43\pi[(6.37 \times 10^6)^3 - (3.48 \times 10^6)^3] \times 4500 = \qty{4.08e24}{kg}$;
कुल \qty{6.0e24}{kg}, यानी $1\%$ के भीतर। (ख) सतह $GM/R^2 = \qty{9.9}{m/s^2}$;
सीमा पर $6.67 \times 10^{-11} \times 1.94 \times 10^{24}/(3.48 \times 10^6)^2 =
\qty{10.7}{m/s^2}$। (ग) $g = GM(r)/r^2$, जहाँ $\dd M = 4\pi r^2\rho_m\,\dd r$:
$\dd g/\dd r = 4\pi G\rho_m - 2GM(r)/r^3 = 4\pi G[\rho_m - \tfrac23\bar\rho(r)]$,
जहाँ $\bar\rho(r) = M(r)/\tfrac43\pi r^3$ $r$ के नीचे का माध्य घनत्व है।
$g$ गहराई के साथ ($\dd g/\dd r < 0$) तभी बढ़ता है जब $\rho_m < \tfrac23\bar\rho$ हो। सतह
से ठीक नीचे $\bar\rho = 5500$, $\tfrac23\bar\rho = 3700 < 4500$: अतः $g$
पहले ज़रा-सा गिरता है; और क्रोड की सीमा पर $\bar\rho = 11000$, $7300 >
4500$: वहाँ वह चढ़कर $10.7$ हो जाता है।
\end{solution}

\begin{solution}{pb:b1:electrostatics-gauss:1}
\textbf{1.} $\vect F = -e\vect E$: ऊपर की ओर। सतह पर नीचे की ओर का क्षेत्र
किसी ऋणात्मक आवेश का क्षेत्र है: यानी पृथ्वी ऋणात्मक है।

\textbf{2.} $R_E$ त्रिज्या का गोला, $E_n = -100$: $Q = 4\pi\varepsilon_0R_E^2E_n =
1.11 \times 10^{-10} \times 4.06 \times 10^{13} \times (-100) = \qty{-4.5e5}{C}$।

\textbf{3.} $\sigma = \varepsilon_0E_n = \qty{-8.9e-10}{C/m^2}$; $8.9 \times 10^{-10}/1.6 \times
10^{-19} = 5.5 \times 10^9$ इलेक्ट्रॉन प्रति वर्ग मीटर, यानी $5.5 \times 10^5$ प्रति
वर्ग सेंटीमीटर।

\textbf{4.} $F = 100e \times 100 = \qty{1.6e-15}{N}$; भार $10^{-9} \times 9.8 =
\qty{9.8e-9}{N}$: यानी साठ लाख गुना बड़ा।

\textbf{5.} बाहर की ओर के अभिलंब: ऊपर $E_n = -5$, नीचे $E_n = +100$, और
पार्श्व शून्य: $\Phi = \qty{95}{V.m}$, अतः प्रति वर्ग मीटर $Q = 95\varepsilon_0 = \qty{+8.4e-10}{C}$
--- यानी धनात्मक आकाश-आवेश, जो भूमि के
$-8.9 \times 10^{-10}$ को लगभग काट देता है।

\textbf{6.} $\rho = 8.4 \times 10^{-10}/10^4 = \qty{8.4e-14}{C/m^3}$: यानी प्रति घन
मीटर $5 \times 10^5$ \omterm{def:b1:electrostatics-gauss:charge}{मूल आवेश}; और हर चिह्न के $10^9$ आयनों के सामने,
$5 \times 10^{-4}$ का असंतुलन।

\textbf{7.} $J = \gamma E_0 = \qty{2e-12}{A/m^2}$; $I = J \times 5.1 \times 10^{14} =
\qty{1.0}{kA}$; $Q/I = 4.5 \times 10^5/1000 = \qty{450}{s}$। पृथ्वी आठ मिनट में
उदासीन हो जाती: यानी कोई चीज़ उसे लगातार फिर से आवेशित करती रहती है।

\textbf{8.} अक्ष उसे रखने वाले हर तल में है, और वे सब \omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल} हैं: अतः
$\vect E$ अक्षीय है। हर $\dd q$ अपनी ही दिशा में $\dd q/4\pi\varepsilon_0(a^2 + z^2)$
देता है, जिसका अक्षीय भाग अंश
$z/\sqrt{a^2 + z^2}$ है: अतः $E_z = qz/4\pi\varepsilon_0(a^2 + z^2)^{3/2}$।

\textbf{9.} $\dd q = 2\pi\sigma a\,\dd a$: $E_z = \dfrac{\sigma z}{2\varepsilon_0}\displaystyle\int_0^R
\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}} = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2
+ R^2}}\Bigr)$।

\textbf{10.} $z \ll R$: $\sigma/2\varepsilon_0$, यानी अनंत तल (गाउस वाला मान);
$z \gg R$: $\sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, यानी बिंदु आवेश।

\textbf{11.} $\sigma = -40/(\pi \times 9 \times 10^6) = \qty{-1.4e-6}{C/m^2}$; $\sigma/2\varepsilon_0 =
\qty{8.0e4}{V/m}$; और $z = \qty{3}{km}$ पर, $1 - 3/\sqrt{18} = 0.29$: $E =
\qty{2.3e4}{V/m}$, जो ऋणात्मक आवेश की ओर \emph{ऊपर} संकेत करता है।

\textbf{12.} ऊपरी चकती, $z = \qty{9}{km}$: $8.0 \times 10^4 \times (1 - 9/\sqrt{90}) =
\qty{4.1e3}{V/m}$ नीचे की ओर। परिणामी $\qty{1.9e4}{V/m}$ ऊपर की ओर, और भूमि
के प्रेरित आवेशों के साथ लगभग $4 \times 10^4$: यानी मापे गए \qty{10}{kV/m} की
सही कोटि (असली आवेश अधिक फैला हुआ होता है और कुछ हद तक ढका भी);
और साफ़ मौसम के क्षेत्र से उलटा तथा सौ गुना।

\textbf{13.} पूरा बादल: परिणामी आवेश शून्य, अतः \omterm{def:b1:electrostatics-gauss:flux}{अभिवाह} शून्य। अकेली निचली

\textbf{24.} $m\ddot x = -mg_0x/R_E$: $\omega = \sqrt{g_0/R_E} = \qty{1.24e-3}{rad/s}$,
$T = 2\pi/\omega = \qty{5060}{s} = \qty{84}{min}$, $v_{\max} = \omega R_E =
\qty{7.9}{km/s}$ --- यानी किसी सतह-छूते उपग्रह की \omterm{thm:b1:central-forces:circular}{कक्षीय चाल}, जिसका
आवर्तकाल भी वही है।

\textbf{25.} उसने पृथ्वी के हर आवेश पर के योग की जगह ली (I), हवा के आयनों
पर के योग की (I), और चट्टान के हर कोश पर के योग की (IV) --- और
II में उसने चकती वाले योग की सीमाएँ तय कीं। याद रखिए: साफ़ मौसम की पृथ्वी
के लिए \qty{-5e5}{C} और \qty{1}{kA}; किसी तूफ़ान के नीचे \qty{40}{C} तथा
\qty{e4}{V/m}; प्रति आघात \qty{e5}{A} और \qty{e9}{J}; और क्रोड की सीमा पर
\qty{10.7}{m/s^2}।
