\begin{solution}{pb:b1:dc-circuits:1}
\textbf{1.} आदर्श \omterm{def:b1:dc-circuits:sources}{विद्युत वाहक बल} $E$, $r$ के साथ श्रेणी में: $u = E - ri$।

\textbf{2.} $u(0) = \qty{12.6}{V}$; $I_N = E/r = \qty{1260}{A}$। सिरों के
आरपार पड़ा कोई पाना उसे ढो लेता: कुछ ग्राम इस्पात में $\qty{16}{kW}$
--- वह पिघलकर छिटक जाता है।

\textbf{3.} \qty{10}{m\ohm} के समांतर में \omterm{def:b1:dc-circuits:sources}{धारा स्रोत} $I_N = \qty{1260}{A}$।

\textbf{4.} $i = E/(R + r)$; $u = ER/(R + r)$; $P_R = E^2R/(R + r)^2$;
$P_r = E^2r/(R + r)^2$.

\textbf{5.} $\dd P_R/\dd R \propto (r - R)$: अतः $R = r$ पर अधिकतम,
$P_{\max} = E^2/4r = 12.6^2/0.040 = \qty{4.0}{kW}$; और दक्षता
$R/(R + r) = 50\%$।

\textbf{6.} $12.6 - 0.010 \times 12 = \qty{12.48}{V}$; $12.6 - 0.010 \times
120 = \qty{11.40}{V}$। $u$ को $i$ के सामने आलेखित करने पर सरल रेखा मिलती है,
जिसका अंतःखंड $E$ और ढाल $-r$ है (\omterm{prop:b1:units-dimensions:regression}{न्यूनतम वर्ग}, \cref{ch:b1:units-dimensions})।

\textbf{7.} दो दीप: \qty{12}{V} पर \qty{110}{W}, यानी \qty{9.2}{A};
$60/9.2 = \qty{6.5}{h}$। ऊर्जा $60 \times 3600 \times 12 = \qty{2.6}{MJ}$।

\textbf{8.} $R_h = 12^2/55 = \qty{2.62}{\ohm}$; समांतर में \qty{1.31}{\ohm}।

\textbf{9.} $i = 12.6/0.090 = \qty{140}{A}$; $u = 12.6 - 1.4 = \qty{11.2}{V}$;
$P_s = 0.080 \times 140^2 = \qty{1.57}{kW}$; खोई $0.010 \times 140^2 =
\qty{196}{W}$।

\textbf{10.} $R = 0.080 \parallel 1.31 = \qty{75.4}{m\ohm}$; $i = 12.6/0.0854
= \qty{148}{A}$; $u = 12.6 - 1.48 = \qty{11.1}{V}$.

\textbf{11.} $i_s = i \times 1.31/(1.31 + 0.080) = \qty{139}{A}$; और दोनों दीपों के लिए $i_h =
\qty{8.5}{A}$।

\textbf{12.} हर दीप: $u^2/R_h = 11.1^2/2.62 = \qty{47}{W}$, जबकि नाममात्र
\qty{55}{W}: यानी $14\%$ की गिरावट।

\textbf{13.} \omterm{thm:b1:dc-circuits:kirchhoff}{पाश नियम}: सिरों का \omterm{def:b1:dc-circuits:voltage}{विभवांतर} $u = E - ri$ है; स्टार्टर के
\qty{140}{A} बैटरी के भीतर \qty{1.4}{V} का पात बनाते हैं, और उन्हीं सिरों पर
समांतर में जुड़े दीपों को वही घटा हुआ \omterm{def:b1:dc-circuits:voltage}{विभवांतर} मिलता है।

\textbf{14.} केवल आंशिक रूप से: मोटे तार तारों में पात घटाते हैं, $r$ में
नहीं; बैटरी के सिरों से खिलाए गए दीप फिर भी $E - ri$ ही देखते हैं।

\textbf{15.} उन्हीं दो संधियों के बीच तीन शाखाएँ: $(E, r)$,
$(E_a, r_a)$ और $R_h = \qty{1.31}{\ohm}$।

\textbf{16.} $V = \dfrac{12.6/0.010 + 14.2/0.050 + 0/1.31}{100 + 20 + 0.763}
= \dfrac{1544}{120.8} = \qty{12.79}{V}$.

\textbf{17.} प्रत्यावर्तित्र $(14.2 - 12.79)/0.050 = \qty{28.3}{A}$; बैटरी
$(12.6 - 12.79)/0.010 = \qty{-18.6}{A}$ (जो उसके \emph{भीतर} बह रही है); दीप
$12.79/1.31 = \qty{9.8}{A}$; और $28.3 = 18.6 + 9.8$।

\textbf{18.} आवेशित हो रही है। उसे $12.79 \times 18.6 = \qty{238}{W}$ मिलते
हैं (\qty{234}{W} संचित, \qty{3.5}{W} ऊष्मा); और प्रत्यावर्तित्र अपने सिरों पर
$12.79 \times 28.3 = \qty{362}{W}$ देता है।

\textbf{19.} अकेली बैटरी (प्रत्यावर्तित्र $\to$ \qty{50}{m\ohm}): भार
$0.050 \parallel 1.31 = \qty{48.2}{m\ohm}$, $u_1 = 12.6 \times 48.2/58.2 =
\qty{10.44}{V}$। अकेला प्रत्यावर्तित्र (बैटरी $\to$ \qty{10}{m\ohm}): भार
$0.010 \parallel 1.31 = \qty{9.92}{m\ohm}$, $u_2 = 14.2 \times 9.92/59.9 =
\qty{2.35}{V}$। योग \qty{12.79}{V}।

\textbf{20.} $u = 5.0\,R_s/(R_0 + R_s)$: \qty{20}{\degreeCelsius} पर
\qty{3.60}{V}, और \qty{90}{\degreeCelsius} पर \qty{1.40}{V}।

\textbf{21.} $\Delta u = 5[R_a/(R_0 + R_a) - R_b/(R_0 + R_b)]$;
$\dd\Delta u/\dd R_0 = 0 \Leftrightarrow R_a/(R_0 + R_a)^2 = R_b/(R_0 +
R_b)^2 \Leftrightarrow R_0^2 = R_aR_b$: अतः $R_0 = \sqrt{2000 \times 300} =
\qty{775}{\ohm}$।

\textbf{22.} $u = 5[\tfrac12 - R_s/(775 + R_s)]$, जो तब शून्य है जब $R_s =
\qty{775}{\ohm}$ हो, यानी उस मध्य-परास ताप पर जहाँ तापप्रतिरोधी वह मान
पकड़ लेता है।

\textbf{23.} प्रति चालक $R = 1.7\times10^{-8} \times 1.5/16\times10^{-6} =
\qty{1.6}{m\ohm}$, कुल \qty{3.2}{m\ohm}; पात $0.0032
\times 140 = \qty{0.45}{V}$; शक्ति \qty{63}{W}।

\textbf{24.} \qty{4}{mm^2}: हर एक \qty{6.4}{m\ohm}, कुल \qty{12.8}{m\ohm}:
पात \qty{1.8}{V}, \qty{250}{W} बरबाद --- स्टार्टर अपने \omterm{def:b1:dc-circuits:voltage}{विभवांतर} का
छठा भाग खो देता। मोटा तार $R_{\text{तार}} \ll R_s$ बनाए रखता है।

\textbf{25.} इंजन घुमाते समय सिरों पर लगभग \qty{11.1}{V}; स्टार्टर को बैटरी
की शक्ति का $1568/(1568 + 196) \approx 89\%$ मिलता है; और बत्तियों का मंद
पड़ना \omterm{thm:b1:dc-circuits:kirchhoff}{पाश नियम} है, $u = E - ri$।
\end{solution}
