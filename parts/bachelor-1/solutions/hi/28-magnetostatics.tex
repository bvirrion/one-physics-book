\begin{solution}{exo:b1:magnetostatics:5}

\begin{solution}{exo:b1:magnetostatics:6}

\[
f(u) = \mu_0NIR^2/2(R^2 + u^2)^{3/2} .
\]

क्षेत्र तीसरी कोटि तक एकसमान है। आँकड़े: $0.716 \times 4\pi \times 10^{-7} \times 100/0.15 =
\qty{0.60}{mT}$।
\end{solution}

\end{solution}

\begin{solution}{pb:b1:magnetostatics:1}
\textbf{1.} अक्ष को रखने वाला हर तल धाराओं का कोई \omterm{prop:b1:electrostatics-gauss:symmetry}{सममिति तल} है: अतः
$\vect B$ उस पर लंब है, यानी लंबत्रिज्य; और अक्ष के अनुदिश तथा उसके चारों ओर
की अपरिवर्तिता से: $\vect B = B(r)\,\vect e_\theta$।

\textbf{2.} $r$ त्रिज्या का वृत्त: $2\pi rB = \mu_0I_{\text{घिरी}}$। क्रोड:
$I_{\text{घिरी}} = Ir^2/a^2$, $B = \mu_0Ir/2\pi a^2$; बीच में: $\mu_0I/2\pi r$;
और बाहर ($r > c$): $I - I = 0$, $B = 0$।

\textbf{3.} $B(a) = 2 \times 10^{-7}/5 \times 10^{-4} = \qty{4.0e-4}{T}$; $B(b) = 2 \times
10^{-7}/2.5 \times 10^{-3} = \qty{8.0e-5}{T}$।

\textbf{4.} नली के भीतर कोई धारा घिरी नहीं होती: अतः वहाँ रैखिक चढ़ाई की जगह
$B = 0$; और बाकी हर जगह कुछ नहीं बदलता --- किसी बेलनाकार धारा के बाहर का
क्षेत्र केवल कुल धारा पर निर्भर करता है।

\textbf{5.} $I_{\text{घिरी}} = I - I(r^2 - b^2)/(c^2 - b^2) = I(c^2 - r^2)/(c^2 -
b^2)$: $B = \dfrac{\mu_0I}{2\pi r}\,\dfrac{c^2 - r^2}{c^2 - b^2}$, जो
$b$ पर $\mu_0I/2\pi b$ के बराबर है और $c$ पर शून्य: यानी संतत। आलेख: \qty{0.5}{mm} पर \qty{4e-4}{T}
तक रैखिक, फिर $1/r$ से \qty{2.5}{mm} पर \qty{8e-5}{T} तक, फिर \qty{3}{mm} पर
गिरकर शून्य, और उसके आगे शून्य।

\textbf{6.} आवरण की धारा क्रोड की धारा से प्रतिसमांतर है: अतः प्रतिकर्षण,
यानी बाहर की ओर कोई दाब। पृष्ठीय धारा $K = I/2\pi b = \qty{64}{A/m}$; और माध्य क्षेत्र
$\tfrac12(8 \times 10^{-5} + 0) = \qty{4e-5}{T}$: अतः $p = KB = \qty{2.5e-3}{Pa}$।
\qty{10}{kA} पर: $\times10^8$, यानी \qty{2.5e5}{Pa} = \qty{2.5}{bar} ---
स्पंदित-शक्ति के तार इसे झेलने लायक बनाए जाते हैं।

\textbf{7.} बाहर कोई क्षेत्र नहीं: अतः तार न कुछ विकीर्ण करता है, न अपने
पड़ोसियों से कोई चुंबकीय व्यवधान उठाता है। तारों का कोई जोड़ा: दोनों क्षेत्र
लगभग कट जाते हैं और \qty{10}{cm} पर $\approx \mu_0Id/2\pi D^2 = 2 \times 10^{-7} \times 2 \times
10^{-3}/10^{-2} = \qty{4e-8}{T}$ बचता है --- छोटा, पर समाक्ष तो शून्य ही
देता है।

\textbf{8.} $D = \sqrt{h^2 + d^2/4} = \qty{10.0}{m}$: हर एक $\mu_0I/2\pi D =
\qty{1.0e-5}{T}$ देता है, जो चालक से जाती रेखा पर लंब है। धाराएँ विपरीत हैं,
अतः चालकों को जोड़ती रेखा के समांतर घटक कट जाते हैं और ऊर्ध्वाधर घटक जुड़
जाते हैं: $B = 2 \times 10^{-5} \times
(d/2)/D = \qty{1.0e-6}{T}$, ऊर्ध्वाधर। यानी पृथ्वी के क्षेत्र से पचास गुना
नीचे, और संस्तुत सीमा से सौ गुना नीचे।

\textbf{9.} $D \gg d$ के लिए, $B \approx \dfrac{\mu_0I}{2\pi}\Bigl(\dfrac1{D_1} - \dfrac1{D_2}\Bigr)
\sim \dfrac{\mu_0I}{2\pi}\dfrac{d}{D^2}$: दोनों $1/D$ क्षेत्र प्रथम कोटि तक कट
जाते हैं और केवल उनका अंतर बचता है, जिसका सापेक्ष आकार $d/D$ है --- यानी कोई
``दो-तार वाला द्विध्रुव''।

\textbf{10.} अपना चालक \qty{1}{m} पर: \qty{1.0e-4}{T} क्षैतिज; और दूसरा
\qty{1.41}{m} पर: \qty{7.1e-5}{T}, \qty{45}{\degree} पर, विरोध में: अतः परिणामी
$(5 \times 10^{-5}, 5 \times 10^{-5})$, यानी \qty{7e-5}{T} --- \qty{70}{\micro T}, जो
सार्वजनिक सीमा से नीचे है और व्यावसायिक सीमा से कहीं नीचे।

\textbf{11.} $F/\ell = \mu_0I^2/2\pi d = 2 \times 10^{-7} \times 2.5 \times 10^5 =
\qty{0.05}{N/m}$, प्रतिकर्षी; और \qty{20}{kA} पर: \qty{80}{N/m}।

\textbf{12.} सामान्य में \qty{15}{N}; और लघुपथन में \qty{24}{kN}, यानी फैलाव
के \qty{2.9}{kN} भार से आठ गुना: चालक एक-दूसरे से दूर उछल जाते हैं, झूलते
हैं और लौटते समय आपस में टकरा भी सकते हैं --- इसीलिए दूरी बनाए रखने वाले
अवरोधक और कुछ ही चक्रों में खुल जाने वाले वियोजक लगाए जाते हैं।

\textbf{13.} $I\cos\omega t + I\cos(\omega t - 2\pi/3) + I\cos(\omega t + 2\pi/3) =
I\cos\omega t\,(1 + 2\cos2\pi/3) = 0$: यानी तीनों कला-सदिशों का योग शून्य है।
$1/D$ वाले क्षेत्र समाक्ष की तरह कट जाते हैं; और दूर का क्षेत्र $1/D^2$ की
तरह या उससे भी तेज़ी से गिरता है।

\textbf{14.} वहाँ लाइन का क्षेत्र ऊर्ध्वाधर है; और कोई दिक्सूचक केवल क्षैतिज
घटक पर ही प्रतिक्रिया करता है: अतः कोई विक्षेप नहीं (नति
$10^{-6}/(2 \times 10^{-5})$, यानी तीन अंश बदलती है, जो दिक्सूचक दिखा ही नहीं सकता)।

\textbf{15.} भुजाएँ $a$ (जो अक्ष के समांतर हैं): त्रिज्य $\vect B$ उन पर
लंब है, अतः हर एक पर बल $NIaB$, जो कुंडली के तल पर लंब है और दोनों भुजाओं पर
विपरीत: यानी एक युग्म। भुजाएँ $b$: वहाँ $\dd\vect l$
त्रिज्य है, यानी $\vect B$ के समांतर: अतः कोई बल नहीं।

\textbf{16.} $\Gamma = 2 \times NIaB \times b/2 = NIabB = NISB = 0.024\,I$ (न्यूटन-मीटर में, जहाँ $I$
\omterm{def:b1:dc-circuits:current}{ऐंपियर} में)। त्रिज्य क्षेत्र हमेशा कुंडली के तल में और भुजाओं $a$ पर लंब
रहता है: अतः बल हमेशा तल पर अभिलंब होते हैं और उनकी \omterm{def:b1:angular-momentum:moment}{उत्तोलक भुजा} $b/2$ कभी
नहीं बदलती।

\textbf{17.} $C\theta = NISB$: $\theta = 2400\,I$ रेडियन, अर्थात् \qty{2.4}{rad/mA}; और पूर्ण
पैमाना \qty{1.2}{rad} पर, $I = \qty{0.50}{mA}$।

\textbf{18.} लंबाई $200 \times 2(0.02 + 0.03) = \qty{20}{m}$, काट $\pi(5 \times
10^{-5})^2 = \qty{7.9e-9}{m^2}$: $R = 1.7 \times 10^{-8} \times 20/7.9 \times 10^{-9} =
\qty{43}{\ohm}$; $U = 43 \times 5 \times 10^{-4} = \qty{22}{mV}$; $P = UI = \qty{11}{\micro W}$।

\textbf{19.} $J\ddot\theta = -C\theta + NISB\,I$: $\omega_0 = \sqrt{C/J} = \sqrt{10^{-5}/6.7
\times 10^{-8}} = \qty{12}{rad/s}$, $T = \qty{0.51}{s}$।

\textbf{20.} $J\ddot\theta + \alpha\dot\theta + C\theta = NISB\,I$: क्रांतिक तब, जब
$\alpha = 2\sqrt{JC} = 2\sqrt{6.7 \times 10^{-13}} = \qty{1.6e-6}{N.m.s}$; तब सुई
बिना आगे निकले सबसे तेज़ी से अपने पाठ पर पहुँच जाती है।

\textbf{21.} वोल्टमापी: $R_s = 10/5 \times 10^{-4} - 43 \approx \qty{20}{k\ohm}$।
धारामापी: शंट \qty{22}{mV} के नीचे \qty{0.9995}{A} ले जाता है: $R_{\text{शंट}} =
\qty{22}{m\ohm}$।

\textbf{22.} $m = NIS = 200 \times 5 \times 10^{-4} \times 6 \times 10^{-4} = \qty{6e-5}{A.m^2}$;
$-mB = \qty{-1.2e-5}{J}$; और स्प्रिंग $\tfrac12C\theta^2 = \tfrac12 \times 10^{-5} \times 1.44 =
\qty{7.2e-6}{J}$: यानी वही कोटि --- स्प्रिंग वही संचित रखती है जो चुंबकीय
\omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} ने कुंडली के घूमते समय किया।

\textbf{23.} किसी एकसमान क्षेत्र में \omterm{def:b1:angular-momentum:moment}{बलाघूर्ण} $mB\cos\theta$ होता (जो तब
अधिकतम है जब तल $\vect B$ को रखता हो, और तब शून्य जब अभिलंब उसके अनुदिश हो): $C\theta =
NISB\,I\cos\theta$ --- यानी केवल छोटे $\theta$ के लिए रैखिक, और पैमाने के
ऊपरी सिरे पर दबा हुआ।

\textbf{24.} \qty{50}{Hz}, जबकि $\omega_0/2\pi = \qty{2}{Hz}$: कुंडली साथ नहीं
चल पाती; वह धारा के माध्य, यानी शून्य, पर ठहरी रहती है, बस
$\approx \theta_{\text{स्थैतिक}}(\omega_0/\omega)^2 = 1.2 \times (12/314)^2 \approx
\qty{2e-3}{rad}$ \omterm{def:b1:wave-propagation:signal}{आयाम} की एक कँपकँपी के साथ, जो दिखती ही नहीं। कोई चल-कुंडली
मापी माध्य पढ़ता है --- प्रत्यावर्ती धारा के लिए उसे कोई दिष्टकारी चाहिए।
