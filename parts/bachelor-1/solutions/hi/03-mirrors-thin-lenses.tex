\begin{solution}{exo:b1:mirrors-thin-lenses:1}

1/\overline{SA}$। \qty{60}{cm} पर: $\overline{SA'} = \qty{-30}{cm}$,
वास्तविक, $\gamma = -0.5$। \qty{30}{cm} पर: $\overline{SA'} = \qty{-60}{cm}$,

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:4}

छोटी दिखती कार \qty{0.95}{m} पर होते हुए भी दूर की कार पढ़ी जाती है --- यानी
\qty{20}{m} से भी दूर की।
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:6}
$1/\overline{OA'} = 1/50 - 1/2000 = \qty{0.0195}{mm^{-1}}$,
$\overline{OA'} = \qty{51.3}{mm}$: अतः \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} संवेदक से \qty{1.3}{mm} दूर
खिसकता है। $\gamma = 51.3/(-2000) = -0.026$। $\abs\gamma = 24/1800

\qty{-3.8}{m}$।
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$; $\overline{OA'} = \qty{5.0}{m}$,
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$;
$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$, $f' = \qty{98}{mm}$।
\omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} उलटा है ($\gamma < 0$), ऊपर--नीचे और बाएँ--दाएँ दोनों: इसलिए स्लाइड
$180^\circ$ घुमाकर डाली जाती है।
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
वस्तु \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} तल में $\Rightarrow$ हर बिंदु से चली किरणें \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} से समांतर
निकलती हैं; \omterm{prop:b1:rays-reflection-refraction:mirror}{समतल दर्पण} उन्हें समांतर ही लौटा देता है (सममित दिशा में); और \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस}
समांतर पुंज को अपने \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} तल में, स्रोत बिंदु के अक्ष-सममित बिंदु पर, केंद्रित
कर देता है: अतः \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} वस्तु के तल में, उलटा, $\gamma = -1$। दर्पण को
$\alpha$ से तिरछा करने पर लौटता पुंज $2\alpha$ घूम जाता है और \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब}
बग़ल में $2\alpha f'$ खिसक जाता है, पर उसी तल में तीखा बना रहता है।
\end{solution}

$4f' = \qty{84}{cm}$, जिस पर $\gamma = -1$।
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
$L_1$ अनंत का \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $F_1'$ पर बनाता है, यानी $L_1$ से \qty{20}{cm} आगे, यानी \qty{5}{cm}
$L_2$ से आगे: अतः आभासी वस्तु, $\overline{O_2A_1} = \qty{+5}{cm}$।
$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$: अंतिम \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} \qty{10}{cm}
$L_2$ से आगे। मध्यवर्ती \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} की ऊँचाई $h_1 = f_1'\alpha = 20\alpha$ (cm);
$\gamma_2 = 10/5 = 2$, अतः $h = 40\alpha$: यानी तुल्य \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी}
\qty{40}{cm}, और नली की लंबाई $15 + 10 = \qty{25}{cm}$ --- यही टेलीफ़ोटो है।
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ से $\overline{SA'} =
3\overline{SA}$; दोनों वास्तविक, आगे की ओर: $\overline{SA} = -x$,
$\overline{SA'} = -3x$, और परदा--दीपक दूरी $2x = \qty{80}{cm}$:
$x = \qty{40}{cm}$। दर्पण दीपक से \qty{40}{cm} और परदे से \qty{120}{cm}
पर; $1/\overline{SF} = -1/40 - 1/120 = -1/30$:
$R = 2 \times 30 = \qty{60}{cm}$।
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:12}
$\theta = 0.53^\circ = \qty{9.3e-3}{rad}$; \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} का व्यास $f'\theta =

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$,
$\gamma = \overline{OA'}/\overline{OA}$; अक्ष रोशनी की दिशा में दिष्ट, और
दूरियाँ $O$ से बीजीय।

\textbf{2.} $p = -\overline{OA}$, $\overline{OA'} = D - p$:
$1/(D - p) + 1/p = 1/f_1'$, यानी $p^2 - Dp + Df_1' = 0$; वास्तविक मूल तभी जब
$D^2 - 4Df_1' \geq 0$, यानी $D \geq 4f_1'$।

\textbf{3.} मूलों का योग $D$ है: $p_1 + p_2 = D$, अतः $p_2 = D - p_1$ ---
वस्तु और \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} की दूरियाँ आपस में बदल जाती हैं, यानी स्थितियाँ मध्यबिंदु के
परितः सममित हैं। $d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$, अतः $f_1' = (D^2 -
d^2)/4D$।

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} इसमें केवल दोनों तीखी सज्जाओं के बीच नली का खिसकाव $d$ और
वस्तु--परदा दूरी $D$ आते हैं: दोनों में से किसी के लिए यह जानना ज़रूरी नहीं कि
नली के भीतर $O$ कहाँ बैठा है।

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$,
$\partial f_1'/\partial d = -d/2D = -0.25$; अतः
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ और $u(f_1') = \qty{0.14}{cm}$:
$f_1' = \qty{15.00\pm0.14}{cm}$।

\textbf{7.} $p = (80 \mp 40)/2 = 20$ या \qty{60}{cm}: $\gamma =
-60/20 = -3$ (स्लाइड \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} के पास, बड़ा \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब}: यही प्रक्षेपक) और
$\gamma = -1/3$।

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$: \omterm{def:b1:mirrors-thin-lenses:image}{वास्तविक प्रतिबिंब} अक्ष के परितः $180^\circ$
घूमा हुआ है, अतः स्लाइड भी $180^\circ$ घुमाकर डाली जाती है (उलटी
और दर्पण-उलट)।

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$: $1/\overline{OA} = 1/1000
- 1/15$, $\overline{OA} = \qty{-15.23}{cm}$: अतः \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस} स्लाइड की ओर
\qty{0.7}{mm} खिसकता है। $\gamma = 1000/(-15.23) = -65.7$: चित्र
\qty{2.36}{m} चौड़ा।

\textbf{13.} देकार्त का अवकलन करने पर: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$, अतः $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$।

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$: फ़िल्म गरम होते ही
\omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} से बाहर सरक जाता है --- इसीलिए काँच में जड़ी स्लाइडें, या
फ़िल्म का पीछा करता स्वतःफोकसन।

\textbf{15.} द्वारक से \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} बिंदु तक समरूप त्रिभुजों से:
धुँधलापन $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$; \qty{5}{m} से यह
$\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$ का कोण घेरता है: पकड़ में नहीं आता।

\textbf{16.} स्लाइड का क्षेत्रफल $8.64\times10^{-4}\ \unit{m^2}$: $E_{\text{स्लाइड}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$; परदे का क्षेत्रफल $\gamma^2$
गुना बड़ा, यानी $\qty{2.16}{m^2}$: $E_{\text{परदा}} = \qty{0.93}{W/m^2}$;
शक्ति वही, और क्षेत्रफल $\gamma^2 = 2500$ के अनुपात में।

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$, $\overline{O_2A} = \qty{+29.3}{cm}$: यानी
मध्यवर्ती \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $L_2$ से \qty{29.3}{cm} \emph{आगे} पड़ता है --- अतः वह
$L_2$ के लिए आभासी वस्तु है ($L_1$ से आती किरणें उसी की ओर अभिसरित होती हैं)।

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$;
$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$, $\overline{O_1A} =
\qty{-25.1}{cm}$।

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$; $\gamma_2 = 1200/29.3 =
41.0$; $\gamma = -60.9$.

\textbf{20.} चौड़ाई $60.9 \times 36 = \qty{2.19}{m}$। अकेला $L_1$:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$, चौड़ाई
\qty{2.86}{m}।

\textbf{21.} $\alpha$ पर आता पुंज: $L_1$ उसे अपने \omterm{def:b1:mirrors-thin-lenses:mirror}{फोकस} तल में ऊँचाई $h_1 = f_1'\alpha$
पर केंद्रित करता है, और वह तल $\overline{O_2A_1} = f_1' - e$ दूरी पर है, $L_2$ से नापी हुई;
$L_2$ इस बिंदु का \omterm{def:b1:mirrors-thin-lenses:image}{प्रतिबिंब} $\overline{O_2A'}$ पर बनाता है, जहाँ $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$, और आवर्धन $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$ देता है, अतः $h = \gamma_2 f_1'\alpha$ और $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$, जिसका व्युत्क्रम वही दिया गया
सूत्र है। $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$,
$f_{\mathrm{eq}} = \qty{19.6}{cm}$।

\textbf{22.} $2.19/2.86 = 0.77$ और $15/19.6 = 0.77$। केवल लगभग इसलिए
कि स्लाइड परिमित दूरी पर है और दोनों प्रक्षेप भिन्न लेंसों से नापे गए हैं;
अनंत पर रखी वस्तु के लिए यह अनुपात यथार्थ होता।

\textbf{23.} किसी \omterm{def:b1:mirrors-thin-lenses:lens}{अपसारी लेंस} के आगे $p > 0$ पर आभासी वस्तु:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$, जो तभी धनात्मक
(यानी \omterm{def:b1:mirrors-thin-lenses:image}{वास्तविक प्रतिबिंब}) है जब $p < \abs{f_2'}$ हो।

\textbf{24.} वही शक्ति, पर चित्र का क्षेत्रफल $(2.19/2.86)^2 = 0.59$
गुना छोटा: अतः विकिरण-तीव्रता $\times 1.7$।

\textbf{25.} परिवर्तक \omterm{def:b1:mirrors-thin-lenses:lens}{फोकस दूरी} को
$f_{\mathrm{eq}}/f_1' = 1.3$ गुना कर देता है; किसी दिए गए प्रक्षेप पर चित्र
हर दिशा में $1.3$ गुना छोटा और $1.7$ गुना अधिक चमकीला हो जाता है --- यानी
दीर्घ-प्रक्षेप \omterm{def:b1:mirrors-thin-lenses:lens}{लेंस}।
\end{solution}
