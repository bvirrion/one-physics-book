\section*{अध्याय \ref{ch:b1:rays-reflection-refraction} --- ज्यामितीय प्रकाशिकी: किरणें, परावर्तन, अपवर्तन}

\begin{solution}{exo:b1:rays-reflection-refraction:1}
$v = c/n$: पानी \qty{2.26e8}{m/s}, काँच \qty{2.00e8}{m/s}, हीरा
\qty{1.24e8}{m/s}। तंतु: $t = nL/c = 1.48 \times 10^3/3.00\times10^8 =
\qty{4.9}{\micro s}$।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:2}
$\sin i_2 = \sin i_1/1.33$: $30^\circ \to 22.1^\circ$; $60^\circ \to
40.6^\circ$। पानी से $50^\circ$ पर: $1.33\sin 50^\circ = 1.02 > 1$,
कोई अपवर्तित किरण नहीं --- पूर्ण परावर्तन ($i_c = 48.8^\circ$)।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:3}
$i_c = \arcsin(1/n)$: काँच $41.8^\circ$, हीरा $24.4^\circ$, पानी
$48.8^\circ$। आकाश की चकती $2 \times 48.8^\circ = 97.5^\circ$ फैली रहती है।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:4}
आँखें \qty{1.60}{m} पर। पैरों से चली किरण आँख की आधी ऊँचाई,
\qty{0.80}{m}, पर परावर्तित होकर आँख तक पहुँचती है; सिर के ऊपरी सिरे से चली
किरण सिरे और आँखों के ठीक बीच, \qty{1.65}{m} पर परावर्तित होती है। अतः दर्पण
\qty{0.80}{m} से \qty{1.65}{m} तक: ऊँचाई \qty{0.85}{m}, यानी शरीर की आधी
ऊँचाई --- और यह दूरी से स्वतंत्र है (मध्यबिंदु का नियम हर दूरी पर लागू रहता
है)।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:5}
छोटे कोणों के लिए: विरल माध्यम से देखने पर आभासी स्थिति $h' = h/n$ होती है,
और सघन माध्यम से देखने पर $h' = nh$। सिक्का: $1.2/1.33 =
\qty{0.90}{m}$। मछुआरा: $1.33 \times 1.8 = \qty{2.4}{m}$। औचित्य:
क्षैतिज दूरी $x$ पर स्थित सतह-बिंदु के लिए $\tan i_2 \approx i_2 =
x/h$ और $\tan i_1 \approx i_1 = x/h'$, जहाँ $i_1 = n i_2$।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:6}
$\sin\frac{A + D_{\min}}{2} = 1.52\sin 30^\circ = 0.760$, अतः
$\frac{A + D_{\min}}{2} = 49.5^\circ$, $D_{\min} = 38.9^\circ$।
निर्गम के लिए $r' \leq i_c$ चाहिए; और $\sin r = \sin i/n \leq 1/n$ से
$r \leq i_c$ मिलता है; अतः $A = r + r' \leq 2i_c$। यहाँ $i_c = \arcsin(1/1.52) =
41.1^\circ$, $2i_c = 82.3^\circ > 60^\circ$: ठीक है।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:7}
$\mathrm{NA} = \sqrt{1.50^2 - 1.48^2} = \sqrt{0.0596} = 0.244$,
$\theta_a = 14.1^\circ$। $\Delta t/L = (1.50/c)(1.50/1.48 - 1) =
5.0\times10^{-9} \times 0.0135 = \qty{68}{ns/km}$। \qty{1}{km} पर:
$B_{\max} \approx 1/(2 \times \qty{68}{ns}) = \qty{7.4}{Mbit/s}$।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:8}
$D \approx (n - 1)A$: लाल $0.510 \times 10^\circ = 5.10^\circ$, नीला
$5.30^\circ$; वर्णक्रम की चौड़ाई $0.20^\circ$।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:9}
प्रवेश: $\sin r = \sin 45^\circ/1.5 = 0.471$, $r = 28.1^\circ$। दोनों
फलक समांतर हैं, अतः दूसरे फलक पर \omterm{def:b1:rays-reflection-refraction:incidence}{आपतन कोण} $r$ है और
$\sin i' = n\sin r = \sin i$: यानी $i' = i$, और निकलती किरण आपतित किरण के
समांतर है। भीतर की $e/\cos r$ लंबाई वाली राह पर लंबवत खिसकाव
$(e/\cos r)\sin(i - r)$ होता है:
$d = 2.0 \times \sin 16.9^\circ/\cos 28.1^\circ = \qty{0.66}{cm}$।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:10}
दो अपवर्तन ($2(i - r)$) और दो परावर्तन ($2(\pi - 2r)$):
$D = 2\pi + 2i - 6r$। $\dd D/\dd i = 2 - 6\cos i/(n\cos r) = 0$ से
$3\cos i = n\cos r$, $9\cos^2 i = n^2 - 1 + \cos^2 i$, $\cos^2 i =
(n^2 - 1)/8$। $n = 1.333$: $\cos^2 i = 0.0971$, $i = 71.8^\circ$,
$\sin r = 0.950/1.333$, $r = 45.5^\circ$, $D = 360^\circ + 143.7^\circ -
272.8^\circ = 230.9^\circ$: अतः धनुष सूर्य-विपरीत बिंदु से $230.9^\circ - 180^\circ =
51^\circ$ पर है, प्राथमिक ($42^\circ$) के बाहर।
लाल ($n = 1.331$) $50.4^\circ$ पर, बैंगनी ($1.343$) $53.6^\circ$ पर:
रंग उलटे हैं, बैंगनी बाहर।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:11}
$\sin i_c = 1.00026/1.00029 = 0.99997$, $i_c = 89.56^\circ$: सड़क की सतह से
$0.44^\circ = \qty{7.7}{mrad}$ के भीतर आती किरणें पूरी तरह
परावर्तित हो जाती हैं (सामान्यतः स्पर्श कोण $\approx \sqrt{2\Delta n/n}$)।
\qty{1.2}{m} की ऊँचाई से दृष्टि-रेखा इस कोण पर
$1.2/0.0077 \approx \qty{160}{m}$ के परे स्पर्श करती है: वहाँ सड़क आकाश को
परावर्तित करती है --- वही ``पोखर''।
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:12}
$t(x) = \dfrac{\sqrt{a^2 + x^2}}{v_1} + \dfrac{\sqrt{b^2 + (d - x)^2}}{v_2}$;
$t'(x) = \dfrac{x}{v_1\sqrt{a^2 + x^2}} - \dfrac{d - x}{v_2\sqrt{b^2 + (d - x)^2}}
= \dfrac{\sin i_1}{v_1} - \dfrac{\sin i_2}{v_2}$, जो ठीक तभी शून्य है जब
$\sin i_1/v_1 = \sin i_2/v_2$ हो, यानी $n_1\sin i_1 = n_2\sin i_2$, जहाँ
$n = c/v$।
\end{solution}

\begin{solution}{pb:b1:rays-reflection-refraction:1}
\textbf{1.} $n = c/v$; $v_1 = 3.00\times10^8/1.480 = \qty{2.03e8}{m/s}$.

\textbf{2.} $t = Ln_1/c = 5.0\times10^4 \times 1.48/3.00\times10^8 =
\qty{247}{\micro s}$.

\textbf{3.} $\nu = c/\lambda_0 = \qty{1.94e14}{Hz}$, जो काँच में अपरिवर्तित रहती है;
$\lambda = \lambda_0/n_1 = \qty{1047}{nm}$।

\textbf{4.} नंगा \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} तभी तक निर्देशन करता है जब तक उसका पृष्ठ पूरी तरह
साफ़ और अछूता रहे: धूल, पानी या पकड़ स्थानीय रूप से हवा की जगह अधिक \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक}
ले आते हैं और प्रकाश बाहर निकल जाता है, तथा खरोंचें उसे प्रकीर्ण कर देती हैं।
\omterm{def:b1:rays-reflection-refraction:fiber}{आवरण} एक बंद, नियंत्रित अंतरापृष्ठ है, जिसकी अपवर्तनांक-छलाँग जानबूझकर चुनी
जाती है।

\textbf{5.} $\sin i_c = 1.465/1.480 = 0.9899$, $i_c = 81.8^\circ$.

\textbf{6.} $90^\circ - 81.8^\circ = 8.2^\circ$.

\textbf{7.} $\sin\theta = n_1\sin\theta'$; निर्देशित तभी जब $\cos\theta' \geq
n_2/n_1$, यानी $\sin\theta' \leq \sqrt{1 - n_2^2/n_1^2}$, यानी
$\sin\theta \leq \sqrt{n_1^2 - n_2^2} = \sqrt{2.1904 - 2.1462} = 0.210$:
$\theta_a = 12.1^\circ$।

\textbf{8.} $\mathrm{NA} = 0.210$। बड़ी छलाँग शंकु को चौड़ा कर देती है, अतः
किसी अपसारी स्रोत (जैसे LED) का अधिक प्रकाश पकड़ में आ जाता है।

\textbf{9.} किरण त्रिज्या $R_b$ के वृत्त (भीतरी किनारा) की स्पर्शरेखा है
और त्रिज्या $R_b + 2a$ के वृत्त (बाहरी किनारा) से मिलती है; केंद्र--स्पर्श
बिंदु--टक्कर बिंदु वाले समकोण त्रिभुज में, टक्कर बिंदु पर किरण और त्रिज्या
(यानी अभिलंब) के बीच का कोण
$\sin i = R_b/(R_b + 2a)$ का पालन करता है। निर्देशित तभी जब $\sin i \geq n_2/n_1$:
$R_b \geq 2a\,\dfrac{n_2/n_1}{1 - n_2/n_1} = \qty{50}{\micro m} \times 97.6
= \qty{4.9}{mm}$। इससे तीखे मोड़ प्रकाश रिसा देते हैं।

\textbf{10.} $t_0 = Ln_1/c$; सबसे तिरछी किरण का पथ $L/\cos\theta'_{\max} =
Ln_1/n_2$, अतः $t_1 = Ln_1^2/(n_2 c)$।

\textbf{11.} $\Delta t/L = (n_1/c)(n_1/n_2 - 1) = 4.93\times10^{-9} \times
0.01024 = \qty{50.5}{ns/km}$; \qty{2}{km} पर: \qty{101}{ns}।

\textbf{12.} $B_{\max} \approx 1/(2\Delta t) = 1/\qty{202}{ns} \approx
\qty{5}{Mbit/s}$.

\textbf{13.} $B_{\max} L = 1/(2\,\Delta t/L) = 1/(2 \times
\qty{50.5}{ns/km}) \approx \qty{10}{Mbit.s^{-1}.km}$.

\textbf{14.} अक्ष से दूर चलती किरणें लंबा पथ तय करती हैं, पर कम \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक}
में, इसलिए तेज़ चलती हैं; ठीक चुने गए परिच्छेद के साथ दोनों प्रभाव एक-दूसरे को
काट देते हैं और सारी किरणें साथ-साथ पहुँचती हैं।

\textbf{15.} $\qty{9}{\micro m} \approx 6\lambda_0$: \omterm{def:b1:rays-reflection-refraction:fiber}{क्रोड} तरंगदैर्घ्य की
तुलना में बड़ा नहीं है, इसलिए विवर्तन हावी हो जाता है और किरणों का कोई अर्थ
नहीं रह जाता। तरंग-विवेचन एक ही निर्देशित प्रतिरूप दिखाता है: अंतरमोड प्रसार
बिलकुल नहीं; केवल वर्णिक विक्षेपण और क्षीणन बचे रहते हैं।

\textbf{16.} \qty{100}{km} $\times$ \qty{0.2}{dB/km} $= \qty{20}{dB}$:
$P/P_0 = 10^{-2}$, यानी एक प्रतिशत।

\textbf{17.} $10\log(P_0/P_{\min}) = \qty{30}{dB}$, अतः $L_{\max} =
30/0.20 = \qty{150}{km}$।

\textbf{18.} \qty{100}{km} में \qty{20}{dB} लगते हैं: अवकाश \qty{10}{dB},
यानी शक्ति में $10$ का गुणक।

\textbf{19.} LED: $\Delta t_c = 17 \times 40 \times 100 = \qty{68000}{ps}
= \qty{68}{ns}$, $B \approx 1/\qty{136}{ns} = \qty{7}{Mbit/s}$। लेज़र:
$\Delta t_c = \qty{170}{ps}$, $B \approx 1/\qty{340}{ps} =
\qty{2.9}{Gbit/s}$।

\textbf{20.} $30/2.0 = \qty{15}{km}$, यानी दस गुना कम। सिलिका
\qty{1550}{nm} के पास सबसे अधिक पारदर्शी है (उसका क्षीणन-न्यूनतम वहीं है), और
प्रकाशिक प्रवर्धक भी वहीं उपलब्ध हैं।

\textbf{21.} प्रवर्धकों के बीच $30/30 = \qty{1}{km}$, जबकि तंतु में
\qty{150}{km}: तंतु $150$ के गुणक से जीतता है।

\textbf{22.} अंतरमोड: \qty{101}{ns} ($\to$ \qty{5}{Mbit/s});
वर्णिक (लेज़र): $17 \times 0.1 \times 2 = \qty{3.4}{ps}$;
क्षीणन \qty{0.4}{dB}, नगण्य। अतः अंतरमोड प्रसार ही सीमित करता है:
\qty{5}{Mbit/s}।

\textbf{23.} अंतरमोड प्रसार नहीं; वर्णिक \qty{170}{ps} $\to$
\qty{2.9}{Gbit/s}; क्षीणन \qty{20}{dB}, बजट के भीतर। अतः वर्णिक
विक्षेपण सीमित करता है, लगभग \qty{3}{Gbit/s} पर।

\textbf{24.} $80 \times 2.9 \approx \qty{230}{Gbit/s}$.

\textbf{25.} सोपान \omterm{def:b1:rays-reflection-refraction:index}{अपवर्तनांक}: \qty{10}{Mbit.s^{-1}.km}; एकल-मोड चैनल:
$2.9 \times 100 = \qty{290}{Gbit.s^{-1}.km}$ --- यानी लगभग
$3\times10^4$ का गुणक। एक ही तथ्य: इकलौता निर्देशित पथ, अतः पहुँचने के समयों
का कोई ज्यामितीय (अंतरमोड) प्रसार नहीं।
\end{solution}
