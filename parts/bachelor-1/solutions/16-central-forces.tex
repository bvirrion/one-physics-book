\begin{solution}{exo:b1:central-forces:3}

\begin{solution}{exo:b1:central-forces:5}
$\Delta E = \tfrac12 mv^2 + GMm(1/R - 1/r) = 2.94\times10^{10} + 3.99\times
10^{17} \times (1.570 - 1.477)\times10^{-7} = 2.94\times10^{10} + 0.37\times
10^{10} = \qty{3.3e10}{J}$ --- \qty{33}{MJ/kg}, the energy of \qty{0.7}{kg}
of gasoline per kilogram placed in orbit (rockets need far more, having
to lift their own fuel).
\end{solution}

\begin{solution}{exo:b1:central-forces:6}
$a = (0.586 + 35.1)/2 = \qty{17.8}{AU}$; $e = (35.1 - 0.586)/35.7 = 0.967$;
$T = 17.8^{3/2} = \qty{75}{yr}$. $v_p^2 = GM(2/r_p - 1/a)$ with $r_p =
8.77\times10^{10}$ m, $a = 2.67\times10^{12}$ m: $v_p = \qty{54}{km/s}$;
$v_a = v_pr_p/r_a = \qty{0.91}{km/s}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:7}
$a = \qty{24490}{km}$. $v_1 = \qty{7.67}{km/s}$; on the ellipse at $r_1$:
$\sqrt{GM(2/r_1 - 1/a)} = \qty{10.07}{km/s}$: $\Delta v_1 = \qty{2.4}{km/s}$.
At $r_2$: $\sqrt{GM(2/r_2 - 1/a)} = \qty{1.62}{km/s}$, circular
\qty{3.07}{km/s}: $\Delta v_2 = \qty{1.45}{km/s}$. Total \qty{3.85}{km/s};
duration $\pi\sqrt{a^3/GM} = \qty{1.9e4}{s} = \qty{5.3}{h}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:8}
$E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ at $r_c = L^2/GMm^2$.
$E_{p,\mathrm{eff}}'' = -2GMm/r^3 + 3L^2/mr^4$; at $r_c$, $L^2 = GMm^2r_c$
gives $E'' = GMm/r_c^3$, so $\omega_r^2 = E''/m = GM/r_c^3 = \omega_{\text{orb}}^2$.
A small radial disturbance oscillates exactly once per revolution: the
orbit closes on itself --- a slightly eccentric ellipse, not a rosette.
\end{solution}

\begin{solution}{exo:b1:central-forces:9}
Only gravity acts on station and crew alike; both fall on the same
orbit, so no contact force is needed between them --- no floor pushes,
hence ``weightlessness''. $g = GM/r^2 = 3.99\times10^{14}/(6.77\times10^6)^2
= \qty{8.7}{m/s^2}$: gravity is almost full strength; weight is not
absent, it is entirely spent bending the path.
\end{solution}

\begin{solution}{exo:b1:central-forces:10}
$a = 4\pi^2r/T^2 = 39.5 \times 3.84\times10^8/(2.36\times10^6)^2 =
\qty{2.7e-3}{m/s^2}$; $g(R/r)^2 = 9.81/60.3^2 = \qty{2.7e-3}{m/s^2}$: the
same force law that drops an apple holds the Moon, diluted as $1/r^2$.
\end{solution}

\begin{solution}{exo:b1:central-forces:11}
$E_k = 2Ze^2/4\pi\varepsilon_0r_{\min}$: $r_{\min} = 8.99\times10^9 \times 2 \times 79
\times (1.60\times10^{-19})^2/8.0\times10^{-13} = \qty{4.5e-14}{m} = \qty{45}{fm}$,
six times the nuclear radius: the $\alpha$ turns back before touching.
To reach \qty{7}{fm}: $E_k \approx \qty{32}{MeV}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:12}
$E_m = -GMm/2r$ decreases $\Rightarrow$ $r$ decreases $\Rightarrow$ $v =
\sqrt{GM/r}$ increases. The \omterm{thm:b1:work-and-energy:ke}{kinetic energy} $GMm/2r$ rises by $\abs{\Delta E}$
while the \omterm{def:b1:work-and-energy:conservative}{potential energy} $-GMm/r$ falls by $2\abs{\Delta E}$: half of the
\omterm{def:b1:work-and-energy:conservative}{potential energy} released becomes kinetic, half is dissipated as heat
by the drag.
\end{solution}

\begin{solution}{pb:b1:central-forces:1}
\textbf{1.} $T = \qty{43082}{s}$: $r = (GMT^2/4\pi^2)^{1/3} = (3.986\times10^{14}
\times 1.856\times10^9/39.48)^{1/3} = \qty{2.656e7}{m}$; altitude
\qty{20190}{km}.

\textbf{2.} $v = \sqrt{GM/r} = \qty{3.87}{km/s}$.

\textbf{3.} $a = v^2/r = GM/r^2 = \qty{0.565}{m/s^2} = g(R/r)^2$, seventeen
times weaker than at the surface --- but not zero: the satellite falls
continuously around the Earth.

\textbf{4.} $E_m/m = -GM/2r = \qty{-7.5}{MJ/kg}$; $E_k/m = \qty{+7.5}{MJ/kg}$,
$E_p/m = \qty{-15.0}{MJ/kg}$.

\textbf{5.} The Earth turns once per sidereal day relative to the stars
(the frame of the orbit); after two orbits the satellite is back over
the same ground point: the ground tracks repeat every sidereal day.

\textbf{6.} $\sin\beta = R/r = 0.240$: $\beta = 13.9^\circ$, a disk
$28^\circ$ wide.

\textbf{7.} Geostationary \qty{42200}{km}: $1.59$ times higher; the
navigation orbit is below it.

\textbf{8.} $GM(1/R - 1/2r) = 3.986\times10^{14}(1.570\times10^{-7} - 1.88\times
10^{-8}) = \qty{5.5e7}{J/kg} = \qty{55}{MJ/kg}$.

\textbf{9.} $r_1 = \qty{6771}{km}$: $v = \qty{7.67}{km/s}$.

\textbf{10.} $a = \qty{16665}{km}$; at $r_1$: $\sqrt{GM(2/r_1 - 1/a)} =
\qty{9.69}{km/s}$, $\Delta v_1 = \qty{2.0}{km/s}$; at $r_2$: \qty{2.47}{km/s}
against circular \qty{3.87}{km/s}, $\Delta v_2 = \qty{1.4}{km/s}$; total
\qty{3.4}{km/s}.

\textbf{11.} $\pi\sqrt{a^3/GM} = \qty{1.07e4}{s} \approx \qty{3.0}{h}$.

\textbf{12.} Escape from low orbit: $(\sqrt2 - 1)\times 7.67 = \qty{3.2}{km/s}$:
the mission's \qty{3.4}{km/s} exceeds it --- reaching this orbit costs
about as much as leaving the Earth altogether.

\textbf{13.} $m_0/m_1 = \eu^{3.4/3.0} = 3.1$: $m_1 = \qty{640}{kg}$,
propellant \qty{1360}{kg} --- two thirds of the mass.

\textbf{14.} $\cos\theta = 0.240$, $\theta = 76^\circ$; fraction $(1 - 0.240)/2
= 0.38$.

\textbf{15.} $24 \times 0.38 = 9$ on average; four is comfortably
exceeded almost always.

\textbf{16.} Zenith: $2.02\times10^7/c = \qty{67}{ms}$; horizon: distance
$\sqrt{r^2 - R^2} = \qty{2.58e7}{m}$, \qty{86}{ms}.

\textbf{17.} $3/c = \qty{10}{ns}$.

\textbf{18.} The receiver's clock offset is a fourth unknown besides its
three coordinates: four distances, four equations.

\textbf{19.} $10^{-8}\ \unit{s}/\qty{86400}{s} \approx \num{1e-13}$.

\textbf{20.} $v^2/2c^2 = (3874)^2/(2 \times 8.99\times10^{16}) = \num{8.3e-11}$:
$8.3\times10^{-11} \times 86400 = \qty{7.2}{\micro s}$ per day, slow.

\textbf{21.} $GM(1/R - 1/r)/c^2 = 3.986\times10^{14} \times 1.19\times10^{-7}/
8.99\times10^{16} = \num{5.3e-10}$: \qty{45.7}{\micro s} per day, fast.

\textbf{22.} Net $+\qty{38.5}{\micro s}$ per day; $c \times 38.5\times10^{-6}
= \qty{11.5}{km}$ of error per day.

\textbf{23.} $(10.23 - 10.22999999543)/10.23 = \num{4.47e-10}$, matching
the net $4.5\times10^{-10}$ (\qty{38.5}{\micro s} over \qty{86400}{s}).

\textbf{24.} The Earth's equatorial bulge, the Moon's and Sun's
attraction, solar radiation pressure (and small thruster firings): the
ground segment measures the orbits continuously and uploads fresh
ephemerides, which the satellites broadcast.

\textbf{25.} Energy and \omterm{def:b1:angular-momentum:L}{angular momentum} fix a \omterm{prop:b1:central-forces:effective}{circular orbit} from its
period; Kepler III places it; geometry fixes coverage and light times;
and the clocks --- the heart of the system --- must be corrected for
relativity by \qty{38}{\micro s} a day, or the fix drifts by ten
kilometers by nightfall.
\end{solution}
