\section*{Chapter \ref{ch:b1:sinusoidal-impedance} --- Sinusoidal Steady State and Impedance}

\begin{solution}{exo:b1:sinusoidal-impedance:1}
$\underline U = 5\eu^{j\pi/3} = 2.5 + 4.33j$ (V). $3\cos\omega t +
4\sin\omega t$: $3 + 4\eu^{-j\pi/2} = 3 - 4j$, modulus $5$, argument
$-\arctan(4/3) = \qty{-0.927}{rad}$: $5\cos(\omega t - 0.927)$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:6}
$\underline Z = 100 + j(250 - 100) = 100 + 150j$: $Z = \qty{180}{\ohm}$,
$\arg = 56^\circ$. $I = 1.0/180 = \qty{5.6}{mA}$, lagging the source by
$56^\circ$. $U_R = \qty{0.56}{V}$, $U_L = \qty{1.39}{V}$, $U_C = \qty{0.56}{V}$;
\omterm{def:b1:sinusoidal-impedance:complex}{phasors}: $U_R$ along $I$, $U_L$ up, $U_C$ down, resultant \qty{1.0}{V}.
$L\omega > 1/C\omega$: above resonance (inductive).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:7}
$f_0 = 1/(2\pi\sqrt{LC})$: \qty{1.59}{MHz} (\qty{50}{pF}) to \qty{0.50}{MHz}
(\qty{500}{pF}). At \qty{1.00}{MHz}: $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$.
$\Delta f = \qty{10}{kHz}$. Station at \qty{1.009}{MHz}: $x = 1.009$,
$Q(x - 1/x) = 1.8$, ratio $1/\sqrt{1 + 3.2} = 0.49$. Capacitor \omterm{def:b1:dc-circuits:voltage}{voltage}
$QE = \qty{1.0}{mV}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:8}
$Q(x - 1/x) = \pm1 \Leftrightarrow x^2 \mp x/Q - 1 = 0$; positive roots
$x_{1,2} = \mp 1/2Q + \sqrt{1 + 1/4Q^2}$. Difference $1/Q$, so
$\Delta\omega = \omega_0/Q$; product $(1 + 1/4Q^2) - 1/4Q^2 = 1$, so
$\omega_1\omega_2 = \omega_0^2$ (the band is symmetric on a log scale).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:9}
$I = P/(U\cos\varphi) = 2000/(230 \times 0.70) = \qty{12.4}{A}$. \omterm{thm:b1:sinusoidal-impedance:power}{Reactive
power} $P\tan\varphi = 2000 \times 1.02 = \qty{2.0}{kvar}$. $C = 2040/(314
\times 230^2) = \qty{123}{\micro F}$. New current $2000/230 = \qty{8.7}{A}$.
Losses $0.50 \times 12.4^2 = \qty{77}{W}$ before, $0.50 \times 8.7^2 =
\qty{38}{W}$ after.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:10}
$\underline Y = 1/R + j(C\omega - 1/L\omega)$; $\underline U = \underline I/
\underline Y$ is maximal when the susceptance vanishes, at $\omega_0$,
where $U = RI$. Half-power: $\abs{C\omega - 1/L\omega} = 1/R$, i.e.\
$RC\omega_0(x - 1/x) = \pm1$: $\Delta\omega = \omega_0/(RC\omega_0) = 1/RC$
and $Q = RC\omega_0 = R\sqrt{C/L}$. Numbers: $\omega_0 = \qty{1.0e6}{rad/s}$
($f_0 = \qty{159}{kHz}$), $Q = 10^4\sqrt{10^{-6}} = 10$, $\Delta f =
\qty{16}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:11}
$U_C = I/C\omega = E/[C\omega\sqrt{R^2 + (L\omega - 1/C\omega)^2}]$;
multiply inside by $C\omega$: $\sqrt{(RC\omega)^2 + (LC\omega^2 - 1)^2} =
\sqrt{x^2/Q^2 + (1 - x^2)^2}$. With $s = x^2$: $(1 - s)^2 + s/Q^2$ is
minimal at $s = 1 - 1/2Q^2$ (if positive), minimum $1/Q^2 - 1/4Q^4$:
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$. $Q = 3.16$: $x_r = 0.975$,
$U_{C,\max} = 3.2E$. For $Q < 1/\sqrt2$ the minimum is at $s = 0$: $U_C$
decreases monotonically from $E$, no peak.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:12}
At resonance $i = I\cos\omega_0 t$ and $u_C = (I/C\omega_0)\sin\omega_0 t$,
so, using $1/C\omega_0^2 = L$,
\[
\tfrac12 Li^2 + \tfrac12 Cu_C^2 = \tfrac12 LI^2\cos^2\omega_0 t
+ \tfrac12 LI^2\sin^2\omega_0 t = \tfrac12 LI^2 .
\]
Dissipated per period: $\tfrac12 RI^2T$. Ratio $\times 2\pi$: $2\pi L/(RT)
= L\omega_0/R = Q$.
\end{solution}

\begin{solution}{pb:b1:sinusoidal-impedance:1}
\textbf{1.} $\underline Z = R + j(L\omega - 1/C\omega)$.

\textbf{2.} $\underline I = E/\underline Z$; $I = E/\sqrt{R^2 + (L\omega -
1/C\omega)^2}$; $\varphi_i = -\arctan[(L\omega - 1/C\omega)/R]$.

\textbf{3.} The \omterm{def:b1:sinusoidal-impedance:impedance}{reactance} vanishes at $\omega_0 = 1/\sqrt{LC}$: $I_{\max}
= E/R$, $\varphi_i = 0$.

\textbf{4.} $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$; $L\omega_0 = 2\times10^{-4}
\times 6.28\times10^6 = \qty{1257}{\ohm}$.

\textbf{5.} $Q = 1257/12.6 = 100$.

\textbf{6.} $L\omega - 1/C\omega = L\omega_0(x - 1/x)$, divide by $R$.

\textbf{7.} $x < 1$: \omterm{def:b1:sinusoidal-impedance:impedance}{reactance} negative, $\varphi_i > 0$, the current
leads (capacitive); $x > 1$: lags (inductive).

\textbf{8.} $Q(x - 1/x) = \pm1$.

\textbf{9.} $x^2 \mp x/Q - 1 = 0$, $x_{1,2} = \mp1/2Q + \sqrt{1 + 1/4Q^2}$,
$x_2 - x_1 = 1/Q$: $\Delta\omega = \omega_0/Q$.

\textbf{10.} $\Delta f = \qty{10}{kHz}$.

\textbf{11.} \qty{1.010}{MHz}: $Q(x - 1/x) = 100 \times 0.0199 = 2.0$,
ratio $1/\sqrt{5.0} = 0.45$, power $0.20$: \qty{-7}{dB}. \qty{1.020}{MHz}:
$3.96$, ratio $0.25$, power $0.06$: \qty{-12}{dB}.

\textbf{12.} Marginal: the neighbor is only \qty{7}{dB} down. A higher
$Q$ (lower $R$), or several tuned stages in cascade, sharpens the
selection.

\textbf{13.} \qty{0.50}{MHz} to \qty{1.59}{MHz}: the AM band.

\textbf{14.} $\underline U_C = \underline I/jC\omega$; at $\omega_0$,
$U_C = I_{\max}/C\omega_0 = E/(RC\omega_0) = QE$.

\textbf{15.} $I_{\max} = 10^{-5}/12.6 = \qty{0.79}{\micro A}$; $U_C = QE =
\qty{1.0}{mV}$: a hundredfold \omterm{def:b1:dc-circuits:voltage}{voltage} gain with no amplifier.

\textbf{16.} $\underline U_L = jL\omega_0\underline I = jQE$, $\underline U_C
= -jQE$: equal and opposite, they cancel; the source sees $R$ alone.

\textbf{17.} Stored $\tfrac12 LI_{\max}^2 = \qty{6.2e-17}{J}$; dissipated
per period $\tfrac12 RI_{\max}^2T = \qty{3.9e-18}{J}$; $2\pi \times 6.2/0.39
\approx 100 = Q$.

\textbf{18.} $\tau = 2Q/\omega_0 = \qty{32}{\micro s}$. The circuit cannot
follow changes faster than $\sim\tau$: a bandwidth $\Delta f$ passes
only modulations slower than $\Delta f$; with $Q = 1000$, $\Delta f =
\qty{1}{kHz}$ and the treble of the program would be lost.

\textbf{19.} $Q$ halves to $50$, $\Delta f$ doubles to \qty{20}{kHz}, $U_C$
halves to \qty{0.5}{mV}.

\textbf{20.} $R_{\text{tot}} = \qty{62.6}{\ohm}$: $Q = 1257/62.6 = 20$,
$\Delta f = \qty{50}{kHz}$, $U_C = 20 \times \qty{10}{\micro V} = \qty{0.2}{mV}$:
sensitivity and selectivity both collapse.

\textbf{21.} $P = E^2/(2R_{\text{tot}}) = 10^{-10}/125 = \qty{8e-13}{W}$;
fraction in $R_a$: $50/62.6 = 80\%$.

\textbf{22.} Matching wants a load resistance equal to $R_a$, i.e.\ a
large total resistance; selectivity wants the smallest possible total
resistance ($Q = L\omega_0/R_{\text{tot}}$). One cannot have both with a
direct connection.

\textbf{23.} A tap or a small coupling capacitor presents the antenna
with only a fraction of the resonant circuit: the antenna's resistance
is ``seen'' by the circuit much reduced, keeping $Q$ high, while the
antenna still delivers power near its optimum.

\textbf{24.} At the half-power frequencies the \omterm{def:b1:sinusoidal-impedance:impedance}{reactance} equals $\pm R$,
so $\varphi = \pm45^\circ$ and $\cos\varphi = 1/\sqrt2$; with $I =
I_{\max}/\sqrt2$ too, $P = \tfrac12 EI\cos\varphi$ is half its resonant
value.

\textbf{25.} $Q$ multiplies the \omterm{def:b1:wave-propagation:signal}{signal} by $Q$ and narrows the band to
$f_0/Q$, at the cost of a slower response ($\tau = 2Q/\omega_0$) and a
fragile coupling to the antenna; for \qty{10}{kHz} channels at
\qty{1}{MHz}, $Q \approx 100$ is the number --- and the antenna must be
coupled lightly to keep it.
\end{solution}
