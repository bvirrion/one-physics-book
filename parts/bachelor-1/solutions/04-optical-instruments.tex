\section*{Chapter \ref{ch:b1:optical-instruments} --- Optical Instruments and the Eye}

\begin{solution}{exo:b1:optical-instruments:1}
Infinity: $f' = \qty{17}{mm}$, $V = 1/0.017 = \qty{58.8}{\delta}$.
\qty{25}{cm}: $V = 58.8 + 1/0.25 = \qty{62.8}{\delta}$, $f' = \qty{15.9}{mm}$.
Amplitude \qty{4.0}{\delta}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:2}
Myope: infinity must be imaged at the \omterm{def:b1:optical-instruments:punctum}{far point}, $f' = \qty{-40}{cm}$,
$V = \qty{-2.5}{\delta}$. Hyperope: an object at \qty{25}{cm} must be
imaged (virtually) at \qty{1.0}{m}: $V = 1/\overline{OA'} - 1/\overline{OA}
= -1 + 4 = \qty{+3.0}{\delta}$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:3}
$G = 25/4.0 = 6.3$; stamp in the focal plane, \qty{4.0}{cm} from the
\omterm{def:b1:mirrors-thin-lenses:lens}{lens}. Detail of \qty{2}{mm}: $\theta' = 2/40 = \qty{0.05}{rad}$, against
$2/250 = \qty{0.008}{rad}$ at \qty{25}{cm}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:4}
$G = 900/25 = 36$ and $900/9 = 100$; tube lengths $f_1' + f_2' =
\qty{925}{mm}$ and \qty{909}{mm}; Moon: $36 \times 0.52^\circ = 19^\circ$
and $52^\circ$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:5}
$\gamma_1 = -160/4 = -40$; $G_2 = 250/20 = 12.5$; $G = 500$.
$\overline{F_1A} = -f_1'^2/\Delta = -16/160 = \qty{-0.10}{mm}$: object
\qty{4.1}{mm} from the \omterm{def:b1:optical-instruments:microscope}{objective}. The object must stay at that distance
from the \omterm{def:b1:optical-instruments:microscope}{objective}, so the whole tube moves by \qty{10}{\micro m} ---
the fine-focus knob.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:6}
Naked \omterm{def:b1:optical-instruments:eye}{eye}: $1.5/3\times10^{-4} = \qty{5}{km}$. Telescope: $\theta_{\min}
= 1.22 \times 5.5\times10^{-7}/0.100 = \qty{6.7e-6}{rad}$; headlights
separated up to $1.5/6.7\times10^{-6} \approx \qty{220}{km}$ (in
principle --- air and curvature intervene first). $G_r = 3\times10^{-4}/
6.7\times10^{-6} = 45$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:7}
$D = 50/2 = \qty{25}{mm}$. Exposure $\propto N^2$: $(8/2)^2 = 16$ times
longer, $16/500 \approx 1/30$~\unit{s}. It buys \omterm{def:b1:optical-instruments:camera}{depth of field} (and
fewer aberrations) at the price of motion blur or a tripod.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:8}
$H = 35^2/(11 \times 0.03) = \qty{3.7}{m}$; sharp from $H/2 =
\qty{1.9}{m}$ to infinity. Preset at $H$, the \omterm{def:b1:optical-instruments:camera}{camera} needs no focusing:
the picture can be taken the instant the scene happens.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:9}
$8\times30$: \omterm{prop:b1:optical-instruments:telescope}{exit pupil} $30/8 = \qty{3.8}{mm} < \qty{7}{mm}$, all light
enters the \omterm{def:b1:optical-instruments:eye}{eye}. $7\times50$: \qty{7.1}{mm}, matching the night pupil,
and $(50/30)^2 = 2.8$ times more light collected: the dusk pair.
$20\times30$: \omterm{prop:b1:optical-instruments:telescope}{exit pupil} \qty{1.5}{mm}, dim image, narrow field and
hand shake magnified twenty times.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:10}
Contact \omterm{def:b1:mirrors-thin-lenses:lens}{lens}: $f' = \qty{-25}{cm}$, $V = \qty{-4.0}{\delta}$. Spectacles:
the \omterm{def:b1:optical-instruments:punctum}{far point} is $25 - 1.5 = \qty{23.5}{cm}$ from the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, so $f' =
\qty{-23.5}{cm}$, $V = \qty{-4.3}{\delta}$ --- stronger. The \omterm{def:b1:mirrors-thin-lenses:lens}{lens} must
form the image at the \omterm{def:b1:optical-instruments:eye}{eye}'s \omterm{def:b1:optical-instruments:punctum}{far point}, whose distance from the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}
depends on where the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} sits.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:11}
Afocal: $F_1' = F_2$, separation $f_1' + f_2' = \qty{20}{cm}$ (shorter
than the \qty{40}{cm} of a Keplerian of the same $G$). $G = -30/(-10) =
+3$: upright. \omterm{prop:b1:optical-instruments:telescope}{Exit pupil} $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1' =
-10 \times 20/30 = \qty{-6.7}{cm}$: virtual, inside the tube, where the
\omterm{def:b1:optical-instruments:eye}{eye} cannot be; the \omterm{def:b1:optical-instruments:eye}{eye} behind the \omterm{def:b1:optical-instruments:microscope}{eyepiece} catches only the beams that
happen to reach it --- a narrow field, the classic complaint about opera
glasses.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:12}
$(200/7)^2 = 820$; $\Delta m = 2.5\log_{10}820 = 7.3$: limiting
magnitude $13.3$. Flux $\propto 1/r^2$, so the lamp is seen
$\sqrt{820} = 29$ times farther.
\end{solution}

\begin{solution}{pb:b1:optical-instruments:1}
\textbf{1.} $F_2 = F_1'$: separation $f_1' + f_2' = \qty{1225}{mm}$
(\qty{25}{mm} \omterm{def:b1:optical-instruments:microscope}{eyepiece}) or \qty{1206}{mm} (\qty{6}{mm}).

\textbf{2.} $L_1$: $\overline{O_1A_1} = f_1'$ (image at $F_1'$). $L_2$:
$\overline{O_2A_1} = -f_2'$, so $1/\overline{O_2A'} = 1/f_2' + 1/(-f_2')
= 0$: image at infinity.

\textbf{3.} The beam focuses in the common focal plane at height $h =
f_1'\alpha$ (ray through $O_1$); from there the ray through $O_2$ sets
the emerging direction, $\alpha' = -h/f_2' = -(f_1'/f_2')\alpha$.

\textbf{4.} $G = -48$ and $-200$; Moon: $25^\circ$ and $104^\circ$.

\textbf{5.} Up and down have no meaning in the sky; a spotting scope
adds an erecting prism pair (or \omterm{def:b1:mirrors-thin-lenses:lens}{lens} relay).

\textbf{6.} In the focal plane of $L_1$, diameter $f_1'\theta = 1200 \times
9.1\times10^{-3} = \qty{11}{mm}$.

\textbf{7.} Object $O_1$ at $\overline{O_2O_1} = -(f_1' + f_2')$:
$1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2') = f_1'/[f_2'(f_1' + f_2')]$;
magnification $\overline{O_2P'}/\overline{O_2O_1} = -f_2'/f_1' = 1/G$,
diameter $D/\abs G$.

\textbf{8.} \qty{25}{mm}: $\overline{O_2P'} = 25 \times 1225/1200 =
\qty{25.5}{mm}$, diameter $150/48 = \qty{3.1}{mm}$. \qty{6}{mm}:
\qty{6.0}{mm} behind, diameter \qty{0.75}{mm}.

\textbf{9.} Every ray that entered the \omterm{def:b1:optical-instruments:microscope}{objective} passes through the \omterm{prop:b1:optical-instruments:telescope}{exit
pupil}: an \omterm{def:b1:optical-instruments:eye}{eye} placed there receives the whole field. Farther back it
intercepts only the central beams: the field shrinks to a keyhole.

\textbf{10.} $D/\abs G \leq \qty{7}{mm}$: $\abs G \geq 150/7 \approx 21$.

\textbf{11.} $\theta_{\min} = 1.22 \times 5.5\times10^{-7}/0.150 =
\qty{4.5e-6}{rad} = 0.92''$.

\textbf{12.} $4.5\times10^{-6} \times 3.84\times10^5 = \qty{1.7}{km}$;
naked \omterm{def:b1:optical-instruments:eye}{eye} $3\times10^{-4} \times 3.84\times10^5 = \qty{115}{km}$.

\textbf{13.} $G_r = 3\times10^{-4}/4.5\times10^{-6} = 67$; the \qty{25}{mm}
\omterm{def:b1:optical-instruments:microscope}{eyepiece} ($48$) is closest, slightly under; the \qty{6}{mm} ($200$) is
three times over.

\textbf{14.} At $G = 200$ the diffraction blur subtends $200 \times 4.5
\times10^{-6} = \qty{9e-4}{rad} > \epsilon$: visibly fuzzy. The Moon's
light is spread over an area $(200/48)^2 = 17$ times larger on the
\omterm{def:b1:optical-instruments:eye}{retina}: dimmer.

\textbf{15.} $1'' = \qty{4.85e-6}{rad}$: $D = 1.22\lambda/\theta =
\qty{140}{mm}$. Beyond that, seeing limits resolution from the ground
(barring adaptive optics); a larger $D$ still collects more light,
reaching fainter objects.

\textbf{16.} $(150/7)^2 = 460$.

\textbf{17.} $\Delta m = 2.5\log_{10}460 = 6.7$: magnitude $12.7$.

\textbf{18.} $P = 2\times10^{-3} \times \pi(0.075)^2 = \qty{3.5e-5}{W}$,
$460$ times the naked \omterm{def:b1:optical-instruments:eye}{eye}'s. The retinal image is $\abs G^2$ times
larger in area: surface brightness ratio $460/G^2 = 0.20$ ($G = 48$),
$0.012$ ($G = 200$) --- equal to $(D/\abs G)^2/d_p^2$, the squared ratio
of \omterm{prop:b1:optical-instruments:telescope}{exit pupil} to \omterm{def:b1:optical-instruments:eye}{eye} pupil.

\textbf{19.} For an extended object the collected light grows as $D^2$
but the image area as $G^2 \geq (D/d_p)^2$: the surface brightness can
at best equal the \omterm{def:b1:optical-instruments:eye}{naked-eye} value (\omterm{prop:b1:optical-instruments:telescope}{exit pupil} $=$ \omterm{def:b1:optical-instruments:eye}{eye} pupil). A star is
unresolved: all its light lands in one diffraction spot, so its
brightness grows as $(D/d_p)^2$.

\textbf{20.} A point at angle $\beta$ images at height $f_1'\beta$ in
the focal plane; the stop passes $\abs{f_1'\beta} \leq \phi/2$: full
field $\phi/f_1'$ (small angles).

\textbf{21.} $20/1200 = \qty{0.0167}{rad} = 0.95^\circ$: the whole Moon
fits. $5/1200 = 0.24^\circ$: it does not.

\textbf{22.} Apparent field $\phi/f_2'$: $20/25 = \qty{0.8}{rad} = 46^\circ$;
$5/6 = 48^\circ$.

\textbf{23.} $0.95^\circ = 3420''$: $3420/15 = \qty{230}{s}$, nearly four
minutes; $0.24^\circ = 864''$: \qty{58}{s}.

\textbf{24.} \qty{25}{mm} \omterm{def:b1:optical-instruments:microscope}{eyepiece} ($G = 48$, near $G_r$, bright, whole
Moon) for the disk; \qty{6}{mm} ($G = 200$) for a closer look at craters,
dimmer and fuzzier. The instrument's number: craters down to
\qty{1.7}{km}.

\textbf{25.} Same $f_1'$ and $D$: same $G$, fields, \omterm{prop:b1:optical-instruments:telescope}{exit pupils},
resolution and light (minus any central obstruction). Gains: no
chromatic aberration. Practical change: the focus lies in front of the
mirror, so a small secondary mirror must fold the beam out to the
\omterm{def:b1:optical-instruments:microscope}{eyepiece}.
\end{solution}
