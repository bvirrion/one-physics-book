\begin{solution}{exo:b1:mirrors-thin-lenses:1}
$\overline{SF} = \qty{-20}{cm}$; $1/\overline{SA'} = 1/\overline{SF} -
1/\overline{SA}$. At \qty{60}{cm}: $\overline{SA'} = \qty{-30}{cm}$,
real, $\gamma = -0.5$. At \qty{30}{cm}: $\overline{SA'} = \qty{-60}{cm}$,
real, $\gamma = -2$. At \qty{10}{cm}: $1/\overline{SA'} = -1/20 + 1/10
= 1/20$, $\overline{SA'} = \qty{+20}{cm}$ behind the mirror, virtual,
$\gamma = +2$ (upright, enlarged).
\end{solution}

$\overline{OA} = \qty{-3.0}{cm}$: $1/\overline{OA'} = 1/5 - 1/3 = -2/15$,

\begin{solution}{exo:b1:mirrors-thin-lenses:4}
$\overline{FA} = \qty{-15}{cm}$: $\overline{F'A'} = -f'^2/\overline{FA}
= +100/15 = \qty{6.7}{cm}$ past $F'$, i.e.\ $\overline{OA'} = \qty{16.7}{cm}$;
$\gamma = f'/\overline{FA} = -0.67$. Descartes: $\overline{OA} =
\qty{-25}{cm}$, $1/\overline{OA'} = 1/10 - 1/25 = 3/50$,
$\overline{OA'} = \qty{16.7}{cm}$, $\gamma = 16.7/(-25) = -0.67$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:5}
$\overline{SF} = \qty{+1.0}{m}$, $\overline{SA} = \qty{-20}{m}$:
$1/\overline{SA'} = 1 + 1/20$, $\overline{SA'} = \qty{0.95}{m}$ behind
the mirror (virtual, upright); $\gamma = -0.95/(-20) = +0.048$, \omterm{def:b1:mirrors-thin-lenses:image}{image}
height \qty{7.1}{cm}. The brain judges distance from apparent size: a
car shown twenty times smaller at \qty{0.95}{m} is read as a car far
away --- farther than \qty{20}{m}.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:6}
$1/\overline{OA'} = 1/50 - 1/2000 = \qty{0.0195}{mm^{-1}}$,
$\overline{OA'} = \qty{51.3}{mm}$: the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} moves \qty{1.3}{mm} away from
the sensor. $\gamma = 51.3/(-2000) = -0.026$. For $\abs\gamma = 24/1800
= 1/75$: $\overline{OA} = f'(1/\gamma - 1) = 50 \times (-76) =
\qty{-3.8}{m}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$; $\overline{OA'} = \qty{5.0}{m}$,
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$;
$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$, $f' = \qty{98}{mm}$. The
\omterm{def:b1:mirrors-thin-lenses:image}{image} is inverted ($\gamma < 0$), top--bottom and left--right: the slide
goes in rotated by $180^\circ$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
Object in the focal plane $\Rightarrow$ rays from each point leave the
\omterm{def:b1:mirrors-thin-lenses:lens}{lens} parallel; the \omterm{prop:b1:rays-reflection-refraction:mirror}{plane mirror} sends them back parallel (symmetric
direction); the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} focuses a parallel beam in its focal plane, at the
point symmetric about the axis of the source point: \omterm{def:b1:mirrors-thin-lenses:image}{image} in the object
plane, inverted, $\gamma = -1$. Tilting the mirror by $\alpha$ turns the
returning beam by $2\alpha$ and shifts the \omterm{def:b1:mirrors-thin-lenses:image}{image} sideways by
$2\alpha f'$, still sharp in the same plane.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:9}
$f' = (100^2 - 40^2)/(4 \times 100) = \qty{21}{cm}$. Shortest distance
$4f' = \qty{84}{cm}$, with $\gamma = -1$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
$L_1$ \omterm{def:b1:mirrors-thin-lenses:image}{images} infinity at $F_1'$, \qty{20}{cm} past $L_1$, i.e.\ \qty{5}{cm}
past $L_2$: a virtual object, $\overline{O_2A_1} = \qty{+5}{cm}$.
$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$: final \omterm{def:b1:mirrors-thin-lenses:image}{image} \qty{10}{cm}
past $L_2$. Intermediate \omterm{def:b1:mirrors-thin-lenses:image}{image} height $h_1 = f_1'\alpha = 20\alpha$ (cm);
$\gamma_2 = 10/5 = 2$, so $h = 40\alpha$: equivalent \omterm{def:b1:mirrors-thin-lenses:lens}{focal length}
\qty{40}{cm}, in a barrel $15 + 10 = \qty{25}{cm}$ long --- a telephoto.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ gives $\overline{SA'} =
3\overline{SA}$; both real, in front: $\overline{SA} = -x$,
$\overline{SA'} = -3x$, and screen--lamp distance $2x = \qty{80}{cm}$:
$x = \qty{40}{cm}$. Mirror \qty{40}{cm} from the lamp, \qty{120}{cm}
from the screen; $1/\overline{SF} = -1/40 - 1/120 = -1/30$:

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:12}
$\theta = 0.53^\circ = \qty{9.3e-3}{rad}$; \omterm{def:b1:mirrors-thin-lenses:image}{image} diameter $f'\theta =
\qty{4.6}{mm}$. Collected power $10^3 \times \pi(0.05)^2 = \qty{7.9}{W}$;
\omterm{def:b1:mirrors-thin-lenses:image}{image} area $\pi(2.3\times10^{-3})^2 = \qty{1.7e-5}{m^2}$; irradiance

\end{solution}

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$,
$\gamma = \overline{OA'}/\overline{OA}$; axis oriented along the light,
distances algebraic from $O$.

\textbf{2.} $p = -\overline{OA}$, $\overline{OA'} = D - p$:
$1/(D - p) + 1/p = 1/f_1'$, i.e.\ $p^2 - Dp + Df_1' = 0$; real roots iff
$D^2 - 4Df_1' \geq 0$, i.e.\ $D \geq 4f_1'$.

\textbf{3.} The roots sum to $D$: $p_1 + p_2 = D$, so $p_2 = D - p_1$ ---
object and \omterm{def:b1:mirrors-thin-lenses:image}{image} distances swap, the positions are symmetric about the
midpoint. $d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$, hence $f_1' = (D^2 -
d^2)/4D$.

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} Only the displacement $d$ of the barrel between the two
sharp settings and the object--screen distance $D$ enter: neither
requires knowing where $O$ sits inside the barrel.

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$,
$\partial f_1'/\partial d = -d/2D = -0.25$; so
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ and $u(f_1') = \qty{0.14}{cm}$:
$f_1' = \qty{15.00\pm0.14}{cm}$.

\textbf{7.} $p = (80 \mp 40)/2 = 20$ or \qty{60}{cm}: $\gamma =
-60/20 = -3$ (slide near the \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, enlarged: the projector) and
$\gamma = -1/3$.

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$: the \omterm{def:b1:mirrors-thin-lenses:image}{real image} is rotated by $180^\circ$
about the axis, so the slide is inserted rotated by $180^\circ$ (upside
down and mirror-reversed).

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$: $1/\overline{OA} = 1/1000
- 1/15$, $\overline{OA} = \qty{-15.23}{cm}$: the \omterm{def:b1:mirrors-thin-lenses:lens}{lens} moves \qty{0.7}{mm}
toward the slide. $\gamma = 1000/(-15.23) = -65.7$: picture
\qty{2.36}{m} wide.

\textbf{13.} Differentiating Descartes: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$, so $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$.

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$: the \omterm{def:b1:mirrors-thin-lenses:image}{image}
drifts out of \omterm{def:b1:mirrors-thin-lenses:mirror}{focus} as the film warms --- hence glass-mounted slides or
an autofocus that tracks the film.

\textbf{15.} Similar triangles from the aperture to the \omterm{def:b1:mirrors-thin-lenses:image}{image} point:
blur $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$; from \qty{5}{m} this
subtends $\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$: not noticed.

\textbf{16.} Slide area $8.64\times10^{-4}\ \unit{m^2}$: $E_{\text{slide}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$; screen area $\gamma^2$
times larger, $\qty{2.16}{m^2}$: $E_{\text{screen}} = \qty{0.93}{W/m^2}$;
same power, areas in ratio $\gamma^2 = 2500$.

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$, $\overline{O_2A} = \qty{+29.3}{cm}$: the
intermediate \omterm{def:b1:mirrors-thin-lenses:image}{image} lies \qty{29.3}{cm} \emph{beyond} $L_2$ --- a virtual
object for $L_2$ (the rays from $L_1$ converge toward it).

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$;
$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$, $\overline{O_1A} =
\qty{-25.1}{cm}$.

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$; $\gamma_2 = 1200/29.3 =
41.0$; $\gamma = -60.9$.

\textbf{20.} Width $60.9 \times 36 = \qty{2.19}{m}$. $L_1$ alone:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$, width
\qty{2.86}{m}.

\textbf{21.} Beam at $\alpha$: $L_1$ focuses it at height $h_1 = f_1'\alpha$
in its focal plane, which is $\overline{O_2A_1} = f_1' - e$ from $L_2$;
$L_2$ \omterm{def:b1:mirrors-thin-lenses:image}{images} this point at $\overline{O_2A'}$ with $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$ and magnifies by $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$, so $h = \gamma_2 f_1'\alpha$ and $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$, whose inverse is the stated
formula. $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$,
$f_{\mathrm{eq}} = \qty{19.6}{cm}$.

\textbf{22.} $2.19/2.86 = 0.77$ and $15/19.6 = 0.77$. Only approximately
because the slide is at finite distance and the two throws are measured
from different lenses; for an object at infinity the ratio would be
exact.

\textbf{23.} Virtual object at $p > 0$ beyond a \omterm{def:b1:mirrors-thin-lenses:lens}{diverging lens}:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$, positive (\omterm{def:b1:mirrors-thin-lenses:image}{real
image}) iff $p < \abs{f_2'}$.

\textbf{24.} Same power on a picture of area $(2.19/2.86)^2 = 0.59$
times smaller: irradiance $\times 1.7$.

\textbf{25.} The converter multiplies the \omterm{def:b1:mirrors-thin-lenses:lens}{focal length} by
$f_{\mathrm{eq}}/f_1' = 1.3$; at a given throw the picture is $1.3$
times smaller in each direction and $1.7$ times brighter --- a
long-throw \omterm{def:b1:mirrors-thin-lenses:lens}{lens}.
\end{solution}
