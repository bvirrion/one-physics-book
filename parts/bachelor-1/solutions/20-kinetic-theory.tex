\begin{solution}{exo:b1:kinetic-theory:1}
$N = PV/k_BT = 1.013\times10^5 \times 10^{-3}/(1.38\times10^{-23} \times 273) =
\num{2.7e22}$; $n^{*-1/3} = (3.7\times10^{-26})^{1/3} = \qty{3.3}{nm}$, ten

\end{solution}

\begin{solution}{exo:b1:kinetic-theory:2}

\begin{solution}{exo:b1:kinetic-theory:3}

\times 293) = \qty{4.3}{mol}$, $\qty{125}{g}$. At \qty{323}{K}, $P \propto T$:
\qty{3.86}{\omterm{def:b1:kinetic-theory:pressure}{bar}} (gauge \qty{2.85}{\omterm{def:b1:kinetic-theory:pressure}{bar}}).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:4}
$n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$; $\lambda = 1/(\sqrt2\pi \times 1.37\times
10^{-19} \times 2.4\times10^{25}) = \qty{6.8e-8}{m}$; frequency $u/\lambda \approx
500/7\times10^{-8} = \qty{7e9}{s^{-1}}$. $\lambda \propto 1/P$: \qty{1}{m} at
$P \approx 10^5 \times 7\times10^{-8} = \qty{7e-3}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:5}
$\rho = PM/RT$: \qty{1.20}{kg/m^3} at \qty{293}{K}, \qty{0.946}{kg/m^3} at
\qty{373}{K}. Lift $(1.20 - 0.95) \times 2800 \times 9.81 = \qty{7.0}{kN}$:
about \qty{720}{kg}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:6}
$P_{\mathrm{O}_2} = 0.21P$: \qty{0.21}{atm}; \qty{0.11}{atm} at \qty{5000}{m};
\qty{0.07}{atm} on Everest --- a third of sea level. Oxygen enters the
blood in proportion to its partial \omterm{def:b1:kinetic-theory:pressure}{pressure}; at a third, the lungs
cannot saturate the hemoglobin.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:7}
\omterm{def:b1:kinetic-theory:model}{Perfect gas}: $P = RT/V = 2494/10^{-3} = \qty{24.9}{bar}$. Van der Waals:
$RT/(V - b) - a/V^2 = 2494/9.57\times10^{-4} - 0.364/10^{-6} = 26.1 - 3.6 =
\qty{22.4}{bar}$: the attraction ($-3.6$ \omterm{def:b1:kinetic-theory:pressure}{bar}) outweighs the excluded
volume ($+1.2$ \omterm{def:b1:kinetic-theory:pressure}{bar}); real CO$_2$ is $10\%$ below the ideal value.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:8}
$n = PV/RT = 10^5 \times 50/(8.314 \times 300) = \qty{2000}{mol}$; $U = \tfrac52
nRT = \qty{1.25e7}{J}$; $\Delta U = \tfrac52 nR \times 5 = \qty{208}{kJ}$;
\qty{104}{s} at \qty{2}{kW}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:9}
$\Delta L = \alpha L\Delta T = 1.2\times10^{-5} \times 30 \times 50 = \qty{18}{mm}$.
Ring: $\Delta d/d = \alpha\Delta T = 0.05/49.95 = 10^{-3}$: $\Delta T = 10^{-3}/
1.9\times10^{-5} = \qty{53}{K}$ (the hole expands like the metal around it).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:10}
$\tfrac32 k_BT = \qty{6.2e-21}{J} = \qty{0.039}{eV}$. Argon $\tfrac32 R =
\qty{12.5}{J/(mol.K)}$; nitrogen $\tfrac52 R = \qty{20.8}{J/(mol.K)}$: the
diatomic molecule also stores energy in rotation (two extra degrees of
freedom), whatever its mass.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:11}
Six-direction model: as in the text, $P = \tfrac13 n^*mu^2$. Isotropy:
$\langle v_x^2\rangle = \langle v_y^2\rangle = \langle v_z^2\rangle$ and their
sum is $\langle v^2\rangle$, so each is $\tfrac13\langle v^2\rangle$. The flux
argument with a distribution gives $P = n^*m\langle v_x^2\rangle$ (molecules
with $v_x > 0$ hitting the wall, momentum $2mv_x$ each, flux $n^*v_x/2$ per
speed class, averaged), i.e.\ $\tfrac13 n^*m\langle v^2\rangle$ --- the same.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:12}
H$_2$: \qty{1.9}{km/s}, O$_2$: \qty{0.48}{km/s}; Earth \qty{11.2}{km/s}, Moon
\qty{2.4}{km/s}. For O$_2$ on Earth, $v_e/u = 23$: the tail of the
distribution beyond that is utterly negligible; for H$_2$, $v_e/u \approx 6$:
a minute but steady fraction escapes each year, and over billions of
years the hydrogen is gone. On the Moon, $v_e/u$ is $5$ for O$_2$ and
$1.2$ for H$_2$: everything leaked away.
\end{solution}

\begin{solution}{pb:b1:kinetic-theory:1}
\textbf{1.} $V = \qty{200}{m^3}$: $n = PV/RT = 1.013\times10^5 \times 200/(8.314
\times 293) = \qty{8320}{mol}$; $N = nN_A = \num{5.0e27}$.

\textbf{2.} $m = nM = \qty{241}{kg}$: three people.

\textbf{3.} $n^* = N/V = \qty{2.5e25}{m^{-3}}$; distance $n^{*-1/3} = \qty{3.4}{nm}$,
nine molecular diameters.

\textbf{4.} N$_2$: $\sqrt{3 \times 8.314 \times 293/0.028} = \qty{511}{m/s}$;
O$_2$: \qty{478}{m/s}. Same $\tfrac32 k_BT$ per molecule: $\tfrac12 mu^2$ fixed,
so $u \propto 1/\sqrt m$.

\textbf{5.} $\tfrac32 k_BT = \qty{6.1e-21}{J}$; total $N \times 6.1\times10^{-21}
= \qty{3.0e7}{J}$; a \qty{1000}{kg} car at \qty{100}{km/h}: \qty{3.9e5}{J} ---
the room's molecular motion is eighty cars' worth.

\textbf{6.} $\lambda = 1/(\sqrt2\pi \times 1.37\times10^{-19} \times 2.5\times10^{25})
= \qty{66}{nm}$; time $\lambda/u = 66\times10^{-9}/500 = \qty{1.3e-10}{s}$.

\textbf{7.} Impacts per second on $S$: $\tfrac16 n^*Su = \tfrac16 \times 2.5\times
10^{25} \times 10^{-4} \times 500 = \qty{2.1e23}{s^{-1}}$.

\textbf{8.} Each gives $2mu = 2 \times 4.8\times10^{-26} \times 500 = \qty{4.8e-23}{kg.m/s}$
($m = M/N_A = \qty{4.8e-26}{kg}$): force $2.1\times10^{23} \times 4.8\times10^{-23}
= \qty{10}{N}$ on \qty{1}{cm^2}: \qty{1.0e5}{Pa}. The model reproduces
atmospheric \omterm{def:b1:kinetic-theory:pressure}{pressure}.

\textbf{9.} $F = PS = 1.013\times10^5 \times 20 = \qty{2.0e6}{N}$ --- two hundred
tonnes; the same \omterm{def:b1:kinetic-theory:pressure}{pressure} acts on the other side.

\textbf{10.} $P' = P \times 303/293 = \qty{1.048e5}{Pa}$; net $\Delta P\,S = 3.5
\times10^3 \times 20 = \qty{70}{kN}$ outward: seven tonnes on the wall.

\textbf{11.} Air flows out until the \omterm{def:b1:kinetic-theory:pressure}{pressure} equalizes: $n \propto 1/T$ at
fixed $P$, $V$: $8320 \times (1 - 293/303) = \qty{275}{mol}$, \qty{8}{kg}
leave.

\textbf{12.} $\rho = PM/RT$: \qty{1.205}{kg/m^3} at \qty{293}{K}; \qty{0.946}{kg/m^3}
at \qty{373}{K}.

\textbf{13.} Hot air $0.946 \times 2800 \times 9.81 = \qty{26.0}{kN}$; displaced
cold air $1.205 \times 2800 \times 9.81 = \qty{33.1}{kN}$; lift \qty{7.1}{kN}
(\qty{725}{kg}).

\textbf{14.} $725 - 300 = \qty{425}{kg}$: five passengers.

\textbf{15.} Lift $\propto \rho_c - \rho_h = \rho_c(1 - 293/T)$: doubling
$1 - 293/373 = 0.215$ needs $T = 293/0.57 = \qty{514}{K}$ (\qty{240}{\degreeCelsius}),
well above what nylon tolerates (about \qty{120}{\degreeCelsius}).

\textbf{16.} Helium $\rho = 1.205 \times 4/29 = \qty{0.166}{kg/m^3}$: lift
$(1.205 - 0.166) \times 2800 \times 9.81 = \qty{28.5}{kN}$ (\qty{2.9}{t}), four
times more --- but helium is expensive and lost at every landing, while
hot air is free and its lift can be adjusted with the burner.

\textbf{17.} $0.21 \times 1.013\times10^5 = \qty{2.1e4}{Pa}$.

\textbf{18.} $n^*_{\mathrm{O}_2} = P_{\mathrm{O}_2}/k_BT$: room $2.1\times10^4/
(1.38\times10^{-23} \times 293) = \qty{5.3e24}{m^{-3}}$; Everest $0.21 \times 0.33
\times 1.013\times10^5/(1.38\times10^{-23} \times 250) = \qty{2.0e24}{m^{-3}}$:
$38\%$ of the room's.

\textbf{19.} $H = RT/Mg = 8.314 \times 260/(0.029 \times 9.81) = \qty{7.6}{km}$;
$P_0/2$ at $H\ln 2 = \qty{5.3}{km}$.

\textbf{20.} $c_s = \sqrt{1.4 \times 8.314 \times 293/0.029} = \qty{343}{m/s}$,
against $u = \qty{503}{m/s}$ for the mixture: $c_s/u = \sqrt{\gamma/3} = 0.68$.
Sound is a disturbance carried by the molecules themselves, so it
travels at a speed of the order of theirs --- a little less, since
only the component along the propagation matters.

\textbf{21.} $c_s \propto \sqrt T$: $343\sqrt{303/293} = \qty{349}{m/s}$.
Helium: $\sqrt{(5/3) \times 8.314 \times 293/0.004} = \qty{1010}{m/s}$: the
resonances of the vocal tract scale with $c_s$, so every formant is
three times higher.

\textbf{22.} Colder: $343\sqrt{250/293} = \qty{317}{m/s}$, $8\%$ slower;
the \omterm{def:b1:kinetic-theory:pressure}{pressure} does not enter.

\textbf{23.} Same $N$ (same $P$, $V$, $T$); mass $\times 40/29 = \qty{333}{kg}$;
speeds $\times\sqrt{29/40}$: $u = \qty{428}{m/s}$; $U = \tfrac32 nRT$ instead of
$\tfrac52$: $60\%$.

\textbf{24.} $U = \tfrac52 \times 8320 \times 8.314 \times 293 = \qty{5.1e7}{J}$;
$\Delta U = \tfrac52 nR \times 10 = \qty{1.7e6}{J}$.

\textbf{25.} One average, $\langle v^2\rangle$, fixes the \omterm{def:b1:kinetic-theory:pressure}{pressure}
($\tfrac13 n^*m\langle v^2\rangle$), defines the \omterm{def:b1:kinetic-theory:temperature}{temperature}
($\tfrac12 m\langle v^2\rangle = \tfrac32 k_BT$), gives the equation of state
and the \omterm{def:b1:kinetic-theory:U}{internal energy}, and sets the scale of the speed of sound.
\end{solution}
