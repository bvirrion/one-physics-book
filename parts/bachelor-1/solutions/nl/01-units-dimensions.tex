\section*{Hoofdstuk \ref{ch:b1:units-dimensions} --- Eenheden, dimensies en meten}

\begin{solution}{exo:b1:units-dimensions:1}
$\unit{J} = \unit{kg.m^2.s^{-2}}$; $\unit{W} = \unit{kg.m^2.s^{-3}}$;
$\unit{Pa} = \unit{kg.m^{-1}.s^{-2}}$; $\unit{C} = \unit{A.s}$.
$\unit{Pa.m^3} = \unit{kg.m^{-1}.s^{-2}.m^3} = \unit{kg.m^2.s^{-2}} = \unit{J}$
(de arbeid van drukkrachten, $P\,\dd V$).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:2}
Druk $M L^{-1} T^{-2}$; vermogen $M L^2 T^{-3}$; lading $I\,T$;
spanning $= $ vermogen/stroom $= M L^2 T^{-3} I^{-1}$.
$[G] = [F r^2/m^2] = M L T^{-2} L^2 M^{-2} = L^3 M^{-1} T^{-2}$;
$[h] = [E/\nu] = M L^2 T^{-2} \cdot T = M L^2 T^{-1}$;
$[k_B] = [E/T] = M L^2 T^{-2} \Theta^{-1}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:3}
$\sqrt{2gh}$: $\sqrt{L T^{-2} \cdot L} = L T^{-1}$, homogeen.
$\tfrac12 mv$: $M L T^{-1} \neq M L^2 T^{-2}$, geen energie.
$\rho g h$: $M L^{-3} \cdot L T^{-2} \cdot L = M L^{-1} T^{-2}$, een druk.
$v_0 t + \tfrac12 g t^2$: beide termen $L$, homogeen.
$2\pi\sqrt{g/\ell}$: $\sqrt{T^{-2}} = T^{-1}$ --- een frequentie, geen periode.
$I_0\eu^{-t/RC}$: $RC$ in $\unit{\ohm.F} = \unit{s}$, argument \omterm{def:b1:units-dimensions:dimension}{dimensieloos}, homogeen.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:4}
Liniaal: $\delta = \qty{1}{mm}$, $u = 1/\sqrt{12} = \qty{0.29}{mm}$.
Balans: $u = 0.01/\sqrt{12} = \qty{0.003}{g}$.
Voltmeter: $a = 0.005 \times 4.80 = \qty{0.024}{V}$,
$u = 0.024/\sqrt3 = \qty{0.014}{V}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:5}
$v = k F^\alpha \mu^\beta$: $L T^{-1} = (M L T^{-2})^\alpha (M L^{-1})^\beta$,
dus $\alpha + \beta = 0$, $\alpha - \beta = 1$, $-2\alpha = -1$:
$\alpha = \tfrac12$, $\beta = -\tfrac12$, $v = k\sqrt{F/\mu}$ (in feite $k = 1$).
Geluid: $P/\rho$ heeft \omterm{def:b1:units-dimensions:dimension}{dimensie} $M L^{-1} T^{-2} / (M L^{-3}) = L^2 T^{-2}$,
dus $v = k\sqrt{P/\rho}$ ($k = \sqrt\gamma \approx 1.2$ voor lucht).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:6}
Som $= \qty{3620}{ms}$, $\bar t = \qty{452.5}{ms}$. Afwijkingen
$-0.5, -5.5, 8.5, 2.5, -3.5, 5.5, -8.5, 1.5$; de kwadraten sommeren tot $226$;
$s = \sqrt{226/7} = \qty{5.7}{ms}$; $u(\bar t) = 5.7/\sqrt8 = \qty{2.0}{ms}$.
Resultaat: $t = \qty{452.5\pm2.0}{ms}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:7}
$R = 4.80/0.0124 = \qty{387}{\ohm}$. Relatieve onzekerheden $0.03/4.80 =
0.63\%$ en $0.2/12.4 = 1.6\%$; gecombineerd $\sqrt{0.63^2 + 1.6^2} = 1.7\%$,
$u(R) = \qty{7}{\ohm}$: $R = \qty{387\pm7}{\ohm}$. De stroom
overheerst --- verbeter eerst de ampèremeter (of haar bereik).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:8}
$z = 7/\sqrt{16 + 9} = 1.4 \leq 2$: verenigbaar.
Derde tegen eerste: $z = 9/\sqrt{4 + 16} = 2.0$: op de grens, net
verenigbaar. Derde tegen tweede: $z = 16/\sqrt{4 + 9} = 4.4$: onverenigbaar
--- minstens een van beide heeft een systematische fout of een
onderschatte onzekerheid.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:9}
$V = \pi d^3/6 = \pi \times 25.40^3/6 = \qty{8580}{mm^3}$.
$u(V)/V = 3\,u(d)/d = 3 \times 0.02/25.40 = 0.24\%$,
$u(V) = \qty{20}{mm^3}$: $V = \qty{8.58\pm0.02}{cm^3}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:10}
$F = k\eta^a R^b v^c$: $M$: $a = 1$; $T$: $-a - c = -2$, $c = 1$;
$L$: $-a + b + c = 1$, $b = 1$: $F = k\eta R v$ (Stokes, $k = 6\pi$).
Met $\rho$ in plaats daarvan: $M$: $a = 1$; $T$: $-c = -2$, $c = 2$; $L$:
$-3 + b + 2 = 1$, $b = 2$: $F = k\rho R^2 v^2$.
Regendruppel: $\eta R v = 1.8\times10^{-5} \times 10^{-3} \times 5 \approx
\qty{1e-7}{N}$; $\rho R^2 v^2 = 1.2 \times 10^{-6} \times 25 =
\qty{3e-5}{N}$, driehonderd keer groter: het traagheidsregime (hoge snelheid)
geldt --- de verhouding $\rho v R/\eta \approx 330$ is het getal van
Reynolds.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:11}
$\bar I = \qty{5.0}{mA}$, $\bar U = \qty{1.000}{V}$; afwijkingen in $I$:
$-3, -1, 1, 3$; in $U$: $-0.59, -0.22, 0.22, 0.59$.
$\sum (I_i - \bar I)(U_i - \bar U) = 3.98$, $\sum (I_i - \bar I)^2 = 20$:
$a = \qty{0.199}{V/mA} = \qty{199}{\ohm}$, $b = 1.000 - 0.199 \times 5 =
\qty{0.005}{V}$. Met $u(U) = \qty{0.02}{V}$ per punt ligt een snijpunt van
\qty{0.005}{V} ruim binnen de onzekerheid: verenigbaar met nul, de
weerstand is ohms, $R \approx \qty{199}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:12}
$\ell = G^a h^b c^c$: $M$: $-a + b = 0$; $T$: $-2a - b - c = 0$; $L$:
$3a + 2b + c = 1$. Dus $b = a$, $c = -3a$, $2a = 1$:
$\ell_P = \sqrt{Gh/c^3} = \sqrt{4.42\times10^{-44}/2.7\times10^{25}}
= \qty{4.1e-35}{m}$; $t_P = \ell_P/c = \qty{1.4e-43}{s}$;
$m_P = \sqrt{hc/G} = \sqrt{1.99\times10^{-25}/6.67\times10^{-11}}
= \qty{5.5e-8}{kg}$. De plancklengte ligt twintig ordes van grootte
onder een proton; de planckmassa, \qty{55}{\micro g}, is een stofje
--- reusachtig voor een deeltje, nietig voor een mens ($10^9$ keer lichter).
\end{solution}

\begin{solution}{pb:b1:units-dimensions:1}
\textbf{1.} $\sqrt{L/(L T^{-2})} = \sqrt{T^2} = T$.

\textbf{2.} $\tau = k\ell^\alpha m^\beta g^\gamma$ geeft
$T = L^{\alpha+\gamma} M^\beta T^{-2\gamma}$, dus $\beta = 0$: geen
combinatie van $\ell$ en $g$ kan een massa wegdelen.

\textbf{3.} $\theta_0$ is \omterm{def:b1:units-dimensions:dimension}{dimensieloos}, zodat elke factor $f(\theta_0)$
onzichtbaar is voor \omterm{def:b1:units-dimensions:dimension}{dimensies}. Houd de uitwijking klein (onder ongeveer
$10^\circ$), waar $f \approx 1$ tot op beter dan $0.2\%$.

\textbf{4.} $g = 4\pi^2\ell/\tau^2$.

\textbf{5.} $\ell = 99.0 + 1.00 = \qty{100.0}{cm} = \qty{1.000}{m}$.

\textbf{6.} $u = \qty{1}{mm}/\sqrt{12} = \qty{0.29}{mm}$.

\textbf{7.} Uitlijning: $5/\sqrt3 = \qty{2.9}{mm}$; gecombineerd
$\sqrt{0.29^2 + 2.9^2} = \qty{2.9}{mm}$ --- de aflezing op het lint is
verwaarloosbaar. $u(\ell)/\ell = 0.29\%$.

\textbf{8.} De fout door de reactietijd (ongeveer \qty{0.1}{s}) is dezelfde
of men één dan wel tien perioden klokt, zodat haar effect op één periode
tien keer kleiner is. Herhalen legt de spreiding bloot (vandaar $s$) en
deelt de onzekerheid van het gemiddelde door $\sqrt8$.

\textbf{9.} Som $= \qty{160.80}{s}$, $\overline{t_{10}} = \qty{20.100}{s}$.
Afwijkingen $0.02, -0.05, 0.08, -0.01, 0.05, -0.08, 0.01, -0.02$; de kwadraten
sommeren tot $\num{188e-4}$; $s = \sqrt{188\times10^{-4}/7} = \qty{0.052}{s}$.

\textbf{10.} $u(\overline{t_{10}}) = 0.052/\sqrt8 = \qty{0.018}{s}$;
$\tau = \qty{2.0100}{s}$, $u(\tau) = \qty{0.0018}{s}$.

\textbf{11.} $u(\tau)/\tau = 0.09\%$, drie keer kleiner dan
$u(\ell)/\ell = 0.29\%$.

\textbf{12.} $g = 4\pi^2 \times 1.000/2.0100^2 = 39.48/4.040 =
\qty{9.772}{m/s^2}$.

\textbf{13.} $g \propto \ell\,\tau^{-2}$:
$\dfrac{u(g)}{g} = \sqrt{\left(\dfrac{u(\ell)}{\ell}\right)^2
+ \left(2\dfrac{u(\tau)}{\tau}\right)^2} = \sqrt{0.29^2 + 0.18^2}\,\%
= 0.34\%$.

\textbf{14.} $u(g) = 0.0034 \times 9.772 = \qty{0.033}{m/s^2}$:
$g = \qty{9.77\pm0.03}{m/s^2}$.

\textbf{15.} $z = \abs{9.772 - 9.81}/0.033 = 1.2 \leq 2$: verenigbaar.

\textbf{16.} De lengte ($0.29\%$ tegen $0.18\%$ voor de periodeterm).
Beide uiteinden tot op $\pm\qty{1}{mm}$ lokaliseren brengt $u(\ell)/\ell$ op
$0.06\%$ en $u(g)/g$ op $\sqrt{0.06^2 + 0.18^2} = 0.19\%$; de lengte
verdubbelen halveert $u(\ell)/\ell$ en verlengt $\tau$, wat beide termen
helpt. Twintig perioden klokken in plaats van tien halveert alleen de
periodeterm: $u(g)/g = \sqrt{0.29^2 + 0.09^2} = 0.30\%$ --- een kleine winst
zolang de lengte de flessenhals is.

\textbf{17.} Nee: een verschuiving die alle aflezingen gemeen hebben is een
\emph{systematische fout} (een bias); de begroting dekt toevallige spreiding
en opgegeven toleranties, geen constante verschuiving. Zij wordt opgespoord
door van methode of instrument te veranderen, of --- Deel V --- door een van
nul verschillend snijpunt van de lijn $\tau^2$--$\ell$.

\textbf{18.} $\tau^2 = (4\pi^2/g)\,\ell$ is een rechte door de
oorsprong: lineariteit en het nulsnijpunt zijn allebei te toetsen met het oog
en met \omterm{prop:b1:units-dimensions:regression}{kleinste kwadraten}; $\tau \propto \sqrt\ell$ niet.

\textbf{19.} $\bar\ell = \qty{0.800}{m}$; $\overline{\tau^2} = 16.16/5 =
\qty{3.232}{s^2}$.

\textbf{20.} Afwijkingen in $\ell$: $-0.4, -0.2, 0, 0.2, 0.4$; in
$\tau^2$: $-1.612, -0.822, 0.018, 0.808, 1.608$. Kruissom $= 1.614$,
$\sum(\ell_i - \bar\ell)^2 = 0.400$: $a = \qty{4.035}{s^2/m}$,
$b = 3.232 - 4.035 \times 0.800 = \qty{0.004}{s^2}$.

\textbf{21.} $g = 4\pi^2/a = 39.48/4.035 = \qty{9.78}{m/s^2}$.

\textbf{22.} $b \approx 0$, zoals het model voorspelt. Een duidelijk van nul
verschillende $b$ zou een systematische verschuiving $\delta$ in elke lengte
verraden ($\tau^2 = a(\ell_{\text{gem}} - \delta)$ geeft $b = -a\delta$), bv.\ een
verkeerd geplaatst middelpunt van de bol --- of een uitwijkingseffect.

\textbf{23.} $u(g)/g = u(a)/a = 0.05/4.035 = 1.2\%$, $u(g) =
\qty{0.12}{m/s^2}$: $g = \qty{9.78\pm0.12}{m/s^2}$.

\textbf{24.} $z = \abs{9.772 - 9.784}/\sqrt{0.033^2 + 0.12^2} = 0.1$:
volledig verenigbaar. De ene zorgvuldige meting is nauwkeuriger; de lijn
bevestigt het model en beschermt tegen een lengteverschuiving.

\textbf{25.} $\tau = 2\pi\sqrt{10/9.8} = \qty{6.3}{s}$; één zwaai geklokt
met een reactietijd van ongeveer \qty{0.2}{s} geeft $u(\tau)/\tau \approx
3\%$, dus $u(g)/g \approx 6\%$, terwijl $u(\ell)/\ell$ tot
$0.03\%$ daalt: twintig keer slechter dan Deel IV. De winst op de lengte gaat
verloren door één periode te klokken. Beste aanpak: lange slinger \emph{en} veel
perioden, herhaald --- tien zwaaien van de slinger van \qty{10}{m}, acht
keer, geeft $u(\tau)/\tau \approx 0.03\%$ en $u(g)/g \approx 0.07\%$.
\end{solution}
