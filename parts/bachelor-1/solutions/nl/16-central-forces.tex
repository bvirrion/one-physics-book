\section*{Hoofdstuk \ref{ch:b1:central-forces} --- Centrale krachten: planeten en satellieten}

\begin{solution}{exo:b1:central-forces:1}

= \qty{5.5e3}{s} = \qty{92}{min}$; $1440/92 \approx 16$ zonsopgangen per dag.
\end{solution}

\begin{solution}{exo:b1:central-forces:2}

\begin{solution}{exo:b1:central-forces:3}
$v_e = \sqrt{2GM/R}$: aarde \qty{11.2}{km/s}; maan $\sqrt{2 \times 4.90\times

\qty{2.4}{km/s}; over geologische tijden lekte het gas van de maan weg.
\end{solution}

\begin{solution}{exo:b1:central-forces:4}
$T = a^{3/2}$ (jaren, AU): Mars \qty{1.88}{yr}; Jupiter \qty{11.9}{yr};

\end{solution}

\begin{solution}{exo:b1:central-forces:5}
$\Delta E = \tfrac12 mv^2 + GMm(1/R - 1/r) = 2.94\times10^{10} + 3.99\times
10^{17} \times (1.570 - 1.477)\times10^{-7} = 2.94\times10^{10} + 0.37\times
10^{10} = \qty{3.3e10}{J}$ --- \qty{33}{MJ/kg}, de energie van \qty{0.7}{kg}
benzine per kilogram die in een \omterm{def:b1:point-kinematics:frame}{baan} wordt gebracht (raketten hebben veel meer
nodig, omdat zij hun eigen brandstof moeten meetillen).
\end{solution}

\begin{solution}{exo:b1:central-forces:6}
$a = (0.586 + 35.1)/2 = \qty{17.8}{AU}$; $e = (35.1 - 0.586)/35.7 = 0.967$;
$T = 17.8^{3/2} = \qty{75}{yr}$. $v_p^2 = GM(2/r_p - 1/a)$ met $r_p =
8.77\times10^{10}$ m, $a = 2.67\times10^{12}$ m: $v_p = \qty{54}{km/s}$;
$v_a = v_pr_p/r_a = \qty{0.91}{km/s}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:7}
$a = \qty{24490}{km}$. $v_1 = \qty{7.67}{km/s}$; op de ellips in $r_1$:
$\sqrt{GM(2/r_1 - 1/a)} = \qty{10.07}{km/s}$: $\Delta v_1 = \qty{2.4}{km/s}$.
In $r_2$: $\sqrt{GM(2/r_2 - 1/a)} = \qty{1.62}{km/s}$, cirkelvormig
\qty{3.07}{km/s}: $\Delta v_2 = \qty{1.45}{km/s}$. Totaal \qty{3.85}{km/s};
duur $\pi\sqrt{a^3/GM} = \qty{1.9e4}{s} = \qty{5.3}{h}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:8}
$E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ in $r_c = L^2/GMm^2$.
$E_{p,\mathrm{eff}}'' = -2GMm/r^3 + 3L^2/mr^4$; in $r_c$ geeft $L^2 = GMm^2r_c$
dat $E'' = GMm/r_c^3$, dus $\omega_r^2 = E''/m = GM/r_c^3 = \omega_{\text{baan}}^2$.
Een kleine radiale verstoring trilt precies eens per omloop: de
\omterm{def:b1:point-kinematics:frame}{baan} sluit zichzelf --- een licht excentrische ellips, geen rozet.
\end{solution}

\begin{solution}{exo:b1:central-forces:9}
Op station en bemanning werkt alleen de zwaartekracht; beide vallen op dezelfde
\omterm{def:b1:point-kinematics:frame}{baan}, zodat er tussen hen geen contactkracht nodig is --- geen vloer die duwt,
vandaar de ``gewichtloosheid''. $g = GM/r^2 = 3.99\times10^{14}/(6.77\times10^6)^2
= \qty{8.7}{m/s^2}$: de zwaartekracht is bijna op volle sterkte; het gewicht is niet
afwezig, het gaat volledig op aan het krommen van de \omterm{def:b1:point-kinematics:frame}{baan}.
\end{solution}

\begin{solution}{exo:b1:central-forces:10}
$a = 4\pi^2r/T^2 = 39.5 \times 3.84\times10^8/(2.36\times10^6)^2 =
\qty{2.7e-3}{m/s^2}$; $g(R/r)^2 = 9.81/60.3^2 = \qty{2.7e-3}{m/s^2}$: dezelfde
krachtwet die een appel laat vallen houdt de maan vast, verdund als $1/r^2$.
\end{solution}

\begin{solution}{exo:b1:central-forces:11}
$E_k = 2Ze^2/4\pi\varepsilon_0r_{\min}$: $r_{\min} = 8.99\times10^9 \times 2 \times 79
\times (1.60\times10^{-19})^2/8.0\times10^{-13} = \qty{4.5e-14}{m} = \qty{45}{fm}$,
zesmaal de kernstraal: het $\alpha$-deeltje keert om vóór het de kern raakt.
Om \qty{7}{fm} te bereiken: $E_k \approx \qty{32}{MeV}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:12}
$E_m = -GMm/2r$ neemt af $\Rightarrow$ $r$ neemt af $\Rightarrow$ $v =
\sqrt{GM/r}$ neemt toe. De \omterm{thm:b1:work-and-energy:ke}{kinetische energie} $GMm/2r$ stijgt met $\abs{\Delta E}$
terwijl de \omterm{def:b1:work-and-energy:conservative}{potentiële energie} $-GMm/r$ daalt met $2\abs{\Delta E}$: de helft van de
vrijkomende \omterm{def:b1:work-and-energy:conservative}{potentiële energie} wordt kinetisch, de helft wordt door de weerstand
als warmte gedissipeerd.
\end{solution}

\begin{solution}{pb:b1:central-forces:1}
\textbf{1.} $T = \qty{43082}{s}$: $r = (GMT^2/4\pi^2)^{1/3} = (3.986\times10^{14}
\times 1.856\times10^9/39.48)^{1/3} = \qty{2.656e7}{m}$; hoogte
\qty{20190}{km}.

\textbf{2.} $v = \sqrt{GM/r} = \qty{3.87}{km/s}$.

\textbf{3.} $a = v^2/r = GM/r^2 = \qty{0.565}{m/s^2} = g(R/r)^2$, zeventien
keer zwakker dan aan het oppervlak --- maar niet nul: de satelliet valt
onophoudelijk om de aarde heen.

\textbf{4.} $E_m/m = -GM/2r = \qty{-7.5}{MJ/kg}$; $E_k/m = \qty{+7.5}{MJ/kg}$,
$E_p/m = \qty{-15.0}{MJ/kg}$.

\textbf{5.} De aarde draait één keer per sterrendag ten opzichte van de sterren
(het stelsel van de \omterm{def:b1:point-kinematics:frame}{baan}); na twee omlopen staat de satelliet weer boven
hetzelfde punt op de grond: de grondsporen herhalen zich elke sterrendag.

\textbf{6.} $\sin\beta = R/r = 0.240$: $\beta = 13.9^\circ$, een schijf van
$28^\circ$ breed.

\textbf{7.} Geostationair \qty{42200}{km}: $1.59$ keer hoger; de
navigatiebaan ligt eronder.

\textbf{8.} $GM(1/R - 1/2r) = 3.986\times10^{14}(1.570\times10^{-7} - 1.88\times
10^{-8}) = \qty{5.5e7}{J/kg} = \qty{55}{MJ/kg}$.

\textbf{9.} $r_1 = \qty{6771}{km}$: $v = \qty{7.67}{km/s}$.

\textbf{10.} $a = \qty{16665}{km}$; in $r_1$: $\sqrt{GM(2/r_1 - 1/a)} =
\qty{9.69}{km/s}$, $\Delta v_1 = \qty{2.0}{km/s}$; in $r_2$: \qty{2.47}{km/s}

\textbf{11.} $\pi\sqrt{a^3/GM} = \qty{1.07e4}{s} \approx \qty{3.0}{h}$.

\textbf{12.} Ontsnappen uit de lage \omterm{def:b1:point-kinematics:frame}{baan}: $(\sqrt2 - 1)\times 7.67 = \qty{3.2}{km/s}$:

\textbf{13.} $m_0/m_1 = \eu^{3.4/3.0} = 3.1$: $m_1 = \qty{640}{kg}$,

\textbf{24.} De afplatting van de aarde aan de evenaar, de aantrekking van maan
en zon, de stralingsdruk van de zon (en kleine correctiestoten): het
grondsegment meet de banen voortdurend en zendt verse
efemeriden omhoog, die de satellieten uitzenden.

\textbf{25.} Energie en \omterm{def:b1:angular-momentum:L}{impulsmoment} leggen een cirkelbaan vast zodra haar
omlooptijd bekend is; Kepler III plaatst haar; de meetkunde bepaalt dekking en
looptijden; en de klokken --- het hart van het systeem --- moeten voor de
relativiteit worden gecorrigeerd met \qty{38}{\micro s} per dag, anders loopt de
plaatsbepaling tegen de avond tien kilometer weg.
\end{solution}
